\label{thm:edelman-kostlan}
Let $X=(X(z))_{z \in D}$ be a GAF with convariance $K(z,w) =
 \mathbb{E}[X(z)\overline{X(w)}]$. Then, the first intensity of
 zeros of $X(z)$ is given by the formula 
\[
 \rho_1(z) = \frac{1}{\pi} \partial_z \partial_{\bar{w}} \log
 K(z,w)\Big|_{w=z}. 
\]
<!-- end thm-1 -->
<!-- prop-1: body_md -->
\label{prop:tightness}
Let $D\subset \mathbb{C}$ be a domain and $\{X_n\}_{n=1}^{\infty}$ be a sequence of random analytic
 functions on $D$. If $\sup_n \mathbb{E}[|X_n(z)|^p]$ is locally
 integrable on $D$ for some $p>0$, then $\{\mu_{X_n}\}_{n=1}^{\infty}$ is tight in
 $\mathcal{P}(\mathcal{H}(D))$. Moreover, if $\{X_n\}_{n=1}^{\infty}$
 converges to $X$ in the sense of finite-dimensional
 distributions, then $\{\mu_{X_n}\}$ converges weakly to $\mu_X$ as
 $n \to \infty$, or equivalently, $\{X_n\}_{n=1}^{\infty}$
 converges to $X$ in law.
<!-- end prop-1 -->
<!-- prop-2: body_md -->
\label{prop:zeros-converge}
A sequence of random analytic functions
 $\{X_n\}_{n=1}^{\infty}$ 
converges in law to $X$. Then, the zero process $\mathcal{Z}_{X_n}$
 converges in law to $\mathcal{Z}_X$ provided that $X \not\equiv
 0$ almost surely.
<!-- end prop-2 -->
<!-- cor-1: body_md -->
\label{cor:weak-conv}
Let $F_n(dt), n=1,2,\dots$ and $F(dt)$ be probability
measures on $\mathbb{T}$ and suppose that $\{F_n(dt)\}_{n=1}^{\infty}$
converges weakly to $F(dt)$.  
Then, $X_{F_n}$ corresponding to $F_n$ converges in law to
 $X_F$ corresponding to $F$. Moreover, the zero process
 $\mathcal{Z}_{X_{F_n}}$ converges in law to that of $\mathcal{Z}_{X_F}$.
<!-- end cor-1 -->
<!-- proof-1: body_md -->
The local integrability of $\sup_n \mathbb{E}[|X_{F_n}(z)|^2]$
 follows from the fact that $\mathbb{E}[|X_{F_n}(z)|^2] = K_{F_n}(z,z) \le (1-|z|)^{-2}$
 for every $n$. Since $s(z,t)$ is bounded continuous in $t$
 for every $z \in \mathbb{D}$, by the weak convergence of
 $\{F_n\}_{n=1}^{\infty}$ to $F$, $K_{F_n}(z,w)$ converges to
 $K_{F}(z,w)$ for every $z,w \in \mathbb{D}$. This implies that
 $X_{F_n}$ converges to $X_{F}$ in the sense of 
 finite-dimensional distributions. 
Therefore, $X_F$ converges in law to $X$ by Proposition~\ref{prop:tightness}. 
The zero processes $\mathcal{Z}_{F_n}, n=1,2,\dots$ converges in law
 to $\mathcal{Z}_F$ Proposition~\ref{prop:zeros-converge} since $X_F
 \not\equiv 0$ almost surely.

<!-- prop-3: body_md -->
\label{prop:prop6.1}
The $n$-point correlation function of the zeros of GAF with covariance kernel $K(z,w)$ 
is given by the formula 
\[
 \rho_n(z_1,z_2,\dots,z_n) = \frac{\per(\partial_z \partial_{\bar{w}} K^{z_1,z_2,\dots,z_n}(z_j,z_k))_{j,k=1}^n}{\det (\pi K(z_j,z_k))_{j,k=1}^n} 
\]
for distinct $z_1,z_2,\dots,z_n \in D$ with 
$\det (K(z_j,z_k))_{j,k=1}^n > 0$, where $\per A$ 
is the permanent of an $n$ by $n$ matrix 
$A = (a_{jk})_{j,k=1}^n$ defined by 
\[
 \per A = \sum_{\sigma \in \mathcal{S}_n} \prod_{j=1}^n a_{j \sigma(j)}, 
\]
where $\mathcal{S}_n$ is the symmetric group of order $n$.

<!-- lem-1: body_md -->
\label{lem:detK} 
For $\mathbf{a}=(a_1,a_2,\dots,a_n)$ and $\mathbf{t}=(t_1,t_2,\dots,t_n)$,
 we have 
\begin{equation}
\det(K(a_j,a_k))_{j,k=1}^n 
= |V(\mathbf{a})|^2 \mu_{\mathbf{a}}(\mathbb{T}^n), 
\label{eq:detS}
\end{equation}
where 
\begin{equation}
\mu_{\mathbf{a}}(d\mathbf{t}) = \frac{1}{n!} 
|V(e^{-it_1}, e^{-it_2},\dots, e^{-it_n})|^2
\cdot |S(\mathbf{a},\mathbf{t})|^2 F^{\otimes n}(d\mathbf{t}). 
\label{eq:muadt}
\end{equation}
<!-- end lem-1 -->
<!-- proof-2: body_md -->
By \eqref{eq:Szw}
and the symmetrization of the variable $\mathbf{t}$, we see that 
\begin{align*}
\det(K(a_j,a_k))_{j,k=1}^n
&= \sum_{\sigma \in \mathcal{S}_n} \sgn(\sigma) 
\int_{\mathbb{T}^{n}} \prod_{k=1}^n s(a_{\sigma(k)},t_k) \cdot 
\overline{\prod_{k=1}^n s(a_k,t_k)}
F^{\otimes n}(d\mathbf{t}) \\
&= \int_{\mathbb{T}^{n}} \det(s(a_j,t_k))_{j,k=1}^n
\overline{\Big(\prod_{k=1}^n s(a_k,t_k)\Big)}
F^{\otimes n}(d\mathbf{t}) \\
&=\frac{1}{n!} 
\int_{\mathbb{T}^{n}} 
\det(s(a_j,t_k))_{j,k=1}^n
\overline{\det(s(a_j,t_k))_{j,k=1}^n}  
F^{\otimes n}(d\mathbf{t}) \\
&=\frac{1}{n!} 
\int_{\mathbb{T}^{n}} |D(\mathbf{a},\mathbf{t})|^2 
F^{\otimes n}(d\mathbf{t}). 
\end{align*}
The assertion follows directly from the identity \eqref{eq:cauchy_identity2}.

<!-- rem-1: body_md -->
\label{rem:DPPonT}
Let $\mathcal{U}(n)$ denote the compact Lie group of $n \times n$
 unitary matrices and let $P_n$ be the Haar measure on it. 
The pair $(\mathcal{U}(n), P_n)$ is called the circular unitary ensemble CUE$_n$. 
The eigenvalues of CUE$_n$ are known to form the $n$-point DPP 
on $\mathbb{T}$ associated with the rank $n$ projection kernel 
\[
 \tilde{k}_n(\theta, \varphi) 
=\sum_{k=0}^{n-1}  e^{i k\theta} \overline{e^{ik\varphi}}
= u(\theta) \frac{\sin
 \frac{n(\theta-\varphi)}{2}}{\sin\frac{\theta-\varphi}{2}}
 \overline{u(\varphi)} 
=: u(\theta) k_n(\theta, \varphi)
 \overline{u(\varphi)}, 
\]
where $u(\theta) = e^{i (n-1)\theta/2}$ and with the background
 measure being the Lebesgue probability measure on $\mathbb{T}$. 
The probability measure on $\mathbb{T}^n$ with density $\mu_{\mathbf{a}}(\mathbf{t})/\mu_{\mathbf{a}}(\mathbb{T}^n)$ can be
 regarded as the $n$-point DPP associated with the
 correlation kernel $k_n(\theta,\varphi)$ and the background measure 
\[
|S(\mathbf{a},t)|^2 F(dt) = \Big(\prod_{j=1}^n \frac{1}{1-r_j^2} \Big) 
\Big(\prod_{j=1}^n P(r_j, \varphi_j - t) \Big) F(dt), 
\]
where $a_j = r_j e^{i\varphi_j}$ and $P(r,\theta)$ is the
 Poisson kernel in \eqref{eq:poisson-kernel}. 
We denote this $n$-point DPP on $\mathbb{T}$ by $\mathbb{P}_{\mathbf{a}}^{(n)}$.

<!-- lem-2: body_md -->
\label{lem:palmn} 
For $\mathbf{a}=(a_1,a_2,\dots,a_n)$ and $\tilde{\mathbf{t}}
 =(t_0,t_1,\dots,t_n)$, 
\begin{align*}
\lefteqn{\det(K(a_j,a_k))_{j,k=1}^n \cdot 
K^{a_1,\dots,a_n}(z,w)} \\
&= |V(\mathbf{a})|^2
\prod_{i=1}^n (z-a_i)\overline{(w-a_i)} \int_{\mathbb{T}^{n+1}} 
S(z,\tilde{\mathbf{t}}) \overline{S(w,\tilde{\mathbf{t}})}
 \tilde{\mu}_{\mathbf{a}}(d\tilde{\mathbf{t}}) \\
&=|V(\mathbf{a})|^2 \tilde{\mu}_{\mathbf{a}}(\mathbb{T}^{n+1})
\prod_{i=1}^n (z-a_i)\overline{(w-a_i)} \cdot K_{\mathbf{a}}^{(n+1)}(z,w) 
\end{align*}
where 
\[
\tilde{\mu}_{\mathbf{a}}(d\tilde{\mathbf{t}}) 
= \frac{1}{(n+1)!}|V(e^{-it_0}, e^{-it_1}, \dots, e^{-it_n})|^2 
|S(\mathbf{a},\tilde{\mathbf{t}})|^2
 F^{\otimes(n+1)}(d\tilde{\mathbf{t}}). 
\]
and 
\[
K_{\mathbf{a}}^{(n+1)}(z,w) = \mathbb{E}_{\mathbf{a}}^{(n+1)}[S(z,\tilde{\mathbf{t}})
 \overline{S(w,\tilde{\mathbf{t}})}],  
\]
where $\mathbb{E}_{\mathbf{a}}^{(n+1)}$ is the expectation with respect to
 $\mathbb{P}_{\mathbf{a}}^{(n+1)}$ defined as in Remark~\ref{rem:DPPonT}.
<!-- end lem-2 -->
<!-- proof-3: body_md -->
As in the previous calculation in the proof of
 Lemma~\ref{lem:detK}, by \eqref{eq:palm_kernel-2}, it follows that 
\begin{align*}
Q_n(z,w)
&:=\det(K(a_j,a_k))_{j,k=1}^n K^{a_1,\dots,a_n}(z,w) \\
&=\det(K(a_j,a_k))_{j,k=0}^n\Big|_{\text{$a_0=z$ or $w$}} \\
&=\frac{1}{(n+1)!} 
\int_{\mathbb{T}^{n+1}} 
\det(s(a_i,t_j))_{i,j=0}^n\Big|_{a_0=z} \cdot 
\overline{\det(s(a_i,t_j))_{i,j=0}^n}\Big|_{a_0=w} 
F^{\otimes n+1}(dt_0 d\mathbf{t}). 
\end{align*}
Setting $\tilde{\mathbf{a}} = (a_0,\mathbf{a})$, by
 \eqref{eq:cauchy_identity}, we have 
\begin{align*}
\det(s(a_j,t_k))_{j,k=0}^n \Big|_{a_0=z}
&= V(\tilde{\mathbf{a}}) V(e^{-it_0}, e^{-it_1}, \dots, e^{-it_n}) S(\tilde{\mathbf{a}}, \tilde{\mathbf{t}})\Big|_{a_0=z} \\
&= \prod_{j=1}^n (a_j-z) \cdot V(\mathbf{a}) 
V(e^{-it_0}, e^{-it_1}, \dots, e^{-it_n}) 
S(z,\tilde{\mathbf{t}}) S(\mathbf{a},\tilde{\mathbf{t}}).  
\end{align*}
Hence, we obtain 
\begin{align*}
Q_n(z,w)
&=\frac{1}{(n+1)!} |V(\mathbf{a})|^2
\prod_{j=1}^n (z-a_j)\overline{(w-a_j)} \\
&\quad \times \int_{\mathbb{T}^{n+1}} 
S(z,\tilde{\mathbf{t}}) \overline{S(w,\tilde{\mathbf{t}})}
\cdot |V(e^{-it_0}, e^{-it_1}, \dots, e^{-it_n})|^2 
|S(\mathbf{a},\tilde{\mathbf{t}})|^2 F^{\otimes(n+1)}(d\tilde{\mathbf{t}}).  
\end{align*}
This completes the proof.

<!-- thm-2: body_md -->
\label{thm:n-correlation} 
For pairwise distinct $\mathbf{a}=(a_1,a_2,\dots,a_n) \in \mathbb{D}^n$, 
\begin{align*}
 \rho_n(\mathbf{a}) 
&= 
\frac{1}{\pi^n} |V(\mathbf{a})|^2 \frac{\tilde{\mu}_{\mathbf{a}}(\mathbb{T}^{n+1})^n}{\mu_{\mathbf{a}}(\mathbb{T}^n)^{n+1}} 
\per (K_{\mathbf{a}}^{(n+1)}(a_p,a_q))_{p,q=1}^n.  
\end{align*}
<!-- end thm-2 -->
<!-- proof-4: body_md -->
By Lemma~\ref{lem:palmn}, we note that 
\begin{align*}
\lefteqn{\partial_z\partial_{\bar{w}} Q_n(z,w))|_{z=a_p, w=a_q} }\\
&= |V(\mathbf{a})|^2 \tilde{\mu}_{\mathbf{a}}(\mathbb{T}^{n+1}) \left(
\prod_{i=1 \atop{i \not = p}}^n (a_p-a_i)
\overline{\prod_{j=1 \atop{j\not=q}}^n (a_q-a_j)} \right) 
K_{\mathbf{a}}^{(n+1)}(a_p,a_q). 
\end{align*}
Then, we see that 
\begin{align*}
\lefteqn{\per((\partial_z\partial_{\bar{w}} Q_n(z,w))|_{z=a_p,
 w=a_q})}\\
&= \left\{|V(\mathbf{a})|^2 \tilde{\mu}_{\mathbf{a}}(\mathbb{T}^{n+1})\right\}^{n} 
\sum_{\sigma \in \mathcal{S}_n} \prod_{p=1}^n 
\left(\prod_{i=1 \atop{i \not = p}}^n (a_p-a_i)
\overline{\prod_{j=1 \atop{j\not=\sigma(p)}}^n
 (a_{\sigma(p)}-a_j)} \right) K_{\mathbf{a}}^{(n+1)}(a_p,a_{\sigma(p)}) \\
&= |V(\mathbf{a})|^{2n+4} \tilde{\mu}_{\mathbf{a}}(\mathbb{T}^{n+1})^n 
\per\big( K_{\mathbf{a}}^{(n+1)}(a_p,a_q)_{p,q=1}^n \big). 
\end{align*}
Here, we used \eqref{eq:prodap-aj} in the second equality.  
Therefore, by Lemma~\ref{lem:detK}, we obtain 
\begin{align*}
 \rho_n(\mathbf{a}) 
&= \frac{\per((\partial_z\partial_{\bar{w}} K^{a_1,\dots,a_n}(z,w))|_{z=a_p,
 w=a_q})}{\det(\pi K)} 
= \frac{\per((\partial_z\partial_{\bar{w}} Q_n(z,w))|_{z=a_p,
 w=a_q})}{\pi^n (\det K)^{n+1}} \\
&= \frac{|V(\mathbf{a})|^2 \tilde{\mu}_{\mathbf{a}}(\mathbb{T}^{n+1})^n}{\pi^n \mu_{\mathbf{a}}(\mathbb{T}^n)^{n+1}} \per(K_{\mathbf{a}}^{(n+1)}(a_p,a_q)_{p,q=1}^n). 
\end{align*}
This is the desired formula.

<!-- rem-2: body_md -->
Theorem~\ref{thm:n-correlation} shows that the
 $n$-correlation function includes the factor of squared Vandermonde determinant
 $|V(\mathbf{a})|^2$. 
For $\mathbf{a}=(a_1,\dots,a_n)$, we consider a monic polynomial
 $p(z) = \prod_{k=1}^n (z-a_k)$ and suppose it has the coeffcients $e_k, 
 k=0,1,\dots, n-1$, i.e., $p(z) = \sum_{k=0}^{n} e_k z^k$
 with $e_n=1$. Then the transformation $T : \mathbb{C}^n \to
 \mathbb{C}^n$ defined by $T(a_1,a_2,\dots, a_n) = (e_{n-1},e_{n-2},
 \dots, e_0)$ has Jacobian determinant $|V(\mathbf{a})|^2$
 (cf. \cite{HKPV09}), which is reflected in the theorem. See also
 \cite{NS12, GP17}.
<!-- end rem-2 -->
<!-- lem-3: body_md -->
Let 
\[
C(\mathbf{z},\mathbf{a}) := \prod_{k=1}^n \prod_{j=1}^n \frac{1}{z_k-a_j}. 
\]
Then, for an analytic function $Q(\mathbf{z})$ that is symmetric in its variables,  
\begin{equation}
\left(\frac{1}{2\pi i}\right)^{n} \oint_{C^n} V(\mathbf{z})^2 C(\mathbf{z},\mathbf{a}) Q(\mathbf{z})
\prod_{k=1}^n dz_k = (-1)^{{n \choose 2}}  n! \cdot Q(\mathbf{a}),  
\label{eq:reproducing-VC} 
\end{equation}
and for pairwise distinct $\mathbf{a}=(a_1,\dots,a_n)$, 
\begin{align}
\lefteqn{
\left(\frac{1}{2\pi i}\right)^{n} \oint_{C^n} V(\mathbf{z})^2 C(\mathbf{z},\mathbf{a}) Q(\mathbf{z})
\prod_{k=1}^n \frac{dz_k}{z_k} } \nonumber \\ 
&= (-1)^{{n \choose 2}}  n! \cdot \prod_{j=1}^n 
 \frac{1}{a_j} \cdot 
\Big\{
Q(\mathbf{a}) 
- \sum_{p=1}^n 
\prod_{k=1 \atop{k\not= p}}^n\frac{a_k}{a_k-a_p}
Q(0,\mathbf{a}^{(p)}) \Big\}, 
\label{eq:reproducing-VC-2} 
\end{align}
where $C = \{z \in \mathbb{C} : |z|=1\}$ and $\mathbf{a}^{(p)} =
 (a_1,\dots,a_{p-1}, a_{p+1},\dots, a_n)$.
<!-- end lem-3 -->
<!-- proof-5: body_md -->
Due to the presence of the Vandermonde term, the residue is
 zero unless the points $(z_1, z_2, \dots, z_n)$ are all distinct and
 form a permutation of $\{a_1, \dots, a_n\}$. 
Since the integral is symmetric in $(z_1, z_2, \dots, z_n)$, it suffices
 to consider the residue at $(z_1,z_2,\dots,z_n) =
 (a_1,a_2,\dots,a_n)$ and multiply it by $n!$. The residue
 is given by 
$\displaystyle V(\mathbf{a})^2 \prod_{k=1\atop{k\not=j}}^n (a_k-a_j)^{-1}
 Q(\mathbf{a})$ 
and thus, using \eqref{eq:prodap-aj}, we obtain \eqref{eq:reproducing-VC}. 

Similarly, 
the residue is zero unless the points $(z_1, z_2, \dots, z_n)$ are all distinct and
 form a permutation of $\{0, a_1, \dots, a_n\} \setminus
 \{a_p\}$ for some $p=1,2,\dots,n$. Since the
 integral is symmetric in $(z_1, z_2, \dots, z_n)$, it suffices
 to consider the residue at $\mathbf{z}=\mathbf{a}$ or $\mathbf{z} = (0,\mathbf{a}^{(p)})$
 for $p=1,2,\dots,n$ and multiply it by $n!$. 
For $\mathbf{z}=\mathbf{a}$, the first term in \eqref{eq:reproducing-VC-2}
 is obtained in the same manner as above with the extra term
 $\prod_{k=1}^n 1/a_k$. 
Since $\displaystyle V(0,\mathbf{a}^{(p)}) = (\prod_{j=1 \atop{j\not= p}}^n a_j)
 V(\mathbf{a}^{(p)})$, the residue at $\mathbf{z} = (0,\mathbf{a}^{(p)})$ is given
 by 
\begin{align*}
\mathrm{Res}_{\mathbf{z}=(0,\mathbf{a}^{(p)})}&=(\prod_{j=1 \atop{j\not= p}}^n a_j)^2 V(\mathbf{a}^{(p)})^2 
\cdot \prod_{j=1}^n \frac{1}{-a_j}
\cdot \prod_{k=1 \atop{k\not=p}}^n 
\prod_{j=1 \atop{j\not=k}}^n \frac{1}{a_k-a_j} 
\cdot \prod_{k=1 \atop{k\not= p}}^n \frac{1}{a_k} \cdot Q(0,\mathbf{a}^{(p)}) \\
&= (-1)^{{n \choose 2}+1} 
\prod_{j=1}^n \frac{1}{a_j}
\prod_{k=1 \atop{k\not=p}}^n \frac{a_k}{a_k-a_p}
\cdot Q(0,\mathbf{a}^{(p)}). 
\end{align*}
Therefore, we obtain \eqref{eq:reproducing-VC-2}.

<!-- lem-4: body_md -->
\label{lem:volume-formula} 
Suppose $F(dt) = dt/(2\pi)$. Then, 
\[
\mu_{\mathbf{a}}(\mathbb{T}^n) =  \tilde{\mu}_{\mathbf{a}}(\mathbb{T}^{n+1}) 
= S(\mathbf{a},\mathbf{a}), \quad 
K_{\mathbf{a}}^{(n+1)}(z,w) = S(z,w) S(z,\mathbf{a}) \overline{S(w,\mathbf{a})}. 
\]
<!-- end lem-4 -->
<!-- proof-6: body_md -->
By \eqref{eq:muadt}, the definition of $\mu_{\mathbf{a}}$, we see
 that 
 \begin{align*}
n! \mu_{\mathbf{a}}(\mathbb{T}^{n}) 
&= \int_{\mathbb{T}^{n}} |V(e^{-it_1}, \cdots, e^{-it_n})|^2 
\prod_{k=1}^n \prod_{j=1}^n \left|\frac{1}{1-a_j
  e^{-it_k}}\right|^2 \prod_{k=1}^n \frac{dt_k}{2\pi} \\
&= \left(\frac{1}{2\pi i}\right)^{n} 
\oint_{C^{n}} |V(z_1^{-1}, \cdots, z_n^{-1})|^2 
\prod_{k=1}^n \prod_{j=1}^n \left|\frac{1}{1-a_j
  z_k^{-1}}\right|^2 \prod_{k=1}^n \frac{dz_k}{z_k} \\
&= (-1)^{{n \choose 2}} \left(\frac{1}{2\pi i}\right)^{n} 
\oint_{C^{n}} V(\mathbf{z})^2 C(\mathbf{z},\mathbf{a})S(\mathbf{z},\mathbf{a}) \prod_{k=1}^n dz_k, 
 \end{align*}
where we used $\bar{z}_k = z_k^{-1}$ on $C$ and the following identity: 
\[
 V(z_1^{-1}, \cdots, z_n^{-1}) = (-1)^{{n \choose 2}} 
\frac{V(z_1, \cdots, z_n)}{(z_1 \cdots z_n)^{n-1}}. 
\]
Therefore, by \eqref{eq:reproducing-VC}, we have
 $\mu_{\mathbf{a}}(\mathbb{T}^n) = S(\mathbf{a},\mathbf{a})$. 

Similarly as above, setting $\tilde{\mathbf{z}} = (z_0,z_1,\dots,z_n)$
 and $\tilde{\mathbf{a}}_z=(z,\mathbf{a}) = (z,a_1,\dots,a_n)$ and using \eqref{eq:reproducing-VC}, we have 
 \begin{align*}
\lefteqn{\tilde{\mu}_{\mathbf{a}}(\mathbb{T}^{n+1}) K_{\mathbf{a}}^{(n+1)}(z,w)}\\
&= \int_{\mathbb{T}^{n+1}} S(z,\tilde{\mathbf{t}})
  \overline{S(w,\tilde{\mathbf{t}})} \tilde{\mu}_{\mathbf{a}}(d\tilde{\mathbf{t}}) \\
&=\frac{1}{(n+1)!}\int_{\mathbb{T}^{n+1}} |V(e^{-it_0}, \cdots, e^{-it_n})|^2 
\prod_{k=0}^n \prod_{j=1}^n \left|\frac{1}{1-a_j
  e^{-it_k}}\right|^2 \\ 
&\quad \times \prod_{k=0}^n \frac{1}{1-z e^{-it_k}}
  \overline{\frac{1}{1-w e^{-it_k}}} 
\prod_{k=0}^n \frac{dt_k}{2\pi} \\
&= (-1)^{{n+1 \choose 2}} \frac{1}{(n+1)!} 
\left(\frac{1}{2\pi i}\right)^{n+1} 
\oint_{C^{n+1}} V(\tilde{\mathbf{z}})^2 C(\tilde{\mathbf{z}}, \tilde{\mathbf{a}}_z)
  S(\tilde{\mathbf{z}},\tilde{\mathbf{a}}_w)\prod_{k=0}^n dz_k \\
&= S(\tilde{\mathbf{a}}_z, \tilde{\mathbf{a}}_w). 
\end{align*}
Hence, $K_{\mathbf{a}}^{(n+1)}(z,w) = S(\tilde{\mathbf{a}}_z,
 \tilde{\mathbf{a}}_w)/\tilde{\mu}_{\mathbf{a}}(T^{n+1})$.  
Since the left-hand side is $1$ when $z=w=0$ 
by the definition of $K_{\mathbf{a}}^{(n+1)}(z,w)$, we see that $\tilde{\mu}_{\mathbf{a}}(\mathbb{T}^{n+1}) =
S(\tilde{\mathbf{a}}_0,\tilde{\mathbf{a}}_0) = S(\mathbf{a},\mathbf{a})$, and thus, 
$K_{\mathbf{a}}^{(n+1)}(z,w) = S(z,w) S(z,\mathbf{a}) \overline{S(w,\mathbf{a})}$.
