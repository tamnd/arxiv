<!-- lemma-1: body_md -->
\label{lemma:special-pf-eqns}
                                                                                                                                                                                     
    Let~$\boldsymbol{x} \in \mathbb{C}^n$ be a nodal voltage state inducing power injections~$\boldsymbol{s} \in \mathbb{C}^n$. There exists a matrix-valued function~$\boldsymbol{F}: \mathbb{C}^n \to \mathbb{C}^{n \times n(n-1)/2}$ such that the complex conjugate of the power injections can be written as a parameterized function of all possible admittance weights~$\boldsymbol{w} \in \mathbb{C}^{{n \choose 2}}$, namely,
    \begin{equation}
    \label{eq:topology-paramterized-pf-eqns}
        \mkern 1.5mu\underline{\mkern-1.5mu\boldsymbol{s}\mkern-1.5mu}\mkern 1.5mu(\boldsymbol{x};\boldsymbol{w}) = \boldsymbol{F}(\boldsymbol{x}) \boldsymbol{w},
    \end{equation}
    where $\boldsymbol{F}: \mathbb{C}^n \to \mathbb{C}^{n \times n(n-1)/2}$ is defined explicitly in \eqref{eq:special-pf-eq-representation}, and $\boldsymbol{w}$ is as in \eqref{eq:def:all-possible-weights}.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
First we prove the claim for the case where the system has no phase-shifting transformers. 

Note that for any number of buses $n$, there exists a mapping~$\mathsf{A}_{(\cdot)} : n \mapsto \{-1,0,1\}^{n(n-1)/2 \times n}$ to the unique incidence matrix of a complete undirected graph with a self-edge at every node. Denote the $i$-th standard basis vector in~$\mathbb{R}^d$ as~$\boldsymbol{e}^{d}_i \in \mathbb{R}^d$; we then have that~$\mathsf{A}$ takes the form
\begin{equation}
    \label{eq:def:agnostic-incidence-matrix}
    \begin{split}
    \mathsf{A}_n &= 
                     
                                                           
                                                                 
                                                  
                              
                                                                           
                                                    
                                         
            
                                                  
                    
        
    \begin{bmatrix}
    \boldsymbol{1}_{n-1} & -\boldsymbol{e}^{n-1}_{1} & -\boldsymbol{e}^{n-1}_{2}  & \dots  & -\boldsymbol{e}^{n-1}_{n-1}\\
    \boldsymbol{0}_{n-2} & \boldsymbol{1}_{n-2} & -\boldsymbol{e}^{n-2}_{1} &\ddots & -\boldsymbol{e}^{n-2}_{n-2}\\
    \vdots & \ddots & \ddots & \ddots & \vdots \\
                                                     
    0 & \dots & 0 & 1 & -1 \\
    \hline
     \cdot & \cdot & \mathbf{I}_{n} & \cdot & \cdot \\
    \end{bmatrix},
       
                        
                  
                
                   
    \end{split}
\end{equation}
where $\boldsymbol{1}_d$ and $\boldsymbol{0}_d$ denote $d$-dimensional vectors of all ones and zeros, respectively, and $\mathbf{I}_d$ denotes an identity matrix of dimension $d \times d$. The $\mathbf{I}_n$ block concatenated at the bottom of \eqref{eq:def:agnostic-incidence-matrix} is due to the presence of self edges at every node. We remark that for \textit{any possible topology weights} $\boldsymbol{w}$, we have
\begin{subequations}
\label{eq:special-pf-eq-representation}
    \begin{align}
        \mkern 1.5mu\underline{\mkern-1.5mu\boldsymbol{s}\mkern-1.5mu}\mkern 1.5mu = \operatorname{diag}(\mkern 1.5mu\underline{\mkern-1.5mu\boldsymbol{x}\mkern-1.5mu}\mkern 1.5mu) Y\boldsymbol{x} &= \operatorname{diag}(\mkern 1.5mu\underline{\mkern-1.5mu\boldsymbol{x}\mkern-1.5mu}\mkern 1.5mu) \mathsf{A}_n^\top \operatorname{diag}(\boldsymbol{w}) \mathsf{A}_n \boldsymbol{x}\\
        &= \underbrace{\operatorname{diag}(\mkern 1.5mu\underline{\mkern-1.5mu\boldsymbol{x}\mkern-1.5mu}\mkern 1.5mu) \mathsf{A}_n^\top \operatorname{diag}\left(\mathsf{A}_n \boldsymbol{x}\right)}_{:= \boldsymbol{F}(\boldsymbol{x})} \boldsymbol{w}\\
        &:= \boldsymbol{F}(\boldsymbol{x}) \boldsymbol{w},
    \end{align}
\end{subequations}
which yields the desired representation of the power flow equations. 
                                                                                                                                                                                

The case where the system has phase-shifting transformers is proven in the same way, with the only difference that now $\mathsf{A}_n$ has the form
\begin{equation}
    \mathsf{A}_n = \begin{bmatrix}
        \boldsymbol{A}(n)\\
        \mathbf{I}
    \end{bmatrix}.
\end{equation}
where $\boldsymbol{A}(n)$ is the \textit{generalized incidence matrix} (see \cite{turizo2023}) of a complete $n$-node graph without self edges, and with phase-shifts chosen to match those of the system transformers.

<!-- assumption-1: body_md -->
\label{assum:no-voltage-collapse}
    Let $(\boldsymbol{v}^\bullet_\mathcal{P},\boldsymbol{\delta}^\bullet) \in \mathbb{R}^{(n-g)}_+ \times (-\pi,\pi]^n$ and $\boldsymbol{x}^\bullet \in \mathbb{C}^n$ be a solution to the power flow equations in polar and rectangular coordinates, respectively, where  $\boldsymbol{x}^\bullet := \boldsymbol{v}(\boldsymbol{v}^\bullet_\mathcal{P}) \circ \exp(\mathrm{j} \boldsymbol{\delta}^\bullet)$. The corresponding power flow Jacobian is assumed to be non-singular at such an operating point.
<!-- end assumption-1 -->
<!-- theorem-1: body_md -->
\label{thm:dvdw}
    Consider a power system with admittance parameters~$\boldsymbol{w}^\bullet := \boldsymbol{g}^\bullet + \mathrm{j} \boldsymbol{b}^\bullet$ and a corresponding power flow solution~$(\boldsymbol{v}^\bullet_\mathcal{P},\boldsymbol{\delta}^\bullet)$. If Assumption~\ref{assum:no-voltage-collapse} holds, the Jacobian of the power flow solution with respect to the network admittance parameters~$\boldsymbol{w} := \boldsymbol{g} + \mathrm{j} \boldsymbol{b}$ is well-defined in a neighborhood of~$(\boldsymbol{g}^\bullet, \boldsymbol{b}^\bullet)$.
<!-- end theorem-1 -->
<!-- proof-2: body_md -->
Denote the real and imaginary components of the vectorized admittance weights as $ \boldsymbol{g} := \operatorname{Re}\{\boldsymbol{w}\}$ and $ \boldsymbol{b} := \operatorname{Im}\{ \boldsymbol{w}\}$. Then, by applying Lemma \ref{lemma:special-pf-eqns}, we can construct the topology-parameterized power flow equations \eqref{eq:topology-paramterized-pf-eqns} in terms of polar coordinates for the state as the function 
\begin{equation}
    \mkern 1.5mu\underline{\mkern-1.5mu\boldsymbol{s}\mkern-1.5mu}\mkern 1.5mu := \boldsymbol{F}(\boldsymbol{v}_\mathcal{P},\boldsymbol{\delta})\boldsymbol{w} := \boldsymbol{F}(\boldsymbol{v}(\boldsymbol{v}_\mathcal{P}) \circ \exp(\mathrm{j} \boldsymbol{\delta})) (\boldsymbol{g} + \mathrm{j} \boldsymbol{b}).
\end{equation}
There is a mapping $\boldsymbol{\kappa} : \mathbb{R}^n \times \mathbb{R}^n \times \mathbb{R}^{n(n-1)/2} \times \mathbb{R}^{n(n-1)/2} \to \mathbb{R}^{(2n-g)}$ parameterized by the network topology parameters $\boldsymbol{g},\boldsymbol{b}$ with entries of the form
\begin{align*}
    &(\boldsymbol{\kappa}(\boldsymbol{v}_\mathcal{P},\boldsymbol{\delta};\boldsymbol{g},\boldsymbol{b}))_k = \nonumber \\ &\hspace*{5em}\begin{cases}
    \left(\operatorname{Re}\{\boldsymbol{F}(\boldsymbol{v}_\mathcal{P},\boldsymbol{\delta})(\boldsymbol{g} + \mathrm{j} \boldsymbol{b})\} - \boldsymbol{p}^\bullet\right)_k & k \leq n \\
    \left(\operatorname{Im}\{\boldsymbol{F}(\boldsymbol{v}_\mathcal{P},\boldsymbol{\delta})(\boldsymbol{g} + \mathrm{j} \boldsymbol{b})\} + \boldsymbol{q}^\bullet\right)_{\mathcal{P}(k-n)} & k > n \label{eq:fixed-point} \\
    \end{cases},
\end{align*} 
such that~$\boldsymbol{\kappa}(\boldsymbol{v}^\bullet_\mathcal{P},\boldsymbol{\delta}^\bullet;\boldsymbol{g}^\bullet,\boldsymbol{b}^\bullet) = \boldsymbol{0}_{2n-g}$. Let~$\partial \boldsymbol{\kappa}$ denote the derivative of~$\boldsymbol{\kappa}$ with respect to the power system state at~$(\boldsymbol{v}^\bullet_\mathcal{P},\boldsymbol{\delta}^\bullet;\boldsymbol{g}^\bullet,\boldsymbol{b}^\bullet)$, then
\begin{equation}
    \partial \boldsymbol{\kappa} = \begin{bmatrix}
    \frac{\partial \boldsymbol{\kappa}}{\partial \boldsymbol{\delta}} &  \frac{\partial \boldsymbol{\kappa}}{\partial \boldsymbol{v}_\mathcal{P}} 
\end{bmatrix} = \boldsymbol{J}(\boldsymbol{v}^\bullet_\mathcal{P},\boldsymbol{\delta}^\bullet),
\end{equation}
where $\boldsymbol{J}(\boldsymbol{v}_\mathcal{P},\boldsymbol{\delta})$ is the standard power flow Jacobian matrix in polar coordinates. If Assumption \ref{assum:no-voltage-collapse} holds, then $\boldsymbol{J}(\boldsymbol{v}^\bullet_\mathcal{P},\boldsymbol{\delta}^\bullet)$ is non-singular, so $\partial \boldsymbol{\kappa}$ is non-singular as well. By the Implicit Function Theorem, there exists unique, continuously differentiable functions $\boldsymbol{v}_\mathcal{P}(\boldsymbol{g},\boldsymbol{b})$ and $\boldsymbol{\delta}(\boldsymbol{g},\boldsymbol{b})$ defined in a neighborhood of $(\boldsymbol{g}^\bullet,\boldsymbol{b}^\bullet)$ such that $\boldsymbol{v}_\mathcal{P}(\boldsymbol{g}^\bullet,\boldsymbol{b}^\bullet) = \boldsymbol{v}_\mathcal{P}^\bullet$, $\boldsymbol{\delta}(\boldsymbol{g}^\bullet,\boldsymbol{b}^\bullet) = \boldsymbol{\delta}^\bullet$, and $\boldsymbol{\kappa}(\boldsymbol{v}_\mathcal{P}(\boldsymbol{g},\boldsymbol{b}),\boldsymbol{\delta}(\boldsymbol{g},\boldsymbol{b});\boldsymbol{g},\boldsymbol{b}) = \boldsymbol{0}_{2n-g}$ for all $(\boldsymbol{g}, \boldsymbol{b})$ in the neighborhood. Moreover, the derivatives of these functions satisfy
\begin{subequations}
\begin{align}
     \begin{bmatrix}
        \frac{\partial \boldsymbol{\delta}}{\partial \boldsymbol{g}} & \frac{\partial \boldsymbol{\delta}}{\partial \boldsymbol{b}}\\
        \frac{\partial \boldsymbol{v}_\mathcal{P}}{\partial \boldsymbol{g}} & \frac{\partial \boldsymbol{v}_\mathcal{P}}{\partial \boldsymbol{b}}
    \end{bmatrix} &= -(\partial \boldsymbol{\kappa})^{-1}\begin{bmatrix}
        \frac{\partial \boldsymbol{\kappa}}{\partial \boldsymbol{g}} & \frac{\partial \boldsymbol{\kappa}}{\partial \boldsymbol{b}}
    \end{bmatrix} \\
                     
                                                                                         
                                                                                           
                    
    &= -\boldsymbol{J}(\boldsymbol{v}^\bullet_\mathcal{P},\boldsymbol{\delta}^\bullet)^{-1}\begin{bmatrix}
        \frac{\partial \boldsymbol{\kappa}}{\partial \boldsymbol{g}} & \frac{\partial \boldsymbol{\kappa}}{\partial \boldsymbol{b}}
    \end{bmatrix}, \label{eq:implicit-function-theorem-sol}
\end{align}
\end{subequations}
so by construction the claim holds.
