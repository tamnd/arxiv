<!-- lem-1: body_md -->
\label{Hou-Zhang}
For any integer $m\geq1$, there exists a real number
\begin{equation*}
N_0(m) :=
\begin{cases}
1, &\quad \text{if}\ m=1,\\
2m \log m-m \log \log m, & \quad \text{if}\ m \geq 2,
\end{cases}
\end{equation*}
such that
\begin{align*}
x^me^{-x}<1,~~~\text{for}~~x\geq N_0(m).
\end{align*}
<!-- end lem-1 -->
<!-- proof-1: body_md -->
For $m=1$, it is immediate that $N_0(m)=1$. For $m \geq 2$, rewrite the inequality $x^me^{-x}<1$ as $f(x):=x-m\log x>0$. Now $f(x)$ is strictly increasing for $x>m$.  In order to show $f(x)>0$ for $x\geq N_0(m)$, first we show that $N_0(m)>m$ and then it is enough to show $f(N_0(m))>0$. To prove $N_0(m)>m$, it is equivalent to show $m^2>e. \log m$ which holds for $m \geq 2$. Next, we observe that
	\begin{equation}\label{lemeqn1}
	f(N_0(m))>0 \Leftrightarrow\log m  >  \log 2+2 \log \log m+\log \Bigl(1-\dfrac{\log \log m}{2\log m}\Bigr)\nonumber\\
	\end{equation}
For $m \geq 3$, $\log \Bigl(1-\dfrac{\log \log m}{2\log m}\Bigr) <0$ and hence, it is sufficient to prove $$\log m  >  \log 2+2 \log \log m \Leftrightarrow m>2\ (\log m)^2,$$
which holds for $m \geq 14$. Therefore, $f(N_0(m))>0$ for all $m \geq 14$ and we conclude the proof by checking numerically that $f(N_0(m))>0$ for $2\leq m \leq 13$.
<!-- end proof-1 -->

<!-- lem-2: body_md -->
\label{T-T-bound}
Let ${\widehat{\nu}}(n)={\widehat{\mu}}(n)({\widehat{\mu}}_1(n)-1)$ and
\begin{align*}
{\widehat{\mu}}_1(n)&=\sum_{k=0}^{m'}{1/2\choose k}\pi^{2k}{\widehat{\mu}}(n)^{-2k},\qquad
{\widehat{\varepsilon}}_1(n)=\left|{1/2\choose m'+1}\right|\pi^{2(m'+1)}{\widehat{\mu}}(n)^{-2(m'+1)},
\\[3pt]
{\widehat{\mu}}_2(n)&=\sum_{k=0}^{m'}{-3/2\choose k}\pi^{2k}{\widehat{\mu}}(n)^{-2k}, \qquad
{\widehat{\varepsilon}}_2(n)=\left|{-3/2\choose m'+1}\right|\pi^{2(m'+1)}{\widehat{\mu}}(n)^{-2(m'+1)},
\\[3pt]
{\widehat{\nu}}_1(n)&=\left({\widehat{\mu}}_1(n)-{\widehat{\varepsilon}}_1(n)-\frac{1}{{\widehat{\mu}}(n)}\right)\sum_{k=0}^m{\widehat{\mu}}(n)^{-k},
\\[3pt]
{\widehat{\nu}}_2(n)&=\left({\widehat{\mu}}_1(n)+{\widehat{\varepsilon}}_1(n)-\frac{1}{{\widehat{\mu}}(n)}\right)
\left(\sum_{k=0}^m{\widehat{\mu}}(n)^{-k}+2{\widehat{\mu}}(n)^{-m-1}\right),
\end{align*}
then
\begin{align}\label{T-ratio-low}
\frac{{\widehat{T}}(n+1)}{{\widehat{T}}(n)}&>{\widehat{\nu}}_1(n)\left({\widehat{\mu}}_2(n)-{\widehat{\varepsilon}}_2(n)\right)
(1-{\widehat{\mu}}(n){\widehat{\varepsilon}}_1(n))\sum_{k=0}^m\frac{{\widehat{\nu}}(n)^k}{k!},
\end{align}
and
\begin{align}\label{T-ratio-upp}
\frac{{\widehat{T}}(n+1)}{{\widehat{T}}(n)}&<{\widehat{\nu}}_2(n)({\widehat{\mu}}_2(n)+{\widehat{\varepsilon}}_2(n))
(1+2{\widehat{\mu}}(n){\widehat{\varepsilon}}_1(n))\left(\sum_{k=0}^m\frac{{\widehat{\nu}}(n)^k}{k!}
+e^{{\widehat{\nu}}(n)}\frac{{\widehat{\nu}}(n)^{m+1}}{(m+1)!}\right)
\end{align}
where ${\widehat{T}}(n)$ is defined as in \eqref{wT(n)}.
<!-- end lem-2 -->
<!-- head-2: body_md -->
By \eqref{wT(n)}, we have
\begin{align}\label{T-ratio}
\frac{{\widehat{T}}(n+1)}{{\widehat{T}}(n)}=\frac{{\widehat{\mu}}(n+1)-1}{{\widehat{\mu}}(n)-1}\cdot\frac{{\widehat{\mu}}(n)^3}{{\widehat{\mu}}(n+1)^3}\cdot e^{{\widehat{\mu}}(n+1)-{\widehat{\mu}}(n)}.
\end{align}
Now we consider the above ratio term by term.
For the first factor, we have
\begin{align*}
\frac{{\widehat{\mu}}(n+1)-1}{{\widehat{\mu}}(n)-1}=\frac{\frac{{\widehat{\mu}}(n+1)}{{\widehat{\mu}}(n)}-\frac{1}{{\widehat{\mu}}(n)}}{1-\frac{1}{{\widehat{\mu}}(n)}}.
\end{align*}
By Taylor's Theorem, we have
\begin{align*}
\left(1-\frac{1}{{\widehat{\mu}}(n)}\right)^{-1}
=\sum_{k=0}^\infty {\widehat{\mu}}(n)^{-k},
\end{align*}
which implies that
\begin{align}\label{wT-n+1-1-1}
\sum_{k=0}^m{\widehat{\mu}}(n)^{-k}<\left(1-\frac{1}{{\widehat{\mu}}(n)}\right)^{-1}
<\sum_{k=0}^m{\widehat{\mu}}(n)^{-k}+2{\widehat{\mu}}(n)^{-m-1}.
\end{align}
Noting that
\begin{align*}
{\widehat{\mu}}(n+1)={\widehat{\mu}}(n)\left(1+\frac{\pi^2}{{\widehat{\mu}}(n)^2}\right)^{\frac{1}{2}}.
\end{align*}
For any integer $m$, let $m'=\lfloor\frac{m}{2}\rfloor$. Since
\begin{align*}
\left(1+\frac{\pi^2}{{\widehat{\mu}}(n)^2}\right)^{\frac{1}{2}}=\sum_{k=0}^{m'}{1/2\choose k}\pi^{2k}{\widehat{\mu}}(n)^{-2k}
+{1/2\choose m'+1}\left(\frac{\pi^2}{{\widehat{\mu}}(n)^2}\right)^{m'+1}\left(1+\xi\right)^{\frac{1}{2}-m'-1},
\end{align*}
where $0<\xi<\frac{\pi^2}{{\widehat{\mu}}(n)^2}$.
We have
\begin{align}\label{wT-n+1-1-2}
{\widehat{\mu}}_1(n)-{\widehat{\varepsilon}}_1(n)<\frac{{\widehat{\mu}}(n+1)}{{\widehat{\mu}}(n)}<{\widehat{\mu}}_1(n)+{\widehat{\varepsilon}}_1(n).
\end{align}
Combining \eqref{wT-n+1-1-1} and \eqref{wT-n+1-1-2}, we deduce that
\begin{align}\label{wT-n+1-1}
{\widehat{\nu}}_1(n)<\frac{{\widehat{\mu}}(n+1)-1}{{\widehat{\mu}}(n)-1}<{\widehat{\nu}}_2(n).
\end{align}
For the second factor, we have
\begin{align*}
\frac{{\widehat{\mu}}(n)^3}{{\widehat{\mu}}(n+1)^3}=\left(1+\frac{\pi^2}{{\widehat{\mu}}(n)^2}\right)^{-3/2}.
\end{align*}
Since
\begin{align*}
\left(1+\frac{\pi^2}{{\widehat{\mu}}(n)^2}\right)^{-\frac{3}{2}}=\sum_{k=0}^{m'}{-3/2\choose k}\pi^{2k}{\widehat{\mu}}(n)^{-2k}
+{-3/2\choose m'+1}\left(\frac{\pi^2}{{\widehat{\mu}}(n)^2}\right)^{m'+1}\left(1+\xi\right)^{-\frac{3}{2}-m'-1},
\end{align*}
where $0<\xi<\frac{\pi^2}{{\widehat{\mu}}(n)^2}$.
We have
\begin{align}\label{wT-n+1-2}
{\widehat{\mu}}_2(n)-{\widehat{\varepsilon}}_2(n)<\frac{{\widehat{\mu}}(n)^3}{{\widehat{\mu}}(n+1)^3}<{\widehat{\mu}}_2(n)+{\widehat{\varepsilon}}_2(n).
\end{align}
For the last factor, using \eqref{wT-n+1-1-2}, then
\begin{align*}
e^{{\widehat{\nu}}(n)-{\widehat{\mu}}(n){\widehat{\varepsilon}}_1(n)}<e^{{\widehat{\mu}}(n+1)-{\widehat{\mu}}(n)}<e^{{\widehat{\nu}}(n)+{\widehat{\mu}}(n){\widehat{\varepsilon}}_1(n)}.
\end{align*}
Evidently, for $0<x<\frac{1}{2}$,
\[e^{-x}>1-x,\qquad e^x<1+2x\]
and for $x>0$
\[\sum_{k=0}^m\frac{x^k}{k!}<e^x<\sum_{k=0}^m\frac{x^k}{k!}+e^x\frac{x^{m+1}}{(m+1)!},\]
so that
\begin{align}\label{wT-n+1-3-low}
e^{{\widehat{\mu}}(n+1)-{\widehat{\mu}}(n)}>(1-{\widehat{\mu}}(n){\widehat{\varepsilon}}_1(n))\sum_{k=0}^m\frac{{\widehat{\nu}}(n)^k}{k!}
\end{align}
and
\begin{align}\label{wT-n+1-3-upp}
e^{{\widehat{\mu}}(n+1)-{\widehat{\mu}}(n)}<(1+2{\widehat{\mu}}(n){\widehat{\varepsilon}}_1(n))\left(\sum_{k=0}^m\frac{{\widehat{\nu}}(n)^k}{k!}
+e^{{\widehat{\nu}}(n)}\frac{{\widehat{\nu}}(n)^{m+1}}{(m+1)!}\right).
\end{align}
Applying the estimates \eqref{wT-n+1-1}, \eqref{wT-n+1-2}, \eqref{wT-n+1-3-low} and \eqref{wT-n+1-3-upp} to \eqref{T-ratio}, we reach \eqref{T-ratio-low} and \eqref{T-ratio-upp}. This completes the proof.

<!-- head-3: body_md -->
By \eqref{p-T-R} and Theorem \ref{wyn-bound}, for any $m\geq2$ there exists the integer $N_1(m)$ such that

  |{\overline{p}}(n)/{\widehat{T}}(n)-1|<\left(\frac 32\right)^{m+1}{\widehat{\mu}}(n)^{-m},~~\forall n\geq N_1(m).

Since ${\widehat{\mu}}(n)$ is a increasing function of $n$, we derive that
\begin{equation*}
  \frac {{\widehat{T}}(n+1)}{{\widehat{T}}(n)} \frac {1-\left(\frac 32\right)^{m+1}{\widehat{\mu}}(n)^{-m}}{1+\left(\frac 32\right)^{m+1}{\widehat{\mu}}(n)^{-m}}<\frac {{\overline{p}}(n+1)}{{\overline{p}}(n)}<\frac {{\widehat{T}}(n+1)}{{\widehat{T}}(n)} \frac {1+\left(\frac 32\right)^{m+1}{\widehat{\mu}}(n)^{-m}}{1-\left(\frac 32\right)^{m+1}{\widehat{\mu}}(n)^{-m}}.
\end{equation*}
We find that for $0<\lambda<1/3$,
\begin{equation*}
  \frac {1+\lambda}{1-\lambda}<1+3\lambda~~~ and~~~\frac {1-\lambda}{1+\lambda}>1-2\lambda.
\end{equation*}
According to
\[0<\frac {\left(\frac 32\right)^{m+1}}{{\widehat{\mu}}(n)^m}<1/3,\]
we have that for all $n\geq N_1(m)$
\begin{equation}\label{p-p-bound}
  \frac {{\widehat{T}}(n+1)}{{\widehat{T}}(n)}\left(1-4\cdot2^m{\widehat{\mu}}(n)^{-m}\right)<\frac {{\overline{p}}(n+1)}{{\overline{p}}(n)}<\frac {{\widehat{T}}(n+1)}{{\widehat{T}}(n)}\left(1+6\cdot2^m{\widehat{\mu}}(n)^{-m}\right).
\end{equation}
By Lemma \ref{T-T-bound}, we can see that ${\widehat{T}}(n+1)/{\widehat{T}}(n)$ is bounded by a pair of polynomials in ${\widehat{\mu}}(n)^{-1}$ whose difference is a polynomials in ${\widehat{\mu}}(n)^{-1}$ with degree at least $m+1$. Combining \eqref{p-p-bound} and $\lim_{n\rightarrow+\infty}{\widehat{\mu}}(n)=+\infty$, we have \eqref{p-p-b}. This completes the proof.

Let $\{a_n\}_{n\geq0}$ be a positive sequence such that $\mathscr{R}^2a_n=a_na_{n+2}/a_{n+1}^2$ has the following asymptotic expression
