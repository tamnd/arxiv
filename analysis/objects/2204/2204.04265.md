\label{Thm:PoissonLp}    Assume that the sequence $\{a_j\}_{j\in \mathbb Z}$ is a $\rho$-lacunary sequence with $\rho >1$. Let $T^*$ be   the operator   defined in \eqref{Formu:MaxSquareFun}. We have the following statements.
\begin{enumerate}[(a)]
    \item For any $1<p<\infty$ and $\omega\in A_p$,  there exists a constant $C$ depending  on $\rho, p,\omega,  \lambda$ and $\left\Vertv\right\Vert_{l^\infty(\mathbb Z)}$ such that
 $$\left\VertT^* f\right\Vert_{L^p(\mathbb R_+, \omega dm_\lambda)}\leq C\left\Vertf\right\Vert_{L^p(\mathbb R^{n}, \omega dm_\lambda)},$$
 for all functions $f\in L^p({\mathbb R}_+, \omega dm_\lambda).$
    \item For any  $\omega\in A_1$, there exists a constant $C$ depending  on $\rho, \omega, \lambda$ and  $\left\Vertv\right\Vert_{\ell^\infty(\mathbb Z)}$   such that
 $$\omega\left({\{x\in {\mathbb R}_+:\left\vertT^* f(x)\right\vert>\sigma\}}\right) \le C\frac{1}{\sigma}\left\Vertf\right\Vert_{L^1(\mathbb R_+, \omega dm_\lambda)}, \quad \sigma>0,$$
for all functions $f\in L^1({\mathbb R}_+, \omega  dm_\lambda).$

\item Let $\{a_j\}_{j\in \mathbb Z}$ be  a $\rho$-lacunary sequence. And in this latter case, there exists a constant $C$ depending  on $\lambda, \rho$ and  $\left\Vertv\right\Vert_{\ell^\infty(\mathbb Z)}$ such that
$$\left\VertT^*f\right\Vert_{BMO(\mathbb R_+, dm_\lambda)}\leq C\left\Vertf\right\Vert_{L^\infty(\mathbb R_+, dm_\lambda)}.$$

\end{enumerate}

<!-- proof-1: body_md -->
Let $f\in \text{Dom}(\Delta_\lambda)$. Using  the Plancherel theorem for Hankel transform, we have
\begin{align*}
 \left\VertT_N f \right\Vert^2_{L^2({\mathbb R}_+, dm_\lambda)} & = \left\Vert\sum_{j=N_1}^{N_2} v_j\left({\mathcal P}_{a_{j+1}} f -{\mathcal P}_{a_{j}}f\right)\right\Vert^2_{L^2({\mathbb R}_+, dm_\lambda)}\\
 &=  \int_{\mathbb{R}_+} \Big\{\sum_{j=N_1}^{N_2} v_j\left({\mathcal P}_{a_{j+1}} f(x) -{\mathcal P}_{a_{j}}f(x)\right) \Big\}^2 dm_\lambda(x) \\
  &=
 \int_{\mathbb{R}_+} \left(H_\lambda{\Big\{\sum_{j=N_1}^{N_2} v_j\left({\mathcal P}_{a_{j+1}} f(\cdot) -{\mathcal P}_{a_{j}}f(\cdot)\right) \Big\}}(y)\right)^2 dm_\lambda(y)\\
  & =  \int_{\mathbb{R}_+} \Big\{\sum_{j=N_1}^{N_2} v_j \int_{a_j}^{a_{j+1}}\partial_t H_\lambda \left({\mathcal P}_t f \right)(y)dt  \Big\}^2 dm_\lambda(y)\\
 &\le
   C\left\Vertv\right\Vert^2_{\ell^\infty(\mathbb Z)}
   \int_{\mathbb{R}_+} \Big\{\sum_{j=N_1}^{N_2} \Big|\int_{a_j}^{a_{j+1}}\partial_t H_\lambda \left({\mathcal P}_t f\right)(y) dt\Big|  \Big\}^2 dm_\lambda(y)\\  &=
   C_v
   \int_{\mathbb{R}_+} \Big\{\sum_{j=N_1}^{N_2} \Big|\int_{a_j}^{a_{j+1}}\left\verty\right\vert e^{-t\left\verty\right\vert} H_\lambda ( f)(y)dt\Big|  \Big\}^2 dm_\lambda(y)\\
   &\le    C_v
   \int_{\mathbb{R}_+} \Big\{\sum_{j=N_1}^{N_2} \int_{a_j}^{a_{j+1}}\left\verty\right\vert e^{-t\left\verty\right\vert} dt|H_\lambda( f)(y)|  \Big\}^2 dm_\lambda(y)\\ &\le  C_v
   \int_{\mathbb{R}_+} \Big\{\Big|\int_0^\infty\left\verty\right\vert e^{-t\left\verty\right\vert} dt\Big||H_\lambda( f)(y)|  \Big\}^2 dm_\lambda(y)\\
   & \le  C_{v, \lambda} \|f\|_{L^2({\mathbb R}_+, dm_\lambda)}^2.
\end{align*}
Since $\text{Dom}(\Delta_\lambda)$ is dense in $L^2({\mathbb R}_+, dm_\lambda),$ $T_N$ extends uniquely to  a bounded linear operator on $L^2({\mathbb R}_+, dm_\lambda).$
Then the proof of the theorem is complete.

\label{Thm:KernelEst}
Let $f\in L^p(\mathbb{R}_+, dm_\lambda), 1\le p \le \infty$. Then
$$T_Nf(x) = \int_{\mathbb{R}_+} K_N(x,y)f(y)dm_\lambda(y)$$
with
\begin{align*}\label{kernel}
&K_N(x, y) =  \sum_{j=N_1}^{N_2}v_j \left({\mathcal P}_{a_{j+1}} (x,y)-{\mathcal P}_{a_{j}}(x,y)\right)\\
&={2\lambda\over \pi}\sum_{j=N_1}^{N_2}v_j \left(\int_0^\pi{a_{j+1}(\sin \theta)^{2\lambda-1}\over {(x^2+y^2+a_{j+1}^2-2xy\cos \theta)^{\lambda+1}}}d\theta -\int_0^\pi{a_{j}(\sin \theta)^{2\lambda-1}\over {(x^2+y^2+a_{j}^2-2xy\cos \theta)^{\lambda+1}}}d\theta\right).
\end{align*}
Moreover, there exists  constant $C>0$ depending  on $\lambda$  and $\left\Vertv\right\Vert_{\ell^\infty(\mathbb Z)}$(not on $N$) such that, for any $x\neq y,$
\begin{enumerate}[\indent i)]
  \item $\displaystyle   |K_N(x,y)|\leq \frac{C}{m_\lambda(I(x,|x-y|))}$,
  \item $\displaystyle   |\partial_x K_N(x,y)|+ |\partial_y K_N(x,y)|\leq \frac{C}{m_\lambda(I(x,|x-y|))\cdot |x-y|}$.
\end{enumerate}
<!-- end prop-2 -->
<!-- proof-2: body_md -->
{\it i)}~Regarding  the size condition for the kernel, we consider it into two cases:

 Case 1: $x\le 2|x-y|$. In this case, $$m_\lambda{(I(x, |x-y|))}\sim |x-y|^{2\lambda+1}.$$    By Lemma \ref{KerPoisson} $iii)$,  we have
\begin{align*}
  |K_N (x, y)| &\leq  \sum_{j=N_1}^{N_2}\left\vertv_j\right\vert \left\vert{\mathcal P}_{a_{j+1}} (x,y)-{\mathcal P}_{a_{j}(x,y)}(x,y)\right\vert\\
&\le C_{\lambda, v} \sum_{j=-\infty}^{\infty}  \left|\int_{a_j}^{a_{j+1}}\partial_t  {\mathcal P}_t(x, y) dt\right|\\
   &\le C_{\lambda, v}\int_{0}^{\infty}{1\over {(|x-y|^2+t^2)^{\lambda+{1}}}}{dt } \\
    &\le  C_{\lambda, v}{1 \over  {|x-y|^{2\lambda+1}}}=\frac{C}{m_\lambda(I(x,|x-y|))}.
\end{align*}


Case 2: $x>2|x-y|$. In this case, $\displaystyle {x\over 2}\le y\le {3x\over 2}$ and $$m_\lambda{(I(x, |x-y|))}\sim  x^{2\lambda}|x-y|\sim  (xy)^{\lambda}|x-y|.$$    By Lemma \ref{KerPoisson} $iii)$,  we have
\begin{align*}
  |K_N (x, y)| &\leq \frac{2\lambda}{ \pi} \sum_{j=N_1}^{N_2}\left\vertv_j\right\vert \left\vert{\mathcal P}_{a_{j+1}} (x,y)-{\mathcal P}_{a_{j}(x,y)}(x,y)\right\vert\\
&\le C_{\lambda, v} \sum_{j=-\infty}^{\infty}  \left|\int_{a_j}^{a_{j+1}}\partial_t  {\mathcal P}_t(x, y) dt\right|\\
   &\le C_{\lambda, v}\int_{0}^{\infty}{1\over {(xy)^\lambda(|x-y|^2+t^2)}}{dt } \\
    &\le  C_{\lambda, v}{1 \over  {(xy)^\lambda |x-y|}}=\frac{C}{m_\lambda(I(x,|x-y|))}.
\end{align*}



{\it ii)}~
 With a similar argument as above in $i)$,  we can get the proof of $ii)$ but by using the estimates in Lemma \ref{KerPoisson} $iv).$
The proof of the  proposition is complete.

<!-- proof-3: body_md -->
By Propositions \ref{Thm:L2Estimate} and \ref{Thm:KernelEst}, we know that the operator $T_N$ is a Calder\'on-Zygmund operator with standard kernel. Previously, we have remarked  that the constants  in the $L^p$ boundedness only depend on the initial constant in $L^{p_0}(\mathbb{R}_+)$(in our case $p_0=2$), the size constant and smoothness constant of the kernel.  Hence the uniform boundedness of the operators $T_N$ in $L^p(\mathbb{R}_+, dm_\lambda)$ spaces is a direct consequence of the vector-valued Calder\'on-Zygmund theory in homogeneous type space $(\mathbb R_+, |\cdot|, dm_\lambda)$.  So, it is bounded on the weighted $L_p$-spaces for $1<p<\infty$, and is of weak type $(1,1)$. Then, we get the proof of $(a)$ and $(b)$.

For $(c)$ and $(d),$ we can prove them as following.
 The finiteness of $T_N$ for functions in  $L^\infty({\mathbb R}_+, dm_\lambda)$ is obvious, since for each $N$, $K_N$ is an integrable function.
  On the other hand, if $f\in BMO({\mathbb R}_+, dm_\lambda)$, we can proceed as follows. Let $I=I(x_0, r_0)$ and $I^*=2I=I(x_0, 2r_0)$ with some $x_0\in {\mathbb R}_+$ and $r_0>0$. We decompose $f$ to  be
$$f=(f-f_{I })\chi_{I^*}+(f-f_{I })\chi_{(I^*)^c}+f_{I }=:f_1+f_2+f_3,$$
where $\displaystyle f_{I }={1\over m_\lambda(I(x_0,r_0))}\int_{I(x_0,r_0)}f(y)\  dm_\lambda(y).$
The function $f_1$ is integrable, hence  $T_N f_1(x)<\infty,$ $a.e.\ x\in {\mathbb R}_+.$ For $T_N f_2$, we note that, for any $x\in I$,   $y\in \{y\in \mathbb R_+: 2^k r_0<|x_0-y|\le 2^{k+1}r_0\}$ with  $k\ge 1$, $|x-y|\sim |y-x_0|$ and we claim that
\begin{equation}\label{eq:max}
 \max \left\{ {t\over{(|x-y|^2+t^2)^{\lambda+1}}},\ {t\over {(xy)^{\lambda}(|x-y|^2+t^2)}}    \right\}\le C {t\over {m_\lambda(I(x_0, 2^kr_0))\cdot 2^kr_0}}.
\end{equation}
In fact, we can prove \eqref{eq:max} by considering the following two cases:


Case 1: $x_0\le 4\cdot 2^{k}r_0.$ In this case, $$m_\lambda(I(x_0, 2^kr_0))\sim (2^k r_0)^{2\lambda+1}.$$
So
$${t\over{(|x-y|^2+t^2)^{\lambda+1}}}\le {t\over{|x-y|^{2\lambda+2}}}\le C {t\over{(2^kr_0)^{2\lambda+2}}}=C {t\over {m_\lambda(I(x_0, 2^kr_0))\cdot 2^kr_0}}.$$

Case 2: $x_0 >4\cdot  2^{k}r_0.$ In this case,  $$m_\lambda(I(x_0, 2^kr_0))\sim x_0^{2\lambda+1}2^k r_0.$$
Since $x\in I$ and $y\in \{y\in \mathbb R_+: 2^k r_0<|x_0-y|\le 2^{k+1}r_0\}$, $x\sim y\sim x_0.$ And then, we get
$${t\over {(xy)^{\lambda}(|x-y|^2+t^2)}}\le {t\over{x_0^{2\lambda}|x_0-y|^{2}}}\le C {t\over{(x_0^{2\lambda}2^kr_0)\cdot 2^kr_0}}=C {t\over {m_\lambda(I(x_0, 2^kr_0))\cdot 2^kr_0}}.$$

Then, for any $t>0,$ by \eqref{eq:max} we have
\begin{align*}
\left\vert{\mathcal P}_t f_2(x)\right\vert&= \left\vert\int_{{\mathbb R}_+}  {\mathcal P}_t(x,y) f_2 (y)dm_\lambda(y)\right\vert\\
 &\le C \sum_{k=1}^\infty\Big|\int_{\{2^kr_0<|x_0-y|\le 2^{k+1}r_0\}\cap \mathbb R_+}
           \max \left\{ {t\over{(|x-y|^2+t^2)^{\lambda+1}}},\ {t\over {(xy)^{\lambda}(|x-y|^2+t^2)}}    \right\}\\
           &\quad\quad\quad\quad\quad\quad\quad\quad\quad\quad\quad\quad\quad\quad
           \quad\quad\quad\quad\quad\quad\quad\quad\quad\quad\quad\quad\quad\quad\quad\quad \cdot ({f(y)-f_{I}})dm_\lambda(y)\Big|\\
&\le Ct\sum_{k=1}^\infty{(2^kr_0)}^{-1}{1\over {m_\lambda(I(x_0, 2^kr_0))}}\left\vert\int_{\{|x_0-y|\le 2^{k+1}r_0\}\cap \mathbb R_+}  ({f(y)-f_{I}})dm_\lambda(y)\right\vert\\
&\le  Ct  \sum_{k=1}^\infty{(2^kr_0)}^{-1}(1+ k)\left\Vertf\right\Vert_{BMO({\mathbb R}_+, dm_\lambda)}<\infty.
\end{align*}
So, ${\mathcal P}_t f_2(x)$ is finite for any $x\in I$ and $t>0.$ Since $T_N f_2(x)$ is a finite summation and $x_0, r_0$ is arbitrary, $T_N f_2(x)<\infty$ $a.e.$ $x\in {\mathbb R}_+.$
Finally we note that $T_N f_3(x)\equiv 0$, since ${\mathcal P}_{a_j} f_3= f_{I}$ for any $j\in \mathbb Z.$
Hence, $T_N f(x)<\infty$ $a.e.$ $x\in {\mathbb R}_+.$ Then, by Propositions \ref{Thm:L2Estimate} and  \ref{Thm:KernelEst}  we  get the proof of part $(c)$ of   Theorem \ref{Thm:BMO}. Since $T_N1=0$, we can get the  conclusion of $(d)$.

Given a $\rho$-lacunary sequence $\{a_j\}_{j\in \mathbb Z}$ and a multiplying sequence $\{v_j\}_{j\in \mathbb Z}\in \ell^\infty(\mathbb Z)$, we can define a $\rho$-lacunary sequence $\{\eta_j\}_{j\in \mathbb Z}$ and $\{\omega_j\}_{j\in \mathbb Z}\in \ell^\infty(\mathbb Z)$ verifying the following properties:

\end{enumerate}

In order to prove $(b)$, we consider the $\ell^\infty(\mathbb Z^2)$-valued operator
$\mathcal{T}f(x) = \{  T_N f(x) \}_{N\in \mathbb Z^2}$. Since  $\|\mathcal{T}f(x) \|_{\ell^\infty(\mathbb Z^2)}= T^*f(x)$,   by using $(a)$ we know that the operator $\mathcal{T}$ is bounded from $L^p({\mathbb R}_+, \omega dm_\lambda) $ into $L^p_{\ell^\infty(\mathbb Z^2)}(\mathbb{R}_+, \omega dm_\lambda) $, for every $1<p<\infty$ and $\omega \in A_p$. The kernel of the operator $\mathcal{T}$ is given by $\mathcal{K}(x) = \{ K_N(x)\} _{N\in \mathbb Z^2}$. Therefore, by the vector valued Calder\'on-Zygmund theory, the operator $\mathcal{T}$ is bounded from $L^1(\mathbb{R}_+, \omega dm_\lambda)$ into weak-$L^1_{\ell^\infty(\mathbb Z^2)}(\mathbb{R}_+, \omega dm_\lambda)$ for $\omega \in A_1$. Hence, as $\|\mathcal{T}f(x) \|_{\ell^\infty(\mathbb Z^2)}= T^*f(x)$, we get the proof of  $(b)$.


For $(c)$, we shall prove that, if $f\in BMO(\mathbb R_+, dm_\lambda)$ and there exists $x_0\in \mathbb R_+$ such that $T^*f(x_0)<\infty,$ then $T^*f(x)<\infty$ for $a.e.$ $x\in {\mathbb R}_+.$
Set $I=I(x_0, 4\left\vertx-x_0\right\vert)$ with $x\neq x_0$. And we decompose $f$ to be
\begin{equation*}
f=(f-f_I)\chi_I+(f-f_I)\chi_{I^c}+f_I=:f_1+f_2+f_3.
\end{equation*}
Note that $T^*$ is $L^p$ bounded for any $1<p<\infty.$ Then $T^*f_1(x)<\infty$, because $f_1\in L^p(\mathbb R_+, dm_\lambda)$, for any $1<p<\infty.$ And  $T^* f_3=0$, since ${\mathcal P}_{a_j}f_3=f_3$ for any $j\in \mathbb Z.$
On the other hand by the smoothness properties of the kernel , we have
\begin{align*}
&\Big|T_N f_2(x)-T_N f_2(x_0)\Big|=\Big| \int_{I^c}\left(K_N(x, y) - K_N (x_0, y)\right)f_2(y)dm_\lambda(y)\Big|\\
&\le C \int_{I^c}  \frac{\left\vertx-x_0\right\vert}{m_\lambda(I(x_0, |y-x_0|))\left\verty-x_0\right\vert}\left\vertf(y)-f_I\right\vert dm_\lambda(y)\\
&\le C \sum_{k=1}^{+\infty}{ |x-x_0|} \int_{2^{k} I\setminus 2^{k-1}I}  {\left\vertf(y)-f_I\right\vert\over {m_\lambda(I(x_0, |y-x_0|))\left\verty-x_0\right\vert}}dm_\lambda(y)\\
&\le C \sum_{k=1}^{+\infty}{ |x-x_0|\over m_\lambda(I(x_0, 2^{k+1}|x-x_0|))2^{k+1}|x-x_0|} \int_{2^kI}  {\left\vertf(y)-f_I\right\vert} dm_\lambda(y) \\ &\le C \sum_{k=1}^{+\infty}2^{-(k+1)}{ 1\over m_\lambda(I(x_0, 2^{k+1}|x-x_0|))} \int_{2^{k}I} \Big(  {\left\vertf(y)-f_{2^kI}\right\vert}+\sum_{l=1}^{k}\left\vertf_{2^lI}-f_{2^{l-1}I}\right\vert\Big)dm_\lambda(y)\\
&\le C\sum_{k=1}^{+\infty}2^{-(k+1)}{ 1\over m_\lambda(I(x_0, 2^{k+1}|x-x_0|))} \int_{2^{k}I} \Big(  {\left\vertf(y)-f_{2^kI}\right\vert}+2k\left\Vertf\right\Vert_{BMO({\mathbb R}_+, dm_\lambda)}\Big)dm_\lambda(y)\\
&\le C\sum_{k=1}^{+\infty}2^{-(k+1)}{(1+2k)\left\Vertf\right\Vert_{BMO({\mathbb R}_+, dm_\lambda)}}\\
&\le  C\left\Vertf\right\Vert_{BMO({\mathbb R}_+,dm_\lambda)},
 \end{align*}
where $2^k I=I(x_0, 2^{k}\cdot 4|x-x_0|)$ for any $k\in \mathbb N.$
 Hence
  \begin{align*}
\left\VertT_N f_2(x)-T_N f_2(x_0)\right\Vert_{l^\infty(\mathbb Z^2)} \le C\left\Vertf\right\Vert_{BMO(\mathbb R_+, dm_\lambda)},
\end{align*}
and therefore $ T^*f(x) = \left\VertT_N f(x)\right\Vert_{l^\infty(\mathbb Z^2)}  \le C < \infty.$

First, we shall see that if $\varphi$ is a test function, then $T_N \varphi(x)$ converges for all $x\in {\mathbb R}_+$.  In order to prove this, it is enough to see that for any  $(L,M)$ with $0<L<M$,  the  series \begin{equation*} A= \sum_{j=L}^M v_j ( {\mathcal P}_{a_{j+1}} \varphi(x) - {\mathcal P}_{a_j} \varphi(x))\quad   \hbox{  and  }\quad  B= \sum_{j=-M}^{-L} v_j ( {\mathcal P}_{a_{j+1}} \varphi(x) - {\mathcal P}_{a_j} \varphi(x))
 \end{equation*}
converge to zero, when $L, M\rightarrow +\infty$.
By the mean value theorem and the $\rho$-lacunarity  of the sequence $\{a_j\}_{j\in \mathbb Z}$, we have
\begin{align*}
|A|   & \le   C_{ v}  { \int_{{\mathbb R}_+}\sum_{j=L}^{M}   \left\vert{\mathcal P}_{a_{j+1}}(x,y)-{\mathcal P}_{a_{j}}(x,y)\right\vert |\varphi(y)|dm_\lambda(y)}\\
&\le C_{ v, \rho, \lambda}   \int_{{\mathbb R}_+} \sum_{j=L}^{M}  \frac{C}{a^{2\lambda+1}_j} |\varphi(y)|dm_\lambda(y) \le  C_{ v, \rho, \lambda}\left({1\over a_L^{2\lambda+1}}\sum_{j=L}^{M} \frac{a_L^{2\lambda+1}}{a_j^{2\lambda+1}}\right) \int_{{\mathbb R}_+} |\varphi(y)|dm_\lambda(y)\\
&\le C_{ v, \rho, \lambda} {\rho^{{2\lambda+1}}\over {\rho^{{2\lambda+1}}-1}}\|\varphi\|_{L^1({\mathbb R}_+, dm_\lambda)}{1\over a_L^{{2\lambda+1}}} \longrightarrow 0, \quad \hbox{as}\ { L,M \to +\infty}.
\end{align*}
On the other hand,  as the integral of the kernels are zero, we can write
\begin{align*}
B&= \int_{{\mathbb R}_+} \sum_{j=-M}^{-L}v_j \left({\mathcal P}_{a_{j+1}}(x,y)- {\mathcal P}_{a_{j}}(x,y) \right)( \varphi(y)- \varphi(x))dm_\lambda(y)\\
&= \sum_{j=-M}^{-L}\Big(\int_{0}^{\sqrt a_j}v_j \left({\mathcal P}_{a_{j+1}}(x,y)- {\mathcal P}_{a_{j}}(x,y) \right)( \varphi(y)- \varphi(x))dm_\lambda(y)\\
&\quad \quad +\int^{+\infty}_{\sqrt a_j}v_j \left({\mathcal P}_{a_{j+1}}(x,y)- {\mathcal P}_{a_{j}}(x,y) \right)( \varphi(y)- \varphi(x))dm_\lambda(y)\Big)\\
&=:B_1+B_2.
\end{align*}
And, then proceeding as in the case $A$, and by using the fact that $\varphi$ is a test function,  we have
\begin{align*}
|B_1|&\le C_{\lambda, \rho} \left\Vertv\right\Vert_{\ell^\infty(\mathbb Z)} \sum_{j=-M}^{-L} \int_{0}^{\sqrt a_j} { a_j\over (|x-y|^2+a_j^2)^{\lambda+1} }\left\vert \varphi(y)- \varphi(x)\right\vert dm_\lambda(y)\\
&\le C_{\lambda, \rho} \left\Vert\nabla\varphi\right\Vert_{L^\infty({\mathbb R}_+, dm_\lambda)}\sum_{j=-M}^{-L} \int_{0}^{\sqrt a_j} { a_j|x-y|\over (|x-y|^2+a_j^2)^{\lambda+1} }dm_\lambda(y)\\
&\le C_{ \lambda, \rho, \varphi} a_{-L}^{1/2}\sum_{j=-M}^{-L} {a_j^{1/2}\over  a_{-L}^{1/2}}\\
&\le C_{ \lambda, \rho, \varphi} {\sqrt \rho\over \sqrt \rho-1} a_{-L}^{1/2}\longrightarrow 0, \quad \hbox{as}\ { L,M \to +\infty}.
\end{align*}
On the other hand,
\begin{align*}
|B_2|&\le C_{\lambda, \rho} \left\Vertv\right\Vert_{\ell^\infty(\mathbb Z)} \sum_{j=-M}^{-L} \int_{\sqrt a_j}^{+\infty} { a_j\over (|x-y|^2+a_j^2)^{\lambda+1} }\left\vert \varphi(y)- \varphi(x)\right\vert dm_\lambda(y)\\
&\le C_{\lambda, \rho} \left\Vert\varphi\right\Vert_{L^\infty({\mathbb R}_+, dm_\lambda)}\sum_{j=-M}^{-L} \int_{\sqrt a_j}^{+\infty} { a_j \over (|x-y|^2+a_j^2)^{\lambda+1} }dm_\lambda(y)\\
&\le C_{ \lambda, \rho, \varphi} a_{-L}^{1/2}\sum_{j=-M}^{-L} {a_j^{1/2}\over  a_{-L}^{1/2}}\\
&\le C_{ \lambda, \rho, \varphi} {\sqrt \rho\over \sqrt \rho-1} a_{-L}^{1/2}\longrightarrow 0, \quad \hbox{as}\ { L,M \to +\infty}.
\end{align*}

As the set of test functions is dense in $L^p(\mathbb{R}_+, dm_\lambda)$, by Theorem \ref{Thm:PoissonLp} we get the $a.e.$ convergence for any function in $L^p(\mathbb{R}_+, dm_\lambda)$. Analogously, since $L^p(\mathbb{R}_+) \cap L^p(\mathbb{R}_+, dm_\lambda)) $ is dense in $L^p(\mathbb{R}_+, \omega dm_\lambda)$, we get the $a.e.$ convergence for functions in $L^p(\mathbb{R}_+, \omega dm_\lambda) $ with $1\le p<\infty$. By using the dominated convergence theorem,  we can prove the convergence in $L^p(\mathbb{R}_+, \omega dm_\lambda)$ norm for $1<p<\infty$, and also in measure.

We will prove it only in the case $1<p<\infty$. For the cases $p=1$ and  $p=\infty$, the proof follows by introducing the obvious changes.  Since $2r<1,$ we know that $\tilde I\backslash \tilde I_{2r}\neq \emptyset.$ Let $f=f_1+f_2$, where $f_1=f\chi_{\tilde I_{2r}}$ and $f_2=f(x)\chi_{\tilde I\backslash \tilde I_{2r}}$. Then
$$\left\vertT^{*} f(x)\right\vert\le \left\vertT^{*} f_1(x)\right\vert+\left\vertT^{*} f_2(x)\right\vert.$$
By Theorem \ref{Thm:PoissonLp}, we have
\begin{multline*}               
\frac{1}{m_\lambda(\tilde I_r)} \int_{\tilde I_r} \left\vertT^{*} f_1 (x)\right\vert dm_\lambda(x)\le \left(\frac{1}{m_\lambda(\tilde I_r)} \int_{\tilde I_r} \left\vertT^{*} f_1 (x)\right\vert^2 dm_\lambda(x)\right)^{1/2}
\\\le C \left(\frac{1}{m_\lambda(\tilde I_r)} \int_{\mathbb{R}_+}\left\vert f_1 (x)\right\vert^2 dm_\lambda(x)\right)^{1/2}\le C\left\Vertf\right\Vert_{L^\infty(\mathbb{R}_+, dm_\lambda)}.
\end{multline*}

We should note that, for any $j\in \mathbb Z,$ by Lemma \ref{KerPoisson},
 \begin{multline}\label{equ:constant}
 \int_{{\mathbb R}_+}\left\vert{\mathcal P}_{a_{j+1}}(x,y) -{\mathcal P}_{a_{j}}(x,y) \right\vert~ dm_\lambda(y)=\int_{{\mathbb R}_+}\left\vert\int_{a_j}^{a_{j+1}}\partial_t{\mathcal P}_{t}(x,y) dt \right\vert~ dm_\lambda(y) \\
 \le C_\lambda\int_{{\mathbb R}_+} {a_{j+1}-a_j\over {(|x-y|^2+a_j^2)^{\lambda+1}}} dm_\lambda(y)  =C_{\lambda, \rho}.
 \end{multline}
And for $1< p < \infty$ and any $N=(N_1, N_2)$, by H\"older's inequality, \eqref{equ:constant}, Fubini's Theorem and Lemma  \ref{KerPoisson} we have
\begin{align*}
&\left\vert\sum_{j=N_1}^{N_2}v_j\left({\mathcal P}_{a_{j+1}}f_2(x)- {\mathcal P}_{a_j}f_2(x)\right)\right\vert\nonumber\\
&\le C\sum_{j=N_1}^{N_2} \left\vertv_j \int_{{\mathbb R}_+}\left({\mathcal P}_{a_{j+1}}(x,y) -{\mathcal P}_{a_{j}}(x,y)\right)  f_2(y)~ dm_\lambda(y)\right\vert\nonumber\\
&\le C\left\Vertv\right\Vert_{l^p(\mathbb Z)}\left( \sum_{j=N_1}^{N_2}\left(\int_{{\mathbb R}_+}\left\vert{\mathcal P}_{a_{j+1}}(x,y) -{\mathcal P}_{a_{j}}(x,y)\right\vert \left\vertf_2(y)\right\vert~ dm_\lambda(y)\right)^{p'}\right)^{1/p'} \nonumber \\
&\le C_v\Big(\sum_{j=N_1}^{N_2} \Big\{\int_{{\mathbb R}_+}\left\vert{\mathcal P}_{a_{j+1}}(x,y) -{\mathcal P}_{a_{j}}(x,y) \right\vert \left\vertf_2(y)\right\vert^{p'}~ dm_\lambda(y)\Big\}  \\
 & \quad \quad  \times  \Big\{\int_{{\mathbb R}_+}\left\vert{\mathcal P}_{a_{j+1}}(x,y) -{\mathcal P}_{a_{j}}(x,y) \right\vert~ dm_\lambda(y)  \Big\}^{p'/p} \Big)^{1/p'}
\nonumber \\
&\le C_{v,p, \lambda, \rho}\left(\sum_{j=N_1}^{N_2} \int_{{\mathbb R}_+}\left\vert{\mathcal P}_{a_{j+1}}(x,y) -{\mathcal P}_{a_{j}}(x,y) \right\vert \left\vertf_2(y)\right\vert^{p'}~ dm_\lambda(y)   \right)^{1/p'}
\nonumber \\
&\le C_{v,p, \lambda, \rho} \left(\int_{{\mathbb R}_+}\sum_{j=-\infty}^{+\infty} \left\vert{\mathcal P}_{a_{j+1}}(x,y) -{\mathcal P}_{a_{j}}(x,y)\right\vert \left\vertf_2(y)\right\vert^{p'}~ dm_\lambda(y)  \right)^{1/p'}
\nonumber \\ \nonumber
&\le C_{v,p, \lambda, \rho}\left(\int_{{\mathbb R}_+} \frac1{|x-y|^{2\lambda+1}} \left\vertf_2(y)\right\vert^{p'}~ dm_\lambda(y)  \right)^{1/p'}.
\end{align*}
For $y\in \tilde I\backslash \tilde I_{2r}$ and $x\in \tilde I_r$,  we have  $r\le |x-y|\le 2$. Then,  we get
\begin{align*}
&\frac{1}{m_\lambda(\tilde I_r)} \int_{\tilde I_r} \left\vertT^{*} f_2(x)\right\vert dm_\lambda(x) \\&\le
C_{v,p, \lambda, \rho}\frac{1}{m_\lambda(\tilde I_r)} \int_{\tilde I_r} \left(\int_{{\mathbb R}_+} \frac1{|x-y|^n} \left\vertf_2(y)\right\vert^{p'}~ dm_\lambda(y)  \right)^{1/p'}dm_\lambda(x) \\
&\le C_{v,p, \lambda, \rho}\frac{\left\Vertf\right\Vert_{L^\infty({\mathbb R}_+, dm_\lambda)}}{m_\lambda(\tilde I_r)} \int_{\tilde I_r} \left(\int_{r\le |x-y| \le 2} \frac1{|x-y|^{2\lambda+1}} ~ dm_\lambda(y) \right)^{1/p'}dm_\lambda(x) \\
 &\sim \Big(\log\frac{2}{r}\Big)^{1/p'}\left\Vertf\right\Vert_{L^\infty({\mathbb R}_+, dm_\lambda)}.
\end{align*}
Hence, $$\frac{1}{m_\lambda(\tilde I_r)} \int_{\tilde I_r} \left\vertT^{*} f(x)\right\vert dm_\lambda\le C\left(1+\Big(\log\frac{2}{r}\Big)^{1/p'}\right)\left\Vertf\right\Vert_{L^\infty({\mathbb R}_+, dm_\lambda)}  \le C\Big(\log\frac{2}{r}\Big)^{1/p'}\left\Vertf\right\Vert_{L^\infty({\mathbb R}_+, dm_\lambda)}.$$
