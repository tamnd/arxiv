\label{rmk: subrepresentations}
Although the space~$\mathscr{F}{\!\!\textnormal{\footnotesize\itshape ock}}$ is not an irreducible representation of $\textnormal{\textbf{SF}} \oplus \overline{\textnormal{\textbf{SF}}}$, it is cyclically generated by the action of the generators of $\textnormal{\textbf{SF}} \oplus \overline{\textnormal{\textbf{SF}}}$ on the vector~$\boldsymbol{\omega}$.
From the explicit basis~\eqref{eq: fock basis} and the anticommutation  relations ---Lie superbrackets--- of the generators,
one can argue that the only proper subrepresentations of $\mathscr{F}{\!\!\textnormal{\footnotesize\itshape ock}}$ are the ones cyclically generated by the other ground states ---$\boldsymbol{\mathbbm{1}}$, $\boldsymbol{\xi}$, and $\boldsymbol{\theta}$---.

\label{thm: characterization of CF}
For each complex number $\alpha\in{\mathbb C}$, there exists a unique collection
of correlation functions with the following properties:
\begin{description}[style=sameline]

                                      
	\item[(FER)]
	The correlation functions of the ground fermions are
	\begin{align*}
		\Big\langle \boldsymbol{\xi}(z_1)\,\boldsymbol{\theta}(w_1)\cdots\boldsymbol{\xi}(z_n)\,\boldsymbol{\theta}(w_n) \Big\rangle_{\Omega; \alpha}^{\textnormal{SyFe}}
		\, = \,
		(4\pi)^n
		\displaystyle\sum_{\sigma\in\mathcal{S}_n} \textnormal{sgn}(\sigma)
		\displaystyle\prod_{i=1}^n
		\mathbf{G}_\Omega\big(z_i,w_{\sigma(i)}\big)
	\end{align*}
	for $(z_1,\ldots,z_n,w_1\ldots,w_n)\in\textnormal{Conf}_{2n}( \Omega )$,
	they vanish when the number of
	\linebreak[4]
	$\boldsymbol{\xi}$-inser\-tions and $\boldsymbol{\theta}$-insertions do not coincide,
	and they satisfy
	\begin{align*}
		\Big\langle \cdots\boldsymbol{\xi}(z)\,\boldsymbol{\theta}(w)\cdots \Big\rangle_{\Omega; \alpha}^{\textnormal{SyFe}}
		\,=
		-\,\Big\langle \cdots\boldsymbol{\theta}(w)\,\boldsymbol{\xi}(z)\cdots \Big\rangle_{\Omega; \alpha}^{\textnormal{SyFe}}\,.
	\end{align*}
	
	
                                        
	\item[(DER)]
	For the Virasoro modes~${\mathsf{L}}_{-1}$ and $ 
    \mathrlap{\hspace{-0.3pt}\overline{\scalebox{0.6}[1]{\phantom{\ensuremath{\mathsf{L}}}}}}\ensuremath{\mathsf{L}}
_{-1}$ we have
	\begin{align*}
		\Big\langle \big[{\mathsf{L}}_{-1}\boldsymbol{\varphi}\big](z)\cdots \Big\rangle_{\Omega; \alpha}^{\textnormal{SyFe}}
		\  = \ 
		\partial_z\,\Big\langle \boldsymbol{\varphi}(z)\cdots \Big\rangle_{\Omega; \alpha}^{\textnormal{SyFe}}

	\begin{align*}

_z\,\Big\langle \boldsymbol{\varphi}(z)\cdots \Big\rangle_{\Omega; \alpha}^{\textnormal{SyFe}}\,,
	\end{align*}
	for any field~$\boldsymbol{\varphi}\in\mathscr{F}{\!\!\textnormal{\footnotesize\itshape ock}}$.

                               
\item[(CUR)]                       
Within correlation functions, we have
\begin{align*}
                   
    \boldsymbol{\chi}(z)\boldsymbol{\varphi}(w)
    \,=\,
    \sum_{k\in{\mathbb Z}}\,
    \frac{\,\big[{\upchi}_{k}\boldsymbol{\varphi}\big](w)\,}{\,(z-w)^{k+1}}\,,
    \mspace{20mu}
    &
    \mspace{59mu}
    \overline{\boldsymbol{\chi}}(z)\boldsymbol{\varphi}(w)
    \,=\,
    \sum_{k\in{\mathbb Z}}\,
    \frac{\,\big[ 
    \mathrlap{\hspace{-0.6pt}\overline{\scalebox{1}[1]{\phantom{\ensuremath{\upchi}}}}}\ensuremath{\upchi}
_{k}\boldsymbol{\varphi}\big](w)\,}{\,(\overline{z}-\overline{w})^{k+1}}\,,
    \\
    \phantom{.}
    \\
    \boldsymbol{\eta}(z)\boldsymbol{\varphi}(w)
    \,=\,
    \sum_{k\in{\mathbb Z}}\,
    \frac{\,\big[{\upeta}_{k}\boldsymbol{\varphi}\big](w)\,}{\,(z-w)^{k+1}}\,,
    \mspace{20mu}
    &
    \text{ and }
    \mspace{20mu}

    \,+\,

    \,

as $\vert z - w \vert \to 0$.
\end{description}

where $\mathbf{g}_\Omega$ is the harmonic part of the Green's function~$\mathbf{G}_\Omega$.

\label{rmk: general wick formula}
The correlation functions of the ground states $\boldsymbol{\xi}$, $\boldsymbol{\theta}$, and $\boldsymbol{\omega}$ satisfy the following formula:
Take three non-intersecting collections $\underline{z}  \coloneqq\{z_1,\ldots,z_n\}$, $\underline{w}  \coloneqq\{w_1,\ldots,w_n\}$ and $\underline{x}  \coloneqq\{x_1,\ldots,x_k\}$ of distinct points in the domain~$\Omega$.
Let ${\mathfrak{B}(\underline{ z}  ,\underline{ w}  ;\underline{ x}  )}$ denote the set of bi\-jec\-tions from the set~${\underline{ z}  \cup\underline{ x}  }$ onto the set~${\underline{w}  \cup\underline{ x}  }$.
We have
\begin{align*}
    \Big\langle \boldsymbol{\xi}(z_1)\boldsymbol{\theta}(w_1)&\cdots\boldsymbol{\xi}(z_n)\boldsymbol{\theta}(w_n)
    \boldsymbol{\omega}(x_k)\cdots\boldsymbol{\omega}(x_1) \Big\rangle_{\Omega; \alpha}^{\textnormal{SyFe}}
    \\
    & = \phantom{\bigg\vert}
    (-1)^k\sum_{b\in\mathfrak{B}(\underline{z}  ,\underline{w}  ;\underline{x}  )}
    (-1)^b(-1)^{F_b}
    \prod_{\underset{b(y)\neq y}{y\in\underline{z}  \cup\underline{x}  }}
    \Big\langle \boldsymbol{\xi}(y)\boldsymbol{\theta}\big(b(y)\big) \Big\rangle_{\Omega; \alpha}^{\textnormal{SyFe}}
    \prod_{\underset{b(x)= x}{x\in\underline{x}  }}
    \Big\langle \boldsymbol{\omega}(x) \Big\rangle_{\Omega; \alpha}^{\textnormal{SyFe}}\,,
\end{align*}
where $(-1)^b$ is the signature of the bijection $b$, and $F_b$ is the number of fixed points of $b$.
\hfill$\diamond$

\label{rmk: factorisation laplacian}
The discrete Laplacian admits the following factorisations in terms of discrete differential operators:
\begin{align*}
       \partial_\sharp 
    \mathrlap{\hspace{1pt}\overline{\scalebox{0.8}[1]{\phantom{\ensuremath{\partial}}}}}\ensuremath{\partial}

    \mathrlap{\hspace{1pt}\overline{\scalebox{0.8}[1]{\phantom{\ensuremath{\partial}}}}}\ensuremath{\partial}

\end{align*}
We will use extensively these factorisations in our computations in later sections.

\label{prop: monomials}
There exists a unique family of ${\mathbb C}$-valued functions $\{\mathbf{u}\mapsto \mathbf{u}^{[n]}\}_{n\in{\mathbb Z}}$ on
${\mathbb Z}^2_\diamond\cup{\mathbb Z}^2_\textnormal{m}$ that satisfies the following properties:
\begin{enumerate}
\item For all $n\in{\mathbb Z}$, the function $\mathbf{u} \mapsto \mathbf{u}^{[n]}$ has the same square-grid symmetries
as the Laurent monomial $z \mapsto z^n$,
i.e., $(\mathbbm{i} \mathbf{u})^{[n]}=\mathbbm{i}^n \mathbf{u}^{[n]}$ and $\overline{\mathbf{u}}^{[n]} = \overline{\mathbf{u}^{[n]}}$ for all
$\mathbf{u} \in {\mathbb Z}^2_\diamond\cup{\mathbb Z}^2_\textnormal{m}$.
\item For all $\mathbf{u}\in{\mathbb Z}^2_\diamond\cup{\mathbb Z}^2_\textnormal{m}$, $\mathbf{u}^{[0]}=1$ and, for all $n\in{\mathbb Z}$, $\partial_\sharp \mathbf{u}^{[n]} = n\,\mathbf{u}^{[n-1]}$.
\item For each $\mathbf{u}\in{\mathbb Z}^2_\diamond\cup{\mathbb Z}^2_\textnormal{m}$, there exists an $N\in{\mathbb N}$ such that $\mathbf{u}^{[n]}=0$ for all $n\geq N$.
\item For $n<0$, we have $\mathbf{u}^{[n]} \to 0$ as $\vert \mathbf{u}\vert\rightarrow \infty$.
\item The first negative-power monomial has the following explicit
failure of discrete holomorphicity near the origin
\begin{align*}
    \frac{1}{2\pi} 
    \mathrlap{\hspace{1pt}\overline{\scalebox{0.8}[1]{\phantom{\ensuremath{\partial}}}}}\ensuremath{\partial}
_\sharp \mathbf{u}^{[-1]}
    =
    \frac{1}{2} \, \delta_{\mathbf{u},0}
    +
    \frac{1}{4}
    \sum_{\vert\mathbf{v}\vert = \frac{1}{2}} 
    \delta_{\mathbf{u},\mathbf{v}}
    +
    \frac{1}{8}\sum_{\vert\mathbf{v}\vert = \frac{1}{\sqrt{2}}}
    \mspace{-5mu}
    \delta_{\mathbf{u},\mathbf{v}}\,.
\end{align*}
\item \label{property: poles}
For any $n\geq 0$ and all $\mathbf{u}\in{\mathbb Z}^2_\diamond\cup{\mathbb Z}^2_\textnormal{m}$ we have $ 
    \mathrlap{\hspace{1pt}\overline{\scalebox{0.8}[1]{\phantom{\ensuremath{\partial}}}}}\ensuremath{\partial}
_\sharp \mathbf{u}^{[n]}=0$. 
For any $n<0$, we have $ 
    \mathrlap{\hspace{1pt}\overline{\scalebox{0.8}[1]{\phantom{\ensuremath{\partial}}}}}\ensuremath{\partial}
_\sharp \mathbf{u}^{[n]}=0$ except at finitely many
points~$\mathbf{u} \in {\mathbb Z}^2_\diamond\cup{\mathbb Z}^2_\textnormal{m}$.
                                                                                                                                    
\item For any $n,m\in{\mathbb Z}$, we have the discrete residue formula
\begin{align*}
    \frac{1}{2\pi\mathbbm{i}}
    \sqint^{\sharp}_{\gamma} \mathbf{u}^{[n]}_\diamond \, \mathbf{u}^{[m]}_\textnormal{m} \, \textnormal{d}^{\sharp} \mathbf{u} 
    \; = \; \delta_{n+m+1}\,,
\end{align*}
for any large enough positively-oriented corner contour $\gamma$ that encircles the origin.
\item For any $n \in {\mathbb Z}$, as~${|\mathbf{u}| \to \infty}$, the discrete monomial has the asymptotics
\begin{align*}
\mathbf{u}^{[n]} = \mathbf{u}^n + o(|\mathbf{u}|^n) \,,
\end{align*}
where in $\mathbf{u}^n$ we interpret $\mathbf{u}\in{\mathbb C}$ via the embedding  ${\mathbb Z}^2 \subset {\mathbb C}$.
\end{enumerate}

\label{rmk: 2n pt function fDGFF}
The correlation function of the monomial~$\xi(\mathbf{z}_1)\cdots\xi(\mathbf{z}_n)\theta(\mathbf{w}_1)\cdots\theta(\mathbf{w}_m)$ vani\-shes unless $n=m$.
For the two-point function, we have
\begin{align*}
    \Big\langle \xi(\mathbf{z})\theta(\mathbf{w}) \Big\rangle_{\Omega^{\delta}}^{\textnormal{fDGFF}}
    \;=\;
    4\pi\,
    \mathbf{G}_{\Omega^{\delta}}(\mathbf{z},\mathbf{w})\,,
\end{align*}
where $\mathbf{G}_{\Omega^{\delta}}$ is ---up to a negative sign--- the inverse of the Dirichlet Laplacian, that is, it satisfies
\begin{align*}
    \big(\Delta_{\Omega^{\delta}}^{\texttt{D}}\big)_{(1)}
    \mathbf{G}_{\Omega^{\delta}} (\mathbf{z},\mathbf{w})
    \,=\,
    -\delta_{\mathbf{z},\mathbf{w}}
\end{align*}
for $\mathbf{z},\mathbf{w}\in\textnormal{int}\,{\Omega^{\delta}}$, where the subindex $(\,\cdot\,)_{(1)}$ indicates the Laplacian operator is taken with respect to the first variable of $\mathbf{G}_{\Omega^{\delta}}$.
Finally, the $2n$-point functions satisfy the \textbf{fermionic Wick's formula}, i.e.~we have
\begin{align*}
    \Big\langle \xi(\mathbf{z}_1)\theta(\mathbf{w}_1)\cdots\xi(\mathbf{z}_n)\theta(\mathbf{w}_n) \Big\rangle_{\Omega^{\delta}}^{\textnormal{fDGFF}}
    \;=\;
    \sum_{\sigma\in\mathfrak{S}_n}
    (-1)^\sigma
    \prod_{i=1}^n
    \Big\langle \xi(\mathbf{z}_i)\theta(\mathbf{w}_{\sigma(i)}) \Big\rangle_{\Omega^{\delta}}^{\textnormal{fDGFF}}
\end{align*}
where $\mathfrak{S}_n$ denotes the symmetric group of order~$n$ and $(-1)^\sigma$ is the signature of the permutation~$\sigma$.
\hfill$\diamond$

<!-- rmk-9: body_md -->
\label{rmk: arbitrary scale}
In each discrete approximation $\Omega^{\delta}$ of the continuum domain~$\Omega$, the linear scale of the system is proportional to $\delta$; the proportionality constant $\lambda$ is arbitrary.
There is no good reason to pick as a linear scale the edge length $\delta$ instead of the diagonal length of the square-grid plaquettes $\sqrt{2}\,\delta$ or half the length of an edge $\delta/2$.
\hfill$\diamond$
<!-- end rmk-9 -->
<!-- rmk-10: body_md -->
This arbitrariness is \emph{not} relevant in non-logarithmic theories---a change in the linear scale can be absorbed as a multiplicative factor in the local observables.
This is also the case for local fields in our theory with well-defined scaling dimensions.
However, our theory contains more exotic fields that feature logarithmic divergencies in the scaling limit.
These are the generalised eigenvectors of ${\mathbf{L}}_{0}+ 
    \mathrlap{\hspace{-1pt}\overline{\scalebox{0.85}[1]{\phantom{\ensuremath{\mathbf{L}}}}}}\ensuremath{\mathbf{L}}
_{0}$ that are not eigenvectors.

Let us focus on the integrand that is an fDGFF correlation function.
By the fDGFF generalised Wick's formula ---Remark~\ref{rmk: discrete Wick with log corrections}---, and by linearity of the fDGFF correlation functions, it can be rewritten as sums of products of factors of the two following types:
\begin{align*}
    \sqint^{\sharp}_{\gamma^\eta_{i;a}(\delta)}
    \!\!\!\!
    \frac{\textnormal{d}^{\sharp} \mathbf{u}_{i;a}}{2\pi\mathbbm{i}} \,
    (\mathbf{u}_{i;a})_\diamond^{[-k_a^{(i)}]}
    \sqint^{\sharp}_{\gamma^\chi_{j;b}(\delta)} \,
    \!\!\!\!
    \frac{\textnormal{d}^{\sharp} \mathbf{v}_{j;b}}{2\pi\mathbbm{i}}
    (\mathbf{v}_{i;b})_\diamond^{[-\ell_b^{(j)}]}
    \big(\partial^\bullet_\sharp\big)_{\!\mathbf{u}}
    \big(\partial^\bullet_\sharp\big)_{\!\mathbf{v}}
    \Big\langle 
    \theta ( \mathbf{z}_i^\delta+\delta(\mathbf{u}_{i;a})_\textnormal{m} )
    \xi ( \mathbf{z}_j^\delta+\delta(\mathbf{v}_{j;b})_\textnormal{m} )
     \Big\rangle_{\Omega^{\delta}}^{\textnormal{fDGFF}}
\end{align*}
and
\begin{align*}
    - \Big\langle 
    \xi \big( \mathbf{z}_i^\delta \big)
    \theta \big( \mathbf{z}_i^\delta \big)
     \Big\rangle_{\Omega^{\delta}}^{\textnormal{fDGFF}}
    - 2\log\big(\lambda\delta\big)\,.
\end{align*}
The latter type of factor, converges to 
\begin{align*}
    \Big\langle \boldsymbol{\omega}(x) \Big\rangle_{\Omega; \alpha(\lambda)}^{\textnormal{SyFe}}
\end{align*}
uniformly for $x$ on compacts of $\Omega$ by Corollary~\ref{coro: converg of partner}.
The former type, when renormalised by $\delta^{-(k_a^{(i)}+\ell_b^{(j)})}$, converges uniformly for $(z_i,z_j)$ on compacts of $\textnormal{Conf}_{2}( \Omega )$ to
\begin{align*}
    \oint_{\Gamma^\eta_{i;a}}
    \!\!
    \frac{\textnormal{d} \zeta^{\eta}_{i;a}}{2\pi\mathbbm{i}} \,
    \big( \zeta^{\eta}_{i;a} \big)^{-k_a^{(i)}}
    \sqint^{\sharp}_{\Gamma^\chi_{j;b}} \,
    \!\!
    \frac{\textnormal{d} \zeta^{\chi}_{j;b}}{2\pi\mathbbm{i}} \,
    \big( \zeta^{\chi}_{j;b} \big)^{-\ell_b^{(j)}}
    \partial_{\zeta^{\eta}_{i;a}}
    \partial_{\zeta^{\chi}_{j;b}}
    \Big\langle 
    \boldsymbol{\theta} \big( z_i+\zeta^{\eta}_{i;a} \big)\,
    \boldsymbol{\xi} \big( z_j+\zeta^{\chi}_{j;b} \big)
     \Big\rangle_{\Omega; \alpha(\lambda)}^{\textnormal{SyFe}}\,,
\end{align*}
where we took $\alpha = \alpha(\lambda)$ since the CFT correlation does not depend on this constant.
To see this, recall that for $\mathbf{u}^\delta\in\delta{\mathbb Z}^2_\diamond$ satisfying $\delta\mathbf{u}^\delta\to\zeta \in {\mathbb C}$ as $\delta\downarrow0$, we have ---Proposition~\ref{prop: monomials} (Property~8)--- the convergence of the discrete monomials
\begin{align*}
    \delta^k \big(\mathbf{u}^\delta\big)^{[k]}\;
    \xlongrightarrow{\;\;\; \delta\downarrow 0\;\;\;}
    \;\zeta^k
\end{align*}
uniformly for $\zeta$ on compacts of ${\mathbb C}\setminus\{0\}$.
We also need the classical result of the scaling-limit convergence of the double derivatives of the discrete (Dirichlet) Green's function to its continuum counterpart when renormalized by $\delta^2$---see \cite[Section~3.3]{ABK-DGFF_local_fields}.
Then, when each of the discrete differentials $\textnormal{d}^{\sharp}$ is renormalised by $\delta$, the expression $\eqref{eq: fDGFF corr fun}$ becomes a Riemann-sum approximation of double integrals.
Collecting the powers of $\delta$ we find that, if we renormalise~\eqref{eq: fDGFF corr fun} by
\begin{align*}
    {{\mathlarger{\delta}}^{\,-\sum_i\big( \sum_a k_{i;a} + \sum_b \ell_{i;b} \big)}},
\end{align*}
it converges uniformly on compacts to \eqref{eq: CFT corr fun} ---after a shift in the integration variables---.
Note that $\sum_a k_{i;a} + \sum_b \ell_{i;b}$ is the generalised scaling dimension of the field $\boldsymbol{\varphi}_i$.
