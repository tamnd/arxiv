\begin{equation}
\begin{aligned}
\mathcal{H}(\underline{\eta},\xi) &= \frac{{\mathcal D}_1(H,\xi)\sum_{i = 1}^k (\xi^2 + |\vec{\eta}_i|^2)G^{N -

&\leq \frac{{\mathcal D}_1(H,\xi) G(\vec\eta_0)^{N - 1}(k\xi^2 + \|\underline{\eta}_{0,k}\|^2)}{\xi^2 + 
\|\underline{\eta}_{0,k}\|^2}\\
&= \frac{\sum_{i = 1}^k  \widetilde H(\vec{\eta}_0,\xi) G^{N - 1}(\underline{\eta}_{0,k}^i)}{\xi^2 + \|\underline{\eta}_{0,k}\|^2}.
\end{aligned}
\end{equation}
Hence
\[
\mathcal{H}(\underline{\eta},\xi) \leq \mathcal{H}(\underline{\eta}_{0,k},\xi).
\]

$\|\vec{\eta}_{k +1}\|,...,\|\vec{\eta}_N\| < \eta_0$, for some $k$. Set $\underline{\eta}_{*} = (\vec{\eta}_1,...,\vec{\eta}_k,0,...,0),$ and
$\underline{\eta}_{**} = (0,...,0,\vec{\eta}_{k + 1},...,\vec{\eta}_N).$ Then we have
\begin{equation}
\begin{aligned}
\mathcal{H}(\underline{\eta},\xi) &= \frac{\sum_{i = 1}^N \widetilde H(\vec{\eta}_i,\xi)G^{N - 1}(\underline{\eta}^i)}{\xi^2 + \|\underline{\eta}\|^2}\\
&\leq\frac{\sum_{i = 1}^k {\mathcal D}_1(H,\xi)(\xi^2 + \|\vec{\eta}_i^2\|)G^{N - 1}(\underline{\eta}^i) + \sum_{i = k + 1}^N

\frac12\xi^2 +\frac12(\|\vec{\eta}_{k + 1}^2\| + \cdots + \|\vec{\eta}_N^2\|)}\\

\end{aligned}
\end{equation}

\[
\frac{\sum_{i = 1}^k {\mathcal D}_1(H,\xi)(\xi^2 + \|\vec{\eta}_i\|^2)G^{N - 1}
(\underline{\eta}^i)}{\xi^2 + (\|\vec{\eta}_1\|^2 + \cdots + \|\vec{\eta}_k^2)\|} \leq 
\mathcal{H}(\underline{\eta}_{*},\xi) \leq \mathcal{H}(\underline{\eta}_{0,k},\xi).

where we have used again the monotonicity of $G$ for the first inequality. On the other hand, the second term satisfies
\[

\]
Since $\vec\eta_{**}$ has all component of norm less than $\eta_0$, combining these two cases yields the claim.

\[

\]
