<!-- lemma-1: body_md -->
\label{lem:a}
Suppose that a parabolic connection $(E_*, D)$ is 
irreducible.  Then $(E_*, D)$ is stable.
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{lem:1.a} Let 
$(E_*, D)$ and $(E'_*, D')$ be semi-stable full flag parabolic connections over $X$ with same  generic weight $\alpha$.
Then we have the followings. 
\begin{enumerate}
\item \label{a1} Suppose $(E_*, D)$ and $(E'_*, D')$ are stable and $p\mu(E_*) = p\mu(E'_*).$
If $$\phi : (E_*, D) \to (E'_*, D')$$  is a non-zero morphism of parabolic connections, then it is an isomorphism.
\item \label{a2} If $(E_*, D)$ is stable, then the only parabolic endomorphisms  of the pair  $(E_*, D)$ are scalars.
\end{enumerate}
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
\mbox{}
\begin{enumerate}
\item First note that a subbundle and its quotient bundle of a parabolic vector bundle admits a parabolic structure from its original parabolic bundle (see the remark after Definition 1.8 of \cite{MS}). We shall use usual technique of Kernel-Image and Image-Coimage sequences to prove the first part.

Let $\ensuremath{\mathrm{Ker} (\phi)}_* \subset E_*$ is a parabolic subbundle.
Then, $\ensuremath{\mathrm{Ker} (\phi)}_*$ is $D$-invariant parabolic subbundle of $E$. Since $(E_*, D)$ is stable, we have $p\mu(\ensuremath{\mathrm{Ker} (\phi)}_*) < p\mu(E_*)$. Since $\phi \neq 0$, we have $\ensuremath{\mathrm{Im} (\phi)} \neq 0$. Now, $\ensuremath{\mathrm{Im} (\phi)}$ inherits a parabolic structure from $E'_*$.

Consider the Kernel-Image short exact sequence 
$$0 \to \ensuremath{\mathrm{Ker} (\phi)}_* \to E_* \to \ensuremath{\mathrm{Im} (\phi)}_* \to 0.$$
Then, $p\mu(E_*) < p\mu(\ensuremath{\mathrm{Im} (\phi)}_*)$.
Next, consider the Image-Coimage short exact sequence 
$$ 0 \to \ensuremath{\mathrm{Im} (\phi)}_* \to E'_* \to (E' / \ensuremath{\mathrm{Im} (\phi)})_* \to 0.$$
Note that $\ensuremath{\mathrm{Im} (\phi)}_*$ is a $D'$-invariant parabolic subbundle of $E'$. Since $(E'_*, D')$ is stable, we have 
$p\mu(\ensuremath{\mathrm{Im} (\phi)}_*) < p\mu (E'_*)$. Thus, we get that 
$p\mu(E_*) < p\mu(E'_*)$ which contradicts the assumption that
$p\mu(E_*) = p\mu (E'_*)$. Therefore, on underlying vector bundles, we have $\ensuremath{\mathrm{Ker} (\phi)} = 0$ and 
$\ensuremath{\mathrm{Im} (\phi)} = E'$.

\item Let  $\psi : (E_*, D) \to (E_*, D)$ be the given endomorphism. Let $x \in X$, and let $\lambda \in \ensuremath{\mathbb{C}}$ be an eigen value of the linear map $\psi(x) : E_x \to E_x$. Note that $X$ is a compact Riemann surface, it does not admit any non-constant holomorphic function, therefore the eigen values and their multiplicities are independent of $x \in X$.
Since $D$ is $\ensuremath{\mathbb{C}}$-linear, 
$\psi - \lambda \ensuremath{\mathbf{1}_{E_*}}$ is an endomorphism of $(E_*, D)$, that is, 
$$D \circ (\psi - \lambda \ensuremath{\mathbf{1}_{E_*}}) = (\psi - \lambda \ensuremath{\mathbf{1}_{E_*}}) \otimes \ensuremath{\mathbf{1}_{\Omega^1_X(S)}} \circ D.$$
Since $(E_*, D)$ is stable, 
from \eqref{a1}, $\psi - \lambda \ensuremath{\mathbf{1}_{E_*}}$ is either a zero morphism or an isomorphism. 
Since $\lambda$ is an eigen value of $\psi(x)$, kernel of 
$\psi - \lambda \ensuremath{\mathbf{1}_{E_*}}$ is non-trivial. Thus, $\psi = \lambda \ensuremath{\mathbf{1}_{E_*}}$, where $\lambda \in \ensuremath{\mathbb{C}}$.

\end{enumerate}

$\ensuremath{\mathrm{Pic} (\ensuremath{\mathcal{M}}'_{pc}(r, d, \alpha))}$ is isomorphic 
to $\ensuremath{\mathrm{Pic} (\ensuremath{\mathcal{M}}(r, d, \alpha))}$.

 We know that the dual of an affine space is a vector space, therefore  the dual $$\pi^{-1}(E_*)^{\vee} = \{\varphi: \pi^{-1}

 $$r^2(g-1)+\frac{m(r^2-r)}{2} +2,$$ follows from Lemma \ref{lem:a.2}.

 denoted by $H_{(E_*, D)}$.

 induced from $\psi$. 

Next, define a map 
\begin{equation}
\label{eq:a27}
i: \ensuremath{\mathcal{M}}'_{pc}(r, d, \alpha)
  \to \ensuremath{\mathbf{P}}(\ensuremath{\mathcal{W}})
\end{equation} by sending  $(E_*, D)$ to the equivalence class of  $H_{(E_*, D)}$,
  which is an open embedding. Set 
\begin{equation}
\label{eq:a28}
{\bf H} = \ensuremath{\mathbf{P}} (\ensuremath{\mathcal{W}}) \setminus \ensuremath{\mathcal{M}}'_{pc}(r,d,\alpha).
\end{equation}  
  Then $\tilde{\psi}^{-1}(E_*) \cap {\bf H} $ is a projective hyperplane in $\tilde{\psi}^{-1}(E_*)$ for every $E_* \in \ensuremath{\mathcal{M}}(r, d, \alpha)$, and hence ${\bf H}$ is a hyperplane at infinity. 


  
 We first 
show that $\pi^*$ is injective.
Let $\eta \to 
\ensuremath{\mathcal{M}}(r, d, \alpha)$ be a line bundle such that $\pi^* \eta$ is a trivial line bundle
over $\ensuremath{\mathcal{M}}'_{pc}(r, d, \alpha)$. A trivialization of $\pi^* \eta$ is equivalent to have 
 a nowhere vanishing section of $\pi^* \eta$ over $\ensuremath{\mathcal{M}}'_{pc}(r, d, \alpha)$. Fix 
$s \in \ensuremath{\mathrm{H}^{0}(\ensuremath{\mathcal{M}}'_{pc}(r, d, \alpha),\pi^* \eta)}$ a nowhere vanishing section. Choose a
point $E_* \in \ensuremath{\mathcal{M}}(r, d, \alpha)$.  Then, from the following commutative diagram 

\begin{equation}
\label{eq:a29}
\xymatrix{
\pi^* \eta \ar[d] \ar[r]^{\tilde{\pi}} & \eta \ar[d] \\
\ensuremath{\mathcal{M}}'_{pc}(r, d, \alpha) \ar[r]^{\pi} & \ensuremath{\mathcal{M}}(r, d, \alpha)\\
}
\end{equation}
we get 
\begin{equation*}
\label{eq:a30}
s|_{\pi^{-1}(E_*)}: \pi^{-1}(E_*) \to \eta(E_*)
\end{equation*}
 a nowhere vanishing map. Notice that $\pi^{-1}(E_*) \cong \ensuremath{\mathbb{C}}^N$ and $\eta(E_*) \cong \ensuremath{\mathbb{C}}$, where $N = r^2(g-1)+\frac{m(r^2-r)}{2} + 1$. Now, any nowhere vanishing algebraic function on an affine space
$\ensuremath{\mathbb{C}}^N$ is a constant function, that is, $s|_{\pi^{-1}(E_*)}$ is a constant
function and hence corresponds to a non-zero vector $\alpha_{E_*} \in \eta(E_*)$.
Since $s$ is constant on each fiber of $\pi$,  the trivialization $s$ of $\pi^*\eta$ descends to a trivialization of the
line bundle $\eta$ over $\ensuremath{\mathcal{M}}(r, d, \alpha)$, and hence giving a nowhere vanishing
section of $\eta$ over $\ensuremath{\mathcal{M}}(r, d, \alpha)$.  Thus,  $\eta$ is a trivial line bundle
over $\ensuremath{\mathcal{M}}(r, d, \alpha)$.

 It remains to show that $\pi^*$ is surjective. Let 
 $\vartheta \to \ensuremath{\mathcal{M}}'_{pc}(r, d, \alpha)$ be an algebraic line 
 bundle.  Since $\ensuremath{\mathbf{P}} (\ensuremath{\mathcal{W}})$ is a smooth compactification of $\ensuremath{\mathcal{M}}'_{pc}(r, d, \alpha)$ follows from the embedding $i : \ensuremath{\mathcal{M}}'_{pc}(r, d, \alpha) \hookrightarrow
 \ensuremath{\mathbf{P}} (\ensuremath{\mathcal{W}}) $ in \eqref{eq:a27}, the homomorphism of Picard groups 
 $$i^* : \ensuremath{\mathrm{Pic} ( \ensuremath{\mathbf{P}} (\ensuremath{\mathcal{W}}))} \to \ensuremath{\mathrm{Pic} (\ensuremath{\mathcal{M}}'_{pc}(r, d, \alpha))}$$ defined by pull back of line bundles via $i$ in \eqref{eq:a27}, is a surjective homomorphism. Therefore, 
 we can extend $\vartheta$ to a line bundle
 $\vartheta'$ over $\ensuremath{\mathbf{P}} (\ensuremath{\mathcal{W}})$.
 Again from the morphism $\tilde{\psi}: \ensuremath{\mathbf{P}}(\ensuremath{\mathcal{W}}) \to \ensuremath{\mathcal{M}}(r, d, \alpha)$ in \eqref{eq:a26}  and from \cite{H}, Chapter III, Exercise 12.5, p.n. 291,
we have
\begin{equation}
\label{eq:a31}
 \ensuremath{\mathrm{Pic} (\ensuremath{\mathbf{P}} (\ensuremath{\mathcal{W}}))} \cong \tilde{ \psi}^*\ensuremath{\mathrm{Pic} (\ensuremath{\mathcal{M}}(r, d, \alpha))}\oplus  \ensuremath{\mathbb{Z}} \ensuremath{\mathcal{O}_{\ensuremath{\mathbf{P}} (\ensuremath{\mathcal{W}})}}(1).
\end{equation}
Therefore,
\begin{equation}
\label{eq:a32}
\vartheta' = \tilde{\psi}^* \Lambda \otimes \ensuremath{\mathcal{O}_{\ensuremath{\mathbf{P}} (\ensuremath{\mathcal{W}})}}(l)
\end{equation}
where $\Lambda$ is a line bundle over $\ensuremath{\mathcal{M}}$ and 
$l \in \ensuremath{\mathbb{Z}}$.
Since ${\bf H} = \ensuremath{\mathbf{P}} (\ensuremath{\mathcal{W}}) \setminus \ensuremath{\mathcal{M}}'_{pc}(r, d, \alpha)$ is the 
hyperplane at infinity, using \eqref{eq:a31} the line bundle 
$\ensuremath{\mathcal{O}_{\ensuremath{\mathbf{P}} (\ensuremath{\mathcal{W}})}}({\bf H})$ associated to the divisor ${\bf H}$ can be expressed
as 
\begin{equation}
\label{eq:a33}
\ensuremath{\mathcal{O}_{\ensuremath{\mathbf{P}} (\ensuremath{\mathcal{W}})}}({\bf H}) = \tilde{\psi}^* \Gamma \otimes \ensuremath{\mathcal{O}_{\ensuremath{\mathbf{P}} (\ensuremath{\mathcal{W}})}}(1)
\end{equation}
for some line bundle $\Gamma$ over $\ensuremath{\mathcal{M}}(r, d, \alpha)$.
Now, from \eqref{eq:a32} and \eqref{eq:a33}, we get
\begin{equation*}
\label{eq:a34}
\vartheta' = \tilde{\psi}^*(\Lambda \otimes 
(\Gamma^{\vee})^{\otimes l}) \otimes \ensuremath{\mathcal{O}_{\ensuremath{\mathbf{P}} (\ensuremath{\mathcal{W}})}}(l{\bf H}).
\end{equation*}
Since, the restriction of the line bundle $\ensuremath{\mathcal{O}_{\ensuremath{\mathbf{P}} 
(\ensuremath{\mathcal{W}})}}({\bf H})$ to the complement $$\ensuremath{\mathbf{P}} (\ensuremath{\mathcal{W}}) \setminus {\bf H} = 
\ensuremath{\mathcal{M}}'_{pc}(r, d, \alpha)$$ is the trivial line bundle and restriction of $\tilde{\psi}$ to $\ensuremath{\mathcal{M}}'_{pc}(r, d, \alpha)$ is the map $\pi$ defined in \eqref{eq:a21.1}, therefore, we have
\begin{equation*}
\label{eq:a35}
\vartheta =  \pi^*(\Lambda \otimes (\Gamma^{\vee})^{\otimes l}).
\end{equation*}
This completes the proof of the theorem.

Consider the dual of the  Atiyah exact sequence in \eqref{eq:a40}, that is, 
 \begin{equation}
\label{eq:a41}
0 \to T^*\ensuremath{\mathcal{M}} 
\xrightarrow{\sigma^*} 
\ensuremath{\mathrm{At} (L)}^* \xrightarrow{\imath^*}   \ensuremath{\mathcal{O}_{\ensuremath{\mathcal{M}}}} \to 0.
\end{equation}

Consider $\ensuremath{\mathcal{O}_{\ensuremath{\mathcal{M}}}}$ as the trivial line 
bundle $\ensuremath{\mathcal{M}} \times \ensuremath{\mathbb{C}}$. Let $$ \alpha : \ensuremath{\mathcal{M}} \to \ensuremath{\mathcal{M}} \times \ensuremath{\mathbb{C}}$$ be a
holomorphic section of the trivial line bundle defined by $x \mapsto (x,1)$.
Let $Y = \ensuremath{\mathrm{Im} (\alpha)} \subset \ensuremath{\mathcal{M}} \times \ensuremath{\mathbb{C}}$ be the image of $\alpha$. Then 
$$\ensuremath{\mathcal{C}}(L) = {\imath^*}^{-1}Y \subset \ensuremath{\mathrm{At} (L)}^*.$$

In fact, for every open subset $U \subset \ensuremath{\mathcal{M}}
$, a holomorphic section of $\ensuremath{\mathcal{C}}(L)|_{U}$ over $U$ gives a holomorphic
splitting of the Atiyah exact sequence \eqref{eq:a40}, 
associated to the holomorphic vector bundle $L|
_{U} \to U$. 
 For instance, suppose $\tau: U \to \ensuremath{\mathcal{C}}(L)|
_{U}$ is a holomorphic section. Then $\tau$ will be a 
holomorphic section of $\ensuremath{\mathrm{At} (L)}^*|_{U}$  over $U$, 
because $ \ensuremath{\mathcal{C}}(L) =
{\imath^*}^{-1}Y \subset \ensuremath{\mathrm{At} (L)}^*$. Since $$ 
 \tau \circ \imath =  \imath^*(\tau) = \ensuremath{\mathbf{1}_{U}},$$ so we 
 get a holomorphic  splitting $\tau$ of the Atiyah 
 exact sequence \eqref{eq:a40}
 associated to $L|_{U}.$ Thus, $L|_{U}$ admits 
 a holomorphic connection. 
 Conversely, given any splitting of Atiyah exact sequence 
 \eqref{eq:a40} over an open subset $U \subset \ensuremath{\mathcal{M}}$, we get a holomorphic section of $\ensuremath{\mathcal{C}}(L)|_{U}$ over $U$.
 
Let 
 $\ensuremath{\mathbf{P}}(\ensuremath{\mathrm{At} (L)})$ be the projectivization of 
$\ensuremath{\mathrm{At} (L)}$, that is, $\ensuremath{\mathbf{P}}(\ensuremath{\mathrm{At} (L)})$ parametrises hyperplanes in $\ensuremath{\mathrm{At} (L)}$.
Let $\ensuremath{\mathbf{P}}({T \ensuremath{\mathcal{M}}})$ be the projectivization of the tangent bundle $T\ensuremath{\mathcal{M}}$. 
Notice that $\ensuremath{\mathbf{P}}(T\ensuremath{\mathcal{M}})$ is a subvariety of 
$\ensuremath{\mathbf{P}} (\ensuremath{\mathrm{At} (L)})$, and  $\ensuremath{\mathbf{P}}(T\ensuremath{\mathcal{M}})$ is the zero
locus of the of a section of the tautological line bundle
$\ensuremath{\mathcal{O}_{\ensuremath{\mathbf{P}} (\ensuremath{\mathrm{At} (L)})}}(1)$. Now, observe that
$$\ensuremath{\mathcal{C}}(L) = \ensuremath{\mathbf{P}}(\ensuremath{\mathrm{At} (L)}) \setminus \ensuremath{\mathbf{P}}(T \ensuremath{\mathcal{M}}).$$ 
So we have 
\begin{equation}
\label{eq:a42}
\ensuremath{\mathrm{H}^{0}(\ensuremath{\mathcal{C}}(L),\ensuremath{\mathcal{O}_{\ensuremath{\mathcal{C}}(L)}})} =
\varinjlim_{k \geq 0} \ensuremath{\mathrm{H}^{0}(\ensuremath{\mathbf{P}} (\ensuremath{\mathrm{At} (L)}),\ensuremath{\mathcal{O}_{\ensuremath{\mathbf{P}} \ensuremath{\mathrm{At} (L)}}}(k))}.
\end{equation}
 
 
Since for any finite dimensional vector space $V$ over 
$\ensuremath{\mathbb{C}}$ and for every $k \geq 0$, we have 
$\ensuremath{\mathrm{H}^{0}(\ensuremath{\mathbf{P}}(V),\ensuremath{\mathcal{O}_{\ensuremath{\mathbf{P}}(V)}}(k))} = \ensuremath{\mathcal{S}}ym^k(V)$, 
where $\ensuremath{\mathcal{S}}ym^k(V)$ denote the $k$-th symmetric powers of $V$. We get a natural isomorphism

\begin{equation}
\label{eq:a43}
\ensuremath{\mathrm{H}^{0}(\ensuremath{\mathcal{C}}(L),\ensuremath{\mathcal{O}_{\ensuremath{\mathcal{C}}(L)}})} = \varinjlim_{k} \ensuremath{\mathrm{H}^{0}(\ensuremath{\mathcal{M}}, \ensuremath{\mathcal{S}}ym^k\ensuremath{\mathrm{At} (L)})}.

on $k$-th symmetric powers of bundles.
In view of the following composition 
\begin{equation*}
\ensuremath{\mathcal{S}}ym^{k-1} \ensuremath{\mathrm{At} (L)} = \ensuremath{\mathcal{O}_{\ensuremath{\mathcal{M}}}} \otimes \ensuremath{\mathcal{S}}ym^{k-1} \ensuremath{\mathrm{At} (L)} \hookrightarrow

Now, to compute $\ensuremath{\mathrm{H}^{0}(T^*\ensuremath{\mathcal{M}},\ensuremath{\mathcal{O}_{T^*\ensuremath{\mathcal{M}}}})}$
we use the Hitchin fibration for the moduli space of

Let $g: T^*\ensuremath{\mathcal{M}} \to \ensuremath{\mathbb{C}}$ be an algebraic function. Since $T^*\ensuremath{\mathcal{M}}$ is an open subset of $\ensuremath{\mathcal{M}}_{Higgs }(r, d, \alpha)$ such that the complement has codimension  at least $2$ (see \eqref{eq:a22.3}), from

Since $h_P^{-1}(a)$ is an abelian variety,  $\tilde{g}$ is a constant 
function. Therefore, on generic fibre $h^{-1}(a)$, $\tilde{g}$ is constant, and $h_P$ is proper, hence gives an algebraic function on

Set $\ensuremath{\mathcal{P}} = \mbox{d}( \ensuremath{\mathrm{H}^{0}(\ensuremath{\mathcal{H}}_P,\ensuremath{\mathcal{O}_{\ensuremath{\mathcal{H}}_P}})}) \subset  \ensuremath{\mathrm{H}^{0}(\ensuremath{\mathcal{H}}_P,\Omega^1_{\ensuremath{\mathcal{H}}_P})}$ the space of all exact algebraic $1$-form.
Define a map

\end{equation}
by $g \mapsto d \tilde{g}$, where $\tilde{g}$ is the function which is defined by descent of $g$.  Then $\theta$ is an isomorphism.

From \eqref{eq:a52} and \eqref{eq:a53.1}, we have
\begin{equation}
\label{eq:a54.2}

\ensuremath{\mathrm{H}^{0}(\ensuremath{\mathcal{M}},\ensuremath{\mathcal{S}}ym^{k} T \ensuremath{\mathcal{M}})} \to \ensuremath{\mathcal{P}}
\end{equation}
which is an isomorphism.
Let us denote the restriction of $h_P$ on $T^*\ensuremath{\mathcal{M}}$ by  $h'_{P}$.

Let $T_{h'_P} = T_{T^* \ensuremath{\mathcal{M}} / \ensuremath{\mathcal{H}}_P} = \ensuremath{\mathcal{K}er (\mbox{d}h'_P)}$
be the relative tangent sheaf on $T^*\ensuremath{\mathcal{M}}$,
where $$\mbox{d}h'_P: T(T^*\ensuremath{\mathcal{M}}) \to {h'}_P^*T \ensuremath{\mathcal{H}}_P$$ morphism 
of bundles.

 Note that 
$ \ensuremath{\mathrm{H}^{0}(\ensuremath{\mathcal{H}}_P,\Omega^1_{\ensuremath{\mathcal{H}}_P})} \subset  \ensuremath{\mathrm{H}^{0}(T^*\ensuremath{\mathcal{M}},T_{h'_P})} $, and hence from \eqref{eq:a54.2}, we have
an injective homomorphism
\begin{equation*}
\label{eq:a55.1}
\nu: \ensuremath{\mathcal{P}} = \bigoplus_{k \geq 0} \theta(
\ensuremath{\mathrm{H}^{0}(\ensuremath{\mathcal{M}},\ensuremath{\mathcal{S}}ym^{k} T \ensuremath{\mathcal{M}})}) \to \ensuremath{\mathrm{H}^{0}(T^*\ensuremath{\mathcal{M}},T_{h'_P})}.
\end{equation*}

Consider the morphism 

$$\ensuremath{\mathrm{H}^{0}(T^*\ensuremath{\mathcal{M}},T_{h'_P})} \to \ensuremath{\mathrm{H}^{1}(T^*\ensuremath{\mathcal{M}},T_{h'_P} \otimes T^* T^* \ensuremath{\mathcal{M}})}$$ defined 
by taking cup product with the first Chern class $c_1(p^* L) \in  \ensuremath{\mathrm{H}^{1}(T^*\ensuremath{\mathcal{M}}, T^* T^* \ensuremath{\mathcal{M}})}$.

Using the pairing $$T_{h'_P} \otimes T^* T^* \ensuremath{\mathcal{M}} \to \ensuremath{\mathcal{O}_{T^* \ensuremath{\mathcal{M}}}},$$ we get a homomorphism
\begin{equation*}
\label{eq:a56.1}
\eta: \ensuremath{\mathrm{H}^{0}(T^*\ensuremath{\mathcal{M}},T_{h'_P})} \to \ensuremath{\mathrm{H}^{1}(T^*\ensuremath{\mathcal{M}},\ensuremath{\mathcal{O}_{T^*\ensuremath{\mathcal{M}}}})}
\end{equation*}

Since $c_1(p^* L) = p^*c_1 (L)$, we have
\begin{equation}
\label{eq:a57.1}
k \eta \circ \nu \circ \theta(\omega_k) = \tilde{\mu}(\omega_k),
\end{equation}
for all $\omega_k \in \ensuremath{\mathrm{H}^{0}(\ensuremath{\mathcal{M}},\ensuremath{\mathcal{S}}ym^{k} T \ensuremath{\mathcal{M}})})$.
Since $\nu$ and $\theta$ are injective homomorphisms, it is enough to 
show that $\eta|_{\nu(\ensuremath{\mathcal{P}})}$ is injective homomorphism. 


Let $\omega \in \ensuremath{\mathcal{P}} \setminus \{0\}$ be a non-zero
exact $1$-form. Choose $a \in \ensuremath{\mathcal{H}}_P$ such that $\omega(a) \neq 0$. 
In view of \eqref{eq:a22.1} and \eqref{eq:a22.3}, the generic fibre has the form
 $${h'_P}^{-1}(a) = A \setminus F,$$ where $A$ is an abelian variety and $F$ is a subvariety of $A$ such that $\mbox{codim}(F,A) \geq 2$.
Now, $\eta(\nu(\omega)) \in \ensuremath{\mathrm{H}^{1}(T^*\ensuremath{\mathcal{M}},\ensuremath{\mathcal{O}_{T^*\ensuremath{\mathcal{M}}}})}$ and we have restriction map $$\ensuremath{\mathrm{H}^{1}(T^*\ensuremath{\mathcal{M}},\ensuremath{\mathcal{O}_{T^*\ensuremath{\mathcal{M}}}})} \to \ensuremath{\mathrm{H}^{1}({h'_P}^{-1}(a),\ensuremath{\mathcal{O}_{{h'_P}^{-1}(a)}})}.$$
Since $ \omega(a) \neq 0$, $\eta(\nu(\omega)) \in \ensuremath{\mathrm{H}^{1}({h'_P}^{-1}(a),\ensuremath{\mathcal{O}_{{h'_P}^{-1}(a)}})}$.
Because of the following isomorphisms
\begin{equation*}
\label{eq:a58}
 \ensuremath{\mathrm{H}^{1}({h'_P}^{-1}(a),\ensuremath{\mathcal{O}_{{h'_P}^{-1}(a)}})} \cong  \ensuremath{\mathrm{H}^{1}(A,\ensuremath{\mathcal{O}_{A}})} \cong  \ensuremath{\mathrm{H}^{0}(A,TA)},
\end{equation*}
it follows that $\eta(\nu(\omega)) \neq 0$.
This completes the proof.

<!-- proposition-4: body_md -->
\label{prop:4}
Consider the smooth divisor $\bf{H}_0$ defined in Proposition \ref{prop:3}. Then,
the divisor $\bf{H}_0$ is numerically effective if and only if the tangent bundle $T \ensuremath{\mathcal{M}}(r, \alpha, \xi)$ is numerically effective.
<!-- end proposition-4 -->
<!-- proof-7: body_md -->
Let $\ensuremath{\mathcal{N}}_{\ensuremath{\mathbf{P}}(\ensuremath{\mathcal{V}})/\bf{H}_0}$ denote the normal 
bundle of the divisor ${\bf H}_0 \subset \ensuremath{\mathbf{P}} (\ensuremath{\mathcal{V}})$,
where $$\Psi :\ensuremath{\mathcal{V}} \longrightarrow \ensuremath{\mathcal{M}}_{\xi}: = \ensuremath{\mathcal{M}}(r, \alpha, \xi)$$ is the vector bundle (see \eqref{eq:a59}) in Proposition \ref{prop:3}.


Recall that the effective divisor $\bf{H}_0$ is numerically effective if and only if the restriction of the line bundle $\ensuremath{\mathcal{O}_{\ensuremath{\mathbf{P}} (\ensuremath{\mathcal{V}})}}(\bf{H}_0)$ to $\bf{H}_0$ is numerically effective.
From Poincar\'e adjunction formula we have the following 
\begin{equation}
\label{eq:a60}
\ensuremath{\mathcal{O}_{\ensuremath{\mathbf{P}} (\ensuremath{\mathcal{V}})}}(\bf{H}_0) \vert_{\bf{H}_0} \cong \ensuremath{\mathcal{N}}_{\ensuremath{\mathbf{P}}(\ensuremath{\mathcal{V}})/\bf{H}_0}.
\end{equation}

 Therefore, $\bf{H}_0$ is numerically effective if and only if the normal bundle $\ensuremath{\mathcal{N}}_{\ensuremath{\mathbf{P}}(\ensuremath{\mathcal{V}})/\bf{H}_0}$
 is numerically effective. Recall that  the tangent bundle $T \ensuremath{\mathcal{M}}_\xi$ is numerically effective if and only if the tautological line bundle $\ensuremath{\mathcal{O}_{\ensuremath{\mathbf{P}} (T \ensuremath{\mathcal{M}}_\xi)}}(1)$ is numerically 
effective.

Thus, to prove the proposition it is enough to show that 
 the normal bundle  $\ensuremath{\mathcal{N}}_{\ensuremath{\mathbf{P}}(\ensuremath{\mathcal{V}})/\bf{H}_0}$ is canonically isomorphic to the tautological 
line bundle $\ensuremath{\mathcal{O}_{\ensuremath{\mathbf{P}} (T \ensuremath{\mathcal{M}}_\xi)}}(1)$. 
 
First we show that the divisor $\bf{H}_0$ is canonically 
isomorphic to projective bundle $\ensuremath{\mathbf{P}} (T \ensuremath{\mathcal{M}}_\xi)$.
Let 
\begin{equation*}
\label{eq:a61}
\tilde{\Psi} : \ensuremath{\mathbf{P}} (\ensuremath{\mathcal{V}}) \longrightarrow \ensuremath{\mathcal{M}}_{\xi}
\end{equation*}
be the projective bundle. Let $E_* \in \ensuremath{\mathcal{M}}_\xi$, and 
\begin{equation}
\label{eq:a62}
\theta \in \tilde{\Psi}^{-1}(E_*) \cap \bf{H}_0 \subset 
\ensuremath{\mathbf{P}} (\ensuremath{\mathcal{V}}).
\end{equation}
Then, $\theta$ represents a hyperplane in the fibre 
$\ensuremath{\mathcal{V}}(E_*) = \pi_{\xi}^{-1}(E_*)^{\vee}$ of the vector bundle $\ensuremath{\mathcal{V}}$, where $\pi_\xi$ is defined in 
\eqref{eq:a21.1}. Let $H_{\theta}$ denote this hyperplane represented by $\theta$.
Note that $H_{\theta} \subset \pi_{\xi}^{-1}(E_*)^{\vee}$ and $\pi_{\xi}^{-1}(E_*)$ is the affine space modelled 
over the vector space $\ensuremath{\mathrm{H}^{0}(X,\Omega^1_X(S) \otimes \ensuremath{\mathrm{SParEnd'}  (E_*)})},$ therefore
for any $f \in H_{\theta}$, we have 
$$\text{d}f \in (T^*_{E_*} \ensuremath{\mathcal{M}}_\xi)^* = T_{E_*}\ensuremath{\mathcal{M}}_\xi.$$
Since $\theta \in \bf{H}_0$, and $H_\theta$ is a hyperplane, the subspace of $T_{E_*} \ensuremath{\mathcal{M}}_{\xi}$
generated by $\{\text{d}f\}_{f \in H_{\theta}}$ is a
hyperplane in  $T_{E_*} \ensuremath{\mathcal{M}}_{\xi}$. Let $\tilde{\theta} \in \ensuremath{\mathbf{P}} (T_{E_*} \ensuremath{\mathcal{M}}_{\xi})$ denote this hyperplane.
Thus, we get a canonical isomorphism 
\begin{equation}
\label{eq:a63}
{\bf H}_0 \cong \ensuremath{\mathbf{P}} (T \ensuremath{\mathcal{M}}_{\xi})
\end{equation}
by sending $\theta$ to $\tilde{\theta}$.

Next, note that the fibre $\ensuremath{\mathcal{N}}_{\ensuremath{\mathbf{P}}(\ensuremath{\mathcal{V}})/\bf{H}_0}(\theta)$ of the normal bundle $\ensuremath{\mathcal{N}}_{\ensuremath{\mathbf{P}}(\ensuremath{\mathcal{V}})/\bf{H}_0}$ is canonically isomorphic to the quotient 
$\ensuremath{\mathcal{V}}(E_*) / H_{\theta}$. 
Consider the morphism 
$$\ensuremath{\mathcal{V}}(E_*) \longrightarrow T_{E_*}\ensuremath{\mathcal{M}}_{\xi}$$
of vector spaces defined by sending $f \mapsto \text{d} f$.
Since image of the hyperplane $H_{\theta}$ is contained in $\tilde{\theta}$, we have well defined morphism on quotients 
$$\ensuremath{\mathcal{V}}(E_*)/ H_{\theta} \longrightarrow T_{E_*} \ensuremath{\mathcal{M}}_{\xi}/ \tilde{\theta},$$
which is an isomorphism.
Recall that the fibre of the tautological line bundle 
$\ensuremath{\mathcal{O}_{\ensuremath{\mathbf{P}} (T \ensuremath{\mathcal{M}}_{\xi})}}(1)$ at $E_*$ is canonically 
identified with the quotient $T_{E_*} \ensuremath{\mathcal{M}}_{\xi}/ \tilde{\theta}$.
This completes the proof.
