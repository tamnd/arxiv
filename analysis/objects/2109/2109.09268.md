<!-- thm-1: body_md -->
\label{main_thm}Let $I$ be the edge ideal of a simple graph $G$. For all $s \le 4$ and all intermediate ideals $J \in \operatorname{Inter}(I^s,\overline{I^s})$, we have
$$\operatorname{reg} J = \operatorname{reg} \overline{I^s} = \operatorname{reg} (I^s).$$
<!-- end thm-1 -->
<!-- exm-1: body_md -->
Let 
\begin{gather*}
I=(x_1y_1, x_2y_1, x_3y_1, x_7y_1, x_9y_1, x_1y_2, x_2y_2, x_4y_2, x_6y_2, x_{10}y_2,x_1y_3, x_3y_3, x_5y_3,x_6y_3,\\
x_8y_3, x_2y_4, x_4y_4, x_5y_4, x_7y_4, x_8y_4,x_3y_5, x_4y_5, x_5y_5, x_9y_5, x_{10}y_5,x_6y_6,x_7y_6, x_8y_6,x_9y_6,\\  x_{10}y_6,z_1z_2,z_2z_3,z_1z_3,z_4z_5,z_5z_6,z_4z_6) \subseteq k[x_1,\ldots,x_{10},y_1,\ldots,y_6,z_1,\ldots,z_6],
\end{gather*}
be the edge ideal of the disjoint union of two triangles and the graph in the Dalili and Kummini's example \cite{DK}. By Corollary \ref{cor_char_dependence}, for each $s \ge 1$, we have 
$$\operatorname{reg} I^s = \operatorname{reg} \overline{I^s} = \operatorname{reg} I^{(s)} = \begin{cases} 5 + 2s & \text{ when } \operatorname{char} \mathrm{k} = 2 \\
4 + 2s &  \text{ when } \operatorname{char} \mathrm{k} \neq 2. \end{cases}$$
In particular, $\operatorname{reg} I^s, \operatorname{reg} \overline{I^s}, \operatorname{reg} I^{(s)},$ and their asymptotic constants all depend on the characteristic of the base field.
<!-- end exm-1 -->
<!-- defn-1: body_md -->
For a squarefree monomial ideal $I$, the Stanley-Reisner complex of $I$ is defined by
$$ \Delta(I) = \{ F \subset [n] \mid x_F \notin I\}.$$

For a simplicial complex $\Delta$, the Stanley-Reisner ideal of $\Delta$ is defined by
$$I_\Delta = (x_F \mid  F \notin \Delta).$$
The Stanley-Reisner ring of $\Delta$ is $\mathrm{k}[\Delta] =  S/I_\Delta.$
<!-- end defn-1 -->
<!-- lem-1: body_md -->
\label{cone} Let $I, J$ be squarefree monomial ideals of  $S = \mathrm{k} [x_1,\ldots, x_n]$. Then 
\begin{enumerate}
    \item $\Delta(I)$ is a cone over $t \in [n]$ if and only if $x_t$ does not divide any minimal generator of $I$.
    \item $I \subseteq J$ if and only if $\Delta(I) \supseteq \Delta(J)$.
    \item $\Delta(I + J) = \Delta(I) \cap \Delta(J).$
    \item $\Delta(I \cap J) = \Delta(I) \cup \Delta(J).$
\end{enumerate}
<!-- end lem-1 -->
<!-- thm-2: body_md -->
\label{thm_mixed_sum} Let $I \subseteq R$ and $J \subseteq S$ be proper monomial ideals in standard graded polynomial rings over a field $\mathrm{k}$. Let $P$ be the sum of the extensions of $I$ and $J$ to $R \otimes_\mathrm{k} S$. Then
$$\operatorname{reg} P^s = \max_{i\in [1,s-1], j\in [1,s]} \left ( \operatorname{reg} I^i + \operatorname{reg} J^{s-i}, \operatorname{reg} I^j + \operatorname{reg} J^{s-j+1} - 1\right ).$$
In particular, 
$$\operatorname{reg} P^s \ge \operatorname{reg} I^j + \operatorname{reg} J^{s-j+1} - 1, \text{ for } j \in [1,s].$$
<!-- end thm-2 -->
<!-- defn-2: body_md -->
Let $I$ be a monomial ideal of $S$. The exponent set of $I$ is $E(I) = \{\mathbf a \in \operatorname{\mathbb {N}}^n | x^\mathbf a \in I\}$. The Newton polyhedron of $I$ is 
$$\operatorname{NP}(I) = \operatorname{conv} (\mathbf a \in \operatorname{\mathbb {N}}^n | x^\mathbf a \in  I ),$$
the convex hull of the exponent set of $I$ in $\operatorname{\mathbb {R}}^n$.
<!-- end defn-2 -->
<!-- lem-2: body_md -->
\label{lem_NP} Let $I$ be a monomial ideal of $S$. Then the integral closure of $I$ is a monomial ideal with the exponent
set $E(\overline {I}) = \operatorname{NP}(I) \cap \operatorname{\mathbb {Z}}^r.$
<!-- end lem-2 -->
<!-- defn-3: body_md -->
A monomial ideal $I$ of $S$ is said to be normal if $\overline{I^s} = I^s$ for all $s \ge 1$.
<!-- end defn-3 -->
<!-- cor-1: body_md -->
\label{restriction} Let $I$ be a monomial ideal. Then 
$$\overline{(I_V)} = (\overline{I})_V.$$
<!-- end cor-1 -->
<!-- proof-1: body_md -->
Follows from Lemma \ref{lem_NP}.

<!-- cor-2: body_md -->
\label{restriction_intermediate} Let $I$ be a monomial ideal. Let $J \in \operatorname{Inter}(I^s,\overline{I^s})$ be an intermediate ideal. Let $V \subseteq [n]$ be a subset. Then $J_V \in \operatorname{Inter}(I_V^s, \overline{I_V^s}).$
<!-- end cor-2 -->
<!-- proof-2: body_md -->
Follows from the definition and Corollary \ref{restriction}.

<!-- lem-3: body_md -->
\label{partial_degree} Let $I$ be a monomial ideal. Then $\rho_j(\overline{I}) \le \rho_j(I),$ for all $j = 1, \ldots, n$. In particular, if $I$ is squarefree, then $\rho_j(\overline{I^s}) \le s$ for all $j = 1, \ldots, n$.
<!-- end lem-3 -->
<!-- proof-3: body_md -->
Let $x^\mathbf a$ be a minimal generator of $\overline{I}$. By Lemma \ref{lem_NP}, there exists non-negative real numbers $c_1, ..., c_m$ with $\sum_{i=1}^m c_i = 1$ and minimal generators $x^{\mathbf b_1}, ..., x^{\mathbf b_m}$ of $I$ such that 
$$a_j \ge \sum_{i=1}^m c_i b_{ij}$$
for all $j$. Since $x^\mathbf a$ is minimal, this implies that $a_j = \lceil \sum_{i=1}^m c_i b_{ij} \rceil$ for all $j$. Since $b_{ij} \le \rho_j(I)$ and $\sum_{i=1}^m c_i = 1$, this implies that $\sum_{i=1}^m c_i b_{ij} \le \rho_j(I)$. Thus $a_j \le \rho_j(I)$. The conclusion follows.
<!-- end proof-3 -->
<!-- thm-3: body_md -->
\label{expansion_edge} Let $I = I(G)$ be the edge ideal of a simple graph $G$. Let $f$ be a minimal generator of $\overline{I(G)^s}$. Then there exist odd cycles $C_1, ..., C_{2a}$, and edges $e_1, ...,e_b$ of $G$ such that
$$f = C_1 \cdots C_{2a} \cdot e_1 \cdots e_b,$$
where $(|C_1| + \cdots + |C_{2a}|)/2 + b = s.$ Furthermore, the odd cycles $C_i$ can be chosen so that $N(C_i) \cap C_j = \emptyset$ for all $i \neq j \in \{1, ..., 2a\}$.
<!-- end thm-3 -->
<!-- proof-4: body_md -->
For each pair $C_1, C_2$ of induced disjoint odd cycles, denote $m_{C_1,C_2} = x_{C_1} \cdot x_{C_2}$. Denote $P$ the set of all pairs of disjoint odd cycles in $G$. By \cite[Proposition 8.7.19]{V}, we have 
$$\overline{\mathcal{R}(It)} = S[It,m_{C_i,C_j}t^{m_{i,j} /2} \mid (C_i,C_j) \in P],$$
where $m_{i,j} = \deg m_{C_i,C_j}$. The conclusion follows by taking the graded component of degree $s$ and a remark that if $N(C_i) \cap C_j \neq \emptyset$, then $x_{C_i}\cdot x_{C_j} \in I^{u}$ with $u= m_{i,j}/2.$
<!-- end proof-4 -->
<!-- lem-4: body_md -->
\label{Key0}
Let $I$ be a monomial ideal in $S$. Then
\begin{multline*}
\operatorname{reg}(S/I)=\max\{|\mathbf a|+i~|~\mathbf a\in\operatorname{\mathbb {N}}^n,i\ge 0,\widetilde{H}_{i-1}(\operatorname{lk}_{\Delta_\mathbf a(I)}F;\mathrm{k})\ne 0\\ \text{ for some $F\in \Delta_\mathbf a(I)$ with $F\cap \operatorname{supp} \mathbf a=\emptyset$}\}.
\end{multline*}
In particular, if $I=I_\Delta$ is the Stanley-Reisner ideal of a simplicial complex $\Delta$ then
$\operatorname{reg}(\mathrm{k}[\Delta])=\max\{i \mid \widetilde{H}_{i-1}(\operatorname{lk}_{\Delta}F;\mathrm{k})\ne 0\text{ for some }F\in \Delta\}$.

\label{restriction_in} Let $I$ be a monomial ideal and $x$ be a variable. Then

<!-- lem-7: body_md -->
\label{restriction_inq} Let $J$ be a monomial ideal in $S$ and $V \subseteq [n]$. We have
$$\operatorname{reg} (J_V) \le \operatorname{reg} (J).$$
In particular, for a monomial ideal $I \subseteq S$ then
$$\operatorname{reg} \overline{I_V^s} \le \operatorname{reg} I^s.$$

\label{radical_colon} Let $I$ be a monomial ideal in $S$ generated by the monomials $f_1, ..., f_s$ and $\mathbf a \in \operatorname{\mathbb {N}}^n$. Then $\sqrt{I:x^\mathbf a}$ is generated by $\sqrt{f_1/\gcd(f_1, x^\mathbf a)}, ..., \sqrt{f_s/\gcd(f_s,x^\mathbf a)}$.

<!-- lem-12: body_md -->
\label{lem_radical_power} Let $I = I(G)$ be an edge ideal of a simple graph $G$. Let $\mathbf a, \mathbf b \in \operatorname{\mathbb {N}}^n$ be exponents, and $f$ a monomial in $S$. Then
\begin{enumerate}
    \item $\operatorname{rp}_\mathbf a(f) \ge \operatorname{ord}_I(x^\mathbf a)$.
    \item $\operatorname{rp}_{\mathbf a + \mathbf b}(f)  \ge \operatorname{rp}_\mathbf a(f) + \operatorname{rp}_\mathbf b(f)$.
    \item If $\operatorname{supp} f \cup \operatorname{supp} \mathbf a \supseteq \operatorname{supp} C$, where $C$ is an odd cycle of length $2\ell+1$, then $\operatorname{rp}_\mathbf a(f) \ge \ell$. Furthermore, if $\operatorname{supp} f \cap \operatorname{supp} C \neq \emptyset$ then $\operatorname{rp}_\mathbf a(f) \ge \ell+1.$

\label{rad_good_colon} Let $I$ be the edge ideal of a simple graph $G$. Let $\mathbf a \in \operatorname{\mathbb {N}}^n$ be an exponent such that $x^\mathbf a \notin I^s$. Assume that $\mathbf a$ has a decomposition $\mathbf a = \mathbf b + \mathbf{c}$ satisfying the following conditions:
\begin{enumerate}
    \item $x_i \in \sqrt{I^s:x^\mathbf a}$ for all $i \in N[\operatorname{supp} \mathbf b]$,
    \item $N[\operatorname{supp} \mathbf b] \cap \operatorname{supp} \mathbf{c} = \emptyset$.
\end{enumerate}
Let $H$ be the restriction of $G$ to $[n] \setminus N[\operatorname{supp} \mathbf b]$. Denote $t = s - \operatorname{ord}_I(x^\mathbf b)$. Then, 
\begin{equation}\label{eq_rad_colon}
    \sqrt{I^s:x^\mathbf a} = \left ( x_i \mid  i \in N[\operatorname{supp} \mathbf b] \right ) + \sqrt{I(H)^{t}:x^\mathbf{c}}.
\end{equation}
In particular, if $(\mathbf a,i)$ is an extremal exponent of $I^s$ then
\begin{equation}\label{eq_reg_inequality}
    \operatorname{reg} (I^s) \le \operatorname{reg} I(H)^t + |\mathbf b|.
\end{equation}
<!-- end lem-13 -->
<!-- proof-8: body_md -->
First, we prove the equality \eqref{eq_rad_colon}. From the assumption, it is clear that the left hand side contains the right hand side. Let $V = [n] \setminus N[\operatorname{supp} \mathbf b]$. It suffices to prove that if $f$ is a minimal generator of  $\sqrt{I^s:x^\mathbf a}$ with $\operatorname{supp} f \subseteq V$, then $f \in \sqrt{I(H)^t:x^\mathbf{c}}$. If $f \in I$, then by definition, $f \in I(H) \subseteq \sqrt{I(H)^t:x^\mathbf{c}}$. Thus, we may assume that $F = \operatorname{supp} f$ is an independent set. By Lemma \ref{criterion_in_power}, we have 
$$\sum_{j \in N(F)} a_j + \operatorname{ord}_I( h )\ge s, \text{ where } h:= \prod_{u \notin N[F]} x_u^{a_u}.$$

Since $N[\operatorname{supp} \mathbf b] \cap \operatorname{supp} \mathbf{c} = \emptyset$, $N[\operatorname{supp} \mathbf{c}] \cap \operatorname{supp} \mathbf b = \emptyset$. In particular, $N[F] \cap \operatorname{supp} \mathbf b = \emptyset$. Hence,
\begin{equation}\label{eq_3_1_2}
   \sum_{j\in N(F) \cap V} c_j = \sum_{j \in N(F)} a_j. 
\end{equation}
Let
$$h_1 := \prod_{u\in V\setminus N[F]} x_u^{c_u} = \prod_{u \in V \setminus N[F]} x_u^{a_u}.$$
Hence, $h = x^\mathbf b h_1$. Since $N[\operatorname{supp} \mathbf b] \cap \operatorname{supp} h_1 = \emptyset$, $\operatorname{ord}_I(h) = \operatorname{ord}_I(x^\mathbf b) + \operatorname{ord}_I(h_1)$. Thus
$$\sum_{j \in N(F) \cap V} c_j + \operatorname{ord}_I(h_1) = \sum_{j \in N(F)} a_j + \operatorname{ord}_I(h) - \operatorname{ord}_I(b) \ge t.$$
By Lemma \ref{criterion_in_power}, $f \in \sqrt{I(H)^t : x^\mathbf{c}}$ as required.

We now prove \eqref{eq_reg_inequality}. By \eqref{eq_rad_colon}, $\Delta_\mathbf a(I^s) = \Delta_\mathbf{c}(I(H)^t)$. Since $(\mathbf a,i)$ is an extremal exponent of $I^s$, $x^\mathbf a \notin I^s$, hence $t \ge 1$ and $x^\mathbf{c} \notin I^t$. By Lemma \ref{Key0}, 
$$\operatorname{reg} I^s = |\mathbf a| +i+1 = |\mathbf b| + |\mathbf{c}| + i+1 \le |\mathbf b| + \operatorname{reg} I(H)^t.$$
The conclusions follow.

Subcase 2.a. $|\{u,v\} \cap \{1,2,3,4,5,6\} | = 2$. Since $N[123] \cap \{4,5,6\} = \emptyset$, we may assume that $u = 5$, $v = 6$. Subcase 2.b. $|\{u,v\} \cap \{1,2,3,4,5,6\}| = 1$. We may assume that $u = 6$, $v = 7$. If $\operatorname{supp} f \cap \{1,2,3\} \neq \emptyset$, then $\operatorname{rp}_\mathbf b(f) \ge 2$, $\operatorname{rp}_\mathbf{c}(f) \ge 2$, hence $f \in \sqrt{I^4:x^\mathbf a}$, a contradiction. 2.b.$\alpha$. If $4 \in \operatorname{supp} f$ or $5 \in \operatorname{supp} f$ then $6\in \operatorname{supp} \mathbf{c}$. Thus $\operatorname{rp}_\mathbf{c}(f) \ge 3$, hence $\operatorname{rp}_\mathbf a(f)  \ge 4$, a contradiction.

$$x_7^2 x_6^2 x_4x_5 x_1x_2 \in I^4 \implies f \in \sqrt{J_1:x^\mathbf a}, \text{ a contradiction}.$$

Let $I\subseteq \mathrm{k}[x_1,\ldots,x_{11}]$ be the following edge ideal:
\begin{gather*}
I=(x_1x_2, x_1x_6, x_1x_7, x_1x_9, x_2x_6, x_2x_8, x_2x_{10}, x_3x_4, x_3x_5, x_3x_7, x_3x_{10},\\
x_4x_5, x_4x_6, x_4x_{11}, x_5x_8, x_5x_9, x_6x_{11}, x_7x_9, x_7x_{10}, x_8x_9, x_8x_{10}, x_8x_{11}, x_{10}x_{11}).
\end{gather*}
Computation with Macaulay2 \cite{M2} shows that $\operatorname{reg} I^2 = 5$ does not depend on the characteristic of the base field. It is interesting if we can prove that 
$$\operatorname{reg} I^s \le \operatorname{reg} I + 2s - 2,$$
for all $s \ge 2$. If this is true, then the mixed sum $P$ of $I$ with the edge ideal of a triangle gives an example of a non-normal edge ideal on $14$ variables with 
$$\operatorname{reg} P^s = \begin{cases} 3 + 2s & \text{ if } \operatorname{char} \mathrm{k} = 2 \\
2 + 2s & \text{ if } \operatorname{char} \mathrm{k} \neq 2.\end{cases},$$
for all $s \ge 1$.
