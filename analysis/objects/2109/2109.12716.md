<!-- definition-1: body_md -->
Let $H=(V_{H},E_{H})$ be a fixed graph with $|V_{H}|=h$ and $G_n$ be the line graph on $n$ vertices. A cylinder graph $\mathcal{G}_n$ is given by the graph Cartesian product
	\begin{align}
		\mathcal{G}_n:=G_n\times H \label{def:Cylgraph}
	\end{align}
	with vertex set $V_{\mathcal{G}_n}=V_{G_n}\times V_{H}$ and the adjacency relation given by $u=(i,j)\sim v=(i',j')$ if either $i\sim i'$ in $G_n$ and $j=j'$ or $i=i'$ and $j\sim j'$ in $H$.

\label{def:RestZ}
	Let $\mathcal{G}_{[k:l]}$ denote the principal subgraph of $\mathcal{G}_n$ generated by the vertices with $G_n$ components in the interval $[k,l]$. The restricted partition function $Z_{[k:l]}$ is defined as the partition function of the monomer-dimer model on $\mathcal{G}_{[k:l]}$.

\label{def:GibbsAverage}
	Let $\mathcal{M}$ be the collection of all matchings on the graph $\mathcal{G}_n$. Let $\mu$ be the Gibbs measure on $\mathcal{M}$ as defined in~\eqref{def:Gibbs}. <!-- lem-1: body_md -->
\label{lem:gauge}
	Consider a vertex $v$ which has weight $\nu_v$ and all adjacent edges $e=(v,w)$ with weights $\omega_{(v,w)}$ for all $w\sim v$. Let $y\in \mathds{R}$ be a constant. The transformation $\nu_v\to \nu_v+y$ and $\omega_{(v,w)}\to \omega_{(v,w)}+y$ for all $u \sim v$ leaves the Gibbs measure invariant.
<!-- end lem-1 -->
<!-- proof-1: body_md -->
Let $\mathcal{H}^y(\mathfrak{m})$ denote the Hamiltonian with the transformed weights. Note that any matching $\mathfrak{m}$ includes exactly one edge adjacent to $v$, or leaves $v$ unpaired. Thus $\mathcal{H}^y(\mathfrak{m})=\mathcal{H}(\mathfrak{m})+y$ for all $\mathfrak{m}$. Moreover for the transformed partition function $Z_n^y$, we have
	\begin{align*}Z^y_n=\sum_{\mathfrak{m} \in \mathcal{M}} \exp \mathcal{H}^y(\mathfrak{m})=\exp(y)\sum_{\mathfrak{m} \in \mathcal{M}}\exp(\mathcal{H}(\mathfrak{m}))=\exp(y)Z_n.
	\end{align*}
	With $\mu^y$ denoting the transformed Gibbs measure, for all $\mathfrak{m}\in \mathcal{M}$ we have
	\begin{align*}\mu^y(\mathfrak{m})=\frac{\exp(\mathcal{H}^y(\mathfrak{m}))}{Z^y_n}=\frac{\exp(y)\exp(\mathcal{H}(\mathfrak{m}))}{\exp(y)Z_n}=\mu(\mathfrak{m}).
	\end{align*}
	In particular, the Gibbs measure stays invariant.

\label{lem:MomentCumulant}
	Let $f:\mathds{R} \to \mathds{R}_{+}$ be $n$--times differentiable, and let $g(t)=\log f(t)$. Let the $n^{th}$ order derivatives of $f$ and $g$ be denoted by $f^{(n)}$ and $g^{(n)}$, respectively. We have,

\label{lem:errbound}
	Let $p\geqslant 1$ be fixed. Assume that, $\bigl\Vert\omega\bigr\Vert_p+ \bigl\Vert\nu\bigr\Vert_p<\infty$. We have
	\begin{align*}
		\bigl\VertR_{n,k}\bigr\Vert_p
		\leqslant h(1+\bigl\Vert\omega\bigr\Vert_p+4\bigl\Vert\nu\bigr\Vert_p).
	\end{align*}
<!-- end lem-4 -->
<!-- proof-2: body_md -->
Key to this lemma is the fact that $R_{n,k}>0$. Note that with $\mathcal{E}$ and $A$ as introduced in~\eqref{def:error} we have,
	\begin{align*}
		Z_{[1:k]}^{A}\leqslant Z_{[1:k]}\prod_{e_{k,i\in A}}\exp(-\nu_{k,i}-\nu_{k+1,i}).
	\end{align*}
	We combine this with the explicit expression for $R_{n,k}$ provided in~\eqref{def:error}. Thus
	\begin{align*}
		R_{n,k}
		\leqslant \log \bigl( 1+ \sum_{A\neq \phi} \prod_{e_{k,i}\in A}\exp(\omega_{k,i}-2\nu_{k,i}-2\nu_{k+1,i})\bigr).
	\end{align*}
	Applying the multinomial theorem, we get
	\begin{align*}
		R_{n,k}
		 & \leqslant \log \prod_{i=1}^{h} \bigl(1+\exp(\omega_{k,i}-2\nu_{k,i}-2\nu_{k+1,i})\bigr)
		\leqslant \sum_{i=1}^{h} \bigl(1+ \bigl\vert\omega_{k,i}-2\nu_{k,i}-2\nu_{k+1,i}\bigr\vert\bigr).
	\end{align*}
	The last inequality follows from $\log (1+e^{z})\leqslant 1+|z|$ for all $z\in \mathds{R}$. The proof follows from	the triangle inequality for norms.

\label{lem:freeenergy_mean}
	Let $Z_n$ be the partition function as defined in~\eqref{def:Z}. Assume that $\operatorname{\mathds{E}}(|\omega|+|\nu|)<\infty$. There exists a finite constant $f$ such that
	\begin{align*}
		n^{-1} \operatorname{\mathds{E}} \log Z_n \to f \text{ as } n\to\infty.
	\end{align*}
<!-- end lem-5 -->
<!-- proof-3: body_md -->
First, assuming convergence we show that $|f|< \infty$.
	Let $v\in V_{\mathcal{G}}$. This vertex can either be paired with an adjacent vertex or unpaired. The contribution of $v$ to any configuration can be bounded above by $1+\exp {\nu_v}+\sum_{w\sim v}\exp{\omega_{(w,w)}}$. The all monomer configuration has energy given by $\sum_{v\in V}\nu_v$. These yield the upper and lower bounds
	\begin{align*}
		\sum_{v\in V}\nu_v \leqslant \log Z_n \leqslant \sum_{v\in V} \log\bigl(1+\exp (\nu_v)+\sum_{w\sim v}\exp{\omega_{(v,w)}}\bigr).
	\end{align*}
	The expectation of the above expression is obviously bounded given the moment condition on $\omega_e$ and $\nu_v$. Now, we recall the decomposition introduced in~\eqref{def:Decomp}. We clearly have
	\begin{align*}\operatorname{\mathds{E}}\log Z_n\leqslant \operatorname{\mathds{E}}\log Z_{[1:k]} +\operatorname{\mathds{E}}\log Z_{[k+1:n]} + \bigl\VertR_{n,k}\bigr\Vert_1. \end{align*}
	It is sufficient for the first moments of $\nu$ and $\omega$ to exist to guarantee $\bigl\VertR_{n,k}\bigr\Vert_1$ being finite and uniformly bounded. Observing that
	\begin{align*} \operatorname{\mathds{E}} \log Z_{[1:k]}=\operatorname{\mathds{E}} \log Z_{k} \text{ and } \operatorname{\mathds{E}} \log Z_{[k+1:n]}=\operatorname{\mathds{E}} \log Z_{n-k} \end{align*}
	immediately yields that $\operatorname{\mathds{E}}\log Z_n$ satisfies the hypothesis of Theorem~\ref{thm:subadditive} with the corresponding $b_n$ being given by $\sup_{1\leqslant k \leqslant n}\bigl\VertR_{n,k}\bigr\Vert_1$ which is uniformly bounded above by a constant. This completes the proof of convergence.

<!-- lem-6: body_md -->
\label{lem:efron}
	There exists a finite constant $C$ such that $\operatorname{Var} (\log Z_n) \leqslant C n$ for all $n$.
<!-- end lem-6 -->
<!-- proof-4: body_md -->
The proof will proceed via the Efron-Stein inequality. Let $e=(u,v)$ denote an edge, let $\omega_e$ the corresponding edge weight, and let $\nu_u$ and $\nu_v$ be the corresponding vertex weights. Let $Z_n^{e}$ denote the partition function obtained when $\omega_e$ is replaced by an independent copy $\omega_e^{'}$. Analogously, let $Z_n^{v}$ denote the partition function obtained by replacing $\nu_v$ with independent copy $\nu_v^{'}$
	\begin{align*}\operatorname{Var} (\log Z_n)\leqslant \sum_{e\in E} \operatorname{\mathds{E}} \left ( \log Z_n^{e}-\log Z_n\right)^{2} +\sum_{v\in V}\left( \log Z_n-\log Z_n^{v}\right)^{2}.\end{align*}
	For an edge $e=(u,v)$, let $\mathcal{G}_n^{e}$ denote the graph $G_n$ with only the edge $e$ removed. Let $\mathcal{G}_n^{u,v}$ denote the principal subgraph obtained on removing vertices $u$ and $v$. We now introduce
	\begin{align*}
		\alpha_e:=Z_{\mathcal{G}_n^{e}} \text{ and } \beta_e:=Z_{\mathcal{G}_n^{u,v}}.
	\end{align*}
	Clearly, we have
	\begin{align*}
		Z_n=\alpha_e +\beta_e\exp(\omega_e).
	\end{align*}
	On taking the difference of $\log Z_n$ and $\log Z_n^{e}$,
	\begin{align*}
		\log Z_n^{e}-\log Z_n
		&=\log \left(\alpha_e + \beta_e \exp({\omega_e'}) \right)-\log\left(\alpha_e +\beta_e \exp({\omega_e})\right)\\
		&=\int_{\omega_e'}^{\omega_e}\frac{\beta_e e^z}{\alpha_e +\beta_e e^z}\, dz.
	\end{align*}
	On applying the triangle inequality, we get
	\begin{align*}
		|\log Z_n^{e}-\log Z_n|\leqslant |\omega_e'-\omega_e|.
	\end{align*}
	Now for the vertices, given a vertex $v$ let $\mathcal{G}_n^{v}$ denote the principal subgraph obtained on the removal of $v$ from $\mathcal{G}_n$. With two vertices $u$ and $v$, we denote $\mathcal{G}_n^{u,v}$ to be the principal subgraph obtained on the removal of $u$ and $v$. We will introduce
	\begin{align*}
		\widehat{\alpha}_v:=\sum_{u\sim v} \exp\left(\omega_{(u,v)} \right)Z_{\mathcal{G}_n^{u,v}} \text{ and }\widehat{\beta}_v:=Z_{\mathcal{G}_n^{v}}.
	\end{align*}
	Recurrence~\eqref{def:recur} then yields
	$Z_n=\widehat{\alpha}_v+\exp(\nu_v)\widehat{\beta}_v.$
	In a process identical to bounding the influence of the edge weights, we find
	$|\log Z_n^{v}-\log Z_n|\leqslant |\nu_v'-\nu_v|.$
	Squaring, and adding over all $e\in E_{\mathcal{G}_n}$ and $v \in V_{\mathcal{G}_n}$ completes the proof.

\label{lem:zvarconv}
	Assume that $\operatorname{\mathds{E}}(\omega^2+\nu^2)<\infty$. We have
	\begin{align*}
		n^{-1} \operatorname{Var} (\log Z_n)\to \sigma_{F}^2 \text{ as } n\to \infty.
	\end{align*}
<!-- end lem-7 -->
<!-- proof-5: body_md -->
We begin with
	\begin{align*}
		\log Z_n=\log Z_{[1:k]}+\log Z_{[k+1:n]} + R_{n,k}.
	\end{align*}
	Recall that for a random variable $X$, we denote its centered version as
	\begin{align*}
		\overline{X}:=X-\operatorname{\mathds{E}} X.
	\end{align*}
	The variance of $X$ is then given by $\bigl\Vert\overline{X}\bigr\Vert_{2}^{2}$. We have
	\begin{align*}
		\overline{\log Z_n}
		=
		\overline{\log Z_{[1:k]}}+\overline{\log Z_{[k+1:n]}} + \overline{R_{n,k}}.
	\end{align*}
	Applying the Cauchy-Schwarz inequality and independence between disjoint blocks, we get
	\begin{align*}
		\bigl\Vert\overline{\log Z_n}\bigr\Vert_{2}^{2}\leqslant \bigl\Vert\overline{\log Z_{k}}\bigr\Vert_{2}^{2} + \bigl\Vert\overline{\log Z_{n-k}} \bigr\Vert_{2}^{2} +\bigl\Vert\overline{R_{n,k}}\bigr\Vert_{2}(\bigl\Vert\overline{\log Z_{k}}\bigr\Vert_{2}+\bigl\Vert\overline{\log Z_{n-k}}\bigr\Vert_{2}+\bigl\Vert\overline{R_{n,k}}\bigr\Vert_{2}).
	\end{align*}
	Applying Lemmas~\ref{lem:errbound} and~\ref{lem:efron}, we have
	\begin{align*}\bigl\Vert\overline{\log Z_n}\bigr\Vert_{2}^{2}\leqslant \bigl\Vert\overline{\log Z_{k}}\bigr\Vert_{2}^{2}+\bigl\Vert\overline{\log Z_{n-k}}\bigr\Vert_{2}^{2}+C(\sqrt{k}+\sqrt{n-k})\end{align*}
	Applying the concavity of the square root, we then have
	\begin{align*}
		\bigl\Vert\log Z_n\bigr\Vert_{2}^{2}\leqslant \bigl\Vert\log Z_{k}\bigr\Vert_{2}^{2}+\bigl\Vert\log Z_{n-k}\bigr\Vert_{2}^{2}+C'\sqrt{n}.
	\end{align*}
	Thus, the sequence $\sigma_{F,n}^{2}=\bigl\Vert\overline{\log Z_n}\bigr\Vert_{2}^{2}$ satisfies the hypothesis of the subadditive lemma, application of which completes the proof of convergence.

<!-- lem-8: body_md -->
\label{lem:zvarlowerbound}
	There exists a constant $c>0$ such that $ \operatorname{Var} (\log Z_n) \geqslant cn$ for all $n$.
<!-- end lem-8 -->
<!-- proof-6: body_md -->
For the moment, we take $G_n$ to be the one dimensional torus $\mathds{T}_n$. To distinguish this from the usual non-transitive case, we denote the partition function as $W_n$. It is equivalent to prove a lower bound for $\operatorname{Var}\log W_n$. This is because a modification of~\ref{def:Decomp} yields that $\log Z_{n-1} + R_{n+1,1} = \log W_{n+1}$, where $\bigl\VertR\bigr\Vert_{2}$ is bounded above. On centering,
	\begin{align*}
		\bigl\Vert\overline{\log Z_{n-1}}\bigr\Vert_{2}+\bigl\Vert\overline{R_{n+1,1}}\bigr\Vert_{2}\geqslant \bigl\Vert\overline{\log W_{n+1}}\bigr\Vert_{2}.
	\end{align*}
	Consider the principal subgraph obtained on restriction to the vertices having fixed $H$--coordinate, let us say $k\in V_{H}$. There are $n$ edges corresponding to this layer, and we enumerate them as $\{e_1,e_{2},\ldots, e_n\}$. We denote their respective weights as $\{\omega_1,\omega_{2},\ldots,\omega_n\}$. Let
	\begin{align*}
		\mathcal{F}_j=\sigma\{\omega_1,\omega_{2},\ldots, \omega_j\}
	\end{align*}
	denote the filtration corresponding to the weights of the edges.
	We will use Doob's martingale decomposition to find a lower bound for the variance.

	We have that $\operatorname{\mathds{E}}\left(\log W_n\mid \mathcal{F}_j\right)$ is a martingale in $j$, and the variance of $\operatorname{\mathds{E}}\left(\log W_n \mid \mathcal{F}_n\right)$ can be written as the sum of the variances of the martingale differences. The $j^{th}$ martingale difference is
	\begin{align}\label{def:martdiff}
		\operatorname{\mathds{E}}\left(\log W_n\mid \mathcal{F}_j\right)-\operatorname{\mathds{E}}\left(\log W_n\mid\mathcal{F}_{j-1}\right).
	\end{align}
	Suppose we replace the weight $\omega_j$ of edge $e_j$ with an independent copy $\omega_j'$, let $W_n^{(j)}$ denote the new partition function. The expression in~\eqref{def:martdiff} is the same as
	\begin{align*}
		\operatorname{\mathds{E}}\bigl(\log W_n-\log W_n^{(j)}\mid \mathcal{F}_j\bigr).
	\end{align*}
	Thus,
	\begin{align*}
		\operatorname{Var}(\log W_n)
		\geqslant \operatorname{Var}(\operatorname{\mathds{E}}\left(W_n\mid \mathcal{F}_n\right))
		= \sum_{j=1}^n \bigl\Vert\operatorname{\mathds{E}}(\log W_n-\log W_n^{(j)}\mid\mathcal{F}_j)\bigr\Vert^2_{2}.
	\end{align*}
	A combination of the tower property and Jensen's inequality yields
	\begin{align}\label{def:varlb}
		\operatorname{Var}(\operatorname{\mathds{E}}\left(\log W_n\mid \mathcal{F}_n\right))
		\geqslant \sum_{j=1}^n\bigl\Vert \operatorname{\mathds{E}}(\log W_n -\log W_n^{(j)}\mid\omega_j)\bigr\Vert^2_{2}
		= \sum_{j=1}^n\operatorname{Var}(\operatorname{\mathds{E}}\left(\log W_n \mid\omega_j\right)).
	\end{align}
	Shift-invariance of $G_n$ implies edge transitivity; therefore, exchangeability of the random variables we are conditioning on. Concretely, the random variables 
	$
		\operatorname{\mathds{E}}\left(\log W_n\mid \omega_i\right)
	$ depends on $i$ only through $\omega_i$, for all $i=1,2,\ldots,n$. We write
	\[
	 g_n\left(\omega_1\right):=\operatorname{\mathds{E}}\left(\log W_n\mid \omega_1\right)
	\]
	as the common conditional expectation function.
	
	The problem of the variance lower bound for the free energy has been reduced to showing that the variance of $g$ is bounded below by a constant, since from~\eqref{def:varlb} we have
	\begin{align}\label{def:reducetog}
		\operatorname{Var}(\log W_n)
		\geqslant n\operatorname{Var}(g_n(\omega_1)).
	\end{align}
	Recall the $\alpha_{e_j}$ and $\beta_{e_j}$ introduced in Lemma~\ref{lem:efron}, so that $W_n=\alpha_{e_j} + \beta_{e_j} e^{\omega_j}$ and 
	\begin{align}\label{def:perturb}
		\log W_n -\log W_n^{(j)}=\int_{\omega_j'}^{\omega_j}\frac{\beta_{e_j} e^z}{\alpha_{e_j} +\beta_{e_j} e^z}\, dz
	\end{align}
	for all $j=1,2,\ldots,n$. W.l.o.g.~we can work with $j=1$.
	Let $u$ and $v$ be the vertices incident to $e_1$, and let $\mathcal{E}_1\subset E_{\mathcal{G}_n}$ denote the collection of edges adjacent to $e_1$. From the definitions of $\alpha_{e_1}$ and $\beta_{e_1}$, on using~\eqref{def:recur} we have
	\begin{align*}
	 e^{\nu_u+\nu_v} \cdot \beta_{e_1} 
	 \leqslant \alpha_{e_1} 
	 \leqslant
	 e^{\nu_u+\nu_v} \prod_{e=(x,y)\in \mathcal{E}_1} (1+e^{\omega_e-\nu_x-\nu_y})
	 \cdot \beta_{e_1}.
	\end{align*}
	Note that, the random variable 
	\[
	 U:=1/(1+e^{\nu_u+\nu_v} \prod_{e=(x,y)\in \mathcal{E}_1} (1+e^{\omega_e-\nu_x-\nu_y}))
	\]
	is bounded and has strictly positive mean as $\bigl\vert\mathcal{E}_1\bigr\vert$ is bounded. 
	Observe that 
	\begin{align*}
	 \operatorname{Var}( g_n(\omega_1))
	 =\frac{1}{2}\operatorname{\mathds{E}}\left(g_n(\omega_1)-g_n(\omega_1')\right)^{2}
	 =\operatorname{\mathds{E}}\left(\1_{\omega_1
	 > \omega_1'}
	 \left(g_n(\omega_1)-g_n(\omega_1')\right)^{2}\right).
	\end{align*}
	Since the integrand in~\eqref{def:perturb} is monotonically increasing in $z$, on the event $\{\omega_1> \omega_1'\}$, we have
	\begin{align*}
	 \log W_n-\log W_n^{(1)}
	 \geqslant \frac{\beta_{e_1}}{\alpha_{e_1} + \beta_{e_1}} (\omega_1-\omega_1')_{+} 
	 \geqslant (\omega_1-\omega^{'}_1)_{+} \cdot U.
	\end{align*}
	Thus,
	\begin{align*}
		\left(g_n(\omega_1)-g_n(\omega_1') \right)\1_{\omega_1> \omega_1'}
		=\1_{\omega_1> \omega_1'}\operatorname{\mathds{E}}\left(\log W_n-\log W_n^1\mid \omega_1,\omega_1'\right)
		\geqslant (\omega_1-\omega_1^{'})_{+} \operatorname{\mathds{E}}(U).
	\end{align*}
	To complete the proof, we square, take the expectation, and apply~\eqref{def:reducetog}.

\label{lem:tightness}
	Let $\mathcal{G}_n$ have maximal degree $d_{max}<\infty$. Then the sequence of empirical measures $\{\rho_n\}$ is tight in probability.
<!-- end lem-12 -->
<!-- proof-10: body_md -->
We begin by constructing a sequence of principal subgraphs of $\mathcal{G}_n$ whose corresponding Lee-Yang zeroes satisfy the interlacing condition in~\eqref{def:interlace}.
	Let $K \in (0,\infty)$ denote a cutoff. Recall the definition of $\Delta_u$ from~\eqref{weightedDegree}, the weighted degree of a vertex $u$. We remove all vertices $u$ with $\Delta_u\geqslant K$ from $\mathcal{G}_n$, and denote the principal subgraph on the remaining vertices as $\mathcal{G}^{K}$. Clearly, as $K$ increases, the vertices are added in one by one until we obtain the full graph $\mathcal{G}$ as $K \to \infty$. Let $N_K$ denote the number of vertices in $\mathcal{G}^{K}$. Clearly, \begin{align*}N_K=\sum_{v\in V}\1_{\Delta_v<K}.\end{align*}
	Now, let $u$ and $v$ be two distinct vertices. Via a coupling argument, it is not hard to show that if $\text{deg}(u)>\text{deg}(v)$, then $\operatorname{\mathds{P}}(\Delta_u<K)\leqslant \operatorname{\mathds{P}}(\Delta_v<K)$. Let $\Delta_{max}$ denote a random variable equal in distribution to the weighted degree of the vertex with maximum degree. We find the following bounds for the mean and variance

	\begin{align*}N\operatorname{\mathds{P}}\left(\Delta_{max}< K \right)\leqslant \operatorname{\mathds{E}} N_K \leqslant N \operatorname{\mathds{P}}\left(\tilde{\omega}_e< \log K \right).\end{align*}
	Since $N=nh$ and $\bigl\vert E_{\mathcal{G}_n} \bigr\vert\leqslant (d_{max}+1)nh/2$,

	\begin{align*}
		\operatorname{Var}(N_K)\leqslant \sum_{v \in V} \operatorname{Var}(\1_{\Delta_v< K}) + \sum_{v\sim w} \operatorname{Cov}(\1_{\Delta_v< K},\1_{\Delta_{w}< K}) \leqslant (d_{max}+3)nh/2.\end{align*}
	These estimates together yield concentration via Chebyshev's inequality:
	\begin{align*} 
	\operatorname{\mathds{P}} \left(\bigl\vert N_K-\operatorname{\mathds{E}} N_K \bigr\vert\geqslant \varepsilon n\right)\leqslant \frac{(d_{max}+3)h}{2\varepsilon^2 n}.
	\end{align*}
	What this means in particular is that for $\varepsilon >0$ with probability approaching $1$, \begin{align*}N_K/n\geqslant \operatorname{\mathds{P}}\left(\Delta_{max}<K\right)-\varepsilon.\end{align*}
	We do not go into the detail of the distribution of $\Delta_{max}$. This is because for proving tightness of $\rho_n$ all that is required is that $\Delta_{max}$ is a tight random variable which is clear since $d_{max}$ is bounded. The localization of the zeroes relates the weighted degree of the vertices to the absolute values of the Lee-Yang zeroes. By hypothesis, on $\mathcal{G}^{K}$ the weighted degrees of all the vertices are bounded above by $K$, thus all Lee-Yang zeroes corresponding to $Z_{\mathcal{G}^{K}}$ are located in the interval $[-K,K]$. Let us now take $K$ and $K+\delta$ such that exactly one vertex is added, that is $N_{K+\delta}=N_K+1$. Let the Lee-Yang zeroes of the corresponding partition functions be denoted as $\{\lambda_{k,K+\delta}\}_{k=1}^{N_{K+\delta}}$ and $\{\lambda_{k,K}\}_{k=1}^{N_K}$. The interlacing~\eqref{def:interlace} yields that
	\begin{align*}\lambda_{N_K,K}\geqslant \lambda_{N_K,K+\delta}.\end{align*} Extending this inductively, let $\{\lambda_{k}\}^{N}_{k=1}$ denote the Lee-Yang zeroes of the partition function corresponding to $\mathcal{G}_n$. We have
	\begin{align*}
		\lambda_{N_K}\leqslant \lambda_{N_K,K}\leqslant K.
	\end{align*}

	Thus, $2(n-N_K)/n$ is an upper bound for the fraction of Lee-Yang zeroes of $Z_{\mathcal{G}_n}(x)$ such that $|\lambda|> K$. Given an $\varepsilon > 0$, we can choose $K$ such that $2(n-N_K)/n\leqslant \varepsilon$ with probability approaching 1. This is exactly $\rho_n[-K,K]^{c}$, which establishes tightness.

	\end{align}

		\end{cases}

			&

		\end{split}

\label{lem:doublelaplace}
	Let $\{\mu_n\}_{n\geqslant 1}$ be a tight sequence of symmetric probability measures on the real line. Consider the following transform of $\mu_n$:
	\begin{align*}
		F_n(z)=\int_{\mathds{R}}\frac{z}{z+\lambda^{2}}\, d\mu_n(\lambda), \text{ } z\in (0,\infty).
	\end{align*}
	If $F_n(z)$ converges pointwise to $F(z)$ as $n\to\infty$, then $\mu_n$ is weakly convergent to a probability measure $\mu$.
<!-- end lem-20 -->
<!-- proof-18: body_md -->
The symmetry is of crucial importance here. Let $L_n$ denote the random variable with distribution $\mu_n$. Symmetry of $\mu_n$ implies that the distribution of $L_n$ can be recovered from the distribution of $L_n^{2}$. The transform $F_n(z)$ is given by
	\begin{align*}
		F_n(z)= \operatorname{\mathds{E}}\frac{z}{z+L_n^{2}}=z\operatorname{\mathds{E}}\int_{0}^{\infty}\exp(-t(z+L_n^{2}))dt, { }z>0.
	\end{align*}
	Applying Fubini's theorem, we obtain that
	\begin{align*}
		F_n(z)=z\int_{0}^{\infty} \exp(-zt)\Gamma_n(t)dt,\text{ }z\geqslant 0
	\end{align*}
	where
	\begin{align*}
		\Gamma_n(t):=\operatorname{\mathds{E}}\exp(-tL_n^{2})=\int_{\mathds{R}}e^{-\lambda^{2}t} d\mu_n(\lambda),\text{ }t\geqslant0
	\end{align*}
	is the Laplace transform of the measure $\mu_n$. Since the sequence $\{\mu_n\}$ is tight, there exists a subsequence $\{\mu_{n_{k}}\}$ weakly converging to a probability measure $\mu$. If $\mu_n$ is not weakly convergent, there must be another subsequence $\{\mu_{m_{k}}\}$ with a distinct weak limit $\mu^{*}$. Since $\exp({-zt})$ is bounded and continuous, we have that $\Gamma_{n_{k}}(t)$ and $\Gamma_{m_{k}}(t)$ both converge pointwise as $k \to \infty$ to limits, say $\Gamma(t)$ and $\Gamma^{*}(t)$, respectively. By our original hypothesis, $F_{n_{k}}(z)$ and $F_{m_{k}}(z)$ both converge to $F(z)$ as $k \to \infty$. Applying dominated convergence (we may since $|\Gamma_n|\leqslant 1$), we obtain that
	\begin{align*}
		\int_{0}^{\infty}\exp(-zt) (\Gamma(t)-\Gamma^{*}(t))dt=0,\text{ }\forall z>0.
	\end{align*}
	Uniqueness of the Laplace transform now implies that $\Gamma=\Gamma^{*}$, which in turn implies that $\mu=\mu^{*}$. Thus, $\mu_n$ is weakly convergent with limit $\mu$.

\label{lem:uannealedvarlb}
	Let both the vertex and edge weights be compactly supported in $[-K,K]$ for fixed $0<K<\infty$, such that the Lee-Yang zeroes of $Z_n$ are localized to $[-M,M]$ for some fixed $0<M<\infty$ depending on $K$. For $e^x\geqslant M+\varepsilon$ for some fixed $\varepsilon>0$ there exists $C=C(\varepsilon)> 0$ such that
	\begin{align*}
		\operatorname{Var} \langle U\rangle_n^x \geqslant C n.
	\end{align*}
<!-- end lem-22 -->
<!-- proof-21: body_md -->
The proof, as with the case of $\sigma_{F}$ is about extracting the bounds on the influence of a single edge. The martingale argument then yields the lower bound. Recall the filtration $\mathcal{F}_j$ defined in Lemma~\ref{lem:zvarlowerbound}. With the same notation, we have
	\begin{align*}
		\partial_x \log W_n-\partial_x \log W_n^{j}=\partial_x \int_{\omega_j'}^{\omega_j }\frac{e^z}{e^z+{\alpha_{e_j}}/{\beta_{e_j}}}\, dz.
	\end{align*}
	We now apply the triangle inequality and Lemma~\ref{lem:derlowerbound} to obtain
	\begin{align*}
		\bigl\vert \partial_x\log W_n-\partial_x\log W^{(j)}_n \bigr\vert
		 & = \bigl\vert \partial_x \left(\frac{\alpha_{e_j}}{\beta_{e_j}}\right) \bigr\vert\int_{\omega_j^{'}\land \omega_j}^{\omega_j'\lor \omega_j} \frac{e^z}{\left(e^z + {\alpha_{e_j}}/{\beta_{e_j}} \right)^2}\, dz \\
		 & \geqslant \left(\frac{e^{2x}-M^2}{e^{2x}+M^2}\right)_{+} \int_{\omega_j^{'}\land \omega_j}^{\omega_j' \lor \omega_j} \frac{e^z}{\left( e^z+{\alpha_{e_j}}/{\beta_{e_j}}\right)^2}\, dz \\
		 & \geqslant \left(\frac{e^{2x}-M^2}{e^{2x}+M^2}\right)_{+}\int_{\omega_j^{'}\land \omega_j}^{\omega_j'\lor \omega_j}\frac{e^z}{(e^z+\exp(2x)+(1+M)^{d_{max}})^{2}}\, dz.
	\end{align*}
	Note that the integrand may be rewritten as $y^{-1}\operatorname{sech}^{2}\left(\frac{z+y}{2}\right)$ where \[y=\exp(2x)+(1+M)^{d_{max}}.\]
	For the fixed compact interval $[-K,K]$ which our weights are restricted to, the integrand is uniformly bounded below in $z$. Applying this lower bound, squaring and then taking the expectation completes the result. The non transitive case follows in a procedure identical to that followed in Lemma~\ref{lem:zvarlowerbound}.

<!-- cor-8: body_md -->
\label{cor:UAvar}
	Let the support of $\omega$ be contained in $[a,b]$ and the support of $\nu$ be contained in $[c,d]$ such that $b-2c<-\log d_{max}$. Then we have a constant $C>0$ such that
	\begin{align*}
		\operatorname{Var} \langle U\rangle_n\geqslant Cn.
	\end{align*}
<!-- end cor-8 -->
<!-- proof-22: body_md -->
Let $e=(u,v)$ be an edge. The gauge transformed weight of $e$ is $\tilde{\omega}_e=\omega_e-\nu_u-\nu_v$. By hypothesis, this implies that with probability $1$,
	\begin{align*}
		\tilde{\omega}_e<-\log d_{max}.
	\end{align*}
	Thus for any vertex $v$,
	\begin{align*}
		\Delta_v=\sum_{e\sim v}\exp(\tilde{\omega}_e)<1
	\end{align*}
	This implies that there is an $M<1$ such that all Lee-Yang zeroes associated to $Z_n(x)$ are located in $[-M,M]$. Lemma~\ref{lem:uannealedvarlb} now applies with $x=0.$

\label{lem:linearGrowth}
	Let $u$ be as given in~\eqref{def:vertexdensity}. There exists a finite constant $C$ such that for all $m\in \mathds{N}$
	\begin{align*}|\operatorname{\mathds{E}}\langle U\rangle _{m}- m\cdot u|\leqslant C\end{align*}
<!-- end lem-23 -->
<!-- proof-23: body_md -->
Recall the dyadic subdivision as introduced in Section~\ref{section:MFE} along with the accompanying notation, and let $n=\sum_{i=0}^{l}a_i2^i$. Recall from~\eqref{def:dyadic}
	\begin{align*}
		\log Z_n=\sum_{v\in V_{k}} \log Z^{v}_{\pi_{k}(\boldsymbol{a})}+\sum_{j=1}^k\bigl(\sum_{v\in V_{j-1}}(R_{j-1}^{v}+a_jT_{j-1}^{v})\bigr).
	\end{align*}
	Taking the derivative, the expectation, and then absolute value of both sides,
	\begin{align*}
		\bigl\vert \operatorname{\mathds{E}}\partial_x\log Z_n -\operatorname{\mathds{E}} \sum_{v\in V_{k}} \partial_x\log Z^{v}_{\pi_{k}(\boldsymbol{a})} \bigr\vert
		\leqslant \sum_{j=1}^k\sum_{v\in V_j}\operatorname{\mathds{E}}\bigl\vert \partial_x R_{j-1}^{v} +a_j\partial_xT_{j-1}^{v} \bigr\vert.
	\end{align*}
	Applying Lemma~\ref{lem:cutbound}, we get
	\begin{align*}
		\bigl\vert \operatorname{\mathds{E}}\partial_x\log Z_n -\operatorname{\mathds{E}} \sum_{v\in V_{k}} \partial_x\log Z^{v}_{\pi_{k}(\boldsymbol{a})}\bigr\vert
		\leqslant C\sum_{i=1}^k2^i=C'2^k.
	\end{align*}
	Dividing both sides by $n$ and simplifying, we have
	\begin{align*}
		\bigl\vert \frac{1}{n}\operatorname{\mathds{E}}\partial_x\log Z_n- \frac{2^k}{n}\operatorname{\mathds{E}} \partial_x\log Z_{m} \bigr\vert\leqslant C'\frac{2^k}{n}
	\end{align*}
	where $m=\pi_{k}(\boldsymbol{a})$. Let $p=l-k$ and
	\begin{align*}
		\gamma_{n,p}:=1-\frac{2^{p}m}{n}.
	\end{align*}
	It is easy to show that
	\begin{align*}
		|\gamma_{n,p}|\leqslant \frac{2}{m}\leqslant 1.
	\end{align*}
	Multiplying the entire expression by $m$ and applying the triangle inequality, we get

	\begin{align*}
		\bigl\vert m\frac{1}{n}\operatorname{\mathds{E}}\partial_x\log Z_n- \operatorname{\mathds{E}} \partial_x\log Z_{m} \bigr\vert-|\gamma_{n,p}|\operatorname{\mathds{E}}\partial_x\log Z_{m}\leqslant C'.
	\end{align*}
	Let
	\begin{align*}
		|\gamma_n|:=\sup_{p}|\gamma_{n,p}|\leqslant \frac{2}{m}.
	\end{align*}
	Rearranging the terms and using the fact that $0\leqslant \partial_x \log Z_{m}\leqslant m$ (there are at most $m$ unpaired vertices), we obtain

	\begin{align*}
		\bigl\vert m\cdot \frac{\operatorname{\mathds{E}} \partial_x\log Z_n}{n} - \operatorname{\mathds{E}} \partial_x \log Z_{m} \bigr\vert\leqslant C' +2.
	\end{align*}
	Note that letting $p\to \infty$ implies that $n \to \infty$ while keeping $m$ fixed. Thus, we have
	$|m\cdot u-\operatorname{\mathds{E}}\langle U\rangle _{m}|\leqslant C'+2.$
	This completes the proof.
