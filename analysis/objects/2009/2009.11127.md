\label{defn:phyp} 
	Suppose that $M$ is a compact  Riemannian manifold with convex boundary.
	Let $\hat M$ be a regular cover of $M$ with the deck transformation group $G_M$. 
	A representation $\rho: G_M \rightarrow {\mathsf{GL}}(n, {\mathbb{R}})$ is {\em partially hyperbolic in a bundle sense} if 
	the following hold for a Riemannian metric on $M$: 
	\begin{enumerate} 
		                                           
		                                                                                         
		                         
		\item[(i)] 
		There exist nontrivial $C^0$-subbundles ${\mathbb V}_+, {\mathbb V}_0$, and ${\mathbb V}_-$ in ${\mathbb{R}}^n_{\rho}$ 
		with fibers \[{\mathbb V}_+(x), {\mathbb V}_0(x), {\mathbb V}_-(x) \subset {\mathbb V}(x)\] for the fiber ${\mathbb V}(x)$ of ${\mathbb V}$ over each point $x$ of $\mathbf{UC} M$
		invariant under the flow $\Phi_t$ of ${\mathbb V}$ over 
		the geodesic flow $\phi_t$ of $\mathbf{UC} M$. 
		                                
		                                                                 
		                                                                        
		\item[(ii)] ${\mathbb V}_+(x), {\mathbb V}_0(x)$ and ${\mathbb V}_-(x) $ for 
		each $x \in \mathbf{UC} M$ are independent subspaces, and their sum equals ${\mathbb V}(x)$.
		\item[(iii)] For any fiberwise metric on ${\mathbb{R}}^n_{\rho}$ over $\mathbf{UC} M$, 
		the lifted action of $\Phi_{t}$ on ${\mathbb V}_{+}$ (resp. ${\mathbb V}_{-}$) is dilating (resp. contracting). 
		That is, for coefficients $A> 0, a> 0$, $A' > 0, a'>0$\/: 
		\begin{enumerate} 
			\item $                
	\left|\mkern-2mu\left|\Phi_{-t}(\mathbf{v})\right|\mkern-2mu\right|_{{\mathbb{R}}^n_\rho, \phi_{-t}(m)} \leq A \exp(-a t)                
	\left|\mkern-2mu\left|\mathbf{v}\right|\mkern-2mu\right|_{{\mathbb{R}}^n_\rho, m}$ for $\mathbf{v} \in {\mathbb V}_+(m)$ as $t \rightarrow \infty$. 
			\item $                
	\left|\mkern-2mu\left|\Phi_{t}(\mathbf{v})\right|\mkern-2mu\right|_{{\mathbb{R}}^n_\rho, {\phi_{t}(m)}} \leq A \exp(-a t)                
	\left|\mkern-2mu\left|\mathbf{v}\right|\mkern-2mu\right|_{{\mathbb{R}}^n_\rho, m}$ for $\mathbf{v} \in {\mathbb V}_-(m))$ as $t \rightarrow \infty$. 
			                                                                                         
			                        
			\item (Dominance properties) 
			\begin{multline} \label{eqn:dominance} 
				\frac{                
	\left|\mkern-2mu\left|\Phi_{t}(\mathbf{w})\right|\mkern-2mu\right|_{{\mathbb{R}}^n_\rho,\phi_{t}(m)}}{                
	\left|\mkern-2mu\left|\Phi_{t}(\mathbf{v})\right|\mkern-2mu\right|_{{\mathbb{R}}^n_\rho, \phi_{t}(m)}} 
				\leq A' \exp(-a't)\frac{                
	\left|\mkern-2mu\left|\mathbf{w}\right|\mkern-2mu\right|_{{\mathbb{R}}^n_\rho, m}}{                
	\left|\mkern-2mu\left|\mathbf{v}\right|\mkern-2mu\right|_{{\mathbb{R}}^n_\rho, m}} \\
				\hbox{ for } \mathbf{v} \in {\mathbb V}_+(m), \mathbf{w} \in {\mathbb V}_0(m)
				\hbox{ as } t \rightarrow \infty, \\
				 \hbox{ or for }
				 \mathbf{v} \in {\mathbb V}_0(m), \mathbf{w} \in {\mathbb V}_-(m)
				 \hbox{ as } t \rightarrow \infty.
			\end{multline} 
		\end{enumerate} 
	\end{enumerate}
	We additionally require that $\dim {\mathbb V}_+ = \dim {\mathbb V}_- \geq 1$.  
	Here $\dim {\mathbb V}_+$ is the {\em partial hyperbolicity index} of $\rho$
	for the associated bundle decomposition ${\mathbb V}_+, {\mathbb V}_0, {\mathbb V}_-$ 
	where $\dim {\mathbb V}_+$ is maximal
	and $\dim {\mathbb V}_+ = \dim {\mathbb V}_- < n/2$, so that $\dim {\mathbb V}_0 \geq 1$. 
                                   
 Furthermore,	${\mathbb V}_0$ is said to be the {\em neutral subbundle} of ${\mathbb V}$. 
	(See Definition 1.5 of \cite{CP} or \cite{BLM}.)

 $K(\pi_1(N), 1)$ is realized by a finite complex.

\label{lem:twogeo} 
	Let $M$ be a compact manifold with 
	the induced path metric $d_{\hat M}$ on $\hat M$
	from a Riemannian metric of $M$.
	Let $\Gamma_M$ be the deck transformation group of $\hat M \rightarrow M$.

for	every pair of
	complete isometric geodesics $l_1$ and $l_2$ in $\hat M$

$l_1$ and $l_2$ under $d$ is bounded above by $C$.

\label{lem:identify} 
	Suppose that $\hat M$ is Gromov hyperbolic. 
	Then there is a quasi-isometric homeomorphism $\mathcal{F}:\mathbf{UC} \hat M \rightarrow G\hat M$
	by taking a unit vector $\vec{u}$ at $x$ to a complete isometric geodesic 
	${\mathbb{R}} \rightarrow \hat M$ passing $x$ tangent to $\vec{u}$.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
There is a map $G\hat{M} \to \mathbf{UC} \hat{M}$ given by sending the complete isometric geodesic $g: \mathbb{R} \to \hat{M}$ to $(g(0), \vec{v}_0)$ where $\vec{v}_0$ is the unit tangent vector at $g(0)$. This map is clearly bijective. In Section III of \cite{Matheus90}, the map $G\hat{M} \to \hat{M}$ defined by $g \mapsto g(0)$ is shown to be a quasi-isometry. Therefore, post-composing this map with a lift that sends each $g(0)$ to a vector in the fiber of $\mathbf{UC} \hat{M}$ over $g(0)$ results in a quasi-isometry. For a sequence $\{g_i\}$ of geodesics in $\hat{M}$, if $d_{GX}(g, g_i) \to 0$, then $(g_i(0), g_i'(0)) \to (g(0), g'(0))$ since otherwise, we would obtain a positive lower bound for the integral \eqref{eqn:dGX}.

The inverse map $\mathbf{UC} \hat{M} \to G\hat{M}$ is also continuous due to the continuity of the exponential map and considerations involving \eqref{eqn:dGX}, where $d_X(g(t), h(t))$ grows sublinearly. The continuity of integral values under $g$ and $h$ can be shown by cutting off for $|t| > N$.

\label{prop:Uc} 
	Let $M$ be a compact manifold with a covering map $\hat M \rightarrow M$
	with a deck transformation group $\Gamma_M$. 
	Suppose that $\Gamma_M$ is word-hyperbolic.

	Then there is 
	a quasi-isometric surjective map 
	$\mathcal{E}: G\hat M \rightarrow \partial^{(2)}_\infty \hat M \times {\mathbb{R}}$ with compact fibers. 
	                                                            
	The map obtained from composing with 
	a projection to the first factor is a map given by taking the endpoints of complete isometric geodesics.

Choose a basepoint $x_0$ in $\hat M$. 
	For each complete isometric geodesic $g: {\mathbb{R}} \rightarrow \hat M$, 
	let $g_{x_0}$ denote the projection point on $g({\mathbb{R}})$ that is of 
	the minimal distance from $x_0$
	and let $\partial_+ g, \partial_-g \in \partial_\infty \hat M$ denote
	the forward and backward endpoints,  
	and let $t(g) \in {\mathbb{R}}$ denote $\pm d(g(0), g_{x_0})$ where we use $+$ 
	if $g(0)$ is ahead of $g_{x_0}$.  
	
	We define 
	$\mathcal{E}(g) = (\partial_+ g, \partial_- g, t(g))$. 
	The surjectivity follows from Proposition 2.1 of Chapter 2 of 
	\cite{CDP90} since $t$ is an isometry on each complete isometric  
	geodesic. 	The compactness of the fiber is implied by Lemma \ref{lem:twogeo}. 
	
	                                                     
	                                       
	                                             
	                                                   
	                                                                  
	                                    
	
	                                                                    
	                                                                       
	                                             
	                                  
	                                                               
	                                       
	                                       
	                                                         
	                                          
	                                           
	                                                      
	                                    
	
	
	Champetier \cite{Champetier94} constructs the space 
	$G\hat \Gamma_M$ from $\hat \Gamma_M$ quasi-isometric to $\Gamma_M$ 
	following Gromov. 
	                                      
	                                 
	                                                 
	                       
	                                                    
                                             
Recall $\partial_\infty \hat{M} = \partial_\infty \hat{\Gamma}_M$. Using the orbit map $\Gamma_M \to \hat{M}$ sending $\gamma$ to $\gamma(x_0)$, we extend the map to $\hat{\Gamma}_M \to \hat{M}$ by sending each edge to a shortest geodesic.

Let $I: G\hat{\Gamma}_M \to G\hat{M}$ be a map that sends a geodesic $g$ in $\hat{\Gamma}_M$ to a geodesic $g'$ in $\hat{M}$ with the same endpoints in $\partial_\infty \hat{M}$, and maps $g(0)$ to the nearest point on the image of $g'$.

The map $I': G\hat{M} \to G\hat{\Gamma}_M$ can be defined by taking a geodesic $g$ in $\hat{M}$ to a geodesic $g'$ in $\hat{\Gamma}_M$ in a similar manner, with $g(0)$ going to one of the elements of $\Gamma_M(x_0)$ nearest to it.

By Theorem 11.72 of \cite{DK2018}, which states that every geodesic in $\hat{\Gamma}_M$ is a quasi-geodesic in $\hat{M}$ and vice versa, and considering that every geodesic in $\hat{M}$ is uniformly bounded away from one in $\hat{\Gamma}_M$ in the Hausdorff distance, it follows that $G\hat{M}$ is quasi-isometric to $G\hat{\Gamma}_M$ via the maps $I$ and $I'$, using \eqref{eqn:dGX}.

                                                                   
                                                                                
                                                                    
                                 
                                               
                                                     
                                                           
                          
                                                       
                                                          
                                         
                                         
                                                                
                                 
                                                             
                                                                
                                                            
                                     
                                                
                                    
                                            
                          
	
	                                                  
	                                                 
	                      
	                                                  
	             
	                                         
	                                                  
	                                                    
	We define 
$\mathcal{E}': G\hat \Gamma_M \rightarrow \partial_\infty^{(2)} \hat \Gamma_M \times {\mathbb{R}}$ 
	as we did for $\mathcal{E}$ above. 
	By Proposition 4.8 and (4.3) of \cite{Champetier94}, 
	$\mathcal{E}'$ sends 
	$G\hat \Gamma_M$ to 
	$\partial_\infty^{(2)} \hat \Gamma_M \times {\mathbb{R}}
	= \partial_\infty^{(2)}\hat M\times {\mathbb{R}}$ as a quasi-isometric homeomorphism. 
	Since $\mathcal{E}(g, t)$ and $\mathcal{E}'\circ I'(g, t)$ are uniformly bounded away from 
	each other,  the result follows.

\label{defn:Anosov} 
	A representation $\rho: \Gamma_M \rightarrow {\mathsf{GL}}(n, {\mathbb{R}})$ is {\em $(P^{+}, P^{-})$-Anosov} if 
	there exist continuous $\rho$-equivariant maps $\xi^+: \partial_\infty \hat \Gamma_M \rightarrow {\mathcal{F}}^+$, 
	$\xi^-:\partial_\infty \hat \Gamma_M \rightarrow {\mathcal{F}}^-$ such that:  
	\begin{enumerate} 
		\item[(i)] for all $(x, y)\in \partial_\infty^{(2)} \hat\Gamma_M$, the pair 
		$(\xi^+(x), \xi^-(y))$ is transverse, and  
		\item[(ii)] for one (and hence any) continuous and equivariant family of norms 
		$(                
	\left|\mkern-2mu\left|\cdot\right|\mkern-2mu\right|_{m})_{m\in F\Gamma_M} $ on 
		\[(T_{\xi^+(\tau^+(m))} {\mathcal{F}}^+)_{m\in F\Gamma_M}
		\Big(\hbox{resp. } 
		(T_{\xi^-(\tau^-(m))} {\mathcal{F}}^-)_{m\in F\Gamma_M}\Big):\]
		\[                 
	\left|\mkern-2mu\left|e\right|\mkern-2mu\right|_{\phi_{-t}m} \leq A e^{-at}                 
	\left|\mkern-2mu\left|e\right|\mkern-2mu\right|_{m}  \Big(\hbox{resp.} 
		                
	\left|\mkern-2mu\left|e\right|\mkern-2mu\right|_{\phi_{t}m} \leq A e^{-at}                 
	\left|\mkern-2mu\left|e\right|\mkern-2mu\right|_{m} \Big).
		   \]
	\end{enumerate}
We call 	$\xi^\pm:\partial_\infty F\Gamma_M \rightarrow {\mathcal{F}}^\pm$ the {\em Anosov maps} associated with 
$\rho: \Gamma_M \rightarrow G$. 
	(See also Section 2.5 of  \cite{GGKW17ii} for more details.)

\label{thm:bundle} 
	Let $\rho$ be partially hyperbolic with an index $k$, $k < n/2$, 
	in the singular value sense. 
	                                              
	                             
	                                                                 
	Then 
	$\rho$ is partially hyperbolic in the bundle sense with index $k$.

<!-- proof-7: body_md -->
By Lemma \ref{lem:phypAnos} and Proposition \ref{lem:Equiv}, 
	$\rho$ is $k$-dominated.

                                                      
                                     
	       
	                           
	Bochi, Potrie, and Sambarino \cite{BPS} 
	construct subbundles $Z$ and $\Theta$ of $E_\rho$ 
	where $Z$ dominates $\Theta$
	and $\dim Z = k, \dim \Theta = n-k$. 

	Now, $\rho$ is also $n-k$-dominated. 
	By Lemma \ref{lem:phypAnos}, 
	we obtain a new splitting of the bundle $E_\rho = Z'\oplus \Theta'$ 
	over $\partial_\infty^{(2)}\hat M \times {\mathbb{R}}$
	where $\dim Z' = n-k$ and $\dim \Theta'$ is $k$. 
	Furthermore, $Z'$ dominates $\Theta'$. 
	Then $Z$ is a subbundle of $Z'$, and 
	 $\Theta'$ is a subbundle of $\Theta$ by 
	Proposition 2.1 of \cite{BPS}. 
	 
	                                                
	                                                                  
	                                                                    
	                                                      
	                     
	We now form bundles $Z, Z'\cap \Theta, \Theta'$ over $\partial_\infty^{(2)}\hat M \times {\mathbb{R}}$. 
	                                           
	                                                         
	                                                                                    
	                                                            
	                                
	The dominance property \eqref{eqn:dominance} of Definition \ref{defn:phyp}  follows from those of $Z, \Theta$ and $Z', \Theta'$. 
		
	There are exponential expansions 
	of $a_i(g)$ for $1\leq i \leq k$ by the partially 
	hyperbolic condition of the premise. 
	                                             
	                          
	Now (iii)(a) of Definition \ref{defn:phyp} follows: 
	              
	                                                       
	                                                  
	                                                  
	                                                        
	Define $U_p(A)$ as the sum of eigenspaces of $\sqrt{AA^*}$ corresponding to $p$ largest eigenvalues. 
	Define $S_{n-p}(A):= U_{n-p}(A^{-1})$. 
	Since \textcolor{black}{$\hat M/\Gamma_M$} is compact, Theorem 2.2 of \cite{BPS} shows 
	\[U_k(\psi^t_{\phi^{-1}_t(x)})\rightarrow \hat Z(x) \hbox{ and } 
	S_{n-k}(\psi^t_x) \rightarrow \hat \Phi(x) \hbox{ uniformly for $x \in \hat \Gamma_M$  as } t \rightarrow \infty.\] 

	Since by the paragraph above Theorem 2.2 of \cite{BPS}, we obtain
	\[\psi^t(S_{n-k}(\psi^t_{\phi^{-1}_t(x)})^{\perp}) = U_k(\psi^t_{\phi^{-1}_t(x)}),\]
	where the unit vectors of $S_{n-k}(\psi^t_{\phi^{-1}_t(x)})^{\perp}$ goes to 
	vectors of norm at least the $k$-th singular value for $\psi_t$. 
		The angle between $\hat Z(x)$ and $\hat \Phi(x)$ is uniformly bounded below by 
a positive constant 
		by the compactness of \textcolor{black}{$\hat M/\Gamma_M$}. 
		Hence, unit vectors in $\hat Z({\phi^{-1}_t(x)})$ have 
		components in $S_{n-k}(\psi^t_{\phi^{-1}_t(x)})^\perp$ whose lengths are uniformly bounded below by a positive constant 
		for sufficiently large $t$. 

	As $t \rightarrow \infty$, $\phi_t$ passes the images of the fundamental domain under $\Gamma_M$. 
	Let $g_i$ denote the sequence of elements of $\Gamma_M$ arising in this way. 
		Since the $a_k(g_i)$ grows exponentially, the expansion properties of $Z$ follow
\textcolor{black}{by the above paragraph}. 
	(iii) of  Definition \ref{defn:phyp}  follows up to reversing the flow. 
	
			By the quasi-isometry of $\mathbf{UC} \hat M$ with $\partial^{(2)}\hat M \times {\mathbb{R}}$ in Theorem \ref{thm:identify},
	                                                  
	we construct our pulled-back partially hyperbolic bundles over $\mathbf{UC} \hat M$.

\label{thm:HKS}
	Let $N$ be a complete affine manifold.
	Let $\rho:\pi_1(N) \rightarrow {\mathsf{GL}}(n, {\mathbb{R}})$ be the linear part of the affine holonomy representation $\rho'$.
	Then $g$ has an eigenvalue equal to $1$ for each $g$ in the Zariski closure $Z(\rho(\pi_1(N)))$ of $\rho(\pi_1(N))$.
<!-- end theorem-6 -->
<!-- proof-8: body_md -->
We define function $f: Z(\rho(\pi_1(N))) \rightarrow {\mathbb{R}}$ 
		by $f(x) = \det(x-{\rm I})$ for each element $x$. 
	                                                        
		Suppose that $f(\rho(g)) \ne 0$, $g\in \Gamma_M$. Then $\rho(g)(x) + b = x$ has 
		a solution for each $b$ by Cramer's rule. 
		Hence $\rho'(g)$ has a fixed point, which is a contradiction.
		We obtain that $f$ is zero on the Zariski closure, 
		implying that each element has at least one eigenvalue equal to $1$.

<!-- proposition-2: body_md -->
\label{prop:PAnosovStrict}  
Assume the following\/{\em :} 
\begin {itemize}
	\item  $G$ is a finitely generated discrete group.                                
	\item $\rho:G \rightarrow {\mathsf{GL}}(n, {\mathbb{R}})$ is a

The main idea for proof is to pull everything back to a fundamental domain 
and find a suitable convex ball which is bounded. 

	                                                           
	                                
	Let $F$ be a compact fundamental domain of $\hat M \subset \mathds{A}^n$, 
	containing the origin $O$. 
	Let $L'$ be any affine subspace of dimension $\dim L \leq n$.
	Let $d_{L'}$ denote the path metric on $L'$ induced from  $ d_{\tilde N}$. 
	Let $r$ be any positive real number. 
	The $d_{L'}$-ball $B^{d_{L'}}_r(x)$ in $L'$ of radius $r> 0$ 
	for $x \in F$ is a subset of 
	$B^{d_{\hat M}}_r(x)$  for a $d_{\hat M}$-ball of radius $r$ 
	with center $x \in F$ 
	                                              
	since the endpoints of a $d_{L'}$-path of length $< r$ has 
	$d_{\hat M}$-distances $< r$ from $x$. 
	Since $\bigcup_{x\in F} B^{d_{\hat M}}_r(x)$ is bounded in $d_E$, 
	there is a constant $R(r, F)$ depending only on $r$ and $F$  
	so that $B^{d_{\hat M}}_r(x) \subset B^{d_E}_{R(r, F)}(O)$ for every $x \in F$
	for the Euclidean ball $B^{d_E}_{R(r, F)}(O)$ of radius 
	$R(r, F)$ with center $O$. 
	
               
                                                          
                                         
                 
                                                                                        
                                                      
                                                   
                                                  
                                             
	We take $C(R, F)$ for each $R> 0$ to be the supremum of 
	\[\{d_{L'}(x, y)| x \in F \cap L', y \in  B^{d_E}_R(O)\cap L'\}\]
where $L'$ varies over the collection of affine subspace $L'$ with $\dim L'=\dim L$
and $L' \cap F \ne \emptyset$. 
Since the set of such subspaces, $F$,  and ${\rm Cl}(B^{d_E}_R(O))$ are compact, 
and $d_{L'}(x, y)$ is a continuous function of $L'$ and $x, y$,
the supremum exists. 
      
                                                                     
	Now, \[B^{d_E}_R(O)\cap L' \subset B^{d_{L'}}_{C(R, F)}(x)\subset L'
	\hbox{ for } x\in F \cap L'\] 
and for any affine subspace $L'$ with $\dim L'=\dim L$ containing $x \in F$. 
	Now, $B^{d_E}_R(O) \cap L'$ is convex
	and is a subset of $B^{d_{L'}}_{C(R, F)}(x)$. 	
	Since \[B^{d_{L'}}_r(x) \subset B^{d_E}_{R(r, F)}(O) \cap L', x\in F,\]  
	$B^{d_{L'}}_r(x)$ is contractible to a point 
	inside $B^{d_{L'}}_{C(R(r, F), F)}(x) \subset L'$.

\[ f_2^*: H^*(\mathpzc{Hom}_{\Gamma}(C_*(\mathbb{A}^n), \mathbb{Z}\Gamma)) \to H^*(\mathpzc{Hom}_{\Gamma}(C_*(\tilde K(\Gamma, 1)), \mathbb{Z}\Gamma)). \]


                                                                         
                                         
                                               
                                                                
                                                                
                                                                        
                
                                                          
                                             
         
                                                                                                   
                       
        
                                                        
                                                                
                                                                            
                                                                           
                                                                         
                                                                           
                                                   
                                                                                                  
                                                                    
 
                                                                       
                                             

(II)(c) Since $\hat M_i$ exhausts $\mathds{A}^n$, 
$C_\ast(\mathds{A}^n)$ equals $\varinjlim C_\ast(\hat M_i)$ as ${\mathbb Z}\Gamma$-modules. 
We have 
\begin{equation} \label{eqn:dlimit} 
\mathpzc{Hom}_{\Gamma}(C_*(\mathds{A}^n), {\mathbb Z}\Gamma) = \varprojlim  \mathpzc{Hom}_{\Gamma}(C_\ast(\hat M_i), {\mathbb Z}\Gamma).
\end{equation} 
Let $\tilde \iota_i: \hat M_i \rightarrow \mathds{A}^n$ be the lift of the inclusion map $\iota_i: M_i \rightarrow N$. The lift is unique since $\pi_1(M_i) \rightarrow \pi_1(N)$ is surjective. 
Then we have
\begin{equation} \label{eqn:varp2-p2}
	\Lambda_i =\tilde \iota_i^\ast: H^l(\mathpzc{Hom}_{\Gamma}(C_*(\mathds{A}^n), {\mathbb Z}\Gamma)) \rightarrow 
	H^l( \mathpzc{Hom}_{\Gamma}(C_\ast(\hat M_i), {\mathbb Z}\Gamma)) \hbox{ for all } l.
\end{equation} 
By Theorem 3.5.8 of \cite{Weibel}, there is a surjective homomorphism 
\begin{equation} \label{eqn:varp-p2}
\Lambda: H^l(\mathpzc{Hom}_{\Gamma}(C_*(\mathds{A}^n), {\mathbb Z}\Gamma)) \rightarrow 
\varprojlim H^l( \mathpzc{Hom}_{\Gamma}(C_\ast(\hat M_i), {\mathbb Z}\Gamma)) \hbox{ for all } l.
\end{equation} 
where $\Lambda$ is the inverse limit of $\Lambda_i$.  
                                   
                                                                                                
                                                                               
                
We may assume that the image of $f_2$ is in $\hat M_i$ for all $i$. 
Let $f_2^i: \tilde K(\Gamma, 1) \rightarrow \hat M_i$  denote the restriction of the range space. 
Then  we have $\tilde \iota_i \circ f_2^i = f_2$, and 
$f_2^* = f_2^{i*} \circ \Lambda_i$ is an isomorphism. 
This means that each $\Lambda_i$ is injective. Hence, we deduced that $\Lambda$ is an isomorphism. 

                                                
                                     
(II)(d) By Lemma 7.4 of Chapter 8 of \cite{Brown}, there is a natural isomorphism
\textcolor{black}{
\[\mathpzc{Hom}_{\Gamma}(C_\ast(\hat M_i), {\mathbb Z}\Gamma) \cong \mathpzc{Hom}_c(C_\ast(\hat M_i), {\mathbb Z}).\] 
}
Since the cohomology of 
$\mathpzc{Hom}_c(C_\ast(\hat M_i), {\mathbb Z})$ is 
$H^*_c(\hat M_i)$, the right side of \eqref{eqn:varp-p2} is zero for $l > \dim L$
 by \eqref{eqn:dimL-p2}.  
By the conclusion of (II)(b), we obtain that 
\[  H^l(\Gamma, {\mathbb Z}\Gamma) = 0 \hbox{ for } l > \dim L.\]
By \eqref{eqn:cd}, we obtain
$\mathrm{cd}(\Gamma) \leq \dim L$.
