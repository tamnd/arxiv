<!-- defn-1: body_md -->
A partition $\Omega  = \{A_0,A_1,\ldots,A_K \}$ of ${\mathbb F}_{p^n}$ is called a bent partition of ${\mathbb F}_{p^n}$ of depth $K$ if every function $f:{\mathbb F}_{p^n}\mapsto {\mathbb F}_p$ such that $f(A_0)\equiv c\in {\mathbb F}_p$ and for every $j\in{\mathbb F}_p$, $D_j$ contains exactly $K/p$ elements of $\Omega \backslash \{A_0\}$, is bent.
<!-- end defn-1 -->
<!-- defn-2: body_md -->
\label{nbp}
Normal partition $\Omega  = \{U, A_1,\ldots,A_K \}$ of ${\mathbb F}_{q^2}$, where $U$ is an $m$-dimensional subspace of ${\mathbb F}_{q^2}$, is called a normal bent partition of ${\mathbb F}_{q^2}$ of depth $K$ if every function $f:{\mathbb F}_{q^2}\mapsto{\mathbb F}_p$ such that $f(U)\equiv c\in {\mathbb F}_p$ and for every $j\in{\mathbb F}_p$, $D_j$ contains exactly $K/p$ elements of $\Omega \backslash \{U\}$, is bent.
<!-- end defn-2 -->
<!-- thm-1: body_md -->
\label{th:f1}
For any function $f_1$ given in~\eqref{f1}, and any $a \in {\mathbb F}_q$, the corresponding Walsh transform coefficient $\mathcal W_{f_1}(a)=q 
\zeta_p^{c}$ and, thus, $f^*_1(a)=c$. Further, for any $a \in {\mathbb F}_{q^2} \backslash {\mathbb F}_q$, consider two cases. 
\renewcommand{\roman{enumi}}{\roman{enumi}}
\renewcommand{(\roman{enumi})}{(\roman{enumi})}
\begin{enumerate}
\item\label{it:1} If $B()$ is linear on ${\mathbb F}_{p^k}$ then $f_1$ is bent if and only if $B(\gamma)={\operatorname{Tr}}_k(t'\gamma)$ for some 
    $t'\in{\mathbb F}^*_{p^k}$. In this case, $f_1$ is a regular bent function and $f^*_1(-a)={\operatorname{Tr}}_m \left(a' \pi_1^{-1}\left(\frac{-t'}{b'}\right) 
    \right)$. Moreover, vectorial function $F_1(z)={\operatorname{Tr}}_k^m(g_1(z))+ c(1-(z+z^q)^{q-1})$ is bent and all its components belong to the 
    completed original class of Maiorana-McFarland bent functions with associated subspace $U$. 
\item\label{it:2} If $\pi_1^{-1}(x)$, considered as a polynomial in ${\mathbb F}_q[x]$, is such that $\pi_1^{-1}(x) \pmod{x^{p^k}-x}$ is a 
    linearized polynomial over ${\mathbb F}_q[x]$ (say, $\pi_1^{-1}(x) = \sum_{i} \alpha_i x^{s_i(p^k-1)+p^{r_i}}$) then $f_1$ is a regular bent 
    function and $f^*_1(-a)=B(-\delta_1)$, where $\delta_1= {\operatorname{Tr}}_k^m \left(\sum_{i} \left(\frac{a'\alpha_i}{ 
    (b')^{s_i(p^k-1)+p^{r_i}}}\right)^{p^{k-r_i}}\right)$. 
\end{enumerate}
In both cases, $a' =\frac{a+a^q}{2} \in {\mathbb F}_q $ and $b' = \frac{\beta(a-a^q)}{2} \in {\mathbb F}_q$.
<!-- end thm-1 -->
<!-- proof-1: body_md -->
Start with calculating the Walsh transform at $a=0$. 
\begin{align*}
        \mathcal W_{f_1}(0) &= \sum_{z \in {\mathbb F}_{q^2}} \zeta_p^{B({\operatorname{Tr}}_k^m(g_1(z)))+ c(1-(z+z^q)^{q-1})}\\
        &=  \sum_{z \in U} \zeta_p^c + \sum_{\gamma \in {\mathbb F}_{p^k}} \sum_{z \in \mathcal A_1(\gamma)} \zeta_p^{B({\operatorname{Tr}}_k^m(g_1(z)))}\\
        &= \lvert U \rvert \zeta_p^{c} + \sum_{\gamma \in {\mathbb F}_{p^k}}  \lvert \mathcal A_1(\gamma) \rvert \zeta_p^{B (\gamma)} \\
        &=q \zeta_p^{c} + p^{m-k}(q-1) \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B (\gamma)}\\
        &=q \zeta_p^{c}.
    \end{align*} 
Now, for any nonzero $a \in {\mathbb F}_{q^2}$, we have
\begin{align*}
     \mathcal W_{f_1}(-a) &= \sum_{z \in {\mathbb F}_{q^2}} \zeta_p^{B({\operatorname{Tr}}_k^m(g_1(z)))+ c(1-(z+z^q)^{q-1}) +{\operatorname{Tr}}_{2m}(az)}\\
        &=  \sum_{z \in U} \zeta_p^{c+{\operatorname{Tr}}_{2m}(az)} + \sum_{\gamma \in {\mathbb F}_{p^k}} \sum_{z \in \mathcal A_1(\gamma)} \zeta_p^{B({\operatorname{Tr}}_k^m  (g_1(z)))+{\operatorname{Tr}}_{2m}(az)}\\
        &= \zeta_p^c \sum_{z \in U} \zeta_p^{{\operatorname{Tr}}_{2m}(az)} + \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{z \in \mathcal A_1(\gamma )} \zeta_p^{{\operatorname{Tr}}_{2m}(az)} \\
     &= \zeta_p^c \sum_{z \in U} \zeta_p^{{\operatorname{Tr}}_{2m}(az)} + \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{\substack{j \in {\mathbb F}_q \\ {\operatorname{Tr}}_k^m(j)=\gamma }}  \sum_{z \in U_j^*} \zeta_p^{{\operatorname{Tr}}_{2m}(az)} \\
     &\stackrel{\eqref{u1}}{=} \zeta_p^c \sum_{z \in U} \zeta_p^{{\operatorname{Tr}}_{2m}(az)} + \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)}  \sum_{\substack{j \in {\mathbb F}_q \\ {\operatorname{Tr}}_k^m(j)=\gamma }}  \sum_{u \in {\mathbb F}_q^*} \zeta_p^{{\operatorname{Tr}}_{2m}\left(\frac{a}{2}(u+j\beta \pi_1(u))\right)} \\
     &= \zeta_p^c \sum_{z \in U} \zeta_p^{{\operatorname{Tr}}_{2m}(az)} + \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)}  \sum_{\substack{j \in {\mathbb F}_q \\ {\operatorname{Tr}}_k^m(j)=\gamma }}  \sum_{u \in {\mathbb F}_q^*} \zeta_p^{{\operatorname{Tr}}_m \left(u \left(\frac{a+a^q}{2} \right)+j \beta \pi_1(u) \left(\frac{a-a^q}{2} \right) \right)}\\
    &= \zeta_p^c \sum_{z \in U} \zeta_p^{{\operatorname{Tr}}_{2m}(az)} + \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)}  \sum_{\substack{j \in {\mathbb F}_q \\ {\operatorname{Tr}}_k^m(j)=\gamma }}  \sum_{u \in {\mathbb F}_q^*} \zeta_p^{{\operatorname{Tr}}_m \left(a' u+jb'\pi_1(u) \right)},
\end{align*}
where $\frac{a+a^q}{2}=a' \in {\mathbb F}_q $ and $\frac{\beta(a-a^q)}{2} =b' \in {\mathbb F}_q$. Now, we consider two cases, namely, $a \in {\mathbb F}_q^*$ and $a \in {\mathbb F}_{q^2} \backslash \Bbb F_q$.\\

\textbf{Case 1.} Let $a \in {\mathbb F}_q^*$. In this case, $a'=a$, $b'=0$ and hence we have
\begin{align*}
    \mathcal W_{f_1}(-a)  &= \zeta_p^c \sum_{z \in U} \zeta_p^{{\operatorname{Tr}}_{2m}(az)} + \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)}  \sum_{\substack{j \in {\mathbb F}_q \\ {\operatorname{Tr}}_k^m(j)=\gamma }}  \sum_{u \in {\mathbb F}_q^*} \zeta_p^{{\operatorname{Tr}}_m (au )}\\
    &= \zeta_p^c \sum_{z \in U} \zeta_p^{{\operatorname{Tr}}_m(a(z+z^q))} - p^{m-k} \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)}\\
    &=q \zeta_p^c.
\end{align*}

\textbf{Case 2.} Let $a \in {\mathbb F}_{q^2} \backslash \Bbb F_q$. In this case, we have $b' \neq 0$. For each $\gamma \in {\mathbb F}_{p^k}$, fix $j_\gamma \in 
{\mathbb F}_q$ such that ${\operatorname{Tr}}_k^m(j_\gamma)= \gamma $. Let $j= j_\gamma + v\in{\mathbb F}_q$ for some $v \in \ker({\operatorname{Tr}}_k^m)$. Then, we have 
\begin{align*}
    \allowdisplaybreaks 
    \mathcal W_{f_1}(-a) &=   \zeta_p^c \sum_{z \in U} \zeta_p^{{\operatorname{Tr}}_{2m}(az)} + \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{\substack{j \in {\mathbb F}_q \\ {\operatorname{Tr}}_k^m(j)=\gamma }}  \sum_{u \in {\mathbb F}_q^*} \zeta_p^{{\operatorname{Tr}}_m \left(a' u+jb' \pi_1(u) \right)}\\
      &= \zeta_p^c \sum_{z \in {\mathbb F}_q} \zeta_p^{{\operatorname{Tr}}_{2m}(a \beta z)} + \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)}  \sum_{v \in \ker({\operatorname{Tr}}_k^m)}  \sum_{u \in {\mathbb F}_q^*} \zeta_p^{{\operatorname{Tr}}_m \left(a' u+(j_{\gamma}+v)b' \pi_1(u)  \right)} \\
      &= \zeta_p^c \sum_{z \in {\mathbb F}_q} \zeta_p^{{\operatorname{Tr}}_m((a-a^q)\beta z)} + \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)}  \sum_{v \in \ker({\operatorname{Tr}}_k^m)}  \sum_{u \in {\mathbb F}_q^*} \zeta_p^{{\operatorname{Tr}}_m \left(a' u+j_{\gamma}b' \pi_1(u) \right)} \zeta_p^{{\operatorname{Tr}}_m \left(b' v \pi_1(u) \right)} \\
        &=  \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{u \in {\mathbb F}_q^*} \zeta_p^{{\operatorname{Tr}}_m \left(a' u+j_{\gamma} b' \pi_1(u) \right)}  \sum_{v \in \ker({\operatorname{Tr}}_k^m)}  \zeta_p^{{\operatorname{Tr}}_m \left(b' v \pi_1(u) \right)} \\
        &=  \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{u \in {\mathbb F}_q^*} \zeta_p^{{\operatorname{Tr}}_m \left(a' u+j_{\gamma} b' \pi_1(u) \right)}  \sum_{v \in \ker({\operatorname{Tr}}_k^m)}  \zeta_p^{{\operatorname{Tr}}_k \left( {\operatorname{Tr}}_k^{m} \left(b' v \pi_1(u) \right) \right)}\\
        &=p^{m-k} \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)}  \sum_{\substack{u \in {\mathbb F}_q^* \\ b' \pi_1(u) \in {\mathbb F}_{p^k}^*}} \zeta_p^{{\operatorname{Tr}}_m \left(a' u+j_{\gamma} b' \pi_1(u) \right)} \\
         &= p^{m-k} \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{\substack{u \in {\mathbb F}_q^* \\ b' \pi_1(u)  \in {\mathbb F}_{p^k}^*}} \zeta_p^{{\operatorname{Tr}}_m (a' u)+{\operatorname{Tr}}_k (\gamma b' \pi_1(u) )}.
    \end{align*}
Since $\pi_1$ is bijective on ${\mathbb F}_q$, the map $u \mapsto b'\pi_1(u)$ is bijective on ${\mathbb F}_q$. Thus, there exist precisely $p^k-1$ element $u 
\in {\mathbb F}_q^*$ such that $b'\pi_1(u) \in {\mathbb F}_{p^k}^*$. Let $b' \pi_1(u)= t$ then $u=\pi_1^{-1}\left(\frac{t}{b'}\right)$, where $\pi_1^{-1}$ is 
the compositional inverse of $\pi_1$. Thus, we have 
\begin{equation}\label{MainEqT1}
    \mathcal W_{f_1}(-a) = p^{m-k} \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{t \in {\mathbb F}_{p^k}^*} \zeta_p^{{\operatorname{Tr}}_m \left(a' \pi_1^{-1}\left(\frac{t}{b'}\right) \right)+{\operatorname{Tr}}_k (\gamma t)}.
    \end{equation}

In particular, if $B$ is linear on ${\mathbb F}_{p^k}$ then
\begin{align*}
\mathcal W_{f_1}(-a)&=p^{m-k} \sum_{t \in {\mathbb F}_{p^k}^*} \zeta_p^{{\operatorname{Tr}}_m \left(a' \pi_1^{-1}\left(\frac{t}{b'}\right) \right)}\sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)+{\operatorname{Tr}}_k (\gamma t)}\\
&=q \sum_{\substack{t \in {\mathbb F}_{p^k}^*\\B(\gamma)\equiv -{\operatorname{Tr}}_k (\gamma t)}} \zeta_p^{{\operatorname{Tr}}_m \left(a' \pi_1^{-1}\left(\frac{t}{b'}\right) \right)}\\
&=q \zeta_p^{{\operatorname{Tr}}_m \left(a' \pi_1^{-1}\left(\frac{-t'}{b'}\right) \right)},
\end{align*}
if $B(\gamma)={\operatorname{Tr}}_k(t'\gamma)$ for some $t'\in{\mathbb F}^*_{p^k}$ and $\gamma\in{\mathbb F}_{p^k}$. Obviously, $|\mathcal W_{f_1}(-a)|\neq q$ if $B$ is not of the 
trace form. From \eqref{MainEqT1} it also follows that taking $B(\gamma)={\operatorname{Tr}}_k(t'\gamma)$ for any $t'\in{\mathbb F}^*_{p^k}$ we have 
$\mathcal W_{f_1}(-a)=q \zeta_p^{{\operatorname{Tr}}_m \left(a' \pi_1^{-1}\left(\frac{-t'}{b'}\right) \right)}$. Therefore, vectorial function $F_1(z)= 
{\operatorname{Tr}}_k^m(g_1(z))+ c(1-(z+z^q)^{q-1})$ is bent and all its components are bent (each component function ${\operatorname{Tr}}_k(t'F_1(z))$ has the form of 
\eqref{f1} with $c{\operatorname{Tr}}_k(t')$ in place of $c$). 

In the general case with balanced $B$, consider $\pi_1^{-1}(x)$ as a polynomial in ${\mathbb F}_q[x]$ and assume that $\pi_1^{-1}(x) 
\pmod{x^{p^k}-x}$ is a linearized polynomial over ${\mathbb F}_q[x]$. This is equivalent to saying that 
\[
\pi_1^{-1}(x) = \sum_{i} \alpha_i x^{s_i(p^k-1)+p^{r_i}}.
\]
Therefore,
\[
\pi_1^{-1}\left( \frac{t}{b'} \right) = \sum_{i}  \frac{\alpha_i t^{p^{r_i}}}{ (b')^{s_i(p^k-1)+p^{r_i}}} = \sum_{i} \beta_i t^{p^{r_i}} =:L_1(t),
\]
where 
\[
\beta_i=\frac{\alpha_i}{ (b')^{s_i(p^k-1)+p^{r_i}}} .
\]
Then \eqref{MainEqT1} reduces to
\begin{align*}
    \mathcal W_{f_1}(-a) &= p^{m-k} \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{t \in {\mathbb F}_{p^k}^*} \zeta_p^{{\operatorname{Tr}}_m(a' L_1(t))+{\operatorname{Tr}}_k (\gamma t)}\\
    &= p^{m-k} \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{t \in {\mathbb F}_{p^k}^*} \zeta_p^{{\operatorname{Tr}}_m( \sum_{i} a'\beta_i t^{p^{r_i}})+{\operatorname{Tr}}_k (\gamma t)}\\
    &= p^{m-k} \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{t \in {\mathbb F}_{p^k}^*} \zeta_p^{{\operatorname{Tr}}_m( t\sum_{i} (a'\beta_i)^{p^{k-r_i}})+{\operatorname{Tr}}_k (\gamma t)}\\
    &= p^{m-k} \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{t \in {\mathbb F}_{p^k}^*} \zeta_p^{{\operatorname{Tr}}_k( t {\operatorname{Tr}}_k^m(\sum_{i} (a'\beta_i)^{p^{k-r_i}}))+{\operatorname{Tr}}_k (\gamma t)}\\
     &= p^{m-k} \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{t \in {\mathbb F}_{p^k}^*} \zeta_p^{{\operatorname{Tr}}_k ((\delta_1+\gamma) t)},\\
     &= p^{m-k} \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{t \in {\mathbb F}_{p^k}} \zeta_p^{{\operatorname{Tr}}_k ((\delta_1+\gamma) t)}\\
     &= q\zeta_p^{B(-\delta_1)},
\end{align*}
where
\[
    \delta_1 = {\operatorname{Tr}}_k^m \left(\sum_{i} (a'\beta_i)^{p^{k-r_i}}\right).
\]
This completes the proof.

Start with calculating the Walsh transform at $a=0$. 
\begin{equation*}
    \begin{split}
        \mathcal W_{f_2}(0) = \sum_{z \in {\mathbb F}_{q^2}} \zeta_p^{f_2(z)} &=  \sum_{z \in V} \zeta_p^c + \sum_{\gamma \in {\mathbb F}_{p^k}} \sum_{z \in \mathcal A_2(\gamma)} \zeta_p^{B({\operatorname{Tr}}_k^m(g_2(z)))}\\
        &= \lvert V \rvert \zeta_p^c + \sum_{\gamma \in {\mathbb F}_{p^k}}  \lvert \mathcal A_2(\gamma) \rvert \zeta_p^{B (\gamma)} \\
        &= \lvert V \rvert \zeta_p^c + p^{m-k}(q-1) \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B (\gamma)} \\
        &=q\zeta_p^c.
    \end{split}
\end{equation*} 
Now, for any nonzero $a \in {\mathbb F}_{q^2}$, we have
\begin{align*}
        \mathcal W_{f_2}(-a) &= \sum_{z \in {\mathbb F}_{q^2}} \zeta_p^{B({\operatorname{Tr}}_k^m(g_2(z)))+ c(1-(\beta(z-z^q))^{q-1}) +{\operatorname{Tr}}_{2m}(az)}\\

      &= \zeta_p^c \sum_{z \in {\mathbb F}_q} \zeta_p^{{\operatorname{Tr}}_{2m}(az)} + \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{\substack{j \in {\mathbb F}_q \\ {\operatorname{Tr}}_k^m(j)=\gamma }}  \sum_{z \in V_j^*} \zeta_p^{{\operatorname{Tr}}_{2m}(az)} \\
      &\stackrel{\eqref{u2}}{=} \zeta_p^c \sum_{z \in {\mathbb F}_q} \zeta_p^{{\operatorname{Tr}}_{2m}(az)} + \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)}  \sum_{\substack{j \in {\mathbb F}_q \\ {\operatorname{Tr}}_k^m(j)=\gamma }}  \sum_{u \in {\mathbb F}_q^*} \zeta_p^{{\operatorname{Tr}}_{2m}\left(\frac{a}{2} (u \beta +j \pi_2(u))\right)}\\
       &= \zeta_p^c \sum_{z \in {\mathbb F}_q} \zeta_p^{{\operatorname{Tr}}_{2m}(az)} + \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)}  \sum_{\substack{j \in {\mathbb F}_q \\ {\operatorname{Tr}}_k^m(j)=\gamma }}  \sum_{u \in {\mathbb F}_q^*} \zeta_p^{{\operatorname{Tr}}_m \left(u \beta \left(\frac{a-a^q}{2} \right)+j \pi_2(u) \left(\frac{a+a^q}{2} \right) \right)}\\
       &= \zeta_p^c \sum_{z \in {\mathbb F}_q} \zeta_p^{{\operatorname{Tr}}_{2m}(az)} + \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{\substack{j \in {\mathbb F}_q \\ {\operatorname{Tr}}_k^m(j)=\gamma }}  \sum_{u \in {\mathbb F}_q^*} \zeta_p^{{\operatorname{Tr}}_m \left(a' u+jb'\pi_2(u) \right)},
    \end{align*}
where $\frac{\beta(a-a^q)}{2}=a' \in {\mathbb F}_q $ and $\frac{a+a^q}{2}  =b' \in {\mathbb F}_q$. Now, we consider two cases, namely, $a \in {\mathbb F}_q^*$ and $a \in {\mathbb F}_{q^2} \backslash \Bbb F_q$.\\
\textbf{Case 1.} Let $a \in {\mathbb F}_q^*$. In this case, $a'=0$ and $b'=a$. For each $\gamma \in {\mathbb F}_{p^k}$, fix $j_\gamma \in {\mathbb F}_q$ such that 
${\operatorname{Tr}}_k^m(j_\gamma)= \gamma $. Let $j= j_\gamma + v$ for some $v \in \ker({\operatorname{Tr}}_k^m)$. Then, we have 
\begin{align*}
        \mathcal W_{f_2}(-a) &= \zeta_p^c \sum_{z \in {\mathbb F}_q} \zeta_p^{{\operatorname{Tr}}_m((a+a^q)z)} + \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{\substack{j \in {\mathbb F}_q \\ {\operatorname{Tr}}_k^m(j)=\gamma }}  \sum_{u \in {\mathbb F}_q^*} \zeta_p^{{\operatorname{Tr}}_m (ja\pi_2(u))}\\
        &=\zeta_p^c \sum_{z \in {\mathbb F}_q} \zeta_p^{{\operatorname{Tr}}_m(2az)} + \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{v \in \ker({\operatorname{Tr}}_k^m)}  \sum_{u \in {\mathbb F}_q^*} \zeta_p^{{\operatorname{Tr}}_m ((j_\gamma +v)a\pi_2(u))}\\
        &= \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)}  \sum_{u \in {\mathbb F}_q^*} \zeta_p^{{\operatorname{Tr}}_m (j_\gamma a\pi_2(u))}  \sum_{v \in \ker({\operatorname{Tr}}_k^m)} \zeta_p^{{\operatorname{Tr}}_k ({\operatorname{Tr}}_k^{m}(av\pi_2(u)))}\\
        &= p^{m-k}\sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)}  \sum_{\substack{u \in {\mathbb F}_q^* \\ a\pi_2(u) \in {\mathbb F}_{p^k}^*}} \zeta_p^{{\operatorname{Tr}}_k (\gamma a\pi_2(u))}  \\
        &= p^{m-k}(p^k-1) +p^{m-k}\sum_{\gamma \in {\mathbb F}_{p^k}^*} \zeta_p^{B(\gamma)}  \sum_{\substack{u \in {\mathbb F}_q^* \\ a\pi_2(u) \in {\mathbb F}_{p^k}^*}} \zeta_p^{{\operatorname{Tr}}_k (\gamma a\pi_2(u))} \\
        &= p^{m-k}(p^k-1)-p^{m-k}\sum_{\gamma \in {\mathbb F}_{p^k}^*} \zeta_p^{B(\gamma)}\\
        &= p^{m}-p^{m-k}\sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)}\\
        &= q.
\end{align*}

\textbf{Case 2.} Let $a \in {\mathbb F}_{q^2} \backslash \Bbb F_q$. In this case, we have $a' \neq 0$. Now we consider two subcases, namely, $b'=0$ and $b' \neq 0$.\\

\textbf{Subase 2.1.} Let $b'= 0$. In this case, we have 
\begin{align*}
        \mathcal W_{f_2}(-a) &= \zeta_p^c \sum_{z \in {\mathbb F}_q} \zeta_p^{{\operatorname{Tr}}_{2m}(az)} + \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{\substack{j \in {\mathbb F}_q \\ {\operatorname{Tr}}_k^m(j)=\gamma }}  \sum_{u \in {\mathbb F}_q^*} \zeta_p^{{\operatorname{Tr}}_m \left(a' u \right)}\\
        &= \zeta_p^c \sum_{z \in {\mathbb F}_q} \zeta_p^{{\operatorname{Tr}}_m((a+a^q)z)} -p^{m-k} \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \\
        &=q \zeta_p^c.
    \end{align*}

\textbf{Subase 2.2.} Let $b' \neq 0$. In this case, we have 
\begin{align*}
        \mathcal W_{f_2}(-a) &= \zeta_p^c \sum_{z \in {\mathbb F}_q} \zeta_p^{{\operatorname{Tr}}_{2m}(az)} + \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{\substack{j \in {\mathbb F}_q \\ {\operatorname{Tr}}_k^m(j)=\gamma }}  \sum_{u \in {\mathbb F}_q^*} \zeta_p^{{\operatorname{Tr}}_m \left(a' u+jb'\pi_2(u) \right)}\\
        &= \zeta_p^c \sum_{z \in {\mathbb F}_q} \zeta_p^{{\operatorname{Tr}}_m((a+a^q)z)} + \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{v \in \ker({\operatorname{Tr}}_k^m)}  \sum_{u \in {\mathbb F}_q^*} \zeta_p^{{\operatorname{Tr}}_m (a' u+(j_\gamma +v)b'\pi_2(u))}\\
        &= \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)}  \sum_{u \in {\mathbb F}_q^*} \zeta_p^{{\operatorname{Tr}}_m (a' u+j_\gamma b'\pi_2(u))}  \sum_{v \in \ker({\operatorname{Tr}}_k^m)} \zeta_p^{{\operatorname{Tr}}_k ({\operatorname{Tr}}_k^{m}(v b' \pi_2(u)))}\\
        &= p^{m-k}\sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)}  \sum_{\substack{u \in {\mathbb F}_q^* \\ b'\pi_2(u) \in {\mathbb F}_{p^k}^*}} \zeta_p^{{\operatorname{Tr}}_m (a' u)+{\operatorname{Tr}}_k (\gamma b'\pi_2(u))}.
    \end{align*}
Since $\pi_2$ is bijective on ${\mathbb F}_q$, the map $u \mapsto b'\pi_2(u)$ is bijective on ${\mathbb F}_q$. Thus, there exist precisely $p^k-1$ element $u 
\in {\mathbb F}_q^*$ such that $b'\pi_2(u) \in {\mathbb F}_{p^k}^*$. Let $b' \pi_2(u)= t$ then $u=\pi_2^{-1}\left(\frac{t}{b'}\right)$, where $\pi_2^{-1}$ is 
the compositional inverse of $\pi_2$. Thus, we have 
\begin{equation}\label{MainEqT2}
        \mathcal W_{f_2}(-a) = p^{m-k}\sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)}  \sum_{t \in {\mathbb F}_{p^k}^*} \zeta_p^{{\operatorname{Tr}}_m \left(a' \pi_2^{-1}\left(\frac{t}{b'}\right)\right)+{\operatorname{Tr}}_k (\gamma t)}.
    \end{equation}

In particular, if $B$ is linear on ${\mathbb F}_{p^k}$ then
\begin{align*}
\mathcal W_{f_2}(-a)&=p^{m-k} \sum_{t \in {\mathbb F}_{p^k}^*} \zeta_p^{{\operatorname{Tr}}_m \left(a' \pi_2^{-1}\left(\frac{t}{b'}\right) \right)}\sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)+{\operatorname{Tr}}_k (\gamma t)}\\
&=q \sum_{\substack{t \in {\mathbb F}_{p^k}^*\\B(\gamma)\equiv -{\operatorname{Tr}}_k (\gamma t)}} \zeta_p^{{\operatorname{Tr}}_m \left(a' \pi_2^{-1}\left(\frac{t}{b'}\right) \right)}\\
&=q \zeta_p^{{\operatorname{Tr}}_m \left(a' \pi_2^{-1}\left(\frac{-t'}{b'}\right) \right)},
\end{align*}
if $B(\gamma)={\operatorname{Tr}}_k(t'\gamma)$ for some $t'\in{\mathbb F}^*_{p^k}$ and $\gamma\in{\mathbb F}_{p^k}$. Obviously, $|\mathcal W_{f_2}(-a)|\neq q$ if $B$ is not of the 
trace form. From \eqref{MainEqT2} it also follows that taking $B(\gamma)={\operatorname{Tr}}_k(t'\gamma)$ for any $t'\in{\mathbb F}^*_{p^k}$ we have 
$\mathcal W_{f_2}(-a)=q \zeta_p^{{\operatorname{Tr}}_m \left(a' \pi_2^{-1}\left(\frac{-t'}{b'}\right) \right)}$. Therefore, vectorial function $F_2(z)= 
{\operatorname{Tr}}_k^m(g_2(z))+ c(1-(\beta^{-1}(z-z^q))^{q-1})$ is bent and all its components are regular (each component function ${\operatorname{Tr}}_k(t'F_2(z))$ has 
the form of \eqref{f1} with $c{\operatorname{Tr}}_k(t')$ in place of $c$). 

In the general case with balanced $B$, consider $\pi_2^{-1}(x)$ as a polynomial in ${\mathbb F}_q[x]$ and assume that $\pi_2^{-1}(x) 
\pmod{x^{p^k}-x}$ is a linearized polynomial over ${\mathbb F}_q[x]$. This is equivalent to saying that 
\[
\pi_2^{-1}(x) = \sum_{i} \alpha_i x^{s_i(p^k-1)+p^{r_i}}.
\]
Therefore,
\[
\pi_2^{-1}\left( \frac{t}{b'} \right) = \sum_{i} \frac{\alpha_i t^{p^{r_i}}}{ (b')^{s_i(p^k-1)+p^{r_i}}} = \sum_{i} \beta_i t^{p^{r_i}} =:L_2(t),
\]
where 
\[
\beta_i=\frac{\alpha_i}{ (b')^{s_i(p^k-1)+p^{r_i}}} .
\]
Then \eqref{MainEqT2} reduces to
\begin{align*}
    \mathcal W_{f_1}(-a) &= p^{m-k} \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{t \in {\mathbb F}_{p^k}^*} \zeta_p^{{\operatorname{Tr}}_m(a' L_2(t))+{\operatorname{Tr}}_k (\gamma t)}\\
    &= p^{m-k} \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{t \in {\mathbb F}_{p^k}^*} \zeta_p^{{\operatorname{Tr}}_m( \sum_{i} a'\beta_i t^{p^{r_i}})+{\operatorname{Tr}}_k (\gamma t)}\\
    &= p^{m-k} \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{t \in {\mathbb F}_{p^k}^*} \zeta_p^{{\operatorname{Tr}}_m( t\sum_{i} (a'\beta_i)^{p^{k-r_i}})+{\operatorname{Tr}}_k (\gamma t)}\\
    &= p^{m-k} \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{t \in {\mathbb F}_{p^k}^*} \zeta_p^{{\operatorname{Tr}}_k( t {\operatorname{Tr}}_k^m(\sum_{i} (a'\beta_i)^{p^{k-r_i}}))+{\operatorname{Tr}}_k (\gamma t)}\\
     &= p^{m-k} \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{t \in {\mathbb F}_{p^k}^*} \zeta_p^{{\operatorname{Tr}}_k ((\delta_2+\gamma) t)},\\
     &= p^{m-k} \sum_{\gamma \in {\mathbb F}_{p^k}} \zeta_p^{B(\gamma)} \sum_{t \in {\mathbb F}_{p^k}} \zeta_p^{{\operatorname{Tr}}_k ((\delta_2+\gamma) t)}\\
     &= q\zeta_p^{B(-\delta_2)},
\end{align*}
where 
\[
\delta_2 = {\operatorname{Tr}}_k^m \left(\sum_{i} (a'\beta_i)^{p^{k-r_i}}\right).
\]
This completes the proof.

<!-- defn-3: body_md -->
Vectorial bent function $F: {\mathbb F}_{p^n} \mapsto {\mathbb F}_{p^m}$ is called vectorial dual-bent if the set consisting of the duals of the component functions of $F$, forms the set of component functions of another vectorial bent function from ${\mathbb F}_{p^n}$ to ${\mathbb F}_{p^m}$, denoted by $F^*$, which is called a vectorial dual of $F$.

<!-- defn-4: body_md -->
Let $n=2m$ be a positive integer and let $0<k\leq m$. A vectorial dual-bent function $F: {\mathbb F}_{p^n} \mapsto {\mathbb F}_{p^k}$ is said to satisfy {\bf 
Condition A}. If 
\begin{enumerate}
    \item all component functions $F_t= {\operatorname{Tr}}_k(tF)$ with $t \in {\mathbb F}_{p^k}^*$, are regular, or all component functions are weakly regular 
        but not regular 
    \item for any $t_1, t_2 \in {\mathbb F}_{p^k}^*$ with $t_1+t_2 \neq 0$, the equality
    \[
    (F_{t_1})^*+ (F_{t_2})^*=(F_{t_1+t_2})^*
    \]
    holds (that is, the sum of the duals equals the dual of the sum).
\end{enumerate}

The vectorial dual-bent property is proven. For any $t_1, t_2 \in {\mathbb F}_{p^k}^*$, with $t_1+t_2 \neq 0$, the equality $(F_{t_1})^*+ (F_{t_2})^*=(F_{t_1+t_2})^*$
    follows directly from the linearity of the trace function.

<!-- defn-5: body_md -->
Let $\mathcal S = \{U, \mathcal A(\gamma) \mid \gamma \in {\mathbb F}_{p^k}\}$ be a normal bent partition of ${\mathbb F}_{p^{2m}}$ of depth $p^k$. Then $\mathcal S$ is said to satisfy {\bf Condition C}. If
    \begin{enumerate}
        \item ${\mathbb F}_p^* \mathcal A(\gamma) = \mathcal A(\gamma)$ for all $\gamma \in {\mathbb F}_{p^k}$.
        \item all bent functions obtained from $\mathcal S$ are regular, or all bent functions obtained from $\mathcal S$ are weakly regular but not regular.
    \end{enumerate}
    Joining $U$ with some $\mathcal A(\gamma)$, we obtain a bent partition of ${\mathbb F}_{p^{2m}}$ satisfying Condition C.

<!-- prop-2: body_md -->
Take appropriate permutations $\pi_1$ (as described in Corollary~\ref{co1}), then normal bent partition $\mathcal S_1$ defined
    in~\eqref{part1}, satisfies Condition C.

<!-- proof-5: body_md -->
Part (2) of Condition C is satisfied because of Corollary~\ref{co1}. Further, since $U= \beta {\mathbb F}_q$, then for any $c \in {\mathbb F}_p^*$, $cU=U$. 
Now we show that for any $j \in {\mathbb F}_{p^k}$, $cU_j=U_j$. It is easy to verify that the function 
    \[
    u \mapsto \frac{1}{2}(\pi^{-1}(u)+j \beta u)
    \]
    is a bijection from ${\mathbb F}_q$ to $U_j$. For any  $c \in {\mathbb F}_p^*$, we have
    \[
    cU_j= \left\{ \frac{1}{2}(c\pi^{-1}(u)+j \beta cu) \mid u \in {\mathbb F}_q \right\} = \left\{ \frac{1}{2}(\pi^{-1}(u)+j \beta u) \mid cu \in {\mathbb F}_q \right\} =U_j,
    \]
    where the second equality follows from the fact that
    \[
    \pi_1^{-1}(cu) = \sum_{i} \alpha_i (cu)^{s_i(p^k-1)+p^{r_i}} = c \sum_{i} \alpha_i u^{s_i(p^k-1)+p^{r_i}}=c\pi_1^{-1}(u) .
    \]
This completes the proof.
