\label{main_estimates}
Put $\theta=1-\frac{1}{12\log 10}$.
There are explicit positive constants $A_k,B_k$,
depending only on $k$, such that, for every
$k\in\mathbb{Z}_{\geq0}$, $q\in\mathbb{Z}_{\geq1}$
and $X>0$,
\begin{align*}
\big|\mathbf{\check m}_q^{(k)}(X)-Q_{k,q}(\log X)\big|
&\leq
\frac{\sqrt{q}}{\varphi_{1/2}(q)}
\frac{A_k}{\sqrt{X}}
+\frac{q^\theta}{\varphi_\theta(q)}
\frac{B_k\,\mathds{1}_{X\geq10^{12}}(X)}{\log X},\\
\big|\mathbf{\tilde m}_q^{(k)}(X)
-\widetilde Q_{k,q}(\log X)\big|
&\leq
\mathpzc p_{1/2}(q)\frac{3.575A_k}{\sqrt{X}}
+\mathpzc p_\theta(q)
\frac{1.693B_k\,\mathds{1}_{X\geq10^{12}}(X)}{\log X}.
\end{align*}
Here $Q_{0,q}=\widetilde Q_{0,q}=0$ and for $k\geq1$, the polynomials $Q_{k,q}$ and
$\widetilde Q_{k,q}$ have degree $k-1$ and respective
leading coefficients $
\frac{kq}{\varphi(q)}$ and $k\zeta(2)\mathpzc p_1(q)$.
Their coefficients are given by \eqref{Qkq} and
\eqref{tilde_Pol}, and the constants $A_k,B_k$ are
specified in \eqref{AB}, \eqref{Ak_rec} and \eqref{Bk_rec}.

<!-- lemma-1: body_md -->
\label{EM}Let $X>0$ and $j\in\mathbb{Z}_{\geq 0}$. Then
\begin{equation}\label{R_j}
\sum_{m\leq X}\frac{\log^jm}{m}=\frac{\log^{j+1}X}{j+1}+\gamma_j+R_j(X)
\end{equation}
where $\gamma_j$ is the $j$-th Stieltjes constant and 
\begin{equation*}
|R_j(X)|\leq \frac{|\log X|^j}{2X}+\frac{P_j(|\log X|)}{X^2}
\end{equation*}
where $P_j$ is the following polynomial
\begin{equation}\label{Pol}
P_0:t\geq 0\mapsto\frac{1}{6},\qquad P_j:t\geq 0\mapsto\frac{t^j}{6}+\frac{jt^{j-1}}{4}+\frac{j!}{2}\sum_{i=2}^{j}\frac{t^{j-i}}{(j-i)!2^{i}},\qquad j\geq 1.
\end{equation}
<!-- end lemma-1 -->
<!-- proofbold-1: body_md -->
Suppose that $X\geq 1$. By applying the Euler-Maclaurin summation formula \eqref{EulerMaclaurin} (of order $2$) to the function $\ell_j:t>0\mapsto \frac{\log^jt}{t}$, we obtain
\begin{align}\label{eq:frommaclau}
\sum_{m\leq X}\frac{\log^jm}{m}&=\mathds{1}_{\{j=0\}}(j)+\sum_{1<m\leq X}\frac{\log^jm}{m}=\frac{\log^{j+1}X}{j+1}+I_j+R_j(X),
\end{align}
where 
\begin{equation*}
I_j=\frac{7\ \mathds{1}_{\{j=0\}}(j)}{12}-\frac{\mathds{1}_{\{j=1\}}(j)}{12}-\frac{1}{2}\int_1^\infty B_2(\{t\})\ell_j''(t)dt\\
\end{equation*}
and
\begin{equation}\label{R_j_alt}
R_j(X)=-B_1(\{X\})\ell_j(X)+\frac12 B_2(\{X\})\ell_j'(X)+\frac{1}{2}\int_X^\infty B_2(\{t\})\ell_j''(t)\ dt.
\end{equation}
Observe that
\begin{align*}
\ell_0'(t)&=\frac{-1}{t^2},\quad\ell_1'(t)=\frac{1-\log t}{t^2},\quad\ell_j'(t)=\frac{j\log^{j-1}t-\log^jt}{t^2}\qquad(j\geq 2),\\
\ell_0''(t)&=\frac{2}{t^3},\quad\ell_1''(t)=\frac{2\log t-3}{t^3},\quad\ell_j''(t)=\frac{\log^{j-2}t\ (2\log^2t-3j\log t+j(j-1))}{t^3}\qquad (j\geq 2).
\end{align*}
By estimates \ref{BernoulliBounds}, $B_1$ and $B_2$ are bounded functions on $[0,1]$; hence, $R_j(X)=O_j\left(\frac{\log^jX}{X}\right)$ and the integral in $I_j$ is convergent to a constant. In fact, by \eqref{gamma:limit}, we have $I_j=\gamma_j$, where $\gamma_j$ is the $j$-th Stieltjes constant. 

Now, by using \eqref{BernoulliBounds} in \eqref{R_j_alt}, we obtain
\begin{equation}\label{eq:Rj-basic}
|R_j(X)|
\leq\frac{\log^jX}{2X}
+\frac{|\ell_j'(X)|}{12}
+\frac1{12}\int_X^\infty|\ell_j''(t)|\ dt.
\end{equation}
For $j=0$, the above reads as
\begin{equation*}
|R_0(X)|
\leq\frac{1}{2X}+\frac{1}{12X^2}
+\frac{1}{12}\int_X^\infty\frac{2}{t^3}\ dt
=\frac{1}{2X}+\frac{1}{6X^2}.
\end{equation*}
Suppose now that $j\geq1$. Then we conclude from \eqref{eq:Rj-basic}
\begin{align}\label{eq:Rj-integrals}
|R_j(X)|
\leq&
\frac{\log^jX}{2X}
+\frac{\log^jX+j\log^{j-1}X}{12X^2}\\
&+\frac1{12}\bigg(
2\int_X^\infty\frac{\log^jt}{t^3}\ dt+3j\int_X^\infty\frac{\log^{j-1}t}{t^3}\ dt+j(j-1)\int_X^\infty\frac{\log^{j-2}t}{t^3}\ dt
\bigg),\nonumber
\end{align}
Note that the above last integral does not vanish only when $j\geq 2$. Integration by parts yields
\[
\int_X^\infty\frac{\log^0t}{t^3}\ dt=\frac{1}{2X^2},
\qquad
\int_X^\infty\frac{\log^rt}{t^3}\ dt=\frac{\log^rX}{2X^2}+\frac{r}{2}\int_X^\infty\frac{\log^{r-1}t}{t^3}\ dt
\qquad(r\geq1).
\]
Hence, by induction, for any $r\in\mathbb{Z}_{\geq 0}$,
\begin{equation}\label{eq:Jr}
\int_X^\infty\frac{\log^rt}{t^3}\ dt
=\frac{1}{2X^2}\sum_{i=0}^r
\frac{r!\ \log^{r-i}X}{(r-i)!\ 2^{i}}.
\end{equation}

Substituting \eqref{eq:Jr} into \eqref{eq:Rj-integrals},
the coefficient of $\frac{\log^jX}{X^2}$ is $
\frac{1}{12}+\frac{1}{12}=\frac{1}{6}$,
and that of $\frac{\log^{j-1}X}{X^2}$ is $\frac{j}{12}+\frac{j}{24}+\frac{j}{8}=\frac{j}{4}$.
For $2\leq i\leq j$, the coefficient of $\frac{\log^{j-i}X}{X^2}$ is
\begin{equation*}
\frac{1}{12}\left(
\frac{j!}{(j-i)!\ 2^{i}}
+\frac{3j!}{(j-i)!\ 2^{i}}
+\frac{j!}{(j-i)!\ 2^{i-1}}
\right)
=\frac{j!}{2(j-i)!\ 2^{i}}.
\end{equation*}
Therefore
\begin{equation*}
|R_j(X)|
\leq\frac{\log^jX}{2X}+\frac{P_j(\log X)}{X^2},
\end{equation*}
where $P_j$ is defined by \eqref{Pol}.

Finally, suppose that $0<X<1$. The Euler--Maclaurin remainder identity \eqref{R_j_alt}
remains valid for every $X>0$, with the leftmost sum
being empty. For every integer $r\geq0$, the
substitution $t=Xe^v$ gives
\begin{align*}
\int_X^\infty\frac{|\log t|^r}{t^3}\ dt
=\frac{1}{X^2}\int_0^\infty
|\log X+v|^r e^{-2v}\ dv&\leq\frac{1}{X^2}\int_0^\infty
\big(|\log X|+v\big)^r e^{-2v}\ dv\\
&=\frac{1}{2X^2}\sum_{i=0}^{r}
\frac{r!}{(r-i)!2^i}|\log X|^{r-i}.
\end{align*}
Hence, applying the above estimate to the derivatives of
$\ell_j(t)=\log^j(t)/t$ in \eqref{eq:Rj-basic}, we obtain the same
polynomial $P_j$, first for $j=0$ and then evaluated at
$|\log X|$, via \eqref{eq:Rj-integrals}, for $j\geq 1$.

\label{recurrence}
Let $k\in\mathbb Z_{\geq2}$. For every $X>0$,
\begin{equation*}
\mathbf{\check m}^{(k)}(X)
=k\log^{k-2}X\log^+X
-k\sum_{j=1}^{k-1}\bigg(\genfrac{}{}{0pt}{}{k-1}{j}\bigg)
(-1)^{k-1-j}\gamma_{k-1-j}
\mathbf{\check m}^{(j)}(X)
+\Xi_k(X),
\end{equation*}
where $\Xi_k(X)$ is defined in \eqref{Xi}. 

Moreover,
\begin{align*}
|\Xi_k(X)|
&\leq k(k-1)\int_1^X
\bigg|m\bigg(\frac{X}{t}\bigg)\bigg|W_{k-2}(t)\ dt,
\end{align*}
where, for $r\geq0$,
\begin{equation}\label{W_r}
W_r:t\geq 1\mapsto
\frac{2^{r-1}\log^r t}{t^2}
+\frac{1}{t^3}\sum_{j=0}^{r}
\bigg(\genfrac{}{}{0pt}{}{r}{j}\bigg)\log^{r-j}t\ P_j(\log t),
\end{equation}
and the $P_j$ are defined by \eqref{Pol}
<!-- end theorem-3 -->
<!-- proofbold-2: body_md -->
If $0<X<1$, then $\Xi_k(X)=0$ and the integral
vanishes, since $m(X/t)=0$ for $X<t<1$.
We may therefore suppose that $X\geq1$.

By Lemma \ref{EM}, the triangle inequality and the
binomial theorem, for every $r\geq0$ and $t\geq1$,
\begin{equation}\label{S_r:bound}
|S_r(t)|
\leq tW_r(t).
\end{equation}
On the other hand, the definition of $R_j$ gives
\begin{equation*}
S_r(t)
=\sum_{m\leq t}\frac{\log^r(t/m)}{m}
-\frac{\log^{r+1}t}{r+1}
-\sum_{j=0}^{r}(-1)^j\bigg(\genfrac{}{}{0pt}{}{r}{j}\bigg)
\gamma_j\log^{r-j}t.
\end{equation*}
For $r\geq1$, the function $S_r$ is continuous
and continuously differentiable between consecutive
integers, with
\begin{equation*}
S_r(1)=(-1)^{r+1}\gamma_r,
\qquad
S_r'(t)=\frac{r}{t}S_{r-1}(t),
\qquad t\geq 1,\quad t\notin\mathbb Z.
\end{equation*}
Since $k\geq2$, partial summation applies to
$S_{k-1}$ and yields
\begin{align*}
\Xi_k(X)
&=(-1)^k k\gamma_{k-1}m(X)
-km(X)S_{k-1}(1)+k\int_1^X m(u)
\frac{\mathrm{d}}{\mathrm{d}u}
\bigg[S_{k-1}\bigg(\frac{X}{u}\bigg)\bigg]\ du\\
&\quad=-k(k-1)\int_1^X m(u)
S_{k-2}\bigg(\frac{X}{u}\bigg)\frac{du}{u}\ =-k(k-1)\int_1^X
m\bigg(\frac{X}{t}\bigg)S_{k-2}(t)\frac{dt}{t}.
\end{align*}
Taking absolute values and applying \eqref{S_r:bound}
with $r=k-2$ proves the result.

We have
\begin{equation*}
|m(X)|\leq
\begin{cases}
\displaystyle\frac{3}{\pi 10^{10}}+\frac{11.39}{\sqrt{X}},
&\text{if }X\geq1,
\quad\text{\cite[Cor.~1.3]{ChirreHelfgott*25a}},\\[8pt]
\displaystyle\frac{362.84}{\log^2X},
&\text{if }X\geq2,
\quad\text{\cite[Thm.~A]{Daval25}},\\[8pt]
\displaystyle\frac{\sqrt{2}}{\sqrt{X}},
&\text{if }0<X\leq10^{14},
\quad\text{\cite[Lemma~5.10]{Helfgott19}}.
\end{cases}
\end{equation*}
We readily verify that the function
\begin{equation*}
f:t>0\mapsto\frac{3t}{\pi10^{10}}+(11.39-\sqrt{2})te^{-\frac{t}{2}}
\end{equation*}
is convex on $[14\log 10,1.949\cdot 10^6]$. Moreover, 
\begin{equation*}
f(14\log 10)<0.00003217\quad\text{and}\quad f(1.949\cdot 10^6)<0.00018612.
\end{equation*}
Thus, if we suppose that $X\in (10^{14},e^{1.949\cdot 10^6}]$, then $f(\log X)<\frac{1.862}{10^4}$,
which gives \eqref{mlog^2} in that interval.
For $X\geq e^{1.949\cdot 10^6}$, we instead use
\begin{equation*}
|m(X)|
\leq\frac{362.84}{\log^2X}
\leq\frac{362.84}{1.949\cdot 10^6\log X}
<\frac{1.862}{10^4\log X}.
\end{equation*}
This proves \eqref{mlog^2} for every $X>0$.

Estimate \eqref{checklog^2} follows from
\cite[Lemma~5.9, Eq.(5.54)]{Helfgott19} for $X\leq 10^{12}$ and
\cite[Thm.~B]{Daval25} for $X\geq 10^{12}$.

For the last estimate, the exact remainder identity
in the proof of Theorem \ref{recurrence}, with $k=2$,
gives, for $X\geq1$,
\begin{equation}\label{R_0}
\check{\check m}(X)-2\log X+2\gamma
=-2\gamma(\check m(X)-1)-2\int_1^X m\bigg(\frac{X}{t}\bigg)
R_0(t)\frac{dt}{t}.
\end{equation}
Here
\begin{equation*}
\frac{|R_0(t)|}{t}
\leq W_0(t):=\frac{1}{2t^2}+\frac{1}{6t^3}.
\end{equation*}
Observe that
\begin{equation*}
\int_1^\infty\sqrt{t}\ W_0(t)\ dt=\frac{10}{9},
\qquad
\int_1^\infty W_0(t)\ dt=\frac{7}{12}
\quad\text{and}\quad
\int_1^\infty W_0(t)\log t\ dt=\frac{13}{24}.
\end{equation*}
Applying \eqref{mlog^2} into \eqref{R_0}, and using the inequality
\begin{equation}\label{inek}
\frac{1}{\log(X/t)}
\leq\frac{1}{\log X}
\bigg(1+\frac{\log t}{12\log 10}\bigg),
\qquad1\leq t\leq \frac{X}{10^{12}},
\end{equation}
we obtain
\begin{align*}
&|\check{\check m}(X)-2\log X+2\gamma|-2\gamma|\check m(X)-1|
\leq 2\int_1^X\bigg|m\bigg(\frac{X}{t}\bigg)\bigg|\frac{|R_0(t)|}{t}dt\\
&\qquad\leq \frac{2\sqrt{2}}{\sqrt{X}}\int_1^X\sqrt{t}W_0(t)dt
+2\cdot\frac{1.862}{10^4}\ \mathds{1}_{{X\geq 10^{12}}}(X)\int_1^{X/10^{12}}\frac{W_0(t)}{\log(X/t)}dt\\
&\qquad\qquad\leq \frac{2\sqrt{2}}{\sqrt{X}}\int_1^X\sqrt{t}W_0(t)dt
+2\cdot\frac{1.862}{10^4}\frac{\mathds{1}_{{X\geq 10^{12}}}(X)}{\log X}\int_1^{X/10^{12}}\bigg(1+\frac{\log t}{12\log 10}\bigg)W_0(t)dt.\end{align*}
Here the integrals with upper limit $X/10^{12}$ are omitted
when $X<10^{12}$. Now, applying \eqref{checklog^2} gives
\begin{equation*}
|\check{\check m}(X)-2\log X+2\gamma|\leq\frac{C}{\sqrt{X}}+\frac{D\ \mathds{1}_{{X\geq 10^{12}}}(X)}{\log X}
\end{equation*}
where
\begin{equation*}
C=2\gamma+\frac{20\sqrt{2}}{9},\qquad D=2\gamma\frac{8.55}{10^6}+\frac{1.862}{10^4}\bigg(\frac{7}{6}+\frac{13}{144\log 10}\bigg).
\end{equation*}
Since $C>4e^{\frac{\gamma}{2}-1}$ and $t\geq e^2\mapsto\frac{\log t}{\sqrt{t}}$ is decreasing, we have, for $X\geq 10^{12}$,
\begin{equation*}
\frac{C}{\sqrt{X}}+\frac{D}{\log X}
\leq\frac{4e^{\frac{\gamma}{2}-1}}{\sqrt{X}}
+\bigg(D+(C-4e^{\frac{\gamma}{2}-1})\frac{12\log 10}{\sqrt{10^{12}}}\bigg)\frac{1}{\log X}.
\end{equation*}
Finally,
\begin{equation*}
D+\bigg(C-4e^{\frac{\gamma}{2}-1}\bigg)
\frac{12\log 10}{10^6}
<\frac{2.98876}{10^4}<\frac{2.989}{10^4}.
\end{equation*}
For $0<X<10^{12}$, the required bound is already supplied
by \cite[Lemma~5.9, Eq.(5.56)]{Helfgott19}. This proves
\eqref{checkcheckmlog^2}.

\label{general_polynomial}
Define
\begin{align*}
U_r&=r!\bigg(4^r+\frac{14}{45}\bigg(\frac{4}{3}\bigg)^r-\frac{1}{5\cdot2^r}\bigg),\\
V_r&=r!\bigg(2^{r-1}+\frac{1}{3}-\frac{1}{2^{r+2}}+\bigg(2^{r-1}(r+1)+\frac{2r-1}{12}+\frac{1}{2^{r+3}}\bigg)\frac{1}{12\log 10}\bigg),
\qquad r\geq 1.
\end{align*}
For $k\geq3$, let
\begin{align}
A_k
&=k\sum_{j=1}^{k-1}\bigg(\genfrac{}{}{0pt}{}{k-1}{j}\bigg)
|\gamma_{k-1-j}|A_j
+k(k-1)\sqrt{2}\ U_{k-2},\label{Ak_rec}\\
B_k
&=k\sum_{j=1}^{k-1}\bigg(\genfrac{}{}{0pt}{}{k-1}{j}\bigg)
|\gamma_{k-1-j}|B_j
+\frac{1.862}{10^4}k(k-1)V_{k-2}.
\label{Bk_rec}
\end{align}
Then, for every integer $k\geq0$ and every $X>0$,
\begin{equation}\label{general_polynomial_bound}
\big|\mathbf{\check m}^{(k)}(X)-Q_k(\log X)\big|
\leq\frac{A_k}{\sqrt{X}}
+\frac{B_k\ \mathds{1}_{X\geq10^{12}}(X)}{\log X},
\end{equation}
where $Q_0=0$ and, for $k\geq1$, $Q_k$ is the
polynomial of degree $k-1$ and leading coefficient $k$
defined by \eqref{Pol_2}.

<!-- proofbold-4: body_md -->
The cases $k=0,1,2$ follow from Lemma
\ref{initial_estimates} and \eqref{AB}, since
$Q_0(t)=0$, $Q_1(t)=1$ and $Q_2(t)=2t-2\gamma$.

For $k\geq 3$ we use induction: suppose that the result holds for all smaller positive integers $1\leq j\leq k-1$ and all $X>0$.

Note now the following binomial identities
\begin{equation*}
\sum_{j=0}^{r}\bigg(\genfrac{}{}{0pt}{}{r}{j}\bigg)=2^r,\quad j\bigg(\genfrac{}{}{0pt}{}{r}{j}\bigg)=r\bigg(\genfrac{}{}{0pt}{}{r-1}{j-1}\bigg),\quad
\sum_{j=i}^{r}\bigg(\genfrac{}{}{0pt}{}{r}{j}\bigg)\frac{j!}{(j-i)!}
=\frac{r!}{(r-i)!}
\sum_{j=i}^{r}\bigg(\genfrac{}{}{0pt}{}{r-i}{j-i}\bigg)
=\frac{r!}{(r-i)!}2^{r-i}.
\end{equation*}
Consequently, the definition \eqref{Pol} of $P_j$ yield
\begin{align*}
\sum_{j=0}^{r}\bigg(\genfrac{}{}{0pt}{}{r}{j}\bigg)t^{r-j}P_j(t)
&=\frac{2^r}{6}t^r
+\frac{r2^{r-1}}{4}t^{r-1}
+\frac{1}{2}\sum_{i=2}^{r}\frac{t^{r-i}}{2^i}
\sum_{j=i}^{r}\bigg(\genfrac{}{}{0pt}{}{r}{j}\bigg)\frac{j!}{(j-i)!}\\
&=\frac{2^r}{6}t^r
+r2^{r-3}t^{r-1}
+\sum_{i=2}^{r}\frac{r!}{(r-i)!}
2^{r-2i-1}t^{r-i}\\
&=\frac{2^r}{6}t^r
+\sum_{i=1}^{r}\frac{r!}{(r-i)!}
2^{r-2i-1}t^{r-i},\qquad r\geq 1.
\end{align*}
Recall the definition \eqref{W_r} of $W_r$. Integrating term by term using
\begin{equation*}
\int_1^\infty\frac{\log^\ell t}{t^{1+\alpha}}\ dt
=\frac{\ell!}{\alpha^{\ell+1}},
\qquad \alpha>0,
\end{equation*}
and summing the resulting geometric sums gives
\begin{equation*}
\int_1^\infty\sqrt{t}\ W_r(t)\ dt=U_r,
\qquad
\int_1^\infty W_r(t)
\bigg(1+\frac{\log t}{12\log 10}\bigg)\ dt=V_r,
\end{equation*}
with
\begin{equation*}
(U_1,V_1)=\bigg(\frac{233}{54},\frac{29}{24}+\frac{103}{576\log 10}\bigg),\qquad
(U_2,V_2)=\bigg(\frac{5347}{162},\frac{109}{24}+\frac{201}{192\log 10}\bigg).
\end{equation*}
Suppose first that $X\geq1$. By estimate \eqref{mlog^2},
\begin{align*}
\int_1^X
\bigg|m\bigg(\frac{X}{t}\bigg)\bigg|W_r(t)\ dt
\leq\phantom{.}&
\frac{\sqrt{2}}{\sqrt{X}}
\int_1^X\sqrt{t}\ W_r(t)\ dt+\frac{1.862\cdot\mathds{1}_{X\geq 10^{12}}}{10^4}
\int_1^{X/10^{12}}\frac{W_r(t)}{\log(X/t)}\ dt.
\end{align*}
By recalling inequality \eqref{inek}, we then have
\begin{equation*}
\int_1^X
\bigg|m\bigg(\frac{X}{t}\bigg)\bigg|W_r(t)\ dt
\leq\frac{\sqrt{2}\ U_r}{\sqrt{X}}
+\frac{1.862\ V_r\ \mathds{1}_{{X\geq 10^{12}}}(X)}
{10^4\log X}.
\end{equation*}
Thus, Theorem \ref{recurrence} with $r=k-2$ gives
\begin{equation*}
|\Xi_k(X)|
\leq\frac{k(k-1)\sqrt{2}\ U_{k-2}}{\sqrt{X}}

\mathds{1}_{{X\geq 10^{12}}}(X)}{10^4\log X}.
\end{equation*}

On the other hand, subtracting \eqref{Pol_2} evaluated at $\log X$
from the recurrence in Theorem \ref{recurrence} gives

=-k\sum_{j=1}^{k-1}\bigg(\genfrac{}{}{0pt}{}{k-1}{j}\bigg)

+\Xi_k(X).
\end{equation*}

bound for $\Xi_k(X)$ proves
\eqref{general_polynomial_bound} for $X\geq1$,
with the constants \eqref{Ak_rec} and \eqref{Bk_rec}.

Finally, for $0<X<1$, we have $\mathbf{\check m}^{(k)}(X)=0$.
Hence, by \eqref{Pol_2} and the induction hypothesis,
\begin{align*}
\big|\mathbf{\check m}^{(k)}(X)-Q_k(\log X)\big|
&=|Q_k(\log X)|\\
&\leq k|\log X|^{k-1}
+k\sum_{j=1}^{k-1}\bigg(\genfrac{}{}{0pt}{}{k-1}{j}\bigg)
|\gamma_{k-1-j}||Q_j(\log X)|\\
&= k|\log X|^{k-1}
+k\sum_{j=1}^{k-1}\bigg(\genfrac{}{}{0pt}{}{k-1}{j}\bigg)
|\gamma_{k-1-j}||\mathbf{\check m}^{(j)}(X)-Q_j(\log X)|\\
&\leq \frac{1}{\sqrt{X}}\bigg(k\bigg(\frac{2(k-1)}{e}\bigg)^{k-1}

\end{align*}
where we have used that the function
$t\mapsto\log^{k-1}t/\sqrt{t}$ is increasing on
$[1,e^{2(k-1)}]$ and decreasing on $[e^{2(k-1)},\infty)$, so that
\begin{equation*}

\leq\bigg(\frac{2(k-1)}{e}\bigg)^{k-1},
\qquad 0<X<1,
\end{equation*}

\geq k!\sqrt{2}\ 4^{k-2}
\geq k\bigg(\frac{2(k-1)}{e}\bigg)^{k-1}.
\end{equation*}
and thus, for $0<X<1$,
\begin{align*}

\end{align*}
completing the proof.

<!-- lemma-5: body_md -->
\label{aux}
Let $q\in\mathbb{Z}_{\geq 1}$, $X>0$, and $s\in\mathbb{C}$
with $\Re(s)>0$. Then
\begin{enumerate}[label=(\alph*),ref=\alph*]
\item\label{aux_a} We have
\begin{equation*}
\sum_{d|q^\infty}\frac{1}{d^s}
=\prod_{p|q}\bigg(1-\frac{1}{p^s}\bigg)^{-1}=\frac{q^s}{\varphi_s(q)}.
\end{equation*}

\item\label{aux_b} Let $X>0$. For every integer $k\geq0$, we have
\begin{equation*}
\sum_{d|q^\infty}\frac{1}{d^s}
\log^k\left(\frac{X}{d}\right)
=
\sum_{j=0}^{k}\bigg(\genfrac{}{}{0pt}{}{k}{j}\bigg)\log^{k-j}X\ 
\frac{\mathrm{d}^{j}}{\mathrm{d}s^{j}}\bigg[\frac{q^s}{\varphi_s(q)}\bigg].
\end{equation*}
\end{enumerate}
<!-- end lemma-5 -->
<!-- proofbold-5: body_md -->
\eqref{aux_a}
Every divisor of $q^\infty$ has a unique representation $d=\prod_{p|q}p^{\alpha_p}$ with $\alpha_p\geq0$.
Since $\Re(s)>0$, the corresponding geometric series
converge absolutely. Hence
\begin{equation*}
\sum_{d|q^\infty}\frac{1}{d^s}
=\prod_{p|q}\sum_{j=0}^{\infty}\frac{1}{p^{js}}
=\prod_{p|q}\left(1-\frac{1}{p^s}\right)^{-1}
=\frac{q^s}{\varphi_s(q)}.
\end{equation*}
\noindent\eqref{aux_b} For every integer $j\geq0$, the series $
\sum_{d|q^\infty}\frac{\log^jd}{d^s}$
converges absolutely and locally uniformly in
$\Re(s)>0$. Thus, differentiating the identity in
(a) term by term gives
\begin{equation}\label{diff}
\frac{\mathrm{d}^{j}}{\mathrm{d}s^{j}}\bigg[\frac{q^s}{\varphi_s(q)}\bigg]
=(-1)^j\sum_{d|q^\infty}\frac{\log^jd}{d^s}.
\end{equation}
Now, the binomial theorem yields
\begin{equation*}
\sum_{d|q^\infty}\frac{\log^k(X/d)}{d^s}=\sum_{j=0}^{k}\bigg(\genfrac{}{}{0pt}{}{k}{j}\bigg)
\log^{k-j}X(-1)^j\sum_{d|q^\infty}\frac{\log^jd}{d^s}
\end{equation*}
By using \eqref{diff}, we derive the result.

<!-- lemma-6: body_md -->
\label{derivatives_phi}
Let $q,j\in\mathbb{Z}_{\geq 1}$. Then
\begin{align*}
\frac{\mathrm{d}^{j}}{\mathrm{d}s^{j}}
\bigg[\frac{q^s}{\varphi_s(q)}
\bigg|_{s=1}
=\phantom{.}&\sum_{i=0}^{j-1}\bigg(\genfrac{}{}{0pt}{}{j-1}{i}\bigg)(-1)^{i+1}
\frac{\mathrm{d}^{j-1-i}}{\mathrm{d}s^{j-1-i}}
\bigg[\frac{q^s}{\varphi_s(q)}
\bigg|_{s=1}\cdot\sum_{p|q}\log^{i+1}p\ 
\frac{\mathrm{d}^{i}}{\mathrm{d}z^{i}}
\bigg[\frac{1}{p/e^{z}-1}
\bigg|_{z=0}.
\end{align*}
<!-- end lemma-6 -->
<!-- proofbold-6: body_md -->
Since $\Re(s)>0$, for every $i\geq0$, we have
\begin{equation*}
\frac{\mathrm{d}^{i+1}}{\mathrm{d}s^{i+1}}
\bigg[\sum_{p|q}\sum_{\ell=1}^{\infty}
\frac{1}{\ell p^{\ell s}}\bigg]
=(-1)^{i+1}\sum_{p|q}\log^{i+1}p\
\sum_{\ell=1}^{\infty}\frac{\ell^i}{p^{\ell s}},
\end{equation*}
where term by term differentiation is justified by normal convergence.

By applying the above identity with $i=0$, we derive
\begin{align*}
\frac{\mathrm{d}}{\mathrm{d}s}
\bigg[\frac{q^s}{\varphi_s(q)}\bigg]&=\frac{\mathrm{d}}{\mathrm{d}s}\bigg[\exp\bigg(-\sum_{p|q}\log\bigg(1-\frac{1}{p^s}\bigg)\bigg)\bigg]\\
&=\frac{\mathrm{d}}{\mathrm{d}s}\bigg[\exp\bigg(\sum_{p|q}\sum_{\ell=1}^{\infty}
\frac{1}{\ell p^{\ell s}}\bigg)\bigg]
=-\frac{q^s}{\varphi_s(q)}
\sum_{p|q}\log p\sum_{\ell=1}^{\infty}
\frac{1}{p^{\ell s}}.
\end{align*}
Applying the Leibniz differentiation rule a further
$j-1$ times in the preceding identity and evaluating at $s=1$, we obtain
\begin{align}\label{diff_Z}
\frac{\mathrm{d}^{j}}{\mathrm{d}s^{j}}
\bigg[\frac{q^s}{\varphi_s(q)}\bigg|_{s=1}
&=\sum_{i=0}^{j-1}\bigg(\genfrac{}{}{0pt}{}{j-1}{i}\bigg)(-1)^{i+1}
\frac{\mathrm{d}^{j-1-i}}{\mathrm{d}s^{j-1-i}}
\bigg[\frac{q^s}{\varphi_s(q)}\bigg|_{s=1}
\cdot\sum_{p|q}\log^{i+1}p
\sum_{\ell=1}^{\infty}\frac{\ell^i}{p^\ell}.
\end{align}
Finally, for $|e^z/p|<1$, we have the series
\begin{equation*}
\frac{1}{p/e^z-1}
=\sum_{\ell=1}^{\infty}\frac{e^{\ell z}}{p^\ell},
\end{equation*}
which converge normally on $\Re z<\log p$. Differentiating $i$ times
and evaluating at $z=0$, we obtain
\begin{equation*}
\frac{\mathrm{d}^{i}}{\mathrm{d}z^{i}}
\bigg[\frac{1}{p/e^z-1}\bigg|_{z=0}
=\sum_{\ell=1}^{\infty}\frac{\ell^i}{p^\ell}.
\end{equation*}
Substituting this identity into \eqref{diff_Z}
proves the result.

<!-- theorem-5: body_md -->
\label{coprime_polynomial}
Let $q\in\mathbb{Z}_{\geq 1}$, $k\in\mathbb{Z}_{\geq 0}$ and $A_k,B_k$ be as in Theorem
\ref{general_polynomial}, and put
$\theta=1-\frac{1}{12\log 10}$.
Then, for every $X>0$,
\begin{equation}\label{coprime_polynomial_bound}
\big|\mathbf{\check m}_q^{(k)}(X)-Q_{k,q}(\log X)\big|
\leq
\frac{\sqrt{q}}{\varphi_{1/2}(q)}
\frac{A_k}{\sqrt{X}}
+\frac{q^\theta}{\varphi_\theta(q)}
\frac{B_k\ \mathds{1}_{X\geq 10^{12}}(X)}{\log X},
\end{equation}
<!-- end theorem-5 -->
<!-- proofbold-7: body_md -->
By Lemma \ref{Idd}, for every $X>0$,
\begin{equation}\label{awe}
\mathbf{\check m}_q^{(k)}(X)
=\sum_{d|q^\infty}\frac{1}{d}
\mathbf{\check m}^{(k)}\bigg(\frac{X}{d}\bigg).
\end{equation}
For $k\geq 1$, since $Q'_k(t)=kQ_{k-1}(t)$, Taylor formula gives
\begin{equation}\label{Taylor}
Q_k(t-\log d)
=\sum_{j=0}^{k-1}\bigg(\genfrac{}{}{0pt}{}{k}{j}\bigg)
Q_{k-j}(t)(-1)^j\log^jd.
\end{equation}
Subsequently, by the identity \eqref{diff}, we therefore obtain
\begin{align}\label{Qkq_alt}
\sum_{d|q^\infty}\frac{1}{d}Q_k(t-\log d)
&=\sum_{j=0}^{k-1}\bigg(\genfrac{}{}{0pt}{}{k}{j}\bigg)Q_{k-j}(t)
(-1)^j\sum_{d|q^\infty}\frac{\log^jd}{d}\nonumber\\
&\qquad=\sum_{j=0}^{k-1}\bigg(\genfrac{}{}{0pt}{}{k}{j}\bigg)Q_{k-j}(t)
\frac{\mathrm{d}^{j}}{\mathrm{d}s^{j}}
\bigg[\frac{q^s}{\varphi_s(q)}\bigg|_{s=1}
:=Q_{k,q}(t).
\end{align}
The exchange of summations above is justified by absolute convergence. Moreover, the same identity holds for $k=0$, since $Q_0=Q_{0,q}=0$. 

By using identiy \eqref{awe} and then identity \eqref{Qkq_alt} with $t=\log X$ we derive
\begin{align*}
\big|\mathbf{\check m}_q^{(k)}(X)-Q_{k,q}(\log X)\big|&=
\bigg|\sum_{d|q^\infty}\frac{1}{d}\bigg(\mathbf{\check m}^{(k)}\bigg(\frac{X}{d}\bigg)-Q_k\bigg(\log\bigg(\frac{X}{d}\bigg)\bigg)\bigg)\bigg|\\
&\leq\sum_{d|q^\infty}\frac{1}{d}\bigg|\mathbf{\check m}^{(k)}\bigg(\frac{X}{d}\bigg)-Q_k\bigg(\log\bigg(\frac{X}{d}\bigg)\bigg)\bigg|.
\end{align*}
Thus, thanks to Theorem \ref{general_polynomial}, we conclude
\begin{equation}\label{marvel}
\big|\mathbf{\check m}_q^{(k)}(X)-Q_{k,q}(\log X)\big|
\leq\frac{A_k}{\sqrt{X}}
\sum_{d|q^\infty}\frac{1}{\sqrt{d}}+B_k\mathds{1}_{X\geq 10^{12}}
\sum_{\substack{d|q^\infty\\d\leq X/10^{12}}}
\frac{1}{d\log(X/d)}.
\end{equation}
Next, for $X\geq 10^{12}$, the function
$t\mapsto\frac{t^{-(1-\theta)}}{\log(X/t)}$
is decreasing on $[1,X/10^{12}]$, where $\theta=1-\frac{1}{12\log 10}$. Consequently,
\begin{equation}\label{calculus}
\frac{1}{d\log(X/d)}
\leq\frac{1}{d^\theta\log X},
\qquad\text{ for }1\leq d\leq X/10^{12},
\end{equation}
and then Lemma \ref{aux}\eqref{aux_a} gives
\begin{equation*}
\sum_{\substack{d|q^\infty\\d\leq X/10^{12}}}
\frac{1}{d\log(X/d)}
\leq\frac{q^\theta}{\varphi_\theta(q)\log X},
\qquad
\sum_{d|q^\infty}\frac{1}{\sqrt{d}}
=\frac{\sqrt{q}}{\varphi_{1/2}(q)}.
\end{equation*}
which we use in the estimate \eqref{marvel} to obtain \eqref{coprime_polynomial_bound}.

<!-- lemma-7: body_md -->
\label{q:id}
Let $n,q,d\in\mathbb{Z}_{\geq1}$. Then
\begin{equation*}
d|n,\quad (d,q)=1,\quad \mu^2(d)=1,\quad
\frac{n}{d}|(dq)^\infty
\quad\Longleftrightarrow\quad
d=\prod_{\substack{p|n\\p\nmid q}}p.
\end{equation*}
<!-- end lemma-7 -->
<!-- proofbold-8: body_md -->
Since $\frac{n}{d}|(qd)^{\infty}$, we have $n|(dq)^{\infty}$. Since $(d,q)=1$, a prime divides $d$ if and only if it divides $n$ and does not divide $q$. We conclude by squarefreeness. 

[$\ \Longleftarrow\ $] If $d=\prod_{\substack{p|n\\p\nmid q}}p$ then $d|n$, $(d,q)=1$ and $\mu^2(d)=1$. Moreover,
\begin{equation*}
\frac{n}{d}
=\prod_{\substack{p^{\alpha}||n\\p|q}}p^{\alpha}\prod_{\substack{p^{\alpha}||n\\p\nmid q}}p^{\alpha}\prod_{\substack{p|n\\p\nmid q}}\frac{1}{p}
=\prod_{\substack{p^{\alpha}||n\\p|q}}p^{\alpha}\prod_{\substack{p^{\alpha}||n\\p\nmid q}}p^{\alpha-1}.
\end{equation*}
Thus, if $p|\frac{n}{d}$ and $p\nmid q$, then $p|d$, whence $\frac{n}{d}|(dq)^{\infty}$.

\label{tilde_convolution}
Let $k\in\mathbb{Z}_{\geq 0}$ and $q\in\mathbb{Z}_{\geq 1}$. For every $X>0$, we have
\begin{equation*}
\mathbf{\tilde m}_q^{(k)}(X)=\sum_{n=1}^{\infty}\frac{w_q(n)}{n}
\mathbf{\check m}^{(k)}
\bigg(\frac{X}{n}\bigg).
\end{equation*}
<!-- end lemma-8 -->

Since $\sigma$ is the sum-of-divisors function, we have
\begin{equation*}
\mu^2(n)\frac{n}{\sigma(n)}
=\mu^2(n)\prod_{p|n}\bigg(1-\frac{1}{p+1}\bigg)
=\mu^2(n)\sum_{d|n}\frac{\mu(d)}{\sigma(d)}.
\end{equation*}
Hence, by recalling definition \eqref{definitions}, we may use the above identity as follows
\begin{align*}
\mathbf{\tilde{m}}_q^{(k)}(X)=\sum_{\substack{n\leq X\\(n,q)=1}}\frac{\mu(n)}{\sigma(n)}\log^k\bigg(\frac{X}{n}\bigg)
&=\sum_{\substack{de\leq X\\(de,q)=1\\(d,e)=1}}
\frac{\mu^2(d)\mu(e)}{d\sigma(d)e}
\log^k\bigg(\frac{X}{de}\bigg)
=\sum_{\substack{d\\(d,q)=1}}
\frac{\mu^2(d)}{d\sigma(d)}
\mathbf{\check m}_{dq}^{(k)}
\bigg(\frac{X}{d}\bigg),
\end{align*}
where the above condition $\{d\leq X\}$ is encoded by the definition of $\mathbf{\check m}_{dq}^{(k)}(X/d)$. Now, by applying Lemma \ref{Idd} to
$\mathbf{\check m}_{dq}^{(k)}(X/d)$ gives
\begin{equation*}
\mathbf{\tilde m}_q^{(k)}(X)=\sum_{\substack{d\\(d,q)=1}}
\frac{\mu^2(d)}{d\sigma(d)}
\sum_{\ell|(dq)^\infty}\frac{1}{\ell}
\mathbf{\check m}^{(k)}\bigg(\frac{X}{d\ell}\bigg)
=\sum_{n\leq X}\frac{1}{n}
\mathbf{\check m}^{(k)}\bigg(\frac{X}{n}\bigg)
\sum_{\substack{d|n\\(d,q)=1\\\frac{n}{d}|(dq)^\infty}}
\frac{\mu^2(d)}{\sigma(d)}.
\end{equation*}
The result is concluded with the help of Lemma \ref{q:id} since
\begin{equation*}
\sum_{\substack{d|n\\(d,q)=1\\\frac{n}{d}|(dq)^\infty}}
\frac{\mu^2(d)}{\sigma(d)}
=\sum_{d=\prod_{p|n,p\nmid q}p}\frac{1}{\sigma(d)}
=\prod_{\substack{p|n\\p\nmid q}}\frac{1}{p+1}
=w_q(n).
\end{equation*}

\mathbf{\check m}^{(k)}\bigg(\frac{X}{n}\bigg)\geq 0.
\end{equation*}

<!-- lemma-9: body_md -->
\label{Hq_s}
Let $q\in\mathbb{Z}_{\geq1}$. For $\Re(s)>0$, define 
$H_q(s)
:=\sum_{n=1}^{\infty}\frac{w_q(n)}{n^s}$. Then
\begin{equation}\label{tilde_euler_product}
H_q(s)
=\frac{q^s}{\varphi_s(q)}
\prod_{p\nmid q}
\bigg(1+\frac{1}{(p+1)(p^s-1)}\bigg)
=\mathpzc p_s(q)
\prod_p\bigg(1+\frac{1}{(p+1)(p^s-1)}\bigg)
\end{equation}
where $\mathpzc p_s(q)$ is defined in \eqref{functions}.
The series and product defining $H_q$ converge normally in the
half-plane $\Re s>0$. In particular, we have
\begin{equation}\label{H_q(1)}
H_q(1)=\zeta(2)\mathpzc p_1(q).
\end{equation}
<!-- end lemma-9 -->
<!-- proofbold-11: body_md -->
The function $w_q$ is multiplicative and for $\alpha\geq 1$, we have 
\begin{equation*}
\frac{w_q(p^\alpha)}{p^{\alpha s}}=\frac{1}{p^{\alpha s}}\quad\text{ if }\ p|q\qquad\text{ and }\qquad \frac{w_q(p^\alpha)}{p^{\alpha s}}=\frac{1}{p^{\alpha s}(p+1)},\quad\text{ if }\ p\nmid q.
\end{equation*}
Since there is only a finite number of primes that divide $q$, we conclude from the second expression above that the series defining $H_q(s)$ converges normally in $\{\Re s>0\}$. Further, in that region, we have the following Euler product
\begin{equation*}
H_q(s)=\prod_{p|q}\bigg(\sum_{\alpha\geq 0}\frac{1}{p^{\alpha s}}\bigg)\prod_{p\nmid q}\bigg(1+\frac{1}{(p+1)}\sum_{\alpha\geq 1}\frac{1}{p^{\alpha s}}\bigg)
=\frac{q^s}{\varphi_s(q)}
\prod_{p\nmid q}
\bigg(1+\frac{1}{(p+1)(p^s-1)}\bigg).
\end{equation*}
Rearranging the Euler product gives
\begin{equation}\label{budew}
H_q(s)
=\mathpzc p_s(q)
\prod_p\bigg(1+\frac{1}{(p+1)(p^s-1)}\bigg),
\end{equation}
with $\mathpzc p_s(q)$ defined in \eqref{functions}. Finally, at $s=1$, we have
\begin{equation*}
H_q(1)
=\frac{q}{\varphi(q)}
\prod_{p\nmid q}\bigg(1-\frac{1}{p^2}\bigg)^{-1}
=\zeta(2)\mathpzc p_1(q).
\end{equation*}

<!-- theorem-6: body_md -->
\label{tilde_general_bound}
Let $q\in\mathbb{Z}_{\geq 1}$ and $k\in\mathbb{Z}_{\geq 0}$. Let $A_k,B_k$ be as in Theorem
\ref{general_polynomial} and put $\theta=1-\frac{1}{12\log 10}$.
Then, for every $X>0$,
\begin{equation*}
\big|\mathbf{\tilde m}_q^{(k)}(X)
-\widetilde Q_{k,q}(\log X)\big|
\leq
\mathpzc p_{1/2}(q) 
\frac{\mathit P_{1/2}A_k}{\sqrt{X}}
+\mathpzc p_\theta(q) 
\frac{\mathit  P_\theta B_k\ \mathds{1}_{X\geq 10^{12}}(X)}{\log X},
\end{equation*}
where
\begin{equation}\label{values}
\mathit P_{1/2}=3.575\quad\text{ and }\quad
\mathit P_\theta=1.693.
\end{equation}
<!-- end theorem-6 -->
<!-- proofbold-12: body_md -->
For $k\geq1$, applying identity \eqref{Taylor} gives
\begin{align}\label{tilde_polynomial_convolution}
&\sum_{n=1}^{\infty}\frac{w_q(n)}{n}Q_k(t-\log n)=\sum_{n=1}^{\infty}\frac{w_q(n)}{n}\sum_{j=0}^{k-1}\bigg(\genfrac{}{}{0pt}{}{k}{j}\bigg)
Q_{k-j}(t)(-1)^j\log^jn\\
&=\sum_{j=0}^{k-1}\bigg(\genfrac{}{}{0pt}{}{k}{j}\bigg)
Q_{k-j}(t)\sum_{n=1}^{\infty}\frac{w_q(n)}{n}(-1)^j\log^jn=\sum_{j=0}^{k-1}\bigg(\genfrac{}{}{0pt}{}{k}{j}\bigg)
Q_{k-j}(t)\frac{\mathrm{d}^j}{\mathrm{d}s^j}\bigg[\sum_{n=1}^{\infty}\frac{w_q(n)}{n^s}\bigg|_{s=1}:=\widetilde Q_{k,q}(t).\nonumber
\end{align}
The exchange of summation and the term by term differentiation are justified since the series $H_q(s)$ defined in Lemma \ref{Hq_s} and all its derivatives converge normally in $\{\Re(s)>0\}$. Moreover, the same identity \eqref{tilde_polynomial_convolution} holds for $k=0$, since both polynomials are identically zero.

On the other hand, by Lemma \ref{tilde_convolution}, identity
\eqref{tilde_polynomial_convolution} with $t=\log X$ and Theorem
\ref{general_polynomial}, for every $X>0$ and $k\in\mathbb{Z}_{\geq 0}$, we obtain
\begin{align*}
\big|\mathbf{\tilde m}_q^{(k)}(X)
-\widetilde Q_{k,q}(\log X)\big|
&\leq
\sum_{n=1}^{\infty}\frac{w_q(n)}{n}
\bigg|\mathbf{\check m}^{(k)}
\bigg(\frac{X}{n}\bigg)
-Q_k\bigg(\log\frac{X}{n}\bigg)\bigg|\\
&\leq
\frac{A_k}{\sqrt{X}}
\sum_{n=1}^{\infty}\frac{w_q(n)}{\sqrt{n}}
+\mathds{1}_{X\geq 10^{12}}\ B_k\sum_{n\leq X/10^{12}}
\frac{w_q(n)}{n\log(X/n)}.
\end{align*}
Moreover, the same analysis to obtain inequality \eqref{calculus} gives
\begin{align*}
\big|\mathbf{\tilde m}_q^{(k)}(X)
-\widetilde Q_{k,q}(\log X)\big|
&\leq
\frac{A_k}{\sqrt{X}}
\sum_{n=1}^{\infty}\frac{w_q(n)}{\sqrt{n}}
+\frac{\mathds{1}_{X\geq 10^{12}}\ B_k}{\log X}\sum_{n=1}^{\infty}
\frac{w_q(n)}{n^{\theta}}\\
&=\frac{A_kH_q(\frac{1}{2})}{\sqrt{X}}
+\frac{B_kH_q(\theta)\
\mathds{1}_{X\geq 10^{12}}(X)}{\log X}.
\end{align*}
The product estimates in the proof of
\cite[Lemma 3.5]{2ZA20} and the expression \eqref{budew} yield
$H_q\big(\frac{1}{2}\big)
\leq\mathpzc p_{1/2}(q) \mathit P_{1/2}$, and
$H_q(\theta)
\leq\mathpzc p_\theta(q) \mathit  P_\theta$,
with $\mathit P_{1/2}$ and $\mathit P_{\theta}$ defined in \eqref{values}. This gives the inequality of the statement.
