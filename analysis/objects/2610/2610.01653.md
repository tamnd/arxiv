<!-- theorem-1: body_md -->
\label{thm:main}
Every four-dimensional parallelohedron $P$ satisfies
\[
  \mathcal I(P)\ge2^{11/4}.
\]
Equality holds if and only if $P$ is similar to the regular $24$-cell.
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{cor:global-stability}
Let
\[
 d_{24}(P)=\inf_{U\in \mathrm{O}(4)}d_H(\widehat P,U\widehat C_{24}),
\]
where $U$ ranges over group $\mathrm{O}(4)$ of orthogonal $4\times4$ matrices, $d_H$ is Hausdorff distance.
There are constants $C,r>0$ such that every four-dimensional parallelohedron $P$ with $\operatorname{def}(P)\le1$ satisfies
\[
 d_{24}(P)\le C\sqrt{\operatorname{def}(P)}.
\]
If $d_{24}(P)<r$, then
\[
 \frac{\sqrt2}{8}d_{24}(P)^2-Cd_{24}(P)^3
 \le\operatorname{def}(P)\le C d_{24}(P)^2.
\]
The coefficient $\sqrt2/8$ is optimal as $d_{24}(P)\to0$.
Thus the deficit is locally equivalent to squared Hausdorff distance modulo similarities, with optimal exponent $1/2$ in the first estimate.
Theorem~\ref{thm:sharp-Hausdorff-coefficient} identifies all extremal first-order deformations.
<!-- end corollary-1 -->
<!-- corollary-2: body_md -->
\label{cor:fixed-inradius}
Let $P$ be a convex fundamental domain for a lattice in $\mathbb R^4$.

<!-- theorem-3: body_md -->
\label{thm:zonotope-gap}

\end{equation*}

 \le Z_{\rm tree}\det\Sigma\le\frac{Z_{\rm tree}}{256},

\end{equation*}

Every independent quotient triple extends to a basis~\cite{McMullen1975,DezaGrishukhin2004}.

Raising to $4/3$ and using the normalized volume bound proves the claim.
Finally, $5^{1/6}>13/10$, $\sqrt3>19/11$, and $(1+2\sqrt3)^4=217+104\sqrt3$ put it above $113438/55>2048$.

<!-- theorem-4: body_md -->
\label{lem:balanced-entropy-bridge}
Let $d\ge2$, let $H>0$ be a real symmetric $d\times d$ matrix, and let $\mathcal N$ be a finite indexed family of unit vectors with nonnegative weights satisfying cofactor balance.
Write
\[
 {D_H}:=\det H\quad\text{and}\quad \sum_{e\in\mathcal N}w_e\Pi_e=\operatorname{adj}H.
\]
Let $\kappa_e>0$, and define
\[
 h_e=e^{\mathsf T}He,\quad
 \lambda_e=\frac{w_eh_e}{{D_H}},\quad
 E=\frac1{D_H}\sum_e w_e\operatorname{ent}_e(H),\quad
 F=E-\sum_e\lambda_e\log\kappa_e,
\]
\[
 W_\kappa=\sum_e\frac{w_e}{\sqrt{\kappa_eh_e}}\Pi_e ,
 \quad\text{and}\quad j_\kappa=\frac{\det W_\kappa}{{D_H}^{d-3/2}}.
\]
Write $\mathcal B$ for the $d$-element bases of positive-weight vectors in $\mathcal N$, and set
\[
 \mu(B)=\frac{c_B}{{D_H}^{d-1}}\quad\text{and}\quad
 \nu(B)=\frac{c_B\prod_{e\in B}(\kappa_eh_e)^{-1/2}}
                    {\det W_\kappa},
\]
where $c_B=\det[e:e\in B]^2\prod_{e\in B}w_e$.
These are probability laws with the same support, and we use $\operatorname{KL}(\mu\|\nu)=\sum_B\mu(B)\log(\mu(B)/\nu(B))$ for their relative entropy.
Then
\begin{equation*}
 E=\log {D_H}-\sum_e\lambda_e\log h_e\ge0
 \quad\text{and}\quad
 \log j_\kappa=\frac F2+\operatorname{KL}(\mu\|\nu).
\end{equation*}
For $|\det L|=1$, define
\[
 \operatorname{surf}_H(L)=\frac{{D_H}^{-1+1/(2d)}}d
       \sum_e\kappa_e^{-1/2}w_e
                         \|L^{-\mathsf T}H^{1/2}e\|.
\]
Write $L^{-\mathsf T}=US$ with $S>0$ and $\det S=1$, and define
\[
 n_e=\frac{H^{1/2}e}{\sqrt{h_e}},\quad
 w_e^{\mathrm{surf}}=\frac{w_e\sqrt{h_e}}{\sqrt{\kappa_e}},\quad
 \widehat W_\kappa=\sum_e w_e^{\mathrm{surf}}\Pi_{n_e}=H^{1/2}W_\kappa H^{1/2},
\]
\[
 T=\frac{\widehat W_\kappa^{1/2}S\widehat W_\kappa^{1/2}}
                {(\det\widehat W_\kappa)^{1/d}},\quad\text{and}\quad
 g_{\mathrm{ang}}=\frac{\sum_e w_e^{\mathrm{surf}}(\|Sn_e\|-n_e^{\mathsf T}Sn_e)}
                  {\operatorname{tr}(S\widehat W_\kappa)}.
\]
Then $\det T=1$, $g_{\mathrm{ang}}\ge0$, and the exact identity
\begin{equation*}
 \log\operatorname{surf}_H(L)
 =\frac{F}{2d}+\frac{\operatorname{KL}(\mu\|\nu)}d
       +\log\frac{\operatorname{tr}T}{d}+\log(1+g_{\mathrm{ang}})
\end{equation*}
decomposes the surface deficit into compensated entropy and three nonnegative terms.
In particular,
\[
 \operatorname{surf}_H(L)\ge j_\kappa^{1/d}\ge\exp(F/(2d)).
\]
<!-- end theorem-4 -->
<!-- proof-8: body_md -->
Tracing cofactor balance against $H$ and $H\log H$ gives
\[
 \sum_e\lambda_e=d\quad\text{and}\quad
 \sum_e w_e e^{\mathsf T}H\log H\,e={D_H}\log {D_H}.
\]
Spectral Jensen gives $E\ge0$, and Cauchy--Binet normalizes $\mu,\nu$ since $\operatorname{adj}H>0$.
Differentiating the weights independently before imposing balance gives
\[
 \Pr_\mu(e\in B)
 =w_e\frac{\partial}{\partial w_e}
       \log\det\Bigl(\sum_f w_f\Pi_f\Bigr)
 =w_e e^{\mathsf T}(\operatorname{adj}H)^{-1}e
 =\lambda_e.
\]
Hence
\[
 \operatorname{KL}(\mu\|\nu)
 =\log\det W_\kappa-(d-1)\log {D_H}
                  +\frac12\sum_e\lambda_e\log(\kappa_eh_e)
 =\log j_\kappa-\frac F2,
\]
which proves the entropy identities.

For unit $n$, Cauchy--Schwarz gives $\|Sn\|\ge n^{\mathsf T}Sn>0$, hence $g_{\mathrm{ang}}\ge0$.
Since $\det\widehat W_\kappa=D_H\det W_\kappa$, the surface formula becomes
\[
 \operatorname{surf}_H(L)
   =j_\kappa^{1/d}\frac{\operatorname{tr}T}{d}(1+g_{\mathrm{ang}}).
\]
Taking logarithms and using $\operatorname{tr}T\ge d$ proves the decomposition and bounds.

<!-- remark-1: body_md -->

<!-- lemma-5: body_md -->
\label{lem:marked-rank-one}
Fix a unit vector $a$, a positive semidefinite $H_-$, and $H(R)=H_-+R\Pi_a>0$ on an open interval of positive $R$.
Suppose a fixed finite family $\mathcal N$ of unit vectors has nonnegative cofactor weights satisfying balance, with $w_e=R\omega_e+\zeta_e$ for $e\perp a$ and $w_e$ independent of $R$ otherwise.
Here $\omega_e,\zeta_e\ge0$ are fixed.
Fix marks $\kappa_e>0$ and use the entropy notation of Theorem~\ref{lem:balanced-entropy-bridge}.
In the orthogonal splitting $\mathbb Ra\oplus a^\perp$, write
\[
 {H_-}=\begin{pmatrix}g_{00}&g^{\mathsf T}\\g&{H_\perp}\end{pmatrix},
 \quad {D_\perp}=\det {H_\perp},\quad
 {h_{\mathrm S}}=g_{00}-g^{\mathsf T}{H_\perp}^{-1}g.
\]
Thus ${H_\perp}>0$, ${h_{\mathrm S}}\ge0$, $\det {H_-}={D_\perp}{h_{\mathrm S}}$, and ${D_H}={D_\perp}(R+{h_{\mathrm S}})$.
For $e\not\perp a$, set
\[
 \alpha_e=(e\cdot a)^2,\qquad
 \beta_e=\frac{e^{\mathsf T}{H_-}e}{\alpha_e},\qquad
 p_e^{\mathrm{dir}}=\frac{w_e\alpha_e}{{D_\perp}}.
\]
Then $p^{\mathrm{dir}}$ is a probability law and $\beta_e\ge{h_{\mathrm S}}$.
For perpendicular normals set $h_e=e^{\mathsf T}{H_\perp}e$ and
\[
 E_\perp=\sum_{e\perp a}\omega_e\operatorname{ent}_e({H_\perp})
 \quad\text{and}\quad
 E_{\mathrm{pen},\perp}=\sum_{e\perp a}\omega_eh_e\log\kappa_e.
\]
The compensated quantity $F_{\mathrm{det}}(R)={D_H}(R)F(R)$ satisfies the exact identity
\[
 F_{\mathrm{det}}'(R)
 =E_\perp-E_{\mathrm{pen},\perp}
   -{D_\perp}\sum_{e\not\perp a}p_e^{\mathrm{dir}}\log(\alpha_e\kappa_e)
 -{D_\perp}\sum_{e\not\perp a}p_e^{\mathrm{dir}}
                    \log\frac{R+\beta_e}{R+{h_{\mathrm S}}}.
\]
If $\alpha_e\kappa_e=e^{-b}$ on positive-weight nonperpendicular normals, the angular term is $D_\perp b$.

Define
\[
 {N_{\perp,0}}=\sum_{e\perp a}\zeta_eh_e\quad\text{and}\quad
 Q=R{D_\perp}-(d-2)\det {H_-}+{N_{\perp,0}}.
\]
One always has $Q\le {D_H}$ and
\[
 \sum_{e\not\perp a}p_e^{\mathrm{dir}}
       \frac{R+\beta_e}{R+{h_{\mathrm S}}}=2-\frac Q{D_H}.
\]
Thus Jensen bounds the final logarithmic average by $\log(2-Q/D_H)$.
No lower bound for $Q$ is asserted.
<!-- end lemma-5 -->
<!-- proof-10: body_md -->
Since $H_\perp>0$, Schur completion gives the determinant formulas and $\beta_e\ge h_{\mathrm S}$.
Differentiating cofactor balance gives
\[
 \sum_{e\perp a}\omega_e\Pi_e=\operatorname{adj}{H_\perp}
 \quad\text{on }\; a^\perp.
\]
The $a,a$ entry gives $\sum_{e\not\perp a}w_e\alpha_e=D_\perp$, normalizing $p_e^{\mathrm{dir}}$.
Transverse traces give
\[
 \sum_{e\perp a}\omega_eh_e=(d-1){D_\perp}\quad\text{and}\quad
 E_\perp={D_\perp}\log {D_\perp}-\sum_{e\perp a}\omega_eh_e\log h_e.
\]

Differentiate $F_{\mathrm{det}}=D_H\log D_H-\sum_e w_eh_e\log(\kappa_eh_e)$ using $D_H'=D_\perp$ and $h'_e=0$ or $\alpha_e$ according as $e\perp a$ or not.
The constant terms cancel, leaving
\[
 F_{\mathrm{det}}'
 =D_\perp\log D_H-\sum_{e\perp a}\omega_eh_e\log(\kappa_eh_e)
 -D_\perp\sum_{e\not\perp a}p_e^{\mathrm{dir}}
                       \log[\kappa_e\alpha_e(R+\beta_e)].
\]
Substitute the transverse trace identity and $D_H=D_\perp(R+h_{\mathrm S})$ to obtain the marked rank-one identity.

Tracing balance against $H$ and subtracting the transverse contribution yields
\[
 D_\perp\sum_{e\not\perp a}p_e^{\mathrm{dir}}(\beta_e-h_{\mathrm S})
 =(d-1)D_\perp h_{\mathrm S}-N_{\perp,0}=D_H-Q\ge0.
\]
This gives the mean ratio and $Q\le D_H$.
Jensen is an equality precisely when $\beta_e$ is constant on positive-weight nonperpendicular normals.

<!-- theorem-5: body_md -->
\label{thm:portable-global-comparison}
Let $d\ge2$, let $a_1,\ldots,a_N\in\mathbb R^d$ be unit vectors, and define
\[
 H(t,x)=t\mathrm{Id}+\sum_{i=1}^Nx_i\Pi_{a_i}>0,
 \quad\text{with}\quad t,x_i\ge0.
\]
Suppose a fixed finite family of unit vectors has nonnegative cofactor weights $w_e(t,x)$ and fixed marks $\kappa_e>0$ satisfying balance.
Assume the weights are continuous, homogeneous of degree $d-1$, $C^1$ in positive $x$-coordinates on each fixed-support face, and satisfy Lemma~\ref{lem:marked-rank-one} in each positive $x_i$.
Let $F$ be the corresponding compensated entropy, and require $w_e(t,0)>0\Rightarrow\kappa_e=1$.
For each largest positive coordinate $R=x_i$, define the transverse budget
\[
 b_i^{\mathrm{ent}}=\frac{E_\perp-E_{\mathrm{pen},\perp}}{{D_\perp}}
       -\sum_{e\not\perp a_i}p_e^{\mathrm{dir}}\log(\alpha_e\kappa_e).
\]
Let $b_i^{\mathrm{ent}}\ge\underline b_i^{\mathrm{ent}}$ and $Q/D_H\ge\underline q_i$ at every such coordinate.
If either test
\begin{equation*}
 \underline b_i^{\mathrm{ent}}\ge\log(2-\underline q_i)
 \quad\text{or}\quad
 \underline b_i^{\mathrm{ent}}-\log2+\frac{\underline q_i}{2}\ge0
\end{equation*}
holds at each point and coordinate, then the compensated entropy $F$ is nonnegative, including on the positive-definite zero-core boundary.
If the chosen test is strict at every such point and coordinate, $F=0$ precisely at $x=0$.

If the cofactor family is the actual facet data of a lattice of covolume $v_\Lambda$, with
\[
 \kappa_e=(\ell_e/\ell_0)^2
 \quad\text{and}\quad \mathcal I_0=2dv_\Lambda^{1/d}/\ell_0,
\]
then $\mathcal I_d(LV_H)\ge \mathcal I_0$ for every $|\det L|=1$.
Under the strict tests, equality holds precisely at the core with orthogonal $L$.
If, on $t=0$, the left-minus-right margin in either chosen test is uniformly at least $\varepsilon>0$, then $F\ge\varepsilon$.
For actual facet data this yields
\begin{equation*}
 \mathcal I_d(LV_H)\ge \mathcal I_0e^{\varepsilon/(2d)}.
\end{equation*}

<!-- proof-11: body_md -->
Write $F_{\mathrm{det}}={D_H}F$.
Lemma~\ref{lem:marked-rank-one} and Jensen give

 \partial_{x_i}F_{\mathrm{det}}
 \ge {D_\perp}\{b_i^{\mathrm{ent}}-\log(2-Q/{D_H})\}
 \ge {D_\perp}\{\underline b_i^{\mathrm{ent}}-\log(2-\underline q_i)\}.
\]
Since $\underline q_i\le Q/D_H\le1$, both logarithms are defined, and $\log(2-q)\le\log2-q/2$ gives the linear test.

Cap $x_i(r)=\min\{x_i,r\}$ for $0<r\le\max_i x_i$.

Exposing a vertex of the projection of $Q+[-z,z]$, $z\ne0$, to $z^\perp$ gives an edge parallel to $z$.
Every two-face containing that edge has $[-z,z]$ as a summand and hence two parallel edges, contradicting the triangular two-faces of $C_{24}$.

Let $P_j\to P_0$ and set $\widetilde P_j=2^{1/4}P_j\to C_{24}$.
Classification allows a fixed cone type along a subsequence.
The principal type is impossible: its at most ten centered segment summands lie in $\widetilde P_j$, so their bounded generators would converge to a zonotope representation of $C_{24}$.

In either nonprincipal type, Lemma~\ref{lem:facet-cofactors} gives
\[
  \widetilde P_j
  =A_j\left(t_jC_{24}+\frac12\sum_{i=1}^9x_{ji}[-a_i,a_i]\right),
  \quad\text{with}\quad t_j,x_{ji}\ge0,

Let $T_j=t_jA_j$ and $z_{ji}=x_{ji}A_ja_i$.
The summands lie in $\widetilde P_j$, so $T_j,z_{ji}$ are bounded and their subsequential limits satisfy
\[
  C_{24}=TC_{24}+\frac12\sum_{i=1}^9[-z_i,z_i].

\[
 \operatorname{cell}(T,y)=T\left(C_{24}+\frac12\sum_i y_i[-a_i,a_i]\right),
 \quad\text{with}\quad y_i\ge0.
\]
We claim, near $(\mathrm{Id},0)$,
\begin{equation}\label{eq:24-cell-inverse-support}
 \|T-\mathrm{Id}\|+\sum_i y_i
 \le C d_H\bigl(\operatorname{cell}(T,y),C_{24}\bigr).
\end{equation}
Otherwise, some $T_j\to \mathrm{Id}$, $y_j\to0$ satisfy
\[
 \varepsilon_j=\|T_j-\mathrm{Id}\|+\sum_i y_{ji}>0
 \quad\text{and}\quad d_H\bigl(\operatorname{cell}(T_j,y_j),C_{24}\bigr)=o(\varepsilon_j).

Normalize $(T_j-\mathrm{Id},y_j)$ by $\varepsilon_j$ and pass to a limit $(B,z)$, with $z_i\ge0$ and $\|B\|+\sum_i z_i=1$.

 u\cdot Bv+\frac12\sum_i z_i|a_i\cdot u|=0,

The Minkowski formula gives $P=\widehat{LV_H}$, and smoothness of the polar logarithm bounds $\|H-\mathrm{Id}\|+\|Y\|$ by $C(\|T-T_0\|+\sum_i y_i)$, as required.

\label{thm:sharp-Hausdorff-coefficient}
There are constants $C,r>0$ such that every four-dimensional parallelohedron with $d_{24}(P)<r$ satisfies
\begin{equation}\label{eq:sharp-Hausdorff-cubic}
 \operatorname{def}(P)\ge
 \frac{\sqrt2}{8}d_{24}(P)^2-Cd_{24}(P)^3.
\end{equation}
The coefficient is optimal:
\begin{equation*}
 \lim_{r\downarrow0}\inf_{0<d_{24}(P)\le r}
 \frac{\operatorname{def}(P)}{d_{24}(P)^2}
 =\frac{\sqrt2}{8}.
\end{equation*}
An extremal family is
\[
 K_{\rm ext} =\operatorname{diag}(12,10,10,0),\quad
 Y_{\rm ext} =\operatorname{diag}(4,-1,-1,-2),\quad
 P_\tau=\widehat{e^{-\tau Y_{\rm ext}}V_{\mathrm{Id}+\tau K_{\rm ext}}},
\]
with $\tau>0$.
For $0<r_P:=d_{24}(P)<r$, define
\[
 \varepsilon(P)=\max\left\{
 \frac{\operatorname{def}(P)}{r_P^2}-\frac{\sqrt2}{8},0\right\}
 \quad\text{and}\quad \tau=\frac{r_P}{8\cdot 2^{-1/4}}.
\]
Then, after increasing $C$ if necessary,
\begin{equation*}
 \inf_{U\in \mathrm{O}(4)}d_H(\widehat P,UP_\tau)
 \le C r_P\sqrt{\varepsilon(P)+r_P}.
\end{equation*}
Hence every sequence with $d_{24}(P_j)\to0$ and $\operatorname{def}(P_j)/d_{24}(P_j)^2\to\sqrt2/8$ lies within $o(d_{24}(P_j))$ of this family, up to rotation.

Let $c_{\rm vol}=2^{-1/4}$, so $\widehat C_{24}=c_{\rm vol}C_{24}$.
In either nonprincipal cone, write
\[
 K=\sum_i z_i\Pi_{a_i},\quad s=\sum_i z_i,\quad X=\operatorname{tf} K,
 \quad\text{with }z_i\ge0,\quad Y\in\mathrm{Sym}_0(\mathbb R^4).
\]
After rotation, every sufficiently close unit-volume parallelohedron has the form
\begin{equation*}
 P(z,Y)=c_{\rm vol}\det(\mathrm{Id}+K)^{-1/8}e^{-Y}(\mathrm{Id}+K)^{-1/2}
 \left(C_{24}+\frac12\sum_i z_i[-a_i,a_i]\right).
\end{equation*}
Lemma~\ref{lem:24-cell-local-converse} gives such coordinates with
\begin{equation}\label{eq:intrinsic-small-coordinates}
 r_{\rm par}:=\|z\|+\|Y\|\le C_1d_{24}(P).
\end{equation}
The traceless projector blocks are orthogonal, with Gram matrix $G=\mathrm{Id}_3-\frac14\1\1^{\mathsf T}>0$ on each retained triple.
Thus the local converse controls the weights, and division by the core coefficient gives the representation.
Trace normalization replaces $X$ by $X/(1+s/4)=X+O(r_{\rm par}^2)$, so the second variation gives
\begin{equation}\label{eq:intrinsic-quadratic-energy}
 \operatorname{def}(P(z,Y))=\operatorname{cost}(z,Y)+O(r_{\rm par}^3),
\end{equation}
where
\[
\operatorname{cost}(z,Y)=\frac1{20}\|X\|^2+
              \frac5{24}\|Y_{\rm res}\|^2
 \quad\text{and}\quad Y_{\rm res}=Y-X/10.
\]
The positive frame Gram matrices make $\operatorname{cost}$ positive definite.

\smallskip\noindent
\emph{Support displacement.}
Let $N_v$ be the closed normal cone at a vertex $v$ of $C_{24}$.
Define
\begin{gather*}
 B=-Y-\frac12K-\frac{s}{8}\mathrm{Id}
   =-Y_{\rm res}-\frac35K-\frac{s}{10}\mathrm{Id},\qquad
 g_z(u)=\frac12\sum_i z_i|a_i\cdot u|,\\
 \operatorname{disp}(z,Y)=\max_{v\in\operatorname{vert}C_{24}}
       \sup_{u\in N_v,\,\|u\|=1}|u\cdot Bv+g_z(u)|.
\end{gather*}
The closed cones retain every incident vertex at a wall.
We claim
\begin{equation}\label{eq:intrinsic-support-upper}
 d_H(P(z,Y),c_{\rm vol}C_{24})\le c_{\rm vol}\operatorname{disp}(z,Y)+O(r_{\rm par}^2).
\end{equation}
The prefactor divided by $c_{\rm vol}$ is $\mathrm{Id}+B+O(r_{\rm par}^2)$, and $g_z((\mathrm{Id}+B)^{\mathsf T}u)=g_z(u)+O(r_{\rm par}^2)$ for unit $u$.
With an active vertex $v_t$ along $w_t=u+tB^{\mathsf T}u$, absolute continuity of the piecewise linear support function gives
\[
 h_{C_{24}}((\mathrm{Id}+B)^{\mathsf T}u)-h_{C_{24}}(u)+g_z(u)
 =\int_0^1\bigl(u\cdot Bv_t+g_z(u)\bigr)\,dt.
\]
Since $w_t\in N_{v_t}$, the integrand has absolute value at most $\|w_t\|\operatorname{disp}(z,Y)+O(r_{\rm par}^2)=\operatorname{disp}(z,Y)+O(r_{\rm par}^2)$.
Taking the supremum over unit $u$ gives the claimed Hausdorff bound.

\smallskip\noindent
\emph{Sharp comparison.}
We prove
\begin{equation*}
 \operatorname{disp}(z,Y)^2\le8\operatorname{cost}(z,Y).
\end{equation*}
Orient the frame containing $v$ so that $v=e_1$, and write $u=(h,b)\in N_v$. The coordinate and half-sign vertices give $h\ge\|b\|_1$.
The functional $u\cdot Bv+g_z(u)$ is
\[
 -u\cdot Y_{\rm res}v+\sum_i\ell_i z_i,
 \quad\text{where }\ell_i=\frac12|a_i\cdot u|
       -\frac35(a_i\cdot u)(a_i\cdot v)-\frac1{10}h.
\]
Because $G^{-1}=\mathrm{Id}_3+\1\1^{\mathsf T}$ and the frame blocks are orthogonal, its squared dual norm for $\operatorname{cost}$ is
\begin{equation*}
 \operatorname{dual}_v(u)=\frac{24}{5}\left\|\operatorname{tf}
       \frac{uv^{\mathsf T}+vu^{\mathsf T}}2\right\|^2
       +20\sum_{j=1}^3\ell_{T_j}^{\mathsf T}G^{-1}\ell_{T_j}.
\end{equation*}
The first term is $(18h^2+12\|b\|^2)/5$, and the dual norm also bounds the functional on $z_i\ge0$.
In either other frame, orient $a=(1,\varepsilon)/2$. Then $\ell_a=\varepsilon\cdot b/10$, and orthogonality gives
\[
 \sum_{a\in\mathrm{fr}_j}\ell_a=0\quad\text{and}\quad
 \sum_{a\in\mathrm{fr}_j}\ell_a^2=\frac{\|b\|^2}{25}.
\]
Omitting any one vector therefore leaves $\ell_{T_j}^{\mathsf T}G^{-1}\ell_{T_j}=\|b\|^2/25$.

If $v$ is retained, write $(p,q,r)$ for the absolute coordinates of $b$, with $r$ along the omitted vector.
The remaining block has coefficients $(-h/5,-h/10+p/2,-h/10+q/2)$, giving
\[
 \operatorname{dual}_v(u)=8h^2-10h(p+q)+14(p^2+q^2)+10pq+4r^2.
\]
Since $h\ge p+q+r$,
\[
 8\|u\|^2-\operatorname{dual}_v(u)
 \ge4(p^2+q^2)+10pq+10r(p+q)+4r^2\ge0,
\]
with equality only when $b=0$.
If $v$ is omitted, let $\ell=\|b\|_1$.
Its own-frame coefficients are $-h/10+|b_i|/2$, and the same calculation gives
\[
 \operatorname{dual}_v(u)=6h^2-8h\ell+9\|b\|^2+5\ell^2
 \le6\|u\|^2,
\]
since $h\ge\ell\ge\|b\|$.
Cauchy--Schwarz proves the sharp comparison, and \eqref{eq:intrinsic-support-upper}, \eqref{eq:intrinsic-small-coordinates}, and \eqref{eq:intrinsic-quadratic-energy} give
\[
 d_{24}(P)^2\le8c_{\rm vol}^2\operatorname{def}(P)+O(d_{24}(P)^3),
\]
proving the sharp lower bound.

\smallskip\noindent
\emph{Extremal family.}
Equality requires a retained $v$ and $u=v$. Choose $v=a_1=e_1$ and omitted vector $e_4$.
Dual Cauchy--Schwarz and $z_i\ge0$ force a positive multiple of
\[
 (z_1,z_2,z_3)=(12,10,10),\quad z_i=0\ (i>3),\quad
 Y_{\rm res}=\frac{24}{5}\operatorname{tf} \Pi_{e_1}.
\]
This gives $K_{\rm ext},Y_{\rm ext}$, with $\operatorname{cost}=8$ and radial displacement $-8$ at $u=v=e_1$.
Thus $\operatorname{disp}=8$ and
\begin{equation*}
 \operatorname{def}(P_\tau)=8\tau^2+O(\tau^3)\quad\text{and}\quad
 d_{24}(P_\tau)=8c_{\rm vol}\tau+O(\tau^2).
\end{equation*}
A minimizing rotation satisfies $d_H(U_\tau C_{24},C_{24})=O(\tau)$. After right multiplication by the inverse of the nearby symmetry, \eqref{eq:24-cell-inverse-support} with no segments gives $U_\tau=\mathrm{Id}+O(\tau)$.
Since $e_1$ exposes a unique vertex of $C_{24}$,
\[
 h_{U_\tau C_{24}}(e_1)=e_1\cdot U_\tau e_1=1+O(\tau^2)
 \quad\text{and}\quad h_{P_\tau}(e_1)=c_{\rm vol}(1-8\tau)+O(\tau^2).
\]
Thus rotations preserve the leading radial displacement, proving the asymptotics and optimality.

\smallskip\noindent
\emph{Quantitative rigidity.}
Recall that $\operatorname{cost}$ is the quadratic area cost and $\operatorname{disp}$ the first-order support displacement.
For $\xi=(z,Y_{\rm res})$, use the norm $\|\xi\|_{\mathrm{cost}}=\sqrt{\operatorname{cost}(\xi)}$ and let $\operatorname{modes}$ be the finite set of positive extremal vectors just identified, indexed by retained vertices.
Each has $\operatorname{cost}=8$.
We claim that
\begin{equation}\label{eq:quantitative-tangent-rigidity}
 \min_{\xi_{\rm ext}\in\operatorname{modes}}
 \left\|\xi-\sqrt{\operatorname{cost}(\xi)/8}\,\xi_{\rm ext}\right\|_{\mathrm{cost}}^2
 \le C\{8\operatorname{cost}(\xi)-\operatorname{disp}(\xi)^2\}.
\end{equation}
Normalize $\operatorname{cost}(\xi)=1$ and set $\gamma=8-\operatorname{disp}(\xi)^2$.
The left side is at most $4$, so only small $\gamma$ matters.
Choose $v,u$ attaining $\operatorname{disp}$ and let $r_u$ represent the functional $\xi\mapsto u\cdot Bv+g_z(u)$ in the $\operatorname{cost}$ inner product. Thus $\|r_u\|_{\mathrm{cost}}^2=\operatorname{dual}_v(u)$.
For $\gamma<1$, the bound $\operatorname{dual}_v(u)\le6$ excludes omitted vertices.
At a retained vertex the dual estimate gives
\[
 7\le \operatorname{disp}^2\le \operatorname{dual}_v(u)\le8
 \quad\text{and}\quad 4\|b\|^2\le8-\operatorname{dual}_v(u)\le\gamma.
\]
Lipschitz dependence on $u$, $\|u-v\|\le\sqrt2\|b\|$, and $r_v=-\xi_{\mathrm{ext},v}$ give
\[
 \left\|\frac{r_u}{\sqrt{\operatorname{dual}_v(u)}}+
                  \frac{\xi_{\mathrm{ext},v}}{\sqrt8}\right\|_{\mathrm{cost}}\le C\sqrt\gamma.
\]
If $\sigma$ is the sign of $\langle\xi,r_u\rangle_{\mathrm{cost}}$, the exact Cauchy--Schwarz remainder gives
\[
 \left\|\xi-\sigma\frac{r_u}{\sqrt{\operatorname{dual}_v(u)}}\right\|_{\mathrm{cost}}^2
 =2\left(1-\frac{\operatorname{disp}}{\sqrt{\operatorname{dual}_v(u)}}\right)
 \le\frac{2\gamma}{7}.
\]
Consequently $\|\xi+\sigma \xi_{\mathrm{ext},v}/\sqrt8\|_{\mathrm{cost}}\le C\sqrt\gamma$.
If $\sigma=1$, the nonnegative $v$-coordinate differs from $-12/\sqrt8$ by at least $12/\sqrt8$, and its squared dual norm $20(G^{-1})_{11}=40$ forces distance at least $3/\sqrt{20}$.
For small $\gamma$ this is impossible, so $\sigma=-1$ and the rigidity estimate follows.

For the coordinates of $P$, write $r_P=d_{24}(P)>0$.
Equations~\eqref{eq:intrinsic-small-coordinates}--\eqref{eq:intrinsic-support-upper} give
\[
 8\operatorname{cost}-\operatorname{disp}^2\le8\left\{\operatorname{def}(P)
                    -\frac{\sqrt2}{8}r_P^2\right\}+Cr_P^3
 \le C r_P^2\{\varepsilon(P)+r_P\}.
\]
The local formula is uniformly Lipschitz, and the frame changes and coordinate permutations identifying $\operatorname{modes}$ preserve $C_{24}$.
Thus \eqref{eq:quantitative-tangent-rigidity} yields an orthogonal $U$ and $\tau_{\rm cost}=\sqrt{\operatorname{cost}/8}$ with
\[
 d_H(\widehat P,UP_{\tau_{\rm cost}})
       \le C r_P\sqrt{\varepsilon(P)+r_P}.
\]
Finally, \eqref{eq:sharp-Hausdorff-cubic} and \eqref{eq:intrinsic-quadratic-energy} imply
\[
 \left|\operatorname{cost}-\frac{\sqrt2}{8}r_P^2\right|
       \le C r_P^2\{\varepsilon(P)+r_P\}.
\]
The inequality $|\sqrt a-\sqrt b|\le\sqrt{|a-b|}$ bounds $|\tau_{\rm cost}-r_P/(8c_{\rm vol})|$ by $Cr_P\sqrt{\varepsilon(P)+r_P}$.
Lipschitz dependence on this parameter proves the claimed proximity to the extremal family.
