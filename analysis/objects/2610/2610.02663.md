<!-- defn-2: body_md -->
\normalfont
 Let $L \in \mathbb{N}$ and $\{N_i\}_{i \in [L]} \subset \mathbb{N}$. An $L$-layer neural network $f: \mathop{\mathbb{R}}\nolimits^D \to \mathop{\mathbb{R}}\nolimits^{N_L}$ is               
\begin{equation}
\label{ee1}
f(x) = A_L \circ \sigma_{L-1} \circ A_{L-1} \circ \dots \circ \sigma_1 \circ A_1 (x),
\end{equation}
where $A_i(y) = W_i y + b_i$, with $W_i \in \mathop{\mathbb{R}}\nolimits^{N_{i} \times N_{i-1}}$, $b_i \in \mathop{\mathbb{R}}\nolimits^{N_i}$, $N_0 = D$, and each $\sigma_j$ applied component-wise.                                                     
Here $\{W_i\}$ are weights, $\{b_i\}$ biases, and $\{\sigma_i\}_{1 \le i \le L-1}$ activation functions. Define: depth $ \mathcal{L}(f) := L$; number of weights $ \mathcal{W}(f) = \sum_{i=1}^L N_i N_{i-1}$; maximum weight $ \mathcal{B}(f) = \max_{1 \le j \le L} (\|b_j\|_\infty \vee \|W_j\|_{\infty})$; and maximum output $ \mathcal{K}(f) = \sup_{x \in \mathop{\mathbb{R}}\nolimits^D}\|f(x)\|_\infty$. Let
\begin{align*}
     \mathcal{N}  \mathcal{N}_{\{\sigma_i\}_{1 \le i \le L-1}} (L, W, B, K) = \big\{  f \text{ of the form \eqref{ee1}}:  \mathcal{L}(f) \le L ,   \mathcal{W}(f) \le W,  \mathcal{B}(f) \le B,  \mathcal{K}(f) \le K \big\}.
\end{align*}
 When $\sigma_j(x) = x \vee 0$ for all $j \in [L-1]$, we write $ \mathcal{R}  \mathcal{N} (L, W, B, K)$ for $ \mathcal{N}  \mathcal{N}_{\{\sigma_i\}_{1 \le i \le L-1}} (L, W, B, K)$.

\begin{align*}
    \varphi_t(\kappa) = \mathbb{E} \exp(\iota \kappa^\top X_t) = \int \exp(\iota \kappa^\top (\alpha_t x_1 + \beta_t x_0 )) \mathrm{d} P_0(x_0) \mathrm{d} P^{\text{emp}}_n(x_1).
\end{align*}
For any $t \in [0,1-\delta]$, $\int |\varphi_t(\kappa)| \mathrm{d} \kappa \lesssim \delta^{-2 \kappa_1}(2\pi)^{D/2} < \infty$, by Lemma~\ref{lem_cond_1}. Hence, applying the Fourier inversion theorem, $X_t$ has a density and is given by,
\begin{align*}
    \varrho_t(x) = (2\pi)^{-D} \int \exp(-\iota\kappa^\top x) \varphi_t(\kappa) \mathrm{d} \kappa.
\end{align*}
By Lemma~\ref{lem_cond_1}, since,
\begin{align}

\begingroup
\allowdisplaybreaks

    = & \int_0^1 \mathbb{E}\|v(X_t, t) - v_\star(X_t,t)\|^2 \mathrm{d} t + \int_0^1 \mathbb{E} \|v_\star(X_t,t) - \dot{X_t}\|^2 \mathrm{d} t \\
    & + 2 \int_0^1 \mathbb{E} \mathbb{E} \left(\langle v(X_t, t) - v_\star(X_t,t), v_\star(X_t,t) - \dot{X_t}\rangle\mid X_t\right) \mathrm{d} t\\
    = &  \mathcal{L}(v) + \int_0^1 \mathbb{E} \|v_\star(X_t,t) - \dot{X_t}\|^2 \mathrm{d} t .
\end{align*}

<!-- lemma-6: body_md -->
\label{lem:score-lip}
Under Assumption~\ref{a1} and Design Choices D~\ref{a2}, for every $t\in[0,1-\delta]$ the map
$x\mapsto\nabla\log\varrho_t(x)$ is differentiable with
\[
  \nabla^2 \log\varrho_t(x)
  \;=\; \frac{\alpha_t^2}{\beta_t^4}\,\mathrm{Var}\big(X_1 \mid X_t = x\big) \;-\; \frac{1}{\beta_t^2}\,I_D ,
  \qquad
  \big\|\nabla^2\log\varrho_t(x)\big\| \;\lesssim\; D\,\delta^{-4\kappa_1},
\]
uniformly in $x\in\mathbb{R}^D$ and $t\in[0,1-\delta]$. In particular $\nabla\log\varrho_t(\cdot)$
is globally Lipschitz with constant $ \mathcal{O}(D\delta^{-4\kappa_1})$, uniformly in $t\in[0,1-\delta]$.

<!-- theorem-3: body_md -->
\label{thm:equivalence}
Fix $\delta \in (0,1/2)$. Under Assumption~\ref{a1} and Design Choices D~\ref{a2}, there is a
pathwise unique strong solution $\{X_t\}_{t\in[0,1-\delta]}$ of
    \begin{equation}
        dX_t = \mathfrak{b}_t(X_t)\,dt + \sigma\,dW_t, \qquad X_0 \sim \mathcal{N}(0,I_D),
        \label{eq:sde-sigma}
    \end{equation}
    with $\mathrm{Law}(X_t) = \varrho_t$ for every $t\in[0,1-\delta]$.
<!-- end theorem-3 -->
<!-- proof-5: body_md -->
Let $\mathcal{L}_t^\sigma f := \mathfrak{b}_t\cdot\nabla f + \tfrac{\sigma^2}{2}\Delta f$ be
the generator of \eqref{eq:sde-sigma}, with formal adjoint
$(\mathcal{L}_t^\sigma)^*$. Since $\varrho_t$ is a Gaussian smoothing of a compactly supported
measure it is smooth and strictly positive on $\mathbb{R}^D$ for every $t\in(0,1)$, so the
following manipulations are classical. Using the score identity
$\varrho_t \nabla\log\varrho_t = \nabla\varrho_t$,
\begin{align}
    (\mathcal{L}_t^\sigma)^*\varrho_t
    = & -\nabla\cdot(\mathfrak{b}_t \varrho_t) + \frac{\sigma^2}{2}\Delta\varrho_t \notag\\
    = & -\nabla\cdot\!\Big(v_\star \varrho_t + \frac{\sigma^2}{2}\nabla\varrho_t\Big)
       + \frac{\sigma^2}{2}\Delta\varrho_t \notag \\
    = & -\nabla\cdot(v_\star \varrho_t) \notag\\
    = &\; \partial_t \varrho_t.
    \label{eq:fpk-cancel}
\end{align}
In the above calculations, \eqref{eq:fpk-cancel} follows from \eqref{eq:continuity}. This holds \emph{for every} $\sigma\geq 0$ simultaneously: the diffusive term
$\tfrac{\sigma^2}{2}\Delta\varrho_t$ contributed by the noise is exactly
cancelled by the score correction in \eqref{eq:drift-def}, not merely
asymptotically, and \eqref{eq:fpk-cancel} does not depend on $\sigma$ at all.
Hence $t\mapsto\varrho_t\,dx$ is a probability solution of
\begin{equation}
    \partial_t \mu_t = (\mathcal{L}_t^\sigma)^*\mu_t, \qquad t\in[0,1-\delta],
    \qquad \mu_0 = \varrho_0 =  \mathcal{N}(0,I_D),
    \label{eq:fpk-sigma}
\end{equation}
for every admissible $\sigma$, the initial condition being the weak limit in
\eqref{eq:continuity}.

By Corollary~\ref{cor14}, $v_\star(\cdot,t)$ is $ \mathcal{O}(D\delta^{-3\kappa_1})$-Lipschitz uniformly
over $t\in[0,1-\delta]$, and by Lemma~\ref{lem:score-lip}, $\nabla\log\varrho_t(\cdot)$ is
$ \mathcal{O}(D\delta^{-4\kappa_1})$-Lipschitz uniformly over the same range. Hence
\[
    \mathfrak{b}_t(x) = v_\star(x,t) + \tfrac{\sigma^2}{2}\nabla\log\varrho_t(x)
\]
is globally Lipschitz in $x$, uniformly over $t\in[0,1-\delta]$, with Lipschitz constant
$\Lambda_\delta :=  \mathcal{O}\big(D(1\vee\sigma^2)\delta^{-4\kappa_1}\big)$, and is of at most linear
growth, since $\|\mathfrak{b}_t(0)\| \lesssim \delta^{-\kappa_1}\sqrt D$ by Corollary~\ref{cor14}
and $\|\nabla\log\varrho_t(0)\| \le \beta_t^{-2}\alpha_t\mathbb{E}\|X_1\| \lesssim \delta^{-2\kappa_1}\sqrt{D}$.
Since $\sigma$ is a fixed positive constant, the classical It\^o
existence--uniqueness theorem for SDEs with globally Lipschitz coefficients of linear growth
\citep[Theorem~2.9, Chapter~5]{karatzas1991brownian} gives a pathwise unique strong solution
$\{X_t\}_{t\in[0,1-\delta]}$ of \eqref{eq:sde-sigma} for any initial law with
finite second moment, in particular for $\mathrm{Law}(X_0)= \mathcal{N}(0,I_D)$.

The drift $\mathfrak{b}$ is globally Lipschitz in $x$ uniformly in $t\in [0,1-\delta]$ and of linear growth, and
the diffusion coefficient $\sigma I_D$ is constant and nondegenerate, so the Cauchy problem
\eqref{eq:fpk-sigma} admits at most one probability solution on $[0,1-\delta]$ with initial datum
$\varrho_0$ \citep[Ch.~9]{bogachev2015fokker}. Equivalently, and without appeal to that theory:
$t\mapsto\mathrm{Law}(X_t)$ for the pathwise-unique solution of \eqref{eq:sde-sigma} is itself a
probability solution of \eqref{eq:fpk-sigma} with the same initial datum, and two probability
solutions of a well-posed SDE with the same initial law have equal marginals. From the above arguments, the map,
$t\mapsto\varrho_t$ is also a probability solution of \eqref{eq:fpk-sigma} with
$\varrho_t|_{t=0}=\varrho_0= \mathcal{N}(0,I_D)$. Uniqueness therefore forces
$\mathrm{Law}(X_t)=\varrho_t$ for every $t\in[0,1-\delta]$.

Write $\Omega = C([0,1-\delta];\mathop{\mathbb{R}}\nolimits^D)$ for path space and let $\boldsymbol{P}$ denote the law on $\Omega$ of the
solution $\{X_t\}$ of
\[
  \mathrm{d} X_t = \big(a_t v_\star(X_t,t) - b_t X_t\big)\mathrm{d} t + \sigma \mathrm{d} W_t,
  \qquad X_0 \sim  \mathcal{N}(0, I_D),
\]
which exists and is pathwise unique by Theorem~\ref{thm:equivalence}. Let $\boldsymbol{Q}$ denote the law on $\Omega$ of the
Euler--Maruyama process \eqref{exponential_integrator}, namely the solution of
\[
  \mathrm{d}  \widehat{X}_t = \big(a_{t_k}\hat{v}( \widehat{X}_{t_k},t_k) - b_t  \widehat{X}_t\big)\mathrm{d} t + \sigma \mathrm{d} W_t,
  \quad t \in [t_k, t_{k+1}), \qquad  \widehat{X}_0 \sim  \mathcal{N}(0,I_D).
\]
Both are driven by the same diffusion coefficient $\sigma > 0$ and started from the same law, so
$\boldsymbol{P}$ and $\boldsymbol{Q}$ differ only through their drifts. Writing these as functionals of the path,
\[
  \mu^{\boldsymbol{P}}_t(\omega) = a_t v_\star(\omega_t, t) - b_t\omega_t,
  \qquad
  \mu^{\boldsymbol{Q}}_t(\omega) = a_{t_k}\hat{v}(\omega_{t_k}, t_k) - b_t\omega_t
  \quad \text{for } t \in [t_k,t_{k+1}),
\]
both are adapted, and the linear terms $-b_t\omega_t$ coincide, so that
\begin{equation}
  \mu^{\boldsymbol{P}}_t(\omega) - \mu^{\boldsymbol{Q}}_t(\omega)
  = a_t v_\star(\omega_t,t) - a_{t_k}\hat{v}(\omega_{t_k},t_k),
  \qquad t \in [t_k, t_{k+1}).
  \label{eq:driftdiff}
\end{equation}
By Corollary~\ref{cor14}, $\|v_\star(x,t)\| \lesssim \delta^{-\kappa_1}(\|x\| + \sqrt D)$ uniformly on
$[0,1-\delta]$; by D~\ref{a2}(a), $\sup_{t \le 1-\delta}|a_t| \lesssim \delta^{-\kappa_1/2}$; and
$\|\hat{v}(\cdot,\cdot)\| \le \sqrt D\, \mathcal{K}(\hat{v}) \le \sqrt D K$. Hence the drift difference
\eqref{eq:driftdiff} satisfies
\[
  \big\|\mu^{\boldsymbol{P}}_t(\omega) - \mu^{\boldsymbol{Q}}_t(\omega)\big\|
  \;\le\; C_\delta\Big(1 + \sup_{s \le t}\|\omega_s\|\Big),
  \qquad t \in [0,1-\delta],
  \qquad C_\delta \;\lesssim\; \delta^{-\kappa_1/2}\sqrt D\,\big(\delta^{-\kappa_1} + K\big),
\]
which is finite for each fixed $\delta > 0$. Since Bene\v{s}'s condition requires only linear growth
of the drift difference in the sup-norm of the path, with no restriction on the constant, it holds here
\citep[Corollary~3.5.16]{karatzas1991brownian}, and the stochastic exponential below is a true
martingale. Girsanov's theorem \citep[Theorem~8.6.5]{oksendal2003stochastic}
therefore gives $\boldsymbol{Q} \sim \boldsymbol{P}$ with 
\[
  \frac{\mathrm{d} \boldsymbol{P}}{\mathrm{d} \boldsymbol{Q}}(\omega)
  = \exp\left(\frac{1}{\sigma}\int_0^{1-\delta}\big\langle \mu^{\boldsymbol{P}}_t - \mu^{\boldsymbol{Q}}_t, \mathrm{d} W_t\big\rangle
    - \frac{1}{2\sigma^2}\int_0^{1-\delta}\big\|\mu^{\boldsymbol{P}}_t - \mu^{\boldsymbol{Q}}_t\big\|^2 \mathrm{d} t\right),
\]
and taking $\mathbb{E}_{\boldsymbol{P}}\log(\mathrm{d}\boldsymbol{P}/\mathrm{d}\boldsymbol{Q})$, the stochastic integral being a true martingale under $\boldsymbol{P}$,
\begin{equation}
  \operatorname{KL}(\boldsymbol{P} \,\|\, \boldsymbol{Q})
  = \frac{1}{2\sigma^2}\sum_{k=0}^{N-1}\int_{t_k}^{t_{k+1}}
    \mathbb{E}\big\|a_t v_\star(X_t,t) - a_{t_k}\hat{v}(X_{t_k},t_k)\big\|^2 \mathrm{d} t .
  \label{eq:pathkl}
\end{equation}
The marginal laws at the terminal time are $P_{1-\delta} = \boldsymbol{P} \circ \pi_{1-\delta}^{-1}$ and
$ \hat{P}_{1-\delta}(\hat{v}) = \boldsymbol{Q} \circ \pi_{1-\delta}^{-1}$, where $\pi_{1-\delta}(\omega) = \omega_{1-\delta}$,
the first identification being Theorem~\ref{thm:equivalence}. Since the Kullback--Leibler divergence contracts
under the pushforward by $\pi_{1-\delta}$,
\begin{equation}
  \operatorname{KL}\left(P_{1-\delta}\,\|\, \hat{P}_{1-\delta}(\hat{v})\right) \le \operatorname{KL}(\boldsymbol{P}\,\|\,\boldsymbol{Q}).
  \label{eq:dpi}
\end{equation}
It remains to split \eqref{eq:pathkl}. Adding and subtracting $a_{t_k}v_\star(X_{t_k},t_k)$ and using
$\|u+v\|^2 \le 2\|u\|^2 + 2\|v\|^2$,
\begin{align*}
  \operatorname{KL}(\boldsymbol{P}\,\|\,\boldsymbol{Q})
  \le & \; \frac{1}{\sigma^2}\sum_{k=0}^{N-1}\int_{t_k}^{t_{k+1}}
    \mathbb{E}\big\|a_t v_\star(X_t,t) - a_{t_k} v_\star(X_{t_k},t_k)\big\|^2 \mathrm{d} t \\
  & + \frac{1}{\sigma^2}\sum_{k=0}^{N-1}\int_{t_k}^{t_{k+1}} a_{t_k}^2\,
    \mathbb{E}\big\|v_\star(X_{t_k},t_k) - \hat{v}(X_{t_k},t_k)\big\|^2 \mathrm{d} t .
\end{align*}
The second sum has an integrand constant in $t$ on each $[t_k,t_{k+1})$ and so equals
$\sigma^{-2}\sum_k h_k a_{t_k}^2\,\mathbb{E}\|v_\star(X_{t_k},t_k) - \hat{v}(X_{t_k},t_k)\|^2$. For the first, adding and
subtracting $a_t v_\star(X_{t_k},t_k)$ and splitting once more,
\[
  \big\|a_t v_\star(X_t,t) - a_{t_k}v_\star(X_{t_k},t_k)\big\|^2
  \le 2a_t^2\big\|v_\star(X_t,t) - v_\star(X_{t_k},t_k)\big\|^2
    + 2(a_t - a_{t_k})^2\big\|v_\star(X_{t_k},t_k)\big\|^2 .
\]
Combining the three displays with \eqref{eq:dpi} gives the claim. \qedhere

Throughout, $t \in [t_k, t_{k+1})$ for the relevant $k$, and all expectations are under the law of
$\{X_t\}_{t\in[0,1-\delta]}$ solving \eqref{eq:sde}. By Theorem~\ref{thm:equivalence},
$\mathrm{Law}(X_s) = \varrho_s = \mathrm{Law}(\alpha_s X_1 + \beta_s X_0)$ for each fixed $s$. Since $X_1 \sim P^{\text{emp}}_n$ is supported in
$[0,1]^D$ by Assumption~\ref{a1} and $X_0 \sim  \mathcal{N}(0,I_D)$,
\begin{equation}
  \mathbb{E}\|X_s\|^2 \le 2\alpha_s^2\,\mathbb{E}\|X_1\|^2 + 2\beta_s^2\,\mathbb{E}\|X_0\|^2 \lesssim D,
  \qquad s \in [0,1-\delta].
  \label{eq:secondmoment}
\end{equation}
Combining \eqref{eq:secondmoment} with Corollary~\ref{cor14},
\begin{equation}
  \mathbb{E}\big\|v_\star(X_s,s)\big\|^2 \lesssim \delta^{-2\kappa_1}\Big(\mathbb{E}\|X_s\|^2 + D\Big)
  \lesssim D\,\delta^{-2\kappa_1}.
  \label{eq:vstarmoment}
\end{equation}

We write $\mu_s(x) := a_s v_\star(x,s) - b_s x$ for the drift
of \eqref{eq:sde}. By \eqref{eq:vstarmoment}, D~\ref{a2}(a) (which gives $a_s^2 \lesssim \delta^{-\kappa_1}$
and $|b_s| \lesssim \delta^{-\kappa_1}$) and \eqref{eq:secondmoment},
\begin{equation}
  \mathbb{E}\|\mu_s(X_s)\|^2 \le 2a_s^2\,\mathbb{E}\|v_\star(X_s,s)\|^2 + 2b_s^2\,\mathbb{E}\|X_s\|^2
  \lesssim D\,\delta^{-3\kappa_1}.
  \label{eq:driftmoment}
\end{equation}
Integrating \eqref{eq:sde} from $t_k$ to $t$,
\[
  X_t - X_{t_k} = \int_{t_k}^{t}\mu_s(X_s)\,\mathrm{d} s + \sigma\big(W_t - W_{t_k}\big).
\]
By $\|u+v\|^2 \le 2\|u\|^2+2\|v\|^2$, the Cauchy--Schwarz inequality applied to the drift
integral, and $\mathbb{E}\|W_t - W_{t_k}\|^2 = D(t-t_k)$,
\begin{align}
  \mathbb{E}\big\|X_t - X_{t_k}\big\|^2
  &\le 2(t-t_k)\int_{t_k}^{t}\mathbb{E}\|\mu_s(X_s)\|^2\,\mathrm{d} s + 2\sigma^2 D\,(t-t_k) \nonumber\\
  &\lesssim D\,\delta^{-3\kappa_1}(t-t_k)^2 + \sigma^2 D\,(t-t_k) \nonumber\\
  &\lesssim\; D\big(\sigma^2 + \delta^{-3\kappa_1}h\big)(t-t_k).
  \label{eq:increment}
\end{align}

Note that the $\sigma^2$ appearing here is exactly cancelled by the factor $\sigma^{-2}$ carried by
Lemma~\ref{lem_oracle}, so the resulting contribution to the KL bound is independent of $\sigma$; see
Remark~\ref{rem:sigma} below.

 Splitting in space and in time,
\begin{align}
  & \mathbb{E}\big\|v_\star(X_t,t) - v_\star(X_{t_k},t_k)\big\|^2 \nonumber\\
  \le\; & 2\,\mathbb{E}\big\|v_\star(X_t,t) - v_\star(X_{t_k},t)\big\|^2
    + 2\,\mathbb{E}\big\|v_\star(X_{t_k},t) - v_\star(X_{t_k},t_k)\big\|^2 .
  \label{eq:split}
\end{align}
For the first summand, Corollary~\ref{cor14} gives $\operatorname{Lip}(v_\star(\cdot,t)) \lesssim D\delta^{-3\kappa_1}$
uniformly in $t \in [0,1-\delta]$. Thus, from \eqref{eq:increment},
\begin{equation}
  \mathbb{E}\big\|v_\star(X_t,t) - v_\star(X_{t_k},t)\big\|^2
  \lesssim D^2\delta^{-6\kappa_1}\,\mathbb{E}\|X_t - X_{t_k}\|^2
  \lesssim D^3\delta^{-6\kappa_1}\big(\sigma^2 + \delta^{-3\kappa_1}h\big)(t-t_k).
  \label{eq:spatial}
\end{equation}
For the second summand, the two evaluations share the spatial argument $X_{t_k}$, so only the
one-time marginal of $X_{t_k}$ is involved. By Lemma~\ref{lip_general},
$\|\partial_t v_\star(x,t)\| \lesssim \delta^{-\kappa_1}(1+\|x\|)$ uniformly on $[0,1-\delta]$. Hence,
by the mean value theorem and \eqref{eq:secondmoment},
\begin{equation}
  \mathbb{E}\big\|v_\star(X_{t_k},t) - v_\star(X_{t_k},t_k)\big\|^2
  \lesssim \delta^{-2\kappa_1}\,\mathbb{E}\big(1 + \|X_{t_k}\|^2\big)(t-t_k)^2
  \lesssim D\,\delta^{-2\kappa_1}(t-t_k)^2 .
  \label{eq:temporal}
\end{equation}
 Using $a_t^2 \lesssim \delta^{-\kappa_1}$, then \eqref{eq:split},
\eqref{eq:spatial} and \eqref{eq:temporal}, and finally
$\int_{t_k}^{t_{k+1}}(t-t_k)\mathrm{d} t = h^2/2$ and $\int_{t_k}^{t_{k+1}}(t-t_k)^2\mathrm{d} t = h^3/3$,
\begin{align*}
   \sum_{k=0}^{N-1}\int_{t_k}^{t_{k+1}} a_t^2\,\mathbb{E}\big\|v_\star(X_t,t) - v_\star(X_{t_k},t_k)\big\|^2 \mathrm{d} t 
  \lesssim\; & \delta^{-\kappa_1}\left[
      D^3\delta^{-6\kappa_1}\big(\sigma^2 + \delta^{-3\kappa_1}h\big)\sum_{k=0}^{N-1}\frac{h^2}{2}
      + D\,\delta^{-2\kappa_1}\sum_{k=0}^{N-1}\frac{h^3}{3}\right] \\
  \lesssim\; & D^3\delta^{-7\kappa_1}\big(\sigma^2 + \delta^{-3\kappa_1}h\big)\,h
      + D\,\delta^{-3\kappa_1}h^2 \\
  \lesssim\; & D^3\big(\sigma^2 \vee 1\big)\,\delta^{-10\kappa_1}\,h ,
\end{align*}
where the second line uses $Nh = 1-\delta \le 1$, so that $\sum_k h^2 \le h$ and $\sum_k h^3 \le h^2$,
and the last uses $h \le 1$ to absorb $h^2$ into $h$. This is \eqref{eq:dis1}.

To bound \eqref{eq:dis2}, we note that this term involves $v_\star$ at the single time $t_k$ only, so
\eqref{eq:vstarmoment} applies directly. By D~\ref{a2}(b), $|a_t - a_{t_k}| \lesssim \delta^{-\kappa_2}(t-t_k)$, whence
\[
  \sum_{k=0}^{N-1}\int_{t_k}^{t_{k+1}} (a_t - a_{t_k})^2\,\mathbb{E}\big\|v_\star(X_{t_k},t_k)\big\|^2 \mathrm{d} t
  \lesssim D\,\delta^{-2\kappa_1 - 2\kappa_2}\sum_{k=0}^{N-1}\int_{t_k}^{t_{k+1}} (t-t_k)^2 \mathrm{d} t
  \lesssim D\,\delta^{-2(\kappa_1+\kappa_2)}h^2,
\]
using $\sum_k h^3 \le h^2$. This is \eqref{eq:dis2}. Since $h \le 1$ and
$2(\kappa_1+\kappa_2) \le 10\kappa_1 + 2\kappa_2$, the two bounds combine as stated. \qedhere

Throughout, fix $t\in(0,1)$ with $\alpha_t\neq 0,\ \beta_t>0$, and recall the interpolant is given by,
\(X_t = \alpha_t X_1 + \beta_t X_0, \, X_1\simP^{\text{emp}}_n,\, X_0\sim N(0,I_D)\ \text{independent of } X_1.\)
Hence, $\varrho_t$, the law of $X_t$, is the Gaussian smoothing
\[
\varrho_t(x) = \int_{\mathop{\mathbb{R}}\nolimits^D} \phi(x;\alpha_t y,\beta_t^2 I_D)\,P^{\text{emp}}_n(dy),
\qquad
\]
where, $\phi(x;\mu,\Sigma) := (2\pi)^{-D/2}\det(\Sigma)^{-1/2}\exp\!\Big(-\tfrac12(x-\mu)^\top \Sigma^{-1}(x-\mu)\Big).$
By definition, $v_\star(x,t) = \mathbb{E}[\dot X_t \mid X_t = x]$. For notational simplicity, we write
\[
m(x,t):=\mathbb{E}[X_1\mid X_t=x]\in\mathop{\mathbb{R}}\nolimits^D,\qquad
\Sigma(x,t):=\operatorname{Var}(X_1\mid X_t=x)\in\mathop{\mathbb{R}}\nolimits^{D\times D}.

Since $\beta_t>0$, we may write $X_0 = \beta_t^{-1}(X_t-\alpha_t X_1)$. Hence
\[
v_\star(x,t) = \mathbb{E}[\dot{\alpha}_t X_1 + \dot{\beta}_t X_0 \mid X_t=x]
= \dot{\alpha}_t\, m(x,t) + \frac{\dot{\beta}_t}{\beta_t}\big(x - \alpha_t m(x,t)\big),
\]
using $\mathbb{E}[X_t\mid X_t=x]=x$. Collecting terms in $m(x,t)$ gives
\begin{equation}
v_\star(x,t) = \frac{\dot{\beta}_t}{\beta_t}\,x + \frac{\dot{\alpha}_t\beta_t-\dot{\beta}_t\alpha_t}{\beta_t}\, m(x,t).
\end{equation}

To prove equation \eqref{i2}, we note that, $\nabla_x \phi(x;\alpha_t y,\beta_t^2 I_D) = -\beta_t^{-2}(x-\alpha_t y)\,\phi(x;\alpha_t y,\beta_t^2 I_D)$, differentiation under the integral sign (justified since $\beta_t>0$ and $P^{\text{emp}}_n$ has finite second moments) gives
\[
\nabla \varrho_t(x) = -\frac{1}{\beta_t^2}\int (x-\alpha_t y)\,\phi(x;\alpha_t y,\beta_t^2 I_D)\,P^{\text{emp}}_n(dy).
\]
Dividing by $\varrho_t(x)>0$ and using that $\phi(x;\alpha_t y,\beta_t^2I_D)P^{\text{emp}}_n(dy)/\varrho_t(x)$ is exactly the conditional law of $X_1$ given $X_t=x$,

\end{equation}

\begin{equation}

\label{i4}
\end{equation}

\[

+ \frac{\dot{\alpha}_t\beta_t-\dot{\beta}_t\alpha_t}{\alpha_t\beta_t}\,x
+ \frac{(\dot{\alpha}_t\beta_t-\dot{\beta}_t\alpha_t)\beta_t}{\alpha_t}\,\nabla\log\varrho_t(x).
\]
The coefficient of $x$ simplifies as
\(\frac{\dot{\beta}_t}{\beta_t}+\frac{\dot{\alpha}_t\beta_t-\dot{\beta}_t\alpha_t}{\alpha_t\beta_t}
=\frac{\dot{\beta}_t\alpha_t+\dot{\alpha}_t\beta_t-\dot{\beta}_t\alpha_t}{\alpha_t\beta_t}
=\frac{\dot{\alpha}_t}{\alpha_t},
\)
and the coefficient of $\nabla\log\varrho_t(x)$ as
\[
\frac{\beta_t(\dot{\alpha}_t\beta_t-\dot{\beta}_t\alpha_t)}{\alpha_t}
=\beta_t^2\cdot\frac{\dot{\alpha}_t\beta_t-\dot{\beta}_t\alpha_t}{\alpha_t\beta_t}
=\beta_t^2\left(\frac{\dot{\alpha}_t}{\alpha_t}-\frac{\dot{\beta}_t}{\beta_t}\right).

\end{equation}
Write $N_i(x):=\int y_i\,\phi(x;\alpha_t y,\beta_t^2I_D)\,P^{\text{emp}}_n(dy)$, so that $m_i(x,t)=N_i(x)/\varrho_t(x)$ for $i=1,\dots,D$. Differentiating under the integral sign as before,

Dividing by $\varrho_t(x)$,

Differentiating $g(x) = -\frac{x - \alpha_t E_1}{\beta_t^2}$ in $x$ and using $\nabla_x E_1 = \frac{\alpha_t}{\beta_t^2}V_1$,

 By the product rule,

\partial_t v_\star(x,t) = \dot u_t\, x + \dot q_t\, g(x) + q_t\, \partial_t g(x), \qquad \dot q_t = \beta_t^2\big(2w_t\kappa_t + \dot\kappa_t\big), \quad \dot\kappa_t = \dot u_t - \dot w_t.

Let $f_c (y) = \sup_{x \in [0,1]^D} \left(f(x) - \|x-y\|^p\right)$ denote the $c$-transform of $f$. It can be shown that $f(x) = \inf_{y \in [0,1]^D}\left(f_c(y) + \|x-y\|^p\right)$. From Kantorovich duality, it is well known that, for any $p \ge 1$,
\begin{align*}
    \mathbb{W}_p^p(\mu, \nu) = \sup_{f \in  \mathcal{C}_b\left([0,1]^D\right)} \left( \int f \mathrm{d} \mu - \int f_c \mathrm{d} \nu\right),
\end{align*}
where $ \mathcal{C}_b$ denotes the set of all bounded continuous functions on $[0,1]^D$. By noting that $(f+a)_c = f_c + a$ for any constant $a$, we can restrict this function class by shifting $f$ such that $\inf_{x \in [0,1]^D} f(x) = 0$. Since $f$ satisfies $f(x) - f(y) \le \|x-y\|^p \le D^{p/2}$ under the optimal transform, this shift guarantees $0 \le f(x) \le D^{p/2}$. 
Consequently, for any $y \in [0,1]^D$, we have
\begin{align*}
    f_c(y) = \sup_{x \in [0,1]^D} \left(f(x) - \|x-y\|^p\right) \ge f(y) - \|y-y\|^p = f(y) \ge 0.
\end{align*}
Hence, we can take the supremum on the set of all bounded continuous functions from $[0,1]^D \to [0,D^{p/2}]$. Thus,
\begin{align*}
    \mathbb{W}_p^p(\mu, \nu) = \sup_{f \in  \mathcal{C}_b\left([0,1]^D\right) \cap \{0 \le f \le D^{p/2}\}} \left( \int f \mathrm{d} \mu - \int f_c \mathrm{d} \nu\right).
\end{align*}
Let 
\begin{align*}
    W(x_1, \dots, x_n) = & \mathbb{W}_p^p(P_{\text{data}},P^{\text{emp}}_n)\\
    = & \sup_{f \in  \mathcal{C}_b\left([0,1]^D\right) \cap \{0 \le f \le D^{p/2}\}} \left( \frac{1}{n} \sum_{i=1}^n f(x_i) - \int f_c \mathrm{d} P_{\text{data}}\right).
\end{align*}
Hence,
\begin{align*}
    & W(x_1, \dots, x_i, \dots, x_n) - W(x_1, \dots, x_i^\prime, \dots, x_n) \\
   \le & \sup_{f \in  \mathcal{C}_b\left([0,1]^D\right) \cap \{0 \le f \le D^{p/2}\}} \left( \frac{1}{n} \sum_{j=1}^n f(x_j) - \int f_c \mathrm{d} P_{\text{data}}\right) \\
   &- \sup_{f \in  \mathcal{C}_b\left([0,1]^D\right) \cap \{0 \le f \le D^{p/2}\}} \left( \frac{1}{n} \sum_{j \neq i} f(x_j) + \frac{f(x_i^\prime)}{n} - \int f_c \mathrm{d} P_{\text{data}} \right)\\
   \le  & \frac{1}{n} \sup_{0 \le f \le D^{p/2}} \left(f(x_i) - f(x_i^\prime)\right) \\
   \le & \frac{D^{p/2}}{n}.
\end{align*}
Applying the bounded difference inequality, we get that with probability at least $1-\xi$,
\begin{align*}
    \mathbb{W}_p^p(P^{\text{emp}}_n,P_{\text{data}}) \le & \ \mathbb{E} \mathbb{W}_p^p(P^{\text{emp}}_n,P_{\text{data}}) + \sqrt{\frac{D^p \log(1/\xi)}{2n}}\\
    \le & \ n^{-p/d} + \sqrt{\frac{D^p \log(1/\xi)}{2n}},
\end{align*}
where the last inequality follows from \citet[Theorem~10]{chakraborty2026diffusion}. The desired inequality now follows from taking the $p$-th root of both sides and applying the subadditivity inequality $(a+b)^{1/p} \le a^{1/p} + b^{1/p}$ (valid for $a,b \ge 0$ and $p \ge 1$).
