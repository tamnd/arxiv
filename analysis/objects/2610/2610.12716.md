<!-- definition-1: body_md -->
\label{def:HT_cond}
    A family of IMAP queues parameterized by their idleness $\varepsilon$ for $\varepsilon\in \mathcal{E}$ that satisfies our \emph{heavy-traffic conditions} is a family of IMAP queues where there exist matrices $\boldsymbol{F}^{(0)}$ and $\boldsymbol{R}^{(0)}$, $\mu^{(0)}\in (0,\infty)$, and a distribution $\boldsymbol{\pi}^{(0)}$ such that:
    \begin{enumerate}
        \item 
        \begin{enumerate}
            \item For all $i\in \mathbb{N}_0$, $\lim_{\varepsilon\to 0}\pi_i^{(\varepsilon)}= \pi_i^{(0)}$.
            
            \item For all $i,j\in \mathbb{N}_0$, $\lim_{\varepsilon\to 0} r_{i,j}^{(\varepsilon)}= r_{i,j}^{(0)}$.

            \item For all $i,j\in \mathbb{N}_0$, $\lim_{\varepsilon\to 0} \boldsymbol{F}_{i,j}^{(\varepsilon)}= \boldsymbol{F}_{i,j}^{(0)}$.

            \item $\lim_{\varepsilon\to 0}\mu^{(\varepsilon)}= \mu^{(0)}$.
        \end{enumerate}

                                                          
                                                                                                                                                                                                                                                                                                                                                                                                                    
                                                                                                                                                                                

        \item \label{M_conv}Let $M^{(0)}=1-(\mu^{(0)})^{-1}\boldsymbol{\pi}^{(0)}\left(-\boldsymbol{F}^{(0)}+\mu^{(0)} \boldsymbol{I}\right)\boldsymbol{R}^{(0)}\left(-\boldsymbol{F}^{(0)}+\mu^{(0)} \boldsymbol{I}\right)e^T$. \\
        For all $z\in (0,1)$,
        $\lim_{\varepsilon\to 0}M(z^\varepsilon;\varepsilon) =M^{(0)}$.
        
        \item $Y^{(\varepsilon)}$ is ergodic for all $\varepsilon\in \mathcal{E}$.
    \end{enumerate}

<!-- definition-2: body_md -->
\label{def:rtbdd}
    A \emph{rate-bounded} family of IMAP queues is a family of IMAP queues where, there exist some $\hat{\lambda}>0, T_{\max}>0,$ and $\hat{\mu} > 0$ such that for all modulation states and all possible idleness $\varepsilon\in \mathcal{E}$,
    \begin{enumerate}
        \item The instantaneous arrival rate to the queue is upper bounded by $\hat{\lambda}$.

        \item The expected time until transition in any modulation state is upper bounded by $T_{\max}$.

        \item The service rate is lower bounded by $\hat{\mu}$.
    \end{enumerate}
<!-- end definition-2 -->
<!-- definition-3: body_md -->
\label{def:bd}
    A \emph{birth-death} family of IMAP queues is a family of IMAP queues where $Y^{(\varepsilon)}$ is a birth-death process on $\mathbb{N}_0$ for all $\varepsilon\in \mathcal{E}$, i.e., for all $\varepsilon\in \mathcal{E}$, $Y^{(\varepsilon)}(t)\in \{0, 1, 2, \dots\}$ and from $Y^{(\varepsilon)} = i$, $Y^{(\varepsilon)}$ can only transition to $i-1$ or $i+1$.
<!-- end definition-3 -->
<!-- definition-4: body_md -->
\label{def:unifstab}
    A \emph{uniformly stable} birth-death family of IMAP queues is a family of IMAP queues where there is some $N$, $a\in(0,1)$, and $b\in (0, \frac{1}{2})$ such that:
        \begin{enumerate}
            \item For all $\varepsilon\in \mathcal{E}$ and all $n\geq N$, $\frac{\alpha^{(\varepsilon)}_{n,n+1}}{\alpha^{(\varepsilon)}_{n+1,n}} \leq a$

            \item For all $\varepsilon\in \mathcal{E}$ and all $n\geq N$, $\frac{\alpha^{(\varepsilon)}_{n,n+1}}{\alpha^{(\varepsilon)}_{n,n-1}+\alpha^{(\varepsilon)}_{n,n+1}} \leq b$
        \end{enumerate}

<!-- definition-5: body_md -->
\label{def:asymp_exp_ind}
    An \emph{asymptotically exponential and independent} family of IMAP queues with a heavy-traffic convergent phase process with mean $M$ is a family of IMAP queues with heavy-traffic convergent phase process where
    \begin{equation}    \label{eq:asympexp}
         \lim_{\varepsilon\to 0} \overline{q}(z^{\varepsilon};\varepsilon) = \frac{1}{1+M\ln z}
    \end{equation}
    and
    \begin{equation}
    \label{eq:asymp_ind}
        \lim_{\varepsilon\to 0} \hat{q}(z^\varepsilon;i,\varepsilon) = \frac{\pi^{(0)}_i}{1+M\ln z}
    \end{equation}
    for all $z\in (0,1)$ and all $i\in \mathbb{N}_0$.

<!-- theorem-1: body_md -->
\label{Thm1}
    If a family of IMAP queues with a heavy-traffic convergent phase process has a light-tailed phase process (\cref{def:light-tailed}, which will be defined in \cref{Full_Thm1}), then $\varepsilon Q^{(\varepsilon)}$ is asymptotically exponential with mean $M^{(0)}$ (where $M^{(0)}$ is defined in \cref{def:HT_cond}) and asymptotically independent from $Y^{(\varepsilon)}$ in the $\varepsilon\to 0$ limit. Equivalently,
    \begin{align*}
        \lim_{\varepsilon\to 0} \overline{q}(z;\varepsilon) &= \frac{1}{1+M^{(0)}\ln z}, \qquad
        \lim_{\varepsilon\to 0} \hat{q}(z;i,\varepsilon) = \frac{\pi^{(0)}_i}{1+M^{(0)}\ln z}.
    \end{align*}

\label{Thm2}

Consequently $\varepsilon Q^{(\varepsilon)}$ is asymptotically exponential with mean $M^{(0)}$ (where $M^{(0)}$ is defined in  \cref{Rdefn}) and asymptotically independent from $Y^{(\varepsilon)}$ in the $\varepsilon\to 0$ limit. Equivalently,

    \end{align*}

<!-- lemma-1: body_md -->
\label{R_exst_lemma}
    For any ergodic continuous time Markov chain $Y$ with generator matrix $\boldsymbol{A}$, there exists a near-inverse matrix $\boldsymbol{R} = [r_{i,j}]_{i,j\in \mathbb{N}_0}$ satisfying
    \[\boldsymbol{A}\boldsymbol{R}= \begin{bmatrix}
        0 & -\frac{1}{\pi_0} \boldsymbol{\pi}_{\overline{0}}\\
        0& I
    \end{bmatrix}=
    \begin{bmatrix}
        0 & -\frac{\pi_1}{\pi_0} & -\frac{\pi_2}{\pi_0} & \dots &-\frac{\pi_n}{\pi_0} &\dots\\
        0 & 1 & 0 &\dots &0&\dots\\
        0&0&1&\dots&0&\dots\\
        \vdots\\
        0&0&0&\dots &1&\dots\\
        \vdots
    \end{bmatrix}\]
    or, equivalently,
    \[\boldsymbol{A}\boldsymbol{R}x=x-\frac{x_0}{\pi_0}\boldsymbol{\pi}\]
    for any vector $x$.
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{ind_exp_lemma}
    For any infinite-phase Markovian arrival process, if there exists a near-inverse matrix $R$ then the system has the following near-independence and near-exponential equations for any $M>0$:
    \begin{equation}
         \hat{\boldsymbol{q}}\left(z\right) = \hat{q}\left(z;0\right)\frac{\boldsymbol{\pi}}{\pi_0} +\hat{q}\left(z;0\right)\frac{\boldsymbol{\pi}}{\pi_0}\left( \boldsymbol{F}\left(1-z\right)+\mu\left(1-\frac{1}{z}\right)\boldsymbol{I}\right)R + \overline{\boldsymbol{Y}}\left(z\right)\label{near-ind-eq}       
    \end{equation}
    \begin{equation}
        \hat{q}\left(z;0\right) = \pi_0\frac{1+U\left(z\right)}{1+M\ln\left(z\right)/\varepsilon +V\left(z\right)}\label{near-exp-eq}
    \end{equation}
    where
    \begin{align*}
                \overline{\boldsymbol{Y}}\left(z\right)
                &=\left(1-\frac{1}{z}\right)\hat{\boldsymbol{q}}\left(z\right)
                \Big(\left(-z \boldsymbol{F}+\mu \boldsymbol{I}\right)\boldsymbol{R}\Big)^2 - \mu\left(1-\frac{1}{z}\right)\boldsymbol{\pi}_\text{idle}\boldsymbol{R}\left(-z \boldsymbol{F}+\mu \boldsymbol{I}\right)\boldsymbol{R} - \mu\boldsymbol{\pi}_\text{idle}\boldsymbol{R}\\
                U\left(z\right) &=-\mu^{-1}\varepsilon^{-1}\left(1-\frac{1}{z}\right)\overline{\boldsymbol{Y}}\left(z\right)\left(-z \boldsymbol{F}+\mu \boldsymbol{I}\right)e^T\\
        V\left(z\right) &= \varepsilon^{-1}\left(1-\frac{1}{z}\right)\left(-z\frac{\overline{\lambda}}{\mu}+\mu^{-1}\boldsymbol{\pi}\left(-z \boldsymbol{F}+\mu \boldsymbol{I}\right)\boldsymbol{R}\left(-z \boldsymbol{F}+\mu \boldsymbol{I}\right)e^T\right) - M\ln \left(z\right)/\varepsilon.
    \end{align*}

\label{def:light-tailed}
    A family of IMAP queues that meets our heavy-traffic conditions (\cref{def:HT_cond}) with a \emph{light-tailed phase process} is a family of IMAP queues that meets our heavy-traffic conditions where:
    \begin{enumerate}
        \item                 
        \label{item:light-tailed-1} For all $z\in\left(0,1\right)$, as $\varepsilon\to 0$:
        \begin{enumerate}
            \item \label{item:light-tailed-1a} $\left(1-z^{-\varepsilon}\right)\hat{\boldsymbol{q}}\left(z^\varepsilon;\varepsilon\right) \left(\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{\left(\varepsilon\right)} \boldsymbol{I}\right)\boldsymbol{R}^{(\varepsilon)}\right)^2\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu \boldsymbol{I}\right)e^T\to 0$

            \item \label{item:light-tailed-1b}$\mu^{\left(\varepsilon\right)}\left(1-z^{\varepsilon}\right)\boldsymbol{\pi}^{\left(\varepsilon\right)}_0\boldsymbol{R}^{(\varepsilon)}\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{\left(\varepsilon\right)} \boldsymbol{I}\right)\boldsymbol{R}^{(\varepsilon)}\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{(\varepsilon)} \boldsymbol{I}\right)e^T\to 0$
            
            \item \label{item:light-tailed-1c}$\boldsymbol{\pi}^{\left(\varepsilon\right)}_0\boldsymbol{R}^{(\varepsilon)}\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{(\varepsilon)} \boldsymbol{I}\right)e^T\to 0$                                                                                                   

            \item \label{item:light-tailed-1d}$\left(1-z^{-\varepsilon}\right)\boldsymbol{\pi}^{\left(\varepsilon\right)}\left(\mu^{\left(\varepsilon\right)} \boldsymbol{I} - z^{\varepsilon}  \boldsymbol{F}^{(\varepsilon)}\right)\boldsymbol{R}^{(\varepsilon)}e^T\to 0$

            \item \label{item:light-tailed-1e} $\left(1-z^{-\varepsilon}\right)\hat{\boldsymbol{q}}\left(z^\varepsilon;\varepsilon\right)\left(\left(\mu^{\left(\varepsilon\right)}I-z \boldsymbol{F}^{(\varepsilon)}\right)\boldsymbol{R}^{(\varepsilon)}\right)^2e^T\to 0$

            \item \label{item:light-tailed-1f}$\mu^{\left(\varepsilon\right)}\left(1-z^{-\varepsilon}\right)\boldsymbol{\pi}_\text{idle}^{\left(\varepsilon\right)}\boldsymbol{R}^{(\varepsilon)}\left(\mu^{\left(\varepsilon\right)}I-z^{\varepsilon} \boldsymbol{F}^{(\varepsilon)}\right)\boldsymbol{R}^{(\varepsilon)}e^T\to 0$

            \item \label{item:light-tailed-1g}$\mu^{\left(\varepsilon\right)}\boldsymbol{\pi}_\text{idle}^{\left(\varepsilon\right)}\boldsymbol{R}^{(\varepsilon)}e^T\to 0$
        \end{enumerate}

        \item \label{def_to0i}For all $z\in\left(0,1\right)$ and all $i\in \mathbb{N}_0$ as $\varepsilon\to 0$:
        \begin{enumerate}
            \item \label{item:light-tailed-2a}$[\left(1-z^{-\varepsilon}\right)\boldsymbol{\pi}^{\left(\varepsilon\right)}\left(\mu^{\left(\varepsilon\right)}I-z \boldsymbol{F}^{(\varepsilon)}\right)\boldsymbol{R}^{(\varepsilon)}]_i\to 0$

            \item \label{item:light-tailed-2b}$[\hat{\boldsymbol{q}}\left(z^\varepsilon;\varepsilon\right) \left(\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{\left(\varepsilon\right)} \boldsymbol{I}\right)R\right)^2]_i \to 0$

            \item \label{item:light-tailed-2c}$[\mu^{\left(\varepsilon\right)}\left(1-z^{-\varepsilon}\right)\boldsymbol{\pi}^{\left(\varepsilon\right)}_0\boldsymbol{R}^{(\varepsilon)}\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)} + \mu^{\left(\varepsilon\right)}I\right)\boldsymbol{R}^{(\varepsilon)}]_i\to 0$

            \item \label{item:light-tailed-2d}$[\mu^{\left(\varepsilon\right)}\boldsymbol{\pi}^{\left(\varepsilon\right)}_0\boldsymbol{R}^{(\varepsilon)}]_i\to 0$
        \end{enumerate}
    \end{enumerate}

<!-- proof-1: body_md -->
We now shift our focus to the scaled system $\varepsilon Q^{(\varepsilon)}$ rather than the original system $Q^{(\varepsilon)}$. Notice that 
    \begin{align*}
        \mathbb{E}[z^{\varepsilon Q^{(\varepsilon)}}] &= \mathbb{E}[(z^\varepsilon)^{Q^{(\varepsilon)}}]= \overline{q}(z^\varepsilon;\varepsilon)
    \end{align*}
    and
    \begin{align*}
        \mathbb{E}\left[z^{\varepsilon Q^{(\varepsilon)}}\mathbb{I}\{Y^{(\varepsilon)} = i\}\right] &= \mathbb{E}\left[(z^\varepsilon)^{Q^{(\varepsilon)}}\mathbb{I}\{Y^{(\varepsilon)} = i\}\right]= \hat{q}(z^\varepsilon;i, \varepsilon).
    \end{align*}
    Therefore, our analysis of $\varepsilon Q^{(\varepsilon)}$ can rely on our existing near-independence and near-exponential equations \cref{near-ind-eq,near-exp-eq} for $Q^{(\varepsilon)}$.
    
    We break our result into three parts.
    In part 1, we show that $[\varepsilon Q^{(\varepsilon)} | Y^{(\varepsilon)} = 0]$ is asymptotically exponential or  
    \begin{align}
        \lim_{\varepsilon\to 0} \hat{q}(z^\varepsilon;0,\varepsilon) &= \frac{\pi^{(0)}_0}{1+M^{(0)}\ln z}. \label{eq:Thm-1-0res}
    \end{align}
    Next, in part 2, we show this for any $i\in \mathbb{N}_0,$ $[\varepsilon Q^{(\varepsilon)} | Y^{(\varepsilon)} = i]$ is asymptotically exponential or  
    \begin{align}
        \lim_{\varepsilon\to 0} \hat{q}(z^\varepsilon;i,\varepsilon) &= \frac{\pi^{(0)}_i}{1+M^{(0)}\ln z}.\tag{\ref{eq:Thm1-res-b}}
    \end{align}
    Finally, in part 3,  we show that $\varepsilon Q^{(\varepsilon)}$ (absent any conditioning on $Y^{(\varepsilon)}$) is asymptotically exponential or
    \begin{align*}
        \lim_{\varepsilon\to 0} \overline{q}(z^\varepsilon;\varepsilon) &= \frac{1}{1+M^{(0)}\ln z}. \tag{\ref{eq:Thm1-res-a}}
    \end{align*}

    First, we show that $[\varepsilon Q^{(\varepsilon)}|Y^{(0)}=0]$ is asymptotically exponential by proving \cref{eq:Thm-1-0res}.
    
    By \cref{ind_exp_lemma}, for any member of the parameterized family we have
    \begin{align*}
        \hat{\boldsymbol{q}}\left(z;\varepsilon\right) &= \hat{q}\left(z;0,\varepsilon\right)\frac{\boldsymbol{\pi}^{\left(\varepsilon\right)}}{\pi_0^{\left(\varepsilon\right)}} +\hat{q}\left(z;0,\varepsilon\right)\frac{\boldsymbol{\pi}^{\left(\varepsilon\right)}}{\pi_0^{\left(\varepsilon\right)}}\left( \boldsymbol{F}^{(\varepsilon)}\left(1-z\right)+\mu^{\left(\varepsilon\right)}\left(1-\frac{1}{z}\right)\boldsymbol{I}\right)\boldsymbol{R}^{(\varepsilon)} + \overline{\boldsymbol{Y}}\left(z;\varepsilon\right)\\
        \hat{q}\left(z;0,\varepsilon\right) &= \pi_0^{\left(\varepsilon\right)}\frac{1+U\left(z;\varepsilon\right)}{1+M^{(0)}\ln \left(z\right)/\varepsilon +V\left(z;\varepsilon\right)}
    \end{align*}

    We begin by showing $U\left(z^\varepsilon;\varepsilon\right) \to 0$. We can consider $U\left(z^\varepsilon;\varepsilon\right)$ in two parts:
 
    \begin{align*}
        U\left(z^\varepsilon;\varepsilon\right)&=\left(-\frac{1-z^{-\varepsilon}}{\varepsilon\mu^{\left(\varepsilon\right)}}\right)\cdot\left(\overline{\boldsymbol{Y}}\left(z^\varepsilon;\varepsilon\right)\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{(\varepsilon)} \boldsymbol{I}\right)e^T\right)
    \end{align*}
    
    We analyze the two multiplicands separately. Because $-\frac{1-z^{-\varepsilon}}{\varepsilon\mu^{\left(\varepsilon\right)}}\to -\frac{\ln\left(z\right)}{\mu^{\left(\varepsilon\right)}}$ and $|-\frac{\ln\left(z\right)}{\mu^{\left(\varepsilon\right)}}|<\infty$ for any $z>0$, if $\overline{\boldsymbol{Y}}\left(z^\varepsilon;\varepsilon\right)\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{(\varepsilon)} \boldsymbol{I}\right)e^T\to 0$ then $U\left(z^\varepsilon;\varepsilon\right)\to 0$ as well.

    Expanding $\overline{\boldsymbol{Y}}\left(z^\varepsilon;\varepsilon\right)\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{(\varepsilon)} \boldsymbol{I}\right)e^T$, we have 3 distinct terms:
    \begin{align}
        &\overline{\boldsymbol{Y}}\left(z^\varepsilon;\varepsilon\right)\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{(\varepsilon)} \boldsymbol{I}\right)e^T\nonumber\\
        &=\left(1-z^{-\varepsilon}\right)\hat{\boldsymbol{q}}\left(z^\varepsilon;\varepsilon\right) \left(\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{\left(\varepsilon\right)} \boldsymbol{I}\right)R\right)^2\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{(\varepsilon)} \boldsymbol{I}\right)e^T\label{eq:Y_FI_e_expdd_1}\\
        &-\mu^{\left(\varepsilon\right)}\left(1-z^{\varepsilon}\right)\boldsymbol{\pi}^{\left(\varepsilon\right)}_0\boldsymbol{R}^{(\varepsilon)}\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{\left(\varepsilon\right)} \boldsymbol{I}\right)\boldsymbol{R}^{(\varepsilon)}\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{(\varepsilon)} \boldsymbol{I}\right)e^T \label{eq:Y_FI_e_expdd_2}\\
        &-\mu^{\left(\varepsilon\right)}\boldsymbol{\pi}^{\left(\varepsilon\right)}_0\boldsymbol{R}^{(\varepsilon)}\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{(\varepsilon)} \boldsymbol{I}\right)e^T \label{eq:Y_FI_expdd_3}
    \end{align}

    Because the family of IMAP queues has a light-tailed phase process, Properties~\ref{item:light-tailed-1a}, \ref{item:light-tailed-1b}, and \ref{item:light-tailed-1c} of \cref{def:light-tailed} guarantee that each of \cref{eq:Y_FI_e_expdd_1,eq:Y_FI_e_expdd_2,eq:Y_FI_expdd_3}, respectively, are negligible as $\varepsilon\to 0$.

    Thus, $U(z^\varepsilon;\varepsilon) \to 0$ as $\varepsilon\to 0$.

    Next, let us show $V\left(z^\varepsilon;\varepsilon\right)\to 0$. It is useful to reformulate $V\left(z^\varepsilon; \varepsilon\right)$ by adding and subtracting $\varepsilon^{-1}\left(1-z^{-\varepsilon}\right)$ and then carefully rearranging terms.
    \begin{align}
        &V\left(z^\varepsilon; \varepsilon\right) = \varepsilon^{-1}\left(1-z^{-\varepsilon}\right)\left(-z^\varepsilon\frac{\overline{\lambda}^{\left(\varepsilon\right)}}{\mu^{\left(\varepsilon\right)}}\right)\nonumber\\
        &+ \varepsilon^{-1}\left(1-z^{-\varepsilon}\right)\left(\left(\mu^{\left(\varepsilon\right)}\right)^{-1}\boldsymbol{\pi}^{\left(\varepsilon\right)}\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{(\varepsilon)} \boldsymbol{I}^{\left(\varepsilon\right)}\right)\boldsymbol{R}^{(\varepsilon)}\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{(\varepsilon)} \boldsymbol{I}\right)e^T\right)\nonumber \\
        &- M^{(0)}\ln\left(z\right)\nonumber\\
        &=\varepsilon^{-1}\left(1-z^{-\varepsilon}\right) \left(1-1-z^{\varepsilon}\frac{\overline{\lambda}^{\left(\varepsilon\right)}}{\mu^{\left(\varepsilon\right)}}\right)\nonumber \\
        &+ \varepsilon^{-1}\left(1-z^{-\varepsilon}\right)\left(-1+\left(\mu^{\left(\varepsilon\right)}\right)^{-1}\boldsymbol{\pi}^{\left(\varepsilon\right)}\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{\left(\varepsilon\right)} \boldsymbol{I}\right)\boldsymbol{R}^{(\varepsilon)}\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{\left(\varepsilon\right)} \boldsymbol{I}\right)e^T\right) \nonumber \\\
        &- M^{(0)}\ln\left(z\right)\nonumber\\
        &=\varepsilon^{-1}\left(1-z^{-\varepsilon}\right) \left(1-z^{\varepsilon}\frac{\overline{\lambda}^{\left(\varepsilon\right)}}{\mu^{\left(\varepsilon\right)}}\right) \label{eq:V-expdd-1}\\
        &+ \varepsilon^{-1}\left(1-z^{-\varepsilon}\right)\left(-1+\left(\mu^{\left(\varepsilon\right)}\right)^{-1}\boldsymbol{\pi}^{\left(\varepsilon\right)}\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{\left(\varepsilon\right)} \boldsymbol{I}\right)\boldsymbol{R}^{(\varepsilon)}\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{\left(\varepsilon\right)} \boldsymbol{I}\right)e^T\right) \label{eq:V-expdd-2}\\
        &- M^{(0)}\ln\left(z\right) \label{eq:V-expdd-3}
    \end{align}
    Let us consider \cref{eq:V-expdd-1} separately from the other terms of $V\left(z^\varepsilon;\varepsilon\right)$, \cref{eq:V-expdd-2,eq:V-expdd-3}. 

    Beginning with \cref{eq:V-expdd-1}, we note that $\varepsilon^{-1}\left(1-z^{-\varepsilon}\right)\to-\ln z$ (where $|-\ln z| < \infty$ for all $z\in (0,1)$) as $\varepsilon\to 0$. Next, using the fact that $\frac{\overline{\lambda}^{(\varepsilon)}}{\mu^{(\varepsilon)}} = (1-\varepsilon)$ we can see that $\left(1-z^{\varepsilon}\frac{\overline{\lambda}^{\left(\varepsilon\right)}}{\mu^{\left(\varepsilon\right)}}\right) = \left(1-z^{\varepsilon}(1-\varepsilon)\right)\to 0$ as $\varepsilon\to 0$. Thus, their product, \cref{eq:V-expdd-1}, is negligible as $\varepsilon\to 0$.

    Next, we move onto \cref{eq:V-expdd-2,eq:V-expdd-3}. We again note that $\varepsilon^{-1}\left(1-z^{-\varepsilon}\right)\to-\ln z$. By property \ref{M_conv} of the heavy-traffic family conditions (\cref{def:HT_cond}), we can see that
    \[-1+\left(\mu^{\left(\varepsilon\right)}\right)^{-1}\boldsymbol{\pi}^{\left(\varepsilon\right)}\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{\left(\varepsilon\right)} \boldsymbol{I}\right)\boldsymbol{R}^{(\varepsilon)}\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{\left(\varepsilon\right)} \boldsymbol{I}\right)e^T\to -M^{(0)}\]

    Thus, \cref{eq:V-expdd-2} approaches $M^{(0)}\ln z$ in the $\varepsilon\to 0$ limit and \cref{eq:V-expdd-2,eq:V-expdd-3} cancel in this limit.

    We therefore conclude that $V\left(z^\varepsilon;\varepsilon\right)\to 0$.

    Having shown that $U\left(z^\varepsilon;\varepsilon\right), V\left(z^\varepsilon;\varepsilon\right)\to 0$, we have our desired result: \begin{equation}
        \hat{q}\left(z^\varepsilon;0,\varepsilon\right) = \pi_0^{\left(\varepsilon\right)} \frac{1+U\left(z^\varepsilon;\varepsilon\right)}{1+M^{(0)}\ln z - V\left(z^\varepsilon;\varepsilon\right)} \to \frac{\pi_0^{\left(0\right)}}{1+M^{(0)}\ln z} \tag{\ref{eq:Thm-1-0res}}
    \end{equation}

    We have thus shown that the scaled queue length process is asymptotically exponential when conditioned on $Y^{\left(\varepsilon\right)}=0$.

     We begin with \cref{eq:Qii_expdd_2}

     \end{align*}

     Because $\frac{\hat{q}\left(z^\varepsilon;0,\varepsilon\right)}{\pi_0^{\left(\varepsilon\right)}}\to \frac{1}{1-M^{(0)}\ln z}$ and, by property \ref{item:light-tailed-2a} of our light-tailed phase process conditions, $\left[\left(1-z^{-\varepsilon}\right)\boldsymbol{\pi}^{\left(\varepsilon\right)}\left(\mu^{\left(\varepsilon\right)}I-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}\right)\boldsymbol{R}^{(\varepsilon)}\right]_i\to 0$, the second summand approaches $0$.

     Moving onto the third summand, we can expand $\overline{Y}^i\left(z^\varepsilon;\varepsilon\right)$ to
     \begin{align}
        \overline{Y}^i\left(z^\varepsilon;\varepsilon\right) 
        &=\left[\left(1-z^{-\varepsilon}\right)\hat{\boldsymbol{q}}\left(z^\varepsilon;\varepsilon\right) \left(\left(-z^{\varepsilon} \boldsymbol{F}^{(\varepsilon)}+\mu^{\left(\varepsilon\right)} \boldsymbol{I}\right)\boldsymbol{R}^{(\varepsilon)}\right)^2\right]_i \\
        &\quad- \left[\mu^{\left(\varepsilon\right)}\left(1-z^{-\varepsilon}\right)\boldsymbol{\pi}_\text{idle}^{\left(\varepsilon\right)}\boldsymbol{R}^{(\varepsilon)}\left(-z^{\varepsilon} \boldsymbol{F}^{(\varepsilon)}+\mu^{\left(\varepsilon\right)} \boldsymbol{I}\right)R\right]_i \\
        &\quad - \left[\mu^{\left(\varepsilon\right)}\boldsymbol{\pi}_\text{idle}^{\left(\varepsilon\right)}\boldsymbol{R}^{(\varepsilon)}\right]_i
     \end{align}

     The light-tailed phase process conditions, specifically properties \ref{item:light-tailed-2a}, \ref{item:light-tailed-2b}, and \ref{item:light-tailed-2c}, provide that each of the three parts above go to $0$ as $\varepsilon\to 0$ and therefore $\overline{Y}^i\left(z^\varepsilon;\varepsilon\right) \to 0$ as well.

    This expression exactly states that $\varepsilon Q^{\left(\varepsilon\right)}$ and $Y^{\left(\varepsilon\right)}$ are asymptotically independent and ${[\varepsilon Q^{\left(\varepsilon\right)}|Y^{(0)}=i]}$ is asymptotically exponential with mean $M^{(0)}$.

Let us prove the three bounds in order.
    \begin{enumerate}[(a)]
        \item $\pi^{\left(\varepsilon\right)}_n \leq C_1a^n$.
        \label{pi_bd}

        The bounded rate conditions (\cref{def:rtbdd}) guarantee that there is some $a\in (0,1)$ and $N$ such that for all $n\geq N$, $\frac{\alpha^{\left(\varepsilon\right)}_{n, n+1}}{\alpha^{\left(\varepsilon\right)}_{n+1,n}}\leq a$ and therefore ${\pi^{\left(\varepsilon\right)}_n = \pi_N^{\left(\varepsilon\right)} \prod_{m=N}^n \frac{\alpha^{\left(\varepsilon\right)}_{n, n+1}}{\alpha^{\left(\varepsilon\right)}_{n+1,n}} \leq \pi^{\left(\varepsilon\right)}_N a^{n-N}}$. Letting $C_1= a^{-N}$, this bound trivially holds for all for all $n\leq N$ and we have our desired bound for all $n$.

        This bound also holds for $\pi^{(\varepsilon)}_{n,0}$ and $\hat{q}(z;n)$ as $\pi^{(\varepsilon)}_{n,0}\leq \pi^{(\varepsilon)}_n$ and $\hat{q}(z;n)\leq \pi^{(\varepsilon)}_n$.

        \item $r^{\left(\varepsilon\right)}_{i,j} \leq C_2 \tilde{b}^{\left(j-i\right)^+}$.
        \label{rijbd}

        It is useful to consider this bound in three cases: when $i= j$, when $i\geq j$, and when $i<j$.
        
        To simplify notation we let $\mathbb{P}(x_1\to x_2\to \dots \to x_n/y_1,y_2, \dots, y_m)$ be the probability that the phase process begins in state $x_1$ and transitions (over any number of steps) to state $x_2$ \dots and then finally to state $x_n$ without entering states $y_1, y_2,\dots, y_m$.
        \begin{enumerate}

            \item Case 1: $i= j$.
            
            Starting in state $i$ there is some probability, $p_i$, that the process reaches state $0$ before returning to state $i$. The number of visits to state $i$ is distributed as a shifted geometric variable, $K_{i,i}\sim\text{Geom}\left(p_i\right)$ (where $K_{i,i} \in \{1, 2, \dots\}$). Thus if we can uniformly lower bound $p_i$ for all $i$ and $\varepsilon$ then we can uniformly upper bound $r^{(\varepsilon)}_{i,i}$.

            Let us start with the case when $i\leq N$. In this case we can naively bound the probability of moving from state $i$ to state $0$ without returning to $i$ by the probability of transitioning downward $i$ times in a row: ${p_{i\downarrow0}:=\prod_{m=1}^i \frac{\alpha_{m,m-1}}{\alpha_{m,m-1}+\alpha_{m,m+1}}}$.

            When $i > N$ we can break the process of moving from $i$ to $0$ into two components: moving from $i$ to $N$ without returning to $i$ and then moving from $N$ to $0$ without returning to $i$. We already have a lower bound on the probability of transitioning from $N$ to $0$ without returning to state $i$, ${p_{N\downarrow 0} =\prod_{m=1}^N \frac{\alpha^{(\varepsilon)}_{m,m-1}}{\alpha_{m,m-1}^{(\varepsilon)}+\alpha_{m,m+1}^{(\varepsilon)}}}$, so we only need to bound the probability $\mathbb{P}(i\to N/i)$.

            To move from $i$ to $N$ the process has to first transition downward from $i$ to $i-1$, which occurs with probability at least $1-b$ by our uniform stability conditions. From there we can think of the process as akin to a gambler's ruin which either reaches state $i$ or state $N$ first.

            Consider a biased gambler's ruin on $\{N, N+1, \dots, i\}$ with ``success" probability $b$ starting in state $i-1$. The probability the gambler runs out of money (i.e., reaches state $N$ before state $i$) is $\frac{1-\left(\frac{b}{1-b}\right)^1}{1-\left(\frac{b}{1-b}\right)^{i-N}}$. We can uniformly upper bound this probability for any $i$ by $1-\frac{b}{1-b}>0$.

            Because the probability of transitioning upward for any state $i\geq N$ is bounded above by $b$, a coupling argument shows that the probability of the process reaching state $i$ before state $N$ from state $i-1$ has the same upper bound of $1-\frac{b}{1-b}$.

            Letting \[\underline{p}:=\left(1-b\right)\left(1-\frac{b}{1-b}\right)\prod_{m=1}^N \left(\frac{\alpha_{m,m-1}}{\alpha_{m,m-1}+\alpha_{m,m+1}}\right)\] we have for any state $i$, ${p_i \geq \underline{p}>0}$
            and therefore 
            \[r_{i,i} \leq T_{\text{max}} \frac{1-\underline{p}}{\underline{p}}\]
            where $T_{\text{max}}$ is the maximum expected time in state for any state as defined in \cref{def:rtbdd}.
            \item Case 2: $i>j$.

            Next we consider the case when $i>j$. Because the phase process is Markovian, if the process enters state $j$ after starting in state $i$ then the expected time in state $j$ before reaching state $0$ is the same as if the process had started in state $j$. With probability $1$, the phase process started in state $i$ eventually reaches state $0$ so, as a birth-death process, it must pass through state $j$ at least once on the way to $0$. Therefore,
            \[r_{i,j}=r_{j,j}\leq T_{max} \frac{1-\underline{p}}{\underline{p}}.\]

            \item Case 3: $i<j$.

            Lastly we consider the case when $i<j$. In this case, the phase process might reach $0$ without ever going through state $j$. If we can bound the probability that the process started in $i$ visits $j$ before $0$ then we can bound $r_{i,j}$.


            Consider the probability in two cases: from state $i$ to state $j$ the process either transitions through state $N$ or not:
            \[\mathbb{P}(i\to j/0) = \mathbb{P}(i\to j/0,N) + \mathbb{P}(i\to N\to j/0)\]

            We can analyze these two possibilities separately. The probability the process moves from from $i$ to $j$ without reaching state $N$ is bounded by $(\frac{b}{1-b})^{j-i}$ by coupling with a biased gambler's ruin.

            Next consider the probability of touching state $N$ but not state $0$ on the way to state $j$ from state $i$. We can immediately upper bound this by the probability of simply moving from state $N$ to state $j$ from state $N$.
            \[\mathbb{P}(i\to N\to j/0) \leq \mathbb{P}(N\to j/0)\]

            We can also notice that the probability that the system reaches $j$ before $0$ after starting in $N$ with some number of returns is the same as the probability that, given this is the last time the system is in $N$, the system reaches state $j$ before $0$. Thus
            \[\mathbb{P}(N\to j/0) = \frac{\mathbb{P}(N\to j/0,N)}{\mathbb{P}(N\to j/0, N) + \mathbb{P}(N\to 0/j,N)} \leq \frac{\mathbb{P}(N\to j/0,N)}{\mathbb{P}(N\to 0/j,N)}\]

            We thus have a bound in terms of two easier to analyze quantities: $\mathbb{P}(N\to j/0,N)$ and $\mathbb{P}(N\to 0/j,N)$. We have an existing lower bound on $\mathbb{P}(N\to 0/j,N)$ in $\underline{p} = \prod_{i=1}^{N} \frac{\alpha_{i, i-1}}{\alpha_{i,i-1}+\alpha_{i, i+1}}$. We can also use our gambler's ruin technique again to upper bound $\mathbb{P}(N\to j/0,N)$. Once the process moves from state $N$ to $N+1$ (with probability no more than $b$), we can couple the process with a gambler's ruin on $\{N, N+1, N+2, \dots, j\}$. The probability the gambler's ruin reaches state $j$ can be upper bounded by $(\frac{b}{1-b})^{j-N-1}$.

            Taken together we have a bound on the probability of reaching state $j$,
            \begin{align*}
                \mathbb{P}(i\to j/0) &\leq (\frac{b}{1-b})^{j-i} + \frac{1}{\underline{p}} (\frac{b}{1-b})^{j-N-1},
            \end{align*}
            and can find our final bound
            \begin{align*}
                r_{i,j}^{(\varepsilon)} &\leq T_{max} \frac{1-\underline{p}}{\underline{p}}\left(\left(\frac{b}{1-b}\right)^{j-i} + \frac{1}{\underline{p}} \left(\frac{b}{1-b}\right)^{j-N-1}\right)\\
                &= T_{max} \frac{1-\underline{p}}{\underline{p}}\left(1 + \frac{1}{\underline{p}} \left(\frac{b}{1-b}\right)^{i-N-1}\right) \left(\frac{b}{1-b}\right)^{j-i}\\
                &\leq T_{max} \frac{1-\underline{p}}{\underline{p}}\left(1 + \frac{1}{\underline{p}} \left(\frac{b}{1-b}\right)^{-N-1}\right) \left(\frac{b}{1-b}\right)^{j-i}
            \end{align*}

            

            
        \end{enumerate}

        Taking these three cases together we have our desired bound that for any $i,j$,
        \begin{align*}
            r_{i,j}^{(\varepsilon)} \leq C_2 (\frac{p}{1-p})^{(j-i)^+},
        \end{align*}
        where $C_2 :=T_{max} \frac{1-\underline{p}}{\underline{p}}\left(1 + \frac{1}{\underline{p}} \left(\frac{b}{1-b}\right)^{-N-1}\right) \geq T_{max} \left(\frac{b}{1-b}\right)$.
        
        \item $\left|[\boldsymbol{R}^{(\varepsilon)}e^T]_i\right| \leq C_3 i$.
        \label{ReTbd}
        \begin{align*}
            \left|[\boldsymbol{R}^{(\varepsilon)}e^T]_i\right| &= \left|\sum_{j=0}^\infty r_{i,j}^{\left(\varepsilon\right)}\right|
            \leq \left|\sum_{j=0}^{i-1} r_{i,j}^{\left(\varepsilon\right)}\right| + \left|\sum_{j=i}^\infty r_{i,j}^{\left(\varepsilon\right)}\right|
        \end{align*}
            Using property \ref{eq:birth-bd-2} we can bound $r_{i,j}^{(\varepsilon)}$ to find that
        \begin{align*}
            \left|[\boldsymbol{R}^{(\varepsilon)}e^T]_i\right| &\leq C_2 i + C_2 \sum_{j=i}^\infty a ^{j-i}
            = C_2i + C_2 \frac{1}{1-a}
            \leq C_3 i,
        \end{align*}
        where $C_3:=C_2\left(1 + \frac{1}{1-a}\right)$.

        
    \end{enumerate}
    We thus have our desired bounds on $\pi_n^{(\varepsilon)}$, $r_{i,j}^{(\varepsilon)}$, and $\left|[\boldsymbol{R}^{(\varepsilon)}e^T]_i\right|$.

            |\pi^{(0)}_n - \pi^{(\varepsilon)}_n | &= \left|\pi^{(0)}_0 \frac{(\lambda^{(0)})^n}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)} -\pi^{(\varepsilon)}_0 \frac{(\lambda^{(\varepsilon)})^n}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)}\right| \\ \nonumber
            &= \left|\pi^{(0)}_0 \frac{(\lambda^{(0)})^n}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)} -\pi^{(\varepsilon)}_0 \frac{(\lambda^{(\varepsilon)})^n + (\lambda^{(0)})^n - (\lambda^{(0)})^n}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)}\right| \\ \nonumber
            &= \left|(\pi^{(0)}_0-\pi_0^{(\varepsilon)}) \frac{(\lambda^{(0)})^n}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)} -\pi^{(\varepsilon)}_0 \frac{(\lambda^{(\varepsilon)})^n - (\lambda^{(0)})^n}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)}\right| \\
            &\leq \left|(\pi^{(0)}_0-\pi_0^{(\varepsilon)}) \frac{(\lambda^{(0)})^n}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)}\right| \label{eq:pi0mpieps-1}\\
            &+ \left|\pi^{(\varepsilon)}_0 \frac{(\lambda^{(\varepsilon)})^n - (\lambda^{(0)})^n}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)}\right|\label{eq:pi0mpieps-2}
        \end{align}

        We can consider \cref{eq:pi0mpieps-1} and \cref{eq:pi0mpieps-2} separately. Let's begin with \cref{eq:pi0mpieps-1}.
        \begin{align*}
            \left|(\pi^{(0)}_0-\pi_0^{(\varepsilon)}) \frac{(\lambda^{(0)})^n}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)}\right| &= \left|-\varepsilon \frac{\mu_2}{\mu_1} \frac{(\lambda^{(0)})^n}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)}\right|\\
            &=\varepsilon \frac{\mu_2}{\mu_1} \frac{(\lambda^{(0)})^n}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)}
        \end{align*}
        Let $\hat{n} = 1+\frac{e\lambda^{(0)}}{\theta}$. For all $n\geq \hat{n}$, $\frac{\lambda^{(0)}}{\left(\mu_1+\left(n-1\right)\theta\right)} \leq e^{-1}$. Letting $A:=\frac{\mu_2}{\mu_1} \sup_{n\leq \hat{n}} e^n\frac{(\lambda^{(0)})^n}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)}$, we have that
        \begin{align*}
            \left|(\pi^{(0)}_0-\pi_0^{(\varepsilon)}) \frac{(\lambda^{(0)})^n}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)}\right| &= \varepsilon \frac{\mu_2}{\mu_1} \frac{(\lambda^{(0)})^n}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)}
            \leq A\varepsilon e^{-n}
        \end{align*}

        Next we turn to \cref{eq:pi0mpieps-2}. By factoring $(\lambda^{(0)}-\lambda^{(\varepsilon)})$ from $((\lambda^{(0)})^n-(\lambda^{(\varepsilon)})^n)$ we can perform a similar analysis,
        \begin{align*}
            \left|\pi^{(\varepsilon)}_0 \frac{(\lambda^{(\varepsilon)})^n - (\lambda^{(0)})^n}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)}\right| &= \pi^{(\varepsilon)}_0 \frac{(\lambda^{(0)})^n-(\lambda^{(\varepsilon)})^n}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)}\\
            &= \pi^{(\varepsilon)}_0 \frac{(\lambda^{(0)}-\lambda^{(\varepsilon)})\sum_{k=0}^{n-1}(\lambda^{(0)})^k(\lambda^{(\varepsilon)})^{n-1-k}}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)}\\
            &\leq \frac{(\lambda^{(0)}-\lambda^{(\varepsilon)})n(\lambda^{(0)})^{n-1}}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)}
        \end{align*}

        Because $\frac{(n+1)(\lambda^{(0)})^{n}}{\prod_{i=1}^{n+1} \left(\mu_1+\left(i-1\right)\theta\right)}/\frac{n(\lambda^{(0)})^{n-1}}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)} = \left(\frac{n+1}{n}\right)\frac{\lambda^{(0)}}{\mu_1 + n\theta}\to 0$, there exists a $\hat{m}$ such that for all $n\geq \hat{m}$, \[\frac{(n+1)(\lambda^{(0)})^{n}}{\prod_{i=1}^{n+1} \left(\mu_1+\left(i-1\right)\theta\right)}/\frac{n(\lambda^{(0)})^{n-1}}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)} \leq e^{-1}.\] Letting $B:= \frac{\mu_2}{\mu_1} \sup_{n\leq \hat{m}} e^n\frac{n(\lambda^{(0)})^{n-1}}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)}$ we have
        \begin{align*}
            \left|\pi^{(\varepsilon)}_0 \frac{(\lambda^{(\varepsilon)})^n - (\lambda^{(0)})^n}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)}\right| 
            &\leq \frac{(\lambda^{(0)}-\lambda^{(\varepsilon)})n(\lambda^{(0)})^{n-1}}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)}\leq B\varepsilon e^{-n}
        \end{align*}

        Using our bounds on \cref{eq:pi0mpieps-1} and \cref{eq:pi0mpieps-2}, we now have our desired bound,
                \begin{align*}
            \left|\pi^{(\varepsilon)}_n - \pi^{(0)}_n\right| &\leq \left|(\pi^{(0)}_0-\pi_0^{(\varepsilon)}) \frac{(\lambda^{(0)})^n}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)}\right| + \left|\pi^{(\varepsilon)}_0 \frac{(\lambda^{(\varepsilon)})^n - (\lambda^{(0)})^n}{\prod_{i=1}^n \left(\mu_1+\left(i-1\right)\theta\right)}\right|\\
            &\leq A\varepsilon e^{-n} + B\varepsilon e^{-n}= \hat{C}_1 \varepsilon e^{-n}
        \end{align*}
        where $\hat{C}_1 := A+B$.

        
        

        

        \item Bounds on $|r_{i,j}^{(0)}-r^{(\varepsilon)}_{i,j}|$.
        
        With our bounds on $|\pi_n^{(\varepsilon)}-\pi_n^{(0)}|$ proven, we next show that
        \begin{equation}
            |r_{i,j}^{(0)}-r^{(\varepsilon)}_{i,j}| \leq \hat{C}_2(\lambda^{(0)} - \lambda^{(\varepsilon)}). \label{eq:rij0-rijeps-bd}
        \end{equation}

        Let $T_{i,j}$ be the random amount of time that the phase process spends in state $j$ between starting in state $i$ and reaching state $0$. Note that $r_{i,j} = \mathbb{E}[T_{i,j}]$.
        
        Consider a coupled pair of systems: one, $Y^{(0)}$, with arrival rate $\lambda^{(0)}$ and one, $Y^{(\varepsilon)}$, with rate $\lambda^{(\varepsilon)}$ for some $\varepsilon>0$. 

        When $(Y^{(0)}, Y^{(\varepsilon)}) = (n,n)$, the pair jointly transition upward at rate $\lambda^{(\varepsilon)}$. The pair jointly transition downward at rate $\mu_1 + (n-1)\theta$. With rate $\lambda^{(0)} - \lambda^{(\varepsilon)}$, $Y^{(0)}$ transitions upward and $Y^{(\varepsilon)}$ doesn't transition. When $Y^{(0)} \ne Y^{(\varepsilon)}$, the two evolve independently. This coupling ensures that $Y^{(0)} \geq Y^{(\varepsilon)}$ almost surely.

        Consider two possibilities: if the systems do not diverge before $Y^{(0)}$ reaches state $0$ then the two have moved in lockstep the entire duration and have spent exactly the same time in state $j$. If the two processes do diverge before this point then the two processes may spend different times in state $j$, thus
        \begin{align*}
            r_{i,j}^{(0)} - r^{(\varepsilon)}_{i,j} = \mathbb{E}[T^{(0)}_{i,j} -T^{(\varepsilon)}_{i,j}]
            &= \mathbb{E}[T^{(0)}_{i,j} -T^{(\varepsilon)}_{i,j}\mid Y^{(0)}, Y^{(\varepsilon)}\text{ diverge before } Y^{(0)}=0] \\&\qquad\cdot \mathbb{P}(Y^{(0)}, Y^{(\varepsilon)}\text{ diverge before } Y^{(0)}=0)
        \end{align*}

        We bound both of these terms separately. First we consider the conditional mean. Let $I$ be the (random) index where the first divergence occurs. At this divergence, the two processes move from $(I,I)$ to $(I,I+1)$. Thus
        \begin{align*}
            |\mathbb{E}[T^{(0)}_{i,j} -T^{(\varepsilon)}_{i,j}\mid Y^{(0)}, Y^{(\varepsilon)}\text{ diverge before } Y^{(0)}=0]| &=| \mathbb{E}[r^{(0)}_{I+1,j}-r^{(\varepsilon)}_{I,j}]|\\
            &\leq |r^{(0)}_{j,j}| + |r^{(\varepsilon)}_{j,j}|
            \leq 2C_2,
        \end{align*}
        where $C_2$ is defined in \cref{FundBDBds}.

        Until a divergence occurs, the divergences occur at rate $\lambda^{(0)}-\lambda^{(\varepsilon)}$. The probability that a divergence occurs before time $t$ is $\mathbb{P}(\text{Pois}((\lambda^{(0)}-\lambda^{(\varepsilon)})t)>0) = 1-\exp(-(\lambda^{(0)}-\lambda^{(\varepsilon)})t)$. Letting $\tau_i$ be the time when $Y^{(0)}$ first reaches $0$,
        \begin{align*}
            \mathbb{P}(Y^{(0)}, Y^{(\varepsilon)}\text{ diverge before } Y^{(0)}=0) & = \mathbb{E}[1-\exp(-(\lambda^{(0)}-\lambda^{(\varepsilon)})\tau_i)]\\
            &\leq (\lambda^{(0)}-\lambda^{(\varepsilon)})\mathbb{E}[\tau_i].
        \end{align*}

        Noting that $\mathbb{E}[\tau_i]$ is the same as $\boldsymbol{R}^{(\varepsilon)}e^T$ in \cref{FundBDBds}, we have that $\mathbb{E}[\tau_i]\leq C_3 i$ where $C_3$ is defined in \cref{FundBDBds}. Thus
        \begin{align*}
            &\mathbb{P}(Y^{(0)}, Y^{(\varepsilon)}\text{ diverge before } Y^{(0)}=0) \leq C_3(\lambda^{(0)}-\lambda^{(\varepsilon)})i \\
            \implies
            &|r_{i,j}^{(0)} - r^{(\varepsilon)}_{i,j}| \leq (2C_2)\cdot(C_3(\lambda^{(0)}-\lambda^{(\varepsilon)})i)=\hat{C}_3(\lambda^{(0)}-\lambda^{(\varepsilon)})i
        \end{align*}
        for $\hat{C}_3 = 2C_2C_3$.

        
        

    \end{enumerate}

    With these bounds, we are now able to show that the hybrid-abandonment systems meet our heavy-traffic condition (\cref{def:HT_cond}). We verify the conditions here. We begin by showing that each of $\boldsymbol{\pi}$, $\mathbf{R}$, $\mathbf{F}$, and $\mu_2$ have the desired convergence properties and then show that $M(z^\varepsilon;\varepsilon)\to M^{(0)}$
    \begin{enumerate}
        \item As $\varepsilon\to 0$ or, equivalently, $\lambda^{(\varepsilon)} \to \lambda^{(0)}$,
        \begin{enumerate}
            \item For any $n$, $|\pi_n^{(\varepsilon)} - \pi_n^{(0)}|\leq \hat{C}_1\varepsilon e^{-n}$ and thus $\pi^{(\varepsilon)}_n\to \pi_{n}^{(0)}$ where $\boldsymbol{\pi}^{(0)}$ is a valid distribution.

            \item For any $i,j$, $|r_{i,j}^{(0)}-r_{i,j}^{(\varepsilon)}| \leq \hat{C}_2 (\lambda^{(0)}-\lambda^{(\varepsilon)})$ so $r_{i,j}^{(0)}\to r_{i,j}^{(\varepsilon)}$.

            \item $\boldsymbol{F}^{(\varepsilon)}$ is constant across all $\varepsilon$ so trivially $\boldsymbol{F}^{(\varepsilon)}$ is convergent as $\varepsilon\to 0$.

            \item $\mu_2^{(\varepsilon)}$ is constant across all $\varepsilon$ so trivially $\mu_2^{(\varepsilon)}$ is convergent as $\varepsilon\to 0$.
        \end{enumerate}

        \item As $\varepsilon\to 0$,
        \begin{align*}
        &1-(\mu^{(\varepsilon)}_2)^{-1} \boldsymbol{\pi}^{(\varepsilon)}(\mu^{(\varepsilon)}\boldsymbol{I}-z^\varepsilon \boldsymbol{F}^{(\varepsilon)})\boldsymbol{R}^{(\varepsilon)}(\mu^{(\varepsilon)}\boldsymbol{I}-z^\varepsilon \boldsymbol{F}^{(\varepsilon)})e^T\\ &\qquad\to 1-(\mu_2^{(0)})^{-1} \boldsymbol{\pi}^{(0)}(\mu^{(0)}\boldsymbol{I}-\boldsymbol{F}^{(0)})\boldsymbol{R}^{(0)}(\mu^{(0)}\boldsymbol{I}-\boldsymbol{F}^{(0)})e^T
        \end{align*}
            

            We can decompose $\boldsymbol{\pi}^{(\varepsilon)}(\boldsymbol{I}-z^\varepsilon \boldsymbol{F}^{(\varepsilon)})\boldsymbol{R}^{(\varepsilon)}(\boldsymbol{I}-z^\varepsilon \boldsymbol{F}^{(\varepsilon)})e^T$ to isolate the effect of $z^\varepsilon$, $\boldsymbol{R}^{(\varepsilon)}$, and $\boldsymbol{\pi}^{(\varepsilon)}$. Because $\boldsymbol{F}^{(\varepsilon)}$ does not depend on $\varepsilon$ in this case, we omit the dependence on $\varepsilon$. Similarly we replace $\mu^{(\varepsilon)}$ by $\mu_2$ for clarity because it does not depend on $\varepsilon$.
            \begin{align}
                &\boldsymbol{\pi}^{(\varepsilon)}(\mu_2\boldsymbol{I}-z^\varepsilon \boldsymbol{F})\boldsymbol{R}^{(\varepsilon)}(\mu_2\boldsymbol{I}-z^\varepsilon \boldsymbol{F})e^T \\&= \boldsymbol{\pi}^{(0)}(\mu_2\boldsymbol{I}-\boldsymbol{F})\boldsymbol{R}^{(0)}(\mu_2\boldsymbol{I}-\boldsymbol{F})e^T\nonumber\\
                &+(\boldsymbol{\pi}^{(\varepsilon)}-\boldsymbol{\pi}^{(0)})(\mu_2\boldsymbol{I}-\boldsymbol{F})\boldsymbol{R}^{(0)}(\mu_2\boldsymbol{I}-\boldsymbol{F})e^T\label{eq:M0_decomp_2}\\
                &+\boldsymbol{\pi}^{(\varepsilon)}(1-z^\varepsilon)\boldsymbol{F}\boldsymbol{R}^{(0)}(\mu_2\boldsymbol{I}-\boldsymbol{F})e^T\label{eq:M0_decomp_3}\\
                &+\boldsymbol{\pi}^{(\varepsilon)}(\mu_2\boldsymbol{I}-z^{\varepsilon}\boldsymbol{F})(\boldsymbol{R}^{(\varepsilon)}-\boldsymbol{R}^{(0)})(\boldsymbol{I}-\boldsymbol{F})e^T\label{eq:M0_decomp_4}\\
                &+\boldsymbol{\pi}^{(\varepsilon)}(\mu_2\boldsymbol{I}-z^\varepsilon \boldsymbol{F})\boldsymbol{R}^{(\varepsilon)}(1-z^\varepsilon) \boldsymbol{F}e^T \label{eq:M0_decomp_5}
            \end{align}

            We show \cref{eq:M0_decomp_2,eq:M0_decomp_3,eq:M0_decomp_4,eq:M0_decomp_5} are each negligible in this limit. We extensively use the bounds derived in \cref{FundBDBds} and \cref{DerBDBds}.
            \begin{enumerate}
                \item \cref{eq:M0_decomp_2} is asymptotically negligible.                                                                                                                                       

                First let us bound $(\boldsymbol{\pi}^{(\varepsilon)}-\boldsymbol{\pi}^{(0)})(\mu_2\boldsymbol{I}-\boldsymbol{F})$. In particular, this has an analogous bound to its underlying quantities, $(\boldsymbol{\pi}^{(\varepsilon)}-\boldsymbol{\pi}^{(0)})$ (\cref{eq:pinmpi0bd}).
                \begin{align*}
                    \left|[(\boldsymbol{\pi}^{(\varepsilon)}-\boldsymbol{\pi}^{(0)})(\mu^{(0)}\boldsymbol{I}-\boldsymbol{F})]_{i}\right| &= \left|\sum_{k} ((\pi^{(\varepsilon)}_i-\pi^{(0)}_i)[\mu_2\boldsymbol{I}-\boldsymbol{F}]_{i,k})\right|\\
                    &=\left|(\pi^{(\varepsilon)}_i -\pi^{(0)}_i)\mu_2 - (\pi^{(\varepsilon)}_{i-1}-\pi^{(0)}_{i-1})\beta_{i,i-1}\mathbb{I}\{i>0\}\right|\\
                    &\leq \mu_2 \hat{C}_1 \varepsilon e^{-i} + \mu_1 \hat{C}_1\varepsilon e^{-(i-1)}\\
                    &=\hat{C}_1(\mu_2+e\mu_1)\varepsilon e^{-i}
                \end{align*}


                Thus using the bound we just derived and the bound on $\boldsymbol{R}(\mu^{(\varepsilon)}\boldsymbol{I}-\boldsymbol{F})e^T$ introduced in \cref{BDandLTMods},
                \begin{align*}
                    &\left|(\boldsymbol{\pi}^{(\varepsilon)}-\boldsymbol{\pi}^{(0)})(\mu^{(0)}\boldsymbol{I}-\boldsymbol{F})\boldsymbol{R}^{(0)}(\mu_2\boldsymbol{I}-\boldsymbol{F})e^T\right|\\ &= \left|\sum_{i}[(\boldsymbol{\pi}^{(\varepsilon)}-\boldsymbol{\pi}^{(0)})(\mu^{(0)}\boldsymbol{I}-\boldsymbol{F})]_{i}\ [\boldsymbol{R}(\mu_2\boldsymbol{I}-\boldsymbol{F})e^T]_i\right|\\
                    &\leq \sum_{i} ((\hat{C}_1(\mu_2+e\mu_1)\varepsilon e^{-i})\ (\tilde{C}_3i))=\varepsilon \hat{C}._1\tilde{C}_3(\mu_2+e\mu_1) \sum_i ie^{-i}
                \end{align*}
                Thus $(\boldsymbol{\pi}^{(\varepsilon)}-\boldsymbol{\pi}^{(0)})(\mu^{(0)}\boldsymbol{I}-\boldsymbol{F})\boldsymbol{R}^{(0)}(\mu_2\boldsymbol{I}-\boldsymbol{F})e^T\to \varepsilon$ as $\varepsilon\to 0$.

                \item \cref{eq:M0_decomp_3} is asymptotically negligible.

                First, we note that
                \begin{align*}
                    [\boldsymbol{\pi}^{(\varepsilon)}\boldsymbol{F}]_j = \sum_{i}\pi^{(\varepsilon)}_i \boldsymbol{F}_{i,j}
                    = \pi^{(\varepsilon)}_{i+1} \beta_{i+1, i}
                    &\leq C_1 \beta_{i+1,i} e^{-(i+1)}
                    = C_1 e^{-1}\beta_{i+1,i} e^{-i}.
                \end{align*}
                Using our newly derived bound we can see that
                \begin{align*}
                    &|\boldsymbol{\pi}^{(\varepsilon)}(1-z^\varepsilon)\boldsymbol{F}\boldsymbol{R}^{(0)}(\boldsymbol{I}-\boldsymbol{F})e^T|  = (1-z^\varepsilon) \left|\sum_{i}
                    \left([\boldsymbol{\pi}^{(\varepsilon)}\boldsymbol{F}]_i\ [\boldsymbol{R}^{(0)}(\mu_2 \boldsymbol{I} -\boldsymbol{F})e^T]_i\right) \right|\\
                    &\leq (1-z^\varepsilon)\left|\sum_{i} \left((C_1 e^{-1}\beta_{i+1,i} e^{-i})\ (\tilde{C}_3 i)\right)\right|
                    =(1-z^\varepsilon)\tilde{C}_3C_1 e^{-1}\beta_{i+1,i}\sum_{i}  e^{-i} i
                \end{align*}
                Thus as $\varepsilon\to0$, $(1-z^{\varepsilon})\to 0$ and $\boldsymbol{\pi}^{(\varepsilon)}(1-z^\varepsilon)\boldsymbol{F}\boldsymbol{R}^{(0)}(\mu_2\boldsymbol{I}-\boldsymbol{F})e^T\to 0$

                \item \cref{eq:M0_decomp_4} is asymptotically negligible.

                \begin{align*}
                    &|\boldsymbol{\pi}^{(\varepsilon)}(\mu_2\boldsymbol{I}-z^{\varepsilon}\boldsymbol{F})(\boldsymbol{R}^{(\varepsilon)}-\boldsymbol{R}^{(0)})(\mu_2\boldsymbol{I}-\boldsymbol{F})e^T|\\&= |[\boldsymbol{\pi}^{(\varepsilon)}(\mu_2\boldsymbol{I}-z^{\varepsilon}\boldsymbol{F})](\boldsymbol{R}^{(\varepsilon)}-\boldsymbol{R}^{(0)})[(\mu_2\boldsymbol{I}-\boldsymbol{F})e^T]|\\
                    &\leq |[\boldsymbol{\pi}^{(\varepsilon)}(\mu_2\boldsymbol{I}-z^{\varepsilon}\boldsymbol{F})](\boldsymbol{R}^{(\varepsilon)}-\boldsymbol{R}^{(0)})[(\mu_2+\mu_1)e^T]|\\
                    &\leq (\mu_1+\mu_2)|[\boldsymbol{\pi}^{(\varepsilon)}(\mu_2\boldsymbol{I}-z^{\varepsilon}\boldsymbol{F})](\boldsymbol{R}^{(\varepsilon)}-\boldsymbol{R}^{(0)})e^T|\\
                    &=(\mu_1+\mu_2) \sum_{i,j}([\boldsymbol{\pi}^{(\varepsilon)}(\mu_2\boldsymbol{I}-z^{\varepsilon}\boldsymbol{F})]_i\ (r_{i,j}^{(\varepsilon)}-r_{i,j}^{(0)}))\\
                    &\leq (\mu_1+\mu_2) \sum_{i}(\tilde{C}_1 e^{-i} (\sum_j (r_{i,j}^{(\varepsilon)}-r_{i,j}^{(0)})))
                \end{align*}

                We use two different bounds for $r^{(\varepsilon)}_{i,j} - r^{(0)}_{i,j}$. In particular we use our coupling bound, \mbox{$|r^{(\varepsilon)}_{i,j} - r^{(0)}_{i,j}| \leq \hat{C}_2(\lambda^{(0)}-\lambda^{(\varepsilon)})i$}, \eqref{eq:rij0-rijeps-bd}, for ``small" $j$ and our bounds on $|r_{i,j}^{(\varepsilon}|$, $|r_{i,j}^{(\varepsilon)}| \leq C_2 e^{-(j-i)^+}$ (property \ref{eq:birth-bd-2}), for ``large" $j$. We change our bound at $j=i+\ln ((\lambda^{(0)}-\lambda^{(\varepsilon)})^{-1})$. Thus, 
                \begin{align*}
                    \left|\sum_{j}(r^{(\varepsilon)}_{i,j} - r^{(0)}_{i,j})\right| &= \left|\sum_{j=0}^{\lfloor i-\ln(\lambda^{(0)}-\lambda^{(\varepsilon)})\rfloor}(r^{(\varepsilon)}_{i,j} - r^{(0)}_{i,j}) + \sum_{j=1+\lfloor i-\ln(\lambda^{(0)}-\lambda^{(\varepsilon)})\rfloor}^{\infty}(r^{(\varepsilon)}_{i,j} - r^{(0)}_{i,j}) \right|\\
                    &\leq |\sum_{j=0}^{\lfloor i-\ln(\lambda^{(0)}-\lambda^{(\varepsilon)})\rfloor}(\hat{C}_2(\lambda^{(0)}-\lambda^{(\varepsilon)})i)| \\
                    &+ |\sum_{j=1+\lfloor i-\ln(\lambda^{(0)}-\lambda^{(\varepsilon)})\rfloor}^{\infty}(C_2e^{-(j-i)^+})|\\
                    &\leq (i-\ln(\lambda^{(0)}-\lambda^{(\varepsilon)}))(\hat{C}_2(\lambda^{(0)}-\lambda^{(\varepsilon)})i)\\ 
                    &+ C_2(\lambda^{(0)}-\lambda^{(\varepsilon)}) \sum_{n=0}^\infty e^{-n}\\
                    &=\hat{C}_2(\lambda^{(0)}-\lambda^{(\varepsilon)})i^2 - \hat{C}_2\ln(\lambda^{(0)}-\lambda^{(\varepsilon)})(\lambda^{(0)}-\lambda^{(\varepsilon)})i \\&+ \frac{C_2}{1-e^{-1}} (\lambda^{(0)}-\lambda^{(\varepsilon)})
                \end{align*}
                With this bound on  $|\sum_{j}(r^{(\varepsilon)}_{i,j} - r^{(0)}_{i,j})|$ we can analyze \cref{eq:M0_decomp_4} directly.
                \begin{align*}
                    &\left|\boldsymbol{\pi}^{(\varepsilon)}(\mu_2\boldsymbol{I}-z^{\varepsilon}\boldsymbol{F})(\boldsymbol{R}^{(\varepsilon)}-\boldsymbol{R}^{(0)})(\mu_2\boldsymbol{I}-\boldsymbol{F})e^T\right| \\
                    &\leq (\mu_1+\mu_2) \sum_{i}( \tilde{C}_1e^{-i} \sum_{j}(r_{i,j}^{(0)}-r_{i,j}^{(\varepsilon)}))\\
                    &\leq  (\mu_1+\mu_2)\sum_{i} (\tilde{C}_1e^{-i}(\hat{C}_2(\lambda^{(0)}-\lambda^{(\varepsilon)})i^2 - \hat{C}_2\ln(\lambda^{(0)}-\lambda^{(\varepsilon)})(\lambda^{(0)}-\lambda^{(\varepsilon)})i ))\\
                    &+  (\mu_1+\mu_2)(\lambda^{(0)}-\lambda^{(\varepsilon)}) \sum_i(\frac{C_2}{1-e^{-1}}e^{-i})\\
                    &= (\mu_1+\mu_2)(\lambda^{(0)}-\lambda^{(\varepsilon)}) \sum_{i}(C_1\hat{C}_2 i^2e^{-i)} \\&-  (\mu_1+\mu_2)(\lambda^{(0)}-\lambda^{(\varepsilon)})\ln (\lambda^{(0)}-\lambda^{(\varepsilon)}) \sum_i(C_1\hat{C}_2 ie^{-i}) \\&+  (\mu_1+\mu_2)(\lambda^{(0)}-\lambda^{(\varepsilon)}) \sum_i(\frac{C_2}{1-e^{-1}}e^{-i})
                \end{align*}
                Thus because $(\lambda^{(0)}-\lambda^{(\varepsilon)})\to 0$ and $-(\lambda^{(0)}-\lambda^{(\varepsilon)})\ln((\lambda^{(0)}-\lambda^{(\varepsilon)}))\to 0$ as $\varepsilon\to 0$,\\ \mbox{
$\boldsymbol{\pi}^{(\varepsilon)}(\mu_2\boldsymbol{I}-z^{\varepsilon}\boldsymbol{F})(\boldsymbol{R}^{(\varepsilon)}-\boldsymbol{R}^{(0)})(\mu_2\boldsymbol{I}-\boldsymbol{F})e^T\to 0$}.                

                \item \cref{eq:M0_decomp_5} is asymptotically negligible.

                \begin{align*}
                    &\left|\boldsymbol{\pi}^{(\varepsilon)}(\boldsymbol{I}-z^\varepsilon \boldsymbol{F})\boldsymbol{R}^{(\varepsilon)}(1-z^\varepsilon) \boldsymbol{F}e^T\right| = (1-z^\varepsilon)\left|\boldsymbol{\pi}^{(\varepsilon)}(\boldsymbol{I}-z^\varepsilon \boldsymbol{F})\boldsymbol{R}^{(\varepsilon)} \boldsymbol{F}e^T\right| \\
                    &=(1-z^\varepsilon)\left|[\boldsymbol{\pi}^{(\varepsilon)}(\boldsymbol{I}-z^\varepsilon \boldsymbol{F})]\ \boldsymbol{R}^{(\varepsilon)} \ [\boldsymbol{F}e^T]\right|
                    \leq(1-z^\varepsilon)\left|[\boldsymbol{\pi}^{(\varepsilon)}(\boldsymbol{I}-z^\varepsilon \boldsymbol{F})]\ \boldsymbol{R}^{(\varepsilon)} \ [\mu_1e^T]\right|\\
                    &= \mu_1(1-z^\varepsilon)\left|[\boldsymbol{\pi}^{(\varepsilon)}(\boldsymbol{I}-z^\varepsilon \boldsymbol{F})]\ \boldsymbol[{R}^{(\varepsilon)}e^T]\right|
                    \leq \mu_1(1-z^\varepsilon) \left|\sum_i ((\tilde{C}_1e^{-i}) (C_3i))\right|\\
                    &=(1-z^\varepsilon) \mu_1 \tilde{C}_1C_3 \sum_i ie^{-i}
                \end{align*}

                Thus as $\varepsilon\to 0$, $(1-z^\varepsilon)\to 0$ and $\boldsymbol{\pi}^{(\varepsilon)}(\boldsymbol{I}-z^\varepsilon \boldsymbol{F})\boldsymbol{R}^{(\varepsilon)}(1-z^\varepsilon) \boldsymbol{F}e^T\to 0$.

            \end{enumerate}

            
        \item For all $\varepsilon$, the phase process is an $M/M/1+M$ queue with $\lambda>0$ and $\theta>0$. Thus the phase process is ergodic for all $\varepsilon$.
    \end{enumerate}

    The hybrid-abandonment system therefore meets our heavy traffic conditions (\cref{def:HT_cond}). Having show this we need only show that the system also meets the birth-death, rate bounded, and uniformly stable conditions.

    The phase process, the first queue, is a birth-death process for all $\varepsilon$, so it immediately satisfies our birth-death condition (\cref{def:bd}). Thus, we need only show that the family is uniformly stable (\cref{def:unifstab}) and rate-bounded (\cref{def:rtbdd}).
    With $a=b=e^{-1}$ and $N=\lceil \frac{\theta e}{\lambda_{\text{max}}}\rceil +1$,
    \begin{enumerate}
            \item $\sup_{n\geq N,\varepsilon} \frac{\alpha_{n,n+1}}{\alpha_{n,n-1}} = \sup_{n\geq N,\varepsilon} \frac{\lambda}{\mu_11\{n>0\} + \left(n-1\right)\theta1\{n>0\}} \leq e^{-1}$.

        \item $\sup_{n\geq N,\varepsilon} \frac{\alpha_{n,n+1}}{\alpha_{n+1, n}} = \sup_{n\geq N,\varepsilon} \frac{\lambda^{\left(\varepsilon\right)}}{\mu_1+n\theta} \leq e^{-1}$.
        \end{enumerate}
        Thus, the family satisfies all the conditions of \cref{def:unifstab} and is uniformly stable.

        Finally, let us show that the family is rate-bounded, with $T_{\max} = (\lambda_{\text{min}})^{-1}$ and $\hat{\lambda} = \mu_1$.
    \begin{enumerate}

The element $[\boldsymbol{R}]_{i,j}$, which we call $r_{i,j}$ for notational convenience, is the expected time the process spends in state $j$ between starting in state $i$ and reaching the $0$ state for the first time. We can represent this as
    \begin{align*}
        [\boldsymbol{R}]_{i,j}=r_{i,j} = \mathbb{E}[\sum^{N_{i,j}}_{k=1} T_k],
    \end{align*}
    where $N_{i,j}$ is the number of visits the process makes to state $j$ between starting in state $i$ and reaching state $0$ and $T_k$ is the length of time spent in state $j$ during the $k^{\text{th}}$ visit to the state.

    Because $T_1, T_2, \dots, T_k, \dots$ is a sequence of i.i.d. exponential random variables with mean $\frac{1}{-\alpha_{j,j}}<\infty$, if $\mathbb{E}[N_{i,j}]<\infty$ then $r_{i,j} = \mathbb{E}[N_{i,j}]\mathbb{E}[T_0] < \infty$ by Wald's lemma. Thus, if we can show that $\mathbb{E}[N_{i,j}]<\infty$ for all $i$ and $j$, then $\boldsymbol{R}$ is well-posed.

    First, let us show that $\mathbb{E}[N_{i,i}]<\infty$ for all $i\in \mathbb{N}_0$ and then show the same for when $i\ne j$. Because the chain is positive recurrent, with probability $1$ the chain to return to state $i$ infinitely many times after starting there. Because the chain is irreducible, there is some positive probability $p_i$ that chain reaches state $0$ before returning to state $i$. Thus, $N_{i,i}\sim \text{Geom}\left(p_i\right)$ and $\mathbb{E}[N_{i,i}] = \frac{1-p_i}{p_i}<\infty$.

    Next consider when $i\ne j$. Consider two possibilities: starting in state $i$ the chain either reaches $0$ before $j$ or vice versa. If the chain reaches $0$ before $j$, then, by definition, $N_{i,j}=0$. Conversely if the chain reaches $j$ before $0$ then the expected number of returns to $j$ is exactly the same as if the process simply began in state $j$. Therefore 
        \[\mathbb{E}[N_{i,j}] = P\left(\text{Reach $j$ before $0$}\right)\mathbb{E}[N_{j,j}]\leq \mathbb{E}[N_{j,j}]<\infty.\]

    Thus, $\mathbb{E}[N_{i,j}]<\infty$ for all $i,j$ and $r_{i,j} = \frac{\mathbb{E}[N_{i,j}]}{-\alpha_{j,j}} < \infty$. The $\boldsymbol{R}$ matrix is well posed.

    Next, let us show that our proposed $\boldsymbol{R}$ has the near-inverse property. Notice that the quantity $N_{i,j}$ is an element of the fundamental matrix of an absorbing DTMC, particularly the embedded DTMC of our CTMC after making $0$ an absorbing state. Let $\boldsymbol{H}_{\overline{0},\overline{0}}$ and  be the one step transition matrix of the transient states in this absorbing embedded DTMC, i.e., all state except $0$.
    \[\boldsymbol{N}\boldsymbol{I}+\boldsymbol{H}_{\overline{0},\overline{0}}+\left(\boldsymbol{H}_{\overline{0},\overline{0}}\right)^2+\dots = \boldsymbol{I}+\boldsymbol{NH}_{\overline{0},\overline{0}} = \boldsymbol{I}+\boldsymbol{H}_{\overline{0},\overline{0}}\boldsymbol{N}\]
    or, after rearranging,
    \[\boldsymbol{I}=\boldsymbol{N}\left(\boldsymbol{I}-\boldsymbol{H}_{\overline{0},\overline{0}}\right)=\left(\boldsymbol{I}-\boldsymbol{H}_{\overline{0},\overline{0}}\right)\boldsymbol{N}.\] This is precisely $\boldsymbol{N}=\left(\boldsymbol{I}-\boldsymbol{H}_{\overline{0},\overline{0}}\right)^{-1}$ and therefore
    \[-\left(\boldsymbol{W}_{\overline{0},\overline{0}}\right)^{-1}\boldsymbol{N} = \left(\boldsymbol{W}_{\overline{0},\overline{0}}\right)^{-1}\left(\boldsymbol{H}_{\overline{0},\overline{0}}-\boldsymbol{I}\right) = \left(\boldsymbol{A}_{\overline{0}, \overline{0}}\right)^{-1}.\]
    where $\boldsymbol{W}_{\overline{0},\overline{0}}$ is the holding rates matrix for all states except $0$ and $\boldsymbol{A}_{\overline{0}, \overline{0}} = [\alpha_{i,j}]_{i,j\in \mathbb{N}_0/0}$.

    Thus, \begin{align*}
        r_{i,j}= \begin{cases}
            \left[(\boldsymbol{A}_{\overline{0}, \overline{0}})^{-1}\right]_{i,j}, \qquad & i,j\in \mathbb{N}_0/0\\
            0, & \text{else}
        \end{cases}
    \end{align*}

    The steady-state balance equation provides that \[\pi_0 \left(\alpha_{0, 1}, \alpha_{0, 2}, \dots\right) + \left(\pi_1, \pi_2, \dots\right) \boldsymbol{A}_{\overline{0}, \overline{0}} = 0\] or, equivalently, \[\left(\alpha_{0,1}, \alpha_{0,2}, \dots\right) \left(\boldsymbol{A}_{\overline{0}, \overline{0}}\right)^{-1} = -\frac{1}{\pi_0}\left(\pi_1, \pi_2, \dots\right).\]

    Thus, we can now construct our matrix $\boldsymbol{AR}$. For $i,j\in \mathbb{N}_0/0$, $[\boldsymbol{A}_{\overline{0}, \overline{0}}\boldsymbol{R}]_{i,j} = \mathbb{I}\{i=j\}$. When $i=0$, $[\boldsymbol{A}_{\overline{0}, \overline{0}}\boldsymbol{R}]_{i,j} = 0$ because $[\boldsymbol{R}]_{0,j} = 0 $ for all $j\in \mathbb{N}_0$. Lastly, when $i\in \mathbb{N}_0/0$, $[\boldsymbol{A}_{\overline{0}, \overline{0}}\boldsymbol{R}]_{i,0} = \frac{\pi_i}{\pi_0}$.
    We have our desired result that

 = \begin{bmatrix}
        0 & -\frac{1}{\pi_0} \boldsymbol{\pi}_{\overline{0}}\\
        0& \boldsymbol{I}
    \end{bmatrix}.
    \end{align*}

            &\leq \left(C_1a^{-1}\hat{\lambda} + C_1 \left(\hat{\mu} + \hat{\lambda}\right) + C_1a \hat{\lambda}\right)a^j\\
            &=\tilde{C}_1a^j
        \end{align*}
        for $\tilde{C}_1 = C_1a^{-1}\hat{\lambda} + C_1 \left(\hat{\mu} + \hat{\lambda}\right) + C_1a \hat{\lambda}$
        \item $\left|[\boldsymbol{R}^{(\varepsilon)}\left(\mu^{\left(\varepsilon\right)}\boldsymbol{I}-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}\right)]_{i,j}\right| \leq \tilde{C}_2\tilde{b}^{\left(j-i\right)^+}$ for some $\tilde{C}_2>0$.

                &=a \tilde{C}_1\tilde{C}_3C_3\left|\sum_{i=0}^\infty a^i(\sum_{k=0}^{i-1}(\tilde{b}^{(k-i)^+}\cdot \tilde{b}^{(j-k)^+})+\sum_{k=i}^\infty(\tilde{b}^{(k-i)^+}\cdot \tilde{b}^{(j-k)^+})) \right|\\
                &=a \tilde{C}_1\tilde{C}_3C_3\left|\sum_{i=0}^\infty a^i(\sum_{k=0}^{i-1}(1)+\sum_{k=i}^\infty(\tilde{b}^{(k-i)^+})) \right|
                =a \tilde{C}_1\tilde{C}_3C_3\left|\sum_{i=0}^\infty a^i(i+\frac{1}{1-\tilde{b}})) \right|\\
                &=\tilde{C}_1\tilde{C}_3C_3\left|\frac{a}{(1-a)^2}+\frac{1}{(1-\tilde{b})(1-a)}) \right| < \infty
            \end{align*}

            Thus because $(1-z^{-\varepsilon})\to 0$ as $\varepsilon\to 0$ and $[\hat{\boldsymbol{q}}\left(z^\varepsilon;\varepsilon\right) \left(\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{\left(\varepsilon\right)} \boldsymbol{I}\right)R\right)^2]_j$ is bounded uniformly across $\varepsilon, z$, their product goes to $0$ as $\varepsilon\to 0$.
                        
            \item $[\mu^{\left(\varepsilon\right)}\left(1-z^{-\varepsilon}\right)\boldsymbol{\pi}^{\left(\varepsilon\right)}_0\boldsymbol{R}^{(\varepsilon)}\left(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)} + \mu^{\left(\varepsilon\right)}I\right)\boldsymbol{R}^{(\varepsilon)}]_j\to 0$.
            \begin{align*}
                &\left|[\boldsymbol{\pi}^{(\varepsilon)}_0R^{(\varepsilon)}(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)} + \mu^{(\varepsilon)} \boldsymbol{I})R^{(\varepsilon)}]_j\right|\\
                &=\left|\sum_{i,k}\pi^{(\varepsilon)}_{i,0} \cdot [R^{(\varepsilon)}(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)}+\mu^{(\varepsilon)} \boldsymbol{I})]_{i,k}\cdot r^{(\varepsilon)}_{k,j} \right|\\
                &=a \left|\sum_{i,k}\tilde{C}_1 a^i \cdot \tilde{C}_3\tilde{b}^{(k-i)^+}\cdot C_3\tilde{b}^{(j-k)^+} \right|\\
                &=a \left|\sum_{i=0}^\infty\sum_{k=0}^\infty(\tilde{C}_1 a^i \cdot \tilde{C}_3\tilde{b}^{(k-i)^+}\cdot C_3\tilde{b}^{(j-k)^+}) \right|\\
                &=a \left|\sum_{i=0}^\infty\tilde{C}_1 a^i(\sum_{k=0}^{i-1}(\tilde{C}_3\tilde{b}^{(k-i)^+}\cdot C_3\tilde{b}^{(j-k)^+})+\sum_{k=i}^\infty(\tilde{C}_3\tilde{b}^{(k-i)^+}\cdot C_3\tilde{b}^{(j-k)^+})) \right|\\
                &=a \tilde{C}_1\tilde{C}_3C_3\left|\sum_{i=0}^\infty a^i(\sum_{k=0}^{i-1}(\tilde{b}^{(k-i)^+}\cdot \tilde{b}^{(j-k)^+})+\sum_{k=i}^\infty(\tilde{b}^{(k-i)^+}\cdot \tilde{b}^{(j-k)^+})) \right|\\
                &=a \tilde{C}_1\tilde{C}_3C_3\left|\sum_{i=0}^\infty a^i(\sum_{k=0}^{i-1}(1)+\sum_{k=i}^\infty(\tilde{b}^{(k-i)^+})) \right|\\
                &=a \tilde{C}_1\tilde{C}_3C_3|\sum_{i=0}^\infty a^i(i+\frac{1}{1-\tilde{b}})) |\\
                &=\tilde{C}_1\tilde{C}_3C_3|\frac{a}{(1-a)^2}+\frac{1}{(1-\tilde{b})(1-a)}) | < \infty
            \end{align*}
            Thus because $\left(1-z^{-\varepsilon}\right)\to 0$ as $\varepsilon\to 0$ and $|[\boldsymbol{\pi}^{(\varepsilon)}_0R^{(\varepsilon)}(-z^\varepsilon \boldsymbol{F}^{(\varepsilon)} + \mu^{(\varepsilon)} \boldsymbol{I})R^{(\varepsilon)}]_j|$ is uniformly bounded across all $z,\varepsilon$, their product also goes to $0$ as $\varepsilon\to 0$.


            \item $[\mu^{\left(\varepsilon\right)}\boldsymbol{\pi}^{\left(\varepsilon\right)}_0\boldsymbol{R}^{(\varepsilon)}]_j\to 0$.

            \begin{align*}
                \left|[\mu^{\left(\varepsilon\right)}\boldsymbol{\pi}^{\left(\varepsilon\right)}_0\boldsymbol{R}^{(\varepsilon)}]_j\right| &\leq \left|\mu^{\left(\varepsilon\right)}| \ |\sum_{i=0}^\infty\pi^{\left(\varepsilon\right)}_{i,0} r^{\left(\varepsilon\right)}_{i,j}\right|
                \leq |\mu^{\left(\varepsilon\right)}| \ \left|\sum_{i=0}^\infty\pi^{\left(\varepsilon\right)}_{i,0} r^{\left(\varepsilon\right)}_{j,j}\right|\\
                &=|\mu^{\left(\varepsilon\right)}| \ \left|r_{j,j}^{\left(\varepsilon\right)}| \ |\sum_{i=0}^\infty \pi^{\left(\varepsilon\right)}_{i,0}\right|

            \end{align*}
            Because $|\mu^{\left(\varepsilon\right)}|$ and $ |r_{j,j}^{\left(\varepsilon\right)}|$ approach constants $|\mu^{\left(0\right)}|$ and $ |r_{j,j}^{\left(0\right)}|$ as $\varepsilon\to 0$, $\varepsilon |\mu^{\left(\varepsilon\right)}| \ |r_{j,j}^{\left(\varepsilon\right)}| \to 0$.
        \end{enumerate}
