<!-- definition-1: body_md -->
\label{def:weighted-spherical-k-design}
Let $k\ge1$ be an integer.  The weighted configuration
\[
        (q_1,\ldots,q_n;m_1,\ldots,m_n)
\]
is called a \emph{weighted spherical $k$-design} if
\begin{equation}\label{eq:weighted-k-design-definition}
        \frac{1}{M}\sum_{i=1}^n m_i p(q_i)
        =
        \int_{S_R(V)}p(x)\,d\nu_R(x)
\end{equation}
for every real polynomial $p:V\to\mathbb R$ of total degree at most $k$.

Thus $k$ specifies the degree up to which the finite weighted configuration
reproduces exactly the corresponding spherical averages.  In particular,
every spherical $k$-design is automatically a spherical $\ell$-design for
every $\ell\le k$.  The largest integer $k$ for which
\eqref{eq:weighted-k-design-definition} holds is called the \emph{strength}
of the design.

<!-- theorem-1: body_md -->
\label{thm:even-power-design}
Let $V\subset\mathbb R^D$ be a $d$-dimensional Euclidean linear subspace,
with $d\ge2$, and let $k\ge1$.  Suppose that positive masses
$m_1,\ldots,m_n$ at points
\[
        q_1,\ldots,q_n\in S_R(V)
\]
form a weighted spherical $k$-design.  Then the configuration is central
for the homogeneous exponent $\sigma=2k$.  More precisely,
\begin{equation}\label{eq:general-even-centrality}
        \sum_{j\ne i}m_j(q_j-q_i)|q_j-q_i|^{2k-2}
        =\lambda_{d,k}q_i,
        \qquad i=1,\ldots,n,
\end{equation}
where
\begin{equation}\label{eq:general-even-lambda}
        \lambda_{d,k}
        =
        -2^{2k-1}MR^{2k-2}
        \frac{\bigl((d-1)/2\bigr)_k}{(d-1)_k}.
\end{equation}
Here
\begin{equation}\label{eq:pochhammer-definition}
        (a)_k=a(a+1)\cdots(a+k-1)
\end{equation}
denotes the rising factorial.

<!-- lemma-1: body_md -->
\label{lem:weighted-design-closure}
Let $V\subset\mathbb R^D$ be a $d$-dimensional Euclidean linear subspace, and
let $X_1,\ldots,X_s$ be weighted spherical $k$-designs on the same sphere
$S_R(V)$.  For each block choose an arbitrary constant $c_a>0$ and an
orthogonal transformation $Q_a\in O(V)$.  Multiply every weight in $X_a$ by
$c_a$, apply $Q_a$ to all its points, and then take the weighted union of the
resulting blocks.  The union is again a weighted spherical $k$-design on
$S_R(V)$.

\label{thm:generic-rotated-design-moduli}
Let $k\ge2$, let $V$ be a $d$-dimensional Euclidean space with $d\ge2$, and
let
\[
        X_1,\ldots,X_s\subset S_R(V),
        \qquad s\ge2,
\]
be finite weighted spherical $k$-designs with fixed positive weights.
For
\[
        (Q_1,\ldots,Q_s)\in SO(V)^s
\]
consider the weighted union
\begin{equation}\label{eq:general-rotated-block-family}
        \mathcal X(Q_1,\ldots,Q_s)
        :=
        Q_1X_1\cup\cdots\cup Q_sX_s.
\end{equation}
Then there exists an open dense subset
\[
        \mathcal U\subset SO(V)^s
\]
such that every configuration \eqref{eq:general-rotated-block-family} with
$(Q_1,\ldots,Q_s)\in\mathcal U$ is collision-free and has the following
properties.

\begin{enumerate}
\item[(i)]
Once the fixed block data $X_1,\ldots,X_s$ are given, the decomposition into
the $s$ rotated design blocks can be recovered from the Euclidean metric of
the weighted union: one uses the finite set of distances which occur inside
the fixed blocks.  In particular, every isometry between two generic unions
must permute the rotated blocks, up to a finite permutation of mutually
congruent blocks carrying compatible weights.

\item[(ii)]
After quotienting by a common orthogonal transformation of $V$, by the
finite orthogonal symmetry groups of the individual blocks, and by the
finite permutations allowed in {\rm(i)}, the map from relative orientations
to similarity classes is locally finite-to-one.

\item[(iii)]
Consequently the generic local dimension of the resulting moduli space of
weighted configurations is
\begin{equation}\label{eq:general-rotated-block-moduli-dimension}
        (s-1)\dim SO(d)
        =
        (s-1)\frac{d(d-1)}2.
\end{equation}
\end{enumerate}

By Corollary~\ref{cor:relative-rotations-designs}, all these collision-free
configurations are central for $\sigma=2k$.

\label{thm:minimal-positive-weighted-2-design}
Let

\]

\begin{equation}\label{eq:minimal-two-design-node-bound}
        N\ge d+1.
\end{equation}
If equality holds, then
\begin{equation}\label{eq:minimal-two-design-equality}
        p_i=\frac1{d+1},
        \qquad
        q_i\cdot q_j=-\frac{R^2}{d}
        \quad (i\ne j).
\end{equation}
Thus equality holds precisely for the equal-weight vertices of a regular
$d$-simplex inscribed in $S_R(V)$, up to an orthogonal transformation.

<!-- proof-2: body_md -->
The first- and second-moment identities
\eqref{eq:design-first-moment} and
\eqref{eq:design-second-moment}, after normalizing the total mass to one, are
equivalently
\begin{equation}\label{eq:normalized-two-design-moments}
        \sum_{i=1}^N p_iq_i=0,
        \qquad
        \sum_{i=1}^N p_iq_iq_i^T
        =\frac{R^2}{d}I_V.
\end{equation}
Here $q_iq_i^T$ denotes the rank-one operator
\[
        u\longmapsto (q_i\cdot u)q_i.
\]
Choose an orthonormal basis of $V$ and let $A$ be the $d\times N$ matrix
whose $i$th column is
\begin{equation}\label{eq:minimal-two-design-column}
        a_i:=\frac{\sqrt{dp_i}}{R}\,q_i.
\end{equation}
The second identity in \eqref{eq:normalized-two-design-moments} gives
\begin{equation}\label{eq:minimal-two-design-AAstar}
        AA^T=I_d.
\end{equation}
Hence $\operatorname{rank}A=d$.  On the other hand, if
\[
        s:=(\sqrt{p_1},\ldots,\sqrt{p_N})^T,
\]
then the first identity in \eqref{eq:normalized-two-design-moments} gives
\[
        As=0.
\]
Since $s\ne0$, the kernel of $A$ is nontrivial, and therefore
$N\ge d+1$, proving \eqref{eq:minimal-two-design-node-bound}.

Suppose now that $N=d+1$.  Then $\ker A$ is one-dimensional and is spanned
by $s$.  Since $\|s\|^2=\sum_i p_i=1$ and
\eqref{eq:minimal-two-design-AAstar} says that the rows of $A$ are
orthonormal, $A^TA$ is the orthogonal projection onto $s^\perp$:
\begin{equation}\label{eq:minimal-two-design-projection}
        A^TA=I_{d+1}-ss^T.
\end{equation}
Comparing the diagonal entries of
\eqref{eq:minimal-two-design-projection} gives
\[
        dp_i=1-p_i,
\]
hence $p_i=1/(d+1)$.  Comparing the off-diagonal entries gives
\[
        \frac{d\sqrt{p_ip_j}}{R^2}\,q_i\cdot q_j
        =-\sqrt{p_ip_j},
\]
and therefore $q_i\cdot q_j=-R^2/d$ for $i\ne j$.  This is precisely
\eqref{eq:minimal-two-design-equality}.

It remains to verify the converse implicit in the word ``precisely'' in the
statement.  Suppose, therefore, that $q_1,\ldots,q_{d+1}\in S_R(V)$ satisfy
\[
        q_i\cdot q_j=-\frac{R^2}{d}
        \qquad (i\ne j),
\]
and give them the equal weights
\[
        p_i=\frac1{d+1}.
\]
These are the vertices of a regular $d$-simplex.

First,
\[
\begin{aligned}
        \left|\sum_{i=1}^{d+1}q_i\right|^2
        &=
        \sum_{i=1}^{d+1}|q_i|^2
        +
        2\sum_{1\le i<j\le d+1}q_i\cdot q_j\\
        &=
        (d+1)R^2
        -
        2\binom{d+1}{2}\frac{R^2}{d}\\
        &=0.
\end{aligned}
\]
Hence
\begin{equation}\label{eq:regular-simplex-first-moment}
        \sum_{i=1}^{d+1}p_iq_i=0.
\end{equation}

Next put
\[
        T:=\sum_{i=1}^{d+1}q_iq_i^T.
\]
Using the preceding centering identity, for every vertex $q_j$ we obtain
\[
\begin{aligned}
        Tq_j
        &=
        \sum_{i=1}^{d+1}q_i(q_i\cdot q_j)\\
        &=
        R^2q_j
        -
        \frac{R^2}{d}
        \sum_{i\ne j}q_i\\
        &=
        R^2q_j+\frac{R^2}{d}q_j\\
        &=
        \frac{d+1}{d}R^2q_j.
\end{aligned}
\]
To see that the vertices span $V$, consider their $(d+1)\times(d+1)$ Gram
matrix
\[
        G=(q_i\cdot q_j)_{i,j=1}^{d+1}.
\]
Using
\[
        |q_i|^2=R^2,
        \qquad
        q_i\cdot q_j=-\frac{R^2}{d}
        \quad (i\ne j),
\]
we may write
\[
        G
        =
        \frac{R^2}{d}
        \left(
        (d+1)I_{d+1}
        -
        \mathbf 1\mathbf 1^T
        \right),
\]
where
\[
        \mathbf 1=(1,\ldots,1)^T.
\]
Now $\mathbf 1\mathbf 1^T$ has eigenvalue $d+1$ on the line spanned by
$\mathbf 1$ and eigenvalue $0$ on its orthogonal complement
$\mathbf 1^\perp$.  Hence $G$ has eigenvalue $0$ on
$\operatorname{span}\{\mathbf 1\}$ and eigenvalue
\[
        \frac{d+1}{d}R^2
\]
on the $d$-dimensional space $\mathbf 1^\perp$.  Therefore
\[
        \operatorname{rank}G=d.
\]
Since the rank of a Gram matrix is the dimension of the span of the vectors
from which it is formed, the vertices $q_1,\ldots,q_{d+1}$ span $V$.
Therefore
\[
        T=\frac{d+1}{d}R^2 I_V,
\]
and hence
\begin{equation}\label{eq:regular-simplex-second-moment}
        \sum_{i=1}^{d+1}p_iq_iq_i^T
        =
        \frac{R^2}{d}I_V.
\end{equation}

Equations
\eqref{eq:regular-simplex-first-moment} and
\eqref{eq:regular-simplex-second-moment} reproduce the spherical averages of
all linear and quadratic polynomials, and constants are reproduced
automatically because the weights sum to one.  Thus the equal-weight vertices
of a regular $d$-simplex form a weighted spherical $2$-design.

        \int_{S^1} z^\ell\,d\nu

        =0.

Applying the design identity
\eqref{eq:weighted-k-design-definition} to the real and imaginary parts of
$z^\ell$, for $\ell=1,\ldots,k$, therefore gives
\begin{equation}\label{eq:circle-moments-section3}
        \sum_{j=1}^N p_jz_j^\ell=0,
        \qquad

\end{equation}
The equations for negative powers are the complex conjugates of
\eqref{eq:circle-moments-section3}.

\begin{equation}\label{eq:circle-vandermonde-matrix}
        W_{rj}:=\sqrt{p_j}\,z_j^r,
        \qquad
        r=0,\ldots,k.
\end{equation}
Using \eqref{eq:circle-moments-section3}, its Gram matrix is
\begin{equation}\label{eq:circle-vandermonde-unitarity}
        WW^*=I_{k+1},
\end{equation}
because the $(r,s)$ entry is
\[
        \sum_{j=1}^N p_jz_j^{\,r-s}
        =\delta_{rs}.

Hence $\operatorname{rank}W=k+1$, which proves

\]
Therefore $z_i\overline{z_j}$ is a nontrivial $(k+1)$st root of unity.  Since
there are exactly $k+1$ distinct nodes, they form one rotated copy of the

At the reference configuration,
Lemma~\ref{lem:angular-jacobian-rank} shows that
$D_\theta\mathcal F$ is surjective onto
$\mathbb R^{2n-2}$.  Hence one may select $2n-2$ of the angular variables
for which the corresponding square Jacobian minor is invertible.

The remaining two angular variables and all $2n$ masses may therefore be
used as parameters in the parameterized implicit-function theorem.  It
follows that, for every mass vector sufficiently close to
\[
        (1,\ldots,1)\in\mathbb R_{>0}^{2n},
\]
the solutions of
\[
        \mathcal F(\theta,m)=0
\]
near the reference configuration form a smooth two-dimensional local family
in the angular variables.


One of these two dimensions is present for an elementary reason: simultaneous
rotation of all points,
\[
        \theta_j\longmapsto\theta_j+\tau,
\]
preserves every equation
\eqref{eq:circle-weighted-fourier}.  We now remove this trivial direction
explicitly rather than merely counting it.

Let $\mathcal S_m$ denote the two-dimensional local solution manifold in the
angular variables for one fixed nearby mass vector $m$.  On $\mathcal S_m$
consider the function
\[
        g(\theta):=\theta_0.
\]
Along the common-rotation orbit one has
\[
        g(\theta+\tau(1,\ldots,1))
        =
        g(\theta)+\tau.
\]
Hence the derivative of $g$ in the common-rotation direction is equal to
$1$.  In particular, $g$ is not locally constant on $\mathcal S_m$, and by
the ordinary implicit-function theorem it may be used as one of the two
local coordinates on $\mathcal S_m$.

We may therefore impose the gauge
\begin{equation}\label{eq:mass-persistence-rotation-gauge}
        \theta_0=\theta_0^0.
\end{equation}
Every sufficiently nearby common-rotation orbit meets this gauge exactly
once: given $\theta$, the unique small rotation which brings it to the gauge
is
\[
        \tau=\theta_0^0-\theta_0.
\]
Fixing the coordinate $g=\theta_0^0$ removes exactly one of the two local
parameters.  Thus, after quotienting by common rotation, the nearby
solutions form a smooth one-dimensional local family.

After shrinking the neighborhoods if necessary, all masses remain positive
and all angular positions remain distinct.  Their cyclic order on the
circle is unchanged.  Hence every resulting configuration consists of
distinct points on one circle and is strictly convex.



It remains only to check that the surviving parameter is not removed by the
full similarity quotient.  Let two nearby members of the family, for the
same fixed mass vector, be related by a similarity
\[
        T(z)=a+sAz,
        \qquad
        s>0,
        \qquad
        A\in O(2),
\]
possibly followed by a permutation of particles carrying equal masses.

Every weighted $k$-design with $k\ge1$ has weighted center of mass equal to
the origin.  Since a similarity carries the weighted center of mass to its
image under the same affine map, the two centered configurations can be
similar only if
\[
        a=0.
\]
Moreover, every point of both configurations lies on the unit circle.
Thus
\[
        1=|T(z)|=s|Az|=s
\]
for every point $z$ of the first configuration, and hence
\[
        s=1.
\]
Therefore every remaining similarity identification is induced by an
element of $O(2)$.

The subgroup $SO(2)$ consists precisely of the common rotations which have
already been removed by the gauge
\eqref{eq:mass-persistence-rotation-gauge}.  The quotient
\[
        O(2)/SO(2)
\]
has two elements: after common rotations have been factored out, the only
additional orthogonal possibility is one reflection class.  For the fixed
mass vector, the permutations which preserve the masses form a subgroup of
the finite symmetric group on the $2n$ particles, and are therefore finite
as well.

Consequently the remaining equivalence relation on the one-dimensional
gauged solution family has finite equivalence classes.  As in the finite
quotient argument used in the proof of
Theorem~\ref{thm:generic-rotated-design-moduli}, quotienting by these finite
identifications does not change the local dimension.  Hence the full
similarity quotient has local dimension one.  In particular, every
sufficiently small neighborhood contains infinitely many mutually
non-similar weighted configurations.

therefore makes every member of the family a central configuration for
\[
        \sigma=2k.
\]
This proves Theorem~\ref{thm:open-mass-persistence}.
