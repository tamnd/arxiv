\label{lemma.chain}
Let $(X, d)$ be a $\delta$-hyperbolic space.
Let $C, R>0$.
\begin{enumerate}[(i)]
\item\label{lemma.chain1}
For all $w, x, y, z \in X$,
if $(w, y)_x\le C$, $(x, z)_y\le C+\delta$, 
and
\[
d(x, y)\ge 2C+2\delta+R,
\]
then 
$(w, z)_x\le C+\delta$.
\item\label{lemma.chain2}
For every $(C, D)$-chain $x_0, \dots, x_n$ with $D \ge 2C+2\delta +R$,
we have
\[
d(x_0, x_n) \ge \sum_{i=0}^{n-1}(d(x_i, x_{i+1})-2(C+\delta))\ge R n.
\]
\item\label{lemma.chain3}
For every $(C, D)$-chain $x_0, \dots, x_n$ with $D \ge 2C+4\delta+R$,
we have
$(x_0, x_n)_{x_i} \le C+2\delta$
for all $i=0, \dots, n$.
\end{enumerate}
<!-- end lemma-1 -->
<!-- head-1: body_md -->
The parts \eqref{lemma.chain1}, \eqref{lemma.chain2}, and \eqref{lemma.chain3} are Lemmas 3.5, 3.7, and 3.8 in \cite{G}, respectively, when $R=1$.
The case when $R>0$ is reduced to the case when $R=1$ by considering the  space $(X, (1/R)d)$, which is $(\delta/R)$-hyperbolic.
\qed

\label{lemma.quasiaxis}
For a hyperbolic isometry $g$ of $X$,
there exists a path (continuous map) $\alpha: {\mathbb R} \to X$ such that
\begin{enumerate}[(i)]
\item $g(\alpha)=\alpha$ and there exists  $T>0$ such that $g(\alpha(t))=\alpha(t+T)$ for every $t \in {\mathbb R}$.
\item Let $t_1, t_2 \in {\mathbb R}$ with $t_1<t_2$.
Then every geodesic between $\alpha(t_1)$ and $\alpha(t_2)$ and the path $\alpha|_{[t_1, t_2]}:[t_1, t_2]\to X$ of $\alpha$ are in the $\Delta$-neighborhoods of each other,
where $\Delta=1672\delta$.
\end{enumerate}
Moreover, $\alpha$ can be parametrized by arc length such that $T>0$.

<!-- lemma-4: body_md -->
\label{lemma.rectangle}
Let $\alpha$ be a $\Delta$-quasi-axis in $X$.
For all $x, y \in X$, let $x', y'$ be points in $\alpha$ such that
\[
d(x, x')= d(x, \alpha) \quad \text{and} \quad d(y, y')= d(y, \alpha).
\]
If $d(x', y')> 10\delta+2\Delta$,
then
\[
d(x, x')+d(x', y')+d(y', y)\le d(x, y)+18\delta+4\Delta.
\]
<!-- end lemma-4 -->
<!-- lemma-5: body_md -->
\label{lemma.tripod}
Let $x, y, z$ in $X$.
For all $y' \in [x, y]$ and all $z'\in [x, z]$ such that $d(x, y')=d(x, z')\le (y,z)_x$,
we have $d(y', z')\le 4\delta$.
<!-- end lemma-5 -->
<!-- head-2: body_md -->
See \cite[21.- Proposition, Chapitre 2]{GH}.
\qed
<!-- end head-2 -->
<!-- head-3: body_md -->
First, we have $(x, y')_{x'} \le 4 \delta+\Delta$.
Indeed, otherwise, by Lemma \ref{lemma.tripod}, we have $z_1 \in [x, x']$ with $d(z_1, x')>4\delta+\Delta$,
and $z_2\in [x', y']$ with $d(z_1, z_2)\le 4\delta$.
Since $\alpha$ is a $\Delta$-quasi-axis, $d(z_2, \alpha)\le \Delta$.
Concatenating $[x, z_1]$, $[z_1, z_2]$, and a geodesic from $z_2$ to $\alpha$ yields a strictly shorter path than $[x, x']$, which is impossible.

Similarly, $(x', y)_{y'} \le 4\delta+\Delta$.
If $d(x', y')> 10 \delta+2\Delta$,
then $(x, y)_{y'}\le 5\delta+\Delta$ by Lemma \ref{lemma.chain} (i). 
By definition of the Gromov product,
we obtain
\begin{align*}
d(x, x')+d(x', y')+d(y', y)
                                       
\le d(x, y)+18\delta+4\Delta.
\end{align*}
This concludes the proof.
\qed

\label{lemma.disp}
Let $U$, $V$ be finite sets of isometries of $X$.
\begin{enumerate}[(1)]
\item\label{item.UV}
If $U\subset V$, then $L(U) \le L(V)$.
\item\label{item.Un}
For every $n \in {\mathbb N}$,
we have

\item\label{item.trL}
For every $u \in U$, we have ${\rm tr}(u) \le L(U)$.
\end{enumerate}

<!-- lemma-9: body_md -->
\label{lemma.independent}
Fix $n\in {\mathbb N}$ and $a \in [1, +\infty)$.
Let $X$ be an $n$-dimensional Hadamard manifold with sectional curvature $K_X \in [-a^2, -1]$.
Let $\delta$ be the hyperbolicity constant of $\Bbb H^2$, so that $X$ 
is $\delta$-hyperbolic. 

Then, there exists a constant $N=N(n, a)$ such that the following holds:
For every finite set $U$ in $\operatorname{{\rm Isom}}(X)$ with $1 \in U$ such that $\<U\>$ is discrete and 
non-virtually-nilpotent, 
we have
\[
L(U^N) \ge K_0 \delta+1.
\]
Here $K_0$ is the absolute constant in Lemma \ref{lemma.independent.2}.
<!-- end lemma-9 -->
<!-- head-6: body_md -->
Let ${\varepsilon}(n, a)$ denote the Margulis constant of $X$.
The Margulis lemma states that if $\<U\>$ is discrete and non-virtually-nilpotent,
then $L(U)\ge {\varepsilon}(n, a)$, see \cite[Section 8.3]{BGS}.
By Lemma \ref{lemma.CAT0},
we have
\[
L(U^m)\ge \frac{\sqrt{m}}{2}L(U) \ge \frac{\sqrt{m}}{2}{\varepsilon}(n, a),
\]
for every $m \in {\mathbb N}_0$.

Letting 
\[
N:=\left\lceil\left(\frac{2K_0\delta+2}{{\varepsilon}(n, a)}\right)^2 \right\rceil,
\]
we obtain the claim since $N$ depends only on $n$ and $a$.
Here, we denote by $\lceil r \rceil$ the smallest integer greater than or equal to $r \in \Bbb R$.
\qed

<!-- definition-1: body_md -->
\label{def.wa}
An isometric action by a group $G$ on a $\delta$-hyperbolic space $X$ 
 is called {\it weakly acylindrical} if
there exists a constant ${\mathcal D} > 0$ such that for every hyperbolic element $g \in G$, 
and every $h \not \in E(g)$, we have
\[
\operatorname{diam} \left(\pi_{{\rm Ax}(g)}(h{\rm Ax}(g))\right) \le {\mathcal D}\left(1+{\rm tr}(g)\right).
\]
We call it ${\mathcal D}$-weakly acylindrical if we want to make the constant ${\mathcal D}$ explicit.

<!-- lemma-10: body_md -->
\label{lemma.diam}
Let $\alpha$, $\beta$ be $\Delta$-quasi-axes of some hyperbolic isometries in a $\delta$-hyperbolic geodesic metric space $X$.
For every $R\ge 0$, we have
\[
\operatorname{diam}(\alpha\cap {\mathcal{N}}_{R}(\beta)) \le \operatorname{diam} \pi_\alpha(\beta) +4R.
\]
Conversely, we have
\[
\operatorname{diam} \pi_\alpha(\beta) \le \operatorname{diam}(\alpha\cap {\mathcal{N}}_{20\delta+2\Delta}(\beta))+10\delta+2\Delta.
\]
<!-- end lemma-10 -->
<!-- head-7: body_md -->
We prove the first inequality.
We assume that $\alpha\cap {\mathcal{N}}_{R}(\beta)\neq \emptyset$ since otherwise the inequality holds.
For every $x \in \alpha \cap {\mathcal{N}}_{R}(\beta)$, 
let $y \in \beta$ with $d(x, y) \le R$.
Since $d(y, \pi_\alpha(y)) \le R$,
we have $x \in {\mathcal{N}}_{2R}(\pi_\alpha(\beta))$.
Hence
\[
\alpha \cap {\mathcal{N}}_{R}(\beta) \subset {\mathcal{N}}_{2R}(\pi_\alpha(\beta)),
\]
implying the first inequality.


We show the second inequality.
Let $a_1, a_2\in \alpha$. 
Further, let $b_1, b_2\in \beta$ be points such that $a_i \in \pi_{\alpha}(b_i)$, $i=1, 2$.
We consider a geodesic rectangle on $a_1, a_2, b_2, b_1$.

Suppose $d(a_1, a_2)>10\delta+2\Delta$.
As in the proof of Lemma \ref{lemma.rectangle}, we have
\[
(b_1, a_2)_{a_1} \le 4\delta+\Delta \quad \text{and} \quad (b_1, b_2)_{a_2} \le 5\delta+\Delta.
\]
Thus,
by definition of the Gromov product, we obtain
\[
(a_1, b_1)_{a_2}=d(a_1, a_2)-(b_1, a_2)_{a_1} \ge d(a_1, a_2)-4\delta-\Delta>5\delta+\Delta\ge (b_1, b_2)_{a_2}.
\]
Consider geodesic triangles on $a_1, a_2, b_1$, and on $b_1, a_2, b_2$.
By using the tripod lemma (Lemma \ref{lemma.tripod}), 
we have
\[
d(a_1, \beta) \le 12 \delta+2\Delta \quad \text{and} \quad d(a_2, \beta)\le 9\delta+2\Delta.
\]
Thus, $a_1, a_2 \in \alpha \cap {\mathcal{N}}_{12\delta+2\Delta}(\beta)$.
Therefore, if $\operatorname{diam} \pi_\alpha(\beta)>10\delta+2\Delta$,
then
$\operatorname{diam} \pi_\alpha(\beta) \le \operatorname{diam} (\alpha \cap {\mathcal{N}}_{20\delta+2\Delta}(\beta))$.
The desired second inequality follows by
combining the case when $\operatorname{diam} \pi_\alpha(\beta)\le 10\delta+2\Delta$,
completing the proof.
\qed

<!-- prop-1: body_md -->
\label{prop.overlap}
There exists a constant $Z>0$ that depends only on $n$ and $a$ with
the following property. 
Let $G < \operatorname{{\rm Isom}}(X)$ be a discrete group on $X$, 
and $g\in G$ hyperbolic.   
For every $h\in G$ such that $ h \not\in E(g)$, 
we have
\[
\operatorname{diam} \left({\rm Ax}(g) \cap {\mathcal{N}}_{50\delta}(h{\rm Ax}(g))\right) \le (n+2) {\rm tr}(g) + Z.
\]
<!-- end prop-1 -->
<!-- cor-3: body_md -->
\label{cor.Hn}
Let $n \in {\mathbb N}$ and $a \in [1, +\infty)$.
Let $X$ be an $n$-dimensional Hadamard manifold $X$ with sectional curvature $K_X\in [-a^2, -1]$.
Let $G <\operatorname{{\rm Isom}}(X)$ be a discrete subgroup. 

Then there exists a constant ${\mathcal D}={\mathcal D}(n, a)$ such that the action of $G$ on $X$ is ${\mathcal D}$-weakly acylindrical.
<!-- end cor-3 -->
<!-- head-8: body_md -->
Let $\delta$ be a hyperbolicity constant of $\mathbb H^2$.
Then $X$ is $\delta$-hyperbolic.
As we said, since the axis of a hyperbolic isometry of $X$ is
a geodesic, we may take $\Delta=0$ in Lemma \ref{lemma.diam}.

Let $g\in G$ be hyperbolic and $h\in G\setminus E(g)$.
By Lemma \ref{lemma.diam} and Proposition \ref{prop.overlap}, we have
\[
\begin{aligned}
\operatorname{diam}\pi_{{\rm Ax}(g)}
  (h{\rm Ax}(g))
&\le
\operatorname{diam} \left(
  {\rm Ax}(g) \cap
  {\mathcal{N}}_{20\delta}(h{\rm Ax}(g)) \right)
  +10\delta\\
&\le (n+2) {\rm tr}(g)+Z+10\delta.
\end{aligned}
\]
Setting
$
\mathcal D=\max\{n+2,Z+10\delta\},
$
we obtain the claim. Note that $\mathcal D$ depends
only on $n$ and $a$.
\qed

<!-- lemma-11: body_md -->
\label{lemma.nonelem}
Keep the same setting as in Corollary \ref{cor.Hn}.
For every finite set $U$ in $\operatorname{{\rm Isom}}(X)$,
if $\<U\>$ is discrete and non-virtually-nilpotent, 
then the action of 
$\<U\>$ is non-elementary on $X$.
<!-- end lemma-11 -->
<!-- head-9: body_md -->
By Lemmas \ref{lemma.independent} and \ref{lemma.independent.2}, $\<U\>$ contains a hyperbolic element $g$. Let $E(g)$ be the elementary closure in $\<U\>$. 
Then, $E(g)$ is virtually cyclic since it preserves $\operatorname{{\rm Ax}}(g)$
and $E(g)$ is discrete.
It implies that $\< U\> \not\subset E(g)$
since $\<U\>$ is not virtually-nilpotent. 

Let $u \in \<U \> \backslash E(g)$. 
Set $h=ugu^{-1} \in \< U\>$, which is hyperbolic. Then $g$ and $h$ are independent. 
Indeed, otherwise, $g$ and $h$ fix a common point in $\partial X$.
Then, $\operatorname{diam} \pi_{\operatorname{{\rm Ax}}(g)} (u \operatorname{{\rm Ax}}(g))$ is infinite. 
This contradicts Corollary \ref{cor.Hn} since it in particular says that 
$\operatorname{diam} \pi_{\operatorname{{\rm Ax}}(g)} (u \operatorname{{\rm Ax}}(g))$ is finite.
\qed

<!-- definition-2: body_md -->

A finite set $S$ of isometries of $X$ is $(\eta, C, D)$-{\bf Schottky} at $o$ if the following holds:
\begin{itemize}
\item For all $x, y \in X$, 
we have
$\#\{s \in S \ : \ (sx, y)_o \le C\} \ge (1-\eta)\#S$.
\item For all $x, y \in X$,
we have 
$\#\{s \in S \ : \ (s^{-1}x, y)_o \le C\} \ge (1-\eta)\#S$.
\item 
For all $s\in S$, 
we have $d(o, so) \ge D$.
\end{itemize}

Let  $w$ and $w'$ be non-empty reduced words on $\{u, u^{-1}, u', u'^{-1}\}$ of length $n$ and $m$, respectively:

\label{prop.schottky}
Let $N \in {\mathbb N}$ and $\delta, {\mathcal D} \ge 0$.
Let $X$ be a $\delta$-hyperbolic geodesic space. 
Let $U$ be a finite set of isometries in $X$ with $1 \in U$ and $U=U^{-1}$. 
Let $g$ be a hyperbolic isometry of $X$.
Suppose that the following conditions hold:
\begin{enumerate}[(i)]
\item\label{assump.h}  $U^N$ contains $g$ such that 
${\rm tr}(g) \ge L(U)+4000\delta+1$.
\item\label{assump.action}
$\<U\>$ does not preserve the set of fixed points of $g$ in $\partial X$.

\item\label{assump.wa} The action of $\langle U\rangle$ on $X$ is ${\mathcal D}$-weakly acylindrical.
\end{enumerate}
Then there exists a positive integer $M=M(\delta, {\mathcal D}, N)$, a point $o\in X$, and positive constants $C(\delta, {\mathcal D}, N)$ and $D(\delta, {\mathcal D}, N)$ such that the following holds:


\begin{itemize}
\item 
We have $D\ge 20C+100\delta+L(U)$ for  
\[
C=C(\delta, {\mathcal D}, N)(L(U)+1) \quad \text{and} \quad D=D(\delta, {\mathcal D}, N)(L(U)+1).
\]


\item $U^M$ contains a $(1/100, C, D)$-Schottky set $S$ at $o$. 

\item We have $\# S \ge 200$.
\end{itemize}

Under the assumptions of Proposition \ref{prop.schottky},

We first prove that $o \in {\mathcal{N}}_{\Lambda}({\rm Ax}(g))$.
Suppose, to the contrary, that $o\notin {\mathcal{N}}_{\Lambda}({\rm Ax}(g))$.
Since $1\in U$, we have $\{g, h\} \subset U^{N+2}$, and thus, by Lemma \ref{lemma.disp} \eqref{item.Un},
\[
L(\{g, h\}) \le L(U^{N+2}) \le (N+2)L(U).
\]
On the one hand, 
by \eqref{eq.basepoint},
\begin{align}\label{eq.tube2}
d(o, g o)	
\le L(\{g, h\})+1
\le (N+2)L(U)+1.
\end{align}
On the other hand,
since $o \notin {\mathcal{N}}_{\Lambda}({\rm Ax}(g))$,
we have
\begin{equation}\label{eq.tube3}
2\Lambda+{\rm tr}(g) \le d(o, g o)+8000\delta.
\end{equation}
To see this, 
choose the points $o, x, g x, g o$ where $x \in {\rm Ax}(g)$ with $d(o, x)= d(o, {\rm Ax}(g))$.
Note that $d(x, g x)\ge {\rm tr}(g) \ge 4000\delta$ by definition of ${\rm tr}(g)$ and the assumption \eqref{assump.h}.
Then, Lemma \ref{lemma.rectangle} shows \eqref{eq.tube3}.

Combining \eqref{eq.tube2} and \eqref{eq.tube3},
we obtain
\[
2\Lambda+{\rm tr}(g) \le (N+2)L(U)+8000\delta+1.
\]
Since ${\rm tr}(g) \ge L(U)+4000\delta+1$ by the assumption \eqref{assump.h},
this implies 
\[
2\Lambda\le (N+1)L(U)+4000\delta.
\]
This is a contradiction since $\Lambda=NL(U)+4000\delta+1$.
Thus, we have shown $o \in {\mathcal{N}}_{\Lambda}({\rm Ax}(g))$.

Similarly, $o \in {\mathcal{N}}_{\Lambda}({\rm Ax}(h))$ since ${\rm tr}(h)={\rm tr}(g)$ and \eqref{eq.basepoint} holds.
Thus, we obtain \eqref{eq.tube1}.



It suffices to prove the second claim for $g^m$ and $h^m$ with $m \ge 0$
since the case for $m <0$ follows by replacing $g$ and $h$ with their inverses. 



We show that
for every $m\ge 0$, 
\begin{equation}\label{eq.tube4}
(g^m o, h^m o)_o -2\Lambda -8\delta\le \operatorname{diam} ({\rm Ax}(g)\cap{\mathcal{N}}_{2\Lambda+20\delta}({\rm Ax}(h))).
\end{equation}

First, by \eqref{eq.tube1},
we have $x \in {\rm Ax}(g)$ and $y \in {\rm Ax}(h)$ such that $d(o, x) \le \Lambda$ and $d(o, y) \le \Lambda$.
Furthermore, by the property of quasi-axes (Lemma \ref{lemma.quasiaxis} (i)),
we have $g^m x \in {\rm Ax}(g)$ and $h^m y \in {\rm Ax}(h)$.
Next, by the tripod lemma (Lemma \ref{lemma.tripod}),
there exist $p \in [o, g^m o]$ and $q \in [o, h^m o]$ such that

d(p, {\rm Ax}(g)) \le \Lambda+8\delta \quad \text{and} \quad d(q, {\rm Ax}(h)) \le \Lambda+8\delta.

The first inequality follows by considering a geodesic rectangle on $o, x, g^m x, g^m o$ and applying the tripod lemma (Lemma \ref{lemma.tripod}) twice, and the second inequality follows similarly.
Let $p' \in {\rm Ax}(g)$ and $q' \in {\rm Ax}(h)$ with $d(p, p') \le \Lambda+8\delta$ and $d(q, q')\le \Lambda+8\delta$, respectively.
By the triangle inequality, we have
$d(x, y) \le 2\Lambda$ and $d(p', q') \le 2\Lambda+20\delta$.
Thus, we obtain

x, p' \in {\rm Ax}(g) \cap {\mathcal{N}}_{2\Lambda+20\delta}({\rm Ax}(h)).

\[

This shows \eqref{eq.tube4}.

we have:
\begin{align*}

&\le {\mathcal D}({\rm tr}(g)+1)+8\Lambda+80\delta.
\end{align*}
Rearranging the terms, we obtain the claim.

<!-- lemma-15: body_md -->
\label{lemma.separation}
Under the assumptions of Proposition \ref{prop.schottky},
we have the following:
For every $m \in {\mathbb Z}$, 
we have 
\begin{equation}\label{eq.qa}
\max\left\{(g^m o, g^{-m}o)_o, (h^m o, h^{-m}o)_o\right\} \le C_0(\delta, N)(L(U)+1),
\end{equation}
and
\begin{equation}\label{eq.wa}
\max\left\{(g^m o, h^m o)_o, (g^m o, h^{-m} o)_o\right\} \le C_1(\delta, {\mathcal D}, N)(L(U)+1),
\end{equation}
where $C_0(\delta, N)=N+10^4\delta+1$ and
$C_1(\delta, {\mathcal D}, N)={\mathcal D} N+10N+10^5\delta+10$.



Furthermore, for every $m \in {\mathbb Z}$, we have 
\begin{equation}\label{eq.D}
\min\big\{d(o, g^m o), d(o, h^m o) \big\}\ge |m|(L(U)+1),
\end{equation}
and
\begin{equation}\label{eq.Dup}
\max\big\{d(o, g^m o), d(o, h^m o)\big\} \le |m|((N+2)L(U)+1).
\end{equation}
<!-- end lemma-15 -->

First, we show \eqref{eq.qa}.
Let us show the required bound on $(g^m o, g^{-m}o)_o$ for every integer $m \ge 1$, 
as the Gromov product equals $0$ when $m=0$.
The bound on $(h^m o, h^{-m}o)_o$ follows analogously.

By Lemma \ref{lemma.tube}, there exists $x \in {\rm Ax}(g)$ such that $d(o, x)\le \Lambda$.
For every $m\ge 1$, 
since ${\rm tr}(g) > 4000\delta$ by the assumption \eqref{assump.h},
we have
\[
d(g^{-m}x, g^m x)=d(x, g^{2m} x) \ge {\rm tr}(g^{2m})=2m{\rm tr}(g) > 8000m\delta.
\]
Applying Lemma \ref{lemma.rectangle} to $g^{-m}o$, $g^{-m}x$, $g^m x$, $g^m o$,
we obtain 
\[
2d(o, x)+d(g^{-m}x, g^m x) \le d(g^{-m}o, g^m o)+8000\delta.
\]
Since $g^{-m}x, x, g^m x$ belong to ${\rm Ax}(g)$, and ${\rm Ax}(g)$ is a $\Delta$-quasi-axis,
the $\Delta$-neighborhood of a geodesic $[g^{-m}x, g^m x]$ contains $x$.
Thus, we have
\[
d(g^{-m}x, x)+d(x, g^m x) \le d(g^{-m}x, g^m x)+2\Delta.
\]
Therefore, we obtain
\[
2d(o, x)+d(g^{-m}x, x)+d(x, g^m x) \le d(g^{-m}o, g^m o)+8000\delta+2\Delta.
\]
By definition of the Gromov product and the triangle inequality,
we obtain:
\begin{align*}
(g^{-m}o, g^m o)_o
&=\frac{1}{2}(d(o, g^{-m}o)+d(o, g^m o)-d(g^{-m}o, g^m o))\\
&\le \frac{1}{2}(4d(o, x)+d(x, g^{-m}x)+d(x, g^m x)-d(g^{-m}o, g^m o))\\
&\le d(o, x)+4000\delta+\Delta\le \Lambda+4000\delta+\Delta.
\end{align*}
Recall that $\Lambda=NL(U)+4000\delta+1$. 
Moreover, $\Delta\le 2000\delta$ by Lemma \ref{lemma.quasiaxis}.
Thus, we have
\[
\Lambda+\Delta+4000\delta \le NL(U)+10^4\delta+1\le C_0(\delta, N)(L(U)+1).
\]
This implies \eqref{eq.qa}.

Next, we show \eqref{eq.wa}.
Since $g \in U^N$,
by Lemma \ref{lemma.disp}, we have ${\rm tr}(g) \le N L(U)$.
By Lemma \ref{lemma.tube}, 
for every $m \ge 0$,
we have
\begin{align*}
(g^m o, h^m o)_o 
&\le {\mathcal D}({\rm tr}(g)+1)+10\Lambda+100\delta\\
&\le {\mathcal D}(N L(U)+1)+10(NL(U)+4000\delta+1)+100\delta\\
&\le ({\mathcal D} N+10N+10^5\delta+10)(L(U)+1).
\end{align*}
For every $m \ge 0$, we have
\[
(g^m o, h^m o)_o \le C_1(\delta, {\mathcal D}, N)(L(U)+1),
\]
where $C_1(\delta, {\mathcal D}, N)={\mathcal D} N+10N+10^5\delta+10$.
The same inequality holds for $(g^m o, h^{-m}o)_o$.
Thus, we have \eqref{eq.wa}.

Finally, we show \eqref{eq.D} and \eqref{eq.Dup}.
Since ${\rm tr}(g)={\rm tr}(g^{-1})$,
by the assumption \eqref{assump.h}, for every $m \in {\mathbb Z}$, we have
\[
d(o, g^m o) \ge {\rm tr}(g^m)=|m|{\rm tr}(g) \ge |m|(L(U)+1).
\]
The analogous inequality holds for $d(o, h^m o)$ since ${\rm tr}(g)={\rm tr}(h)$.
Hence we obtain \eqref{eq.D}.
Furthermore, 
we have
\[
d(o, g^m o) \le |m|((N+2)L(U)+1),
\]
by the triangle inequality, \eqref{eq.basepoint}, and Lemma \ref{lemma.disp} \eqref{item.Un} since $\{g, h\}\subset U^{N+2}$.
The analogous inequality holds for $d(o, h^m o)$.
Thus, we obtain \eqref{eq.Dup}.
\qed

<!-- proof-1: body_md -->
By \cite[Lemma 4.9]{G}, for all $i \in {\mathbb Z}$,
we have
\[
\operatorname{{\bf P}}(\#P_{n+1}\ge i)\ge \operatorname{{\bf P}}(\#P_n+U \ge i),
\]
where $U$ is a random variable independent from $\#P_n$ distributed as
\[
\operatorname{{\bf P}}(U=-j)=9\left(\frac{1}{10}\right)^{j+1} \ \text{for $j>0$}, \
\operatorname{{\bf P}}(U=0)=0 \ \text{and} \
\operatorname{{\bf P}}(U=1)=\frac{9}{10}. 
\]
Thus, letting $U_i$, $i\in {\mathbb N}$, be independent copies of $U$, independently from the sequence $\# P_n$, $n\in {\mathbb N}_0$, for each $i\in {\mathbb Z}$ and each $n \in {\mathbb N}$, we have
\[
\operatorname{{\bf P}}(\# P_n\ge i)\ge \operatorname{{\bf P}}(U_1+\cdots+U_n \ge i).
\]
Therefore, for every $r>0$ and every $n \in {\mathbb N}_0$, we have
\[
\operatorname{{\bf P}}(\# P_n \le r n) \le \operatorname{{\bf P}}(U_1+\cdots+U_n \le r n).
\]
We show that for every $r\in (0, \operatorname{{\bf E}} U)$, there exists $c_r>0$ such that
for all $n \in {\mathbb N}_0$,
\begin{equation}\label{eq:ldp}
\operatorname{{\bf P}}(U_1+\cdots+U_n \le r n) \le e^{-c_r n}.
\end{equation}
This implies the claim by taking $r=1/2<\operatorname{{\bf E}} U=71/90$.

Let us show \eqref{eq:ldp}.
For every $r \in (0, \operatorname{{\bf E}} U)$, define ${\varepsilon}:=\operatorname{{\bf E}} U-r>0$.
For every $t>0$, by the Markov inequality and the independence of $U_i$, $i\in {\mathbb N}$, we have the following:
\begin{align*}
\operatorname{{\bf P}}\Big(U_1+\cdots+U_n \le rn\Big) 
&=\operatorname{{\bf P}}\Big(-\sum_{i=1}^n (U_i-\operatorname{{\bf E}} U) \ge {\varepsilon} n\Big)\\
&\le e^{-t {\varepsilon} n}\Big(\operatorname{{\bf E}} e^{-t(U-\operatorname{{\bf E}} U)}\Big)^n.
\end{align*}
                                            
Since $\operatorname{{\bf E}} e^{-t (U-\operatorname{{\bf E}} U)}=e^{o(t)}$,
for a small enough $t>0$ with ${\varepsilon}_t:={\varepsilon} -o(t)/t>0$,
the last term in the above displayed inequality is at most $\exp(-t{\varepsilon}_t n)$ for all $n \in {\mathbb N}_0$.
Setting $c_r:=t {\varepsilon}_t>0$ yields \eqref{eq:ldp}.
This completes the proof.

\label{thm.meta}
Let $\delta, {\mathcal D} \ge 0$, $N, q \in {\mathbb N}$, and $\theta_0\in (0, 1]$.
Then there exist constants $\kappa_\star(\delta, {\mathcal D}, N, q, \theta_0)>0$ and $c_\star(\delta, {\mathcal D}, N, q, \theta_0)>0$ such that the following holds.

Let $U$ be a finite symmetric set in $\operatorname{{\rm Isom}} (X)$ with $1 \in U$. 
Assume

Let $\mu$ be a countably supported probability measure on $\operatorname{{\rm Isom}} (X)$
with 
$$\min_{u\in U}\mu^{\ast q}(u)\ge \theta_0.$$





Then, for all large enough $n$,
we have
\[
\operatorname{{\bf P}}_\mu\Bigg(\frac{1}{n L(U)}d(o, w_n o) \le \kappa_\star(\delta, {\mathcal D}, N, q, \theta_0) \Bigg)\le  \exp\Bigg(-c_\star(\delta, {\mathcal D}, N, q, \theta_0) n\Bigg).
\]
Consequently, if $\mu$ has a finite first moment, then
\[
\ell(\mu)\ge \kappa_\star(\delta, {\mathcal D}, N, q, \theta_0) L(U).
\]
Moreover, there exists $M\in {\mathbb N}$ depending only on $\delta$, ${\mathcal D}$, and $N$, such that
\[
\kappa_\star(\delta, {\mathcal D}, N, q, \theta_0)=\frac{\theta_0^{2M}}{16qM}.
\]

\label{thm.drift_general}
Let $\delta, {\mathcal D} \ge 0$, and $N \in {\mathbb N}$.
Let $(X, d)$ be a $\delta$-hyperbolic geodesic space. 
Let $\Gamma$ be a finitely generated group, and let $\nu$ be an admissible probability measure on $\Gamma$ with finite first moment.
Let $\Sigma$ be a finite symmetric generating set of $\Gamma$ with $1 \in \Sigma$.



Then, there exists a constant  $\kappa(\delta, {\mathcal D}, N, \Gamma, \Sigma, \nu)>0$ such that for
every faithful, non-elementary and ${\mathcal D}$-weakly acylindrical isometric action $\rho:\Gamma \to \operatorname{{\rm Isom}} (X)$  satisfying $L(\rho(\Sigma)^N)\ge K_0\delta+1$,
we have
\[
\ell(\nu, \rho)\ge \kappa(\delta, {\mathcal D}, N, \Gamma, \Sigma, \nu)L(\rho(\Sigma)).
\]
Here $K_0$ is the absolute constant in Theorem \ref{thm.meta}.
<!-- end thm-8 -->
<!-- head-17: body_md -->
Let $U:=\rho(\Sigma)$, and $\mu:=\rho_\ast \nu$.
We may assume that  $1 \in \operatorname{{\rm supp}} \nu$ since otherwise Lemma \ref{lemma.lazy}
reduces the general case to this one at the cost of a factor of $2$.  
To apply Theorem \ref{thm.meta}, we first verify its assumptions and identify the relevant constants. 
We claim that there exist $q=q(\Gamma, \Sigma, \nu) \in {\mathbb N}$ and $\theta_0=\theta_0(\Gamma, \Sigma, \nu) \in (0, 1]$ such that for every faithful isometric action $\rho$ of $\Gamma$ on $X$,
we have
$\min_{u \in \rho(\Sigma)}(\rho_\ast \nu)^{\ast q} \ge \theta_0$.

Indeed, since $1 \in \operatorname{{\rm supp}} \nu$ and $\nu$ is admissible, the support of $\nu^{\ast n}$ is weakly increasing in $n$, and the union of these supports coincides with $\Gamma$.
Moreover, since $\Sigma$ is a finite set,
there exists $q \in {\mathbb N}$ such that $\Sigma \subset \operatorname{{\rm supp}} \nu^{\ast q}$.
Letting $\theta_0:=\min\{\nu^{\ast q}(s) \ : \ s \in \Sigma\}>0$,
we have $\min_{s\in \Sigma}\nu^{\ast q}(s)\ge \theta_0$.
Furthermore, by the injectivity of $\rho$,
we have $\min_{u\in \rho(\Sigma)}(\rho_\ast \nu)^{\ast q}(u)\ge \theta_0$, as required.
By definition, $q$ and $\theta_0$ depend only on $\Gamma$, $\Sigma$, and $\nu$, but not on $\rho$.
Moreover, we have $\rho(\Sigma)^{-1}=\rho(\Sigma)$ since $\Sigma$ is symmetric.




Note that $\mu=\rho_\ast \nu$ has a finite first moment on $(X, d)$ for every $\rho$.
Therefore, by Theorem \ref{thm.meta},
letting $\kappa(\delta, {\mathcal D}, N, \Gamma, \Sigma, \nu):=\kappa_\star(\delta, {\mathcal D}, N, q, \theta_0)$,
we obtain
\[
\ell(\nu, \rho)=\ell(\rho_\ast \nu)\ge \kappa(\delta, {\mathcal D}, N, \Gamma, \Sigma, \nu) L(\rho(\Sigma)).
\]
This shows the theorem.

if $\ell (\mu_U) \le A$, then 
$L(U) \le A/\kappa_1(\delta, m_0, m)$.
Then, there exists an element $g$ in $G$ such that $g^{-1}Ug$ is contained in the ball of radius
$A/\kappa_1(\delta, m_0, m)+1$ centered at $1$ in ${\rm Cay}(G, U_0)$.
Thus, 
there are only finitely many such $U$ up to conjugation by an element in $G$.
In particular, this implies that for each  $A>0$,
the set $\mathcal{L}(G, U_0, m) \cap (0, A]$ is finite, so that the set $\mathcal{L}(G, U_0, m)$ is discrete.

Finally, the set is unbounded since $L(U)$ is unbounded if we vary $U$.
To see this, take two independent infinite order 
elements $g,h \in G$, set 
\[
U=\{g^n, g^{-n}, h^n, h^{-n}\}
\]
and let $n \to \infty$, so that the set $\mathcal{L}(G, U_0, m)$ is infinite and unbounded. 
\qed

<!-- remark-2: body_md -->
\label{rem.KKS}
Corollary \ref{cor.hyperbolic.spec} may be viewed as a partial generalization
of \cite[Theorem D]{KKS} on free groups to 
hyperbolic groups, possibly with torsion.
In \cite{KKS}, they discuss the set of values of {\it generic stretching factors} on finite rank non-abelian free groups (see their definition).

To explain the relation, fix a Cayley graph of a free group of rank $m$ associated to a free basis.
For the simple random walk associated with another free basis, the drift is equal to the generic stretching factor multiplied by $(m-1)/m$ (see \cite[Proof of Theorem 2.2, p.16]{KKS}).

Thus, in our terminology, \cite[Theorem D]{KKS} implies that the drifts obtained by varying the free basis belong to $\frac{1}{N}\mathbb Z$ for some positive integer $N$. We do not know  whether 
the set defined in 
Corollary \ref{cor.hyperbolic.spec} is contained in $r {\mathbb Z}$ for some $r \in {\mathbb R}$ for general hyperbolic groups.

<!-- lemma-18: body_md -->
\label{lemma.mcg}
There exists a constant $N=N(\mathcal S)$ such that for 
every finite set $U$ in ${\rm Mod}(\mathcal S)$ with
$\<U\>$ containing a pseudo-Anosov element, 
we have
\[
L(U^N) \ge K_0\delta+1.
\]
Here $K_0$ is the absolute constant in Theorem \ref{thm.meta}, and $\delta$ is the hyperbolicity constant of $\mathcal C(\mathcal S)$.
<!-- end lemma-18 -->
<!-- head-24: body_md -->
It is known that there exists a constant $N_1=N_1(\mathcal S)$ such that 
if $\<U\>$ contains a pseudo-Anosov element, then
$U^{N_1}$ contains a pseudo-Anosov element $g$ \cite[Theorem 1.1]{M}.

Also, there is a constant $\tau=\tau(\mathcal S)>0$ such that ${\rm tr}(u) \ge \tau$ for every pseudo-Anosov
element $u$ in ${\rm Mod}(\mathcal S)$ \cite[Proposition 4.6]{MM}.
Hence, for every $n \in {\mathbb N}_0$,
we have $L(U^{N_1 n}) \ge {\rm tr}(g^n) \ge n\tau$ by Lemma \ref{lemma.disp} \eqref{item.trL}.
Letting $n:=\lceil (K_0\delta+1)/\tau \rceil$ and $N:=N_1 n$,
we have
\[
L(U^N) \ge K_0\delta+1.
\]
We note that $N$ depends only on $\mathcal S$.
\qed

<!-- thm-10: body_md -->
\label{thm.mcg}
Let $\mathcal S$ be a closed orientable surface of genus at least two, and $m \in {\mathbb N}$. 
Let us consider the action of ${\rm Mod}(\mathcal S)$ on the curve graph $\mathcal C(\mathcal S)$ endowed with the combinatorial path metric.
Then, there exists a constant $\kappa(\mathcal S,m)>0$
such that the following holds:

Let $U$ be a finite symmetric set in ${\rm Mod}(\mathcal S)$ with $\#U=m$ such that $\<U\>$ contains a pseudo-Anosov element 
and not virtually cyclic. 
Consider the uniform distribution $\mu_U$ on $U$.
Then, we have
$$\ell(\mu_U) \ge \kappa(\mathcal S,m) L(U).$$
<!-- end thm-10 -->
<!-- head-25: body_md -->
The curve graph $\mathcal C(\mathcal S)$ is $\delta$-hyperbolic, and the action of ${\rm Mod}(\mathcal S)$ on $\mathcal C(\mathcal S)$ is $\mathcal D$-weakly acylindrical, where $\delta$ and ${\mathcal D}$ depend only on $\mathcal S$. 
By assumption, $\<U\>$ contains a pseudo-Anosov element.
By Lemma \ref{lemma.mcg}, 
there exists $N=N(\mathcal S) \in {\mathbb N}$, which depends only on $\mathcal S$, such that $L(U^{N}) \ge K_0 \delta +1$.

The elementary closure of every pseudo-Anosov element is virtually ${\mathbb Z}$.
Since by assumption $\<U\>$ contains a pseudo-Anosov element and is not virtually cyclic, 
there exists $u \in \<U\>$ such that $u \notin E(g)$.
Note that $g$ and $ugu^{-1}$ have disjoint sets of fixed points in the boundary of $\mathcal C(\mathcal S)$. Indeed, if otherwise, then the diameter of $\pi_{{\rm Ax}(g)}(u{\rm Ax}(g))$ is infinite, which contradicts the (weak) acylindricity of the ${\rm Mod}(\mathcal S)$-action on $\mathcal C(\mathcal S)$.
Thus, the action of $\<U\>$ on $\mathcal C (\mathcal S)$ is non-elementary.

Therefore, by Theorem \ref{thm.drift2}, we have
$$\ell(\mu_U) \ge \kappa(\delta, \mathcal D, N(\mathcal S), m) L(U).$$
Taking $\kappa(\mathcal S,m)=\kappa(\delta, \mathcal D, N(\mathcal S), m)$ gives the desired conclusion. 
\qed

<!-- cor-10: body_md -->
\label{cor.gap.mcg}
Let $\mathcal S$ be a closed orientable surface of genus at least two, and 
$m \in {\mathbb N}$. 
Then, there exists a constant $c(\mathcal S, m)>0$ such that the following holds:

For every finite set $U$ in ${\rm Mod}(\mathcal S)$ with $\# U=m$ and $U=U^{-1}$,
we have
\begin{enumerate}[(1)]
\item
$\ell(\mu_U) \ge c(\mathcal S, m)$, if
$\<U\>$ contains at least one pseudo-Anosov element, and $\< U\>$ is not virtually cyclic.


\item
Otherwise, $\ell(\mu_U)=0$.
\end{enumerate}
<!-- end cor-10 -->
<!-- head-26: body_md -->
(1) 
Since $\<U\>$ contains a pseudo-Anosov element, 
by Lemma \ref{lemma.mcg}, 
there exists $N=N(\mathcal S)$, which depends only on $\mathcal S$, such that $L(U^{N}) \ge K_0 \delta +1$, where $K_0$ is the absolute constant in Theorem \ref{thm.meta} and $\delta$ is the hyperbolicity constant of the curve graph $\mathcal C(\mathcal S)$.
Hence, by Theorem \ref{thm.mcg}, we have
\[
\ell(\mu_U) \ge \kappa(\mathcal S,m) L(U) \ge \kappa(\mathcal S, m)\frac{K_0 \delta+1}{N(\mathcal S)}.
\]
Letting $c(\mathcal S, m):=\kappa(\mathcal S, m)(K_0\delta+1)/N(\mathcal S)$ yields the desired inequality.


(2) We use the classification of subgroups of mapping class groups.
First, suppose that $\<U\>$ is finite, then its orbit is finite, hence clearly
$\ell(\mu_U)=0$. Second, suppose that 
$\< U\>$ is a reducible subgroup in ${\rm Mod}(\mathcal S)$. Then there is a finite set
of vertices $V$ in $\mathcal C(\mathcal S)$ that is invariant by $\< U\>$.
Again, the orbit by $\< U\>$ is finite, so that $\ell(\mu_U)=0$.
Finally, suppose $\<U\>$ contains a pseudo-Anosov element $g$, and $\<U\>$ is virtually cyclic.
Then, we have $\ell(\mu_U)=0$ (cf.\ the proof of Corollary \ref{cor.dichotomy}).
This completes the argument. 
\qed
