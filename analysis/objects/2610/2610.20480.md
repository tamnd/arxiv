<!-- definition-1: body_md -->
\label{defn}
Two dispersive equations of the form (\ref{dispersive}) corresponding to $\psi_1,\psi_2$ are defined to be {\it of comparable oscillation} if there exist $\Lambda>0$ and $C>0$, such that 
\begin{equation*}
\left|\psi_1(\|\xi\|)-\psi_2(\|\xi\|)\right| \le C\:, \text{ for all } \|\xi\| > \Lambda\:.
\end{equation*}
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{transference_theorem}
Let $\psi_1$ and $\psi_2$ be radial, real-valued functions on $\mathbb R^n$. For two dispersive equations with phases $\psi_1,\psi_2$ which are of comparable oscillation, we have
\begin{equation*}
\mathfrak d_{\psi_1,n}(s)=\mathfrak d_{\psi_2,n}(s),\:\:\:\text{ for all } s>0\:.
\end{equation*}
<!-- end theorem-1 -->
<!-- remark-1: body_md -->
To properly contextualize Theorem \ref{transference_theorem}, it should be remarked that all functions $\mathfrak d_{\psi,n}$ are well-defined for all $s>0$, with $\mathfrak d_{\psi,n}(s) \ge n-2s$ (with $\mathfrak d_{\psi,n}(s) = 0$ for all $s>n/2$). For more details, we refer to Section \ref{sec3}.
<!-- end remark-1 -->
<!-- remark-2: body_md -->
As applications of Theorem \ref{transference_theorem},
\begin{enumerate}
\item  we obtain Hausdorff dimension estimates of divergence sets for families of abstract dispersive equations, classified solely in terms of the oscillation of their Fourier multipliers in high frequency (Corollaries \ref{corollary1}, \ref{corollary2}, \ref{corollary3} and \ref{corollary4});
\item as special cases,
\begin{enumerate}
\item we recover the known one-dimensional results for the linearized Boussinesq and the Beam propagations (Remark \ref{recovering_results_beam});
\item we also obtain genuinely new results for the linearized Boussinesq and the Beam propagations in higher dimensions (Remark \ref{new_results_beam}) and the Klein-Gordon propagations in all dimensions (Remark \ref{new_results_klein}).
\end{enumerate}
\end{enumerate}
For more details, we refer to Section \ref{sec5}.
<!-- end remark-2 -->
<!-- remark-3: body_md -->
We note that, in (\ref{radial_dimension_defn}) the supremum is taken only over the class of radial elements in $H^s(\mathbb R^n)$ and thus trivially
\begin{equation*}
\widetilde{\mathfrak d}_{\psi,n} \le \mathfrak d_{\psi,n}\:.
\end{equation*}
But nothing more can be concluded from $\mathfrak d_{\psi,n}$ and hence, estimating $\widetilde{\mathfrak d}_{\psi,n}$ is a variation of the original problem, not a mere special case.

<!-- theorem-2: body_md -->
\label{radial_concave_thm}
For the fractional Schr\"odinger equations (\ref{frac_schrodinger}) with radial initial data, we have
\begin{enumerate}
\item for concave phase i.e. $a \in (0,1)$,
\begin{enumerate}
\item the sharp result in the range,
\begin{equation*}
\widetilde{\mathfrak d}_{\psi,n}(s) =\begin{cases}
n\:,\:\: &s<\frac{a}{4}\\
n-2s\:,\:\: &\frac{1}{4} \le s<\frac{1}{2}\\
0\:,\:\: &\frac{1}{2} < s\:;
\end{cases}
\end{equation*}
\item and in the remaining range, \\
the upper bound,
\begin{equation*}
\widetilde{\mathfrak d}_{\psi,n}(s) \le n-\frac{1}{2}+\frac{1-4s}{2(1-a)}\:,\:\: \frac{a}{4} \le s \le \frac{1}{4}\:,
\end{equation*}
and the lower bound,
\begin{equation*}
\widetilde{\mathfrak d}_{\psi,n}(s) \ge n-2s\:,\:\: \frac{a}{4} \le s \le \frac{1}{4}\:;
\end{equation*}
\end{enumerate}
\item for the Wave propagation i.e. $a=1$,
\begin{equation*}
\widetilde{\mathfrak d}_{\psi,n}(s) =\begin{cases}
n\:,\:\: &s\le \frac{1}{2}\\
0\:,\:\: &\frac{1}{2} < s\:.
\end{cases}
\end{equation*}
\end{enumerate}

In the same spirit, Theorem \ref{radial_concave_thm} refines the results of Walther \cite{Walther1, Walther2, Walther3} for the concave phases and the Wave propagation.

<!-- theorem-3: body_md -->
\label{H3_theorem}
For radial initial data on $\mathbb H^3$, we have
\begin{enumerate}
\item for the Schr\"odinger equation, the fractional Schr\"odinger equations with convex phase, the Beam and the linearized Boussinesq equation,
\begin{equation*} 
\widetilde{\mathfrak d}_{\psi,\mathbb H^3}(s)= \begin{cases}
3,\:\: &s<\frac{1}{4}\\
3-2s,\:\: &\frac{1}{4} \le s<\frac{1}{2}\\
0,\:\: &\frac{1}{2} < s\:;
\end{cases}
\end{equation*}  
\item for the fractional Schr\"odinger equations with concave phase,  i.e. $a \in (0,1)$,
\begin{enumerate}
\item the sharp result in the range,
\begin{equation*}
\widetilde{\mathfrak d}_{\psi,\mathbb H^3}(s) =\begin{cases}
3\:,\:\: &s<\frac{a}{4}\\
3-2s\:,\:\: &\frac{1}{4} \le s<\frac{1}{2}\\
0\:,\:\: &\frac{1}{2} < s\:;
\end{cases}
\end{equation*}
\item and in the remaining range, \\
the upper bound,
\begin{equation*}
\widetilde{\mathfrak d}_{\psi,\mathbb H^3}(s) \le \frac{5}{2}+\frac{1-4s}{2(1-a)}\:,\:\: \frac{a}{4} \le s \le \frac{1}{4}\:,
\end{equation*}
and the lower bound,
\begin{equation*}
\widetilde{\mathfrak d}_{\psi,\mathbb H^3}(s) \ge 3-2s\:,\:\: \frac{a}{4} \le s \le \frac{1}{4}\:;
\end{equation*}
\end{enumerate}
\item for the Wave equation and the Klein-Gordon equations, we have
\begin{equation*} 
\widetilde{\mathfrak d}_{\psi,\mathbb H^3}(s)= \begin{cases}
3,\:\: &s \le \frac{1}{2}\\
0,\:\: &\frac{1}{2} < s\:.
\end{cases}
\end{equation*}
\end{enumerate}

<!-- lemma-1: body_md -->
\label{N-maximal-estimate}
Let $s>0$ and $\alpha >n-2s$. Then for any $f \in H^s(\mathbb R^n)$ and any $\mu \in \mathcal M^\alpha(\mathbb{B}(0,1))$, we have
\begin{equation*}
\left\|\sup_{N \ge 1} \left|S^N_{\psi,t}f\right|\right\|_{L^2(d\mu)} \lesssim \sqrt{c_\alpha(\mu)}\:\|f\|_{H^s(\mathbb R^n)}\:.
\end{equation*}
<!-- end lemma-1 -->
<!-- corollary-1: body_md -->
\label{fractal-Sobolev-embedding}
Let $s>0$ and $\alpha >n-2s$. Then for any $f \in H^s(\mathbb R^n)$, any $\mu \in \mathcal M^\alpha(\mathbb{B}(0,1))$,  we have
\begin{equation*}
\left\|u(\cdot,t)\right\|_{L^2(d\mu)} \lesssim \sqrt{c_\alpha(\mu)}\:\|f\|_{H^s(\mathbb R^n)}\:.
\end{equation*}
<!-- end corollary-1 -->
<!-- remark-7: body_md -->
\label{Zubrinic_remark}
The index $n-2s$ showing up above, is actually the critical index for Hausdorff dimension of singular sets of Sobolev functions in $H^s(\mathbb R^n)$ for $s \in (0,n/2)$. In \cite{Zubrinic}, it was shown that the Bessel potential representation can be singular on sets with dimension smaller than this. Consequently, our problem of estimating Hausdorff dimension of divergence sets become ill-posed for $\mathfrak d_{\psi,n}(s)<n-2s$.

<!-- corollary-2: body_md -->
\label{corollary1}
For a dispersive equation with asymptotically convex phase $\psi$ on $\mathbb R$ i.e. real-valued, even function $\psi$ on $\mathbb R$ satisfying
\begin{equation*}
\psi(|\xi|)=|\xi|^a \:+\:\mathcal{O}(1)\:,\:\:|\xi| \gg 1\:,\:\:a>1\:,
\end{equation*}
we have
\begin{equation*} 
\mathfrak d_{\psi,1}(s) = \begin{cases}
1\:,\:\:& s<\frac{1}{4} \\
1-2s\:,\:\:& \frac{1}{4} \le s \le \frac{1}{2} \\
0\:,\:\:& \frac{1}{2} \le s\:. 
\end{cases}
\end{equation*}
<!-- end corollary-2 -->
<!-- remark-8: body_md -->
\label{recovering_results_beam}
\begin{enumerate}
\item Apart from the fractional Schr\"odinger equations with convex phases, the class of dispersive equations in Corollary \ref{corollary1} also contains the known examples: the linearized Boussinesq and the Beam equation, also known as the fourth order wave equation (see Table \ref{table:1}). The Beam equation physically models how flexural stiffness drives structural bending, causing shorter mechanical waves to travel faster than longer ones.    
\item Corollary \ref{corollary1} recovers the results for the linearized Boussinesq \cite[Theorem 1.3]{LiLi} and the Beam propagations \cite[Theorem 1]{LLN} by means of a simple one-line proof, compared to the oscillatory integral estimates involving higher derivatives in \cite{LiLi,LLN}.
\end{enumerate}
<!-- end remark-8 -->
<!-- proof-1: body_md -->
In the case of the fractional Schr\"odinger equations with convex phase, in dimension one, Sj\"olin proved that $\mathfrak d_{\psi,1}(s) =1$ for $s < 1/4$ \cite{Sjolin}. Then Barcel\'o et. al. established that $\mathfrak d_{\psi,1}(s) = 1-2s$  for $1/4 \le s \le 1/2$, in \cite{BBCR}. The result for $s>1/2$ follows simply by Sobolev embedding.

The result for dispersive equations with asymptotically convex phase, in Corollary \ref{corollary1}, now follows by Theorem \ref{transference_theorem} and the above mentioned results for the fractional Schr\"odinger equations.

<!-- corollary-3: body_md -->
\label{corollary2}
For a dispersive equation with asymptotically concave phase $\psi$ on $\mathbb R$ i.e. real-valued, even function $\psi$ on $\mathbb R$ satisfying
\begin{equation*}
\psi(|\xi|)=|\xi|^a \:+\:\mathcal{O}(1)\:,\:\:|\xi| \gg 1\:,\:\:a \in (0,1)\:,
\end{equation*}
we have
\begin{enumerate}
\item the sharp result in the range,
\begin{equation*}
\mathfrak d_{\psi,1}(s) =\begin{cases}
1\:,\:\: &s<\frac{a}{4}\\
1-2s\:,\:\: &\frac{1}{4} \le s \le \frac{1}{2}\\
0\:,\:\: &\frac{1}{2} \le s\:;
\end{cases}
\end{equation*}
\item and in the remaining range, \\
the upper bound,
\begin{equation*}
\mathfrak d_{\psi,1}(s) \le \frac{1}{2}+\frac{1-4s}{2(1-a)}\:,\:\: \frac{a}{4} \le s \le \frac{1}{4}\:,
\end{equation*}
and the lower bound,
\begin{equation*}
\mathfrak d_{\psi,1}(s) \ge 1-2s\:,\:\: \frac{a}{4} \le s \le \frac{1}{4}\:;
\end{equation*}
\end{enumerate}
<!-- end corollary-3 -->
<!-- proof-2: body_md -->
In the case of the fractional Schr\"odinger equations with concave phase i.e. $a \in (0,1)$, in dimension one, Walther proved that $\mathfrak d_{\psi,1}(s)=1$ for $s<a/4$ \cite{Walther1}.  Then in the remaining range, Cho-Shiraki recently established that
\begin{equation*} 
\mathfrak d_{\psi,1}(s) \le \begin{cases}
 \frac{1}{2}+\frac{1-4s}{2(1-a)}\:,\:\:& \frac{a}{4} \le s \le \frac{1}{4} \\
 1-2s\:,\:\:& \frac{1}{4} \le s \le \frac{1}{2}\:, 
\end{cases}
\end{equation*}
in \cite{CS}. Their bound is sharp in the range $s \in [1/4,1/2]$, by the singularity of Sobolev functions \cite{Zubrinic}, which for the range $s \in [a/4,1/4]$ also yields the lower bound $\mathfrak d_{\psi,1}(s) \ge 1-2s$.  The result for $s>1/2$ follows simply by Sobolev embedding.  
 
The result for dispersive equations with asymptotically concave phase, with $a \in (0,1)$, in Corollary \ref{corollary2}, now follows by Theorem \ref{transference_theorem} and the above mentioned results for the fractional Schr\"odinger equations.

<!-- corollary-4: body_md -->
\label{corollary3}
For a dispersive equation with asymptotically quadratic phase $\psi$ on $\mathbb R^n,\:n \ge 2$ i.e. real-valued, radial function $\psi$ on $\mathbb R^n,\:n \ge 2$, satisfying
\begin{equation*}
\psi(\|\xi\|)=\|\xi\|^2 \:+\:\mathcal{O}(1)\:,\:\:\|\xi\| \gg 1\:,
\end{equation*}
we have,
\begin{enumerate}
\item the sharp result for the following range:
\begin{equation*} 
\mathfrak d_{\psi,n}(s)  =\begin{cases}
n\:,\:\:& s<\frac{n}{2(n+1)} \\
 n-2s\:,\:\:& \frac{n}{4} \le s \le \frac{n}{2} \\
0\:,\:\:& \frac{n}{2} \le s\:; 
\end{cases}
\end{equation*}
\item in the remaining range, 
\begin{enumerate}
\item the upper bound,
\begin{equation*}
\mathfrak d_{\psi,n}(s)  \le n+1-\frac{2(n+1)s}{n}\:,\:\: \frac{n}{2(n+1)} \le s < \frac{n}{4}\:;
\end{equation*}
\item the lower bounds (with $m_0=\lfloor (n-1)/3\rfloor$ and $m_1=\lfloor (n/2)-1\rfloor$),
\begin{enumerate}
\item for dimensions $n=2,3$,
\begin{equation*}
\mathfrak d_{\psi,n}(s)  \ge \alpha^{(1)}_0(s)\:,\:\:\frac{n}{2(n+1)} \le s \le \frac{n}{4}\:;
\end{equation*}
\item for dimensions $n=4,5,6,7$,
\begin{equation*}
\mathfrak d_{\psi,n}(s)  \ge \begin{cases}
\max\left\{\alpha^{(1)}_{m-1}(s)\:,\:\alpha^{(2)}_m(s)\right\}\:,\:\:& \mathfrak{s}^{(1)}_{m-1} \le s \le \mathfrak{s}^{(1)}_{m}\:,\:\: 1\le m \le m_0  \\
\alpha^{(1)}_{m_0}(s)\:,\:\:& \mathfrak{s}^{(1)}_{m_0} \le s \le \frac{n}{4}\:; 
\end{cases}
\end{equation*}
\item for dimensions $n=8,9,10,11,13$,
\begin{equation*}
\mathfrak d_{\psi,n}(s)  \ge \begin{cases}
\max\left\{\alpha^{(1)}_{m-1}(s)\:,\:\alpha^{(2)}_m(s)\right\}\:,\:\:& \mathfrak{s}^{(1)}_{m-1} \le s \le \mathfrak{s}^{(1)}_{m}\:,\:\: 1\le m \le m_0  \\
\max\left\{\alpha^{(1)}_{m_0}(s)\:,\:\alpha^{(3)}_{m_0+1}(s)\right\}\:,\:\:& \mathfrak{s}^{(1)}_{m_0} \le s \le \mathfrak{s}^{(2)}_{m_0+1}  \\
\alpha^{(1)}_{m_0+1}(s)\:,\:\:& \mathfrak{s}^{(2)}_{m_0+1} \le s \le \frac{n}{4}\:; 
\end{cases}
\end{equation*}
\item for dimensions $n\ge 12$, even,
\begin{equation*}
\mathfrak d_{\psi,n}(s)  \ge \begin{cases}
\max\left\{\alpha^{(1)}_{m-1}(s)\:,\:\alpha^{(2)}_m(s)\right\}\:,\:\:& \mathfrak{s}^{(1)}_{m-1} \le s \le \mathfrak{s}^{(1)}_{m}\:,\:\: 1\le m \le m_0  \\
\max\left\{\alpha^{(1)}_{m_0}(s)\:,\:\alpha^{(3)}_{m_0+1}(s)\right\}\:,\:\:& \mathfrak{s}^{(1)}_{m_0} \le s \le \mathfrak{s}^{(2)}_{m_0+1}  \\
\max\left\{\alpha^{(1)}_{m-1}(s)\:,\:\alpha^{(3)}_m(s)\right\}\:,\:\:& \mathfrak{s}^{(2)}_{m-1} \le s \le \mathfrak{s}^{(2)}_{m}\:,\:\: m_0+2\le m \le m_1\:; 
\end{cases}
\end{equation*}
\item for dimensions $n\ge 15$, odd,
\begin{equation*}
\mathfrak d_{\psi,n}(s)  \ge \begin{cases}
\max\left\{\alpha^{(1)}_{m-1}(s)\:,\:\alpha^{(2)}_m(s)\right\}\:,\:\:& \mathfrak{s}^{(1)}_{m-1} \le s \le \mathfrak{s}^{(1)}_{m}\:,\:\: 1\le m \le m_0  \\
\max\left\{\alpha^{(1)}_{m_0}(s)\:,\:\alpha^{(3)}_{m_0+1}(s)\right\}\:,\:\:& \mathfrak{s}^{(1)}_{m_0} \le s \le \mathfrak{s}^{(2)}_{m_0+1}  \\
\max\left\{\alpha^{(1)}_{m-1}(s)\:,\:\alpha^{(3)}_m(s)\right\}\:,\:\:& \mathfrak{s}^{(2)}_{m-1} \le s \le \mathfrak{s}^{(2)}_{m}\:,\:\: m_0+2\le m \le m_1\\
\alpha^{(1)}_{m_1}(s)\:,\:\:& \mathfrak{s}^{(2)}_{m_1} \le s \le \frac{n}{4}\:.
\end{cases}
\end{equation*}
\end{enumerate}
\end{enumerate} 
\end{enumerate}

<!-- corollary-5: body_md -->
\label{corollary4}
For a dispersive equation with asymptotically linear phase $\psi$ i.e. real-valued, radial function $\psi$ on $\mathbb R^n$ satisfying
\begin{equation*}
\psi(\|\xi\|)=\|\xi\| \:+\:\mathcal{O}(1)\:,\:\:\|\xi\| \gg 1\:,
\end{equation*}
we have,
\begin{enumerate}
\item for general $n \ge 1$, the sharp result for the following range:
\begin{equation*}
\mathfrak d_{\psi,n}(s) = \begin{cases}
n, \:\:\:&s \le \frac{1}{2}\\
0, \:\:\:&s>\frac{n}{2}\:;
\end{cases}
\end{equation*}
\item in the remaining range, 
\begin{enumerate}
\item for dimensions $n=2,3$, the sharp results:
\begin{equation*}
\mathfrak d_{\psi,2}(s) = \begin{cases}
4-4s, \:\:\:&\frac{1}{2} \le s <\frac{3}{4}\\
2-2s, \:\:\:&\frac{3}{4}< s \le 1\:;
\end{cases}
\text{ and } \mathfrak d_{\psi,3}(s) = \begin{cases}
5-4s, \:\:\:&\frac{1}{2} \le s \le 1\\
3-2s, \:\:\:&1 \le s \le \frac{3}{2}\:;
\end{cases}
\end{equation*}
\item for dimensions $n \ge 4$, 
\begin{enumerate}
\item  the sharp result in the range,
\begin{equation*}
\mathfrak d_{\psi,n}(s) = n-2s, \:\:\:\frac{n+1}{4} \le s \le \frac{n}{2}\:;
\end{equation*}
\item the upper bound,
\begin{equation*}
\mathfrak d_{\psi,n}(s) \le \begin{cases}
\frac{n^2-n-1}{n-2}-\frac{2(n-1)s}{n-2}, \:\:\:&\frac{1}{2} < s \le \frac{n}{4}\\
\frac{3n+1}{2}-4s, \:\:\:&\frac{n}{4} \le s \le \frac{n+1}{4}\:;
\end{cases}
\end{equation*}
\item the lower bound,
\begin{equation*}
\mathfrak d_{\psi,n}(s) \ge \begin{cases}
n+2-4s, \:\:\:&\frac{1}{2} < s \le 1\\
n-2s, \:\:\:&1 \le s \le \frac{n+1}{4}\:.
\end{cases}
\end{equation*}
\end{enumerate}
\end{enumerate}
\end{enumerate}

<!-- remark-10: body_md -->
\label{new_results_klein}
As special cases, Corollary \ref{corollary4} provides new results for the family of Klein-Gordon propagations (see Table \ref{table:1}), in particular, solving the classical Carleson's problem for Lebesgue measure. They are related to relativistic wave equations that describe how mass affects wave propagation in high-energy physics.
<!-- end remark-10 -->
<!-- proof-4: body_md -->
For the Wave propagation, by Walther's negative result \cite{Walther2} and the Sobolev embedding, we have for general $n \ge 1$,
\begin{equation*}
\mathfrak d_{\psi,n}(s) = \begin{cases}
n, \:\:\:&s\le\frac{1}{2}\\
0, \:\:\:&s>\frac{n}{2}\:.
\end{cases}
\end{equation*}
In dimension $n=2$, Barcel\'o et. al. \cite{BBCR} established the sharp result:
\begin{equation*}
\mathfrak d_{\psi,2}(s) = \begin{cases}
4-4s, \:\:\:&\frac{1}{2} \le s <\frac{3}{4}\\
2-2s, \:\:\:&\frac{3}{4}< s \le 1\:.
\end{cases}
\end{equation*}
In dimension $n=3$, Ham-Ko-Lee \cite{Ham} established the sharp result:
\begin{equation*}
\mathfrak d_{\psi,3}(s) = \begin{cases}
5-4s, \:\:\:&\frac{1}{2} \le s \le 1\\
3-2s, \:\:\:&1 \le s \le \frac{3}{2}\:.
\end{cases}
\end{equation*}
Finally, for dimension $n \ge 4$, the results stated in (2.b), in the statement of Corollary \ref{corollary4} are due to upper bounds in \cite{Ham} and lower bounds in \cite{BBCR}.   

The result for dispersive equations with asymptotically linear phase, in Corollary \ref{corollary4}, now follows by Theorem \ref{transference_theorem} and the above mentioned results for the Wave propagation.

\label{Annulus-Sobolev-embedding}
Let $s>0$ and $\alpha >n-2s$. Then for any $f \in H^s(\mathbb R^n)$, any $\mu \in \mathcal M^\alpha(\mathbb{A}(r_1,r_2))$, for $0<r_1<r_2<\infty$, we have
\begin{equation*}
\left\|u(\cdot,t)\right\|_{L^2(d\mu)} \lesssim \sqrt{c_\alpha(\mu)}\:\|f\|_{H^s(\mathbb R^n)}\:,
\end{equation*}
where the implicit constant depends only  on $r_1,r_2$\:.
<!-- end lemma-3 -->
<!-- remark-11: body_md -->
\label{BR_remark}
The index $n-2s$ showing up above, is actually the critical index for Hausdorff dimension of singular sets of radial Sobolev functions $f \in H^s(\mathbb R^n)$ for $s \in (0,1/2)$. In \cite{BR}, the authors constructed radial functions $f \in H^s(\mathbb R^n)$ for $s \in (0,1/2)$, having singularities on sets of Hausdorff dimension $<n-2s$. Consequently, our problem of estimating Hausdorff dimension of divergence sets becomes ill-posed for $\widetilde{\mathfrak d}_{\psi,n}(s)<n-2s$ in the range $s \in (0,1/2)$.

<!-- lemma-4: body_md -->
\label{lemma_s>1/2}
We have for all phases $\psi$ and $s>1/2$,
\begin{equation*}
\left\|\sup_{t \in \mathbb I} \sup_{N \ge 1} \left|S^N_{\psi,t}f\right|\right\|_{L^1(d\mu)} \lesssim \|f\|_{H^s(\mathbb R^n)}\:,
\end{equation*}
where $\mu \in \mathcal M^\alpha(\mathbb{A}(r_1,r_2))$, for $0<r_1<r_2<\infty,\:\alpha>0,\:f \in H^s(\mathbb R^n)$ radial and $S^N_{\psi,t}f$ is as in (\ref{N-scale}). The implicit constant in the above inequality depends only on $\mu,\:r_1,\:r_2$\:.
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
Let $G_s$ be the Bessel kernel on $\mathbb R^n$. Then writing $f=G_s*g$, it suffices to prove that 
\begin{equation} \label{lemma_pf_eq1}
\int_{\mathbb{A}(r_1,r_2)} \sup_{t \in \mathbb I} \sup_{N \ge 1} \left|\int_{\mathbb R^n} \frac{\Phi\left(N^{-1}\xi\right)\:e^{i\left(x\cdot \xi + it\psi(\|\xi\|)\right)}\:\widehat{g}(\xi)}{(1+\|\xi\|^2)^{s/2}}\:d\xi\right|\:d\mu(x) \lesssim \|g\|_{L^2(\mathbb R^n)}\:,\:\:\text{ if } s>\frac{1}{2}\:,
\end{equation} 
uniformly in $g \in L^2(\mathbb R^n)$ radial. As $\mu$ is a Radon measure on $\mathbb{A}(r_1,r_2)$, to prove (\ref{lemma_pf_eq1}), it is enough to prove the pointwise inequality,
\begin{equation} \label{lemma_pf_eq2}
\left|\int_{\mathbb R^n} \frac{\Phi\left(N^{-1}\xi\right)\:e^{i\left(x\cdot \xi + it\psi(\|\xi\|)\right)}\:\widehat{g}(\xi)}{(1+\|\xi\|^2)^{s/2}}\:d\xi\right| \lesssim \|g\|_{L^2(\mathbb R^n)}\:,\:\:\text{ if } s>\frac{1}{2}\:,
\end{equation}
uniformly in $N \ge 1,\:t \in \mathbb I,\:x \in \mathbb{A}(r_1,r_2)$ and $g \in L^2(\mathbb R^n)$ radial. Now using the radial symmetry of the phase $\psi$ and the function $g$, by polar decomposition, we have by writing $\lambda=\|\xi\|$,
\begin{equation*}
\left|\int_{\mathbb R^n} \frac{\Phi\left(N^{-1}\xi\right)\:e^{i\left(x\cdot \xi + it\psi(\|\xi\|)\right)}\:\widehat{g}(\xi)}{(1+\|\xi\|^2)^{s/2}}\:d\xi\right| = C \left|\int_{0}^\infty \frac{\Phi\left(N^{-1}\lambda\right)\lambda^{n-1}\:e^{it\psi(\lambda)}\:\:\widehat{g}(\lambda)}{(1+\lambda^2)^{s/2}}\frac{J_{\frac{n-2}{2}}(\lambda\|x\|)}{(\lambda\|x\|)^{\frac{n-2}{2}}}\:d\lambda\right|\:,
\end{equation*}
where $J_{(n-2)/2}$ denotes the Bessel function of the first kind of order $(n-2)/2$ and $C$ is some positive constant depending only on $n$. Then using the boundedness of Bessel functions of the first kind near $0$ and decay away from $0$ (see \cite{Szego}), we have the well-known estimate,
\begin{equation*}
\left|(\lambda\|x\|)^{\frac{1}{2}}\:J_{\frac{n-2}{2}}(\lambda\|x\|)\right| \lesssim 1\:,
\end{equation*}
with the implicit constant depending only on $n$. Thus by the Cauchy-Schwarz inequality and the Plancherel theorem, we get uniformly in $N \ge 1,\:t \in \mathbb I,\:x \in \mathbb{A}(r_1,r_2)$ and $g \in L^2(\mathbb R^n)$ radial,
\begin{eqnarray*}
&&\left|\int_{\mathbb R^n} \frac{\Phi\left(N^{-1}\xi\right)\:e^{i\left(x\cdot \xi + it\psi(\|\xi\|)\right)}\:\widehat{g}(\xi)}{(1+\|\xi\|^2)^{s/2}}\:d\xi\right| \\
&\lesssim & \frac{1}{\|x\|^{\frac{n-1}{2}}} \int_0^\infty \frac{\lambda^{\frac{n-1}{2}} \:|\widehat{g}(\lambda)|}{(1+\lambda^2)^{s/2}}\:d\lambda\\
& \lesssim & \frac{1}{\|x\|^{\frac{n-1}{2}}} \left(\int_0^\infty \lambda^{n-1}\:|\widehat{g}(\lambda)|^2\:d\lambda\right)^\frac{1}{2} \left(\int_0^\infty \frac{d\lambda}{(1+\lambda^2)^s}\right)^\frac{1}{2} \\
& \lesssim & \|g\|_{L^2(\mathbb R^n)}\:,
\end{eqnarray*}
as $x \in \mathbb{A}(r_1,r_2)$ and $s>1/2$. Hence we obtain (\ref{lemma_pf_eq2}) and this completes the proof of Lemma \ref{lemma_s>1/2}.

<!-- lemma-5: body_md -->
\label{CS_lemma}
Let $\alpha \in (0,1]$ and $\nu$ be an $\alpha$-dimensional measure on $\mathbb R$. There exists a constant $C>0$, such that for any interval $[c,d] \subset \mathbb R$, we have
\begin{equation*}
\left|\int_{\mathbb R} \int_{\mathbb R} g_1(x)\:g_2(x')\:\chi_{[c,d]}(x-x')\:d\nu(x)\:d\nu(x')\right| \le C(d-c)^{\alpha}\:\|g_1\|_{L^2(d\nu)}\:\|g_2\|_{L^2(d\nu)}\:.
\end{equation*}
Moreover, if $\rho \in (0,\alpha)$, then  there exists a constant $C>0$ such that
\begin{equation*}
\left|\int_{\mathbb R} \int_{\mathbb R} g_1(x)\:g_2(x')\:|x-x'|^{-\rho}\:d\nu(x)\:d\nu(x')\right| \le C\:\|g_1\|_{L^2(d\nu)}\:\|g_2\|_{L^2(d\nu)}\:.
\end{equation*}
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
We first prove part (1). The result for the range $s \in (0,a/4)$ follows from Walther's work in \cite{Walther3}. We now focus on the range $s \in (a/4,1/2)$ and prove the maximal estimate,
\begin{equation} \label{concave_pf_eq1}
\|S^*_\psi f\|_{L^1(d\mu)} \lesssim \|f\|_{H^s(\mathbb R^n)}\:,
\end{equation} 
for $\mu \in \mathcal M^\alpha(\mathbb{A}(r_1,r_2))\:,\: 0<r_1<r_2<\infty$, all $f \in \mathcal S(\mathbb R^n)$ radial and 
\begin{equation*}
\alpha > \max \left\{n-\frac{1}{2}\:+\:\frac{1-4s}{2(1-a)}\:,\: n-2s\right\}\:,
\end{equation*}
or equivalently for 
\begin{equation*}
s>\max\left\{\frac{1}{2}-\frac{a}{4}-\frac{(1-a)(\alpha-n+1)}{2}\:,\: \frac{1}{2}-\frac{(\alpha-n+1)}{2}\right\}\:,
\end{equation*}
where $\alpha \in (n-1,n)$\:. Then by considering the Bessel kernel $G_s$, it suffices to prove that
\begin{equation} \label{concave_pf_eq2}
\|S^*_\psi (G_s *f)\|_{L^1(d\mu)} \lesssim \|f\|_{L^2(\mathbb R^n)}\:,
\end{equation}
for $\mu \in \mathcal M^\alpha(\mathbb{A}(r_1,r_2))\:,\: 0<r_1<r_2<\infty$, all $f \in \mathcal S(\mathbb R^n)$ radial. Denoting $\|x\|$ by $r$, using the radial symmetry of the phase $\psi$ and the function $f$, by polar decomposition, we have 
\begin{eqnarray*}
S_{\psi,t} \left(G_s*f\right)(x)&=&S_{\psi,t} \left(G_s*f\right)(r) \\
&=& \frac{C}{r^{\frac{n-1}{2}}} \int_{0}^\infty \widehat{f}(\lambda) \:e^{it\lambda^a} (\lambda r)^{\frac{1}{2}} J_{\frac{n-2}{2}}(\lambda r)\frac{\lambda^{\frac{n-1}{2}} }{(1+\lambda^2)^{s/2}}\:d\lambda\:.
\end{eqnarray*} 
Thus it suffices to work with radial measures. We consider, $d\mu_0(r):=d\mu(x)$, the one-dimensional pushforward of $\mu$. Now as for each interval $I_v$ of length $v$, we can find an $n$-ball $B_{nv}$ of radius $nv$ so that,
\begin{equation*}
\mu_0\left(I_v\right)\:v^{n-1} \lesssim \mu\left(B_{nv}\right)\:,
\end{equation*}
it follows that $\mu_0 \in \mathcal M^{\alpha-(n-1)}(r_1,r_2)$\:. Hence to prove (\ref{concave_pf_eq2}), it suffices to prove the following estimate:
\begin{equation*} 
\int_{r_1}^{r_2} \sup_{t \in \mathbb I} \left|S_{\psi,t} \left(G_s*f\right)(r)\right|\:r^{\frac{n-1}{2}}\:d\mu_0(r) \lesssim \|f\|_{L^2(\mathbb R^n)}\:,
\end{equation*}
which in turn by linearization, is equivalent to prove 
\begin{equation} \label{concave_pf_eq3}
\int_{r_1}^{r_2} \left|S_{\psi,t(r)} \left(G_s*f\right)(r)\right|\:r^{\frac{n-1}{2}}\:d\mu_0(r) \lesssim \|f\|_{L^2(\mathbb R^n)}\:,
\end{equation}
for $\mu_0 \in \mathcal M^{\alpha_0}(r_1,r_2)\:,\: 0<r_1<r_2<\infty$, all $f \in \mathcal S(\mathbb R^n)$ radial and 
\begin{equation*}
s>\max\left\{\frac{1}{2}-\frac{a}{4}-\frac{(1-a)\alpha_0}{2}\:,\: \frac{1}{2}-\frac{\alpha_0}{2}\right\}\:,
\end{equation*}
where $\alpha_0 \in (0,1)$ and $t : (r_1,r_2) \to \mathbb I$ is a measurable function. Finally, by duality, to prove (\ref{concave_pf_eq3}), it suffices to prove that
\begin{equation} \label{concave_pf_eq4}
\|Tg\|_{L^2(0,\infty)} \lesssim \|g\|_{L^\infty(d\mu_0)}\:,
\end{equation}
for $g \in L^\infty(d\mu_0)$, where 
\begin{equation*}
Tg(\lambda):= \frac{1}{(1+\lambda^2)^{s/2}} \int_{r_1}^{r_2} (\lambda r)^{\frac{1}{2}} J_{\frac{n-2}{2}}(\lambda r)\:e^{it(r)\lambda^a}\:g(r)\:d\mu_0(r)\:.
\end{equation*}
Now invoking the facts (see \cite{Szego}) that there are complex numbers $a_1,a_2$ and some constant $C>0$, such that 
\begin{eqnarray*}
\left|y^{\frac{1}{2}}\:J_{\frac{n-2}{2}}(y) -\left(a_1e^{iy}+a_2e^{-iy}\right)\right| &\le& Cy^{-1}\:,\:y>1\:,\\
\left|y^{\frac{1}{2}}\:J_{\frac{n-2}{2}}(y) -\left(a_1e^{iy}+a_2e^{-iy}\right)\right| &\le& C\:,\:0<y \le 1\:,
\end{eqnarray*}
we decompose,
\begin{equation*}
Tg(\lambda)=a_1T_1g(\lambda)\:+\:a_2T_1g(-\lambda)\:+\:T_2g(\lambda)\:,
\end{equation*}
where
\begin{equation*}
T_1g(\lambda) := \frac{1}{(1+\lambda^2)^{s/2}} \int_{r_1}^{r_2} e^{i\left(\lambda r+t(r)|\lambda|^a\right)}\:g(r)\:d\mu_0(r)\:,
\end{equation*}
and $T_2g$ is the part corresponding to the remainder term. By \cite[pp. 7-8]{BR}, we have
\begin{equation} \label{concave_pf_eq5}
\|T_2g\|_{L^2(0,\infty)} \lesssim \|g\|_{L^\infty(d\mu_0)}\:.
\end{equation}
We now focus on $T_1$. We set,
\begin{equation*}
Ug(\lambda):=\frac{1}{(1+\lambda^2)^{s/2}} \int_{r_1}^{r_2} e^{i\left(\lambda r+t(r)|\lambda|^a\right)}\:g(r)\:d\mu_0(r)\:,\:\:\lambda \in \mathbb R\:,
\end{equation*}
and shall prove that
\begin{equation} \label{concave_pf_eq6}
\|Ug\|_{L^2(\mathbb R)} \lesssim \|g\|_{L^\infty(d\mu_0)}\:,
\end{equation}
from which the required estimates for $T_1$ would follow.

We now consider an auxiliary non-negative, even function $\eta \in C^\infty_c(\mathbb R)$ such that 
\begin{equation*}
Supp(\eta) \subset \left\{\lambda \in \mathbb R : \frac{1}{2} < |\lambda| < 2\right\}\:\:\text{ and } \sum_{k=-\infty}^\infty \eta\left(2^{-k}\lambda\right)=1\:,\:\text{ for } \lambda \ne 0\:.
\end{equation*}
Then setting,
\begin{equation*}
\eta_1(\lambda):=\sum_{k=-\infty}^0 \eta\left(2^{-k}\lambda\right)\:,\:\:\text{ and } \eta_2(\lambda):=\sum_{k=1}^\infty \eta\left(2^{-k}\lambda\right)\:,
\end{equation*}
we conider the decomposition,
\begin{equation*}
Ug(\lambda)= U_1g(\lambda)\:+\:U_2g(\lambda)\:,\:\lambda \in \mathbb R,
\end{equation*}
where for $j=1,2$,
\begin{eqnarray*}
U_jg(\lambda):=\frac{\eta_j(\lambda)}{(1+\lambda^2)^{s/2}} \int_{r_1}^{r_2} e^{i\left(\lambda r+t(r)|\lambda|^a\right)}\:g(r)\:d\mu_0(r)\:,\:\:\lambda \in \mathbb R\:.
\end{eqnarray*}

We trivially have that
\begin{equation*}
\|U_1g\|_{L^\infty(\mathbb R)} \lesssim \|g\|_{L^\infty(d\mu_0)}\:.
\end{equation*}
Now as $Supp(U_1g) \subset (-2,2)$, the above estimate yields 
\begin{equation}\label{concave_pf_eq7}
\|U_1g\|_{L^2(\mathbb R)} \lesssim \|g\|_{L^\infty(d\mu_0)}\:.
\end{equation}
Next, to estimate $U_2g$, we start by noting that
\begin{equation*}
\|U_2g\|^2_{L^2(\mathbb R)} \lesssim \sum_{k=1}^\infty \|\eta(2^{-k}\cdot)\:Ug\|^2_{L^2(\mathbb R)}\:,
\end{equation*}
and then by Fubini's theorem, for each $k \ge 1$,
\begin{eqnarray} \label{sum_estimate}
&& \|\eta(2^{-k}\cdot)\:Ug\|^2_{L^2(\mathbb R)} \nonumber\\
&= &\int_{\mathbb R}\frac{\eta\left(2^{-k}\lambda\right)^2}{(1+\lambda^2)^s} \int_{r_1}^{r_2}e^{i\left(\lambda r + t(r) |\lambda|^a\right)}\:g(r)\:d\mu_0(r) \int_{r_1}^{r_2}e^{-i\left(\lambda r' + t(r') |\lambda|^a\right)}\:\overline{g(r')}\:d\mu_0(r')\:d\lambda \nonumber\\
\:\:\:\:\:\:\:&\lesssim & \left(2^k\right)^{1-2s} \int_{r_1}^{r_2} \int_{r_1}^{r_2} |g(r)|\:|g(r')|\:|I_{k}(r,r')|\:dr\:dr'\:, 
\end{eqnarray}
where
\begin{equation*}
I_k(r,r'):= \int_\mathbb R \eta(\lambda)^2\:e^{i\left(2^k\lambda(r-r')\:+\:2^{ka}|\lambda|^a(t(r)-t(r'))\right)}\:d\lambda\:.
\end{equation*}
For notational convenience, setting $y:=r-r'$ and $t:=t(r)-t(r')$, we estimate the above oscillatory integral by considering the cases:
\begin{equation*}
|y| \le 2^{-\frac{2ks_*}{\alpha_0}}\:,\:\:\text{ or } |y| > 2^{-\frac{2ks_*}{\alpha_0}}\:,
\end{equation*} 
where 
\begin{equation*}
s_*:=\min\left\{\frac{a}{4}+\frac{(1-a)\alpha_0}{2}\:,\: \frac{\alpha_0}{2}\right\}\:,
\end{equation*}
that is, $s=1/2-s_*+\varepsilon$, for any $\varepsilon>0$. In the former case, it follows trivially that
\begin{equation}\label{concave_pf_eq8}
\left|\int_\mathbb R \eta(\lambda)^2\:e^{i\left(2^k\lambda y\:+\:2^{ka}|\lambda|^a t)\right)}\:d\lambda\right| \lesssim 1\:.
\end{equation}
We now focus when $|y| > 2^{-\frac{2ks_*}{\alpha_0}}$. In particular, we note that $2^k|y| \ge 1$. For notaional converience, writing, $$\phi(\lambda):= 2^k\lambda y\:+\:2^{ka}|\lambda|^a t\:,$$ we decompose the integral as follows,
\begin{equation*}
\int_{\mathbb R} e^{i\phi(\lambda)}\:\eta(\lambda)^2\:d\lambda = \int_{\Omega_1} e^{i\phi(\lambda)}\:\eta(\lambda)^2\:d\lambda \:+\: \int_{\Omega_2} e^{i\phi(\lambda)}\:\eta(\lambda)^2\:d\lambda=: I^{(1)}_k\:+\:I^{(2)}_k\:,
\end{equation*}
where
\begin{eqnarray*}
\Omega_1&:=& \left\{\frac{1}{2} < |\lambda| <2 : 2^{1+ka} |t||\lambda|^{a-1} \ge 2^{4ks_*}|y|^{2\alpha_0}\right\}\:,\\
\Omega_2&:=& \left\{\frac{1}{2} < |\lambda| <2 : 2^{1+ka} |t||\lambda|^{a-1} < 2^{4ks_*}|y|^{2\alpha_0}\right\}\:.
\end{eqnarray*}
We first focus on $I^{(1)}_k$. In this case, since 
\begin{equation*}
\left|\frac{d^2}{d\lambda^2}\phi(\lambda)\right| \gtrsim 2^{4ks_*}|y|^{2\alpha_0}\:,
\end{equation*}
by van der Corput's lemma, we have
\begin{equation}\label{concave_pf_eq9}
\left|I^{(1)}_k\right| \lesssim \left(2^{4ks_*}|y|^{2\alpha_0}\right)^{-\frac{1}{2}} = \left(2^k\right)^{-2s_*}\:|y|^{-\alpha_0}\:.
\end{equation} 
For $I^{(2)}_k$, by the defining condition of $\Omega_2$ and the fact that $t \in \mathbb I$, it follows that
\begin{equation*}
2^{ka}\:|t|^{\max\left\{1,2\alpha_0\right\}}\:|\lambda|^{a-1} \le 2^{4ks_*-1}\:|y|^{2\alpha_0}\:.
\end{equation*} 
We also note that
\begin{equation*}
2^{ka}\left(\left(2^k\right)^{4s_*-a}\:|y|^{2\alpha_0}\right)^{\left(\max\left\{1,2\alpha_0\right\}\right)^{-1}} = \left(2^k|y|\right)^{\min\left\{2\alpha_0,1\right\}} \le 2^k|y|\:.
\end{equation*}
Hence, it follows that
\begin{eqnarray*}
\left|\frac{d}{d\lambda}\phi(\lambda)\right| \gtrsim 2^k|y|-a2^{ka}|t||\lambda|^{a-1} &\gtrsim & 2^k|y|-2^{ka}\left(\left(2^k\right)^{4s_*-a}\:|y|^{2\alpha_0}\right)^{\left(\max\left\{1,2\alpha_0\right\}\right)^{-1}} \\
&\gtrsim&  2^k|y|\:,
\end{eqnarray*}
which by van der Corput's lemma yields,
\begin{equation}\label{concave_pf_eq10}
\left|I^{(2)}_k\right| \lesssim \left(2^{k}|y|\right)^{-1} \le \left(2^{k}|y|\right)^{-\alpha_0} \le \left(2^k\right)^{-2s_*}\:|y|^{-\alpha_0}\:.
\end{equation} 
Thus by (\ref{concave_pf_eq8})-(\ref{concave_pf_eq10}), we have
\begin{equation*}
\left|I_k(r,r')\right| \lesssim \chi_{\left\{|r-r'| \le 2^{-\frac{2ks_*}{\alpha_0}}\right\}}(r-r')\:+\: \left(2^k\right)^{-2s_*}\:|r-r'|^{-\alpha_0}\chi_{\left\{|r-r'| > 2^{-\frac{2ks_*}{\alpha_0}}\right\}}(r-r')\:.
\end{equation*}
Thus by Lemma \ref{CS_lemma}, it follows that
\begin{equation*}
\int_{r_1}^{r_2} \int_{r_1}^{r_2} |g(r)|\:|g(r')|\:|I_{k}(r,r')|\:dr\:dr' \lesssim \left(2^k\right)^{-2s_*}\:\|g\|^2_{L^2(d\mu_0)}\:,
\end{equation*}
which upon plugging in (\ref{sum_estimate}) yields for $k \ge 1$\:,
\begin{equation} \label{concave_pf_eq11}
\|\eta(2^{-k}\cdot)\:Ug\|^2_{L^2(\mathbb R)} \lesssim \left(2^k\right)^{1-2(s+s_*)}\|g\|^2_{L^2(d\mu_0)} = 2^{-2k\varepsilon}\|g\|^2_{L^2(d\mu_0)} \:.
\end{equation}
Therefore by (\ref{concave_pf_eq7}) and (\ref{concave_pf_eq11}), we get (\ref{concave_pf_eq6}) as,
\begin{eqnarray*}
\|Ug\|_{L^2(\mathbb R)} &\le & \|U_1g\|_{L^2(\mathbb R)}\:+\:\|U_2g\|_{L^2(\mathbb R)} \\ &\lesssim & \|g\|_{L^\infty(d\mu_0)}\:+\:\left(\sum_{k=1}^\infty 2^{-2k\varepsilon}\right)^\frac{1}{2}  \|g\|_{L^2(d\mu_0)} \\
&\lesssim & \|g\|_{L^\infty(d\mu_0)}\:,
\end{eqnarray*}
which along with (\ref{concave_pf_eq5}) yields the desired estimate (\ref{concave_pf_eq4}). Therefore we have proved (\ref{concave_pf_eq1}). 

Then by the method of maximal functions, density of $\mathcal S(\mathbb R^n)$, the fractal Sobolev embedding for the initial data (Lemma \ref{Annulus-Sobolev-embedding}) and Frostmann's lemma, for $a \in (0,1)$, all radial $f \in H^s(\mathbb R^n),\:s \in (a/4,1/2)$ and 
\begin{equation} \label{alpha_range}
\alpha > \max \left\{n-\frac{1}{2}\:+\:\frac{1-4s}{2(1-a)}\:,\: n-2s\right\}\:,
\end{equation}
we have
\begin{equation*}
\mathcal{H}^\alpha \left\{x \in \mathbb{A}(r_1,r_2) : \displaystyle\lim_{t \to 0} e^{it(-\Delta)^{a/2}}f(x) \ne f(x)\right\}=0\:.
\end{equation*}
Then by countable additivity of Hausdorff measures, we get that
\begin{equation*}
\mathcal{H}^\alpha \left\{x \in \mathbb R^n : \displaystyle\lim_{t \to 0} e^{it(-\Delta)^{a/2}}f(x) \ne f(x)\right\}=0\:,
\end{equation*}
for $\alpha$ satisfying (\ref{alpha_range}). Therefore, we get 
\begin{equation*}
\widetilde{\mathfrak d}_{\psi,n}(s) \le \max \left\{n-\frac{1}{2}\:+\:\frac{1-4s}{2(1-a)}\:,\: n-2s\right\}\:,\:\:s \in (a/4,1/2)\:,
\end{equation*}
which is also sharp in the range $s \in [1/4,1/2)$ by Remark \ref{BR_remark}. Remark \ref{BR_remark} also yields the lower bound $\widetilde{\mathfrak d}_{\psi,n}(s) \ge n-2s$ for the range $s \in [a/4,1/4)$.

For the range $s>1/2$, the result follows by the above arguments (of the last paragraph) combined with Lemma \ref{lemma_s>1/2}. This completes the proof of part (1) of Theorem \ref{radial_concave_thm}.

\medskip

For part (2) of Theorem \ref{radial_concave_thm}, the result in the range $s \in (0,1/2]$ follows from Walther's work in \cite{Walther2}, whereas for the range $s>1/2$, the result follows by the above arguments combined with Lemma \ref{lemma_s>1/2}. This completes the proof of Theorem \ref{radial_concave_thm}.

and the lower bound,
\begin{equation*}

<!-- lemma-6: body_md -->
\label{fractal_measure_correspondence_lemma}
On a non-positively curved Hadamard manifold, for any $0<r_1<r_2<\infty$ and $\alpha \ge 0$, the Riemannian exponential map $\exp_o$ induces an one-to-one correspondence between $\mathcal M^\alpha\left(\mathbb{A}(r_1,r_2)\right)$ and $\mathcal M^\alpha\left(A(r_1,r_2)\right)$.
<!-- end lemma-6 -->
<!-- proof-11: body_md -->
We start off by making the canonical identification $T_o\mathbb X \cong \mathbb R^n$, where $n$ is the dimension of $\mathbb X$. Due to the underlying geometry of $\mathbb X$,  $\exp_o$ is a diffeomorphism from $\mathbb R^n$ onto $\mathbb X$ and is also a radial isometry. Hence $\exp_o\left(\mathbb{A}(r_1,r_2)\right)=A(r_1,r_2)$ and moreover, it suffices to prove that for any $\mu \in \mathcal M^\alpha\left(\mathbb{A}(r_1,r_2)\right)$, its pushforward,
\begin{equation*}
(\left(\exp_o\right)_*\mu)(B):=\mu\left(\exp^{-1}_o(B)\right)\:,\:\:B \in \mathscr{B}(\mathbb X)\:,
\end{equation*}
is a member of $\mathcal M^\alpha\left(A(r_1,r_2)\right)$. Let $\mu \in \mathcal M^\alpha\left(\mathbb{A}(r_1,r_2)\right)$. Then there exists $C_1>0$ such that 
\begin{equation} \label{euclidean_inequality}
\mu\left(\mathbb{B}(x,r)\right) \le C_1r^\alpha\:,\:\:\text{ for all } x \in \mathbb R^n\:,\:r>0\:.
\end{equation}

It is clear that $\left(\exp_o\right)_*\mu$ is a positive Radon measure on $\mathbb X$ with support contained in $A(r_1,r_2)$. We now consider $(\left(\exp_o\right)_*\mu)(B(x,r))$ for geodesic balls $B(x,r) \subset \mathbb X$. For $r \ge 1$, we trivially have,
\begin{equation} \label{fractal_correspondence_eq1}
(\left(\exp_o\right)_*\mu)(B(x,r))=\mu\left(\exp^{-1}_o(B(x,r))\right) \le \mu\left(\mathbb{A}(r_1,r_2)\right) \le \mu\left(\mathbb{A}(r_1,r_2)\right) r^\alpha\:. 
\end{equation}
We next consider the case $r \in (0,1)$ and pay attention to the location of the center $x$. If $d(o,x) \ge r_2+1$, then $B(x,r) \cap  A(r_1,r_2)=\emptyset$ and hence,
\begin{equation} \label{fractal_correspondence_eq2}
(\left(\exp_o\right)_*\mu)(B(x,r))=0 \le r^\alpha\:.
\end{equation}
Finally if $d(o,x)<r_2+1$, then we note that $B(x,r) \subset B(o,r_2+2)$. Since $\exp_o$ is diffeomorphism from $\mathbb R^n$ onto $\mathbb X$, it is bi-Lipschitz from $\mathbb{B}(0,r_2+2)$ onto $B(o,r_2+2)$, that is, there exists a constant $C_2>0$ such that 
\begin{equation*}
\frac{1}{C_2} \|a-b\| \le d(\exp_o(a),\exp_o(b)) \le C_2 \|a-b\|\:,\:\:a,b\in \mathbb{B}(0,r_2+2)\:.
\end{equation*}
Thus $$\exp^{-1}_o\left(B(x,r)\right) \subset B\left(\exp^{-1}_o(x),C_2r\right)\:,$$
which in turn by (\ref{euclidean_inequality}) yields
\begin{equation} \label{fractal_correspondence_eq3}
(\left(\exp_o\right)_*\mu)(B(x,r)) \le \mu\left(B\left(\exp^{-1}_o(x),C_2r\right)\right) \le (C_1 C^\alpha_2)\:r^\alpha\:.
\end{equation}
Then by (\ref{fractal_correspondence_eq1})-(\ref{fractal_correspondence_eq3}), the result follows by considering $\max\left\{\mu\left(\mathbb{A}(r_1,r_2)\right),\:1,\:C_1 C^\alpha_2\right\}$.

<!-- lemma-7: body_md -->
\label{schwartz_correspondence_lemma}
Given any radial $f \in \mathcal{S}^2(\mathbb H^3)$ there exists a unique radial $F \in \mathcal{S}(\mathbb R^3)$ (and vice-versa) such that 
\begin{itemize}
\item $\mathscr{F}f=\widehat{F}$ on $[0,\infty)$ and 
\item $\|f\|_{H^s(\mathbb H^3)}\asymp\|F\|_{H^s(\mathbb R^3)}$, for all $s \ge 0$\:.
\end{itemize}
<!-- end lemma-7 -->
<!-- proof-12: body_md -->
By the well-known properties of the Abel transform on Ch\'ebli-Trim\`eche Hypergroups \cite{BX} (for symmetric spaces see \cite{AnkerJFA} and also \cite[Lemma 2.3]{Helgason-b}), we have the following commutative diagram, where
\begin{itemize}
\item $\mathscr{A}_1$ is the Abel transform defined from $\mathcal{S}^2(\mathbb H^3)_{radial}$ onto the space of even Schwartz class functions on $\mathbb R$, ${\mathcal{S}(\mathbb R)}_{even}$\:.
\item $\mathscr{A}_2$ is the Abel transform defined from the space of radial Schwartz class functions on $\mathbb R^3$, $\mathcal{S}(\mathbb R^3)_{radial}$ onto ${\mathcal{S}(\mathbb R)}_{even}$\:.
\item $\mathscr{F}$ denotes the Spherical Fourier transform from $\mathcal{S}^2(\mathbb H^3)_{radial}$ onto ${\mathcal{S}(\mathbb R)}_{even}$\:.
\item $\wedge$ denotes the Euclidean Fourier transform from $\mathcal{S}(\mathbb R^3)_{radial}$ onto ${\mathcal{S}(\mathbb R)}_{even}$\:.
\item $\sim$ denotes the 1-dimensional Euclidean Fourier transform from ${\mathcal{S}(\mathbb R)}_{even}$ onto itself.
\end{itemize}

\[
\begin{tikzcd}[row sep=1.4cm,column sep=1.4cm]
\mathcal{S}^2(\mathbb H^3)_{radial} \arrow[r,"\mathscr{A}_1"] \arrow[dr,swap,"\mathscr{F}"] & {\mathcal{S}(\mathbb R)}_{even}
\arrow[d,"\sim"] &
\arrow[l,"\mathscr{A}_2",swap]  \mathcal{S}(\mathbb R^3)_{radial} \arrow[dl,"\wedge"] \\
&  {\mathcal{S}(\mathbb R)}_{even}&
\end{tikzcd}
\]

Then, since all the maps above are topological isomorphisms, defining $\mathscr{A}:=\mathscr{A}^{-1}_2 \circ \mathscr{A}_1$, one can reduce matters to the following simplified commutative diagram:

\[
\begin{tikzcd}[row sep=1.4cm,column sep=1.4cm]
\mathcal{S}^2(\mathbb H^3)_{radial}\arrow[r,"\mathscr{A}"] \arrow[dr,swap,"\mathscr{F}"] & \mathcal{S}(\mathbb R^3)_{radial}
\arrow[d,"\wedge"]   \\
&  {\mathcal{S}(\mathbb R)}_{even}&
\end{tikzcd}
\]

Hence, for $f \in \mathcal{S}^2(\mathbb H^3)_{radial}$, we consider $F:= \mathscr{A}f \in \mathcal{S}(\mathbb R^3)_{radial}$ to observe that  
\begin{equation*}
\mathscr{F}f(\lambda)= \widehat{F}(\lambda)\:,\:\:\lambda \ge 0\:.
\end{equation*}
Moreover, for any $s \ge 0$, 
\begin{eqnarray*} 
{\|f\|}_{H^s(\mathbb H^3)} &=& {\left(\int_0^\infty {(\lambda^2 + 1)}^s \:{\left|\mathscr{F}f(\lambda)\right|}^2 \: \lambda^2 \: d\lambda\right)}^{1/2} \\
&=& {\left(\int_0^\infty {(\lambda^2 + 1)}^s \:{\left|\widehat{F}(\lambda)\right|}^2 \: \lambda^2 \: d\lambda\right)}^{1/2} \nonumber\\
&\asymp&  {\|F\|}_{H^s(\mathbb R^3)}\:.
\end{eqnarray*}
This completes the proof of Lemma \ref{schwartz_correspondence_lemma}.

<!-- lemma-8: body_md -->
\label{H3-N-maximal-estimate-radial}
Let $s>0$ and $\alpha >3-2s$. Then for any radial $f \in H^s(\mathbb H^3)$ and any radial $\mu \in \mathcal M^\alpha(A(r_1,r_2))$, for $0<r_1<r_2<\infty$, we have
\begin{equation*}
\left\|\sup_{N \ge 1} \left|S^N_{\psi,t}f\right|\right\|_{L^2(d\mu)} \lesssim \|f\|_{H^s(\mathbb H^3)}\:,
\end{equation*}
where the implicit constant depends only on $\alpha, \mu, r_1,r_2$\:.
<!-- end lemma-8 -->
<!-- proof-13: body_md -->
We fix $s>0,\:\alpha >3-2s$ and radial $\mu \in \mathcal M^\alpha(A(r_1,r_2))$ with its one-dimensional pushforward denoted by $\mu_0$. By density, it suffices to prove the inequality for all radial functions in $\mathcal{S}^2(\mathbb H^3)$. Let $f \in \mathcal{S}^2(\mathbb H^3)_{radial}$. By Lemma \ref{schwartz_correspondence_lemma}, there exists $F \in \mathcal{S}(\mathbb R^3)_{radial}$ such that 
\begin{equation*}
\mathscr{F}f=\widehat{F}\:,\:\:\text{ on } [0,\infty) \text{ and } 
\|f\|_{H^s(\mathbb H^3)}\asymp\|F\|_{H^s(\mathbb R^3)}\:\:.
\end{equation*}
Thus for all $r \in (r_1,r_2)$ and all $t \in \mathbb R$,
\begin{eqnarray} \label{H3-N-eq1}
S^N_{\psi,t}f(r)&=& \int_0^\infty \Phi\left(N^{-1}\lambda\right)\: \varphi_\lambda(r)\:e^{it\psi(\sqrt{\lambda^2+1})}\:\mathscr{F}f(\lambda)\:\lambda^2\:d\lambda \nonumber\\
&=& \int_0^\infty \Phi\left(N^{-1}\lambda\right)\:\frac{\sin(\lambda r)}{\lambda \sinh(r)} \:e^{it\psi(\sqrt{\lambda^2+1})}\:\mathscr{F}f(\lambda)\:\lambda^2\:d\lambda \nonumber\\
&=& \left(\frac{ r}{\sinh(r)}\right) \int_0^\infty \Phi\left(N^{-1}\lambda\right)\: \frac{\sin(\lambda r)}{\lambda r} \: e^{it\widetilde{\psi}(\lambda)} \: \widehat{F}(\lambda) \: \lambda^2 \: d\lambda \nonumber\\
&=& \left(\frac{ r}{\sinh(r)}\right) \int_0^\infty \Phi\left(N^{-1}\lambda\right)\left( \frac{J_{\frac{1}{2}}(\lambda r)}{{(\lambda r)}^{\frac{1}{2}}} \right) e^{it\widetilde{\psi}(\lambda)} \: \widehat{F}(\lambda) \: \lambda^2 \: d\lambda \nonumber\\
&=& C\left(\frac{ r}{\sinh(r)}\right) S^N_{\widetilde{\psi},t}F(r)\:,
\end{eqnarray}
for some $C>0$, where $S^N_{\widetilde{\psi},t}F$ is the Euclidean counter-part given by (\ref{N-scale}), corresponding to the dispersive equation:
\begin{equation*} 
\begin{cases}
	 i\frac{\partial u}{\partial t} +\widetilde{\psi}(\sqrt{-\Delta} )u=0\:,\:  &(x,t) \in \mathcal \mathbb R^3 \times \mathbb R\:, \\
	u(\cdot,0)=F\:,\: &\text{ on } \mathcal \mathbb R^3 \:,
	\end{cases}
\end{equation*}
where $\widetilde{\psi}(\lambda):=\psi(\sqrt{\lambda^2+1})$. Thus by (\ref{H3-N-eq1}), we have for all $r \in (r_1,r_2)$ and all $t \in \mathbb R$, the pointwise comparability,
\begin{equation} \label{H3-N-eq2}
\sup_{N \ge 1} \left|S^N_{\psi,t}f(r)\right| \asymp \sup_{N \ge 1} \left|S^N_{\widetilde{\psi},t}F(r)\right|\:.
\end{equation}
Now by Lemma \ref{fractal_measure_correspondence_lemma}, identifying $T_o\mathbb H^3 \cong \mathbb R^3$, there exists radial $\nu \in \mathcal M^\alpha(\mathbb{A}(r_1,r_2))$ (with its one-dimensional pushforward $\nu_0$) such that $\mu=(\exp_o)_*\nu$. We note that the radial symmetry is preserved as $\exp_o$ is a radial isometry. Combining this with (\ref{H3-N-eq2}), we have,
\begin{eqnarray*}
\left\|\sup_{N \ge 1} \left|S^N_{\psi,t}f\right|\right\|_{L^2(d\mu)} &=& \left(\int_{r_1}^{r_2}\left(\sup_{N \ge 1} \left|S^N_{\psi,t}f(r)\right|\right)^2\:d\mu_0(r)\right)^\frac{1}{2} \\
&\asymp & \left(\int_{r_1}^{r_2}\left(\sup_{N \ge 1} \left|S^N_{\widetilde{\psi},t}F(r)\right|\right)^2\:d\nu_0(r)\right)^\frac{1}{2} \\
&=& \left\|\sup_{N \ge 1} \left|S^N_{\widetilde{\psi},t}F\right|\right\|_{L^2(d\nu)}\:,
\end{eqnarray*}
which along with Lemma \ref{N-maximal-estimate-radial} yields
\begin{eqnarray*}
\left\|\sup_{N \ge 1} \left|S^N_{\psi,t}f\right|\right\|_{L^2(d\mu)}  \lesssim  \left\|\sup_{N \ge 1} \left|S^N_{\widetilde{\psi},t}F\right|\right\|_{L^2(d\nu)} \lesssim \|F\|_{H^s(\mathbb R^3)} \lesssim \|f\|_{H^s(\mathbb H^3)}\:.
\end{eqnarray*}
Since $f$ was arbitrary, this completes the proof of Lemma \ref{H3-N-maximal-estimate-radial}.

<!-- lemma-11: body_md -->
\label{lemma_bessel_kernel}
The Bessel kernel of order $s>0$ on $\mathbb H^3$ is radial and has the explicit form:
\begin{equation*}
\mathcal{G}_s(r):= C\:\left(\frac{r}{\sinh r}\right)\:r^{\frac{s-3}{2}}\:K_{\frac{3-s}{2}}(r)\:,
\end{equation*}
where $K_\nu$ is the modified Bessel function of the second kind of order $\nu$ and $C$ is a positive constant depending only on $s$. Consequently, the Bessel kernel satisfies the following asymptotics,
\begin{equation*}
\mathcal{G}_s(r) \asymp \begin{cases}
r^{s-3}\:,\:\:&r\in(0,1)\:,\\
r^{\frac{s-2}{2}}\:e^{-2r}\:,\:\:&r \ge 1\:.
\end{cases}
\end{equation*}
<!-- end lemma-11 -->
<!-- proof-15: body_md -->
By the heat semi-group representation of $(-\Delta_{\mathbb H^3})^{-s/2}$, we have,
\begin{equation*}
(-\Delta_{\mathbb H^3})^{-s/2} = \frac{1}{\Gamma(s/2)} \int_0^\infty t^{\frac{s}{2}-1}\:e^{t\Delta_{\mathbb H^3}}\:dt\:.
\end{equation*}
Now as $(-\Delta_{\mathbb H^3})^{-s/2}f$ for an $f \in C^\infty_c(\mathbb H^3)$ is given by a right convolution with the radial kernel $\mathcal{G}_s$, we get
\begin{equation} \label{bessel_eq1}
\mathcal{G}_s(r) = \frac{1}{\Gamma(s/2)} \int_0^\infty t^{\frac{s}{2}-1}\:h_t(r)\:dt\:,
\end{equation}
where $h_t$ is the heat kernel of $\mathbb H^3$, explicitly given by \cite{AO},
\begin{equation*}
h_t(r):=ct^{-\frac{3}{2}}\:e^{-t}\:\left(\frac{r}{\sinh r}\right)\:e^{-\frac{r^2}{4t}}\:.
\end{equation*}
Plugging the last expression in (\ref{bessel_eq1}), we have
\begin{equation*} 
\mathcal{G}_s(r) = \frac{c}{\Gamma(s/2)} \left(\frac{r}{\sinh r}\right)\int_0^\infty t^{\frac{s-3}{2}-1}\:e^{-\left(t+\frac{r^2}{4t}\right)}\:dt\:.
\end{equation*}
Then performing the change of variable: $t=(r/2)u$, we get
\begin{equation*} 
\mathcal{G}_s(r) = \frac{c}{\Gamma(s/2)} \left(\frac{r}{\sinh r}\right) \left(\frac{r}{2}\right)^{\frac{s-3}{2}}\int_0^\infty u^{\frac{s-3}{2}-1}\:e^{-\frac{r}{2}\left(u+\frac{1}{u}\right)}\:du\:.
\end{equation*}
Identifying the last integral as the modified Bessel function of the second kind of the appropriate order, we get the result.
<!-- end proof-15 -->
<!-- proof-16: body_md -->
We first focus on part (1). We fix $s \in (0,3/2)$ and choose a compact subset $\mathcal{K}' \subset \mathbb R^3$ such that 
\begin{equation*}
dim_{\mathcal{H}}(\mathcal{K}')=dim_{\mathcal{M}}(\mathcal{K}')<3-2s\:,
\end{equation*}
where $dim_{\mathcal{M}}(\mathcal{K}')$ is the Minkowski dimension of $\mathcal{K}'$. For example, if the dimension is integer, we can simply take a compact line segment or a compact 2-disk and for non-integer dimension say $\lambda$, we write $\lambda=\zeta+\lfloor \lambda \rfloor$, $\zeta \in (0,1)$ and consider $\mathcal{K}'=\mathcal{C} \times [0,1]^{\lfloor \lambda \rfloor}$ where $\mathcal{C}$ is a generalized Cantor set with both Hausdorff and Minkowski dimension equal to $\zeta$. 

By identifying $T_o\mathbb H^3 \cong \mathbb R^3$, we then consider
\begin{equation*}
\mathcal{K}:=\exp_o(\mathcal{K}')\:.
\end{equation*}
Since $\exp_o$ is locally bi-Lipschitz, we get that
\begin{equation*}
dim_{\mathcal{H}}(\mathcal{K})=dim_{\mathcal{M}}(\mathcal{K})=dim_{\mathcal{H}}(\mathcal{K}')=dim_{\mathcal{M}}(\mathcal{K}')<3-2s\:.
\end{equation*}

We choose and fix $\alpha \in \left(dim_{\mathcal{M}}(\mathcal{K})\:,\:3-2s\right)$. For $\varepsilon>0$, we consider the $\varepsilon$-neighborhood of $\mathcal{K}$,
\begin{equation*}
\mathcal{K}_\varepsilon:=\left\{x \in \mathbb H^3 : d(x,\mathcal{K})<\varepsilon\right\}\:.
\end{equation*}
Next for $\gamma \in \left(s,(3-\alpha)/2\right)$, let us define, $g : \mathbb H^3 \to [0,+\infty]$ by,
\begin{equation*}
g(x):=\begin{cases}
d(x,\mathcal{K})^{-\gamma}\:,\:\:& x \in \mathcal{K}_\varepsilon\\
0\:,\:\:& x \in \mathbb H^3 \setminus \mathcal{K}_\varepsilon\:.
\end{cases}
\end{equation*}
Now since, $\alpha > dim_{\mathcal{M}}(\mathcal{K})$, the $\alpha$-dimensional Minkowski content of $\mathcal{K}$ defined as,
\begin{equation*}
\mathscr{M}^\alpha(\mathcal{K}):=\lim_{\varepsilon \to 0+} \frac{vol\left(\mathcal{K}_\varepsilon\right)}{\varepsilon^{3-\alpha}}\:,
\end{equation*}
is $0$ and thus
\begin{eqnarray*}
\|g\|^2_{L^2(\mathbb H^3)} &=& \int_{\mathcal{K}_\varepsilon} d(x,\mathcal{K})^{-2\gamma}\:dvol(x) \\
& \le & \sum_{j=0}^\infty  \int_{\mathcal{K}_{2^{-j}\varepsilon} \setminus \mathcal{K}_{2^{-j-1}\varepsilon}} d(x,\mathcal{K})^{-2\gamma}\:dvol(x) \\
& \le & \sum_{j=0}^\infty \left(2^{-j-1}\varepsilon\right)^{-2\gamma}\:\:vol\left(\mathcal{K}_{2^{-j}\varepsilon}\right) \\
& \lesssim & \sum_{j=0}^\infty \left(2^{-j-1}\varepsilon\right)^{-2\gamma}\:\left(2^{-j}\varepsilon\right)^{3-\alpha} \\
&=& \frac{2^{2\gamma} \varepsilon^{3-\alpha-2\gamma}}{1-2^{-(3-\alpha-2\gamma)}} \\
&<& \infty\:,
\end{eqnarray*}
where we have used the fact that  $2\gamma < 3-\alpha$. Hence $g \in L^2(\mathbb H^3)$. Consequently, the convolution with the Bessel kernel,
\begin{equation*} 
f:= g * \mathcal{G}_s \in H^s(\mathbb H^3)\:.
\end{equation*}
We next look at the singularities of $f$. Let $x \in \mathcal{K}_\varepsilon\setminus \mathcal{K}$ and choose $x_0 \in \mathcal{K}$ such that 
\begin{equation*}
d(x,x_0)=d(x,\mathcal{K})=:r\:.
\end{equation*}
We next consider the geodesic ball $B(x_0,r/2)$ and note that for $y \in B(x_0,r/2)$,
\begin{equation*}
d(y,\mathcal{K}) \le d(y,x_0) <\frac{r}{2} < \varepsilon\:.
\end{equation*}
Thus $B(x_0,r/2) \subset \mathcal{K}_\varepsilon$ and hence,
\begin{eqnarray*}
f(x)&=& \int_{\mathcal{K}_\varepsilon} \mathcal{G}_s\left(d(x,y)\right)\:d(y,\mathcal{K})^{-\gamma}\:dvol(y) \\
&\ge & \int_{B(x_0,r/2)} \mathcal{G}_s\left(d(x,y)\right)\:d(y,\mathcal{K})^{-\gamma}\:dvol(y)\:.
\end{eqnarray*}
At this juncture we note that 
\begin{equation*}
d(y,\mathcal{K})^{-\gamma} \ge d(y,x_0)^{-\gamma}\:,
\end{equation*}
and also that for $y \in B(x_0,r/2)$,
\begin{equation*}
d(x,y) \le d(x,x_0)+d(x_0,y) < \frac{3r}{2}\:.
\end{equation*}
Using the above inequalities along with the asymptotic lower bound for the Bessel kernel for small geodesic distances given by Lemma \ref{lemma_bessel_kernel}, we get
\begin{equation*}
f(x) \gtrsim r^{s-3} \int_{B(x_0,r/2)} d(y,x_0)^{-\gamma}\:dvol(y)\:.
\end{equation*} 
Then by the volume growth asymptotics of small geodesic balls given by (\ref{volume_asymptotics}), 
\begin{equation*}
f(x) \gtrsim r^{s-\gamma} \:.
\end{equation*}
Since $r=d(x,\mathcal{K})$, we get that for all $x \in \mathcal{K}_\varepsilon$,
\begin{equation*}
f(x) \gtrsim d(x,\mathcal{K})^{s-\gamma} \:,
\end{equation*}
that is, $f$ is singular on $\mathcal{K}$ (as $\gamma>s$). Then it is easy to see that $f$ is continuous on $\mathbb H^3 \setminus \overline{\mathcal{K}_\varepsilon}$ (by an easy application of the Dominated Convergence Theorem) and dominated by a continuous function on $\overline{\mathcal{K}_\varepsilon}\setminus \mathcal{K}$. Hence $\mathcal{K}$ is precisely the set of singularities of $f$. This completes the proof of part (1).

For part (2), we first consider the radial set in $\mathbb R^3$, considered by Bennett-Rogers in \cite[p.8]{BR} and then take its image under the Riemannian exponential map into $\mathbb H^3$. Then we proceed as in the proof of part (1). This time, we only additionally use the fact that the Riemannian exponential map is a radial isometry.

<!-- theorem-5: body_md -->
\label{H3_transference}
For any $s>0$ and any phase $\psi$, we have the following transference between the Hausdorff dimensions of (i) divergence sets for $\psi$-dispersive propagations on $\mathbb H^3$ with radial initial data and (ii) divergence sets for $\widetilde{\psi}$-dispersive propagations on $\mathbb R^3$ with radial initial data,
\begin{equation*}
\widetilde{\mathfrak d}_{\psi,\mathbb H^3}(s)=\widetilde{\mathfrak d}_{\widetilde{\psi},3}(s)\:,
\end{equation*}
where $$\widetilde{\psi}(\lambda):=\psi\left(\sqrt{\lambda^2+1}\right)\:,\:\:\lambda \ge 0\:.$$
<!-- end theorem-5 -->
<!-- proof-17: body_md -->
By part (2) of Theorem \ref{sobolev_singularity_theorem} (Remark \ref{BR_remark} resp.) the function $\widetilde{\mathfrak d}_{\psi,\mathbb H^3}$ ($\widetilde{\mathfrak d}_{\widetilde{\psi},3}$ resp.) is well-defined for all $s>0$, with $\widetilde{\mathfrak d}_{\psi,\mathbb H^3}(s) \ge 3-2s$ ($\widetilde{\mathfrak d}_{\widetilde{\psi},3}(s) \ge 3-2s$ resp.) and in particular, $\widetilde{\mathfrak d}_{\psi,\mathbb H^3}(s)=0$ for all $s>1/2$ by Lemma \ref{H3-lemma_s>1/2} ($\widetilde{\mathfrak d}_{\widetilde{\psi},3}(s)=0$ for all $s>1/2$ by Lemma \ref{lemma_s>1/2} resp.). 

Let $s_0>0$. Then by definition, 
\begin{equation*}
dim_{\mathcal H}\mathfrak{D}_{\widetilde{\psi}}(f) \le \widetilde{\mathfrak d}_{\widetilde{\psi},3}(s_0)\:,\:\:\text{ for all } f \in H^{s_0}(\mathbb R^3)_{radial}\:.
\end{equation*}  
So in particular, for any fixed $0<r_1<r_2<\infty$,
\begin{equation*}
dim_{\mathcal H}\left(\mathfrak{D}_{\widetilde{\psi}}(f) \cap \mathbb{A}(r_1,r_2)\right) \le \widetilde{\mathfrak d}_{\widetilde{\psi},3}(s_0)\:,\:\:\text{ for all } f \in H^{s_0}(\mathbb R^3)_{radial}\:.
\end{equation*} 
Thus given $\varepsilon>0$, we have for $\alpha:=\widetilde{\mathfrak d}_{\widetilde{\psi},3}(s_0)+\varepsilon$,
\begin{equation*}
\mathcal H^\alpha\left(\mathfrak{D}_{\widetilde{\psi}}(f) \cap \mathbb{A}(r_1,r_2)\right)=0\:,\:\:\text{ for all } f \in H^{s_0}(\mathbb R^3)_{radial}\:.
\end{equation*}
Then by Frostmann's lemma,
\begin{equation*}
\nu\left(\mathfrak{D}_{\widetilde{\psi}}(f) \cap \mathbb{A}(r_1,r_2)\right)=0\:,\:\:\forall \text{ radial } \nu \in \mathcal M^\alpha(\mathbb{A}(r_1,r_2))\:,\:\:\forall f \in H^{s_0}(\mathbb R^3)_{radial}\:.
\end{equation*}
This means that for all radial $\nu \in \mathcal M^\alpha(\mathbb{A}(r_1,r_2))$ and for all $f \in H^{s_0}(\mathbb R^3)_{radial}$,
\begin{equation*} 
e^{it\widetilde{\psi}(\sqrt{-\Delta})} f(x) \to f(x)\:,\:\text{ as } t \to 0\:,\: \nu-\text{a.e. } x \in \mathbb{A}(r_1,r_2)\:\:.
\end{equation*}
Then proceeding as in the Step 1 of the proof of Theorem \ref{transference_theorem}, we obtain that there exists a $\nu$-measurable radial weight $w$ with $0<w\le 1,\:\nu$-a.e. on $\mathbb{A}(r_1,r_2)$ such that the following weighted, weak, fractal maximal estimate holds,
\begin{equation*}
\left\|S^*_{\widetilde{\psi}}f\right\|_{L^{2,\infty}(w\:d\nu)} \lesssim \|f\|_{H^{s_0}(\mathbb R^3)}\:,\:\:\text{ for all } f \in \mathcal S(\mathbb R^3)_{radial}\:.
\end{equation*} 
Next proceeding as in the proof of Lemma \ref{H3-N-maximal-estimate-radial}, we get that for all radial $\mu \in \mathcal M^\alpha(A(r_1,r_2))$,
\begin{equation*}
\left\|S^*_{\psi}f\right\|_{L^{2,\infty}\left(\left(w\:\circ\: \exp^{-1}_o\right)\:d\mu \right)} \lesssim \|f\|_{H^{s_0}(\mathbb H^3)}\:,
\end{equation*}
holds for all $f \in \mathcal S^2(\mathbb H^3)_{radial}$. Now by the standard method of maximal functions using the above maximal estimate, density of $\mathcal S^2(\mathbb H^3)_{radial}$ and the fractal Sobolev embedding for the initial data (Lemma \ref{H3-Sobolev-embedding}), we get that for all radial $\mu \in \mathcal M^\alpha(A(r_1,r_2))$ and all $f \in H^{s_0}(\mathbb H^3)_{radial}$,
\begin{equation*}
\mu \left\{x \in A(r_1,r_2) : \displaystyle\lim_{t \to 0} e^{it\psi\left(\sqrt{-\Delta_{\mathbb H^3}}\right)}f(x) \ne f(x)\right\}=0\:.
\end{equation*}
Thus by Frostmann's lemma,
\begin{equation*}
\mathcal{H}^\alpha  \left\{x \in A(r_1,r_2) : \displaystyle\lim_{t \to 0} e^{it\psi\left(\sqrt{-\Delta_{\mathbb H^3}}\right)}f(x) \ne f(x)\right\}=0\:,
\end{equation*}
for all $f \in H^{s_0}(\mathbb H^3)_{radial}$. By countable additivity of Hausdorff measure, we get that
\begin{equation*}
\mathcal{H}^\alpha\left(\mathfrak{D}_{\psi}(f)\right)=\mathcal{H}^\alpha \left\{x \in \mathbb H^3 : \displaystyle\lim_{t \to 0} e^{it\psi\left(\sqrt{-\Delta_{\mathbb H^3}}\right)}f(x) \ne f(x)\right\}=0\:,
\end{equation*}
for all $f \in H^{s_0}(\mathbb H^3)_{radial}$. This implies that 
\begin{equation*} 
\widetilde{\mathfrak d}_{\psi,\mathbb H^3}(s_0) \le \alpha = \widetilde{\mathfrak d}_{\widetilde{\psi},3}(s_0)+\varepsilon\:.
\end{equation*}
Since, the above inequality is true for any $\varepsilon>0$, we get that
\begin{equation*} 
\widetilde{\mathfrak d}_{\psi,\mathbb H^3}(s_0) \le \widetilde{\mathfrak d}_{\widetilde{\psi},3}(s_0)\:.
\end{equation*}
By the symmetry of the argument, switching the roles of $\psi$ and $\widetilde{\psi}$, we get the opposite inequality and thus getting the result.
