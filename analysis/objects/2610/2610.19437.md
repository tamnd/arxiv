A Mukai vector $v=(r,\xi,a) \in H^*(X,{\mathbb Z})_{\operatorname{alg}}$ is positive if
\begin{enumerate}
\item $r>0$ or 
\item $r=0$ and $\xi \ne 0$ is effective or
\item $r=\xi=0$ and $a>0$.
\end{enumerate}
We denote a positive Mukai vector $v$ by $v>0$.

<!-- thm-1: body_md -->
\label{thm:BM}
Let $(X,H)$ be a pair of K3 surface $X$ and an ample divisor $H$ on $X$
and  $(X',H')$ be a pair of K3 surface $X'$ and an ample divisor $H'$ on $X'$. 
Then the following holds.
\begin{enumerate}
\item[(1)]
$M_H(v)$ is birationally equivalent to $M_{H'}(v')$ if and only if 
there is a (contravariant) equivalence $\Phi:{\bf D}(X) \to {\bf D}(X')$ 
such that $\overline{\Phi}(v)=v'$, where
$\overline{\Phi}:H^*(X,{\mathbb Z}) \to H^*(X',{\mathbb Z})$ is the Hodge
isometry induced by $\Phi$.
\item[(2)]
Under the condition of (1),
we can choose $\Phi$ to satisfy that
$\Phi(E) \in M_{H'}(v')$ for a general $E \in M_H(v)$.
\end{enumerate}

\label{thm:intro:bir-auto}
Let $(X,H)$ and $(X',H')$ be polarized K3 surfaces.
Let $\varphi:M_H(v) \cdots \to M_{H'}(v')$ be a birational
map, where $v \in H^*(X,{\mathbb Z})$ and $v' \in H^*(X',{\mathbb Z})$ are positive Mukai vectors.
Assume that $\operatorname{Pic}(X)={\mathbb Z}H$.
Then there is a (contravariant) Fourier-Mukai transform
$\Phi:{\bf D}(X) \to {\bf D}(X')$ 
which induces $\varphi_*$.

\label{defn:theta}
Assume that $v$ is primitive and $\langle v^2 \rangle \geq 0$.
\begin{enumerate}
\item[(1)]
\begin{equation}\label{eq:theta}
\theta_{v}:v^{\perp} \to H^2(M_{H}(v),{\mathbb Z})
\end{equation}
denotes the Mukai's homomorphism.
\item[(2)]
If there is a universal family
${\bf E}$ on $M_{H}(v)$, e.g.,
there is a Mukai vector $w$ with
$\langle v,w \rangle=1$,
then 
$$
\theta_{v}(x)=
c_1(p_{M_{H}(v)*}
(\operatorname{ch} ({\bf E})p_X^*(1+\varrho_X) p_X^*(x^{\vee})))=
-c_1(p_{M_{H}(v)*}(\operatorname{ch} ({\bf E}^{\vee}) p_X^*(1+\varrho_X) p_X^*(x))).
$$ 
where $p_X, p_{M_H(v)}$ are projections
from $X \times M_H(v)$ to $X$ and
$M_H(v)$ respectively. 
\end{enumerate}

<!-- lem-3: body_md -->
\label{lem:lifting}
Let $X$ and $Y$ be K3 surfaces.
Assume that there is no $(-2)$-curves on $Y$.
\begin{enumerate}
\item[(1)]
Let $f:H^*(X,{\mathbb Z}) \to H^*(Y,{\mathbb Z})$ be an orientation preserving Hodge isometry.
Then there is an equivalence  
$\Phi_{X \to Y}^{{\mathcal E}^{\vee}[k]}:{\bf D}(X) \to {\bf D}(Y)$
such that ${\mathcal E}$ is a family of stable sheaves on $X$ 
and $\Phi_{X \to Y}^{{\mathcal E}^{\vee}[k]}$ induces $f$.
\item[(2)]
Let $f:H^*(X,{\mathbb Z}) \to H^*(Y,{\mathbb Z})$ be an orientation reversing Hodge isometry.
Then there is an equivalence  
$\Phi_{X \to Y}^{{\mathcal E}[k]}:{\bf D}(X) \to {\bf D}(Y)$
such that ${\mathcal E}$ is a family of stable sheaves on $X$ 
and ${\mathcal D}_{Y} \circ \Phi_{X \to Y}^{{\mathcal E}[k]}=\Phi_{X \to Y}^{{\mathcal E}^{\vee}[2-k]} \circ {\mathcal D}_X$ induces $f$.
\item[(3)]
Let $f:H^*(X,{\mathbb Z}) \to H^*(Y,{\mathbb Z})$ be an orientation reversing Hodge isometry.
Then there is an equivalence  
$\Phi_{X \to Y}^{{\mathcal E}^{\vee}[k]}:{\bf D}(X) \to {\bf D}(Y)$
such that ${\mathcal E}$ is a family of stable sheaves on $X$ 
and ${\mathcal D}_Y \circ \Phi_{X \to Y}^{{\mathcal E}^{\vee}[k]}$ induces $f$.
\end{enumerate}
<!-- end lem-3 -->
<!-- proof-2: body_md -->
(1)
Let $f:H^*(X,{\mathbb Z}) \to H^*(Y,{\mathbb Z})$ be an orientation preserving Hodge isometry.
We take $u \in H^*(X,{\mathbb Z})_{\operatorname{alg}}$ with $f(u)=\varrho_{Y}$.
Replacing $f$ by $-f$, we assume that $u>0$.
We set $X':=M_H(u)$, where $H$ is a general ample divisor.
Since $\langle f^{-1}(v({\mathcal O}_X),u \rangle=\langle v({\mathcal O}_Y),\varrho_X \rangle=-1$,
there is a universal family ${\mathcal E}$ on $X \times X'$.
Let $\Phi_{X \to X'}^{{\mathcal E}^{\vee}}:{\bf D}(X) \to {\bf D}(X')$ be an equivalence
such that $\overline{\Phi}_{X \to X'}^{{\mathcal E}^{\vee}}(u)=\varrho_{X'}$.
                                      
                                                            
Since $\Phi_{X \to X'}^{{\mathcal E}^{\vee}}$ is orientation preserving (\cite{HMS}),
$\varphi:=\overline{\Phi}_{X \to X'}^{{\mathcal E}^{\vee}} \circ f^{-1}$ is
also orientation preserving. Thus
$\varphi:H^*(Y,{\mathbb Z}) \to H^*(X',{\mathbb Z})$ is an orientation preserving Hodge isometry
such that 
$\varphi(\varrho_Y)=\varphi_{X'}$ and
$\varphi(v({\mathcal O}_Y))=v({\mathcal O}_{X'}(D))$ $(D \in \operatorname{NS}(X'))$.
Replacing ${\mathcal E}$ by ${\mathcal E}(D)$,
we get $\varphi(v({\mathcal O}_Y))=v({\mathcal O}_{X'})$.
Then we have a Hodge isometry
$H^2(Y,{\mathbb Z}) \to H^2(X',{\mathbb Z})$ preserving positive cone, 
which shows that $Y \cong X'$.
                                                                                  
Since there is no $(-2)$-curves on $Y$, $\varphi$ is induced by an isomorphism
$\psi:Y \to X'$.
Thus ${\mathcal E}_1:=(1_X \times \psi)^*({\mathcal E})$
defines a desired Fourier-Mukai transform
$\Phi_{X \to Y}^{{\mathcal E}_1^{\vee}}$.
\begin{NB}
$\Phi_{X \to Y}^{{\mathcal E}_1^{\vee}}=\psi^* \circ \Phi_{X \to X'}^{{\mathcal E}^{\vee}}$.
\end{NB}

(2)
Since $f \circ \overline{\mathcal D}_{X}$ is orientation preserving, the claim follows from (1).

(3)
Since $\overline{\mathcal D}_Y \circ f$ is orientation preserving, 
the claim also follows from (1).
