<!-- definition-1: body_md -->
\label{defweak}
Assume that (AOP) and (AG) hold, and let \(\rho>-1\). A function
\(u\in C([0,\infty);V_1)\) with \(u_t\in C([0,\infty);V_0)\) satisfying
\((u(0);u_t(0))=(u_0;u_1)\) is called a \textit{global strong solution}
to Problem \eqref{abstract} on \([0,\infty)\) if:
\begin{description}
\item[(i)] \(u\in W^{1,\infty}(0,T;V_1)\) and \(u_t\in W^{1,\infty}(0,T;V_0)\)
for all \(0<T<\infty\);
\item[(ii)] Equation \eqref{abstract} holds in \(V_0\) for almost every
\(t\in[0,\infty)\).
\end{description}
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{thmwellposedness}
Let Assumptions (AOP) and (AG) hold and let \(\rho>-1\). For every initial data
\((u_0;u_1)\in\mathcal{H}_1\), Problem \eqref{abstract} admits a unique global solution
\(u\) on \([0,\infty)\). Moreover, this solution satisfies
\[
(u;u_t)\in L^{\infty}(0,\infty;\mathcal{H}_1),\qquad
u_{tt}\in L^{\infty}(0,\infty;V_0),
\]
and hence
\[
u\in C([0,\infty);V_1),\qquad u_t\in C([0,\infty);V_0).
\]
Consequently, \(u\) is a global solution in the sense of Definition~\ref{defweak}.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{main}
Assume that (AOP) and (AG) hold, and let \(E(0)>0\).
\begin{enumerate}
\item For \(-1<\rho<0\),
\[
E(t)\leq C(\rho,\mu_0,\lambda_1,g(0),G_0,k)
\left(1+E(0)^{-\frac{\rho}{2\rho+2}}\right)^{\frac{2\rho+2}{|\rho|}}
t^{\frac{2\rho+2}{\rho}},\qquad t\geq t_0,
\]
where \(C(\rho,\mu_0,\lambda_1,g(0),G_0,k)>0\) is a constant depending only on the indicated parameters, and \(t_0>0\) is defined by
\[
\int_0^{t_0}g(s)ds=\frac12 G_0,
\]
with \(G_0=\int_0^\infty g(s)ds\) given in \eqref{G0}. Equivalently, since
\(-\frac{\rho}{2\rho+2}\in(0,\frac12)\) and
\(-\frac{2\rho+2}{\rho}=\frac{2\rho+2}{|\rho|}>0\), one may replace the
constant factor above by
\(\left(1+E(0)^{-1}\right)^{\frac{2\rho+2}{|\rho|}}\), at the cost of
enlarging the constant \(C\).

\item For \(\rho\geq0\), if the constant \(k\) in Assumption (AG) satisfies \(k\geq2g(0)\), then
\[
E(t)\leq
\begin{cases}
E(0)e^{-\kappa(t-1)^+}, & \rho=0,\\[1ex]
\left[\dfrac{\rho}{2\mu}(t-1)^++E(0)^{-\frac{\rho}{2}}\right]^{-\frac{2}{\rho}}, & \rho>0,
\end{cases}
\qquad t\geq0.
\]
Here
\[
\mu:=\left[2(E(0))^{\frac{\rho}{\rho+2}}+4\mu_1\right]^{1+\frac{\rho}{2}},\qquad
\kappa=-\log(1-1/\mu).
\]
Since \(\mu_1>0\) and \(\rho>0\), we have \(\mu>1\), so the constant
\(\kappa\) is well defined. Moreover,
\[
\mu_1:=\frac{32}{\mu_0\lambda_1}+\frac{4(2E(0))^\rho}{\mu_0\lambda_1}+\frac{3}{2}+\frac{2}{k}(E(0))^{\frac{\rho}{\rho+2}},
\]
with \(s^+:=\frac{1}{2}(s+|s|)\), and the positive constants \(\mu_0\) and \(\lambda_1\) given in \eqref{mu_0} and \eqref{lam}, respectively.
\end{enumerate}

<!-- remark-1: body_md -->
The restriction \(\rho>-1\) is essential for the present approach. For \(\rho>-1\), the operator \(l(v)=\|v\|^\rho v\) is continuous, monotone, and semicontinuous on \(V_0\), and \(l(0)=0\). These properties are used both in the Faedo--Galerkin compactness argument and in the uniqueness proof. When \(\rho\le -1\), the operator \(l\) is no longer continuous and monotone at the origin; although the dissipation rate \(\|u_t\|^{\rho+2}\) remains nonnegative for \(\rho>-2\), the well-posedness theory requires a different approach. We therefore leave the case \(\rho\le -1\) for future work.
<!-- end remark-1 -->
<!-- lemma-1: body_md -->
\label{continuity}
Assume that \(\rho>-1\), and let
\[
l(v):=\|v\|^{\rho}v,\qquad v\in V_0.
\]
Then \(l:V_0\to V_0\) is monotone and semicontinuous. More precisely:
\begin{enumerate}
\item \((l(u)-l(v),u-v)\geq0\) for all \(u,v\in V_0\);
\item for all \(u,v,w\in V_0\), the map \(\lambda\mapsto(l(u+\lambda v),w)\) is continuous from \(\mathbb{R}\) to \(\mathbb{R}\).
\end{enumerate}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
If \(u=0\), then
\[
(l(u)-l(v),u-v)=(-\|v\|^{\rho}v,-v)=\|v\|^{\rho+2}\ge0;
\]
if \(v=0\), then
\[
(l(u)-l(v),u-v)=(\|u\|^{\rho}u,u)=\|u\|^{\rho+2}\ge0.
\]
Hence assume \(u\neq0\) and \(v\neq0\). For \(u,v\in V_0\), we have
\begin{align*}
(l(u)-l(v),u-v)
&=(\|u\|^{\rho}u-\|v\|^{\rho}v,u-v)\\
&=\|u\|^{\rho}(\|u\|^2-(u,v))+\|v\|^{\rho}(\|v\|^2-(u,v))\\
&\geq\|u\|^{\rho+2}-\|u\|^{\rho+1}\|v\|+\|v\|^{\rho+2}-\|v\|^{\rho+1}\|u\|\\
&=(\|u\|-\|v\|)(\|u\|^{\rho+1}-\|v\|^{\rho+1}).
\end{align*}
Since \(\rho+1>0\), the function \(t\mapsto t^{\rho+1}\) is nondecreasing on \([0,\infty)\), and hence the last expression is nonnegative. This proves (i).

To prove (ii), fix \(u,v,w\in V_0\) and set
\[
\phi(\lambda):=(l(u+\lambda v),w)=\|u+\lambda v\|^{\rho}((u,w)+\lambda(v,w)),\qquad \lambda\in\mathbb{R}.
\]
If \(u=0\) and \(v=0\), then \(\phi\equiv0\). If \(u=0\) and \(v\neq0\), then
\[
\phi(\lambda)=|\lambda|^{\rho}\lambda\|v\|^{\rho}(v,w),
\]
which is continuous at \(\lambda=0\) because \(\rho>-1\) implies \(\rho+1>0\), and continuous elsewhere by the continuity of \(\lambda\mapsto|\lambda|^\rho\lambda\) on \(\mathbb{R}\setminus\{0\}\). If \(u\neq0\) and \(v=0\), then \(\phi\) is constant.

Finally, if \(u\neq0\) and \(v\neq0\), we distinguish two cases.

\medskip
\noindent\emph{Case 1: there exists \(\lambda_0\in\mathbb{R}\) such that \(u+\lambda_0 v=0\).}
Such a \(\lambda_0\) is unique. On \(\mathbb{R}\setminus\{\lambda_0\}\), the continuity of \(\phi\) follows from the continuity of \(\lambda\mapsto\|u+\lambda v\|\) and of \(t\mapsto t^\rho\) on \((0,\infty)\). At \(\lambda=\lambda_0\), we have \(u+\lambda_0 v=0\), hence \((u,w)+\lambda_0(v,w)=0\), and
\[
\phi(\lambda)=\|u+\lambda v\|^{\rho}((\lambda-\lambda_0)(v,w))
=|\lambda-\lambda_0|^{\rho}(\lambda-\lambda_0)\|v\|^{\rho}(v,w)\to0
\]
as \(\lambda\to\lambda_0\), since \(\rho+1>0\). Thus \(\phi\) is continuous at \(\lambda_0\) as well, hence on all of \(\mathbb{R}\).

\medskip
\noindent\emph{Case 2: no such \(\lambda_0\) exists.}
Then \(u+\lambda v\neq0\) for every \(\lambda\in\mathbb{R}\), so \(\|u+\lambda v\|>0\) for every \(\lambda\). The continuity of \(\phi\) on \(\mathbb{R}\) follows from the continuity of \(\lambda\mapsto\|u+\lambda v\|\) and of \(t\mapsto t^\rho\) on \((0,\infty)\).

Thus \(\phi\) is continuous in all cases. This proves (ii).
<!-- end proof-1 -->
<!-- proof-2: body_md -->
Let \(\{\omega_k\}_{k=1}^\infty\) be the orthonormal basis of \(V_0\) given in \eqref{basis}. For each \(n\in\mathbb{N}\), we consider the approximate problem
\begin{align}
\left(u_{tt}^n(t),\omega_k\right)&+\left(Au^n(t),\omega_k\right)\notag\\
&-\left(\int_{0}^{t}g(t-s)Au^n(s)ds,\omega_k\right)
+\|u_t^n(t)\|^{\rho}\left(u_t^n(t),\omega_k\right)=0,\quad t>0,\label{weak}
\end{align}
for \(k=1,\dots,n\), where
\begin{equation}\label{bc1}
u^n(t)=\sum_{j=1}^n c_{nj}(t)\omega_j,\qquad u^n(0)=u^{n}_0,\qquad u^n_t(0)=u^{n}_1,
\end{equation}
and
\begin{align}
\left(u^{n}_0;u^{n}_1\right)\rightarrow(u_0;u_1)\quad\text{strongly in }\mathcal{H}_1\text{ as }n\to\infty.\label{defineun}
\end{align}

Substituting \eqref{bc1} into \eqref{weak}, we obtain the system of ordinary differential equations
\begin{align}
c''_{nk}&+\sum_{j=1}^{n}(A^{\frac12}\omega_j,A^{\frac12}\omega_k)c_{nj}\notag\\
&-\sum_{j=1}^{n}\left(A^{\frac12}\omega_j,A^{\frac12}\omega_k\right)
\int_{0}^{t}g(t-s)c_{nj}(s)ds
+F_k\left(c'_{n1},\dots,c'_{nn}\right)=0,\label{withouttime}
\end{align}
where
\[
F_k(c'_{n1},\dots,c'_{nn})
:=
\begin{cases}
\left(\sum_{j=1}^n(c'_{nj})^2\right)^{\rho/2}c'_{nk}, & \sum_{j=1}^n(c'_{nj})^2\ne0,\\[1ex]
0, & \sum_{j=1}^n(c'_{nj})^2=0.
\end{cases}
\]
Since \(\rho>-1\), the map \((c'_{n1},\dots,c'_{nn})\mapsto F_k(c'_{n1},\dots,c'_{nn})\) is continuous: indeed, with \(r=(\sum_{j=1}^n(c'_{nj})^2)^{1/2}\), we have \(|F_k|\leq r^{\rho+1}\to0\) as \(r\to0\), because \(\rho+1>0\). Moreover, the memory term is linear and continuous in \(c_{nj}\). Hence the right-hand side of \eqref{withouttime} is continuous in \((c_{nj},c'_{nj})\), and by the Peano existence theorem there exists a local solution \(c_{nk}\in C^2([0,T_n])\), \(k=1,\dots,n\), for some \(T_n>0\).

We now derive uniform a priori estimates on \([0,T_n]\). Multiplying \eqref{weak} by \(c'_{nk}(t)\) and summing over \(k=1,\dots,n\), we obtain
\begin{align}
\frac12\frac{d}{dt}&\left(\left\|u_t^n(t)\right\|^2+\left\|A^{\frac12}u^n(t)\right\|^2\right)\notag\\
&-\left(\int_{0}^{t}g(t-s)A^{\frac12}u^n(s)ds,A^{\frac12}u_t^n(t)\right)
+\|u_t^n\|^{\rho+2}=0.\label{oneorder}
\end{align}
We use the standard identity for the memory term,
\begin{align}
&-\left(\int_{0}^{t}g(t-s)A^{\frac12}u^n(s)ds,A^{\frac12}u_t^n(t)\right)\notag\\
=&\frac12\frac{d}{dt}\int_{0}^{t}g(t-s)\left\|A^{\frac12}u^n(t)-A^{\frac12}u^n(s)\right\|^2ds
-\frac12\int_{0}^{t}g'(t-s)\left\|A^{\frac12}u^n(t)-A^{\frac12}u^n(s)\right\|^2ds\notag\\
&+\frac12g(t)\left\|A^{\frac12}u^n(t)\right\|^2
-\frac12\frac{d}{dt}\left(\int_{0}^{t}g(s)ds\left\|A^{\frac12}u^n(t)\right\|^2\right).\label{import}
\end{align}
Substituting \eqref{import} into \eqref{oneorder} and integrating from \(0\) to \(t\), we obtain
\begin{align}
E_n(t)+\int_{0}^{t}\left\|u_t^n(s)\right\|^{\rho+2}ds=&E_n(0)+\frac12\int_{0}^{t}\int_{0}^{s}g'(s-\tau)\left\|A^{\frac12}u^n(s)-A^{\frac12}u^n(\tau)\right\|^2d\tau ds\notag\\
&-\frac12\int_{0}^{t}g(s)\left\|A^{\frac12}u^n(s)\right\|^2ds,\label{energyidentity}
\end{align}
where
\begin{align}
E_n(t):=&\frac12\left\|u_t^n(t)\right\|^2
+\frac12\left(1-\int_{0}^{t}g(s)ds\right)\left\|A^{\frac12}u^n(t)\right\|^2\notag\\
&+\frac12\int_{0}^{t}g(t-s)\left\|A^{\frac12}u^n(t)-A^{\frac12}u^n(s)\right\|^2ds.\label{En}
\end{align}
Since
\[
E_n(0)=\frac12\left(\|u_1^n\|^2+\left\|A^{\frac12}u_0^n\right\|^2\right)
\le C\left(\|(u_0^n;u_1^n)\|_{\mathcal{H}_1}\right),
\]
and \((u_0^n;u_1^n)\) is uniformly bounded in \(\mathcal{H}_1\) by \eqref{defineun}, the term \(E_n(0)\) is uniformly bounded. Since \(g'\leq0\) and \(g\geq0\), the last two terms on the right-hand side of \eqref{energyidentity} are nonpositive. Hence, by \eqref{mu_0},
\begin{align}\label{weakestimateone}
\left\|u_t^n(t)\right\|^2&+\left\|A^{\frac12}u^n(t)\right\|^2\notag\\
&+\int_0^t g(t-s)\left\|A^{\frac12}u^n(t)-A^{\frac12}u^n(s)\right\|^2ds
\leq C\left(\|(u_0;u_1)\|_{\mathcal{H}_1}\right),
\end{align}
for all \(t\in[0,T_n]\). In particular, by \eqref{defineun},
\begin{equation}\label{weakestimateoneb}
\left\|u_t^n(t)\right\|^2+\left\|A^{\frac12}u^n(t)\right\|^2\leq C\left(\|(u_0;u_1)\|_{\mathcal{H}_1}\right),\qquad t\in[0,T_n].
\end{equation}

Next, multiplying \eqref{weak} by \(\lambda_kc'_{nk}(t)\) and summing over \(k=1,\dots,n\), we obtain
\begin{align*}
\frac12\frac{d}{dt}&\left(\left\|A^{\frac12}u_t^n(t)\right\|^2+\left\|Au^n(t)\right\|^2\right)
\\
&-\left(\int_{0}^{t}g(t-s)Au^n(s)ds,Au_t^n(t)\right)
+\|u_t^n\|^{\rho}\left\|A^{\frac12}u^n_t\right\|^2=0.
\end{align*}
Using the same identity for the memory term with \(A^{1/2}u^n\) replaced by \(Au^n\), and noting that \(\|u_t^n\|^{\rho}\|A^{1/2}u_t^n\|^2\geq0\), we deduce
\begin{align}
\left\|A^{\frac12}u_t^n(t)\right\|^2+\left\|Au^n(t)\right\|^2\leq C\left(\|(u_0;u_1)\|_{\mathcal{H}_1}\right),\qquad t\in[0,T_n].\label{weaktwo}
\end{align}

Finally, multiplying \eqref{weak} by \(c''_{nk}(t)\) and summing over \(k=1,\dots,n\), we get
\[
\|u_{tt}^n\|^2=\sum_{i=1}^{3}Y_i,
\]
where
\[
Y_1:=-\left(Au^n(t),u_{tt}^n\right),\quad
Y_2:=\left(\int_{0}^{t}g(t-s)Au^n(s)ds,u_{tt}^n\right),\quad
Y_3:=-\|u_t^n\|^{\rho}\left(u_t^n,u_{tt}^n\right).
\]
By Cauchy's inequality, Young's inequality, \eqref{mu_0}, \eqref{G0}, \eqref{weakestimateoneb} and \eqref{weaktwo}, we obtain
\[
|Y_1|\leq\frac16\left\|u_{tt}^n\right\|^2+C\left(\|(u_0;u_1)\|_{\mathcal{H}_1}\right),
\]
\[
|Y_2|\leq\frac16\left\|u_{tt}^n\right\|^2+C\left(\|(u_0;u_1)\|_{\mathcal{H}_1}\right),
\]
\[
|Y_3|\leq\frac16\left\|u_{tt}^n\right\|^2+C\left(\|(u_0;u_1)\|_{\mathcal{H}_1}\right).
\]
Indeed, for \(Y_2\) we used
\[
\left\|\int_0^t g(t-s)Au^n(s)ds\right\|
\leq G_0\|Au^n\|_{L^\infty(0,\infty;V_0)}\leq C,
\]
and for \(Y_3\) we used \(\|u_t^n\|\leq C\) together with the fact that,
since \(\rho+1>0\), the function \(s\mapsto s^{\rho+1}\) is continuous on
\([0,\infty)\) and hence bounded on \([0,C]\), which gives
\(\|u_t^n\|^{\rho+1}\leq C\). Hence
\begin{align}
\|u_{tt}^n\|^2\leq C(\|(u_0;u_1)\|_{\mathcal{H}_1}),\qquad t\in[0,T_n].\label{utt}
\end{align}
Combining \eqref{defineun}, \eqref{weakestimateoneb}, \eqref{weaktwo} and \eqref{utt}, we arrive at
\begin{equation}\label{bc8}
\|u_{t}^n(t)\|^2+\|u_{tt}^n(t)\|^2+\left\|A^{\frac12}u_t^n(t)\right\|^2+\left\|Au^n(t)\right\|^2\leq C\left(\|(u_0;u_1)\|_{\mathcal{H}_1}\right),\quad t\in[0,T_n].
\end{equation}
Since the bound on the right-hand side of \eqref{bc8} is independent of \(T_n\), the local solution can be extended to all \(t\geq0\), and hence \(T_n=\infty\). Consequently, the estimates \eqref{weakestimateoneb}, \eqref{weaktwo}, \eqref{utt} and \eqref{bc8} hold for all \(t\in[0,\infty)\).

From \eqref{bc8}, \eqref{G0} and \(\rho>-1\), we further obtain, for all \(t\in[0,\infty)\),
\[
\left\|\|u_t^n(t)\|^{\rho}u_t^n(t)\right\|=\|u_t^n(t)\|^{\rho+1}\leq C\left(\|(u_0;u_1)\|_{\mathcal{H}_1}\right),
\]
since \(\|u_t^n(t)\|\leq C\) by \eqref{bc8} and the function \(s\mapsto s^{\rho+1}\) is continuous on \([0,\infty)\) with \(\rho+1>0\). Moreover,
\begin{align*}
 \left\|\int_{0}^{t}g(t-s)A^{\frac12}u^n(s)ds\right\|_1
&=\left\|A^{\frac12}\int_{0}^{t}g(t-s)A^{\frac12}u^n(s)ds\right\|
\leq\int_{0}^{t}g(t-s)\|Au^n(s)\|ds
\\
&\leq G_0\|Au^n\|_{L^\infty(0,\infty;V_0)}
\leq C\left(\|(u_0;u_1)\|_{\mathcal{H}_1}\right).
\end{align*}
Therefore, there exist a subsequence of \(\{u^n\}\), still denoted by \(\{u^n\}\), a function \(u\in L^{\infty}(0,\infty;V_2)\) with \(u_t\in L^{\infty}(0,\infty;V_1)\), and a function \(\chi\in L^\infty(0,\infty;V_0)\) such that, as \(n\to\infty\),
\begin{equation}\label{strongsolu}
\left\{
\begin{split}
&u^n\rightharpoonup u\quad \text{weakly star in }L^{\infty}(0,\infty;V_2),\\
&u_t^n\rightharpoonup u_t\quad \text{weakly star in }L^{\infty}(0,\infty;V_1),\\
&u_{tt}^n\rightharpoonup u_{tt}\quad \text{weakly star in }L^{\infty}(0,\infty;V_{0}),\\
&\int_{0}^{t}g(t-s)A^{\frac12}u^n(s)ds\rightharpoonup\int_{0}^{t}g(t-s)A^{\frac12}u(s)ds\quad \\
&\qquad\qquad\qquad\qquad\qquad\text{weakly star in }L^{\infty}(0,\infty;V_1),\\
&\|u_t^n\|^{\rho}u_t^n\rightharpoonup\chi\quad \text{weakly star in }L^{\infty}(0,\infty;V_0).
\end{split}
\right.
\end{equation}

Since \(V_2\hookrightarrow V_1\hookrightarrow V_0\) compactly, the Aubin--Lions--Simon theorem \cite{MR0916688} together with \eqref{bc8} and \eqref{strongsolu} yields, after passing to a subsequence,
\begin{equation}\label{bc5}
\begin{split}
&u_{t}^n\rightarrow u_{t}\quad \text{strongly in }C([0,T];V_{0})\text{ for every }T>0,\\
&u^n\rightarrow u\quad \text{strongly in }C([0,T];V_{1})\text{ for every }T>0.
\end{split}
\end{equation}
In particular, \(u^n(0)\to u(0)\) strongly in \(V_1\) and \(u^n_t(0)\to u_t(0)\) strongly in \(V_{0}\). By \eqref{defineun},
\begin{equation}\label{initialdy}
u(0)=u_0,\qquad u_t(0)=u_1.
\end{equation}

We next identify the weak limit \(\chi\). Since \(l:V_0\to V_0\) is monotone and semicontinuous, we argue as in the classical monotone operator method. For any \(\varphi\in L^2(0,t;V_0)\), the monotonicity of \(l\) gives
\[
\int_{0}^{t}\left(\|u_{\tau}^n\|^{\rho}u_{\tau}^n-\|\varphi\|^{\rho}\varphi,u_{\tau}^n-\varphi\right)d\tau\geq0.
\]
Taking \(\liminf\) as \(n\to\infty\), and using \eqref{strongsolu} and \eqref{bc5}, we obtain
\[
\int_{0}^{t}\left(\chi-\|\varphi\|^{\rho}\varphi,u_{\tau}-\varphi\right)d\tau\geq0.
\]
Here we used that \(\|u_{\tau}^n\|^{\rho}u_{\tau}^n\rightharpoonup\chi\) weakly in \(L^2(0,t;V_0)\) and \(u_{\tau}^n\to u_{\tau}\) strongly in \(L^2(0,t;V_0)\), so that the product converges. Choosing \(\varphi=u_\tau-\lambda\omega\) with \(\lambda\geq0\) and \(\omega\in L^2(0,t;V_0)\), and then letting \(\lambda\to0^+\), we get
\[
\int_{0}^{t}\left(\chi-\|u_\tau\|^{\rho}u_\tau,\omega\right)d\tau\geq0.
\]
Repeating the argument with \(\lambda\leq0\) yields the reverse inequality. Hence
\begin{equation}\label{chiey}
\chi=\|u_t\|^{\rho}u_t.
\end{equation}

Now, letting \(n\to\infty\) in \eqref{weak}, and using \eqref{strongsolu} and \eqref{chiey}, we obtain
\[
(u_{tt}(t),\omega_k)+(Au(t),\omega_k)-\left(\int_{0}^{t}g(t-s)Au(s)ds,\omega_k\right)+\|u_t\|^{\rho}(u_t,\omega_k)=0
\]
for \(t\in(0,\infty)\) and \(k=1,2,\cdots\). Since \(\{\omega_k\}_{k=1}^\infty\) is a basis of \(V_0\), it follows that
\[
u_{tt}(t)+Au(t)-\int_{0}^{t}g(t-s)Au(s)ds+\|u_t\|^{\rho}u_t=0
\]
holds in \(V_0\) for almost every \(t\in(0,\infty)\). Together with \eqref{initialdy}, this proves the existence part of the theorem.

Moreover, the above estimates show that \((u;u_t)\in L^{\infty}(0,\infty;\mathcal{H}_1)\) and \(u_{tt}\in L^{\infty}(0,\infty;V_0)\). Since \(u\in L^\infty(0,\infty;V_2)\), \(u_t\in L^\infty(0,\infty;V_1)\), we have \(u\in W^{1,\infty}(0,T;V_1)\subset C([0,T];V_1)\) for every \(T>0\). Similarly, since \(u_t\in L^\infty(0,\infty;V_1)\) and \(u_{tt}\in L^\infty(0,\infty;V_0)\), we have \(u_t\in W^{1,\infty}(0,T;V_0)\subset C([0,T];V_0)\) for every \(T>0\). Thus \(u\in C([0,\infty);V_1)\) and \(u_t\in C([0,\infty);V_0)\), which matches Definition~\ref{defweak}.

It remains to prove uniqueness. Let \(u\) and \(v\) be two global solutions of \eqref{abstract} with initial data \(z_0=(u_0;u_1)\in\mathcal{H}_1\) and \(\widetilde{z}_0=(v_0;v_1)\in\mathcal{H}_1\), respectively, and set \(\omega:=u-v\). Then
\begin{equation}\label{unique}
\omega_{tt}+A\omega-\int_{0}^{t}g(t-s)A\omega(s)ds+\|u_t\|^{\rho}u_t-\|v_t\|^{\rho}v_t=0.
\end{equation}
Taking the multiplier \(\omega_t\) in \eqref{unique}, we obtain
\[
\frac12\frac{d}{dt}\left(\|\omega_t\|^2+\|A^{1/2}\omega\|^2\right)
-\left(\int_0^t g(t-s)A\omega(s)ds,\omega_t\right)
+\left(\|u_t\|^\rho u_t-\|v_t\|^\rho v_t,\omega_t\right)=0.
\]
Define
\begin{align*}
 E_\omega(t):=&\frac12\left(\|\omega_t(t)\|^2+\left(1-\int_0^t g(s)ds\right)\|A^{1/2}\omega(t)\|^2\right)\\
&+\frac12\int_0^t g(t-s)\|A^{1/2}\omega(t)-A^{1/2}\omega(s)\|^2ds.
\end{align*}
Using the standard identity for the memory term, we get
\begin{align*}
E_\omega'(t)&+\left(\|u_t\|^\rho u_t-\|v_t\|^\rho v_t,\omega_t\right)\\
&+\frac12g(t)\|A^{1/2}\omega(t)\|^2
-\frac12\int_0^t g'(t-s)\|A^{1/2}\omega(t)-A^{1/2}\omega(s)\|^2ds=0.
\end{align*}
Equivalently,
\begin{align*}
 E_\omega'(t)
=
&-\left(\|u_t\|^\rho u_t-\|v_t\|^\rho v_t,\omega_t\right)\\
&-\frac12g(t)\|A^{1/2}\omega(t)\|^2
+\frac12\int_0^t g'(t-s)\|A^{1/2}\omega(t)-A^{1/2}\omega(s)\|^2ds.
\end{align*}
By Lemma \ref{continuity}, the map \(l(v)=\|v\|^\rho v\) is monotone for \(\rho>-1\). Hence
\[
\left(\|u_t\|^\rho u_t-\|v_t\|^\rho v_t,\omega_t\right)
=(l(u_t)-l(v_t),u_t-v_t)\ge0.
\]
Since \(g\ge0\) and \(g'\le0\), we deduce that \(E_\omega'(t)\le0\). But \(E_\omega(0)=0\) and \(E_\omega(t)\ge0\), so \(E_\omega(t)=0\) for all \(t\ge0\). Therefore \(\omega\equiv0\), and uniqueness follows for every \(\rho>-1\).

\]
This completes the proof for \(-1<\rho<0\).

\int_{t_1}^{t_2}E(s)ds
&\leq\frac32\int_{t_1}^{t_2}\|u_s(s)\|^2ds+\sum\limits_{i=1}^{2}J_i+\frac2k\int_{t}^{t+1}(-E'(s))ds\\
&\leq\frac32\int_{t_1}^{t_2}\|u_s(s)\|^2ds+\sum\limits_{i=1}^{2}J_i+\frac2k[D(t)]^2,
\end{align*}
that is,
\begin{equation}\label{mid}
\int_{t_1}^{t_2}E(s)ds\leq\frac32\int_{t_1}^{t_2}\|u_s(s)\|^2ds+\sum\limits_{i=1}^{2}J_i+\frac2k[D(t)]^2.
\end{equation}

Since \(t\leq t_1<t_2\leq t+1\), by \eqref{klll},
\begin{equation}\label{klkl}
\int_{t_1}^{t_2}\|u_s(s)\|^2ds\leq\int_{t}^{t+1}\|u_s(s)\|^2ds\leq[D(t)]^{\frac{4}{\rho+2}}.
\end{equation}
Using \eqref{embed} with \(s_1=1\) and \(s_2=0\), we have \(\|u(t_i)\|\leq C_1\|A^{\frac12}u(t_i)\|\) with \(C_1:=\lambda_1^{-1/2}\). Hence, by \eqref{mu_0}, \eqref{energy}, and \(E(t)\geq E(t_i)\) for \(t_i\geq t\),
\[
\|u(t_i)\|\leq C_1\left\|A^{\frac12}u(t_i)\right\|\leq C_1\sqrt{\frac{2}{\mu_0}E(t_i)}\leq C_1\sqrt{\frac{2}{\mu_0}E(t)}.
\]
Together with \eqref{mean}, this gives
\[
J_1^2\leq2\left(\|u(t_1)\|^2\|u_t(t_1)\|^2+\|u(t_2)\|^2\|u_t(t_2)\|^2\right)\leq32C_1^2\mu_0^{-1}E(t)[D(t)]^{\frac{4}{\rho+2}},
\]
and hence, by Young's inequality,
\[
J_1\leq\epsilon E(t)+\frac{8C_1^2}{\epsilon\mu_0}[D(t)]^{\frac{4}{\rho+2}}
\]
for any \(\epsilon>0\).

By \eqref{energy} and \(E'(t)\leq0\), we have \(\|u_t\|^2\leq2E(0)\). Let \(C_2:=(2E(0))^{\frac{\rho}{2}}\). By H\"older's inequality, \eqref{klkl}, and Young's inequality, we obtain
\begin{align*}
J_2
&\leq C_2\int_{t_1}^{t_2}\|u(s)\|\|u_s(s)\|ds
\leq C_1C_2\int_{t_1}^{t_2}\|A^{\frac12}u(s)\|\|u_s(s)\|ds\\
&\leq C_1C_2\sqrt{2}\mu_0^{-\frac12}[E(t)]^{\frac12}\int_{t_1}^{t_2}\|u_s(s)\|ds
\leq C_1C_2\sqrt{2}\mu_0^{-\frac12}[E(t)]^{\frac12}\left(\int_{t_1}^{t_2}\|u_s(s)\|^2ds\right)^{\frac12}\\
&\leq C_1C_2\sqrt{2}\mu_0^{-\frac12}[E(t)]^{\frac12}[D(t)]^{\frac{2}{\rho+2}}
\leq\epsilon E(t)+\frac{C_1^2C_2^2}{2\epsilon\mu_0}[D(t)]^{\frac{4}{\rho+2}}
\end{align*}
for any \(\epsilon>0\).

Combining the estimates for \(J_1\) and \(J_2\) with \eqref{klkl} and \eqref{mid}, we get
\[
\int_{t_1}^{t_2}E(s)ds\leq2\epsilon E(t)+\left(\frac{8C_1^2}{\epsilon\mu_0}+\frac{C_1^2C_2^2}{2\epsilon\mu_0}+\frac32\right)[D(t)]^{\frac{4}{\rho+2}}+\frac2k[D(t)]^2.
\]
Since \([D(t)]^2\leq E(t)\) and \(E'(t)\leq0\), we have
\begin{equation}\label{lko}
[D(t)]^2=[D(t)]^{\frac{4}{\rho+2}}[D(t)]^{\frac{2\rho}{\rho+2}}\leq E(0)^{\frac{\rho}{\rho+2}}[D(t)]^{\frac{4}{\rho+2}}.
\end{equation}
Thus
\begin{equation}\label{next}
\int_{t_1}^{t_2}E(s)ds\leq2\epsilon E(t)+\mu_1[D(t)]^{\frac{4}{\rho+2}}
\end{equation}
for any \(\epsilon>0\), where
\[
\mu_1:=\frac{8C_1^2}{\epsilon\mu_0}+\frac{C_1^2C_2^2}{2\epsilon\mu_0}+\frac32+\frac2kE(0)^{\frac{\rho}{\rho+2}}.
\]
Since \(E'(t)\leq0\), the mean value theorem gives some \(\beta\in[t_1,t_2]\subset[t,t+1]\) such that
\begin{align}
\int_{t_1}^{t_2}E(s)ds=E(\beta)(t_2-t_1)\geq\frac12E(t+1).\label{most}
\end{align}
Combining \eqref{nba}, \eqref{next}, \eqref{most}, and \eqref{lko}, we obtain
\begin{align*}
E(t)
&=[D(t)]^2+E(t+1)\leq[D(t)]^2+2\int_{t_1}^{t_2}E(s)ds\\
&\leq[D(t)]^2+2\mu_1[D(t)]^{\frac{4}{\rho+2}}+4\epsilon E(t)\\
&\leq\left[E(0)^{\frac{\rho}{\rho+2}}+2\mu_1\right][D(t)]^{\frac{4}{\rho+2}}+4\epsilon E(t).
\end{align*}
Taking \(\epsilon=\frac18\), we get
\[
E(t)\leq2\left[E(0)^{\frac{\rho}{\rho+2}}+2\mu_1\right][D(t)]^{\frac{4}{\rho+2}},
\]
and hence
\[
[E(t)]^{1+\frac{\rho}{2}}\leq\mu[D(t)]^2=\mu[E(t)-E(t+1)],
\]
where
\[
\mu:=\left[2E(0)^{\frac{\rho}{\rho+2}}+4\mu_1\right]^{1+\frac{\rho}{2}}.
\]
Since \(E'(t)\leq0\),
\[
\sup_{t\leq s\leq t+1}[E(t)]^{1+\frac{\rho}{2}}\leq[E(t)]^{1+\frac{\rho}{2}}\leq\mu[E(t)-E(t+1)].
\]
Applying Lemma \ref{lemmp}, we obtain the desired decay estimates. This completes the proof for \(\rho\geq0\).
