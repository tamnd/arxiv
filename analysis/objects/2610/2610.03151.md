\label{thMain}
           Let $G$ be a finite group, $x \in G$, and $K \leqslant G$.
           Then
           $$ |N_G(x) \cap K| \qquad \mbox{divides} \qquad |\mathop{\mathrm{Sol_G}}(x) \cap K| . $$

<!-- theorem-2: body_md -->
\label{thHall}
Let $G$ be a finite group, $x\in G$, and $\ell \geq 1$. Then
\[
|C_G(x)| \qquad \mbox{divides} \qquad |\mathrm{Sol}_{G,\ell}(x)|.
\]
<!-- end theorem-2 -->
<!-- lemma-1: body_md -->
\label{lemCL}
        If a finite group $H$ acts on a finite set $A$, then the cardinality of the set
        $$ \{ ({a,h}) \in A \times H : a^h = a \} $$
        is a multiple of $|H|$. The ratio is the number of $H$-orbits on $A$.
<!-- end lemma-1 -->
<!-- theorem-3: body_md -->
\label{thHR}
         Let $G$ be a finite group and $x \in G$. Then $|C_G(x)|$ divides $|\mathop{\mathrm{Sol}}_G(x)|$.
<!-- end theorem-3 -->
<!-- proof-1: body_md -->
We work by induction on $|G|$,
           and let $Z=Z(G)$ be the center of $G$.
         It is easy to see that $\frac{C_G(x)}{Z} \leqslant C_{G/Z}(Zx)$, and $\frac{\mathop{\mathrm{Sol}}_G(x)}{Z} = \mathop{\mathrm{Sol}}_{G/Z}(Zx)$.
         Now $|C_G(x)|$ divides $|\mathop{\mathrm{Sol}}_G(x)|$ if and only if $\frac{|C_G(x)|}{|Z|}$ divides $\frac{|\mathop{\mathrm{Sol}}_G(x)|}{|Z|}$,
         and so, by induction, we can assume $Z=1$.
         
         Let $A=\mathop{\mathrm{Sol}}_G(x)$, $H=C_G(x)$, and consider the action of $H$ on $A$ by conjugation.
        By Cauchy--Frobenius counting lemma, the quantity
        $$ \sum_{h \in H} |C_A(h)| $$
        is a multiple of $|H|$.
          For any $h \in H$ we have $x \in C_G(h)$, and
          $$ C_A(h) = C_G(h) \cap \mathop{\mathrm{Sol_G}}(x) = \mathrm{Sol}_{C_G(h)}(x) . $$
          We write
         $$ \sum_{h \in H} |C_A(h)| = |A| + \sum_{h \in H \setminus 1} |\mathrm{Sol}_{C_G(h)}(x)| , $$
          so it is sufficient to prove that $\sum_{h \in H \setminus 1} |\mathrm{Sol}_{C_G(h)}(x)|$ is a multiple of $|H|$.
          
          Let $h \in H \setminus 1$. 
          Since $h$ is noncentral in $G$, by induction
          we have $|\mathrm{Sol}_{C_G(h)}(x)| = n_h |C_{C_G(h)}(x)|$ for some integer $n_h$.
          We write
          $$ \sum_{h \in H \setminus 1} |\mathrm{Sol}_{C_G(h)}(x)| = \sum_{h \in H \setminus 1} n_h \> |C_{C_G(h)}(x)| . $$
          It is easy to see that $n_h = n_{h^g}$ for each $g \in H$.
          If $1,h_2,...,h_s \in H$ are representatives of the $H$-orbits on $H$ by conjugation, we have
          \begin{equation} \label{eq:3}
          \sum_{h \in H \setminus 1} |\mathrm{Sol}_{C_G(h)}(x)| = \sum_{i=2}^s n_{h_i} |(h_i)^H| |C_{C_G(h_i)}(x)| .
          \end{equation}
         Note that $C_{C_G(h_i)}(x) = C_G(h_i) \cap C_G(x) = C_H(h_i)$.
         By the orbit-stabilizer lemma, the right side of (\ref{eq:3}) is a multiple of $|H|$ as desired.

<!-- proof-2: body_md -->
Let $F=F(X,Y)$ be the free group on $X$ and $Y$,
and set $A:=\mathrm{Sol}_{G,\ell}(x)$.
For each $y \in G\setminus A$, there is a word $w_y\in F^{(\ell)}$
such that $w_y(x,y)\neq 1$.
It is easy to check that $A$ is the set of solutions in $G$ of the system of equations
\[
w_y(x,Y)=1,
\qquad y\in G\setminus A.
\]
Since $F^{(\ell)} \subseteq F^{(1)}$, the sum of the exponents of $Y$ in each equation is
zero. By Hall's theorem \cite[Theorem~II]{Hal36}, $|A|$ is divisible by the order of the centralizer of all the
coefficients occurring in the system.
All coefficients belong to $\langle x\rangle$, so their centralizer
contains $C_G(x)$, and the proof follows.

<!-- remark-1: body_md -->
Since $N_G(x) \subseteq \mathop{\mathrm{Sol}}_{G,2}(x)$,
looking at Theorem~\ref{thHall}, a tempting approach for the normalizer-solubilizer conjecture would be to first show that
          $|N_G(x)|$ divides $|\mathop{\mathrm{Sol}}_{G,2}(x)|$, and then that $|\mathop{\mathrm{Sol}}_{G,\ell}(x)|$ divides $|\mathop{\mathrm{Sol}}_{G,\ell+1}(x)|$ for all $\ell \geq 2$.
          However, there are examples in $\mathop{\mathrm{Alt}}(6)$ where $|N_G(x)|$ does not divide $|\mathop{\mathrm{Sol}}_{G,2}(x)|$ (but still divides $|\mathop{\mathrm{Sol}}_G(x)|$).

<!-- proof-3: body_md -->
We set
\[
H:=N_G(x)\cap K,
\qquad
A:=\mathop{\mathrm{Sol_G}}(x)\cap K,
\]
and argue by induction on $|K|$.
Since $N_G(x)$ normalizes $\mathop{\mathrm{Sol}}_G(x)$, the group $H$ acts on $A$ by conjugation.

Let $T \unlhd H$ be maximal with respect to inclusion with the properties that
\[
T\unlhd K \qquad \mbox{and} \qquad
\langle x,a,T\rangle \text{ is soluble for every } a\in A.
\]

We claim that
\begin{equation} \label{eqClaim}
    (H/T)\cap Z(K/T)=1.
\end{equation}
Let $D$ be the inverse image of \((H/T)\cap Z(K/T)\) in $H$.
Then $D\unlhd K$. Suppose that $D>T$. Fix $a\in A$ and put
\[
U:=\langle x,a,T\rangle ,
\]
so that $U$ is soluble by the choice of $T$.
Moreover,
\[
U\unlhd UD=\langle x,a,D\rangle
\]
and $UD/U$ is abelian. Thus $\langle x,a,D\rangle$ is soluble for
every $a\in A$, contradicting the maximality of $T$.
This proves (\ref{eqClaim}).

For every $a\in A$ and $t\in T$ we have that
\(
\langle x,at\rangle \leqslant \langle x,a,T\rangle
\) is soluble,
and therefore $at\in A$. Hence $A$ is a union of $T$-cosets.
Write $H:=H/T$, and $A:=A/T$ for the set of $T$-cosets contained in $A$.
Note that
\begin{equation} \label{eqQuot}
    |\overline H|=\frac{|H|}{|T|}
\qquad\text{and}\qquad
|\overline A|=\frac{|A|}{|T|}.
\end{equation}

Let $1\neq\overline h\in\overline H$, and let $L$ be the inverse image
in $K$ of $C_{K/T}(\overline h)$. By (\ref{eqClaim}), $L<K$.
Moreover,
\[
C_{\overline H}(\overline h)
   =\frac{H\cap L}{T},
\qquad
C_{\overline A}(\overline h)
   =\frac{A\cap L}{T}.
\]

By induction applied to $L$,
$|N_G(x)\cap L|$ divides $|\mathop{\mathrm{Sol_G}}(x)\cap L|$.
Since $L \subseteq K$, this says that
$|H\cap L|$ divides $|A\cap L|$, and we obtain
\begin{equation} \label{eqInd}
    |C_{\overline H}(\overline h)|
   \qquad \mbox{divides} \qquad
|C_{\overline A}(\overline h)|.
\end{equation}

By Cauchy--Frobenius counting lemma applied to the action of $\overline H$ on
$\overline A$ by conjugation, we have that
\begin{equation} \label{eqBL}
    \sum_{\overline h\in\overline H}
|C_{\overline A}(\overline h)|
\qquad \mbox{is a multiple of} \qquad |\overline H| .
\end{equation}
By the orbit-stabilizer lemma, for every nonidentity conjugacy class
$\overline h^{\,\overline H}$, the contribution to this sum is
\[
|\overline h^{\,\overline H}|\,
|C_{\overline A}(\overline h)|
=
\frac{|\overline H|}
     {|C_{\overline H}(\overline h)|}
|C_{\overline A}(\overline h)|,
\]
which is a multiple of $|\overline H|$ by (\ref{eqInd}).
Therefore all nonidentity conjugacy classes contribute a multiple of
$|\overline H|$ to (\ref{eqBL}). Hence the identity contribution
does as well, meaning precisely that
$|\overline H|$ divides $|\overline A|$.
The proof follows by (\ref{eqQuot}).
