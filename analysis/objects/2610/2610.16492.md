<!-- theorem-1: body_md -->
\label{thm:main}
	If $A=A^T\in\mathbb R^{n\times n}$, $s=2$ and $d(A,v_0)\ge3$, then the even-index sequence $\{v_{2k}\}$ in \eqref{eqn:a1}--\eqref{eqn:a2} converges to a single limit vector. Hence also the odd-index sequence $\{v_{2k+1}\}$ converges to a single limit vector.
<!-- end theorem-1 -->
<!-- lemma-1: body_md -->
\label{lem:polys}
The polynomials $p_k$ in \eqref{eqn:pk-Hk} have uniformly bounded coefficients, and there is a constant $C>0$ such that
\begin{equation}\label{eq:variation}
 \|p_{k+2}-p_k\|\le C(H_{k+1}-H_k),\quad k\ge0.
\end{equation}
Both sequences $\{p_{2k}\}$ and $\{p_{2k+1}\}$ converge.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Write $p_k(z)=z^2+b_kz+a_k$. The orthogonality conditions \eqref{eqn:arn-proj} for $p_k$ are
\begin{equation}\label{eq:orthogonality}
 \langle p_k,1\rangle_k=\langle p_k,z\rangle_k=0.
\end{equation}
They are equivalent to the linear system
\[
 G_k\begin{bmatrix}a_k\\b_k\end{bmatrix}
 =-\begin{bmatrix}\langle z^2,1\rangle_k\\\langle z^3,1\rangle_k\end{bmatrix},
 \quad
 G_k:=\begin{bmatrix}
 \langle 1,1\rangle_k&\langle z,1\rangle_k\\
 \langle z,1\rangle_k&\langle z,z\rangle_k
 \end{bmatrix}.
\]
Here $G_k$ is the Gram matrix of the two vectors $v_k, Av_k$. Since every vector of the sequence \eqref{eqn:a1}--\eqref{eqn:a2} has grade at least three, $G_k$ is positive definite. Moreover, every limit vector also has grade at least three, and hence compactness gives a uniform bound $G_k\ge cI_2$ for some $c>0$. Also note that the right-hand side of the system above is uniformly bounded, so that the coefficients $a_k,b_k$ are uniformly bounded as well. For any polynomial $f(z)=bz+a$ we have 
 
$$\|f\|_k^2=\langle f,f\rangle_k=b^2\langle z,z\rangle_k+2ab\langle z,1\rangle_k+a^2\langle 1,1\rangle_k=[a,b]G_k[a,b]^T.$$ 
 
Therefore,
 
\begin{equation}\label{eq:linear-lower-bound}
 c\|f\|^2\leq \|f\|_k^2,
\end{equation}
for every polynomial $f$ of degree at most one and every $k\geq 0$, which we will use below.

Since $p_{k+1}-p_k$ has degree at most one, orthogonality gives
\begin{equation}\label{eqn:qk-1}
 \langle q_k,1\rangle_k=\langle p_k,p_{k+1}\rangle_k
 =\langle p_k,p_k\rangle_k=H_k.
\end{equation}
Equation~\eqref{eqn:ip2} with $f=g=1$ yields $\langle q_k,q_k\rangle_k=H_kH_{k+1}$. Define $r_k:=q_k-H_k$. Then
\begin{equation}\label{eq:energy}
 \|r_k\|_k^2
 =\langle q_k,q_k\rangle_k-2H_k\langle q_k,1\rangle_k+H_k^2
 =H_k(H_{k+1}-H_k).
\end{equation}
 
For any polynomial $f$ of degree at most one, \eqref{eqn:ip1} and
orthogonality imply that
\[
 \langle p_kf,q_k\rangle_k
 =\langle p_kf,p_kp_{k+1}\rangle_k
 =H_k\langle f,p_{k+1}\rangle_{k+1}=0,
\]
and hence
\begin{equation}\label{eq:double-orthogonality}
 \langle p_kf,r_k\rangle_k
 =\langle p_kf,q_k\rangle_k-H_k\langle p_k,f\rangle_k=0.
\end{equation}
Define $D_k:=p_{k+2}-p_k$, which has degree at most one. We can assume that $D_k\neq 0$, because for $D_k=0$ the bound \eqref{eq:variation} is trivial. Orthogonality and \eqref{eqn:ip2} yield
\[
 \langle D_k,f\rangle_{k+2}
 =-\langle p_k,f\rangle_{k+2}
 =-\frac{\langle p_kf,q_k^2\rangle_k}{H_kH_{k+1}}.
\]
Using $q_k^2=(r_k+H_k)^2=r_k^2+2r_kH_k+H_k^2$, as well as \eqref{eq:orthogonality} and
\eqref{eq:double-orthogonality} shows that
\begin{equation}\label{eq:factor-difference}
 \langle D_k,f\rangle_{k+2}
 =-\frac{\langle p_kf,r_k^2\rangle_k}{H_kH_{k+1}}.
\end{equation}
Considering the term in the numerator on the right-hand side with $f=D_k$, an evaluation on the fixed spectrum of $A$ gives
\[
 |\langle p_kD_k,r_k^2\rangle_k|
 \le\max_{\lambda\in\operatorname{spec}(A)}
             |p_k(\lambda)D_k(\lambda)|\,\|r_k\|_k^2
 \le C\|D_k\|\|r_k\|_k^2,
\]
where the last inequality uses the uniform coefficient bound for $p_k$. Combining  \eqref{eq:linear-lower-bound}, \eqref{eq:energy}, and \eqref{eq:factor-difference}
we obtain
\[
 c\|D_k\|^2
 \le\|D_k\|_{k+2}^2
 =\frac{|\langle p_kD_k,r_k^2\rangle_k|}{H_kH_{k+1}}
 \le C\|D_k\|(H_{k+1}-H_k).
\]
Here $H_{k+1}\ge H_0>0$ has been absorbed into the constant. Division by $\|D_k\|$ now proves \eqref{eq:variation}. Summing \eqref{eq:variation} gives
\[
\sum_{k=0}^{\infty}\|p_{k+2}-p_k\|
\leq C\sum_{k=0}^{\infty}(H_{k+1}-H_k)
= C(H-H_0)<\infty.
\]
Thus, for each $r\in\{0,1\}$, the series
\[
p_r+\sum_{j=0}^{\infty}
\bigl(p_{2j+r+2}-p_{2j+r}\bigr)
\]
converges absolutely in the coefficient norm. Its partial sums are
$p_{2m+r}$, so both the even- and the odd-index polynomial sequences
converge.

<!-- lemma-3: body_md -->
\label{lem:exterior-decay}
Assume that $|S|=4$ and that $\Sigma_A^*$ contains a vector of grade 4. Then there are constants $C>0$ and $0<\vartheta<1$ such that, for all sufficiently large $k$,
\begin{equation}\label{eq:exterior-decay}
 \varepsilon_k:=\sum_{\mu_j\notin S}\nu_{k,j}^2
 \le C\vartheta^k.
\end{equation}
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
If $v_k$ has no exterior eigenvalues, then $\varepsilon_k=0$ and the assertion is immediate. Otherwise, suppose that $v_k$ is as in \eqref{eq:spectral-coordinates} with $\nu_{k,j}\neq 0$ for some exterior eigenvalue $\mu_j\notin S$. By construction,
 
\begin{equation}\label{eq:component}
 \nu_{k+2,j}
 =\frac{q_k(\mu_j)}{\sqrt{H_kH_{k+1}}}\nu_{k,j}.
\end{equation}
 
We know from fact (F1) that $\sqrt{H_k H_{k+1}}\uparrow H$, and from \eqref{eq:product-limit} that $q_k(\mu_j)\rightarrow Q_*(\mu_j)$. If $|Q_*(\mu_j)|>H$, then the absolute value of the multiplier in \eqref{eq:component} will eventually (for sufficiently large $k$) become larger than $1$, contradicting that the exterior coefficients converge to zero. Thus, we must have $|Q_*(\mu_j)|\leq H$. The equality $Q_*(\mu_j)=H$ is excluded by the definition of $S$ (since $\mu_j$ is an exterior eigenvalue). We will now show that also $Q_*(\mu_j)\neq -H$, which implies $|Q_*(\mu_j)|<H$.  

Using the polynomial $\Pi$ as in \eqref{eq:four-factor} we consider the standard Lagrange basis polynomials
 
$$L_i(z):=\frac{\Pi(z)}{(z-\lambda_i)\Pi'(\lambda_i)},\quad i=1,\dots ,4,$$
 
which give $g(z)=\sum_{i=1}^4 g(\lambda_i)L_i(z)$ for every polynomial $g$ of degree at most $3$. Thus, 
 
\begin{equation}\label{eq:lagrange-exterior}
g(\mu_j)=\sum_{i=1}^4 g(\lambda_i) \beta_{i,j},\quad 
\beta_{i,j}=\frac{\Pi(\mu_j)}{(\mu_j-\lambda_i)\Pi'(\lambda_i)},\;\; i=1,\dots,4,\quad \sum_{i=1}^4\beta_{i,j}=1,
\end{equation}
 
where the last equality follows from taking $g(z)=1$.

As described above, for sufficiently large $k$ all coefficients of $v_k$ in its expansion \eqref{eq:spectral-coordinates} are assumed nonzero, since we can consider the iteration after the last ``deactivation'' of an active eigenvalue. For any exterior coefficient $\nu_{k,j}$ we may therefore define
 
$$Y_{j,k}:=\log\nu_{k,j}^2
       +\sum_{i=1}^4\beta_{i,j}\log\nu_{k,i}^2,$$
 
where $\nu_{k,i}$, $i=1,\dots,4$, are the coefficients corresponding to the active eigenvalues in $S$. Taking logarithms of the squared recurrence \eqref{eq:component} gives
 
$$\frac{\log\nu_{k+2,j}^2-\log\nu_{k,j}^2}{2}
=\log|q_k(\mu_j)|-\frac12(\log H_k+\log H_{k+1}).$$
 
The same relation holds for each coefficient corresponding to $\lambda_i$.
Substitution into the definition of $Y_{j,k}$ therefore yields
\begin{align*}
\frac{Y_{j,k+2}-Y_{j,k}}2
={}&\log|q_k(\mu_j)|
+\sum_{i=1}^4\beta_{i,j}\log|q_k(\lambda_i)|
-\frac{1+\sum_{i=1}^4\beta_{i,j}}2
 (\log H_k+\log H_{k+1}).
\end{align*}
Using $\sum_{i=1}^4\beta_{i,j}=1$ from \eqref{eq:lagrange-exterior}, and
$q_k(\lambda_i)>0$ for all sufficiently large $k$ (which holds since $q_k(\lambda_i)\to H>0$), we obtain
 
\begin{equation}\label{eq:log-difference}
\frac{Y_{j,k+2}-Y_{j,k}}2
=\log|q_k(\mu_j)|+\sum_{i=1}^4\beta_{i,j}\log q_k(\lambda_i)
-\log H_k-\log H_{k+1}.
\end{equation}
 
Write $h_k:=H-H_k$ and $e_k:=q_k-Q_*$, which has degree at most $3$. Then, by \eqref{eq:tail}, $\|e_k\|=O(h_k)$. Recall that we want to show $Q_*(\mu_j)\neq -H$. Assume the contrary, i.e., that $Q_*(\mu_j)=-H$. Then $q_k(\mu_j)=-H+e_k(\mu_j)$, and a Taylor expansion at $H>0$ gives
\begin{align*}
 \log|q_k(\mu_j)|
  &=\log H-\frac{e_k(\mu_j)}H+O(h_k^2),\\
 \log q_k(\lambda_i)
  &=\log H+\frac{e_k(\lambda_i)}H+O(h_k^2),\\
 -\log H_k-\log H_{k+1}
  &=-2\log H+\frac{h_k+h_{k+1}}H+O(h_k^2).
\end{align*}
Thus, in \eqref{eq:log-difference} the constant terms and the terms involving $e_k$ cancel, so that
\begin{equation}\label{eq:log-increase}
 \frac{Y_{j,k+2}-Y_{j,k}}2
 =\frac{h_k+h_{k+1}}H+O(h_k^2)>0
\end{equation}
for all sufficiently large $k$. Here $0<h_{k+1}\le h_k\to0$ by \eqref{eq:strict-height}, so the positive term is at least $h_k/H$ and dominates the remainder. Along any even subsequence converging to a limit vector with grade $4$, the weights $\nu_{k,i}^2$, $i=1,\dots,4$, tend to positive values, whereas $\nu_{k,j}^2\to0$. Thus $Y_{j,k}\to-\infty$ along this subsequence, which contradicts \eqref{eq:log-increase} and shows that $Q_*(\mu_j)\neq -H$.

Consequently, every remaining exterior eigenvalue $\mu_j$ satisfies $|Q_*(\mu_j)|<H$. Choose a constant $\vartheta$ such that
\[
\max_{\mu_j\notin S}\frac{|Q_*(\mu_j)|}{H}<\vartheta<1.
\]
Since the two-step multipliers in \eqref{eq:component} converge to $Q_*(\mu_j)/H$, and there are only finitely many exterior eigenvalues, there is an index $K$
such that
 
$$\frac{|q_k(\mu_j)|}{\sqrt{H_kH_{k+1}}}\le\vartheta,\quad\mbox{for every}\; 
k\ge K\;\mbox{and}\; \mu_j\notin S.$$
 
Squaring \eqref{eq:component} and summing over the exterior eigenvalues gives $\varepsilon_{k+2}\le\vartheta^2\varepsilon_k$ for $k\ge K$. Iterating separately on the two parities yields
\[
\varepsilon_{K+r+2\ell}\le\vartheta^{2\ell}\varepsilon_{K+r}
\quad\mbox{for}\; r\in\{0,1\}\;\mbox{and}\; \ell\ge0.
\]
Hence, choosing a positive constant $C$ no smaller than
\[
\max_{r\in\{0,1\}}\left\{\vartheta^{-(K+r)}\varepsilon_{K+r}\right\},
\]
we obtain $\varepsilon_k\le C\vartheta^k$ for every $k\ge K$, which proves \eqref{eq:exterior-decay}.

<!-- lemma-4: body_md -->
\label{lem1}
For all sufficiently large $k$,
\begin{equation}\label{eq:lanczos-alpha}
|\alpha_{k+1}-\alpha_k|\le C(\alpha_k^2+\vartheta^k),
\end{equation}
where $\vartheta$ is as in Lemma~~\ref{lem:exterior-decay}.
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
For each $k$, let $f_k(z)=z^3+c_kz^2+b_kz+a_k$ be defined by
\begin{equation}
 \langle f_k,1\rangle_k
 =\langle f_k,p_k\rangle_k
 =\langle f_k,zp_k\rangle_k=0.
 \label{eq:lanczos-cubic}
\end{equation}
Expanding these conditions and using the orthogonality relations \eqref{eq:orthogonality}, together with the identities $\langle 1,1\rangle=1$, $\langle z,1\rangle=\rho_k$, and $\langle z^2,p_k\rangle=H_k$, we obtain the linear algebraic system
\[
B_k\begin{bmatrix}a_k\\b_k\\c_k\end{bmatrix}
=-\begin{bmatrix}\langle z^3,1\rangle\\\langle z^3,p_k\rangle\\\langle z^3,zp_k\rangle\end{bmatrix},
\quad\mbox{where}\quad
B_k=\begin{bmatrix}
1&\rho_k&\langle z^2,1\rangle\\
0&0&H_k\\
0&H_k&\langle z^3,p_k\rangle
\end{bmatrix}.
\]
The identity $\langle z^2,p_k\rangle=H_k$ follows because $z^2-p_k$ has degree at most one and is orthogonal to $p_k$. Since $\det B_k=-H_k^2$ and $H_k\ge H_0>0$, the system has a unique solution. Its entries and right-hand side are uniformly bounded, so $B_k^{-1}$ and the coefficients of $f_k$ are uniformly bounded as well.

The polynomial
\[
g_k:=r_k-\Pi-\alpha_k f_k
\]
has degree at most two. Equations \eqref{eqn:qk-1} and \eqref{eq:double-orthogonality} give
\[
\langle r_k,1\rangle=\langle r_k,p_k\rangle=\langle r_k,zp_k\rangle=0.
\]
Since $\Pi$ vanishes on $S$ and all polynomial values involved are uniformly bounded on the fixed spectrum, we have
\[
|\langle \Pi,1\rangle|+|\langle \Pi,p_k\rangle|+|\langle \Pi,zp_k\rangle|
\le C\varepsilon_k,
\quad\|\Pi\|_k^2\le C\varepsilon_k.
\]
The defining orthogonality conditions for $f_k$ and $r_k$ imply
\[
\langle g_k,1\rangle=-\langle \Pi,1\rangle,\quad
\langle g_k,p_k\rangle=-\langle \Pi,p_k\rangle,\quad
\langle g_k,zp_k\rangle=-\langle \Pi,zp_k\rangle.
\]
Writing $g_k(z)=g_{k,0}+g_{k,1}z+g_{k,2}z^2$, these relations form a system with the same coefficient matrix $B_k$. Its right-hand side is $O(\varepsilon_k)$, so $\|g_k\|\le C\varepsilon_k$ in the coefficient norm. Using $r_k=\Pi+\alpha_k f_k+g_k$, the uniform bounds above, and $\varepsilon_k\le1$, equation \eqref{eq:energy} now gives
 
\begin{equation}
 H_{k+1}-H_k=\frac{\|r_k\|_k^2}{H_k}
 \le C(\alpha_k^2+\varepsilon_k).
 \label{eq:lanczos-energy-bound}
\end{equation}
 
Write $p_k(z)=z^2+\beta_kz+\gamma_k$. Since
$q_{k+1}-q_k=p_{k+1}(p_{k+2}-p_k)$, comparison of the cubic coefficients and Lemma~\ref{lem:polys} yield
\[
|\alpha_{k+1}-\alpha_k|=|\beta_{k+2}-\beta_k|
\le C(H_{k+1}-H_k).
\]
Combining the last two inequalities with $\varepsilon_k\le C\vartheta^k$ proves the assertion.

<!-- lemma-5: body_md -->
\label{lem2}
For all sufficiently large $k$,
\begin{align}
 \rho_{k+2}-\rho_k
 &=-\kappa_k\alpha_k+E_k\quad\mbox{with}\quad \kappa_k:=\frac{2\Delta_k}{H_k^2H_{k+1}},\quad 0\leq \kappa_k\leq C,\label{eq:lanczos-rho} \\
 |E_k| &\leq C(H_{k+1}-H_k+\vartheta^k)\quad\mbox{and}\quad \sum_{k=0}^\infty |E_k|<\infty,
\label{eq:lanczos-rho1}
\end{align}
and where \(\vartheta\) is as in Lemma~\ref{lem:exterior-decay}.
<!-- end lemma-5 -->
<!-- proof-4: body_md -->
Let $f_k$ and $g_k$ be the auxiliary polynomials from the proof of Lemma~\ref{lem1}. Write $p_k(z)=z^2+\beta_kz+\gamma_k$ and use the triangular matrices
\[
L_k=\begin{bmatrix}
1&0&0&0\\0&1&0&0\\\gamma_k&\beta_k&1&0\\0&\gamma_k&\beta_k&1
\end{bmatrix},\quad
R_k=\begin{bmatrix}
1&0&0&a_k\\0&1&0&b_k\\0&0&1&c_k\\0&0&0&1
\end{bmatrix}.
\]
Left multiplication by $L_k$ replaces the row polynomials
$(1,z,z^2,z^3)$ by $(1,z,p_k,zp_k)$, and right multiplication by $R_k$ replaces the column polynomials by $(1,z,z^2,f_k)$. Both matrices have determinant one. The last column of $L_kG_kR_k$ is
$[0,\langle z,f_k\rangle,0,0]^T$. Expansion along that column gives
\begin{equation}
\Delta_k =\langle z,f_k\rangle_k \det(B_k)=-H_k^2\langle z,f_k\rangle_k.
 \label{eq:lanczos-determinant}
\end{equation}
 
This identity also holds when the grade is three.

Since $\langle z,\Pi\rangle=O(\varepsilon_k)$ and $\|g_k\|\le C\varepsilon_k$, the decomposition $r_k=\Pi+\alpha_k f_k+g_k$ gives
 
\begin{equation}
 \langle z,r_k\rangle_k
 =-\frac{\Delta_k}{H_k^2}\alpha_k+O(\vartheta^k).
 \label{eq:lanczos-first-moment}
\end{equation}
 
Equations \eqref{eqn:ip2} and \eqref{eq:energy}, together with $q_k=H_k+r_k$, yield the exact identity
\[
\rho_{k+2}-\rho_k
=\frac{2\langle z,r_k\rangle}{H_{k+1}}
+\frac{\langle z-\rho_k,r_k^2\rangle}{H_kH_{k+1}}.
\]
The first term equals $-\kappa_k\alpha_k+O(\vartheta^k)$.
Using $|\rho_k|\le\|A\|$ and \eqref{eq:energy}, the second term is bounded in absolute value by
\[
\frac{2\|A\|\|r_k\|_k^2}{H_kH_{k+1}}
=\frac{2\|A\|}{H_{k+1}}(H_{k+1}-H_k)
\le C(H_{k+1}-H_k).
\]
This proves the recurrence and the stated bound for $E_k$. Since
$H_k\uparrow H$ and $0<\vartheta<1$, telescoping the height increments and summing the geometric series give $\sum_k|E_k|<\infty$.

<!-- lemma-6: body_md -->
For any fixed $\eta\in(\vartheta,1)$, where \(\vartheta\) is as in Lemma~\ref{lem:exterior-decay}, exactly one of the following
alternatives holds:
\begin{enumerate}
\item[(i)] $|\alpha_k|<\eta^k$ for all sufficiently large $k$, and in this case $\alpha_k=O(\eta^k)$.
\item[(ii)] $|\alpha_k|\ge\eta^k$ for all sufficiently large $k$, and $\alpha_k$ has a fixed nonzero sign on this tail.
\end{enumerate}
 
Moreover, the sequence $\{\rho_{2k}\}$ has finite total variation, i.e., $\sum_{k=0}^{\infty}|\rho_{2k+2}-\rho_{2k}|<\infty$, and hence it converges.
<!-- end lemma-6 -->
<!-- proof-5: body_md -->
We have $\alpha_k\to0$, and the estimate of Lemma~\ref{lem1} holds for all sufficiently large $k$. Choose $K_0$ so large that this estimate holds and that
\[
C|\alpha_k|+C(\vartheta/\eta)^k<1-\eta
\]
for all $k\geq K_0$. Such an index exists because both terms on the left tend to zero.

If $|\alpha_k|<\eta^k$ for every $k\ge K_0$, then alternative (i) holds. Otherwise, there is an index $k_0\ge K_0$ satisfying $|\alpha_{k_0}|\ge\eta^{k_0}$. At any index $k\ge K_0$ with $|\alpha_k|\ge\eta^k$, Lemma~\ref{lem1} gives
\[
\frac{|\alpha_{k+1}-\alpha_k|}{|\alpha_k|}
\le C|\alpha_k|+\frac{C\vartheta^k}{|\alpha_k|}
\le C|\alpha_k|+C(\vartheta/\eta)^k<1-\eta.
\]
Thus, $\alpha_{k+1}$ has the same sign as $\alpha_k$, and $|\alpha_{k+1}|>\eta|\alpha_k|\ge\eta^{k+1}$. Induction shows that the threshold remains satisfied and the sign remains unchanged for every $k\ge k_0$. Hence alternative (ii) holds. The two alternatives are mutually exclusive.

In order to show that the sequence $\{\rho_{2k}\}$ has finite total variation we consider any fixed $\eta\in(\vartheta,1)$. Note that finitely many initial terms do not affect the conclusion.

In alternative (i) we have $\alpha_k=O(\eta^k)$. The recurrence of Lemma~\ref{lem2},
the boundedness of $\kappa_k$, and the summability of $E_k$ imply, for
sufficiently large $J$,
\[
\sum_{k=J}^{\infty}|\rho_{2k+2}-\rho_{2k}|
\le C\sum_{k=J}^{\infty}|\alpha_{2k}|
+\sum_{k=J}^{\infty}|E_{2k}|<\infty.
\]
In alternative (ii), define
\[
\widehat\rho_k:=\rho_{2k}+\sum_{\ell=k}^{\infty}E_{2\ell}.
\]
The series is well defined by Lemma~\ref{lem2}, and $\widehat\rho_{k+1}-\widehat\rho_k=-\kappa_{2k}\alpha_{2k}$. Since $\kappa_{2k}\ge0$ and $\alpha_{2k}$ has a fixed sign eventually, $\widehat\rho_k$ is eventually monotone. The series is bounded because $|\rho_{2k}|\le\|A\|$ and the error tail tends to zero. Hence it converges and has finite total variation. Moreover,
\[
\rho_{2k+2}-\rho_{2k}
=(\widehat\rho_{k+1}-\widehat\rho_k)+E_{2k}.
\]
The summability of $E_{2k}$ therefore gives finite total variation of $\rho_{2k}$ as well, which finishes the proof.
