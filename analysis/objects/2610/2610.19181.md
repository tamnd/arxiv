<!-- theorem-1: body_md -->
\label{thm:optimal_weights_exist}
Let $\boldsymbol{\pi}^* \in \mathbb{R}^m$ be an optimal dual solution of the Lagrangian dual \eqref{eq:lagrangian_dual}.
For each group $\ell \in [r]$, let $\boldsymbol{\pi}^{*,\ell} \in
\mathbb{R}^{m_{\ell}}$ denote the subvector of $\boldsymbol{\pi}^*$ associated with that group.
Then $\mathbf{w}^* = \boldsymbol{\pi}^*$
(i.e., $\mathbf{w}_{\ell}^* = \boldsymbol{\pi}^{*,\ell}$ for each $\ell \in [r]$) is a maximizer of the generalized surrogate dual~\eqref{eq:surrogate_dual_problem}, and
\begin{equation}
\label{eq:optimal_weights_formula}
z^{SU} = z^{\text{surr}}(\boldsymbol{\pi}^*) = z^{CG}.
\end{equation}
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
Since $z^{\text{surr}}(\boldsymbol{\pi}^*) \le z^{SU} \le z^{CG}$ are immediate, it suffices to prove that $z^{\text{surr}}(\boldsymbol{\pi}^*) \ge z^{CG}$.

                      
                                          
                                                                            
Evaluating the Lagrangian relaxation~\eqref{eq:lagrangian_subproblem} at $\boldsymbol{\pi}^*$ and using
$z^{CG} = z^{LD}$ from~\eqref{eq:lagrangian_dw_equivalence} gives
\begin{equation}
\label{eq:lagrangian_at_pistar}
z^{CG}
= \sum_{\ell=1}^{r} (\boldsymbol{\pi}^{*,\ell})^\top \mathbf{b}^\ell
  + \min_{\mathbf{x} \in \text{conv}(\mathcal{X})} \Bigl\{ \mathbf{c}^\top \mathbf{x}
    - \sum_{\ell=1}^{r} (\boldsymbol{\pi}^{*,\ell})^\top A^\ell \mathbf{x} \Bigr\}.
\end{equation}
Let $\tilde{\mathbf{x}}$ be any feasible solution of~\eqref{eq:aggregated_lp_weighted}
with $\mathbf{w}_{\ell} = \boldsymbol{\pi}^{*,\ell}$.
Then $\tilde{\mathbf{x}} \in \text{conv}(\mathcal{X})$ and
$(\boldsymbol{\pi}^{*,\ell})^\top A^\ell \tilde{\mathbf{x}} = (\boldsymbol{\pi}^{*,\ell})^\top \mathbf{b}^\ell$
for all $\ell \in [r]$.
Substituting the aggregated-constraint feasibility into the Lagrangian objective:
\begin{align*}
\mathbf{c}^\top \tilde{\mathbf{x}}
&= \Bigl( \mathbf{c}^\top \tilde{\mathbf{x}} - \sum_{\ell=1}^{r} (\boldsymbol{\pi}^{*,\ell})^\top A^\ell \tilde{\mathbf{x}} \Bigr) +
\sum_{\ell=1}^{r} (\boldsymbol{\pi}^{*,\ell})^\top \mathbf{b}^\ell \\
&\ge \min_{\mathbf{x} \in \text{conv}(\mathcal{X})} \Bigl\{ \mathbf{c}^\top x - \sum_{\ell=1}^{r} (\boldsymbol{\pi}^{*,\ell})^\top
A^\ell \mathbf{x} \Bigr\} + \sum_{\ell=1}^{r} (\boldsymbol{\pi}^{*,\ell})^\top \mathbf{b}^\ell
= z^{CG}.
\end{align*}
Since every feasible solution of~\eqref{eq:aggregated_lp_weighted} with weights $\boldsymbol{\pi}^*$
has objective value at least $z^{CG}$, taking the minimum gives $z^{\text{surr}}(\boldsymbol{\pi}^*) \ge z^{CG}$.
Combined with $z^{\text{surr}}(\boldsymbol{\pi}^*) \leq z^{SU} \leq z^{CG}$, this implies
\[
z^{\text{surr}}(\boldsymbol{\pi}^*) = z^{SU} = z^{CG},
\]
which completes the proof.

<!-- example-1: body_md -->
\label{ex:surrogate_nonconcave}
Consider a three-scenario problem with first-stage copies $x_1, x_2, x_3$,
consensus variable $\bar{x}$, and second-stage recourse variables $y_1, y_2, y_3$ for
scenarios~1-3.
The NACs are $x_j = \bar{x}$, $j=1,2,3$:
\begin{equation}
\label{eq:lp}
\begin{aligned}
&\max && x_1\\
&\text{s.t.} && x_j = \bar{x}, \quad j=1,2,3 \\
&&& x_1 + y_1 \le 14,
\ x_2+y_2 \le 12, \ x_3 + y_3 \le 10 ,\\
&&& x_j \geq 0, y_j \geq 0, \quad j=1,2,3 
\end{aligned}
\end{equation}
The surrogate relaxation with weights $\mathbf{w} = (w_1, w_2,w_3)$ is:
\begin{equation}
\label{eq:inner-lp}
\begin{aligned}
z^{\text{surr}}(\mathbf{w})\;:=\;&\max && x_1\\
&\text{s.t.} && w_1 x_1 + w_2 x_2 + w_3 x_3 \;=\; (w_1 + w_2 + w_3)\bar{x}, \,\\
&&& x_1 + y_1 \le 14,
\ x_2+y_2 \le 12, \ x_3 + y_3 \le 10 ,\\
&&& x_j \geq 0, y_j \geq 0, \quad j=1,2,3 .
\end{aligned}
\end{equation}
Take \(\mathbf{w}^{(1)}=(1,-1,0)\) and \(\mathbf{w}^{(2)}=(0,1,-1)\), with midpoint
\(\bar{\mathbf{w}}=\tfrac12(\mathbf{w}^{(1)}+\mathbf{w}^{(2)})=(1/2,0,-1/2)\).
Evaluating \eqref{eq:inner-lp} at each gives
\[
z^{\text{surr}}(\mathbf{w}^{(1)})=12,\qquad
z^{\text{surr}}(\mathbf{w}^{(2)})=14,\qquad
z^{\text{surr}}(\bar{\mathbf{w}})=10.
\]
Since \(z^{\text{surr}}(\bar{\mathbf{w}})=10 < \min\{z^{\text{surr}}(\mathbf{w}^{(1)}),z^{\text{surr}}(\mathbf{w}^{(2)})\}= 12 \),
the function \(z^{\text{surr}}(\cdot)\) is not quasiconcave.

<!-- theorem-2: body_md -->
\label{thm:quasigradient_convergence}
Let $\mathcal{X}$ be compact
and let the step-size sequence $\{\alpha_t\}_{t=0}^\infty$ satisfy the Robbins--Monro conditions:
\begin{equation}
\label{eq:step_size_conditions}
\sum_{t=0}^\infty \alpha_t = \infty
\quad \text{and} \quad
\sum_{t=0}^\infty \alpha_t^2 < \infty.
\end{equation}
Then the sequence of bounds $\{z^{\text{surr}}(\mathbf{w}^{(t)})\}_{t=0}^\infty$
produced by Algorithm~\ref{alg:quasigradient_weighted} with the stopping conditions removed satisfies
\begin{equation}
\label{eq:convergence_result}
\liminf_{t \to \infty}  z^{\text{surr}}(\mathbf{w}^{(t)})  =  z^{CG}.
\end{equation}
<!-- end theorem-2 -->
<!-- proof-2: body_md -->
By Theorem~\ref{thm:optimal_weights_exist},
there exist optimal weights $\mathbf{w}^* = \boldsymbol{\pi}^* \in \mathbb{R}^m$
such that $z^{\text{surr}}(\mathbf{w}^*) = z^{CG}$.
Define the optimality gap at iteration $t$ as $\Delta_t \coloneqq z^{CG} - z^{\text{surr}}(\mathbf{w}^{(t)}) \ge 0$.
Since $\mathbf{x}^{(t)}$ is feasible for the aggregated formulation~\eqref{eq:aggregated_lp_weighted},
it satisfies every weighted aggregated constraint:
\begin{equation}
\label{eq:orthogonality_condition}
\mathbf{w}_{\ell}^{(t)\top} \mathbf{g}_{\ell}^{(t)} = 0,
\qquad \forall\, \ell \in [r],
\end{equation}
where $\mathbf{g}_{\ell}^{(t)} = \mathbf{b}^\ell - A^\ell \mathbf{x}^{(t)}$
is the residual vector for group $\ell$.
Summing over groups gives
\begin{equation}
\label{eq:weight_orthogonality}
\sum_{\ell=1}^r \mathbf{w}_{\ell}^{(t)\top} \mathbf{g}_{\ell}^{(t)} = 0.
\end{equation}

Since $z^{CG} = z^{LD}$ and the Lagrangian dual value at $\mathbf{w}^* = \boldsymbol{\pi}^*$ equals $z^{LD}$,
\[
z^{CG} = \sum_{\ell=1}^r \mathbf{w}_{\ell}^{*\top} \mathbf{b}^\ell + \min_{\mathbf{x} \in \mathcal{X}} \Bigl\{
\mathbf{c}^\top \mathbf{x} - \sum_{\ell=1}^r \mathbf{w}_{\ell}^{*\top} A^\ell \mathbf{x} \Bigr\}.
\]
Substituting $\mathbf{x}^{(t)} \in \mathcal{X}$ in place of the minimizer gives an upper bound:
\begin{align}
z^{CG}
&\le \sum_{\ell=1}^r \mathbf{w}_{\ell}^{*\top} \mathbf{b}^\ell + \mathbf{c}^\top \mathbf{x}^{(t)} -
\sum_{\ell=1}^r \mathbf{w}_{\ell}^{*\top} A^\ell \mathbf{x}^{(t)} \notag \\
&= z^{\text{surr}}(\mathbf{w}^{(t)}) + \sum_{\ell=1}^r \mathbf{w}_{\ell}^{*\top} \mathbf{g}_{\ell}^{(t)}. \notag 
\end{align}
Rearranging yields
\begin{align}
\Delta_t
&\le \sum_{\ell=1}^r \mathbf{w}_{\ell}^{*\top} \mathbf{g}_{\ell}^{(t)}. \label{eq:dual_gap_bound}
\end{align}

Expanding the squared distance after the weight update~\eqref{eq:weight_update_vector}:
\begin{align*}
\| \mathbf{w}^{(t+1)} - \mathbf{w}^* \|^2
&= \| \mathbf{w}^{(t)} + \alpha_t \mathbf{g}_{}^{(t)} - \mathbf{w}^* \|^2 \\
&= \| \mathbf{w}^{(t)} - \mathbf{w}^* \|^2
+ 2\alpha_t \sum_{\ell=1}^r (\mathbf{w}_{\ell}^{(t)} - \mathbf{w}_{\ell}^{*})^\top \mathbf{g}_{\ell}^{(t)}
+ \alpha_t^2 \| \mathbf{g}_{}^{(t)} \|^2.
\end{align*}
The cross term simplifies using~\eqref{eq:weight_orthogonality} and~\eqref{eq:dual_gap_bound}:
\[
\sum_{\ell=1}^r (\mathbf{w}_{\ell}^{(t)} - \mathbf{w}_{\ell}^{*})^\top \mathbf{g}_{\ell}^{(t)}
= \sum_{\ell=1}^r \mathbf{w}_{\ell}^{(t)\top} \mathbf{g}_{\ell}^{(t)}
- \sum_{\ell=1}^r \mathbf{w}_{\ell}^{*\top} \mathbf{g}_{\ell}^{(t)}
\le 0 -\Delta_t = - \Delta_t.
\]
Since $\mathcal{X}$ is compact, the residuals are uniformly bounded:
there exists $M > 0$ such that $\sum_{\ell=1}^r \| \mathbf{g}_{\ell}^{(t)} \|_2^2 \le M$ for all $t$.
Combining these bounds:
\begin{align*}
\| \mathbf{w}^{(t+1)} - \mathbf{w}^* \|^2
&\le \| \mathbf{w}^{(t)} - \mathbf{w}^* \|^2 - 2\alpha_t \Delta_t + \alpha_t^2 M, \\
\Rightarrow \quad 2\alpha_t \Delta_t
&\le \| \mathbf{w}^{(t)} - \mathbf{w}^* \|^2 - \| \mathbf{w}^{(t+1)} - \mathbf{w}^* \|^2 + \alpha_t^2 M.
\end{align*}

Summing from $t = 0$ to $T$, the distance terms telescope:
\[
2 \sum_{t=0}^T \alpha_t \Delta_t
\le \| \mathbf{w}^{(0)} - \mathbf{w}^* \|^2
   - \underbrace{\| \mathbf{w}^{(T+1)} - \mathbf{w}^* \|^2}_{\ge \, 0}
   + M \sum_{t=0}^T \alpha_t^2
\le \| \mathbf{w}^{(0)} - \mathbf{w}^* \|^2
   + M \sum_{t=0}^T \alpha_t^2.
\]
Taking $T \to \infty$, the right-hand side remains finite
by the step-size condition $\sum \alpha_t^2 < \infty$, so
\begin{equation}
\label{eq:weighted_gaps_finite}
\sum_{t=0}^\infty \alpha_t \Delta_t < \infty.
\end{equation}
Since $\Delta_t \ge 0$, $\sum \alpha_t = \infty$, and $\sum \alpha_t \Delta_t < \infty$,
we conclude $\liminf_{t \to \infty} \Delta_t = 0$:
if $\liminf \Delta_t > 0$, there would exist $\epsilon > 0$ and $T_0$ such that $\Delta_t \ge \epsilon$ for all $t \ge T_0$,
giving $\sum_{t \ge T_0} \alpha_t \Delta_t \ge \epsilon \sum_{t \ge T_0} \alpha_t = \infty$,
contradicting~\eqref{eq:weighted_gaps_finite}.
Hence $\liminf_{t \to \infty} z^{\text{surr}}(\mathbf{w}^{(t)}) = z^{CG}$.

<!-- corollary-1: body_md -->
\label{cor:inequality_constraints}
Theorem~\ref{thm:quasigradient_convergence} extends to inequality constraints
$A^\ell \mathbf{x} \le \mathbf{b}^\ell$ (or $A^\ell \mathbf{x} \ge \mathbf{b}^\ell$)
by converting them to equalities with bounded slack variables.
For each group $\ell \in [r]$ with constraints $A^\ell \mathbf{x} \le \mathbf{b}^\ell$,
introduce $\mathbf{s}^\ell \in \mathbb{R}^{m_{\ell}}$ satisfying
\begin{equation}
\label{eq:slack_bounds}
\mathbf{0} \le \mathbf{s}^\ell \le \mathbf{b}^\ell - \min_{\mathbf{x} \in \mathcal{X}} A^\ell \mathbf{x}
\end{equation}
and rewrite the constraints as $A^\ell \mathbf{x} + \mathbf{s}^\ell = \mathbf{b}^\ell$.
Compactness of $\mathcal{X}$ ensures the upper bound in~\eqref{eq:slack_bounds} is finite,
so $\mathbf{s}^\ell$ remains bounded.
The quasi-gradient becomes
$\mathbf{g}_{\ell}^{(t)} = \mathbf{b}^\ell - A^\ell \mathbf{x}^{(t)} - {\mathbf{s}^\ell}^{(t)}$,
and the analysis carries through unchanged.
<!-- end corollary-1 -->
