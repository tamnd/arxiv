\label{def:energy}
Let $T>0$ and $(u_0,u_1)\in V\times H$.
A function $u$ is called an energy solution on $[0,T]$ if
\[
u\in C([0,T];V),
\quad
u_t\in C([0,T];H)\cap L^{m(x)}(Q_T),
\]
where $u_t$ denotes the distributional time derivative
of $u$, and $u(0)=u_0\in V$, $u_t(0)=u_1\in H.$
Moreover, for every $z\in X_m$, 
\begin{equation}
\label{eq6}
\begin{split}
&\frac{d}{dt}(u_t(t),z)
+(\nabla u(t),\nabla z)
-\int_0^t g(t-s)(\nabla u(s),\nabla z)\,ds\\
&\qquad+\int_\Omega b(x,u_t(t))z\,dx
=\int_\Omega f(x,u(t))z\,dx
\end{split}
\end{equation}
in $\mathcal D'(0,T)$.

<!-- theorem-1: body_md -->
\label{thm:local}
Suppose that Assumption \ref{ass1} holds.
Given any initial pair $(u_0,u_1)\in V\times H$, there exists some $T>0$ together with an energy solution $u$ defined on $[0,T]$ for problem \eqref{eq1} satisfying the initial conditions $(u(0),u_t(0))=(u_0,u_1)$.
$T$ may be taken uniformly positive for initial data lying within bounded subsets of $V\times H$.

Each local energy solution possesses a maximal extension onto an interval $[0,T_{\max})$, with $T_{\max}\in(0,\infty]$.
For arbitrary $0<T<T_{\max}$, we have
$
f(\cdot,u)\in L^{m'(x)}(Q_T),
$
and the energy identity
\begin{equation}\label{eq7}
\begin{aligned}
E(t)
&+\int_s^t\int_\Omega
|u_\tau(x,\tau)|^{m(x)}\,dx\,d\tau+\frac12\int_s^t
g(\tau)\|\nabla u(\tau)\|_2^2\,d\tau\\
&+\frac12\int_s^t\int_0^\tau
[-g'(\tau-r)]
\|\nabla u(\tau)-\nabla u(r)\|_2^2\,dr\,d\tau
=E(s)
\end{aligned}
\end{equation}
is valid for all $0\le s\le t\le T$.

Whenever $T_{\max}<\infty$, the blow-up alternative
\begin{equation}\label{eq8}
\limsup_{t\uparrow T_{\max}}
\bigl(\|u(t)\|_V+\|u_t(t)\|_2\bigr)=\infty
\end{equation}
holds true.

\label{thm:negative}
Let Assumption~\ref{ass1} and \ref{ass2} hold. Suppose that
\[
\mathcal E(0)\le0,
\quad
(u_0,u_1)\ne(0,0).
\]
Then $T_{\max}<\infty$ and
\[
\limsup_{t\uparrow T_{\max}}
\bigl(\|u(t)\|_V+\|u_t(t)\|_2\bigr)=\infty.
\]
More precisely, the following lifespan estimates hold.
\begin{enumerate}[label=\textup{(\roman*)}]
\item
If $\mathcal E(0)<0$, set $h_0=-\mathcal E(0)$.
There exist positive constants $\varepsilon,c,A$,
chosen as in Section~\ref{sec:negative} and possibly
depending on the initial data, such that
\[
Y_0=h_0^{1-\alpha}+\varepsilon(u_0,u_1)>0
\]
and
\[
T_{\max}
\le\frac{A}{\varepsilon c(r-1)}Y_0^{1-r}.
\]

\item
If $\mathcal E(0)=0$, then
\[
\mathcal E(\tau)<0
\quad\text{for every }0<\tau<T_{\max}.
\]
For each such $\tau$, set $h_\tau=-\mathcal E(\tau)$.
There exist positive constants
$\varepsilon_\tau,c_\tau,A_\tau$, chosen as in
Section~\ref{sec:negative}, such that
\[
Y_\tau=h_\tau^{1-\alpha}
+\varepsilon_\tau(u(\tau),u_t(\tau))>0
\]
and
\[
T_{\max}
\le\tau+
\frac{A_\tau}{\varepsilon_\tau c_\tau(r-1)}
Y_\tau^{1-r}.
\]
                                                                                             
\end{enumerate}

$m_+\le2^*$ and Assumption  \ref{assadd} hold. 
Then the following decay estimates hold.

\[

\]

\begin{equation}\label{eq17}

\end{equation}

\begin{equation}\label{eq18}
E(t)\le CE(0)\Psi^{-1}\left(c\int_0^t\xi(s)\,ds\right),
\quad t\ge0.
\end{equation}
\end{enumerate}
Here $c,C>0$ are constants that may vary between estimates.
They depend only on the fixed parameters and, in case
\textup{(ii)}, may also depend on $t_*$.

<!-- lemma-1: body_md -->
\label{lem:growth}
Choose
\[
0<\sigma<\min\left\{2^*-p_+,\,
\frac{2^*}{m_-'}-(p_+-1)\right\},
\] there is a constant $C>0$,
independent of $x$ and $s$, such that
\[
\begin{aligned}
|f(x,s)|
&\le C\bigl(1+|s|^{p_+-1+\sigma}\bigr),\\
|\partial_s f(x,s)|
&\le C\bigl(1+|s|^{p_+-2+\sigma}\bigr),\\
|F(x,s)|
&\le C\bigl(1+|s|^{p_++\sigma}\bigr)
\end{aligned}
\]
for almost every $x\in\Omega$ and every $s\in\mathbb R$.
Moreover, $f(x,\cdot)\in C^1(\mathbb R)$, with
\[
\partial_s f(x,s)
=
|s|^{p(x)-2}
\bigl[1+(p(x)-1)\log|s|\bigr]
\quad\text{for }s\ne0,
\quad
\partial_s f(x,0)=0.
\]
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
By direct differentiation, one obtains 
\begin{align*}
\frac{d}{dr}(r^{-\sigma}\log r)
&=r^{-\sigma-1}(1-\sigma\log r),\quad r\ge1,\\
\frac{d}{dr}(-r^a\log r)
&=r^{a-1}(-a\log r-1),\quad 0<r\le1,\quad a>0.
\end{align*}
The derivatives change from positive to negative at $e^{1/\sigma}$ and
$e^{-1/a}$, respectively. Therefore, 
\begin{equation}
\label{eq22}
\sup_{r\ge1}r^{-\sigma}\log r=\frac1{e\sigma},
\quad
\sup_{0<r\le1}r^a|\log r|=\frac1{ea},
\quad
\lim_{r\downarrow0}r^a|\log r|=0.
\end{equation}


For $s \neq 0$,
\[
\partial_s f(x, s) = |s|^{p(x)-2}\big[1 + (p(x)-1)\log|s|\big].
\]
For $0 < |h| \leq 1$, as $h\to 0$, 
\[
\left| \frac{f(x, h) - f(x, 0)}{h} \right|
= |h|^{p(x)-2}|\log|h||
\leq |h|^{p_- -2}|\log|h|| \to 0,
\]
so $\partial_s f(x, 0) = 0$. Moreover, for $0 < |s| \leq 1$, as $s\to 0$, 
\[
|\partial_s f(x, s)| \leq |s|^{p_- -2}\big[1 + (p_+ -1)|\log|s||\big] \to 0,
\]
hence $\partial_s f(x, \cdot)$ is continuous at $0$. Therefore $f(x, \cdot) \in C^1(\mathbb R)$.



(1) For $0<|s|\le1$, by using \eqref{eq22}, we obtain 
\begin{align*}
|f(x,s)|
&=|s|^{p(x)-1}|\log|s||
\le |s|^{p_--1}|\log|s||
\le\frac1{e(p_--1)},\\
|\partial_s f(x,s)|
&\le |s|^{p(x)-2}+(p(x)-1)|s|^{p(x)-2}|\log|s||\\
&\le1+(p_+-1)|s|^{p_--2}|\log|s||
\le1+\frac{p_+-1}{e(p_--2)},\\
|F(x,s)|
&\le\frac{|s|^{p(x)}}{p(x)}|\log|s||
+\frac{|s|^{p(x)}}{p(x)^2}\\
&\le\frac{|s|^{p_-}}{p_-}|\log|s||+\frac1{p_-^2}
\le\frac1{ep_-^2}+\frac1{p_-^2}.
\end{align*}
These bounds also hold at zero by continuity. 

(2) For $|s|\ge1$, using again \eqref{eq22}, we obtain 
\begin{align*}
|f(x,s)|
&=|s|^{p(x)-1}\log|s|
\le\frac1{e\sigma}|s|^{p(x)-1+\sigma}
\le\frac1{e\sigma}|s|^{p_+-1+\sigma},\\
|\partial_s f(x,s)|
&\le |s|^{p(x)-2}+(p_+-1)|s|^{p(x)-2}\log|s|\\
&\le |s|^{p_+-2}+\frac{p_+-1}{e\sigma}|s|^{p_+-2+\sigma}
\le\left(1+\frac{p_+-1}{e\sigma}\right)|s|^{p_+-2+\sigma},\\
|F(x,s)|
&\le\frac1{p_-}|s|^{p(x)}\log|s|+\frac1{p_-^2}|s|^{p(x)}\\
&\le\frac1{e\sigma p_-}|s|^{p_++\sigma}+\frac1{p_-^2}|s|^{p_+}
\le\left(\frac1{e\sigma p_-}+\frac1{p_-^2}\right)|s|^{p_++\sigma}.
\end{align*}
This completes the proof.

<!-- lemma-2: body_md -->
\label{lem:cutoff}
Let $\vartheta\in C_c^\infty(\mathbb R)$ be even and satisfy
\[
0\le\vartheta\le1,
\quad
\vartheta(s)=1\quad\text{for }|s|\le1,
\quad
\vartheta(s)=0\quad\text{for }|s|\ge2.
\]
For $n\in\mathbb N$, set
\[
f_n(x,s)=\vartheta(s/n)f(x,s).
\]
For each fixed $n$, 
\begin{enumerate}
  \item the Nemytskii operator
$f_n:H\to H$ is globally Lipschitz continuous.
  \item the operators
\[
f_n:L^{q_\sigma}(\Omega)\to L^{m_-'}(\Omega), \quad f_n:L^{q_\sigma}(\Omega)\to L^{m'(x)}(\Omega)
\]
are locally Lipschitz continuous, with Lipschitz constants
on bounded sets independent of $n$. Here
\[q_\sigma:=(p_+-1+\sigma)m_-'.\]
\end{enumerate}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Clearly, 
\[
\partial_s f_n(x,s)
=
\vartheta(s/n)\partial_s f(x,s)
+\frac1n\vartheta'(s/n)f(x,s).
\]
Lemma~\ref{lem:growth} gives
\[
|\vartheta(s/n)\partial_s f(x,s)|
\le C(1+|s|^{p_+-2+\sigma}).\]
Since $\vartheta'(s/n)$ is supported on $n\le|s|\le2n$, where $|s|\ge1$ and
$1/n\le2/|s|$, by \eqref{eq22} we derive 
\[\left|
\frac1n\vartheta'(s/n)f(x,s)
\right|
\le
\frac{2\|\vartheta'\|_\infty}{|s|}
|s|^{p(x)-1}\log|s|
\le
\frac{2\|\vartheta'\|_\infty}{e\sigma}
|s|^{p(x)-2+\sigma}\le
\frac{2\|\vartheta'\|_\infty}{e\sigma}
|s|^{p_+-2+\sigma}.
\]
Combining the above two estimates arrives at 
\begin{equation}
\label{eq23}
|\partial_s f_n(x,s)|
\le C
\bigl(1+|s|^{p_+-2+\sigma}\bigr).
\end{equation}

(1) For fixed $n$, the derivative vanishes for $|s|>2n$.
Thus
\[
\sup_{s\in\mathbb R}|\partial_s f_n(x,s)|
\le
C\bigl(1+(2n)^{p_+-2+\sigma}\bigr)
=:C_n<\infty.
\]
Therefore the fundamental theorem of calculus  implies
\[
|f_n(x,a)-f_n(x,b)|\le C_n|a-b|.
\]
Since $f_n(x,0)=0$, we also have $|f_n(x,a)|\le C_n|a|$.
Hence $f_n$ maps $H$ into $H$.

(2)  To derive an estimate uniform in $n$, we apply \eqref{eq23}, then 
\begin{align*}
|f_n(x,a)-f_n(x,b)|
&=\left|(a-b)\int_0^1 \partial_s f_n(x,b+t(a-b))\,dt\right|\\
&\le C|a-b|\int_0^1 \big(1+|b+t(a-b)|^{p_+-2+\sigma}\big)\,dt\\
&\le C|a-b|\big(1+|a|^{p_+-2+\sigma}+|b|^{p_+-2+\sigma}\big),
\end{align*}
where we used $|b+t(a-b)|=|(1-t)b+ta|\le\max\{|a|,|b|\}$ for $0\le t\le1$ and $p_+-2+\sigma>0$.
                                                                                

By the definition of $q_\sigma$,
\[
\frac1{m_-'}
=
\frac1{q_\sigma}
+\frac{p_+-2+\sigma}{q_\sigma}.
\]
H\"older's inequality implies 
\begin{align*}
\bigl\|
|u-v|\,|u|^{p_+-2+\sigma}
\bigr\|_{m_-'}
\le
\|u-v\|_{q_\sigma}
\bigl\||u|^{p_+-2+\sigma}\bigr\|_
{q_\sigma/(p_+-2+\sigma)}=
\|u-v\|_{q_\sigma}
\|u\|_{q_\sigma}^{p_+-2+\sigma}.
\end{align*}
The same estimate holds with $u$ replaced by $v$.
Furthermore, since $q_\sigma>m_-'$ and $|\Omega|<\infty$,
\[
\|u-v\|_{m_-'}
\le
|\Omega|^{1/m_-'-1/q_\sigma}
\|u-v\|_{q_\sigma}.
\]
Combining these estimates yields
\begin{align*}
\|f_n(\cdot,u)-f_n(\cdot,v)\|_{m_-'}
\le
C
\left(
|\Omega|^{1/m_-'-1/q_\sigma}
+\|u\|_{q_\sigma}^{p_+-2+\sigma}
+\|v\|_{q_\sigma}^{p_+-2+\sigma}
\right)
\|u-v\|_{q_\sigma}.
\end{align*}
If $\|u\|_{q_\sigma}+\|v\|_{q_\sigma}\le L$, then
\begin{equation*}
\|f_n(\cdot,u)-f_n(\cdot,v)\|_{m_-'}
\le C
\left(
|\Omega|^{1/m_-'-1/q_\sigma}
+2L^{p_+-2+\sigma}
\right)\|u-v\|_{q_\sigma}.
\end{equation*}


The continuous inclusion
$L^{m_-'}(\Omega)\hookrightarrow L^{m'(x)}(\Omega)$  directly indicates that $f_n:L^{q_\sigma}(\Omega)\to L^{m'(x)}(\Omega)$ 
is locally Lipschitz continuous.

\label{lem:pairing-estimate}
Assume that Assumptions~\ref{ass1} and~\ref{ass2} hold. For almost every
$t$,
\begin{equation}\label{eq47}
\begin{split}
\frac d{dt}(u,u_t)
                                                                                                                                                                                                                                                                                                                                         
\ge{}&
\left(1+\frac {\hat{q}}{2}\right)\|u_t\|_2^2
+\left(\frac{\hat{q}-2}{2}\ell-\frac{1-\ell}{4\mu}\right)\|\nabla u\|_2^2\\
&+\left(1-\frac {\hat{q}}{p_-}\right)
\int_\Omega|u|^{p(x)}\log|u|dx
-\hat{q}\mathcal E(t)-\left|\int_\Omega b(x,u_t)udx\right|,
\end{split}
\end{equation}

<!-- proof-3: body_md -->
Since $u\in C([0,T];V)$ and $m_+<2^*$, the Sobolev embedding
$V\hookrightarrow L^{2^*}(\Omega)$ yields
\[
\int_{Q_T}|u(x,t)|^{m(x)}\,dx\,dt
\le T|\Omega|+\int_0^T\|u(t)\|_{2^*}^{2^*}\,dt
\le T|\Omega|+TC^{2^*}
\sup_{0\le t\le T}\|\nabla u(t)\|_2^{2^*}<\infty.
\]
Thus
$u\in L^{m(x)}(Q_T)$. 
We may therefore test the weak equation with $u$ after
time regularization. Applying the product rule to $(u,u_t)$ leads  to 
\begin{equation}
\label{eq44}
\begin{split}
\frac d{dt}(u,u_t)
&=\|u_t\|_2^2
-\left(1-\int_0^tg(s)ds\right)\|\nabla u\|_2^2+\int_0^tg(t-s)(\nabla u(t),\nabla u(s)-\nabla u(t))ds\\
&\quad-\int_\Omega b(x,u_t)udx
+\int_\Omega f(x,u)u\,dx.
\end{split}
\end{equation}
For any $\mu>0$, Cauchy--Schwarz and $ab\le a^2/(4\mu)+\mu b^2$ results in 
\begin{equation}
\label{eq45}
\begin{split}
&\left|\int_0^tg(t-s)(\nabla u(t),\nabla u(s)-\nabla u(t))ds\right|\\
&\quad\le\left(\int_0^tg(s)ds\right)^{1/2}\|\nabla u(t)\|_2
\left(\int_0^tg(t-s)\|\nabla u(t)-\nabla u(s)\|_2^2ds\right)^{1/2}\\
&\quad\le\frac{1-\ell}{4\mu}\|\nabla u(t)\|_2^2
+\mu\int_0^tg(t-s)\|\nabla u(t)-\nabla u(s)\|_2^2ds.
\end{split}
\end{equation}
We may choose
$\hat q\in(2,p_-)$ sufficiently close to $p_-$ such that
$\ell>1/(\hat q-1)^2$, which implies $\frac{\hat{q}-2}{2}\ell-\frac{1-\ell}{2\hat{q}}>0.
$
Choose $\mu<\hat{q}/2$ sufficiently close to $\hat{q}/2$ so that
\begin{equation}\label{eq46}
\frac{\hat{q}-2}{2}\ell-\frac{1-\ell}{4\mu}>0,
\quad \frac {\hat{q}}{2}-\mu>0.
\end{equation}
Split the source integral into the fractions $\hat q/p_-$ and
$1-\hat q/p_-$. Multiplying \eqref{eq43} by $\hat q/p_-$ establishes
\[
\begin{aligned}
\int_\Omega f(x,u)u\,dx
\ge{}&\frac{\hat q}{2}\|u_t\|_2^2
+\frac{\hat q}{2}\left(1-\int_0^t g(s)\,ds\right)\|\nabla u\|_2^2+\frac{\hat q}{2}\int_0^t g(t-s)\|\nabla u(t)-\nabla u(s)\|_2^2\,ds
\\
&-\hat q\mathcal E(t)+\left(1-\frac{\hat q}{p_-}\right)
\int_\Omega|u|^{p(x)}\log|u|\,dx.
\end{aligned}
\]
Combining this with \eqref{eq44}--\eqref{eq45} and
$1-\int_0^t g\,ds\ge\ell$ yields \eqref{eq47}.

\label{lem:Y-estimate}
Under \eqref{eq9}, define
\[
Y(t)=H^{1-\alpha}(t)+\varepsilon(u(t),u_t(t)),
\]
where $H(t)\ge h_*>0$ is an auxiliary energy function and
$\varepsilon>0$ is a constant. Their precise definitions will be
given in the proofs of Theorems~\ref{thm:negative} and~\ref{thm:high}.
Then
\begin{equation}\label{eq53}
Y^r(t)\le A\left[
H(t)+\|u_t(t)\|_2^2+\|\nabla u(t)\|_2^2
+\frac{|\Omega|}{ep_-}
+\int_\Omega |u|^{p(x)}\log|u|\,dx
\right],
\end{equation}

If, in addition,
\[
\int_\Omega |u|^{p(x)}\log|u|\,dx\ge0,
\]
then
\begin{equation}\label{eq54}
Y^r(t)\le A\left[
H(t)+\|u_t(t)\|_2^2+\|\nabla u(t)\|_2^2
+\int_\Omega |u|^{p(x)}\log|u|\,dx
\right].
\end{equation}
Here $A$ is given in \eqref{1517}.

<!-- proof-4: body_md -->
For any $x,y\ge0$, $k\ge2$ and $a>0$, Young's inequality shows
\[
xy^{k-1}\le ay^k+
\frac{(k-1)^{k-1}}{k^k}a^{-(k-1)}x^k
\le ay^k+a^{-(k-1)}x^k.
\]
Take $k=m(x)$, $a=\lambda H^{-\alpha}$, and $\lambda\ge1$. Since
$\lambda^{-(m(x)-1)}\le\lambda^{-1}$, we have 
\begin{equation}\label{eq48}
\begin{aligned}
\left|\int_\Omega b(x,u_t)udx\right|
\le\lambda H^{-\alpha}\int_\Omega|u_t|^{m(x)}dx+\lambda^{-1}\int_\Omega H^{\alpha(m(x)-1)}|u|^{m(x)}dx.
\end{aligned}
\end{equation}
To estimate the second term in the right-hand side of \eqref{eq48}, split $\Omega$ into
$\{|u|\le e\}$ and $\{|u|>e\}$.
For  $\{|u|>e\}$, we have $\log|u|>1$ and
$|u|^{p_-}\le |u|^{p(x)}$. Therefore, 
\begin{equation}
\label{eq49}
\begin{split}
\int_{\{|u|>e\}}|u|^{p_-}\,dx
&\le \int_{\{|u|>e\}}|u|^{p(x)}\log|u|\,dx\\
&=\int_\Omega |u|^{p(x)}\log|u|\,dx
-\int_{\{1\le |u|\le e\}}|u|^{p(x)}\log|u|\,dx\\
&\quad+\int_{\{|u|<1\}}|u|^{p(x)}|\log|u||\,dx\\
&\le \frac{|\Omega|}{ep_-}+\int_\Omega |u|^{p(x)}\log|u|\,dx.
\end{split}
\end{equation}
Here  we used 
\[
\int_{\{|u|<1\}}|u|^{p(x)}|\log|u||\,dx
\le\int_{\{|u|<1\}}\frac1{ep(x)}\,dx
\le\frac{|\Omega|}{ep_-}\]
by \eqref{eq22}.  


Since $|u|^{m(x)}\le e^{m_+}$ on $\{|u|\le e\}$,
H\"{o}lder's inequality and \eqref{eq49} lead to 
\begin{equation}
\label{eq50}
\begin{split}
\int_\Omega |u|^{m(x)}\,dx
&\le e^{m_+}|\Omega|
+\int_{\{|u|>e\}}|u|^{m_+}\,dx\\
&\le e^{m_+}|\Omega|
+|\Omega|^{1-\beta}
\left(\int_{\{|u|>e\}}|u|^{p_-}\,dx\right)^\beta\\
&\le e^{m_+}|\Omega|
+|\Omega|^{1-\beta}
\left(\frac{|\Omega|}{ep_-}+\int_\Omega |u|^{p(x)}\log|u|\,dx\right)^\beta\\
&\le C_1
\left(1+\frac{|\Omega|}{ep_-}+\int_\Omega |u|^{p(x)}\log|u|\,dx\right)^\beta,
\end{split}
\end{equation}
where $C_1=e^{m_+}|\Omega|+|\Omega|^{1-\beta}$, $\beta=m_+/p_-<1$.
Similarly,
\begin{equation}\label{eq51}
\|u\|_2\le C_2
\left(1+\frac{|\Omega|}{ep_-}+\int_\Omega|u|^{p(x)}\log|u|\,dx\right)^{1/p_-},
\quad C_2=\left(e^2|\Omega|+|\Omega|^{1-2/p_-}\right)^{1/2}.
\end{equation}
Since $r/p_-+r/2<1$ by recalling \eqref{eq9},
Cauchy--Schwarz, \eqref{eq51}, and generalized Young's inequality with weights $r/p_-$, $r/2$, and $1-r/p_--r/2$ illustrate 
\begin{align*}
|(u,u_t)|^r
&\le\|u\|_2^r\|u_t\|_2^r\\
&\le C_2^r\left(1+\frac{|\Omega|}{ep_-}+\int_\Omega|u|^{p(x)}\log|u|\,dx\right)^{r/p_-}
\|u_t\|_2^r\\
&\le C_2^r\left(1+\frac{|\Omega|}{ep_-}+\int_\Omega|u|^{p(x)}\log|u|\,dx
+\|u_t\|_2^2\right).
\end{align*}
Here  we used 
\begin{equation}
\label{eq52}
\int_\Omega|u|^{p(x)}\log|u|\,dx
=\int_{\{|u|\ge1\}}|u|^{p(x)}\log|u|\,dx-\int_{\{|u|<1\}}|u|^{p(x)}|\log|u||\,dx
\ge-\frac{|\Omega|}{ep_-}.
\end{equation}

Since
$r(1-\alpha)=1$ by recalling \eqref{eq9}, the preceding estimate entails
\[
\begin{aligned}
Y^r(t)
&\le2^{r-1}\left(H(t)+\varepsilon^r|(u,u_t)|^r\right)\\
&\le2^{r-1}\left[
H(t)+\varepsilon^rC_2^r
\left(1+\frac{|\Omega|}{ep_-}+\int_\Omega|u|^{p(x)}\log|u|\,dx
+\|u_t\|_2^2\right)\right].
\end{aligned}
\]
Using $1\le H(t)/h_*$ and \eqref{eq52}, we obtain \eqref{eq53}
\begin{equation}\label{1517}
A=2^{r-1}\left(
1+\varepsilon^rC_2^r
\max\left\{1,\frac{1+\frac{|\Omega|}{ep_-}}{h_*}\right\}
\right).
\end{equation}
If the source integral is nonnegative, absorbing $1+\frac{|\Omega|}{ep_-}$
into $H(t)$ due to $1+\frac{|\Omega|}{ep_-}\le(1+\frac{|\Omega|}{ep_-})H(t)/h_*$ instead gives \eqref{eq54}, with the same constant $A$. 

In fact, \eqref{eq54} applies when $\mathcal E(t)<0$, otherwise,
we retain form \eqref{eq53}.

\label{lem:firstmult}
Under the hypotheses of Theorem \ref{thm:global} and $m_+\le2^*$,
for almost every $t>0$,
\begin{equation}\label{eq69}
\frac d{dt}(u,u_t)
\le\|u_t\|_2^2-\frac\ell4\|\nabla u\|_2^2
+C\|\nabla w\|_2^2
+C\int_\Omega|u_t|^{m(x)}\,dx,
\end{equation}
where 
\[
w(t):=\int_0^t g(t-s)[u(t)-u(s)]\,ds.
\]
<!-- end lemma-6 -->
<!-- proof-5: body_md -->
By Cauchy--Schwarz, \eqref{eq67}, and \eqref{eq41}, one establishes 
\begin{equation}\label{eq70}
\begin{aligned}
\|\nabla w(t)\|_2^2
&\le\left(\int_0^t g(s)\,ds\right)
\int_0^t g(t-s)
\|\nabla u(t)-\nabla u(s)\|_2^2\,ds\\
&\le(1-\ell)\int_0^t g(t-s)
\|\nabla u(t)-\nabla u(s)\|_2^2\,ds\\
&\le\frac{1-\ell}{c_E}E(t)
\le\frac{1-\ell}{c_E}E(0).
\end{aligned}
\end{equation}

It follows from \eqref{eq68} and \eqref{eq70}
that, for $z=u(t)$ or $z=w(t)$,
\begin{align*}
\int_\Omega|z|^{m(x)}\,dx
\le\|z\|_{m_-}^{m_-}+\|z\|_{m_+}^{m_+}
\le C^{m_-}\|\nabla z\|_2^{m_-}
+C^{m_+}\|\nabla z\|_2^{m_+}
\le C\|\nabla z\|_2^2,
\end{align*}
by splitting into $|z|\le1$ and $|z|>1$.
By Young's inequality and the above inequality, we get
\begin{equation}\label{eq71}
\begin{aligned}
\left|\int_\Omega b(x,u_t)z\,dx\right|
=\left|\int_\Omega |u_t|^{m(x)-2}u_tz\,dx\right|\le\varepsilon C\|\nabla z\|_2^2
+C_\varepsilon\int_\Omega|u_t|^{m(x)}\,dx.
\end{aligned}
\end{equation}



By \eqref{eq66}, Young's inequality, and
\eqref{eq71},
\[
\int_\Omega f(x,u)u\,dx
\le\frac{\ell}{2}\|\nabla u\|_2^2,
\]
\[
|(\nabla u,\nabla w)|
\le\frac\ell8\|\nabla u\|_2^2
+\frac2\ell\|\nabla w\|_2^2,
\]
\[
\left|\int_\Omega b(x,u_t)u\,dx\right|
\le\frac\ell8\|\nabla u\|_2^2
+C\int_\Omega|u_t|^{m(x)}\,dx.
\]
Combining these estimates with \eqref{eq44} and using
$1-\int_0^t g(s)\,ds\ge\ell$ proves
\eqref{eq69}.

<!-- lemma-7: body_md -->
\label{lem:secondmult}
Under the hypotheses of Lemma~\ref{lem:firstmult},
fix $t_0>0$ and set
$\Gamma_0=\int_0^{t_0}g(s)\,ds>0$.
For every $\delta>0$ and almost every $t\ge t_0$,
\begin{equation}\label{eq72}
\begin{aligned}
\frac d{dt}[-(u_t,w)]
\le{}&-\frac{\Gamma_0}{2}\|u_t\|_2^2
+\delta\|\nabla u\|_2^2
+C_\delta\|\nabla w\|_2^2\\
&+C_\delta\int_\Omega|u_t|^{m(x)}\,dx+C_\delta\int_0^t[-g'(t-s)]
\|\nabla u(t)-\nabla u(s)\|_2^2\,ds,
\end{aligned}
\end{equation}
where $C_\delta$ may depend on $t_0$ but is
independent of time.
<!-- end lemma-7 -->
<!-- proof-6: body_md -->
Since $w\in C([0,T];V)\cap L^{m(x)}(Q_T)$ and
$w_t\in C([0,T];H)$, time regularization justifies
the product rule and spatial integration by parts,
yielding
\begin{equation}
\label{eq73}
\begin{split}
\frac d{dt}[-(u_t,w)]
={}&\left(1-\int_0^t g(s)\,ds\right)(\nabla u,\nabla w)
+\|\nabla w\|_2^2\\
&+\int_\Omega b(x,u_t)w\,dx
-\int_\Omega f(x,u)w\,dx
-\left(\int_0^t g(s)\,ds\right)\|u_t\|_2^2\\
&-\left(u_t,\int_0^t g'(t-s)[u(t)-u(s)]\,ds\right)
\end{split}
\end{equation}
for almost every $t$.

By \eqref{eq22}, for $0<|s|\le1$,
\[
|f(x,s)|\le |s|\,|s|^{p_--2}|\log|s||
\le\frac{|s|}{e(p_--2)},
\]
and for $|s|>1$,
$|f(x,s)|\le(e\sigma)^{-1}|s|^{p_++\sigma-1}$.
Since $(p_++\sigma-1)(2^*)'<2^*$, Sobolev embedding
and \eqref{eq68} give
\begin{align*}
\|f(x,u)\|_{(2^*)'}
&\le C\bigl(\|u\|_{(2^*)'}+
\|u\|_{(p_++\sigma-1)(2^*)'}^{p_++\sigma-1}\bigr)\\
&\le C\bigl(\|\nabla u\|_2+
\|\nabla u\|_2^{p_++\sigma-1}\bigr)
\le C(1+R^{p_++\sigma-2})\|\nabla u\|_2.
\end{align*}
Consequently, for every $z\in V$,
\[
|\langle f(x,u),z\rangle|
\le\|f(x,u)\|_{(2^*)'}\|z\|_{2^*}
\le C\|\nabla u\|_2\|z\|_V.
\]
Taking the supremum over $\|z\|_V\le1$, we obtain
\begin{equation}\label{eq74}
\|f(x,u)\|_{V^*}\le C\|\nabla u\|_2.
\end{equation}


For $t\ge t_0$, we have
\[
-\left(\int_0^t g(s)\,ds\right)\|u_t(t)\|_2^2
\le
-\left(\int_0^{t_0}g(s)\,ds\right)\|u_t(t)\|_2^2
=-\Gamma_0\|u_t(t)\|_2^2.
\]
Using \eqref{eq74}, Young's inequality, \eqref{eq71}, the
Cauchy--Schwarz inequality, and Poincar\'e's inequality, we obtain, for every
$\delta>0$,
\[\left|
\left(1-\int_0^t g(s)\,ds\right)(\nabla u,\nabla w)
\right|
+\left|\int_\Omega f(x,u)w\,dx\right|
\le\delta\|\nabla u\|_2^2+C_\delta\|\nabla w\|_2^2,\]
\[\left|\int_\Omega b(x,u_t)w\,dx\right|
\le \delta\|\nabla w\|_2^2
+C_\delta\int_\Omega|u_t|^{m(x)}\,dx,\]
\[
\begin{aligned}
\left|
\left(u_t,\int_0^t g'(t-s)[u(t)-u(s)]\,ds\right)
\right|
\le\frac{\Gamma_0}{2}\|u_t\|_2^2
+\frac{C_p^2g(0)}{2\Gamma_0}
\int_0^t[-g'(t-s)]
\|\nabla u(t)-\nabla u(s)\|_2^2\,ds.
\end{aligned}
\]
Combining these estimates with \eqref{eq73} proves \eqref{eq72}.

\label{lem:basicL}
Under the hypotheses of Lemmas~\ref{lem:firstmult}
and~\ref{lem:secondmult}, fix $t_0>0$.
There exist constants $N_1,N_2,C_L,c,C>0$ such that
\[
L(t)=N_1E(t)+(u(t),u_t(t))-N_2(u_t(t),w(t))
\]
satisfies
\begin{equation}\label{eq75}
\begin{aligned}
E(t)\le L(t)\le C_LE(t),\quad t\ge t_0
\end{aligned}
\end{equation}
and
\begin{equation}\label{eq76}
L'(t)\le-cE(t)
+C\int_0^t g(t-s)
\|\nabla u(t)-\nabla u(s)\|_2^2\,ds, \quad \text{a.e. } t\ge t_0.
\end{equation}

To establish equivalence with the energy, Cauchy--Schwarz inequality,
Poincar\'e's inequality, \eqref{eq14}, and
\eqref{eq70} yield
\[
\begin{aligned}
|(u,u_t)|
&\le\frac{C_p}{2}
\bigl(\|\nabla u\|_2^2+\|u_t\|_2^2\bigr)
\le C_5E(t),\\
|(u_t,w)|
&\le\frac{C_p}{2}
\bigl(\|u_t\|_2^2+\|\nabla w\|_2^2\bigr)
\le C_6E(t).
\end{aligned}
\]
Increasing $N_1$, if necessary, so that
$N_1\ge C_5+N_2C_6+1$, we obtain
\[
\bigl(N_1-C_5-N_2C_6\bigr)E(t)
\le L(t)
\le\bigl(N_1+C_5+N_2C_6\bigr)E(t).
\]
This proves \eqref{eq75} with $C_L=N_1+C_5+N_2C_6$.

Set $N_2=4/\Gamma_0$ and choose $\delta=\ell/(8N_2)$
in \eqref{eq72}. Adding \eqref{eq69}
and $N_2$ times \eqref{eq72}, we obtain
\[
\begin{aligned}
\frac{d}{dt}\bigl[(u,u_t)-N_2(u_t,w)\bigr]
\le{}&
-\|u_t\|_2^2-\frac{\ell}{8}\|\nabla u\|_2^2
+C_7\|\nabla w\|_2^2+C_8\int_\Omega|u_t|^{m(x)}\,dx\\
&+C_9\int_0^t[-g'(t-s)]
\|\nabla u(t)-\nabla u(s)\|_2^2\,ds.
\end{aligned}
\]
Choosing $N_1\ge\max\{C_8,2C_9\}$, the energy identity
\eqref{eq41} absorbs the last two terms and gives
the differential inequality
\begin{equation}
\label{eq77}
L'(t)\le
-c_1\bigl(\|u_t(t)\|_2^2+\|\nabla u(t)\|_2^2\bigr)
+C_7\|\nabla w(t)\|_2^2

with $c_1=\min\{1,\ell/8\}$. \eqref{eq14} implies

\|u_t(t)\|_2^2+\|\nabla u(t)\|_2^2

-\int_0^t g(t-s)
\|\nabla u(t)-\nabla u(s)\|_2^2\,ds.

Therefore, it follows from \eqref{eq70} and \eqref{eq77} that

L'(t)\le

\int_0^t g(t-s)
\|\nabla u(t)-\nabla u(s)\|_2^2\,ds,
\end{aligned}

which proves \eqref{eq76}.

<!-- lemma-9: body_md -->
\label{lem:energyintegrability}
Assume the hypotheses of Lemma~\ref{lem:basicL} and
$-g'>0$ almost everywhere on $(0,\infty)$.
Then there exists $C>0$, independent of time and of the
individual initial data in the stable region, such that
\begin{equation}\label{eq78}
\int_0^\infty E(t)\,dt\le CE(0),
\quad
E(t)\le\frac{CE(0)}{1+t},
\quad t\ge0.
\end{equation}
<!-- end lemma-9 -->
<!-- proof-8: body_md -->
Fix $t_0>0$ and use the constants from
Lemma~\ref{lem:basicL}. For $\delta>0$, set
\[
c_\delta=\int_0^\infty
\frac{g(s)^2}{\delta g(s)-g'(s)}\,ds.
\]
Since $\delta g(s)-g'(s)\ge\delta g(s)>0$, we have
\[
c_\delta\le\frac{1}{\delta}
\int_0^\infty g(s)\,ds
=\frac{1-\ell}{\delta}<\infty,\quad 0\le
\frac{\delta g(s)^2}{\delta g(s)-g'(s)}
\le g(s).
\]
The assumption $-g'(s)>0$ almost everywhere implies
$
\frac{\delta g(s)^2}{\delta g(s)-g'(s)}
\to 0$ for almost every $s>0$  as  $\delta\to 0$. 
Since $g\in L^1(0,\infty)$, the dominated convergence
theorem yields
\begin{equation}\label{eq79}
\delta c_\delta
=\int_0^\infty
\frac{\delta g(s)^2}{\delta g(s)-g'(s)}\,ds
\to 0
\quad\text{as }\delta\to 0.
\end{equation}

By the definition of $w$ and 
Cauchy--Schwarz inequality,
\begin{equation}
\label{eq80}
\begin{split}
\|\nabla w(t)\|_2^2
&\le
\left(\int_0^t
\frac{g(t-s)^2}{\delta g(t-s)-g'(t-s)}\,ds\right)
\int_0^t[\delta g(t-s)-g'(t-s)]
\|\nabla u(t)-\nabla u(s)\|_2^2\,ds\\
&\le c_\delta\int_0^t
[\delta g(t-s)-g'(t-s)]
\|\nabla u(t)-\nabla u(s)\|_2^2\,ds.
\end{split}
\end{equation}

Define the nonnegative tail convolution
\[
A_g(t)=\int_0^t
\left(\int_{t-s}^\infty g(r)\,dr\right)
\|\nabla u(s)\|_2^2\,ds.
\]
Then
\begin{equation}
\label{eq81}
0\le A_g(t)
\le(1-\ell)\int_0^t\|\nabla u(s)\|_2^2\,ds.
\end{equation}
Using $\|a-b\|_2^2\le2\|a\|_2^2+2\|b\|_2^2$, we obtain
\begin{equation}\label{eq82}
\begin{split}
A_g'(t)&=(1-\ell)\|\nabla u(t)\|_2^2
-\int_0^t g(t-s)\|\nabla u(s)\|_2^2\,ds\\
&\le
-\frac12\int_0^t g(t-s)
\|\nabla u(t)-\nabla u(s)\|_2^2\,ds
+C_{10}\|\nabla u(t)\|_2^2
\end{split}
\end{equation}
with $C_{10}=2(1-\ell).$ 
Combining \eqref{eq82} with \eqref{eq41} and \eqref{eq80} leads to
\begin{equation}
\label{eq83}
\begin{split}
c_\delta(\delta A_g'+E')
&\le-\frac{c_\delta}{2}\int_0^t
[\delta g(t-s)-g'(t-s)]
\|\nabla u(t)-\nabla u(s)\|_2^2\,ds
+C_{10}\delta c_\delta\|\nabla u(t)\|_2^2\\
&\le-\frac12\|\nabla w(t)\|_2^2
+C_{10}\delta c_\delta\|\nabla u(t)\|_2^2.
\end{split}
\end{equation}

Set $\kappa=2C_7+2$. By \eqref{eq79}, choose
$\delta>0$ sufficiently small and then $\rho>0$ such that
\[
\kappa C_{10}\delta c_\delta\le\frac{c_1}{2},
\quad
\rho C_{10}\le\frac{c_1}{4}.
\]
For
\[
\mathcal Y(t)
=L(t)+\kappa c_\delta E(t)
+(\kappa\delta c_\delta+\rho)A_g(t),
\]
then it follows from \eqref{eq77}, \eqref{eq82} and \eqref{eq83} that 
\begin{equation}
\label{eq84}
\begin{split}
\mathcal Y'(t)
&\le-c_1\|u_t(t)\|_2^2
-\bigl(c_1-\kappa C_{10}\delta c_\delta-\rho C_{10}\bigr)
 \|\nabla u(t)\|_2^2
+\left(C_7-\frac{\kappa}{2}\right)\|\nabla w(t)\|_2^2\\
&\quad-\frac{\rho}{2}\int_0^t g(t-s)
\|\nabla u(t)-\nabla u(s)\|_2^2\,ds\\
&\le-c_1\|u_t(t)\|_2^2
-\frac{c_1}{4}\|\nabla u(t)\|_2^2
-\|\nabla w(t)\|_2^2\\
&\quad-\frac{\rho}{2}\int_0^t g(t-s)
\|\nabla u(t)-\nabla u(s)\|_2^2\,ds
\le-c_2E(t)
\end{split}
\end{equation}
for almost every $t\ge t_0$, where
$
c_2=\frac1{C_E}
\min\left\{\frac{c_1}{4},\frac{\rho}{2}\right\}>0
$
by \eqref{eq14}.

By \eqref{eq75}, \eqref{eq14},
\eqref{eq41}, and \eqref{eq81}, one has 
\[
\mathcal Y(t)\ge L(t)\ge E(t),
\quad
A_g(t_0)\le\frac{(1-\ell)t_0}{c_E}E(0),
\quad
\mathcal Y(t_0)\le CE(0).
\]
Integrating \eqref{eq84} results in 
\[
c_2\int_{t_0}^T E(t)\,dt
\le\mathcal Y(t_0)-\mathcal Y(T)
\le CE(0).
\]
Letting $T\to\infty$ and using \eqref{eq41} on
$[0,t_0]$ proves the first estimate in \eqref{eq78}.
Since 
\[
tE(t)\le\int_0^t E(s)\,ds
\le\int_0^\infty E(s)\,ds\le CE(0), 
\]
adding $E(t)\le E(0)$ shows
\[
E(t)\le\frac{(1+C)E(0)}{1+t},
\quad t\ge0,
\]
which proves the second estimate in \eqref{eq78}.This completes the proof.
