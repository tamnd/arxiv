<!-- definition-1: body_md -->
Let $\mathcal{G}=(V(\mathcal{G}),E(\mathcal{G}))$ be a finite directed graph, with vertex set $V(\mathcal{G})=\{1,...,N\}$ for $N\in\mathbb{N}$ and edge set $E(\mathcal{G})\subseteq \{(i,j)\in V(\mathcal{G})\times V(\mathcal{G})~|~i\neq j\}$. A \textit{state} $n\in V(\mathcal{G})$ is called \textit{absorbing} if $\forall m\in V(\mathcal{G}):(n,m)\notin E(\mathcal{G})$ and \textit{non-absorbing} otherwise. We denote the set of non-absorbing states as $V^\circ(\mathcal{G})\subseteq V(\mathcal{G})$. A finite \textit{semi-Markov network} on $\mathcal{G}$ is specified by a family of finite non-negative measures $\big(Q_{ij}\big)_{(i,j)\in E(\mathcal{G})}$ on $[0,\infty)$ with $Q_{ij}(\{0\})=0$ for all $(i,j)\in E(\mathcal{G})$ such that the quantities
    \begin{align}
      p_{ij}:=Q_{ij}([0,\infty))>0
    \end{align}
    satisfy for every $i\in V^\circ(\mathcal{G})$:
    \begin{align}
        \sum_{j:(i,j)\in E(\mathcal{G})}p_{ij}=1.
    \end{align}
    The measure $Q_{ij}(dt)$ is interpreted as the joint probability that, conditional on entering state $i$, the next state is $j$ and the sojourn time lies in $dt$.

<!-- lemma-1: body_md -->
Let $\text{diag}\big(\tilde{\phi}(s)\big):=\text{diag}\big(\tilde{\phi}_1(s),..., \tilde{\phi}_N(s)\big)$ and $\tilde{W}(s)$ and $\tilde{Q}(s)$ as defined above. Then the following identity holds:
    \begin{align}
        (s\mathbb{1}-\tilde{W}(s))\text{diag}\big(\tilde{\phi}(s)\big)=\mathbb{1}-\tilde{Q}^T(s),
    \end{align}
    where $\mathbb{1}$ denotes the identity matrix in $N$-dimensions.
    
    \noindent Moreover, $\text{diag}\big(\tilde{\phi}(s)\big)$ is invertible, so also
    \begin{align}
        s\mathbb{1}-\tilde{W}(s)=\big(\mathbb{1}-\tilde{Q}^T(s)\big)\text{diag}\big(\tilde{\phi}(s)\big)^{-1}.
    \end{align}
    \label{lemma:Q_W_identity}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Using that $\forall i\in V^\circ(\mathcal{G})$:
    \begin{align}
        \tilde{\phi}_i(s)=&\frac{1}{s}\big(1-\tilde{Q}_i(s)\big),
    \end{align}
    and $\tilde{\phi}_i(s)=\frac{1}{s}$ for $i\in V(\mathcal{G})\setminus V^\circ(\mathcal{G})$, the identity can be explicitly verified: 
    \begin{align}
        \Big[(s\mathbb{1}-\tilde{W}(s))\text{diag}\big(\tilde{\phi}(s)\big)\Big]_{ij}&=-\tilde{K}_{ji}(s)\tilde{\phi}_j(s)\\ &=-\tilde{Q}_{ji}(s),
    \end{align}
    and for $i=j\in V^\circ(\mathcal{G})$:
        \begin{align}
        \Big[(s\mathbb{1}-\tilde{W}(s))\text{diag}\big(\tilde{\phi}(s)\big)]_{ii}&=\Big(s+\sum_{l=1}^N\tilde{K}_{il}(s)\Big)\frac{1-\tilde{Q}_i(s)}{s}\\ &=1-\tilde{Q}_i(s)+\sum_{l=1}^N\tilde{Q}_{il}(s)\\ &=1.
    \end{align}
    For absorbing states, $\tilde{Q}_i(s)=0$, $\tilde {K}_{il}(s)=0$, and $\tilde\phi_i(s)=\frac{1}{s}$ yield the same identity.

    Lastly, since $\forall i\in V(\mathcal{G})$ $\phi_i\geq0 $ and $\phi_i(0) =1 $, each Laplace transform is strictly positive, $\text{diag}\big(\tilde{\phi}(s)\big)$ is invertible.

<!-- proposition-1: body_md -->
Let $X_t$ be a semi-Markov process on a network on $\mathcal{G}=\big(V(\mathcal{G}), E(\mathcal{G})\big)$ with $\tilde{W}(s)$ as defined in Eq.~(\ref{eq:def_W_matrix}) with initial distribution $p_0=(p_{0,i})_{i\in V(\mathcal{G})}$. Then the Laplace transform of the occupation probabilities $p(t)=\big(p_i(t)\big)_{i\in V(\mathcal{G})}$ obeys the following relation:  
    \begin{align}
        (s\mathbb{1}-\tilde{W}(s))\tilde{p}(s)=p_0.\label{eq:generalized_ME_Laplace}
    \end{align}

<!-- proof-2: body_md -->
Consider
    \begin{align}
        R_i(B):=\mathbb{E}\big[\sum_{n\geq 0}\mathbb{1}_{\{X_n^d=i,\ t_n\in B\}}\big]
    \end{align}
    for Borel sets $B\subseteq[0,\infty)$. Then
    \begin{align}
         R_i(B)=p_{0,i}\delta_0(B)+\sum_{j\in V(\mathcal{G})}(R_j*Q_{ji})(B).
    \end{align}
    In Laplace space, this yields
    \begin{align}
        \tilde{R}_i(s)=p_{0,i}+\sum_{j\in V(\mathcal{G})}\tilde{Q}_{ji}(s)\tilde{R}_j(s).
    \end{align}
    Writing $(\tilde{R}(s))_i=\tilde{R}_i(s)$, $\tilde{R}(s)$ can therefore be expressed as:
    \begin{align}
        (\mathbb{1}-\tilde{Q}^T(s))\tilde{R}(s)=p_0. \label{eq:R_Laplace_space}      
    \end{align}
    The occupation probabilities then follow as:
    \begin{align}
         p_i(t)=\int_{[0,t]}\phi_i(t-u)R_i(du),
    \end{align}
    which in Laplace space reads:
    \begin{align}
        \tilde{p}(s)=\text{diag}\big(\tilde{\phi}(s)\big)\tilde{R}(s),
    \end{align}
    where $(\tilde{p}(s))_i=\tilde{p}_i(s)$ and $\text{diag}\big(\tilde{\phi}(s)\big):=\text{diag}\big(\tilde{\phi}_1(s),...,\tilde{\phi}_N(s)\big)$. Combining this expression with Eq.~(\ref{eq:R_Laplace_space}) yields
    \begin{align}
        (\mathbb{1}-\tilde{Q}^T(s))\text{diag}\big(\tilde{\phi}_1(s),...,\tilde{\phi}_N(s)\big)^{-1}\tilde{p}(s)=p_0,
    \end{align}
    using that $\tilde{\phi}_i(s)>0$ since $\phi_i\geq 0$ and $\phi_i(0)=1$. 
    Applying Lemma \ref{lemma:Q_W_identity}, we can alternatively express this equation as:
    \begin{align}
        (s\mathbb{1}-\tilde{W}(s))\tilde{p}(s)=p_0.
    \end{align}

<!-- definition-2: body_md -->
Let $\mathcal{G}=(V(\mathcal{G}),E(\mathcal{G}))$. For $i,j\in V(\mathcal{G})$, a (non-self-intersecting) \textit{path from i to j}, $\mathcal{P}^{i\to j}\subseteq\mathcal{G}$, is a subgraph of the shape:
    \begin{align}
        \mathcal{P}^{i\to j}=\big(\{i, l_1,...,l_{n-1},j\},\{(i, l_1), (l_1,l_2),...,(l_{n-1},j)\}\big).
    \end{align}
    This includes the trivial path $\mathcal{P}^{i\to i}=(\{i\},\emptyset )$. If such a path $\mathcal{P}^{i\to j}$ exists, we say that $i$ is \textit{connected to} $j$ (in the edge-directed sense).

    Given a set of \textit{roots} $I=\{i_1,...,i_n\}\subseteq V(\mathcal{G})$, $I\neq \emptyset$, a \textit{spanning forest} 
    \begin{align}
        \mathcal{F}_{[i_1,...,i_n]}=\big(V(\mathcal{G}),E(\mathcal{F}_{[i_1,...,i_n]})\big)\subseteq\mathcal{G}
    \end{align}
    is a subgraph, where for every $j\in V(\mathcal{G})$, there is exactly one path $\mathcal{P}^{j\to i}$ from $j$ to exactly one $i\in I$. The set of all spanning forests of $\mathcal{G}$ with root set $I=\{i_1,...,i_n\}$ is denoted as $F_{[i_1,...,i_n]}$.
    Moreover, a spanning forest with exactly one root $i\in V(\mathcal{G})$ is called a \textit{spanning tree}. Instead of $\mathcal{F}_{[i]}$ and $F_{[i]}$, we then write $\mathcal{T}_{[i]}$ and $T_{[i]}$ to emphasize this particular structure.

<!-- definition-3: body_md -->
Let $\mathcal{G} =\big(V(\mathcal{G}), E(\mathcal{G})\big)$ be a directed graph, and let
    \begin{align}
        w:E(\mathcal{G})&\to (0,\infty)\\ (i,j)&\mapsto w(i,j)
    \end{align}
    be a weight function that assigns a weight $w(i,j)>0$ to every $(i,j)\in E(\mathcal{G})$. The \textit{adjacency matrix} $A\in\mathbb{R}^{N\times N}$ of the weighted graph $(\mathcal{G},w)$ is defined as:
    \begin{align}
        A_{ij}:=\begin{cases}
            w(i,j)&\text{if }(i,j)\in E(\mathcal{G})\\~~~~0&\text{else},
        \end{cases}
    \end{align}
    and the \textit{(out-degree) Laplacian matrix} $L$ is defined as:
    \begin{align}
        L_{ij}:=\begin{cases}
            ~~ -A_{ij}&\text{if }i\neq j\\\sum_{l\in V(\mathcal{G})}A_{il}&\text{if }i=j,
        \end{cases}.
    \end{align}

<!-- theorem-1: body_md -->
Let $(\mathcal{G},w)$ be a directed and weighted graph with vertex set $V(\mathcal{G})=\{1,...,N\}$. Let $L$ be its Laplacian matrix, $i,j\in V(\mathcal{G})\setminus\{N\}$ and let $L_{\{N\},\{N\}}$ and $L_{\{i,N\},\{j,N\}}$ be the $N$-th principal  minor and the minor obtained from deleting the rows $i$ and $N$ and the columns indexed by $j$ and $N$, respectively. Then the following identity holds for their determinants:
    \begin{align}
        \det{L_{\{N\},\{N\}}}&=\sum_{\mathcal{T}\in T_{[N]}}w(\mathcal{T}),\label{eq_n-th-principal-minor}\\
        \det{L_{\{i,N\},\{j,N\}}}&=(-1)^{i+j}\sum_{\mathcal{F}\in F^{j\to i}_{[i,N]}}w(\mathcal{F}).\label{eq:_n_i_n_j-minor}
    \end{align}
    \label{theorem:AMMTT}
<!-- end theorem-1 -->
<!-- proof-3: body_md -->
We refer to \cite{chaiken1982combinatorial} for details. The key idea is to expand the determinant and to interpret the resulting terms as weights of directed subgraphs. One then shows that precisely the terms containing a cycle cancel by sign-reversing pairing, while the acyclic terms correspond exactly to the spanning forests.

<!-- proposition-2: body_md -->
Let $\mathcal{G}=\big(V(\mathcal{G}),E(\mathcal{G})\big)$ with $V(\mathcal{G})=\{1,...,N\}$ be a semi-Markov network with $\tilde{W}(s)$ as defined in Eq.~(\ref{eq:def_W_matrix}). Then, for every $s>0$, $\big(s\mathbb{1}-\tilde{W}(s)\big)^T$ is a principal minor of a Laplacian matrix of an \textit{augmented graph}:
    \begin{align}
        \mathcal{G}^\ast=\big(V(\mathcal{G}^\ast),E(\mathcal{G}^\ast)\big) :=\big(V(\mathcal{G})\cup\{\ast\},E(\mathcal{\mathcal{G}})\cup\{(i,\ast)~|~i\in V(\mathcal{G})\}\big),
    \end{align}
    with the weight function defined by:
    \begin{align}
        w:E(\mathcal{G}^\ast)&\to(0,\infty)\\ (i,j)&\mapsto\begin{cases}
            ~~~~s&\text{if }j=\ast\\ \tilde{W}_{ji}(s)&\text{else}.
        \end{cases}
    \end{align}
    Moreover, $s\mathbb{1}-\tilde{W}(s)$ is invertible and its inverse reads:
    \begin{align}
        \big(s\mathbb{1}-\tilde{W}(s)\big)^{-1}_{ij}=\frac{\sum_{\mathcal{F}^{s, j\to i}_{[i,\ast]}}w(\mathcal{F})}{\sum_{\mathcal{T}^{s}_{[\ast]}}w(\mathcal{T})},
    \end{align}
    where $\mathcal{T}^s$ and $\mathcal{F}^s$ indicate the fact that these are the spanning trees and forests of the augmented network $\mathcal{G}^\ast$.\label{Prop:inverse_of_s-W}
<!-- end proposition-2 -->
<!-- proof-4: body_md -->
The augmented network is obtained by adding a vertex $\ast$ to $V(\mathcal{G})$ and connecting all other vertices to it. The weights for the edges $(i,j)\in E(\mathcal{G})$ are $\tilde{W}_{ji}(s)=\tilde{K}_{ij}(s)$ and $w(i,\ast)=s$ for $i\in V(\mathcal{G})$. Hence, its Laplacian matrix reads for $n,m\in V(\mathcal{G}^\ast)$:
    \begin{align}
        L_{nm}=\begin{cases}
            ~~~-\tilde{W}_{mn}(s)& \text{for }n,m\in V(\mathcal{G})~n\neq m,\\
            \qquad-s& \text{for }n\in V(\mathcal{G}), m=\ast,\\
            \qquad ~~0& \text{for }n=\ast, m\in V(\mathcal{G}^\ast),\\ 
            s+\sum_{l\in V(\mathcal{G})\setminus\{n\}}\tilde{W}_{ln}(s)& \text{for }m=n\neq\ast.
        \end{cases}
    \end{align}
    Therefore, the $\ast$-principal minor of $L$ reads:
    \begin{align}
        \big(L_{\{\ast\},\{\ast\}}\big)_{nm}&=\begin{cases}
            ~~~-\tilde{W}_{mn}(s)& \text{for }n,m\in V(\mathcal{G})~n\neq m,\\
            s+\sum_{l\in V(\mathcal{G})\setminus\{n\}}\tilde{W}_{ln}(s)& \text{for }m=n\neq\ast.
        \end{cases}\\ &=(s\mathbb{1}-\tilde{W}(s))_{mn}\\ &=\big((s\mathbb{1}-\tilde{W}(s))^T\big)_{nm}.
    \end{align}
    So we see that for $s>0$:
    \begin{align}
        \det(s\mathbb{1}-\tilde{W}(s))=\det L_{\{\ast\},\{\ast\}}=\sum_{\mathcal{T}^{s}_{[\ast]}}w(\mathcal{T})\geq s^N>0,
    \end{align}
    since the augmented graph always contains the spanning tree where each original vertex points directly to $\ast$ and all the other weights are non-negative. We can apply Theorem \ref{theorem:AMMTT} to obtain the inverse of $s\mathbb{1}-\tilde{W}(s)$:
    \begin{align}
        \big(s\mathbb{1}-\tilde{W}(s)\big)^{-1}_{ij}&=\frac{(-1)^{i+j}\det\big(\big(s\mathbb{1}-\tilde{W}(s)\big)_{\{j\}\{i\}}\big)}{\det\big(s\mathbb{1}-\tilde{W}(s)\big)}\\ &=\frac{(-1)^{i+j}\det L_{\{i,\ast\}\{j,\ast\}}}{\det L_{\{\ast\},\{\ast\}}}\\ &=\frac{\sum_{\mathcal{F}^{s,j\to i}_{[i,\ast]}}w(\mathcal{F})}{\sum_{\mathcal{T}^{s}_{[\ast]}}w(\mathcal{T})}.
    \end{align}

<!-- theorem-2: body_md -->
\label{theorem:transient_semi_MCTT}
    Let $X_{t\geq 0}$ be a semi-Markov process on a network $\mathcal{G}$ with initial distribution $\big(p_0\big)_{j\in V(\mathcal{G})}$. The Laplace transform of the occupation probability $\tilde{p}_i(s)$ for state $i\in V(\mathcal{G})$ is given by:
    \begin{align}
        \tilde{p}_i(s)=\sum_{j\in V(\mathcal{G})}\frac{\sum_{\mathcal{F}^{s,j\to i}_{[i,\ast]}}w(\mathcal{F})}{\sum_{\mathcal{T}^{s}_{[\ast]}}w(\mathcal{T})}p_{0,j}\label{eq:transient_semi_MCTT}.
    \end{align}
<!-- end theorem-2 -->

This is an immediate consequence of applying Proposition \ref{Prop:inverse_of_s-W} to Eq.~(\ref{eq:generalized_ME_Laplace}).

<!-- assumption-1: body_md -->
\label{assumption:small_s_sojourn_times}
    For every non-absorbing state $i\in V^\circ(\mathcal{G})$, assume that the Laplace-Stieltjes transforms $\tilde{Q}_{i}(s)$ permit a small-$s$ expansion as follows:
    \begin{align}
         \Tilde{Q}_{i}(s)=1-a_{i}s^{\alpha_{i}}+o(s^{\alpha_{i}}),
    \end{align}
    where $a_{i}>0$ and $\alpha_{i}\in (0,1]$. For an absorbing state $i$, we may set $\alpha_i=0$.
<!-- end assumption-1 -->
<!-- theorem-3: body_md -->
\label{theorem:SMCTT}
    Let $X_{t\geq 0}$ be a semi-Markov process on a network on $\mathcal{G}$ with at least one spanning tree  and sojourn times that satisfy Assumption \ref{assumption:small_s_sojourn_times}. If the limiting state distribution $\pi :=\lim_{t\to \infty}p(t)$ exists for every initial condition, it is unique in the sense that it is independent of the initial distribution, and for $i\in V(\mathcal{G})$, it is given by:
    \begin{align}
    \pi_i=\begin{cases}
        \frac{\sum_{\mathcal{T}_{[i]}}w_\kappa(\mathcal{T})}{\sum_{l\in V_{\alpha^\ast}}\sum _{\mathcal{T}_{[l]}}w_\kappa(\mathcal{T})}&\text{if}~\alpha_i=\alpha^\ast,\\ \qquad~~0 &\text{else}.
    \end{cases}\label{eq:stationary_solution}
\end{align}
<!-- end theorem-3 -->
<!-- proof-6: body_md -->
Observe that $\mathcal{G}$ can contain at most one absorbing vertex $n\in V(\mathcal{G})$ because it contains a spanning tree. If this is the case, all spanning trees have to be rooted at $n$, so Eq.~(\ref{eq:transient_semi_MCTT}) implies that:
    \begin{align}
        \pi_i=\lim_{s\to 0}s\tilde{p}_i(s)=\sum_{j\in V(\mathcal{G})}\frac{\sum_{\mathcal{T}_{[i]}}w(\mathcal{T})}{\sum_{\mathcal{T}_{[n]}}w(\mathcal{T})}p_{0,j}=\delta_{in}.
    \end{align}
    Since $\alpha_n=0$ and $\alpha_i>0$ for $i\in V(\mathcal{G})\setminus\{n\}$, Eq.~(\ref{eq:stationary_solution}) holds in this case.
    
    Now consider the case that there are no absorbing vertices and observe that Assumption \ref{assumption:small_s_sojourn_times} implies the following small-$s$ scaling for a spanning forest $\mathcal{F}$ of $\mathcal{G}$ with roots $\{i_1,...,i_n\}\subseteq V(\mathcal{G})$: 
    \begin{align}
    w(\mathcal{F})=w_\kappa(\mathcal{F})s^{|E(\mathcal{F})|-\sum_{k\in V(\mathcal{G})}\alpha_k+\sum_{k=1}^n\alpha_{i_k}}+o(s^{|E(\mathcal{F})|-\sum_{k\in V(\mathcal{G})}\alpha_k+\sum_{k=1}^n\alpha_{i_k}}).
    \end{align}
    Therefore, 
    \begin{align}
        \sum_{\mathcal{F}^{s,j\to i}_{[i,\ast]}}w(\mathcal{F})&=\sum_{\mathcal{T}_{[i]}}w_\kappa(\mathcal{T})s^{N-1-\sum_{k\in V(\mathcal{G})}\alpha_k+\alpha_{i}}+o(s^{N-1-\sum_{k\in V(\mathcal{G})}\alpha_k+\alpha_{i}}),\\\sum_{\mathcal{T}^{s}_{[\ast]}}w(\mathcal{T})&=\sum_{l:\alpha_l=\alpha^\ast}\sum _{\mathcal{T}_{[l]}}w_\kappa(\mathcal{T})s^{N-\sum_{k\in V(\mathcal{G})}\alpha_k+\alpha^\ast}+o(s^{N-\sum_{k\in V(\mathcal{G})}\alpha_k+\alpha^\ast}).
    \end{align}
    Note that the leading-order term of the sum over the spanning forests is independent of $j$.
    
    By assumption, the limiting state distribution $\pi_i$ for $i\in V(\mathcal{G})$ satisfies $\pi_i=\lim_{s\to 0}s\tilde{p}_i(s)$, and Eq.~(\ref{eq:transient_semi_MCTT}) implies:
    \begin{align}
        \pi_i&=\lim_{s\to 0}s\tilde{p}_i(s)\\&=\sum_{j\in V(\mathcal{G})}p_{0,j}\lim_{s\to0}\frac{\sum_{\mathcal{T}_{[i]}}w_\kappa(\mathcal{T})s^{\alpha_{i}}}{\sum_{l:\alpha_l=\alpha^\ast}\sum _{\mathcal{T}_{[l]}}w_\kappa(\mathcal{T})s^{\alpha^\ast}}\\ &=\begin{cases}
        \frac{\sum_{\mathcal{T}_{[i]}}w_\kappa(\mathcal{T})}{\sum_{l:\alpha_l=\alpha^\ast}\sum _{\mathcal{T}_{[l]}}w_\kappa(\mathcal{T})}&\text{if}~\alpha_i=\alpha^\ast,\\ \qquad~~0 &\text{else},
    \end{cases}
    \end{align}
    since $\sum_{j\in V(\mathcal{G})}p_{0,j}=1$. In particular, the limit is independent of the choice of the initial distribution $p_0$, i.e., the limiting distribution is uniquely determined by the network.

<!-- example-1: body_md -->
Consider the graph with two vertices $\mathcal{G}=\big(\{1,2\},\{(1,2),(2,1)\}\big)$ and the semi-Markov network with $\delta$-like transition-time law:
    \begin{align}
        Q_{12}=\delta_\tau,~~Q_{21}=\delta_\tau.
    \end{align}
    Then the transitions are deterministic and it is clear that the occupation probabilities read:
    \begin{align}
        p_1(t)=1-p_2(t)=\begin{cases}
            p_{0,1}~~\text{if }2k\tau \leq t<(2k+1)\tau \text{ for }k\in \mathbb{N}_0,\\ p_{0,2}~~\text{if }(2k+1)\tau \leq t<2(k+1)\tau \text{ for }k\in \mathbb{N}_0. 
        \end{cases}
    \end{align}
    In particular, the limit $t\to \infty$ does not exist but the process alternates indefinitely between the two states. This is reflected by purely imaginary zeros of $\det(s\mathbb{1}-\tilde{W}(s))$. Note that $\tilde{K}_{12}(s)=\tilde{K}_{21}(s)=\frac{s}{e^{\tau s}-1}$, so:
    \begin{align}
         &0=\det\big(s\mathbb{1}-\tilde{W}(s)\big)=\sum_{\mathcal{T}^{s}_{[\ast]}}w(\mathcal{T}) =s^2\big(1+\frac{2}{e^{\tau s}-1}\big) ~\Leftrightarrow ~s=0~~\lor~~e^{\tau s}=-1.
    \end{align}
    Thus, $s=i \frac{2k+1}{\tau}\pi$ for $k\in \mathbb{Z}$ are also zeros of the determinant.

\label{prop_fpt_distribution_semi_Markov}
    Let $X_t$ be a semi-Markov process on a network on $\mathcal{G}=\big(\{1,...,N\}, E(\mathcal{G})\big)$, where $N$ is an absorbing state, with initial distribution $p_0$. Then
    \begin{align}
        F(t)&=p_N(t)=1-\sum_{j=1}^{N-1}p_j(t),
        \\\tilde{f}(s)&=p_{0,N}+\sum_{j=1}^{N-1}p_{0,j}\frac{\sum_{i=1}^{N-1}\Tilde{K}_{iN}(s)\sum_{\mathcal{F}^{s,j\to i}_{[i,N,\ast]}}w(\mathcal{F})}{\sum_{\mathcal{F}^{s}_{[N,\ast]}}w(\mathcal{F})}\label{eq:FPT_distribution_graph_theory_1},
    \end{align}
    where the sums are taken over the augmented graph $\mathcal{G}^\ast$ as before.
<!-- end proposition-3 -->
<!-- proof-8: body_md -->
The first identity follows immediately from the fact that $N$ is absorbing, as once $X_t$ reaches $N$, it stays in this state, so $p_N(t)=\mathbb{P}(T\leq t)=F(t)$.

    The Laplace transform of $F(t)$ therefore reads:
    \begin{align}
        \tilde{F}(s)=\frac{1}{s}-\sum_{j=1}^{N-1}\tilde{p}_j(s),
    \end{align}
    and applying Eq.~(\ref{eq:FPT_distribution}) and Eq.~(\ref{eq:transient_semi_MCTT}) then yields:
    \begin{align}
    \Tilde{f}(s)&=s\tilde{F}(s)\\&=1-s\sum_{i=1}^{N-1}\Tilde{p}_i(s)&\\&=1-\sum_{j=1}^Np_{0,j}\frac{s\sum_{i=1}^{N-1}\sum_{\mathcal{F}^{s,j\to i}_{[i,\ast]}}w(\mathcal{F})}{\sum_{\mathcal{T}^{s}_{[\ast]}}w(\mathcal{T})}\\ &=\sum_{j=1}^Np_{0,j}\frac{\sum_{\mathcal{T}^{s}_{[\ast]}}w(\mathcal{T})-s\sum_{i=1}^{N-1}\sum_{\mathcal{F}^{s,j\to i}_{[i,\ast]}}w(\mathcal{F})}{\sum_{\mathcal{T}^{s}_{[\ast]}}w(\mathcal{T})}.
    \end{align}
    Observe that for every $j\in V(\mathcal{G})\setminus\{N\}$, the sum over the spanning trees of the augmented graph can be decomposed as follows:
    \begin{align}
        \sum_{\mathcal{T}^{s}_{[\ast]}}w(\mathcal{T})&=s\sum_{i=1}^{N-1}\sum_{\mathcal{F}^{s,j\to i}_{[i,\ast]}}w(\mathcal{F})+s\sum_{\mathcal{F}^{s,j\to N}_{[N,\ast]}}w(\mathcal{F})\\&=s\sum_{i=1}^{N-1}\sum_{\mathcal{F}^{s,j\to i}_{[i,\ast]}}w(\mathcal{F})+s\sum_{i=1}^{N-1}\Tilde{K}_{iN}(s)\sum_{\mathcal{F}^{s,j\to i}_{[i,N,\ast]}}w(\mathcal{F}).
    \end{align}
    Thus, the Laplace transform of the distribution can be expressed as:
    \begin{align}
        \Tilde{f}(s)&=p_{0,N}+\sum_{j=1}^{N-1}p_{0,j}\frac{\sum_{i=1}^{N-1}\Tilde{K}_{iN}(s)\sum_{\mathcal{F}^{s,j\to i}_{[i,N,\ast]}}w(\mathcal{F})s}{\sum_{\mathcal{T}^{s}_{[\ast]}}w(\mathcal{T})}\\ &=p_{0,N}+\sum_{j=1}^{N-1}p_{0,j}\frac{\sum_{i=1}^{N-1}\Tilde{K}_{iN}(s)\sum_{\mathcal{F}^{s,j\to i}_{[i,N,\ast]}}w(\mathcal{F})}{\sum_{\mathcal{F}^{s}_{[N,\ast]}}w(\mathcal{F})},
    \end{align}
    using that the only outgoing edge of $N$ in the augmented graph is $(N,\ast)$, which therefore contributes a factor $s$ to all terms in the sum over the spanning trees.

<!-- lemma-2: body_md -->
\label{lemma:finite_K}
    Consider a semi-Markov network on the graph $\mathcal{G}=(\{1,...,N\},E(\mathcal{G}))$. Let $(i,j)\in E(\mathcal{G})$, and
    \begin{align}
        \int_{0}^{\infty}t^nQ_i(\text{d}t)<\infty
    \end{align}
    for an $n\in \mathbb{N}$, i.e., the $n$-th moment of the sojourn time in state $i$ is finite. Then $\lim_{s\to 0}\frac{\text{d}^m}{\text{d}s^m}\tilde{K}_{ij}(s)$ is finite for all $m\in\{0,...,n-1\}$ and:
    \begin{align}
        \lim_{s\to 0}s\frac{\text{d}^n}{\text{d}s^n}\tilde{K}_{ij}(s)=0.
    \end{align}
<!-- end lemma-2 -->
<!-- proof-9: body_md -->
A finite $n$-th moment implies that also all moments of order $m\leq n$ are finite, and since $Q_{ij}\leq Q_i$, we also have 
    \begin{align}
        \int_{0}^{\infty}t^mQ_{ij}(\text{d}t)<\infty,
    \end{align}
    for $m\leq n$.

    For the Laplace transforms, this means that $\forall m\in\{1,...,n\}$:
    \begin{align}
        \lim_{s\to 0}\frac{\text{d}^m\Tilde{Q}_{ij}}{\text{d}s^m}(s)&=(-1)^m\int_{0}^{\infty}t^mQ_{ij}(\text{d}t)<\infty\\ \lim_{s\to 0}\frac{\text{d}^m\Tilde{Q}_{i}}{\text{d}s^m}(s)&=(-1)^m\int_{0}^{\infty}t^mQ_{i}(\text{d}t)<\infty.
    \end{align}
    Since $Q_i\geq 0$, the following holds for $m\leq n-1$:
    \begin{align}
      \lim_{s\to 0}\frac{\text{d}^{m}\tilde{\phi}_i}{\text{d}s^m}(s)&=(-1)^m\int_{0}^{\infty}t^m\phi_i(t)\text{d}t\\&=(-1)^m\int_{0}^{\infty}t^m\int_{0}^{\infty}\mathbb{1}_{\{t'>t\}}Q_i(\text{d}t')\text{d}t\\&=(-1)^m\int_{0}^{\infty}\Big(\int_{0}^{t'}t^m\text{d}t\Big)Q_i(\text{d}t')\\&=\frac{(-1)^m}{m+1}\int_{0}^{\infty}t'^{m+1}Q_i(\text{d}t')<\infty\label{eq:relation_phi_psi_moments}.
    \end{align}
    Therefore, since $\tilde{\phi}_i(0)>0$ and $\tilde{K}_{ij}(s)=\frac{\Tilde{Q}_{ij}(s)}{\tilde{\phi}_i(s)}$, for all $m\leq n-1$, $\lim_{s\to 0}\frac{\text{d}^m}{\text{d}s^m}\tilde{K}_{ij}(s)$ exists and is finite.

    For the final claim, it remains to show that $\lim_{s\to 0}s\frac{\text{d}^n\tilde{\phi}_i}{\text{d}s^n}(s)=0$. Note that
    \begin{align}
        s\frac{\text{d}^n\tilde{\phi}_i}{\text{d}s^n}(s)&=-\big(n\frac{\text{d}^{n-1}\tilde{\phi}_i}{\text{d}s^{n-1}}(s)+\frac{\text{d}^{n}\Tilde{Q}_i}{\text{d}s^{n}}(s)\big),
    \end{align}
    which follows by induction as
    \begin{align}
        s\frac{\text{d}\tilde{\phi}_i}{\text{d}s}(s)&=-\big(\tilde{\phi}_i(s)+\frac{\text{d}\Tilde{Q}_i}{\text{d}s}(s)\big),
    \end{align}
    and
    \begin{align}
        s\frac{\text{d}^{n+1}\tilde{\phi}_i}{\text{d}s^{n+1}}(s)&=\frac{\text{d}}{\text{d}s}\Big(s\frac{\text{d}^{n}\tilde{\phi}_i}{\text{d}s^{n}}(s)\Big)-\frac{\text{d}^{n}\tilde{\phi}_i}{\text{d}s^{n}}(s)\\ &=-\big((n+1)\frac{\text{d}^{n}\tilde{\phi}_i}{\text{d}s^{n}}(s)+\frac{\text{d}^{n+1}\Tilde{Q}_i}{\text{d}s^{n+1}}(s)\big),
    \end{align}
    using the induction hypothesis for the second equality.
    By Eq.~(\ref{eq:relation_phi_psi_moments}), we have:
    \begin{align}
        n\lim_{s\to0}\frac{\text{d}^{n-1}\tilde{\phi}_i}{\text{d}s^{n-1}}(s)=-\lim_{s\to0}\frac{\text{d}^{n}\Tilde{Q}_i}{\text{d}s^{n}}(s)\Rightarrow \lim_{s\to 0}s\frac{\text{d}^n\tilde{\phi}_i}{\text{d}s^n}(s)=0.
    \end{align}
    Therefore, we conclude that $\lim_{s\to 0}s\frac{\text{d}^n}{\text{d}s^n}\tilde{K}_{ij}(s)=0$.

<!-- theorem-4: body_md -->
\label{theorem_fpt_semi_Markov}
    Let $X_t$ be a semi-Markov process on a network on $\mathcal{G}=\big(V(\mathcal{G}), E(\mathcal{G})\big)$, where $V(\mathcal{G})=\{1,...,N\}$, with initial distribution $p_0$. Let $\mathcal{G}$ contain at least one spanning tree rooted at $N$, and let $T$ be the hitting time to state $N$. If the sojourn times for all states $i\in V(\mathcal{G})\setminus\{N\}$ have finite moments up to order $n\in \mathbb{N}$, then the following holds:
    \begin{align}
        \mathbb{E}[T^k]&=(-1)^{k-1}k\sum_{j=1}^Np_{0,j}\frac{\text{d}^{k-1}}{\text{d}s^{k-1}}\frac{\sum_{i=1}^{N-1}\sum_{\mathcal{F}^{s,j\to i}_{[i,N,\ast]}}w(\mathcal{F})}{\sum_{\mathcal{F}^{s}_{[N,\ast]}}w(\mathcal{F})}\Big|_{s=0}<\infty,\label{eq:moments_fpt}
    \end{align}
    for every $k\in\{1,...,n\}$, and for the mean in particular:
    \begin{align}
        \mathbb{E}[T]=\sum_{j=1}^{N}p_{0,j}\frac{\sum_{i=1}^{N-1}\sum_{\mathcal{F}^{j\to i}_{[i,N]}}w_\kappa(\mathcal{F})}{\sum_{\mathcal{T}_{[N]}}w_\kappa(\mathcal{T})}.\label{eq:MFPT}
    \end{align}
    \begin{proof}
        Using Eq.~(\ref{eq:FPT_distribution_graph_theory_1}), we can express the moments as follows:
        \begin{align}
             \mathbb{E}[T^k]=&(-1)^k\lim_{s\to 0}\frac{\text{d}^k\tilde{f}}{\text{d}s^k}(s)\\=&(-1)^k\lim_{s\to 0}\frac{\text{d}^k}{\text{d}s^k}\Big(1-p_{0,N}-\sum_{j=1}^{N-1}p_{0,j}\frac{s\sum_{i=1}^{N-1}\sum_{\mathcal{F}^{s,j\to i}_{[i,N,\ast]}}w(\mathcal{F})}{\sum_{\mathcal{F}^{s}_{[N,\ast]}}w(\mathcal{F})}\Big)\\ =&(-1)^{k-1}\lim_{s\to 0}\sum_{j=1}^{N-1}p_{0,j}\Big(k\frac{\text{d}^{k-1}}{\text{d}s^{k-1}}\frac{\sum_{i=1}^{N-1}\sum_{\mathcal{F}^{s,j\to i}_{[i,N,\ast]}}w(\mathcal{F})}{\sum_{\mathcal{F}^{s}_{[N,\ast]}}w(\mathcal{F})}\\&+s\frac{\text{d}^{k}}{\text{d}s^k}\frac{\sum_{i=1}^{N-1}\sum_{\mathcal{F}^{s,j\to i}_{[i,N,\ast]}}w(\mathcal{F})}{\sum_{\mathcal{F}^{s}_{[N,\ast]}}w(\mathcal{F})}\Big)\label{eq:FPT_moments_intermediate_step}.
        \end{align}
        The second summand contains terms of at most $s\frac{\text{d}^{n}\tilde{K}_{ij}}{\text{d}s^n}(s)$, which all vanish by Lemma \ref{lemma:finite_K} for $s\to 0$. Since by assumption, there is a spanning tree $\mathcal{T}$ rooted in $N$, and it has weight $w_\kappa(\mathcal{T})>0$ because the sojourn times have finite first moments, i.e., finite means, we have:
        \begin{align}
            \lim_{s\to0}\sum_{\mathcal{F}^{s}_{[N,\ast]}}w(\mathcal{F})=\sum_{\mathcal{T}_{[N]}}w_\kappa(\mathcal{T})>0.
        \end{align}
        Hence, the second summand in Eq.~(\ref{eq:FPT_moments_intermediate_step}) vanishes and 
        \begin{align}
            \mathbb{E}[T^k]&=(-1)^{k-1}k\sum_{j=1}^{N-1}p_{0,j}\frac{\text{d}^{k-1}}{\text{d}s^{k-1}}\frac{\sum_{i=1}^{N-1}\sum_{\mathcal{F}^{s,j\to i}_{[i,N,\ast]}}w(\mathcal{F})}{\sum_{\mathcal{F}^{s}_{[N,\ast]}}w(\mathcal{F})}\Big|_{s=0},
        \end{align}
        which by Lemma \ref{lemma:finite_K} is finite.

        The formula for the mean follows immediately by setting $k=1$:
        \begin{align}
            \mathbb{E}[T]=\sum_{j=1}^{N}p_{0,j}\frac{\sum_{i=1}^{N-1}\sum_{\mathcal{F}^{j\to i}_{[i,N]}}w_\kappa(\mathcal{F})}{\sum_{\mathcal{T}_{[N]}}w_\kappa(\mathcal{T})}.
        \end{align}
    \end{proof}

<!-- proposition-4: body_md -->
Consider a semi-Markov network on $\mathcal{G}=\big(V(\mathcal{G}), E(\mathcal{G})\big)$, where $V(\mathcal{G})=\{1,...,N\}$, with finite-mean nonlattice sojourn time distributions. Let $\mathcal{G}$ contain a spanning tree rooted at every $i\in V(\mathcal{G})$. Let $T_{i\to j}$ be the hitting time to state $j$ for a process starting at $i$, i.e., $X_0=i$. Then, there exists $K\in(0,\infty)$ (\textit{Kemeny's constant}) such that $\forall i\in V(\mathcal{G})$:
    \begin{align}

    \end{align}

    \begin{align}
        K=\frac{\sum_{1\leq j<l\leq N}\sum_{\mathcal{F}_{[l,j]}}w_\kappa(\mathcal{F})}{\sum_{l=1}^N\sum_{\mathcal{T}_{[l]}}w_\kappa(\mathcal{T})}.

\label{Theorem:FPT_asymptotics}
    Let $X_{t\geq 0}$ be a semi-Markov process on a network $\mathcal{G}$ with at least one spanning tree with root $N\in V(\mathcal{G})$ with initial condition $p_0$ such that $p_{0,N}<1$ and sojourn times that satisfy Assumption \ref{assumption:small_s_sojourn_times}. Then the small-$s$ behavior of the hitting-time distribution in Laplace space is given by:
    \begin{align}
     \tilde{f}(s)=1-\sum_{j=1}^Np_{0,j}\frac{\sum_{i:\alpha_i=\alpha_{p_0}(\mathcal{G})}\sum_{\mathcal{F}^{ j\to i}_{[i,N]}}w_\kappa(\mathcal{F})}{\sum_{\mathcal{T}_{[N]}}w_\kappa(\mathcal{T})}s^{\alpha_{p_0}(\mathcal{G})}+o(s^{\alpha_{p_0}(\mathcal{G})})\label{eq:FPT_distribution_Laplace_asymptotics}.
     \end{align}
    If $\alpha_{p_0}(\mathcal{G})<1$, this yields the following asymptotic scaling in time domain for the survival probability:
    \begin{align}
        \mathbb{P}(T>t)\sim \frac{\sum_{j=1}^Np_{0,j}\sum_{i:\alpha_i=\alpha_{p_0}(\mathcal{G})}\sum_{\mathcal{F}^{ j\to i}_{[i,N]}}w_\kappa(\mathcal{F})}{\Gamma(1-\alpha_{p_0}(\mathcal{G}))\sum_{\mathcal{T}_{[N]}}w_\kappa(\mathcal{T})}t^{-\alpha_{p_0}(\mathcal{G})}.
    \end{align}

<!-- proof-12: body_md -->
We again use that Assumption \ref{assumption:small_s_sojourn_times} yields the following small-$s$ scaling for a spanning forest $\mathcal{F}$ of $\mathcal{G}$ with roots $\{i_1,...,i_n\}\subseteq V(\mathcal{G})$: 
    \begin{align}
    w(\mathcal{F})=w_\kappa(\mathcal{F})s^{|E(\mathcal{F})|-\sum_{k\in V(\mathcal{G})}\alpha_k+\sum_{k=1}^n\alpha_{i_k}}+o(s^{|E(\mathcal{F})|-\sum_{k\in V(\mathcal{G})}\alpha_k+\sum_{k=1}^n\alpha_{i_k}}).
    \end{align}
    Since $N$ is an absorbing state, we have:
    \begin{align}
        \sum_{\mathcal{F}^{s,j\to i}_{[i,N,\ast]}}w(\mathcal{F})&=\sum_{\mathcal{F}^{s,j\to i}_{[i,N]}}w_\kappa(\mathcal{F})s^{N-2-\sum_{k\in V(\mathcal{G})}\alpha_k+\alpha_{i}}+o(s^{N-2-\sum_{k\in V(\mathcal{G})}\alpha_k+\alpha_{i}}),\\\sum_{\mathcal{F}^{s}_{[N,\ast]}}w(\mathcal{F})&=\sum _{\mathcal{T}_{[N]}}w_\kappa(\mathcal{T})s^{N-1-\sum_{k\in V(\mathcal{G})}\alpha_k}+o(s^{N-1-\sum_{k\in V(\mathcal{G})}\alpha_k}).
    \end{align}
    Hence, using Eq.~(\ref{eq:FPT_distribution_graph_theory_1}), we find the following small-$s$ regime for $\tilde{f}(s)$:
    \begin{align}
     \tilde{f}(s)&=1-s\sum_{j=1}^Np_{0,j}\frac{\sum_{i=1}^{N-1}\sum_{\mathcal{F}^{s,j\to i}_{[i,N,\ast]}}w(\mathcal{F})}{\sum_{\mathcal{F}^{s}_{[N,\ast]}}w(\mathcal{F})}\\ &=1-\sum_{j=1}^Np_{0,j}\frac{\sum_{i:\alpha_i=\alpha_{p_0}(\mathcal{G})}\sum_{\mathcal{F}^{ j\to i}_{[i,N]}}w_\kappa(\mathcal{F})}{\sum_{\mathcal{T}_{[N]}}w_\kappa(\mathcal{T})}s^{\alpha_{p_0}(\mathcal{G})}+o(s^{\alpha_{p_0}(\mathcal{G})}).
     \end{align}

    \begin{align}
        \int_{0}^{\infty}\mathbb{P}(T>t)e^{-st}\text{d}t&=\int_{0}^{\infty}(1-F(t))e^{-st}\text{d}t\\&=\frac{1-\tilde{f}(s)}{s}\\ &=\sum_{j=1}^Np_{0,j}\frac{\sum_{i:\alpha_i=\alpha_{p_0}(\mathcal{G})}\sum_{\mathcal{F}^{ j\to i}_{[i,N]}}w_\kappa(\mathcal{F})}{\sum_{\mathcal{T}_{[N]}}w_\kappa(\mathcal{T})}s^{\alpha_{p_0}(\mathcal{G})-1}+o(s^{\alpha_{p_0}(\mathcal{G})-1}).
    \end{align}

    \begin{align}
        \mathbb{P}(T>t)\sim \frac{\sum_{j=1}^Np_{0,j}\sum_{i:\alpha_i=\alpha_{p_0}(\mathcal{G})}\sum_{\mathcal{F}^{ j\to i}_{[i,N]}}w_\kappa(\mathcal{F})}{\Gamma(1-\alpha_{p_0}(\mathcal{G}))\sum_{\mathcal{T}_{[N]}}w_\kappa(\mathcal{T})}t^{-\alpha_{p_0}(\mathcal{G})}.
    \end{align}

<!-- definition-5: body_md -->
Let $X_t$ be a semi-Markov process on a network $\mathcal{G}=(V(\mathcal{G}), E(\mathcal{G}))$ with the underlying discrete-time Markov process $(X^d_n)_{n\in\mathbb{N}_0}$. We define the \textit{number of transitions along an edge }$(i,j)\in E(\mathcal{G})$ as:
    \begin{align}
        N_{ij}(t):=\big|\{n\in\mathbb{N}~|~t_n\leq t,~{X^d_{n-1}=i\land X^d_n=j} \}\big|.
    \end{align}
    A \textit{time-additive observable} is a random variable of the shape:
    \begin{align}
    A(t)=\int_{0}^{t}g(X_{t'})\text{d}t'+\sum_{(i,j)\in E(\mathcal{G})}h(i,j)N_{ij}(t),
    \end{align}
    where 
    \begin{align*}
        g:V(\mathcal{G})&\to \mathbb{R}\\
        i&\mapsto g_i
    \end{align*}
    defines the \textit{state weights} and
    \begin{align*}
        h:E(\mathcal{G})&\to \mathbb{R}\\
        (i,j)&\mapsto h_{ij}
    \end{align*}
    defines the \textit{edge weights} of $A(t)$. If $h=0$, we call $A(t)$ a \textit{time-additive state observable}, and if $g=0$, we call $A(t)$ a \textit{time-additive transition observable}.

<!-- proposition-5: body_md -->
Let $A(t)$ be a time-additive observable of a semi-Markov network on $\mathcal{G}=(V(\mathcal{G}), E(\mathcal{G}))$. Define $\phi_i^{(\lambda)}(t):=\phi_i(t)e^{\lambda g_it}$ for $i\in V(\mathcal{G})$ and $Q_{ij}^{(\lambda)}(\text{d}t):=e^{\lambda\left(g_i t+h_{i,j}\right)}Q_{ij}(\text{d}t)$ for $(i,j)\in E(\mathcal{G})$. Then, for $\tilde{G}(\lambda,s)=(\tilde{G}_i(\lambda,s))_{i\in V(\mathcal{G})}$ and every $ s>0$, there exists $\epsilon_s>0$ such that for every $\lambda\in (-\epsilon_s,\epsilon_s)$, the following identity holds:
     \begin{align}
         \big(\mathbb{1}-\tilde{Q}^{(\lambda)}(s)\big)\tilde{G}(\lambda,s)=\tilde{\phi}^{(\lambda)}(s).\label{eq:generalized_Feynmann_Kac}
     \end{align}
<!-- end proposition-5 -->
<!-- proof-13: body_md -->
For a fixed $s>0$, $\tilde{Q}_{ij}^{(\lambda)}(s)=e^{\lambda h_{ij}}\tilde{Q}_{ij}(s-\lambda g_i)$ and $\tilde{\phi}_{i}^{(\lambda)}(s)=\tilde{\phi}_{i}(s-\lambda g_i)$ are finite for all $\lambda$ in a sufficiently small neighborhood around $0$, and we can calculate $\tilde{G}_i(\lambda,s)$ by splitting the expectation value into a contribution before the first jump and after the first jump:
    \begin{align}
        G_i(t,\lambda)&=\mathbb{E}[e^{\lambda A(t)}~|~X_0=i]\\ &=\mathbb{P}(t<t_1~|~X_0=i)\mathbb{E}[e^{\lambda A(t)}~|~X_0=i, t<t_1]\\&+\mathbb{P}(t\geq t_1~|~X_0=i)\mathbb{E}[e^{\lambda A(t)}~|~X_0=i, t\geq t_1].
    \end{align}
    The first term is simply the survival probability up to time $t$ times the contribution to $A$ accumulated up to that time:
    \begin{align}
        \mathbb{P}(t<t_1~|~X_0=i)\mathbb{E}[e^{\lambda A(t)}~|~X_0=i, t<t_1]&=\phi_i(t)e^{\lambda g_it}\\ &=\phi_i^{(\lambda)}(t).
    \end{align}
    The second term describes the case that at least one transition away from the initial state $i$ occurs, so all possible transition times $t_1\leq t$ have to be considered, which contribute $e^{\lambda(g_it_1+h_{ij})}$ to $A$, depending on the next state $j\in V(\mathcal{G})$. Afterwards, the contribution is identical to a process starting in $j$ at time $t_1$, so we may express this term as:
    \begin{align}
        \mathbb{P}(t\geq t_1)\mathbb{E}[e^{\lambda A(t)}~|~X_0=i, t\geq t_1]&=\int_{[0,t]}^{}\sum_{j:(i,j)\in E(\mathcal{G})}e^{\lambda(g_it'+h_{ij})}\mathbb{E}[e^{\lambda A(t-t')}~|~X_0=j]~Q_{ij}(\text{d}t')\\ &=\sum_{j:(i,j)\in E(\mathcal{G})}\int_{[0,t]}^{}e^{\lambda(g_it'+h_{ij})}G_j(t-t',\lambda)Q_{ij}(\text{d}t')\\ &=\sum_{j:(i,j)\in E(\mathcal{G})}\int_{[0,t]}^{}G_j(t-t',\lambda)Q^{(\lambda)}_{ij}(\text{d}t').
    \end{align}
    Combining these two summands and taking the Laplace transform of $G(t,\lambda)$ yields:
    \begin{align}
        \tilde{G}_i(s,\lambda)=\tilde{\phi}_i^{(\lambda)}(s)+\sum_{j:(i,j)\in E(\mathcal{G})}\tilde{Q}^{(\lambda)}_{ij}(s)\tilde{G}_j(s,\lambda).
    \end{align}
    Written in matrix notation, we see that:
    \begin{align}
         \big(\mathbb{1}-\tilde{Q}^{(\lambda)}(s)\big)\tilde{G}(\lambda,s)=\tilde{\phi}^{(\lambda)}(s).
     \end{align}
    Moreover, since $\sum_{j}\tilde{Q}_{ij}(s)<1$ if $i$ is not an absorbing state, also $\sum_{j}\tilde{Q}^{(\lambda)}_{ij}(s)<1$ for sufficiently small $|\lambda|$, so also $\tilde{G}(\lambda,s)$ is finite around $\lambda =0$.

    \end{align}

<!-- proof-15: body_md -->
By Proposition \ref{Prop:time_additive_observable_mean}, the mean of $A(t)$ is given by:
    \begin{align}
        \mathbb{E}[A(t)]=\sum_{i\in V(\mathcal{G})}g_i\int_{0}^{t}p_i(t')\text{d}t',
    \end{align}
    so it suffices to show that $\frac{1}{t}\int_{0}^{t}p_i(t')\text{d}t'\overset{t\to\infty}{\longrightarrow}\pi_i$. Since $\lim_{t\to \infty}p_i(t)=\pi_i$, we have:
    \begin{align}
        \forall \epsilon >0:\exists t_\epsilon\geq 0:\forall t\geq t_\epsilon:|\pi_i-p_i(t)|<\epsilon.
    \end{align}
    So choose an arbitrary $\epsilon>0$ and the corresponding $t_\epsilon$. Then:
    \begin{align}
        \Big|\frac{1}{t}\int_{0}^{t}p_i(t')\text{d}t'-\pi_i\Big|&=\Big|\frac{1}{t}\int_{0}^{t_\epsilon}p_i(t')\text{d}t'+\frac{1}{t}\int_{t_\epsilon}^{t}p_i(t')\text{d}t'-\pi_i\Big|\\&\leq\Big|\frac{1}{t}\int_{0}^{t_\epsilon}p_i(t')\text{d}t'\Big|+\Big|\frac{1}{t}\int_{t_\epsilon}^{t}p_i(t')\text{d}t'-\pi_i\Big|\\&\leq\frac{t_\epsilon}{t}+\Big|\frac{1}{t}\int_{t_\epsilon}^{t}p_i(t')\text{d}t'-\pi_i\Big|\\&\leq\frac{t_\epsilon}{t}+\Big|\frac{t-t_\epsilon}{t}\epsilon+\frac{t_\epsilon}{t}\pi_i\Big|\\ &\leq \big(1-\frac{t_\epsilon}{t}\big)\epsilon+2\frac{t_\epsilon}{t},
    \end{align}
    using that $p_i(t)\in (\pi_i-\epsilon,\pi_i+\epsilon)$ for $t\geq t_\epsilon$ to bound the integral. Hence, we see that:
    \begin{align}
        \limsup_{t\to\infty}\Big|\frac{1}{t}\int_{0}^{t}p_i(t')\text{d}t'-\pi_i\Big|\leq\epsilon,
    \end{align}
    and since $\epsilon>0$ was arbitrary, we conclude that:
    \begin{align}
        \lim_{t\to\infty}\frac{1}{t}\int_{0}^{t}p_i(t')\text{d}t'=\pi_i.
    \end{align}

Observe that:
    \begin{align}
        \frac{\text{d}\ln{\tilde{K}_{ij}(s)}}{\text{d}\ln{s}}&=\frac{s}{\tilde{K}_{ij}(s)}\frac{\text{d}}{\text{d}s}\tilde{K}_{ij}(s)\\&=\frac{s}{\tilde{K}_{ij}(s)}\Big(\frac{\tilde{K}_{ij}(s)}{s}+\frac{s\frac{\text{d}}{\text{d}s}\Tilde{Q}_{ij}(s)}{1-\Tilde{Q}_i(s)}+\frac{\frac{\text{d}}{\text{d}s}\Tilde{Q}_{i}(s)}{1-\Tilde{Q}_i(s)}\tilde{K}_{ij}(s)\Big)\\ &=1+\frac{s\frac{\text{d}}{\text{d}s}\Tilde{Q}_{ij}(s)}{\Tilde{Q}_{ij}(s)}+\frac{\frac{\text{d}}{\text{d}s}\Tilde{Q}_{i}(s)}{\tilde{\phi}_i(s)}.
    \end{align}
    Consider the second term in the limit $s\to 0$. The denominator satisfies $\tilde{Q}_{ij}(0)=p_{ij}>0$ and for the numerator, we can use that $xe^{-x}\leq e^{-1}$ for all $x\geq 0$ to see that for all $y\geq 0$, we have
    \begin{align}
        s\frac{\text{d}}{\text{d}s}\Tilde{Q}_{ij}(s)&=-s\int_{0}^{\infty}te^{-st}Q_{ij}(\text{d}t)\\ &\geq-s\int_{0}^{y}te^{-st}Q_{ij}(\text{d}t)-e^{-1}\int_{y}^{\infty}Q_{ij}(\text{d}t),

    and also $s\frac{\text{d}}{\text{d}s}\Tilde{Q}_{ij}(s)\leq 0$.

    \begin{align}
        \lim_{s\to 0}s\frac{\text{d}}{\text{d}s}\Tilde{Q}_{ij}(s)&=0 
        \Rightarrow \lim_{s\to 0}\frac{s\frac{\text{d}}{\text{d}s}\Tilde{Q}_{ij}(s)}{\Tilde{Q}_{ij}(s)}=0,
    \end{align}

    \end{align}

    \begin{align}
        \lim_{s\to 0}\frac{\text{d}}{\text{d}\ln s}\ln\tilde{K}_{ij}(s)=1+\lim_{s\to 0}\frac{\frac{\text{d}}{\text{d}s}\Tilde{Q}_i(s)}{\tilde{\phi}_i(s)},
    \end{align}

        \frac{\text{d}}{\text{d}s}\Tilde{Q}_j(s)&=-\int_{0}^{\infty}te^{-st}Q_{j}(\text{d}t)<0\\ \tilde{\phi}_j(s)&=\frac{1-\Tilde{Q}_j(s)}{s}>0,

    \end{align}
    using the standard bound $ste^{-st}\leq 1-e^{-st}$ for $st\geq 0$.

    \begin{align}
        \alpha_n>\alpha_m~\land~\pi_m>0 \Rightarrow \pi_n=0.
    \end{align}

    \begin{align}
        \lim_{s\to 0}\frac{\text{d}}{\text{d}\ln s}\ln\frac{\tilde{K}_{mi}(s)}{\tilde{K}_{nj}(s)}=\lim_{s\to 0}\frac{\text{d}}{\text{d}\ln s}\ln\tilde{K}_{mi}(s)-\lim_{s\to 0}\frac{\text{d}}{\text{d}\ln s}\ln\tilde{K}_{nj}(s)=\alpha_n-\alpha_m>0.
    \end{align}
    Hence, there is a $\delta >0$ such that $\forall s\leq\delta$:
    \begin{align}
        \frac{\alpha_n-\alpha_m}{2}&\leq\frac{\text{d}}{\text{d}\ln s}\ln\frac{\tilde{K}_{mi}(s)}{\tilde{K}_{nj}(s)}\leq 3\frac{\alpha_n-\alpha_m}{2} \\ \Rightarrow\frac{\alpha_n-\alpha_m}{2}\ln{\frac{\delta}{s}}&\leq\ln\frac{\tilde{K}_{mi}(\delta)}{\tilde{K}_{nj}(\delta)}+\ln\frac{\tilde{K}_{nj}(s)}{\tilde{K}_{mi}(s)}\leq 3\frac{\alpha_n-\alpha_m}{2}\ln{\frac{\delta}{s}}, 
    \end{align}
    by integration from $s$ to $\delta$. Letting $s\to 0$, it follows that:
    \begin{align}
        \lim_{s\to 0}\ln\frac{\tilde{K}_{nj}(s)}{\tilde{K}_{mi}(s)}=\infty\Leftrightarrow \lim_{s\to 0}\frac{\tilde{K}_{mi}(s)}{\tilde{K}_{nj}(s)}=0.
    \end{align}
    Also note that $\lim_{s\to0}\frac{\tilde{K}_{ij}(s)}{\tilde{K}_{il}(s)}=\frac{p_{ij}}{p_{il}}\in (0,\infty)$.
    Thus, for the ratio of the weights of two spanning trees $\mathcal{T}_m$ and $\mathcal{T}_n$ rooted at $m$ and $n$, respectively, we have:
    \begin{align}
        \lim_{s\to 0}\frac{w(\mathcal{T}_n)}{w(\mathcal{T}_m)}&=\lim_{s\to 0}\frac{\prod_{(a,b)\in E(\mathcal{T}_n)}\tilde{K}_{ab}(s)}{\prod_{(c,d)\in E(\mathcal{T}_m)}\tilde{K}_{cd}(s)}\\&=\Big(\lim_{s\to 0}\frac{\tilde{K}_{mi}(s)}{\tilde{K}_{nj}(s)}\Big)\frac{p_{nj}}{p_{mi}}\frac{\prod_{(a,b)\in E(\mathcal{T}_n)}p_{ab}}{\prod_{(c,d)\in E(\mathcal{T}_m)}p_{cd}}\\&=0,
    \end{align}
    where $(m,i)\in E(\mathcal{T}_n)$ and $(n,j)\in E(\mathcal{T}_m)$.     \begin{align}
        \frac{\pi_n}{\pi_m}=\lim_{s\to 0}\frac{\sum_{\mathcal{T}_{[n]}}w(\mathcal{T})}{\sum_{\mathcal{T}_{[m]}}w(\mathcal{T})}=0,
    \end{align}
    since $\lim_{s\to 0}\frac{s}{\tilde{K}_{ij}(s)}=\lim_{s\to 0}\frac{1-\tilde{Q}_i(s)}{\tilde{Q}_{ij}(s)}=0$ and because the number of spanning trees is finite in a finite graph.
    Since $m,n\in V(\mathcal{G})$ were arbitrary, $\pi_n,\pi_m>0$ implies $\alpha_n=\alpha_m$.

\label{theorem:weak_ergodicity_breaking}
    Let $A(t)$ be a time-additive state observable on a semi-Markov network on $\mathcal{G}=(V(\mathcal{G}),E(\mathcal{G}))$ with a unique limiting state distribution $\pi$ for all initial conditions. Let the limit $\lim_{s\to 0}\frac{\text{d}}{\text{d}\ln s}\ln\tilde{K}_{ij}(s)=1-\alpha_i$ exist for all $(i,j)\in E(\mathcal{G})$. Then:
    \begin{align}
        \lim_{t\to \infty}\text{Var}\big(\frac{A(t)}{t}\big)&=\big(1-\alpha\big)\Big(\sum_{j\in V(\mathcal{G})}\pi_jg_j^2-\big(\sum_{j\in V(\mathcal{G})}\pi_jg_j\big)^2\Big),\label{eq:convergence_variance}
    \end{align}
    where $\alpha =\alpha_j$ for all $j\in V(\mathcal{G})$ such that $\pi_j>0$.
    Moreover, $A(t)$ exhibits weak ergodicity breaking if there is at least one state $n\in V(\mathcal{G})$ with $\pi_n>0$ and $\alpha_n<1$ and $ m\in V(\mathcal{G})$ with $\pi_n,\pi_m>0$ such that $g_n\neq g_m$.

If the limiting distribution is concentrated on an absorbing state, the statement is trivial. In the following, we therefore assume that the states are not absorbing. 

   Consider the long-time limit of the variance of the time average of $A(t)$, given by:
    \begin{align}
        \text{Var}\big(\frac{A(t)}{t}\big)=&\mathbb{E}\Big[\Big|\frac{A(t)}{t}-\mathbb{E}\Big[\frac{A(t)}{t}\Big]\Big|^2\Big]\\=&\frac{1}{t^2}\big(\mathbb{E}[A(t)^2]-\mathbb{E}[A(t)]^2\big),
    \end{align}
    so $\text{Var}\big(\frac{A(t)}{t}\big)\overset{t\to\infty}{\longrightarrow}0$ means that $\frac{A(t)}{t}$ converges to its mean in $L^2$. Since $\big|\frac{A(t)}{t}\big|\leq \max_{i\in V(\mathcal{G})}\{|g_i|\}$, i.e., $\frac{A(t)}{t}$ is uniformly bounded, almost-sure convergence to a constant would imply $L^2$-convergence by dominated convergence. Hence a strictly positive limiting variance rules out almost-sure, and indeed, convergence in probability to a constant, so we can prove weak ergodicity breaking from showing that the limit is different from zero.

    We calculate the second moment from the second $\lambda$-derivative of $\tilde{G}(\lambda,s)$ at $\lambda=0$. 
    Taking the second derivative with respect to $\lambda$ of Eq.~(\ref{eq:generalized_Feynmann_Kac}) yields:
    \begin{align}
        &\big(-\frac{\partial^2}{\partial\lambda^2}\tilde{Q}^{(\lambda)}(s)\big)\tilde{G}(\lambda,s)-2\big(\frac{\partial}{\partial\lambda}\tilde{Q}^{(\lambda)}(s)\big)\big(\frac{\partial}{\partial\lambda}\tilde{G}(\lambda,s)\big)+\big(\mathbb{1}-\tilde{Q}^{(\lambda)}(s)\big)\frac{\partial^2}{\partial\lambda^2}\tilde{G}(\lambda,s)=\frac{\partial^2}{\partial\lambda^2}\tilde{\phi}^{(\lambda)}(s)\\ \Rightarrow&\big(\mathbb{1}-\tilde{Q}^{(\lambda)}(s)\big)\frac{\partial^2}{\partial\lambda^2}\tilde{G}(\lambda,s)=\frac{\partial^2}{\partial\lambda^2}\tilde{\phi}^{(\lambda)}(s)+\big(\frac{\partial^2}{\partial\lambda^2}\tilde{Q}^{(\lambda)}(s)\big)\tilde{G}(\lambda,s)+2\big(\frac{\partial}{\partial\lambda}\tilde{Q}^{(\lambda)}(s)\big)\big(\frac{\partial}{\partial\lambda}\tilde{G}(\lambda,s)\big)    
    \end{align}
    Again, we evaluate this expression at $\lambda = 0$, using that $\tilde{G}_i(s,0)=\frac{1}{s}$ for all $i\in V(\mathcal{G})$ and express $(\mathbb{1}-\tilde{Q}^{(\lambda)}(s))\big|_{\lambda=0}=(\mathbb{1}-\tilde{Q}(s))$ in terms of $\tilde{W}(s)$:
    \begin{align}
        \text{diag}\big(\tilde{\phi}(s)\big)\big(s\mathbb{1}-\tilde{W}^T(s)\big)\frac{\partial^2}{\partial\lambda^2}\tilde{G}(\lambda,s)\Big|_{\lambda=0}=&\frac{\partial^2}{\partial\lambda^2}\big(\tilde{\phi}^{(\lambda)}(s)+\frac{1}{s}\tilde{Q}^{(\lambda)}(s)\mathbf{1}\big)\Big|_{\lambda=0}\\&+2\big(\frac{\partial}{\partial\lambda}\tilde{Q}^{(\lambda)}(s)\big)\big(\frac{\partial}{\partial\lambda}\tilde{G}(\lambda,s)\big)\Big|_{\lambda=0}.
    \end{align}
    For $s>0$, $\text{diag}\big(\tilde{\phi}(s)\big)\big(s\mathbb{1}-\tilde{W}^T(s)\big)$ can be inverted according to Proposition \ref{Prop:inverse_of_s-W}:
     \begin{align}
        \frac{\partial^2}{\partial\lambda^2}\tilde{G}(\lambda,s)\Big|_{\lambda=0}=&\big(s\mathbb{1}-\tilde{W}^T(s)\big)^{-1}\text{diag}\big(\tilde{\phi}(s)\big)^{-1}\Big(\frac{\partial^2}{\partial\lambda^2}\big(\tilde{\phi}^{(\lambda)}(s)+\frac{1}{s}\tilde{Q}^{(\lambda)}(s)\mathbf{1}\big)\\&+2\big(\frac{\partial}{\partial\lambda}\tilde{Q}^{(\lambda)}(s)\big)\big(\frac{\partial}{\partial\lambda}\tilde{G}(\lambda,s)\big)\Big)\Big|_{\lambda=0}.
    \end{align}
    We can evaluate the right-hand side using that:
    \begin{align}
        \frac{\partial}{\partial\lambda}\tilde{Q}^{(\lambda)}_{ij}(s)\Big|_{\lambda=0}&=-g_i\frac{\text{d}}{\text{d}s}\Tilde{Q}_{ij}(s),\\\frac{\partial^2}{\partial\lambda^2}\tilde{Q}^{(\lambda)}_{ij}(s)\Big|_{\lambda=0}&=g_i^2\frac{\text{d}^2}{\text{d}s^2}\Tilde{Q}_{ij}(s),\\ \frac{\partial^2}{\partial\lambda^2}\tilde{\phi}_i^{(\lambda)}(s)\Big|_{\lambda=0}&=g_i^2\frac{\text{d}^2}{\text{d}s^2}\tilde{\phi}_i(s)\\ &=g_i^2\frac{\text{d}^2}{\text{d}s^2}\frac{1-\Tilde{Q}_i(s)}{s}\\ &=g_i^2\big(2\frac{\tilde{\phi}_i(s)}{s^2}+\frac{2}{s^2}\frac{\text{d}}{\text{d}s}\Tilde{Q}_i(s)-\frac{1}{s}\frac{\text{d}^2}{\text{d}s^2}\Tilde{Q}_i(s)\big),
    \end{align}
    so:
    \begin{align}
        \frac{\partial^2}{\partial\lambda^2}\big(\tilde{\phi}^{(\lambda)}(s)+\frac{1}{s}\tilde{Q}^{(\lambda)}(s)\mathbf{1}\big)_{i}\Big|_{\lambda=0}&=\frac{2g_i^2}{s^2}\big(\tilde{\phi}_i(s)+\frac{\text{d}}{\text{d}s}\Tilde{Q}_i(s)\big)\\&=-\frac{2g_i^2}{s}\frac{\text{d}}{\text{d}s}\tilde{\phi}_i(s)
    \end{align}
    To evaluate $\partial_\lambda\tilde{G}_i^{(\lambda)}(s)$, we use the expression from Proposition \ref{Prop:time_additive_observable_mean}:
    \begin{align}
        \partial_\lambda\tilde{G}_i^{(\lambda)}(s)\big|_{\lambda=0}&=\frac{1}{s}\sum_{l\in V(\mathcal{G})}g_l\frac{\sum_{\mathcal{F}^{s,i\to l}_{[l,\ast]}}w(\mathcal{F})}{\sum_{\mathcal{T}^{s}_{[\ast]}}w(\mathcal{T})},
    \end{align}
    so:
    \begin{align}
        2\big(\partial_\lambda\tilde{Q}^{(\lambda)}(s)\partial_\lambda\tilde{G}^{(\lambda)}(s)\big)_{i}\big|_{\lambda=0}&=-\frac{2g_i}{s}\sum_{j,l\in V(\mathcal{G})}g_l\big(\frac{\text{d}}{\text{d}s}\Tilde{Q}_{ij}(s)\big)\frac{\sum_{\mathcal{F}^{s,j\to l}_{[l,\ast]}}w(\mathcal{F})}{\sum_{\mathcal{T}^{s}_{[\ast]}}w(\mathcal{T})}.
    \end{align}
    Combining these terms and expressing the inverse of $s\mathbb{1}-\tilde{W}^T(s)$ as in Proposition \ref{Prop:inverse_of_s-W}, we obtain:
    \begin{align}
        \frac{\partial^2}{\partial\lambda^2}\tilde{G}_i(\lambda,s)\Big|_{\lambda=0}=&\frac{2}{s}\sum_{j\in V(\mathcal{G})}g_j\frac{\sum_{\mathcal{F}^{s,i\to j}_{[j,\ast]}}w(\mathcal{F})}{\sum_{\mathcal{T}^{s}_{[\ast]}}w(\mathcal{T})}\Big(\frac{g_j}{s}\big(1+\frac{\frac{\text{d}}{\text{d}s}\Tilde{Q}_j(s)}{\tilde{\phi}_j(s)}\big)\\&-\sum_{m,l\in V(\mathcal{G})}g_l\big(\frac{\frac{\text{d}}{\text{d}s}\Tilde{Q}_{jm}(s)}{\tilde{\phi}_j(s)}\big)\frac{\sum_{\mathcal{F}^{s,m\to l}_{[l,\ast]}}w(\mathcal{F})}{\sum_{\mathcal{T}^{s}_{[\ast]}}w(\mathcal{T})}\Big).
    \end{align}
    Eq.~(\ref{eq:transient_semi_MCTT}) implies that the ratios of the sums over the spanning forests and spanning trees can be interpreted as the Laplace transforms of the occupation probabilities under suitable initial conditions. For the $s\to 0$ limit, as the limiting state distribution was unique, this implies that:
    \begin{align}
        \frac{\sum_{\mathcal{F}^{s,i\to j}_{[j,\ast]}}w(\mathcal{F})}{\sum_{\mathcal{T}^{s}_{[\ast]}}w(\mathcal{T})}=\frac{\pi_j}{s}+o(s^{-1}).
    \end{align}
    

    Therefore, we get the following small-$s$ regime:
    \begin{align}
        \frac{\partial^2}{\partial\lambda^2}\tilde{G}_i(\lambda,s)\Big|_{\lambda=0}=&\frac{2}{s^3}\sum_{j\in V(\mathcal{G})}g_j\pi_j\Big(g_j\big(1-\alpha_j\big)+\sum_{l\in V(\mathcal{G})}g_l\pi_l\alpha_j\Big)+o(s^{-3}),
    \end{align}
    using that $\lim_{s\to 0}\frac{\text{d}}{\text{d}\ln s}\ln\tilde{K}_{ij}(s)=1-\alpha_i=1+\lim_{s\to 0}\frac{\frac{\text{d}}{\text{d}s}\Tilde{Q}_i(s)}{\tilde{\phi}_i(s)}$, by assumption and by Lemma \ref{lemma:memory_kernels_small_s}.

    Now define $g_b:=-\min_{i\in V(\mathcal{G})}g_i$ and consider  
    \begin{align}
        B(t):=\int_{0}^{t}(g(X_{t'})+g_b)\text{d}t'=A(t)+g_bt
    \end{align}
    Note that 
    \begin{align}
        \frac{B(t)}{t}=\frac{A(t)}{t}+g_b\Rightarrow \text{Var}\big(\frac{B(t)}{t}\big)=\text{Var}\big(\frac{A(t)}{t}\big).
    \end{align}
    Moreover, $B(t)^2$, and hence $\mathbb{E}[B(t)^2]$, is non-decreasing in $t$ since $g(X_{t})+g_b\geq 0$, so Karamata’s Tauberian theorem can be applied, yielding the following long-time scaling for the second moment of $B(t)$:
    \begin{align}
        \mathbb{E}[B(t)^2]=\sum_{j\in V(\mathcal{G})}\pi_j(g_j+g_b)\Big((g_j+g_b)\big(1-\alpha_j\big)+\sum_{l\in V(\mathcal{G})}(g_l+g_b)\pi_l\alpha_j\Big)t^2+o(t^2).
    \end{align}
    By Proposition \ref{prop:time_average_mean_converges}, we have:
    \begin{align}
        \mathbb{E}[B(t)]^2&=\Big(\sum_{j\in V(\mathcal{G})}\pi_j(g_j+g_b)\Big)^2t^2+o(t^2)\\ &=\big(\sum_{j,l\in V(\mathcal{G})}\pi_j\pi_l(g_j+g_b)(g_l+g_b)\big)t^2+o(t^2).
    \end{align}
    Thus, we obtain:
    \begin{align}
        \lim_{t\to \infty}\text{Var}\big(\frac{A(t)}{t}\big)&= \lim_{t\to \infty}\text{Var}\big(\frac{B(t)}{t}\big)\\&=\lim_{t\to \infty}\frac{\mathbb{E}[B(t)^2]-\mathbb{E}[B(t)]^2}{t^2}\\&=\sum_{j\in V(\mathcal{G})}\pi_j(g_j+g_b)\Big((g_j+g_b)\big(1-\alpha_j\big)-\sum_{l\in V(\mathcal{G})}(g_l+g_b)\pi_l(1-\alpha_j)\Big)\\ &=\sum_{j\in V(\mathcal{G})}\pi_j(g_j+g_b)\big(1-\alpha_j\big)\sum_{l\in V(\mathcal{G})}(g_j-g_l)\pi_l.
    \end{align}
   By Lemma \ref{lemma:memory_kernels_small_s}, there is an $\alpha\in[0,1]$ such that $\pi_j>0$ implies $\alpha_j=\alpha$, which yields:
   \begin{align}
        \lim_{t\to \infty}\text{Var}\big(\frac{A(t)}{t}\big)&=\big(1-\alpha\big)\Big(\sum_{j\in V(\mathcal{G})}\pi_jg_j^2-\big(\sum_{j\in V(\mathcal{G})}\pi_jg_j\big)^2\Big),
    \end{align}
    which is zero if and only if $\alpha=1$, as in the finite-mean sojourn time case, or if the variance of $g_j$ with respect to the limiting state distribution $\pi$ vanishes, i.e., $g_i=g_j$ wherever $\pi_i,\pi_j>0$. 

     Since positive limiting variance is enough to rule out convergence in probability, it also rules out almost-sure convergence to a constant, and since the mean of the time average always converges according to Proposition \ref{prop:time_average_mean_converges}, this implies that $A(t)$ exhibits weak ergodicity breaking.

<!-- lemma-5: body_md -->
Let $f:[0,\infty)\to [0,\infty)$ be a locally integrable function for $t\geq 0$ and $\tilde{f}(s)$ its Laplace transform for $s>0$. If $\lim_{t\to \infty}f(t)$ exists and is finite, then

    \end{align}

<!-- proof-19: body_md -->
Let $c:=\lim_{t\to \infty}f(t)<\infty$. Then, for every $\epsilon >0$, there exists $t_\epsilon\in (0,\infty)$ such that
    \begin{align}
        \big|f(t)-c\big|<\epsilon
    \end{align}
    for $t>t_\epsilon$. Consider,
    \begin{align}
        s\tilde{f}(s)=s\int_{0}^{t_\epsilon}f(t)e^{-st}\text{d}t+s\int_{t_\epsilon}^{\infty}f(t)e^{-st}\text{d}t.
    \end{align}
    Since $\int_{0}^{t_\epsilon}f(t)\text{d}t<\infty$, 
    \begin{align}
        \lim_{s\to 0}s\int_{0}^{t_\epsilon}f(t)e^{-st}\text{d}t=0.
    \end{align}
    For the second integral, the following bounds must hold:
    \begin{align}
        (c-\epsilon)e^{-st_\epsilon} \leq s\int_{t_\epsilon}^{\infty}f(t)e^{-st}\text{d}t\leq (c+\epsilon)e^{-st_\epsilon},
    \end{align}
    so in the limit $s\to 0$, we get
    \begin{align}
        c-\epsilon \leq \lim_{s\to 0}s\int_{t_\epsilon}^{\infty}f(t)e^{-st}\text{d}t\leq c+\epsilon.
    \end{align}
    Since $\epsilon>0$ was arbitrary, this implies
    \begin{align}
        \lim_{s\to 0}s\tilde{f}(s)=\lim_{s\to 0}s\int_{t_\epsilon}^{\infty}f(t)e^{-st}\text{d}t=c=\lim_{t\to \infty}f(t).
    \end{align}

<!-- theorem-7: body_md -->
Let $f:[0,\infty)\to [0,\infty)$ be a locally integrable function and ultimately monotone. Let $\tilde{f}(s)$ be its Laplace transform. Then the following holds:
    \begin{align}
        f(t)\sim c\frac{t^{\alpha-1}}{\Gamma(\alpha)}~~\text{for }t\to \infty \Leftrightarrow \tilde{f}(s)\sim cs^{-\alpha}~~\text{for }s\to 0,
    \end{align}
    for $c>0$ and $\alpha >0$.
