\label{lem:HM}
	Let $\theta\in\overline{\mathbb{F}((X^{-1}))}$. Let $\mathcal{M}_\theta=(m_{i,j})_{i,j\geq 0}\in\mathbb{F}^{\mathbb N_0\times\mathbb N_0}$ be the Hankel matrix associated with $\theta$. 	\begin{enumerate}

		\item The integer $k$ is the degree of the denominator of a best approximation to $\theta=\sum_{j=1}^\infty m_{0,j-1}X^{-j}$.

<!-- defi-3: body_md -->
\label{def:KTseq}
	Let $\mathbb{F}_q$ be a finite field and let $\theta\in\overline{\mathbb{F}_q((X^{-1}))}$. For $n\in\mathbb N_0$, the $n$th element $x_n$ of the Kronecker-type sequence determined by $\theta$ is obtained as follows:
	\begin{enumerate}
		\item Expand $n$ in base $q$, that is, write $n=\sum_{i=0}^\infty n_iq^i$ with $n_i\in\{0,1,\ldots,q-1\}$, and associate with $n$ the polynomial $n(X)\in\mathbb{F}_q[X]$ defined by
		$$n(X)=\sum_{i=0}^\infty \eta(n_i)X^i,$$
		where $\eta:\{0,1,\ldots,q-1\}\to\mathbb{F}_q$ is a fixed bijection satisfying $\eta(0)=0$. 
		
		\item Compute the product $n(X)\theta$ and consider the Laurent-series expansion of its fractional part,
		$$\langle n(X)\theta\rangle=\sum_{i=1}^\infty u_iX^{-i}.$$
		
		\item Finally, set
		$$x_n=\sum_{i=1}^\infty \eta^{-1}(u_i)q^{-i},$$
		which is an element of the interval $[0,1]$. We refer to the resulting sequence as
		$$\bigl([\langle n(X)\theta\rangle]_q\bigr)_{n\geq 0}.$$
	\end{enumerate}
More generally, given $(\theta_1,\ldots,\theta_s)\in
\overline{\mathbb{F}_q((X^{-1}))}^s$, the associated $s$-dimensional Kronecker-type sequence is
$$\bigl(([\langle n(X)\theta_1\rangle]_q,\ldots,[\langle n(X)\theta_s\rangle]_q)\bigr)_{n\geq 0},$$
which is a sequence in $[0,1]^s$.

<!-- remark-2: body_md -->
\label{rem:KTseq}
	Note that the Kronecker-type sequence determined by $\theta$ can also be defined using the concept of an $\mathbb N_0\times\mathbb N_0$ generating matrix $\mathcal{M}_\theta$ over $\mathbb{F}_q$ from Definition~\ref{def:Hankelmatrix}, together with the digital method introduced in \cite{nie1987}. The digital method works as follows. 
	
	We again fix a bijection $\eta:\{0,1,\ldots,q-1\}\to\mathbb{F}_q$ satisfying $\eta(0)=0$, associate with each index $n\in\mathbb N_0$, expanded in base $q$, the digit vector $\vec{n}=(\eta(n_0),\eta(n_1),\ldots)^T\in\mathbb{F}_q^{\mathbb N_0}$, and choose a generating matrix
	$$\mathcal{C}:=(c_{i,j})_{i\geq 0,j\geq 0}\in\mathbb{F}_q^{\mathbb N_0\times\mathbb N_0}.$$
	We then compute
	$$\mathcal{C}\vec{n}=:\vec{y}_n=(y_{n,0},y_{n,1},\ldots)^T\in\mathbb{F}_q^{\mathbb N_0}.$$
	Finally, the $n$th point of the sequence $(x_n)_{n\geq 0}$ is defined by
	$$x_n:=\sum_{j=0}^\infty \eta^{-1}(y_{n,j})q^{-j-1}.$$
	
	If the generating matrix $\mathcal{C}$ is chosen to be the identity matrix, the resulting sequence is the well-known van der Corput sequence in base $q$ (see also Example~\ref{ex:vdc} in Section~\ref{sec:4}). 
	
	If, instead, the generating matrix $\mathcal{C}$ is chosen to be $\mathcal{M}_\theta$, then the digital method constructs the Kronecker-type sequence
	$$\bigl([\langle n(X)\theta\rangle]_q\bigr)_{n\geq 0},$$
	by Definition~\ref{def:Hankelmatrix} (see, e.g., \cite{LarNie1993,larnie1993}).                                                                                                                                              
	
	Although the Hankel matrix $\mathcal{M}_\theta$ can be defined over an arbitrary field $\mathbb{F}$, the construction of a sequence in $[0,1]$ by this method relies on the finiteness of the field $\mathbb{F}$.  
		
	More generally, an $s$-dimensional digital sequence is generated by
	$s$ matrices
	$\mathcal{C}^{(1)},\ldots,\mathcal{C}^{(s)}$ with
	$\mathcal{C}^{(i)}\in\mathbb{F}_q^{\mathbb N_0\times\mathbb N_0}$ for $1\leq i\leq s$.
	Writing
	$$
	\mathcal{C}^{(i)}\vec{n}
	=(y_{n,0}^{(i)},y_{n,1}^{(i)},\ldots)^T,
	$$
	its $n$th point is
	$$
	\boldsymbol{x}_n
	=
	\left(
	\sum_{j=0}^{\infty}\eta^{-1}(y_{n,j}^{(1)})q^{-j-1},
	\ldots,
	\sum_{j=0}^{\infty}\eta^{-1}(y_{n,j}^{(s)})q^{-j-1}
	\right)\in[0,1]^s.
	$$

\label{algo:OstrRep}
	Let $\theta\in\overline{\mathbb{F}((X^{-1}))}$ and let $P(X)\in\mathbb{F}[X]$. If $P(X)=0$, set
	\[
	(z_0,z_1,z_2,\ldots)=(0,0,0,\ldots)\in\mathbb{F}^{\mathbb N_0}.

	\begin{itemize}
		\item Set $P_r(X):=P(X)$. Choose $z_r\in\mathbb{F}\setminus\{0\}$ such that
		\[
		P_{r-1}(X):=P_r(X)-z_rB_r(X)
		\]
		satisfies $\deg( P_{r-1}(X))<r$.
		
		\item For $i\in\{1,\ldots,r-1\}$, suppose that $P_i(X)$ has already been defined. If $\deg( P_i(X))<i$, set
		\[
		z_i:=0
		\qquad\text{and}\qquad
		P_{i-1}(X):=P_i(X).
		\]
		Otherwise, choose $z_i\in\mathbb{F}\setminus\{0\}$ such that
		\[
		P_{i-1}(X):=P_i(X)-z_iB_i(X)
		\]
		satisfies $\deg (P_{i-1}(X))<i$.

	P(X)=\sum_{n=0}^{r}z_nB_n(X).

Hence the first item is an easy consequence using the definition of $\mathcal{U}_\phi$ and $\mathcal{L}_\phi$.\\
	
	We next prove the Hankel property in item 3. Write
	\[
	\mathcal{L}_{\theta}\mathcal{I}_{\theta}\mathcal{U}_{\theta}
	=:(k_{i,j})_{i,j\geq 0}.

	By the definitions of $\mathcal{L}_{\theta}$ and
	$\mathcal{U}_{\theta}$, and using the fact that

		&=\boldsymbol{l}_i\mathcal{I}_{\theta}\boldsymbol{u}_j\\

<!-- defi-9: body_md -->
\label{def:sobol_construction}
	Let $q=p^r$, where $p\in\mathbb P$, $r\in\mathbb N$, and $s\in\mathbb N$. Choose $s$ pairwise coprime monic polynomials
	\[
	b_1(X),\ldots,b_s(X)\in\mathbb{F}_q[X]
	\]
	of degrees $e_1,\ldots,e_s\geq 1$.
	
	For each $i\in\{1,\ldots,s\}$, construct a matrix
	$\mathcal{C}^{(i)}\in\mathbb{F}_q^{\mathbb N_0\times\mathbb N_0}$ using the base polynomial
	\[
	b_i(X)=X^{e_i}+a^{(i)}_{e_i-1}X^{e_i-1}+\cdots+a^{(i)}_1X+a^{(i)}_0,
	\]
	where $a^{(i)}_{e_i-1},\ldots,a^{(i)}_1,a^{(i)}_0\in\mathbb{F}_q$, by recursively defining its columns. We begin with the $e_i$ initial columns
	\[
	\begin{pmatrix}|&&|\\
		c^{(i)}_0&\cdots&c^{(i)}_{e_i-1}\\
		|&&|
	\end{pmatrix}
	=
	\begin{pmatrix}
		B_i\\
		0\\
		0\\
		\vdots
	\end{pmatrix}
	\in\mathbb{F}_q^{\mathbb N_0\times e_i},
	\]
	where $B_i$ is an $e_i\times e_i$ NUT matrix over $\mathbb{F}_q$. All subsequent columns are defined recursively by
	\begin{equation}\label{equ:sobol_construction}
		c^{(i)}_{e_i+r}
		+a^{(i)}_{e_i-1}c^{(i)}_{e_i-1+r}
		+\cdots
		+a^{(i)}_1c^{(i)}_{1+r}
		+a^{(i)}_0c^{(i)}_{r}
		=
		\begin{pmatrix}
			0\\
			\vdots\\
			0\\
			c^{(i)}_{r}
		\end{pmatrix},
	\end{equation}
	where the column on the right-hand side begins with $e_i$ zero entries.
	
	We call the sequence $(\boldsymbol{x}_n)_{n\geq 0}$ constructed from the matrices
	$\mathcal{C}^{(1)},\ldots,\mathcal{C}^{(s)}$ via the algorithm described in Remark~\ref{rem:KTseq} the \emph{Sobol'-type sequence} associated with the base polynomials
	$b_1(X),\ldots,b_s(X)\in\mathbb{F}_q[X]$.

\label{lem:inversematrices}

	(\mathcal{C}^{(1)})^{-1}=:\mathcal{C}^{(-1)},\ldots,
	(\mathcal{C}^{(s)})^{-1}=:\mathcal{C}^{(-s)}.

<!-- proof-5: body_md -->
Let
	\[
	\mathcal{X}:=
	\begin{pmatrix}
		\boldsymbol{0}\\
		\mathcal{I}
	\end{pmatrix},
	\]
	where $\mathcal{I}$ denotes the infinite identity matrix over $\mathbb{F}_q$. By the recursion in \eqref{equ:sobol_construction}, the matrix $\mathcal{C}^{(i)}$ satisfies
	\[
	\mathcal{C}^{(i)}b_i(\mathcal{X})
	=
	\mathcal{X}^{e_i}\mathcal{C}^{(i)}.
	\]
	Multiplying this identity from the left by $(\mathcal{C}^{(i)})^{-1}=\mathcal{C}^{(-i)}$ and from the right by $\mathcal{C}^{(-i)}$ gives
	\begin{equation}\label{eq:sobol_construction}
		b_i(\mathcal{X})\mathcal{C}^{(-i)}
		=
		\mathcal{C}^{(-i)}\mathcal{X}^{e_i}.
	\end{equation}
	In particular, if
	\[
	\mathcal{C}^{(-i)}
	=
	\begin{pmatrix}
		|&|&&\\
		c^{(-i)}_0&c^{(-i)}_1&\cdots&\\
		|&|&&
	\end{pmatrix},
	\]
	then \eqref{eq:sobol_construction} implies
	\[
	b_i(\mathcal{X})c^{(-i)}_j=c^{(-i)}_{j+e_i},
	\qquad j\in\mathbb N_0.
	\]
	
	For a polynomial
	\[
	p(X)=\sum_{j=0}^r a_jX^j\in\mathbb{F}_q[X],
	\]
	we write
	\[
	\vec{p}:=(a_0,a_1,\ldots,a_r,0,0,\ldots)^{\mathsf T}\in\mathbb{F}_q^{\mathbb N_0}.
	\]
	Then
	\begin{equation}\label{eq:identity_vec_Xi}
		\mathcal{I}\vec{p}
		=
		p(\mathcal{X})
		\begin{pmatrix}
			1\\0\\0\\\vdots
		\end{pmatrix}.
	\end{equation}
	
	We now verify the linear independence criterion from Definition~\ref{def:dig_ues_seq}. Let $l_1,\ldots,l_s\in\mathbb N_0$ and put
	\[
	m:=l_1e_1+\cdots+l_se_s.
	\]
	We have to show that the $m$ row vectors
	\[
	\bigl(c^{(i)}_{j,0},c^{(i)}_{j,1},\ldots,c^{(i)}_{j,m-1}\bigr),
	\qquad
	i\in\{1,\ldots,s\},\quad
	j\in\{0,\ldots,e_il_i-1\},
	\]
	are linearly independent over $\mathbb{F}_q$.
	
	By Definition~\ref{def:dig_ues_seq}, it suffices to show that the
common kernel of the corresponding $m\times m$ matrix is trivial.
Let $\vec n$ belong to this kernel. Thus, $\vec n$ is the coefficient
vector of a polynomial $n(X)\in\mathbb{F}_q[X]$ with $\deg(n)<m$, and the
first $e_il_i$ entries of $\mathcal{C}^{(i)}\vec n$ vanish for every
$i\in\{1,\ldots,s\}$. Therefore, there exists
$v^{(i)}(X)\in\mathbb{F}_q[X]$ such that
\[
\mathcal{C}^{(i)}\vec n=\mathcal{X}^{e_il_i}\vec v^{(i)}.
\]
	Since $\mathcal{C}^{(-i)}\vec{v}^{(i)}$ is again the coefficient vector of a polynomial, say $w^{(i)}(X)$, it follows from \eqref{eq:sobol_construction} and \eqref{eq:identity_vec_Xi} that
	\[
	n(\mathcal{X})
	=
	b_i(\mathcal{X})^{l_i}w^{(i)}(\mathcal{X}).
	\]
	Consequently,
	\[
	b_i(X)^{l_i}\mid n(X)
	\qquad\text{for every }i\in\{1,\ldots,s\}.
	\]
	Because the polynomials $b_1(X),\ldots,b_s(X)$ are pairwise coprime, their powers are pairwise coprime, and hence
	\[
	\prod_{i=1}^s b_i(X)^{l_i}\mid n(X).
	\]
	The degree of the polynomial on the left-hand side is
	\[
	\sum_{i=1}^s l_i\deg(b_i(X))
	=
	\sum_{i=1}^s l_ie_i
	=
	m.
	\]
	Since $\deg(n(X))<m$, this is possible only if $n(X)=0$. Therefore, the stated row vectors are linearly independent.
	
	By Definition~\ref{def:dig_ues_seq}, the matrices
	$\mathcal{C}^{(1)},\ldots,\mathcal{C}^{(s)}$ generate a digital
	$(0,\boldsymbol{e},s)$-sequence over $\mathbb{F}_q$, where
	\[
	\boldsymbol{e}=(e_1,\ldots,e_s).
	\]
