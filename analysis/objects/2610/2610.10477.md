<!-- lemma-3: body_md -->
\label{lem:odd-rounding}

<!-- proof-2: body_md -->

    f(X) = g(X) = \operatorname{sign}\left(\frac{1}{\sqrt{{n}}}\sum_{i = 1}^{n} \operatorname{He}_{d}(X_i)\right)
\end{equation}

$\frac{\pi}{2}$ is the maximum possible integrality gap for PSD matrices.
The standard construction for little Grothendieck uses ${d}=1$ but it turns out that arbitrary ${d}$ has the same gap.

$s\in[-1,1]$. Thus, for independent uniform $s,t$ we have
\begin{equation}\label{eq:odd-threshold-rounding}
 \operatorname{\mathbb{E}}_{s,t}c(R_s f,R_t g)=c(f,g).
\end{equation}
Thus $\mathcal{B} \subseteq \mathcal{K}$. The same proof shows that if odd $|f|,|g| \leq 1$ depend on only finitely many grade-one variables, then $c(f,g) \in \mathcal C$.

We next prove compactness of $\mathcal B$.
Take a sequence of odd $(f_n,g_n), |f_n|,|g_n| \leq 1$. By \cref{thm:closure}(i),
after passing to a subsequence there are
$p_d,r_d\in\overline{\mathcal H}_d$ such that
\begin{equation}\label{eq:ou-1}
 (\Pi_d f_n,\Pi_d g_n)_{d \in \mathbb{N}}
 \xrightarrow{d}(p_d,r_d)_{d\in \mathbb{N}}
\end{equation}
with convergence of the mixed moments of every finite
subtuple. For each fixed $D$,
\[\sum_{d \le D} \|p_d\|_2^2 = \lim_{n \to \infty} \sum_{d \le D} \|\Pi_d f_n\|_2^2 \overset{\text{Parseval}}{\leq} 1\,.\]
Taking sup over $D$ shows $\sum_d\|p_d\|_2^2\le1$.
So $f=\sum_dp_d$ is well-defined in $\overline{L_2}$ and $\Pi_d f = p_d$.
The same holds for $g$.
Both are odd since their even-degree components vanish.

Note that we may \emph{not} have $f_n \xrightarrow{d} f$ since Hermite mass
for $f_n$ could escape to higher and higher degree.
We still prove $|f|,|g| \leq 1$. Fix $0<r<1$ and define the graded Ornstein--Uhlenbeck operator $T_r = \sum_d r^d \Pi_d$.
\cref{eq:ou-1} together with
\[
 \sup_n\left\|T_rf_n-\sum_{d\le D}r^d\Pi_df_n\right\|_2
 \le r^{D+1},
\]
implies $T_rf_n\xrightarrow{d}T_rf$.
For the nonnegative continuous function
$\varphi(x)=(|x|-1)_+$, the Portmanteau inequality gives
\[
 \operatorname{\mathbb{E}}\varphi(T_rf)
 \le\liminf_{n\to\infty}\operatorname{\mathbb{E}}\varphi(T_rf_n)=0,
\]
since $|T_rf_n|\le1$. Finally, $T_rf\to f$ in $L_2$
as $r\uparrow1$, and $\varphi$ is $1$-Lipschitz, so
$\operatorname{\mathbb{E}}\varphi(f)=0$. Hence $|f|\le1$ almost surely.
The argument for $g$ is identical.

We have
\[c_d(f_n,g_n) = \operatorname{\mathbb{E}}[(\Pi_d f_n)(\Pi_d g_n)] \underset{n \to\infty}{\longrightarrow} \operatorname{\mathbb{E}}[(\Pi_d f)(\Pi_d g)] = c_d(f,g)
\]
Thus every sequence in $\mathcal{B}$ has a coordinatewise convergent subsequence.
Since the coordinatewise topology is metrizable, sequential compactness implies $\mathcal{B}$ is (topologically) compact.

We now prove
$\mathcal B\subseteq\overline{\mathcal C}$.
Fix odd $(f,g), |f|,|g| \leq 1$, and put
\[
 f^{(D)}=\sum_{d\le D}\Pi_df,\qquad
 g^{(D)}=\sum_{d\le D}\Pi_dg,\qquad
 \varepsilon_D=
 \|f-f^{(D)}\|_2+\|g-g^{(D)}\|_2.
\]
Then $\varepsilon_D\to0$ as $D \to \infty$.
\cref{thm:closure} (ii) gives odd finite-dimensional
polynomials $F_n,G_n$ of degree at most $D$ such that, also with convergence of all mixed moments,
\[
 (\Pi_dF_n,\Pi_dG_n)_{d\le D}
 \xrightarrow{d}
 (\Pi_df,\Pi_dg)_{d\le D}.
\]
In particular,
\[
 c(F_n,G_n)\longrightarrow c(f^{(D)},g^{(D)})
 \quad\text{in }\ell_1,
 \qquad
 \|F_n\|_2\to\|f^{(D)}\|_2\le1,
\]
and similarly for $G_n$.

Let $\chi(u)=\max\{-1,\min\{u,1\}\}$ be the clipping function to $[-1,1]$.
Convergence in distribution and uniformly bounded fourth
moments imply
 \begin{align}\label{eq:clip1}
 \|F_n-\chi(F_n)\|_2
 &\longrightarrow
 \|f^{(D)}-\chi(f^{(D)})\|_2
 \leq\|f^{(D)}-f\|_2,\\ \notag
 \|G_n-\chi(G_n)\|_2
 &\longrightarrow
 \|g^{(D)}-\chi(g^{(D)})\|_2
 \leq\|g^{(D)}-g\|_2.
 \end{align}
The right inequalities use that clipping is the pointwise projection onto $[-1,1]$.
Orthogonality and Cauchy--Schwarz give the general continuity relations,
\begin{equation}\label{eq:coefficient-continuity}
 \begin{aligned}
 \|c(f,g)\|_1&\le\|f\|_2\|g\|_2,\\
 \|c(f,g)-c(f',g')\|_1
 &\le\|f-f'\|_2\|g\|_2+\|f'\|_2\|g-g'\|_2.
 \end{aligned}
\end{equation}


Using \cref{eq:clip1} and \eqref{eq:coefficient-continuity}, we obtain
\[
 \limsup_{n \to \infty}
 \|c(\chi(F_n),\chi(G_n))-c(f,g)\|_1
 \le2\varepsilon_D.
\]
The clipped functions are bounded by 1, odd, and finite-dimensional,
so their coefficient sequences belong to $\mathcal C$ by
the sentence after \eqref{eq:odd-threshold-rounding}. Letting $D\to\infty$
proves the inclusion.

It remains to pass from $\mathcal{B}$ to $\mathcal{K}$.
The set $\mathcal K$ is exactly the set of barycenters of
probability measures on $\mathcal B$.
Indeed, if $c(\mu) \in \mathcal{K}$, then $c(\mu) = \operatorname{\mathbb{E}}_{(f,g) \sim \mu}c(f,g)$
is also a mixture of elements $c(f,g) \in \mathcal{B}$.
Conversely, a probability
measure on $\mathcal B$ can be lifted\footnote{
The space of bounded odd pairs is a closed subset of
$\overline{L_2}^{\,2}$, hence Polish. Its coefficient map
onto $\mathcal B$ is continuous by
\eqref{eq:coefficient-continuity}. The Jankov--von Neumann
theorem gives a section measurable for the completion of
the given probability measure, which supplies the lift.}
to a measure $\mu$ on odd pairs $(f,g), |f|,|g| \leq 1$,
and then threshold rounding in \eqref{eq:odd-threshold-rounding} ensures $f,g \in \{\pm1\}$ without changing $c(\mu)$.

Probability measures on the compact metric space
$\mathcal B$ form a compact space for weak convergence.
Their barycenters depend continuously on each coordinate,
so $\mathcal K$ is coordinatewise compact and convex.

The inclusion map $\mathcal B\to\ell_1$ is Borel measurable,
and $\|c\|_1\le1$ on $\mathcal B$, so these barycenters are
also Bochner integrals in $\ell_1$. Since
$\overline{\mathcal C}$ is closed and convex
and contains $\mathcal B$, it contains every such
barycenter. Hence
$\mathcal K\subseteq\overline{\mathcal C}$.
Conversely, $\mathcal C\subseteq\mathcal K$, and
coordinatewise compactness makes $\mathcal K$ closed
in $\ell_1$. Therefore
$\mathcal K=\overline{\mathcal C}$.

 \left(\sum_{k\ge0}\rho^{k/2}\psi_{k+l}(v)^2\right)^{1/2}.
 \end{aligned}
\]
For fixed $p,l$ and $0<\rho<1$,
\[
 \frac{\sqrt{(k+p)!(k+l)!}}{k!}\rho^{k/2}
 \le(k+p+l+1)^{(p+l)/2}\rho^{k/2}\longrightarrow0\quad(k\to\infty).
\]
Mehler's identity with equal arguments gives
\[
 \sum_{k\ge0}\rho^{k/2}\psi_{k+p}(u)^2
 \le\frac{\rho^{-p/2}}{\sqrt{1-\rho}}
 \exp\!\left(\frac{\sqrt\rho\,u^2}{1+\sqrt\rho}\right).
\]
Apply the preceding sum inequality to tails. The displayed limit and
these bounds for $(p,u)$ and $(l,v)$ give uniform convergence on compact
sets of $u,v\in\mathbb{R}$.
