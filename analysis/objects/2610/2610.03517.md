<!-- head-1: body_md -->
The \emph{shift graph} $S_k$ is the graph whose vertices are the pairs $(a,b)$ with $1\leq a<b\leq k$, and two vertices $(a,b)$ and $(c,d)$ are joined by an edge if $b=c$ or $a=d$. These graphs were introduced by Erd\H{o}s and Hajnal \cite{shift}, who showed that $S_k$ is triangle-free on $n=\binom{k}{2}$ vertices of chromatic number $\Omega(\log n)$. Ordering the vertices lexicographically, the resulting ordered graph avoids every three-edge ordered path $x_1x_2x_3x_4$ with $x_1,x_3<x_2,x_4$ as an induced subgraph.
<!-- end head-1 -->

For infinitely many $n$, there exists a triangle- and cross-path-free ordered graph on $n$ vertices with independence number $n^{1-\Omega(1/\log \log n)}$, and thus chromatic number $n^{\Omega(1/\log \log n)}$.

<!-- lemma-4: body_md -->
\label{lemma:multidim}
Let $N$ and $t$ be integers, and let $\varepsilon\in \{-1,0,1\}^{t}$ be a sign-pattern containing $p$ non-zero entries. If $S\subset [N]^t$ such that $|S|>pN^{t-1}$, then there exist $x,y\in S$ such that $\text{sign}(x-y)=\varepsilon$.
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
We prove this by induction on $p$. In case $p=0$, the statement is trivial as $S$ is non-empty and we can take any $x=y\in S$. Assume that $p\geq 1$ and let $j\in  [t]$ be an index such that $\varepsilon_j\neq 0$. Let $\varepsilon'\in \{-1,0,1\}^t$ be the sign-pattern  which agrees with $\varepsilon$ on every index, except for $j$, and $\varepsilon'_j=0$. Delete every element $z\in S$ for which there exists no $z'\in S$ with $z'_j>z_j$ and $z'_i=z_i$ for $i\in [t]\setminus \{j\}$; let $S'$ be the resulting set. We deleted at most $N^{t-1}$ elements, so $|S'|\geq |S|-N^{t-1}>(p-1)N^{t-1}$. Therefore, by our induction hypothesis, $S'$ contains two elements, $x'$ and $y'$, such that $\text{sign}(x'_i-y'_i)=\varepsilon'_i$. In particular, $x'_j=y'_j$. If $\varepsilon_j=1$, we can find an element $x\in S$ such that $x_j>x'_j$. But then taking $y=y'$, the pair $(x,y)$ satisfies the required  properties. Similarly, if $\varepsilon_j=-1$, we can find $y\in S$ such that $y_j>y'_j$. But then setting $x=x'$, the pair $(x,y)$ suffices.

<!-- lemma-6: body_md -->
\label{lemma:box_realization}
$G$ can be realized as an intersection graph of boxes in $\mathbb{R}^3$.
<!-- end lemma-6 -->
<!-- proof-6: body_md -->
Let $b=10s$, and let 
$$\overline{x}=\sum_{h=1}^{r}x_hb^{r-h}\quad\text{and}\quad\hat{x}_{[i]}=\sum_{h=i+1}^r x_h b^{h-1},$$
that is, $\overline{x}$ is equal to the number $x_1x_2\dots x_r$ written in base $b$, while $\hat{x}_{[i]}$ is the base-$b$ number with digits $x_r,x_{r-1},\dots,x_{i+1}$ followed by $i$ zeros. For $(x,i)\in Q\times [r]$, define the box $B_{x,i}=X_{x,i}\times Y_{x,i}\times Z_{x,i}$, where
\begin{align*}
X_{x,i}&=[2\overline{x},2\overline{x}+b^{r-i}],\\
Y_{x,i}&=[2\overline{x}-(2x_i+1)b^{r-i},2\overline{x}],\\
Z_{x,i}&=[2\hat{x}_{[i]}-2b^{i-1},2\hat{x}_{[i]}+(2x_i-1)b^{i-1}].
\end{align*}
Our goal is to prove that $B_{x,i}\cap B_{y,j}\neq \emptyset$ if and only if $(x,i)\sim (y,j)$. 

First, consider the case $x=y$. We show that $Z_{x,i}$ and $Z_{x,j}$ are disjoint, so $B_{x,i}$ and $B_{x,j}$ are also disjoint. Assume that $i<j$, then the right endpoint of $Z_{x,j}$ is $$2\hat{x}_{[j]}+(2x_j-1)b^{j-1}=2\hat{x}_{[j-1]}-b^{j-1},$$
while the left endpoint of $Z_{x,i}$ is 
$$2\hat{x}_{[i]}-2b^{i-1}>2\hat{x}_{[j-1]}-b^{j-1}.$$
Thus, $Z_{x,i}$ is completely to the right of $Z_{x,j}$.

In the rest of the argument, we assume that $x\neq y$. Let $p$ be the smallest index such that $x_p\neq y_p$, and let $q$ be the largest index such that $x_q\neq y_q$. Furthermore, without loss of generality, assume that $i\leq j$.

\begin{claim}\label{claim:boxx}
$$X_{x,i}\cap X_{y,j}\neq \emptyset \quad\Leftrightarrow\quad \begin{cases}x_p<y_p\text{ and }p>i\text{, or}\\ x_p>y_p\text{ and }p>j. \end{cases}$$
\end{claim}
\begin{proof}
We have $X_{x,i}\cap X_{y,j}\neq \emptyset$ if and only if
$$2\overline{x}\leq 2\overline{y}+b^{r-j}\quad\text{and}\quad 2\overline{y}\leq 2\overline{x}+b^{r-i}.$$ If $x_p<y_p$, then $\overline{x}<\overline{y}$, so the first inequality is satisfied, and the second inequality is satisfied if and only if $y_1=x_1,\dots,y_i=x_i$.
Similarly, if $x_p>y_p$, then $\overline{x}>\overline{y}$, so the second inequality is satisfied, and the first inequality is satisfied if $y_1=x_1,\dots,y_j=x_j$.
\end{proof}

\begin{claim}\label{claim:boxy}
$$Y_{x,i}\cap Y_{y,j}\neq \emptyset \quad\Leftrightarrow\quad \begin{cases}x_p<y_p\text{ and }p\geq j\text{, or}\\ x_p>y_p\text{ and }p\geq i. \end{cases}$$
\end{claim}

\begin{proof}
We have $Y_{x,i}\cap Y_{y,j}\neq \emptyset$ if and only if
$$2\overline{x}-(2x_i+1)b^{r-i}\leq 2\overline{y}\quad\text{and}\quad 2\overline{y}-(2y_j+1)b^{r-j}\leq 2\overline{x}.$$ If $x_p<y_p$, then $\overline{x}<\overline{y}$, so the first inequality is satisfied, and the second is satisfied if $x_1=y_1,\dots,x_{j-1}=y_{j-1}$. Similarly, if $x_p>y_p$, then $\overline{x}>\overline{y}$, so the second inequality is satisfied, and the first is satisfied if $x_1=y_1,\dots,x_{i-1}=y_{i-1}$.
\end{proof}

\begin{claim}\label{claim:boxz}
$$Z_{x,i}\cap Z_{y,j}\neq \emptyset \quad\Leftrightarrow\quad \begin{cases}i=j\text{ and }q\leq i\text{, or}\\i<j\text{ and }q\leq j\text{ and }x_j<y_j.
\end{cases}$$
\end{claim}
\begin{proof}
We have $Z_{x,i}\cap Z_{y,j}\neq \emptyset$ if and only if $$2\hat{x}_{[i]}-2b^{i-1}\leq 2\hat{y}_{[j]}+(2y_j-1)b^{j-1}\quad\text{and}\quad 2\hat{y}_{[j]}-2b^{j-1}\leq 2\hat{x}_{[i]}+(2x_i-1)b^{i-1}.$$
Consider three cases.

\medskip
\noindent
\textbf{Case 1.} $i=j$.

\noindent
If $\hat{x}_{[i]}\leq \hat{y}_{[i]}$, then the first inequality is satisfied. Also, the second inequality is then satisfied if and only if $x_r=y_r,\dots,x_{i+1}=y_{i+1}$. The same conclusion holds if $\hat{y}_{[i]}\leq \hat{x}_{[i]}$. Thus, $q\leq i$.

\medskip
\noindent
\textbf{Case 2.}  $i<j$ and $\hat{x}_{[i]}< \hat{y}_{[j]}$.

\noindent
The first inequality is satisfied. Moreover, $\hat{x}_{[i]}< \hat{y}_{[j]}$ implies that $q\geq j+1$. But then $\hat{y}_{[j]}-\hat{x}_{[i]}\geq b^{q-1}/2\geq b^{j}/2$, so the second inequality cannot be satisfied.

\medskip
\noindent
\textbf{Case 3.} $i<j$ and $\hat{y}_{[j]}\leq \hat{x}_{[i]}$.

\noindent
    The second inequality is satisfied. The first inequality is satisfied if and only if
    $x_r=y_r,\dots,x_{j+1}=y_{j+1}$ and $x_j<y_j$. Thus, $q\leq j$ and $x_j<y_j$.
\end{proof}
Assume that $B_{x,i}\cap B_{y,j}\neq \emptyset$. Then we have 
$$X_{x,i}\cap X_{y,j}\neq \emptyset\quad\text{and}\quad Y_{x,i}\cap Y_{y,j}\neq  \emptyset\quad\text{and}\quad Z_{x,i}\cap Z_{y,j}\neq \emptyset.$$
First, if $i=j$, then \Cref{claim:boxx} implies that $p>i$, and \Cref{claim:boxz} implies $q\leq i$. As $p\leq q$, this is a contradiction, so we must have $i<j$. Next, the combination of \Cref{claim:boxx} and \Cref{claim:boxy} implies that $p\geq j$. In contrast, \Cref{claim:boxz} implies $q\leq j$, so we must have $p=q=j$. Then  \Cref{claim:boxx} ensures that $x_p<y_p$. To summarize, we proved that $x$ and $y$ only differ in coordinate $p=j$, and $x_j<y_j$. This is exactly the condition which ensures that $(x,i)\sim (y,j)$ in $G$.

To finish the proof, it remains to check that if $(x,i)\sim (y,j)$ for some $i<j$, then $B_{x,i}\cap B_{y,j}\neq \emptyset$. This follows by checking that the right-hand sides of \Cref{claim:boxx}, \Cref{claim:boxy}, and \Cref{claim:boxz} are all satisfied. We have $p=j$, $x_p<y_p$ and $i<p$, so the first formula of \Cref{claim:boxx} is true. Moreover, $x_p<y_p$ and $p\geq j$, so the first formula of \Cref{claim:boxy} is true. Finally, we have $i<j$, $p=q\leq j$, and $x_j<y_j$, so the second formula of \Cref{claim:boxz} is satisfied. This finishes the proof.

Moreover, for every fingerprint $S$, we have $|S|\leq N/\Delta_0$, which implies 
$$|\mathcal{C}|\leq \binom{N}{\leq N/\Delta_0}\leq \exp\left(\frac{2Nrm\log k}{k}\right).$$
For $i\in [k]$ and $x\in V(H)^k=V(G)$, a subset of vertices in $G$ is called the \emph{$i$-slice of $x$} if it is the set of vertices that are either $x$, or only differ from $x$ in the $i$-th coordinate. A \emph{slice} refers to an $i$-slice for any $i\in [k]$. Each slice in $H^{\square k}$ induces a copy of $H$, and there are $km^{k-1}=kN/m$ slices. 

As the chromatic number of $H$ is at least $r$, no $r-1$ independent sets can cover $V(H)$. We show that this effect in $H^{\square k}$ is greatly amplified by proving that no $r-1$ containers can cover a large portion of $G$.

\begin{claim}
    Let $C_1,\dots,C_{r-1}\in \mathcal{C}$. Then
    $$|V(G)\setminus (C_1\cup \dots\cup C_{r-1})|\geq  \frac{N}{2m}.$$
\end{claim}
\begin{proof}
Let $D=C_1\cup\dots\cup C_{r-1}$. For $i\in [k]$, let $q_i$ be the number of $i$-slices contained completely in $D$. As $H$ is not $(r-1)$-colorable, every $i$-slice contained in $D$ sends at least one edge to one of $C_1,\dots,C_{r-1}$. Therefore,
$$q_1+\dots+q_k\leq \sum_{i=1}^{r-1} e(G[C_i])\leq (r-1)\cdot \frac{kN}{2rm}<\frac{kN}{2m}.$$
On the other hand, as the $i$-slices partition $V(G)$ into $N/m$ parts of size $m$, we have
$$|D|\leq q_i m+\left(\frac{N}{m}-q_i\right)(m-1)=q_i+\frac{N(m-1)}{m}.$$
Hence, we get
$$k|D|\leq q_1+\dots+q_k+\frac{kN(m-1)}{m}\leq \frac{kN}{2m}+\frac{kN(m-1)}{m}=\frac{kN(m-1/2)}{m},$$
which implies
$$|V(G)\setminus D|\geq \frac{N}{2m}.$$
\end{proof}

Let $p=k^{-3/4}$, and let $U$ be a random $p$-sample of $V(G)$. We prove that with high probability, $G[U]$ has chromatic number at least $r$ robustly, that is, the chromatic number remains at least $r$  even after removing some vertices.   We prove this by arguing that most of $U$ cannot be covered by $r-1$ containers. Say that $U$ is \emph{$q$-robust}, if $G[U']$ has chromatic number at least $r$ for every $U'\subset U$ with $|U'|\geq |U|-q$. 

\begin{claim}
    For $q=pN/(4m)$, we have
    $$\mathbb{P}(U\text{ is }q\text{-robust})\geq \frac{2}{3}.$$
\end{claim}

\begin{proof}
 Let $C_1,\dots,C_{r-1}\in \mathcal{C}$, $D=C_1\cup\dots \cup C_{r-1}$ and let $D^c=V(G)\setminus D$. Then $|D^c|\geq N/(2m)$ by the previous claim. Write $X=|U\cap D^c|$, then $\mathbb{E}X\geq pN/2m$, and by Chernoff's inequality
\begin{align*}
\mathbb{P}(X\leq q)&\leq \mathbb{P}\left(X\leq \frac{1}{2}\mathbb{E}X\right)\leq \exp\left(-\frac{\mathbb{E}X}{8}\right)< \exp\left(-\frac{pN}{16m}\right).
\end{align*}

If $U$ is not $q$-robust, then for some $U'\subset U$ with $|U'|\geq |U|-q$, $U'$ can be covered by $r-1$ independent sets, and thus $r-1$ containers. In other words, there is an $(r-1)$-tuple of containers $C_1,\dots,C_{r-1}$ such that $|U\setminus (C_1\cup\dots\cup C_{r-1})|\leq q$. Hence, by the union bound over all $(r-1)$-tuples of containers, we get
$$\mathbb{P}(U\text{ is not }q\text{-robust})\leq |\mathcal{C}|^{r-1}\exp\left(-\frac{pN}{16m}\right)<\exp\left(\frac{2Nr^2m\log k}{k}-\frac{k^{-3/4}N}{16m}\right).$$
If $k$ is sufficiently large, then the right-hand side is at most $1/3$, finishing the proof.
\end{proof}

In what follows, we show that random sampling destroys short cycles. 

\begin{claim}
    For $\ell\geq 3$, the number of cycles in $G$ of length $\ell$ is at most 
    $$O_{m,\ell}(Nk^{\lfloor \ell/2\rfloor}).$$
\end{claim}
\begin{proof}
A cycle $v_0,\dots,v_{\ell-1}$ in $G$ can be uniquely encoded by a sequence $v_0,s_1,e_1,s_2,e_2,\dots,s_{\ell},e_{\ell}$, where  $s_i$ is the coordinate in which $v_{i-1}$ and $v_i$ differ, and $e_i$ is the corresponding edge of $H$. Every coordinate that appears in $\{s_1,\dots,s_{\ell}\}$ must appear at least twice in order to ensure that $v_0,\dots,v_{\ell-1}$ is a cycle. Thus, we have at most $\binom{k}{\leq \lfloor\ell/2\rfloor}\leq 2k^{\lfloor\ell/2\rfloor}$ choices for the set $S=\{s_1,\dots,s_{\ell}\}$. From this, we get that the number of cycles of length $\ell$ is at most
$$2Nk^{\lfloor \ell/2\rfloor}\ell^{\ell}m^{2\ell},$$
as there are $N$ choices for $v_0$, at most $2k^{\lfloor \ell/2\rfloor}$ choices for the set $S$, at most $\ell^{\ell}$ choices to arrange $s_1,\dots,s_{\ell}$ once the set $S$ is fixed, and at most $m^{2\ell}$ choices for the edges $e_1,\dots,e_{\ell}$.
\end{proof}

Let $X_{\ell}$ denote the number of cycles of length $\ell$ in $G[U]$. Then 
$$\sum_{\ell=3}^{g-1} \mathbb{E}X_{\ell}\leq \sum_{\ell=3}^{g-1} O_{m,\ell}(p^{\ell}Nk^{\lfloor \ell/2\rfloor})=\sum_{\ell=3}^{g-1} O_{m,\ell}(Nk^{\lfloor \ell/2\rfloor-3\ell/4})\leq O_{m,g}(Nk^{-1}).$$
Let $q=pN/(4m)$. If $k$ is sufficiently large with respect to $m$ and $g$, the right-hand side is at most $q/3$. But then, by Markov's inequality, 
$$\mathbb{P}\left(X_3+\dots+X_{g-1}\geq q\right)\leq \frac{1}{3}.$$
Therefore, with positive probability,  there is a choice for $U$ such that $U$ is $q$-robust, and the number of cycles in $G[U]$ of length less than $g$ is at most $q$. But then removing a vertex of each cycle, we get a subset $V\subset U$ such that $\chi(G[V])\geq r$ and $G[V]$ has girth at least $g$.

<!-- lemma-10: body_md -->
\label{lemma:integer_sat}
Let $m$ be sufficiently large, and let $U\subset [m]^2$ be of size at least $m^{1.9}$. Then
$$\Delta(G_U)\geq \Omega\left(\frac{|U|\log m}{m}\right).$$
<!-- end lemma-10 -->
<!-- proof-19: body_md -->
If $(a,b)$ is an integer vector such that $a^2+b^2$ is a square (also known as \emph{Pythagorean vector}), then $[m]^2\cap \{x+t(a,b):t\in \mathbb{Z}\}$ forms a clique in $G_Q$ for every $x\in \mathbb{Z}^2$. If $1\leq a,b\leq 2m^{1/2}$ and $a,b$ are coprime, then the parallel lines $\{x+t(a,b):t\in \mathbb{Z}\}$ partition $Q$ into at most $m(a+b)$ parts. Indeed, each such line has an element in $([m]\times [b])\cup ([a]\times [m])$. Therefore, writing $L_1,\dots,L_k$ for the lines of direction $(a,b)$ intersecting $Q$, we have $k\leq m(a+b)$, and the number of edges of $G_U$ contained in one of $L_1,\dots,L_k$ is at least
$$\sum_{i=1}^k \binom{|L_i\cap U|}{2}=\frac{1}{2}\sum_{i=1}^k |L_i\cap U|^2-\frac{|U|}{2}\geq \frac{(\sum_{i=1}^k |L_i\cap U|)^2}{2k}-\frac{|U|}{2}=\frac{|U|^2}{2k}-\frac{|U|}{2}\geq \frac{|U|^2}{4m(a+b)}.$$
The first inequality is due to the Cauchy-Schwarz inequality, while in the last inequality, we used that $|U|\geq 16m^{3/2}>4m(a+b)$. Let $(u,v)$ be a pair of integers such that $1\leq u<v\leq m^{1/4}$, $u\not\equiv v\pmod{2}$ and $u,v$ are coprime. Then $(a,b)=(v^2-u^2,2uv)$ gives a unique Pythagorean vector with $a,b$ coprime and $1\leq a,b\leq 2m^{1/2}$. As lines of different directions yield different edges of $G_U$, we get
$$e(G_U)\geq \sum_{\substack{1\leq u<v\leq m^{1/4}\\ \text{gcd}(u,v)=1\\ u\not\equiv v\pmod{2}}}\frac{|U|^2}{16mv^2}.$$
By elementary number theoretic results, the sum on the right-hand side is at least $\Omega(|U|^2\log m/m)$. Indeed, for every sufficiently large $t$, there are $\Omega(t^2)$ coprime pairs $t\leq u<v\leq 2t$ with different parities, so the inequality follows by dyadic partitioning. Thus, by averaging, $G_U$ contains a vertex of degree at least $\Omega(|U|\log m/m)$.

<!-- lemma-11: body_md -->
\label{lemma:int_ind}
Let $\varepsilon\in (0,1/2)$, $p=\frac{\varepsilon}{(\log m)^{1/2}m}$, and assume that $m$ is sufficiently large. Let $U$ be a random $p$-sample of $[m]^2$. Then with probability at least $2/3$, we have $|U|\geq pm^2/2$ and
$$\frac{\alpha(G_U)}{|U|}\leq O_{\varepsilon}\left(\frac{\log\log m}{(\log m)^{1/2}}\right).$$
<!-- end lemma-11 -->
<!-- proof-20: body_md -->
By the multiplicative Chernoff bound, we have $\mathbb{P}(|U|\geq pm^2/2)\geq 3/4$ for $m$ sufficiently large. 

Let $c_0$ be the constant hidden by the $\Omega(.)$ notation in \Cref{lemma:integer_sat}. Let
$$L=\frac{m^2}{c_0\log m}\quad\text{and}\quad\Delta_0=m.$$
Run the Container algorithm (\Cref{alg:cont}) on the graph $G_Q$ with parameter $\Delta_0$, and let $$\mathcal{C}=\{S\cup T_S:S\text{ is a fingerprint}\}$$ be the collection of containers.

Next, we bound the size of $S$ and $T_S$. The simple bound $|S|\leq |Q|/(\Delta_0+1)$ is not enough for our purposes, but a more delicate counting argument yields a stronger bound. 

\begin{claim}
    We have
    $$|S|\leq \frac{8m\log \log m}{c_0\log m}\quad\text{and}\quad |T_S|\leq L.$$
\end{claim}

\begin{proof}
 The Container algorithm starts with $T=V(G)$, and in each step, removes at least one vertex of $T$. In case $|T|\geq L$, we have $\Delta(G[T])\geq \frac{c_0L\log m}{m}=m=\Delta_0$ by \Cref{lemma:integer_sat}, which shows that  $|T_S|\leq L$. Moreover, \Cref{lemma:integer_sat} ensures that   $$\Delta(G[T])\geq \frac{c_0|T|\log m}{m}.$$ This means that every time the algorithm adds a vertex to the set $S$, it removes at least $\frac{c_0|T|\log m}{m}$ vertices from $T$. Hence, while the size of $T$ drops from $t$ to $t/2$ for some $t\geq 2L$, at most $\frac{2m}{c_0\log m}$ vertices have been added to $S$. This ensures that as the size of $T$ drops from $m^2$ to $L=m^2/(c_0\log m)$, at most $\frac{4m\log \log m}{c_0\log m}$  vertices have been added to $S$. On the other hand, if $|T|\leq L$, the algorithm added at most $L/\Delta_0=O(m/\log m)$ vertices to $S$. Thus, we have
 $$|S|\leq \frac{4m\log \log m}{c_0\log m}+O\left(\frac{m}{\log m}\right)\leq \frac{8m\log \log m}{c_0\log m}.$$
\end{proof}

Let $\beta=\lfloor \frac{8m\log\log m}{c_0\log m}\rfloor$ and $\alpha=4\beta$. If $\alpha(G_U)\geq \alpha$, then there exists an independent set $I\subset U$ of size at least $\alpha$, and  a fingerprint $S$ such that $S\subset I\subset T_S\cup S$. As $|S|\leq \beta$, the latter implies that 
$|T_S\cap U|\geq 3\beta$. The probability of this event can be bounded by a simple union bound as
$$\mathbb{P}(|T_S\cap U|\geq 3\beta)\leq \binom{|T_S|}{3\beta}p^{3\beta}\leq \left(\frac{
eLp}{3\beta}\right)^{3\beta}\leq \exp\left(3\beta \log \frac{Lp}{\beta} \right).$$
Hence, applying the union bound over all possible fingerprints, and noting that the events $(S\subset U)$ and $(|T_S\cap U|\geq 3\beta)$ are independent by the disjointness of $S$ and $T_S$, we get
\begin{align*}
\mathbb{P}(\alpha(G_U)\geq \alpha)&\leq \mathbb{P}(\exists S\text{ fingerprint}:\ S\subset U\text{ and }|T_S\cap U|\geq 3\beta)\\
&\leq \sum_{S\text{ fingerprint}} p^{|S|} \mathbb{P}(|T_S\cap U|\geq 3\beta)\\
&\leq \sum_{k=0}^{\beta}\binom{m^2}{k}p^k \exp\left(3\beta \log \frac{Lp}{\beta} \right)\\
&\leq \sum_{k=0}^{\beta}\exp\left(k+k\log \frac{m^2p}{k}+3\beta \log \frac{Lp}{\beta}\right).
\end{align*}
The function $x\mapsto x+x\log \frac{m^2p}{x}$ is monotone increasing for $x<\beta<m^2p/e$, so for $k\leq \beta$, the right-hand side can be further bounded as
\begin{equation}\label{eq:int1}
    (\beta+1) \exp\left(\beta+\beta\log \frac{m^2p}{\beta}+3\beta \log \frac{Lp}{\beta}\right)=(\beta+1)\exp\left(\beta+\beta\log \left[\frac{m^2p}{\beta}\cdot \left(\frac{Lp}{\beta}\right)^3\right]\right).
\end{equation}
Recalling that  $L=\Theta(m^2/\log m)$, $p=\varepsilon/(m(\log m)^{1/2})$ and $\beta=\Theta(m\log\log m/\log m)$, we have inside the logarithm
$$\frac{m^2p}{\beta}\cdot \left(\frac{Lp}{\beta}\right)^3=\Theta_{\varepsilon}\left(\frac{1}{(\log \log m)^4\log m}\right).$$  Therefore, the right-hand side of (\ref{eq:int1}) is tending to 0 as $m$ tends to infinity. Thus, with probability at least $2/3$, we have $\alpha(G_U)\leq \alpha$ and $|U|\geq pm^2/2$, which implies that 
$$\frac{\alpha(G_U)}{|U|}\leq O_{\varepsilon}\left(\frac{\log \log m}{(\log m)^{1/2}}\right).$$

<!-- proof-25: body_md -->
Let us briefly outline the proof. For integers $t,u,q$, consider the map $\phi=\phi_{t,u,q}:\mathbb{R}^2\rightarrow\mathbb{R}^2$ defined as
$$\phi((x,y))=(t+qx,u+q^3y).$$
The important property of this map is that it maps edges to edges. We show that $G_V$ has many edges by considering the intersections of $V$ with  the images $\phi_{t,u,q}([N]\times [N^3])$ for an appropriately chosen $N$ and tuples of integers $(t,u,q)$.  $$|J_q|\leq (m+sN)(m^3+s^3N^3)<4m^4.$$

$$\Gamma_{t,u,q}=\phi_{t,u,q}([N]\times [N^3])=\{(t+qx,u+q^3y):(x,y)\in [N]\times [N^3]\}.$$
Note that every element $(a,b)\in [m]\times [m^3]$ is contained in $\Gamma_{t,u,q}$ for exactly $N^4$ pairs $(t,u)\in J_q$. Indeed, for every $(x,y)\in [N]\times [N^3]$ there is a unique choice of $(t,u)\in J_q$ such that $(a,b)=(t+qx,u+q^3y)$. Thus,
$$\sum_{(t,u)\in J_q} |V\cap \Gamma_{t,u,q}|=|V|N^4\geq Cf(m)m^4N^4.$$
Say that a pair $(t,u)$ is rich if $|V\cap \Gamma_{t,u,q}|\geq Cf(m)N^4/8$. As $|J_q|\leq 4m^4$, we have
$$\sum_{\substack{(t,u)\in J_q\\ (t,u)\text{ is not rich}}} |V\cap \Gamma_{t,u,q}|\leq Cf(m)m^4N^4/2.$$
Thus, using that $|V\cap \Gamma_{t,u,q}|\leq |\Gamma_{t,u,q}|=N^4$, there are at least $Cf(m)m^4/2$ rich pairs $(t,u)$. In case $C$ is sufficiently large with respect to $\varepsilon$, we have $Cf(m)N^4/8>f(N)N^4$, which then ensures that for every rich pair $(t,u)$, $\Gamma_{t,u,q}\cap V$ induces at least one edge in $G_V$. 

Next, we bound the number of sets $\Gamma_{t,u,q}\cap V$ with $s/2\leq q\leq s$ and $(t,u)\in J_q$ that contain a given edge $(a,b)\sim (A,B)$ in $G_V$. Assume that
$$(t+qx,u+q^3y)=(a,b)\quad\text{and}\quad (t+qX,u+q^3Y)=(A,B).$$
Then $A-a=q(X-x)$, so $q$ divides $A-a$. Thus, as $0\neq |A-a|\leq m$, there are $m^{o(1)}$ choices for $q$. If $q$ is given, we have  $t\equiv a\pmod{q}$, thus there are at most $2m/q=O(N)$ choices for $t$. Similarly, as
$u\equiv b\pmod{q^3}$,  there are at most $2m^3/q^3=O(m^3/s^3)=O(N^3)$ choices for $u$. Thus, in total, any edge may appear in $\Gamma_{t,u,q}\cap V$ for at most $m^{o(1)}N^{4}$ tuples $(t,u,q)$. 

Thus, we can lower bound the number of edges of $G_V$ by $f(m)m^{5-o(1)}N^{-5}$. Indeed, there are at least $(s/2)\cdot (Cf(m)m^4/2)=\Omega(Cf(m)m^5 N^{-1})$ tuples $(t,u,q)$ such that $V\cap \Gamma_{t,u,q}$ induces an edge, and each such edge is counted for at most $m^{o(1)}N^{4}$ different tuples. For $m$ sufficiently large, we have $f(m)m^{5-o(1)}N^{-5}>m^{5-\delta}$, which implies $\Delta(G_V)\geq m^{1-\delta}$.

Let $c_0$ be the constant hidden by the $\Omega(.)$ notation in \Cref{lemma:circle_saturation}. Let $Q=[m]^3$, $G=G_Q$, 
$$\mu=\frac{1000(\log \log m)^2}{c_0\log m},\quad L=\mu m^3\quad\text{and}\quad \Delta_0=\frac{c_0 L\log m}{m^2}=c_0\mu m\log m.$$ Let $<$ be an arbitrary ordering on the vertices of $G$. \begin{claim}
    We have
    $$|S|\leq \frac{8m^2}{c_0\log m}\log \frac{1}{\mu}=:\beta\quad\text{and}\quad |T_S|\leq L.$$
\end{claim}

\begin{proof}
 The Container algorithm starts with $T=V(G)$, and in each step, removes at least one vertex of $T$. In case $|T|\geq L$, we have $\Delta(G[T])\geq \Delta_0$, which ensures $|T_S|\leq L$. However, \Cref{lemma:circle_saturation} ensures the even stronger bound   $$\Delta(G[T])\geq \frac{c_0|T|\log m}{m^2}.$$ This means that every time the algorithm adds a vertex to the set $S$, it removes at least $\frac{c_0|T|\log m}{m^2}$ vertices from $T$. On the other hand, if $|T|\leq L$, the algorithm added at most $|T|/\Delta_0=O(m^2/\log m)$ vertices to $S$. Thus, we have
 $$|S|\leq \frac{4m^2}{c_0\log m}\log \frac{1}{\mu}+O\left(\frac{m^2}{\log m}\right)\leq \frac{8m^2}{c_0\log m}\log \frac{1}{\mu}.$$
\end{proof}

Let $\alpha=2\mu m^2$. If $\alpha(G_U)\geq \alpha$, then there exists an independent set $I\subset U$ of size at least $\alpha$, and  a fingerprint $S$ such that $S\subset I\subset T_S\cup S$. As $|S|< \mu m^2=\alpha/2$, the latter implies that 
$$|T_S\cap U|\geq \mu m^2\geq 2p L\geq 2\mathbb{E}|T_S\cap U|.$$ The probability of this event can be bounded by the Chernoff bound as
$$\mathbb{P}(|T_S\cap U|\geq \mu m^2)\leq \exp\left(-\frac{\mu m^2}{6}\right).$$
Hence, applying the union bound over all possible fingerprints, and noting that the events $(S\subset U)$ and $(|T_S\cap U|\geq \mu m^2)$ are independent by the disjointness of $S$ and $T_S$, we get
\begin{align*}
\mathbb{P}(\alpha(G_U)\geq \alpha)&\leq \mathbb{P}(\exists S\text{ fingerprint}:\ S\subset U\text{ and }|T_S\cap U|\geq \mu m^2)\\
&\leq \sum_{S\text{ fingerprint}} p^{|S|} \mathbb{P}(|T_S\cap U|\geq \mu m^2)\\
&\leq \sum_{k=0}^{\lfloor \beta\rfloor}\binom{m^3}{k}p^k \exp\left(-\frac{\mu m^2}{6}\right)\\
&\leq \sum_{k=0}^{\lfloor \beta\rfloor}\exp\left(k+k\log \frac{m^3p}{k}-\frac{\mu m^2}{6}\right).
\end{align*}
The function $x+x\log \frac{m^3p}{x}$ is monotone increasing for $x<m^3p/e=\Theta(m^2)$, so for $k\leq \beta$, the right-hand side can be further bounded as
\begin{align*}
(\beta+1) \exp\left(2\beta\log \frac{m^3p}{\beta}-\frac{\mu m^2}{6}\right)&=(\beta+1) \exp\left(\frac{16m^2}{c_0\log m}\cdot \log\frac{1}{\mu}\cdot \log \frac{c_0\varepsilon \log m}{8\log (1/\mu)}-\frac{\mu m^2}{6}\right)\\
&\leq (\beta+1)\exp\left(\frac{16m^2(\log \log m)^2}{c_0\log m}-\frac{\mu m^2}{6}\right)
\end{align*}
The choice of $\mu$ ensures that the right-hand side is at most $0.1$ for $m$ sufficiently large. Thus, with probability at least $2/3$, we have $\alpha(G_U)\leq \alpha$ and $|U|\geq pm^3/2$, which implies that 
$$\frac{\alpha(G_U)}{|U|}\leq O_{\varepsilon}\left(\frac{(\log \log m)^2}{\log m}\right).$$

<!-- lemma-25: body_md -->
\label{lemma:common_neighbor}
Let $u,v\in [m]^3$ be distinct. 
\begin{enumerate}
    \item If $u\not\sim v$, then there are at most $m^{o(1)}$ common neighbors of $u$ and $v$.
    \item If $u\sim v$, then all common neighbors of $u$ and $v$ are on the line spanned by $u$ and $v$.
\end{enumerate}
<!-- end lemma-25 -->
<!-- proof-32: body_md -->
Let $w\in [m]^3$ be a common neighbor of $u$ and $v$. Write
$$v-u=(a,b,c),\quad w-u=(x,y,z)\quad\text{ and }\quad D=ac-b^2.$$
Note that $D=0$ is equivalent to $u\sim v$. Since $u\sim w$, we have $$xz=y^2.$$
Similarly, $v\sim w$ implies $(x-a)(z-c)=(y-b)^2$, which can be rewritten as
\begin{equation}\label{eq:7.2}
cx+az-2by=D.
\end{equation}
Multiply both sides by $x$ and substitute $xz=y^2$, then we get
$$cx^2+ay^2-2bxy=xD.$$
Assume that $a\neq 0$. Then multiplying by $a$ and completing the square, we get
$$(ac-b^2)x^2+(ay-bx)^2=axD,$$
which we can further write as $Dx^2+(ay-bx)^2=axD$. Moreover, multiplying by 4 and completing the other square, we get
\begin{equation}\label{equ:7.1}
    D(2x-a)^2+4(ay-bx)^2=Da^2.
\end{equation}
If $D\neq 0$, this is an equation of the form $AX^2+BY^2=C$, for which \Cref{lemma:quadratic} applies. Therefore, as the coefficients in this equation are bounded by $O(m^4)$, there are at most $m^{o(1)}$ solutions $(X,Y)$, which give $m^{o(1)}$ solutions $x,y$ of (\ref{equ:7.1}). Thus, $u$ and $v$ have $m^{o(1)}$ common neighbors.

If $D=0$, then (\ref{equ:7.1}) implies $ay=bx$, and (\ref{eq:7.2}) yields $cx=az$. From this,
$w-u=\frac{x}{a}(v-u),$ so the common neighbors of $u$ and $v$  lie on the line spanned by $u$ and $v$.

If $a=0$ and $c\neq 0$, then the same argument works by symmetry. Finally, if $a=c=0$, then $b\neq 0$ (as $u\neq v$). Then $D=-b^2\neq 0$ and (\ref{eq:7.2}) implies $y=b/2$.  But then $xz=y^2$ becomes $xz=b^2/4.$
The number of solutions $(x,z)$ of this equation is the number of divisors of $b^2/4$, which is also $m^{o(1)}$.

Assume that $u_i$ is in the blow-up interval of the vertex $v_i\in V(K)$. As the neighborhoods of $u_1$ and $u_2$ are different, we have $v_1<v_2$. Similarly, $v_3<v_4$. Also, we have $v_1v_3,v_1v_4,v_2v_4\in E(K)$. If $v_2=v_3$, then $K$ contains a triangle, contradiction. If $v_2<v_3$, then $u_2u_3\not\in E(K^{(b)})$ implies that $v_2v_3\not\in E(K)$, so $v_1,v_2,v_3,v_4$ contradict that $K$ is cross-path-free.

$G$ is triangle- and cross-path-free.

<!-- proof-35: body_md -->
First, we show that $G$ is triangle-free. Assume to the contrary that $u_1<u_2<u_3$ are vertices of a triangle. Then $u_1,u_2,u_3$ cannot be contained in the same interval $J_w$ as $K^{(b)}$ is triangle-free. Also, if two of them are contained in the same interval, but not the third, then $\tau(u_1)=\tau(u_2)=\tau(u_3)$. However, two vertices in the same interval and assigned the same number are not connected by an edge. Finally, it cannot be that all three vertices are contained in different intervals as $L$ is triangle-free.

Next, we prove that $G$ is cross-path-free. Assume to the contrary that $u_1<u_2<u_3<u_4$ are vertices of $G$ such that $u_1u_3,u_1u_4,u_2u_4$ are edges, but $u_2u_3$ is not an edge. Assume that $u_i$ is contained in the interval $J_{w_i}$ for $i\in [4]$, then $w_1\leq w_2\leq w_3\leq w_4$. We analyze 8 different cases depending on the equalities between $w_1,w_2,w_3,w_4$.
\begin{description}
    \item[$w_1=w_2=w_3=w_4$:] we get a contradiction as $K^{(b)}$ is cross-path-free.
    \item[$w_1<w_2=w_3=w_4$:] $u_1u_3$ and $u_1u_4$ are edges across intervals, which forces $\tau(u_1)=\tau(u_3)=\tau(u_4)$. However, as $u_2<u_3<u_4$ are contained in $J_{w_2}$, this forces $u_3$ and $u_4$ to be in different copies  of $K^{(b)}$, and thus $u_2$ and $u_4$ are also in different copies. But then there is no edge between $u_2$ and $u_4$, contradiction.
    \item[$w_1=w_2<w_3=w_4$:]  We have $\tau(u_1)=\tau(u_2)=\tau(u_3)=\tau(u_4)$, and $w_1w_3\in E(L)$, which implies that $u_2u_3$ is also an edge of $G$, contradiction. 
    \item[$w_1=w_2=w_3<w_4$:] analogous to the case $w_1<w_2=w_3=w_4$.
    \item[$w_1<w_2<w_3=w_4$:] as $u_1u_4$, $u_2u_4$, $u_1u_3$ are edges across intervals, we must have $\tau(u_1)=\tau(u_2)=\tau(u_3)=\tau(u_4)$. Moreover $w_1w_3\in E(L)$ and $w_2w_3=w_2w_4\in E(L)$, which forces $u_2u_3$ to be an edge, contradiction.
    \item[$w_1<w_2=w_3<w_4$:] we get a contradiction as $L$ is triangle-free.
    \item[$w_1=w_2<w_3<w_4$:] analogous to the case $w_1<w_2<w_3=w_4$.
    \item[$w_1<w_2<w_3<w_4$:] we get a contradiction as $L$ is cross-path-free.
\end{description}

Assume that 
$$a\geq \rho_K\rho_L^2\quad\text{and}\quad b\geq 2\rho_K^2\rho_L^2.$$

<!-- proof-36: body_md -->
Let $\mathcal{I}=(I_{w,p})_{w\in V(L),p\in [a]}$ be a tuple, where $I_{w,p}$ is an independent set in $K_{w,p}$ such that if some vertex is contained in $I_{w,p}$, then all its blow-up twins are also contained in $I_{w,p}$. Define  $S(\mathcal{I})=\bigcup_{w\in V(L), p\in [a]} I_{w,p}$. Note that there are at most $2^{k}$ choices for the set $I_{w,p}$ for every fixed $(w,p)$, so the number of tuples $\mathcal{I}$ can be upper bounded as 
$$2^{ak\ell}=2^{n/b}.$$ Clearly, every independent set of $G$ is contained in some $S(\mathcal{I})$.

Next, we reveal our random assignment $\tau$, and thus the random edges across the intervals $(J_w)_{w\in V(L)}$. Fix some tuple $\mathcal{I}$ and set $S=S(\mathcal{I})$. For $w\in V(L)$ and $i\in [kb]$, let $N_{w,i}$ be the number of vertices of $S(\mathcal{I})\cap J_w$ labeled $i$. Define
$$Z_i=\max_{\substack{I\subset V(L)\\I\text{ is independent}}} \sum_{w\in I} N_{w,i}$$
and let $Z=Z_1+\dots+Z_{kb}$. Then $Z$ is an upper bound on the maximum size of an independent set of $G[S]$. Next, we bound the expectation of $Z$. We have
\begin{equation}\label{equ:order1}
    \mathbb{E}N_{w,i}\leq \frac{\alpha(K)}{k}a=:t.
\end{equation}
Indeed,  $|S\cap V(K_{w,p})|\leq b\alpha(K)$, and each vertex of $S\cap V(K_{w,p})$ receives the label $i$ with probability $1/(bk)$.  Moreover, as $N_{w,i}$ is a sum of independent indicator random variables, we also have
$$\operatorname{Var}(N_{w,i})\leq \mathbb{E}N_{w,i}\leq  t.$$
For every independent set $I$ of $L$, 
$$\sum_{w\in I}N_{w,i}\leq \sum_{w\in I}(t+|N_{w,i}-\mathbb{E}N_{w,i}|)\leq t\alpha(L) + \sum_{w\in I} |N_{w,i}-\mathbb{E}N_{w,i}|\leq t\alpha(L) + \sum_{w\in V(L)} |N_{w,i}-\mathbb{E}N_{w,i}|.$$
Thus, we also have
\begin{equation}\label{equ:order2}
Z_i\leq t\alpha(L)+\sum_{w\in V(L)} |N_{w,i}-\mathbb{E}N_{w,i}|.
\end{equation}
Here,
$$\mathbb{E}|N_{w,i}-\mathbb{E}N_{w,i}|\leq \sqrt{\operatorname{Var}(N_{w,i})}\le \sqrt{t}.$$
Taking the expectation of both sides of (\ref{equ:order2}), we get
$$\mathbb{E}Z_i\leq \alpha(L)t+\ell\sqrt{t}.$$
By our lower bound $a\geq (k/\alpha(K))\cdot(\ell/\alpha(L))^2$, the right-hand side is at most $2\alpha(L)t$. Thus, we conclude
$$\mathbb{E}Z\leq 2kbt\alpha(L)=2ab\cdot\alpha(K)\alpha(L)=\frac{2n}{\rho_K\rho_L}.$$

Next, we argue that $Z$ is concentrated around its mean. For $j=0,\dots,n$, define the martingale
$$M_j=\mathbb{E}(Z|\tau\text{ is revealed on the first }j\text{ vertices of }G).$$
The main observation is that $|M_j-M_{j+1}|\leq 2$, because for every $(w,p)$, the assignment $\tau|_{V(K_{w,p})}$ is an independent random permutation of $[kb]$, and swapping any two elements of the permutation changes the value of $Z$ by at most $2$. Hence, we can apply the Azuma-Hoeffding inequality to get that
$$\mathbb{P}\left(Z-\mathbb{E}Z\geq \frac{2n}{\rho_K\rho_L}\right)\leq \exp\left(-\frac{n}{2\rho_K^2\rho_L^2}\right).$$
Thus, the probability that $S$ contains an independent set of size at least $4n/(\rho_K\rho_L)$ is at most $\exp(-n/(2\rho_K^2\rho_L^2))$. However, recall that the number of choices for $S$ is at most $2^{n/b}$, thus, by the union bound,
$$\mathbb{P}\left(\alpha(G)\geq \frac{4n}{\rho_K\rho_L}\right)\leq 2^{n/b}\exp\left(-\frac{n}{2\rho_K^2\rho_L^2}\right)$$
Our lower bound $b\geq 2\rho_K^2\rho_L^2$ ensures that the right-hand side is strictly less than 1, finishing the proof.
