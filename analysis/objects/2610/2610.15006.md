<!-- lemma-1: body_md -->
\label{lem:disint}
For every $\mu\in\mathcal{P}_n$ and partition $\mathcal{I}$,
there exists a $\Phi_k$-a.e.\ unique probability kernel
$t\mapsto\mu(\cdot\mid t)$ supported on $\Pi^{-1}(t)$ such that
\[
\int_{\mathbb{I}^n}f(u)\,d\mu(u)
=
\int_{\mathbb{I}^k}
\left(
\int_{\Pi^{-1}(t)} f(u)\,d\mu(u\mid t)
\right)
d\Phi_k(t).
\]
<!-- end lemma-1 -->
<!-- defn-1: body_md -->
The family $\mathcal{P}_\bullet=(\mathcal{P}_n)_{n\ge1}$ with symmetric group action
$\sigma\cdot\mu=(\sigma^{-1})_*\mu$ is a symmetric cooperad via the
cocomposition map
\[
\boxed{
\Delta_{\mathcal{I}}:\mathcal{P}_n
\longrightarrow
\Bigl(\bigotimes_{j=1}^k\mathcal{P}_{m_j}\Bigr)\otimes\mathcal{P}_k,
\qquad
\Delta_{\mathcal{I}}(\mu)
=
(\mu_1,\dots,\mu_k;\,\Phi_k)
}
\]
where $\mu_j=(\pi_{I_j})_*\mu$ and
$\Phi_k=(\Psi_1\times\cdots\times\Psi_k)_*\mu$.
<!-- end defn-1 -->
<!-- defn-2: body_md -->
The same family $\mathcal{P}_\bullet$ carries a symmetric operad structure
with composition
\[
\boxed{
\gamma\bigl(\Phi_k;\mu_1,\dots,\mu_k\bigr)
:=
\int_{\mathbb{I}^k}
\left(
\bigotimes_{j=1}^k\mu_j(\cdot\mid t_j)
\right)
d\Phi_k(t_1,\dots,t_k)
}
\tag{OP}
\]
where $\mu_j(\cdot\mid t_j)$ is the $\Phi_k$-a.e.\ unique regular
conditional measure of $\mu_j\in\mathcal{P}_{m_j}$ along its own CDF summary
$\Psi_j$. Conditional kernels are not stored in the objects;
they are invoked inside the definition of $\gamma$.
<!-- end defn-2 -->
<!-- prop-1: body_md -->
\label{prop:asym}
Let $\mathcal{I}$ be a partition of $\{1,\dots,n\}$ into $k$ blocks.
\begin{enumerate}
\item[(i)] \textbf{Combination then disintegration is the identity
on admissible inputs:}
\[
\Delta_{\mathcal{I}}\bigl(\gamma(\Phi_k;\mu_1,\dots,\mu_k)\bigr)
=
(\mu_1,\dots,\mu_k;\,\Phi_k)
\]
holds for every admissible input
$(\Phi_k;\mu_1,\dots,\mu_k)$.

\item[(ii)] \textbf{Disintegration then combination is not the identity
on $\mathcal{P}_n$ in general:}
\[
\gamma\bigl(\Delta_{\mathcal{I}}(\mu)\bigr)
=
\int_{\mathbb{I}^k}
\left(\bigotimes_{j=1}^k\mu_j(\cdot\mid t_j)\right)
d\Phi_k(t)
\neq\mu
\]
whenever $\mu\notin\mathrm{im}(\gamma)$.
\end{enumerate}
<!-- end prop-1 -->
<!-- proof-1: body_md -->
(i) Let $\hat\mu=\gamma(\Phi_k;\mu_1,\dots,\mu_k)$.
By the definition of $\gamma$, for $\Phi_k$-a.e.\ $t$ we have
$\hat\mu(\cdot\mid t)=\bigotimes_{j=1}^k\mu_j(\cdot\mid t_j)$
---this is a definitional consequence of how $\gamma$ is constructed,
not a claim about arbitrary $\mu$.
Applying $\Delta_{\mathcal{I}}$ to $\hat\mu$ recovers the same
$(\mu_1,\dots,\mu_k;\,\Phi_k)$ because the block margins and the
CDF-summary pushforward are unchanged.

(ii) For arbitrary $\mu\in\mathcal{P}_n$, Lemma~\ref{lem:disint} gives the
true conditional distribution $\mu(\cdot\mid t)$.
In contrast, $\gamma(\Delta_{\mathcal{I}}(\mu))$ uses
$\bigotimes_j\mu_j(\cdot\mid t_j)$---the conditionally independent
product---as its integrand.
These two integrands differ whenever $\mu$ has residual cross-block
dependence beyond what $\Phi_k$ captures:
\[
\mu(\cdot\mid t)\neq\bigotimes_{j=1}^k\mu_j(\cdot\mid t_j),
\qquad\Phi_k\text{-a.e. on a positive-measure set.}
\]
Thus the integral recovers $\mu$ iff
$\mu\in\mathrm{im}(\gamma)$, i.e., iff $\mu$ was generated by
combination from some $(\Phi_k;\mu_1,\dots,\mu_k)$.
<!-- end proof-1 -->
<!-- theorem-1: body_md -->
\label{thm:kl}
For any $\mu\in\mathcal{P}_n$ and partition $\mathcal{I}$,
\[
D_{\mathrm{KL}}\bigl(\mu\,\|\,\mu_1\otimes\cdots\otimes\mu_k\bigr)
=
D_{\mathrm{KL}}\bigl(\Phi_k\,\|\,\Phi_k^{\mathrm{ind}}\bigr)
+
\sum_{j=1}^k
\mathbb{E}_{t\sim\Phi_k}\bigl[
D_{\mathrm{KL}}\bigl(\mu_j(\cdot\mid t_j)\,\|\,\mu_j\bigr)
\bigr].
\]
The residual loss of the disintegration--combination round trip is
\[
\mathbb{E}_{t\sim\Phi_k}\Bigl[
D_{\mathrm{KL}}\Bigl(
\mu(\cdot\mid t)\,\Big\|\,
\bigotimes_{j=1}^k\mu_j(\cdot\mid t_j)
\Bigr)
\Bigr]\ge 0,
\]
with equality iff $\mu\in\mathrm{im}(\gamma)$.
<!-- end theorem-1 -->
