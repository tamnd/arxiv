<!-- defi-1: body_md -->
A  \emph{tree decomposition} of a graph $G$ is a pair $\mathcal{T}=(T,\{X_t\}_{t\in V(T)})$, where $T$ is
	a tree whose every node $t$ is assigned a vertex subset
	$X_t\subseteq V(G)$, called a \emph{bag}, such that the following conditions hold:
	\begin{enumerate}
		\item[(T1)] $\displaystyle\bigcup_{t\in V(T)}X_t=V(G)$. In other words, every vertex of $G$ is in at least one bag.
		
		\item[(T2)] For every edge $uv\in E(G)$, there exists a node $t$ of $T$ such that bag $X_t$ contains both $u$ and $v$.
		
		\item[(T3)] For every $u\in V(G)$, the set $T_u=\{t\in V(T):u\in X_t\}$, i.e., the set of nodes whose corresponding bags contain $u$, induces a connected subtree of $T$.
	\end{enumerate}
<!-- end defi-1 -->
<!-- lemma-1: body_md -->
If a graph $G$ admits a tree decomposition of width at most $k$, then it also admits a nice tree decomposition of width at most $k$. Moreover, given a tree decomposition
	 $\mathcal{T}=(T,\{X_t\}_{t\in V(T)})$ of $G$ of width at most $k$, one can compute a nice tree decomposition of $G$ of width at most $k$ that has at most $O(k|V(G)|)$ nodes in time $O\left(k^2\cdot\max(|V(T)|,|V(G)|)\right)$.
<!-- end lemma-1 -->
<!-- defi-2: body_md -->
The \emph{clique-width} of a graph $G$, denoted by $\texttt{cw}(G)$, is the minimum number of labels needed to construct $G$ using the following four operations:
\begin{enumerate}
\item \textbf{Introducing a vertex.}
$\Phi=v(i)$ creates a new vertex $v$ with label $i$. Thus, $G_{\Phi}$ consists of a single vertex $v$ with label $i$.

\item \textbf{Disjoint union.}
$\Phi=\Phi'\oplus\Phi''$ constructs the disjoint union of the labeled graphs $G_{\Phi'}$ and $G_{\Phi''}$.

\item \textbf{Introducing edges.}
$\Phi=\eta_{i,j}(\Phi')$ adds an edge between every vertex with label $i$ and every vertex with label $j$, where $i\neq j$, in $G_{\Phi'}$.

\item \textbf{Renaming labels.}
$\Phi=\rho_{i\rightarrow j}(\Phi')$ changes the label of every vertex with label $i$ to label $j$ in $G_{\Phi'}$.
\end{enumerate} 

An expression constructed using these four operations and using at most $w$ labels is called a \emph{$w$-expression}. In other words, the clique-width of a graph $G$ is the minimum integer $w$ for which there exists a $w$-expression that constructs $G$.

It is easy to verify that \textsc{strong odd $k$-colorability} is in NP. Now we show that \textsc{strong odd $k$-colorability} is NP-hard by reducing an instance $(G, k)$ of $k$-\textsc{colorability} to an instance $(H,k)$ of \textsc{strong odd $k$-colorability} such that $G$ is $k$-colorable if and only if $H$ is strong odd $k$-colorable. 
	
	\begin{construction}\label{construction_PEBG}
		Let $G$ be a graph with $V(G) = \{v_1, v_2, \ldots, v_n\}$ and $E(G) = \{e_1, e_2, \ldots, e_m\}$. We construct a graph $H$ from $G$ in polynomial time using the following steps.
		
		\vspace{-0.2cm}
		
		\begin{enumerate}
			\item Take a copy vertex $v_i'$ of each vertex $v_i\in V(G)$. Let $V'=\{v_1', v_2',\ldots,v_n'\}$.
			
			\item Take a vertex $v_{e_i}$ corresponding to each edge $e_i\in E(G)$ and make $v_{e_i}$ adjacent to the copy of both the endpoints of $e_i$ in $V'$. Let $V_{E}^1 = \{v_{e_1}, v_{e_2}, \ldots, v_{e_m}\}$. Clearly, each $v_{e_i}$ is a vertex of degree $2$.
			
			\item For each vertex $v_{e_i}\in V_{E}^1$, introduce a new vertex $v_{e_i}'$. Let $V_{E}^2 = \{v_{e_1}', v_{e_2}', \ldots, v_{e_m}'\}$. We add an edge between $v_{e_i}'\in V_{E}^2$ and $v_j\in V'$ whenever there is an edge between $v_{e_i}$ and $v_j$. Therefore, each $v_{e_i}'\in V_{E}^2$ is a vertex of degree $2$.
			
			\item Attach $k-1$ number of pendant vertices to each vertex $v_i\in V'$, denote them by $\{v_1^i, v^i_2, \ldots v^i_{k-1}\}$. Let $P=\{v^1_1, v^1_2, \ldots v^1_{k-1}, v^2_1, v^2_2, \ldots v^2_{k-1}, \ldots, v^n_1, v^n_2, \ldots v^n_{k-1}\}$.
		\end{enumerate}
		
	\end{construction}

	Note that $G$ is a bipartite graph. Let $\sigma = (v_1v^1_1, v_2v^2_1, \ldots, v_nv^n_1)$ and $S_i = \{v_1, v^1_1, v_2, v^2_1, \\\ldots, v_i, v^i_1\}$. Every edge $v_{i+1} v^{i+1}_1$ is bisimplicial in $H[V(H) \setminus S_i]$ for each $i\in [n]$. Moreover, there are no edges in the graph $H[V(H) \setminus S_n]$. Therefore, $\sigma$ is a perfect edge elimination scheme of the graph $H$. Thus, $H$ is a perfect elimination bipartite graph.
	
	
	\begin{claim}\label{SO_npc}
		$G$ is $k$-colorable if and only if $H$ is strong odd $k$-colorable.
	\end{claim}
	\begin{proof}
	Let $\varphi$ be a proper $k$-coloring of $G$. We extend this coloring to a strong odd $k$-coloring of $H$, say $\varphi^*$. For each $v_i'\in V'$, $\varphi^*(v_i') = \varphi(v_i)$. Color each vertex $v_{e_i}\in V_{E}^1$ by avoiding the colors assigned to both of its neighbors in $V'$. Then color each vertex $v_{e_i}'\in V_E^2$ by assigning the same color as assigned to $v_{e_i}$. Assign distinct colors to each vertex of the set  $\{v^i_1, v^i_2, \ldots, v^i_{k-1}\}$ from the set of colors $[k]\setminus\varphi^*(v_i')$.  
	
	Now, we show that $\varphi^*$ is a strong odd coloring of $H$. Since each vertex $x\in P$ is a pendant vertex, either a color $c$ does not appear at all or it appears an odd number of times in $N(x)$. Let $v_i'$ and $v_j'$ be two neighbors of a vertex $v_{e_i}\in V_E^1$. Since $v_i$ and $v_j$ are adjacent in $G$, $v_i'$ and $v_j'$ are assigned distinct colors under $\varphi^*$. Therefore, either a color $c$ does not appear at all, or it appears an odd number of times in $N(v_{e_i})$ for each $v_{e_i}\in V_{E}^1$. Similarly, either a color $c$ does not appear at all, or it appears an odd number of times in $N(v_{e_i}')$ for each $v_{e_i}'\in V_{E}^2$. The set $V'$ is an independent set in $H$. Thus, neighbors of each $v_i'\in V'$ are only in the sets $V_{E}^1$, $V_{E}^2$, and $P$. Since $v_i'\in V'$ is adjacent to $v_{e_i}'$ whenever $v_i'$ is adjacent to $v_{e_i}$ and $\varphi^*(v_{e_i})=\varphi^*(v_{e_i}')$, each color that appears in $N_{H}(v_i)\cap (V_{E}^1\cup V_{E}^2)$ appears an even number of times. Note that each $c\in [k]\setminus\{\varphi^*(v_i')\}$ appears exactly once in $N(v_i')\cap P$. Consequently, each color of the set $[k]\setminus\{\varphi^*(v_i)\}$ appears an odd number of in $N_{H}(v_i)$. Thus, $\varphi^*$ is a strong odd coloring of $G$.
	
	Conversely, let $\psi$ be a strong odd $k$-coloring of $H$. We restrict the coloring $\psi$ to the vertices of the set $V'$, denoted by $\psi'$. Since $v_i$ and $v_j$ have a common $2$-neighbor whenever there is an edge between $v_i$ and $v_j$ in $G$, $\psi'(v_i)\neq \psi'(v_j)$ for each edge $v_iv_j\in E(G)$. Thus, $\psi'$ is a proper $k$-coloring of $G$.
\end{proof}
	
	
It is well known that $k$-\textsc{colorability} is NP-complete, for $k\geq 3$. Therefore, \textsc{strong odd $k$-colorability} is NP-complete for perfect elimination bipartite graphs, $k\geq3$. This completes the proof of Theorem~\ref{pebg_thm}.

<!-- proof-3: body_md -->
We construct a graph $H$ from an arbitrary graph $G$ such that a polynomial time $O(n^{\frac{1}{2}-\varepsilon})$-approximation algorithm $\mathcal{A}$ for \textsc{strong odd chromatic number} gives a polynomial time $O(n^{1-\varepsilon})$-approximation algorithm for \textsc{chromatic number}.
	
	\begin{construction}
		Let $G$ be a graph with the vertex set $V(G) = \{v_1, v_2, \ldots, v_n\}$ and the edge set $E(G) = \{e_1, e_2, \ldots, e_m\}$. We construct a graph $H$ from the graph $G$ in polynomial time using the following steps.
		
		\vspace{-0.2cm}
		
		\begin{enumerate}
			
			\item Take two copies of $G$, say $G_1$ and $G_2$. Let $V(G_1) = V_1 = \{u_1, u_2, \ldots, u_n\}$ and $V(G_2) = V_2 = \{u_1', u_2', \ldots, u_n'\}$.
			
		
			
			\item Let $v_i\in V(G)$ be the vertex corresponding to $u_i\in V_1$ and $u_i'\in V_2$. Attach $d(v_i)+n$ pendant vertices to each vertex $u_i\in V_1$ and $u_i'\in V_2$. Let $X_{u_i} = \{u_i^1, u_i^2, \ldots, u_i^{d(v_i)+n}\}$ and
			$X_{u_i'} = \{(u_i')^1, (u_i')^2, \ldots, (u_i')^{d(v_i)+n}\}$
			represent the sets of all the pendant vertices attached to
			$u_i\in V_1$ and $u_i'\in V_2$, respectively.  Let $X = X_{u_1}\cup X_{u_2}\cup\ldots X_{u_n} \cup X_{u'_1}\cup X_{u'_2}\cup\ldots X_{u'_n}$. 
			
			
			
			\item Add a vertex $z$ and make it adjacent to each vertex of $V_1\cup V_2\cup X$.
			
			\item Attach $n$ number of pendant vertices to the vertex $z$. Let $S = \{s_1, s_2, \ldots, s_n\}$ be the set of all the pendant vertices attached to the vertex $z$.
		\end{enumerate}
		
	\end{construction}
	

	
	\begin{claim} \label{reduction_iff}
		$\chi_{\text{so}}(H)\leq\chi(G)+2$.
	\end{claim}
\begin{proof}
	We prove this by showing that $H$ admits a strong odd coloring with at most $\chi(G)+2$ colors.
	
	Let $\varphi:V(G)\rightarrow [k]$ be an optimal proper coloring of $G$. Note that $\chi(G)=k$. We construct a strong odd coloring $\varphi^*$ of $H$ with at most $k+2$ colors. For each $u_i\in V_1$ and $u_i'\in V_2$, assign $\varphi^*(u_i)=\varphi^*(u_i')=\varphi(v_i)$. Assign $\varphi^*(z)=k+1$. For each $s_i\in S$ with $i\le k$, assign $\varphi^*(s_i)=i$. If $n-k$ is even, then choose a color $\beta\in[k]$ and color each uncolored vertex in $S$ with color $\beta$. If $n-k$ is odd, then choose a color $\beta\in[k]$, color all uncolored vertices in $S\setminus\{s_n\}$ with color $\beta$, and assign color $k+2$ to $s_n$. It remains to color the vertices in $X$. Let $u_i\in V_1$. Recall that $u_i$ corresponds to the vertex $v_i\in V(G)$. We color $\{u_i^1, u_i^2, \ldots, u_i^{d(v_i)}\}$ in $X_{u_i}$ with the same colors as the vertices in $N(v_i)$ i.e., for every color $j\in[k]$, the number of vertices among $\{u_i^1, u_i^2, \ldots, u_i^{d(v_i)}\}$ receiving the color $j$ is equal to the number of vertices in $N_G(v_i)$ receiving the color $j$. We color $\{u_i^{d(v_i)+1}, u_i^{d(v_i)+2}, \ldots, u_i^{d(v_i)+k-1}\}$ in $X_{u_i}$ with $k-1$ colors in $[k]\setminus\{\varphi^*(v)\}$ so that each color is used exactly once. If $n-d(v_i)-k$ is even, then choose a color $\beta\in[k]\setminus\{\varphi^*(v)\}$ and color each vertex in $\{u_i^{d(v_i)+1}, u_i^{d(v_i)+2}, \ldots, u_i^{d(v_i)+k}\}$ with color $\beta$. If $n-d(v_i)-k$ is odd, then choose a color $\beta\in[k]\setminus\{\varphi^*(v)\}$, color all uncolored vertices in $X_v\setminus\{v^{d(v)+n}\}$ with color $\beta$, and assign color $k+2$ to $v^{d(v_i)+n}$. Then color each vertex of $X_{u_i}'$ with the same colors used in $X_{u_i}$, so that the number of vertices colored with color $j$ in $X_{u_i}$ is equal to the number of vertices colored with color $j$ in $X_{u_i'}$.
	
	\textbf{Proper coloring:} We first show that $\varphi^*$ is a proper coloring of $H$. Since $\varphi$ is proper, $\varphi^*$ is proper on $G_1$ and $G_2$. Note that $\varphi^*(z)=k+1$, which is distinct from the color assigned to any of its neighbors. Finally, each vertex $x\in X_v$ receives a color different from $\varphi^*(v)$, and each $s_i\in S$ receives a color different from $\varphi^*(z)$. Thus, $\varphi^*$ is proper.
	
	\textbf{Strong odd condition:} Now we show that each vertex $v\in V(H)$ satisfies the strong odd condition. 
	
	\textbf{Vertices of $V_1\cup V_2$:} First, consider a vertex $v\in V_1\cup V_2$. Fix a color $i\in[k]$ with $i\neq\varphi^*(v)$. By construction, the number of vertices of color $i$ in $(N(v)\cap(V_1\cup V_2))\cup\{v^1, v^2, \ldots, v^{d(v)}\}$ is even. The color $i$ appears an odd number of times among $\{v^{d(v)+1}, v^{d(v)+2}, \ldots, v^{n}\}$. Therefore, color $i$ appears an odd number of times in $N(v)$. Note that the color $k+1$ appears exactly once in $N(v)$, and the color $k+2$ appears at most once in $N(v)$. Thus, every vertex $v\in V_1\cup V_2$ satisfies the strong odd condition.
	
	\textbf{Vertices of $X\cup S$:} Let $x\in X_v$. Then $x$ is adjacent to its corresponding vertex $v\in V_1\cup V_2$ and to $z$. Since $\varphi^\ast(v)\neq\varphi^\ast(z)$, each of these colors appears exactly once in $N(x)$. Therefore, $x$ satisfies the strong odd condition. For every $s_i\in S$, we have $N(s_i)={z}$. Thus, the color $k+1$ appears exactly once in $N(s_i)$, so $s_i$ satisfies the strong odd condition. Therefore, every vertex in $X\cup S$ satisfies the strong odd condition.
	
	\textbf{Vertex $z$:} Finally, consider $z$. We show that every color in $[k]\cup\{k+2\}$ appears an odd number of times in $N(z)$. Fix a color $i\in[k]\cup\{k+2\}$. The number of vertices of color $i$ in $V_1\cup X_{u_1}\cup X_{u_2}\cup \ldots \cup X_{u_n} $ is equal to the number of vertices of color $i$ in $V_2\cup X_{u_1'}\cup X_{u_2'}\cup \ldots \cup X_{u_n'} $. Hence, the total number of vertices of color $i$ in $V_1\cup V_2\cup X$ is even. Note that the color $i$ appears an odd number of times in $S$. Therefore, color $i$ appears an odd number of times in $N(z)$. Hence, $z$ also satisfies the strong odd condition. 
	
	Thus, $\varphi^*$ is a strong odd coloring of $H$ with at most $\chi(G)+2$ colors.
\end{proof}

Assume that there exists a polynomial time algorithm $\mathcal{A}$ that approximates \textsc{strong odd chromatic number} within a factor of $N^{\frac{1}{2}-\varepsilon}$ for a graph with $N$ vertices. Let $\mathcal{A}$ return a strong odd coloring $\psi$ of $H$ with $\alpha$ number of colors. Therefore, 
$\alpha\le N^{\frac{1}{2}-\varepsilon}\cdot\chi_{\text{so}}(H).$
Note that $N=4m+2n^2+3n+1\le6n^2$. By Claim~\ref{reduction_iff}, $\chi_{\text{so}}(H)\le\chi(G)+2$. Since $\chi(G)\ge1$, $\chi(G)+2\le3\cdot\chi(G)$. Thus, 
$\alpha\le (4m+2n^2+3n+1)^{\frac{1}{2}-\varepsilon}\cdot(\chi(G)+2)\le3\cdot(6n^2)^{\frac{1}{2}-\varepsilon}\cdot\chi(G).$
Since $z\in V(H)$ is adjacent to every vertex of $G_1$, $\chi(G)-1\le\chi(H)\le\chi_{\text{so}}(H)$. Since restriction of $\psi$ on $G_1$ is a proper coloring of $G$ with at most $\alpha-1$ colors, $\chi(G)\leq\alpha-1$. We have
$\alpha-1<\alpha\le 3\cdot6^{\frac{1}{2}-\varepsilon}\cdot n^{1-2\varepsilon}\cdot\chi(G)$.
Therefore,

$$\frac{\alpha-1}{\chi(G)}<3\cdot6^{\frac{1}{2}-\varepsilon}\cdot n^{1-2\varepsilon}=O(n^{1-\varepsilon'}),$$

where $\varepsilon'=2\varepsilon$. This is a contradiction to Theorem~\ref{chromaticnumber}. Thus, Theorem~\ref{inapproximation} holds.

<!-- theorem-5: body_md -->
\textsc{strong odd $k$-colorability} parameterized by treewidth and $k$ admits an FPT algorithm with running time $O(n^{O(1)}\cdot(k\cdot9^k)^{\texttt{tw}+1})$.
<!-- end theorem-5 -->
<!-- proof-10: body_md -->
We assume that a nice tree decomposition $\mathcal{T}=(T,\{X_i\}_{i\in V(T)})$ of $G$ with $|X_i|\le \texttt{tw}+1$ for each $i\in V(T)$ is given. Thus, each node of $T$ is of one of the types: (i) leaf node, (ii) introduce node, (iii) forget node, and (iv) join node. Before going into the algorithm, we define an operation that will be used in this section. We define a binary operation $\star$ such that $0\star0=0$, $0\star\texttt{odd}=\texttt{odd}$, $0\star\texttt{even}=\texttt{even}$, $\texttt{odd}\star\texttt{odd}=\texttt{even}$, $\texttt{odd}\star\texttt{even}=\texttt{odd}$, and $\texttt{even}\star\texttt{even}=\texttt{even}$. Now, we proceed with the dynamic programming algorithm for treewidth. Given a bag $X_i$ corresponding to the vertex $i\in V(T)$, a state for the bag is a tuple $(i, \texttt{col},\texttt{par})$, where $i$ determines the bag; $\texttt{col}:X_i\to[k]$ is a vertex coloring of $X_i$; $\texttt{par}:X_i\times[k]\to\{0,\texttt{odd}, \texttt{even}\}$ is a parity function. The value $\texttt{par}(v,c)$ indicates the current parity status of the color $c$ in $N(v)$ among the edges processed so far. Precisely, $\texttt{par}(v,c)=\texttt{odd}$ if $c$ appears an odd number of times, $\texttt{par}(v,c)=\texttt{even}$ if $c$ appears a non-zero even number of times, and $\texttt{par}(v,c)=0$ if $c$ has not yet appeared in $N(v)$. 
    
    
We denote the dynamic programming table corresponding to the node $i\in V(T)$ by $\mathcal{S}_i$ and the value corresponding to the state $(i,\texttt{col}, \texttt{par})$ is denoted by $\mathcal{S}_i(i,\texttt{col}, \texttt{par})$. For a node $i\in T$, let $V_i$ denote the set of vertices of $G$ appearing in bags of the subtree rooted at $i$. For a state $(i, \texttt{col}, \texttt{par})$, suppose there is a proper coloring $\varphi: V_i\to[k]$ such that (i) its restriction to $X_i$ is the coloring $\texttt{col}$, (ii) For each $(v,c)\in X_i\times[k]$, if $\texttt{par}(v, c)=\texttt{odd}$, $c$ appears an odd number of times in $N(v)\cap V_i$; if $\texttt{par}(v, c)=\texttt{even}$, $c$ appears an even($\ne0$) number of times $N(v)\cap V_i$; if $\texttt{par}(v, c)=0$, then $c$ does not appear at all in $N(v)\cap V_i$, (iii) $\texttt{par}(v,c)\in\{0,\texttt{odd}\}$ for each $c\in[k]$ if $i$ is a forget node forgetting the vertex $v$. Then $\mathcal{S}_i(i,\texttt{col},\texttt{par})=1$. Otherwise, $\mathcal{S}_i(i,\texttt{col},\texttt{par})=0$. Intuitively, we are assigning $0$ to only those states, for which $\texttt{col}$ cannot be extended to a strong odd coloring of the graph $G$. For a coloring to be strong odd, it should be proper, and each vertex should satisfy the strong odd condition. Therefore, we have condition (i) and (iii). Note that $\texttt{par}$ should correctly store the parity of each color in the neighborhood of each vertex $v\in X_i$ for a state $(i,\texttt{col},\texttt{par})$. Therefore, we have condition (ii).


Our dynamic programming algorithm will compute $\mathcal{S}_i$ for each state corresponding to each node $i\in V(T)$. For the root $X_r=\{z\}$, if there is a state $(r, \texttt{col}, \texttt{par})$ such that $\texttt{par}(z,c)\in\{0, \texttt{odd}\}$ for each $c\in[k]$ and $\mathcal{S}(r, \texttt{col}, \texttt{par})=1$, then $G$ has a strong odd $k$-coloring. Now, we describe our dynamic programming in cases according to the types of nodes of the tree decomposition.


\textbf{Leaf node:} Let $i$ be a leaf node with $X_i=\{z\}$. Then,
$$\mathcal{S}_i(i,\texttt{col},\texttt{par}) = \begin{cases} 1, \text{ if } \texttt{par}(z,c)=0 \text{ for each } c\in[k];\\
0, \text{ otherwise};\end{cases}$$

For the correctness, note that no neighbor of $z$ is colored yet in the graph processed so far, as a leaf node only contains $z$.

\smallskip

\textbf{Introduce node:} Let $i$ be an introduce node with child $j$ introducing a vertex $v$. Note that $X_i=X_j\cup\{v\}$. Consider a state $(i, \texttt{col}, \texttt{par})$. Let $\texttt{col}(v)=\alpha$. Since $\mathcal{T}$ is a tree decomposition, each vertex $u\in N(v)\cap V_i$ is in the bag $X_i$. Thus, $\texttt{par}(v,c)$ should be as per the occurrence of color $c$ in $N(v)\cap X_i$. Moreover, $\texttt{par}(z,\alpha)$ changes for each $z\in N(v)\cap X_i$. Note that $\texttt{par}(z,c)$ remains the same for each $(z,c)\in N(v)\times([k]\setminus\{\alpha\})$. Also, $\texttt{par}(z,c)$ remains the same for each $(z,c)\in (X_i\setminus N(v))\times[k]$. Let $(j, \texttt{col}', \texttt{par}')$ be a state such that (i) $\texttt{col}'=\texttt{col}|_{X_j}$, (ii) $\texttt{par}'(z,c)=\texttt{par}(z,c)$, for each $(z,c)\in (X_i\setminus N(v))\times[k]$, (iii) $\texttt{par}'(z, \alpha)=\texttt{par}(z,\alpha)\star\texttt{odd}$, for each $z\in N(v)$, and (iv) for each $c\in[k]$, $\texttt{par}(v,c)=\Phi_{v}^c$, where $$\Phi_v^c=\begin{cases} 0, \text{ if } c \text{ does not appear in } N(v)\cap X_i;\\\texttt{odd}, \text{ if } c \text{ appears an odd number of times in } N(v)\cap X_i;\\\texttt{even}, \text{ if } c \text{ appears an even } (\ne0) \text{ number of times in } N(v)\cap X_i.\end{cases}$$

Then, we have 
$\mathcal{S}_i(i, \texttt{col}, \texttt{par}) = \mathcal{S}_j(j, \texttt{col}', \texttt{par}').$
Otherwise,
$\mathcal{S}_i(i,\texttt{col}, \texttt{par}) = 0.$


\smallskip

\textbf{Forget Node:}
Let $i$ be a forget node with child $j$ forgetting a vertex $v$. Note that $X_i=X_j\setminus\{v\}$. Consider a state $(i, \texttt{col}, \texttt{par})$. Note that $\texttt{col}$ cannot be extended to a strong odd coloring of $G$ if $\texttt{par}(v,c)\notin\{0, \texttt{odd}\}$.
Let $(j, \texttt{col}', \texttt{par}')$ be a state such that (i) $\texttt{col}'|_{X_i}=\texttt{col}$, (ii) $\texttt{par}'(z,c)=\texttt{par}(z,c)$ for each $(z,c)\in (X_j\setminus\{v\})\times[k]$, (iii) $\texttt{par}'(v,c)\in\{0,\texttt{odd}\}$ for each $c\in[k]$, and (iv) $\mathcal{S}_j(j, \texttt{col}', \texttt{par}')=1$.

Then, we have $\mathcal{S}_i(i, \texttt{col}, \texttt{par}) = 1$.
Otherwise, $\mathcal{S}(i,\texttt{col}, \texttt{par}) = 0$.


\smallskip

\textbf{Join Node:}
Let $i$ be a join node with children $j$ and $k$ such that $X_i=X_j=X_k$. Consider the states $s=(i, \texttt{col}, \texttt{par}), s'=(j, \texttt{col}', \texttt{par}')$, and $s''=(k, \texttt{col}'', \texttt{par}'')$. We say that the states $s, s'$, and $s''$ are consistent with each other if (i) $\texttt{col} = \texttt{col}' = \texttt{col}''$, and (ii) for each $(v,c)\in X_i\times[k]$, $\texttt{par}(v,c) = ((\texttt{par}'(v,c)\star\texttt{par}''(v,c))\star\Phi_v^c)$, where $$\Phi_v^c=\begin{cases} 0, \text{ if } c \text{ does not appear in } N_{X_i}(v);\\ \texttt{odd}, \text{ if } c \text{ appears an odd number of times in } N_{X_i}(v);\\ \texttt{even}, \text{ if } c \text{ appears an even number of times in } N_{X_i}(v);\end{cases}$$




Since $\texttt{col}=\texttt{col}'=\texttt{col}''$, if a color $c$ appears in $N(v)\cap X_i$ for some $v\in X_i$, then this occurrence is counted in both subtrees rooted at $j$ and $k$. Hence, directly combining $\texttt{par}'$ and $\texttt{par}''$ results in counting this occurrence twice. Therefore, we use $\Phi_v^c$ to correct this double counting and obtain the actual parity.

If there exist states $s'$ corresponding to $j$ and $s''$ corresponding to $k$ such that $\mathcal{S}(s')=\mathcal{S}(s'')=1$ and $s,s',$ and $s''$ are consistent, then $\mathcal{S}(s) = 1$. Otherwise, $\mathcal{S}(s) = 0$.

\smallskip

Now we prove the recurrence formulas used in the dynamic programming algorithm above.

\textit{Introduce node:}
We prove the correctness of the recurrence formula by showing that $\mathcal{S}_i(i,\texttt{col},\texttt{par})=1$ if and only if $\mathcal{S}_j(j,\texttt{col}',\texttt{par}')=1$.

First assume that $\mathcal{S}_i(i,\texttt{col},\texttt{par})=1$. Note that there is exactly one state, say $(j,\texttt{col}|_{X_j},\texttt{par}')$, that satisfies all the conditions. Since $\texttt{par}'$ satisfies the above-mentioned conditions, there exists a proper coloring $\varphi$ satisfying all the conditions for $\mathcal{S}_j(j,\texttt{col}|_{X_j},\texttt{par}')$ to be $1$.


Conversely, assume that $\mathcal{S}_j(j,\texttt{col}|_{X_j},\texttt{par}')=1$. Since $v$ is introduced in $X_i$ and $\texttt{par}$ satisfies the condition (iv), there exists a proper coloring $\varphi$ satisfying all the conditions for $\mathcal{S}_i(i,\texttt{col},\texttt{par})$ to be $1$. Therefore, $\mathcal{S}_i(i,\texttt{col},\texttt{par})=1$.

\smallskip

\noindent\textit{Forget node:}
Let $i$ be a forget node with child $j$ such that $X_i=X_j\setminus\{v\}$. We prove the correctness of the recurrence formula by showing that $\mathcal{S}_i(i,\texttt{col},\texttt{par})=1$ if and only if $\mathcal{S}_j(j,\texttt{col}',\texttt{par}')=1$ and $\texttt{par}(v,c)\in\{0, \texttt{odd}\}$ for each $c\in[k]$.

First, assume that $\mathcal{S}_i(i,\texttt{col},\texttt{par})=1$. Note that there $2^k$ states that satisfies all the required conditions except (iv). Although there is at most one state, say $(j,\texttt{col}|_{X_j},\texttt{par}')$, that satisfies all the conditions. Since $\texttt{col}'$ and $\texttt{par}'$ satisfies the above-mentioned conditions, there exists a proper coloring $\varphi$ satisfying all the conditions for $\mathcal{S}_j(j,\texttt{col}|_{X_j},\texttt{par}')$ to be $1$. Since we are forgetting a vertex $v$, the neighborhood of $v$ is processed completely. If $\texttt{par}(v,c)\notin\{0,1\}$ for some color $c\in[k]$, then $v$ does not satisfy strong odd condition. Therefore, $\texttt{col}$ cannot be extended to a strong odd coloring of $G$. Then, $\mathcal{S}_i(i,\texttt{col},\texttt{par})=0$. Therefore, $\texttt{par}(v,c)\in\{0, \texttt{odd}\}$


Conversely, assume that $\mathcal{S}_j(j,\texttt{col}|_{X_j},\texttt{par}')=1$ and $\texttt{par}(v,c)\in\{0, \texttt{odd}\}$ for each $c\in[k]$. Then, there exists a proper coloring $\varphi$ satisfying all the conditions for $\mathcal{S}_i(i,\texttt{col},\texttt{par})$ to be $1$. Therefore, $\mathcal{S}_i(i,\texttt{col},\texttt{par})=1$.


\smallskip
\noindent\textit{Join node:}
First, assume that $\mathcal{S}_i(i,\texttt{col},\texttt{par})=1$. Let $s=(i, \texttt{col}, \texttt{par})$. Note that there exists two states $s'=(j, \texttt{col}', \texttt{par}')$ and $s''=(k, \texttt{col}'', \texttt{par}'')$ such that $s, s'$ and $s''$ are consistent with each other. By the definition of consistency $\mathcal{S}_j(j,\texttt{col}',\texttt{par}')=1$ and $\mathcal{S}_k(k,\texttt{col}'',\texttt{par}'')=1$. 

Conversely, assume that $\mathcal{S}_j(j,\texttt{col}',\texttt{par}')=1$ and $\mathcal{S}_k(k,\texttt{col}'',\texttt{par}'')=1$. Since $\texttt{col} = \texttt{col}' = \texttt{col}''$ and for each $(v,c)\in X_i\times[k]$, $\texttt{par}(v,c) = ((\texttt{par}'(v,c)\star\texttt{par}''(v,c))\star\Phi_v^c)$, $(i, \texttt{col}, \texttt{par})$ satisfoes all the conditions for $S_i(i, \texttt{col}, \texttt{par})$ to be $1$.

\textbf{Running time:}
Note that there is exactly one state corresponding to the child of an introduce node satisfying all the required conditions and it can be searched in $O(1)$ time. Thus, leaf node and introduce node can be processed in time $O(n^{O(1)}\cdot(k.3^k)^{\texttt{tw}+1})$. Now consider a forget node. There is exactly one state corresponding to child of a forget node satisfying all the conditions required and it can be searched in $k\cdot2^k$ time. Therefore, forget node can be processed in time $O(n^{O(1)}\cdot(k.3^k)^{\texttt{tw}+1}\cdot k\cdot2^k)$. Now, consider a join node $i$ with children $j$ and $k$. Fix a coloring $\texttt{col}$ of $X_i$. We enumerate every pair $(\texttt{par}',\texttt{par}'')$ of parity functions corresponding to the two children. For each such pair, the parity function $\texttt{par}$ of the parent is uniquely determined. Since there are $k^{|X_i|}$ possible colorings of $X_i$, join node can be processed in time $O(n^{O(1)}\cdot(k\cdot9^k)^{\texttt{tw}+1})$. Therefore, the total running of the algorithm is $O(n^{O(1)}\cdot (k\cdot9^k)^{\texttt{tw}+1})$.


This shows that strong odd coloring is fixed-parameter tractable when parameterized by the treewidth of the input graph and the number of colors.

We give a parameterized reduction from $k$-\textsc{colorability} parameterized by treewidth. 
	
\begin{construction}\label{construction} Given an instance $(G, \texttt{tw}, k)$ of $k$-\textsc{colorability}, where $\texttt{tw}$ denotes the treewidth of $G$, we construct a graph $H$ from $G$ as follows:
\begin{enumerate}
    \item Initialize $H$ as a copy of $G$. For each vertex $v \in V(G)$, let $v'$ denote the copy of $v$ in $V(H)$, and let 
    $V_1 = \{v' \mid v\in V(G)\}$.
    \item For each vertex $v' \in V_1$, attach $d_G(v) + k - 1$ pendant 
    vertices to $v'$, where $d_G(v)$ denotes the degree of $v$ in $G$. Let $P(v')=\{v_1', v_2', \ldots, v_{(d_G(v)+k-1)}'\}$ be the set of pendant vertices attached to the vertex $v'\in V_1$.
\end{enumerate}
\end{construction}

If $k\geq |V(G)|$, then \textsc{strong odd $k$-colorability} is trivially yes. So
we may assume that $k \leq |V(G)|$. Therefore, $H$ has $|V(G)|+\sum_{v\in V(G)}\bigl(d_G(v)+k-1\bigr)
=|V(G)|+2|E(G)|+|V(G)|\cdot(k-1)=2|E(G)|+|V(G)|\cdot k$ vertices, which is polynomial in the size of the input.

\begin{claim}\label{treewidth_lower_bound}
    $\texttt{tw}(H)=\texttt{tw}(G)$.
\end{claim}
\begin{proof}
Let $\mathcal{T}_G$ be a tree decomposition of $G$ with bag size at most $\texttt{tw}+1$. We construct a tree decomposition $\mathcal{T}_H$ of $H$ starting with $\mathcal{T}_G$. Since $\mathcal{T}_G$ is a tree decomposition, there exists a bag $B_v$ containing $v$ in $\mathcal{T}_G$. Now, attach $d_G(v)+k-1$ number of child bags $B_1, B_2, \ldots, B_{(d(v)+k-1)}$ to $B_v$ with $V(B_i)=\{v', v_i'\}$ for each $i\in[d_G(v)+k-1]$ where $v'$ is the copy of $v\in V(G)$. Do this for every vertex $v\in V(G)$. Then, replace $v$ with its copy $v'$ in $H$. 


Now, we show that $\mathcal{T}_H$ a tree decomposition of $H$. For the copy $v'\in V(H)$ of every vertex $v\in V(G)$, we have added a bag to every edge $v'v_i'$ for each $i\in[d(v)+k-1]$. There is a bag containing $v'$ for every vertex $v'\in V(H)$, and there is a bag containing $u'$ and $v'$ for every edge $u'v'\in E(H)$. Since we have attached bags containing $B_1, B_2, \ldots, B_{(d(v)+k-1)}$ to a bag $B_v$ containing $v$ with $V(B_i)=\{v', v_i'\}$ for each $i\in[d_G(v)+k-1]$ and $v_i'$ is not contained in any other bag, bags containing $v\in V(H)$ induces a connected subtree of $\mathcal{T}_H$. Moreover, we have added bags of size at most $2$. Since $\texttt{tw}\geq1$ and bag size of $G$ is at most $\texttt{tw}+1$, $\mathcal{T}_H$ is a tree decomposition of $H$ with bag size at most $\texttt{tw}(G)+1$. Thus, $\texttt{tw}(G)=\texttt{tw}(H)$.
\end{proof}


\begin{claim}\label{cons:iff}
    $G$ is $k$-colorable if and only if $H$ is strong odd $k$-colorable.
\end{claim}
\begin{proof}
Assume that $\psi : V(G) \to [k]$ is a proper $k$-coloring of $G$. We define a coloring $\varphi$ for $H$. For each vertex $v'\in V_1$, $\varphi(v') = \psi(v)$. For each vertex $v \in V(G)$, let $N(v) = \{u_1, \dots, u_{d_G(v)}\}$. For each $v\in V_1$, $\varphi(v_i') = \psi(u_i)$ for $i\in[d_G(v)]$. For each $v\in V_1$, assign distinct colors from the set $[k] \setminus \{\varphi(v)\}$ to the vertices of 
$\{v_{d_G(v)+1}', v_{d_G(v)+2}' \ldots, v_{d_G(v)+k-1}'\}$, one color per vertex. 

\smallskip

We claim that $\varphi$ is a strong odd coloring of $H$. Since $\psi$ is a proper coloring of $G$ and we do not assign $\varphi(v)$ to any vertex from the set $P(v)$, $\varphi$ is a strong odd coloring of $H$. Now, consider any vertex $v\in V(G)$. Let $c$ be a color that appears in $N_H(v)$. If $c$ appears $m$ number of times in $N(v)$, then $c$ appears $m$ number of times in $\{v_1, v_2, \ldots, v_{d_G(v)}\}$ and once in $\{v_{d_G(v)+1}, v_{d_G(v)+2}, \ldots, v_{d_G(v)+k-1}\}$. Therefore, $c$ appears an odd number of times in $N_H(v)$. Now, assume that $c$ does not appear in $N(v)$. The color $c$ appears exactly once in $N_H(v)$. Therefore, each color appears an odd number of times in $N_H(v)$. Since each vertex $v\in V(H)\setminus V(G)$ is a vertex of degree 1, every color appearing in $N_H(v)$ appears an odd number of times.
Thus, $\varphi$ is a strong odd $k$-coloring of $H$.

Let $\varphi$ be a strong odd $k$ coloring of $H$. Since $G$ is an induced subgraph of $H$, restriction of $\varphi$ to $G$ is a proper $k$-coloring of $G$.
\end{proof}

Lokshtanov et al.~\cite{Lokshtanov2018} have shown that $k$\textsc{-Colorability} cannot be solved in time $(k-\varepsilon)^{\texttt{tw}(G)}n^{O(1)}$ time under SETH, for any fixed $k\ge3$. Therefore, \textsc{strong odd $k$-colorability} parameterized by treewidth $\texttt{tw}$ cannot be solved in time $(k-\varepsilon)^{\texttt{tw}}n^{O(1)}$ under SETH, for $k \geq 3$.
\qed
