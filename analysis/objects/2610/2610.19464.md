\label{thm:global}
Let $f\in\mathcal{B}_A$. Suppose $(a,q)=1$, $|\alpha-a/q|\le q^{-2}$, and
$2\le R\le q\le N/R$. Then
\begin{equation}\label{eq:global}
 |S_f(N,\alpha)|\ll A\bigg(\frac NL+\frac N{\sqrt R}\bigg).
\end{equation}
For $(a,q)=1$, $1\le q\le N$ and $|\alpha-a/q|\le q^{-2}$, one also has
\begin{equation}\label{eq:three-term}
 |S_f(N,\alpha)|\ll A\bigg(\frac NL+\frac N{\sqrt q}+\sqrt{Nq}\bigg).
\end{equation}
Consequently, if $1\le q\le N/L^2$, then
\begin{equation}\label{eq:folklore}
 |S_f(N,\alpha)|\ll A\bigg(\frac NL+\frac N{\sqrt q}\bigg).
\end{equation}

\label{thm:aux}
Let $f\in\mathcal{B}_A$, $Q_0=N/L^3$, and suppose

 (a,r)=1,\qquad 1\le r\le Q_0,\qquad
 \alpha=\frac ar+\beta,\qquad |\beta|\le\frac1{rQ_0}.

 |S_f(N,\alpha)|\ll A\bigg(\frac NL+\frac N{\sqrt{rB}}\bigg).
\end{equation}

\label{thm:refined}
Let $f\in\mathcal{B}_A$, $(a,q)=1$, $q\ge1$, and $\beta\in\mathbb{R}$. Put
$B=\max\{1,N|\beta|\}$ and $D=qB$. Then
\begin{equation}\label{eq:displacement-global}
 |S_f(N,a/q+\beta)|\ll A\bigg(\frac NL+\frac N{\sqrt D}+\sqrt{ND}\bigg).
\end{equation}
If
\begin{equation}\label{eq:wider-local}
 D\le N/L^2,
\end{equation}
one has
\begin{equation}\label{eq:refined}
 |S_f(N,a/q+\beta)|\ll A\bigg(\frac NL+
 \frac N{\sqrt{q\max\{1,N|\beta|\}}}\bigg).
\end{equation}
Condition \eqref{eq:wider-local} is equivalent to $q\le N/L^2$ and
$|\beta|\le1/(qL^2)$. The estimate \eqref{eq:refined} also holds when
$q\le Q_2=N/L^2$ and $|\beta|\le1/(qQ_2)$. In the conclusions,
$\max\{1,N|\beta|\}$ may be replaced by $1+N|\beta|$.

<!-- theorem-4: body_md -->
\label{thm:sharpness-summary}
For every fixed $A>1$, there are arbitrarily large odd squarefree $q$
and functions $f_q\in\mathcal{F}_A$ such that
\begin{equation}\label{eq:counter-summary}
 \sum_{n\le x}|f_q(n)|^2\le x\quad(x\ge1),\qquad
 |f_q(p^k)|\le A\quad(p\text{ prime},\ k\ge1),
\end{equation}
and, for every sufficiently large real $N$ depending on $q$,
simultaneously for every $(a,q)=1$,
\begin{equation}\label{eq:counter-order}
 |S_{f_q}(N,a/q)|\asymp\frac N{\sqrt{\varphi(q)}}.
\end{equation}
One may take $q/\varphi(q)\asymp\log\log q$, and $N$ with $L\ge q$;
then $q\le N/L^2$ and $N/L\le N/q$.

\label{prop:GHS-long}
Let $U$ be sufficiently large, $\ell=\log U$, $r\le\ell^4$, and let $g$
be completely multiplicative with $|g(n)|\le1$. There is a set $\mathcal{E}$
of at most $63$ characters modulo $r$ such that, for every $v$ with
$U^{1/4}\le v\le U$,
\begin{equation}\label{eq:GHS-long}
 \bigg(\sum_{\chi\notin\mathcal{E}}|A_g(v,\chi)|^2\bigg)^{1/2}
 \ll v\ell^{-7/8}(\log\ell)^{15/8}.
\end{equation}
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Apply \cite[Theorems~1.8 and~1.9]{GHS} with $X=U^{1/2}$, modulus $r$,
$\kappa=1$ and $J=63$. The requirement $r\le X^{1/4}=U^{1/8}$ follows
from $r\le\ell^4$, and the range $\sqrt X\le x\le X^2$ of those
theorems is $U^{1/4}\le x\le U$. With $Q=r\log X$, the bound of
\cite[Theorem~1.9]{GHS} for the left side of \eqref{eq:GHS-long} is
\[
 v\Big(\frac{\log Q}{\log v}\Big)^{7/8}
 \log\Big(\frac{\log v}{\log Q}\Big).
\]
Here $\log Q\le5\log\ell$ and $\log v\ge\ell/4$, and
$t^{7/8}\log(1/t)$ is increasing for small $t>0$, so this is bounded by
the right side of \eqref{eq:GHS-long}. The exceptional set in
\cite[Theorem~1.8]{GHS} is the same for every $v$ in the stated range.

<!-- proposition-2: body_md -->
\label{prop:GHS-short-input}
The following estimates hold.
\begin{enumerate}
\item[\textup{(i)}] Let $u\in\mathcal C(1)$, let $x$ be sufficiently
large, $0<\delta<1/4$, and $z=x^{1-\delta}$. Then
\begin{equation}\label{eq:short-halasz}
 \bigg|\sum_{x<n\le x+z}u(n)\bigg|
 \ll z\bigg((1+M)e^{-M}+\frac{\log(x/z)+\log\log x}{\log x}\bigg),
\end{equation}
where $M$ is defined by
\begin{equation}\label{eq:short-M}
 e^{-M}\log x=\max_{|t|\le(x/z)\log x}
 \bigg|\sum_{n\ge1}\frac{u(n)}{n^{1+1/\log x+it}}\bigg|.
\end{equation}
\item[\textup{(ii)}] Let $K\ge1$ be fixed and let $\chi$ be a
nonprincipal character modulo $r$. If $X\ge10$, $y\ge r\log X$,
$\sigma\ge1+1/\log X$ and $|t|\le(\log X)^K$, then
\begin{equation}\label{eq:rough-L}
 |L_y(\sigma+it,\chi)|\ll_K1,
 \qquad L_y(s,\chi)=\prod_{p>y}\big(1-\chi(p)p^{-s}\big)^{-1}.
\end{equation}
\item[\textup{(iii)}] Let $\kappa>0$ be fixed and $u\in\mathcal C(\kappa)$.
If $x$ is sufficiently large in terms of $\kappa$, then
\begin{equation}\label{eq:Shiu}
 \sum_{n\le x}|u(n)|\ll_\kappa\frac x{\log x}
 \exp\bigg(\sum_{p\le x}\frac{|u(p)|}p\bigg).
\end{equation}
\end{enumerate}
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
Part~(i) is \cite[Corollary~1.4]{GHS} with $\kappa=1$, and part~(ii) is
\cite[Lemma~5.1]{GHS}. For part~(iii), put $g(n)=|u(n)|$. By
\cite[Section~1]{GHS}, $g(n)\le d_\kappa(n)$, where $d_\kappa(n)$ is
the coefficient of $n^{-s}$ in $\zeta(s)^\kappa$, so that
$g(p^j)\le d_\kappa(p^j)=\binom{\kappa+j-1}{j}$.
Differentiating $\sum_{j\ge0}d_\kappa(p^j)z^j=(1-z)^{-\kappa}$ gives
\[
 \sum_{j\ge2}\frac{j g(p^j)}{p^j}
 \le\frac\kappa p\Big((1-p^{-1})^{-\kappa-1}-1\Big)
 \ll_\kappa p^{-2}.
\]
Since $g(p)\le\kappa$, the Chebyshev bound
\cite[Corollary~2.6]{MV2} now yields
\[
 \sum_{p\le t}g(p)\log p\ll_\kappa t\quad(t\ge1),\qquad
 \sum_p\sum_{j\ge2}\frac{j g(p^j)\log p}{p^j}
 \ll_\kappa\sum_p\frac{\log p}{p^2}<\infty.
\]
Thus \cite[Theorem~2.14 and Corollary~2.15]{MV2}, applied to the
nonnegative multiplicative function $g$, give, for $x\ge2$,
\[
 \sum_{n\le x}g(n)
 \ll_\kappa\frac x{\log x}
 \prod_{p\le x}\Big(1+\sum_{j\ge1}\frac{g(p^j)}{p^j}\Big)
 \ll_\kappa\frac x{\log x}
 \exp\bigg(\sum_{p\le x}\frac{g(p)}p\bigg),
\]
where the last step uses $\log(1+t)\le t$ for $t\ge0$ and the first
prime-power bound. For short intervals and arithmetic progressions,
see \cite[Lemma~2.3]{GHS} and Shiu's theorem \cite[Theorem~1]{S}.

<!-- proposition-3: body_md -->
\label{prop:sampling}
Let $c=(c_n)$ be a sequence of complex numbers supported on at most $V$
consecutive integers, put $E=\sum_n|c_n|^2$, and let
$M_c(\theta)$ be the maximum over integers $k$ of
$|\sum_{n\le k}c_n\operatorname{e}(n\theta)|$. Then
\begin{align}
 \int_0^1M_c(\theta)^2\,d\theta&\ll E,\label{eq:CH}\\
 \sum_{b\bmod r}M_c(\theta+b/r)^2&\ll(V+r)E\label{eq:fullsample}
\end{align}
for every integer $r\ge1$ and every real $\theta$. The implied
constants are absolute.
<!-- end proposition-3 -->
<!-- proof-3: body_md -->
When the support is contained in $\{1,\ldots,V\}$, these are
\cite[Proposition~2.3 and (4.4)]{Robles}. A shift of the support
multiplies every sum $\sum_{n\le k}c_n\operatorname{e}(n\theta)$ by the same factor of
modulus $1$, and so does not change $M_c$.

<!-- lemma-1: body_md -->
\label{lem:Gallagher}
Let $(a_n)$ be a finitely supported sequence in a complex Hilbert
space. For $\delta>0$ and $h_0=1/(2\delta)$,
\begin{equation}\label{eq:Gallagher}
 \int_{-\delta}^{\delta}\bigg\|\sum_na_n\operatorname{e}(n\theta)\bigg\|^2\,d\theta
 \ll h_0^{-2}\int_{\mathbb{R}}\bigg\|\sum_{t<n\le t+h_0}a_n\bigg\|^2\,dt.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-4: body_md -->
Let $\Phi(t)=\sum_na_n\mathbf 1_{[n-h_0,n)}(t)$, so that
$\Phi(t)=\sum_{t<n\le t+h_0}a_n$. The Fourier transform of $\Phi$ at
$\theta$ is the product of $\sum_na_n\operatorname{e}(-n\theta)$ and a factor of
modulus $|\sin(\pi h_0\theta)|/(\pi|\theta|)$, which is at least
$2h_0/\pi$ for $|\theta|\le1/(2h_0)$. Plancherel's theorem, applied to
the coordinates of $\Phi$ in an orthonormal basis of the span of the
$a_n$, gives \eqref{eq:Gallagher}.

\label{prop:old-blocks}
Let $f\in\mathcal{F}_A$, suppose that \eqref{eq:aux-hyp} holds, and put $D=rB$.

\begin{equation}\label{eq:reduction}
 L\,S_f(N,\alpha)=\sum_{\substack{Y\le P<N\\P\textup{ dyadic}}}T(P)+O_A(N),
\end{equation}
and for every dyadic $P$ with $Y\le P<N$, writing
$U=\lfloor N/P\rfloor$,
\begin{align}

 +\sqrt{\frac{rNP}{\varphi(r)}},\label{eq:old-block}\\
 |T(P)|&\ll_AN\sqrt{\frac{\log\log(3U)}U}
 \qquad(U\le r).\label{eq:old-short}
\end{align}
\item[\textup{(iii)}] Under the hypotheses of \textup{(ii)}, let $P$ be
dyadic with $Y\le P<N$, let $U=\lfloor N/P\rfloor$ and

 \sum_{P<p\le2P}M_c(U;\alpha p)^2\log p
 \ll\frac1{\varphi(r)}\sum_{b\bmod r}^{*}\int_P^{2P}
 \big(&M_c(U;ab/r+\beta t)^2\notag\\
 &+h^2|\beta|^2M_d(U;ab/r+\beta t)^2\big)\,dt,\label{eq:positive-transfer}
\end{align}

\end{enumerate}

\]

\begin{equation}\label{eq:h-tail}
 \sum_{d>x}\frac{|h(d)|}d\ll_Ax^{-1/4},\qquad
 \sum_{d\le x}|h(d)|\ll_Ax^{3/4}.
\end{equation}

<!-- proof-6: body_md -->
Set $g(p)=f(p)/\max\{1,|f(p)|\}$ and extend $g$ completely
multiplicatively. The function $h$ is then determined by
\[
 h(p)=f(p)-g(p),\qquad
 h(p^k)=f(p^k)-g(p)f(p^{k-1})\quad(k\ge2).
\]
For a finite set of primes with $|f(p)|>1$, multiplicativity gives
$\prod_p|f(p)|\le A$. Since a product of numbers $1+b_p$ with
$b_p\ge0$ is at least $1+\sum_pb_p$, it follows that
\[
 \sum_p\big(|f(p)|-1\big)_+\le A-1.
\]
Moreover, $|h(p)|=(|f(p)|-1)_+$ and $|h(p^k)|\le2A$ for $k\ge2$. The
Euler product for the left side of \eqref{eq:h-norm} is at most
\[
 \exp\bigg(A-1+2A\sum_p\frac{p^{-2\sigma}}{1-p^{-\sigma}}\bigg)<\infty.
\]
Taking $\sigma=3/4$ and comparing powers proves \eqref{eq:h-tail}.

<!-- lemma-3: body_md -->
\label{lem:linear-A}
Let $A\ge1$, $X\ge1$ and $f\in\mathcal{B}_A$. There are finitely many functions
$g_j\in\mathcal B_1$ and real numbers $w_j$ with $\sum_j|w_j|\le A$
such that
\begin{equation}\label{eq:decomposition-A}
 f(n)=\sum_jw_jg_j(n)\qquad(n\le X).
\end{equation}
Consequently, if $\mathscr N$ is a seminorm on
$\mathbb{C}^{\lfloor X\rfloor}$, then
\begin{equation}\label{eq:seminorm-A}
 \mathscr N\big((f(n))_{n\le X}\big)
 \le A\sup_{g\in\mathcal B_1}\mathscr N\big((g(n))_{n\le X}\big),
\end{equation}
and in particular, for all complex numbers $c_n$,
\begin{equation}\label{eq:linear-A}
 \bigg|\sum_{n\le X}c_nf(n)\bigg|
 \le A\sup_{g\in\mathcal B_1}\bigg|\sum_{n\le X}c_ng(n)\bigg|.
\end{equation}
<!-- end lemma-3 -->
<!-- proof-7: body_md -->
For each prime $p\le X$, put
\[
 m_p=\max\big(\{1\}\cup\{|f(p^k)|:p^k\le X\}\big).
\]
For each $p$ with $m_p>1$, choose a prime power attaining the maximum,
and let $n_0$ be the product of these prime powers. Multiplicativity
and the bound $|f(n_0)|\le A$ give $\prod_{p\le X}m_p=|f(n_0)|\le A$.
The integer $n_0$ need not be at most $X$.

Define $u(p^k)=f(p^k)/m_p$ for $p^k\le X$. For each vector
$\varepsilon=(\varepsilon_p)_{p\le X}$ of signs $\varepsilon_p=\pm1$, let $g_\varepsilon$ be the
multiplicative function with
\[
 g_\varepsilon(p^k)=\varepsilon_pu(p^k)\quad(p^k\le X),
\]
and $g_\varepsilon(p^k)=0$ at all other prime powers, and put
\[
 w(\varepsilon)=\prod_{p\le X}\frac{1+\varepsilon_pm_p}2.
\]
Each $g_\varepsilon$ belongs to $\mathcal B_1$. For $m\ge1$,
\[
 \sum_{\varepsilon=\pm1}\frac{1+\varepsilon m}2=1,\qquad
 \sum_{\varepsilon=\pm1}\varepsilon\,\frac{1+\varepsilon m}2=m,\qquad
 \sum_{\varepsilon=\pm1}\bigg|\frac{1+\varepsilon m}2\bigg|=m.
\]
Taking products of these identities over $p\le X$ gives
\[
 f(n)=\sum_\varepsilon w(\varepsilon)g_\varepsilon(n)\quad(n\le X),
 \qquad\sum_\varepsilon|w(\varepsilon)|=\prod_{p\le X}m_p\le A,
\]
which is \eqref{eq:decomposition-A}. The triangle inequality and the
homogeneity of $\mathscr N$ give
\[
 \mathscr N\big((f(n))_{n\le X}\big)
 \le\sum_\varepsilon|w(\varepsilon)|\,\mathscr N\big((g_\varepsilon(n))_{n\le X}\big),
\]
which proves \eqref{eq:seminorm-A}. Applying this to
$\mathscr N((z_n))=|\sum_{n\le X}c_nz_n|$ gives \eqref{eq:linear-A}.

<!-- lemma-4: body_md -->
\label{lem:Gauss-mass}
If $\chi$ is a character modulo $r$ of conductor $d$, then
\begin{equation}\label{eq:Gamma-mass}
 \sum_{v\bmod r}|\Gamma_\chi(v;r)|
 =\sqrt d\,2^{\omega(r)-\omega(d)}\le\frac4{\sqrt6}\sqrt r.
\end{equation}
Also $|\Gamma_\chi(n;r)|\le1$.
<!-- end lemma-4 -->
<!-- proof-8: body_md -->
For a character $\psi$ modulo $m$, write
\[
 \tau(\psi)=\sum_{v\bmod m}\psi(v)\operatorname{e}(v/m),
\]
with the character and the Gauss sum modulo $1$ both equal to $1$. Let
$\chi^*$ be the primitive character modulo $d$ inducing $\chi$. By
\cite[Theorem~9.7]{MV2},
\begin{equation}\label{eq:induced-Gauss}
 |\tau(\chi^*)|=\sqrt d.
\end{equation}
Put $m=r/(n,r)$ and $n'=n/(n,r)$. By \cite[Theorem~9.12]{MV2},
$\Gamma_\chi(n;r)=0$ unless $d\mid m$, and for $d\mid m$
\[
 \Gamma_\chi(n;r)=\overline{\chi^*(n')}\,\chi^*(m/d)\,\mu(m/d)\,
 \frac{\tau(\chi^*)}{\varphi(m)}.
\]
Here $|\chi^*(n')|=1$, because $(n',m)=1$ and $d\mid m$. Call $m$
admissible if $d\mid m$ and $m/d$ is squarefree and coprime to $d$.
By \eqref{eq:induced-Gauss}, $|\Gamma_\chi(n;r)|=\sqrt d/\varphi(m)$ if
$m$ is admissible, and $\Gamma_\chi(n;r)=0$ otherwise.
For each $m\mid r$ there are $\varphi(m)$ residue classes
$n$ modulo $r$ with $r/(n,r)=m$, so each admissible $m$ contributes
$\sqrt d$ to the sum in \eqref{eq:Gamma-mass}. The admissible $m$ are
the products $dk$, where $k$ is a product of distinct primes dividing
$r$ but not $d$; there are $2^{\omega(r)-\omega(d)}$ of them. This
gives the equality in \eqref{eq:Gamma-mass}.

For $p^\nu\Vert r$ and $p^\delta\Vert d$ with $\delta>0$, the local
factor of the ratio of $\sqrt d\,2^{\omega(r)-\omega(d)}$ to $\sqrt r$
is $p^{(\delta-\nu)/2}\le1$. If $\delta=0$, it is $2p^{-\nu/2}$. Only
$p=2$ and $p=3$ can give a factor larger than $1$, and the product of
these two factors is at most $4/\sqrt6$. The constant $4/\sqrt6$ is
attained by the principal character modulo $6$, for which the sum in
\eqref{eq:Gamma-mass} is $4$. The underlying bound
$2^{\omega(n)}\le(4/\sqrt6)\sqrt n$ is also used in \cite{GL}, after
(7.22) there. The bound $|\Gamma_\chi(n;r)|\le1$ is immediate from
\eqref{eq:Gamma}.

<!-- lemma-5: body_md -->
\label{lem:one-character}
Let $U\ge1$, let $|c_n|\le A$ for $1\le n\le U$, and let $C_\chi$ and
$F_y$ be as in \eqref{eq:char-expansion}. Then
\begin{align}
 \sup_{y\le U,\,\theta\in\mathbb{R}}|C_\chi(y,\theta)|
 &\ll A\frac{U+r}{\sqrt r},\label{eq:one-point}\\
 \int_I\max_{y\le U}|C_\chi(y,\theta)|^2\,d\theta
 &\ll A^2\frac{U+r}r\label{eq:one-energy}
\end{align}
for every real interval $I$ of length at most $1/r$. If $\mathcal{E}$ is a
set of $J$ characters modulo $r$, then for every such interval $I$ one has
\begin{equation}\label{eq:finite-rank-energy}
 \frac1{\varphi(r)}\sum_{b\bmod r}^{*}\int_I
 \max_{y\le U}|P_\mathcal{E} F_y(b,\theta)|^2\,d\theta
 \ll A^2J^2\frac{U+r}r.
\end{equation}
<!-- end lemma-5 -->
<!-- proof-9: body_md -->
Group $n$ according to its residue $v$ modulo $r$. Each group has at
most $U/r+1$ terms, and \eqref{eq:one-point} follows from
\eqref{eq:Gamma-mass}. For the integral, the sum over the $n\le y$ in one group
is $\operatorname{e}(v\theta)\sum_kc_{v+rk}\operatorname{e}(rk\theta)$, where $k$ runs over the
integers with $1\le v+rk\le y$. Its modulus is a function of
$\phi=r\theta$ of period $1$, and $I$ corresponds to an interval of
$\phi$ of length at most $1$. By \eqref{eq:CH}, applied in the variable
$\phi$, and $d\theta=d\phi/r$, the integral over $I$ of the square of
its maximum over $y\le U$ is $\ll A^2(U/r+1)/r$. Minkowski's inequality and \eqref{eq:Gamma-mass}
then give \eqref{eq:one-energy}. Finally, the Cauchy-Schwarz inequality
gives, for every $b$ and $\theta$,
\[

 \le J\sum_{\chi\in\mathcal{E}}\max_{y\le U}|C_\chi(y,\theta)|^2,
\]
and \eqref{eq:finite-rank-energy} follows from \eqref{eq:one-energy}
on integrating over $I$ and averaging over $b$.

<!-- lemma-6: body_md -->
\label{lem:smooth-weights}
For every $r\ge1$ and $x\ge1$,
\begin{align}
 &\sum_{s\in\mathcal{S}_r}s^{-1}=H_r,\qquad
 \sum_{s\in\mathcal{S}_r}s^{-1/2}\le r^2,\label{eq:smooth-sums}\\
 &\sum_{\substack{s\in\mathcal{S}_r\\s>x}}s^{-1}\le r^2x^{-1/2},\qquad
 \#\{s\in\mathcal{S}_r:s\le x\}\le r^2x^{1/2},\label{eq:smooth-tail}\\
 &W_r:=\sum_{s\in\mathcal{S}_r}\frac{\max_\chi|\Gamma_\chi(s;r)|}s
 \le\frac{H_rV_r}{\sqrt r}.\label{eq:W}
\end{align}
For fixed nonnegative $a,b$ and every $\nu>0$,
\begin{equation}\label{eq:subpower}
 H_r^aV_r^b\ll_{a,b,\nu}r^\nu.
\end{equation}
<!-- end lemma-6 -->
<!-- proof-10: body_md -->
The assertions in \eqref{eq:smooth-sums} follow from Euler products and
$(1-p^{-1/2})^{-1}\le p^2$, and those in \eqref{eq:smooth-tail} follow
by comparing powers. For \eqref{eq:W}, group $s$ according to
$m=r/(s,r)$, and write $H_m=m/\varphi(m)$. The condition $r/(s,r)=m$
means that, for each prime $p\mid r$, the power of $p$ dividing $s$ is
exactly that dividing $r/m$ if $p\mid m$, and is at least that dividing
$r$ if $p\nmid m$. Hence the sum of $1/s$ over the $s\in\mathcal{S}_r$ with
$r/(s,r)=m$ is
\[
 \frac mr\prod_{\substack{p\mid r\\p\nmid m}}(1-p^{-1})^{-1}
 =\frac mr\frac{H_r}{H_m}.
\]
The proof of Lemma~\ref{lem:Gauss-mass} gives
$\max_\chi|\Gamma_\chi(s;r)|\le\sqrt m/\varphi(m)=H_m/\sqrt m$ whenever
$r/(s,r)=m$. Hence
\[
 W_r\le\sum_{m\mid r}\frac{H_m}{\sqrt m}\cdot\frac mr\frac{H_r}{H_m}
 =\frac{H_r}r\sum_{m\mid r}\sqrt m
 =\frac{H_rV_r}{\sqrt r}.
\]
Finally, the Euler factors of $H_r^aV_r^b$ are at most
$(1-p^{-1})^{-a}(1-p^{-1/2})^{-b}$. For all sufficiently large $p$
these are at most $p^\nu$, and the remaining primes contribute a
constant independent of their exponents. This proves
\eqref{eq:subpower}.

\label{lem:coarse-grid}
Let $c_n$ be supported on $1\le n\le U$, with $E=\sum_n|c_n|^2$, and
let $F_y$ be as in \eqref{eq:char-expansion}. Let $P$ be an orthogonal
projection on the space of functions of the reduced residues modulo
$r$, and put $R_y=(1-P)F_y$. For a real interval $I$ and a function
$\Psi(b,\theta)$, write
\[
 \|\Psi\|_{I,r}^2=\frac1{\varphi(r)}\sum_{b\bmod r}^{*}
 \int_I|\Psi(b,\theta)|^2\,d\theta.
\]
Let $0=y_0<y_1<\cdots<y_G=W$ be integers with $W\ge U$, and put
$\Delta=\max_j(y_{j+1}-y_j)$. Then
\begin{equation}\label{eq:coarse-grid-transfer}
 \big\|\max_{y\le U}|F_y|\big\|_{I,r}^2
 \le3\big\|\max_{0\le j\le G}|R_{y_j}|\big\|_{I,r}^2
 +3\big\|\max_{y\le W}|PF_y|\big\|_{I,r}^2
 +C\frac{|I|(\Delta+r)}{\varphi(r)}E,
\end{equation}
where $C$ is an absolute constant.

<!-- proof-11: body_md -->
If $y_j\le y\le y_{j+1}$, then
\[
 F_y=R_{y_j}+PF_{y_j}+(F_y-F_{y_j}).
\]
Let $M_j(b,\theta)$ be the maximum over $y_j\le y\le y_{j+1}$ of
$|F_y(b,\theta)-F_{y_j}(b,\theta)|$. Then, for every $b$ and $\theta$,
\[
 \max_{y\le U}|F_y|
 \le\max_j|R_{y_j}|+\max_{y\le W}|PF_y|+\max_jM_j.
\]
The square of the right side is at most three times the sum of the
three squares, and $\max_jM_j^2\le\sum_jM_j^2$. For fixed $\theta$,

 \ll\frac{\Delta+r}{\varphi(r)}\sum_j\sum_{y_j<n\le y_{j+1}}|c_n|^2
 =\frac{\Delta+r}{\varphi(r)}E.
\]
Integrating over $I$ proves \eqref{eq:coarse-grid-transfer}.

<!-- lemma-9: body_md -->
\label{lem:repulsion}
Let $g$ be completely multiplicative with $|g(n)|\le1$, let $K\ge1$ be
fixed and let $J\ge1$ be an integer. Let $X$ be sufficiently large,
$r\log X\le y\le\sqrt X$, and let $\chi_1,\ldots,\chi_J$ be distinct
characters modulo $r$. Put
\[
 \mathcal G_\chi(s;y)
 =\prod_{p>y}\big(1-g(p)\overline{\chi(p)}p^{-s}\big)^{-1}.
\]
Then, for all real $\delta_j\ge0$ and $t_j$ with $|t_j|\le(\log X)^K$,
\begin{equation}\label{eq:repulsion}
 \min_{1\le j\le J}
 \log\big|\mathcal G_{\chi_j}(1+1/\log X+\delta_j+it_j;y)\big|
 \le\frac1{\sqrt J}\log\frac{\log X}{\log y}+O_K(\sqrt J).
\end{equation}

<!-- proof-14: body_md -->
Write $p(n)$ for the least prime factor of an integer $n>1$, put
$\sigma_0=1+1/\log X$ and
\[
 Z=\sum_{p(n)>y}\frac{\Lambda(n)}{n^{\sigma_0}\log n}
 =\log\frac{\log X}{\log y}+O(1).
\]
The sum is $\log\zeta(\sigma_0)=\log\log X+O(1)$ less the contribution
of the primes $p\le y$, which is $\log\log y+O(1)$ by Mertens' theorem
\cite[Theorem~2.7(d)]{MV2},
since $0\le p^{-k}-p^{-k\sigma_0}\le kp^{-k}(\log p)/\log X$ for
$k\ge1$, $\sum_{p\le y}(\log p)/p\ll\log y$ by \cite[(2.2)]{Robles},
and $y\le\sqrt X$. Since $g$ is completely multiplicative,
$\log\mathcal G_\chi(s;y)$ is the sum of
$\Lambda(n)g(n)\overline{\chi(n)}/(n^s\log n)$ over $p(n)>y$ when
$\Re s>1$. Put $s_j=\sigma_0+\delta_j+it_j$. Taking real parts, using
$|g(n)|\le1$ and applying the Cauchy-Schwarz inequality, we obtain
\[
 \sum_{j=1}^J\log|\mathcal G_{\chi_j}(s_j;y)|
 \le\bigg(Z\sum_{p(n)>y}\frac{\Lambda(n)}{n^{\sigma_0}\log n}
 \bigg|\sum_{j=1}^J\overline{\chi_j(n)}n^{-\delta_j-it_j}\bigg|^2
 \bigg)^{1/2}.
\]
We expand the square. The terms with $j=k$ contribute at most $JZ$ to
the sum over $n$. For $j\ne k$, the term $(j,k)$ contributes
\[
 \log L_y\big(\sigma_0+\delta_j+\delta_k+i(t_j-t_k),
 \overline{\chi_j}\chi_k\big),
\]
whose real part is the logarithm of the modulus. The character
$\overline{\chi_j}\chi_k$ is nonprincipal and
$|t_j-t_k|\le2(\log X)^K\le(\log X)^{K+1}$, so
Proposition~\ref{prop:GHS-short-input}(ii), with $K+1$ in place of $K$,
shows that this real part is at most $O_K(1)$. The sum over $n$ is
real, and is therefore at most $JZ+O_K(J^2)$. Hence
\[
 \sum_{j=1}^J\log|\mathcal G_{\chi_j}(s_j;y)|
 \le\sqrt{Z\big(JZ+O_K(J^2)\big)}\le\sqrt J\,Z+O_K(J^{3/2}).
\]
The minimum does not exceed the mean, and \eqref{eq:repulsion} follows
on dividing by $J$.

<!-- lemma-10: body_md -->
\label{lem:short-characters}
Let $K\ge1$ be fixed, let $f\in\mathcal{B}_A$, let $U$ be sufficiently large in
terms of $A$ and $K$, and let $r\le\ell^4$. There is a set $\mathcal{E}$ of at
most $4095$ characters modulo $r$ such that, for every
$\chi\notin\mathcal{E}$,
\begin{equation}\label{eq:short-character}
 \bigg|\sum_{x<n\le x+z}f(n)\overline{\chi(n)}\bigg|
 \ll_{A,K}z\ell^{-31/32}
\end{equation}
for all $x$ and $z$ with $U^{1/2}\le x\le U$ and $x/\ell^K\le z\le x$.
<!-- end lemma-10 -->
<!-- proof-15: body_md -->
Let $g$ and $h$ be as in Lemma~\ref{lem:companion}. Put $X=4U$ and
$y=r\log X$, so that $y\le\sqrt X$ and $\log\ell\le\log y\ll\log\ell$
for large $U$. For a constant $C_K$, let $\mathcal{E}$ be the set of
characters $\chi$ modulo $r$ for which
\[
 \sup_{\substack{\sigma\ge1+1/\log X\\|t|\le(\log X)^{K+3}}}
 |\mathcal G_\chi(\sigma+it;y)|
 >C_K\Big(\frac{\log X}{\log y}\Big)^{1/64}.
\]
If $\mathcal{E}$ contained $4096$ distinct characters $\chi_j$, there would
be for each of them a point $\sigma_j+it_j$ in the stated range at which
$|\mathcal G_{\chi_j}(\sigma_j+it_j;y)|$ exceeds the right side. This
contradicts Lemma~\ref{lem:repulsion}, applied with $J=4096$ and with
$K+3$ in place of $K$, once $\log C_K$ is larger than the term
$O_K(\sqrt J)$ in \eqref{eq:repulsion}. With this choice of $C_K$, the
set $\mathcal{E}$ has at most $4095$ elements. It depends only on $g$, $r$,
$U$ and $K$.

Write $G_\chi(s)=\sum_{n\ge1}g(n)\overline{\chi(n)}n^{-s}$. This is the
product of $\mathcal G_\chi(s;y)$ and the Euler factors at the primes
$p\le y$, and for $\Re s>1$ these factors have a product of modulus at
most $\prod_{p\le y}(1-p^{-1})^{-1}\ll\log y\ll\log\ell$. If
$U^{1/4}\le x\le2U$, then $1+1/\log x\ge1+1/\log X$. Hence, for
$\chi\notin\mathcal{E}$, $U^{1/4}\le x\le2U$ and $|t|\le(\log X)^{K+3}$,
\begin{equation}\label{eq:G-nonexceptional}
 |G_\chi(1+1/\log x+it)|\ll_K\ell^{1/64}(\log\ell)^{63/64}.
\end{equation}

Now let $\chi\notin\mathcal{E}$, $U^{1/4}\le x\le2U$ and $x/\ell^K\le z<x$.
We apply Proposition~\ref{prop:GHS-short-input}(i) to
$u=g\overline\chi\in\mathcal C(1)$, with $\delta=\log(x/z)/\log x$. For large $U$ we
have $0<\delta<1/4$, because $\log(x/z)\le K\log\ell$ and
$\log x\ge\ell/4$, and
$(x/z)\log x\le\ell^K\log(2U)\le(\log X)^{K+3}$. By \eqref{eq:short-M}
and \eqref{eq:G-nonexceptional}, we have $e^{-M}\le b_U$, where
$b_U=C_K'\ell^{-63/64}(\log\ell)^{63/64}$ for a suitable constant
$C_K'\ge1$. For sufficiently large $U$ we have $b_U<1$, so that
$M\ge\log(1/b_U)>0$. Since $(1+m)e^{-m}$ is decreasing for $m\ge0$,
\[
 (1+M)e^{-M}\le\big(1+\log(1/b_U)\big)b_U
 \ll_K\ell^{-63/64}(\log\ell)^2.
\]
The second term on the right of \eqref{eq:short-halasz} is
$\ll_K(\log\ell)/\ell$. Therefore
\begin{equation}\label{eq:g-short}
 \bigg|\sum_{x<n\le x+z}g(n)\overline{\chi(n)}\bigg|
 \ll_Kz\ell^{-31/32}\qquad(\chi\notin\mathcal{E})
\end{equation}
for $U^{1/4}\le x\le2U$ and $x/\ell^K\le z<x$. If $z=x$ and
$U^{1/4}\le x\le U$, we divide $(x,2x]$ into the intervals $(x,3x/2]$
and $(3x/2,2x]$. Their left endpoints are in $[U^{1/4},2U]$, and their
length $x/2$ is smaller than the left endpoint and at least $\ell^{-K}$
times it. Thus \eqref{eq:g-short} also holds for $z=x$ when
$U^{1/4}\le x\le U$.

Finally let $U^{1/2}\le x\le U$ and $x/\ell^K\le z\le x$. Since
$f=g*h$,
\[
 \sum_{x<n\le x+z}f(n)\overline{\chi(n)}
 =\sum_{d\le x+z}h(d)\overline{\chi(d)}
 \sum_{x/d<m\le(x+z)/d}g(m)\overline{\chi(m)}.
\]
For $d\le U^{1/8}$ we have $U^{3/8}\le x/d\le U$, and the ratio of the
length $z/d$ to the left endpoint $x/d$ is $z/x$. By \eqref{eq:g-short}
and \eqref{eq:h-norm} with $\sigma=1$, these $d$ contribute
$\ll_{A,K}z\ell^{-31/32}$. For $d>U^{1/8}$ the inner sum has at most
$z/d+1$ terms, and \eqref{eq:h-tail} gives
\[
 \sum_{U^{1/8}<d\le2x}|h(d)|\Big(\frac zd+1\Big)
 \ll_AzU^{-1/32}+(2x)^{3/4}.
\]
Since $z\ge x/\ell^K$ and $x\ge U^{1/2}$, both terms are
$\ll_Kz\ell^{-31/32}$. This proves \eqref{eq:short-character}.

\label{cor:short-dilated}
Let $f\in\mathcal{B}_A$, let $U$ be sufficiently large in terms of $A$, and let

There is an absolute constant $U_0$ such that, if $f\in\mathcal{B}_A$,
$U\ge U_0$, $r\le(\log U)^{15/8}$ and $1\le T\le(\log U)^8$, then

<!-- proof-17: body_md -->
By \eqref{eq:energy-A-transfer}, we may assume $f\in\mathcal B_1$. All
constants and all lower bounds for $U$ in this proof are then absolute.
Put $I=[-T/U,T/U]$ and $h_0=U/(2T)$, and let $\mathcal{E}$ be the set of
Corollary~\ref{cor:short-dilated}.

We first remove the terms with $n\le U^{3/4}$. Let $c_n=f(n)$ for
$U^{3/4}<n\le U$ and $c_n=0$ otherwise. Then
$M_f(U;\alpha)\le M_{f-c}(U;\alpha)+M_c(U;\alpha)$ for every real
$\alpha$, and the coefficients $f(n)-c_n$ with $n\le U$ are supported
on $n\le U^{3/4}$. Since $|I|<1$, the estimate \eqref{eq:CH} gives
$\int_IM_{f-c}(U;b/r+\theta)^2\,d\theta\ll U^{3/4}$ for every $b$, and
this is $\ll U/r$. In the rest of the proof $F_y$ is formed with the
coefficients $c_n$, and $R_y=(1-P_\mathcal{E})F_y$. It remains to show that
\begin{equation}\label{eq:large-goal}
 \big\|\max_{y\le U}|F_y|\big\|_{I,r}^2\ll\frac Ur,
\end{equation}
in the notation of Lemma~\ref{lem:coarse-grid}.

Let $a_n=(1-P_\mathcal{E})\big[c_n\operatorname{e}(nb/r)\big]$, a function of the reduced
residues $b$ modulo $r$, so that
$R_y(\cdot,\theta)=\sum_{n\le y}a_n\operatorname{e}(n\theta)$. We claim that
\begin{equation}\label{eq:window-vector}
 \bigg\|\sum_{v<n\le v'}a_n\bigg\|_{2,r}\ll h_0\varepsilon_U H_rV_r
 \qquad(U^{3/4}\le v\le v'\le U,\ v'-v\le h_0).
\end{equation}
By Parseval's identity for \eqref{eq:char-expansion} and by
\eqref{eq:smooth-expansion}, the square of the left side is
\[
 \sum_{\chi\notin\mathcal{E}}\bigg|\sum_{s\in\mathcal{S}_r}f(s)\Gamma_\chi(s;r)
 \sum_{v/s<m\le v'/s}f(m)\overline{\chi(m)}\bigg|^2.
\]
For $s\le U^{1/8}$ the inner sum is
$\ll(h_0\varepsilon_U+U\ell^{-16})/s$ by \eqref{eq:short-dilated}. Since there
are at most $r$ characters, \eqref{eq:W} shows that the terms with
$s\le U^{1/8}$ contribute to the norm at most
\[
 C\sqrt r\,W_r\big(h_0\varepsilon_U+U\ell^{-16}\big)
 \le CH_rV_r\big(h_0\varepsilon_U+U\ell^{-16}\big).
\]
For $s>U^{1/8}$ the inner sum has at most $h_0/s+1$ terms and
$|\Gamma_\chi(s;r)|\le1$, so by \eqref{eq:smooth-tail} these $s$
contribute at most $r^{5/2}(h_0U^{-1/16}+U^{1/2})$. Now $T\le\ell^8$ and
$r\le\ell^{15/8}$ give $h_0\varepsilon_U\ge U\ell^{-9}/2$ and
$r^{5/2}\le\ell^5$. Hence $U\ell^{-16}$, $r^{5/2}h_0U^{-1/16}$ and
$r^{5/2}U^{1/2}$ are all $\ll h_0\varepsilon_U$. As $H_rV_r\ge1$, this proves
\eqref{eq:window-vector}.

Next let $U^{3/4}\le v\le v'\le U$, with no condition on $v'-v$, and
apply Lemma~\ref{lem:Gallagher} with $\delta=T/U$ to the functions
$a_n$ with $v<n\le v'$. For every real $t$, the integers $n$ with
$t<n\le t+h_0$ and $v<n\le v'$ are those of an interval to which
\eqref{eq:window-vector} applies, and there are none unless
$v-h_0<t<v'$. Therefore
\begin{equation}\label{eq:interval-energy}
 \int_I\bigg\|\sum_{v<n\le v'}a_n\operatorname{e}(n\theta)\bigg\|_{2,r}^2\,d\theta
 \ll(v'-v+h_0)\varepsilon_U^2H_r^2V_r^2.
\end{equation}

Let $W$ be the least power of $2$ with $W\ge U$, and let $G$ be the
power of $2$ with $H_rT\le G<2H_rT$. For sufficiently large $U$ we have
$G\le W$, so that $\Delta=W/G$ is an integer; put $y_j=j\Delta$ for
$0\le j\le G$. For $0\le k\le\log_2G$, let $\mathcal D_k$ be the set of
the $2^k$ intervals $((i-1)W/2^k,iW/2^k]$ with $1\le i\le2^k$. By the
binary expansion of $j/G$, each interval $(0,y_j]$ with $1\le j\le G$
is a disjoint union of intervals taken from the sets $\mathcal D_k$, at
most one from each. Hence, for all $b$ and $\theta$,
\[
 \max_{0\le j\le G}|R_{y_j}(b,\theta)|
 \le\sum_{k=0}^{\log_2G}\bigg(\sum_{D\in\mathcal D_k}
 \bigg|\sum_{n\in D}a_n(b)\operatorname{e}(n\theta)\bigg|^2\bigg)^{1/2},
\]
and Minkowski's inequality for the norm $\|\cdot\|_{I,r}$ gives
\begin{equation}\label{eq:dyadic-grid-maximal}
 \big\|\max_{0\le j\le G}|R_{y_j}|\big\|_{I,r}
 \le\sum_{k=0}^{\log_2G}\bigg(\sum_{D\in\mathcal D_k}\int_I
 \bigg\|\sum_{n\in D}a_n\operatorname{e}(n\theta)\bigg\|_{2,r}^2\,d\theta\bigg)^{1/2}.
\end{equation}
Since $a_n=0$ unless $U^{3/4}<n\le U$, the estimate
\eqref{eq:interval-energy} applies to the intersection of each $D$ with
$(U^{3/4},U]$. For fixed $k$ these intersections have total length at
most $U$, and there are at most $G$ of them. As $Gh_0<H_rU$, the sum
over $D\in\mathcal D_k$ in \eqref{eq:dyadic-grid-maximal} is
\[
 \ll(U+Gh_0)\varepsilon_U^2H_r^2V_r^2\ll U\varepsilon_U^2H_r^3V_r^2.
\]
The number of $k$ is $1+\log_2G\ll\log\ell$. Hence
\begin{equation}\label{eq:large-grid}
 \big\|\max_{0\le j\le G}|R_{y_j}|\big\|_{I,r}^2
 \ll U\varepsilon_U^2H_r^3V_r^2(\log\ell)^2\ll\frac Ur.
\end{equation}
For the last step, \eqref{eq:subpower} with $\nu=1/128$ and
$r\le\ell^{15/8}$ give
\[
 rH_r^3V_r^2\varepsilon_U^2(\log\ell)^2
 \ll r^{129/128}\ell^{-31/16}(\log\ell)^2
 \le\ell^{-49/1024}(\log\ell)^2.
\]

Since $|I|=2T/U\le1/r$ and $c_n=0$ for $n>U$, the estimate
\eqref{eq:finite-rank-energy}, with $J\le4095$, gives
$\|\max_{y\le W}|P_\mathcal{E} F_y|\|_{I,r}^2\ll U/r$. We now apply
Lemma~\ref{lem:coarse-grid} to the coefficients $c_n$, with $P=P_\mathcal{E}$
and the points $y_j$. The first two terms on the right of
\eqref{eq:coarse-grid-transfer} have just been bounded. Since
$\sum_n|c_n|^2\le U$ and $\Delta<2U/(H_rT)$, the third term is
\[
 \ll\frac{T(\Delta+r)}{\varphi(r)}\ll\frac Ur+TH_r\ll\frac Ur.
\]
This proves \eqref{eq:large-goal}, and with it \eqref{eq:energy-large}
for $f\in\mathcal B_1$.

\label{lem:new-block}
Assume the hypotheses of Proposition~\ref{prop:old-blocks}(ii), with
$f\in\mathcal{B}_A$, and let $P$ be dyadic with $Y\le P<N$ and
$U=\lfloor N/P\rfloor$. If the hypotheses of
Proposition~\ref{prop:energy-small} or of

\begin{equation}\label{eq:new-block}

\end{equation}

Let $Q$ be the left side of \eqref{eq:positive-transfer} with
$c_n=f(n)$, so that $d_n=nf(n)$. For every integer $k\le U$ and every
real $\xi$, partial summation gives
\[
 \sum_{n\le k}nf(n)\operatorname{e}(n\xi)=k\sum_{n\le k}f(n)\operatorname{e}(n\xi)
 -\sum_{m<k}\sum_{n\le m}f(n)\operatorname{e}(n\xi),
\]
and therefore
\begin{equation}\label{eq:weighted-max}
 M_d(U;\xi)\le2UM_f(U;\xi).
\end{equation}
Since $h|\beta|U\le1$, it follows from \eqref{eq:positive-transfer}

\begin{equation}\label{eq:Q-transfer}

\end{equation}
We show that
\begin{equation}\label{eq:Q-bound}
 Q\ll_A\frac{NU}{rB}.
\end{equation}

Suppose first that $B>1$, so that $|\beta|=B/N$. For $P\le t\le2P$ we

multiplication by $a$ permutes the reduced residues modulo $r$. The
substitution $\theta=\beta t$ in \eqref{eq:Q-transfer} therefore gives
\[

\]

\[

 \ll_AP\frac{NU}{rB}\le\frac{N^2}{rB}.
\]
This proves \eqref{eq:new-block}.

<!-- corollary-3: body_md -->
\label{cor:log-weights}
Let $f\in\mathcal{B}_A$, $(a,q)=1$, $1\le q\le N$ and $|\alpha-a/q|\le q^{-2}$.
Then
\begin{equation}\label{eq:log-weights}
 \bigg|\sum_{n\le N}\frac{f(n)}n\operatorname{e}(n\alpha)\bigg|
 \ll A\bigg(\log(2q)+\frac L{\sqrt q}\bigg).
\end{equation}
<!-- end corollary-3 -->
<!-- proof-23: body_md -->
Since $|f(n)|\le A$, the sum of $|f(n)|/n$ over $n\le x$ is at most
$A(1+\log x)$ for every $x\ge1$. This gives \eqref{eq:log-weights} when
$q=1$, and also when $q^2\ge N$, since then $L\le2\log q$. Let
$2\le q<\sqrt N$. The terms with $n\le q^2$ contribute
$\ll A\log(2q)$. For $q^2\le t\le N$ we have $2\le q\le t/q$, and
\eqref{eq:global}, with $t$ in place of $N$ and with $R=q$, gives
\[
 |S_f(t,\alpha)|\ll A\bigg(\frac t{\log t}+\frac t{\sqrt q}\bigg).
\]
By partial summation,
\begin{align*}
 \bigg|\sum_{q^2<n\le N}\frac{f(n)}n\operatorname{e}(n\alpha)\bigg|
 &\le\frac{|S_f(N,\alpha)|}N+\frac{|S_f(q^2,\alpha)|}{q^2}
 +\int_{q^2}^N\frac{|S_f(t,\alpha)|}{t^2}\,dt\\
 &\ll A\bigg(1+\log(2L)+\frac L{\sqrt q}\bigg).
\end{align*}
If $q\ge L$, then $\log(2L)\le\log(2q)$. If $q<L$, then
$L/\sqrt q>\sqrt L\gg\log(2L)$. This proves \eqref{eq:log-weights}.

<!-- proposition-7: body_md -->
\label{prop:stationary}
Let $N\ge3$ be an integer, $B\ge2$ and $f_B(n)=n^{-i\pi B}$. There is
a real number $\vartheta=\vartheta(N,B)$ such that
\begin{equation}\label{eq:stationary}
 S_{f_B}(N,B/N)=\frac N{\sqrt{2B}}\,e^{i\vartheta}
 +O\Big(\frac NB+BL+1\Big).
\end{equation}
In particular, if $B\to\infty$ and $B^{3/2}L=o(N)$, then
$|S_{f_B}(N,B/N)|\sim N/\sqrt{2B}$.
<!-- end proposition-7 -->

For $B=L$ this is \cite[(8.2)]{Robles}, with
$\vartheta=\pi B(1+\log2)-\pi B\log N+\pi/4$. The proof given there
applies to every $B\ge2$. It compares $S_{f_B}(N,B/N)$ with the
integral of $\exp(i(2\pi Bx/N-\pi B\log x))$ over $1\le x\le N$, with
an error $O(1+BL)$, and it evaluates this integral, after the
substitution $x=Nu$, by the method of stationary phase at $u=1/2$,
with an error $O(N/B)$. The value $B=L$ is used in \cite{Robles} only
to verify the hypotheses of \cite[Theorem~1.1]{Robles} and to compare
the error term with the main term.

Let $y\ge3$, and put
\[
 q=\prod_{3\le p\le y}p,\qquad\delta=\frac1{\log(2H_q)}.
\]
Then $H_q\ge3/2$ and $0<\delta<1$. Let $\chi$ be a primitive character
modulo $q$, for example the product of the Legendre symbols modulo the
primes dividing $q$. For $z>y$, let $\mathcal{P}$ be the set of primes $p$ with
$y<p\le z$, and put
\begin{equation}\label{eq:counter-f}
 f_q(n)=\chi(n)(1+\delta)^{\omega_{\mathcal{P}}(n)/2},\qquad
 \Pi_\delta=\prod_{p\in\mathcal{P}}\Big(1+\frac\delta p\Big),
\end{equation}
where $\omega_{\mathcal{P}}(n)$ is the number of primes of $\mathcal{P}$ dividing $n$.
The function $f_q$ is multiplicative, and for $k\ge1$ we have
$|f_q(p^k)|=0$ if $p\mid q$, $|f_q(p^k)|=\sqrt{1+\delta}$ if
$p\in\mathcal{P}$, and $|f_q(p^k)|=1$ otherwise. Since $H_q\to\infty$ as
$y\to\infty$, we have $\sqrt{1+\delta}\le A$ when $y$ is large in terms
of $A$. The parameters $y$ and $z$ are chosen below.

We first bound the mean square. Let $u(n)=|f_q(n)|^2$, so that
$u(n)=(1+\delta)^{\omega_{\mathcal{P}}(n)}$ if $(n,q)=1$ and $u(n)=0$
otherwise. For $p\in\mathcal{P}$,
\[
 \sum_{k\ge0}\frac{u(p^k)}{p^{ks}}=\frac{1+\delta p^{-s}}{1-p^{-s}},
 \qquad\text{so that}\qquad
 \Lambda_u(p^k)=\big(1+(-1)^{k-1}\delta^k\big)\log p.
\]
Moreover $\Lambda_u(p^k)=0$ if $p\mid q$, and $\Lambda_u(p^k)=\log p$
for the remaining primes. Hence $|\Lambda_u(n)|\le2\Lambda(n)$, that
is, $u\in\mathcal C(2)$. By \eqref{eq:Shiu} there are absolute
constants $x_0$ and $C$ such that
\[
 \sum_{n\le x}u(n)\le C\frac x{\log x}
 \exp\bigg(\sum_{p\le x}\frac{u(p)}p\bigg)\qquad(x\ge x_0).
\]
Let $y\ge x_0$ and $x>y$. Since $u(p)=0$ for $p\mid q$,
$u(p)=1+\delta$ for $p\in\mathcal{P}$, and $u(p)=1$ for the other primes,
\[
 \sum_{p\le x}\frac{u(p)}p\le\sum_{p\le x}\frac1p-\sum_{p\mid q}\frac1p
 +\delta\sum_{p\in\mathcal{P}}\frac1p.
\]
Here $\sum_{p\mid q}1/p=\log H_q+O(1)$ and
$\delta\sum_{p\in\mathcal{P}}1/p=\log\Pi_\delta+O(1)$, because
$\sum_pp^{-2}<\infty$ and $\delta\le1$, and
$\sum_{p\le x}1/p=\log\log x+O(1)$ by Mertens' theorem
\cite[Theorem~2.7(d)]{MV2}. Therefore
\begin{equation}\label{eq:counter-mean}
 \sum_{n\le x}|f_q(n)|^2\le C_1x\frac{\Pi_\delta}{H_q}\qquad(x>y),
\end{equation}
where $C_1\ge1$ is an absolute constant; it does not depend on $y$,
$z$ or $A$. For $1\le x\le y$, the integers $n\le x$ coprime to $q$
are powers of $2$. They have no prime factor in $\mathcal{P}$, and there are at
most $x$ of them, so that $\sum_{n\le x}|f_q(n)|^2\le x$.

We now choose $y$ so large that $y\ge x_0$, $\sqrt{1+\delta}\le A$ and
$H_q>2C_1$. Since $\sum_p1/p$ diverges, there is a least prime $z>y$
for which $\Pi_\delta\ge H_q/(2C_1)$. As $H_q/(2C_1)>1$ and each factor
of $\Pi_\delta$ is at most $2$, this $z$ satisfies
\begin{equation}\label{eq:product-stop}
 \frac{H_q}{2C_1}\le\Pi_\delta\le\frac{H_q}{C_1}.
\end{equation}
The bound \eqref{eq:counter-mean} and the case $x\le y$ then give
\begin{equation}\label{eq:all-x-verified}
 \sum_{n\le x}|f_q(n)|^2\le x\qquad\text{for every real }x\ge1.
\end{equation}
This proves \eqref{eq:counter-summary}, and $f_q\in\mathcal{F}_A$.

It remains to prove \eqref{eq:counter-order}. Put
\[
 \eta=\sqrt{1+\delta}-1,\qquad q_1=q\prod_{p\in\mathcal{P}}p,\qquad
 \Pi_\eta=\prod_{p\in\mathcal{P}}\Big(1+\frac\eta p\Big).
\]
The function $f_q(n)\operatorname{e}(an/q)$ has period $q_1$, and the weight
$(1+\delta)^{\omega_{\mathcal{P}}(n)/2}$ is the product over $p\in\mathcal{P}$ of
factors equal to $\sqrt{1+\delta}$ if $p\mid n$ and to $1$ otherwise.
By the Chinese remainder theorem, the sum of $f_q(n)\operatorname{e}(an/q)$ over a
period is therefore the product of
$\sum_{m\bmod q}\chi(m)\operatorname{e}(am/q)=\overline{\chi(a)}\tau(\chi)$ and of
the numbers $p-1+\sqrt{1+\delta}=p(1+\eta/p)$ for $p\in\mathcal{P}$. Since
$|\tau(\chi)|=\sqrt q$ by \eqref{eq:induced-Gauss}, we obtain, when $N$
is a multiple of $q_1$,
\begin{equation}\label{eq:CRT-exact}
 S_{f_q}(N,a/q)=\frac Nq\,\overline{\chi(a)}\tau(\chi)\,\Pi_\eta,
 \qquad|S_{f_q}(N,a/q)|=\frac N{\sqrt q}\,\Pi_\eta.
\end{equation}
We compare $\Pi_\eta$ with $\Pi_\delta$. Since $\delta=2\eta+\eta^2$,
we have $(1+\eta/p)^2\le1+\delta/p$, and so
$\Pi_\eta\le\Pi_\delta^{1/2}$. Put $\gamma=\eta/\delta
=1/(1+\sqrt{1+\delta})$. The concavity of the logarithm gives
$\log(1+\gamma\delta/p)\ge\gamma\log(1+\delta/p)$, and so
$\Pi_\eta\ge\Pi_\delta^\gamma$. Now
\[
 0\le\frac12-\gamma=\frac\delta{2(1+\sqrt{1+\delta})^2}\le\frac\delta8,
 \qquad0<\delta\log\Pi_\delta\le\delta\log(2H_q)=1,
\]
by \eqref{eq:product-stop}. Consequently
\begin{equation}\label{eq:product-comparison}
 e^{-1/8}\,\Pi_\delta^{1/2}\le\Pi_\eta\le\Pi_\delta^{1/2}.
\end{equation}
By \eqref{eq:product-stop}, \eqref{eq:CRT-exact},
\eqref{eq:product-comparison} and $H_q/q=1/\varphi(q)$, we obtain for
the multiples $N$ of $q_1$
\[
 \frac{e^{-1/8}}{\sqrt{2C_1}}\,\frac N{\sqrt{\varphi(q)}}
 \le|S_{f_q}(N,a/q)|\le\frac1{\sqrt{C_1}}\,\frac N{\sqrt{\varphi(q)}}.
\]

For arbitrary real $N\ge1$, write $N=kq_1+s$ with an integer $k\ge0$
and $0\le s<q_1$, and put
$\mu_a=\overline{\chi(a)}\tau(\chi)\Pi_\eta/q$. By periodicity,
$S_{f_q}(N,a/q)=kq_1\mu_a+S_{f_q}(s,a/q)$. The Cauchy-Schwarz
inequality and \eqref{eq:all-x-verified} give
$|S_{f_q}(s,a/q)|\le s$, and $|\mu_a|\le1$ by the upper bound just
proved. Hence
\begin{equation}\label{eq:counter-all-lengths}
 S_{f_q}(N,a/q)=N\mu_a+O(q_1),\qquad
 \frac{e^{-1/8}}{\sqrt{2C_1\varphi(q)}}\le|\mu_a|
 \le\frac1{\sqrt{C_1\varphi(q)}},
\end{equation}
where the error term has modulus at most $2q_1$. It is at most
$N|\mu_a|/2$ as soon as
$N\ge4e^{1/8}\sqrt{2C_1}\,q_1\sqrt{\varphi(q)}$, and for these $N$
\eqref{eq:counter-order} holds, with absolute constants and for every
$a$ coprime to $q$.

Finally, $H_q\asymp\log\log q$. Indeed,
\[
 H_q=\prod_{3\le p\le y}\Big(1-\frac1p\Big)^{-1}
 \ge\sum_{\substack{n\le y\\n\textup{ odd}}}\frac1n\gg\log y,
\]
while $\log q\le y\log y$ gives $\log\log q\ll\log y$; and
$H_q\ll\log\log(3q)$ by \cite[(2.3)]{Robles}. If $N$ is such that
$L\ge q$, then $q\le N/L^2$ and $N/L\le N/q$, as in
Section~\ref{sec:q-sharpness}.
