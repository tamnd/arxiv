    For a representation $\pi$ of the form \eqref{matrix}, $\mathfrak{H}_j$ has no $G$-invariant complement in $\mathfrak{H}_{j+1}$ for each $j = 1, \cdots, n-1$ if and only if the following conditions hold
    \begin{enumerate}

        \begin{align}
            c_{k, k+1} \notin B^1(\pi_{k+1}, \pi_k)\,, \quad\mbox{for}\quad k = 1, \cdots, n-1 \,.
        \end{align}}

        \item{
        \begin{align}
            \delta c_{kl} = -\sum_{j=k+1}^{l-1} c_{kj} \times c_{jl}\,, \quad\mbox{for}\quad 1 \leqslant k < l +1 \leqslant n+1 \,.
        \end{align}}        
    \end{enumerate}

\textbf{Step I.} Let $P_1$, $P_2 \overset{\scalebox{1.6}[1.3]{$\curvearrowright$}}{\big(=}P_{2^\prime} + P_{{\overline{2}^\prime}})$, and $P_3$ denote the projections associated with the decomposition \eqref{Araki-direct-sum-bulk333}. Explicitly (see Table \ref{Table 1}),
\begin{align}
    P_1 &\colon\; V_{\mathrm{tot}} \; \longrightarrow\; V_g \,, \qquad P_1 \Phi = \sum_{L=0}^{p-1} \left\langle \Phi \,,\, \phi^{(s)}_L \right\rangle_{\vphantom{}_{\mathrm{KG}}} \;\phi^{(g)}_L \,, \\[3pt]
    P_2 \overset{\scalebox{1.6}[1.3]{$\curvearrowright$}}{\big(=}P_{2^\prime} + P_{{\overline{2}^\prime}}) &\colon\; V_{\mathrm{tot}} \; \longrightarrow\; V^\prime\oplus \overline{V^\prime} \,, \qquad P_2 \Phi = \underbrace{\sum_{L=p}^\infty \left\langle \Phi \,,\, \phi^{}_L \right\rangle_{\vphantom{}_{\mathrm{KG}}} \;\phi^{}_L}_{P^{}_{2^\prime} \colon\; V_{\mathrm{tot}} \; \longrightarrow\; V^\prime} \;+\;\, \underbrace{\sum_{L=p}^\infty \left\langle \Phi \,,\, -\overline{\phi^{}_L} \right\rangle_{\vphantom{}_{\mathrm{KG}}} \;\overline{\phi^{}_L}}_{P_{{\overline{2}^\prime}} \colon\; V_{\mathrm{tot}} \; \longrightarrow\; \overline{V^\prime}} \,, \\[5pt] 
    P_3 &\colon\; V_{\mathrm{tot}} \; \longrightarrow\; V_s \,, \qquad P_3 \Phi = \sum_{L=0}^{p-1} \left\langle \Phi \,,\, \phi^{(g)}_L \right\rangle_{\vphantom{}_{\mathrm{KG}}} \;\phi^{(s)}_L \,.
\end{align}
We recall that $\{\phi^{(g)}_{L<p}\}$ and $\{\phi^{(s)}_{L<p}\}$ are KG-paired bases of $V_g$ and $V_s$, respectively, while $\{\phi_{L\geqslant p}\}$ and $\{\overline{\phi_{L\geqslant p}}\}$ are KG-``orthonormal'' bases of $V^\prime$ and $\overline{V^\prime}$, respectively; see Table \ref{Table 1}. Moreover, the maps $P_i$, $i=1,2 \overset{\scalebox{1.6}[1.3]{$\curvearrowright$}}{\big(=}2^\prime,{\overline{2}^\prime}\big), 3$, satisfy
\begin{align}
    P_iP_j=\delta_{ij}P_i \,,
\end{align}
where $\delta_{ij}$ denotes the Kronecker delta.

\textbf{Step III.} We observe the following:
\begin{enumerate}
