<!-- theorem-4: body_md -->
\label{thm:main}
Let $(M^n,g,J)$ be a connected complete K\"ahler manifold of complex
dimension $n\ge2$ with strictly positive holomorphic bisectional
curvature. Suppose that there are a compact set $C\subset M$ and a
constant $\kappa>0$ such that, on $M\setminus C$, the real sectional
curvature is nonnegative and
\[
  \operatorname{Ric}_{\min}\ge\kappa\,\operatorname{Ric}_{\max}.
\]
Then $M$ is compact.

<!-- proposition-1: body_md -->
\label{prop:examples}
For every $n\ge2$ and $\delta>0$, there exists a complete
$\mathrm U(n)$-invariant K\"ahler metric on $\mathbb C^n$ with positive
bisectional curvature, Euclidean volume growth, and quadratic curvature
decay such that
\begin{equation}\label{eq:example-ratio}
  \frac{\operatorname{Ric}_{\min}(x)}{\operatorname{Ric}_{\max}(x)}
  \asymp\frac{1}{(\log r(x))^{1+\delta}}
\end{equation}
as $r(x)\to\infty$. Here $r$ is the distance from the origin, and
$\asymp$ means that the ratio of the two sides is bounded above and below
by positive constants.

\label{prop:top-mass-euclidean}
Let $M^n$ be a complete noncompact K\"ahler manifold with nonnegative
Ricci curvature and Euclidean volume growth. Suppose that either the
bisectional curvature is nonnegative, or the curvature has quadratic
decay. Then
\begin{equation}\label{eq:top-mass-euclidean}
  \int_M\rho^n<\infty.
\end{equation}

<!-- lemma-1: body_md -->
\label{lem:log-integral}
Let $(M^n,\omega)$, $n\ge2$, be a complete K\"ahler manifold with
$\rho\ge0$ and $\int_M\rho^n<\infty$. If \eqref{eq:log-pinching} holds
outside a compact set, then
\begin{equation}\label{eq:log-integral}
  \int_{B(p,R)}S^n\,dV=o(\log R).
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Let $0\le\lambda_1\le\cdots\le\lambda_n$ be the eigenvalues of $\rho$
with respect to $\omega$. At a point where $\lambda_1\ge\kappa\lambda_n$,
we have
\begin{equation}\label{eq:ratio-determinant}
  \prod_{j=1}^n\lambda_j
  \ge\kappa^{n-1}\lambda_n^n
  \ge\frac{\kappa^{n-1}}{n^n}S^n.
\end{equation}
Since $\rho^n=n!\bigl(\prod_j\lambda_j\bigr)dV$, condition
\eqref{eq:log-pinching} implies
\begin{equation}\label{eq:weighted-mass}
  S^n\,dV\le C\log(2+r)\,\rho^n
\end{equation}
outside a compact set. Fix $R_0$ sufficiently large that this set is
contained in $B(p,R_0)$. For $R>R_0$, we obtain
\begin{align*}
 \frac1{\log R}\int_{B(p,R)}S^n\,dV
 &\le \frac1{\log R}\int_{B(p,R_0)}S^n\,dV\\
 &\quad+C\frac{\log(2+R)}{\log R}
       \int_{M\setminus B(p,R_0)}\rho^n.
\end{align*}
First let $R\to\infty$, then let $R_0\to\infty$. The lemma follows from
the finiteness of the total mass.

<!-- lemma-2: body_md -->
\label{lem:euclidean-gap}
Let $M^n$, $n\ge2$, be a complete noncompact K\"ahler manifold with
nonnegative bisectional curvature and Euclidean volume growth. If
\eqref{eq:log-integral} holds, then $M$ is flat.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
By \cite[Theorems~2 and~3]{Liu24} and the existence of the asymptotic
volume ratio, the limit
\begin{equation}\label{eq:scalar-limit}
  \ell=\lim_{R\to\infty}R^{2-2n}
            \int_{B(p,R)}S\,dV
\end{equation}
exists and is finite. Set
$A_j=B(p,2^{j+1})\setminus B(p,2^j)$. Condition
\eqref{eq:log-integral} gives
\[
  \sum_{j=0}^{N-1}\int_{A_j}S^n\,dV=o(N).
\]
Hence there is a sequence $j_k\to\infty$ such that
$\int_{A_{j_k}}S^n\,dV\to0$. Put $R_k=2^{j_k}$. By H\"older's
inequality and Bishop--Gromov volume comparison,
\begin{align*}
 R_k^{2-2n}\int_{B(p,2R_k)\setminus B(p,R_k)}S\,dV
 &\le R_k^{2-2n}\operatorname{Vol}(B(p,2R_k))^{(n-1)/n}
       \left(\int_{A_{j_k}}S^n\,dV\right)^{1/n}\\
 &\le C(n)\left(\int_{A_{j_k}}S^n\,dV\right)^{1/n}
       \longrightarrow0.
\end{align*}
On the other hand, \eqref{eq:scalar-limit} implies that the left hand
side converges to $(2^{2n-2}-1)\ell$. Since $n\ge2$, we obtain
$\ell=0$. Euclidean volume growth now gives
\[
  \lim_{R\to\infty}
    \frac{R^2}{\operatorname{Vol}(B(p,R))}\int_{B(p,R)}S\,dV=0.
\]
By the second author's optimal gap theorem~\cite{Ni12}, $M$ is flat.
<!-- end proof-2 -->
<!-- proof-3: body_md -->
It suffices to prove Theorem~\ref{thm:logarithmic}, since a uniform
positive pinching constant implies \eqref{eq:log-pinching}.
Proposition~\ref{prop:top-mass-euclidean} and Lemma~\ref{lem:log-integral}
give \eqref{eq:log-integral} in both cases. If the bisectional curvature
is nonnegative, Lemma~\ref{lem:euclidean-gap} implies that $M$ is flat.
If the Ricci curvature is nonnegative and the curvature has quadratic
decay, \cite[Corollary~7]{Liu24} implies that $M$ is Ricci flat.

<!-- lemma-3: body_md -->
\label{lem:scalar}
Let $(M,g)$ be a connected complete noncompact Riemannian manifold with
$\operatorname{Ric}_g\ge0$, and let $2\le p<\infty$. For $f\in L^p(M)$,
\begin{equation}\label{eq:scalar-estimates}
 \left\lVert P_tf\right\rVert_{L^p}\le\left\lVert f\right\rVert_{L^p},
 \qquad
 \left\lVert \nabla P_tf\right\rVert_{L^p}
 \le C_p t^{-1/2}\left\lVert f\right\rVert_{L^p}
 \quad(t>0).
\end{equation}
Moreover,
\begin{equation}\label{eq:scalar-decay}
  \lim_{t\to\infty}\left\lVert P_tf\right\rVert_{L^p}=0.
\end{equation}
The same statements hold after a fixed positive rescaling of the scalar
Laplacian.
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
The heat semigroup is positivity preserving and contractive on $L^p$
(cf.\ Proposition~15.1 of \cite{Li12}).
For a bounded compactly supported smooth function $f$, the Bochner
inequality under $\operatorname{Ric}_g\ge0$ implies the reverse Poincar\'e estimate
\begin{equation}\label{eq:reverse-poincare}
  |\nabla P_tf|^2
  \le\frac{C}{t}\bigl(P_t(f^2)-(P_tf)^2\bigr)
  \le\frac{C}{t}P_t(f^2).
\end{equation}
One way to see the first inequality is to differentiate
\[
  s\longmapsto P_s\bigl((P_{t-s}f)^2\bigr),
  \qquad 0\le s\le t.
\]
Its derivative is a fixed positive multiple (depending on the definitions
of the Laplacian and gradient) of $P_s|\nabla P_{t-s}f|^2$. The
gradient-contraction inequality, which follows from the Bochner formula
and nonnegativity of Ricci curvature, bounds this from below by the same
multiple of $|\nabla P_tf|^2$. Integration in $s$ gives the first
estimate in \eqref{eq:reverse-poincare}.

Since $p/2\ge1$, Jensen's inequality and the $L^p$ contraction
mentioned above give
\[
 \begin{split}
  \int_M|\nabla P_tf|^p\,dV
  &\le C_p t^{-p/2}
       \int_M\bigl(P_t(f^2)\bigr)^{p/2}\,dV\\
  &\le C_p t^{-p/2}\left\lVert f\right\rVert_{L^p}^{p}.
 \end{split}
\]
Approximation by compactly supported smooth functions proves the gradient
estimate for all $f\in L^p$.

We next prove \eqref{eq:scalar-decay}. First, $M$ has infinite volume.
For an elementary verification, let $\gamma$ be a minimizing ray from $o$.
The balls
\[
  B\bigl(\gamma(4^j),4^j/2\bigr),\qquad j=1,2,\ldots,
\]
are pairwise disjoint. Bishop--Gromov comparison and
$B(o,1)\subset B(\gamma(r),r+1)$ show that each of these balls has volume
bounded below by a fixed positive multiple of $\operatorname{Vol} B(o,1)$. Their union
therefore has infinite volume.

The $L^2$ kernel of $-\Delta$ is zero. Indeed, if $h\in L^2$ is harmonic,
standard distance cutoffs $\chi_R$ give
\[
  \int_M\chi_R^2|\nabla h|^2\,dV
  \le \frac{C}{R^2}\left\lVert h\right\rVert_{L^2}^2.
\]
Letting $R\to\infty$ shows that $h$ is constant. Infinite volume then gives
$h=0$. The spectral theorem for the nonnegative self-adjoint realization
of $-\Delta$ consequently yields
\begin{equation}\label{eq:L2-decay}
  \left\lVert P_th\right\rVert_{L^2}\longrightarrow0
  \qquad(h\in L^2).
\end{equation}
The local parabolic mean-value estimate in Theorem~14.7 of
\cite{Li12} also gives
$\left\lVert P_th\right\rVert_{L^\infty(K)}\longrightarrow0$ for every compact
$K\subset M$. Only the $L^2$ convergence in \eqref{eq:L2-decay} is
needed below.

If $h$ is bounded and compactly supported, then
\[
  \left\lVert P_th\right\rVert_{L^p}^{p}
  \le \left\lVert h\right\rVert_\infty^{p-2}\left\lVert P_th\right\rVert_{L^2}^{2}
  \longrightarrow0.
\]
For $p=2$, this is simply \eqref{eq:L2-decay}. Density of bounded compactly
supported functions in $L^p$, together with $L^p$-contractivity, proves
\eqref{eq:scalar-decay}.

<!-- lemma-4: body_md -->
\label{lem:hodge}
Let $(M^n,\omega)$, $n\ge2$, be a connected complete noncompact K\"ahler
manifold with nonnegative Ricci curvature and nonnegative quadratic
orthogonal bisectional curvature. Let $\alpha$ be a smooth real
$(1,1)$-form such that
\[
  d\alpha=0,
  \qquad F:=|\alpha|\in L^n(M),
  \qquad f:=\Lambda_\omega\alpha.
\]
There is a smooth real Hodge heat evolution $\eta(x,t)$ with initial
value $\alpha$ for which
\begin{equation}\label{eq:heat-form}
  \eta_t+\Delta_{\bar\partial}\eta=0,
  \qquad d\eta(x,t)=0,
\end{equation}
\begin{equation}\label{eq:norm-domination}
  |\eta(x,t)|\le P_tF(x),
\end{equation}
and
\begin{equation}\label{eq:trace-heat}
  u(x,t):=\Lambda_\omega\eta(x,t)=P_tf(x).
\end{equation}
In particular,
\begin{equation}\label{eq:exact-evolution}
  \eta_t=dd^c u.
\end{equation}
The evolution converges locally to $\alpha$ as $t\downarrow0$.
No sign condition is imposed on $\alpha$ or asserted for $\eta(t)$.
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
For a fixed base point $o$, put
\[
  k_F(R)=\frac1{V(o,R)}\int_{B(o,R)}F\,dV.
\]
H\"older's inequality gives
\begin{equation}\label{eq:average-growth}
  k_F(R)\le V(o,R)^{-1/n}\left\lVert F\right\rVert_{L^n}.
\end{equation}
In particular, $k_F(R)=o(R^2)$. The Hodge--Laplace heat construction
of Tam and the second author~\cite[Theorem~4.1]{NT13}, whose curvature hypotheses are
precisely nonnegative Ricci curvature and \eqref{eq:qobc}, therefore
applies and gives a closed evolution. Reality and the domination
\eqref{eq:norm-domination} are part of that construction; see also
\cite[Lemmas~3.1 and~3.2]{NT13}. The Bochner inequality supplied by
\eqref{eq:qobc} makes the norm of a homogeneous $(1,1)$-form heat
solution a scalar heat subsolution, which is the input for this
pointwise domination.

Tracing the Hodge heat equation gives the scalar heat equation with
initial value $f$. Since $|f|\le C_nF$ and
$|u(t)|\le C_nP_tF$, scalar $L^n$-contractivity gives a uniform $L^n$
bound for $u$ on every finite time interval. Uniqueness in this class,
or \cite[Lemma~2.2]{NT13}, identifies $u$ with $P_tf$ and proves
\eqref{eq:trace-heat}.

Finally, the K\"ahler identities for a closed $(1,1)$-form give
\[
  -\Delta_{\bar\partial}\eta
  =\sqrt{-1}\,\partial\bar\partial\Lambda_\omega\eta;
\]
see \cite[Lemma~2.1]{NT13}. This proves \eqref{eq:exact-evolution}.
The local initial convergence is part of the heat construction.

<!-- theorem-5: body_md -->
\label{thm:critical}
Under the hypotheses of Lemma~\ref{lem:hodge}, the top-degree integral
is absolutely convergent and
\begin{equation}\label{eq:signed-vanish}
  \int_M\alpha^n=0.
\end{equation}
If, in addition, $\alpha\ge0$, then
\begin{equation}\label{eq:rank-vanish}
  \alpha^n\equiv0.
\end{equation}
In this semipositive case, $\alpha$ cannot be positive definite at any
point. For a semipositive form, the integrability hypothesis may
equivalently be written as $\Lambda_\omega\alpha\in L^n(M)$.
<!-- end theorem-5 -->
<!-- proof-6: body_md -->
Let $\eta(t)$ be the evolution of Lemma~\ref{lem:hodge}, and let
$u=P_tf$. Define, also at $t=0$ with $\eta(0)=\alpha$,
\[
  \mu_t=\frac{\eta(t)^n}{n!},
  \qquad J(t)=\int_M\mu_t.
\]
The form $\mu_t$ may have either sign. If $\mu_t=q_t\,dV$, write
$|\mu_t|=|q_t|\,dV$ for its total variation. The elementary pointwise
bound for wedge products, followed by \eqref{eq:norm-domination}, gives
\begin{equation}\label{eq:mass-upper}
  |\mu_t|\le C_n|\eta(t)|^n\,dV
  \le C_n(P_tF)^n\,dV
  \qquad(t>0).
\end{equation}
The analogous bound at $t=0$ is
$|\mu_0|\le C_nF^n\,dV$. Thus all these top-degree integrals are
absolutely convergent. By Lemma~\ref{lem:scalar},
\begin{equation}\label{eq:mass-finite}
  |J(t)|\le\int_M|\mu_t|
  \le C_n\left\lVert P_tF\right\rVert_{L^n}^{n}
  \le C_n\left\lVert F\right\rVert_{L^n}^{n}.
\end{equation}
Moreover,
\begin{equation}\label{eq:form-gradient-bounds}
  \left\lVert \eta(t)\right\rVert_{L^n}\le\left\lVert F\right\rVert_{L^n},
  \qquad
  \left\lVert \nabla u(t)\right\rVert_{L^n}
  \le C_n t^{-1/2}\left\lVert f\right\rVert_{L^n}
  \le C_n t^{-1/2}\left\lVert F\right\rVert_{L^n}.
\end{equation}

We now justify conservation of $J(t)$, including its initial value.
Choose compactly supported distance cutoffs $\chi_R$ with
\[
  0\le\chi_R\le1,\qquad
  \chi_R=1\text{ on }B(o,R),\qquad
  \operatorname{supp}\chi_R\subset B(o,2R),\qquad
  |d\chi_R|\le C/R.
\]
Lipschitz cutoffs are sufficient, and the integrations by parts below
are valid by smooth approximation on their compact supports. Fix
$0<a<b$. Since $d\eta(t)=0$ and $\eta_t=dd^c u$, we have
\begin{align}
 \int_M\chi_R(\mu_b-\mu_a)
 &=\frac1{(n-1)!}
   \int_a^b\int_M\chi_R\,dd^c u\wedge\eta^{n-1}\,dt\notag\\
 &=-\frac1{(n-1)!}
   \int_a^b\int_Md\chi_R\wedge d^cu\wedge\eta^{n-1}\,dt.
 \label{eq:flux}
\end{align}
The algebraic estimate $|\eta^{n-1}|\le C_n|\eta|^{n-1}$ does not
require a sign condition. It yields
\begin{align}
 \left|\int_M\chi_R(\mu_b-\mu_a)\right|
 &\le\frac{C_n}{R}
       \int_a^b\int_M|\nabla u|\,|\eta|^{n-1}\,dV\,dt\notag\\
 &\le\frac{C_n}{R}
       \int_a^b\left\lVert \nabla u(t)\right\rVert_{L^n}
                    \left\lVert \eta(t)\right\rVert_{L^n}^{n-1}\,dt\notag\\
 &\le\frac{C_n\left\lVert F\right\rVert_{L^n}^{n}}{R}
       \int_a^bt^{-1/2}\,dt\notag\\
 &\le\frac{C_n\left\lVert F\right\rVert_{L^n}^{n}}{R}
       (\sqrt b-\sqrt a).
 \label{eq:flux-bound}
\end{align}
Here we used H\"older's inequality and
\eqref{eq:form-gradient-bounds}.

Keep $R$ fixed and let $a\downarrow0$. The local convergence of
$\eta(a)$ to $\alpha$ on the compact support of $\chi_R$ gives
\begin{equation}\label{eq:initial-flux}
  \left|\int_M\chi_R(\mu_b-\mu_0)\right|
  \le \frac{C_n\left\lVert F\right\rVert_{L^n}^{n}}{R}\sqrt b.
\end{equation}
Since both $\mu_b$ and $\mu_0$ are absolutely integrable, dominated
convergence now allows $R\to\infty$. We conclude that
\begin{equation}\label{eq:mass-conservation}
  J(b)=J(0)=\int_M\frac{\alpha^n}{n!}
  \qquad(b>0).
\end{equation}
This order of limits uses only local convergence at initial time; no
interchange of a global integral with $a\downarrow0$ has been assumed.
The integrability of $t^{-1/2}$ near zero is what permits conservation
back to the initial form.

On the other hand, \eqref{eq:scalar-decay} and
\eqref{eq:mass-finite} give
\[
  |J(t)|\le C_n\left\lVert P_tF\right\rVert_{L^n}^{n}
  \longrightarrow0\qquad(t\to\infty).
\]
Together with \eqref{eq:mass-conservation}, this proves
\eqref{eq:signed-vanish}. If $\alpha\ge0$, then $\alpha^n$ is a smooth
nonnegative measure, so its zero integral implies
\eqref{eq:rank-vanish}. In that case $|\alpha|$ and
$\Lambda_\omega\alpha$ are pointwise comparable up to dimensional
constants, which proves the final assertion.

Notice that we have not inferred pointwise vanishing of $\eta(t)^n$
from $J(t)=0$, since $\eta(t)$ need not be semipositive. The sign of the
\emph{initial} form is used only after its integral has been identified
with the conserved signed integral.

<!-- theorem-6: body_md -->
\label{thm:finite}
Let $(M^n,\omega)$, $n\ge2$, be a connected complete noncompact K\"ahler
manifold with nonnegative Ricci curvature and nonnegative quadratic
orthogonal bisectional curvature. Let $\rho$ be its Ricci form and put
$S=\Lambda_\omega\rho$. Suppose that, for a constant $\varepsilon>0$,
\begin{equation}\label{eq:trace-pinching}
  \rho\ge\varepsilon S\omega,
  \qquad
  \int_M\rho^n<\infty.
\end{equation}
Then $\operatorname{Rm}_g\equiv0$.
<!-- end theorem-6 -->
<!-- proof-7: body_md -->
Let $\lambda_1,\ldots,\lambda_n\ge0$ be the eigenvalues of $\rho$
relative to $\omega$. The pinching condition gives
$\lambda_j\ge\varepsilon S$ for every $j$. Therefore
\begin{equation}\label{eq:pinched-determinant}
  \varepsilon^n S^n\,dV
  \le\frac{\rho^n}{n!}
  =\Bigl(\prod_{j=1}^n\lambda_j\Bigr)dV
  \le n^{-n}S^n\,dV.
\end{equation}
The finite-mass hypothesis implies $S\in L^n(M)$. Since $\rho\ge0$,
we also have $|\rho|\le C_nS$, so $|\rho|\in L^n(M)$. The Ricci form
is smooth, real, and closed. Theorem~\ref{thm:critical} applied to
$\alpha=\rho$ therefore gives $\rho^n\equiv0$. The first inequality
in \eqref{eq:pinched-determinant} then implies $S\equiv0$, and
nonnegative Ricci curvature gives $\rho\equiv0$.

To conclude full flatness under \eqref{eq:qobc}, fix a point $x\in M$
and a unit $(1,0)$-vector $X\in T_x^{1,0}M$. Extend $e_1=X$ to a
unitary frame and take $(a_1,a_2,\ldots,a_n)=(1,0,\ldots,0)$ in
\eqref{eq:qobc}. Writing
$H(X)=R(X,\overline X,X,\overline X)$, we obtain
\begin{equation}\label{eq:holomorphic-nonpositive}
  0\le2\sum_{j=2}^nR_{1\bar1 j\bar j}
  =2\bigl(\operatorname{Ric}(X,\overline X)-H(X)\bigr)
  =-2H(X).
\end{equation}
Thus $H(X)\le0$ for every unit $X$.

Let $d\sigma$ be the unitary-invariant probability measure on the
unit sphere in $T_x^{1,0}M$. The elementary moment identity
\[
  \int_{|z|=1}z_i\bar z_j z_k\bar z_l\,d\sigma(z)
  =\frac{\delta_{ij}\delta_{kl}+\delta_{il}\delta_{kj}}{n(n+1)}
\]
and the K\"ahler curvature symmetries give
\begin{equation}\label{eq:holomorphic-average}
  \int_{|X|=1}H(X)\,d\sigma(X)
  =\frac{2S(x)}{n(n+1)}=0.
\end{equation}
A continuous nonpositive function with zero average is identically
zero. Hence all holomorphic sectional curvatures vanish at $x$.
Polarization of the K\"ahler curvature tensor gives $\operatorname{Rm}(x)=0$.
Since $x$ was arbitrary, $\operatorname{Rm}_g\equiv0$.

<!-- lemma-5: body_md -->
\label{lem:busemann}
Let $(M^n,g,J)$ be a connected complete noncompact K\"ahler manifold with
strictly positive holomorphic bisectional curvature. Assume that the
real sectional curvature is nonnegative outside a compact set.
Then $M$ admits a smooth strictly plurisubharmonic exhaustion $\psi$
such that
\begin{equation}\label{eq:busemann-gradient}
  \sup_M|\nabla\psi|\le1.
\end{equation}
<!-- end lemma-5 -->
<!-- proof-8: body_md -->
Fix $o\in M$. We use the full Busemann function
\begin{equation}\label{eq:full-busemann}
  B(x)=\sup_{\gamma}B_\gamma(x),
  \qquad
  B_\gamma(x)=\lim_{s\to\infty}\bigl(s-d(x,\gamma(s))\bigr),
\end{equation}
where $\gamma$ ranges over all unit-speed minimizing rays from $o$.
In particular, $B$ is not the Busemann function of an arbitrary single
ray. It is a continuous plurisubharmonic function with Lipschitz
constant one; see \cite[Section~4]{NT03}. The exhaustion argument
invoked in \cite[proof of Theorem~4.2]{NT03} shows that $B$ is proper
when the real sectional curvature is nonnegative outside a compact
set.

Let $v(\cdot,t)=P_tB$. By \cite[Lemma~4.1(i)--(iii)]{NT03}, at each
positive time $v(\cdot,t)$ is smooth and strictly plurisubharmonic,
its gradient has norm at most one, and it is an exhaustion because
$B$ is an exhaustion. The harmless constant rescaling of heat time in
our Laplacian convention does not affect these statements. Fixing
one positive time $t_0$ and taking $\psi=v(\cdot,t_0)$ proves the
lemma.

<!-- proposition-3: body_md -->
\label{prop:top-mass-busemann}
Under the hypotheses of Lemma~\ref{lem:busemann}, the Ricci form
satisfies
\begin{equation}\label{eq:DPS}
  \int_M\rho^n<\infty.
\end{equation}
No Ricci pinching assumption is required for this conclusion.
<!-- end proposition-3 -->
<!-- proof-9: body_md -->
Lemma~\ref{lem:busemann} gives a smooth strictly plurisubharmonic
exhaustion with uniformly bounded gradient. Together with positive
holomorphic bisectional curvature, these are exactly the hypotheses in
the general clause of Datar--Pingali--Seshadri
\cite[Theorem~1.1]{DPS26}. That result gives \eqref{eq:DPS}.
<!-- end proof-9 -->
<!-- proof-10: body_md -->
Suppose, to obtain a contradiction, that $M$ is noncompact. Positive
holomorphic bisectional curvature implies $\rho>0$ everywhere.
The continuous function
\[
  x\longmapsto
  \frac{\lambda_{\min}(\operatorname{Ric}_g)(x)}
       {\lambda_{\max}(\operatorname{Ric}_g)(x)}
\]
has a positive minimum on the compact set in the theorem. Thus, after
decreasing the exterior pinching constant, we obtain
\eqref{eq:ratio} on all of $M$, with a constant $0<\kappa\le1$.

Write $0<\lambda_1\le\cdots\le\lambda_n$ for the Hermitian Ricci
eigenvalues. By \eqref{eq:ratio},
\[
  S=\sum_{j=1}^n\lambda_j
  \le\lambda_1+(n-1)\lambda_n
  \le\left(1+\frac{n-1}{\kappa}\right)\lambda_1.
\]
Therefore
\begin{equation}\label{eq:epsilon-kappa}
  \rho\ge\varepsilon S\omega,
  \qquad
  \varepsilon=\frac{\kappa}{n-1+\kappa}>0.
\end{equation}
Proposition~\ref{prop:top-mass-busemann} gives \eqref{eq:DPS}.
Nonnegative Ricci curvature and nonnegative quadratic orthogonal
bisectional curvature also follow from positive holomorphic
bisectional curvature. All the hypotheses of
Theorem~\ref{thm:finite} are now satisfied, so $\operatorname{Rm}_g\equiv0$.
This contradicts strictly positive holomorphic bisectional curvature.
Hence $M$ is compact.

<!-- remark-2: body_md -->
Real sectional curvature is used only to obtain exhaustiveness in
Lemma~\ref{lem:busemann}. Consequently, one may instead assume that
the full Busemann function is exhaustive, while retaining positive
holomorphic bisectional curvature everywhere and uniform Ricci
pinching outside a compact set. More generally, a smooth strictly
plurisubharmonic exhaustion with bounded gradient is sufficient.
Positive bisectional curvature alone is not being asserted to supply
this exhaustion. In the special case $K_g\ge0$ everywhere, the full
Busemann function is convex as well as exhaustive; global convexity
is not required by the proof above.

<!-- remark-3: body_md -->
The proof uses the Hodge-Laplace heat equation deformation of a $(1,1)$-form, not a global solution
of $dd^c\phi=\rho$. In particular, it does not require the stronger
average-integrability hypothesis for the Poincar\'e--Lelong theorem in
\cite[Theorem~1.2]{NT13}. It also does not require a quantitative rate
for $\left\lVert P_t|\rho|\right\rVert_{L^n}\to0$. Norm domination of the form evolution
and conservation of the signed top-degree integral, including its initial
value, allow qualitative $L^n$ decay to suffice. Semipositivity is used
only for the initial Ricci form, not for the evolved forms. The Busemann
heat deformation serves solely to verify the geometric finiteness input;
it is separate from the Hodge heat evolution of the Ricci form.

<!-- remark-4: body_md -->
The analytic Theorem~\ref{thm:finite} requires only nonnegative Ricci
curvature and nonnegative quadratic orthogonal bisectional curvature,
together with Ricci pinching and finite-mass assumptions. This weaker
analytic curvature hypothesis does not by itself supply a finiteness
theorem for the top Ricci integral. In

The restriction $n\ge2$ cannot be omitted. The cigar metric

<!-- proof-11: body_md -->
Fix $0<a<1$ and $\delta>0$. We choose
\begin{equation}\label{eq:profile}
  \xi(t)=a\left(1-\bigl(\log(e+t)\bigr)^{-\delta}\right).
\end{equation}
Then $\xi(0)=0$, $\xi'(t)>0$, and $\xi(t)<a<1$. The Wu--Zheng
criterion implies that \eqref{eq:radial-metric} has positive bisectional
curvature. Smoothness at the origin follows from
\[
  \xi(t)=\frac{a\delta}{e}t+O(t^2).
\]
For $t\ge1$, the inequality $\xi\le a$ gives
$h(t)\ge h(1)t^{-a}$. Thus the integral in
\eqref{eq:radial-distance} diverges, and the metric is complete.

Put $b=1-a>0$. Since
\[
  \frac{th'(t)}{h(t)}=-\xi(t)\longrightarrow-a,
\]
l'H\^opital's rule gives
\begin{equation}\label{eq:radial-asymptotics}
  H(t)\sim\frac{th(t)}b,
  \qquad
  r(t)\sim\frac{\sqrt{2th(t)}}b.
\end{equation}
Here and below $u\sim v$ means $u/v\to1$. Indeed,
$(th)'=(1-\xi)h$, and differentiating $\sqrt{th}$ gives the second
limit. Also,
\begin{equation}\label{eq:log-asymptotics}
  \log h(t)=-a\log t+o(\log t),
  \qquad
  \log r(t)\sim\frac b2\log t.
\end{equation}
The determinant of \eqref{eq:radial-metric} is $f^{n-1}h$.
Integration in polar coordinates yields
\[
  \operatorname{Vol}(B(0,r(t)))=2^n\omega_{2n}H(t)^n.
\]
Combining this with \eqref{eq:radial-asymptotics}, we obtain
\begin{equation}\label{eq:example-avr}
  \lim_{r\to\infty}
  \frac{\operatorname{Vol}(B(0,r))}{\omega_{2n}r^{2n}}=b^n>0.
\end{equation}

We next compute the curvature. By \eqref{eq:profile},
\[
  \xi'(t)=\frac{a\delta}{(e+t)(\log(e+t))^{1+\delta}},
\]
and hence
\begin{equation}\label{eq:A-asymptotic}
  A(t)\sim\frac{a\delta}{th(t)(\log t)^{1+\delta}}.
\end{equation}
To estimate $B$, set $J(t)=\int_0^t\xi'(u)H(u)\,du$.
By \eqref{eq:radial-asymptotics},
\[
  J'(t)\sim\frac{a\delta}{b}
                     \frac{h(t)}{(\log t)^{1+\delta}}.
\]
Since
\[
  \frac{d}{dt}\left(\frac{th(t)}{(\log t)^{1+\delta}}\right)
  \sim\frac{bh(t)}{(\log t)^{1+\delta}},
\]
another application of l'H\^opital's rule gives
\[
  J(t)\sim\frac{a\delta}{b^2}
                     \frac{th(t)}{(\log t)^{1+\delta}}.
\]
It follows that
\begin{equation}\label{eq:BD-asymptotic}
  B(t)\sim\frac{a\delta}{th(t)(\log t)^{1+\delta}},
  \qquad
  D(t)\sim\frac{2ab}{th(t)}.
\end{equation}
For the second relation, we used $\int_0^t\xi h\sim aH(t)$,
which follows from $\xi\to a$ and $H(t)\to\infty$.

Equations~\eqref{eq:radial-ricci}, \eqref{eq:A-asymptotic},
and~\eqref{eq:BD-asymptotic} imply
\[
  \lambda_{\mathrm{rad}}(t)
      \sim\frac{na\delta}{th(t)(\log t)^{1+\delta}},
  \qquad
  \lambda_{\mathrm{tan}}(t)\sim\frac{nab}{th(t)}.
\]
In particular, $\lambda_{\mathrm{rad}}<\lambda_{\mathrm{tan}}$ for
sufficiently large $t$, and
\[
  \frac{\operatorname{Ric}_{\min}}{\operatorname{Ric}_{\max}}
  \sim\frac\delta b(\log t)^{-1-\delta}
  \asymp(\log r)^{-1-\delta}.
\]
Finally, \eqref{eq:radial-curvature}, \eqref{eq:radial-asymptotics},
\eqref{eq:A-asymptotic}, and~\eqref{eq:BD-asymptotic} show that
$|\operatorname{Rm}|=O(r^{-2})$. On a compact neighborhood of the origin the
curvature is bounded by smoothness. This proves all the assertions.
