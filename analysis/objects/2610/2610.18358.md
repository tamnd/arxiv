<!-- theorem-1: body_md -->
\label{maintheorem}
Assume that $k\ge2$, $\phi_1\not\equiv0$ and \eqref{gh}--\eqref{g00} hold.
Let $T_\delta$ denote the lifespan of the smooth solution $\phi$.
Then we have
\begin{enumerate}[label=(\roman*)]
\item when $\varepsilon_k^*<\varepsilon_0<1$, for small $\delta>0$, the solution $\phi$ develops a singularity in finite time, and
\begin{equation}\label{lifespan}
\limsup_{\delta\to0^+}
\delta^{(k+1)(\varepsilon_k^*-\varepsilon_0)}(T_\delta-1)
\le\frac{2}{(k+1)\Xi}.
\end{equation}
\item when $\varepsilon_0=\varepsilon_k^*$ and \eqref{critical-compression}
holds, for small $\delta>0$,  the solution $\phi$ will blow up in finite time and
\begin{equation}\label{critical-lifespan}
\limsup_{\delta\to0^+}T_\delta
\le S_{\mathrm{crit}}
:=\big(1-\frac{k-1}{(k+1)\Xi}\big)^{-\frac{2}{k-1}}.
\end{equation}
\end{enumerate}

<!-- remark-2: body_md -->
\label{Rem.3}
Consider the small data solution problem for the 2D quasilinear wave equation
\begin{equation}\label{quasi-0}
\text{$\sum_{\alpha,\beta=0}^2g^{\alpha\beta}(\partial v)\partial_{\alpha\beta}^2 v=0$ with $(v,\partial_t v)(1,x)=\delta(v_0(x),v_1(x))$},
\end{equation}
where $(v_0, v_1)(x)\in C_0^{\infty}(\mathbb R^2)$ and $\delta>0$ is small.
Assume that for small $\partial v$,
\begin{equation}\label{H-01}
\displaystyle g^{\alpha\beta}(\partial v)=m^{\alpha\beta}+\sum_{\kappa=0}^{2} g^{\alpha\beta,\kappa}\partial_\kappa v+\sum_{\kappa,\varsigma=0}^2h^{\alpha\beta,\kappa\varsigma}\partial_\kappa v\partial_\varsigma v+O(|\partial v|^3),
\end{equation}
where $m^{\alpha\beta}$ denotes the Minkowski metric, $g^{\alpha\beta,\kappa}$
and $h^{\alpha\beta,\kappa\varsigma}$ are constants.
It is shown in \cite{Ali3} that the problem \eqref{quasi-0} admits a global smooth solution $v$ satisfying $|\partial v|\le C\delta(1+t)^{-1/2}$ when both the first and the second null conditions hold (i.e.,
$\displaystyle\sum_{\alpha,\beta,\kappa=0}^2g^{\alpha\beta,\kappa}\omega_\alpha\omega_\beta\omega_\kappa\equiv0$ and $\displaystyle\sum_{\alpha,\beta,\kappa,\varsigma=0}^2 h^{\alpha\beta,\kappa\varsigma}\omega_\alpha\omega_\beta\omega_\kappa\omega_\varsigma\equiv0$
for $\omega_0=-1$ and $(\omega_1,\omega_2)\in\mathbb S$, respectively).
Otherwise, if either the first or the second null condition fails,
under the generic non-degenerate assumption on $(v_0(x),v_1(x))$, then $v$ blows up in finite time (see \cite{Ali1} and \cite{Ali2}).


For the short pulse problem considered here, no additional generic non-degeneracy assumption is needed for $\varepsilon_k^* \leq \varepsilon_0 < 1$ except the usual condition $\phi_1 \not\equiv 0$.

\label{Local}
Under the assumptions \eqref{gh}--\eqref{g00}, when $\delta>0$ is small, \eqref{quasi} with \eqref{id}
admits a local smooth solution $\phi\in C^\infty([1, t_0]\times\mathbb R^2)$. Moreover,

\end{equation}

\end{align}

Denote by $Z_g$ any fixed vector field in $\{S, H_1, H_2\}$.
Suppose that for $1\leq t\leq t_0$
and $N_0\in\mathbb N_0$ with $N_0\ge 6$, one has
	\begin{equation}\label{ba}
	|\partial^\kappa\Omega^pZ_g^\nu\phi|\leq\delta^{2-\epsilon-|\nu|\varepsilon_k-|\kappa|}\quad (|\kappa|+p+|\nu|\leq N_0,\quad |\nu|\leq 1),
	\end{equation}
where $\epsilon\in(\varepsilon0, 1)$ is a fixed constant.
Define the following energy for \eqref{quasi} and for $n\in\mathbb N_0$,
$$
M_n(t)=\sum_{\tiny\begin{array}{c}|\kappa|+p+|\nu|\leq n\\|\nu|\leq 1\end{array}}\delta^{2|\kappa|+2|\nu|\varepsilon_k}\|\partial\partial^\kappa\Omega^pZ_g^\nu\phi(t,\cdot)\|_{L^2(\mathbb R^2)}^2.
$$

Let $w=\delta^{|\kappa|+|\nu|\varepsilon_k}\partial^\kappa\Omega^pZ_g^\nu\phi\quad(|\kappa|+p+|\nu|\leq 2N_0-2)$. It follows from \eqref{quasi}
and integration by parts that
\begin{equation}\label{Y-4}
\begin{split}
&\int_1^t\int_{\Sigma_\tau}(\partial_twg^{\alpha\beta}\partial_{\alpha\beta}^2w)(\tau,x) dxd\tau\\
=&\int_{\Sigma_{t}}\big(g^{0\alpha}\partial_\alpha w\partial_t w-\frac12g^{\alpha\beta}\partial_\alpha w\partial_\beta w\big)(t,x)dx-\int_{\Sigma_{1}}\big(g^{0\alpha}\partial_\alpha w\partial_t w-\frac12g^{\alpha\beta}\partial_\alpha w\partial_\beta w\big)(1,x)dx\\
&+\int_1^t\int_{\Sigma_\tau}\big(-(\partial_\alpha g^{\alpha\beta})\partial_\beta w\partial_t w+\frac12(\partial_tg^{\alpha\beta})\partial_\alpha w\partial_\beta w\big)(\tau,x) dxd\tau
\end{split}
\end{equation}
with
\begin{equation}\label{w}
\begin{split}
|g^{\alpha\beta}&\partial_{\alpha\beta}^2w|\lesssim\delta^{|\kappa|+|\nu|\varepsilon_k}\sum_{\tiny\begin{array}{c}
	|\kappa_1|+|\kappa_2|\leq|\kappa|,\ p_1+p_2\leq p,\\|\nu_1|+|\nu_2|\leq|\nu|,\\ |\kappa_2|+p_2+|\nu_2|<|\kappa|+p+|\nu|\end{array}}|\partial^{\kappa_1}\Omega^{p_1}Z_g^{\nu_1}	(g^{\alpha\beta}-m^{\alpha\beta})||\partial^{\kappa_2}\Omega^{p_2}Z_g^{\nu_2}\partial^2\phi|.
\end{split}
\end{equation}
By \eqref{ba}-\eqref{w} and Gronwall's inequality, we obtain that for $1\leq t\leq t_0$,
$$
M_{2N_0-2}(t)\lesssim M_{2N_0-2}(1)\lesssim\delta^{3-2\varepsilon_0}.
$$
In addition, by the following Sobolev embedding theorem on the circle $\mathbb S_r$
	(with center at the origin and radius $r$), one has that
	$$
	|w(t,x)|\lesssim \frac{1}{\sqrt{r}}\|\Omega^{\leq 1}w\|_{L^2(\mathbb S_r)}.
	$$
This, together with $r\sim 1$ for $t\in [1,t_0]$ and $(t,x)\in \textrm{supp}\ w$, yields that when $|\kappa|+p+|\nu|\leq N_0$,
	\begin{equation}\label{le}
	|\partial^\kappa\Omega^pZ_g^\nu\phi(t,x)|\lesssim \|\Omega^{\leq 1}\partial^\kappa\Omega^pZ_g^\nu\phi\|_{L^2(\mathbb S_r)}\lesssim\delta^{1/2}\|\partial\Omega^{\leq 1}\partial^\kappa\Omega^pZ_g^\nu\phi\|_{L^2(\Sigma_{t})}\lesssim\delta^{2-|\kappa|-\varepsilon_0-|\nu|\varepsilon_k},\quad
	\end{equation}
	where $N_0\geq 6$ has been used. Therefore, \eqref{ba} can be closed
	for suitably small $\delta>0$ and $\epsilon>\varepsilon_0$.
	
	
Next we improve the $L^\infty$ estimate of $\phi(t_0,x)$ on some special space domains. To this end,
one rewrites \eqref{quasi} as
\begin{equation}\label{me}
L\underline L\phi=\frac{1}{2r}L\phi-\frac{1}{2r}\underline L\phi+\frac{1}{r^2}\Omega^2\phi+g^{\alpha\beta,
	\gamma_1,\cdots,\gamma_k}(\partial_{\gamma_1}\phi)\cdots(\partial_{\gamma_k}\phi)\partial^2_{\alpha\beta}\phi+h^{\alpha\beta}(\partial\phi)\partial^2_{\alpha\beta}\phi.
\end{equation}


Applying the operator $L\bar\partial^{\iota}\Omega^p$ to both sides of \eqref{me}, where $\bar\partial\in\{\partial_t,\partial_r\}$ and $\iota\in\mathbb N_0^2$,
we can obtain an expression of $\underline LL^2\bar\partial^{\iota}\Omega^p\phi$ by $\underline LL^2\bar\partial^{\iota}\Omega^p=L\bar\partial^{\iota}\Omega^pL\underline L$  and a direct computation.
Together with \eqref{le}, this implies that for $|\iota|+p\leq 2N_0-6$,
\begin{equation}\label{nonull}
	|\underline L(L^2\bar\partial^{\iota}\Omega^p\phi)|\lesssim \delta^{-2\varepsilon_k-|\iota|},
\end{equation}
here  the null condition \eqref{null} is not used. Due to the vanishing property of $\phi$ on $\{(t, x): t\ge 1, t=r\}$,
integrating \eqref{nonull} along the integral curves of $\underline L$ yields that for $(t,r)\in D^1=\{(t,r): 1\le t\le t_0, 2-t\le r\le t\}$ and for $|\iota|+p\leq 2N_0-6$,
\begin{equation}\label{D}
|L^2\bar\partial^{\iota}\Omega^p\phi(t,x)|\lesssim\delta^{1-2\varepsilon_k-|\iota|}.
\end{equation}

\label{prop:bootstrap-higher-derivatives}
Under the bootstrap assumptions \eqref{bootstrap-assumptions},
for small $\delta>0$, the following estimates
hold that for $t_0\leq s<T_\delta$ and $|\alpha|\leq N-2$:
\begin{equation}\label{eq:bootstrap-higher-derivatives}
\delta^{l+m\varepsilon_k}|Z^\alpha\phi|
\lesssim\delta^{2-\varepsilon_0},
\quad
\delta^{l+m\varepsilon_k}|Z^\alpha\varphi_\gamma|
\lesssim\delta^{1-\varepsilon_0},
\quad \gamma=0,1,2,
\end{equation}
where $Z \in \{ \mathring{L}, T, R \}$, $l$ and $m$
denote the numbers of $T$  and $\mathring{L}$ in $Z^\alpha$, respectively.
Moreover,
\begin{align}
\bigl|
\varrho^{1/2}T\varphi_\gamma(s,u,\vartheta)
-\varrho_0^{1/2}T\varphi_\gamma(t_0,u,\vartheta)
\bigr|
&\lesssim M^{k+1}\delta^{-\varepsilon_k},
\label{Tvp}\\
\bigl|
\varrho^{1/2}\varphi_\gamma(s,u,\vartheta)
-\varrho_0^{1/2}\varphi_\gamma(t_0,u,\vartheta)
\bigr|
&\lesssim M^{k+1}\delta^{(k+1)(1-\varepsilon_0)}
\label{r0vp}
\end{align}
with $\varrho=s-u$ and $\varrho_0=t_0-u$.

<!-- proof-5: body_md -->
From \eqref{fequation} and Proposition~\ref{prop:higher-order-L-infty-estimates}, we have that for all $|\alpha| \leq N-2$,
\begin{equation*}
\begin{split}
&\bigl| \mathring{L}{\bar Z}^\alpha \underline{\mathring L} \varphi_\gamma + \frac{1}{2\varrho}{\bar Z}^\alpha\underline{\mathring L} \varphi_\gamma \bigr| \\
&= \big|{\bar Z}^\alpha (\mu\slashed\triangle\varphi_\gamma+H_\gamma )+ [\mathring{L}, {\bar Z}^\alpha] \underline{\mathring L} \varphi_\gamma
- \sum_{\substack{\alpha_1 + \alpha_2 =\alpha\\ |\alpha_1| > 0}}{\bar Z}^{\alpha_1} \big( \frac{1}{2\varrho} \big){\bar Z}^{\alpha_2} \underline{\mathring L} \varphi_\gamma \big| \\
&\lesssim M^{k+1}\delta^{-2\varepsilon_k-l},
\end{split}
\end{equation*}
where $\bar Z\in\{T, R\}$ and $l$ is the number of $T$ operators in $\bar Z^\alpha$.
Consequently,
\[
\bigl| \mathring{L} \bigl( \varrho^{1/2}{\bar Z}^\alpha \underline{\mathring L} \varphi_\gamma \bigr) \bigr|
\lesssim M^{k+1}\delta^{-2\varepsilon_k-l},
\]
which implies
\[
\bigl| \varrho^{1/2} {\bar Z}^\alpha \underline{\mathring L} \varphi_\gamma(s,u,\vartheta) -\varrho_0^{1/2}{\bar Z}^\alpha \underline{\mathring L} \varphi_\gamma(t_0,u,\vartheta) \bigr|
\lesssim M^{k+1}\delta^{-\varepsilon_k-l}.
\]
From this, \eqref{Tvp} and $|{\bar Z}^\alpha T \varphi_\gamma| \lesssim \delta^{-\varepsilon_0-l}$ can be obtained.
Then commuting $T$ with $\bar Z^\alpha$ yields $|T\bar{Z}^\alpha \varphi_\gamma| \lesssim \delta^{-\varepsilon_0-l}$.
In addition, by \eqref{Xi} and \eqref{bootstrap-assumptions}, we deduce
\begin{equation}\label{barzvp}
\Bigl| \frac{\partial}{\partial u}{\bar Z}^\alpha \varphi_\gamma \Bigr|
\lesssim \bigl| T{\bar Z}^\alpha \varphi_\gamma \bigr| + \bigl| \Theta^X \slashed{d}_X {\bar Z}^\alpha \varphi_\gamma \bigr|
\lesssim \delta^{-\varepsilon_0-l}
\end{equation}
and
\begin{equation}\label{YHCC-A1}
\begin{split}

\end{split}
\end{equation}

The case of $m=0$ follows by integrating \eqref{barzvp}. For the induction argument, assume that for some integer $p\ge 0$ 
and for all $|\alpha|\leq N-2$ containing at most $p$  $\mathring{L}$'s, it holds
\[
|Z^\alpha\varphi_\gamma|\lesssim\delta^{1-\varepsilon_0-l-m\varepsilon_k},

\begin{split}
|Z^\beta\mathring L\underline{\mathring L}\varphi_\gamma|
&=\bigl|Z^\beta\bigl(\mu\slashed\triangle\varphi_\gamma+H_\gamma-\frac{1}{2\varrho}\underline{\mathring L}\varphi_\gamma\bigr)\bigr|\\
&\lesssim\delta^{-l-p\varepsilon_k}\bigl(\delta^{-\varepsilon_0}+\delta^{-2\varepsilon_k}M^{k+1}\bigr),
\end{split}
\end{equation*}
which implies
\[
|Z^\beta\mathring L T\varphi_\gamma|\lesssim\delta^{-l-p\varepsilon_k}\bigl(\delta^{-\varepsilon_0}+\delta^{-2\varepsilon_k}M^{k+1}\bigr).
\]

\[
|T Z^\beta\mathring L\varphi_\gamma|\lesssim\delta^{-l-p\varepsilon_k}\bigl(\delta^{-\varepsilon_0}+\delta^{-2\varepsilon_k}M^{k+1}\bigr).
\]

\[
\Bigl|\frac{\partial}{\partial u}Z^\beta\mathring L\varphi_\gamma\Bigr|
\lesssim\delta^{-l-p\varepsilon_k}\bigl(\delta^{-\varepsilon_0}+\delta^{-2\varepsilon_k}M^{k+1}\bigr).
\]

\[

This completes the induction argument and obtains the second bound in~\eqref{eq:bootstrap-higher-derivatives}.

\]
Integrating this from $0$ to $u$ yields
$|Z^\alpha \phi| \lesssim \delta^{2-\varepsilon_0-l-m\varepsilon_k}$,
which is just the first estimate in~\eqref{eq:bootstrap-higher-derivatives}.

\quad
g(s,0,\vartheta)=0.

+ \delta^{l_1+p_1\varepsilon_k} \| Z^{m_1} \operatorname{tr}\check{\mathscr{X}} \|_{s,u} + \delta^{k(1-\varepsilon_0)} \delta^{l_0+p_0\varepsilon_k} \| Z^{m_0}x \|_{s,u}\\
&+ \delta(1+\delta^{2k(1-\varepsilon_0)-\varepsilon_0}) \delta^{l_0+p_0\varepsilon_k} \| Z^{m_0}\mu \|_{s,u} + \delta^{k(1-\varepsilon_0)} \delta^{l_1+p_1\varepsilon_k} \| \slashed{\mathcal L}_Z^{m_1}\slashed{g} \|_{s,u} \\
&+ \delta^{k(1-\varepsilon_0)+\varepsilon_0} \bigl( \sqrt{\mathcal E_{1,\leq m+2}(s,u)} + \sqrt{\mathcal E_{2,\leq m+2}(s,u)} \bigr),
\end{split}
\end{equation}
where the bound \(\|1\|_{s,u}\lesssim\delta^{1/2}\) is used. Here \(l\) and \(p\) (resp.\ \(l_n\) and \(p_n\), \(n=0,1\)) denote the numbers of \(T\) and \(\varrho\mathring{L}\) in the string \(Z^{m+1}\) (resp.\ \(Z^{m_n}\)) with \(0 \leq m_n \leq m+1-n\).

Similarly, from the identities \(\varrho\mathring Lx^i=\varrho\check L^i+x^i\), \(Tx^i=\mu(-g^{0i}-\check L^i-\frac{x^i}{\varrho})\) and \(Rx^i=\Omega^i+\upsilon(g^{0i}+\check L^i+\frac{x^i}{\varrho})\), one has
\begin{equation}\label{Zkx}
\begin{split}
&\delta^{l+p\varepsilon_k}\|Z^{m+2}x^i\|_{s,u}\\
\lesssim&\delta^{1/2}+\delta^{l_0+p_0\varepsilon_k}\|Z^{m_0}\upsilon\|_{s,u}
+(\delta^{k(1-\varepsilon_0)}+\delta^{\varepsilon_k})\sum_{j=1}^2\delta^{l_0+p_0\varepsilon_k}\|Z^{m_0}\check L^j\|_{s,u}\\
&+\delta\delta^{l_{0}+p_0\varepsilon_k}\|Z^{m_{0}}\mu\|_{s,u}
+\delta^{2k(1-\varepsilon_0)+\varepsilon_0}\bigl( \sqrt{\mathcal E_{1,\leq m+2}(s,u)} + \sqrt{\mathcal E_{2,\leq m+2}(s,u)} \bigr).
\end{split}
\end{equation}
In addition, by \(\slashed{\mathcal L}_{\varrho\mathring L}R=\varrho\leftidx{^{(R)}}{\slashed\pi}_{\mathring L}\), \(\slashed{\mathcal L}_TR=\leftidx{^{(R)}}{\slashed\pi}_T\) and \eqref{Rpi}, it holds that
\begin{equation}\label{ZR}
\begin{split}
&\delta^{l+p\varepsilon_k}\|\slashed{\mathcal L}_Z^{m+1}R\|_{s,u}\\
\lesssim&\delta^{\varepsilon_0+1/2}+(\delta^{2k(1-\varepsilon_0)}+\delta^{\varepsilon_0})\delta^{l_0
+p_0\varepsilon_k}\|Z^{m_0}x\|_{s,u}+\delta^{k(1-\varepsilon_0)}\delta^{l_1+p_1\varepsilon_k}\|Z^{m_1}\upsilon\|_{s,u}\\
&+\delta^{\varepsilon_k}\delta^{l_1+p_1\varepsilon_k}\|Z^{m_1}\operatorname{tr}\check{\mathscr X}\|_{s,u}+(\delta^{2k(1-\varepsilon_0)}+\delta^{\varepsilon_k})\sum_{j=1}^2\delta^{l_1+p_1\varepsilon_k}\|Z^{m_1}\check L^j\|_{s,u}\\
&+(\delta^{\varepsilon_0}+\delta^{2k(1-\varepsilon_0)})\delta^{l_1+p_1\varepsilon_k}\|\slashed{\mathcal L}_Z^{m_1}\slashed g\|_{s,u}+\delta^{1+k(1-\varepsilon_0)}\delta^{l_{0}+p_0\varepsilon_k}\|Z^{m_{0}}\mu\|_{s,u}\\
&+\delta^{\varepsilon_0+2k(1-\varepsilon_0)}\bigl(\sqrt{\mathcal E_{1,\leq m+2}(s,u)} + \sqrt{\mathcal E_{2,\leq m+2}(s,u)}\bigr).
\end{split}
\end{equation}
On the other hand, from \eqref{omega}, \eqref{rrho} and the relations \(\slashed{\mathcal L}_{\varrho\mathring L}\slashed g=2\varrho\chi\), \(\slashed{\mathcal L}_T\slashed g=\leftidx{^{(T)}}{\slashed\pi}\), \(\slashed{\mathcal L}_R\slashed g=\leftidx{^{(R)}}{\slashed\pi}\), we have
\begin{equation}\label{Zku}
\begin{split}
\delta^{l+p\varepsilon_k}\|Z^{m+1}\upsilon\|_{s,u}
\lesssim& \delta^{k(1-\varepsilon_0)}\delta^{l_0+p_0\varepsilon_k}\|Z^{m_0}x\|_{s,u}+\sum_{i=1}^2\delta^{l_0+p_0\varepsilon_k}\|Z^{m_0}\check L^i\|_{s,u}\\
&+\delta^{k(1-\varepsilon_0)+\varepsilon_0}\bigl( \sqrt{\mathcal E_{1,\leq m+2}(s,u)} + \sqrt{\mathcal E_{2,\leq m+2}(s,u)} \bigr),
\end{split}
\end{equation}
\begin{equation}\label{Zkr}
\begin{split}
\delta^{l+p\varepsilon_k}\|Z^{m+1}\check\varrho\|_{s,u}
\lesssim & \delta^{k(1-\varepsilon_0)}\delta^{l_0+p_0\varepsilon_k}\|Z^{m_0}x\|_{s,u}
+\sum_{i=1}^2\delta^{l_0+p_0\varepsilon_k}\|Z^{m_0}\check L^i\|_{s,u}\\
&+\delta^{2k(1-\varepsilon_0)+1/2}+ \delta^{k(1-\varepsilon_0)+\varepsilon_0}\bigl(\sqrt{\mathcal E_{1,\leq m+2}(s,u)} + \sqrt{\mathcal E_{2,\leq m+2}(s,u)}\bigr),
\end{split}
\end{equation}
\begin{equation}\label{Zkg}
\begin{split}

Combining \eqref{ZkL}--\eqref{Zkg} and using induction on the derivative orders yield
\begin{align}
\delta^{l+p\varepsilon_k}\|Z^{m+2}x^i\|_{s,u}\lesssim&\delta^{1/2}+\Pi(s,u),\label{Zk1x}\\
\delta^{l+p\varepsilon_k}\|\slashed{\mathcal L}_Z^{m+1}\slashed g\|_{s,u}\lesssim&\delta^{1/2}(\delta^{k(1-\varepsilon_0)}+\delta^{\varepsilon_k})+\Pi(s,u),\label{Zk1g}\\

Collecting all the estimates from \eqref{RmcX}, \eqref{RMMu}, \eqref{1lpchi}--\eqref{1lpmu} and \eqref{Zk1x}--\eqref{ZkTLpi},
we finish the proof of Proposition \ref{L2chi}.

Recalling that $T\mathring L^i$ satisfies \eqref{TL}, we have
\begin{equation}\label{TLp}
T(\mathring L^\alpha \varphi_\alpha)=\mu\tilde T^i(\mathring L\varphi_i)+(\slashed d_X\mu)\slashed d^X\phi-(G_{X\mathring L}^\gamma T\varphi_\gamma)\slashed d^X\phi+\frac12(G_{\mathring L\mathring L}^\gamma T\varphi_\gamma)\tilde T^i\varphi_i+\dots.
\end{equation}
With the help of \eqref{GT} and \eqref{gLLL}, substituting \eqref{TLp} into \eqref{ZLp} yields
\begin{align}
&\|R^{m+1}\mathring{L}\phi\|_{s,u}\nonumber\\
\lesssim&\ \delta^{(k+1)(1-\varepsilon_0)}\Bigl(\sum_{i=1}^2\|R^{m_0}\check{L}^i\|_{s,u}+\|R^{m_0}\check{\varrho}\|_{s,u}
+\|R^{m_0}\varphi\|_{s,u}+\delta\|TR^{m_0}\varphi\|_{s,u}\Bigr)\nonumber\\
&+\delta^{(k+2)(1-\varepsilon_0)}\bigl(\|R^{m_{-1}}x\|_{s,u}+\|R^{m_0}\mu\|_{s,u}\bigr)
+\delta^{k(1-\varepsilon_0)+1}\|\mu\slashed{d}R^{m_0}\varphi\|_{s,u}\nonumber\\
&+\delta\|\mu\mathring{L}R^{m_0}\varphi\|_{s,u}
+\delta^{(k+2)(1-\varepsilon_0)+1}\|\slashed{\mathcal{L}}_{R}^{m_1}\,{}^{(R)}\!\slashed{\pi}_T\|_{s,u}
+\delta^{k(1-\varepsilon_0)}\|\slashed{d}R^{m_0}\phi\|_{s,u}\nonumber\\
&+\delta^{3-\varepsilon_0}\|\slashed{d}R^{m+1}\mu\|_{s,u}
+\delta^{(k+1)(1-\varepsilon_0)}(\delta+\delta^{(2k+1)(1-\varepsilon_0)})\|\slashed{\mathcal L}_{R}^{m_1}\leftidx{^{(R)}}{\slashed\pi}\|_{s,u}\nonumber\\
&+\delta^{2-\varepsilon_0}\|\slashed{\mathcal{L}}_{R}^{m_1}\,{}^{(R)}\!\slashed{\pi}_{\mathring{L}}\|_{s,u}
+\delta^{3-\varepsilon_0}\|\slashed{\mathcal{L}}_{R}^{m_1}\,{}^{(T)}\!\slashed{\pi}_{\mathring{L}}\|_{s,u}\nonumber\\
\lesssim&\ \delta^{(k+2)(1-\varepsilon_0)+1/2}+\delta^{(2k+1)(1-\varepsilon_0)}\Xi^2_{m+2}(s,u)+\delta\sqrt{\mathcal E_{1,\leq m+2}(s,u)}\nonumber\\
&+\delta^{(k+1)(1-\varepsilon_0)+1}\sqrt{\mathcal E_{2,\leq m+2}(s,u)}+\delta^{3-\varepsilon_0}\|R^m\slashed\triangle\mu\|_{s,u}+\delta^{k(1-\varepsilon_0)}\|\slashed{d}R^{m_0}\phi\|_{s,u},\nonumber
\end{align}
where $1\leq m_p\leq m+1-p$ with $p=-1,0,1$. If $\bar Z^{m+1}=\bar Z^{p_1}TR^{p_2}$ with $p_1+p_2=m$, then $\bar Z^{m+1}\mathring L\phi=\bar Z^{p_1}R^{p_2}T(\mathring L^\alpha\varphi_\alpha)+\bar Z^{p_1}[T,R^{p_2}]\mathring L\phi$. Hence, for any $\bar Z^{m+1}$ containing $l$ occurrences of the vector field $T$, one has
\begin{equation}\label{bZLp}
\begin{split}
\delta^l\|\bar Z^{m+1}\mathring L\phi\|_{s,u}
\lesssim & \ \delta^{(k+2)(1-\varepsilon_0)+1/2}+\delta^{(2k+1)(1-\varepsilon_0)}\Xi^2_{m+2}(s,u)+\delta\sqrt{\mathcal E_{1,\leq m+2}(s,u)}\\
&+\delta^{(k+1)(1-\varepsilon_0)+1}\sqrt{\mathcal E_{2,\leq m+2}(s,u)}+\delta^{3-\varepsilon_0}\|R^m\slashed\triangle\mu\|_{s,u}\\
&+\delta^{k(1-\varepsilon_0)}\delta^{l_0}\|\slashed{d}\bar Z^{m_0}\phi\|_{s,u}.
\end{split}
\end{equation}



In addition, due to $|\slashed{\mathcal L}_{\bar Z}^{m+1}\slashed d\phi|\lesssim |R\bar Z^{m+1}\phi|$, it suffices to
treat the $L^2$ norm of $R\bar Z^{m+1}\phi$. As before, for $\bar Z^{m+1}=R^{m+1}$, we have
\begin{equation}\label{Rp}
\begin{split}
\|R^{m+2}\phi\|_{s,u}
\lesssim & \ \delta\|R^{m+1}TR\phi\|_{s,u}+\delta^{3-\varepsilon_0}\|\slashed{\mathcal L}_{R}^{m_1}\leftidx{^{(R)}}{\slashed\pi}_T\|_{s,u}+\delta^{2k(1-\varepsilon_0)}\|R^{m_0}R\phi\|_{s,u}\\
&+\delta^{2k(1-\varepsilon_0)+2-\varepsilon_0}\|\slashed{\mathcal L}_{R}^{m_1}\leftidx{^{(R)}}{\slashed\pi}\|_{s,u}.
\end{split}
\end{equation}
Note that $TR\phi=\leftidx{^{(R)}}{\slashed\pi}_{TX}\slashed d^X\phi+(R\mu)\tilde T^i\varphi_i-\mu(Rg^{0i})\varphi_i-\mu(R\mathring L^i)\varphi_i+\mu \tilde T^iR\varphi_i$. It follows from \eqref{Rpi} and \eqref{deL} that
\begin{equation*}
\begin{split}
TR\phi=&\ \upsilon\slashed d^X\mu\slashed d_X\phi+\mu(G_{X\mathring L}^\gamma R\varphi_\gamma)\slashed d^X\phi+\mu(G_{X\tilde T}^\gamma R\varphi_\gamma)\slashed d^X\phi+\mu\epsilon_i^a\check T^ig_{ab}(\slashed d_Xx^b\slashed d^X\phi)\\
&-\frac{\mu}{\varrho}R\phi+\upsilon(G_{X\tilde T}^\gamma T\varphi_\gamma)\slashed d^X\phi-\frac12\upsilon\mu(G_{\tilde T\tilde T}^\gamma\slashed d^X\varphi_\gamma)\slashed d_X\phi+\mu(G_{\mathring L\tilde T}^\gamma R\varphi_\gamma)\tilde T^i\varphi_i\\
&+\frac12\mu(G_{\tilde T\tilde T}^\gamma R\varphi_\gamma)\tilde T^i\varphi_i+(R\mu)\tilde T^i\varphi_i-\mu(Rg^{0i})\varphi_i+\mu \tilde T^iR\varphi_i.
\end{split}
\end{equation*}
Thus,
\begin{align}
\|R^{m+2}\phi\|_{s,u}
\lesssim&\ \delta^{2-\varepsilon_0}\bigl\{\|\slashed dR^{m+1}\mu\|_{s,u}+\|R^{m_0}\mu\|_{s,u}\bigr\}+\delta^{3-\varepsilon_0}\|\slashed{\mathcal L}_{R}^{m_1}\leftidx{^{(R)}}{\slashed\pi}_T\|_{s,u}\nonumber\\
&+\delta\|R^{m_0}\varphi\|_{s,u}+\delta \|\mu\slashed dR^{m+1}\varphi\|_{s,u}+\delta^{2k(1-\varepsilon_0)+2}\|TR^{m+1}\varphi\|_{s,u}\nonumber\\
&+\delta^{(k+1)(1-\varepsilon_0)+1}\|R^{m_0}\upsilon\|_{s,u}+\delta^{(2k+1)(1-\varepsilon_0)+1}\|\slashed{\mathcal L}_{R}^{m_1}\leftidx{^{(R)}}{\slashed\pi}\|_{s,u}\nonumber\\
&+\delta^{2-\varepsilon_0}\sum_{i=1}^2\bigl(\|R^{m_0}\check L^i\|_{s,u}+\|R^{m_{-1}}x^i\|_{s,u}\bigr)\nonumber\\
\lesssim & \ \delta^{5/2-\varepsilon_0}+\delta\sqrt{\mathcal E_{1,\leq m+2}(s,u)}+\delta^2\sqrt{\mathcal E_{2,\leq m+2}(s,u)}\label{Rmp}\\
&+\delta^{2-\varepsilon _0}\|R^m\slashed\triangle\mu\|_{s,u}+\delta^{1+k(1-\varepsilon_0)}\Xi^2_{m+2}(s,u).\nonumber
\end{align}
For $\bar Z^{m+1}=\bar Z^{p_1}TR^{p_2}$ with $p_1+p_2=m$, due to $R\bar Z^{m+1}\phi=R\bar Z^{p_1}R^{p_2}(\mu \tilde T^i\varphi_i)+R\bar Z^{p_1}[T,R^{p_2}]\phi$, one has
\begin{equation}\label{DXY-8}
\begin{split}
\delta^l\|R\bar Z^{m+1}\phi\|_{s,u}
\lesssim & \ \delta^{3-\varepsilon_0}\delta^{l_1}\|\slashed{\mathcal L}_{\bar Z}^{m_1}\leftidx{^{(R)}}{\slashed\pi}_T\|_{s,u}+\delta^{2k(1-\varepsilon_0)}\delta^{l_0}\|\bar Z^{m_0}\phi\|_{s,u}+\delta^{(2k+1)(1-\varepsilon_0)+1}\delta^{l_1}\|\slashed{\mathcal L}_{\bar Z}^{m_1}\slashed g\|_{s,u}\\
&+\delta^{2-\varepsilon_0}\delta^{l_0}\bigl(\|\bar Z^{m_0}\mu\|_{s,u}+\sum_{i=1}^2(\|\bar Z^{m_0}\check L^i\|_{s,u}+\|\bar Z^{m_0}x^i\|_{s,u})\bigr)+\delta\delta^{l_0}\|\bar Z^{m_0}\varphi\|_{s,u}.
\end{split}
\end{equation}
Note that the terms on the right-hand side in \eqref{DXY-8} are exactly the same as the ones in \eqref{lowerphi}. Therefore,
we can obtain \eqref{Zdphi}, and then \eqref{ZLphi} follows by substituting \eqref{Zdphi} into \eqref{bZLp}.



Analogously, owing to
\[
\|R^m\mathring L^2\phi\|_{s,u}
\lesssim \delta\bigl\|[T,R^m\mathring L]\mathring L\phi\bigr\|_{s,u}
+\delta\bigl\|R^m\mathring L T\bigl(\mathring L^\alpha\varphi_\alpha\bigr)\bigr\|_{s,u},
\]
one deduces from \eqref{TLp} that
\begin{align*}
&\|R^m\mathring L^2\phi\|_{s,u}\\
\lesssim &\delta^{1-\varepsilon_k}\biggl(
\sum_{i=1}^2\bigl(\delta^{-\varepsilon_k}\|R^{m_0}\check L^i\|_{s,u}
+\|R^{m_1}\mathring L\check L^i\|_{s,u}\bigr)
+\delta^{-\varepsilon_k}\|R^{m_0}\check\varrho\|_{s,u}
+\|R^{m_1}\mathring L\check\varrho\|_{s,u}
\biggr)\\
&+\delta^{(2k+2)(1-\varepsilon_0)-\varepsilon_0}\|R^{m_{0}}x\|_{s,u}
+\delta^{3-\varepsilon_0-\varepsilon_k}
\bigl\|\slashed{\mathcal L}_{R}^{m_1} \leftidx{^{(T)}}{\slashed\pi}_{\mathring L}\bigr\|_{s,u}+\delta^{3-\varepsilon_0-2\varepsilon_k}
\bigl\|\slashed{\mathcal L}_{R}^{m_1} \leftidx{^{(R)}}{\slashed\pi}_T\bigr\|_{s,u}\\
&+\delta^{k(1-\varepsilon_0)}\|R^{m_0}\mathring L\phi\|_{s,u}+\delta^{k(1-\varepsilon_0)-\varepsilon_k}\|R^{m_0}\phi\|_{s,u}
+\delta^{3-\varepsilon_0}\bigl\|\slashed d R^{m}\mathring L\mu\bigr\|_{s,u}\\
&+\delta^{2-\varepsilon_0-\varepsilon_k}\|R^{m_1}\mathring L\mu\|_{s,u}
+\delta^{2-\varepsilon_0-2\varepsilon_k}\|R^{m_0}\mu\|_{s,u}+\delta^{2-\varepsilon_0-\varepsilon_k}
\bigl\|\slashed{\mathcal L}_{R}^{m_1} \leftidx{^{(R)}}{\slashed\pi}_{\mathring L}\bigr\|_{s,u}\\
&+\delta^{1-\varepsilon_k}\sqrt{\mathcal E_{1,\leq m+2}(s,u)}+\delta^{2(k+1)(1-\varepsilon_0)} \sqrt{\mathcal E_{2,\leq m+2}(s,u)}\\
\lesssim & \delta^{5/2-\varepsilon_0-2\varepsilon_k}
+\delta^{2-\varepsilon_0+2k(1-\varepsilon_0)}\|R^m\slashed\triangle\mu\|_{s,u}+\delta^{1-\varepsilon_k}\sqrt{\mathcal E_{1,\leq m+2}(s,u)}\\
&+\delta^{2(k+1)(1-\varepsilon_0)} \sqrt{\mathcal E_{2,\leq m+2}(s,u)}+\delta^{3k(1-\varepsilon_0)-2\varepsilon_0+1}\Xi_{m+2}^2(s,u).
\end{align*}


Meanwhile, it follows from \eqref{gLLL} that
\begin{align*}
\delta^l\bigl\lVert\bar{Z}^{m+1}\bigl(G_{\mathring{L}\mathring{L}}^\gamma \mathring{L}\gamma\bigr)\bigr\rVert
&\lesssim \delta^{(k-1)(1-\varepsilon_0)}\delta^{l_0}\bigl(
\lVert\bar{Z}^{m_0}\varphi\rVert{s,u}
\sum_{i=1}^2\lVert\bar{Z}^{m_0}\check{L}^i\rVert_{s,u}
\lVert\bar{Z}^{m_0}\check{\varrho}\rVert_{s,u}
\bigr) \\
&\quad + \delta^{k(1-\varepsilon_0)}\delta^{l_{-1}}\lVert\bar{Z}^{m_{-1}}x\rVert_{s,u}
\delta^{(k-2)(1-\varepsilon_0)}\delta^{l_0}\bigl(
\lVert\bar{Z}^{m_0}\mathring{L}\phi\rVert_{s,u}
\lVert\slashed{d}\bar{Z}^{m_0}\phi\rVert_{s,u}
\bigr).
\end{align*}
Together with Proposition \ref{L2chi} and the estimates \eqref{ZLphi}--\eqref{Zdphi}, this yields \eqref{ZGL}.
In addition, \eqref{ZmLphi} follows from \eqref{ZLphi} and Proposition \ref{L2chi}.
