\label{prop:DZ}
Let $R\ge1$, let $X$ and $\nu_{X,\alpha}$ be as above, and let $h\in L^2(\mathbb{R}^n)$ with $\operatorname{supp}\widehat h\subset B^n_2(0)$. Then, for every $\varepsilon>0$,
                            
\begin{equation}\label{eq:DZ}
\lVert e^{it\Delta}h\rVert_{L^2(X)}\ \lesssim_\varepsilon\ \nu_{X,\alpha}^{\frac1{n+1}}\,(R)^{\frac{\alpha}{2(n+1)}+\varepsilon}\,\lVert h\rVert_2 .
\end{equation}

<!-- theorem-1: body_md -->
\label{thm:main}
Given $\Theta\subset \mathbb{R}^n$ compact, $\beta=\dim_{M}(\Theta)$ and $\gamma:\mathbb{R}^n\times[0,1)\times\Theta\to\mathbb{R}^n$, with $\gamma(x,0,\theta)=x$, satisfying \eqref{C1},\eqref{C2} and \eqref{C3} we have
\begin{equation}\label{eq:main}
\lVert M_{\Theta} f\rVert_{L^2(B^n_1)}\ \lesssim_s\ \lVert f\rVert_{H^s(\mathbb{R}^n)},
\end{equation}
for all $f\in H^s(\mathbb{R}^n)$ whenever
\begin{equation}\label{eq:thresholds}
s\ >\sigma:= \
\begin{cases}
	\dfrac{n(\beta+1)}{2(n+1)}, & 0\le\beta\le n-1,\\[2mm]
	\dfrac{\beta}{2}+\dfrac{1}{2(n+1)}, & n-1<\beta\le n-\dfrac{1}{n+1},\\[2mm]
	\dfrac n2, & n-\dfrac{1}{n+1}<\beta\le n.
\end{cases}
\end{equation}
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
Given $\Theta\subset \mathbb{R}^n$ compact, $\beta=\dim_{M}(\Theta)$ and $\gamma:\mathbb{R}^n\times[0,1)\times\Theta\to\mathbb{R}^n$, with $\gamma(x,0,\theta)=x$, satisfying \eqref{C1},\eqref{C2} and \eqref{C3}, we have
\begin{equation}\label{eq:aeconverg}
\lim _{\substack{(y, t) \rightarrow(x, 0) \\ y \in \Gamma_{x,t},\ t>0}} e^{i t \Delta} f(y)=f(x) \;\text { a.e. } x\in\mathbb{R}^n
\end{equation}
for all $f\in H^s(\mathbb{R}^n)$ with $s>\sigma$ and $\sigma$ as defined in \eqref{eq:thresholds}.

<!-- remark-1: body_md -->
\label{rem:sanity}
The threshold $\sigma$ matches some cases where an answer is independently known:
\begin{itemize}
\item[(i)] At $n=1$, the best known value is $\tfrac{\beta+1}{4}$, the theorem of Cho--Lee--Vargas \cite{CLV} (see also Shiraki \cite{Shi} for the fractional Schrödinger operator), conjectured sharp for $\beta\in(0,1)$ and known to be sharp for $\beta\in\{0,1\}$. Note that $\tfrac{n(\beta+1)}{2(n+1)} = \tfrac{\beta+1}{4}$ at $n=1$. However our $\sigma$ only takes that value at $\beta\in\{0,1\}$.
\item[(ii)] At $\beta=0$, when there is a single curve, $\sigma=\tfrac{n}{2(n+1)}$ is the sharp threshold for the almost everywhere convergence problem, by Du--Guth--Li \cite{DGL}, Du--Zhang \cite{DZ} and the counterexample of Bourgain \cite{Bou}.
\item[(iii)] At $n=2$, the best known value is $\tfrac{\beta+1}{3}$ obtained by Li--Wang--Yan \cite{LWY}. Our $\sigma =\tfrac{n(\beta+1)}{2(n+1)}= \tfrac{\beta+1}{3}$ for all $\beta\in[0,1]\cup\{2\}$ and $\sigma>\tfrac{\beta+1}{3}$ for all $\beta\in(1,2)$.
\item[(iv)] At $\beta=n$, the sufficiency threshold $s>\tfrac n2$ is given by Sobolev embedding. Convergence on a cone of directions requires $s> \frac n2$ by Sjögren--Sjölin \cite{SS}. Our argument in Section \refeq{sec:highbeta} allows us to reach the sufficiency of $s>\tfrac n2+\tfrac1{2(n+1)}$, not lower.
                                                                                                                                                                                                                                                                                                       
\end{itemize}

<!-- proposition-2: body_md -->
\label{prop:lcp}
Let $c\ge1$ and $K\ge1$, and let $f\in L^2(\mathbb{R}^n)$ with $\operatorname{supp}\widehat f\subset B^n_c(0)$. For $\mathfrak l\in\mathbb{Z}^n$ write
\begin{equation}\label{eq:htrans}
\widehat{f_{\mathfrak l}}(\xi)=e^{2\pi i\mathfrak l\cdot\xi}\,\widehat f(\xi).
\end{equation}
 Let $(x,t)\in\mathbb{R}^{n+1}$ and let $\mathcal Q\subset\mathbb{R}^{n+1}$ be a unit cube such that $\lvert (x,t)-(x',t')\rvert\le K$ for all $(x',t')\in\mathcal Q$. Then, for every $p\ge1$,
\begin{equation}\label{eq:lcp-cubes}
\bigl|e^{it\Delta}f(x)\bigr|^{p}\ 
                
\lesssim\
 \sum_{\mathfrak l\in\mathbb{Z}^{n}}\frac{1}{(1+\lvert \mathfrak l\rvert)^{n+1}}\int_{\mathcal Q}\bigl|e^{is\Delta}f_{\mathfrak l}(y)\bigr|^{p}\,dy\,ds ,
\end{equation}
where the implicit constant may depend on $c$ and $K$.

<!-- corollary-2: body_md -->
\label{cor:lcp}
Let $h\in L^2(\mathbb{R}^n)$ with $\operatorname{supp} \hat h \subset A(1)+B^n_{1/100}$. Let $E\subset\mathbb{R}^{n+1}$ be non-empty. Then there is a lattice unit cube $\mathcal Q\subset\mathbb{R}^{n+1}$, containing a point of $E$, such that
\begin{equation}\label{eq:lcp-sup}
\sup_{(z,t)\in E}\lvert e^{it\Delta}h(z)\rvert^2\ \lesssim\ \sum_{\mathfrak l\in\mathbb{Z}^{n}}\frac{1}{(1+\lvert \mathfrak l\rvert)^{n+1}}\,\lVert e^{i\cdot\Delta}h_{\mathfrak l}\rVert_{L^2(\mathcal Q)}^2 .
\end{equation}
<!-- end corollary-2 -->
<!-- proof-1: body_md -->
We observe that $e^{it\Delta}h(\cdot)$ is bounded. Indeed, since $\widehat h$ is supported on a bounded set, Cauchy--Schwarz and Plancherel give
\begin{equation}\label{eq:L1L2}
	\lVert \widehat h\rVert_{1}\ \le\ \lvert \operatorname{supp}\widehat h\rvert^{1/2}\,\lVert \widehat h\rVert_{2}\ \lesssim\ \lVert h\rVert_2 ,
\end{equation}
so $\widehat h\in L^1$ and $\lvert e^{it\Delta}h(\cdot)\rvert\lesssim\lVert \widehat h\rVert_{L^1}\lesssim\lVert h\rVert_2$ uniformly in $t$.
	
Since $e^{it\Delta}h(\cdot)$ is bounded, the supremum in \eqref{eq:lcp-sup} is finite, so we may pick $a_*=(x_*,t_*)\in E$ with
\begin{equation}\label{eq:halfsup}
\bigl|e^{it_*\Delta}h(x_*)\bigr|^2\ \ge\ \dfrac12\sup_{(z,t)\in E}\lvert e^{it\Delta}h(z)\rvert^2.
\end{equation}
Let $\mathcal Q$ be a lattice unit cube containing $a_*$. Every point of $\mathcal Q$ lies at distance $\sqrt{n+1}$ or less from $a_*$. Notice that $A(1)+B^n_{1/100}\subset B^n_{2}+B^n_{1/100}$. We apply Proposition~\ref{prop:lcp} with $p=2$, $c=2+\frac1{100}$, and $K=\sqrt{n+1}$, and obtain \eqref{eq:lcp-sup}.

<!-- lemma-3: body_md -->
\label{lem:homeo}
Each $\Psi_\theta$ is a homeomorphism of $\mathbb{R}^n$ onto $\mathbb{R}^n$. In particular $\Psi_\theta^{-1}$ is defined on all of $\mathbb{R}^n$ and is $C_1$-Lipschitz.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Write $\Psi=\Psi_\theta$ and recall $C_1^{-1}\lvert x-x'\rvert\le\lvert \Psi(x)-\Psi(x')\rvert\le C_1\lvert x-x'\rvert$.

We first prove that $\Psi$ is a homeomorphism onto its image. The lower bound in \eqref{C1R} gives injectivity. The lower bound also implies the $C_1$-Lipschitz continuity of $\Psi^{-1}: \Psi(\mathbb{R}^n)\to \mathbb{R}^n$. Indeed, if $\Psi(x_k){\to}\Psi(x)$ as $k\to\infty$, then $\lvert \Psi^{-1}(\Psi(x_k))-\Psi^{-1}(\Psi(x))\rvert=\lvert x_k-x\rvert\le C_1\lvert \Psi(x_k)-\Psi(x)\rvert\to0$ as $k\to\infty$. With the continuity of $\Psi$ this proves the claim.

We now check that the image is open. By the Invariance of Domain Theorem (see Corollary 19.9 in \cite{Bredon}), a continuous injective map $\mathbb{R}^n\to\mathbb{R}^n$ is open, so $\Psi(\mathbb{R}^n)$ is open.

We now prove that the image is closed. If $K\subset\mathbb{R}^n$ is compact, then $\Psi^{-1}(K)$ is the image of the bounded set $K\cap\Psi(\mathbb{R}^n)$ under the $C_1$-Lipschitz map $\Psi^{-1}$, therefore $\Psi^{-1}(K)$ is bounded. It is also closed, by the continuity of $\Psi$, so it is compact. Since $K$ was arbitrary, we deduce that $\Psi:\mathbb{R}^n\to\mathbb{R}^n$ is a proper map. Since a proper continuous map between locally compact Hausdorff spaces is closed (see Proposition 11.5 in \cite{Bredon}), we deduce that $\Psi(\mathbb{R}^n)$ is closed in $\mathbb{R}^n$.

We conclude. The set $\Psi(\mathbb{R}^n)$ is nonempty, open and closed in the connected space $\mathbb{R}^n$, therefore it is equal to $\mathbb{R}^n$.

<!-- lemma-5: body_md -->
\label{lem:cover}
For every $\varepsilon>0$ there exists $C_\varepsilon>0$ such that for all $1\le r\le C_1C_3R$,
\begin{equation}\label{eq:cover}
N\bigl(\Phi_v(\Theta),r\bigr)\ \lesssim\ C_\varepsilon\Bigl(\frac R r\Bigr)^{\beta+\varepsilon}.

\begin{equation}\label{eq:covpush}
N(\Phi_v(\Theta),r)\ \le\ N\Big(\Theta,\frac{r}{C_0R}\Big)
=N(\Theta,\delta)
                                      
\end{equation}
It remains to bound $N(\Theta,\delta)$ for all $\delta\in(0,1]$. Then, for all $\delta\in(0,1]$,
\begin{equation}\label{eq:covtheta}
N(\Theta,\delta)\le C_\varepsilon\,\delta^{-\beta-\varepsilon},
\end{equation}
and \eqref{eq:cover} follows.

Let $j$ be a base point from the set above and let $p\in(\mathcal E_j(\tilde t)+B^n_{c})\cap \pi(Q)$.
	Notice $\pi(Q)= \pi\big(Q^{n+1}_1(w)\big)=Q^n_1(v)$.
	Then $p\in Q_1^n(v)$, so $\lvert p-v\rvert\le\sqrt n$, and $p\in\mathcal E_j(\tilde t)+B^n_c$, so there is $\theta\in\Theta$ with $\lvert \Psi_\theta(j)-p\rvert<c$. Hence $\Psi_\theta(j)\in v+B^n_{c'}$ with $c':=c+\sqrt n$. Applying the Lipschitz bound for $\Psi_\theta^{-1}$ (Lemma~\ref{lem:homeo}),
                                             
	\begin{equation}\label{eq:multkey}
		\lvert j-\Phi_v(\theta)\rvert=\bigl|\Psi_\theta^{-1}(\Psi_\theta(j))-\Psi_\theta^{-1}(v)\bigr|\ \le\ C_1\lvert \Psi_\theta(j)-v\rvert\ \le\ C_1c' .
	\end{equation}
	So every such $j$ lies in $\Phi_v(\Theta)+B^n_{C_1c'}$. This $C_1c'$-neighborhood is covered by at most a quantity of unit balls comparable to $N(\Phi_v(\Theta),1)$, each ball containing at most $3^n$ lattice points. This proves the first inequality in \eqref{eq:mult}. Lemma~\ref{lem:cover} at $r=1$ gives the second inequality in \eqref{eq:mult}.

<!-- lemma-7: body_md -->
\label{lem:pack}
Let $c\ge1$ and $C\ge1$, fix $v\in\mathbb{R}^n$, and let $1\le r\le CR$. Then
\begin{equation}\label{eq:pack}
\#\bigl\{j\in\mathbb{Z}^n:\ (\mathcal E_j(\tilde t)+B^n_{c})\cap B^n_r(v)\ne\emptyset\bigr\}\ \lesssim_{\varepsilon,c,C}\ R^{\beta+\varepsilon}\,r^{\,n-\beta} .
\end{equation}
<!-- end lemma-7 -->
<!-- proof-7: body_md -->
Apply the Lipschitz bound as in Lemma~\ref{lem:mult}. If $j$ is a base point from the set above, a point $p\in(\mathcal E_j(\tilde t)+B^n_{c})\cap B^n_r(v)$ satisfies $\lvert p-v\rvert<r$ and $\lvert \Psi_\theta(j)-p\rvert<c$ for some $\theta\in\Theta$. Therefore $\Psi_\theta(j)\in v+B^n_{r+c}$, hence, $j$ lies in $\Phi_{v}(\Theta)+B^n_{C_1(r+c)}$.

Write $C_0=C_1C_3$. Suppose first that $1\le r\le C_0R$. By Lemma~\ref{lem:cover}, cover $\Phi_{v}(\Theta)$ by $N(\Phi_{v}(\Theta),r)\lesssim_\varepsilon(\frac{R}{r})^{\beta+\varepsilon}$ balls of radius $r$ and pack at most about $r^n$ lattice points into each ball, which gives \eqref{eq:pack}. Suppose now that $C_0R\le r\le CR$, that is $r\sim R$. Since $\Phi_v$ is $(C_0R)$-Lipschitz on the bounded set $\Theta$, the set $\Phi_{v}(\Theta)$ has diameter at most about $r$, so it can be covered by a number of balls of radius $r$ only depending on the diameter of $\Theta$. Therefore $N(\Phi_{v}(\Theta),r)\lesssim 1$. The count of lattice points in each ball is at most about $r^n$. Combine this with $r^\beta\le R^{\beta+\varepsilon}$ to obtain \eqref{eq:pack}.
