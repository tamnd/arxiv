\label{variations}
More generally, whenever $v \in \operatorname{Lip} \cap W^{2,1}(B_1,\mathbb{R}^n)$ and $F$ is a smooth function on $\mathbb{R}^{n \times m}$, we have the implication 
\begin{equation}\label{eq:sys0}
\partial_i(F_{p^l_i}(Dv)) = H^l,\, l =1,\dots, n \quad \Longrightarrow \quad \partial_i(F(Dv)\delta_{ij} - F_{p^l_i}(Dv)v^l_j) = -H^lv^l_j, \, j =1,\dots, m.
\end{equation}
Here $F_{p^l_i}$ denotes the derivative of $F$ in the $p^l_i$ direction. The system on the left reflects that $v$ is a critical point of the energy 
\begin{equation}\label{ene}
\text{E}(v) = \int_{B_1} F(Dv) + (H,v)dx
\end{equation}
with respect to outer variations (i.e., $\text{E}(v + t\varphi) = \text{E}(v) + o(t)$ for all maps $\varphi \in C^{\infty}_c(B_1,\mathbb{R}^n)$), and the system on the right reflects that $v$ is a critical point of $\text{E}$ with respect to domain variations (i.e., $\text{E}(v(x + tV)) = \text{E}(v) + o(t)$ for all vector fields $V \in C^{\infty}_c(B_1,\mathbb{R}^m)$).\ If $D^2F > 0$ and $H \in L^2_{\operatorname{loc}}$, then Lipschitz solutions to the system on the left hand side of \eqref{eq:sys0} are in $W^{2,2}_{\operatorname{loc}}$ by standard regularity theory, and hence solve the system on the right hand side of \eqref{eq:sys0}. In this paper we are considering the area integrand $F(p) = \sqrt{\det(I + p^Tp)}$. This integrand is polyconvex (a convex function of the sub-determinants of $p$), and it is convex if and only if $\min\{m,\,n\} = 1$.

<!-- def-1: body_md -->
Let $\Omega \subset \mathbb{R}^2$ be open. The map $\varphi\in W^{1,2}(\Omega,\mathbb{R}^2)$ is $(K-)$quasiregular if there exists $K \ge 1$ such that
\begin{equation}\label{Kqc1}
|D\varphi|^2(x) \le K\det(D\varphi(x)), \quad \text{for a.e. $x \in \Omega$.}
\end{equation}
We say that it is $(K-)$quasiconformal if, in addition, $\varphi$ is a homeomorphism\footnote{Note that every quasiregular map is continuous, compare Theorem \ref{thm:quas}\eqref{quasma4}.} between $\Omega$ and the open set $\varphi(\Omega)$.

\label{thm:quas}
Let $\varphi \in W^{1,2}(\Omega_1,\Omega_2)$ be quasiregular and $\psi \in W^{1,2}(\Omega_2,\Omega_3)$ be quasiconformal with $\Omega_3 = \psi(\Omega_2)$. Then:
\begin{enumerate}
\item\label{quasma1} $\psi^{-1}: \Omega_3 \to \Omega_2$ is a $W^{1,2}(\Omega_3,\Omega_2)$ quasiconformal mapping (with the same quasiregularity constant $K$ of $\psi$, compare \eqref{Kqc1}), with gradient
\[
D(\psi^{-1}(y)) = (D\psi)^{-1}(\psi^{-1}(y)), \text{ for a.e. } y \in \Omega_3;
\]
\item\label{quasma2} the chain rule holds: for every $f \in W^{1,2}(\Omega_2)$, $f\circ \varphi \in W^{1,2}(\Omega_1)$ with
\[
D(f\circ\varphi)(x) = Df(\varphi(x))D\varphi(x), \text{ for a.e. } x \in \Omega_1;
\]
\item\label{quasma3} the composition $\psi \circ \varphi$ is quasiregular;
\item\label{quasma4} there exists $p = p(K) > 2$ such that $\varphi \in W^{1,p}_{\operatorname{loc}}(\Omega_1,\Omega_2)$, if $K$ is the quasiregularity constant of \eqref{Kqc1}, with bounds depending solely on the $W^{1,2}$ norm of $\varphi$;
\item\label{quasma5} $\det D\psi\neq 0$ a.e..
\end{enumerate}

\begin{equation}\label{sptnull}
|X|^2 \le K \det X, |Y|^2 \le K\det Y\; \Longrightarrow\; |X-Y|^2 \le C\det(X-Y) +C\frac{|H(X)-H(Y)|^2}{\max\{|X|,|Y|\}^2}.

The constant $C$ appearing in the next computations will always depend solely on $K$ and can change line-by-line. We start by noticing that, by symmetry, we can assume that $|Y|\le |X|$ and, in addition, that
\begin{equation}\label{YlessX} 
0 <|Y| \le |X|.
\end{equation}
Indeed if not $Y = 0$ and \eqref{sptnull} is true simply by quasiconformality of $X$, up to taking $C \ge K$.

We start by reducing the proof of \eqref{sptnull} to the proof of a similar inequality. To state it, we need to use the complex coordinates introduced in Section \ref{s:cd}. For simplicity, let us denote 
\[
a \doteq [X]_{\mathcal{H}},\;a' \doteq [X]_{\overline{\mathcal{H}}},\; b \doteq [Y]_{\mathcal{H}} ,\;b' \doteq  [Y]_{\overline{\mathcal{H}}}.
\] 
We claim that there exists $C = C(K)> 0$ and $\lambda = \lambda(K) \in [0,1)$ such that
\begin{equation}\label{totalc}
|a'-b'|^2 \le \lambda|a-b|^2 + C\frac{|\overline{a}a' - \overline{b}b'|^2}{|a|^2}.
\end{equation}
Let us first show how the claim implies the Lemma. Notice that
\begin{equation}\label{qccomp1}
\max\{|X|^2,|Y|^2\} = |X|^2 \overset{\eqref{alg}}{=} 2|a|^2 + 2|a'|^2 \overset{\eqref{qccomp}}{\le} C|a|^2.
\end{equation}
We fix $T = T(K)> 0$ large enough so that
\begin{equation}\label{T}
(1-T) +\lambda(1+T) < 0.
\end{equation}
We can do so since $\lambda = \lambda(K)$ appearing in \eqref{totalc} belongs to $[0,1)$. We can then compute:
\begin{align*}
2^{-1}|X-Y|^2 &\overset{\eqref{alg}}{=} |a - b|^2 +|a'-b'|^2 \\
&\overset{\eqref{alg}}{=} (1-T)|a-b|^2 +(1+T)|a'-b'|^2 + T\det(X-Y) \\
&\overset{\eqref{totalc}}{\le} (1-T)|a-b|^2 +\lambda(1+T)|a-b|^2 + T\det(X-Y) + (1+T)C\frac{|\overline{a}a' -\overline{b}b'|^2}{|a|^2} \\
&\overset{\eqref{T}}{\le} T\det(X-Y) + (1+T)C\frac{|\overline{a}a' - \overline{b}b'|^2}{|a|^2} \overset{\eqref{eqH}-\eqref{qccomp1}}{\le} T\det(X-Y) + C\frac{|H(X) - H(Y)|^2}{\max\{|X|,|Y|\}^2}
\end{align*}
This shows the validity of \eqref{sptnull} assuming \eqref{totalc}. Now, as a final step, we show \eqref{totalc}. Using the fact that $Y \neq 0$, \eqref{YlessX}, we can estimate for an $\varepsilon = \varepsilon(K) > 0$ to be fixed later,
\begin{equation}\label{a'b'}
\begin{split}
|a' - b'|^2  &= \left|\frac{a'\overline{a}}{\overline{a}} -\frac{b'\overline{b}}{\overline{b}}\right|^2 = \left|\frac{a'\overline{a} -b'\overline{b}}{\overline{a}} 
+ \left(\frac{1}{\overline{a}}-\frac{1}{\overline{b}}\right)b'\overline{b}\right|^2 \\
&\le \left(1+\frac{1}{\varepsilon^2}\right)\left|\frac{a'\overline{a} -b'\overline{b}}{\overline{a}}\right|^2 +(1 +\varepsilon^2)|a-b|^2\frac{|b'|^2}{|a|^2}.
\end{split}
\end{equation}
By \eqref{YlessX} and \eqref{alg}, we have that $|b'|^2 + |b|^2 \le |a'|^2 + |a|^2$. Using \eqref{qccomp} in this inequality, we find
\begin{equation}\label{kappa}
(1 + \kappa^{-1})|b'|^2 \le (1+\kappa)|a|^2\; \Rightarrow\; |b'|^2 \le \kappa |a|^2.
\end{equation}
Let us remark once again that $\kappa = \kappa(K) \in (0,1)$. We can then find $\varepsilon = \varepsilon(K) > 0$ such that $\lambda \doteq (1+ \varepsilon^2)\kappa < 1$. For any such $\varepsilon>0$, we see that \eqref{a'b'} implies \eqref{totalc} through \eqref{kappa} and we conclude the proof.

\label{thm:regv}
Let $r > 2$ and let $v \in W^{1,2}(B_1,\mathbb{R}^n)$, $\psi \in W^{1,2}(B_1,\mathbb{R}^2)$, $f \in L^r(B_1,\mathbb{R}^n)$. Assume $\psi$ is quasiconformal and that the maps are linked through \eqref{eq:gainfi}, \eqref{eq:yU} and \eqref{eq:compara} for all $p > 2$, namely:
\begin{numcases}
\displaystyle D\psi \in L^p_{\operatorname{loc}}(B_1,\mathbb{R}^2) \Rightarrow \Delta v \in L_{\operatorname{loc}}^{\beta}(B_1,\mathbb{R}^n), \quad \text{for } \beta = \beta(p) = \frac{r'p}{(r'-1)p + 2}, \label{feB}\\
\displaystyle H(D\psi) = -H(Dv) \text{ a.e. in }B_1, \label{seB}\\
\displaystyle |Dv(y)| \le C|D\psi(y)| \text{ a.e. in }B_1.\label{thB}
\end{numcases}
Then, there exists $\varepsilon > 0$ such that $v \in W^{2,2+\varepsilon}_{\operatorname{loc}}(B_1,\mathbb{R}^n)$.
<!-- end teo-4 -->
<!-- proof-2: body_md -->
The proof is based on an iterative argument.\ Namely, starting with a given regularity  $D\psi \in L^{p_i}_{\operatorname{loc}}(B_1,\mathbb{R}^2)$ for some $p_i > 2$, we can employ \eqref{feB} to deduce higher regularity of $v$. From \eqref{seB}, Lemma \ref{lem:dett} and a few additional observations we can then infer $D\psi \in L^{p_{i + 1}}_{\operatorname{loc}}(B_1,\mathbb{R}^2)$ with $p_{i + 1} > p_i$, compare \eqref{eq:p}. The goal is to show that, for sufficiently large $i$, $\beta(p_i) > 2$ and conclude the assertion through \eqref{feB} and standard Calder\'on-Zygmund theory. The iterative argument is hidden in the definition of $\sigma$ of \eqref{sup}, but it is the underlying idea of the proof. We start by describing one step of the iteration.

\medskip

Assume $D\psi \in L^p_{\operatorname{loc}}(B_1,\mathbb{R}^2)$ for some $p > 2$. The existence of a starting $p$ for which this is true is guaranteed by Theorem \ref{thm:quas}\eqref{quasma4}. We notice preliminarily that
\begin{equation}\label{bdend}
p > \frac{4}{2-r'}\quad \Longrightarrow\quad \beta(p) > 2.
\end{equation}
Thus we can, without loss of generality, assume that $2 < p <\frac{4}{2-r'}$.

\medskip

Consider $\eta \in C^{\infty}_c(B_1)$ with $\eta \equiv 1$ on $B_{\rho'}$ and $\operatorname{spt}(\eta)\subset B_\rho \subset B_1$.\ We employ \eqref{sptnull} with $X = D\psi(x+h)$ and $Y = D\psi(x)$, where $|h| <(1-\rho)/2$.\ For any map $f$, we set $f_h(x) \doteq f(x+h)-f(x)$.\ Since the determinant is a Null Lagrangian (see \cite[Theorem 2.3(ii)]{DMU}):
\begin{equation}\label{eq:routine0}
0 = \int_{B_1}\det(D(\eta(x)\psi_h(x)))dx = \int_{B_1}\eta^2\det(D\psi_h)dx + 2\int_{B_1}\eta\langle\psi_h\otimes D\eta, \operatorname{cof}^T(D\psi_h)\rangle dx.
\end{equation}
Using \eqref{sptnull} and routine estimates, see for instance \cite[Proposition 6.6]{RT}, this leads to
\begin{equation}\label{eq:routine}
\int_{B_1}\eta^2|D\psi_h|^2dx \le C\int_{B_1}|\eta||\psi_h||D\eta||D\psi_h| dx + C\int_{B_1}\eta^2\frac{|H(D\psi(x+h))-H(D\psi(x))|^2}{\max\{|D\psi(x+h)|,|D\psi(x)|\}^2}dx.
\end{equation}
In what follows, $C$ is a constant which depends on $p$ and on $\rho$ and is independent of $h$, and will change line-by-line. We observe that every component of $H(X)$ is a quadratic form, compare \eqref{defhopf}. Thus:
\begin{equation*}\label{eq:quadra}
\begin{split}
|H(D\psi(x+h))-H(D\psi(x))|^2& \overset{\eqref{seB}}{=} |H(Dv(x+h))-H(Dv(x))|^2 \\
&\le C\max\{|Dv|(x+h),|Dv|(x)\}^2|Dv_h|^2(x)\\
&\overset{\eqref{thB}}{\le} C\max\{|D\psi|(x+h),|D\psi|(x)\}^2|Dv_h|^2(x),
\end{split}
\end{equation*}
where the first inequality is true precisely due to the fact that $H$ is a quadratic form.\ This and Young's inequality allow us to estimate \eqref{eq:routine} as
   
                                                                                                                 
   
                                  
\begin{equation}\label{vpsi}
\int_{B_1}\eta^2|D\psi_h|^2dx \le C\int_{B_1}|D\eta|^2|\psi_h|^2 dx + C\int_{B_1}\eta^2|Dv_h|^2(x)dx.
\end{equation}
As $\psi \in W^{1,2}(B_1,\mathbb{R}^2)$, we can bound the first addendum simply by:
\begin{equation}\label{eq:routine1}
\int_{B_1}\eta^2|D\psi_h|^2dx \le C|h|^2 + C\int_{B_1}\eta^2|Dv_h|^2(x)dx.
\end{equation}
Estimating the second addendum requires more work.\ By \eqref{feB}, $v \in W_{\operatorname{loc}}^{2,\beta}(B_1,\mathbb{R}^n)$. Set $f^h(x) \doteq \eta(x)|Dv_h|(x)$ and choose
                                                                                                                                                
                  
$\theta \in (0,1)$ as
\begin{equation}\label{theta}
	\theta \doteq \frac{4}{\beta}\left[1-\frac{1}{\beta}\right].
\end{equation}
We have $2= \theta \beta + (1-\theta)\beta^*$ and, using H\"older's inequality, we get:
\begin{equation}\label{fh}
\int_{B_1}|f^h|^2(x) dx \le \left(\int_{B_1}|f^h|^\beta(x) dx\right)^{\theta}\left(\int_{B_1}|f^h|^{\beta^*}(x) dx\right)^{1-\theta} \le C\left(\int_{B_1}|f^h|^\beta(x) dx\right)^{\theta},
\end{equation}
where here
\[
C = \sup_{h \in B_{\frac{1-\rho}{2}}}\left(\int_{B_1}|f^h|^{\beta^*}(x) dx\right)^{1-\theta} < + \infty.
\]
As $Dv \in W_{\operatorname{loc}}^{1,\beta}(B_1,\mathbb{R}^n)$, from \eqref{fh} we can then estimate \eqref{eq:routine1} as:
   
                                                 
   
\begin{equation}\label{eq:routine2}
\int_{B_{\rho'}}|D\psi_h|^2dx \le \int_{B_1}\eta^2|D\psi_h|^2dx \le C |h|^{\theta\beta}.
\end{equation}
This means that $D\psi$ belongs to the local version of Besov space $B^{\frac{\theta \beta}{2},2}_\infty$, and we could already invoke embedding theorems concerning Besov spaces.\ Let us, instead, work with more classical fractional Sobolev spaces, which we denote by $H^t = W^{t,2}$.\ \eqref{eq:routine2} implies that $D\psi \in H^{t}(B_{\rho'},\mathbb{R}^2)$ for all $t < t_0 \doteq \frac{\beta\theta}{2}$. Indeed, 
 
                                    
               
                                                                                                                                                              
             
                                                             
\begin{align*}
\int_{B_{\rho'}\times B_{\rho'}}\frac{|D\psi(x) - D\psi(y)|^2}{|x-y|^{2(1+t)}}dxdy &= \int_{B_{\rho'}}\int_{B_{\rho'}}\frac{|D\psi(x) - D\psi(y)|^2}{|x-y|^{2(1+t)}}dydx \\
&= \int_{B_{\rho'}}\int_{B_{\frac{1-\rho}{2}}(x)}\frac{|D\psi(x) - D\psi(y)|^2}{|x-y|^{2(1+t)}}dydx \\
&\quad + \int_{B_{\rho'}}\int_{B_{\rho'}\setminus B_{\frac{1-\rho}{2}}(x)}\frac{|D\psi(x) - D\psi(y)|^2}{|x-y|^{2(1+t)}}dydx\\
& \le \int_{B_{\rho'}}\int_{B_{\frac{1-\rho}{2}}}\frac{|D\psi(x) - D\psi(x+h)|^2}{|h|^{2(1+t)}}dhdx + C\|D\psi\|_{2}^2.
\end{align*}
The last integral is finite due to \eqref{eq:routine2} and our choice of $t_0$. Thus, $D\psi \in H^{t}(B_{\rho'},\mathbb{R}^2)$, see \cite[Section 2]{HITCH}. From the fractional Sobolev embeddings \cite[Theorem 6.5]{HITCH} we find that then $D\psi \in L_{\operatorname{loc}}^\tau(B_{\rho'},\mathbb{R}^2)$ for $\tau = \frac{4}{2-2t} = \frac{2}{1-t}.$ In other words, if $\tau_0 \doteq \frac{2}{1-t_0}$ then, due to the arbitrariness of $\rho> 0$,
\[
D\psi \in L_{\operatorname{loc}}^{\tau_0 - \delta}(B_1,\mathbb{R}^2), \quad \forall \delta > 0.
\]
We can write more explicitly
\begin{equation}\label{tau0}
\frac{1}{\tau_0} =\frac{1}{2} -\frac{t_0}{2} = \frac{1}{2} -\frac{\beta\theta}{4} \overset{\eqref{theta}}{=}  \frac{1}{2} - \left[1-\frac{1}{\beta}\right] = \frac{1}{\beta} - \frac{1}{2} = \frac{1}{\beta^*}, \text{ i.e. } \tau_0 = \beta^* \overset{\eqref{feB}}{=} \frac{2r'p}{4 +(r'-2)p}.
\end{equation}
By hand, we can check that:
   
                                                                                                                 
   
      
   
                                             
   
                        
   
                                                           
   
                                                                                                                                                              
                
\begin{equation}\label{eq:p}
\tau_0(p) \overset{\eqref{bdend}}{>} p,\; \forall p \in \left(2,\frac{4}{2-r'}\right) \;\; \text{ and } \;\; \tau_0(p) > \frac{4}{2-r'}, \; \forall   p \in \left(\frac{8}{4-(r')^2},\frac{4}{2-r'}\right).
\end{equation}
                  
                             
                                                                    
               
Let now
\begin{equation}\label{sup}
\sigma \doteq \sup\left\{s>1: D\psi \in L^s_{\operatorname{loc}}(B_1,\mathbb{R}^2)\right\}.
\end{equation}
As remarked above, $\sigma > 2$ due to Theorem \ref{thm:quas}\eqref{quasma4}. We claim 
\begin{equation}\label{eq:sigma}
\sigma >  \frac{8}{4-(r')^2}.
\end{equation}
If this is not the case, then
\[
\sigma \le \frac{8}{4-(r')^2} = \frac{2}{2 + r'}\frac{4}{2-r'} < \frac{4}{2-r'}.
\]
Thus one could repeat the reasoning above with $p < \sigma$ but very close to $\sigma$, and use the first inequality of \eqref{eq:p} to get a contradiction with the fact that $\sigma$ is a supremum.\ Since \eqref{eq:sigma} is verified, we can repeat once again the reasoning above with any $p \in\left(\frac{8}{4-(r')^2},\frac{4}{2-r'}\right)$ for which $D\psi \in L^p_{\operatorname{loc}}(B_1,\mathbb{R}^2)$ to obtain that $\sigma > \frac{4}{2-r'}$ by the second inquality of \eqref{eq:p} and conclude using \eqref{bdend}.

\label{equn}
	Let $K$ be defined in \eqref{diffinc}. Then, the following are equivalent:
	\begin{enumerate}
		\item \eqref{P1} holds; \label{appsol} 
		\item for every sequence $(u_j)_j$ weakly-$*$ converging to a Lipschitz map $u$ with

	\end{enumerate}

We start assuming $\eqref{pde}$. Consider $(w_j)_j$ as in \eqref{P1}, and define $w_j(x) = (u_j(x),v_j(x))$ with $u_j, v_j: B_1 \to \mathbb{R}^n$ equi-Lipschitz.\ Since $\operatorname{dist}(Dw_j,K) \to 0$ strongly in $L^1(B_1)$, we can set $F_j \doteq D\mathcal{A}(Du_j) +Dv_jJ$, which is equibounded in $L^\infty(B_1,\mathbb{R}^{n\times 2})$, and infer $F_j \to 0 \text{ strongly in $L^1(B_1,\mathbb{R}^{n\times 2})$}$.\ Thus,
	\[
	\operatorname{div}(D\mathcal{A}(Du_j)) = \operatorname{div}(D\mathcal{A}(Du_j) + Dv_jJ) = \operatorname{div}(F_j),
	\]
	and hence by \eqref{approarea0} $(Du_j)_j$ converges strongly in $L^1$. 	\]
	Setting $w_j = (u_j,v_j)$, this map is easily seen to fulfill the assumptions of property \eqref{P1}, and hence converges strongly due to our assumption \eqref{appsol}. It follows that \eqref{pde} holds.

\label{thm:stronguq}
Let $n \ge 1$, $r>2$, $g \in L^{r}(B_1)$, $B_1\subset \mathbb{R}^2$, and $w \in W^{2,2+\gamma}_{\operatorname{loc}} \cap C^{1,\gamma}(B_1,\mathbb{R}^n)$ be a solution to

\]

<!-- prop-1: body_md -->
\label{filem}
Let $w$ fulfill the assumptions of Theorem \ref{thm:stronguq}. Then, for all $k \ge 1$ the following holds:
\begin{equation}\label{eq:secasew}
Dw(z) = o(|z-x|^{k-1})\quad \Longrightarrow \quad\lim_{z \to x}\frac{\partial_zw_j(z)}{(z-x)^{k}} \text{ exists for all $j$}.
\end{equation}
<!-- end prop-1 -->
<!-- proof-5: body_md -->
Without loss of generality, we assume that $x = 0$. Consider $0<\rho< \frac{1}{2}$ to be chosen later. We take any $y \in B_{\rho}\setminus\{0\}$ and consider, for each $j$, $f_j(z) \doteq z^{-k}(z-y)^{-1}\partial_zw_j(z).$
   
                                                
   
Let $0 <\sigma < |y|$ be so small that $B_\sigma(y) \cap B_\sigma(0) = \emptyset$ and set $\Omega_{\sigma} \doteq B_{\rho}\setminus (B_{\sigma}(y)\cup B_{\sigma}(0))$.\ By the divergence theorem and Lemma \ref{teccomp}, we see that
\begin{equation*}
\int_{\partial \Omega_{\sigma}} f_j(z)d\gamma(z) = 2i\int_{\Omega_{\sigma}} \partial_{\overline z}f_j(z)dz  = 2i\int_{\Omega_{\sigma}}z^{-k}(z-y)^{-1}\partial_{z\overline{z}}w_j(z)dz = \frac{i}{2}\int_{\Omega_{\sigma}}z^{-k}(z-y)^{-1}\Delta w_j(z)dz.
\end{equation*}
Using the structure of $\Omega_{\sigma}$ we get:
\begin{equation}\label{fiid}
\begin{split}
\int_{\partial B_{\rho}} f_j(z)d\gamma(z)- \int_{\partial B_{\sigma}(y)} f_j(z)d\gamma(z) - \int_{\partial B_{\sigma}} f_j(z)d\gamma(z)= \frac{i}{2}\int_{B_{\rho} \setminus (B_{\sigma}(y)\cup B_{\sigma})}z^{-k}(z-y)^{-1}\Delta w_j(z)dz.
\end{split}
\end{equation}
We have:
\[
\lim_{\sigma \to 0}\int_{\partial B_{\sigma}(y)} f_j(z)d\gamma(z) = 2\pi i y^{-k}\partial_zw_j(y).
\]
Furthermore by assumption we know that $\partial_zw_j(z) = o(|z|^{k - 1})$ and hence 
\[
\lim_{\sigma \to 0}\int_{\partial B_{\sigma}} f_j(z)d\gamma(z) = 0.
\]
We thus let $\sigma \to 0$ in \eqref{fiid} to get:
\begin{equation}\label{eqA}
\int_{\partial B_{\rho}} z^{-k}(z-y)^{-1}\partial_zw_j(z)d\gamma(z)-2\pi i y^{-k}\partial_zw_j(y) = \frac{i}{2}\int_{B_{\rho}}z^{-k}(z-y)^{-1}\Delta w_j(z)dz.
\end{equation}
The last passage to the limit in the right-hand side is justified by the fact that, for all $y \in \mathbb{R}^2\setminus\{0\}$,
\[
z^{-k}(z-y)^{-1}\Delta w_j(z) \in L^1_{\operatorname{loc}}(B_1).
\]
To see this, notice that the only non-trivial pole is $0$, but, since $Dw(z) = o(|z|^{k-1})$, we have for a.e. $z \in B_\rho$,
\begin{equation}\label{boundw}
|z^{-k}(z-y)^{-1}\Delta w_j(z)| \overset{\eqref{eq:intineq}}{\le} |z|^{-k}|z-y|^{-1}g(z)|D w(z)| \le C'|z|^{-1}|z-y|^{-1}g(z).
\end{equation}
The constant $C'$ depends on $k$.\ Instead, $C > 0$ will denote a universal constant that can change line by line but that does not depend on $k$. Resuming our reasoning, the last function in \eqref{boundw} belongs to $L_{\operatorname{loc}}^1$ since $r > 2$. We need to estimate the right-hand side of \eqref{eqA}. To do so, we first estimate \eqref{eqA} using \eqref{eq:intineq}:
\[
| y^{-k}\partial_zw_j(y)| \le \int_{\partial B_{\rho}} |z|^{-k}|z-y|^{-1}|\partial_zw_j(z)|d|\gamma|(z) + C\int_{B_{\rho}}|z|^{-k}|z-y|^{-1}g(z)|Dw(z)|dz.
\]
Summing over $j$, we get to
\begin{equation}\label{eqzy}
| y|^{-k}|Dw(y)| \le C\int_{\partial B_{\rho}} |z|^{-k}|z-y|^{-1}|Dw(z)|d|\gamma|(z) + C\int_{B_{\rho}}|z|^{-k}|z-y|^{-1}g(z)|Dw(z)|dz.
\end{equation}
For any $z_0 \in B_\rho\setminus\{0\}$, multiply by $g(y)|y - z_0|^{-1}$ this expression and integrate over $B_{\rho}$ to get
\begin{equation}\label{eqzya}
\begin{split}
\int_{B_\rho}| y|^{-k}|Dw(y)|g(y)|y-z_0|^{-1}dy & \le C\int_{\partial B_{\rho}} |z|^{-k}\left[\int_{B_\rho}g(y)|y-z_0|^{-1}|z-y|^{-1}dy\right] |D w(z)|d|\gamma|(z) \\
&+ C\int_{B_{\rho}}|z|^{-k}\left[\int_{B_\rho}g(y)|y-z_0|^{-1}|z-y|^{-1}dy\right] g(z)|Dw(z)|dz.
\end{split}
\end{equation}
Notice that $|(z-y)(y-z_0)|^{-1} = |z-z_0|^{-1}|(z-y)^{-1} + (y-z_0)^{-1}|$, therefore:
\begin{equation}\label{eq:prelll}
\begin{split}
\int_{B_\rho}g(y)|(z-y)(y-z_0)|^{-1} dy &=\int_{B_\rho}g(y)|z-z_0|^{-1}|(z-y)^{-1} + (y-z_0)^{-1}| dy\\
& \le |z-z_0|^{-1}\int_{B_\rho}g(y)[|z-y|^{-1} + |y-z_0|^{-1}] dy.
\end{split}
\end{equation}
We observe that, due to H\"older's inequality and the fact that $r > 2$, for all $b \in \overline{B_\rho}$
\[
\int_{B_\rho}g(y)|y-b|^{-1}dy \le \|g\|_{L^r}\left(\int_{B_{\rho}}|y-b|^{-r'}dy\right)^{\frac{1}{r'}} \le C\left(\int_{B_{2\rho}(b)}|y-b|^{-r'}dy\right)^{\frac{1}{r'}}\le C\rho^{\frac{2}{r'}-1}.
\]
Thus, we can continue \eqref{eq:prelll} as:
\begin{equation}\label{eq:prelll1}
\begin{split}
\int_{B_\rho}g(y)|(z-y)(y-z_0)|^{-1} dy  \le C|z-z_0|^{-1}\rho^{\frac{2}{r'}-1}.
\end{split}
\end{equation}
We can finally estimate \eqref{eqzya}:
\begin{equation}\label{eqzy1}
\begin{split}
\int_{B_\rho}g(y)|y-z_0|^{-1}|y|^{-k}|Dw(y)|dy &\le C\rho^{\frac{2}{r'}-1}\int_{\partial B_{\rho}} |z|^{-k}|z-z_0|^{-1}|Dw(z)|d|\gamma|(z) \\
&   + C\rho^{\frac{2}{r'}-1}\int_{B_{\rho}}|z|^{-k}|z-z_0|^{-1}g(z)|Dw(z)|dz.
\end{split}
\end{equation}
Observe that the second addendum is finite, due to the same bounds as in \eqref{boundw}. Now fix $\rho$ so small that
\begin{equation}\label{eqzy1200}
1 -C\rho^{\frac{2}{r'}-1} > 0
\end{equation}
 and absorb the second addendum of the right-hand side of \eqref{eqzy1} in the left-hand side to get
\begin{equation}\label{eqzy12}
\int_{B_\rho}g(z)|z-y|^{-1}|z|^{-k}|Dw(z)|dz \le C\rho^{\frac{2}{r'}-1}\int_{\partial B_{\rho}} |z|^{-k}|z-y|^{-1}|Dw(z)|d|\gamma|(z).
\end{equation}
Notice that we changed the name of the variable of integration from $y$ to $z$ in the first integral on the left-hand side, and we renamed $z_0$ as $y$. Going back to \eqref{eqzy}, this shows that $Dw = O(|y|^k)$.\ Hence, if $F_y(z) \doteq z^{-k}(z-y)^{-1}\Delta w_j(z)$, we can bound for some constants $c_1,c_2 > 0$:
\[
|F_y(z)| \overset{\eqref{eq:intineq}}{\le} c_1 |z|^{-k}|z-y|^{-1}g(z)|Dw|(z) \le c_2|z-y|^{-1}g(z),
\]
Since $g \in L^r(B_1)$ and $r > 2$, $F_y(\cdot)$ has equibounded $L^p(B_\rho)$ norms for some $p > 1$.\ In addition, $\lim_{y \to 0}F_y(z) = F_0(z)$ for a.e. $z \in B_\rho$.\ Thus, we can pass to the limit in \eqref{eqA} to show the assertion, namely:
\[
-2\pi i \lim_{y \to 0}y^{-k}\partial_zw_j(y) = \frac{i}{2}\int_{B_{\rho}}z^{-k-1}\Delta w_j(z)dz - \int_{\partial B_{\rho}} z^{-k-1}\partial_zw_j(z)d\gamma(z), \quad \forall j = 1,\dots, n.
\]

<!-- prop-2: body_md -->
\label{selem}
Let $w$ fulfill the assumptions of Theorem \ref{thm:stronguq}. If $Dw(z) = o(|z-x|^k)$ for some $x \in B_1$ and for all $k \in \mathbb{N}$, then $Dw \equiv 0$ in $B_r(x)$ for some $r>0$.
<!-- end prop-2 -->
<!-- proof-6: body_md -->
We again assume without loss of generality that $x = 0$, and moreover that $g \ge 1$ a.e in $B_1$. Due to our hypotheses, we see that \eqref{eqzy12} now holds for $y = 0$, any (fixed) $\rho > 0$ which fulfills \eqref{eqzy1200}, and all $k$:
\begin{equation}\label{eqzy000}
\int_{B_\rho}g(z)|z|^{-k-1}|Dw(z)|dz \le C\rho^{\frac{2}{r'}-1}\int_{\partial B_{\rho}} |z|^{-k-1}|Dw(z)|d|\gamma|(z) =  C\rho^{\frac{2}{r'}-2-k}\int_{\partial B_{\rho}}|Dw(z)|d|\gamma|(z).
\end{equation}
From the proof of Proposition \ref{filem} we know that the constant $C$ of \eqref{eqzy000} is independent of $k$. 
                                                                                                                                                                        
Using that $g \ge 1$ a.e. and dividing both sides of \eqref{eqzy000} by $\rho^{-k-1}$, we get:
                        
                                
                                                                                                          
               
                                                
\begin{equation}\label{eqzy1234}
	\int_{B_\rho}\frac{\rho^{k +1}}{|z|^{k+1}}|Dw(z)|dz \le C\rho^{\frac{2}{r'}-1}\int_{\partial B_{\rho}}|Dw(z)|d|\gamma|(z).
\end{equation}
Since $|z| \le \rho$ and the right-hand side does not depend on $k$, letting $k \to \infty$ shows $Dw\equiv 0$ on $B_\rho$ by Fatou's Lemma, as wanted.

<!-- proof-7: body_md -->
Consider the closed set $$Z = \{x \in B_1: \forall r > 0 \text{ with }B_r(x) \subset B_1, B_r(x) \text{ contains at least two zeroes of }Dw\}.$$ Through Propositions \ref{filem}-\ref{selem} we see that $Z$ is open as well.\ Hence, either $Dw \equiv 0$ or the zeroes of $Dw$ are isolated. In the latter case, the claimed asymptotic expansion of $\partial_zw_j$ is then a consequence of Proposition \ref{filem}, and the fact that $a_{j_0} \neq 0$ for some $j_0$ is a consequence of Proposition \ref{selem}.
