<!-- definition-1: body_md -->
\label{def:h}
Let $\mathbb{T}$ be a compact metric tree with leaves $\xi_{1},\ldots,\xi_{\kappa}$. The $L^2$
Hausdorff distance on $\mathbb{T}$ is
\begin{equation}
\label{eq:h}
h_\kappa(s,t) = \left( \frac{1}{\kappa} \sum_{j=1}^{\kappa} \,\big|d(\xi_{j},s) -
d(\xi_{j},t)\big|^2 \right)^{1/2},
\qquad s,t\in \mathbb{T}.
\end{equation}
<!-- end definition-1 -->
<!-- proposition-1: body_md -->
\label{thm:struct}
Let $\mathbb{T}$ be a compact metric tree with $\kappa$ leaves and let $h_\kappa$ be given by
\eqref{eq:h}. Then the following hold.
\begin{enumerate}
\item[\rm (i)] $h_\kappa$ is a metric on $\mathbb{T}$.

\item[\rm (ii)] The two metrics are comparable in both directions, in the sense that
\begin{equation}
\label{eq:bilip}
\sqrt{2/\kappa}\; d(s,t) \;\leq\; h_\kappa(s,t) \;\leq\; d(s,t), \qquad s,t\in \mathbb{T}.
\end{equation}

\item[\rm (iii)] Equality holds on the right of \eqref{eq:bilip} if and only if no vertex of
degree three or more lies strictly between $s$ and $t$. In particular $h_{\kappa}(s,t)=d(s,t)$
whenever $s$ and $t$ lie in a common edge.
\end{enumerate}
<!-- end proposition-1 -->
<!-- proposition-2: body_md -->
\label{thm:local}
Let $\mathbb{T}$ be a compact metric tree with $\kappa$ leaves and let $s\neq t$. Then
\begin{equation}
\label{eq:local}
h_\kappa(s,t) \;\geq\; \big(1-\kappa_{st}/\kappa\big)^{1/2}\, d(s,t) .
\end{equation}
The left-hand inequality of \eqref{eq:bilip} is the case $\kappa_{st}=\kappa-2$, and the equality
of part (iii) is the case $\kappa_{st}=0$.
<!-- end proposition-2 -->
<!-- theorem-1: body_md -->
\label{thm:pd}
Let $\mathbb{T}$ be a compact metric tree with $\kappa$ leaves and let $h_\kappa$ be given by
\eqref{eq:h}. Then, for every $\varphi\in\Phi_\kappa$ and every $\sigma^2>0$,
\begin{equation}
\label{eq:model}
C(s,t) = \sigma^2\,\varphi\big(h_\kappa(s,t)\big), \qquad s,t\in\mathbb{T},
\end{equation}
is a covariance function on $\mathbb{T}$.
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{cor:local}
Consider the model \eqref{eq:model}, and let $s$ and $t$ lie in a common edge. Under the Mat\'ern model the field is $m$ times mean square differentiable along that edge if and only if $\nu>m$. Its sample paths there have fractal index $2\nu$ for $\nu<1$ under the Mat\'ern model, and $\alpha$ under the powered exponential and generalized Cauchy models.
<!-- end corollary-1 -->
<!-- corollary-2: body_md -->
\label{cor:multi}
Let $\bm{\varphi}=(\varphi_{ij})_{i,j=1}^{p}$ be a mapping from $[0,\infty)$ into the $p\times p$
real matrices such that $(\bm{x},\bm{y})\mapsto\bm{\varphi}(\|\bm{x}-\bm{y}\|)$ is a matrix-valued
covariance function on $\mathbb{R}^\kappa$. Then
\begin{equation}
\label{eq:multi}
\bm{C}(s,t) \;=\; \bm{\varphi}\big(h_\kappa(s,t)\big), \qquad s,t\in\mathbb{T},
\end{equation}
is a matrix-valued covariance function on $\mathbb{T}$, so that there exists a $p$-variate
random field $\bm{Z}=(Z_1,\hdots,Z_p)^\top$ on $\mathbb{T}$ with
$\mathrm{cov}\{Z_i(s),Z_j(t)\}=\varphi_{ij}(h_\kappa(s,t))$ for every $i,j=1,\hdots,p$.
<!-- end corollary-2 -->
<!-- example-1: body_md -->
\label{ex:bimatern}
Let $p=2$ and write $\mathcal{M}(r\,|\,\nu,a)=\big(2^{1-\nu}/\Gamma(\nu)\big)(r/a)^{\nu}K_\nu(r/a)$
for the Mat\'ern correlation of Table \ref{tab:phi} with smoothness $\nu$ and scale $a$. In the
bivariate Mat\'ern model each entry of \eqref{eq:multi} is a Mat\'ern correlation with its own
smoothness and its own scale,
\[
\bm{C}(s,t)=
\begin{pmatrix}
\sigma_1^2\,\mathcal{M}(h_\kappa(s,t)\,|\,\nu_{11},a_{11}) &
\rho\,\sigma_1\sigma_2\,\mathcal{M}(h_\kappa(s,t)\,|\,\nu_{12},a_{12})\\[3pt]
\rho\,\sigma_1\sigma_2\,\mathcal{M}(h_\kappa(s,t)\,|\,\nu_{12},a_{12}) &
\sigma_2^2\,\mathcal{M}(h_\kappa(s,t)\,|\,\nu_{22},a_{22})
\end{pmatrix},
\]
where $\sigma_1^2,\sigma_2^2>0$ and $\rho\in[-1,1]$. Part (d) of Theorem 3 of
\cite{gneiting2010matern} covers the case
\begin{equation}
\label{eq:bimatern}
\nu_{12}=\tfrac{1}{2}(\nu_{11}+\nu_{22}),
\qquad
a_{12}\;\leq\;\min(a_{11},a_{22}),
\end{equation}
in which the model is a matrix-valued covariance function on $\mathbb{R}^\kappa$, and therefore on
$\mathbb{T}$ by Corollary \ref{cor:multi}, if and only if
\begin{equation}
\label{eq:bimatern2}
|\rho| \;\leq\;
\left(\frac{a_{12}}{a_{11}}\right)^{\!\nu_{11}}
\left(\frac{a_{12}}{a_{22}}\right)^{\!\nu_{22}}
\left\{\frac{\Gamma(\nu_{11}+\kappa/2)}{\Gamma(\nu_{11})}\,
\frac{\Gamma(\nu_{22}+\kappa/2)}{\Gamma(\nu_{22})}\right\}^{1/2}
\frac{\Gamma(\nu_{12})}{\Gamma(\nu_{12}+\kappa/2)} .
\end{equation}
The number of leaves enters \eqref{eq:bimatern2} only through the gamma factors, which equal one
when $\nu_{11}=\nu_{22}$ and decrease as $\kappa$ grows and as the two marginal smoothness
parameters separate. The scales contribute a factor that does not depend on $\kappa$ and is at most
one by \eqref{eq:bimatern}. Equal scales and equal smoothness
place no restriction on $\rho$, and setting $a_{11}=a_{22}=a_{12}$ gives the parsimonious model,
whose bound does not involve the scale.
<!-- end example-1 -->
<!-- corollary-3: body_md -->
\label{cor:st}
Let $\widetilde{\varphi}$ be a mapping from $[0,\infty)^2$ into $\mathbb{R}$ such that
$\big((\bm{x},u),(\bm{y},w)\big)\mapsto\widetilde{\varphi}(\|\bm{x}-\bm{y}\|,|u-w|)$ is a covariance
function on $\mathbb{R}^\kappa\times\mathbb{R}$. Then
\begin{equation}
\label{eq:st}
C\big((s,u),(t,w)\big) \;=\; \widetilde{\varphi}\big(h_\kappa(s,t),|u-w|\big),
\qquad (s,u),(t,w)\in\mathbb{T}\times\mathbb{R},
\end{equation}
is a covariance function on $\mathbb{T}\times\mathbb{R}$, so that there exists a
random field $\{Z(s,u),\,s\in\mathbb{T},\,u\in\mathbb{R}\}$ with

<!-- lemma-1: body_md -->
\label{lem:id}
Let $s\neq t$ in $\mathbb{T}$. Then
\begin{equation}
\label{eq:xsq}
h_\kappa^2(s,t) = d^2(s,t)\left\{ 1 - \frac{4}{\kappa}\sum_{j=1}^{\kappa}
x_j(s,t)\{1-x_j(s,t)\} \right\} .
\end{equation}
<!-- end lemma-1 -->
<!-- head-1: body_md -->
Every path from $\xi_{j}$ to a point of $P$
passes through $m_j(s,t)$, so that $d(\xi_{j},s)=d(\xi_{j},m_j(s,t))+p_j(s,t)$ and
$d(\xi_{j},t)=d(\xi_{j},m_j(s,t))+p_j(t,s)$. Subtracting, and using $p_j(s,t)+p_j(t,s)=d(s,t)$,
\[
d(\xi_{j},s) - d(\xi_{j},t) = p_j(s,t) - p_j(t,s) = \{2x_j(s,t)-1\}\,d(s,t) .
\]
Squaring, averaging over $j=1,\hdots,\kappa$ and using \eqref{eq:h},
$h_\kappa^2(s,t)=d^2(s,t)\kappa^{-1}\sum_{j=1}^\kappa\{2x_j(s,t)-1\}^2$, which is \eqref{eq:xsq} because
$(2x-1)^2=1-4x(1-x)$.
\hfill$\square$
