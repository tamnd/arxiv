<!-- definition-1: body_md -->
\label{def:esp}
The reservoir \eqref{eq:general_reservoir} has the \gls{ESP} if, for every ${u\in\mathcal U}$ and every ${x_0,x_0'\in\mathbb{R}^n}$, there holds
\begin{align}
    \lim_{t\to\infty}\|\varphi_t(x_0,u)-\varphi_t(x_0',u)\| = 0.
    \label{eq:esp}
\end{align}
<!-- end definition-1 -->
<!-- lemma-1: body_md -->
\label{lem:linear_invariant_map}
Consider the reservoir \eqref{eq:linear_reservoir} driven by the exosystem \eqref{eq:linear_exosystem} through the interconnection \eqref{eq:interconnection}. Assume \textup{(A\ref{asmp:linear_hurwitz})} and \textup{(A\ref{asmp:linear_neutral})} hold. Then the Sylvester equation
\begin{align}
    F\Pi+GL &= \Pi S
    \label{eq:sylvester}
\end{align}
has a unique solution ${\Pi\in\mathbb{R}^{n\times\nu}}$, and the subspace
\begin{align}
    \mathcal{V} &= \left\{(w,x)\in\mathbb{R}^{\nu}\times\mathbb{R}^n \,\middle|\, x=\Pi w\right\}
    \label{eq:invariant_subspace}
\end{align}
is invariant under the dynamics of the system \eqref{eq:linear_reservoir}--\eqref{eq:interconnection}.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
By \textup{(A\ref{asmp:linear_hurwitz})} and \textup{(A\ref{asmp:linear_neutral})}, ${\sigma(F)\cap\sigma(S)=\varnothing}$, hence \eqref{eq:sylvester} has a unique solution \cite{antoulas_approximation_2005}. Let $(w,x)$ be a solution of \eqref{eq:linear_reservoir}--\eqref{eq:interconnection} and let ${\varepsilon=x-\Pi w}$. Then \eqref{eq:sylvester} implies
\begin{align*}
    \dot\varepsilon = Fx+GLw-\Pi Sw = F\varepsilon+(F\Pi+GL-\Pi S)w = F\varepsilon .
\end{align*}
If ${(w(0),x(0))\in\mathcal V}$, then ${\varepsilon(0)=0}$, so ${\varepsilon(t)=0}$ for all ${t\geq0}$ by uniqueness of solutions of ${\dot\varepsilon=F\varepsilon}$, \textit{i.e.}, $\mathcal V$ is invariant.
<!-- end proof-1 -->
<!-- proposition-1: body_md -->
\label{prop:luenberger}
Consider the linear reservoir \eqref{eq:linear_reservoir} driven by the linear exosystem \eqref{eq:linear_exosystem} through the interconnection \eqref{eq:interconnection}. Assume \textup{(A\ref{asmp:linear_hurwitz})} and \textup{(A\ref{asmp:linear_neutral})} hold. Then the following statements hold.
{\renewcommand{\roman{enumi}}{\roman{enumi}}\renewcommand{(\roman{enumi})}{(\roman{enumi})}\setlength{\leftmargini}{1.75em}
\begin{enumerate}
    \item\label{item:pi_integral} The unique solution of \eqref{eq:sylvester} is
    \begin{align}
        \Pi &= \int_0^\infty e^{Ft}GL e^{-St}\,\mathrm{d} t.
        \label{eq:pi_integral}
    \end{align}
    \item\label{item:sync_attractive} Along every solution of \eqref{eq:linear_reservoir}--\eqref{eq:interconnection}, the error \eqref{eq:manifold_error_dynamics} satisfies ${\dot{\varepsilon} = F\varepsilon}$, and $\mathcal{V}$ is globally exponentially attractive.
        \item\label{item:exact_readout} There exists ${W_{\mathrm{out}}\in\mathbb{R}^{p\times n}}$ such that
    \begin{align}
        W_{\mathrm{out}}\Pi &= Q
        \label{eq:perfect_readout}
    \end{align}
    if and only if ${\operatorname{rank}\begin{bsmallmatrix}\Pi\\ Q\end{bsmallmatrix}=\operatorname{rank}\Pi}$. Any such $W_{\mathrm{out}}$ gives a readout error \eqref{eq:linear_readout_error} which converges exponentially to zero.
\end{enumerate}
}
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
(\ref{item:pi_integral}) By \textup{(A\ref{asmp:linear_hurwitz})} and \textup{(A\ref{asmp:linear_neutral})}, the integral in \eqref{eq:pi_integral} is absolutely convergent. Denoting its value by $\tilde\Pi$, we obtain
\begin{align*}
    F\tilde\Pi-\tilde\Pi S
    &= \int_0^\infty \frac{\,\mathrm{d}}{\,\mathrm{d} t}\left(e^{Ft}GL e^{-St}\right)\,\mathrm{d} t = -GL,
\end{align*}
so $\tilde\Pi$ solves \eqref{eq:sylvester} and ${\tilde\Pi=\Pi}$ by Lemma \ref{lem:linear_invariant_map}.

(\ref{item:sync_attractive}) The identity ${\dot\varepsilon=F\varepsilon}$ is established in the proof of Lemma \ref{lem:linear_invariant_map}. By \textup{(A\ref{asmp:linear_hurwitz})}, there exist ${M\geq1}$ and ${\alpha>0}$ such that ${\|e^{Ft}\|\leq Me^{-\alpha t}}$, so ${\|\varepsilon(t)\|\leq Me^{-\alpha t}\|\varepsilon(0)\|}$ for all ${t\geq0}$.

(\ref{item:exact_readout}) The rank condition holds if and only if the row space of $Q$ is contained in the row space of $\Pi$, \textit{i.e.}, if and only if ${Q=W_{\mathrm{out}}\Pi}$ for some ${W_{\mathrm{out}}\in\mathbb{R}^{p\times n}}$. For every $W_{\mathrm{out}}$ satisfying \eqref{eq:perfect_readout},
$    e=W_{\mathrm{out}} x-Qw=W_{\mathrm{out}}(x-\Pi w)=W_{\mathrm{out}}\varepsilon,$
which converges exponentially to zero by (\ref{item:sync_attractive}).
<!-- end proof-2 -->
<!-- proposition-2: body_md -->
\label{prop:free_modes}
Consider the closed-loop reservoir \eqref{eq:loop_closure}, with ${p=m}$ and ${Q=L}$, where $W_{\mathrm{out}}$ satisfies the exact readout condition \eqref{eq:perfect_readout}. Assume \textup{(A\ref{asmp:linear_hurwitz})}, \textup{(A\ref{asmp:linear_neutral})}, and \textup{(A\ref{asmp:linear_observability})} hold. Then
\begin{align}
    (F+GW_{\mathrm{out}})\Pi &= \Pi S,
    \label{eq:closed_loop_intertwining}
\end{align}
and $\Pi$ is injective, so that ${\sigma(S)\subseteq\sigma(F+GW_{\mathrm{out}})}$.
<!-- end proposition-2 -->
<!-- proof-3: body_md -->
Since ${Q=L}$, \eqref{eq:perfect_readout} and \eqref{eq:sylvester} give $${(F+GW_{\mathrm{out}})\Pi=F\Pi+GL=\Pi S}.$$ If ${\Pi v=0}$, then ${Lv=W_{\mathrm{out}}\Pi v=0}$ and $${\Pi Sv=F\Pi v+GLv=0}$$ by \eqref{eq:sylvester}, so $\ker\Pi$ is an $S$-invariant subspace of $\ker L$, hence ${\ker\Pi=\{0\}}$ by \textup{(A\ref{asmp:linear_observability})}, that is, ${\operatorname{rank}\Pi=\nu}$. Thus, by \eqref{eq:closed_loop_intertwining}, the restriction of ${F+GW_{\mathrm{out}}}$ to $\operatorname{range}\Pi$ is similar to $S$, and ${\sigma(S)\subseteq\sigma(F+GW_{\mathrm{out}})}$.
<!-- end proof-3 -->
<!-- remark-1: body_md -->
\label{rem:transverse}
Proposition \ref{prop:free_modes} implies that the closed-loop modes outside $\operatorname{range}\Pi$ are not fixed by training. To see this, let $\Pi^\dagger$ be the Moore--Penrose pseudoinverse of $\Pi$, and let $N$ have orthonormal columns spanning $\ker\Pi^\top$. In the coordinates ${(\Pi^\dagger x,N^\top x)}$, the matrix ${F+GW_{\mathrm{out}}}$ is block upper triangular, with diagonal blocks $S$ and ${N^\top(F+GW_{\mathrm{out}})N}$, whose ${n-\nu}$ eigenvalues we call \emph{transverse}. Under \textup{(A\ref{asmp:linear_controllability})}, an application of the \gls{pbh} test \cite{wonham_linear_1985} shows that, by appropriately choosing $W_{\mathrm{out}}$ among the solutions of \eqref{eq:perfect_readout}, the transverse eigenvalues can be assigned to any multiset of ${n-\nu}$ complex numbers closed under conjugation \cite{wonham_linear_1985}. The stability of the predictor~\eqref{eq:linear_readout_definition}, \eqref{eq:loop_closure} is thus decided not by the target, but by the transverse eigenvalues, which depend on the transients and noise in the data entering \eqref{eq:least_squares_minimization}, and on regularization, if any, as observed for trained predictors in~\cite{lu_attractor_2018}.
<!-- end remark-1 -->
<!-- proposition-3: body_md -->
\label{prop:moment_matching}
Consider the reservoir \eqref{eq:linear_reservoir} with output \eqref{eq:linear_readout_definition} and transfer matrix \eqref{eq:transfer_function}. Assume \textup{(A\ref{asmp:linear_hurwitz})} and \textup{(A\ref{asmp:linear_neutral})} hold. Then the following statements hold.
{\renewcommand{(\roman{enumi})}{(\roman{enumi})}\setlength{\leftmargini}{2.5em}
\begin{enumerate}
    \item The unique solution $\Pi$ of \eqref{eq:sylvester} satisfies
    \begin{align}
        W_{\mathrm{out}}\Pi V
        =
        \begin{bmatrix}
            \eta_0(s_1)Lv_1 &
            \cdots &
            \eta_0(s_\nu)Lv_\nu
        \end{bmatrix}.
        \label{eq:tangential_moments}
    \end{align}
    \item The exact readout condition \eqref{eq:perfect_readout} holds if and only if
    \begin{align}
        \eta_0(s_i)Lv_i = Qv_i, \qquad i=1,\ldots,\nu.
        \label{eq:interpolation_conditions}
    \end{align}
\end{enumerate}
}
<!-- end proposition-3 -->
<!-- proof-4: body_md -->
(i) Let $Sv_i=s_i v_i$. Right-multiplying \eqref{eq:sylvester} by $v_i$ gives ${(s_iI_n-F)\Pi v_i=GLv_i}$. By \textup{(A\ref{asmp:linear_hurwitz})} and \textup{(A\ref{asmp:linear_neutral})}, ${s_i\notin\sigma(F)}$, hence ${W_{\mathrm{out}}\Pi v_i=W_{\mathrm{out}}(s_iI_n-F)^{-1}GLv_i=\eta_0(s_i)Lv_i}$ for ${i=1,\ldots,\nu}$, which is \eqref{eq:tangential_moments}.

(ii) Since $V$ is invertible, \eqref{eq:perfect_readout} holds if and only if ${W_{\mathrm{out}}\Pi V=QV}$, \textit{i.e.}, by \eqref{eq:tangential_moments}, if and only if \eqref{eq:interpolation_conditions} holds.
<!-- end proof-4 -->
<!-- lemma-2: body_md -->
\label{lem:nonlinear_synchronization}
Consider the reservoir \eqref{eq:general_reservoir} driven by the exosystem \eqref{eq:nonlinear_exosystem} through the interconnection \eqref{eq:interconnection}. Assume \textup{(A\ref{asmp:nonlinear_regularity})}--\textup{(A\ref{asmp:nonlinear_neutral_stability})} hold. Then the invariance equation
\begin{align}
    \frac{\partial\pi}{\partial w}(w)s(w)
    &=
    f(\pi(w), \ell(w))
    \label{eq:nonlinear_invariance}
\end{align}
has a locally unique solution ${\pi:\mathcal O_\pi\to\mathbb R^n}$ of class $C^2$ with ${\pi(0)=0}$, defined on an open neighborhood ${\mathcal O_\pi\subseteq\mathcal O}$ of ${w=0}$, and the manifold
\begin{align}
    \mathcal M
    &=
    \left\{(w,x)\in\mathcal O_\pi\times\mathbb R^n
    \,\middle|\,
    x=\pi(w)\right\}
    \label{eq:nonlinear_invariant_manifold}
\end{align}
is locally invariant under the dynamics of the interconnected system \eqref{eq:general_reservoir}, \eqref{eq:interconnection}, \eqref{eq:nonlinear_exosystem}.
<!-- end lemma-2 -->
<!-- proof-5: body_md -->
The proof follows standard arguments of nonlinear output regulation (see, \textit{e.g.}, \cite{isidori_nonlinear_1995}). By \textup{(A\ref{asmp:nonlinear_regularity})}, the interconnected system is of class $C^2$ with an equilibrium at the origin. By \textup{(A\ref{asmp:nonlinear_stability})} and \textup{(A\ref{asmp:nonlinear_neutral_stability})}, its linearization at the origin has $n$ eigenvalues with negative real part and $\nu$ eigenvalues on the imaginary axis, so the center manifold theorem yields a locally invariant $C^2$ manifold which is the graph of a map $\pi$ with ${\pi(0)=0}$. Local invariance of this graph is equivalent to \eqref{eq:nonlinear_invariance}, and local uniqueness follows from the Lyapunov and Poisson stability in \textup{(A\ref{asmp:nonlinear_neutral_stability})}.
<!-- end proof-5 -->
<!-- proposition-4: body_md -->
\label{prop:kkl_synchronization}
Consider the reservoir \eqref{eq:linear_reservoir} driven by the exosystem \eqref{eq:nonlinear_exosystem} through the interconnection \eqref{eq:interconnection}. Assume \textup{(A\ref{asmp:linear_hurwitz})}, \textup{(A\ref{asmp:nonlinear_regularity})}, \textup{(A\ref{asmp:nonlinear_neutral_stability})}, and \textup{(A\ref{asmp:nonlinear_backward})} hold. Then the following statements hold.
{\renewcommand{\roman{enumi}}{\roman{enumi}}\renewcommand{(\roman{enumi})}{(\roman{enumi})}\setlength{\leftmargini}{2.em}
\begin{enumerate}
    \item\label{item:kkl_integral} The locally unique solution $\pi$ of \eqref{eq:kkl_invariance} is
    \begin{align}
        \pi(w) &= \int_0^\infty e^{Ft}G\ell\bigl(\varphi_{-t}(w)\bigr)\,\mathrm{d} t, \quad w\in\mathcal K.
        \label{eq:nonlinear_synchronization_map}
    \end{align}
    \item\label{item:kkl_sync} Along every solution of \eqref{eq:linear_reservoir}, \eqref{eq:nonlinear_exosystem}, \eqref{eq:interconnection} with ${w(0)\in\mathcal K}$, the error \eqref{eq:nonlinear_manifold_error_dynamics} satisfies ${\dot{\varepsilon} = F\varepsilon}$, and $\mathcal M$ is exponentially attractive.
    \item\label{item:kkl_readout} There exists ${W_{\mathrm{out}}\in\mathbb{R}^{p\times n}}$ such that
    \begin{align}
        W_{\mathrm{out}}\pi(w)
        &=
        q(w),
        \qquad w\in\mathcal K,
        \label{eq:nonlinear_perfect_readout}
    \end{align}
    if and only if 
    ${\dim\operatorname{span}\begin{bsmallmatrix}\pi\\ q\end{bsmallmatrix}(\mathcal K)=\dim\operatorname{span}\pi(\mathcal K)}.$
    For ${w(0)\in\mathcal K}$, every such $W_{\mathrm{out}}$ gives a readout error \eqref{eq:nonlinear_readout_error} which converges exponentially to zero along the solutions of \eqref{eq:linear_reservoir},  \eqref{eq:interconnection}, \eqref{eq:nonlinear_exosystem}.
\end{enumerate}
}
<!-- end proposition-4 -->
<!-- proof-6: body_md -->
(\ref{item:kkl_integral}) The argument follows \cite{andrieu_existence_2006}. Fix ${w\in\mathcal K}$. By assumption \textup{(A\ref{asmp:nonlinear_backward})}, ${\varphi_{-t}(w)\in\mathcal K\subseteq\mathcal O_\pi}$ for all ${t\geq0}$. Then \eqref{eq:kkl_invariance} implies
$${\frac{\,\mathrm{d}}{\,\mathrm{d} t}\bigl[e^{Ft}\pi(\varphi_{-t}(w))\bigr]=-e^{Ft}G\ell(\varphi_{-t}(w))}.$$ Since $\pi$ and $\ell$ are bounded on the compact set $\mathcal K$ and $F$ is Hurwitz by \textup{(A\ref{asmp:linear_hurwitz})}, integrating over $\mathbb{R}_{+}$ yields \eqref{eq:nonlinear_synchronization_map}.

(\ref{item:kkl_sync}) 
By assumption \textup{(A\ref{asmp:nonlinear_backward})}, ${w(0)\in\mathcal K}$ implies ${w(t)\in\mathcal K}$ for all ${t\geq0}$. Then \eqref{eq:kkl_invariance} gives 
$${\dot{\varepsilon} = Fx+G\ell(w)-\frac{\partial\pi}{\partial w}(w)s(w) = F\varepsilon},$$
which converges to zero exponentially by \textup{(A\ref{asmp:linear_hurwitz})}.

(\ref{item:kkl_readout}) By the equality of row and column rank, ${\dim\operatorname{span}\begin{bsmallmatrix}\pi\\ q\end{bsmallmatrix}(\mathcal K)}$ and ${\dim\operatorname{span}\pi(\mathcal K)}$ are the numbers of linearly independent components of $\begin{bsmallmatrix}\pi\\ q\end{bsmallmatrix}$ and of $\pi$, viewed as functions on $\mathcal K$. They coincide if and only if each component of $q$ is a linear combination of the components of $\pi$ on $\mathcal K$, that is, if and only if \eqref{eq:nonlinear_perfect_readout} holds for some $W_{\mathrm{out}}$, as in the proof of Proposition \ref{prop:luenberger}(\ref{item:exact_readout}). For every such $W_{\mathrm{out}}$, along a solution with ${w(0)\in\mathcal K}$, ${e=W_{\mathrm{out}} x-q(w)=W_{\mathrm{out}}\varepsilon}$, which converges exponentially to zero by (\ref{item:kkl_sync}).

<!-- corollary-1: body_md -->
\label{cor:nonlinear_immersion}
Consider the reservoir \eqref{eq:linear_reservoir} driven by the exosystem \eqref{eq:nonlinear_exosystem} through the interconnection \eqref{eq:interconnection}, with ${p=m}$ and ${q=\ell}$, and let $\pi$ be the map of Lemma \ref{lem:nonlinear_synchronization}. Assume \textup{(A\ref{asmp:linear_hurwitz})}, \textup{(A\ref{asmp:nonlinear_regularity})}, \textup{(A\ref{asmp:nonlinear_neutral_stability})}, and \textup{(A\ref{asmp:nonlinear_backward})} hold, and that $W_{\mathrm{out}}$ satisfies \eqref{eq:nonlinear_perfect_readout}. Then, along every solution with ${w(0)\in\mathcal K}$, ${\xi=\pi(w)}$ satisfies
\begin{align}
    \dot{\xi}
    &=
    \bigl(F+GW_{\mathrm{out}}\bigr)\xi,
    &
    y
    &=
    W_{\mathrm{out}}\xi .
    \label{eq:nonlinear_immersed_system}
\end{align}
<!-- end corollary-1 -->
<!-- proof-7: body_md -->
By \textup{(A\ref{asmp:nonlinear_backward})}, along a solution with ${w(0)\in\mathcal K}$, ${w(t)\in\mathcal K}$ for all ${t\geq0}$. Then \eqref{eq:kkl_invariance} and \eqref{eq:nonlinear_perfect_readout}, with ${q=\ell}$, imply ${\dot\xi=\frac{\partial\pi}{\partial w}(w)s(w) = F\pi(w)+G\ell(w) = (F+GW_{\mathrm{out}})\xi}$ and ${y=\ell(w)=W_{\mathrm{out}}\pi(w)=W_{\mathrm{out}}\xi}$.

<!-- remark-2: body_md -->
\label{rem:moments}
The invariance equation \eqref{eq:nonlinear_invariance} is also the starting point of model reduction by moment matching for nonlinear systems \cite{astolfi_model_2010}. Under \textup{(A\ref{asmp:nonlinear_regularity})}--\textup{(A\ref{asmp:nonlinear_neutral_stability})}, which ensure that \eqref{eq:nonlinear_invariance} admits a locally unique solution $\pi$, the \emph{moment} of the reservoir \eqref{eq:general_reservoir}, \eqref{eq:linear_readout_definition} at the signal generator $(s,\ell)$ is the steady-state response ${w\mapstoW_{\mathrm{out}}\pi(w)}$ of the reservoir output to the signal generator \cite{astolfi_model_2010}. The exact readout condition \eqref{eq:nonlinear_perfect_readout} thus states that the moment of the trained reservoir at $(s,\ell)$ coincides with the target map $q$, and reservoir computing can be regarded as enforcing moment matching in a data-driven fashion \cite{scarciotti_data_2017}.
