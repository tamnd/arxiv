<!-- theorem-1: body_md -->
\label{thm:standard}
Fix $b\in\{1,3\}$ and $s\in\mathbb C$ such that $\operatorname{Re}(s)>(k+1)/2$. Then
\begin{align}\label{eq:standarduniform}
 \sup_{0\neq h\in S_k}
 \#\{p\in\mathscr P_{4,b}(X):L(h,\chi_p,s)=0\}=o(P_b(X)).
\end{align}
Equivalently, for any $\epsilon>0$, there exists some $X_0$ independent of $h$, such that 
$$\#\{p\in\mathscr P_{4,b}(X):L(h,\chi_p,s)=0\}<\epsilon P_b(X)$$
for every $X\geq X_0$ and every $0\neq h\in S_k$.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:symmetric}
Fix $b\in\{1,3\}$ and $s\in\mathbb C$ such that $\operatorname{Re}(s)>k$. Then
\begin{align}\label{eq:symuniform}
 \sup_{\boldsymbol{c}\in\mathbb C^d\setminus\{0\}}
 \#\left\{p\in\mathscr P_{4,b}(X):
 \sum_{i=1}^dc_iL(\operatorname{Sym}^2f_i,\chi_p,s)=0\right\}=o(P_b(X)).
\end{align}
Equivalently, for any $\epsilon>0$, there exists some $X_0$ independent of $\boldsymbol{c}=(c_1,\ldots,c_d)$, such that 
$$\#\left\{p\in\mathscr P_{4,b}(X):
\sum_{i=1}^dc_iL(\operatorname{Sym}^2f_i,\chi_p,s)=0\right\}<\epsilon P_b(X)$$
for every $X\geq X_0$ and every $\boldsymbol{c}\in\mathbb C^d\setminus\{0\}$.
<!-- end theorem-2 -->
<!-- corollary-1: body_md -->
\label{cor:determinant}
Fix $b\in\{1,3\}$, $m\in\{1,2\}$, and $s\in\mathbb C$ with
$\operatorname{Re}(s)>\sigma_m$. Then
\begin{align}\label{eq:detcount}
 \#\left\{(p_1,\ldots,p_d)\in\mathscr P_{4,b}(X)^d:
 \det(L^{(m)}(f_i,\chi_{p_j},s))_{1\leq i,j\leq d}=0\right\}
 =o(P_b(X)^d).
\end{align}
Moreover, outside a set of $o(P_b(X)^d)$ tuples, the determinants are simultaneously nonzero for any fixed finite collection of pairs $(m,s)$ with $m\in\{1,2\}$ and $\operatorname{Re}(s)>\sigma_m$.

<!-- theorem-3: body_md -->
\label{thm:abstract}
Let $d\geq1$ and let $u_i(r)\in[-2,2]$ for every prime $r$ and
$1\leq i\leq d$. Suppose that, for each $i\neq j$, we have
\begin{align}\#\{r~{\rm prime}:u_i(r)\neq u_j(r)\}=\infty.\label{eq:abstractseparation}\end{align} Fix
$w\in\mathbb C$ with $\operatorname{Re}(w)>1$. Define
\begin{align*}
 Z_i&=\prod_r(1-u_i(r)\varepsilon_r r^{-w}+r^{-2w})^{-1},\\
 Y_i(p)&=\prod_r(1-u_i(r)\chi_p(r)r^{-w}
+\chi_p(r)^2r^{-2w})^{-1},
\end{align*}
where the products are over
all primes, and $r^{-w}:=\exp(-w\log r)$. Then these products converge absolutely and are nonzero. For every
$\boldsymbol{c}=(c_1,\ldots,c_d)\in\mathbb C^d\setminus\{0\}$,
\begin{align}
 \mu\left(\sum_{i=1}^d c_i Z_i=0\right)=0. \label{eq:randomsignP=0}
\end{align}
Moreover, for each $b\in\{1,3\}$,
\begin{align}
 \sup_{\boldsymbol{c}\neq0}
 \#\left\{p\in\mathscr P_{4,b}(X):\sum_{i=1}^d c_iY_i(p)=0\right\}
 =o(P_b(X)). \label{eq:twistP=0}
\end{align}

<!-- lemma-2: body_md -->
\label{lem:signs}
Fix $b\in\{1,3\}$, distinct primes $r_1,\ldots,r_t$, and signs
$\eta_1,\ldots,\eta_t\in\{-1,1\}$. Then
\begin{align}\label{eq:signequidistribution}
 \lim_{X\longrightarrow\infty}
 \frac{\#\{p\in\mathscr P_{4,b}(X):\chi_p(r_j)=\eta_j\ (1\leq j\leq t)\}}
      {P_b(X)}=2^{-t}.
\end{align}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Let $N(X)$ denote the numerator in \eqref{eq:signequidistribution}. We may ignore the finitely many primes $p$ among  $r_1,\dots,r_t$ since they don't affect the limit. 
For an odd $r_j$, quadratic reciprocity gives
\begin{align*}
    \chi_p(r_j)=(-1)^{\frac{r_j-1}{2}\frac{p-1}{2}}\left(\frac{p}{r_j}\right).
\end{align*}
Since $p\equiv b\pmod 4$, $(-1)^{\frac{r_j-1}{2}\frac{p-1}{2}}$ is fixed. Thus, to make $\chi_p(r_j)=\eta_j$, $p$ must lie in a residue class modulo $r_j$ such that $\left(\frac{\cdot}{r_j}\right)$ has the required sign. Note that there are $(r_j-1)/2$ such nonzero residue classes modulo $r_j$.
For $r_j=2$, we have 
\begin{align*}
    \chi_p(2)=(-1)^{\frac{p^2-1}{8}}=\begin{cases}1& p\equiv1,7\pmod 8,\\ -1&p\equiv 3,5\pmod 8.\end{cases}
\end{align*}
Note that $p\equiv b\pmod 4$ gives $p\equiv b\pmod8$ or $p\equiv b+4 \pmod 8$, and each residue class gives different values of $\chi_p(2)$. Thus, to make $\chi_p(2)=\eta_j$, $p$ needs to lie in exactly one residue class modulo $8$.

Now, we write $\delta=1$ if some $r_j=2$ and $\delta=0$ otherwise. Then $t=\delta+\#\{j:r_j\neq2\}$. Let
$M=8\prod_{r_j\neq 2}r_j.$
By the Chinese remainder theorem and the argument above, there are
\begin{align*}
    2^{1-\delta}\prod_{r_j\neq2}\frac{r_j-1}{2}=\frac{\varphi(M)}{2^{t+1}}
\end{align*}
different residue classes modulo $M$ satisfying all conditions. Denote by $\mathcal{C}$ the set of these $\varphi(M)/2^{t+1}$ residue classes modulo $M$. Summing over  $\mathcal{C}$ and applying \eqref{eq:SiegelWalfisz}, we get 
\begin{align*}
N(X)&=\sum_{a\in\mathcal{C}}\pi(X;M,a)=|\mathcal{C}|\left(\frac{\operatorname{Li}(X)}{\varphi(M)}+o(\operatorname{Li}(X))\right)=2^{-(t+1)}\operatorname{Li}(X)+o(\operatorname{Li}(X)),
\end{align*}
which implies that
\begin{align*}
\lim_{X\rightarrow\infty}\frac{N(X)}{P_b(X)}&=\lim_{X\rightarrow\infty}\frac{2^{-(t+1)}\operatorname{Li}(X)+o(\operatorname{Li}(X))}{\frac{\operatorname{Li}(X)}{\varphi(4)}+o(\operatorname{Li}(X))}=2^{-t},
\end{align*}
as desired.

<!-- lemma-3: body_md -->
\label{lem:uniformproduct}
The products in \eqref{eq:abstractY} converge absolutely and uniformly
in $\boldsymbol{e}$, and their values are nonzero. The set
\begin{align}
 K=Y\left(\{-1,0,1\}^{\{{\rm primes}\}}\right) \label{eq:compactK}
\end{align}
is a compact subset of $(\mathbb C^\times)^d$. Moreover, for $R\geq2$,
\begin{align}\label{eq:uniformtail}
 \sup_{\boldsymbol{e}}\left|Y_i(\boldsymbol{e})-\prod_{r\leq R}V_{i,r}(e_r)\right|
 \ll_w\sum_{r>R}r^{-\sigma}
 \qquad(1\leq i\leq d).
\end{align}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Since $u_i(r)\in[-2,2]$, write $u_i(r)=2\cos\theta_{i,r}$ with
$\theta_{i,r}\in\mathbb R$. Then
\begin{align*}
 1-u_i(r)e r^{-w}+e^2r^{-2w}
 =(1-e\exp(\mathrm{i}\theta_{i,r})r^{-w})
  (1-e\exp(-\mathrm{i}\theta_{i,r})r^{-w}),
\end{align*}
where both factors are nonzero, since $|e r^{-w}|\leq r^{-\sigma}<1$.
The triangle inequality gives
\begin{align*}
 (1+r^{-\sigma})^{-2}\leq |V_{i,r}(e)|
 \leq(1-r^{-\sigma})^{-2}.
\end{align*}
Put $a_{i,r}(e)=V_{i,r}(e)-1$. Then
\begin{align*}
 |a_{i,r}(e)|
 \leq\frac{2r^{-\sigma}+r^{-2\sigma}}{(1-r^{-\sigma})^2}
 \leq C_\sigma r^{-\sigma},
 \qquad
 C_\sigma:=\frac{2+2^{-\sigma}}{(1-2^{-\sigma})^2},
\end{align*}
uniformly in $i$ and $e$.
Since $\sigma>1$, we have
\begin{align*}
 \sum_r r^{-\sigma}\leq\sum_{n=2}^{\infty}n^{-\sigma}<\infty,
 \qquad
 \sum_r|a_{i,r}(e_r)|
 \leq C_\sigma\sum_r r^{-\sigma}=:A_\sigma<\infty.
\end{align*}
Thus, $Y_i(\boldsymbol{e})=\prod_r(1+a_{i,r}(e_r)) $ converges absolutely, independently of $\boldsymbol{e}$. The non-vanishing of $Y_i(\boldsymbol{e})$ is from the fact that
\begin{align}\label{eq:productbounds}
|Y_i(\boldsymbol{e})|&\geq \prod_r(1+r^{-\sigma})^{-2}>0.
                                                   
                                               
\end{align}


We now prove the uniform  estimate \eqref{eq:uniformtail}. Note that for any finite set $B$ of primes, we have
\begin{align*}\left|\prod_{r\in B} V_{i,r}(e_r)\right|&\leq\prod_{r\in B}|1+a_{i,r}(e_r)|\\&\leq\exp\left(\sum_{r\in B}|a_{i,r}(e_r)|\right)\\&\leq \exp(A_{\sigma}).\end{align*}
It follows that for $S>R$, we have
\begin{align*}
 \left|\prod_{r\leq S}V_{i,r}(e_r)
       -\prod_{r\leq R}V_{i,r}(e_r)\right|
 &\leq \exp(A_{\sigma})\sum_{R<r\leq S}|a_{i,r}(e_r)|\\
 &\leq \exp(A_{\sigma})\cdot C_\sigma\sum_{R<r\leq S}r^{-\sigma},
\end{align*}
where the final sum tends to zero
as $R\to\infty$, independently of $S$ and $\boldsymbol{e}$. Hence the
partial products $\prod_{r\leq R}V_{i,r}(e_r)$ form a uniformly Cauchy family. Letting $S\to\infty$
proves uniform convergence and \eqref{eq:uniformtail}.


Finally, give $\{-1,0,1\}^{\{\text{primes}\}}$ the product topology,
with each factor discrete. Then $\{-1,0,1\}^{\{\text{primes}\}}$ is compact by Tychonoff's Theorem. Each finite Euler product $\prod_{r\leq R}V_{i,r}(e_r)$ depends on only
finitely many coordinates and is therefore continuous. Uniform
convergence shows that $Y$ is continuous. Since the domain of $Y$ is compact, the image $K$ is compact. Finally, by \eqref{eq:productbounds},
$K\subseteq(\mathbb C^\times)^d$. This completes the proof.

\label{lem:blockconditioning}
Let $B$ be a finite set of primes and
$\mathcal G_B=\sigma(\varepsilon_r:r\notin B)$ be the $\sigma$-algebra generated by the events $\{\varepsilon_r=1\}$ for $r\notin B$.
For $\omega\in\Omega$ and $\eta\in\{-1,1\}^B$, let
$\omega^{B,\eta}$ be obtained by replacing the coordinates of $\omega$
in $B$ by $\eta$. Then, for every $A\in\mathcal F$,
\begin{align}\label{eq:blockconditioning}
 \mu(A\mid\mathcal G_B)(\omega)
 =2^{-|B|}\sum_{\eta\in\{-1,1\}^B}
   \mathbf1_A(\omega^{B,\eta})
 \qquad\text{almost surely}.
\end{align}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Write $\Omega=\Omega_B\times\Omega_{B^c}$ and
$\mu=\mu_B\otimes\mu_{B^c}$, where
$\Omega_B=\{-1,1\}^B$ and $\mu_B(\{\eta\})=2^{-|B|}$.
The function
\begin{align*}
 g(\xi)=2^{-|B|}\sum_{\eta\in\Omega_B}\mathbf1_A(\eta,\xi)
\end{align*}
is measurable on $\Omega_{B^c}$.
Note that every $C\in\mathcal G_B$ has the form $\Omega_B\times D$ for a
measurable set $D\subseteq\Omega_{B^c}$, and
\begin{align*}
 \int_C g\,d\mu
 =2^{-|B|}\sum_{\eta\in\Omega_B}
   \int_D\mathbf1_A(\eta,\xi)\,d\mu_{B^c}(\xi)
 =\mu(A\cap C).
\end{align*}
Thus, $g$ has the defining property of $\mu(A\mid\mathcal G_B)$.

<!-- lemma-6: body_md -->
\label{lem:tensors}
For each pair $1\leq i<j\leq d$, choose a prime $r_{ij}$, with all
$\binom d2$ primes distinct, and put $B=\{r_{ij}:i<j\}$. For every
$r\in B$ and $1\leq i\leq d$, let
$v_{i,r}:\{-1,1\}\to\mathbb C$ be a nonzero function. Suppose that
$v_{i,r_{ij}}$ and $v_{j,r_{ij}}$ are not proportional for every $i<j$.
Then the following functions on the vectors
$\eta=(\eta_r)_{r\in B}\in\{-1,1\}^{B}$ are linearly independent:
\begin{align*}
 F_i(\eta)=\prod_{r\in B}v_{i,r}(\eta_r)
 \qquad(1\leq i\leq d).
\end{align*}
<!-- end lemma-6 -->
<!-- proof-5: body_md -->
For $d=1$, we have $B=\varnothing$ and hence $F_1\equiv1$,
so $\{F_1\}$ is linearly independent. Suppose $d\geq2$ and
$\sum_{k=1}^d a_kF_k=0$.
Fix $i$. We construct a linear functional $L_i$ such that 
$L_i(F_k)=0$ for $k\neq i$ and $L_i(F_i)\neq0$.

Write $r_{ji}=r_{ij}$. For $r=r_{ij}$ with $j\neq i$, define a
linear functional on the vector space of functions $h:\{-1,1\}\to\mathbb C$ by
\begin{align*}
 \ell_r(h)=v_{j,r}(-1)h(1)-v_{j,r}(1)h(-1).
\end{align*}
It follows that $\ell_r(v_{j,r})=0$, and
$
 \ell_r(v_{i,r})
 =v_{j,r}(-1)v_{i,r}(1)-v_{j,r}(1)v_{i,r}(-1)\neq0$
since $v_{i,r}$ and $v_{j,r}$ are not proportional. At every remaining coordinate $r\in B$, choose
$e_r\in\{-1,1\}$ with $v_{i,r}(e_r)\neq0$ and set
$\ell_r(h)=h(e_r)$. Such a sign exists since $v_{i,r}$ is a nonzero
function. Because the primes $r_{ij}$ are distinct, $\ell_r$ is well-defined.                                                          
For convenience, we write
$\ell_r(h)=b_r(1)h(1)+b_r(-1)h(-1)$. We define a linear functional
\begin{align*}
 L_i(H)=\sum_{\eta\in\{-1,1\}^{B}}
 \left(\prod_{r\in B}b_r(\eta_r)\right)H(\eta)
\end{align*}
for functions $H:\{-1,1\}^{B}\rightarrow\mathbb{C}$. It follows that
\begin{align*}
 L_i(F_k)
 &=\sum_{\eta\in\{-1,1\}^{B}}
       \prod_{r\in B}\bigl(b_r(\eta_r)v_{k,r}(\eta_r)\bigr)\\
 &=\prod_{r\in B}\left(\sum_{e\in\{-1,1\}}b_r(e)v_{k,r}(e)\right)
  \\&=\prod_{r\in B}\ell_r(v_{k,r}),
\end{align*}
which is nonzero when $k=i$ and zero otherwise.  Applying $L_i$ to the relation $\sum_{k=1}^d a_kF_k=0$ gives
$a_iL_i(F_i)=0$, hence $a_i=0$. Repeating this for each $i$
proves the lemma.

\end{align*}
It follows that
\begin{align}\label{eq:localindependent}
 v_{i,r}\not\parallel v_{j,r}

\end{align*}
Therefore, we have $\mu(A)=0$ from Lemma \ref{lem:conditionalgap}, completing the proof.

\label{prop:primeproducts}

\begin{align}\label{eq:abstractuniform}

For $R\geq2$, define the truncated coordinates and random vectors by
\begin{align*}

\end{align*}
which proves \eqref{eq:abstractuniform}.

Fix $b\in\{1,3\}$, $m\in\{1,2\}$, and $s\in\mathbb C$ with
$\operatorname{Re}(s)>\sigma_m$. Put
\begin{align*} v(p)&=(L^{(m)}(f_1,\chi_p,s),\ldots,L^{(m)}(f_d,\chi_p,s))^T

By \eqref{eq:standarduniform} and \eqref{eq:symuniform},
$B_X=o(P)$. 
Note that every proper subspace $W\subsetneq\mathbb C^d$ is contained in
$\{z:\boldsymbol{c}\cdot z=0\}$ for some $\boldsymbol{c}\neq0$. Hence
\[
 \#\{p\in\mathscr P_{4,b}(X):v(p)\in W\}\le B_X
 \qquad(W\subsetneq\mathbb C^d).
\]
Since each coordinate of $v(p)$ is a nonzero Euler product,  the
case $d=1$ is immediate. Now we assume $d\ge2$. If the matrix with
columns $v(p_1),\ldots,v(p_d)$ is singular, then there is a smallest

$v(p_1),\ldots,v(p_{j-1})$ are linearly independent and 
 $v(p_j)\in
 W:=\operatorname{Span}_{\mathbb C}\{v(p_1),\ldots,v(p_{j-1})\}.$
Fix this index $j$. There are at most $P^{j-1}$ ordered choices
of the preceding primes $p_1,\ldots,p_{j-1}$ whose corresponding columns  
$v(p_1),\ldots,v(p_{j-1})$ are linearly independent. For each such choice, $\dim W=j-1<d$, so at most
$B_X$ primes can be chosen for $p_j$. This bound is independent
of the preceding choices. The remaining primes
$p_{j+1},\ldots,p_d$ can be chosen in $P^{d-j}$ ways.
Therefore, the number of  tuples $(p_1,\ldots,p_d)$ whose first dependence
occurs at $j$ is at most
$$
 P^{j-1}B_XP^{d-j}=B_XP^{d-1}.$$
Summing over the possible indices of $j$ gives
\[
 \#\{(p_1,\ldots,p_d)\in\mathscr P_{4,b}(X)^d:
       \det(v(p_1),\ldots,v(p_d))=0\}
 \le (d-1)B_XP^{d-1}=o(P^d).
\]

\]
Outside the above union, the determinants are nonzero for every
$(m,s)\in T$. Thus, the proof is complete.

\label{thm:kernelspanning}
Fix $b\in\{1,3\}$, $m\in\{1,2\}$, and $s$ with
$\operatorname{Re}(s)>\sigma_m$. Then
\begin{align}\label{eq:kerneluniform}
 B_{m,s,b}(X):=\sup_{W\subsetneq S_k}
 \#\{p\in\mathscr P_{4,b}(X):K^{(m)}_{k,p,s}\in W\}
 =o(P_b(X)),
\end{align}
where $W$ runs over proper complex linear subspaces of $S_k$.
Consequently:
\begin{enumerate}
 \item For every $\delta>0$, there exists $X_0$ such that, for every
 $X\geq X_0$ and every $\mathcal A_X\subseteq\mathscr P_{4,b}(X)$ with
 $\#\mathcal A_X\geq\delta P_b(X)$,
 \begin{align*}
  \operatorname{Span}_{\mathbb C}\{K^{(m)}_{k,p,s}:p\in\mathcal A_X\}=S_k.
 \end{align*}
 \item If $\mathcal A\subseteq\mathscr P_{4,b}$ satisfies
 $\overline\delta_b(\mathcal A)>0$, then, for every $Y>0$,
 \begin{align*}
  \operatorname{Span}_{\mathbb C}\{K^{(m)}_{k,p,s}:p\in\mathcal A,\ p>Y\}=S_k.
 \end{align*}
 \item Almost every ordered tuple $(p_1,\ldots,p_d)\in\mathscr P_{4,b}(X)^d$
 gives a basis $K^{(m)}_{k,p_1,s},\ldots,K^{(m)}_{k,p_d,s}$ of $S_k$.
\end{enumerate}
The estimate \eqref{eq:kerneluniform} and assertions (i)--(iii) hold
simultaneously for any fixed finite collection of pairs $(m,s)$ with
$m\in\{1,2\}$ and $\operatorname{Re}(s)>\sigma_m$.

Note that every proper complex subspace $W\subsetneq S_k$ is contained in
$h^\perp=\{g\in S_k:\langle h,g\rangle=0\}$ for some $h\neq0$.

\end{align*}
which implies that $\overline\delta_b(\mathcal A_{>Y})=0$,

least one $(m,s)\in\mathscr T$. This completes the proof.

Since $f_1,\ldots,f_d$ form a basis of $S_k$, the periods form a
basis of $S_k^*$ exactly when the matrix
$R_t=(r_{t,\chi_{p_j}}(f_i))_{i,j=1}^d$ is nonsingular.

\begin{align*}
 \det R_t
 =\left(\frac{t!}{(-2\pi i)^{t+1}}\right)^d
\det(L(f_i,\chi_{p_j},t+1))_{1\leq i,j\leq d}.
\end{align*}
Here $t+1\geq k/2+1>(k+1)/2$, so
Corollary~\ref{cor:determinant} applies.

Now suppose $0\leq t\leq k/2-2$. For $f\in\mathcal H_k$, put
\begin{align*}
 \Lambda_{f,p}(s)
 =\left(\frac p{2\pi}\right)^s\Gamma(s)L(f,\chi_p,s).
\end{align*}
The quadratic-twist functional equation
\cite[Chapter~14]{IwaniecKowalski2004}, in the normalization recorded
in \cite[Section~2.1]{Ni2026}, is
\begin{align*}
 \Lambda_{f,p}(s)=i^k\chi_p(-1)\Lambda_{f,p}(k-s).
\end{align*}
Consequently,
\begin{align*}
 L(f,\chi_p,s)
 &=i^k\chi_p(-1)\left(\frac p{2\pi}\right)^{k-2s}
 \frac{\Gamma(k-s)}{\Gamma(s)}L(f,\chi_p,k-s).
\end{align*}
Setting $s=t+1$ and using \eqref{eq:defoftwistperiod} gives
\begin{align*}
 r_{t,\chi_{p_j}}(f_i)
 &=C_{p_j,t}L(f_i,\chi_{p_j},k-t-1),\\
 C_{p,t}
 &:=\frac{i^k\chi_p(-1)(k-t-2)!}{(-2\pi i)^{t+1}}
 \left(\frac p{2\pi}\right)^{k-2t-2},
\end{align*}
where the factor $C_{p,t}$ is nonzero and independent of $f_i$. Hence
\begin{align*}
 \det R_t
 =\left(\prod_{j=1}^d C_{p_j,t}\right)
 \det(L(f_i,\chi_{p_j},k-t-1))_{i,j=1}^d.
\end{align*}
Since $k-t-1\geq k/2+1>(k+1)/2$,
Corollary~\ref{cor:determinant} also applies in this range.
Finally, since there are only finitely many admissible indices $t$, the union
of their exceptional sets still has size $o(P_b(X)^d)$.

For $f\in\mathcal H_K$, the Rankin--Selberg identity recalled in
\cite[Lemma 2.1]{Ni2026} gives

 \langle f,F_p\rangle

\end{align}

\begin{align*}
 k_2+c>(K+1)/2,\qquad k_1+k_2+c-1>(K+1)/2.

 A=(\langle f_i,F_{p_j}\rangle)_{1\leq i,j\leq d_K},\qquad
 M=(L(f_i,\chi_{p_j},k_2+c))_{1\leq i,j\leq d_K}.

 A\overline{\boldsymbol{c}}=0
 \quad\Longleftrightarrow\quad
 \sum_{j=1}^{d_K}c_jF_{p_j}=0.
\end{align*}

linearly independent. Equation~\eqref{eq:rankinselbergid} gives

 \,M\,\operatorname{diag}(C_{p_j})_{j=1}^{d_K}.
\end{align*}
Note that both diagonal matrices are invertible, so $A$ is nonsingular exactly
when $M$ is. Corollary~\ref{cor:determinant}, applied in weight $K$,

<!-- lemma-9: body_md -->
\label{lem:determinantexpansion}
Suppose $d\geq2$. For any odd primes $p_1,\ldots,p_d$ and $w\in\mathbb C$
with $\operatorname{Re}(w)>1$,
\begin{align}\label{eq:determinantexpansion}
 \det(F_{i,p_j}(w))_{1\leq i,j\leq d}
 =\sum_{1\leq n_1<\cdots<n_d}
\frac{\det(c_i(n_\ell))_{1\leq i,\ell\leq d}
      \cdot \det(\chi_{p_j}(n_\ell))_{1\leq \ell,j\leq d}}
      {(n_1\cdots n_d)^w},
\end{align}
and the series converges absolutely.
<!-- end lemma-9 -->
<!-- proof-15: body_md -->
Put $u=\operatorname{Re}(w)>1$. For $N\geq d$, define the $d\times N$ and $N\times d$ matrices
\begin{align*}
 A_N&=(c_i(n)n^{-w})_{1\leq i\leq d,1\leq n\leq N},
 &C_N&=(\chi_{p_j}(n))_{\substack{1\leq n\leq N,1\leq j\leq d}},
\end{align*}
which gives
\begin{align*}
 M_N=A_NC_N
 =\left[\sum_{n\leq N}c_i(n)\chi_{p_j}(n)n^{-w}\right]_{1\leq i,j\leq d}.
\end{align*}
By the Cauchy--Binet formula \cite[p.~28]{Hornbook}, we have
\begin{align}\label{eq:finitecauchybinet}
 \det M_N
 =\sum_{1\leq n_1<\cdots<n_d\leq N}
 \frac{\det(c_i(n_\ell))_{1\leq i,\ell\leq d}
    \cdot\det(\chi_{p_j}(n_\ell))_{1\leq \ell,j\leq d}}
      {(n_1\cdots n_d)^w}.
\end{align}
Let $M=(F_{i,p_j}(w))_{1\leq i,j\leq d}$.
By \eqref{eq:reducedcoefficients} and $|n^{-w}|=n^{-u}$,
\begin{align*}
 \max_{i,j}|(M_N)_{ij}-M_{ij}|
 \leq\sum_{n>N}\frac{\tau(n)}{n^u}\longrightarrow0
 \qquad(N\to\infty),
\end{align*}
since $\sum_{n\geq1}\tau(n)n^{-u}=\zeta(u)^2<\infty$.
Continuity of the determinant therefore gives
\begin{align*}
 \lim_{N\to\infty}\det M_N=\det M.
\end{align*}
Hadamard's inequality \cite[p.~506]{Hornbook} and
$|\chi_{p_j}(n)|\leq1$ give
\begin{align}\label{eq:minorbound}
\left|\det(c_i(n_\ell))_{1\leq i,\ell\leq d}\cdot
       \det(\chi_{p_j}(n_\ell))_{1\leq \ell,j\leq d}\right|
 \leq d^d\prod_{\ell=1}^d\tau(n_\ell).
\end{align}
Consequently, for every integer $N\geq d$,
\begin{align}
 &\sum_{\substack{1\leq n_1<\cdots<n_d\nonumber\\ n_d>N}} \frac{\left|\det(c_i(n_\ell))_{1\leq i,\ell\leq d}
    \cdot\det(\chi_{p_j}(n_\ell))_{1\leq \ell,j\leq d}\right|}
      {(n_1\cdots n_d)^u}\\&\leq
 d^d\sum_{t>N}\frac{\tau(t)}{t^u}
       \sum_{n_1<\cdots<n_{d-1}<t}
       \prod_{\ell=1}^{d-1}\frac{\tau(n_\ell)}{n_\ell^u}\nonumber\\&
 \leq\frac{d^d\zeta(u)^{2d-2}}{(d-1)!}
\sum_{t>N}\frac{\tau(t)}{t^u},\label{eq:middlestep}
\end{align}
where we used the fact
\begin{align*}
 (d-1)!\sum_{n_1<\cdots<n_{d-1}<t}
       \prod_{\ell=1}^{d-1}\frac{\tau(n_\ell)}{n_\ell^u}
 &\leq \sum_{n_1,\ldots,n_{d-1}\geq1}
       \prod_{\ell=1}^{d-1}\frac{\tau(n_\ell)}{n_\ell^u}\\
 &=\left(\sum_{n\geq1}\frac{\tau(n)}{n^u}\right)^{d-1}
 \\&=\zeta(u)^{2d-2}.
\end{align*}
Taking $N=d$ in \eqref{eq:middlestep} proves absolute convergence
of the tail with $n_d>d$. The only remaining tuple is
$(1,\ldots,d)$, so the whole series in
\eqref{eq:determinantexpansion} converges absolutely.
As $N\to\infty$, the bound in \eqref{eq:middlestep} tends to zero,
so the finite sums in
\eqref{eq:finitecauchybinet} converge to the series in
\eqref{eq:determinantexpansion}. Together with
$\det M_N\to\det M$, this proves the identity.

Put $w=s-\sigma_m+1$, $u=\operatorname{Re}(w)$, and
$B=(\chi_{p_j}(n))_{1\leq n,j\leq d}$.
For $d=2$, condition~(i) gives $|\det B|=2$.
For $d=3$, its columns are $(1,\epsilon_j,\eta_j)^T$, where
the $(\epsilon_j,\eta_j)$ are three distinct elements of
$\{-1,1\}^2$. Every such choice gives $|\det B|=4$.

Separating the term $(n_1,\ldots,n_d)=(1,\ldots,d)$ from
the remaining terms in \eqref{eq:determinantexpansion}, we write
\begin{align}\label{eq:leadingdeterminant}
 \det(F_{i,p_j}(w))_{1\leq i,j\leq d}=D_{k,m}(\det B)(d!)^{-w}+E_d(w).
\end{align}
Let $Q_2=3$ and $Q_3=10$. Every nonzero term in $E_d(w)$ has
$n_1\cdots n_d\geq Q_d$. For $d=2$ this is immediate.
For $d=3$, the only remaining triple with product less than $10$
is $(1,2,4)$; the corresponding determinant $\det(\chi_{p_j}(n_\ell))_{1\leq\ell,j\leq3}$ vanishes because
$\chi_{p_j}(1)=\chi_{p_j}(4)=1$ for every $j$.
By \eqref{eq:minorbound}, for
$u\geq2$ we obtain
\begin{align*}
 |E_d(w)|
 &\leq d^dQ_d^{2-u}
 \sum_{n_1<\cdots<n_d}
       \prod_{\ell=1}^d\frac{\tau(n_\ell)}{n_\ell^2}\\
 &\leq\frac{d^d}{d!}Q_d^{2-u}\zeta(2)^{2d}.
\end{align*}
                            
               
                                                                                                                                                                                                                      
             
In particular,
\begin{align}\label{eq:determinanttails}
 |E_2(w)|\leq18\zeta(2)^4\,3^{-u},\qquad
 |E_3(w)|\leq450\zeta(2)^6\,10^{-u}.
\end{align}
For either $m\in\{1,2\}$, Lemma~\ref{lem:smallcoefficientdet}
gives $|D_{k,m}|\geq1/4$. Hence the leading term in
\eqref{eq:leadingdeterminant} satisfies
\begin{align*}
 |D_{k,m}(\det B)(d!)^{-w}|
 =|D_{k,m}\det B|(d!)^{-u}
 \geq\begin{cases}
 \frac12\,2^{-u},&d=2,\\
 6^{-u},&d=3.
 \end{cases}
\end{align*}
For $u\geq18$, the bounds in \eqref{eq:determinanttails} give
\begin{align*}
 \frac{|E_2(w)|}{\frac12\,2^{-u}}
 &\leq36\zeta(2)^4\left(\frac23\right)^u<1,\\
 \frac{|E_3(w)|}{6^{-u}}
 &\leq450\zeta(2)^6\left(\frac35\right)^u<1.
\end{align*}
Thus, the leading term in \eqref{eq:leadingdeterminant} strictly exceeds the error term $E_d(w)$ in absolute
value, and $\det(F_{i,p_j}(w))\neq0$ for
$u\geq18$, equivalently $\operatorname{Re}(s)\geq\sigma_m+17$.
For $m=2$, the Euler product gives $L(w,\chi_{p_j})\neq0$
when $\operatorname{Re}(w)>1$. Multiplying the $j$th column of
$(F_{i,p_j}(w))_{1\leq i,j\leq d}$ by this factor recovers
the matrix in \eqref{eq:explicitdet}.
The matrix $(\mathcal L^{(m)}_{p_j,s}(f_i))_{1\leq i,j\leq d}$
is therefore nonsingular, so these $d$ linear functionals form a
basis of $S_k^*$. The kernel representation \eqref{eq:kernelspectral}
then gives the corresponding basis of $S_k$.
