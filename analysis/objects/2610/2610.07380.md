<!-- definition-1: body_md -->
A nonnegative $(\mathcal F_t)_{t\ge0}$-adapted process
		$A=(A_t)_{t\ge0}$ is called a \emph{positive continuous additive functional}
		(PCAF) of $Y$ if $A_0=0$ and, under every admissible starting law, its paths
		are almost surely continuous and nondecreasing, and satisfy
		\begin{equation}
			\label{eq:PCAF-additivity}
			A_{t+s}
			=
			A_t+A_s\circ\theta_t,
			\qquad s,t\ge0.
		\end{equation}

<!-- lemma-1: body_md -->
\label{lem:AF-conditioning}
		Let \(F\) be a nonnegative measurable functional of the future base path such that
		\[
		K:=\sup_{z\in E}\operatorname{E}_z F<\infty.
		\]
		For deterministic $0\le a\le b$,
		\begin{equation}\label{eq:AF-conditioning}
			\operatorname{E}_y\int_a^b F\circ\theta_s\,dA_s
			\le K\operatorname{E}_y(A_b-A_a).
		\end{equation}
		Moreover, for every integer $p\ge1$,
		\begin{equation}\label{eq:abstract-A-moment}
			\sup_y\operatorname{E}_y A_t^p\le p!M(t)^p\le p!C_T^pt^{p\vartheta}.
		\end{equation}
		For every finite real $q\ge1$ this implies
		\begin{equation}\label{eq:abstract-A-increments}
			\sup_y\operatorname{E}_y(A_v-A_u)^q\le C_{T,q}(v-u)^{q\vartheta},
			\qquad 0\le u<v\le T.
		\end{equation}

<!-- proof-1: body_md -->
Set $\tau_\ell=\inf\{s:A_s>\ell\}$, with $\inf\emptyset=\infty$.
		Then $\tau_\ell$ is a stopping time, and the Stieltjes change of variables gives  (see \cite[Proposition~0.4.10]{revuz1999Continuous_martingales})
		\[
		\int_a^b F\circ\theta_s\,dA_s
		=\int_0^\infty\mathbf1_{\{a\le\tau_\ell<b\}}
		F\circ\theta_{\tau_\ell}\,d\ell.
		\]
		Endpoint and plateau levels have zero Lebesgue measure. Conditioning at
		$\tau_\ell$ and using the strong Markov property gives the bound
		$K\operatorname{E}_y\int_0^\infty\mathbf1_{\{a\le\tau_\ell<b\}}d\ell$, which is
		\eqref{eq:AF-conditioning}. 
		The interchange of expectation and integration is justified by localization in time and level and then applying monotone convergence.
		
		For integer $p\ge2$, continuity and monotonicity give
		\[
		A_t^p=p\int_0^t(A_t-A_s)^{p-1}\,dA_s.
		\]
		Since $A_t-A_s=A_{t-s}\circ\theta_s\le A_t\circ\theta_s$, induction and \eqref{eq:AF-conditioning} give
		\[
		\operatorname{E}_y A_t^p\le p(p-1)!M(t)^{p-1}\operatorname{E}_y A_t\le p!M(t)^p.
		\]
		All integrals are nonnegative, so Tonelli and truncation justify the induction
		without any prior moment assumption. Applying the same argument after time
		$u$ gives the increment estimate for integer powers; Lyapunov's inequality
		with $p=\lceil q\rceil$ gives the real exponents. 
		This is the standard ordered-integral (Kac moment) argument for positive continuous additive functionals; see \cite[Section~3.1]{fitzsimmons1999kac}.
		The same Kac moment machinery also yields the two-point local-time moment identity used in Example~\ref{ex:abstract-benchmarks}.

<!-- theorem-1: body_md -->
\label{thm:abstract-averaging}
		Assume \eqref{eq:abstract-A-first}. The following estimates hold uniformly in the initial states $y\in E$ and $i\in S$, for $0<\varepsilon\le1$.
		\begin{enuroman}
			\item\label{item:abstract-random-time}
			For every $[0,T]$-valued random time $\tau$ measurable with respect to the base path,
			\begin{equation}\label{eq:abstract-random-time}
				\operatorname{E}_{y,i}|J_\varepsilon(\tau)|^2\le C_{T,h}\varepsilon^\vartheta.
			\end{equation}
			No stopping-time property is required. Once the base path is fixed,
			both $dA$ and the upper limit $\tau$ are deterministic from the point of
			view of the chain.
			\item\label{item:abstract-max}
			The uniform estimate is
			\begin{equation}\label{eq:abstract-max}
				\operatorname{E}_{y,i}\sup_{0\le t\le T}|J_\varepsilon(t)|^2
				\le C_{T,h}\varepsilon^\vartheta(1+|\log\varepsilon|)^2.
			\end{equation}
			In particular, $J_\varepsilon\to0$ in $L^2$ for the uniform norm on $C([0,T])$ under every starting law, uniformly in the starting states.
		\end{enuroman}

<!-- proof-2: body_md -->
\emph{Step 1: mixing and discounted potentials.} 
		Since \(S\) is finite and the chain is irreducible, the transition
		semigroup \(P_t^\alpha=e^{tQ}\) is exponentially ergodic: there exist
		constants \(C,c>0\) such that
		\[
		\sup_{i,j\in S}
		\left|P_{ij}^\alpha(t)-\pi_j\right|
		\le Ce^{-ct},
		\qquad t\ge0;
		\]
		see, e.g., \cite[Section~6.6]{anderson1991_book}. For bounded \(h\),
		with
		\[
		\widetilde h:=h-\pi(h),
		\qquad
		\pi(\widetilde h)=0,
		\]
		we have
		\[
		\|P_t^\alpha\widetilde h\|_\infty
		\le C_h e^{-ct},
		\qquad t\ge0,
		\]
		where \(C_h:=|S|C\|\widetilde h\|_\infty\). Therefore, for \(r\le s\),
		the Markov property gives
		\begin{align}\label{eq:abstract-covariance}
			\left|\operatorname{E}_i^\alpha[\widetilde h(\alpha_{r/\varepsilon})
			\widetilde h(\alpha_{s/\varepsilon})]\right|
			&=\left|\operatorname{E}_i^\alpha[\widetilde h(\alpha_{r/\varepsilon})
			(P^\alpha_{(s-r)/\varepsilon}\widetilde h)(\alpha_{r/\varepsilon})]\right|\notag\\
			&\le C_he^{-c(s-r)/\varepsilon}.
		\end{align}
		No stationarity is used here. Integrating by parts against $\operatorname{E}_yA_s$ gives
		\begin{align}\label{eq:abstract-weighted-potential}
			U_\varepsilon(T):=\sup_y\operatorname{E}_y\int_0^T e^{-cs/\varepsilon}\,dA_s
			&\le e^{-cT/\varepsilon}M(T)
			+\frac c\varepsilon\int_0^T e^{-cs/\varepsilon}M(s)\,ds\notag\\
			&\le C_T\varepsilon^\vartheta.
		\end{align}
		
		\medskip\noindent\emph{Step 2: interval and random-time estimates.}
		Lemma~\ref{lem:AF-conditioning} gives the integrability needed for Fubini.
		Condition on the base path. Since \(\tau\) is measurable with respect to
		that path, both \(A\) and \(\tau\) are then fixed. Since \(A\) is continuous,
		its Stieltjes measure \(dA\) is atomless, so the diagonal has zero
		\(dA\otimes dA\)-measure.
		Therefore,
		\[
		\operatorname{E}_{y,i}|J_\varepsilon(\tau)|^2
		\le 2C_h\operatorname{E}_y\int_0^T\int_s^T
		e^{-c(r-s)/\varepsilon}\,dA_r\,dA_s
		\le 2C_h U_\varepsilon(T)\operatorname{E}_y A_T,
		\]
		where \eqref{eq:AF-conditioning} is applied to
		$F=\int_0^T e^{-cu/\varepsilon}\,dA_u$ after enlarging the remaining
		future horizon to $T$. 
		Together with
		\eqref{eq:abstract-weighted-potential} and
		our assumption \eqref{eq:abstract-A-first}, this proves
		\eqref{eq:abstract-random-time}.
		
		The same calculation on a deterministic interval \([a,b]\) yields
		\begin{equation}\label{eq:abstract-interval}
			\operatorname{E}_{y,i}|J_\varepsilon(b)-J_\varepsilon(a)|^2
			\le
			C_{T,h}\varepsilon^\vartheta
			\operatorname{E}_y(A_b-A_a),
			\qquad 0\le a\le b\le T.
		\end{equation}
		The dependence on the interval through the increment \(A_b-A_a\) will be
		important below. Indeed, if
		\[
		0=t_0<t_1<\cdots<t_N=T
		\]
		is any deterministic partition of \([0,T]\), then
		\[
		\sum_{k=0}^{N-1}
		\operatorname{E}_y\!\left(A_{t_{k+1}}-A_{t_k}\right)
		=
		\operatorname{E}_y A_T.
		\]
		Thus, summing \eqref{eq:abstract-interval} over all intervals at one dyadic
		level introduces no factor proportional to the number of intervals.
		
		\medskip\noindent\emph{Step 3: dyadic maximal bound.}
		Let \(m\in\mathbb N\), and let
		\[
		G_m=\{kT2^{-m}:0\le k\le2^m\}
		\]
		be the dyadic grid of mesh size \(T2^{-m}\) on \([0,T]\), and for
		\(0\le j\le m\) and \(0\le k<2^j\), define
		\[
		D_{j,k}
		:=
		J_\varepsilon((k+1)T2^{-j})
		-
		J_\varepsilon(kT2^{-j}),
		\]
		the increment of \(J_\varepsilon\) over the \(k\)-th dyadic interval at
		level \(j\).
		Every interval \([0,t]\), with \(t\in G_m\), admits a dyadic decomposition
		into disjoint intervals, containing at most one interval from each level
		\(0\le j\le m\). Hence the corresponding telescoping decomposition of
		\(J_\varepsilon(t)\) yields
		\[
		\max_{t\in G_m}|J_\varepsilon(t)|
		\le\sum_{j=0}^m\max_{k<2^j}|D_{j,k}|.
		\]
		At each level the intervals partition $[0,T]$, and \eqref{eq:abstract-interval} implies
		\[
		\operatorname{E}_{y,i}\max_{k<2^j}|D_{j,k}|^2
		\le\sum_{k<2^j}\operatorname{E}_{y,i}|D_{j,k}|^2
		\le C_{T,h}\varepsilon^\vartheta\operatorname{E}_y A_T.
		\]
		Therefore, by \eqref{eq:abstract-A-first} and Minkowski's inequality,
		\begin{equation}\label{eq:abstract-grid-points}
			\left(\operatorname{E}_{y,i}\max_{t\in G_m}|J_\varepsilon(t)|^2\right)^{1/2}
			\le C_{T,h}(m+1)\varepsilon^{\vartheta/2}.
		\end{equation}
		
		Choose an integer $p>\max\{2,1/\vartheta\}$ and set $\delta:=T2^{-m}$. Between grid points,
		\[
		|J_\varepsilon(t)-J_\varepsilon(k\delta)|
		\le\|\widetilde h\|_\infty(A_{(k+1)\delta}-A_{k\delta}).
		\]
		Using \eqref{eq:abstract-A-increments} and concavity of $x^{2/p}$,
		\[
		\operatorname{E}_y\max_{k<2^m}(A_{(k+1)\delta}-A_{k\delta})^2
		\le\left(\sum_{k<2^m}\operatorname{E}_y(A_{(k+1)\delta}-A_{k\delta})^p\right)^{2/p}
		\le C_{T,p}\delta^{2\vartheta-2/p}.
		\]
		Consequently,
		\begin{equation}\label{eq:abstract-balance}
			\operatorname{E}_{y,i}\sup_{t\le T}|J_\varepsilon(t)|^2
			\le C_{T,h,p}\big((m+1)^2\varepsilon^\vartheta
			+\delta^{2\vartheta-2/p}\big).
		\end{equation}
		For small $\varepsilon$, choose the dyadic mesh so that
		$\delta\asymp\varepsilon^{\vartheta/(2\vartheta-2/p)}$. Then
		$m= \log_2(T/\delta) =O(1+|\log\varepsilon|)$ and the interpolation term is
		$O(\varepsilon^\vartheta)$. The remaining values of $\varepsilon\in(0,1]$
		are absorbed into the constant using
		$|J_\varepsilon|\le\|\widetilde h\|_\infty A_T$.
		This proves the uniform-in-time bound, and completes the proof.

<!-- corollary-1: body_md -->
\label{cor:abstract-all-moments}
		Under \eqref{eq:abstract-A-first}, for every $0<\zeta<\vartheta$, 
		\begin{equation}\label{eq:abstract-all-moments-max}
			\sup_{y,i}\operatorname{E}_{y,i}\sup_{t\le T}|J_\varepsilon(t)|^2

		\end{equation}

<!-- remark-1: body_md -->
The integrand may also depend on the base process, provided
the switching chain remains independent of it.
Indeed, let \(g:E\times S\to\mathbb R\) be bounded and measurable, and set
\[
\overline g(y):=\sum_{i\in S}\pi_i g(y,i).
\]
Then the same estimates apply to
\[
J_\varepsilon^g(t)
=
\int_0^t
\bigl[g(Y_s,\alpha_{s/\varepsilon})-\overline g(Y_s)\bigr]\,dA_s.
\]
Indeed, conditional on the base path, the functions
\[
h_s(i):=g(Y_s,i)-\overline g(Y_s)
\]
are deterministic, centered with respect to \(\pi\), and uniformly bounded by
\(2\|g\|_\infty\). Exponential ergodicity of the chain therefore gives, uniformly
in \(r\le s\),
\[
\left|
\operatorname{E}_i^\alpha
\left[
h_r(\alpha_{r/\varepsilon})
h_s(\alpha_{s/\varepsilon})
\right]
\right|
\le
C_g e^{-c(s-r)/\varepsilon}.
\]
The proof of Theorem~\ref{thm:abstract-averaging} then applies without change.
Consequently, \eqref{eq:abstract-random-time} and
\eqref{eq:abstract-max} remain valid, with constants depending on
\(\|g\|_\infty\).

<!-- example-1: body_md -->
\label{ex:abstract-benchmarks}
		For bounded nonnegative $V$, $A_t=\int_0^t V(Y_s)\,ds$ satisfies \eqref{eq:abstract-A-first} with $\vartheta=1$. A singular example, without a reflecting boundary, is the symmetric local time $\ell_t^0$ at zero of  standard one-dimensional Brownian motion $B$. 
		With $p_s(x,y)$ its heat kernel,
		\[
		\sup_x\operatorname{E}_x\ell_t^0
		=\sup_x\int_0^t p_s(x,0)\,ds
		=\sqrt{2t/\pi},
		\]
		so the bound~\eqref{eq:abstract-A-first} holds for $\vartheta=1/2$. 
		The identity
		\[
		\operatorname{E}_x\ell_t^0=\int_0^t p_s(x,0)\,ds
		\]
		follows from the occupation-density formula
		\cite[Corollary~VI.1.9]
		{revuz1999Continuous_martingales}.
		
		The exponent $\vartheta=1/2$ is attained in the following simple benchmark.
		Let \(B_0=0\) and let \(H\) be an independent sign chain on
		\(S=\{-1,1\}\) with generator and invariant law
		\[
		Q=
		\begin{pmatrix}
			-a & a\\
			a & -a
		\end{pmatrix},
		\qquad
		\pi=
		\left(\frac12,\frac12\right).
		\]
		
		Assume that \(H\) is started from \(\pi\). Then \(H\) is stationary and centered in the sense that \(\operatorname{E}(H_t)=0\) for all \(t\ge 0\), and its covariance formula reads
		\begin{equation}
			\label{eq:covariance_formula}
			\operatorname{E}[H_sH_r]=e^{-2a|s-r|}.
		\end{equation}
		
		Conditioning on the Brownian path and using the independence of \(H\) and
		\(B\), together with~\eqref{eq:covariance_formula}, gives
		\begin{align}
			\operatorname{E}\left|\int_0^T H_{s/\varepsilon}\,d\ell_s^0\right|^2
			&= \operatorname{E}\left[\int_0^T \int_0^T H_{r/\varepsilon}\,H_{s/\varepsilon}\,d\ell_r^0\,d\ell_s^0\right]
			\\
			&= \operatorname{E}\left[
			\int_0^T\int_0^T
			e^{-2a|s-r|/\varepsilon}\,
			d\ell_r^0\,d\ell_s^0
			\right].
		\end{align}
		For every nonnegative Borel function \(g\), applying the space--time
		occupation-times formula
		\cite[Exercise~VI.1.15]{revuz1999Continuous_martingales}
		to \((s,a)\mapsto g(s)\eta_\delta(a)\), where
		\[
		\eta_\delta(a):=\frac{1}{2\delta}\mathbf 1_{\{|a|<\delta\}},
		\]
		then taking expectations and letting \(\delta\downarrow0\), gives
		\[
		\operatorname{E}_x\left[\int_0^t g(s)\,d\ell_s^0\right]
		=
		\int_0^t g(s)p_s(x,0)\,ds.
		\]
		Using this identity together with Tonelli's theorem, the additivity of
		local time, and the Markov property, for every nonnegative Borel function
		\(\varphi\) on \(\{(r,s):0<r<s<T\}\),
		\[
		\begin{aligned}
			&\operatorname{E}_0\left[
			\int_0^T\int_0^s
			\varphi(r,s)\,d\ell_r^0\,d\ell_s^0
			\right]
			\\
			&\qquad=
			\operatorname{E}_0\left[
			\int_0^T
			\left(
			\int_r^T \varphi(r,s)\,d\ell_s^0
			\right)
			d\ell_r^0
			\right]
			\\
			&\qquad=
			\int_0^T
			p_r(0,0)
			\left(
			\int_0^{T-r}
			\varphi(r,r+u)\,
			p_u(0,0)\,du
			\right)dr
			\\
			&\qquad=
			\int_0^T\int_0^s
			\varphi(r,s)\,
			p_r(0,0)p_{s-r}(0,0)\,dr\,ds.
		\end{aligned}
		\]
		
		Since \(\ell^0\) is continuous, its Stieltjes measure \(d\ell^0\) is
		atomless, and therefore \(d\ell^0\otimes d\ell^0\) gives no mass to the
		diagonal. Taking
		\[
		\varphi(r,s)=e^{-2a(s-r)/\varepsilon}
		\]
		and using symmetry, we obtain
		\begin{align}
			\operatorname{E}\left|\int_0^T H_{s/\varepsilon}\,d\ell_s^0\right|^2
			&=
			2\int_{0<r<s<T}
			e^{-2a(s-r)/\varepsilon}
			p_r(0,0)p_{s-r}(0,0)\,dr\,ds
			\\
			&=
			2\int_0^T e^{-2au/\varepsilon}p_u(0,0)
			\int_0^{T-u}p_s(0,0)\,ds\,du
			\\
			&=
			\frac2\pi\int_0^T e^{-2au/\varepsilon}
			\sqrt{\frac{T-u}{u}}\,du
			\sim
			\sqrt{\frac{2T}{\pi a}}\,\varepsilon^{1/2}.
			\label{eq:interior-localtime-variance}
		\end{align}
		For the asymptotic, substitute $u=\varepsilon v$ and apply dominated
		convergence with the integrable bound
		\(\sqrt T\,e^{-2av}v^{-1/2}\). 
		
		By contrast, using the covariance formula~\eqref{eq:covariance_formula} with \(\lambda=2a/\varepsilon\),
		we get 
		\begin{align}
			\operatorname{E}\left|\int_0^T H_{s/\varepsilon}\,ds\right|^2
			&=
			\int_0^T\int_0^T e^{-\lambda|s-r|}\,dr\,ds \\
			&=
			2\int_0^T \frac{1-e^{-\lambda s}}{\lambda}\,ds \\
			&=
			2\left(
			\frac{T}{\lambda}
			-
			\frac{1-e^{-\lambda T}}{\lambda^2}
			\right)
			\sim \frac{T}{a}\varepsilon.
			\label{eq:interior-occupationtime-variance}
		\end{align}
		The asymptotics~\eqref{eq:interior-localtime-variance} and
		\eqref{eq:interior-occupationtime-variance} show that ordinary occupation
		and local-time occupation fluctuate on different scales. In particular,
		within this class the exponent \(1/2\) in the terminal-time second-moment
		estimate for local-time occupation cannot be improved.

<!-- remark-2: body_md -->
\label{rem:abstract-boundary-local-time}
		Unlike the one-dimensional interior-local-time benchmark of
		Example~\ref{ex:abstract-benchmarks}, we consider here the boundary
		local time of reflected Brownian motion in a \(C^2\) domain \(D\).
		Appendix~\ref{app:averaging} shows that \(\vartheta=1/2\) in this case.
		Consequently, the random-time squared \(L^2\) error is
		\(O(\varepsilon^{1/2})\), while the maximal error is
		\(O(\varepsilon^{1/2}(1+|\log\varepsilon|)^2)\), and hence
		\(O(\varepsilon^\zeta)\) for every \(0<\zeta<1/2\).
		
		By Lemma~\ref{lem:AF-conditioning}, the first-moment estimate for the
		boundary local time also yields, for every finite \(q\ge1\),
		\[
		\sup_{x\in\overline D}\operatorname{E}_x[L_t^q]
		\le C_{T,q}t^{q/2},
		\qquad 0\le t\le T,
		\]
		and
		\[
		\sup_{x\in\overline D}
		\operatorname{E}_x[(L_v-L_u)^q]
		\le C_{T,q}(v-u)^{q/2},
		\qquad 0\le u<v\le T.
		\]
		Thus no additional moment assumption on the boundary local time is
		required beyond the short-time first-moment estimate.

<!-- definition-2: body_md -->
\label{def:SR_EK}
	We use the following terminology and associated smooth boundary classes.
	\begin{enuroman}
		\item The instantaneously reflected Brownian motion corresponds to

<!-- proof-4: body_md -->
The independent processes \(W^+\) and \(\alpha\) are strong Markov, hence
		their product
		\[
		Z_t=(W_t^+,\alpha_t)
		\]
		is a c\`adl\`ag strong Markov process on
		\(E=\overline D\times S\).
		Its path regularity follows from the continuity of \(W^+\) and the
		c\`adl\`ag paths of \(\alpha\).
		
		\emph{Step 1: Applying the time-change.} The clock
		\[
		A(t)=t+\int_0^t\rho(\alpha_s)\,dL_s
		\]
		is a continuous additive functional of \(Z\), by the additivity of the
		boundary local time. Moreover, for \(0\le s\le t\),
		\[
		A(t)-A(s)
		=
		t-s+\int_s^t\rho(\alpha_u)\,dL_u
		\ge t-s.
		\]
		Hence \(A\) is strictly increasing and tends to infinity, and its inverse
		\(\gamma\) is finite, continuous, nondecreasing, and \(1\)-Lipschitz:
		\[
		A(\gamma(t))=t,
		\qquad
		|\gamma(t)-\gamma(s)|\le |t-s|.
		\]
		
		The inverse clock satisfies the usual composition rule
		\[
		\gamma(t+s)
		=
		\gamma(t)+\gamma(s)\circ\theta_{\gamma(t)},
		\]
		as well as the corresponding stopping-time version; see
		\cite[Chapter~V, Proposition~2.3]{BlumenthalGetor68}.
		
		By the standard time-change theorem for continuous additive functionals,
		the process
		\[
		\widehat Z_t:=Z_{\gamma(t)}
		=
		\bigl(W^+_{\gamma(t)},\alpha_{\gamma(t)}\bigr)
		=
		(X_t,I_t)
		\]
		is strong Markov with respect to the time-changed filtration
		\[
		\widehat{\mathcal F}_t=\mathcal F_{\gamma(t)};
		\]
		see \cite[Chapter~V, Proposition~2.3 and Exercise~2.11]{BlumenthalGetor68} or \cite[Theorem~10.10]{dynkin1965Markov_processes}.
		
		\emph{Step 2: Applying the killing functional.} Since \(\gamma\) is continuous and nondecreasing, \(X_t=W^+_{\gamma(t)}\)
		is continuous and \(I_t=\alpha_{\gamma(t)}\) is c\`adl\`ag. Hence
		\(\widehat Z=(X,I)\) is c\`adl\`ag.
		
		Define the operational-time killing functional
		\[
		B(t):=\int_0^t\kappa(\alpha_s)\,dL_s.
		\]
		Then \(B\) is a continuous additive functional of \(Z\), and
		\[
		K(t)=B(\gamma(t)).
		\]
		By the inverse-clock composition rule,
		\[
		K(t+s)=K(t)+K(s)\circ\widehat\theta_t,
		\]
		with the corresponding stopping-time version. Consequently,
		\[
		M_t:=e^{-K(t)}
		\]
		is a continuous strong multiplicative functional of \(\widehat Z\); see
		\cite[Chapter~III, Definition~3.11]{BlumenthalGetor68}.
		
		The standard multiplicative-functional construction therefore yields the
		killed subprocess
		\[
		\mathcal Y_t=
		\begin{cases}
			(X_t,I_t),&t<\zeta,\\
			\dagger,&t\ge\zeta,
		\end{cases}
		\]
		which is a c\`adl\`ag strong Markov process; see
		\cite[Chapter~III, Theorem~3.3 and Proposition~3.12]{BlumenthalGetor68}.
		
		
		\emph{Step 3: Identification of transition law.} 
		Since \(K(t)\) is measurable with respect to the path of \(Z\), independence
		of \(\mathcal E\sim\operatorname{Exp}(1)\) gives
		\[
		\operatorname{P}(t<\zeta\mid \sigma(Z_s:s\ge0))
		=
		\operatorname{P}(\mathcal E\ge K(t)\mid \sigma(Z_s:s\ge0))
		=
		e^{-K(t)}.
		\]
		Hence, for every bounded Borel function \(\Phi\),
		\begin{align}
			\operatorname{E}_{x,i}\!\left[
			\Phi(X_t,I_t)\mathbf 1_{\{t<\zeta\}}
			\right]
			&=
			\operatorname{E}_{x,i}\!\left[
			\Phi(X_t,I_t)e^{-K(t)}
			\right]
			\\
			&=
			\operatorname{E}_{x,i}\!\left[
			\Phi\!\left(
			W^+_{\gamma(t)},\alpha_{\gamma(t)}
			\right)
			e^{-K(t)}
			\right].
		\end{align}
		This is \eqref{eq:construction-semigroup}.

		\label{eq:Delta_epsilon}

		\ge -c_D|z-w|^2,
		\qquad z\in\partial D,\quad w\in\overline D.

		\[
		dZ_t
		=
		-\mathbf n(W_t^x)\,dL_t^x
		+\mathbf n(W_t^y)\,dL_t^y.

		\[
		d|Z_t|^2
		=
		-2 Z_t\cdot \mathbf n(W_t^x)\,dL_t^x
		+2 Z_t\cdot \mathbf n(W_t^y)\,dL_t^y.
		\]

		observables converge in probability. Since they are uniformly bounded by
		\(\|\Phi\|_\infty\), the convergence also holds in \(L^1\). Therefore
		\[
		(\mathcal S_t\Phi)(x_m,i)\to(\mathcal S_t\Phi)(x,i).
		\]
		
		\noindent\emph{Step 3: strong continuity.}
		Let
		\[
		q_*:=\max_{i\in S}(-q_{ii}).
		\]
		Because $\gamma(t)\le t$, the chain changes state before
		$\gamma(t)$ with probability at most $q_*t$. The reflected motion is
		uniformly stochastically continuous on the compact state space (recall the proof of Theorem~\ref{thm:quenched_sticky_elastic}). Hence uniform
		continuity of $\Phi$ gives
		\[
		\sup_{(x,i)}
		\operatorname{E}_{x,i}\!\left[
		\left|
		\Phi(W^+_{\gamma(t)},\alpha_{\gamma(t)})
		-\Phi(x,i)
		\right|
		\right]
		\longrightarrow0.
		\]
		The killing term satisfies
		\[
		\operatorname{E}_{x,i}[1-e^{-K(t)}]
		\le
		\|\kappa\|_\infty\operatorname{E}_x[L_t],
		\]
		which also tends to zero uniformly in $x$. Thus
		\[
		\|\mathcal S_t\Phi-\Phi\|_\infty\longrightarrow0,
		\qquad t\downarrow0.
		\]
		This proves strong continuity.
		
		\medskip
		\noindent\emph{Step 4: identification of the generator.}
		Let \(\Phi\in\mathcal D_0\), and set
		\[
		B(s):=\int_0^s\kappa(\alpha_u)\,dL_u.
		\]
		Applying the generalized It\^o formula for switching diffusions to
		\(\Phi(W_s^+,\alpha_s)\), together with the reflection term, gives
		\[
		d\Phi(W_s^+,\alpha_s)
		=
		d\mathcal M_s
		+
		(\mathcal L\Phi)_{\alpha_s}(W_s^+)\,ds
		-
		\partial_{\mathbf n}\Phi_{\alpha_s}(W_s^+)\,dL_s,
		\]
		where \(\mathcal M\) is a local martingale. Here the compensated jumps of
		\(\alpha\) produce the term
		\[
		\sum_{j\in S}q_{\alpha_s j}\Phi_j(W_s^+)\,ds
		\]
		appearing in \(\mathcal L\); see, e.g.,
		\cite[Section~2.2]{ngyuen2025_book}.
		
		Since
		\[
		d(e^{-B(s)})
		=
		-e^{-B(s)}\kappa(\alpha_s)\,dL_s,
		\]
		the product rule yields
		\begin{align}
			&e^{-B(\gamma(t))}
			\Phi(W^+_{\gamma(t)},\alpha_{\gamma(t)})
			-\Phi(x,i)
			\nonumber\\
			&=
			\mathcal N_t
			+
			\int_0^{\gamma(t)}
			e^{-B(s)}
			(\mathcal L\Phi)_{\alpha_s}(W_s^+)\,ds
			\nonumber\\
			&\quad
			-
			\int_0^{\gamma(t)}
			e^{-B(s)}
			\left(
			\partial_{\mathbf n}\Phi_{\alpha_s}
			+\kappa(\alpha_s)\Phi_{\alpha_s}
			\right)(W_s^+)\,dL_s,
			\label{eq:annealed-ito-formula}
		\end{align}
		where \(\mathcal N_t\) has zero expectation; indeed,
		\(\gamma(t)\le t\) is a bounded stopping time and all coefficients involved are
		bounded on \(\overline D\times S\).
		
		For \(\Phi\in\mathcal D_0\),
		\[
		-
		\left(
		\partial_{\mathbf n}\Phi_i+\kappa_i\Phi_i
		\right)
		=
		\rho_i(\mathcal L\Phi)_i
		\qquad\text{on }\partial D.
		\]
		Since \(dL_s\) is supported on \(\{W_s^+\in\partial D\}\), taking expectations
		in \eqref{eq:annealed-ito-formula} and using
		\[
		dA(s)=ds+\rho(\alpha_s)\,dL_s
		\]
		gives
		\begin{equation}
			\label{eq:annealed-dynkin-formula}
			(\mathcal S_t\Phi)(x,i)-\Phi(x,i)
			=
			\operatorname{E}_{x,i}\!\left[
			\int_0^{\gamma(t)}
			e^{-B(s)}
			(\mathcal L\Phi)_{\alpha_s}(W_s^+)\,dA(s)
			\right].
		\end{equation}
		
		Since \(A\) is continuous and strictly increasing, with inverse \(\gamma\),
		the change of variables \(r=A(s)\) gives
		\[
		\mathcal S_t\Phi-\Phi
		=
		\int_0^t\mathcal S_r(\mathcal L\Phi)\,dr.
		\]
		As \(\mathcal L\Phi\in C(\overline D\times S)\), strong continuity implies
		\[
		\left\|
		\frac{\mathcal S_t\Phi-\Phi}{t}-\mathcal L\Phi
		\right\|_\infty
		\le
		\sup_{0\le r\le t}
		\|\mathcal S_r(\mathcal L\Phi)-\mathcal L\Phi\|_\infty
		\longrightarrow0.
		\]
		Thus
		\[
		\Phi\in D(\mathcal G),
		\qquad
		\mathcal G\Phi=\mathcal L\Phi.
		\]
		
		\medskip
		\noindent\emph{Step 5: the backward equation.}
		The semigroup now gives the abstract Cauchy problem directly:
		$u(t)=\mathcal S_t\Phi$ is its unique mild solution.
		
		If $u$ is continuously differentiable in time with values in $C(\overline D\times S)$ and $u(t)\in\mathcal D_0$, the generator formula gives
		\eqref{eq:coupled-backward-system}, while membership in $\mathcal D_0$ gives
		\eqref{eq:coupled-sticky-elastic-bc}. The initial condition follows from
		strong continuity. Finally, substituting
		\[
		\partial_tu_i
		=
		\frac12\Delta u_i+\sum_{j\in S}q_{ij}u_j
		\]
		into \eqref{eq:coupled-sticky-elastic-bc} gives
		\eqref{eq:dynamic-wentzell-bc}.

				\le

			\end{equation}
			If \(\kappa\equiv0\), then total mass is conserved.
			
			\item The map \(t\mapsto\mu_t^\alpha\) is narrowly continuous.
			
			\item For every \(f\in C^2(\overline D)\),
			\begin{equation}
				\label{eq:quenched-forward-dynkin}
				\begin{aligned}
					&\int_{\overline D}f(y)\,\mu_t^\alpha(dy)
					-
					\int_{\overline D}f(y)\,\mu_0(dy)
					\\
					&=
					\int_{\overline D}
					\operatorname{E}_x\!\left[
					\frac12
					\int_0^{\gamma^\alpha(t)}
					H^\alpha_s
					\Delta f(W_s^+)\,ds
					\right]\mu_0(dx)
					\\
					&\quad
					-
					\int_{\overline D}
					\operatorname{E}_x\!\left[
					\int_0^{\gamma^\alpha(t)}
					H^\alpha_s
					\left(
					\partial_{\mathbf n}f
					+\kappa(\alpha_s)f
					\right)(W_s^+)\,dL_s
					\right]\mu_0(dx).
				\end{aligned}
			\end{equation}

				\label{eq:weak-forward-equation}

				-
				\int_{\overline D}f\,d\mu_0
				=
				\frac12
				\int_0^t
				\int_{\overline D}\Delta f(y)\,
				\mu_s^\alpha(dy)\,ds.
			\end{equation}
			Equivalently,
			\begin{equation}
				\label{eq:weak-forward-derivative}
				\frac{d}{dt}
				\int_{\overline D}f\,d\mu_t^\alpha
				=
				\frac12
				\int_{\overline D}\Delta f\,d\mu_t^\alpha
			\end{equation}
			for every \(t\ge0\).
			
			\item More generally, let

			\varphi(t,\cdot)\in\mathcal D_\alpha,

					\right)
					\mu_t^\alpha(dx)\,dt.
				\end{aligned}
			\end{equation}
		\end{enumerate}

\emph{Step 1: bound for \(W^{+}\).}
		We first prove that, for \(\theta<1/2\) and \(q\ge2\), there exists a
		constant \(C_{D,T,\theta,q}>0\), independent of the starting position \(x\),
		such that
		\begin{equation}
			\label{eq:Holder_bound_Wp}
			\sup_{x\in\overline D}
			\operatorname{E}_x\left[[W^+]_{\theta,T}^q\right]
			\le C_{D,T,\theta,q}<\infty.
		\end{equation}
		
		The reflected heat kernel satisfies the Gaussian bound
		\cite[Theorem~3.1 and Remark~3.11]{bass1991_some_potential_theory}:
		\begin{equation}
			\label{eq:neumann-heat-kernel-upper-bound}
			p^{W^+}(u,y,z)
			\le
			C^{(1)}_{D,T} u^{-d/2}
			\exp\left(
			-\frac{|z-y|^2}{C^{(2)}_D u}
			\right),
			\qquad
			0<u\le T,
			\quad y,z\in\overline D.
		\end{equation}
		
		The Markov property and
		\eqref{eq:neumann-heat-kernel-upper-bound} give, for
		\(0\le s<t\le T\) and \(a\ge2\),
		\begin{align}
			\operatorname{E}_x\left[
			|W_t^+-W_s^+|^a
			\right]
			&= \operatorname{E}_x\left[ \operatorname{E}  \left[{ 
				|W_t^+-W_s^+|^a \mid \mathcal F_{s} }\right] 
			\right]
			\\
			&=
			\operatorname{E}_x\left[
			\int_D
			|z-W_s^+|^a
			p^{W^+}(t-s,W_s^+,z)\,dz
			\right]
			\nonumber\\
			&\le
			C^{(1)}_{D,T}
			\operatorname{E}_x\left[
			\int_{\mathbb R^d}
			|\xi-W_s^+|^a
			(t-s)^{-d/2}
			\exp\left(
			-\frac{|\xi-W_s^+|^2}{C^{(2)}_D(t-s)}
			\right)\,d\xi
			\right]
			\nonumber\\
			&=
			C^{(1)}_{D,T}
			\operatorname{E}_x\left[
			\int_{\mathbb R^d}
			|\zeta|^a
			(t-s)^{-d/2}
			\exp\left(
			-\frac{|\zeta|^2}{C^{(2)}_D(t-s)}
			\right)\,d\zeta
			\right]
			\nonumber\\
			&=
			C^{(1)}_{D,T}
			\int_{\mathbb R^d}
			|\zeta|^a
			(t-s)^{-d/2}
			\exp\left(
			-\frac{|\zeta|^2}{C^{(2)}_D(t-s)}
			\right)\,d\zeta
			\nonumber\\
			&=
			C_{a,D,T}|t-s|^{a/2},
			\label{eq:reflected-bm-increment-moment}
		\end{align}
		where in the last step we used the change of variables
		\(\zeta=\sqrt{t-s}\,\eta\) and absorbed the resulting Gaussian moment
		into the constant \(C_{a,D,T}\).
		
		Choose now \(a>q\) so large that
		\[
		\theta < \frac12 - \frac1a.
		\]
		Kolmogorov's theorem in moment form
		\cite[Theorem~A.10]{friz2010_book}, with constants uniform in the starting
		point, gives
		\[
		\operatorname{E}_x\left[[W^+]_{\theta,T}^q\right]
		\le C_{D,T,\theta,q}<\infty,
		\]
		which proves \eqref{eq:Holder_bound_Wp}.
		
		We turn to the regularity of the regulator \(L\).
		
		\emph{Step 2: smooth distance local time.}
		Let \(R_t=r(W_t^+)\). We first relate \(L\) to the right local time of \(R\).
		With our normalization,
		\begin{equation}
			\label{eq:sloctime_limit}
			L_t = \lim_{\varepsilon \to 0} \frac{1}{2\varepsilon}
			\int_{0}^{t}
			\mathbf{1}_{\operatorname{dist}(W^+_s,\, \partial D)< \varepsilon} \,\mathrm{d} s. 
		\end{equation}
		The reflected SDE, with outward normal, is
		\cite{bass2005_uniqueness_for_reflecting}
		\begin{equation}
			\mathrm{d} W^+_t = \mathrm{d} W_t - \mathbf n(W^+_t) \,\mathrm{d} L_t,
		\end{equation}
		where \(W\) is a \(d\)-dimensional Brownian motion and \(\mathbf n\) is the
		outward unit normal vector on \(\partial D\).
		It\^o's formula gives
		\begin{align}
			\mathrm{d} R_t
			&= \nabla r(W^+_t) \cdot \,\mathrm{d} W^+_t
			+ \frac{1}{2} \sum_{i,j=1}^{d}
			\partial_i \partial_j r(W^+_t)
			\,\mathrm{d} \langle (W^+)^i, (W^+)^j\rangle_t \\
			&= \nabla r(W^+_t) \cdot \,\mathrm{d} W_t
			- \nabla r(W^+_t) \cdot \mathbf n(W^+_t) \,\mathrm{d} L_t
			+ \frac{1}{2} \Delta r(W^+_t) \,\mathrm{d} t,
		\end{align}
		where we used that the finite-variation reflection term does not contribute
		to quadratic variation, and hence
		\[
		\mathrm{d}\langle (W^+)^i,(W^+)^j\rangle_t
		=
		\delta_{ij}\,\mathrm{d} t.
		\]
		On the boundary, \(r\) is the inward distance, hence
		\[
		\nabla r(x) = -\mathbf n(x),
		\qquad
		\nabla r(x)\cdot \mathbf n(x) = -1.
		\]
		Since \(dL_t\) is supported on
		\(\{W^+_t\in\partial D\}=\{R_t=0\}\), we obtain
		\begin{equation}
			R_t = R_0
			+ \int_{0}^{t} \nabla r(W^+_u) \cdot \,\mathrm{d} W_u
			+ L_t
			+ \frac{1}{2} \int_{0}^{t} \Delta r(W^+_u) \,\mathrm{d} u.
		\end{equation}
		Its quadratic variation is
		\begin{equation}
			\langle R \rangle_t
			=
			\sum_{i=1}^{d} \int_{0}^{t}
			 \left[{ \partial_{i} r(W^+_u) }\right] ^{2} \,\mathrm{d} u.
		\end{equation}
		
		Since \(R\ge0\), Tanaka's formula
		\cite[Theorem~VI.1.2]{revuz1999Continuous_martingales} gives
		\begin{equation}
			R_t - R_0
			=
			\int_{0}^{t} \mathbf{1}_{\{R_u>0\}} \,\mathrm{d} R_u
			+ \frac{1}{2} L^{0}_{t}(R),
		\end{equation}
		where \(L^{0}_{}(R)\) is the right local time of \(R\).
		Comparison with the preceding decomposition yields
		\begin{equation}
			\int_{0}^{t}
			\mathbf{1}_{\{R_u = 0\}} \nabla r(W^+_u) \cdot \,\mathrm{d} W_u
			+ L_t
			+ \frac{1}{2} \int_{0}^{t}
			\mathbf{1}_{\{R_u = 0\}} \Delta r(W^+_u) \,\mathrm{d} u
			=
			\frac{1}{2} L^{0}_{t}(R).
		\end{equation}
		
		Since \(\partial D\) has zero Lebesgue measure and \(W_u^+\) admits a
		transition density for every \(u>0\),
		\[
		\operatorname{E}_x\left[
		\int_0^T \mathbf{1}_{\{W_u^+\in\partial D\}}\,\,\mathrm{d} u
		\right]
		=
		\int_0^T
		\Pr_x(W_u^+\in\partial D)\,\,\mathrm{d} u
		=0.
		\]
		Hence
		\[
		\int_0^T \mathbf{1}_{\{R_u=0\}}\,\,\mathrm{d} u=0
		\qquad\text{a.s.}
		\]
		The boundary-supported martingale therefore has zero quadratic variation
		and hence vanishes
		\cite[Proposition~IV.1.13]{revuz1999Continuous_martingales}:
		\begin{equation}
			\left\langle
			\int_{0}^{\cdot}
			\mathbf{1}_{\{R_u = 0\}} \nabla r(W^+_u) \cdot \,\mathrm{d} W_u
			\right\rangle_t
			=
			\int_0^t \mathbf{1}_{\{R_u=0\}}\,
			|\nabla r(W^+_u)|^2\,\,\mathrm{d} u
			=0.
		\end{equation}
		Likewise,
		\[
		\int_0^t \mathbf{1}_{\{R_u=0\}}\Delta r(W^+_u)\,\,\mathrm{d} u=0.
		\]
		Thus, with our normalization,
		\begin{equation}
			\label{eq:loctime_relation}
			L_t=\frac12L^{0}_{t}(R).
		\end{equation}
		This factor \(1/2\) is precisely the convention fixed in
		Section~\ref{ssec:notation}; compare
		\cite[Section~VI.1]{revuz1999Continuous_martingales}.
		
		\medskip\noindent\emph{Step 3: bound for \(L\).}
		By Tanaka's formula applied to the nonnegative continuous semimartingale \(R\),
		and using the identity~\eqref{eq:loctime_relation}, for \(0\le s<t\le T\),
		\begin{align}
			L_t - L_s
			&=
			 \left({R_t - R_s}\right) 
			-
			\int_s^t \mathbf{1}_{\{R_u>0\}}\,\,\mathrm{d} R_u \\
			&=
			 \left({R_t - R_s}\right) 
			-
			 \left({M_t-M_s}\right) 
			-
			 \left({A_t-A_s}\right) ,
		\end{align}
		where
		\[
		M_t := \int_{0}^{t}
		\mathbf{1}_{\{R_u > 0\}} \nabla r(W^+_u) \cdot \,\mathrm{d} W_u,
		\qquad
		A_t := \frac{1}{2} \int_{0}^{t}
		\mathbf{1}_{\{R_u > 0\}} \Delta r(W^+_u) \,\mathrm{d} u.
		\]
		Here we used that \(dL_u\) is supported on \(\{R_u=0\}\), so the reflection
		term vanishes after multiplication by \(\mathbf{1}_{\{R_u>0\}}\).
		
		By convexity,
		\begin{equation}
			\operatorname{E}_x  \left[{|L_t - L_s|^{q}}\right] 
			\le
			3^{q-1}
			 \left({
				\operatorname{E}_x  \left[{|R_t - R_s|^{q}}\right] 
				+ \operatorname{E}_x  \left[{|M_t - M_s|^{q}}\right] 
				+ \operatorname{E}_x  \left[{|A_t - A_s|^{q}}\right] 
			}\right) .
		\end{equation}
		
		Since \(r\) is Lipschitz on \(\overline D\),
		\begin{equation}
			|R_t - R_s|
			=
			|r(W^+_t) - r(W^+_s)|
			\le
			\mathrm{Lip}(r)\,|W^{+}_t - W^{+}_s|.
		\end{equation}
		Therefore, by Step~1,
		\begin{equation}
			\operatorname{E}_x  \left[{|R_t - R_s|^{q}}\right] 
			\le
			\mathrm{Lip}(r)^q C_{q,D,T} |t-s|^{q/2}.
		\end{equation}
		
		For the martingale term, BDG gives
		(see, e.g., \cite[Section~IV.4]{revuz1999Continuous_martingales})
		\begin{align}
			\operatorname{E}_x  \left[{|M_t - M_s|^{q}}\right] 
			&\le C_{q}
			\operatorname{E}_x  \left[{
				| \left\langle {M} \right\rangle _t -  \left\langle {M} \right\rangle _s|^{q/2}}\right]  \\
			&= C_{q} \operatorname{E}_x  \left[{
				 \left|{
					\int_{s}^{t}
					\mathbf{1}_{\{R_u > 0\}}
					|\nabla r(W^+_u)|^{2}
					\,\mathrm{d} u}\right|^{q/2}}\right]  \\
			&\le C_q  \left\|{\nabla r}\right\|_{\infty}^{q} |t-s|^{q/2}.
		\end{align}
		
		For the finite-variation term,
		\begin{align}
			\operatorname{E}_x  \left[{|A_t - A_s|^{q}}\right] 
			&\le
			\frac{1}{2^q}
			 \left\|{\Delta r}\right\|_{\infty}^{q}
			|t-s|^{q} \\
			&\le
			\frac{1}{2^q}
			 \left\|{\Delta r}\right\|_{\infty}^{q}
			T^{q/2}
			|t-s|^{q/2}.
		\end{align}
		
		Combining the three bounds gives
		\begin{equation}
			\operatorname{E}_x  \left[{ |L_t - L_s|^{q} }\right] 
			\le
			C_{q,D,T}|t-s|^{q/2},
			\qquad 0\le s<t\le T.
		\end{equation}
		The same argument applies with \(q\) replaced by any \(a\ge2\). Hence
		\begin{equation}
			\label{eq:local-time-increment-moment}
			\sup_{x\in\overline D}
			\operatorname{E}_x \left[{|L_t-L_s|^a}\right] 
			\le
			C_{a,D,T}|t-s|^{a/2},
			\qquad 0\le s<t\le T.
		\end{equation}
		
		Now fix \(\theta<1/2\) and \(q\ge2\), and choose \(a>q\) large enough so that
		\[
		\theta < \frac12-\frac1a.
		\]
		Kolmogorov's theorem in moment form then gives
		\[
		\sup_{x\in\overline D}
		\operatorname{E}_x\left[[L]_{\theta,T}^q\right]
		\le C_{D,T,\theta,q}<\infty.
		\]
		
		It remains to treat \(1\le q<2\). By Jensen's inequality,
		\[
		\operatorname{E}_x[Z^q]
		=
		\operatorname{E}_x[(Z^2)^{q/2}]
		\le
		\operatorname{E}_x[Z^2]^{q/2}.
		\]
		Applying this with \(Z=[W^+]_{\theta,T}\) and
		\(Z=[L]_{\theta,T}\) extends the preceding \(q=2\) estimate to all
		\(1\le q<2\).
		This completes the proof.

		\operatorname{E}^\alpha\operatorname{E}_x^W
		\left[
		\sup_{0\le t\le T}R_\varepsilon(t)
		\right]\,\mu_0(dx)
		\\
		&\qquad=
		\int_{\overline D}
		\operatorname{E}_{x,\pi}
		\left[
		\sup_{0\le t\le T}R_\varepsilon(t)
		\right]\,\mu_0(dx)
		\\
		&\qquad\le
		C m_0\varepsilon^\beta,
	\end{aligned}
	\]
	where the last inequality follows from
	\eqref{eq:forward-proof-coupling-bound}.
	This proves \eqref{eq:forward-BL-uniform}.
	The fixed-time estimate \eqref{eq:forward-BL-fixed} follows from
	\[
	\sup_{0\le t\le T}
	\operatorname{E}^\alpha
	d_{\mathrm{BL}}
	\bigl(\mu_t^{\varepsilon,\alpha},\mu_t^0\bigr)
	\le
	\operatorname{E}^\alpha
	\left[
	\sup_{0\le t\le T}
	d_{\mathrm{BL}}
	\bigl(\mu_t^{\varepsilon,\alpha},\mu_t^0\bigr)
	\right].
	\]
	
	Finally, for every \(\eta>0\), Markov's inequality gives
	\[
	\operatorname{P}^\alpha
	\left(
	\sup_{0\le t\le T}
	d_{\mathrm{BL}}
	\bigl(\mu_t^{\varepsilon,\alpha},\mu_t^0\bigr)
	>\eta
	\right)
	\le
	\frac{C m_0}{\eta}\varepsilon^\beta
	\longrightarrow0.
	\]
	Both forward measures have mass at most \(m_0\).
	Since \(\overline D\) is compact, the bounded-Lipschitz distance
	metrizes narrow convergence on this set of measures.
	This proves the asserted uniform-in-time convergence in probability.

Fix \(T>0\), and set
	\[
	a_f
	:=
	\left\|cf+\frac12\Delta f\right\|_\infty.
	\]
	As in the proof of
	Theorem~\ref{thm:improved-smooth-observable}, the proportionality condition
	\(\kappa_i=c\rho_i\) yields
	\[
	K_\varepsilon(t)=c\bigl(t-\gamma_\varepsilon(t)\bigr),
	\qquad
	K_0(t)=c\bigl(t-\gamma_0(t)\bigr).
	\]
	For a fixed environment, apply It\^o's formula to
	\(F(s,x)=e^{cs}f(x)\) at the bounded stopping times
	\(\gamma_\varepsilon(t)\) and \(\gamma_0(t)\).
	The Neumann condition removes the boundary local-time term.
	Taking expectations and subtracting the two identities yields
	\[
	u_\alpha^\varepsilon(t,x;f)-u^0(t,x;f)
	=
	\operatorname{E}_x^W
	\left[
	\int_{\gamma_0(t)}^{\gamma_\varepsilon(t)}
	e^{-c(t-s)}
	\left(cf+\frac12\Delta f\right)(W_s^+)\,ds
	\right],
	\]
	where the integral is understood in the oriented sense.
	
	Both inverse clocks are at most \(t\), and \(c\ge0\).
	Thus \(e^{-c(t-s)}\le1\) on the interval between them, and
	\begin{equation}
		\label{eq:forward-proof-smooth-clock-bound}
		\left|
		u_\alpha^\varepsilon(t,x;f)-u^0(t,x;f)
		\right|
		\le
		a_f\operatorname{E}_x^W
		\left|
		\gamma_\varepsilon(t)-\gamma_0(t)
		\right|.
	\end{equation}
	Using duality and the nonnegativity of \(\mu_0\), we obtain
	\begin{equation}
		\label{eq:forward-proof-smooth-duality-bound}
		\left|
		\int_{\overline D}
		f\,d\bigl(\mu_t^{\varepsilon,\alpha}-\mu_t^0\bigr)
		\right|
		\le
		a_f\int_{\overline D}
		\operatorname{E}_x^W
		\left|
		\gamma_\varepsilon(t)-\gamma_0(t)
		\right|\,\mu_0(dx).
	\end{equation}
	
	The fixed-time clock estimate
	\eqref{eq:fixed-time-gamma-L2-rate} holds uniformly in the starting
	point, by
	Theorem~\ref{thm:abstract-averaging} and
	Lemma~\ref{lem:short-time-growth-local-time}.
	Consequently, Tonelli's theorem and Lyapunov's inequality give
	\[

		\operatorname{E}^\alpha
		\left|
		\int_{\overline D}
		f\,d\bigl(\mu_t^{\varepsilon,\alpha}-\mu_t^0\bigr)
		\right|
		\\

		a_f\int_{\overline D}
		\sup_{0\le t\le T}
		\operatorname{E}_{x,\pi}
		\left|
		\gamma_\varepsilon(t)-\gamma_0(t)
		\right|\,\mu_0(dx)

		\left|
		\gamma_\varepsilon(t)-\gamma_0(t)

		\right)^{1/2}\,\mu_0(dx)

	\end{aligned}

	\sup_{0\le t\le T}
	\left|
	\gamma_\varepsilon(t)-\gamma_0(t)
	\right|^2
	\right]
	\le
	C\varepsilon^{1/2}
	\bigl(1+|\log\varepsilon|\bigr)^2.

	\eqref{eq:forward-proof-smooth-duality-bound}, then using again Tonelli's
	theorem and Lyapunov's inequality, gives
	\[
	\begin{aligned}
		&\operatorname{E}^\alpha
		\left[
		\sup_{0\le t\le T}
		\left|
		\int_{\overline D}
		f\,d\bigl(\mu_t^{\varepsilon,\alpha}-\mu_t^0\bigr)
		\right|
		\right]
