\label{app:thm:upper}
Let $\Lambda$ be a finite rectangular box with open boundary conditions in
fixed dimension $\nu\ge1$, and let
\begin{align}
 H=\sum_{\substack{Z\subseteq\Lambda:\ 1\le|Z|\le k\\
                    \operatorname{diam}(Z)\le r_0}}h_Z,
 \qquad h_Z=h_Z^\dagger,
 \qquad \max_{i\in\Lambda}\sum_{Z\ni i}\|h_Z\|\le1.
 \label{app:eq:local-assumptions}
\end{align}
The on-site dimension is at most $d_0$, and $\nu,d_0,k,r_0$ are fixed.
Assume that $H$ has a unique normalized ground state $\Omega$, energy $E_0$,
and gap $\Delta>0$. Fix a planar cut $L|R$ with cross-section $\mathcal A$.
There are constants $C,C_*$, independent of $|\Lambda|,\mathcal A,
\epsilon,\Delta$, such that for every $0<\epsilon\le1/4$ there exist
$K=K^\dagger$ and a normalized state $\Omega_K$ satisfying
\begin{align}
 K|\Omega_K\rangle&=|\Omega_K\rangle,&
 \|\Omega-\Omega_K\|&\le\delta_K,&
 \|K(1-|\Omega_K\rangle\langle\Omega_K|)\|&\le\epsilon_K,
 \label{app:eq:agsp-definition}\\
 \delta_K+\epsilon_K&\le\epsilon,&
 \operatorname{SR}(K)&\le D_K,&
 \log D_K&\le
 C\left[\mathcal A+
 \left(\frac{\mathcal A L_\epsilon}{\sqrt{\Delta_*}}\right)^{2/3}\right]
 \log^2\!\left(\frac{C_*\mathcal A L_\epsilon}{\Delta_*}\right).
 \label{app:eq:upper-theorem}
\end{align}
In particular, $\|K-|\Omega\rangle\langle\Omega|\|\le\epsilon$.
Neither frustration freeness nor a local-gap assumption is required.

\label{app:thm:lower}
There are fixed constants $d_0\le6$, $k\le5$, $r_0\le2$, $\Delta_0>0$,
and $C>0$ with the following properties. For every $m\ge3$, every fixed
$\nu\ge2$, and every rectangular $(\nu-1)$-dimensional cross-section
$\Sigma$ of size $M\ge1$, there is a Hamiltonian satisfying
\eqref{app:eq:local-assumptions} on
\begin{align}
 \Lambda=\{-16m,\ldots,16m\}\times\Sigma,
 \qquad n=M(32m+1),\qquad \mathcal A=M.
 \label{app:eq:lower-geometry}
\end{align}
The cut lies between $x=0$ and $x=1$. The ground state
$\Omega_{M,m}^{(\nu)}$ is unique, the full-system gap is at least
$\Delta_0$, and a product root $o$ obeys
$|\langle o|\Omega_{M,m}^{(\nu)}\rangle|^2\ge1/2$.
There is a fixed projection $P_F=P_L\otimes I_R$, independent of any
approximating state, such that, with
$w_{M,m}:=\|P_F\Omega_{M,m}^{(\nu)}\|^2$,
\begin{align}
 w_{M,m}&\ge
 \exp\!\left[-C\left(M+\sqrt M\,m^{3/2}\right)\right],
 \label{app:eq:target-theorem}\\
 (\mathcal V_L\otimes\mathcal V_R)
 \frac{P_F|\Omega_{M,m}^{(\nu)}\rangle}{\sqrt{w_{M,m}}}
 &=|\Phi\rangle^{\otimes Mm}\otimes|\Xi\rangle,
 \qquad \|\Xi\|=1,
 \label{app:eq:bell-theorem}
\end{align}
where $\mathcal V_L,\mathcal V_R$ are isometries and
$\Phi=(|00\rangle+|11\rangle)/\sqrt2$.
For every integer $1\le D\le2^{Mm}$,
\begin{align}
 \inf_{\operatorname{SR}(\phi)\le D}
 \|\Omega_{M,m}^{(\nu)}-\phi\|^2
 \ge w_{M,m}\left(1-\frac{D}{2^{Mm}}\right).
 \label{app:eq:finite-tail-theorem}
\end{align}
The approximating vector $\phi$ need not be normalized.
In one dimension there is a nearest-neighbor, two-body construction on
$n=32m$ sites, with $\mathcal A=1$, the same constant-gap and root-overlap
properties, and a fixed projection extracting $m$ Bell pairs with
$w_m\ge e^{-Cm^{3/2}}$.

\label{app:lem:moment}
For $s_0=1/(4k)$,
\begin{align}
 \|e^{s_0D}\Omega\|\le e^{3b_\partial s_0},
 \qquad
 \|\mathbf1_{(T,\infty)}(D)\Omega\|
 \le e^{-s_0T+3b_\partial s_0}\quad(T\ge0).
 \label{app:eq:moment}
\end{align}
<!-- end applemma-1 -->
<!-- appproof-1: body_md -->
Up to a scalar, $D$ is a sub-sum of the original microscopic local terms,
so its interaction strength per site is still at most one. Write
$\operatorname{ad}_A(B):=[A,B]=AB-BA$ and
$\operatorname{ad}_A^0(B):=B$. If $O_Z$ is supported on at most $k$ sites, then
\begin{align}
 \|\operatorname{ad}_D^r(O_Z)\|
 \le(2k)^r r!\,\|O_Z\|,\qquad r\ge0.
 \label{app:eq:microscopic-commutator}
\end{align}
Indeed, after $j$ nonzero commutators the support has at most
$k+j(k-1)$ sites. The sum of the norms of terms that can contribute to the
next commutator is at most $k+j(k-1)\le k(j+1)$. Summing over ordered
sequences of local terms and using $\|[X,Y]\|\le2\|X\|\|Y\|$ yields
$\prod_{j=0}^{r-1}2k(j+1)=(2k)^rr!$.
Applying this estimate term by term to $B$ gives, for $0\le s\le s_0$,
\begin{align}
 \|e^{sD}Be^{-sD}\|
 \le\sum_{r\ge0}\frac{s^r}{r!}
                  \|\operatorname{ad}_D^r(B)\|
 \le\frac{b_\partial}{1-2ks}\le2b_\partial.
 \label{app:eq:conjugated-boundary}
\end{align}
Let $F(s)=e^{sD}e^{-s(D+B)}$. Differentiation gives
$F'(s)=-(e^{sD}Be^{-sD})F(s)$ and $F(0)=I$, so the integral equation and
Gronwall's inequality imply $\|F(s)\|\le e^{2b_\partial s}$.
Since $e^{sD}\Omega=e^{se_0}F(s)\Omega$, the first bound in
\eqref{app:eq:moment} follows. The spectral theorem gives the second.
The argument uses microscopic sites rather than treating an entire layer
as one site; in particular, $s_0$ has no dependence on $\mathcal A$,
$\ell$, or $n$.

\label{app:lem:variational}
Let $G\ge0$ have a unique normalized zero-energy ground state $\Omega$ and
gap $\Delta>0$. For an orthogonal projection $P$ with $P\Omega\ne0$, set
\begin{align}
 G_P:=PGP\big|_{\operatorname{Ran}P},
 \qquad
 e_P:=\frac{\langle\Omega|PGP|\Omega\rangle}{\|P\Omega\|^2}.
 \label{app:eq:variational-energy}
\end{align}
If $e_P<\Delta$, the ground state $\Omega_P$ of $G_P$ is unique and,
when embedded in the original space with its phase aligned with $\Omega$,
\begin{align}
 0\le\lambda_0(G_P)\le e_P,
 \qquad \Delta_P\ge\Delta-e_P,
 \qquad \|\Omega_P-\Omega\|\le\sqrt{2e_P/\Delta}.
 \label{app:eq:variational-conclusions}
\end{align}
If $\operatorname{Ran}P$ is one-dimensional, the excited-state assertion
is vacuous.
<!-- end applemma-2 -->
<!-- appproof-2: body_md -->
Use $P\Omega/\|P\Omega\|$ as a trial state. For a restriction to a
subspace, the min--max principle gives
\begin{align}
 \lambda_1(G_P)=
 \min_{\substack{U\subseteq\operatorname{Ran}P\\\dim U=2}}
 \max_{\substack{u\in U\\\|u\|=1}}\langle u|G|u\rangle
 \ge\lambda_1(G)=\Delta.
 \label{app:eq:minmax}
\end{align}
This proves uniqueness and the gap estimate. Since
$G\ge\Delta(I-|\Omega\rangle\langle\Omega|)$,
$1-|\langle\Omega|\Omega_P\rangle|^2\le e_P/\Delta$.
For phase-aligned unit vectors, the squared distance is at most twice
this quantity.

\label{app:lem:cutcount}
Suppose $F$ is norm-absolutely convergent in a neighborhood of the closed
unit polydisc. For an integer $Q\ge0$, set
\begin{align}
 \delta_Q(F):=\sum_{|q|>Q}\|F_q\|,\qquad
 p_*:=\lfloor2Q/\ell\rfloor,\qquad
 N_{\mathrm{ph}}:=\ell(2Q+1)^{\lceil\log_2\ell\rceil}.
 \label{app:eq:cutcount-data}
\end{align}
There are indexed families $\mathcal Z_s$ of unit phases and coefficients
$C_{s,z}$ such that
\begin{align}
 F(\mathbf1)
 &=\sum_{s=1}^\ell\sum_{z\in\mathcal Z_s} C_{s,z}
       \sum_{p=0}^{p_*}[x^p]
       F(z_1,\ldots,z_{s-1},xz_s,z_{s+1},\ldots,z_\ell)+R_Q,
 \label{app:eq:cutcount}\\
 |z_t|&=1,\qquad |C_{s,z}|\le1,\qquad
 \sum_s|\mathcal Z_s|\le N_{\mathrm{ph}},\qquad
 \|R_Q\|\le2N_{\mathrm{ph}}\delta_Q(F).
 \label{app:eq:cutcount-error}
\end{align}
<!-- end applemma-3 -->
<!-- appproof-3: body_md -->
First consider a polynomial of total degree at most $Q$.
Organize the $\ell$ cuts into a binary tree, splitting each set into
children whose cardinalities differ by at most one. For a monomial
$\lambda^q$, compare the \emph{total} costs of the two children $I,J$:
\begin{align}
 d_{I,J}(q):=\sum_{s\in I}q_s-\sum_{s\in J}q_s.
 \label{app:eq:total-cost}
\end{align}
Choose $I$ if $d_{I,J}\le0$ and $J$ otherwise. This tie convention gives
every monomial a unique leaf. The selected cost is at most half the parent
cost at each step. A leaf depth $h$ lies between
$\lfloor\log_2\ell\rfloor$ and $\lceil\log_2\ell\rceil$, so its selected
cut satisfies $q_s\le Q2^{-h}\le2Q/\ell$, and thus $q_s\le p_*$.

Let $N_F=2Q+1$ and $\omega=e^{2\pi i/N_F}$.
The selection functions $\chi_L(d)=\mathbf1_{d\le0}$ and
$\chi_R(d)=\mathbf1_{d>0}$ have the exact discrete Fourier representations
on $-Q\le d\le Q$
\begin{align}
 \chi_b(d)=\sum_{r=0}^{N_F-1}\widehat\chi_b(r)\omega^{rd},
 \qquad
 \widehat\chi_b(r)=\frac1{N_F}\sum_{a=-Q}^Q\chi_b(a)\omega^{-ra},
 \qquad |\widehat\chi_b(r)|\le1.
 \label{app:eq:selection-fourier}
\end{align}

$\mathsf L=G^{-1}JG$. With
$q_{xy}=-J_{xy}\sqrt{\pi(y)/\pi(x)}\ge0$ for $x\ne y$, we have
\begin{align}
 (\mathsf Lf)(x)=\sum_{y\ne x}q_{xy}(f(x)-f(y)),\qquad
 c_{xy}=\pi(x)q_{xy}=c_{yx}.
 \label{app:eq:Doob-transform}
\end{align}
Thus $\mathsf L$ is the nonnegative-sign Markov Laplacian, whose
transition semigroup is $e^{-t\mathsf L}$, and
$\langle f,\mathsf Lg\rangle_\pi=\mathcal E(f,g)$.

\begin{align}
 h:=s(\mathsf L+sI)^{-1}\frac{\mathbf1_a}{p}.
 \label{app:eq:resolvent-density}
\end{align}
Connectivity and the positive resolvent kernel give $h>0$.
Stationarity gives $\mathbb E_\pi h=1$. At a positive maximum away from
$a$, $(\mathsf L+sI)h>0$ would contradict the resolvent equation, so
$h(a)=\max h\ge1$.

$(\mathsf L+sI)h=s\mathbf1_a/p$ with $1/h$. Since
$\langle1/h,h\rangle_\pi=1$, this gives
\begin{align}
 -\mathcal E(1/h,h)=s\left(1-\frac1{h(a)}\right)\le s.
 \label{app:eq:log-energy-start}
\end{align}

$(\log x-\log y)^2\le(x-y)^2/(xy)$. Hence
\begin{align}
 \mathcal E(\log h,\log h)
 \le-\mathcal E(1/h,h)\le s,
 \qquad \operatorname{Var}_\pi(\log h)\le s\tau.
 \label{app:eq:log-density-variance}
\end{align}
Let $\ell_h=\mathbb E_\pi\log h$. The source contribution to the variance
and $\log h(a)\ge0$ imply $\ell_h\ge-\sqrt{s\tau/p}$.

$|\mathbb E_{\pi(\cdot|F)}\log h-\ell_h|\le\sqrt{s\tau/q}$.

 \ge-(p^{-1/2}+q^{-1/2})\sqrt{s\tau}.

\end{align}

 s\langle v_F|(J+sI)^{-1}|a\rangle
       =\sqrt{pq}\,\mathbb E_{\pi(\cdot|F)}h.
 \label{app:eq:resolvent-density-identity}
\end{align}
Jensen's inequality completes the proof.

We apply the entangling-rate and exact quasi-adiabatic-continuation

lower bound on each nonzero coupling is needed for the resulting

quasi-local decomposition of exact spectral flow. Terms supported
entirely on one side of the cut do not change its entanglement.
A radius-$cr$ ball contains $O(r^\nu)$ sites, and at most
$O(\mathcal A r)$ centers have such a ball crossing a planar cut.
The small-incremental-entangling bound (Theorem~3 of
Ref.~\cite{Marien2016}) controls each crossing term by
its norm times the logarithm of its supported Hilbert-space dimension,
independently of the remaining sites treated as ancillas. Hence the
integrated entangling-rate bound gives
\begin{align}
 |S_1(\Omega(s))-S_1(\Omega(0))|
 \le C\mathcal A\log6\int_0^s
       \sum_{r\ge1}r^{\nu+1}C_N(1+r)^{-N}\,du
 \le C_\nu\mathcal A
 \label{app:eq:path-rate-integral}
\end{align}
for any fixed $N>\nu+2$. The integrated form also covers points at
which a reduced density matrix has zero eigenvalues. Finally,
$S_1(\Omega(0))=0$ because $o$ is a product state.
This explicitly verifies the finite-size-uniform area-law stability
argument for the present rectangular geometries.

<!-- appthm-3: body_md -->
\label{app:thm:thermal-transfer}
There are constants $C,\Delta_0>0$, independent of $m,\beta,\alpha$,
such that, for every $m\ge3$ and $0<\alpha<1$,
\begin{align}
 \beta\Delta_0\ge32m\log6+Cm^{3/2}+\log16
 \quad\Longrightarrow\quad
 E_{p,\alpha}(\rho_{\beta,m})
 \ge m\log2-\log2-\kappa_\alpha(Cm^{3/2}+\log8).
 \label{app:eq:thermal-1d-finite}
\end{align}
The row has $n=32m$ sites, on-site dimension at most six, and physical
gap at least $\Delta_0$. Its local data and gap bound are uniform in
$m$. The lower bound applies to every purification in
\eqref{app:eq:thermal-Ep-def}.

<!-- applemma-10: body_md -->
\label{app:lem:near-pure}
Let $\langle\Omega|\rho|\Omega\rangle=1-\delta>0$ and
$d_\delta:=\sqrt{2-2\sqrt{1-\delta}}\le\sqrt{2\delta}$.
For every integer $R\ge1$ and $0<\alpha<1$,
\begin{align}
 E_{p,\alpha}(\rho)\ge\log R+
 \kappa_\alpha\log\left(\sqrt{T_\Omega(R)}-d_\delta\right)_+^2.
 \label{app:eq:near-pure-transfer}
\end{align}
A zero argument of the logarithm gives no useful bound for that $R$.
<!-- end applemma-10 -->
<!-- appproof-14: body_md -->
For any purification $\Psi\in LL'RR'$, define the normalized ancillary
vector $\eta$ by
$(\langle\Omega|\otimes I)|\Psi\rangle=\sqrt{1-\delta}\,|\eta\rangle$.
Then $\|\Psi-\Omega\otimes\eta\|=d_\delta$, and $\eta$ may itself be
entangled across $L'|R'$. If $p_i,q_j$ are the Schmidt probabilities of
$\Omega,\eta$, those of $\Omega\otimes\eta$ are $p_iq_j$. The sum of
any $R$ of these products is at most the sum of the $R$ largest $p_i$:
group the selected pairs by $i$ and use $\sum_jq_j=1$.
Hence $T_{\Omega\otimes\eta}(R)\ge T_\Omega(R)$. Distance to the set
of vectors of Schmidt rank at most $R$ is $1$-Lipschitz, giving
\begin{align}
 T_\Psi(R)\ge
       \left(\sqrt{T_\Omega(R)}-d_\delta\right)_+^2.
 \label{app:eq:purification-tail}
\end{align}
For Schmidt probabilities $s_1\ge s_2\ge\cdots$, put
$M_\alpha=\sum_js_j^\alpha$. Then
\begin{align}
 s_{R+1}\le(M_\alpha/R)^{1/\alpha},\qquad
 T(R)\le s_{R+1}^{1-\alpha}\sum_{j>R}s_j^\alpha
       \le M_\alpha^{1/\alpha}R^{-(1-\alpha)/\alpha}.
 \label{app:eq:tail-to-entropy}
\end{align}
Thus $S_\alpha\ge\log R+\kappa_\alpha\log T(R)$.
Apply this to \eqref{app:eq:purification-tail} and take the infimum
over all purifications and ancillary bipartitions.

<!-- appproof-15: body_md -->
The full excited-state weight satisfies
\begin{align}
 \delta_\beta:=1-\langle\Omega_m|\rho_{\beta,m}|\Omega_m\rangle
 =\frac{\sum_{j\ge1}e^{-\beta(E_j-E_0)}}
        {1+\sum_{j\ge1}e^{-\beta(E_j-E_0)}}
 \le(6^{32m}-1)e^{-\beta\Delta_0}.
 \label{app:eq:thermal-excited-weight}
\end{align}
This counts all physical excitations, including legal, non-history and
illegal sectors. Set $F=Cm^{3/2}$. The cooling condition in
\eqref{app:eq:thermal-1d-finite} implies $\delta_\beta\le e^{-F}/16$.
The one-dimensional Bell-sector construction gives
$T_{\Omega_m}(2^{m-1})\ge\tfrac12e^{-F}$. Consequently,
\begin{align}
 \left(\sqrt{T_{\Omega_m}(2^{m-1})}-d_{\delta_\beta}\right)_+^2
 \ge\left(\sqrt{e^{-F}/2}-\sqrt{e^{-F}/8}\right)^2
 =e^{-F}/8.
 \label{app:eq:thermal-tail-constant}
\end{align}
Lemma~\ref{app:lem:near-pure} with $R=2^{m-1}$ proves the claim.

<!-- appcor-4: body_md -->
\label{app:cor:thermal-1d}
There exist constants $c>0$, $\beta_0>0$ and $0<\alpha_0<1$ such that,
for every $\beta\ge\beta_0$ and $0<\alpha\le\alpha_0$,
\begin{align}
 \sup_{m\ge3}E_{p,\alpha}(\rho_{\beta,m})
 \ge c\min\!\left\{\beta^{2/3},\left(\frac{1-\alpha}{\alpha}\right)^2\right\}.
 \label{app:eq:thermal-1d-joint}
\end{align}
For a sufficiently small fixed $a>0$, taking
$\alpha=a\beta^{-1/3}$ and a suitable $m=\Theta(\beta^{2/3})$ gives
\begin{align}
 E_{p,\alpha}(\rho_{\beta,m})=\Theta(\beta^{2/3}).
 \label{app:eq:thermal-1d-diagonal}
\end{align}
The system size is selected for the prescribed $(\beta,\alpha)$.

Write $\kappa=\kappa_\alpha$ and
$R_0=\min\{\beta^{2/3},\kappa^{-2}\}$. Choose $0<\zeta<1$ such that

 32\zeta\log6+C\zeta^{3/2}\le\Delta_0/2,

 \label{app:eq:thermal-1d-zeta}

Set $m=\lfloor\zeta R_0\rfloor$. Increase $\beta_0\ge1$ and decrease

<!-- appprop-2: body_md -->
\label{app:prop:gibbs-compression}
There are constants $B,c>0$ and $L_0$ such that, for every $L\ge L_0$,
with $\epsilon=e^{-L}$ and $\beta=BL$, one can choose a row with
$m=\Theta(L^{2/3})$ for which every operator $X$ obeying
\begin{align}
 \|X-\rho_{\beta,m}\|_1\le\epsilon
 \quad\text{has}\quad
 \log\operatorname{SR}(X)\ge cL^{2/3}
 \label{app:eq:gibbs-rank}
\end{align}
across the central cut. The operator need not be positive, normalized,
Hermitian, or exactly ground-state preserving.
<!-- end appprop-2 -->
<!-- appproof-17: body_md -->
Choose a fixed small $a>0$ with $Ca^{3/2}\le1$ and set
$m=\lfloor aL^{2/3}\rfloor\ge3$, so $F=Cm^{3/2}\le L$.
A fixed sufficiently large $B$ gives
$BL\Delta_0\ge L+32m\log6+\log2$ for every $L\ge L_0$.
By \eqref{app:eq:thermal-excited-weight} and commutation with
$P_0=|\Omega_m\rangle\langle\Omega_m|$,
\begin{align}
 \|\rho_{\beta,m}-P_0\|_1=2\delta_\beta\le e^{-L},
 \qquad \|X-P_0\|\le2e^{-L}.
 \label{app:eq:gibbs-ground-distance}
\end{align}
Choose the phase so $a_o=\langle\Omega_m|o\rangle\ge1/\sqrt2$.
Since $o$ is a product state, $v=Xo/a_o$ has
$\operatorname{SR}(v)\le\operatorname{SR}(X)$ and
$\|v-\Omega_m\|^2\le8e^{-2L}$.
If $\operatorname{SR}(X)\le2^{m-1}$, the established Schmidt-tail
bound gives instead
\begin{align}
 \|v-\Omega_m\|^2\ge\tfrac12e^{-Cm^{3/2}}
                      \ge\tfrac12e^{-L},
 \label{app:eq:gibbs-contradiction}
\end{align}
contradicting $L>\log16$. Thus $\operatorname{SR}(X)>2^{m-1}$.
