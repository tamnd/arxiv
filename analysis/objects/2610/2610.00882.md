<!-- theorem-1: body_md -->
\label{thm:3.1}
   The distribution of the error $x^{\dagger} - x^K$ is given by

\begin{equation}\label{errordist}
    x^{\dagger} - x^K \sim \mathcal{N}\left( {G^K x^{\dagger}},  \Gamma_K \right),
\end{equation}
where
\begin{equation}\label{eq:Cov_k}
    \Gamma_K =  \sigma^2\omega^2 \left (\sum_{k=0}^{K-1} G^k\right) A^T A \left (\sum_{j=0}^{K-1} G^j\right).
\end{equation}
<!-- end theorem-1 -->
<!-- remark-1: body_md -->
In particular, this shows that the error follows a normal distribution with a mean given by the bias term
\begin{equation} \label{bias1}
    \mathbb{E}[x^{\dagger} - x^K] = G^K x^\dagger=\left(I - \sum_{k=0}^{K-1} G^k \omega A^T A \right) x^{\dagger},
\end{equation}
and a covariance given by
\begin{equation} \label{cov1}
    {\rm Cov}(x^{\dagger} - x^K) = \Gamma_K.
\end{equation}
<!-- end remark-1 -->
<!-- proof-1: body_md -->
We know that the Landweber iterates follow the form
\begin{equation}
    x^{(k+1)} = G x^{(k)} + f,
\end{equation}
where 
\begin{equation}
    G = I - \omega A^T A, \quad f = \omega A^T y.
\end{equation}
Recall that the data is given by, 
$ y = A x^{\dagger} + \epsilon, \,\text{where } \epsilon \sim \mathcal{N}(0, \sigma^2 I).
$
Substituting this into the formula for $f$ gives
\begin{equation}
    f = \omega A^T(A x^{\dagger} + \epsilon) = \omega A^T A x^{\dagger} + \omega A^T \epsilon.
\end{equation}
Assuming the iterations to start with $x_0=0$, after $K$ iterations we have
\begin{align}
    x^K =  \sum_{k=0}^{K-1} G^k f= \left(\sum_{k=0}^{K-1} G^k \omega A^T A \right) x^{\dagger} + \sum_{k=0}^{K-1} G^k \omega A^T \epsilon .
\end{align}
The first term, 
$\left(\sum_{k=0}^{K-1} G^k \omega A^T A \right) x^{\dagger}=(I-G^K)x^\dagger,
$
is deterministic, while the second term, 
$\sum_{k=0}^{K-1} G^k \omega A^T \epsilon,
$
is a linear transformation of an independent Gaussian variable $\epsilon \sim \mathcal{N}(0,\sigma^2 I)$. Since Gaussian distributions remain Gaussian under linear transformations and sums, we get that
\begin{equation}
    \sum_{k=0}^{K-1} G^k \omega A^T \epsilon \sim \mathcal{N}\!\left(0, \; \sigma^2 \omega^2 \left (\sum_{k=0}^{K-1} G^k\right) A^T A \left (\sum_{j=0}^{K-1} G^j\right)\right).
\end{equation}
Thus, we obtain
\begin{equation}
    x^K \sim \mathcal{N}\left ((I-G^K) x^{\dagger}, \Gamma_K\right),
\end{equation}
where $\Gamma_K$ is given in \eqref{eq:Cov_k}. Since $x^\dagger$ is deterministic, we therefore obtain 

                                                         
            
               
                                                                                                                                                  
                                                                                                                                 
             
       
\begin{equation}
    x^{\dagger} - x^K \sim \mathcal{N}\left(G^K x^{\dagger}, \Gamma_K\right).
\end{equation}

<!-- lemma-1: body_md -->
\label{thm:3.2}
Let us assume the source condition such that for some $\nu>0$,  $x^\dagger=(A^T A)^\nu w,\,
    \|w\|_2\leq M.$ Let us denote an eigendecomposition of, $A^TA$ as 
    \[ A^T A =V\operatorname{diag}(s_1^2,\ldots,s_n^2)V^T,\] where \(s_1,\ldots,s_n\) are the singular values of $A$. Let $G=I-\omega A^T A$ as above. Then the regularization bias after $K$ iterations satisfies
    \[\|G^Kx^\dagger\|_2 \leq  M\max_{1\leq i\leq n}
    \left|1-\omega s_i^2\right|^K(s_i^2)^\nu:=b_K(\nu,A).\] Moreover, if $\omega=\frac{1}{2s_{\max}^2},
    \, s_{\max}:=\max_i s_i,$ then \begin{align} \label{bkest}
    b_K(\nu,A) \leq
    2^\nu s_{\max}^{2\nu}M
    \frac{\nu^\nu K^K}{(\nu+K)^{\nu+K}}:=b_K^{ub}(\nu,A).
\end{align}
where $b_K^{ub}(\nu,A)$ denotes a global upper bound on the bias.
<!-- end lemma-1 -->
<!-- remark-2: body_md -->
The estimate \eqref{bkest} also shows that $b_K(\nu,A) \leq C_{\nu}M K^{-\nu}$ where $C_{\nu}$ is a constant that depends on $\nu$ and $\|A\|$. See also \cite[Theorem 3.1]{hanke_95} for a similar result.
<!-- end remark-2 -->
<!-- proof-2: body_md -->
Note that, \[
G^K(A^T A)^\nu
=V\operatorname{diag}_n\left((1-\omega s_i^2)^K(s_i^2)^\nu\right) V^T.\] Thus it follows that, \begin{align*}
    \|G^Kx^\dagger\|_2
    &=
    \|G^K(A^T A)^\nu w\|_2 \leq
    \|G^K(A^T A)^\nu\|_2\,\|w\|_2 \\
    &\leq
    M\max_{1\leq i\leq n}
    \left|1-\omega s_i^2\right|^K(s_i^2)^\nu,
\end{align*} Now if we choose, $\omega=\frac{1}{2s_{\max}^2}$ and let $ t_i=\omega s_i^2$ then $0\leq t_i\leq 1/2$. Thus $s_i^{2\nu}= \left(\frac{t_i}{\omega}\right)^\nu =2^\nu s_{\max}^{2\nu}t_i^\nu.$ From this it follows that, 
\[ b_K(\nu,A)
    \leq
    2^\nu s_{\max}^{2\nu}M
    \max_{0\leq t\leq 1/2}
    (1-t)^Kt^\nu\leq 2^\nu s_{max}^{2\nu} M \frac{\nu^\nu K^K}{(\nu+K)^{\nu+K}}.\]

\label{thm:3.3}
Suppose that assumptions of Lemma \ref{thm:3.2} hold. For $0<\alpha<1$, define \begin{align}
C_K(y)
:=
\left\{
x:\ (x-x^K_\sigma)^T(\Gamma_K)^{-1}(x-x^K_\sigma)
\le
\Big(\sqrt{\chi^2_{n,1-\alpha}}+\delta_K(\nu,A)\Big)^2
\right\}. 
\end{align}
 Then, \[
\mathbb{P}_{x^\dagger}\left(x^\dagger\in  C_K(Y)\right)\ \geq\ 1-\alpha.
\]
Hence, $C_K(Y)$ is a $(1-\alpha)$-level confidence region for
$x^\dagger$.
<!-- end theorem-2 -->
<!-- proof-3: body_md -->
Recall that \eqref{errordist} implies that 
 \[x^{\dagger}-x_{\sigma}^K=G^Kx^\dagger+Z_K\] where $Z_K\sim N(0,\Gamma_K)$. Thus, $G^Kx^\dagger$ is the deterministic regularization bias,
whereas $Z_K$ is random reconstruction error. In order to construct confidence regions we look at the distribution of the random variable given by $(x^\dagger-x_{\sigma}^K)^T\Gamma_K^{-1}(x^\dagger-x_{\sigma}^K):=\|(x^\dagger-x_{\sigma}^K)\|^2_{\Gamma^{-1}_K}$. 
By the triangle inequality,  \[
\|x^\dagger-x^k_{\sigma}\|_{{\Gamma_K}^{-1}}
\leq
\|Z_K\|_{{\Gamma_K}^{-1}} + \|G^K x^\dagger\|_{{\Gamma_K}^{-1}}.
\]  Since $Z_K\sim N(0,\Gamma_K)$, it follows that
$\Gamma_K^{-1/2}Z_K\sim N(0,I_n).
$
Thus
$\|Z_K\|_{(\Gamma_K)^{-1}}^2
\sim
\chi_n^2.
$ Therefore, by the definition of $\chi^2_{n,1-\alpha}$,
\[\mathbb{P}\left(
\|Z_K\|_{\Gamma_K^{-1}}\leq
\sqrt{\chi^2_{n,1-\alpha}}
\right)
=1-\alpha.
\]
Also, from Lemma \ref{thm:3.2} and from the definition of $\delta_K(\nu, A)$ we have, 
$\|G^Kx^\dagger\|_{(\Gamma_K)^{-1}}
\leq
\delta_K(\nu,A).
$ Thus, on the event
$\left\{
\|Z_K\|_{\Gamma_K^{-1}}\leq
\sqrt{\chi^2_{n,1-\alpha}}\right\},
$
we have
\[
\|x^\dagger-x_\sigma^K\|_{\Gamma_K^{-1}}
\leq\sqrt{\chi^2_{n,1-\alpha}}+\delta_K(\nu,A).
\]
Equivalently,
\[
(x^\dagger-x_\sigma^K)^T
\Gamma_K^{-1}
(x^\dagger-x_\sigma^K)
\leq\Big(
\sqrt{\chi^2_{n,1-\alpha}}
+\delta_K(\nu,A)
\Big)^2,
\]
which implies that $x^\dagger\in C_K(Y)$. Hence,
\[\mathbb{P}_{x^\dagger}\left(x^\dagger\in C_K(Y)
\right)\geq \mathbb{P}\left(
\|Z_K\|_{\Gamma_K^{-1}}\leq
\sqrt{\chi^2_{n,1-\alpha}}
\right)
=1-\alpha.
\]

 \begin{align}

\leq
M

|1-\omega s_i^2|^K
s_i^{2\nu}, \,\text{ and }\,  \|G^K(I-P_r)x^\dagger\|_2
\leq
\rho^K M s_{r+1}^{2\nu} .
 \end{align}

\leq

$(I-P_r)x^\dagger

$(1-P_r)x^\dagger = \sum_{i=r+1}^n
s_i^{2\nu} \langle w,v_i\rangle v_i,
$ so that $
\|(I-P_r)x^\dagger\|_2
\leq
s_{r+1}^{2\nu}
\left(
\sum_{i=r+1}^n
|\langle w,v_i\rangle|^2
\right)^{1/2}.
$ Hence,
$$
|G^K(I-P_r)x^\dagger|2
\leq
\rho^K s_{r+1}^{2\nu}
\left(
\sum_{i=r+1}^n
|\langle w,v_i\rangle|^2
\right)^{1/2}.
$$

\label{lem:1}
 Let $B=A^TA,\ 
G=I-\omega B$ and let $\widetilde B_r:=\widetilde{V_r}\widetilde{\Lambda}_r\widetilde{V}_r^T$ be the randomized rank $r-$ SVD approx to $B$ obtained by using \cite[Algorithm 5.3]{halko2011} with $r-$ as the traget rank and $p$ as an oversampling parameter. We also define $\widetilde G_r:=I-\omega\widetilde B_r$. Let the step size $\omega$ be chosen such that, $\|G\|\leq\rho<1$ and $\|\widetilde G_r\|\le1.$ For notational convenience we denote $S(G):=\sum_{k=0}^{K-1}G^k 
$ and similarly for $S(\widetilde{G}_r)$. We note that the matrices, $G,\, \widetilde{G}_r$ commute respectively with $B,$ and $\widetilde{B}_r$. Then we can write the covariance matrix from Theorem \ref{thm:3.1} and its approximation resulting from  truncated randomized SVD approximation to $B$ respectively as, \begin{align}
    \Gamma_K&=
\sigma^2\omega^2 S(G)\,B\, S(G)=\sigma^2\omega^2B(S(G))^2,\\
\widetilde\Gamma_{K,r}
&=\sigma^2\omega^2
S(\widetilde G_r)\,\widetilde B_r\,S(\widetilde G_r)=\sigma^2\omega^2\widetilde{B}_r(S(\widetilde G_r))^2.
\end{align}
Then $\mathbb{P}_{\mathrm{rSVD}}(\lVert\Gamma_K-\widetilde{\Gamma}_{K,r}\rVert\leq \Delta_{K,r,p})\geq(1-3e^{-p})$ for any $p\geq 4.$ Here, $\Delta_{K,r,p}$ is defined below.

<!-- proof-6: body_md -->
We estimate the error in approximating the covariance matrix by the randomized algorithm given in \cite[Algorithm 5.3]{halko2011}. To that end, consider first
\begin{align}
\Gamma_K&=\sigma^2\omega^2B(S(G))^2=\sigma^2 \omega(I-G)(S(G))^2\nonumber\\
    &=\sigma^2\omega \left(\sum_{k=0}^{K-1}G^k -\sum_{k=0}^{K-1}G^{k+1}\right)\sum_{k=0}^{K-1}G^k \nonumber\\
    &=\sigma^2\omega(I-G^K)\sum_{k=0}^{K-1}G^k =\sigma^2\omega \left(\sum_{k=0}^{K-1}G^k -\sum_{k=K}^{2K-1}G^k \right)
\end{align}
Similarly, it is easy to verify that $
   \widetilde\Gamma_{K,r}=\sigma^2\omega \left(\sum_{k=0}^{K-1}\widetilde G_r^k -\sum_{k=K}^{2K-1}\widetilde G_r^k \right).
  $
Thus, we get,
\begin{align*}
\Gamma_{K}-\widetilde\Gamma_{K,r}
=\sigma^2\omega
\Bigg[\sum_{k=0}^{K-1}
\left(
G^k-\widetilde G_r^k
\right)-
\sum_{k=K}^{2K-1}
\left(
G^k-\widetilde G_r^k
\right)
\Bigg].
\end{align*}





            
               
                                        
                        
                            
                
                   
                             
                   
            
                                    
                                                                                              
                              
                                         
      
                        
                              
                    
                       
             
                        
                       
             
         
                                      
             
                                                                                                                                                                                            
            
          
                       
          
      
                           
      
                                                                                                   
                                                                       
\noindent Using the identity, 
$\widetilde G_r^m-G^m
=\sum_{j=0}^{m-1}
\widetilde G_r^{\,j}
(\widetilde G_r-G)
G^{\,m-1-j},
$
 and the fact that \(
\|G\|_2=\rho<1,\|\widetilde G_r\|_2\le1
\),  we get
\begin{align*}\|\widetilde G_r^m-G^m\|_2
\le \sum_{j=0}^{m-1}\rho^{m-1-j}\,
\|\widetilde G_r-G\|_2=\frac{1-\rho^m}{1-\rho}\omega\lVert \widetilde B_r- B\rVert_2.
\end{align*}
It follows that
\begin{align}\label{rand_err}
\|\Gamma_{K}-\widetilde\Gamma_{K,r}\|_2
&\le \sigma^2\omega
\sum_{k=1}^{2K-1}
\|G^k-\widetilde G_r^k\|_2
\le
\sigma^2\omega^2\left(
\sum_{k=1}^{2K-1}\frac{1-\rho^k}{1-\rho}
\right)
\|B-\widetilde B_r\|_2
\nonumber\\&= \sigma^2\omega^2 H_{K,\rho}
\|B-\widetilde B_r\|_2 \text{ where } H_{K,\rho}=\frac{2K-1}{1-\rho}-\frac{\rho(1-\rho^{2K-1})}{(1-\rho)^2}.
\end{align}






            
                              
                              
    
         
                  
                          
          
      
                  
   
                        
             
    
     
                                                                  
             
       
       
                  
             
                                  

                                                                                                                                                                                                                
   
                              
   
              
\noindent It thus remains for us to estimate $\lVert B-\widetilde{B}_r\rVert_2$. In order to do this, let us unpack how \cite[Algorithm 5.3]{halko2011} is used to find the randomized SVD approximation to $B:=A^TA$. First of all let $Q$ denote the randomized range finder from \cite[Algorithm 4.1]{halko2011} (see also remark \ref{rem2} below) such that, \begin{align}\label{eq:App1}
    \lVert B- QQ^*B\rVert_2 =\lVert (I-P) B\rVert_2\leq \epsilon
\end{align} for some chosen tolerance level $\epsilon$. In practice, this corresponds to choosing a target rank $r$ and oversampling parameter $p\geq 4$ in the approximation so that \eqref{eq:App1} holds with a high probability. In particular, we have the following result from \cite[Corollary 10.9]{halko2011},
\begin{align}\label{eq:App1.2}
  \mathbb{P}_{\mathrm{rSVD}}\left(  \|(I-P)B\|_2\leq\left(
1+16\sqrt{1+\frac{r}{p+1}}
\right)s_{r+1}^2+\frac{8\sqrt{r+p}}{p+1}
\left(\sum_{j>r}s_j^4\right)^{1/2}\right)\geq 1-3e^{-p}
\end{align}
For notational convenience, let us denote $\alpha_{r,p}:=\left(1+16\sqrt{1+\frac{r}{p+1}}
\right)s_{r+1}^2+\frac{8\sqrt{r+p}}{p+1}
\left(\sum_{j>r}s_j^4\right)^{1/2}$. Now as per \cite[Algorithm 5.3]{halko2011},  for the symmetric positive semi-definite (PSD) matrix, $B$, we first form another PSD matrix, $C=Q^*BQ$, evaluate the eigen-decomposition of $C= V\Lambda V^*$ and then subsequently, $B\approx U\Lambda U^*$ where $U=QV$. Effectively then, we have the matrix $B$ approximated as follows:
\[B\approx QV \Lambda V^*Q^*=QCQ^*=QQ^*BQQ^*=PBP\]
\noindent Now $\lVert B-PBP\rVert_2\leq \lVert (I-P) B\rVert_2+\lVert PB(I-P)\rVert_2\leq 2\lVert (I-P) B\rVert_2 $. Note also that the matrix $Q\in \mathbb{R}^{n\times( r+p)}.$ Thus to get the final randomized rank $r-$ approximation, we truncate
\begin{align}
  \widetilde{B}_r=QV_r\Lambda_rV_r^*Q^*  
\end{align}
Clearly, $\lVert PBP-\widetilde{B}_r\rVert_2=\lambda_{r+1}(PBP)\leq \lambda_{r+1}(B)=s^2_{r+1}$ where $\lambda_{r+1}(M)$ denotes the $r+1-$th ordered eigenvalue of the matrix $M$. Thus we get
\begin{align}\label{eq:App2}
    \lVert B-\widetilde{B}_r\rVert_2&\leq \lVert B-PBP\rVert_2+\lVert PBP-\widetilde{B}_r\rVert_2 \nonumber\\
    &\leq 2\lVert (I-P)B\rVert_2+s_{r+1}^2.
\end{align}
Thus if we define, $\beta_{r,p}:=2\alpha_{r,p}+s_{r+1}^2$, and $E_{r,p}=\{\lVert B-\widetilde{B}\rVert_2\leq \beta_{r,p}\}$ then from \eqref{eq:App1} and \eqref{eq:App2} we get
    $\mathbb{P}(E_{r,p})\geq 1-3e^{-p}.$ Finally, on the event $E_{r,p}$, it follows from \eqref{rand_err} that,
    \begin{align*}
        \lVert \Gamma_K-\widetilde{\Gamma}_{K,r}\rVert_2\leq \sigma^2 \omega^2 H_{K,\rho}\beta_{r,p}:=\Delta_{K,r,p}
    \end{align*}
Thus, $\mathbb{P}_{\mathrm{rSVD}}(\lVert \Gamma_K -\widetilde{\Gamma}_{K,r}\rVert_2\leq \Delta_{K,r,p})\geq 1-3 e^{-p}.$ 

                                                                                                                                                                                                                                
                                                                 

                                                                                                                                                                                                                                                                                                      

\begin{remark}\label{rem2}
 For a chosen tolerance level $\epsilon$, the rank $r$ may in fact adaptively be chosen to ensure $\beta_{r,p}\leq \epsilon$ with high probability by employing \cite [Algorithm 4.2]{halko2011} in lieu of \cite[Algorithm 4.1]{halko2011} at some extra one-time computational cost. Additionally, more recent developments in randomized numerical linear algebra allow one to employ `single-pass' algorithms, such as ``Nystr\"om method" to more efficiently compute the randomised PSD approximation to PSD matrices directly, see e.g. \cite[section 8]{tropp2023}. We also note that the singular values of the Radon transform decay like $j^{-1/2}$, and so the bound in terms of $\sum_{j>r} s_j^4$ is expected to decay as $r \rightarrow \infty$ independent of discretization.
\end{remark}
