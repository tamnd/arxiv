<!-- theorem-1: body_md -->
\label{thm-BM-intro}
		There are integers $\mu_\sigma(\bar\rho)\geq0$,
		depending only on $\bar\rho$ and $\sigma$, such that for any $p$-adic Hodge type $(\mathbf{w},\tau,\psi)$, the following holds
		\[
		e\bigl(R/\varpi\bigr)
		=
		\sum_{\sigma\in W(\bar\rho)}
		[\bar\sigma(\mathbf{w},\tau)^{\operatorname{cr},\operatorname{ss}}:\sigma]\,
		\mu_\sigma(\bar\rho),
		\]
		where $[\bar\sigma(\mathbf{w},\tau)^{\operatorname{cr},\operatorname{ss}}:\sigma]$ is the multiplicity of $\sigma$ occurring in the semisimplification of the mod-$\varpi$ reduction of $\sigma(\mathbf{w},\tau)^{\operatorname{cr}}$.
<!-- end theorem-1 -->
<!-- question-1: body_md -->
\label{question-1}
		For a minimal prime $\mathfrak{p}\in\operatorname{Min}(R)$, what is the
		Breuil--M\'ezard multiplicity
		$
		e\bigl(R/(\mathfrak{p},\varpi)\bigr)
		$
		of the corresponding irreducible component?
<!-- end question-1 -->
<!-- definition-1: body_md -->
We call $\bar\rho$
		\emph{reducible and very generic} if
		\[
		\bar\rho\simeq
		\begin{pmatrix}
			\mathrm{unr}(\bar\beta_1)\omega^{a+b+1}&*\\
			0&\mathrm{unr}(\bar\beta_2)\omega^b
		\end{pmatrix},
		\qquad
		2\leq a\leq p-5,\quad 0\leq b\leq p-2,
		\]
		where $\bar\beta_1,\bar\beta_2\in\mathbb{F}^\times$.
		We call $\tau$ \emph{wildly ramified} if the least integer
		$m\geq1$ such that $\theta_1\theta_2^{-1}$ factors through
		$(\mathbb{Z}/p^m\mathbb{Z})^\times$ satisfies $m\geq2$.
<!-- end definition-1 -->
<!-- theorem-2: body_md -->
\label{thm-Main-results}
		Assume that $p\geq11$.
		Let $\bar\rho$ be reducible and very generic, and let
		$(\mathbf{w},\tau,\psi)$ be a $p$-adic Hodge type such that $\tau$
		is a wildly ramified principal series type.
		Put
		$R=R_{\bar\rho}^{\square,\psi,\operatorname{cr}}(\mathbf{w},\tau)$.
		For every $\mathfrak{p}\in\operatorname{Min}(R)$, write $R(\mathfrak{p}):=R/\mathfrak{p}$.
		Then
		\[
		e(R(\mathfrak{p})/\varpi)=
		\begin{cases}
			1,&\text{if $\bar\rho$ is non-split},\\
			1\text{ or }2,&\text{if $\bar\rho$ is split}.
		\end{cases}
		\]
		Moreover, if $\bar\rho$ is split, then
		$e(R(\mathfrak{p})/\varpi)=1$ (resp.\ $2$) if and only if some (equivalently, every)
		closed point of $\operatorname{Spec} R(\mathfrak{p})[1/\varpi]$ corresponds
		to a reducible (resp. an irreducible) Galois representation.
<!-- end theorem-2 -->
<!-- remark-1: body_md -->
\begin{enumerate}
			\item By Lemma~\ref{lem comm alg2}, whenever $e(R(\mathfrak{p})/\varpi)=1$,
			the ring $R(\mathfrak{p})$ is simply a formal power series ring over $\mathcal{O}$. However, our method still cannot determine how these components intersect in $R$.
			\item 
			We also determine the number of irreducible components
			of $R$.
			If $\bar\rho$ is non-split, then
			$
			\#\operatorname{Min}(R)=e(R/\varpi)$ (Theorem~\ref{thm-BM=1-non-split}).
			If $\bar\rho$ is split, then
			$
			2\#\operatorname{Min}(R)-e(R/\varpi)\in\{0,1,2\}$; see
			Corollary~\ref{cor-split-compare-mult} and
			Theorem~\ref{thm-main-thm-split} for explicit formulas.
					\end{enumerate}
<!-- end remark-1 -->
<!-- remark-2: body_md -->
We emphasize that Theorem~\ref{thm-Main-results} imposes no further restriction

		$\Delta$, where we identify $\Delta$ with the subgroup $\operatorname{diag}(\Delta,1)$
		of $\bar T$.

		$\mathrm{Iw}_m\backslash\mathrm{Iw}$ has one fixed point $\{0\}$ and
		$(p^{m-1}-1)/(p-1)$ free orbits (i.e. the stabilizer is trivial), from which we obtain
		\[

		\[

		lift of $\lambda$. Indeed,

		u(t)\begin{pmatrix}[\lambda]&0\\0&1\end{pmatrix}
		=\begin{pmatrix}[\lambda]&0\\pt[\lambda]&1\end{pmatrix}

<!-- proof-15: body_md -->
By construction, $A$ is a reduced complete Noetherian local
		$\mathcal{O}$-flat algebra.
		The map $A\to A^\square$ is formally smooth and induces an
		isomorphism on residue fields, hence 
		$
		A^\square\simeq A[[X_1,\ldots,X_n]]$.
		Consequently,
		$
		A^\square/\varpi
		\simeq (A/\varpi)[[X_1,\ldots,X_n]]$, and this gives
		$e(A/\varpi)=e(A^\square/\varpi)$.
		
		For every prime ideal $\mathfrak{p}$ of $A$, we have
		$
		A^\square/\mathfrak{p} A^\square
		\simeq (A/\mathfrak{p})[[X_1,\ldots,X_n]]$.
		Thus $\mathfrak{p} A^\square$ is prime and
		$\mathfrak{p} A^\square\cap A=\mathfrak{p}$.
		If $\mathfrak{p}$ is minimal and $\mathfrak q\subseteq\mathfrak{p} A^\square$
		is a prime ideal, then $\mathfrak q\cap A=\mathfrak{p}$, which forces
		$\mathfrak{p} A^\square\subseteq\mathfrak q$.
		Hence $\mathfrak{p} A^\square$ is minimal, and
		\eqref{eq-pf-2} is well-defined and injective.
		Conversely, if $\mathfrak q\in\operatorname{Min}(A^\square)$, going-down
		for the flat map $A\to A^\square$
		\cite[Lemma~10.11]{eis} implies that
		$\mathfrak{p}:=\mathfrak q\cap A$ is a minimal prime of $A$.
		Since $\mathfrak{p} A^\square$ is a prime ideal contained in
		$\mathfrak q$, minimality gives
		$\mathfrak q=\mathfrak{p} A^\square$.
		This proves the bijection.
		
		By Theorem~\ref{thm-Kisin}, $A^\square$ is equidimensional.
		For every $\mathfrak{p}\in\operatorname{Min}(A)$, the preceding bijection gives
		$
		\dim(A/\mathfrak{p})
		=
		\dim(A^\square/\mathfrak{p} A^\square)-n
		=
		\dim A^\square-n
		=
		\dim A$.
		Thus $A$ is equidimensional.
		Moreover,
		$
		A^\square/(\mathfrak{p} A^\square,\varpi)
		\simeq
		\bigl(A/(\mathfrak{p},\varpi)\bigr)[[X_1,\ldots,X_n]]$
		gives the asserted equality
		for the special fibers of corresponding irreducible components.
		
		Finally, $A\to A^\square$ is a flat local homomorphism,
		hence faithfully flat \cite[Lemma~10.39.17]{stack}.
		Its base change
		$
		A[1/\varpi]\longrightarrow A^\square[1/\varpi]$
		is also faithfully flat by

		Let $a\in\mathfrak{m}$ be a non-zero-divisor, and set
		$
		\mathrm{ord}_{\mathfrak{m}}(a):=\max\{j\geq 0:a\in\mathfrak{m}^j\}$.
		Then we have
		$
		e(A/a)\geq \mathrm{ord}_{\mathfrak{m}}(a)\cdot e(A)$.
<!-- end lemma-8 -->
<!-- proof-22: body_md -->
Since $a\in\mathfrak{m}$ is a non-zero-divisor,
		$\dim(A/a)=\dim A-1$, so $a$ is part of a system of parameters
		of $A$; see, for example,
		\cite[Lemma~10.60.13]{stack} and its proof.
		The assertion then follows from \cite[Theorem~14.9]{mat}.

		$
		\mathrm{ord}(c):=
		\max\{j\geq 0:c\in(T_1,\dots,T_n)^j\}

		Then we have
		$
		e(A)=\min\{2,\mathrm{ord}(c)\}\leq 2$.

<!-- proof-23: body_md -->
Set $S=\mathbb{F}[\![x,y,T_1,\dots,T_n]\!]$,
		$f=xy-c$, and $\mathfrak{m}=(x,y,T_1,\dots,T_n)\subseteq S$.
		Since $S$ is a regular local ring and $f\neq 0$,
		\cite[(40.2)]{Nagata} gives
		$
		e(A)=\mathrm{ord}_{\mathfrak{m}}(f)$.
		As $c$ involves only $T_1,\dots,T_n$, we have
		$
		\mathrm{ord}_{\mathfrak{m}}(f)=\min\{2,\mathrm{ord}(c)\}$,
		which proves the assertion.

<!-- lemma-10: body_md -->
\label{lem comm alg2}
		Let $A$ be a complete Noetherian local $\mathcal{O}$-flat algebra
		that is equidimensional of dimension $n$ and has residue
		field $\mathbb{F}$.
		Suppose that $A[1/\varpi]$ is reduced and $e(A/\varpi)=1$.
		Then
		$
		A\simeq\mathcal{O}[\![X_1,\dots,X_{n-1}]\!]$.
<!-- end lemma-10 -->
<!-- proof-24: body_md -->
This is a slight generalization of \cite[Lemme~5.1.8]{BM02},
		which treats the case $n=2$ under the stronger assumption
		that $A[1/\varpi]$ is regular.
		By Nagata's multiplicity-one criterion
		\cite[Theorem~40.6]{Nagata}, a complete unmixed Noetherian
		local ring $R$ with $e(R)=1$ is regular.
		We first apply this criterion to show that $A$ is regular.
		
		Since $A$ is $\mathcal{O}$-flat, $\varpi$ is a non-zero-divisor
		in $A$.
		Thus Lemma~\ref{lem-ca1} gives
		$
		\mathrm{ord}_{\mathfrak{m}}(\varpi)\cdot e(A)
		\leq e(A/\varpi)=1$.
		It follows that $e(A)=1$ and
		$\varpi\in\mathfrak{m}\setminus\mathfrak{m}^2$.
		Moreover, $\mathcal{O}$-flatness implies that the natural map
		$A\longrightarrow A[1/\varpi]$ is injective.
		Since $A[1/\varpi]$ is reduced, so is $A$.
		Consequently,
		$
		\operatorname{Ass}(A)=\operatorname{Min}(A)$.
		Together with the equidimensionality of $A$, this gives
		$\dim A/\mathfrak{p}=\dim A$ for every $\mathfrak{p}\in\operatorname{Ass}(A)$.
		Thus $A$ is unmixed, and Nagata's criterion implies that
		$A$ is regular.
		
		Since $A$ is regular of dimension $n$ and
		$\varpi\in\mathfrak{m}\setminus\mathfrak{m}^2$, we may choose
		$x_1,\dots,x_{n-1}\in\mathfrak{m}$ such that
		$\bar\varpi,\bar x_1,\dots,\bar x_{n-1}$ form an
		$\mathbb{F}$-basis of $\mathfrak{m}/\mathfrak{m}^2$.
		Nakayama's lemma then gives
		$
		\mathfrak{m}=(\varpi,x_1,\dots,x_{n-1})$.
		By \cite[Theorem~7.16(a)]{eis}, there is a unique continuous
		homomorphism of $\mathcal{O}$-algebras
		\[
		\phi\colon B:=\mathcal{O}[\![X_1,\dots,X_{n-1}]\!]
		\longrightarrow A,
		\qquad X_i\longmapsto x_i.
		\]
		Since $\mathcal{O}/(\varpi)\simeq\mathbb{F}\simeq A/\mathfrak{m}$ and
		$\mathfrak{m}=(\varpi,x_1,\dots,x_{n-1})$,
		\cite[Theorem~7.16(b)]{eis} implies that $\phi$ is surjective.
		Since $A$ and $B$ are regular and $\phi$ identifies their
		chosen regular systems of parameters, the induced map on
		associated graded rings is an isomorphism.
		Hence \cite[Theorem~7.16(c)]{eis} implies that $\phi$ is
		injective, and we obtain the required isomorphism
		$
		\mathcal{O}[\![X_1,\dots,X_{n-1}]\!]\simeq A$.

<!-- lemma-11: body_md -->
\label{lem multiplicity-components}
		Let $A$ be a complete Noetherian local $\mathcal{O}$-flat algebra
		that is equidimensional.
		Suppose that $A[1/\varpi]$ is regular.
		Write
		$
		\operatorname{Min}(A)
		=\{\mathfrak p_1,\dots,\mathfrak p_r\}$, and put
		$A_i:=A/\mathfrak p_i$.
		Then we have
		\[
		e(A/\varpi)
		=\sum_{i=1}^r e(A_i/\varpi)\geq r.
		\]
<!-- end lemma-11 -->
<!-- proof-25: body_md -->
By \cite[Lemma~5.1.6]{BM02}, we only need to check that the condition ``$A\sim \Pi_iA_i$" defined in \cite[Definition~5.1.3]{BM02} holds, namely
		\begin{enumerate}
			\item The natural map $A\rightarrow \Pi_iA_i\twoheadrightarrow A_i$ is surjective for any $i$, and 
			\item 
			the above map induces an isomorphism on the generic fiber $A[1/\varpi]\simeq\Pi_iA_i[1/\varpi]$.
		\end{enumerate}
		(1) is clear in our case and (2) follows from the reducedness of $A$ and the regularity of $A[1/\varpi]$. Indeed, it suffices to show the natural homomorphism $$\varphi:A[1/\varpi]\rightarrow \Pi_iA[1/\varpi]/\mathfrak{p}_iA[1/\varpi]=\Pi_i A_i[1/\varpi]$$ is an isomorphism. The kernel of $\varphi$ is $\cap_i(\mathfrak{p}_iA[1/\varpi])=(\cap_i\mathfrak{p}_i)A[1/\varpi]=0$ as $A$ is reduced. The regularity of $A[1/\varpi]$ ensures that irreducible components of $\operatorname{Spec} A[1/\varpi]$ are pairwise disjoint, namely, $\mathfrak{p}_iA[1/\varpi]+\mathfrak{p}_jA[1/\varpi]=A[1/\varpi]$ for any $i\neq j$. By the Chinese remainder theorem we deduce that $\varphi$ is an isomorphism. This lemma is proved.

<!-- lemma-12: body_md -->
\label{lem-minimal-primes}
		Let $A$ be a ring and let $I,J\subseteq A$ be ideals. 
		\begin{enumerate}
			\item If
			$\mathfrak p$ is minimal over $I\cap J$ and $\mathfrak p\in V(I)$, then $\mathfrak p$ is minimal over $I$.
			\item If
			$\mathfrak p$ is minimal over $I$ and $\mathfrak p\in V(I)-V(J)$, then $\mathfrak p$ is minimal over $I\cap J$.
		\end{enumerate}
<!-- end lemma-12 -->
<!-- proof-26: body_md -->
For (1),
		let $\mathfrak q$ be a prime ideal satisfying
		$
		I\subseteq\mathfrak q\subseteq\mathfrak p$.
		Then $I\cap J\subseteq I\subseteq\mathfrak q\subseteq\mathfrak{p}$. Since $\mathfrak p$ is minimal
		over $I\cap J$, it follows that $\mathfrak q=\mathfrak p$. Hence
		$\mathfrak p$ is minimal over $I$.
		
		For (2), let $\mathfrak q$ be a prime ideal such that
		$
		I\cap J\subseteq \mathfrak q\subseteq \mathfrak p$.
		Since
		$
		IJ\subseteq I\cap J\subseteq \mathfrak q$
		and $\mathfrak q$ is prime, either $I\subseteq\mathfrak q$ or
		$J\subseteq\mathfrak q$. Our assumption $\mathfrak{p}\notin V(J)$ implies that $J$ is not contained in $\mathfrak q$, hence $I\subseteq\mathfrak q$. Since
		$\mathfrak p$ is minimal over $I$, we conclude that
		$\mathfrak q=\mathfrak p$. Hence $\mathfrak p$ is minimal over
		$I\cap J$.
