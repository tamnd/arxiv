<!-- theorem-1: body_md -->
\label{thm:main phi4}
For every $L\geq1$ and $m,\lambda>0$, consider the
$\Phi^4_2$ model or the sine-Gordon model with fixed
$\beta^2<2\pi$ on $\mathbb T_L^2$.
Fix $\sigma_p\geq\sigma>0$ such that
\eqref{eq:lsi-lattice} and~\eqref{eq:poincare-lattice}
hold uniformly in $\delta$.

There exist finite constants $C>0$ and
$R=R(m,\lambda,\sigma)$, with $C$ universal and $R$
possibly depending on $L$ only through $\sigma$, such that
for every $\gamma>0$, $t\geq0$, and
$f\in L^2(\mu\otimes\kappa)$ with
$\|f\|_{L^2(\mu\otimes\kappa)}=1$,
\begin{equation}\label{eq:main-convergence}
\left\|P_t f-\int f\mathop{}\!\mathrm d(\mu\otimes\kappa)\right\|_{L^2(\mu\otimes\kappa)}
\leq C\exp\left(
-\frac{\sigma_p\gamma t}
{C(\sqrt{\sigma_p}+R+\gamma)^2}
\right).
\end{equation}
Moreover, for the $\Phi^4_2$ model and every
$\varepsilon>0$, there exists a constant
$C_{L,\varepsilon}>0$ such that the estimate
\eqref{eq:main-convergence} also holds with
$R\leq C_{L,\varepsilon}m$, for all
$0<m\leq1$ and
$0<\lambda\leq\lambda_0(m)\wedge m^{4+\varepsilon}$.

<!-- remark-1: body_md -->
\label{remark:comparison}
The lattice log-Sobolev and Poincar\'e inequalities pass to the
continuum limit, so that $\sigma$ and $\sigma_p$ are also
admissible constants for the corresponding inequalities under
$\mu$. Denoting by $\sigma_p^{\mathrm{opt}}(\mu)$ the optimal
Poincar\'e constant of $\mu$, we thus have
$\sigma_p^{\mathrm{opt}}(\mu)\geq\sigma_p\geq\sigma$.
For the $\Phi^4_2$ model in the weak-coupling regime of
Theorem~\ref{thm:main phi4}, the susceptibility estimates of
Bauerschmidt and Dagallier~\cite{bauerschmidt_logsobolev_2024}
give $\sigma_p^{\mathrm{opt}}(\mu)\lesssim m^2$,
uniformly in $L$, by testing the Poincar\'e inequality with
the spatial average of the field. Since one may choose
$\sigma_p=\sigma\gtrsim m^2$, it follows that
$\sigma_p^{\mathrm{opt}}(\mu) \simeq m^2$.
Together with the bound $R\leq C_{L,\varepsilon}m$,
Theorem~\ref{thm:main phi4} then yields a kinetic relaxation
rate of at least
\begin{equation}
\frac{m^2\gamma}{C_{L,\varepsilon}(m+\gamma)^2}.
\end{equation}
Choosing $\gamma=m$ yields a kinetic relaxation rate of at least
order $m$, compared with the optimal overdamped relaxation rate
$\sigma_p^{\mathrm{opt}}(\mu) \simeq m^2$. 
This establishes the announced acceleration in the small-mass regime.
<!-- end remark-1 -->
<!-- remark-2: body_md -->
\label{rem:regular-one-dimensional}
In one spatial dimension, analogous $L^2$ convergence estimates
follow from the dimension-free result of Cao, Lu, and
Wang~\cite{cao_explicit_2023}. For a smooth, bounded-below
interaction potential $V$ satisfying suitable growth conditions
and $\operatorname{Hess} V\geq-R^2$, their result yields, provided the
corresponding lattice Gibbs measures satisfy a uniform
Poincar\'e inequality with constant $\sigma_p>0$,
a relaxation rate
\[
\alpha\geq
\frac{\sigma_p\gamma}
{C(\sqrt{\sigma_p}+R+\gamma)^2},
\]
uniformly in the lattice discretization.
The corresponding continuum estimate follows by passing
to the limit as in~\cite{LATTICE}.

\begin{equation}\label{eq:potential assumption exponential}
\sup_{i\in\mathbb Z^d}
\log\int
\exp\left(\frac{4}{\sigma}Q_{\delta,i}(u)\right)
\mathop{}\!\mathrm d\mu_\delta(u)
\leq\frac{4R^2}{\sigma},
\end{equation}
where $\sigma$ is as in Assumption~\ref{ass:log-sobolev}.

<!-- theorem-2: body_md -->
\label{thm:decay of variance}
Under Assumption~\ref{ass:log-sobolev} and either
Assumption~\ref{ass:hessian finite} or~\ref{ass:potential-intro}, together with
the technical Assumption~\ref{ass:technical} stated
in Section~\ref{sec:clw}, there exists a universal
finite constant $C>0$ such that for every
$\delta>0$ with $L/\delta\in\mathbb N$,
$\tilde\gamma>0$, $t'\geq0$, and
$f\in L^2(\mu_\delta\otimes\kappa_\delta)$,
\[
\left\|P_{t'}^\delta f-\int f\mathop{}\!\mathrm d(\mu_\delta\otimes\kappa_\delta)\right\|_{L^2(\mu_\delta\otimes\kappa_\delta)}
\leq C e^{-\alpha_\delta t'}
\left\|f-\int f\mathop{}\!\mathrm d(\mu_\delta\otimes\kappa_\delta)\right\|_{L^2(\mu_\delta\otimes\kappa_\delta)},
\]
where
\[
\alpha_\delta=
\frac{\sigma_p\delta^d\tilde\gamma}
{C\bigl[\delta^{d/2}(\sqrt{\sigma_p}+\tilde R)
+\tilde\gamma\bigr]^2},
\]
with $\tilde R=R$ under Assumption~\ref{ass:hessian finite}
and $\tilde R=\sqrt{R^2+C_d}$ under
Assumption~\ref{ass:potential-intro},
where $C_d$ only depends on the dimension of the lattice~$\Lambda$.

\label{thm:integrability}
Let $d=2$, and consider the lattice $\Phi^4_2$ model or
the lattice sine-Gordon model with fixed $0<\beta^2<4\pi$.

\begin{enumerate}
\item[(i)] For every $L\geq1$, $m,\lambda>0$, and
$\delta>0$ with $L/\delta\in\mathbb N$, there exist functions
$Q_{\delta,i}:\mathbb R^{\Lambda_\delta}\to\mathbb R$,
indexed by $i\in\mathbb Z^2$, such that for every
$i\in\mathbb Z^2$, $u\in\mathbb R^{\Lambda_\delta}$,
$h\in\mathbb R^{\Lambda_\delta}$ supported in $B(i,2)$,
and $q>0$,
\begin{equation}\label{eq:integrability bounds}
\begin{split}
\delta^2\sum_{x\in\Lambda_\delta}
\left[
\bigl(V_\delta''(u(x))+m^2\bigr)h(x)^2
+\frac12|\nabla_\delta h(x)|^2
\right]
&\geq
-Q_{\delta,i}(u)\|h\|_{L^2(\Lambda_\delta)}^2,\\
\sup_{i\in\mathbb Z^2}\log\int
\exp\bigl(qQ_{\delta,i}(u)\bigr)\mathop{}\!\mathrm d\mu_\delta(u)
&\leq \mathcal R(q,m,\lambda),
\end{split}
\end{equation}
where $\mathcal R(q,m,\lambda)<\infty$ is independent
of $\delta$ and $L$.

\item[(ii)] For the lattice $\Phi^4_2$ model, let
$Q_\delta(u)=-\lambda_{\min}(\operatorname{Hess} U(u))$,
as in Assumption~\ref{ass:hessian finite}.
There exists a universal constant $c_0>0$ such that
for every $L\geq1$ and $\varepsilon>0$, there is a
constant $C_{L,\varepsilon}<\infty$ with the following
property: for all $0<m\leq1$ and
$0<\lambda\leq\lambda_0(m)\wedge m^{4+\varepsilon}$,
\begin{equation}\label{eq:global spectral estimate phi4}
\sup_{\substack{\delta>0\\L/\delta\in\mathbb N}}
\log\int
\exp\left(\frac{4}{c_0m^2}Q_\delta(u)\right)
\mathop{}\!\mathrm d\mu_\delta(u)
\leq C_{L,\varepsilon}.
\end{equation}
\end{enumerate}

Item~(i) is proved in Section~\ref{sec:measures}.
Item~(ii) follows immediately, since the additional $\tfrac 1 2 |\nabla_\delta h|^2$ term
only improves the estimates.

<!-- remark-3: body_md -->
\label{rem:global-spectral-estimate}
The global estimate in Theorem~\ref{thm:integrability}(ii)
verifies Assumption~\ref{ass:hessian finite} with
$R=O(m)$ at fixed volume.
Indeed, in the weak-coupling regime under consideration,
$\sigma_*=c_0m^2$ is an admissible log-Sobolev constant,
uniformly in $\delta$.
By~\eqref{eq:global spectral estimate phi4},
Assumption~\ref{ass:hessian finite} therefore holds with
\[
R^2=\frac{\sigma_*}{4}C_{L,\varepsilon}
=\frac{c_0C_{L,\varepsilon}}{4}m^2.
\]
This is the bound needed to establish the small-mass
acceleration discussed in Remark~\ref{remark:comparison}.

<!-- remark-4: body_md -->
\label{rem:bottom-spectrum}
The functions $Q_{\delta,i}$ in
Theorem~\ref{thm:integrability}(i) can be chosen as
negatives of the lowest eigenvalues of modified discrete
Schr\"odinger operators with localized potentials.
By contrast, $-Q_\delta$ in
Theorems~\ref{thm:integrability}(ii)
and~\ref{thm:3 dimensions} is the lowest eigenvalue
of $\operatorname{Hess} U$.
These results thus provide exponential lower-tail
estimates for the spectra of singular random
Schr\"odinger operators.

In the $\Phi^4_3$ model, the potential $V_\delta''(u)$
is non-Gaussian and renormalized, with spatial regularity
analogous to that of two-dimensional white noise.
Its spectral analysis is therefore closely related
to that of the Anderson Hamiltonian; see
\cite{allez_continuous_2015,gubinelli_semilinear_2020,
labbe_continuous_2019}.
For the white-noise Anderson Hamiltonian, exponential
lower-tail estimates for the lowest eigenvalue are known;
see~\cite{allez_continuous_2015,labbe_continuous_2019}.
To our knowledge, such estimates have not previously been established for the non-Gaussian, renormalized potential considered here. They may be of independent interest.

<!-- remark-5: body_md -->
To obtain convergence estimates uniform also in the volume

\label{thm:ngs bound}
Assume that the measure $\mathop{}\!\mathrm d\mu$ satisfies a log-Sobolev inequality with constant $\sigma$.
Let $\psi \colon \mathbb R^\Lambda \to \mathbb R$ be a continuous function such that

<!-- lemma-3: body_md -->
\label{thm:clw 2.3}
Let $g \in H^2(\tau \otimes \mu_\delta)$ be such that
$(\partial_t g, \nabla_u g) \in H^1_0(\tau \otimes \mu_\delta)^{1 + |\Lambda|}$.
Then
\[
\begin{gathered}
\|D^2 g\|_{L^2(\tau \otimes \mu)}^2
\leq 2\|(-\partial_{tt} + \nabla_u^\ast \nabla_u) g\|_{L^2(\tau \otimes \mu)}^2
    + 2(R^2+2^d) \delta^d \|\nabla_u g\|_{L^2(\tau \otimes \mu)}^2, \text{ and}\\
\|\nabla_u^2 g\|_{L^2(\tau \otimes \mu)}^2
\leq 2\|\nabla_u^\ast \nabla_u g\|_{L^2(\tau \otimes \mu)}^2
    + 2(R^2+2^d) \delta^d \|\nabla_u g\|_{L^2(\tau \otimes \mu)}^2.
\end{gathered}
\]
Here $D^2$ is the Hessian with respect to all of
$t$ and $u(x)$, $x \in \Lambda$.
If  Assumption~\ref{ass:hessian finite} holds,
then the above bounds hold with $R^2+2^d$ replaced by $R^2$.
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
We only prove the second statement, the first is proved analogously.
As in \cite[Eq.~(20)]{cao_explicit_2023}, we start from the following Bochner identity,
also known as the $\Gamma_2$ identity \cite[Section~1.16]{bakry_analysis_2014}:
\begin{equation}\label{eq:clw 2.3 beginning}
\begin{split}
\|\nabla_u^2 g\|_{L^2(\tau \otimes \mu)}^2
&= \sum_{x,y \in \Lambda} \iint |\partial_{u(x)} \partial_{u(y)} g(u)|^2 \mathop{}\!\mathrm d\mu_\delta(u) \mathop{}\!\mathrm d\tau(t)\\
&= \|\nabla_u^\ast \nabla_u g\|_{L^2(\tau \otimes \mu)}^2\\
&\qquad -\iint
    (\nabla_u g(t,u))^\top (\nabla_u^2 U(u)) (\nabla_u g(t,u)) \mathop{}\!\mathrm d\mu_\delta(u) \mathop{}\!\mathrm d\tau(t).
\end{split}
\end{equation}
We estimate the second term from above.
By computing the Hessian of the energy $U$, we find that the second term equals
\begin{equation}
-\iint \delta^d \sum_{x \in \Lambda} \Bigl[ (V''(u(x)) + m^2) (\partial_{u(x)} g(t,u))^2
        + |\nabla_\delta \partial_{u(x)} g(t,u)|^2 \Bigr]
    \mathop{}\!\mathrm d\mu_\delta(u) \mathop{}\!\mathrm d\tau(t).
\end{equation}
We now decompose $h(x)^2 \coloneqq \partial_{u(x)} g(t,u)^2$ as $\sum_{i\in \mathbb Z^d} h_i^2$
given by $h_i=\partial_{u(x)} g(t,u) \chi_{i}$,
where $\chi^{2}_i$ is a suitable partition of unity supported on $B(i,2)$ for $i\in \mathbb Z^d$.
We will also require a family $\tilde{\chi}_i$ of smooth cutoff functions supported in $B(i,4)$
and equal to $1$ on the support of $\chi_i$,
such that $\sum_{i\in \mathbb Z^d} \tilde{\chi}^2_i = 2^d$.
Then we can write the above expression as 
\begin{equation}
-\sum_{i \in \mathbb Z^d}\iint \delta^d \sum_{x \in \Lambda} \Bigl[ V''(u(x)) h_i^2
        + m^2 h_i^2 + |\chi_i \nabla_\delta h|^2 \Bigr]
    \mathop{}\!\mathrm d\mu_\delta(u) \mathop{}\!\mathrm d\tau(t).
\end{equation}
We may write
\begin{equation}
\chi_i \nabla_\delta h
= \chi_i \nabla_\delta (\tilde\chi_i h)
= \nabla (\chi_i \tilde\chi_i h) - (\nabla \chi_i) \tilde\chi_i h
= \nabla (\chi_i h) - (\nabla \chi_i) \tilde\chi_i h.
\end{equation}
Therefore we can estimate with Young's inequality
\begin{equation}
-|\chi_i \nabla_\delta h|^2
\leq -\frac 1 2 |\nabla_\delta (\chi_i h)|^2
    + |(\nabla \chi_i) \tilde\chi_i h|^2.
\end{equation}
By construction $|\nabla \chi_i| \leq C$ for some uniform, dimension-dependent constant.
Therefore
\begin{equation}
\begin{split}
&\mathrel{\phantom{=}}
-\iint \delta^d \sum_{x \in \Lambda} \Bigl[ V''(u(x)) h_i(x)^2 + m^2 h_i(x)^2
        + | \chi_i \nabla_\delta h|^2 \Bigr]
    \mathop{}\!\mathrm d\mu_\delta(u) \mathop{}\!\mathrm d\tau(t)\\
&\leq -\iint \delta^d \sum_{x \in \Lambda} \Bigl[ V''(u(x)) h_i(x)^2 + m^2 h_i(x)^2
        + \frac{1}{2}| \nabla_\delta h_i|^2 \Bigr] + C \delta^d \sum_{x \in \Lambda} \tilde{\chi}^{2}_{i}  h^2 \mathop{}\!\mathrm d\mu_\delta(u) \mathop{}\!\mathrm d\tau(t).
\end{split}
\end{equation}
Now by Assumption~\ref{ass:potential-intro} this can be bounded by
\begin{equation}\label{eq:clw 2.3 sum}
\sum_{x \in \Lambda} \delta^d \iint (Q_i(u)) (h_i)^2 
    \mathop{}\!\mathrm d\mu_\delta(u) \mathop{}\!\mathrm d\tau(t) + C \delta^d \sum_{x \in \Lambda} \tilde{\chi}^{2}_{i}  h^2 \mathop{}\!\mathrm d\mu_\delta(u) \mathop{}\!\mathrm d\tau(t)
\end{equation}
Now by Lemma~\ref{thm:ngs bound} (taking $\psi(u) = 2 Q(u)$ and $f(u) = 2^{-1/2} h_i(x)$
with $x$ and $i$ fixed)
and $R^2$ from Assumption~\ref{ass:potential-intro}, the integral is bounded by
\begin{equation}
\int \Bigl[ \delta^d R^2 \int
        (h_i(t,x,u))^2 \mathop{}\!\mathrm d\mu_\delta(u)
    + \frac 1 2 \int |\nabla_u h_i(t,x,u)|^2 \mathop{}\!\mathrm d\mu_\delta(u) \Bigr] \mathop{}\!\mathrm d\tau(t)+C \delta^d \sum_{x \in \Lambda} \tilde{\chi}^{2}_{i}  h^2 \mathop{}\!\mathrm d\mu_\delta(u) \mathop{}\!\mathrm d\tau(t)
\end{equation}
Summing over $x \in \Lambda$ and $i\in \mathbb Z^d$ we obtain 
\begin{equation}
\begin{split}
&\mathrel{\phantom{=}}
    \|\nabla_u^2 g\|_{L^2(\tau \otimes \mu)}^2\\
&\leq \|\nabla_u^\ast \nabla_u g\|_{L^2(\tau \otimes \mu)}^2
    + \delta^d (R^2+2^d C) \iint |\nabla_u g(t,u)|^2 \mathop{}\!\mathrm d\mu_\delta(u) \mathop{}\!\mathrm d\tau(t)
    + \frac{1}{2} \|\nabla_u^2 g\|_{L^2(\tau \otimes \mu)}^{2}.
\end{split}
\end{equation}
Subtraction of the last term on the right-hand side gives the desired bound. 
Under Assumption~\ref{ass:hessian finite} the proof proceeds
without using the partition of unity,
which removes the $2^d C$~term on the right-hand side.

<!-- lemma-4: body_md -->
\label{lem:Grad-bottom}
Assume $V \in C^3(\mathbb R)$.
For any $\delta > 0$, we have
\[
-\partial_{u(x)} Q(u) = \chi(x) V'''(u(x))(\psi^u(x))^2,
\]
where $\psi^u$ of unit $L^2(\Lambda)$ norm minimizes
\[
    \psi \mapsto
    \sum_{x \in\Lambda}
    \left(\chi V''(u)\psi^2
    +m^2\psi^2+ \frac 1 2 |\nabla_\delta \psi|^2\right).
\]
<!-- end lemma-4 -->
<!-- proof-7: body_md -->
The gradient of $Q(u)$ is well-defined by Kato perturbation theory.
Indeed Q is the bottom eigenvalue of a discrete Schrödinger operator,
which is known to be simple by comparison principle,
so Kato perturbation theory implies that $Q$ is smooth if $V$ depends smoothly on $u$~\cite[Chapter~2]{kato_1996}. 
For any $v \in L^2(\Lambda)$ and $t > 0$,
we can plug $\psi^u$ into both infima to estimate
\begin{equation}
\begin{split}
&\mathrel{\phantom{=}} \frac{1}{t} \inf_{\|\psi\|_{L^2(\Lambda)} = 1}
    \left( \sum_{x \in\Lambda} \chi V'' (u + t v) \psi^2
        + \frac 1 2 |\nabla_\delta \psi|^2
        + m^2 \psi^2  \right)\\
&\qquad - \frac{1}{t} \inf_{\|\psi\|_{L^2(\Lambda)} = 1}
    \left( \sum_{x \in\Lambda} \chi V'' (u) \psi^2
        + \frac 1 2 |\nabla_\delta \psi|^2
        + m^2 \psi^2   \right)\\
&\leq \frac{1}{t} \left( \sum_{x \in\Lambda} \chi V''(u + t v) (\psi^u)^2 
    - \sum_{x \in\Lambda} \chi V''(u) (\psi^u)^2  \right)\\
&\to \sum_{x \in\Lambda} \chi V'''(u)(\psi^u)^2 v
\end{split}
\end{equation}
in the $t \to 0$ limit.
We obtain the converse inequality by replacing $v$ with $-v$.
Therefore
\begin{equation}
-\langle \nabla_u Q (u), v \rangle_{L^2 (\Lambda)}
= \langle \chi V''' (u) (\psi^u)^2, v \rangle_{L^2 (\Lambda)},
\end{equation}
from which the statement follows by duality.

<!-- proof-8: body_md -->
Let $\psi=\psi^{W+I}$ be as in Lemma~\ref{lem:Grad-bottom},
minimizing the quadratic form
\begin{equation}\label{eq:estimate-psi-phi42 qf}

Since $\|\psi\|_{L^2}^2=1$ by assumption,
\begin{equation}
\|\nabla_\delta \psi\|_{L^2}^2

\begin{equation}\label{eq:estimate-psi-phi42 w terms}
\begin{split}

\end{equation}

\begin{equation}
\begin{split}
\eqref{eq:estimate-psi-phi42 w terms}

&\leq \frac{1}{4} \|\psi\|_{H^1}^2

\begin{equation}\label{eq:estimate-product-WI}

&\leq C q\lambda  \|W\|_{\mathcal C^{-\varepsilon}(\rho)}

\end{split}

\end{equation}

    + \lambda^{2\varepsilon} \|\chi^{1/2} W\|_{\mathcal C^{-\varepsilon}(\rho)}^\alpha X^{1+\alpha}
    + \lambda^{2\varepsilon} X^{1+4\varepsilon}
    + X
\right]\!.
\end{equation}
We extract~$X^2$ from each term with Young's inequality (which requires $\varepsilon<1/4$).

\begin{equation}
\begin{split}
&\mathbin{\phantom{+}}

    \|\chi^{1/2} W\|_{\mathcal C^{-\varepsilon}(\rho)}^{2\alpha/(1-\alpha)}\\

&+ C_\kappa q^2 \lambda^2 \|W\|_{\mathcal C^{-\varepsilon}(\rho)}^2\\
&+ \kappa m^2 X^2.
\end{split}

\label{prop:coupling}
Let $\mu_{\beta}$ be the lattice sine-Gordon measure for $\beta^2 < 4 \pi$,
and $\rho$ a weight as in Section~\ref{sec:phi42 measure}.
There exists a probability space $\mathord{\mathbb P}$ and random variables

and $I \in L^{\infty} (\mathord{\mathbb P};\, \mathcal C^1(\Lambda))$ such that
\[
\mu_{\beta} = \operatorname{Law}(W + I), \quad \nu = \operatorname{Law}(W).
\]
such that $\sup_{L} \sup_{\delta} \|I\|_{L^{\infty} (\mathord{\mathbb P}, \mathcal C^1 (\Lambda))}
\leq A(m,\lambda)$,
where $A(m,\lambda) = C\lambda
    + \frac{C \lambda^2}{m^2} \exp \!\left( \frac{C \lambda}{m^2} \right)$.

<!-- proof-14: body_md -->
This statement has essentially been proved
in~\cite{barashkov_stochastic_2022,gubinelli_fbsde_2026,bauerschmidt_logsobolev_2021,bauerschmidt_maximum_2022}.
We reprove it here in the context of lattice approximations and uniformly in the volume,
while also tracking the dependence on~$\lambda$ and~$m$.
We will opt for the method of Gubinelli and Meyer~\cite{gubinelli_fbsde_2026}.

By~\cite[Lemma~2.5]{gubinelli_fbsde_2026},
there exists an optimal~$h\in \mathbb{H}_a$ in Theorem~\ref{thm:boue-dupuis}
that satisfies almost surely the equation
\begin{equation}\label{eq:FBDSE}
h_{t}
=- \lambda\beta J_{t} \mathord{\mathbb E} [ \mathord{\mathopen: \sin (\beta W_{\infty} + \beta I_{\infty}[h]) \mathclose:} \mid \mathcal{F}_t ] .
\end{equation}
One can show \cite{barashkov_variational_2023,gubinelli_fbsde_2026}
that then $\mu_{\beta} = \operatorname{Law}(W+I)$.
Note that in this case we do not need to pass to the space $\overline{\mathcal{X}}$
to ensure existence of the minimizer.

Denoting $X_t = W_t + I_t$, we can rewrite~\eqref{eq:FBDSE} as a system.
For any $F\in C^2(\mathbb R_+ \times \mathbb R^\Lambda, \mathbb R^\Lambda)$,
we set $R_t = \lambda\beta \mathord{\mathbb E} [ \mathord{\mathopen: \sin (\beta W_{\infty} + \beta I_{\infty}[h]) \mathclose:} \mid \mathcal{F}_t ]
    - F_t (W_t+I_t(h))$.
As discussed in~\cite[Section~2.3]{gubinelli_fbsde_2026},
$R_t$ then solves the system
\begin{equation}
\begin{split}
R_t &= \mathord{\mathbb E} \left[ \int^{\infty}_t H_s (X_s) + \mathrm{D} F_s (X_s) J^2_s R_s \mathop{}\!\mathrm d s
    \,\middle|\, \mathcal{F}_t \right]\!,\\
H_t &\coloneqq \partial_t F_t + \frac{1}{2} \operatorname{Tr} (J^2_t \mathrm{D}^2 F_t)
    -\frac{1}{2} \mathrm{D} \langle F_t, J^2_t F_t \rangle_{L^2 (\Lambda)} .
\end{split}
\end{equation}

In our setting we will set
\begin{equation}
F_t (\varphi) = \lambda \beta e^{\frac{\beta^2 c_{t}}{2}} \sin (\beta \varphi),
\end{equation}
where $c_t = \mathord{\mathbb E} [W_{t}(0)^2] = \log(t \wedge \delta^2)/4\pi + O(1)$.
(This estimate follows by adapting~\cite[Appendix~A]{gubinelli_fbsde_2026} to the periodic setup.)
It can be checked that this satisfies
\begin{equation}
\partial_t F_t + \frac{1}{2} \text{Tr} (J^2_t \mathrm{D}^2 F_t) = 0, \qquad
F_{\infty} (\varphi) = \lambda \beta \mathord{\mathopen: \sin (\beta \varphi) \mathclose:}.
\end{equation}
In addition we have, with the notation $\lambda_{t}=\lambda e^{\frac{\beta^2 c_{t}}{2}}$, the identity
\begin{equation}
\langle F_t, J^2_t F_t \rangle_{L^2 (\Lambda)}
= \lambda^2_t \beta^2 \iint \sin (\beta \varphi) (x) C^{\ast 2}_t (x - y) \sin (\beta \varphi) (y) \mathop{}\!\mathrm d x\mathop{}\!\mathrm d y,
\end{equation}
where $C^{\ast 2}_t$ is as in Lemma~\ref{thm:heat kernel bounds}.
(For brevity we replace the sums over $\Lambda$ with integrals in convolutions.)
So taking the gradient we have
\begin{equation}
H_t = -\frac 1 2 \mathrm{D} \langle F_t, J^2_t F_t \rangle_{L^2 (\Lambda)}
= -\lambda^2_t \beta^3
    \iint \cos (\beta \varphi) (x) C^{\ast 2}_t (x - y) \sin (\beta \varphi) (y) \mathop{}\!\mathrm d x\mathop{}\!\mathrm d y.
\end{equation}
Observe that this implies the estimate
\begin{equation} \label{eq:estimate-bilinear}
\begin{split}
\|\mathrm{D} \langle F_t, J^2_t F_t \rangle_{L^2 (\Lambda)}\|_{L^{\infty}(\Lambda)}
&\leq \lambda^2_t \beta^3 \|C^{\ast 2}_t\|_{L^1}\\
&\leq \frac{1}{m^2} \beta^3 \lambda^2_t t^{-2}\\
&\leq \frac{1}{m^2} \lambda^2 \beta^3 \langle t \rangle^{-1 - \varepsilon},
\end{split}
\end{equation}
where $\varepsilon$ is taken small enough.
In particular $R$ satisfies the bound
\begin{equation}
\begin{split}
\|R_t\|_{L^{\infty}(\mathord{\mathbb P}, L^{\infty} (\Lambda))}
&\leq \frac{C \lambda^2}{m^2} \beta^3 + \lambda \beta^2
    \int_t^\infty \langle s \rangle^{\beta^2 / 4 \pi}
        \|J^2_s R_s\|_{L^{\infty}(\mathord{\mathbb P}, L^{\infty} (\Lambda))} \mathop{}\!\mathrm d s\\
&\leq \frac{C \lambda^2}{m^2} \beta^3 + \frac{\lambda \beta^2}{m^2}
    \int_t^\infty \langle s \rangle^{- 2 - \varepsilon + \beta^2 / 4 \pi}
        \|R_s\|_{L^{\infty}(\mathord{\mathbb P}, L^{\infty} (\Lambda))} \mathop{}\!\mathrm d s.
\end{split}
\end{equation}

\begin{equation}

&= -24\lambda\delta^3 \sum_{x\in\Lambda} u(x)K(x)(\psi^u(x))^2.\\
&= -24\lambda\delta^3 \sum_{x\in\Lambda} (W+W^{[3]}+K)(x)K(x)(\psi^u(x))^2\\
&\geq -24\lambda\delta^3 \sum_{x\in\Lambda} (W+W^{[3]})(x)K(x)(\psi^u(x))^2.
\end{split}
\end{equation}
We now apply the same reasoning as in the proof of Lemma~\ref{lem:lower-bound},
more precisely equations~\eqref{eq:estiamte-3d-KW-1},
\eqref{eq:estiamte-3d-KW-2}, and~\eqref{eq:estiamte-3d-KW-3},
as well as those inequalities with $W$ replaced by $W^{[3]}$ to get 
\begin{equation}\label{eq:tilt-pairing}
\begin{split}
&\mathrel{\phantom{=}} q \langle\nabla Q(u),K\rangle  \\ 
&\geq  -C_\eta (\lambda^{7/8}q m^{-1/2} (\|W\|_{\mathcal C^{-1/2-\varepsilon}}+\|W^{[3]}\|_{\mathcal C^{-1/2-\varepsilon}}))^{16/(1-8\varepsilon_5)}
- \eta \mathcal{E}(K)
- \eta \|\psi\|_{H^{1-\varepsilon}}^2\\
&\geq -\eta \mathcal{E}(K)- \mathcal{B}^2_{\eta}(W,\lambda,m)
    -C_\eta [\lambda^{7/8}q m^{-1/2} (\|W\|_{\mathcal C^{-1/2-\varepsilon}}+\|W^{[3]}\|_{\mathcal C^{-1/2-\varepsilon}})]^{16/(1-\varepsilon)},
\end{split}
\end{equation}
where in the last line we have applied Proposition~\ref{bound-3d-ground}.
We set
\begin{equation}
\mathcal{B}^4_{\eta}(W,\lambda,m,q)
\coloneqq C_\eta [\lambda^{7/8}\beta m^{-1/2} (\|W\|_{\mathcal C^{-1/2-\varepsilon}}
    +\|W^{[3]}\|_{\mathcal C^{-1/2-\varepsilon}})]^{16/(1-8\varepsilon)}.
\end{equation}

We now derive the equation satisfied by $K$, which  follows from the decomposition $K=u-W-W^{[3]}$. Writing
$\mathrm L_\delta=\partial_t+m^2-\operatorname{\mathrm{\Delta}}_\delta$ and
$\mathrm P_0 = \mathrm{Id} - \mathrm P_{\geq1}$, we obtain
\begin{equation}
\mathrm L_\delta K
    + 4\lambda\bigl(\mathord{\mathopen: u^3 \mathclose:}-\mathord{\mathopen: W^3 \mathclose:}\bigr)
    + 2\lambda^2 a_\delta u
    - q\nabla Q(u)
= \sqrt2\, \mathrm P_0\xi.
\label{eq:K-equation}
\end{equation}
Using Itô's formula we obtain
\begin{equation}
0 = \frac12 \frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t} \mathord{\mathbb E} \|K\|_{L^2}^2
= \mathord{\mathbb E}\langle F_K,K\rangle_{L^2} - \operatorname{Tr}_{L^2}(P_0),
\end{equation}
where
\begin{equation}
F_K = (m^2-\operatorname{\mathrm{\Delta}}_\delta)K
    + 4\lambda\bigl(\mathord{\mathopen: u^3 \mathclose:}-\mathord{\mathopen: W^3 \mathclose:}\bigr)
    + 2\lambda^2a_\delta u-q\nabla Q(u).
\end{equation}
Since $\mathrm P_0$ projects onto the normalized constant mode,
$\operatorname{Tr}_{L^2}(P_0)=1$.
Proposition~\ref{prop-gub-hov} and~\eqref{eq:tilt-pairing} therefore give
\begin{equation}
(1-\eta)\mathord{\mathbb E}\mathcal E(K)
\leq 1+\mathord{\mathbb E}\!\left[ \mathcal B^3(W,\lambda,m,q)
    + \mathcal B^2_{\eta}(W,\lambda,m)
    + \mathcal B^4_{\eta}(W,\lambda,m)\right].
\label{inequality-energy}
\end{equation}
Finally observe that if $\|\psi\|_{L^2}=1$,
we have $\|\psi\|^2_{H^{1-\varepsilon}}\geq 1$, so
\begin{equation}
q \left((1-\lambda/2) \|\psi\|^2_{H^{1-\varepsilon}}-1\right)
\geq -\frac{\lambda q}{2}.
\end{equation}
Thus we obtain, applying Lemma~\ref{lem:lower-bound},
\begin{equation}
\begin{split}
q\int Q(u)\,\mathop{}\!\mathrm d\nu^{qQ}(u)
&= -q\,\mathord{\mathbb E}\inf_{\|\psi\|_2=1}\mathcal A
       (W+W^{[3]}+K)(\psi)\\
&\leq C\, \mathord{\mathbb E}\!\left[1 +\lambda q/2 +\mathcal E(K)
    + \mathcal B^1(W,\lambda,m,q)\right],
\end{split}
\end{equation}
where the inequality is the negative of the lower bound in Lemma~\ref{lem:lower-bound}.
Using the energy estimate~\eqref{inequality-energy} and the asserted uniform moments of~$\mathcal B^i$,
this is the required \emph{upper} bound on $q\int Q\,d\nu^{qQ}$.
It can then be fed into Lemma~\ref{lem:hairer-steele} to bound $Z^{qQ}$, and obtain $\mathcal{R}$.

Finally by examining each $\mathcal B^i$, we observe that $\mathcal R$ indeed satisfies
\begin{equation}
\sup_{q, m, \lambda}  \mathcal{R}(q,m,\lambda) < \infty
\end{equation}
in the regime $\lambda\lesssim m^{3}$, $m\leq 1$, $q\lesssim m^{-2}$.

<!-- theorem-9: body_md -->
\label{thm:besov properties}
Let the ambient space be either~$\Lambda_\delta$ or~$\mathbb T^2$,
let $\rho_1$ and $\rho_2$ be weights,
let $p = 1/p_1 + 1/p_2$ and $r = 1/r_1 + 1/r_2$ be Hölder conjugate pairs in~$[1,\infty]$,
and let $s \in \mathbb R$.
Then the following hold:
\begin{enumerate}
\renewcommand{(\alph{enumi})}{(\alph{enumi})}
\item Multiplication: $\|fg\|_{B^s_{p,r}(\rho_1 \rho_2)}
\lesssim \|f\|_{B^s_{p_1, r}(\rho_1)} \|g\|_{B^t_{p_2, r}(\rho_2)}$
    for $t \geq s$ such that $s + t > 0$.
\item Duality: $\|\rho_1 \rho_2 fg\|_{L^1}
        \lesssim \|f\|_{B^{-s}_{p_1, r_1}(\rho_1)} \|g\|_{B^s_{p_2, r_2}(\rho_2)}$,
    where we set $p = r = 1$.
\item Interpolation: let $\theta \in (0,1)$ and
    \[
        s = \theta s_1 + (1-\theta) s_2, \quad
        \frac 1 p = \frac{\theta}{p_1} + \frac{1-\theta}{p_2}, \quad
        \frac 1 r = \frac{\theta}{r_1} + \frac{1-\theta}{r_2}.
    \]
    Then $\|f\|_{B^s_{p,r}(\rho_1^{\theta} \rho_2^{1-\theta})}
        \leq \|f\|_{B^{s_1}_{p_1, r_1}(\rho_1)}^\theta
            \|f\|_{B^{s_2}_{p_2, r_2}(\rho_2)}^{1-\theta}$.
\item Besov embedding: $\|f\|_{B^s_{p,r}(\rho_1)} \lesssim \|f\|_{B^t_{q,r}(\rho_1)}$
    where $q < p$ and $t \geq s + d(1/q - 1/p)$.
\item $L^p$ embedding: $\|f\|_{B^0_{p,\infty}(\rho_1)} \lesssim \|f\|_{L^p(\rho_1)}$.
\item Second parameter tradeoff:
    $\|f\|_{B^s_{p,r}(\rho_1)} \lesssim \|f\|_{B^{s+\varepsilon}_{p,\infty}(\rho_1)}$
    for all $1 \leq r \leq \infty$ and $\varepsilon > 0$.
\end{enumerate}
The implicit constants are uniform in $\delta$ (small enough).

\label{thm:besov paraproducts}
The product of two smooth functions can be decomposed into two paraproducts and resonant product
\[
fg = f \prec g + f \succ g + f \odot g
\]
as in~\cite[Section~2.8]{bahouri_fourier_2011}.

For any weights $\rho_1$, $\rho_2$,
and $r$, $p$, $p_1$, $p_2 \in [1, \infty]$ such that $1/p = 1/p_1 + 1/p_2$,

\[
\|f \prec g\|_{B^{\min(\alpha,0) + \beta}_{p,r}(\rho_1 \rho_2)}
\lesssim \|f\|_{B^\alpha_{p_1, \infty}(\rho_1)}
    \|g\|_{B^\beta_{p_2, r}(\rho_2)},
\quad \alpha, \beta \in \mathbb R,
\]
and the resonant product satisfies
\[
\|f \odot g\|_{B^{\alpha + \beta}_{p,r}(\rho_1 \rho_2)}
\lesssim \|f\|_{B^\alpha_{p_1, \infty}(\rho_1)}
    \|g\|_{B^\beta_{p_2, r}(\rho_2)},
\quad \alpha + \beta > 0.
\]

<!-- theorem-12: body_md -->
\label{wick-power-estimate}
Let $\Lambda_\delta \subset \mathbb R^2$,
and let $W$ be distributed according to the Gaussian free field on~$\Lambda_\delta$
with mass~$m^2$.
Then for $k \in \mathbb N$ and $p < \infty$ we have
\[
\mathord{\mathbb E} \|\mathord{\mathopen: W^k \mathclose:}\|_{\mathcal C^{-\kappa}(\rho)}^p \lesssim 1 + |\log(m)|^{pk/2}.
\]
Here we assume that $\rho$ is sufficiently integrable over $\Lambda_\delta$.
<!-- end theorem-12 -->
<!-- proof-31: body_md -->
We first use Theorem~\ref{thm:besov properties}
to go from $\mathcal C^{-\kappa}$ to $B_{p,p}^{-\kappa/2}$ for $p$ large enough.
We then use hypercontractivity, translation invariance, and Wick formula
to estimate
\begin{equation}
\begin{split}
\mathord{\mathbb E} \|\mathord{\mathopen: W^k \mathclose:}\|_{B_{p,p}^{-\kappa}(\rho)}^p
&\lesssim \sum_{j \geq -1} 2^{-pj\kappa} [\mathord{\mathbb E} (\Delta_j \mathord{\mathopen: W^k \mathclose:}(0))^2]^{p/2} \int \rho(x)^p \mathop{}\!\mathrm d x\\
&\lesssim \sum_{j \geq -1} 2^{-pj\kappa} \int K_j(-x) K_j(-y) \mathord{\mathbb E} [\mathord{\mathopen: W^k \mathclose:}(x) \, \mathord{\mathopen: W^k \mathclose:}(y)]^{p/2} \mathop{}\!\mathrm d x \mathop{}\!\mathrm d y\\
&\lesssim \sum_{j \geq -1} 2^{-pj\kappa} \int K_j(-x) K_j(-y) G(x,y)^{pk/2} \mathop{}\!\mathrm d x \mathop{}\!\mathrm d y,
\end{split}
\end{equation}
where $K_j$ are the kernels of the Littlewood--Paley blocks $\Delta_j$,
and $G$ is the Green's function of the free field.

The Green's function is of order $\log(m |x-y|)$ near the diagonal
and rapidly decaying away from it~\cite[Chapter~7]{glimm_quantum_1981}.
On the other hand, the kernels $K_j$ are uniformly bounded in $L^1$ norm
and of order $2^j$ in $L^\infty$ norm.
Therefore we can interpolate
\begin{equation}
\|K_j\|_{L^{q/(q-1)}} \lesssim 2^{j/q}.
\end{equation}
These considerations imply that
\begin{equation}
\mathord{\mathbb E} \|\mathord{\mathopen: W^k \mathclose:}\|_{B_{p,p}^{-\kappa}(\rho)}^p
\lesssim \sum_{j \geq -1} 2^{-pj\kappa} (|\log(m)|^{pk/2} + 2^{j/q}).
\end{equation}
By choosing $q$ large enough, we get the claim.

<!-- proof-32: body_md -->
We express $W = W_{\infty} = \int^{\infty}_0 J_t \mathop{}\!\mathrm d B_t$,
where $J_t$ is as in Section~\ref{sec:sine-gordon measure}
and $B_t$ is an $L^2 (\Lambda)$ cylindrical Brownian motion.
It follows from the Itô formula that
$\mathord{\mathopen: \sin (\beta W_\infty) \mathclose:} = \beta \int_0^\infty \mathord{\mathopen: \cos (\beta W_t) \mathclose:} \mathop{}\!\mathrm d W_t$.
Its quadratic variation is given by
\begin{equation}
\begin{split}
&\mathrel{\phantom{=}}
\beta^2 \left\langle \int^{\infty}_0 \mathord{\mathopen: \cos (\beta W_t) \mathclose:} (x) \mathop{}\!\mathrm d W_t (x),
    \int^{\infty}_0 \mathord{\mathopen: \cos (\beta W_t) \mathclose:} (y) \mathop{}\!\mathrm d W_t (y)
\right\rangle\\
& = \beta^2 \int_0^\infty C^{\ast 2}_t (x - y) \mathord{\mathopen: \cos (\beta W_t) (x) \mathclose:}
    \mathord{\mathopen: \cos (\beta W_t) (y) \mathclose:} \mathop{}\!\mathrm d t.
\end{split}
\end{equation}
Here $C^{\ast 2}_t$ is the convolution kernel of $J_t^2$,
as in Section~\ref{sec:sine-gordon measure}.

Using this we can deduce that the Littlewood--Paley blocks satisfy
\begin{equation}
\begin{split}
&\mathrel{\phantom{=}} \langle (\Delta_j \mathord{\mathopen: \sin (\beta W) \mathclose:}) (z) \rangle\\
&= \beta^2 \int^{\infty}_0 \!\!\! \iint C^{\ast 2}_t(x - y) K_j (x - z) K_j (y - z)
    \mathord{\mathopen: \cos(\beta W_t) (x) \mathclose:} \mathord{\mathopen: \cos (\beta W_t) (y) \mathclose:} \mathop{}\!\mathrm d x \mathop{}\!\mathrm d y \mathop{}\!\mathrm d t\\
&\leq \beta^2 \int^{\infty}_0 \!\!\! \iint C^{\ast 2}_t(x - y)
    |K_j (x- z) K_j (y - z)| \langle t \rangle^{\beta^2 / 2 \pi} \mathop{}\!\mathrm d x \mathop{}\!\mathrm d y \mathop{}\!\mathrm d t\\
&= \beta^2 \int^{\infty}_0 \!\!\! \iint C^{\ast 2}_t(x - y)
    |K_j (x) K_j (y)| \langle t \rangle^{\beta^2 / 2 \pi} \mathop{}\!\mathrm d x \mathop{}\!\mathrm d y \mathop{}\!\mathrm d t,
\end{split}
\end{equation}
where $K_j$ is again the convolution kernel of the block $\Delta_j$.

Now it is not hard to show that (see~\cite[Appendix~A]{bauerschmidt_logsobolev_2021})
\begin{equation}
\begin{split}
\int_0^\infty C^{\ast 2}_t(x - y) \langle t \rangle^{\beta^2 / 2 \pi} \mathop{}\!\mathrm d t
&\lesssim \int_0^\infty e^{- \frac{|x - y|^2}{2 t} - m t} \langle t \rangle^{\beta^2 / 2 \pi} \mathop{}\!\mathrm d t\\
&\lesssim \frac{1}{|x-y|^{\beta^2 / 2 \pi}} + m^{\beta^2 / 2 \pi}.
\end{split}
\end{equation}
Now for $2 / q > \beta^2 / 4 \pi$ we can estimate
\begin{equation}
\begin{split}
&\mathrel{\phantom{=}} \iint
    \left( \frac{1}{| x - y |^{\beta^2 / 2 \pi}} + m^{\beta^2 / 2\pi} \right)
    |K_j (x) K_j (y)| \langle t \rangle^{\beta^2 / 2 \pi} \mathop{}\!\mathrm d x \mathop{}\!\mathrm d y\\
&\lesssim \|K_j\|^2_{L^q} + m^{-\beta^2 / 2 \pi} \|K_j\|^2_{L^1}.
\end{split}
\end{equation}
From this we can conclude by applying the Burkholder--Davis--Gundy inequality:
for $\kappa > \beta^2 / 4 \pi$ and $p$ large enough, we have
\begin{equation}
\begin{split}
&\mathrel{\phantom{=}} \mathord{\mathbb E} \|\mathord{\mathopen: \sin (\beta W) \mathclose:}\|^p_{\mathcal C^{-\kappa}(\rho)}\\
&\lesssim \mathord{\mathbb E} \|\mathord{\mathopen: \sin (\beta W) \mathclose:}\|^p_{B_{p, p}^{- \kappa + \varepsilon / 2}(\rho)}\\
&\leq \sum 2^{- j p (\kappa + \varepsilon / 2)}
    \mathord{\mathbb E} \|\Delta_j \mathord{\mathopen: \sin (\beta W) \mathclose:}\|_{L^p (\rho)}^p\\
&\leq \sum 2^{- j p (\kappa + \varepsilon / 2)}
    \mathord{\mathbb E} |\Delta_j \mathord{\mathopen: \sin (\beta W) \mathclose:} (0) |^p\\
&\lesssim \sum 2^{- j p (\kappa + \varepsilon / 2)}
    (\mathord{\mathbb E} \langle \Delta_j \mathord{\mathopen: \sin (\beta W) \mathclose:} (0) \rangle)^{p / 2}\\
&\lesssim \sum 2^{- j p (\kappa + \varepsilon / 2)}
    (\|K_j\|^p_{L^q} + m^{-p\beta^2 / 4 \pi} \|K_j\|^p_{L^1})\\
&\lesssim (m \wedge 1)^{-p \beta^2 / 4 \pi}.
\end{split}
\end{equation}
The proof for the $\cos$ is analogous.
