<!-- theorem-1: body_md -->
\label{thm:main}
Let $B$ be a standard one-dimensional Brownian motion. Then $\Gamma_B$ is almost surely not conformally removable.

<!-- theorem-2: body_md -->
\label{thm:construction}
Almost surely, there exist a nonzero finite positive Radon measure $\mu$ supported on $\Gamma_B$ and a continuous function $f:\mathbb C \to \mathbb C$ with $\bar\partial f=\mu$ such that, for every sufficiently small $\epsilon>0$, the map $z \mapsto z+\epsilon f(z)$ extends to a non-M\"obius homeomorphism of $\widehat{\mathbb C}$ that is conformal on $\widehat{\mathbb C} \setminus \operatorname{supp} \mu$.

<!-- proposition-1: body_md -->
\label{prop:criterion}
Let $f:\mathbb C \to \mathbb C$ be continuous, and suppose that, in the sense of distributions,
\begin{equation}\label{eq:criterion-derivatives}
    \bar\partial f=\nu, \qquad \partial f=\nu+g \mathop{}\!\mathrm{d} A,
\end{equation}
where $\nu$ is a nonzero finite positive measure with compact support, $g \in L_{\rm loc}^1(\mathbb C)$ and $\operatorname{Re} g \in L^\infty(\mathbb C)$. For any $\epsilon>0$ such that $\epsilon\|\operatorname{Re} g\|_{L^\infty(\mathbb C)}<1$, write $c_0:=1-\epsilon\|\operatorname{Re} g\|_{L^\infty(\mathbb C)}>0$.
Then $F(z):=z+\epsilon f(z)$ is a homeomorphism of $\mathbb C$ onto itself and satisfies
\begin{equation}\label{eq:strong-monotonicity}
    (F(z)-F(w)) \cdot (z-w) \ge c_0|z-w|^2 \qquad \mbox{for } z,w\in\mathbb C.
\end{equation}
Moreover, $F$ extends to a homeomorphism of $\widehat{\mathbb C}$, is conformal at every point of
$\widehat{\mathbb C}\setminus\operatorname{supp}\nu$, and is not a
M\"obius transformation.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Write $f=u+iv$ and $g=\alpha+i\beta$, where $\alpha=\operatorname{Re} g$. By the identities for distributional derivatives~\eqref{eq:distributional-derivatives}, the conditions~\eqref{eq:criterion-derivatives} are equivalent to
\[ D_xu=2\nu+\alpha \mathop{}\!\mathrm{d} A, \qquad D_yu=-\beta \mathop{}\!\mathrm{d} A, \qquad D_xv=\beta \mathop{}\!\mathrm{d} A, \qquad D_yv=\alpha \mathop{}\!\mathrm{d} A. \]
Equivalently, the distributional Jacobian of $f$ is the matrix-valued Radon measure
\begin{equation}\label{eq:matrix-Df}
    Df=
    \begin{pmatrix}
        2\nu+\alpha \mathop{}\!\mathrm{d} A & -\beta \mathop{}\!\mathrm{d} A \\
        \beta \mathop{}\!\mathrm{d} A & \alpha \mathop{}\!\mathrm{d} A
    \end{pmatrix}.
\end{equation}
Let $e_1=(1,0)^T$. Since $DF=I \mathop{}\!\mathrm{d} A+\epsilon Df$, taking symmetric parts in~\eqref{eq:matrix-Df} gives
\begin{equation}\label{eq:sym-lower}
    \operatorname{sym} DF=(1+\epsilon\alpha)I \mathop{}\!\mathrm{d} A +2\epsilon\nu \, e_1\otimes e_1 \ge c_0 I \mathop{}\!\mathrm{d} A,
\end{equation}
where we used $1+\epsilon \alpha \ge c_0$ almost everywhere and the positivity of $\nu$.

Choose a nonnegative function $\eta\in C_c^\infty(\mathbb C)$ satisfying $\operatorname{supp} \eta \subset B(0,1)$ and $\int_{\mathbb C} \eta \mathop{}\!\mathrm{d} A=1$, and let $\eta_\delta(z)=\delta^{-2}\eta(z/\delta)$ be the standard mollifier. Define $F_\delta(z):=z+\epsilon(f*\eta_\delta)(z)$, then we have $DF_\delta=(DF)*\eta_\delta$. Convolution with nonnegative $\eta_\delta$ preserves the inequality~\eqref{eq:sym-lower}, which implies that $\operatorname{sym} DF_\delta(z)\ge c_0 I$ for all $z \in \mathbb C$.

For $z,w \in \mathbb C$, write $v=z-w$, then
\[ (F_\delta(z)-F_\delta(w)) \cdot v=\int_0^1 v^T DF_\delta(w+tv) v \mathop{}\!\mathrm{d} t = \int_0^1 v^T (\operatorname{sym} DF_\delta)(w+tv)v \mathop{}\!\mathrm{d} t \ge c_0|v|^2. \]
Because $f$ is continuous, $F_\delta \to F$ locally uniformly. Letting $\delta\downarrow0$ therefore gives
\eqref{eq:strong-monotonicity}.

By the Cauchy--Schwarz inequality,~\eqref{eq:strong-monotonicity} implies
\begin{equation}\label{eq:colipschitz}
    |F(z)-F(w)|\ge c_0|z-w| \qquad \mbox{for } z,w\in\mathbb C,
\end{equation}
so $F$ is injective, and its inverse on its image is continuous.

By invariance of domain, $F(\mathbb C)$ is open. Note that $F(\mathbb C)$ is also closed: if $F(z_j)$ converges in $\mathbb C$, then
\eqref{eq:colipschitz} shows that $(z_j)$ is a Cauchy sequence.
Writing $z=\lim_jz_j$ and using the continuity of $F$, we obtain $\lim_j F(z_j)=F(z) \in F(\mathbb C)$. Thus $F(\mathbb C)$ is a nonempty subset of the connected space $\mathbb C$ that is both open and closed, and hence $F(\mathbb C)=\mathbb C$. This proves that $F$ is a homeomorphism of $\mathbb C$ onto itself.

Moreover, taking $w=0$ in~\eqref{eq:colipschitz} gives $|F(z)| \ge |F(z)-F(0)|-|F(0)| \ge c_0 |z|-|F(0)|$. Hence $|F(z)| \to \infty$ as $|z| \to \infty$. Therefore, setting $F(\infty)=\infty$ extends $F$ to a homeomorphism of $\widehat{\mathbb C}$.

Since $\bar\partial F=\epsilon \nu$ vanishes on $\mathbb C \setminus \operatorname{supp} \nu$, by Weyl's lemma~\cite[Lemma A.6.10]{AIM09}, it follows that $F$ is holomorphic on $\mathbb C \setminus \operatorname{supp} \nu$. Moreover, $F$ is injective, so its complex derivative does not vanish there. Hence $F$ is conformal on $\mathbb C \setminus \operatorname{supp} \nu$. At infinity, in the coordinate $\zeta=1/z$, the function $G(\zeta):=\frac{1}{F(1/\zeta)}$ is holomorphic in a punctured neighborhood of 0 and satisfies $G(\zeta) \to 0$ as $\zeta \to 0$. Therefore, $G$ extends holomorphically to 0. The extension is injective near 0, and hence $G'(0) \neq 0$. Therefore $F$ is also conformal at infinity.

Finally, if $F$ were a M\"obius map, then, since $F$ fixes $\infty$, its restriction to $\mathbb C$ would be holomorphic. But this contradicts $\bar\partial F=\epsilon\nu \neq 0$. This completes the proof.

<!-- proposition-2: body_md -->
\label{prop:criterion+}
Let $q>2$, and let $v \in C_c(\mathbb C)$ be real-valued. Suppose that, in the sense of distributions,
\begin{equation}\label{eq:derivative-v}
    D_x v=\mu+a\mathop{}\!\mathrm{d} A, \qquad D_y v=b\mathop{}\!\mathrm{d} A,
\end{equation}
where $\mu$ is a nonzero finite positive Radon measure with compact support, and the functions $a \in C_c^\infty(\mathbb C)$ and $b \in L_c^q(\mathbb C)$ are real-valued. Suppose further that $\mathcal R_x\mathcal R_y b \in L^\infty(\mathbb C)$. Then there exists a continuous function $f:\mathbb C \to \mathbb C$ with $\bar \partial f = \mu$ such that, for every sufficiently small $\epsilon>0$, the map $z\mapsto z+\epsilon f(z)$ extends to a non-M\"obius homeomorphism of $\widehat{\mathbb C}$ that is conformal on $\widehat{\mathbb C} \setminus \operatorname{supp}\mu$.
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
Define $\rho=a+ib$ and $f=2v-\mathcal C(\rho \mathop{}\!\mathrm{d} A)$. Since $\rho \in L_c^q(\mathbb C)$ and $q>2$,~\cite[Theorem 4.3.13]{AIM09} implies that $\mathcal C (\rho \mathop{}\!\mathrm{d} A) \in C^{0,\alpha}(\mathbb C)$, where $\alpha=1-2/q>0$. In particular, $\mathcal C(\rho \mathop{}\!\mathrm{d} A)$ is continuous, so $f$ is continuous.

    The Cauchy-transform identities from Section~\ref{subsec:transforms-prelim} give $\bar\partial f=\mu$ and $\partial f=\mu+g \mathop{}\!\mathrm{d} A$, where, by~\eqref{eq:beurling-riesz},
    \[ g=(a-ib)-\mathcal{B}(a+ib)=(a-ib)-(\mathcal R_y^2-\mathcal R_x^2+2i \mathcal R_x \mathcal R_y)(a+ib). \]
    The $L^q$-boundedness of $\mathcal B$ implies $g \in L^q(\mathbb C) \subset L_{\rm loc}^1(\mathbb C)$. Moreover, since $a$ and $b$ are real-valued, taking the real part gives $\operatorname{Re} g = (\operatorname{id}-\mathcal R_y^2+\mathcal R_x^2)a+ 2 \mathcal R_x \mathcal R_y b$. The first term is bounded since $a \in C_c^\infty(\mathbb C)$, and the second term is bounded by assumption. Thus $\operatorname{Re} g \in L^\infty(\mathbb C)$, and the conclusion now follows from Proposition~\ref{prop:criterion}.

<!-- lemma-1: body_md -->
\label{lem:row}
For every $\eta \in (0,1)$ there exist a rational number $r \in (0,1/8)$, $c \in (0,1)$, and a positive integer $K_0$, which depend only on $\eta$, such that for every integer $K \ge K_0$ with $rK \in \mathbb{N}$, the following holds with $h=H/K$. For an $H$-upbox $P=[T_{H,k},T_{H,k+1}] \times [b,b+H]$, the event that
\[ P \mbox{ has at least } cK \mbox{ many } K\mbox{-descendants in row } j \mbox{ for each } j \in \mathcal J_K:=\{-3rK,-3rK+1,\ldots,K-rK-1\} \]
occurs with conditional probability at least $1-\eta$.
<!-- end lemma-1 -->
<!-- proof-3: body_md -->
This is essentially~\cite[Lemma~2.2]{DM26}. More precisely, it is shown that for every $\eta \in (0,1)$ there exist a rational number $r \in (0,1/8)$, $c \in (0,1)$, and a positive integer $K_0$, which depend only on $\eta$, such that for every integer $K \ge K_0$ with $rK\in\mathbb{N}$, the following event occurs with probability at least $1-\eta$: the stopped Brownian motion $(B_t)_{t \in [0,\tau_1]}$ conditional on $B_{\tau_1}=1$ makes at least $cK$ upcrossings of every interval of the form
\[ [j/K,(j+1)/K], \quad \mbox{for } j \in \mathcal J_K=\{-3rK,-3rK+1,\ldots,K-rK-1\}. \]

Note that each upcrossing of the interval $[b+jh,b+(j+1)h]$ during the time interval $I(P)$ naturally induces (immediately before this upcrossing ends) an $h$-upbox whose height range is $[b+jh,b+(j+1)h]$, and different upcrossings correspond to different upboxes. The desired claim then follows from translation and Brownian scaling.

<!-- lemma-2: body_md -->
\label{lem:conditional-independence}
Fix an integer $K \ge2$ and an $H$-upbox $P=[T_{H,k},T_{H,k+1}] \times [b,b+H]$, and set $h=H/K$. There is an index $r_0$ such that $T_{h,r_0}=T_{H,k}$, and we write $\sigma_\ell=T_{h,r_0+\ell}$ for $\ell \ge 0$. Let $N_P$ be the first index for which $\sigma_{N_P}=T_{H,k+1}$. Write $\epsilon_\ell=h^{-1}(B_{\sigma_{\ell+1}}-B_{\sigma_\ell}) \in \{-1,1\}$.

Let $\mathcal S_P$ be the sigma-algebra generated by $N_P$ and $\epsilon_0,\ldots,\epsilon_{N_P-1}$. Conditional on $\mathcal S_P$, the rescaled stopped Brownian path segments together with their durations,
\[ \left(h^{-2}(\sigma_{\ell+1}-\sigma_\ell),(h^{-1}(B_{\sigma_\ell+h^2t}-B_{\sigma_\ell}))_{0\le t\le h^{-2}(\sigma_{\ell+1}-\sigma_\ell)}\right), \quad 0\le\ell<N_P,
\]
are independent. Moreover, each rescaled Brownian path segment has law $\mathbb P^+$ or its vertical reflection depending on its sign.
<!-- end lemma-2 -->
<!-- proof-4: body_md -->
We first work under the unconditioned Brownian law after $T_{H,k}$, and apply the strong Markov property successively at the stopping times $\sigma_\ell$. Before their signs are revealed, the stopped increments are independent Brownian paths run until first exit from $(-h,h)$. On the event $\{N_P=n\}$, these $n$ stopped Brownian path segments are conditionally independent given the sign sequence $(\epsilon_0,\ldots,\epsilon_{n-1})$. Since the index $N_P$ is determined by the sign sequence, conditional independent still holds given $\mathcal S_P$. After translation and Brownian scaling, conditioning an increment on its exit sign gives $\mathbb P^+$ for a positive sign and the reflected law for a negative sign.

<!-- lemma-3: body_md -->
\label{lem:uniform-count}
Fix $H_0>0$ and $ K \ge 2$, and set $H_n=H_0K^{-n}$. Almost surely, there exists a finite random constant $C_B<\infty$ which depends only on $H_0$, $K$, and $(B_t)_{t \ge 0}$, such that, for every $n$, every integer $\ell$, and every time interval $W \subseteq [0,1]$ of length $R \ge H_n^2$, we have
\begin{equation}\label{eq:uniform-count}
    N_{H_n,\ell}^+(W) \le C_B\sqrt{n+1} \frac{\sqrt{R}}{H_n}.
\end{equation}
<!-- end lemma-3 -->
<!-- proof-5: body_md -->
The proof is divided into three steps. First, we obtain a Gaussian tail bound for a fixed triple $(H,\ell,W)$. Next, a union bound and Borel--Cantelli give the desired estimate simultaneously over the relevant height ranges and dyadic time intervals at all sufficiently small scales. Finally, we pass from dyadic intervals to arbitrary intervals and absorb the finitely many remaining scales into the random constant.

\textit{Step 1. Tail bound for a fixed interval.} Suppose $W$ has length $R$, and $N^+_{H,\ell}(W) \ge k \ge 4$ for some $H$ and $\ell$. After ignoring the at most two crossing intervals containing the endpoints of $W$, there are at least $k-2$ upsteps inside $W$. Between any two consecutive upsteps, there must be a downcrossing of the interval $[\ell H,(\ell+1)H]$. Hence, the Brownian motion makes at least $k-3$ disjoint downcrossings during $W$, and these downcrossings are independent by the strong Markov property.

Recall that for a Brownian motion $(B_t)_{t \ge 0}$, the one-sided exit time $\sigma_H:=\inf\{t>0: B_t=-H\}$ has Laplace transform $\mathbb E e^{-\lambda \sigma_H}=e^{-H\sqrt{2\lambda}}$ for $\lambda>0$ (see~\cite[Exercise 2.18]{MP10}). Let $\sigma_H^1,\ldots,\sigma_H^{k-3}$ denote i.i.d.\ copies of $\sigma_H$. Then, by Markov's inequality,
\[ \mathbb P[N^+_{H,\ell}(W) \ge k] \le \mathbb P[\sigma_H^1+\cdots+\sigma_H^{k-3} \le R] \le e^{\lambda R} \mathbb E[e^{-\lambda(\sigma_H^1+\cdots+\sigma_H^{k-3})}]=\exp(\lambda R-(k-3)H\sqrt{2\lambda}). \]
Optimizing over $\lambda>0$ gives, for $k \ge 4$,
\begin{equation}\label{eq:count-tail}
    \mathbb P[N^+_{H,\ell}(W)\ge k] \le \exp\left(-\frac{(k-3)^2H^2}{2R}\right) \le \exp\left(-\frac{k^2H^2}{32R}\right).
\end{equation}

\textit{Step 2. Proving~\eqref{eq:uniform-count} for dyadic intervals.} First, by the reflection principle,
\[ \sum_{n=1}^\infty \mathbb P\left[\sup_{0 \le t \le 5} |B_t|>n \right] \le 4\sum_{n=1}^\infty \mathbb P[B_5>n]<\infty. \]
It follows from the Borel--Cantelli lemma that, almost surely, there exists an integer $N_0 \ge 2$ such that $\sup_{0 \le t \le 5} |B_t| \le N_0$. Increasing $N_0$ if necessary, we may assume that $H_n=H_0 K^{-n} \le 1$ for all $n \ge N_0$. We now fix $n \ge N_0$. Then
\begin{equation}\label{eq:event-condition}
    \mbox{any } H_n\mbox{-upstep that intersects } [0,5] \mbox{ must have its height range contained in } [-n-2,n+2].
\end{equation}
For each dyadic number $D \in [H_n^2,2]$, consider the following collection of intervals of length $2D$,
\[ \mathcal V_D:=\{[0,2D], \ [D,3D], \ [2D,4D],\ldots\}. \]
Let $\mathsf A \ge 4$ be a large constant to be determined later. For each $V \in \mathcal V_D$, by~\eqref{eq:count-tail} we have
\begin{equation}\label{eq:count-tail-2}
    \mathbb P \left[N^+_{H_n,\ell}(V) \ge \mathsf A \sqrt{n+1} \tfrac{\sqrt{2D}}{H_n} \right] \le \exp\left(-\frac{\mathsf A^2 (n+1)}{32}\right).
\end{equation}

For fixed $n \in \mathbb N$, we now apply a union bound using~\eqref{eq:count-tail-2} over all pairs $(V,\ell)$, where
\begin{itemize}
    \item $V \in \mathcal V_D$ for some dyadic $D \in [H_n^2,2]$, such that $V$ intersects $[0,1]$, and
    \item $\ell$ is an integer satisfying $|\ell| \le 2n/H_n$. (For larger $|\ell|$, we have $N_{H_n,\ell}^+(V)=0$ by~\eqref{eq:event-condition}.)
\end{itemize}
The number of dyadic levels $D$ is at most $\log_2(2/H_n^2)+1$. For fixed dyadic $D \ge H_n^2$, the number of $V \in \mathcal V_D$ that intersect $[0,1]$ is at most $1/D+1 \le 1/H_n^2+1$. Therefore, the number of pairs $(V,\ell)$ is at most
\[ (\log_2(2/H_n^2)+1) \times (1/H_n^2+1) \times (4n/H_n+1), \]
which is $O(n^2 H_n^{-3})=O(n^2 K^{3n})$ with the implicit constant depending only on $H_0$ and $K$. Hence, by choosing $\mathsf A \ge 4$ sufficiently large, the sum of~\eqref{eq:count-tail-2} over all pairs $(V,\ell)$ decays exponentially in $n$ and is therefore summable. By the Borel--Cantelli lemma, almost surely there exists an integer $N_1 \ge N_0$ such that for every $n \ge N_1$, none of the events in~\eqref{eq:count-tail-2}, indexed by the pairs $(V,\ell)$ considered above, occurs.

\textit{Step 3. Concluding the proof.} Fix $n \ge N_1$. For each dyadic $D \in [H_n^2,2]$ and every interval $W \subseteq [0,1]$ with length $R \in [D/2,D]$, it is clear that $W$ is contained in $V \cap [0,1]$ for some $V \in \mathcal V_D$. It follows that
\[ N_{H_n,\ell}^+(W) \le N_{H_n,\ell}^+(V)< \mathsf A \sqrt{n+1} \frac{\sqrt{2D}}{H_n} \le 2 \mathsf A \sqrt{n+1} \frac{\sqrt{R}}{H_n}. \]

Finally, at the finitely many remaining levels $n<N_1$, since $H_n$ is bounded away from 0, the number of $H_n$-upsteps that intersect $[0,1]$ is finite almost surely. Thus, we can choose a finite random constant $C_B \ge 2\mathsf A$ for which the asserted estimate holds at every level. This concludes the proof.

<!-- proposition-3: body_md -->
\label{prop:construct-tree}
Fix $q>2$ and $\eta \in (0,1/40)$. There exist a rational number $r \in (0,1/8)$, $s \in (0,1)$, $\mathsf C>0$, and a positive integer $K$ with $rK \in \mathbb N$, such that $m=\lfloor sK \rfloor>K^{1-1/q}$, and the following event occurs almost surely: there exists an infinite $m(1+2r)K$-regular tree of upboxes such that

\begin{enumerate}[(a)]
\item The root of the tree is an $H_0$-upbox $[t_1,t_2] \times [B_{t_1},B_{t_2}]$ for some interval $[t_1,t_2]=[T_{H_0,k},T_{H_0,k+1}] \subseteq [0,1]$ and a rational $H_0 \in (0,1)$; the $n$-th level of the tree consists of $H_n$-upboxes, with $H_n=H_0K^{-n}$.
\item Every $H_n$-upbox in the tree has exactly $m$ many $K$-descendants in each row $j \in \mathcal J_K$.
\item Every $H_n$-upbox $P$ in the tree satisfies $\mathrm{len}(I(P))\le \mathsf C H_n^2$.
\end{enumerate}
<!-- end proposition-3 -->
<!-- proof-6: body_md -->
The proof proceeds in two steps. First, we use Lemma~\ref{lem:row} to show that a fixed upbox has a uniform positive probability of inducing an infinite regular tree. We then upgrade this positive-probability statement to an almost-sure one via the independence of Brownian motion.

\textbf{Step 1. Estimate for a single upbox.} As mentioned earlier, when considering an upbox $P=[T_{H,k},T_{H,k+1}] \times [B_{T_{H,k}},B_{T_{H,k+1}}]$, we work under the conditional law of $(B_t)_{t \in [T_{H,k},T_{H,k+1}]}$. Let $G(P)$ be the event considered in Lemma~\ref{lem:row}. Then $G(P)$ is measurable with respect to $\mathcal S_P$ and has conditional probability at least $1-\eta$.

For an upbox $P$, we say that $P$ is \emph{0-good} if $P$ is fast. For each $N \ge 0$, we say that $P$ is \emph{$(N+1)$-good} if the following conditions hold simultaneously:
\begin{itemize}
    \item $P$ is fast;
    \item $G(P)$ occurs, and moreover, for every $j \in \mathcal J_K$, there are at least $m$ indices $\ell \in \{1,\ldots,M\}$ for which $Q_{j,\ell}$ is $N$-good. Here, $Q_{j,1},\ldots,Q_{j,M}$ are the first $M$ elements, in chronological order according to the time interval, among the $K$-descendants of $P$ in row $j$.
\end{itemize}
A simple induction shows that if $P$ is $(N+1)$-good, then it is also $N$-good. We say that $P$ is $\infty$-good if $P$ is $N$-good for every $N \ge 0$. Note that if $P$ is $\infty$-good, then there exists an infinite regular tree rooted at $P$ that satisfies conditions (b) and (c).

By translation invariance and Brownian scaling, the conditional probability of $P$ being $N$-good does not depend on the choice of $P$. We denote this common value by $t_N=\mathbb P^+ [\mbox{the box } [0,\tau_1] \times [0,1] \mbox{ is } N\mbox{-good}]$. Lemma~\ref{lem:conditional-independence} implies that, conditional on $\mathcal S_P$, the events that $Q_{j,1},\ldots,Q_{j,M}$ are $N$-good are independent, for each fixed $j$, and each has probability $t_N$. Hence, by a union bound,
\begin{equation}\label{eq:recursion}
    t_{N+1} \ge 1-\epsilon_0-\eta -(K+2rK) \cdot \mathbb P[\mathsf{Bin}(M,t_N)<m].
\end{equation}
By the choice of $\epsilon_0$ in~\eqref{eq:parameter-C}, we have $t_0=1-\epsilon_0>0.9$. Moreover, since $\eta<1/40$ and $K$ is chosen as in~\eqref{eq:parameter-K}, an induction using~\eqref{eq:recursion} shows that $t_N \ge 0.9$ for every $N \ge 0$. \[ A_k^a:=\{B_{T_{a,k+1}}-B_{T_{a,k}}=a \ \mbox{ and } \ [T_{a,k},T_{a,k+1}] \times [B_{T_{a,k}},B_{T_{a,k+1}}] \mbox{ is } \infty\mbox{-good} \} \]

\[ \mathbb P[E^c] \le \mathbb P[T_{a,N}>1]+\mathbb P[\cap_{k=0}^{N-1}(A_k^a)^c] \le Na^2+(1-\mathsf p)^N<\delta. \]
Since $\delta>0$ was arbitrary, it follows that $\mathbb P[E]=1$. <!-- lemma-4: body_md -->
\label{lem:conservation}
    The following properties hold.
    \begin{enumerate}[(a)]
        \item Let $n \ge 1$, and let $Q \in \mathscr T_n$ have height range $[\ell H_n,(\ell+1)H_n]$, then $\operatorname{supp} \theta_Q \subseteq \ell H_n+H_n[-3r,1-2r]$.

<!-- lemma-5: body_md -->
\label{lem:theta-derivatives}
For every integer $k \ge 0$, there is a constant $C_k<\infty$ such that, for every $n \ge 0$, every upbox $P \in \mathscr T_n$, and every $j \in \mathcal J_K$,
\begin{equation}\label{eq:theta-phi-derivative-bound}
    \|(\theta_P \phi_{P,j})^{(k)}\|_{L^\infty} \le C_k m^{-n} H_{n+1}^{-k}.
\end{equation}
<!-- end lemma-5 -->
<!-- proof-8: body_md -->
We first show that for some constant $C_k'<\infty$,
\begin{equation}\label{eq:theta-derivative-bound}
    \|\theta_P^{(k)}\|_{L^\infty} \le C_k' m^{-n} H_n^{-k}.
\end{equation}

Let $P_0=\mathsf R, P_1,\ldots,P_{n-1}, P_n=P$ be the chain of ancestors of $P$, and suppose that $P_{i+1}$ is a descendant of $P_i$ in row $j_i$. Iterating~\eqref{eq:theta-child}, we obtain
\[ \theta_P(y)=m^{-n} \cdot \omega(y) \prod_{i=0}^{n-1} \phi_{P_i,j_i}(y). \]
For every integer $a \ge 0$, the definition~\eqref{eq:def-phi} of $\phi_{P_i,j_i}$ gives $\| \phi_{P_i,j_i}^{(a)} \|_{L^\infty} \le c_a H_{i+1}^{-a}$, where $c_a$ depends only on $a$ and $\phi$. When the above product is differentiated $k$ times, each derivative on a factor $\phi_{P_i,j_i}$ contributes a factor bounded by a constant multiple of $H_{i+1}^{-1}$, while the derivatives of $\omega$ are uniformly bounded. It follows that $\|\theta_P^{(k)}\|_{L^\infty} \le C m^{-n} (1+\sum_{i=1}^n 1/H_i)^k$. This proves~\eqref{eq:theta-derivative-bound} since $1+\sum_{i=1}^n 1/H_i \lesssim 1/H_n$.

Note that $(\theta_P \phi_{P,j})^{(k)}=\sum_{i=0}^k \binom{k}{i} \theta_P^{(i)} \phi_{P,j}^{(k-i)}$. Since $\| \phi_{P,j}^{(a)} \|_{L^\infty} \le c_a H_{n+1}^{-a}$, the estimate~\eqref{eq:theta-derivative-bound} and $H_n \ge H_{n+1}$ imply~\eqref{eq:theta-phi-derivative-bound}. This proves the lemma.

<!-- lemma-6: body_md -->
\label{lem:measure-construction}
    The sequence $(u_n)_{n \ge 0}$ converges uniformly to a continuous function $u$, with $u-u_0$ continuous and compactly supported. Moreover, $\mu:=D_x u$ is a nonzero finite positive Radon measure such that $\operatorname{supp} \mu \subseteq \Gamma_0$.

<!-- proof-9: body_md -->
We first prove the uniform convergence. Substituting~\eqref{eq:theta-child} into~\eqref{eq:def-un} yields
\begin{align}\label{eq:un-increment}
    u_{n+1}(x,y)-u_n(x,y) &= \sum_{P \in \mathscr T_n} \sum_{j \in \mathcal J_K} \sum_{\ell=1}^m \theta_{P_{j,\ell}}(y) F_{I(P_{j,\ell})}(x) - \sum_{P \in \mathscr T_n} \theta_P(y)F_{I(P)}(x) \nonumber \\
    &= \sum_{P \in \mathscr T_n} \sum_{j \in \mathcal J_K} \sum_{\ell=1}^m \frac{1}{m} \theta_P(y) \phi_{P,j}(y) F_{I(P_{j,\ell})}(x) - \sum_{P \in \mathscr T_n} \sum_{j \in \mathcal J_K} \theta_P(y) \phi_{P,j}(y) F_{I(P)}(x) \nonumber \\
    &=\sum_{P \in \mathscr T_n}\sum_{j \in \mathcal J_K} \theta_P(y) \phi_{P,j}(y) G_{P,j}(x),
\end{align}
where we used Lemma~\ref{lem:conservation}(b) in the second equality, and~\eqref{eq:G-p-j} in the last equality.

The function $G_{P,j}$ vanishes outside $\operatorname{int}(I(P))$. Since the intervals $I(P)$, $P \in \mathscr T_n$, have pairwise disjoint interiors, for each fixed $x$ at most one $P \in \mathscr T_n$ contributes nontrivially to~\eqref{eq:un-increment}. Once $P$ is fixed, $\operatorname{supp} \phi_{P,j} \subseteq J(P;j)$ implies that, for each fixed $y$, at most two indices $j \in \mathcal J_K$ contribute. Since $|G_{P,j}(x)| \le 1$, it follows from Lemma~\ref{lem:theta-derivatives} (with $k=0$) that $\|u_{n+1}-u_n\|_{L^\infty} \le 2C_0 m^{-n}$. Thus $u_n$ converges uniformly to a continuous function $u$. Since each $u_{n+1}-u_n$ is supported in the same compact set $I(\mathsf R) \times \operatorname{supp} \omega$, we obtain that $u-u_0 \in C_c(\mathbb C)$.

    We next consider the horizontal derivative. Since $F_I'=p_I$, differentiating~\eqref{eq:def-un} with respect to $x$ yields
\begin{equation}\label{eq:Dxun}
    D_xu_n =\sum_{P \in \mathscr T_n} \theta_P(y) p_{I(P)}(x) \mathop{}\!\mathrm{d} A.
\end{equation}
In particular, each $D_x u_n$ is a positive Radon measure.
By Lemma~\ref{lem:conservation}(c), the measures in~\eqref{eq:Dxun} have a common finite total mass $\int_{\mathbb R} \omega(y) \mathop{}\!\mathrm{d} y>0$. Moreover, $\operatorname{supp} D_x u_n \subseteq I(\mathsf R) \times \operatorname{supp} \omega$ for every $n$. Uniform convergence of $u_n$ implies $D_x u_n \to D_x u$ in the sense of distributions, so $\mu:=D_x u$ is a positive Radon measure supported on the same compact set. Since the total mass of $\mu$ is finite and positive, $\mu$ is finite and nonzero.

It remains to determine the support of $\mu$. For $n \ge 1$, let $A_n=\bigcup_{P \in \mathscr T_n} I(P) \times \operatorname{supp} \theta_P$. Each $A_n$ is compact. If $Q$ is a descendant of $P$, then $I(Q) \subset I(P)$ and $\operatorname{supp} \theta_Q \subset \operatorname{supp} \theta_P$, and therefore $A_{n+1} \subseteq A_n$. By~\eqref{eq:Dxun}, every $D_xu_N$ with $N\ge n$ is supported in $A_n$, and hence $\operatorname{supp} \mu \subset \cap_n A_n$.

Now take $(t,y) \in \cap_n A_n$. For each $n$, choose $P_n \in \mathscr T_n$ such that $(t,y) \in I(P_n) \times \operatorname{supp} \theta_{P_n}$ and let $t_n$ denote the left endpoint of $I(P_n)$. By Condition (c) of Proposition~\ref{prop:construct-tree}, $|t-t_n|\le \mathsf C H_n^2$. On the other hand, Lemma~\ref{lem:conservation}(a) gives $|y-B_{t_n}| \le \max\{3r,1-2r\} H_n \le H_n$. Since $H_n \to 0$, we have $t_n \to t$ and $B_{t_n} \to y$, and continuity of the Brownian path gives $y=B_t$. Thus $(t,y) \in \Gamma_0$, which proves $\operatorname{supp} \mu \subset \Gamma_0$. This completes the proof.

\label{lem:single}
Let $a_1 \ge 1$, $a_2>0$, $h>0$, and $L>0$. Let $I,J \subset \mathbb R$ be bounded intervals with $\mathrm{len}(I)=L$ and $\mathrm{len}(J) \le a_1h$. Suppose that

Write $(G \otimes \gamma)(x,y)=G(x) \gamma(y)$. Then, for every $z=(x,y) \in \mathbb R^2$,

\textbf{Case 1.} $d \ge h$. Let $d_x=\operatorname{dist}(x,I)$ and $d_y=\operatorname{dist}(y,J)$, then we have $\max\{d_x,d_y\} \ge d/\sqrt{2}$.

    &=\frac{1}{2} a_1 a_2 L d^{-2} \le 2 a_1 a_2 L \, (d+h)^{-2}.
\end{align*}
The case $d_x \ge d/\sqrt{2}$ can be handled similarly, using instead $|K_{\mathcal T}(X,Y)| \le 1/(4X^2)$.

\textbf{Case 2.} $d \le h$. For $\delta>0$, let $\mathcal T_\delta$ denote the truncation of $\mathcal T$ with kernel $K_{\mathcal T_\delta}(X,Y)=K_{\mathcal T}(X,Y) \mathbf{1}_{X^2+Y^2>\delta^2}$. Then
\[ \mathcal T_\delta (G \otimes \gamma)(x,y)=\iint_{s^2+t^2>\delta^2} K_{\mathcal T}(s,t) G(x-s) \gamma(y-t) \mathop{}\!\mathrm{d} s \mathop{}\!\mathrm{d} t. \]

\[ \mathcal T_\delta (G \otimes \gamma)(x,y) = \int_{\mathbb R} G(x-s) \int_{s^2+t^2>\delta^2} K_{\mathcal T}(s,t) (\gamma(y-t)-\gamma(y)) \mathop{}\!\mathrm{d} t \mathop{}\!\mathrm{d} s. \]

Therefore its uniform limit agrees with $\mathcal T b$ almost everywhere. Identifying $\mathcal T b$ with this bounded representative, we conclude that $\mathcal T b \in L^\infty(\mathbb C)$.

Fix $n$ sufficiently large and $z \in \mathbb C$. For $P \in \mathscr T_n$ and $j \in \mathcal J_K$, we have $\|G_{P,j}\|_{L^\infty} \le 1$, $\operatorname{supp} G_{P,j} \subseteq I(P)$, and $\mathrm{len}(I(P)) \le \mathsf CH_n^2$ by Condition (c) of Proposition~\ref{prop:construct-tree}. Moreover, $\operatorname{supp} (\theta_P \phi_{P,j}) \subseteq \operatorname{supp} \phi_{P,j} \subseteq J(P;j)$ and $J(P;j)$ has length $(1+r)H_{n+1}$. By Lemma~\ref{lem:theta-derivatives}, the derivative bounds in Lemma~\ref{lem:single} hold for $\gamma=(\theta_P \phi_{P,j})'$ with $a_2=\max\{C_1,C_2\} m^{-n}$ and $h=H_{n+1}$. Applying Lemma~\ref{lem:single} with $a_1=1+r$ gives
\[ \left| \mathcal T (G_{P,j} \otimes (\theta_P \phi_{P,j})')(z) \right| \lesssim m^{-n} H_n^2 \cdot (H_{n+1}+\operatorname{dist}(z,I(P) \times J(P;j)))^{-2}. \]
Summing over all $P \in \mathscr T_n$ and $j \in \mathcal J_K$, we obtain
\[ |\mathcal T \mathsf b_n(z)| \lesssim m^{-n} H_n^2 \sum_{P \in \mathscr T_n} \sum_{j \in \mathcal J_K} (H_{n+1}+\operatorname{dist}(z,I(P) \times J(P;j)))^{-2}. \]

For $P \in \mathscr T_n$ and $j \in \mathcal J_K$, let $z_{P,j}$ be the center of the rectangle $I(P) \times J(P;j)$. Since $\mathrm{len}(I(P)) \le \mathsf C H_n^2$ and $J(P;j)$ has length $(1+r)H_{n+1}$, we have, for all sufficiently large $n$, that the diameter of this rectangle is at most $\mathsf CH_n^2+(1+r)H_{n+1} \le 2H_{n+1}$. It follows that
\[ |z-z_{P,j}| \le \operatorname{dist}(z,I(P) \times J(P;j))+H_{n+1}. \]

Define the discrete measure $\Lambda_n =\sum_{P \in \mathscr T_n} \sum_{j \in \mathcal J_K} \delta_{z_{P,j}}$, where $\delta_z$ is the Dirac measure at $z$. Then
\[ |\mathcal T \mathsf b_n(z)| \lesssim m^{-n} H_n^2 \sum_{P \in \mathscr T_n} \sum_{j \in \mathcal J_K} (H_{n+1}+|z-z_{P,j}|)^{-2}= m^{-n} H_n^2 \int_{\mathbb C} (H_{n+1}+|z-w|)^{-2} \mathop{}\!\mathrm{d} \Lambda_n(w). \]

We now estimate the mass growth of $\Lambda_n$. Let $R>H_{n+1}$. If $z_{P,j} \in B(z,R)$, then $I(P) \times J(P;j)$ intersects $B(z,R)$. We therefore count the rectangles $I(P) \times J(P;j)$ that can intersect this ball. Recall that if $P$ has height range $[\ell H_n,(\ell+1)H_n]$, then $J(P;j) = (\ell K+j)H_{n+1}+H_{n+1}[-3r,1-2r]$. Note that the vertical intervals $\ell' H_{n+1}+H_{n+1}[-3r,1-2r]$ are translates of each other by integer multiples of $H_{n+1}$. Thus the projection of $B(z,R)$ onto the $y$-axis can intersect at most $(2R/H_{n+1})+4$ such intervals. Since $r \in (0,1/8)$, for each integer $\ell'$, there are at most two pairs $(\ell,j) \in \mathbb Z \times \mathcal J_K$ such that $\ell'=\ell K+j$.

Fix such a pair $(\ell,j)$. For sufficiently large $n$, we have $R>H_{n+1}>H_n^2$. If $R<1/2$, by Lemma~\ref{lem:uniform-count} and extending the interval if necessary, we derive that the number of $P \in \mathscr T_n$ with height range $[\ell H_n,(\ell+1)H_n]$ such that $I(P)$ intersects the $x$-projection of $B(z,R)$ is at most $C_B \sqrt{n+1} \frac{\sqrt{2R}}{H_n}$. If $R \ge 1/2$, the same upper bound also holds by applying Lemma~\ref{lem:uniform-count} to $W=[0,1]$.

Combining the vertical and horizontal enumerations gives
\begin{equation}\label{eq:Lambda_mass}
    \Lambda_n(B(z,R)) \le \left(\frac{2R}{H_{n+1}}+4 \right) \times 2 \times C_B \sqrt{n+1} \frac{\sqrt{2R}}{H_n} \lesssim C_B \sqrt{n+1} \frac{R^{3/2}}{H_n H_{n+1}}.
\end{equation}

For $j \ge 1$, define the dyadic annulus $A_j=\{w \in \mathbb C: 2^j H_{n+1} \le |w-z| < 2^{j+1} H_{n+1}\}$. Applying~\eqref{eq:Lambda_mass} to each ball $B(z,2^j H_{n+1})$, $j \ge 1$, yields
\begin{align*}
    &\quad \int_{\mathbb C} (H_{n+1}+|z-w|)^{-2} \mathop{}\!\mathrm{d} \Lambda_n(w) \\
    &=\int_{B(z,2H_{n+1})} (H_{n+1}+|z-w|)^{-2} \mathop{}\!\mathrm{d} \Lambda_n(w) + \sum_{j=1}^\infty \int_{A_j} (H_{n+1}+|z-w|)^{-2} \mathop{}\!\mathrm{d} \Lambda_n(w) \\
    &\le H_{n+1}^{-2} \cdot\Lambda_n(B(z,2H_{n+1}))+\sum_{j=1}^\infty (2^j H_{n+1})^{-2} \cdot \Lambda_n(B(z,2^{j+1} H_{n+1})) \\
    &\lesssim C_B \sqrt{n+1} (H_n H_{n+1})^{-1} \left( H_{n+1}^{-2} \cdot (2H_{n+1})^{3/2} + \sum_{j=1}^\infty (2^j H_{n+1})^{-2} (2^{j+1} H_{n+1})^{3/2} \right) \\
    &\lesssim C_B \sqrt{n+1} H_n^{-1} H_{n+1}^{-3/2}.
\end{align*}
Substituting this into the preceding estimate for $\mathcal T \mathsf b_n$, we find
\[ |\mathcal T \mathsf b_n(z)| \lesssim m^{-n} H_n^2 \times C_B \sqrt{n+1} H_n^{-1} H_{n+1}^{-3/2} \lesssim \sqrt{n+1} (\sqrt{K}/m)^n. \]
Since the bound is uniform in $z$, this proves~\eqref{eq:T-b-bounded} and thereby completes the proof.

<!-- lemma-11: body_md -->
\label{lem:holder-regularity}
The sequence $(f_n)_{n \ge 0}$ converges uniformly to an $\alpha$-H\"older continuous function $f:[0,1] \to \mathbb R$.
<!-- end lemma-11 -->
<!-- proof-16: body_md -->
Since $v([0,1])=[-r,1+r]$, replacing $f_n$ by $f_{n+1}$ on an interval $P \in \mathscr A_n$ changes the function by at most $(1+r)K^{-n}$. Hence $\|f_{n+1}-f_n\|_{L^\infty([0,1])} \le (1+r)K^{-n}$, and thus $f_n$ converges uniformly to a continuous function $f:[0,1]\to\mathbb R$.

We next prove the H\"older estimate. We first claim that the oscillation of $f$ on every $S$-adic interval $P \in \mathscr B_n$ is at most $5 K^{-n}$.

Suppose first that $P \in \mathscr A_n$, then the oscillation of $f$ on $P$ is at most $K^{-n}+2 \sum_{j \ge n} (1+r)K^{-j} \le 5K^{-n}$. If instead $P \notin \mathscr A_n$, then for some $j \le n-1$ we have that $f=f_{j+1}$ on $P$. On this interval, $f_{j+1}$ is linear with slope of absolute value at most $(S/K)^{j+1}$, hence the oscillation is at most $(S/K)^{j+1} \cdot S^{-n} \le K^{-n}<5K^{-n}$.

Now, for $0 \le x<y \le 1$, choose $n \ge 0$ so that $S^{-(n+1)} \le |x-y| \le S^{-n}$. Then the interval $[x,y]$ is contained in the union of at most two adjacent $S$-adic intervals in $\mathscr B_n$, and consequently
\[ |f(x)-f(y)| \le 10K^{-n} \le 10S^{-\alpha n} \le 10 S^\alpha \cdot |x-y|^\alpha, \]
where we used~\eqref{eq:parameter-choice}. This shows that $f$ is $\alpha$-H\"older continuous.

<!-- proposition-5: body_md -->
\label{prop:riesz-bounded-holder}
    The series $\sum_{n \ge 0} \mathcal T \mathsf b_n$ converges uniformly on $\mathbb C$ to $\mathcal Tb$. In particular, $\mathcal T b \in L^\infty(\mathbb C)$.
<!-- end proposition-5 -->
<!-- proof-21: body_md -->
Recall from~\eqref{eq:b-n-holder} that each $\mathsf b_n$ is smooth and compactly supported, so $\mathcal T \mathsf b_n \in L^\infty(\mathbb C)$. We claim that there exists a constant $C<\infty$ such that for all $n \ge 0$,
\begin{equation}\label{eq:T-b-bounded-holder}
    \|\mathcal T \mathsf b_n\|_{L^\infty} \le C (n+1) (K^2/(2MS))^n.
\end{equation}
Since $K^2/(2MS)<1$ by~\eqref{eq:parameter-choice}, this estimate implies $\sum_{n=0}^\infty \|\mathcal T \mathsf b_n \|_{L^\infty}<\infty$, and hence $\sum_{n \ge 0} \mathcal T \mathsf b_n$ converges uniformly to a bounded function. On the other hand, Lemma~\ref{lem:vertical-derivative-holder}, together with the boundedness of $\mathcal T$ on $L^q(\mathbb C)$, shows that the same series converges to $\mathcal T b$ in $L^q(\mathbb C)$. Therefore its uniform limit agrees with $\mathcal T b$ almost everywhere, proving the proposition once~\eqref{eq:T-b-bounded-holder} is established.

Fix $n \ge 0$ and $z=(x,y) \in \mathbb R^2$. For $P \in \mathscr A_n$ and $j \in \widetilde{\mathcal J}_K$, write $d_P=\operatorname{dist}(x,P)$, $e_{P,j}=\operatorname{dist}(y,J(P;j))$, and $D_{P,j}=\operatorname{dist}(z,P \times J(P;j))$. By~\eqref{eq:G-p-j-holder}, Lemmas~\ref{lem:theta-derivatives-holder} and~\ref{lem:G-p-j-sum-zero}, the pair $(G_{P,j}, (\theta_P\phi_{P,j})')$ satisfies the hypotheses of Lemma~\ref{lem:single} with $a_1=1+2\delta$ and $a_2=\max\{C_1,C_2\} (2M)^{-n}$. Moreover, $\int_{\mathbb R} G_{P,j}=0$.

For fixed $P$, recall that the intervals $J(P;j)$ are translates of each other by multiples of $H_{n+1}$ and have length $(1+2\delta)H_{n+1}<2H_{n+1}$. Hence, for each integer $k \ge 0$, there are at most four indices $j \in \widetilde{\mathcal J}_K$ satisfying $e_{P,j} \in [kH_{n+1},(k+1)H_{n+1})$. It follows that for every nonnegative decreasing integrable function $\mathsf F:[0,\infty) \to [0,\infty)$,
\begin{equation}\label{eq:summation-bound}
    \sum_{j \in \widetilde{\mathcal J}_K} \mathsf F(e_{P,j}) \le 4 \sum_{k=0}^\infty \mathsf F(kH_{n+1}) \le 4 \left( \mathsf F(0)+\frac{1}{H_{n+1}} \int_0^\infty \mathsf F(t) \mathop{}\!\mathrm{d} t \right).
\end{equation}

We split the sum~\eqref{eq:b-n-holder} according to the value of $d_P$.

\begin{itemize}
    \item \textit{Intervals with $d_P<L_n$.} Since $\mathscr A_n$ consists of $S$-adic intervals of length $L_n$, there are at most three such intervals. Lemma~\ref{lem:single} gives
    \[ |\mathcal T(G_{P,j} \otimes (\theta_P \phi_{P,j})')(z)| \lesssim (2M)^{-n} L_n (H_{n+1}+D_{P,j})^{-2} \lesssim (2M)^{-n} L_n (H_{n+1}+e_{P,j})^{-2}. \]
    Applying~\eqref{eq:summation-bound} with $\mathsf F(t)=(H_{n+1}+t)^{-2}$ gives
    \begin{equation}\label{eq:near}
        \sum_{\substack{P \in \mathscr A_n \\ d_P<L_n}} \sum_{j \in \widetilde{\mathcal J}_K} |\mathcal T(G_{P,j} \otimes (\theta_P \phi_{P,j})')(z)| \lesssim (2M)^{-n} L_n H_{n+1}^{-2}.
    \end{equation}
    
    \item \textit{Intervals with $d_P \ge L_n$.} Since $\int_{\mathbb R} G_{P,j}=0$, Lemma~\ref{lem:single+} and $D_{P,j}\ge(d_P+e_{P,j})/\sqrt{2}$ imply
    \begin{equation}\label{eq:far-2}
        |\mathcal T(G_{P,j} \otimes (\theta_P \phi_{P,j})')(z)| \lesssim (2M)^{-n} L_n^2 (d_P+e_{P,j})^{-1} (H_{n+1}+d_P+e_{P,j})^{-2}.
    \end{equation}

    For $d>0$, the function $\mathsf F_d(t)=(d+t)^{-1} (H_{n+1}+d+t)^{-2}$ is decreasing and satisfies $\int_0^\infty \mathsf F_d(t) \mathop{}\!\mathrm{d} t \le d^{-1} \int_0^\infty (H_{n+1}+d+t)^{-2} \mathop{}\!\mathrm{d} t=(d(H_{n+1}+d))^{-1}$. Using~\eqref{eq:summation-bound} and~\eqref{eq:far-2}, we obtain
    \begin{align}\label{eq:far-3}
        \sum_{j\in\widetilde{\mathcal J}_K} |\mathcal T(G_{P,j} \otimes (\theta_P \phi_{P,j})')(z)|
        &\lesssim (2M)^{-n} L_n^2 \left( \frac{1}{d_P(H_{n+1}+d_P)^2} + \frac{1}{H_{n+1} d_P(H_{n+1}+d_P)} \right) \nonumber \\
        &\lesssim (2M)^{-n} \frac{L_n^2}{H_{n+1} d_P (H_{n+1}+d_P)}.
    \end{align}
    For each integer $k \ge 1$, there are at most two $S$-adic intervals of length $L_n$ such that $d_P \in [kL_n,(k+1)L_n)$. Since $t \mapsto (t(H_{n+1}+t))^{-1}$ is decreasing, we have
    \begin{align*}
        \sum_{\substack{P \in \mathscr A_n \\ d_P \ge L_n}} \frac{1}{d_P(H_{n+1}+d_P)} &\le 2 \sum_{k=1}^\infty \frac{1}{kL_n(H_{n+1}+kL_n)} \le \frac{2}{L_n(H_{n+1}+L_n)}+\frac{2}{L_n} \int_{L_n}^\infty \frac{1}{t(H_{n+1}+t)} \mathop{}\!\mathrm{d} t \\
        &\lesssim (L_n H_{n+1})^{-1} \left(1+\log \left(1+\frac{H_{n+1}}{L_n} \right) \right).
    \end{align*}
    Together with~\eqref{eq:far-3}, this yields
    \begin{equation}\label{eq:far}
        \sum_{\substack{P \in \mathscr A_n \\ d_P \ge L_n}} \sum_{j \in \widetilde{\mathcal J}_K} |\mathcal T(G_{P,j} \otimes (\theta_P \phi_{P,j})')(z)| \lesssim (2M)^{-n} L_n H_{n+1}^{-2} \left(1+\log \left(1+\frac{H_{n+1}}{L_n} \right) \right).
    \end{equation}
\end{itemize}

Combining~\eqref{eq:near} and~\eqref{eq:far}, and
recalling~\eqref{eq:b-n-holder}, we conclude that
\[ \|\mathcal T \mathsf b_n\|_{L^\infty} \lesssim (2M)^{-n} L_n H_{n+1}^{-2} \left(1+\log \left(1+\frac{H_{n+1}}{L_n} \right) \right). \]
Since $L_n=S^{-n}$ and $H_{n+1}=K^{-(n+1)}$, this implies~\eqref{eq:T-b-bounded-holder} and thereby completes the proof.
