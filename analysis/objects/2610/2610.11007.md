<!-- assumption-1: body_md -->
\label{ass:mono}
$q_i(S \cup T) \le q_i(S)$ for every finite $S$, $T$ and every $i \in S$.
<!-- end assumption-1 -->
<!-- assumption-2: body_md -->
\label{ass:bounded}
$\limsup_{n \to \infty} M(n)/n < \rho$.
<!-- end assumption-2 -->
<!-- assumption-3: body_md -->
\label{ass:capacity}
$Q(S) \le \bar Q$ for every finite $S$, for a constant $\bar Q < \infty$.
<!-- end assumption-3 -->
<!-- proposition-1: body_md -->
\label{prop:size}
Under Assumption~\ref{ass:capacity}, every file with $R(S) \ge 0$, and
in particular every optimal file $S^\star$, satisfies
\[
\kappa\, C(S) \le \bar w\, \bar Q
\qquad\text{and}\qquad
|S| \le \frac{\bar w\, \bar Q}{\kappa\, c_{\min}} = \frac{\bar Q}{\rho}.
\]
Neither bound depends on the pool size $N$ or the budget $B$. If
$B \ge \bar w \bar Q / \kappa$, the budget never binds. Under~\eqref{eq:sd}
the capacity is $\bar Q = \chi \bar G$.
<!-- end proposition-1 -->
<!-- corollary-1: body_md -->
\label{cor:accretion}
Assume~\eqref{eq:sd} with $\chi \bar G < \infty$, and let $S^\star_N$ be
optimal for the pool $[N]$. Every $S \subseteq [N]$ satisfies
$R(S) \le \bar w\, \chi \bar G - \kappa c_{\min} |S|$, hence
\[
R(S^\star_N) - R(S_N^{\mathrm{acc}}) \ \ge\ \kappa c_{\min}\, |S_N^{\mathrm{acc}}| - \bar w\, \chi \bar G,
\]
which tends to infinity whenever $|S_N^{\mathrm{acc}}| \to \infty$.
<!-- end corollary-1 -->
<!-- proposition-2: body_md -->
\label{prop:nodilution}
Suppose $q_i(S) = q_i$ for every $S \ni i$, so that
$v_i^{\mathrm{solo}} = w_i q_i - \kappa c_i$. Then $R(S) = \sum_{i \in S} v_i^{\mathrm{solo}}$. If $B \ge C(S_N^{\mathrm{acc}})$, then $S_N^{\mathrm{acc}}$ is
optimal, and the optimal files are exactly the $S$ with
$S_N^{\mathrm{acc}} \subseteq S \subseteq \overline{S}_N^{\mathrm{acc}}$ and $C(S) \le B$.
<!-- end proposition-2 -->
<!-- assumption-4: body_md -->
\label{ass:feedback}
In session $t$ the curator observes the first two signals above. All
draws are independent across instructions and sessions, and their law
depends on the file only through $q_i(S_t)$. The gains $r_i$ enter no
observation. Under magnitude feedback the curator also observes
$X_{i,t}$ for each followed $i \in S_t$.
<!-- end assumption-4 -->
<!-- proposition-3: body_md -->
\label{prop:blind}
Suppose Assumption~\ref{ass:feedback} holds, with a fixed file $S$ played in every
session. A deletion policy that uses only compliance data has the same
law of decisions under every parameter vector with the same $(p, q)$.
Take an instance of~\eqref{eq:sd} with $S = \{1, 2\}$. Instruction~1
is helpful but often ignored; instruction~2 is mostly followed but worth
little or harmful. Deleting the most-ignored instruction removes
instruction~1 with probability tending to one as sessions accumulate.
<!-- end proposition-3 -->
<!-- theorem-1: body_md -->
\label{thm:threshold}
Assumption~\ref{ass:sight} with $\pi, \varepsilon \in (0, 1]$, $W \ge 2$,
$\Lambda_F > 0$ and $T \ge 1$. If
\[
x < \omega\, (2 - 1/\psi_W) \qquad \text{and} \qquad
\omega / \psi_W < 1 - x_T / 2 - x,
\]
then $k = 2$ is the unique minimizer of $\mathcal{L}$ over $k \ge 1$. As
$\psi_W \to 1$ the condition reads $x < \omega < 1 - x_T / 2 - x$.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:etc}
Assumption~\ref{ass:feedback} with magnitude feedback and
compliance~\eqref{eq:sd} under tied loads; $m \ge 2$, $B \ge c_{\max}$.
For every $n_0 \ge 8 \ln(6NT) / p_{\min}$,
\[
\mathrm{Reg}_T \ \le\ \Delta_{\max}\, L \Big( J + \Big\lceil \frac{s + n_B}{m} \Big\rceil \Big)
\ +\ \Lambda_T(n_0),
\]
where the learning term $\Lambda_T(n_0)$ is given in
Theorem~\ref{thm:etcfull}.
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm:lower}
Let $n_m = \lfloor N / m \rfloor$, $T \ge L n_m$, $N, m \ge 1$, $B > 0$.
There is a family of $N$ instances of Assumption~\ref{ass:feedback} with
magnitude feedback and compliance~\eqref{eq:sd}. They share
$c \le B$, $p$, $\theta$ and $a$, with noiseless gains and no harm. Under
instance $j$ the optimal file is $\{j\}$, missing $j$ costs at least
$\Delta = p\, \theta\, g(a) - \kappa c > 0$ per session, and
$\Delta_{\max} \le \Delta + \kappa B$. Every policy that adds at most $m$
instructions per epoch from the empty file satisfies
\[
\max_j \mathrm{Reg}_T(j) \ \ge\ \frac{\Delta L}{2} (n_m - 1)
\ \ge\ \frac{\Delta L N}{4 m} \quad (N \ge 4m).
\]

<!-- propositiona-1: body_md -->
\label{prop:sizefull}
Assume~\eqref{eq:sd} with $\chi \bar G < \infty$.
(i) Every $S$ satisfies
$R(S) \le \bar w\, \Theta(S)\, g(A(S)) - \kappa C(S)
\le \bar w\, \chi \bar G - \kappa C(S)$, where $\Theta(S) = \sum_{i \in S} \theta_i$.
(ii) Every $S$ with $R(S) \ge 0$ satisfies
$\kappa C(S) \le \bar w\, \chi \bar G$ and $|S| \le \chi \bar G / \rho$.
(iii) If $B \ge \bar w \chi \bar G / \kappa$, the token budget never binds: the capped
optima are exactly the uncapped optima.
(iv) Every optimum $S^\star$ has $R(S^\star) \ge 0$, so (ii) applies to it
for every $N$ and every $B$.
(v) In the single-choice case $\chi \bar G = 1$, every $S$ with $R(S) \ge 0$
has $C(S) \le \bar w / \kappa$ and $|S| \le 1/\rho$.
The proof uses~\eqref{eq:sd} only through Assumption~\ref{ass:capacity}
with $\bar Q = \chi \bar G$; Proposition~\ref{prop:size} is (ii)-(iv) so
stated.
<!-- end propositiona-1 -->
<!-- proof-1: body_md -->
(i) Since $w_i \le \bar w$, $q_i(S) \ge 0$ and $\bar w \ge 0$,
$R_0(S) \le \bar w\, Q(S) = \bar w\, \Theta(S)\, g(A(S))$. As
$\theta_i \le \chi a_i$ for every $i$, $\Theta(S) \le \chi A(S)$, so
$\Theta(S)\, g(A(S)) \le \chi\, A(S)\, g(A(S)) \le \chi \bar G$.
(ii) $R(S) \ge 0$ and (i) give $\kappa C(S) \le \bar w \chi \bar G$, and
$C(S) \ge c_{\min} |S|$ gives $|S| \le \bar w \chi \bar G / (\kappa c_{\min})
= \chi \bar G / \rho$. (If $\bar w = 0$ then $R(S) = -\kappa C(S) \ge 0$
forces $S = \emptyset$, and $\chi \bar G / \rho = 0$.)
(iii) Let $S_\infty$ be an uncapped optimum. It has
$R(S_\infty) \ge R(\emptyset) = 0$, so $C(S_\infty) \le \bar w \chi \bar G /
\kappa \le B$ by (ii): $S_\infty$ is feasible under the cap, hence a
capped optimum. Conversely a capped optimum $S$ has
$R(S) \ge R(S_\infty)$, so it is an uncapped optimum.
(iv) $\emptyset$ is feasible for every $B \ge 0$ and $R(\emptyset) = 0$;
the constants $\bar w$, $\chi$ and $\bar G$ are taken over all candidates.
(v) Put $\chi \bar G = 1$ in (ii).
<!-- end proof-1 -->
<!-- proof-2: body_md -->
Proposition~\ref{prop:sizefull}(i) with $C(S) \ge c_{\min} |S|$ gives the
bound on $R(S)$. Apply it to $S = S_N^{\mathrm{acc}}$ and use
$R(S^\star_N) \ge 0$ (Proposition~\ref{prop:sizefull}(iv)).
<!-- end proof-2 -->
<!-- proof-3: body_md -->
Additivity is immediate from $q_i(S) = q_i$. As $v_i^{\mathrm{solo}} > 0$ exactly on
$S_N^{\mathrm{acc}}$ and $v_i^{\mathrm{solo}} \le 0$ off it,
$R(S) \le \sum_{i \in S \cap S_N^{\mathrm{acc}}} v_i^{\mathrm{solo}} \le R(S_N^{\mathrm{acc}})$, with equality iff
$S_N^{\mathrm{acc}} \subseteq S$ and $v_i^{\mathrm{solo}} = 0$ on $S \setminus S_N^{\mathrm{acc}}$, that is
$S \subseteq \overline{S}_N^{\mathrm{acc}}$; $S_N^{\mathrm{acc}}$ is feasible because $B \ge C(S_N^{\mathrm{acc}})$.
<!-- end proof-3 -->
<!-- propositiona-2: body_md -->
\label{prop:inside}
Under Assumption~\ref{ass:mono}, with $w_i$ of any sign, every optimal
$S^\star$ satisfies $S^\star \subseteq \{i : w_i > 0\} \cap \overline{S}_N^{\mathrm{acc}}$, and
$S^\star \cap S_N^{\mathrm{acc}}$ is also optimal.
<!-- end propositiona-2 -->
<!-- proof-4: body_md -->
Let $P = \{i : w_i > 0\}$; subsets of feasible files are feasible. Two
inequalities hold for every $S$ under Assumption~\ref{ass:mono}.
(L1) $R(S \cap P) \ge R(S) + \kappa C(S \setminus P)$: indeed
$R(S \cap P) - R(S) = \sum_{j \in S \cap P} w_j [q_j(S \cap P) - q_j(S)]
- \sum_{j \in S \setminus P} w_j q_j(S) + \kappa C(S \setminus P)$; the first
sum is $\ge 0$ by monotone dilution and $w_j > 0$, the second is $\le 0$
because $w_j \le 0$.
(L2) If $w_j \ge 0$ for all $j \in S$ and $i \in S$, then
$R(S) - R(S \setminus \{i\}) \le v_i^{\mathrm{solo}}$: indeed
$R(S) - R(S \setminus \{i\}) = w_i q_i(S)
+ \sum_{j \in S \setminus \{i\}} w_j [q_j(S) - q_j(S \setminus \{i\})]
- \kappa c_i$; each bracket is $\le 0$ and $w_i q_i(S) \le w_i q_i(\{i\})$.
Now let $S^\star$ be optimal. If $S^\star \not\subseteq P$, then
$C(S^\star \setminus P) > 0$ and (L1) gives
$R(S^\star \cap P) > R(S^\star)$, a contradiction; so $S^\star \subseteq P$.
If some $i \in S^\star$ had $v_i^{\mathrm{solo}} < 0$, (L2) would give
$R(S^\star \setminus \{i\}) > R(S^\star)$; so $S^\star \subseteq \overline{S}_N^{\mathrm{acc}}$.
Finally remove the items with $v_i^{\mathrm{solo}} = 0$ from $S^\star$ one at a time; each
intermediate set $S'$ satisfies $S' \subseteq P$, so (L2) gives
$R(S' \setminus \{i\}) \ge R(S')$, the value never drops, and the last set
is $S^\star \cap S_N^{\mathrm{acc}}$.
<!-- end proof-4 -->
<!-- propositiona-3: body_md -->
\label{prop:sizegeneral}
(i) Every $S$ satisfies $R(S) \le \bar w M(|S|) - \kappa C(S)$.
(ii) If $R(S) \ge 0$, then $\kappa C(S) \le \bar w M(|S|)$ and $|S| \in \mathcal{N}$.
(iii) Under Assumption~\ref{ass:bounded}, $\mathcal{N}$ is finite. With
$\bar n = \max \mathcal{N}$, every $S$ with $R(S) \ge 0$ has $|S| \le \bar n$ and
$C(S) \le \bar w M^*(\bar n) / \kappa$, and a cap
$B \ge \bar w M^*(\bar n) / \kappa$ never binds.
(iv) Every optimal $S^\star$ has $R(S^\star) \ge 0$, so (ii) and (iii)
apply to it, with bounds that depend on neither $N$ nor $B$.
<!-- end propositiona-3 -->
<!-- proof-5: body_md -->
(i) Since $w_i \le \bar w$, $q_i(S) \ge 0$ and $\bar w \ge 0$,
$R_0(S) \le \bar w\, Q(S) \le \bar w M(|S|)$.
(ii) Combine $R(S) \ge 0$ with (i) and $C(S) \ge c_{\min} |S|$.
(iii) If $\bar w = 0$, then $\mathcal{N} = \{0\}$. Otherwise
Assumption~\ref{ass:bounded} gives $M(n) < \rho\, n$, that is
$n \notin \mathcal{N}$, for all large $n$. Then
$\kappa C(S) \le \bar w M(|S|) \le \bar w M^*(\bar n)$. The cap never
binds by the argument of Proposition~\ref{prop:sizefull}(iii), with
$\bar w M^*(\bar n)$ in place of $\bar w \chi \bar G$.
(iv) $\emptyset$ is feasible and $R(\emptyset) = 0$; $\bar w$ and $M$ are
taken over all candidates.
<!-- end proof-5 -->
<!-- lemmaa-1: body_md -->
\label{lem:mono}
Under Assumption~\ref{ass:mono}, $M(n+1)/(n+1) \le M(n)/n$ for $n \ge 1$.
Hence Assumption~\ref{ass:bounded} holds iff $M(n_1) < \rho\, n_1$ for some
$n_1 \ge 1$, and $\mathcal{N}$ is an initial segment of $\mathbb N$.
<!-- end lemmaa-1 -->
<!-- proof-6: body_md -->
Fix $S'$ with $|S'| = n + 1$ and let $j \in S'$ minimize $q_j(S')$. Removing
the smallest of $n + 1$ terms keeps at least a share $n/(n+1)$ of the sum,
and Assumption~\ref{ass:mono} gives $q_i(S' \setminus \{j\}) \ge q_i(S')$,
so
$M(n) \ge \sum_{i \in S' \setminus \{j\}} q_i(S' \setminus \{j\})
\ge \sum_{i \in S' \setminus \{j\}} q_i(S')
\ge \frac{n}{n+1} \sum_{i \in S'} q_i(S')$.
Take the supremum over $S'$. A nonincreasing sequence has limit superior
equal to its infimum, which is below $\rho$ iff some term is. For
$n \ge 1$, $n \in \mathcal{N}$ iff $M(n)/n \ge \rho$, which is preserved as $n$
decreases, and $0 \in \mathcal{N}$.
<!-- end proof-6 -->
<!-- head-1: body_md -->
Without Assumption~\ref{ass:mono}, $\mathcal{N}$ need not be an interval
(Lemma~\ref{lem:mono} needs the assumption), which is why $\bar n$ is
defined as a maximum: take identical candidates with
$w_i = c_i = 1$, $\kappa = 0.5$, and $q_i(S) = 0.1$ if $|S| \le 2$, $1$ if
$|S| = 3$, and $0$ if $|S| \ge 4$; then $\mathcal{N} = \{0, 3\}$ and $R(S) = 1.5$
for $|S| = 3$. The strict inequality in Assumption~\ref{ass:bounded}
cannot be relaxed: for identical instructions with
$q(n) = \rho + (1 - \rho)/n^2$ and $0 < \rho < 1$, Assumption~\ref{ass:mono}
holds, $M(n)/n \to \rho$, every instruction is in $S_N^{\mathrm{acc}}$, and
$R(S) = \bar w (1 - \rho)/|S|$, so $R(S_N^{\mathrm{acc}}) \to 0$ while
$R(S^\star_N) = \bar w (1 - \rho)$ for $B \ge c_{\min}$.
<!-- end head-1 -->
<!-- head-2: body_md -->
Table~\ref{tab:examples} takes identical instructions ($w_i = \bar w > 0$,
$c_i = c_{\min}$) whose compliance $q(n)$ in a file of $n$ is
nonincreasing. Then Assumption~\ref{ass:mono} holds, $M(n) = n q(n)$,
$R(S) = \bar w\, [M(n) - \rho\, n]$ for $|S| = n$, and
$\bar n = \max \{n : q(n) \ge \rho\}$ ($0$ if $q(1) < \rho$). When $M$ is
bounded, the cost bound $\bar w \sup_n M(n) / \kappa$ is linear in
$\bar w / \kappa$. Under power dilution ($0 < \beta < 1$), $\bar n$ grows
like $(\bar w / \kappa)^{1/\beta}$ and, for $\alpha > \rho$, blows up as
$\beta \downarrow 0$, the no-dilution limit.
<!-- end head-2 -->
<!-- propositiona-4: body_md -->
\label{prop:load}
Assume~\eqref{eq:sd} with $\bar w > 0$ and $\bar\theta < \infty$, and let
$S^\star$ be a nonempty optimum under any budget. Then $w_i > 0$ on
$S^\star$, $\kappa c_i \le w_i \theta_i\, g(A(S^\star))$ for every
$i \in S^\star$, and $g(A(S^\star)) \ge \rho / \bar\theta$. Hence, if
$\rho / \bar\theta > 1$ the only optimum is $\emptyset$; and if
$g(\infty) < \rho / \bar\theta \le 1$ then
$A(S^\star) \le g^{-1}(\rho / \bar\theta) < \infty$ and, when
$a_{\min} > 0$, $|S^\star| \le g^{-1}(\rho / \bar\theta) / a_{\min}$.
<!-- end propositiona-4 -->
<!-- proof-7: body_md -->
By Assumption~\ref{ass:mono}, $w_i > 0$ on $S^\star$ by
Proposition~\ref{prop:inside}. Fix $i \in S^\star$; the file
$S^\star \setminus \{i\}$ is feasible. By Assumption~\ref{ass:mono} and
$w_l > 0$ for $l \ne i$,
$R(S^\star \setminus \{i\}) \ge \sum_{l \in S^\star \setminus \{i\}} w_l q_l(S^\star)
- \kappa C(S^\star) + \kappa c_i = R(S^\star) - w_i q_i(S^\star) + \kappa c_i$,
and optimality gives $w_i q_i(S^\star) \ge \kappa c_i$, that is
$\kappa c_i \le w_i \theta_i\, g(A(S^\star))$. Since $w_i \le \bar w$,
$\theta_i \le \bar\theta$ and $c_i \ge c_{\min}$,
$g(A(S^\star)) \ge \kappa c_{\min} / (\bar w \bar\theta) = \rho / \bar\theta$.
As $g \le 1$, this is impossible when $\rho / \bar\theta > 1$. When
$g(\infty) < \rho / \bar\theta \le 1$, $A(S^\star)$ belongs to
$\{x : g(x) \ge \rho / \bar\theta\}$, whose supremum $g^{-1}(\rho / \bar\theta)$
is finite; and $|S^\star|\, a_{\min} \le A(S^\star)$.
<!-- end proof-7 -->
<!-- lemmaa-2: body_md -->
\label{lem:single}
For nonempty $S$ and $i \in S$,
$R_0(S) - R_0(S \setminus \{i\}) = \theta_i\, g(A - a_i)\, [w_i - \phi\, e_i(S)]$.
<!-- end lemmaa-2 -->
<!-- proof-8: body_md -->
$R_0(S) - R_0(S \setminus \{i\}) = g(A) W(S) - g(A - a_i)(W(S) - w_i \theta_i)
= w_i \theta_i\, g(A - a_i) - W(S)[g(A - a_i) - g(A)]$. By the definition
of $e_i$, $g(A - a_i) - g(A) = \theta_i e_i(S)\, g(A)\, g(A - a_i)$, so the
last term is $\phi\, \theta_i e_i(S)\, g(A - a_i)$.
<!-- end proof-8 -->
<!-- propositiona-5: body_md -->
\label{prop:sharp}
Assume~\eqref{eq:sd}, let $w_i$ be of any sign, and let $S$ be
deletion-stable at $\mu > 0$.
(i) $w_i > \phi\, e_i(S)$ for every $i \in S$, and
$W(S)\, [1 - g(A) E(S)] > 0$. If $g(A) E(S) < 1$, in particular if
$|S| = 1$, then $\phi > 0$ and $w_i > 0$ on $S$.
(ii) If $\phi > 0$, then $w_i > 0$ on $S$ and
\[
\mu\, C(S) \le \phi \cdot \frac{g(A - a^S_{\max})}{g(A)}\, \big[1 - g(A) E(S)\big]
\le \phi, \qquad \text{so } R_\mu(S) \ge 0 .
\]
(iii) If $\phi > 0$, then $\mu\, C(S) \le \bar w\, X_g(S) \le a_g$, and
$X_g(S) \le Q(S)$; under $\chi \bar G < \infty$,
$a_g \le \bar w \chi \bar G$.
Hence every optimum at any price $\mu > 0$ under any budget has
$\mu\, C(S^\star) \le a_g$, and a cap $B \ge a_g / \kappa$ never binds:
its optima are exactly the uncapped optima.
<!-- end propositiona-5 -->
<!-- proof-9: body_md -->
(i) Stability at $\mu$ and Lemma~\ref{lem:single} give
$\theta_i\, g(A - a_i)\, [w_i - \phi e_i(S)] \ge \mu c_i > 0$, and
$\theta_i g(A - a_i) > 0$, so $w_i > \phi e_i(S)$. Multiply by $\theta_i$
and sum over $S$: $W(S) > \phi E(S) = g(A) W(S) E(S)$, that is
$W(S)[1 - g(A)E(S)] > 0$. If $1 - g(A)E(S) > 0$ then $W(S) > 0$, so
$\phi = g(A) W(S) > 0$ and $w_i > \phi e_i(S) \ge 0$. For $S = \{i\}$,
$E(S) = 1/g(a_i) - 1$ and $g(A)E(S) = 1 - g(a_i) < 1$.
(ii) If $\phi > 0$ then $W(S) > 0$, so $1 - g(A)E(S) > 0$ by (i), and then
$w_i > \phi e_i(S) \ge 0$. Sum the stability inequalities over $S$, using
$g(A - a_i) \le g(A - a^S_{\max})$ (as $A - a_i \ge A - a^S_{\max}$ and $g$
is nonincreasing) and the positive brackets:
$\mu C(S) \le g(A - a^S_{\max}) \sum_{i \in S} \theta_i [w_i - \phi e_i(S)]
= g(A - a^S_{\max}) [W(S) - \phi E(S)]$, and $W(S) = \phi / g(A)$ gives the
first bound. For the second, every term of $E(S)$ is nonnegative, so
$E(S) \ge 1/g(A) - 1/g(A - a^S_{\max})$, hence
$1 - g(A)E(S) \le g(A)/g(A - a^S_{\max})$ and the first bound is at most
$\phi$.
(iii) $W(S) \le \bar w\, \Theta(S)$ since $w_i \le \bar w$, $\theta_i > 0$
and $\bar w \ge 0$. Insert this in
$\mu C(S) \le g(A - a^S_{\max}) [W(S) - \phi E(S)] = g(A - a^S_{\max})\, W(S)\,
[1 - g(A)E(S)]$, where the bracket is positive by (ii), to get
$\mu C(S) \le \bar w X_g(S)$; by (ii) $w_i > 0$ on $S$, so
$X_g(S) \le \bar X_g$. The inequality
$g(A - a^S_{\max})[1 - g(A)E(S)] \le g(A)$ of (ii) gives
$X_g(S) \le \Theta(S) g(A) = Q(S)$, and $Q(S) \le \chi \bar G$ as in
Proposition~\ref{prop:sizefull}(i).
For the consequences, a nonempty optimum $S^\star$ at price $\mu$ is
deletion-stable at $\mu$ and has $w_i > 0$ on $S^\star$ by
Proposition~\ref{prop:inside} at price $\mu$, so $\phi > 0$ and (iii)
gives $\mu C(S^\star) \le a_g$; the empty optimum has cost $0$. The cap
never binds by the argument of Proposition~\ref{prop:sizefull}(iii), with
$a_g$ in place of $\bar w \chi \bar G$.

$\mathbb E \tau_2 = \frac1\pi (1 + 1/\psi_W)$ with

<!-- end lemmaa-6 -->
<!-- proof-17: body_md -->
(1) The policy cannot fire before the $k$-th sighting, whose time is a sum
of $k$ i.i.d.\ $\mathrm{Geom}(\pi)$ waiting times with mean $k/\pi$.
(2) Let $Z_1, Z_2, \dots$ be the i.i.d.\ $\mathrm{Geom}(\pi)$ times between
sightings ($Z_1$ is the first sighting time). The window count changes only
at sightings, and at the $j$-th sighting the window holds two sightings iff
$Z_j \le W - 1$. So the policy fires at the $j^*$-th sighting,
$j^* = \min\{j \ge 2 : Z_j \le W - 1\}$, and $\tau_2 = Z_1 + \dots + Z_{j^*}$.
Here $j^* - 1 \sim \mathrm{Geom}(\psi_W)$ with $\psi_W = \Pr(Z \le W - 1)$,
and $j^*$ is a stopping time for $(Z_j)_{j \ge 2}$ with finite mean, so
Wald's identity gives
$\mathbb E \tau_2 = \mathbb E Z_1 + \mathbb E[j^* - 1]\, \mathbb E Z
= 1/\pi + 1/(\psi_W \pi)$. Finally $\tau_1 = Z_1$.

Write $\beta_\varepsilon = 1 - (1 - \varepsilon)^{W - 1}$ for the chance
that a spurious sighting has another inside its window. We first show that
$k = 2$ is the unique minimizer under the two conditions
\begin{align*}
\text{(A)}\ & \Lambda_F \big[1 - (1 - \varepsilon)^T - T \varepsilon\, \beta_\varepsilon\big]
  > \Lambda_D / \psi_W, \\
\text{(B)}\ & \Lambda_F\, T \varepsilon\, \beta_\varepsilon < \Lambda_D (2 - 1/\psi_W),
\end{align*}
which need only $\varepsilon \in [0, 1]$ and $\Lambda_F \ge 0$, and then
that the band implies them.
By Lemmas~\ref{lem:fi}(1) and~\ref{lem:delay},
$\mathcal{L}(1) = \Lambda_F (1 - (1 - \varepsilon)^T) + \Lambda_D$. By
Lemma~\ref{lem:fi}(2) with $\Pr(\mathrm{Bin}(W-1, \varepsilon) \ge 1) =
\beta_\varepsilon$ and Lemma~\ref{lem:delay}(2),
$\mathcal{L}(2) \le \Lambda_F T \varepsilon\, \beta_\varepsilon
+ \Lambda_D (1 + 1/\psi_W)$. By Lemma~\ref{lem:delay}(1),
$\mathcal{L}(k) \ge \Lambda_D k \ge 3\Lambda_D$ for $k \ge 3$. Condition (A) says
exactly that the bound on $\mathcal{L}(2)$ is below $\mathcal{L}(1)$, and (B) that it is
below $3\Lambda_D$; so $\mathcal{L}(2) < \mathcal{L}(k)$ for every $k \ne 2$, $k \ge 1$.
For the band, use $(1 - \varepsilon)^T \le e^{-x_T}$ and
$\beta_\varepsilon \le (W - 1)\varepsilon \le x$: (A) follows from
$\Lambda_F (1 - e^{-x_T} - x_T x) > \Lambda_D / \psi_W$, and (B) from
$\Lambda_F x_T x < \Lambda_D (2 - 1/\psi_W)$. Dividing by
$\Lambda_F \varepsilon T > 0$, the second is $x < \omega (2 - 1/\psi_W)$, and
the first follows from $1 - e^{-x_T} \ge x_T - x_T^2/2$ and
$1 - x_T/2 - x > \omega/\psi_W$.

<!-- lemmaa-7: body_md -->
\label{lem:batch}
Assumptions~\ref{ass:mag} and~\ref{ass:tied}. Let $\mathcal G_0$ be the
event that for every $i \in [N]$, with $j$ the batch of $i$: (G1)
$N_i = \sum_{t \in I_j} \xi_{i,t} \ge n p_i / 2$; (G2)
$|\hat q_i - q_i(\mathcal J_j)| \le \epsilon_0$; (G3)
$|\bar Y_i - \mathbb E Y_i| \le 2 \epsilon_0$, where
$\mathbb E Y_i = w_i q_i(\mathcal J_j) = \beta_i g(A(\mathcal J_j))$. If
$n_0 \ge 8 \ln(6NT) / p_{\min}$, then $\Pr(\mathcal G_0^c) \le 5 / (6T)$,
and on $\mathcal G_0$, for every $j$ and $i \in \mathcal J_j$,
\[
|\hat A_j - A(\mathcal J_j)| \le \frac{s \epsilon_0}{\lambda \gamma_s} =: \epsilon_A,
\qquad
|\hat a_i - a_i| \le \frac{\epsilon_0}{\lambda g_s} + \frac{L_{1/g}}{\lambda} \epsilon_A =: \epsilon_a,
\qquad
|\hat\beta_i - \beta_i| \le \frac{2 \epsilon_0}{g_s} + L_{1/g}\, \epsilon_A =: \epsilon_\beta .
\]
<!-- end lemmaa-7 -->
<!-- proof-19: body_md -->
\emph{Probability.} The exploration schedule is fixed in advance, so for
$i \in \mathcal J_j$ the latent draws $(\xi_{i,t}, F_{i,t}, X_{i,t})_{t \in I_j}$
are i.i.d.\ across $t$ by Assumption~\ref{ass:obs}, and $X_{i,t}$ is
independent of $(\xi_{i,t}, F_{i,t})$ because it depends on the other
draws only through $\zeta_{i,t}$, which is independent of them; hence
$\mathbb E Y_i = p_i q_i(\mathcal J_j) r_i = w_i q_i(\mathcal J_j)$. By the
multiplicative Chernoff bound,
$\Pr(N_i < n p_i/2) \le e^{-n p_i/8} \le e^{-n_0 p_{\min}/8} \le 1/(6NT)$.
Conditional on the relevance draws, the compliance draws at the relevant
sessions are i.i.d.\ $\mathrm{Bern}(q_i(\mathcal J_j))$, so on (G1) Hoeffding's
inequality gives $\Pr(|\hat q_i - q_i(\mathcal J_j)| > \epsilon_0 \mid \xi)
\le 2 e^{-2 N_i \epsilon_0^2} \le 2 e^{-n p_i \epsilon_0^2}
\le 2 e^{-\ln(6NT)} = 1/(3NT)$. The $Y_{i,t}$ are i.i.d.\ in $[-1, 1]$, so
Hoeffding's inequality gives
$\Pr(|\bar Y_i - \mathbb E Y_i| > 2\epsilon_0) \le 2 e^{-2 n \epsilon_0^2}
\le 1/(3NT)$. A union bound over the $N$ items gives
$\Pr(\mathcal G_0^c) \le N (1/(6NT) + 2/(3NT)) = 5/(6T)$.

\emph{Load.} Write $A_j = A(\mathcal J_j) \le x_s$ and
$Q_j = Q(\mathcal J_j) = \lambda G(A_j)$, so $Q_j / \lambda \in [0, G(x_s)]$.
By (G2), $|\sum_{l} \hat q_l / \lambda - Q_j / \lambda| \le s \epsilon_0 / \lambda$,
and clipping to $[0, G(x_s)]$, which contains $Q_j/\lambda$, does not
increase the distance. On $[0, x_s]$, $G' \ge \gamma_s > 0$, so $G$ is
increasing there and $G^{-1}$ is $1/\gamma_s$-Lipschitz on
$[0, G(x_s)]$; hence $|\hat A_j - A_j| \le \epsilon_A$.

\emph{Items.} Since $\hat A_j, A_j \in [0, x_s]$, $g(\hat A_j) \ge g_s$ and
$|1/g(\hat A_j) - 1/g(A_j)| \le L_{1/g} \epsilon_A$. With
$q_i = q_i(\mathcal J_j) = \lambda a_i g(A_j) \le 1$,
\[
\frac{\hat q_i}{\lambda g(\hat A_j)} - a_i
= \frac{\hat q_i - q_i}{\lambda g(\hat A_j)}
+ \frac{q_i}{\lambda} \Big( \frac{1}{g(\hat A_j)} - \frac{1}{g(A_j)} \Big),
\]
of absolute value at most $\epsilon_a$; clipping to
$[0, a_{\max}] \ni a_i$ does not increase the error. Likewise, with
$|\mathbb E Y_i| \le 1$,
$\hat\beta_i - \beta_i = (\bar Y_i - \mathbb E Y_i)/g(\hat A_j)
+ \mathbb E Y_i\, (1/g(\hat A_j) - 1/g(A_j))$, of absolute value at most
$\epsilon_\beta$.

<!-- lemmaa-8: body_md -->
\label{lem:plugin}
Assumptions~\ref{ass:mag} and~\ref{ass:tied}. On $\mathcal G_0$, every
$S \subseteq [N]$ with $|S| \le n_B$ satisfies
$|\hat R(S) - R(S)| \le |S|\, \varepsilon_{\mathrm{SD}}(n_0)$.
<!-- end lemmaa-8 -->
<!-- proof-20: body_md -->
Both $A(S) = \sum_S a_i$ and $\hat A(S) = \sum_S \hat a_i$ lie in
$[0, x_B]$, and $|\hat A(S) - A(S)| \le |S| \epsilon_a$ by
Lemma~\ref{lem:batch}. Decompose
\[
\hat R(S) - R(S)
= g(\hat A(S)) \sum_{i \in S} (\hat\beta_i - \beta_i)
+ \big( g(\hat A(S)) - g(A(S)) \big) \sum_{i \in S} \beta_i .
\]
The first term is at most $|S| \epsilon_\beta$ in absolute value, as
$g \le 1$. For the second, $|\sum_S \beta_i| \le \lambda \sum_S |w_i| a_i
\le \lambda A(S)$, and by the definition of $K_g$,
$A(S)\, |g(\hat A(S)) - g(A(S))| \le K_g |\hat A(S) - A(S)|
\le K_g |S| \epsilon_a$. So
$|\hat R(S) - R(S)| \le |S| (\epsilon_\beta + \lambda K_g \epsilon_a)$, and
$\epsilon_\beta + \lambda K_g \epsilon_a = \varepsilon_{\mathrm{SD}}(n_0)$ by
the definitions of $\epsilon_a$, $\epsilon_\beta$ and $\epsilon_A$.

<!-- theorema-2: body_md -->
\label{thm:etcfull}
Assumptions~\ref{ass:obs}, \ref{ass:mag} and~\ref{ass:tied}; $m \ge 2$,
$B \ge c_{\max}$, $c_i > 0$, $\kappa > 0$, $N, L, T \ge 1$. If
$n_0 \ge 8 \ln(6NT) / p_{\min}$, then
\[
\mathrm{Reg}_T \le \Delta_{\max} L \Big( J + \Big\lceil \frac{s + n_B}{m} \Big\rceil \Big)
+ \Lambda_T(n_0),
\qquad
\Lambda_T(n_0) = \Delta_{\max} J n_0 + 2 n_B T\, \varepsilon_{\mathrm{SD}}(n_0) + \Delta_{\max} .
\]
<!-- end theorema-2 -->
<!-- proof-21: body_md -->
\emph{Feasibility.} A batch costs at most $s c_{\max} \le B$, the sets
played during the move are subsets of the last batch or of $\hat S$, and
$C(\hat S) \le B' \le B$; so every played file is feasible and every
session's expected regret is at most $\Delta_{\max}$. A batch switch uses
at most $2s \le m$ edits and each epoch of the move at most $m$.

\emph{Step 1 (decomposition).} Exploration lasts
$J L \lceil n_0/L \rceil < J(n_0 + L)$ sessions and the move
$L \lceil (s + n_B)/m \rceil$ sessions, in all at most
$\Delta_{\max} [L(J + \lceil (s + n_B)/m \rceil) + J n_0]$ regret. Every
later session costs $R(S^\star) - R(\hat S)$.

\emph{Step 2 (commit error).} Some optimal file lies in $\mathcal S'$ and
every $S \in \mathcal S'$ has $|S| \le n_B$. On $\mathcal G_0$,
Lemma~\ref{lem:plugin} gives
$|\hat R(S) - R(S)| \le n_B \varepsilon_{\mathrm{SD}}(n_0)$ for all
$S \in \mathcal S'$, so
$R(S^\star) - R(\hat S) \le [R(S^\star) - \hat R(S^\star)]
+ [\hat R(S^\star) - \hat R(\hat S)] + [\hat R(\hat S) - R(\hat S)]
\le 2 n_B \varepsilon_{\mathrm{SD}}(n_0)$, the middle bracket being
$\le 0$ by the choice of $\hat S$.

\emph{Step 3 (expectation).} The commit phase lasts at most $T$ sessions,
so its regret is at most $2 n_B T \varepsilon_{\mathrm{SD}}(n_0)$ on
$\mathcal G_0$ and at most $T \Delta_{\max}$ off $\mathcal G_0$, whose
probability is at most $5/(6T) \le 1/T$ by Lemma~\ref{lem:batch}; in
expectation this adds at most $\Delta_{\max}$. Summing with Step~1 gives
the bound.

<!-- corollarya-2: body_md -->
\label{cor:tuned}
In the setting of Theorem~\ref{thm:etcfull}, let
$n_{\min} = \lceil 8 \ln(6NT) / p_{\min} \rceil$,
$C_1 = 2 n_B \big[ (2 + K_g)/g_s + (1 + K_g) L_{1/g}\, s / (\lambda \gamma_s) \big]
\sqrt{\ln(6NT) / p_{\min}}$, so that
$2 n_B \varepsilon_{\mathrm{SD}}(n_0) = C_1 n_0^{-1/2}$, and
$n_0 = \max\{n_{\min}, \lceil (T C_1 / (2 J \Delta_{\max}))^{2/3} \rceil\}$,
reading $y/0 = 0$. Then
\[
\Lambda_T(n_0) \le 1.9\, (J \Delta_{\max})^{1/3} (C_1 T)^{2/3}
+ J \Delta_{\max} (n_{\min} + 1) + \Delta_{\max} .
\]
<!-- end corollarya-2 -->
<!-- proof-22: body_md -->
If $\Delta_{\max} = 0$ then $\mathrm{Reg}_T \le T \Delta_{\max} = 0$ and there is
nothing to prove. Otherwise the first two terms of $\Lambda_T(n_0)$ are
$\mathfrak a n_0 + \mathfrak b n_0^{-1/2}$ with $\mathfrak a = J \Delta_{\max}$
and $\mathfrak b = C_1 T$; put $n^* = (\mathfrak b/(2\mathfrak a))^{2/3}$, so
$\mathfrak a n^* = 2^{-2/3} \mathfrak a^{1/3} \mathfrak b^{2/3}$ and
$\mathfrak b (n^*)^{-1/2} = 2^{1/3} \mathfrak a^{1/3} \mathfrak b^{2/3}$.
If $n_0 = \lceil n^* \rceil \ge n_{\min}$, then
$\mathfrak a n_0 + \mathfrak b n_0^{-1/2} \le \mathfrak a (n^* + 1)
+ \mathfrak b (n^*)^{-1/2} = 3 \cdot 2^{-2/3} \mathfrak a^{1/3} \mathfrak b^{2/3}
+ \mathfrak a \le 1.9\, \mathfrak a^{1/3}\mathfrak b^{2/3} + \mathfrak a$.
If $n_0 = n_{\min} > n^*$, then $\mathfrak a n_0 = \mathfrak a n_{\min}$ and
$\mathfrak b n_0^{-1/2} < \mathfrak b (n^*)^{-1/2} \le 1.9\, \mathfrak a^{1/3} \mathfrak b^{2/3}$.
In both cases the two terms are at most
$1.9\, \mathfrak a^{1/3} \mathfrak b^{2/3} + \mathfrak a (n_{\min} + 1)$.

The hypotheses of Theorem~\ref{thm:etc} are those of
Theorem~\ref{thm:etcfull}, with $n_B$ the largest size of a file in
$\mathcal S'$, so the bound is Theorem~\ref{thm:etcfull}. Under the
tuning of Corollary~\ref{cor:tuned}, $C_1$ is
$O(\sqrt{\ln(NT)})$ and $n_{\min}$ is $O(\ln(NT))$ with the other
parameters fixed, so
$\Lambda_T = O(T^{2/3} (\ln NT)^{1/3})$. For the edit-cap term, if
$m \le 2B/c_{\max}$ then $s = \lfloor m/2 \rfloor \ge m/3$, so
$J \le 3N/m + 1$, and $n_B \le N$ gives

<!-- proof-24: body_md -->
\emph{Instance.} Fix $c \in (0, B]$, $p \in (0, 1]$, $\theta > 0$ and
$a > 0$ with $\theta g(a) < 1$ and $\kappa c < p \theta g(a)$; such
values exist since $c$ may be taken small. All items share $c$, $p$,
$\theta_i = \theta$ and $a_i = a$; harm $h_i = 0$; gap detection
$\delta$; under instance $j$ the gains are $r_j = 1$ and $r_i = 0$ for
$i \ne j$, observed without noise ($X_{i,t} \equiv r_i$). Then
$R(S) = p\, \theta\, g(|S| a)\, \mathbf 1\{j \in S\} - \kappa c |S|$. If
$j \in S$ this is at most $p \theta g(a) - \kappa c = R(\{j\})$, as $g$
is nonincreasing, and if $j \notin S$ it is $-\kappa c |S| \le 0 < R(\{j\})$.
So $S^\star_j = \{j\}$, and $R(S^\star_j) - R(S_t) \ge \Delta$ whenever
$j \notin S_t$; also $R(S) \ge -\kappa C(S) \ge -\kappa B$ for feasible
$S$, so $\Delta_{\max} \le \Delta + \kappa B$.

\emph{Reduction.} Let instance $0$ have $r_i = 0$ for all $i$. For a file
$S \not\ni j$, the observation law of a session with file $S$ is the same
under instance $j$ and instance $0$: in-file items have the same
compliance $\theta g(|S| a)$ and the same gain $0$ under both, and $j$
contributes only $G_{j,t} \sim \mathrm{Bern}(\delta p)$. This holds for every
joint law of the flags within a session, since the flag laws do not
depend on the gains. Let $\sigma_j = \min\{t : j \in S_t\}$ be the first
session that tries $j$. The law of a history
$(S_1, O_1, \dots, S_t, O_t)$ is the product of the policy kernels
$\mathsf P(S_s \mid \text{past})$ and the session laws
$\mathsf K(O_s \mid S_s)$; on the event $\{\sigma_j > t\}$ every factor
agrees under instances $j$ and $0$, so
$\Pr_j(\sigma_j > t) = \Pr_0(\sigma_j > t)$.

\emph{Counting.} From $S_0 = \emptyset$ with at most $m$ additions per
epoch, $|\bigcup_{s \le t} S_s| \le \sum_{s \le t} |S_s \setminus S_{s-1}|
\le m\, e(t)$ almost surely, so
$\sum_j \Pr_0(\sigma_j \le t) \le \mathbb E_0 |\bigcup_{s \le t} S_s|
\le m\, e(t)$, and
$\sum_j \Pr_j(j \notin S_t) \ge \sum_j \Pr_0(\sigma_j > t) \ge N - m\, e(t)$.

\emph{Summing.} $\mathrm{Reg}_T(j) \ge \Delta \sum_{t \le T} \Pr_j(j \notin S_t)$,
so, using $T \ge L n_m$ and $N - m e \ge 0$ for $e \le n_m$,
\[
\sum_j \mathrm{Reg}_T(j) \ge \Delta \sum_{e = 1}^{n_m} L (N - m e)
= \Delta L \Big(N n_m - \frac{m n_m (n_m + 1)}{2}\Big)
\ge \Delta L\, \frac{N (n_m - 1)}{2},
\]
because $m n_m \le N$. The maximum over $j$ is at least the average,
$(\Delta L/2)(n_m - 1)$. For $N \ge 4m$, $n_m - 1 > N/m - 2 \ge N/(2m)$.
