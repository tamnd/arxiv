<!-- assumption-1: body_md -->
\label{ass:main}
The operator $F:\mathbb R^d\to\mathbb R^d$ in \eqref{eq:problem} is monotone, $G:\mathbb R^d\rightrightarrows\mathbb R^d$ is maximally monotone, and
\begin{equation}\label{eq:msl}
\frac1n\sum_{i=1}^n\left\lVertF_i(x)-F_i(y)\right\rVert^2\le L^2\left\lVertx-y\right\rVert^2
\qquad\text{for all }x,y\in\mathbb R^d.
\end{equation}
There is a point $u^*\in\operatorname{zer}(F+G)$ satisfying $\left\lVertu_0-u^*\right\rVert\le R$, where positive bounds $L,R$ are available to the algorithm.
<!-- end assumption-1 -->
<!-- theorem-1: body_md -->
\label{thm:upper}
Under Assumption~\ref{ass:main}, Algorithm~\ref{alg:main} returns a computable pair $(y,g)$ satisfying $g\in G(y)$ and
\begin{equation}\label{eq:upper-residual}
\left\lVertF(y)+g\right\rVert_{ L^2}\le\frac{29}{32}\varepsilon<\varepsilon.
\end{equation}
Its expected numbers of component evaluations and resolvent evaluations are each at most
\begin{equation}\label{eq:upper-cost}
C\left(n+\frac{\sqrt n LR}{\varepsilon}\right)
\end{equation}
for a universal constant $C$. No logarithm of a problem parameter is hidden in $C$.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:lower}
Let $n\ge1$, $L,R>0$, and $0<\varepsilon\le LR/2$. Any randomized linear-span component-oracle algorithm guaranteeing expected residual at most $\varepsilon$ on every instance satisfying Assumption~\ref{ass:main} has worst-case expected component cost $M$ satisfying
\begin{equation}\label{eq:lower}
M\ge\frac1{64}\left(n+\frac{\sqrt n LR}{\varepsilon}\right).
\end{equation}
The lower bound holds for $G=0$ and affine monotone components.
<!-- end theorem-2 -->
<!-- corollary-1: body_md -->
\label{cor:optimal}
For $0<\varepsilon\le LR/2$, the worst-case expected component complexity is
\[
\Theta\left(n+\frac{\sqrt n LR}{\varepsilon}\right)
\]
within the randomized linear-span component-oracle model that also permits the specified $G$-resolvent outputs. In particular, the upper-bound algorithm is linear-span on the lower-bound subclass $G=0$.

<!-- lemma-1: body_md -->
\label{lem:early}
Under Assumption~\ref{ass:main}, let $z$ be the zero of $F+B_{\lambda,b}$. There is a routine \textnormal{\textsc{VRContract}} starting at $a$ such that
\[
\mathbb E[\left\lVerta^+-z\right\rVert^2\mid a,b]\le\frac1{16}\left\lVerta-z\right\rVert^2
\]
at expected cost $\mathcal O(n+\sqrt n L/\lambda)$. The resolvent count satisfies the same order bound.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Apply \citet[Theorem 4.6]{cad24} with $A=F$, $B=B_{\lambda,b}$, and strong monotonicity parameter $\lambda$. For $n\ge2$, initialize $u^0=w^0=w^{-1}=a$, set
\[
p=\frac1n,\qquad \eta=\frac{\sqrt{p(1-p)}}{2L},\qquad
M_\lambda=\left\lceil14\left(n+\frac{\sqrt n L}{\lambda}\right)\log(4\sqrt6)\right\rceil,
\]
and for $t=0,\ldots,M_\lambda-1$ draw a uniform component $i_t$ and compute
\begin{align*}
u^{t+1}&=J_{\eta B_{\lambda,b}}\left((1-p)u^t+pw^t
-\eta\big[F(w^t)-F_{i_t}(w^{t-1})+F_{i_t}(u^t)\big]\right),\\
w^{t+1}&=\begin{cases}u^{t+1},&\text{with probability }p,\\w^t,&\text{with probability }1-p.\end{cases}
\end{align*}
The refresh decision is independent of the component draw and past history. Cache $F(w^t)$, computing a full average only when the reference point changes. Return $a^+=u^{M_\lambda}$. The cited theorem's linear estimate gives the stated distance contraction. One full initialization and at most $2+pn=3$ expected component calls per iteration give the cost claim. The same estimate applies after conditioning on the data of an outer stage. If $n=1$, use two identical copies of $F_1$; this only changes universal constants. If $a=z$, the initialized method remains at the root, so no division by a zero initial distance is needed.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:centered}
Fix $a,v,b\in\mathbb R^d$ and $\lambda>0$. Let $e=v-F(a)$ and let $z$ be the zero of
\[
\widetilde H=F+B_{\lambda,b}+e.
\]
The iteration \eqref{eq:centered} with parameters \eqref{eq:inner-parameters} satisfies
\[
\mathbb E[\left\lVertw_N-z\right\rVert^2\mid a,v,b]\le\frac1{64}\left\lVerta-z\right\rVert^2.
\]
It uses at most $2N$ component evaluations and $N$ resolvent evaluations, where $N=\mathcal O(1+(L/\lambda)^2)$.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Condition on the stage data $a,v,b$ throughout. Write $D=\left\lVerta-z\right\rVert$ and $r=\tau\lambda$. At iteration $t$, write $Q_t=F(w_t)+e+\xi_t$. Conditional on the inner history $\mathcal F_t$,
\begin{equation}\label{eq:noise-bound}
\mathbb E[\xi_t\mid\mathcal F_t]=0,\qquad
\mathbb E[\left\lVert\xi_t\right\rVert^2\mid\mathcal F_t]\le L^2\left\lVertw_t-a\right\rVert^2.
\end{equation}
Indeed, $\xi_t$ is the centered version of $F_{i_t}(w_t)-F_{i_t}(a)$, and removing its mean cannot increase its second moment.

Since $z=J_{\tau B_{\lambda,b}}(z-\tau(F(z)+e))$, the resolvent contraction gives
\begin{align}
\mathbb E[\left\lVertw_{t+1}-z\right\rVert^2\mid\mathcal F_t]
&\le\frac{1}{(1+r)^2}\mathbb E\left[\left\lVertw_t-z-\tau(F(w_t)-F(z))-\tau\xi_t\right\rVert^2\mid\mathcal F_t\right]\notag\\
&\le\frac{(1+\tau^2L^2)\left\lVertw_t-z\right\rVert^2+\tau^2L^2\left\lVertw_t-a\right\rVert^2}{(1+r)^2}\notag\\
&\le\frac{(1+3\tau^2L^2)\left\lVertw_t-z\right\rVert^2+2\tau^2L^2D^2}{(1+r)^2}.
\label{eq:centered-recursion}
\end{align}
The second inequality uses monotonicity and Lipschitz continuity of the mean $F$, as well as \eqref{eq:noise-bound}; it does not use component monotonicity.

The choice of $\tau$ yields
\[
0<r\le\frac1{256},\qquad \tau^2L^2\le\frac r{256}.
\]
Thus $3\tau^2L^2\le3r/256\le r-r^2-r^3$, implying
\[
\frac{1+3\tau^2L^2}{(1+r)^2}\le1-r,
\qquad
\frac{2\tau^2L^2}{(1+r)^2}\le\frac r{128}.
\]
Taking expectations in \eqref{eq:centered-recursion} and iterating gives
\[
\mathbb E\left\lVertw_N-z\right\rVert^2
\le\left((1-r)^N+\frac1{128}\right)D^2
\le\frac{D^2}{64},
\]
where $(1-r)^N\le e^{-rN}\le1/128$. Finally,
\[
\frac1r=256\left(1+\frac{L^2}{\lambda^2}\right).
\]
Each sampled difference costs at most two component calls. The logarithm in $N$ is the fixed numerical constant $\log128$.
<!-- end proof-2 -->
<!-- lemma-3: body_md -->
\label{lem:invariants}
Let $D_s=\left\lVertx_s-z_s\right\rVert_{ L^2}$, $\delta_s=\left\lVertx_{s+1}-z_s\right\rVert_{ L^2}$, and $\bar D_s=R4^{-s}$. Then
\begin{equation}\label{eq:stage-bounds}
D_s\le\bar D_s,\qquad
\delta_s\le\begin{cases}D_s/4,&s<K,\\D_s/8,&s\ge K,\end{cases}
\end{equation}
and $\left\lVerte_S\right\rVert_{ L^2}\le\mu R/4$.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
The initial root is $z_0=J_{T/\mu}(u_0)$. Since the displacement map $I-J_{T/\mu}$ is nonexpansive and vanishes at $u^*$,
\[
D_0=\left\lVertu_0-J_{T/\mu}(u_0)\right\rVert\le\left\lVertu_0-u^*\right\rVert\le R.
\]
Lemmas~\ref{lem:early} and \ref{lem:centered}, applied conditional on each stage's data and then averaged, give the displayed bounds on $\delta_s$.

For $s<K$, including the transition $s=K-1$ if it occurs, the update of $b_s$ gives
\[
\widetilde H_{s+1}=\widetilde H_s+\lambda_s(I-x_{s+1}),\qquad
z_{s+1}=J_{\widetilde H_s/\lambda_s}(x_{s+1}).
\]
Here and below $I-a$ denotes $x\mapsto x-a$. The point $z_s$ is a fixed point of $J_{\widetilde H_s/\lambda_s}$. Nonexpansiveness of the corresponding displacement map yields, pathwise,
\[
\left\lVertx_{s+1}-z_{s+1}\right\rVert\le\left\lVertx_{s+1}-z_s\right\rVert.
\]
Consequently $D_{s+1}\le\delta_s\le D_s/4$.

For $K\le s<S$, define $\Delta_s=e_{s+1}-e_s$. The estimator update implies
\begin{equation}\label{eq:delta-definition}
\Delta_s=\frac1{m_s}\sum_{j=1}^{m_s}
\big[F_{i_{s,j}}(x_{s+1})-F_{i_{s,j}}(x_s)\big]
-\big[F(x_{s+1})-F(x_s)\big].
\end{equation}
Let $\mathcal F_{s,+}$ be the history through $x_{s+1}$ before this fresh batch. Then
\[
\mathbb E[\Delta_s\mid\mathcal F_{s,+}]=0,\qquad
\mathbb E[\left\lVert\Delta_s\right\rVert^2\mid\mathcal F_{s,+}]
\le\frac{L^2}{m_s}\left\lVertx_{s+1}-x_s\right\rVert^2.
\]
Since $\left\lVertx_{s+1}-x_s\right\rVert_{ L^2}\le D_s+\delta_s\le9D_s/8$, the batch size gives
\begin{equation}\label{eq:bias-increment}
\left\lVert\Delta_s\right\rVert_{ L^2}\le\frac{9L}{8\sqrt{m_s}}D_s
\le\frac{\lambda_sD_s}{8}\le\frac{\lambda_s\bar D_s}{8}.
\end{equation}
Introduce the intermediate root
\[
\bar z_{s+1}=J_{\widetilde H_s/\lambda_s}(x_{s+1}).
\]
Its distance from $x_{s+1}$ is at most $\left\lVertx_{s+1}-z_s\right\rVert$. Moreover,
\[
\widetilde H_{s+1}
=\widetilde H_s+\lambda_s(I-x_{s+1})+\Delta_s.
\]
The operator before the constant shift $\Delta_s$ is $2\lambda_s$-strongly monotone. The perturbation estimate in Appendix~\ref{app:vr} therefore gives
\[
\left\lVertz_{s+1}-\bar z_{s+1}\right\rVert\le\frac{\left\lVert\Delta_s\right\rVert}{2\lambda_s}.
\]
By Minkowski's inequality,
\[
D_{s+1}\le\delta_s+\frac{\left\lVert\Delta_s\right\rVert_{ L^2}}{2\lambda_s}
\le\frac{D_s}{8}+\frac{D_s}{16}\le\frac{D_s}{4}.
\]
Induction proves \eqref{eq:stage-bounds}. Since $e_K=0$, one more use of Minkowski's inequality gives
\begin{equation}\label{eq:total-bias}
\left\lVerte_S\right\rVert_{ L^2}\le\sum_{s=K}^{S-1}\left\lVert\Delta_s\right\rVert_{ L^2}
\le\frac{\mu R}{8}\sum_{s=K}^{S-1}2^{-s}
\le\frac{\mu R}{4}.
\end{equation}
This argument does not assume independence between stage roots or between accumulated stage errors. In particular, conditional centering of $\Delta_s$ is not used to set $e_s$ to zero.

<!-- lemma-4: body_md -->
\label{lem:regularizer}
For $x=x_{S+1}$,
\[
\left\lVert\lambda_Sx-b_S\right\rVert_{ L^2}\le\frac52\mu R.
\]
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Lemma~\ref{lem:invariants} gives
\[
\left\lVertx_{s+1}-x_s\right\rVert_{ L^2}\le\frac54R4^{-s}.
\]
Summing these increments from $s$ onward yields
\begin{equation}\label{eq:tail-displacements}
\left\lVertx-x_s\right\rVert_{ L^2}\le\frac53R4^{-s}.
\end{equation}
The definitions of $\lambda_s,b_s$ imply the exact identity
\[
\lambda_Sx-b_S=\mu(x-u_0)+\sum_{j=0}^{S-1}\lambda_j(x-x_{j+1}).
\]
Using \eqref{eq:tail-displacements},
\[
\left\lVert\lambda_Sx-b_S\right\rVert_{ L^2}
\le\frac53\mu R+\frac53\mu R\sum_{j=0}^{S-1}2^j4^{-(j+1)}
\le\frac53\mu R+\frac56\mu R=\frac52\mu R.
\]

<!-- lemma-5: body_md -->
\label{lem:chain}
If $z$ is supported on the first $q<m$ coordinates, then
\[
\left\lVertDz-ae_1\right\rVert^2\ge\frac{a^2}{q+1}.
\]
Equality is attained by $z_j=a(q+1-j)/(q+1)$ for $1\le j\le q$ and zero elsewhere.
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
The first $q+1$ coordinates of $Dz-ae_1$ sum to $-a$ because the differences telescope and $z_{q+1}=0$. Cauchy--Schwarz gives the lower bound. For the displayed $z$, these $q+1$ coordinates are each $-a/(q+1)$, and the remaining coordinates vanish.
