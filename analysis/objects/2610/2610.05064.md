<!-- definition-1: body_md -->
\label{def:oriented_varifolds}
    A finite non-negative Radon measure $V$ on $\Omega\times S^{d-1}$ is called an oriented varifold on $\Omega$.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
Let $V$ be an oriented varifold. Then the weight measure $|\mu|$ is the finite non-negative Radon measure on $\Omega$ defined by
\begin{equation}
\label{eq:intro-weight-measure}
|\mu| \coloneqq (\pi_\Omega)_\# V,
\end{equation}
with the projection $\pi_\Omega(x,n)\coloneqq x$, that is, $|\mu|(A)=V(A\times S^{d-1})$ for every Borel set $A\subset\Omega$.
Likewise, the current $\vv{\mu}$ associated with $V$ is the finite $\mathbb{R}^d$-valued Radon measure on $\Omega$ defined by
\begin{equation}
\label{eq:intro-first-moment-current}
\begin{aligned}
\vv{\mu}(\varphi)
\coloneqq
\int_{\Omega \times S^{d-1}} \varphi(x) \cdot n \d V(x,n)
&&
\text{for every compactly supported } \varphi \in C(\Omega;\mathbb{R}^d).
\end{aligned}
\end{equation}
<!-- end definition-2 -->
<!-- remark-1: body_md -->
\label{rem:hutchinson}
The preceding definition is a measure-theoretic version of the current associated with an oriented varifold by \citet{Hutchinson86}.
In the rectifiable hypersurface case, an oriented integral varifold may be written, up to the identification between oriented tangent planes and unit normals, in the form
\(V = \mathcal{H}^{d-1}\llcorner\Gamma \otimes \pab{ m_+(x)\delta_{n(x)} + m_-(x)\delta_{-n(x)} }\), where $\Gamma$ is a $(d-1)$-rectifiable set with unit normal $n$ and \(m_\pm(x)\in\mathbb{Z}_{\ge0}\) are the multiplicities of the two orientations.
Then
\(|\mu| = (m_+ + m_-)\mathcal{H}^{d-1}\llcorner \Gamma\) and \(\vv{\mu} = (m_+ - m_-)n\mathcal{H}^{d-1}\llcorner \Gamma\).
The weight measure thus has the total multiplicity $m_+ + m_-$, whereas the current has the signed excess $m_+ - m_-$.

The measure $\kappa_x$ may charge both orientations at the same point.

<!-- definition-7: body_md -->
\label{def:coherent-perimeter-functional}
For $\sigma > 0$, the weight measure and the current are mollified by convolution with the kernel $\psi_{\sigma}$,
\[
\begin{aligned}
    (\psi_{\sigma} \ast |\mu|)(x) \coloneqq \int_{\mathbb{R}^d}\psi_{\sigma}(x-y)\d |\mu|(y),
&&
(\psi_{\sigma} \ast \vv{\mu})(x) \coloneqq \int_{\mathbb{R}^d}\psi_{\sigma}(x-y)\d\vv{\mu}(y),
\end{aligned}
\]
and the \emph{visibility} is defined by their ratio:
\[
q_\sigma[V](x)
\coloneqq
\begin{cases}
\dfrac{\abs{(\psi_{\sigma} \ast \vv{\mu})(x)}}{(\psi_{\sigma} \ast |\mu|)(x)},&\text{if }(\psi_{\sigma} \ast |\mu|)(x) > 0,\\
0&\text{otherwise}.
\end{cases}
\]
Since $(\psi_{\sigma} \ast \vv{\mu})(x)/(\psi_{\sigma} \ast |\mu|)(x)$ is the $\psi_{\sigma}(x-\cdot)\d V$-weighted average of the orientations $n$, the visibility is the length of the mean of the local unit normals (for fixed $x$, the mean resultant length of directional statistics, see \citealp{MardiaJupp00}). It equals $1$ where all normals near $x$ agree, and it vanishes where the mollified current vanishes, for instance where two pieces of the interface with opposite normals cancel exactly within the support of $\psi_{\sigma}(x-\cdot)$ (a hidden boundary in the sense of \zcref{def:hidden-boundary}).
The \emph{visible perimeter} functional is
\[
P_\sigma(V) \coloneqq \int_\Omega q_\sigma[V](x) \d |\mu|(x).
\]

<!-- assumption-2: body_md -->
\label{ass:piecewise-regular-support}
There exist finitely many compact sets $\Gamma_1,\dots,\Gamma_L \subset \Omega$, which we call \emph{branches}, and a set $\Sigma \subset \Omega$ which is a finite union of compact $C^1$ submanifolds of dimension $d-2$ (a finite set if $d=2$), such that
\(
\Gamma = \bigcup_{l=1}^L \Gamma_l\) and
\(\Gamma_l\cap\Gamma_{l'} \subset \Sigma\) for \(l\ne l'\).
Moreover,
\begin{enumerate}[label=\textup{(\roman*)},ref=(\roman*)]
\item each $\Gamma_l$ is an embedded $(d-1)$-dimensional $C^{1,\alpha}$ manifold with (possibly empty) boundary $\partial\Gamma_l\subset\Sigma$, for a fixed exponent $\alpha \in (0,1]$,
\item\label{item:weight-measure-hausdorff} the weight measure is
\(|\mu| = \mathcal{H}^{d-1} \llcorner \Gamma\),
\item\label{item:branch-separation} there exist $c_{\textup{s}}\in(0,1]$ and $r_{\textup{sep}}>0$ such that, with $d_\Sigma(y)\coloneqq\min\Bab{\operatorname{dist}(y,\Sigma),r_{\textup{sep}}}$ for $y\in\mathbb{R}^d$ ($d_\Sigma\equiv r_{\textup{sep}}$ if $\Sigma=\emptyset$),
\begin{equation}
\label{eq:branch-separation-global}
\begin{aligned}
\operatorname{dist}(x,\Gamma_{l'}) \ge c_{\textup{s}}d_\Sigma(x)
&&
\text{for every } x\in\Gamma_l \text{ and every } l'\ne l.
\end{aligned}
\end{equation}
\end{enumerate}
\zcref{item:branch-separation} rules out distinct branches meeting tangentially at $\Sigma$. Since the branches are compact and meet only in $\Sigma$, it suffices to require \zcref{eq:branch-separation-global}, with a possibly smaller $c_{\textup{s}}$, for $x$ with $\operatorname{dist}(x,\Sigma)\le r_{\textup{sep}}$. For instance, it holds there if $\Gamma_l\pitchfork\Gamma_{l'}$ for all $l\ne l'$.
When $\Sigma=\emptyset$ (for instance, when $\Gamma$ is a finite union of disjoint closed surfaces), it only states that distinct branches are at positive distance from each other.
\zcref{item:weight-measure-hausdorff} fixes the multiplicity of $\Gamma$ to $1$, and hidden boundaries are carried by $\kappa_x$ as in \zcref{rem:hidden-boundary-samples-allowed}.

<!-- lemma-1: body_md -->
\label{lem:ahlfors-regularity}
Under \zcref{ass:piecewise-regular-support}, there exist constants $c_\Gamma, C_\Gamma, r_\Gamma > 0$ such that
\begin{equation}
\label{eq:ahlfors-regularity}
c_\Gamma r^{d-1} \le |\mu|(B_r(x)) \le C_\Gamma r^{d-1}
\end{equation}
for every $x \in \Gamma$ and every $r \in (0,r_\Gamma]$.
This property is called the Ahlfors regularity of $|\mu|$.
Moreover, there exists a constant $C_\Sigma > 0$ such that, for every $r \in (0,r_\Gamma]$,
\begin{equation}
\label{eq:singular-neighbourhood-measure}
|\mu|\pab{\Setdef x \in \Gamma | \operatorname{dist}(x,\Sigma) \le r \Setdef} \le C_\Sigma r.
\end{equation}
In particular, $|\mu|(\Sigma)=0$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
A compact $C^1$ manifold $\Gamma_l$ with boundary has $(d-1)$-dimensional density bounded above and below at small scales, uniformly in the point (at a boundary point the ball $B_r(x)$ still contains a half-ball portion of $\Gamma_l$), and since there are finitely many manifolds the constants can be chosen uniformly in $l$.
A ball $B_r(x)$ centred at $x\in\Gamma$ meets at most $L$ of the manifolds, so that the upper bound holds for $\Gamma$ after changing the constant, and the lower bound holds because $x$ belongs to at least one $\Gamma_l$.
This proves \zcref{eq:ahlfors-regularity}.
For \zcref{eq:singular-neighbourhood-measure}, the $r$-neighbourhood of a compact $C^1$ submanifold of dimension $d-2$ is covered by $Cr^{-(d-2)}$ balls of radius $2r$. A ball of radius $2r$ that meets $\Gamma$ is contained in a ball of radius $4r$ centred at a point of $\Gamma$, which carries $|\mu|$-mass at most $C_\Gamma(4r)^{d-1}$ by \zcref{eq:ahlfors-regularity} if $4r\le r_\Gamma$. Hence the total mass is at most $4^{d-1}CC_\Gamma r$ for $r\le r_\Gamma/4$, and for $r_\Gamma/4<r\le r_\Gamma$ the bound holds once $C_\Sigma\ge4|\mu|(\Gamma)/r_\Gamma$.
For $d=2$, $\Sigma$ is a finite set and the bound follows directly from \zcref{eq:ahlfors-regularity}.

\end{aligned}

q_{\sigma,\delta,N}(x)

<!-- lemma-5: body_md -->
\label{lem:density-at-sample}
Under \zcref{ass:kernel,ass:piecewise-regular-support,ass:sampling-density,ass:marked-iid-model}, there exists $C>0$ such that, for every $\delta\in(0,\delta_0]$ and every $N\ge2$,
\[
\mathbb E \bab{\abs{\theta_{\delta,N}(X_1)-\theta(X_1)}}
\le
C\pab{\delta^{\alpha\wedge\gamma}+\frac{1}{\sqrt{N\delta^{d-1}}}+\frac{1}{N\delta^{d-1}}}.
\]

<!-- proof-7: body_md -->
Conditionally on $X_1=x$, we have $\theta_{\delta,N}(x)=\psi_{}(0)/(NZ_{\psi_{}}^{(d-1)}\delta^{d-1})+\frac{N-1}{N}\widetilde\theta(x)$, where $\widetilde\theta(x)\coloneqq((N-1)Z_{\psi_{}}^{(d-1)}\delta^{d-1})^{-1}\sum_{j=2}^N\psi_{}((x-X_j)/\delta)$ is the average of $N-1$ independent random variables, each with mean $\theta_\delta(x)$ and, by \zcref{eq:ahlfors-regularity}, with variance at most $C\delta^{1-d}$ (as in the proof of \citealp[Proposition 3.5]{Boricaud_Buet25}).
Hence $\widetilde\theta(x)$ has mean $\theta_\delta(x)$ and variance at most $C/((N-1)\delta^{d-1})$, and the Cauchy--Schwarz inequality with $N-1\ge N/2$ gives
\[
\mathbb E\left[ 
  \IfBooleanTF{\theta_{\delta,N}(X_1)-\theta(X_1)}{\vab*{#3}}{ 
    \IfValueTF{\,}{\vab[\,]{#3}}{ 
      \mathchoice{\vab{#3}}{\vab*{#3}}{\vab*{#3}}{\vab*{#3}}}}\middle|\,X_1=x\right]
\le
\frac{C}{N\delta^{d-1}}
+\frac{\theta_\delta(x)}{N}
+\frac{C}{\sqrt{N\delta^{d-1}}}

Integrating with respect to $\mu_X(\d x)$, using $\theta_\delta\le C$ on $\Gamma$ and \zcref{lem:density-bias} gives the claim.

<!-- lemma-6: body_md -->
\label{lem:weighted-empirical}
Under \zcref{ass:piecewise-regular-support,ass:sampling-density,ass:marked-iid-model}, let $G>0$ and let $g\colon\Gamma\to[0,G]$ be Borel measurable.
Then there exists $C>0$, depending only on $\Gamma$ and $d$, such that
\[
\begin{aligned}
\mathbb E \bab{d_{\textup{BL},\overline\Omega}\pab{\frac1N\sum_{i=1}^Ng(X_i)\delta_{X_i},\ g\mu_X}}
\le
CGN^{-\nicefrac{1}{(d+1)}}
&&
\text{for every } N\in\mathbb{Z}_{>0}.
\end{aligned}
\]

Write $\lambda_N^g\coloneqq\frac1N\sum_ig(X_i)\delta_{X_i}$ and $\rho\coloneqq N^{-\nicefrac{1}{(d+1)}}$.
If $\rho>r_\Gamma$, then $d_{\textup{BL},\overline\Omega}(\lambda_N^g,g\mu_X)\le\lambda_N^g(\Gamma)+G\le2G\le2Gr_\Gamma^{-1}N^{-\nicefrac{1}{(d+1)}}$, so we assume $\rho\le r_\Gamma$.
Among the half-open cubes $\rho(k+[0,1)^d)$, $k\in\mathbb{Z}^d$, which partition $\mathbb{R}^d$, let $\mathcal Q$ be the family of those that meet $\Gamma$, and for $Q\in\mathcal Q$ pick $x_Q\in Q\cap\Gamma$.
The balls $B_\rho(x_Q)$ are contained in the concentric cubes of side $3\rho$, each point of $\mathbb{R}^d$ lies in at most $3^d$ of these enlarged cubes, and $|\mu|(B_\rho(x_Q))\ge c_\Gamma\rho^{d-1}$ by \zcref{eq:ahlfors-regularity}. Hence the number $\#\mathcal Q$ of cubes in $\mathcal Q$ satisfies $\#\mathcal Q\le3^d|\mu|(\Gamma)/(c_\Gamma\rho^{d-1})$.
Let $\varphi\in\operatorname{BL}(\overline\Omega)$ with $ 
  \IfBooleanTF{\varphi}{\Vab*{#3}}{ 
    \IfValueTF{_}{\Vab[_]{#3}}{ 
      \mathchoice{\Vab{#3}}{\Vab*{#3}}{\Vab*{#3}}{\Vab*{#3}}}}{\textup{BL}(\overline\Omega)}\le1$ and set $\varphi_\rho\coloneqq\sum_{Q\in\mathcal Q}\varphi(x_Q)\chi_{Q}$, so that $ 
  \IfBooleanTF{\varphi-\varphi_\rho}{\vab*{#3}}{ 
    \IfValueTF{\le}{\vab[\le]{#3}}{ 
      \mathchoice{\vab{#3}}{\vab*{#3}}{\vab*{#3}}{\vab*{#3}}}}\sqrt d\rho$ on $\Gamma$.
Then
\[
 
  \IfBooleanTF{\int\varphi\d(\lambda_N^g-g\mu_X)}{\vab*{#3}}{ 
    \IfValueTF{\le}{\vab[\le]{#3}}{ 
      \mathchoice{\vab{#3}}{\vab*{#3}}{\vab*{#3}}{\vab*{#3}}}}
\sqrt d\rho\pab{\lambda_N^g(\Gamma)+G}
+\sum_{Q\in\mathcal Q} 
  \IfBooleanTF{\lambda_N^g(Q)-(g\mu_X)(Q)}{\vab*{#3}}{ 
    \IfValueTF{,}{\vab[,]{#3}}{ 
      \mathchoice{\vab{#3}}{\vab*{#3}}{\vab*{#3}}{\vab*{#3}}}}
\]
and the right-hand side does not depend on $\varphi$.

\le
\sum_{Q\in\mathcal Q}\sqrt{\frac{G^2\mu_X(Q)}{N}}
\le

\]
where we used $\sum_Q\mu_X(Q)=1$.
