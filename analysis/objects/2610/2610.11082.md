<!-- proof-1: body_md -->
\textbf{Step 1: Construct hard functions.}
    For every $z \in \mathrm{supp}(\nu_T)$, define
    \begin{align}
        f_z := \frac{B}{4}\sqrt{\frac{\rho}{d_T}} V_T^{-1} \phi(z) \in \mathcal{H}_k.
    \end{align}
    We first show that $f_z \in \mathcal{F}_k(B)$.
    \Cref{lem:positive_definite} shows that
    $V_T^{-2} \preceq \rho^{-1}V_T^{-1}  $ and $V_T^{-1}, V_T^{-1/2}$ are self-adjoint.
    Then
    \begin{align}
        \lVertf_z\rVert_k^2
        & = \frac{B^2\rho}{16 d_T} \langle \phi(z), V_T^{-2}\phi(z) \rangle_k \\
        & \leq \frac{B^2}{16 d_T} \langle \phi(z), V_T^{-1}\phi(z) \rangle_k \\
        & = \frac{B^2}{16} \label{eq:use_fact_2} \\
        & \leq B^2,
    \end{align}
    where \cref{eq:use_fact_2} follows from \cref{eq:fact_2,eq:ux}. We have thus shown $f_z \in \mathcal{F}_k(B)$. 
    For any $x \in \mathcal{X}$,
    the reproducing property and definition of $u_x$ give that
    \begin{align}
        f_z(x) 
        = \frac{B}{4}\sqrt{\rho d_T} \langle u_z, u_x \rangle_k, \label{eq:f_z_x}
    \end{align}
    which combined with \cref{eq:fact_1,eq:fact_2} gives
    \begin{align}
        f_z(z) = B\sqrt{\rho d_T}/4,
        \quad
        \lvertf_z(x)\rvert \leq B\sqrt{\rho d_T}/4.
        \label{eq:max_f_z}
    \end{align}

    \textbf{Step 2: Change of measure.} 
    We will apply a well-known change of measure result, stated in \Cref{lem:change_of_measure} in \Cref{app:technical_lemmas}.
    We first bound the relevant KL divergence.
    Fix any policy $\pi$, and let
    $H_{t} := (X_1,Y_1,\ldots, X_{t},Y_{t}) $ be the history until time $t$ (the uppercase $X_t$ and $Y_t$ are random variables, whereas lowercase $x_t$ and $y_t$ are realizations). Let $P$ be the joint law on $(Z,H_T)$ obtained by first drawing $Z \sim \nu_T$, then running $\pi$ against $f_Z$. Let $Q$ be the joint law on $(Z,H_T)$ when the policy is instead run on the zero function, and $Z \sim \nu_T$ is drawn independently from the interaction.
    By the chain rule for KL divergence, 
    \begin{align}
        & D_{\mathrm{KL}} (Q \Vert P) \notag \\
        &= D_{\mathrm{KL}}(Q_Z \Vert P_Z) + \sum_{t=1}^T \mathbb{E}_Q[D_{\mathrm{KL}}(Q_{X_t\vert Z, H_{t-1}}  \Vert P_{X_t\vert Z, H_{t-1}} )] \notag \\
        &\quad +  \sum_{t=1}^T \mathbb{E}_Q[D_{\mathrm{KL}}(Q_{Y_t\vert Z, H_{t-1}, X_t}  \Vert P_{Y_t\vert Z, H_{t-1}, X_t}) ] \label{eq:kl_decompose} \\
        &= \sum_{t=1}^T \mathbb{E}_Q[D_{\mathrm{KL}}(Q_{Y_t\vert Z, H_{t-1}, X_t} \Vert P_{Y_t\vert Z, H_{t-1}, X_t} ) ] \label{eq:same_policy} \\
        &= \frac{1}{2\sigma^2} \sum_{t=1}^T \mathbb{E}_Q [f_Z(X_t)^2], \label{eq:KL_gaussian}
    \end{align}
    where \cref{eq:same_policy} follows because $P_Z = Q_Z$ and the same policy is used for both $f_Z$ and the zero function, and \cref{eq:KL_gaussian} follows from the KL divergence formula between two Gaussians.
    Under $Q$, the random variables $Z$ and $X_t$ are independent,
    which implies  $\mathbb{E}_Q [ f_Z(X_t)^2] = \mathbb{E}_Q[\mathbb{E}_Q[f_Z(X_t)^2 \vert X_t] ] = \mathbb{E}_Q [\mathbb{E}_{Z}[f_Z(X_t)^2]]  $. 
    Observe that, for any fixed $x \in \mathcal{X}$,
    \begin{align}
        \mathbb{E}_Z[f_Z(x)^2]
        & = \frac{B^2\rho d_T}{16} \mathbb{E}_Z[\lvert\langle u_Z, u_x \rangle_k\rvert^2] \label{eq:use_f_z_x} \\
        & = \frac{B^2\rho d_T}{16} \langle u_x, \mathbb{E}_Z[u_Z\otimes u_Z]u_x \rangle_k \label{eq:use_otimes} \\
        & \leq \frac{B^2\rho}{16} \langle u_x, u_x \rangle_k \label{eq:use_fact_3} \\
        & \leq \frac{B^2\rho}{16}, \label{eq:use_fact_1}
    \end{align}
    where \cref{eq:use_f_z_x} follows from \cref{eq:f_z_x}, \cref{eq:use_otimes} follows from the definition of $\otimes$ and linearity of expectation, \cref{eq:use_fact_3} follows from \cref{eq:fact_3}, and \cref{eq:use_fact_1} follows from \cref{eq:fact_1}.
    Combining with \cref{eq:KL_gaussian} gives
    \begin{align}
        D_{\mathrm{KL}} (Q \Vert P)
        \leq \frac{TB^2\rho}{32\sigma^2}
        = \frac{1}{32}. \label{eq:kl_bound}
    \end{align}
    We now proceed to bound $\mathbb{E}_Q[R_T(\pi, f_Z)]$.
    For any $x \in \mathcal{X}$, we have
    \begin{align}
        \mathbb{E}_Z[f_Z(x)] =
        & \frac{B\sqrt{\rho d_T}}{4} \langle \mathbb{E}_Z[u_Z], u_x \rangle_k \label{eq:use_f_z_x_2} \\
        & \leq \frac{B\sqrt{\rho}}{4} \lVertu_x\rVert_k \label{eq:use_fact_4} \\
        & \leq \frac{B\sqrt{\rho}}{4}, \label{eq_f_z_xt}
    \end{align}
    where \cref{eq:use_f_z_x_2} follows from \cref{eq:f_z_x}, \cref{eq:use_fact_4} follows from \cref{eq:fact_4} and Cauchy-Schwarz, and \cref{eq_f_z_xt} follows from \cref{eq:fact_1}.
    Recalling $\mathbb{E}_Q [ f_Z(X_t)] = \mathbb{E}_Q [\mathbb{E}_{Z}[f_Z(X_t)]]$,  \cref{eq:max_f_z,eq_f_z_xt} then give
    \begin{align}
        \mathbb{E}_Q[R_T(\pi, f_Z)] 
        & = \sum_{t=1}^T \mathbb{E}_Q \Bigl[\max_{x\in\mathcal{X}}f_Z(x) - f_Z(X_t) \Bigr] \\
        & \geq BT \biggl( \frac{\sqrt{\rho d_T}}{4} - \frac{\sqrt{\rho}}{4} \biggr) \geq \frac{TB\sqrt{\rho d_T}}{8}, \label{eq:regret_under_q}
    \end{align}
    where the last step uses $d_T \geq 4$.
    Finally, \cref{eq:max_f_z} implies $0 \leq R_T(\pi, f_Z) \leq BT\sqrt{\rho d_T} / 2$.
    Applying the standard change of measure result from \Cref{lem:change_of_measure} in \Cref{app:technical_lemmas} (total variation + Pinsker's inequality) with $H = R_T(\pi, f_Z)$, $M = BT\sqrt{\rho d_T} / 2$, and \cref{eq:regret_under_q,eq:kl_bound} gives
    \begin{align}
        \mathbb{E}_P[R_T(\pi, f_Z)]
        & \geq \frac{TB\sqrt{\rho d_T}}{8} - \frac{TB\sqrt{\rho d_T}}{2}\cdot \frac{1}{8} \\
        & = \frac{\sigma}{16} \sqrt{T d_T},
    \end{align}
    which implies \Cref{lem:regret_high_ds}
    since $\sup_{f\in\mathcal{F}_k(B)}\mathbb{E}[R_T(\pi, f)]$ is at least $\mathbb{E}_{Z\sim \nu_T}\mathbb{E}[R_T(\pi, f_Z)]$ and $\pi$ is arbitrary.

<!-- lemma-9: body_md -->
\label{lem:positive_definite}
    Consider the setup of \Cref{sec:setup_finite} where $\mathcal{X}$ is finite. For every $\nu \in \Delta(\mathcal{X})$,
    \begin{align}
        M_{\nu} + \rho I_E 
        \succ M_{\nu}
        \succeq 0. \label{eq:psd_claims}
    \end{align}
    Consequently, $(M_{\nu} + \rho I_E)^{-1} \succ 0$, $(M_{\nu} + \rho I_E)^{-1/2}$ is self-adjoint, and $(M_{\nu} + \rho I_E)^{-2} \preceq \rho^{-1} (M_{\nu} + \rho I_E)^{-1} $.
<!-- end lemma-9 -->
<!-- proof-4: body_md -->
To show $M_{\nu} + \rho I_E \succ M_{\nu}$, it suffices to show that $\rho I_E \succ 0$ and $M_{\nu} \succeq 0$.  The condition $\rho I_E \succ 0$ holds since $\langle \rho I_E v, v \rangle_k = \rho \lVertv\rVert_k^2 > 0 $ for every $v \in E$ and $v \neq 0$.
    To show $M_{\nu} \succeq 0$, we recall that $M_{\nu} = \sum_{x\in\mathcal{X}} \nu(x) \phi(x)\otimes \phi(x)$, such that $\langle M_{\nu} v, v \rangle_k = \sum_{x\in\mathcal{X}}\nu(x) \lvert\langle \phi(x), v \rangle_k\rvert^2 \geq 0 $ for every $v \in E$.  This establishes \cref{eq:psd_claims}.

    Since $M_{\nu} + \rho I_E$ is self-adjoint, it follows that $(M_{\nu} + \rho I_E)^{-1}$ is self-adjoint, which combined with the fact that all eigenvalues of $(M_{\nu} + \rho I_E)^{-1}$ are positive proves $(M_{\nu} + \rho I_E)^{-1} \succ 0$. This also proves $(M_{\nu} + \rho I_E)^{-1/2}$ is self-adjoint.
    To show $(M_{\nu} + \rho I_E)^{-2} \preceq \rho^{-1} (M_{\nu} + \rho I_E)^{-1} $, observe that $\rho^{-1} (M_{\nu} + \rho I_E)^{-1} -(M_{\nu} + \rho I_E)^{-2} = \rho^{-1} (M_{\nu} + \rho I_E)^{-1}M_{\nu} (M_{\nu} + \rho I_E)^{-1} $.
    For every $u \in E$, observe that $\langle (M_{\nu} + \rho I_E)^{-1}M_\nu (M_{\nu} + \rho I_E)^{-1} u, u \rangle_k = \langle M_\nu (M_{\nu} + \rho I_E)^{-1} u, (M_{\nu} + \rho I_E)^{-1}u \rangle_k \geq 0 $, which proves $(M_{\nu} + \rho I_E)^{-1}M_{\nu} (M_{\nu} + \rho I_E)^{-1} \succeq 0$, and thus $(M_{\nu} + \rho I_E)^{-2} \preceq \rho^{-1} (M_{\nu} + \rho I_E)^{-1}$.

\label{lem:bound_by_two_d_T}
    Consider the setup of \Cref{sec:setup_finite} where $\mathcal{X}$ is finite. For every $\nu \in \Delta(\mathcal{X})$,
    \begin{align}
        \operatorname{tr}_E(M_{\nu} (M_{\nu} + \rho I_{E})^{-1} )
        \leq 2d_T.
    \end{align}
<!-- end lemma-11 -->
<!-- proof-5: body_md -->
Fix $\nu \in \Delta(\mathcal{X})$. 
    Since $M_T := M_{\nu_T} $ where $\nu_T \in \Delta(\mathcal{X})$,
    \Cref{lem:positive_definite} shows that $M_{\nu}, M_T \succeq 0 $, such that
    \begin{align}
        \operatorname{tr}_E (M_{\nu}(M_{\nu}+ \rho I_E)^{-1} )
        & \leq \operatorname{tr}_E ((M_{\nu} + M_T) (M_{\nu} + M_T+ \rho I_E)^{-1} ) \\
        & \leq \operatorname{tr}_E ((M_{\nu} + M_T) (M_T+ \rho I_E)^{-1} ) \\
        & = \operatorname{tr}_E (M_{\nu} (M_T+ \rho I_E)^{-1} ) + d_T, \label{eq:one_d_T}
    \end{align}
    where the equality follows from the linearity of the trace and \cref{eq:effective_dimension}. For the first term in \cref{eq:one_d_T}, observe that
    \begin{align}
        \operatorname{tr}_E (M_{\nu} (M_T+ \rho I_E)^{-1} )
        & = \sum_{x \in \mathcal{X}} \nu(x) \operatorname{tr}_E ((\phi(x)\otimes \phi(x)) (M_T+ \rho I_E)^{-1} ) \\
        & = \sum_{x \in \mathcal{X}} \nu(x) \langle \phi(x), (M_T+ \rho I_E)^{-1}\phi(x) \rangle_k \label{eq:use_trace_ideneity} \\
        & \leq \max_{x\in\mathcal{X}} \langle \phi(x), (M_T+ \rho I_E)^{-1}\phi(x) \rangle_k \label{eq:mean_by_max} \\
        & = d_T, \label{eq:use_lem_15}
    \end{align}
    where \cref{eq:use_trace_ideneity} applies \Cref{lem:trace_to_inner_product}, \cref{eq:mean_by_max} follows since the average is upper bounded by the maximum, and \cref{eq:use_lem_15} applies \Cref{lem:camilleri_lem_15}.
    Combining \cref{eq:one_d_T,eq:use_lem_15} proves the result.

\label{lem:det_ratio}
    Consider the setup of \Cref{sec:setup_finite} where $\mathcal{X}$ is finite. Fix $u \in E$, and let $A$ be an operator on $E$ such that $A - u\otimes u \succ 0 $. Then

<!-- proof-6: body_md -->

\label{lem:unique_operator}
    Consider the setup of \Cref{sec:setup_finite} where $\mathcal{X}$ is finite.
    Suppose $\nu_1,\nu_2\in\Delta(\mathcal{X})$, and that both $\nu_1,\nu_2$ are regularized $D$-optimal designs satisfying \cref{eq:d_optimal_design} for some regularization level $\rho > 0$. Then $M_{\nu_1} = M_{\nu_2} $.
<!-- end lemma-17 -->
<!-- proof-11: body_md -->
If $\nu_1 = \nu_2$, then the result trivially follows, so we assume that $\nu_1 \neq \nu_2$.
    Fix $\nu \in \Delta(\mathcal{X})$. By \Cref{lem:positive_definite}, $M_\nu + \rho I_E$ is positive definite.
    We now verify that $\log\det_E $ is strictly concave in the space of positive definite operators. Let $A,B$ be distinct positive definite operators on $E$, and let $t \in (0,1)$. Define $C := A^{-1/2}BA^{-1/2} $. Then $C$ is positive definite. Let $\lambda_1,\ldots, \lambda_n > 0$ be its eigenvalues. Since $(1-t) A + tB = A^{1/2}\bigl( (1-t)I_E + tC \bigr)A^{1/2} $, it follows that
    \begin{align}
        \log\det_E\bigl( (1-t)A + tB \bigr)
        = \log\det_E A + \sum_{j=1}^n\log \bigl( (1-t) + t\lambda_j \bigr).
    \end{align}
    Also notice that $\log\det_E(B) = \log\det_E(A) + \sum_{i=1}^n \log\lambda_i $. Since the logarithm is strictly concave on $(0, \infty)$, we have $\log\bigl( (1-t) + t\lambda_i \bigr) \geq t\log \lambda_i $, with equality when $\lambda_i = 1$. Since $A \neq B$, it follows that $C \neq I_E$, such that at least one of $\lambda_1,\ldots, \lambda_n$ does not equal $1$. Combining, we have
    \begin{align}
        \log\det_E\bigl( (1-t)A + tB \bigr)
        > (1-t) \log\det_E (A) + t \log \det_E(B). \label{eq:strong_concave}
    \end{align}
    Now suppose by contradiction that $M_{\nu_1} \neq M_{\nu_2}$. Consider the mixed design $\nu_t := (1-t)\nu_1 + t\nu_2 \in \Delta(\mathcal{X}) $. Since the mapping $\nu \mapsto M_\nu$ is linear, $M_{\nu_t} = (1-t) M_{\nu_1} + t M_{\nu_2} $.
    Then, applying \cref{eq:strong_concave} with $A = M_{\nu_1} + \rho I_E$ and $B = M_{\nu_2} + \rho I_E$ shows that
    \begin{align}
        \log\det_E\bigl( M_{\nu_t} + \rho I_E \bigr)
        > (1-t) \log\det_E (M_{\nu_1} + \rho I_E) + t \log \det_E(M_{\nu_2} + \rho I_E),
    \end{align}
    which implies the design value at $\nu_t$ (notice $\nu_t$ is different from $\nu_1$ and $\nu_2$ since $\nu_1 \neq \nu_2$) is larger than the $D$-optimal design value, establishing a contradiction.
    Hence, $M_{\nu_1} = M_{\nu_2}$.

        & = \frac{1}{2} \log \det_E (I_E + \rho^{-1} M_{T}), \label{eq:compare_gamma_T}
    \end{align}
    where the max operation in \cref{eq:max_over_empirical_distribution} is over all empirical distributions on $\mathcal{X}$ from $T$ queries, and the last equality follows from the definition of $M_T$.

    \Cref{lem:trace_of_M_T} in \Cref{app:technical_lemmas} gives $\operatorname{tr}_E(M_T) \leq 1$.    
    Then, the definition of $d_T$ in \cref{eq:effective_dimension} gives $d_T = \sum_{i=1}^n \frac{\lambda_i}{\lambda_i + \rho} \leq \rho^{-1} \sum_{i=1}^n \lambda_i = \rho^{-1} \operatorname{tr}_E(M_T) \leq \rho^{-1}$, proving the desired upper bound.
    For later use, we note that the inequality $\operatorname{tr}_E(M_T) \leq 1$ also implies $\lambda_i \leq 1$ for $i \in [n]$.
    
    It remains to prove the lower bound.
    For $i \in [n]$, define $w_i := \rho^{-1}\lambda_i$, such that $0 \leq w_i \leq \rho^{-1} $. Then, the formula $d_T = \sum_{i=1}^n\frac{\lambda_i}{\lambda_i + \rho}$ equivalently gives $d_T = \sum_{i=1}^n \frac{w_i}{1+w_i} $.
    For $w > 0$, define $q(w):= \frac{1+w}{w}\log(1+w) $, with $q(0) := 1$. Since $q'(w) = \frac{w -\log(1+w)}{w^2} > 0 $, it follows that $q(\cdot)$ is strictly increasing. 
    Then, \cref{eq:compare_gamma_T} implies that
    \begin{align}
        \gamma_T
        & \leq \frac{1}{2} \sum_{i=1}^n \log (1+w_i)
        = \frac{1}{2} \sum_{i=1}^n \frac{w_i}{1+w_i} q(w_i) \\
        & \leq \frac{q(\rho^{-1})}{2} \sum_{i=1}^n \frac{w_i}{1+w_i}
        = \frac{q(\rho^{-1})d_T}{2}. \label{eq:gamma_T_d_T}
    \end{align}
    Since $\log(1+w) \leq w$, it follows that $q(w) = \frac{1}{w}\log(1+w) + \log(1+w) \leq 1 + \log(1+w) $. Then, \cref{eq:gamma_T_d_T} gives
    \begin{align}
        d_T 
        \geq \frac{2\gamma_T}{1+\log(1+\rho^{-1})},
    \end{align} 
    which proves the desired lower bound.

The proof is similar to \Cref{lem:regret_high_ds} in structure, and the main differences are in numerical constants, so we omit details where appropriate.
    
    \textbf{Step 1: Construct hard functions.}
    For every $z \in \mathrm{supp}(\nu_T)$, define
    \begin{align}
        f_z := \frac{B}{6} \sqrt{\frac{\rho}{d_T}} (M_T + \rho I_E)^{-1} \phi(z)
        \in \mathcal{H}_k.
    \end{align}
    It follows that $\lVertf_z\rVert_k^2 \leq \frac{B^2}{36} \leq B^2$, and hence $f_z \in \mathcal{F}_k(B)$.
    In addition, observe that
    \begin{align}
        f_z(x) = \frac{B}{6}\sqrt{\rho d_T} \langle u_z, u_x \rangle_k. \label{eq:fz_at_x}
    \end{align}
    Combining \Cref{lem:off_net_bound} and \cref{eq:fact_1,eq:fact_2} gives
    \begin{align}
        f_z(z) = \frac{B}{6} \sqrt{\rho d_T},
        \qquad
        \lvertf_z(x)\rvert \leq \frac{3B}{16} \sqrt{\rho d_T}
        \qquad
        \forall x \in \mathcal{X}. \label{eq:f_values}
    \end{align}

    \textbf{Step 2: Change of measure.} 
    To use \Cref{lem:change_of_measure}, we first bound the KL divergence.
    Let $H_{t} := (X_1,Y_1,\ldots, X_{t},Y_{t}) $ be the history until time $t$. Fix a policy $\pi$. Let $P$ be the joint law on $(Z,H_T)$ obtained by first drawing $Z \sim \nu_T$, then running $\pi$ against $f_Z$. Let $Q$ be the joint law on $(Z,H_T)$ when the policy is instead run on the zero function, and $Z \sim \nu_T$ is drawn independently from the interaction.
    Then
    \begin{align}
        D_{\mathrm{KL}} (Q \Vert P)
        & = \frac{1}{2\sigma^2} \sum_{t=1}^T \mathbb{E}_Q [f_Z(X_t)^2] \label{eq:kl_chain_rule} \\
        & = \frac{1}{2\sigma^2} \sum_{t=1}^T \mathbb{E}_Q [\mathbb{E}_{Z}[f_Z(X_t)^2]] \label{eq:conditional_independence} \\
        & \leq \frac{T}{2\sigma^2} \cdot \frac{9B^2\rho}{256} \label{eq:bound_eqzfzxt} \\
        & = \frac{9}{512}, \label{eq:kl_bound_2}
    \end{align}
    where \cref{eq:kl_chain_rule} follows from the chain rule for KL divergence,
    \cref{eq:conditional_independence} follows because the random variables $Z$ and $X_t$ are independent under $Q$,
    \cref{eq:bound_eqzfzxt} is explained below, and \cref{eq:kl_bound_2} uses $\rho = \sigma^2/(B^2T)$.  To verify \cref{eq:bound_eqzfzxt}, observe that for every $x \in \mathcal{X}$, $\mathbb{E}_Z[f_Z(x)^2] = \frac{B^2\rho d_T}{36} \langle u_x, \mathbb{E}_Z[u_Z\otimes u_Z]u_x \rangle_k \leq \frac{B^2\rho}{36} \langle u_x, u_x \rangle_k \leq \frac{9B^2\rho}{256}$,
    where the first inequality follows from \cref{eq:fact_3_new}, and the second inequality follows from \Cref{lem:off_net_bound}.
   
    Next, we proceed to bound $\mathbb{E}_Q[R_T(\pi, f_Z)]$.
    For any $x \in \mathcal{X}$, we have
    \begin{align}
        \mathbb{E}_Z[f_Z(x)] =
        \frac{B\sqrt{\rho d_T}}{6} \langle \mathbb{E}_Z[u_Z], u_x \rangle_k
        \leq \frac{B\sqrt{\rho}}{6} \lVertu_x\rVert_k
        \leq \frac{3B\sqrt{\rho}}{16}, \label{eq:EZ_fZ}
    \end{align}
    where the equality follows from \cref{eq:fz_at_x}, the first inequality follows from \cref{eq:fact_4} and Cauchy-Schwarz, and the second inequality follows from \Cref{lem:off_net_bound}.
    The independence of $Z$ and $X_t$ under $Q$ implies $\mathbb{E}_Q [ f_Z(X_t)] = \mathbb{E}_Q [\mathbb{E}_{Z}[f_Z(X_t)]]  $, which combined with \cref{eq:f_values,eq:EZ_fZ} and $d_T \geq 4$ gives
    \begin{align}
        \mathbb{E}_Q[R_T(\pi, f_Z)] 
        = \sum_{t=1}^T \mathbb{E}_Q \bigl[\max_{x}f_Z(x) - f_Z(X_t) \bigr] 
        \geq BT \biggl( \frac{\sqrt{\rho d_T}}{6} - \frac{3\sqrt{\rho}}{16} \biggr)
        \geq \frac{7TB\sqrt{\rho d_T}}{96}. \label{eq:EQ_RT}
    \end{align}
    Finally, \cref{eq:f_values} implies $0 \leq R_T(\pi, f_Z) \leq \frac{3BT\sqrt{\rho d_T}}{8} $.
    Applying \Cref{lem:change_of_measure} with $H = R_T(\pi, f_Z)$, $M = \frac{3BT\sqrt{\rho d_T}}{8}$, and \cref{eq:EQ_RT,eq:kl_bound_2} gives
    \begin{align}
        \mathbb{E}_P[R_T(\pi, f_Z)]
        \geq \frac{7TB\sqrt{\rho d_T}}{96} -  \frac{3BT\sqrt{\rho d_T}}{8} \cdot \frac{3}{32}
        = \frac{29\sigma}{32\cdot24} \sqrt{Td_T}
        \geq \frac{\sigma}{32} \sqrt{T d_T},
    \end{align}
    which proves \Cref{lem:regret_high_dT_compact}
    since $\sup_{f\in\mathcal{F}_k(B)}\mathbb{E}[R_T(\pi, f)] \geq \mathbb{E}_{Z\sim \nu_T}\mathbb{E}[R_T(\pi, f_Z)]$ and the policy $\pi$ is arbitrary.

<!-- proof-27: body_md -->
We will first prove a high-probability lower bound expressed using $d_T = d_T(\tilde{\mathcal{X}})$, then transfer to $\gamma_T(\mathcal{X})$ using \Cref{lem:d_T_bound_compact}.
    We consider two cases.

    \textbf{Case 1: $d_T \geq 4$.}
    We follow the setup in \Cref{lem:regret_high_dT_compact}: we consider the hard functions $\{f_z\}$, where $z \in \mathrm{supp}(\nu_T)$. We fix some policy $\pi$, and consider probability measures $P$, $Q$, which are joint laws on $(Z,H_T)$ where $Z \sim \nu_T$, and $H_T$ is the history until time $T$ obtained by running $\pi$ against $f_Z$. 
    Specifically, $P$ is obtained by drawing $Z\sim \nu_T$ and running $\pi$ against $f_Z$, and $Q$ is obtained by drawing $Z$ independently and running $\pi$ against the zero function.
    For convenience we define $S:= \sqrt{\sigma^2 T d_T}$.

    By \cref{eq:f_values}, $f_Z(Z) = \frac{B}{6} \sqrt{\rho d_T}$, which yields
    \begin{align}
        R_T(\pi, f_Z)
        \geq \frac{BT}{6} \sqrt{\rho d_T} - \sum_{t=1}^T f_Z(X_t)
        = \frac{S}{6} - \sum_{t=1}^T f_Z(X_t). \label{eq:R_T_in_S}
    \end{align}
    Moreover, by \cref{eq:fz_at_x}, $f_z(x) = \frac{B}{6}\sqrt{\rho d_T} \langle u_z, u_x \rangle_k$, which gives $\sum_{t=1}^T f_Z(X_t) = \frac{B}{6}\sqrt{\rho d_T} \langle u_Z, \sum_{t=1}^T u_{X_t} \rangle_k $.
    We then observe that
    \begin{align}
        \mathbb{E}_Q \biggl[ \biggl(\sum_{t=1}^T f_Z(X_t)\biggr)^2 \,\bigg\vert\, X_{1:T} \biggr]
        & = \frac{B^2\rho d_T}{36} \mathbb{E}_Q \biggl[  \biggl\langle \sum_{t=1}^T u_{X_t}, u_Z\otimes u_Z \sum_{t=1}^T u_{X_t}\biggr \rangle_k \,\bigg\vert\, X_{1:T} \biggr] \\
        & = \frac{B^2\rho d_T}{36} \biggl\langle \sum_{t=1}^T u_{X_t}, \mathbb{E}[u_Z\otimes u_Z] \sum_{t=1}^T u_{X_t}\biggr \rangle_k, \label{eq:use_conditional_independence} \\
        & \leq \frac{B^2\rho}{36} \biggl\lVert \sum_{t=1}^T u_{X_t} \biggr \rVert_k^2, \label{eq:use_fact_new} \\
        & \leq \frac{B^2\rho}{36}\cdot T \sum_{t=1}^T \lVertu_{X_t}\rVert_k^2 \label{eq:use_cauchy} \\
        & \leq \frac{9\sigma^2 T}{256} \label{eq:use_off_net_bound} \\
        & \leq \frac{9S^2}{1024}, \label{eq:use_def_S}
    \end{align}
    where \cref{eq:use_conditional_independence} follows because $Z$ is independent of the zero-function interaction under $Q$, \cref{eq:use_fact_new} follows from \cref{eq:fact_3_new}, 
    \cref{eq:use_cauchy} applies Cauchy-Schwarz,
    \cref{eq:use_off_net_bound} follows from combining \Cref{lem:off_net_bound} and $\rho = \sigma^2/(B^2T)$, and \cref{eq:use_def_S} applies the definition of $S$ and $d_T \geq 4$.

    We now combine the above findings to lower bound the regret.  
    First observe that if $\sum_{t=1}^T f_Z(X_t) \leq \frac{13S}{96}$, then \cref{eq:R_T_in_S} gives $R_T(\pi, f_Z) \geq \frac{S}{32}$. 
    Moreover, applying Markov's inequality to $\bigl(\sum_{t=1}^T f_Z(X_t)\bigr)^2$ gives
    \begin{align}
        \mathbb{P}_Q(R_T(\pi, f_Z) \geq S / 32 )
        \geq \mathbb{P}_Q\biggl(\sum_{t=1}^T f_Z(X_t) \leq \frac{13S}{96}\biggr)
        \geq 1 - \mathbb{P}_Q\biggl(\biggl(\sum_{t=1}^T f_Z(X_t)\biggr)^2 \geq \biggl(\frac{13S}{96} \biggr)^2 \biggr)
        \geq 1 - \frac{81}{169}
        = \frac{88}{169}.
    \end{align}
    Recall that \cref{eq:kl_bound_2} gives $D_{\mathrm{KL}}(Q \Vert P) \leq \frac{9}{512}$.
    Then, applying \Cref{lem:change_of_measure} with $H = \mathbb{I}_{\{R_T(\pi, f_Z) \geq S/32\}}$ and $M = 1$ gives
    \begin{align}
        \mathbb{P}_P(R_T(\pi, f_Z) \geq S/32)
        \geq \mathbb{P}_Q(R_T(\pi, f_Z) \geq S/32) - \sqrt{\frac{1}{2}D_{\mathrm{KL}}(Q \Vert P)}
        \geq \frac{88}{169} - \frac{3}{32}
        = \frac{2309}{5408}
        \geq \frac{1}{4}.
    \end{align}
    Then, since $S/32 = \sqrt{\sigma^2 T d_T} / 32 \geq \min \{1, 2\Delta_k \} \cdot \frac{\sigma}{32} \sqrt{Td_T}$, we have
    \begin{align}
        P\biggl(R_T(\pi, f_Z) \geq \min \{1, 2\Delta_k \} \cdot \frac{\sigma}{32} \sqrt{Td_T}\biggr)
        \geq \frac{1}{4}.
    \end{align}
    This holds for $Z\sim \nu_T$, and it follows that there exists some fixed $z \in \mathrm{supp}(\nu_T)$ such that $R_T(\pi, f_z) \geq \min \{1, 2\Delta_k \} \cdot \frac{\sigma}{32} \sqrt{Td_T}$ with probability at least $1/4$.
    

    \textbf{Case 2: $0 < d_T < 4$.}
    Here we follow the setup in \Cref{lem:regret_2}. We define $a := \min\{B, \sigma/(2\sqrt{T}) \} $, and consider the hard functions $f_v := va \cdot g$, where $v \in \{-1,+1\}$ and $g$ is defined in \cref{eq:function_g}. We again fix some policy $\pi$, and consider probability measures $P$, $Q$, which are joint laws on $(V,H_T)$ where $V$ is uniform on $\{-1,+1\}$, and $H_T$ is the history until time $T$ obtained by running $\pi$ against $f_V$. Specifically, $P$ is obtained by drawing $V$ and running $\pi$ against $f_V$, and $Q$ is obtained by drawing $V$ independently and running $\pi$ against the zero function.
    We define $R_{+} := \sum_{t=1}^T (\max_{x\in\mathcal{X}} f_{+1}(x) - f_{+1}(x_t) ) $ and $R_{-} := \sum_{t=1}^T (\max_{x\in\mathcal{X}} f_{-1}(x) - f_{-1}(x_t) ) $, and consider any fixed realization $x_1,\ldots, x_T \in \mathcal{X}$. 
    
    Recall that $\Delta_k := \max_{x,x' \in \mathcal{X}} \lVert\phi(x) - \phi(x')\rVert_k $ and the proof of \Cref{lem:regret_2} gives $\max_{x\in\mathcal{X}} g(x) - \min_{x\in\mathcal{X}} g(x) = \Delta_k $. We observe that
    \begin{align}
        R_{+} + R_{-}
        = a \sum_{t=1}^T \biggl( \max_{x\in\mathcal{X}}g(x) - g(x_t) + g(x_t) - \min_{x\in\mathcal{X}}g(x) \biggr)
        = a T \Delta_k,
    \end{align}
    which implies that at least one of $R_{+}$, $R_{-}$ is at least $aT\Delta_k/2$.  Hence,
    \begin{align}
        \mathbb{P}_Q(R_T(\pi, f_V) \geq aT\Delta_k/2 \vert X_{1:T} )
        \geq \frac{1}{2}.
    \end{align}

    \begin{align}
        k_{\mathrm{RQ}, a}(h) = \int_{\mathbb{R}^d} \exp(2\pi i \xi\cdot h) s(\xi) \ d\xi. \label{eq:spectral_density}
    \end{align}
    
    \textbf{Step 1: Deriving the rate of $s(\xi)$.} This step is similar to \citep[pp.~21--22]{Lee25}, but extends to $d$ dimensions.
    For $u > 0$, the Gamma function satisfies $\int_{0}^{\infty} t^{a-1} \exp(-ut) \ dt = u^{-a} \Gamma(a)$. Then
    \begin{align}
        k_{\mathrm{RQ},a}(h) 
        = \biggl( 1 + \frac{\lVerth\rVert_2^2}{2a\ell^2} \biggr)^{-a} 
        = \frac{1}{\Gamma(a)} \int_{0}^{\infty} t^{a-1}\exp(-t) \exp \biggl( - \frac{t \lVerth\rVert_2^2}{2a\ell^2} \biggr) \ dt. \label{eq:rq_to_gaussian}
    \end{align}
    In $d$ dimensions, the Fourier transform of a Gaussian is again a Gaussian (e.g., see \cite[eq. (4.6) and discussion after eq. (4.9)]{Rasmussen05}):
    \begin{align}
        \int_{\mathbb{R}^d} \exp \biggl( - \frac{t \lVerth\rVert_2^2}{2a\ell^2} \biggr) \exp(-2\pi i \xi \cdot h) \ dh

    \end{align}
    where the right-hand side is the spectral density of the Gaussian in \cref{eq:rq_to_gaussian}, which we denote as $q_t(\xi)$. Then

    \end{align}

    Substituting \cref{eq:gaussian_spectral_density} into \cref{eq:rq_to_gaussian} gives

Fix a nonempty $\mathcal{V} \subseteq \mathcal{X}$, and denote $m := \lvert\mathcal{V}\rvert $. 
    Following \cref{eq:d_optimal_design}, we define $\nu$ as a regularized $D$-optimal design over $\Delta(\mathcal{V})$. 
    We also define $E_1 :=\mathrm{span} \{\phi(v): v \in \mathcal{V} \} $.
    Define $D := \operatorname{diag}(\nu(v), v \in \mathcal{V} ) $ and the weighted Gram matrix $H_k := D^{1/2} [k(v,w)]_{v,w\in\mathcal{V}} D^{1/2} $. 
    By an argument similar to that in \Cref{lem:same_eigenvalues}, we can show that $H_k$ and $M_{\nu}$ have the same nonzero eigenvalues (including multiplicities) as follows: Define $\Phi: \mathbb{R}^m \to E_1 $ by $\Phi b_i = \sqrt{\nu(v_i)} \phi(v_i) $, and define $\Phi^*$ to be its adjoint. Then $H_k = \Phi^*\Phi $ and $M_{\nu} = \Phi \Phi^* $, and the result follows.
    
    Then, \citep[Lemma 3]{Camilleri21} gives $h(\mathcal{V}, \rho) \leq \operatorname{tr}_{E_1}(M_{\nu} (M_{\nu} + \rho I_{E_1})^{-1})$, which combined with the above fact on $H_k, M_{\nu}$ implies

    To proceed, we apply \citep[Theorem 1]{Janz26}.
    Suppose that for every $M \in \mathbb{Z}_{+}$, there exist
    positive semidefinite kernels $k_{M,1}, k_{M,2}: \mathcal{X}\times \mathcal{X} \to \mathbb{R}$ and $\epsilon_M > 0$ such that
    \begin{align}
        k \preceq k_{M,1} + k_{M,2},
        \qquad
        \mathrm{rank}[k_{M,1}(x,y)]_{x,y\in\mathcal{X}} =  O(M^d),
        \qquad
        \max_{x \in\mathcal{X}} k_{M,2}(x,x) \leq \epsilon_M^2, \label{eq:kernel_condition}
    \end{align}
    where $k \preceq k_{M,1} + k_{M,2}$ denotes the corresponding Gram-matrix inequality. Following $H_k$, we define $H_{k_{M,1}}$ and $H_{k_{M,2}}$ as the weighted Gram matrices for $k_{M,1}$ and $k_{M,2}$, respectively. Then
    \begin{align}
        H_k \preceq H_{k_{M,1}} + H_{k_{M,2}},
        \qquad
        \mathrm{rank} \ H_{k_{M,1}} =  O(M^d),
        \qquad
        \operatorname{tr} H_{k_{M,2}} = \sum_{v\in\mathcal{V}} \nu(v)k_{M,2}(v,v) \leq \epsilon_M^2. \label{eq:H_facts}
    \end{align}
    Then, we follow the proof of \citep[Theorem 1]{Janz26} to upper bound $\operatorname{tr}(H_k (H_k + \rho I)^{-1})$.
    For convenience, we define $H := H_{k_{M,1}} + H_{k_{M,2}} $.
    Since $H_k \preceq H$, it follows that $\operatorname{tr}(H_k (H_k + \rho I)^{-1}) \leq \operatorname{tr} (H (H + \rho I)^{-1}) $. Let $P$ be the orthogonal projection onto the range of $H_{k_{M,1}}$, and define $Q := I - P$. Then
    \begin{align}
        \operatorname{tr}(H_k (H_k + \rho I)^{-1}) 
        & \leq \operatorname{tr} (H (H + \rho I)^{-1}) \\
        & = \operatorname{tr} (PH (H + \rho I)^{-1}P) + \operatorname{tr} (QH (H + \rho I)^{-1}Q) \\
        & \leq \mathrm{ rank } \ H_{k_{M,1}} + \operatorname{tr} (QH (H + \rho I)^{-1}Q) \\
        & \leq \mathrm{ rank } \ H_{k_{M,1}} + \rho^{-1} \operatorname{tr} (QH_{k_{M,2}} Q), \label{eq:trace_h_k}
    \end{align}
    where the second inequality uses $H (H + \rho I)^{-1} \preceq I$ and $\operatorname{tr} P = \mathrm{ rank } \ H_{k_{M,1}} $,
    the third inequality follows since $H(H+\rho I)^{-1} \preceq \rho^{-1}H $ and $Q H_{k_{M,1}} Q  = 0$. Applying \cref{eq:f_V_by_H_k,eq:H_facts} gives

        = O ( M^d + \epsilon_M^2 / \rho )
        = O ( M^d + T\epsilon_M^2 ). \label{eq:f_V_by_r_eta}
    \end{align}

    We now specialize \cref{eq:f_V_by_r_eta} to each kernel.

    \textbf{Squared exponential kernel.}
    \citep[Remark 2]{Janz26} shows the squared exponential kernel has spectral density $s(\xi) = \Theta \bigl( \exp(-\tau \lVert\xi\rVert^2_2) \bigr) $ for some $\tau > 0$. Denote $\mu$ as its spectral measure. Then
    \begin{align}
        \int_{\mathbb{R}^d} \exp\biggl( \frac{\tau}{2} \lVert\xi\rVert_2^2 \biggr) \mu(d\xi)
        = \int_{\mathbb{R}^d} \exp\biggl(  \frac{\tau}{2} \lVert\xi\rVert_2^2 \biggr) s(\xi) \ d\xi
        \leq C \int_{\mathbb{R}^d} \exp\biggl( - \frac{\tau}{2} \lVert\xi\rVert_2^2 \biggr) \ d\xi
        = C (2\pi / \tau)^{d/2} < \infty,
    \end{align}
    where the second equality follows by applying $\int_{\mathbb{R}} \exp(-\tau x^2/2) \ dx = \sqrt{2\pi/\tau} $ to each dimension.
    Then the conditions of \citep[Proposition 15]{Janz26} are satisfied with $\tau_0 = \tau / 2$ and $\gamma = 2$, which implies $k_{\mathrm{SE}}$ (defined on $[-1,1]^d\times [-1,1]^d $) is entire of finite order at most $p = 2$.
    Then \citep[Proposition 14]{Janz26} gives $\epsilon_M = O\bigl(\exp (-cM\log(e+M))\bigr) $ for some $c > 0$, and restricting to $\mathcal{X}\times \mathcal{X}$ preserves this bound.

    \begin{align}
        h(\mathcal{V}, \rho)
        = O \bigl( (\log T / \log\log T)^d + T^{-1} \bigr)
        = O \bigl( (\log T / \log\log T)^d \bigr).
    \end{align}
    
    \textbf{Rational quadratic kernel}. We define $\tilde{k}_{\mathrm{RQ}}$ as the extension of $k_{\mathrm{RQ}}$ to $\mathbb{R}^d \times \mathbb{R}^d$. Then $\tilde{k}_{\mathrm{RQ}}$ is positive semidefinite, because it can be represented as a positive mixture of squared exponential kernels \citep[(4.19)--(4.20)]{Rasmussen05}. In addition, $\tilde{k}_{\mathrm{RQ}}$ is real analytic on $\mathbb{R}^d \times \mathbb{R}^d$, such that $\tilde{k}_{\mathrm{RQ}}$ belongs to the Gevrey class $G^1(\mathbb{R}^d \times \mathbb{R}^d)$, which is the class of real-analytic functions on $\mathbb{R}^d \times \mathbb{R}^d$ \citep{Janz26}.  Then \citep[Corollary 11]{Janz26} is satisfied with $U = \mathbb{R}^d$, $\sigma = 1$, $\tilde{k} =  \tilde{k}_{\mathrm{RQ}}$, such that $\tilde{k}_{\mathrm{RQ}} $ (restricted to $[0,1]^d\times [0,1]^d$) has $\epsilon_M = O (\exp(-cM)) $, and further restricting to $\mathcal{X}\times \mathcal{X}$ preserves this bound.
    Then, choosing $M = \lceil c^{-1} \log T \rceil$ gives
    \begin{align}
        h(\mathcal{V}, \rho)
        = O \bigl( (\log T)^d + T^{-1} \bigr)
        = O \bigl( (\log T)^d \bigr),
    \end{align}
    and hence $\max_{\mathcal{V} \subseteq \mathcal{X}}h(\mathcal{V}, \rho) = O((\log T)^d)$.
    
    \textbf{One-dimensional Dirichlet kernel.}
    Observe that
    \begin{align}
        k_{\mathrm{Dir},n}(x,x')
        & = \frac{1}{2n+1} \sum_{k=-n}^n \exp(-ik(x-x')) \\
        & = \frac{1}{2n+1} \biggl(1 +  2\sum_{j=1}^n \cos(j(x-x')) \biggr) \\
        & = \frac{1}{2n+1} \biggl(1 +  2\sum_{j=1}^n [\cos(jx)\cos(jx') + \sin(jx)\sin(jx')] \biggr), \label{eq:kernel_decompose}
    \end{align}
    where each of the $2n+1$ terms in \cref{eq:kernel_decompose} is a positive semidefinite kernel with rank at most $1$. 
    Then, $k_{\mathrm{Dir},n}(r)$ has rank at most $2n+1$. Choosing $k_{M,1} = k$ and $k_{M,2} \equiv 0$ gives $\mathrm{rank} \ H_{k_{M,1}} \leq 2n + 1$ and $\operatorname{tr} H_{k_{M,2}} = 0$, which combined with
    \cref{eq:f_V_by_H_k,eq:trace_h_k} gives
    \begin{align}
        h(\mathcal{V}, \rho)
        \leq 2n + 1,
    \end{align}
    and hence $\max_{\mathcal{V} \subseteq \mathcal{X}}h(\mathcal{V}, \rho) \le 2n+1$.

Fix $x \in \mathcal{X}$, and consider a grid cell in $\tilde{\mathcal{X}}$ containing $x$.
    Then there exists $j\in \{0, \ldots, m-1 \}^d $, such that the vertices of the cell can be written as 
    \begin{align}
        v_b = m^{-1} (j + b) \in \tilde{\mathcal{X}}
        \qquad
        \forall b \in \{0, 1\}^d.
    \end{align}
    Likewise, there exists $\theta \in [0,1]^d$, such that $x = m^{-1}(j + \theta) $.
    For $\ell \in [d]$, we define $b_\ell$ to be the $\ell$-th coordinate of $b$, and similarly $\theta_{\ell}$.
    Then, we define the multilinear interpolation weights
    \begin{align}
        w_b(x) := \prod_{\ell:b_{\ell} = 1} \theta_{\ell} \prod_{\ell: b_{\ell} = 0} (1 - \theta_{\ell}). 
    \end{align}
    It follows that
    \begin{align}
        w_b(x) \geq 0,
        \qquad
        \sum_{b \in \{0,1\}^d } w_b(x) = 1,
        \qquad
        \sum_{b \in \{0,1\}^d } w_b(x) v_b = x,
        \qquad
        \lVertv_b - x\rVert_2 \leq \sqrt{d}/m. \label{eq:weight_facts}
    \end{align}
    Define $I_m f(x) := \sum_{b \in \{0,1\}^d } w_b(x) f(v_b) $. 
    Because $w_b(x)$ are weights that sum to one, 
    we have $I_m f(x) \leq \max_{a\in\tilde{\mathcal{X}}} f(a) $.
    Consequently, $ f(x) - \max_{a \in \tilde{\mathcal{X}}}f(a) \leq f(x) - I_m f(x) \leq \lvertf(x) - I_m f(x)\rvert  $, and taking the maximum over $x \in \mathcal{X}$ gives
    \begin{align}
        \max_{x \in \mathcal{X}} f(x) - \max_{a \in \tilde{\mathcal{X}}} f(a)
        \leq \sup_{x\in \mathcal{X} } \lvertf(x) - I_m f(x)\rvert. \label{eq:approximation_error}
    \end{align}
    We proceed to upper bound ${f(x) - I_m f(x)}$.
    By the reproducing property,
    \begin{align}
        f(x) - I_m f(x)
        = \biggl \langle f, \phi(x) - \sum_{b \in \{0,1\}^d } w_b(x) \phi(v_b) \biggr \rangle_k.
    \end{align}
    Applying the Cauchy-Schwarz inequality and $\lVertf\rVert_k \leq B$ gives
    \begin{align}
        \lvertf(x) - I_mf(x)\rvert^2
        \leq B^2  \biggl\lVert \phi(x) - \sum_{b \in \{0,1\}^d } w_b(x) \phi(v_b) \biggr\rVert_k^2. \label{eq:Imf_bound}
    \end{align}
    For the squared norm term in \cref{eq:Imf_bound}, we have
    \begin{align}
        & \biggl\lVert \phi(x) - \sum_{b \in \{0,1\}^d } w_b(x) \phi(v_b) \biggr\rVert_k^2 \notag \\
        &= k(x,x) + \sum_{b,c} w_b(x) w_c(x) k(v_b, v_c) - \sum_b w_b(x) (k(x,v_b) +k(v_b, x)) \\
        &=\int_{\mathbb{R}^d} \biggl[ 1 + \sum_{b,c} w_b(x)w_c(x) \exp(iw^T(v_b - v_c)) - \sum_{b} w_b(x) \exp(iw^T(v_b - x)) - \sum_{b} w_b(x)\exp(iw^T(x-v_b))  \biggr] p(w) \ dw \\
        &= \int_{\mathbb{R}^d} \biggl( 1 - \sum_b w_b(x) \exp(iw^T (v_b - x)) \biggr) \biggl( 1 - \sum_c w_c(x) \exp(-iw^T (v_c - x)) \biggr) p(w) \ dw \\
        &= \int_{\mathbb{R}^d} \biggl\lvert 1 - \sum_b w_b(x) \exp(iw^T (v_b - x)) \biggr\rvert^2 p(w) \ dw, \label{eq:norm_bound}
    \end{align}
    where the second equality applies \cref{eq:spectral_representation},
    and the last equality follows because the weights are real and $\overline{e^{it}} = e^{-it}$.
    Now define $D_x(w) := 1 - \sum_{b} w_b (x) \exp(i w^{T}(v_b -x))$. Since $\lvert\exp(ui)\rvert = \lvert\cos u + i \sin u\rvert =  1$ for real $u$, we have $\lvertD_x(w)\rvert \leq 1 + \sum_b \lvertw_b(x)\rvert \lvert\exp(iw^T(v_b - x))\rvert = 1 + \sum_b w_b(x) = 2 $. For a sharper bound, we first show
    \begin{align}
        \lvert\exp(iu) - 1 - ui\rvert \leq u^2 / 2. \label{eq:integral_bound}
    \end{align}
    Indeed, $\exp(ui) - 1 - ui = ui \int_0^1 \exp(iut) - 1 \ dt = -u^2 \int_{0}^1\int_{0}^t \exp(isu) \ dsdt $. Since $\lvert\exp(isu)\rvert = 1$, we have $\lvert\exp(ui) - 1 - ui\rvert \leq u^2 \int_{0}^1\int_{0}^t 1 \ dsdt = u^2/2 $.
    Then
    \begin{align}
        \lvertD_x(w)\rvert
        & = \biggl\lvert \sum_b w_b(x)(\exp(iw^T(v_b -x) - 1 - iw^T(v_b-x))  \biggr\rvert \\
        & \leq \frac{1}{2} \sum_{b} w_b(x) (w^T (v_b - x))^2 \\
        & \leq \frac{1}{2} \sum_{b} w_b(x) \lVertw\rVert_2^2 \lVertv_b - x\rVert_2^2 \\
        & \leq \frac{d}{2m^2} \lVertw\rVert_2^2,
    \end{align}
    where the first equality uses $\sum_{b}w_b(x)w^T(v_b - x) = 0 $ from \cref{eq:weight_facts},
    the first inequality applies \cref{eq:integral_bound},
    the second inequality applies Cauchy-Schwarz,
    and the last inequality uses $\lVertv_b - x\rVert_2 \leq \sqrt{d}/m$ from \cref{eq:weight_facts}.
    Then
    \begin{align}
        \lvertD_x(w)\rvert^2 \leq C_d\cdot \min \bigl\{1, m^{-4}  \lVertw\rVert^4_2  \bigr\}, \label{eq:dxw_bound}
    \end{align}
    where $C_d > 0 $ is a constant that depends on $d$.
    Combining
    \cref{eq:norm_bound,eq:Imf_bound} gives
    \begin{align}
        \lvertf(x) - I_mf(x)\rvert^2
        & \leq B^2  \int_{\mathbb{R}^d} \lvertD_x(w)\rvert^2 p(w) \ dw \\
        & \leq AB^2C_d  \int_{\mathbb{R}^d} \min \{1, m^{-4} \lVertw\rVert^4_2  \} (1 + \lVertw\rVert_2^2)^{-(s + d/2)} \ dw \\
        & \leq AB^2C_d \lvert\mathbb{S}^{d-1}\rvert \int_0^{\infty} \min \{1, m^{-4} r^4  \} (1 + r^2)^{-(s + d/2)}r^{d-1} \ dr,
    \end{align}
    where the second inequality uses \cref{eq:spectral_assumption} and \cref{eq:dxw_bound}, and the third inequality uses the polar-coordinate with $\lvert\mathbb{S}^{d-1}\rvert$ being the unit sphere's surface area. Now we decompose the integral according to the value of $r$.
    When $0 \leq r \leq 1$, $m^{-4} r^4 \leq 1$ and $(1+r^2)^{-(s+d/2)} \leq 1 $. Then the integral on $(0,1)$ is upper bounded by $\int_{0}^1 m^{-4} r^{d+3} dr = \frac{m^{-4}}{d+4} \leq m^{-4} $. For $1\leq r \leq m$, $(1+r^2)^{-(s+d/2)} \leq r^{2\cdot-(s+d/2)} $, and the integrand is at most $m^{-4} r^{3-2s} $. For $r \geq m$, the integral is at most $(1+r^2)^{-(s+d/2)} r^{d-1}\leq r^{-2s-d+d-1} = r^{-2s-1} $. Thus
    \begin{align}
        \lvertf(x) - I_mf(x)\rvert^2
        & \leq AB^2C_d\lvert\mathbb{S}^{d-1}\rvert \cdot \biggl[ m^{-4} + m^{-4}\int_{1}^{m} r^{3-2s} \ dr + \int_{m}^{\infty} r^{-2s-1} \biggr]. \label{eq:sum_integral}
    \end{align}
    
    
    
    
    
    For $0 < s < 2$, $m^{-4} \int_{1}^{m} r^{3 - 2s} \ dr = \frac{m^{-2s} - m^{-4}}{4 - 2s} \leq \frac{m^{-2s}}{4 - 2s}  $, and $\int_{m}^{\infty} r^{-2s-1} = \frac{m^{-2s}}{2s} $. Also $m^{-4} \leq m^{-2s}$. Hence,
    \begin{align}
        \lvertf(x) - I_mf(x)\rvert^2
        \leq CB^2 m^{-2s}.
    \end{align}
    Taking the square root and combining with \cref{eq:approximation_error} proves the result.

\label{cor:gamma_T_upper_bound}
    Consider the setup of \Cref{sec:setup_compact} where $\mathcal{X}$ is compact. 
    Suppose there exists a sequence $(U_t)_{t\in \mathbb{Z}_{+}}$ such that $R^*_t \leq U_t$ for every $t \in \mathbb{Z}_{+}$.
    Then,
    \begin{align}
        \gamma_T
        \leq \gamma_1 + \frac{1 + \lambda^{-1}}{c_1^2} \sum_{t=2}^T \frac{U_t^2}{t^2},
    \end{align}
    where $c_1 = \min \bigl\{ \frac{\sigma}{32}, \frac{\sigma \Delta_k}{16}  \bigr\}$.
