<!-- theorem-1: body_md -->
\label{thm:star-quotient}
Let $1\leq l<k\leq n$, and let $u\in C^3(\overline\Omega)$ solve
\eqref{eq:quotient-problem}. Assume that $\Omega$ is strictly
star-shaped with respect to a point $o$. If $K>0$, assume also that
$\overline\Omega$ lies in the open hemisphere centered at $o$.
Then
 $\Omega$ is a geodesic ball and $u$ is radial with respect
to its center.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:nonstar-quotient}
Let $1\leq l<k\leq n$, and let $u\in C^3(\overline\Omega)$ solve
\eqref{eq:quotient-problem}. If $K^2c_1\leq1$, then
 $\Omega$ is a geodesic ball and $u$ is radial with respect
to its center.
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm:sum}
Let $1\leq k\leq n$, and let $u\in C^3(\overline\Omega)$ solve
\eqref{eq:sum-problem}. If $k\geq2$, assume that $K^2c_1\leq1$.
Then
 $\Omega$ is a geodesic ball and $u$ is radial with respect
to its center.
<!-- end theorem-3 -->
<!-- remark-1: body_md -->
\begin{enumerate}
\item When $k=1$, the additional restriction on $c_1$ is not required.
\item When $\alpha=0$, Theorems~\ref{thm:star-quotient} and \ref{thm:sum}
recover the corresponding results of Gao, Ma and Yang
\cite[Theorems~1.1 and 1.4]{GaoMaYang}.
\item When $\alpha=0$, Theorem~\ref{thm:nonstar-quotient} gives a non-star-shaped
Hessian quotient result under $K^2c_1\leq1$, whereas the quotient
theorem in \cite{GaoMaYang} assumes star-shapedness.
For $K=0$, the condition $K^2c_1\leq1$ holds automatically.
\end{enumerate}
<!-- end remark-1 -->
<!-- lemma-1: body_md -->
\label{lem:cone-inclusion}
If $k\geq2$ and $A\in\widetilde\Gamma_k$, then $A\in\Gamma_{k-1}$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
We prove $E_j(A)>0$ successively for $1\leq j\leq k-1$.
Assume that $E_{j-1}>0$, starting with $E_0=1$.
If $E_j\leq0$, Newton's inequality \eqref{eq:Newton} gives
$$
 \begin{aligned}
 G_{j+1}
 &=E_{j+1}+\alpha E_j\\
 &\leq\frac{E_j^2}{E_{j-1}}+\alpha E_j
 =\frac{E_j}{E_{j-1}}G_j\leq0.
 \end{aligned}
$$
This contradicts $G_{j+1}>0$. The induction proves the assertion.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:automatic-cone}
Let $u\in C^2(\overline\Omega)$ solve either
\eqref{eq:quotient-problem} or \eqref{eq:sum-problem}.
Then $b\in\widetilde\Gamma_k$ on $\overline\Omega$.
For the quotient problem, one also has
\begin{equation}\label{eq:G-lower-bound}
 G_k(b)=G_l(b)\geq1\quad\text{on }\overline\Omega.
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
The function $u$ attains its minimum on $\overline\Omega$.
This minimum cannot occur at a boundary point $y$. Indeed, the
curve $\exp_y(-t\nu)$ lies in $\Omega$ for small $t>0$, and
$$
 \left.\frac{d}{dt}u(\exp_y(-t\nu))\right|_{t=0}=-c_2<0.
$$
Hence $u$ has an interior minimum at some $x_0$. At this point, we can see that
$$
 b(x_0)=\nabla^2u(x_0)+Ku(x_0)g\geq0.
$$
All elementary symmetric functions of $b(x_0)$ are nonnegative.
For the quotient problem, $G_l(b(x_0))\ne0$ and $G_k=G_l$,
so $G_k(b(x_0))>0$. For the sum problem,
$G_k(b(x_0))=1+\alpha>0$.

If $\alpha=0$, positivity of $G_k=E_k$ at a nonnegative
eigenvalue vector implies positivity of $E_j$ for $j\leq k$.
If $\alpha>0$, positivity of $G_k$ implies that at least $k-1$
eigenvalues are positive when $k\geq2$. Thus $G_j(b(x_0))>0$
for $j<k$ as well. The case $k=1$ follows directly from
$G_1(b(x_0))>0$. We have proved that
$b(x_0)\in\widetilde\Gamma_k$.

Let
$$
 \mathcal U=\{x\in\overline\Omega:b(x)\in\widetilde\Gamma_k\}.
$$
This set is nonempty and relatively open. We prove that it is
relatively closed.

First consider the quotient problem. On $\mathcal U\cap\Omega$,
the equation and \eqref{eq:G-ratios} give
$$
 1=\frac{G_k}{G_l}
 =\frac{G_{l+1}}{G_l}\cdots\frac{G_k}{G_{k-1}},
 \qquad
 \frac{G_1}{G_0}\geq\cdots\geq\frac{G_k}{G_{k-1}}>0.
$$
The first factor in the product is at least one. Each of the
preceding ratios is therefore also at least one. It follows that
$$
 G_l=\frac{G_1}{G_0}\cdots\frac{G_l}{G_{l-1}}\geq1.
$$
Continuity gives $G_k=G_l\geq1$ on all of $\mathcal U$.
For the sum problem, $G_k=1+\alpha$ on $\overline\Omega$ by
continuity.

Now take $x_m\in\mathcal U$ with $x_m\to x\in\overline\Omega$.
The positive form of \eqref{eq:G-Maclaurin} gives
$$
 G_j(b(x_m))\geq G_k(b(x_m))^{j/k},
 \qquad 1\leq j\leq k.
$$
In the quotient case the right-hand side is at least one.
In the sum case it equals $(1+\alpha)^{j/k}>0$.

Thus $\mathcal U$ is relatively closed.

<!-- lemma-3: body_md -->
\label{lem:individual-ellipticity}
If $A\in\widetilde\Gamma_k$, then $(G_j^{pq}(A))$ is positive
definite for every $1\leq j\leq k$.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
The assertion for $j=1$ follows from $G_1^{pq}=\delta_{pq}/n$.
Let $j\geq2$, and let $\lambda=(\lambda_1,\ldots,\lambda_n)$
be the eigenvalues of $A$. Set
$$
 \beta_j=\alpha\frac{n-j+1}{j}.
$$
The elementary symmetric function in $n+1$ variables satisfies
\begin{equation}\label{eq:augmented-G}
 \sigma_j(\lambda_1,\ldots,\lambda_n,\beta_j)
 =\sigma_j(\lambda)+\beta_j\sigma_{j-1}(\lambda)
 =\binom njG_j(\lambda)>0.
\end{equation}
It follows from  Lemma~\ref{lem:cone-inclusion} that $A\in\Gamma_{k-1}$.
For $m<j$, we therefore have
$$
 \sigma_m(\lambda_1,\ldots,\lambda_n,\beta_j)
 =\sigma_m(\lambda)+\beta_j\sigma_{m-1}(\lambda)>0.
$$
Thus $(\lambda,\beta_j)\in\Gamma_j$ in dimension $n+1$.

The elementary symmetric function $\sigma_j$ is elliptic on
$\Gamma_j$; see \cite{CNS} and
\cite[Proposition~2.3]{GaoMaYang}. Hence, for every $i$,
$$
 \frac{\partial G_j}{\partial\lambda_i}
 =\binom nj^{-1}
 \left(\sigma_{j-1}(\lambda\mid i)
       +\beta_j\sigma_{j-2}(\lambda\mid i)\right)>0.
$$
Here $\lambda\mid i$ means that the $i$-th entry is omitted.
In an eigenframe of $A$, these derivatives are the eigenvalues
of $(G_j^{pq})$. The first-variation formula for a symmetric
polynomial also holds at repeated eigenvalues. Therefore
$(G_j^{pq})$ is positive definite.

<!-- proposition-1: body_md -->
\label{prop:quotient-ellipticity}
Let $1\leq l<k\leq n$. The operator
$$
 F(A)=\frac{G_k(A)}{G_l(A)}
$$
is elliptic on $\widetilde\Gamma_k$.
<!-- end proposition-1 -->
<!-- proof-4: body_md -->
Fix an eigenvalue vector $\lambda$ in $\widetilde\Gamma_k$
and an index $i$. By Lemma~\ref{lem:individual-ellipticity},
$$
 a_j=\frac{\partial G_j}{\partial\lambda_i}(\lambda)>0,
 \qquad 1\leq j\leq k.
$$
Each $G_j$ is affine in $\lambda_i$. Consequently,
$$
 g_j(t)=G_j(\lambda+te_i)=G_j(\lambda)+a_jt.
$$
Set
$$
 t_0=\max_{1\leq j\leq k}
 \left(-\frac{G_j(\lambda)}{a_j}\right)<0.
$$
Then $g_j(t)>0$ for $t>t_0$, while $g_j(t_0)\geq0$ for all
$j\leq k$, and at least one of these values vanishes.

Let $r$ be the first index for which $g_r(t_0)=0$.
We claim that $r=k$. Suppose that $r<k$.
If $r=1$, \eqref{eq:first-Newton} and continuity give
$g_2(t_0)=0$. Applying the same inequality at $t_0+s$,
where $s>0$, gives
$$
 a_1^2s^2\geq a_2s.
$$
Division by $s$ and passage to the limit $s\downarrow0$
contradict $a_2>0$.

If $r\geq2$, then $g_{r-1}(t_0)>0$. Ren's inequality
\eqref{eq:Ren}, first at $t>t_0$ and then by continuity,
gives $g_{r+1}(t_0)=0$. At $t=t_0+s$, it reads
$$
 a_r^2s^2\geq
 \bigl(g_{r-1}(t_0)+a_{r-1}s\bigr)a_{r+1}s.
$$
After division by $s$ and passage to the limit, we obtain
$0\geq g_{r-1}(t_0)a_{r+1}>0$, a contradiction.
Thus $r=k$, and
$$
 g_k(t_0)=0,\qquad g_l(t_0)>0.
$$

The affine formulas now yield
$$
 a_kG_l(\lambda)-a_lG_k(\lambda)=a_kg_l(t_0)>0.
$$
Therefore
\begin{equation}\label{eq:quotient-eigenvalue-derivative}
 \frac{\partial F}{\partial\lambda_i}
 =\frac{a_kG_l-a_lG_k}{G_l^2}>0.
\end{equation}
This holds for every $i$. The spectral first-variation formula
then shows that $(F^{pq})$ is positive definite.

<!-- lemma-4: body_md -->
\label{lem:quotient-algebra}
Let $A\in\widetilde\Gamma_k$, where $1\leq l<k\leq n$, and
assume $G_k(A)=G_l(A)=G$. Then
\begin{equation}\label{eq:quotient-neighbours}
 G\geq1,\qquad
 G_{k-1}\geq G,\qquad
 G_{l-1}\leq G,\qquad
 G_{l+1}\geq G.
\end{equation}
If $k<n$, one also has $G_{k+1}\leq G$. Moreover,
\begin{equation}\label{eq:quotient-E}
 E_1\geq1,\qquad
 E_{k-1}-E_{l-1}\geq0,\qquad
 E_l-E_k=\alpha(E_{k-1}-E_{l-1}).
\end{equation}
If $E_{k-1}=E_{l-1}$, then $A=I$.
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
We suppress the argument $A$ in this proof. The ratios in
\eqref{eq:G-ratios} are non-increasing, and
$$
 \frac{G_{l+1}}{G_l}\cdots\frac{G_k}{G_{k-1}}=1.
$$
Their first factor is at least one and their last factor is at
most one. Hence
$$
 \frac{G_l}{G_{l-1}}\geq
 \frac{G_{l+1}}{G_l}\geq1
 \geq\frac{G_k}{G_{k-1}}.
$$
These inequalities give all the comparisons in
\eqref{eq:quotient-neighbours}. In particular, the first $l$
ratios are at least one, so their product $G_l$ is at least one.
If $k<n$, Ren's inequality at index $k$ gives
$$
 G_{k+1}\leq\frac{G_k^2}{G_{k-1}}\leq G.
$$
This estimate does not require $G_{k+1}>0$.

Since $l\geq1$, the ratio $G_2/G_1$ precedes or equals
$G_{l+1}/G_l$. By Newton's inequality and $G_1>0$,
$$
 1\leq\frac{G_2}{G_1}
 =\frac{E_2+\alpha E_1}{E_1+\alpha}
 \leq\frac{E_1^2+\alpha E_1}{E_1+\alpha}=E_1.
$$

Next, the equality $G_k=G_l$ gives
\begin{equation}\label{eq:E-comparison}
 G(E_{k-1}-E_{l-1})
 =E_{k-1}E_l-E_{l-1}E_k.
\end{equation}
Lemma~\ref{lem:cone-inclusion} gives $E_j>0$ for
$j\leq k-1$. Thus the right-hand side of
\eqref{eq:E-comparison} is positive if $E_k\leq0$.
If $E_k>0$, then $A\in\Gamma_k$, and
\eqref{eq:Maclaurin} gives
$$
 \frac{E_k}{E_{k-1}}\leq\frac{E_l}{E_{l-1}}.
$$
The right-hand side of \eqref{eq:E-comparison} is again
nonnegative. This proves $E_{k-1}\geq E_{l-1}$.
The last identity in \eqref{eq:quotient-E} follows by
subtracting $G_k=G_l$.

Suppose that $E_{k-1}=E_{l-1}$. The case $E_k\leq0$
has been excluded. Equality therefore holds between two
distinct positive Newton quotients in $\Gamma_k$.
The equality case in \eqref{eq:Maclaurin} shows that all
eigenvalues equal a number $a>0$. The equation becomes
$$
 a^{l-1}(a+\alpha)(a^{k-l}-1)=0.
$$
Since $a+\alpha>0$, we obtain $a=1$.

<!-- lemma-5: body_md -->
\label{lem:quotient-P}

$$
 F^{ij}\nabla_{ij}^2P\geq0\quad\text{in }\Omega.
$$

<!-- proof-6: body_md -->
Differentiation gives
\begin{equation}\label{eq:P-gradient}
 \frac12\nabla_iP
 =u_mu_{mi}+Kuu_i-u_i=b_{mi}u_m-u_i.
\end{equation}
Using the Codazzi identity \eqref{eq:Codazzi}, we obtain
\begin{align}
 \frac12\nabla_{ij}^2P
 &=(\nabla_jb_{mi})u_m+b_{mi}u_{mj}-u_{ij}\notag\\
 &=(\nabla_mb_{ij})u_m
   +b_{mi}(b_{mj}-Ku\delta_{mj})
   -(b_{ij}-Ku\delta_{ij})\notag\\
 &=(\nabla_mb_{ij})u_m+b_{mi}b_{mj}
   -(Ku+1)b_{ij}+Ku\delta_{ij}.
 \label{eq:P-Hessian}
\end{align}
The equation $F=1$ implies
$F^{ij}\nabla_mb_{ij}=0$. On the solution,
$$
 F^{ij}=\frac{G_k^{ij}-G_l^{ij}}G.
$$
The identities \eqref{eq:Euler-G}--\eqref{eq:square-G} therefore give
\begin{align}
 F^{ij}b_{ij}
 &=k-l-\alpha\frac{E_{k-1}-E_{l-1}}G,
 \label{eq:Fb}\\
 F^{ij}\delta_{ij}
 &=\frac{(k-1)G_{k-1}-(l-1)G_{l-1}
          +E_{k-1}-E_{l-1}}G,
 \label{eq:Ftrace}\\
 F^{ij}b_{mi}b_{mj}
 &=\frac{-(n-k)G_{k+1}+(n-l)G_{l+1}
          -\alpha(E_k-E_l)}G.
 \label{eq:Fsquare}
\end{align}
The term containing $G_{k+1}$ is absent when $k=n$.

Lemma~\ref{lem:quotient-algebra} yields
$$
 \begin{aligned}
 F^{ij}\delta_{ij}
 &\geq k-l+\frac{E_{k-1}-E_{l-1}}G,\\
 F^{ij}b_{mi}b_{mj}
 &\geq k-l+\alpha^2\frac{E_{k-1}-E_{l-1}}G.
 \end{aligned}
$$
Contracting \eqref{eq:P-Hessian}, and using $Ku\geq0$ in
the trace term, gives
\begin{align}
 \frac12F^{ij}\nabla_{ij}^2P
 &\geq k-l+\alpha^2\frac{E_{k-1}-E_{l-1}}G\notag\\
 &\quad-(Ku+1)
    \left(k-l-\alpha\frac{E_{k-1}-E_{l-1}}G\right)\notag\\
 &\quad+Ku\left(k-l+\frac{E_{k-1}-E_{l-1}}G\right)\notag\\
 &=\frac{(1+\alpha)(\alpha+Ku)}G
       (E_{k-1}-E_{l-1})\geq0.
 \label{eq:quotient-LP}
\end{align}
The operator is uniformly elliptic on $\overline\Omega$.
The maximum principle, its strong form, and
\eqref{eq:P-boundary} give the assertions;
see \cite[Chapter~3]{GilbargTrudinger}.

 \label{eq:Pohozaev}
\end{align}

\begin{align}
 &\int_\Omega(Kc_1-u)
 \bigl(H^{ij}\delta_{ij}-H^{ij}b_{ij}\bigr)V\,d\mu\notag\\

 \label{eq:Pohozaev-P}
\end{align}

We follow the integration by parts used for the $k$-Hessian
equation in \cite[Section~6]{GaoMaYang}. The proof below uses
only the properties in Lemma~\ref{lem:H-properties}.
Since $\Phi_{lj}=V\delta_{lj}$ and
$b_{ij}=u_{ij}+Ku\delta_{ij}$,
\begin{align}
 H^{ij}b_{ij}uV
 &=\nabla_j(H^{ij}u_{il}u\Phi_l)
   -H^{ij}(\nabla_ju_{il})u\Phi_l\notag\\
 &\quad-H^{ij}u_{il}u_j\Phi_l
   +Ku^2VH^{ij}\delta_{ij}.
 \label{eq:Pohozaev-start}
\end{align}
The Codazzi identity and \eqref{eq:H-properties} imply
$$
 \begin{aligned}
 H^{ij}(\nabla_ju_{il})\Phi_l
 &=H^{ij}(\nabla_jb_{il}-Ku_j\delta_{il})\Phi_l\\
 &=H^{ij}\nabla_lb_{ij}\,\Phi_l-KH^{ij}u_j\Phi_i\\
 &=-KH^{ij}u_j\Phi_i.
 \end{aligned}
$$
Thus
\begin{equation}\label{eq:Pohozaev-u-square}
 -H^{ij}(\nabla_ju_{il})u\Phi_l
 =\frac K2\nabla_j(H^{ij}\Phi_i u^2)
  -\frac K2H^{ij}\delta_{ij}u^2V.
\end{equation}

$$
 H^{ij}u_{il}u_j\Phi_l
 =H^{li}u_{ij}u_j\Phi_l

\begin{equation}\label{eq:Pohozaev-gradient}

 =-\frac12\nabla_i(H^{li}\Phi_l|\nabla u|^2)

\end{equation}

\eqref{eq:Pohozaev-gradient} into \eqref{eq:Pohozaev-start},

 =\int_\Omega H^{ij}\delta_{ij}V\,d\mu.
\end{equation}
Second, the Codazzi identity gives
$$
 \begin{aligned}
 \int_{\partial\Omega}H^{ij}b_{il}\Phi_l\nu_j\,dS
 &=\int_\Omega
    \left(H^{ij}\nabla_jb_{il}\,\Phi_l
          +H^{ij}b_{il}\Phi_{lj}\right)d\mu\\

    \left(H^{ij}\nabla_lb_{ij}\,\Phi_l

 \end{aligned}

\begin{equation}\label{eq:boundary-uHess}

$$

 &=Kc_1\int_\Omega H^{ij}b_{ij}V\,d\mu\\

  \bigl(|\nabla u|^2+Ku^2-K^3c_1^2-c_2^2\bigr)
  H^{ij}\delta_{ij}V\,d\mu.
 \end{aligned}
$$
This is \eqref{eq:Pohozaev}. Finally,

$$
 \frac12G_1^{ij}\nabla_{ij}^2P
 =\frac1n|b|^2-1\geq0.
$$
The last inequality is the Cauchy--Schwarz inequality
$|b|^2\geq(\operatorname{tr}b)^2/n=n$.
Equality holds only when $b=g$. The maximum principle
gives the assertions in this case.

Assume now that $k\geq2$.
We first prove $E_{k-1}\geq1$.
Lemma~\ref{lem:cone-inclusion} gives $E_{k-1}>0$.
If $E_{k-1}<1$ and $E_k\leq0$, then
$E_k+\alpha E_{k-1}<1+\alpha$, contrary to the equation.
If $E_{k-1}<1$ and $E_k>0$, then $b\in\Gamma_k$, and
the Maclaurin inequality \eqref{eq:Maclaurin} gives
$$
 E_k\leq E_{k-1}^{k/(k-1)}<1.
$$
Again $G_k<1+\alpha$, a contradiction.
Thus $E_{k-1}\geq1$. Applying \eqref{eq:Maclaurin}
in $\Gamma_{k-1}$ gives $E_1\geq1$ and, if $k\geq3$,
$E_{k-2}\geq1$. For $k=2$, the last assertion is
$E_0=1$. This proves \eqref{eq:sum-basic}.

The equation also gives
\begin{equation}\label{eq:sum-Ek}
 E_k=1-\alpha(E_{k-1}-1).
\end{equation}
Using \eqref{eq:Euler-G}, \eqref{eq:trace-G} and
\eqref{eq:sum-Ek}, we obtain the exact identity
\begin{align}
 G_k^{ij}\delta_{ij}-G_k^{ij}b_{ij}
 &=k(E_{k-1}-E_k)
   +\alpha(k-1)(E_{k-2}-E_{k-1})\notag\\
 &=(k+\alpha)(E_{k-1}-1)
   +\alpha(k-1)(E_{k-2}-1)\geq0.
 \label{eq:sum-trace-exact}
\end{align}
This proves \eqref{eq:sum-trace-gap}.

We next estimate the quadratic contraction.
Equations \eqref{eq:Euler-G}, \eqref{eq:square-G} and
\eqref{eq:sum-Ek} give
\begin{align}
 &G_k^{ij}b_{mi}b_{mj}-G_k^{ij}b_{ij}\notag\\
 &=n(1+\alpha)E_1-(n-k)E_{k+1}
   -\alpha(n-k+1)E_k-k(1+\alpha)+\alpha E_{k-1}\notag\\
 &=n(1+\alpha)(E_1-1)+(n-k)(1-E_{k+1})\notag\\
 &\quad+
   \bigl(\alpha^2(n-k+1)+\alpha\bigr)(E_{k-1}-1).
 \label{eq:sum-square-first}
\end{align}
Newton's inequality \eqref{eq:Newton} gives
$$
 E_{k+1}\leq\frac{E_k^2}{E_{k-1}}.
$$
For $k=n$, we use $E_{n+1}=0$ and the terms multiplied
by $n-k$ vanish. Notice that this use of Newton's inequality
does not require $E_k>0$.

To separate the nonnegative terms in
\eqref{eq:sum-square-first}, write
$$
 1-E_{k+1}
 =1-\frac{E_k^2}{E_{k-1}}
  +\frac{E_k^2}{E_{k-1}}-E_{k+1}.
$$
Equation \eqref{eq:sum-Ek} gives
$$
 1-\frac{E_k^2}{E_{k-1}}
 =(E_{k-1}-1)
   \left(\frac{(1+\alpha)^2}{E_{k-1}}-\alpha^2\right).
$$
Consequently,
$$
 \begin{aligned}
 &(n-k)\left(1-\frac{E_k^2}{E_{k-1}}\right)
   +\bigl(\alpha^2(n-k+1)+\alpha\bigr)(E_{k-1}-1)\\
 &\qquad=
 (1+\alpha)
 \left(\alpha+\frac{(n-k)(1+\alpha)}{E_{k-1}}\right)
 (E_{k-1}-1).
 \end{aligned}
$$
Substitution into \eqref{eq:sum-square-first} now yields
\begin{align}
 &G_k^{ij}b_{mi}b_{mj}-G_k^{ij}b_{ij}\notag\\
 &=n(1+\alpha)(E_1-1)
   +(n-k)\left(\frac{E_k^2}{E_{k-1}}-E_{k+1}\right)\notag\\
 &\quad+
 (1+\alpha)
 \left(\alpha+\frac{(n-k)(1+\alpha)}{E_{k-1}}\right)
 (E_{k-1}-1)\geq0.
 \label{eq:sum-square-gap}
\end{align}
Each term on the right is nonnegative by
\eqref{eq:sum-basic} and Newton's inequality.

Since $G_k$ is constant, contraction of
\eqref{eq:P-Hessian} gives
\begin{align}
 \frac12G_k^{ij}\nabla_{ij}^2P
 &=G_k^{ij}b_{mi}b_{mj}-G_k^{ij}b_{ij}\notag\\
 &\quad+Ku\bigl(G_k^{ij}\delta_{ij}-G_k^{ij}b_{ij}\bigr).
 \label{eq:sum-P-decomposition}
\end{align}
Equations \eqref{eq:sum-trace-exact} and
\eqref{eq:sum-square-gap}, together with $Ku\geq0$,
prove \eqref{eq:sum-LP}. The maximum principle and
$P=P_0$ on $\partial\Omega$ give $P\leq P_0$.

Suppose that $P$ is constant. Both terms on the right-hand
side of \eqref{eq:sum-P-decomposition} are nonnegative.
The first must therefore vanish. Its decomposition in
\eqref{eq:sum-square-gap} implies $E_1=1$. On the other hand,
\eqref{eq:sum-basic} and the Maclaurin inequality in
$\Gamma_{k-1}$ give
$$
 1\leq E_{k-1}\leq E_1^{k-1}=1.
$$
Hence $E_{k-1}=1$, and \eqref{eq:sum-Ek} gives $E_k=1$.
We now have $b\in\Gamma_k$ and equality in
$E_k^{1/k}\leq E_1$. The equality case in
\eqref{eq:Maclaurin} gives $b=g$.

Identity \eqref{eq:Pohozaev-P} reduces to

 \int_\Omega(P-P_0)V\,d\mu=0.

Lemma~\ref{lem:sum-P} gives $P\leq P_0$, and $V>0$.

gives $b=g$. The rigidity argument in the final part of

This argument does not use $K^2c_1\leq1$.

Assume that $k\geq2$. By \eqref{eq:sum-basic},

 \Delta w+\frac2V\langle\nabla V,\nabla w\rangle
 =\frac{n(E_1-K^2c_1)}V\geq0.
$$
Since $w=0$ and $w_\nu=c_2/V>0$ on $\partial\Omega$,

 \int_\Omega(P-P_0)G_k^{ij}\delta_{ij}V\,d\mu=0.
