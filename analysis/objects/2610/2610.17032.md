\label{thm:intrinsic_recovery_intro}
Let $M$ be a compact smooth manifold without boundary of dimension $d\ge1$, fix $t>0$, and let $\Theta_{\mathrm{int}}$ be a compact class as in Theorem~\ref{thm:top_embedding_intrinsic_intro}. Fix $(g_0,p_0)\in\Theta_{\mathrm{int}}$ and draw $X_1,X_2,\ldots$ independently from $p_0\,\mu_{g_0}$. Form the empirical operator $L_{n,t}^{d_{g_0}}$ on $C(M)$ as in Definition~\ref{def:discrete_GL}. Every measurable exact or asymptotic minimum-distance estimator $(\widehat g_n,\widehat p_n)$ over $\Theta_{\mathrm{int}}$, as in Definition~\ref{def:minimum_distance_estimator} and fitted to this operator using $d_*$ and $\Phi_t^{\mathrm{int}}$, satisfies
\[
\widehat g_n\to g_0\quad\text{in }C^2,\qquad \widehat p_n\to p_0\quad\text{uniformly on }M,
\]
both in probability. For an exact estimator, or an asymptotic estimator whose defining approximation error tends to zero almost surely, both conclusions also hold almost surely.

\label{def:discrete_GL}
Let $\ell$ be a continuous (possibly extended) distance on $M$, with the convention $k(+\infty)=0$. The corresponding unnormalized discrete graph Laplace operator is the operator on $C(M)$ defined at every $x\in M$ by
\begin{equation}\label{eq:discrete_GL}
(L_{n,t}^{\ell} f)(x)
:= \frac{1}{n\,t^{d/2+1}} \sum_{j=1}^n k\!\left(\frac{\ell(x,X_j)^2}{t}\right)\bigl(f(x)-f(X_j)\bigr), \qquad f \in C(M).
\end{equation}

<!-- definition-2: body_md -->
\label{def:continuous_GL}
Let $\ell$ be a continuous (possibly extended) distance on $M$. The corresponding unnormalized continuous graph Laplace operator is defined by
\begin{equation}\label{eq:continuous_GL}
L_t^{\ell} f(x)
:= \frac{1}{t^{d/2+1}} \int_M k\!\left(\frac{\ell(x,y)^2}{t}\right)\bigl(f(x)-f(y)\bigr) p(y)\,d\mu_g(y), \qquad f \in C(M).
\end{equation}

\label{def:weak-metric}
   Since $M$ is compact, the Banach space $C(M)$ is separable. Fix once and for all a countable dense subset $\{f_m\}_{m\ge 1}$ of the unit ball $\{f\in C(M):\left\lVertf\right\rVert_\infty\le 1\}$, and a countable dense subset $\{z_q\}_{q\ge 1}$ of $M$. For $S,T\in \mathcal L(C(M),C(M))$, define
\[
d_{*}(S,T):=\sum_{m=1}^{\infty}\sum_{q=1}^{\infty}2^{-(m+q)}\min\Bigl\{1,\bigl|Sf_m(z_q)-Tf_m(z_q)\bigr|\Bigr\}.
\]

<!-- definition-10: body_md -->
\label{def:minimum_distance_estimator}
Let $\Theta$ be a compact subset of a model class, let $\Phi:\Theta\to\mathcal L(C(M),C(M))$ be a map, and let $d_*$ be the operator metric in Definition \ref{def:weak-metric}. Fix a \textit{true parameter} $\theta_0\in\Theta$ and write $T_0:=\Phi(\theta_0)$ for its \textit{true operator}.

For each $n\ge 1$, let $\widehat T_n$ be a $\mathcal L(C(M),C(M))$-valued random element built from the data $X_1,\dots,X_n$. The main examples in this paper are the intrinsic and extrinsic discrete graph Laplace operators at fixed $t>0$.

Define the \textit{criterion function} $Q_n(\theta):=d_*(\widehat T_n,\Phi(\theta))$ for $\theta\in\Theta$.

A $\Theta$-valued random element $\widehat\theta_n$ is an \emph{exact minimum-distance estimator w.r.t. the weak metric $d_*$} if $$Q_n(\widehat\theta_n)=\inf_{\theta\in\Theta}Q_n(\theta).$$ It is an \emph{asymptotic minimum-distance estimator w.r.t. the weak metric $d_*$} if
\[
\inf_{\theta\in\Theta}Q_n(\theta) \le Q_n(\widehat\theta_n)\le \inf_{\theta\in\Theta}Q_n(\theta)+r_n,
\]
where $r_n$ is a nonnegative random variable with $r_n\to 0$ in probability, or almost surely, according to the mode of convergence being considered. The defining inequality is required to hold almost surely for each $n$.

\label{thm:top_embedding_intrinsic_intro}

\[

\]

<!-- definition-11: body_md -->
\label{def:rigid_motion_equiv}
Let $D\in \mathbb{N}$, and let $\operatorname{Emb}^{2}(M,\mathbb{R}^{D})$ denote the space of $C^{2}$ embeddings $\iota:M\to\mathbb{R}^{D}$, endowed with the $C^{2}$-topology.

Define an equivalence relation $\sim$ on $\operatorname{Emb}^{2}(M,\mathbb{R}^{D})$ by declaring $\iota_1\sim \iota_2$ if there exist $A\in O(D)$ and $b\in \mathbb{R}^{D}$ such that $\iota_2 = A\circ \iota_1 + b$. Denote by $[\iota]$ the equivalence class of $\iota$, and let
\[
\mathcal E^{2}(M,\mathbb{R}^{D}):=\operatorname{Emb}^{2}(M,\mathbb{R}^{D})/\sim
\]
be the quotient space, endowed with the quotient topology.

This quotient is metrizable. Indeed, choose a $C^2$ norm on maps $M\to\mathbb{R}^D$ using a reference metric on $M$ and Euclidean norms on values and derivatives. The rigid-motion group acts by isometries for the associated distance between maps. The formula
\[
d_{\mathcal E}([\iota],[\kappa]):=\inf_{A\in O(D),\,b\in\mathbb{R}^D}\left\lVert\kappa-(A\iota+b)\right\rVert_{C^2}
\]
defines a metric: if the infimum is zero, take a sequence $(A_j,b_j)$ realizing the infimum asymptotically; compactness of $O(D)$ and boundedness of $b_j$ (by evaluation at one point of $M$) give a subsequence with $\kappa=A\iota+b$. The triangle inequality follows from the isometric group action. Balls for $d_{\mathcal E}$ are images under the quotient map of $C^2$ balls, so this metric induces precisely the quotient topology. In particular, compact parameter subclasses of $\mathcal E^2(M,\mathbb{R}^D)\times C(M)$ are compact metrizable spaces.

For any fixed $t>0$ and $[\iota]\in \mathcal E^{2}(M,\mathbb{R}^{D})$, the kernel
\[
K_{[\iota]}(x,y):=k\!\left(\frac{\left\lVert\iota(x)-\iota(y)\right\rVert^{2}}{t}\right)
\]
is well-defined, because rigid motions preserve Euclidean distances. Likewise, the induced metric
\[
g_{[\iota]}:=\iota^{*}\langle\cdot,\cdot\rangle_{\mathbb{R}^{D}}
\]
is well-defined.

<!-- lemma-2: body_md -->
\label{lem:metric_determined_by_distance}
Let $M$ be a smooth manifold and let $g_1,g_2$ be $C^2$ Riemannian metrics on $M$ with geodesic distance functions $d_{g_1},d_{g_2}$. If
\[
d_{g_1}(x,y)=d_{g_2}(x,y)\qquad\text{for all }x,y\in M,
\]
then $g_1=g_2$.
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
Fix $x\in M$, choose a coordinate neighborhood, and let $y_s$ have coordinate difference $sv$ from $x$, for $v\in T_xM$. For each $i$ and every $\varepsilon>0$, continuity of $g_i$ gives a smaller coordinate neighborhood in which, under coordinate identification of tangent spaces, $(1-\varepsilon)g_i(x)\leq g_i(z)\leq(1+\varepsilon)g_i(x)$ as quadratic forms. The straight coordinate segment from $x$ to $y_s$ has length at most $\sqrt{1+\varepsilon}\,|s|\sqrt{g_i(x)(v,v)}$. For sufficiently small $s$, any curve leaving this smaller neighborhood has length bounded away from zero, while the segment length tends to zero. For curves remaining inside, comparison with the constant quadratic form $g_i(x)$ gives the lower bound $\sqrt{1-\varepsilon}\,|s|\sqrt{g_i(x)(v,v)}$. Taking infima over curves, then $s\to0$ and $\varepsilon\to0$, yields
\[
\lim_{s\to0}\frac{d_{g_i}(x,y_s)^2}{s^2}=g_i(x)(v,v).
\]
Equality of the distances gives equality of these quadratic forms for every $x,v$; polarization gives $g_1=g_2$.

<!-- proof-4: body_md -->
Write $\mu_i:=\mu_{g_i}$ and $K_i(x,y):=k(d_{g_i}(x,y)^2/t)$. For $i=1,2$ define the integral operators
\[
(T_if)(x):=\int_M K_i(x,y)\, f(y)\,p(y)\,d\mu_i(y), \qquad f\in C(M).
\]
Then
\[
(L^{(g_i)}_{t,p} f)(x)= \frac{1}{t^{d/2+1}}\Big((T_i1)(x) f(x) - (T_if)(x)\Big).
\]
Assuming $L^{(g_1)}_{t,p}=L^{(g_2)}_{t,p}$, we obtain for all $f\in C(M)$ and all $x\in M$,
\begin{equation}\label{eq:T_identity_metric}
(T_1f)(x) - (T_2f)(x) = \big((T_11)(x) - (T_21)(x)\big) f(x).
\end{equation}
Define $a(x):=(T_11)(x) - (T_21)(x)$.

\paragraph{Step 1: $T_11=T_21$ and hence $T_1=T_2$.}
Fix $x\in M$ and define the Borel measures
\[
m_{i,x}(B):=\int_B K_i(x,y)p(y)\,d\mu_i(y), \qquad B\subset M \text{ Borel}.
\]
Then $(T_if)(x)=\int_M f(y)\,d m_{i,x}(y)$ for all $f\in C(M)$, so \eqref{eq:T_identity_metric} becomes
\[
\int_M f(y)\,d(m_{1,x}-m_{2,x})(y)=a(x)f(x) \qquad (\forall f\in C(M)).
\]
By the Riesz representation theorem, the functional $f\mapsto a(x)f(x)$ is represented by the measure $a(x)\delta_x$, where $\delta_x$ denotes the Dirac mass at $x$, hence
\begin{equation}\label{eq:measure_identity_metric}
m_{1,x}-m_{2,x}=a(x)\delta_x\qquad\text{as finite signed measures.}
\end{equation}
Evaluating \eqref{eq:measure_identity_metric} on $\{x\}$ yields
\[
(m_{1,x}-m_{2,x})(\{x\})=a(x).
\]
On the other hand,
\[
m_{i,x}(\{x\}) = \int_{\{x\}} K_i(x,y)p(y)\,d\mu_i(y)=0,
\]
since $\mu_i(\{x\})=0$ for the Riemannian volume measure $\mu_i$. Thus $(m_{1,x}-m_{2,x})(\{x\})=0$, hence $a(x)=0$. Since $x$ was arbitrary, $a\equiv0$, so $T_11=T_21$ and \eqref{eq:T_identity_metric} implies $T_1=T_2$.

\paragraph{Step 2: identification of the kernel and the distance.}
Fix $x\in M$. From $T_1=T_2$ we have, for all $f\in C(M)$,
\[
\int_M f(y)K_1(x,y)p(y)\,d\mu_1(y)=\int_M f(y)K_2(x,y)p(y)\,d\mu_2(y).
\]
Since $\mu_1$ and $\mu_2$ are Riemannian volume measures on $M$, they are mutually absolutely continuous; write $d\mu_2=w\,d\mu_1$ for a continuous function $w>0$ using the local chart expression $d\mu_i=\sqrt{\det g_i}$. Then
\[
\int_M f(y)\big(K_1(x,y)-K_2(x,y)w(y)\big)p(y)\,d\mu_1(y)=0\qquad(\forall f\in C(M)).
\]
Hence
\[
\big(K_1(x,y)-K_2(x,y)w(y)\big)p(y)=0\qquad\text{for }\mu_1\text{-a.e. }y.
\]
Using $p>0$, this gives
\[
K_1(x,y)-K_2(x,y)w(y)=0\qquad\text{for }\mu_1\text{-a.e. }y.
\]
The map $y\mapsto K_1(x,y)-K_2(x,y)w(y)$ is continuous since $K_i$ and $w$ are so, and $\mu_1$ has full support, so the equality holds for all $y\in M$. Setting $y=x$ gives $k(0)=k(0)w(x)$, hence $w\equiv1$ and $\mu_1=\mu_2$. Therefore $K_1(x,y)=K_2(x,y)$ for all $x,y$. Strict decrease of $k$ gives $d_{g_1}=d_{g_2}$ on each connected component; both distances are infinite between components. By Lemma~\ref{lem:metric_determined_by_distance}, $g_1=g_2$.

\[

\qquad

\]
where $\rho_\ell,\rho\in C(M)$ are positive. Since $g_\ell\to g$ in $C^2$, in particular $g_\ell\to g$ uniformly. It follows from the local coordinate expression of the Riemannian volume density that $\rho_\ell\to \rho$ uniformly on $M$.

Now define
\[
K_\ell(x,y):=k\!\left(\frac{d_{g_\ell}(x,y)^2}{t}\right),
\qquad

\[
(1-\varepsilon)g \le g_\ell \le (1+\varepsilon)g
\]
as quadratic forms on $TM$. Comparing lengths of $C^1$ curves and taking infima yields
\[
\sqrt{1-\varepsilon}\, d_g(x,y)\le d_{g_\ell}(x,y)\le \sqrt{1+\varepsilon}\, d_g(x,y)
\qquad (x,y\text{ in the same connected component}).
\]
Each of the finitely many connected components of compact $M$ has finite $g$-diameter, so the distances converge uniformly on the union of the same-component products. Uniform continuity of $k$ on the resulting bounded range gives uniform convergence of the kernels there. On pairs in different components both kernels vanish by our convention. Therefore $K_\ell\to K$ uniformly on $M\times M$.

Since $p_\ell\rho_\ell\to p\rho$ uniformly, for every $\|f\|_\infty\le1$ and $x\in M$ the defining integral gives
\[
\bigl|((L^{(g_\ell)}_{t,p_\ell}-L^{(g)}_{t,p})f)(x)\bigr|
\le 2t^{-d/2-1}\mu_h(M)\,\|K_\ell p_\ell\rho_\ell-Kp\rho\|_{L^\infty(M\times M)}\to0.

\paragraph{Step 1: continuity of $\Phi_t^{\mathrm{ext}}$.}
Let $q:\operatorname{Emb}^{2}(M,\mathbb{R}^{D})\to \mathcal E^{2}(M,\mathbb{R}^{D})$ be the quotient map. Consider the lifted map
\[
\widetilde{\Phi}_t^{\mathrm{ext}}:\operatorname{Emb}^{2}(M,\mathbb{R}^{D})\times C(M)\to \mathcal L(C(M),C(M)),
\]
defined by
\[
\widetilde{\Phi}_t^{\mathrm{ext}}(\iota,p):=L_{t,p}^{\mathrm{ext}}(\iota^{*}\langle\cdot,\cdot\rangle_{\mathbb{R}^{D}};\iota).
\]
Rigid motions preserve both the kernel and the induced metric, so $\widetilde{\Phi}_t^{\mathrm{ext}}$ is invariant in its first argument. The map $q$ is open: for open $U\subset\operatorname{Emb}^{2}(M,\mathbb{R}^{D})$, its saturation $q^{-1}(q(U))$ is the union of the rigid-motion translates of $U$, each open. Therefore $q\times\mathrm{id}_{C(M)}$ is an open quotient map. Once continuity of the lifted map is shown, the quotient universal property gives continuity of the descended map on $\mathcal E^2(M,\mathbb{R}^D)\times C(M)$, and hence of its restriction $\Phi_t^{\mathrm{ext}}$ to $\Theta_{\mathrm{ext}}$.

Let $(\iota_\ell,p_\ell)\to (\iota,p)$ in $\operatorname{Emb}^{2}(M,\mathbb{R}^{D})\times C(M)$, that is, $\iota_\ell\to \iota$ in $C^{2}$ and $p_\ell\to p$ uniformly on $M$. Set
\[
g_\ell:=\iota_\ell^{*}\langle\cdot,\cdot\rangle_{\mathbb{R}^{D}},\qquad g:=\iota^{*}\langle\cdot,\cdot\rangle_{\mathbb{R}^{D}}.
\]
Then $g_\ell\to g$ in $C^{1}$, hence uniformly on $M$. Fix a smooth reference metric $h$ on $M$, and write
\[
d\mu_{g_\ell}=\rho_\ell\,d\mu_h,\qquad d\mu_{g}=\rho\,d\mu_h.
\]
Since $g_\ell\to g$ uniformly, we have $\rho_\ell\to \rho$ uniformly on $M$.

Define
\[
K_\ell(x,y):=k\!\left(\frac{\left\lVert\iota_\ell(x)-\iota_\ell(y)\right\rVert^{2}}{t}\right),\qquad K(x,y):=k\!\left(\frac{\left\lVert\iota(x)-\iota(y)\right\rVert^{2}}{t}\right).
\]
Since $\iota_\ell\to \iota$ uniformly, the squared chord distances converge uniformly and stay in a bounded interval; uniform continuity of $k$ there gives $K_\ell\to K$ uniformly on $M\times M$.

Since $p_\ell\rho_\ell\to p\rho$ uniformly, the same direct estimate as in the intrinsic case gives
\[
\|L_{t,p_\ell}^{\mathrm{ext}}(g_\ell;\iota_\ell)-L_{t,p}^{\mathrm{ext}}(g;\iota)\|_{C(M)\to C(M)}
\le 2t^{-d/2-1}\mu_h(M)\,\|K_\ell p_\ell\rho_\ell-Kp\rho\|_{L^\infty(M\times M)}\to0.
\]

\paragraph{Step 2: injectivity of $\Phi_t^{\mathrm{ext}}$.}
Assume
\[
L_{t,p_1}^{\mathrm{ext}}(g_{[\iota_1]};[\iota_1])=L_{t,p_2}^{\mathrm{ext}}(g_{[\iota_2]};[\iota_2])\qquad\text{as operators on }C(M).
\]
Choose representatives $\iota_1,\iota_2$ of the two classes. The proof of Theorem~\ref{thm:extrinsic_joint_identifies_metric_and_density} gives
\[
\left\lVert\iota_1(x)-\iota_1(y)\right\rVert=\left\lVert\iota_2(x)-\iota_2(y)\right\rVert\qquad\text{for all }x,y\in M,
\]
and
\[
p_1=p_2 \quad \mu_g\text{-almost everywhere},
\]
with respect to the common induced volume measure $\mu_g$.

Now the map
\[
F:\iota_1(M)\to \iota_2(M),\qquad F(\iota_1(x)):=\iota_2(x),
\]
is a distance-preserving bijection between subsets of $\mathbb{R}^{D}$. To see that it extends to a rigid motion, choose $u_0\in\iota_1(M)$ and put $V=\operatorname{span}\{u-u_0:u\in\iota_1(M)\}$. The polarization identity for squared Euclidean distances shows that $u-u_0\mapsto F(u)-F(u_0)$ preserves all pairwise inner products. It therefore extends linearly to an isometry from $V$ onto $\operatorname{span}\{F(u)-F(u_0):u\in\iota_1(M)\}$. Extend this isometry by orthonormal bases to $A\in O(D)$, and set $b=F(u_0)-Au_0$. Then
\[
F(u)=Au+b\qquad\text{for all }u\in \iota_1(M).
\]
Equivalently, $\iota_2 = A\circ \iota_1 + b$, so that $[\iota_1]=[\iota_2]$. Since $p_1,p_2\in C(M)$ and $\mu_g$ has full support, almost everywhere equality implies pointwise equality. Hence $p_1=p_2$ on $M$. Therefore $\Phi_t^{\mathrm{ext}}$ is injective.
