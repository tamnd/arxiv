<!-- head-1: body_md -->
We denote by $\varepsilon>0$ the required accuracy in function value: the method must output a point $\hat x$ with $f(\hat x)-f^\ast\le\varepsilon$. The symbol $\|\cdot\|$ always stands for the Euclidean norm of a vector and the spectral norm of a matrix, $[x,y]$ is the segment with endpoints $x,y\in\mathbb R^d$, $B(x,r)$ is the closed ball of radius $r$ centered at $x$, and C--S stands for the Cauchy--Schwarz inequality. The sum of the first $k$ terms of a sequence is denoted by the corresponding capital letter: for example, $A_k=a_1+\dots+a_k$ and $A_0=0$.
<!-- end head-1 -->
<!-- head-2: body_md -->
We consider arbitrary deterministic first-order methods: a method may query the gradient at any points and output any point, and every oracle call is counted. We prove that the smallest number of calls $N$ for which there exists a deterministic first-order method solving the problem for all functions of the class in all dimensions satisfies\footnote{Here and below, the lower bounds are proved for $L_0R^2\ge\varepsilon$.}
\begin{align*}
 1+N&\ge\frac1{29}\Bigl[1+\sqrt{\frac{L_0R^2}{\varepsilon}}+\frac{L_1R}{\sqrt{\log(2+L_1R)}}\Bigr],\\
 1+N&\le5\sqrt{\frac{L_0R^2}{\varepsilon}}+C\,(1+L_1R)\log(2+L_1R)
\end{align*}
with an absolute constant $C$ (Theorems~\ref{thm:upper} and~\ref{thm:lower-radius}). Thus, the optimal complexity is $\sqrt{L_0R^2/\varepsilon}+L_1R$ up to logarithmic factors, and the logarithmic factors affect only the $L_1R$ term. The method achieving the upper bound uses only gradients. The lower bound remains valid for methods that receive the function value together with the gradient.
<!-- end head-2 -->
<!-- definition-1: body_md -->
\label{def:class}
Let $\ell:[0,\infty)\to(0,\infty)$ be a continuous nondecreasing function. We denote by $\mathcal F_\ell^d$ the class of convex differentiable functions $f:\mathbb R^d\to\mathbb R$ such that for all $x,y\in\mathbb R^d$
\begin{equation}\label{eq:class}
 \|\nabla f(x)-\nabla f(y)\|
 \le\ell\Bigl(\sup_{u\in[x,y]}\|\nabla f(u)\|\Bigr)\|x-y\| .
\end{equation}
For $\ell(s)=L_0+L_1s$ we write $\mathcal F_{L_0,L_1}^d$ and call the functions of this class \emph{$(L_0,L_1)$-smooth}.
<!-- end definition-1 -->
<!-- lemma-1: body_md -->
\label{lem:class}
Let $f:\mathbb R^d\to\mathbb R$ be convex and differentiable.
\begin{enumerate}
\item[(a)] $f\in\mathcal F_\ell^d$ if and only if for all $x,v\in\mathbb R^d$ the map $t\mapsto\nabla f(x+tv)$ is locally absolutely continuous on $\mathbb R$ and for almost all $t$
\begin{equation}\label{eq:line-derivative}
 \Bigl\|\frac{d}{dt}\nabla f(x+tv)\Bigr\|\le\ell\bigl(\|\nabla f(x+tv)\|\bigr)\|v\| .
\end{equation}
\item[(b)] If $f\in C^2(\mathbb R^d)$, then $f\in\mathcal F_\ell^d$ if and only if for all $x$
\begin{equation}\label{eq:hessian-bound}
 0\preceq\nabla^2f(x)\preceq\ell\bigl(\|\nabla f(x)\|\bigr)I .
\end{equation}
\end{enumerate}
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{lem:safe-step}
For every point $y\in\mathbb R^d$, the point
\[
 y^+=y-\frac{\nabla f(y)}{4\bigl(L_0+L_1\|\nabla f(y)\|\bigr)}
\]
satisfies
\begin{gather*}
 f(y^+)\le f(y)-\frac{5\|\nabla f(y)\|^2}{24\bigl(L_0+L_1\|\nabla f(y)\|\bigr)},\\
 \|y^+-x^\ast\|^2\le\|y-x^\ast\|^2-\frac{\|\nabla f(y)\|^2}{24\bigl(L_0+L_1\|\nabla f(y)\|\bigr)^2} .
\end{gather*}

<!-- lemma-3: body_md -->
\label{lem:anchored}
For every point $y\in\mathbb R^d$
\[
 f(y)+\langle \nabla f(y),x^\ast-y\rangle+\Phi\bigl(\|\nabla f(y)\|\bigr)\le f^\ast .

\label{lem:search}
Let $r,\delta>0$, let the points $v$ and $x$ lie in the ball $B(x^\ast,r)$, and let a positive integer $m$ satisfy

 2^m\ge128\Bigl(1+\frac{L_0r^2}\delta+L_1r\Bigr)^2 .
\end{equation}

\item[(b)] The inequalities
\begin{equation}\label{eq:search}
\begin{aligned}
 f(y)&\le f(x),\\
 \langle \nabla f(y),y-v\rangle&\le\frac{\|\nabla f(y)\|^2}{16\bigl(L_0+L_1\|\nabla f(y)\|\bigr)}
\end{aligned}

<!-- lemma-5: body_md -->
\label{lem:energy}
Let $\|x^0-x^\ast\|\le R$, and suppose that Algorithm~\ref{alg:am} completes iterations $1,\dots,k$ without meeting the stopping criterion. Then
\begin{equation}\label{eq:energy}
 A_k\bigl(f(x^k)-f^\ast\bigr)+\frac12\sum_{i=1}^ka_i^2\|\nabla f(y^i)\|^2+\frac12\|v^k-x^\ast\|^2\le\frac{R^2}2 ,
\end{equation}
and all points $x^i$, $v^i$, $y^i$, $i\le k$, lie in the ball $B(x^\ast,R)$.
<!-- end lemma-5 -->
<!-- proof-1: body_md -->
Induction on $k$. For $k=0$, inequality~\eqref{eq:energy} coincides with the condition $\|x^0-x^\ast\|\le R$. Suppose the statement holds for $k-1$.
\begin{itemize}
\item \emph{The iteration is well defined.} By the induction hypothesis, the points $v^{k-1}$ and $x^{k-1}$ lie in the ball $B(x^\ast,R)$. The ball is convex, so the segment $[v^{k-1},x^{k-1}]$ and the point $y^k$ lie in it, and Lemma~\ref{lem:search} applies. The stopping criterion was not triggered, so inequalities~\eqref{eq:search} hold. Together with Lemma~\ref{lem:safe-step} they give $x^k\in B(x^\ast,R)$ and
\begin{equation}\label{eq:energy-step}
\begin{aligned}
 &f(y^k)\le f(x^{k-1}),\\
 &\langle \nabla f(y^k),y^k-v^{k-1}\rangle\le\frac{\|\nabla f(y^k)\|^2}{16\bigl(L_0+L_1\|\nabla f(y^k)\|\bigr)},\\
 &f(x^k)\le f(y^k)-\frac{5\|\nabla f(y^k)\|^2}{24\bigl(L_0+L_1\|\nabla f(y^k)\|\bigr)} .
\end{aligned}
\end{equation}
\item \emph{Distance.} Since $v^k=v^{k-1}-a_k\nabla f(y^k)$,
\begin{equation}\label{eq:energy-dist}
\begin{aligned}
 \MoveEqLeft\frac12\|v^k-x^\ast\|^2-\frac12\|v^{k-1}-x^\ast\|^2\\
 &=a_k\langle \nabla f(y^k),y^k-v^{k-1}\rangle-a_k\langle \nabla f(y^k),y^k-x^\ast\rangle+\frac12a_k^2\|\nabla f(y^k)\|^2 .
\end{aligned}
\end{equation}
We bound the first two terms on the right:
\begin{align*}
 a_k\langle \nabla f(y^k),y^k-v^{k-1}\rangle
 &\mathrel{\underset{\mathmakebox[\widthof{$\scriptstyle\text{Lemma~\ref{lem:anchored}}$}]{\eqref{eq:energy-step}}}{\le}}\frac{a_k\|\nabla f(y^k)\|^2}{16\bigl(L_0+L_1\|\nabla f(y^k)\|\bigr)},\\
 -a_k\langle \nabla f(y^k),y^k-x^\ast\rangle
 &\mathrel{\underset{\text{Lemma~\ref{lem:anchored}}}{\le}}-a_k\bigl(f(y^k)-f^\ast\bigr)-a_k\Phi\bigl(\|\nabla f(y^k)\|\bigr).
\end{align*}
\item \emph{Optimality gap.} Since $A_k=A_{k-1}+a_k$,
\begin{align*}
 \MoveEqLeft A_k\bigl(f(x^k)-f^\ast\bigr)-A_{k-1}\bigl(f(x^{k-1})-f^\ast\bigr)\\
 &\mathrel{\underset{\eqref{eq:energy-step}}{\le}}A_k\bigl(f(y^k)-f^\ast\bigr)-\frac{5A_k\|\nabla f(y^k)\|^2}{24\bigl(L_0+L_1\|\nabla f(y^k)\|\bigr)}-A_{k-1}\bigl(f(y^k)-f^\ast\bigr)\\
 &\mathrel{\underset{\mathmakebox[\widthof{$\scriptstyle\eqref{eq:energy-step}$}]{}}{=}}a_k\bigl(f(y^k)-f^\ast\bigr)-\frac{5A_k\|\nabla f(y^k)\|^2}{24\bigl(L_0+L_1\|\nabla f(y^k)\|\bigr)} .
\end{align*}
\item \emph{Sum.} Multiplying equation~\eqref{eq:coef} by $\|\nabla f(y^k)\|^2$, we obtain
\[
 a_k\Phi\bigl(\|\nabla f(y^k)\|\bigr)=a_k^2\|\nabla f(y^k)\|^2-\frac{A_k\|\nabla f(y^k)\|^2}{7\bigl(L_0+L_1\|\nabla f(y^k)\|\bigr)} .
\]
We add~\eqref{eq:energy-dist}, the two bounds on its terms and the bound on the optimality gap, and then substitute this equality:
\begin{align*}
 \MoveEqLeft\Bigl[A_k\bigl(f(x^k)-f^\ast\bigr)+\frac12\|v^k-x^\ast\|^2\Bigr]-\Bigl[A_{k-1}\bigl(f(x^{k-1})-f^\ast\bigr)+\frac12\|v^{k-1}-x^\ast\|^2\Bigr]\\
 &\mathrel{\underset{\mathmakebox[\widthof{$\scriptstyle\eqref{eq:coef-basic}$}]{}}{\le}}\Bigl(\frac{a_k}{16}-\frac{5A_k}{24}\Bigr)\frac{\|\nabla f(y^k)\|^2}{L_0+L_1\|\nabla f(y^k)\|}-a_k\Phi\bigl(\|\nabla f(y^k)\|\bigr)+\frac12a_k^2\|\nabla f(y^k)\|^2\\
 &\mathrel{\underset{\mathmakebox[\widthof{$\scriptstyle\eqref{eq:coef-basic}$}]{}}{=}}\Bigl(\frac{a_k}{16}-\frac{5A_k}{24}+\frac{A_k}7\Bigr)\frac{\|\nabla f(y^k)\|^2}{L_0+L_1\|\nabla f(y^k)\|}-\frac12a_k^2\|\nabla f(y^k)\|^2\\
 &\mathrel{\underset{\eqref{eq:coef-basic}}{\le}}-\frac{A_k\|\nabla f(y^k)\|^2}{336\bigl(L_0+L_1\|\nabla f(y^k)\|\bigr)}-\frac12a_k^2\|\nabla f(y^k)\|^2\\
 &\mathrel{\underset{\mathmakebox[\widthof{$\scriptstyle\eqref{eq:coef-basic}$}]{}}{\le}}-\frac12a_k^2\|\nabla f(y^k)\|^2 .
\end{align*}
Adding this inequality to inequality~\eqref{eq:energy} for $k-1$, we obtain~\eqref{eq:energy} for $k$. It implies $\|v^k-x^\ast\|\le R$, which completes the induction.\qedhere
\end{itemize}

\begin{itemize}
\item \emph{Two sums.} By Lemma~\ref{lem:energy} and the condition on the optimality gap
\begin{equation}\label{eq:count-anchor-budget}
\begin{aligned}
 &L_0A_K<64L_1^2R^2,\\
 &\sum_{i=1}^Ka_i^2\|\nabla f(y^i)\|^2\le R^2 .
\end{aligned}
\end{equation}
\item \emph{Bounds for one iteration.} From inequalities~\eqref{eq:coef-basic}
\begin{equation}\label{eq:count-step}
\begin{gathered}
\begin{aligned}
 \sqrt{L_0A_i}-\sqrt{L_0A_{i-1}}=\frac{\sqrt{L_0}\,a_i}{\sqrt{A_i}+\sqrt{A_{i-1}}}
 &\mathrel{\underset{A_{i-1}\le A_i}{\ge}}\frac{\sqrt{L_0}\,a_i}{2\sqrt{A_i}}\\
 &\mathrel{\underset{\mathmakebox[\widthof{$\scriptstyleA_{i-1}\le A_i$}]{\eqref{eq:coef-basic}}}{\ge}}\frac1{2\sqrt7}\sqrt{\frac{L_0}{L_0+L_1\|\nabla f(y^i)\|}},
\end{aligned}\\
\begin{aligned}
 L_0A_i=\frac{L_0A_i^2}{A_i}
 &\mathrel{\underset{\eqref{eq:coef-basic}}{\ge}}\frac{L_0a_i^2}{A_i}\\
 &\mathrel{\underset{\eqref{eq:coef-basic}}{\ge}}\frac{L_0}{7\bigl(L_0+L_1\|\nabla f(y^i)\|\bigr)} .
\end{aligned}
\end{gathered}
\end{equation}

\begin{equation}\label{eq:anchor-identity}
 L_1a_i\|\nabla f(y^i)\|
 =\frac{L_1\Phi(\|\nabla f(y^i)\|)}{\|\nabla f(y^i)\|}
 +\frac{A_i}{7a_i}\cdot\frac{L_1\|\nabla f(y^i)\|}{L_0+L_1\|\nabla f(y^i)\|} .
\end{equation}
Both terms on the right are nonnegative. By definition~\eqref{eq:Phi}
\begin{equation}\label{eq:Phi-ratio}
\begin{aligned}
 \frac{L_1\Phi(\|\nabla f(y^i)\|)}{\|\nabla f(y^i)\|}
 &=\Bigl(1+\frac{L_0}{L_1\|\nabla f(y^i)\|}\Bigr)\log\Bigl(1+\frac{L_1\|\nabla f(y^i)\|}{L_0}\Bigr)-1\\
 &\ge\log\Bigl(1+\frac{L_1\|\nabla f(y^i)\|}{L_0}\Bigr)-1 .
\end{aligned}
\end{equation}

\begin{equation}\label{eq:count-two-groups}
\begin{aligned}
 \MoveEqLeft2\sqrt7\Bigl(\sqrt{L_0A_i}-\sqrt{L_0A_{i-1}}\Bigr)+\Bigl(\frac{7L_1a_i^2\|\nabla f(y^i)\|}{A_i}\Bigr)^{2/3}\\

\end{aligned}
\end{equation}
Here step (a) uses the inequalities $t^{1/2}\ge t$ and $t^{2/3}\ge t$ for $0\le t\le1$.
\item \emph{Iterations with a large product $L_1a_i\|\nabla f(y^i)\|$.} Let $J$ be the set of indices $i\le K$ with $L_1a_i\|\nabla f(y^i)\|\ge\sqrt{L_1R}$. By the second inequality of~\eqref{eq:count-anchor-budget}
\[
 |J|\,L_1R\le\sum_{i\in J}L_1^2a_i^2\|\nabla f(y^i)\|^2\le L_1^2R^2 ,
\]
that is, $|J|\le L_1R$.

\[
 \log\Bigl(1+\frac{L_1\|\nabla f(y^i)\|}{L_0}\Bigr)-1\le L_1a_i\|\nabla f(y^i)\|<\sqrt{L_1R} ,
\]
and the second inequality of~\eqref{eq:count-step} gives
\[
 L_0A_i\ge\frac17\Bigl(1+\frac{L_1\|\nabla f(y^i)\|}{L_0}\Bigr)^{-1}>\frac17e^{-\sqrt{L_1R}-1} .
\]
Let $j$ be the smallest index in $I$. The weights $A_i$ increase, so for $i>j$ we have $A_{i-1}>0$ and
\[
 \frac{a_i}{A_i}\le-\log\Bigl(1-\frac{a_i}{A_i}\Bigr)=\log\frac{A_i}{A_{i-1}} .
\]
Together with the inequality $a_j\le A_j$ this gives
\begin{align*}
 \sum_{i\in I}\frac{a_i}{A_i}
 &\le1+\log\frac{A_K}{A_j}
 \mathrel{\underset{\eqref{eq:count-anchor-budget}}{<}}1+\log\bigl(448L_1^2R^2e^{\sqrt{L_1R}+1}\bigr)\\
 &\mathrel{\underset{\mathmakebox[\widthof{$\scriptstyle\eqref{eq:count-anchor-budget}$}]{}}{=}}\sqrt{L_1R}+2+\log448+2\log(L_1R)\\
 &\mathrel{\underset{\mathmakebox[\widthof{$\scriptstyle\eqref{eq:count-anchor-budget}$}]{(\mathrm b)}}{<}}5\sqrt{L_1R}+\frac92
 \mathrel{\underset{L_1R\ge1}{\le}}\frac{19}2\sqrt{L_1R} .
\end{align*}
Here step (b) uses the inequalities $\log448<13/2$ and $\log t\le2\sqrt t-2$ for $t\ge1$.
Sum inequalities~\eqref{eq:count-two-groups} over $i\in I$. The sum of the first terms is at most $2\sqrt7\sqrt{L_0A_K}<16\sqrt7L_1R$. To the second terms we apply Hölder's inequality with exponents $3$ and $3/2$:
\begin{align*}
 |I|
 &<16\sqrt7L_1R+7^{2/3}\sum_{i\in I}\bigl(L_1a_i\|\nabla f(y^i)\|\bigr)^{2/3}\Bigl(\frac{a_i}{A_i}\Bigr)^{2/3}\\

\end{align*}
Here step (c) uses the inequalities $16\sqrt7<42.4$ and $(133/2)^{2/3}<16.5$, which are equivalent to $1792<42.4^2$ and $133^2<4\cdot16.5^3$.

\[
 K=|J|+|I|<L_1R+59L_1R=60L_1R .\qedhere
\]
\end{itemize}

 N=\Bigl\lceil\sqrt{\frac{18L_0R^2}\varepsilon}\,\Bigr\rceil

\item The author of \citet{tyurin2026accelerated} states the result for twice differentiable functions, but notes (ibid., after Lemma~2.2) that instead of the Hessian bound the inequality

suffices. For our class this is Lemma~\ref{lem:gronwall}(b).

<!-- lemma-7: body_md -->
\label{lem:conj}
Let $K\subset\mathbb R^n$ be a nonempty convex compact set, let the function $H$ be continuous on $K$, and suppose that for all $z,z'\in K$ its restriction to the segment $[z,z']$ is strongly convex with modulus
\[
 \frac1{L_0+L_1\max\{\|z\|,\|z'\|\}} .
\]
Then for every $\sigma\in\mathbb R^n$ the function
\[
 \varphi(y)=\max_{z\in K}\bigl\{\langle y+\sigma,z\rangle-H(z)\bigr\}
\]
is convex and differentiable, its gradient $\nabla\varphi(y)$ is the unique maximizer $z(y)$, and for all $y,y'$
\begin{equation}\label{eq:conj-lip}
 \|\nabla\varphi(y)-\nabla\varphi(y')\|\le\Bigl(L_0+L_1\max\bigl\{\|\nabla\varphi(y)\|,\|\nabla\varphi(y')\|\bigr\}\Bigr)\|y-y'\| .
\end{equation}
In particular, $\varphi\in\mathcal F_{L_0,L_1}^n$.
<!-- end lemma-7 -->
<!-- proof-5: body_md -->
\leavevmode
\begin{itemize}
\item \emph{Gradient.} Denote $\Psi_y(z)=\langle y+\sigma,z\rangle-H(z)$. The function $H$ is strictly convex on every segment, so $\Psi_y$ has a unique maximizer $z(y)$ on $K$. By Danskin's theorem, $\varphi$ is convex, differentiable and $\nabla\varphi(y)=z(y)$.
\item \emph{Growth around the maximizer.} Let $z=z(y)$, $z'\in K$, $z_t=z+t(z'-z)$, and let $m$ be the modulus from the assumption for the segment $[z,z']$. By assumption, the function $\Psi_y(z_t)$ remains concave in $t$ after adding the quadratic term with modulus $m$, therefore
\begin{align*}
 \Psi_y(z)
 &\mathrel{\underset{z=z(y)}{\ge}}\Psi_y(z_t)\\
 &\mathrel{\underset{\mathmakebox[\widthof{$\scriptstylez=z(y)$}]{(\mathrm a)}}{\ge}}(1-t)\Psi_y(z)+t\Psi_y(z')+\frac m2t(1-t)\|z'-z\|^2 ,
\end{align*}
where step (a) uses the strong concavity of $\Psi_y$ on the segment $[z,z']$ with modulus $m$.
Subtracting $(1-t)\Psi_y(z)$, dividing by $t$ and letting $t$ tend to zero, we obtain
\begin{equation}\label{eq:conj-growth}
 \Psi_y(z(y))-\Psi_y(z')\ge\frac m2\|z'-z(y)\|^2 .
\end{equation}
\item \emph{Lipschitz continuity of the gradient.} Let $z=z(y)$, $z'=z(y')$, and let $m$ be the modulus for the segment $[z,z']$. Add~\eqref{eq:conj-growth} for the pair $(y,z')$ to the same inequality for the pair $(y',z)$. The values of $H$ cancel, and
\begin{align*}
 m\|z-z'\|^2&\mathrel{\underset{\mathmakebox[\widthof{$\scriptstyle\text{C--S}$}]{}}{\le}}\langle y-y',z-z'\rangle\\
 &\mathrel{\underset{\text{C--S}}{\le}}\|y-y'\|\,\|z-z'\| .
\end{align*}
If $z=z'$, inequality~\eqref{eq:conj-lip} is obvious, otherwise dividing by $m\|z-z'\|$ gives~\eqref{eq:conj-lip}. The maximum of the gradient norm over the endpoints of a segment does not exceed its supremum over the segment, so $\varphi\in\mathcal F_{L_0,L_1}^n$.\qedhere
\end{itemize}

\item[(a)] Let $\varphi:\mathbb R^n\to\mathbb R$ be a convex chain, $d\ge2n$, and let every $d\times n$ matrix $V$ with orthonormal columns $v_1,\dots,v_n$ be assigned the function

<!-- lemma-8: body_md -->
\label{lem:ord-class}
The function $H$ satisfies the assumption of Lemma~\ref{lem:conj}. Consequently, $\varphi\in\mathcal F_{L_0,L_1}^n$ and its gradient satisfies inequality~\eqref{eq:conj-lip}.
<!-- end lemma-8 -->
<!-- proof-6: body_md -->
\leavevmode
\begin{itemize}
\item \emph{Continuity.} The function $r$ is convex and $r(0)=0$, so on $[0,1]$
\[
 0\le r(p)\le p\,r'(p)\le p\,r'(1) .
\]
Hence, the second term of $H$ is at most $Z_nr'(1)$ and tends to zero together with $Z_n$. At the other points of $K$ continuity is obvious.
\item \emph{Second derivative along a segment.} Let $z,z'\in K$ be distinct, $u=z'-z$ and $w=z+tu$, $t\in[0,1]$. Denote $W_n=w_1+\dots+w_n$ and $U_n=u_1+\dots+u_n$, so that $W_n$ depends on $t$ and its derivative in $t$ equals $U_n$. Set
\[
 p=\frac w{W_n} .
\]
Coordinates that vanish at both points vanish on the whole segment, and the others are positive for $0<t<1$. Therefore, on $(0,1)$ the function $t\mapsto H(w)$ is twice continuously differentiable. The second term of $H$ is the perspective of the function $r(p_1)+\dots+r(p_n)$. Along the segment
\begin{align*}
 \frac{dp}{dt}&=\frac{u-pU_n}{W_n},\\
 \Phi''(W_n)&\mathrel{\underset{\eqref{eq:Phi-derivatives}}{=}}\frac1{L_0+L_1W_n},
\end{align*}
and differentiation gives
\begin{equation}\label{eq:H-second}
 \frac{d^2}{dt^2}H(w)=\frac{2U_n^2}{L_0+L_1W_n}+\sum_{i=1}^n\frac{r''(p_i)}{W_n}\,(u_i-p_iU_n)^2 .
\end{equation}
The vector $p$ lies in the simplex, so each of its coordinates is at most its norm and
\begin{equation}\label{eq:simplex-p}
 \frac1{\sqrt n}\le\|p\|\le1 .
\end{equation}
Moreover, $\|w\|=W_n\|p\|$. We bound the coefficients in~\eqref{eq:H-second}:
\begin{align*}
 \frac{r''(p_i)}{W_n}
 &\mathrel{\underset{\eqref{eq:r-derivatives}}{=}}\frac{4\sqrt n}{L_1W_n\,(1+\sqrt n\,p_i)}
 \mathrel{\underset{\eqref{eq:simplex-p}}{\ge}}\frac2{L_1W_n\|p\|}\\
 &\mathrel{\underset{\mathmakebox[\widthof{$\scriptstyle\eqref{eq:r-derivatives}$}]{}}{=}}\frac2{L_1\|w\|}
 \ge\frac2{L_0+L_1\|w\|},\\
 \frac2{L_0+L_1W_n}
 &\mathrel{\underset{\|p\|\le1}{\ge}}\frac{2\|p\|^2}{L_0+L_1W_n\|p\|}
 =\frac{2\|p\|^2}{L_0+L_1\|w\|} .
\end{align*}
Substituting them into~\eqref{eq:H-second}, we obtain
\begin{align*}
 \frac{d^2}{dt^2}H(w)
 &\ge\frac2{L_0+L_1\|w\|}\Bigl(\|p\|^2U_n^2+\|u-pU_n\|^2\Bigr)\\
 &\mathrel{\underset{(\mathrm a)}{\ge}}\frac{\|u\|^2}{L_0+L_1\|w\|}
 \mathrel{\underset{(\mathrm b)}{\ge}}\frac{\|u\|^2}{L_0+L_1\max\{\|z\|,\|z'\|\}} .
\end{align*}
Here step (a) uses the inequality $\|u\|^2\le2\|pU_n\|^2+2\|u-pU_n\|^2$, and step (b) uses the convexity of the norm.
\item \emph{Conclusion.} The restriction of $H$ to the segment minus the quadratic term from Lemma~\ref{lem:conj} has a nonnegative second derivative on $(0,1)$ and is continuous on $[0,1]$. Hence, it is convex, and Lemma~\ref{lem:conj} applies.\qedhere
\end{itemize}

\label{lem:ord-local}
The function $\varphi$ is a chain: if $y\in\mathcal K_{j-1}$, then $\nabla\varphi(y)\in\mathcal K_j$.

\begin{itemize}
\item \emph{Optimality conditions.} In a neighborhood of $z$ the function $H$ is differentiable and $\partial_iH(z)=\nu+r'(p_i)$, where
\[
 \nu=2\Phi'(Z_n)+\sum_{k=1}^n\bigl[r(p_k)-p_kr'(p_k)\bigr].
\]

\begin{equation}\label{eq:ord-kkt}
 y_i+\sigma_i-\frac\lambda{G_i}\ \begin{cases}=\nu+r'(p_i), & z_i>0,\\ \le\nu, & z_i=0.\end{cases}
\end{equation}

\begin{equation}\label{eq:ord-nu-upper}
 \nu\le2\Phi'(G_1)\mathrel{\underset{\eqref{eq:Phi-derivatives}}{=}}\frac2{L_1}\log\Bigl(1+\frac{L_1G_1}{L_0}\Bigr).
\end{equation}
\item \emph{Lower bound on $\nu$.} Since $0\le r'(p_j)\le r'(1)$ and $y_j=0$, in both cases~\eqref{eq:ord-kkt} gives
\begin{equation}\label{eq:ord-nu-lower}
 -\lambda\le G_j\nu-G_j\sigma_j+G_jr'(1) .
\end{equation}
\item \emph{Coordinate $l$ is zero.} Since $y_l=0$ and $G_j\ge G_l$,

 &\mathrel{\underset{\mathmakebox[\widthof{$\scriptstyle\eqref{eq:ord-nu-upper}$}]{\eqref{eq:ord-nu-lower}}}{\le}}G_l\sigma_l-G_j\sigma_j+G_jr'(1)+(G_j-G_l)\nu\\

\end{align*}

\end{itemize}

<!-- lemma-13: body_md -->
\label{lem:gronwall}
For all points $y,z\in\mathbb R^d$
\begin{align*}
 &(\text{a})&L_0+L_1\|\nabla f(z)\|&\le\bigl(L_0+L_1\|\nabla f(y)\|\bigr)e^{L_1\|z-y\|},\\
 &(\text{b})&\|\nabla f(z)-\nabla f(y)\|&\le\bigl(L_0+L_1\|\nabla f(y)\|\bigr)\frac{e^{L_1\|z-y\|}-1}{L_1},\\
 &(\text{c})&0\le f(z)-f(y)-\langle \nabla f(y),z-y\rangle&\le\bigl(L_0+L_1\|\nabla f(y)\|\bigr)\frac{e^{L_1\|z-y\|}-1-L_1\|z-y\|}{L_1^2} .
\end{align*}
<!-- end lemma-13 -->

We track the gradient at the points $y+t(z-y)$, $t\in[0,1]$.
\begin{itemize}
\item \emph{Integral inequality.} By Lemma~\ref{lem:class}(a), the map $t\mapsto\nabla f(y+t(z-y))$ is absolutely continuous, and for almost all $t$ the norm of its derivative is at most $(L_0+L_1\|\nabla f(y+t(z-y))\|)\|z-y\|$. Therefore, for all $t\in[0,1]$
\begin{equation}\label{eq:gronwall-int}
 \|\nabla f\bigl(y+t(z-y)\bigr)-\nabla f(y)\|\le\|z-y\|\int_0^t\Bigl(L_0+L_1\|\nabla f\bigl(y+\tau(z-y)\bigr)\|\Bigr)d\tau .
\end{equation}
By the triangle inequality this implies
\begin{align*}
 \MoveEqLeft L_0+L_1\|\nabla f\bigl(y+t(z-y)\bigr)\|\\
 &\le L_0+L_1\|\nabla f(y)\|+L_1\|z-y\|\int_0^t\Bigl(L_0+L_1\|\nabla f\bigl(y+\tau(z-y)\bigr)\|\Bigr)d\tau .
\end{align*}
\item \emph{Grönwall's inequality.} The left-hand side of the last inequality is continuous in $t$, and by Grönwall's inequality in integral form
\[
 L_0+L_1\|\nabla f\bigl(y+t(z-y)\bigr)\|\le\bigl(L_0+L_1\|\nabla f(y)\|\bigr)e^{L_1\|z-y\|t} .
\]
For $t=1$ this is statement~(a).

\begin{align*}
 \|\nabla f\bigl(y+t(z-y)\bigr)-\nabla f(y)\|
 &\le\bigl(L_0+L_1\|\nabla f(y)\|\bigr)\|z-y\|\int_0^te^{L_1\|z-y\|\tau}d\tau\\
 &=\bigl(L_0+L_1\|\nabla f(y)\|\bigr)\frac{e^{L_1\|z-y\|t}-1}{L_1} .
\end{align*}
For $t=1$ this is statement~(b).

\end{align*}
The second step uses the Cauchy--Schwarz inequality and statement~(b).\qedhere

Denote by $z(t)=v+t(x-v)$ the points of the segment $[v,x]$. The function $t\mapsto f(z(t))$ is convex on $[0,1]$, and its derivative $\langle \nabla f(z(t)),x-v\rangle$ is nondecreasing. For the point $y=z(t)$
\[
 \langle \nabla f(y),y-v\rangle=t\,\langle \nabla f(z(t)),x-v\rangle .
\]
\begin{itemize}
\item \emph{Stopping criterion.} If it was triggered at a point $z\in[v,x]$, then $z$ lies in the ball $B(x^\ast,r)$, and by convexity
\[
 f(z)-f^\ast\le\langle \nabla f(z),z-x^\ast\rangle\le r\|\nabla f(z)\|\le\delta .
\]
From now on we assume that the stopping criterion was not triggered.
\item \emph{Return at the endpoints of the segment.} If $\langle \nabla f(v),x-v\rangle\ge0$, then the derivative is nonnegative on $[0,1]$. Hence, $f(v)\le f(x)$, and for $y=v$ the left-hand side of the second inequality of~\eqref{eq:search} is zero. If $\langle \nabla f(x),x-v\rangle\le0$, then for $y=x$ the first inequality of~\eqref{eq:search} is obvious, and the left-hand side of the second one is nonpositive.
\item \emph{Bisection.} The algorithm maintains the inequalities
\[
 \langle \nabla f(z(t_-)),x-v\rangle<0<\langle \nabla f(z(t_+)),x-v\rangle .

If the derivative at the midpoint is zero, then this midpoint minimizes the function on the segment, and both inequalities~\eqref{eq:search} hold for it. Otherwise, after $m$ steps $t_+-t_-=2^{-m}$.
\item \emph{Decrease for $y=z(t_+)$.} The derivative at the point $t_+$ is positive, so the function is nondecreasing on $[t_+,1]$, and $f(y)\le f(x)$.

 L_1\|z(t_+)-z(t_-)\|\le2^{1-m}L_1r\le\frac{2L_1r}{128\bigl(1+L_1r\bigr)^2}\le\frac1{32} .

\]
and the second inequality of~\eqref{eq:search} holds.

\end{itemize}

\label{lem:eq-chain}
The function $\psi$ is convex, continuously differentiable, and
\begin{enumerate}
\item[(a)] its minimum is zero, and the set of minimizers is
\[
 \bigl\{y\in\mathbb R^n:\ y_i\le-(\sigma_i+b)\ \text{for}\ i=1,\dots,n\bigr\}.
\]
The minimizer closest to zero, $-b\,(2n,2n-1,\dots,n+1)$, has norm $R$,
\item[(b)] $\psi(0)=(4-2/n)\varepsilon$,
\item[(c)] $\psi(y)\ge2\varepsilon$ whenever $y_n=0$,
\item[(d)] $\psi$ is a chain,
\item[(e)] $\nabla\psi(y)=z(y)=G\,(p_1,\dots,p_n)$, where $z(y)$ is the unique maximizer in~\eqref{eq:eq-chain}.
\end{enumerate}

<!-- proof-19: body_md -->
\leavevmode
\begin{itemize}
\item \emph{Gradient.} The strong convexity of $r$ gives the strong convexity of $H$ on $K$ with a constant modulus. Therefore, the maximizer is unique and, as in the proof of Lemma~\ref{lem:conj}, $\nabla\psi(y)=z(y)$ and the gradient is continuous. This is statement~(e).
\item \emph{Optimality conditions.} Set $\theta_0=0$ and $\theta_i=y_i+\sigma_i$ for $i\ge1$. The maximization problem in~\eqref{eq:eq-chain} is equivalent to maximizing
\[
 \sum_{i=0}^n\theta_ip_i-\sum_{i=0}^nr(p_i)
\]
over the simplex. Moving weight between two positive coordinates and from a positive coordinate to a zero one gives a number $\nu$ such that for $0\le i\le n$
\begin{equation}\label{eq:eq-kkt}
 \theta_i\ \begin{cases}=\nu+r'(p_i), & p_i>0,\\ \le\nu, & p_i=0.\end{cases}
\end{equation}
These conditions are also sufficient by the convexity of $r$. Since $0\le r'(p_i)\le b$, we always have $\nu\ge\theta_i-b$.
\item \emph{Statement~(a).} The point $z=0$ is feasible, so $\psi\ge0$. Equality holds exactly when the vector $p=e_0$ is optimal. \item \emph{Statement~(b).} For $y=0$, conditions~\eqref{eq:eq-kkt} hold for $p=e_1$ and $\nu=\sigma_1-b=(2n-2)b$: all other $\theta_i$, including $\theta_0=0$, are at most $\nu$. Therefore, $\psi(0)=G\sigma_1=(4-2/n)\varepsilon$.

A positive $p_l$ would contradict~\eqref{eq:eq-kkt}, since $r'(p_l)>0$. Hence, the $l$-th coordinate of the gradient $Gp_l$ is zero.\qedhere
\end{itemize}

\label{lem:eq-conj-class}
If $r$ is twice continuously differentiable on $(0,1)$ and for $0<p<1$
\begin{equation}\label{eq:eq-dual-curvature}
 r''(p)\ge\frac{G}{L_0+L_1Gp},
\end{equation}
then $\psi\in\mathcal F_{L_0,L_1}^n$.

<!-- proof-20: body_md -->
Let $z,z'\in K$ be distinct and $u=z'-z$. On the interval $0<t<1$ the coordinates $p_i$ of the point $z+tu$ are either positive or identically zero. Therefore,
\begin{align*}
 \frac{d^2}{dt^2}H(z+tu)
 &=\frac1G\sum_{i=1}^nr''(p_i)u_i^2+\frac1Gr''(p_0)U_n^2\\
 &\mathrel{\underset{\eqref{eq:eq-dual-curvature}}{\ge}}\frac{\|u\|^2}{L_0+L_1\|z+tu\|}
 \ge\frac{\|u\|^2}{L_0+L_1\max\{\|z\|,\|z'\|\}} ,
\end{align*}
where $U_n=u_1+\dots+u_n$. The terms with identically zero coordinates are omitted. If $p_0\equiv0$, then $U_n=0$ and the last term is absent as well. \label{thm:lower-gap}
Let a positive integer $n$ satisfy condition~\eqref{eq:eq-quad-cond} or condition~\eqref{eq:eq-log-cond}. \begin{equation}\label{eq:lower-gap}
 1+\mathcal N(\mathcal P_{4\varepsilon})\ge\frac1{12}\Bigl[1+\sqrt{\frac{L_0R^2}\varepsilon}+\Bigl(\frac{L_1R}{\log(2+L_1^2\varepsilon/L_0)}\Bigr)^{2/3}\Bigr].
\end{equation}

Let $\|x^0-x^\ast\|\le R_0$, let the initial optimality gap $f(x^0)-f^\ast$ be at most $\Delta_0$, and let

\end{equation}
and all points of the method and of the line search lie in the ball $B(x^\ast,r)$. Moreover, for all $i\le k$
\begin{equation}\label{eq:gap-monotone}
\begin{aligned}
 f(x^{i-1})&\ge f(y^i),\\
 f(y^i)&\ge f(x^i)+\frac{\|\nabla f(y^i)\|^2}{8\bigl(L_0+L_1\|\nabla f(y^i)\|\bigr)} .
\end{aligned}
\end{equation}

\item \emph{Replacements.} Instead of Lemma~\ref{lem:anchored}, the convexity inequality $\langle \nabla f(y^k),y^k-x^\ast\rangle\ge f(y^k)-f^\ast$ is used, and Lemma~\ref{lem:search} is used in the same way. Inequalities~\eqref{eq:gap-monotone} follow from it and Lemma~\ref{lem:safe-step}.

<!-- lemma-18: body_md -->
\label{lem:block}
Let $\|x^0-x^\ast\|\le R$, let Algorithm~\ref{alg:am-gap} be run with $a_0=0$ and $r=R$, and suppose that it completes iterations $k+1,\dots,k+N$, $N\ge1$, without meeting the stopping criterion. Then
\begin{equation}\label{eq:block}
 f(x^{k+N})-f^\ast\le\frac{96L_0R^2}{N^2}+\frac{3072\,L_1^2R^2}{N^3}\bigl(f(x^k)-f^\ast\bigr).
\end{equation}
<!-- end lemma-18 -->
<!-- proof-23: body_md -->
All sums in the proof are over $i=k+1,\dots,k+N$.
\begin{itemize}
\item \emph{Optimality gap through the weights.} We have
\begin{align*}
 f(x^{k+N})-f^\ast
 &\mathrel{\underset{\eqref{eq:energy-gap}}{\le}}\frac{R^2}{2A_{k+N}}
 \le\frac{R^2}{2\bigl(\sqrt{A_{k+N}}-\sqrt{A_k}\bigr)^2}\\
 &\mathrel{\underset{\eqref{eq:coef-gap}}{\le}}\frac{64R^2}{\Bigl(\sum_i\bigl(L_0+L_1\|\nabla f(y^i)\|\bigr)^{-1/2}\Bigr)^2} .
\end{align*}
\item \emph{Hölder's inequality.} With exponents $3/2$ and $3$
\begin{align*}
 N&=\sum_i\bigl(L_0+L_1\|\nabla f(y^i)\|\bigr)^{-1/3}\bigl(L_0+L_1\|\nabla f(y^i)\|\bigr)^{1/3}\\
 &\le\Bigl(\sum_i\bigl(L_0+L_1\|\nabla f(y^i)\|\bigr)^{-1/2}\Bigr)^{2/3}\Bigl(\sum_i\bigl(L_0+L_1\|\nabla f(y^i)\|\bigr)\Bigr)^{1/3} .
\end{align*}
Substituting into the previous bound, we obtain
\begin{equation}\label{eq:block-g}
\begin{aligned}
 f(x^{k+N})-f^\ast
 &\le\frac{64R^2}{N^3}\sum_i\bigl(L_0+L_1\|\nabla f(y^i)\|\bigr)\\
 &=\frac{64L_0R^2}{N^2}+\frac{64L_1R^2}{N^3}\sum_i\|\nabla f(y^i)\| .
\end{aligned}
\end{equation}
\item \emph{The sum of gradient norms is small when the optimality gap is small.} Summing inequalities~\eqref{eq:gap-monotone} over the block, we obtain
\[
 \sum_i\frac{\|\nabla f(y^i)\|^2}{L_0+L_1\|\nabla f(y^i)\|}\le8\bigl(f(x^k)-f^\ast\bigr).
\]
By the Cauchy--Schwarz inequality
\begin{align*}
 \Bigl(\sum_i\|\nabla f(y^i)\|\Bigr)^2
 &\le\sum_i\frac{\|\nabla f(y^i)\|^2}{L_0+L_1\|\nabla f(y^i)\|}\cdot\sum_i\bigl(L_0+L_1\|\nabla f(y^i)\|\bigr)\\
 &\le8\bigl(f(x^k)-f^\ast\bigr)\Bigl(NL_0+L_1\sum_i\|\nabla f(y^i)\|\Bigr).
\end{align*}
If the sum $\sum_i\|\nabla f(y^i)\|$ exceeded $\sqrt{8NL_0(f(x^k)-f^\ast)}+8L_1(f(x^k)-f^\ast)$, then its square would exceed this quantity times the sum itself, which is at least the right-hand side of the last inequality. Therefore,
\[
 \sum_i\|\nabla f(y^i)\|\le\sqrt{8NL_0\bigl(f(x^k)-f^\ast\bigr)}+8L_1\bigl(f(x^k)-f^\ast\bigr).
\]
\item \emph{Substitution.} Substitute this bound into~\eqref{eq:block-g}. The second term in~\eqref{eq:block-g} becomes at most
\[
 \frac{64L_1R^2}{N^3}\sqrt{8NL_0\bigl(f(x^k)-f^\ast\bigr)}+\frac{512\,L_1^2R^2\bigl(f(x^k)-f^\ast\bigr)}{N^3} .
\]
By the inequality between the arithmetic and geometric means, the first of these terms is at most
\[
 \frac{32L_0R^2}{N^2}+\frac{256\,L_1^2R^2\bigl(f(x^k)-f^\ast\bigr)}{N^3} .
\]
Therefore,
\[
 f(x^{k+N})-f^\ast\le\frac{96L_0R^2}{N^2}+\frac{768\,L_1^2R^2\bigl(f(x^k)-f^\ast\bigr)}{N^3} .
\]
Since $768\le3072$, we obtain~\eqref{eq:block}.\qedhere
\end{itemize}

<!-- lemma-19: body_md -->
\label{lem:count-weight}
If Algorithm~\ref{alg:am-gap} with parameters~\eqref{eq:weight-params} completes $K$ iterations without meeting the stopping criterion and $f(x^K)-f^\ast>\varepsilon$, then
\[
 K<16\sqrt{\frac{L_0R^2}\varepsilon}+21\,(L_1R)^{2/3}\log^{2/3}\frac{2\Delta}\varepsilon .
\]
<!-- end lemma-19 -->
<!-- proof-24: body_md -->
The right-hand side of~\eqref{eq:energy-gap} is at most $R^2/2+R^2/2=R^2$. Since $f(x^K)-f^\ast>\varepsilon$, this gives
\begin{equation}\label{eq:count-weight}
\begin{aligned}
 A_K&<\frac{R^2}\varepsilon,\\
 \sum_{i=1}^Ka_i^2\|\nabla f(y^i)\|^2&\le2R^2 .
\end{aligned}
\end{equation}
We split the iterations into two groups: with a small gradient, $L_1\|\nabla f(y^i)\|\le L_0$, and with a large one, $L_1\|\nabla f(y^i)\|>L_0$.
\begin{itemize}
\item \emph{Iterations with a small gradient.} For them $L_0+L_1\|\nabla f(y^i)\|\le2L_0$, and by~\eqref{eq:coef-gap} each such iteration increases $\sqrt{L_0A_i}$ by at least $1/16$. Hence, their number is at most
\[
 16\sqrt{L_0A_K}\mathrel{\underset{\eqref{eq:count-weight}}{<}}16\sqrt{\frac{L_0R^2}\varepsilon} .
\]
\item \emph{Iterations with a large gradient.} For them $L_0+L_1\|\nabla f(y^i)\|<2L_1\|\nabla f(y^i)\|$, therefore
\begin{align*}
 a_i\|\nabla f(y^i)\|&\mathrel{\underset{\eqref{eq:coef-gap}}{=}}\frac{A_i\|\nabla f(y^i)\|}{32\bigl(L_0+L_1\|\nabla f(y^i)\|\bigr)a_i}\\
 &\mathrel{\underset{\mathmakebox[\widthof{$\scriptstyle\eqref{eq:coef-gap}$}]{}}{>}}\frac{A_i}{64L_1a_i} .
\end{align*}
This and~\eqref{eq:count-weight} give
\[
 \sum_{L_1\|\nabla f(y^i)\|>L_0}\Bigl(\frac{A_i}{a_i}\Bigr)^2<(64L_1)^2\cdot2R^2=8192\,L_1^2R^2 .
\]
Since $A_{i-1}\ge a_0>0$, by~\eqref{eq:coef-gap-log}
\begin{align*}
 \sum_{L_1\|\nabla f(y^i)\|>L_0}\frac{a_i}{A_i}
 &\le\log\frac{A_K}{a_0}

<!-- proposition-3: body_md -->
\label{prop:fixed-dim}
Let $L_0,L_1,R,\varepsilon>0$ and let the dimension $d$ be fixed. The class $\mathcal P_{\rm R}$ in dimension $d$ is solved by the center-of-gravity method with
\begin{equation}\label{eq:centroid-count}
 1+\Bigl\lfloor\frac{d\log\bigl(2\sqrt d\,R/\min\{R,\rho_\varepsilon\}\bigr)}{-\log(1-1/e)}\Bigr\rfloor
\end{equation}
oracle calls, that is, with
\[
 O\Bigl(d\log\Bigl(2\sqrt d\max\Bigl\{1,\frac R{\rho_\varepsilon}\Bigr\}\Bigr)\Bigr)
\]
calls. In dimension $d=1$, bisection uses at most
\begin{equation}\label{eq:one-dim-count}
 \Bigl\lceil\max\Bigl\{\log_2\frac R{\rho_\varepsilon},\,0\Bigr\}\Bigr\rceil
\end{equation}
calls. If the output must coincide with a query point, one additional call suffices.
<!-- end proposition-3 -->
<!-- proof-26: body_md -->
\leavevmode
\begin{itemize}
\item \emph{Ball of points with small optimality gap.} By Lemma~\ref{lem:radial}, for every $x$
\begin{equation}\label{eq:fixed-envelope}
 f(x)-f^\ast\le\frac{L_0}{L_1^2}\bigl(e^{L_1\|x-x^\ast\|}-1-L_1\|x-x^\ast\|\bigr).
\end{equation}
Consequently, in the ball $B(x^\ast,\min\{R,\rho_\varepsilon\})$ the optimality gap is at most $\varepsilon$.
\item \emph{Center-of-gravity method.} Without loss of generality $x^0=0$. Start with the cube $K_0=[-2R,2R]^d$. It contains the ball $B(x^\ast,\min\{R,\rho_\varepsilon\})$, since $\|x^\ast\|\le R$. At step $k$, compute the center of gravity $c_k$ of the polytope $K_{k-1}$ and call the oracle at this point. If $\nabla f(c_k)=0$, then $c_k$ is a minimizer. Otherwise set
\[
 K_k=K_{k-1}\cap\bigl\{z:\ \langle \nabla f(c_k),z-c_k\rangle\le0\bigr\}.
\]
After the prescribed number of steps, return the query point with the smallest function value. Computing the centers of gravity between oracle calls is not counted.
\item \emph{The ball is preserved.} Suppose that at all of the first $N$ query points the optimality gap exceeds $\varepsilon$. For every point $z$ of the ball $B(x^\ast,\min\{R,\rho_\varepsilon\})$, convexity and~\eqref{eq:fixed-envelope} give
\[
 \langle \nabla f(c_k),z-c_k\rangle\le f(z)-f(c_k)<0 .
\]
Hence, the whole ball is preserved after each cut.
\item \emph{Comparison of volumes.} By Grünbaum's theorem \citep{grunbaum1960partitions}, every hyperplane through the center of gravity of a convex body leaves in each of the two half-spaces at least $1/e$ of its volume. The unit ball contains the cube $[-1/\sqrt d,1/\sqrt d]^d$, so its volume is at least $(2/\sqrt d)^d$. Consequently,
\begin{align*}
 \Bigl(\frac{2\min\{R,\rho_\varepsilon\}}{\sqrt d}\Bigr)^d
 &\mathrel{\underset{(\mathrm a)}{\le}}\operatorname{vol}(K_N)\\
 &\mathrel{\underset{(\mathrm b)}{\le}}\Bigl(1-\frac1e\Bigr)^N(4R)^d ,
\end{align*}
where step (a) uses the preservation of the ball and step (b) uses Grünbaum's theorem.
For $N$ equal to~\eqref{eq:centroid-count}, the right-hand side is strictly smaller than the left-hand side. This contradiction shows that at least one of the query points has optimality gap at most $\varepsilon$.
\item \emph{One-dimensional case.} Start with the segment $[-R,R]$. At each step, call the oracle at the midpoint of the current segment. If the derivative there is positive, keep the left half, if it is negative, keep the right half, and if it is zero, an exact minimizer has been found. The monotonicity of the derivative of a convex function guarantees that the minimizer stays in the segment. After $n$ calls, the midpoint $\hat x$ of the remaining segment satisfies
\begin{align*}
 |\hat x-x^\ast|&\mathrel{\underset{\mathmakebox[\widthof{$\scriptstyle\eqref{eq:fixed-envelope}$}]{}}{\le}}R\,2^{-n},\\
 f(\hat x)-f^\ast&\mathrel{\underset{\eqref{eq:fixed-envelope}}{\le}}\frac{L_0}{L_1^2}\bigl(e^{L_1R2^{-n}}-1-L_1R\,2^{-n}\bigr).
\end{align*}
For $n$ equal to~\eqref{eq:one-dim-count}, we have $R2^{-n}\le\rho_\varepsilon$, and the right-hand side is at most $\varepsilon$. The oracle may not have been called at the output midpoint. If the output must coincide with a query point, one more call is needed.\qedhere
\end{itemize}

<!-- remark-1: body_md -->
\label{rem:fixed-dim-growth}
For fixed $d$ and $L_1^2\varepsilon/L_0$, Proposition~\ref{prop:fixed-dim} gives only $O(\log(2+L_1R))$ calls. Therefore, a lower bound that is linear in $L_1R$ up to logarithms cannot be realized by a family of fixed dimension. It requires a mechanism that hides information in a number of directions growing together with the number of calls. This is exactly why Theorems~\ref{thm:lower-gap} and~\ref{thm:lower-radius} use a dimension proportional to the chain length.
