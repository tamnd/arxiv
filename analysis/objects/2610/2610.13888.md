<!-- question-1: body_md -->
\label{converse}
Assume that $R$ is quasi-unmixed with infinite residue field, and let
$E_1,E_2\subseteq F$ have finite colength. Does
\[
e_{\mathrm{BR}}(E_1^{[q-i]},E_2^{[i]})
=
e_{\mathrm{BR}}(E_1)
=
e_{\mathrm{BR}}(E_2)
\qquad(0\leq i\leq q)
\]
imply $\overline{E_1}=\overline{E_2}$ in $F$?
<!-- end question-1 -->
<!-- lem-1: body_md -->
\label{Coro16.5.7SH}
Suppose that $R$ is formally equidimensional, and let
$E_1\subseteq E_2\subseteq F$ satisfy $\ell_R(F/E_1)<\infty$.
Then $E_1$ is a reduction of $E_2$ in $F$ if and only if
$e_{\mathrm{BR}}(E_1)=e_{\mathrm{BR}}(E_2)$.
<!-- end lem-1 -->
<!-- prop-1: body_md -->
\label{prop:br-degree-polynomiality}
Under the preceding assumptions, there exists a polynomial
$P_E(n;N)\in\mathbb Q[n]$ of degree at most $q$ such that
\[
P_E(n;N)=\ell_R\bigl((S_{en}/E^n)\otimes_RN\bigr)
\qquad(n\gg0).
\]
<!-- end prop-1 -->
<!-- proof-1: body_md -->
Give the Veronese algebra $B=S^{(e)}$ its standard grading
$B_n=S_{en}$, and set $T=B\otimes_RN$ and $J=EB$.
Then $T$ is generated in degree zero, and right exactness gives
\[
H(n):=\ell_R\bigl((S_{en}/E^n)\otimes_RN\bigr)
\]
Equip $G=\operatorname{gr}_J(T)$ with the bigrading
\[
G_{i,j}=\frac{E^iT_j}{E^{i+1}T_{j-1}}
\qquad(i,j\geq0),
\]
where $T_{-1}=0$. The algebra $\operatorname{gr}_J(B)$ is generated
over $R$ in bidegrees $(1,0)$ and $(0,1)$, and $G$ is finite over it.
Consequently, $K=\bigoplus_{i\geq0,\,j\geq1}G_{i,j}$ is a finite
bigraded submodule.

By \eqref{eq:br-degree-colength}, there is $c\geq1$ with
$\mathfrak m^cT_1\subseteq ET_0$. Since $T_j=B_{j-1}T_1$, this implies
$\mathfrak m^cE^iT_j\subseteq E^{i+1}T_{j-1}$ for $j\geq1$.
Thus $\mathfrak m^cK=0$. Using total degree $i+j$, Hilbert--Serre
shows that $\ell_R(K_n)$ is eventually polynomial. The filtration
\[
T_n\supseteq ET_{n-1}\supseteq\cdots\supseteq E^nT_0
\]
gives $H(n)=\sum_{i=0}^{n-1}\ell_R(G_{i,n-i})=\ell_R(K_n)$,
proving polynomiality.

Iterating $\mathfrak m^cT_1\subseteq ET_0$ yields
$\mathfrak m^{cn}T_n\subseteq E^nT_0$. Therefore
\[
H(n)\leq\ell_R(T_n/\mathfrak m^{cn}T_n)
 =\binom{en+p-1}{p-1}\ell_R(N/\mathfrak m^{cn}N).
\]
Since $\dim_RN=d$, one has
$\ell_R(N/\mathfrak m^{cn}N)=O(n^d)$, while
$\binom{en+p-1}{p-1}=O(n^{p-1})$. Hence $H(n)=O(n^q)$,
which proves the degree bound.
which proves the degree bound.

<!-- defn-1: body_md -->
\label{def:br-degree}
The Buchsbaum--Rim multiplicity of $E$ in degree $e$, with
coefficients in $N$ and relative to $S$, is defined by
\[
\widetilde e_{\mathrm{BR}}(E;N)
:=q!\lim_{n\to\infty}
\frac{\ell_R((S_{en}/E^n)\otimes_RN)}{n^q}.
\]
When $N=R$, we omit $N$
from the notation.
<!-- end defn-1 -->
<!-- rem-1: body_md -->
\label{rem:br-degree-conventions}
The number $\widetilde e_{\mathrm{BR}}(E;N)$ is a nonnegative integer:
$P_E(n;N)$ is eventually integer-valued and nonnegative, so its
normalized coefficient in degree $q$ is integral and nonnegative.
It may vanish; for example, it vanishes when $E=S_e$.
For $e=1$, the definition is the classical Buchsbaum--Rim
multiplicity $e_{\mathrm{BR}}(E;N)$. For $e>1$, it is relative to
multiplication in $S$, and must not be confused with the classical
multiplicity of the inclusion into the free module $S_e$, whose
ambient symmetric algebra would instead be
$\operatorname{Sym}_R(S_e)$.

<!-- lem-2: body_md -->
\label{lem:br-degree-fiber-bound}
Let $(A,\mathfrak n)$ be a Noetherian local ring of dimension
$\delta\geq1$, let $C=A[t_1,\ldots,t_p]$, and let $U\subseteq C_e$
have finite colength, where $e\geq1$. Grade $\mathcal A=A[U]$ by
assigning degree one to $U$. Then
\[
\dim(\mathcal A/\mathfrak n\mathcal A)\leq\delta+p-1.
\]
<!-- end lem-2 -->
<!-- proof-2: body_md -->
It suffices to consider the images of $\mathcal A$ over the minimal
prime quotients of $A$. Indeed, the kernel of the map to the product
of these images is nilpotent, and the resulting finitely many
closed subsets cover $\operatorname{Spec}(\mathcal A)$.

Suppose first that $A$ is a domain of positive dimension
$\delta'$. On $X=\mathbb P_A^{p-1}$, the forms in $U$ define a
linear system which is the complete degree-$e$ system outside the
closed fiber. Its graph closure $Y$ is the blowup of its base ideal.
It is integral, has dimension $\delta'+p-1$, and maps properly and
surjectively to $\operatorname{Proj}(A[U])$. Surjectivity follows
because its generic image is dense and the morphism is proper.
The closed fiber $Y_{\mathfrak n}$ is a proper closed subset of $Y$,
so its dimension is at most $\delta'+p-2$. Surjectivity on closed
fibers now gives
$\dim(A[U]/\mathfrak nA[U])\leq\delta'+p-1$.
If the domain quotient has dimension zero, it is a field, and its
algebra of forms has dimension at most $p$. Both bounds are at most
$\delta+p-1$.
<!-- end proof-2 -->
<!-- lem-3: body_md -->
\label{lem:br-degree-shift}
With $B=S^{(e)}$, $T=B\otimes_RN$, and $H(n)=\ell_R(T_n/E^nT_0)$,
for every fixed integer $h\geq1$ one has
\[
\ell_R(T_{n+h}/E^nT_h)=H(n+h)+O(n^{q-1}).
\]
<!-- end lem-3 -->
<!-- proof-3: body_md -->
Set $R'=R/\operatorname{Ann}_R(N)$ and let $E'$ be the image of $E$
in $S_e\otimes_RR'$. The module
$(S_e\otimes_RR')/E'$ has finite length: outside the closed point
its tensor product with the nonzero localized module $N$ vanishes,
so Nakayama's lemma forces it to vanish. Also $\dim R'=d$.
Thus Lemma~\ref{lem:br-degree-fiber-bound}, applied to
$\mathcal A=R'[E']$, gives
$\dim(\mathcal A/\mathfrak m\mathcal A)\leq q$.

For fixed $h$, the graded module
\[
V_h=\bigoplus_{n\geq0}\frac{E^nT_h}{E^{n+h}T_0}
\]
is finite over $\mathcal A$, since its numerator is generated by
$T_h$ in degree zero. Choose $c$ as in the proof of
Proposition~\ref{prop:br-degree-polynomiality}. Then
$\mathfrak m^{ch}T_h\subseteq E^hT_0$, whence
$\mathfrak m^{ch}V_h=0$ and $\dim_{\mathcal A}V_h\leq q$.
Hilbert--Serre yields
$\ell_R((V_h)_n)=O(n^{q-1})$.
The assertion follows from the exact sequence
\[
0\longrightarrow\frac{E^nT_h}{E^{n+h}T_0}
\longrightarrow\frac{T_{n+h}}{E^{n+h}T_0}
\longrightarrow\frac{T_{n+h}}{E^nT_h}
\longrightarrow0.
\]

<!-- prop-2: body_md -->
\label{prop:br-degree-KT-identification}
Under the assumptions of Proposition~\ref{prop:br-degree-polynomiality},
one has
\[
\widetilde e_{\mathrm{BR}}(E;N)=e^{q,0}(E;N),
\]
Here $e^{q,0}(E;N)$ is the Buchsbaum--Rim multiplicity of
Kleiman--Thorup \cite[Section~8]{Kleiman-Thorup2}, computed for
$S\otimes_RN$ and the degree-$e$ submodule $E$.
<!-- end prop-2 -->
<!-- proof-4: body_md -->
Condition~\eqref{eq:br-degree-colength} implies that the base locus
of $E$ on $\operatorname{Proj}(S)$ meets the support of
$\widetilde{S\otimes_RN}$ only in the closed fiber. Hence
\cite[Proposition~8.2]{Kleiman-Thorup2} applies to
\[
\lambda(n,b)=\ell_R\left(
\frac{S_{en+b}\otimes_RN}{E^n(S_b\otimes_RN)}\right).
\]
For $n,b\gg0$, this is a polynomial of total degree at most $q$,
whose homogeneous component of degree $q$ is
\[
\sum_{i+k=q}\frac{e^{i,k}(E;N)}{i!\,k!}n^ib^k.
\]
Choose a fixed $h\geq1$ sufficiently large that $b=eh$ belongs to
the polynomial range. Then
\[
\lambda(n,eh)=\frac{e^{q,0}(E;N)}{q!}n^q+O(n^{q-1}).
\]
On the other hand, Lemma~\ref{lem:br-degree-shift} gives
\[
\lambda(n,eh)=H(n+h)+O(n^{q-1})
\]
Comparison proves the assertion.

<!-- rem-3: body_md -->
\label{remark1}
Let $M\subseteq F$ have finite colength, and let $r\geq1$.
Regard $M^r$ as a submodule of $S_r$. Since
$(M^r)^n=M^{rn}$ inside $S_{rn}$, the defining Hilbert functions
satisfy
\[
\ell_R\!\left((S_{rn}/(M^r)^n)\otimes_RN\right)
=\ell_R\!\left((S_{rn}/M^{rn})\otimes_RN\right).
\]
The leading term on the right is
$\widetilde e_{\mathrm{BR}}(M;N)(rn)^q/q!$.
Consequently,
\[
\widetilde e_{\mathrm{BR}}(M^r;N)
=r^q\widetilde e_{\mathrm{BR}}(M;N).
\]
In particular, for $N=R$,
$\widetilde e_{\mathrm{BR}}(M^r)=r^{d+p-1}e_{\mathrm{BR}}(M)$.
<!-- end rem-3 -->
<!-- rem-4: body_md -->
Combining \cite[Proposition~8.2]{Kleiman-Thorup2} with
multigraded Hilbert--Serre yields a polynomial in
$\mathbb Q[n_1,n_2]$ that agrees with $h_R(n_1,n_2;N)$
for $n_1,n_2\gg0$.
Indeed, one applies these results to the two outer terms of
the exact sequence

0\longrightarrow
\frac{E_2^{n_2}T_{n_1}}{E_1^{n_1}E_2^{n_2}T_0}
\longrightarrow
\frac{T_{n_1+n_2}}{E_1^{n_1}E_2^{n_2}T_0}
\longrightarrow
\frac{T_{n_1+n_2}}{E_2^{n_2}T_{n_1}}
\longrightarrow0,
\]
where $T=S\otimes_RN$, using the associated graded filtration
induced by $E_1$ for the left-hand term.
Moreover, choosing $c\geq1$ with
$\mathfrak m^cF\subseteq E_1\cap E_2$ and setting $u=n_1+n_2$,
we obtain
\[

\leq
\binom{u+p-1}{p-1}\ell_R(N/\mathfrak m^{cu}N)
\]
Since eventual polynomiality has already been established, this

<!-- prop-3: body_md -->
\label{prop:mixed-br-arbitrary-degree}
Fix positive integers $e_1,e_2$, and let
$E_i\subseteq S_{e_i}=F^{e_i}$ be submodules satisfying
$\ell_R(S_{e_i}/E_i)<\infty$ for $i=1,2$. Then the function
\[
h_R(n_1,n_2;N) = \ell_R\left( \left( S_{e_1n_1+e_2n_2}/E_1^{n_1}E_2^{n_2} \right)\otimes_RN \right)
\]
agrees, for $n_1,n_2\gg0$, with a polynomial in $\mathbb Q[n_1,n_2]$ of total degree at most $q$.
<!-- end prop-3 -->
<!-- proof-5: body_md -->
Set $T=S\otimes_RN$, $T_v=S_v\otimes_RN$. All products acting on $T$ are understood as the images of the corresponding multiplication maps. By right exactness of tensor products, $\left( S_u/E_1^{n_1}E_2^{n_2} \right)\otimes_RN \cong T_u/E_1^{n_1}E_2^{n_2}T_0$, where $u=e_1n_1+e_2n_2$.

Since $E_1^{n_1}E_2^{n_2}T_0 \subseteq E_2^{n_2}T_{e_1n_1} \subseteq T_u$, we obtain an exact sequence
\[
0\longrightarrow \frac{E_2^{n_2}T_{e_1n_1}}{E_1^{n_1}E_2^{n_2}T_0} \longrightarrow \frac{T_u}{E_1^{n_1}E_2^{n_2}T_0} \longrightarrow \frac{T_u}{E_2^{n_2}T_{e_1n_1}} \longrightarrow 0.
\]
Consequently, $h_R(n_1,n_2;N)=A(n_1,n_2)+B(n_1,n_2)$, where $A(n_1,n_2)$ and $B(n_1,n_2)$ denote the lengths of the left-hand and right-hand terms, respectively.

We first consider $B(n_1,n_2)$. Since $\ell_R(S_{e_2}/E_2)<\infty$, the hypotheses of \cite[Proposition~8.2]{Kleiman-Thorup2} are satisfied. Hence there exists a polynomial $Q(a,b)\in\mathbb Q[a,b]$ such that $\ell_R\left( T_{e_2a+b}/E_2^aT_b \right) = Q(a,b)$ for $a,b\gg0$. Taking $a=n_2$ and $b=e_1n_1$, we obtain $B(n_1,n_2)=Q(n_2,e_1n_1)$ for $n_1,n_2\gg0$. Thus $B$ is eventually polynomial in $n_1,n_2$.

We now prove the eventual polynomiality of $A$. Consider the externally bigraded algebra and module $\mathcal C = \bigoplus_{j,b\geq0}E_2^bS_{e_1j}$ and $\mathcal T = \bigoplus_{j,b\geq0}E_2^bT_{e_1j}$, where the summand $E_2^bS_{e_1j}$ has bidegree $(j,b)$.

Since $S_{e_1j}=(S_{e_1})^j$, the algebra $\mathcal C$ is generated over $R$ by its components of degrees $(1,0)$ and $(0,1)$. Moreover, $\mathcal T$ is a finite graded $\mathcal C$-module generated by $\mathcal T_{0,0}=T_0=N$.

Let $\mathcal J$ be the ideal of $\mathcal C$ generated by $E_1$ in bidegree $(1,0)$. Set $\mathcal A=\operatorname{gr}_{\mathcal J}(\mathcal C)$ and $G=\operatorname{gr}_{\mathcal J}(\mathcal T)$. Then $G$ is a finite graded module over the Noetherian algebra $\mathcal A$.

Combining the associated grading with the original bigrading, and writing $j$ for the residual first degree after removing the $\mathcal J$-filtration degree, we obtain
\[
G_{i,j,b} = \frac{ E_1^iE_2^bT_{e_1j} }{ E_1^{i+1}E_2^bT_{e_1(j-1)} }, \qquad i,j,b\geq0,
\]
where the denominator is understood to be zero when $j=0$.

The generators of $\mathcal A$ have tridegrees $(1,0,0)$, $(0,1,0)$, and $(0,0,1)$. The submodule
\[
K=\bigoplus_{i,b\geq0,\,j\geq1}G_{i,j,b}
\]
is stable under the action of $\mathcal A$, and hence is finite over $\mathcal A$.

Choose $c\geq1$ such that $\mathfrak m^cS_{e_1}\subseteq E_1$. For $j\geq1$, the equality $T_{e_1j}=S_{e_1}T_{e_1(j-1)}$ implies $\mathfrak m^cE_1^iE_2^bT_{e_1j} \subseteq E_1^{i+1}E_2^bT_{e_1(j-1)}$. Thus $\mathfrak m^cK=0$.

Now replace the trigrading by the bigrading $(i,j,b)\longmapsto(i+j,b)$. Under this regrading, $\mathcal A$ becomes a standard bigraded algebra generated in degrees $(1,0)$ and $(0,1)$. Therefore $K$ is a finite bigraded module over $\mathcal A/\mathfrak m^c\mathcal A$, whose degree-zero component is the Artinian ring $R/\mathfrak m^c$.

The multigraded Hilbert--Serre theorem gives a polynomial
$P_K(X,Y)\in\mathbb Q[X,Y]$ such that
\[
\ell_R(K_{n_1,n_2})=P_K(n_1,n_2)
\qquad(n_1,n_2\gg0).
\]

By the definition of the regrading, $K_{n_1,n_2} = \bigoplus_{i=0}^{n_1-1} G_{i,n_1-i,n_2}$. The filtration $$E_2^{n_2}T_{e_1n_1} \supseteq E_1E_2^{n_2}T_{e_1(n_1-1)} \supseteq\cdots\supseteq E_1^{n_1}E_2^{n_2}T_0$$ has successive quotients $G_{i,n_1-i,n_2}$ for $0\leq i\leq n_1-1$. Therefore,
\[
A(n_1,n_2) = \sum_{i=0}^{n_1-1} \ell_R\left(G_{i,n_1-i,n_2}\right) = \ell_R(K_{n_1,n_2}) = P_K(n_1,n_2)
\]
for $n_1,n_2\gg0$. Hence $h_R(n_1,n_2;N) = P_K(n_1,n_2)+Q(n_2,e_1n_1)$ for all sufficiently large $n_1,n_2$. Thus $h_R$ agrees eventually with a polynomial $P(n_1,n_2)\in\mathbb Q[n_1,n_2]$.

It remains to bound the total degree of $P$. Choose $c\geq1$ such that $\mathfrak m^cS_{e_i}\subseteq E_i$ ($i=1,2$). Since $S$ is a polynomial ring, $(S_{e_1})^{n_1}(S_{e_2})^{n_2}=S_u$ for $u=e_1n_1+e_2n_2$. Consequently, $\mathfrak m^{c(n_1+n_2)}T_u \subseteq E_1^{n_1}E_2^{n_2}T_0$. Since $T_u\cong N^{\binom{u+p-1}{p-1}}$ as an $R$-module, we obtain
\[
\begin{aligned}
0\leq h_R(n_1,n_2;N)
&\leq \ell_R\!\left(T_u/\mathfrak m^{c(n_1+n_2)}T_u\right)\\
&=\binom{u+p-1}{p-1}
\ell_R\!\left(N/\mathfrak m^{c(n_1+n_2)}N\right)\\
&=O\!\left((n_1+n_2)^q\right).
\end{aligned}
\]

Suppose that $P$ had total degree $D>q$, and let $P_D$ denote its homogeneous component of degree $D$. Since $P_D\neq0$, there exist positive integers $a,b$ such that $P_D(a,b)\neq0$. Hence $P(an,bn) = P_D(a,b)n^D+O(n^{D-1})$, which contradicts the estimate $P(an,bn)=O(n^q)$. Therefore, $\deg P\leq q=d+p-1$. This proves the proposition.

<!-- lem-4: body_md -->
\label{lem:mixed-product-formula}
Let $E_1,E_2\subseteq F$ have finite colength. Regarding
$E_1E_2$ as a submodule of
$F^2=\operatorname{Sym}_R^2(F)$, one has
\[
\widetilde e_{\mathrm{BR}}(E_1E_2;N)
=
\sum_{i=0}^{q}
\binom{q}{i}
e_{\mathrm{BR}}
\left(E_1^{[q-i]},E_2^{[i]};N\right).
\]
<!-- end lem-4 -->
<!-- proof-6: body_md -->
Set $S=\operatorname{Sym}_R(F)$.
First, $E_1E_2$ has finite colength in $S_2$:
if $\mathfrak m^cF\subseteq E_1\cap E_2$, then
$\mathfrak m^{2c}S_2\subseteq E_1E_2$.
Thus $\widetilde e_{\mathrm{BR}}(E_1E_2;N)$ is defined.

By Proposition~\ref{prop:mixed-br-arbitrary-degree}, for
$n_1,n_2\gg0$ one has
\[
\ell_R\!\left(
(S_{n_1+n_2}/E_1^{n_1}E_2^{n_2})\otimes_RN
\right)
=
\sum_{i=0}^{q}
\frac{e_{\mathrm{BR}}(E_1^{[q-i]},E_2^{[i]};N)}
{(q-i)!\,i!}\,
n_1^{q-i}n_2^i
+P_{<q}(n_1,n_2),
\]
where $P_{<q}$ has total degree less than $q$.

\label{lem:mixed-br-basic-properties}

\begin{enumerate}
\item For finite-colength submodules $E_1,E_2\subseteq F$ and

$0\leq i\leq q$,

\]

finite-colength submodules of $F$, with $K_j$ a reduction of $M_j$
for $j=1,2$, then
\[
e_{\rm BR}\!\left(K_1^{[q-i]},K_2^{[i]}\right)
=e_{\rm BR}\!\left(M_1^{[q-i]},M_2^{[i]}\right)
\qquad(0\leq i\leq q).
\]
\end{enumerate}

\smallskip
\noindent\textup{(i)}
For every pair of positive integers $r,s$, Corollary~\ref{mixed}
and the identity
$(E_1^u)^r(E_2^v)^s=E_1^{ur}E_2^{vs}$ give
\[
\begin{aligned}
\sum_{i=0}^{q}\binom{q}{i}
\widetilde e_{\rm BR}
\left((E_1^u)^{[q-i]},(E_2^v)^{[i]}\right)
r^{q-i}s^i&= \widetilde e_{\rm BR}(E_1^{ur}E_2^{vs})\\
=&
\sum_{i=0}^{q}\binom qi
e_{\rm BR}\left(E_1^{[q-i]},E_2^{[i]}\right)
u^{q-i}v^i r^{q-i}s^i.
\end{aligned}
\]
Since this equality holds for all positive integers $r,s$,
it is a polynomial identity. Comparing coefficients of
$r^{q-i}s^i$ proves \textup{(i)}.

\smallskip
\noindent\textup{(ii)}
By Corollary~\ref{mixed} and Remark~\ref{remark1},
\[
\begin{aligned}
\sum_{i=0}^{q}\binom qi
e_{\rm BR}\left(E_1^{[q-i]},E_1^{[i]}\right)
r^{q-i}s^i
&=\widetilde e_{\rm BR}(E_1^{r+s})\\
&=e_{\rm BR}(E_1)(r+s)^q\\
&=\sum_{i=0}^{q}\binom qi
e_{\rm BR}(E_1)r^{q-i}s^i.
\end{aligned}
\]
Comparing coefficients proves \textup{(ii)}.

\noindent\textup{(iii)}
We first establish reduction invariance for multiplicities
in arbitrary degree. Let $A\subseteq B\subseteq S_e$ have
finite colength, and suppose that $A$ is a reduction of $B$.
Choose $h\geq1$ such that
\(
B^n=A^{n-h}B^h
\,\,(n\geq h).
\)
Since $A^n\subseteq B^n\subseteq A^{n-h}S_{eh}$, we obtain
\[
\ell_R(S_{en}/A^{n-h}S_{eh})
\leq
\ell_R(S_{en}/B^n)
\leq
\ell_R(S_{en}/A^n).
\]
Applying Lemma~\ref{lem:br-degree-shift} to $A$, with
coefficient module $R$ and index $n-h$, gives
\[
\ell_R(S_{en}/A^{n-h}S_{eh})
=
\ell_R(S_{en}/A^n)+O(n^{q-1}).
\]
Thus the two outer terms have the same normalized leading
coefficient. Dividing by $n^q/q!$ and taking limits yields
\[
\widetilde e_{\rm BR}(A)=\widetilde e_{\rm BR}(B).
\]

of $M_1$ and $K_2$ is a reduction of $M_2$, for $n\gg0$ we have
\[
M_1^{r(n+1)}=K_1^rM_1^{rn},
\qquad
M_2^{s(n+1)}=K_2^sM_2^{sn}.
\]
Consequently,
\[
(M_1^rM_2^s)^{n+1}
=(K_1^rK_2^s)(M_1^rM_2^s)^n.
\]
Hence $K_1^rK_2^s$ is a reduction of $M_1^rM_2^s$, both
regarded as submodules of $S_{r+s}$.

\subseteq K_1^rK_2^s
\subseteq M_1^rM_2^s.
\]
The reduction invariance proved above therefore gives
\[
\widetilde e_{\rm BR}(K_1^rK_2^s)
=\widetilde e_{\rm BR}(M_1^rM_2^s).
\]
By Corollary~\ref{mixed}, this equality becomes
\[
\begin{aligned}
&\sum_{i=0}^{q}\binom qi
e_{\rm BR}\left(K_1^{[q-i]},K_2^{[i]}\right)r^{q-i}s^i=
\sum_{i=0}^{q}\binom qi
e_{\rm BR}\left(M_1^{[q-i]},M_2^{[i]}\right)r^{q-i}s^i.
\end{aligned}
\]
Since it holds for all positive integers $r,s$, comparison
of coefficients proves \textup{(iii)}.

The following conditions are equivalent:
\begin{enumerate}
\item[(a)]
All mixed multiplicities

\qquad(1\leq i\leq q-1),

The bracketed expressions vanish for $i=0,q$ and are

bracketed expression must vanish. This proves
\textup{(b)}, and hence also \textup{(a)}.

{\rm Equality of all mixed Buchsbaum--Rim multiplicities does
not imply equality of integral closures, even over a
complete, reduced, Cohen--Macaulay local ring.

Let $k$ be an infinite field and set
\(
R=k[[x,y]]/(xy),\,\, \mathfrak m=(x,y),
\)
where $x,y$ also denote their residue classes in $R$.
The ring $R$ is complete, reduced, Cohen--Macaulay, and
equidimensional of dimension one; in particular, it is
quasi-unmixed. Consider
\[
I=(x,y^2),\qquad J=(x^2,y),\qquad
F=Rf_1\oplus Rf_2,
\]
and set $E_1=IF$ and $E_2=JF$.
Both modules have finite colength in $F$, and
$q=\dim R+\operatorname{rank}F-1=2$.

For integers $a,b\geq1$, the relation $xy=0$ gives
\(
I^aJ^b=(x^{a+2b},y^{2a+b}).
\)
Since
\[
\ell_R\bigl(R/(x^u,y^v)\bigr)=u+v-1
\qquad(u,v\geq1),
\]
we obtain $\ell_R(R/I^aJ^b)=3(a+b)-1$.
Writing $S=\operatorname{Sym}_R(F)=R[t_1,t_2]$, we have
\[
E_1^aE_2^b=I^aJ^bS_{a+b}.
\]
The module $S_{a+b}$ is free of rank $a+b+1$, so
\[
\begin{aligned}
\ell_R(S_{a+b}/E_1^aE_2^b)
&=(a+b+1)\ell_R(R/I^aJ^b)\\
&=(a+b+1)\bigl(3(a+b)-1\bigr)\\
&=3a^2+6ab+3b^2+2a+2b-1.
\end{aligned}
\]
Comparing its quadratic terms with
\[
\frac{e_{\rm BR}(E_1)}{2}a^2
+e_{\rm BR}(E_1^{[1]},E_2^{[1]})ab
+\frac{e_{\rm BR}(E_2)}{2}b^2
\]
yields
\[
e_{\rm BR}(E_1)
=e_{\rm BR}(E_1^{[1]},E_2^{[1]})
=e_{\rm BR}(E_2)=6.
\]

Nevertheless, $\overline{E_1}\neq\overline{E_2}$ in $F$.
Indeed, $xf_1\in E_1$. Suppose that
$xf_1\in\overline{E_2}$. Identifying $f_1,f_2$ with
$t_1,t_2$, respectively, there would exist an equation
\[
(xt_1)^n+a_1(xt_1)^{n-1}+\cdots+a_n=0,
\qquad a_j\in E_2^j,
\]
for some $n\geq1$.

\[
\varphi:R[t_1,t_2]\longrightarrow k[[x]],
\qquad
x\longmapsto x,\quad y\longmapsto0,\quad
t_1\longmapsto1,\quad t_2\longmapsto0.
\]
Since $\varphi(E_2)\subseteq(x^2)$, this gives
\[
x^n+\varphi(a_1)x^{n-1}+\cdots+\varphi(a_n)=0,
\qquad \varphi(a_j)\in(x^{2j}).
\]
Thus $x$ would be integral over $(x^2)$ in the discrete
valuation ring $k[[x]]$. This is impossible because
$(x^2)$ is integrally closed and $x\notin(x^2)$.
Therefore
\(
xf_1\in\overline{E_1}\setminus\overline{E_2}.
\)

\end{enumerate}

Conversely, assume \textup{(i)}, and write
$A=e_{\rm BR}(E_1)=e_{\rm BR}(E_2)$.

\[
e_{\rm BR}\left((IF)^{[q-i]},(JF)^{[i]}\right)=A
\qquad(0\leq i\leq q).

\(
(IF)^a(JF)^b=I^aJ^bS_{a+b},
\)
we have
\[
\ell_R\left(S_{a+b}/(IF)^a(JF)^b\right)
=
\binom{a+b+p-1}{p-1}\ell_R(R/I^aJ^b).
\]

\[
\frac{A}{q!}(X+Y)^q

\frac{(X+Y)^{p-1}}{(p-1)!\,d!}
\sum_{j=0}^{d}\binom dj
e\left(I^{[d-j]},J^{[j]}\right)X^{d-j}Y^j.
\]

\[
e\left(I^{[d-j]},J^{[j]}\right)
=\frac{A}{\binom qd}
=e(I)=e(J)
\qquad(0\leq j\leq d).
\]

\[
e(IJ)=2^d e(I),
\]
and hence
\[
e(IJ)^{1/d}=e(I)^{1/d}+e(J)^{1/d}.
\]
Since $R$ is quasi-unmixed and $d\geq2$, the equality
criterion in \cite[Theorem~1.16]{Verma} yields positive
integers $r,s$ such that
$\overline{I^r}=\overline{J^s}$.
Taking multiplicities gives
\(
r^d e(I)=s^d e(J).
\)
Thus $r=s$, and consequently $\overline I=\overline J$.

=H^{n+1}S_{n+1}
=IH^nS_{n+1}

is a reduction of $HF$. It follows that
\[
\overline{E_1}
=\overline{IF}
=\overline{HF}
=\overline{JF}
=\overline{E_2},
\]
as required.
