<!-- theorem-1: body_md -->
\label{thm:main}
    For a polytope $P\subset\mathbb{R}^d$ holds
     
    \begin{equation}
        \label{eq:main}
        \operatorname{vol}(P) \operatorname{vol}(P-P)^\circ \le \frac{\mathcal F(P)}{(d!)^2}
    \end{equation}
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{res:cs}
    For a centrally symmetric polytope $P\subset\mathbb{R}^d$ holds
     
    \begin{equation}
        \label{eq:cs}
        \operatorname{vol}(P)\operatorname{vol}(P^\circ) \le \frac{2^d}{(d!)^2} \mathcal F(P).
    \end{equation}
<!-- end corollary-1 -->
<!-- proof-1: body_md -->
We have $\operatorname{vol} (P-P)^\circ = \operatorname{vol} (2P)^\circ = 2^{-d}\operatorname{vol}(P^\circ)$.
    Hence
     
    \[\operatorname{vol}(P) \operatorname{vol}(P^\circ) = 2^d \operatorname{vol}(P) \operatorname{vol}(P-P)^\circ \overset{\text{ 
  \hyperref[{eq:main}]{\textup{\tagform@{\ref*{eq:main}}}} 
}}\le \frac{2^d}{(d!)^2} \mathcal F(P).\qedhere\]

<!-- lemma-1: body_md -->
For each flag $\boldsymbol{F}$ holds
     
    $$\sum_v \operatorname{vol}(\Delta_v(\boldsymbol{F}))\operatorname{vol}(Q_v)\le \frac1{(d!)^2}.$$
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
We have
     
    $$
    \operatorname{vol}(\Delta_v(\boldsymbol{F})) = \frac1{d!}
    \left|\smash{\,\det\!\underbrace{
    \begin{pmatrix}
        v(F_0) - v(F_1)
        \\
        \vdots
        \\
        v(F_{d-1}) - v(F_d)
    \end{pmatrix}}_{=:D_v}}
    \mathrlap{\phantom{
    \begin{matrix}
        x \\ x \\ x  
    \end{matrix}}}\right|
    .
    \mathrlap{\phantom{
    \underbrace{\begin{matrix}
        x \\ x \\ x  
    \end{matrix}}_D
    }}
    $$
     
    and therefore
     
    $$\sum_v\operatorname{vol}(\Delta_v(\boldsymbol{F}))\operatorname{vol}(Q_v)= \frac1{d!}\sum_v|\kern-1pt\det(D_v)|\operatorname{vol}(Q_v)=\frac1{d!} \sum_v\operatorname{vol}(D_v Q_v).$$
     
    Thus, it suffices to bound the right sum by $1/d!$.
    This follows with two ingredients:
     
    \begin{myenumerate}
        \item $D_vQ_v\subseteq \boldsymbol{\Delta}_d:=\{x\in\mathbb{R}^d_{\ge0}\mid x_1+\cdots+x_d \le 1\}$, and
        \item if $v\ne w$ then $D_vQ_v$ and $D_w Q_w$ have disjoint interiors.
    \end{myenumerate}
     
    Then
     
    \begin{equation}
    \label{eq:le_simplex}
        \sum_v \operatorname{vol}(D_v Q_v) \overset{\text{{\upshape(\textit{\romannumeral2\relax})}}}= \operatorname{vol}\Big(\bigcup_v D_vQ_v\Big) \overset{\text{{\upshape(\textit{\romannumeral1\relax})}}}\le \operatorname{vol}(\boldsymbol{\Delta}_d) =
        \frac1{d!}.        
    \end{equation}

    Proof of {\upshape(\textit{\romannumeral1\relax})}:
    fix $t\in Q_v$. Then $(D_v t)_k = \<t,v(F_{k-1})\>-\<t,v(F_k)\> \ge0$ by  
  \hyperref[{eq:order}]{\textup{\tagform@{\ref*{eq:order}}}} 
.\nolinebreak\space
    Also, through telescoping we get
     
    \begin{equation}
    \label{eq:le_one}
        \sum_k (D_v t)_k = \<t,v(F_0)\>-\<t,v(F_d)\>\le 1,
    \end{equation}
     
    where the inequality follows from
     
    $$t\in Q=(P-P)^\circ = \big\{\,t\in\mathbb{R}^d \;\big\vert\; \max_{x\in P}\<t,x\> - \min_{x\in P} \<t,x\> \le 1\,\big\}.$$

    Proof of {\upshape(\textit{\romannumeral2\relax})}:
    For sake of contradiction suppose that $D_vQ_v$ and $D_wQ_w$ intersect~in an interior point
    $D_vt_v=D_wt_w$. In particular, $D_v$ and $D_w$ are non-singular, and~$t_v\in \operatorname{int}(Q_v)$ and $t_w\in \operatorname{int}(Q_w)$.
    Unfolding the definition of the matrices yields
     
    \begin{equation}
        \label{eq:eq}
        \<t_v,v(F_{k-1})\>-\<t_v,v(F_k)\> = \<t_w,w(F_{k-1})\>-\<t_w,w(F_k)\>,\;\;\text{for $k\in\{1,...,d\}$}.
    \end{equation}
    
    Note that $v(F_0)=w(F_0)$, and we may translate $P$ so that $v(F_0)=w(F_0)=0$.
    Then the system  
  \hyperref[{eq:eq}]{\textup{\tagform@{\ref*{eq:eq}}}} 
 can be equivalently expressed as
     
    \begin{equation}
        \label{eq:eq2}
        \<t_v,v(F_k)\> = \<t_w,w(F_k)\>,\;\;\text{for $k\in\{0,...,d\}$}.
    \end{equation}

    We show that the linear forms $\<t_v,\_\>$ and $\<t_w,\_\>$ agree on $F_k$ for all $k$.
    This~is clear for $k=0$.
    Suppose then that they agree on $F_{k-1}$.
                                       
    Since on $F_k$ the vertex~$v(F_k)$ \mbox{minimizes~$\<t_v,\_\>$}, and the vertex $w(F_k)$ minimizes $\<t_w,\_\>$, we get
     
                    
                                                                                                         
            
                                                                                                         
                  
     
    $$
        \<t_v,w(F_k)\> \ge \<t_v,v(F_k)\> \overset{ 
  \hyperref[{eq:eq2}]{\textup{\tagform@{\ref*{eq:eq2}}}} 
}= \<t_w,w(F_k)\>
    $$
    \vspace{-3ex}
    $$
        \<t_v,v(F_k)\> \overset{ 
  \hyperref[{eq:eq2}]{\textup{\tagform@{\ref*{eq:eq2}}}} 
}= \<t_w,w(F_k)\> \le \<t_w,v(F_k)\>
    $$
     
    Since $v(F_k)$ and $w(F_k)$ lie on the same side of $F_{k-1}$ in $F_k$, we must have $\<t_v,\_\>=\<t_w,\_\>$ on both points, and actually, on all of $F_k$.
                                                                                                          
     
                                                                                
    In particular, since $F_d$ is spanning, we get $t_v=t_w$.
    This means $t_v,t_w\in \operatorname{int}(Q_v)\cap \operatorname{int}(Q_w)$, which implies $v=w$. 
    \qedhere
