<!-- definition-1: body_md -->
For $d \in \mathbb{N}$, $1 \leq p<d$, define the set 
    \begin{align*}
        K^p:= \{f: \mathbb{R}^d \to \mathbb{R}|\ f \geq 0, f \in L^{\frac{pd}{d-p}}(\mathbb{R}^d), Df \in L^p(\mathbb{R}^d, \mathbb{R}^d)\}.
    \end{align*}
    Given a set $A \subset \mathbb{R}^d$, define the $p$-capacity as
    \begin{align*}
        \mathrm{Cap}_p(A):= \inf\Big\{ \int_{\mathbb{R}^n} |Df|^p\, {\rm d} x \big|\ f \in K^p, A \subset \{f \geq 1\}^\circ\Big\},
    \end{align*}
    where $\{f \geq 1\}^\circ$ denotes the interior of the set $\{f \geq 1\}$.

<!-- theorem-1: body_md -->
\label{basic_properties_capa}
    Assume $A \subset \mathbb{R}^d$. Then
    \begin{itemize}
        \item[(i)] $\mathrm{Cap}_p(A) = \inf\{\mathrm{Cap}_p(U)|\ U\ \mathrm{open,\ } A \subset U\}$. 
        \item[(ii)] $\mathrm{Cap}_p(\lambda A)= \lambda^{d-p} \mathrm{Cap}_p(A)$  for $\lambda>0$.
        \item[(iii)] $\mathrm{Cap}_p(LA)= \mathrm{Cap}_p(A)$ for any affine isometry $L: \mathbb{R}^d \to \mathbb{R}^d$.
        \item[(iv)] $\mathrm{Cap}_p(A)\leq C \mathcal{H}^{d-p}(A)$ for some constant $C$ depending only on $d$ and $p$. 
        \item[(v)] $\mathcal{L}^d(A) \leq C\mathrm{Cap}_p(A)^{\frac{d}{d-p}}$ for some constant $C$ depending only on $d$ and $p$. 
    \end{itemize}

<!-- lemma-2: body_md -->
\label{capacity_integral_estimate_one_line2d}
    Let $L >1$, $\lambda \in (0,L)$, $b>0$. Let $\gamma:[0,1] \to \mathbb{R}^2$ be a continuous curve such that $\gamma([0,1]) \subset B^{\mathbb{R}^2}_{L}(0)$ and $|\gamma_0-\gamma_1|_{\mathbb{R}^2} \geq \lambda$. Now let $f \in C_c^{0,1}(\mathbb{R}^2, [0,1])$ such that $f \equiv b$ on $\gamma([0,1])$. Then
    \begin{align*}
        \int_{B_{4L}(0)} |f| + |Df| {\rm d} x \geq 2b\lambda.
    \end{align*}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
By potentially applying an orthogonal linear transformation, we may assume that $\gamma_{1}-\gamma_{0} \in \mathrm{span}(e_1)$. 
    Denote by $P_1: \mathbb{R}^2 \to \mathbb{R}$ the projection on the first coordinate. 
    Denote $y_1:= \min(P_1(\gamma_0), P_1(\gamma_1))$.
    \begin{align}\label{first_fubini2d}
        \int_{B_{4L}(0)} |f| + |Df| {\rm d} x \geq \int_{[-2L, 2L]} \int_{\{y\}\times [-2L, 2L]} |f| + |Df|\,{\rm d} \mathcal{L}^{1}(x) {\rm d} \mathcal{L}^1(y).
    \end{align}
    Now, by the intermediate value theorem and the choice of $f$, we have that for each $y \in [y_1, y_1 + \lambda]$ there exists a $x_y \in [-L, L]$ such $(y, x_y) = \gamma_t$ for some $t \in [0,1]$ and hence $f((y, x_y)) =b$. 
    Now fix a $y \in [y_1, y_1 + \lambda]$. 
    Then, 
    \begin{align}\label{2dslice_integral}
        \int_{\{y\}\times [-2L, 2L]} |f| + |Df| {\rm d} x \geq \int_{\{y\}\times [x_y-L, x_y+L]} |f| + |Df|{\rm d} x.
    \end{align}
    The last integral can be rewritten as
    \begin{align*}
       \sum_{v \in \{-1, +1\}} \int_0^L (|f| + |Df|)((y,x_y)+ \rho (0,v))\, {\rm d} \rho.
    \end{align*}
    Fix a direction $v \in \{-1, +1\}$ and denote $f_v\colon [0,L] \to [0,1],\ \rho \mapsto f((y,x_y)+ \rho (0,v))$. Denote $\sigma := \min_{\rho \in [0,L]} f_v(\rho)$. 
    Then, we have that 
    \begin{align}\label{function_integral_on_line2d}
        \int_0^L  |f| ((y,x_y)+ \rho (0,v))\, {\rm d} \rho = \int_0^L|f_v| {\rm d} \rho \geq L\sigma \geq \sigma.
    \end{align}
    On the other hand, Lemma \ref{twodim_rotation_gradient} applied to $\max(f_v-\sigma, 0)$ yields that  
    \begin{align}\label{gradient_integral_on_line2d}
        \int_0^L |Df| ((y,x_y)+ \rho (0,v))\, {\rm d} \rho = \int_0^L |f_v'| {\rm d} \rho \geq b-\sigma.
    \end{align}
    Summing up \eqref{function_integral_on_line2d} and \eqref{gradient_integral_on_line2d} gives
    \begin{align}
        \int_0^L  (|f| + |Df|)((y,x_y)+ \rho (0,v))\, {\rm d} \rho \geq b. 
    \end{align}
    As this estimate holds independently of $v$, we get that 
    \begin{align}
         \sum_{v \in \{-1, +1\}} \int_0^L  (|f| + |Df|)((y,x_y)+ \rho (0,v))\, {\rm d} \rho \geq  2b.
    \end{align}
    Combining this with \eqref{first_fubini2d} and \eqref{2dslice_integral}, we get that 
    \begin{align*}
        \int_{B_{4L}(0)} |f| + |Df| {\rm d} x \geq 2b\lambda.
    \end{align*}
    This proves the lemma.

<!-- lemma-3: body_md -->
\label{2dcapacity_of_line_segments} 
    Let $L >1$. Let $K \subset [0,1]$ be the finite union of pairwise disjoint compact intervals $I_1, \ldots, I_n$ and let $\gamma:K \to B^{\mathbb{R}^2}_L(0)$ be a $(\frac{1}{L}, L)$-bi-Lipschitz continuous map. Now let $f \in C_c^{0,1}(\mathbb{R}^2, [0,1])$ such that $f \equiv 1$ on $\gamma(K)$. Then it holds 
    \begin{align*}
        \int_{B_{4L}(0)}|f|+|Df| {\rm d} x \geq \frac{2\mathcal{L}^1(K)}{L}.
    \end{align*}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
We proceed by induction. \\
    \textbf{Step 1.} $n=1$. \\
    We have that $\gamma(K) = \gamma(I_1) \subset B_L(0)$ and $|\gamma_{\min I_1}-\gamma_{\max I_1}|_{\mathbb{R}^2} \geq \frac{\mathcal{L}^1(I_1)}{L}$. Then $\gamma$ is in particular a curve and the case $n=1$ now follows from Lemma \ref{capacity_integral_estimate_one_line2d}, after reparametrising $\gamma$ and setting $b=1$. \\
    \textbf{Step 2.} $n>1$. \\
    Assume the property holds true for all $m \leq n-1$. Define 
    \begin{align*}
        a:= \sup\{s \in [0,1]: I_1, I_n\ \mathrm{are\ in\ the\ same \ connected\ component\ of\ } f^{-1}((s, \infty)) \cap \overline{B_{4L}(0)}\}.
    \end{align*}
    In the following all statements about openness and compactness will be made with respect to the Euclidean subspace topology in $\overline{B_{4L}(0)}$.
    Let $\varepsilon \in (0, \min(a, 1-a)/2)$.
    Then $\gamma(I_1)$ and $\gamma(I_n)$ are path connected in $f^{-1}(a-\varepsilon, \infty)\cap \overline{B_{4L}(0)}$. Hence, there exists a continuous curve $\Tilde{\gamma}:[0,1] \to f^{-1}(a-\varepsilon, \infty)\cap \overline{B_{4L}(0)}$ such that $\Tilde{\gamma}_0 = \gamma_{\min {I_1}}$ and $\Tilde{\gamma}_1 = \gamma_{\max {I_n}}$. Note that
    \begin{align}
        |\gamma_{\min {I_1}}-\gamma_{\max I_n}|_{\mathbb{R}^2} \geq \frac{1}{L}\mathcal{L}^1(K).
    \end{align}
    Then Lemma \ref{capacity_integral_estimate_one_line2d} shows that for $\Tilde{f}:= \min(f, a-\varepsilon)$, it holds
    \begin{align}
        \int_{B_{4L}(0)}|\Tilde{f}|+|D\Tilde{f}| {\rm d} x \geq \frac{2(a-\varepsilon)\mathcal{L}^1(K)}{L}. 
    \end{align}
    Now consider $\check{f}:= \max(f, a+2\varepsilon)-a-2\varepsilon$. Then $\mathrm{supp}\, D\Tilde{f} \cap \mathrm{supp}\, D \check{f} = \emptyset$. Now there exists $j \in \{2, \ldots, n\}$, pairwise disjoint nonempty sets $B_1, \ldots, B_j \subset \{1, \ldots, n\}$, and pairwise disjoint open sets $O_1, \ldots, O_j \subset f^{-1}((a+\varepsilon, \infty)) \cap \overline{B_{4L}(0)}$ such that for each $k$, $O_k$ is a connected component of $f^{-1}((a+\varepsilon, \infty))$ and
    \begin{align}
        \bigcup_{i \in B_k} \gamma(I_i) \subset O_k.
    \end{align}
    Fix a $k \in \{1, \ldots, j\}$. Define
    \begin{align}
        \check{f}_k:= \check{f}\mathbbm{1}_{O_k}. 
    \end{align}
    Note that 
    \begin{align*}
        \check{f}_k \in C^{0,1}_c(B_{4L}),
    \end{align*}
    as $f^{-1}([a+ \frac{3\varepsilon}{2}, \infty)) \cap \overline{B_{4L}(0)}$ is compactly contained in $f^{-1}((a+\varepsilon, \infty))\cap \overline{B_{4L}(0)}$, so for each $k$ there exists a function $ \eta_k\in  C_c^\infty(O_k\cap\overline{B_{4L}(0))}, [0,1])$ such that $\eta_k \equiv 1$ on $f^{-1}([a+ \frac{3\varepsilon}{2}, \infty)) \cap \overline{B_{4L}(0)}$, hence $\check{f}_k = \eta_k \check{f}$.
    
    Denote $J_k:= \bigcup_{i \in B_k} I_i \subset [0,1]$ and $\gamma_k:= \gamma|_{J_k}$. Then $\gamma_{k}$ is $(\frac{1}{L}, L)$-bi-Lipschitz. 
    Now as $|B_k|< n$, rescaling the induction hypothesis yields that 
    \begin{align}
        \int_{B_{4L}(0)}|\check{f}_k|+|D\check{f}_k| {\rm d} x \geq \frac{2\mathcal{L}^1(J_k)}{L} (1-a-2\varepsilon). 
    \end{align}
    Using that $f \geq \check{f}+ \Tilde{f}$, as well as $\check{f}, \Tilde{f}\geq 0$, and for $k \neq k'$, it holds $\mathrm{supp}\, \check{f}_{k'} \cap \mathrm{supp} \check{f}_k =\emptyset$, we get that 
    \begin{align*}
         \int_{B_{4L}(0)}|f|+|Df| {\rm d} x &\geq  \int_{B_{4L}(0)}|\Tilde{f}|+|D\Tilde{f}| {\rm d} x + \sum_{k=1}^j \int_{B_{4L}(0)}|\check{f}_k|+|D\check{f}_k| {\rm d} x \\
         & \geq \frac{2(a-\varepsilon)\mathcal{L}^1(K)}{L} + \sum_{k=1}^j \frac{2\mathcal{L}^1(J_k)}{L} (1-a-2\varepsilon) \geq \frac{(2-6\varepsilon)\mathcal{L}^1(K)}{L}.
    \end{align*}
    As $\varepsilon$ was arbitrary, the lemma follows.

<!-- definition-3: body_md -->
Let $\gamma \in C([0,1],X)$ be a curve. Then we say that $\gamma$ is absolutely continuous if there exists a function $g \in L^1([0,1], [0, \infty))$ such that for all $0 \leq s<t \leq 1$, it holds
    \begin{align*}
        \mathsf d(\gamma_s, \gamma_t) \leq \int_s^t g(r) \, {\rm d} r.
    \end{align*}
    There exists a minimal such function $g$, which we call the metric speed of $\gamma$ and denote it by $|\dot{\gamma}_t|$.
<!-- end definition-3 -->
<!-- definition-4: body_md -->
\label{test_plan}
    Let $\boldsymbol{\pi} \in \mathcal{P}(C([0, 1], X))$. We say that $\boldsymbol{\pi}$ is a \emph{test plan} if there exists a constant $C(\boldsymbol{\pi})>0$ such that 
    \begin{align*}
        (e_t)_\#\boldsymbol{\pi} \leq C(\boldsymbol{\pi})\mathfrak{m}\ \mathrm{for \ all \ } t \in [0,1]
    \end{align*}
    and 
    \begin{align*}
        \int \int_0^1 |\dot{\gamma}_t|^2\,{\rm d} t{\rm d}\boldsymbol{\pi}(\gamma) < \infty. 
    \end{align*}
    We use the convention that if $\gamma$ is not absolutely continuous, then $\int_0^1 |\dot{\gamma}_t|^2\,{\rm d} t =\infty$.

<!-- definition-5: body_md -->
Given { $f: X \to \mathbb{R}$ } a $\mathfrak{m}$-measurable function, then a $\mathfrak{m}$-measurable function $G: X \to [0, \infty]$ is called a weak upper gradient of $f$, if 
   \begin{align}\label{weakgradient}
       \int|f(\gamma_1)-f(\gamma_0)|\,{\rm d}\boldsymbol{\pi}(\gamma)\,  \leq \int\int_0^1G(\gamma_t)|\dot{\gamma}_t|\, {\rm d} t{\rm d}\boldsymbol{\pi}(\gamma) < \infty, \quad \mathrm{for \ all \ test \ plans } \ \boldsymbol{\pi}. 
   \end{align}

<!-- definition-6: body_md -->
The Cheeger energy is defined in the class of $\mathfrak{m}$-measurable functions by
    \begin{align*}
        \mathrm{Ch}(f):= \left\{ \begin{array}{ll}
           \frac{1}{2}\int_X |Df|^2_w\, {\rm d}\mathfrak{m}  & \mathrm{if}\ f\ \mathrm{has\ a\ weak\ upper\ gradient\ in }\ L^2(X, \mathfrak{m}),  \\
           \infty  & \mathrm{otherwise},
        \end{array}\right.
    \end{align*}
    with proper domain $D(\mathrm{Ch})= \{f:X \to [0, \infty], \mathfrak{m}\mathrm{-measurable}, \ \mathrm{Ch}(f) < \infty\}$.

<!-- definition-7: body_md -->
A metric measure space is called \emph{infinitesimally Hilbertian} if  $\mathrm{Ch}$ is a quadratic form.

<!-- assumption-1: body_md -->

The manifold $(M,g)$ satisfies one of the following:
    \begin{itemize}
    \item[(i)] The manifold $M$ has finite $\mathrm{vol}_g$-volume. 
    \item[(ii)] We can cover $M$ with coordinate charts such that the smallest eigenvalue of $g$ is uniformly bounded below in these charts and $M$ is complete or the interior of a complete manifold (where the completeness is with respect to the distance induced by charts).
\end{itemize}

<!-- proposition-2: body_md -->
\label{metric_speed_depends_on_direction}
 Let $M$ be a smooth manifold and $g$ a Riemannian metric such that $g, g^{-1} \in L^\infty_{\rm loc}$. 
     Let $\gamma:[0,1] \to M$ be a Lipschitz continuous curve and $t\in [0,1]$ such that the metric speed $|\dot{\gamma}_t|$ exists at $t$ and $\gamma$ is differentiable at $t$. Let $c:[0,1] \to M$ be another curve such that $c_t = \gamma_t$, $c$ is differentiable at $t$ and $\dot{c}_t = \dot{\gamma}_t$. Then the metric speed of $|\dot{c}_t|$ of $c$ at $t$ exists and $|\dot{c}_t| = |\dot{\gamma}_t|$.

<!-- lemma-5: body_md -->
\label{metric_speed_exists_ae}
Let $d \geq 2$. 
    Let $U \subset M$ such that $U$ lies in one coordinate patch and let $v \in \mathbb{R}^d$. Then for almost every $x \in U$ the metric speed of the curve $\gamma^x: t \mapsto x+tv$ exists at $t=0$.
<!-- end lemma-5 -->
<!-- proof-4: body_md -->
We may assume that $U \subset \mathbb{R}^d$.
    Denote $\vartheta:= |v|_{euc}$.
    Fix an $y\in U$ and find $r>0$ such that $B^{euc}_{4r\vartheta}(y) \subset U$.
    For $x \in B_{r\vartheta}(y)$, reparametrise the curve $\gamma^x:[0,1]\to U, t \mapsto x+r(\frac{1}{2}-t)v$. 
    Define the test plan $\boldsymbol{\pi}$ via
    \begin{align*}
        {\rm d} \boldsymbol{\pi}(\gamma)= \left\{ \begin{array}{ll}
            \frac{1}{\mathcal{L}^{d+1}(B_{r\vartheta}(y))}{\rm d} \mathcal{L}^{d}(x)  & \mathrm{if}\ \gamma=\gamma^x \ \mathrm{for\ some\ } x \in B_{r\vartheta}(y), \\
            0 & \mathrm{otherwise.}
        \end{array} \right.
    \end{align*}
    Denote by $N$ the set of points $x$ for which the metric speed of $\gamma^x$ at $t=0$ does not exist. Now, as $\gamma^x$ is absolutely continuous, we have that $\gamma^x \perp N$ for every $x \in B_{r\vartheta}(y)$. Note that there exists a $c>0$ such that for each $t\in [\frac{1}{4},\frac{3}{4}]$, it holds 
    \begin{align*}
        (e_t)_{\#}\boldsymbol{\pi} \geq c \mathcal{L}^d{\mathchoice
{\,\hbox{\vrule width.2pt \vbox to7pt{\vfill \hrule
width7pt height.2pt}}\,}{\,\hbox{\vrule width.2pt \vbox to7pt{\vfill \hrule
width7pt height.2pt}}\,}{\,\hbox{\vrule width.2pt \vbox to5pt{\vfill \hrule
width5pt height.2pt}}\,}{\,\hbox{\vrule width.2pt \vbox to3.35pt{\vfill \hrule
width3.35pt height.2pt}}\,}} {B_{r\vartheta/8}(y)}.
    \end{align*}
    Now Fubini's theorem yields that
    \begin{align*}
        \frac{c}{2} \mathcal{L}^d (N \cap B_{r\vartheta/8}(y)) \leq  \int_0^1 \int_M \mathbbm{1}_{N}(y)\,{\rm d}(e_t)_\#\boldsymbol{\pi}(y){\rm d} t =  \int_{C([0,1], M)} \int_0^1 \mathbbm{1}_{N}(\gamma_t)\,{\rm d} t{\rm d}\boldsymbol{\pi}(\gamma)=0.
    \end{align*}
    This can be done for any $y \in U$ and hence proves the lemma.

<!-- lemma-6: body_md -->
\label{lemma_directional_metric_speed_measurable}
Let $d \geq 2$.
    Let $U \subset M$ such that $U$ lies in one coordinate patch and let $v \in \mathbb{R}^d$. Define the curve $\gamma^x: t \mapsto x+tv$. Then the $\mathcal{L}^2$-a.e. defined map $x \mapsto |\dot{\gamma}^x_0|$ is Lebesgue measurable.
<!-- end lemma-6 -->
<!-- proof-5: body_md -->
We may assume that $U \subset \mathbb{R}^d$.
    Denote $\gamma^x:[-\delta,\delta] \to U, t \mapsto x+tv$. By Lemma \ref{metric_speed_exists_ae}, we get that the metric speed of the curve $\gamma^x$ exists at $0$ for almost every $x$, and it is given by 
    \begin{align*}
        |\dot{\gamma}^x_0| = \lim_{h \to 0} \frac{\mathsf d_g(x, x+hv)}{|h|}.
    \end{align*}
    Now for $n\in \mathbb{N}$ define the function 
    \begin{align*}
        s(x, n):= n\mathsf d_g\Big(x, x+\cfrac{v}{n}\Big).  
    \end{align*}
    As $\mathsf d_g$ is continuous on $U^2$, we get that for all $n$, it holds that $s(\cdot, n)$ is a measurable function. Hence, $\limsup_{n \to \infty} s(\cdot, n)$ is measurable. Noting that $\limsup_{n \to \infty} s(x, n) = |\dot{\gamma}^x_0|$ for almost every $x$, we get that $x \mapsto |\dot{\gamma}^x_0|$ is Lebesgue measurable.

<!-- proposition-4: body_md -->
\label{2d_metric_speed_sobolev}
    Let $d=2$ and $p \geq 1$. Let $M$ be a $d$-dimensional manifold and $g$ a Riemannian metric on $M$ such that $g, g^{-1} \in L^\infty_{\rm loc} \cap W^{1,p}_{\rm loc}$ and suppose that Assumption \ref{assumption_on_mfd} is satisfied. Then the following holds for almost every $x \in M$. Given a curve $\gamma\colon[0,1] \to M$ with $\gamma_t =x$ for some $t\in (0,1)$ and such that both the metric speed $|\dot{\gamma}_t|$ of $\gamma$ at time $t$ and the derivative of $\dot{\gamma}_t \in T_xM$ at time $t$ exist, then $|\dot{\gamma}_t| = |\dot{\gamma}_t|_{g(x)}$.
<!-- end proposition-4 -->
<!-- proof-6: body_md -->
Proving the proposition for $p=1$ is sufficient as $W^{1,p}_{\rm loc} \subset W^{1,1}_{\rm loc}$ for all $p \geq 1$. Hence, we may assume that $p =1$.
We argue by contradiction and suppose the statement is false. \\
\textbf{Step 1.} Reduction to one coordinate patch and one direction. \\
    Let $(U_i, \psi_i)_{i \in \mathbb{N}}$ be a countable atlas of relatively compact coordinate patches of $M$. Let furthermore $(v_j)_{j\in \mathbb{N}}$ be a dense set in $\mathbb{R}^{2}$. Fixing one $i \in \mathbb{N}$, we may assume that $U_i \subset \mathbb{R}^2$. 
    By Lemma \ref{metric_speed_exists_ae}, we have that for almost every $x \in U_i$, the metric speed of the curve $t \mapsto x+tv_j$ exists at $0$ for all $j \in \mathbb{N}$. We denote the complement of that set by $N_i$ and note that $N_i$ is a null set.
    Fix an $x \in U_i \setminus N_i$ such that there exists a curve $\gamma:[0,1] \to M$ with $\gamma_t =x$ for some $t$ and such that the metric speed $|\dot{\gamma}_t|$ of $\gamma$ at time $t$ and the derivative of $\dot{\gamma}_t \in T_xM$ at time $t$ exist, but $|\dot{\gamma}_t| \neq |\dot{\gamma}_t|_{g(x)}$. Denote $v:= \dot{\gamma}_t \in T_xM$. Now by Proposition \ref{metric_speed_continuous_from_direction} and Proposition \ref{metric_speed_depends_on_direction}, we get that there exists a $j \in \mathbb{N}$ such that $\gamma^{x,j}: t \mapsto x+tv_j$ satisfies $|\dot{\gamma}^{x,j}_0| \neq |v_j|_{g(x)}$.
    Define the set 
    \begin{align*}
        S_{i,j, n}:= \Big\{x \in U_i \setminus N_i: \ ||\dot{\gamma}^{x,j}_0| - |v_j|_{g(x)}| \geq \frac{1}{n} \Big\}. 
    \end{align*}
    Note that by Lemma \ref{lemma_directional_metric_speed_measurable}, $S_{i,j,n}$ is measurable.
    Now if the statement of the proposition was not true, that translates into 
    \begin{align*}
       \mathrm{vol}_g\Big( \bigcup_{i,j,n \in \mathbb{N}} S_{i,j,n}\Big) >0.
    \end{align*}
    It then follows that there must exist some $i, j, n \in \mathbb{N}$ such that $\mathrm{vol}_g (S_{i,j,n})>0$. Using that each $U_i$ is relatively compact, and hence $\mathcal{L}^2$ and $\mathrm{vol}_g$ are equivalent on $U_i$, it follows that: \\

    There exists an open, relatively compact set $U \subset \mathbb{R}^2$, a vector $v \in \mathbb{R}^2$ and $\zeta > 0$, $\delta >0$, and a Lebesgue measurable set $S \subset U$ such that
    \begin{align}
        \mathcal{L}^2(S)>\delta,
    \end{align}
     and such that for each $x \in S$, the metric speed $|\dot{\gamma}^{x,v}_0|$ of the curve $\gamma^{x,v}: t \mapsto x+tv$ at $0$ exists and satisfies
    \begin{align}\label{set_zeta}
        |\dot{\gamma}^{x,v}_0| \leq |v|_{g(x)}-\zeta.
    \end{align}
    We may furthermore assume that $|v|_{euc} =1$.
    This concludes the first step. \\

    \textbf{Step 2.} The existence of faulty lines and their properties. \\
     We now define the $\mathrm{Cap}_1$/$\mathcal{L}^d$-null set $N$ as the set of non-Lebesgue points of $g$. Now for each $x \in S \setminus N$, there must exist a $r_x>0$ such that for $r \in (0,r_x)$ it holds $\mathsf d_g(x, x+rv) < |v|_{g(x)}-\zeta$.
     Recall that $\mathsf d_g = \lim_{\varsigma \to 0} \mathsf d_{g_\varsigma}$, where $g_\varsigma := \rho_\varsigma * g$. Now, it follows that for each $x \in S \setminus N$ and $r \in (0, r_x)$, there exists a $\varsigma_{x,r}>0$ such that for all $\varsigma \in (0, \varsigma_{x,r})$, it holds
    \begin{align}\label{distance_assumption_step2_2d}
        |g(x)-g_\varsigma(x)|_{op} \leq \frac{\zeta}{4} \ \mathrm{and}\ \mathsf d_{g_\varsigma}(x, x+rv) < r\Big(|v|_{g(x)}- \frac{\zeta}{2}\Big).
    \end{align}
    As $g_\varsigma$ is smooth, the Arzela-Ascoli theorem yields that there exists length minimising Lipschitz continuous curve $\gamma^{r, \varsigma}:[0,r] \to U$ such that
    \begin{align}\label{shorter_curves2d}
        \gamma^{r, \varsigma}_0 =x, \ \gamma^{r, \varsigma}_r = x+rv, \ \int_0^r |\dot{\gamma}^{r, \varsigma}_t|_{g_\varsigma(\gamma^{r, \varsigma}_t)}\,  {\rm d} t = \mathsf d_{g_\varsigma}(x, x+rv) < r\Big(|v|_{g(x)}- \frac{\zeta}{2}\Big).
    \end{align}
    Let $0<\lambda'<1< \Lambda'$ such that on $U$ it holds $\lambda' \mathrm{Id}_2 \leq g \leq \Lambda' \mathrm{Id}_2$ and such that $\mathsf d_g$ and $\mathsf d_{euc}$ are $(\lambda', \Lambda')$-equivalent in $U$. As $U$ is relatively compact, we can shrink $\varsigma_{x, r}>0$ and find $0< \lambda' < 2\lambda <1<\Lambda<2\Lambda'$ such that for all $\varsigma \in (0, \varsigma_{x, r})$ it holds that $\lambda \mathrm{Id}_2 \leq g_\varsigma \leq \Lambda \mathrm{Id}_2$ and such that $\mathsf d_{g_\varsigma}$ and $\mathsf d_{euc}$ are $(\lambda, \Lambda)$-equivalent in $U$.
    
    Fix a $\varsigma \in (0, \varsigma_{x, r})$ and write from now on $\gamma^r:= \gamma^{r, \varsigma}$.
    It follows that 
    \begin{align}
         \int_0^r |\dot{\gamma}^r_t|_{euc} {\rm d} t \leq \Lambda \int_0^r |\dot{\gamma}^r_t|_{g_\varsigma(\gamma^r_t)}\,  {\rm d} t \leq 2 \Lambda^2r,
    \end{align}
    hence we may reparametrise $\gamma^r$ to be of constant Euclidean speed and get that the Lipschitz constant is bounded from above by $2\Lambda^2$. As $\mathsf d_{g_\varsigma}(x, x+rv) \geq \frac{\lambda^2}{2}r$, we get that the Euclidean speed is bounded from below by $\lambda^2/2$.
    Now note that for each $s, s' \in [0,r]$, $s<s'$ it holds
    \begin{align}\label{minimal_length_of_curve_segment_2d}
        \int_s^{s'} |\dot{\gamma}^r_t|_{g_\varsigma} {\rm d} t \geq \lambda \int_s^{s'} |\dot{\gamma}^r_t|_{euc} {\rm d} t \geq \frac{\lambda^3}{2} |s-s'|.
    \end{align} 
   \textbf{Claim.} For each $s, s' \in [0,r]$, $s<s'$ it holds 
    \begin{align}
        \frac{\lambda^3}{4\Lambda^2} |s-s'| \leq \mathsf d_{euc}(\gamma_s, \gamma_{s'}).
    \end{align}
    \textit{Proof of the claim.}
    Suppose this was not the case for some $s, s' \in [0,r]$. Then we could replace the segment between $\gamma_s$ and $\gamma_{s'}$ by a straight line segment between $\gamma_s$ and $\gamma_{s'}$. That would give a curve $\Tilde{\gamma}:[s, s'] \to U$ such that $\Tilde{\gamma}_s = \gamma_s$ and $\Tilde{\gamma}_{s'} = \gamma_{s'}$ with 
    \begin{align*}
        \int_s^{s'} |\dot{\Tilde{\gamma}}_t|_{euc} {\rm d} t \leq \frac{\lambda^3}{4\Lambda^2} |s-s'|
    \end{align*}
    which using \eqref{minimal_length_of_curve_segment_2d} yields
    \begin{align}
        \int_s^{s'} |\dot{\Tilde{\gamma}}_t|_{g_\varsigma} {\rm d} t \leq \frac{\lambda^3}{4\Lambda} |s-s'| < \int_s^{s'} |\dot{\gamma}^r_t|_{g_\varsigma} {\rm d} t,
    \end{align}
    so replacing $\gamma^r$ with $\Tilde{\gamma}$ on $[s, s']$ yields a shorter curve.
    This contradicts the length optimality of $\gamma^r$ with respect to $g_\varsigma$ and hence proves the claim.  
    
    This yields that we can assume that there exists a constant $C>1$ that only depends on $U$ and $g$ such that for each $0 \leq s< s' \leq 1$, it holds
    \begin{align}\label{bi-Lipschitz_curves2d}
        \frac{1}{C}|s-s'|\leq \mathsf d_{euc}(\gamma^r_s, \gamma^r_{s'})\leq C |s-s'|.
    \end{align}
    Observe that by the properties of inner products on $\mathbb{R}^2$, it holds
    \begin{align*}
        |rv|_{g(x)} =\Big|\int_0^r \dot{\gamma}^r_t\, {\rm d} t\Big|_{g(x)} \leq \int_0^r |\dot{\gamma}^r_t|_{g_\varsigma(x)}\, {\rm d} t .
    \end{align*}
    Note that in the last integral, we evaluate every vector with respect to $g_\varsigma(x) = g_\varsigma(\gamma^r_0)$ instead of $g_\varsigma(\gamma^r_t)$. 
    By \eqref{distance_assumption_step2_2d}, it follows that 
    \begin{align*}
        \int_0^r |\dot{\gamma}^r_t|_{g(x)} {\rm d} t \geq |rv|_{g(x)} \geq \mathsf d_{g_\varsigma}(x, x+rv) + \frac{\zeta r}{2} = \int_0^r |\dot{\gamma}^r_t|_{g_\varsigma(\gamma^r_t)} {\rm d} t.
    \end{align*}
    Hence, 
    \begin{align}\label{integral_speed_comparison2d}
        \int_0^r |\dot{\gamma}^r_t|_{g(x)} -|\dot{\gamma}^r_t|_{g_\varsigma(\gamma^r_t)} {\rm d} t \geq \frac{\zeta r}{2}.
    \end{align}
    Note that \eqref{bi-Lipschitz_curves2d} implies that for all $t \in [0,r]$, it holds $|\dot{\gamma}^r_t|_{g(x)} -|\dot{\gamma}^r_t|_{g_\varsigma(\gamma^r_t)} \leq C(U, g)$ and hence, combining this with \eqref{integral_speed_comparison2d} yields
    \begin{align}
        \mathcal{L}^1\Big(\Big\{t: \ |\dot{\gamma}^r_t|_{g(x)} -|\dot{\gamma}^r_t|_{g_\varsigma(\gamma^r_t)} \geq \frac{\zeta}{2} \Big\}\Big) \geq \frac{\zeta}{2C(U, g)}.
    \end{align}
    Using \eqref{bi-Lipschitz_curves2d} once more, gives that there exists a constant $C=C(g, U)$ and a set $A^{x, r, \varsigma} \subset [0,r]$ such that 
    \begin{align}\label{discontinuity_of_g_along_line2d}
        &\mathcal{L}^1(A^{x, r, \varsigma}) \geq \frac{r\zeta}{2C},  \nonumber \\
        &\forall t \in A:\ \norm{g(x)-g_\varsigma(\gamma^{r, \varsigma}_t)}_{euc} \geq \frac{\zeta}{2C},
    \end{align}
    where in the last inequality we evaluate the Euclidean matrix norm. This finishes the second step.\\

    \textbf{Step 3.} The faulty lines have big $1$-capacity. \\
    Let us fix some constants for this step. We fix $C=C(U, g)>1$ for \eqref{bi-Lipschitz_curves2d} and \eqref{discontinuity_of_g_along_line2d} to hold. We fix $\zeta \in (0,1)$ for \eqref{set_zeta} (and hence \eqref{discontinuity_of_g_along_line2d}) to be true. Finally, $C_2$ shall denote the constant from the Besicovitch covering theorem in $\mathbb{R}^2$ and $c_{1,2}$ denotes the constant such that for each $\phi \in W^{1,1}(\mathbb{R}^2)$, it holds $c_{1,2}\norm{\phi}_{L^2} \leq \norm{D\phi}_{L^1}$.  
    Define
    \begin{align}\label{set_eps}
        \epsilon:= \frac{\zeta}{400\pi C_2C^5}\cdot \frac{\delta}{2} >0,
    \end{align}
    By Theorem \ref{quasicontinuity} and Theorem \ref{basic_properties_capa} (v), we can find an open set $\Omega \subset U$ such that 
    \begin{align}\label{Capacity_omega}
        \mathrm{Cap}_1(\Omega) \leq \min\Big(1, \frac{c_{1,2}}{\sqrt{\mathcal{L}^2(U)}}\Big) \cdot \frac{\epsilon}{40}, \ \mathcal{L}^2(\Omega) \leq \frac{\delta}{2}
    \end{align}
    such that $g$ is continuous when restricted to $U \setminus \Omega$ and by the discussion after Theorem \ref{quasicontinuity} we may assume that (a subsequence of) $g_\varsigma$ converges uniformly on $U \setminus \Omega$. Note that then 
    \begin{align}\label{remaining_S}
        \mathcal{L}^2(S \setminus (\Omega\cup N)) \geq \frac{\delta}{2}.
    \end{align}
    Let $f \in W^{1,1}(\mathbb{R}^2, [0,1])$ such that $f \equiv 1$ on $\Omega$ and 
    \begin{align}\label{function_with_low_capacity}
        \int_{\mathbb{R}^2} |Df|\, {\rm d} x \leq 2 \mathrm{Cap}_1(\Omega).
    \end{align}
    Now, for every $x \in S \setminus (\Omega\cup N)$, there exists $\varepsilon_x >0$ such that for all $y \in B_{\varepsilon_x}(x) \setminus \Omega$, it holds 
    \begin{align}\label{g_continuous_away_omega2d}
        \norm{g(x)-g(y)}_{euc} \leq \frac{\zeta}{8C}.
    \end{align}
    Now for each $x \in S \setminus (\Omega\cup N)$ pick an $\rho_x \in (0, \min(r_x, \varepsilon_x))$, where $r_x$ is chosen as in the beginning of Step 2 (see the discussion leading up to \eqref{distance_assumption_step2_2d}). Then 
    \begin{align*}
       S \setminus (\Omega\cup N) \subset \bigcup_{x \in S \setminus (\Omega\cup N)} B_{\rho_x}(x). 
    \end{align*}
    The Besicovitch covering theorem states that there exists $C_2$ countable families $T_1, \ldots, T_{C_2} \subset S \setminus (\Omega\cup N)$ such that for each $j \neq j'$, it holds $T_j \cap T_{j'} = \emptyset$ and such that for each $x \neq x' \in T_j$, it holds
    \begin{align*}
        B_{\rho_{x}}(x) \cap B_{\rho_{x'}}(x') = \emptyset.
    \end{align*}
    Then, by the pigeonhole principle, there exists a $j \in \{1, \ldots, C_2\}$ such that 
    \begin{align}\label{disjoint_balls_big_measure}
        \mathcal{L}^2\Big(\bigcup_{x \in T_j} B_{\rho_x}(x)\Big) = \sum_{x \in T_j} \mathcal{L}^2( B_{\rho_x}(x)) \geq \frac{1}{C_2} \mathcal{L}^2(S \setminus (\Omega\cup N)) \geq \frac{\delta}{2C_2}.
    \end{align}
    Write $T_j = \{x_i\}_{i\in \mathbb{N}}$. Now fix an $i \in \mathbb{N}$.  
    Denote $x:= x_i \in T_j$ and write $\rho:= \rho_x$.
    Now, by the second step, more precisely \eqref{discontinuity_of_g_along_line2d}, there exists a $\varsigma_{x, \rho}>0$ such that for each $\varsigma \in (0, \varsigma_{x, \rho})$ there exists a $(\frac{1}{C}, C)$-bi-Lipschitz curve $\gamma = \gamma^{\rho/5C^2}:[0,\rho/5C^2] \to B_{\rho/5C}(x)$ and a set $A \subset [0,\rho/5C^2]$ such that $\mathcal{L}^1(A) \geq \frac{\rho\zeta}{20C^3}$ and for all $y \in \gamma(A)$, it holds
    \begin{align*}
        \norm{g(x)-g_\varsigma(y)}_{euc} \geq \frac{\zeta}{2C}.
    \end{align*}
    Using the locally uniform convergence of $g_\varsigma$ to $g$ on $U\setminus \Omega$, we may choose $\varsigma \in (0, \varsigma_{x, \rho})$ small enough such that additionally 
    \begin{align*}
        \norm{g(y)-g_\varsigma(y)}_{euc} \geq \frac{\zeta}{8C}, \ \mathrm{for\ all\ } y \in U \setminus\Omega,
    \end{align*}
    implying 
     \begin{align*}
        \norm{g(x)-g(y)}_{euc} \geq \frac{\zeta}{4C}, \ \mathrm{for\ all\ } y \in \gamma(A).
    \end{align*}
    Recalling \eqref{g_continuous_away_omega2d}, this implies that $\gamma(A) \subset \Omega \cap B_\rho(x)$. In particular, $O_\gamma:=\gamma^{-1}(\Omega) \subset [0,\rho]$ is an open set with $\mathcal{L}^1(O_\gamma) \geq \frac{\rho\zeta}{20C^3}$. Now, we can find a compact set $K \subset [0,1]$ that is the finite union of compact intervals $I_1,\ldots, I_n$ such that $K \subset O_\lambda$ and $\mathcal{L}^1(K) \geq \frac{\rho\zeta}{40C^3}$.

    Using mollification, we can find a function $\Tilde{f} \in C_c^\infty(\mathbb{R}^2)$ such that $\Tilde{f} \equiv 1$ on $\gamma(K)$ and 
    \begin{align}\label{replace_by_smooth_function}
        \norm{f-\Tilde{f}}_{W^{1,1}(B_{\rho}(x))} \leq 2^{-i-2}\epsilon.
    \end{align}
    By translating, we may assume that $x =0$. Now consider $F(x) := \Tilde{f}(\rho x) \in C_c^\infty(\mathbb{R}^2)$. We immediately get that 
    \begin{align}\label{rescale_smooth_function}
        \norm{F}_{W^{1,1}(B_1(0))} \leq \frac{1}{\rho^2} \norm{\Tilde{f}}_{W^{1,1}(B_\rho(0))}.
    \end{align}
    Moreover define the curve $\boldsymbol{\gamma}:[0, \frac{1}{5C^2}] \to B_{1/5C}(0)$ by $\boldsymbol{\gamma}(t) = \frac{1}{\rho}\gamma(\rho t)$. Then $\boldsymbol{\gamma}$ is $(\frac{1}{C}, C)$-bi-Lipschitz continuous.
    By construction, we have that $F$ is a $C_c^\infty$-function that satisfies $F(\boldsymbol{\gamma}(\frac{1}{\rho}K)) = \{1\}$. Using that $\mathcal{L}^1(\frac{1}{\rho}K) \geq \frac{\zeta}{40C^3}$, we may apply Lemma \ref{2dcapacity_of_line_segments}, to get that 
    \begin{align}
        \int_{B_{1}(0)} |F| + |DF| {\rm d} y \geq \frac{\zeta}{80C^5}.
    \end{align}
    Combining this with \eqref{rescale_smooth_function} and \eqref{replace_by_smooth_function}, we get that 
    \begin{align}\label{integral_estimate_on_specific_ball}
        \int_{B_{\rho}(0)} |f| + |Df| {\rm d} x \geq \frac{\rho^2\zeta}{80C^5} -2^{-i-2}\epsilon
    \end{align}
    Using that $\rho^2 = \mathcal{L}^2(B_\rho(0))/\pi$ together with \eqref{disjoint_balls_big_measure} and \eqref{integral_estimate_on_specific_ball} we get that 
    \begin{align}\label{lower_bound_w11}
        \int_{\mathbb{R}^2} |f| + |Df| {\rm d} y \geq \sum_{x \in T_j} \int_{B_{\rho_x}(x)} |f| + |Df| {\rm d} y \geq  \frac{\zeta}{80\pi C_2C^5}\mathcal{L}^2(S \setminus (\Omega\cup N)) - \epsilon.
    \end{align}
    Now note that $2$ is the Sobolev conjugate of $p=1$ in $\mathbb{R}^2$. 
    We now distinguish two cases: \\
    \textbf{Case 1.} 
    \begin{align}\label{condition_case_1}
         \int_{\mathbb{R}^2} |f| {\rm d} y \geq \frac{\zeta}{160\pi C_2C^5}\mathcal{L}^2(S \setminus (\Omega\cup N)) - \epsilon.
    \end{align}
    Then, by the H\"older inequality, we get that 
    \begin{align}
        \norm{f}_{L^2(\mathbb{R}^2)} \geq \norm{f}_{L^2(U)} \geq  \frac{1}{\sqrt{\mathcal{L}^2(U)}} \Big(\frac{\zeta}{160\pi C_2C^5}\mathcal{L}^2(S \setminus (\Omega\cup N)) - \epsilon \Big).
    \end{align}
    Then, by the Sobolev embedding theorem, we get that there exists a constant $c_{1,2}>0$ such that 
    \begin{align}\label{conclusion_case_1}
        \int_{\mathbb{R}^2} |Df| {\rm d} y \geq \frac{c_{1,2}}{\sqrt{\mathcal{L}^2(U)}} \Big(\frac{\zeta}{160\pi C_2C^5}\mathcal{L}^2(S \setminus (\Omega\cup N)) - \epsilon \Big).
    \end{align}
  \textbf{Case 2.}  
      \begin{align}\label{condition_case_2}
         \int_{\mathbb{R}^2} |f| {\rm d} y < \frac{\zeta}{160\pi C_2C^5}\mathcal{L}^2(S \setminus (\Omega\cup N)) - \epsilon.
    \end{align}
    Then \eqref{lower_bound_w11} implies that
    \begin{align}\label{conclusion_case2}
         \int_{\mathbb{R}^2} |Df| {\rm d} y \geq \frac{\zeta}{160\pi C_2C^5}\mathcal{L}^2(S \setminus (\Omega\cup N)) - \epsilon.
    \end{align}
    As one of case 1 and case 2 must be true, we get that 
    \begin{align}\label{final_bound_norm_df}
        \int_{\mathbb{R}^2} |Df| {\rm d} y \geq  \min\Big(1, \frac{c_{1,2}}{\sqrt{\mathcal{L}^2(U)}}\Big)\cdot\Big(\frac{\zeta}{160\pi C_2C^5}\mathcal{L}^2(S \setminus (\Omega\cup N)) - \epsilon\Big).
    \end{align}
    Now this contradicts \eqref{set_eps}, \eqref{Capacity_omega}, and \eqref{remaining_S} and hence proves the proposition.

We will not provide the full proof, as with the above ingredients, it follows the strategy of \cite{mondino2025equivalence}. The main steps are the following:
   \begin{itemize}
       \item For $f \in C^1$, we have that $|Df|_w = |\nabla_g f|_g$ $\mathrm{vol}_g$-almost everywhere. This follows exactly as in \cite[Proposition 4.24]{mondino2025equivalence}, using Proposition \ref{2d_metric_speed_sobolev} and localising the test plan around a Lebesgue point of $g$.  
       \item Following the argument in \cite[Proposition 4.25 and Lemma 4.26]{mondino2025equivalence}, we can infer that $f\in W^{1,2}_w(M)$ if and only if $f\in W^{1,2}_{\rm loc}(M)$ and $|\nabla_gf|_g \in L^2(M)$. In particular, we again get that for $f\in W^{1,2}_{\rm loc}(M)$, it holds $|Df|_w = |\nabla_g f|_g$ $\mathrm{vol}_g$-almost everywhere.
       \item Infinitesimal Hilbertianity then follows as in \cite[Corollary 4.27]{mondino2025equivalence}, from the linearity of differentiation and the fact that $g$ is an inner product on the tangent space. 
   \end{itemize}
