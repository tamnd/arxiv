<!-- theorem-1: body_md -->
\label{thm:main}
Let $0<s<1$, and let $\mu_1$ and $\mu_2$ be $s$-Frostman probability measures on $[0,1]$.
If
\begin{equation}\label{eq:profile-separation}
  \limsup_{t\to\infty}\abs{H_{\mu_1}^s(t)-H_{\mu_2}^s(t)}>0,
\end{equation}
then
\[
  \operatorname{Law}(W_*\mu_1)\perp\operatorname{Law}(W_*\mu_2)
  \quad\text{on }\mathcal{P}(\mathbb{R}^2).
\]

<!-- theorem-2: body_md -->
\label{thm:profile-asymptotics}
Let $\mu$ be the natural self-similar probability measure of an IFS satisfying the open set
condition, with similarity dimension $s\in(0,1)$.
If the IFS is of lattice type with span $\ell$, then there is a continuous, strictly positive,
non-constant $\ell/2$-periodic function $\widetilde H_\mu^s$ such that
\[
  H_\mu^s(t)-\widetilde H_\mu^s(t)\to0
  \qquad\text{as }t\to\infty.
\]
If the IFS is of non-lattice type, then
\[
  H_\mu^s(t)\to C_\mu\in(0,\infty)
  \qquad\text{as }t\to\infty.
\]

<!-- theorem-3: body_md -->
\label{thm:fixed-dimension-separation}
Let $s\in(0,1)$. For every $p_1\in(0,1/2]$, the set of $p_2\in(0,1/2]$ with
$\overline H_s(p_2)=\overline H_s(p_1)$ is finite. Consequently, for $p_2\in(0,1/2]$ not in this
exceptional set,
$\overline H_s(p_1)\neq\overline H_s(p_2)$, and
\[
  \operatorname{Law}(W_*\mu_{p_1})\perp\operatorname{Law}(W_*\mu_{p_2}).
\]
The set $D_s$ has no accumulation point in $(0,1/2]$, and the mean is strictly monotone on
every component of $(0,1/2)\setminus D_s$.

  \operatorname{Law}(W_*\mu_{p_1})\perp\operatorname{Law}(W_*\mu_{p_2}).

<!-- theorem-5: body_md -->
\label{thm:minkowski-reconstruction}
Let $\{S_i\}_{i=1}^N$ be a self-similar IFS on $[0,1]$ satisfying the open set condition,
with contraction ratios $r_i$ and attractor $K$. Assume that its
similarity dimension $s$ lies in $(0,1)$, and let $\mu$ be the self-similar probability measure
with weights $r_i^s$. Then
\begin{equation}\label{eq:minkowski-reconstruction}
  \nu_{W(K),e^{-n}}\to W_*\mu
  \quad\text{weakly almost surely as }n\to\infty.
\end{equation}
In particular, $W_*\mu$ is almost surely a Borel function of the compact set $W(K)$.
<!-- end theorem-5 -->

\label{thm:cantor-set-application}
Let $K_A$ and $K_B$ be the attractors of two IFSs on $[0,1]$ satisfying the open set condition,

second is of non-lattice type, then
\[
  \operatorname{Law}(W(K_A))\perp\operatorname{Law}(W(K_B))
  \quad\text{on }\mathcal{K}(\mathbb{R}^2).
\]

<!-- lemma-1: body_md -->
\label{thm:gaussian-reduction}
For every $r>0$,
\begin{equation}\label{eq:gaussian-reduction}
  \begin{split}
    \mathbb{E}\mathcal{C}_r(W_*\mu)
    &=\int_{[0,1]^2}\left(1-\exp\left(-\frac{r^2}{2\abs{u-t}}\right)\right)
      \,\mathrm{d}\mu(u)\,\mathrm{d}\mu(t)\\
    &=\int_0^1\left(1-\exp\left(-\frac{r^2}{2\delta}\right)\right)
      \,\mathrm{d}\mathcal{C}_\delta(\mu).
  \end{split}
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
For $u\neq t$, the increment $W_u-W_t$ is a centred planar Gaussian with covariance
$\abs{u-t}I_2$. Polar integration gives
\[
  \mathbb{P}(\abs{W_u-W_t}<r)=1-\exp\left(-\frac{r^2}{2\abs{u-t}}\right).
\]
Since $\mu$ is atomless, this holds for $\mu\times\mu$-almost every $(u,t)$.
Integrating with respect to $\mu\times\mu$ and applying Fubini's theorem gives the first
identity, while pushing $\mu\times\mu$ forward by $(u,t)\mapsto\abs{u-t}$ gives the second.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{thm:renewal-defect-left-tail}
The function $z(t)$ is continuous and for $t\leqslant0$,
\[
  z(t)=e^{st}\left(1-\sum_i r_i^{2s}\right)>0.
\]
In particular, $z(t)$ decays exponentially as $t\to-\infty$.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Continuity follows from $z(t)=G(t)-\sum_i p_iG(t-\alpha_i)$, since $G$ is continuous,
and non-negativity follows from the definition of the cross term. For $t\leqslant0$, every
pair lies within distance $e^{-t}\geqslant1$, so
\[
  \mathcal{C}^\times_{e^{-t}}(\mu)=\sum_{i\neq j}p_ip_j=1-\sum_i p_i^2.
\]
The claim follows because $p_i=r_i^s\in(0,1)$ and $\sum_i p_i=1$.
<!-- end proof-2 -->
<!-- lemma-3: body_md -->
\label{thm:cross-piece-mass}
There exist $C>0$ and $\eta\in(0,1-s)$ such that
\begin{equation}\label{eq:cross-piece-mass}
  0\leqslant\mathcal{C}^\times_\delta(\mu)\leqslant C\delta^{s+\eta},\qquad0<\delta\leqslant1.
\end{equation}
Consequently, $0\leqslant z(t)\leqslant Ce^{-\eta t}$ for all $t\geqslant0$.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
By Ahlfors regularity, for each $x\in K$ the $\mu$-mass of points $y$ satisfying
  $\abs{S_i(x)-S_j(y)}\leqslant\delta$ is at most $C\delta^s$, and it vanishes unless
  $\operatorname{dist}(S_i(x),S_jK)\leqslant\delta$. Fubini's theorem gives
  \[
    \mathcal{C}^\times_\delta(\mu)\leqslant C\delta^s\sum_{i\neq j}p_ip_j\,
    \mu\set{x\in K\colon\operatorname{dist}(S_i(x),S_jK)\leqslant\delta},
  \]
  so it suffices to prove, for some $\eta\in(0,1-s)$,
  \begin{equation}\label{eq:boundary-mass}
    \mu\set{x\in K\colon\operatorname{dist}(S_i(x),S_jK)\leqslant\delta}
    \leqslant C\delta^\eta,\qquad0<\delta\leqslant1,\quad i\neq j.
  \end{equation}
  By Schief~\cite{Schief1994} the IFS satisfies the strong open set condition, so
  Lalley's estimate~\cite[Proposition~3]{Lalley1988} applies: there is $\gamma>0$
  such that every $\delta$-separated subset of
  $A_{ij}(\delta)=\set{y\in S_iK\colon\operatorname{dist}(y,S_jK)\leqslant\delta}$ has at most
  $C\delta^{\gamma-s}$ points. The closed balls of radius $\delta$ about a maximal
  $\delta$-separated subset cover $A_{ij}(\delta)$, and each has $\mu$-mass at most
  $C\delta^s$ by Ahlfors regularity, so $\mu(A_{ij}(\delta))\leqslant C\delta^\gamma$.
  Since $\mu(S_iA)\geqslant p_i\mu(A)$ for every Borel set $A$ by self-similarity,
  \cref{eq:boundary-mass} follows with $\eta=\min\set{\gamma,(1-s)/2}$, and with it
  \cref{eq:cross-piece-mass}.
  Taking $\delta=e^{-t}$ and multiplying by $e^{st}$ gives the bound on $z$.
<!-- end proof-3 -->
<!-- proposition-1: body_md -->
\label{thm:renewal-limits}
  If the IFS is of non-lattice type, then
  \begin{equation}\label{eq:g-non-lattice-limit}
    G(t)\to\frac{1}{m}\int_{\mathbb{R}}z(x)\,\mathrm{d} x\in(0,\infty),
    \qquad t\to\infty.
  \end{equation}
  If the IFS is of lattice type with span $\ell$, then $\widetilde G$ is continuous,
  strictly positive and $\ell$-periodic, and
  \[
    G(t)-\widetilde G(t)\to0,\qquad t\to\infty.
  \]
<!-- end proposition-1 -->
<!-- proof-4: body_md -->
By \cref{thm:renewal-defect-left-tail,thm:cross-piece-mass}, the defect $z$ satisfies the
hypotheses above. Since $G$ is bounded and tends to zero at $-\infty$, the renewal theorem
gives \cref{eq:g-non-lattice-limit} and the lattice approximation. Positivity of the integral
and of $\widetilde G$ follows from $z\geqslant0$ and $z(t)>0$ for $t\leqslant0$.
<!-- end proof-4 -->
<!-- lemma-4: body_md -->
\label{thm:lattice-nonconvergence}
If the IFS is of lattice type, then the periodic profile $\widetilde G$ is non-constant.
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
Fix $i\neq j$. The maps $S_{ij}$ and $S_{ji}$ have the contraction ratio, while their
images must be non-trivially translationally separated by the open set condition.
So, there exists $\rho\coloneqq\abs{S_{ij}(0)-S_{ji}(0)}>0$.

For $0<\delta<\rho/(2r_{ij})$, put $\delta_1=\rho-2r_{ij}\delta$ and
$\delta_2=\rho+r_{ij}\delta$. We have
$\delta_1<\abs{S_{ij}(x)-S_{ji}(y)}\leqslant\delta_2$.
Expanding the difference $\mathcal{C}^\times_{\delta_2}(\mu)-\mathcal{C}^\times_{\delta_1}(\mu)$ therefore gives
\begin{align*}
  \mathcal{C}^\times_{\delta_2}(\mu)-\mathcal{C}^\times_{\delta_1}(\mu)
  &\geqslant p_i^2p_j^2(\mu\times\mu)
    \set{(x,y)\colon\delta_1<\abs{S_{ij}(x)-S_{ji}(y)}\leqslant\delta_2}\\*
  &\geqslant p_i^2p_j^2\mathcal{C}_\delta(\mu).
\end{align*}
Hence,
\begin{align*}
  \delta_2^s\widetilde G(-\log\delta_2)-\delta_1^s\widetilde G(-\log\delta_1)
  &=\frac{\ell}{m}\sum_{k\in\mathbb{Z}}e^{sk\ell}
    \bigl(\mathcal{C}^\times_{\delta_2e^{-k\ell}}(\mu)-\mathcal{C}^\times_{\delta_1e^{-k\ell}}(\mu)\bigr)\\*
  &\geqslant\frac{\ell}{m} p_i^2p_j^2\mathcal{C}_\delta(\mu)\geqslant c\delta^s.
\end{align*}
Since $\widetilde G$ is bounded and $\delta_2^s-\delta_1^s=O(\delta)$, this gives
\[
  \delta_2^s\bigl(\widetilde G(-\log\delta_2)-\widetilde G(-\log\delta_1)\bigr)
  \geqslant c\delta^s-C\delta>0
\]
for all sufficiently small $\delta>0$, since $s<1$.
<!-- end proof-5 -->
<!-- lemma-5: body_md -->
\label{thm:smoothing-injective}
The operator $T$ is injective on the continuous $\ell$-periodic functions. Further,
$Tg$ can be constant only when $g$ is constant.
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
Since $g$ is $\ell$-periodic, its Fourier coefficients are
\[
  \widehat g(k)=\frac1\ell\int_0^\ell g(x)e^{-2\pi ikx/\ell}\,\mathrm{d} x,\qquad k\in\mathbb{Z}.
\]
The $\ell$-periodic function $x\mapsto Tg(x/2)=\int_0^\infty\phi(\eta)g(x-\log\eta)\,\mathrm{d}\eta$
is an average of translates of $g$. Translating $g$ by $\log\eta\in\mathbb{R}$ multiplies
$\widehat g(k)$ by $e^{-2\pi ik\log(\eta)/\ell}=\eta^{-2\pi ik/\ell}$. 
Averaging over $\eta$, Fubini's theorem gives
\begin{equation}\label{eq:fourier-multiplier}
  \widehat{Tg}(k)=\left(\int_0^\infty\phi(\eta)\eta^{-2\pi ik/\ell}\,\mathrm{d}\eta\right)\widehat g(k),
  \qquad k\in\mathbb{Z},
\end{equation}
where $\widehat{Tg}(k)$ denotes the $k$-th Fourier coefficient of $Tg$ with respect to
its period $\ell/2$.

The multiplier, when substituting $x=1/(2\eta)$, becomes
\[
  \int_0^\infty\phi(\eta)\eta^{-2\pi ik/\ell}\,\mathrm{d}\eta
  =2^{-s+2\pi ik/\ell}\int_0^\infty x^{-s+2\pi ik/\ell}e^{-x}\,\mathrm{d} x.
\]
The last integral is $\Gamma(z)=\int_0^\infty x^{z-1}e^{-x}\,\mathrm{d} x$ for
$z=1-s+2\pi ik/\ell$. Thus
\begin{equation}\label{eq:gamma-multiplier}
  \int_0^\infty\phi(\eta)\eta^{-2\pi ik/\ell}\,\mathrm{d}\eta
  =2^{-s+2\pi ik/\ell}\Gamma(1-s+2\pi ik/\ell),
\end{equation}
which is non-zero.

Now let $g_1$ and $g_2$ be continuous and $\ell$-periodic with $Tg_1=Tg_2$. Linearity
gives $T(g_1-g_2)=0$, so by \cref{eq:fourier-multiplier} and the non-vanishing of the
multipliers, $\widehat{(g_1-g_2)}(k)=0$ for every $k\in\mathbb{Z}$. By the uniqueness theorem
$g_1=g_2$, so $T$ is injective. The conclusion on constant functions follows directly from the
non-vanishing of multipliers.
<!-- end proof-6 -->
<!-- proof-7: body_md -->
In the non-lattice case, \cref{thm:renewal-limits} and dominated convergence in
\cref{eq:h-definition} give
\[
  H_\mu^s(t)\to\frac1m\int_\mathbb{R} z(x)\,\mathrm{d} x
    \int_0^\infty\phi(\eta)\,\mathrm{d}\eta\in(0,\infty).
\]

In the lattice case with span $\ell$, let $\widetilde G$ be the periodic limit from
\cref{thm:renewal-limits} and put $\widetilde H_\mu^s=T\widetilde G$. This function is
continuous, strictly positive, and $\ell/2$-periodic, and it is non-constant by
\cref{thm:lattice-nonconvergence,thm:smoothing-injective}. Since $G$ and $\widetilde G$ are
bounded, dominated convergence gives
\[
  H_\mu^s(t)-\widetilde H_\mu^s(t)
  =\int_0^\infty\phi(\eta)
    \bigl(G(2t-\log\eta)-\widetilde G(2t-\log\eta)\bigr)\,\mathrm{d}\eta
  \to0.\qedhere
\]

<!-- proposition-2: body_md -->
\label{thm:four-point-integral}
There is a constant $C>0$, depending only on $s$ and $A$, such that, for every $0<r\leqslant1$,
\[
  \int_{\set{I\cap I'\neq\varnothing}}
  \mathbb{P}(\abs{Z}<r,\abs{Z'}<r)\,\mathrm{d}\mu^4
  \leqslant C V_s(r).
\]
<!-- end proposition-2 -->
<!-- proof-8: body_md -->
Since $\mu$ is non-atomic, we may ignore tuples with repeated endpoints. For overlapping
intervals, put
\[
  a=|I\cap I'|\quad\text{and}\quad b=|I\mathbin{\triangle}I'|.
\]
The coordinate pairs $(Z_1,Z'_1)$ and $(Z_2,Z'_2)$ are independent centred Gaussian vectors,
each with covariance matrix
\[
  \begin{pmatrix}|I|&\pm a\\\pm a&|I'|\end{pmatrix}.
\]
The determinant of this matrix is
\[
  |I||I'|-a^2=ab+(|I|-a)(|I'|-a)\geqslant ab>0.
\]
Hence the joint density $f$ of $(Z,Z')$ satisfies
\[
  f(z,z')\leqslant\frac{1}{(2\pi)^2(|I||I'|-a^2)}\leqslant\frac{1}{(2\pi)^2ab}.
\]
Hence
\[
  \mathbb{P}(\abs{Z}<r,\abs{Z'}<r)
  =\int_{B(0,r)}\int_{B(0,r)}f(z,z')\,\mathrm{d} z'\,\mathrm{d} z
  \leqslant\frac{(\pi r^2)^2}{(2\pi)^2ab}
  =\frac{r^4}{4ab}.
\]
We may assume, relabelling if necessary, that $|I|\geqslant|I'|$, so $|I|\geqslant(a+b)/2$. Since
\[
  \{\abs{Z}<r,\abs{Z'}<r\}\subseteq\{\abs{Z}<r\},
\]
we obtain
\[
  \mathbb{P}(\abs{Z}<r,\abs{Z'}<r)
  \leqslant\int_{B(0,r)}\frac{e^{-\abs{z}^2/(2|I|)}}{2\pi|I|}\,\mathrm{d} z
  =1-e^{-r^2/(2|I|)}
  \leqslant\frac{r^2}{2|I|}\leqslant\frac{r^2}{a+b}.
\]
Together with the trivial full probability bound, these inequalities yield
\begin{align*}
  \mathbb{P}(\abs{Z}<r,\abs{Z'}<r)
  &\leqslant\min\left\{1,\frac{r^2}{a+b},\frac{r^4}{4ab}\right\}\\
  &\leqslant\min\left\{1,\frac{r^2}{a},\frac{r^2}{b},\frac{r^4}{ab}\right\}
  =\min\left\{1,\frac{r^2}{a}\right\}\min\left\{1,\frac{r^2}{b}\right\}.
\end{align*}

The Frostman bound and $\mu(\mathbb{R})=1$ give, for $a,b>0$,
\[
  \mu^4\set{0<|I\cap I'|<a,\ |I\mathbin{\triangle}I'|<b}
  \leqslant4!A^3\min\{1,a\}^s\min\{1,b\}^{2s}.
\]
These bounds and Fubini's theorem give
\begin{align*}
  \int_{\set{I\cap I'\neq\varnothing}}
    \mathbb{P}(\abs{Z}<r,\abs{Z'}<r)\,\mathrm{d}\mu^4
  &\leqslant r^4\int_{r^2}^\infty\int_{r^2}^\infty
    \mu^4\set{0<|I\cap I'|<a,\ |I\mathbin{\triangle}I'|<b}
    \frac{\,\mathrm{d} b\,\mathrm{d} a}{a^2b^2}\\
  &\leqslant4!A^3
    \left(r^2\int_{r^2}^\infty\frac{\min\{1,a\}^s}{a^2}\,\mathrm{d} a\right)
    \left(r^2\int_{r^2}^\infty\frac{\min\{1,b\}^{2s}}{b^2}\,\mathrm{d} b\right).
\end{align*}
Their product is at most $C V_s(r)$, since
\begin{align*}
  r^2\int_{r^2}^\infty\frac{\min\{1,a\}^s}{a^2}\,\mathrm{d} a
  &=r^2+r^2\int_{r^2}^1a^{s-2}\,\mathrm{d} a
  =\frac{r^{2s}-sr^2}{1-s}\leqslant\frac{r^{2s}}{1-s},
\end{align*}
and
\begin{align*}
  r^2\int_{r^2}^\infty\frac{\min\{1,b\}^{2s}}{b^2}\,\mathrm{d} b
  &=r^2+r^2\int_{r^2}^1b^{2s-2}\,\mathrm{d} b\\*
  &\leqslant
  \begin{cases}
    r^{4s}/(1-2s),&0<s<1/2,\\
    r^2(1+2\log(1/r)),&s=1/2,\\
    2sr^2/(2s-1),&1/2<s<1.
  \end{cases}
\end{align*}
These are the required bounds, up to a multiplicative factor.

<!-- theorem-6: body_md -->
\label{thm:variance}
Let $0<s<1$ and let $\mu$ be an $s$-Frostman probability measure on $[0,1]$ with Frostman
constant $A\geqslant1$. There is a constant $C>0$, depending only on $s$ and $A$, such that
\[
  \operatorname{Var}\bigl(\mathcal{C}_r(W_*\mu)\bigr)\leqslant C V_s(r),
  \qquad 0<r\leqslant1.
\]
Moreover, $V_s(r)=o(r^{4s})$ as $r\to0$.
<!-- end theorem-6 -->
<!-- proof-9: body_md -->
Expanding the variance and applying Fubini's theorem gives
\[
  \operatorname{Var}\bigl(\mathcal{C}_r(W_*\mu)\bigr)=\int\left(
  \mathbb{P}(\abs{Z}<r,\abs{Z'}<r)
  -\mathbb{P}(\abs{Z}<r)\mathbb{P}(\abs{Z'}<r)
  \right)\,\mathrm{d}\mu^4.
\]
By independence of disjoint Brownian increments, the integrand vanishes when
$I\cap I'=\varnothing$. Otherwise it is at most $\mathbb{P}(\abs{Z}<r,\abs{Z'}<r)$, so
\cref{thm:four-point-integral} gives the variance bound. In each case of
\cref{eq:variance-scale}, $V_s(r)/r^{4s}\to0$ as $r\to0$.
<!-- end proof-9 -->
<!-- theorem-7: body_md -->
\label{thm:uniform-concentration}
Let $0<s<1$ and let $\mu$ be an $s$-Frostman probability measure on $[0,1]$. Then
\[
  \limsup_{t\to\infty}\abs{Y_\mu(t)-H_\mu^s(t)}=0
  \qquad\text{almost surely}.
\]
<!-- end theorem-7 -->
<!-- proof-10: body_md -->
Fix an integer $m\geqslant1$ and put $t_{j,m}\coloneqq j/m$. By \cref{eq:y-variance},
\[
  \sum_{j=0}^\infty\operatorname{Var}(Y_\mu(t_{j,m}))<\infty.
\]
One can then estimate $\mathbb{P}\{\abs{Y_\mu(t)-H_\mu^s(t)}>\varepsilon\}$ by Chebyshev's inequality and apply
the Borel--Cantelli lemma, giving
\begin{equation}\label{eq:grid-convergence}
  Y_\mu(t_{j,m})-H_\mu^s(t_{j,m})\to0
  \qquad\text{almost surely as }j\to\infty.
\end{equation}

If $t\in[t_{j,m},t_{j+1,m}]$, monotonicity of $r\mapsto\mathcal{C}_r(W_*\mu)$ gives
\begin{equation}\label{eq:monotone-fill}
  e^{-2s/m}Y_\mu(t_{j+1,m})\leqslant Y_\mu(t)\leqslant e^{2s/m}Y_\mu(t_{j,m}).
\end{equation}
Combining \cref{eq:grid-convergence,eq:monotone-fill} gives, almost surely,
\[
  \limsup_{t\to\infty}\abs{Y_\mu(t)-H_\mu^s(t)}
  \leqslant (e^{2s/m}-1)\sup_{t\geqslant0}H_\mu^s(t)
  +\sup_{\substack{t,u\geqslant0\\\abs{t-u}\leqslant1/m}}
  \abs{H_\mu^s(t)-H_\mu^s(u)}.
\]
Taking the countable intersection over $m$ and using the boundedness of $H_\mu^s$
and its uniform continuity from \cref{eq:h-definition} gives the desired limit.
<!-- end proof-10 -->
<!-- proof-11: body_md -->
For $i\in\set{1,2}$, define
\[
  \mathcal{E}_i\coloneqq
  \set{\nu\in\mathcal{P}(\mathbb{R}^2)\colon
  \lim_{n\to\infty}\sup_{\substack{q\in\mathbb{Q}\\q\geqslant n}}
  \abs{e^{2sq}\mathcal{C}_{e^{-q}}(\nu)-H_{\mu_i}^s(q)}=0}.
\]
Each $\mathcal{E}_i$ is Borel, since $\nu\mapsto\mathcal{C}_r(\nu)$ is Borel for every $r>0$.
Suppose that $\nu\in\mathcal{E}_1\cap\mathcal{E}_2$. The triangle inequality gives
\[

\label{thm:neighbourhood-moments}
For every $q\geqslant1$, there is $C_q>0$ such that, for $0<r\leqslant1$,

  \mathbb{E}\bigl[\mathcal{L}^2(\langle E\rangle_r)^q\bigr]

\end{equation}
There is also $c>0$ such that, for $0<r\leqslant1$,

\end{equation}
For each $w\in\mathcal{W}(r^2)$, choose an endpoint $t_w$ of $S_w([0,1])$ and put

  \leqslant\pi\sum_{w\in\mathcal{W}(r^2)}(r+\omega_w)^2.

\[
  \mathbb{E}(r+\omega_w)^{2q}\leqslant C_qr^{2q}.

By convexity,

  \leqslant\bigl(\#\mathcal{W}(r^2)\bigr)^{q-1}\sum_w(r+\omega_w)^{2q}.

  \int_{\mathbb{R}^2}h_{r/2}(x)\,\mathrm{d} x&=\frac{\pi r^2}{4},\\

  &\leqslant\frac{\pi r^2}{4}\mathcal{C}_r(\nu).

Some ball of radius $r/2$ in a finite cover of the support of $\nu$ has positive $\nu$-measure.

\end{equation}

\label{thm:neighbourhood-concentration}
There exist constants $C,\gamma>0$ such that, for every $t\geqslant0$,
\[
  \operatorname{Var}\bigl(e^{(2-2s)t}M_{e^{-t}}(E)\bigr)\leqslant Ce^{-\gamma t}.
\]
Consequently, for every fixed $\tau\in\mathbb{R}$, we have
\begin{equation}\label{eq:neighbourhood-grid-concentration}
  e^{(2-2s)(n+\tau)}M_{e^{-(n+\tau)}}(E)-Q(n+\tau)\to0
  \qquad\text{almost surely as }n\to\infty.
\end{equation}

Fix $0<\gamma<\min\{s,2\eta/3\}$ and put $r=e^{-t}$. For $r_{\min}\leqslant r\leqslant1$,
\cref{thm:neighbourhood-moments} gives
\[
  \operatorname{Var}\bigl(r^{2s-2}M_r(E)\bigr)\leqslant r^{4s-4}\mathbb{E}[M_r(E)^2]\leqslant C
  \leqslant Cr_{\min}^{-\gamma}e^{-\gamma t}.
\]
We may therefore, without loss of generality, assume $0<r<r_{\min}$.
For $w\in\mathcal{W}(r)$, we have $r_w>r_{\min}r>r^2$, so $r/\sqrt{r_w}<1$.
By \cref{eq:cylinder-neighbourhood-scaling,thm:neighbourhood-moments}, for every $q\geqslant1$,
\[
  \mathbb{E}[M_r(W(S_wK))^q]
  \leqslant C_qr_w^q\left(\frac{r}{\sqrt{r_w}}\right)^{q(2-2s)}
  =C_qr^{q(2-2s)}p_w^q.
\]
As in \cref{sec:osc-geometry}, we partition $\mathcal{W}(r)$ into classes
$\mathcal{W}_1(r),\ldots,\mathcal{W}_M(r)$ such that the time intervals $I_w$ have pairwise disjoint interiors
within each class.
By independence within each class, Cauchy--Schwarz, and $p_w\leqslant r^s$, we obtain
\begin{equation}\label{eq:stopping-neighbourhood-variance}
  \begin{split}
    r^{4s-4}\operatorname{Var}\left(\sum_{w\in\mathcal{W}(r)}M_r(W(S_wK))\right)
    &\leqslant Mr^{4s-4}\sum_{j=1}^M
      \operatorname{Var}\left(\sum_{w\in\mathcal{W}_j(r)}M_r(W(S_wK))\right)\\
    &=Mr^{4s-4}\sum_{w\in\mathcal{W}(r)}\operatorname{Var}(M_r(W(S_wK)))\\
    &\leqslant C\sum_{w\in\mathcal{W}(r)}p_w^2
     \leqslant Cr^s\sum_{w\in\mathcal{W}(r)}p_w=Cr^s.
  \end{split}
\end{equation}

We next bound the total overlap defect. Let $\mathcal{T}(r)$ be the set of proper prefixes of words
in $\mathcal{W}(r)$, including the empty word. For $v\in\mathcal{T}(r)$, put
\[
  D_{v,r}=\sum_i M_r(W(S_vS_iK))-M_r(W(S_vK))\geqslant0.
\]
The terms for internal cylinders cancel, giving
\[
  \sum_{v\in\mathcal{T}(r)}D_{v,r}
  =\sum_{w\in\mathcal{W}(r)}M_r(W(S_wK))-M_r(E).
\]
For $v\in\mathcal{T}(r)$, we have $r_v>r$. Brownian scaling and
\cref{eq:neighbourhood-defect-mean} give
\begin{align*}

  &=r^{2s-2}r_v\mathbb{E} D_{r/\sqrt{r_v}}
  \leqslant Cr^{2s-2}r_v\left(\frac{r}{\sqrt{r_v}}\right)^{2-2s+2\eta}

\end{align*}
Since $r<r_v\leqslant(\max_i r_i)^{\abs{v}}$, the set $\mathcal{T}(r)$ has at most
$C(1+\log(1/r))$ generations. In each generation,
\[

  \leqslant\sum_{\abs{v}=k}p_v=1.
\]
Consequently,
\begin{equation}\label{eq:stopping-neighbourhood-defect}
  \begin{split}
    r^{2s-2}\sum_{v\in\mathcal{T}(r)}\mathbb{E} D_{v,r}
    &\leqslant Cr^{2\eta}\sum_{v\in\mathcal{T}(r)}p_vr_v^{-\eta}
     \leqslant Cr^\eta\sum_{v\in\mathcal{T}(r)}p_v\\
    &\leqslant Cr^\eta(1+\log(1/r)).

\end{equation}
The total defect is nonnegative and at most $\sum_{w\in\mathcal{W}(r)}M_r(W(S_wK))$.
Since $\sum_{w\in\mathcal{W}(r)}p_w=1$, Jensen's inequality and the fourth-moment bound give
\begin{align*}

  &\leqslant r^{8s-8}\sum_{w\in\mathcal{W}(r)}p_w^{-3}\mathbb{E}[M_r(W(S_wK))^4]\\
  &\leqslant C\sum_{w\in\mathcal{W}(r)}p_w=C.
\end{align*}
H\"older's inequality and \cref{eq:stopping-neighbourhood-defect} now give
\begin{align*}
  r^{4s-4}\mathbb{E}\left[\left(\sum_{v\in\mathcal{T}(r)}D_{v,r}\right)^2\right]
  &\leqslant r^{4s-4}\left(\sum_{v\in\mathcal{T}(r)}\mathbb{E} D_{v,r}\right)^{2/3}
    \left(\mathbb{E}\left[\left(\sum_{v\in\mathcal{T}(r)}D_{v,r}\right)^4\right]\right)^{1/3}\\
  &\leqslant Cr^{2\eta/3}(1+\log(1/r))^{2/3}.
\end{align*}

\[

  \abs{\int f\,\mathrm{d}\nu_{E,e^{-n}}-\sum_{\abs{w}=k}p_wf(x_w)}
  \leqslant
  \operatorname{Lip}(f)\max_{\abs{w}=k}\operatorname{diam}(W(S_w(K))).
\]

To prove the claim, we first replace the maximum of the cylinder cut-offs by their sum. The
resulting

\cref{eq:level-neighbourhood-defect}. Moreover, if
$x\in\langleW(S_w(K))\rangle_{e^{-n}}$, then
\[
  \abs{f(x)-f(x_w)}
  \leqslant\operatorname{Lip}(f)(\operatorname{diam}(W(S_w(K)))+e^{-n}).
\]
Integrating against the cylinder cut-offs and dividing by $M_{e^{-n}}(E)$ bounds the error from
replacing $f(x)$ by $f(x_w)$ on each cylinder by
\[
  \operatorname{Lip}(f)
  \sum_{\abs{w}=k}(\operatorname{diam}(W(S_w(K)))+e^{-n})

\]
Finally, \cref{eq:cylinder-neighbourhood-ratio} shows that the additional error incurred by
replacing the remaining cylinder ratios by $p_w$ tends to zero.

\[

\[
  \sum_{\abs{w}=k}p_wf(x_w)\to\int f\,\mathrm{d}(W_*\mu).
\]
Letting first $n\to\infty$ and then $k\to\infty$ proves the weak convergence
\cref{eq:minkowski-reconstruction}.

Finally, $F\mapsto\nu_{F,r}$ is continuous for each $r>0$ by dominated convergence. Since
$\mathcal{P}(\mathbb{R}^2)$ is Polish, the limit $\lim_{n\to\infty}\nu_{F,e^{-n}}$ is Borel measurable on
its Borel set of convergence. The almost-sure convergence above therefore proves the Borel
assertion.

\label{thm:base-five-profiles}
For every integer $n\geqslant1$,
\begin{equation}\label{eq:base-five-correlation}
  \mathcal{C}_{5^{-n}}(\mu_1)=\frac32\,3^{-n}-\frac12\,9^{-n},
  \qquad \mathcal{C}_{5^{-n}}(\mu_2)=3^{-n}.
\end{equation}
In particular, the expected Brownian correlation profiles satisfy
\[
  \limsup_{t\to\infty}\abs{H_{\mu_1}^s(t)-H_{\mu_2}^s(t)}>0.
\]

<!-- proof-18: body_md -->
For $n\geqslant1$, each level-$n$ cylinder interval has length
$5^{-n}$ and mass $3^{-n}$. Pairs in the same cylinder contribute $3^{-n}$ to the
correlation integral. Distinct cylinder intervals either touch or are separated by at least
$5^{-n}$, with equality contributing zero since the measures are atomless. For two touching
intervals, the distance is less than $5^{-n}$ precisely when the rescaled right point is smaller
than the rescaled left point. By exchangeability, each order therefore contributes
$9^{-n}/2$.

For $K_1$, touching pairs have words with a common prefix of length $k$, followed by $0$ and
$1$, and then respectively $n-k-1$ copies of $4$ and $0$. There are consequently
$\sum_{k=0}^{n-1}3^k=(3^n-1)/2$ such pairs, whereas $K_2$ has none. Adding these
contributions proves \cref{eq:base-five-correlation}.

Both IFSs are of lattice type with span $\ell$, so the renewal limits in \cref{thm:renewal-limits}
give continuous $\ell$-periodic functions $\widetilde G_j$ with
$G_j(t)-\widetilde G_j(t)\to0$ as $t\to\infty$, and
\cref{eq:base-five-correlation} gives
\[
  \widetilde G_1(0)=\lim_{n\to\infty}G_1(n\ell)=\frac32,
  \qquad
  \widetilde G_2(0)=\lim_{n\to\infty}G_2(n\ell)=1.
\]
The expected profiles converge to their smoothed periodic limits, as in the proof of
\cref{thm:profile-asymptotics}. These limits are distinct by the injectivity of smoothing in
\cref{thm:smoothing-injective}, giving the separation.

    &=\begin{cases}
      6,&k=0,\\
      1,&\abs k\in\set{1,\ldots,13,16,17},\\

    \end{cases}

\end{equation}

\label{thm:level-sums}

bounded in modulus by some $\rho<1$, with $\rho B^2<1$. Let $D\subset\mathbb{C}$ be a bounded open convex set

<!-- lemma-11: body_md -->
\label{thm:fixed-dimension-analyticity}
For every fixed $s\in(0,1)$, the functions
\[
  p\mapsto\int_0^1\delta^{-s}\,\mathrm{d}\mathcal{C}^\times_\delta(\mu_p)
  \quad\text{and}\quad p\mapsto\overline H_s(p)
\]
are real analytic on $(0,1)$.
<!-- end lemma-11 -->
<!-- proof-22: body_md -->
Fix $p_0\in(0,1)$ and $\rho\in(\max\{\lambda_1(p_0),\lambda_2(p_0)\},1)$. On a small open disc
$U$ centred at $p_0$, the branches $\lambda_1$ and $\lambda_2$ are holomorphic with
$\abs{\lambda_1},\abs{\lambda_2}\leqslant\rho$, and $B=\sup_U(\abs z+\abs{1-z})$ is as close to $1$
as we wish, so we may assume $\rho B^2<1$.
Iterating the contraction maps gives
$\abs{x_w(z)-x_w(p_0)}\leqslant C\abs{z-p_0}/(1-\rho)$ uniformly in $w$. Since
$g_0=1-\lambda_1(p_0)-\lambda_2(p_0)>0$, we may shrink $U$ and choose a bounded convex

  \psi(z;x,y)=\bigl(1-\lambda_2(z)+\lambda_2(z)y-\lambda_1(z)x\bigr)^{-s}

on $U$ with
\[
  \Lambda(p)=\frac1{2pq}\int_0^1\delta^{-s}\,\mathrm{d}\mathcal{C}^\times_\delta(\mu_p)
\]
for real $p\in U$. Multiplication by $2z(1-z)$ extends the cross-term integral holomorphically.
After shrinking $U$, the principal-branch extension $h(z)=-z\log z-(1-z)\log(1-z)$ is nonzero,
since $h(p_0)>0$, so \cref{eq:fixed-dimension-mean} also extends the mean holomorphically.
These local extensions agree on overlapping discs by the identity theorem, giving extensions to
a complex neighbourhood of $(0,1)$. Their restrictions to $(0,1)$ are real analytic.

  \overline H_s(p)=O\!\left(\frac{p}{h(p)}\right)

and real analytic on $(0,1)$. Its derivative and each difference
