<!-- assumption-1: body_md -->
\label{ass:path-space}
The admissible trajectory set \(\Xi\), equipped with the metric \(d_{\kappa,\lambda}\) in \hyperref[eq:kappa-lambda-metric]{(\ref*{eq:kappa-lambda-metric})}, is a Polish metric space with finite diameter: $
\operatorname{diam}_{\kappa,\lambda}(\Xi)
\coloneqq
\sup_{X,\widehat X\in\Xi} d_{\kappa,\lambda}(X,\widehat X)
<\infty.$
<!-- end assumption-1 -->
<!-- assumption-2: body_md -->
\label{ass:nonstationary-drift}
The trajectories \(X_1,\dots,X_N\) are independent, with \(X_n\sim P_n\) for \(n=1,\dots,N\). Moreover, there exists \(\varrho\ge0\) such that
\[
W_\infty(P_n,P_{n+1})\le\varrho,
\qquad n=1,\dots,N.
\]
<!-- end assumption-2 -->
<!-- theorem-1: body_md -->
\label{thm:drift-aware-next-period-risk}
Let $p\in(1,\infty)$ and suppose Assumptions~\ref{ass:path-space} and~\ref{ass:nonstationary-drift} hold. There exist constants $c_1,c_2>0$ and $\nu\in(0,1/2)$, depending only on $d(T+1)$, $p$, $\operatorname{diam}_{\kappa,\lambda}(\Xi)$, such that the following holds for every $w\in\Delta_N$ and $\beta\in(0,1)$. Define
\[
\varepsilon_\varrho(w,\beta)
\coloneqq
D_p(w)\varrho
+
\left(
c_2N_{\mathrm{eff}}(w)^{-\nu}
+
\sqrt{\frac{\log(1/\beta)}{c_1N_{\mathrm{eff}}(w)}}
\right)^{1/p}.
\]
With probability at least $1-\beta$, we have $P_{N+1}\in\mathcal B_{\varepsilon_\varrho(w,\beta)}(\mu_w)$ and $\mathbb E_{X\sim P_{N+1}}[\ell_{\mathrm{DH}}(\vartheta;X)]\le V_{\vartheta}(\varepsilon_\varrho(w,\beta))$ for every $\vartheta\in\Theta\times\mathbb R$, where $V_{\vartheta}$ uses reference law $\mu_w$.
<!-- end theorem-1 -->
<!-- definition-1: body_md -->
\label{def:phi-divergence}
A function \(\phi:[0,\infty)\to[0,\infty]\) is an \emph{entropy function} if it
is proper, convex, lower semicontinuous, and satisfies \(\phi(1)=0\). For
\(\pi',\pi\in\mathcal P(\Xi\times\Xi)\), define the $\phi$-divergence of $\pi'$ with respect to~$\pi$ induced by $\phi$ as
\[
D_\phi(\pi'\|\pi)
\coloneqq
\begin{cases}
\displaystyle\int_{\Xi\times\Xi}
\phi\!\left(\frac{d\pi'}{d\pi}\right)\,d\pi, & \pi'\ll\pi,\\
+\infty, & \pi'\not\ll\pi.
\end{cases}
\]
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:two_budget_phi_ot_clean}
For transport budget $\varepsilon\ge0$ and reweighting budget $\tau\ge0$, define
\begin{equation}
\label{eq:short-two-budget-set}
\mathcal B_{\varepsilon,\tau}(\mu_w)
\coloneqq
\left\{
\eta\in\mathcal P(\Xi):
\begin{array}{l}
\text{there exist }\pi,\pi'\in\mathcal P(\Xi\times\Xi)
\text{ such that}\\[0.4ex]
\pi_1=\mu_w,\quad \pi'_2=\eta,\quad
D_\phi(\pi'\|\pi)\le\tau,\\[0.8ex]
\displaystyle
\int_{\Xi\times\Xi}
d_{\kappa,\lambda}(x,\widehat x)^p\,
d\pi'(x,\widehat x)\le\varepsilon^p
\end{array}
\right\}.
\end{equation}
<!-- end definition-2 -->
<!-- remark-1: body_md -->
\label{rem:two-budget-coverage}
For every \(\tau\ge0\),
\(\mathcal B_\varepsilon(\mu_w)\subseteq\mathcal B_{\varepsilon,\tau}(\mu_w)\),
since any Wasserstein-feasible coupling is feasible in \hyperref[eq:short-two-budget-set]{(\ref*{eq:short-two-budget-set})} by taking \(\pi'\!=\!\pi\). Hence the transport radius calibrated in Section~\ref{subsec:short-coverage} retains its next-period coverage guarantee under the two-budget formulation. This inclusion does not justify reducing the calibrated transport radius in exchange for a reweighting budget.
<!-- end remark-1 -->
<!-- proposition-1: body_md -->
\textnormal{(Dual geometry of the weighted path metric)}
\label{prop:dual-geometry-main}
Let $p\in(1,\infty)$, $q=p/(p-1)$, and $\kappa_t,\lambda_i>0$. For $\Delta X,G\in\mathbb R^{d(T+1)}$, define
\begin{equation}
\label{eq:dual_norm_main}
\|\Delta X\|_{\kappa,\lambda,p}
\coloneqq\Bigl(\sum_i(\lambda_i\max_t\kappa_t|\Delta X_t^i|)^p\Bigr)^{1/p},
\quad
\|G\|_{\kappa,\lambda,p}^{\ast}
\coloneqq\Bigl(\sum_i\Bigl(\sum_t\frac{|G_t^i|}{\lambda_i\kappa_t}\Bigr)^q\Bigr)^{1/q}.
\end{equation}
The first is a norm inducing the metric in \hyperref[eq:kappa-lambda-metric]{(\ref*{eq:kappa-lambda-metric})} through \(d_{\kappa,\lambda}(X,\widehat X)=\|X-\widehat X\|_{\kappa,\lambda,p}\), and the second is its dual norm under the Euclidean inner product. Moreover, there exists a Borel map \(h:\mathbb R^{d(T+1)}\to\mathbb R^{d(T+1)}\), with \(h(0)=0\), such that \(\|h(G)\|_{\kappa,\lambda,p}=1\) and \(\langle G,h(G)\rangle=\|G\|_{\kappa,\lambda,p}^{\ast}\) whenever \(G\ne0\).
<!-- end proposition-1 -->
<!-- assumption-3: body_md -->
\label{ass:loss-regularity}
The trajectories $X_n$ lie in the interior of $\Xi$, and for every $\vartheta\in\Theta\times\mathbb R$, the loss $\ell_{\mathrm{DH}}(\vartheta;\cdot)$ is Borel measurable, bounded on $\Xi$ and admits at each $X_n$ the expansion
\[
\ell_{\mathrm{DH}}(\vartheta;X_n+\Delta)
=L_n(\vartheta)+\langle g_n(\vartheta),\Delta\rangle+o(\|\Delta\|_{\kappa,\lambda,p}), \quad \|\Delta\|_{\kappa,\lambda,p}\to0.
\]

<!-- theorem-2: body_md -->
\textnormal{(Joint first-order approximation)}
\label{thm:two_budget_ot_phi_taylor}
Suppose $\phi$ is locally regular,  $p\in(1,\infty)$ and Assumption~\ref{ass:loss-regularity} holds. For every fixed $\vartheta\in\Theta\times\mathbb R$, as $(\varepsilon,\tau)\to(0,0)$

\label{eq:two_budget_joint_taylor}
V_{\vartheta}(\varepsilon,\tau)
=
\bar L_w(\vartheta)
+
\Upsilon_w(\vartheta)\,\varepsilon
+
\sqrt{\frac{2}{\phi''(1)}}\,\sigma_w(\vartheta)\sqrt{\tau}
+
o\big(\varepsilon+\sqrt{\tau}\big).
\end{equation}

\label{rem:payoff-regularity}
Many vanilla payoffs, such as calls or puts, are globally Lipschitz in $S_T$ and hence in $X$. If differentiability everywhere is desired, kinked terms such as $\max(\cdot,0)$ or $|\cdot|$ can be replaced by smooth approximations (\textit{e.g.}~softplus for the positive part). The same approach applies to nonsmooth OCE losses, such as the CVaR loss, while preserving convexity, monotonicity, and the OCE normalization. These approximations preserve the structure of the DRO formulation.

Assumption~\ref{ass:path-space} is satisfied when all $\lambda_i$ and $\kappa_t$ are positive and \(\Xi\) is compact. In implementations, compactness corresponds to working with normalized and bounded features or clipping extreme inputs. The Polish-space condition ensures that the Wasserstein ambiguity sets and couplings are well defined, while the finite-diameter condition is used in the concentration bounds and sensitivity analysis.

\textit{Primal norm.}\ 

Homogeneity and definiteness follow directly from the positivity of the weights
\(\kappa_t\) and \(\lambda_i\). For the triangle inequality, define, for each component \(i\),
\[
\|\Delta X^i\|_{\infty,\kappa}
\coloneqq
\max_{t=0,\dots,T}\kappa_t|\Delta X_t^i|,
\qquad
\Delta X^i=(\Delta X_0^i,\dots,\Delta X_T^i).
\]
The weighted sup-norm satisfies
\[
\|\Delta X^i+\Delta Y^i\|_{\infty,\kappa}
\le \|\Delta X^i\|_{\infty,\kappa}
+ \|\Delta Y^i\|_{\infty,\kappa}.
\]
Therefore,
\[
\begin{aligned}
\|\Delta X+\Delta Y\|_{\kappa,\lambda,p}
&=
\left(
\sum_{i=1}^d
\left(
\lambda_i
\|\Delta X^i+\Delta Y^i\|_{\infty,\kappa}
\right)^p
\right)^{1/p} \\
&\le
\left(
\sum_{i=1}^d
\left(
\lambda_i\|\Delta X^i\|_{\infty,\kappa}
+
\lambda_i\|\Delta Y^i\|_{\infty,\kappa}
\right)^p
\right)^{1/p}.
\end{aligned}
\]
Applying Minkowski's inequality to the outer \(\ell_p\) aggregation, with
\(a_i=\lambda_i\|\Delta X^i\|_{\infty,\kappa}\) and
\(b_i=\lambda_i\|\Delta Y^i\|_{\infty,\kappa}\), gives
\[
\begin{aligned}
\|\Delta X+\Delta Y\|_{\kappa,\lambda,p}
&\le
\left(
\sum_{i=1}^d
\left(
\lambda_i\|\Delta X^i\|_{\infty,\kappa}
\right)^p
\right)^{1/p}
+
\left(
\sum_{i=1}^d
\left(
\lambda_i\|\Delta Y^i\|_{\infty,\kappa}
\right)^p
\right)^{1/p} \\
&=
\|\Delta X\|_{\kappa,\lambda,p}
+
\|\Delta Y\|_{\kappa,\lambda,p}.
\end{aligned}
\]
Thus the triangle inequality holds, and \(\|\cdot\|_{\kappa,\lambda,p}\) is a norm.

\smallskip\noindent\textit{Dual norm.} For \(G,\Delta X\in\mathbb{R}^{d(T+1)}\), write
\[
\langle G,\Delta X\rangle
\coloneqq
\sum_{i=1}^d\sum_{t=0}^T G_t^i\Delta X_t^i.
\]
Let \(q\) be the H\"older conjugate of \(p\in(1,\infty)\), so that \(1/p+1/q=1\).

For each component \(i\), define
\(b_i\coloneqq \sum_{t=0}^T |G_t^i|/\kappa_t\). For any
\(\Delta X\in\mathbb{R}^{d(T+1)}\), the componentwise pairing satisfies
\[
\sum_{t=0}^T G_t^i\Delta X_t^i
\le
\sum_{t=0}^T |G_t^i|\,|\Delta X_t^i|
\le
\|\Delta X^i\|_{\infty,\kappa}\, b_i,
\]
where
\(\|\Delta X^i\|_{\infty,\kappa}\coloneqq\max_{t=0,\dots,T}\kappa_t|\Delta X_t^i|\).
Summing over \(i\), we obtain
\[
\langle G,\Delta X\rangle
\le
\sum_{i=1}^d
\left(\lambda_i\|\Delta X^i\|_{\infty,\kappa}\right)
\frac{b_i}{\lambda_i}.
\]
H\"older's inequality in \(\mathbb{R}^d\) gives
\[
\langle G,\Delta X\rangle
\le
\|\Delta X\|_{\kappa,\lambda,p}
\left(
\sum_{i=1}^d
\left(\frac{b_i}{\lambda_i}\right)^q
\right)^{1/q}.
\]
Taking the supremum over all \(\Delta X\) with
\(\|\Delta X\|_{\kappa,\lambda,p}\le 1\) yields
\begin{equation}
\label{eq:dual_norm_upper_bound}
\|G\|_{\kappa,\lambda,p}^\ast
\le
\left(
\sum_{i=1}^d
\left(
\sum_{t=0}^T
\frac{|G_t^i|}{\lambda_i\kappa_t}
\right)^q
\right)^{1/q}.
\end{equation}

It remains to show that the upper bound is tight. Since the dual norm is the
supremum over all feasible perturbations, it is at least the supremum over any
feasible subclass. Choose
\(\Delta X_t^i=\operatorname{sgn}(G_t^i)\alpha_i/\kappa_t\), with
\(\alpha_i\ge 0\). Then
$\|\Delta X^i\|_{\infty,\kappa}\le\alpha_i$,
$\|\Delta X\|_{\kappa,\lambda,p}
\le(\sum_{i=1}^d(\lambda_i\alpha_i)^p)^{1/p}$, and
\(\langle G,\Delta X\rangle=\sum_{i=1}^d\alpha_i b_i\). Hence
\[
\|G\|_{\kappa,\lambda,p}^\ast
\ge
\sup_{\left(\sum_{i=1}^d(\lambda_i\alpha_i)^p\right)^{1/p}\le 1}
\sum_{i=1}^d \alpha_i b_i.
\]
Setting \(c_i=\lambda_i\alpha_i\), the right-hand side becomes
\[
\sup_{\|c\|_p\le 1}
\sum_{i=1}^d c_i\frac{b_i}{\lambda_i}
=
\left(
\sum_{i=1}^d
\left(\frac{b_i}{\lambda_i}\right)^q
\right)^{1/q},
\]
by the standard duality between \(\ell_p\) and \(\ell_q\). Substituting
\(b_i=\sum_{t=0}^T |G_t^i|/\kappa_t\) gives
\begin{equation}
\label{eq:dual_norm_lower_bound}
\|G\|_{\kappa,\lambda,p}^\ast
\ge
\left(
\sum_{i=1}^d
\left(
\sum_{t=0}^T
\frac{|G_t^i|}{\lambda_i\kappa_t}
\right)^q
\right)^{1/q}.
\end{equation}
Combining (\ref{eq:dual_norm_upper_bound}) and
(\ref{eq:dual_norm_lower_bound}) proves
(\ref{eq:dual_norm_main}).

\smallskip\noindent\textit{Maximizer map.} The maximizer map \(h\) is applied to dual variables, such as pathwise gradients
\(G\in\mathbb{R}^{d(T+1)}\), rather than to trajectories. For
\(G\in\mathbb{R}^{d(T+1)}\), define
\(a_i(G)\coloneqq \sum_{t=0}^T |G_t^i|/(\lambda_i\kappa_t)\) and
\(A(G)\coloneqq (\sum_{j=1}^d a_j(G)^q)^{1/q}\). By
(\ref{eq:dual_norm_main}), \(A(G)=\|G\|_{\kappa,\lambda,p}^\ast\).

For
\(G\neq 0\), define
\(r_i(G)\coloneqq a_i(G)^{q-1}/A(G)^{q-1}\) and set
\begin{equation}
\label{eq:h_explicit_app}
h(G)_t^i
\coloneqq
\frac{r_i(G)}{\lambda_i\kappa_t}\operatorname{sign}(G_t^i),
\qquad
i=1,\dots,d,\quad t=0,\dots,T.
\end{equation}
For \(G=0\), we set
\(h(0)=0\).

By construction, $\kappa_t|h(G)_t^i|\le r_i(G)/\lambda_i$. If $r_i(G)>0$, some $G_t^i$ is nonzero, so equality is attained at that date; if $r_i(G)=0$, the entire component vanishes. Thus $\lambda_i\max_t\kappa_t|h(G)_t^i|=r_i(G)$. Hence
\[
\|h(G)\|_{\kappa,\lambda,p}
=
\left(
\sum_{i=1}^d r_i(G)^p
\right)^{1/p}.
\]
From \(r_i(G)=a_i(G)^{q-1}/A(G)^{q-1}\) and the identity
\((q-1)p=q\), we obtain
\[
\sum_{i=1}^d r_i(G)^p
=
\frac{\sum_{i=1}^d a_i(G)^q}{A(G)^q}
=
1,
\]
so \(\|h(G)\|_{\kappa,\lambda,p}=1\).

The same construction attains the dual norm. Substituting \hyperref[eq:h_explicit_app]{(\ref*{eq:h_explicit_app})} into the pairing gives
\[
\langle G,h(G)\rangle
=
\sum_{i=1}^d r_i(G)
\sum_{t=0}^T
\frac{|G_t^i|}{\lambda_i\kappa_t}
=
\sum_{i=1}^d r_i(G)a_i(G).
\]
Substituting the definition of \(r_i(G)\), this becomes
\[
\sum_{i=1}^d r_i(G)a_i(G)
=
\frac{\sum_{i=1}^d a_i(G)^q}{A(G)^{q-1}}
=
A(G).
\]
Since \(A(G)=\|G\|_{\kappa,\lambda,p}^\ast\) by
(\ref{eq:dual_norm_main}), we conclude that
\(\langle G,h(G)\rangle=\|G\|_{\kappa,\lambda,p}^\ast\).

<!-- lemma-1: body_md -->
\label{lem:expectation-bound}
Suppose Assumption~\ref{ass:path-space} holds. For
\(p\in(1,\infty)\) and \(P\in\mathcal P(\Xi)\), let \(X_1,\dots,X_N\sim P\) be i.i.d., and define
\(\mu_w\coloneqq\sum_{n=1}^N w_n\delta_{X_n}\) for \(w\in\Delta_N\). Then
there exist constants \(c_0>0\) and \(\nu\in(0,1/2)\), depending only on the
ambient dimension \(d(T+1)\), \(p\), and
\(\operatorname{diam}_{\kappa,\lambda}(\Xi)\), such that
\[
\mathbb E\!\left[
W_p(\mu_w,P)^p
\right]
\le
c_0 N_{\mathrm{eff}}(w)^{-\nu}.
\]
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
The proof reduces the weighted empirical measure to a convex combination of ordinary empirical measures and then uses the standard Wasserstein convergence rate for i.i.d.~samples.

Since \(X_1,\dots,X_N\) are i.i.d., the law of
\(\sum_{n=1}^N w_n\delta_{X_n}\) is invariant under permutations of the weights. Since \(N_{\mathrm{eff}}(w)\) is also permutation-invariant, it is enough to prove the claim for weights sorted in nonincreasing order. We therefore write \(w_1\ge\cdots\ge w_N\) and set \(w_{N+1}=0\).
For \(k=1,\dots,N\), define 
\[
\bar P_k \coloneqq \frac1k\sum_{i=1}^k \delta_{X_i},
\qquad
\alpha_k\coloneqq k(w_k-w_{k+1}).
\]
Then \(\mu_w\) admits the decomposition $\mu_w =\sum_{k=1}^N \alpha_k \bar P_k$. 
Indeed, \(\alpha_k\ge0\) and
\[
\sum_{k=1}^N\alpha_k
=\sum_{k=1}^N k(w_k-w_{k+1})
=\sum_{k=1}^N w_k
=1,
\]
so this is a convex combination. Moreover, the coefficient of \(\delta_{X_i}\)
in \(\sum_{k=1}^N\alpha_k\bar P_k\) is
\[
\sum_{k=i}^N \frac{\alpha_k}{k}
=\sum_{k=i}^N (w_k-w_{k+1})
=w_i.
\]
Hence the decomposition is exact.

The map \(Q\mapsto W_p(Q,P)^p\) is convex in its first argument. Therefore,
\[
W_p(\mu_w,P)^p
\le
\sum_{k=1}^N \alpha_k W_p(\bar P_k,P)^p.
\]
Taking expectations gives
\[
\mathbb E\!\left[W_p(\mu_w,P)^p\right]
\le
\sum_{k=1}^N \alpha_k
\mathbb E\!\left[W_p(\bar P_k,P)^p\right].
\]
Since \(\Xi\) has finite diameter under \(d_{\kappa,\lambda}\), standard empirical Wasserstein convergence bounds on compact subsets of finite-dimensional normed spaces, such as \citet{Fournier2015on}, imply that there exist constants
\(C>0\) and \(\nu\in(0,1/2)\) such that
\[
\mathbb E\!\left[W_p(\bar P_k,P)^p\right]
\le
Ck^{-\nu},
\qquad k=1,\dots,N.
\]
The ground metric \(d_{\kappa,\lambda}\) is induced by a weighted norm on a finite-dimensional space, so the usual Euclidean-rate statement applies up to a change of constants. Thus,
\[
\mathbb E\!\left[W_p(\mu_w,P)^p\right]
\le
C\sum_{k=1}^N k(w_k-w_{k+1})k^{-\nu}
=
C\sum_{k=1}^N k^{1-\nu}(w_k-w_{k+1}).
\]
By summation by parts,
\[
\sum_{k=1}^N k^{1-\nu}(w_k-w_{k+1})
=
\sum_{k=1}^N
\big(k^{1-\nu}-(k-1)^{1-\nu}\big)w_k,
\]
where \(0^{1-\nu}=0\). In the remainder of the proof, \(C_\nu<\infty\) denotes a constant depending only on \(\nu\), whose value may change from line to line. Since \(x\mapsto x^{1-\nu}\) is concave and increasing,
\[
k^{1-\nu}-(k-1)^{1-\nu}
\le
C_\nu k^{-\nu}.
\]
Hence
\[
\mathbb E\!\left[W_p(\mu_w,P)^p\right]
\le
C C_\nu \sum_{k=1}^N k^{-\nu}w_k.
\]

It remains to control the last sum by the effective sample size. Let
\(m\coloneqq \lfloor N_{\mathrm{eff}}(w)\rfloor\). Since
\(N_{\mathrm{eff}}(w)\ge1\), we have \(m\ge1\). Split
\[
\sum_{k=1}^N k^{-\nu}w_k
=
\sum_{k=1}^m k^{-\nu}w_k
+
\sum_{k=m+1}^N k^{-\nu}w_k
\coloneqq H+R.
\]
For the head term, Cauchy--Schwarz and the definition of
\(N_{\mathrm{eff}}(w)\) give
\[
H
\le
\left(\sum_{k=1}^m k^{-2\nu}\right)^{1/2}
\left(\sum_{k=1}^m w_k^2\right)^{1/2}
\le
N_{\mathrm{eff}}(w)^{-1/2}
\left(\sum_{k=1}^m k^{-2\nu}\right)^{1/2}.
\]
Because \(\nu<1/2\),
\[
\sum_{k=1}^m k^{-2\nu}
\le
C_\nu m^{1-2\nu}.
\]
Therefore,
\[
H
\le
C_\nu
N_{\mathrm{eff}}(w)^{-1/2}m^{1/2-\nu}
\le
C_\nu N_{\mathrm{eff}}(w)^{-\nu},
\]
where the last inequality uses \(m\le N_{\mathrm{eff}}(w)\).

For the remainder term, since \(k^{-\nu}\) is decreasing and the weights sum to one,
\[
R
\le
(m+1)^{-\nu}\sum_{k=m+1}^N w_k
\le
(m+1)^{-\nu}
\le
N_{\mathrm{eff}}(w)^{-\nu}.
\]
Combining the bounds on \(H\) and \(R\), we obtain
\[
\sum_{k=1}^N k^{-\nu}w_k
\le
C_\nu N_{\mathrm{eff}}(w)^{-\nu}.
\]
Absorbing constants into \(c_0\) gives
\[
\mathbb E\!\left[
W_p(\mu_w,P)^p
\right]
\le
c_0N_{\mathrm{eff}}(w)^{-\nu},
\]
which proves the claim.

<!-- proposition-2: body_md -->
\label{prop:concentration-bound}
Let \(p\in(1,\infty)\) and assume \(\mathrm{diam}_{\kappa,\lambda}(\Xi)<\infty\). Let
\(P\in\mathcal P(\Xi)\), and let \(\nu\in(0,1/2)\) be the exponent from Lemma~\ref{lem:expectation-bound}. Let \(X_1,\dots,X_N\sim P\) be independent and identically distributed, and define \(\mu_w\coloneqq \sum_{i=1}^N w_i\delta_{X_i}\). There exist constants \(c_1,c_2>0\) such that, for all \(N\in\mathbb N\), \(w\in\Delta_N\), and
\(\varepsilon>0\),
\[
\mathbb P\!\left(W_p(\mu_w,P)\ge \varepsilon\right)
\le
\exp\!\left(-c_1N_{\mathrm{eff}}(w)\Big(\varepsilon^p-c_2N_{\mathrm{eff}}(w)^{-\nu}\Big)_+^2\right).
\]
In particular, whenever
\(\varepsilon \ge 2(c_2 N_{\mathrm{eff}}(w)^{-\nu})^{1/p}\), we have
\[
\mathbb P\!\left(
W_p(\mu_w,P)\ge \varepsilon
\right)
\le
\exp\!\left(
-\frac{c_1}{4}N_{\mathrm{eff}}(w)\varepsilon^{2p}
\right).
\]
<!-- end proposition-2 -->
<!-- proof-3: body_md -->
We apply McDiarmid's inequality to the random variable
\[
f(X_1,\dots,X_N)
\coloneqq
W_p\!\left(\sum_{i=1}^N w_i\delta_{X_i},P\right)^p .
\]
The key step is to verify a bounded-differences property. Changing one observation affects only the mass \(w_j\) attached to that observation; boundedness of \(\Xi\) therefore yields a bounded-differences constant proportional to \(w_j\).

Fix \(j\in\{1,\dots,N\}\), and let \(X_j'\in\Xi\) be an arbitrary replacement of the
\(j\)-th trajectory. Define
\[
\mu_w^{(j)}
\coloneqq
\sum_{i\neq j}w_i\delta_{X_i}+w_j\delta_{X_j'} .
\]
We claim that
\begin{equation}
\label{eq:bounded-difference-wp}
\left|
W_p(\mu_w,P)^p
-
W_p(\mu_w^{(j)},P)^p
\right|
\le
w_j\,\mathrm{diam}_{\kappa,\lambda}(\Xi)^p .
\end{equation}

To prove the claim, let
\(\pi^{(j)}\in\Pi(\mu_w^{(j)},P)\) be arbitrary. Since
\(\mu_w^{(j)}\) is discrete, we may disintegrate \(\pi^{(j)}\) according to
the indexed atoms of \(\mu_w^{(j)}\). Thus, for some conditional laws
\(\beta_1,\dots,\beta_N\in\mathcal P(\Xi)\),
\[
\pi^{(j)}(dx,dy)
=
\sum_{i\neq j}w_i\delta_{X_i}(dx)\,\beta_i(dy)
+
w_j\delta_{X_j'}(dx)\,\beta_j(dy).
\]
If some atoms coincide, we split the mass according to this indexed
representation; this does not change the marginal measure or the transport
cost.

Now construct a coupling \(\pi\in\Pi(\mu_w,P)\) by keeping the same
conditional laws and replacing only the source point \(X_j'\) by \(X_j\):
\[
\pi(dx,dy)
=
\sum_{i\neq j}w_i\delta_{X_i}(dx)\,\beta_i(dy)
+
w_j\delta_{X_j}(dx)\,\beta_j(dy).
\]
This coupling is feasible for \(W_p(\mu_w,P)\). Therefore,
\[
\begin{aligned}
W_p(\mu_w,P)^p
&\le
\int_{\Xi\times\Xi}
d_{\kappa,\lambda}(x,y)^p\,d\pi(x,y) \\
&=
\sum_{i\neq j}w_i
\int_\Xi d_{\kappa,\lambda}(X_i,y)^p\,\beta_i(dy)
+
w_j
\int_\Xi d_{\kappa,\lambda}(X_j,y)^p\,\beta_j(dy).
\end{aligned}
\]
Since \(X_j,X_j'\), and \(y\) all belong to \(\Xi\), we have
\(d_{\kappa,\lambda}(X_j,y)^p\le \mathrm{diam}_{\kappa,\lambda}(\Xi)^p\), and
hence
\[
d_{\kappa,\lambda}(X_j,y)^p
\le
d_{\kappa,\lambda}(X_j',y)^p
+
\mathrm{diam}_{\kappa,\lambda}(\Xi)^p .
\]
Substituting this bound gives
\[
\begin{aligned}
W_p(\mu_w,P)^p
&\le
\sum_{i\neq j}w_i
\int_\Xi d_{\kappa,\lambda}(X_i,y)^p\,\beta_i(dy)
+
w_j
\int_\Xi d_{\kappa,\lambda}(X_j',y)^p\,\beta_j(dy) \\
&\qquad
+
w_j\,\mathrm{diam}_{\kappa,\lambda}(\Xi)^p \\
&=
\int_{\Xi\times\Xi}
d_{\kappa,\lambda}(x,y)^p\,d\pi^{(j)}(x,y)
+
w_j\,\mathrm{diam}_{\kappa,\lambda}(\Xi)^p .
\end{aligned}
\]
Since \(\pi^{(j)}\in\Pi(\mu_w^{(j)},P)\) was arbitrary, taking the infimum
over all such couplings yields
\[
W_p(\mu_w,P)^p
-
W_p(\mu_w^{(j)},P)^p
\le
w_j\,\mathrm{diam}_{\kappa,\lambda}(\Xi)^p .
\]
Exchanging the roles of \(X_j\) and \(X_j'\) gives the reverse inequality, and
therefore proves (\ref{eq:bounded-difference-wp}).

Thus \(f\) satisfies McDiarmid's bounded-differences condition with constants
\[
a_j\coloneqq w_j\,\mathrm{diam}_{\kappa,\lambda}(\Xi)^p,
\qquad j=1,\dots,N.
\]
Hence, for every \(u>0\),
\[
\mathbb P\big(f-\mathbb E[f]\ge u\big)
\le
\exp\!\left(
-\frac{2u^2}{\sum_{j=1}^N a_j^2}
\right).
\]
Since
\[
\sum_{j=1}^N a_j^2
=
\mathrm{diam}_{\kappa,\lambda}(\Xi)^{2p}
\sum_{j=1}^N w_j^2
=
\mathrm{diam}_{\kappa,\lambda}(\Xi)^{2p}
N_{\mathrm{eff}}(w)^{-1},
\]
we obtain
\[
\mathbb P\big(f-\mathbb E[f]\ge u\big)
\le
\exp\!\left(
-c_1N_{\mathrm{eff}}(w)u^2
\right),
\qquad
c_1\coloneqq 2\,\mathrm{diam}_{\kappa,\lambda}(\Xi)^{-2p}.
\]

By Lemma~\ref{lem:expectation-bound}, there exists \(c_2>0\) such that
\[
\mathbb E[W_p(\mu_w,P)^p]
\le
c_2N_{\mathrm{eff}}(w)^{-\nu}.
\]
Therefore, for any \(\varepsilon>0\),
\[
\begin{aligned}
\mathbb P\!\left(W_p(\mu_w,P)\ge \varepsilon\right)
&=
\mathbb P\!\left(W_p(\mu_w,P)^p\ge \varepsilon^p\right) \\
&\le
\mathbb P\!\left(
W_p(\mu_w,P)^p-\mathbb E[W_p(\mu_w,P)^p]
\ge
\varepsilon^p-c_2N_{\mathrm{eff}}(w)^{-\nu}
\right) \\
&\le
\exp\!\left(
-c_1N_{\mathrm{eff}}(w)
\Big(
\varepsilon^p-c_2N_{\mathrm{eff}}(w)^{-\nu}
\Big)_+^2
\right).
\end{aligned}
\]
This proves the first claim.

For the simplified bound, suppose \(\varepsilon \ge 2(c_2N_{\mathrm{eff}}(w)^{-\nu})^{1/p}\).
Then \(c_2N_{\mathrm{eff}}(w)^{-\nu} \le 2^{-p}\varepsilon^p\).
Since \(p\in(1,\infty)\), we have \(1-2^{-p}\ge 1/2\), and therefore
\[
\Big(\varepsilon^p-c_2N_{\mathrm{eff}}(w)^{-\nu}\Big)_+
\ge
\frac12\varepsilon^p.
\]
Substituting this into the first concentration bound gives
\[
\mathbb P\!\left(
W_p(\mu_w,P)\ge \varepsilon
\right)
\le
\exp\!\left(
-\frac{c_1}{4}N_{\mathrm{eff}}(w)\varepsilon^{2p}
\right),
\]
which completes the proof.

<!-- proposition-3: body_md -->
\label{prop:weighted-coupling}
Let $p\in(1,\infty)$, $\mu_w=\sum_{n=1}^N w_n\delta_{X_n}$, and $\widehat\mu_w=\sum_{n=1}^N w_n\delta_{\widehat X_n}$.
Then,
\begin{equation}
\label{eq:weighted-coupling}
W_p(\mu_w,\widehat\mu_w)^p
\le
\sum_{n=1}^N w_n\, d_{\kappa,\lambda}(X_n,\widehat X_n)^p.
\end{equation}
<!-- end proposition-3 -->
<!-- proof-4: body_md -->
We construct an explicit coupling by assigning mass \(w_n\) to the pair
\((X_n,\widehat X_n)\) for each \(n=1,\dots,N\), namely
\(\pi_w \coloneqq \sum_{n=1}^N w_n\,\delta_{(X_n,\widehat X_n)}\). The first marginal of \(\pi_w\) is \(\mu_w\), and its second marginal is
\(\widehat\mu_w\). Hence \(\pi_w\in\Pi(\mu_w,\widehat\mu_w)\).

The \(p\)-Wasserstein distance on \((\Xi,d_{\kappa,\lambda})\) is defined by
\[
W_p(\mu_w,\widehat\mu_w)^p
=
\inf_{\pi\in\Pi(\mu_w,\widehat\mu_w)}
\int_{\Xi\times\Xi}
d_{\kappa,\lambda}(x,\widehat x)^p\,d\pi(x,\widehat x).
\]
Since \(\pi_w\) is an admissible coupling, evaluating the transport cost at
\(\pi_w\) gives the upper bound
\[
W_p(\mu_w,\widehat\mu_w)^p
\le
\int_{\Xi\times\Xi}
d_{\kappa,\lambda}(x,\widehat x)^p\,d\pi_w(x,\widehat x).
\]
Finally, substituting the definition of \(\pi_w\) 
\[
\int_{\Xi\times\Xi}
d_{\kappa,\lambda}(x,\widehat x)^p\,d\pi_w(x,\widehat x)
=
\sum_{n=1}^N
w_n\,d_{\kappa,\lambda}(X_n,\widehat X_n)^p.
\]
We therefore obtain the desired bound,
\[
W_p(\mu_w,\widehat\mu_w)^p
\le
\sum_{n=1}^N
w_n\,d_{\kappa,\lambda}(X_n,\widehat X_n)^p.
\]

<!-- lemma-2: body_md -->
\label{lem:drift-aware-wasserstein-bound}
Suppose Assumption~\ref{ass:nonstationary-drift} holds, and let \(p\in(1,\infty)\) and \(w\in\Delta_N\).
Then there exist coupled pairs \((X_n,\widetilde X_n)\), independent across
\(n\), with \(X_n\sim P_n\) and \(\widetilde X_n\sim P_{N+1}\), such that, for
\(\mu_w\coloneqq\sum_{n=1}^N w_n\delta_{X_n}\) and
\(\widetilde\mu_w\coloneqq\sum_{n=1}^N w_n\delta_{\widetilde X_n}\),
\[
W_p(\mu_w,P_{N+1})
\le
W_p(\widetilde\mu_w,P_{N+1})+D_p(w)\varrho
\qquad \text{a.s.}
\]
In particular, \(\widetilde X_1,\dots,\widetilde X_N\) are i.i.d.~with law
\(P_{N+1}\).

<!-- proof-5: body_md -->
By Assumption~\ref{ass:nonstationary-drift}, for each
\(k=1,\dots,N\) there is a coupling of \(P_k\) and \(P_{k+1}\) whose \(d_{\kappa,\lambda}\)-cost is at most \(\varrho\) almost surely. Gluing these couplings from time \(n\) to time \(N+1\) yields a coupling of \(P_n\) and \(P_{N+1}\) under which
\[
d_{\kappa,\lambda}(X_n,\widetilde X_n)
\le
\sum_{k=n}^N \varrho
=
(N-n+1)\varrho
\qquad \text{a.s.}
\]
Choosing these couplings independently across \(n\) gives the coupled pairs in the statement. Since the couplings are independent across \(n\), and each proxy variable has marginal law \(P_{N+1}\), the variables \(\widetilde X_1,\dots,\widetilde X_N\) are i.i.d.~with law \(P_{N+1}\).

We compare \(\mu_w\) with the proxy measure \(\widetilde\mu_w\) using the triangle inequality. By the triangle inequality for \(W_p\),
\[
W_p(\mu_w,P_{N+1})
\le
W_p(\widetilde\mu_w,P_{N+1})
+
W_p(\mu_w,\widetilde\mu_w).
\]
The second term is controlled by the weighted coupling bound in Proposition~\ref{prop:weighted-coupling}. Applying it to the paired samples \((X_n,\widetilde X_n)\) yields
\[
W_p(\mu_w,\widetilde\mu_w)^p
\le
\sum_{n=1}^N w_n d_{\kappa,\lambda}(X_n,\widetilde X_n)^p .
\]
Using the pairwise drift bound and the definition of \(D_p(w)\), we obtain
\[
W_p(\mu_w,\widetilde\mu_w)^p
\le
\varrho^p\sum_{n=1}^N w_n (N-n+1)^p
=
\varrho^p D_p(w)^p .
\]
Hence \(W_p(\mu_w,\widetilde\mu_w)\le D_p(w)\varrho\). Substituting this into the previous triangle inequality gives
\[
W_p(\mu_w,P_{N+1})
\le
W_p(\widetilde\mu_w,P_{N+1})
+
D_p(w)\varrho
\qquad \text{a.s.}
\]

<!-- theorem-4: body_md -->
\label{thm:dlb-coverage}
Let \(p\in(1,\infty)\) and suppose Assumptions~\ref{ass:path-space} and
\ref{ass:nonstationary-drift} hold. Then there exist constants
\(c_1,c_2>0\) and an exponent \(\nu\in(0,1/2)\) such that, for every
\(w\in\Delta_N\) and every \(\varepsilon>0\),
\begin{equation}
\label{eq:dlb-bound}
\mathbb P\!\left(W_p(\mu_w,P_{N+1}) \ge \varepsilon\right)
\le
\exp\!\left(
-c_1 N_{\mathrm{eff}}(w)
\Big(
\big((\varepsilon-D_p(w)\varrho)_+^p
-
c_2N_{\mathrm{eff}}(w)^{-\nu}\big)_+
\Big)^2
\right).
\end{equation}
In particular, if \(\varepsilon
\ge D_p(w)\varrho + 2\big(c_2N_{\mathrm{eff}}(w)^{-\nu}\big)^{1/p}\),
then
\begin{equation}
\label{eq:dlb-bound-tail}
\mathbb P\!\left(W_p(\mu_w,P_{N+1}) \ge \varepsilon\right)
\le
\exp\!\left(
-\frac{c_1}{4}
N_{\mathrm{eff}}(w)
(\varepsilon-D_p(w)\varrho)_+^{2p}
\right).
\end{equation}

<!-- proof-6: body_md -->
We compare the nonstationary weighted empirical measure \(\mu_w\) to a proxy
weighted empirical measure drawn from the target distribution \(P_{N+1}\). By
Lemma~\ref{lem:drift-aware-wasserstein-bound}, we may construct coupled pairs
\((X_n,\widetilde X_n)\), independent across \(n\), with \(X_n\sim P_n\) and
\(\widetilde X_n\sim P_{N+1}\), such that, for
\(\widetilde\mu_w\coloneqq\sum_{n=1}^N w_n\delta_{\widetilde X_n}\),
\[
W_p(\mu_w,P_{N+1})
\le
W_p(\widetilde\mu_w,P_{N+1})+D_p(w)\varrho
\qquad \text{a.s.}
\]
Consequently, for any \(\varepsilon>0\),
\[
\mathbb P\!\left(W_p(\mu_w,P_{N+1})\ge \varepsilon\right)
\le
\mathbb P\!\left(
W_p(\widetilde\mu_w,P_{N+1})
\ge
(\varepsilon-D_p(w)\varrho)_+
\right).
\]

The proxy variables \(\widetilde X_1,\dots,\widetilde X_N\) are i.i.d.~with law
\(P_{N+1}\). Hence Proposition~\ref{prop:concentration-bound}, applied with
\(P=P_{N+1}\) and with \(\mu_w\) replaced by \(\widetilde\mu_w\), gives, for
all \(t>0\),
\[
\mathbb P\!\left(W_p(\widetilde\mu_w,P_{N+1})\ge t\right)
\le
\exp\!\left(
-c_1N_{\mathrm{eff}}(w)
\big(t^p-c_2N_{\mathrm{eff}}(w)^{-\nu}\big)_+^2
\right).
\]
If \(t=(\varepsilon-D_p(w)\varrho)_+=0\), the desired bound is trivial because
the right-hand side of (\ref{eq:dlb-bound}) equals one. Otherwise, substituting
\(t=(\varepsilon-D_p(w)\varrho)_+\) into the preceding display yields
\[
\mathbb P\!\left(W_p(\mu_w,P_{N+1})\ge \varepsilon\right)
\le
\exp\!\left(
-c_1 N_{\mathrm{eff}}(w)
\Big(
\big((\varepsilon-D_p(w)\varrho)_+^p
-
c_2N_{\mathrm{eff}}(w)^{-\nu}\big)_+
\Big)^2
\right),
\]
which proves (\ref{eq:dlb-bound}).

The simplified tail bound follows by lower-bounding the residual radius. Suppose that
\[
\varepsilon \ge D_p(w)\varrho
+
2(c_2N_{\mathrm{eff}}(w)^{-\nu})^{1/p}.
\]
Then
\[
c_2N_{\mathrm{eff}}(w)^{-\nu}
\le
2^{-p}(\varepsilon-D_p(w)\varrho)_+^p .
\]
Since \(p>1\), we have \(1-2^{-p}\ge 1/2\), and therefore
\[
\Big(
(\varepsilon-D_p(w)\varrho)_+^p
-
c_2N_{\mathrm{eff}}(w)^{-\nu}
\Big)_+
\ge
\frac12(\varepsilon-D_p(w)\varrho)_+^p .
\]
Substituting this lower bound into (\ref{eq:dlb-bound}) gives
\[
\mathbb P\!\left(W_p(\mu_w,P_{N+1}) \ge \varepsilon\right)
\le
\exp\!\left(
-\frac{c_1}{4}
N_{\mathrm{eff}}(w)
(\varepsilon-D_p(w)\varrho)_+^{2p}
\right),
\]
which proves (\ref{eq:dlb-bound-tail}).

<!-- proposition-4: body_md -->
\label{prop:two_budget_exact_reduction}
For \(p\in(1,\infty)\), define
\[
Q_\phi(\tau)
\coloneqq
\Big\{
 a\in[0,\infty)^N:
 \sum_{n=1}^N w_n a_n = 1,
 \ \sum_{n=1}^N w_n\phi(a_n)\le \tau
\Big\},
\]
and for $a\in Q_\phi(\tau)$ define
\[
T_a(\varepsilon)
\coloneqq
\sup_{\nu_1',\dots,\nu_N'\in\mathcal P(\Xi)}
\Bigg\{
\sum_{n=1}^N w_n a_n\int_\Xi \ell_{\mathrm{DH}}(\vartheta;\widehat{x})\,\nu_n'(d\widehat{x})
:
\sum_{n=1}^N w_n a_n\int_\Xi d_{\kappa,\lambda}(X_n,\widehat{x})^p\,\nu_n'(d\widehat{x})\le \varepsilon^p
\Bigg\}.
\]
The robust value admits the exact reduction
\begin{equation}
\label{eq:two_budget_exact_reduction}
V_{\vartheta}(\varepsilon,\tau)
=
\sup_{a\in Q_\phi(\tau)} T_a(\varepsilon).
\end{equation}

<!-- proof-8: body_md -->
We prove the claim in two steps. First, we show that any feasible
\(\eta\in\mathcal B_{\varepsilon,\tau}\) admits a representation
\(\eta=\sum_{n=1}^N w_n a_n\nu_n'\), where \(a\) satisfies the
\(\phi\)-divergence constraint and the conditional laws \(\nu_n'\) satisfy the
transport constraint. Second, we show that any such representation induces a
feasible law in \(\mathcal B_{\varepsilon,\tau}\).

Fix any feasible $\eta \in \mathcal{B}_{\varepsilon,\tau}$. By definition, there exist couplings $\pi,\pi' \in \mathcal P(\Xi \times \Xi)$ such that
\[
\pi_1=\mu_w,
\qquad
\pi'_2=\eta,
\qquad
\int_{\Xi\times\Xi} d_{\kappa,\lambda}(x,\widehat{x})^p\,d\pi'(x,\widehat{x})\le \varepsilon^p,
\qquad
D_\phi(\pi'\,\|\,\pi)\le \tau.
\]
Since \(\pi_1=\mu_w=\sum_{n=1}^N w_n\delta_{X_n}\), the first coordinate has support contained in the finite set \(\{X_1,\dots,X_N\}\). Hence \(\pi\) admits the disintegration
\[
\pi(dx,d\widehat{x})
=
\sum_{n=1}^N w_n\,\delta_{X_n}(dx)\,\nu_n(d\widehat{x}),
\]
where $\nu_n \in \mathcal P(\Xi)$ are the reference conditional laws.

Because $D_\phi(\pi'\,\|\,\pi)<\infty$, we necessarily have $\pi'\ll\pi$. Let $W=d\pi'/d\pi$ denote the Radon-Nikodym derivative, and define its slices as
\[
W_n(\widehat{x})\coloneqq W(X_n,\widehat{x}),
\qquad
a_n\coloneqq\int_\Xi W_n(\widehat{x})\,\nu_n(d\widehat{x}).
\]
The quantity $a_n$ represents the total mass assigned by $\pi'$ to the slice $\{x=X_n\}$ relative to $\pi$. For each $n$ such that $a_n > 0$, define the normalized conditional probability measure $\nu_n'$ by
\[
\nu_n'(d\widehat{x})\coloneqq\frac{W_n(\widehat{x})}{a_n}\,\nu_n(d\widehat{x}),
\]
and choose an arbitrary probability measure $\nu_n'\in\mathcal P(\Xi)$ when $a_n=0$. This construction cleanly separates the distorted coupling into a mass component $a_n$ and a normalized conditional slice $\nu_n'$. 

The distorted coupling therefore factors as:
\[
\pi'(dx,d\widehat{x})
=
\sum_{n=1}^N w_n a_n\,\delta_{X_n}(dx)\,\nu_n'(d\widehat{x}).
\]
Taking the second marginal (integrating out $dx$) directly yields:
\[
\pi'_2
=
\eta
=
\sum_{n=1}^N w_n a_n\nu_n',
\qquad\text{with}\quad
\sum_{n=1}^N w_n a_n=1,
\]
thus any feasible \(\eta\) can be written as a finite mixture with weights \(w_n a_n\). Consequently, its expected loss equals the objective appearing in
\(T_a(\varepsilon)\):
\[
\mathbb E_{\widehat{X}\sim\eta}\big[\ell_{\mathrm{DH}}(\vartheta;\widehat{X})\big]
=
\sum_{n=1}^N w_n a_n\int_\Xi \ell_{\mathrm{DH}}(\vartheta;\widehat{x})\,\nu_n'(d\widehat{x}).
\]
Moreover, the transport constraint is preserved under this representation:
\[
\int_{\Xi\times\Xi} d_{\kappa,\lambda}(x,\widehat{x})^p\,d\pi'(x,\widehat{x})
=
\sum_{n=1}^N w_n a_n\int_\Xi d_{\kappa,\lambda}(X_n,\widehat{x})^p\,\nu_n'(d\widehat{x})
\le \varepsilon^p.
\]
Finally, by Jensen's inequality applied to the convex function $\phi$,
\begin{align*}
D_\phi(\pi'\,\|\,\pi)
&=
\sum_{n=1}^N w_n\int_\Xi \phi\big(W_n(\widehat{x})\big)\,\nu_n(d\widehat{x})\\
&\ge
\sum_{n=1}^N w_n\phi\!\left(\int_\Xi W_n(\widehat{x})\,\nu_n(d\widehat{x})\right)
=
\sum_{n=1}^N w_n\phi(a_n),
\end{align*}
so $a\in Q_\phi(\tau)$ since $D_\phi(\pi'\,\|\,\pi)\leq \tau$. Thus, for this specific $a$, the conditional slices $(\nu_1',\dots,\nu_N')$ are feasible for the inner supremum $T_a(\varepsilon)$, attaining the exact same objective value. This proves
\[
V_{\vartheta}(\varepsilon,\tau)
\le
\sup_{a\in Q_\phi(\tau)}T_a(\varepsilon).
\]

Conversely, fix any $a\in Q_\phi(\tau)$ and any $\nu_1',\dots,\nu_N'\in\mathcal P(\Xi)$ feasible in the definition of $T_a(\varepsilon)$. Set
\[
\pi(dx,d\widehat{x})
\coloneqq
\sum_{n=1}^N w_n\,\delta_{X_n}(dx)\,\nu_n'(d\widehat{x}),
\qquad
\pi'(dx,d\widehat{x})
\coloneqq
\sum_{n=1}^N w_n a_n\,\delta_{X_n}(dx)\,\nu_n'(d\widehat{x}).
\]
Then $\pi_1=\mu_w$, $\pi'_2=\sum_{n=1}^N w_n a_n\nu_n'$, and the transport cost evaluates to:
\[
\int_{\Xi\times\Xi} d_{\kappa,\lambda}(x,\widehat{x})^p\,d\pi'(x,\widehat{x})
=
\sum_{n=1}^N w_n a_n\int_\Xi d_{\kappa,\lambda}(X_n,\widehat{x})^p\,\nu_n'(d\widehat{x})
\le \varepsilon^p.
\]
Moreover, $\pi'\ll\pi$.  Where several components overlap,
\(d\pi'/d\pi\) is a weighted average of the corresponding \(a_n\)'s, so applying Jensen's inequality with the convexity of $\phi$ yields
\[
D_\phi(\pi'\,\|\,\pi)
\le
\sum_{n=1}^N w_n\phi(a_n)
\le \tau.
\]
Hence $\eta\coloneqq\sum_{n=1}^N w_n a_n\nu_n'$ belongs to $\mathcal B_{\varepsilon,\tau}$ and attains exactly the same objective value as in $T_a(\varepsilon)$. Taking the supremum over all $a \in Q_\phi(\tau)$ and their corresponding feasible conditional slices $(\nu_1', \dots, \nu_N')$ proves the reverse inequality
\[
V_{\vartheta}(\varepsilon,\tau)
\ge
\sup_{a\in Q_\phi(\tau)}T_a(\varepsilon).
\]

\label{lem:uniform_ot_reweighted}
Let \(p\in(1,\infty)\) and suppose Assumption~\ref{ass:loss-regularity} holds, with
\(\|\cdot\|_{\kappa,\lambda,p}^{\ast}\) and \(h\) as in
Proposition~\ref{prop:dual-geometry-main}. For \(\delta\in(0,1)\), define
\begin{equation}

A_\delta
\coloneqq
\Big\{

\sum_{n=1}^N w_n a_n=1,\

\Big\}.

\coloneqq

-

-

\right|

\qquad \text{as } \varepsilon \downarrow 0.

<!-- proof-9: body_md -->
We establish matching upper and lower bounds uniformly over \(a\in A_\delta\). The upper bound follows from a local linearization of \(\ell_{\mathrm{DH}}\) around the nominal anchors; the lower bound uses an explicit deterministic perturbation along the dual maximizing directions.


Let
\[
B_\ell \coloneqq
\sup_{\widehat x\in\Xi}
|\ell_{\mathrm{DH}}(\vartheta;\widehat x)|<\infty
\]
and fix \(\zeta>0\). By the local expansion in
Assumption~\ref{ass:loss-regularity} and the finiteness of \(N\), there exists
\(r_\zeta>0\) such that, for every \(n=1,\dots,N\),
\begin{equation}
\label{eq:linearization_error}
\big|\ell_{\mathrm{DH}}(\vartheta;\widehat{x}) - L_n(\vartheta) - \langle g_n(\vartheta), \widehat{x}-X_n\rangle\big| \le \zeta d_{\kappa,\lambda}(X_n, \widehat{x})
\end{equation}
for all $\widehat{x} \in \Xi$ with $d_{\kappa,\lambda}(X_n, \widehat{x}) \le r_\zeta$.

To prove the upper bound, fix $a \in A_\delta$ and let $(\nu_1', \dots, \nu_N')$ be any feasible family of conditional slices for the inner problem $T_a(\varepsilon)$. Define the local neighborhood $B_n \coloneqq \{\widehat{x} \in \Xi : d_{\kappa,\lambda}(X_n, \widehat{x}) \le r_\zeta\}$. Since $\Xi = B_n \cup B_n^c$, we decompose the expected loss as 
\[
\int_\Xi \ell_{\mathrm{DH}}(\vartheta; \widehat{x}) \, \nu_n'(d\widehat{x}) = \int_{B_n} \ell_{\mathrm{DH}}(\vartheta; \widehat{x}) \, \nu_n'(d\widehat{x}) + \int_{B_n^c} \ell_{\mathrm{DH}}(\vartheta; \widehat{x}) \, \nu_n'(d\widehat{x}).
\]
On $B_n$, the linearization gives $\ell_{\mathrm{DH}}(\vartheta;\widehat{x}) \le L_n(\vartheta) + \langle g_n(\vartheta), \widehat{x} - X_n\rangle + \zeta d_{\kappa,\lambda}(X_n, \widehat{x})$, while on $B_n^c$, global boundedness implies $\ell_{\mathrm{DH}}(\vartheta;\widehat{x}) - L_n(\vartheta) \le 2B_\ell$.

Integrating these bounds and utilizing the identity $L_n(\vartheta) \nu_n'(B_n) + L_n(\vartheta) \nu_n'(B_n^c) = L_n(\vartheta)$, we obtain
\[
\int_\Xi \ell_{\mathrm{DH}}(\vartheta; \widehat{x})\,\nu_n'(d\widehat{x}) \le L_n(\vartheta) + \int_{B_n}\langle g_n(\vartheta), \widehat{x} - X_n\rangle\,\nu_n'(d\widehat{x}) + \zeta \int_{B_n} d_{\kappa,\lambda}(X_n, \widehat{x})\,\nu_n'(d\widehat{x}) + 2B_\ell\nu_n'(B_n^c).
\]
By the definition of the dual norm, $\langle g_n(\vartheta), \widehat{x} - X_n\rangle \le \|g_n(\vartheta)\|_{\kappa,\lambda,p}^* d_{\kappa,\lambda}(X_n,\widehat{x})$. Since this upper bound is non-negative, we can extend the integration domain from $B_n$ to the entire space $\Xi$ by monotonicity. Taking the weighted sum with $\widehat w_n$ yields the nominal reweighted loss alongside three distinct error terms: a gradient bound ($I_{\mathrm{grad}}$), a linearization error ($I_{\mathrm{lin}}$), and a tail bound ($I_{\mathrm{tail}}$):
\begin{align*}
&\sum_{n=1}^N \widehat w_n
\int_\Xi \ell_{\mathrm{DH}}(\vartheta;\widehat x)\,
\nu_n'(d\widehat x) \\
&\le
\sum_{n=1}^N \widehat w_n L_n(\vartheta)
+ \underbrace{
\sum_{n=1}^N \widehat w_n
\int_\Xi
\|g_n(\vartheta)\|_{\kappa,\lambda,p}^*
d_{\kappa,\lambda}(X_n,\widehat x)\,
\nu_n'(d\widehat x)
}_{I_{\mathrm{grad}}}  \\
&\quad
+ \underbrace{\zeta
\sum_{n=1}^N \widehat w_n
\int_\Xi d_{\kappa,\lambda}(X_n,\widehat x)\,
\nu_n'(d\widehat x)
}_{I_{\mathrm{lin}}}
+
\underbrace{
2B_\ell\sum_{n=1}^N \widehat w_n\nu_n'(B_n^c)
}_{I_{\mathrm{tail}}}.
\end{align*}
For the gradient bound $I_{\mathrm{grad}}$, H\"older's inequality and the feasibility of $(\nu_n')$ imply that  $\sum \widehat w_n \int_\Xi d_{\kappa,\lambda}(X_n, \widehat{x})^p \, \nu_n'(d\widehat{x}) \le \varepsilon^p$, hence we have
\begin{align*}
I_{\mathrm{grad}}
&\le
\left(
\sum_{n=1}^N \widehat w_n
\|g_n(\vartheta)\|_{\kappa,\lambda,p}^{*\,q}
\right)^{1/q}
\left(
\sum_{n=1}^N \widehat w_n
\int_\Xi d_{\kappa,\lambda}(X_n,\widehat x)^p\,\nu_n'(d\widehat x)
\right)^{1/p}  \\
&\le
\Upsilon_{\widehat w}(\vartheta)\varepsilon.
\end{align*}
Similarly, the linearization error satisfies $
I_{\mathrm{lin}}
\le
\zeta\varepsilon$.
Finally, since \(d_{\kappa,\lambda}(X_n,\widehat x)>r_\zeta\) on \(B_n^c\), Markov's inequality gives
\[
\nu_n'(B_n^c)
\le
r_\zeta^{-p}
\int_\Xi d_{\kappa,\lambda}(X_n,\widehat x)^p\,\nu_n'(d\widehat x),
\]
and therefore
\[
I_{\mathrm{tail}}
\le
2B_\ell r_\zeta^{-p}\varepsilon^p.
\]

Substituting these three bounds and taking the supremum over all feasible slices yields the uniform upper bound:
\begin{equation}
\label{eq:uniform_ot_upper}
T_a(\varepsilon) \le \sum_{n=1}^N \widehat w_n L_n(\vartheta) + \varepsilon\Upsilon_{\widehat w}(\vartheta) + \zeta\varepsilon + 2B_\ell r_\zeta^{-p}\varepsilon ^p.
\end{equation}

We now construct a matching lower bound by evaluating a specific discrete attack. If $\Upsilon_w(\vartheta) = 0$, then $\|g_n(\vartheta)\|_{\kappa,\lambda,p}^{\ast\,q}=0$ for every $n$ (since $w_n>0$), meaning $\Upsilon_{\widehat w}(\vartheta)=0$ for all $a\in A_\delta$. Setting $\nu_n'=\delta_{X_n}$ yields $T_a(\varepsilon)\ge \sum_{n=1}^N \widehat w_n L_n(\vartheta)$, which is sufficient. We therefore assume $\Upsilon_w(\vartheta) > 0$.

For every $a\in A_\delta$, the constraint $\|a-\mathbf{1}\|_\infty \le \delta$ implies $a_n \ge 1-\delta$. Thus, $\Upsilon_{\widehat w}(\vartheta)^q = \sum_{n=1}^N w_n a_n \|g_n(\vartheta)\|_{\kappa,\lambda,p}^{\ast\,q} \ge (1-\delta)\sum_{n=1}^N w_n \|g_n(\vartheta)\|_{\kappa,\lambda,p}^{\ast\,q} = (1- \delta)\Upsilon_w(\vartheta)^q$, meaning $\Upsilon_{\widehat w}(\vartheta)$ is uniformly bounded away from $0$. Define the deterministic adversarial perturbations:
\[
\widehat X_n^a \coloneqq X_n + \varepsilon \, h(g_n(\vartheta)) \, \|g_n(\vartheta)\|_{\kappa,\lambda,p}^{*\,(q-1)} \, \Upsilon_{\widehat w}(\vartheta)^{1-q}.
\]
The transport distance is $d_{\kappa,\lambda}(X_n,\widehat X_n^a) = \varepsilon\, \|g_n(\vartheta)\|_{\kappa,\lambda,p}^{*\,(q-1)} \Upsilon_{\widehat w}(\vartheta)^{1-q}$ using that $h(g_n(\vartheta))$ is a unit vector. Recalling that $(q-1)p=q$, the total transport cost evaluates exactly to the budget:
\[
\sum_{n=1}^N \widehat w_n d_{\kappa,\lambda}(X_n,\widehat X_n^a)^p = \varepsilon^p\,\Upsilon_{\widehat w}(\vartheta)^{-q} \sum_{n=1}^N \widehat w_n \|g_n(\vartheta)\|_{\kappa,\lambda,p}^{\ast\,q} = \varepsilon^p\,\Upsilon_{\widehat w}(\vartheta)^{-q} \Upsilon_{\widehat w}(\vartheta)^q = \varepsilon^p.
\]
Hence, the discrete point-mass choice $\nu_n'=\delta_{\widehat X_n^a}$ is feasible for $T_a(\varepsilon)$. Furthermore, $d_{\kappa,\lambda}(X_n,\widehat X_n^a)\to 0$ uniformly in $a\in A_\delta$ as $\varepsilon\downarrow0$, and by Assumption~\ref{ass:loss-regularity} these perturbations remain in $\Xi$.

Using the linearization error \hyperref[eq:linearization_error]{(\ref*{eq:linearization_error})} at the perturbed points yields:
\[
\ell_{\mathrm{DH}}(\vartheta; \widehat X_n^a) \ge L_n(\vartheta) + \langle g_n(\vartheta),\widehat X_n^a-X_n\rangle - \zeta d_{\kappa,\lambda}(X_n,\widehat X_n^a).
\]
Multiplying by $\widehat w_n$ and summing over $n$, we explicitly evaluate the gradient inner product. We proceed in three steps: first, we substitute the definition of the perturbation $\widehat X_n^a$; second, we apply the dual maximizer property $\langle g_n(\vartheta), h(g_n(\vartheta)) \rangle = \|g_n(\vartheta)\|_{\kappa,\lambda,p}^*$; and third, we recognize the definition of $\Upsilon_{\widehat w}(\vartheta)^q = \sum_{n=1}^N \widehat w_n \|g_n(\vartheta)\|_{\kappa,\lambda,p}^{\ast\,q}$. This yields:
\begin{align*}
\sum_{n=1}^N \widehat w_n \langle g_n(\vartheta),\widehat X_n^a-X_n\rangle 
&= \sum_{n=1}^N \widehat w_n \Big\langle g_n(\vartheta), \, \varepsilon h(g_n(\vartheta)) \|g_n(\vartheta)\|_{\kappa,\lambda,p}^{*\,(q-1)} \Upsilon_{\widehat w}(\vartheta)^{1-q} \Big\rangle \\
&= \varepsilon \Upsilon_{\widehat w}(\vartheta)^{1-q} \sum_{n=1}^N \widehat w_n \langle g_n(\vartheta), h(g_n(\vartheta)) \rangle \|g_n(\vartheta)\|_{\kappa,\lambda,p}^{*\,(q-1)} \\
&= \varepsilon \Upsilon_{\widehat w}(\vartheta)^{1-q} \sum_{n=1}^N \widehat w_n \|g_n(\vartheta)\|_{\kappa,\lambda,p}^{*\,q} \\
&= \varepsilon \Upsilon_{\widehat w}(\vartheta)^{1-q} \Upsilon_{\widehat w}(\vartheta)^q \\
&= \varepsilon \Upsilon_{\widehat w}(\vartheta).
\end{align*}
Substituting this into the aggregated linear bound directly confirms the lower bound:
\begin{align*}
T_a(\varepsilon) &\ge \sum_{n=1}^N \widehat w_nL_n(\vartheta) + \varepsilon \Upsilon_{\widehat w}(\vartheta) - \zeta \sum_{n=1}^N \widehat w_n d_{\kappa,\lambda}(X_n,\widehat X_n^a)\\
&\ge \sum_{n=1}^N \widehat w_nL_n(\vartheta) + \varepsilon\Upsilon_{\widehat w}(\vartheta) - \zeta\varepsilon,
\end{align*}
uniformly over $a\in A_\delta$. Combining this with (\ref{eq:uniform_ot_upper}) and noting $\varepsilon^p=o(\varepsilon)$ as $\varepsilon \downarrow 0$ proves (\ref{eq:uniform_ot_reweighted}).

Let \(\phi\) be a locally regular entropy function. Then, as
\(\tau\downarrow0\), the reweighting problem admits the expansion
\begin{equation}
\label{eq:phi_reweighting_expansion}

We prove the expansion by establishing matching upper and lower bounds in the small-budget regime. The key idea is that for sufficiently small $\tau$, every feasible reweighting stays strictly close to the uniform baseline $a = \mathbf{1}$. This forces the divergence constraint into a local regime where $\phi$ is regular and reduces the optimization to maximizing a linear objective subject to a weighted variance constraint.

If $\sigma_w(\vartheta)=0$, then $L_n(\vartheta) = \bar L_w(\vartheta)$ for all $n$ (since $w_n>0$). For any feasible $a \in Q_\phi(\tau)$, the expected loss is exactly $\sum_{n=1}^N w_n a_n L_n(\vartheta) = \bar L_w(\vartheta) \sum_{n=1}^N w_n a_n = \bar L_w(\vartheta)$, yielding $R_\phi(\tau)=\bar L_w(\vartheta)$ for all $\tau \ge 0$, which trivially satisfies the expansion. We therefore assume $\sigma_w(\vartheta)>0$.

\noindent Define the centered losses $Y_n(\vartheta) \coloneqq L_n(\vartheta)-\bar L_w(\vartheta)$. By definition of the weighted empirical mean and variance, they satisfy $\sum_{n=1}^N w_n Y_n(\vartheta) = 0$ and $\sum_{n=1}^N w_n Y_n(\vartheta)^2 = \sigma_w(\vartheta)^2$.

Fix $0<\zeta<\phi''(1)/2$. Because \(\phi\) is locally regular, \(1\) is its unique minimizer and
\(\phi'(1)=0\). Since $\phi$ is twice continuously differentiable near $1$, a second-order Taylor expansion yields $\phi(1+s) = \frac{1}{2}\phi''(1)s^2 + o(s^2)$. Thus, there exists $\delta_\zeta\in(0,1)$ such that for all $|s|\le \delta_\zeta$, the divergence is locally bounded by quadratics:
\begin{equation}
\label{eq:phi_local_quadratic_bounds}
\left(\frac{\phi''(1)}{2}-\zeta\right)s^2
\le \phi(1+s)
\le \left(\frac{\phi''(1)}{2}+\zeta\right)s^2.
\end{equation}

For this quadratic bound to be useful, we must guarantee that small budgets restrict all feasible weights to this neighborhood. For any feasible $a \in Q_\phi(\tau)$, the constraints $a_n \ge 0$ and $\sum_{n=1}^N w_n a_n = 1$ strictly bound the weights: $a_n \le 1/\min_k w_k$. Define the compact set $K_{\delta_\zeta} \coloneqq [0, 1/\min_k w_k] \setminus (1-\delta_\zeta, 1+\delta_\zeta)$. Since $1 \notin K_{\delta_\zeta}$ and $\phi$ is strictly positive away from $1$, its minimum on this set is strictly positive: $m_\zeta \coloneqq \min_{s\in K_{\delta_\zeta}} \phi(s)>0$. If any weight strays such that $|a_n-1|\ge \delta_\zeta$, then $a_n \in K_{\delta_\zeta}$, forcing $\tau \ge \sum_{k=1}^N w_k\phi(a_k) \ge w_n\phi(a_n) \ge (\min_k w_k) m_\zeta$. Consequently, for all $\tau < (\min_k w_k)m_\zeta$, every feasible $a \in Q_\phi(\tau)$ must satisfy $|a_n-1|<\delta_\zeta$ for all $n$.

We next bound the objective. For any feasible $a$, define the deviations $u_n \coloneqq a_n-1$. Since $\sum_{n=1}^N w_n u_n = 0$, the objective simplifies to $\sum_{n=1}^N w_n a_n L_n(\vartheta) = \sum_{n=1}^N w_n (1+u_n)(\bar L_w(\vartheta)+Y_n(\vartheta)) = \bar L_w(\vartheta) + \sum_{n=1}^N w_n u_n Y_n(\vartheta)$. For sufficiently small $\tau$, we know $|u_n|<\delta_\zeta$. Applying the lower bound in (\ref{eq:phi_local_quadratic_bounds}) with $s=u_n$ bounds the variance of the weights:
\[
\tau
\ge \sum_{n=1}^N w_n\phi(1+u_n)
\ge \left(\frac{\phi''(1)}{2}-\zeta\right)\sum_{n=1}^N w_n u_n^2.
\]
Applying the Cauchy--Schwarz inequality to the objective deviation yields:
\[
\sum_{n=1}^N w_n u_n Y_n(\vartheta) \le \left(\sum_{n=1}^N w_n u_n^2\right)^{1/2} \left(\sum_{n=1}^N w_n Y_n(\vartheta)^2\right)^{1/2} \le \sqrt{\frac{\tau}{\phi''(1)/2-\zeta}}\,\sigma_w(\vartheta).
\]
Taking the supremum over all feasible $a$ gives the uniform upper bound:
\begin{equation}
\label{eq:phi_upper_local}
R_\phi(\tau) \le \bar L_w(\vartheta) + \frac{\sigma_w(\vartheta)}{\sqrt{\phi''(1)/2-\zeta}}\sqrt{\tau}.
\end{equation}

For the matching lower bound, we construct an explicit feasible reweighting. Define the weights $a_n \coloneqq 1 + \alpha_\tau Y_n(\vartheta)$, where $\alpha_\tau \coloneqq \sqrt{\tau} \big/ \big(\sigma_w(\vartheta)\sqrt{\phi''(1)/2+\zeta}\big)$. By construction, $\sum_{n=1}^N w_n a_n=1$. For sufficiently small $\tau$, we have $|\alpha_\tau Y_n(\vartheta)| \le \delta_\zeta$, allowing us to use the upper bound in (\ref{eq:phi_local_quadratic_bounds}) to verify feasibility:
\[
\sum_{n=1}^N w_n\phi(a_n)
\le
\left(\frac{\phi''(1)}{2}+\zeta\right)\alpha_\tau^2 \sum_{n=1}^N w_n Y_n(\vartheta)^2
=
\left(\frac{\phi''(1)}{2}+\zeta\right)\alpha_\tau^2 \sigma_w(\vartheta)^2
=
\tau,
\]
confirming $a \in Q_\phi(\tau)$. Evaluating the expected loss under $a$ yields $\sum_{n=1}^N w_n a_n L_n(\vartheta) = \bar L_w(\vartheta) + \alpha_\tau\sum_{n=1}^N w_n Y_n(\vartheta) L_n(\vartheta)$. Since $\sum_{n=1}^N w_n Y_n(\vartheta) \bar L_w(\vartheta) = 0$, the inner product simplifies to $\sum_{n=1}^N w_n Y_n(\vartheta) (Y_n(\vartheta) + \bar L_w(\vartheta)) = \sigma_w(\vartheta)^2$. Substitution provides the lower bound:
\[
R_\phi(\tau)
\ge
\bar L_w(\vartheta)
+
\frac{\sigma_w(\vartheta)}{\sqrt{\phi''(1)/2+\zeta}}\sqrt{\tau}.
\]

Combining this lower bound with (\ref{eq:phi_upper_local}) and letting $\zeta\downarrow 0$ completes the proof of (\ref{eq:phi_reweighting_expansion}).

Combining \hyperref[eq:structured_two_budget_OT]{(\ref*{eq:structured_two_budget_OT})} and \hyperref[eq:structured_two_budget_phidiv]{(\ref*{eq:structured_two_budget_phidiv})} yields
\[
\mathbb E_{X\sim\widehat\eta_{\varepsilon,\tau}}
\big[\ell_{\mathrm{DH}}(\vartheta;X)\big]
=
\bar L_w(\vartheta)
+ \Upsilon_w(\vartheta)\varepsilon
+ \sqrt{\frac{2}{\phi''(1)}}\,\sigma_w(\vartheta)\sqrt{\tau}
+ o\big(\varepsilon+\sqrt{\tau}\big).
\]
Comparing this expansion with Theorem~\ref{thm:two_budget_ot_phi_taylor} gives
\[
V_{\vartheta}(\varepsilon,\tau)
-
\mathbb E_{X\sim\widehat\eta_{\varepsilon,\tau}}
\big[\ell_{\mathrm{DH}}(\vartheta;X)\big]
=
o\big(\varepsilon+\sqrt{\tau}\big),
\]
which is \hyperref[eq:short-attack-accuracy]{(\ref*{eq:short-attack-accuracy})} and completes the proof.

<!-- remark-7: body_md -->
In Section~\ref{sec:short-two-budget}, we construct a first-order adversary using linear tilting of the form $a_n=1+\alpha_\tau(L_n-\bar L_w)$. While this construction is first-order accurate as $\tau\downarrow0$, at larger budgets the nonnegativity constraint can prevent further increases in $\alpha_\tau$, leaving part of the divergence budget unused. In the numerical experiments, we therefore replace linear tilting with exponential tilting, $a_n=\exp(sL_n)/\sum_j w_j\exp(sL_j)$, where $s\ge0$ is chosen by scalar bisection. The two constructions have the same small-budget leading behavior, preserving the first-order expansion while permitting the divergence budget to remain active beyond the range allowed by linear tilting.
