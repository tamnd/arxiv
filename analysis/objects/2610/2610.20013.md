\label{Thm} 
 Let $f$ satisfy the assumptions \eqref{sem0}-\eqref{defF} and \eqref{qlim}. Let $u_0$ and $\Omega$ be as in \eqref{Omega-u0} such that $T=T(u_0)<\infty$. Assume that  $u_t\ge 0$ or $q>q_S:=p_S/(p_S-1)$. Then we have
	\begin{equation}\label{infer1}
	u(x,t)=(1+o(1))G^{-1}\bigg(T-t+\frac{|x|^2}{4q|\log|x|^2|}\bigg)\quad \text{as } (x,t)\to (0,T).
	\end{equation}

<!-- cor-1: body_md -->
\label{Debo}
	Under the assumptions of Theorem~\ref{Thm}, we have 
	
		\begin{itemize}
			\item [(i)]\textbf{(final profile)} \begin{equation}\label{fp}
			u(x,T)=(1+o(1)) G^{-1}\bigg(\frac{|x|^2}{4q|\log|x|^2|}\bigg)   \quad\text{as }\ x\to 0;
			\end{equation}
			\item [(ii)] \textbf{(refined space-time profile)}
			\begin{equation}\label{lou}
			u\bigl(\xi\sqrt{(T-t)|\log(T-t)|},t\bigr)= (1+o(1))G^{-1}\bigg[(T-t)\bigg(1+\frac{|\xi|^2}{4q}\bigg)\bigg] 
			\quad\text{as }\ t\to T,
			\end{equation}
			uniformly for $\xi$ bounded.
		\end{itemize}

<!-- thm-2: body_md -->
\label{theo}
	Under assumptions of Theorem \ref{Thm}, we have:

$(i)$ if $q=1$
\begin{equation}\label{A1}
\log \frac{T-t}{G(u(y\sqrt{T-t},t))}=-\frac{ |y|^2-2n+o(1)}{4|\log(T-t)|},\quad \text{as} \ t\to T,
	\end{equation}
	
$(ii)$ if $q>1$
\begin{equation}\label{A1-2}
\Big(\frac{T-t}{G(u(y\sqrt{T-t},t)}\Big)^{(q-1)}=1-\frac{q-1}{q}\frac{ |y|^2-2n+o(1)}{4|\log(T-t)|},\quad \text{as} \ t\to T,
	\end{equation}
	with convergence in $H^1_\rho(\mathbb{R}^n)$ and uniformly for bounded $y$.

<!-- remark-1: body_md -->
$(i)$ {\bf(Alternative type I assumption)}
For $n\le 2$, Theorems \ref{Thm}-\ref{theo} remain valid 
under the type I blow-up assumption, namely,
 \begin{equation}\label{dam1a}
	\|u(\cdot,t)\|_\infty\le G^{-1}(C(T-t)),\quad T-\delta<t<T,
	\end{equation}
for some $C,\delta>0$.
In any dimension, property \eqref{dam1a} is in particular guaranteed if $u_t\ge 0$
(see \cite[Lemma~4.7]{C25-2}, which follows from arguments in \cite[Section 4]{FM85}).
However, in higher dimensions, the conclusions of Theorems \ref{Thm}-\ref{theo}
are not true in general under the weaker assumption \eqref{dam1a}.
In \cite{souplet2022universal}, it proved that \eqref{dam1a} holds for $q>p_S/(p_S-1)$ for any positive solution of \eqref{SHE}. This was extended in \cite{FK25} to the case of $q=1$ and $3\le n\le 9$ or $q_{JL}<q<q_S$ and $n\ge3$  in the radial setting, where 
$$
q_{JL}=\begin{cases} 1\quad\hbox{if }\ n\le 10,\\
\frac{n-2\sqrt{n-1}}{4}\quad\hbox{if } n\ge 11.
\end{cases}
$$

$(ii)$ {\bf(Bounded domain)} For simplicity, we have restricted our study to the case $\Omega=\mathbb{R}^n$. However, the same arguments apply, with minor modifications, to the case $\Omega=B_R$, following the approach in \cite{C25-2,CS25}.

<!-- remark-2: body_md -->
\label{rem1}
 {\bf Structural universality of the profile.}
 As in \cite{C25-2,CS25}, our main result recovers  
 the universal structure of the global blow-up profile. More precisely, the profile is obtained by applying the ``resolvent'' $G^{-1}$ of the associated ODE to the universal time-space building block
\begin{equation}\label{block}
T-t+\frac{|x|^2}{4q|\log|x|^2|}.
\end{equation}
It turns out that
 the influence of $f$ on the building block appears only at the coefficient of the logarithmic spatial correction which is governed by 
 \begin{equation} \label{checkq}
 \lim_{X\to\infty}\Big(f'(X)\int_X^\infty\frac{ds}{f(s)}\Big)=q.
 \end{equation}
  The same phenomenon also holds for the final and refined space-time profiles, as well as for the sharp asymptotics in backward space-time parabolas (cf.~\eqref{fp}, \eqref{lou} and \eqref{A1}).

{\bf(Examples of nonlinearities)}\label{exampleofL}
Our results apply in particular to the fast growing nonlinearities $f(s)=exp_m(u^p)$, with $m,p\ge 1$,
where $\exp_0:=id$ and $\exp_m(\cdot)=\exp(\exp_{m-1}(\cdot))$, and they are already completely new in this case. More generally, typical examples of nonlinearities fulfilling assumptions \eqref{sem0}-\eqref{qlim},
so that Theorems~\ref{Thm}-\ref{theo} apply,
 are given by~ 
\begin{equation} \label{examplesf} 
 f(u)=\exp_m(u^p)L(\exp_m(u^p))\quad\hbox{  for }\ m\in\{0,1,2,\cdots\}\quad\hbox{ and }\  \begin{cases}p>1\quad \hbox{if  $m=0$},\\
 p\ge1\quad \hbox{if  $m\in \mathbb{N}$}\end{cases}
\end{equation} 
  where $L\in C^1([0,\infty))$ is a slow variation function satisfying 
 \begin{equation} \label{semC-CS}
 \frac{XL'(X)}{L(X)}=o(\log^{-\alpha_0} X)\quad\hbox{ and }\quad \frac{X^2L''(X)}{L(X)}=o(1)\quad\hbox{as }\ X\to \infty.
\end{equation}
Such $L$ include logarithms and their powers and iterates, as well as some strongly oscillating functions. 
  We here provide a few examples of such function $L$:
$$
 	\left\{\begin{aligned}
 		&\hbox{\ $\star$\  $\log^a (K+s)$ for $K>1$ and $a\in\mathbb{R}$ or $K=1$ and $a\ge1$,} \\
 		\noalign{\vskip 1mm}
 		&\hbox{\ $\star$\  the iterated logarithms $\log_m(K+s)$, with $K=\exp\circ\exp\circ\cdots\circ \exp(0)$,}  \\
 		\noalign{\vskip 1mm}
 		&\hbox{\ $\star$\  $\exp(|\log s|^\nu)$ with $\nu\in (0,1/2)$,} \\
 		\noalign{\vskip 1mm}
 		&\hbox{\ $\star$\  the strongly oscillating functions} \\
 		&\hbox{\qquad $\bigl[\log(3+s)\bigr]^{\sin[\log\log(3+s)]}
 			\quad\hbox{ and }\quad
 			\exp\bigl[|\log s|^\nu\cos(|\log s|^\gamma)\bigr],\ \ \nu, \gamma>0,\ \nu+\gamma<1/2$,} \\
 		\noalign{\vskip 1mm}
 		&\hbox{\ $\star$\  $1 +a \sin\bigl(\log^\nu(2+s)\bigr)$ with $ \nu\in (0,1/2)$ and $|a|<1$.}
 	\end{aligned}
 	\right.
 	$$
 	We observe that the factor $L$ does not affect the limit $q$ in \eqref{qlim}. 
Indeed, a sufficient condition for the limit $q$ in \eqref{checkq} to exist is that
$$
\lim_{X\infty}\frac{f'^2}{ff''}=q.
$$

{\bf (Known results on the blow-up profile)}
$(i)$ Regarding \eqref{examplesf}, we see that 
$$
\begin{cases}
q=1\quad\hbox{if }\quad m\in\mathbb{N} \quad \hbox{and }\ p\ge1\\
q=\frac{p}{p-1}\quad\hbox{if }\quad m=0 \quad \hbox{and }\ p>1.\\
\end{cases}
$$

Our results cover all the nonlinearities considered in \cite{C25-2} and \cite{CS25}, namely $e^uL(e^u)$ and $u^pL(u)$ with $p>1$, respectively. We obtain the same results under weaker assumptions on $f$. Indeed, they obtained the sharp global blow-up profile given in \eqref{infer1} under the  assumptions \eqref{semC-CS} and 
$$
\Big(\frac{XL'(X)}{L(X)}\Big)'=o\Big(\frac{1}{X\log X}\Big) \quad \hbox{as }\quad X\to\infty.
$$
This assumption is used in their proofs only to control the linear term $\tau(s)$ in the rescaled equation; see \cite[Formula~(6.14)]{C25-2} and \cite[Formula~(6.27)]{CS25}. This term disappears in our rescaled equation, thanks to the transformation \eqref{dam11} below. Consequently, we are able to obtain the same sharp blow-up profile without requiring this additional assumption involving $L''$.



$(ii)$ Choosing $m=1$ and $p>1$ in \eqref{examplesf}, $a=q/p$ and $L(s)=\log^a(K+s)$, with either $K>0$ and $a=0$, or $K=0$ and $a\ge1/p$, we obtain
\begin{equation} \label{ryoequ}
f(u)=u^q e^{u^p},
\end{equation}
where $q\in\{0\}\cup[1,\infty)$. Thus, our assumptions cover the entire class of nonlinearities considered in \cite{Ryoto}. In that case, we have 
$$
G^{-1}(Y)=(1+o(1))|\log Y|^{1/p}\quad\hbox{as}\quad Y\to 0^+.
$$
Combining this with \eqref{infer1}, we obtain the explicit blow-up asymptotics of $u$, which takes the form
$$
u(x,t)= (1+o(1))\Big|\log\bigg(T-t+\frac{|x|^2}{4|\log|x|^2|}\bigg)\Big|^{1/p}\quad \text{as } (x,t)\to (0,T).
$$

\label{CSLem} {\bf (On the convexity assumption)}
Let $f$ satisfy \eqref{sem0}-\eqref{defF} and that 
$$
\lim_{X\to \infty} f'(X)G(X)=q>1.
$$
Then $f$ has regular variation at $\infty$ with index $p=\frac{q}{q-1}$. This follows by showing that $L(X)=X^{-p}f(X)$ satisfies
$$
\lim_{X\to \infty} \frac{XL'(X)}{L(X)}=0.
$$
 Moreover, if $f$ is a $ C^2$ function at $\infty$, it was proved in \cite[Page~4526]{CS25} that $\frac{f}{\log f}$ is convex for large $X$, assuming
$$
\lim_{X\to \infty} \frac{X^2L''(X)}{L(X)}=0.
$$

\label{qssblowup}
Let $p > 1$.
 Assume \eqref{sem0}-\eqref{defF}, \eqref{qlim}, \eqref{Omega-u0} and $T=T(u_0)<\infty$. We also assume that $u$ is of type~I. For any $t_j\to T$, there exist a subsequence (still denoted $t_j$) and a nonnegative radially decreasing function $\phi$ solving
$$
\Delta\phi-\frac12 y\cdot\nabla\phi=\begin{cases}1-e^\phi\quad\hbox{if }\ q=1,\\ (q-1)\phi-\phi^{q/(q-1)}\quad\hbox{if }\ q>1,\end{cases}
\quad\text{in }\mathbb{R}^n,
$$
 such that
\begin{equation}\label{eq:qssblowup}
	\phi(|y|)=\lim_{t_j \rightarrow T} \begin{cases}\log\Big(\frac{(T-t_j)}{G(u(y\sqrt{T-t_j},t_j))}\Big)\quad\hbox{if }\ q=1,\\ \Big(\frac{(q-1)(T-t_j)}{G(u(y\sqrt{T-t_j},t_j))}\Big)^{q-1} \quad\hbox{if }\ q>1,\end{cases}
\end{equation}
uniformly for $y$ bounded. In particular, under the assumptions of Theorem~\ref{Thm}, \eqref{eq:qssblowup} holds with $\phi\equiv(q-1)^{q-1}$ and $\phi\equiv0$ if $q=1$.

<!-- lemma-1: body_md -->
\label{lemmaFH}
Let $f$ and $G$ be as in \eqref{sem0}-\eqref{defF} with
$$
 f'(X)G(X)\to q\ge 1\quad\hbox{as }\ X\to \infty.
$$
We also define 
\begin{equation} \label{Hqlim}
H(X):=\int_X^\infty \frac{A+\log f(s)}{f(s)}ds\quad\hbox{for large $X>0$ and some $A>0$}.
\end{equation}
$(i)$ As $X\to \infty$, we have 
\begin{equation} \label{propFf}
|\log G(X)|\sim \frac1q\log f(X)
\end{equation}
and 
\begin{equation} \label{propFH2}
H(X)\sim qG(X)|\log G(X)|.
\end{equation}

\noindent$(ii)$ Moreover, as $Y\to 0^+$, we have 
\begin{equation} \label{propinversFH2}
H^{-1}(Y)\sim G^{-1}\Big(\frac{Y}{q|\log Y|}\Big).
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
$(i)$ The proof of \eqref{propFf} and \eqref{propFH2} is based on the L'H\^{o}pital's rule. Indeed, using $f(s)\to \infty$ and $G(X)\to 0$ as $X\to \infty$, we have
$$
\begin{aligned}
\lim_{X\to \infty}\frac{\log G(X)}{\log f(X)}&=\lim_{X\to \infty}\frac{\frac{-1 }{f(X)G(X)}}{\frac{f'(X)}{f(X)}}=\lim_{X\to \infty}\frac{-1}{ f'(X)G(X)}=\frac{-1}{q}.
\end{aligned}
$$
Which yields the proof of \eqref{propFf}. Building on the latter, we derive \eqref{propFH2} by the same way. Namely, we compute:
$$
\begin{aligned}
\lim_{X\to \infty}\frac{G(X)|\log G(X)|}{H(X)}&=\lim_{X\to \infty}\frac{\frac{\log G(X)}{f(X)}+\frac{1}{f(X)}}{-\frac{\log f(X)}{f(X)}}=\lim_{X\to \infty}\Big(\frac{-\log G(X)}{\log f(X)}-\frac{1}{\log f(X)}\Big)=\frac{1}{q}.
\end{aligned}
$$

\noindent$(ii)$ We shall derive \eqref{propinversFH2} from \eqref{propFf}-\eqref{propFH2}. 
By \eqref{propFH2}, we first note that $\log H(X)\sim\log G(X)$ at $\infty$, and then 
$$
H(X)\sim qG(X)|\log H(X)|.
$$
Thus, writing $Y:=H(X)$, we get 
$$
G(H^{-1}(Y))\sim\frac{Y}{q|\log Y|}.
$$
Which implies that 
$$
H^{-1}(Y)=G^{-1}(G(H^{-1}(Y)))=G^{-1}\Big((1+o(1))\frac{Y}{q|\log Y|}\Big).
$$
This along with \eqref{equivGG} below (which is independant of \eqref{propinversFH2}) concludes the proof.

\label{ell12a}
	Let $f$ satisfy the assumptions \eqref{sem0}-\eqref{qlim} and $F$ be defined by \eqref{defF}.
	\smallskip
	 There exist $C_0,C_1,X_1>0$ such that, if $0\le \varepsilon<1/2$ then for all $X\ge X_1$, we have 
	
		\begin{equation} \label{ell12a1}
f((1-\varepsilon))X)\ge e^{-C_0\varepsilon}f(X)	
		\end{equation}
		and
		\begin{equation} \label{ell12a2}
	G((1-\varepsilon )X)\le (1+C_1\varepsilon)G(X).
	\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
By remark~\ref{CSLem}, we refer to \cite[Lemma~7.1]{CS25} for the proof of this lemma in the case of $q>1$ and also \cite[Lemma~7.2]{C25-2} in the particular case of $m=1$ and $p=1$ in \eqref{examplesf}. We give here a proof using the convexity of $f/\log f$ and $f\to \infty$ at $\infty$.
To this end, let $X_1>0$ be such that 
$$
(f/\log f)''\ge 0,\quad\log f(X)\ge 2,\quad \hbox{and }\ f'\ge 0\quad\forall \ X\ge X_1.
$$ 

 Set $h(X)=f(X)/\log f(X)$. By the first part of \eqref{qlim}, we have $h,h''\ge0$ for $X\ge X_1$, and  hence 
$$
h(X/2)\le h(X)-\frac{X}{2} h'(X).
$$ 
This in particular implies that 
$$
h'(X)\le \frac2X (h(X)-h(X/2))\le \frac{2 h(X)}{X}.
$$
Thus by a simple computation for large $X$, we get 
$$
0\le \frac{f'(X)}{f(X)}=\frac{h'(X)}{h(X)(1-1/\log f(X))}\le \frac{2h'(X)}{h(X)}\le \frac{4}{X}.
$$
We are ready to complete the proof of \eqref{ell12a1} as follow:
$$
\begin{aligned}
\log \frac{f(X)}{f((1-\varepsilon)X)}&=\int_{(1-\varepsilon)X}^X\frac{f'(s)}{f(s)}ds\le 4 \int_{(1-\varepsilon)X}^X\frac{1}{s}ds\\
&\le-4\log(1-\varepsilon)\le 8\varepsilon,
\end{aligned}
$$
where we obtain the last inequality by using $-\log(1-\varepsilon)\le 2\varepsilon$ for any $0<\varepsilon<1/2$. Compose with $x\mapsto\exp(x)$, increasing function, we get
$$
 \frac{f(X)}{f((1-\varepsilon)X)}\le e^{ 8\varepsilon} \quad \hbox{for large }\ X.
$$
This conclude the proof of \eqref{ell12a1}.

By the definition of $F$ as in \eqref{defF}, \eqref{ell12a2} is a direct consequence of \eqref{ell12a1}.

\label{lem1-1}
$(i)$ If a function $\varepsilon$ satisfies $\lim_{Y\to 0^+}\varepsilon(Y)=0$ then
 		\begin{equation}\label{equivGG}
 		\lim_{Y\to 0^+} \frac{G^{-1}\Big((1+\varepsilon(Y))Y\Big)}{G^{-1}(Y)}=1
 			\end{equation}
 		and the similar properties hold for  $H^{-1}$.
 		
 		\smallskip
 		
$(ii)$ For any $\beta>0$, we have 
 		\begin{equation} \label{eqen0}
 		\lim_{Y\to 0^+} Y^{\beta}(G^{-1}(Y))^\alpha=0,\quad\hbox{for any } \alpha\in (0,\beta/q).
 		\end{equation}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
\noindent$(i)$  We observe that \eqref{equivGG} for $G$ is a direct consequence of \eqref{ell12a2} and decreasing property of $G$. Therefore, by using \eqref{propFH2} we also deduce the conclusion for $H$.

\noindent$(ii)$ Let $X=G^{-1}(Y)$ and $g(Y)=f(G^{-1}(Y))$. By a simple computation, we have 
$$
g'(Y)=f'(X)\frac{d X}{dy}=-f'(X)f(X),
$$
which implies that 
$$
-\frac{Yg'(Y)}{g(Y)}\to q\quad\hbox{as }\ Y\to 0^+.
$$
Hence, there exists $Y_0>0$ such that 
$$
\frac{g'(Y)}{g(Y)}\ge -\frac{q+1}{Y}\quad\hbox{for all }\  0<Y\le Y_0.
$$
Therefore, we have 
$$
\log g(Y)-\log g(Y_0)\le (q+1)\log \frac{Y_0}{Y} \quad\hbox{for all }\  0<Y\le Y_0.
$$
Thus 
$$
g(Y)\le  Y^{-(q+1)} \Big(Y_0^{(q+1)}g(Y_0)\Big) \quad\hbox{for all }\  0<Y\le Y_0.
$$
Now we use 
$$
\begin{aligned}
G^{-1}(Y)-G^{-1}(Y_0)&=-\int_Y^{Y_0}(G^{-1}(z))'dz=\int_Y^{Y_0}f(G^{-1}(z))dz=\int_Y^{Y_0}g(z)dz\\
&\le  Y_0^{(q+1)}g(Y_0)\int_Y^{Y_0}z^{-(q+1)}dz\le C(q,Y_0,g(Y_0))(Y^{-q}-Y_0^{-q})\le CY^{-q}
\end{aligned}
$$
Hence for any $\alpha,\beta,>0$ we have 
$$
Y^\beta(G^{-1}(Y))^\alpha\le CY^{\beta-\alpha q}+Y^\beta G^{-1}(Y_0) \quad \hbox{for all }\ 0<Y\le Y_0
$$
This conclude the proof of \eqref{eqen0}.

\label{rappelH}
Let $\Omega=\mathbb{R}^n \text{ or }\Omega=B_R$, and $u_0\ge 0$ be radially symmetric, nonincreasing in $|x|$ and  $T<\infty$.
Assume that  
\begin{equation*}
 f\in C^1([0,\infty)),\quad f(0)\ge 0, \quad \lim_{s\to \infty}f(s)=\infty,
 \end{equation*}
 \begin{equation*}
	\hbox{$f$ is $C^2$ and } \frac{f}{\log f} \hbox{ is convex at $\infty$,} \quad \int^\infty \frac{\log f(s)}{f(s)}ds <\infty.
	\end{equation*}
	If $\Omega=\mathbb{R}^n$ assume also that $u_0$ is nonconstant. 
	Then there exist $A_0,\eta>0$ such that
	\begin{equation}\label{upperTheoL}
	u(x,t)\le H^{-1}\bigg(H(u(0,t))+\frac{|x|^2}{4}\bigg)\quad \text{in }\ B_\eta\times(T-\eta,T),
	\end{equation}
	where $H$ is define as in \eqref{Hqlim}.
Moreover, there exist constants $A>0$ such that for $M>0$ as in \eqref{fposM},
\begin{equation} \label{Rappelupper}
-u_r\ge \frac{rf(u)}{2(A+\log f(u))},\quad \hbox{in }\ Q:=\big\{(x,t)\in B_{R/2}\times(T/2,T); \ u(x,t)\ge M\big\}.
\end{equation}
<!-- end thm-3 -->
<!-- proof-4: body_md -->
Since $m(t):=u(0,t)=\|u(t)\|_\infty\to \infty$, we have  $G(m(t))\to0^+$ 
 as $t\to T$. 
 
 Applying property \eqref{propFH2} to  inequality \eqref{upperTheoL}, 
 we get
$$u(x,t)\le H^{-1}\left(H(m)+\frac{|x|^2}{4}\right)
 =H^{-1}\left((1+o(1))\left(qG(m)|\log(G(m))|+\frac{|x|^2}{4}\right)\right)$$
as $(x,t)\to (0,T)$.  Next applying \eqref{propinversFH2} and \eqref{equivGG},
we obtain
 \begin{equation}\label{nuitt}
 u(x,t)\le (1+o(1)) G^{-1}\left(\frac{G(m)|\log(G(m))|+\frac{|x|^2}{4q}}
 {\Bigl|\log\Bigl(G(m)|\log(G(m))|+\frac{|x|^2}{4}\Bigr)\Bigr|}\right).
\end{equation}
 Also, for $(x,t)$ close enough to $(0,T)$,
 we have $G(m)\le G(m)|\log(G(m))|+\frac{|x|^2}{4}<1$,
 so that

\end{aligned}

This concludes the proof of Lemma~\ref{controlRL}.

		+\frac{\mathcal{R}(y,|\log \sigma|)}{|\log\sigma|}\Big)^{1/(1-q)}\Big]\quad\hbox{if }\ q>1;
		\end{cases}
		$$
		with
		$$\lim_{s\to\infty}\|\mathcal{R}(\cdot,s)\|_{H^1_\rho}=0\quad\hbox{ and uniformly for bounded $y$}.$$
		
	$\bullet$ {\bf Case $q>1$.} 
 Recall that $f$ is regularly varying at $\infty$ with index $q/(q-1)$; see Remark~\ref{CSLem}. In particular, $G^{-1}$ is regularly varying at $0$ with index $1-q$. Hence, for bounded $|y|$ and sufficiently small $\sigma$, we obtain
	$$
	u(x,T-\sigma)\ge G^{-1}(\sigma)\left(1+ \frac{2n(q-1)}{4q|\log\sigma|}-\frac{(q-1)|x|^2}{4q\sigma|\log\sigma|}
		+\frac{\mathcal{R}_\sigma(x)}{|\log\sigma|}\right),
	$$		
		where 
		$$
		 \mathcal{R}_\sigma(x)=\mathcal{R}(x\sigma^{-1/2},|\log \sigma|)+\mathcal{R}_0(x\sigma^{-1/2},|\log \sigma|),\ \hbox{with }\ \|\mathcal{R}_0(\cdot,s)\|_{H_\rho^1}\underset{s\to \infty}{\to }0.
		$$ 
		
	$\bullet$ {\bf Case $q=1$.} Using \eqref{qlim} and Taylor expansion for bounded $|y|$ and sufficiently small $\sigma$, we obtain
	 	$$
	 	\frac{u(x,T-\sigma)}{G^{-1}(\sigma)}\ge 1+ \frac{f(G^{-1}(\sigma))}{G^{-1}(\sigma)f'(G^{-1}(\sigma))}\left(\frac{2n}{4|\log\sigma|}-\frac{|x|^2}{4\sigma|\log\sigma|}
		+\frac{\mathcal{R}_\sigma(x)}{|\log\sigma|}\right)
	 	$$
		where 
		$$
		 \mathcal{R}_\sigma(x)=\mathcal{R}(x\sigma^{-1/2},|\log \sigma|)+\mathcal{R}_1(x\sigma^{-1/2},|\log \sigma|),\ \hbox{with }\ \sup_{s\ge s_0}\|\mathcal{R}_1(\cdot,s)\|_{H_\rho^1}<\infty.
		$$ 

In all cases, we then have 
\begin{equation} \label{usigma}
u(x,T-\sigma)\ge G^{-1}(\sigma)\left(1+ \frac{2ng(q,\sigma)}{4|\log\sigma|}-\frac{g(q,\sigma)|x|^2}{4\sigma|\log\sigma|}
		+\frac{\mathcal{R}_2(\sigma,x)}{|\log\sigma|}\right),
\end{equation}
	where 
	\begin{equation} \label{defgqsig}
	0\le g(q,\sigma)=\begin{cases} \frac{q-1}{q}\quad\hbox{if }\ q>1\\
	\frac{f(G^{-1}(\sigma))}{G^{-1}(\sigma)f'(G^{-1}(\sigma))} \quad\hbox{if }\ q=1;
	\end{cases}
	\end{equation}
	and 
	$$
	 \mathcal{R}_2(\sigma,x)=\mathcal{R}(x\sigma^{-1/2},|\log \sigma|)+\mathcal{R}_3(x\sigma^{-1/2},|\log \sigma|),\quad \hbox{with }\quad \|\mathcal{R}_3(\cdot,s)\|_{H_\rho^1}\underset{s\to \infty}{\to }0.
	 	$$ 
Therefore we have 
$$
\begin{aligned}
 V(x,t)&= M-S(t-t_0) M+S(t-t_0)\Big( u(t_0)\Big)\ge 
S(t-t_0)\Big( u(t_0\Big)\\
&\ge G^{-1}(\sigma)S(t-t_0)\Bigl(1+\frac{2ng(q,\sigma)}{4|\log\sigma|}-\frac{g(q,\sigma)|x|^2}{4\sigma|\log\sigma|}\Bigl)
+\frac{G^{-1}(\sigma)}{|\log\sigma|^2}S(t-t_0)\mathcal{R}_2\equiv V_1+V_2\\
\end{aligned}$$
hence, since $ V\ge 0$,
\begin{equation} \label{VV1V2}
V\ge (V_1)_+-|V_2|\quad \hbox{ in $B_\delta\times(t_0,T)$.}
\end{equation}
Using Lemma~\ref{lemSrho}(i) and recalling \eqref{deft0}, 
we may estimate $(V_1)_+$ from below for $t\in[t_0,T)$ as:
\begin{equation} \label{V1below}
\begin{aligned}
(V_1(t))_+
&\ge e^{-\lambda(t-t_0)} G^{-1}(\sigma)\Bigl[1+\frac{2ng(q,\sigma)}{4|\log\sigma|}-\frac{g(q,\sigma)(|x|^2+2n(t-t_0))}{4\sigma|\log\sigma|}\Bigl]_+
\chi_{B_{\delta/2}} \\
&{\strut =\strut} e^{-\lambda\sigma} G^{-1}(\sigma) \Bigl[1-\frac{g(q,\sigma)|x|^2}{4\sigma|\log\sigma|}\Bigl]_+
\chi_{B_{\delta/2}}. 
\end{aligned}
\end{equation}
To control $V_2$, we observe that, for $t\in[t_0,T)$,
$$
\begin{aligned}
\int_{B_\delta}\mathcal{R}_2^2(\sigma,z)e^{-\frac{|z|^2}{4(t-t_0)}}dz&
\le 2\int_{B_\delta} (\mathcal{R}^2+\mathcal{R}_3^2)(z\sigma^{-1/2},|\log \sigma|)e^{-\frac{|z|^2}{4\sigma}}dz\\
&\le 2\sigma^{n/2}(\|\mathcal{R}(\cdot,|\log \sigma|)\|^2_{L^2_\rho}+\|\mathcal{R}_3(\cdot,|\log \sigma|)\|^2_{L^2_\rho})
\end{aligned}
$$
and then we use Lemma~\ref{lemSrho}(ii) 
to write, for $x\in B_\delta,\ t\in[t_0,T)$:
$$\begin{aligned}
\bigl|S(t-t_0)\mathcal{R}_2(\sigma,\cdot)\bigr|(x)
&\le C(n)(t-t_0)^{-n/4}\bigg(\int_{B_\delta}(\mathcal{R}_2^2(\sigma,z)e^{-\frac{|z|^2}{4(t-t_0)}}dz\bigg)^{1/2}e^{\frac{|x|^2}{2(t-t_0)}}\\
&\le C(n)(t-t_0)^{-n/4}\sigma^{n/4}e^{\frac{|x|^2}{2(t-t_0)}}(\|\mathcal{R}(\cdot,|\log \sigma|)\|_{L^2_\rho}+\|\mathcal{R}_3(\cdot,|\log \sigma|)\|_{L^2_\rho}).
\end{aligned}$$
Denoting $D_\sigma=\overline B_{\sqrt\sigma}\times [T-\textstyle\frac{\sigma}{2},T)$ and using \eqref{deft0},
we obtain
\begin{equation} \label{controlV2}
\sup_{D_\sigma}|V_2|
\le C(n)G^{-1}(\sigma)\frac{\|\mathcal{R}(\cdot,|\log \sigma|)\|_{L^2_\rho}+\|\mathcal{R}_3(\cdot,|\log \sigma|)\|_{L^2_\rho}}{|\log\sigma|}
=\frac{\varepsilon(\sigma)G^{-1}(\sigma)}{|\log\sigma|}.
\end{equation}
Here and in the rest of the proof $\varepsilon$ denotes a generic function such that $\lim_{\sigma\to 0}\varepsilon(\sigma)=0$.
 Assuming $\sigma\le \delta^2/4$, it follows from \eqref{VV1V2}-\eqref{controlV2} that 
\begin{equation} \label{controlV}
\tilde V\ge e^{-\lambda\sigma} G^{-1}(\sigma)\Bigl(1-\frac{g(q,\sigma)+\varepsilon(\sigma)}{4|\log\sigma|}\Bigl)\quad\hbox{in $D_\sigma$}.
\end{equation}
Moreover, we may choose $\sigma_0\in(0,\delta^2/4)$ such that the RHS of \eqref{controlV} is $> M$ 
for all $\sigma\in(0,\sigma_0]$.
Therefore
$$
0<G(V)+t_0-t\le  \mu_q(\sigma,t):=G\biggl[e^{-\lambda\sigma} G^{-1}(\sigma)\Bigl(1-\frac{g(q,\sigma)+\varepsilon(\sigma)}{4|\log\sigma|^2}\Bigl)\biggr]+T-t-\sigma
\quad\hbox{in $D_\sigma$},
$$
where the RHS is $<G( M)$ hence, by \eqref{a1}-\eqref{defUV},
\begin{equation} \label{fact}
u\ge G^{-1}(\mu_q(\sigma,t))\quad\hbox{in $D_\sigma$}.
\end{equation}



\vspace{.2cm}

\noindent{\bf Step 3.} {\it Asympotics of $\mu_q(\sigma,t)$ and conclusion.}
Fix any $x_0$ such that $|x_0|^2\le\sigma_0$. Choosing 
\begin{equation} \label{choicesigma}
\sigma=|x_0|^2,
\end{equation}
it follows from \eqref{fact} that
\begin{equation} \label{fact2}
u(x_0, t)\ge G^{-1}(\mu_q(\sigma,t))\quad\hbox{ for $T-\frac{\sigma}{2}\le t<T$.}
\end{equation}

To estimate $\mu_1$ from above, we first use \eqref{ell12a2} in Lemma~\ref{ell12a} to get, for some constant $C>0$,
$$
\mu_1(\sigma,t)\le (1+C\sigma)G\biggl[G^{-1}(\sigma)\Bigl(1-\frac{g(q,\sigma)+\varepsilon(\sigma)}{4|\log\sigma|}\Bigl)\biggr]+T-t-\sigma.
$$
Next, by $G'=-1/f$ and the mean value theorem, there exists $\theta(\sigma)\in(0,1)$ such that

$$
\mu_1(\sigma,t)\le (1+C\sigma)\biggl\{\sigma+
\frac{
\frac{g(q,\sigma)+\varepsilon(\sigma)}{4|\log\sigma|}G^{-1}(\sigma)}{f\Bigl[G^{-1}(\sigma)\Bigl(1-\theta(\sigma)\frac{g(q,\sigma)+\varepsilon(\sigma)}{4|\log\sigma|}\Bigl)\Bigl]}\biggr\}+T-t-\sigma.
$$


By \eqref{ell12a1} in Lemma~\ref{ell12a} and the fact that $g(q,\sigma)\le C$ for any $q\ge 1$ and small $\sigma$, we have
$$
f\bigl[G^{-1}(\sigma)\bigl(1-\theta(\sigma)\frac{g(q,\sigma)+\varepsilon(\sigma)}{4|\log\sigma|}\bigl)\bigl]
\ge (1+\varepsilon(\sigma))f\bigl(G^{-1}(\sigma)).
$$

\begin{itemize}

\item {\bf \underline{$q=1$ case}:}
Using the definition of $g(q,\sigma)$ in \eqref{defgqsig} for $q=1$, it follows that
$$
\begin{aligned}
\mu_1(\sigma,t)
&\le (1+C\sigma)\biggl\{\sigma+\frac{(g(q,\sigma)+\varepsilon(\sigma))G^{-1}(\sigma)}{4|\log\sigma|f(G^{-1}(\sigma))}\biggr\}+T-t-\sigma\\
&=T-t+C\sigma^2+ \frac{(g(q,\sigma)+\varepsilon(\sigma))G^{-1}(\sigma)}{4|\log\sigma|f(G^{-1}(\sigma))}\\
&\le T-t+C\sigma^2+ \frac{(1+\varepsilon(\sigma))}{4|\log\sigma|}\times\frac{g(q,\sigma)G^{-1}(\sigma)}{f(G^{-1}(\sigma))}\\
&\le T-t+C\sigma^2+\frac{(1+\varepsilon(\sigma))\sigma}{4|\log\sigma|(f'G)(G^{-1}(\sigma))}\\
&\le(1+\varepsilon(\sigma))\Bigl(T-t+\frac{\sigma}{4|\log\sigma|}\Bigr).
\end{aligned}$$

\item {\bf \underline{$q>1$ case}:}
Using $g(q,\sigma)=(q-1)/q$ along with the fact that $X/(G(X)f(X))\to 1/(q-1)$ as $X\to \infty$, we obtain
$$
\begin{aligned}
\mu(\sigma,t)
&\le (1+C\sigma)\biggl\{\sigma+\frac{(g(q,\sigma)+\varepsilon(\sigma))G^{-1}(\sigma)}{4|\log\sigma|f(G^{-1}(\sigma))}\biggr\}+T-t-\sigma\\
&=T-t+C\sigma^2+ \frac{(q-1+\varepsilon(\sigma))G^{-1}(\sigma)}{4q|\log\sigma|f(G^{-1}(\sigma))}\\
&\le T-t+C\sigma^2+ \frac{(1+\varepsilon(\sigma))\sigma}{4q|\log\sigma|}\times\frac{(q-1)G^{-1}(\sigma)}{(Gf)(G^{-1}(\sigma))}\\
&\le T-t+C\sigma^2+\frac{(1+\varepsilon(\sigma))\sigma}{4q|\log\sigma|}\\
&\le(1+\varepsilon(\sigma))\Bigl(T-t+\frac{\sigma}{4q|\log\sigma|}\Bigr).
\end{aligned}$$
\end{itemize}

By \eqref{choicesigma}, \eqref{fact2}, using \eqref{equivGG} again, we thus deduce that for all $q\ge 1$, as $(x,t)\to (0,T)$,
$$u(x,t)\ge(1+o(1))G^{-1}\bigg(T-t+\frac{|x|^2}{4q|\log|x|^2|}\bigg),
\quad\hbox{ uniformly for $ |x|^2\ge 2(T-t)$.}$$
		
	On the other hand, by the second assertion of Proposition~\ref{qssblowup} and \eqref{equivGG} again, 
	 we have, as $(x,t)\to (0,T)$,
		$$u(x,t)= (1+o(1))G^{-1}(T-t)=(1+o(1))G^{-1}\bigg(T-t+\frac{|x|^2}{4|\log|x|^2|}\bigg),
		\ \hbox{ uniformly for $ |x|^2\le 2(T-t)$.}$$
		 Estimate \eqref{infer1-lower}, and so the lower part of \eqref{infer1}, follows.
