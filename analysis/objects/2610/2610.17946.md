We use $\xrightarrow{\makebox[1em][c]{$\scriptstylep$}}$ and $\xrightarrow{\makebox[1em][c]{$\scriptstyled$}}$ for convergence in probability and in distribution respectively. We will throughout use standard Mann-Wald notation to describe orders of magnitude of quantities. Given two sequences of positive constants $(a_n), (b_n)$, we write $a_n = o(b_n)$ when $\lim_{n \to \infty} (a_n/b_n) = 0$, and $a_n = O(b_n)$ when $\limsup_{n \to \infty} (a_n/b_n) = K < \infty$. For a sequence of random variables $\{X_n\}$ and a sequence of positive constants $\{a_n\}$, we write $X_n = o_p(a_n)$ if $X_n / a_n \xrightarrow{\makebox[1em][c]{$\scriptstylep$}} 0$ as $n \to \infty$; and $X_n = O_p(a_n)$ when $X_n / a_n$ is bounded in probability as $n \to \infty$ i.e. $\lim_{C \to \infty}\limsup_{n\to\infty}\mathbb{P}(|X_n|\ge Ca_n)=0$. In particular, for a constant $c$, $X_n = c + o_p(1)$ means that $X_n \xrightarrow{\makebox[1em][c]{$\scriptstylep$}} c$.

\label{prop:sphere_backprojection}
Consider $\mathcal{M} = S^{d-1}(b,r)$ the sphere geometry defined in Section~\ref{sec:shift-sphere-geom}. Let $z \in \mathcal{M}$ and $v \in T_z\mathcal{M}$ satisfy $\|v\|<r$. Then:
\begin{enumerate}
\item[(i)] The projection scalar is $\alpha = -r + \sqrt{r^2 - \|v\|^2}$, and the proposed point is $z' = b + v + \sqrt{r^2 - \|v\|^2}\,\hat{n}$.

\item[(ii)] The new unit normal is $\hat{n}' = (z'-b)/r = (1 + \alpha/r)\,\hat{n} + v/r$.

\item[(iii)] The reverse tangent increment is $v' = -v - \alpha(\hat{n} + \hat{n}')$, and $\|v'\| = \|v\|$.
\end{enumerate}
<!-- end prop-1 -->
<!-- pf-1: body_md -->
All identities follow by direct calculation from $\|z-b\| = r$, $v \perp \hat{n}$, and $z' = z + v + \alpha\hat{n} \in \mathcal{M}$. Note that we select the plus root of the projection scalar characterised by $\alpha\to0$ as $v\to0$.

\label{prop:Rc_sphere}
Under the notation above,
\begin{align}
  \langle v',b\rangle - \langle v,b\rangle
  &= -\bigl(2 + \alpha/r\bigr)\,g,
  \label{eq:vb_diff} \\
  \|P_{z'}(b)\|^2 - \|P_z(b)\|^2
  &= -\frac{2\langle \hat{n},b\rangle}{r}\,g - \frac{g^2}{r^2}.
  \label{eq:Pb_diff}
\end{align}
Consequently, the exact Metropolis defect of the proposal \eqref{eq:general_proposal} is
\begin{equation}
  R =
  \left(-1-\frac{2c}{\sigma^2} -\frac{c}{\sigma^2}\,\frac{\alpha}{r}
    +\frac{c^2}{\sigma^2}\,\frac{\langle\hat{n},b\rangle}{r} \right) g +
  \frac{c^2}{2\sigma^2}\,\frac{g^2}{r^2}.
  \label{eq:Rcancel_general}
\end{equation}
In particular, for tangent pCN
\begin{equation}
    R=
    \left( \frac{1-\rho}{1+\rho} + \frac{\alpha}{(1+\rho)r} \right) g + 
    \frac{1-\rho}{1+\rho}\left(  \frac{\langle \hat{n},b\rangle}{r}\,g + \frac{g^2}{2r^2} \right).
  \label{eq:Rcancel_exact}
\end{equation}

<!-- lem-4: body_md -->
\label{lem:v2_moment}
For any $z \in \mathcal{M}$, under the tangent pCN proposal,
\begin{equation}
  \mathbb{E}[\|v\|^2 \mid z] = \beta^2(d-1) + \gamma^2 \|P_z(b)\|^2.
  \label{eq:v2_mean}
\end{equation}
Define $M_d := \mathbb{E}_\pi[\|v\|^2]$, then under the critical scaling $\beta = \ell d^{-1/6}$ and stationarity,
\begin{equation}
  M_d = \beta^2(d-1)\bigl(1 + O(\beta^2)\bigr)
  = \ell^2 d^{2/3}\bigl(1 + o(1)\bigr),
  \label{eq:Md_leading}
\end{equation}
\begin{equation}
  \mathrm{Var}_\pi(\|v\|^2) = 2\beta^4(d-1)\bigl(1+o(1)\bigr),
  \label{eq:v2_total_var}
\end{equation}
and consequently
\begin{equation}
  \mathbb{E}_\pi\,\left[\left(\frac{\|v\|^2}{M_d} - 1\right)^2\right] = O(d^{-1}).
  \label{eq:eta_L2}
\end{equation}
<!-- end lem-4 -->
<!-- pf-9: body_md -->
Conditional on $z$, applying the Gaussian second-moment identity to the tangent pCN proposal gives $\mathbb{E}[\|v\|^2 \mid z] = \| \gamma P_z(b) \|^2 + \mathrm{tr}(\beta^2 P_z)$ with $\mathrm{tr}(P_z) = d-1$ gives \eqref{eq:v2_mean}. Averaging under $\pi$, we have $M_d = \beta^2(d-1) + \gamma^2 \mathbb{E}_\pi[\|P_z(b)\|^2]$. We can write $\|P_z(b)\|^2=\delta^2 d(1-t^2)$. Since $t \xrightarrow{\makebox[1em][c]{$\scriptstylep$}} t^*$ by Lemma~\ref{lem:concentration_t} and $0 \le1 - t^2 \le 1$, bounded convergence gives $\mathbb{E}_\pi[1-t^2] \to 1-(t^*)^2$. Therefore, $\mathbb{E}_\pi[\|P_z(b)\|^2] = \delta^2 d\,\mathbb{E}_\pi[1-t^2] = \delta^2 d(1-(t^*)^2)(1+o(1)) = \tau^2 d(1+o(1))$. Since $\gamma^2=\beta^4 / 4+O(\beta^6)$, it follows that $\gamma^2\mathbb{E}_\pi[\|P_z(b)\|^2]=O(\beta^4 d)$, and hence $M_d=\beta^2(d-1)\bigl(1+O(\beta^2)\bigr)$. Under $\beta=\ell d^{-1/6}$, this becomes $M_d=\ell^2 d^{2/3}(1+o(1))$, proving \eqref{eq:Md_leading}.

For the total variance, basis decomposition gives $\mathrm{Var}(\|v\|^2 \mid z) = 2\beta^4(d-1) + 4\beta^2\gamma^2\|P_z(b)\|^2$, using $\mathrm{Var}(X^2)=2\sigma^4+4\sigma^2\mu^2$ for $X\sim\mathcal N(\mu,\sigma^2)$. We will use $\mathbb{E}_\pi[\|P_z(b)\|^2] = O(d)$ and $\mathrm{Var}_\pi(\|P_z(b)\|^2) = O(d)$, both of which follow from $\|P_z(b)\|^2 = \delta^2 d(1-t^2)$ together with Lemma~\ref{lem:concentration_t}. By the law of total variance, the first term contributes
\begin{equation*}
  \mathbb{E}_\pi[\mathrm{Var}(\|v\|^2 \mid z)]
  = 2\beta^4(d-1) + 4\beta^2\gamma^2\mathbb{E}_\pi[\|P_z(b)\|^2]
  = 2\beta^4(d-1)(1+o(1)),
\end{equation*}
using $\gamma^2 \mathbb{E}_\pi[\|P_z(b)\|^2] = O(\beta^4 d)$, and the second contributes
\begin{equation*}
  \mathrm{Var}_\pi(\mathbb{E}[\|v\|^2 \mid z]) = \gamma^4 \mathrm{Var}_\pi(\|P_z(b)\|^2)
  = O(\beta^8 d) = o(\beta^4 d).
\end{equation*}
Hence $\mathrm{Var}_\pi(\|v\|^2) = 2\beta^4(d-1)(1+o(1))$, proving \eqref{eq:v2_total_var}. Finally, since $\|v\|^2/M_d - 1$ has mean zero,
\begin{equation*}
  \mathbb{E}_\pi\,\left[\left(\frac{\|v\|^2}{M_d} - 1\right)^2\right]
  = \frac{\mathrm{Var}_\pi(\|v\|^2)}{M_d^2}
  = \frac{O(\beta^4 d)}{O(\beta^4 d^2)}
  = O(d^{-1}),
\end{equation*}
proving \eqref{eq:eta_L2}.

Under the critical scaling $\beta = \ell d^{-1/6}$,

  = M_d \cdot \mathbb{E}_\pi[a(z,v)] + o(M_d)\,,

<!-- pf-10: body_md -->
Write $\|v\|^2 = M_d(1 + \eta_d)$ where $\eta_d := \|v\|^2/M_d - 1$. Then
\[

  = M_d \mathbb{E}_\pi[a] + M_d \mathbb{E}_\pi[\eta_d \cdot a].

  \le \mathbb{E}_\pi[\eta_d^2]^{1/2}

A proposal with $\|v\|\geq r$ is rejected by the convention of Section~\ref{sec:exact-metr-defect}, so $a(z,v)=0$ on $\{\|v\| \geq r\}$ and the integrand in \eqref{eq:ESJD_def} vanishes there; all expectations below thus reduce to integrals over $\{\|v\| <  r\}$, where Lemma~\ref{lem:geodesic_identity} applies. \begin{equation}

  = \mathbb{E}_\pi\bigl[\|v\|^2\,a(z,v)\bigr]

  \le \mathbb{E}_\pi\bigl[|d_{\mathrm{geo}}^2-\|v\|^2|\,\mathds 1_{\{\|v\| < r\}}\bigr]
  \le \frac{C}{r^2}\,\mathbb{E}_\pi[\|v\|^4].

\end{equation*}

\end{equation*}

\label{thm:joint_diffusion_limit}

 \label{eq:joint_process_convergence}

\end{equation}

\label{rmk:K_properties}
We have $K(0)=1$ and $K'(\sigma)=\sigma\Phi(-\sigma/2)-2\phi(\sigma/2)<0$ for $\sigma\ge0$ by Mills' inequality, so $K$ is strictly decreasing. We also have that $K(\sigma) <2\Phi(-\sigma/2)$ for $\sigma>0$ by the same inequality. Considering the different coordinate scales, the speeds satisfy
\begin{equation}
    \frac{h(\ell)}{(\tau/(\delta\kappa))^2h_\perp(\ell)} =\frac{K(\sigma_\ell)}{2\Phi(-\sigma_\ell/2)}<1.
\end{equation}
The leading longitudinal increment and the log acceptance ratio depend on the same Gaussian variable $\xi_1$ and hence cannot be treated as asymptotically independent. In a transverse direction, the leading noise is independent of $\xi_1$, its odd moments against the acceptance indicator vanish, and its acceptance-weighted second moment converges to the acceptance probability. This gives the factor $K$ in $h$ and the factor $2\Phi(-\sigma_\ell/2)$ in $h_\perp$.

\label{thm:fixed_step}
Fix $\delta,\kappa>0$ and $\beta\in(0,1)$. Consider the chain generated by the intrinsic proposal \eqref{eq:intrinsic_proposal}, started in stationarity, $z_0\sim\pi_d$. Set $g=J_s=0$ and $R=-\infty$ on projection failure.
\begin{enumerate}
\item[(i)] The projection failure probability tends to zero, and
\begin{equation}
  \frac{g}{\sqrt d}\xrightarrow{\makebox[1em][c]{$\scriptstyled$}}\bar g,
  \qquad
  \sqrt d\,J_s\xrightarrow{\makebox[1em][c]{$\scriptstyled$}}\bar J_s,
  \qquad
  R\xrightarrow{\makebox[1em][c]{$\scriptstyled$}}\bar R,
  \label{eq:fixed_step_limit}
\end{equation}
jointly. Here $\zeta\sim\mathcal{N}(0,\sigma_{t^*}^2)$ and $\xi_1,\omega\sim\mathcal{N}(0,1)$ are independent, and
\begin{align}
  \bar g &=
  \delta\kappa\frac{\rho^2-q}{q}\,\zeta +\frac{\rho\beta\tau}{q\sqrt s}\,\xi_1 +\frac{\beta^2\tau^2}{\sqrt2\,q\,s}\,\omega,
\label{eq:gbar}\\
  \bar J_s &=
  \frac1s\left[\frac{1-\rho}{1+\rho}\left(\frac{\delta\rho}{\kappa q}\,\zeta +\frac{\beta\tau}{\sqrt s\,\kappa^2q}\,\xi_1\right) +\frac{\gamma}{\sqrt2\,\kappa^2q}\,\omega \right],
\label{eq:Jbar}\\
  \bar R &=
  \bar J_s\bar g +\frac{1-\rho}{1+\rho} \frac{\bar g^2}{2s\kappa^2}.
\label{eq:Rbar}
\end{align}
Consequently, the stationary acceptance probability satisfies
\begin{equation}
  a_d(\beta)\longrightarrow \mathbb{E}[1 \wedge e^{\bar R}]\in(0,1).
  \label{eq:fixed_step_acceptance}
\end{equation}
\item[(ii)] For every fixed $m\in\mathbb N$, the process $(\zeta_k,X_k)_{0\le k\le m}$ converges in law to the stationary Markov chain whose initial law is given by $(\zeta_0,X_0) \sim \mathcal{N}(0,\sigma_{t^*}^2) \otimes \mathcal{N}(0,s^{-1})$ and the following transition: from the current state $(\zeta_k,X_k)$, draw independent $\xi_1,\xi_2, \omega\sim\mathcal{N}(0,1)$ and $u\sim U(0,1)$. \begin{equation}

    & \log u<\bar R,\\[1.5ex]

  \end{cases}
\label{eq:limit_kernel}
\end{equation}
The limiting kernel is reversible with respect to its initial law.
\end{enumerate}

Set $a:=C_*\tau\ell^3$ so that $R_0=a\xi_1-a^2 /2$ and $\sigma_\ell=|a|$. The transformation $(\xi_1,a)\mapsto(-\xi_1,-a)$ preserves the law of $\xi_1$ and leaves $R_0$ unchanged, while $\xi_1-\frac a2 \,\mapsto\, -\left(\xi_1-\frac a2\right)$. So identities involving the square of this quantity are unchanged and the identities involving its first power change sign on both
sides. It therefore suffices to treat $a>0$, and then $\sigma_\ell=a$.

For $a>0$, $R_0<0$ exactly when $\xi_1<a/2$ and so $\mathcal A(R_0) = \mathds1_{\{\xi_1\ge a/2\}} +e^{R_0}\mathds1_{\{\xi_1<a/2\}}$. Completing the square gives $e^{ax-a^2/2}\phi(x)=\phi(x-a)$. Consequently, for each of the polynomial functions $F$ used below,
\begin{equation}
\mathbb{E}\!\left[F(\xi_1)\mathcal A(R_0)\right] = \int_{a/2}^{\infty}F(x)\phi(x)\,dx +\int_{-\infty}^{-a/2}F(y+a)\phi(y)\,dy,
\label{eq:acceptance_master}
\end{equation}
where the second integral follows from the substitution $y=x-a$. Taking $F\equiv1$ in \eqref{eq:acceptance_master} proves \eqref{eq:A_moment_diffusion}. Taking $F(x)=x-a/2$ and substituting $y=-x$ in the second integral of \eqref{eq:acceptance_master}, the two integrals cancel and so we have $\mathbb{E}\!\left[\left(\xi_1-\frac a2\right)\mathcal A(R_0)\right]=0$, showing~\eqref{eq:A_first_moment_disappears}.
For \eqref{eq:K_moment_diffusion}, set $c:=a/2$ and take

the two integrals are equal, and using truncated Gaussian moments we get
\begin{align*}
\mathbb{E}\!\left[\left(\xi_1-\frac a2\right)^2\mathcal A(R_0)\right] &= 2\int_c^\infty(x-c)^2\phi(x)\,dx\\
&=2\bigl[(1+c^2)\Phi(-c)-c\phi(c)\bigr]\\
&=\left(2+\frac{\sigma_\ell^2}{2}\right) \Phi(-\sigma_\ell/2) -\sigma_\ell\phi(\sigma_\ell/2) = K(\sigma_\ell)\,.
\end{align*}
The random variable $(\xi_1-a/2)^2\mathcal A(R_0)$ is nonnegative and a.s.~not zero, so $K(\sigma_\ell)>0$.

\begin{equation}
\mathbb{E}\!\left[\left(\xi_1-\frac a2\right)e^{R_0}\mathds1_{\{R_0<0\}}\right]= \int_{-\infty}^{-a/2}\left(y+\frac a2\right)\phi(y)\,dy = \frac a2\Phi(-a/2)-\phi(a/2).
\end{equation}

\begin{enumerate}
    \item[(i)] for every $R,\eta>0$, the quantities $\mathsf b_d$, $\mathsf a_d$ and $\mathsf q_d^\eta$ converge uniformly on $\{|x|\le R\}$ to $b(x)$, $a(x)$ and zero respectively, where $b:\mathds{R}^n \to \mathds{R}^n$ and $a:\mathds{R}^n \to \mathds{R}^{n \times n}$ are continuous, with $a(x)$ symmetric and nonnegative definite;
    \item[(ii)] the $C([0,\infty),\mathds{R}^n)$ martingale problem for
    \begin{equation*}
        \mathcal LF(x)= \sum_{i=1}^n b_i(x)\partial_i F(x)+\tfrac12 \sum_{i,j=1}^n a_{ij}(x)\partial_{ij} F(x), \qquad F\in C_c^\infty(\mathds{R}^n),
    \end{equation*}
    has a unique solution for every initial law in $\mathcal P(\mathds{R}^n)$;
    \item[(iii)] there exists $\nu\in\mathcal P(\mathds{R}^n)$ such that $Y_0^{(d)}\xrightarrow{\makebox[1em][c]{$\scriptstyled$}}\nu$.
\end{enumerate}
Then, as $d\to\infty$,
\begin{equation*}
\bigl(Y^{(d)}_{\lfloor c_du\rfloor}\bigr)_{u\ge0}\;\Rightarrow\;Y \quad\text{in }D([0,\infty),\mathds{R}^n),
\end{equation*}
where $Y$ solves the martingale problem for $(\mathcal L,\nu)$.
