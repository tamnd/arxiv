<!-- theorem-1: body_md -->
\label{thm:letter-rate}
For any two finite-dimensional Lindbladians,
\begin{align}
 \gamma(&\mathcal L_1\|\mathcal L_2)=\sup_{\rho_{AE},\sigma_{AE}}
 \left.\frac{\mathrm d}{\mathrm d s}
 D\!\left(e^{s\mathcal L_1}(\rho_{AE})\middle\|e^{s\mathcal L_2}(\sigma_{AE})\right)\right|_{s=0}\nonumber\\
 &=\sup_{\rho_{AE},\sigma_{AE}}\Bigl(
 \operatorname{Tr}\bigl[\mathcal L_1(\rho_{AE})
\bigl(\log\rho_{AE}-\log\sigma_{AE}\bigr)\bigr]\nonumber\\
 &\qquad-\operatorname{Tr}\bigl[\rho_{AE}\,\Omega_{\sigma_{AE}}
 \bigl(\mathcal L_2(\sigma_{AE})\bigr)\bigr]\Bigr),
 \label{eq:differential}
\end{align}
where the supremum ranges over all finite-dimensional auxiliary systems $E$ and positive definite states $\rho_{AE},\sigma_{AE}$.\footnote{For simplicity, we write $\mathcal L_i$ and $e^{s\mathcal L_i}$ for $\mathcal L_i\otimes\operatorname{id}_E$ and $e^{s\mathcal L_i}\otimes\operatorname{id}_E$ in \eqref{eq:differential}.}
Here $\Omega_{\sigma_{AE}}$ is the derivative of the operator logarithm at $\sigma_{AE}$, defined in \eqref{sm:omega}.

<!-- theorem-2: body_md -->
\label{thm:letter-span}
The following conditions are equivalent:
\begin{enumerate}
 \item $\gamma(\mathcal L_1\|\mathcal L_2)<\infty$,
 \item\label{item:letter-span-algebraic} $L_j^{(1)}\in\mathcal S_2$ for all $j$, and $G_1-G_2\in\mathcal S_2$,
 \item There exists $0<\varepsilon<1$ such that $\beta_{\varepsilon,t}(\mathcal L_1\|\mathcal L_2)>0$ for all $t\ge0$.
\end{enumerate}
Furthermore, if these conditions hold, then for $0\le\varepsilon<1$ and all $t\ge0$,
\begin{equation}
 \beta_{\varepsilon,t}(\mathcal L_1\|\mathcal L_2)\ge 2^{-[t\gamma(\mathcal L_1\|\mathcal L_2) +h(\varepsilon)]/(1-\varepsilon)},
 \label{eq:finitefloor}
\end{equation}
where $h(\varepsilon)=-\varepsilon\log\varepsilon-(1-\varepsilon)\log(1-\varepsilon)$ is the binary entropy.

<!-- theorem-3: body_md -->
\label{thm:letter-resolution}
Suppose \eqref{eq:choi} holds and $\gamma(\mathcal L_1\|\mathcal L_2)=\infty$.
Then for all $0<\varepsilon<1$, there exist $s_0>0$, $C_\varepsilon\ge c_\varepsilon>0$ and $t_\varepsilon<\infty$, depending only on the generators and $\varepsilon$, such that
\begin{equation}
 \left(\frac{s_{\min}}{2s_0}\right)^{C_\varepsilon t}
 \le\beta^{(s_{\min})}_{\varepsilon,t}(\mathcal L_1\|\mathcal L_2)
 \le\left(\frac{s_{\min}}{2s_0}\right)^{c_\varepsilon t}
 \label{eq:resolutionlaw}
\end{equation}
for all $t\ge t_\varepsilon$ and $0<s_{\min}\le s_0$.
Moreover, the threshold $t_\varepsilon$ can be chosen to satisfy $t_\varepsilon\le K\,(\,1+\log(1/\varepsilon))$, for some constant $K\ge 0$ depending only on $\mathcal L_1$ and $\mathcal L_2$.

In particular, this gives $\beta^{(s_{\min})}_{\varepsilon,t}(\mathcal L_1\|\mathcal L_2)>0$ for all $s_{\min}>0$ and $t\ge0$ and further $\beta_{\varepsilon,t}(\mathcal L_1\|\mathcal L_2) = \lim_{s_{\min}\to 0} \beta^{(s_{\min})}_{\varepsilon,t}(\mathcal L_1\|\mathcal L_2)  =0$ for all $t\ge t_\varepsilon.$

<!-- theorem-5: body_md -->
\label{thm:rate}
The Lindbladian Stein exponent satisfies
\begin{align}
 \nonumber \gamma(\mathcal L_1\|\mathcal L_2)
 &=\sup_{s>0}\frac{D^{\mathcal A}\left(e^{s\mathcal L_1}\middle\|e^{s\mathcal L_2}\right)}s
   =\lim_{s\to0}\frac{D^{\mathcal A}\left(e^{s\mathcal L_1}\middle\|e^{s\mathcal L_2}\right)}s \\
 &=\sup_{E,\rho_{AE},\sigma_{AE}>0}
   \left.\frac{\mathrm d}{\mathrm d s}
       D\left(\left(e^{s\mathcal L_1}\otimes\operatorname{id}_E\right)(\rho_{AE})\|
         \left(e^{s\mathcal L_2}\otimes\operatorname{id}_E\right)(\sigma_{AE})\right)\right|_{s=0}.
 \label{sm:diff}
\end{align}
The supremum in \eqref{sm:diff} ranges over finite-dimensional auxiliary systems $E$ and positive definite states $\rho_{AE},\sigma_{AE}$ on $AE$.
Moreover, replacing the inner $\liminf_{t\to\infty}$ in the definition of $\gamma(\mathcal L_1\|\mathcal L_2)$ by $\limsup_{t\to\infty}$ gives the same result.
Furthermore, for every fixed $s_{\min}>0$, the optimal type-II error probability $\beta^{(s_{\min})}_{\varepsilon,t}(\mathcal L_1\|\mathcal L_2)$ defined in \eqref{eq:beta} satisfies
\begin{align*}
 \Gamma_{s_{\min}}(\mathcal L_1\|\mathcal L_2)
 &=\lim_{\varepsilon\to0}\liminf_{t\to\infty}
 -\frac1t\log\beta^{(s_{\min})}_{\varepsilon,t}(\mathcal L_1\|\mathcal L_2)=\lim_{\varepsilon\to0}\limsup_{t\to\infty}
 -\frac1t\log\beta^{(s_{\min})}_{\varepsilon,t}(\mathcal L_1\|\mathcal L_2).
\end{align*}
Moreover, whenever $\Gamma_{s_{\min}}(\mathcal L_1\|\mathcal L_2)<\infty$, we have, for all $0\le\varepsilon<1$,
\begin{equation}
 \beta^{(s_{\min})}_{\varepsilon,t}(\mathcal L_1\|\mathcal L_2)\ge
 2^{-[t\Gamma_{s_{\min}}(\mathcal L_1\|\mathcal L_2)+h(\varepsilon)]/(1-\varepsilon)}.
 \label{sm:telescopebound}
\end{equation}
The same bound with $\gamma(\mathcal L_1\|\mathcal L_2)$ holds for $\beta_{\varepsilon,t}(\mathcal L_1\|\mathcal L_2)$ whenever $\gamma(\mathcal L_1\|\mathcal L_2)<\infty$.

<!-- proof-1: body_md -->
We first establish achievability using parallel strategies, which are a special case of the adaptive protocols considered here.
Fix $s>0$ and prepare a state on $A^nE$, involving $n$ probes and an auxiliary system $E$.
Apply the unknown dynamics to each probe for time $s$, followed by a joint binary measurement.
This applies $(e^{s\mathcal L_i})^{\otimes n}\otimes\operatorname{id}_E$ under hypothesis $i$ and uses total evolution time $ns$.
It can be implemented sequentially by storing the remaining probes and all outputs in the auxiliary system and swapping each probe into $A$ for its evolution period~\cite{Bergh2024}.

The negative logarithm of the optimal type-II error probability for this strategy is the hypothesis-testing relative entropy of the tensor-power channels.
Its asymptotic rate is the regularised channel relative entropy~\cite{WangWilde2019Channels}:
\begin{equation*}
 \lim_{\varepsilon\to0}\liminf_{n\to\infty}
 \frac1n D_H^\varepsilon\left(\mathcal N^{\otimes n}\middle\|\mathcal M^{\otimes n}\right)
 =D^{\mathrm{reg}}(\mathcal N\|\mathcal M).
\end{equation*}
For a total time budget $t$, choose $n=\lfloor t/s\rfloor$.
Since these parallel strategies are admissible adaptive protocols,
\begin{equation*}
 -\log\beta_{\varepsilon,t}(\mathcal L_1\|\mathcal L_2)
 \ge  D_H^\varepsilon\left(\left(e^{s\mathcal L_1}\right)^{\otimes n}
 \middle\|\left(e^{s\mathcal L_2}\right)^{\otimes n}\right),
\end{equation*}
Dividing by $t$ and taking limits, using $1/s-1/t<n/t\le1/s$ and \eqref{sm:chain}, gives
\begin{align*}
 \gamma(\mathcal L_1\|\mathcal L_2)
 &=\lim_{\varepsilon\to0}\liminf_{t\to\infty}
 -\frac1t\log\beta_{\varepsilon,t}(\mathcal L_1\|\mathcal L_2)\\
 &\ge\lim_{\varepsilon\to0}\liminf_{t\to\infty}
 \frac nt\,\frac1n D_H^\varepsilon\left(\left(e^{s\mathcal L_1}\right)^{\otimes n}
 \middle\|\left(e^{s\mathcal L_2}\right)^{\otimes n}\right)\\
 &=\frac1sD^{\mathrm{reg}}\left(e^{s\mathcal L_1}\middle\|e^{s\mathcal L_2}\right)
 =\frac1sD^{\mathcal A}\left(e^{s\mathcal L_1}\middle\|e^{s\mathcal L_2}\right).
\end{align*}
Taking the supremum over $s>0$ gives the required lower bound.
At fixed positive resolution time, the same argument with $s\ges_{\min}$ gives the lower bound $\Gamma_{s_{\min}}(\mathcal L_1\|\mathcal L_2)$.

For the converse, first consider a protocol with fixed durations $s_1,\ldots,s_n$ and common control channels $\mathcal A_k$ on $AE$.
Starting from $\rho_{0,AE}^{(1)}=\rho_{0,AE}^{(2)}=\omega_{AE}$, the states after the $k$th evolution and control operation are, for $i=1,2$,
\begin{equation*}
 \rho_{k,AE}^{(i)}
 =\mathcal A_k\left(\left(e^{s_k\mathcal L_i}\otimes\operatorname{id}_E\right)
 (\rho_{k-1,AE}^{(i)})\right).
\end{equation*}
Data processing through $\mathcal A_k$ and the definition of amortised relative entropy give
\begin{align*}
 D(\rho_{k,AE}^{(1)}\|\rho_{k,AE}^{(2)})
 &\le
 D\left(\left(e^{s_k\mathcal L_1}\otimes\operatorname{id}_E\right)(\rho_{k-1,AE}^{(1)})
 \middle\|\left(e^{s_k\mathcal L_2}\otimes\operatorname{id}_E\right)(\rho_{k-1,AE}^{(2)})\right)
 \\
 &\le D(\rho_{k-1,AE}^{(1)}\|\rho_{k-1,AE}^{(2)})
 +D^{\mathcal A}\left(e^{s_k\mathcal L_1}\middle\|e^{s_k\mathcal L_2}\right).
\end{align*}
If the preceding relative entropy is infinite, the second inequality is trivial. Otherwise, it follows directly from the definition of $D^{\mathcal A}$.
The initial relative entropy is zero, so iterating gives
\begin{equation*}
 D(\rho_{n,AE}^{(1)}\|\rho_{n,AE}^{(2)})
 \le\sum_{k=1}^nD^{\mathcal A}\left(e^{s_k\mathcal L_1}\middle\|e^{s_k\mathcal L_2}\right).
\end{equation*}

The same argument applies when the evolution durations $s_k$ and the number of calls are chosen adaptively based on preceding measurement outcomes.
At each stage, retain the preceding classical history $h$ in a separate classical register $C$ and write its probabilities as $p_i(h)$ and its conditional states as $\rho_{h,AE}^{(i)}$.
The direct-sum states act on $CAE$, and their relative entropy is
\begin{equation*}
 D\!\left(\bigoplus_h p_1(h)\rho_{h,AE}^{(1)}\,\middle\|\,
           \bigoplus_h p_2(h)\rho_{h,AE}^{(2)}\right)
 =D(p_1\|p_2)+\sum_h p_1(h)D(\rho_{h,AE}^{(1)}\|\rho_{h,AE}^{(2)}).
\end{equation*}
Applying the same additive bound to the conditional states bounds the relative entropy after a history-dependent call by its value before the call plus
$\sum_h p_1(h)D^{\mathcal A}\left(e^{s_h\mathcal L_1}\middle\|e^{s_h\mathcal L_2}\right)$.
Each subsequent instrument is a common channel on the state and record and cannot increase the total relative entropy.
Iterating therefore gives an expectation under the null hypothesis, denoted by $\mathbb E_1$.
Since the total time is at most $t$ on every branch, for $t>0$ we obtain
\begin{align*}
 D(\rho_{AE}^{(1),\mathsf P}\|\rho_{AE}^{(2),\mathsf P})
 &\le\mathbb E_1\!\left[\sum_kD^{\mathcal A}\left(e^{s_k\mathcal L_1}\middle\|e^{s_k\mathcal L_2}\right)\right]\le t\sup_{s>0}\frac{D^{\mathcal A}\left(e^{s\mathcal L_1}\middle\|e^{s\mathcal L_2}\right)}s.
\end{align*}
Nodes with zero probability under the null hypothesis make no contribution.
At fixed positive resolution time, the supremum is restricted to $s\ges_{\min}$ and hence equals $\Gamma_{s_{\min}}(\mathcal L_1\|\mathcal L_2)$.

To turn this entropy bound into an error bound, use the relation between hypothesis-testing and quantum relative entropy~\cite{WangRenner2012}.
For every protocol $\mathsf P$ with $\alpha(\mathsf P)\le\varepsilon$,
\begin{align*}
 -\log\beta(\mathsf P)
 &\le D_H^\varepsilon\left(\rho_{AE}^{(1),\mathsf P}\middle\|\rho_{AE}^{(2),\mathsf P}\right)
 \le\frac{D(\rho_{AE}^{(1),\mathsf P}\|\rho_{AE}^{(2),\mathsf P})+h(\varepsilon)}{1-\varepsilon}
 \\
 &\le\frac{t\displaystyle\sup_{s>0}\frac1sD^{\mathcal A}\left(e^{s\mathcal L_1}\middle\|e^{s\mathcal L_2}\right)+h(\varepsilon)}{1-\varepsilon}.
\end{align*}
Optimising over all admissible protocols gives the corresponding lower bound on the optimal type-II error probability, with the supremum restricted to $s\ges_{\min}$ for \eqref{sm:telescopebound}.
Dividing by $t$, taking $\limsup_{t\to\infty}$ and then $\varepsilon\to0$ gives the upper bound $\sup_{s>0}D^{\mathcal A}\left(e^{s\mathcal L_1}\middle\|e^{s\mathcal L_2}\right)/s$ on the iterated exponent, or $\Gamma_{s_{\min}}(\mathcal L_1\|\mathcal L_2)$ at fixed positive resolution time.
Together with achievability, this proves the operational equality with $\sup_{s>0}D^{\mathcal A}\left(e^{s\mathcal L_1}\middle\|e^{s\mathcal L_2}\right)/s$, the fixed-resolution statement, and the equality of the iterated limits using $\liminf$ or $\limsup$.
If the corresponding supremum is infinite, achievability already proves these assertions.
Finally, Lemma~\ref{lem:amort-differential} below identifies the unrestricted supremum with the small-time limit and the differential in \eqref{sm:diff}, completing the proof.

\label{lem:amort-differential}
For two finite-dimensional Lindbladians,
\begin{align}
 \sup_{s>0}\frac{D^{\mathcal A}\left(e^{s\mathcal L_1}\middle\|e^{s\mathcal L_2}\right)}s

\end{align}

 p_0=\frac{p(\tau)}2,\qquad t_\varepsilon=\tau\left(2+\frac8{p_0}\log\frac1\varepsilon\right),\qquad c=\frac{p_0}{4\tau}.
 \label{sm:amplification-parameters}
\end{equation}

The sum is strictly positive because the term corresponding to $X$ is nonzero by our choice of $|v\rangle$.

Suppose that, for all $s>0$,
\begin{equation}
 \operatorname{supp} J^{(1)}_{AA'}(s)\subseteq\operatorname{supp} J^{(2)}_{AA'}(s).
 \label{sm:support}
\end{equation}
There are constants $C,s_*>0$, depending only on the generators, such that, for all $0<s\le s_*$,
\begin{equation}
 \frac{D^{\mathcal A}\left(e^{s\mathcal L_1}\middle\|e^{s\mathcal L_2}\right)}s\le C[1+\log(s_*/s)].
 \label{sm:localupper}
\end{equation}
Moreover, if $\gamma(\mathcal L_1\|\mathcal L_2)=\infty$, the reverse inequality holds with another positive constant for all sufficiently small $s$.
Furthermore, in this case of infinite Stein exponent, for all fixed $0<\varepsilon<1$, there are $s_0>0$, $0<c_\varepsilon\le C_\varepsilon$ and $t_\varepsilon<\infty$, depending only on the generators and $\varepsilon$, such that
\begin{equation}
 \left(\frac{s_{\min}}{2s_0}\right)^{C_\varepsilon t}
 \le\beta^{(s_{\min})}_{\varepsilon,t}(\mathcal L_1\|\mathcal L_2)
 \le\left(\frac{s_{\min}}{2s_0}\right)^{c_\varepsilon t},
 \label{sm:jointlaw}
\end{equation}
where $t\ge t_\varepsilon$ and $0<s_{\min}\le s_0$.
Moreover, the threshold $t_\varepsilon$ can be chosen to satisfy $t_\varepsilon\le K[1+\log(1/\varepsilon)]$, where $K>0$ depends only on the generators.
Furthermore, for all $s_{\min}>0$ and $t\ge0$, $\beta^{(s_{\min})}_{\varepsilon,t}(\mathcal L_1\|\mathcal L_2)>0$.
In particular, when $\gamma(\mathcal L_1\|\mathcal L_2)=\infty$, the upper bound in \eqref{sm:jointlaw} gives $\beta_{\varepsilon,t}(\mathcal L_1\|\mathcal L_2)=\lim_{s_{\min}\to0}\beta^{(s_{\min})}_{\varepsilon,t}(\mathcal L_1\|\mathcal L_2)=0$ for all $t\ge t_\varepsilon$.

<!-- lemma-2: body_md -->
\label{lem:rank}
Let $\mathcal L$ be  finite-dimensional Lindbladian. Then we have that the rank of $J_{AA'}\left(e^{s\mathcal L}\right)$ is constant on $s>0$.
<!-- end lemma-2 -->
<!-- proof-5: body_md -->
To compare the Choi ranks at different times, we retain the no-jump contribution during the additional evolution interval.
Write $\mathcal L(X)=GX+XG^*+\mathcal J(X)$, where $\mathcal J(X)=\sum_jL_jXL_j^*$, and define $\mathcal F_u(X)=e^{uG}Xe^{uG^*}$ for $u\ge0$.
Duhamel's formula gives
\begin{equation*}
 e^{u\mathcal L}=\mathcal F_u+\int_0^u\mathcal F_{u-v}\circ\mathcal J\circ e^{v\mathcal L}\,\mathrm d v.
\end{equation*}
Every map in the integrand is completely positive, so $e^{u\mathcal L}-\mathcal F_u$ is completely positive.
Consequently, applying $(e^{u\mathcal L}-\mathcal F_u)\otimes\operatorname{id}_{A'}$ to the positive operator $J_{AA'}\left(e^{s\mathcal L}\right)$ gives a positive operator.
Using the semigroup property and the definition of the Choi state, we therefore obtain, for $s,u>0$,
\begin{align*}
 J_{AA'}\left(e^{(s+u)\mathcal L}\right)
 &=\left(e^{u\mathcal L}\otimes\operatorname{id}_{A'}\right)\left(J_{AA'}\left(e^{s\mathcal L}\right)\right)\\
 &\ge\left(\mathcal F_u\otimes\operatorname{id}_{A'}\right)\left(J_{AA'}\left(e^{s\mathcal L}\right)\right)\\
 &=(e^{uG}\otimes\1_{A'})J_{AA'}\left(e^{s\mathcal L}\right)(e^{uG^*}\otimes\1_{A'}).
\end{align*}
Since $e^{uG}$ is invertible, the preceding inequality shows that the rank of the Choi state $J_{AA'}\left(e^{s\mathcal L}\right)$ cannot decrease with $s$.
Suppose it increased from $r$ at some time $s_1>0$ to a larger value at $s_2>s_1$.
Then some minor of order $r+1$, meaning the determinant of an $(r+1)\times(r+1)$ submatrix, would be nonzero at $s_2$.
However, the rank is at most $r$ for $0<s<s_1$, so this minor vanishes throughout that interval.
This is impossible because each minor is real analytic in $s$, and an analytic function that vanishes on an open interval must vanish identically.

<!-- proof-6: body_md -->
To prove the lower bound on the optimal type-II error probability in \eqref{sm:jointlaw}, we establish the relative entropy estimate \eqref{sm:localupper}.
For one channel use, support inclusion ensures that an outcome possible under the null hypothesis is also possible under the alternative hypothesis.
To quantify how much smaller its probability can be under the alternative hypothesis, we first bound the smallest positive Choi eigenvalue under that hypothesis from below.
Let $J^{(i)}_{AA'}(s)$ denote the normalised Choi state of the channel under hypothesis $i$ after exposure time $s>0$.
Let $r$ denote the rank of $J^{(2)}_{AA'}(s)$, which is constant for positive $s$ by Lemma~\ref{lem:rank}.
Consider the product $P(s)=\prod_{j=1}^r\lambda_j(s)$, where $\lambda_1(s),\ldots,\lambda_r(s)$ are its positive eigenvalues.
This product equals the sum of all principal minors of order $r$, meaning determinants of submatrices obtained by selecting the same $r$ row and column indices, so it extends analytically to $s=0$, where the rank may drop.
Since $P(s)>0$ for $s>0$, its Taylor expansion at zero has a first nonzero term.
Every eigenvalue is at most one, so the smallest positive eigenvalue $\lambda_{\min}^+$ is at least $P(s)$.
Consequently, there are $s_*,a_*>0$ and an integer $q\ge1$ such that, for $0<s\le s_*$,
\begin{equation}
 \lambda_{\min}^+\left(J^{(2)}_{AA'}(s)\right)\ge P(s)\ge a_*\ \left(\frac{s}{s_*}\right)^q.
 \label{sm:choieigen}
\end{equation}

Define $\mathcal R_s=e^{s\mathcal L_1}-e^{sG_1}(\,\cdot\,)e^{sG_1^*}$, the remainder after subtracting the no-jump contribution, which is completely positive by Duhamel's formula as in the proof of Lemma~\ref{lem:rank}.
Let $d_G=\left\lVertG_1-G_2\right\rVert_\infty$ and
$\kappa=\left\lVert\sum_jL_j^{(1)*}L_j^{(1)}\right\rVert_\infty$.
The operators $e^{sG_i}$ are contractions in operator norm.\footnote{Since $G_i+G_i^*=-\sum_jL_j^{(i)*}L_j^{(i)}$, for $s\ge0$ and any vector $|\phi\rangle$ we have $\frac{\mathrm d}{\mathrm d s}\|e^{sG_i}|\phi\rangle\|^2=-\sum_j\|L_j^{(i)}e^{sG_i}|\phi\rangle\|^2\le0$.
Thus $\|e^{sG_i}|\phi\rangle\|\le\||\phi\rangle\|$, giving $\left\lVerte^{sG_i}\right\rVert_\infty\le1$.}
Duhamel's formula and trace preservation therefore give
\begin{equation}
 \left\lVerte^{sG_1}-e^{sG_2}\right\rVert_\infty\le d_Gs,\qquad
 \mathcal R_s^*(\1)=\1-e^{sG_1^*}e^{sG_1}\le\kappa s\1.
 \label{sm:nearidentity}
\end{equation}
Choose Kraus operators $M_{s,j}$ for $\mathcal R_s$ and a common finite dilation space $F$ for the two channels, padding the dilations if necessary. Define
\begin{equation*}
 W_s=\left(e^{sG_1}-e^{sG_2}\right)\otimes|0\rangle_F+\sum_jM_{s,j}\otimes|j\rangle_F,\qquad
 V_s(z)=e^{sG_2}\otimes|0\rangle_F+zW_s .
\end{equation*}
The labels $j$ in this expression are orthogonal to $0$.
At $z=1$, $V_s(1)$ is a Stinespring isometry for $e^{s\mathcal L_1}$.
For $|z|=2$, \eqref{sm:nearidentity} yields
\begin{equation*}
 V_s(z)^*V_s(z)
 \le[1+(4d_G+4d_G^2s_*+4\kappa)s]\1 .
\end{equation*}
Thus, for a fixed $a_0\ge0$, we have $\left\lVertV_s(z)\right\rVert_\infty\le2^{a_0s}$ on this circle.

To compare this dilation with the channel under the alternative hypothesis, consider the completely positive map $\mathcal B_s=\left(e^{sG_1}-e^{sG_2}\right)(\,\cdot\,)\left(e^{sG_1^*}-e^{sG_2^*}\right)+\mathcal R_s$.
Its Choi support lies inside $\operatorname{supp} J^{(2)}_{AA'}(s)$.
For $\mathcal R_s$ and the Kraus vector of $e^{sG_1}$ this follows from \eqref{sm:support}. The Kraus vector of $e^{sG_2}$ already belongs to that support, so the same holds for their difference.
Its Choi trace is at most $d_G^2s^2+\kappa s$.
\eqref{sm:choieigen} therefore gives
\begin{equation}
 \mathcal B_s\preceq b_s^2e^{s\mathcal L_2},\qquad
 0\le b_s\le b_*(s_*/s)^{(q-1)/2}
 \label{sm:bs}
\end{equation}
for some finite $b_*$.

Choose a dilation $V_2(s)$ of the channel under the alternative hypothesis with $e^{sG_2}$ as its explicit no-jump Kraus operator.
There are environment maps $A_s,B_s$ with
\begin{equation*}
 (\1_A\otimes A_s)V_2(s)=e^{sG_2}\otimes|0\rangle_F,\quad
 (\1_A\otimes B_s)V_2(s)=W_s,\quad
 \left\lVertA_s\right\rVert_\infty\le1,\quad \left\lVertB_s\right\rVert_\infty\le b_s.
\end{equation*}
For completeness, the second statement is the finite-dimensional factorisation behind completely positive order.
If the Choi factors for $\mathcal B_s$ and $e^{s\mathcal L_2}$ are $XX^*$ and $YY^*$, their order gives $X=YC$ with $\left\lVertC\right\rVert_\infty\le b_s$ by the pseudoinverse on the range of $Y$.
Transposing the coefficient matrix gives the indicated map on Kraus labels.
The first map simply selects the explicit no-jump Kraus operator.

To control the approximation uniformly over inputs, we truncate the joint dilation polynomial at a degree depending only on the total exposure time $ns$.
For any pure input $|\psi\rangle_{A^nE}$ on $A^nE$, identify the joint dilation output $(AF)^nE$ with $A^nEF^n$ by reordering tensor factors, and consider
\begin{align*}
 |\Psi(z)\rangle_{A^nEF^n}&=\left(V_s(z)^{\otimes n}\otimes\1_E\right)|\psi\rangle_{A^nE}
       =\sum_{j=0}^nz^j|\Psi_j\rangle_{A^nEF^n},\qquad
 |\Psi^{(2)}\rangle_{A^nEF^n}=\left(V_2(s)^{\otimes n}\otimes\1_E\right)|\psi\rangle_{A^nE} .
\end{align*}
On $|z|=2$ the norm is at most $2^{a_0ns}$.
Cauchy's coefficient estimate gives
$\||\Psi_j\rangle_{A^nEF^n}\|\le2^{a_0ns-j}$.
Choose $K=\lceil a_0ns+2\rceil$ and truncate the polynomial at degree $K$, or retain it all if $K\ge n$.
The omitted norm at $z=1$ is at most $2^{a_0ns-K}\le1/4$.
The retained vector equals $(\1_{A^nE}\otimes C_{\le K})|\Psi^{(2)}\rangle_{A^nEF^n}$, where
\begin{equation*}
 \left\lVertC_{\le K}\right\rVert_\infty
 \le\sum_{j=0}^{\min(K,n)}\binom{n}{j}b_s^j
 \le e^{ns/s_*}\left[\frac{s_*}{s}(1+b_s)\right]^K .
\end{equation*}
To prove the last inequality, set $x=s/[s_*(1+b_s)]\le1$.
For $j\le K$, $x^j\ge x^K$, and hence the sum is bounded by
$x^{-K}(1+xb_s)^n\le x^{-K}e^{ns/s_*}$.
This estimate avoids a spurious factor $\log n$.

Consider any final binary measurement $\{Q,\1-Q\}$ on the accessible output $A^nE$, where $0\le Q\le\1$, which accepts the null hypothesis for outcome $Q$ and the alternative hypothesis otherwise.
Suppose the probability of outcome $Q$ under the null hypothesis is at least $1/2$.
Its acceptance amplitude on the retained vector is at least
$1/\sqrt2-1/4\ge1/(2\sqrt2)$.
Since $Q$ commutes with the environment map, this amplitude is at most
$\left\lVertC_{\le K}\right\rVert_\infty\sqrt\beta$, where $\beta$ is the probability of outcome $Q$ under the alternative hypothesis, hence the type-II error probability.
It follows that
\begin{equation}
 -\log\beta\le
 3+2(\log e)\frac{ns}{s_*}
 +2\lceil a_0ns+2\rceil
       \log\left[\frac{s_*}{s}(1+b_s)\right].
 \label{sm:polynomialtest}
\end{equation}
All constants are independent of the number of probes and the auxiliary system.

To obtain the regularised relative entropy bound \eqref{sm:localupper}, fix $s$, a block size $k$ and an input in \eqref{sm:chain}.
Repeat the block $r$ times and apply the state Stein lemma with acceptance at least $1/2$.
\eqref{sm:polynomialtest} holds for $n=rk$.
Divide by $n$ and let $r\to\infty$.
Every achievable state error exponent is bounded, so
\begin{align*}
 \frac1kD\left(\left(\left(e^{s\mathcal L_1}\right)^{\otimes k}\otimes\operatorname{id}_E\right)(\omega_{A^kE})\|
                \left(\left(e^{s\mathcal L_2}\right)^{\otimes k}\otimes\operatorname{id}_E\right)(\omega_{A^kE})\right)
 &\le 2(\log e)\frac{s}{s_*}
 +2a_0s\log\left[\frac{s_*}{s}(1+b_s)\right].
\end{align*}
Taking the suprema over $k$ and $\omega_{A^kE}$, using \eqref{sm:chain}, and then \eqref{sm:bs} proves \eqref{sm:localupper}.
The Stein limit was taken at each fixed $s$. No exchange with a simultaneous small-$s$ limit is required.

By \eqref{sm:subdivision}, for $s_{\min}\le s_*/2$ this bounds
$\Gamma_{s_{\min}}(\mathcal L_1\|\mathcal L_2)$ by a constant times $1+\log(s_*/s_{\min})$.
For all $s_{\min}>0$, \eqref{sm:telescopebound} therefore gives positivity for all $t\ge0$, using monotonicity in $s_{\min}$ if $s_{\min}>s_*/2$.

If $\gamma(\mathcal L_1\|\mathcal L_2)=\infty$, choose a common smaller cutoff $s_0$ for this estimate and Theorem~\ref{thm:infinite}.
After adjusting a constant $B>0$ depending only on the generators, \eqref{sm:telescopebound} gives
\begin{equation*}
 -\log\beta^{(s_{\min})}_{\varepsilon,t}(\mathcal L_1\|\mathcal L_2)
 \le\frac{Bt[1+\log(s_0/s_{\min})]+h(\varepsilon)}{1-\varepsilon}.
\end{equation*}
Keep the choice of $t_\varepsilon$ in \eqref{sm:amplification-parameters}, as used in the proof of Theorem~\ref{thm:infinite}.
Since $h(\varepsilon)\le1$ and $t\ge t_\varepsilon\ge2\tau$, the right-hand side is at most $C_\varepsilon t[1+\log(s_0/s_{\min})]$ with $C_\varepsilon=(B+1/(2\tau))/(1-\varepsilon)$.
Exponentiating gives the lower bound in \eqref{sm:jointlaw}, while Theorem~\ref{thm:infinite} gives the upper bound.
Thus the same threshold satisfies the full estimate and its stated logarithmic upper bound.

To obtain the matching logarithmic lower bound on the relative entropy per unit time in this case, use the vectors and $\tau$ from Section~\ref{sec:zeno-general} and take $n=\lfloor\tau/s\rfloor$.
Since $0\le\tau-ns<s$, \eqref{sm:product} and continuous differentiability of $p(u)$ imply that the accepting probabilities satisfy $p\ge p_0=p(\tau)/2$ and $q\le C_1s$ for some $C_1>0$ and sufficiently small $s>0$.
The entropy budget and binary data processing then give, for $C_1s\le p_0/e$,
\begin{equation}
 \frac1sD^{\mathcal A}\left(e^{s\mathcal L_1}\middle\|e^{s\mathcal L_2}\right)
 \ge\frac{D((p,1-p)\|(q,1-q))}{ns}
 \ge\frac{p_0}{\tau}\log\frac{p_0}{eC_1s}.
 \label{sm:event-entropy}
\end{equation}
Here we used $ns\le\tau$ and $D((p,1-p)\|(q,1-q))\ge p\log[p/(eq)]$.
Thus \eqref{sm:event-entropy} supplies the claimed lower bound for all sufficiently small $s$.

Finally, letting $s_{\min}\to0$ in the upper bound in \eqref{sm:jointlaw} shows that the unrestricted optimal type-II error probability vanishes for all $t\ge t_\varepsilon$, since it is no larger than the optimum at any positive resolution time.

\label{lem:support-attainment}
For two finite-dimensional Lindbladians and a adaptive protocols $\mathsf P$ as defined in \eqref{eq:beta}, the following are equivalent:
\begin{enumerate}[label=(\roman*)]
 \item\label{item:support-violation} $\operatorname{supp} J^{(1)}_{AA'}(s)\not\subseteq\operatorname{supp} J^{(2)}_{AA'}(s)$ for some $s>0$,
 \item\label{item:zero-error-attainment} for all $s_{\min}>0$ and $0<\varepsilon<1$, there exist a finite $t\ge0$ and a protocol $\mathsf P$ with total exposure time at most $t$ and resolution time $s_{\min}$ such that $\alpha(\mathsf P)\le\varepsilon$ and $\beta(\mathsf P)=0$.
\end{enumerate}
If instead support inclusion holds at all positive times, then for all protocols $\mathsf P$  with total time $t\ge0$ and time $s_{\min}>0$ fixed and further $\alpha(\mathsf P)<1$ has $\beta(\mathsf P)>0$.
<!-- end lemma-3 -->
<!-- proof-7: body_md -->
We first prove \ref{item:support-violation}$\Rightarrow$\ref{item:zero-error-attainment}.
By assumption, support inclusion fails at some time $s>0$.
We show that, for any prescribed resolution time $s_{\min}>0$, it also fails at some time $\tilde s\ges_{\min}$.
Write $J^{(i)}_{AA'}(u)=J_{AA'}\left(e^{u\mathcal L_i}\right)$ and let $r_2$ be the rank of $J^{(2)}_{AA'}(u)$, which is constant for $u>0$ by Lemma~\ref{lem:rank}.
Since the Choi states are positive, for $u>0$ we have
\begin{equation*}
 \operatorname{supp} J^{(1)}_{AA'}(u)\subseteq\operatorname{supp} J^{(2)}_{AA'}(u)
 \quad\Longleftrightarrow\quad
 \operatorname{rank}\left(J^{(1)}_{AA'}(u)+J^{(2)}_{AA'}(u)\right)=r_2.
\end{equation*}
Since support inclusion fails at $u=s$ by \ref{item:support-violation}, we have $\operatorname{rank}\left(J^{(1)}_{AA'}(s)+J^{(2)}_{AA'}(s)\right)>r_2$.
Hence some minor of order $r_2+1$ of the sum $J^{(1)}_{AA'}(u)+J^{(2)}_{AA'}(u)$ is nonzero at $u=s$.
This minor is real analytic in $u$ and not identically zero, so it cannot vanish throughout $(s_{\min},2s_{\min})$.
Hence support inclusion also fails at some $\tilde s\in(s_{\min},2s_{\min})$, providing an allowed evolution duration.

To construct the protocol $\mathsf P$, let $Q$ be the projection onto $\ker J^{(2)}_{AA'}(\tilde s)$.
In each run, prepare the maximally entangled state $|\Phi\rangle=d^{-1/2}\sum_{a=1}^d|a\rangle_A|a\rangle_{A'}$, with $d=\dim A$ and $A'\simeq A$, and let $A$ evolve under the unknown dynamics for time $\tilde s$, leaving $A'$ unchanged.
The resulting state under hypothesis $i$ is $J^{(i)}_{AA'}(\tilde s)$, which we measure with $\{Q,\1-Q\}$.
By support violation, outcome $Q$ has probability $p=\operatorname{Tr}\left(QJ^{(1)}_{AA'}(\tilde s)\right)>0$ under the null hypothesis and zero under the alternative hypothesis.
Choose an integer $m\ge1$ such that $(1-p)^m\le\varepsilon$, and repeat this preparation, evolution and measurement independently $m$ times.
The protocol accepts the null hypothesis if outcome $Q$ occurs in at least one run, and the alternative hypothesis otherwise.
Thus $\alpha(\mathsf P)=(1-p)^m\le\varepsilon$ and $\beta(\mathsf P)=0$, with total exposure time $t=m\tilde s$ and evolution durations $\tilde s\ges_{\min}$ as required in \eqref{eq:beta}.

We now prove \ref{item:zero-error-attainment}$\Rightarrow$\ref{item:support-violation} by contraposition.
Suppose support inclusion holds at all positive times and fix a protocol $\mathsf P$ with total exposure time at most $t<\infty$, resolution time $s_{\min}>0$ and $\alpha(\mathsf P)<1$.
Choose $0<\varepsilon<1$ with $\alpha(\mathsf P)\le\varepsilon$.
By \eqref{eq:beta} and Theorem~\ref{thm:resolution}, $\beta(\mathsf P)\ge\beta^{(s_{\min})}_{\varepsilon,t}(\mathcal L_1\|\mathcal L_2)>0$.
This also proves the converse implication.

Exactly one of the following three cases occurs:

\label{thm:identity}
If $g>0$, then for all $0\le\varepsilon<1$,
\begin{equation}
 \beta_{\varepsilon,t}(0\|\mathcal L)=0
 \quad\Longleftrightarrow\quad
 t\ge T_\varepsilon:=\frac{\arccos\sqrt\varepsilon}{g}.
 \label{sm:threshold}
\end{equation}
Moreover, at total time $T_\varepsilon$, protocols with positive resolution time have exactly $\alpha=\varepsilon$ and $\beta=O(s_{\min})$.
<!-- end theorem-9 -->
<!-- proof-10: body_md -->
We use the protocol of Section~\ref{sec:zeno-common} with an auxiliary system $E\simeq A$ and input and target states $|\psi\rangle,|v\rangle$ on $AE$ chosen so that $|\langlev|(G\otimes\1_E)|\psi\rangle|=g$.
We first show that such states satisfying the protocol's orthogonality conditions exist, then choose the total exposure time to attain the threshold and finally prove the converse below it.

For $g>0$, operator-norm and trace-norm duality on the quotient by $\mathcal S$ gives
\begin{equation*}
 g=\max_{\substack{\left\lVertW\right\rVert_1\le1\\
                   \operatorname{Tr}(W^*C)=0\ \forall C\in\mathcal S}}
          |\operatorname{Tr}(W^*G)|.
\end{equation*}
Let $W$ be a maximiser, which has trace norm one because $g>0$.
Since $\1\in\mathcal S$, the constraint also gives $\operatorname{Tr} W=0$.
Write its polar decomposition as $W=U|W|$.
Choose corresponding orthonormal bases $\{|a\rangle_A\}_a$ and $\{|a\rangle_E\}_a$ for $A$ and $E$.
For any operator $X$ on $A$, define its unnormalised vectorisation on $AE$ by $|X\rangle\!\rangle=(X\otimes\1_E)\sum_a|a\rangle_A|a\rangle_E$.
Applying this construction to $|W|^{1/2}$ and $U|W|^{1/2}$ gives the vectors
\begin{equation*}
 |\psi\rangle=\bigl||W|^{1/2}\bigr\rangle\!\rangle,\qquad
 |v\rangle=\bigl|U|W|^{1/2}\bigr\rangle\!\rangle
\end{equation*}
which are normalised and satisfy, for every operator $X$ on $A$,
$\langlev|(X\otimes\1_E)|\psi\rangle=\operatorname{Tr}(W^*X)$.
Hence $\langle v|\psi\rangle=\operatorname{Tr}(W^*)=0$ and $\langlev|(L_j\otimes\1_E)|\psi\rangle=0$ for all $j$.
Since $W$ is a maximiser, these vectors also satisfy
\begin{equation}
 |\langlev|(G\otimes\1_E)|\psi\rangle|=|\operatorname{Tr}(W^*G)|=g.
 \label{sm:witnessg}
\end{equation}

To exploit the identity null hypothesis, apply the protocol of Section~\ref{sec:zeno-common} with $\mathcal L_1=0$, $\mathcal L_2=\mathcal L$, the vectors constructed above and the control $H_c$ defined there.
The interleaved projection in the Zeno product \eqref{sm:zeno-limit} is $P=|\psi\rangle\!\langle\psi|+|v\rangle\!\langlev|$, and by \eqref{sm:witnessg} we can fix the phase of $|v\rangle$ so that $z=\langlev|(G\otimes\1_E)|\psi\rangle=g>0$.
For $s=\tau/n\le s_1$, we have
\begin{equation}
 \alpha_n=\cos^2(|z|\tau),\qquad \beta_n\le\frac{C_0\tau^2}{n},
 \label{sm:identity-block}
\end{equation}
where the bound on $\beta_n$ follows from \eqref{sm:block-errors}, while the equality for $\alpha_n$ follows by using $\mathcal L_1=0$ in \eqref{sm:error-equalities} and $H_c=PH_cP$, $H_c^2=|z|^2P$, so that every intermediate measurement returns $P$ and $\alpha_n=1-|\langlev|e^{-i\tau H_c}|\psi\rangle|^2=1-\sin^2(|z|\tau)=\cos^2(|z|\tau)$.

For $0\le\varepsilon<1$ and $0<s_{\min}\le\min\{T_\varepsilon/2,s_1/2\}$, choosing total exposure time $\tau=T_\varepsilon=\arccos\sqrt\varepsilon/g$ and the subdivision in \eqref{sm:block-resolution}, we obtain from \eqref{sm:identity-block}
\begin{equation*}
 \alpha_n=\cos^2(gT_\varepsilon)=\varepsilon,\qquad \beta_n\le2C_0T_\varepsilons_{\min}.
\end{equation*}
Letting $s_{\min}\to0$ proves that the optimal type-II error probability vanishes at the endpoint of \eqref{sm:threshold}.
Using the same protocol within any larger total exposure time budget proves the same for $t\ge T_\varepsilon$.


To prove the converse for arbitrary adaptive protocols, choose a minimiser $C=a\1+\sum_jc_jL_j$ in \eqref{sm:g}, which in particular gives $\left\lVertG-C\right\rVert_\infty=g$.
To control the change of direction of the no-jump state by $g$, we choose an equivalent Lindblad representation in which the drift equals $G-C$ up to a scalar multiple of the identity.
Following \eqref{sm:gauge}, shift the jump operators to $L'_j=L_j+\overline c_j\1$ and replace the drift by
\begin{equation*}
 G'=G-\sum_jc_jL_j-\tfrac12\sum_j|c_j|^2\1
    =G-C+\zeta\1,\qquad
 \zeta=a-\tfrac12\sum_j|c_j|^2 .
\end{equation*}
This change of jump operators and drift leaves the Lindbladian $\mathcal L$ invariant.
Let
$K=\sum_j(L'_j)^*L'_j=-(G'+G'^*)\ge0$ and $\nu=\left\lVertK\right\rVert_\infty$.
The channel under the alternative hypothesis contains its no-jump completely positive part:
\begin{equation}
 \mathcal Q_s(X)=e^{sG'}Xe^{sG'^*},\qquad \mathcal Q_s\preceq e^{s\mathcal L}.
 \label{sm:nojump}
\end{equation}

Consider an arbitrary finite adaptive protocol.
Purify the initial state and every common control, retaining a purification environment.
For an instrument, keep an orthogonal outcome record and an additional untouched copy so that discarding the latter implements the original classical record.
All subsequent operations are controlled by that record.
The original final test acts trivially on the added purification systems.

Pad any branch which stops at accumulated time $T_h<t$ with a call of duration $t-T_h$ on a fresh disposable probe.
Discarding that probe leaves the original output and errors unchanged.
The padding is only a converse device and need not respect a positive resolution time.
Now every branch has total time exactly $t$.
Synchronise the finitely many branch endpoints on the common axis for the accumulated time $u\in[0,t]$.
Between endpoints, every branch is being sampled, either on its physical probe or on a padding probe.
The no-jump generator under the alternative hypothesis is therefore a direct sum of copies of $G-C+\zeta\1$.
Its scalar part is a global scalar and its residual operator has norm at most $g$.

Take a common finite enlarged system $AEF$ for the protocol, with $F$ containing the retained purification systems and $E$ also containing the padding probes.
On $AEF$, let $|\phi_u\rangle$ be the normalised purified vector under the null hypothesis and let $|\psi_u\rangle$ be the subnormalised vector under the alternative hypothesis obtained by selecting the no-jump term in \eqref{sm:nojump} at every call.
Positivity of the omitted Dyson terms means that the output on $AEF$ under the alternative hypothesis dominates this no-jump substate.
Common isometries preserve its norm.
During each call,
\begin{equation*}
 \frac{\mathrm d}{\mathrm d u}\||\psi_u\rangle\|^2
 =-\langle\psi_u|K_u|\psi_u\rangle\ge-\nu\||\psi_u\rangle\|^2,
\end{equation*}
where $K_u$ is the controlled copy of $K$.
Consequently,
\begin{equation}
 \||\psi_t\rangle\|^2\ge e^{-\nu t}.
 \label{sm:survival}
\end{equation}

Normalise $|\widehat{\psi}_u\rangle=|\psi_u\rangle/\||\psi_u\rangle\|$.
After removing the common isometries and the global scalar, its derivative orthogonal to itself is
$\left(\1-|\widehat{\psi}_u\rangle\!\langle\widehat{\psi}_u|\right)(G-C)_u|\widehat{\psi}_u\rangle$, where $(G-C)_u$ is the controlled copy of $G-C$ on the enlarged system and has operator norm at most $g$.
Thus the projective angle traversed per unit total time is at most $g$.
Under the null hypothesis there is no unknown evolution, and common isometries preserve the angle.
Since the two vectors agree initially,
\begin{equation}
 \arccos|\langle\phi_t|\widehat{\psi}_t\rangle|\le gt.
 \label{sm:angle}
\end{equation}
Only the range $gt\le\pi/2$ will be needed.
One may equivalently extend all common isometries to unitaries on a fixed larger space and use the interaction picture there.

To turn the angle bound \eqref{sm:angle} into a testing bound, consider two unit vectors with overlap $\cos\theta$, where $0\le\theta\le\pi/2$.
Let $a$ and $b$ be the probabilities of accepting the null hypothesis on the first and second states, respectively, using the binary measurement $\{Q,\1-Q\}$, where outcome $Q$ means accepting the null hypothesis.
If $a\ge1-\varepsilon$ and $\theta<\arccos\sqrt\varepsilon$, then $b\ge\left[\sqrt{1-\varepsilon}\cos\theta-\sqrt\varepsilon\sin\theta\right]^2$.
To verify this directly, write the two acceptance probabilities as
$a=\cos^2 A$ and $b=\cos^2 B$, with $A,B\in[0,\pi/2]$.
Monotonicity of fidelity under the binary measurement gives
$\cos\theta\le\sqrt{ab}+\sqrt{(1-a)(1-b)}=\cos(A-B)$.
Hence $|A-B|\le\theta$.
Since $A\le\arcsin\sqrt\varepsilon$, it follows that
$B\le\theta+\arcsin\sqrt\varepsilon<\pi/2$, which gives the claimed lower bound on $b$.

Apply this bound to $|\phi_t\rangle,|\widehat{\psi}_t\rangle$ and lift the original final effect to the purification space.
The actual output under the alternative hypothesis dominates the no-jump substate, so \eqref{sm:survival} and \eqref{sm:angle} imply, for $t<T_\varepsilon$,
\begin{equation}
 \beta_{\varepsilon,t}(0\|\mathcal L)\ge
 e^{-\nu t}
 \left[\sqrt{1-\varepsilon}\cos(gt)-\sqrt\varepsilon\sin(gt)\right]^2>0.
 \label{sm:thresholdfloor}
\end{equation}
The same lower bound holds at all positive resolution times.
Together with endpoint achievability, this proves \eqref{sm:threshold}.
This completes the proof of Theorem~\ref{thm:identity}.

\label{prop:classical}
For two finite-state continuous-time Markov generators $L_1,L_2$ with exit rates $r_i(x)$, let
$\Lambda=\max_{i,x}r_i(x)$.
Every classical protocol of pathwise total time at most $t$ satisfies
\begin{equation}
 \alpha+\beta\ge e^{-\Lambda t},\qquad
 p^{\mathrm{class}}_{\mathrm{err},t}(L_1,L_2)\ge\tfrac12e^{-\Lambda t}.
 \label{sm:classical}
\end{equation}
The statement allows arbitrary finite classical memory, stochastic controls, resets, adaptive observation times and even observation of the entire physical jump record.

For a fixed protocol, let $P_i(h)$ be the probability of accepting hypothesis $h\in\{1,2\}$ under generator $L_i$, and let $P_0(h)$ be the same probability with the unknown dynamics replaced by the identity.
An identity run has evolution durations $s_k$ and states $x_k$ during these intervals.
Under hypothesis $i$, the probability of no physical jump along these intervals is~\cite[Section~2.6]{Norris1997}
\begin{equation*}
 \prod_k e^{-r_i(x_k)s_k}
 =e^{-\sum_k r_i(x_k)s_k}
 \ge e^{-\Lambda\sum_k s_k}\ge e^{-\Lambda t}.
\end{equation*}
Weighting each run under identity dynamics by its no-jump probability and averaging over the random preparations, controls and final decision gives $P_i(h)\ge e^{-\Lambda t}P_0(h)$.
This remains valid for adaptive controls and durations, since $\sum_k s_k\le t$ on every run.
Consequently,
\begin{equation*}
 \alpha+\beta=P_1(2)+P_2(1)
 \ge e^{-\Lambda t}\left(P_0(2)+P_0(1)\right)=e^{-\Lambda t}.
\end{equation*}
Taking the infimum of $(\alpha+\beta)/2$ over protocols proves the symmetric error bound.
