\label{hd:draw-intro}

<!-- theorem-2: body_md -->
\label{thm:intro-covariance}
If $0\ne S\in\mathcal T_{\mathrm{TT}}$ satisfies $\Delta S=\lambda S$, then
\[
 \operatorname{Cov}_0(S^\vee,S^\vee)=\kappa_n(\lambda)
       \|S^\vee\|_{L^2(SM,m_0)}^2,
\]
where
\begin{equation*}
 \kappa_n(\lambda)=\frac{J(a+\mu)J(a-\mu)}{J(n+3)},
 \qquad a=\frac{n+3}{2},
\end{equation*}
and
\[
 J(z)=\sqrt\pi\,\frac{\Gamma(z/2)}{\Gamma((z+1)/2)},\qquad
 \mu^2=\frac{(n-1)^2+8}{4}-\lambda.
\]
We choose $\mu\geq0$ when $\mu^2\geq0$, and $\mu$ with positive
imaginary part otherwise; the formula is unchanged by $\mu\mapsto-\mu$.

<!-- theorem-3: body_md -->
\label{hd:main}
For every $n\geq3$, the constrained Hessian at $g_0$ is diagonal in an
$L^2$-orthogonal TT eigenbasis of $\Delta$. If $0\ne S\in\mathcal T_{\mathrm{TT}}$ and $\Delta S=\lambda S$,
then $\lambda\geq n$ and
\begin{equation}\label{hd:main-mode}
 \mathcal H(S,S)=\frac{(\lambda-2)(n-1)}{8}
 \bigl(\kappa_n(\lambda)-\frac{n+2}{n-1}\bigr)
 \|S^\vee\|_{L^2(SM,m_0)}^2.
\end{equation}
The function $\kappa_n$ is strictly decreasing on
$[n,\infty)$ and tends to zero.

\begin{itemize}
\item For $n=3$, $\mathcal H$ is negative definite on $\mathcal T_{\mathrm{TT}}$.
\item For every $n\geq4$, there is exactly one $\lambda_*(n)>n$ satisfying
\begin{equation*}
 \kappa_n(\lambda_*(n))=\frac{n+2}{n-1}.
\end{equation*}
The positive space of $\mathcal H$ is the sum of the TT eigenspaces with
$n\leq\lambda<\lambda_*(n)$; its kernel is the eigenspace at
$\lambda_*(n)$; and its negative space is the sum of those above it.
The positive space and kernel are finite-dimensional, and the negative
space is infinite-dimensional.
\end{itemize}

<!-- corollary-1: body_md -->
\label{cor:local-entropy-curves}
Let $(M^3,g_0)$ be a closed hyperbolic three-manifold. There is a
$C^\infty$ neighborhood $\mathcal U$ of $g_0$ such that every smooth

<!-- corollary-2: body_md -->
\label{hd:saddle-intro}
If $n\geq4$ and $(M,g_0)$ admits a nonzero trace-free Codazzi tensor,

Such closed hyperbolic manifolds exist in every dimension $n\geq4$.
The curves converge smoothly to $g_0$ as $s\to0$.

<!-- proposition-1: body_md -->
\label{prop:flaminio}
Let $\widehat g_s$ be a $C^2$ curve of $C^5$ negatively curved metrics
such that $\operatorname{Vol}(M,\widehat g_s)=V_0$, $\widehat g_0=g_0$, and
$\dot{\widehat g}_0=S$.  Then
\begin{equation*}
 \left.\frac{d^2}{ds^2}\right|_0 h(\widehat g_s)
 =\operatorname{Cov}_0\left(T(S)^\vee,\frac{n-1}{2}S^\vee\right),
\end{equation*}
where
\[
T(S)=\frac14 \nabla^*\nabla S-\frac12\delta^*\delta S+\frac12(\operatorname{tr}_{g_0}S)g_0-\frac12 S.
\]
In particular, $T(S)=\tfrac14PS$ for $S\in\mathcal T_{\mathrm{TT}}$, where
$P=\Delta-2$.

<!-- lemma-1: body_md -->
\label{lem:integrability}
For every $S\in\mathcal T_{\mathrm{TT}}$ there is a smooth curve
$g_s\in\mathcal M_{-n(n-1)}$ with $g_{s=0}=g_0$ and
$\left.\frac{d}{ds}\right|_{s=0}g_s=S$.
For sufficiently small $s$, the metrics $g_s$ remain negatively curved.

<!-- proof-1: body_md -->
Fix $k\geq4$ and $0<\alpha<1$. On the open neighborhood of $(0,0)$
where $g_0+sS+ug_0$ is positive definite, consider
\[
 \Phi(s,u):=\operatorname{Sc}(g_0+sS+u g_0)+n(n-1),
 \qquad
 \Phi:\mathbb R\times C^{k,\alpha}(M)
       \longrightarrow C^{k-2,\alpha}(M).
\]
For $h=fg_0$ one has $\operatorname{tr} h=nf$ and $\delta h=-df$.
Equation \eqref{eq:scalar-linearization-hyperbolic} therefore gives
\begin{equation*}
 D_u\Phi(0,0)f
 =D\operatorname{Sc}_{g_0}(fg_0)
 =(n-1)(\nabla^*\nabla+n)f.
\end{equation*}
The operator on the right is an isomorphism
$C^{k,\alpha}\to C^{k-2,\alpha}$ because it is elliptic, self-adjoint,
and strictly positive.  The implicit function theorem produces a curve
$u(s)$ with $u(0)=0$ and $\Phi(s,u(s))=0$.  Since $S$ is TT,
\eqref{eq:scalar-linearization-hyperbolic} gives
$D_s\Phi(0,0)=0$; differentiating $\Phi(s,u(s))=0$ then shows
$\dot u(0)=0$.  Thus
\[
 g_s=g_0+sS+u(s)g_0
\]
has $\dot g_0=S$ and the required scalar curvature.  Elliptic regularity
and the smooth implicit function theorem give a smooth curve when $S$ is
smooth.  Finally, negative sectional curvature is open in the $C^2$
topology.

<!-- proposition-2: body_md -->
\label{hd:bochner}
For $S\in\mathcal T_{\mathrm{TT}}$, put $C_{ijk}=\nabla_iS_{jk}-\nabla_jS_{ik}$. Then
\begin{equation}\label{hd:codazzi-energy}
 \frac12\|C\|_{L^2}^2=\langle (\nabla^*\nabla-n)S,S\rangle_{L^2}.
\end{equation}
In particular $\nabla^*\nabla\geq n$ on $\mathcal T_{\mathrm{TT}}$, with equality exactly for
trace-free Codazzi tensors, that is, tensors satisfying $C=0$.
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
The bundle-valued Bochner identity (see \cite{Besse}*{p.~355}) is
\[
 (d_\nabla d_\nabla^*+d_\nabla^*d_\nabla)S
 =\nabla^*\nabla S-R^\circ(S)+\operatorname{Ric}_{g_0}\circ S.
\]
Here $S$ is viewed as a $T^*M$-valued one-form. For trace-free
$S$ at curvature $-1$, the right-hand side is $\Delta S-nS$.
For $S\in\mathcal T_{\mathrm{TT}}$, we also have $d_\nabla^*S=0$ and $d_\nabla S=C$.
With $|C|^2=\sum_{i,j,k}C_{ijk}^2$ in an orthonormal frame, the
bundle-valued two-form norm is
$\|d_\nabla S\|_{L^2}^2=\tfrac12\|C\|_{L^2}^2$,
so integration gives \eqref{hd:codazzi-energy}.
If $S$ is Codazzi, the same identity gives $\Delta S=nS$.
Conversely, equality in \eqref{hd:codazzi-energy} forces $C=0$.
Finally, contracting the Codazzi equation gives
$\nabla^iS_{ij}=\nabla_j\operatorname{tr} S$, so a trace-free Codazzi tensor is TT.

<!-- theorem-4: body_md -->
\label{thm:constrained-hessian}
Let $g_s\in\mathcal M_{-n(n-1)}$ be a $C^2$ curve through $g_0$ with
$S=\dot g_0\in\mathcal T_{\mathrm{TT}}$.  Then
\begin{equation}\label{eq:main-hessian}
 \boxed{\begin{aligned}
 \mathcal H(S,S)
 :=\left.\frac{d^2}{ds^2}\right|_0h(g_s)
 &=\frac{n-1}{8}\operatorname{Cov}_0\bigl((PS)^\vee,S^\vee\bigr)
 -\frac{1}{4nV_0}\langle PS,S\rangle_{L^2(M)},\\
 &\hspace{35mm} P=\nabla^*\nabla-2.
 \end{aligned}}
\end{equation}
The right-hand side depends only on $S$ and not on the acceleration of the
chosen constrained curve.  By Lemma~\ref{lem:integrability}, every
$S\in\mathcal T_{\mathrm{TT}}$ occurs in such a curve.

 =\frac{n-1}{8}\operatorname{Cov}_0\bigl((PS)^\vee,S^\vee\bigr).
\]
Multiplying \eqref{eq:volume-second} by

\label{pro:correlation}
For $S\in E_\lambda$ and every smooth trace-free symmetric two-tensor
$A$, the correlation satisfies
\begin{equation}\label{eq:correlation-factor}
 \int_{SM}A^\vee(S^\vee\circ\phi_t)\,dm_0
 =\varphi_{n,\lambda}(t)\int_{SM}A^\vee S^\vee\,dm_0
 \qquad(t\in\mathbb R),
\end{equation}
where $\varphi_{n,\lambda}\colon\mathbb R\to\mathbb R$ is the unique
smooth even solution of
\begin{equation}\label{hd:correlation-ode}
 \varphi''+(n+3)\coth r\,\varphi'+(\lambda+2n)\varphi=0,
 \qquad \varphi(0)=1,\quad \varphi'(0)=0.
\end{equation}
The differential equation holds for $r>0$.

Fix $x\in M$ and abbreviate $F(t)=F_{x,A,S}(t)$.
The geodesic flow is smooth, so $F$ is smooth.  Replacing $\xi$ by
$-\xi$ changes $\gamma_{x,\xi}(t)$ to $\gamma_{x,\xi}(-t)$ and
reverses its velocity.  Since both tensor evaluations are quadratic,
this substitution gives $F(-t)=F(t)$ and hence $F'(0)=0$.

Set $Y_{A_x}(\xi)=A_x(\xi,\xi)$.  In the polar coordinates above,
\[
 F(r)=\int_{S_xM}Y_{A_x}(\xi)U(r,\xi)\,d\sigma_x(\xi).
\]
Since $A_x$ is trace-free, $Y_{A_x}$ is a spherical harmonic of
degree two and $\Delta_\sigma Y_{A_x}=2nY_{A_x}$.  Thus integration
by parts on the sphere gives
\[
 \int_{S_xM}Y_{A_x}\Delta_\sigma U\,d\sigma_x
 =\int_{S_xM}(\Delta_\sigma Y_{A_x})U\,d\sigma_x
 =2nF(r).
\]

\[
 F''+(n+3)\coth r\,F'
 +\left(\lambda+2n\coth^2r-\frac{2n}{\sinh^2r}\right)F=0.
\]
Since $\coth^2r-(\sinh r)^{-2}=1$, this is precisely the
differential equation in \eqref{hd:correlation-ode}.

smooth even solutions directly.
Put $c=\lambda+2n$ and $p(r)=(\sinh r)^{n+3}$.  A smooth solution
$f$ satisfies $(pf')'=-cpf$ and $p(r)f'(r)\to0$ as $r\downarrow0$.

\[
 f(r)=f(0)-c(Kf)(r),\qquad
 (Kf)(r)=\int_0^r\frac{1}{p(u)}\int_0^u p(v)f(v)\,dv\,du.
\]
Monotonicity of $p$ gives the estimate
\[
 \|Kh\|_{C[0,\varepsilon]}
 \leq \frac{\varepsilon^2}{2}\|h\|_{C[0,\varepsilon]}.
\]
For $|c|\varepsilon^2/2<1$, the map $h\mapsto b-cKh$ is a

has a convergent even power series with arbitrary constant term $b$.  Continuation for the linear equation on $(0,\infty)$
and even reflection give the global solution; taking $b=1$

\begin{equation*}
 F_{x,A,S}(t)=\varphi_{n,\lambda}(t)F_{x,A,S}(0)
 \qquad(t\in\mathbb R).
\end{equation*}
Integrating over $x$ with measure

\label{lem:entropy-remainder}

smoothly on the metric, losing one spatial derivative. Structural
stability gives a positive H\"older time-change function $a_g$ whose
dependence on a $C^4$ vector field is $C^3$ with values in a fixed
H\"older space (see \cite{KKPW}*{Proposition~2.2 and its proof,
pp.~591--592}). Thus $g\mapsto a_g$ is $C^3$ on a sufficiently small

\]

The scalar and tangent components of its closedness condition give
\[
 g_0(A(X),Y)=g_0(X,A(Y)),\qquad
 (\nabla_XA)(Y)=(\nabla_YA)(X).
\]
Thus $A$ is symmetric and Codazzi.  To remove its trace, observe that
\[
 D(\nabla f,-f)(X)=((\operatorname{Hess} f-f\operatorname{Id})(X),0).
\]
The trace of its tangent component is $-(\Delta+n)f$, since
$\Delta=-\operatorname{tr}\operatorname{Hess}$ on functions.  Solve
$-(\Delta+n)f=\operatorname{tr} A$, using invertibility of $\Delta+n$ on the
closed manifold $M$.  Subtracting this exact form removes the trace.
The new representative is still closed and has scalar component zero,
so it is still symmetric and Codazzi.

To prove uniqueness, suppose a trace-free representative is exact:
$(A,0)=D(Y,s)$.  Its scalar component gives $Y=-\nabla s$,
so $A=-\operatorname{Hess} s+s\operatorname{Id}$.  Taking the trace gives
$(\Delta+n)s=0$, hence $s=0$ and $A=0$.
This proves the isomorphism.

Choose a nonzero trace-free Codazzi tensor $S^+$. Its Hessian is positive
by Corollary~\ref{hd:codazzi-sign}. Choose any TT eigentensor $S^-$
with eigenvalue greater than $\lambda_*(n)$; its Hessian is negative.
Lemma~\ref{lem:integrability} integrates both tensors to constrained curves.
Their first entropy variations vanish, and therefore
\[
 h(g_s^\pm)=n-1+\frac{s^2}{2}\mathcal H(S^\pm,S^\pm)+o(s^2).
\]
This gives the two strict inequalities for sufficiently small nonzero
$s$. Proposition~\ref{hd:bending-input} and
Theorem~\ref{hd:flat-bundle} provide such manifolds in every dimension
$n\geq4$. The curves are smooth and negatively curved near zero.
