\label{asmp:EB_t-ell} 
There exist maps $t:{\cal  P}(\mathbb{Z}_+)\to \mathbb R$ and $\ell:\mathbb R^2\to\mathbb R$ such that, for any two probability mass functions $q_1,q_2\in{\cal  P}(\mathbb{Z}_+)$,
\begin{displaymath}
d(q_1\|q_2)=t(q_1)+\sum_{y\geq0}\ell(q_1(y),q_2(y)),
\end{displaymath}
where, for every $a>0$, the map $b\mapsto \ell(a,b)$ is strictly decreasing and strictly convex, and $\ell(0,b)=0$ for all $b\geq0$.

\label{def:coarsening}

\label{def_ratio}
For any probability measure $G$ on $\mathbb{R}_+$, define $m_G(x):=\frac{A_x(G)}{B_x(G)}$, where
\begin{displaymath}
A_x(G)=\int_{\mathbb{R}_+} e^{-\theta}\theta^x I(\theta<x)\,dG(\theta),\qquad B_x(G)=\int_{\mathbb{R}_+} e^{-\theta}\theta^x\,dG(\theta)=x!\,f_G(x).
\end{displaymath}
Whenever $B_x(G)>0$, the ratio $m_G(x)=A_x(G)/B_x(G)$ has its usual conditional-probability interpretation. If $B_x(G)=0$, we define $m_G(x)$ by the deterministic value of the event $\{\theta<x\}$ whenever this event is $G$-almost surely constant. Thus, we set $m_G(x)=1$ if $G([0,x))=1$ and $m_G(x)=0$ if $G([x,\infty))=1$. This convention is used only outside the relevant finite window, where the posterior event is deterministic under the support constraint.

\end{equation}

\begin{equation}\label{eq:par_above_pred}
\widehat{S}^{\text{\tiny{[Pred]}}}_{n,>}({G^\ast}_\tau)=\sum_{i=1}^{n}\left[1-(1+\tau)^{-X_i}\right],
\end{equation}

\begin{displaymath}
\widehat{\tau}^{\text{\tiny{[ML]}}}_{m}=\frac{m}{\sum_{j=1}^{m}X'_j}.
\end{displaymath}
Accordingly, the resulting parametric maximum-likelihood estimates of $S_{n,\kappa}$ and $S_{n,>}$, together with those of their predictive counterparts, are obtained by replacing $\tau$ in ~(\ref{eq:par_tot})--~(\ref{eq:par_above_pred}) with its estimate $\widehat{\tau}^{\text{\tiny{[ML]}}}_{m}$. \begin{displaymath}
\widehat{\tau}^{\text{\tiny{[B]}}}_{m}=\frac{m+\alpha}{\beta+\sum_{j=1}^{m}X'_j}.
\end{displaymath}
Accordingly, the resulting parametric Bayes estimates of $S_{n,\kappa}$ and $S_{n,>}$, together with those of their predictive counterparts, are obtained by replacing $\tau$ in ~(\ref{eq:par_tot})--~(\ref{eq:par_above_pred}) with its estimate $\widehat{\tau}^{\text{\tiny{[B]}}}_{m}$.

<!-- lemma-5: body_md -->
\label{lmm:hel-bound-genr}
	Suppose that $Z_1,\dots,Z_m\stackrel{\mbox{\scriptsize{iid}}}{\sim} f$ for some distribution $f\in {\cal  P}(\mathbb{Z})$ and fix $J\geq 1$. Let $\widehat q_m$ be the empirical distribution of $Z_1,\dots,Z_m$, and $\widehat f \in {\cal  P}(\mathbb{Z})$ satisfies
	$
	d(\widehat q_m\|\widehat f)\leq d(\widehat q_m\|f).
	$ 
	Then we have 
	$$\mathbb{E}_f\!\left[H^2(\widehat f,f)\right]\le\frac{4c_2}{c_1}\,\frac{J}{m}+\Big(\frac{4c_2}{c_1}+2m\Big)\mathbb{P}_{Z\sim  f}[Z\geq J].$$
<!-- end lemma-5 -->
<!-- proof-1: body_md -->
Using $d(\widehat q_m\|\widehat f)\le d(\widehat q_m\|f)$, the triangle inequality for the Hellinger metric, and \prettyref{asmp:dist_sandwich}, we get
\begin{align}
	H^2(\widehat f,f)
	&\le\Big(H(\widehat q_m,\widehat f)+H(\widehat q_m,f)\Big)^2
	\le 2H^2(\widehat q_m,\widehat f)+2H^2(\widehat q_m,f) \nonumber\\
	&
	\le\frac{2}{c_1}d(\widehat q_m\|\widehat f)+\frac{2}{c_1}d(\widehat q_m\|f)
	\le\frac{4}{c_1}d(\widehat q_m\|f).
	\label{eq:hell-by-optimality-feasible}
\end{align}
Let $Z_{(m)}$ denote the maximum of $Z_1,\dots,Z_m$.
Using ~(\ref{eq:hell-by-optimality-feasible}) and the fact that $H^2\le 2$, we obtain
\begin{align}
	\mathbb{E}_f\!\left[H^2(\widehat f,f)\right]
	&=\mathbb{E}_f\!\left[H^2(\widehat f,f)I(Z_{(m)}<J)\right]+\mathbb{E}_f\!\left[H^2(\widehat f,f)I(Z_{(m)}\ge J)\right]
	\nonumber\\
	&\le\frac{4}{c_1}\mathbb{E}_f\!\left[d(\widehat q_m\|f)I(X'_{(m)}<J)\right]+2\,\mathbb{P}_{f}(Z_{(m)}\ge J).
	\label{eq:split-Ymax-feasible}
\end{align}
By the upper bound in \prettyref{asmp:dist_sandwich}, we bound the first term in the right hand side of \prettyref{eq:split-Ymax-feasible} as
\begin{align}
	\label{eq:app0}
	\frac{1}{c_2}\mathbb{E}_f\!\left[d(\widehat q_m\|f)I(Z_{(m)}<J)\right]\le\mathbb{E}_f\!\left[\chi^2(\widehat q_m\|f)I(Z_{(m)}<J)\right].
\end{align}
The right hand side of the above display can be further bounded as
\begin{align}
	\mathbb{E}_f\!\left[\chi^2(\widehat q_m\|f)I(Z_{(m)}<J)\right]
	&=
	\sum_{y\ge 0}\frac{\mathbb{E}_f[(\widehat q_m(y)-f(y))^2I(Z_{(m)}<J)]}{f(y)}
	\nonumber\\
	&=\sum_{y<J}\frac{\mathbb{E}_f[(\widehat q_m(y)-f(y))^2I(Z_{(m)}<J)]}{f(y)}+\sum_{y\ge J}\frac{\mathbb{E}_f[(\widehat q_m(y)-f(y))^2I(Z_{(m)}<J)]}{f(y)}.
	\label{eq:chi-split-feasible}
\end{align}
On the event $\{Z_{(m)}<J\}$ we have $\widehat q_m(y)=0$ for all $y\ge J$, so we can bound the second sum as
\[
\sum_{y\ge J}\frac{f(y)^2\,\mathbb{P}_f(Z_{(m)}<J)}{f(y)}=\mathbb{P}_{Z\sim f}[Z\geq J]\,\mathbb{P}_G(Z_{(m)}<J)\le\mathbb{P}_{Z\sim f}[Z\geq J].
\]
For the first sum in \prettyref{eq:chi-split-feasible}, we use the bound
\[
\sum_{y<J}\frac{\mathbb{E}_f[(\widehat q_m(y)-f(y))^2I(Z_{(m)}<J)]}{f(y)}\le\sum_{y<J}\frac{\mathbb{E}_f[(\widehat q_m(y)-f(y))^2]}{f(y)}.
\]
Using the following results
\[
\mathbb{E}_f[\widehat q_m(y)]=f(y),\qquad\mathrm{Var}_f(\widehat p_m(y))=\frac{f(y)(1-f(y))}{m},
\]
we get,
\[
\sum_{y<J}\frac{\mathbb{E}_f[(\widehat p_m(y)-f(y))^2]}{f(y)}=\frac1m\sum_{\substack{y<J\\ f(y)>0}}(1-f(y))\le\frac{J}{m}.
\]
Combining these bounds with ~(\ref{eq:chi-split-feasible}) and ~(\ref{eq:app0}) yields
\[
\mathbb{E}_f\!\left[d(\widehat q_m\|f)I(Z_{(m)}<J)\right]\le c_2\Big(\frac{J}{m}+\mathbb{P}_{Z\sim f}[Z\geq J]\Big).
\]
In view of the last result and the fact that we can bound $\mathbb{P}_f(Z_{(m)}\ge J)$ from above using the union bound $\mathbb{P}_f(Z_{(m)}\ge J)\le m\,\mathbb{P}_{Z\sim f}[Z\geq J]$, we continue ~(\ref{eq:split-Ymax-feasible}) to get the desired inequality
\[
\mathbb{E}_f\!\left[H^2(\widehat f,f)\right]\le\frac{4c_2}{c_1}\Big(\frac{J}{m}+\mathbb{P}_{Z\sim f}[Z\geq J]\Big)+2m\,\mathbb{P}_{Z\sim f}[Z\geq J].
\]

\label{lem:ratio-stab}
Let $A,B,\widetilde A,\widetilde B\ge 0$ with $B>0$, $\widetilde B>0$, and $A\le B$, $\widetilde A\le \widetilde B$. Then
\begin{displaymath}
\frac{B\widetilde B}{B+\widetilde B}\Big(\frac{A}{B}-\frac{\widetilde A}{\widetilde B}\Big)^2\le 8\Big\{(\sqrt A-\sqrt{\widetilde A})^2+(\sqrt B-\sqrt{\widetilde B})^2\Big\}.
\end{displaymath}
<!-- end lemma-13 -->
<!-- proof-8: body_md -->
Set $a=\sqrt A$, $b=\sqrt B$, $\widetilde a=\sqrt{\widetilde A}$, and $\widetilde b=\sqrt{\widetilde B}$.  Since $A\le B$ and $\widetilde A\le \widetilde B$, we have $0\le a\le b$ and $0\le \widetilde a\le \widetilde b$. 
Now
\begin{displaymath}
\frac{A}{B}-\frac{\widetilde A}{\widetilde B}=\Big(\frac{a}{b}\Big)^2-\Big(\frac{\widetilde a}{\widetilde b}\Big)^2=\Big(\frac{a}{b}-\frac{\widetilde a}{\widetilde b}\Big)\Big(\frac{a}{b}+\frac{\widetilde a}{\widetilde b}\Big).
\end{displaymath}
Because $a/b,\widetilde a/\widetilde b\in[0,1]$, we have $\frac{a}{b}+\frac{\widetilde a}{\widetilde b}\le 2$. Therefore $\Big(\frac{A}{B}-\frac{\widetilde A}{\widetilde B}\Big)^2\le4\Big(\frac{a}{b}-\frac{\widetilde a}{\widetilde b}\Big)^2$. \begin{displaymath}
\frac{B\widetilde B}{B+\widetilde B}\Big(\frac{A}{B}-\frac{\widetilde A}{\widetilde B}\Big)^2\le4\cdot \frac{b^2\widetilde b^2}{b^2+\widetilde b^2}\Big(\frac{a}{b}-\frac{\widetilde a}{\widetilde b}\Big)^2=4\cdot \frac{(a\widetilde b-\widetilde a b)^2}{b^2+\widetilde b^2}.
\end{displaymath}
Next, decompose $a\widetilde b-\widetilde a b=(a-\widetilde a)\widetilde b+\widetilde a(\widetilde b-b)$. Hence, by $(u+v)^2\le 2u^2+2v^2$, $(a\widetilde b-\widetilde a b)^2\le2(a-\widetilde a)^2\widetilde b^2+2\widetilde a^2(\widetilde b-b)^2$. Since $\widetilde a\le \widetilde b$, this yields
\begin{displaymath}
(a\widetilde b-\widetilde a b)^2\le2(a-\widetilde a)^2\widetilde b^2+2\widetilde b^2(\widetilde b-b)^2.
\end{displaymath}
Divide by $b^2+\widetilde b^2\ge \widetilde b^2$:
\begin{displaymath}
\frac{(a\widetilde b-\widetilde a b)^2}{b^2+\widetilde b^2}\le2(a-\widetilde a)^2+2(\widetilde b-b)^2.
\end{displaymath}

\begin{displaymath}

\end{displaymath}
This proves the lemma.

<!-- lemma-14: body_md -->
\label{lem:quad-convert}
For any $a,b\ge 0$ (with the convention $0/0=0$),
\begin{displaymath}
a \le 4\,\frac{ab}{a+b} + 2\,\frac{(a-b)^2}{a+b}.
\end{displaymath}
Consequently, for any $u\in[0,1]$,
\begin{displaymath}
a\,u \le 4\,\frac{ab}{a+b}\,u + 2\,\frac{(a-b)^2}{a+b}.
\end{displaymath}
<!-- end lemma-14 -->
<!-- proof-9: body_md -->
If $a+b=0$, then $a=b=0$ and the inequality is trivial. Assume now $a+b>0$. Multiplying the desired inequality by $(a+b)$, it is enough to prove that $a(a+b)\le 4ab+2(a-b)^2$. Expand the right-hand side: $4ab+2(a-b)^2=4ab+2(a^2-2ab+b^2)=2a^2+2b^2$. So it suffices to prove that $a(a+b)\le 2a^2+2b^2$. But
\begin{displaymath}
2a^2+2b^2-a(a+b)=a^2-ab+2b^2.
\end{displaymath}
Now $a^2-ab+2b^2=\frac12(a-b)^2+\frac12 a^2+\frac32 b^2\ge 0$. Hence $a(a+b)\le 2a^2+2b^2=4ab+2(a-b)^2$, which proves
\begin{displaymath}
a \le 4\,\frac{ab}{a+b}+2\,\frac{(a-b)^2}{a+b}.
\end{displaymath}
For the second statement, simply multiply both sides by $u\in[0,1]$ and use $u\le 1$:
\begin{displaymath}
a\,u\le4\,\frac{ab}{a+b}\,u+2\,\frac{(a-b)^2}{a+b}\,u\le4\,\frac{ab}{a+b}\,u+2\,\frac{(a-b)^2}{a+b}.
\end{displaymath}
This proves the lemma.

<!-- lemma-15: body_md -->
\label{lem:chi2-hell}
For any $a,b\ge 0$ (with $0/0=0$),
\begin{displaymath}
\frac{(a-b)^2}{a+b}\le 4(\sqrt a-\sqrt b)^2.
\end{displaymath}
Hence for pmfs $p,q$ on a countable alphabet,
\begin{displaymath}
\sum_t \frac{(p(t)-q(t))^2}{p(t)+q(t)}\le4\sum_t(\sqrt{p(t)}-\sqrt{q(t)})^2.
\end{displaymath}
<!-- end lemma-15 -->
<!-- proof-10: body_md -->
If $a=b=0$, the claim is trivial. Otherwise, use $a-b=(\sqrt a-\sqrt b)(\sqrt a+\sqrt b)$, so $(a-b)^2=(\sqrt a-\sqrt b)^2(\sqrt a+\sqrt b)^2$. Since $(\sqrt a+\sqrt b)^2=a+b+2\sqrt{ab}\le 2(a+b)$, we obtain
\begin{displaymath}
(a-b)^2\le 2(a+b)(\sqrt a-\sqrt b)^2.
\end{displaymath}
In fact this already gives the sharper bound with constant \(2\); hence certainly $\frac{(a-b)^2}{a+b}\le 4(\sqrt a-\sqrt b)^2$. Summing the pointwise inequality with $a=p(t)$ and $b=q(t)$ yields
\begin{displaymath}
\sum_t \frac{(p(t)-q(t))^2}{p(t)+q(t)}\le4\sum_t(\sqrt{p(t)}-\sqrt{q(t)})^2.
\end{displaymath}
This proves the lemma.
