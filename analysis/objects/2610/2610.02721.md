\label{thm:globalize-en}
Let $u$ be the maximal-lifespan solution to \eqref{eq:SM-en} with initial time $t=0$, and write its lifespan interval as $I_{\max}=[0,T_+)$. Assume
\[
u(0)-Q\in L_x^2,
\qquad
u(0)\in \dot H^1\cap\dot H^3,
\qquad
E(u_0)<4\pi.
\]
If
\begin{equation}\label{eq:forward-finite-en}
\|\nabla u\|_{\ell^2L^4_{t,x}([0,T_+)\times{\mathbb R}^2)}<\infty,
\end{equation}
then
\[
T_+=+\infty,
\qquad
I_{\max}=[0,+\infty),
\qquad
u\in L^\infty_{t,\mathrm{loc}}\bigl([0,+\infty);\dot H^1\cap\dot H^3\bigr).
\]
In particular, for every $0<T<\infty$,
\[
\|\nabla u\|_{\ell^2L^4_{t,x}([0,T]\times{\mathbb R}^2)}<\infty.
\]

<!-- lemma-1: body_md -->
\label{lem:Jparallel-en}
For any tangential vector field $X$ along $u$, one has
\[
\nabla_\alpha(J_uX)=J_u(\nabla_\alpha X).
\]
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Differentiate $u\times X$ and project onto the tangent space:
\[
\nabla_\alpha(J_uX)
=\Pi_u\bigl(\partial_\alpha u\times X + u\times \partial_\alpha X\bigr).
\]
Here $\partial_\alpha u\times X$ is orthogonal to $T_u\mathbb{S}$, and so its tangential projection vanishes. Hence
\[
\nabla_\alpha(J_uX)=u\times \Pi_u(\partial_\alpha X)=J_u\nabla_\alpha X.
\qedhere
\]

<!-- proposition-1: body_md -->
\label{prop:covdiff-en}
Let $u$ be a smooth solution of \eqref{eq:SM-en}. Then
\begin{equation}\label{eq:cov-sch-psi-en}
\nabla_t\psi_m = J_u\nabla_j\nabla_j\psi_m + R(\psi_m,\psi_j)J_u\psi_j.
\end{equation}
If, for each fixed $t$, we denote by $u(s,t,x)$ the harmonic map heat flow with initial data $u(t,\cdot)$, then
\begin{equation}\label{eq:cov-heat-psi-en}
\nabla_s\psi_m = \nabla_j\nabla_j\psi_m + R(\psi_m,\psi_j)\psi_j.
\end{equation}
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
For \eqref{eq:cov-sch-psi-en}, by the original equation and Lemma \ref{lem:Jparallel-en},
\[
\nabla_t\psi_m = \nabla_m\partial_tu
= \nabla_m(J_u\nabla_j\psi_j)
= J_u\nabla_m\nabla_j\psi_j.
\]
Now commute $\nabla_m$ and $\nabla_j$:
\[
\nabla_m\nabla_j\psi_j
=\nabla_j\nabla_m\psi_j+R(\psi_m,\psi_j)\psi_j.
\]
By the torsion-free identity \eqref{eq:torsion-en},
\[
\nabla_j\nabla_m\psi_j=\nabla_j\nabla_j\psi_m,
\]
and this yields \eqref{eq:cov-sch-psi-en}.

For \eqref{eq:cov-heat-psi-en}, one only needs to substitute $\partial_su=\tau(u)=\nabla_j\psi_j$:
\[
\nabla_s\psi_m
=\nabla_m\nabla_j\psi_j
=\nabla_j\nabla_m\psi_j + R(\psi_m,\psi_j)\psi_j
=\nabla_j\nabla_j\psi_m + R(\psi_m,\psi_j)\psi_j.
\qedhere
\]

<!-- proposition-2: body_md -->
\label{prop:cov-W-en}
The fields $W_m$ satisfy
\begin{equation}\label{eq:cov-sch-W-en}
(\mathrm{i}\nabla_t+\nabla_j\nabla_j)W_m
=\frac{\mathrm{i}}{2}\,\Im(W_m\mid W_j)\,W_j,
\end{equation}
and
\begin{equation}\label{eq:cov-heat-W-en}
(\nabla_s-\nabla_j\nabla_j)W_m

(I+\mathrm{i} J_u)J_u=-\mathrm{i}(I+\mathrm{i} J_u),

the principal part becomes $-\mathrm{i}\nabla_j\nabla_jW_m$, and thus

(\mathrm{i}\nabla_t+\nabla_j\nabla_j)W_m
=\mathrm{i}(I+\mathrm{i} J_u)R(\psi_m,\psi_j)J_u\psi_j.
\]
By the spherical curvature formula \eqref{eq:curvature-en},

R(\psi_m,\psi_j)J_u\psi_j
=-\langle \psi_m,J_u\psi_j\rangle\,\psi_j.
\]
Using \eqref{eq:WW-en} and $(I+\mathrm{i} J_u)\psi_j=W_j$, one obtains \eqref{eq:cov-sch-W-en}.

The heat-flow equation \eqref{eq:cov-heat-W-en} follows similarly from \eqref{eq:cov-heat-psi-en}, because

R(\psi_m,\psi_j)\psi_j=|\psi_j|^2\psi_m-\langle \psi_m,\psi_j\rangle\psi_j.

<!-- lemma-2: body_md -->
\label{lem:gamma-basic-en}
If $Z$ is a complex tangential field along $u$, then
\begin{equation}\label{eq:gammaZ-body-en}
\Gamma_j Z=u\,\langle \psi_j,Z\rangle,
\qquad
\Gamma_j^2 Z=-\psi_j\,\langle \psi_j,Z\rangle,
\end{equation}
and therefore
\begin{equation}\label{eq:gammaZ-bound-en}
|\Gamma_jZ|\le |\psi_j|\,|Z|,
\qquad
|\Gamma_j^2Z|\le |\psi_j|^2|Z|.
\end{equation}
In addition,
\begin{equation}\label{eq:divGamma-body-en}
\partial_j\Gamma_j
=u\otimes(\Delta u)^\top-(\Delta u)\otimes u^\top.
\end{equation}
<!-- end lemma-2 -->
<!-- proof-4: body_md -->
Since $Z$ is tangential, $\langle u,Z\rangle=0$, and hence
\[
\Gamma_jZ
=u\,\langle \psi_j,Z\rangle-\psi_j\,\langle u,Z\rangle
=u\,\langle \psi_j,Z\rangle,
\]
which is the first identity in \eqref{eq:gammaZ-body-en}. Applying $\Gamma_j$ once more, and using $\langle \psi_j,u\rangle=0$ and $\langle u,u\rangle=1$, yields
\[
\Gamma_j^2Z
=\Gamma_j\bigl(u\,\langle \psi_j,Z\rangle\bigr)
=-\psi_j\,\langle \psi_j,Z\rangle,
\]
which proves the second identity in \eqref{eq:gammaZ-body-en} and therefore \eqref{eq:gammaZ-bound-en}.

Differentiating the explicit formula for $\Gamma_j$ gives
\[
\partial_j\Gamma_j
=\psi_j\otimes\psi_j^\top+u\otimes(\partial_j\psi_j)^\top
-(\partial_j\psi_j)\otimes u^\top-\psi_j\otimes\psi_j^\top,
\]
and the quadratic terms cancel pairwise. This yields \eqref{eq:divGamma-body-en}.

<!-- proposition-3: body_md -->
\label{prop:ambient-W-en}
After expanding the covariant equations from the previous subsection into the ambient space, the complex tangential fields $W_m$ satisfy
\begin{equation}\label{eq:ambient-W-en}
(\mathrm{i}\partial_t+\Delta)W_m
=-2\Gamma_j\partial_jW_m-\Gamma_j^2W_m
+\frac{\mathrm{i}}{2}\,\Im(W_m\mid W_j)\,W_j.
\end{equation}
Along the harmonic map heat flow one also has
\begin{equation}\label{eq:ambient-heat-en}
(\partial_s-\Delta)W_m
=2\Gamma_j\partial_jW_m + V_m^{(s)},
\end{equation}
where
\begin{equation}\label{eq:Vsbound-en}
|V_m^{(s)}|
\lesssim \bigl(|\partial_j\Gamma_j|+|\Gamma_s|+|\Gamma_j|^2+|W_x|^2\bigr)|W_m|.
\end{equation}
<!-- end proposition-3 -->
<!-- proof-5: body_md -->
Expand the covariant Laplacian in \eqref{eq:cov-sch-W-en}:
\[
\nabla_j\nabla_jW_m
=\Delta W_m+2\Gamma_j\partial_jW_m+(\partial_j\Gamma_j+\Gamma_j^2)W_m.
\]
Since $\nabla_tW_m=\partial_tW_m+\Gamma_tW_m$, it follows that
\[
(\mathrm{i}\partial_t+\Delta)W_m
=-2\Gamma_j\partial_jW_m-(\mathrm{i}\Gamma_t+\partial_j\Gamma_j+\Gamma_j^2)W_m
+\frac{\mathrm{i}}{2}\,\Im(W_m\mid W_j)\,W_j.
\]
Thus it remains only to prove the cancellation
\begin{equation}\label{eq:cancel-en}
(\mathrm{i}\Gamma_t+\partial_j\Gamma_j)W_m=0.
\end{equation}
Since $W_m$ is tangential, Lemma \ref{lem:gamma-basic-en} and \eqref{eq:divGamma-body-en} give
\[
\Gamma_tW_m = u\,\langle \partial_tu,W_m\rangle,
\qquad
(\partial_j\Gamma_j)W_m = u\,\langle \Delta u,W_m\rangle.
\]
Now $\partial_tu=J_u\Delta u$, and \eqref{eq:JW-en} implies that for any tangential vector $Y$,
\[
\langle J_uY,W_m\rangle = -\langle Y,J_uW_m\rangle=\mathrm{i}\langle Y,W_m\rangle.
\]
Therefore
\[
\mathrm{i}\langle \partial_tu,W_m\rangle+\langle \Delta u,W_m\rangle
=\mathrm{i}\langle J_u\Delta u,W_m\rangle+\langle \Delta u,W_m\rangle
=\mathrm{i}(\mathrm{i}\langle \Delta u,W_m\rangle)+\langle \Delta u,W_m\rangle=0,
\]
which proves \eqref{eq:cancel-en} and hence \eqref{eq:ambient-W-en}.

The heat-flow equation \eqref{eq:ambient-heat-en} is entirely analogous, except that the above cancellation is not needed there, so all zeroth-order terms are absorbed into $V_m^{(s)}$. The bound \eqref{eq:Vsbound-en} follows immediately from the expansion and \eqref{eq:Gammasize-en}.

\[

\]
This is exactly parallel to the role played by $2A_j\partial_j\psi_m$ in the caloric gauge.

<!-- proposition-4: body_md -->
\label{prop:energy-balance-en}
Define
\begin{equation}\label{eq:energy-density-en}
e[u]:=\frac12|\psi_x|^2=\frac14\sum_{m=1}^2(W_m\mid W_m),
\qquad
q_j[u]:=-\langle J_u\nabla_j\psi_m,\psi_m\rangle
=-\frac12\Im(\nabla_jW_m\mid W_m).
\end{equation}
Then one has the local conservation law
\begin{equation}\label{eq:energy-balance-en}
\partial_t e[u]+\partial_jq_j[u]=0.
\end{equation}
Equivalently, in terms of $W$,
\begin{equation}\label{eq:mass-balance-en}
\partial_t\frac14\sum_{m=1}^2(W_m\mid W_m)
-
\frac12\partial_j\Im\sum_{m=1}^2(\nabla_jW_m\mid W_m)=0.
\end{equation}
<!-- end proposition-4 -->
<!-- proof-6: body_md -->
We first explain the second identity in \eqref{eq:energy-density-en}. For any tangential vectors $A,X$, write
$W_A:=A+\mathrm{i} J_uA$ and $W_X:=X+\mathrm{i} J_uX$. Then one has the algebraic identity
\[
-\langle J_uA,X\rangle=-\frac12\Im(W_A\mid W_X).
\]
Substituting $A=\nabla_j\psi_m$ and $X=\psi_m$ gives
$q_j[u]= -\frac12\Im(\nabla_jW_m\mid W_m)$.

Now start from Proposition \ref{prop:covdiff-en}:
\[
\partial_t e[u]
=\langle \nabla_t\psi_m,\psi_m\rangle
=\langle J_u\nabla_j\nabla_j\psi_m,\psi_m\rangle
+\langle R(\psi_m,\psi_j)J_u\psi_j,\psi_m\rangle.
\]
Here both $m$ and $j$ are summed under the Einstein convention. The curvature term can be written as
\[
\langle R(\psi_m,\psi_j)J_u\psi_j,\psi_m\rangle
=-\langle \psi_m,J_u\psi_j\rangle\,\langle \psi_j,\psi_m\rangle.
\]
Set
\[
a_{mj}:=\langle \psi_m,J_u\psi_j\rangle,\qquad s_{mj}:=\langle \psi_j,\psi_m\rangle.
\]
Then $a_{mj}=-a_{jm}$ is an antisymmetric matrix, whereas $s_{mj}=s_{jm}$ is symmetric, and hence
\[
\sum_{m,j}a_{mj}s_{mj}=0.
\]
Therefore the total curvature contribution vanishes.

For the principal part,
\[
\langle J_u\nabla_j\nabla_j\psi_m,\psi_m\rangle
=\partial_j\langle J_u\nabla_j\psi_m,\psi_m\rangle
-\langle J_u\nabla_j\psi_m,\nabla_j\psi_m\rangle,
\]
and the last term vanishes because $\langle J_uX,X\rangle=0$ for every tangential vector $X$. Thus
\[
\partial_t e[u]+\partial_jq_j[u]=0.
\]
Substituting the $W$-expressions for $e[u]$ and $q_j[u]$ from \eqref{eq:energy-density-en} yields \eqref{eq:mass-balance-en}.

<!-- lemma-3: body_md -->
\label{lem:multiplier-en}
Let $a:I\times{\mathbb R}^2\to \mathbb{C}^{m\times m}$ satisfy
\[
a\in L^\infty_{t,x},
\qquad
\nabla_x a\in L_t^\infty L_x^2.
\]
Then for any $\mathbb{C}^m$-valued field $g$, one has
\begin{equation}\label{eq:multiplier-en}
\lVert ag\rVert_{\ell^2L_{t,x}^4(I\times{\mathbb R}^2)}
\lesssim
\Bigl(
\lVert a\rVert_{L_{t,x}^\infty}
+\lVert \nabla_x a\rVert_{L_t^\infty L_x^2}
\Bigr)
\lVert g\rVert_{\ell^2L_{t,x}^4(I\times{\mathbb R}^2)}.
\end{equation}
The implicit constant is independent of the interval $I$, the frequency scale, and the matrix dimension.
<!-- end lemma-3 -->
<!-- proof-7: body_md -->
Write $\Delta_j:=P_j$, $S_{j-4}:=P_{<j-4}$, and use Bony's decomposition
\[
ag=T_ag+T_ga+R(a,g),
\]
where
\[
T_ag:=\sum_j S_{j-4}a\,\Delta_j g,
\qquad
T_ga:=\sum_j S_{j-4}g\,\Delta_j a,
\qquad
R(a,g):=\sum_{|j-\ell|\le 3}\Delta_j a\,\Delta_\ell g.
\]
See \cite{Bony,BCD}.

First consider the low-high interaction $T_ag$. By frequency support, $P_kT_ag$ receives contributions only from terms with $|j-k|\le 4$, and therefore
\[
\lVert P_kT_ag\rVert_{L_{t,x}^4}
\lesssim
\sum_{|j-k|\le 4}
\lVert S_{j-4}a\,\Delta_j g\rVert_{L_{t,x}^4}
\le
\lVert a\rVert_{L_{t,x}^\infty}
\sum_{|j-k|\le 4}\lVert \Delta_j g\rVert_{L_{t,x}^4}.
\]
Taking the $\ell^2$ norm in $k$ and using finite overlap gives
\[
\lVert T_ag\rVert_{\ell^2L_{t,x}^4}
\lesssim
\lVert a\rVert_{L_{t,x}^\infty}\lVert g\rVert_{\ell^2L_{t,x}^4}.
\]

Next consider the high-low interaction $T_ga$. Again by frequency support,
\[
P_kT_ga
=
\sum_{|j-k|\le 4}P_k\bigl(S_{j-4}g\,\Delta_j a\bigr).
\]
Hence
\[
\lVert P_kT_ga\rVert_{L_{t,x}^4}
\lesssim
\sum_{|j-k|\le 4}
\lVert S_{j-4}g\rVert_{L_{t,x}^4}\lVert \Delta_j a\rVert_{L_t^\infty L_x^\infty}.
\]
By the Littlewood--Paley square-function estimate,
\[
\lVert g\rVert_{L_{t,x}^4}
\lesssim
\Biggl\|
\Bigl(\sum_\ell |\Delta_\ell g|^2\Bigr)^{1/2}
\Biggr\|_{L_{t,x}^4}
\lesssim
\Bigl(\sum_\ell \lVert \Delta_\ell g\rVert_{L_{t,x}^4}^2\Bigr)^{1/2}
=
\lVert g\rVert_{\ell^2L_{t,x}^4},
\]
and so $\lVert S_{j-4}g\rVert_{L_{t,x}^4}\le \lVert g\rVert_{L_{t,x}^4}\lesssim \lVert g\rVert_{\ell^2L_{t,x}^4}$. By Bernstein,
\[
\lVert \Delta_j a\rVert_{L_t^\infty L_x^\infty}
\lesssim
2^j\lVert \Delta_j a\rVert_{L_t^\infty L_x^2}.
\]
Therefore
\[
\lVert P_kT_ga\rVert_{L_{t,x}^4}
\lesssim
\lVert g\rVert_{\ell^2L_{t,x}^4}
\sum_{|j-k|\le 4}2^j\lVert \Delta_j a\rVert_{L_t^\infty L_x^2}.
\]
Taking the $\ell^2$ norm in $k$ and again using finite overlap yields
\[
\lVert T_ga\rVert_{\ell^2L_{t,x}^4}
\lesssim
\lVert \nabla_x a\rVert_{L_t^\infty L_x^2}\,
\lVert g\rVert_{\ell^2L_{t,x}^4}.
\]

Finally consider the resonant term $R(a,g)$. Let
\[
\widetilde\Delta_j:=\Delta_{j-1}+\Delta_j+\Delta_{j+1}.
\]
Then
\[
R(a,g)=\sum_j \Delta_j a\,\widetilde\Delta_j g,
\]
and $P_kR(a,g)$ receives contributions only from terms with $|j-k|\le 5$. Thus
\[
\lVert P_kR(a,g)\rVert_{L_{t,x}^4}
\lesssim
\sum_{|j-k|\le 5}
\lVert \Delta_j a\rVert_{L_t^\infty L_x^\infty}
\lVert \widetilde\Delta_j g\rVert_{L_{t,x}^4}
\lesssim
\sum_{|j-k|\le 5}
2^j\lVert \Delta_j a\rVert_{L_t^\infty L_x^2}
\lVert \widetilde\Delta_j g\rVert_{L_{t,x}^4}.
\]
Taking the $\ell^2$ norm in $k$ and using finite overlap once more gives
\[
\lVert R(a,g)\rVert_{\ell^2L_{t,x}^4}
\lesssim
\lVert \nabla_x a\rVert_{L_t^\infty L_x^2}\,
\lVert g\rVert_{\ell^2L_{t,x}^4}.
\]
Adding the three parts yields \eqref{eq:multiplier-en}.

<!-- lemma-4: body_md -->
\label{lem:compare-en}
Fix $E_0<4\pi$. Then there exists a constant $C(E_0)\ge 1$ such that, if $E(u)\le E_0$ and $u-Q\in L_x^2$, then for every compact interval $I$ one has
\begin{equation}\label{eq:compare-en}
C(E_0)^{-1}\lVert \nabla u\rVert_{\ell^2L^4_{t,x}(I\times{\mathbb R}^2)}
\le
\lVert W_x\rVert_{\ell^2L^4_{t,x}(I\times{\mathbb R}^2)}
\le
C(E_0)\lVert \nabla u\rVert_{\ell^2L^4_{t,x}(I\times{\mathbb R}^2)}.
\end{equation}
<!-- end lemma-4 -->
<!-- proof-8: body_md -->
By \eqref{eq:intro-basic-notation-en}--\eqref{eq:intro-complex-notation-en}, and writing $J_QX:=Q\times X$, one has
\[
W_x=(I+\mathrm{i} J_Q)\nabla u + \mathrm{i}\bigl((u-Q)\times \nabla u\bigr).
\]
Here $I+\mathrm{i} J_Q$ is a constant-coefficient zeroth-order multiplier, and hence
\[
\lVert (I+\mathrm{i} J_Q)\nabla u\rVert_{\ell^2L_{t,x}^4}
\lesssim
\lVert \nabla u\rVert_{\ell^2L_{t,x}^4}.
\]
On the other hand, pointwise one has $|u-Q|\le 2$, while
\[
\lVert \nabla_x(u-Q)\rVert_{L_t^\infty L_x^2}
=
\lVert \nabla u\rVert_{L_t^\infty L_x^2}
\le (2E_0)^{1/2}.
\]
Applying Lemma \ref{lem:multiplier-en} with
\[
a=u-Q,
\qquad
g=\nabla u,
\]
one obtains
\[
\lVert (u-Q)\times \nabla u\rVert_{\ell^2L_{t,x}^4}
\lesssim
\bigl(1+E_0^{1/2}\bigr)\lVert \nabla u\rVert_{\ell^2L_{t,x}^4}.
\]
Thus
\[
\lVert W_x\rVert_{\ell^2L_{t,x}^4}
\lesssim_{E_0}
\lVert \nabla u\rVert_{\ell^2L_{t,x}^4}.
\]

The reverse estimate is even more direct. Since
\[
\partial_m u=\frac12\bigl(W_m+\overline{W_m}\bigr),
\]
and the Littlewood--Paley projections are real Fourier multipliers, one has
\[
P_k\partial_m u=\frac12\bigl(P_kW_m+\overline{P_kW_m}\bigr).
\]
Hence for each $k$,
\[
\lVert P_k\partial_m u\rVert_{L_{t,x}^4}
\le
\frac12\lVert P_kW_m\rVert_{L_{t,x}^4}
+\frac12\lVert \overline{P_kW_m}\rVert_{L_{t,x}^4}
=
\lVert P_kW_m\rVert_{L_{t,x}^4}.
\]
Summing over $m=1,2$ and then taking the $\ell^2$ norm in $k$ yields
\[
\lVert \nabla u\rVert_{\ell^2L_{t,x}^4}
\lesssim
\lVert W_x\rVert_{\ell^2L_{t,x}^4}.
\]
Combining the two bounds proves \eqref{eq:compare-en}.

<!-- lemma-5: body_md -->
\label{lem:free-linear-en}
Let
\[
S_k(t)f:=1_I(t)e^{\mathrm{i}(t-t_0)\Delta}P_k f.
\]
Then for every $f\in L_x^2({\mathbb R}^2;\mathbb{C}^3)$,
\begin{align}
\lVert S_k f\rVert_{L_t^\infty L_x^2}
+\lVert S_k f\rVert_{L_{t,x}^4}
&\lesssim \lVert f\rVert_{L_x^2},\label{eq:freeL4-en}\\
2^{-k/6}\sup_{\theta\in\mathbb S^1}\lVert S_k f\rVert_{L_\theta^{3,6}}
+
2^{k/6}\sup_{|j-k|\le 20}\sup_{\theta\in\mathbb S^1}\lVert P_{j,\theta}S_kf\rVert_{L_\theta^{6,3}}
&\lesssim \lVert f\rVert_{L_x^2},\label{eq:freelateral-en}\\
2^{k/2}\sup_{|j-k|\le20}\sup_{\theta\in\mathbb S^1}\sup_{|\lambda|<2^{k-40}}
\lVert P_{j,\theta}S_kf\rVert_{L_{\theta,\lambda}^{\infty,2}}
+
2^{-k/2}\sup_{\theta\in\mathbb S^1}\lVert S_kf\rVert_{L_{\theta,W_{k+40}}^{2,\infty}}
&\lesssim \lVert f\rVert_{L_x^2}.\label{eq:freesmooth-en}
\end{align}
In particular, by Bernstein,
\begin{equation}\label{eq:free-max-en}
2^{-k/2}\lVert S_kf\rVert_{L_x^4L_t^\infty}
\lesssim
\lVert S_kf\rVert_{L_t^\infty L_x^2}
\lesssim
\lVert f\rVert_{L_x^2}.
\end{equation}
<!-- end lemma-5 -->
<!-- proof-9: body_md -->
The proof is divided into four steps. Since time translation, rotation, and scaling preserve the form of these mixed norms, it is enough to prove the scalar case with $t_0=0$ and $k=0$; the vector-valued case follows by summing over components.

First, unitarity directly gives
\[
\lVert S_0 f\rVert_{L_t^\infty L_x^2}\le \lVert f\rVert_{L_x^2}.
\]
Let $Tf(t,x):=1_I(t)e^{\mathrm{i} t\Delta}P_0f(x)$. By the $TT^*$ principle, it suffices to prove
\[
\lVert TT^*F\rVert_{L_{t,x}^4}\lesssim \lVert F\rVert_{L_{t,x}^{4/3}}.
\]
Since
\[
(TT^*F)(t)=\int_I e^{\mathrm{i}(t-s)\Delta}P_0^2F(s)\,ds,
\]
and the kernel of $e^{\mathrm{i}\tau\Delta}P_0^2$ satisfies the stationary phase bound
\[
|K(\tau,x)|\lesssim_N (1+|\tau|)^{-1}\bigl(1+|x|/(1+|\tau|)\bigr)^{-N},
\]
we have
\[
\lVert e^{\mathrm{i}\tau\Delta}P_0^2 g\rVert_{L_x^4}
\lesssim (1+|\tau|)^{-1/2}\lVert g\rVert_{L_x^{4/3}}.
\]
Thus
\[
\lVert TT^*F(t)\rVert_{L_x^4}
\lesssim \int_I (1+|t-s|)^{-1/2}\lVert F(s)\rVert_{L_x^{4/3}}\,ds.
\]
The one-dimensional Hardy--Littlewood--Sobolev inequality in time gives
\[
\lVert TT^*F\rVert_{L_t^4L_x^4}
\lesssim \lVert F\rVert_{L_t^{4/3}L_x^{4/3}},
\]
which proves \eqref{eq:freeL4-en}.

Second, we prove the local-smoothing endpoint. Fix $|j|\le20$ and $\theta=e_1$, and set
\[
\nu(t,x):=P_{j,e_1}e^{\mathrm{i} t\Delta}P_0f.
\]
Then $\widehat\nu$ is supported where $|\xi|\sim1$ and $|\xi_1|\sim1$. For fixed $x_1$, taking the Fourier transform in $(x_2,t)$ gives
\[
\mathcal F_{x_2,t}\nu(x_1,\eta,\tau)
=\int e^{\mathrm{i} x_1\xi_1}m_j(\xi_1,\eta)\widehat f(\xi_1,\eta)
\,\delta(\tau-\xi_1^2-\eta^2)\,d\xi_1.
\]
Since $|\xi_1|\sim1$, the coarea formula and Cauchy--Schwarz imply
\[
\int_{{\mathbb R}^2}|\mathcal F_{x_2,t}\nu(x_1,\eta,\tau)|^2\,d\eta\,d\tau
\lesssim \lVert f\rVert_{L_x^2}^2.
\]
By Plancherel in $(x_2,t)$,
\[
\lVert P_{j,e_1}e^{\mathrm{i} t\Delta}P_0f\rVert_{L_{e_1}^{\infty,2}}
\lesssim \lVert f\rVert_{L_x^2}.
\]
Scaling gives
\[
\lVert P_{j,\theta}S_kf\rVert_{L_{\theta,0}^{\infty,2}}
\lesssim 2^{-k/2}\lVert f\rVert_{L_x^2},
\qquad |j-k|\le20.
\]
The Galilean commutation relation with the free Schrödinger group then yields
\[
\lVert P_{j,\theta}S_kf\rVert_{L_{\theta,\lambda}^{\infty,2}}
\lesssim 2^{-k/2}\lVert f\rVert_{L_x^2},
\qquad |j-k|\le20,\\ |\lambda|<2^{k-40}.
\]
This is the local-smoothing part of \eqref{eq:freesmooth-en}.

Third, we prove the lateral Strichartz estimates and the two-dimensional Galilean maximal-function endpoint. Again take $k=0$ and $\theta=e_1$. For the Galilean endpoint, let $I\subset[0,2^{2K}]$ and set $\delta:=2^{-2K-40}$. Decompose the unit annulus in the variable $2\xi_1$ as
\[
1=\sum_{\lambda\in W_{40}}\chi_\lambda(2\xi_1),
\qquad
\operatorname{supp}\chi_\lambda\subset\{|2\xi_1-\lambda|\le4\delta\},
\]
with uniformly bounded overlap. Let $f_\lambda$ be defined by
\[
\widehat f_\lambda(\xi)=\chi_\lambda(2\xi_1)\widehat{P_0f}(\xi),
\qquad
\nu_\lambda(t,x):=1_I(t)e^{\mathrm{i} t\Delta}f_\lambda(x).
\]
It suffices to prove
\begin{equation}\label{eq:gal-piece-en}
\lVert \nu_\lambda\rVert_{L_{e_1,\lambda}^{2,\infty}}
\lesssim \delta^{1/2}\lVert f_\lambda\rVert_{L_x^2}.
\end{equation}
Indeed, summing over $\lambda$ and using Cauchy--Schwarz, the effective number $O(\delta^{-1})$ of strips, and almost orthogonality yields
\[
\lVert S_0f\rVert_{L_{e_1,W_{40}}^{2,\infty}}
\lesssim \lVert f\rVert_{L_x^2}.
\]
Scaling gives
\begin{equation}\label{eq:gal-max-proof-en}
2^{-k/2}\sup_{\theta\in\mathbb S^1}
\lVert S_kf\rVert_{L_{\theta,W_{k+40}}^{2,\infty}}
\lesssim \lVert f\rVert_{L_x^2}.
\end{equation}

To prove \eqref{eq:gal-piece-en}, apply the Galilean transform to $\nu_\lambda$ and ignore unimodular phases. One obtains
\[
T_{\lambda e_1}\nu_\lambda(t,x)
=1_I(t)\int e^{\mathrm{i}(x_1a/2+x_2\eta+t(\eta^2+a^2/4))}
 b_\lambda(a,\eta)\,da\,d\eta,
\]
where $|a|\lesssim\delta$, $|\eta|\lesssim1$, and $\lVert b_\lambda\rVert_{L^2_{a,\eta}}\sim\lVert f_\lambda\rVert_{L_x^2}$. For fixed $x_1$, the map
\[
(a,\eta)\mapsto(\tau,\eta):=(\eta^2+a^2/4,\eta)
\]
has fiber length $O(\delta)$ when $|a|\lesssim\delta$. The coarea formula and Cauchy--Schwarz therefore give
\[
\lVert T_{\lambda e_1}\nu_\lambda(x_1,\cdot)\rVert_{H^2_{t,x_2}}
\lesssim
\delta^{1/2}
\Bigl(\int |\mathcal F^{-1}_{a\to x_1}b_\lambda(x_1,\eta)|^2\,d\eta\Bigr)^{1/2}.
\]
The embedding $H^2_{t,x_2}\hookrightarrow L^\infty_{t,x_2}$, followed by $L^2$ in $x_1$ and Plancherel, proves \eqref{eq:gal-piece-en}.

For the lateral Strichartz estimates, first assume that the Fourier support lies in a fixed angular sector with $|\xi_1|\sim1$. Let
\[
Ef(t,x_1,x_2):=e^{\mathrm{i} t\Delta}P_0f(x_1,x_2).
\]
For fixed $x_1$,
\[
\mathcal F_{x_2,t}Ef(x_1,\eta,\tau)
=\int e^{\mathrm{i} x_1\xi_1}m(\xi_1,\eta)\widehat f(\xi_1,\eta)
\delta(\tau-\xi_1^2-\eta^2)\,d\xi_1.
\]
Since $|\xi_1|\sim1$, the change of variables $(\xi_1,\eta)\mapsto(\tau,\eta)$ has Jacobian $2|\xi_1|\sim1$. This nondegeneracy, together with Hausdorff--Young, the coarea formula, and $TT^*$, gives
\[
\lVert Ef\rVert_{L_{e_1}^{p,q}}
\lesssim
\lVert f\rVert_{L_x^2},
\qquad
2<p\le\infty,
\quad 2\le q<\infty,
\quad \frac1p+\frac1q=\frac12.
\]
After a finite angular decomposition and rotation,
\[
\lVert S_0 f\rVert_{L_\theta^{3,6}}
+
\sup_{|j|\le20}\lVert P_{j,\theta}S_0f\rVert_{L_\theta^{6,3}}
\lesssim \lVert f\rVert_{L_x^2}.
\]
Scaling gives
\begin{equation}\label{eq:lateral-proof-en}
2^{-k/6}\sup_{\theta\in\mathbb S^1}\lVert S_k f\rVert_{L_\theta^{3,6}}
+
2^{k/6}\sup_{|j-k|\le20}\sup_{\theta\in\mathbb S^1}\lVert P_{j,\theta}S_kf\rVert_{L_\theta^{6,3}}
\lesssim \lVert f\rVert_{L_x^2}.
\end{equation}
Equations \eqref{eq:gal-max-proof-en} and \eqref{eq:lateral-proof-en} imply \eqref{eq:freelateral-en} and the remaining part of \eqref{eq:freesmooth-en}.

Finally, Bernstein gives
\[
\lVert P_k h\rVert_{L_x^4}\lesssim 2^{k/2}\lVert P_k h\rVert_{L_x^2}.
\]
Taking $h=e^{\mathrm{i}(t-t_0)\Delta}P_kf$ and then the supremum in $t$ proves \eqref{eq:free-max-en}.

<!-- proof-12: body_md -->
Fix a physical time $t$ and regard $u(s,x):=u(s,t,x)$ as a harmonic map heat flow depending only on the heat variable $s$. The energy dissipation identity
\[
\frac{d}{ds}E(u(s))=-\int_{{\mathbb R}^2}|\partial_su(s,x)|^2\,dx
\]
implies
\begin{equation}\label{eq:heat-energy-monotone-proof-en}
\sup_{s>0}\lVert W_x(s)\rVert_{L_x^2}\lesssim E_0^{1/2},
\qquad
\int_0^\infty\lVert \partial_su(s)\rVert_{L_x^2}^2\,ds\lesssim E_0.
\end{equation}

The first step is the scale-invariant first-order bound. We claim that
\begin{equation}\label{eq:heat-gradient-critical-bound-en}
\sup_{s>0}s^{1/2}\lVert \nabla_xu(s)\rVert_{L_x^\infty}\le C(E_0).
\end{equation}
If this failed, there would be points $(s_n,x_n)$ such that $s_n|\nabla_xu(s_n,x_n)|^2\to\infty$. Set
\[
\lambda_n:=|\nabla_xu(s_n,x_n)|^{-1},
\qquad
\nu_n(\sigma,y):=u(s_n+\lambda_n^2\sigma,x_n+\lambda_n y).
\]
Since $s_n/\lambda_n^2\to\infty$, each $\nu_n$ is defined at least on a fixed backward unit parabolic cylinder, and satisfies
\[
|\nabla_y\nu_n(0,0)|=1,
\qquad
E(\nu_n(\sigma))=E(u(s_n+\lambda_n^2\sigma))\le E_0<4\pi.
\]
By local parabolic compactness, a subsequence converges to a nontrivial ancient limit $\nu_\infty$. This limit is stationary in the heat variable, and hence gives a nonconstant finite-energy harmonic map ${\mathbb R}^2\to\mathbb{S}^2$ with energy not exceeding $E_0<4\pi$. This contradicts the fact that the least energy of a nonconstant sphere-valued harmonic map is $4\pi$. Hence \eqref{eq:heat-gradient-critical-bound-en} holds. Since
\[
|W_x|\sim |\nabla_xu|,
\qquad
|\Gamma_x|\lesssim |\nabla_xu|,
\]
we obtain
\begin{equation}\label{eq:heat-first-order-pointwise-en}
\sup_{s>0}s^{1/2}\lVert W_x(s)\rVert_{L_x^\infty}
+
\sup_{s>0}s^{1/2}\lVert \Gamma_x(s)\rVert_{L_x^\infty}
\lesssim_{E_0}1.
\end{equation}

The second step upgrades this scale-invariant bound to higher parabolic smoothing. For each component $W_m$, equation \eqref{eq:ambient-heat-en} takes the form
\[
(\partial_s-\Delta)W_m=2\Gamma_j\partial_jW_m+V_m^{(s)}.
\]
Using \eqref{eq:Vsbound-en}, \eqref{eq:heat-first-order-pointwise-en}, and \eqref{eq:heat-energy-monotone-proof-en}, the right-hand side has scale-uniform $L_x^2$ control on each heat layer $[s/2,s]$. More precisely,
\begin{equation}\label{eq:heat-force-L2-proof-en}
\sup_{s'>0}
\Bigl(
(s')^{1/2}\lVert \Gamma_x(s')\rVert_{L_x^\infty}
+(s')\lVert \partial_x\Gamma_x(s')\rVert_{L_x^2}
+(s')\lVert \Gamma_s(s')\rVert_{L_x^2}
+(s')\lVert V_x(s')\rVert_{L_x^2}
\Bigr)
\lesssim_{E_0}1.
\end{equation}
Now fix $s>0$ and integrate the heat equation from $s/2$ to $s$:
\[
W_m(s)=e^{\frac s2\Delta}W_m(s/2)
+
\int_{s/2}^s e^{(s-s')\Delta}\bigl(2\Gamma_j\partial_jW_m+V_m^{(s')}\bigr)(s')\,ds'.
\]
For any integer $r\ge0$, the heat kernel estimates
\[
\lVert \nabla_x^re^{\tau\Delta}f\rVert_{L_x^2}\lesssim \tau^{-r/2}\lVert f\rVert_{L_x^2},
\qquad
\lVert \nabla_x^re^{\tau\Delta}f\rVert_{L_x^\infty}\lesssim \tau^{-(r+1)/2}\lVert f\rVert_{L_x^2}
\]
together with \eqref{eq:heat-energy-monotone-proof-en} and \eqref{eq:heat-force-L2-proof-en} yield
\[
\sup_{s>0}s^{r/2}\lVert \nabla_x^rW_x(s)\rVert_{L_x^2}
+
\sup_{s>0}s^{(r+1)/2}\lVert \nabla_x^rW_x(s)\rVert_{L_x^\infty}
\lesssim_{E_0,r}1.
\]
Taking the supremum over the physical time $t$ gives \eqref{eq:heat-smooth-W-en}.

It remains to handle $\Gamma_x$ and $\Gamma_s$. By \eqref{eq:intro-complex-notation-en} and
\[
\partial_su=\Delta u+|\nabla_xu|^2u,
\]
$\nabla_x^r\Gamma_x$ is a linear combination of derivatives up to order $r+1$ of $u$, while $\nabla_x^r\Gamma_s$ is built from derivatives up to order $r+2$ of $u$ and lower-order products. Substituting the higher estimates for $W_x$ obtained above gives \eqref{eq:heat-smooth-Gamma-en} and the $L_x^\infty$ bound following it.

Put
\[
w_k(s):=(1+s2^{2k})^{-4}.
\]
We run a joint bootstrap on $0\le s,\tilde s\le S$. Assume that, for a sufficiently large constant $M\ge1$,
\begin{align}
\lVert P_kW_x(s)\rVert_{X_k(I)}+
\lVert P_k\Gamma_x(s)\rVert_{L_{t,x}^4(I\times{\mathbb R}^2)}
&\le M\upsilon_kw_k(s),
\label{eq:heat-bootstrap-en}\\
\lVert (P_j\Gamma_x(s))(P_kW_x(\tilde s))\rVert_{L_{t,x}^2(I\times{\mathbb R}^2)}
&\le M2^{-\frac12|j-k|}\upsilon_j\upsilon_kw_j(s)w_k(\tilde s).
\label{eq:heat-bootstrap-bilin-en}
\end{align}
We prove that, if $\varepsilon_0=\varepsilon_0(E_0)$ is sufficiently small, then the constant in \eqref{eq:heat-bootstrap-en} can be improved to $M/2$.

First, we prove $L_x^2$ decay for the initial data of the heat-flow slices. Fix $t=t_0$ and write the Duhamel formula for \eqref{eq:ambient-heat-en}:
\begin{equation}\label{eq:heat-duhamel-in-proof-en}
P_kW_x(s,t_0)
=e^{s\Delta}P_kW_x(0,t_0)
+\int_0^s e^{(s-s')\Delta}P_k\Bigl(2\Gamma_j\partial_jW_x+V^{(s')}\Bigr)(t_0)\,ds'.
\end{equation}
The two terms in \eqref{eq:heat-duhamel-in-proof-en} are estimated separately. The linear term is bounded by frequency-localized heat kernel decay:
\begin{equation}\label{eq:heat-initial-linear-en}
\lVert e^{s\Delta}P_kW_x(0,t_0)\rVert_{L_x^2}
\lesssim e^{-cs2^{2k}}\beta_k
\lesssim \beta_kw_k(s).
\end{equation}
For the Duhamel term, split the integral into $[0,s/2]$ and $[s/2,s]$. On the first subinterval, use
\[
e^{-c(s-s')2^{2k}}\lesssim_N w_k(s)
\qquad (0\le s'\le s/2)
\]
and the parabolic smoothing input of Lemma \ref{lem:heat-smooth-en}. This gives
\[
\lVert e^{(s-s')\Delta}P_k\bigl(2\Gamma_j\partial_jW_x+V^{(s')}\bigr)(t_0)\rVert_{L_x^2}
\lesssim w_k(s)(s')^{-1}\beta_k.
\]
On the second subinterval $s'\in[s/2,s]$, one uses the same parabolic smoothing bounds directly and the fact that $s'\sim s$, obtaining again
\[
\int_{s/2}^s
\lVert e^{(s-s')\Delta}P_k\bigl(2\Gamma_j\partial_jW_x+V^{(s')}\bigr)(t_0)\rVert_{L_x^2}\,ds'
\lesssim \beta_kw_k(s).
\]
Together with \eqref{eq:heat-initial-linear-en}, this yields
\begin{equation}\label{eq:heat-initial-slice-en}
\lVert P_kW_x(s,t_0)\rVert_{L_x^2}
\lesssim \beta_kw_k(s).
\end{equation}

Second, we prove forcing decay for each fixed heat time $s$. By \eqref{eq:ambient-W-en}, $W_x(s)$ satisfies, as a function of $(t,x)$,
\begin{equation}\label{eq:heat-slice-sch-en}
(\mathrm{i}\partial_t+\Delta)P_kW_x(s)=P_kN(s),
\end{equation}
where
\begin{equation}\label{eq:heat-slice-nonlinearity-en}
N(s):=-2\Gamma_j(s)\partial_jW_x(s)-\Gamma_j(s)^2W_x(s)
+\frac{\mathrm{i}}{2}\Im(W_x(s)\mid W_j(s))W_j(s).
\end{equation}
The decomposition \eqref{eq:heat-slice-nonlinearity-en} separates the transport term from the zeroth-order cubic terms. The latter are treated by Lemma \ref{lem:product-en}, the bootstrap assumption \eqref{eq:heat-bootstrap-en}, and
\[
\sum_{\ell\in\mathbb Z}\lVert P_\ell W_x(s)\rVert_{L_{t,x}^4}^2
\lesssim M^2\sum_\ell \alpha_\ell^2w_\ell(s)^2
\lesssim M^2\varepsilon_0^2.
\]
Thus
\begin{equation}\label{eq:heat-perturbative-force-en}
\lVert P_k\Bigl(\Gamma_j(s)^2W_x(s)
-\frac{\mathrm{i}}{2}\Im(W_x(s)\mid W_j(s))W_j(s)\Bigr)\rVert_{Y_k(I)}
\lesssim \varepsilon_0^2\upsilon_kw_k(s).
\end{equation}
For the transport term, decompose $\Gamma_j(s)\partial_jW_x(s)$ into low-high, comparable, and high-high parts. The comparable-frequency and high-high pieces are estimated directly using the dyadic decomposition, the small $L^4$ norm, Bernstein, and Lemma \ref{lem:product-en}:
\begin{equation}\label{eq:heat-transport-pert-en}
\lVert P_k\bigl(\Gamma_j(s)\partial_jW_x(s)\bigr)^{\mathrm{comp+hh}}\rVert_{Y_k(I)}
\lesssim \varepsilon_0^2\upsilon_kw_k(s).
\end{equation}
For the low-high part, put the derivative on the high-frequency factor and use the joint bootstrap assumption \eqref{eq:heat-bootstrap-bilin-en}:
\begin{align}
\lVert P_k\bigl(\Gamma_j(s)\partial_jW_x(s)\bigr)^{\mathrm{lh}}\rVert_{Y_k(I)}
&\lesssim
\upsilon_kw_k(s)\sum_{\ell\le k-20}\upsilon_\ell w_\ell(s)\,\upsilon_\ell\notag\\
&\lesssim_E \upsilon_kw_k(s),
\label{eq:heat-transport-lh-en}
\end{align}
where the last step uses only the $\ell^2$ summability of $\{\upsilon_\ell\}$. Combining \eqref{eq:heat-perturbative-force-en}, \eqref{eq:heat-transport-pert-en}, and \eqref{eq:heat-transport-lh-en}, we obtain
\begin{equation}\label{eq:heat-force-en}
\lVert P_kN(s)\rVert_{Y_k(I)}
\lesssim_E \upsilon_kw_k(s).
\end{equation}

Third, apply the linear estimate to \eqref{eq:heat-slice-sch-en}. Using Proposition \ref{prop:linear-en}, \eqref{eq:heat-initial-slice-en}, and \eqref{eq:heat-force-en}, we get
\[
\lVert P_kW_x(s)\rVert_{X_k(I)}
\lesssim \beta_kw_k(s)+C_E\upsilon_kw_k(s)
\lesssim C_E\upsilon_kw_k(s).
\]
Taking the bootstrap constant $M=2C_E$, this becomes
\begin{equation}\label{eq:heat-bootstrap-improved-W-en}
\lVert P_kW_x(s)\rVert_{X_k(I)}
\le \frac{M}{2}\upsilon_kw_k(s).
\end{equation}

Fourth, improve the bootstrap bound for $\Gamma_x$. Since
\[
\Gamma_x=u\otimes(\partial_xu)^\top-(\partial_xu)\otimes u^\top,
\qquad
\partial_xu=\frac12(W_x+\overline{W_x}),
\]
Bony decomposition gives
\[
\lVert P_k\Gamma_x(s)\rVert_{L_{t,x}^4}
\lesssim
\lVert P_k\partial_xu(s)\rVert_{L_{t,x}^4}
+
\sum_{\ell\ge k-10}2^{k-\ell}\lVert P_\ell\partial_xu(s)\rVert_{L_{t,x}^4}.
\]
Using \eqref{eq:heat-bootstrap-improved-W-en} and the slow-variation summation for frequency envelopes, we get
\[
\lVert P_k\Gamma_x(s)\rVert_{L_{t,x}^4(I\times{\mathbb R}^2)}
\le \frac{M}{2}\upsilon_kw_k(s).
\]
Thus the first part of the joint bootstrap has been improved; the conclusion \eqref{eq:heatdecay-en} follows by the continuity argument.

\label{lem:free-bilin-en}
Let $f_j,g_k\in L_x^2({\mathbb R}^2;\mathbb{C}^3)$ have Fourier supports contained in $\Xi_j$ and $\Xi_k$, respectively. Then

We first prove the scalar case. By symmetry it is enough to consider $j\ge k$. If $|j-k|\le4$, then H\"older and the $L^4_{t,x}$ Strichartz estimate in Lemma \ref{lem:free-linear-en} give

\le \lVert e^{it\Delta}f_j\rVert_{L^4_{t,x}}\lVert e^{it\Delta}g_k\rVert_{L^4_{t,x}}
\lesssim \lVert f_j\rVert_{L_x^2}\lVert g_k\rVert_{L_x^2}.
\]
Since $2^{-\frac12|j-k|}\sim1$ in this case, \eqref{eq:free-bilin-en} follows.

\[
\nu_j(t,x):=e^{it\Delta}f_j(x),
\qquad

\]
With the space-time Fourier transform convention

\]
we have
\[
\widetilde{\nu_j}(\tau,\xi)=\delta(\tau-|\xi|^2)\widehat f_j(\xi),
\qquad
\widetilde{\omega_k}(\tau,\xi)=\delta(\tau-|\xi|^2)\widehat g_k(\xi).
\]
Therefore
\begin{equation}\label{eq:bilin-conv-en}
\widetilde{\nu_j\omega_k}(\tau,\xi)
=\int_{{\mathbb R}^2}\widehat f_j(\eta)\widehat g_k(\xi-\eta)
\,\delta\bigl(\tau-|\eta|^2-|\xi-\eta|^2\bigr)\,d\eta.
\end{equation}
Applying Cauchy--Schwarz to the right-hand side of \eqref{eq:bilin-conv-en} gives
\[
|\widetilde{\nu_j\omega_k}(\tau,\xi)|^2
\le K_{j,k}(\tau,\xi)
\int_{{\mathbb R}^2}|\widehat f_j(\eta)|^2|\widehat g_k(\xi-\eta)|^2
\delta\bigl(\tau-|\eta|^2-|\xi-\eta|^2\bigr)\,d\eta,
\]
where
\[
K_{j,k}(\tau,\xi)
:=\int_{{\mathbb R}^2}\mathbf 1_{\Xi_j}(\eta)\mathbf 1_{\Xi_k}(\xi-\eta)
\delta\bigl(\tau-|\eta|^2-|\xi-\eta|^2\bigr)\,d\eta.
\]
It remains to prove
\begin{equation}\label{eq:K-bound-en}
K_{j,k}(\tau,\xi)\lesssim 2^{k-j}.
\end{equation}
For fixed $(\tau,\xi)$, set
\[
\Phi_\xi(\eta):=|\eta|^2+|\xi-\eta|^2
=2\Bigl|\eta-\frac\xi2\Bigr|^2+\frac{|\xi|^2}{2}.
\]
The level set $\Phi_\xi(\eta)=\tau$ is a circle centered at $\xi/2$. On the support
\[

\qquad
\xi-\eta\in\Xi_k,
\]
and since $j\ge k+5$,
\[
|\nabla_\eta\Phi_\xi(\eta)|=2|2\eta-\xi|=2|\eta-(\xi-\eta)|
\ge 2\bigl||\eta|-|\xi-\eta|\bigr|
\gtrsim 2^j.
\]
Also, $\xi-\eta\in\Xi_k$ restricts $\eta$ to an annulus centered at $\xi$ of radius and thickness $\sim2^k$; hence the intersection of the level circle with this annulus has length at most $C2^k$. By the coarea formula,
\[
K_{j,k}(\tau,\xi)
\le\int_{\Phi_\xi=\tau}\frac{\mathbf 1_{\Xi_j}(\eta)\mathbf 1_{\Xi_k}(\xi-\eta)}{|\nabla_\eta\Phi_\xi(\eta)|}\,d\sigma(\eta)
\lesssim 2^{-j}2^k,
\]
which proves \eqref{eq:K-bound-en}.

By Plancherel and Fubini,
\begin{align*}
\lVert \nu_j\omega_k\rVert_{L^2_{t,x}}^2
&=\lVert \widetilde{\nu_j\omega_k}\rVert_{L^2_{\tau,\xi}}^2\\
&\lesssim 2^{k-j}
\int_{{\mathbb R}}\int_{{\mathbb R}^2}\int_{{\mathbb R}^2}
|\widehat f_j(\eta)|^2|\widehat g_k(\xi-\eta)|^2
\delta\bigl(\tau-|\eta|^2-|\xi-\eta|^2\bigr)
\,d\eta\,d\xi\,d\tau\\

<!-- lemma-11: body_md -->
\label{lem:X-bilinear-en}

\le \lVert U\rVert_{L_{t,x}^4}\lVert V\rVert_{L_{t,x}^4}
\le \lVert U\rVert_{X_j}\lVert V\rVert_{X_k},

\]
Summing over $\nu$ gives the result. If $k\ge j+11$, exchange the roles of $U$ and $V$.
