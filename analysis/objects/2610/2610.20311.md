<!-- theorem-1: body_md -->
\label{thm:main}
There is an absolute positive integer $N_0$ such that, for every $N\in\mathbb N$ with $N\geq N_0$,
\begin{equation}\label{eq:main}
 f(N)\geq \frac{a\kappa}{24}\,N^{N-1}(\log \log \log N),
\end{equation}
where the positive real constants $a,b,\kappa$ are
\[
 a=\frac12\log 2,\qquad b=4\log 2,\qquad
 \kappa=\frac{1}{32(1+c_2b)}.
\]
Here $c_2>0$ denotes the absolute constant in part~\textup{(b)} of
\Cref{thm:lau-wu}.

<!-- lemma-1: body_md -->
\label{lem:cheb}
There is a real constant \(t_0\geq2\) such that, for every real \(t\geq t_0\),
\begin{equation}\label{eq:cheb}
 at\leq\vartheta(t)\leq bt,
 \qquad
 \frac{at}{\log t}\leq\pi(t)\leq\frac{8t}{\log t},
\end{equation}
where \(a=\frac12\log2\) and \(b=4\log2\).
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
We first compare $\psi$ with central binomial coefficients, then remove
the higher prime powers and estimate $\pi$.

\medskip
\noindent\textbf{Step 1. The binomial comparison.}
For $p\in\mathcal P$ and $m\in\mathbb N$, let $v_p(m)$ denote the exponent
of $p$ in the prime factorisation of $m$.  Counting multiples of each
power of $p$ in $m!$ gives
\begin{equation*}
 v_p(m!)=
 \sum_{\substack{j\in\mathbb N\\p^j\leq m}}
 \left\lfloor\frac{m}{p^j}\right\rfloor.
\end{equation*}
Indeed, a factor of $m!$ divisible by $p^r$ but not by $p^{r+1}$,
where $r\in\mathbb Z_{\geq0}$, is counted once for each
$j\in\{1,\ldots,r\}$.  Empty sums are understood to be zero.
Unique factorisation therefore gives, for every $n\in\mathbb N$,
\begin{equation*}
 \log\binom{2n}{n}
 =\sum_{\substack{p\in\mathcal P\\p\leq2n}}
   \sum_{j=1}^{\lfloor\log(2n)/\log p\rfloor}
 \left(
 \left\lfloor\frac{2n}{p^j}\right\rfloor
 -2\left\lfloor\frac{n}{p^j}\right\rfloor
 \right)\log p.
\end{equation*}
Each parenthesised expression is $0$ or $1$, since
$\lfloor2u\rfloor-2\lfloor u\rfloor\in\{0,1\}$ for every real
$u\geq0$.  It equals $1$ whenever $n<p^j\leq2n$.  Thus
\begin{equation}\label{eq:cheb-binomial-psi}
 \psi(2n)-\psi(n)
 \leq\log\binom{2n}{n}\leq\psi(2n)
 \qquad(n\in\mathbb N).
\end{equation}
The central coefficient is the largest of the coefficients
$\displaystyle \binom{2n}{k}$, $k\in\{0,\ldots,2n\}$: their successive ratio is
$(2n-k)/(k+1)$ for $k\in\{0,\ldots,2n-1\}$, which is greater than
$1$ for $k<n$ and less than $1$ for $k\geq n$.  The binomial theorem gives
\begin{equation}\label{eq:cheb-binomial-size}
 \sum_{k=0}^{2n}\binom{2n}{k}=2^{2n},
 \qquad
 \frac{2^{2n}}{2n+1}\leq\binom{2n}{n}\leq2^{2n}.
\end{equation}

\medskip
\noindent\textbf{Step 2. Linear bounds for $\psi$.}
For real $t\geq2$, set $n=\lfloor t/2\rfloor\in\mathbb N$.
Since $t-2\leq2n\leq t$, the monotonicity of $\psi$ and
\eqref{eq:cheb-binomial-psi}--\eqref{eq:cheb-binomial-size} imply
\begin{equation}\label{eq:cheb-psi-lower}
 \psi(t)\geq\psi(2n)
 \geq2n\log2-\log(2n+1)
 \geq(t-2)\log2-\log(t+1).
\end{equation}
For the upper bound, let $u\geq1$ be real and put
$J=\lfloor\log u/\log2\rfloor\in\mathbb Z_{\geq0}$.
Then $2^J\leq u<2^{J+1}$.  Apply
\eqref{eq:cheb-binomial-psi}--\eqref{eq:cheb-binomial-size}
with $n=2^j$ for each $j\in\{0,\ldots,J\}$.
Using $\psi(1)=0$, we obtain
\begin{align}
 \psi(u)
 \leq\psi(2^{J+1})
 =\sum_{j=0}^{J}\bigl(\psi(2^{j+1})-\psi(2^j)\bigr)
\leq(\log2)\sum_{j=0}^{J}2^{j+1}
 =(2^{J+2}-2)\log2
 \leq 4u\log2.
 \label{eq:cheb-psi-upper}
\end{align}
In particular, $\vartheta(u)\leq\psi(u)\leq bu$ for every real $u\geq1$.

\medskip
\noindent\textbf{Step 3. Removing higher prime powers.}
For real $t\geq2$, let
$K=\lfloor\log t/\log2\rfloor\in\mathbb N$.
No prime power $p^j\leq t$ can have $j>K$, since $p\geq2$.
Grouping the terms of $\psi(t)$ by their exponent, and applying
\eqref{eq:cheb-psi-upper}, gives
\begin{align*}
 0\leq\psi(t)-\vartheta(t)
 =\sum_{j=2}^{K}\vartheta(t^{1/j})
 \leq b\sum_{j=2}^{K}t^{1/j}
 \leq b(K-1)\sqrt t
 \leq4\sqrt t\log t.
\end{align*}
Here $t^{1/j}\leq\sqrt t$ for every integer $j\geq2$, and
$K\leq\log t/\log2$; the sums are empty when $K=1$.
Together with \eqref{eq:cheb-psi-lower}, this yields
\begin{equation}\label{eq:cheb-theta-lower}
 \vartheta(t)\geq t\log2-E(t),\qquad
 E(t)=2\log2+\log(t+1)+4\sqrt t\log t.
\end{equation}
Since $E(t)/t\to0$ as $t\to\infty$, there is a real $t_1\geq2$
such that $E(t)\leq at$ for all real $t\geq t_1$.
As $\log2=2a$, \eqref{eq:cheb-psi-upper} and
\eqref{eq:cheb-theta-lower} give
\begin{equation*}
 at\leq\vartheta(t)\leq bt\qquad(t\geq t_1).
\end{equation*}

\medskip
\noindent\textbf{Step 4. Bounds for the prime-counting function.}
For real $t\geq t_1$, each prime $p\leq t$ satisfies
$\log p\leq\log t$, so
\begin{equation*}
 \pi(t)\geq\frac{\vartheta(t)}{\log t}
 \geq\frac{at}{\log t}.
\end{equation*}
For the upper bound, there are at most $\sqrt t$ primes at most
$\sqrt t$, and every prime $p>\sqrt t$ satisfies
$\log p>\tfrac12\log t$.  Consequently,
\begin{align*}
 \pi(t)
 =\pi(\sqrt t)+
   \sum_{\substack{p\in\mathcal P\\\sqrt t<p\leq t}}1
 \leq\sqrt t+
   \frac{2}{\log t}
   \sum_{\substack{p\in\mathcal P\\\sqrt t<p\leq t}}\log p
 \leq\left(8\log2+\frac{\log t}{\sqrt t}\right)
       \frac{t}{\log t}.
\end{align*}
Because $\log2<1$ and $(\log t)/\sqrt t\to0$, we may choose a real
$t_0\geq t_1$ such that $$(\log t)/\sqrt t\leq8(1-\log2)$$ for
every real $t\geq t_0$.  The last display then gives
$\pi(t)\leq8t/\log t$, completing the proof.

\label{lem:explicit}
Define $h:(0,\infty)\to\mathbb R$ and
$W:\{s\in\mathbb C:\Re s>-35\}\to\mathbb C$ by
\[
 h(t)=t^{35}\mathrm e^{-24t}(t-1)(2-t),
 \qquad
 W(s)=\int_0^\infty h(t)t^{s-1}\,dt.
\]
                                                                                                              
For a primitive Dirichlet character \(\chi:\mathbb Z\to\mathbb C\) of conductor \(Q\in\mathbb N\), define
\[
 T_x(\chi)=\sum_{p\in\mathcal P}(\log p)\chi(p)h(p/x),
 \qquad x\in\mathbb R,\quad x\geq2.
\]
Let $\mathcal Z_\chi$ be the multiset consisting of all nontrivial zeros
$\rho\in\mathbb C$ of $L(s,\chi)$ with $0<\Re\rho<1$, together with
the zero $\rho=0$ if $L(0,\chi)=0$.  Each zero is repeated according
to its order of vanishing. Then
\begin{equation}\label{eq:explicit}
 T_x(\chi)=\1_{\chi=\chi_1}\,xW(1)
 -\sum_{\rho\in\mathcal Z_\chi}W(\rho)x^\rho
 +O\!\left(\sqrt{x}\log^2(x+2)+x^{-1/2}\log(Q+2)\right).
\end{equation}
The sum over zeros is absolutely convergent.  Moreover, there is a real constant $C_h>0$, depending only on $h$, such that
\begin{equation}\label{eq:Wdecay}
 W(1)>0,
 \qquad
 |W(\sigma+i\tau)|\leq C_h\mathrm e^{-|\tau|}
 \quad\left(\sigma,\tau\in\mathbb R,\ -\frac12\leq\sigma\leq2\right).
\end{equation}

<!-- proof-2: body_md -->
We first evaluate the Mellin transform, then apply the residue theorem
to the weighted sum over prime powers, and finally remove the higher
prime powers.  All implied constants below are absolute, since $h$
is fixed.

\medskip\noindent
\textbf{Step 1. The Mellin transform.}
The function $h$ is positive on $(1,2)$ and nonpositive on
$(0,1]\cup[2,\infty)$.  It satisfies $h(t)=O(t^{35})$ as $t\downarrow0$
and decays exponentially as $t\to\infty$.  Hence the integral defining
$W(s)$, and its derivatives with respect to $s$, converge locally
uniformly for $\Re s>-35$.  Thus $W$ is holomorphic there.  Expanding
$(t-1)(2-t)=-t^2+3t-2$ and integrating each term gives
\begin{equation*}
\begin{aligned}
 W(s)&=-24^{-s-37}\Gamma(s+37)
       +3\,24^{-s-36}\Gamma(s+36)
       -2\,24^{-s-35}\Gamma(s+35)\\
 &=24^{-s-37}\Gamma(s+35)(108+s-s^2).
\end{aligned}
\end{equation*}
Here $\Gamma$ is Euler's gamma function.  In particular,
\begin{equation*}
 W(1)=108\,24^{-38}\Gamma(36)
     >0.
\end{equation*}
The vertical-strip form of Stirling's formula
\cite[pp.~87--88 and~293--295]{Olver}, applied to $s+35$, gives
\begin{equation}\label{eq:W-stirling}
 |W(\sigma+i\tau)|
 \ll(1+|\tau|)^{77/2}\mathrm e^{-\pi|\tau|/2}
 \ll\mathrm e^{-|\tau|}
 \qquad\left(-\tfrac12\leq\sigma\leq2,\ \tau\in\mathbb R\right).
\end{equation}
The estimate is uniform because $\Re(s+35)\in[34.5,37]$;
bounded $\tau$ are covered by continuity.  This proves
\eqref{eq:Wdecay}.

\medskip\noindent
\textbf{Step 2. A Mellin integral for the prime powers.}
For $n\in\mathbb N$, let $\Lambda(n)=\log p$ if $n=p^k$ for a prime
$p\in\mathcal P$ and an integer $k\geq1$, and let $\Lambda(n)=0$
otherwise.  Put
\begin{equation}\label{eq:weighted-von-mangoldt}
 S_x(\chi)=\sum_{n=1}^{\infty}\Lambda(n)\chi(n)h(n/x).
\end{equation}
This series is absolutely convergent.  Write $\displaystyle \int_{(c)}$ for upward
integration along $s=c+i\tau$, $\tau\in\mathbb R$, with $c\in\mathbb R$.
Mellin inversion \cite[Chapter~4, Appendix; Equation~(4.106)]{IwaniecKowalski}
gives
\begin{equation*}
 h(t)=\frac1{2\pi i}\int_{(2)}W(s)t^{-s}\,ds
 \qquad(t\in\mathbb R,\ t>0).
\end{equation*}
Its hypotheses hold because $h$ is continuous, of bounded variation,
and of Mellin type $(-35,\infty)$: the displayed asymptotics give
$h(t)=O(t^{35})$ at $0$ and $h(t)=O(t^{37}\mathrm e^{-24t})$ at infinity,
while $\displaystyle\int_0^\infty|h'(t)|\,dt<\infty$.  The inversion integral is
absolutely convergent by \eqref{eq:W-stirling}.  Substitution into
\eqref{eq:weighted-von-mangoldt} is justified by the finite majorant
\begin{equation*}
 x^2\sum_{n=1}^{\infty}\frac{\Lambda(n)}{n^2}
       \int_{-\infty}^{\infty}|W(2+i\tau)|\,d\tau<\infty.
\end{equation*}
Since $$-L'/L(s,\chi)=\sum_{n=1}^{\infty}\Lambda(n)\chi(n)n^{-s}$$
for $\Re s>1$, it follows that
\begin{equation}\label{eq:mellin}
 S_x(\chi)=\frac1{2\pi i}\int_{(2)}
       -\frac{L'}{L}(s,\chi)W(s)x^s\,ds.
\end{equation}
Here $L'$ denotes differentiation with respect to $s\in\mathbb C$.

\medskip\noindent
\textbf{Step 3. Bounds for the contour shift.}
Let $a_\chi\in\{0,1\}$ satisfy $\chi(-1)=(-1)^{a_\chi}$, and let
$\overline\chi(n)=\overline{\chi(n)}$ for $n\in\mathbb Z$.
The functional equation for a primitive character of conductor $Q$ is
\begin{equation*}
\begin{aligned}
 \Lambda_\chi(s)&=\left(\frac Q\pi\right)^{(s+a_\chi)/2}
       \Gamma\!\left(\frac{s+a_\chi}{2}\right)L(s,\chi),\\
 \Lambda_\chi(s)&=\varepsilon_\chi\Lambda_{\overline\chi}(1-s),
       \qquad\varepsilon_\chi\in\mathbb C,\quad|\varepsilon_\chi|=1;
\end{aligned}
\end{equation*}
see \cite[Theorem~4.15; Equations~(4.73)--(4.75)]{IwaniecKowalski}.
For $\chi=\chi_1$ and $Q=1$, this is the meromorphic functional
equation of $\zeta(s)$.  Thus it also applies in the principal case.
Set $\psi_\Gamma(z)=\Gamma'(z)/\Gamma(z)$ for
$z\in\mathbb C\setminus\{0,-1,-2,\ldots\}$.  Logarithmic differentiation
gives the meromorphic identity
\begin{equation*}
\begin{aligned}
 \frac{L'}{L}(s,\chi)
 ={}-\frac{L'}{L}(1-s,\overline\chi)-\log\frac Q\pi
 -\frac12\psi_\Gamma\!\left(\frac{s+a_\chi}{2}\right)
  -\frac12\psi_\Gamma\!\left(\frac{1-s+a_\chi}{2}\right).
\end{aligned}
\end{equation*}
On $s=-1/2+i\tau$, the reflected logarithmic derivative has an
absolutely convergent Dirichlet series on $\Re(1-s)=3/2$ and is
bounded by $$\sum_{n=1}^\infty\Lambda(n)n^{-3/2}<\infty.$$
The digamma estimate \cite[pp.~87--88 and~293--295]{Olver} bounds the other
terms by $O(\log(|\tau|+2))$.  For bounded $\tau$, their arguments
stay a fixed distance from the poles of $\psi_\Gamma$.
Consequently,
\begin{equation*}
 \left|\frac{L'}{L}\left(-\frac12+i\tau,\chi\right)\right|
 \ll\log\bigl(Q(|\tau|+2)\bigr)\qquad(\tau\in\mathbb R),
\end{equation*}
and the new vertical integral satisfies
\begin{equation*}
 \left|\frac1{2\pi i}\int_{(-\frac{1}{2})}
       -\frac{L'}{L}(s,\chi)W(s)x^s\,ds\right|
 \ll x^{-\frac{1}{2}}\int_{-\infty}^{\infty}
       \mathrm e^{-|\tau|}\log\bigl(Q(|\tau|+2)\bigr)\,d\tau
 \ll x^{-\frac{1}{2}}\log(Q+2).
\end{equation*}

We also justify the disappearance of the horizontal sides.
Let $\mathcal Z_\chi^\circ$ denote the multiset of nontrivial zeros
$\rho$ with $0<\Re\rho<1$.  The standard local zero estimates are
\begin{align}
 \#\{\rho\in\mathcal Z_\chi^\circ:|\Im\rho-\tau|\leq1\}
 &\ll\log\bigl(Q(|\tau|+2)\bigr),
 \label{eq:local-zero-count}\\
 \frac{L'}{L}(s,\chi)
 &=\sum_{\substack{\rho\in\mathcal Z_\chi^\circ\\|s-\rho|<1}}
       \frac1{s-\rho}
       +O\!\left(\log\bigl(Q(|\tau|+2)\bigr)\right),
 \label{eq:local-log-derivative}
\end{align}
where $s=\sigma+i\tau$, $-1/2\leq\sigma\leq2$, $|\tau|\geq2$,
and $s$ is not a zero in \eqref{eq:local-log-derivative}.
Zeros are counted with multiplicity.  These are
\cite[Proposition~5.7; Equations~(5.27)--(5.28)]{IwaniecKowalski}:
primitive Dirichlet $L$-functions have degree one and analytic
conductor comparable to $Q(|\tau|+3)$, as verified in
\cite[Section~5.9]{IwaniecKowalski}.  Thus the constants are uniform
in $Q$ and $\chi$.

For each integer $j\geq3$, let $M_j$ count the zeros in
$\mathcal Z_\chi^\circ$ with $j-1\leq|\Im\rho|\leq j+2$.
Covering these two intervals by finitely many intervals of length
$2$ and using \eqref{eq:local-zero-count} gives
$M_j\ll\log(Q(j+2))$.  Set $\delta_j=1/(4(M_j+1))$.
Intervals of radius $\delta_j$ about the $M_j$ absolute ordinates
remove length at most $2M_j\delta_j<1/2$ from $[j,j+1]$.
Choose $T_j$ outside them.  Then both $T_j$ and $-T_j$ have distance
at least $\delta_j$ from every zero ordinate; ordinates outside the
specified intervals have distance at least $1$. 
\eqref{eq:local-zero-count}--\eqref{eq:local-log-derivative} yield
\begin{equation*}
 \sup_{-1/2\leq\sigma\leq2}
 \left|\frac{L'}{L}(\sigma\pm iT_j,\chi)\right|
 \ll\log^2(Q(j+2)).
\end{equation*}
Each horizontal integral in the rectangle with vertical sides
$\Re s=2,-1/2$ and heights $\pm T_j$ is therefore
\begin{equation*}
 O\!\left(x^2\mathrm e^{-j}\log^2(Q(j+2))\right)=o(1)
 \qquad(j\to\infty),
\end{equation*}
with $x$ and $Q$ fixed.

\medskip\noindent
\textbf{Step 4. The residues and the zero sum.}
The only pole of $L(s,\chi)$ is the simple pole at $s=1$ when
$\chi=\chi_1$.  Dirichlet $L$-functions have no zeros on
$\Re s=1$; for nonprincipal primitive characters this is
\cite[Theorem~5.26]{IwaniecKowalski}, while for $\chi_1$ it is the
classical zero-free line for $\zeta(s)$, recorded in
\cite[Equation~(4.17.2), p.~89]{Titchmarsh} (with its simple pole at $s=1$).
The functional equation then shows that their only
possible zero on $\Re s=0$ is at $s=0$.  For a nonprincipal even
primitive character this zero is simple: the gamma factor has a
simple pole at $0$, whereas $\Lambda_{\overline\chi}(1)\ne0$.
For an odd primitive character the gamma factor is finite and
nonzero at $0$, so $L(0,\chi)\ne0$.  In the principal case,
$\zeta(0)=-1/2$.  The remaining trivial zeros lie at negative
integers, outside the rectangle.  Thus the poles of the integrand
in \eqref{eq:mellin} inside it are exactly the principal pole at $1$
and the zeros in $\mathcal Z_\chi$ with $|\Im\rho|<T_j$.

At the principal pole the residue of $-L'/L$ is $+1$.
At a zero $\rho$ of order $m\in\mathbb N$, it is $-m$, since locally
$L(s,\chi)=(s-\rho)^m g(s)$ with $g$ holomorphic and $g(\rho)\ne0$.
Their contributions are therefore $xW(1)$ and
$-mW(\rho)x^\rho$, respectively, including the possible zero at $0$.
There are no poles contributed by $W$, which is holomorphic on
the whole rectangle.

To pass to the limit, write
$N_\chi(T)=\#\{\rho\in\mathcal Z_\chi^\circ:|\Im\rho|\leq T\}$,
with multiplicity, for real $T\geq2$.
The bound $N_\chi(T)\ll T\log(Q(T+2))$ is the Dirichlet case of
\cite[Theorem~5.24]{IwaniecKowalski}.  By \eqref{eq:Wdecay},
\begin{equation*}
 \sum_{\rho\in\mathcal Z_\chi^\circ}|W(\rho)x^\rho|
 \ll x\sum_{j=0}^{\infty}\mathrm e^{-j}N_\chi(j+2)
 \ll x\sum_{j=0}^{\infty}\mathrm e^{-j}(j+2)\log(Q(j+4))<\infty.
\end{equation*}
The possible term at $0$ is finite.  The residue theorem, followed
by $j\to\infty$, now gives
\begin{equation}\label{eq:vonmangoldt}
 S_x(\chi)=\1_{\chi=\chi_1}xW(1)
 -\sum_{\rho\in\mathcal Z_\chi}W(\rho)x^\rho
 +O\!\left(x^{-1/2}\log(Q+2)\right).
\end{equation}

\medskip\noindent
\textbf{Step 5. Removing the higher prime powers.}
For real $Y>0$, define
\begin{equation*}
 A(Y)=\sum_{\substack{p\in\mathcal P,\ k\in\mathbb N\\k\geq2,\ p^k\leq Y}}
             \log p.
\end{equation*}
This is zero for $Y<4$.  The estimate $\vartheta(u)\ll u$ for
all real $u>0$ follows from \Cref{lem:cheb}, after increasing
the constant on a bounded interval.  For $Y\geq4$ it gives
\begin{equation*}
 A(Y)=\sum_{k=2}^{\lfloor\log Y/\log2\rfloor}\vartheta(Y^{1/k})
 \ll Y^{1/2}+Y^{1/3}\log Y\ll Y^{1/2}.
\end{equation*}
Indeed, every term with $k\geq3$ is $O(Y^{1/3})$, and
$\log Y\leq(6/\mathrm e)Y^{1/6}$ for $Y\geq1$, by maximizing
$(\log Y)/Y^{1/6}$.  Since $|\chi(n)|\leq1$, partial summation yields
\begin{equation}\label{eq:prime-power-removal}
\begin{aligned}
 & |S_x(\chi)-T_x(\chi)|
 \leq\int_0^\infty|h(t/x)|\,dA(t)
 \leq\frac1x\int_0^\infty A(t)|h'(t/x)|\,dt \\
 & \hspace{6cm}
 \ll\sqrt x\int_0^\infty u^{1/2}|h'(u)|\,du
 \ll\sqrt x.
\end{aligned}
\end{equation}
Here $|h|$ is absolutely continuous with
$|(|h|)'|\leq|h'|$ almost everywhere.  The boundary terms vanish
by $A(t)=0$ near $0$ and the exponential decay at infinity.
The last integral is finite because $h'(u)=O(u^{34})$ near $0$
and $h'(u)=O(u^{37}\mathrm e^{-24u})$ as $u\to\infty$.
Combining \eqref{eq:vonmangoldt} and \eqref{eq:prime-power-removal}
proves \eqref{eq:explicit}.

For all sufficiently large real $y$, there is a set

Put
\begin{equation*}
 \Omega_y=\{s\in\mathbb C:\Re s\geq1-2\eta(y),\ |\Im s|\leq H\}.
\end{equation*}
By \Cref{thm:lau-wu}(a), the $L$-functions for $\chi\in\mathcal C_y$
have at most one zero in $\Omega_y$, counted with multiplicity.
If this zero exists, it is real.  The four characters in
$\mathcal C_y$ whose discriminants have no odd prime factor are
\[
 \chi_1,\qquad\chi_{-4},\qquad\chi_8,\qquad\chi_{-8}.
\]
For the three nonprincipal characters, $L(1,\chi)\ne0$ by \cite[Theorem~6.20, p.~141]{Apostol}; for the
principal character, $L(s,\chi_1)=\zeta(s)$ has a simple pole at $s=1$.
Consequently, for some fixed $\delta>0$, none of these four $L$-functions
has a zero in the disk $|s-1|<\delta$ (for $\chi_1$, use that
$(s-1)\zeta(s)$ is holomorphic and nonzero at $s=1$).  Since
$\eta(y)\to0$, we may assume
\[
 2\eta(y)<\min(\delta,1/2).
\]

If no $L(s,\chi)$ with $\chi\in\mathcal C_y$ has a zero in
$\Omega_y$, set $E_y=\varnothing$.
Then $P_y^*=P_y$ and $\mathcal D_y=\mathcal C_y$.
The conclusion follows because
$\{s\in\mathbb C:\Re s\geq1-\eta(y),\ |\Im s|\leq H\}\subseteq\Omega_y$.

Suppose now that some $L(s,\chi)$ with $\chi\in\mathcal C_y$ has a zero in $\Omega_y$.  By
\Cref{thm:lau-wu}(a), it is a unique real zero $\beta$ of a unique
character $\chi_*\in\mathcal C_y$, counted with multiplicity.  We have
$\beta\geq1-2\eta(y)>0$; the Euler product gives zero-freeness for
$\Re s>1$, and the nonvanishing at $s=1$ rules out $\beta=1$.  Hence
\[
 0<\beta<1,\qquad 1-\beta\leq2\eta(y)<\delta.
\]
Thus $|\beta-1|<\delta$, and the disk exclusion above shows that
$\chi_*$ is not one of the four characters with no odd prime factor in
its discriminant.

Write $\chi_*=\chi_{D_*}$. By the preceding paragraph, $|D_*|$
has an odd prime divisor; choose one and denote it by $r_y$.
Since $|D_*|\mid4P_y$ and $r_y>2$, we have $r_y\mid P_y$, hence
$r_y\leq y$. Set $E_y=\{r_y\}$ and $P_y^*=P_y/r_y$.
The integer $P_y$ is squarefree, so $r_y\nmid4P_y^*$.
Every $\chi_D\in\mathcal D_y$ satisfies $|D|\mid4P_y^*$ and therefore
$r_y\nmid|D|$. In contrast, $r_y\mid|D_*|$, so
$\chi_*\notin\mathcal D_y$.
The zero $\beta$ belonged only to $L(s,\chi_*)$, and no other
$L$-function in $\mathcal C_y$ has a zero in $\Omega_y$.
Since $\mathcal D_y\subseteq\mathcal C_y\setminus\{\chi_*\}$, every
$L(s,\chi)$ with $\chi\in\mathcal D_y$ is zero-free in $\Omega_y$.
In particular, it has no zero with $\Re s\geq1-\eta(y)$ and
$|\Im s|\leq H$, as required.  Finally, $P_y^*\in\mathbb N$ because $r_y$ is a prime factor of
$P_y$.

\label{lem:residue}
Let $q\equiv5\pmod8$ be prime, let $K\in\mathbb N$ satisfy $1\leq K<q$,
and let $E$ be empty or a singleton containing an odd prime $r\leq K$.
Suppose
\begin{equation}\label{eq:residue-hypothesis}

 \qquad(\ell\leq K,\ \ell\text{ an odd prime},\ \ell\notin E).
\end{equation}

\begin{equation}\label{eq:Ssize}
 |S|\geq\frac K6-1
\end{equation}
and
\begin{equation}\label{eq:Sproperty}
 \left(\frac{s^2+t^2}{q}\right)=-1\qquad(s,t\in S).
\end{equation}
Every multiplicative translate $cS$, $c\in\mathbb F_q^\times$, has the same

<!-- proof-4: body_md -->
We divide the proof into three steps.

\medskip\noindent
\textbf{Step 1. Selecting the integers.}
Put
\begin{equation}\label{eq:U-initial}
 U_0=\{u\in\mathbb N:1\leq u\leq K,\ u\equiv1\pmod4\},
 \qquad |U_0|=\left\lfloor\frac{K+3}{4}\right\rfloor\geq\frac K4.
\end{equation}
If $E=\varnothing$, take $U=U_0$.  Suppose $E=\{r\}$.  Since
$\gcd(4,r)=1$, the simultaneous conditions $u\equiv1\pmod4$ and
$u\equiv0\pmod r$ specify one residue class modulo $4r$.  Thus the
number of such integers in $[1,K]$ is at most $K/(4r)+1$.  Set
$V=\{u\in U_0:r\nmid u\}$; then
\begin{equation*}
 |V|\geq\frac K4-\frac K{4r}-1.
\end{equation*}

The nonzero residue classes modulo $r$ split into disjoint pairs
$\{a,-a\}$.  Indeed, $a=-a$ would give $2a=0$, hence $a=0$,
because $r$ is odd.  For each pair \(\{a,-a\}\) of nonzero residue classes modulo \(r\), choose the class represented by more elements of \(V\). If the two classes are represented equally often, choose either one.
Let $A\subset(\mathbb Z/r\mathbb Z)^\times$ be the retained classes and put
$U=\{u\in V:u\bmod r\in A\}$.  The construction gives
\begin{equation}\label{eq:U-no-opposites}
 A\cap(-A)=\varnothing,
 \qquad r\nmid u,\quad r\nmid(u+v)\qquad(u,v\in U).
\end{equation}
At least half of $V$ is retained.  Since
$r\geq3$, we obtain
\begin{equation*}
 |U|\geq\frac{|V|}{2}
 \geq\frac K8\left(1-\frac1r\right)-\frac12
 \geq\frac K{12}-\frac12.
\end{equation*}
This last bound also holds when $E=\varnothing$, by
\eqref{eq:U-initial}.

\medskip\noindent
\textbf{Step 2. The quadratic characters of the sums.}
Every $u\in U$ is a positive odd integer at most $K$, and none of its
prime factors belongs to $E$.  By \eqref{eq:residue-hypothesis} and
multiplicativity,
\begin{equation}\label{eq:U-residues}
 \left(\frac{u}{q}\right)
 =\prod_{\ell\mid u\atop \ell\in\mathcal P}\left(\frac{\ell}{q}\right)^{v_\ell(u)}=1.
\end{equation}
For $u,v\in U$, define
\begin{equation*}
 w=\frac{u+v}{2},\qquad 1\leq w\leq K,\qquad w\equiv1\pmod2.
\end{equation*}
Because \(K<q\) and $q\equiv5\pmod8$, we have \(q\nmid w\) and \(q\nmid u+v=2w\).  If $E=\{r\}$,
then $r\nmid w$, since $r\mid w$ would contradict
\eqref{eq:U-no-opposites}.  Thus all prime factors of $w$ also satisfy
\eqref{eq:residue-hypothesis}.  Consequently $\displaystyle \left(\frac{w}{q}\right)=1$.
The supplementary law $$\left(\frac{2}{q}\right)=(-1)^{(q^2-1)/8}=-1,$$ valid
because $q\equiv5\pmod8$, now gives
\begin{equation}\label{eq:U-sum-nonresidue}
 \left(\frac{u+v}{q}\right)=\left(\frac{2}{q}\right)\left(\frac{w}{q}\right)=-1
 \qquad(u,v\in U).
\end{equation}
In particular, the sum is nonzero modulo $q$: $w<q$ and $q$ is odd.

\medskip\noindent
\textbf{Step 3. Taking square roots and multiplicative translates.}
Regard $U\subset[1,K]$ as a subset of $\mathbb F_q^\times$; this is an
injective identification since $K<q$.  Define
\begin{equation*}
 S=\{s\in\mathbb F_q^\times:s^2\in U\}.
\end{equation*}
By \eqref{eq:U-residues}, each $u\in U$ has exactly two square roots
in $\mathbb F_q^\times$.  They are distinct because $q$ is odd, and roots
of different $u$ are disjoint.  Therefore
\begin{equation*}
 |S|=2|U|\geq\frac K6-1.
\end{equation*}
For $s,t\in S$, apply \eqref{eq:U-sum-nonresidue} to $u=s^2$ and
$v=t^2$ to obtain \eqref{eq:Sproperty}.  This includes $s=t$ and
$s=-t$, since the preceding argument allowed $u=v$.  Finally,
for $c\in\mathbb F_q^\times$,
\begin{equation}\label{eq:S-translate}
 \left(\frac{(cs)^2+(ct)^2}{q}\right)
 =\left(\frac{c^2}{q}\right)\left(\frac{s^2+t^2}{q}\right)=-1
 \qquad(s,t\in S),
\end{equation}
because $c^2$ is a nonzero square.  This proves the assertion for $cS$.

We divide the proof into six steps and use the notation introduced above.

\medskip\noindent
\textbf{Step 1. The scales and the size of the character family.}
Let $x$ be a sufficiently large real number and put
\begin{equation*}
 L=\log x,\qquad J=\log L,\qquad T=\log J,
 \qquad y=\kappa LT,\qquad\varepsilon=J^{-1/2}.
\end{equation*}
We take $x$ large enough that $y$ exceeds the thresholds in
\Cref{lem:cheb,thm:lau-wu,lem:deletion} and $0<\varepsilon<1/2$.
Use the notation $P_y$, $H=\log P_y$, and $\mathcal C_y$ from
\eqref{eq:PyH}--\eqref{eq:Cy}, and choose $E_y$, $P_y^*$, and
$\mathcal D_y$ by \Cref{lem:deletion}.  Then
\begin{equation}\label{eq:scale-bounds}
 ay\leq H\leq by,\qquad
 \log H=J+\log T+O(1)\sim J.
\end{equation}
Indeed, $H\asymp y$ and $\log y=J+\log T+\log\kappa$.
By \eqref{eq:character-family-size} and the upper bound for $\pi(y)$
in \Cref{lem:cheb},
\begin{equation}\label{eq:family-log-size}
 R_y=|\mathcal C_y|=2^{\pi(y)+1},\qquad
 \log R_y=O\!\left(\frac{LT}{J}\right)=o(\varepsilon L).
\end{equation}
The last relation follows from $T/\sqrt J\to0$.  In particular,
$R_y=\exp(o(L))$, $H=O(LT)$, and $\log H=O(J)=o(L)$, so
\begin{equation}\label{eq:family-small}
 R_y\left(\sqrt{x}\log^2(x+2)+x^{-1/2}H\right)=o(x).
\end{equation}

\medskip\noindent
\textbf{Step 2. Zeros close to the line $\Re s=1$.}
For $0\leq t<1/2$, let $M(t)$ count, with multiplicity and separately
for every $\chi\in\mathcal D_y$, the zeros
$\rho=\beta+i\gamma\in\mathcal Z_\chi$ satisfying
$\beta\geq1-t$ and $|\gamma|\leq H$.  Put
\begin{equation*}
 \eta=\frac{c_1\sqrt{\log H}}{2H},\qquad
 M(t)=0\quad(0\leq t\leq\eta).
\end{equation*}
The equality is the conclusion of \Cref{lem:deletion}.  Since
$\eta/\varepsilon\asymp J/(LT)\to0$, we also have $0<\eta<\varepsilon$
for large $x$.  The four discriminant types listed in \Cref{sec:quadratic-characters}
partition $\mathcal C_y$ between the two products in
\Cref{thm:lau-wu}(b).  Thus
\begin{equation}\label{eq:M-comparison}
 M(t)\leq N_1(1-t;y)+N_4(1-t;y).
\end{equation}
Here $1-t>1/2$, so the pole at $1$ and the possible zero at $0$
do not affect the zero counts.

We apply the two bounds for $N_\ell(1-t;y)$ in \Cref{thm:lau-wu}(b).
By \eqref{eq:scale-bounds}, $T/\sqrt J\to0$, and
$\log(1/\varepsilon)=T/2$,
\begin{equation}\label{eq:density-coeff}
 \frac{c_2H}{\sqrt{\log H}}\leq\frac L4,
 \qquad
 \frac{c_2H}{\log(1/\varepsilon)}
 \leq2c_2b\kappa L
 =\frac{c_2b}{16(1+c_2b)}L\leq\frac L4
\end{equation}
for all sufficiently large $x$.  If $0<t\leq\varepsilon$, then
$\log(1/t)\geq\log(1/\varepsilon)$. For $t\leq\eta_1(y)$,
the theorem gives $N_\ell(1-t;y)\leq C\sqrt{\log H}\exp(tL/4)$.
For $t>\eta_1(y)$, it gives $N_\ell(1-t;y)\leq C\exp(tL/4)$.
Adding the bounds for $\ell=1,4$ in \eqref{eq:M-comparison} and
enlarging $C$, we obtain
\begin{equation*}
 M(t)\leq C\sqrt{\log H}\,\mathrm e^{tL/4}
 \qquad(0<t\leq\varepsilon).
\end{equation*}

For each zero counted by $M(\varepsilon)$, write $t_\rho=1-\beta$.
Then $0\leq t_\rho\leq\varepsilon$ and
\begin{equation}\label{eq:zero-weight-identity}
 \mathrm e^{-Lt_\rho}=\mathrm e^{-L\varepsilon}
       +L\int_{t_\rho}^{\varepsilon}\mathrm e^{-Lt}\,dt.
\end{equation}
The number of these zeros is finite by \Cref{thm:lau-wu}(c).
Summing \eqref{eq:zero-weight-identity}, with multiplicities, gives
\begin{align}
 &\sum_{\chi\in\mathcal D_y}
  \sum_{\substack{\rho=\beta+i\gamma\in\mathcal Z_\chi\\
                  \beta\geq1-\varepsilon,\ |\gamma|\leq H}}
 x^\beta \notag\\
 &= x^{1-\varepsilon}M(\varepsilon)
       +xL\int_0^\varepsilon\mathrm e^{-Lt}M(t)\,dt \notag\\
 &\leq Cx\sqrt{\log H}\mathrm e^{-3\varepsilon L/4}
       +CxL\sqrt{\log H}\int_\eta^\varepsilon
                  \mathrm e^{-3Lt/4}\,dt \notag\\
 &\leq Cx\sqrt{\log H}\mathrm e^{-3\varepsilon L/4}
       +\frac{4C}{3}x\sqrt{\log H}\mathrm e^{-3\eta L/4}
       =o(x).
 \label{eq:near-zero}
\end{align}
To justify the last equality, use $\log H\sim J$ and
$\log\log H\sim T$. Since $H\leq b\kappa LT$, we have
\begin{equation*}
 \eta L=\frac{c_1L\sqrt{\log H}}{2H}
 \geq\frac{c_1}{2b\kappa}\frac{\sqrt{\log H}}{T}
 \gg\frac{\sqrt J}{T}.
\end{equation*}
Consequently,
\begin{equation*}
 \log\log H\sim T,\qquad
 \frac{\varepsilon L}{\log\log H}\asymp\frac{L}{\sqrt J\,T}
       \longrightarrow\infty,\qquad
 \frac{\eta L}{\log\log H}
       \gg\frac{\sqrt J}{T^2}\longrightarrow\infty.
\end{equation*}
Thus both exponential factors in \eqref{eq:near-zero} tend to
zero faster than $1/\sqrt{\log H}$.  Since $|W(\rho)|\leq C_h$ in the critical
strip by \eqref{eq:Wdecay}, the same $o(x)$ bound holds with
$x^\beta$ replaced by $|W(\rho)x^\rho|$.

\medskip\noindent
\textbf{Step 3. The remaining zeros and the aggregate estimate.}
For $\beta<1-\varepsilon$ and $|\gamma|\leq H$, we have
$|W(\rho)x^\rho|\leq C_hx^{1-\varepsilon}$.  Each character has
conductor $Q\leq4P_y$, so \Cref{thm:lau-wu}(c), including its assertion
at $0$, gives
\begin{align}
 &\sum_{\chi\in\mathcal D_y}
  \sum_{\substack{\rho=\beta+i\gamma\in\mathcal Z_\chi\\
                  \beta<1-\varepsilon,\ |\gamma|\leq H}}
       |W(\rho)x^\rho| \notag\\
 &\qquad\ll x^{1-\varepsilon}R_yH\log\bigl(4P_y(H+2)\bigr)
       \ll x^{1-\varepsilon}R_yH^2=o(x).
 \label{eq:low-zeros}
\end{align}
Indeed, $\log(4P_y(H+2))=H+O(\log H)$, and the logarithm of
the last bound divided by $x$ is
\begin{equation*}
 -\varepsilon L+\log R_y+2\log H
       =-\varepsilon L+o(\varepsilon L)\longrightarrow-\infty.
\end{equation*}
Here \eqref{eq:family-log-size} applies and
$\log H=O(J)=o(L/\sqrt J)$.

For the zeros with $|\gamma|>H$, let $m_0=\lfloor H\rfloor$ and
partition the absolute ordinates into intervals $[m,m+1)$,
$m\geq m_0$.  The number of zeros of any one character in such
an interval is at most its total number with $|\gamma|\leq m+1$,
which is $O((m+1)\log(Q(m+3)))$ by \Cref{thm:lau-wu}(c).
Also $x^\beta\leq x$, $|W(\rho)|\leq C_h\mathrm e^{-m}$, and
$\log Q\leq H+\log4$.  Hence
\begin{align}
 &\sum_{\chi\in\mathcal D_y}
  \sum_{\substack{\rho\in\mathcal Z_\chi\\|\Im\rho|>H}}
       |W(\rho)x^\rho| \notag\\
 &\qquad\ll xR_y\sum_{m=m_0}^{\infty}
       \mathrm e^{-m}(m+1)\bigl(H+\log(m+3)+1\bigr) \notag\\
 &\qquad\ll xR_y\mathrm e^{-H}(H+1)^2=o(x).
 \label{eq:high-zeros}
\end{align}
For the second inequality, set $m=m_0+j$ and bound the summand by
$O(\mathrm e^{-H-j}(H+1)^2(j+1)^2)$; the sum over $j\geq0$ converges.
For the last equality, \eqref{eq:scale-bounds} and
\eqref{eq:family-log-size} give
$$\log R_y+2\log(H+1)=o(H).$$

Apply \Cref{lem:explicit} to every $\chi\in\mathcal D_y$.
Its error terms sum to $o(x)$ by \eqref{eq:family-small}, since
$|\mathcal D_y|\leq R_y$ and $\log(Q+2)=O(H)$.
The three ranges in \eqref{eq:near-zero}, \eqref{eq:low-zeros}, and
\eqref{eq:high-zeros} exhaust $\mathcal Z_\chi$.  The triangle inequality
therefore yields
\begin{equation}\label{eq:aggregate}
 \sum_{\chi\in\mathcal D_y}
 \left|T_x(\chi)-\1_{\chi=\chi_1}xW(1)\right|=o(x).
\end{equation}
The bounds above use $|\mathcal D_y|\leq R_y$ and
$M(t)\leq N_1(1-t;y)+N_4(1-t;y)$. Their right-hand sides do not
depend on which prime is omitted. Hence the $o(x)$ error in
\eqref{eq:aggregate} is uniform over the choices of $E_y$ in
\Cref{lem:deletion}.

\medskip\noindent
\textbf{Step 4. A prime with the required residue pattern.}
For every integer $n$, define
\begin{equation}\label{eq:F-definition}
 F_y(n)=(1+\chi_{-4}(n))(1-\chi_8(n))
       \prod_{\substack{r\in\mathcal P,\ r\leq y\\r>2,\ r\notin E_y}}
             (1+\chi_{r^*}(n)).
\end{equation}
Each character value belongs to $\{-1,0,1\}$, so $F_y(n)\geq0$.
The integer $P_y^*/2$ is the product of the odd primes
$r\leq y$ with $r\notin E_y$, each taken once. Thus the products
formed by choosing any of these primes are precisely the divisors
$d\mid P_y^*/2$; the empty choice gives $d=1$.
Expanding \eqref{eq:F-definition} and using
\eqref{eq:quadratic-products} gives
\begin{equation*}
 F_y(n)=\sum_{\substack{d\in\mathbb N\\d\mid P_y^*/2}}
 \bigl(\chi_{d^*}(n)+\chi_{-4d^*}(n)
       -\chi_{8d^*}(n)-\chi_{-8d^*}(n)\bigr).
\end{equation*}
The discriminant classification in \Cref{sec:quadratic-characters} shows that these are
exactly the characters of $\mathcal D_y$, each occurring once.
Thus $$F_y=\sum_{\chi\in\mathcal D_y}c_\chi\chi,$$ where
$c_\chi\in\{-1,1\}$ and $c_{\chi_1}=1$.
For fixed $x$, the formula for $h$ gives
\begin{equation*}
 \sum_{p\in\mathcal P}(\log p)|h(p/x)|
 \ll_x\sum_{n=2}^{\infty}n^{37}(\log n)\mathrm e^{-24n/x}<\infty.
\end{equation*}
Since there are finitely many characters and $|\chi(p)|\leq1$,
we may sum the character expansion over primes. By \eqref{eq:aggregate},
\begin{align}
 \sum_{p\in\mathcal P}(\log p)h(p/x)F_y(p)
 =\sum_{\chi\in\mathcal D_y}c_\chi T_x(\chi)
 =xW(1)+o(x)>0,
 \label{eq:positive}
\end{align}
since $W(1)>0$ by \eqref{eq:Wdecay}.

For $p\notin(x,2x)$, the corresponding summand in
\eqref{eq:positive} is nonpositive, because $h(p/x)\leq0$ and
$F_y(p)\geq0$.  There must therefore be a prime $q\in(x,2x)$ with
$F_y(q)>0$.  Since $y=\kappa\log x\log_3x=o(x)$, we have $q>y$
for large $x$.  The prime $q$ consequently divides none of the
conductors in \eqref{eq:F-definition}; all character values there
are $\pm1$.  Positivity of the product forces
\begin{equation*}
 \chi_{-4}(q)=1,\qquad\chi_8(q)=-1,\qquad
 \chi_{r^*}(q)=1\quad(r\leq y,\ r>2,\ r\notin E_y).
\end{equation*}
The first identity gives $q\equiv1\pmod4$.  The supplementary law
at $2$ gives $$\chi_8(q)=\left(\frac{2}{q}\right)=-1$$ only for
$q\equiv3,5\pmod8$; hence $q\equiv5\pmod8$.
Also $r^*=\pm r$, and $\displaystyle \left(\frac{-1}{q}\right)=1$ because $q\equiv1\pmod4$.
We conclude that
\begin{equation}\label{eq:qpattern}
 \begin{gathered}
 x<q<2x,\qquad q\equiv5\pmod8,\\
 \left(\frac{r}{q}\right)=1\quad(r\leq y,\ r\text{ an odd prime},\ r\notin E_y).
 \end{gathered}
\end{equation}

\medskip\noindent
\textbf{Step 5. An admissible subset of the prime-power box.}
Now let $N$ be sufficiently large, take $x=p_N$, and set
$K=\lfloor y\rfloor$.  Then $1\leq K<q$.  Every prime in $E_y$
is at most $y$ and is an integer, so it is at most $K$.
Thus \eqref{eq:qpattern} verifies every hypothesis of
\Cref{lem:residue}, with $E=E_y$.  It supplies $S\subset\mathbb F_q^\times$
with $|S|\geq K/6-1$ and \eqref{eq:Sproperty}.
Every $n\in\Phi_N$ is nonzero modulo $q$, since its prime factors
are at most $p_N<q$.  For each $c\in\mathbb F_q^\times$, put
\begin{equation*}
 B_c=\{n\in\Phi_N:n\bmod q\in cS\}.
\end{equation*}
If $u,v\in B_c$ and $w\in\mathbb N$ satisfied $u^2+v^2=w^2$, choose
$s,t\in S$ with $u\equiv cs$ and $v\equiv ct\pmod q$.
Then
\begin{equation*}
 w^2\equiv c^2(s^2+t^2)\pmod q,\qquad
 \left(\frac{c^2(s^2+t^2)}{q}\right)=-1.
\end{equation*}
The second equality follows from \eqref{eq:S-translate}.
It contradicts the first, since a square has Legendre symbol $0$ or
$1$.  Thus every $B_c$ is admissible.

For any fixed $n\in\Phi_N$ and $s\in S$, there is a unique
$c\in\mathbb F_q^\times$ with $n\equiv cs\pmod q$, namely
$c=(n\bmod q)s^{-1}$.  Distinct $s$ give distinct $c$, since
$n\bmod q\ne0$.  Therefore exactly $|S|$ translates contain
$n\bmod q$, and double counting yields
\begin{equation}\label{eq:translate-average}
 \sum_{c\in\mathbb F_q^\times}|B_c|=|\Phi_N||S|=N^N|S|,
 \qquad f(N)\geq\max_c|B_c|\geq\frac{N^N|S|}{q-1}.
\end{equation}

\medskip\noindent
\textbf{Step 6. The quantitative lower bound.}
Increase $N$ so that $K\geq6$.  Since $q<2p_N$,
$K\geq y-1$, and $N=\pi(p_N)\geq ap_N/\log p_N$ by
\Cref{lem:cheb}, \eqref{eq:translate-average} gives
\begin{align}
 \frac{f(N)}{N^{N-1}}
 &\geq\frac{N}{2p_N}\left(\frac K6-1\right)
  \geq\frac{a}{2\log p_N}\left(\frac K6-1\right) \notag\\
 &\geq\frac{a}{2\log p_N}
       \left(\frac{\kappa\log p_N\log_3p_N}{6}-\frac76\right)
  =\frac{a\kappa}{12}\log_3p_N-\frac{7a}{12\log p_N}.
 \label{eq:final-intermediate}
\end{align}
The use of the lower bound for $N$ is valid because $K/6-1\geq0$.
Since $p_N\geq N$, monotonicity of the iterated logarithm gives
$\log_3p_N\geq\log_3N$ for sufficiently large $N$.
The error term in \eqref{eq:final-intermediate} tends to zero,
whereas $\log_3N\to\infty$.  We can therefore choose an absolute
$N_0$ satisfying all earlier thresholds and
\begin{equation*}
 \frac{7a}{12\log p_N}\leq\frac{a\kappa}{24}\log_3N
 \qquad(N\geq N_0).
\end{equation*}
Substitution into \eqref{eq:final-intermediate} proves
\eqref{eq:main} for every $N\geq N_0$.

\label{rem:log-two-obstruction}
The factor $\log_3N$ comes from the choice
$y=\kappa\log x\log_3x$. Indeed, with $x=p_N$, the averaging
argument and the bound $|S|\gg y$ give
\begin{equation*}
 f(N)\gg N^{N-1}\frac{y}{\log x},
\end{equation*}
provided that a prime $q\in(x,2x)$ with the required residue
pattern is available. Chebyshev's estimates give
$\log_2p_N\sim\log_2N$. To obtain a factor of order $\log_2N$
by this construction, one would therefore seek that pattern at
the larger scale
\begin{equation*}
 y=\delta LJ,\qquad
 L=\log x,\quad J=\log L,\quad \delta>0\text{ fixed}.
\end{equation*}
The estimates used above do not establish it, for the following reasons.

At this scale, the bounds for $\vartheta(y)$ give
$H\asymp\delta LJ$ and $\log H\sim J$. Consequently,
\begin{equation*}
 \eta L=\frac{c_1L\sqrt{\log H}}{2H}
       \asymp\frac{1}{\delta\sqrt J}\longrightarrow0,
 \qquad
 \frac{c_2H}{L\sqrt{\log H}}
       \asymp\delta\sqrt J\longrightarrow\infty.
\end{equation*}
The first relation shows that the zero-free region no longer
supplies the decay used after
\eqref{eq:near-zero}: the factor
$\sqrt{\log H}\exp(-3\eta L/4)$ is now asymptotic to $\sqrt J$.
The second relation prevents the first inequality in
\eqref{eq:density-coeff}.
Thus the argument for making the contribution of zeros near $1$
equal to $o(x)$ fails at this scale.

Changing only the cutoff $\varepsilon\in(0,1/2)$ does not repair
the estimates in Steps~2 and~3. By \eqref{eq:character-family-size}
and the bounds for $\pi(y)$, we have $\log R_y\asymp\delta L$.
To obtain an $o(x)$ estimate from the final bound in
\eqref{eq:low-zeros}, one needs
\begin{equation*}
 \log\bigl(x^{-\varepsilon}R_yH^2\bigr)
 =-\varepsilon L+\log R_y+2\log H
 \longrightarrow-\infty.
\end{equation*}
Since $\log R_y\geq c\delta L$ for some absolute $c>0$,
this expression can tend to $-\infty$ only if
$\varepsilon\geq c\delta/2$ for all sufficiently large $x$.
Such a cutoff exceeds $\eta_1(y)$, which tends to zero.
We then have $\log(1/\varepsilon)=O_\delta(1)$, whereas
\begin{equation*}
 \frac{c_2H}{L\log(1/\varepsilon)}
       \gg_\delta J\longrightarrow\infty.
\end{equation*}
Hence the second inequality in \eqref{eq:density-coeff} cannot
hold either.

A direct enlargement of $y$ therefore does not yield the desired
improvement. To make the argument work at this larger scale, one would
need stronger control of zeros near $\Re s=1$, or an argument that avoids
the present absolute estimates
over the character family. These observations concern the estimates
used here; they do not assert that the factor $\log_3N$ is optimal.
