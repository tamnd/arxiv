<!-- assumption-1: body_md -->
\label{ass:c11-deformations}
Let $\mathfrak T\subset\mathscr C^{1,1}(\widehat\Gamma;\mathbb R^3)$ satisfy
the following conditions.
\begin{itemize}
 \item[\normalfont{(i)}] The set $\mathfrak T$ is compact.
 \item[\normalfont{(ii)}] For each ${\normalfont{\boldsymbol r}}\in\mathfrak T$, the image
 $\Gamma_{\normalfont{\boldsymbol r}}\coloneqq {\normalfont{\boldsymbol r}}(\widehat\Gamma)$ is the boundary of a bounded, simply connected
 $\mathscr C^{1,1}$ domain $\normalfont\text{D}_{\normalfont{\boldsymbol r}}$ with connected boundary.
 \item[\normalfont{(iii)}] For each ${\normalfont{\boldsymbol r}}\in\mathfrak T$, the map
 ${\normalfont{\boldsymbol r}}:\widehat\Gamma\to\Gamma_{\normalfont{\boldsymbol r}}\subset\mathbb R^3$ is a $\mathscr C^{1,1}$
 diffeomorphism.
\end{itemize}

\label{lem:c11-jacobian-multiplier}
Let $\mathfrak T$ satisfy Assumption~\ref{ass:c11-deformations}.
For each ${\normalfont{\boldsymbol r}}\in\mathfrak T$, the functions $J_{\text S,{\normalfont{\boldsymbol r}}}$ and
$J_{\text S,{\normalfont{\boldsymbol r}}}^{-1}$ belong to $\mathscr C^{0,1}(\widehat\Gamma)$.
Multiplication by either function is bounded on
$\mathscr C^{0,\alpha}(\widehat\Gamma)$ for $0<\alpha<1$,
and on $H^s(\widehat\Gamma)$ for $-1\leq s\leq1$.

\label{lem:c11-geometric-holomorphy}

satisfying Assumption~\ref{ass:c11-deformations}.

    $\mathfrak T\ni{\normalfont{\boldsymbol r}}\mapsto J_{\text S,{\normalfont{\boldsymbol r}}}^{-1} \in \mathscr C^{0,1}(\widehat\Gamma)$

\label{lem:c11-sobolev-extension}
For a given $0<\alpha<1$, let
$\widehat{\mathsf T}\in\mathscr L(\mathscr C^{0,\alpha}(\widehat\Gamma),
\allowbreak\mathscr C^{1,\alpha}(\widehat\Gamma))$ satisfy
\begin{equation}\label{eq:symmetry_Lax}
	\int_{\widehat\Gamma}(\widehat{\mathsf T}u)(\widehat{\mathbf{x}})
	    v(\widehat{\mathbf{x}})\,\,\mathrm d s_{\widehat{\mathbf{x}}}
	=
	\int_{\widehat\Gamma}u(\widehat{\mathbf{x}})
	    (\widehat{\mathsf T}v)(\widehat{\mathbf{x}})\,
	    \,\mathrm d s_{\widehat{\mathbf{x}}},
\end{equation}
for all $u,v\in\mathscr C^{0,\alpha}(\widehat\Gamma)$.
Then the following assertions hold.
\begin{itemize}
 \item[\normalfont{(i)}] The operator $\widehat{\mathsf T}$ has a unique
 bounded extension
 \begin{equation}
 	\mathcal E(\widehat{\mathsf T}):
 	H^{-\frac{1}{2}}(\widehat\Gamma)\to H^{\frac{1}{2}}(\widehat\Gamma).
 \end{equation}
 \item[\normalfont{(ii)}] For operators satisfying \eqref{eq:symmetry_Lax}, the extension
 operation $\mathcal E$ is complex linear and satisfies
 \begin{equation}
 	\|\mathcal E(\widehat{\mathsf T})\|_

We first prove a Sobolev estimate for $\widehat{\mathsf T}$ on its
original domain $\mathscr C^{0,\alpha}(\widehat\Gamma)$, and then
extend it by density. For this estimate, we use the Laplace single
layer BIO to define an equivalent norm on
$H^{-\frac12}(\widehat\Gamma)$.
Let $\widehat{\mathsf V}_0$ denote the Laplace single layer BIO on
$\widehat\Gamma$, defined, for $u\in\mathscr C^0(\widehat\Gamma)$, by 
\begin{equation}
	(\widehat{\mathsf V}_0u)(\widehat{\mathbf{x}})
	\coloneqq
	\int_{\widehat\Gamma}
	\frac{u(\widehat{\mathbf{y}})}
	{4\pi\|\widehat{\mathbf{x}}-\widehat{\mathbf{y}}\|}
	\,\,\mathrm d s_{\widehat{\mathbf{y}}},
	\quad\forall\widehat{\mathbf{x}}\in\widehat\Gamma,
\end{equation}
It is a symmetric isomorphism from
$\mathscr C^{0,\alpha}(\widehat\Gamma)$
to $\mathscr C^{1,\alpha}(\widehat\Gamma)$ for $0<\alpha<1$
\cite[Lemma~2.5]{BonnaillieDallaRivaDambrineMusolino2018}.
Its extension to the Sobolev spaces is a bounded, positive and coercive
isomorphism from $H^{-\frac12}(\widehat\Gamma)$ to
$H^{\frac12}(\widehat\Gamma)$, see e.g.,
\cite[Corollary~8.13]{McLean2000}.
This extension agrees with $\widehat{\mathsf V}_0$ on
$\mathscr C^{0,\alpha}(\widehat\Gamma)$.

The operator $\widehat{\mathsf V}_0$ is symmetric and coercive,
thus implying that
\begin{equation}
	(u,v)_0\coloneqq
	\langle u,\widehat{\mathsf V}_0\overline v\rangle_{\widehat\Gamma},
\end{equation}
for all $u,v\in H^{-\frac12}(\widehat\Gamma)$, defines an inner product in $ H^{-\frac{1}{2}}(\widehat\Gamma)$
and an induced norm $\|u\|_0\coloneqq(u,u)_0^{1/2}$ that is equivalent to
$\|u\|_{H^{-\frac{1}{2}}(\widehat\Gamma)}$.
The embedding of $\mathscr C^{0,\alpha}(\widehat\Gamma)$ in
$H^{-\frac12}(\widehat\Gamma)$ is continuous.

Set $\widehat{\mathsf K}\coloneqq
\widehat{\mathsf V}_0^{-1}\widehat{\mathsf T}
\in\mathscr L(\mathscr C^{0,\alpha}(\widehat\Gamma),
              \mathscr C^{0,\alpha}(\widehat\Gamma))$, and let
$\widehat{\mathsf J}$ be conjugation on
$\mathscr C^{0,\alpha}(\widehat\Gamma)$.
Symmetry of $\widehat{\mathsf T}$ and the real symmetric kernel of
$\widehat{\mathsf V}_0$ give
\begin{equation}
	(\widehat{\mathsf K}u,v)_0
	=\langle\overline v,\widehat{\mathsf T}u\rangle_{\widehat\Gamma}
	=\langle u,\widehat{\mathsf T}\overline v\rangle_{\widehat\Gamma}
	=(u,\widehat{\mathsf J}\widehat{\mathsf K}\widehat{\mathsf J}v)_0,
\end{equation}
for all $u,v\in\mathscr C^{0,\alpha}(\widehat\Gamma)$.
Conjugation is isometric on $\mathscr C^{0,\alpha}(\widehat\Gamma)$.
Hence $\widehat{\mathsf J}\widehat{\mathsf K}\widehat{\mathsf J}$
is bounded and complex linear, with the same operator norm as
$\widehat{\mathsf K}$.
We apply Lax's lemma (Theorem~\ref{thm:c11-operator-extension}) with
$X=Y=\mathscr C^{0,\alpha}(\widehat\Gamma)$, the inner product
$(\cdot,\cdot)_0$, $A=\widehat{\mathsf K}$, and
$B=\widehat{\mathsf J}\widehat{\mathsf K}\widehat{\mathsf J}$,
to obtain
\begin{equation}
	\|\widehat{\mathsf K}u\|_0
	\leq\|\widehat{\mathsf K}\|_
	 {\mathscr L(\mathscr C^{0,\alpha}(\widehat\Gamma),
	             \mathscr C^{0,\alpha}(\widehat\Gamma))}\|u\|_0,
	\quad \forall u\in\mathscr C^{0,\alpha}(\widehat\Gamma).
\end{equation}
Since $\widehat{\mathsf T}u=\widehat{\mathsf V}_0\widehat{\mathsf K}u$
for all $u\in\mathscr C^{0,\alpha}(\widehat\Gamma)$, boundedness of
$\widehat{\mathsf V}_0:H^{-\frac12}(\widehat\Gamma)\to
H^{\frac12}(\widehat\Gamma)$ and norm equivalence give
\begin{equation}
\label{eq:c11-original-operator-sobolev-bound}
	\begin{aligned}
	\|\widehat{\mathsf T}u\|_{H^{\frac12}(\widehat\Gamma)}
	&=\|\widehat{\mathsf V}_0\widehat{\mathsf K}u\|_
	    {H^{\frac12}(\widehat\Gamma)}\\
	&\leq C\|\widehat{\mathsf K}u\|_0
	 \leq C\|\widehat{\mathsf K}\|_
	 {\mathscr L(\mathscr C^{0,\alpha}(\widehat\Gamma),
	             \mathscr C^{0,\alpha}(\widehat\Gamma))}\|u\|_0\\
	&\leq C(\widehat\Gamma,\alpha)\|\widehat{\mathsf T}\|_
	 {\mathscr L(\mathscr C^{0,\alpha}(\widehat\Gamma),
	             \mathscr C^{1,\alpha}(\widehat\Gamma))}
	 \|u\|_{H^{-\frac12}(\widehat\Gamma)}.
	\end{aligned}
\end{equation}
The last inequality also uses boundedness of
$\widehat{\mathsf V}_0^{-1}:
\mathscr C^{1,\alpha}(\widehat\Gamma)\to
\mathscr C^{0,\alpha}(\widehat\Gamma)$.
The constant in the last line of
\eqref{eq:c11-original-operator-sobolev-bound} reads
\begin{equation}
\label{eq:c11-sobolev-extension-constant}
\begin{aligned}
	C(\widehat\Gamma,\alpha)\coloneqq
	&
	\|\widehat{\mathsf V}_0\|^{\frac32}_{\mathscr L(H^{-\frac12}(\widehat\Gamma),H^{\frac12}(\widehat\Gamma))}
	\|\widehat{\mathsf V}_0^{-1}\|^{\frac12}_{\mathscr L(H^{\frac12}(\widehat\Gamma),H^{-\frac12}(\widehat\Gamma))}
	\|\widehat{\mathsf V}_0^{-1}\|_
	{\mathscr L(\mathscr C^{1,\alpha}(\widehat\Gamma),
     \mathscr C^{0,\alpha}(\widehat\Gamma))},
\end{aligned}
\end{equation}
which only depends on $\widehat\Gamma$ and $\alpha$.

Since $\widehat{\mathsf V}_0$ is the Laplace single layer BIO on the
fixed reference boundary, this constant depends only on $\alpha$ and
$\widehat\Gamma$, and is independent of $\widehat{\mathsf T}$ and $u$.
Therefore, \eqref{eq:c11-original-operator-sobolev-bound} bounds the original
operator $\widehat{\mathsf T}$ in the fractional Sobolev norms,
with the constant given in \eqref{eq:c11-sobolev-extension-constant}.

By density of $\mathscr C^{0,\alpha}(\widehat\Gamma)$ in
$H^{-\frac12}(\widehat\Gamma)$,
estimate~\eqref{eq:c11-original-operator-sobolev-bound} gives a unique
bounded extension
$\mathcal E(\widehat{\mathsf T}):H^{-\frac12}(\widehat\Gamma)
\to H^{\frac12}(\widehat\Gamma)$
satisfying~\eqref{eq:c11-sobolev-extension-agreement}, with the same norm bound.
Uniqueness also implies complex linearity of $\mathcal E$.

$\mathfrak T\to\mathscr L(\mathscr C^{0,\alpha}(\widehat\Gamma),

\cite[Sections~5 and~13]{Mujica1986}.
Since the inclusion

 =\mathcal E(\mathrm D_{\normalfont{\boldsymbol r}}\mathsf B_{\normalfont{\boldsymbol r}}[\boldsymbol h]).

	=\widehat{\mathsf R}^{(D)}(J_{\text S,{\normalfont{\boldsymbol r}}})\tau_{\normalfont{\boldsymbol r}}^s
	    \operatorname{div}_{\Gamma_{\normalfont{\boldsymbol r}}^s}\boldsymbol j.
	\label{eq:c11-screen-piola-divergence}
\end{equation}
The field formula for $\mathcal P_{\normalfont{\boldsymbol r}}^s$,
\eqref{eq:c11-screen-piola-divergence}, and the multiplier and
pullback estimates in Lemma~\ref{lem:c11-screen-scalar-operators},
items~(i) and~(ii),
give boundedness from $X_{\normalfont{\boldsymbol r}}^s$ to $\widehat X^s$.

Applying the same divergence calculation to the inverse surface
diffeomorphism ${\normalfont{\boldsymbol r}}^{-1}:\Gamma_{\normalfont{\boldsymbol r}}^s\to\widehat\Gamma^s$, together with
the screen multiplier and pullback estimates, gives the bounded map
\begin{equation}
	(\mathcal P_{\normalfont{\boldsymbol r}}^s)^{-1}\widehat{\boldsymbol j}
	=(\tau_{\normalfont{\boldsymbol r}}^s)^{-1}
	  \bigl(\widehat{\mathsf R}^{(D)}(J_{\text S,{\normalfont{\boldsymbol r}}}^{-1}F_{\normalfont{\boldsymbol r}})
	         \widehat{\boldsymbol j}\bigr)\in X_{\normalfont{\boldsymbol r}}^s,
	\quad\forall\widehat{\boldsymbol j}\in\widehat X^s.
\end{equation}
The tangential-projector identities
$F_{\normalfont{\boldsymbol r}}^{-1}F_{\normalfont{\boldsymbol r}}=\widehat P$ and
$F_{\normalfont{\boldsymbol r}} F_{\normalfont{\boldsymbol r}}^{-1}=\mathsf I_3-\boldsymbol N_{\normalfont{\boldsymbol r}}\boldsymbol N_{\normalfont{\boldsymbol r}}^\top$,
together with tangency and injectivity of $\tau_{\normalfont{\boldsymbol r}}^s$, show that these
maps are mutually inverse.
Since $\widehat{\mathsf E}_X$ acts component-wise as
$\widehat{\mathsf E}^{(N)}$,
\eqref{eq:c11-screen-piola-components} is
\eqref{eq:c11-screen-piola-extension}, completing the proof.

Fix ${\normalfont{\boldsymbol r}}\in\mathfrak T$.
By change of variables and the positive minimum and finite maximum
of $J_{\text S,{\normalfont{\boldsymbol r}}}$ on $\widehat\Gamma$, we obtain
\begin{equation}
	\|\tau_{\normalfont{\boldsymbol r}} f\|_{L^2(\widehat\Gamma)}
	\leq C_{\normalfont{\boldsymbol r}}\|f\|_{L^2(\Gamma_{\normalfont{\boldsymbol r}})}
	\quad\text{and}\quad
	\|\tau_{\normalfont{\boldsymbol r}}^{-1}\widehat f\|_{L^2(\Gamma_{\normalfont{\boldsymbol r}})}
	\leq C_{\normalfont{\boldsymbol r}}\|\widehat f\|_{L^2(\widehat\Gamma)},
\end{equation}
for all $f\in L^2(\Gamma_{\normalfont{\boldsymbol r}})$ and
$\widehat f\in L^2(\widehat\Gamma)$.
Recalling the tangential inverse $F_{\normalfont{\boldsymbol r}}^{-1}$,
the tangential chain rule gives, for all
$f\in\mathscr C^1(\Gamma_{\normalfont{\boldsymbol r}})$ and
$\widehat f\in\mathscr C^1(\widehat\Gamma)$,
\begin{equation}
	\nabla_{\widehat\Gamma}(\tau_{\normalfont{\boldsymbol r}} f)
	=F_{\normalfont{\boldsymbol r}}^\top\tau_{\normalfont{\boldsymbol r}}(\nabla_{\Gamma_{\normalfont{\boldsymbol r}}}f)
	\quad\text{and}\quad
	\tau_{\normalfont{\boldsymbol r}}\bigl(\nabla_{\Gamma_{\normalfont{\boldsymbol r}}}
	                   (\tau_{\normalfont{\boldsymbol r}}^{-1}\widehat f)\bigr)
	=F_{\normalfont{\boldsymbol r}}^{-\top}\nabla_{\widehat\Gamma}\widehat f.
\end{equation}
Using these chain rules, the boundedness of $F_{\normalfont{\boldsymbol r}}$ and $F_{\normalfont{\boldsymbol r}}^{-1}$,
and the preceding estimates, we obtain
\begin{equation}
	\|\tau_{\normalfont{\boldsymbol r}} f\|_{H^1(\widehat\Gamma)}
	\leq C_{\normalfont{\boldsymbol r}}\|f\|_{H^1(\Gamma_{\normalfont{\boldsymbol r}})}
	\quad\text{and}\quad
	\|\tau_{\normalfont{\boldsymbol r}}^{-1}\widehat f\|_{H^1(\Gamma_{\normalfont{\boldsymbol r}})}
	\leq C_{\normalfont{\boldsymbol r}}\|\widehat f\|_{H^1(\widehat\Gamma)}.
\end{equation}
By density, these identities and bounds hold for all
$f\in H^1(\Gamma_{\normalfont{\boldsymbol r}})$ and
$\widehat f\in H^1(\widehat\Gamma)$.
By interpolation \cite[Theorem~B.11]{McLean2000}, these bounds give
boundedness of $\tau_{\normalfont{\boldsymbol r}}$ and
$\tau_{\normalfont{\boldsymbol r}}^{-1}$ between $H^s(\Gamma_{\normalfont{\boldsymbol r}})$ and
$H^s(\widehat\Gamma)$ for $0<s<1$.
The identities $\tau_{\normalfont{\boldsymbol r}}^{-1}\tau_{\normalfont{\boldsymbol r}} f=f$ and
$\tau_{\normalfont{\boldsymbol r}}\tau_{\normalfont{\boldsymbol r}}^{-1}\widehat f=\widehat f$ hold for
$f\in H^s(\Gamma_{\normalfont{\boldsymbol r}})$ and $\widehat f\in H^s(\widehat\Gamma)$,
since these spaces are contained in $L^2(\Gamma_{\normalfont{\boldsymbol r}})$ and
$L^2(\widehat\Gamma)$, respectively.

For $f\in\mathscr C^{0,\alpha}(\Gamma_{\normalfont{\boldsymbol r}})$,
\begin{equation}
	|f\circ{\normalfont{\boldsymbol r}}|_{\mathscr C^{0,\alpha}(\widehat\Gamma)}
	\leq\operatorname{Lip}({\normalfont{\boldsymbol r}})^\alpha
	      |f|_{\mathscr C^{0,\alpha}(\Gamma_{\normalfont{\boldsymbol r}})}.
\end{equation}
The seminorm estimate gives boundedness of $\tau_{\normalfont{\boldsymbol r}}$ from
$\mathscr C^{0,\alpha}(\Gamma_{\normalfont{\boldsymbol r}})$ to
$\mathscr C^{0,\alpha}(\widehat\Gamma)$.
The two surface gradient chain rules give boundedness from
$\mathscr C^{1,\alpha}(\Gamma_{\normalfont{\boldsymbol r}})$ to
$\mathscr C^{1,\alpha}(\widehat\Gamma)$, using the Lipschitz coefficients
$F_{\normalfont{\boldsymbol r}}$ and $F_{\normalfont{\boldsymbol r}}^{-1}$.
Applying these estimates with ${\normalfont{\boldsymbol r}}^{-1}$ and the boundaries
interchanged gives the bounds for $\tau_{\normalfont{\boldsymbol r}}^{-1}$.

For $-1\leq s<0$, consider the maps
$\widehat\phi\mapsto\tau_{\normalfont{\boldsymbol r}}^{-1}
(J_{\text S,{\normalfont{\boldsymbol r}}}^{-1}\widehat\phi)$ from
$H^{-s}(\widehat\Gamma)$ to $H^{-s}(\Gamma_{\normalfont{\boldsymbol r}})$ and
$\phi\mapsto J_{\text S,{\normalfont{\boldsymbol r}}}\tau_{\normalfont{\boldsymbol r}}\phi$ in the opposite direction.
Composing these two maps cancels the Jacobian factors and the
pullbacks, so each composition is the identity.
Their boundedness between $H^{-s}(\widehat\Gamma)$ and
$H^{-s}(\Gamma_{\normalfont{\boldsymbol r}})$ follows from the
positive-order pullback bounds and
Lemma~\ref{lem:c11-jacobian-multiplier}.
By~\eqref{eq:c11-distribution-pullbacks}, $\tau_{\normalfont{\boldsymbol r}}$ and $\tau_{\normalfont{\boldsymbol r}}^{-1}$
on $H^s(\Gamma_{\normalfont{\boldsymbol r}})$ and $H^s(\widehat\Gamma)$
are the duals of these maps. They are therefore bounded and
mutually inverse.
All constants may depend on ${\normalfont{\boldsymbol r}}$,
but not on the functions to which the operators are applied.

\label{thm:c11-operator-extension}
Let $X$ and $Y$ be complex normed spaces with inner products
$(\cdot,\cdot)_X$ and $(\cdot,\cdot)_Y$. Let
$\|u\|_{X,0}\coloneqq(u,u)_X^{1/2}$ and
$\|v\|_{Y,0}\coloneqq(v,v)_Y^{1/2}$, and assume that
\begin{equation}
	\|u\|_{X,0}\leq c_X\|u\|_X
	\quad\text{and}\quad
	\|v\|_{Y,0}\leq c_Y\|v\|_Y,
	\quad\forall u\in X,\ \forall v\in Y,
\end{equation}
for some $c_X,c_Y>0$.
Suppose that $A\in\mathscr L(X,Y)$ and
$B\in\mathscr L(Y,X)$ satisfy
\begin{equation}
	(Au,v)_Y=(u,Bv)_X,
	\quad\forall u\in X,\ \forall v\in Y.
\end{equation}
Then
\begin{equation}
	\|Au\|_{Y,0}^2
	\leq\|A\|_{\mathscr L(X,Y)}
	     \|B\|_{\mathscr L(Y,X)}\|u\|_{X,0}^2,
	\quad\forall u\in X.
\end{equation}
Consequently, $A$ has a unique bounded extension between the
Hilbert completions of $X$ and $Y$ in their inner product norms,
with operator norm at most
$\bigl(\|A\|_{\mathscr L(X,Y)}
\|B\|_{\mathscr L(Y,X)}\bigr)^{1/2}$.
