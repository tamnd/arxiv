<!-- theorem-1: body_md -->
\label{thm:bounded}
Assume \eqref{eq:source}. Let $\mu$ be supported in a ball of radius
$R>0$, and let $\nu\in\mathcal P_2(\mathds R^d)$ be arbitrary. Then, with
$\delta=W_2(\mu,\nu)$,
\begin{equation}\label{eq:bounded}
\lVert T_\mu-T_\nu\rVert_{L^2(\rho)}
\leqslant \delta+2\sqrt2\,R^{1-\alpha_p}(\mathsf S_p\delta)^{\alpha_p}.
\end{equation}
In particular, $\alpha_p=(p-1)/(4p-2)$ and $\alpha_\infty=1/4$.

<!-- theorem-2: body_md -->
\label{thm:moment}
Assume \eqref{eq:source}. Let $\nu\in\mathcal P_2(\mathds R^d)$ be arbitrary, and
suppose that, for some $q>2$ and $A>0$,
\[
\int |y|^q\,\mathrm d\mu(y)\leqslant A^q.
\]
Set
\begin{equation}\label{eq:beta}
\beta_{p,q}=\frac{\alpha_p(q-2)}{q-2\alpha_p}
=\frac{q-2}{2q(p'+1)-2}.
\end{equation}
Then, with $\delta=W_2(\mu,\nu)$,
\begin{equation}\label{eq:moment}
\lVert T_\mu-T_\nu\rVert_{L^2(\rho)}
\leqslant\delta+10A^{1-\beta_{p,q}}(\mathsf S_p\delta)^{\beta_{p,q}}.
\end{equation}
For $p=\infty$, the exponent is $(q-2)/(4q-2)$.

<!-- proposition-1: body_md -->
\label{prop:sharpness}
For every $p\in(1,\infty]$, there is a fixed source on $\mathds R^2$ satisfying
\eqref{eq:source} for which the following hold.
\begin{enumerate}[(i)]
\item There are targets $\mu_n$, $\nu_n$ in the unit ball with
$\delta_n=W_2(\mu_n,\nu_n)\to0$ such that, for every $a>\alpha_p$,
\[
\frac{\lVert T_{\mu_n}-T_{\nu_n}\rVert_{L^2(\rho)}}{\delta_n^a}\longrightarrow\infty.
\]
\item For each $q>2$, there are targets with both $q$-moments at most one
and $\delta_n=W_2(\mu_n,\nu_n)\to0$ such that, for every $b>\beta_{p,q}$,
\[
\frac{\lVert T_{\mu_n}-T_{\nu_n}\rVert_{L^2(\rho)}}{\delta_n^b}\longrightarrow\infty.
\]
\end{enumerate}
For $p=\infty$, the source itself has a bounded BV density.

<!-- proposition-2: body_md -->
\label{prop:dimension}
There are universal $c,C>0$ such that, in every sufficiently large
dimension $d$, the source $\rho=\gamma_d=N(0,I_d)$ admits targets
$\mu_d$, $\nu_d$ in the unit ball satisfying
\[
W_2(\mu_d,\nu_d)\leqslant C d^{-1/2},\qquad
\lVert T_{\mu_d}-T_{\nu_d}\rVert_{L^2(\gamma_d)}\geqslant c.
\]
Consequently, a uniform estimate $\lVert T_\mu-T_\nu\rVert_2\leqslant C_dW_2(\mu,\nu)^a$
with $a>0$ requires $C_d\geqslant c_a d^{a/2}$.
<!-- end proposition-2 -->
<!-- proof-1: body_md -->
First let $d=2n$. Choose a binary linear code $\mathcal C\subset\{0,1\}^n$
of dimension $k=\lfloor n/4\rfloor$ whose nonzero words have weight
greater than $n/8$. Such codes exist for large $n$: for a random linear
map from $\{0,1\}^k$ to $\{0,1\}^n$, a union bound gives a failure
probability at most $2^{k-n(1-H_2(1/8))}$, which tends to zero; here
$H_2$ is the binary entropy and $1-H_2(1/8)>1/4$.

Write the associated sign vectors as
$v_i=n^{-1/2}((-1)^{c_j})_{j=1}^n$, $1\leqslant i\leqslant N=2^k$.
They have unit norm, satisfy $|v_i-v_j|^2\geqslant1/2$ for $i\ne j$,
and are permuted transitively by orthogonal coordinate sign changes.
Put
\[
r=\frac{\log N-1}{\log N+1},\qquad
a_i=(v_i,0),\qquad b_i=(rv_i,\sqrt{1-r^2}\,v_i),
\]
and let $\mu_d$, $\nu_d$ be uniform on the $a_i$, $b_i$, respectively.
For independent standard Gaussian vectors $X,Z\in\mathds R^n$, set
$Y=rX+\sqrt{1-r^2}Z$ and $I(x)=\arg\max_i\langle v_i,x\rangle$.
Ties have probability zero, and transitivity implies
$\operatorname{\mathds P}[I(X)=i]=1/N$. The gradients of the two maxima of linear functions
are therefore the Brenier maps:
\[
T_{\mu_d}(X,Z)=a_{I(X)},\qquad T_{\nu_d}(X,Z)=b_{I(Y)}.
\]
Nelson's Gaussian hypercontractivity theorem
(see~\cite[Theorem~2.1]{ChafaiLehecLSI}) implies, for the cells
$A_i=\{I=i\}$,
\[
\operatorname{\mathds P}[I(X)=I(Y)]
\leqslant\sum_i\gamma_n(A_i)^{2/(1+r)}
=N^{-(1-r)/(1+r)}=e^{-1}.
\]
Since $|a_i-b_j|^2=2(1-r)+r|v_i-v_j|^2$, diagonal matching is optimal
between the two uniform targets. Therefore
\[
W_2(\mu_d,\nu_d)^2=\frac4{\log N+1},\qquad
\lVert T_{\mu_d}-T_{\nu_d}\rVert_2^2\geqslant\frac r2(1-e^{-1}).
\]
For odd dimensions, add an independent Gaussian coordinate on which
both maps are constant.

<!-- lemma-1: body_md -->
\label{lem:gap-stability}
Assume \eqref{eq:source}, and set $H_p=\Lambda B_0^{1/p'}$.
For any convex $R$-Lipschitz function $\phi$, define almost everywhere
\begin{equation}\label{eq:pointwise-bias}
a_t=\phi_t-\phi+\frac t2|\nabla\phi|^2,\qquad
A_t=\int a_t\,\mathrm d\rho.
\end{equation}
Then
\begin{equation}\label{eq:bias}
A_t\leqslant\frac{H_p}{2\,3^{1/p'}}
R^{2+1/p'}t^{1+1/p'}.
\end{equation}
If $\nabla\phi$ pushes $\rho$ to $\mu$, then for every
$\sigma\in\Pi(\rho,\mu)$,
\begin{equation}\label{eq:gap-stability}
0\leqslant\mathcal G_t(\sigma)-\mathcal G(\sigma)
=\mathcal G_t(\sigma_*)=A_t.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
By \eqref{eq:Moreau-dual} and Fenchel equality,
$a_t(x)=g_{\phi_t}(x,\nabla\phi(x))\geqslant0$ almost everywhere.
Since $\phi_t\leqslant\phi$ and $|\nabla\phi|\leqslant R$, we also have
$0\leqslant a_t\leqslant tR^2/2$. If $(\nabla\phi)_\#\rho=\mu$, the
common marginals and \eqref{eq:Moreau-dual} give
\[
\mathcal G_t(\sigma)-\mathcal G(\sigma)
=\int(\phi_t-\phi)\,\mathrm d\rho+\frac t2\int|y|^2\,\mathrm d\mu=A_t.
\]
Evaluating at $\sigma_*$ proves \eqref{eq:gap-stability}.
All integrals are finite: the potentials have linear growth and
$\phi^*$ is integrable against $\mu$ by Fenchel equality on $\sigma_*$.

We next prove the second-order estimate
\begin{equation}\label{eq:BV-bias}
\int a_t\,\mathrm d\rho_0\leqslant\frac{R^3B_0}{6}t^2.
\end{equation}
First take $\rho=\rho_0$ and suppose that $\phi$ is smooth.
The Moreau envelope satisfies
the Hamilton--Jacobi equation
\[
\partial_s\phi_s+\frac12|\nabla\phi_s|^2=0,\qquad \phi_0=\phi.
\]
Differentiating in time, we find $A_0=A'_0=0$ and
\[
A''_s=\int\langle \nabla\phi_s,D^2\phi_s\nabla\phi_s\rangle\,f_0\,\mathrm d x.
\]
Since $D^2\phi_s\geqslant0$, we have
\[
\operatorname{div}(|\nabla\phi_s|^2\nabla\phi_s)
\geqslant3\langle \nabla\phi_s,D^2\phi_s\nabla\phi_s\rangle.
\]
Integration by parts and the bound $|\nabla\phi_s|\leqslant R$ imply
\begin{equation}\label{eq:bias-second-derivative}
0\leqslant A''_s
\leqslant-\frac13\int |\nabla\phi_s|^2\nabla\phi_s\cdot\,\mathrm d Df_0
\leqslant\frac{R^3B_0}{3}.
\end{equation}
Integrating twice in time, we obtain
\[
A_t=\int_0^t(t-s)A''_s\,\mathrm d s
\leqslant\frac{R^3B_0}{6}t^2.
\]
In the integration by parts, cutoffs on balls of radius $a$
produce an error bounded by $CR^3/a$, which tends to zero.
For a general convex $R$-Lipschitz $\phi$, apply the estimate to smooth
convolutions. As the mollification parameter vanishes, the potentials
and their envelopes converge uniformly, while the gradients of the initial
potentials converge almost everywhere to $\nabla\phi$ and remain
bounded by $R$. Dominated convergence proves \eqref{eq:BV-bias} for
the original convex Lipschitz potential.

For $\rho=h\rho_0$ and $1<p<\infty$, H\"older's inequality gives
\[
A_t=\int a_t h\,\mathrm d\rho_0
\leqslant\Lambda\biggl(\int a_t^{p'}\,\mathrm d\rho_0\biggr)^{\!1/p'}.
\]
Using $0\leqslant a_t\leqslant tR^2/2$ and \eqref{eq:BV-bias}, we have
\[
\int a_t^{p'}\,\mathrm d\rho_0
\leqslant\biggl(\frac{tR^2}{2}\biggr)^{\!p'-1}\int a_t\,\mathrm d\rho_0
\leqslant\biggl(\frac{tR^2}{2}\biggr)^{\!p'-1}\frac{R^3B_0t^2}{6}.
\]
Taking the $1/p'$ power and using $(p'-1)/p'=1/p$, we conclude that
\[
A_t\leqslant\Lambda
\biggl(\frac{tR^2}{2}\biggr)^{\!1/p}
\biggl(\frac{R^3B_0t^2}{6}\biggr)^{\!1/p'}
=\frac{H_p}{2\,3^{1/p'}}R^{2+1/p'}t^{1+1/p'}.
\]
For $p=\infty$, the same bound follows directly from
$A_t\leqslant\Lambda\int a_t\,\mathrm d\rho_0$ and \eqref{eq:BV-bias}, with $p'=1$.

<!-- lemma-2: body_md -->
\label{lem:coercivity}
For $t>0$, the regularized potential satisfies
\begin{equation}\label{eq:smooth}
\lVert \nabla\phi_t\rVert_\infty\leqslant R,\qquad
\operatorname{Lip}(\nabla\phi_t)\leqslant\frac1t=:L_t.
\end{equation}
For every $x,y\in\mathds R^d$,
\begin{equation}\label{eq:Fenchel}
g_{\phi_t}(x,y)\geqslant\frac t2|y-\nabla\phi_t(x)|^2.
\end{equation}
Integrating against any $\sigma\in\Pi(\rho,\mu)$, we obtain
\begin{equation}\label{eq:coercivity}
\int|y-\nabla\phi_t(x)|^2\,\mathrm d\sigma(x,y)\leqslant\frac2t\mathcal G_t(\sigma).
\end{equation}
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
Let $z_t(x)$ be the unique minimizer in \eqref{eq:regularizedgap}.
The optimality condition gives
\[
\nabla\phi_t(x)=\frac{x-z_t(x)}t\in\partial\phi(z_t(x)),
\]
so its norm is at most $R$. Monotonicity of $\partial\phi$ implies
\[
t|\nabla\phi_t(x)-\nabla\phi_t(x')|^2
\leqslant\langle \nabla\phi_t(x)-\nabla\phi_t(x'),x-x'\rangle,
\]
which proves the Lipschitz bound. In the definition of $\phi_t^*(y)$,
use the inequality
\[
\phi_t(z)\leqslant\phi_t(x)+\langle \nabla\phi_t(x),z-x\rangle
+\frac1{2t}|z-x|^2
\]
at $z=x+t(y-\nabla\phi_t(x))$ to obtain \eqref{eq:Fenchel}.
Integration proves \eqref{eq:coercivity}. This is the
smoothness--strong-convexity argument used by Li and
Nochetto~\cite[Lemma~2.2 and Proposition~3.1]{LiNochettoQuantitative}.
For proximal versions of the Fenchel--Young inequality with a remainder,
see Carlier~\cite{CarlierFenchelYoung}.

<!-- lemma-3: body_md -->
\label{lem:gap-distance}
Let $\mu,\nu\in\mathcal P_2(\mathds R^d)$ and $\delta=W_2(\mu,\nu)$.
There exist jointly distributed $(X_\nu,Y_\nu,Y_\mu)$ with
$X_\nu\sim\rho$, $Y_\nu=T_\nu(X_\nu)$ and $(Y_\mu,Y_\nu)$ an optimal
coupling of the targets, such that $\sigma=\operatorname{Law}(X_\nu,Y_\mu)$
satisfies
\begin{equation}\label{eq:gap-distance}
\mathcal G(\sigma)\leqslant2M_\rho\delta.
\end{equation}
Here $\mathcal G$ is defined by \eqref{eq:gap}.
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
Glue a $W_2$-optimal target coupling to the two optimal graph plans to
obtain $(X_\mu,X_\nu,Y_\mu,Y_\nu)$ with $X_\eta\sim\rho$ and
$Y_\eta=T_\eta(X_\eta)$ for $\eta\in\{\mu,\nu\}$.
Optimality of $T_\nu$ gives
$\operatorname{\mathds E}[\langle X_\mu,Y_\nu\rangle]\leqslant\operatorname{\mathds E}[\langle X_\nu,Y_\nu\rangle]$. Hence
\begin{align*}
\mathcal G(\sigma)
&=\operatorname{\mathds E}[\langle X_\mu-X_\nu,Y_\mu\rangle]\\
&\leqslant\operatorname{\mathds E}[\langle X_\mu-X_\nu,Y_\mu-Y_\nu\rangle]\\
&\leqslant\lVert X_\mu-X_\nu\rVert_2\lVert Y_\mu-Y_\nu\rVert_2
\leqslant2M_\rho\delta.
\end{align*}
The last bound uses
$\lVert X_\mu-X_\nu\rVert_2\leqslant\lVert X_\mu-m_\rho\rVert_2+
\lVert X_\nu-m_\rho\rVert_2$.

<!-- remark-2: body_md -->
\label{rem:one-sided}
For bounded, compactly supported source densities and bounded $\mu$,
M\'erigot's coercivity estimate~\cite[Theorem~3.4]{MerigotSharpStability}
combined with Lemma~\ref{lem:gap-distance} gives
$\lVert T_\mu-T_\nu\rVert_{L^2(\rho)}\leqslant\delta+C\delta^{1/4}$ for arbitrary
$\nu\in\mathcal P_2(\mathds R^d)$, where $\delta=W_2(\mu,\nu)$.
<!-- end remark-2 -->
<!-- proof-5: body_md -->
Translate both targets to arrange $\operatorname{supp}\mu\subset\overline B(0,R)$.
Choose the coupling $\sigma$ from Lemma~\ref{lem:gap-distance}.
Applying Lemma~\ref{lem:coercivity} to $\sigma$ and $\sigma_*$ gives
\begin{align}
\int|T_\mu(x)-y|^2\,\mathrm d\sigma(x,y)
&\leqslant4L_t\bigl(\mathcal G_t(\sigma_*)+\mathcal G_t(\sigma)\bigr)\notag\\
&\leqslant4L_t\mathcal G(\sigma)+8L_tA_t,\label{eq:graph}
\end{align}
where the second inequality is Lemma~\ref{lem:gap-stability}.
Since $Y_\nu=T_\nu(X_\nu)$ and
$\lVert Y_\mu-Y_\nu\rVert_2=\delta$, the $L^2$ triangle inequality implies,
with $D=\lVert T_\mu-T_\nu\rVert_{L^2(\rho)}$,
\begin{equation}\label{eq:balance}
(D-\delta)_+^2\leqslant\frac{8M_\rho\delta}{t}
+\frac4{3^{1/p'}}H_pR^{2+1/p'}t^{1/p'}.
\end{equation}
For $\delta>0$, choose
\[
s=\biggl(\frac{M_\rho\delta}{H_pR^{2+1/p'}}\biggr)^{\!p'/(p'+1)},
\qquad t=2s.
\]
Since $(2/3)^{1/p'}\leqslant1$, the right-hand side of
\eqref{eq:balance} is at most $8M_\rho\delta/s$.
Using $M_\rho H_p^{p'}=\mathsf S_p$ and taking square roots gives
\eqref{eq:bounded}. When $\delta=0$, the two maps agree by uniqueness.

<!-- remark-4: body_md -->
\label{rem:heat}
The proof also applies to the heat regularization
$P_t\phi$, where $P_t=e^{t\Delta}$. Convexity gives $P_t\phi\geqslant\phi$
and $(P_t\phi)^*\leqslant\phi^*$, so the change in the duality gap is
bounded by $\int(P_t\phi-\phi)\,\mathrm d\rho$.
Set $\kappa_d=\sqrt2\,\operatorname{\mathds E}[|G_d|]$ for $G_d\sim N(0,I_d)$.
The pointwise bound $0\leqslant P_t\phi-\phi\leqslant R\kappa_d\sqrt t$
and the BV estimate
$\int(P_t\phi-\phi)\,\mathrm d\rho_0\leqslant RB_0t$, combined with
H\"older's inequality, imply
\[
\int(P_t\phi-\phi)\,\mathrm d\rho
\leqslant R\Lambda\kappa_d^{1/p}B_0^{1/p'}t^{(1+1/p')/2}.
\]
The regularized gradient satisfies
\[
\operatorname{Lip}(\nabla P_t\phi)\leqslant\frac{R}{\sqrt{\pi t}}.
\]
The same coercivity and balancing argument proves a bound of the form
\eqref{eq:bounded}, with a universal numerical constant and source
parameter $\mathsf S_p\kappa_d^{p'-1}$. The H\"older exponent is $\alpha_p$.
Since $\kappa_d$ is of order $\sqrt d$, the extra factor contributes to
the dimension dependence when $1<p<\infty$ and equals one when
$p=\infty$.

\label{prop:tailestimate}
Assume \eqref{eq:source}, and let $\mu$, $\nu\in\mathcal P_2(\mathds R^d)$.
For every $R>0$ such that $\mu(\overline B(0,R))>0$, with
$\delta=W_2(\mu,\nu)$, one has
\begin{equation}\label{eq:tailestimate}
\lVert T_\mu-T_\nu\rVert_{L^2(\rho)}
\leqslant\delta+2\sqrt2\,R^{1-\alpha_p}(\mathsf S_p\delta)^{\alpha_p}
+3\sqrt2\,\tau_\mu(R).
\end{equation}

<!-- proof-6: body_md -->
The case $\delta=0$ follows from uniqueness, so assume $\delta>0$.
Let $\phi$ be a proper lower semicontinuous Brenier potential from
$\rho$ to $\mu$. Define the slope truncation
\begin{equation}\label{eq:truncation}
\phi_R(x)=(\phi^*+\iota_{\overline B(0,R)})^*(x)
=\sup_{|y|\leqslant R}{\{\langle x,y\rangle-\phi^*(y)\}},
\end{equation}
where $\iota_E$ is zero on $E$ and $+\infty$ outside $E$.
Since $\phi^*$ is finite $\mu$-almost everywhere and
$\mu(\overline B(0,R))>0$, the supremum includes a finite affine function.
Lower semicontinuity on the compact ball bounds $\phi^*$ from below
there. Hence $\phi_R$ is finite, convex and $R$-Lipschitz, with
\begin{equation}\label{eq:truncateddual}
\phi_R\leqslant\phi,\qquad
\phi_R^*=\phi^*+\iota_{\overline B(0,R)}.
\end{equation}
This is the Pasch--Hausdorff Lipschitz regularization,
\[
\phi_R(x)=\inf_z{\{\phi(z)+R|x-z|\}};
\]
see
Bauschke and Combettes~\cite[Proposition~12.17]{BauschkeCombettesConvex}
for the envelope properties and Beck and
Teboulle~\cite[Lemma~2.1]{BeckTeboulleExact} for its dual representation.

Write $T=\nabla\phi$, $T_R=\nabla\phi_R$ and let
$\phi_{R,t}$ be the Moreau--Yosida regularization of $\phi_R$.
Set
\[
a_{R,t}=\phi_{R,t}-\phi_R+\frac t2|T_R|^2,\qquad
A_{R,t}=\int a_{R,t}\,\mathrm d\rho.
\]
Applying Lemma~\ref{lem:gap-stability} to $\phi_R$, we obtain
\begin{equation}\label{eq:truncated-bias}
A_{R,t}\leqslant\frac{H_p}{2\,3^{1/p'}}R^{2+1/p'}t^{1+1/p'}.
\end{equation}
Also $|T_R|$, $|\nabla\phi_{R,t}|\leqslant R$. Wherever $|T(x)|\leqslant R$,
Fenchel equality gives $\phi_R(x)=\phi(x)$, hence $T_R(x)=T(x)$
almost everywhere on that set. Consequently,
\begin{equation}\label{eq:truncated-moment}
0\leqslant\int\bigl(|T|^2-|T_R|^2\bigr)\,\mathrm d\rho
\leqslant\tau_\mu(R)^2.
\end{equation}

For $|y|\leqslant R$, the conjugacy identities
\eqref{eq:truncateddual} and \eqref{eq:Moreau-dual} imply
\begin{equation}\label{eq:smallslopes}
g_{\phi_{R,t}}(x,y)
\leqslant g_\phi(x,y)+a_{R,t}(x)
+\frac t2\bigl(|y|^2-|T_R(x)|^2\bigr).
\end{equation}
For any $\sigma\in\Pi(\rho,\mu)$, the function
$|y|^2-|T_R(x)|^2$ is nonnegative on $\{|y|>R\}$.
Thus its integral over $\{|y|\leqslant R\}$ is bounded
above by its integral over all pairs, which is at most
$\tau_\mu(R)^2$ by \eqref{eq:truncated-moment}.
Since $g_\phi$ and $a_{R,t}$ are nonnegative, coercivity gives
\[
\int_{|y|\leqslant R}|\nabla\phi_{R,t}(x)-y|^2\,\mathrm d\sigma
\leqslant\frac2t\bigl(\mathcal G(\sigma)+A_{R,t}\bigr)+\tau_\mu(R)^2.
\]
On $\{|y|>R\}$, use
$|\nabla\phi_{R,t}(x)-y|^2\leqslant4|y|^2$ instead. We obtain
\begin{equation}\label{eq:tailgraph1}
\int|\nabla\phi_{R,t}(x)-y|^2\,\mathrm d\sigma
\leqslant\frac2t\bigl(\mathcal G(\sigma)+A_{R,t}\bigr)+5\tau_\mu(R)^2.
\end{equation}
Here $\mathcal G$ is the gap for the original potential $\phi$ as in
\eqref{eq:gap}. Its potential terms are integrable: affine lower
bounds for $\phi$ and $\phi^*$ control their negative parts, while
Fenchel equality on the optimal graph and the second moments control
their positive parts.

On the optimal graph, $T_R=T$ when $|T|\leqslant R$, so the moment correction
vanishes on this set:
\[
\int|\nabla\phi_{R,t}-T|^2\,\mathrm d\rho
\leqslant\frac2t A_{R,t}+4\tau_\mu(R)^2.
\]
Combining this with \eqref{eq:tailgraph1} gives
\[
\int|T(x)-y|^2\,\mathrm d\sigma
\leqslant\frac4t\mathcal G(\sigma)+\frac8t A_{R,t}+18\tau_\mu(R)^2.
\]
Choose the coupling from Lemma~\ref{lem:gap-distance} and use
\eqref{eq:truncated-bias}. We bound the first two terms as
in \eqref{eq:balance}. The same choice of $t$, followed by
$\sqrt{a+b}\leqslant\sqrt a+\sqrt b$ and the target-coupling triangle
inequality, proves \eqref{eq:tailestimate}.

Fix $p\in(1,\infty]$ and its conjugate $p'$. We construct one source
whose mass is concentrated on pairs of thin rectangles escaping to
infinity. For integers $n\geqslant N$, with $N$ sufficiently large, set
\begin{equation}\label{eq:sharp-scales}
\begin{gathered}
m_n=e^{-n^2},\qquad b_n=n^{2(p'-1)}m_n^{p'},\qquad
\ell_n=n^2b_n,\\
L_n=\frac1{n\sqrt{m_n}},\qquad c_n=1-2^{-n}.
\end{gathered}
\end{equation}
Let $u_n^\pm$ be the uniform probability measures on the rectangles
\[
Q_n^\pm=
\biggl[c_n\pm\ell_n-\frac{\ell_n}{4},
      c_n\pm\ell_n+\frac{\ell_n}{4}\biggr]
\times[\mp L_n-1,\mp L_n+1],
\]
and let $u_0$ be uniform on $[-2,-1]\times[-1,1]$. Define
\begin{align*}
\rho&=\biggl(1-\sum_{n\geqslant N}m_n\biggr)u_0
       +\sum_{n\geqslant N}\frac{m_n}{2}(u_n^++u_n^-),\\
\rho_0&=\biggl(1-\sum_{n\geqslant N}b_n\biggr)u_0
       +\sum_{n\geqslant N}\frac{b_n}{2}(u_n^++u_n^-).
\end{align*}
These are probabilities on pairwise disjoint rectangles. Since
$b_n\leqslant m_n$ and $m_nL_n^2=n^{-2}$, both measures have finite second
moments. Each $Q_n^\pm$ has area $\ell_n$ and perimeter $4+\ell_n$,
so the density $f_0$ of $\rho_0$ satisfies
\[
|Df_0|(\mathds R^2)\leqslant C+\sum_{n\geqslant N}
       b_n\frac{4+\ell_n}{\ell_n}<\infty.
\]
For $p<\infty$, the density $h=\,\mathrm d\rho/\,\mathrm d\rho_0$ obeys
\[
\int h^p\,\mathrm d\rho_0
=C_p+\sum_{n\geqslant N}\frac{m_n^p}{b_n^{p-1}}
=C_p+\sum_{n\geqslant N}\frac1{n^2}<\infty.
\]
For $p=\infty$, $b_n=m_n$, so $\rho=\rho_0$ and $h=1$.
The density is then bounded, since its value on $Q_n^\pm$ is
$m_n/(2\ell_n)=1/(2n^2)$.

Put $s_n=\sum_{j>n}m_j$ and $r_n=m_n/2+s_n$, so that
$r_n\sim m_n/2$ and $s_n<m_n/4$. The half-space $\{x_1>c_n\}$
contains $Q_n^+$ and all rectangles with index $j>n$, hence has
$\rho$-mass $r_n$. Set
\[
\eta_n=\frac{4\ell_n}{L_n}
       =4n^{2p'+1}m_n^{p'+1/2},\qquad
v_n=\frac{e_1+\eta_ne_2}{\sqrt{1+\eta_n^2}}.
\]
Under the projection $F_n(x)=x_1+\eta_nx_2$, the two rectangles
exchange order:
\[
F_n(Q_n^+)\subset
\biggl[c_n-\frac72\ell_n,c_n-\frac52\ell_n\biggr],\qquad
F_n(Q_n^-)\subset
\biggl[c_n+\frac52\ell_n,c_n+\frac72\ell_n\biggr].
\]
Their centers project to $c_n\mp3\ell_n$ and their projected
half-width is at most $\ell_n/4+\eta_n\leqslant\ell_n/2$.
Every rectangle with index $j<n$ projects below
$c_n-4\ell_n$. For $N$ large, the right edges of the rectangles
increase with their index and $L_{n-1}+1\leqslant L_n$, so their projections
are at most
\[
c_{n-1}+\frac54\ell_{n-1}+4\ell_n<c_n-4\ell_n.
\]
The last inequality follows from $\ell_n\leqslant n^2e^{-n^2}$ and
$c_n-c_{n-1}=2^{-n}$. The base rectangle satisfies the same bound
directly.

Choose $t_n$ so that $\rho(F_n>t_n)=r_n$; the projected distribution
has no atoms. The two current rectangles lie above all earlier ones
and have mass $m_n>r_n$. Since the later rectangles have total mass
$s_n$, the level set must take at least $r_n-s_n=m_n/2$ from the current
pair, starting with $Q_n^-$. It therefore contains all of $Q_n^-$ and
at most mass $s_n$ from $Q_n^+$.

For any amplitude $a_n>0$, consider the two-point targets
\[
\mu_n=(1-r_n)\delta_0+r_n\delta_{a_ne_1},\qquad
\nu_n=(1-r_n)\delta_0+r_n\delta_{a_nv_n}.
\]
Their Brenier maps are
\[
T_{\mu_n}=a_ne_1\mathbf 1_{\{x_1>c_n\}},\qquad
T_{\nu_n}=a_nv_n\mathbf 1_{\{F_n>t_n\}},
\]
as gradients of scaled positive parts of affine functions.
On mass at least $m_n-s_n$, exactly one of these maps vanishes.
Matching corresponding target atoms, we obtain
\begin{equation}\label{eq:sharp-rates}
\begin{aligned}
\lVert T_{\mu_n}-T_{\nu_n}\rVert_2&\geqslant a_n\sqrt{m_n-s_n},\\
\delta_n&\leqslant a_n\sqrt{r_n}\,|e_1-v_n|
\leqslant C a_n n^{2p'+1}m_n^{p'+1}.
\end{aligned}
\end{equation}
Taking $a_n=1$ gives targets in the unit ball. Since $m_n=e^{-n^2}$,
\eqref{eq:sharp-rates} rules out every exponent larger than
$1/[2(p'+1)]=\alpha_p$.

For $q>2$, take $a_n=r_n^{-1/q}$. Both targets then
have $q$-moment exactly one, while
\[
\lVert T_{\mu_n}-T_{\nu_n}\rVert_2\geqslant c m_n^{1/2-1/q},\qquad
\delta_n\leqslant Cn^{2p'+1}m_n^{p'+1-1/q}.
\]
Every exponent larger than
$(1/2-1/q)/(p'+1-1/q)=\beta_{p,q}$ is therefore impossible.
