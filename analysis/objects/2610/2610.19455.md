<!-- enpdefinition-1: body_md -->
\label{ENP}
	We say that a bounded domain $\Omega \subset \mathbb R^n$ satisfies the
	\emph{Enhanced Neumann P\'olya inequality} (the \emph{\ref{ENP} inequality} for short)
	if there exist positive constants $C_1(\Omega)$ and
	$C_2(\Omega)$ such that 
	\[
	\mathcal{N}_\Omega^N(\lambda) \ge
	L_{0,n}|\Omega|\lambda^{\frac n2}
	+ 	\max\left( 	C_1(\Omega)\lambda^{\frac{n-1}2}, C_2(\Omega) \right), \qquad \forall \lambda>0.
	\]
<!-- end enpdefinition-1 -->
<!-- theorem-1: body_md -->
\label{general}
	Let $\Omega\subset\mathbb{R}^n$ be a bounded domain satisfying the \ref{ENP} inequality, and let $U\subset\mathbb{R}^n$ be a bounded domain with smooth boundary. Then there exists a constant $a_0>0$, depending only on $\Omega$ and $U$, such that, for every $a\in(0,a_0)$ and every $x\in\mathbb{R}^n$ satisfying $\overline{x+aU}\subset\Omega$,  
	the domain 
	\[
	\Omega'=\Omega\setminus\overline{x+aU}
	\]
	also satisfies the \ref{ENP} inequality. In particular, the Neumann eigenvalues of $\Omega'$ satisfy P\'olya's conjecture.
<!-- end theorem-1 -->
<!-- remark--1: body_md -->
Theorem \ref{general} admits a Dirichlet analogue. More precisely, we say that a bounded domain $\Omega$ satisfies the \emph{EDP inequality} if  there exists a constant $C(\Omega)>0$ such that 
	\begin{equation}\label{AssDiri}
	\mathcal{N}_\Omega^D(\lambda) \le
	L_{0,n}|\Omega|\lambda^{\frac n2}
	\left(1-C(\Omega)\lambda^{-\frac12}\right)_+, \qquad  \forall \lambda>0.
	\end{equation}
	By an argument essentially identical to that used in \cite[Theorem 1.9]{JL25}, one can show that the EDP inequality is also stable under the removal of sufficiently small holes. More precisely, for any $U$, there exists a constant $a_0>0$, depending only on $\Omega$ and $U$, such that, for every $a\in(0,a_0)$ and every $x\in\mathbb R^n$ satisfying	$x+aU\subset\Omega$, 
	the domain 	$\Omega\setminus\overline{(x+aU)}	$  also satisfies the EDP inequality.  We omit the details and focus only on the Neumann case in this paper.
<!-- end remark--1 -->
<!-- theorem-2: body_md -->
\label{mthm}
    Let $\Omega\subset \mathbb{R}^n$ be a bounded domain satisfying the \ref{ENP} inequality. For any $M\in\mathbb{Z}_+$, let $\{Q_i\}_{i=1}^M$ be a family of pairwise disjoint open cubes whose closures are contained in $\Omega$. Then there exist positive constants $A(M,\Omega)$ and $V(M,\Omega)$ such that, if
    \[
    \mathcal{A}_{\mathcal{Q}}:=\sum_{k=1}^M \ell(Q_k)^{n-1}<A(M,\Omega),\qquad
    \mathcal{V}_{\mathcal{Q}}:=\sum_{k=1}^M \ell(Q_k)^n<V(M,\Omega),
    \]
    the Neumann Laplacian eigenvalues of
    \[\Omega'=\Omega\setminus \overline{\cup_{k=1}^M Q_k}\]
    satisfy P\'olya's conjecture.
<!-- end theorem-2 -->

	A bounded domain $\Omega\subset\mathbb{R}^n$ is called a \emph{strip-tiling domain} if there exists a constant $w>0$ such that, by translations and rotations in the first $n-1$ directions,
$\Omega$ tiles $\mathbb{R}^{n-1}\times (0,w)$.

<!-- proposition-1: body_md -->
\label{strip}
	Every strip-tiling domain satisfies the \ref{ENP} inequality. Moreover, for a strip-tiling domain with  $w=\pi$,  the constants in the \ref{ENP} inequality can be chosen as 
	\[
	C_1(\Omega)=\frac{\left(4-\pi\right)L_{0,n-1}}{4\pi}|\Omega|, \qquad 
	C_2(\Omega) = 	\min\left( 	\frac12, \frac{\left(4-\pi\right)|\Omega|^{\frac1n}}{2\sqrt\pi}
	L_{0,n-1} 	\right).
	\]
<!-- end proposition-1 -->
<!-- corollary-1: body_md -->
\label{cor:strip}
	Let $\Omega$ be a strip-tiling domain tiling
	$\mathbb{R}^{n-1}\times(0,\pi)$.
	For any $M\in\mathbb{Z}_+$, let
	$\{Q_i\}_{i=1}^M$
	be a family of pairwise disjoint open cubes whose closures are contained in $\Omega$, 
	such that 
	\[
	\mathcal{A}_{\mathcal{Q}} \le \frac{(4-\pi)|\Omega|\pi^{n-2}L_{0,n-1}}{16n}.
	\] 
	Then every Neumann eigenvalue of $\Omega'=\Omega\setminus \overline{\cup_{k=1}^M Q_k}$ with 
	\[
	\mu_k(\Omega') >  \left( \frac{8\pi M}{(4-\pi)|\Omega|L_{0,n-1}} \right)^{\frac{2}{n-1}}
	\]
	satisfies P\'olya's inequality.
<!-- end corollary-1 -->
<!-- remark--2: body_md -->
The lower bound on $\mu_k(\Omega')$ is needed here because the proof of Theorem~\ref{mthm} relies on an explicit $L^\infty$ estimate for Neumann eigenfunctions (see \eqref{Linftybd}). While such an estimate is available in general, no explicit upper bound for the corresponding constant is currently known for strip-tiling domains. Thus Corollary \ref{cor:strip} reduces P\'olya's conjecture to verifying P\'olya's inequality for the finitely many eigenvalues below the explicit threshold above.

<!-- proof-1: body_md -->
Decomposing a nonnegative lattice point according to its set of nonzero coordinates gives
	\begin{equation}\label{eq:decomposition}
		\mathcal{N}_{I_n}^N(\lambda)
		=
		\sum_{j=0}^n\binom{n}{j}A_j(\sqrt{\lambda}),
	\end{equation}
	where $A_0(r):=1$ and for $j\ge1$, 
	\[
	A_j(r)
	:=	\#\left\{m\in\mathbb Z_{>0}^j:\ |m|<r\right\}
	\le  \frac{\omega_j}{2^j}r^j.
	\]
	Here $\omega_j$ denotes the volume of the
	unit ball in $\mathbb R^j$.  
	
	First suppose  $\lambda \ge 4$. 
	                  
	Then $\frac{{\sqrt \lambda}^j}{2^j}\le	2^{1-n}\lambda^{\frac{n-1}2}$ for  $0\le j\le n-1$, and thus 
	\[
	\mathcal{N}_{I_n}^N(\lambda)
	\le 	\sum_{j=0}^{n}
	\binom{n}{j}\frac{\omega_j}{2^j}\sqrt{\lambda}^j
	\le 	\frac{\omega_n}{2^n}\lambda^{\frac n2}+
	2^{1-n}\lambda^{\frac{n-1}2}
	\sum_{j=0}^{n-1}\binom{n}{j}\omega_j.
	\]	 
	So the desired inequality follows from the fact 
	$\frac{\omega_n}{2^n} = L_{0,n}\pi^n$ 
	and 
	\[
	2^{1-n} \sum_{j=0}^{n-1}\binom{n}{j}\omega_j
	\le 2^{1-n} \left( 1+2\sum_{j=1}^{n-1}j\binom{n}{j}
	\right) 
	=\left( 2n+(1-2n)2^{1-n} \right) \le 2n,
	\]
	where we used the trivial identity $\omega_j \le 2j$ and the combinatorial identity 	
	\[ \sum_{j=1}^{n-1}j\binom{n}{j}=\sum_{j=1}^{n-1}n\binom{n-1}{j-1}=n(2^{n-1}-1).\] 
	
	It remains to consider $0<\lambda<4$. Since  
	\[
	\mathcal{N}_{I_n}^N(\lambda)
	=\begin{cases}
		1, & 0<\lambda \le 1,\\
		n+1, & 1 < \lambda \le 2, \\
		\frac{n^2+n+2}{2}, & 2 < \lambda \le 3, \\
		\frac{n^3+5n+6}{6}, & 3 < \lambda \le 4,
	\end{cases}
	\]
	the desired inequality holds for $0<\lambda \le 1$ trivially, and it is not hard to verify for $\lambda \in (k,k+1]$, where $k=1,2,3$, that

	If $0<\lambda\le1$, the desired inequality holds trivially. 
	To prove that the same inequality holds for all $\lambda>1$, we consider, on each interval $(k^2,(k+1)^2]$, $k\ge1$, the function
	\[
	f_k(\lambda)
	:=\sum_{i=0}^{k}\sqrt{\lambda-i^2}
	-\frac{\pi}{4}\lambda-\left(1-\frac{\pi}{4}\right)\sqrt\lambda =\frac{\pi}4 \sqrt \lambda+\sum_{i=1}^{k}\sqrt{\lambda-i^2}
	-\frac{\pi}{4}\lambda.
	\]
	It is concave and thus its minimum is attained at an endpoint. So it
	suffices to verify $f_k(k^2) \ge 0$ and $f_k((k+1)^2) \ge 0$ for all $k$, or equivalently, to prove 
	\begin{equation}\label{eq:square-points}
		\sum_{i=0}^{k}
		\sqrt{k^2-i^2}
		-\frac{\pi}{4}k^2 \ge \left(1-\frac{\pi}{4}\right) k,
		\qquad k\in\mathbb N.
	\end{equation}
	
	Write
	$g_k(x)=\sqrt{k^2-x^2}$ on $[0,k]$.  Since
	$\int_0^k g_k(x)\,dx=\pi k^2/4$,                                 
	\begin{align*}
		\sum_{i=0}^{k}
		\sqrt{k^2-i^2}
		-\frac{\pi}{4}k^2
		&=\sum_{i=0}^{k-1}
		\left(g_k(i)-\int_i^{i+1}g_k(x)\,dx\right) \\
		&\ge \frac12\sum_{i=1}^{k-1}
		\frac{i}{\sqrt{k^2-i^2}} \\
		&\ge \frac12\int_0^{k-1}
		\frac{x}{\sqrt{k^2-x^2}}\,dx
		=\frac12\left(k-\sqrt{2k-1}\right).
	\end{align*}
    This gives \eqref{eq:square-points} for $k\ge 6$.

	p_n(x)=1-\frac{n^2}{x^2},
	\qquad
	S_n(x)=J_n(x)^2+\frac{J_n'(x)^2}{p_n(x)}.
	\]
	Using the Bessel equation, one obtains for \(x>n\),
	\[
	S_n'(x)
	=
	-\frac{2J_n'(x)^2}{x\,p_n(x)}
	-\frac{p_n'(x)J_n'(x)^2}{p_n(x)^2}
	\leq0.
	\]
	Thus the absolute values of the successive extrema of \(J_n\) are
	nonincreasing.  Since \(j'_{n,1}\) is the first positive critical point and $|J_n|$ is even,
	\[
	M_n(a)=J_n(j'_{n,1})=\sup_{x\in\mathbb{R}}|J_n(x)|.
	\]
	
	By \eqref{eq:BesselFunctionInt}, $J_n$ is entire. Moreover, it is  of exponential type at most $1$ since 
	\[
	|J_n(z)|
	\leq
	\frac{1}{\pi}\int_0^\pi e^{|y||\sin\theta|}\,d\theta
	\leq e^{|y|}
	\leq e^{|z|}.
	\]
	Hence by Bernstein's inequality
	\cite{Bernstein1923}, 
	\[
	\|J_n'\|_{L^\infty(\mathbb R)}
	\leq
	\|J_n\|_{L^\infty(\mathbb R)}
	=M_n(a).
	\]
	As a consequence, for $x\in[j'_{n,1}-1,j'_{n,1}]$,  
	\[
	J_n(x)
	\geq J_n(j'_{n,1})-(j'_{n,1}-x)M_n(a)
	=(1-j'_{n,1}+x)M_n(a).
	\]
	Since \(j'_{n,1}\geq n\geq1\) and \(a\geq j'_{n,1}\), it follows that
	\begin{align*}
		\int_0^a\! xJ_n(x)^2\,dx
		\geq
		M_n(a)^2\int_{j'_{n,1}\!-\!1}^{j'_{n,1}}\! x(1\!-\!j'_{n,1}\!+\!x)^2\,dx=
		\frac{4j'_{n,1}\!-\!1}{12}M_n(a)^2\geq\frac{n}{4}M_n(a)^2.\!
	\end{align*}
