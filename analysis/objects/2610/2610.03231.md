<!-- theorem-1: body_md -->
\label{thm:main}
Let $n\ge3$ and let $\mathcal C\subset\mathbb R^n$ be a compact
$C^\infty$ nondegenerate curve. For every $\varepsilon>0$, $Q\ge2$,
and $0<\delta\le1$,
\begin{equation}\label{eq:intro-refined-bound}
\mathcal R_{\mathcal C}(\delta,Q)
\lesssim_{\varepsilon,\mathcal C}
\delta^{n-1}Q^2+Q^\varepsilon
\left(Q+\sum_{k=1}^{n-1}
\delta^{k^2/(k+1)}Q^{(2k+1)/(k+1)}\right).
\end{equation}
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
Let $n\ge3$ and let $\mathcal C$ be as in Theorem~\ref{thm:main}.
Set $D_n=n-1+\left\lfloor\frac{(n-1)^2}{4}\right\rfloor$.
For every $\varepsilon>0$, if $Q\ge2$ is sufficiently large depending on
$\varepsilon$ and $\mathcal C$, and
$Q^{-1/D_n+\varepsilon}\leq\delta\leq1$, then
\begin{equation}
 c\delta^{n-1}Q^2
 \leq \mathcal R_{\mathcal C}(\delta,Q) \leq C\delta^{n-1}Q^2,
 \label{eq:intro-volume-range}
\end{equation}
where $c,C>0$ depend only on $\varepsilon$ and $\mathcal C$.
<!-- end corollary-1 -->
<!-- proposition-1: body_md -->
\label{cor:intro-moment-count}
Let $n\ge3$ and let $\Gamma_n$ be the moment curve defined above.
For every $Q\ge2$ and $0<\delta\le1$,
\[
\delta^{n-1}Q^2+
\sum_{k=0}^{n-1}\delta^{k^2/(k+1)}Q^{(2k+1)/(k+1)}
\lesssim_n \mathcal R_{\Gamma_n}(\delta,Q).
\]
<!-- end proposition-1 -->
<!-- remark-1: body_md -->
{\rm
    The idea of transferring the original counting problem to a dual counting problem was first used by Huang \cite{Huang2020} to count rationals near hypersurfaces. 
    }
<!-- end remark-1 -->
<!-- definition-1: body_md -->
Fix $r\ge 1$. Let $\boldsymbol{\lambda}=(\lambda_1,\dots,\lambda_{n-1})$ with $\lambda_i\in(0,1]$ and let $A\subset \{1,\dots,n-1\}$. We say $(\boldsymbol{\lambda},A)$ is admissible at scale $r$ in dimension $n$ if $r^{-1}=\lambda_1^n\lambda_2^{n-1}\dots\lambda_{n-1}^2$, and $i\in A$ if and only if $\lambda_i<1$.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
Given $\boldsymbol{\lambda}=(\lambda_1,\dots,\lambda_{n-1})$, set $\rho_0=1$ and, for $1\le k\le n-1$, define
    \begin{equation}\label{defrho}
        \rho_k=\lambda_1^k\lambda_2^{k-1}\dots\lambda_k.
    \end{equation}
    We also set $\rho_n=\lambda_1^n\dots\lambda_{n-1}^2=r^{-1}$.
<!-- end definition-2 -->
<!-- remark-2: body_md -->
{\rm
    It is easy to see that
\begin{equation}\label{rho n-1}
    \rho_n\rho_{n-2}=\rho_{n-1}^2 \;\;\;\; \mathrm{and}  \qquad \rho_{i}\rho_{i+2}\le \rho_{i+1}^2 \quad (0\le i\le n-3). 
\end{equation} 
    }
<!-- end remark-2 -->
<!-- definition-3: body_md -->
Let $(\boldsymbol{\lambda},A)$ be admissible at scale $r\ge 1$. Define the $(\boldsymbol{\lambda},A)$-plank as follows
    \begin{equation}
        P[\boldsymbol{\lambda},A]:=\Biggl\{(x_1,\dots,x_{n+1})\in\mathbb R^{n+1}:
\begin{array}{l}
|x_i|\lesssim r\rho_{i-1}
 \quad (1\le i\le n-1,\ i\notin A),\\[1mm]
|x_i|\sim r\rho_{i-1}
 \quad (i\in A),\\[1mm]
|x_n|\lesssim r\rho_{n-1},\quad |x_{n+1}|\lesssim r\rho_n=1
\end{array}\Biggr\}.
    \end{equation}
<!-- end definition-3 -->
<!-- remark-3: body_md -->
{\rm
    For $i\in A$, the condition $|x_i|\sim r\rho_{i-1}$ reads as either $x_i\in [-Cr\rho_{i-1},-cr\rho_{i-1}]$ or $x_i\in [cr\rho_{i-1},Cr\rho_{i-1}]$ for some constants $0<c<C$.
    Thus, $P[\boldsymbol{\lambda},A](t)$ consists of $2^{|A|}$ boxes, each with dimensions $r\times r\rho_1\times \dots\times r\rho_{n-1}\times 1$. For our purpose, it is harmless to consider a single piece of $P[\boldsymbol{\lambda},A]$ which satisfies $x_i>0$ for $i\in A$. The other pieces can be handled similarly.
    When only the dimensions of $P[\boldsymbol{\lambda},A](t)$ are relevant, we refer to it as a $\boldsymbol{\lambda}$-plank.
    }
<!-- end remark-3 -->
<!-- lemma-1: body_md -->
\label{lemcompare}
Let $h=\prod_{i=1}^{n-1}\lambda_i$.
    If $|s-s'|\lesssim h$, then $P[\boldsymbol{\lambda},A](s)$ and $P[\boldsymbol{\lambda},A](s')$ are comparable. If $|s-s'|\ge C h$, then $P[\boldsymbol{\lambda},A](s)$ and $P[\boldsymbol{\lambda},A](s')$ are disjoint.
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
Fix $r\ge 1$. Let $\mathcal I(h)$ be a $h$-net of $I$.
For $x\in \mathbb{R}^{n+1}$ with $|x|\sim r$, we have 
    \begin{equation}\label{gmo0}
        |\widehat \mu_\Gamma(x)|\lesssim \sum_{\begin{subarray}{c}
          (\boldsymbol{\lambda},A) \\
          \text{~admissible at~}r
\end{subarray}}h\sum_{t\in \mathcal I(h)}\mathbf{1}_{P[\boldsymbol{\lambda},A](t)}(x).
    \end{equation}
Here, the first sum runs over all pairs $(\boldsymbol{\lambda},A)$ admissible at scale $r$, with each $\lambda_i$ dyadic.
<!-- end lemma-2 -->
<!-- definition-4: body_md -->
Let $1\le Q_*$ and $0<\delta_*\le Q_*$.
    Suppose $(\boldsymbol{\lambda},A)$ is admissible at scale $\delta_*^{-1}Q_*$. Define the anisotropic counting function
\begin{equation}\label{defanisotropic}
       N_n(\delta_*,Q_*;\boldsymbol{\lambda}):=\#\left\{ x\in Q_*^{-1}\mathbb{Z}^{n+1}:x\in \bigcup_{t\in \mathcal I(h)} p[\boldsymbol{\lambda},A](t)\right\}.
    \end{equation}
<!-- end definition-4 -->
<!-- remark-4: body_md -->
{\rm We give two remarks.

    \begin{itemize}
        \item When $n=2$ and $\delta_*\le 1\le Q_*$, let $\lambda_1=(\delta_* Q_*^{-1})^{1/2}$. Then the anisotropic counting function $N_2(\delta_*,Q_*;\lambda_1)$ agrees with the isotropic counting function $N_2(\delta_*,Q_*)$.

        \item It may happen that $\delta_*\ge 1$. In this case, the counting problem is trivial, since the shortest edge of the $\boldsymbol{\lambda}$-plank is larger than $Q_*^{-1}$, the spacing of the lattice $Q_*^{-1}\mathbb{Z}^{n+1}$. We have the trivial volume bound:
    \begin{equation}\label{volumebound}
        N_n(\delta_*,Q_*;\boldsymbol{\lambda})\lesssim Q_*^{n+1} \left|\bigcup_{t\in \mathcal I(h)} p[\boldsymbol{\lambda},A](t)\right|\lesssim Q_*^{n+1}h^{-1}\prod_{i=1}^n\rho_i. 
    \end{equation}
    \end{itemize}}
<!-- end remark-4 -->
<!-- proposition-2: body_md -->
\label{input1}Let $0<\delta\le 1\le Q$. Then,
    \begin{equation}
        N_n(\delta,Q)\lesssim \delta^{n-1}Q^2+ \delta^{n-1}Q^2  \sum_{\begin{subarray}{c}
          Q\le r\le \delta^{-1}Q \\
          \text{dyadic}
\end{subarray}}\sum_{\begin{subarray}{c}
          (\boldsymbol{\lambda},A) \\
          \text{~admissible at~}r
\end{subarray}}h\cdot N_n(Q^{-1},\frac rQ;\boldsymbol{\lambda}).
    \end{equation}
<!-- end proposition-2 -->
<!-- proposition-3: body_md -->
\label{input2} Fix $m\ge 3$. Let $0<\delta_*\le 1\le Q_*$. 
Let $\boldsymbol{\lambda}=(\lambda_1,\dots,\lambda_{m-1})$ be admissible at scale $\delta_*^{-1}Q_*$. Let $\tau_\circ=\min\{Q_*^{-1}, \prod_{i=1}^{m-2}\lambda_i^{m+1-i}\}$. Then for any $\tau\in [\delta_*Q_*^{-1},\tau_\circ]$, we have
    \begin{equation}
\begin{split}
N_m&(\delta_*,Q_*;\boldsymbol{\lambda})\lesssim \frac{\tau}{\tau_\circ}N_{m-1}(\prod_{i=1}^{m-2}\lambda_i^{m-i}Q_*,Q_*;\lambda_1,\dots,\lambda_{m-2})\\
&+(\prod_{i=1}^{m-2}\rho_i)\cdot(\rho_{m-2}\tau)^{1/2}\tau Q_*^{m+1}N_2(\rho_{m-2}^{-1}Q_*^{-1},\tau^{-1}Q_*^{-1})\prod_{i=1}^{m-3}(1+\rho_{i}^{-1}Q_*^{-1}).
\end{split}
    \end{equation}
<!-- end proposition-3 -->
<!-- proposition-4: body_md -->
\label{input0}
    Let $\delta_m=Q_*\prod_{i=1}^{m-1}\lambda_i^{m+1-i}$ and $\delta_{m-1}=Q_*\prod_{i=1}^{m-2}\lambda_i^{m-i}$. Then
    \[ N_m(\delta_m,Q_*;\lambda_1,\dots,\lambda_{m-1})\lesssim (1+\delta_m)\cdot N_{m-1}(\delta_{m-1},Q_*;\lambda_1,\dots,\lambda_{m-2}). \]
    The additional factor $1+\delta_m$ counts the choice of the last component of the lattice points when their first $m$ components are fixed.
<!-- end proposition-4 -->
<!-- proposition-5: body_md -->
\label{input3}
    Let $0<\delta\le 1\le Q$. Then for any $\epsilon>0$,
    \begin{equation}
        N_2(\delta,Q)\lesssim_\epsilon \delta Q^2+Q^{1+\epsilon}.
    \end{equation}
<!-- end proposition-5 -->
<!-- theorem-2: body_md -->
\label{thm:section-two-main}
Let $0<\delta\leq1\leq Q$. Then for every $\epsilon>0$,
\begin{equation}\label{eq:section-two-main}
 N_n(\delta,Q)
 \lesssim_\epsilon
 \delta^{n-1}Q^2+Q^\epsilon\left(
 Q+\sum_{k=1}^{n-1}
 \delta^{\,k^2/(k+1)}Q^{(2k+1)/(k+1)}\right).
\end{equation}
<!-- end theorem-2 -->
<!-- proof-1: body_md -->
\textbf{Step 1. Derivation of \eqref{eq:section-two-normalized-recurrence}.}
We first treat $Q^{-1}\leq\delta\leq1$.  In
Proposition~\ref{input1}, write
\[
 M:=\frac rQ.
\]
Then the range of $M$ is
\[ 1\leq M\leq\delta^{-1}\leq Q.\]
Fix one tuple
$\boldsymbol{\lambda}=(\lambda_1,\ldots,\lambda_{n-1})$ that is admissible at scale $r$, and put
$h:=\prod_{i=1}^{n-1}\lambda_i$.  Admissibility gives
\begin{equation}\label{eq:section-two-full-relation}
 \prod_{i=1}^{n-1}\lambda_i^{\,n+1-i}=r^{-1}=(QM)^{-1}.
\end{equation}

To make the computation simple, we introduce some new parameters. Let
\begin{equation}\label{eq:section-two-Lj}
 \ell_0:=M,
 \qquad
 \ell_j:=M\prod_{i=1}^{j}\lambda_i^{\,j+1-i}
 \quad(1\leq j\leq n-1),
\end{equation}
and, for $2\leq m\leq n$, set
\begin{equation}\label{eq:section-two-deltam}
 \delta_m:=M\prod_{i=1}^{m-1}\lambda_i^{\,m+1-i}
      =\frac{\ell_{m-1}^2}{\ell_{m-2}}.
\end{equation}
Hence the shortest side of the $m$-dimensional $(\lambda_1,\dots,\lambda_{m-1})$-plank is
$\delta_mM^{-1}$, and \eqref{eq:section-two-full-relation} says that
$\delta_n=Q^{-1}$.  Write
\[
 \mathbf N_m:=N_m(\delta_m,M;\lambda_1,\ldots,\lambda_{m-1}).
\]
Start from $m=n$.
If $\delta_m\le 1$, we use Proposition \ref{input2}. If $\delta_m>1$, the iteration stops and we use the volume bound \eqref{volumebound}.   Introduce also the volume scale
\begin{equation}\label{eq:section-two-volume}
 \mathcal V_m
 :=\left(\prod_{j=0}^{m-2}\ell_j\right)\ell_{m-1}^2.
\end{equation}
This is the volume of the union of the  $(\lambda_1,\dots,\lambda_{m-1})$-planks multiplied by the
lattice density $M^{m+1}$. The same reasoning as in \eqref{volumebound} shows that if $\delta_m>1$, then
\begin{equation}\label{volumeestimate}
    \mathbf N_m\lesssim \mathcal V_m.
\end{equation} 

We next rewrite Proposition~\ref{input2} in these parameters. We use $(\delta_m,M)$ in place of the $(\delta_*,Q_*)$ from Proposition \ref{input2}. For a fixed
$m\geq3$, put
\begin{equation}\label{eq:section-two-local-parameters}
 \ell:=\ell_{m-2},\qquad \ell_-:=\ell_{m-3},\qquad
 S:=\prod_{j=0}^{m-3}\ell_j,\qquad
 g:=\prod_{i=1}^{m-2}\lambda_i=\frac \ell{\ell_-},\qquad
 w:=\ell g^2=\frac{\ell^3}{\ell_-^2}.
\end{equation}
Since $\rho_i=\ell_i/M$ for $i\leq m-2$, the cutoff in
Proposition~\ref{input2} is
\[
 \tau_\circ=M^{-1}\min\{1,w\}.
\]
If $\tau$ denotes the parameter in Proposition~\ref{input2}, put
$\sigma:=M\tau$.  Thus the condition $\tau\in [\delta_* Q_*^{-1},\tau_\circ]$ can be rewritten as
\begin{equation}\label{eq:section-two-sigma-range}
 \delta_m\leq\sigma\leq\min\{1,w\}.
\end{equation}


The projected count $N_{m-1}(\prod_{i=1}^{m-2}\lambda_i^{m-i}Q_*,Q_*;\lambda_1,\dots,\lambda_{m-2})$ in Proposition~\ref{input2} is $\mathbf N_{m-1}$, and
\[
 \frac{\tau}{\tau_\circ}
 =\frac{\sigma}{\min\{1,w\}}
 \sim\sigma(1+w^{-1}).
\]
Moreover,
\[
 \prod_{i=1}^{m-2}\rho_i=\frac{S\ell}{M^{m-1}},
 \qquad
 (\rho_{m-2}\tau)^{1/2}=\frac{(\ell\sigma)^{1/2}}M,
\]
and the planar count $N_2(\rho_{m-2}^{-1}Q_*^{-1},\tau^{-1}Q_*^{-1})$ appearing there is
\[
 N_2(\ell^{-1},\sigma^{-1}).
\]
The full coefficient multiplying this planar count is therefore
\[
 \left(\prod_{i=1}^{m-2}\rho_i\right)
 (\rho_{m-2}\tau)^{1/2}\tau M^{m+1}
 =S\ell^{3/2}\sigma^{3/2}.
\]
When $\ell>1$, the remaining fiber factor is harmless: indeed
$\ell_0\geq \ell_1\geq\cdots\geq \ell_{m-2}=\ell$, and hence
\[
 \prod_{i=1}^{m-3}(1+\rho_i^{-1}M^{-1})
 =\prod_{i=1}^{m-3}(1+\ell_i^{-1})\lesssim_m1.
\]
Since $\ell>1$ and $\sigma\leq1$, the hypotheses
$\ell^{-1}\leq1\leq\sigma^{-1}$ of Proposition~\ref{input3} hold, and it gives
\[
 N_2(\ell^{-1},\sigma^{-1})
 \lesssim_\epsilon \ell^{-1}\sigma^{-2}+\sigma^{-1-\epsilon}.
\]
In the rest of this subsection, we write $X\lessapprox Y$ if, for every $\eta>0$,
$X\lesssim_\eta Q^\eta Y$. Notice that $\sigma\ge \delta_m\ge \delta_n=Q^{-1}$. Hence $\sigma^{-\epsilon}$ is absorbed by $Q^\epsilon$.
Consequently Propositions~\ref{input2} and \ref{input3} imply that if $\ell>1$ and \eqref{eq:section-two-sigma-range} hold, then  
\begin{equation}\label{eq:section-two-normalized-recurrence}
 \mathbf N_m\lessapprox
 \sigma(1+w^{-1})\mathbf N_{m-1}
 +S\left(\frac \ell\sigma\right)^{1/2}
 +S\ell^{3/2}\sigma^{1/2}.
\end{equation}


\medskip
\noindent\textbf{Step 2. The inductive bound and preliminary estimates.}
We now iterate \eqref{eq:section-two-normalized-recurrence}.  For
$0\leq k\leq n-1$, let
\begin{equation}\label{eq:section-two-Dnk}
 D_{n,k}:=(n-1)(k+1)-k^2=n-1+k(n-1-k),
\end{equation}
and define
\begin{equation}\label{eq:section-two-En}
 E_n(Q^{-1},M)
 :=\max_{0\leq k\leq n-1}
 \left(Q^{-1}M^{D_{n,k}}\right)^{1/(k+1)},
 \qquad
 U:=\frac{E_n(Q^{-1},M)}h.
\end{equation}
We claim that $U$ and the volume scale $\mathcal V_m$ control $\mathbf N_m$:
\begin{equation}\label{eq:section-two-induction}
 \mathbf N_m\lessapprox\mathcal V_m+U,
 \qquad 2\leq m\leq n.
\end{equation}

We will prove this bound by induction on $m$. We first establish the numerical inequalities needed for the induction. From
\eqref{eq:section-two-full-relation} and \eqref{eq:section-two-Lj}, for $0\leq j\leq n-1$,
\[
 Q\ell_j=\prod_{i=1}^j\lambda_i^{-(n-j)}\prod_{i=j+1}^{n-1}\lambda_i^{-(n+1-i)}\leq h^{-(n-j)}.
\]
Fix $0\leq j\leq n-1$ and choose $k=n-j-1$ in
\eqref{eq:section-two-En}. Then 
\[
 D_{n,n-j-1}
 =(n-1)(n-j)-(n-j-1)^2
 =(j+1)(n-j)-1.
\]
Therefore,
\[
 U
 \geq h^{-1}
 \left(Q^{-1}M^{D_{n,n-j-1}}\right)^{1/(n-j)}
 =h^{-1}Q^{-1/(n-j)}M^{j+1-1/(n-j)}.
\]
Moreover, the preceding estimate for $Q\ell_j$ gives
$h^{-1}\geq(Q\ell_j)^{1/(n-j)}$.  Substituting this into the preceding inequality yields
\begin{equation}\label{eq:section-two-basic-U}
 U
 \geq (Q\ell_j)^{1/(n-j)}Q^{-1/(n-j)}
 M^{j+1-1/(n-j)} 
 =M^{j+1}\left(\frac{\ell_j}{M}\right)^{1/(n-j)}.
\end{equation}
In particular,
\begin{equation}\label{eq:section-two-U-consequences}
 U\geq M,\qquad
 U\geq M^{m-3/2}\ell^{1/2},\qquad
 U\geq M^{m-1}\ell_{m-1}.
\end{equation}
Indeed, for $j=m-2$ the exponent in
\eqref{eq:section-two-basic-U} is $1/(n-m+2)\leq1/2$, while $\ell/M\leq1$.
For $j=m-1$, use $1/(n-m+1)\leq1$ and $\ell_{m-1}/M\leq1$.
Since every $\ell_j\leq M$ and $S\leq M^{m-2}$, it follows that
\begin{equation}\label{eq:section-two-SL}
 S\ell\le M^{m-2}M^{1/2}\ell^{1/2} \leq U.
\end{equation}
We shall also use
\begin{equation}\label{eq:section-two-min-bound}
 \min\{S\ell^2,S\ell_-^{2/3}\}
 \leq (S\ell^2)^{1/4}(S\ell_-^{2/3})^{3/4}
 =S(\ell \ell_-)^{1/2}\le M^{m-2}\ell^{1/2}M^{1/2} \leq U
\end{equation}
and
\begin{equation}\label{eq:section-two-volume-over-w}
 \frac{\mathcal V_m}{w}\leq U.
\end{equation}
Indeed,
$\ell_{m-1}\leq \ell^2/\ell_-$ and
$\mathcal V_m=S\ell\,\ell_{m-1}^2$, so
\[
 \frac{\mathcal V_m}{w}
 \leq S\ell_-\ell_{m-1}
 \leq M^{m-1}\ell_{m-1}\leq U.
\]

\medskip
\noindent\textbf{Step 3. Proof of \eqref{eq:section-two-induction} by induction.}
For the case $m=2$ of \eqref{eq:section-two-induction}, if $\delta_2\geq1$ the
elementary volume estimate \eqref{volumeestimate} gives $\mathbf N_2\lesssim\mathcal V_2$.  If
$\delta_2<1$, Proposition~\ref{input3} gives
$\mathbf N_2\lessapprox\delta_2M^2+M=\mathcal V_2+M$, which is bounded by
$\mathcal V_2+U$ because $U\geq M$.

Suppose the claim is known in dimension $m-1$. If $\delta_m\geq1$, the volume estimate \eqref{volumeestimate} gives
$\mathbf N_m\lessapprox\mathcal V_m$.  If $\delta_m<1$, Proposition \ref{input0} gives
$\mathbf N_m\lessapprox(1+\delta_m)\mathbf N_{m-1}\lesssim\mathbf N_{m-1}$. Thus,
when $\mathcal V_{m-1}\leq U$, the induction hypothesis already gives
$\mathbf N_m\lessapprox U$.

It remains only to consider
\begin{equation}\label{eq:section-two-hard-case}
 \delta_m<1,
 \qquad
 \mathcal V_{m-1}=S\ell^2>U.
\end{equation}
The induction hypothesis then gives
$\mathbf N_{m-1}\lessapprox S\ell^2$.  Since $S\ell^2>U\geq S\ell$ (see \eqref{eq:section-two-SL}), we have
$\ell>1$.  Furthermore, \eqref{eq:section-two-min-bound} implies
$S\ell_-^{2/3}\leq U$.  Comparing this with $S\ell^2>U$ gives
$\ell>\ell_-^{1/3}$, and hence
\begin{equation}\label{eq:section-two-admissibility-bound}
 \frac{S\ell_-}{\ell}\leq S\ell_-^{2/3}\leq U.
\end{equation}

Choose
\begin{equation}\label{eq:section-two-sigma-choice}
 T:=\max\{\mathcal V_m,U\},
 \qquad
 \sigma_0:=\frac{S^2\ell}{T^2},
 \qquad
 \sigma=\max\{\delta_m,\sigma_0\}.
\end{equation}
We verify that $\sigma$ satisfies \eqref{eq:section-two-sigma-range} and that each of the three terms on the right-hand side of \eqref{eq:section-two-normalized-recurrence} is $\lessapprox T$.

Because $S\ell\leq U\leq T$ and $\ell>1$,
\[
 \sigma_0=\frac{(S\ell)^2}{T^2\ell}\leq \ell^{-1}<1.
\]
Also, by \eqref{eq:section-two-admissibility-bound},
\[
 \frac{\sigma_0}{w}
 =\left(\frac{S\ell_-}{\ell T}\right)^2\leq1.
\]
Finally,
\[
 \delta_m=w\lambda_{m-1}^2\leq w.
\]
Together with \eqref{eq:section-two-hard-case}, these estimates verify
\eqref{eq:section-two-sigma-range}.

We apply \eqref{eq:section-two-normalized-recurrence}.  The middle term is
at most $T$, since
\[ S\left(\frac{\ell}{\sigma}\right)^{1/2} \le S\left(\frac{\ell}{\sigma_0}\right)^{1/2}=T.\]
For the first term, 
\[
\begin{aligned}
 \sigma(1+w^{-1})\mathbf N_{m-1}
 &\lessapprox(\delta_m+\sigma_0)(1+w^{-1})S\ell^2 \\
 &=\mathcal V_m+\frac{\mathcal V_m}{w}
   +\frac{(S\ell)^3+(S\ell_-^{2/3})^3}{T^2} \\
 &\leq T+U+\frac{2U^3}{T^2}
 \leq4T.
\end{aligned}
\]
Indeed, the second line uses
\[
 \delta_mS\ell^2=\mathcal V_m,
 \qquad
 \sigma_0S\ell^2=\frac{(S\ell)^3}{T^2},
 \qquad
 \frac{(S\ell)^3}{w}=(S\ell_-^{2/3})^3.
\]
The last line follows from $\mathcal V_m\leq T$,
\eqref{eq:section-two-volume-over-w}, the bounds
$S\ell\leq U$ and $S\ell_-^{2/3}\leq U$ obtained above, and $U\leq T$.
Finally,
\[
 \bigl(S\ell^{3/2}\sigma^{1/2}\bigr)^2
 =\max\left\{(S\ell)\mathcal V_m,\frac{(S\ell)^4}{T^2}\right\}
 \leq T^2.
\]
This closes the induction and proves \eqref{eq:section-two-induction}.

\medskip
\noindent\textbf{Step 4. Completion of the proof of Theorem~\ref{thm:section-two-main}.}
At $m=n$, the volume term can be absorbed into $E_n(Q^{-1},M)$.
We first compute it directly from \eqref{eq:section-two-Lj} and
\eqref{eq:section-two-volume}:
\[
\begin{aligned}
 h\mathcal V_n
 &=\left(\prod_{i=1}^{n-1}\lambda_i\right)
   \left(\prod_{j=0}^{n-2}
      \left(M\prod_{i=1}^{j}\lambda_i^{\,j+1-i}\right)\right)
   \left(M\prod_{i=1}^{n-1}\lambda_i^{\,n-i}\right)^2 \\
 &=M^{n+1}\prod_{i=1}^{n-1}
   \lambda_i^{\,1+\sum_{j=i}^{n-2}(j+1-i)+2(n-i)} \\
 &=M^{n+1}\prod_{i=1}^{n-1}
   \lambda_i^{(n+1-i)(n+2-i)/2}.
\end{aligned}
\]
The admissibility relation \eqref{eq:section-two-full-relation} therefore gives
\[
\begin{aligned}
 h\mathcal V_n
 &=M^{n+1}
   \left(\prod_{i=1}^{n-1}\lambda_i^{\,n+1-i}\right)
   \prod_{i=1}^{n-1}\lambda_i^{(n-i)(n+1-i)/2} \\
 &=Q^{-1}M^n
   \prod_{i=1}^{n-1}\lambda_i^{(n-i)(n+1-i)/2}.
\end{aligned}\]
Consequently, using $M\le \delta^{-1}\leq Q$,
\[
 h\mathcal V_n
 \leq Q^{-1}M^n(Q^{-1}M^{-1})^{1/2}  
 \leq Q^{-1}M^{n-1}
 \leq E_n(Q^{-1},M).
\]

Multiplying \eqref{eq:section-two-induction} by $h$ gives the estimate needed
in Proposition~\ref{input1}:
\begin{equation}\label{eq:section-two-inner-bound}
 hN_n(Q^{-1},M;\boldsymbol{\lambda})
 \lessapprox E_n(Q^{-1},M)
 \leq\sum_{k=0}^{n-1}
 Q^{-1/(k+1)}M^{D_{n,k}/(k+1)}.
\end{equation}

It remains to sum over the dyadic radii.  There are only $O((\log(2Q))^{n-1})$
admissible dyadic $\boldsymbol{\lambda}$-tuples at each radius, while the auxiliary set $A$
has $O_n(1)$ possibilities.  Since $r$ is dyadic, $M=r/Q$ ranges over a
dyadic sequence.  As $D_{n,k}>0$,
\[
 \sum_{\substack{1\leq M\leq\delta^{-1}\\M=r/Q,\ r\ {\rm dyadic}}}
 M^{D_{n,k}/(k+1)}
 \lesssim \delta^{-D_{n,k}/(k+1)}.
\]
Substituting \eqref{eq:section-two-inner-bound} into
Proposition~\ref{input1}, we find that the $k$-th contribution is
\begin{align*}
 \delta^{n-1}Q^2Q^{-1/(k+1)}
 \delta^{-D_{n,k}/(k+1)}
 &=\delta^{\,k^2/(k+1)}Q^{(2k+1)/(k+1)}.
\end{align*}
This finishes the proof when $\delta\ge Q^{-1}$.

Finally, if $0<\delta<Q^{-1}$, monotonicity gives
$N_n(\delta,Q)\leq N_n(Q^{-1},Q)$.  The estimate already proved at
$\delta=Q^{-1}$ is $O_\epsilon(Q^{1+\epsilon})$: indeed the exponent of $Q$ in every
$k\geq1$ term is
$(2k+1-k^2)/(k+1)\leq1$, and the main term is also $O(Q)$.  Since the
right-hand side of \eqref{eq:section-two-main} contains $Q^{1+\epsilon}$, this
proves the remaining range.

We write
\begin{equation}\label{eq:dyadic-decomposition}

=\sum_{{j\ge0:2^{-j}Q\ge2}}
\bigl(\mathcal R_{\mathcal C}(\delta,2^{-j}Q)

\end{equation}
We will show that the $j=0$ term satisfies
\begin{equation}\label{eq:dyadic-bound}

-\mathcal R_{\mathcal C}(\delta,Q/2)\lesssim_\epsilon

+Q^\epsilon\left(

\delta^{k^2/(k+1)}Q^{(2k+1)/(k+1)}
\right).
\end{aligned}
\end{equation}
The same estimate applies to every other term after replacing $Q$ by $2^{-j}Q$.

\subsection*{Case 1: \texorpdfstring{$2Q^{-1}\le\delta\le1$}{}}
Choose a nonnegative $\Phi\in\mathcal S(\mathbb R^n)$ such that
$\Phi(x)\ge1$ for $|x|\le2$ and $\widehat{\Phi}\in C_c^\infty(\mathbb R^n)$.
For every $\mathbf p\in\mathbb Z^n$ and every integer $q$ with $Q/2<q\le Q$, one may see that
\[
\mathbf1_{\{\operatorname{dist}(\mathbf p/q,\mathcal C)\le\delta/q\}}
\lesssim_\gamma\frac Q\delta
\int_{\mathbb R}\chi(t)
\Phi\!\left(\frac{\mathbf p-q\gamma(t)}{\delta}\right)\,dt.
\]
Summing over $\mathbf p$ and $q$, and using $\omega(2q/Q)\ge1$ for
$Q/2<q\le Q$, we obtain
\begin{equation}\label{eq:counting-integral-majorant}
\begin{aligned}

\sum_{q\in\mathbb Z}\omega(2q/Q)
\int_{\mathbb R}\chi(t)\sum_{\mathbf p\in\mathbb Z^n}
\Phi\!\left(\frac{\mathbf p-q\gamma(t)}{\delta}\right)\,dt.
\end{aligned}
\end{equation}
Poisson summation in $\mathbf p$ gives
\[
\sum_{\mathbf p\in\mathbb Z^n}
\Phi\!\left(\frac{\mathbf p-q\gamma(t)}{\delta}\right)

\widehat{\Phi}(\delta\mathbf j_{\geq 1})e(-q\mathbf j_{\geq 1}\cdot\gamma(t)).

\widehat{\Phi}(\delta\mathbf j_{\geq 1})\widehat{d\sigma_\chi}(q\mathbf j_{\geq 1})\right|.
\end{aligned}
\end{equation}
In the remaining term, apply Poisson summation in $q$:
\[
\begin{aligned}

\sum_{\mathbf j_{\geq 1}\in\mathbb Z^n\setminus\{0\}}
\widehat{\Phi}(\delta\mathbf j_{\geq 1})
\widehat{d\sigma_\chi}(q\mathbf j_{\geq 1})=\frac Q2\sum_{\substack{\mathbf j\in\mathbb Z^{n+1}:
\mathbf j_{\geq 1}\neq0}}
\widehat{\Phi}(\delta\mathbf j_{\geq 1})
\widehat{\mu_{\omega,\chi}}\bigl(\tfrac Q2\mathbf j\bigr).
\end{aligned}
\]
Hence, it suffices to prove that
\[

\mathbf j_{\geq 1}\neq0}}

\widehat{\mu_{\omega,\chi}}\bigl(\tfrac Q2\mathbf j\bigr)\right|
\lesssim_\epsilon Q^\epsilon E_n(Q^{-1},\delta^{-1}).
\]

$\Psi(u_0,u):=\rho(u_0)\widehat{\Phi}(u)$. We decompose the sum into two terms
\[
\begin{aligned}

\mathbf j_{\geq 1}\neq0}}
\Psi(\delta\mathbf j)
\widehat{\mu_{\omega,\chi}}\bigl(\tfrac Q2\mathbf j\bigr)+\sum_{\substack{\mathbf j\in\mathbb Z^{n+1}\\
\mathbf j_{\geq 1}\neq0}}

\end{aligned}
\]
For the first term, Proposition~\ref{prop:conical-n}, applied with height $Q/2$, gives
\[
\begin{aligned}

\mathbf j_{\geq 1}\neq0}}
\Psi(\delta\mathbf j)
\widehat{\mu_{\omega,\chi}}\bigl(\tfrac Q2\mathbf j\bigr)\right|
&\lesssim_\epsilon Q^\epsilon E_n(2Q^{-1},\delta^{-1})\lesssim_\epsilon Q^\epsilon E_n(Q^{-1},\delta^{-1}).
\end{aligned}
\]
By the choice of $\rho$, every nonzero summand in the second term satisfies
\[

\qquad(t\in\operatorname{supp}\chi).
\]

\[

\lesssim_M (Q|j_0|)^{-M}.
\]
Summing this estimate with $M$ sufficiently large shows that the second term is
$O_N(Q^{-N})$ for every $N>0$.

\[

\quad\Longrightarrow\quad
\begin{cases}
|(p_1,p_2)-q(t,\gamma_2(t))|\le\delta,\\
|t-p_1/q|\le\delta/q.
\end{cases}

\lesssim_{\epsilon,\gamma}Q^{1+\epsilon},
\end{aligned}
\]
where the last inequality uses $\delta<2Q^{-1}$.
This proves \eqref{eq:dyadic-bound} in Case~2 and completes the proof.

<!-- proposition-11: body_md -->
\label{prop:n-projected-curves}
Let $r,m$ be integers with $0\le r\le n-2$ and $2\le m\le n-r$.
Define
\[
\mathcal C_{r,m}
:=\{(\mathcal D_r\gamma_{r+1}(t),\ldots,
       \mathcal D_r\gamma_{r+m}(t)):t\in I^*\}\subset\mathbb R^m.
\]
Then $\mathcal C_{r,m}$ is nondegenerate, and
$s=\mathcal D_r\gamma_{r+1}(t)$ gives a smooth reparametrization.
The same statement holds with $I^*$ replaced by any compact interval
in the neighborhood where $\operatorname{Wr}(\gamma_0,\ldots,\gamma_k)\ne0$
for $0\le k\le n$.
<!-- end proposition-11 -->
<!-- proof-5: body_md -->
Let $t_0\in I^*$. To prove nondegeneracy, suppose that real numbers
$a_{r+1},\ldots,a_{r+m}$ satisfy
\[
\sum_{\nu=r+1}^{r+m}a_\nu
\frac{d^k}{dt^k}(\mathcal D_r\gamma_\nu)(t_0)=0
\qquad(1\le k\le m).
\]
We show that all these coefficients are zero.
By the recursive definition of the normalized derivatives,
$\mathcal D_{r+k}\gamma_\nu$ is a linear combination of the first
$k$ ordinary derivatives of $\mathcal D_r\gamma_\nu$,
with coefficients independent of $\nu$.
Indeed, each step differentiates the preceding expression and divides
by the nonzero function $(\mathcal D_{r+k-1}\gamma_{r+k})'$.
Consequently,
\[
\sum_{\nu=r+1}^{r+m}a_\nu\mathcal D_{r+k}\gamma_\nu(t_0)=0
\qquad(1\le k\le m).
\]
Since $\mathcal D_{r+k}\gamma_\nu=0$ for $\nu<r+k$ and
$\mathcal D_{r+k}\gamma_{r+k}=1$, these equations become
\[
0=a_{r+k}+\sum_{\nu=r+k+1}^{r+m}
 a_\nu\mathcal D_{r+k}\gamma_\nu(t_0)
\qquad(1\le k\le m).
\]
Taking $k=m$ gives $a_{r+m}=0$; then $k=m-1$ gives
$a_{r+m-1}=0$. Continuing down to $k=1$ gives
$a_{r+1}=\cdots=a_{r+m}=0$.
Thus the first $m$ derivative vectors are linearly independent at
$t_0$, proving nondegeneracy.

Finally, $ds/dt=(\mathcal D_r\gamma_{r+1})'\ne0$, so the change of
parameter is smooth and preserves nondegeneracy.
The argument applies throughout the neighborhood where the
normalizing denominators are nonzero, proving the last assertion.

<!-- lemma-7: body_md -->
\label{lem:bp36-planar-bound}
Let $r\in\mathcal R_{\boldsymbol\lambda}$ with $r\le n-3$.
For every $\eta>0$ and $D\ge1$,
\[
\mathfrak R_{\mathcal C_{r,n-r-1}}(u,D)
\lesssim_{\eta,\gamma}D^\eta(uD^2+D),\qquad 0<u\le1.

<!-- end lemma-7 -->
<!-- proof-7: body_md -->
The planar projection of $\mathcal C_{r,n-r-1}$ is
$\mathcal C_{r,2}=\{(s,f_r(s))\}$, with
\[
f_r(s)=\mathcal D_r\gamma_{r+2}(t(s)),\qquad f_r'(s)=\frac{(\mathcal D_{r}\gamma_{r+2})'(t(s))}{(\mathcal D_r\gamma_{r+1})'(t(s))}=\mathcal D_{r+1}\gamma_{r+2}(t(s)),\qquad
f_r''(s)=\frac{(\mathcal D_{r+1}\gamma_{r+2})'(t(s))}{(\mathcal D_r\gamma_{r+1})'(t(s))}\ne0.
\]
The curve is smooth, and $|f_r''|$ is bounded above and away from zero. 
The planar counting theorem of Vaughan and Velani~\cite[Theorem~3 and the following remark]{VV2006},
applied with thickness $\max\{u,D^{-1}\}$, gives
\[
\mathcal R_{\mathcal C_{r,2}}(u,D)
\lesssim_{\eta,\gamma}D^\eta(uD^2+D),\qquad 0<u\le1.
\]
Recall that $\mathcal R_{\mathcal C_{r,n-r-1}}(u,D)$ counts pairs
$(\mathbf p,d)\in\mathbb Z^{n-r-1}\times\mathbb N$ satisfying
\[
d\lesssim D,\qquad
\operatorname{dist}(\mathbf p/d,\mathcal C_{r,n-r-1})\le u/d.
\]
Their projections $((p_1,p_2),d)$ are counted by
$\mathcal R_{\mathcal C_{r,2}}(u,D)$.
For given $d,p_1,p_2$, the first coordinate condition
$|ds-p_1|\le u$ confines $s$ to an interval of length $2u/d$.
Each remaining coordinate function has bounded first derivative,
so its values multiplied by $d$ lie in an interval of length
$O_\gamma(u)$. Allowing an error of $u$ therefore leaves
$O_\gamma(1)$ choices for each remaining integer coordinate.
There are $n-r-3$ such coordinates, and hence
\[
\mathcal R_{\mathcal C_{r,n-r-1}}(u,D)
\lesssim_\gamma \mathcal R_{\mathcal C_{r,2}}(u,D)
\lesssim_{\eta,\gamma}D^\eta(uD^2+D)
\]
for $0<u\le1$. 

It remains to sum the weights defining $\mathfrak R$.
For any $v>0$ and a fixed denominator $d$, there are
$O_\gamma(d+v)$ choices for the first integer coordinate.
Once it is fixed, the same argument as above gives
$O_\gamma(1+v)$ choices for each remaining coordinate. Hence
\[
\begin{aligned}
\mathcal R_{\mathcal C_{r,n-r-1}}(v,D)
&\lesssim_\gamma\sum_{1\le d\le D}(d+v)(1+v)^{n-r-2}\\
&\lesssim_\gamma D(D+v)(1+v)^{n-r-2}
\le D^2(1+v)^{n-r-1}.
\end{aligned}
\]
Since $C$ is fixed, this bound and the preceding small-thickness estimate give
\[
\mathcal R_{\mathcal C_{r,n-r-1}}(Cv,D)
\lesssim_{\eta,\gamma}D^\eta(vD^2+D),\qquad 0<v\le1.
\]
Indeed, if $Cv>1$, then $v>C^{-1}$ and the elementary bound applies.
Thus the terms with $2^\nu u\le1$ contribute at most
\[
D^\eta\sum_{\substack{\nu\ge0\\2^\nu u\le1}}
2^{-A\nu}(1+2^\nu u)(2^\nu uD^2+D)
\lesssim_{\eta,\gamma}D^\eta(uD^2+D).
\]
For the remaining terms, the elementary bound gives
\[
\begin{aligned}
&\sum_{\substack{\nu\ge0\\2^\nu u>1}}
2^{-A\nu}(1+2^\nu u)
\mathcal R_{\mathcal C_{r,n-r-1}}(C2^\nu u,D)\\
&\qquad\lesssim_\gamma
D^2u^{n-r}\sum_{\substack{\nu\ge0\\2^\nu u>1}}
2^{-(A-n+r)\nu}\lesssim_\gamma D^2u^A\le uD^2.
\end{aligned}
\]
Here $A>n-r$ and $0<u\le1$. These estimates prove the lemma.

$z_k=H_{\mathbf j}^{(k)}(s_0)/L_{r+k}$ and $s_0-s=-\frac{L_{r+2}}{L_{r+3}}y$, and applying \eqref{twoscalingineq}, we obtain
\begin{equation}\label{boundzk}
    |z_k|
\lesssim_\gamma
\sum_{m=0}^{n-r-k}\frac{L_{r+k+m}}{L_{r+k}}\left( \frac{L_{r+2}}{L_{r+3}} \right)^m|y|^m+\frac{\delta^{-1}}{L_{r+k}}\left(\frac{L_{r+2}}{L_{r+3}}\right)^{n-r-k}|y|^{n-r-k}
\lesssim_\gamma1
\qquad (|y|\le2c).
\end{equation}

Consequently,
\begin{equation}\label{zbounded}
    \chi(y)\widetilde a(y,\mathbf z)\ne0
\quad\Longrightarrow\quad |\mathbf z|\lesssim_\gamma1.
\end{equation}


\eqref{dphi} shows that $y=0$ is a critical point of the phase function.
By \eqref{eq:bp52-derivative-bounds}, $|H_{\mathbf j}''(s)|\asymp_\gamma L_{r+2}$ and $|H_{\mathbf j}'''(s)|\lesssim L_{r+3}$ on this support. If $c$ is chosen sufficiently small,
by \eqref{d2phi}, one sees that
\begin{equation}\label{uniformnondeg}
    |\partial_y^2\Phi_{\mathbf z}(y)|\asymp_\gamma1,
\end{equation} 
on the support of $\chi\widetilde{a}$.
Hence $y=0$ is the only stationary point of the phase function $\Phi_{\mathbf z}(y)$ on the support of $\chi\widetilde{a}$, and it is uniformly nondegenerate in the sense of \eqref{uniformnondeg}.

\medskip
The next goal is to prove
\begin{equation}\label{boundtildea}
|\partial_y^q\partial_{\mathbf z}^{\alpha}\widetilde a(y,\mathbf z)| \lesssim_{q,\alpha,\gamma}1,
\end{equation}
when $(y,\mathbf z)$ lies in the support of $\chi \widetilde{a}$. For simplicity, we write \[h:=\frac{L_{r+2}}{L_{r+3}}(\le 1).\] 
Set
\[
b(y,\mathbf z)=\frac{H_{\mathbf j}''(s_0+hy)}{L_{r+2}},
\]
so that $\widetilde a=a|b|$. Here, $\mathbf j_{\ge r+2}=\mathbf j_{\ge r+2}(\mathbf z)$ is a function of $\mathbf z$ as in \eqref{changeofcoord}. We also remark $b\neq 0$ in the support of $\chi\widetilde a$, so $\partial_y^q\partial_{\mathbf z}^\alpha |b|=\pm \partial_y^q\partial_{\mathbf z}^\alpha b$.

Recall that the majorant $a$ is chosen in \eqref{defa} as a finite product of fixed smooth cutoffs in the normalized
variables
\[
s,\qquad \delta\mathbf j_{\ge r+2},\qquad
\delta H_{\mathbf j}'(s),\qquad
\frac{(\mathcal D_{r+\ell}P_{\mathbf j})(t(s))}{L_{r+\ell}}
\quad(2\le\ell\le n-r),
\]
so we just need to prove $O(1)$ upper bounds when $\partial_y^q\partial_{\mathbf z}^\alpha$ is applied to these variables and applied to $b$. 
\medskip

\begin{enumerate}
    \item It is easy to see $|\partial_y^q\partial_{\mathbf z}^\alpha s|\le h^q\le 1$, since $s=s_0+hy$ and $s$ is independent of $\mathbf z$.\\

    \item Since $\mathbf j_{\ge r+2}$ is linear in $\mathbf z$ and is independent of $y$, only $q=0$ and $|\alpha|\le 1$ contribute. By \eqref{bounddj}, one has $|\partial_y^q\partial_{\mathbf z}^\alpha (\delta\mathbf j_{\ge r+2})|\lesssim \delta\max_j L_j\le 1$.\\

    \item 
    A direct computation gives
    $|\partial_y^q\partial_{\mathbf z}^\alpha (\delta H_{\mathbf j}'(s))|\lesssim \delta (|\mathbf{j}_{\ge r+2}|+|\partial_{\mathbf z}\mathbf{j}_{\ge r+2}| )\lesssim 1  $. Here, we only need to consider $|\alpha|\le 1$ since $H_\mathbf{j}'$ is linear in $\mathbf{j}$ and hence in $\mathbf z$. \\

     \item Denote $F_\ell(s,\mathbf z):=(\mathcal D_{r+\ell}P_{\mathbf j})(t(s))$. We want to prove
     \begin{equation}\label{boundFl}
         | \partial_y^q\partial_{z_k}^\alpha F_\ell |\lesssim L_{r+\ell}
     \end{equation}
     
     By \eqref{eq:n-normalized-derivatives},
     \[ \frac{d}{ds}F_\ell(s,\mathbf z)=c_\ell(s)F_{\ell+1}(s,\mathbf z)\]
     with $|c_\ell|\asymp 1$. Also, \eqref{Hj'} implies
     \[ F_1=j_{r+1}+H'_{\mathbf{j}}, \]
     so
     \[ H''_{\mathbf j}=\frac{d}{ds}F_1=c_1 F_2. \]
     Consequently,
     \[ F_2=c_1^{-1}H''_{\mathbf j},\ \ \ F_3=c_2^{-1}\frac{d}{ds}F_2=c_2^{-1}(c_1^{-1}H''_{\mathbf j})'. \]
     
Iterating this process yields 
\begin{equation}\label{Fl}
    F_\ell(s,\mathbf z)=\sum_{v=2}^{\ell}a_{\ell,v}(s)H^{(v)}_{\mathbf j}(s), 
\end{equation} 
where all $a_{\ell,v}$ are bounded. In other words, $F_\ell$ is a lower-triangular combination of $H_{\mathbf j}^{(v)}$ ($2\le v\le \ell$). Since $F_\ell$ is linear in $\mathbf j_{\ge r+2}$ and hence linear in $\mathbf z$, we just need to show
\[ |\partial_y^q F_\ell|\lesssim L_{r+\ell},\ \ \  |\partial_y^q \partial_{z_k} F_\ell|\lesssim L_{r+\ell}. \]

To prove the first one, we observe that for $2\le v\le \ell$,
\begin{equation}\label{derivativebounddH0}
    |\partial_y^qH^{(v)}_{\mathbf j}(s)|=|h^q H^{(v+q)}_{\mathbf j}(s)|\lesssim h^qL_{\min\{r+v+q,n\}}\lesssim L_{r+v}. 
\end{equation} 
The last inequality uses \eqref{twoscalingineq}. Using \eqref{Fl}, we see that $|\partial_y^qF_\ell|\lesssim L_{r+\ell}$.
Next, we prove the second one. It suffices to prove
\begin{equation}\label{derivativeboundH}
|\partial_y^q\partial_{z_k}H^{(v)}_{\mathbf j}(s)|\lesssim L_{r+v}.
\end{equation}

Recalling \eqref{eq:bp52-derivative-bounds} and\eqref{bounddj} that $|\mathbf{j}_{\ge r+2}|\lesssim \delta^{-1}$ and $|\partial_{z_k}\mathbf{j}_{\ge r+2}|\lesssim \delta^{-1}$ on the support of $\chi\widetilde{a}$, by the definition \eqref{Hj}, one sees that
\begin{equation}\label{trivialbound}
|\partial_y^q\partial_{z_k}H^{(v)}_\mathbf{j}(s)|\lesssim |\mathbf{j}_{\ge r+2}|+| \partial_{z_k}\mathbf{j}_{\ge r+2}|\lesssim \delta^{-1}.
\end{equation}

When $v=n-r$, \eqref{derivativeboundH} holds, so we just need to consider $v\le n-r-1$.
By Taylor's expansion,
\begin{equation}\label{taylorHv}
     H^{(v)}_{\mathbf j}(s)=\sum_{m=0}^{n-r-v-1}H^{(v+m)}_\mathbf{j}(s_0)\frac{(hy)^m}{m!}+\frac{(hy)^{n-r-v}}{(n-r-v-1)!}\int_{0}^1(1-\theta)^{n-r-v-1}H^{(n-r)}_\mathbf{j}(s_0+\theta hy)d\theta. 
\end{equation}

First consider when $\partial_y^q\partial_{z_k}$ acts on the first term. Recalling that $z_k=\frac{H^{(k)}(s_0)}{L_{r+k}}$, we have
\[ \partial_y^q\partial_{z_k}(\text{First term})\lesssim h^{m}L_{r+k}(1+|y|)^{(m-q)_+}. \]
with $v+m=k$. By \eqref{twoscalingineq}, $h^{k-v}L_{r+k}\le L_{r+v}$, and noting $|y|\lesssim 1$, this finishes
the estimate for the first term.

For the remainder term, using \eqref{trivialbound} we have
\[ \partial_y^q\partial_{z_k}(\text{Remainder term})\lesssim h^{n-r-v}\delta^{-1} (1+|y|)^{n-r-v}\lesssim L_{r+v}. \]
The last inequality is by \eqref{twoscalingineq}.\\

\item To prove $|\partial_y^q\partial_{z_k}b|\lesssim 1$, we just note that this is a special case of \eqref{derivativebounddH0} and \eqref{derivativeboundH} with $v=2$.

\end{enumerate}

This finishes the proof of \eqref{boundtildea}.
\medskip

Finally, noting
$\partial_y\Phi_{\mathbf z}=-yb$, so the bounds
for $b$ also give the required bounds for the phase function. Hence we obtain
\begin{equation}
|\partial_y^q\partial_{\mathbf z}^{\alpha}\Phi_{\mathbf z}(y)|
+|\partial_y^q\partial_{\mathbf z}^{\alpha}\widetilde a(y,\mathbf z)|
\lesssim_{q,\alpha,\gamma}1
\end{equation}
on the bounded parameter region contributing to $\chi\widetilde a$.

We now apply Lemma~\ref{lem:stationary-phase-parameters} with
\[
\Lambda=\frac{dL_{r+2}^3}{L_{r+3}^2},
\qquad \phi(y;\mathbf z)=\Phi_{\mathbf z}(y),
\qquad w(\Lambda;y,\mathbf z)=\chi(y)\widetilde a(y,\mathbf z).
\]
The assumed lower bound
$d\gtrsim1+L_{r+3}^2/L_{r+2}^3$ gives $\Lambda\gtrsim1$.
\eqref{uniformnondeg} verifies uniqueness and nondegeneracy of the
critical point. The amplitude is independent of
$\Lambda$, so its positive $\Lambda$-derivatives vanish. Lemma \ref{lem:stationary-phase-parameters} yields
\begin{equation}
\int\chi(y)\widetilde a(y,\mathbf z)
 e\!\left(\Lambda\Phi_{\mathbf z}(y)\right)\,dy=\Lambda^{-1/2}e(\Lambda\Phi_{\mathbf z}(0))w_1(\Lambda;\mathbf z).
\end{equation}
Since $\Phi_{\mathbf z}(0)=0$ and $w_1(\Lambda;\mathbf z)$ have bounded $\mathbf z$-derivatives, we have
\begin{equation}
\sup_{\mathbf z}(1+|\mathbf z|)^L
\left|\partial_{\mathbf z}^{\alpha}
\int\chi(y)\widetilde a(y,\mathbf z)
 e\!\left(\Lambda\Phi_{\mathbf z}(y)\right)\,dy\right|
\lesssim_{\alpha,L,\gamma}\Lambda^{-1/2}
=\left(\frac{dL_{r+2}^3}{L_{r+3}^2}\right)^{-1/2}.
\end{equation}
Here $(1+|\mathbf z|)^L$ is harmless because $|\mathbf z|\lesssim 1$ by \eqref{zbounded}.  Thus the above estimate gives the required bounds for \eqref{eq:bp52-amplitude-bounds} when the $y$-integral is restricted to the cutoff by $\chi$.

$\bullet$ \textbf{The contribution away from $y=0$.}
Write $1-\chi=\sum_{\nu\ge0}\chi_\nu$, where
\[
\operatorname{supp}\chi_\nu\subset\{|y|\asymp2^\nu c\},
\qquad |\partial_y^q\chi_\nu(y)|\lesssim_{q,\gamma}2^{-\nu q}.
\]

We record two useful estimates. Assuming $2\le v\le n-r$, then
\begin{equation}\label{usefulestimate}
|\partial_y^qH^{(v)}_\mathbf{j}(s)|\lesssim L_{r+v},\ \ \ |\partial_y^q\partial_{z_k} H^{(v)}_\mathbf{j}(s)|\lesssim L_{r+v}(1+|y|)^{n-r-2}.
\end{equation}
The first one is just \eqref{derivativebounddH0}; the second one follows the same proof as \eqref{derivativeboundH} in which we also retain $y$ variable that contributes an additional factor $\lesssim (1+|y|)^{n-r-v}\le (1+|y|)^{n-r-2}$.


For $q\ge2$, direct differentiation gives
\[
\partial_y^q\Phi_{\mathbf z}(y)
=-\frac1{L_{r+2}}\left[
(q-1)\left(\frac{L_{r+2}}{L_{r+3}}\right)^{q-2}H_{\mathbf j}^{(q)}(s)
+y\left(\frac{L_{r+2}}{L_{r+3}}\right)^{q-1}H_{\mathbf j}^{(q+1)}(s)
\right].
\]
Hence, on the support of $\chi_\nu\widetilde a$,
\[
|\partial_y\Phi_{\mathbf z}|\asymp_\gamma2^\nu,
\qquad |\partial_y^q\Phi_{\mathbf z}|\lesssim_{q,\gamma}2^\nu\quad(q\ge2),
\qquad
\left|\partial_y^q\frac1{\partial_y\Phi_{\mathbf z}}\right|
\lesssim_{q,\gamma}2^{-\nu}\quad(q\ge0).
\]
To control the size of $\mathbf z$, expand each $H_{\mathbf j}^{(k)}(s_0)$
about $s$ as in \eqref{boundzk}. For $2\le k\le n-r$,
\[
|z_k|\lesssim_\gamma
\sum_{m=0}^{n-r-k}\frac{L_{r+k+m}}{L_{r+k}}
\left(\frac{L_{r+2}}{L_{r+3}}\right)^{m}|y|^{m}
+\frac{\delta^{-1}}{L_{r+k}}
\left(\frac{L_{r+2}}{L_{r+3}}\right)^{n-r-k}|y|^{n-r-k}
\lesssim_\gamma(1+|y|)^{n-r-k}.
\]
In particular,
\begin{equation}\label{boundofz}
    \chi_\nu(y)\widetilde a(y,\mathbf z)\ne0
\quad\Longrightarrow\quad
1+|\mathbf z|\lesssim_\gamma2^{\nu(n-r-2)}.
\end{equation}


The next goal is to prove
\begin{equation}
    |\partial_y^q\partial_{\mathbf z}^\alpha\widetilde{a}(y,\mathbf z)|\lesssim_{q,\alpha,\gamma} 2^{\nu(n-r-2)|\alpha|}. 
\end{equation}
The proof follows the same strategy when proving \eqref{boundtildea}; the only additional thing is to take consideration the effect of the $y$ variable.
Using the previous notation, one has $\widetilde a=a|b|$. 
One needs to estimate the action of $\partial_y^q\partial_{\mathbf z}^\alpha$ on
\[
s,\qquad \delta\mathbf j_{\ge r+2},\qquad
\delta H_{\mathbf j}'(s),\qquad
\frac{(\mathcal D_{r+\ell}P_{\mathbf j})(t(s))}{L_{r+\ell}}
\quad(2\le\ell\le n-r),\qquad b.
\]
Follow the same proof as in the previous part, we see that the action of $\partial_y^q\partial_{\mathbf z}^\alpha$ on $s, \delta\mathbf{j}_{\ge r+2}$ and $\delta H_\mathbf{j}'(s)$ results in an $O(1)$ bound. Following the proof strategy of \eqref{boundFl} together with \eqref{usefulestimate}, one has
\[ |\partial_y^q\partial_{\mathbf z}^\alpha\frac{(\mathcal D_{r+\ell}P_{\mathbf j})(t(s))}{L_{r+\ell}}|\lesssim (1+|y|)^{n-r-2}. \]
From \eqref{defa}, by using Leibniz rule and noting $|y|\sim 2^\nu$, one has
\[ |\partial_y^q\partial_{\mathbf z}^\alpha a|\lesssim 2^{\nu (n-r-2)|\alpha| } \]
in the support of $\chi_\nu \widetilde a$. One can similarly prove the bound
\[ |\partial_y^q\partial_{\mathbf z}^\alpha b|\lesssim 2^{\nu(n-r-2)|\alpha|}. \]
Hence, using $\widetilde a=a |b|$ and $b\neq 0$ in the support of $\chi_\nu \widetilde a$, one shows that
\[
|\partial_y^q\partial_{\mathbf z}^{\alpha}
 \widetilde a(y,\mathbf z)|
\lesssim_{q,\alpha,\gamma}
2^{\nu(n-r-2)|\alpha|}.
\]

One can also show
\begin{align}
|\partial_y^q\partial_{z_k}\Phi_{\mathbf z}(y)|
&\lesssim_{q,\gamma}2^{\nu(n-r)}.
\end{align}
(Since $\Phi_{\mathbf z}$ is linear in $\mathbf z$, all
of its $\mathbf z$-derivatives of order at least two vanish.) When $q\ge 1$, the estimate follows from $\partial_y\Phi_{\mathbf z}=-yb$ and the derivative bounds on $b$. When $q=0$, we use the expression \eqref{taylorphi} to see that it suffices to prove for any $|s'-s|\lesssim |hy|$ that
\begin{equation}\label{want}
|\partial_{z_k}H_{\mathbf{j}}''(s')|\lesssim L_{r+2} |y|^{n-r-2}. 
\end{equation} 
Using the Taylor's expansion \eqref{taylorHv} with $v=2$, the follow-up estimates give the desired \eqref{want}.

Set
\[
\Lambda:=\frac{dL_{r+2}^3}{L_{r+3}^2}\gtrsim1.
\]
Leibniz rule gives
\begin{equation}\label{eqqq1}
    \partial_{\mathbf z}^{\alpha}
 \bigl((\chi_\nu\widetilde a) e(\Lambda\Phi_{\mathbf z})\bigr)
=e(\Lambda\Phi_{\mathbf z})
 \sum_{\beta\le\alpha}
 \binom{\alpha}{\beta}
 (2\pi i\Lambda)^{|\beta|}
 \partial_{\mathbf z}^{\alpha-\beta}(\chi_\nu \widetilde a)
 \prod_{k=2}^{n-r}
 \bigl(\partial_{z_k}\Phi_{\mathbf z}\bigr)^{\beta_k}.
\end{equation}

By
the preceding bounds, the Leibniz rule in $y$, and the estimates for
$\chi_\nu$, for every $q\ge0$ we have
\begin{align}\label{eqqq2}
\left|\partial_y^q\left[
 \partial_{\mathbf z}^{\alpha-\beta}(\chi_\nu \widetilde a)
 \prod_{k=2}^{n-r}
 \bigl(\partial_{z_k}\Phi_{\mathbf z}\bigr)^{\beta_k}
 \right]\right|
\qquad\lesssim  
2^{\nu(n-r)|\alpha-\beta|}
2^{\nu(n-r)|\beta|}
=2^{\nu(n-r)|\alpha|}.
\end{align}
Indeed, derivatives falling on $\chi_\nu$ only improve this estimate,
while every $y$-derivative of a factor
$\partial_{z_k}\Phi_{\mathbf z}$ satisfies the same bound as the factor
itself.

For fixed $\mathbf z$, introduce the formal adjoint of the
nonstationary-phase operator,
\[
T_{\mathbf z}f:=-\partial_y\!\left(
 \frac{f}{\partial_y\Phi_{\mathbf z}}\right).
\]
The identity
\[
e(\Lambda\Phi_{\mathbf z})
=\frac{1}{2\pi i\Lambda\,\partial_y\Phi_{\mathbf z}}
 \partial_y e(\Lambda\Phi_{\mathbf z})
\]
and $N$ integrations by parts give, with no boundary terms,
\[
\int f(y)e(\Lambda\Phi_{\mathbf z}(y))\,dy
=(2\pi i\Lambda)^{-N}
 \int T_{\mathbf z}^{N}f(y)
 e(\Lambda\Phi_{\mathbf z}(y))\,dy.
\]
Each application of $T_{\mathbf z}$ gains one factor $2^{-\nu}$.  More
precisely, the reciprocal-phase bounds above imply
\[
\begin{aligned}
|\partial_y^q(T_{\mathbf z}f)|
&\le C_q\sum_{j=0}^{q+1}|\partial_y^j f|
 \left|\partial_y^{q+1-j}
       \frac1{\partial_y\Phi_{\mathbf z}}\right|\\
&\lesssim_{q,\gamma}2^{-\nu}
 \max_{0\le j\le q+1}|\partial_y^j f|.
\end{aligned}
\]
Iterating this inequality yields
\[|T_{\mathbf z}^{N}f|
\lesssim_{N,\gamma}2^{-N\nu}
 \max_{0\le q\le N}|\partial_y^qf|.\]
 Hence, for $f$ supported in $\operatorname{supp}\chi_\nu$, whose length is
$O_\gamma(2^\nu)$, we obtain
\begin{equation}\label{eqqq3}
   |\int f(y)e(\Lambda\Phi_{\mathbf z}(y))\,dy|
\lesssim_{N,\gamma} \Lambda^{-N}2^{(1-N)\nu}
\max_{0\le q\le N}\|\partial_y^qf\|_{L^\infty_y}.
\end{equation}

Combining \eqref{eqqq1},\eqref{eqqq2},\eqref{eqqq3}, we obtain
\[
\left|\partial_{\mathbf z}^{\alpha}
\int\chi_\nu(y)\widetilde a(y,\mathbf z)
 e\!\left(\Lambda\Phi_{\mathbf z}(y)\right)\,dy\right|\lesssim_{N,\alpha,\gamma}
\Lambda^{|\alpha|-N}2^{-N\nu}2^\nu
2^{\nu(n-r)|\alpha|}.
\]

It remains to insert the weight in $\mathbf z$. By \eqref{boundofz},
$1+|\mathbf z|\lesssim_\gamma2^{\nu(n-r-2)}$.  Hence
\[
\begin{aligned}
&\sup_{\mathbf z}(1+|\mathbf z|)^L
\left|\partial_{\mathbf z}^{\alpha}\left[
\Lambda^{1/2}
\int\chi_\nu(y)\widetilde a(y,\mathbf z)
 e\!\left(\Lambda\Phi_{\mathbf z}(y)\right)\,dy
\right]\right|\\
&\qquad\lesssim_{N,\alpha,L,\gamma}
\Lambda^{|\alpha|+\frac12-N}2^{-N\nu}2^\nu
2^{\nu(n-r)(|\alpha|+L)}.
\end{aligned}
\]
Choosing  $N$ large enough, we get
\[
\begin{aligned}
\sup_{\mathbf z}(1+|\mathbf z|)^L
\left|\partial_{\mathbf z}^{\alpha}\left[
\left(\frac{dL_{r+2}^3}{L_{r+3}^2}\right)^{1/2}
\int\chi_\nu(y)\widetilde a(y,\mathbf z)
 e\!\left(\frac{dL_{r+2}^3}{L_{r+3}^2}
 \Phi_{\mathbf z}(y)\right)\,dy
\right]\right|
\lesssim_{\alpha,L,\gamma}2^{-\nu}.
\end{aligned}
\]
Summing over $\nu\ge0$ and adding the contribution near $y=0$ proves
\eqref{eq:bp52-amplitude-bounds}.

Write $s_0=p/d$ and recall the following notation.
\[
\mathsf M=
\left(\left.\partial_s^k\bigl[\mathcal D_r\gamma_\nu(t(s))\bigr]
\right|_{s=s_0}\right)_{
\substack{2\le k\le n-r\\ r+2\le\nu\le n }}.
\]
The nondegeneracy of $\mathcal C_{r,n-r}$ in
Proposition~\ref{prop:n-projected-curves} gives
\[
|\det\mathsf M|\asymp_\gamma1,
\qquad
\|\mathsf M\|+\|\mathsf M^{-1}\|\lesssim_\gamma1.
\]
The normalized derivative coordinates from \eqref{derivativecoord} give
\[
\mathbf z(\mathbf{j}_{\ge r+2})
=\operatorname{diag}(L_{r+2},\ldots,L_n)^{-1}\mathsf M\mathbf \mathbf{j}_{\ge r+2}.
\]
In particular, \eqref{eq:bp52-amplitude} reads
\[
I_{d,p}(\mathbf j_{\ge r+2})
=\sqrt{\frac{L_{r+2}}d}\,
 e\!\left(d\sum_{\nu=r+2}^n j_\nu \mathcal D_r\gamma_\nu(t(s_0))\right)
 B_{d,s_0}\bigl(\mathbf z(\mathbf j_{\ge r+2})\bigr).
\]

Poisson summation in the original integer coordinates gives
\[
\sum_{\mathbf j_{\ge r+2}\in\mathbb Z^{n-r-1}}
 I_{d,p}(\mathbf j_{\ge r+2})
=\sqrt{\frac{L_{r+2}}d}
\sum_{\mathbf a\in\mathbb Z^{n-r-1}}
\int_{\mathbb R^{n-r-1}}
 B_{d,s_0}\bigl(\mathbf z(\mathbf x)\bigr)
\cdot e\!\left(
-\bigl[\mathbf a-d\bigl(\mathcal D_r\gamma_\nu(t(s_0))\bigr)_{\nu=r+2}^n\bigr]
\cdot\mathbf x\right)\,d\mathbf x.
\]
To estimate these Fourier integrals, use
\[
\mathbf x=\mathsf M^{-1}
\operatorname{diag}(L_{r+2},\ldots,L_n)\mathbf z,
\qquad
 d\mathbf x=\frac{\prod_{k=r+2}^nL_k}{|\det\mathsf M|}\,d\mathbf z(\mathbf x).
\]
This Jacobian produces the factor $\prod_{k=r+2}^nL_k$.
Moreover,
\[
\boldsymbol\eta\cdot\mathbf x
=\left(\operatorname{diag}(L_{r+2},\ldots,L_n)
\mathsf M^{-T}\boldsymbol\eta\right)\cdot\mathbf z.
\]
Consequently,
\[
\begin{aligned}
\int_{\mathbb R^{n-r-1}}
 B_{d,s_0}\bigl(\mathbf z(\mathbf x)\bigr)
 e(-\boldsymbol\eta\cdot\mathbf x)\,d\mathbf x=
\frac{\prod_{k=r+2}^nL_k}{|\det\mathsf M|}
\widehat B_{d,s_0}\!\left(
\operatorname{diag}(L_{r+2},\ldots,L_n)
\mathsf M^{-T}\boldsymbol\eta\right).
\end{aligned}
\]
The derivative bounds in \eqref{eq:bp52-amplitude-bounds} imply, by
integration by parts, that
\[
\left|\widehat B_{d,s_0}(\boldsymbol\xi)\right|
=\left|\int_{\mathbb R^{n-r-1}}B_{d,s_0}(\mathbf z)
 e(-\boldsymbol\xi\cdot\mathbf z)\,d\mathbf z\right|
\lesssim_{L,\gamma}(1+|\boldsymbol\xi|)^{-L}.
\]
Since $L_{r+2}\le\cdots\le L_n$, we also have
\[
\begin{aligned}
\left|\operatorname{diag}(L_{r+2},\ldots,L_n)
\mathsf M^{-T}\boldsymbol\eta\right|^2
=\sum_{k=2}^{n-r}L_{r+k}^2
 \left|(\mathsf M^{-T}\boldsymbol\eta)_k\right|^2\ge L_{r+2}^2|\mathsf M^{-T}\boldsymbol\eta|^2
\gtrsim_\gamma L_{r+2}^2|\boldsymbol\eta|^2.
\end{aligned}
\]
Thus the smallest scale $L_{r+2}$ controls the decay in every direction,
and the Fourier integral is bounded by
\[
\left|\int_{\mathbb R^{n-r-1}}
 B_{d,s_0}\bigl(\mathbf z(\mathbf x)\bigr)
 e(-\boldsymbol\eta\cdot\mathbf x)\,d\mathbf x\right|
\lesssim_{L,\gamma}
\left(\prod_{k=r+2}^nL_k\right)
(1+L_{r+2}|\boldsymbol\eta|)^{-L}.
\]
Substituting
$\boldsymbol\eta=\mathbf a-d\bigl(\mathcal D_r\gamma_\nu(t(s_0))\bigr)_{\nu=r+2}^n$
in the Poisson formula and summing over $\mathbf a$ gives
\[
\left|\sum_{\mathbf j_{\ge r+2}\in\mathbb Z^{n-r-1}}
 I_{d,p}(\mathbf j_{\ge r+2})\right|
\lesssim_{L,\gamma}{}
\sqrt{\frac{L_{r+2}}d}
\left(\prod_{k=r+2}^nL_k\right)
\sum_{\mathbf a\in\mathbb Z^{n-r-1}}
\left(1+L_{r+2}
\left|\mathbf a-d\bigl(\mathcal D_r\gamma_\nu(t(s_0))\bigr)_{\nu=r+2}^n\right|\right)^{-L}.
\]
Since $s_0=p/d$, this is \eqref{eq:bp52-second-poisson}.

<!-- lemma-10: body_md -->
\label{lem:sharp-integer-choices}
Under the assumptions above, for each $2\leq j\leq k$ and every $p_j$
arising from the choices in \eqref{eq:sharp-pk-range} and
\eqref{eq:sharp-inverse-interval},
\[
\#\Bigl(\mathbb Z\cap F^{-1}
\bigl([p_j-c_0\delta,p_j+c_0\delta]\bigr)\Bigr)
\gtrsim_n(\delta^kQ)^{1/(k+1)}.
\]
Moreover, writing $t_j=(p_j/q)^{1/j}$, every coordinate chosen in this
construction satisfies
\[
\frac13\left(\frac{c_0\delta}{Q}\right)^{1/(k+1)}
\leq t_j\leq
\frac34\left(\frac{c_0\delta}{Q}\right)^{1/(k+1)}
\qquad(1\leq j\leq k).
\]
<!-- end lemma-10 -->
<!-- proof-17: body_md -->
We first prove the bounds for $t_j$ by descending induction,
from $j=k$ to $j=1$.
By \eqref{eq:sharp-pk-range},
\[
\frac12\left(\frac{c_0\delta}{Q}\right)^{1/(k+1)}
\leq t_k\leq
\frac23\left(\frac{c_0\delta}{Q}\right)^{1/(k+1)},
\]
so the case $j=k$ holds. Let $2\leq j\leq k$ and assume the stated bounds
hold for $t_j,\ldots,t_k$. We show that they also hold for $t_{j-1}$
whenever $p_{j-1}$ is chosen according to \eqref{eq:sharp-inverse-interval}.
Since $j\leq k\leq n-1$, $q\asymp Q$, and $0<c_0\delta\leq1$,
\[
\frac{p_j}{c_0\delta}
=\frac{qt_j^j}{c_0\delta}
\gtrsim_n\left(\frac{Q}{c_0\delta}\right)^{\frac{k+1-j}{k+1}}
\geq Q^{1/n},
\]
where $(k+1-j)/(k+1)\geq1/n$.
Taking $Q_n$ sufficiently large, uniformly in $\delta,j,k$, we obtain
\[
0<p_j-c_0\delta\asymp p_j+c_0\delta\asymp p_j.
\]
For every such choice of $p_{j-1}$, \eqref{eq:sharp-inverse-interval}
gives $|F(p_{j-1})-p_j|\leq c_0\delta$.
By computations, the definition of $F$ gives $F(p_{j-1})=qt_{j-1}^j$.
Dividing $|F(p_{j-1})-p_j|\leq c_0\delta$ by $q$ gives
\begin{equation}\label{eq:sharp-tj-stability}
\begin{aligned}
|t_{j-1}^j-t_j^j|=\frac{|F(p_{j-1})-p_j|}{q}\leq\frac{c_0\delta}{q}.
\end{aligned}
\end{equation}
We claim that
$
|t_{j-1}-t_j|
\lesssim_n\left(\frac{c_0\delta}{Q}\right)^{2/(k+1)}.
$
To prove the claim, use $t_{j-1},t_j>0$ to write
\[
\begin{aligned}
|t_{j-1}^j-t_j^j|
=|t_{j-1}-t_j|
\sum_{\ell=0}^{j-1}t_{j-1}^{\ell}t_j^{j-1-\ell}\geq |t_{j-1}-t_j|\,t_j^{j-1}.
\end{aligned}
\]
Combining this with \eqref{eq:sharp-tj-stability}
gives $|t_{j-1}-t_j|\leq c_0\delta/(qt_j^{j-1})$.
Next, using $q\asymp Q$ and the assumed lower bound for $t_j$, we obtain
\[
\begin{aligned}
\frac{c_0\delta}{qt_j^{j-1}}
\lesssim_n\frac{c_0\delta}{Q}
\left(\frac{c_0\delta}{Q}\right)^{-(j-1)/(k+1)}=\left(\frac{c_0\delta}{Q}\right)^{(k-j+2)/(k+1)}
\leq\left(\frac{c_0\delta}{Q}\right)^{2/(k+1)}.
\end{aligned}
\]
The last inequality uses $j\leq k$ and $c_0\delta/Q\leq1$.
This proves the claim.

To complete the induction step, sum the bounds from the claim:
\[
\begin{aligned}
|t_{j-1}-t_k|
&\leq\sum_{\ell=j-1}^{k-1}|t_\ell-t_{\ell+1}|\\
&\lesssim_n\left(\frac{c_0\delta}{Q}\right)^{2/(k+1)}
\leq\left(\frac{c_0}{Q}\right)^{1/n}
\left(\frac{c_0\delta}{Q}\right)^{1/(k+1)}.
\end{aligned}
\]
Here we used $k\leq n-1$ and
$(c_0\delta/Q)^{1/(k+1)}\leq(c_0/Q)^{1/n}$. This gives the asserted bounds for $t_{j-1}$
and completes the induction.

For every $x\in F^{-1}\bigl([p_j-c_0\delta,p_j+c_0\delta]\bigr)$, we have
\[
x\asymp_n q\left(\frac{c_0\delta}{Q}\right)^{(j-1)/(k+1)},
\]
and therefore
\begin{equation}\label{eq:sharp-F-derivative}
F'(x)
=\frac{j}{j-1}\left(\frac{x}{q}\right)^{1/(j-1)}
\asymp_n\left(\frac{c_0\delta}{Q}\right)^{1/(k+1)}.
\end{equation}
Since $F$ is strictly increasing, the interval in
\eqref{eq:sharp-inverse-interval} has endpoints
$F^{-1}(p_j-c_0\delta)$ and $F^{-1}(p_j+c_0\delta)$.
The mean value theorem gives a point
$\xi\in F^{-1}\bigl([p_j-c_0\delta,p_j+c_0\delta]\bigr)$ such that
\[
F'(\xi)\bigl(F^{-1}(p_j+c_0\delta)-F^{-1}(p_j-c_0\delta)\bigr)
\asymp c_0\delta.
\]
Using \eqref{eq:sharp-F-derivative}, we therefore obtain
\begin{equation}\label{eq:sharp-I-length}
\begin{aligned}
\left|F^{-1}\bigl([p_j-c_0\delta,p_j+c_0\delta]\bigr)\right|
&\asymp\frac{c_0\delta}{F'(\xi)}
\asymp_n(\delta^kQ)^{1/(k+1)}.
\end{aligned}
\end{equation}
Our assumption on $\delta^kQ$ ensures that each interval contains
$\gtrsim_n(\delta^kQ)^{1/(k+1)}$ integers, proving the counting estimate.

\label{lem:stationary-phase-parameters}
Let $Y\subset\mathbb R$ and $Z\subset\mathbb R^{d_0}$ be open sets of
diameter at most $10$, and let
$\phi\in C^\infty(Y\times Z;\mathbb R)$. Suppose that there is a smooth
map $y_*:Z\to Y$ such that, for every $\mathbf z\in Z$, the point
$y_*(\mathbf z)$ is the unique critical point of

\]

$w\in C^\infty(\mathbb R_{>0}\times\mathbb R\times\mathbb R^{d_0})$
be supported in $\mathbb R_{>0}\times Y\times Z$. If, for some
integer $A>1$,
\[
\sup_{\Lambda>0,\,y,\,\mathbf z}
\bigl|\partial_\Lambda^\iota\partial_y^N
\partial_{\mathbf z}^{\boldsymbol\beta}w(\Lambda;y;\mathbf z)\bigr|
\leq C_A(1+\Lambda)^{-\iota}
\quad
(0\leq N,\iota,|\boldsymbol\beta|\leq A),
\]
then, for $\Lambda\geq1$,
\[
\int_{\mathbb R}e\!\left(\Lambda\phi(y;\mathbf z)\right)
 w(\Lambda;y;\mathbf z)\,dy
=\Lambda^{-1/2}
 e\!\left(\Lambda\phi(y_*(\mathbf z);\mathbf z)\right)
 w_1(\Lambda;\mathbf z),
\]
where
\[
\bigl|\partial_\Lambda^\iota
\partial_{\mathbf z}^{\boldsymbol\beta}w_1(\Lambda;\mathbf z)\bigr|
\lesssim_{A,c_0,C_A,\phi,y_*}\Lambda^{-\iota}
\quad(0\leq\iota,|\boldsymbol\beta|\leq A).
\]
For a family of phases, the last constant is uniform provided the
finitely many required seminorms of $\phi$ and $y_*$ are uniform.

\label{lem:app-frenet-covering}
For each Frenet box $\mathcal P_{a,\boldsymbol\lambda,\boldsymbol\nu}$
above, set $A=\{i:\nu_i\ne0\}$. Then, for $|\xi|\sim R$,
\[
\mathbf1_{\mathcal P_{a,\boldsymbol\lambda,\boldsymbol\nu}}(B^T\xi)
\leq\sum_{\substack{\tau\in\mathcal I(\lambda_{\mathrm{prod}})\\
|t_\tau-a|\leq C_n\lambda_{\mathrm{prod}}}}
\mathbf1_{\Pi_{\tau,\boldsymbol\lambda,A}}(\xi).
\]
<!-- end lemma-13 -->
<!-- proof-19: body_md -->
The idea of the proof is to approximate the function $P_{\xi}$ by a polynomial of a fixed degree, and use the fundamental theorem of algebra. To do so, choose an integer $M\geq n+\lceil2/\beta\rceil$ and partition $I$ into
$O_{\beta,\gamma}(R^\beta)$ intervals of length at most $cR^{-\beta}$,
where $0<c\leq1$ is a sufficiently small fixed constant. On each such
interval $J$, let $p$ be the Taylor polynomial of order $M$ for $P_\xi$ at the
left endpoint. Since $|\xi|\sim R$ and $\gamma$ is smooth, Taylor's formula gives
\[
\max_{0\leq j\leq n}\|P_\xi^{(j)}-p^{(j)}\|_{L^\infty(J)}
\lesssim_{\beta,\gamma}R(cR^{-\beta})^{M+1-n}
\lesssim_{\beta,\gamma}R^{-1}.
\]
Since $0<\lambda_i\leq1$, we have $\lambda_{\mathrm{prod}}\leq1$.
Taking $c$ sufficiently small, we therefore have
\[
|p(t)|\lesssim R^\beta,\qquad |p^{(k)}(t)|\gtrsim \lambda_{\mathrm{prod}}^{-k}
\quad(t\in I_0\cap J).
\]
If $I_0\cap J$ is nonempty, the preceding lower bound implies that $p$ is nonconstant.
By the fundamental theorem of algebra, we may write
\[
p(t)=a\prod_{\ell=1}^{\deg p}(t-z_\ell),\qquad \deg p\leq M,
\]
where $z_1,\ldots,z_{\deg p}\in\mathbb C$ are the roots, counted with multiplicity.
For $p(t)\neq0$, the product rule gives
\[
|p^{(k)}(t)|\lesssim_M
|p(t)|\bigl(\min_{1\leq\ell\leq\deg p}|t-z_\ell|\bigr)^{-k}.
\]
Consequently every $t\in I_0\cap J$, including any zero of $p$, satisfies
\[
\min_{1\leq\ell\leq\deg p}|t-z_\ell|
\lesssim_{\beta,\gamma}R^{\beta/k}\lambda_{\mathrm{prod}}
\leq R^\beta \lambda_{\mathrm{prod}}.
\]
Thus $I_0\cap J$ is covered by at most $M$ real intervals centered at
$\operatorname{Re}z_\ell$, each of length $\lesssim_{\beta,\gamma}R^\beta \lambda_{\mathrm{prod}}$.
Since $I_0$ is $\lambda_{\mathrm{prod}}$-separated,
\[
\#(I_0\cap J)\lesssim_{\beta,\gamma}1+R^\beta
\lesssim_{\beta,\gamma}R^\beta.
\]
Summing over the $O_{\beta,\gamma}(R^\beta)$ intervals $J$ proves
\eqref{eq:app-derivative-packing}.
