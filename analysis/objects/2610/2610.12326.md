<!-- theorem-1: body_md -->
There is an orthogonal basis of $\Lambda[z^{-1}]$ indexed by pairs $(\lambda,s)$,
where $\lambda$ is a partition and $s\in\mathcal A_\lambda$ is an addable corner.
Its vectors $\Psi_\lambda^s$ satisfy
\[
\mathbf{L}\Psi_\lambda^s=[s]\Psi_\lambda^s,
\qquad
\pi_0\Psi_\lambda^s=J_\lambda,
\]
where $[s]=\mathsf t^i\mathsf q^{-j}$ is the content of the (zero-based) corner $s=(i,j)$.
Furthermore, the integral Macdonald function decomposes as
\begin{equation}
J_\lambda = \sum_{s \in \mathcal A_\lambda} \tau_\lambda^s \Psi_\lambda^s,
\end{equation}
where $\tau_\lambda^s$ are the Kerov transition measures
(Definition~\ref{def:kerov}).
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:main}
For all partitions $\mu,\nu$, addable corners $s\in\mathcal A_\mu$, $t\in\mathcal A_\nu$, the resolvent of the Macdonald Lax operator satisfies
\begin{equation}\label{eq:maintheorem}
\pi_\infty(u-\mathbf{L})^{-1}\bigl(\Psi_\mu^s\Psi_\nu^t\bigr)
=
\frac{\mathbf{T}_{\mu\star\nu}(u)}{u-[s+t]}\;  \pi_\infty\bigl(\Psi_\mu^s\Psi_\nu^t\bigr).
\end{equation}
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm:maintwo}
For all partitions $\mu,\nu$,
\begin{equation}\label{eq:kappa-multiplicativity}
\kappa_{u,v}(J_\mu J_\nu)
=\kappa_{u,v}(J_\mu)\kappa_{u,v}(J_\nu)
\mathbf{T}_{\mu\star\nu}(uv).
\end{equation}

<!-- lemma-1: body_md -->
\label{lem:A-corner-factor}
\[
\frac{A_\lambda(u)}{A_\lambda(u\mathsf t^{-1})}
=
\prod_{b\in\lambda}\mathcal N(u/[b])^{-1}.
\]
Equivalently,
\[
\frac{A_\lambda(u)}{A_\lambda(u\mathsf t^{-1})}
=
\frac{\,\,\,\,\,\,\prod\limits_{a\in\mathcal A_\lambda}\,\,\,\,\,\,(1-[a]/u)}
{\prod\limits_{b\in\mathcal R_\lambda^+\cup \{(0,0)\}}(1-[b]/u)}.
\]
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
For the boxes in row $i-1$,
\begin{align*}
\prod_{j=0}^{\lambda_i-1}\mathcal N(u/[i-1,j])^{-1}
&=
\frac{(1-\mathsf t^{i-1}\mathsf q^{-\lambda_i}/u)(1-\mathsf t^i/u)}
     {(1-\mathsf t^{i-1}/u)(1-\mathsf t^i\mathsf q^{-\lambda_i}/u)}\\
&=
\frac{(\mathsf q^{-\lambda_i}-u\mathsf t^{1-i})(1-u\mathsf t^{-i})}
     {(1-u\mathsf t^{1-i})(\mathsf q^{-\lambda_i}-u\mathsf t^{-i})}.
\end{align*}
This is the $i$th factor of $A_\lambda(u)/A_\lambda(u\mathsf t^{-1})$, and the
product over $i$ gives the first identity.  In the product over boxes the
factors shared by adjacent boxes cancel, which leaves the second form.

<!-- lemma-2: body_md -->
\label{lem:shift}
As operators on $\Lambda[z^{-1}]$,
\[
z\,\mathbf{L}^+\,z^{-1}=\mathsf q^{-1}\mathsf t\,\mathbf{L} .
\]
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
We start from \(\mathbf{L}^+=\mathcal D_\mathsf q^{-1}\mathcal L^{NS}=\mathcal D_\mathsf q^{-1}Q^*(z^{-1})\mathcal D_\mathsf t\pi_-Q(z)\). The series
\(Q(z)\) and \(Q^*(z^{-1})\) commute with multiplication by \(z\), so conjugation
by \(z(\cdot)z^{-1}\) only acts on the diagonal operators and the projection:
\[
z\,\mathcal D_\mathsf q^{-1}\,z^{-1}=\mathsf q^{-1}\mathcal D_\mathsf q^{-1},\qquad
z\,\mathcal D_\mathsf t\,z^{-1}=\mathsf t\,\mathcal D_\mathsf t,\qquad
z\,\pi_-\,z^{-1}=\pi_{\le 0}.
\]
Therefore
\[
z\,\mathbf{L}^+\,z^{-1}
=
\mathsf q^{-1}\mathcal D_\mathsf q^{-1}\,Q^*(z^{-1})\,\mathsf t\,\mathcal D_\mathsf t\,\pi_{\le 0}\,Q(z)
=
\mathsf q^{-1}\mathsf t\,\mathcal D_\mathsf q^{-1}\,Q^*(z^{-1})\,\mathcal D_\mathsf t\,\pi_{\le 0}\,Q(z)
=
\mathsf q^{-1}\mathsf t\,\mathbf{L},
\]
where the last step uses $\mathcal D_{\mathsf q^{-1}\mathsf t}=\mathcal D_\mathsf q^{-1}\mathcal D_\mathsf t$ and
$
Q^*(\mathsf q^{-1}z^{-1})\mathcal D_\mathsf q^{-1}
=
\mathcal D_\mathsf q^{-1}Q^*(z^{-1}).
$
<!-- end proof-2 -->
<!-- lemma-3: body_md -->
\label{lem:blockform}
With respect to \(\Lambda[z^{-1}]=\Lambda\oplus\Lambda^\infty\), we have
\[
\mathbf{L}
=
\begin{pmatrix}
1 & \mathcal Q\\
\mathcal D_\mathsf q^{-1}\mathcal Q^* & \mathcal D_\mathsf q^{-1}(\mathcal Q^*\mathcal Q+\mathcal L^{NS})
\end{pmatrix}
\]
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
By the previous lemma,
\[
\mathbf{L}=(\mathsf q\mathsf t^{-1})z\mathcal D_\mathsf q^{-1}\mathcal L^{NS}z^{-1}.
\]
For \(f\in\Lambda\) and \(\chi\in\Lambda^\infty\),
\[
\pi_-Q(z)z^{-1}(f+\chi)
=
z^{-1}\left(f+\mathcal Q\chi+\pi_-Q(z)\chi\right).
\]
Using \(\mathcal D_\mathsf t z^{-1}=\mathsf t z^{-1}\mathcal D_\mathsf t\) and
\[
\mathsf q z\mathcal D_\mathsf q^{-1}z^{-1}=\mathcal D_\mathsf q^{-1},
\]
we get
\[
\mathbf{L}(f+\chi)
=
\mathcal D_\mathsf q^{-1}Q^*(z^{-1})\mathcal D_\mathsf t
\left(f+\mathcal Q\chi+\pi_-Q(z)\chi\right).
\]
Since \(f+\mathcal Q\chi\in\Lambda\), this equals
\[
\begin{pmatrix}
f+\mathcal Q\chi\\
\mathcal D_\mathsf q^{-1}\mathcal Q^*(f+\mathcal Q\chi)
+\mathcal D_\mathsf q^{-1}\mathcal L^{NS}\chi
\end{pmatrix}.
\]

<!-- lemma-4: body_md -->
\label{lem:pi0homo}
For $\eta\in\Lambda=\pi_0\Lambda[z^{-1}]$ and $b\in\Lambda[z^{-1}]$,
\[
\mathbf{L}(\eta b)=\mathbf{L}(\eta)\mathbf{L}(b).
\]
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Since $\eta$ has no $z$-dependence, $\mathbf Q(\eta b)=\eta\mathbf Qb$ and
$\mathcal D_{\mathsf q^{-1}\mathsf t}\eta=\eta$.  The operator $\mathbf Q^\dagger$ is an exponential
of commuting derivations and hence preserves multiplication.  Therefore
\[
\mathbf{L}(\eta b)
=\mathbf Q^\dagger\bigl(\eta\mathcal D_{\mathsf q^{-1}\mathsf t}\mathbf Qb\bigr)
=(\mathbf Q^\dagger\eta)\mathbf{L}(b)
=\mathbf{L}(\eta)\mathbf{L}(b).
\]

\mathcal{T}(u) := \pi_0 (1-u^{-1}\mathbf{L})^{-1}\big|_{\Lambda}
=\sum_{k=0}^{\infty} u^{-k} \mathcal{T}_k ,
\qquad \mathcal T_k\in\operatorname{End}(\Lambda).

<!-- proposition-1: body_md -->
\label{cor:transfer-hierarchy}
For every partition $\lambda$,
\[
\mathcal{T}(u) J_\lambda = T_\lambda(u)J_\lambda,
\qquad
T_\lambda(u):=1+\frac{S_\lambda(u)}{u-1},
\qquad
S_\lambda(u) := \frac{A_\lambda(u\mathsf t^{-1})}{A_\lambda(u)} = \prod_{b\in\lambda}\mathcal N(u/[b]),
\]
and
\begin{equation}\label{eq:Tcorner}
T_\lambda(u)=1+u^{-1}\,
\frac{\prod_{b\in\mathcal R_\lambda^+}(1-[b]/u)}{\prod_{a\in\mathcal A_\lambda}(1-[a]/u)} .
\end{equation}
In particular the operators $\mathcal T_k$ commute and are diagonal on the
Macdonald polynomials.
<!-- end proposition-1 -->
<!-- proof-5: body_md -->
Set
\[
F=u^{-1}(1-u^{-1}\mathbf{L})^{-1}J_\lambda=f+\chi,
\qquad
f=\pi_0F,\quad \chi=\pi_-F.
\]
Using the block form of \(\mathbf{L}\), the equation \((u-\mathbf{L})F=J_\lambda\)
is equivalent to
$
(u-1)f-\mathcal Q\chi=J_\lambda
$
and
\[
(u\mathcal D_\mathsf q-\mathcal L^{NS})\chi
=
\mathcal Q^*(f+\mathcal Q\chi).
\]
The first equation gives \(f+\mathcal Q\chi=uf-J_\lambda\).  Substituting this into
the second equation and applying \(\mathcal Q(u\mathcal D_\mathsf q-\mathcal L^{NS})^{-1}\) gives
$
\mathcal Q\chi=I(u)(uf-J_\lambda).
$
Substitution back into the first equation gives
\[
\bigl((u-1)-uI(u)\bigr)f=(1-I(u))J_\lambda.
\]
By Theorem~\ref{thm:nsmain} and \eqref{eq:Aeigen}, \(I(u)\) is diagonal in the Macdonald basis, and
on \(J_\lambda\)
\[
I(u)=\frac{u-1}{u}\bigl(1-S_\lambda(u)^{-1}\bigr).
\]
Hence
\[
(u-1)-uI(u)=(u-1)S_\lambda(u)^{-1},
\qquad
1-I(u)=\frac{1+(u-1)S_\lambda(u)^{-1}}{u}.
\]
Therefore
\[
f
=
\frac{1+(u-1)S_\lambda(u)^{-1}}
{u(u-1)S_\lambda(u)^{-1}}
J_\lambda,
\qquad
\mathcal{T}(u)J_\lambda
=
u f
=
\left(
1+\frac{S_\lambda(u)}{u-1}
\right)J_\lambda
=
T_\lambda(u)J_\lambda .
\]
The corner form \eqref{eq:Tcorner} follows from
$S_\lambda(u)=(1-1/u)\prod_{b\in\mathcal R^+_\lambda}(1-[b]/u)/\prod_{a\in\mathcal A_\lambda}(1-[a]/u)$,
which is Lemma~\ref{lem:A-corner-factor}.

<!-- definition-5: body_md -->
\label{def:kerov}
The \emph{transition measure}  of $\lambda$ is the family
$\{\tau_\lambda^s\}_{s\in\mathcal A_\lambda}$ determined by
\[
T_\lambda(u)
=
\sum_{s\in\mathcal A_\lambda}\frac{\tau_\lambda^s}{1-[s]/u} .
\]
Explicitly,
\[
\tau_\lambda^s
=\frac{1}{[s]}\,
\frac{\prod_{b\in\mathcal R^+_\lambda}(1-[b-s])}
{\prod_{a\in\mathcal A_\lambda,\,a\ne s}(1-[a-s])}.
\]
The \emph{co-transition measure} at an outer corner $b\in\mathcal R^+_\lambda$ is
\[
\tilde\tau_\lambda^{\,b}
:=\frac{\prod_{a\in\mathcal A_\lambda}(1-[a-b])}
{\prod_{b'\in\mathcal R^+_\lambda,\,b'\ne b}(1-[b'-b])}.
\]
<!-- end definition-5 -->
<!-- remark-1: body_md -->
\label{rem:kerov}
We follow Kerov (see also
\cite{Kerov:1998}) and \cite{MicklerMoll:2023} in these names: the transition
measure lives on the addable corners and the co-transition measure on the outer
corners.  Applied to \eqref{eq:Tminus1},
his expansion \cite[(2.1)]{Kerov:2000} would give the weights $[s]\tau_\lambda^s$; both families have
total mass $1$ (Lemma~\ref{lem:kerovmoments}), and they agree in the Jack limit,
where every $[s]\to1$.  Likewise $\tilde\tau_\lambda^{\,b}=-\nu_b/[b]^2$,
where $1/(T_\lambda(u)-1)=u-c-\sum_{b\in\mathcal R^+_\lambda}\nu_b/(u-[b])$ is Kerov's
expansion \cite[(2.5)]{Kerov:2000}.  Kerov normalizes the $\nu_b$ to total
mass $1$; we do not.
<!-- end remark-1 -->
<!-- lemma-5: body_md -->
\label{lem:kerovmoments}
For every partition $\lambda$,
\[
\sum_{a\in\mathcal A_\lambda}\tau_\lambda^a=1,
\qquad
\sum_{a\in\mathcal A_\lambda}[a]\,\tau_\lambda^a=1,
\]
and, as identities of rational functions of $u$,
\[
\sum_{a\in\mathcal A_\lambda}\frac{\tau_\lambda^a[a]}{1-[a]/u}=u(T_\lambda(u)-1),
\qquad
\frac{1}{u(T_\lambda(u)-1)}=1-\frac1u
+\sum_{b\in\mathcal R^+_\lambda}\frac{[b]\,\tilde\tau_\lambda^{\,b}}{u-[b]}.
\]
For $s\in\mathcal A_\lambda$,
\begin{equation}\label{eq:pfvalues}
\sum_{b\in\mathcal R^+_\lambda}\frac{[b]\,\tilde\tau_\lambda^{\,b}}{[b]-[s]}=1-\frac1{[s]}.
\end{equation}
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
By \eqref{eq:Tcorner}, $T_\lambda(u)=1+u^{-1}+O(u^{-2})$ as $u\to\infty$, while
Definition~\ref{def:kerov} gives
$T_\lambda(u)=\sum_a\tau_\lambda^a\,u/(u-[a])
=\sum_a\tau_\lambda^a+u^{-1}\sum_a[a]\tau_\lambda^a+O(u^{-2})$.  Comparing
coefficients gives the first two identities.  The third follows from
$[a]/(1-[a]/u)=u/(1-[a]/u)-u$, Definition~\ref{def:kerov} and the first.
The left side of the fourth is $1$ at $u=\infty$, has residue $-1$ at $u=0$
because $T_\lambda(0)=0$, and has residue $[b]\tilde\tau_\lambda^{\,b}$ at
$u=[b]$ by \eqref{eq:cotau-residue}.  Finally, \eqref{eq:pfvalues} is the
fourth identity at $u=[s]$, where $T_\lambda$ has a pole.

<!-- proposition-2: body_md -->
\label{prop:spectral}
For a partition $\lambda\ne\varnothing$ and $s \in \mathcal A_\lambda$, let
$\Psi_\lambda^s\in\Lambda[z^{-1}]$ be the value at $u=[s]$ of the $\Lambda[z^{-1}]$-valued
rational function of $u$
\[ J_\lambda + \bigl(\mathcal D_\mathsf q-\mathcal L^{NS}/u\bigr)^{-1}\mathcal Q^*J_\lambda . \]
For the empty
partition, set $\Psi_\varnothing^{(0,0)}=1$.  Then $\Psi_\lambda^s$ is well
defined,
\[
\mathbf{L} \Psi_\lambda^{s}
=
[s] \Psi_\lambda^{s},
\qquad
\pi_0\Psi_\lambda^s=J_\lambda,
\]
and the vectors $\Psi_\lambda^s$, over all partitions $\lambda$ and all
$s\in\mathcal A_\lambda$, form a basis of $\Lambda[z^{-1}]$.
<!-- end proposition-2 -->
<!-- proof-7: body_md -->
Work in a fixed total degree.  The empty-partition case is immediate.  Let
$\lambda\ne\varnothing$.  By Theorem~\ref{thm:nsmain} and \eqref{eq:Aeigen},
\[
uI(u)J_\lambda=(u-1)\left(1-\frac{A_\lambda(u)}{A_\lambda(u\mathsf t^{-1})}\right)J_\lambda .
\]
By Lemma~\ref{lem:A-corner-factor}, the poles of
$A_\lambda(u)/A_\lambda(u\mathsf t^{-1})$ lie at the contents $[b]$ with
$b\in\mathcal R^+_\lambda\cup\{(0,0)\}$, and the pole at $[0,0]=1$ is cancelled by the
factor $u-1$.  Since $s\notin\mathcal R^+_\lambda\cup\{(0,0)\}$ and distinct boxes have
distinct contents, $uI(u)J_\lambda$ is regular at $u=[s]$.  Moreover the ratio
vanishes at $u=[s]$, so the value there is $([s]-1)J_\lambda$.

By the block form, the
adjoint of $\mathcal Q$ for $\left\langle \cdot,\cdot\right\rangle_\infty$ is $\mathcal D_\mathsf q^{-1}\mathcal Q^*$, and $\mathbf{L}^+$
is self-adjoint, being the lower-right block of $\mathbf{L}$ minus
$\mathcal D_\mathsf q^{-1}\mathcal Q^*\mathcal Q$.  Temporarily choose real $0<\mathsf q,\mathsf t<1$ away from poles and
content coincidences, so that the scalar product is positive definite.  Put
$v_\lambda=\mathcal D_\mathsf q^{-1}\mathcal Q^* J_\lambda$.  Then
\[
\langle J_\lambda,I(u)J_\lambda\rangle_{\mathsf q,\mathsf t}
=\langle v_\lambda,(u-\mathbf{L}^+)^{-1}v_\lambda\rangle_\infty.
\]
In the spectral decomposition of $\mathbf{L}^+$, the residue on the right at each
eigenvalue is the squared norm of the corresponding component of
$v_\lambda$.  Regularity at $u=[s]$ thus implies that
$(u-\mathbf{L}^+)^{-1}v_\lambda$ is regular there as well.  This holds over
$\mathbb Q(\mathsf q,\mathsf t)$: the entries of $(u-\mathbf{L}^+)^{-1}v_\lambda$ are rational functions of
$\mathsf q,\mathsf t,u$, and a pole of an entry along $u=[s]$ would persist after specializing
$(\mathsf q,\mathsf t)$ to a Zariski-dense set of points, hence to points of the open set
chosen above.  Consequently the rational vector
\[
\psi(u):=u(u-\mathbf{L}^+)^{-1}v_\lambda=(\mathcal D_\mathsf q-\mathcal L^{NS}/u)^{-1}\mathcal Q^*J_\lambda
\]
is regular at $u=[s]$.

Now write $\Psi_\lambda^s=J_\lambda+\psi$ with $\psi=\psi([s])\in\Lambda^\infty$, so
that $\pi_0\Psi_\lambda^s=J_\lambda$.  Since $\mathcal Q\psi(u)=uI(u)J_\lambda$ for
generic $u$, the first paragraph gives
\[
\mathcal Q\psi=([s]-1)J_\lambda .
\]
Moreover, $(\mathcal D_\mathsf q-\mathcal L^{NS}/u)\psi(u)=\mathcal Q^*J_\lambda$ identically in $u$, so
$\mathcal D_\mathsf q^{-1}\mathcal L^{NS}\psi(u)=u\psi(u)-u\mathcal D_\mathsf q^{-1}\mathcal Q^*J_\lambda$, and evaluating at
$u=[s]$ gives
\[
\mathcal D_\mathsf q^{-1}\mathcal L^{NS}\psi=[s]\psi-[s]\mathcal D_\mathsf q^{-1}\mathcal Q^*J_\lambda .
\]
By the block form, the $\Lambda$-component of $\mathbf{L}\Psi_\lambda^s$ is
$J_\lambda+\mathcal Q\psi=[s]J_\lambda$, and its $\Lambda^\infty$-component is
\[
\mathcal D_\mathsf q^{-1}\mathcal Q^*J_\lambda+\mathcal D_\mathsf q^{-1}\mathcal Q^*\mathcal Q\psi+\mathcal D_\mathsf q^{-1}\mathcal L^{NS}\psi
=\mathcal D_\mathsf q^{-1}\mathcal Q^*J_\lambda+([s]-1)\mathcal D_\mathsf q^{-1}\mathcal Q^*J_\lambda
+[s]\psi-[s]\mathcal D_\mathsf q^{-1}\mathcal Q^*J_\lambda
=[s]\psi .
\]
Hence $\mathbf{L}\Psi_\lambda^s=[s]\Psi_\lambda^s$.

It remains to prove that these eigenvectors form a basis.  Fix a total degree
$N$ and suppose that
\[
\sum_{\lambda\vdash N}\sum_{s\in\mathcal A_\lambda}
 c_{\lambda,s}\Psi_\lambda^s=0.
\]
For each corner content $a$ occurring in degree $N$, apply
$\prod_{b\ne a}(\mathbf{L}-b)$, the product over the other corner contents $b$
occurring in degree $N$, and then project with $\pi_0$.  After dividing by the resulting nonzero common
scalar, and using $\pi_0\Psi_\lambda^s=J_\lambda$, this gives
\[
\sum_{\substack{\lambda\vdash N,\ s\in\mathcal A_\lambda\\ [s]=a}}
 c_{\lambda,s}J_\lambda=0.
\]
The addable corners of a fixed partition have distinct contents, so at most one
term occurs for each $J_\lambda$.  Their linear independence implies
$c_{\lambda,s}=0$ for every pair $(\lambda,s)$.

It remains only to count the vectors.  If $p(N)$ is the number of partitions of
$N$, then
\[
\sum_{N\ge0}\left(\sum_{\lambda\vdash N}|\mathcal A_\lambda|\right)x^N
=\frac{1}{1-x}\prod_{m\ge1}\frac{1}{1-x^m}.
\]
Indeed, $|\mathcal A_\lambda|$ is one plus the number of distinct positive parts of
$\lambda$.  Hence
\[
\sum_{\lambda\vdash N}|\mathcal A_\lambda|
=\sum_{k=0}^{N}p(k)
=\dim\Lambda[z^{-1}]_N.
\]
<!-- end proof-7 -->
<!-- proposition-3: body_md -->
\label{prop:jackpsiexp}
For every partition $\lambda$,
\begin{equation}\label{eq:jackpsiexp}
J_\lambda=\sum_{s\in\mathcal A_\lambda}\tau_\lambda^s\Psi_\lambda^s .
\end{equation}
<!-- end proposition-3 -->
<!-- proof-8: body_md -->
Expand $J_\lambda=\sum_{\mu\vdash|\lambda|}\sum_{r\in\mathcal A_\mu}c_{\mu,r}\Psi_\mu^r$ in
the basis of Proposition~\ref{prop:spectral}.  Applying $\pi_0(1-\mathbf{L}/u)^{-1}$
and using Proposition~\ref{cor:transfer-hierarchy},
\[
T_\lambda(u)J_\lambda=\sum_{\mu\vdash|\lambda|}\Bigl(\sum_{r\in\mathcal A_\mu}\frac{c_{\mu,r}}{1-[r]/u}\Bigr)J_\mu .
\]
The addable corners of $\mu$ have distinct contents, so the functions
$(1-[r]/u)^{-1}$, $r\in\mathcal A_\mu$, are linearly independent.  Comparing
coefficients of $J_\mu$ gives $c_{\mu,r}=0$ for $\mu\ne\lambda$, and for
$\mu=\lambda$ Definition~\ref{def:kerov} gives $c_{\lambda,s}=\tau_\lambda^s$.
<!-- end proof-8 -->
<!-- proposition-4: body_md -->
\label{prop:orthogonal}
The basis $\{\Psi_\lambda^s\}$ of Proposition~\ref{prop:spectral} is orthogonal
for $\left\langle \cdot,\cdot\right\rangle_\infty$.  More precisely, for a partition $\lambda$ put
\[
Z_\lambda:=\{f(\mathbf{L})J_\lambda:f\in\mathbb Q(\mathsf q,\mathsf t)[w]\}.
\]
Then $Z_\lambda=\operatorname{span}\{\Psi_\lambda^s:s\in\mathcal A_\lambda\}$, the
spaces $Z_\lambda$ are mutually orthogonal, and within each $Z_\lambda$ the
vectors $\Psi_\lambda^s$ are orthogonal.
<!-- end proposition-4 -->
<!-- proof-9: body_md -->
The argument follows the cyclic-space proof of
\cite[Section~3]{MicklerMoll:2023}.  Since the eigenvalues $[s]$,
$s\in\mathcal A_\lambda$, are distinct, Proposition~\ref{prop:jackpsiexp} and Lagrange
interpolation give
\[
\Psi_\lambda^s
=\frac{1}{\tau_\lambda^s}
 \prod_{\substack{r\in\mathcal A_\lambda\\r\ne s}}
 \frac{\mathbf{L}-[r]}{[s]-[r]}J_\lambda,
\]
so $Z_\lambda=\operatorname{span}\{\Psi_\lambda^s:s\in\mathcal A_\lambda\}$.  Here
$\tau_\lambda^s\ne0$, since no factor in its formula in
Definition~\ref{def:kerov} vanishes: distinct boxes have distinct contents.  By
Proposition~\ref{cor:transfer-hierarchy}, $\pi_0f(\mathbf{L})J_\lambda\in\mathbb Q(\mathsf q,\mathsf t)J_\lambda$
for every polynomial $f$.  If $\lambda\ne\mu$, $F=f(\mathbf{L})J_\lambda\in Z_\lambda$
and $G=g(\mathbf{L})J_\mu\in Z_\mu$, then self-adjointness of $\mathbf{L}$ gives
\begin{align*}
\langle F,G\rangle_\infty
&=\langle J_\lambda,f(\mathbf{L})g(\mathbf{L})J_\mu\rangle_\infty\\
&=\langle J_\lambda,\pi_0f(\mathbf{L})g(\mathbf{L})J_\mu\rangle_{\mathsf q,\mathsf t}=0,
\end{align*}
because the $J_\lambda$ are orthogonal.  Thus $Z_\lambda\perp Z_\mu$ for
$\lambda\ne\mu$.  Within a fixed $Z_\lambda$, the vectors $\Psi_\lambda^s$ have
distinct eigenvalues and are therefore orthogonal by self-adjointness of
$\mathbf{L}$.

<!-- lemma-6: body_md -->
\label{lem:transmeas}
The transition measure admits the hook product form
\begin{equation}\label{eq:tauhooks}
\tau_\lambda^s
=
\mathsf q^{a'(s)}
\prod_{\substack{b\in\lambda\\ b\in \mathrm{Row}_s(\lambda)}}
\frac{h_\lambda(b)}{h_{\lambda+s}(b)}
\prod_{\substack{b\in\lambda\\ b\in \mathrm{Col}_s(\lambda)}}
\frac{h'_\lambda(b)}{h'_{\lambda+s}(b)},
\end{equation}
where \(\mathrm{Row}_s(\lambda)\) is the set of boxes of \(\lambda\) in the same row as
\(s\), \(\mathrm{Col}_s(\lambda)\) the set of boxes in the same column as \(s\), and
$h,h'$ are the hook factors of Section~\ref{ssec:notation}.
<!-- end lemma-6 -->
<!-- proof-11: body_md -->
In the corner coordinates of Section~\ref{ssec:notation}, write
\[
a_k=(I_k,L_{k+1})\quad(0\le k\le m),
\qquad
r_k=(I_k,L_k)\quad(1\le k\le m)
\]
for the addable and outer corners.  Fix
\[
s=a_p=(I_p,L_{p+1})=(i,j).
\]
In these coordinates, the formula for $\tau_\lambda^s$ in
Definition~\ref{def:kerov} reads
\begin{equation}
\tau_\lambda^s
=\frac{1}{[s]}
 \frac{\displaystyle\prod_{k=1}^{m}(1-[r_k]/[s])}
      {\displaystyle\prod_{\substack{0\le k\le m\\k\ne p}}
       (1-[a_k]/[s])}.
\label{eq:tau-corner-product}
\end{equation}

We now compute the two hook products in \eqref{eq:tauhooks}.  First consider the
boxes $(i,c)$ in the row of $s$.  For
\[
L_{k+1}\le c<L_k,
\qquad p+1\le k\le m,
\]
one has $\ell_\lambda(i,c)=I_k-I_p-1$, while adding $s$ increases the arm by
one.  Hence the factors in each such block telescope:
\begin{align*}
\prod_{c=L_{k+1}}^{L_k-1}
\frac{h_\lambda(i,c)}{h_{\lambda+s}(i,c)}
&=
\frac{1-\mathsf q^{j-L_k}\mathsf t^{I_k-I_p}}
     {1-\mathsf q^{j-L_{k+1}}\mathsf t^{I_k-I_p}}\\
&=
\frac{1-[r_k]/[s]}{1-[a_k]/[s]}.
\end{align*}
Therefore
\begin{equation}
\prod_{b\in \mathrm{Row}_s(\lambda)}
\frac{h_\lambda(b)}{h_{\lambda+s}(b)}
=
\prod_{k=p+1}^{m}
\frac{1-[r_k]/[s]}{1-[a_k]/[s]}.
\label{eq:tau-row-product}
\end{equation}

Next consider the boxes $(c,j)$ in the column of $s$.  For
\[
I_{k-1}\le c<I_k,
\qquad 1\le k\le p,
\]
one has $a_\lambda(c,j)=L_k-j-1$, while adding $s$ increases the leg by one.
Thus
\begin{align*}
\prod_{c=I_{k-1}}^{I_k-1}
\frac{h'_\lambda(c,j)}{h'_{\lambda+s}(c,j)}
&=
\frac{1-\mathsf q^{L_k-j}\mathsf t^{I_p-I_k}}
     {1-\mathsf q^{L_k-j}\mathsf t^{I_p-I_{k-1}}}\\
&=
\frac{1-[s]/[r_k]}{1-[s]/[a_{k-1}]}\\
&=
\frac{[a_{k-1}]}{[r_k]}
\frac{1-[r_k]/[s]}{1-[a_{k-1}]/[s]}.
\end{align*}
Since
\[
\prod_{k=1}^{p}\frac{[a_{k-1}]}{[r_k]}
=
\prod_{k=1}^{p}\mathsf t^{I_{k-1}-I_k}
=\mathsf t^{-I_p}=\mathsf t^{-i},
\]
we obtain
\begin{equation}
\prod_{b\in \mathrm{Col}_s(\lambda)}
\frac{h'_\lambda(b)}{h'_{\lambda+s}(b)}
=
\mathsf t^{-i}
\prod_{k=1}^{p}
\frac{1-[r_k]/[s]}{1-[a_{k-1}]/[s]}.
\label{eq:tau-column-product}
\end{equation}

Multiplying \eqref{eq:tau-row-product} and
\eqref{eq:tau-column-product} accounts for every shifted removable corner and
every addable corner except $a_p=s$.  Since $[s]=\mathsf t^i\mathsf q^{-j}$,
\eqref{eq:tau-corner-product} becomes
\[
\tau_\lambda^s
=\mathsf q^j
\prod_{b\in \mathrm{Row}_s(\lambda)}
\frac{h_\lambda(b)}{h_{\lambda+s}(b)}
\prod_{b\in \mathrm{Col}_s(\lambda)}
\frac{h'_\lambda(b)}{h'_{\lambda+s}(b)}.
\]
Finally $j=a'(s)$, proving \eqref{eq:tauhooks}.

\label{lem:phivectors}
Let $\gamma\ne\varnothing$.
\begin{enumerate}
\item[(i)] For $r\in\mathcal R_\gamma$, $\varphi_\gamma^r\in\Lambda^\infty$ and

Fix $r$ and put $c_a=\tau_\gamma^a[a]/([a]-[r^+])$,
so that $\varphi_\gamma^r=\sum_ac_a\Psi_\gamma^a$.  By \eqref{eq:Tminus1},
$T_\gamma([r^+])=1$, so Lemma~\ref{lem:kerovmoments} gives
$\sum_ac_a=1-T_\gamma([r^+])=0$.  Hence
$\pi_0\varphi_\gamma^r=(\sum_ac_a)J_\gamma=0$ and $\varphi_\gamma^r\in\Lambda^\infty$.

\[
\sum_ac_a[a]=\sum_a\tau_\gamma^a[a]+[r^+]\sum_a\frac{\tau_\gamma^a[a]}{[a]-[r^+]}

<!-- proof-18: body_md -->
By induction on $|\lambda|$, following \cite[Theorem~3.4]{Mickler:2023} with
Lemma~\ref{lem:pi0homo} in place of the derivation property used there, we
show that $z^{-1}\Psi_\lambda^s=\mathsf q^{-a'(s)}\varphi_{\lambda+s}^s$.

If $\lambda=\varnothing$ then $s=(0,0)$ and $\Psi_\varnothing^{(0,0)}=1$.  The
space $\Lambda^\infty_1$ is spanned by $z^{-1}$, and $\mathcal Q z^{-1}=Q_1=(1-\mathsf t)p_1=J_{(1)}$,
so Lemma~\ref{lem:phivectors}(ii) forces $\varphi_{(1)}^{(0,0)}=z^{-1}$.

Now let $|\lambda|=n\ge1$ and assume the statement for all partitions of size
$n-1$.  By Corollary~\ref{cor:NSspectral}, $\mathbf{L}^+$ is diagonalizable on
$\Lambda^\infty_{n+1}$, with eigenvectors $z^{-1}\Psi_\mu^a$ ($\mu\vdash n$,
$a\in\mathcal A_\mu$) of eigenvalue $[a^+]$.  Let $P$ be the
projection of $\Lambda^\infty_{n+1}$ onto the $[s^+]$-eigenspace of $\mathbf{L}^+$
along the other eigenspaces, and let $P'$ be the projection of
$\Lambda[z^{-1}]_n$ onto the $[s]$-eigenspace of $\mathbf{L}$.  Distinct boxes have
distinct contents, so $[a^+]=[s^+]$ only when $a=s$.  Hence $z^{-1}$
maps the $[s]$-eigenspace of $\mathbf{L}$ onto the $[s^+]$-eigenspace of
$\mathbf{L}^+$, and the sum of the other eigenspaces of $\mathbf{L}$ onto the sum of the
other eigenspaces of $\mathbf{L}^+$, so $Pz^{-1}=z^{-1}P'$.

By the Pieri rule \cite[Chapter~VI, (6.24)(iv) and (6.23)]{Macdonald:1995},
together with $J_\mu=\prod_{b\in\mu}h_\mu(b)\,P_\mu$ \cite[Chapter~VI, (8.3)]{Macdonald:1995},
\[
J_{(1)}J_\lambda=\sum_{a\in\mathcal A_\lambda}g^aJ_{\lambda+a},
\qquad g^a\ne0 .
\]
Apply $\mathbf{L}$.  By Lemma~\ref{lem:pi0homo} the left side becomes
$\mathbf{L}(J_{(1)})\mathbf{L}(J_\lambda)$.  The block form gives
$\mathbf{L} J_\gamma=J_\gamma+\mathcal D_\mathsf q^{-1}\mathcal Q^*J_\gamma$ for every $\gamma$, and in
particular $\mathbf{L} J_{(1)}=J_{(1)}+cz^{-1}$ with $c=(1-\mathsf t)(1-\mathsf q)/\mathsf q$.  Subtracting
$J_{(1)}J_\lambda$ from both sides gives
\begin{equation}\label{eq:creation-pieri}
\sum_{a\in\mathcal A_\lambda}g^a\,\mathcal D_\mathsf q^{-1}\mathcal Q^*J_{\lambda+a}
=J_{(1)}\,\mathcal D_\mathsf q^{-1}\mathcal Q^*J_\lambda+c\,z^{-1}\mathbf{L} J_\lambda .
\end{equation}
We apply $P$ to the three terms.

By Lemma~\ref{lem:phivectors}(i) and (iii), $\mathcal D_\mathsf q^{-1}\mathcal Q^*J_{\lambda+a}$ is a
combination of eigenvectors $\varphi_{\lambda+a}^r$ with eigenvalues
$[r^+]$, $r\in\mathcal R_{\lambda+a}$.  Such an eigenvalue equals $[s^+]$ only
if $r=s$, and $s\in\mathcal R_{\lambda+a}$ only if $a=s$.  So the left side of
\eqref{eq:creation-pieri} projects to
$-g^s[s^+]\,\tilde\tau_{\lambda+s}^{\,s^+}\varphi_{\lambda+s}^s$, and this
coefficient is nonzero: no factor of $\tilde\tau_{\lambda+s}^{\,s^+}$
vanishes, because $s^+\notin\mathcal A_{\lambda+s}$.

By \eqref{eq:jackpsiexp},
$c\,z^{-1}\mathbf{L} J_\lambda=c\sum_a\tau_\lambda^a[a]\,z^{-1}\Psi_\lambda^a$ projects to
$c\,\tau_\lambda^s[s]\,z^{-1}\Psi_\lambda^s$.

For the remaining term, Lemma~\ref{lem:phivectors}(iii) for $\lambda$ and the
induction hypothesis $\varphi_\lambda^r=\mathsf q^{a'(r)}z^{-1}\Psi_{\lambda-r}^r$ give
\[
J_{(1)}\,\mathcal D_\mathsf q^{-1}\mathcal Q^*J_\lambda
=-\sum_{r\in\mathcal R_\lambda}[r^+]\,\tilde\tau_\lambda^{\,r^+}\mathsf q^{a'(r)}\,
z^{-1}\bigl(J_{(1)}\Psi_{\lambda-r}^r\bigr).
\]
By Lemma~\ref{lem:pi0homo},
$\mathbf{L}(J_{(1)}\Psi_{\lambda-r}^r)=(J_{(1)}+cz^{-1})[r]\Psi_{\lambda-r}^r$, that is,
\[
(\mathbf{L}-[r])\bigl(J_{(1)}\Psi_{\lambda-r}^r\bigr)=c[r]\,z^{-1}\Psi_{\lambda-r}^r
=c[r]\,\mathsf q^{-a'(r)}\varphi_\lambda^r .
\]
Apply $P'$, which commutes with $\mathbf{L}$.  Since $\varphi_\lambda^r\in Z_\lambda$
and the addable corners of $\lambda$ have distinct contents, $P'$ keeps only
the $\Psi_\lambda^s$-component of $\varphi_\lambda^r$, and the definition of
$\varphi_\lambda^r$ gives
$P'\varphi_\lambda^r=\tau_\lambda^s[s]\,\Psi_\lambda^s/([s]-[r^+])$; since
$[s]\ne[r]$ and $[s]\ne[r^+]$,
\[
P'\bigl(J_{(1)}\Psi_{\lambda-r}^r\bigr)
=\frac{c[r]\,\mathsf q^{-a'(r)}\tau_\lambda^s[s]}{([s]-[r])([s]-[r^+])}\,\Psi_\lambda^s .
\]
Thus both terms on the right of \eqref{eq:creation-pieri} project to multiples
of $z^{-1}\Psi_\lambda^s$, so $\varphi_{\lambda+s}^s$ is a multiple of
$z^{-1}\Psi_\lambda^s$.  Since $\mathcal Q\varphi_{\lambda+s}^s=J_{\lambda+s}\ne0$, the
multiple is nonzero, and $z^{-1}\Psi_\lambda^s$ is a multiple of
$\varphi_{\lambda+s}^s$.  Applying $\pi_\infty$, Proposition~\ref{prop:piinftypsi}
and Lemma~\ref{lem:phivectors}(ii) identify the multiple as
$([s]-1)\varpi_\lambda/\varpi_{\lambda+s}=\mathsf q^{-a'(s)}$, by
\eqref{eq:varpi-ratio}.

For \eqref{eq:hat-zcreation}, let first $\lambda\ne\varnothing$.  The
coefficient in \eqref{eq:zcreation} is multiplied by
\[
\frac{([d]-1)\varpi_{\lambda+s}}{([s]-1)\varpi_\lambda},
\qquad
\mathsf q^{-a'(s)}\frac{\varpi_{\lambda+s}}{([s]-1)\varpi_\lambda}=1,
\]
by \eqref{eq:varpi-ratio}.  For
$\lambda=\varnothing$ we have $\widehat\Psi_\varnothing^{(0,0)}=\Psi_\varnothing^{(0,0)}=1$
and $a'(s)=0$, and the multiplier is $([d]-1)\varpi_{(1)}=[d]-1$, which gives the
same formula.  Applying $\pi_\infty$ to \eqref{eq:hat-zcreation} and using
$\pi_\infty\widehat\Psi_\gamma^c=z^{-|\gamma|}$ (Definition~\ref{def:hatpsi})
shows that the coefficients sum to $1$.

<!-- definition-7: body_md -->

<!-- definition-8: body_md -->
\label{def:beta}
For $f,g\in\Lambda[z^{-1}]$, define the \emph{normalized product defect}
\begin{equation}\label{eq:beta-def}
\beta(f,g):=\mathbf{L}^{-1}\bigl(\mathbf{L}(fg)-\mathbf{L}(f)\mathbf{L}(g)\bigr)
=fg-\mathbf{L}^{-1}\bigl(\mathbf{L}(f)\mathbf{L}(g)\bigr).
\end{equation}
The inverse exists degree by degree, since the eigenvalues of $\mathbf{L}$ are
nonzero corner contents.

<!-- lemma-9: body_md -->
\label{lem:beta-structure}
For $f,g\in\Lambda[z^{-1}]$,
\begin{align}
\beta(f,g)
&=fg-\mathbf Q^{-1}\bigl((\mathbf Qf)(\mathbf Qg)\bigr)
\label{eq:beta-Q}\\
&=\mathbf Q(fg)-(\mathbf Qf)g-f(\mathbf Qg)+fg.
\label{eq:beta-coboundary}
\end{align}
In particular, $\beta$ is symmetric and $\Lambda$-bilinear, vanishes if either
argument belongs to $\Lambda$, and satisfies $\pi_\infty\beta(f,g)=0$.
<!-- end lemma-9 -->
<!-- proof-20: body_md -->
The operator $\mathbf Q$ is $\Lambda$-linear, and $\mathbf Q-\operatorname{Id}$ strictly
lowers the degree in $z^{-1}$, so $\mathbf Q$ has a $\Lambda$-linear inverse.
Since $p_n^\perp=n\,\partial/\partial p_n$, the exponent defining
$\mathbf Q^\dagger$ is a derivation that lowers the ordinary polynomial
degree in the power sums.  Its exponential is therefore locally finite,
and Taylor's formula gives the simultaneous substitution
\[
\mathbf Q^\dagger F
=\left.F\right|_{p_n\mapsto p_n+(\mathsf q^{-n}-1)z^{-n}\ (n\ge1)},
\qquad F\in\Lambda[z^{-1}],
\]
with $z^{-1}$ fixed.  This substitution preserves products, and reversing
the translations gives its inverse.  Thus $\mathbf Q^\dagger$ is a
$\mathbb Q(\mathsf q,\mathsf t)$-algebra automorphism.  The operator $\mathcal D_{\mathsf q^{-1}\mathsf t}$ is also
an algebra automorphism, since it rescales $z^{-1}$.  The factorization
\eqref{eq:Laxdef} gives
\[
\mathbf{L}^{-1}\bigl(\mathbf{L}(f)\mathbf{L}(g)\bigr)
=\mathbf Q^{-1}\bigl((\mathbf Qf)(\mathbf Qg)\bigr),
\]
which proves \eqref{eq:beta-Q}.  This expression is $\Lambda$-bilinear and
vanishes when either argument is in $\Lambda$, because $\mathbf Q$ fixes $\Lambda$.

To prove \eqref{eq:beta-coboundary}, it therefore suffices to consider
$f=z^{-m}$, $g=z^{-n}$.  The identity
\[
\mathbf Qz^{-m-1}=z^{-1}\mathbf Qz^{-m}+Q_{m+1}
\]
and \eqref{eq:beta-Q} give, for $m\ge0$, $n\ge1$,
\[
\beta(z^{-m},z^{-n})-\beta(z^{-m-1},z^{-n+1})
=Q_{m+1}z^{-n+1}-Q_nz^{-m}.
\]
Iterating down to $\beta(z^{-m-n},1)=0$ yields
\begin{equation}\label{eq:beta-monomials}
\beta(z^{-m},z^{-n})
=\sum_{j=0}^{n-1}
\left(Q_{m+j+1}z^{-(n-j-1)}-Q_{n-j}z^{-(m+j)}\right).
\end{equation}
Comparing this with
$\mathbf Qz^{-k}=\sum_{i=0}^kQ_i z^{-(k-i)}$ proves
\eqref{eq:beta-coboundary} on monomials, hence in general by
$\Lambda$-bilinearity.  Finally, $\pi_\infty Q_i=0$ for $i\ge1$, so
\eqref{eq:beta-monomials} also gives $\pi_\infty\beta(f,g)=0$.
<!-- end proof-20 -->
<!-- lemma-10: body_md -->
\label{lem:beta-recursion}
For $f,g\in\Lambda[z^{-1}]$,
\begin{equation}\label{eq:beta-recursion}
\beta(f,g)=\beta(z\,\pi_- f,z^{-1}g)
+(z\,\pi_- f)\mathcal Q(z^{-1}g)-\mathcal Q(\pi_-f)g.
\end{equation}
<!-- end lemma-10 -->
<!-- proof-21: body_md -->
Equation~\eqref{eq:beta-coboundary} identifies $\beta$ as the derivation
defect of $\mathbf Q-\operatorname{Id}$.  Expanding it gives the Hochschild cocycle identity
\[
\beta(fg,h)-\beta(f,gh)=f\beta(g,h)-\beta(f,g)h.
\]
Also, $\mathbf Q(z^{-1}g)=z^{-1}\mathbf Qg+\mathcal Q(z^{-1}g)$, so
\[
\beta(z^{-1},g)=\mathcal Q(z^{-1}g)-Q_1g.
\]
Apply the cocycle identity to $(z\,\pi_- f,z^{-1},g)$, using
$z^{-1}(z\,\pi_-f)=\pi_-f$ and $\beta(\pi_0f,g)=0$.  The two terms containing
$Q_1$ cancel, leaving \eqref{eq:beta-recursion}.

<!-- lemma-11: body_md -->
\label{lem:universal-traces}
For $f,g\in\Lambda[z^{-1}]$,
\begin{equation}\label{eq:universal-traces}
\mathcal Q\bigl(z^{-1}\beta(f,g)\bigr)=0,
\qquad
\mathcal Q\bigl(\pi_-\beta(f,g)\bigr)=-\mathcal Q(\pi_-f)\mathcal Q(\pi_-g).
\end{equation}
<!-- end lemma-11 -->
<!-- proof-22: body_md -->
By $\Lambda$-bilinearity and the vanishing of $\beta$ on $\Lambda$, it suffices to

\end{aligned}

<!-- proposition-8: body_md -->
\label{prop:ytrace}
For all partitions $\mu,\nu$ and addable corners $s\in\mathcal A_\mu$, $t\in\mathcal A_\nu$,
\begin{equation}\label{eq:y-beta}
\mathbf{y}_u(\beta_{\mu,\nu}^{s,t})=\mathbf{T}_{\mu\star\nu}(u)-1.
\end{equation}
<!-- end proposition-8 -->
<!-- proof-25: body_md -->
We follow the Jack induction of \cite[Theorem~4.11]{Mickler:2023}, at fixed
total size $N=|\mu|+|\nu|$, inducting on the size of the smaller partition.
If either partition is empty, one argument of $\beta$ is $1$, so both sides
of \eqref{eq:y-beta} vanish.  By symmetry, assume
$0<|\mu|\le|\nu|$ and fix $t\in\mathcal A_\nu$.  For every $r\in\mathcal R_\mu$, the pair
$(\mu-r,\nu+t)$ has total size $N$ and smaller size $|\mu|-1$.
The induction hypothesis applies to this pair, and
Lemma~\ref{lem:y-product-defect} also gives its product formula
\eqref{eq:yutarget}.

By \eqref{eq:hat-pidown}, write
\[
z\,\pi_-\widehat\Psi_\mu^s
=\sum_{r\in\mathcal R_\mu}c_r^s\widehat\Psi_{\mu-r}^{\,r},
\qquad \sum_{r\in\mathcal R_\mu}c_r^s=1.
\]
Likewise the coefficients of $z^{-1}\widehat\Psi_\nu^t$ in the hatted basis
of $Z_{\nu+t}$ sum to $1$, by \eqref{eq:hat-zcreation}.
Lemmas~\ref{lem:beta-recursion}
and~\ref{lem:hat-projections} give
\begin{equation}\label{eq:beta-induction}
\beta_{\mu,\nu}^{s,t}
=\sum_{r\in\mathcal R_\mu}c_r^s\left(
\beta(\widehat\Psi_{\mu-r}^{\,r},z^{-1}\widehat\Psi_\nu^t)
+\widehat\Psi_{\mu-r}^{\,r}\widehat J_{\nu+t}\right)
-\widehat J_\mu\widehat\Psi_\nu^t.
\end{equation}
Expanding $z^{-1}\widehat\Psi_\nu^t$ by the creation formula and applying the
induction hypothesis gives
\[
\mathbf{y}_u\bigl(\beta(\widehat\Psi_{\mu-r}^{\,r},z^{-1}\widehat\Psi_\nu^t)\bigr)
=\mathbf{T}_{(\mu-r)\star(\nu+t)}(u)-1.
\]
For the other term in the sum, expand $\widehat J_{\nu+t}$ using
\eqref{eq:hat-kerov-moments}.  The product formula for the smaller pair and
\eqref{eq:hat-kerov-kernel} give
\[
\begin{aligned}
\mathbf{y}_u(\widehat\Psi_{\mu-r}^{\,r}\widehat J_{\nu+t})
&=\mathbf{T}_{(\mu-r)\star(\nu+t)}(u)
\sum_{d\in\mathcal A_{\nu+t}}\frac{\widehat\tau_{\nu+t}^d}{1-[r][d]/u}\\
&=\mathbf{T}_{\mu\star(\nu+t)}(u)-\mathbf{T}_{(\mu-r)\star(\nu+t)}(u).
\end{aligned}
\]
Thus the two terms combine independently of $r$.  Put
$H:=\mathbf{y}_u(\widehat J_\mu\widehat\Psi_\nu^t)$.  Taking the trace of
\eqref{eq:beta-induction} and using $\sum_r c_r^s=1$ yields
\begin{equation}\label{eq:beta-trace-step}
\mathbf{y}_u(\beta_{\mu,\nu}^{s,t})
=\mathbf{T}_{\mu\star(\nu+t)}(u)-1-H.
\end{equation}
The right-hand side is independent of $s$.  Using this independence,
\eqref{eq:hat-kerov-moments}, Lemma~\ref{lem:y-product-defect}, and
\eqref{eq:hat-kerov-kernel}, we obtain
\[
\begin{aligned}
H&=\sum_{a\in\mathcal A_\mu}\widehat\tau_\mu^a\,
\mathbf{y}_u(\widehat\Psi_\mu^a\widehat\Psi_\nu^t)\\
&=\bigl(1+\mathbf{y}_u(\beta_{\mu,\nu}^{s,t})\bigr)
\sum_{a\in\mathcal A_\mu}\frac{\widehat\tau_\mu^a}{1-[a][t]/u}\\
&=\bigl(1+\mathbf{y}_u(\beta_{\mu,\nu}^{s,t})\bigr)
\bigl(\mathbf{T}_{\mu\star t}(u)-1\bigr).
\end{aligned}
\]
Substituting $H$ into \eqref{eq:beta-trace-step} and using \eqref{eq:boxmult}
gives
\[
\bigl(1+\mathbf{y}_u(\beta_{\mu,\nu}^{s,t})\bigr)\mathbf{T}_{\mu\star t}(u)
=\mathbf{T}_{\mu\star(\nu+t)}(u)
=\mathbf{T}_{\mu\star\nu}(u)\mathbf{T}_{\mu\star t}(u).
\]
Cancelling the nonzero rational factor $\mathbf{T}_{\mu\star t}(u)$ proves
\eqref{eq:y-beta} and completes the induction.

<!-- lemma-15: body_md -->
\label{lem:TasExp}
As a formal power series at $u=0$,
\begin{equation}\label{eq:TasExp}
\mathbf{T}_{\mu\star\nu}(u)
=\Pi\bigl(\mathsf t u\,\overline{\mathfrak s}_\mu,\,\overline{\mathfrak s}_\nu\bigr),
\qquad
\log\mathbf{T}_{\mu\star\nu}(u)
=-\sum_{k\ge1}\frac{u^k}{k}(1-\mathsf q^k)(1-\mathsf t^{-k})
\left(\sum_{x\in\mu}[x]^{-k}\right)
\left(\sum_{y\in\nu}[y]^{-k}\right).
\end{equation}
<!-- end lemma-15 -->
<!-- proof-32: body_md -->
From $[1,1]=\mathsf t/\mathsf q$, $[0,0]=1$, $[0,1]=\mathsf q^{-1}$, and $[1,0]=\mathsf t$,
\eqref{eq:Nfactor} is
\begin{equation}\label{eq:Nregular}
\mathcal N(u)=\frac{(1-u\mathsf q/\mathsf t)(1-u)}{(1-u\mathsf q)(1-u/\mathsf t)},
\end{equation}
so
\begin{align}
\log \mathcal N(u)
&=-\sum_{k\ge1}\frac{u^k}{k}
 \left(\mathsf q^k\mathsf t^{-k}+1-\mathsf q^k-\mathsf t^{-k}\right)\notag\\
&=-\sum_{k\ge1}\frac{u^k}{k}(1-\mathsf q^k)(1-\mathsf t^{-k}).
\label{eq:logN}
\end{align}
Since $[x+y]=[x][y]$, summing \eqref{eq:logN} at $u/[x+y]$ over $x\in\mu$,
$y\in\nu$ gives the second identity.  By the definition of plethystic substitution,
$p_k[\mathsf t u\,\overline{\mathfrak s}_\mu]=-u^k(1-\mathsf q^k)\sum_{x\in\mu}[x]^{-k}$
and $p_k[\overline{\mathfrak s}_\nu]=-\mathsf t^{-k}(1-\mathsf q^k)\sum_{y\in\nu}[y]^{-k}$,
so the $k$th summand in the exponent of \eqref{eq:macrkernel} is
\[
\frac{1-\mathsf t^k}{1-\mathsf q^k}\,
p_k[\mathsf t u\,\overline{\mathfrak s}_\mu]\,p_k[\overline{\mathfrak s}_\nu]
=-u^k(1-\mathsf q^k)(1-\mathsf t^{-k})
 \left(\sum_{x\in\mu}[x]^{-k}\right)
 \left(\sum_{y\in\nu}[y]^{-k}\right).
\]
Dividing by $k$ and summing over $k\ge1$ gives $\log\mathbf{T}_{\mu\star\nu}(u)$,
which proves the first identity.

<!-- corollary-6: body_md -->
\label{cor:Warnaar-rewritten}
Let $\mu,\nu$ be partitions and let $n\ge|\mu|+|\nu|$ be an integer.  Then
\begin{equation}\label{eq:Warnaar-P-finite-rank}
\kappa_{u,\mathsf t^n}(J_\mu J_\nu)
=\kappa_{u,\mathsf t^n}(J_\mu)\kappa_{u,\mathsf t^n}(J_\nu)
\mathbf{T}_{\mu\star\nu}(u\mathsf t^n).
\end{equation}
<!-- end corollary-6 -->
<!-- proof-33: body_md -->
Put $N=|\mu|+|\nu|$ and write
$P_\mu P_\nu=\sum_{\gamma\vdash N}f^\gamma_{\mu\nu}P_\gamma$ in $\Lambda$.  Apply
Proposition~\ref{prop:Warnaar-finite-rank} with $(\lambda,\mu,\nu)$ renamed
$(\mu,\nu,\gamma)$ and with $z=u\mathsf t^n$.  Since $n\ge N$, the monomial
symmetric functions of degree $N$ remain linearly independent in $n$
variables, so the $n$-variable structure constants are the
$f^\gamma_{\mu\nu}$ of $\Lambda$, and the sum runs over all $\gamma\vdash N$.
Here $P_\gamma(\mathsf t^\delta)=P_\gamma(1,\mathsf t,\ldots,\mathsf t^{n-1})=\epsilon_{\mathsf t^n}(P_\gamma)$,
and, by the definition of $(a;\mathsf q,\mathsf t)_\gamma$ and
Theorem~\ref{thm:principal-spec},
\[
\frac{(z\mathsf t^{-n};\mathsf q,\mathsf t)_\gamma}{(z;\mathsf q,\mathsf t)_\gamma}
=\prod_{i=1}^{n}\prod_{\rho=0}^{\gamma_i-1}
\frac{1-u\mathsf t^{1-i}\mathsf q^\rho}{1-u\mathsf t^{n+1-i}\mathsf q^\rho}
=\prod_{b\in\gamma}\frac{1-u/[b]}{1-u\mathsf t^n/[b]}
=\frac{\kappa_{u,\mathsf t^n}(J_\gamma)}{\epsilon_{\mathsf t^n}(J_\gamma)}.
\]
Since $J_\lambda=\prod_{b\in\lambda}h_\lambda(b)\,P_\lambda$,
\[
g_{\mu\nu}^{\gamma}=f_{\mu\nu}^{\gamma}\,
\frac{\prod_{b\in\mu}h_\mu(b)\prod_{b\in\nu}h_\nu(b)}{\prod_{b\in\gamma}h_\gamma(b)},
\]
and multiplying Warnaar's identity by
$\prod_{b\in\mu}h_\mu(b)\prod_{b\in\nu}h_\nu(b)$ gives
\begin{equation}\label{eq:Warnaar-specialized}
\kappa_{u,\mathsf t^n}(J_\mu J_\nu)
=\epsilon_{\mathsf t^n}(J_\mu)\epsilon_{\mathsf t^n}(J_\nu)
\prod_{i,j=1}^{n}
\frac{(u\mathsf t^{n+1-i-j};\mathsf q)_{\mu_i+\nu_j}}
     {(u\mathsf t^{n+2-i-j};\mathsf q)_{\mu_i+\nu_j}}.
\end{equation}
For fixed $i,j$ put $c=u\mathsf t^{n+1-i-j}$.  By \eqref{eq:Nregular},
\[
\mathcal N(\mathsf t c\mathsf q^{\rho+\sigma})
=\frac{(1-c\mathsf q^{\rho+\sigma+1})(1-\mathsf t c\mathsf q^{\rho+\sigma})}
      {(1-\mathsf t c\mathsf q^{\rho+\sigma+1})(1-c\mathsf q^{\rho+\sigma})},
\]
whose product over $0\le\rho<\mu_i$ and $0\le\sigma<\nu_j$ telescopes to
\[
\frac{(c\mathsf q^{\mu_i};\mathsf q)_{\nu_j}(\mathsf t c;\mathsf q)_{\nu_j}}
{(\mathsf t c\mathsf q^{\mu_i};\mathsf q)_{\nu_j}(c;\mathsf q)_{\nu_j}}.
\]
With
$(c;\mathsf q)_{\mu_i+\nu_j}=(c;\mathsf q)_{\mu_i}(c\mathsf q^{\mu_i};\mathsf q)_{\nu_j}$ this gives
\begin{equation}\label{eq:Warnaar-row-split}
\frac{(c;\mathsf q)_{\mu_i+\nu_j}}{(\mathsf t c;\mathsf q)_{\mu_i+\nu_j}}
=\frac{(c;\mathsf q)_{\mu_i}}{(\mathsf t c;\mathsf q)_{\mu_i}}
 \frac{(c;\mathsf q)_{\nu_j}}{(\mathsf t c;\mathsf q)_{\nu_j}}
 \prod_{\rho=0}^{\mu_i-1}\prod_{\sigma=0}^{\nu_j-1}
 \mathcal N(\mathsf t c\mathsf q^{\rho+\sigma}).
\end{equation}
Multiplying the first factor on the right of \eqref{eq:Warnaar-row-split}
over $j=1,\ldots,n$ telescopes, for fixed $i$, to
$(u\mathsf t^{1-i};\mathsf q)_{\mu_i}/(u\mathsf t^{n+1-i};\mathsf q)_{\mu_i}$, and the product over
$i$ is
\[
\prod_{b\in\mu}\frac{1-u/[b]}{1-u\mathsf t^n/[b]}
=\frac{\kappa_{u,\mathsf t^n}(J_\mu)}{\epsilon_{\mathsf t^n}(J_\mu)};
\]
the second factor likewise gives
$\kappa_{u,\mathsf t^n}(J_\nu)/\epsilon_{\mathsf t^n}(J_\nu)$.  Finally,
for boxes $x=(i-1,\rho)\in\mu$ and $y=(j-1,\sigma)\in\nu$,
$\mathsf t c\mathsf q^{\rho+\sigma}=u\mathsf t^{n+2-i-j}\mathsf q^{\rho+\sigma}=u\mathsf t^n/[x+y]$, so
the product of the remaining factors is $\mathbf{T}_{\mu\star\nu}(u\mathsf t^n)$.
Substituting into \eqref{eq:Warnaar-specialized} gives
\eqref{eq:Warnaar-P-finite-rank}.

<!-- corollary-7: body_md -->
\label{cor:kappa-double-residue}
Let $x\in\mu\cap\nu$ satisfy $x\ne(0,0)$, and let $y$ be a box such that
$x+y\notin\mu\cup\nu$.  Then

\sum_{\substack{\lambda:\ x+y\in\lambda\setminus(\mu\cup\nu)}}
 g_{\mu\nu}^{\lambda}
 \mathsf t^{n(\lambda)-n(\mu)-n(\nu)}
\prod_{b\in\lambda\setminus(\mu\cup\nu)\setminus\{x+y\}}
\frac{(1-[x-b])(1-[y-b])}{1-[x+y-b]}

\end{equation}

Apply \eqref{eq:kappa-diagonal-residue} with its box $y$ replaced by $x+y$.

=\frac{(1-[x-b])(1-[y-b])}{1-[x+y-b]}.
\]
This proves \eqref{eq:kappa-double-residue}.  Multiplying by
$\Phi_{\mu\cap\nu}$ before evaluation avoids assuming a simple pole: when $y\in\mu\cap\nu$, that
pole can be double.

<!-- corollary-8: body_md -->
\label{cor:kappa-root-residue}
Assume $\mu,\nu\ne\varnothing$, and let $y\notin\mu\cup\nu$ be a box.  Then
\begin{equation}\label{eq:kappa-root-residue}
\sum_{\substack{\lambda:\ y\in\lambda\setminus(\mu\cup\nu)}}
\widehat g_{\mu\nu}^{\lambda}
=
\frac{1}{[y]}
\operatorname*{Res}_{w=[y]}\mathbf{T}_{\mu\star\nu}(w).
\end{equation}
<!-- end corollary-8 -->
<!-- proof-37: body_md -->
Take the residue of \eqref{eq:kappa-diagonal-residue} at $u=1$, corresponding
to the box $(0,0)\in\mu\cap\nu$.  The residue of the right-hand side is
\[
-\frac{\operatorname*{Res}_{w=[y]}\mathbf{T}_{\mu\star\nu}(w)}{[y]-1}.
\]
On the left, the root-box factor contributes the common residue $-1$.  For
every supported $\lambda$, the definition of $\varpi$ gives
\[
\frac{\varpi_\lambda}{\varpi_\mu\varpi_\nu}
=
\mathsf t^{n(\lambda)-n(\mu)-n(\nu)}
\frac{\displaystyle
 \prod_{b\in\lambda\setminus(\mu\cup\nu)}(1-1/[b])}
{\displaystyle
 \prod_{a\in(\mu\cap\nu)\setminus\{(0,0)\}}(1-1/[a])}.
\]
The diagonal residue has already removed the factor indexed by $y$.  Hence its
$\lambda$-summand is
\[
g_{\mu\nu}^{\lambda}
 \mathsf t^{n(\lambda)-n(\mu)-n(\nu)}
\frac{\displaystyle
 \prod_{b\in\lambda\setminus(\mu\cup\nu)\setminus\{y\}}(1-1/[b])}
{\displaystyle
 \prod_{a\in(\mu\cap\nu)\setminus\{(0,0)\}}(1-1/[a])}
=
\frac{\widehat g_{\mu\nu}^{\lambda}}{1-1/[y]}.
\]
Therefore the residue of the left-hand side is
\[
-\frac{1}{1-1/[y]}
\sum_{\substack{\lambda:\ y\in\lambda\setminus(\mu\cup\nu)}}
\widehat g_{\mu\nu}^{\lambda}.
\]
Equating the two residues and using
$1-1/[y]=([y]-1)/[y]$ proves \eqref{eq:kappa-root-residue}.
