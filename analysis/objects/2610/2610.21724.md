<!-- lemma-1: body_md -->
\label{lem:rank-one}
Let $A$ be the adjacency matrix of a graph in $\mathscr G_3(\theta_0,\theta_1,\theta_2)$.  
There is a positive $\theta_0$-eigenvector $\bm\alpha$ such that
\begin{equation}\label{eqn:3ev}
    A^2-(\theta_1+\theta_2)A+\theta_1\theta_2I
       =\bm\alpha\bm\alpha^ 
  {\mathpalette\@transpose{}} 
.
\end{equation}
For each vertex $\mathsf x$ of degree $d_{\mathsf x}$, we have $\alpha_{\mathsf x}^2=d_{\mathsf x}+\theta_1\theta_2$.
Moreover, $\theta_1\geqslant0$ and  $\theta_2\leqslant-\sqrt2$.

<!-- theorem-1: body_md -->
\label{thm:equitableValency}
    Let $\Gamma \in \mathscr{G}_3(\theta_0,\theta_1,\theta_2)$ have at most three distinct degrees. Then the degree partition $\mathfrak{D}(\Gamma)$ is equitable, and $\theta_0$ is an eigenvalue of the quotient matrix $Q_{\mathfrak{D}(\Gamma)}(\Gamma)$.

<!-- theorem-2: body_md -->
\label{thm:bireg}
    Let $\Gamma \in \mathscr{G}_3(\theta_0,\theta_1,\theta_2)$ be a non-bipartite biregular graph with degrees $k_1$ and $k_2$, and let $Q=Q_{\mathfrak D(\Gamma)}(\Gamma)$.  
    Then
	\begin{enumerate}
		\item[\textup{(i)}] All eigenvalues of $\Gamma$ are integers.
        \item[\textup{(ii)}] The quotient matrix $Q$ has eigenvalues $\theta_0$ and $\theta$, where $\theta \in \{\theta_1,\theta_2\}$.
		\item[\textup{(iii)}] Letting $\{\theta, \theta'\} = \{\theta_1, \theta_2\}$, we have $\alpha_1\alpha_2 = -\theta(\theta' +1)$.
        \item[\textup{(iv)}] If either part of the degree partition induces an empty subgraph, then $\theta = \theta_2$.
  \end{enumerate}

<!-- theorem-3: body_md -->
\label{thm:2-des}
Let $\Gamma\in\mathscr G_3(\theta_0,\theta_1,\theta_2)$ be connected and biregular, with degree partition $\{V_1,V_2\}$, adjacency matrix $\left[\begin{smallmatrix}A_1&B\\B^ 
  {\mathpalette\@transpose{}} 
&A_2\end{smallmatrix}\right]$, and degree-partition quotient matrix $\left[\begin{smallmatrix}k_{11}&k_{12}\\k_{21}&k_{22}\end{smallmatrix}\right]$.
Suppose that $n_1=mn$ for integers $m,n\geqslant2$, and that $\Gamma[V_1]=\overline{mK_n}$ or $\Gamma[V_1]=mK_n$.
Then $B$ is the incidence matrix of a GDD $\mathcal D$ with parameters $(n_1,n_2;k_{12},k_{21};\lambda_1,\lambda_2;m,n)$, where
\begin{equation}
\label{eqn:l1l2}
    (\lambda_1,\lambda_2)=
\begin{cases}
\bigl(k_{12}+\theta_1\theta_2,
      k_{12}+(\theta_1+1)(\theta_2+1)-1+n\bigr),
   &\text{if $\Gamma[V_1]=\overline{mK_n}$},\\[3pt]
\bigl(k_{12}+(\theta_1+1)(\theta_2+1),
      k_{11}+k_{12}+\theta_1\theta_2\bigr),
   &\text{if $\Gamma[V_1]=mK_n$}.
\end{cases}
\end{equation}
If $\mathcal D$ is semiregular, then both of the following conclusions hold:
\[
n=
\begin{cases}
-\theta_2,&\text{if $\Gamma[V_1]=\overline{mK_n}$},\\
\theta_1+1,&\text{if $\Gamma[V_1]=mK_n$},
\end{cases}
\]
and
\[
\operatorname{Spec}(A_2)\subseteq
\begin{cases}
\{k_{22},\theta_1,-n,\theta_1-n\},
    &\text{if $\Gamma[V_1]=\overline{mK_n}$},\\
\{k_{22},n-1,\theta_2,n+\theta_2\},
    &\text{if $\Gamma[V_1]=mK_n$}.
\end{cases}
\]

<!-- proof-1: body_md -->
By Theorem~\ref{thm:equitableValency} the degree-partition is equitable.
Hence, $B\mathbf1_{n_2}=k_{12}\mathbf1_{n_1}$ and $B^ 
  {\mathpalette\@transpose{}} 
\mathbf1_{n_1}=k_{21}\mathbf1_{n_2}$.
The $(1,1)$-block of \eqref{eqn:3ev} is
\begin{equation}\label{eq:BBstar}
BB^ 
  {\mathpalette\@transpose{}} 

   =-A_1^2+(\theta_1+\theta_2) A_1-\theta_1\theta_2I+\alpha_1^2J,
\end{equation}

We first verify the products involving $\mathsf A_1$.  
These are instances of the matched parallel-class calculations in \cite[Proposition~2.2]{Muzychuk2007}:
\begin{align}
    \mathsf A_1^2&=q^3I+(q-1)\mathsf A_{\neq}+q^2\mathsf A_3,
        \label{eqn:A1sq}\\
    \mathsf A_1\mathsf A_3&=(q^2-1)\mathsf A_1+(q-1)\mathsf A_2.
        \label{eqn:A1A3}
\end{align}
Every vertex has a line of $q^2$ neighbours in each of the $q$ other copies, so its degree is $q^3$.  
Two distinct vertices in the same copy have common neighbours in another copy precisely when they lie on the same line of the
class assigned to that copy.  
Each selected class is used once, so their number of common neighbours is $q^2$ when their joining line is in a
selected class and zero otherwise.  
For vertices in distinct copies, each of the $q-1$ remaining copies contributes the intersection of two lines from distinct parallel classes, hence one common neighbour.  
This proves \eqref{eqn:A1sq}.

For \eqref{eqn:A1A3}, take vertices $x\in\Omega_a$ and $y\in\Omega_b$, where $a\ne b$.  
The neighbours of $x$ in $\Omega_b$ form a line $L$ in a selected class.  
If $y\in L$, then the $q^2-1$ other points of $L$ are adjacent to $y$ in $L_{q,b}$, and there are no further contributions.  
If $y\notin L$, each of the other $q-1$ selected classes supplies exactly one point of $L$ adjacent to $y$.  
These points are distinct.  
The product is zero when the two vertices lie in the same copy, proving the identity.

Since $\mathsf A_2 = \mathsf A_{\neq} - \mathsf A_1$ and $\mathsf A_4 = J - I - \mathsf A_{\neq} - \mathsf A_3$, and since $\{I, \mathsf A_{\neq}, \mathsf A_3, \mathsf A_4\}$ spans the Bose--Mesner algebra of $\mathcal{A}_q^\prime$, \eqref{eqn:A1sq} and \eqref{eqn:A1A3}, together with

    q^3 + q(-q^2) + \operatorname{rank}(\mathsf E_2)\cdot 0 + m_3 q^2 - m_4 q^2 = 0 ,
\]
so $m_3 = m_4$; and $m_3 + m_4 = \operatorname{rank}(\mathsf E_3^\prime) = q(q+1)(q^2-1)$ implies that $m_3 = m_4 = \frac{q(q+1)(q^2-1)}{2}$.

The dimensions are positive, so both eigenspaces occur.  
Their projections are the remaining primitive idempotents $\mathsf E_3,\mathsf E_4$, since the algebra has dimension five.
The remaining entries follow by linearity: since $\mathsf A_2 = \mathsf A_{\neq} - \mathsf A_1$, its eigenvalues are $\alpha - \theta$; and since $\sum_{j=0}^4 \mathsf A_j = J$, the eigenvalues of $\mathsf A_4$ on the non-principal idempotents are $-1 - \theta - (\alpha - \theta) - \beta = -1 - \alpha - \beta$.

<!-- corollary-2: body_md -->
\label{cor:auxiliary-coherent-rank}
Let $q$ be a prime power.
Then
\[
    \mathcal W(G_q)
      =\operatorname{span}_{\mathbb{R}}\{I,\mathsf A_1,\mathsf A_2,
                                    \mathsf A_3,\mathsf A_4\}.
\]
<!-- end corollary-2 -->
<!-- proof-13: body_md -->
By Theorem~\ref{thm:main-scheme-and-eigenmatrix} the coherent closure $\mathcal W(G_q)$ is contained in $\operatorname{span}_{\mathbb{R}}\{I,\mathsf A_1,\mathsf A_2,                                    \mathsf A_3,\mathsf A_4\}$.  
    To prove the reverse inclusion, put $H=\mathsf A_1^2-q^3I$.
By \eqref{eqn:A1sq}, $H=(q-1)\mathsf A_{\neq}+q^2\mathsf A_3$.
Since $\mathsf A_{\neq}$ and $\mathsf A_3$ have disjoint binary supports,
\[
    \mathsf A_3=
       \frac{H\circ H-(q-1)H}{q^2(q^2-q+1)},
    \qquad
    \mathsf A_{\neq}=\frac{H-q^2\mathsf A_3}{q-1}.
\]
Both matrices therefore belong to $\mathcal W(G_q)$.
The identities
\[
    \mathsf A_2=\mathsf A_{\neq}-\mathsf A_1,
    \qquad
    \mathsf A_4=J-I-\mathsf A_{\neq}-\mathsf A_3
\]
show that the remaining basis matrices also belong to the coherent closure.  
This proves equality.

<!-- lemma-9: body_md -->
\label{lem:degree-block-rank}
Let $\Gamma$ be a biregular graph with degree partition $\{V_1,V_2\}$.
Let $G_i=\Gamma[V_i]$ for $i=1,2$.
Suppose $A(\Gamma) = \left [\begin{smallmatrix}A(G_1)&B\\B^ 
  {\mathpalette\@transpose{}} 
&A(G_2)\end{smallmatrix}\right ]$, where $B\notin\{O,J\}$.
Then
\[
    \operatorname{cr}(\Gamma)\geqslant
    \operatorname{cr}(\Gamma[V_1])+\operatorname{cr}(\Gamma[V_2])+4.
\]
<!-- end lemma-9 -->
<!-- proof-14: body_md -->
Write $A=A(\Gamma)$ and let $d_1\ne d_2$ be the two degrees.  
The diagonal matrix $D=A^2\circ I$ belongs to $\mathcal W(\Gamma)$, so the diagonal indicator matrices
\[
    F_1=\frac{D-d_2I}{d_1-d_2},\qquad F_2=I-F_1
\]
of $V_1,V_2$ belong to it.  
Restricted to $V_i\times V_i$, the corner $F_i\mathcal W(\Gamma)F_i$ is a coherent algebra containing $A(\Gamma[V_i])$.
Its dimension is therefore at least $\operatorname{cr}(\Gamma[V_i])$.  
Each off-diagonal corner contains the two linearly independent matrices $B,J_{|V_1|,|V_2|}$ or their transposes.
The four corners have disjoint supports, so their dimensions add, giving the stated bound.

<!-- proposition-4: body_md -->
\label{prop:family-coherent-type}
Let $q$ be a prime power.
Then the coherent closure $\mathcal W(\Gamma_q)$ has type $\left[\begin{smallmatrix}3&2\\2&5\end{smallmatrix}\right]$.
<!-- end proposition-4 -->
<!-- proof-15: body_md -->
Let $m=q^3+1$, $v=mq^2$, $b=q^4(q+1)$.  
The adjacency matrix of $\Gamma_q$ is
\[
    A=\begin{bmatrix}J_v-I_m\otimes J_{q^2}&M_q\\M_q^ 
  {\mathpalette\@transpose{}} 
&\mathsf A_1\end{bmatrix}.
\]
By Corollary~\ref{cor:main-three}, its point and block sets are its two degree classes.  
We exhibit a coherent algebra of dimension $12$ containing $A$, then apply Lemma~\ref{lem:degree-block-rank} to prove that it is the coherent closure.

Let
\[
\begin{gathered}
    \mathcal L=\operatorname{span}_{\mathbb{R}}\{I_v,I_m\otimes J_{q^2},J_v\},\qquad
    \mathcal U=\operatorname{span}_{\mathbb{R}}\{M_q,J_{v,b}\},\\
    \mathcal R=\operatorname{span}_{\mathbb{R}}
       \{I_b,\mathsf A_1,\mathsf A_2,\mathsf A_3,\mathsf A_4\}.
\end{gathered}
\]
Here $\mathcal L$ is the rank-three Bose--Mesner algebra of the complete multipartite graph on the points, and $\mathcal R$ is the rank-five coherent closure of the auxiliary graph by Corollary~\ref{cor:auxiliary-coherent-rank}.

Since $G_q$ is an auxiliary graph of $\mathcal D_q$ by Lemma~\ref{lem:kernel-completion}, we can apply \eqref{eqn:aux-block} to obtain
\[
    M_q^ 
  {\mathpalette\@transpose{}} 
 M_q=q^4I+(q^3+q)J-\mathsf A_1^2.
\]
Using \eqref{eqn:A1sq} and $J=I+\mathsf A_{\neq}+\mathsf A_3+\mathsf A_4$, we obtain
\begin{equation}\label{eqn:MtM}
    M_q^ 
  {\mathpalette\@transpose{}} 
 M_q=(q^4+q)I+(q^3+1)\mathsf A_{\neq}
       +(q^3-q^2+q)\mathsf A_3+(q^3+q)\mathsf A_4.
\end{equation}

To check multiplication across the two degree classes, we use
\begin{align*}
    M_q\mathsf A_1&=q^2J,&
    M_q\mathsf A_2&=(q^4-q^2)J,\\
    M_q\mathsf A_3&=-qM_q+q^2J,&
    M_q\mathsf A_4&=(q-1)M_q+(q^3-q^2)J.
\end{align*}
The first identity follows from \eqref{eqn:aux-mixed}. 
Every row of $M_q$ has $q^3$ ones in each copy, so $M_q\mathsf A_{\neq}=q^4J$, giving the second.  
For the third, each of the $q$ used-class lines through a fixed vertex meets the row's $q$ selected free-class lines in $q$
points.  
Removing the vertex itself subtracts $q$ precisely when its incidence entry is one, giving $q^2-qM_q$.  
Since $I+\mathsf A_3+\mathsf A_4$ is the block-diagonal matrix with one all-ones block for each copy, and every row of $M_q$ has $q^3$ ones in each copy, we have
\[
    M_q(I+\mathsf A_3+\mathsf A_4)=q^3J.
\]
Using the third identity, we obtain
\[
    M_q\mathsf A_4
      =q^3J-M_q-M_q\mathsf A_3
      =(q-1)M_q+(q^3-q^2)J.
\]
These four identities show that $M_qR\in\mathcal U$ for every $R\in\mathcal R$.  
Since each such $R$ has constant column sum, we also have $JR\in\operatorname{span}_{\mathbb R}\{J\}$.
As $\mathcal U=\operatorname{span}_{\mathbb R}\{M_q,J\}$, it follows that $\mathcal U\mathcal R\subseteq\mathcal U$.

By Remark~\ref{rem:semireg}, we have $(I_m\otimes J_{q^2})M_q=qJ$.
Together with the constant row and column sums of $M_q$, one obtains $\mathcal L\mathcal U\subseteq\mathcal U$.
By \eqref{eqn:inc}, we have $\mathcal U\mathcal U^ 
  {\mathpalette\@transpose{}} 
\subseteq\mathcal L$, and by \eqref{eqn:MtM}, we have $\mathcal U^ 
  {\mathpalette\@transpose{}} 
\mathcal U\subseteq\mathcal R$.
Thus
\[
    \mathcal C=
    \left\{\begin{bmatrix}L&U\\V&R\end{bmatrix}:
       L\in\mathcal L,\ U\in\mathcal U,\
       V\in\mathcal U^ 
  {\mathpalette\@transpose{}} 
,\ R\in\mathcal R\right\},
\]
with its four blocks chosen independently, is closed under ordinary multiplication and transpose.  
It is also closed under Schur multiplication: its disjoint binary basis consists of $I_v,I_m\otimes J_{q^2}-I_v,J_v-I_m\otimes J_{q^2}$ on the point side, $I_b,\mathsf A_1,\ldots,\mathsf A_4$ on the block side, and $M_q,J-M_q$ and their transposes between the two sides, each extended by zero outside its corner.  
These twelve matrices are nonzero, and $\mathcal C$ contains $I,J,A$.  
Hence $\mathcal W(\Gamma_q)\subseteq\mathcal C$ and $\dim\mathcal C=12$.

The induced point graph has coherent rank $3$, and the induced block graph has coherent rank $5$ by Corollary~\ref{cor:auxiliary-coherent-rank}.  
Also $M_q\notin\{O,J\}$ by its row sums.  
By Lemma~\ref{lem:degree-block-rank}, we have $\operatorname{cr}(\Gamma_q)\geqslant3+5+4=12$.
It follows that $\mathcal W(\Gamma_q)=\mathcal C$.
Its two diagonal basis relations are the identities on the point and block sets, giving the asserted fibres and type.
