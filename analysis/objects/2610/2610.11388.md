\label{prop:double_robust}

\begin{enumerate}
\item[\rm (i)] \textbf{Lipschitz convergence.} Suppose that, uniformly in $z, y$, $|r_n(x,z,y)-r_n(x',z,y)| \leq b_n(z,y)\,|x-x'|$ for some $b_n$. Define $a_n(Z) := \mathcal W_1(K_n(\cdot\mid Z), K(\cdot\mid Z))$. Then, for every $z, y$,
   
    $\left|\mathbb E_{K_n}[r_n(\widetilde X_n,z,y)] - \mathbb E_{K}[r_n(X,z,y)]\right| \leq a_n b_n.$
   

\item[\rm (ii)] \textbf{Uniform convergence.} Suppose that $\sup_{x,z,y}|r_n(x,z,y)| \leq b_n(z,y)$ for some $b_n$. Define $a_n(Z) := \mathrm d_{\mathrm{TV}}(K_n(\cdot\mid Z), K(\cdot\mid Z))$. Then, for every $z, y$,
$
    \left|\mathbb E_{K_n}[r_n(\widetilde X_n,z,y)] - \mathbb E_{K}[r_n(X,z,y)]\right| \leq 2a_nb_n.
$

\item[\rm (iii)] \textbf{$L_2$ convergence.} Suppose that $\mathbb E_{K}[r_n(X,Z,Y)^2 \mid \mathcal F_{n-1}] \leq b_n^2$ for some $b_n$. Define $a_n := \sqrt{\mathbb{E}\left[\mathrm d_{\chi^2}(K_n(\cdot\mid Z) \Vert K(\cdot\mid Z))^2\right]}$. Then
$
    \left|\mathbb E[r_n(\widetilde X_n',Z,Y) - r_n(X,Z,Y)]\right| \leq a_nb_n.
$
\end{enumerate}

In this $0$--$1$ loss setting we can provide an explicit expression for the e-values
that optimize the e-power across the coin-betting and model-X classes, which
turn out to coincide.

We begin by optimizing $\eta$ for $E^\eta_{\mathrm{ML}}$. Observe that
\begin{align*}
\mathbb{E}_{X \mid Y,Z}\left[\log\left(E^\eta_{\mathrm{ML}}\right)\right]
&=
\mathbb{E}_{X \mid Y,Z}\left[
-\eta \mathds{1}_{\{m(X,Z)\neq Y\}}
-
\log
\mathbb{E}_{\widetilde{X}\mid Z}\left[
e^{-\eta \mathds{1}_{\{m(\widetilde X,Z)\neq Y\}}}
\right]
\right].
\end{align*}

Differentiating with respect to $\eta$ gives
\begin{align*}
-
\mathbb{E}_{X\mid Y,Z}\!\left[\mathds{1}_{\{m(X,Z)\neq Y\}}\right]
+
\frac{
\mathbb{E}_{\widetilde{X}\mid Z}\left[
\mathds{1}_{\{m(\widetilde X,Z)\neq Y\}}
e^{-\eta \mathds{1}_{\{m(\widetilde X,Z)\neq Y\}}}
\right]
}{
\mathbb{E}_{\widetilde{X}\mid Z}\left[
e^{-\eta \mathds{1}_{\{m(\widetilde X,Z)\neq Y\}}}
\right]
}.
\end{align*}

Let
\[
p_1 = P_{X\mid Y,Z}(m(X,Z)\neq Y \mid Y,Z),
\qquad
p_0 = P_{\widetilde X\mid Z}(m(\widetilde X,Z)\neq Y \mid Y,Z).
\]

Note that
\[
\mathbb{E}_{\widetilde{X}\mid Z}\left[
\mathds{1}_{\{m(\widetilde X,Z)\neq Y\}}
e^{-\eta \mathds{1}_{\{m(\widetilde X,Z)\neq Y\}}}
\right]
=
e^{-\eta} p_0.
\]

Thus the first-order condition becomes
\begin{align*}
-p_1
+
\frac{e^{-\eta_\star} p_0}{e^{-\eta_\star} p_0 + 1 - p_0}
=0.
\end{align*}

Solving yields
\begin{align*}
\eta_\star
=
\log
\frac{p_0(1-p_1)}{p_1(1-p_0)}.
\end{align*}

With this optimal $\eta_\star$, the e-variable $E^{\eta_\star}_{\mathrm{ML}}$
is
\begin{align*}
E^{\eta_\star}_{\mathrm{ML}}
&=
\frac{e^{-\eta_\star l(m(X,Z),Y)}}
{\mathbb{E}_{\widetilde{X}\mid Z}\left[e^{-\eta_\star l(m(\widetilde X,Z),Y)}\right]}
\\
&=
\frac{\left(e^{-\eta_\star}\right)^{\mathds{1}_{\{m(X,Z)\neq Y\}}}}
{p_0 e^{-\eta_\star} + 1 - p_0}
\\
&=
\frac{\left(\frac{p_1(1-p_0)}{p_0(1-p_1)}\right)^{\mathds{1}_{\{m(X,Z)\neq Y\}}}}
{p_0\frac{p_1(1-p_0)}{p_0(1-p_1)} + 1 - p_0}
\\
&=
\left(
\frac{p_1(1-p_0)}{p_0(1-p_1)}
\right)^{\mathds{1}_{\{m(X,Z)\neq Y\}}}
\frac{1-p_1}{1-p_0}.
\end{align*}

Therefore, there are two cases:
\[
E =
\begin{cases}
\dfrac{1-p_1}{1-p_0}, & m(X,Z)=Y, \\
\dfrac{p_1}{p_0}, & m(X,Z)\neq Y.
\end{cases}
\]

We can proceed similarly for the coin-betting e-value. Optimizing
$\lambda$ yields
\[
\lambda_\star
=
\frac{p_1-p_0}{p_0(p_0-1)}.
\]

Substituting this value into the e-variable gives two cases.

If the prediction is correct, $Y=m(X,Z)$, then
\[
E_{\mathrm{CB}}
=
1+\lambda_\star p_0
=
1+\frac{p_1-p_0}{p_0-1}
=
\frac{1-p_1}{1-p_0}.
\]

If the prediction is incorrect,
\[
E_{\mathrm{CB}}
=
1+\frac{p_1-p_0}{p_0(p_0-1)}(p_0-1)
=
\frac{p_1}{p_0}.
\]

Thus the coin-betting and model-X optimal e-variables coincide.

\begin{align*}
    A_B(t)&=\int f_{T|Y,Z}(t\mid y,z)\,f_{Y,Z\mid\widetilde{T}^B}(y,z\mid\widetilde{t}^B)\,dy\,dz,\\
    D_B(t)&=\int f_{\widetilde{T}|Y,Z}(t\mid y,z)\left(\frac{1}{B+1}\sum_{j=1}^B w(\widetilde{t}_j\mid y,z)\right)f_{Y,Z\mid\widetilde{T}^B}(y,z\mid\widetilde{t}^B)\,dy\,dz.
\end{align*}
From now on we evaluate at $\widetilde{t}^B=\widetilde{T}^B$.
 
\textbf{Step 2 ($A_B$-term).}
For fixed $t$, the function $(y,z)\mapsto f_{T|Y,Z}(t\mid y,z)$ is continuous and bounded by $C$ (Assumption 1). Hence, by the definition of weak convergence and Assumption 3,
\[
    A_B(t)\to\int f_{T|Y,Z}(t\mid y,z)\,\Pi_{\varphi(Y,Z)}(dy,dz)=f_{T|\varphi(Y,Z)}(t\mid\varphi(Y,Z)),
    \qquad\text{and}\qquad \frac{A_B(t)}{B+1}\le\frac{C}{B+1}\to0 .
\]
 
\textbf{Step 3 (the likelihood-ratio average converges uniformly in $(y,z)$).}
Let
\[
    \overline{w}_B(y,z)\overset{ \mathrm{def}}{=}\frac1B\sum_{j=1}^B w(\widetilde{T}_j\mid y,z),
    \qquad
    r(y,z)\overset{\mathrm{def}}{=}\int w(t'\mid y,z)\,f_{\widetilde{T}|Y,Z}(t'\mid Y,Z)\,dt',
\]
so that $r(y,z)$ is the mean of $w(\widetilde{T}_j\mid y,z)$. By Assumption 2,
$|r(y,z)-r(y',z')|\le \ell(Y,Z)\, ||(y,z) - (y',z')||$. Moreover $r(Y,Z)=\int f_{T|Y,Z}(t'\mid Y,Z)\,dt'=1$, so $r$ is bounded by $1+\ell(Y,Z)\cdot\operatorname{diam}<\infty$. We show that, almost surely,
\begin{equation}\label{eq:ULLN}
    \sup_{y,z}\,\bigl|\overline{w}_B(y,z)-r(y,z)\bigr|\to0 .
\end{equation}
Fix $\delta>0$. By compactness, the space of $(y,z)$ is covered by finitely many balls of radius $\delta$, with centres $(y_1,z_1),\dots,(y_K,z_K)$. By the strong law of large numbers (applied to finitely many i.i.d.\ sequences with finite means), almost surely
\[
    \overline{w}_B(y_k,z_k)\to r(y_k,z_k)\ \ (k=1,\dots,K),
    \qquad
    \frac1B\sum_{j=1}^B L(\widetilde{T}_j)\to\ell(Y,Z) .
\]
For any $(y,z)$, choose $k$ with $||(y,z)-(y_k,z_k)||<\delta$. Using Assumption 2 twice,
\begin{align*}
    \bigl|\overline{w}_B(y,z)-r(y,z)\bigr|
    &\le \bigl|\overline{w}_B(y,z)-\overline{w}_B(y_k,z_k)\bigr| + \bigl|\overline{w}_B(y_k,z_k)-r(y_k,z_k)\bigr| + \bigl|r(y_k,z_k)-r(y,z)\bigr| \\
    &\le \delta\,\frac1B\sum_{j=1}^B L(\widetilde{T}_j) + \max_{k\le K}\bigl|\overline{w}_B(y_k,z_k)-r(y_k,z_k)\bigr| + \delta\,\ell(Y,Z) .
\end{align*}
The right-hand side does not depend on $(y,z)$, and its $\limsup$ is $2\delta\,\ell(Y,Z)$. Since $\delta$ was arbitrary, we get \eqref{eq:ULLN}.
 
\textbf{Step 4 ($D_B$-term).}
Since $\frac{1}{B+1}\sum_{j=1}^B w(\widetilde{T}_j\mid y,z)=\frac{B}{B+1}\overline{w}_B(y,z)$, we can split
\begin{multline*}
    \frac{B+1}{B}\,D_B(t)=\int f_{\widetilde{T}|Y,Z}(t\mid y,z)\,r(y,z)\,f_{Y,Z\mid\widetilde{T}^B}(y,z\mid\widetilde{T}^B)\,dy\,dz\\
    +\int f_{\widetilde{T}|Y,Z}(t\mid y,z)\bigl(\overline{w}_B(y,z)-r(y,z)\bigr)\,f_{Y,Z\mid\widetilde{T}^B}(y,z\mid\widetilde{T}^B)\,dy\,dz.
\end{multline*}
The second term is at most $C\sup_{y,z}|\overline{w}_B(y,z)-r(y,z)|$ in absolute value, which tends to $0$ by \eqref{eq:ULLN}. In the first term, $(y,z)\mapsto f_{\widetilde{T}|Y,Z}(t\mid y,z)\,r(y,z)$ is continuous and bounded (Assumption 1 and Step 3), so by Assumption 3 it converges to $\int f_{\widetilde{T}|Y,Z}(t\mid y,z)\,r(y,z)\,\Pi_{\varphi(Y,Z)}(dy,dz)$. For every $(y,z)$ in the fiber, $Q(\widetilde{T}|Y=y,Z=z)=Q(\widetilde{T}|Y=Y,Z=Z)$, and therefore
\[
    f_{\widetilde{T}|Y,Z}(t\mid y,z)=f_{\widetilde{T}|Y,Z}(t\mid Y,Z),
    \qquad
    r(y,z)=\int w(t'\mid y,z)\,f_{\widetilde{T}|Y,Z}(t'\mid y,z)\,dt'=\int f_{T|Y,Z}(t'\mid y,z)\,dt'=1 .
\]
Since $\Pi_{\varphi(Y,Z)}$ is supported on the fiber, $D_B(t)\to f_{\widetilde{T}|Y,Z}(t\mid Y,Z)$.
 
\textbf{Step 5 (conclusion).}
The almost-sure events used in Steps 2--4 (Assumption 3, and the countably many strong laws in Step 3) do not depend on $t$. Hence, almost surely, Steps 2 and 4 hold simultaneously for all $t$, and in particular for $t=T$. Moreover $f_{\widetilde{T}|Y,Z}(T\mid Y,Z)>0$ a.s., because $T$ has density $f_{T|Y,Z}(\cdot\mid Y,Z)$, which vanishes wherever $f_{\widetilde{T}|Y,Z}(\cdot\mid Y,Z)$ does (Assumption 2). Plugging Steps 2 and 4 into \eqref{eq:EB-rewritten},
\begin{equation*}
    \lim_{B \to \infty} E_B^{\text{exch}}(T,\widetilde{T}^B)
    = \frac{\int f_{T|Y,Z}(T\mid y,z)\, \Pi_{\varphi(Y,Z)}(dy,dz)}{ f_{\widetilde{T}|Y,Z}(T\mid Y,Z) }
    = \frac{f_{T|\varphi(Y,Z)}(T\mid\varphi(Y,Z))}{  f_{\widetilde{T}|Y,Z}(T\mid Y,Z)}.
    \qedhere
\end{equation*}

Define the cumulative inflated-controlled event
               
$G_n
:=
\bigcap_{t=1}^n A_n.$ 
             
By the definition of the approximate e-process,
\begin{align*}
\mathbb{E}_{P}
\left[
E_n\mathds{1}_{G_n}
\middle|
\mathcal{F}_{n-1}
\right]
=
\mathds{1}_{G_{n-1}}
\mathbb{E}_{P}
\left[
E_n\mathds{1}_{A_n}
\middle|
\mathcal{F}_{n-1}
\right]
\leq
(1+\epsilon_n)
\mathds{1}_{G_{n-1}}E_{n-1}.
\end{align*}

Define
\begin{align*}
M_n
:=
\frac{
E_{n}\mathds{1}_{G_n}
}{
\prod_{t=1}^n(1+\epsilon_n)
}.
\end{align*}
Then $(M_n)_{n\geq0}$ is a nonnegative supermartingale with $M_0=1$. By Ville's inequality,
\begin{align*}
P
\left(
\sup_{1\leq n\leq N}M_n
\geq \frac{1}{\alpha}
\right)
\leq \alpha.
\end{align*}

Let
\begin{align*}
C_N
:=
\prod_{t=1}^N(1+\epsilon_n).
\end{align*}
On the event $G_N$, for every $n\leq N$,
\begin{align*}
E_{n}
&\leq
C_N M_n.
\end{align*}
Consequently,
\begin{align*}
\left\{
\sup_{1\leq n\leq N}
E_{n}
\geq \frac{1}{\alpha}
\right\}\cap G_N
&\subseteq
\left\{
\sup_{1\leq n\leq N}
M_n
\geq
\frac{1}{\alpha C_N}
\right\}.
\end{align*}
Applying Ville's inequality at level $\alpha C_N$ gives
\begin{align*}
P
\left(
\sup_{1\leq n\leq N}
E_n
\geq \frac{1}{\alpha},
\ G_N
\right)
&\leq
\alpha C_N.
\end{align*}

It remains to control the probability that at least one of the events $A_n$ fails. By the probability assumption on the approximate e-process definition and union bound,
\begin{align*}
P(G_N^c)
=
P
\left(
\bigcup_{n=1}^N A_n^c
\right)
\
\leq
\sum_{n=1}^N
P(A_n^c)
\
\leq
\sum_{n=1}^N\delta_n.
\end{align*}
Therefore,
\begin{align*}
P
\left(
\sup_{1\leq n\leq N}
E_n
\geq \frac{1}{\alpha}
\right)
&\leq
\sum_{n=1}^N\delta_n
+
\alpha
\prod_{t=1}^N(1+\epsilon_n).
\end{align*}

Finally, since $\log(1+x)\leq x$ for all $x\geq0$,
\begin{align*}
\prod_{t=1}^N(1+\epsilon_n)
=
\exp\left(
\sum_{n=1}^N\log(1+\epsilon_n)
\right)\leq
\exp\left(
\sum_{n=1}^N\epsilon_n
\right).
\end{align*}
This proves the result.

We prove the three statements separately.

\@startsection{paragraph}{4}{\z@}{1.5ex plus
   0.5ex minus .2ex}{-1em}{\normalsize\bfseries}{(i) Lipschitz convergence.}

                
 
   
                    
       
                      
   

By the Kantorovich--Rubinstein dual representation of the Wasserstein distance, we have that for every Lipschitz function $f$

$$
\left|
\int f\,dQ-\int f\,dP
\right|
\leq
\operatorname{Lip}(f)\mathcal W_1(Q,P).
$$

By assumption, $x\mapsto r_n(x,z)$
is $b_n$-Lipschitz uniformly on $z, y$. Therefore, for any $z, y$, we have that


\begin{align*}
\left|
\mathbb E[r_n(\widetilde X'_n,z, y)]
-
\mathbb E[r_n(X,z, y)]
\right|
&\leq
b_n(z, y)\mathcal W_1(K_n,K)
\\
&=:
a_nb_n.
\end{align*}


                 

\@startsection{paragraph}{4}{\z@}{1.5ex plus
   0.5ex minus .2ex}{-1em}{\normalsize\bfseries}{(ii) Uniform convergence.} Fix $z$ and $y$. Since $
\sup_x|r_n(x,z, y)|\leq b_n(z, y),
$
we have
$
\|r_n(\cdot,z, y)\|_\infty\leq b_n.
$

Total variation gives
$$
\left|
\int f\,dQ-\int f\,dP
\right|
\leq
2\|f\|_\infty
\mathrm d_{\mathrm{TV}}(P,Q).
$$

Applying this to $f=r_n(,\cdot,z)$, we obtain

$$
\begin{aligned}
\left|
\mathbb E[r_n(\widetilde X'_t,z, y)]
-
\mathbb E[r_n(X,z, y)]
\right|
&\leq
2b_n(z, y)\mathrm d_{\mathrm{TV}}(K_n, K)
\\
&=
2a_nb_n(z, y).
\end{aligned}
$$


\@startsection{paragraph}{4}{\z@}{1.5ex plus
   0.5ex minus .2ex}{-1em}{\normalsize\bfseries}{(iii) $L_2$ convergence.} the chi-squared divergence $\mathrm d_{\chi^2}$ is defined as

$$
\mathrm d_{\chi^2}(P\Vert Q)
:=
\left(
\int
\left(
\frac{dP}{dQ}-1
\right)^2dQ
\right)^{1/2},
\qquad P\ll Q.
$$

Suppose that $
K_n\ll K
$ and let $(X, Z, Y)$ sampled from $P_\star(X, Z, Y)$ and $(\widetilde{X}', Z, Y)$ from $P_n(\widetilde{X}', Z, Y)$. Denote by

$$
D_n(x,z, y)
:=\frac{dP_n(\widetilde{X}', Z, Y)}{dP_\star(X, Z, Y)}(x, z, y)=
\frac{dK_n(\,\cdot\mid z)}
     {dK(\,\cdot\mid z)}(x),
$$
the Radon-Nikodym derivative.

Then $\mathbb E_{K_n}[r_n]
=
\mathbb E_{K}[D_nr_n].
$ Therefore, we have that,

$$
\mathbb E_{P_\star}[r_n]-\mathbb E_{P_n}[r_n]
=
\mathbb E_{P_\star}[(D_n-1)r_n].
$$

By the Cauchy--Schwarz inequality and the total expectation law, we obtain that


\begin{align*}
\left|
\mathbb E_{P_t}[r_n]-\mathbb E_{P_\star}[r_n]
\right|
&\leq
\left(
\mathbb E_{P_\star}\left[(D_t-1)^2\right]
\right)^{1/2}
\left(
\mathbb E_{P_\star}\left[r_n^2\right]
\right)^{1/2}
\\
&\leq
\left(
\mathbb E_{P_\star(Z)}\left[\mathbb{E}_{K}\left[(D_t-1)^2\right]\right]
\right)^{1/2}
\left(
\mathbb E_{P_\star}[r_n^2]
\right)^{1/2}
\\
&=
\sqrt{\mathbb{E}\left[\mathrm d_{\chi^2}(K_n\Vert K)^2\right]}
\,
\|r_n\|_{L_2}.
\end{align*}

By assumption, $\|r_n\|_{L_2}
\leq b_n.$ Hence, we conclude that

$$
\left|
\mathbb E_{P_t}[r_n]-\mathbb E_{P_\star}[r_n]
\right|
\leq
a_nb_n.
$$

\sim
K_n(\cdot\mid Z_n).

E_n
\mid
\mathcal F_{n-1},Z_n,Y_n
\right]=

\mathbb E_{K}
\left[

\mid
\mathcal F_{n-1},Z_n,Y_n
\right]

\mathbb E_{K_n}

\mid
\mathcal F_{n-1},Z_n,Y_n
\right].
\end{align*}

\begin{align*}
\mathbb E &
\left[
E_n
\mid
\mathcal F_{n-1},Z_n,Y_n
\right]-1
\\&=
\lambda
\Bigg[
\mathbb E_{K}
\left[
l(\widehat m_n(X,Z_n),Y_n)-l_0(Z_n,Y_n)
\mid
\mathcal F_{n-1},Z_n,Y_n

\mathcal F_{n-1},Z_n,Y_n
\right]
\Bigg].
\end{align*}

\mathbb E_{K}
\left[

\right]

\mathbb E_{K_n}

\right]

\mathbb E[E_t\mid\mathcal F_{n-1}]
\leq
1+\lambda a_nb_n.

+
\alpha

\right)

\end{align*}
