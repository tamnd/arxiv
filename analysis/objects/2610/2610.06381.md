\label{thm:upper}

hold uniformly in $e\in\mathbb S^{d-1}$.

 u(t,x)\leq C(1+z_e(t,x))\mathrm e^{-z_e(t,x)}

\end{enumerate}

\label{thm:axial}
Assume \eqref{eq:periodic-kpp}, \eqref{eq:u0} and
\eqref{eq:transverse-linearization}. Let $\lambda_*>0$ minimize
$[2(\cosh\lambda-1)+a_{\mathrm{eff}}]/\lambda$, put
$c_*=2\sinh\lambda_*$, and define
\begin{equation*}                   
 m_d(t)=c_*t-\frac{d+2}{2\lambda_*}\log t,\qquad t\geq2.
\end{equation*}
\begin{enumerate}
\item For each $\theta\in(0,1)$ there are $C_\theta,T_\theta$ such that
\begin{equation}\label{eq:axial-position}
 \left|\sup\{j\in\mathbb Z:u(t,j,0)\geq\theta\}-m_d(t)\right|
 \leq C_\theta\qquad(t\geq T_\theta).
\end{equation}
For these times, $u(t,j,0)\geq\theta$ whenever
$0\leq j\leq m_d(t)-C_\theta$.
\item For each $\theta\in(0,1)$ and $A\geq0$, all sufficiently large
$t$ admit $j_t\in\mathbb Z$ such that
\begin{equation}\label{eq:diffusive-section}
 |j_t-m_d(t)|\leq C_{\theta,A},\qquad
 \inf_{|y|\leq A\sqrt t}u(t,j_t,y)\geq\theta.
\end{equation}
\item For every $A\geq0$ and $\sigma>0$ there are positive constants
$b_{A,\sigma},B_{A,\sigma},T_{A,\sigma}$ such that, with
$\xi=j-m_d(t)$,
\begin{equation}\label{eq:axial-leading-edge}
 b_{A,\sigma}(1+\xi)\mathrm e^{-\lambda_*\xi}
 \leq u(t,j,y)\leq
 B_{A,\sigma}(1+\xi)\mathrm e^{-\lambda_*\xi}
\end{equation}
whenever $t\geq T_{A,\sigma}$, $1\leq\xi\leq\sigma\sqrt t$ and
$|y|\leq A\sqrt t$.
\end{enumerate}

\label{thm:critical-uniqueness}
Assume \eqref{eq:periodic-kpp}. Fix $n\in\mathbb S^{d-1}$, let
$\lambda>0$ minimize $H(\eta n)/\eta$, and put
$c=H(\lambda n)/\lambda$, $p=\lambda n$,
$D_n=\tfrac12n^{\mathsf T}\nabla^2H(p)n$ and
$\dot\phi_p=\left.\partial_\eta\phi_{\eta n}\right|_{\eta=\lambda}$.
Let $\Phi$ be a $\Gamma$-periodic pulsating profile in direction $n$
with speed $c$ and end states $1,0$, satisfying
\begin{equation}\label{eq:front-critical-upper}
 0<\Phi(s,x)\leq C(1+s)\mathrm e^{-\lambda s}
             \qquad(s\geq0,\ x\in\mathcal C).
\end{equation}
Then the following statements hold.
\begin{enumerate}
\item There are $A_\Phi>0$, $B_\Phi\in\mathbb R$ and $\varepsilon>0$ such
that, uniformly in the cell as $s\to+\infty$,
\begin{equation}\label{eq:critical-tail-expansion}
 \Phi(s,x)=\mathrm e^{-\lambda s}
 \left[(A_\Phi s+B_\Phi)\phi_p(x)-A_\Phi\dot\phi_p(x)

Let $L$ be a Lipschitz constant for $G$, and set $h=(v-w)_+$ in $D$,
with zero exterior values. The scalar positive-part chain rule and
the nonnegative neighbor coefficients give
$h_t\le\mathcal L_t h+Lh$ almost everywhere, including almost every

 h(t,x)\le h(t_0,x)+(q_0+L)\int_{t_0}^tE(s)\,\mathrm d s.
$$
Taking the supremum yields
$E(t)\le(q_0+L)\int_{t_0}^tE(s)\,\mathrm d s$.
Gronwall's inequality gives $E=0$.

\label{lem:spectral}
\begin{enumerate}[label=\textup{(\roman*)}]
\item The function $H$ is real analytic, even, and superlinear, with
positive definite Hessian. For $H_0(p)=2\sum_i(\cosh p_i-1)$,
\begin{equation}\label{eq:H-comparison}
 H_0(p)+\min a\leq H(p)\leq H_0(p)+\max a.
\end{equation}
\item The operators $Q_p$ and $\widehat Q_p$ are Markov generators
related by time reversal with respect to $\pi_p$. Their mean
displacements, averaged over the stationary cell distribution,
are $-\nabla H(p)$ and $\nabla H(p)$, respectively. Both have
asymptotic covariance $\Sigma_p=\nabla^2H(p)$.
\end{enumerate}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
The finite matrix $B_p$ is irreducible and has nonnegative off-diagonal
entries. Adding a sufficiently large scalar multiple of the identity and
applying Perron--Frobenius theory gives a simple principal eigenvalue and
positive eigenvectors; see \cite[Chapters~1--2]{Seneta} for the general
nonnegative-matrix result. Because the entries of $B_p$ are real analytic in
$p$ and the principal eigenvalue is simple, analytic perturbation theory gives
real-analytic $H$, $\phi_p$, and $\phi_p^*$ under the chosen normalizations;
see \cite[Chapter~II]{Kato}. Index reversal on the cell gives
$B_p^{\mathsf T}=B_{-p}$, whence $H(p)=H(-p)$. Testing the positive constant
vector in the Collatz--Wielandt inequalities \cite{Seneta} gives
\eqref{eq:H-comparison}, including superlinearity.

The eigenvector equation gives
$Q_p=\phi_p^{-1}(B_p-H(p))\phi_p$ on periodic functions.
It also gives the exit-rate terms in \eqref{eq:Q}.
For every oriented nearest-neighbor edge $(x,x+z)$,
$$
 \pi_p(x)q_p(x,z)
 =\phi_p^*(x)\mathrm e^{-p\cdot z}\phi_p(x+z)
 =\pi_p(x+z)\widehat q_p(x+z,-z).
$$
Hence
\begin{equation}\label{eq:reverse-edge}
 \pi_p(x)q_p(x,z)=\pi_p(x+z)\widehat q_p(x+z,-z).
\end{equation}
The right and left eigenvector equations also give
$$
 \sum_zq_p(x,z)=H(p)+2d-a(x)=\sum_z\widehat q_p(x,z).
$$
Together with \eqref{eq:reverse-edge}, this equality proves
invariance of $\pi_p$ and identifies $\widehat Q_p$ as the time
reversal of $Q_p$; see \cite[Chapter~3]{Norris}.

These walks can be constructed with a rate-$\Lambda$ Poisson clock,
where $\Lambda$ exceeds all exit rates. At each clock event an
independent mark chooses a jump $z$ with probability
$q_p(x,z)/\Lambda$, or leaves the site unchanged. Use
$\widehat q_p$ for the reversed walk. The rates are periodic, so a
common $\Lambda$ works on every compact set of $p$ values. The number
of jumps is bounded by the Poisson count, which proves nonexplosion.
We write $\mathbb P_x,\mathbb E_x$ for the law and expectation with
$X_0=x$, and use the completed, right-continuous clock-and-mark
filtration. Subtracting conditional jump means gives the generator
martingale. For an affine coordinate plus a bounded periodic
corrector, its path supremum on a bounded time interval is bounded
by a constant times one plus the Poisson count, and has finite
second moment. These bounds justify the bounded stopping times
used below.

For the walk generated by $Q_p$, multiplying each jump $z$ by
$\mathrm e^{k\cdot z}$ gives, on periodic functions, the matrix
$$
 \mathcal L_{p,k}
 =\phi_p^{-1}\bigl(B_{p-k}-H(p)I\bigr)\phi_p.
$$
Its principal eigenvalue is therefore $H(p-k)-H(p)$. This is the
finite-state continuous-time analogue of the transform-kernel construction
for Markov additive processes in \cite{NeyNummelin}. Exponential moments
over time $t$ are
a positive principal-eigenvector factor times
$\exp(t[H(p-k)-H(p)])$, plus an exponentially smaller term locally
in $k$. Differentiation at zero gives the mean displacement
$-\nabla H(p)$ and limiting covariance $\nabla^2H(p)$.
The reversed generator has exponent $H(p+k)-H(p)$ because $H$ is even.

To prove positive definiteness of the covariance, fix $\eta$ and write
$b(x)=\sum_zq_p(x,z)z$ and $\bar b=\sum_{\mathcal C}\pi_p(x)b(x)$.
The centered finite-cell equation
\begin{equation}\label{eq:corrector-vector}
 Q_p\chi_\eta(x)=\eta\cdot(\bar b-b(x)),\qquad
 \sum_{\mathcal C}\pi_p\chi_\eta=0
\end{equation}
has a unique solution: constants are the kernel of $Q_p$ and the
right-hand side has invariant mean zero. The process
$\eta\cdot(X_t-X_0-\bar b t)+\chi_\eta(X_t)-\chi_\eta(X_0)$
is a martingale. Under the stationary cell distribution its quadratic
variation has mean $t$ times
\begin{equation}\label{eq:variance-formula}
 \sum_{x\in\mathcal C}\pi_p(x)\sum_zq_p(x,z)
 \bigl(\eta\cdot z+\chi_\eta(x+z)-\chi_\eta(x)\bigr)^2.
\end{equation}
By Cauchy--Schwarz, adding
or subtracting its bounded corrector changes the variance by at most
$O(\sqrt t)+O(1)$. After division by $t$ this error vanishes, so
\eqref{eq:variance-formula} equals
$\eta^{\mathsf T}\nabla^2H(p)\eta$. If it were zero, every edge
increment in parentheses would vanish. Summing along a path from
$x$ to $x+L_i e_i$ gives $L_i\eta_i=0$ for every $i$.
Thus $\eta=0$, proving positive definiteness. Continuity and
compactness make the eigenvector bounds, corrector bounds,
jump-rate bounds, and covariance lower bounds uniform on compact
sets of $p$ values.

\label{lem:geometry}
\begin{enumerate}[label=\textup{(\roman*)}]
\item The set $\mathcal W$ is a compact strictly convex body with
smooth boundary and zero in its interior. The quantities in
\eqref{eq:pe} depend continuously on $e$, and $\inf_e\alpha_e>0$.
\item The minimum defining $c(n)$ is attained at a unique
$\lambda(n)>0$. The normal speed and radial speed satisfy
\begin{equation}\label{eq:normal-ray}
 c(n)=\max_{v\in\mathcal W}n\cdot v,\qquad
 w(e)=\min_{n\cdot e>0}\frac{c(n)}{n\cdot e}.
\end{equation}
The second minimum is attained uniquely at $n_e=p_e/|p_e|$, with
$\lambda(n_e)=|p_e|$.
\end{enumerate}

<!-- proof-3: body_md -->
Strict convexity and superlinearity imply that $\nabla H$ maps $\mathbb R^d$
bijectively onto $\mathbb R^d$: the minimum of $H(p)-p\cdot v$ exists and
is unique for each $v$, and its Hessian is positive definite.
The inverse function theorem gives a smooth inverse and
$\nabla^2\mathcal I(v)=(\nabla^2H(\nabla\mathcal I(v)))^{-1}$.
Since $H$ is even, $\mathcal I$ is even and
$\mathcal I(0)=-H(0)<0$. It is coercive: for any $R>0$,
$\mathcal I(v)\geq R|v|-\max_{|p|=R}H(p)$.
Its gradient vanishes only at zero. Hence $\mathcal I$ is strictly convex,
$\mathcal W=\{\mathcal I\leq0\}$ is compact, and $\mathcal I(0)<0$ places
zero in the interior of $\mathcal W$. Since $\nabla\mathcal I$ does not vanish
on $\partial\mathcal W$, that boundary is smooth. Strict convexity implies
that each positive ray meets $\partial\mathcal W$ at exactly one point. At
$v_e=w(e)e$ the outward normal is $p_e=\nabla\mathcal I(v_e)$, so
$\alpha_e=p_e\cdot e>0$. Continuity on the sphere then gives
$\inf_e\alpha_e>0$.

The derivative of $H(\lambda n)/\lambda$ has numerator
$g_n(\lambda)=\lambda n\cdot\nabla H(\lambda n)-H(\lambda n)$.
Here $g_n(0)=-H(0)<0$ and
$g_n'(\lambda)=\lambda n^{\mathsf T}\nabla^2H(\lambda n)n>0$.
If this numerator stayed nonpositive, $H(\lambda n)/\lambda$
would be nonincreasing, contradicting superlinearity. Hence it has
exactly one zero. Fenchel's inequality gives
$n\cdot v\leq H(\lambda n)/\lambda$ on $\mathcal W$.
At the minimizing $\lambda$, equality is attained by
$v=\nabla H(\lambda n)$, for which $\mathcal I(v)=0$.
Thus $c(n)=\max_{v\in\mathcal W}n\cdot v$. Intersecting the half-spaces
$\{v:n\cdot v\leq c(n)\}$ with the ray $\{re:r\geq0\}$ gives the second
formula in \eqref{eq:normal-ray}; the minimizing normal is unique because
$\partial\mathcal W$ has a unique outward normal at $v_e$. Finally,
$H(p_e)=p_e\cdot v_e$ is Fenchel equality at $\mathcal I(v_e)=0$.

\label{thm:quantitative-geometry}
\begin{enumerate}
\item The map $E(n)=v(n)/|v(n)|$ is a smooth diffeomorphism of the

At the critical pair the function
$G(\lambda,n)=\lambda n\cdot\nabla H(\lambda n)-H(\lambda n)$
has derivative $\partial_\lambda G=\lambda n^{\mathsf T}\Sigma n>0$,
where $\Sigma=\nabla^2H(\lambda n)$. The implicit function theorem
makes $\lambda(n)$ smooth. For $\tau\perp n$, differentiation gives
\begin{align}
 D\lambda(n)[\tau]
   &=-\lambda\frac{n^{\mathsf T}\Sigma\tau}{n^{\mathsf T}\Sigma n},
       \nonumber                                          
\\
 Dv(n)[\tau]
   &=\lambda\left(\Sigma\tau-
       \frac{n^{\mathsf T}\Sigma\tau}{n^{\mathsf T}\Sigma n}\Sigma n\right).
       \label{eq:normal-velocity-derivative}
\end{align}
In particular $n\cdot Dv(n)[\tau]=0$. On the tangent space the
quadratic form of this derivative is
$$
 \lambda\left(\tau^{\mathsf T}\Sigma\tau
       -\frac{(n^{\mathsf T}\Sigma\tau)^2}{n^{\mathsf T}\Sigma n}\right)>0
       \qquad(\tau\ne0).
$$
This is strict Cauchy--Schwarz in the $\Sigma$ inner product; equality
would make $\tau$ parallel to $n$. Consider the compact set
$\mathcal K=\{(n,\tau):n\in\mathbb S^{d-1},\ \tau\perp n,\ |\tau|=1\}$.
The displayed quadratic form is continuous and strictly positive on
$\mathcal K$, because $\lambda(n)$ and $\nabla^2H(\lambda(n)n)$ are
continuous. Its minimum and maximum on $\mathcal K$ are therefore numbers
$0<c_0\le C_0<\infty$, which gives the uniform tangent-space bounds. Differentiating $c(n)=n\cdot v(n)$
therefore gives
\begin{equation*}                                 
 v(n)=c(n)n+\nabla_{\mathbb S^{d-1}}c(n),\qquad
 \nabla^2_{\mathbb S^{d-1}}c+cI=Dv|_{n^\perp}.
\end{equation*}
Here $\nabla_{\mathbb S^{d-1}}$ and $\nabla^2_{\mathbb S^{d-1}}$ denote
the gradient and Hessian for the unit sphere, and $I$ is the identity on
$n^\perp$. The first identity uses $n\cdot Dv=0$; the second follows by
taking the tangential part of its derivative.

Lemma~\ref{lem:geometry} already gives a bijection between the radial
and normal parametrizations of $\partial\mathcal W$. Formula
\eqref{eq:normal-velocity-derivative} makes the normal parametrization
a local diffeomorphism. Radial projection is also a local
diffeomorphism on this boundary, since $n\cdot v(n)=c(n)>0$.
Thus $E$ is a smooth diffeomorphism. In particular all quantities
$e\mapsto p_e$, $\alpha_e$, and $w(e)$ are smooth.

Fix $e$ and write $n_0=n_e$. The nonnegative function
$f_e(n)=c(n)-n\cdot v_e$ vanishes only at $n_0$. Its first derivative
there is zero and its spherical Hessian there equals
$\nabla^2_{\mathbb S^{d-1}}c(n_0)+c(n_0)I$.
By the preceding tangent-space estimate, the spherical Hessian of $f_e$ at
$n_e$ has eigenvalues in a fixed interval $[c_0,C_0]$, independently of $e$.
Continuity of this Hessian in $(e,n)$ provides $\delta>0$ and constants
$0<c_1<C_1$ such that
$c_1|n-n_e|^2\le f_e(n)\le C_1|n-n_e|^2$ whenever
$|n-n_e|<\delta$. On the compact set
$\{(e,n)\in\mathbb S^{d-1}\times\mathbb S^{d-1}:|n-n_e|\ge\delta\}$,
$f_e(n)>0$ because $n_e$ is the unique supporting normal; hence its minimum
there is positive. Since $\lambda(n)$ is continuous and has positive minimum
and finite maximum on the sphere, multiplying these estimates by
$\lambda(n)$ proves \eqref{eq:quadratic-support-gap}.

Since $\min_n H(p(n))>0$, the right-hand sides defining
$\mathcal O_t(C)$ are positive for large $t$. Its radial function is
\begin{equation}\label{eq:radial-minimum}
 r_t(e)=\min_{n\cdot e>0}
 \left\{t\frac{c(n)}{n\cdot e}
       -\frac{\kappa_d\log t-C}{\lambda(n)n\cdot e}\right\}.
\end{equation}
There is $c_0>0$, independent of $e$ and all sufficiently large $t$,
such that no minimizer in \eqref{eq:radial-minimum} satisfies
$0<n\cdot e<c_0$: on that region the quotient is larger than the value
obtained by testing $n_e$. On the compact set $n\cdot e\ge c_0$, put
$L_t=\kappa_d\log t-C$ and $d_e(n)=1/(\lambda(n)n\cdot e)$.
The difference from the value at $n_e$ in \eqref{eq:radial-minimum}
is
$$
 t\left(\frac{c(n)}{n\cdot e}-w(e)\right)
       -L_t\bigl(d_e(n)-d_e(n_e)\bigr).
$$
The first parenthesis is bounded below by a positive constant times
$|n-n_e|^2$, by \eqref{eq:quadratic-support-gap}; the second has
absolute value at most $C|n-n_e|$. At a minimizing normal the
difference is nonpositive, hence $|n-n_e|\leq C_C\log t/t$.
Substituting $|n-n_e|\leq C_C\log t/t$ and using the quadratic upper
bound obtained from \eqref{eq:quadratic-support-gap} on the compact set
$n\cdot e\geq c_0$ changes the value in \eqref{eq:radial-minimum} by at most
$C_C(\log t)^2/t$. The value at $n_e$ is exactly
the three displayed terms in \eqref{eq:radial-support-expansion}.
This proves the theorem.

The following estimates hold uniformly for $p\in\mathcal P$ and
the starting site.
\begin{enumerate}[label=\textup{(\roman*)}]
\item The transition probabilities of $Q_p$ and $\widehat Q_p$
are bounded by $C(1+t)^{-d/2}$.
\item For each fixed $D\geq0$ and all sufficiently large $t$,
\begin{equation}\label{eq:crossing}
 \mathbb P_x(Y_t\leq-D-r)\geq\frac25-
                  \frac{C_D(1+r_+)}{\sqrt t},\qquad r\in\mathbb R,
\end{equation}
and
$$
 \mathbb P_x(Y_t\ge D+r)\ge\frac25-
                  \frac{C_D(1+r_+)}{\sqrt t},\qquad r\in\mathbb R.
$$
\end{enumerate}

<!-- proof-5: body_md -->
We use a finite-cell Bloch representation \cite{Kuchment} and keep
the estimates uniform on $\mathcal P$.
Take the Bloch parameter in
$\mathcal B=\prod_i[-\pi/L_i,\pi/L_i]$, with opposite faces
identified. For a vector $f$ indexed by $\mathcal C$, define the Bloch matrix
$$
 (A_p(k)f)(r)=\sum_{|z|_1=1}q_p(r,z)
       \bigl(e^{\mathrm i k\cdot z}f(r+z)-f(r)\bigr),
$$
where the argument of $f$ is reduced modulo $\Gamma$. The transition
kernel is given by the finite-cell inverse Fourier formula
\begin{equation}\label{eq:bloch-kernel-formula}
 P_t^{Q_p}(x,y)=\frac{|\mathcal C|}{(2\pi)^d}
   \int_{\mathcal B} e^{\mathrm i k\cdot(x-y)}
       [e^{tA_p(k)}]_{r_x,r_y}\,\mathrm d k,
 \qquad r_x=x\bmod\Gamma,\quad r_y=y\bmod\Gamma.
\end{equation}
Indeed the integral at $t=0$ is $\1_{\{x=y\}}$ by Fourier orthogonality
on $\Gamma$, and applying $Q_p$ in $x$ differentiates the matrix exponential.
The resulting bounded solution is unique by
Lemma~\ref{lem:fixed-domain-comparison}. Near zero, the principal eigenvalue
of $A_p(k)$ is
$$
 H(p-\mathrm i k)-H(p)=-\mathrm i v_p\cdot k-\tfrac12 k^{\mathsf T}\Sigma_p k
                         +O(|k|^3).
$$
Its eigenprojection is analytic and uniformly bounded; all the other
eigenvalues have real parts bounded above by a negative constant near
zero. Away from zero on this torus the whole spectrum has negative real
part. Indeed, if an eigenvalue had real part zero, choose a component
of an eigenvector with maximal modulus. Equality in the real-part
inequality forces every Bloch-weighted neighboring value to equal
that component. Irreducibility propagates this equality along all
edges. A path with net displacement $L_i e_i$ then forces
$\mathrm e^{\mathrm i k_iL_i}=1$ for all $i$. This is the zero point of the torus.
We choose the uniform spectral gaps in two stages. At $k=0$, the
nonprincipal spectrum is bounded to the left of $-4\eta$ for some
$\eta>0$, uniformly in $p\in\mathcal P$. Fix $\delta>0$ small enough
that, for $|k|\leq\delta$, this spectrum stays to the left of $-2\eta$
and the principal eigenvalue has real part greater than $-\eta/2$
and at most $-c|k|^2$, as follows from the uniform expansion above.
If the cell has one site, the complementary subspace is empty and
this part is omitted. On the remaining compact set $|k|\geq\delta$
in the Bloch torus, the full spectral bound is at most $-\eta_1<0$.

The matrices have uniformly bounded norm. Near zero, choose a fixed
rectangular contour $\mathcal G$ enclosing only the complementary
spectrum, with right edge $\operatorname{Re}z=-\eta$ and the other
edges outside a common bound for the spectra. If $P^\perp_p(k)$ is
the complementary spectral projection, the Dunford formula gives
$$
 e^{tA_p(k)}P^\perp_p(k)=\frac1{2\pi\mathrm i}
 \int_{\mathcal G}e^{tz}(zI-A_p(k))^{-1}P^\perp_p(k)\,dz.
$$
The resolvent is uniformly bounded on the compact parameter and
contour sets, so this term is bounded by $Ce^{-\eta t}$. Away from
zero, a second fixed contour with right edge
$\operatorname{Re}z=-\eta_1/2$ gives
$\|e^{tA_p(k)}\|\leq Ce^{-\eta_1t/2}$ in the same way.
For the principal mode near zero, the real part of the exponent is
at most $-c|k|^2$, so
$\int_{|k|\le\delta}e^{-ct|k|^2}dk\le Ct^{-d/2}$. This proves the kernel
bound for $Q_p$. Since $B_{-p}=B_p^{\mathsf T}$, the positive eigenvector of
$B_{-p}$ is proportional to $\phi_p^*$ and hence
$\widehat Q_p=Q_{-p}$; because $-p\in\mathcal P$, the identical estimates
hold for $\widehat Q_p$.

For the scalar projection, use the finite-cell matrix with jump factors
$\mathrm e^{\mathrm i\theta p\cdot z}$. Multiplication at the endpoints by
$\mathrm e^{\mathrm i\theta\chi_p^\pm}$ includes the bounded corrector.
The preceding expansion implies, for some uniform $\delta,c,C>0$,
\begin{equation}\label{eq:char-small}
 |\mathbb E_x\mathrm e^{\mathrm i\theta Y_t}|
       \leq C\mathrm e^{-ct\theta^2}+C\mathrm e^{-ct}
       \qquad(|\theta|\leq\delta).
\end{equation}
Fix $\theta$ in a bounded set and insert $\theta/\sqrt t$ into the
finite-cell characteristic matrix. The principal exponent equals
$-\sigma_p^2\theta^2/2+O(|\theta|^3/\sqrt t)$ and the remaining spectral
modes are $O(e^{-ct})$, uniformly in $p\in\mathcal P$ and the initial cell,
where $\sigma_p^2=p^{\mathsf T}\Sigma_pp$ is bounded above and away from
zero. Hence
$\mathbb E_x e^{\mathrm i\theta Y_t/\sqrt t}\to
 e^{-\sigma_p^2\theta^2/2}$ uniformly on bounded $\theta$-sets. L\'evy's
continuity theorem \cite[Section~26]{Billingsley} then gives convergence of
$Y_t/\sqrt t$ to the centered normal law with variance $\sigma_p^2$.
Uniformity follows by contradiction: from any violating sequence one may
extract $p_k\to p$ and a fixed initial residue, after which the preceding
characteristic-function convergence and L\'evy's theorem give the claimed
limit.

We also need a concentration estimate for intervals. There is a nonnegative
integrable function $g$, with Fourier transform supported in $[-\delta,\delta]$,
which is bounded below by a positive constant on $[-1,1]$ after a
fixed rescaling; for example take a rescaled square of
$\sin s/s$. Applying its translate and rescaling to an interval $J$
of length $\ell$ and using Fourier inversion and
\eqref{eq:char-small} gives
\begin{equation}\label{eq:concentration}
 \sup_J\mathbb P_x(Y_t\in J)\leq C(1+\ell)t^{-1/2},\qquad t\geq1.
\end{equation}
For lengths below a fixed constant, enlarge the interval to that
constant; for larger lengths rescale $g$, so its Fourier support stays
inside $[-\delta,\delta]$. The Fourier integral is bounded by
$C(1+\ell)\int_{-\delta}^{\delta}(\mathrm e^{-ct\theta^2}+\mathrm e^{-ct})\mathrm d\theta$.
This proves \eqref{eq:concentration} for all $p\in\mathcal P$.
The limiting normal variances stay in a fixed positive compact
interval. Uniform convergence of the distribution functions near
zero, together with $-D/\sqrt t\to0$, therefore gives
$\mathbb P_x(Y_t\le-D)\ge9/20$ for all sufficiently large $t$.
If $r>0$, subtracting the concentration
bound \eqref{eq:concentration} for the interval $[-D-r,-D]$ gives
\eqref{eq:crossing}; if $r\le0$, the event
$\{Y_t\le-D\}$ is contained in $\{Y_t\le-D-r\}$. Replacing the
characteristic parameter $\theta$ by $-\theta$ in the preceding
characteristic-function and concentration calculations gives the stated
upper-tail estimate.

All weighted moments used below are finite on bounded time intervals.

\eqref{eq:global-exponential-bound}. Then every sum obtained below by
multiplying $u(t,x)$ by a fixed polynomial in $|x|$ and by
$\mathrm e^{p\cdot x}$, $p\in\mathcal P$, is absolutely summable uniformly on
bounded time intervals. The same domination applies to $u_t$, since
$|u_t(t,x)|\le\sum_{|z|_1=1}u(t,x+z)+(2d+L_F)u(t,x)$.
These bounds justify termwise differentiation of the weighted sums
and give a $C^1$ weighted density in $\ell^1$ on bounded intervals.
The operator $\widehat Q_p^{\mathsf T}$ is bounded on $\ell^1$,
with norm at most twice the maximal exit rate, so
variation of constants also holds there. Equation

\begin{equation}\label{eq:M-derivative}
 M'(t)=-\sum_x\mathcal R(t,x)\leq0.
\end{equation}
There is also an unweighted bound
\begin{equation}\label{eq:unweighted-polynomial}
 \sum_xu(t,x)\leq C(1+t)^d.
\end{equation}
Fix $\varepsilon>0$. Estimate \eqref{eq:global-exponential-bound} gives
$u(t,x)\le C_\varepsilon\exp(c_\varepsilon t-\varepsilon|x|_1)$.
Split the sum into $|x|_1\le R(1+t)$ and its complement, with
$R>c_\varepsilon/\varepsilon+1$. The first set contains
$O((1+t)^d)$ lattice points and $u\le1$ there. On the complement,
$c_\varepsilon t-\varepsilon|x|_1\le-\varepsilon|x|_1/2$ after increasing
$R$ if necessary, and the lattice exponential tail is summable uniformly in
$t$. This proves \eqref{eq:unweighted-polynomial}.

To measure how much mass lies ahead of the moving plane, put $L=d+3$ and
$$
\begin{gathered}
 z(t,x)=p\cdot x-ht+\chi^+(x)+L\log(1+t),\\
 J(t)=\sum_xz(t,x)W(t,x),\qquad J_+(t)=\sum_xz(t,x)_+W(t,x).
\end{gathered}

bounded for $p\in\mathcal P$ and $x\in\mathcal C$,
\eqref{eq:unweighted-polynomial} implies
\begin{equation}\label{eq:negative-moment}
 \sum_{z(t,x)<0}(1+|z(t,x)|)W(t,x)\leq C(1+t)^{-3}.
\end{equation}

       \leq\frac{L}{1+t}M(t)+C(1+t)^{-3}.
\end{equation}

 \end{split}
\end{equation*}

\end{equation}

$M(t)\le C(1+t)^{-1/2}$.

\label{lem:survival}
For either corrected scalar martingale $Y$ in
Lemma~\ref{lem:Markov-estimates}, with $Y_0=0$, fix a starting height
$r>0$ and set
$\tau=\inf\{s\geq0:r+Y_s\leq0\}$. Then
\begin{equation}\label{eq:survival}
 \mathbb P_x(\tau>t)\leq
       \min\{1,C(1+r)/\sqrt{1+t}\}.
\end{equation}
<!-- end lemma-6 -->
<!-- proof-7: body_md -->
The corrected jump sizes and the continuous drift between jumps are
uniformly bounded. A continuous exit has height zero, and a jump exit
overshoots by at most one corrected jump, so
$r+Y_\tau\in[-B,0]$ on $\{\tau<\infty\}$ for a uniform $B$.
The Poisson construction gives
$\mathbb E_x\sup_{0\le v\le s+1}|Y_v|^2<\infty$.
Bounded optional stopping therefore gives
$\mathbb E_xY_{\tau\wedge s}=0$: one may approximate the stopping
time from above on dyadic grids and pass to the limit by
right-continuity and the integrable path supremum. Splitting
according to $\{\tau>s\}$ yields
$$
 \mathbb E_x[(r+Y_s)\1_{\{\tau>s\}}]\leq r+B.
$$
At time $s$, condition on the surviving walk and its cell position.
The crossing estimate with $D=0$ bounds the probability of ending
above zero another $s$ units of time later by
$3/5+C(1+r+Y_s)/\sqrt s$. Survival is a subset of this event.
With $h(s)=\mathbb P_x(\tau>s)$, the preceding conditional estimate gives
$$
 h(2s)\leq\frac35h(s)+\frac{C(1+r+B)}{\sqrt s}.
$$
Fix $s_0$ large enough for the displayed recurrence and set
$s_n=2^ns_0$ and $b_n=\sqrt{s_n}\,h(s_n)$. Multiplying the recurrence by
$\sqrt{2s_n}$ yields
$$
 b_{n+1}\le \frac{3\sqrt2}{5}b_n+C(1+r+B).
$$
Since $3\sqrt2/5<1$, iteration gives
$b_n\le C(1+r+B)$ uniformly in $n$. Monotonicity of $h$ then interpolates
between consecutive $s_n$ and proves \eqref{eq:survival}; increasing $C$
covers $0<t<s_0$.
The conditioning uses the Markov property of the pair consisting
of the height and the cell position.

\label{lem:killed-kernel-upper}
The following bounds are uniform for $p\in\mathcal P$ and $b\in\mathbb R$.
\begin{enumerate}[label=\textup{(\roman*)}]

 \mathsf K_p(s,t;x,y)
 \le C\frac{(1+d_{p,b}(t,x))_+(1+d_{p,b}(s,y))_+}
              {(1+t-s)^{d/2+1}}.
\end{equation}
\item If $d^\chi_{p,b}(t,x)>0$ and $d^\chi_{p,b}(s,y)>0$, then
\begin{equation}\label{eq:corrected-killed-kernel-upper}
 \mathsf K_{p,\chi}(s,t;x,y)
 \le C(C_0)\frac{(1+d^\chi_{p,b}(t,x))(1+d^\chi_{p,b}(s,y))}
 {(1+t-s)^{d/2+1}}.
\end{equation}
\end{enumerate}

\begin{equation}\label{eq:equal-exit-rates}
 \Lambda_p(z):=\sum_{|b|_1=1}q_p(z,b)

       \prod_{i=1}^m q_p(z_{i-1},z_i-z_{i-1}).

\eqref{eq:killed-kernel-upper}. All constants depend on $\chi$ only
through $C_0$ and the already uniform corrector bounds.

A transverse principal eigenvector, extended constantly in the
longitudinal coordinate, is a positive eigenvector of the full tilted
cell matrix. Its eigenvalue is the right-hand side of
\eqref{eq:transverse-factorization}, so Perron--Frobenius theory
identifies this value with $H(p_1,p')$. At $p'=0$ the transverse
eigenvalue is $a$. Applying the proof of Lemma~\ref{lem:spectral}
to the transverse cell gives the evenness of $H_\perp$ and the
positive definiteness of $\Sigma_\perp$.

Conjugating $\mathcal A_\perp-a$ by $\psi_\perp$ gives the Markov
generator
$$
 Q_\perp h(y)=\sum_{|b|_1=1}
   \frac{\psi_\perp(y+b)}{\psi_\perp(y)}(h(y+b)-h(y)).
$$
It is reversible with cell probability $\psi_\perp^2$, and its
stationary mean displacement is zero. If $P_\perp(s;y,z)$ is its
transition kernel, then
$\mathrm e^{-as}K_\perp(s;y,z)=\psi_\perp(y)P_\perp(s;y,z)/\psi_\perp(z)$.
The Bloch decomposition in the proof of
Lemma~\ref{lem:Markov-estimates} applies to this zero-drift generator
in dimension $n\geq1$ and gives \eqref{eq:transverse-kernel-upper}.

\label{lem:logconcavity}
The tilted kernel $G(t,j)$ from Lemma~\ref{lem:longitudinal} is
strictly positive and log-concave as a sequence in $j$ for every $t>0$:
$$
    G(t,j)^2\geq G(t,j-1)G(t,j+1)\qquad(j\in\mathbb Z).
$$
Consequently, $G(t,j+1)/G(t,j-1)$ is nonincreasing in $j$.
<!-- end lemma-12 -->
<!-- proof-14: body_md -->
We use preservation of log-concavity under convolution and verify it
for the sequences in this argument. A nonnegative
summable sequence $a$ with interval support and
$a_k^2\geq a_{k-1}a_{k+1}$ has a Toeplitz matrix
$T(a)_{ij}=a_{i-j}$ whose $2\times2$ minors are nonnegative. Indeed,
for $i<j$ and $r<s$, the two indices in
$a_{i-r}a_{j-s}$ lie between the two indices in
$a_{i-s}a_{j-r}$, and their sums are equal. The monotonicity of
successive ratios on the positive support gives
$$
    a_{i-r}a_{j-s}\geq a_{i-s}a_{j-r}.
$$
If the right-hand product is positive, interval support justifies all
successive ratios between its two indices and the displayed inequality follows
from their monotonicity. If the right-hand product is zero, the inequality follows directly
from nonnegativity.

If both $a$ and $b$ have this property, the finite Cauchy--Binet formula
shows that every $2\times2$ minor of
$T(a)_{\{i,j\},[-N,N]}T(b)_{[-N,N],\{r,s\}}$ is nonnegative.
Its entries converge to those of $T(a*b)_{\{i,j\},\{r,s\}}$ as
$N\to\infty$, since the convolution sums converge absolutely.
Thus $a*b$ is log-concave as well.

The distribution of a Poisson random variable has interval support and
successive ratio $\rho/(k+1)$ on its positive support, where $\rho>0$
is its parameter. It is therefore log-concave. Reflection preserves
log-concavity. The kernel $G(t,\cdot)$ is the convolution of the
Poisson distribution with parameter $\mathrm e^{\lambda_*}t$ and the
reflected Poisson distribution with parameter $\mathrm e^{-\lambda_*}t$.
The convolution property proves the asserted inequality. Moreover,
$G(t,j)>0$ for every $j\in\mathbb Z$ because the two independent Poisson variables
can realize, for example, the pair $(j,0)$ when $j\ge0$ and $(0,-j)$ when
$j<0$. Its successive ratios are
positive and nonincreasing, so the product of two adjacent ratios,
$G(t,j+1)/G(t,j-1)$, is nonincreasing.

\label{lem:entire-comparison}
Let $V,W$ be entire $[0,1]$-valued solutions for the same periodic
reaction. Choose $\delta,\nu>0$ such that
$F_s(x,s)\leq-\nu$ for $1-\delta\leq s\leq1$ and every $x$.
\begin{enumerate}[label=\textup{(\roman*)}]
\item If $V\geq1-\delta$ on a set $E\subset\mathbb R\times\mathbb Z^d$ and
$W\leq V$ on its complement, then $W\leq V$ everywhere.
\item If $W\leq V$ and equality holds at one space--time point,

\end{enumerate}

<!-- proof-20: body_md -->
Put $D=(W-V)_+$. Wherever $D>0$, the assumed ordering outside $E$
places both solution values in $[1-\delta,1]$. The reaction difference
is therefore at most $-\nu D$. The discrete positive-part inequality
gives, almost everywhere in time,
$$
 D_t\leq\Delta D-\nu D.
$$
At a site where $D=0$, this follows because the positive part has zero
derivative almost everywhere and $\Delta D\geq0$. Since $0\leq D\leq1$,
comparison from time $t-h$ yields
$D(t)\leq\mathrm e^{-\nu h}\mathrm e^{h\Delta}D(t-h)\leq\mathrm e^{-\nu h}$.
Letting $h\to\infty$ proves the first assertion.

For the second, the nonnegative difference $D=V-W$ satisfies
$D_t=\Delta D+b(t,x)D$ with $b$ uniformly bounded. If
$D(t_0,x_0)=0$ and $K\geq\|b\|_\infty$, comparison with the heat
equation gives, for every $h>0$,
$$
 0\geq\mathrm e^{-Kh}
 \sum_{y\in\mathbb Z^d}\mathsf p_d(h,x_0-y)D(t_0-h,y).
$$
Every summand is nonnegative and the heat kernel is strictly positive.
Thus $D$ vanishes at every time before $t_0$, and uniqueness for the
linear lattice equation gives the same conclusion afterwards.

We also need this zero-contact argument for the linear limits below,
which can grow exponentially in space. For $\rho>0$, define
$$
 X_\rho=\left\{f:\|f\|_\rho=
       \sup_{x\in\mathbb Z^d}\mathrm e^{-\rho|x|_1}|f(x)|<\infty\right\}.
$$
Neighbor shifts have norm at most $\mathrm e^\rho$ on $X_\rho$, so
$\|\Delta\|\leq2d(\mathrm e^\rho+1)$ there. Multiplication by bounded $b$
is also bounded. A solution bounded in this norm on compact time
intervals satisfies the integral equation in $X_\rho$, where Gronwall's
inequality gives uniqueness and the same heat comparison. The heat-kernel
sum converges because the Poisson jump count has finite exponential
moments. Consequently zero contact also forces equality in this class.

\cite[Chapter~II]{Kato} gives its spectral projection
$\phi_{p-zn}(\phi^*_{p-zn})^{\mathsf T}$, under the normalization
$(\phi^*_{p-zn})^{\mathsf T}\phi_{p-zn}=1$, and an analytic inverse on
the complementary subspace. This proves
\eqref{eq:critical-resolvent-local}.
The identities $H(p)=c\lambda$ and $n\cdot\nabla H(p)=c$ cancel

                +\mathrm i c\tau\eta(x)=0.

$$
 J_N(t)=\int_0^t\mathcal U_{D_N}(t,s)(v_N(s)^2)\,ds
 \le\int_0^t\mathcal U_{D_{N+1}}(t,s)(v_{N+1}(s)^2)\,ds=J_{N+1}(t).
$$ Their monotone limits are therefore $v$ and $J_v$ by the
definitions of $\mathcal U_D$ and \eqref{eq:quadratic-Dirichlet-potential}.
The bound $J_N(t)\leq M_{T_0}t\,v_N(t)$ controls the limit for
$t\leq T_0$.
Where $v>0$ the quotients converge to the stated quotient. Where $v=0$
the bound $0\leq w_N\leq v_N$ gives convergence to zero.
Thus \eqref{eq:Dirichlet-quotient-bound} holds, including at deletion
times. To justify the subsolution property without differentiating a
pointwise limit, first integrate the finite-box inequality between any
two times $s<t$, including the nonpositive deletion jumps. Since
$0\leq w_N\leq1$, the integrands are uniformly bounded on $[s,t]$;
the Laplacian at a fixed vertex contains only finitely many neighbors.
Dominated convergence therefore passes this integral inequality to the
limiting quotient. The resulting integral subsolution takes values in
$[0,1]$ and has the prescribed post-deletion values. Taylor's lower
bound $a(x)r-Kr^2\leq F(x,r)$ on this interval gives precisely the
subsolution inequality for the original equation.

\label{lem:Dirichlet-bridge}
The weights $\mathfrak b_{t,x}(s,\cdot)$ sum to one and are consistent
with these backward kernels. Moreover,
\begin{equation}\label{eq:normalized-potential-identity}
 \frac{J_v(t,x)}{v(t,x)^2}
  =\int_0^t\sum_y\mathfrak b_{t,x}(s,y)
                   \frac{v(s,y)}{v(t,x)}\,\mathrm d s.
\end{equation}

<!-- proposition-3: body_md -->
\label{prop:general-lower-criterion}
Assume \eqref{eq:periodic-kpp}--\eqref{eq:u0}. Suppose that, for some
$m,C_*,R,t_0>0$, every $t\geq t_0$ admits $x_t\in D(t)$ with
\begin{equation}\label{eq:criterion-seed-inputs}
 |x_t-m_e^T(t)e|\leq R,\qquad v(t,x_t)\geq m,\qquad
 J_v(t,x_t)\leq C_*v(t,x_t)^2.
\end{equation}
For every $\theta\in(0,1)$ and finite $\rho\geq0$, there are
$C,t_1>0$ such that
\begin{equation}\label{eq:criterion-inner-tube}
 u(t,x)\geq\theta
 \quad\hbox{if }t\geq t_1,\quad |x-re|\leq\rho,\quad
                     0\leq r\leq m_e^T(t)-C.
\end{equation}
Consequently, \eqref{eq:general-target} holds for $\rho\geq\sqrt d$.
<!-- end proposition-3 -->
<!-- proof-34: body_md -->
Theorem~\ref{thm:Dirichlet-quotient} and
\eqref{eq:criterion-seed-inputs} give, for every $s\geq t_0$,
$u(s,x_s)\geq\eta:=\min\{1/2,(m^{-1}+KC_*)^{-1}\}>0$.
There is no assumption on the size of $v$ away from these seed sites.
Restart \eqref{eq:heat-comparison} at time $s$. Since
$u(s,x_s)\ge\eta$ and the heat semigroup is positive,
$$
 u(s+1,x)\ge(e^{\Delta_{\mathbb Z^d}}u(s))(x)
 \ge\eta\mathsf p_d(1,x-x_s).
$$
For $|x-m_e^T(s)e|\le\rho$ and
$|x_s-m_e^T(s)e|\le R$, the displacement $x-x_s$ belongs to the finite set
$\{z\in\mathbb Z^d:|z|\le R+\rho\}$. Hence
$\eta_\rho:=\eta\min_{|z|\le R+\rho}\mathsf p_d(1,z)>0$ and the displayed
bound becomes $u(s+1,x)\ge\eta_\rho$.
The last minimum is over the finite set of lattice displacements with
length at most $R+\rho$, and is positive. The amplification lemma
then gives $u(t,x)\geq\theta$ whenever $t\geq s+1+S_\rho$, with
$S_\rho$ uniform in $s,x$.

Since $\alpha_e w(e)=h$, the derivative of $m_e^T$ is positive for
our choice of $T$ and is bounded above by $w(e)$. For every sufficiently
large $r$ there is a time $s\geq t_0$ with $m_e^T(s)=r$. If
$r\leq m_e^T(t)-C$, where $C>w(e)(1+S_\rho)$, then
$s\leq t-1-S_\rho$. Applying Lemma~\ref{lem:amplification} to the seed produced at time $s+1$
gives \eqref{eq:criterion-inner-tube} for these $r$. The omitted bounded range of $r$ contains only finitely many lattice
vertices in the tube. At time $1$, \eqref{eq:heat-comparison} is strictly
positive at each of these vertices; taking the minimum over this finite set
produces a positive seed level. Lemma~\ref{lem:amplification} applied to that
minimum supplies one time after which all of these vertices exceed
$\theta$.

For $\rho\geq\sqrt d$, round the coordinates of
$(m_e^T(t)-C-2\sqrt d)e$ to the nearest integers. The rounded vertex
is within $\sqrt d/2$ of this point, belongs to the tube in
\eqref{eq:tube-level-definition}, and has projection
$m_e^T(t)-O(1)$. Hence
$R_{\theta,e,\rho}(t)\geq m_e^T(t)-O(1)$. Theorem~\ref{thm:upper} gives
$R_{\theta,e,\rho}(t)\leq m_e^T(t)+O(1)$. Finally
$m_e^T(t)=w(e)t-(\kappa_d/\alpha_e)\log t+O(1)$ for fixed $T$.
Combining the two bounds proves \eqref{eq:general-target}.
