\label{thm.7.2.003}
{\rm (Milnor \cite{Milnor})} 
In the situation as above, assume that 
the complex hypersurface $X_0=f^{-1}(0)\subset X$ 
has an isolated singular point at a point 
$x \in X_0=f^{-1}(0)\subset X$. 
Then the Milnor fiber $F_x \subset X \setminus X_0$ of $f$ at it 
is homotopy equivalent to the bouquet of some 
$(n-1)$-dimensional spheres $S^{n-1}$: 
\begin{equation}
F_x  \sim  S^{n-1}\vee\cdots\vee S^{n-1}.
\end{equation}
In particular, for its cohomology groups 
$H^j(F_x; {\mathbb C} )$ ($j \in {\mathbb Z}$) we have 
\begin{equation}
H^j(F_x; {\mathbb C} ) \simeq 
\begin{cases}
\ {\mathbb C} & (j=0) \\
& \\
\ {\mathbb C}^{\mu} & (j=n-1) \\
 & \\
\ 0 & (j \neq 0,n-1), 
\end{cases}
\end{equation}
where $\mu>0$ is the number of spheres $S^{n-1}$ in 
the above bouquet decomposition.

\label{def.7.1.001}
{\rm (Deligne~\cite{Deligne})} 
For ${\cal F} \in {\bf D}^{\mathrm{b}} (X)$ we define its nearby cycle 
complex $\psi_f( {\cal F} )\in {\bf D}^{\mathrm{b}} (X_0)$ along $f$ by 
\begin{equation}
\psi_f( {\cal F} ) = i^{-1} {\rm R} (j\circ\pi)_*(j\circ\pi)^{-1} {\cal F}

\end{equation}
Moreover, we define the vanishing cycle 
complex $\phi_f( {\cal F} )\in {\bf D}^{\mathrm{b}} (X_0)$ of ${\cal F} \in {\bf D}^{\mathrm{b}} (X)$

\phi_f( {\cal F} ) \overset{+1}{\longrightarrow}, 
\end{equation}
where the morphism $i^{-1} {\cal F}  \longrightarrow \psi_f( {\cal F} )$

${\rm id} \longrightarrow {\rm R} (j\circ\pi)_*(j\circ\pi)^{-1}$ 
of functors (for the more precise definition, see 
Kashiwara-Schapira \cite[page 351]{K-S}).

\begin{equation}
\psi_f( {\rm R} \rho_* {\cal G} ) \simeq {\rm R} (\rho|_{Y_0})_*\psi_g( {\cal G} ), \qquad 
\phi_f( {\rm R} \rho_* {\cal G} ) \simeq {\rm R} (\rho|_{Y_0})_*\phi_g( {\cal G} ). 
\end{equation}

{\rm (Broughton \cite{Broughton})} \label{tame}
Assume that $f$ is tame at infinity. 
Then for any point $c \in {\mathbb C}$ its fiber $f^{-1}(c)$ of $f$ 
is homotopy equivalent to the bouquet of some 
$(n-1)$-dimensional spheres $S^{n-1}$: 
\begin{equation}
f^{-1}(c)  \sim  S^{n-1}\vee\cdots\vee S^{n-1}.
\end{equation}
Moreover the number of spheres $S^{n-1}$ in this bouquet 
decomposition is equal to $\mu - \mu_c \geq 0$. 
In particular, we have isomorphisms 
\begin{equation}
H^j(f^{-1}(c); {\mathbb C} ) \simeq 
\begin{cases}
\ {\mathbb C} & (j=0) \\
& \\
\ {\mathbb C}^{\mu - \mu_c} & (j=n-1) \\
 & \\
\ 0 & (j \neq 0,n-1). 
\end{cases}
\end{equation}

<!-- proof-1: body_md -->
Let $i_c: \{ c \} \hookrightarrow {\mathbb C}$ be the inclusion map 
and consider the distinguished triangle 
\begin{equation}\label{d-t-1} 
i_c^{-1} {\rm R}f_! {\mathbb C}_X \longrightarrow \psi_{t-c} {\rm R}f_! {\mathbb C}_X 
\longrightarrow 
\phi_{t-c} {\rm R}f_! {\mathbb C}_X \overset{+1}{\longrightarrow}. 
\end{equation}
Note that by Theorem \ref{thm.7.2.006} there exist isomorphisms 
\begin{align} 
i_c^{-1} {\rm R}f_! {\mathbb C}_X & \simeq {\rm R}\Gamma_c 
(f^{-1}(c); {\mathbb C}_{f^{-1}(c)}), \\ 
\psi_{t-c} {\rm R}f_! {\mathbb C}_X & \simeq {\rm R}\Gamma_c 
(f^{-1}(c+ \eta ); {\mathbb C}_{f^{-1}(c+ \eta)}) 
\quad (0< \eta \ll 1). 
\end{align}
Since for $0< \eta \ll 1$ the fiber $f^{-1}(c+ \eta )$ of $f$ is 
generic, by Theorem \ref{tame} we thus obtain 
isomorphisms 
\begin{equation}
H^j \psi_{t-c} {\rm R}f_! {\mathbb C}_X  \simeq 
\begin{cases}
\ {\mathbb C} & (j=2n-2) \\
& \\
\ {\mathbb C}^{\mu} & (j=n-1) \\
 & \\
\ 0 & (j \neq n-1, 2n-2). 
\end{cases}
\end{equation}
By the tameness at infinity of $f$, 
there exist $r_0 \gg 0$ and a small disk 
$D_{\eta_0}(c) := \{ t \in {\mathbb C} \ | \ 
 |t-c| < \eta_0 \} \subset {\mathbb C}$ ($0< \eta_0 \ll 1$) 
in ${\mathbb C}$ centered at the point $c \in {\mathbb C}$ such that 
for any $r \geq r_0$ and $t \in D_{\eta_0}(c)$ 
the sphere $\partial B(0;r)$ intersects $f^{-1}(t)$ 
transversally (see e.g. \cite[Proposition 3.4]{Tak-4}). 
By the 
non-characteristic deformation lemma \cite[Proposition 2.7.2]{K-S}, 
this implies that we have isomorphisms 
\begin{align} 
i_c^{-1} {\rm R}f_! {\mathbb C}_{B(0;r_0)} & \simeq 
i_c^{-1} {\rm R}f_! {\mathbb C}_X, \\ 
\psi_{t-c} {\rm R}f_! {\mathbb C}_{B(0;r_0)} & \simeq 
\psi_{t-c} {\rm R}f_! {\mathbb C}_X
\end{align}
and hence the distinguished triangle 
\begin{equation}\label{d-t-2} 
i_c^{-1} {\rm R}f_! {\mathbb C}_{B(0;r_0)} \longrightarrow 
\psi_{t-c} {\rm R}f_! {\mathbb C}_{B(0;r_0)} \longrightarrow 
\phi_{t-c} {\rm R}f_! {\mathbb C}_{B(0;r_0)} \overset{+1}{\longrightarrow}. 
\end{equation}
is isomorphic to the one \eqref{d-t-1}. Moreover, by 
Proposition \ref{prp.7.1.002}
there exists an isomorphism 
\begin{equation} 
\phi_{t-c} {\rm R}f_! {\mathbb C}_{B(0;r_0)} \simeq 
{\rm R}\Gamma (f^{-1}(c); \phi_{f-c} {\mathbb C}_{B(0;r_0)}).  
\end{equation}
We can easily see that the support of the vanishing 
cycle sheaf $\phi_{f-c} {\mathbb C}_{B(0;r_0)}$ is contained 
in the finite set of the (isolated) singular points 
of the complex hypersurface $f^{-1}(c) \subset X= {\mathbb C}^n$. 
Then we obtain a concentration 
\begin{equation}
H^j\phi_{t-c} {\rm R}f_! {\mathbb C}_{B(0;r_0)} \simeq 0 
\quad (j \not= n-1) 
\end{equation}
and $\dim H^{n-1}\phi_{t-c} {\rm R}f_! {\mathbb C}_{B(0;r_0)}
= \mu_c$. By the cohomology long exact sequence 
associated to the distinguished triangle \eqref{d-t-1}, 
we thus can show that 
\begin{equation}
H^j_c(f^{-1}(c); {\mathbb C} ) \simeq 0 \quad (j \not= n-1, n, 2n-2), 
\quad H^{2n-2}_c(f^{-1}(c); {\mathbb C} ) \simeq {\mathbb C}
\end{equation}
and 
\begin{equation}
\dim H^{n-1}_c(f^{-1}(c); {\mathbb C} ) - \dim H^{n}_c(f^{-1}(c); {\mathbb C} )
= \mu - \mu_c. 
\end{equation}
Then the assertion immediately follows from the 
distinguished triangle 
\begin{equation}
{\rm R}\Gamma_c(X; {\mathbb C}_{X \setminus f^{-1}(c)}) \longrightarrow {\rm R}\Gamma_c(X; {\mathbb C}_X) 
\longrightarrow 
{\rm R}\Gamma_c(X; {\mathbb C}_{f^{-1}(c)}) \overset{+1}{\longrightarrow}
\end{equation}
and the (generalized) Poincar\'e duality for the complex 
manifold $X \setminus f^{-1}(c)$. 
\hfill $\Box$

i_c^{-1} {\rm R}f_! {\mathbb C}_{U \cap B(0;r_0)} & \simeq

\psi_{t-c} {\rm R}f_! {\mathbb C}_X. 
\end{align}

\begin{equation} 
 \phi_{t-c} {\rm R} f_! {\mathbb C}_{U \cap B(0;r_0)}  \simeq 
\phi_{t-c} {\rm R} f_! {\mathbb C}_U. 
\end{equation}
Let $q_1, \ldots, q_m \in f^{-1}(c)$ be the 
(isolated) singular points of $f^{-1}(c) \subset 
U=X \setminus Q^{-1}(0)$ and $\rho : X \times {\mathbb C} 
\longrightarrow {\mathbb C}$ the projection. Then for 
the (not necessarily) closed embedding 
\begin{equation}
i_f: U=X \setminus Q^{-1}(0) \hookrightarrow 
X \times {\mathbb C}_t \qquad (x \longmapsto (x,f(x))).  
\end{equation}
we have $f= \rho \circ i_f$ and hence 
\begin{equation}
{\rm R} f_! {\mathbb C}_{U \cap B(0;r_0)}  
\simeq 
{\rm R} \rho_* i_{f!} {\mathbb C}_{U \cap B(0;r_0)}, 
\end{equation}
where we used the fact that $\rho$ is proper 
on the support of $i_{f!} {\mathbb C}_{U \cap B(0;r_0)}$.

the meromorphic 
vanishing cycle functor $\phi_{f-c}^{{\rm mero,c}} ( \cdot )$ 
with compact support, we thus obtain an isomorphism 
\begin{equation} 
\phi_{t-c} {\rm R}f_! {\mathbb C}_{U \cap B(0;r_0)} \simeq 
{\rm R}\Gamma (X ; \phi_{f-c}^{{\rm mero,c}} {\mathbb C}_{B(0;r_0)}).  
\end{equation}

\end{equation}

(see Theorem \ref{the-1} (ii)) we obtain the desired concentration 
\begin{equation}
H^j \phi_{t-c} {\rm R} f_! {\mathbb C}_U \simeq 0 \quad (j \not= n-1).  
\end{equation}

By the (generalized) Poincar\'e duality for the complex 
manifold $f^{-1}(c)$ it suffices to show that we have 
\begin{equation}
H^j_c(f^{-1}(c); {\mathbb C} ) \simeq 0 \quad (j \not= n-1, 2n-3, 2n-2) 
\end{equation}
and $H^{2n-3}_c(f^{-1}(c); {\mathbb C} ) \simeq H^{2n-2}_c(f^{-1}(c); {\mathbb C} ) \simeq {\mathbb C}$. 
Since for any $a \in {\mathbb C}$ its fiber $f^{-1}(a)= 
(P-aQ)^{-1}(0) \setminus Q^{-1}(0) \subset U=X \setminus Q^{-1}(0)$ of $f$ 
is an affine variety and ${\mathbb C}_{f^{-1}(a)} [n-1] \in {\bf D}_{\mathrm{c}}^{\mathrm{b}} (f^{-1}(a))$ 
is a perverse sheaf on it, by Artin's 
vanishing theorem for perverse sheaves we have 
\begin{equation}
H^j_c(f^{-1}(a); {\mathbb C} ) \simeq 0 \quad (j \notin [n-1,2n-2]) 
\end{equation}
(see e.g. \cite[Corollary 5.2.19]{Dimca}). 
This implies that we have a 
vanishing 
\begin{equation}\label{van-di}
H^j {\rm R} f_! {\mathbb C}_U \simeq 0 \quad (j \notin [n-1,2n-2]).  
\end{equation}
Note also that by the (generalized) Poincar\'e duality for the complex 
manifold $U= X \setminus Q^{-1}(0)$ Propositions 
\ref{prp.L-T} and \ref{prp.gen-L-T}  
imply that we have 
\begin{equation}
H^j_c(U; {\mathbb C} ) \simeq 0 \quad 
(j \not= n,n+1,2n-1,2n) 
\end{equation}
and $H^{2n-1}_c(U; {\mathbb C} ) \simeq H^{2n}_c(U; {\mathbb C} ) \simeq {\mathbb C}$. 
Let us calculate the cohomology groups 
$H_c^j(U; {\mathbb C} )$ ($j \in {\mathbb Z}$) with compact supports of $U=X \setminus Q^{-1}(0)$ 
by the Leray type spectral sequence $E_r^{p,q}$ ($r \geq 1, p,q \in {\mathbb Z}$) 
such that 
\begin{equation}
E_2^{p,q}=H^p_c( {\mathbb C} ; H^q {\rm R} f_! {\mathbb C}_U) \quad (p,q \in {\mathbb Z} ). 
\end{equation}
By \cite[page 43]{GM} we see that $H^j {\rm R} f_! {\mathbb C}_U$ is 
an algebraic constructible sheaf on ${\mathbb C}$ for any $j \in {\mathbb Z}$. 
Moreover, as in the proof of \cite[Lemma 5.3]{Sabbah-2}, 
by Lemma \ref{vc-lem} we can also show that 
$H^j {\rm R} f_! {\mathbb C}_U$ is a constant sheaf on ${\mathbb C}$ for any $j \geq n+1$. 
For general algebraic constructible sheaves on ${\mathbb C}$ we have the 
following result. 

\begin{lemma}\label{acs-lem} 
Let ${\cal F}$ be an algebraic constructible sheaf on the affine line ${\mathbb C}$. 
Then we have 
\begin{equation}
H^j_c( {\mathbb C} ; {\cal F} ) \simeq 0 \quad (j \notin [0,2]).  
\end{equation}
\end{lemma} 

\begin{proof}
Let $S \subset {\mathbb C}$ be the singular support of ${\cal F}$ and 
consider the exact sequence
\begin{equation}
0 \longrightarrow {\cal F}_{{\mathbb C} \setminus S} \longrightarrow 
{\cal F} \longrightarrow {\cal F}_S \longrightarrow 0. 
\end{equation}
Note that $S$ is a finite subset of ${\mathbb C}$ and hence 
${\cal F}_S$ is a skyscraper sheaf on ${\mathbb C}$. Moreover, 
as ${\mathbb C} \setminus S$ is an affine variety, by Artin's 
vanishing theorem for perverse sheaves (see e.g. \cite[Corollary 5.2.19]{Dimca}) 
we obtain 
\begin{equation}
H^j_c( {\mathbb C} ; {\cal F}_{{\mathbb C} \setminus S} ) 
\simeq H^j_c( {\mathbb C} \setminus S ; {\cal F}|_{{\mathbb C} \setminus S}) \simeq 
0 \quad (j \notin [1,2]).  
\end{equation}
Then the assertion immediatedly follows. 
\hfill $\Box$
\end{proof} 
Summarizing these results, by Lemma \ref{acs-lem} we see that 
$E_2^{p,q}=0$ unless $(p,q) \in {\mathbb Z}^2$ is contained in the set 
\begin{equation}
\{ (0,n-1), (1,n-1), (2,n-1), (0,n), (1,n), (2,n), 
(2,n+1), \ldots (2, 2n-2) \} \quad \subset {\mathbb Z}^2. 
\end{equation}
This in particular implies that for $(p,q)=(2,n)$ the 
spectral sequence $E_r^{2,n}$ ($r \geq 1$) degenerates at $E_2$ 
i.e. $E_2^{2,n}=E_{\infty}^{2,n}$ and hence there exists 
an isomorphism 
\begin{equation}
E_2^{2,n}=H^2_c( {\mathbb C} ; H^n {\rm R} f_! {\mathbb C}_U) \simeq H^{n+2}_c(U; {\mathbb C} ). 
\end{equation}
Note that if $n>3 \Longleftrightarrow 2n-1>n+2$ (resp. $n=3$) 
then we have $H^{n+2}_c(U; {\mathbb C} ) \simeq 0$ (resp. 
$H^{n+2}_c(U; {\mathbb C} )=H^{2n-1}_c(U; {\mathbb C} ) \simeq {\mathbb C}$). 
Let $S \subset \Sigma \subset {\mathbb C}$ be the singular support of the 
algebraic constructible sheaf ${\cal F}_n :=
H^n {\rm R} f_! {\mathbb C}_U$ on ${\mathbb C}$ and denote the 
local system ${\cal F}_n |_{{\mathbb C} \setminus S}$ on ${\mathbb C} \setminus S$ by 
${\mathcal L}$. For $c \in {\mathbb C}$ let $i_c: \{ c \} \hookrightarrow {\mathbb C}$ 
be the inclusion map. Then as in the proof of \cite[Lemma 8.5]{Sabbah-2} 
we obtain an exact sequence 
\begin{equation}
i_c^{-1} {\cal F}_n \longrightarrow \psi_{t-c} {\cal F}_n \longrightarrow 0. 
\end{equation}
This implies that the monodromy of the local system 
${\mathcal L}$ around each point of $S$ is trivial and hence 
${\mathcal L}$ is a constant sheaf on ${\mathbb C} \setminus S$. By the proof of 
Lemma \ref{acs-lem} we thus find that the dimension of 
\begin{equation}
E_2^{2,n}=H^2_c( {\mathbb C} ; {\cal F}_n ) \simeq H^{2}_c( {\mathbb C} \setminus S; {\mathcal L} )
\end{equation}
is equal to the rank of ${\mathcal L}$ i.e. the dimension of 
$H_c^n(f^{-1}(c); {\mathbb C} )$ for the generic fiber $f^{-1}(c) 
\subset X \setminus Q^{-1}(0)$ of $f$. In particular, 
if $n>3$ (resp. $n=2$) we have $H_c^n(f^{-1}(c); {\mathbb C} ) \simeq 0$ 
(resp. $H_c^n(f^{-1}(c); {\mathbb C} )= H_c^{2n-3}(f^{-1}(c); {\mathbb C} ) 
\simeq {\mathbb C}$). For $j \geq n+1$ and $(p,q)=(2,j)$ the spectral sequence 
$E_r^{2,j}$ ($r \geq 1$) degenerates at $E_2$ 
i.e. $E_2^{2,j}=E_{\infty}^{2,j}$ and hence there exists 
an isomorphism 
\begin{equation}
E_2^{2,j}=H^2_c( {\mathbb C} ; H^j {\rm R} f_! {\mathbb C}_U) \simeq H^{j+2}_c(U; {\mathbb C} ). 
\end{equation}
Since for $j \geq n+1$ the algebraic constructible sheaf 
$H^j {\rm R} f_! {\mathbb C}_U$ on ${\mathbb C}$ is constant, we see also that 
the dimension of 
$E_2^{2,j}=H^2_c( {\mathbb C} ; H^j {\rm R} f_! {\mathbb C}_U) \simeq H^{j+2}_c(U; {\mathbb C} )$ 
is equal to that of $H_c^j(f^{-1}(c); {\mathbb C} )$ 
for the generic fiber $f^{-1}(c) 
\subset X \setminus Q^{-1}(0)$ of $f$. For $j \geq n+1$ and the 
generic fiber $f^{-1}(c)$ of $f$ then there exist isomorphisms 
\begin{equation}
H^j_c( f^{-1}(c); {\mathbb C} ) \simeq 
\begin{cases}
\ {\mathbb C} & (j=2n-3, 2n-2) \\
& \\
\ 0 & (j \not= 2n-3,2n-2) 
\end{cases}
\end{equation}
(if $n=3$ the case where $j \geq n+1=4$ and $j=2n-3=3$ does 
not occur). For the generic fiber $f^{-1}(c) 
\subset X \setminus Q^{-1}(0)$ of $f$ we thus obtain 
\begin{equation}
H^j_c(f^{-1}(c); {\mathbb C} ) \simeq 0 \quad (j \not= n-1, 2n-3, 2n-2) 
\end{equation}
and $H^{2n-3}_c(f^{-1}(c); {\mathbb C} ) \simeq H^{2n-2}_c(f^{-1}(c); {\mathbb C} ) \simeq {\mathbb C}$. 
This completes the proof. 
\hfill $\Box$

\label{conj-3}
In the situation of Theorem \ref{MainT-2}, 
any generic fiber $f^{-1}(c) \subset U=X \setminus Q^{-1}(0)$ 
($c \in {\mathbb C} \setminus \Sigma$) of $f: U=X \setminus Q^{-1}(0) \longrightarrow {\mathbb C}$ 
has the homotopy type of 
\begin{equation}
S^1 \vee (S^{n-1} \vee \cdots \vee S^{n-1}), 
\end{equation}
where $S^{n-1} \vee \cdots \vee S^{n-1}$ is 
the bouquet of the $\mu_f-( \mu - \mu_0)$ copies of the 
$(n-1)$-sphere $S^{n-1}$.
