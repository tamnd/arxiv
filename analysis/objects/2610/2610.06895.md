<!-- definition-1: body_md -->
Operator \eqref{eq:2} is called a Volterra quadratic stochastic operator of a two-sex population (VQSOBP) if
\begin{equation}
\label{eq:3}
\begin{cases}
 p_{ik,j}^{(f)}=0, & \text{if }j\notin\{i,k\},\\
 p_{ik,l}^{(m)}=0, & \text{if }l\notin\{i,k\}.
\end{cases}
\end{equation}
<!-- end definition-1 -->
<!-- definition-2: body_md -->
A continuous function $\Phi$, defined on the positively invariant subset of $\Omega$, under consideration, is called a Lyapunov function for system \eqref{eq:5} if, along every trajectory $(x(t),y(t))$ in this subset, the finite limit
\[
\lim_{t\to+\infty}\Phi(x(t),y(t))
\]
exists.
<!-- end definition-2 -->
<!-- lemma-1: body_md -->
Let $n<\nu$. Then
\[
\Phi(x,y)=\prod_{j=n+1}^{\nu}y_j
\]
is a Lyapunov function for system \eqref{eq:5}.

If $b_j\ge0$ and $\sum_{j=n+1}^{\nu}b_j=1$, then
\[
\Phi_b(x,y)=\prod_{j=n+1}^{\nu}y_j^{b_j}
\]
is also a Lyapunov function; factors with $b_j=0$ are not taken into account. If
\[
\mu_b=\min_{1\le k\le n}\sum_{j=n+1}^{\nu}b_jq_{kj}>0,
\]
then
\begin{equation}
\label{eq:6}
\Phi_b(x,y)\le \Phi_b(x^0,y^0)e^{-\mu_b t}.
\end{equation}
If
\[
\mu_{0j}=\min_{1\le k\le n}q_{kj}>0,\qquad j=n+1,\ldots,\nu,
\]
then
\[
y_j\le y_j^0e^{-\mu_{0j}t}
\]
and
\[
\omega(x^0,y^0)\subset S^{n-1}\times\Upsilon,
\qquad
\Upsilon=\{y\in S^{\nu-1}:y_{n+1}=\cdots=y_{\nu}=0\}.
\]
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
For $j>n$,
\[
\dot y_j=-y_j\sum_{k=1}^n q_{kj}x_k\le0.
\]
Hence $y_j$, and with it $\Phi(x,y)$, is non-increasing and bounded below by zero.

If $y_j>0$ for every $j$ with $b_j>0$, then
\[
\frac{\dot\Phi_b}{\Phi_b}
=-\sum_{k=1}^n x_k\sum_{j=n+1}^{\nu}b_jq_{kj}\le-\mu_b.
\]
If one of the factors involved is initially zero, it remains zero and $\Phi_b\equiv0$. Inequality \eqref{eq:6} follows from Gr\"onwall's lemma. The coordinatewise estimates are obtained analogously and give the stated inclusion for the $\omega$-limit set.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
Let $n=\nu$, $1\le r<n$ and
\[
p_{jk,j}^{(f)}+p_{jk,j}^{(m)}<1,\qquad
p_{kj,j}^{(f)}+p_{kj,j}^{(m)}<1
\]
for $1\le j\le r<k\le n$. Then
\[
\Phi(x,y)=\sum_{j=1}^{r}(x_j+y_j)
\]
is a Lyapunov function. If
\[
\Phi_0=\Phi(x^0,y^0)<2
\]
and
\[
\vartheta=
\min_{\substack{1\le j\le r\\r<k\le n}}
\left\{1-p_{jk,j}^{(f)}-p_{jk,j}^{(m)},\,
1-p_{kj,j}^{(f)}-p_{kj,j}^{(m)}\right\}>0,
\]
then
\[
\Phi(x,y)\le \Phi_0e^{-\vartheta(1-\Phi_0/2)t}.
\]
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Put $X=\sum_{j=1}^r x_j$, $Y=\sum_{j=1}^r y_j$, then
\[
\dot\Phi\le-\vartheta[X(1-Y)+Y(1-X)]
\le-\vartheta\Phi\left(1-\frac{\Phi_0}{2}\right).
\]
Since $\Phi_0<2$ this gives the required exponential estimate.
<!-- end proof-2 -->
<!-- lemma-3: body_md -->
Suppose there exist $l\in\{n+1,\ldots,\nu\}$, $q\in\{1,\ldots,n\}$ such that
\[
\max_{1\le i\le n}p_{il,l}^{(m)}
\le
\min_{1\le k\le\nu}p_{qk,q}^{(f)}.
\]
Then, on the relative interior of $\Omega$,
\[
\Phi(x,y)=\frac{y_l}{x_q}
\]
is a Lyapunov function.

If
\[
\vartheta_1=
\min_{1\le k\le\nu}p_{qk,q}^{(f)}-
\max_{1\le i\le n}p_{il,l}^{(m)}>0,
\]
then
\[
\dot\Phi<0,\qquad \Phi(x,y)\le\Phi_0e^{-\vartheta_1t}.
\]
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
The logarithmic derivative satisfies
\[
\frac{d}{dt}\ln\Phi
\le
\max_{1\le i\le n}p_{il,l}^{(m)}-
\min_{1\le k\le\nu}p_{qk,q}^{(f)}.
\]
The non-strict condition guarantees that $\Phi$, does not increase, while the strict condition gives its exponential decay.
<!-- end proof-3 -->
<!-- lemma-4: body_md -->
Let $a_qc_q<b_qd_q$. If $n<\nu$ assume in addition $\varepsilon_q>0$. Then
\[
\Phi_q(x,y)=(1-x_q)+\varrho_q(1-y_q)
\]
is a Lyapunov function relative to $E_q$. The equilibrium $E_q$ is locally exponentially asymptotically stable, and
\[
\mathcal{A}(E_q)=\Omega\setminus\Upsilon_q.
\]
Since $\operatorname{codim}_{\Omega}\Upsilon_q=2$, the equilibrium $E_q$ is almost globally asymptotically stable: its basin of attraction $\mathcal{A}(E_q)$ has full measure in $\Omega$ with respect to Lebesgue measure on the affine hull of $\Omega$.
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Put
\[
u=1-x_q,\qquad
v=\sum_{\substack{l=1\\l\ne q}}^n y_l,\qquad
w=\sum_{l=n+1}^{\nu}y_l,
\]
where, when $n=\nu$, $w=0$. Then $\Phi_q=0$ only at $E_q$.

From system \eqref{eq:5}, using the stochastic and Volterra conditions, we obtain
\[
\dot u\le a_qx_qv-b_qy_qu.
\]
For $n<\nu$,
\[
\dot w\le-\varepsilon_qx_qw-T_q,
\qquad
\dot v\le c_qy_qu-d_qx_qv+T_q,
\]
where
\[
T_q=\sum_{\substack{i=1\\i\ne q}}^n\sum_{l=n+1}^{\nu}
\left(1-p_{il,l}^{(m)}\right)x_i y_l.
\]
For $n=\nu$ the term $T_q$ is absent from the estimate for $\dot v$.

Since
\[
b_q-\varrho_qc_q=\varrho_qd_q-a_q=\gamma_q,
\]
we have
\[
\dot\Phi_q\le-\gamma_qy_qu-\gamma_qx_qv-\varrho_q\varepsilon_qx_qw\le0,
\]
where, for $n=\nu$, the last term is absent.

Choose a neighborhood of $E_q$, in which $x_q\ge\delta$ and $y_q\ge\delta$ for some $\delta>0$, and then a sufficiently small $\varepsilon>0$ so that the set $\{\Phi_q\le\varepsilon\}$ is contained in this neighborhood. This set is positively invariant, and on it
\[
\dot\Phi_q\le-\gamma_0\Phi_q,
\]
where
\[
\gamma_0=\delta\min\left\{\gamma_q,\frac{\gamma_q}{\varrho_q},\varepsilon_q\right\}\quad(n<\nu),
\]
and at $n=\nu$,
\[
\gamma_0=\delta\min\left\{\gamma_q,\frac{\gamma_q}{\varrho_q}\right\}.
\]
Since $\Phi_q$ is equivalent to the distance to $E_q$ on $\Omega$, this estimate proves local exponential stability.

From the global estimate for $\dot\Phi_q$ it follows that $\dot\Phi_q<0$ outside $\{E_q\}\cup\Upsilon_q$. On $\Upsilon_q$,
\[
\Phi_q=1+\varrho_q,
\]
and $\Upsilon_q$ is positively invariant.

If $(x^0,y^0)\notin\Upsilon_q$, then
\[
\Phi_q(x^0,y^0)<1+\varrho_q,
\]
so the monotonicity of $\Phi_q$ rules out any $\omega$-limit set with $\Upsilon_q$. By LaSalle's invariance principle,
\[
(x,y)\to E_q\qquad\text{as }t\to+\infty.
\]
Hence $\mathcal{A}(E_q)=\Omega\setminus\Upsilon_q$.
<!-- end proof-4 -->
<!-- lemma-5: body_md -->
The equilibrium $E_q$ is locally exponentially asymptotically stable relative to $\Omega$ if and only if
\[
\Delta_{qi}>0\qquad\text{for all }i\in\{1,\ldots,n\}\setminus\{q\},
\]
and, if $n<\nu$, $\varepsilon_q>0$.

If $\Delta_{qi}<0$ for at least one $i\ne q$, then $E_q$ is unstable. If $\Delta_{qi}=0$ for at least one $i\ne q$ or if $n<\nu$, $\varepsilon_q=0$, then $E_q$ is non-isolated and is not asymptotically stable.
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
After eliminating the dependent coordinates $x_q,y_q$ the reduced Jacobian of system \eqref{eq:5} has two-dimensional blocks
\[
J_{qi}^{(5)}=
\begin{pmatrix}
-A_{qi}&B_{qi}\\
C_{qi}&-D_{qi}
\end{pmatrix},
\]
where, for $i\ne q$,
\[
A_{qi}=p_{iq,q}^{(f)},\qquad
B_{qi}=1-p_{qi,q}^{(f)},\qquad
C_{qi}=1-p_{iq,q}^{(m)},\qquad
D_{qi}=p_{qi,q}^{(m)},
\]
with $0\le A_{qi},B_{qi},C_{qi},D_{qi}\le1$. For $n<\nu$ there are additional one-dimensional blocks
\[
-p_{ql,q}^{(m)},\qquad l>n.
\]
The block structure of the reduced Jacobian together with the Hurwitz criterion ensures local exponential asymptotic stability whenever the conditions of the first part of the lemma hold. Consider the Cauchy problem for the continuous analogue of $W_9$ \cite{ref10}:

\begin{cases}
\dot x_1=y_1-x_1,\\
\dot x_2=y_2-x_2,\\
\dot y_1=x_1y_2,\\
\dot y_2=x_2y_2-y_2,
\end{cases}
\qquad
(x_1(0),x_2(0),y_1(0),y_2(0))=(x_1^0,x_2^0,y_1^0,y_2^0)\in\Omega_2,
\]
for which no solution in quadratures was obtained in \cite{ref26}.

Eliminating $x_2=1-x_1$ and $y_2=1-y_1$, we obtain
\begin{equation}
\label{eq:20}
\begin{cases}
\dot x_1=y_1-x_1,\\
\dot y_1=x_1(1-y_1),
\end{cases}
\qquad
(x_1(0),y_1(0))=(x_1^0,y_1^0).
\end{equation}
Its equilibria are $(0,0)$ -- a hyperbolic saddle, and $(1,1)$ -- an asymptotically stable improper node.

Set $u=x_1-1$, $v=y_1-1$. Then
\begin{equation}
\label{eq:21}
\begin{cases}
\dot u=v-u,\\
\dot v=-uv-v,
\end{cases}
\end{equation}
and, since $v=\dot u+u$,
\begin{equation}
\label{eq:22}
\ddot u+2\dot u+u\dot u+u+u^2=0,
\end{equation}
\begin{equation}
\label{eq:23}
u(0)=u_0,\qquad \dot u(0)=v_0-u_0.
\end{equation}
The stationary solutions are $u\equiv0$ and $u\equiv-1$. For $v_0=0$ the trajectory lies on the invariant line $v=0$, and
\[
u(t)=u_0e^{-t}.
\]

Consider now the more general problem
\begin{equation}
\label{eq:24}
\ddot u+2\dot u+u=f(u,\dot u),
\end{equation}
\begin{equation}
\label{eq:25}
u(0)=u_0,\qquad \dot u(0)=v_0-u_0,
\end{equation}
where it is assumed that for $|U|<r_1$, $|V|<r_2$ the function $f$ admits the absolutely convergent power series representation
\[
f(U,V)=\sum_{\mathfrak n,\mathfrak m\ge0}a_{\mathfrak n,\mathfrak m}U^{\mathfrak n}V^{\mathfrak m},
\qquad a_{\mathfrak n,\mathfrak m}\in\mathbb{R},
\]
where $0<r_1,r_2\le+\infty$.

Set
\[
\mathcal{N}_1[u]=\ddot u+2\dot u+u-f(u,\dot u),
\qquad
\mathcal{L}_1[\varphi]=\ddot\varphi+2\dot\varphi+\varphi.
\]
The solution of the problem $\mathcal{L}_1[\varphi]=g$ with initial conditions \eqref{eq:25} has the form
\[
\varphi(t)=(u_0+v_0t)e^{-t}+\int_0^t(t-\tau)e^{-(t-\tau)}g(\tau)\,d\tau.
\]
For zero initial data and a bounded continuous function $g$, the estimates
\begin{equation}
\label{eq:26}
\|\varphi\|_\infty\le\|g\|_\infty,
\qquad
\|\dot\varphi\|_\infty\le2\|g\|_\infty
\end{equation}
hold.

For $\hbar\ne0$ introduce the deformation equation
\[
(1-p)\mathcal{L}_1[U]=\hbar p\mathcal{N}_1[U],
\]
\[
U(0,p,\hbar)=u_0,\qquad \dot U(0,p,\hbar)=v_0-u_0.
\]
We seek the solution as a formal series
\begin{equation}
\label{eq:27}
U(t,p,\hbar)=\sum_{k=0}^{\infty}u^{[k]}(t,\hbar)p^k,
\qquad
u^{[k]}=\left.\frac{1}{k!}\frac{\partial^kU}{\partial p^k}\right|_{p=0}.
\end{equation}
Here
\[
u^{[0]}(t)=(u_0+v_0t)e^{-t},
\qquad
u^{[k]}(0,\hbar)=\dot u^{[k]}(0,\hbar)=0,\quad k\ge1.
\]
Let
\[
f(U,\dot U)=\sum_{k=0}^{\infty}F^{[k]}(t,\hbar)p^k.
\]

\[
u^{[k]}(t,\hbar)=\int_0^t(t-\tau)e^{-(t-\tau)}R_k(\tau)\,d\tau,
\qquad k\ge1,
\]
where
\[
R_1=-\hbar F^{[0]},\qquad
R_k=(1+\hbar)\mathcal{L}_1[u^{[k-1]}]-\hbar F^{[k-1]},\quad k\ge2.
\]
For $t\ge0$ we have
\begin{equation}
\label{eq:28}
|u^{[0]}(t)|\le |u_0|+\frac{|v_0|}{e}=:U_0,
\qquad
|\dot u^{[0]}(t)|\le |v_0-u_0|+\frac{|v_0|}{e}=:V_0.
\end{equation}
It is further assumed that $U_0<r_1$, $V_0<r_2$. For fixed $\hbar$ set
\[
M_k=\|u^{[k]}\|_\infty,
\qquad
N_k=\|\dot u^{[k]}\|_\infty,
\qquad
\rho_k=\|R_k\|_\infty,
\qquad k\ge1.
\]

From \eqref{eq:26} it follows that
\begin{equation}
\label{eq:29}
M_k\le\rho_k,\qquad N_k\le2\rho_k,\qquad k\ge1.
\end{equation}
Set
\[
U=u^{[0]}+\xi_1,\qquad \dot U=\dot u^{[0]}+\eta_1,
\]
where
\[
\xi_1=\sum_{k=1}^{\infty}u^{[k]}(t,\hbar)p^k,
\qquad
\eta_1=\sum_{k=1}^{\infty}\dot u^{[k]}(t,\hbar)p^k.
\]
Define
\[
\widetilde f(U,V)=\sum_{\mathfrak n,\mathfrak m\ge0}
|a_{\mathfrak n,\mathfrak m}|U^{\mathfrak n}V^{\mathfrak m},
\qquad
\Psi(z)=\sum_{k=1}^{\infty}\rho_k z^k.
\]
Then
\[
\widehat F(z)=\sum_{k=0}^{\infty}\widehat F^{[k]}z^k
\preceq
\widetilde f(U_0+\Psi(z),V_0+2\Psi(z)),
\]
where
\[
\widehat F^{[k]}=\sup_{t\ge0}|F^{[k]}(t,\hbar)|.
\]
Let $-2<\hbar<0$,
\[
\kappa=\frac{|\hbar|}{1-\upsilon},
\]
and
\[
\widetilde T(V)=\kappa\widetilde f(U_0+V,V_0+2V),
\qquad
V_{\max}=\min\left\{r_1-U_0,\frac{r_2-V_0}{2}\right\}.
\]

\begin{lemma}
If there exists $V^*\in(0,V_{\max})$, for which
\[
\widetilde T(V^*)\le V^*,
\]
then
\[
\sum_{k=1}^{\infty}\rho_k\le V^*.
\]
\end{lemma}

\begin{proof}
Set
\[
S_m=\sum_{k=1}^{m}\rho_k,\qquad S_0=0.
\]
From the recurrence estimates we have
\[
S_m\le\upsilon S_{m-1}+|\hbar|\sum_{k=0}^{m-1}\widehat F^{[k]}.
\]
The coefficientwise majorant for $\widehat F$ and the non-negativity of the coefficients of $\widetilde f$ give, for $S_{m-1}\le V^*$,
\[
\sum_{k=0}^{m-1}\widehat F^{[k]}
\le\widetilde f(U_0+V^*,V_0+2V^*).
\]
Hence $S_m\le V^*$. By induction, $S_m\le V^*$ for all $m\ge1$. Passing to the limit as $m\to\infty$, gives the required inequality.
\end{proof}

For $A\in(0,V_{\max})$ denote
\[
\zeta_A=\kappa\left[
\widetilde f_U(U_0+A,V_0+2A)+
2\widetilde f_V(U_0+A,V_0+2A)
\right].
\]

\begin{theorem}
Let $-2<\hbar<0$ and suppose there exists $A\in(0,V_{\max})$, for which
\begin{equation}
\label{eq:30}
\zeta_A<1,
\end{equation}

\label{eq:31}

\end{equation}

\end{theorem}

\begin{proof}
From conditions \eqref{eq:30}, \eqref{eq:31} we obtain $\widetilde T(A)\le A$. By Lemma~11,
\[

\]
From \eqref{eq:29} and
\[
\ddot u^{[k]}=R_k-2\dot u^{[k]}-u^{[k]},\qquad k\ge1,

we get

\le\widetilde f(U_0+A,V_0+2A)<\infty,
\qquad
\sum_{k=0}^{\infty}F^{[k]}=f(u,\dot u).

Passing to the limit in the recurrence equations gives
\[

\]
whence $\mathcal{L}_1[u]=f(u,\dot u)$ for $\hbar\ne0$. The initial conditions hold by construction, and uniqueness follows from the analyticity of $f$.
\end{proof}

Taking into account the simplex conditions, we get
\[
x_2(t)-y_2(t)=c_4e^{-t},
\qquad c_4=x_2^0-y_2^0.

\mathcal{L}_2[\varphi]=\dot\varphi+(1-c_4e^{-t})\varphi,
\qquad

\]

\[
\mathcal{L}_2[Y_2]=pY_2^2,
\qquad Y_2(0,p)=y_2^0.
\]
\[
Y_2(t,p)=\sum_{k=0}^{\infty}y_2^{[k]}(t)p^k,
\qquad y_2^{[0]}(t)=y_2^0E_2(t),
\]
and the recurrence equations have the form
\[
\mathcal{L}_2[y_2^{[k]}]
=\sum_{j=0}^{k-1}y_2^{[j]}y_2^{[k-1-j]},
\qquad y_2^{[k]}(0)=0,\quad k\ge1.
\]

\[
I_3(t)=\int_0^tE_2(s)\,ds
=\begin{cases}
\displaystyle\frac{\exp[c_4(1-e^{-t})]-1}{c_4},&c_4\ne0,\\[2ex]
1-e^{-t},&c_4=0.
\end{cases}

\[
\mathcal{L}_2[E_2(t)I_3^k(t)]=kE_2^2(t)I_3^{k-1}(t),\qquad k\ge1,

\[
y_2^{[k]}(t)=(y_2^0)^{k+1}E_2(t)I_3^k(t),
\qquad k=0,1,\ldots.
\]

E_2(t)\le e^{-[1-\max\{c_4,0\}]t},
