\label{thm:intro-wz}
The functions \(F(n,k)\) and \(G(n,k)\) form a meromorphic WZ pair:
\[
F(n+1,k)-F(n,k)
=
G(n,k+1)-G(n,k).
\]
Moreover, for every \(x>0\),
\[
\frac{4^x\sqrt{\pi}\,\Gamma(x)\Gamma(x+1)}
     {\Gamma(2x+1/2)}
=
\sum_{n=0}^{\infty}
\frac{6n+4x+1}{(2n+1)(n+x)}
\frac{(1/2)_n(1/2+2x)_n}
     {(1)_n(1+x)_n}
4^{-n}.
\]
Taking the limit \(x\to0^+\) gives
\[
3\ln 2
=
2+\sum_{n=1}^{\infty}
\frac{6n+1}{2n(2n+1)}
\binom{2n}{n}^{2}\frac{1}{2^{6n}}.
\]

\]
and relabeling $k$ as $x$ proves the theorem.

G\left(n+\frac12,k+y\right)-\sum_{n=0}^{\infty}
G\left(n+\frac12,k+y+1\right).

  \sum_{k=0}^{\infty}
F\left(\frac12,k+y\right)

\sum_{n=0}^{\infty}
G\left(n+\frac12,y\right)-\lim_{k\to\infty}\sum_{n=0}^{\infty}
G\left(n+\frac12,k+y\right).
\end{equation}

\[

\frac{2}{\pi^2(2k+1)}\,4^{-n}
\frac{(1)_n(1+2k)_n}

\frac{(1+2k)_n}{(k+3/2)_n}=\prod_{j=0}^{n-1}\frac{2k+1+j}{k+3/2+j} \le 2^n\quad \mathrm{and}\quad \frac{(1)_n}{(3/2)_n}\le 1,

\frac{2}{\pi^2(2k+1)}2^{-n}.

For the lower truncation, using
\[
a_n=\frac{4n+1}{2n(2n+1)}
\binom{2n}{n}^{2}\frac{1}{2^{4n}}=
\frac{1}{2}
\frac{\binom{2n}{n}^{2}}{n16^n}
+
\frac{\binom{2n}{n}^{2}}{(2n+1)16^n},
\]
we obtain
\[
\sum_{n=1}^{(p-3)/2}a_n
=
\frac12
\sum_{n=1}^{(p-3)/2}
\frac{\binom{2n}{n}^{2}}{n16^n}
+
\sum_{n=1}^{(p-3)/2}
\frac{\binom{2n}{n}^{2}}{(2n+1)16^n}.
\]
By \eqref{eq:sun1},
\[
\sum_{n=1}^{(p-3)/2}
\frac{\binom{2n}{n}^{2}}{n16^n}
\equiv
-2H_{(p-1)/2}
-
\frac{\binom{p-1}{(p-1)/2}^{2}}{4^{p-1}(p-1)/2}
\pmod p.
\]
Since $\binom{p-1}{(p-1)/2}
\equiv(-1)^{(p-1)/2}\pmod p,
4^{(p-1)}\equiv1\pmod p$ and $\frac{1}{(p-1)/2}\equiv-2\pmod p$,
\begin{equation}\label{eq:2log2-lower-left}
\frac12
\sum_{n=1}^{(p-3)/2}
\frac{\binom{2n}{n}^{2}}{n16^n}
\equiv
-H_{(p-1)/2}+1
\pmod p.  
\end{equation}
On the other hand, \eqref{eq:sun3} gives
\begin{equation}\label{eq:2log2-lower-right}
 \sum_{n=1}^{(p-3)/2}
\frac{\binom{2n}{n}^{2}}{(2n+1)16^n}
\equiv
-2q_p(2)-1
\pmod p, 
\end{equation}
where the term \(n=0\), which is equal to \(1\), has been removed. \cite[Formula (45)]{Leh38} implies,
\[
H_{(p-1)/2}=\sum_{k=1}^{(p-1)/2}\frac1k\equiv-2q_p(2)\pmod p,
\]
Combining \eqref{eq:2log2-lower-left} and \eqref{eq:2log2-lower-right}, we conclude that
\[
\sum_{n=1}^{(p-3)/2}\frac{4n+1}{2n(2n+1)}
\binom{2n}{n}^{2}\frac{1}{2^{4n}}
\equiv
-H_{(p-1)/2}-2q_p(2)
\equiv0\pmod p.
\]
This proves \eqref{eq:2log2-lower}.

For the upper truncation, \eqref{eq:sun2} and
\eqref{eq:sun4} give
\begin{align*}
\sum_{n=(p+1)/2}^{p-1}\frac{4n+1}{2n(2n+1)}
\binom{2n}{n}^{2}\frac{1}{2^{4n}}
&=
\frac12
\sum_{n=(p+1)/2}^{p-1}
\frac{\binom{2n}{n}^{2}}{n16^n}
+
\sum_{n=(p+1)/2}^{p-1}
\frac{\binom{2n}{n}^{2}}{(2n+1)16^n}\\
&\equiv
\frac12\left(\frac72p^2B_{p-3}\right)
-\frac74p^2B_{p-3}\\
&=0
\pmod{p^3}.
\end{align*}
This proves \eqref{eq:2log2-upper}.

The cases $p=3$ and $p=5$ follow by direct computation.

\begin{equation}\label{eq:N-wz-sum}
F(N+1,x)=
\sum_{n=0}^{N}G(n,x+1)
-\sum_{n=0}^{N}G(n,x).
\end{equation}
By the definition of $H(n,x)$, we have
\[
G(n,x)=\frac{x}{2}
\frac{(1/4)_x(3/4)_x}{(1)_x^2}H(n,x).
\]
Moreover,
\[
\frac{
(1/4)_{x+1}(3/4)_{x+1}/(1)_{x+1}^2
}{(1/4)_x(3/4)_x/(1)_x^2}=
\frac{(4x+1)(4x+3)}{16(x+1)^2}.
\]
Substituting these expressions into
\eqref{eq:N-wz-sum} and cancelling the common
factor yields
\begin{equation}
\frac{(4x+1)(4x+3)}{16(x+1)}
\sum_{n=0}^{N}H(n,x+1)
-x\sum_{n=0}^{N}H(n,x)=-\frac{N+1}{N+1+x}4^{-(N+1)}
\frac{(1/2)_{N+1}(1/2+2x)_{N+1}}
{(1)_{N+1}(1+x)_{N+1}}.
\label{eq:N-wz-identity}
\end{equation}
This identity is initially obtained for generic $x$ and
extends as a rational identity wherever the displayed
expressions are defined.

Now set
\[
x_N=-\frac{2N+1}{4}.
\]
Since$\left(\frac12+2x_N\right)_{N+1}=(-N)_{N+1}=0$,
the right-hand side of \eqref{eq:N-wz-identity}
vanishes.

Now we prove $\displaystyle\sum_{n=0}^N H(n,x_N)=0$ by induction. For $N=0,1$, the coefficient
$(4x_N+1)(4x_N+3)$ also vanishes. Since $x_N\neq0$,
we immediately obtain
\[
\sum_{n=0}^{N}H(n,x_N)=0\quad(N=0,1).
\]
For $N\geq2$, assuming that the identity holds for $N-2$, observe that
\[
x_N+1=x_{N-2},
\qquad
\frac12+2(x_N+1)=-(N-2).

Since $x_N\neq0$, the desired identity follows.

Applying Lemma~\ref{lem:terminating-identity} with $N=\frac{p-1}{2}$, we obtain
\begin{equation}\label{eq:lower-specialization}
\sum_{n=0}^{\frac{p-1}{2}}4^{-n}\frac{(1/2)_n(-(p-1)/{2})_n}{(1)_n((4-p)/4)_n}
\left(
\frac{1}{n-(p/4)}
+\frac{4}{2n+1}
\right)=0.
\end{equation}
For convenience, denote the $n$th summand by
\[
T_n=
4^{-n}\frac{(1/2)_n(-(p-1)/{2})_n}{(1)_n((4-p)/4)_n}
\left(
\frac{1}{n-(p/4)}
+\frac{4}{2n+1}
\right).
\]

First, when $n=0$,
\begin{equation}\label{eq:lower-first-term}
T_0=-\frac4p+4.
\end{equation}

Next, for $1\leq n\leq \frac{p-3}{2}$, since$-(p-1)/2\equiv1/2\pmod p$ and $(4-p)/4\equiv1\pmod p$, we have
\[
4^{-n}\frac{(1/2)_n(-(p-1)/{2})_n}{(1)_n((4-p)/4)_n}
\equiv
4^{-n}\frac{(1/2)_n^2}{((1)_n)^2}
=
\binom{2n}{n}^{2}\frac1{2^{6n}}
\pmod p.
\]
Furthermore,
\[
\frac{1}{n-\frac p4}+\frac4{2n+1}
\equiv
\frac{6n+1}{n(2n+1)}
\pmod p.

\sum_{n=1}^{\frac{p-3}{2}}T_n
\equiv
2\sum_{n=1}^{\frac{p-3}{2}}
\frac{6n+1}{2n(2n+1)}
\binom{2n}{n}^{2}\frac1{2^{6n}}=2\sum_{n=1}^{\frac{p-3}{2}}b_n
\pmod p.
\end{equation}

\[
T_{\frac{p-1}{2}}
=4^{-\frac{p-1}{2}}
\frac{8(p-1)}{p(p-2)}
\frac{(-1)^{\frac{p-1}{2}}
(1/2)_{\frac{p-1}{2}}}
{((4-p)/4)_{\frac{p-1}{2}}}.

Observe that
\[
4^{-\frac{p-1}{2}}
\frac{(-1)^{\frac{p-1}{2}}
(1/2)_{\frac{p-1}{2}}}
{((4-p)/4)_{\frac{p-1}{2}}}
=(-1)^{\frac{p-1}{2}}\binom{p-1}{(p-1)/2}
\frac{\prod_{j=1}^{(p-1)/2}
\left(1-\frac{p}{4j}\right)^{-1}}{4^{p-1}}.

\[
(-1)^{\frac{p-1}{2}}
\binom{p-1}{(p-1)/2}
\equiv 4^{p-1}\pmod{p^3}.

\[
\prod_{j=1}^{(p-1)/2}
\left(1-\frac{p}{4j}\right)^{-1}
\equiv
1+\frac{p}{4}
\sum_{j=1}^{(p-1)/2}\frac1j
\pmod{p^2}.
\]
\cite[Formula (45)]{Leh38} implies
\[
\sum_{j=1}^{(p-1)/2}\frac1j
\equiv -2q_p(2)\pmod p.

\begin{align*}
\prod_{j=1}^{(p-1)/2}

\equiv
1-\frac p2q_p(2)
\pmod{p^2}.
\end{align*}

\end{equation}

\frac{6n+1}{2n(2n+1)}
\binom{2n}{n}^{2}\frac{1}{2^{6n}}
\equiv q_p(2)-1\pmod p.
\end{align*}
This finishes the proof. \hfill$\square$

\begin{align*}
 &U(n,k)=4^{-n}\frac{(1/2)_n(1/2+2k)_n}{(1)_n(1+k)_n}
                  \frac{(1/4)_k(3/4)_k}{(1)_k^2}\\
 F(n,k)&=-\frac{n}{2(n+k)}U(n,k),\qquad

\begin{equation}\label{eq:F(N+1/2,k)}

\end{aligned}

\binom N\ell\binom{2N-2\ell}{N}}
{2\ell\binom{2N}{N}}\sum_{j=N-\ell}^{N-1}\frac{1}{(2j+1)^2}.
\]
This finishes the proof.

\label{lem:pfaff}
For every integer $N\geq 0$,
\[
\sum_{\ell=1}^{N}
\frac{(-1)^\ell}{\ell}
\binom{N}{\ell}\binom{N+\ell}{\ell}
\sum_{j=1}^{\ell}\frac1{j^2}
=
-2\sum_{j=1}^{N}\frac1{j^3}.
\]

Recall that the generalized hypergeometric function is defined by
\[
{}_pF_q\left(
\begin{matrix}
a_1,\ldots,a_p\\
b_1,\ldots,b_q
\end{matrix};z
\right)
=
\sum_{m=0}^{\infty}
\frac{(a_1)_m\cdots(a_p)_m}
     {(b_1)_m\cdots(b_q)_m}
\frac{z^m}{m!}.
\]
In the present case, one of the upper parameters is $-N$,
so the series terminates at $m=N$.

By the terminating Pfaff--Saalsch\"utz formula
\cite[Formula 16.4.3]{DLMF},
\begin{equation}\label{eq:pfaff-saalschutz}
    \begin{aligned}
      {}_3F_2\left(
\begin{matrix}
-N,\ N+1,\ u\\
1+z,\ 1+u-z
\end{matrix};1
\right):&=\sum_{\ell=0}^{N} \frac{(-N)_{\ell}(N+1)_{\ell}(u)_{\ell}}{(1+z)_{\ell}(1+u-z)_{\ell}\,\ell!}\\
&=\frac{(1-z)_N(1+z-u)_N}
     {(1+z)_N(1+u-z)_N}.  
    \end{aligned}
\end{equation}

Since the series on the left-hand side is finite, we may
differentiate term by term with respect to $u$:
\[
\left.
\frac{d}{du}
\frac{(u)_\ell}{(1+u-z)_\ell}
\right|_{u=0}
=
\frac{\frac{d}{du}|_{u=0}(u)_\ell}{(1+u-z)_\ell}.
\]
For $\ell\geq1$,
\[
\left.\frac{d}{du}(u)_\ell\right|_{u=0}
=(\ell-1)!,
\]
whereas $(u)_\ell|_{u=0}=0$. Thus the derivative of the left-hand side of
\eqref{eq:pfaff-saalschutz} at $u=0$ is
\[
\sum_{\ell=1}^{N}
\frac{(-N)_\ell(N+1)_\ell}
     {(1+z)_\ell(1-z)_\ell}
\frac{(\ell-1)!}{\ell!}=\sum_{\ell=1}^{N}\frac{(-N)_\ell(N+1)_\ell}
     {(1+z)_\ell(1-z)_\ell}\frac{1}{\ell}.
\]
Using
\[
(-N)_\ell(N+1)_\ell
=
(-1)^\ell(\ell!)^2
\binom{N}{\ell}\binom{N+\ell}{\ell}
\]
and
\[
(1+z)_\ell(1-z)_\ell
=
(\ell!)^2
\prod_{j=1}^{\ell}
\left(1-\frac{z^2}{j^2}\right),
\]
this becomes
\begin{equation}\label{eq:pfaff-left-derivative}
\sum_{\ell=1}^{N}
\frac{(-1)^\ell}{\ell}
\binom{N}{\ell}\binom{N+\ell}{\ell}
\prod_{j=1}^{\ell}
\left(1-\frac{z^2}{j^2}\right)^{-1}.
\end{equation}

We now differentiate the right-hand side of
\eqref{eq:pfaff-saalschutz}. Put
\[
R(u)=
\frac{(1-z)_N(1+z-u)_N}
     {(1+z)_N(1+u-z)_N}.
\]
Since $R(0)=1$, the logarithmic differentiation gives
\[
\left.\frac{d}{du}R(u)\right|_{u=0}
=
-\sum_{j=1}^{N}
\left(
\frac1{j+z}+\frac1{j-z}
\right)
=
-2\sum_{j=1}^{N}\frac{j}{j^2-z^2}.
\]
Consequently,
\begin{equation}\label{eq:pfaff-differentiated}
\sum_{\ell=1}^{N}
\frac{(-1)^\ell}{\ell}
\binom{N}{\ell}\binom{N+\ell}{\ell}
\prod_{j=1}^{\ell}
\left(1-\frac{z^2}{j^2}\right)^{-1}
=
-2\sum_{j=1}^{N}\frac{j}{j^2-z^2}.
\end{equation}

Finally, around $z=0$,
\[
\prod_{j=1}^{\ell}
\left(1-\frac{z^2}{j^2}\right)^{-1}
=
1+
z^2\sum_{j=1}^{\ell}\frac1{j^2}
+O(z^4),
\]
while
\[
\frac{j}{j^2-z^2}
=
\frac1j+\frac{z^2}{j^3}+O(z^4).
\]
Comparing the coefficients of $z^2$ in
\eqref{eq:pfaff-differentiated}, we obtain
\[
\sum_{\ell=1}^{N}
\frac{(-1)^\ell}{\ell}
\binom{N}{\ell}\binom{N+\ell}{\ell}
\sum_{j=1}^{\ell}\frac1{j^2}
=
-2\sum_{j=1}^{N}\frac1{j^3},
\]
as claimed.

We first record the following elementary reflection congruence.
For $1\leq k\leq (p-1)/2$,
\begin{equation}\label{eq:binom-reflection}
\frac1p\binom{2p-2k}{p-k}
\equiv
-\frac{2}{k\binom{2k}{k}}
\pmod p.
\end{equation}
Indeed, there is exactly one factor $p$ in $(2p-2k)!$
and none in $(p-k)!$. Hence
\[
\frac1p\binom{2p-2k}{p-k}
\equiv
-\frac{(p-2k)!}{((p-k)!)^2}
\pmod p.
\]
By Wilson's theorem,
\[
(p-r)!(r-1)!\equiv (-1)^r\pmod p,
\]
and therefore
\[
-\frac{(p-2k)!}{((p-k)!)^2}
\equiv
-\frac{((k-1)!)^2}{(2k-1)!}
=
-\frac{2}{k\binom{2k}{k}}
\pmod p.
\]\\

Applying \eqref{eq:binom-reflection} and making the
substitution $n=p-k$, we obtain
\begin{equation}
 \begin{aligned}
\frac1{p^2}
\sum_{n=\frac{p+1}{2}}^{p-1}
\frac{6n+1}{2n(2n+1)}
\binom{2n}{n}^2\frac1{2^{6n}}
&=\sum_{k=1}^{\frac{p-1}{2}} \frac{6(p-k)+1}{2(p-k)(2p-2k+1)} \left( \frac{1}{p} \binom{2p-2k}{p-k} \right)^2 \frac{1}{2^{6(p-k)}}\\
&\equiv
\sum_{k=1}^{\frac{p-1}{2}}
\frac{2(1-6k)2^{6(k-1)}}
{k^3(2k-1)\binom{2k}{k}^2}
\pmod p.
\label{eq:upper-reflection}
\end{aligned}    
\end{equation}
Here we have also used $64^{p-1}\equiv1\pmod p$. To use Lemma \ref{lem:finite-derived-wz}, we next reindex the sum by setting
\[
k=\frac{p-1}{2}-m,
\qquad
0\leq m\leq \frac{p-3}{2}.
\]
Then
\[
2(1-6k)
\equiv
4(3m+2)
\pmod p\quad\mathrm{and}\quad k^3(2k-1)
\equiv
\frac{(2m+1)^3(m+1)}{4}
\pmod p.
\]
Moreover, Fermat's little theorem gives
\[
2^{6(k-1)}=
2^{3(p-1)-6m-6}
\equiv
2^{-6m-6}
\pmod p.
\]
It remains to transform the central binomial coefficient.
For $0\leq r\leq (p-1)/2$, we have
\[
4^{-r}\binom{2r}{r}
\equiv
(-1)^r\binom{(p-1)/2}{r}
\pmod p.
\]
Applying this with $r=(p-1)/2-m$ and $r=m$, and using
the symmetry of the binomial coefficients, we obtain
\[
\binom{p-1-2m}{(p-1)/2-m}
\equiv
(-1)^{(p-1)/2}
16^{-m}\binom{2m}{m}
\pmod p,
\]
where $4^{(p-1)/2}\equiv1\pmod p$ has been used.
Consequently,
\[
\binom{p-1-2m}{(p-1)/2-m}^{-2}
\equiv
\frac{256^m}{\binom{2m}{m}^2}
\pmod p.
\]
Combining these congruences, we have
\begin{equation}\label{eq:upper-inverse-binomial}
\frac1{p^2}
\sum_{n=\frac{p+1}{2}}^{p-1}
\frac{6n+1}{2n(2n+1)}
\binom{2n}{n}^2\frac1{2^{6n}}
\equiv
\frac14
\sum_{m=0}^{\frac{p-3}{2}}
\frac{(3m+2)4^m}
{(2m+1)^3(m+1)\binom{2m}{m}^2}
\pmod p.
\end{equation}\\

We now apply Lemma \ref{lem:finite-derived-wz} with $N=\frac{p-1}{2}$. Since $\left\lfloor \frac{N}{2}\right\rfloor=\left\lfloor \frac{p}{4}\right\rfloor$,
Lemma \ref{lem:finite-derived-wz} gives
\begin{align}
\sum_{m=0}^{(p-3)/2}
\frac{(3m+2)4^m}
{(2m+1)^3(m+1)\binom{2m}{m}^2}=
2\sum_{j=0}^{(p-3)/2}\frac1{(2j+1)^3}
+
\sum_{\ell=1}^{\lfloor p/4\rfloor}
A_\ell
\sum_{j=(p-1)/2-\ell}^{(p-3)/2}
\frac1{(2j+1)^2},
\label{eq:lemma44-specialized}
\end{align}
where
\[
A_\ell=
\frac{(-1)^{\ell+1}
\binom{(p-1)/2}{\ell}
\binom{p-1-2\ell}{(p-1)/2}}
{2\ell\binom{p-1}{(p-1)/2}}.
\]

Using the standard relations between generalized binomial
coefficients and Pochhammer symbols (see
 \cite[Formula (1.2.6); Formula (5.2.8)]{DLMF}),
we have, for $1\leq\ell\leq\lfloor p/4\rfloor$,
\[
\binom{(p-1)/2}{\ell}
\equiv
(-1)^\ell4^{-\ell}\binom{2\ell}{\ell}
\pmod p,
\]
while
\[
\frac{\binom{p-1-2\ell}{(p-1)/2}}
{\binom{p-1}{(p-1)/2}}
=
\frac{\binom{(p-1)/2}{2\ell}}
{\binom{p-1}{2\ell}}
\equiv
16^{-\ell}\binom{4\ell}{2\ell}
\pmod p.
\]
Consequently,
\begin{align}
A_\ell\equiv
-\frac{1}{2\ell}
\frac{\binom{2\ell}{\ell}\binom{4\ell}{2\ell}}
{64^\ell}
=
-\frac{1}{2\ell}
\frac{(\frac14)_\ell(\frac34)_\ell}{(\ell!)^2} \equiv(-1)^{\ell+1}\frac{1}{2\ell}
\binom{\lfloor p/4\rfloor}{\ell}\binom{\lfloor p/4\rfloor+\ell}{\ell}
\pmod p.
\label{eq:Aell-congruence}
\end{align}
The last equivalence holds since $\{1/4,3/4\}\equiv\{-\lfloor p/4\rfloor, \lfloor p/4\rfloor+1\}\pmod p$.

Moreover, writing
\[
j=\frac{p-1}{2}-r,
\qquad 1\leq r\leq\ell,
\]
we have $2j+1=p-2r\equiv-2r\pmod p$. Hence
\begin{equation}\label{eq:terminal-odd-square}
\sum_{j=(p-1)/2-\ell}^{(p-3)/2}
\frac1{(2j+1)^2}
\equiv
\frac14\sum_{r=1}^{\ell}\frac1{r^2}
\pmod p.
\end{equation}

Substituting
\eqref{eq:Aell-congruence} and
\eqref{eq:terminal-odd-square}
into \eqref{eq:lemma44-specialized}, we obtain
\begin{align*}
\sum_{m=0}^{(p-3)/2}
\frac{(3m+2)4^m}
{(2m+1)^3(m+1)\binom{2m}{m}^2}\equiv
2\sum_{j=0}^{(p-3)/2}\frac1{(2j+1)^3}
-\frac18
\sum_{\ell=1}^{\lfloor p/4\rfloor}
\frac{(-1)^\ell}{\ell}
\binom{\lfloor p/4\rfloor}{\ell}\binom{\lfloor p/4\rfloor+\ell}{\ell}
\sum_{r=1}^{\ell}\frac1{r^2}
\pmod p.
\end{align*}
Applying Lemma \ref{lem:pfaff}, we conclude that
\begin{equation}\label{eq:upper-harmonic-reduction}
\sum_{m=0}^{(p-3)/2}
\frac{(3m+2)4^m}
{(2m+1)^3(m+1)\binom{2m}{m}^2}
\equiv
2\sum_{j=0}^{(p-3)/2}\frac1{(2j+1)^3}
+
\frac14\sum_{j=1}^{\lfloor p/4\rfloor}\frac1{j^3}
\pmod p.
\end{equation}\\

It remains to evaluate the two sums in \eqref{eq:upper-harmonic-reduction}.
Recall that the Bernoulli polynomials $B_n(x)$ are defined by
\[
\frac{te^{xt}}{e^t-1}
=
\sum_{n=0}^{\infty} B_n(x)\frac{t^n}{n!},\quad B_n=B_n(0);
\]
see \cite[Formula 24.2.3]{DLMF}.

We first note that the Bernoulli numbers which occur below are
$p$-integral. Indeed, by the von Staudt--Clausen theorem
(see \cite[Formula 24.10.1]{DLMF}), the denominator of
$B_{2r}$ is the product of the primes $q$ satisfying
$q-1\mid 2r$. Since $p-1\nmid p-3$, the denominator of
$B_{p-3}$ is prime to $p$.

By the special-value formula for Bernoulli polynomials
(see \cite[Formula 24.4.27]{DLMF}),
\[
B_n\left(\frac12\right)
=
\left(2^{1-n}-1\right)B_n.
\]
Taking $n=p-3$ and using Fermat's little theorem, we obtain
\begin{equation}\label{eq:Bernoulli-half}
B_{p-3}\left(\frac12\right)
=
\left(2^{4-p}-1\right)B_{p-3}
\equiv
7B_{p-3}
\pmod p.
\end{equation}
Moreover, by the reflection formula
(see \cite[Formula 24.4.3]{DLMF}),
\[
B_{p-3}\left(\frac14\right)
=
B_{p-3}\left(\frac34\right),
\]
since $p-3$ is even. The multiplication formula
\cite[Formula 24.4.21]{DLMF}, together with the
preceding symmetry, gives
\[
B_{p-3}\left(\frac14\right)
=
B_{p-3}\left(\frac34\right)
=
2^{3-p}B_{p-3}\left(\frac12\right).
\]
Hence, by \eqref{eq:Bernoulli-half} and Fermat's little theorem,
\begin{equation}\label{eq:Bernoulli-quarter}
B_{p-3}\left(\frac14\right)
=
B_{p-3}\left(\frac34\right)
\equiv
28B_{p-3}
\pmod p.
\end{equation}

We now use the power-sum formula
\cite[Formula 24.4.7]{DLMF},
\[
\sum_{j=1}^{M}j^r
=
\frac{B_{r+1}(M+1)-B_{r+1}}{r+1}.
\]
Since $j^{-3}\equiv j^{p-4}\pmod p$ for $1\leq j\leq p-1$, taking $r=p-4$ gives
\begin{equation}\label{eq:power-sum-cubic}
\sum_{j=1}^{M}\frac1{j^3}
\equiv
\frac{B_{p-3}(M+1)-B_{p-3}}{p-3}
\pmod p.
\end{equation}
Taking $M=(p-1)/2$ in \eqref{eq:power-sum-cubic}, and using
$(p+1)/2\equiv 1/2\pmod p$ together with
\eqref{eq:Bernoulli-half}, we obtain
\begin{equation}\label{eq:half-harmonic-cubic}
\begin{aligned}
\sum_{j=1}^{(p-1)/2}\frac1{j^3}\equiv
\frac{B_{p-3}(\frac12)-B_{p-3}}{p-3} \equiv
-2B_{p-3}
\pmod p.
\end{aligned}
\end{equation}
Similarly, let $M=\lfloor p/4\rfloor$, since $\{1/4,3/4\}\equiv\{-\lfloor p/4\rfloor, \lfloor p/4\rfloor+1\}\pmod p$, by \eqref{eq:Bernoulli-quarter} and
\eqref{eq:power-sum-cubic},
\begin{equation}\label{eq:quarter-harmonic-cubic}
\begin{aligned}
\sum_{j=1}^{\lfloor p/4\rfloor}\frac1{j^3}\equiv
\frac{28B_{p-3}-B_{p-3}}{p-3}  \equiv
-9B_{p-3}
\pmod p.
\end{aligned}
\end{equation}

It remains to evaluate the sum over the odd denominators. Observing that $2j+1$ represents the odd numbers from 1 to $p-2$, we can write the odd residues modulo $p$ in the form
$p-2r$, $1\leq r\leq(p-1)/2$. Then
\begin{align}
\sum_{j=0}^{(p-3)/2}\frac1{(2j+1)^3}\equiv
\sum_{r=1}^{(p-1)/2}\frac1{(p-2r)^3}\equiv
-\frac18
\sum_{r=1}^{(p-1)/2}\frac1{r^3}\equiv
\frac14 B_{p-3}
\pmod p,
\label{eq:odd-harmonic-cubic}
\end{align}
where the last congruence follows from
\eqref{eq:half-harmonic-cubic}. Substituting \eqref{eq:quarter-harmonic-cubic} and
\eqref{eq:odd-harmonic-cubic} into \eqref{eq:upper-harmonic-reduction}, we have
\begin{align}
\sum_{m=0}^{(p-3)/2}
\frac{(3m+2)4^m}
{(2m+1)^3(m+1)\binom{2m}{m}^2}\equiv
-\frac74 B_{p-3}
\pmod p.
\label{eq:inverse-binomial-Bernoulli}
\end{align}

Finally, combining \eqref{eq:inverse-binomial-Bernoulli}
with \eqref{eq:upper-inverse-binomial}, 
\[
\frac1{p^2}
\sum_{n=(p+1)/2}^{p-1}
\frac{6n+1}{2n(2n+1)}
\binom{2n}{n}^2\frac1{2^{6n}}
\equiv
-\frac7{16}B_{p-3}
\pmod p.
\]
i.e.,
\[
\sum_{n=(p+1)/2}^{p-1}
\frac{6n+1}{2n(2n+1)}
\binom{2n}{n}^2\frac1{2^{6n}}
\equiv
-\frac7{16}p^2B_{p-3}
\pmod{p^3}.
\]

This proves Formula \eqref{eq:3log2 upper} and completes the proof of Theorem \ref{thm:3log2 congruence}. \hfill$\square$
