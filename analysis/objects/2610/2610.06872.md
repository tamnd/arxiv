<!-- lemma-1: body_md -->
\label{lemma1}
	\textit{Let \( (\varphi_k) \) be an arbitrary ONS. Then}
	\begin{equation}
		\frac{1}{n^2} \sum_{k=1}^{n} \varphi_k^2(x) = O(1)n^{-\frac{1}{2}}, \quad \text{a.e. on } [0,1].
		\label{2.5}
	\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Since \[
	\frac{1}{n^2} \sum_{k=1}^{n} \varphi_k^2(x) \le \frac{1}{\sqrt{n}} \sum_{k=1}^{n} k^{-\frac{3}{2}} \varphi_k^2(x),
	\]
	we can conclude that
	\[
	\frac{1}{\sqrt{n}} \sum_{k=1}^{\infty} k^{-\frac{3}{2}} \int_0^1 \varphi_k^2(x)\,dx = \frac{1}{\sqrt{n}} \sum_{k=1}^{\infty} k^{-\frac{3}{2}} < +\infty,
	\]
	and it follows from Levi's theorem that the convergence holds almost everywhere, which proves that \eqref{2.5} holds.

<!-- lemma-2: body_md -->
\label{Lemma2} 
	For every integer \( i \) with \( 1 \leq i \leq n \) and for all \( x \in [0,1] \), the following inequality holds (see \eqref{2.3}):
	\begin{equation}
		\int_{\frac{i-1}{n}}^{\frac{i}{n}} |Q_n(u, x)|\,du \leq \left( \frac{1}{n^2} \sum_{k=1}^n \varphi_k^2(x) \right)^{\frac{1}{2}}.
		\label{2.6}
	\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
From the Bessel inequality, it follows that
	
	\begin{equation}
		\sum_{k=1}^\infty g_k^2(u) = \sum_{k=1}^\infty \left( \int_0^u \varphi_k(t)\,dt \right)^2 \le \int_0^u dt \le 1.
		\label{2.7}
	\end{equation}
	Here, the function \( g_k \) is defined by
	\[
	g_k(u) = \int_0^u \varphi_k(t)\, dt, \quad u \in [0,1].
	\]
	
	By applying the Cauchy and H\"older inequalities, as well as inequality \eqref{2.7}, and taking into account that \( D_{nk}^{\alpha} < 1 \), we can conclude that, for each \( i = 1, \dots, n \),
	
	\begin{align*}
		\int_{\frac{i-1}{n}}^{\frac{i}{n}} |Q_n(u,x)|\,du
		&\le \frac{1}{\sqrt{n}} \left( \int_{\frac{i-1}{n}}^{\frac{i}{n}} Q_n^2(u,x)\,du \right)^{\frac{1}{2}} \\
		& = \frac{1}{\sqrt{n}} \left( \int_{\frac{i-1}{n}}^{\frac{i}{n}} \left( \sum_{k=1}^n D_{nk}^{\alpha} g_k(u) \varphi_k(x) \right)^2 du \right)^{\frac{1}{2}}\\
		&\le \frac{1}{\sqrt{n}} \left( \int_{\frac{i-1}{n}}^{\frac{i}{n}}  \sum_{k=1}^n (D_{nk}^{\alpha})^2 g_k^2(u)\, du \cdot \sum_{k=1}^n \varphi_k^2(x) \right)^{\frac{1}{2}} \\
		&\le \frac{1}{n} \left( \sum_{k=1}^n \varphi_k^2(x) \right)^{\frac{1}{2}}.
	\end{align*}
	The proof is complete.

<!-- lemma-3: body_md -->
\label{lemma3}
	Let   \( (\varphi_n) \) be an ONS on \([0,1]\), and let \( f \in C_L \). Then
	\begin{equation}
		\sigma_n^{\alpha}(x, f) = f(1)\, \sigma_n^{\alpha}(x, q) - \int_0^1 f'(u)\, Q_n(u,x)\,du,
		\label{2.8}
	\end{equation}
	where \( q(u) = 1 \) for all \( u \in [0,1] \).
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
By integrating by parts, we have that
	\[
	C_k(f) = \int_0^1 f(u)\, \varphi_k(u)\,du
	= f(1) \int_0^1 \varphi_k(u)\,du - \int_0^1 f'(u)\, g_k(u)\,du,
	\]
	that is,
	\begin{equation}
		C_k(f) = f(1) C_k(q) - \int_0^1 f'(u)\, g_k(u)\,du.
		\label{2.9}
	\end{equation}
	Therefore, by \eqref{2.9}
	\begin{align*}
	\sum_{k=0}^{n-1} D_{nk}^{\alpha} C_k(f)\, \varphi_k(x)
& = f(1) \sum_{k=0}^{n-1} D_{nk}^{\alpha} C_k(q)\, \varphi_k(x)

	& \qquad - \int_0^1 f'(u) \sum_{k=0}^{n-1} D_{nk}^{\alpha} g_k(u) \varphi_k(x)\, du.
	\end{align*}
	By also taking equation \eqref{2.3} into consideration, we obtain the desired result \eqref{2.8}. The proof is complete.

First, we suppose that 
	\begin{equation*}
		\limsup_{n \to \infty} \left| \int_0^1 Q_n(u, x_0) \right| = +\infty.
	\end{equation*} 
	Then (see \eqref{3.3} with \(f=q\))
	\begin{equation*}
		\limsup_{n \to \infty} \left| B_n(q, x_0) \right| = \limsup_{n \to \infty} \left| \int_0^1 Q_n(u, x_0) \right|=+\infty.
	\end{equation*}
	and since \(q\in Lip_1\), \eqref{eq:3.3} holds for this case.
	
	Next, we assume that
	\begin{equation}
		\label{eq3.5} \left| B_n (q, x_0) \right| = O(1).
	\end{equation}
	
	On the space \( \mathrm{Lip}_1 \), we consider the sequence of functions \( (r_n) \), defined as follows:
	
	\begin{equation}\label{eq3.6}
		r_n(u) = \int_0^u \operatorname{sign} \left( \int_0^y Q_n(v, x_0) \, dv \right) dy, \quad n = 1, 2, \ldots
	\end{equation}
	
	The following equality is valid (see \eqref{eq:3.2}, with \(f=r_n\))
	\begin{align}\label{eq3.7}
		\int_0^1 r_n(u) Q_n(u, x_0) \, du 
		&= \sum_{i=1}^{n-1} \left( r_n\left(\frac{i}{n}\right) - r_n\left(\frac{i+1}{n}\right) \right) \int_0^{\frac{i}{n}} Q_n(u, x_0) \, du \notag \\
		&\quad + \sum_{i=1}^{n-1} \int_{\frac{i-1}{n}}^{\frac{i}{n}} \left( r_n(u) - r_n\left(\frac{i}{n}\right) \right) Q_n(u, x_0) \, du  \\
		&\quad + r_n(1) \int_0^1 Q_n(u, x_0) du := A_1 + A_2 + A_3. \notag
	\end{align}
	
	Next, we proceed to by estimating the quantities \( |A_1| \), \( |A_2| \), and \( |A_3| \).
	
	By equation \eqref{2.6} (see  Lemma \ref{Lemma2}) we have that
	\begin{align}\label{al1}
		|A_2| &\leq \frac{1}{n} \sum_{i=1}^{n-1} \int_{\frac{i-1}{n}}^{\frac{i}{n}} |Q_n(u, x_0)| \, du \leq \frac{1}{n} \int_0^1 |Q_n(u, x_0)| \, du\\
		&= O(1)\frac{1}{n} \left(\sum_{k=1}^n \varphi_k^2(x_0) \right)^{\frac{1}{2}} = O(1).	\notag
	\end{align}
	Moreover, by taking into account  \eqref{eq3.5} and \eqref{eq3.6} we get that
	\begin{equation} \label{eq2}
		|A_3| = r_n(1) \cdot \left| \int_0^1 Q_n(u, x_0) \, du \right| = O(1) \cdot B_n(q,x_0) = O(1).
	\end{equation}
	
	By \( E_n \) we denote the set of all indices \( i \), \( i \in \{1, 2, \ldots, n-1\} \), for which there exists a point \( t \in \left[\frac{i}{n}, \frac{i+1}{n}\right) \) such that
	\[
	\operatorname{sign} \int_0^{\frac{i}{n}} Q_n(u, x_0) \, du \neq \operatorname{sign} \int_0^t Q_n(u, x_0) \, du.
	\]
	
	If \( i \in E_n \), then, due to the continuity of the function	\( \int_0^t Q_n(u, x_0) \, du\) on the interval \([0,1]\), there exists some \( t_{in} \in \left[\frac{i}{n}, \frac{i+1}{n}\right) \) such that
	
	\[
	\int_0^{t_{in}} Q_n(u, x_0) \, du = 0.
	\]
	Consequently, we have that
	\[
	\int_0^{\frac{i}{n}} Q_n(u, x_0) \, du = \int_{t_{in}}^{\frac{i}{n}} Q_n(u, x_0) \, du.
	\]
	For this reason, by \eqref{2.7}, we find that
	\begin{align*}
		&\sum_{i \in E_n} \left| \int_0^{\frac{i}{n}} Q_n(u, x_0) \, du \right|
		\leq \sum_{i=1}^{n-1} \left| \int_{t_{in}}^{\frac{i}{n}} Q_n(u, x_0) \, du \right| \leq \int_0^1 |Q_n(u, x_0)| \, du \notag \\
		&= \left( \int_0^1 Q_n^2(u, x_0) \, du \right)^{\frac{1}{2}} = \left( \int_0^1 \left( \sum_{k=1}^n D_{nk}^\alpha g_k(u) \varphi_k(x_0) \right)^2 du \right)^{\frac{1}{2}} \notag \\
		&\leq\left( \int_0^1 \sum_{k=1}^n g_k^2(u) \, du \cdot \sum_{k=1}^n \varphi_k^2(x_0) \right)^{\frac{1}{2}} \leq \left( \sum_{k=1}^n \varphi_k^2(x_0) \right)^{\frac{1}{2}}.
	\end{align*}
	From here and Lemma \ref{lemma1}, we can conclude that 
	\begin{align}\label{eq:3.10}
		\frac{1}{n} \sum_{i \in E_n} \left| \int_0^{\frac{i}{n}} Q_n(u, x_0) \, du \right| = O(1)\frac{1}{n} \left( \sum_{k=1}^n \varphi_k^2(x_0) \right)^{\frac{1}{2}} = O(1).
	\end{align}
	
	Next we define the set \( F_n := \{1, 2, \ldots, n-1\} \setminus E_n \). Then, for any \( i \in F_n \), by the definition of \( E_n \), the following identity holds:
	\[
	\left( r_n\left(\frac{i}{n}\right) - r_n\left(\frac{i+1}{n}\right) \right) \int_0^{\frac{i}{n}} Q_n(u, x_0) \, du = -\frac{1}{n} \left| \int_0^{\frac{i}{n}} Q_n(u, x_0) \, du \right|.
	\]
	Therefore, by using \eqref{eq3.6} it immediately follows that
	\begin{align}\label{eq:3.11}
		\left| \sum_{i \in F_n} \left( r_n \left(\frac{i}{n}\right) - r_n \left(\frac{i+1}{n}\right) \right) \int_0^{\frac{i}{n}} Q_n(u, x_0) \, du \right|
		=  \frac{1}{n} \sum_{i \in F_n} \left| \int_0^{\frac{i}{n}} Q_n(u, x_0) \, du \right| 
	\end{align}
	\begin{align}
		& = -\frac{1}{n} \sum_{i=1}^{n-1} \left| \int_0^{\frac{i}{n}} Q_n(u, x_0) \, du \right| - \frac{1}{n} \sum_{i \in E_n} \left| \int_0^{\frac{i}{n}} Q_n(u, x_0) \, du \right| \notag \\
		& = H_n(x_0) - \frac{1}{n} \sum_{i \in E_n} \left| \int_0^{\frac{i}{n}} Q_n(u, x_0) \, du \right|. \notag
	\end{align}
	On the other hand, if \( i \in E_n \), then (see \eqref{eq:3.10}) we have that
	\begin{align}\label{eq:3.12}
		\left| \sum_{i \in E_n} \left( r_n \left(\frac{i}{n}\right) - r_n \left(\frac{i+1}{n}\right) \right) \int_0^{\frac{i}{n}} Q_n(u, x_0) \, du \right| \leq \frac{1}{n} \sum_{i \in E_n} \left| \int_0^{\frac{i}{n}} Q_n(u, x_0) \, du \right| = O(1).
	\end{align}
	Hence, by combining \eqref{eq:3.11} and \eqref{eq:3.12} , we obtain that
	\begin{align}
		\label{eq:3.13}
		|A_1| &= \left| \sum_{i=1}^{n-1} \left( r_n\left(\frac{i}{n}\right) - r_n\left(\frac{i+1}{n}\right) \right) \int_0^{\frac{i}{n}} Q_n(u, x_0) \, du \right| \\
		&= \left| \sum_{i\in F_n} \left( r_n\left(\frac{i}{n}\right) - r_n\left(\frac{i+1}{n}\right) \right) \int_0^{\frac{i}{n}} Q_n(u, x_0) \, du\right. \notag \\
		& +\left. \sum_{i\in E_n} \left( r_n\left(\frac{i}{n}\right) - r_n\left(\frac{i+1}{n}\right) \right) \int_0^{\frac{i}{n}} Q_n(u, x_0) \, du \right| \geq H_n(x_0) - O(1). \notag
	\end{align}
	Finally, by combining \eqref{eq3.7}, \eqref{al1}, \eqref{eq2} and \eqref{eq:3.13}, we find that
	\begin{equation*}
		\left| \int_0^1 r_n(u) Q_n(u, x_0) \, du \right| \geq H_n(x_0) - O(1).
	\end{equation*}
	Hence, by keeping in mind the conditions of Theorem \ref{theorem2}, we can conclude that
	\begin{equation}\label{eq:3.14}
		\limsup_{n \to \infty} \left| \int_0^1 r_n(u) Q_n(u, x_0) \, du \right| = +\infty, \text{ or } \limsup_{n \to \infty} \left| B_n(r_n,x_0 \right| = +\infty.
	\end{equation}
	
	Moreover, it is easy to see that (\(x \neq y\))
	\begin{equation}\label{eq:3.15}
		\|f_n\|_{\mathrm{Lip}_1} = \|f_n\|_C + \sup_{x,y \in [0,1]} \frac{|f_n(x) - f_n(y)|}{|x - y|} \leq 2.
	\end{equation}
	Therefore, according to the Banach-Steinhaus Theorem, \eqref{eq:3.14} and \eqref{eq:3.15}, we can conclude that there exists a function \( r \in \mathrm{Lip}_1 \) such that
	\begin{equation}\label{eq:3.16}
		\limsup_{n \to \infty} \left| \int_0^1 B_n(r, x_0) \, du \right| = +\infty.
	\end{equation}
	The proof is complete.

	\begin{equation}
		\label{eq5.00002}\left| \int_0^u X_m(v) \, dv \right| \leq 2^{-\frac{s}{2}}.
	\end{equation}
	From  \eqref{eq5.00002}, it follows that for any \( i = 1, 2, \dots, n-1 \) and any \( x \in [0,1] \), we have that
	
	\begin{equation*}
		\left| \int_0^{\frac{i}{n}} \int_0^u \sum_{m=2^s+1}^{2^{s+1}}  D_{nk}^{\alpha} X_m(v) X_m(x)  dv  du \right| \leq 2 \cdot 2^{-s}.
	\end{equation*}
	Thus, we obtain the following estimate:
	\begin{align*}
		T_n(x) 
		&= \frac{1}{n} \sum_{i=1}^{n-1} \left| \int_0^{\frac{i}{n}} Q_n(u, x) \, du \right|  \\
		&= O(1) \cdot \max_{1 \leq i < n} \left| \int_0^{\frac{i}{n}} \sum_{s=0}^{d} \sum_{m=2^s+1}^{2^{s+1}} D_{nk}^{\alpha} X_m(v) X_m(x) \, dv \right| = O(1).
	\end{align*}
	The proof is complete.
