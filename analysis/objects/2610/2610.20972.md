<!-- repeatedtheorem-1: body_md -->
Let $M>0$, $K=\{\rho \in \mathscr{P}_2(\mathbb{R}^d)\colon \|\rho\|_{L^\infty}\leq M\}$, and $\nu \in \mathscr{P}_2(\mathbb{R}^d)$. 
                                                                                                                                                                                                                                                          
    Assume that $\nu$ satisfies the technical assumption \eqref{assumption nu}. 
                                                                                                   
                                                         
                 
                      
                                                                                         
                                                                                                 
 
                                                           
For any $\rho_0,\rho_1 \in K$, let $\rho_t$ be the constant-speed Wasserstein geodesic connecting them. Then, it holds
\begin{equation*}
    \mathscr{F}_\nu(\rho_t)\leq(1-t)\mathscr{F}_\nu(\rho_0)+t\mathscr{F}_\nu(\rho_1)-\frac{\lambda}{2}t(1-t)W^2_2(\rho_0,\rho_1),
\end{equation*}
with $$\lambda=\frac{1}{d}-cM^{\frac{1}{d}},$$
where $c$ is a constant only depending on $\nu$ and not on $\rho_0,\rho_1,\in K$.

<!-- corollary-1: body_md -->
\label{uniqueness corollary}
    If $\nu \in \mathscr{P}_2$ satisfies \eqref{assumption nu} and $\rho_0 \in \mathscr{P}_2(\mathbb{R}^d)\cap L^\infty(\mathbb{R}^d)$, then the sliced Wasserstein flow is well-posed in $L^\infty(\mathbb{R}^d)$, i.e., there exists a unique solution $\{\rho_t\}_{t\geq 0}$ such that $t\mapsto ||\rho_t||_{L^\infty}$ is locally bounded.

<!-- definition-1: body_md -->
\label{costs}
    Let $\mu, \nu \in \mathscr{P}\left(\Omega\right)$ and let $c:\Omega\times\Omega\to\mathbb R$ be a continuous cost function . 
    The map $T : \mathbb{R}^d \to \mathbb{R}^d$ is said to be an optimal transport map for the cost $c$ if it solves the problem
    \begin{equation}
        \label{Monge_pb}
        \min_{S} \left\{\int_{\Omega}c(x,S(x))d\mu(x), \hspace{2mm} S \hspace{2mm} \text{Borel map such that} \hspace{2mm} S_{\#}\mu=\nu \right\}.
    \end{equation}
    The map $\varphi: \mathbb{R}^d \to \mathbb{R}$ is said to be a Kantorovich potential if it solves the problem
    \begin{equation}
    \label{Potential_pb}
        \max_{\psi}\left\{\int_{\Omega}\psi(x)d\mu(x)+\int_{\Omega}\psi^c(x)d\nu(x)\right\},
    \end{equation}
    where $\psi^c(x) \coloneqq \inf_{y \in \mathbb{R}^d} \left\{c(x,y)-\psi(y)\right\}$.

Let $\Pi_\mu^\nu\coloneqq \{\gamma\ \in \mathcal{P}(\Omega\times\Omega)\colon p^1_{\#}\gamma=\mu, p^2_{\#}\gamma=\nu\}$, where $p^i (x_1,x_2)=x_i$. We define the $p$-Wasserstein distance from $\mu$ to $\nu$ to be
\begin{equation*}
    W_p(\mu,\nu)=\left(\int_{\Omega\times \Omega} |x_1-x_2|^{p}\,d\pi(x_1,x_2)\right)^{\frac{1}{p}}=\inf_{\gamma \in \Pi_\mu^\nu}\left(\int_{\Omega\times\Omega}|x_1-x_2|^p\,d\gamma(x_1,x_2)\right)^{\frac{1}{p}}.
\end{equation*}
     It is well known that $W_p$ defines a distance over $\mathscr{P}_p(\Omega)$ for any $p\geq 1$ (see, for example, \cite[Chapter 5]{santambrogio2015optimal}).
<!-- end definition-1 -->
<!-- remark-1: body_md -->
The problems \eqref{Monge_pb} and \eqref{Potential_pb} do not always admit a solution. However, when they exist, the two quantities defined by \eqref{Monge_pb} with \say{$\min$} replaced by \say{$\inf$}, and \eqref{Potential_pb} with \say{$\max$} replaced by \say{$\sup$} coincide. Moreover, when a solution $T$ to the Monge problem exists, then the joint density $\pi$ appearing in the definition of $W_p$ is the measure concentrated on $\text{gr}(T)\subset \Omega \times \Omega$, where $\text{gr}T$ denotes the graph of $T$, namely $\text{gr}T=\{(x,T(x))\in \Omega\times\Omega\colon x\in \Omega\}$. As a consequence, if $T$ is an optimal transport map and $\varphi$ is the Kantorovich potential for the associated problem, it holds
\begin{equation*}
\label{Wassersein distance, optimal transport, kantorovich potential}
    W_p^p(\mu,\nu)=\int_{\Omega} |T(x)-x|^p\,d\mu=\int_{\Omega}\varphi(x)\,d\mu(x)+\int_{\Omega}\varphi^c(x)\,d\nu(x).
\end{equation*}
<!-- end remark-1 -->
<!-- theorem-1: body_md -->
\label{Brenier's_thm}
    Let $\mu$, $\nu$ be two absolutely continuous probability measures in $\mathscr{P}_2(\mathbb R^d)$. A map $T \colon \mathbb{R}^d \to \mathbb{R}^d$ such that $T_{\#}\mu=\nu$ is optimal for the optimal transport problem \eqref{Monge_pb} with the cost $c(x,y)=\frac 12|x-y|^2$ if and only if there exists a convex function $u \colon \mathbb{R}^d \to \mathbb{R}$ such that $T=\nabla u$ a.e.                                                                                                               
    Moreover if this is the case, then $u$ must satisfy the Monge-Amp\`ere equation
    \begin{equation*}
    \nu(\nabla u(x))    \det D^2 u(x)=\mu(x)
    \end{equation*}
    a.e. on $\mathbb{R}^d$.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{brenier 1d}
    Let $\mu$, $\nu$ be two probability densities with finite second moments. Then there exists a unique non-decreasing transport map from $\mu$ to $\nu$, which is optimal and given by
    \[T(x)=F_\nu^{[-1]}\left(F_{\mu}(x)\right),\]
    where $F_{\mu}(x)=\mu((-\infty,x])$ and $F^{[-1]}(y)=\inf\{t\in \mathbb{R} \colon F(t)\geq y\}$.
<!-- end theorem-2 -->
<!-- definition-2: body_md -->
A curve satisfying \eqref{constant speed geo} is called \textit{constant speed geodesic} from $\rho_0$ to $\rho_1$.
    
    A functional $\mathscr{F}\colon \mathscr{P}_2(\mathbb{R}^d)\to \mathbb{R}$ is $\lambda$-geodesically convex if for every two points $\rho_0, \rho_1 \in \mathscr{P}_2(\mathbb{R}^d)$ there exists a constant speed geodesic $\rho(t)$ connecting $\rho(0)=\rho_0$ and $\rho(1)=\rho_1$ such that
\begin{equation}
\label{lambda convexity def}
    \mathscr{F}(\rho_t)\leq (1-t)\mathscr{F}(\rho_0)+t\mathscr{F}(\rho_1)-\frac{\lambda}{2}t(1-t)W_2^2(\rho_0,\rho_1).
\end{equation}

<!-- theorem-3: body_md -->
{\cite[Theorem 5.27]{santambrogio2015optimal}}
\label{geodetiche}
Let $\rho_0,\rho_1 \in \mathscr{P}_2(\mathbb{R}^d)$ with $\rho_0$ absolutely continuous
with respect to the Lebesgue measure, and suppose there exists an optimal transport map $T$ for the quadratic cost $c(x,y)=|x-y|^2$, i.e., a solution of \eqref{Monge_pb}. Then, the unique geodesic $\rho_t \colon [0,1]\to \mathscr{P}_2(\mathbb{R}^d)$ from $\rho(0)=\rho_0$ to $\rho(1)=\rho_1$ is given by
\begin{equation*}
    \rho_t=(T_t)_{\#}\rho_0,
\end{equation*}
where
\begin{equation*}
    T_t(x)=(1-t)x+tT(x)
\end{equation*}
for any $x \in \text{spt }\,\rho_0$.
Then $T_t$ is injective for $t \in [0,1)$ and the curve $(\rho_t)_{t\geq 0}$ solves
\begin{equation*}
    \partial_t \rho_t +\nabla \cdot (\rho w )=0,
\end{equation*}
where $w_t(x)=(T-\mathrm{id})\circ T_t^{-1}(x)$.

<!-- lemma-1: body_md -->
\label{Lemma 1}
    Let $v[\rho]$ be defined by \eqref{velocity field}. Assume that $K\subset \mathscr{P}_2(\mathbb{R}^d)$ is a geodesically convex set and that for any $\rho \in K$
    \begin{equation*}
        \text{Lip}\left(v[\rho]+\frac{\mathrm{id}}{d}\right)\leq L.
    \end{equation*}
    Then $\rho\mapsto \mathscr{F}_\nu(\rho)$ is $\lambda$-geodesically convex on $K$ for $\lambda=\frac{1}{d}-L$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Let $\rho_0,\rho_1$ be two probability densities in $K$, and let $(\rho_t)_{t\geq 0}\subset K$ be the Wasserstein geodesic connecting them. Let $T$ be the optimal transport map from $\rho_0$ to $\rho_1$ and $T_t\coloneqq (1-t)\mathrm{id}+tT$. Then, by Theorem \ref{geodetiche} $\rho_t$ solves \begin{equation*}
    \partial_t \rho_t +\text{div}(w_t\rho_t)=0,
\end{equation*}
where $w_t(x)\coloneqq (T-\mathrm{id})\circ T_t^{-1}(x)$ . We have
\begin{equation*}
    \begin{split}
        \frac{d}{dt}\mathscr{F}_\nu(\rho_t)=& \int \frac{\delta \mathscr{F}_\nu}{\delta \rho}(\rho_t)\partial_t \rho_t=\int \nabla \frac{\delta\mathscr{F}_\nu}{\delta\rho}(\rho_t)\cdot w_t \rho_t = -\int v[\rho_t]\cdot w_t \rho_t,
    \end{split}
\end{equation*}
\begin{equation}
\label{derivata seconda}
    \begin{split}
        \frac{d^2}{dt^2}\mathscr{F}_\nu(\rho_t)=&-\underbrace{\int \partial_t v[\rho_t]\cdot w_t \rho_t}_{\coloneqq I_1} - \underbrace{\int v[\rho_t]\cdot \partial_t(w_t\rho_t)}_{\coloneqq I_2}.
    \end{split}
\end{equation}
The term $I_2$ can be estimated as follows; let $\overline{v}$ be a time independent vector field, with the property that 
\begin{equation*}
    \left\|D\left(\overline{v}(x)+\frac{\mathrm{id}}{d}\right)\right\|_{L^\infty}\leq L.
\end{equation*}
Then,
\begin{equation*}
\label{derivata dist secondo termine}
    \begin{split}
        &\frac{d}{dt}\int \overline{v}\cdot w_t \rho_t=\frac{d}{dt}\int \overline{v}\cdot (T-\mathrm{id})(T_t^{-1}(x))T_{t \#}\rho_0\\ &=\frac{d}{dt}\int \overline{v}(T_t(x))\cdot (T(x)-x)\rho_0
        = \int (T(x)-x)\cdot D \overline v (T_t(x))\,  (T(x)-x)\rho_0\\
        &=\int (T(x)-x)\cdot D\left(\overline{v}+\frac{\mathrm{id}}{d}\right)(T_t(x))(T(x)-x)\,\rho_0-\frac{1}{d}\int |T(x)-x|^2\rho_0\\
        &\leq \left(L-\frac{1}{d}\right)\int |T(x)-x|^2\,\rho_0. 
    \end{split}
\end{equation*}
Taking $\overline v=v[\rho_t]$, we obtain
\begin{equation}
\label{primo termine}
    \begin{split}
        -I_2=-\int v[\rho_t]\cdot \partial_t(w_t \rho_t) \geq \left(\frac{1}{d}-L\right)W^2_2(\rho_0,\rho_1).
    \end{split}
\end{equation}
On the other hand, we claim that $I_1$ is negative. Indeed,
\begin{equation*}
    \begin{split}
        I_1=&\int \partial_t(v[\rho_t])\cdot w_t\rho_t= \int \partial_t\left(\fint_{\mathbb{S}^{d-1}}  T_{t,\vartheta}(x\cdot\vartheta)\vartheta \,d\sigma(\vartheta)\right)\cdot w_t \rho_t\,dx\\
        =&\int\partial_t\left[\fint_{\mathbb{S}^{d-1}}    N_\vartheta^{-1}(R_{t,\vartheta}(x\cdot \vartheta))\vartheta \,d\sigma(\vartheta)  \right] \cdot w_t\rho_t\,dx\\
        =& \int \fint_{\mathbb{S}^{d-1}}  \frac{\partial_t \int_{y\cdot \vartheta \leq x \cdot \vartheta} \rho_t(y)dy}{\nu^\vartheta(T_{t,\vartheta}(x\cdot \vartheta))}\vartheta\,\cdot w_t \rho_t\,d\sigma(\vartheta)\,dx\\
        =&-\int \fint_{\mathbb{S}^{d-1}}  \frac{\int_{y\cdot \vartheta \leq x \cdot \vartheta}\text{div}(w_t\rho_t)\,dy}{\nu^\vartheta(T_{t,\vartheta}(x\cdot \vartheta))}\vartheta\,\cdot w_t \rho_t\,d\sigma(\vartheta)\,dx.
    \end{split}
    \end{equation*}
By Fubini, we change the order of integration and perform the integration on level sets $x\cdot \vartheta =c$:
\begin{equation}
\begin{split}
    \label{secondo termine}
        I_1=&-\fint_{\mathbb{S}^{d-1}} \int_{-\infty}^{+\infty} \frac{1}{\nu^\vartheta(T_{t,\vartheta}(c))} \left(\int_{y\cdot \vartheta \leq c}\text{div}(w_t\rho_t)\,dy\right) \left(\int_{x\cdot \vartheta=c}\vartheta\,\cdot w_t \rho_t\, \,dx\right)dc\,d\sigma(\vartheta)\\
        =&-\fint_{\mathbb{S}^{d-1}} \int_{-\infty}^{+\infty} \frac{1}{\nu^\vartheta(T_{t,\vartheta}(c))} \bigg(\int_{x\cdot \vartheta=c}\vartheta\,\cdot w_t \rho_t\,dx\bigg)^2\,d\sigma(\vartheta)\, dc\leq 0.
    \end{split}
\end{equation}
Combining \eqref{derivata seconda}, \eqref{primo termine} and \eqref{secondo termine}, we obtain
\begin{equation*}
    \begin{split}
        \frac{d^2}{dt^2}\mathscr{F}_\nu(\rho_t)\geq\left(\frac{1}{d}-L\right)W^2_2(\rho_0,\rho_1).
    \end{split}
\end{equation*}

<!-- lemma-2: body_md -->
\label{lemma 2}
    Let $v[\rho]$ be defined by \eqref{velocity field}, $\nu \in \mathcal{P}_2(\mathbb{R}^d)$, and $K=\{\rho \in \mathscr{P}_2(\mathbb{R}^d)\colon \|\rho\|_{L^p}\leq M\}$, for some $p>d'$, $M>0$.
    Assume that, for almost every $\vartheta\in\mathbb S^{d-1}$, the support of $\nu^\vartheta$ is an interval $I_\vartheta$, and that $\nu^\vartheta\geq c_0>0$ almost everywhere on $I_\vartheta$.
    Then, there exists $L=L(c_0,M,p,d)$ such that, for any $\rho \in K$,
    $$\text{Lip}\left(v[\rho]+\frac{\mathrm{id}}{d}\right)\leq L,$$ and $\mathscr{F}_\nu$ is $\lambda$-geodesically convex on $K$ with $\lambda=\frac{1}{d}-L$.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Deriving the relation (see Theorem \ref{brenier 1d})
        \begin{equation*}
        N_\vartheta(T_\vartheta(r))=R_\vartheta(r) \quad \text{for any }r\in \text{spt }\rho^\vartheta,
\end{equation*}
we deduce that
\begin{equation*}
    T'_\vartheta(r)=\frac{\rho^\vartheta(r)}{\nu^\vartheta(T_\vartheta(r))}.
\end{equation*}
Therefore,
    \begin{equation*}
    \begin{split}
        \left|Dv[\rho](x)+\frac{\mathrm{id}}{d}\right|&= \,\left|\fint_{\mathbb{S}^{d-1}} T_{\vartheta}'(x\cdot \vartheta)\vartheta \otimes \vartheta\right| \leq \fint_{\mathbb{S}^{d-1}} \bigg|T_{\vartheta}'(x\cdot \vartheta)\bigg|d\sigma(\vartheta)\\
        &\leq \fint_{\mathbb{S}^{d-1}} \frac{\rho^\vartheta(x\cdot \vartheta)}{\nu^\vartheta(T_{\vartheta}(x\cdot \vartheta))}\,d\sigma(\vartheta)\leq \frac{1}{c_0}\fint_{\mathbb{S}^{d-1}} \rho^\vartheta(x\cdot \vartheta)\,d\sigma(\vartheta)\\
        &=\frac{1}{c_0}\fint_{\mathbb{S}^{d-1}}\int_{e\cdot\vartheta=0}\int_0^{+\infty}\rho(x+er)r^{d-2}\,dr\,d\mathcal H^{d-2}(e)\,d\sigma(\vartheta)=\frac{(d-1)\omega_{d-1}}{d\omega_d c_0}\int \frac{\rho(y)}{|x-y|}\,dy,
    \end{split}
    \end{equation*}
where we used \cite[Lemma 5.1]{cozzi2025long} to obtain the last equality.
Moreover, if $\rho \in L^p$ for $p>d'$, we have
\begin{equation}
\label{rho/x stima Lp}
    \begin{split}
        &\int_{\mathbb{R}^d} \frac{\rho(y)}{|x-y|}\,dy\leq\int_{B_R(x)}\frac{\rho(y)}{|x-y|}\,dy+\int_{B_R^c(x)}\frac{\rho(y)}{|x-y|}\,dy\\
        &\leq \left(\int_{B_R}\frac{1}{|y|^{p'}}\,dy\right)^{\frac{1}{p'}}\left(\int_{\mathbb{R}^d}\rho^p\,dx\right)^{\frac{1}{p}}+\frac{1}{R} \leq c(d,p)R^{\frac{d}{p'}-1}\|\rho\|_{L^p}+\frac{1}{R}.
    \end{split}
\end{equation}

\left(\frac{c(d,p)(d-p')}{p'}\right)^{\frac{p'}{d}}=: L
\end{equation}
for any $\rho \in K$. Since $K$ is geodesically convex (see, for example \cite[Chapter 7]{santambrogio2015optimal}), we can apply Lemma \ref{Lemma 1} and deduce the claim.

\label{proiezioni limitate}
    We do not expect that measures with the property that $\nu^\vartheta\geq c_0>0$ for a.e. $\vartheta \in \mathbb{S}^{d-1}$ exist in $\mathbb{R}^d$ for $d>3$. We will show in this remark that this is the case when we strengthen condition \eqref{eq: bounded projections} replacing it with the following property (which in turn implies \eqref{eq: bounded projections}): for almost every $\vartheta \in \mathbb{S}^{d-1}$ there exists an interval $I_\vartheta$ such that
    \begin{equation}
        \label{stronger condition bounded projections}
        \begin{split}
        \nu^\vartheta\geq c_0 \quad&\text{ on }I_\vartheta\\
        \nu^\vartheta=0 \quad &\text{ on }\mathbb{R}\setminus I_\vartheta \\
        C\coloneqq \bigcap_{\vartheta \in \mathbb{S}^{d-1}} \{x\cdot \vartheta \in I_\vartheta\} \quad &\text{ is uniformly convex}
        \end{split}
    \end{equation}
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                
    Indeed, for any $\vartheta \in \mathbb{S}^{d-1}$, let $r_\vartheta=\sup_{x\in C}{x\cdot \vartheta}$ and $x_\vartheta \in \partial C$ be the boundary point such that $x_\vartheta\cdot \vartheta =r_\vartheta$. Then, consider $S_{\vartheta,\varepsilon}=\{x \in C \colon |x\cdot\vartheta-r_\vartheta|\leq\varepsilon\}$, which is a slice of size $\varepsilon>0$ containing $x_\vartheta$ in its boundary. Since, by assumption, $C$ is uniformly convex and smooth (notice that without the assumption it would just be convex), given any $\varepsilon>0$, for any ball $B_{\sqrt{\varepsilon}}(x_\vartheta)$, it holds $S_{\vartheta,\varepsilon}\subset B_{\sqrt{\varepsilon}}(x_\vartheta)$. By \eqref{stronger condition bounded projections}, we have that $\nu(S_{\vartheta, \varepsilon})\gtrsim \varepsilon$, and hence $\nu(B_{\sqrt{\varepsilon}}(x_\vartheta))\gtrsim \varepsilon$. We now consider a collection of points $\{y_i\}\subset \partial C$ such that $|y_i-y_j|\geq 2\sqrt{\varepsilon}$. The number of such points is of the order $(\sqrt{\varepsilon})^{-(d-1)}$, since $\partial C$ is a $d-1$ dimensional surface. Define $(\partial C)_\varepsilon\coloneqq \{x\in \mathbb{R}^d\colon \text{dist}(\partial C, x)\leq \varepsilon\}$. Considering the above argument for any ball $B_{\sqrt{\varepsilon}}(y_i)$ we obtain that
    \begin{equation*}
    \label{catena disuguaglianze epsilon arbitrario}1=\nu(C)\geq \nu((\partial C)_\varepsilon)\geq \sum_{i=1}^{\varepsilon^{-\frac{d-1}{2}}} \nu(B_{\sqrt{\varepsilon}}(y_i))\gtrsim \sum_{i=1}^{\varepsilon^{-\frac{d-1}{2}}} \varepsilon\gtrsim \varepsilon^{\frac{-d+3}{2}},\end{equation*}
    where we denoted by $\gtrsim$ inequalities holding up to multiplication by a dimensional constant. By arbitrariness of $\varepsilon$, we deduce that it is necessary to assume $d\leq 3$. Moreover, when $d=3$, we also observe that $\nu$ must give positive mass to $\partial C$.

\label{proiezioni uniformi}
The measure $\nu$ with uniform projections (i.e.$, \nu^\vartheta(r)=\chi_{[-1/2,1/2]}(r)$ for any $\vartheta\in \mathbb{S}^{d-1}$) satisfies \eqref{stronger condition bounded projections}: by Remark \ref{proiezioni limitate} such a measure only exists when $d=2,3$, and in this case the set $C$ mentioned there coincides with the closed ball of radius $1/2$ centered at the origin. \begin{equation*}
    \begin{split}
        \frac{SW_2^{2}(\rho,\nu)}{2}=\int \frac{|x|^2}{2d}\rho(x)\,dx-\frac{\omega_{d-1}}{2d\omega_{d}}\iint \rho(x)\rho(y)|x-y|\,dx\,dy+\frac{1}{24} \hspace{5mm} \text{for any }\rho \in \mathcal{P}_2(\mathbb{R}^d).
    \end{split}
\end{equation*}

\label{Main theorem}
    Let $M>0$, $K=\{\rho \in \mathscr{P}_2(\mathbb{R}^d)\colon \|\rho\|_{L^\infty}\leq M\}$, and $\nu \in \mathscr{P}_2(\mathbb{R}^d)$.
    Assume that there exist $0<\alpha<(d-1)/d, \kappa>0$
                                                                                   
    such that\\

\begin{split}
\label{assumption nu}
                                                                                                 
\nu^\vartheta(r)\geq  \kappa\left(\min \{N_\vartheta(r),1-N_\vartheta(r)\}\right)^\alpha
\hspace{3mm} &\text{for a.e. } \vartheta \in \mathbb{S}^{d-1}\text{ and every }r\in\mathbb{R}.
\end{split}
\end{equation}  
Then, $\mathscr{F}_\nu$ is $\lambda$-geodesically convex on $K$ with $$\lambda=\frac{1}{d}-\frac{c(d)}{\left(1-\frac{\alpha d}{d-1}\right)\kappa}M^{\frac{1}{d}},$$

<!-- remark-5: body_md -->
If $\nu$ satisfies \eqref{assumption nu}, then $\nu$ must be compactly supported. 
Indeed, since $N_\vartheta'=\nu^\vartheta$, assumption \eqref{assumption nu} is equivalent to 
\begin{equation*}
    N'_\vartheta(r)\geq \kappa\min\{(1-N(r))^\alpha,N(r)^\alpha\}.
\end{equation*}
Moreover, since $\lim_{r\to +\infty} N(r)=1$, for $r$ large enough this leads to 
\begin{equation}
\label{equation not bounded solutions}
    N'_\vartheta(r)\geq \kappa(1- N_\vartheta(r))^\alpha.
\end{equation}
This in turn gives $(1-N_\vartheta)^{1-\alpha}\leq -c<0$, and it is not possible to have a nonnegative solution of this differential 
inequality on an unbounded domain for $\alpha <1$.

Thanks to Lemma \ref{Lemma 1} and the fact that the set $K$ is geodesically convex, we just need to prove that \eqref{LipConstant} holds true for any $\rho \in K$ and any $\nu$ satisfying assumption \eqref{assumption nu}. By definition of $v[\rho,\nu]$, fixing a point $x \in \text{spt }\,\rho$, we have (the first inequality is obtained as in the proof of Lemma \ref{lemma 2})
\begin{equation*}
\label{lip velocity 1}
    \begin{split}
        \left|Dv[\rho](x)+\frac{\mathrm{id}}{d}\right|\,&\leq \fint_{\mathbb{S}^{d-1}}  \frac{\rho^\vartheta(x\cdot \vartheta)}{\nu^\vartheta(T_{\vartheta}(x\cdot\vartheta))}\,d\sigma(\vartheta)\\
        \leq& \frac{1}{\kappa}\fint_{\mathbb{S}^{d-1}}  \frac{\rho^\vartheta(x\cdot \vartheta)}{(\min \{N_\vartheta(T_{\vartheta}(x\cdot \vartheta)),1-N_\vartheta(T_{\vartheta}(x\cdot \vartheta))\})^\alpha}\,d\sigma(\vartheta)\\
        =& \frac{1}{\kappa}\fint_{\mathbb{S}^{d-1}}  \frac{\rho^\vartheta(x\cdot \vartheta)}{(\min \{R_\vartheta(x\cdot \vartheta),1-R_\vartheta(x\cdot \vartheta)\})^{\alpha}}\,d\sigma(\vartheta),
    \end{split}
\end{equation*}
where in the last equality, we used again the relation
\begin{equation*}
N_\vartheta(T_\vartheta(r))=R_\vartheta(r) \quad \text{for any }r\in \text{spt }\rho^\vartheta.
\end{equation*}
Denoting by $\chi^+(\vartheta)=\chi_{\{R_\vartheta(x\cdot \vartheta) \leq 1/2\}}(\vartheta)$ and $\chi^-(\vartheta)=\chi_{\{R_\vartheta(x\cdot \vartheta) \geq 1/2\}}(\vartheta)$, we obtain
\begin{equation}
\label{eq lip velocity 1}
    \begin{split}
        \left|Dv[\rho](x)+\frac{\mathrm{id}}{d}\right|\,\leq&\frac{1}{\kappa}\fint_{\mathbb{S}^{d-1}}\frac{\rho^\vartheta(x\cdot \vartheta)}{R_\vartheta(x\cdot\vartheta)^\alpha}\chi^+(\vartheta)\,d\sigma(\vartheta)+\frac{1}{\kappa}\fint_{\mathbb{S}^{d-1}}\frac{\rho^\vartheta(x\cdot \vartheta)}{(1-R_\vartheta(x\cdot\vartheta))^\alpha}\chi^-(\vartheta)\,d\sigma(\vartheta)\\
        \leq& \frac{1}{\kappa}\fint_{\mathbb{S}^{d-1}}\frac{\rho^\vartheta(x\cdot \vartheta)}{R_\vartheta(x\cdot\vartheta)^\alpha}\,d\sigma(\vartheta)+\frac{1}{\kappa}\fint_{\mathbb{S}^{d-1}}\frac{\rho^\vartheta(x\cdot \vartheta)}{(1-R_\vartheta(x\cdot\vartheta))^\alpha}\,d\sigma(\vartheta)\\
        =&\frac{2}{\kappa}\fint_{\mathbb{S}^{d-1}}\frac{\rho^\vartheta(x\cdot \vartheta)}{R_\vartheta(x\cdot\vartheta)^\alpha}\,d\sigma(\vartheta),
    \end{split}
\end{equation}
where the last equality was obtained by symmetry.
For any $e \in \mathbb{S}^{d-1}$, define
\begin{equation}
    \label{d-2 m(e) integral} 
    m_{x}(e)=\int_{0}^{+\infty}\rho(x+e\,r)r^{d-2}\,dr.
    \end{equation}
We have that, for any $\vartheta \in \mathbb{S}^{d-1}$
\begin{equation}
    \label{rho theta in x theta}
    \begin{split}
    \rho^\vartheta(x\cdot \vartheta)&=\int_{e\cdot\vartheta=0}m_x(e)\,d\mathcal H^{d-2}(e),\\ R_\vartheta(x\cdot \vartheta)&=\int_{e\cdot \vartheta\leq 0}\int_{0}^{+\infty}\rho(x+e\,r)r^{d-1}\,dr\,d\sigma(e).
    \end{split}
\end{equation}
Here and below, integrals over equators $\{e\in\mathbb S^{d-1}:e\cdot\vartheta=0\}$ and their subsets are taken with respect to $\mathcal H^{d-2}$, while integrals over hemispheres use $\sigma$. When $d=2$, $\mathcal H^0$ is counting measure.

Applying \eqref{rho theta in x theta} in \eqref{eq lip velocity 1}, we obtain
\begin{equation}
\label{eq: main thm}
        \left|Dv[\rho](x)+\frac{\mathrm{id}}{d}\right|\leq
        \frac{2}{\kappa}\fint_{\mathbb{S}^{d-1}}  \frac{\int_{e\cdot \vartheta=0}m_x(e)\,d\mathcal H^{d-2}(e)}{\left(\int_{e\cdot \vartheta \leq 0} \int_{0}^{+\infty}\rho(x+e\,r)r^{d-1}\,dr\,d\sigma(e)\right)^\alpha}\,d\sigma(\vartheta).
\end{equation}
Consider now
\begin{equation*}
    q(M,m_x(e)):=\inf_{0\leq\mu\leq M} \bigg\{\int_{0}^{+\infty}\mu(r)r^{d-1}\,dr \,\colon \int_{0}^{+\infty}\mu(r)r^{d-2}\,dr=m_{x}(e) \bigg\}.
\end{equation*}
The above infimum is achieved by $\mu(r)=M\chi_{(0,R)}(r)$. Here, necessarily,
\[R=\left(\frac{(d-1)m_{x}(e)}{M}\right)^{\frac{1}{d-1}},\]
since
\begin{equation*}
    \begin{split}
    m_{x}(e)=M\int_{0}^{+\infty} \chi_{(0,R)}(r)r^{d-2}\,dr=M \int_0^Rr^{d-2}\,dr=\frac{M}{d-1}R^{d-1}.
    \end{split}
\end{equation*}
Therefore,
\begin{equation}
    \label{optimization problem}
    \begin{split}
        q(M,m_x(e))=M^{-\frac{1}{d-1}}\frac{(d-1)^{\frac{d}{d-1}}}{d}m_x(e)^{\frac{d}{d-1}}.
    \end{split}
\end{equation}
Recalling \eqref{d-2 m(e) integral}, and thanks to \eqref{optimization problem}, we are able to estimate the denominator on the right hand side of \eqref{eq: main thm}, obtaining 
\begin{equation}
\label{Lip Est part 1}
    \begin{split}
        \left|Dv[\rho](x)+\frac{\mathrm{id}}{d}\right|\leq&\frac{2\,d^\alpha\,M^{\frac{\alpha}{d-1}}}{(d-1)^{\frac{\alpha d}{d-1}}\kappa}\fint_{\mathbb{S}^{d-1}}  \frac{\int_{e\cdot \vartheta =0}m_{x}(e)\,d\mathcal H^{d-2}(e)}{\left(\int_{e\cdot \vartheta \leq 0} m_x^{\frac{d}{d-1}}(e)\,d\sigma(e)\right)^{\alpha}}\,d\sigma(\vartheta)\\
        \leq&\frac{c(d)\,M^{\frac{\alpha}{d-1}}}{\kappa} \fint_{\mathbb{S}^{d-1}}  \frac{\int_{e\cdot \vartheta =0}m_{x}(e)\,d\mathcal H^{d-2}(e)}{\left(\int_{e\cdot \vartheta \leq 0} m_{x}(e)\,d\sigma(e)\right)^{\frac{d\alpha}{d-1}}}\,d\sigma(\vartheta),
    \end{split}
\end{equation}
where $c(d)>0$ denotes a constant depending only on $d$ (the term $(d-1)^{\frac{\alpha d}{d-1}}$ can indeed be bounded uniformly for $0\leq\alpha<(d-1)/d$).

We first present the proof in dimension $d=2$, for the reader's convenience. Next, we will generalize the same argument to any dimension.

\noindent\underline{Proof in dimension $d=2$}

Since in this case it holds
\begin{equation*}
    \fint_{\mathbb{S}^{d-1}}  \frac{\int_{e\cdot \vartheta =0}m_{x}(e)\,d\mathcal H^{d-2}(e)}{\left(\int_{e\cdot \vartheta \leq 0} m_{x}(e)\,d\sigma(e)\right)^{\frac{d\alpha}{d-1}}}\,d\sigma(\vartheta)=\fint_{\mathbb{S}^1}\frac{m_x(\vartheta+\pi/2)+m_x(\vartheta-\pi/2)}{\left(\int_{\vartheta-\pi/2}^{\vartheta+\pi/2}m_x(e)\,d\sigma(e)\right)^{2\alpha}}\,d\sigma(\vartheta),
\end{equation*}
the idea is to estimate this integral in the variable $\vartheta$ writing the numerator as a derivative of the denominator. Fix an orthonormal basis $(e_1,e_2)$ of $\mathbb{R}^2$. We parametrize $\mathbb{S}^1$ defining $\theta(e_1,e_2,t)=e_1\cos t+e_2\sin t$. Define $A(e_1,e_2,t)=\{e\in \mathbb{S}^{d-1}\colon e\cdot \theta (e_1,e_2,t)\leq 0\}$, and $A^+(e_1,e_2,t)=A(e_1,e_2,t)\,\cap \{e \in \mathbb{S}^{d-1}\colon e\cdot e_2\geq 0\}$. Define also
\begin{equation*}
    \varphi_{e_1,e_2}(t)= \int_{A^+(e_1,e_2,t)}m_x(e)\,d\sigma(e).
\end{equation*}
The function $\varphi$ is positive, decreasing for $t\in [0,\pi/2)\cup[3\pi/2,2\pi)$ and increasing in $[\pi/2,3\pi/2)$. Moreover it holds
\begin{equation}
\begin{split}
\label{AuxLemma in 2d}
    \theta(e_1,e_2,t)&=\theta(e_1,-e_2,-t),\\
    \varphi_{e_1,e_2}(t)&\leq \int_{A(e_1,e_2,t)}m_x(e)\,d\sigma(e),\\
    |\varphi'_{e_1,e_2}(t)|+|\varphi'_{e_1,-e_2}(-t)|&=\int_{e\cdot \theta(e_1,e_2,t)=0}m_x(e)\,d\mathcal H^{d-2}(e).
\end{split}
\end{equation}
Incidentally, let us note that two of the above identities have been obtained by exploiting the symmetries in our construction when considering an orthonormal basis with a different orientation with respect to the fixed one. This trick, even if more involved, will be the same at the core of the identities in Lemma \ref{AuxLemma}, needed for the construction in dimension $d\geq 3$.

We return to the proof in dimension $d=2$. By our definitions and \eqref{AuxLemma in 2d}, we have
\begin{equation}
\label{stima 2 2d}
\begin{split}
    &\int_{\mathbb{S}^{1}}  \frac{\int_{e\cdot \vartheta =0}m_{x}(e)\,d\mathcal H^{d-2}(e)}{\left(\int_{e\cdot \vartheta \leq 0} m_{x}(e)\,d\sigma(e)\right)^{2\alpha}}\,d\sigma(\vartheta)=\int_{0}^{2\pi}\frac{\int_{e\cdot \theta(e_1,e_2,t)=0}m_x(e)\,d\mathcal H^{d-2}(e)}{\left(\int_{A(e_1,e_2,t)}m_x(e)\,d\sigma(e)\right)^{2\alpha}}\,dt\\
    &=\int_{0}^{2\pi}\frac{|\varphi_{e_1,e_2}'(t)|}{\left(\int_{A(e_1,e_2,t)}m_x(e)\,d\sigma(e)\right)^{2\alpha}}\,dt+\int_{0}^{2\pi}\frac{|\varphi_{e_1,-e_2}'(-t)|}{\left(\int_{A(e_1,-e_2,-t)}m_x(e)\,d\sigma(e)\right)^{2\alpha}}\,dt\\
    &\leq\int_{0}^{2\pi}\frac{|\varphi_{e_1,e_2}'(t)|}{\left(\varphi_{e_1,e_2}(t)\right)^{2\alpha}}\,dt+\int_{0}^{2\pi}\frac{|\varphi_{e_1,-e_2}'(-t)|}{\left(\varphi_{e_1,-e_2}(-t)\right)^{2\alpha}}\,dt\\
    &\leq \frac{1}{1-2\alpha}TV\left[\varphi_{e_1,e_2}^{1-2\alpha}\right]+\frac{1}{1-2\alpha}TV\left[\varphi_{e_1,-e_2}^{1-2\alpha}\right],
                                                                                                                                                                 
    \end{split}
    \end{equation}
    Where $TV$ stands for the total variation. Using the piecewise monotone behavior of $t\mapsto \varphi_{e_1,e_2}(t)$, we have $TV\left[\varphi_{e_1,e_2}^{1-2\alpha}\right]\leq 2\left(\max_{t\in[0,2\pi)}\varphi_{e_1,e_2}(t)\right)^{1-2\alpha}$. Then, by \cite[Lemma 5.1]{cozzi2025long} it holds
\begin{equation}
\label{stima 3 2d}
    \max_{t\in[0,2\pi)}\varphi_{e_1,e_2}(t)\leq \int_{\mathbb{S}^{1}}\int_0^{+\infty}\rho(x+er)dr\,d\sigma(e)=\int_{\mathbb{R}^2}\frac{\rho(y)}{|x-y|}\,dy.
\end{equation}
Applying estimate \eqref{rho/x stima Lp,2} with $p=\infty$ to the right hand side of \eqref{stima 3 2d}, we deduce the existence of a constant $c$ such that 
\begin{equation}
\label{last 2d estimate}
    \max_{t\in[0,2\pi)}\varphi_{e_1,e_2}(t)\leq  c\,M^{\frac{1}{2}},
\end{equation}
 and the estimate does not depend on the chosen basis $(e_1,e_2)$. Combining \eqref{Lip Est part 1}, \eqref{stima 2 2d} and \eqref{last 2d estimate}, we conclude the proof of Proposition \ref{proposition assumption A1} when $d=2$.

\noindent\underline{Proof in dimension $d\geq 3$} 

Define $E$ to be the $(2d-3)-$dimensional set in $\mathbb{R}^{2d}$ given by couples $(e_1,e_2)\in\mathbb{S}^{d-1}\times \mathbb{S}^{d-1}$ such that $e_1\cdot e_2=0$, and endow it with the restriction of $\mathcal H^{2d-3}$. For any given triplet $(e_1,e_2,t)\in E\times [0,2\pi)$, define $\theta(e_1,e_2,t)=e_1\cos t+e_2\sin t$. For fixed $t$, the fibres of $(e_1,e_2)\mapsto\theta(e_1,e_2,t)$ carry $\mathcal H^{d-2}$; the fibres of the full map $\theta:E\times[0,2\pi)\to\mathbb S^{d-1}$ carry $\mathcal H^{d-1}$. All these measures are induced by the Euclidean product metric. As before, define \begin{equation*}
\begin{split}
A(e_1,e_2,t)&=\{e\in \mathbb{S}^{d-1}\colon e\cdot \theta (e_1,e_2,t)\leq 0\},\\
A^+(e_1,e_2,t)&=A(e_1,e_2,t)\,\cap \{e \in \mathbb{S}^{d-1}\colon e\cdot e_2\geq 0\},\\
\partial A(e_1,e_2,t)&=\{e\in \mathbb{S}^{d-1}\colon e\cdot \theta (e_1,e_2,t)= 0\},\\
\partial^+A(e_1,e_2,t)&=\partial A(e_1,e_2,t)\,\cap \{e \in \mathbb{S}^{d-1}\colon e\cdot e_2\geq 0\}.
\end{split}
\end{equation*}
Since no ambiguity can occur, we will simplify the notation, removing the dependence on $e_i,t$. We introduce again the function
\begin{equation*}
    \label{funzione_phi}
    \varphi_{e_1,e_2}(t)\coloneqq \int_{A^+} m_x(e)\, d\sigma(e).
\end{equation*}
Notice that $\varphi_{e_1,e_2}(t)$ is a positive function, decreasing for $t\in [0,\pi/2)\cup [3\pi/2,2\pi)$ and increasing in $[\pi/2,3\pi/2)$.
For the rest of the proof, we will make use of the following 
\begin{lemma} 
\label{AuxLemma}
Let $m_x$, $\varphi, \theta$ be defined as above, and $(e_1,e_2,t)\in E\times{[0,2\pi]}$. Then,
\begin{enumerate}
    \item There exists a function $h\colon \mathbb{R}^{2}\to \mathbb{R}$ such that, for any $t\in[0,2\pi)$, and for any $e_1,e_2\in E$, it holds
\begin{equation*}
\left|\int_{\partial^+A}m(e)
                                                            
h(e\cdot e_1,e\cdot e_2)
                               
\, d\mathcal H^{d-2}(e)\right|=
                                                               
|\varphi'_{e_1,e_2}(t)|
                                
\end{equation*}
    \item Fix $t\in [0,2\pi)$. The value of
\begin{equation*}
    \int_{\theta(e_1,e_2,t)=\vartheta,\, e\cdot e_2 \geq 0} h(e\cdot e_1,e\cdot e_2)\,d\mathcal H^{d-2}(e_1,e_2)
\end{equation*}
is constant with respect to $(e,\vartheta)\in E$.
    \item There exists a dimensional constant $c(d)$ such that, for any two given measurable functions $f,g$ it holds
\begin{equation*}
    \fint_{\mathbb{S}^{d-1}} \frac{\widetilde{f}(\vartheta)}{g(\vartheta)}\,d\sigma(\vartheta)=c(d)\int _{E\times[0,2\pi)}\frac{f(e_1,e_2,t)}{g(\theta(e_1,e_2,t))}\,d\mathcal H^{2d-3}(e_1,e_2)\,dt,
\end{equation*}
    where
\begin{equation*}
    \widetilde{f}(\vartheta)=\int_{\{\theta(e_1,e_2,t)=\vartheta\}}f(e_1,e_2,t)\,d\mathcal H^{d-1}(e_1,e_2,t).
\end{equation*}
\end{enumerate}
\end{lemma}
We postpone the proof of Lemma \ref{AuxLemma}. Combining the results therein contained, we can now conclude the proof of Proposition \ref{proposition assumption A1}. 
Fix $t\in [0,2\pi)$. Using the fact that, for any $(e_1,e_2,t)\in E\times [0,2\pi]$ it holds
\begin{equation*}
    \varphi_{e_1,e_2}(t)\leq \int_{A}m_x(e) \,d\sigma(e),
\end{equation*}
together with point $(1)$ of Lemma \ref{AuxLemma}, one obtains
\begin{equation*}
    \begin{split}
    \label{chain 1}    
    \int_{E}\frac{|\,\varphi'_{e_1,e_2}(t)|}{(\varphi_{e_1,e_2}(t))^{\frac{d\alpha}{d-1}}}\,d\mathcal H^{2d-3}(e_1,e_2)\geq &      
    \left|\int_{E}\frac{\int_{\partial^+A}m(e)h(e\cdot e_1,e\cdot e_2)\,d\mathcal H^{d-2}(e)}{\left(\int_{A}m(e)\,d\sigma(e)\right)^{\frac{d\alpha }{d-1}}}\,d\mathcal H^{2d-3}(e_1,e_2)\right|.
\end{split}
\end{equation*}
Integrating both sides with respect to $t \in [0,2\pi)$, we get
\begin{equation*}
    \int_0^{2\pi}\int_{E}\frac{|\,\varphi'_{e_1,e_2}(t)|}{(\varphi_{e_1,e_2}(t))^{\frac{d\alpha}{d-1}}}\,d\mathcal H^{2d-3}(e_1,e_2)\,dt\geq\left| \int_{E\times[0,2\pi)}\frac{\int_{\partial^+A}m(e)h(e\cdot e_1,e\cdot e_2)\,d\mathcal H^{d-2}(e)}{\left(\int_{A}m(e)\,d\sigma(e)\right)^{\frac{d\alpha }{d-1}}}\,d\mathcal H^{2d-3}(e_1,e_2)\,dt\right|.
\end{equation*}
Applying point $(3)$ of Lemma \ref{AuxLemma} to the right hand side of the above expression, we deduce
\begin{equation*}
\begin{split}
\int_0^{2\pi}\int_{E}&\frac{|\,\varphi'_{e_1,e_2}(t)|}{(\varphi_{e_1,e_2}(t))^{\frac{d\alpha}{d-1}}}\,d\mathcal H^{2d-3}(e_1,e_2)\,dt\\
\ge c(d)&\left|\fint_{\mathbb{S}^{d-1}}\int_{\{(e_1,e_2,t)\colon \theta(e_1,e_2,t)=\vartheta\}}\frac{\int_{e\cdot\vartheta=0, e\cdot e_2 \geq 0} m(e)h(e\cdot e_1, e\cdot e_2)\,d\mathcal H^{d-2}(e)}{\left(\int_{e\cdot \vartheta\leq0}m(e)\,d\sigma(e)\right)^{\frac{d\alpha}{d-1}}}\,d\mathcal H^{d-1}(e_1,e_2,t)\,d\sigma(\vartheta)\right|\\
    =c(d)&\left|\fint_{\mathbb{S}^{d-1}}\frac{\int_{e\cdot \vartheta=0}m_x(e)\int_{\{(e_1,e_2,t)\colon \theta(e_1,e_2,t)=\vartheta, e\cdot e_2 \geq 0\}}h(e\cdot e_1,e\cdot e_2)\, d\mathcal H^{d-1}(e_1,e_2,t)\, d\mathcal H^{d-2}(e)}{\left(\int_{e\cdot \vartheta\leq0}m(e)\,d\sigma(e)\right)^{\frac{d\alpha}{d-1}}}\,d\sigma(\vartheta)\right|.
    \end{split}
\end{equation*}
Finally, by point $(2)$ of Lemma \ref{AuxLemma}, we infer that
\begin{equation*}
    \begin{split}
        \fint_{\mathbb{S}^{d-1}}\frac{\int_{e\cdot \vartheta=0}m_x(e)\, d\mathcal H^{d-2}(e)}{\left(\int_{e\cdot \vartheta\leq0}m_x(e)\,d\sigma(e)\right)^{\frac{d\alpha}{d-1}}}\,d\sigma(\vartheta)&\leq c(d)\int_{E}\int_{0}^{2\pi}\frac{|\varphi'_{e_1,e_2}(t)|}{(\varphi_{e_1,e_2}(t))^{\frac{d\alpha}{d-1}}}\,dt\,d\mathcal H^{2d-3}(e_1,e_2)\\
        &\leq \frac{c(d)}{1-\frac{d\alpha}{d-1}}2\int_{E}\max_{t\in(0,2\pi)}\,\left|\varphi_{e_1,e_2}(t)\right|^{-\frac{d\alpha}{d-1}+1}\, d\mathcal H^{2d-3}(e_1,e_2).
    \end{split}
\end{equation*}
On the other hand, using again \cite[Lemma 5.1]{cozzi2025long} together with the estimate proved in \eqref{rho/x stima Lp,2} with $p=\infty$, we have 
\begin{equation*}
    \begin{split}
        \max_{t\in (0,2\pi)}\varphi_{e_1,e_2}(t)
        \leq& \int_{\mathbb{S}^{d-1}}\int_0^{+\infty}\rho(x+er)r^{d-2}\,dr\,d\sigma(e)
        =\int_{\mathbb{R}^d} \frac{\rho(y)}{|x-y|}\,dy\leq c_dM^{1/d}.
    \end{split}
\end{equation*}
                                                                                                                                                                                           
                   
                                                               
                 
Therefore, it holds
\begin{equation}
\begin{split}
    \label{m_x estimate}
    \int_{E} \max_{(0,2\pi)}|\varphi_{e_1,e_2}(t)|^{1-\frac{d\alpha}{d-1}}\,d\mathcal H^{2d-3}(e_1,e_2)\leq  \int_{E} \left| c(d)\,M^{\frac{1}{d}}\right|^{1-\frac{d\alpha}{d-1}}d\mathcal H^{2d-3}(e_1,e_2)\leq c(d)\,M^{\frac{1}{d}-\frac{\alpha}{d-1}}.
    \end{split}
\end{equation}
Finally, using estimate \eqref{m_x estimate} in \eqref{Lip Est part 1}, we obtain
\begin{equation*}
    \left|Dv[\rho](x)+\frac{\mathrm{id}}{d}\right|\leq \frac{c(d)}{(1-\frac{d\alpha}{d-1})\kappa}\,M^{\frac{1}{d}},
\end{equation*}
which concludes the proof of Proposition \ref{proposition assumption A1}.

    \end{split}
    \end{equation*}

\end{aligned}
\]
Thus, the differential is represented

\cos t\,\mathrm{id} &

\label{derivative of distance}
    Let $p\in[1,+\infty)$ and let $(\rho_i(t))_{t\geq0}$, $i=1,2$, be locally absolutely continuous curves in time with values in $(\mathscr{P}_p(\mathbb{R}^d),W_p)$, consisting of probability measures absolutely continuous with respect to Lebesgue measure and solving
    \begin{equation*}
        \partial_t\rho_i+\nabla\cdot(\rho_i w_i)=0.
    \end{equation*}
    Then $t\mapsto W_p^p(\rho_1(t),\rho_2(t))$ is locally absolutely continuous, and the following identities hold for almost every $t$.

    If $\varphi=\varphi_t$ is a Kantorovich potential from $\rho_1(t)$ to $\rho_2(t)$ for the cost $\frac 1p|x-y|^p$, then we have 
\begin{equation}\label{diff1}
        \frac{1}{p}\frac{d}{dt}W_p^p(\rho_1(t),\rho_2(t)) =\int \nabla\varphi\cdot w_1\,d\rho_1(t)+\int\nabla\varphi^c\cdot w_2\,d\rho_2(t).
    \end{equation}
        This can be re-written with transport maps.
    If $p>1$ and $T=T_t$ is the optimal transport map from $\rho_1(t)$ to $\rho_2(t)$ for the same cost, then
    \begin{equation*}
        \frac{1}{p}\frac{d}{dt}W_p^p(\rho_1(t),\rho_2(t))
        =\int |x-T(x)|^{p-2}(x-T(x))\cdot\bigl(w_1(t,x)-w_2(t,T(x))\bigr)\,d\rho_1(t)(x),
    \end{equation*}
    where $|z|^{p-2}z$ is understood to be zero at $z=0$.

    If $p=1$, we choose $\varphi$ to be $1$-Lipschitz, so that $\varphi^c=-\varphi$; for any optimal transport map $T$ for the cost $|x-y|$, we have
    \begin{equation*}
        \frac{d}{dt}W_1(\rho_1(t),\rho_2(t))
        =\int\nabla\varphi\cdot(w_1-w_2\circ T)\,d\rho_1(t).
    \end{equation*}

The existence of a solution is proved in \cite{bonnotte2013unidimensional} together with local (in time) $L^\infty$ bounds. 
   Let $(\rho_i(t))_{t\geq 0}$ for $i=1,2$ be two solutions of the sliced Wasserstein flow \eqref{SWF eq rho_0}, and let $T>0$. Let $M$ be a constant such that $\rho_i(t,x)\leq M$ for a.e. $x$ and every $t\in [0,T]$ and $i=1,2$. 
   By Theorem \ref{Main theorem} we know that, under assumption \eqref{assumption nu} $\mathscr{F}_\nu$ is $\lambda$-geodesically convex on $K=\{\rho\,:\, \rho\leq M\}$ for some $\lambda \in \mathbb{R}$. We then use formula \eqref{diff1} and a well-known estimate due to geodesic convexity (see \cite{SantSurvey}, Lemma 4.12), to obtain
   \begin{equation*}
       \frac{d}{dt}W^2_2(\rho_1(t),\rho_2(t))\leq -2\lambda W^2_2(\rho_1(t),\rho_2(t)),
   \end{equation*}. Uniqueness then follows by Gronwall inequality
   which proves $\rho_1(t)=\rho_2(t)$ if $\rho_1(0)=\rho_2(0)$ (see \cite{ambrosio2005gradient}).

<!-- remark-8: body_md -->
We provide here an alternative proof of the uniqueness of the SWF,
which does not use Theorem \ref{Main theorem}, by computing the
variation of the distance $W_1$ between two solutions.
We assume that, for almost every $\vartheta\in\mathbb{S}^{d-1}$,
the support of $\nu^\vartheta$ is an interval $I_\vartheta$,
and that there exists $\kappa>0$, independent of $\vartheta$,
such that $\nu^\vartheta\geq \kappa$ almost everywhere on
$I_\vartheta$. The assumption on $\nu$ implies that it is compactly supported, and this uniqueness result also concerns compactly supported solutions. Let $\rho_1, \rho_2$ be compactly supported solutions of \eqref{SWF eq rho_0}. We will denote by $T,\varphi$ the optimal transport map from $\rho_1$ to $\rho_2$ and the associated Kantorovich potential, respectively. Moreover, we will denote by $T_i$ the optimal transport map from $\rho_i$ to $\nu$, and by $T_i^\vartheta$ the one from $\rho_i^\vartheta$ to $\nu^\vartheta$. By Lemma \ref{derivative of distance}, for almost every $t$,
    \begin{equation*}
    \begin{split}
        \partial_t W_1(\rho_1(t),\rho_2(t))=&\int \varphi(\partial_t \rho_1-\partial_t \rho_2)=\int \nabla \varphi\cdot(v[\rho_1]\rho_1-v[\rho_2]\rho_2)\\
        =&\int \nabla \varphi\cdot(v[\rho_1]-v[\rho_2]\circ T)\rho_1,
    \end{split}
    \end{equation*}
                                                                                                                                                                                                                  
    Then, recalling that $\text{Lip}(\varphi)\leq 1$,
    \begin{equation}
    \label{remark W1 first}
        \begin{split}
            \partial_t W_1(\rho_1(t),\rho_2(t))=&\int \nabla \varphi\,\cdot \left(v[\rho_1]-v[\rho_2]\right)d\rho_1+\int \nabla\varphi\, \cdot \left(v[\rho_2]-v[\rho_2]\circ T\right)d\rho_1\\
            \leq& \int \fint_{\mathbb{S}^{d-1}} \left| T_1^{\vartheta}(x\cdot \vartheta)-T_2^\vartheta(x\cdot \vartheta)\right|d\rho_1(x)\,d\sigma(\vartheta)+\text{Lip}(v[\rho_2])\int|T(x)-x|d\rho_1.
        \end{split}
    \end{equation}
    Thanks to the assumption $\nu^\vartheta\geq \kappa$ (which in turn implies that $N^{[-1]}_\vartheta$ is $ \kappa^{-1}-$Lipschitz continuous for every $\vartheta$), we have
    \begin{equation}
    \label{remark W1 second}
    \begin{split}
        \left|T_{\vartheta,1}(x\cdot \vartheta)-T_{\vartheta,2}(x\cdot \vartheta)\right|&=\left|N_\vartheta^{[-1]}(R_{1,\vartheta}(x\cdot \vartheta))-N_\vartheta^{[-1]}(R_{2,\vartheta}(x\cdot \vartheta))\right|\\
        &\leq \frac{1}{\kappa}  \left|R_{1,\vartheta}(x\cdot \vartheta)-R_{2,\vartheta}(x\cdot \vartheta)\right|.\\
        \end{split}
    \end{equation}
    We recall that, for any $\mu,\gamma \in \mathscr{P}_2(\mathbb{R})$ it holds
    \begin{equation}
    \label{W_1 dimensione 1}
        W_1(\mu,\gamma)=\int_0^1|R^{-1}_{\mu}(t)-R^{-1}_{\gamma}(t)|dt= \int_{\mathbb{R}}|R_\mu(t)-R_\gamma(t)|\,dt.   
    \end{equation}
    Moreover, since $\rho_1 \in L^\infty$ and it is compactly supported, we can provide an upper bound to $\rho_1^\vartheta$.
     Combining \eqref{remark W1 first}, \eqref{remark W1 second} and \eqref{W_1 dimensione 1}, together with the fact that $\rho^\vartheta_1\in L^\infty$, we obtain
    \begin{equation*}
    \begin{split}
        \partial_t W_1(\rho_1(t),\rho_2(t))\leq&\frac{1}{\kappa}\fint_{\mathbb{S}^{d-1}} \int_{\mathbb{R}} |R_{1,\vartheta}(r)-R_{2,\vartheta}(r)|\,d\rho_1^{\vartheta}(r)\,d\sigma(\vartheta)+\text{Lip}(v)W_1(\rho_1(t),\rho_2(t))\\
        \leq &\frac{C}{\kappa}W_1(\rho_1(t),\rho_2(t))
    \end{split}
    \end{equation*}
for some constant $C>0$ that is fixed once $T>0$ is. By Gronwall lemma, we deduce again the uniqueness of SWF.

<!-- remark-9: body_md -->
As shown in \cite[Section 4.1]{cozzi2025long}, if in \eqref{SWF eq rho_0} both the initial datum $\rho_0$ and the target measure $\nu$ have bounded support, then the solution $\rho_t$ also has this property for any $t\geq 0$. Under this assumption, and if the target measure $\nu$ has uniform projections (namely, $\nu^{\vartheta}(r)=\chi_{[-1/2,1/2]}(r)$ for any $\vartheta\in \mathbb{S}^{d-1}$), then the uniqueness of solutions to \eqref{SWF eq rho_0} can be proved following a strategy presented in \cite{lagoutiere2024vanishing}. Indeed, repeating verbatim the computations made in the proof of \cite[Theorem 4.1]{lagoutiere2024vanishing}, for any $\rho_1,\rho_2$ compactly supported solutions of the SWF with target measure $\nu$, and for $p\geq 1$, the following estimate holds. Here we fix a finite time interval $[0,T]$ and choose $R>0$ so that the supports of both solutions lie in $B_R$ throughout this interval; $\text{Lip}_{B_R}$ denotes the Lipschitz constant on this ball.
\begin{equation*}
              
    \frac{1}{p}\frac{d}{dt}W_p^p(\rho_1(t),\rho_2(t))
    \leq \underbrace{\text{Lip}_{B_R}(v[\rho_1])\int|T_t(x)-x|^p\,d\rho_1}_{\coloneqq I_1}+\underbrace{W_p(\rho_1,\rho_2)^{p-1}\left(\int|v[\rho_1]-v[\rho_2]|^p(T_t(x))\,d\rho_1\right)^{1/p}}_{\coloneqq I_2}.
            
\end{equation*}
Then, for $I_1$, recalling $W(z)=|z|$ and setting $a_d=\omega_{d-1}/(d\omega_d)$, we have
\begin{equation*}
\begin{split}
    \text{Lip}_{B_R}(v[\rho_1(t)])
    &\leq \frac1d+a_d\|D^2W*\rho_1(t)\|_{L^\infty(B_R)}\\
    &\leq \frac1d+a_d\|D^2W\|_{L^{p_0}(B_{2R})}\|\rho_1(t)\|_{L^{p_0'}}.
\end{split}
\end{equation*}
On the other hand, for $I_2$, following again the argument presented in \cite{lagoutiere2024vanishing}, we have that for any $q\geq 1$ it holds
\begin{equation*}
    I_2 \leq a_d W^p_p(\rho_1,\rho_2)\|D^2 W\|_{L^{pq'}(B_{2R})}\|\rho_2(t)\|^{\frac{1}{p}}_{L^q}.
\end{equation*}
Choose $1\leq p<p_0<d$ and set $q=p_0/(p_0-p)$, so that $pq'=p_0$. We deduce
\begin{equation*}
\begin{split}
    \frac1p\frac{d}{dt}W_p^p(\rho_1(t),\rho_2(t))
    &\leq I_1+I_2\\
    &\leq W_p^p(\rho_1(t),\rho_2(t))\left[\frac1d+a_d\|D^2W\|_{L^{p_0}(B_{2R})}\left(\|\rho_1(t)\|_{L^{p_0'}}+\|\rho_2(t)\|_{L^q}^{1/p}\right)\right].
\end{split}
\end{equation*}
By \eqref{W} we observe that $|D^2W|\lesssim |z|^{-1}\in L^{p_0}_{\text{loc}}$ for any $p_0<d$, hence $p_0'>d'$.
Applying again the Gronwall lemma, and thanks to Theorem \ref{stime Bonnotte Lp}, we can conclude that $\rho_1=\rho_2$.
