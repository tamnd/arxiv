<!-- definition-1: body_md -->
We say that $A$ is a \emph{$(W,*)$-algebra}, or a \emph{$W$-algebra with involution $*$}, if $A$ is a $(W,*)$-bimodule satisfying the following associative relations: for all $w\in W$, $ a_1,a_2\in A$,
\begin{align}\label{eq: compat}
		w(a_1a_2)=(wa_1)a_2, \quad
		(a_1a_2)w=a_1(a_2w), \quad
		(a_1w)a_2=a_1(wa_2).
\end{align}
<!-- end definition-1 -->
<!-- remark-1: body_md -->
\label{rk: A op}
	Given an algebra $A$, we can define the opposite algebra $A^{\operatorname{op}}$ as the underlying vector space of $A$ endowed with the opposite product $\cdot^{\operatorname{op}}$, defined by $a	\cdot^{\operatorname{op}}b=b \cdot a$, where $\cdot$ is the product of $A$.
<!-- end remark-1 -->
<!-- definition-2: body_md -->
Define $\mathcal{M}(A)$ as the subset of $ \operatorname{End}_F^{\operatorname{op}}(A) \times \operatorname{End}_F(A)$ of the pairs $(R,L)$ satisfying,
for all $a,b \in A$,
\begin{equation}\label{eq: comp L,R}
R(ab) = aR(b), \quad L(ab) = L(a)b, \quad R(a)b = aL(b).
\end{equation}
We refer to $R$ and $L$ as a \emph{right} and \emph{left multiplier} of $A$, respectively, and to the elements of $\mathcal{M}(A)$ as \emph{multipliers} of $A$.
<!-- end definition-2 -->
<!-- definition-3: body_md -->
A right multiplier $R$ and a left multiplier $L$ of $A$ are \emph{permutable} if
\begin{equation}\label{eq: permutability}
RL = LR.
\end{equation}
<!-- end definition-3 -->
<!-- example-1: body_md -->
\label{ex: non permutable}
    Let $A=Fe_{12}\oplus Fe_{13} \oplus Fe_{23}$, where the $e_{ij}$’s are the usual matrix units. Consider the multiplier  $(R,L)\in \mathcal{M}(A)$ defined by
    \begin{equation*}
        \begin{split}
            R(e_{12})=e_{12}, \quad R(e_{13})=0, \quad R(e_{23})=e_{13},\\
            L(e_{12})=e_{13}, \quad L(e_{13})=0, \quad L(e_{23})=e_{23}.
        \end{split}
    \end{equation*}
Then $R$ and $L$ are not permutable. Indeed, $ R(L(e_{12}))=R(e_{13})=0,$ whereas $L(R(e_{12}))=L(e_{12})=e_{13}.$ Notice also that $(R,L)\notin \mathcal{I}\mathcal{M}(A)$. Indeed, suppose there exists $m=\alpha e_{12} + \beta  e_{13} + \gamma e_{23}\in A$ such that $(R,L)=(R_m,L_m)$, then applying the right multiplier to $e_{12}$, we obtain $e_{12}=\gamma e_{13}$,  which is a contradiction.
<!-- end example-1 -->
<!-- proposition-1: body_md -->
\label{pro: unitary}
    Let $A$ be a unital $*$-algebra. Then $A\cong \mathcal{I}\mathcal{M}(A)= \mathcal{M}(A)$ as $*$-algebras and $R_aL_b=L_bR_a$ for every $a,b\in A$.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
From \cite[Corollary 2.5]{MarRiz}, we have that $\mathcal{I}\mathcal{M}(A)=\mathcal{M}(A)$, and moreover, $A\cong \mathcal{I}\mathcal{M}(A)$ as $F$-algebras via the isomorphism $\mu \colon A \to \mathcal{I}\mathcal{M}(A)$ defined by $\mu(m)=(R_m,L_m)$, for all $m\in A$. Since $R_{m^*}(a)=L_{m}^{*}(a)$ and $L_{m^*}(a)=R_{m}^{*}(a)$ for all $a,m\in A$, it follows that $\mu$ is an isomorphism of $*$-algebras, which completes the proof.
<!-- end proof-1 -->
<!-- definition-4: body_md -->
\label{def: equivalence}
We say that $(W,*)$-algebra structures $A^\Phi$ and
$A^{\Phi'}$ are \emph{equivalent} if $\Phi(W)=\Phi'(W)$ as $*$-subalgebras of $\mathcal{M}(A)$.
<!-- end definition-4 -->
<!-- proposition-2: body_md -->
\label{pro: unital subalgebra}
If $A^{\overline{W}}$ is $(W,*)$-algebra with unit $1_A$, then the action of $W$ on $A$ is induced by the left and right multiplication actions of a $*$-subalgebra $B$ of $A$ that is isomorphic to $\overline{W}$. Moreover, if $W$ is unital, $1_A\in B$.
<!-- end proposition-2 -->
<!-- theorem-1: body_md -->
\label{th: WM invol} Let $F$ be a field of characteristic 0.
	 Any finite dimensional $F$-algebra $A$ with involution $*$ has a decomposition of the form
     \begin{equation}\label{eq: W-M decomposition}
         A=B+J
     \end{equation}
	where the Jacobson radical $J=J(A)$ is a $*$-ideal and $B$ is a maximal semisimple $*$-subalgebra of $A$. If the field $F$ is algebraically closed, then
	$$
	B=B_1\oplus \cdots \oplus B_m,
	$$
    where the $B_i$s are $*$-simple subalgebras of $A$.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{th: WM for W*A}
 	Let $W$ be a $\symbol{35}$-algebra over a field $F$ of characteristic zero and $A$ be a finite dimensional $(W,*)$-algebra. Then the Jacobson radical $J$ of $A$ is a $(W,*)$-ideal of $A$. Moreover, if $F$ is algebraically closed, then $A=B_1\oplus \cdots \oplus B_m+J$, where $B_i$ is a $*$-simple subalgebra such that $WB_i,\, B_iW\subseteq B_i+J$ for all $1\leq i \leq m$.
<!-- end theorem-2 -->
<!-- proof-2: body_md -->
The result follows from Theorem \ref{th: WM invol} and by \cite[Theorem 3.3]{MarRiz}.
<!-- end proof-2 -->
<!-- corollary-1: body_md -->
\label{cor: *simple W*simple}
 			Let $A$ be a finite dimensional $(W,*)$-algebra over an algebraically closed field $F$ of characteristic zero. Then $A$ is $(W,*)$-simple if and only if it is $*$-simple.
<!-- end corollary-1 -->
<!-- theorem-3: body_md -->
\label{th: *-simple characterization}
            Let $F$ be an algebraically closed field of characteristic zero. Then, up to isomorphisms, all finite dimensional $*$-simple algebras are the following ones
 		\begin{itemize}
 			\item $(M_n(F),t)$, the algebra of $n \times n$ matrices with transpose involution;
 			\item $(M_{2k}(F),s)$, the algebra of $2k \times 2k$ matrices with symplectic involution;
 			\item $(M_n(F)\oplus M_n(F)^{op},ex)$, the direct sum of the algebra of $n\times n$ matrices and its opposite algebra with the exchange involution.
 		\end{itemize}
<!-- end theorem-3 -->
<!-- corollary-2: body_md -->
If $A$ is a finite dimensional $(W,*)$-simple algebra over an algebraically closed field $F$ of characteristic zero, then $A$ is isomorphic to one of the following $(W,*)$-algebras:
 			\begin{itemize}
 				\item $M_n(F)$ with transpose or symplectic involution,
 				\item $M_n(F)\oplus M_n(F)^{op}$ with exchange involution $(a,b)^*=(b,a)$,
 			\end{itemize}
 			with the $W$-action given by left and right multiplication of an appropriate unital $*$-subalgebra.
<!-- end corollary-2 -->
<!-- remark-2: body_md -->
Since $W$ is a $*$-algebra, we can consider a basis $\mathcal{B}_W$ of $W$ which is the union $\mathcal{B}_W=\mathcal{B}_{W^+}\cup \mathcal{B}_{W^-}$ of bases of $W^+$ and $W^-$, respectively, where $\mathcal{B}_{W^+}:=\{w_i^+ \, : \, i\geq 1\}$ and $\mathcal{B}_{W^-}:=\{w_i^- \, : \, i\geq 1\}$. Then, for $n \geq 1$ we can write
    \begin{align*}
        P_n^{W,*} = \operatorname{span}_F\{w_{i_0}^{\eta_0}x_{\sigma (1)}^{\epsilon_1} w_{i_1}^{\eta_1}x_{\sigma(2)}^{\epsilon_2}w_{i_2}^{\eta_2}\cdots w_{i_{n-1}}^{\eta_{n-1}}x_{\sigma(n)}^{\epsilon_n}w_{i_{n}}^{\eta_n}\, : \, \epsilon_k,\eta_l\in \{+,-\},\, w_{i_l}^{\eta_l}\in \mathcal{B}_{W^{\eta_l}},  \ \ \ &\\
         1\leq k \leq n,\, 0\leq l \leq n,\, \sigma \in S_{n}  &\ \}.
    \end{align*}
<!-- end remark-2 -->
<!-- theorem-4: body_md -->
\label{th: ss}
 Let $W$ be a unital algebra with involution over a field $F$ of characteristic zero, and $A^{\overline{W}}$ be a finite-dimensional unital
    $(W,*)$-algebra.
     If $\Bar{J}\subseteq \mathcal{I}\mathcal{M}(J(A))$, where $\Bar{J}=J(\overline{W})$ and $J(A)$ are the Jacobson radicals of $\overline{W}$ and $A$, respectively, then $A^{\overline{W}_{ss}}\in \operatorname{var}^{W,*}(A^{\overline{W}}).$
<!-- end theorem-4 -->
<!-- definition-5: body_md -->
Let $A$ be a finite-dimensional $(W,*)$-algebra such that $A=B_1 \oplus \cdots \oplus B_m + J$, where $B_1,\ldots, B_m$ are $*$-simple algebras.
 	 We say that a $*$-subalgebra $B_{i_1}\oplus \cdots \oplus B_{i_k}$, where the $B_{i_j}$'s are distinct simple $*$-algebras, is \textit{admissible} if for some permutation $(l_1,\ldots,l_k)$ of $(i_1,\ldots, i_k)$ we have that
 	 $$B_{l_1}JB_{l_2}J\cdots JB_{l_k}\neq 0.$$
<!-- end definition-5 -->
<!-- lemma-1: body_md -->
\label{lem: lower bound}
    Let $A$ be a finite dimensional $(W,*)$-algebra over an algebraically closed field $F$ of characteristic zero. Then there exist constants $C>0$ and $t$ such that
    $$
    c_n^{W,*}(A)\geq Cn^t d^n,
    $$
    where $d$ is the maximal dimension of an admissible $*$-subalgebra of $A$.
<!-- end lemma-1 -->
<!-- proof-3: body_md -->
By \cite[Theorem 5]{Gordienko2013} and \eqref{eq: lower} there exist constants $C>0$ and $t$ such that $c_n^{W,*}(A)\geq c_n^* (A) \geq Cn^t d^n$,
    where $d$ is the maximal dimension of an admissible $*$-subalgebra of $A$, as required.
<!-- end proof-3 -->
<!-- proposition-3: body_md -->
\label{pro: relatively free algebra}
Let $W\langle x_1^+, \ldots, x_n^+,x_1^-,\ldots , x_n^-\rangle $ be the free $(W,*)$-algebra in $n$ symmetric variables and $n$ skew-symmetric variables, and  $$\psi\colon W\langle x_1^+, \ldots, x_n^+,x_1^-,\ldots , x_n^-\rangle \to H$$
be the $(W,*)$-homomorphism induced by $\psi(x_i^+)=\xi_i^+$ and $\psi(x_i^-)=\xi_i^-$, $1\leq i \leq n$. Then 
$$
\ker\psi=\operatorname{Id}^{W,*}(A)\cap W\langle x_1^+, \ldots, x_n^+,x_1^-,\ldots , x_n^-\rangle.
$$

\label{lem: upper bound}

 		closed field $F$, where $W$ is a unital $F$-algebra. Then there exist constants $C$, $u$ such that

We begin by determining an upper bound for $\dim_F \mathcal{P}_{r,n-r}$.

       Consider a monomial $w_{i_0}\xi_{\sigma(1)}^{\epsilon_{\sigma(1)}} w_{i_1} \cdots w_{i_{n-1}} \xi_{\sigma(n)}^{\epsilon_{\sigma(n)}}w_{i_n}\in \mathcal{P}_{r,n-r} $, where $\sigma\in S_n$, $w_{i_l}\in \mathcal{B}_W$ for $1\leq l \leq n$, $\epsilon_i=+$ for $1\leq i \leq r$, and $\epsilon_i=-$ for $r+1\leq i \leq n$.
       Using \eqref{eq: generic elem}, such a monomial can be expressed as a linear combination of products of $n$ elements from the fixed basis $\mathcal{B}_A=\mathcal{B}^+\cup\mathcal{B}^-$ of $A$ with coefficients that are polynomials in the variables $\xi_{ij}$. More precisely, when expanded with respect to the basis $\mathcal{B}_A$, it is a linear combination of elements of the form $c\otimes \xi_{1,j_1} \cdots \xi_{n,j_{n}}$, where $c\in \mathcal{B}_A.$ Thus, we will estimate $\dim_F \mathcal{P}_{r,n-r}$ through an estimate of the number of possible monomials $\xi_{1,j_1} \cdots \xi_{n,j_{n}}$ of this last type which can appear as coefficients of $w_{i_0}\xi_{\sigma(1)}^{\epsilon_{\sigma(1)}} w_{i_1} \cdots w_{i_{n-1}} \xi_{\sigma(n)}^{\epsilon_{\sigma(n)}}w_{i_n}$.

       Since $F$ is algebraically closed of characteristic zero, by Theorem \ref{th: WM for W*A} we write $A=B_1\oplus \cdots \oplus B_m+J$, where $J$ is the Jacobson radical of $A$ and each $B_i$ is a simple $*$-algebra satisfying $WB_l,B_lW \subseteq B_l+J$ for $1\leq l \leq m$. We may further assume that the chosen basis $\mathcal{B}_A=\mathcal{B}^+\cup\mathcal{B}^-$ of $A$ is adapted to this decomposition, in the sense that both $\mathcal{B}^+$ and $\mathcal{B}^-$ consist of elements drawn from $J$ and of each of the $B_i$.
       Since each commutative variable $\xi_{i,j}$ in \eqref{eq: generic elem} is attached to an element of either $\mathcal{B}^+$ or $\mathcal{B}^-$, by a slight abuse of terminology, we therefore refer to $\xi_{i,j}$ as a symmetric radical, skew-symmetric radical, symmetric semisimple, or skew-symmetric semisimple variable.



       Take a monomial $ \xi_{1,j_1} \cdots \xi_{n,j_{n}}$ of degree $n$ that contains $i_1$ symmetric radical variables and $i_2$ skew-symmetric radical variables. Since $J$ is a nilpotent $(W,*)$-ideal of $A$, such a monomial is nonzero only if $i_1+i_2\leq u$, where $u$ is such that $J^u\neq 0$ and $J^{u+1}= 0$. 
       Now each such monomial involves $r-i_1$ symmetric semisimple variables and $n-r-i_2$ skew-symmetric semisimple variables which come from some distinct $*$-simple components $B_{l_1}, \ldots, B_{l_p}.$ Since $W B_l, B_l W\subseteq B_l+J$, there exist distinct indices $q_1, \ldots, q_v \in \{l_1,\ldots, l_p\}$, with $1\leq v\leq p$ such that $D:=B_{q_1}\oplus \cdots \oplus B_{q_v}$ is an admissible $*$-subalgebra of $A$.

       Now, fix an admissible $*$-subalgebra $D$ of $A$.

So, the number of possible monomials with symmetric and skew-symmetric semisimple variables coming from $D$ is bounded by $C_2n^u(\dim D^+)^r(\dim D^-)^{n-r}.$

        Now, let $M$ be the number of maximal admissible $*$-subalgebras of $A$ and let $E$ be an admissible $*$-subalgebra of maximum dimension. Then, an upper bound for the number of possible nonzero monomials is
 	 	$MC_2n^u(\dim E^+)^r(\dim E^-)^{n-r}.$
 	 	Since we rewrote any product of $n$ basis elements of $A$ as a linear combination of basis elements, we get that
$$
 	 	\dim_F \mathcal{P}_{r,n-r}\leq C_3n^u(\dim_F E^+)^r(\dim_F E^-)^{n-r},
 	 	$$
 	 	with $C_3$ constant. Since, by Proposition \ref{pro: relatively free algebra}, $\mathcal{P}_{r,n-r}\cong P_{r,n-r}(A)$, we have that $\dim_F \mathcal{P}_{r,n-r}=c_{r,n-r}(A).$ Thus, if $d=\dim_F E^+ + \dim_F E^-=\dim_F E,$ by \eqref{eq: decomp codim} we obtain
    $$
    c_n^{W,*}(A)= \sum_{r=0}^n\binom{n}{r}\dim_F \mathcal{P}_{r,n-r} \leq  C_3n^u \sum_{r=0}^n\binom{n}{r}(\dim_F E^+)^r(\dim_F E^-)^{n-r}=C_3n^ud^n,
 	$$ 	
    as required.

Additionally, applying the Jacobi identity, \eqref{eq: consequence w_1 3} and \eqref{eq: consequence w_1 4}, we may suppose that $l_1<l_2$. \begin{split}
     f= & \sum_{\mathcal{I}, k} \alpha_{\mathcal{I}, k} x_{i_1}^+\cdots x_{i_{m}}^+[x_{k}^+,x_{j_1}^+,\ldots, x_{j_{n-m-1}}^+] + \sum_{\mathcal{I}, k} \beta_{\mathcal{I}, k} x_{i_1}^+\cdots x_{i_{m}}^+[x_{k}^+,x_{j_1}^+,\ldots, x_{j_{n-m-1}}^+]w_1^+ \\

First, suppose that $\alpha_{\mathcal{I},0} \neq 0$ or $\beta_{\mathcal{I},0} \neq 0$ in \eqref{eq: id}  with  $ \mathcal{I}=\{1,\ldots, n\}.$ From the evaluation $x_1^+=\cdots = x_n^+=e_{11}+e_{22}+e_{33}+e_{44}$ it follows that $\alpha_{\mathcal{I},0} (e_{11}+e_{22}+e_{33}+e_{44})+ \beta_{\mathcal{I},0}(e_{11}+e_{44}-e_{22}-e_{33})=0 $, thus $\alpha_{\mathcal{I},0} =\beta_{\mathcal{I},0}=0$, a contradiction. Now, let $k\in \{1,\ldots, n\}$ and let $\mathcal{I}=\{i_1, \ldots, i_m\}$ be a proper subset of $\{1,\ldots, n\}\setminus\{k\}$ where $i_1<\cdots <i_m$ and $ 0 \leq m\leq n-2$ is the maximum such that $\alpha_{\mathcal{I},k} \neq 0$ or $\beta_{\mathcal{I},k}\neq 0$ in \eqref{eq: id}. Thus $ \alpha_{\mathcal{I}, k} -  \beta_{\mathcal{I},k}=0$ and $\alpha_{\mathcal{I}, k} +  \beta_{\mathcal{I},k}=0.$
As a consequence, it follows that $\alpha_{\mathcal{I}, k}= \beta_{\mathcal{I},k}=0$, a contradiction. By repeating this argument we obtain that $\alpha_{\mathcal{I}, k}= \beta_{\mathcal{I},k}=0$ for all $\mathcal{I}$ and $k$.

Now suppose that $\mathcal{H}=\{h_1, \ldots, h_p\}$ is a proper subset of $\{1,\ldots, n\}$ where $h_1<\cdots <h_p$ and $ 0 \leq p\leq n-2$ is the maximum such that $\gamma_{\mathcal{H}} \neq 0$ or $\delta_{\mathcal{H}}\neq 0$ in \eqref{eq: id}, and consider the evaluation $x_{h_1}^+=\cdots =x_{h_p}^+=e_{11}+e_{22}+e_{33}+e_{44},$

As a consequence, since $\operatorname{char} F=0$, we get that $\gamma_{\mathcal{H}}=  \delta_{\mathcal{H}}=0$, a contradiction. As above, by repeating this argument we obtain that $\gamma_{\mathcal{H}}=  \delta_{\mathcal{H}}=0$ for all $\mathcal{H}$. As a consequence, all the scalars appearing in $f$ are zero, i.e., $f=0.$

Thus, the polynomials in \eqref{eq: generators P_n,0} are linearly independent modulo $\operatorname{Id}^{W,*}(M^D)$ and,
as in the previous case, this implies that the $(W,*)$-polynomials in \eqref{eq: generators P_n,0} are a basis of $P_{n,0}$ modulo $P_{n,0}\cap \operatorname{Id}^{W,*}(M^D).$ Hence, by counting we get
\begin{equation}\label{eq: c_{n,0}(M^D)}
    c_{n,0}(M^D)= 2+2\sum_{m=2}^n\binom{n}{m}(m-1)+ 2\sum_{m=0}^{n-1}\binom{n}{m}=2\sum_{m=0}^n\binom{n}{m}m +2= n2^n +2.
\end{equation}

Since we have shown that $P_{r,n-r}\cap \operatorname{Id}^{W,*}(M^D) \supseteq P_{r,n-r}\cap I$  for all $n\in\mathbb{N}$ and $0\leq r\leq n$, it follows that $\operatorname{Id}^{W,*}(M^D)=I$, as claimed.



Finally, since $P_{r,n-r}\equiv 0 \pmod{(P_{r,n-r}\cap \operatorname{Id}^{W,*}(M^D))}$ for $0\leq r \leq n-2$, applying Formula~\eqref{eq: decomp codim} together with \eqref{eq: c_{n-1,1}(M^D)} and \eqref{eq: c_{n,0}(M^D)}, we obtain
\begin{align*}
 	c_n^{W,*}(M^D)=\sum_{r=0}^n\binom{n}{r}c_{r,n-r}^{W,*}(M^D)=c_{n,0}^{W,*}(M^D)+ n c_{n-1,1}^{W,*}(M^D)=n2^{n}+2+ n  2^{n}= n2^{n+1}  +2,
 	\end{align*}
    as desired.

\label{th: cocharacter} 
Let $M^D$ be the $(W,*)$-algebra $M$ where $W$ acts on it as the algebra $D$ by left and right multiplication. For $n\geq 1$ and $0\leq r \leq n$, consider 
 $$\chi_{r,n-r}^{W,*}(M^D)=\sum_{(\lambda,\mu)\vdash (r,n-r)}m_{\lambda,\mu}(\chi_\lambda \otimes \chi_\mu),$$
 the $(r,n-r)$-th $(W,*)$-cocharacter of $M^D$.  Then: 
 \begin{enumerate}
 \item for $r=n$,
$$
m_{\lambda,\mu}= \begin{cases}
             4(q+1) & \text{if }\lambda=(p+q,p), \mu=\emptyset\\
             2(q+1) & \text{if } \lambda=(p+q,p,1), \mu=\emptyset\\
              2n+2 & \text{if } \lambda=(n),  \mu=\emptyset\\
              0 & \text{otherwise}\\
    \end{cases};
$$
\vspace{1mm}
     \item for $r=n-1$,
$$
m_{\lambda,\mu}= \begin{cases}
             2(q+1)  & \text{if } \lambda=(p+q,p),  \mu=(1)\\
              2n & \text{if }  \lambda=(n-1),  \mu=(1)\\
            0 & \text{otherwise}
    \end{cases};
$$
\vspace{1mm}

\item for $0\leq r \leq n-2$, we have $m_{\lambda,\mu}=0$.
 \end{enumerate}

\begin{align*}
       \sum_{\sigma \in S_{N'}}x_{\sigma(1)}^+\cdots x_{\sigma(K')}^+x_{N'+1}^-x_{\sigma(K'+1)}^+\cdots x_{\sigma(N')}^+
        +\sum_{i<K'}\sum_{\sigma \in S_{N'}} \delta_i x_{\sigma(1)}^+\cdots x_{\sigma(i)}^+x_{N'+1}^-x_{\sigma(i+1)}^+\cdots x_{\sigma(N')}^+ &\\

    \end{align*}
Now, we multiply on the right by $y_1^+ \cdots y_{K'}^+$ where $y_i^+$ are new symmetric variables, and alternate $x_i^+$ with $y_i^+$ for all $1\leq i \leq K'$.  Since $[x_1^+,w_1^+]$ is skew-symmetric,
    $x_1^- x_2^- \in \operatorname{Id}^{W,*}(M^D)\subset\operatorname{Id}^{W,*}(\mathcal{U})$ and by Lemma \ref{Lem: consequence x_1^- x_2^-}, we can move $w_1^+$ to the right of any monomial in which it appears. Moreover, 
    by Lemma \ref{Lem: consequence x_1^- x_2^-} (see \cite[Remark 2]{MisVal}), any symmetric variables to the right (and to the left) of $x_{N'+1}^-$ can be reordered modulo $\operatorname{Id}^{W,*}(M^D)\subset\operatorname{Id}^{W,*}(\mathcal{U})$. Hence, we get

     \overline{x_1^+}\cdots \widetilde{x_{K'}^+}x_{N'+1}^-\overline{y_1^+}\cdots \widetilde{y_{K'}^+}{x_{K'+1}^+}\cdots {x_{N'}^+}\ \in \operatorname{Id}^{W,*}(\mathcal{U}).

 \end{equation}

    n-1-\lambda_1\leq N'+1}}\bar{N}n^{N'+1} \leq (N'+1)^2\bar{N}(n+1)n^{N'+1},
   \end{align*}
    since the number of partitions such that $r-\lambda_1\leq N'+1$ is bounded by $(N'+1)^2$.
    Therefore, $c^{W,*}_n(\mathcal{U})$ is polynomially bounded and $\operatorname{var}^{W,*}(M^D)$ has almost polynomial growth, as required.
