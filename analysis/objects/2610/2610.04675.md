<!-- lemma-1: body_md -->
\label{l:JohnsonHomomor}
    If $n \geq 2k$, then there exists a surjective weak homomorphism from $J(n+1, k)$ to $J(n, k)$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Define $f:V(J(n+1, k)) \to V(J(n, k))$ in the following way. For any $u \in V(J(n+1, k))$, let 
    \[ 
f(u)= \left\{
\begin{array}{ll}
      u, & n+1 \notin u, \\
      (u\setminus \{n+1\}) \cup \max([n]\setminus u),  & n+1 \in u.\\
\end{array} 
\right. 
\]
Function $f$ is clearly surjective since for any $x \in V(J(n, k)) \subseteq V(J(n+1, k))$ it holds that $f(x)=x$. Moreover, let $x,y$ be two adjacent vertices of $J(n+1, k)$. Then $|x \cap y|=k-1$ and hence either $|f(x) \cap f(y)|=k-1$ or $f(x)=f(y)$.
<!-- end proof-1 -->
<!-- theorem-2: body_md -->
\label{thm:monotonicity}
    If $n\geq 2k$, then $\gamma(J(n, k))\leq \gamma(J(n+1, k))$.
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm:strict - 2steps}
    Let $k\geq2$ and $n \geq 2k$. Then, $\gamma(J(n, k))+1\leq\gamma(J(n+2, k))$. 
                                                                                                                                           
    Moreover, if $k\geq 5$ then $\gamma(J(n, k))+2\leq\gamma(J(n+2, k))$.
<!-- end theorem-3 -->
<!-- proof-2: body_md -->
Let $D$ be a $\gamma$-set of $J(n+2, k)$ and $S\subseteq D$ such that $h_D=|S|$. Note that $h_D(S)\geq 2$. Without loss of generality assume that $\{n+1,n+2\}\subseteq\bigcap_{x\in S}x$. By maximality of $S$, every vertex $v$ in $D_S=D\setminus S$ satisfies $|v\cap\{n+1,n+2\}|\leq1$.

For any $v \in D_S$  define
    \[ 
v'= \left\{
\begin{array}{ll}
      v, &  \text{if }v \cap \{n+1,n+2\}=\emptyset, \\
      (v\setminus \{n+1\}) \cup \max([n]\setminus v),  & \text{if }n+1 \in v,\\
      (v\setminus \{n+2\}) \cup \max([n]\setminus v),  & \text{if }n+2 \in v.\\
\end{array} 
\right. 
\]
Then $D'=\{v': v \in D_S\}$ is a dominating set of $J(n, k)$. Indeed, let $x \in V(J(n, k))\setminus D'$. Then $x \in V(J(n+2, k))$ and hence there exists $y \in D$ with $|x \cap y|=k-1$. Since $n+1,n+2 \notin x$, it follows that $\{n+1,n+2\} \nsubseteq y$ and $|x \cap y'|=k-1$ (note that $|x \cap y'|=k$ is not possible, since $x \notin D'$). Therefore, $xy' \in E(J(n, k))$ and thus $D'$ is a dominating set of $J(n, k)$.

Hence, $\gamma(J(n, k))\leq |D'|=|D|-|S|=\gamma(J(n+2, k))-h_D$. As $h_D\geq 1$ for every $n\geq 2k$, it follows $\gamma(J(n, k))+1\leq \gamma(J(n+2, k))$.

Finally, assume $k\geq 5$. Notice that there exist two vertices in $D$ which are at distance at most $3$. Otherwise, let $u\in D$, and $w$ a vertex in $J(n+2, k)$ which is at distance $2$ from $u$. Since the distance between $u$ and any other vertex of $D$ is at least $4$, $w\notin D$ and has no neighbor in $D$, contradicting the fact that $D$ is a dominating set. Therefore, there exist $u,v\in D$ such that $\mathrm{dist}(u,v)=k-|u\cap v|\leq 3$, this is, $|u\cap v|\geq k-3 \geq 2$. It follows $h_D\geq 2$ and as a consequence $\gamma(J(n, k))+2\leq \gamma(J(n+2, k))$.

Using an argument similar to the one at the end of the proof of Theorem \ref{thm:strict - 2steps}, we obtain $h_D\geq 2$, and hence $\gamma(J(n, k))+2\leq\gamma(J(n+2, k))$.

<!-- theorem-4: body_md -->
\label{thm: recursive-upper-bound}
For all $m\geq 2k$ and $n\geq m+1$, we have
$$\gamma(J(n, k))\leq \gamma(J(m, k))+\sum_{i=m}^{n-1}\gamma(J(i, k-1)).$$
<!-- end theorem-4 -->
<!-- proof-3: body_md -->
We will show by induction that for $n\geq m+1$, the graph $J(n, k)$ has a dominating set of cardinality
$$f(n)=\gamma(J(m, k))+\sum_{i=m}^{n-1}\gamma(J(i, k-1)).$$

As mentioned above,
$$\gamma(J(m+1, k))\leq \gamma(J(m, k))+\gamma(J(m, k-1)),$$
so the claim holds for $n=m+1$.

Now let $n\geq m+2$ and suppose $J(n-1,k)$ has a dominating set $D_1$ with $|D_1|=f(n-1)$. Let $D_2$ be a minimum dominating set of $J(n-1,k-1)$. By Remark~\ref{rem:recursive-bound}, the set
$$D=D_1\cup \{w\cup\{n\}:w\in D_2\}$$
is a dominating set of $J(n, k)$ and satisfies
\begin{align*}
\gamma(J(n, k))\leq |D|&=\overbrace{\left(\gamma(J(m, k))+\sum_{i=m}^{n-2}\gamma(J(i, k-1))\right)}^{=|D_1|=f(n-1)}+\overbrace{\gamma(J(n-1, k-1))}^{=|D_2|}\\
&=\gamma(J(m, k))+\sum_{i=m}^{n-1}\gamma(J(i, k-1))=f(n),
\end{align*}
which completes the proof.

Recall that by Theorem~\ref{th:k=3}, for $j

\phi_j=
\begin{cases}
\frac{j^2-j+c_j}{12}, &\text{if $j$ is odd,}\\
\frac{j^2-2j+c_j}{12}, &\text{if $j$ is even,}
\end{cases}
\end{equation}
with $c_j$ as in~(\ref{eq: cn}). Note that $\phi_j$ is defined for $j\geq 6$. where $p_{k-1}$ is a polynomial in $n$ of degree $k-3$.

\gamma(J(n, k))
&\leq \gamma(J(2k, k))+\sum_{i=2k}^{n-1} \gamma(J(i, k-1))\\
&\leq \gamma(J(2k, k))+\sum_{i=2k}^{n-1}\left(\frac{1}{6}\binom{i}{k-2}-p_{k-1}(i)\right)\\
& = \sum_{i=k-2}^{n-1}\frac{1}{6}\binom{i}{k-2}-\left(\sum_{i=k-2}^{2k-1} \frac{1}{6}\binom{i}{k-2}+\sum_{i=2k}^{n-1}p_{k-1}(i)-\gamma(J(2k, k))\right)\\

\end{align*}

\begin{equation}\label{eq: p(n)}
p_k(n)=\sum_{i=2k}^{n-1}p_{k-1}(i)+\frac{1}{6}\binom{2k}{k-1}-\gamma(J(2k, k)).
\end{equation}

In this way we obtain
\[
\gamma(J(n, k))\leq \frac{1}{6}\binom{n}{k-1}-p_k(n),
\]
where $p_k$ is a polynomial in $n$ of degree $k-2$. This proves the claim.

<!-- theorem-5: body_md -->
\label{thm: monotonicity total Johnson}
    If $n\geq 2k$, then $\gamma_t(J(n, k))\leq\gamma_t(J(n+1, k))$.
<!-- end theorem-5 -->
<!-- proof-6: body_md -->
Let $D$ be a minimum total dominating set of $J(n+1, k)$, $D=D_0\cup \hat{D}$, where $D_0=\{w\in D:n+1\notin w\}$ and $\hat{D}=\{w\in D:n+1\in D\}$. 
    We index the vertices in $\hat{D}$, $\hat{D}=\{w_1,\ldots,w_r\}$, and iteratively build sets $D_j$ for $j\in [r]$ as follows. For each $j\in[r]$, let $\hat{w_j}=w_j\setminus \{n+1\}$. If for every $x\in [n]\setminus \hat{w_j}$, $\hat{w_j}\cup\{x\}\in D_{j-1}$, then $D_j:=D_{j-1}$. Otherwise, let $x$ be the largest number in $[n]\setminus \hat{w_j}$ such that $\hat{w_j}\cup\{x\}\notin D_{j-1}$. We define $w_j'=\hat{w_j}\cup\{x\}$, and $D_{j}:=D_{j-1}\cup\{w_j'\}$.
    
    Let $D'=D_r$. We will show that $D'$ is a total dominating set of $J(n, k)$. Let $v\in V(J(n, k))$. As $v\inJ(n+1, k)$, there exists $w\in D\cap N_{n+1}(v)$. 
    If $w\in D_0$, then $w\in D_0\cap N_n(v)\subseteq D'\cap N_n(v)$. Otherwise, $w=\hat{w}\cup \{n+1\}$, and $v\cap w=\hat{w}$. By the construction of $D'$, $\hat{w}\cup\{x\}\in D'$ for some $x\in [n]$.
    If for some $x\in[n]\setminus v$, $u=\hat{w}\cup\{x\}\in D'$, then $u\in D'\cap N_n(v)$. 
    Thus, assume $v$ is the only such vertex in $D'$, i.e.\ the only vertex in $D'$ that contains all elements of $\hat{w}$ is $v$. Hence $v=\hat{w}\cup\{x\}$ and by construction $x=\max([n]\setminus w)=\max([n]\setminus \hat{w})$. Besides, it follows that no vertex in $D_0$ contain $\hat{w}$.

    On the other hand, since $w$ is totally dominated by $D$, there exists $u\in D\cap N_{n+1}(w)$. If $u\in D_0$, then as $u$ and $w$ are adjacent and $n+1\notin u$, it follows $\hat{w}\subseteq u$.  which is not possible. Thus we may assume $u\notin D_0$. 
     So $u=\hat{u}\cup \{n+1\}$ with $|\hat{u}\cap \hat{w}|=k-2$. By construction of $D'$, $u'=\hat{u}\cup\{y\}\in D'$, where $y=\max([n]\setminus u)=\max([n]\setminus \hat{u})$.
   If $x=\max([n]\setminus \hat{w})=\max([n]\setminus \hat{u})=y$, then $|u'\cap v|=|(\hat{w}\cap \hat{u})\cup\{x\}|=k-1$. If $x<y$, then $y\in \hat{w}$, so $|u'\cap v|=|\hat{w}|=k-1$. Finally, if $x>y$, then $x\in \hat{u}$, so $|u'\cap v|=|\hat{u}|=k-1$. In any case, $u'\in D'\cap N_n(v)$.
    
    Therefore, $D'$ is a total dominating set of $J(n, k)$ with $|D'|\leq|D|$ and so $\gamma_t(J(n, k))\leq |D'|\leq |D|=\gamma_t(J(n+1, k))$, which completes the proof.

<!-- theorem-6: body_md -->
\label{teo: tdom Johnson r=2}
    For $n\geq 4$, $\gamma_t(J(n, 2))=\left\lceil\frac{2}{3}(n-1)\right\rceil$.
<!-- end theorem-6 -->
<!-- proof-7: body_md -->
Let $D$ be a minimum total dominating set of $J(n, 2)$.
    We define a bipartite graph $G$ with $V(G)=D\cup [n]$, where for $u\in D$ and $x\in [n]$, $u$ and $x$ are adjacent in $G$ if and only if $x\in u$. Let $X\subseteq [n]$ be the set of all $x\in [n]$ that occur in more than one vertex of $D$, and $H=G[D\cup X]$, that is, the subgraph of $G$ induced by $D\cup X$.

    Observe that if $u\in D$, since $D$ is a total dominating set, there exists $v\in D\setminus\{u\}$ such that $|u\cap v|=1$. Then, for some $x\in X$, we have $u\cap v=\{x\}$, so $\text{deg}_H(u)\geq 1$.
    Also, for each $x\in X$, we have $\text{deg}_H(x)\geq 2$.
    Therefore,
    $$2\left(\sum_{x\in X}\text{deg}_H(x)\right)=2|E(H)|=\sum_{u\in D}\text{deg}_H(u)+\sum_{x\in X}\text{deg}_H(x)\geq |D|+2|X|.$$
    Then
    $$\sum_{x\in X} (\text{deg}_H(x)-1)\geq \frac{|D|}{2}.$$
    On the other hand, let $X'=\{x\in [n]:x\in u\text{ for some }u\in D\}=\{x\in u:\text{deg}_G(x)\geq 1\}$. Note that if $x,y\in [n]\setminus X'$, then the vertex $\{x,y\}\in V(J(n, 2))$ has no neighbor in $D$, contradicting that $D$ is dominating. Therefore, there cannot be two elements outside $X'$, and so $|X'|\geq n-1$. Then
    $$2|D|=\sum_{u\in D}\text{deg}_G(u)=\sum_{x\in [n]}\text{deg}_G(x)= |X'|+\sum_{x\in X}(\text{deg}_H(x)-1)\geq n-1+\frac{|D|}{2},$$
    so
    $$|D|\geq \left\lceil\frac{2(n-1)}{3}\right\rceil.$$
    
    Now let us see that there exists a total dominating set of $J(n, 2)$ of cardinality $\left\lceil\frac{2(n-1)}{3}\right\rceil$. Let $k,\ell$ be non-negative integers such that $n=3k+\ell$ with $0\leq \ell\leq 2$. We define, for $\ell\in \{0,1\}$, the set
    $$D=\bigcup_{i=0}^{k-1}\left\{\{3i+1,3i+2\},\{3i+2,3i+3\}\right\},$$
    and for $\ell=2$, the set
    $$D=\left(\bigcup_{i=0}^{k-1}\left\{\{3i+1,3i+2\},\{3i+2,3i+3\}\right\}\right)\cup\{n-2,n-1\}.$$
    It is easy to see that $D$ is a total dominating set of $J(n, 2)$ of cardinality $\left\lceil\frac{2}{3}(n-1)\right\rceil$.

<!-- prop-1: body_md -->
\label{prop:covering-overlapping}
    Let $D\subseteq V(J(n, 3))$. Then $D$ is a total dominating set of $J(n, 3)$ if and only if $D$ is a CEOT of some graph $G$ of order $n$ with $\alpha(G)\leq 2$.
<!-- end prop-1 -->
<!-- proof-8: body_md -->
Let $D$ be a total dominating set of $J(n, 3)$. It follows that each $u\in D$ is a triangle of $G^D$, and each edge of $G^D$ belongs to at least one triangle in $D$. Thus $D$ is an edge triangle covering of $G^D$. Moreover, as $D$ is a total dominating set of $J(n, 3)$, for each $u\in D$, there exists $v\in D$ with $|u\cap v|=2$. This is, there exists a triangle $v\in D$ such that $u$ and $v$ overlap. Therefore $D$ is a CEOT of $G^D$.
        Suppose that $\alpha(G^D) \geq 3$. Then, there exist three vertices $x$, $y$ and $z$ pairwise non adjacent in $G^D$. By definition of $G^D$, $\{x,y,z\} \in V(J(n, 3))$ is not in $D$ and has no neighbor in $D$, arising to a contradiction. So $\alpha(G^D) \leq 2$. Therefore, $D$ is a CEOT of $G^D$ which has order $n$ and $\alpha(G^D)\leq 2$.

        Conversely, let $D'$ be a CEOT of a graph $G$ of order $n$ with $\alpha(G)\leq 2$, and let $u\in V(J(n, 3))$. If $u\in D'$, since the triangles in $D'$ overlap, there exists $v\in D'$ with $|u\cap v|=2$. If $u\notin D'$, then as $\alpha(G)\leq 2$, there exist $a,b\in u$ such that $ab\in E(G)$. Since $D'$ is an edge triangle covering, there exists $v\in D'$ such that $\{a,b\}\subseteq v$, so $|u\cap v|=2$. That is, in either case, there exists $v\in D'$ such that $u$ and $v$ are adjacent. Therefore, $D'$ is a total dominating set of $J(n, 3)$.

<!-- lemma-2: body_md -->
\label{lem: edges CEOT}
    Given a graph $G$, if $D$ is a CEOT of $G$, then $|E(G)|\leq \frac{5}{2}|D|$.
<!-- end lemma-2 -->
<!-- proof-9: body_md -->
Let $D$ be a CEOT of $G$, and $E'$ the set of edges of $G$ that are in more than one triangle of $D$.
    We define a bipartite graph $H$ with $V(H)=D\cup E'$, where for $t\in D$ and $e\in E'$, $t$ and $e$ are adjacent in $H$ if and only if $e$ is an edge in $t$.                                                        
    Observe that by the definition of $E'$, $\text{deg}_H(e)\geq 2$ for each $e\in E'$. Since $D$ is a CEOT, $\text{deg}_H(t)\geq 1$ for each $t\in D$. Moreover, since $H$ is bipartite with bipartition $D\cup E'$, we have $\sum_{t\in D}\text{deg}_H(t)=\sum_{e\in E'}\text{deg}_H(e)$. Then, we have
    \begin{equation}\notag
    \begin{aligned}
        2\left(\sum_{e\in E'}\text{deg}_H(e)\right)=\sum_{t\in D}\overbrace{\text{deg}_H(t)}^{\geq 1}+\sum_{e\in E'}\overbrace{\text{deg}_H(e)}^{\geq 2}\geq |D|+2|E'|. 
    \end{aligned}
    \end{equation}

    On the other hand, if $\hat{E}$ is the set of edges appearing in exactly one triangle, we have
    that
    \begin{equation}\notag
    \begin{aligned}
        3|D|&=\sum_{d\in D} |E(d)|= |\hat{E}|+\sum_{e\in E'}\text{deg}_H(e)= \overbrace{|\hat{E}|+|E'|}^{=|E(G)|}+\sum_{e\in E'}(\text{deg}_H(e)-1)\\
        &= |E(G)|+\sum_{e\in E'}(\text{deg}_H(e)-1)=|E(G)|+\underbrace{\left(\sum_{e\in E'}\text{deg}_H(e)-|E'|\right)}_{\geq \frac{|D|}{2}}\geq |E(G)|+\frac{|D|}{2}
    \end{aligned}
    \end{equation}
    Therefore, 
    $$|E(G)|\leq 3|D|-\frac{|D|}{2}=\frac{5}{2}|D|.$$

Let $n \geq 8$ and let $m\in \left[4..\left\lfloor\frac{n}{2}\right\rfloor\right]$. 
Then
\begin{equation}\label{eq: f_n(m)2}
\begin{aligned}
    f_n(m)&=\frac{m^2-m+a(m)}{5}+\frac{(n-m)^2-(n-m)+a(n-m)}{5}\\
    &=\frac{m^2-m+n^2-2nm+m^2-n+m}{5}+\frac{a(m)+a(n-m)}{5}\\
    &=\frac{n^2-n}{5}+\frac{2}{5}m(m-n)+\frac{a(m)+a(n-m)}{5}.
\end{aligned}
\end{equation}
Note that $m=\left\lfloor\frac{n}{2}\right\rfloor-t$ for some $t\in\left\{0,\ldots,\left\lfloor\frac{n}{2}\right\rfloor-4\right\}$. Then, from (\ref{eq: f_n(m)2}) we have $f_n(m)=f_n\left(\left\lfloor\frac{n}{2}\right\rfloor-t\right)$ where
\begin{equation}\notag
\begin{aligned}
    f_n\left(\left\lfloor\frac{n}{2}\right\rfloor-t\right)&=\frac{n^2-n}{5}+\frac{2}{5}\left(\left\lfloor\frac{n}{2}\right\rfloor-t\right)\left(\left\lfloor\frac{n}{2}\right\rfloor-t-n\right)+\frac{a(\left\lfloor\frac{n}{2}\right\rfloor-t)+a(n-\left\lfloor\frac{n}{2}\right\rfloor+t)}{5}\\
    &=\frac{n^2-n}{5}+\frac{2}{5}\left(\left\lfloor\frac{n}{2}\right\rfloor-t\right)\left(-\left\lceil\frac{n}{2}\right\rceil-t\right)+\frac{a(\left\lfloor\frac{n}{2}\right\rfloor-t)+a(\left\lceil\frac{n}{2}\right\rceil+t)}{5}\\
                                                                                                                                                                                                                                     
                                                                                                                                                                                                                                                                                                          
    &=\frac{n^2-n}{5}-\frac{2}{5}\left\lfloor\frac{n}{2}\right\rfloor\left\lceil\frac{n}{2}\right\rceil + \frac{1}{5}\left(2t\left(t+\left\lceil\frac{n}{2}\right\rceil-\left\lfloor\frac{n}{2}\right\rfloor\right)+a\left(\left\lfloor\frac{n}{2}\right\rfloor-t\right)+a\left(\left\lceil\frac{n}{2}\right\rceil+t\right)\right).
\end{aligned}
\end{equation}
That is, determining $m$ that minimizes $f_n(m)$ is equivalent to determining $t\in\left\{0,\ldots,\left\lfloor\frac{n}{2}\right\rfloor-4\right\}$ that minimizes

\end{equation}

<!-- lemma-4: body_md -->
\label{lema: dom total vert-disj}
     If $n\geq 10$ and $D$ is a minimum total dominating set of $J(n, 3)$, then $V(G^D)$ can be partitioned into two cliques.
<!-- end lemma-4 -->
<!-- proof-12: body_md -->
Let $n\geq 10$, let $D$ be a minimum total dominating set of $J(n, 3)$, and let $G=G^D$. From the definition of $G$, it follows that $G$ is a graph of order $n$ with $\alpha(G)\leq 2$, and that $D$ is a CEOT of $G$.

    Suppose $V(G)$ does not admit a partition into two cliques. Consequently, the graph $\overline{G}$ is not bipartite. Also, $\omega(\overline{G})=\alpha(G)\leq 2$, so $\overline{G}$ is triangle-free. By Lemma~\ref{lem:edges-trianglefree}, we obtain
    $$|E(\overline{G})|\leq \frac{(n-1)^2}{4}+1,$$
    and we have
    $$|E(G)|=\binom{n}{2}-|E(\overline{G})|\geq \frac{n(n-1)}{2}-\frac{(n-1)^2}{4}-1=\frac{n^2-5}{4}.$$
    On the other hand, by Lemma~\ref{lem: edges CEOT} and Proposition~\ref{prop: cota sup tdom varphi_n} it follows that
    $$|E(G)|\leq \frac{5}{2}\,|D|=\frac{5}{2}\,\gamma_t(J(n, 3))\leq \frac{5}{2}\,\varphi_n.$$
    Note that, from (\ref{eq: varphi cases}), we have 
    $$\varphi_n=\frac{n^2-2n+c(n)}{10},$$
    where $c(n)$ is a constant depending on $n$ with $c(n)\in\{0,1,7,10,11,12,15,16,17\}$.
    Note that
    \begin{equation}\notag
    \begin{aligned}
        \frac{n^2-5}{4}-\frac{5}{2}\,\varphi_n&=\frac{n^2-5}{4}-\frac{5}{2}\left(\frac{n^2-2n+c(n)}{10}\right)\\
        &=\frac{n^2-5}{4}-\frac{n^2-2n+c(n)}{4}=\frac{2n-5-c(n)}{4}.
    \end{aligned}
    \end{equation}
    Since from the values of $n$ and $c(n)$ we can easily deduce that $2n-5-c(n) > 0$ for any $n \geq 10$, we get
    $$\frac{n^2-5}{4}\leq |E(G)|\leq \frac{5}{2}\,\varphi_n <\frac{n^2-5}{4},$$
    and we arise to a contradiction. Therefore, we conclude that $V(G)$ admits a partition into two cliques.

<!-- theorem-8: body_md -->
\label{teo: tdom J(n,3)}
    For $n\geq 10$, it holds $\gamma_t(J(n, 3))=\varphi_n$.
<!-- end theorem-8 -->
<!-- proof-13: body_md -->
By Proposition~\ref{prop: cota sup tdom varphi_n} we know that $\gamma_t(J(n,3)) \leq \varphi_n$. To show the reversed inequality, let $n\geq 10$, let $D$ be a minimum total dominating set of $J(n, 3)$ and let $G=G^D$. By Lemma~\ref{lema: dom total vert-disj} it follows that $V(G)$ can be partitioned into two cliques $A$ and $B$, with $|A|\leq |B|$, $|A|=m$, and $|B|=n-m$. 
    
    Let $E(A,B)$ be the set of edges with one endpoint in $A$ and one endpoint in $B$. 
    If $E(A,B)=\emptyset$, then $G$ is a disjoint union of complete graphs and we obtain $|D|=\varphi_n$. Therefore, suppose $E(A,B)\neq \emptyset$. In this case, if $e=ab\in E(A,B)$ with $a\in A$ and $b\in B$, then since $D$ is an edge triangle covering, there exists $c\in [n]\setminus\{a,b\}$ such that $\{a,b,c\}\in D$. If $c\in A$, then $bc\in E(A,B)$ and if $c\in B$, then $ac\in E(A,B)$. In either case, we have $|E(A,B)|\geq 2$.

    Note that we may assume $m\neq 2$. Indeed, if $A=\{x_1,x_2\}$, since $x_1x_2\in E(G)$ and $D$ is a CEOT, there exists a triangle $T=\{x_1,x_2,b\}\in D$ for some $b\in B$. Then, considering $A'=A\cup\{b\}$ and $B'=B\setminus \{b\}$, we obtain a new partition of $V(G)$ into two cliques $A'$ and $B'$ with $|A'|= 3$ and $|B'|=n-3\geq 7$. Therefore, the partition $(A,B)$ of $V(G)$ can be chosen so that $m \neq 2$.
    
    Let us now see that $m\neq 3$.
    Note that
    \begin{equation}\notag
        \begin{aligned}
            |E(G)|&=\binom{m}{2}+\binom{n-m}{2}+|E(A,B)|\\
            &=\frac{m(m-1)}{2}+\frac{(n-m)(n-m-1)}{2}+|E(A,B)|\\
                                                         
            &=\frac{n^2-n+2m^2-2nm}{2}+|E(A,B)|=\frac{n^2-n}{2}+m(m-n)+|E(A,B)|.
        \end{aligned}
    \end{equation}
    Moreover, by Lemma~\ref{lem: edges CEOT}, it follows that
    \begin{equation}\label{eq: cota sup D}
        |D|\geq \frac{2}{5}|E(G)|=\frac{n^2-n}{5}+\frac{2}{5}m(m-n)+\frac{2}{5}|E(A,B)|
    \end{equation}
    On the other hand, let $m'=\left\lfloor\frac{n}{2}\right\rfloor$ for $n\geq 11$, and $m'=4$ for $n=10$. By Remark~\ref{obs: m-min-f_n(m)} we have $\varphi_n=f_n\left(m'\right)$, where
    $$f_n(m')=\frac{n^2-n}{5}+\frac{2}{5}m'(m'-n)+\frac{a(m')+a(n-m')}{5}.$$
    Observe that if $n=10$, then $a(m')+a(n-m')=a(4)+a(6)=3+0=3\leq 8$. If $n=11$, then $a(m')+a(n-m')=a(5)+a(6)=5+0=5\leq 8$. Finally, if $n\geq 12$, then $\left\lfloor\frac{n}{2}\right\rfloor\geq 6$ and $n-\left\lfloor\frac{n}{2}\right\rfloor\geq 6$. Therefore, $a(m')+a(n-m')\leq 4+4=8$. That is, for every $n\geq 10$ we have $a(m')+a(n-m')\leq 8$.
    
    By Proposition~\ref{prop: cota sup tdom varphi_n} we have $|D|=\gamma_t(J(n, 3))\leq\varphi_n$, then for every $n\geq 10$, we have 
    \begin{equation}\label{eq: cota inf D}
        |D|\leq\varphi_n=f_n(m')\leq \frac{n^2-n}{5}+\frac{2}{5}m'(m'-n)+\frac{8}{5}.
    \end{equation}
    From (\ref{eq: cota sup D}) and (\ref{eq: cota inf D}), it follows that
    \begin{equation}\label{eq: comp m y m'}
        \begin{aligned}
            \frac{n^2-n}{5}+\frac{2}{5}m(m-n)+\frac{2}{5}|E(A,B)|&\leq \frac{n^2-n}{5}+\frac{2}{5}m'(m'-n)+\frac{8}{5}\\
            m(m-n)+|E(A,B)|&\leq m'(m'-n)+4.
        \end{aligned}
    \end{equation}
    If $m=3$, since $|E(A,B)|\geq 2$, then
    $$3(3-n)+2\leq 3(3-n)+|E(A,B)|\leq m'(m'-n)+4,$$
    so
    \begin{equation}\label{eq: cond m'}
        m'(m'-n)-3(3-n)+2\geq 0.
    \end{equation}
    If $n=10$, then $m'=4$ and
    $$m'(m'-n)-3(3-n)+2=4(-6)-3(-7)+2=-24+21+2=-1<0.$$
    If $n\geq 11$, then $m'=\left\lfloor\frac{n}{2}\right\rfloor$, so
    \begin{equation}\notag
    \begin{aligned}
        &m'(m'-n)-3(3-n)+2=-\left\lfloor\frac{n}{2}\right\rfloor\left\lceil\frac{n}{2}\right\rceil-3(3-n)+2\leq \frac{-n^2+1}{4}-3(3-n)+2\\
        &=-\frac{1}{4}\left(n^2-1+12(3-n)-8\right)=-\frac{1}{4}\left(n^2-12n+27\right)=-\frac{1}{4}\underbrace{(n-3)}_{>0}\underbrace{(n-9)}_{>0}<0.
    \end{aligned}
    \end{equation}
    In either case, we obtain $m'(m'-n)-3(3-n)+2<0$, contradicting (\ref{eq: cond m'}). This contradiction arises from assuming $m=3$. Therefore, we have $m\notin\{2,3\}$.

    Let $T_1,\ldots, T_k$ be the triangles of $D$ having nonempty intersection with both $A$ and $B$, ordered so that for some $\ell\in\{0,1,\ldots,k\}$ we have $|T_i\cap A|=2$ if and only if $i\leq \ell$. 

    Let $D_0:=D$. We will iterate until obtaining a collection of triangles $D'$ with $|D'|\leq |D|$ such that the graph $G^{D'}$ is a disjoint union of two complete graphs. To do this, at step $i$, we will obtain a collection of triangles $D_{i}$ of $G$ such that
    \begin{enumerate}
        \item $D_i\cap \{T_1,\ldots, T_i\}=\emptyset$,
        \item $\{T_{i+1},\dots, T_k\}\subseteq D_i$, and these are the only triangles of $D_i$ having nonempty intersection with both $A$ and $B$,
        \item $|D_i|\leq |D|$, 
        \item $D_i$ covers all edges of $E(G)\setminus E(A,B)$, and
        \item\label{eq: condicion D_{i-1}} for each $T\in D_i$ such that $T\subseteq A$ or $T\subseteq B$, there exists $T'\in D_i$ such that $|T\cap T'|=2$.
    \end{enumerate}

    At step $i$ we proceed as follows. Let $D_{i-1}$ be the collection of triangles obtained at step $(i-1)$, which satisfies these 5 conditions.
    
    If $i\leq \ell$, then $T_i=\{a_1^i,a_2^i,b^i\}$ for some $a_1^i,a_2^i\in A$ and $b^i\in B$. We define $D_i$ as follows.
    \begin{itemize}
        \item If $|\{a_1^i,a_2^i\}\cap T|\leq 1$ for every $T\in D_{i-1}$ with $T\subseteq A$, then we choose $a\in A\setminus\{a_1^i,a_2^i\}$ (which exists since $|A|\neq 2$). We define $T_i'=\{a_1^i,a_2^i,a\}$ and $D_i=(D_{i-1}\setminus\{T_i\})\cup \{T_i'\}$.

        \item If there is a unique $T_0\in D_{i-1}$ with $\{a_1^i,a_2^i\}\subseteq T_0\subseteq A$, then we choose $a\in A\setminus T_0$ (which exists since $|A|\neq 3$). We again define $T_i'=\{a_1^i,a_2^i,a\}$ and $D_i=(D_{i-1}\setminus\{T_i\})\cup \{T_i'\}$.

        \item If there exist two triangles $T_0,S_0\in D_{i-1}$ such that $\{a_1^i,a_2^i\}\subseteq T_0\subseteq A$ and $\{a_1^i,a_2^i\}\subseteq S_0\subseteq A$, then we define $D_i=D_{i-1}\setminus\{T_i\}$.
    \end{itemize}
    Observe that the triangles $T_i'$ defined in the first two cases do not belong to $D_{i-1}$.

    Note that the only edge of $E(G)\setminus E(A,B)$ covered by $T_i$ is $a_1^ia_2^i$. In the first two cases, we see that this edge is also covered by $T_i'$, while in the last case, there exist at least two triangles in $D_{i-1}\setminus\{T_i\} \subseteq D_i$ that cover it. In either case, $D_i$ covers all edges of $E(G)\setminus E(A,B)$. 
    Moreover, the only triangles in $D_i$ having nonempty intersection with both $A$ and $B$ are the triangles $T_j$ for $j>i$.

    Finally, let $T\in D_i$ be such that $T\subseteq A$ or $T\subseteq B$. Let us see that there exists $T'\in D_i$ such that $|T\cap T'|=2$.
    First, if $T\in D_{i-1}\setminus\{T_i\}$, since (\ref{eq: condicion D_{i-1}}) holds we know there exists $T'\in D_{i-1}$ such that $|T\cap T'|=2$. If $T'\neq T_i$, then $T'\in D_i$, and we have found $T'\in D_i$ such that $|T\cap T'|=2$. If $T'=T_i$, note that since $T\subseteq A$ or $T\subseteq B$, we must have $T\cap T'=\{a_1^i,a_2^i\}$. If $T'$ is the only triangle in $D_{i-1}$ such that $\{a_1^i,a_2^i\}\subseteq T'\subseteq A$, then by how we defined $D_i$, $T_i'\in D_i$ also satisfies $|T\cap T_i'|=2$. Otherwise, that is, if there is another triangle $T''\in D_{i-1}$ such that $\{a_1^i,a_2^i\}\subseteq T''\subseteq A$, then $T''\in D_i=D_{i-1}\setminus\{T_i\}$, and $|T\cap T''|=|\{a_1^i,a_2^i\}|=2$ holds.
    If $T=T_i'=\{a_1^i,a_2^i,a\}$, then since edge $a_1^ia$ is covered by some triangle $T'\in D_{i-1}\setminus\{T_i\}\subseteq D_i$, we find $T'\in D_i$ such that $|T\cap T'|=|\{a_1^i,a\}|=2$.

    That is, we obtain a collection of triangles $D_i$ satisfying the 5 mentioned conditions.
    If $i>\ell$ then $T_i=\{a^i,b_1^i,b_2^i\}$ for some $a^i\in A$ and $b_1^i,b_2^i\in B$. We proceed analogously to what was done for $i\leq \ell$ and obtain a collection of triangles $D_i$ that also satisfies the 5 conditions.

    Let now $D'=D_k$. We have $D'\cap \{T_1,\ldots, T_k\}=\emptyset$ and for each $T\in D'$, either $T\subseteq A$ or $T\subseteq B$. Moreover $D'$ covers all edges in $E(G)\setminus E(A,B)$. That is, $D'$ is an edge triangle covering of the graph $G'$, the disjoint union of the complete graphs $G[A]$ and $G[B]$. Moreover, since condition (\ref{eq: condicion D_{i-1}}) holds, $D'$ is a CEOT of $G'$. As $\alpha(G')=2$, it follows from Proposition~\ref{prop:covering-overlapping} that $D'$ is a total dominating set of $J(n,3)$ and thus $\gamma_t(J(n,3)) \leq |D'| \leq |D|= \gamma_t(J(n,3))$. Since the definition of $\varphi_n$ implies that $$\varphi_n \leq \rho_{\Delta}^s(K_{|A|})+\rho_{\Delta}^s(K_{|B|}) \leq |D'|=\gamma_t(J(n,3)),$$ we get $\gamma_t(J(n,3)) \geq \varphi_n$ as desired.

<!-- prop-5: body_md -->
\label{prop:hypergraph}
    Let $D \subseteq V(J(n, k))$ and let $\mathcal{F}$ be the family of all $(k-1)$-uniform hypergraphs $G$ of order $n$ with $\alpha(G) \leq k-1$. Then $D$ is a dominating set of $J(n, k)$ if and only if $D$ is a covering of hyperedges of a hypergraph $H$ by complete hypergraphs of order $k$ of some $H\in \mathcal{F}$.
<!-- end prop-5 -->
<!-- proof-14: body_md -->
Let $H$ be an $(k-1)$-uniform hypergraph of order $n$ with $V(H)=[n]$. For a family $D$ of $k$-element subsets of $[n]$ we define a hypergraph $H^D$ as follows: $V(G^D)=[n]$ and for each $u \in D$ every $(k-1)$-subset of $u$ is an hyperedge of $H^D$, i.e.\ $u$ induces a complete subhypergraph in $H^D$. Note that $H^D$ is $(k-1)$-uniform hypergraph. 

Let $D$ be a dominating set of $J(n, k)$ and let $\{x_1,\ldots ,x_{k-1}\}$ be an arbitrary hyperedge of $H^D$. Definition of $H^D$ implies that there exists $a \in [n]$ such that $x=\{x_1,\ldots ,x_{k-1},a\} \in D$. Hence, any $(k-1)$-subset of $x$ is an hyperedge of $G^D$ and thus the set $\{x_1,\ldots ,x_{k-1},a\}$ induces a complete subhypergraph of $H^D$. Therefore, every hyperedge of $H^D$ is covered with a complete subhypergraph of order $k$. It is also clear that $\alpha(H^D) \leq k-1$. (Note that a set of vertices $A$ of a hypergraph is an independent set if no hyperedge of the hypergraph is contained in $A$). Indeed, if there exists an independent set $A=\{x_1,\ldots ,x_k\}$ of cardinality $k$, then no hyperedge of $H^D$ is contained in $A$. This implies that for any $z=\{z_1,\ldots ,z_{k-1}\} \subseteq A$, it holds that $z \notin E(H^D)$. Hence, vertex $u=A \in V(J(n, k))$ is not dominated by $D$, a contradiction.

Conversely, let $H$ be an $(k-1)$-uniform hypergraph of order $n$ with $\alpha(H)\leq k-1$, where each edge of $H$ is contained in a complete subhypergraph of $H$ of order $k$. Let $D'$ be a covering of hyperedges of $H$ with complete subhypergraphs of $H$ of order $k$. Let $u=\{u_1,\ldots ,u_k\} \in V(J(n, k))\setminus D'$. Since $\alpha(H) \leq k-1$, $u$ is not an independent set of $H$. Hence, there exists an hyperedge $e$ of $H$ that is contained in $u$. Note that $|e|=k-1$. Since $D'$ is a covering of hyperedges of $H$, there exists $v \in D'$ such that $e \subseteq v$. Hence $u=v$ or $uv \in E(J(n, k))$, and consequently $D'$ is a dominating set of $J(n, k)$.
