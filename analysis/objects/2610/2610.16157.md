\label{thm:upper}
		Let $U\Subset M$ be open, $w\in C_c^\infty(U)$, and
		$e_\lambda\in C^\infty(U)\cap L^2(U)$ satisfy, for $\lambda\ge1$,
		\[
		-\Delta_g e_\lambda=\lambda^2e_\lambda\quad\text{on }U,
		\qquad \|e_\lambda\|_{L^2(U)}=1.
		\]
		The following bounds hold.
		
		\textup{(i)} Without a curvature assumption,
		\begin{equation}\label{eq:uniform-bound}
			\left|\int_Uw e_\lambda^3\,d V_g\right|
			\le C_{g,w,U}.
		\end{equation}
		
		\textup{(ii)} If $K_g\ne0$ almost everywhere on $\{w\ne0\}$, then as $\lambda\to\infty$
		\begin{equation}\label{eq:upper}
			\left|\int_Uw e_\lambda^3\,d V_g\right|=o(1).
		\end{equation}
		
		\textup{(iii)}  If $\sum_{j=0}^m|\nabla^jK_g(x)|_g\ge\kappa$ on $\operatorname{supp} w$,
		where $m\ge0$ is an integer and $\kappa>0$, then
		\begin{equation}\label{eq:finite-order-corollary}
			\left|\int_Uw e_\lambda^3\,d V_g\right|
			\le C_{g,w,U,m,\kappa}\lambda^{-\frac1{6+2m}}.
		\end{equation}
		The same bounds hold with $e_\lambda^3$ replaced by $u_1u_2u_3$,
		where each $u_j\in C^\infty(U)\cap L^2(U)$ satisfies
		$-\Delta_gu_j=\lambda^2u_j$ on $U$ and
		$\|u_j\|_{L^2(U)}=1$. If $M$ is closed, we may take $U=M$ and
		$w\in C^\infty(M)$.

<!-- corollary-1: body_md -->
\label{cor:constant-curvature}
		Let $(M,g)$ be a closed Riemannian surface with everywhere positive
		or everywhere negative Gaussian curvature. If
		$e_\lambda\in C^\infty(M)$ satisfies
		$-\Delta_g e_\lambda=\lambda^2e_\lambda$ on $M$, $\lambda\ge1$,
		and $\|e_\lambda\|_{L^2(M)}=1$, then
		\begin{equation}
			\left|\int_M e_\lambda^3\,d V_g\right|\le C_{M,g}\lambda^{-\frac16}.
		\end{equation}
		In particular, this holds for the standard round sphere $S^2$ and
		all closed hyperbolic surfaces.

\label{thm:global}
		For every $s_0>0$ there exists $C_{s_0}>0$ such that the
		following holds. Let $(M,g)$ be a complete hyperbolic surface
		without boundary with $\operatorname{sys}(M)\ge s_0$.
		Every $L^2$-normalized eigenfunction
		$e_\lambda\in C^\infty(M)\cap L^2(M)$ satisfying
		$-\Delta_g e_\lambda=\lambda^2e_\lambda$, $\lambda\ge2$,
		belongs to $L^3(M)$. Moreover, for every
		$w\in C^\infty(M)$ with $\|w\|_{C_b^2(M)}<\infty$,
		\begin{equation}\label{eq:global}
			\left|\int_M w e_\lambda^3\,d V_g\right|
			\le C_{s_0}\|w\|_{C_b^2(M)}
			(\log\lambda)^{-3/2}.
		\end{equation}
		The same bound holds with $e_\lambda^3$ replaced by
		$u_1u_2u_3$, where $u_1,u_2,u_3$ are any three such
		$L^2$-normalized eigenfunctions with eigenvalue $\lambda^2$.

<!-- remark-1: body_md -->
\label{rem:cusp-free}
		By Patterson's theorem \cite{Patterson1975}, geometrically finite hyperbolic surfaces of infinite area have no $L^2$ eigenvalues in $[1/4,\infty)$. In particular, noncompact geometrically finite cusp-free surfaces admit no high-frequency $L^2$ eigenfunctions. For geometrically infinite cusp-free surfaces, the existence of such eigenfunctions remains open in general \cite[Introduction]{WeichWolf}. Nevertheless, if a complete cusp-free surface with $\operatorname{sys}(M)\ge s_0$ admits an $L^2$-normalized eigenfunction $e_\lambda$ as in Theorem~\ref{thm:global}, then \eqref{eq:global-shallow}, with $w_H=1$, yields the stronger estimate
		\[
		\left|\int_M w e_\lambda^3\,d V_g\right|
		\le C_{s_0}\|w\|_{C_b^2(M)}\lambda^{-1/6}.
		\]
		The same bound holds with $e_\lambda^3$ replaced by $u_1u_2u_3$, where the $u_j$ are any three such $L^2$-normalized eigenfunctions with eigenvalue $\lambda^2$.

		and equal to one where $d_g(y,\operatorname{supp} w)\le6\rho_0$, and define

		V_\lambda(y)=\chi_1(y)\lambda^{1/2}
		\int e^{i\lambda d_g(y,Y(s))}f(s)\,d s.

For fixed $x$, let
		\[
		\begin{gathered}
			y_\rho(\tau,\sigma)=\exp_x(\rho\tau v(x,\sigma)),\qquad 1<\tau<2,\\
			D_\rho(\tau,\sigma,s)=
			\frac{\rho\tau+d_g(y_\rho(\tau,\sigma),Y(s))-d_g(x,Y(s))}{\rho}.
		\end{gathered}
		\]
		Gauss's lemma and Taylor's formula give, uniformly in $x,\tau,\sigma,s$, 
		\[
		\begin{gathered}
			dV_g(y_\rho)=\rho|\partial_\sigma y_\rho|_g\,d\tau\,d\sigma,
			\qquad \rho^{-1}|\partial_\sigma y_\rho|_g\approx1,\\
			D_\rho=\tau(1-\langle v(x,\sigma),v(x,s)\rangle_g)
			+O_{C^N}(\rho).
		\end{gathered}
		\]
		The triangle inequality is an equality at $\sigma=s$. Thus
		\[
		D_\rho(\tau,s,s)=\partial_\sigma D_\rho(\tau,s,s)=0,\qquad
		\partial_\sigma^2D_\rho(\tau,s,s)
		=\tau|\partial_sv(x,s)|_g^2+O(\rho)\ge c.
		\]
		Take $\eta_1\in C_c^\infty((1,2))$ with
		$\int\eta_1=1$, and define $b_{\lambda,\rho}$ by
		\[
		b_{\lambda,\rho}(x,y_\rho(\tau,s))
		\frac{|\partial_sy_\rho(\tau,s)|_g}{\rho}
		=\frac{e^{-i\pi/4}}{\sqrt{2\pi}}\eta_1(\tau)a_\lambda(x,s)
		\sqrt{\partial_\sigma^2D_\rho(\tau,s,s)}.
		\]
		Extend it by zero. The map
		$(\tau,\sigma)\mapsto(y_\rho(\tau,\sigma)-x)/\rho$ and its inverse
		have uniform smooth bounds, giving the asserted derivative bounds.
		On sufficiently small fixed intervals, $\partial_\sigma^2D_\rho\ge c$.
		One-dimensional stationary phase
		\cite[Corollary~1.1.8 and (1.1.20)]{Sogge} gives
		\[
		\begin{aligned}
			&\int e^{i\lambda\rho D_\rho}
			b_{\lambda,\rho}(x,y_\rho(\tau,\sigma))
			\frac{|\partial_\sigma y_\rho(\tau,\sigma)|_g}{\rho}\,d\sigma\\
			&\quad=(\lambda\rho)^{-1/2}\eta_1(\tau)a_\lambda(x,s)
			+O((\lambda\rho)^{-3/2}).
		\end{aligned}
		\]
		The error is uniform in $x,\tau,s$. Since $\int\eta_1=1$,
		\[
		\begin{aligned}
			&\lambda\rho^{-3/2}\int
			e^{i\lambda[d_g(x,y)+d_g(y,Y(s))]}b_{\lambda,\rho}(x,y)\,d V_g(y)\\
			&\quad=\lambda^{1/2}e^{i\lambda d_g(x,Y(s))}a_\lambda(x,s)
			+O(\lambda^{-1/2}\rho^{-1}),
		\end{aligned}
		\]
		where $\lambda\rho^{-3/2}\rho^2(\lambda\rho)^{-1/2}=\lambda^{1/2}$.
		For $x$ near $\operatorname{supp} w$, $\chi_1=1$ on the integration set.
		Integrate against $f(s)$ and use $\|f\|_1\le C\|f\|_2$.

\label{thm:pan-sogge}
		Let $\phi(x,y)$ be real and smooth, $x,y\in\mathbb R^2$, and let
		$a\in C_c^\infty(\mathbb R^2\times\mathbb R^2)$. Let $h=\det\phi_{xy}$.
		Suppose $|\phi_{x_1y_1}|\ge c>0$ and
		\[
		\left|\left(\partial_{y_2}
		-\frac{\phi_{x_1y_2}}{\phi_{x_1y_1}}\partial_{y_1}\right)h\right|\ge c,
		\qquad
		\left|\left(\partial_{x_2}
		-\frac{\phi_{x_2y_1}}{\phi_{x_1y_1}}\partial_{x_1}\right)h\right|\ge c
		\]
		on a fixed neighborhood of $\operatorname{supp} a$. Then, for $\mu\ge1$,
		\[
		\left\|\int_{\mathbb R^2}e^{i\mu\phi(x,y)}a(x,y)f(y)\,d y\right\|_{L^2(dx)}
		\le C\mu^{-5/6}\|f\|_{L^2(dy)}.
		\]
		The constant depends only on $c$, the fixed support, and finitely many
		smooth bounds for $\phi$ and $a$.

\label{lem:trilinear}
		On a sufficiently small fixed product of intervals, suppose
		\[
		\begin{gathered}
			\Phi_\varepsilon(s)=H(s_1+s_2+s_3)+\varepsilon\Psi_\varepsilon(s),
			\qquad |H''|\ge c,\\
			\left|(\Phi_{23}\partial_1-\Phi_{13}\partial_2)q_\Phi\right|
			\ge c\varepsilon,\qquad \Phi=\Phi_\varepsilon,
		\end{gathered}
		\]
		where $H,\Psi_\varepsilon$ are real with uniform smooth bounds.
		For $\lambda\ge1$, $0<\varepsilon\le\varepsilon_0$, with $\varepsilon_0$
		fixed and sufficiently small, smooth amplitudes $a_\lambda$ of fixed
		compact support and uniform smooth bounds, and $f_j\in L^2(\mathbb R)$ supported
		in the corresponding intervals,
		\begin{equation}\label{eq:uniform-trilinear}
			\left|\int e^{i\lambda\Phi_\varepsilon(s)}a_\lambda(s)
			\prod_{j=1}^3 f_j(s_j)\,d s\right|
			\le C\lambda^{-1/2}(1+\lambda\varepsilon)^{-1/6}\prod_{j=1}^3\|f_j\|_2.

<!-- proof-3: body_md -->
If $\lambda\varepsilon\le1$, apply \eqref{eq:nondegenerate-L2} in
		$(s_1,s_2)$ for fixed $s_3$ and use $\|f_3\|_1\le C\|f_3\|_2$.
		Assume $\lambda\varepsilon\ge1$ and write $s=(x,y,z)$, $\Phi=\Phi_\varepsilon$.
		Define
		\[
		\begin{aligned}
			K_\lambda(x,\eta,u,y)
			&=\int e^{i\lambda[\Phi(x,y,z)-\Phi(u+y,\eta,z)]}
			a_\lambda(x,y,z)\overline{a_\lambda(u+y,\eta,z)}\,d z,\\
			\mathcal K_\lambda h(x,\eta)
			&=\iint K_\lambda(x,\eta,u,y)h(u,y)\,d u\,d y.
		\end{aligned}
		\]
		Cauchy--Schwarz in $z$, followed by expansion of the square and the
		substitution $u=\xi-y$, gives
		\begin{equation}\label{eq:crossed-identity}
			\begin{aligned}
				\left|\int e^{i\lambda\Phi(s)}a_\lambda(s)
				\prod_{j=1}^3f_j(s_j)\,d s\right|^2&\le\|f_3\|_2^2\iint\mathcal K_\lambda
				[f_2(y)\overline{f_1(u+y)}](x,\eta)
				f_1(x)\overline{f_2(\eta)}\,d x\,d\eta\\
				&\le\|\mathcal K_\lambda\|_{L^2\to L^2}\prod_{j=1}^3\|f_j\|_2^2.
			\end{aligned}
		\end{equation}
		Take the unitary Fourier transform in $x$ and multiply by
		$e^{i\lambda p\eta}$, obtaining the kernel
		\[
		\widehat K_\lambda(p,\eta,u,y)
		=\left(\frac\lambda{2\pi}\right)^{1/2}
		\int e^{-i\lambda p(x-\eta)}K_\lambda(x,\eta,u,y)\,d x.
		\]
		A tilde denotes evaluation at $(u+y,\eta,z)$. The critical point equations in $x,z$ and the Hessian in these
		variables are
		\[
		\Phi_1=p,\qquad \Phi_3=\widetilde\Phi_3,\qquad
		H_*=\begin{pmatrix}\Phi_{11}&\Phi_{13}\\
			\Phi_{13}&\Phi_{33}-\widetilde\Phi_{33}\end{pmatrix},
		\qquad h_*=\det H_*.
		\]
		On $y=\eta$, $x=u+\eta$ and $h_*=-\Phi_{13}^2$.
		On fixed smaller parameter sets, $h_*\le-c$ and the stationary point
		$(x_*,z_*)$ is unique and smooth. Let
		\[
		\theta_\varepsilon(p,\eta,u,y)
		=\left[\Phi(x,y,z)-\Phi(u+y,\eta,z)-p(x-\eta)\right]_{(x,z)=(x_*,z_*)}.
		\]
		Stationary phase \cite[Corollary~1.1.8 and (1.1.20)]{Sogge} on a fixed
		enlargement of the amplitude support, with the amplitude extended by zero, gives
		\begin{equation}\label{eq:transformed-kernel}
			\widehat K_\lambda
			=\lambda^{-1/2}e^{i\lambda\theta_\varepsilon}B_\lambda+r_\lambda,
			\qquad \|B_\lambda\|_{C^N}\le C_N,\qquad
			\|r_\lambda\|_{L^2(dp\,d\eta\,du\,dy)}\le C\lambda^{-3/2}.
		\end{equation}
		Here $B_\lambda$ has fixed compact support. On the complementary region, where the phase is uniformly nonstationary,
		integration by parts gives rapid decay in $\lambda$ and $1+|p|$.
		
		In the following identities, evaluate untilded derivatives at
		$(x_*,y,z_*)$ and tilded derivatives at $(u+y,\eta,z_*)$. Since
		\[
		\begin{gathered}
			\partial_p\theta_\varepsilon=-x_*+\eta,\qquad
			\partial_\eta\theta_\varepsilon=-\widetilde\Phi_2+p,\\
			H_*\binom{\partial_u x_*}{\partial_u z_*}
			=\binom0{\widetilde\Phi_{13}},\qquad
			H_*\binom{\partial_y x_*}{\partial_y z_*}
			=\binom{-\Phi_{12}}{\widetilde\Phi_{13}-\Phi_{23}},
		\end{gathered}
		\]
		the matrix $\mathcal M=\partial_{(p,\eta)}\partial_{(u,y)}\theta_\varepsilon$
		satisfies
		\begin{equation}\label{eq:crossed-determinant}
			\begin{gathered}
				\mathcal M_{11}=\frac{\Phi_{13}\widetilde\Phi_{13}}{h_*},\qquad
				\det\mathcal M=\frac{\Phi_{12}\widetilde\Phi_{12}}{h_*}
				(q_\Phi-\widetilde q_\Phi),\\
				\left.\partial_y\det\mathcal M\right|_{y=\eta}
				=\frac{\Phi_{12}^2}{\Phi_{13}^3}
				(\Phi_{23}\partial_1-\Phi_{13}\partial_2)q_\Phi.
			\end{gathered}
		\end{equation}
		For the last identity, use $\partial_yx_*=1-\Phi_{23}/\Phi_{13}$ on
		$y=\eta$. The $\partial_yz_*$ terms in
		$\partial_y(q_\Phi-\widetilde q_\Phi)$ cancel.
		
		At $\varepsilon=0$, $\theta_0=-pu$. Therefore, on fixed smaller sets,
		\begin{equation}\label{eq:crossed-normal-form}
			\begin{gathered}
				\theta_\varepsilon=-pu+\varepsilon\psi_\varepsilon,\qquad
				\det\mathcal M=\varepsilon(y-\eta)e_\varepsilon,\\
				|e_\varepsilon|\ge c,\qquad
				\|\psi_\varepsilon\|_{C^N}+\|e_\varepsilon\|_{C^N}\le C_N.
			\end{gathered}
		\end{equation}
		On $y=\eta$, the determinant vanishes and
		\eqref{eq:crossed-determinant} gives
		$|\partial_y(\varepsilon^{-1}\det\mathcal M)|\ge c$.
		
		Set $\mu=\lambda\varepsilon$. Choose
		$\vartheta\in C_c^\infty((-2,2))$ with
		$\sum_{k\in\mathbb Z}\vartheta(p/\varepsilon-k)=1$, and let
		$\mathcal K_{\lambda,k}$ have kernel
		\[
		\lambda^{-1/2}\vartheta(p/\varepsilon-k)
		e^{i\lambda\theta_\varepsilon(p,\eta,u,y)}B_\lambda(p,\eta,u,y).
		\]
		By \eqref{eq:transformed-kernel} and unitarity, it remains to estimate
		$\sum_k\mathcal K_{\lambda,k}$, up to an $O(\lambda^{-3/2})$ error in
		$L^2\to L^2$.
		In the $k$th term substitute $p=\varepsilon(k+p')$. After multiplying
		$h(u,y)$ by $e^{-i\mu ku}$, the phase at frequency $\mu$ is
		\[
		\Theta_k(p',\eta,u,y)=-p'u+\psi_\varepsilon(\varepsilon(k+p'),\eta,u,y),
		\qquad |p'|<2.
		\]
		The phase and amplitude have uniform smooth bounds. Let
		$h_k=\det(\Theta_k)_{(p',\eta),(u,y)}$:
		\[
		|(\Theta_k)_{p'u}|\ge\tfrac12,\qquad
		h_k=(y-\eta)e_\varepsilon(\varepsilon(k+p'),\eta,u,y).
		\]
		Writing $e_\varepsilon$ for its value at $(\varepsilon(k+p'),\eta,u,y)$, we obtain
		\[
		\begin{aligned}
			\left(\partial_y-\frac{(\Theta_k)_{p'y}}{(\Theta_k)_{p'u}}\partial_u\right)h_k
			&=e_\varepsilon+O(|y-\eta|),\\
			\left(\partial_\eta-\frac{(\Theta_k)_{\eta u}}{(\Theta_k)_{p'u}}\partial_{p'}\right)h_k
			&=-e_\varepsilon+O(|y-\eta|).
		\end{aligned}
		\]
		Since $|e_\varepsilon|\ge c$, both absolute values are at least $c/2$
		for small fixed $|y-\eta|$, uniformly in $k,\varepsilon$.
		Theorem~\ref{thm:pan-sogge} and $\,d p=\varepsilon\,d p'$ give
		\[
		\|\mathcal K_{\lambda,k}\|_{L^2\to L^2}
		\le C\lambda^{-1/2}\varepsilon^{1/2}\mu^{-5/6}
		=C\lambda^{-1}\mu^{-1/3}.
		\]
		For $|p/\varepsilon-k|,|\widetilde p/\varepsilon-\ell|<2$ and
		$|k-\ell|>C_0$,
		\[
		\bigl|\partial_u\{\lambda[\theta_\varepsilon(p,\eta,u,y)
		-\theta_\varepsilon(\widetilde p,\widetilde\eta,u,y)]\}\bigr|
		\ge c\mu|k-\ell|.
		\]
		Three integrations by parts in $u$ bound the kernel of
		$\mathcal K_{\lambda,k}\mathcal K_{\lambda,\ell}^*$ by
		$C\lambda^{-1}\mu^{-3}(1+|k-\ell|)^{-3}$.
		The $p,\widetilde p$ supports have lengths $O(\varepsilon)$.
		Schur's test and the preceding norm bound therefore give, for all $k,\ell$,
		\[
		\|\mathcal K_{\lambda,k}\mathcal K_{\lambda,\ell}^*\|_{L^2\to L^2}
		\le C\lambda^{-2}\mu^{-2/3}(1+|k-\ell|)^{-2}.
		\]
		Bounded overlap of the $p$ intervals and Schur's test on this matrix yield
		\[
		\|\mathcal K_\lambda\|_{L^2\to L^2}
		\le C\left(\sup_k\sum_\ell
		\|\mathcal K_{\lambda,k}\mathcal K_{\lambda,\ell}^*\|_{L^2\to L^2}\right)^{1/2}
		+C\lambda^{-3/2}
		\le C\lambda^{-1}\mu^{-1/3}.
		\]
		Taking the square root in \eqref{eq:crossed-identity} proves the lemma.

Let $I$ be the Euclidean metric and $J_g=\sqrt{\det g}$.
	The radial Jacobi equation and Gauss's lemma
	\cite[Section~3.2.4 and Example~3.2.12]{Petersen} give
	\[
	\begin{gathered}
		J_g(x)-1=-\varepsilon|x|^2\int_0^1t(1-t)F(tx)J_g(tx)\,d t,\\
		g-I=(J_g+1)\frac{J_g-1}{|x|^2}(|x|^2I-x\otimes x).
	\end{gathered}
	\]
	The quotient extends smoothly to $0$, and hence
	\begin{equation}\label{eq:flat-comparison}
		\|g-I\|_{C^N}\le C_N\varepsilon.
	\end{equation}
	Since $|\nabla_x\phi_j|_g=1$, there are smooth $\alpha_j,\beta_j$ with
	\[
	\nabla_x^2\phi_j=\alpha_jn_j\otimes n_j,\qquad
	\partial_{s_j}v_j=\beta_jn_j,\qquad c\le\alpha_j,\beta_j\le C.
	\]
	Here the bounds follow from \eqref{eq:flat-comparison}.
	At $(x,s)=(0,0)$, $v_j=v_j^0$ and
	$\nabla_x^2(\phi_1+\phi_2+\phi_3)\ge cI$.
	The implicit function theorem therefore gives a unique smooth $x_c(s)$ near $0$ with
	\begin{equation}\label{eq:reduced-phase}
		\left.\nabla_x(\phi_1+\phi_2+\phi_3)\right|_{x=x_c(s)}=0,
		\qquad x_c(0)=0.
	\end{equation}
	Thus $x_c(s)$ is the local Fermat point of $Y_1(s_1),Y_2(s_2),Y_3(s_3)$:
	it minimizes $\phi_1+\phi_2+\phi_3$ near $0$, and the three joining
	geodesics meet there at angles of $2\pi/3$.
	
	Define
	$\Phi(s)=\phi_1(x_c(s),s_1)+\phi_2(x_c(s),s_2)+\phi_3(x_c(s),s_3)$.
	In formulas involving only $s$, evaluate $v_j,n_j,\alpha_j,\beta_j$
	at $(x_c(s),s_j)$. Since $v_1+v_2+v_3=0$, we may interchange $1,2$ so that
	\[
	n_1=-\frac{\sqrt3}{2}v_3-\frac12n_3,\qquad
	n_2=\frac{\sqrt3}{2}v_3-\frac12n_3.
	\]
	The Hessian $H_g=\nabla_x^2(\phi_1+\phi_2+\phi_3)|_{x=x_c}$ has the
	following matrix in the basis $(v_3,n_3)$:
	\begin{equation}\label{eq:spatial-hessian}
		\begin{gathered}
			H_g=\frac14\begin{pmatrix}
				3(\alpha_1+\alpha_2)&\sqrt3(\alpha_1-\alpha_2)\\
				\sqrt3(\alpha_1-\alpha_2)&\alpha_1+\alpha_2+4\alpha_3
			\end{pmatrix},\\
			\det H_g=\frac34(\alpha_1\alpha_2+\alpha_1\alpha_3+\alpha_2\alpha_3)\ge c.
		\end{gathered}
	\end{equation}
	Stationary phase \cite[Corollary~1.1.8 and (1.1.20)]{Sogge}
	gives \eqref{eq:stationary-kernel} with
	$b_\lambda(s)=2\pi i\,a_\lambda(x_c,s)/\sqrt{\det H_g}$.
	
	Differentiating \eqref{eq:reduced-phase} gives
	$H_g\partial_jx_c=\beta_jn_j$. Differentiating $\Phi$ and using
	\eqref{eq:spatial-hessian}, we obtain
	\begin{equation}\label{eq:geometric-mixed}
		\begin{gathered}
			\Phi_{12}=\frac{3\alpha_3\beta_1\beta_2}{4\det H_g}\approx1,\qquad
			\Phi_{13}=\frac{3\alpha_2\beta_1\beta_3}{4\det H_g}\approx1,\qquad
			\Phi_{23}=\frac{3\alpha_1\beta_2\beta_3}{4\det H_g}\approx1,\\
			\Phi_{33}=\partial_{s_3}^2\phi_3(x_c,s_3)
			-\frac{3\beta_3^2(\alpha_1+\alpha_2)}{4\det H_g},\\
			q_\Phi=\Phi_{33}-\frac{\Phi_{13}\Phi_{23}}{\Phi_{12}}
			=\left[\partial_{s_3}^2\phi_3-\frac{\beta_3^2}{\alpha_3}\right]_{x=x_c}.
		\end{gathered}
	\end{equation}
	The same equations give
	\begin{equation}\label{eq:critical-direction}
		\begin{aligned}
			H_g(\Phi_{23}\partial_1x_c-\Phi_{13}\partial_2x_c)
			&=\frac{3\beta_1\beta_2\beta_3}{4\det H_g}(\alpha_1n_1-\alpha_2n_2)\\
			&=-\frac{\sqrt3\,\beta_1\beta_2\beta_3}{2\det H_g}H_gv_3.
		\end{aligned}
	\end{equation}
	Before evaluation at $x_c$, the Riccati equation
	\cite[Corollary~3.2.10]{Petersen} and the $s_3$-derivatives of
	$\nabla_{v_3}v_3=0$ and $|\nabla_x\phi_3|_g^2=1$ give
	\begin{equation}\label{eq:riccati-identities}
		v_3(\alpha_3)=\alpha_3^2+K_g,\qquad
		v_3(\beta_3)=\alpha_3\beta_3,\qquad
		v_3(\partial_{s_3}^2\phi_3)=\beta_3^2.
	\end{equation}
	Consequently,
	\begin{equation}\label{eq:directional-curvature}
		v_3\left(\partial_{s_3}^2\phi_3-\frac{\beta_3^2}{\alpha_3}\right)
		=\frac{\beta_3^2}{\alpha_3^2}K_g.
	\end{equation}
	Since $H_g$ is invertible and
	$\Phi_{23}\partial_1-\Phi_{13}\partial_2$ leaves $s_3$ fixed,
	\eqref{eq:geometric-mixed}, \eqref{eq:critical-direction}, and
	\eqref{eq:directional-curvature} give, by the chain rule,
	\begin{align}
		(\Phi_{23}\partial_1-\Phi_{13}\partial_2)q_\Phi
		&=-\frac{\sqrt3\,\beta_1\beta_2\beta_3^3}{2\alpha_3^2\det H_g}K_g(x_c),
		\label{eq:curvature-identity}\\
		\left|(\Phi_{23}\partial_1-\Phi_{13}\partial_2)q_\Phi\right|
		&\approx\varepsilon.\label{eq:curvature-nondegeneracy}
	\end{align}
	
	For $g=I$, denote the reduced phase by $\Phi_0$ and choose
	$O_j\in SO(2)$ with $O_jv_j^0=v_1^0$.
	The $2\pi/3$ angles and continuity from $s=0$ imply that
	$O_j(Y_j-x_c)$ have the same direction. Since $O_1+O_2+O_3=0$,
	\begin{equation}\label{eq:flat-phase}
		\begin{gathered}
			\Phi_0(s)=|O_1Y_1+O_2Y_2+O_3Y_3|=H(s_1+s_2+s_3),\\
			H(\sigma)=\sqrt{(t_1+t_2+t_3)^2+\sigma^2},\qquad
			H''(\sigma)=\frac{(t_1+t_2+t_3)^2}{H(\sigma)^3}\approx1.
		\end{gathered}
	\end{equation}
	By \eqref{eq:flat-comparison} and the implicit function theorem,
	\[
	\Phi=\Phi_0+\varepsilon\Psi_\varepsilon,\qquad
	\|\Psi_\varepsilon\|_{C^N}\le C_N.
	\]
	In particular, $\Phi_{ij}\approx1$ for all $1\le i,j\le3$.
	Together with \eqref{eq:curvature-nondegeneracy}, this verifies
	Lemma~\ref{lem:trilinear}.

		\]

\label{lem:cusp-fourier}

				u(x,y)&=\sqrt y\sum_{n\ne0}c_nW_\lambda(2\pi|n|y)e^{2\pi inx},\\
				\|u\|_{L^2(Y)}^2
				&=\sum_{n\ne0}|c_n|^2
				\int_{y_0}^\infty W_\lambda(2\pi|n|y)^2\frac{\,d y}{y}.

		\eqref{eq:global-fourier} restricted to $|n|\ge N$.

		\begin{equation}\label{eq:global-high-bound}
			|u_{\ge N}(x,y)|\le Ce^{-cNy}\|u\|_{L^2(Y)},\qquad y\ge H.
		\end{equation}

		\begin{aligned}
			\delta_m^{1/2}e^{-i\lambda\delta_m y}
			u_{1,\lambda}(\delta_m x,\delta_m y)&\longrightarrow G(x),\\
			\delta_m^{1/2}e^{-i\lambda[\xi_*\delta_m x-s_{\xi_*}(\delta_m y)]}
			u_{2,\lambda}(\delta_m x,\delta_m y)&\longrightarrow G(x+\sqrt3y).
		\end{aligned}
		\]
		Indeed $b_\xi(\delta_m y)\to1$, the quadratic Taylor error in the phase
		is $O_m((\lambda\delta_m)^{-1})$, and
		$\ell(\delta_m x,\delta_m y)/\delta_m\to x+\sqrt3y$.
		Reflection gives the third limit with $G(x-\sqrt3y)$.
		Moreover,
		\[
		y-2s_{\xi_*}(y)=-\frac3{m+3}y^{m+3}+O_m(y^{2m+5}),\qquad
		\lambda\delta_m^{m+3}=c_*^{m+3}.
		\]
		Since $(1+|z|)^{-3}\in L^1(\mathbb R^2)$, dominated convergence after
		$z\mapsto\delta_m z$ gives
		\begin{equation}\label{eq:finite-sharp-limit}
			\begin{aligned}
				&\lim_{\lambda\to\infty}\delta_m^{-1/2}
				\operatorname{Re}\int_Uw\prod_{j=1}^3u_{j,\lambda}\,d V_{g_m}\\
				&\quad=w(0)\int_{\mathbb R^2}G(x)G(x+\sqrt3y)G(x-\sqrt3y)
				\cos\left(\frac{3c_*^{m+3}}{m+3}y^{m+3}\right)\,d x\,d y>0.
			\end{aligned}
		\end{equation}
		The last integral is positive for sufficiently small $c_*>0$, by
		continuity at $c_*=0$.
		Equation~\eqref{eq:real-conversion} gives the lower bound in
		\eqref{eq:power-sharpness}. Theorem~\ref{thm:upper}\textup{(iii)} gives the upper bound.
		
		\medskip
		\textup{(ii)} Use the same cutoff $\chi$. For $v=(v_1,v_2)\in\mathbb R^2$, let
		$v^\perp=(-v_2,v_1)$. For even $n\ge2$, let
		\[
		\begin{gathered}
			\delta_0=c_*n^{-1/3},\qquad
			\theta_j=2\pi(j-1)/3,\qquad v_j=(\cos\theta_j,\sin\theta_j),\\
			\zeta_j(t)=(\sin t\,v_j^\perp+i v_j,\cos t)\in\mathbb C^3.
		\end{gathered}
		\]
		The dot product on $\mathbb C^3$ is the complex bilinear extension
		of the Euclidean inner product. Since
		$\zeta_j(t)\cdot\zeta_j(t)=0$,
		\[
		\Delta_{\mathbb R^3}(\zeta_j(t)\cdot x)^n
		=n(n-1)(\zeta_j(t)\cdot\zeta_j(t))(\zeta_j(t)\cdot x)^{n-2}=0.
		\]
		Thus the functions
		\begin{equation}\label{eq:sphere-average}
			u_{j,n}(x)=\left(\frac{n}{\delta_0}\right)^{1/2}
			\int_\mathbb R\chi(t/\delta_0)(\zeta_j(t)\cdot x)^n\,d t,
			\qquad x\in S^2,
		\end{equation}
		satisfy $-\Delta_{S^2}u_{j,n}=n(n+1)u_{j,n}$.
		
		For $\xi_1,\ldots,\xi_{2k}\in\mathbb C^3$, Isserlis's formula
		\cite[Theorem~1.28]{Janson}, extended by complex multilinearity, and
		polar integration give
		\[
		\begin{gathered}
			(2\pi)^{-3/2}\int_0^\infty r^{2k+2}e^{-r^2/2}\,d r
			=\frac{(2k+1)!}{4\pi\,2^k k!},\\
			\int_{S^2}\prod_{j=1}^{2k}(\xi_j\cdot x)\,d V_g(x)
			=\frac{4\pi\,2^k k!}{(2k+1)!}
			\sum_{\pi}\prod_{\{i,j\}\in\pi}(\xi_i\cdot\xi_j).
		\end{gathered}
		\]
		Here $k\ge0$ is an integer and the sum is over all pairings of
		$\{1,\ldots,2k\}$. Since $\zeta_j\cdot\zeta_j=0$, only the $n!$
		pairings between the two groups contribute to the $L^2$ integral.
		Using $\zeta_j(t)\cdot\overline{\zeta_j(s)}=2\cos^2((t-s)/2)$ gives
		\[
		\begin{aligned}
			&\int_{S^2}(\zeta_j(t)\cdot x)^n
			(\overline{\zeta_j(s)}\cdot x)^n\,d V_g(x)\\
			&\quad=\frac{4^{n+1}\pi(n!)^2}{(2n+1)!}
			\cos^{2n}\left(\frac{t-s}{2}\right)
			\le Cn^{-1/2}e^{-cn(t-s)^2},\qquad |t|,|s|\le1.
		\end{aligned}
		\]
		Stirling's formula gives the last bound. Young's inequality yields
		\begin{equation}\label{eq:sphere-norm}
			\begin{aligned}
				\|u_{j,n}\|_{L^2(S^2)}^2
				&\le\frac{C\sqrt n}{\delta_0}
				\iint\chi(t/\delta_0)\chi(s/\delta_0)e^{-cn(t-s)^2}\,d t\,d s\\
				&\le\frac{C\sqrt n}{\delta_0}
				\|\chi(\cdot/\delta_0)\|_2^2\|e^{-cn(\cdot)^2}\|_1\le C.
			\end{aligned}
		\end{equation}
		
		For $t=(t_1,t_2,t_3)$ near zero, define
		\[
		\mathcal A(t)=\frac8{27}
		\prod_{1\le i<j\le3}\bigl(\zeta_i(t_i)\cdot\zeta_j(t_j)\bigr),
		\qquad \mathcal A(0)=1.
		\]
		In the triple product, let $m_{ij}$ count the pairs joining groups
		$i,j$. Since $\zeta_j\cdot\zeta_j=0$, every nonzero term satisfies
		\[
		m_{12}+m_{13}=m_{12}+m_{23}=m_{13}+m_{23}=n,
		\qquad m_{12}=m_{13}=m_{23}=n/2.
		\]
		The number of such pairings is
		\[
		\binom{n}{n/2}^{\!3}\bigl((n/2)!\bigr)^3
		=\frac{(n!)^3}{((n/2)!)^3}.
		\]
		Isserlis's formula and polar integration with $2k=3n$ give
		\begin{equation}\label{eq:sphere-triple-moment}
			\int_{S^2}\prod_{j=1}^3(\zeta_j(t_j)\cdot x)^n\,d V_g(x)
			=\frac{4\pi\,3^{3n/2}(n!)^3(3n/2)!}
			{(3n+1)!((n/2)!)^3}\mathcal A(t)^{n/2}.
		\end{equation}
		The coefficient is positive and comparable to $n^{-1}$ by Stirling's formula.
		For $i\ne j$,
		\[
		\zeta_i(t_i)\cdot\zeta_j(t_j)
		=\cos t_i\cos t_j-\tfrac12\sin t_i\sin t_j+\tfrac12
		+i\sin(\theta_j-\theta_i)(\sin t_i-\sin t_j).
		\]
		Hence $\partial_{t_i}\log\mathcal A(0)=0$ and
		$\partial_{t_i}\partial_{t_j}\log\mathcal A(0)=-2/3$ for $1\le i,j\le3$.
		With the branch $\log\mathcal A(0)=0$, Taylor's formula gives
		\begin{equation}\label{eq:sphere-phase-expansion}
			\log\mathcal A(t)=-\frac{(t_1+t_2+t_3)^2}{3}+O(|t|^3).
		\end{equation}
		On the support of $\prod_{j=1}^3\chi(t_j/\delta_0)$, choose $c_*>0$ small enough that
		\[
		\frac n2|\operatorname{Im}\log\mathcal A(t)|
		\le Cn\delta_0^3=Cc_*^3\le\pi/3,
		\qquad \operatorname{Re}(\mathcal A(t)^{n/2})
		\ge\tfrac12|\mathcal A(t)|^{n/2}\ge0.
		\]
		\begin{samepage}
			For sufficiently large $n$, the set
			\[
			|t_1|,|t_2|\le\delta_0/8,\qquad
			|t_1+t_2+t_3|\le n^{-1/2}
			\]
			has volume $\delta_0^2/(8\sqrt n)$ and satisfies $|t_j|\le\delta_0/2$.
			The weight equals one there, and \eqref{eq:sphere-phase-expansion} gives
			\[
			|\mathcal A(t)|^{n/2}
			\ge\exp\left[-C\bigl(n(t_1+t_2+t_3)^2+n\delta_0^3\bigr)\right]\ge c.
			\]
		\end{samepage}
		Fubini, \eqref{eq:sphere-triple-moment}, and nonnegativity on the entire
		integration set therefore imply
		\begin{equation}\label{eq:sphere-global-lower}
			\operatorname{Re}\int_{S^2}\prod_{j=1}^3u_{j,n}\,d V_g
			\ge c\left(\frac n{\delta_0}\right)^{3/2}
			n^{-1}\delta_0^2n^{-1/2}=c\delta_0^{1/2}.
		\end{equation}
		Together with \eqref{eq:sphere-norm}, \eqref{eq:real-conversion} gives real
		normalized eigenfunctions with cubic integral at least
		$c\delta_0^{1/2}\approx\lambda_n^{-1/6}$.
		Theorem~\ref{thm:upper}\textup{(iii)} gives the upper bound.
