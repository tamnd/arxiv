<!-- definition-1: body_md -->
Consider a convex function \(f:[0,\infty)\rightarrow \mathbb{R}\cup \{+\infty\}\) with \(f(1)=0\). We say that the function \(f\) is subadditive if, for all random variables \(X\), \(Y\), and \(Z\) defined over finite sets, we have
    \begin{align}
 \widetilde{I}_{f}(X;YZ)&\leq \widetilde{I}_{f}(X;Y) + \widetilde{I}_{f}(X;Z)\label{sub_I_def}
\end{align}
whenever \(Y \rightarrow X \rightarrow Z\) forms a Markov chain, i.e., \(p_{XYZ} = p_X p_{Y|X} p_{Z|X}\).
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{thm:mutual-f-subadditivity}
    Let \(f:[0,\infty)\rightarrow \mathbb{R}\) with \(f(1)=0\) be a continuous convex function on $[0,\infty)$. Assume that $f$ is also four times continuously differentiable on $(0,\infty)$. Let \(g(x)=x^2f''(x)\). If \(g''(x)<0\) for all \(x>0\) and there exist constants \(\eta_g>0\) and \(\ell_g\in\mathbb{R}\) such that
    \begin{align}
       g(x)+\dfrac{2(g'(x)-\eta_g)^2}{-g''(x)}\leq 2\eta_g+(x-1)\ell_g~~\forall x>0,
    \end{align}
    then \(\widetilde{I}_{f}\) is subadditive.
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
The proof can be found in Appendix~\ref{app:mutual-f-subadditivity}.
<!-- end proof-1 -->
<!-- corollary-1: body_md -->
\label{coro:1}
Let \(s(x)\) be a positive concave function on \(x>0\) such that \(s''(x)<0\) for all \(x>0\). Assume that the function
\[
\mathtt{K}_{s}(x)= s(x)+\frac{2\left(s'(x)\right)^2}{-s''(x)}
\]
is concave. Let \(\alpha\) be a positive number such that
\[
\alpha \geq \mathtt{K}_{s}(1).
\]
Let \(g(x)=\alpha x+s(x)\). Then, by choosing
\[
(\eta_g,\ell_g)=\left(\alpha,\; \alpha+ \frac{d}{dx}\mathtt{K}_{s}(x)\Big|_{x=1}\right),
\]
the function \(g(x)\) satisfies the assumptions of Theorem~\ref{thm:mutual-f-subadditivity}.

Moreover, the corresponding \(f(x)\) is given by
\begin{align}
        f(x)=\alpha x\log x+\int_{1}^{x}(x-t)\frac{s(t)}{t^2}\,dt,\label{int:1:1}
\end{align}
for which  \(\widetilde{I}_{f}\) is subadditive.
<!-- end corollary-1 -->
<!-- proof-2: body_md -->
See Appendix~\ref{coro:1:proof}.

<!-- lemma-1: body_md -->
\label{domain:lem}
For arbitrary probability distributions \(p_X\) and \(q_X\) defined on a set \(\mathcal{X}\), the following inequality holds:
\begin{align}
    D_f(p_X\|q_X) \leq \mathsf{D_m}(f) \triangleq f(0) + \lim_{t\rightarrow\infty}\frac{f(t)}{t}. \label{defDmf}
\end{align}
<!-- end lemma-1 -->
<!-- definition-2: body_md -->
\label{def:admissible}
A pair of functions \((G(x),f(x))\) is said to be admissible if \(f:[0,\infty)\rightarrow\mathbb{R}\) is convex and satisfies \(f(1)=0\), while \(G:[0,\mathsf{D_m}(f))\rightarrow [0,\infty)\) is non-decreasing and satisfies \(G(0)=0\). Here, \(\mathsf{D_m}(f)\) is the quantity introduced in \eqref{defDmf}.
<!-- end definition-2 -->
<!-- definition-3: body_md -->
\label{def:div:mutual}
Let \((G,f)\) be an admissible pair. For distributions \(p_X\ll q_X\) on a finite alphabet \(\mathcal{X}\), the \((G,f)\)-divergence \(\mathcal{D}_{G,f}\) is defined as
\begin{align}
\mathcal{D}_{G,f}(p_X\|q_X)
&\triangleq G\!\big(D_f(p_X\|q_X)\big).
\end{align}
<!-- end definition-3 -->
<!-- definition-4: body_md -->
For random variables \(X\) and \(Y\), the corresponding mutual information \(I_{G,f}(X;Y)\) is defined by
\begin{align*}
I_{G,f}(X;Y)
&\triangleq \min_{q_Y} \sum_x p_X(x)\,\mathcal{D}_{G,f}(p_{Y|X=x}\|q_Y)
\\&= \min_{q_Y} \sum_x p_X(x)\, G\!\big(D_f(p_{Y|X=x}\|q_Y)\big).
\end{align*}
<!-- end definition-4 -->
<!-- remark-1: body_md -->
It follows immediately that \(I_{G,f}(X;Y)\) coincides with Shannon's mutual information when \(G(x)=x\) and \(f(x)=x\log x\). Furthermore, choosing
\[
G(x)=\frac{1}{\alpha-1}\log\bigl(1+(\alpha-1)x\bigr),
\qquad
f(x)=\frac{x^\alpha-1}{\alpha-1}
\]
recovers Rényi information for \(\alpha\in(0,1)\cup(1,\infty)\).
In general, however, \(I_{G,f}(X;Y)\) need not be symmetric, i.e.,
\[
I_{G,f}(X;Y) \neq I_{G,f}(Y;X),
\]
which differs from the behavior of Shannon's mutual information. We refer to \cite{abin2026class:arx} for several basic properties of \(I_{G,f}\).
<!-- end remark-1 -->
<!-- definition-5: body_md -->
An admissible pair \((G,f)\) is said to be \emph{divergence-subadditive} if, for any distributions \(q_Y\ll r_Y\) on a finite set \(\mathcal{Y}\) and any distributions \(q_Z\ll r_Z\) on a finite set \(\mathcal{Z}\), the following inequality holds:
\begin{align}
  \mathcal{D}_{G,f}(q_Y q_Z \,\|\, r_Y r_Z)
  \;&\leq\;
  \mathcal{D}_{G,f}(q_Y \,\|\, r_Y)\nonumber\\
  &+
  \mathcal{D}_{G,f}(q_Z \,\|\, r_Z). \label{gen-def-1}
\end{align}
Moreover, a pair \((G,f)\) is called \emph{information-subadditive} if, for any random variables \(X,Y,Z\) defined on finite sets, we have
\begin{align}
  I_{G,f}(X;YZ)&\leq I_{G,f}(X;Y) + I_{G,f}(X;Z)\label{sub_I_def:2}
\end{align}
whenever \(Y \rightarrow X \rightarrow Z\) forms a Markov chain, i.e., \(p_{XYZ} = p_X p_{Y|X} p_{Z|X}\).
<!-- end definition-5 -->
<!-- theorem-2: body_md -->
\label{gen-thm1}
    Let \((G,f)\) be an admissible pair (see Definition \ref{def:admissible}), where \(G\) is a continuous function satisfying \(G(x)>0\) for all \(x>0\), and \(f(x)\) is strictly convex at \(x=1\). Then \((G,f)\) is \emph{divergence-subadditive} if and only if it is \emph{information-subadditive}.

<!-- lemma-2: body_md -->
\label{gen-lemma1}
Let $G:[0,\nu)\rightarrow[0,+\infty)$ be an increasing function with $G(0)=0$ such that the range of $G$ is $[0, \infty)$. Moreover, assume that for every $c>0$, the function
\begin{align}
x \mapsto G^{-1}\big(G(x) + G(c)\big)\label{cond:G}
\end{align}
is concave on $[0,\nu)$.  
Let $f$ be a convex function on $[0,\infty)$ with $f(1)=0$ such that $\mathsf{D_m}(f)\leq \nu$, where $\mathsf{D_m}(f)$ is defined in \eqref{defDmf}.  Assume that \eqref{gen-def-1} holds for any distributions $q_Y,r_Y$ on $\mathcal{Y}$ and $q_Z,r_Z$ on $\mathcal{Z}$ when both alphabets $\mathcal{Y}$ and $\mathcal{Z}$ are binary (i.e., of size two).  
Then \eqref{gen-def-1} also holds for this pair and for any distributions $q_Y,r_Y$ and $q_Z,r_Z$ defined on finite alphabets $\mathcal{Y}$ and $\mathcal{Z}$ of arbitrary size.
<!-- end lemma-2 -->
<!-- lemma-3: body_md -->
\label{corol:g}
Let \(u(t)\) be any positive concave function on \([0,\nu)\). Define a function \(G(x)\) on \([0,\nu)\) by

Then \(G\) satisfies the assumptions in Lemma~\ref{gen-lemma1}.

<!-- theorem-3: body_md -->
\label{th:concave:G}
Assume that \[
G(x) = \int_0^x \frac{1}{u(t)}\,dt
\]
for a positive increasing concave function \(u(t)\) on \([0,\infty)\).  Let \(f\) be a four-times continuously differentiable convex function with \(f(1)=0\). For a positive parameter \(\gamma\), define the set
\begin{align}
    \mathcal{T}_{\gamma}=\Bigl\{g:[0,\infty)\rightarrow [0,\infty)\;\Big|\;&x^2g''(x)-\gamma\, g(\alpha)\,g\!\left(\frac{x}{\alpha}\right)\leq 0,\nonumber\\& \forall x>0,~\forall\alpha>0\Bigr\}.
\end{align}
If \(x^2f''(x)\in \mathcal{T}_{\gamma}\) for
\[
0<\gamma\leq \frac{G^{-1}\!\bigl(2G(\mathsf{D_m}(f))\bigr)-2\mathsf{D_m}(f)}{\mathsf{D_m}^{2}(f)},
\]
then the pair \((G,f)\) is divergence-subadditive. Here, \(\mathsf{D_m}(f)\) is as defined in \eqref{defDmf}.
<!-- end theorem-3 -->
<!-- proof-3: body_md -->
The proof can be found in Appendix~\ref{th:concave:G:p}.

One may reuse these parametric functions for \(\mathcal{T}_{\gamma}\) when $\gamma\neq 1$ after suitable adjustments.

<!-- theorem-4: body_md -->
\label{th:convex:G}
Assume that \[
G(x) = \int_0^x \frac{1}{u(t)}\,dt
\]
for a positive decreasing concave function \(u(t)\) on \([0,\nu)\) for some $\nu>0$. Then \(G(x)\) is convex. Let \(f\) be a four-times continuously differentiable convex function with \(f(1)=0\). For a positive parameter \(\gamma\), define the set
\begin{align}
    \mathcal{T}_{\gamma}^{-}=\Bigl\{g:[0,\infty)\rightarrow [0,\infty)\;\Big|\;&x^2g''(x)+\gamma\, g(\alpha)\,g\!\left(\frac{x}{\alpha}\right)\leq 0,\nonumber\\& \forall x>0,~\forall\alpha>0\Bigr\}.
\end{align}
If \(x^2f''(x)\in \mathcal{T}_{\gamma}^{-}\) for
\[
\gamma\geq  \frac{2\mathsf{D_m}(f)-G^{-1}\!\bigl(2G(\mathsf{D_m}(f))\bigr)}{\mathsf{D_m}^{2}(f)},
\]
then the pair \((G,f)\) is divergence-subadditive. Here, \(\mathsf{D_m}(f)\) is as defined in \eqref{defDmf}.
<!-- end theorem-4 -->
<!-- proof-4: body_md -->
The proof can be found in Appendix \ref{th:convex:G:p}.
<!-- end proof-4 -->
<!-- remark-3: body_md -->
Theorem \ref{th:convex:G} suggests that, in order to find a suitable function \(f\), one should characterize the set \(\mathcal{T}_{\gamma}^{-}\). Suppose that \(\ell_{\gamma}(x)\in \mathcal{T}_{\gamma}^{-}\). Then the equation \(x^2 f_\gamma''(x)=\ell_{\gamma}(x)\) yields the following solution:
\begin{align}
    f_{\gamma}(x)=\widetilde{a}(x-1)+\int_{1}^{x}(x-t)\frac{\ell_{\gamma}(t)}{t^2}\,dt.
\end{align}
Note that the parameter \(\mathsf{D_m}(f)\) depends on \(\gamma\). After computing \(f_{\gamma}(x)\) using the formula above, one should calculate \(\mathsf{D_m}(f_{\gamma})\) and then choose the parameter \(\gamma\) such that
\[
\gamma\geq \frac{2\mathsf{D_m}(f_{\gamma})-G^{-1}\!\bigl(2G(\mathsf{D_m}(f_{\gamma}))\bigr)}{\mathsf{D_m}^{2}(f_{\gamma})}.
\]
The most effective approach for designing the function \(f\) is to consider a parametric family in \(\mathcal{T}_{\gamma}^{-}\) and then select the parameters so that all conditions are satisfied.
<!-- end remark-3 -->
<!-- remark-4: body_md -->
The set \(\mathcal{T}^{-}_{\gamma}\) for $\gamma=1$ was already introduced in \cite{abin2026class:isit} when studying the special case of \(G(x)=-\log(1-x)\).

<!-- lemma-4: body_md -->
\label{yhn:1}
Under the assumptions of Theorem~\ref{thm:mutual-f-subadditivity},
for any fixed  $p_{X}$ and $p_{Y|X}$, the map \(\Theta_{p_X,p_{Y|X}}\left(r_X\right)\) is a concave function of \(r_{X}\),

&\Theta_{p_X,p_{Y|X}}\left(r_{X}\right)\triangleq\nonumber
\\&\sum_{y\in\mathcal{Y}}r_{Y}(y)\left[D_{f}\left(r_{X|y}\,\Big\|\,p_X\right)\right]-D_{f}\bigl(r_{X}\|p_X\bigr)

and $r_{YX}=r_Xp_{Y|X}$.\label{lemma4}

<!-- proof-5: body_md -->
Since \(r_X\) is a probability distribution, it suffices to show that for every function \(v_X\) satisfying \(\sum_{x}v_X(x)=0\), the following quantity is non-positive:
\begin{align}
    &\frac{d^2}{d\epsilon^2}\Theta_{p_X,p_{Y|X}}\left(r_X+\epsilon v_X\right)\Big|_{\epsilon=0}
\nonumber\\&=\vec{v}_{X}^{T}\,\mathcal{H}\!\left(\Theta_{p_X,p_{Y|X}}\left(r_X\right)\right)\vec{v}_{X}.
\end{align}
Here, \(\mathcal{H}\!\left(\Theta_{p_X,p_{Y|X}}\left(r_X\right)\right)\) denotes the Hessian matrix of \(\Theta_{p_X,p_{Y|X}}\left(r_X\right)\), and \(\vec{v}_{X}\) is the column vector corresponding to the function \(v_X\), while \(T\) denotes the transpose operator. Let \(g(x)\triangleq x^2f''(x)\)
and set
$$\beta_X(x)=\frac{v_X(x)}{r_X(x)}, \qquad\forall x.$$
Then,  routine calculations show that
\begin{align}
     &\frac{d^2}{d\epsilon^2}\Theta_{p_X,p_{Y|X}}\left(r_X+\epsilon v_X\right)\Big|_{\epsilon=0} \nonumber\\&=
     \sum_{y\in\mathcal{Y}}r(y)\Bigg[\sum_{x}p(x)\,g\!\left(\frac{r(x|y)}{p(x)} \right)\nonumber\\&\quad\times\left(\beta(x)-\sum_{x'\in \mathcal{X}}r(x'|y)\,\beta(x')\right)^2\Bigg]\nonumber\\&-\sum_{x\in\mathcal{X}}p(x)\,g\!\left(\frac{r(x)}{p(x)}\right)\beta^{2}(x).
\end{align}
For every real-valued function \(u_X\) and all distributions \(p_X\) and $q_X$, define the function
\begin{align}
    \mathcal{L}_{u_X,p_X}(q_X)\triangleq \mathbb{E}_{p_X}\!\left[g\!\left(\frac{q_X}{p_X}\right)\left(u_X-\mathbb{E}_{q_X}(u_X)\right)^2\right].
\end{align}
Then, one can observe that
\begin{align}
     &\frac{d^2}{d\epsilon^2}\Theta_{p_X,p_{Y|X}}\left(r_X+\epsilon v_X\right)\Big|_{\epsilon=0} \nonumber\\&=
 \sum_{y\in\mathcal{Y}}r(y)\mathcal{L}_{\beta_X,p_X}(r_{X|y}) \\&-\mathcal{L}_{\beta_X,p_X}(r_{X}).
\end{align}
Thus, it suffices to show that $\mathcal{L}_{u_X,p_X}(q_X)$ is concave in  $q_X$ for any $u_X$ and $p_X$. 

 As before, consider an arbitrary real-valued function \(k_X\) such that \(\sum_{x\in \mathcal{X}}k_X(x)=0\). To show that \(\mathcal{L}_{u_X,p_X}(q_X)\) is concave over probability distributions, it suffices to show that
\begin{align}
        &\frac{d^2}{d\epsilon^2}\mathcal{L}_{u_X,p_X}(q_X+\epsilon k_X)\Big|_{\epsilon=0} \leq 0
\end{align}
for all distributions \(p_X\) and \(q_X\) and all functions \(u_X\) and \(k_X\) satisfying \(\sum_{x\in\mathcal{X}}k_X(x)=0\). Routine calculations show that
\begin{align}
    &\frac{d^2}{d\epsilon^2}\mathcal{L}_{u_X,p_X}(q_X+\epsilon k_X)\Big|_{\epsilon=0}=\label{diff;2}\\
    &\mathbb{E}_{p_X}\!\left[\widetilde{\omega}_{X}^2\,g''\!\left(\frac{q_X}{p_X}\right)-4\widetilde{\omega}_{X}\widetilde{\gamma}\,g'\!\left(\frac{q_X}{p_X}\right)+2\widetilde{\gamma}^{2}g\!\left(\frac{q_X}{p_X}\right)\right],\nonumber
\end{align}
where
\begin{align}
    \widetilde{\omega}_{X}&=\frac{\left(u_X-\mathbb{E}_{q}\left(u_X\right)\right)k_X}{p_X},\\
    \widetilde{\gamma}&=\mathbb{E}_{p_X}\left(\widetilde{\omega}_{X}\right).\label{qw:1}
\end{align}


Now, we wish to design a function \(g(x)\) such that the expression in \eqref{diff;2} is non-positive. Let \(\eta_g\) be a constant. Using \(g'=(g'-\eta_g)+\eta_g\) together with \eqref{qw:1}, we obtain
\begin{align}
    &\frac{d^2}{d\epsilon^2}\mathcal{L}_{u_X,p_X}(q_X+\epsilon k_X)\Big|_{\epsilon=0}=
    \mathbb{E}_{p_X}\!\left[\widetilde{\omega}_{X}^2\,g''\!\left(\frac{q_X}{p_X}\right)\right.\nonumber\\&\left.-4\widetilde{\omega}_{X}\widetilde{\gamma}\left(g'\!\left(\frac{q_X}{p_X}\right)-\eta_g\right)+2\widetilde{\gamma}^{2}g\!\left(\frac{q_X}{p_X}\right)\right]-4\widetilde{\gamma}^2\eta_g.\label{diff;3}
\end{align}
Assuming \(g''(x)\neq 0\), rearranging \eqref{diff;3} yields
\begin{align}
    &\frac{d^2}{d\epsilon^2}\mathcal{L}_{u_X,p_X}(q_X+\epsilon k_X)\Big|_{\epsilon=0}=
    \mathtt{A}_1+2\widetilde{\gamma}^2\mathtt{A}_{2}-4\widetilde{\gamma}^2\eta_g,
\end{align}
where
\begin{align}
\mathtt{A}_1&=\mathbb{E}_{p_X}\!\left[\frac{\left(\widetilde{\omega}_{X}\,g''\!\left(\frac{q_X}{p_X}\right)-2\widetilde{\gamma}\left(g'\!\left(\frac{q_X}{p_X}\right)-\eta_g\right)\right)^2}{g''\!\left(\frac{q_X}{p_X}\right)}\right],\\
 \mathtt{A}_2&=\mathbb{E}_{p_X}\!\left[g\!\left(\frac{q_X}{p_X}\right)+\frac{2\left(g'\!\left(\frac{q_X}{p_X}\right)-\eta_g\right)^2}{-g''\!\left(\frac{q_X}{p_X}\right)}\right].
\end{align}
It is clear that if \(g''(x)<0\) for all \(x>0\), then \(\mathtt{A}_1\leq 0\). Moreover, if we assume that for all \(x>0\) the function \(g(x)\) satisfies
\begin{align}
    g(x)+\frac{2\left(g'(x)-\eta_g\right)^2}{-g''(x)}\leq \theta_1+\theta_2(x-1)\label{iu:1}
\end{align}
for some constants \(\theta_1\) and \(\theta_2\), then \(\mathtt{A}_2\leq \theta_1\). Thus, we have
\begin{align}
    \frac{d^2}{d\epsilon^2}\mathcal{L}(q_X+\epsilon k_X)\Big|_{\epsilon=0}&=
\mathtt{A}_1+2\widetilde{\gamma}^2\mathtt{A}_{2}-4\widetilde{\gamma}^2\eta_g,\nonumber\\
    &\leq 2\widetilde{\gamma}^2(\theta_1-2\eta_g).
\end{align}
By choosing \(\theta_1= 2\eta_g\), we obtain \(\frac{d^2}{d\epsilon^2}\mathcal{L}(q_X+\epsilon k_X)\big|_{\epsilon=0}\leq 0\). 
Recall that \(f\) is a convex function. Since \(x^2f''(x)=g(x)\) and we assume that \(g''(x)<0\), the left-hand side of \eqref{iu:1} is a positive function, and thus \(\mathtt{A}_{2}\) is positive. Since \(\mathtt{A}_{2}\leq \theta_1\), we conclude that \(\eta_g\) must be positive.
 This completes the proof of Lemma~\ref{yhn:1}.

According to Theorem~3 in \cite{abin2026class:isit}, if the function \(x^2f''\) belongs to the set \(\mathcal{T}^{+}\) defined as
    \begin{align}
    \mathcal{T}^{+}=\Bigl\{g:[0,\infty)\rightarrow [0,\infty)\;\Big|\;&x^2g''(x)-\, g(\alpha)\,g\!\left(\frac{x}{\alpha}\right)\leq 0,\nonumber\\& \forall x>0,~\forall\alpha>0\Bigr\},
\end{align}
then the pair \((\log(1+x),f)\) is divergence-subadditive. Let \(\gamma\) be a positive constant. Since
\[
\mathcal{D}_{\log(1+\gamma x),f}(\cdot\|\cdot)=\mathcal{D}_{\log(1+x),\gamma f}(\cdot\|\cdot),
\]
if \(x^2(\gamma f)''=\gamma\, x^2 f''\in \mathcal{T}^{+}\), then the pair \((\log(1+x),\gamma f)\), and equivalently the pair \((\log(1+\gamma x),f)\), is divergence-subadditive. Let \(x^2f''=g(x)\). The condition \(x^2(\gamma f)''\in \mathcal{T}^{+}\) implies
  \begin{align*}
     &x^2 (\gamma g)''(x)-[\gamma g](\alpha)\,[\gamma g]\!\left(\frac{x}{\alpha}\right)\\
     &=\gamma\left(x^2 g''(x)-\gamma\, g(\alpha)\,g\!\left(\frac{x}{\alpha}\right)\right)\leq 0.
  \end{align*}
Thus, if \(x^2f''\in \mathcal{T}_{\gamma}\), then the pair \((\log(1+\gamma x),f)\) is divergence-subadditive. Divergence-subadditivity of the pair \((\log(1+\gamma x),f)\) implies
  \begin{align}
    \log(1+\gamma c)\leq \log(1+\gamma a)+\log(1+\gamma b),
\end{align}
which is equivalent to
\begin{align}
   c\leq a+b+\gamma ab.
\end{align}
