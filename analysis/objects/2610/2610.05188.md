<!-- assumption-1: body_md -->
\label{ass:main}
Let $\{(v_k(x),u_k(x))\}_{k=1}^N$ be a pair of input and output real-valued functions, $\omega$ denotes NN parameters. The following conditions hold.
\begin{enumerate}
\item The query points $\{x_j\}_{j=1}^J \subset \mathbb{R}^n$ are i.i.d. sampled from a spherical Gaussian distribution $\mathcal{N}(0, I_n)$.
\item The output function $u_k(x)$ is approximated by an integral operator
\[
u_k(x) \approx \frac{1}{J}\sum_{\ell=1}^J \kappa(x,y_\ell) v_k(y_\ell),
\]
with a kernel $\kappa(x,y)$ that admits the separable form
\[
\kappa(x,y)=\sigma(w^\top x)\,\sigma(w^\top y),
\]
where $\sigma(\cdot)=\mathrm{ReLU}(\cdot)$.
\item The input functions are
\[
v_k(x)=\mu_k\,\sigma((w^*)^\top x),\qquad \mu_k\in\mathbb{R},\; w^*\in\mathbb{R}^n,
\]
and the output functions predicted by the network are
\[
u_k(x;w)=\frac{1}{J}\sum_{\ell=1}^J \mu_k\,\sigma(w^\top x)\,\sigma(w^\top y_\ell)\,\sigma((w^*)^\top y_\ell).
\]
The ground‑truth outputs are $u_k(x;w^*)$.
\item The parameters evolve according to the gradient flows
\[
\frac{dw_{L^2}}{dt}=-\mathbb{E}_X\Bigl[\frac{\partial\mathcal{L}_{L^2}}{\partial w}\Bigr],\qquad
\frac{dw_{\mathrm{Sob}}}{dt}=-\mathbb{E}_X\Bigl[\frac{\partial\mathcal{L}_{\mathrm{Sob}}}{\partial w}\Bigr],
\]
with
\[
\mathcal{L}_{L^2}=\frac1N\sum_{k=1}^N\bigl\|u_k(\cdot;w)-u_k(\cdot;w^*)\bigr\|_{L^2}^2,\qquad
\mathcal{L}_{\mathrm{Sob}}=\mathcal{L}_{L^2}+\mathcal{L}_{\mathrm{Der}},
\]
\[
\mathcal{L}_{\mathrm{Der}}=\frac1N\sum_{k=1}^N\bigl\|D_x u_k(\cdot;w)-D_x u_k(\cdot;w^*)\bigr\|_{L^2}^2.
\]
\item The initial weight vector satisfies $\|w_0-w^*\|<\|w^*\|$.
\end{enumerate}

<!-- theorem-1: body_md -->
\label{thm:main}
Under Assumption~\ref{ass:main}, we have
\[
\frac{d}{dt}\|w_{L^2}-w^*\|^2 < 0
\]
whenever $\|w_0-w^*\|<\|w^*\|$, and $w_{L^2}(t)\to w^*$ as $t\to\infty$.
For the Sobolev loss we have
\[
\frac{d}{dt}\|w_{\mathrm{Sob}}-w^*\|^2 \;\le\; \frac{d}{dt}\|w_{L^2}-w^*\|^2 \;<\; 0.
\]
Moreover, if $w$ and $w^*$ are not parallel (i.e. the angle $\theta(w,w^*)\in(0,\pi)$), then the inequality is strict
\[
\frac{d}{dt}\|w_{\mathrm{Sob}}-w^*\|^2 \;<\; \frac{d}{dt}\|w_{L^2}-w^*\|^2 \;<\; 0.
\]
Hence incorporating derivative information accelerates the convergence rate.
