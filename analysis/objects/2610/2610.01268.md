<!-- lemma-5: body_md -->
\label{lemma:l1bound}
    Let $T > 0$. Then there exists constants  $M_1(T) > 0$ and $M_2(T) > 0$ such that
    \begin{align}
    \label{L1ess}
    \int_\Omega c^i(\cdot,t)\leq M_1(T)
    \qquad \text{for all } i\in\{E,M,F\},\quad t\in(0,\widehat T),
\end{align}
    and
    \begin{align}
\label{sqc}
\int_0^{\widehat{T}}\int_\Omega (c^j)^2
\leq M_2(T)
\qquad \text{for all } j\in\{E,F\}.
\end{align}
    where $\widehat{T} := \min\{T, T_{\max}\}$.
<!-- end lemma-5 -->
<!-- proof-4: body_md -->
We integrate the first three equations of \textcolor{magenta}{\textcolor{magenta}{\textcolor{magenta}{\textcolor{magenta}{\eqref{model1}}}}} over $\Omega$ and add them. Owing to the non-negativity of the solution components, this yields the following differential inequality:
   \begin{align}
       \label{L11}
       \frac{d}{dt} \left\{\int_\Omega c^E + \int_\Omega c^M + \int_\Omega c^F\right\} + \rho_E \int_\Omega \left(c^E\right)^2 +  \rho_F \int_\Omega \left(c^F\right)^2 \leq \rho \left\{\int_\Omega c^E + \int_\Omega c^M + \int_\Omega c^F\right\}
   \end{align}
   for all $t \in (0, T_{\max})$, where $\rho = \max\{\rho_E, 1, \rho_F\}$. Dropping the non-negative quadratic terms from the left-hand side of \textcolor{magenta}{\textcolor{magenta}{\textcolor{magenta}{\textcolor{magenta}{\eqref{L11}}}}}, we infer that
\begin{align}
\label{L111}
\frac{d}{dt}\left\{\int_\Omega c^E +\int_\Omega c^M +\int_\Omega c^F\right\} \leq \rho \left\{\int_\Omega c^E +\int_\Omega c^M +\int_\Omega c^F\right\}
\end{align}
for all $t\in(0,T_{\max})$. An application of Gronwall's inequality to \textcolor{magenta}{\textcolor{magenta}{\textcolor{magenta}{\textcolor{magenta}{\eqref{L111}}}}} on $(0,\widehat T)$ therefore yields
\begin{align*}
\int_\Omega c^E(\cdot,t) +\int_\Omega c^M(\cdot,t) +\int_\Omega c^F(\cdot,t)
&\leq e^{\rho t} \left\{\int_\Omega c^E_0 +\int_\Omega c^M_0 +\int_\Omega c^F_0\right\}\\[5pt]
&\leq e^{\rho T}\left\{\int_\Omega c^E_0+\int_\Omega c^M_0+\int_\Omega c^F_0 \right\}
\end{align*}
for all $t\in(0,\widehat T)$. Since $c^E$, $c^M$ and $c^F$ are non-negative, each individual mass is bounded by the total mass. Hence, upon defining
\begin{align*}
M_1(T) := e^{\rho T} \left\{\int_\Omega c^E_0 +\int_\Omega c^M_0 +\int_\Omega c^F_0\right\},
\end{align*}
we directly obtain \textcolor{magenta}{\textcolor{magenta}{\textcolor{magenta}{\textcolor{magenta}{\eqref{L1ess}}}}}. Set, 
$$y_1(t):=\sum_{i\in\{E,M,F\}}\int_\Omega c^i(\cdot,t).$$
Integrating \textcolor{magenta}{\textcolor{magenta}{\textcolor{magenta}{\textcolor{magenta}{\eqref{L11}}}}}, over $(0,\widehat{T})$ in view of \textcolor{magenta}{\textcolor{magenta}{\textcolor{magenta}{\textcolor{magenta}{\eqref{L1ess}}}}}, we further obtain 
\begin{align*} 
\rho_E \int_0^{\widehat{T}}\int_\Omega (c^E)^2 + \rho_F \int_0^{\widehat{T}}\int_\Omega (c^F)^2 & \leq  y_1(0) +\rho\int_0^{\widehat{T}}y_1(t)\,dt\\[5pt]
&\leq  y_1(0)+\rho \widehat{T} M_1(T) \\[5pt]
& \leq  y_1(0)+\rho T M_1(T) 
\end{align*} 
Consequently,
\begin{align*}
\int_0^{\widehat{T}}\int_\Omega (c^E)^2 \leq \frac{y_1(0)+\rho T M_1(T)}{\rho_E}, \qquad
\int_0^{\widehat{T}}\int_\Omega (c^F)^2 \leq \frac{y_1(0)+\rho T M_1(T)}{\rho_F}.
\end{align*}
Hence, upon defining
\begin{align*}
M_2(T):= \max\left\{\frac{y_1(0)+\rho T M_1(T)}{\rho_E}, \frac{y_1(0)+\rho T M_1(T)}{\rho_F}\right\},
\end{align*}
we conclude \textcolor{magenta}{\textcolor{magenta}{\textcolor{magenta}{\textcolor{magenta}{\eqref{sqc}}}}}

 Let $T > 0$. Then there exist constants $a^\ast(T),u^\ast(T), w^\ast(T) > 0$ such that
\begin{equation}

\int_\Omega u(\cdot,t) \leq u^\ast(T)

\int_\Omega w(\cdot,t) \leq w^\ast(T) 
\quad \text{for all } t \in (0,\widehat{T}),
\end{equation}

<!-- proof-5: body_md -->
Using the definitions of $a$, $u$ and $w$ from \textcolor{magenta}{\textcolor{magenta}{\textcolor{magenta}{\textcolor{magenta}{\eqref{cov}}}}}, together with \textcolor{magenta}{\textcolor{magenta}{\textcolor{magenta}{\textcolor{magenta}{\eqref{v_es1}}}}}, we obtain
\begin{align*}
& \int_\Omega a =  \int_\Omega e^{-\chi_E v} c^E \leq \int_\Omega c^E \leq M_1(T),\\[5pt]
& \int_\Omega u 
= \int_\Omega e^{-\chi_M v}\, c^M
