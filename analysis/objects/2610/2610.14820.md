\label{scohomology}
For any $j$, we have:
$$
\mathrm{H}^{j}(\mathfrak{osp}((1|2),\mathfrak D_{{\lambda},\mu})=\mathrm{H}^{j}_{\mathrm{diff}}(\mathfrak{osp}((1|2),\mathfrak D_{{\lambda},\mu})=\mathrm{H}^{j}(\mathfrak{osp}((1|2),M),
$$
where $M$ is the submodule of $\mathfrak{D}_{{\lambda},\mu}$ consisting of the polynomial operators and $\mathrm{H}^{j}_\mathrm{diff}$ denotes the differential cohomology.
Moreover, up to a coboundary, any $j$-cocycle
$c\in \mathrm{Z}^j(\mathfrak{osp}(1|2),\mathfrak{D}_{\lambda,\mu})$ is reduced, i.e.\ for any $u_2,\dots,u_j$ in $\mathfrak{osp}(1|2)$, $c(X_1,u_2,\dots,u_j)=0$.

\label{cocycles}
For each $j$, we have:
$$\aligned
Z^{j}(\mathfrak{osp}(1|2),\mathfrak D_{{\lambda},\mu})&=Z^{j}_{\mathrm{red}}(\mathfrak{osp}(1|2),M)+B^{j}(\mathfrak{osp}(1|2),\mathfrak D_{{\lambda},\mu})\\
Z^{j}_{\mathrm{diff}}(\mathfrak{osp}(1|2),\mathfrak D_{{\lambda},\mu})&=Z^{j}_{\mathrm{red}}(\mathfrak{osp}(1|2),M)+B^{j}_{\mathrm{diff}}(\mathfrak{osp}(1|2),\mathfrak D_{{\lambda},\mu})\\
Z^{j}(\mathfrak{osp}(1|2),M)&=Z^{j}_{\mathrm{red}}(\mathfrak{osp}(1|2),M)+B^{j}(\mathfrak{osp}(1|2),M).
\endaligned
$$

Notice that the operator $X_1\cdot$ is simply the derivation operator with respect to $x$:
$$
X_1\cdot a_\alpha(x,\theta,\partial_\theta)\Omega^\alpha=a'_\alpha(x,\theta,\partial_\theta)\Omega^\alpha.
$$
Consequently, this operator is surjective, and the space $M$ is given by:
$$
M=\{A\in\mathfrak D_{{\lambda},\mu}~|~\exists\, k\in\mathbb{N}, (X_1)^k\cdot A=0\}.
$$
In particular, its kernel is: $$\hbox{ker}(X_1\cdot)=\mathrm{Span}_{\mathbb R}\{\Omega^\alpha,\,\theta\Omega^\alpha,\,\partial_\theta\Omega^\alpha,\,\theta\partial_\theta\Omega^\alpha\}.$$

\noindent
{$\mathbf{Case~} j=0$}

If $j=0$, there is nothing to prove. Indeed, there are no coboundaries, and any $0$-cocycle $c \in \mathfrak D_{{\lambda},\mu}$ is differential and satisfies $X_1\cdot c = 0$. This implies that $c \in M$, yields:
$$
Z^{0}(\mathfrak{osp}(1|2),\mathfrak D_{{\lambda},\mu})=Z^{0}_{\mathrm{diff}}(\mathfrak{osp}(1|2),\mathfrak D_{{\lambda},\mu})=Z^{0}(\mathfrak{osp}(1|2),M)=Z^{0}_{\mathrm{red}}(\mathfrak{osp}(1|2),M).
$$

\noindent
{$\mathbf{Case~} j=1$}

If $a$ is a 1-cochain, there exists $b \in \mathfrak D_{{\lambda},\mu}$ such that $X_1 \cdot b = a(X_1)$. 
If $a$ is a cocycle, the cocycle $c = a - \delta b$ is thus reduced (i.e., it vanishes on $X_1$). 
The cocycle relations applied to the pairs $(X_1,X_x)$, $(X_1,X_{x^2})$, $(X_1,X_{\theta})$, and $(X_1,X_{x\theta})$ then yield
$$X_1 \cdot c(X_x) = 0, \quad X_1^2 \cdot c(X_{x^2}) = 0, \quad X_1 \cdot c(X_{\theta}) = 0, \quad X_1^2 \cdot c(X_{x\theta}) = 0,$$
which implies that all values of $c$ lie in $M$. Note that $b$ is differential. Moreover, if $a$ has values in $M$, we can choose $b \in M$, which gives:
$$\aligned
Z^{1}(\mathfrak{osp}(1|2),\mathfrak D_{{\lambda},\mu})&=Z^{1}_{\mathrm{red}}(\mathfrak{osp}(1|2),M)+B^{1}(\mathfrak{osp}(1|2),\mathfrak D_{{\lambda},\mu}),\\
Z^{1}_{\mathrm{diff}}(\mathfrak{osp}(1|2),\mathfrak D_{{\lambda},\mu})&=Z^{1}_{\mathrm{red}}(\mathfrak{osp}(1|2),M)+B^{1}_{\mathrm{diff}}(\mathfrak{osp}(1|2),\mathfrak D_{{\lambda},\mu}),\\
Z^{1}(\mathfrak{osp}(1|2),M)&=Z^{1}_{\mathrm{red}}(\mathfrak{osp}(1|2),M)+B^{1}(\mathfrak{osp}(1|2),M).
\endaligned
$$

\noindent
{$\mathbf{Case~} j=2$}

Now, let $a\inZ^{2}(\mathfrak{osp}(1|2),\mathfrak D_{{\lambda},\mu})$ be a $2$-cocycle. Modifying \(a\) by a reduced coboundary $\delta b$ satisfying the relations:
$$
\aligned
X_1\cdot b(X_x)&=a(X_1,X_x),\quad ~&X_1\cdot b(X_\theta)&=a(X_1,X_\theta),\\
X_1\cdot b(X_{x\theta})&=a(X_1,X_{x\theta})-b(X_\theta),\quad &X_1\cdot b(X_{x^2})&=a(X_1,X_{x^2})+2b(X_x).
\endaligned
$$
 Then the $2$-cocycle $c=a-\delta b$ satisfies:
$$\aligned
c(X_1,X_x)&=a(X_1,X_x)-X_1\cdot b(X_x)=0,\\
c(X_1,X_{x^2})&=a(X_1,X_{x^2})-X_1\cdot b(X_{x^2})-b([X_1,X_{x^2}])=0,\\
c(X_1,X_{\theta})&=a(X_1,X_{\theta})-X_1\cdot b(X_{\theta})=0,\\
c(X_1,X_{x\theta})&=a(X_1,X_{x\theta})-X_1\cdot b(X_{x\theta})-b(X_{\theta})=0.
\endaligned
$$
This proves that $c$ is reduced.

Now, observe that the general coboundary relation $\delta c(X_1, X_{h_1}, X_{h_2}) = 0$ simplifies to:
\begin{equation}\label{hh}
X_1 \cdot c(X_{h_1}, X_{h_2}) = c([X_1, X_{h_1}], X_{h_2}) - (-1)^{h_1h_2} c([X_1,  X_{h_2}], X_{h_1}).
\end{equation}
Therefore, the relation $\delta c(X_1,X_x,X_{x^2})=0$ reduces to: $X_1\cdot c(X_x,X_{x^2})=0$, then we have $c(X_x,X_{x^2})=\sum c_\alpha \Omega^\alpha$ (with $c'_\alpha=0$) and then $c(X_x,X_{x^2})\in M$. Similarly, the relation $\delta c(X_1,X_x,X_{x\theta})=0$ gives $X_1\cdot c(X_x,X_{x\theta})=0$. The relation $\delta c(X_1,X_\theta,X_{x\theta})=0$ yields $X_1\cdot c(X_\theta,X_{x\theta})=c(X_\theta,X_{\theta})$.  Applying $X_1$ once more yields $X^2_1 \cdot c(X_\theta,X_{x\theta}) = X_1 \cdot c(X_\theta,X_{\theta}) = 0$, therefore $c(X_\theta,X_{x\theta}) \in M$. Using \eqref{hh} and the fact that $c(X_x,X_{x^2})$ and $c(X_x,X_{x\theta})$ belong to $M$, it easily follows that $c(X_\theta,X_{x^2})$ and $c(X_{x^2},X_{x\theta})$ are also in $M$.\\


The 2-cocycle $c$ is also differential since, if $c$ is a reduced 2-cocycle, its values are completely determined by the parity of its arguments.\\

Now, let ${X}_{h_i}$ denote the even fields associated with the even functions $h_i(x)$, and ${X}_{\phi_i}$ denote the odd fields associated with the odd functions $\phi_i(x)$.
 If $a$ is a differential 2-cocycle, it decomposes explicitly into three distinct parity sectors as follows:

\begin{enumerate}
    \item {Even-Even Sector:} For any two even generators, the 2-cocycle is expressed using standard Wronskian coefficients:
    $$
    a({X}_{h_1}, {X}_{h_2}) = \sum_{\alpha} \left[ (h_1 h'_2 - h_2 h'_1) A_\alpha + (h_1 h''_2 - h_2 h''_1) B_\alpha + (h'_1 h''_2 - h'_2 h''_1) C_\alpha \right] \Omega^\alpha.
    $$

    \item {Even-Odd Sector:} For one even and one odd generator, the cross-terms yield:
    $$
    a({X}_{h}, {X}_{\phi}) = \sum_{\alpha} \left[ (h \phi' - h' \phi) A_{\alpha,\text{odd}} + (h \phi'' - h'' \phi) B_{\alpha,\text{odd}} \right] \Omega^\alpha.
    $$

    \item {Odd-Odd Sector:} Due to the anti-commutation rule of the odd elements, the signs switch to a symmetric sum:
    $$
    a({X}_{\phi_1}, {X}_{\phi_2}) = \sum_{\alpha} \left[ (\phi_1 \phi'_2 + \phi'_1 \phi_2) A_{\alpha,\text{even}} + (\phi_1 \phi''_2 + \phi''_1 \phi_2) B_{\alpha,\text{even}} \right] \Omega^\alpha.
    $$
\end{enumerate}
Then, we choose the 1-cochain $b$ to be also differential, vanishing on $X_1$, and defined by components according to parity:
$$
b({X}_{h}) = \sum_{\alpha} \left[ h' E_\alpha + h'' F_\alpha \right] \Omega^\alpha, \quad b({X}_{\phi}) = \sum_{\alpha} \left[ \phi' E_{\alpha,\text{odd}} + \phi'' F_{\alpha,\text{odd}} \right] \Omega^\alpha.
$$

A direct line-by-line computation of the super-coboundary $$\delta b({X}_u, {X}_v) = {X}_u \cdot b({X}_v) - (-1)^{uv} {X}_v \cdot b({X}_u) - b([{X}_u, {X}_v])$$ yields:
$$
\aligned
\delta b({X}_{h_1}, {X}_{h_2}) =& \sum_{\alpha} \Big[ \left|\begin{matrix}h_1&h_2\\ h'_1&h'_2\end{matrix}\right|E'_\alpha +\left|\begin{matrix}h_1&h_2\\ h''_1&h''_2\end{matrix}\right| F'_\alpha\\+& \frac{1}{2} \left|\begin{matrix}h'_1&h'_2\\ h''_1&h''_2\end{matrix}\right|\Big(\sum_{i} (\alpha_i - t_i)E_{\alpha + \varepsilon_i} + (k - |\alpha| + 1)F_\alpha \Big) \Big] \Omega^\alpha. \\
\endaligned
$$

Similarly, evaluating $\delta b$ on the remaining cross-sectors and choosing $E'_\alpha = A_\alpha$, $F'_\alpha = B_\alpha$ (and their corresponding odd counterpart parameters),
 we successfully obtain a reduced and differential 2-cocycle $c = a - \delta b$ whose coefficients are strictly polynomial. We have proved:
\begin{align*}
Z^2(\mathfrak{osp}(1|2), \mathcal{D}_{\lambda,\mu}) &= Z_{\text{red}}^2(\mathfrak{osp}(1|2), M) + B^2(\mathfrak{osp}(1|2), \mathcal{D}_{\lambda,\mu}) \\
Z_{\text{diff}}^2(\mathfrak{osp}(1|2), \mathcal{D}_{\lambda,\mu}) &= Z_{\text{red}}^2(\mathfrak{osp}(1|2), M) + B_{\text{diff}}^2(\mathfrak{osp}(1|2), \mathcal{D}_{\lambda,\mu}) \\
Z^2(\mathfrak{osp}(1|2), M) &= Z_{\text{red}}^2(\mathfrak{osp}(1|2), M) + B^2(\mathfrak{osp}(1|2), M)
\end{align*}


\subsection*{Case $j=3$}
It is also differential. If $c \in Z^3(\mathfrak{osp}(1|2), \mathcal{D}_{\lambda,\mu})$ is a 3-cocycle, and $b$ is a 2-cochain such that $b(X_1, \cdot) = 0$ and $X_1 \cdot b(u, v) = c(X_1, u, v)$, then $c = \delta b$. If $c$ has its values in $M$ and is differential, it can be written using the $3 \times 3$ block-determinants:
$$c({X}_{H_1}, {X}_{H_2}, {X}_{H_3}) = \left[ \begin{vmatrix} h_1 & h_2 & h_3 \\ h'_1 & h'_2 & h'_3 \\ h''_1 & h''_2 & h''_3 \end{vmatrix} + \theta \begin{vmatrix} \phi_1 & \phi_2 & \phi_3 \\ \phi'_1 & \phi'_2 & \phi'_3 \\ \phi''_1 & \phi''_2 & \phi''_3 \end{vmatrix} \right] \sum_{\alpha} E_\alpha \Omega^\alpha,$$
where $H_i=h_i+\theta\phi_i$
Then, the corresponding 2-cochain $b$ is also differential and takes the form:
$$b({X}_{H_1}, {X}_{H_2}) = \left[ \begin{vmatrix} h'_1 & h'_2 \\ h''_1 & h''_2 \end{vmatrix} + \theta \begin{vmatrix} \phi'_1 & \phi'_2 \\ \phi''_1 & \phi''_2 \end{vmatrix} \right] \sum_{\alpha} F_\alpha \Omega^\alpha$$
with $F'_\alpha = E_\alpha$. Consequently, the coefficients of the 2-cochain $b$ are uniquely determined and confined within the polynomial sub-module $M$. We have proved the trivialization of the spaces for degree 3:
\begin{align*}
Z^3(\mathfrak{osp}(1|2), \mathcal{D}_{\lambda,\mu}) &= B^3(\mathfrak{osp}(1|2), \mathcal{D}_{\lambda,\mu}), \\
Z_{\text{diff}}^3(\mathfrak{osp}(1|2), \mathcal{D}_{\lambda,\mu}) &= B_{\text{diff}}^3(\mathfrak{osp}(1|2), \mathcal{D}_{\lambda,\mu}), \\
Z^3(\mathfrak{osp}(1|2), M) &= B^3(\mathfrak{osp}(1|2), M).
\end{align*}

\subsection{Case $j > 3$}
There is no non-trivial $j$-cochain, and there is nothing to prove.

By definition,  we have $B^{j}(\mathfrak{osp}(1|2),M)\subset Z^{j}(\mathfrak{osp}(1|2),M)\cap B^{j}(\mathfrak{osp}(1|2),\mathfrak D_{{\lambda},\mu})$ and $B^{j}(\mathfrak{osp}(1|2),M)\subset Z^{j}(\mathfrak{osp}(1|2),M)\cap B^{j}_{\mathrm{diff}}(\mathfrak{osp}(1|2),\mathfrak D_{{\lambda},\mu}).$ So, let us prove the inverse inclusion. This is holding if $j\geq 3$ thanks to the proof of Lemma \ref{cocycles}, this holds also if $j=0$ since the only coboundary is then 0.\\


\noindent
{$\mathbf{Case~} j=1$}

Let $c$ be in $Z^{1}(\mathfrak{osp}(1|2),M)$ and $b$ in $\mathfrak D_{{\lambda},\mu}$ such that $c=\delta b$ (then $\delta b$ is in  $B^{1}_{\mathrm{diff}}(\mathfrak{osp}(1|2),\mathfrak D_{{\lambda},\mu})=B^{1}(\mathfrak{osp}(1|2),\mathfrak D_{{\lambda},\mu})$). Then $X_1\cdot b=c(X_1)\in M$, therefore $b$ itself belongs to $M$, and $c$ in $B^{1}(\mathfrak{osp}(1|2),M)$. This proves the lemma if $j=1$.\\

\noindent
{$\mathbf{Case~} j=2$}

Let $c$ be in $Z^{2}(\mathfrak{osp}(1|2),M)$ and $b$ a 1-cochain such that $c=\delta b$. There is $a$ in $\mathfrak D_{{\lambda},\mu}$ such that $b(X_1)=X_1\cdot a$, we replace $b$ by the reduced 1-cochain $b'=b-\delta a$, which is differential if $b$ was, we  get:
$$\aligned
c(X_1,X_x)&=X_1\cdot b'(X_x),\\
c(X_1,X_{x^2})&=X_1\cdot b'(X_{x^2})-2b(X_x).
\endaligned
$$
This implies that $b(X_1)$ and $b(X_x)$ belong to $M$, $b\in B^{2}(\mathfrak{osp}(1|2),M)$, and proves the lemma if $j=2$.\\

<!-- thm-2: body_md -->
\label{coho}{\rm (The groups $\mathrm{H}^j(\mathfrak{osp}(1|2),M)$)}

Let us denote by $\hbox{ker} A$ (respectively $\hbox{ker} B$) the subspaces of $M$, kernel of the morphism $v\mapsto Av$ (respectively $v\mapsto Bv$) in $M$. Then we have the following linear isomorphisms:
\begin{itemize}
\item[\sl(i)]  $\mathrm{H}^0(\mathfrak{osp}(1|2),M)=\hbox{ker} A\cap \hbox{ker} B$.
\item[\sl(ii)]  $\mathrm{H}^1(\mathfrak{osp}(1|2),M)\simeq(\hbox{ker} A\cap\hbox{ker} B)\oplus ((\hbox{ker} A)^{-\frac{1}{2}} /B((\hbox{ker} A)^0))$.
\item[\sl(iii)]  $\mathrm{H}^2(\mathfrak{osp}(1|2),M)\simeq(\hbox{ker} A)^{-\frac{1}{2}}/B((\hbox{ker} A)^0)$.
\item[\sl(iv)] $\mathrm{H}^j(\mathfrak{osp}(1|2),M)=0$ if $j>2$.
\end{itemize}
