<!-- theorem-1: body_md -->
\label{thm:main} Let \(A\) be a commutative associative algebra
		over a field of positive characteristic,
		generated by elements \(a_1,\ldots,a_m\). 	Let	$d_1,\ldots,d_n\in \operatorname{Der} A$.	If every commutator \(\rho\) in
		\(d_1,\ldots,d_n\) acts finitely on each 		generator \(a_i\),  \(i=1,\ldots,m\), then the associative  subalgebra $U$ of \(\operatorname{End}_{\mathbb F}A\)  generated by $d_1,\ldots,d_n$ is finite-dimensional.
<!-- end theorem-1 -->
<!-- lemma-1: body_md -->
\label{lem:polynomial}
		Let \(\rho\) be a commutator in \(d_1,\ldots,d_n\). Put
		\[
		S_i=
		\dim_\mathbb{F}\sum_{k=0}^{\infty}\mathbb{F}\rho^k a_i,
		\qquad
		1\le i\le m, \qquad 		\text{and} \qquad	n(\rho)=S_1+\cdots+S_m.
		\]
		Then there exists a nonzero polynomial of the form
		\[
		f_\rho(t)
		=
		\alpha_0t+\alpha_1t^p+\alpha_2t^{p^2}
		+\cdots+\alpha_{n(\rho)}t^{p^{n(\rho)}} \qquad	\text{such that} \qquad 	f_\rho(\rho)=0.	\]	Moreover, one may assume that \(f_\rho(t)\) is monic.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Since	$f_\rho(\rho)\in\operatorname{Der} A,$ 	it is enough to find \(f_\rho(t)\) such that	$f_\rho(\rho)a_i=0,$	$	1\le i\le m.$	Consider the vector space
		\[
		V=\sum_{k=0}^{n(\rho)}\mathbb{F}\, t^{p^k}.
		\]
		Then	$\dim_\mathbb{F} V=n(\rho)+1.$	For \(1\le i\le m\), define a linear mapping	$\varphi_i\colon V\to A,$ $\varphi_i(f(t))=f(\rho)a_i.$	By the definition of \(S_i\), we have	$\operatorname{codim}_V \operatorname{Ker}\varphi_i\le S_i.$	Consequently,
		\[
		\operatorname{codim}_V\bigcap_{i=1}^{m}\operatorname{Ker}\varphi_i
		\le
		S_1+\cdots+S_m
		=
		n(\rho).
		\]
		Since \(\dim_\mathbb{F} V=n(\rho)+1\), it follows that
		\[
		\bigcap_{i=1}^{m}\operatorname{Ker}\varphi_i\ne 0.
		\]
		Choose a nonzero polynomial
		\[
		f_\rho(t)\in \bigcap_{i=1}^{m}\operatorname{Ker}\varphi_i.
		\]
	
	Then \(f_\rho(\rho)a_i=0\) for \(1\le i\le m\). Hence,
	\(f_\rho(\rho)\) vanishes on the generators of \(A\).
	Since \(f_\rho(\rho)\) is a derivation, we obtain
	\(f_\rho(\rho)=0\).
	Multiplying by a nonzero scalar if necessary, we may assume
	that \(f_\rho(t)\) is monic.
	
	If \(\deg f_\rho(t)=p^j\) for some \(j<n(\rho)\), then
	the polynomial
	\[
	g_\rho(t)=f_\rho(t)^{p^{n(\rho)-j}}
	\]
	is monic of degree \(p^{n(\rho)}\). Moreover,
	\[
	g_\rho(\rho)
	=f_\rho(\rho)^{p^{n(\rho)-j}}=0.
	\]
	Thus, we may assume that \(f_\rho(t)\) is monic
	of degree \(p^{n(\rho)}\).
<!-- end proof-1 -->
<!-- proposition-1: body_md -->
\label{prop:zelmanov}
	
	Let \(R\) be an associative \(\mathbb{F}\)-algebra generated by
	elements \(u_1,\ldots,u_n\), and let \(L\subseteq R\) be the Lie
	subalgebra generated by these elements. Suppose that
	\begin{enumerate}
		\item[(1)] \(L\) satisfies a polynomial identity;
		
		\item[(2)] for every commutator \(\rho\) in \(u_1,\ldots,u_n\),
		there exists an integer \(N_\rho\geq 1\) such that
		\[
		\rho^{N_\rho}=0.
		\]
	\end{enumerate}
		Then the algebra \(R\) is nilpotent.
<!-- end proposition-1 -->
<!-- remark-1: body_md -->
In fact, Theorem 1.1 of \cite{Zelmanov} asserts that if
		$L$ satisfies a polynomial identity and the adjoint
		operator $\operatorname{ad}(\rho)$ is nilpotent for every commutator
		$\rho$ in $u_1,\ldots,u_n$, then the Lie algebra $L$
		is finite-dimensional.
		
		If $\rho^n=0$ in the associative algebra $R$, then	$\operatorname{ad}(\rho)^{2n-1}=0.$	Proposition~\ref{prop:zelmanov} then follows from
		this theorem and the Poincar\'e--Birkhoff--Witt theorem.
<!-- end remark-1 -->
<!-- proof-2: body_md -->
We show that the algebra \(U\subseteq End_{\mathbb{F}} A\) generated  by $d_{1},\ldots, d_{n}$ is finite-dimensional. It suffices to prove
		that \(U\) is spanned by products	$d_{i_1}\cdots d_{i_s}$ of length $		s<N.$
		
		Consider an arbitrary product	$	d_{i_1}\cdots d_{i_N}.$	Since
		\[
		\mathbb{F}\langle x_1,\ldots,x_n\rangle^N
		\subseteq I,
		\]
		the corresponding word \(x_{i_1}\cdots x_{i_N}\)
		belongs to \(I\) and can therefore be expressed as a
		linear combination of elements involving generators of \(I\):
		\[
		\begin{aligned}
			x_{i_1}\cdots x_{i_N}
			={}&
			\sum \alpha\,
			x_{j_1}\cdots x_{j_k}\,
			\rho^{p^{n(\rho)}} x_{\mu_1}\cdots x_{\mu_s}
				+
			\sum \beta\,
			x_{\nu_1}\cdots x_{\nu_r}\,
			p(\sigma_1,\ldots,\sigma_l)\,
			x_{w_1}\cdots x_{w_t},
		\end{aligned}
		\]
		where \(\alpha,\beta\in \mathbb{F}\).	The preceding equality holds in the free  associative algebra
		\(\mathbb{F}\langle x_1,\ldots,x_n\rangle\).
		
		Since the elements
		\[
		\rho(x_1,\ldots,x_n)^{p^{n(\rho)}},
		\qquad
		P(\sigma_1,\ldots,\sigma_l)
		\]
		are homogeneous, it follows that  each
		summand on the right-hand side has degree
		\[
		k+p^{n(\rho)}\deg(\rho)+s
		=
		r+\deg P(\sigma_1,\ldots,\sigma_l)+t
		=
		N.
		\]
	After substituting	$x_i\mapsto d_i,$ $ 1\leq i\leq n,$	we get 	\[
d_{i_1}\cdots d_{i_N}= \sum	\alpha\,
	d_{j_1}\cdots d_{j_k}
	\rho(d_1,\ldots,d_n)^{p^{n(\rho)}}
	d_{\mu_1}\cdots d_{\mu_s},
	\qquad
	\alpha\in\mathbb{F}.
	\]

	By Lemma~\ref{lem:polynomial},	$f_\rho(\rho)=0.$	Since \(f_\rho(t)\) is monic and has the form
	\[
	f_\rho(t)=
	\alpha_0t+\alpha_1t^p+\cdots+
	\alpha_{n(\rho)-1}t^{p^{n(\rho)-1}}
	+t^{p^{n(\rho)}},
	\]
	it follows that
	\[
	\rho^{p^{n(\rho)}}
	\]
	is a linear combination of
	\[
	\rho,\rho^p,\rho^{p^2},\ldots,
	\rho^{p^{n(\rho)-1}}.
	\]
	Consequently, every term of the form
	\[
	d_{j_1}\cdots d_{j_k}
	\rho(d_1,\ldots,d_n)^{p^{n(\rho)}}
	d_{\mu_1}\cdots d_{\mu_s}
	\]
	can be expressed as a linear combination of terms of the form
	\[
	d_{j_1}\cdots d_{j_k}
	\rho(d_1,\ldots,d_n)^{p^t}
	d_{\mu_1}\cdots d_{\mu_s},
	\qquad
	0\leq t<n(\rho).
	\]

	Thus, every product	$d_{i_1}\cdots d_{i_N}$	is a linear combination of products of length less than  \(N\). 	Since \(I\) is homogeneous, the same reduction applies to every product of length at least \(N\). Thus,  \(U\) is spanned by all products \(d_{i_1}\cdots d_{i_s}\) with \(1\le s<N\).
		There are only finitely many such products. Hence,	$\dim_{\mathbb{F}} U<\infty.$	This completes the proof.
