\begin{equation}\label{eq:sc-intro-metric-cotype}
\sum_{j=1}^{n}\mathbb E_x d_{\mathcal M}\left(f\left(x+\frac m2e_j\right),f(x)\right)^q\leq\Gamma^qm^q\mathbb E_{x,\varepsilon}d_{\mathcal M}(f(x+\varepsilon),f(x))^q,
\end{equation}
$x$ and $\varepsilon$ are independent and uniform on $\mathbb Z_m^n$ and $\{-1,0,1\}^n$, respectively, and addition on the torus is modulo $m$.

<!-- lemma-1: body_md -->
\label{lem:cross}
If $S,T,D\in M_{d}$ are positive and $S+T\leq D$, then
\begin{equation}
 \left\lVert S^{1/2}D^{\dagger}T^{1/2}\right\rVert_{\infty}\leq\frac12.
 \label{eq:cross}
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Let
\[
 L=\bigl[\,(D^{\dagger})^{1/2}S^{1/2}
            \quad(D^{\dagger})^{1/2}T^{1/2}\,\bigr],
\]
viewed as an operator from $\mathbb{C}^{d}\oplus\mathbb{C}^{d}$ to $\mathbb{C}^{d}$. Then
\[
 LL^{*}=(D^{\dagger})^{1/2}(S+T)(D^{\dagger})^{1/2}
 \leq(D^{\dagger})^{1/2}D(D^{\dagger})^{1/2}
 =P_{D}\leq I_{d}.
\]
Consequently $\left\lVert L\right\rVert_{\infty}\leq1$ and
$0\leq L^{*}L\leq I_{2d}$. The eigenvalues of $L^{*}L$ lie in $[0,1]$,
so
\[
 \left\lVert L^{*}L-\tfrac12I_{2d}\right\rVert_{\infty}\leq\frac12.
\]
For $u,v\in\mathbb{C}^{d}$, block multiplication gives
\[
\left|\left\langle S^{1/2}D^{\dagger}T^{1/2}v,u\right\rangle\right|=\left|\left\langle\left(L^{*}L-\tfrac12I_{2d}\right)(0,v),(u,0)\right\rangle\right|\leq\frac12\left\lVert u\right\rVert_{\mathbb{C}^{d}}\left\lVert v\right\rVert_{\mathbb{C}^{d}}.
\]
Taking the supremum over unit vectors proves \eqref{eq:cross}.

<!-- lemma-2: body_md -->
\label{lem:matrix}
The operation $\Phi_{mix}$ is invariant under every permutation of its arguments.
For each fixed $B,C\in M_{d}^{\mathrm{sa}}$ there exists a map
$V_{B,C}:M_{d}^{\mathrm{sa}}\to M_{d}$ such that, for every $A\in M_{d}^{\mathrm{sa}}$,
\begin{align}
 \Phi_{mix}(A,B,C)
 &=B+V_{B,C}(A)|C-B|^{1/2}
      +|C-B|^{1/2}V_{B,C}(A)^{*},
 \label{eq:factor}\\
 \left\lVert V_{B,C}(A)\right\rVert_{2}
 &\leq\frac12\left\lVert A-B\right\rVert_{1}^{1/2}.
 \label{eq:Zbound}
\end{align}
In particular,
\begin{equation}
 \left\lVert \Phi_{mix}(A,B,C)-B\right\rVert_{1}
 \leq\sqrt{\left\lVert A-B\right\rVert_{1}\left\lVert C-B\right\rVert_{1}}.
 \label{eq:medianbound}
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Interchanging $B,C$ in \eqref{eq:median} interchanges its two summands
and leaves $D$ unchanged. To interchange $A,B$, subtract the two
defining formulas. Since $|A-B|=|B-A|$, this gives
\[
 \begin{aligned}
 &\Phi_{mix}(A,B,C)-\Phi_{mix}(B,A,C)\\
 =&A-B+\operatorname{Re}\bigl[
       (B-A)D^{\dagger}\left(|C-A|+|C-B|\right)\\
 &+\left((C-A)-(C-B)\right)D^{\dagger}|A-B|\bigr]\\
 =&A-B+\operatorname{Re}\left[
       (B-A)D^{\dagger}
       \left(|C-A|+|C-B|+|A-B|\right)\right]\\
 =&A-B+\operatorname{Re}\left[(B-A)D^{\dagger}D\right]=0.
 \end{aligned}
\]
The last equality follows from $D^{\dagger}D=P_{D}$ and
$(B-A)P_{D}=B-A$, by \eqref{eq:support}. The transpositions of the
first two and the last two arguments generate every permutation, proving
symmetry.

For fixed $B,C$, define, with $D=D(A,B,C)$,
\begin{equation}\label{eq:Z}
 V_{B,C}(A)=\frac12\bigl[(A-B)D^{\dagger}|C-B|^{1/2}+|A-B|D^{\dagger}|C-B|^{1/2}\operatorname{sgn}(C-B)\bigr].
\end{equation}
The spectral functions $|C-B|^{1/2}$ and $\operatorname{sgn}(C-B)$ commute and satisfy
\[
 |C-B|^{1/2}\operatorname{sgn}(C-B)|C-B|^{1/2}=C-B.
\]
Therefore
\[
V_{B,C}(A)|C-B|^{1/2}
 =\frac12\bigl[(A-B)D^{\dagger}|C-B|+|A-B|D^{\dagger}(C-B)\bigr].
\]
Adding the adjoint, using $\operatorname{Re}(T^{*})=\operatorname{Re}T$ and the definition in \eqref{eq:median} proves \eqref{eq:factor}.

By \eqref{eq:cross}, applied to $|A-B|+|C-B|\leq D$,
\[
 \left\lVert |A-B|^{1/2}D^{\dagger}|C-B|^{1/2}\right\rVert_{\infty}\leq\frac12.
\]
By \eqref{eq:Z} and \eqref{eq:holder}, we have
\begin{align*}
\left\lVert V_{B,C}(A)\right\rVert_{2}&\leq \frac{1}{2}\left\lVert (A-B)D^{\dagger}|C-B|^{1/2}\right\rVert_{2}\\
&\quad+\frac{1}{2}\left\lVert |A-B|D^{\dagger}|C-B|^{1/2}\operatorname{sgn}(C-B)\right\rVert_{2}\\
&\leq \left\lVert |A-B|D^{\dagger}|C-B|^{1/2}\right\rVert_{2}\\
&\leq \||A-B|^{1/2}\|_{2}\cdot\left\lVert |A-B|^{1/2}D^{\dagger}|C-B|^{1/2}\right\rVert_{\infty}\\
&\leq \frac{1}{2}\left\lVert |A-B|^{1/2}\right\rVert_{2}=\frac{1}{2} \left\lVert A-B\right\rVert_{1}^{1/2},
\end{align*}
which verifies \eqref{eq:Zbound}. Applying \eqref{eq:holder} to
\eqref{eq:factor} now gives
\[
 \left\lVert \Phi_{mix}(A,B,C)-B\right\rVert_{1}
 \leq2\left\lVert V_{B,C}(A)\right\rVert_{2}\left\lVert |C-B|^{1/2}\right\rVert_{2}
 \leq\sqrt{\left\lVert A-B\right\rVert_{1}\left\lVert C-B\right\rVert_{1}},
\]
as required.

<!-- proposition-1: body_md -->
\label{prop:probability}
Let $X$ be a random matrix in $M_d^{\mathrm{sa}}$ with finite range, and let $B,C$
be independent copies of $X$. For every $A_0\in M_d^{\mathrm{sa}}$,
\begin{equation}
 \left\lVert \mathbb{E}\Phi_{mix}(X,B,C)-A_0\right\rVert_{1}
 \leq 3\mathbb{E}\left\lVert X-A_0\right\rVert_{1}.
 \label{eq:approx}
\end{equation}
Suppose that $X$ and real random variables $\xi_1,\ldots,\xi_n$ are defined
on the same probability space, that $B,C$ are independent of that space
and of each other, and that, for some $\kappa\geq0$,
\begin{equation}
 \mathbb{E}\xi_j=0,\qquad \mathbb{E}(\xi_i\xi_j)=\kappa^2\delta_{ij}.
 \label{eq:scores}
\end{equation}
Here $\delta_{ij}$ is $1$ if $i=j$ and $0$ otherwise. Then
\begin{equation}
 \left(\sum_{j=1}^n
   \left\lVert \mathbb{E}\bigl[\xi_j\Phi_{mix}(X,B,C)\bigr]\right\rVert_{1}^2\right)^{1/2}
 \leq\kappa\mathbb{E}\left\lVert X-B\right\rVert_{1}.
 \label{eq:scorebound}
\end{equation}
No independence between $X$ and the $\xi_j$ is required.
<!-- end proposition-1 -->
<!-- proof-3: body_md -->
By independence of $X$ and $C$ for fixed $B$, followed by the scalar
Cauchy--Schwarz inequality,
\begin{align}
 \mathbb{E}\sqrt{\left\lVert X-B\right\rVert_{1}\left\lVert C-B\right\rVert_{1}}
 &=\mathbb{E}_B\left(\mathbb{E}_X\sqrt{\left\lVert X-B\right\rVert_{1}}\right)^2\leq\mathbb{E}_B\mathbb{E}_X\left\lVert X-B\right\rVert_{1}.
 \label{eq:meanroot}
\end{align}
Consequently, \eqref{eq:medianbound} and the triangle inequality give
\begin{align*}
 \left\lVert \mathbb{E}\Phi_{mix}(X,B,C)-A_0\right\rVert_{1}
 &\leq\mathbb{E}\left\lVert \Phi_{mix}(X,B,C)-B\right\rVert_{1}+\mathbb{E}\left\lVert B-A_0\right\rVert_{1}\\
 &\leq\mathbb{E}\left\lVert X-B\right\rVert_{1}+\mathbb{E}\left\lVert B-A_0\right\rVert_{1}\\
 &\leq\mathbb{E}\left\lVert X-A_0\right\rVert_{1}+2\mathbb{E}\left\lVert B-A_0\right\rVert_{1}
 =3\mathbb{E}\left\lVert X-A_0\right\rVert_{1}.
\end{align*}
This proves \eqref{eq:approx}.

Assume without loss of generality that $\kappa>0$. For fixed
$B,C$, use $V_{B,C}(X)$ from Lemma~\ref{lem:matrix} and define only the
coefficients needed for the orthogonality argument:
\[
 Y_j(B,C)=\mathbb{E}_\omega\bigl[\xi_j(\omega)V_{B,C}(X(\omega))\bigr],
 \qquad 1\leq j\leq n.
\]
The subscript $\omega$ means that $B,C$ remain fixed. Expanding the
Hilbert--Schmidt norm, using that the $\xi_j$ are real and
$\mathbb{E}(\xi_i\xi_j)=\kappa^2\delta_{ij}$, gives
\begin{align*}
 0&\leq\mathbb{E}_\omega\left\lVert
 V_{B,C}(X)-\kappa^{-2}\sum_{j=1}^n\xi_jY_j(B,C)
 \right\rVert_2^2\\
 &=\mathbb{E}_\omega\left\lVert V_{B,C}(X)\right\rVert_{2}^2
   -\kappa^{-2}\sum_{j=1}^n\left\lVert Y_j(B,C)\right\rVert_{2}^2.
\end{align*}
In particular, \eqref{eq:Zbound} implies
\begin{equation}
 \sum_{j=1}^n\left\lVert Y_j(B,C)\right\rVert_{2}^2
 \leq\kappa^2\mathbb{E}_X\left\lVert V_{B,C}(X)\right\rVert_{2}^2
 \leq\frac{\kappa^2}{4}\mathbb{E}_X\left\lVert X-B\right\rVert_{1}.
 \label{eq:bessel}
\end{equation}
The equality defining $Y_j$ averages $\xi_j$ jointly with $X$; the last
expectation in \eqref{eq:bessel} only uses the marginal distribution of $X$.

Since $\mathbb{E}\xi_j=0$ and $\xi_j$ is real, \eqref{eq:factor} yields
\[
 \mathbb{E}_\omega[\xi_j\Phi_{mix}(X,B,C)]
 =Y_j(B,C)|C-B|^{1/2}+|C-B|^{1/2}Y_j(B,C)^*.
\]
For any integrable finite family of random matrices $(T_j)_{j=1}^n$,
\begin{equation}
 \left(\sum_j\left\lVert \mathbb{E} T_j\right\rVert_{1}^2\right)^{1/2}
 \leq\mathbb{E}\left(\sum_j\left\lVert T_j\right\rVert_{1}^2\right)^{1/2}.
 \label{eq:familytriangle}
\end{equation}
Indeed, $\left\lVert \mathbb{E} T_j\right\rVert_{1}\leq\mathbb{E}\left\lVert T_j\right\rVert_{1}$, and the Euclidean
triangle inequality for integrals gives the remaining inequality. Applying
\eqref{eq:familytriangle}, then \eqref{eq:holder} and \eqref{eq:bessel},
we obtain
\begin{align*}
& \left(\sum_{j=1}^n\left\lVert \mathbb{E}[\xi_j\Phi_{mix}(X,B,C)]\right\rVert_{1}^2\right)^{1/2}\\
 \leq& 2\mathbb{E}_{B,C}\left[
   \left\lVert |C-B|^{1/2}\right\rVert_{2}
   \left(\sum_{j=1}^n\left\lVert Y_j(B,C)\right\rVert_{2}^2\right)^{1/2}\right]\\
 \leq&\kappa\mathbb{E}_{B,C}
   \sqrt{\left\lVert C-B\right\rVert_{1}\,\mathbb{E}_X\left\lVert X-B\right\rVert_{1}}\\
 \leq&\kappa\mathbb{E}_B
   \sqrt{\bigl(\mathbb{E}_C\left\lVert C-B\right\rVert_{1}\bigr)
               \bigl(\mathbb{E}_X\left\lVert X-B\right\rVert_{1}\bigr)}\\
 =&\kappa\mathbb{E}_B\mathbb{E}_X\left\lVert X-B\right\rVert_{1}.
\end{align*}
For the penultimate line, Cauchy--Schwarz is applied only to $C$, with
$B$ fixed. The last equality holds because $C$ and $X$ have the same
distribution.

<!-- lemma-3: body_md -->
\label{lem:kernel}
The function
\begin{equation}
 \psi(t)=84|t|^5(1-|t|)^2\mathbf1_{[-1,1]}(t)
 \label{eq:kernel}
\end{equation}
is an even probability density in $C_c^1(\mathbb{R})$, and
\begin{equation}
 \int_\mathbb{R} |t|\psi(t)\,dt=\frac23,
 \qquad
 \int_{\{\psi>0\}}\frac{\psi'(t)^2}{\psi(t)}\,dt=70.
 \label{eq:kernelmoments}
\end{equation}
Let $W=(W_1,\ldots,W_n)$ have independent coordinates with density $\psi$.
For $1\leq j\leq n$, define
\begin{equation}
 \xi_j(w)=
 \begin{cases}
 -\psi'(w_j)/\psi(w_j),&\psi(w_j)>0,\\
 0,&\psi(w_j)=0.
 \end{cases}
 \label{eq:xidef}
\end{equation}
Then
\begin{equation}
 \mathbb{E}\xi_j(W)=0,\qquad \mathbb{E}[\xi_i(W)\xi_j(W)]=70\delta_{ij}.
 \label{eq:kernelcovariance}
\end{equation}
If $U$ is independently uniform on $[0,1)^n$, then
$\lfloor U+W\rfloor$ is uniform on $\{-1,0,1\}^n$. Here the floor
function is applied to each coordinate.
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
The function and its derivative vanish at $\pm1$, and the factor $|t|^5$
makes the derivative continuous at zero. Hence $\psi\in C_c^1(\mathbb{R})$.
For $0<t<1$,
\[
 \psi'(t)=84t^4(1-t)(5-7t),\qquad
 \frac{\psi'(t)^2}{\psi(t)}=84t^3(5-7t)^2.
\]
Evenness and integration give
\begin{align*}
 \int_\mathbb{R}\psi(t)\,dt
 &=168\left(\frac16-\frac27+\frac18\right)=1,\\
 \int_\mathbb{R}|t|\psi(t)\,dt
 &=168\left(\frac17-\frac28+\frac19\right)=\frac23,\\
 \int_{\{\psi>0\}}\frac{\psi'(t)^2}{\psi(t)}\,dt
 &=168\left(\frac{25}{4}-\frac{70}{5}+\frac{49}{6}\right)=70.
\end{align*}
Since $\{\psi(W_j)=0\}$ has probability zero,
\[
 \mathbb{E}\xi_j(W)=-\int_\mathbb{R}\psi'(t)\,dt=0,
 \qquad \mathbb{E}\xi_j(W)^2=70.
\]
For $i\ne j$, independence yields
$\mathbb{E}[\xi_i(W)\xi_j(W)]=\mathbb{E}\xi_i(W)\mathbb{E}\xi_j(W)=0$, which proves the assertion \eqref{eq:kernelcovariance}.

Fix $w\in[-1,1]$. The subsets of $[0,1)$ on which
$\lfloor u+w\rfloor$ equals $1,-1,0$ have lengths, respectively,
\[
 \max\{w,0\},\qquad \max\{-w,0\},\qquad 1-|w|.
\]
Therefore, by evenness and \eqref{eq:kernelmoments},
\[
 \mathbb{P}(\lfloor U_j+W_j\rfloor=1)
 =\mathbb{P}(\lfloor U_j+W_j\rfloor=-1)=\frac12\mathbb{E}|W_j|=\frac13,
\]
and the remaining probability is $1-\mathbb{E}|W_j|=1/3$.
The pairs $(U_j,W_j)$ are independent across $j$. Thus, for every
$\varepsilon\in\{-1,0,1\}^n$,
\begin{equation}
 \mathbb{P}(\lfloor U+W\rfloor=\varepsilon)
 =\prod_{j=1}^n\mathbb{P}(\lfloor U_j+W_j\rfloor=\varepsilon_j)=3^{-n}.
 \label{eq:productrounding}
\end{equation}

<!-- proposition-2: body_md -->
\label{prop:smoothing}
The function $g$ in \eqref{eq:smoothing} belongs to
$C^1(\mathbb{R}^n;M_d^{\mathrm{sa}})$. For every $z\in\mathbb{R}^n$, the following two inequalities hold
\begin{equation}\label{eq:localapprox}
 \left\lVert g(z)-F(z)\right\rVert_{1}\leq3\mathbb{E}_W\left\lVert F(z+W)-F(z)\right\rVert_{1},
\end{equation}
and
\begin{equation}\label{eq:localgradient}
\left(\sum_{j=1}^n\left\lVert \partial_jg(z)\right\rVert_{1}^2\right)^{1/2}\leq6\sqrt{70}\,\mathbb{E}_W\left\lVert F(z+W)-F(z)\right\rVert_{1}.
\end{equation}
<!-- end proposition-2 -->
<!-- proof-5: body_md -->
Suppose without loss of generality that $\mathrm{rang}(F)=\{A_{1},\cdots,A_{q}\}\subseteq M^{\mathrm{sa}}_{d}$. Define density as follows
\[
 \rho(w)=\prod_{i=1}^n\psi(w_i),\qquad
 p_a(z)=\int_{\mathbb{R}^n}\mathbf1_{\{F(y)=A_a\}}\rho(y-z)\,dy,\qquad a\in \{1,\cdots, q\}.
\]
Changing variables gives $p_a(z)=\mathbb{P}(F(z+W)=A_a)$ for each $a\in \{1,\cdots, q\}$. In particular,
$p_a\geq0$ and $\sum_{a\in \{1,\cdots,q\}} p_a=1$.

By independence of the three copies and the fact that $F$ has finite range, we have
\begin{equation}\label{summation representation 1}
g(z)=\sum_{a,b,c=1}^q p_a(z)p_b(z)p_c(z)\Phi_{mix}(A_a,A_b,A_c).
\end{equation}

We next prove that $p_{a}\in C^{1}(\mathbb{R}^{n})$
for every $a\in\{1,\ldots,q\}$.
Recall that
\[
p_{a}(z)=
\int_{\mathbb{R}^{n}}
\mathbf{1}_{\{F(y)=A_{a}\}}\rho(y-z)\,dy,
\qquad
\rho(w)=\prod_{i=1}^{n}\psi(w_{i}).
\]
By Lemma \ref{lem:kernel}, $\psi\in C_{c}^{1}(\mathbb{R})$ is supported in $[-1,1]$. Consequently, $\rho=\prod_{i=1}^n\psi(w_i)\in C_{c}^{1}(\mathbb{R}^{n})$, and
\[
\partial_{j}\rho(w)=\psi^{\prime}(w_{j})\prod_{\substack{i=1\\i\neq j}}^{n}\psi(w_{i}),
\qquad 1\leq j\leq n.
\]
In particular, both $\rho$ and its first partial derivatives vanish outside $[-1,1]^{n}$.

Fix $z_{0}\in\mathbb{R}^{n}$ and
$j\in\{1,\ldots,n\}$, and consider the cube
\[
K=z_{0}+[-2,2]^{n}.
\]
For $0<|v|<1$, we have
\[
\frac{p_{a}(z_{0}+ve_{j})-p_{a}(z_{0})}{v}=\int_{\mathbb{R}^{n}}\mathbf{1}_{\{F(y)=A_{a}\}}\frac{\rho(y-z_{0}-ve_{j})-\rho(y-z_{0})}{v}\,dy.
\]
Since $\rho$ is continuously differentiable,
the fundamental theorem of calculus gives
\[
\frac{\rho(y-z_{0}-ve_{j})-\rho(y-z_{0})}{v}=-\int_{0}^{1}
\partial_{j}\rho(y-z_{0}-tve_{j})\,dt.
\]
For every $t\in[0,1]$ and $|v|<1$, the support of the function $y\mapsto \partial_{j}\rho(y-z_{0}-tve_{j})$ is contained in $z_{0}+tve_{j}+[-1,1]^{n}
\subseteq K$. Therefore we have
\[
\left|
\mathbf{1}_{\{F(y)=A_{a}\}}\left(\frac{\rho(y-z_{0}-ve_{j})-\rho(y-z_{0})}{v}\right)\right|\leq\left\|\partial_{j}\rho\right\|_{\infty}\mathbf{1}_{K}(y),
\]
where $K$ is a compact subset in $\mathbb{R}^{n}$. Moreover, for every fixed $y\in\mathbb{R}^{n}$,
\[
\lim_{v\to0}
\frac{\rho(y-z_{0}-ve_{j})-\rho(y-z_{0})}{v}=-\partial_{j}\rho(y-z_{0}),
\]
the dominated convergence entails that
\[
\partial_{j}p_{a}(z_{0})=-\int_{\mathbb{R}^{n}}\mathbf{1}_{\{F(y)=A_{a}\}}\partial_{j}\rho(y-z_{0})\,dy.
\]
Since $z_{0}$ and $j$ were arbitrary, we obtain
\begin{equation}\label{integral representation 1}
\partial_{j}p_{a}(z)=-\int_{\mathbb{R}^{n}}\mathbf{1}_{\{F(y)=A_{a}\}}\partial_{j}\rho(y-z)\,dy
\end{equation}
for every $z\in\mathbb{R}^{n}$ and
$j\in\{1,\ldots,n\}$.

It remains to verify the continuity of these partial derivatives. Suppose that $\{z^{(k)}\}_{k=1}^{\infty}\subseteq \mathbb{R}^{n}$ such that $z^{(k)}\to z_{0}$. For all sufficiently large $k$, we have
$z^{(k)}\in z_{0}+[-1,1]^{n}$, and therefore
\[
\left|\mathbf{1}_{\{F(y)=A_{a}\}}\partial_{j}\rho(y-z^{(k)})\right|\leq\left\|\partial_{j}\rho\right\|_{\infty}\mathbf{1}_{K}(y).
\]
On the other hand, the continuity of
$\partial_{j}\rho$ implies that
\[
\partial_{j}\rho(y-z^{(k)})
\longrightarrow
\partial_{j}\rho(y-z_{0})
\]
for every $y\in\mathbb{R}^{n}$. We apply the dominated convergence theorem again to obtain
\[
\lim_{k\to\infty}
\partial_{j}p_{a}(z^{(k)})
=
\partial_{j}p_{a}(z_{0}).
\]
Thus every first partial derivative of
$p_{a}$ is continuous, proving that
$p_{a}\in C^{1}(\mathbb{R}^{n})$.

We now express these derivatives in terms
of the random variables $\xi_{j}$.
Since $\psi$ is nonnegative and continuously
differentiable, $\psi'(t)=0$ whenever
$\psi(t)=0$. Hence, by the definition of
$\xi_{j}$ in (5.3), we have
\[
-\partial_{j}\rho(w)=\xi_{j}(w)\rho(w),
\qquad w\in\mathbb{R}^{n}.
\]
Furthermore, we have $\mathbb{E}_{W}|\xi_{j}(W)|=\int_{\mathbb{R}^{n}}|\xi_{j}(w)|\rho(w)\,dw=
\int_{\mathbb{R}}|\psi'(t)|\,dt<\infty$, where the last inequality follows from
$\psi\in C_{c}^{1}(\mathbb{R})$.
Changing variables $y=z+w$ in the integral representation \eqref{integral representation 1}, we obtain
\begin{equation}
\begin{aligned}
\partial_{j}p_{a}(z)
&=
-\int_{\mathbb{R}^{n}}
\mathbf{1}_{\{F(y)=A_{a}\}}
\partial_{j}\rho(y-z)\,dy\\
&=
\int_{\mathbb{R}^{n}}
\mathbf{1}_{\{F(z+w)=A_{a}\}}
\xi_{j}(w)\rho(w)\,dw\\
&=
\mathbb{E}_{W}\left[
\xi_{j}(W)
\mathbf{1}_{\{F(z+W)=A_{a}\}}
\right].
\end{aligned}
\end{equation}

Since each $p_{a}$ belongs to $C^{1}(\mathbb{R}^{n})$, \eqref{summation representation 1} implies that $g\in C^{1}(\mathbb{R}^{n};M_{d}^{\mathrm{sa}})$.
Applying the product rule to \eqref{summation representation 1}, we obtain
\begin{align*}
\partial_{j}g(z)&=\sum_{a,b,c=1}^{q}\Big[(\partial_{j}p_{a}(z))p_{b}(z)p_{c}(z)+p_{a}(z)(\partial_{j}p_{b}(z))p_{c}(z)\\
&\qquad+p_{a}(z)p_{b}(z)(\partial_{j}p_{c}(z))
\Big]\Phi_{\mathrm{mix}}(A_{a},A_{b},A_{c}).
\end{align*}
By Lemma \ref{lem:matrix}, $\Phi_{\mathrm{mix}}$ is invariant
under every permutation of its arguments. Hence,
\begin{equation}\label{summation representation 2}
\partial_{j}g(z)=3\sum_{a,b,c=1}^{q}(\partial_{j}p_{a}(z))p_{b}(z)p_{c}(z)\Phi_{\mathrm{mix}}(A_{a},A_{b},A_{c}).
\end{equation}

Finally, by \eqref{integral representation 1} and the independence
of $W^{(1)},W^{(2)},W^{(3)}$, we have
\begin{equation}\label{representation of partial derivative}
\begin{split}
&(\partial_{j}p_{a}(z))p_{b}(z)p_{c}(z)\\
=&\mathbb{E}\Big[
\xi_{j}(W^{(1)})
\mathbf{1}_{\{F(z+W^{(1)})=A_{a}\}}\cdot\mathbf{1}_{\{F(z+W^{(2)})=A_{b}\}}\mathbf{1}_{\{F(z+W^{(3)})=A_{c}\}}\Big]
\end{split}
\end{equation}
Substituting \eqref{representation of partial derivative} into \eqref{summation representation 2} yields
\begin{equation}\label{derivative}
\partial_{j}g(z)=3\mathbb{E}\Big[\xi_{j}(W^{(1)})\Phi_{\mathrm{mix}}\big(F(z+W^{(1)}),F(z+W^{(2)}),F(z+W^{(3)})\big)\Big].
\end{equation}
For each fixed $z$, we apply \eqref{eq:approx} to the three independent
matrices $F(z+W^{(1)}),F(z+W^{(2)}),F(z+W^{(3)})$, with $A_0=F(z)$. This verifies \eqref{eq:localapprox}. Combining \eqref{eq:scorebound},  \eqref{eq:kernelcovariance} with \eqref{derivative} gives
\begin{align*}
 \left(\sum_{j=1}^n\left\lVert \partial_jg(z)\right\rVert_{1}^2\right)^{1/2}
 &\leq3\sqrt{70}\,
 \mathbb{E}\left\lVert F(z+W^{(1)})-F(z+W^{(2)})\right\rVert_{1}\\
 &\leq6\sqrt{70}\,\mathbb{E}_W\left\lVert F(z+W)-F(z)\right\rVert_{1},
\end{align*}
where the last step follows from the triangle inequality and the same distribution of $W^{(1)}$ and $W^{(2)}$. This proves \eqref{eq:localgradient}. Finally, any identity $F(z+me_j)=F(z)$ for all $z$ holds in each argument of \eqref{eq:smoothing}, and gives $g(z+me_j)=g(z)$.

{\bf Step I. Finite dimensional self-adjoint case.} Suppose that $f:\mathbb{Z}_m^n\to M_d^{\mathrm{sa}}$ and equip $\mathbb{T}_m^n=(\mathbb{R}/m\mathbb{Z})^n$ with normalized Lebesgue measure
$d\nu(z)=m^{-n}\,dz$ on $[0,m)^n$. In the sequel, integer vectors in the argument of $f$ are reduced coordinatewise modulo $m$. Define
\begin{equation}\label{eq:cells}
 F(z)=f(\lfloor z\rfloor)=f(\lfloor z_{1}\rfloor,\cdots, \lfloor z_{n}\rfloor),\qquad z\in\mathbb{R}^n,
\end{equation}
It is clear that $F$ is measurable and has finite range such that
\[
F(z+me_j)=F(z),\qquad z\in \mathbb{R}^{n}.
\]
In the sequel, we view $F$ as the function defined on $\mathbb{T}^{n}_{m}$ taking values in $M^{\mathrm{sa}}_{d}$. For any integrable function $h:\mathbb{T}^{n}_{m}\to \mathbb{R}$, by the decomposition of the cube $\mathbb{T}^{n}_{m}$, we have the following
\begin{equation}\label{eq:cellintegration}
\begin{split}
 \int_{\mathbb{T}_m^n}h(z)\,d\nu(z)&=\frac{1}{m^{n}}\int_{\sqcup_{x\in \{0,\cdots, m-1\}^{n}}[x,x+1)^{n}}h(z)~dz\\
&=\frac{1}{m^{n}}\sum_{x\in \{0,\cdots, m-1\}^{n}}\int_{[x,x+1)^{n}}h(z)~dz\\
&=\frac{1}{m^{n}}\sum_{x\in\{0,\ldots,m-1\}^n}\int_{[0,1)^n}h(x+z)\,dz.
\end{split}
\end{equation}
                                                            
   
                     
                                                           
   
                                                                       
                                                                      
                                                                             

Let $x$ be uniform on $\{0,\ldots,m-1\}^n$, and let $U$, $W$ be independent
of $x$ and of each other such that $U$ is uniform on $[0,1)^n$ and $W$ is as in
Lemma \ref{lem:kernel}. Then by the definition of $F$, it is clear that
\[
 F(x+U)=f(x),\qquad
 F(x+U+W)=f(x+\lfloor U+W\rfloor).
\]
We claim here that the variable $\lfloor U+W\rfloor$ is a random variable which is independent of $x$ such that the distribution of $\lfloor U+W\rfloor$ is uniform on $\{-1,0,1\}^n$. Indeed, for each $j\in \{1,\cdots,n\}$, we have
\begin{align*}
\mathbb{P}\{\theta: \lfloor U_{j}+W_{j}\rfloor(\theta)=1\}&=\int_{[0,1)}\int_{\mathbb{R}}\psi(t)\chi_{[1,\infty)}(\omega+t)~dt~d\omega\\
&=\int_{\mathbb{R}}\psi(t)\left(\int_{[0,1)}\chi_{[1,\infty)}(\omega+t)~d\omega\right)dt\\
&=\int_{0}^{\infty}t\psi(t)dt=\frac{1}{3}.
\end{align*}
For other cases, it is also easy to compute that
\[
\mathbb{P}\{\theta:\lfloor U_{j}+W_{j} \rfloor(\theta)=0\}=\mathbb{P}\{\theta:\lfloor U_{j}+W_{j}\rfloor(\theta)=-1\}=\frac{1}{3}.
\]
This completes the proof of our claim. Hence, \eqref{eq:cellintegration} yields that
\begin{equation}\label{eq:shortidentity}
 \int_{\mathbb{T}_m^n}\mathbb{E}_W\left\lVert F(z+W)-F(z)\right\rVert_{1}^2\,d\nu(z)=\mathbb{E}_{(x,\varepsilon)\in\mathbb{Z}_m^n\times\{-1,0,1\}^{n}}\left\lVert f(x+\varepsilon)-f(x)\right\rVert_{1}^2.
\end{equation}

We now apply the smoothing approximation argument as follows. Let $g$ be defined by \eqref{eq:smoothing}. By Proposition \ref{prop:smoothing}, the function $g$ is $C^{1}$ and periodic. For every $z$, the Cauchy--Schwarz inequality gives
\[
 \bigl(\mathbb{E}_W\left\lVert F(z+W)-F(z)\right\rVert_{1}\bigr)^2\leq\mathbb{E}_W\left(\left\lVert F(z+W)-F(z)\right\rVert_{1}^2\right).
\]
Hence, applying Proposition \ref{prop:smoothing} and \eqref{eq:shortidentity} we obtain that
\begin{equation}\label{eq:globalapprox}
\begin{split}
\left\lVert F-g\right\rVert_{L_{2}(\mathbb{T}^{n}_{m};S_{1})}&\leq 3\left(\int_{\mathbb{T}^{n}_{m}}\left(\mathbb{E}_{W}\left\|F(z+W)-F(z)\right\|_{1}\right)^{2}~d\nu(z)\right)^{1/2}\\
&\leq 3\left(\int_{T^{n}_{m}}\mathbb{E}_{W}\left\|F(z+W)-F(z)\right\|^{2}_{1}~d\nu(z)\right)^{1/2}\\
&=3\left(\mathbb{E}_{(x,\varepsilon)\in\mathbb{Z}_m^n\times\{-1,0,1\}^{n}}\left\lVert f(x+\varepsilon)-f(x)\right\rVert_{1}^2\right)^{1/2},
\end{split}
\end{equation}
and
\begin{equation}\label{eq:globalgradient}
\left(\sum_{j=1}^n\left\lVert \partial_jg\right\rVert_{L_2(\mathbb{T}^{n}_{m};S_1)}^2\right)^{1/2}\leq6\sqrt{70}\left(\mathbb{E}_{(x,\varepsilon)\in\mathbb{Z}_m^n\times\{-1,0,1\}^{n}}\left\lVert f(x+\varepsilon)-f(x)\right\rVert_{1}^2\right)^{1/2}.
\end{equation}

For an integer vector $h\in \mathbb{N}^{n}$ and the fact $F(x+u+h)=f(x+h)$ for each $u\in [0,1)^{n}$, we have
\begin{equation}\label{representation 1}
\begin{split}
\left\lVert F(\cdot+h)-F\right\rVert_{L_2(\mathbb{T}^{n}_{m};S_1)}^2&=\int_{T^{n}_{m}}\|F(z+h)-F(z)\|^{2}_{1}~d\nu(z)\\
&=\mathbb{E}_{x\in \mathbb{Z}^{n}_{m}}\int_{[0,1)^{n}}\|F(x+u+h)-F(x+u)\|^{2}_{1}~d u\\
&=\mathbb{E}_{x\in\mathbb{Z}_m^n}\left\lVert f(x+h)-f(x)\right\rVert_{1}^{2}.
\end{split}
\end{equation}
Since $m$ is even, it follows from \eqref{representation 1} that
\begin{equation}\label{eq:longidentity}
 \left(\sum_{j=1}^{n}\mathbb{E}_{x\in \mathbb{Z}^{n}_{m}}\left\lVert f(x+\tfrac m2e_j)-f(x)\right\rVert_{1}^2\right)^{1/2}=\left(\sum_{j=1}^n
   \left\lVert F(\cdot+\tfrac m2e_j)-F\right\rVert_{L_2(\mathbb{T}^{n}_{m}; S_1)}^2\right)^{1/2}.
\end{equation}
For each $j$, by $g\in C^{1}(\mathbb{R}^{n};M^{\mathrm{sa}}_{d})$, the fundamental theorem of calculus gives
\[
g(z+\tfrac m2e_j)-g(z)=\int_0^{m/2}\partial_jg(z+te_j)\,dt.
\]
Hence, the Minkowski inequality yields
\begin{equation}\label{fundamental 1}
\begin{split}
\left\lVert g(\cdot+\tfrac m2e_j)-g\right\rVert_{L_2(\mathbb{T}^{n}_{m};S_1)}\leq\int_0^{m/2}\left\lVert \partial_jg(\cdot+te_j)\right\rVert_{L_2(\mathbb{T}^{n}_{m};S_1)}\,dt=\frac m2\left\lVert \partial_jg\right\rVert_{L_2(\mathbb{T}^{n}_{m}; S_1)}.
\end{split}
\end{equation}
where we used the fact that uniform measure on $\mathbb{T}^{n}_{m}$ is translation invariant. Applying \eqref{eq:longidentity}, we obtain
\begin{equation}\label{triangle 1}
\begin{split}
&\left(\sum_{j=1}^{n}\mathbb{E}_{x\in \mathbb{Z}^{n}_{m}}\left\lVert f(x+\frac{m}{2}e_{j})-f(x)\right\rVert_{1}^{2}\right)^{1/2}\\
\leq&\left(\sum_{j=1}^{n}\left\lVert F(\cdot+\frac{m}{2}e_{j})-g(\cdot+\frac{m}{2}e_{j})\right\rVert_{L_{2}(\mathbb{T}^{n}_{m};S_{1})}^{2}\right)^{1/2}\\
&+\left(\sum_{j=1}^{n}\left\lVert g(\cdot+\frac{m}{2}e_{j})-g\right\rVert_{L_{2}(\mathbb{T}^{n}_{m};S_{1})}^{2}\right)^{1/2}+\sqrt{n}\left\lVert F-g\right\rVert_{L_{2}(\mathbb{T}^{n}_{m};S_{1})}\\
=&2\sqrt{n}\left\lVert F-g\right\rVert_{L_{2}(\mathbb{T}^{n}_{m};S_{1})}+\left(\sum_{j=1}^{n}\left\lVert g(\cdot+\frac{m}{2}e_{j})-g\right\rVert_{L_{2}(\mathbb{T}^{n}_{m};S_{1})}^{2}\right)^{1/2}.
\end{split}
\end{equation}
Due to \eqref{eq:globalapprox}, we now apply Proposition \ref{prop:smoothing} and \eqref{eq:globalgradient} to \eqref{triangle 1} and obtain that
\begin{equation}\label{self-adjoint case}
\begin{split}
&\left(\sum_{j=1}^{n}\mathbb{E}_{x\in \mathbb{Z}^{n}_{m}}\left\lVert f(x+\frac{m}{2}e_{j})-f(x)\right\rVert_{1}^{2}\right)^{1/2}\\
\leq& 2\sqrt{n}\left\lVert F-g\right\rVert_{L_{2}(\mathbb{T}^{n}_{m};S_{1})}+3\sqrt{70}m\left(\mathbb{E}_{(x,\varepsilon)\in\mathbb{Z}_m^n\times\{-1,0,1\}^{n}}\left\lVert f(x+\varepsilon)-f(x)\right\rVert_{1}^2\right)^{1/2}\\
\leq & (6\sqrt{n}+3\sqrt{70}m)\left(\mathbb{E}_{(x,\varepsilon)\in \mathbb{Z}^{n}_{m}\times \{-1,0,1\}^{n}}\left\lVert f(x+\varepsilon)-f(x)\right\rVert_{1}^{2}\right)^{1/2},
\end{split}
\end{equation}
for every $f:\mathbb{Z}^{n}_{m}\to M^{\mathrm{sa}}_{d}$. This completes the proof for the self-adjoint matrices setting.

{\bf Step II. Finite dimensional case.} For a matrix $A\in M_{d}$, we define the Hermitian dilation of $A$ by
\[
\widetilde{A}\coloneqq\frac{1}{2}
\begin{pmatrix}
0&A\\ A^*&0
\end{pmatrix}\in M_{2d}^{\mathrm{sa}}
\]
It is easy to see that
\[
\left\lVert \widetilde{A}\right\rVert_{1}=\frac12\operatorname{Tr}(AA^*)^{1/2}+\frac12\operatorname{Tr}(A^*A)^{1/2}=\left\lVert A\right\rVert_{1}.
\]
By the Hermitian dilation of matrices, we now define the Hermitian dilation of a function $f:\mathbb{Z}^{n}_{m}\to M_{d}$ as follows
\[
 \widetilde f(x)=\frac{1}{2}
\begin{pmatrix}
0&f(x)\\ f(x)^*&0
\end{pmatrix}\in M_{2d}^{\mathrm{sa}}, \qquad x\in \mathbb{Z}^{n}_{m}.
\]

Hence, for a given $f:\mathbb{Z}^{n}_{m}\to M_{d}$, we apply \eqref{self-adjoint case} to $\widetilde{f}$ to obtain
\begin{align*}
&\left(\sum_{j=1}^{n}\mathbb{E}_{x\in \mathbb{Z}^{n}_{m}}\left\lVert f(x+\frac{m}{2}e_{j})-f(x)\right\rVert_{1}^{2}\right)^{1/2}\\
=&\left(\sum_{j=1}^{n}\mathbb{E}_{x\in \mathbb{Z}^{n}_{m}}\left\lVert \widetilde{f}(x+\frac{m}{2}e_{j})-\widetilde{f}(x)\right\rVert_{1}^{2}\right)^{1/2}\\
\leq & (6\sqrt{n}+3\sqrt{70}m)\left(\mathbb{E}_{(x,\varepsilon)\in \mathbb{Z}^{n}_{m}\times \{-1,0,1\}^{n}}\left\lVert \widetilde{f}(x+\varepsilon)-\widetilde{f}(x)\right\rVert_{1}^{2}\right)^{1/2}\\
=&(6\sqrt{n}+3\sqrt{70}m)\left(\mathbb{E}_{(x,\varepsilon)\in \mathbb{Z}^{n}_{m}\times \{-1,0,1\}^{n}}\left\lVert f(x+\varepsilon)-f(x)\right\rVert_{1}^{2}\right)^{1/2}.
\end{align*}
This proves the assertion for mappings taking values in arbitrary finite dimensional matrices.

{\bf Step III. Infinite dimensional case via approximation.}
For each $x\in\mathbb{Z}_m^n$, by the singular value decomposition of $f(x)$, we have
\[
 f(x)=\sum_{k=1}^\infty s_k(x)u_k(x)\otimes v_k(x)
\]
with
\[
\|f(x)\|_{1}=\sum_{k=1}^\infty s_k(x)<\infty,
\]
and $u\otimes v$ denotes the operator $h\mapsto\langle h,v\rangle u$. For each $N\in \mathbb{N}$, let
\[
 f_N(x)=\sum_{k=1}^N s_k(x)u_k(x)\otimes v_k(x),\qquad
 \eta_N=\max_{x\in\mathbb{Z}_m^n}\sum_{k>N}s_k(x).
\]
By the definition of the Schatten trace class norm, it follows that $\left\lVert f_N(x)-f(x)\right\rVert_{1}=\sum_{k>N}s_k(x)\to 0$ as $N\to \infty$. Since $\mathbb{Z}^{n}_{m}$ is a finite set, we have
$\eta_N\to0$. By {\bf Step II}, it follows that for each $N\in \mathbb{N}$
\begin{equation}\label{finite version}
\begin{split}
&\left(\sum_{j=1}^{n}\mathbb{E}_{x\in \mathbb{Z}^{n}_{m}}\left\lVert f_{N}(x+\frac{m}{2}e_{j})-f_{N}(x)\right\rVert_{1}^{2}\right)^{1/2}\\
\leq&(6\sqrt{n}+3\sqrt{70}m)\left(\mathbb{E}_{(x,\varepsilon)\in \mathbb{Z}^{n}_{m}\times \{-1,0,1\}^{n}}\left\lVert f_{N}(x+\varepsilon)-f_{N}(x)\right\rVert_{1}^{2}\right)^{1/2}
\end{split}
\end{equation}
Taking $N\to \infty$ on the both sides of \eqref{finite version} yields the desired result.

\label{cor:scale}
If $m\geq\sqrt n$ is even and $m\geq4$, then for every $f:\mathbb{Z}_m^n\to S_1$ we have
\begin{align*}
&\left(\sum_{j=1}^{n}\mathbb{E}_{x\in \mathbb{Z}^{n}_{m}}\left\lVert f(x+\frac{m}{2}e_{j})-f(x)\right\rVert_{1}^{2}\right)^{1/2}\\
\leq&32m\left(\mathbb{E}_{(x,\varepsilon)\in \mathbb{Z}^{n}_{m}\times \{-1,0,1\}^{n}}\left\lVert f(x+\varepsilon)-f(x)\right\rVert_{1}^{2}\right)^{1/2}.
\end{align*}
Conversely, if $C>0$ and even integers $\ell_{n} \geq4$ satisfy
\begin{align*}
&\left(\sum_{j=1}^{n}\mathbb{E}_{x\in \mathbb{Z}^{n}_{\ell_{n}}}\left\lVert f(x+\frac{\ell_{n}}{2}e_{j})-f(x)\right\rVert_{1}^{2}\right)^{1/2}\\
\leq&C\ell_{n}\left(\mathbb{E}_{(x,\varepsilon)\in \mathbb{Z}^{n}_{\ell_{n}}\times \{-1,0,1\}^{n}}\left\lVert f(x+\varepsilon)-f(x)\right\rVert_{1}^{2}\right)^{1/2}
\end{align*}
for every $n\geq 1$ and every $f:\mathbb{Z}^{n}_{\ell_{n}}\to S_{1}$, then
\begin{equation}
 \ell_n\geq\frac{\sqrt n}{C\sqrt{1-3^{-n}}}\geq\frac{\sqrt n}{C}.
 \label{eq:necessity}
\end{equation}
