<!-- theorem-1: body_md -->
\label{thm:BD_fourier_decay}
Let $X = \Gamma \backslash \mathbb{H}^2$ be a non-elementary infinite-area convex cocompact hyperbolic surface, and let $\delta \in (0,1)$ be the Hausdorff dimension of its limit set $\Lambda$. Let $\mu$ be the Patterson--Sullivan probability measure based at $\mathrm{i}\in\mathbb{H}^2$. Assume that
\[
\varphi \in C^2(\mathbb{R}; \mathbb{R}), \qquad g \in C^1(\mathbb{R}; \mathbb{C})
\]
are functions satisfying the following bounds for some constant $C_{\varphi,g}$:
\[
\|\varphi\|_{C^2} + \|g\|_{C^1} \le C_{\varphi,g}, \qquad \inf_{\Lambda} |\varphi'(x)| \ge C_{\varphi,g}^{-1}.
\]
Then there exist constants $\beta > 0$ depending only on $\delta$, and $C > 0$ depending only on $X$ and $C_{\varphi,g}$, such that
\begin{equation}\label{eq:BD_decay}
\left| \int_{\Lambda} e^{i\xi \varphi(x)} g(x) \, d\mu(x) \right| \le C |\xi|^{-\beta/2}
\qquad \text{for all } |\xi| > 1.
\end{equation}

<!-- theorem-2: body_md -->
\label{thm:main}
Let $X$ be a non-elementary infinite-area convex cocompact hyperbolic surface. Then, for all $\sigma\in(\delta/2,\delta]$, we have
\begin{equation}\label{eq:main_counting_bound}
m_X(\sigma)\le \left( 2 + \frac{\beta_{\mathrm{F}}}{\delta} \right)\left(\delta-\sigma\right) - \frac{\beta_{\mathrm{F}}}{2} = \delta-  \left(2+\frac{\beta_{\mathrm{F}}}{\delta}\right) \left(\sigma - \frac{\delta}{2}\right),
\end{equation}
where $\beta_{\mathrm{F}}$ is the Fourier decay exponent defined after Theorem~\ref{thm:BD_fourier_decay}. In particular, we have
\begin{equation}\label{eq:main_spectral_gap}
g(X)\le \delta - \frac{\delta \beta_{\mathrm{F}}}{4\delta + 2\beta_{\mathrm{F}}}.
\end{equation}

<!-- prop-1: body_md -->
\label{prop:RPF}
For $\sigma \in \mathbb{R}$, the transfer operator $\mathcal{L}_\sigma$ on $C^1(\overline{I})$ is a positive operator with spectral radius $e^{P(\sigma)}$, which is a simple eigenvalue with strictly positive eigenfunction $\varphi_\sigma$. The essential spectral radius is strictly smaller than $e^{P(\sigma)}$. The spectral projector onto $\{e^{P(\sigma)}\}$ is given by
\[
\mathbb{P}_\sigma(f) = \varphi_\sigma \int_{\Lambda} f \, d\mu_\sigma,
\]
where $\mu_\sigma$ is the unique probability measure satisfying $\mathcal{L}_\sigma^* \mu_\sigma = e^{P(\sigma)} \mu_\sigma$. Moreover, the measure $\mu_\sigma$ is supported on $\Lambda$, and we normalize $\varphi_\sigma$ by $\int_\Lambda\varphi_\sigma\,d\mu_\sigma=1$.

<!-- lemma-1: body_md -->
\label{lem:fredholmRefined}
For any partition $Z \subset \mathcal{W}_{\ge 2}$, there exists a holomorphic family of trace class operators $\mathcal{T}_{Z,s}$ such that $1-\mathcal{L}_{Z,s} = (1-\mathcal{T}_{Z,s})(1-\mathcal{L}_{s}),$ where $\mathcal{L}_s$ is the standard transfer operator. More generally, for any finite collection of partitions $Z_1, \dots, Z_k \subset \mathcal{W}_{\ge 2}$, there exists a holomorphic family of trace class operators $\mathcal{T}_{s}$, depending on $Z_1, \dots, Z_k$, such that
\[
1 - \mathcal{L}_{Z_1,s} \cdots \mathcal{L}_{Z_k,s} = (1 - \mathcal{T}_{s})(1 - \mathcal{L}_s).
\] 
In particular, the Fredholm determinant of $\mathcal{L}_{Z_1,s} \cdots \mathcal{L}_{Z_k,s}$ is a holomorphic multiple of the Selberg zeta function $Z_X(s)$, i.e., there exists an entire function $H(s)$, depending on $Z_1, \dots, Z_k$, such that
\[
\det(1 - \mathcal{L}_{Z_1,s} \cdots \mathcal{L}_{Z_k,s}) = Z_X(s) H(s).
\]

<!-- lemma-2: body_md -->
\label{lem:bounded_distortion}
For all $\textbf{a} = a_1\cdots a_n \in \mathcal{W}^\circ$ and $z \in D_{b}$ with $\textbf{a} \to b$, we have
\[
\frac{\gamma''_{\textbf{a}}(z)}{\gamma'_{\textbf{a}}(z)} \asymp 1.
\]
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
We can give a direct proof by writing $\gamma_{\textbf{a}}$ as a matrix in $\mathrm{SL}_2(\mathbb{R})$:
\[
\gamma_{\textbf{a}} = \begin{pmatrix} a_{\textbf{a}}&b_{\textbf{a}}\\c_{\textbf{a}}&d_{\textbf{a}}\end{pmatrix}.
\]
We can then compute
\[
\gamma_{\textbf{a}}'(z) = \frac{1}{(c_{\textbf{a}} z + d_{\textbf{a}})^2}
= \frac{1}{c_{\textbf{a}}^2 (z - x_{\textbf{a}})^2}, \quad \text{where } x_{\textbf{a}} = -\frac{d_{\textbf{a}}}{c_{\textbf{a}}} = \gamma_{\textbf{a}}^{-1}(\infty),
\]
and
\begin{equation}\label{eq:log_der}
\frac{\gamma_{\textbf{a}}''(z)}{\gamma_{\textbf{a}}'(z)} = -\frac{2}{z - x_{\textbf{a}}}.
\end{equation}
Since $\infty$ is not contained in any of the Schottky disks $D_b$, the point $x_{\textbf{a}} = \gamma_{\textbf{a}}^{-1}(\infty)$ lies inside $D_{\overline{a}_n}$. Recall that $\textbf{a} \to b$ means that $\overline{a}_n \neq b$. Hence, since the Schottky disks have mutually disjoint closures, we have $|z - x_{\textbf{a}}| \asymp 1$ for $z \in D_b$ with $\textbf{a} \to b$, which implies the desired bounded distortion property.

<!-- lemma-3: body_md -->
\label{lem:boundsderivatives}
We have the following:
\begin{enumerate}[(i)]

\item $|\gamma_{\textbf{a}}'(z)^s| \ll e^{C(h|t| + |\sigma|)}  \Upsilon_{\textbf{a}}^\sigma.$
\item If we assume in addition that $\textbf{a} \in Y(\tau)\cup Z(\tau)$, then we have $ |\gamma_{\textbf{a}}'(z)^s| \ll  e^{C(h|t| + |\sigma|)} \tau^\sigma.$
\end{enumerate}

We will eventually choose $h$ to be roughly $|t|^{-1}$, so the exponential factor $e^{C(h|t| + |\sigma|)}$ will be bounded by a constant multiple of $e^{C|\sigma|}$, eliminating exponential growth in $t$. This is the reason for working with refined spaces.

<!-- proof-3: body_md -->
The bound in (iii) follows from (ii) and Lemma~\ref{lem:YZ}(i). To prove (ii), we write

\gamma_{\textbf{a}}'(z)^s = \gamma_{\textbf{a}}'(z)^{\sigma} \exp ( it \mathbb{L}(\gamma_{\textbf{a}}'(z)) ).

Taking absolute values and applying the bound from (i) gives

\end{align*}
This completes the proof.

<!-- lemma-9: body_md -->
\label{lem:separation}
There exist $C_{\mathrm{sep}}, h_0 > 0$ depending only on $\Gamma$ such that for $0 < h < h_0$ and $\tau \ge C_{\mathrm{sep}} h$ the following holds true: for all $\ell_1, \ell_2 \in \mathcal{A}$, $\textbf{a} \in Y_{\ell_1}(\tau)$, $\textbf{b} \in Y_{\ell_2}(\tau)$, $z_1 \in D_{\ell_1}$, and $z_2 \in D_{\ell_2}$,
\[
\gamma_{\textbf{a}}(z_1),\, \gamma_{\textbf{b}}(z_2) \text{ lie in the same connected component of } \Omega(h) \quad \Longrightarrow \quad \textbf{a} \sim_{\mathrm{pre}} \textbf{b}.
\]
<!-- end lemma-9 -->
<!-- proof-4: body_md -->
Fix words $\textbf{a} \in Y_{\ell_1}(\tau)$ and $\textbf{b} \in Y_{\ell_2}(\tau)$. Suppose for contradiction that $\textbf{a}$ is not a prefix of $\textbf{b}$ and $\textbf{b}$ is not a prefix of $\textbf{a}$, yet $\gamma_{\textbf{a}}(z_1)$ and $\gamma_{\textbf{b}}(z_2)$ lie in the same connected component of $\Omega(h)$ for some $z_1 \in D_{\ell_1}$, $z_2 \in D_{\ell_2}$. By Lemma~\ref{lem:structureLambdah}, all points in a single connected component of $\Omega(h)$ lie within distance $Ch$ of each other, so
\begin{equation}\label{eq:separation1}
|\gamma_{\textbf{a}}(z_1) - \gamma_{\textbf{b}}(z_2)| \le Ch.
\end{equation}
Let $\textbf{c}$ be the maximal common prefix of $\textbf{a}$ and $\textbf{b}$. If $\textbf{c} = \emptyset$, then $\textbf{a}$ and $\textbf{b}$ have distinct first letters, and so $\gamma_{\textbf{a}}(z_1)$ and $\gamma_{\textbf{b}}(z_2)$ lie in distinct Schottky disks. Since the Schottky disks have mutually disjoint closures, this contradicts \eqref{eq:separation1} for sufficiently small $h$. Thus, we may assume that the prefix $\textbf{c}$ is a non-empty word. Since $\textbf{a}$ and $\textbf{b}$ are not prefixes of each other, we can write $\textbf{a} = \textbf{c}\textbf{a}_2$ and $\textbf{b} = \textbf{c}\textbf{b}_2$, where $\textbf{a}_2$ and $\textbf{b}_2$ are non-empty words with different first letters. Then $\gamma_{\textbf{a}_2}(z_1)$ and $\gamma_{\textbf{b}_2}(z_2)$ lie in distinct Schottky disks, so
\[
|\gamma_{\textbf{a}_2}(z_1) - \gamma_{\textbf{b}_2}(z_2)| \gg 1.
\]
Using the identity $|\gamma(u) - \gamma(w)| = |\gamma'(u)|^{1/2} |\gamma'(w)|^{1/2} |u - w|$, valid for all M\"obius transformations $\gamma$ and points $u,w\in \mathbb{C}$, and using also Lemma~\ref{lem:boundsderivatives}(i), we get
\begin{align*}
|\gamma_{\textbf{a}}(z_1) - \gamma_{\textbf{b}}(z_2)| & = |\gamma_{\textbf{c}}(\gamma_{\textbf{a}_2}(z_1)) - \gamma_{\textbf{c}}(\gamma_{\textbf{b}_2}(z_2))| \\
& \asymp \Upsilon_\textbf{c} \cdot |\gamma_{\textbf{a}_2}(z_1) - \gamma_{\textbf{b}_2}(z_2)| \\
& \gg \Upsilon_\textbf{c}.
\end{align*}
Since $\textbf{c} \prec \textbf{a}$, Lemma~\ref{lem:boundsderivatives}(iv) and Lemma~\ref{lem:YZ}(i) give $\Upsilon_\textbf{c} \gg \Upsilon_\textbf{a} \asymp \tau$. Thus we have shown $|\gamma_{\textbf{a}}(z_1) - \gamma_{\textbf{b}}(z_2)| \gg \tau$, which contradicts \eqref{eq:separation1} if $\tau \ge C_{\mathrm{sep}} h$ for some constant $C_{\mathrm{sep}} > 0$ large enough.

<!-- prop-2: body_md -->
\label{prop:phaseDerivative}
For $b \in \mathcal{A}$, $x \in I_b$, and distinct $\textbf{a}, \textbf{b} \in \mathcal{W}^\circ$ with $\textbf{a}, \textbf{b} \to b$,
\[
|\Phi_{\textbf{a}, \textbf{b}}'(x)| \asymp \mathcal{D}_{\textbf{a}, \textbf{b}},
\]
<!-- end prop-2 -->
<!-- proof-6: body_md -->
We follow the proof of \cite[Proposition~3.2]{Soa25}. Write for all words $\textbf{a}$ (not necessarily reduced),
\[
\gamma_{\textbf{a}} = \begin{pmatrix} a_{\textbf{a}}&b_{\textbf{a}}\\c_{\textbf{a}}&d_{\textbf{a}}\end{pmatrix}\in \mathrm{SL}_2(\mathbb{R}).
\]
The key observation is that for all words $\textbf{a}$ with $\mathsf{red}(\textbf{a})\ne\emptyset$, we have
\[
|c_{\textbf{a}}| \asymp \Upsilon_{\mathsf{red}(\textbf{a})}^{-1/2}.
\]
To see this, note first that $\gamma_{\textbf{a}} = \gamma_{\mathsf{red}(\textbf{a})}$, so we may assume $\textbf{a}$ is non-empty and reduced. We have
\[
\gamma_{\textbf{a}}'(z) = \frac{1}{c_{\textbf{a}}^2 (z - x_{\textbf{a}})^2}, \quad \text{where } x_{\textbf{a}} = -\frac{d_{\textbf{a}}}{c_{\textbf{a}}} = \gamma_{\overline{\textbf{a}}}(\infty).
\]
As in the proof of Lemma~\ref{lem:bounded_distortion}, we have $|z - x_{\textbf{a}}| \asymp 1$ for all $z \in D_b$ and $\textbf{a} \to b$. This implies that $|\gamma_{\textbf{a}}'(z)| \asymp \frac{1}{c_{\textbf{a}}^2}$ for all $z \in D_b$. At the same time, Lemma~\ref{lem:boundsderivatives}(i) tells us that $|\gamma_{\textbf{a}}'(z)| \asymp \Upsilon_{\textbf{a}}$ for all $z \in D_b$, so we get $\Upsilon_{\textbf{a}} \asymp \frac{1}{c_{\textbf{a}}^2}$, which proves the claim. 

Now fix $b \in \mathcal{A}$, $x \in I_b$, and distinct $\textbf{a}, \textbf{b} \in \mathcal{W}^\circ$ with $\textbf{a}, \textbf{b} \to b$. Direct calculation yields
\begin{equation}\label{firstDer}
\Phi_{\textbf{a}, \textbf{b}}'(x) = 2\left( \frac{c_{\textbf{a}}}{c_{\textbf{a}} x + d_{\textbf{a}}} - \frac{c_{\textbf{b}}}{c_{\textbf{b}} x + d_{\textbf{b}}}  \right)
= 2 \cdot \frac{c_{\textbf{a}} d_{\textbf{b}} - c_{\textbf{b}} d_{\textbf{a}}}{(c_{\textbf{a}} x + d_{\textbf{a}})(c_{\textbf{b}} x + d_{\textbf{b}})}.
\end{equation}
Therefore,
\[
|\Phi_{\textbf{a}, \textbf{b}}'(x)| = 2 |c_{\textbf{a}} d_{\textbf{b}} - c_{\textbf{b}} d_{\textbf{a}}| \cdot |\gamma_{\textbf{a}}'(x)|^{1/2} |\gamma_{\textbf{b}}'(x)|^{1/2}.
\]
By Lemma~\ref{lem:boundsderivatives}(i),
\[
|\Phi_{\textbf{a}, \textbf{b}}'(x)| \asymp |c_{\textbf{a}} d_{\textbf{b}} - c_{\textbf{b}} d_{\textbf{a}}| \cdot \Upsilon_{\textbf{a}}^{1/2} \Upsilon_{\textbf{b}}^{1/2}.
\]
Note that
\[
\gamma_{\textbf{a} \overline{\textbf{b}}} = \gamma_{\textbf{a}} \gamma_{\textbf{b}}^{-1} =
\begin{pmatrix}
\ast & \ast \\
c_{\textbf{a}} d_{\textbf{b}} - c_{\textbf{b}} d_{\textbf{a}} & \ast
\end{pmatrix},
\]
so by the observation above,
\[
|c_{\textbf{a}} d_{\textbf{b}} - c_{\textbf{b}} d_{\textbf{a}}| \asymp \Upsilon_{\mathsf{red}(\textbf{a} \overline{\textbf{b}})}^{-1/2}.
\]
Going back to \eqref{firstDer}, we deduce that
\[
|\Phi_{\textbf{a}, \textbf{b}}'(x)| \asymp \frac{\Upsilon_{\textbf{a}}^{1/2} \Upsilon_{\textbf{b}}^{1/2}}{\Upsilon_{\mathsf{red}(\textbf{a} \overline{\textbf{b}})}^{1/2}} = \mathcal{D}_{\textbf{a}, \textbf{b}},
\]
as claimed.

<!-- lemma-11: body_md -->
\label{lem:BergmanBasicProp}
\begin{enumerate}[(i)]
\item For all $z,w\in\Omega$,
\[
|B_\Omega(z,w)|^2\le B_\Omega(z,z)B_\Omega(w,w).
\]
\item If $z \in \Omega_1 \subset \Omega_2$, then $B_{\Omega_2}(z,z) \le B_{\Omega_1}(z,z)$.
\item For the Euclidean disk $D(c,r)$ of center $c$ and radius $r$:

\]
\end{enumerate}

<!-- lemma-12: body_md -->
\label{lem:UpperBoundBergman}
For all $z, w \in \Omega$,
\begin{equation}\label{UpperBoundBergmanineq}
|B_\Omega(z,w)| \ll \frac{1}{\mathrm{dist}(z, \partial\Omega) \, \mathrm{dist}(w, \partial\Omega)},
\end{equation}
where $\mathrm{dist}(z, \partial\Omega)$ denotes the Euclidean distance from $z$ to the boundary of $\Omega$. Moreover, if $\Omega$ is a finite disjoint union of Euclidean disks, then for all $z_0, w_0 \in \Omega$,
\begin{equation}\label{DerivBergmanineq}
|\partial_z B_\Omega(z_0,w_0)|, |\partial_{\overline w} B_\Omega(z_0,w_0)| \ll \frac{1}{\mathrm{dist}(z_0, \partial\Omega)^{3/2} \mathrm{dist}(w_0, \partial\Omega)^{3/2}}.
\end{equation}
The implied constants in \eqref{UpperBoundBergmanineq} and \eqref{DerivBergmanineq} are absolute.
<!-- end lemma-12 -->
<!-- proof-7: body_md -->
Both bounds rest on the elementary inequality
\begin{equation}\label{eq:key_Bergman_ineq}
|r^2 - uv|^2 \ge r^2(r - |u|)(r - |v|), \quad |u|, |v| \le r.
\end{equation}
We start with the first bound. For $\Omega = D(c,r)$, Lemma~\ref{lem:BergmanBasicProp}(iii) gives 
\[
|B_\Omega(z,w)| \ll \frac{r^2}{|r^2-(z-c)(\overline{w}-\overline{c})|^2}.
\]
Applying \eqref{eq:key_Bergman_ineq} with $u = z-c$, $v = \overline{w}-\overline{c}$, and using $r-|u| = \mathrm{dist}(z,\partial\Omega)$, $r-|v| = \mathrm{dist}(w,\partial\Omega)$, yields \eqref{UpperBoundBergmanineq}. For general $\Omega$, note that $D(z,r_z), D(w,r_w) \subset \Omega$ with $r_z = \mathrm{dist}(z,\partial\Omega)$, $r_w = \mathrm{dist}(w,\partial\Omega)$, and so we can use Lemma~\ref{lem:BergmanBasicProp}(i)--(ii) to get
\[
|B_\Omega(z,w)|^2 \le B_\Omega(z,z) B_\Omega(w,w) \le B_{D(z, r_z)}(z,z) B_{D(w, r_w)}(w,w) \ll \frac{1}{r_z^2 r_w^2}.
\]
Now we prove \eqref{DerivBergmanineq}. The bound is local, so we may assume $\Omega = D(c,r)$. Differentiating Lemma~\ref{lem:BergmanBasicProp}(iii) with respect to $z$ gives
\[
\partial_z B_\Omega(z_0, w_0) = \frac{2r^2(\overline{w_0} - \overline{c})}{\pi(r^2 - (z_0-c)(\overline{w_0}-\overline{c}))^3}.
\]
Since $|\overline{w_0} - \overline{c}| = |w_0 - c| < r$, we have 
\[
|\partial_z B_\Omega(z_0,w_0)| \ll \frac{r^3}{|r^2-(z_0-c)(\overline{w_0}-\overline{c})|^3}.
\]
From \eqref{eq:key_Bergman_ineq} we have $|r^2-(z_0-c)(\overline{w_0}-\overline{c})|^3 \ge r^3\,\mathrm{dist}(z_0,\partial\Omega)^{3/2}\mathrm{dist}(w_0,\partial\Omega)^{3/2}$, yielding \eqref{DerivBergmanineq}. The bound on $\partial_{\overline w} B_\Omega$ follows by conjugate symmetry.

<!-- lemma-13: body_md -->
\label{lem:bergman_estimates}
For every bounded open subset $\Omega \subset \mathbb{C}$, $f \in H^2(\Omega)$, $x_0\in \Omega$, and $k\in \mathbb{N}_0$, we have
  \[
  |f^{(k)}(x_0)| \ll \frac{2^k k!}{\mathrm{dist}(x_0, \partial \Omega)^{k+1}} \|f\|_{L^2(\Omega)},

|f(x_0)| \le \left( \int_\Omega |B_\Omega(x_0,w)|^2 \operatorname{dvol}(w) \right)^{1/2} \|f\|_{L^2(\Omega)} = B_\Omega(x_0, x_0)^{1/2} \|f\|_{L^2(\Omega)}.

finishing the proof.

<!-- lemma-15: body_md -->
\label{lem:polynomial-norming}
Fix $K\ge1$, $C_0>0$, $0<\delta<1$, and an integer $N\ge1$. There exists $C_{K,C_0,\delta,N}>0$ such that, for every probability measure $\nu$ supported on $[-K,K]$ satisfying
\begin{equation}\label{eq:measure_condition}
\nu((u-r,u+r))\le C_0r^\delta,\qquad u\in\mathbb R,\quad 0<r\le1,
\end{equation}
and every complex polynomial $p$ of degree at most $N-1$, we have
\[
\sup_{\substack{w\in\mathbb C\\|w|\le K}}|p(w)|^2 \le C_{K,C_0,\delta,N}\int |p(y)|^2\,d\nu(y).
\]
<!-- end lemma-15 -->
<!-- proof-10: body_md -->
Suppose the statement is false. Then there exists a sequence of measures $\nu_j$ satisfying \eqref{eq:measure_condition} and a sequence of polynomials $p_j$ such that
\[
\sup_{|w|\le K}|p_j(w)|=1,\qquad \int|p_j|^2\,d\nu_j \to 0, \quad j\to \infty.
\]
After passing to a subsequence, we may assume that $p_j$ converges uniformly on the closed complex disk $\{|w|\le K\}$ to some polynomial $p$ with
\begin{equation}\label{eq:limit_normalization}
\sup_{|w|\le K}|p(w)|=1,
\end{equation}
and that $\nu_j$ converges weakly to some probability measure $\nu$ supported on $[-K,K]$. The first claim holds because polynomials of degree at most $N-1$ form a finite-dimensional space. The second follows from Prokhorov's theorem, since all $\nu_j$ are probability measures on the same compact interval $[-K,K]$. For every open interval $(u-r,u+r)$, the Portmanteau theorem gives
\[
\nu((u-r,u+r)) \le\liminf_{j\to\infty}\nu_j((u-r,u+r)) \le C_0r^\delta,\qquad 0<r\le1.
\]
Consequently,
\[
\nu(\{u\})\le C_0r^\delta,\qquad 0<r\le1.
\]
Letting $r\downarrow0$, using $\delta>0$, shows that $\nu$ has no atoms.

By the triangle inequality, we have
\[
\left|\int|p_j|^2\,d\nu_j-\int|p|^2\,d\nu\right| \le\sup_{y\in[-K,K]}
\bigl||p_j(y)|^2-|p(y)|^2\bigr| + \left|\int|p|^2\,d\nu_j-\int|p|^2\,d\nu\right|.
\]
The first term tends to zero by uniform convergence of $p_j$ to $p$. The second tends to zero by weak convergence, since $|p|^2$ is continuous on $[-K,K]$. Therefore, letting $j\to\infty$ in the inequality above gives
\begin{equation}
\int|p|^2\,d\nu=0.
\end{equation}
However, by \eqref{eq:limit_normalization}, $p$ is a non-zero polynomial of degree at most $N-1$, so the above equality can only be true if $\nu$ is supported on the zero set of $p$. But $\nu$ has no atoms, and a non-zero polynomial has only finitely many zeros, so this is impossible.

Assume that $h>0$ is sufficiently small so that $\Omega(h)\subset D$. Let $D_{\ell}(h)$, $\ell = 1, \dots, N(h)$, be the disks comprising $\Omega(h)$ and let $I_\ell(h) = D_\ell(h) \cap \mathbb{R}$. \begin{align*}
\sup_{z\in D_\ell(h)}|p(T_{\textbf{a},\ell}(z))|^2 &\leq \sup_{|w|\le K}|p(w)|^2\\
&\ll_N\int |p(y)|^2\,d\nu_{\textbf{a},\ell}(y)\\
&= \frac{1}{\mu_\delta(I_\ell(h))} \int_{I_\ell(h)\cap \Lambda} |p(T_{\textbf{a},\ell}(x))|^2\,d\mu_\delta(x).

\begin{equation}\label{eq:branch-polynomial-norming}

\[
|f(w)-p_{\textbf{a},\ell}(w)| \leq \sum_{k=N}^{\infty} \frac{|f^{(k)}(c_{\textbf{a},\ell})|}{k!} |w-c_{\textbf{a},\ell}|^k, \qquad w\in \mathbb{C}.

\[
|f^{(k)}(c_{\textbf{a},\ell})| \leq C \frac{2^k k!}{h^{k+1}} \|f\|_{L^2(\Omega(h))}

\begin{align*}

&\le C \|f\|_{L^2(\Omega(h))}\sum_{k=N}^\infty  \frac{2^k}{h^{k+1}}  |w-c_{\textbf{a},\ell}|^k\\
&\le C \|f\|_{L^2(\Omega(h))} \sum_{k=N}^\infty \frac{(2K\tau h)^k}{h^{k+1}}\\
&= C h^{-1} \|f\|_{L^2(\Omega(h))} \sum_{k=N}^\infty (2K\tau)^k\\
&= C h^{-1} \|f\|_{L^2(\Omega(h))} \frac{(2K\tau)^N}{1-2K\tau}\\
&\le 2 C h^{-1} \|f\|_{L^2(\Omega(h))} (2K\tau)^N.
\end{align*}

\begin{equation}\label{eq:branch-taylor-norming}

Note that the factor $(2K)^{2N}$ is absorbed into the implicit constant $\ll_N$.

Fix an orthonormal basis $\{\psi_m\}_{m\in \mathbb{N}}$ of $H^2(\Omega(h))$, and recall that the Bergman kernel of $H^2(\Omega(h))$ can be expressed for all $z, w \in \Omega(h)$ as the absolutely convergent series
\[
B_{\Omega(h)}(z,w) = \sum_{m\in \mathbb{N}} \psi_m(z) \overline{\psi_m(w)}.
\]
By the definition of the refined operators, we have for $z\in\Omega_b(h)$, $b\in\mathcal{A}$,
\[
\mathcal{L}_{\tau', \tau''; s}f(z) = \sum_{\textbf{a}_2\in Y_b(\tau')} \sum_{\textbf{a}_1\in Y_{\mathrm{first}(\textbf{a}_2)}(\tau'')} \gamma_{\textbf{a}_1\textbf{a}_2}'(z)^s f(\gamma_{\textbf{a}_1\textbf{a}_2}(z)),
\]
where
\[
S_b:=\{(\textbf{a}_1,\textbf{a}_2):\textbf{a}_2\in Y_b(\tau'),\ \textbf{a}_1\in Y_{\mathrm{first}(\textbf{a}_2)}(\tau'')\}.
\]
Here, $\mathrm{first}(\textbf{a}_2)$ denotes the first letter of the word $\textbf{a}_2$. Therefore, we have the following expansion for every $x \in \Lambda\cap I_b$:
\begin{equation}
\begin{aligned}
\sum_m|\mathcal L_{\tau',\tau'';s}\psi_m(x)|^2
&=\sum_{(\textbf{a}_1,\textbf{a}_2),(\textbf{b}_1,\textbf{b}_2)\in S_b}
\gamma_{\textbf{a}_1\textbf{a}_2}'(x)^s\overline{\gamma_{\textbf{b}_1\textbf{b}_2}'(x)^s}\\
&\quad\times B_{\Omega(h)}(\gamma_{\textbf{a}_1\textbf{a}_2}(x),\gamma_{\textbf{b}_1\textbf{b}_2}(x)).
\end{aligned}
\end{equation}
Define the function
\[
g_{\textbf{a},\textbf{b}}(z;s) := \gamma'_\textbf{a}(z)^s \overline{\gamma'_\textbf{b}(z)^s} B_{\Omega(h)}(\gamma_\textbf{a}(z), \gamma_\textbf{b}(z)).

\begin{equation}\label{eq:explicit_formula}
\int_{\Lambda}\sum_{m\in \mathbb{N}}|\mathcal{L}_{\tau', \tau''; s} \psi_m(x)|^2 d\mu_\delta = \sum_{b\in\mathcal A} \sum_{(\textbf{a}_1,\textbf{a}_2),(\textbf{b}_1,\textbf{b}_2)\in S_b} \int_{\Lambda\cap I_b} g_{\textbf{a}_1\textbf{a}_2,\textbf{b}_1\textbf{b}_2}(x;s) \, d\mu_\delta(x).

For sufficiently small scales, all composite branches $\textbf{a}=\textbf{a}_1\textbf{a}_2$ with $(\textbf{a}_1,\textbf{a}_2)\in S_b$ have length at least $N_0$. Hence Lemma~\ref{lem:contractionInH} gives, for $z\in\Omega_b(h)$,
\[

and similarly for $\gamma_\textbf{b}(z)$. Thus Lemma~\ref{lem:UpperBoundBergman} gives $|B_{\Omega(h)}(\gamma_\textbf{a}(z),\gamma_\textbf{b}(z))|\ll h^{-2}$. Combining this with Lemma~\ref{lem:derivative_bounds_omega} gives
\begin{equation}\label{eq:pointwise_g_bound}

When $x$ is real, we have $|g_{\textbf{a},\textbf{b}}(x;s)|=|g_{\textbf{a},\textbf{b}}(x;\sigma)|,$ in which case we can drop the exponential factor $e^{Ch|t|}.$

Recall that $\tau'' = C_{\mathrm{sep}} h $, where $C_{\mathrm{sep}}$ is the constant from Lemma~\ref{lem:separation}. This lemma, applied with $z_1=\gamma_{\textbf{a}_2}(z)$ and $z_2=\gamma_{\textbf{b}_2}(z)$, gives $\textbf{a}_1\sim_{\mathrm{pre}}\textbf{b}_1$ or otherwise $\gamma_{\textbf{a}_1\textbf{a}_2}(z)$ and $\gamma_{\textbf{b}_1\textbf{b}_2}(z)$ belong to different connected components of $\Omega(h)$. We conclude the sum on the right of \eqref{eq:explicit_formula} reduces to terms where $\textbf{a}_1$ and $\textbf{b}_1$ are prefix-comparable.

Before estimating this norm, fix $x_b\in\Lambda\cap I_b$. For every composite branch $\textbf{a}$ occurring here, $\gamma_\textbf{a}(x_b)\in\Lambda$ and the derivative bounds give
\[
|\gamma_\textbf{a}(x)-\gamma_\textbf{a}(x_b)|\ll\tau'\tau''\asymp h\tau',\qquad x\in I_b.
\]
Since $\tau'\to0$, the entire interval $\gamma_\textbf{a}(I_b)$ lies in $\Omega(h/2)$ for sufficiently large $T$ and has distance $\gg h$ from $\partial\Omega(h)$; the same holds for $\textbf{b}$. Thus $g_{\textbf{a},\textbf{b}}(\cdot;\sigma)$ is well-defined and $C^1$ on $I_b$, and the pointwise bound in \eqref{eq:pointwise_g_bound} holds there. It gives
\[
\Vert g_{\textbf{a},\textbf{b}}(\cdot;\sigma)\Vert_{C^0(I_b)} \ll e^{C|\sigma|} h^{-2}(\tau'\tau'')^{2\sigma}.
\]
To bound the maximum of the derivative, write
\begin{align*}
\frac{d}{dx}g_{\textbf{a},\textbf{b}}(x;\sigma)
&=\frac{d}{dx}\left(\gamma'_\textbf{a}(x)^\sigma\gamma'_\textbf{b}(x)^\sigma\right)
B_{\Omega(h)}(\gamma_\textbf{a}(x),\gamma_\textbf{b}(x))\\
&\quad+\gamma'_\textbf{a}(x)^\sigma\gamma'_\textbf{b}(x)^\sigma
\frac{d}{dx}B_{\Omega(h)}(\gamma_\textbf{a}(x),\gamma_\textbf{b}(x)).
\end{align*}
Since $|\gamma_\textbf{a}''(x)/\gamma_\textbf{a}'(x)| \ll 1$ for $x\in I_b$, we get $ \frac{d}{dx} \left( \gamma'_\textbf{a}(x)^\sigma \gamma'_\textbf{b}(x)^\sigma \right) \ll e^{C|\sigma|} (\tau'\tau'')^{2\sigma}$. Moreover, writing
$$
\begin{aligned}
\frac{d}{dx} B_{\Omega(h)}(\gamma_\textbf{a}(x), \gamma_\textbf{b}(x))
&= \partial_z B_{\Omega(h)}(\gamma_\textbf{a}(x), \gamma_\textbf{b}(x)) \cdot \gamma_\textbf{a}'(x)\\
&\quad+\partial_{\overline w} B_{\Omega(h)}(\gamma_\textbf{a}(x), \gamma_\textbf{b}(x)) \cdot \gamma_\textbf{b}'(x),
\end{aligned}
$$
we obtain from Lemma~\ref{lem:UpperBoundBergman},
$$
\left|\frac{d}{dx}B_{\Omega(h)}(\gamma_\textbf{a}(x),\gamma_\textbf{b}(x))\right| \ll h^{-3}\tau'\tau''.
$$
Combining the above estimates, we get $ \sup_{x\in I_b} \left| \frac{d}{dx} g_{\textbf{a},\textbf{b}}(x;\sigma) \right| \ll e^{C|\sigma|} h^{-2}(\tau'\tau'')^{2\sigma}(1+h^{-1}\tau'\tau'')$. Consequently,
\begin{equation*}
\|g_{\textbf{a},\textbf{b}}(\cdot;\sigma)\|_{C^1(I_b)} \ll e^{C|\sigma|} h^{-2}(\tau'\tau'')^{2\sigma}\bigl(1+h^{-1}\tau'\tau''\bigr).
\end{equation*}
Since $\tau''\asymp h$ and $\tau'\ll 1$, this simplifies to
\begin{equation*}
\|g_{\textbf{a},\textbf{b}}(\cdot;\sigma)\|_{C^1(I_b)} \ll e^{C|\sigma|} h^{-2}(\tau'\tau'')^{2\sigma}.
\end{equation*}

Now we estimate the quantity $\mathcal{D}_{\textbf{a}, \textbf{b}} $. Suppose first that $(\textbf{a},\textbf{b})=(\textbf{c}\textbf{a}_2,\textbf{c}\textbf{b}_2)\in Q_b^{(2)}$. Recall that
\begin{itemize}
\item $\textbf{c}\in Y(\tau'')$ and $\textbf{a}_2,\textbf{b}_2\in Y(\tau')$,
\item $\textbf{c}\to \textbf{a}_2,\textbf{b}_2$, and
\item $\textbf{a}_2\nsim_{\mathrm{suf}}\textbf{b}_2$.
\end{itemize}
Let $\textbf{d}$ be the largest common suffix of $\textbf{a}_2$ and $\textbf{b}_2$ and write $\textbf{a}_2 = \textbf{e} \textbf{d}$ and $\textbf{b}_2 = \textbf{f} \textbf{d}$. Since neither $\textbf{a}_2$ nor $\textbf{b}_2$ is a suffix of the other, both $\textbf{e}$ and $\textbf{f}$ are non-empty words with distinct ending letters. This gives
\[
\mathsf{red}(\textbf{a}\overline{\textbf{b}})
=\textbf{c}\textbf{e}\overline{\textbf{f}}\overline{\textbf{c}}
=\textbf{c}\,\mathsf{red}(\textbf{a}_2\overline{\textbf{b}_2})\,\overline{\textbf{c}},
\]
and $\textbf{c}\to \mathsf{red}(\textbf{a}_2\overline{\textbf{b}_2})\to\overline{\textbf{c}}.$ Repeated applications of Lemma~\ref{lem:boundsderivatives} now give
\[
\mathcal{D}_{\textbf{a}, \textbf{b}}
= \left( \frac{\Upsilon_{\textbf{a}} \Upsilon_{\textbf{b}}}{\Upsilon_{\mathsf{red}(\textbf{a} \overline{\textbf{b}})}} \right)^{1/2}
\asymp \left( \frac{\Upsilon_{\textbf{a}_2} \Upsilon_{\textbf{b}_2}}{\Upsilon_{\mathsf{red}(\textbf{a}_2 \overline{\textbf{b}_2})}} \right)^{1/2}
\asymp \frac{\tau'}{\Upsilon_{\mathsf{red}(\textbf{a}_2 \overline{\textbf{b}_2})}^{1/2}}.
\]
Thus, \eqref{eq:BD_bound_applied} and the preceding $C^1$ bound imply
\begin{align*}
\left|\int_{\Lambda\cap I_b}g_{\textbf{a},\textbf{b}}(x;s)\,d\mu_\delta(x)\right|
&\ll e^{C|\sigma|}h^{-2}(\tau'\tau'')^{2\sigma}
\left(\frac{\tau'|t|}{\Upsilon_{\mathsf{red}(\textbf{a}_2\overline{\textbf{b}_2})}^{1/2}}\right)^{-\beta_{\mathrm{F}}/2}\\
&=e^{C|\sigma|}h^{-2}(\tau'\tau'')^{2\sigma}
(\tau')^{-\beta_{\mathrm{F}}/2}|t|^{-\beta_{\mathrm{F}}/2}
\Upsilon_{\mathsf{red}(\textbf{a}_2\overline{\textbf{b}_2})}^{\beta_{\mathrm{F}}/4}.
\end{align*}
For fixed $\textbf{a}_2,\textbf{b}_2$, the common prefix $\textbf{c}$ ranges over a subset of $Y(\tau'')$. Hence, using $|Y(\tau'')|\ll(\tau'')^{-\delta}$, we obtain
\[
\Sigma_2 \ll e^{C |\sigma|} h^{-2} (\tau'\tau'')^{2\sigma} (\tau'')^{-\delta} (\tau')^{-\beta_{\mathrm{F}}/2} |t|^{-\beta_{\mathrm{F}}/2}\sum_{\textbf{a}_2,\textbf{b}_2\in Y(\tau')} \Upsilon_{\mathsf{red}(\textbf{a}_2 \overline{\textbf{b}_2})}^{\beta_{\mathrm{F}}/4}.
\]
To estimate the sum on the right, we apply Lemma~\ref{lem:sum_of_Upsilons} below with $\alpha=\beta_{\mathrm{F}}/4<\delta/2$. We obtain
\begin{equation}\label{eq:sigma_2}
\Sigma_2 \ll e^{C |\sigma|} h^{-2} (\tau'\tau'')^{2\sigma} (\tau'')^{-\delta} (\tau')^{-2\delta} |t|^{-\beta_{\mathrm{F}}/2}.
\end{equation}

Finally, we consider the set $Q_b^{(3)}$. Suppose without loss of generality that $(\textbf{a},\textbf{b})$ belongs to the first part of $Q_b^{(3)}$. The second part of $Q_b^{(3)}$ is handled similarly. Then there exists a non-empty word $\mathbf{v}\in F$ such that $(\textbf{a},\textbf{b})  = (\textbf{c}\textbf{a}_2,\textbf{c}\mathbf{v}\textbf{b}_2)$. Put $\widetilde{\textbf{b}}_2:=\mathbf{v}\textbf{b}_2$. We claim that $\textbf{a}_2\nsim_{\mathrm{suf}}\widetilde{\textbf{b}}_2$. Suppose not. If $\widetilde{\textbf{b}}_2$ is a suffix of $\textbf{a}_2$, then $\textbf{b}_2$ is a suffix of $\textbf{a}_2$, contradicting $\textbf{a}_2\nsim_{\mathrm{suf}}\textbf{b}_2$. If, on the other hand, $\textbf{a}_2$ is a suffix of $\widetilde{\textbf{b}}_2$, then $\textbf{a}_2\sim_{\mathrm{suf}}\textbf{b}_2$, again a contradiction.

Hence, we can apply the same argument as before to $\textbf{a}_2$ and $\widetilde{\textbf{b}}_2$ instead of $\textbf{b}_2$ to deduce that
\[
\mathcal{D}_{\textbf{c}\textbf{a}_2,\textbf{c}\mathbf{v}\textbf{b}_2}
\asymp_F \frac{\tau'}{\Upsilon_{\mathsf{red}(\textbf{a}_2\overline{\textbf{b}_2}\overline{\mathbf{v}})}^{1/2}}.
\]
For $\mathbf{v}\in F$, let
\[
\mathcal{V}_{\mathbf{v}}:=\{\mathbf{v}\textbf{b}_2:\textbf{b}_2\in Y(\tau'),\ \mathbf{v}\to\textbf{b}_2\}.
\]
The same chain of inequalities as before shows that
\[
\Sigma_3 \ll e^{C |\sigma|} h^{-2} (\tau'\tau'')^{2\sigma} (\tau'')^{-\delta} (\tau')^{-\beta_{\mathrm{F}}/2} |t|^{-\beta_{\mathrm{F}}/2}
\sum_{\mathbf{v}\in F}\sum_{\textbf{a}_2\in Y(\tau'),\,\textbf{b}_2\in\mathcal{V}_{\mathbf{v}}}
\Upsilon_{\mathsf{red}(\textbf{a}_2\overline{\textbf{b}_2})}^{\beta_{\mathrm{F}}/4}.
\]
Since $F$ is finite, for each fixed $\mathbf{v}\in F$, the set $\mathcal{V}_{\mathbf{v}}$ consists of words whose $\Upsilon$-size is comparable to $\tau'$. We can apply Lemma~\ref{lem:sum_of_Upsilons} below to $\mathcal{U} = Y(\tau')$ and $\mathcal{V} = \mathcal{V}_{\mathbf{v}}$ and then sum over $F$ to get
\begin{equation}\label{eq:sigma_3}
\Sigma_3 \ll e^{C |\sigma|} h^{-2} (\tau'\tau'')^{2\sigma} (\tau'')^{-\delta} (\tau')^{-2\delta} |t|^{-\beta_{\mathrm{F}}/2}.
\end{equation}

Combining the estimates for $\Sigma_1$, $\Sigma_2$, and $\Sigma_3$ yields
\begin{equation*}
\sum_{m\in \mathbb{N}}\int_{\Lambda}|\mathcal{L}_{\tau', \tau''; s} \psi_m|^2 d\mu_\delta
\ll e^{C |\sigma|} h^{-2} (\tau'\tau'')^{-\delta + 2\sigma} \left(1 + (\tau')^{-\delta} |t|^{-\beta_{\mathrm{F}}/2} \right).
\end{equation*}
Finally, choosing $\tau' = T^{-\beta_{\mathrm{F}}/(2\delta)}$ and recalling that $T/2 \le |t| \le 2T$, we see that the term in parentheses is $O(1)$, and hence we get the desired bound.

\label{lem:sum_of_Upsilons}
Let $C_0\ge1$, and let $\mathcal{U},\mathcal{V}\subset\mathcal{W}$ be finite sets such that for all $\textbf{a}\in\mathcal{U}$ and $\textbf{b}\in\mathcal{V}$,
\[
C_0^{-1}\tau\le\Upsilon_{\textbf{a}}\le C_0\tau,
\qquad
C_0^{-1}\tau\le\Upsilon_{\textbf{b}}\le C_0\tau.
\]
Then, for all $0<\tau\le1$, $\alpha\in\mathbb{R}$, and $\epsilon>0$,
\[
\sum_{\textbf{a}\in\mathcal{U},\,\textbf{b}\in\mathcal{V}}
\Upsilon_{\mathsf{red}(\textbf{a}\overline{\textbf{b}})}^{\alpha}
\ll_{\alpha,\epsilon,C_0} \tau^{-\delta}+\tau^{-2\delta+2\alpha-\epsilon}.
\]
If $\alpha<\delta/2$, we can remove the $\epsilon$-loss:
\[
\sum_{\textbf{a}\in\mathcal{U},\,\textbf{b}\in\mathcal{V}}
\Upsilon_{\mathsf{red}(\textbf{a}\overline{\textbf{b}})}^{\alpha}
\ll_{\alpha,C_0}\tau^{-2\delta+2\alpha}.
\]

By \cite[Lemma~10]{CalMag25}, the Frobenius norm satisfies $\|\gamma_\textbf{a}\| \asymp \Upsilon_\textbf{a}^{-1/2}$. Using the identity $\|\gamma\|^2 = 2\cosh(d(\mathrm{i},\gamma(\mathrm{i})))$ and the lattice point estimate $\#\{\gamma\in\Gamma: d(\mathrm{i},\gamma(\mathrm{i}))<R\}\ll e^{\delta R}$ from \cite{Pa88} (see also \cite[Theorem~14.22]{Bor16}), it follows that
\begin{equation}\label{eq:cardsets}
\#\left\{ \textbf{a} \in \mathcal{W} : \Upsilon_{\textbf{a}} > \tau \right\} \ll \tau^{-\delta}.
\end{equation}
Thus, using a dyadic decomposition of the set $\{\textbf{a} \in \mathcal{W} : \Upsilon_{\textbf{a}} > \tau\}$, we get for every $\kappa\in\mathbb{R}$ and $\epsilon>0$,
\begin{align*}
\sum_{\substack{\textbf{a} \in \mathcal{W}\\ \Upsilon_{\textbf{a}}>\tau}} \Upsilon_{\textbf{a}}^{\kappa} 
&\ll O(1) + \sum_{k\ge 0: 2^{-k}>\tau} 2^{-\kappa k} \#\left\{ \textbf{a} \in \mathcal{W} : \Upsilon_{\textbf{a}} > 2^{-k-1} \right\}\\
&\ll O(1) + \sum_{k\ge 0: 2^{-k}>\tau} 2^{(\delta-\kappa) k},
\end{align*}
where $O(1)$ accounts for the finitely many terms with $\Upsilon_{\textbf{a}}>1$. A geometric series summation gives
\begin{equation}\label{eq:weighted_Upsilon_sum}
\sum_{\substack{\textbf{a} \in \mathcal{W}\\ \Upsilon_{\textbf{a}}>\tau}} \Upsilon_{\textbf{a}}^{\kappa} \ll_{\kappa,\epsilon} \tau^{-\epsilon}+\tau^{-\delta+\kappa}.
\end{equation}
If $\kappa>\delta$, the same geometric series converges uniformly in $\tau$, so the left-hand side is $O_\kappa(1)$.

Now we can prove the statement of Lemma~\ref{lem:sum_of_Upsilons}. For $\textbf{a}\in\mathcal{U}$ and $\textbf{b}\in\mathcal{V}$, let $\textbf{c}$ be their maximal common suffix and write $\textbf{a}=\textbf{a}_0\textbf{c}$ and $\textbf{b}=\textbf{b}_0\textbf{c}$. If both words $\textbf{a}_0$ and $\textbf{b}_0$ are non-empty, then by maximality of $\textbf{c}$ they end with different letters. In this case $\textbf{a}_0 \overline{\textbf{b}_0}$ is reduced. The same is trivially true when at least one of $\textbf{a}_0$ and $\textbf{b}_0$ is empty. In any case, $\mathsf{red}(\textbf{a}\overline{\textbf{b}}) = \textbf{a}_0 \overline{\textbf{b}_0}$, and using the convention $\Upsilon_\emptyset=1$, Lemma~\ref{lem:boundsderivatives} gives
\[
\Upsilon_{\mathsf{red}(\textbf{a}\overline{\textbf{b}})} \asymp \Upsilon_{\textbf{a}_0}\Upsilon_{\textbf{b}_0},
\]
and
\[
 C^{-1} \frac{\tau}{\Upsilon_\textbf{c}} \le \Upsilon_{\textbf{a}_0}, \Upsilon_{\textbf{b}_0} \le C \frac{\tau}{\Upsilon_\textbf{c}},
\qquad
\Upsilon_\textbf{c}\ge C^{-1}\tau,
\]
for some constant $C=C(C_0,\Gamma)>0$. Therefore, since the map $(\textbf{a},\textbf{b})\mapsto(\textbf{a}_0,\textbf{b}_0,\textbf{c})$ is injective, we can estimate
\begin{equation}
\sum_{\textbf{a}\in\mathcal{U},\,\textbf{b}\in\mathcal{V}} \Upsilon_{\mathsf{red}(\textbf{a}\overline{\textbf{b}})}^{\alpha} \ll_{\alpha,C_0} \tau^{2\alpha}
\sum_{\substack{\textbf{c}\in\mathcal{W}\\\Upsilon_\textbf{c}\ge C^{-1}\tau}}
\#\left\{(\textbf{a}_0,\textbf{b}_0)\in\mathcal{W}^2:
\Upsilon_{\textbf{a}_0},\Upsilon_{\textbf{b}_0}\ge C^{-1} \frac{\tau}{\Upsilon_\textbf{c}}\right\}
\Upsilon_\textbf{c}^{-2\alpha}.
\end{equation}
Using \eqref{eq:cardsets} and \eqref{eq:weighted_Upsilon_sum}, we obtain
\begin{align*}
\sum_{\textbf{a}\in\mathcal{U},\,\textbf{b}\in\mathcal{V}} \Upsilon_{\mathsf{red}(\textbf{a}\overline{\textbf{b}})}^{\alpha} &\ll_{\alpha,C_0} \tau^{-2\delta+2\alpha}
\sum_{\substack{\textbf{c}\in\mathcal{W}\\\Upsilon_\textbf{c}\ge C^{-1}\tau}}
\Upsilon_\textbf{c}^{2\delta-2\alpha}\\
&\ll_{\alpha,\epsilon,C_0}
\tau^{-2\delta+2\alpha}\left(\tau^{-\epsilon}+\tau^{-\delta+(2\delta-2\alpha)}\right)\\
&=\tau^{-\delta}+\tau^{-2\delta+2\alpha-\epsilon},
\end{align*}
which is the sought-after estimate.
If $\alpha<\delta/2$, then $2\delta-2\alpha>\delta$, and the sum over $\textbf{c}$ in the first line is uniformly bounded by the loss-free weighted estimate above (with a fixed adjustment of the cutoff to include equality). This gives $\ll_{\alpha,C_0}\tau^{-2\delta+2\alpha}$ and proves the loss-free case.
