<!-- lemma-1: body_md -->
\label[Lemma]{lem:Lindbladian_properties}
The Lindbladian $\mathcal{L}$ defined in
\eqref{eq:Lindbladian_unparametric} is a linear operator
$\mathcal{L}:\mathcal H\to A_0$ and satisfies
$(\ker\mathcal{L})\cap S\neq\emptyset$
and 
$\ker\mathcal{L}
=
\operatorname{span}_{\ensuremath\mathbb{R}}\bigl((\ker\mathcal{L})\cap S\bigr).$

In particular, the steady-state problem, 
i.e., finding $\rho_{\mathrm{ss}} \in S$ such that $\mathcal{L}\rho_{\mathrm{ss}} = 0$,
has a solution in~$S$,
and a solution is unique if and only if $\dim\ker\mathcal{L}=1$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Linearity, Hermiticity preservation and
$\operatorname{tr}(\mathcal{L} X)=0$ follow directly from
\eqref{eq:Lindbladian_unparametric}. 
The existence of stationary states 
and the relation $\ker\mathcal{L}
=
\operatorname{span}_{\ensuremath\mathbb{R}}\bigl((\ker\mathcal{L})\cap S\bigr)$
for finite-dimensional GKSL generators
are established in~\cite[Sec.~3.6 and~4.1]{BaumgartnerNarnhofer2008}. 
The final assertion follows from $\ker\mathcal{L}
=
\operatorname{span}_{\ensuremath\mathbb{R}}\bigl((\ker\mathcal{L})\cap S\bigr)$ and the trace-one
normalization; see also \cite[Sec.~3.6 and~4.1]{BaumgartnerNarnhofer2008}.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label[lemma]{lem:geometric_properties}
It holds that
\[
\mathcal H=A_0\oplus\operatorname{span}_{\ensuremath\mathbb{R}}\{I\},
\qquad
A_1
=
{I}/{N}
+A_0,
\qquad
S=A_1\cap\mathcal H_+.
\]
In particular, $A_1$ is an affine convex subset of $\mathcal H$,
$S$ is a compact convex subset of $\mathcal H$, and
\[
{1}/{\sqrt N} \leq \|X\|  
\quad\text{for }X\in A_1,
\qquad
{1}/{\sqrt N}\leq\|\rho\|\leq1
\quad\text{for }\rho\in S.
\]
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Since $\langle X,I\rangle=\operatorname{tr} X$, we have
$A_0=\operatorname{span}_{\ensuremath\mathbb{R}}\{I\}^{\perp}$, and every $X\in A_1$ has the unique
representation $X=I/N+X_0$ with $X_0\in A_0$. Moreover,
$1=\langle X,I\rangle\leq\sqrt N\|X\|$ for $X\in A_1$. If
$\lambda_1,\ldots,\lambda_N\geq0$ are the eigenvalues of $\rho\in S$, then
$\sum_i\lambda_i=1$ and
$1/N\leq\|\rho\|^2=\sum_i\lambda_i^2\leq1$. Finally,
$S=A_1\cap\mathcal H_+$ is closed and convex, and its boundedness together
with finite dimensionality gives compactness.
<!-- end proof-2 -->
<!-- remark-1: body_md -->
\label[Remark]{rem:steady_state_eqn_as_LES}
By \Cref{lem:geometric_properties}, every
$\rho_{\mathrm{ss}}\in S$ can be written uniquely 
as $\rho_{\mathrm{ss}}=I/N+X$ with $X\in A_0$. 
Consequently,
\begin{equation}
\mathcal{L}\rho_{\mathrm{ss}}=0
\qquad\Longleftrightarrow\qquad
\mathcal{L} X
=
- 
N^{-1}
\mathcal{L} I.
\label{eq:steady_state_eqn_as_LES}
\end{equation}
Therefore, the steady-state problem on $A_1$ is
equivalently an inhomogeneous linear equation on~$A_0$.
<!-- end remark-1 -->
<!-- assumption-1: body_md -->
\label[assumption]{ass:affine_decomp_param_Lindbladian}
For a compact domain $\mathcal{P}\subseteq\ensuremath\mathbb{R}^d$, the parametric
Lindbladian $\mathcal{L}:\mathcal{P}\to\mathcal B(\mathcal H)$ has the form
\begin{equation}
\mathcal{L}(\mu)X
=
\sum_{p=1}^{P}
\theta_p(\mu) \mathcal{K}_p(X)
+
\sum_{q=1}^{Q}
\phi_q(\mu)\mathcal D_q(X),
\label{eq:Lindbladian_affine_decomposition}
\end{equation}
where 
\begin{align*}
\mathcal{K}_p(X)
\coloneqq \, 
-\mathrm{i} \big[ H_p, X \big]
\qquad \text{and} \qquad
\mathcal D_q(X)
& \coloneqq \, 
\sum_{\ell=1}^{m_q}
\bigl(
L_{q,\ell}XL_{q,\ell}^\dagger
-
\frac12
\{L_{q,\ell}^\dagger L_{q,\ell},X\}
\bigr)
\end{align*}
for $p=1,\ldots,P$ and $q=1,\ldots,Q$,
with each $H_p\in\mathcal H$ 
and $L_{q,\ell}\in\ensuremath\mathbb{C}^{N\timesN}$ fixed.
The coefficient functions
$\theta_p:\mathcal{P}\to\ensuremath\mathbb{R}$ and
$\phi_q:\mathcal{P}\to[0,\infty)$ are also fixed scalar functions.
<!-- end assumption-1 -->
<!-- definition-1: body_md -->
\label[definition]{def:sol_set}
For $\mathcal{L}: \mathcal{P} \to \mathcal{B}(\mathcal{H})$ as in \Cref{ass:affine_decomp_param_Lindbladian},
its {\emph{steady state solution set}} is defined as 
\begin{align*} 
\mathcal{M} \; \coloneqq \; \{ \rho \in S \, | \, \exists \mu \in \mathcal{P} : \mathcal{L}(\mu) \rho = 0 \}. 
\end{align*}
The \emph{Kolmogorov-$n$-width} of $\mathcal{M}$ is defined as 
\begin{align*}
d_n(\mathcal{M})
\;\coloneqq
\inf_{\substack{V\subseteq\mathcal H\\ \dim V\leq n}}
\sup_{\rho \in \mathcal{M}}
\mathrm{dist}(\rho,V)
\qquad \text{for } n\in \ensuremath\mathbb{N}.
\end{align*}
<!-- end definition-1 -->
<!-- lemma-3: body_md -->
\label[lemma]{lem:linear_affine_widths_equivalent}
For $n\geq0$, define the affine Kolmogorov-$n$-width by 
\[
d_n^{\mathrm{aff}}(\mathcal M)
\coloneqq
\inf_{\substack{V\subseteq\mathcal H,\,\dim V\leq n\\X_0\in\mathcal H}}
\sup_{\rho\in\mathcal M}\mathrm{dist}(\rho,X_0+V).
\]
Then it holds that 
\(
d_{n+1}(\mathcal M)
\leq d_n^{\mathrm{aff}}(\mathcal M)
\leq d_n(\mathcal M).
\)
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
The upper bound follows by choosing $X_0=0$. 
For the lower bound, 
$X_0+V$ is contained in $\operatorname{span}_{\ensuremath\mathbb{R}}(\{X_0\}\cup V)$, 
whose dimension is at most $n+1$.
<!-- end proof-3 -->
<!-- assumption-2: body_md -->
We assume access to a solver which, 
for every selected $\mu\in\mathcal{P}$, 
yields a $\rho_\mu$, 
such that $\mathrm{dist} \big(\rho_\mu, S \cap\ker \mathcal{L}(\mu) \big)$ is below any prescribed numerical accuracy.
We denote $\rho_{\mathrm{ss}}(\mu) \coloneqq \rho_\mu$, 
and regard it as an \emph{exact steady state}.
<!-- end assumption-2 -->
<!-- remark-2: body_md -->
\label[remark]{rem:RBM_pLindblad_outline} 
Given a parametric Lindbladian $\mathcal{L}(\mu)$ as in \Cref{ass:affine_decomp_param_Lindbladian},
the RB procedure consists of the following two phases: 
\begin{itemize}
\item 
The \emph{offline phase} assembles a reduced space $V$ based on snapshot solutions $\rho_{\mathrm{ss}}(\mu_1),\dots, \rho_{\mathrm{ss}}(\mu_J)$.
Common assembly strategies include \emph{proper orthogonal decomposition (POD)},
or \emph{greedy methods}, see \Cref{subsubsec:POD_vs_greedy}.
Once $V$ is constructed, 
with an orthonormal basis (ONB) $B_1, \dots,  B_r$ available,
we compute and store parameter-independent quantities \eqref{eq:offline_HH_matrix}--\eqref{eq:offline_DD_matrix},
which will be used for reduced approximation of the steady states.
\item 
In the \emph{online phase}, given any $\mu \in \mathcal{P}$,
we assemble the reduced model \eqref{eq:online_assembly_reduced_normal_matrix}, 
choose a reduced formulation \eqref{eq:abstract_EVP} or \eqref{eq:abstract_constraint_res_min}, 
compute a corresponding reduced solution 
$\rho_{\mathrm{eig}}^{(r)}(\mu)$ or $\rho_{\mathrm{aff}}^{(r)}(\mu)$ 
as specified later in \eqref{eq:trace_normalized_reduced_eigenvector} or \eqref{eq:reduced_affine_approximation}, respectively.
The resulting approximation is Hermitian. 
It has unit trace in the affine
formulation \eqref{eq:abstract_constraint_res_min} and, 
provided that the trace normalization is well-defined, also
in the eigenvalue formulation \eqref{eq:abstract_EVP}.
Post-processing procedures for positivity, 
as described in \Cref{subsec:post_processing}, 
can be optionally performed. 
\end{itemize}

<!-- remark-3: body_md -->
\label{rem:observable}
For an observable $O \in \mathcal{H}$, 
one is interested in {its expectation value with respect to the steady state $\rho(\mu)$},
i.e.,
\[
\langle O\rangle_{\rho(\mu)}
\coloneqq
\langle O,\rho(\mu)\rangle
=
\operatorname{tr}\bigl(O\rho(\mu)\bigr),
\]
for different parameters $\mu\in\mathcal{P}$.
Let $V$ be an offline-constructed reduced space, 
with ONB $B_1,\ldots,B_r$. 
For $\star\in\{\mathrm{eig},\mathrm{aff}\}$, 
let $x_\star^{(r)}(\mu)$ denote the coefficient vector 
of $\rho_\star^{(r)}(\mu)$ in this basis.
Once $O$ is supplied, 
its reduced representation $o_r
\coloneqq
(
\langle O,B_1\rangle,\ldots,\langle O,B_r\rangle
)^\top$ is computed once, 
and the expectation is evaluated as
\[
\langle O\rangle_{\rho_\star^{(r)}(\mu)}
\, = \, 
o_r^\top x_\star^{(r)}(\mu).
\]
Thus, for the evaluation of the observable $O$,
only the one-time construction of~$o_r$ involves full-order quantities.
Subsequent parameter queries require no state reconstruction or other full-order computations,
unless positivity postprocessing is applied.

<!-- lemma-4: body_md -->
\label[lemma]{prop:projection_postprocessing_error}
For $\rho\in A_1$ and
$\rho_{\mathrm{ss}}(\mu)\in S$,
it holds that 
\begin{equation*}
\left\|
\Pi_S(\rho)
-
\rho_{\mathrm{ss}}(\mu)
\right\| 
\leq
\left\|
\rho-\rho_{\mathrm{ss}}(\mu)
\right\| .
\end{equation*}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Metric projections onto a nonempty closed convex subset 
of a Hilbert space are nonexpansive 
by \cite[Prp.~4.8]{BauschkeCombettes2017}.
Hence, for $\Pi_S$ from \Cref{subsubsec:metric_proj} and $S$ convex,
we have $\|\Pi_S(X)-\Pi_S(Y)\|\leq\|X-Y\|$ for all $X,Y\in\mathcal{H}$. 
With $\Pi_S\bigl(\rho_{\mathrm{ss}}(\mu)\bigr) = \rho_{\mathrm{ss}}(\mu) \in S$,
the assertion follows.

<!-- lemma-5: body_md -->

Let $\rho\in A_1$, $\rho_{\mathrm{ss}}(\mu)\in S$, 
and $\Pi_{\mathrm{clip},k}(\rho)$ the partially clipped approximation from
\eqref{eq:partial_clipping_map}. 
Then, it holds that 
\begin{equation*}
\left\|
\Pi_{\mathrm{clip},k}(\rho)
-
\rho_{\mathrm{ss}}(\mu)
\right\| 
\, \leq \, 
\bigl(2+\sqrt{n_{\mathrm{clip}}}\bigr)
\left\|
\rho-\rho_{\mathrm{ss}}(\mu)
\right\| ,
\end{equation*}

\coloneqq

\bigr\}

\end{equation*}

For brevity, set
$C_k\coloneqq C_k(\rho)$,
$s_k\coloneqq s_k(\rho)$ 
and
$E\coloneqq\rho-\rho_{\mathrm{ss}}(\mu)$.
Let $\mu_1\geq\cdots\geq\mu_N\geq0$ denote the eigenvalues of

$\max\{-\lambda_j,0\}\leq|\lambda_j-\mu_j|$,
the Hoffman--Wielandt inequality
\cite[Thm.~6.3.5 and Cor.~6.3.8]{HornJohnson2012} yields

\|C_k\|^2
=
\sum_{j=N-k+1}^{N}
\max\{-\lambda_j,0\}^2 
\leq
\sum_{j=1}^{N}|\lambda_j-\mu_j|^2
\leq

\end{aligned}

=
\operatorname{tr} C_k
\leq
\sqrt{n_{\mathrm{clip}}}\,\|C_k\|
\leq
\sqrt{n_{\mathrm{clip}}}\,\|E\|.

\leq

\leq
\bigl(2+\sqrt{n_{\mathrm{clip}}}\bigr)\|E\|,
\end{aligned}
\]
which proves the assertion.

\geq

<!-- theorem-1: body_md -->
\label[theorem]{thm:weak_greedy_convergence_rates}
Suppose there exists a \emph{solution map} $u:\mathcal{P}\to\mathcal H$,
such that the solution set $\mathcal M=u(\mathcal{P})$ is compact and contained
in the closed unit ball of~$\mathcal H$.
Let $(\mu_r)_{r\in \ensuremath\mathbb{N}}$ be a sequence of snapshot parameters and 
$(V_r)_{r\in \ensuremath\mathbb{N}}$  a sequence of RB spaces, 
so that
$V_{r+1}
=
V_r+\operatorname{span}_{\ensuremath\mathbb{R}}\{u(\mu_{r})\}.$
Moreover, assume that the following conditions hold:
\begin{enumerate}
\item \label{item:weak_greedy}
\emph{(Weak greedy selection)}
Suppose that there exists a constant $\vartheta\in(0,1]$,
called the \emph{weakness parameter}, 
such that
\begin{equation*}
\mathrm{dist}(u(\mu_r), V_r)
\, \geq \, 
\vartheta
\sup_{\mu\in\mathcal{P}}
\mathrm{dist}(u(\mu), V_r)
\qquad \text{for all }
r \in \ensuremath\mathbb{N}.
\end{equation*}
\item \label{item:rate_decay} 
\emph{(Decaying Kolmogorov-$n$-width)}
There exist constants $\alpha_0, \alpha_1,\alpha_2>0$, such that
\[
d_n(\mathcal M)
\, \leq \,
\alpha_0 \exp \bigl(-\alpha_1 n^{\alpha_2} \bigr)
\qquad
\text{for all }n\in \ensuremath\mathbb{N},
\]
where $d_n(\mathcal M)$ is the Kolmogorov-$n$-width of $\mathcal{M}$ from \Cref{def:sol_set}.

\end{enumerate}
Then, with $C_1 \coloneqq \sqrt{2 \alpha_0} / \vartheta$ and
$C_2 \coloneqq 2^{-1-2\alpha_2}\alpha_1$, 
it holds that
\begin{equation}
\sup_{\mu\in\mathcal{P}}
\mathrm{dist}\bigl(u(\mu),V_{1+r}\bigr)
\,\leq\,
C_1\exp\bigl(-C_2\,  r^{\alpha_2}\bigr)
\qquad\text{for all }r\in\ensuremath\mathbb{N}.
\label{eq:weak_greedy_exponential_rate}
\end{equation}
In particular, the greedy procedure converges at an exponential rate if $\alpha_2=1$, 
or at a sub-exponential rate if $0<\alpha_2<1$.

\leq

\end{equation*}
Hence, the standard theorem can be applied to the projected greedy sequence in~$V_1^\perp$.
This explains the index shift on the left hand side of  \eqref{eq:weak_greedy_exponential_rate}.

with $\|\cdot\|_{\mathrm{op}}$ denoting the operator norm.

Let $T:\mathbb{R}^{d}\to\mathbb{R}^{d}$ be an affine bijection
satisfying $T([-1,1]^{d})=\mathcal{P}$,
and set $F\coloneqq\rho_{\mathrm{ss}}\circ T$.
By \Cref{prop:analyticity_steady_state_map},
$F$ is real-analytic on an open neighborhood of the compact cube
$[-1,1]^{d}$.
By the standard complexification result 
for finite-dimensional real-analytic maps, 
cf.~\cite[Sec.~2.2]{KrantzParks2002}, 
$F$ admits a holomorphic extension 
to a complex neighborhood of the cube, 
with values in 
$\mathcal H \otimes \mathbb{C} = \mathbb{C}^{N \times N}$.
Hence, after possibly reducing this neighborhood,
there exists $R>1$, 
such that this extension is bounded 
and holomorphic on the product Bernstein polyellipse $\mathcal E_R^{d} \subseteq \mathbb{C}^{d}$,
where $\mathcal E_R\subseteq\ensuremath\mathbb{C}$ denotes the Bernstein ellipse with parameter~$R$,
cf.~\cite[Sec.~2.2]{KrantzParks2002}.

For every integer $m\geq0$, 
let $\mathcal I_mF$ be the tensorized Chebyshev-Lobatto interpolant
of degree~$m$ in each parameter direction.
The standard interpolation estimate for functions that are holomorphic on a
Bernstein polyellipse gives
\begin{equation}
\sup_{y\in[-1,1]^{d}}
\|
F(y)-\mathcal I_mF(y)
\|
\,\leq\,
C_0R^{-m}
\label{eq:vector_chebyshev_interpolation_error}
\end{equation}
for some constant $C_0>0$ independent of~$m$;
see, e.g., \cite{NobileTamelliniTempone2016,ChkifaCohenSchwab2014,AdcockBrugiapagliaDexterMoraga2024}
for polynomial approximation and interpolation of Hilbert- or Banach-space-valued maps.

Let $(y_\nu)_{\nu\in\{0,\ldots,m\}^{d}}$ be the tensorized Chebyshev nodes.
The Lagrange representation of the interpolant
with respect to a Lagrangian basis 
$\{\ell_\nu\}_{\nu\in\{0,\ldots,m\}^d}$ 
reads
\begin{equation*}
\mathcal I_mF(y)
\, = \!
\sum_{\nu\in\{0,\ldots,m\}^{d}}
\ell_\nu(y)F(y_\nu).
\end{equation*}
Consequently, $\mathcal I_mF(y)$ belongs to the snapshot space
$\operatorname{span}_\ensuremath\mathbb{R}
\bigl\{
F(y_\nu):
\nu\in\{0,\ldots,m\}^{d}
\bigr\},$
which has dimension at most $(m+1)^{d}$.
Together with \eqref{eq:vector_chebyshev_interpolation_error}, this yields
\begin{equation}
d_{(m+1)^{d}}(\mathcal M)
\,\leq\,
C_0R^{-m}.
\label{eq:width_at_tensor_degrees}
\end{equation}

For arbitrary $n\in\ensuremath\mathbb{N}$, choose
$m_n\coloneqq\lfloor n^{1/d}\rfloor-1$.
Then $m_n\geq0$ and $(m_n+1)^{d}\leq n$.
By monotonicity of the Kolmogorov widths,
\eqref{eq:width_at_tensor_degrees}
and using
$\lfloor n^{1/d}\rfloor\geq n^{1/d}-1$,
we obtain
\[
d_n(\mathcal M)
\,\leq\,
C_0R^{-m_n}
=
C_0R^{1-\lfloor n^{1/d}\rfloor}
\,\leq\,
C_0R^2\exp\Bigl(-(\log R)n^{1/d}\Bigr).
\]
This proves \eqref{eq:generalized_exponential_width_decay}
with $\alpha_0\coloneqq C_0R^2$ and $\alpha_1\coloneqq\log R$.

\label[proposition]{prop:finite_training_set_convergence}
Suppose the assumptions of \Cref{prop:analytic_width_decay} hold,
and let $\Xi_{\mathrm{train}}\subseteq\mathcal{P}$ be nonempty and finite.
For either formulation $\star\in\{\mathrm{eig},\mathrm{aff}\}$, let
$V_{1+k,\star}^{\Xi}$ denote the space produced by
\Cref{alg:greedy_reduced_basis} after $k$ snapshot enrichments.  
Then, there exist constants $C_1,C_2>0$, 
independent of~$r$ and of the training set, 
such that
\begin{equation}
\sup_{\xi\in\Xi_{\mathrm{train}}}
\mathrm{dist}\bigl(\rho_{\mathrm{ss}}(\xi),V_{1+r,\star}^{\Xi}\bigr)
\,\leq\,
C_1\exp\Bigl(-C_2 r^{1/d}\Bigr)
\label{eq:discrete_training_manifold_convergence}
\end{equation}
for every snapshot enrichment performed by the algorithm.
If the algorithm terminates because $\Xi_{\mathrm{act}}=\emptyset$,
then its final space~$V_{r,\star}^{\Xi}$ additionally satisfies
\begin{equation}
\sup_{\mu\in\mathcal{P}}
\mathrm{dist}\bigl(\rho_{\mathrm{ss}}(\mu),V_{r,\star}^{\Xi}\bigr)
\,\leq\,
\sqrt{C_\star\varepsilon_{\mathrm{tol}}}
+
\delta_\Xi
\,\leq\,
\sqrt{C_\star\varepsilon_{\mathrm{tol}}}
+
L_u h_\Xi,
\label{eq:finite_training_set_stopping_error}
\end{equation}
where
$C_\mathrm{eig}\coloneqq\gamma_{\min}^{-1}$ and
$C_\mathrm{aff}\coloneqq\beta_{\min}^{-2}$.

First, observe that the implementation of the active training set $\Xi_{\mathrm{act}}$ 
in \Cref{alg:greedy_reduced_basis} 
selects a maximizer over the full training set $\Xi_{\mathrm{train}}$ at every enrichment: 
once a parameter is removed from~$\Xi_{\mathrm{act}}$,
its indicator is bounded by~$\varepsilon_{\mathrm{tol}}$ and remains so,
because the indicators are monotonically nonincreasing with~$r$.
As long as $\Xi_{\mathrm{act}}$ is nonempty, 
every active parameter has indicator value
strictly larger than~$\varepsilon_{\mathrm{tol}}$.
Hence, the maximizer over the active set is also a maximizer over~$\Xi_{\mathrm{train}}$.

The error equivalences from \Cref{prp:valid_weak_greedy} remain valid after restricting
the parameter set to~$\Xi_{\mathrm{train}}$.
Hence, for each formulation the algorithm is a weak greedy procedure
for the finite set~$\mathcal M_\Xi$ with the corresponding weakness parameter
as given in the proof of \Cref{cor:idealized_greedy_convergence}.
Since $\mathcal M_\Xi\subseteq\mathcal M$, 
we have
$d_n(\mathcal M_\Xi)
\,\leq\,
d_n(\mathcal M)
\,\leq\,
\alpha_0\exp\bigl(-\alpha_1 n^{1/d}\bigr)$
for all $n\in\ensuremath\mathbb{N}$.
Hence, the weak greedy theorem, 
together with the fixed-initial-space argument from
\Cref{rem:fixed_initial_space_weak_greedy},
 proves \eqref{eq:discrete_training_manifold_convergence}.


For any subspace $V\subseteq\mathcal H$ and $\mu\in\mathcal{P}$, choose
$\xi\in\Xi_{\mathrm{train}}$ minimizing the state-space distance in
\eqref{eq:intrinsic_training_set_distance}.  Since the distance to a fixed
subspace is $1$-Lipschitz, we have 
\[
\mathrm{dist}\bigl(\rho_{\mathrm{ss}}(\mu),V\bigr)
\leq
\mathrm{dist}\bigl(\rho_{\mathrm{ss}}(\xi),V\bigr)
+
\|\rho_{\mathrm{ss}}(\mu)-\rho_{\mathrm{ss}}(\xi)\|.
\]
If $\Xi_{\mathrm{act}}$ is empty, monotonicity of $\Delta_\star^{(r)}$ implies
\(
\max_{\xi\in\Xi_{\mathrm{train}}}
\Delta_\star^{(r)}(\xi)
\,\leq\,
\varepsilon_{\mathrm{tol}}.
\)
The reliability inequalities from \Cref{prp:valid_weak_greedy} then show that
$\sup_{\xi\in\Xi_{\mathrm{train}}}
\mathrm{dist}\bigl(\rho_{\mathrm{ss}}(\xi),V_{r,\star}^{\Xi}\bigr)
\,\leq\,
\sqrt{C_\star\varepsilon_{\mathrm{tol}}}.$
This estimate and the preceding discretization argument yield 
the first inequality in \eqref{eq:finite_training_set_stopping_error}.

It remains to prove that $\delta_\Xi\leq L_uh_\Xi$.
By \Cref{prop:analyticity_steady_state_map},
$D\rho_{\mathrm{ss}}$ is continuous on the compact set~$\mathcal{P}$,
so $L_u<\infty$.
Moreover, the hyperrectangle~$\mathcal{P}$ is convex.
Hence, the line segment between $\mu$ and~$\xi$ lies in~$\mathcal{P}$, 
and the mean-value estimate gives
\(
\|
\rho_{\mathrm{ss}}(\mu)-\rho_{\mathrm{ss}}(\xi)
\|
\,\leq\,
L_u\|\mu-\xi\|_2.
\)
Taking the supremum proves $\delta_\Xi\leq L_u h_\Xi$,
 and hence, 
 the remaining
inequality in \eqref{eq:finite_training_set_stopping_error}.
