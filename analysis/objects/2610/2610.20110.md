&=-\iint_{\mathbb{R}^{2N}}u(x)v(y)h(x,y)\,dx\,dy-\iint_{\mathbb{R}^{2N}}u(y)v(x)h(x,y)\,dx\,dy.
\end{split}
\]
Notice that the two integrals on the right hand side are absolutely convergent. Indeed, their integrands are supported in $\mbox{supp}(u)\times \mbox{supp}(v)$ and~$\mbox{supp}(v)\times \mbox{supp}(u)$, respectively,  which are compact subsets of $\mathbb{R}^{2N}\setminus\Delta$. Thus, their integrability follows from the fact that $h\in L^1_{loc}(\mathbb{R}^{2N}\setminus\Delta)$. 

By using the symmetry of $h$ and exchanging $x$ and~$y$ in the second integral, the two terms on the right hand side are equal.  Hence, for every~$u\in C_c^{\infty}(U)$ and~$v\in C_c^{\infty}(V)$,
\begin{equation}\label{èplqsdkowABJNCODKMsajcndbvzjknlkò}
\iint_{U\times V} u(x)v(y)h(x,y)\,dy\,dx=0.
\end{equation}

Since $h\in L^1_{loc}(U\times V)$, it defines a distribution $T_h\in\mathcal D'(U\times V)$ such that
\[
T_h(\varphi):=\iint_{U\times V} \varphi(x,y)\,h(x,y)\,dx\,dy,\quad{\mbox{for all }}\,\varphi\in C_c^{\infty}(U\times V).
\]
Accordingly, by~\eqref{èplqsdkowABJNCODKMsajcndbvzjknlkò}, we get
\[
T_h\left(u(x)v(y)\right)=0
\]
for every~$u\in C_c^{\infty}(U)$ and~$v\in C_c^{\infty}(V)$.

The uniqueness part of \cite[Theorem~5.1.1]{MR1996773}, applied to the zero distributions in $\mathcal D'(U)$ and~$\mathcal D'(V)$, implies that the only distribution~$T\in\mathcal D'(U\times V)$ satisfying
\[
T\left(u(x)v(y)\right)=0\quad\mbox{ for all } u\in C_c^{\infty}(U),\ v\in C_c^{\infty}(V),
\]
is the zero distribution. Hence
\[
T_h=0\quad\mbox{ in }\mathcal D'(U\times V).
\]
Since $T_h$ is the distribution induced by the locally integrable function $h$,  it follows that
\begin{equation}\label{vfcdxszaesrdtfghj}
h=0\quad\mbox{ a.e. in }U\times V.
\end{equation}

We now extend~\eqref{vfcdxszaesrdtfghj} to the whole $\mathbb{R}^{2N}\setminus\Delta$. To accomplish this goal, for every~$m\in\mathbb{N}$, we set
\[
K_m:=\left\{(x,y)\in\mathbb{R}^{2N} \mbox{ s.t. } |x|\leqslant m,\ |y|\leqslant m,\ |x-y|\geqslant \frac1m\right\}.
\]
Then $K_m$ is compact, $K_m\subset K_{m+1}$, and
\[
\mathbb{R}^{2N}\setminus\Delta=\bigcup_{m=1}^{+\infty} K_m.
\]
For every~$(x_0,y_0)\in K_m$, since $x_0\ne y_0$, there exist two open neighborhoods $U, V$ with $x_0\in U$, $y_0\in V$,  such that $\overline U\cap\overline V=\varnothing$. 

The sets $U\times V$ form an open cover of $K_m$. By compactness,  there exist finitely many such products $U_j\times V_j$ covering $K_m$. Since, thanks to~\eqref{vfcdxszaesrdtfghj}, $h=0$ a.e. in each $U_j\times V_j$, 
it follows that $h=0$ a.e. in $K_m$.
Since this holds for every~$m\in\mathbb{N}$, we conclude that $h=0$ a.e. in $\mathbb{R}^{2N}\setminus\Delta$, namely $k_1=k_2$ a.e. in $\mathbb{R}^{2N}\setminus\Delta$, as desired.

We first show that $J_{N,s}$ is independent of the choice of $\omega\in\mathbb S^{N-1}$. To this end, let~$\omega_1,\omega_2\in\mathbb S^{N-1}$. Hence, there exists an orthogonal matrix
