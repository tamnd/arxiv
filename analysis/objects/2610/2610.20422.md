<!-- remark-1: body_md -->
\label{remark:bilevel_inf_unb}
    We note that for $\lambda \in [\lambda_1,\lambda_2]$ such that the lower-level is unbounded, our min--min problem~\eqref{min--min} is unbounded. Moreover, for $\lambda \in [\lambda_1,\lambda_2]$ such that the lower-level is infeasible, our max--min problem~\eqref{max--min} is unbounded. However, if the lower-level was formalized instead as the following constraint within our bilevel programs:
    $$\mathbf{x} \in \arg \min_{\mathbf x' \ge 0} \; \left\{ \mathbf c^\top \mathbf x' \;\middle|\; A\mathbf x' + \lambda D \mathbf x' \leq \mathbf b \right\},$$
    a $\lambda \in [\lambda_1,\lambda_2]$ such that (i) the lower-level is unbounded would result in min--min problem~\eqref{min--min} being infeasible, and (ii) the lower-level is infeasible would result in max--min problem~\eqref{max--min} being infeasible. We adopt the formulations~\eqref{min--min} and~\eqref{max--min} because our goal is to bound the value function over $\Lambda$: they return $f(\lambda) = -\infty$ and $f(\lambda) = +\infty$ as the correct extrema, whereas the argmin-constrained formulation would instead declare the bilevel problem infeasible. This detail is relevant for the NP membership and algorithmic approaches considered in this paper.

<!-- proof-1: body_md -->
We show that every member of Matsui's NP-hard family of linear multiplicative programs~\cite{matsui1996np} can be rewritten in polynomial time as an instance of~\eqref{prob:p_min} with the same optimal value. This family of linear multiplicative programs is therefore a special case of~\eqref{prob:p_min}, from which NP-hardness follows. Matsui's family is
\begin{equation*}
\min \quad z_1 z_2 \tag{$P_1(M)$}\label{prob:p1_m}
\end{equation*}
with decision variables
\[
\mathbf u =(u_1,\dots,u_n)\in\mathbb{R}^n,\qquad
\mathbf v=(v_{ij})_{1\le i,j\le n}\in\mathbb{R}^{n\times n},
\qquad u^0,v^0,z_1,z_2\in\mathbb{R},
\]
subject to
\begin{align}
0\le u_i\le 1 &\qquad (i=1,\dots,n), \label{eq:u-bounds}\\
0\le v_{ij}\le 1 &\qquad (i,j=1,\dots,n), \label{eq:v-bounds}\\
v_{ij}\le u_i,\qquad v_{ij}\le u_j,\qquad v_{ij}\ge u_i+u_j-1
&\qquad (i\neq j), \label{eq:mc}\\
v_{ii}=u_i &\qquad (i=1,\dots,n), \label{eq:diag-rel}\\
M\mathbf u=\mathbf 1,& \label{eq:Matsui-partition}\\
u^0=\sum_{i=1}^n p^i u_i,& \label{eq:u0-def}\\
v^0=\sum_{i=1}^n\sum_{j=1}^n p^{i+j} v_{ij},& \label{eq:v0-def}\\
z_1=v^0-p+2p^{4n}+2p^{2n}u^0,& \label{eq:z1-def}\\
z_2=v^0-p+2p^{4n}-2p^{2n}u^0,& \label{eq:z2-def}
\end{align}
where $M$ is an $m\times n$ binary matrix with $n>m$ and $p=n^{n^4}$.

\noindent
We now write~\eqref{prob:p1_m} as an instance of~\eqref{prob:p_min} with decision vector
\[
\mathbf x=(\,s,\ u_1,\dots,u_n,\ v_{11},\dots,v_{nn},\ u^0,\ v^0,\ z_2,\ t\,),
\]
and parameter $\lambda$ representing $z_1$. The objective is $\mathbf c^\top \mathbf x=t$.

\noindent
All linear constraints \eqref{eq:u-bounds}–\eqref{eq:v0-def} and \eqref{eq:z2-def} are included in $A \mathbf x\le \mathbf b$, together with $s=1$ enforced by $s\le 1$, $-s\le -1$. The auxiliary variable $s$ is needed because $\lambda$ enters~\eqref{prob:p_min} only through the products $\lambda D\mathbf x$; the product $\lambda s$ therefore plays the role of $\lambda$ itself.

\noindent
The relation \eqref{eq:z1-def} is encoded as
\[
\lambda s=v^0-p+2p^{4n}+2p^{2n}u^0,
\]
i.e.,
\begin{align*}
\lambda s-v^0-2p^{2n}u^0 &\le -p+2p^{4n},\\
-\lambda s+v^0+2p^{2n}u^0 &\le p-2p^{4n}.
\end{align*}
The epigraph constraint
\[
-t+\lambda z_2\le 0
\]
enforces $t\ge \lambda z_2$.

\noindent
From the bounds on $u$ and $v$, we obtain
\[
-p+2p^{4n}\le \lambda \le n^2p^{2n}-p+2p^{4n}+2np^{3n}.
\]
For $n\ge 5$, this yields $\lambda\in[p^{4n},5p^{4n}]$ and we set $\lambda_1=p^{4n}$, $\lambda_2=5p^{4n}$.
These bounds preserve all feasible values arising from $(P_1(M))$.
Moreover, $\lambda_1$ and $\lambda_2$ are polynomially encodable, since $\log p = O(n^4 \log n)$ \cite{matsui1996np}.

\noindent
Given any feasible solution of $(P_1(M))$, define $\mathbf x=(1,\mathbf u,\mathbf v,u^0,v^0,z_2,z_1z_2)$ and $\lambda=z_1$. Then, $(\mathbf x,\lambda)$ is feasible for $(\mathcal P_{\min})$ and $\mathbf c^\top \mathbf x=z_1z_2$, hence
\[
\operatorname{opt}(\mathcal P_{\min})\le \operatorname{opt}(P_1(M)).
\]

\noindent
Conversely, let $(\mathbf x,\lambda)$ be any feasible solution for \eqref{prob:p_min}. Then, $s=1$ and the constraints imply
\[
\lambda=v^0-p+2p^{4n}+2p^{2n}u^0.
\]
Setting $z_1=\lambda$, we obtain a feasible solution of $(P_1(M))$ whose objective value satisfies $z_1z_2=\lambda z_2\le t=\mathbf c^\top\mathbf x$, by the epigraph constraint.
                                                                                                                       
Since this holds for every feasible solution of~\eqref{prob:p_min},
\[
\operatorname{opt}(P_1(M))\le \operatorname{opt}(\mathcal P_{\min}).
\]

\noindent
Thus, $\operatorname{opt}(\mathcal P_{\min})=\operatorname{opt}(P_1(M))$ and NP-hardness follows.

\label{thmmax}
The max--min program~\eqref{max--min} is NP-hard.
<!-- end theorem-3 -->
<!-- proof-2: body_md -->
We show that the specific min--min instance built in the proof of Theorem~\ref{thmmin} can in turn be rewritten in polynomial time as an instance of~\eqref{max--min} whose optimal value is its negative.
Composing the two constructions, every linear multiplicative program~\eqref{prob:p1_m} is rewritten as an instance of~\eqref{max--min}, from which NP-hardness follows.

\noindent
Consider the min--min instance built in the proof of Theorem~\ref{thmmin}, written as
\begin{equation*}
\min_{\lambda \in [\lambda_1,\lambda_2]} \;
\min_{\boldsymbol{\mu} \ge \mathbf 0} \left\{ \mathbf c_M^\top \boldsymbol{\mu} \;\middle|\; A_M \boldsymbol{\mu} + \lambda D_M \boldsymbol{\mu} \le \mathbf b_M \right\}.
\end{equation*}
Since the feasible region of~\eqref{prob:p1_m} is nonempty and bounded~\cite[Corollary~2.3]{matsui1996np}, this instance is feasible and its optimal value is finite. Note also that $\mathbf c_M \ge \mathbf 0$, since the objective of that instance is the single variable $t$.

\noindent
Consider now the max--min problem
\[
\max_{\lambda \in [\lambda_1,\lambda_2]} \;
\min_{\mathbf x \ge \mathbf 0} \left\{ \mathbf c^\top \mathbf x \;\middle|\; (A + \lambda D) \mathbf x \le \mathbf b \right\}.
\]
For a fixed value of $\lambda$, the inner minimization problem is a linear program.
Using the dual reformulation introduced in Section~\ref{subsec:aux} and the change of variables $\boldsymbol\mu = -\mathbf y \ge \mathbf 0$, its dual can be expressed as
\begin{equation*}
-\min_{\boldsymbol{\mu} \ge \mathbf 0} \left\{ \mathbf b^\top \boldsymbol{\mu} \;\middle|\; -A^\top \boldsymbol{\mu} + \lambda (-D)^\top \boldsymbol{\mu} \le \mathbf c \right\}.
\end{equation*}
We now choose the data $(A,D,\mathbf b,\mathbf c)$ so that this dual problem coincides with the min--min instance above. Setting
\begin{equation*}
A := -A_M^\top, \qquad D := -D_M^\top, \qquad \mathbf b := \mathbf c_M, \qquad \mathbf c := \mathbf b_M,
\end{equation*}
and keeping the same interval $[\lambda_1,\lambda_2]$, the dual of the inner problem becomes
\[
-\min_{\boldsymbol{\mu} \ge \mathbf 0} \left\{ \mathbf c_M^\top \boldsymbol{\mu} \;\middle|\; A_M \boldsymbol{\mu} + \lambda D_M \boldsymbol{\mu} \le \mathbf b_M \right\}. \qquad \textrm{($\boldsymbol\mu$-problem)}
\]


\noindent
Fix $\lambda \in [\lambda_1,\lambda_2]$ and consider the $\boldsymbol\mu$-problem obtained by fixing $\lambda$ in the min--min instance, whose feasible set is $\{\boldsymbol\mu \ge \mathbf 0 \mid A_M \boldsymbol\mu + \lambda D_M \boldsymbol\mu \le \mathbf b_M\}$. Whenever this set is nonempty, the $\boldsymbol\mu$-problem is bounded: its objective is the variable $t$, the epigraph constraint gives $t \ge \lambda z_2$, and $\lambda$ and $z_2$ are bounded, the latter because the construction of Theorem~\ref{thmmin} preserves the feasible points of~\eqref{prob:p1_m}, whose feasible region is bounded~\cite[Corollary~2.3]{matsui1996np}. Two cases arise. If the feasible set is nonempty, the $\boldsymbol\mu$-problem has a finite optimal value; the same then holds for its dual, which is the inner problem of the max--min instance, and strong duality yields
\[
\min_{\mathbf x \ge \mathbf 0} \left\{ \mathbf c^\top \mathbf x \;\middle|\; (A + \lambda D) \mathbf x \le \mathbf b \right\}
=
-\min_{\boldsymbol{\mu} \ge \mathbf 0} \left\{ \mathbf c_M^\top \boldsymbol{\mu} \;\middle|\; A_M \boldsymbol{\mu} + \lambda D_M \boldsymbol{\mu} \le \mathbf b_M \right\}.
\]
If instead the feasible set is empty, the $\boldsymbol\mu$-problem is infeasible and the right-hand side above equals $-\infty$, under the convention that a minimization over an empty feasible set equals $+\infty$. The left-hand side equals $-\infty$ as well: the point $\mathbf x = \mathbf 0$ is feasible for the inner problem since $\mathbf b = \mathbf c_M \ge \mathbf 0$, and an infeasible dual then forces the inner problem to be unbounded. The identity above therefore holds for every $\lambda \in [\lambda_1,\lambda_2]$.

\noindent
By substituting this expression into the max--min formulation, we have 
\begin{align*}
&\max_{\lambda \in [\lambda_1, \lambda_2]} \min_{\mathbf x \ge \mathbf 0} 
\left\{ \mathbf c^\top \mathbf x \;\middle|\; (A+\lambda D) \mathbf x \le \mathbf b \right\} \\
&\qquad = \max_{\lambda \in [\lambda_1, \lambda_2]} 
\left( -\min_{\boldsymbol{\mu} \ge \mathbf 0} 
\left\{ \mathbf c_M^\top \boldsymbol{\mu} \;\middle|\; A_M \boldsymbol{\mu} + \lambda D_M \boldsymbol{\mu} \le \mathbf b_M \right\} \right) \\
&\qquad = -\min_{\lambda \in [\lambda_1, \lambda_2]} \; \min_{\boldsymbol{\mu} \ge \mathbf 0} 
\left\{ \mathbf c_M^\top \boldsymbol{\mu} \;\middle|\; A_M \boldsymbol{\mu} + \lambda D_M \boldsymbol{\mu} \le \mathbf b_M \right\}.
\end{align*}
The constructed max--min instance therefore has optimal value $-\operatorname{opt}(P_1(M))$ by Theorem~\ref{thmmin}, and computing it is NP-hard.

We rewrite the lower-level linear program~\ref{prob:parametric_lp} in standard form by introducing slack variables $\mathbf{s} \in \mathbb{R}^m_{\ge 0}$. Setting $N = n+m$, $\mathbf{z} = (\mathbf{x}, \mathbf{s}) \in \mathbb{R}^N_{\ge 0}$, and $\mathbf{\bar c} = (\mathbf{c}, \mathbf{0}) \in \mathbb{R}^N$, the value function becomes
\[ 
f(\lambda)=\min_{\mathbf{z} \in \mathbb{R}^N} \left\{\mathbf{\bar c}^\top \mathbf{z}  \;\middle|\;  \mathbf{z} \in \mathcal{X}(\lambda):= \left\{\mathbf{z} \ge \mathbf{0} \ \middle|\ \left[ A+ \lambda D \;\middle|\; I_m \right] \mathbf{z}= \mathbf{b}\right\}\right\}.
\]

                                  
\subsubsection{Proof for \ref{D-min}}
Problem \ref{D-min} is a YES-instance if and only if there exists $\lambda \in [\lambda_1, \lambda_2]$ such that $\mathcal{X}(\lambda) \neq \emptyset$ and either $f(\lambda) \le K$ or $f(\lambda) = -\infty$. 

The nondeterministic polynomial-time machine for \ref{D-min} operates as follows:
\begin{enumerate}[label=(\roman*)]
    \item Nondeterministically guess a basis $\mathcal{B} \subseteq \{1, \ldots, N\}$ with $|\mathcal{B}| = m$, and an indicator $j \in \bar{\mathcal{B}} \cup \{\emptyset\}$, where $\bar{\mathcal{B}} = \{1, \ldots, N\} \setminus \mathcal{B}$.
    \item\label{step:optimalii}  If $j = \emptyset$, verify whether there exists $\lambda \in [\lambda_1, \lambda_2]$ such that $\mathcal{B}$ yields a basic feasible solution satisfying $\mathbf{\bar c}^\top \mathbf{z}(\lambda) \le K$.
    \item\label{step:unboundediii}  If $j \in \bar{\mathcal{B}}$, verify whether there exists $\lambda \in [\lambda_1, \lambda_2]$ such that $\mathcal{B}$ is primal feasible and $j$ induces an unbounded recession ray certifying $f(\lambda) = -\infty$.
    \item Accept if the verification in either step (ii) or step (iii) succeeds.
\end{enumerate}

To perform the verification, let $M_{\mathcal{B}}(\lambda)$ be the $m \times m$ submatrix of $[A+\lambda D \mid I_m]$ corresponding to $\mathcal{B}$. By Cramer's rule, the basic variables $\mathbf{z}_{\mathcal{B}}(\lambda) = M_{\mathcal{B}}(\lambda)^{-1}\mathbf{b}$ and the components of the direction vector $\mathbf{d}_{\mathcal{B}}^{(j)}(\lambda) = -M_{\mathcal{B}}(\lambda)^{-1}\left[A+\lambda D \mid I_m\right]_j$ are the rational functions in $\lambda$
\[
z_{{\mathcal{B}},i}(\lambda) = \frac{p_{\mathcal{B},i}(\lambda)}{q_{\mathcal{B}}(\lambda)}, 
\qquad
d_{{\mathcal{B}},i}^{(j)}(\lambda) = \frac{r_{\mathcal{B},i}^{(j)}(\lambda)}{q_{\mathcal{B}}(\lambda)}, 
\qquad
q_{\mathcal{B}}(\lambda) = \det(M_{\mathcal{B}}(\lambda)),
\]
where $p_{\mathcal{B},i}(\lambda) = \det(M_{\mathcal{B},i}(\lambda))$ and $r_{\mathcal{B},i}^{(j)}(\lambda) = \det(M_{\mathcal{B},i}^{(j)}(\lambda))$, with $M_{\mathcal{B},i}(\lambda)$ and $M_{\mathcal{B},i}^{(j)}(\lambda)$ obtained by replacing the $i$-th column of $M_{\mathcal{B}}(\lambda)$ with $\mathbf{b}$ and $-\left[A+\lambda D \mid I_m\right]_j$, respectively. By Lemma~\ref{lem:parametric-det} in Appendix~\ref{appendix:np}, the polynomials $p_{\mathcal{B},i}(\lambda)$, $r_{\mathcal{B},i}^{(j)}(\lambda)$, and $q_{\mathcal{B}}(\lambda)$ have degree at most $m$ with polynomially bounded coefficients.

For $j = \emptyset$, step~\ref{step:optimalii} checks the feasibility over $\lambda \in [\lambda_1, \lambda_2]$ of
\begin{equation} \label{sys:dmin_bounded}
\mathbf{z}_{\mathcal{B}}(\lambda) \ge \mathbf{0},  \qquad \mathbf{\bar c}_{\mathcal{B}}^\top \mathbf{z}_{\mathcal{B}}(\lambda) \le K, \qquad q_{\mathcal{B}}(\lambda) \neq 0.
\end{equation}
For $j \in \bar{\mathcal{B}}$, step~\ref{step:unboundediii} sets $d_j^{(j)}(\lambda) = 1$ and $d_k^{(j)}(\lambda) = 0$ for $k \in \bar{\mathcal{B}} \setminus \{j\}$, and checks the feasibility over $\lambda \in [\lambda_1, \lambda_2]$ of
\begin{equation} \label{sys:dmin_unbounded_rigorous}
\mathbf{z}_{\mathcal{B}}(\lambda) \ge \mathbf{0}, \qquad \mathbf{d}_{\mathcal{B}}^{(j)}(\lambda) \ge \mathbf{0}, \qquad \mathbf{\bar c}^\top \mathbf{d}^{(j)}(\lambda) \le -1, \qquad q_{\mathcal{B}}(\lambda) \neq 0.
\end{equation}

Write $K=\frac{u}{v}$, where $u\in\mathbb{Z}$ and $v\in\mathbb{Z}^+_0$.
Multiplying the objective inequality by $v$ and then the inequalities in \eqref{sys:dmin_bounded} and \eqref{sys:dmin_unbounded_rigorous} by $\left(q_{\mathcal{B}}(\lambda)\right)^2$ clears all denominators without altering inequality signs. To enforce $q_{\mathcal{B}}(\lambda) \neq 0$, we include the strict inequality $\left(q_{\mathcal{B}}(\lambda)\right)^2 > 0$. In either step, this yields an equivalent system of univariate polynomial inequalities in $\lambda$ of degree at most $2m$ with coefficients of polynomial encoding length.
By Lemma~\ref{lem:rational-to-integer} in Appendix~\ref{appendix:np}, the resulting system can be transformed in polynomial time into an equivalent system of univariate polynomial constraints with integer coefficients. Therefore, Lemma~\ref{lem:univariate} in Appendix~\ref{appendix:np} applies and feasibility of this resulting system can be decided in polynomial time. This completes the proof that \ref{D-min} belongs to NP.

                                  
\subsubsection{Proof for \ref{D-max}}
By Remark~\ref{remark:bilevel_inf_unb}, if the lower-level problem~\ref{prob:parametric_lp} is infeasible, $f(\lambda) = +\infty \ge K$ holds trivially. Otherwise, for any lower-level dual feasible solution $\mathbf{y} \le \mathbf{0}$ satisfying $(A+\lambda D)^\top \mathbf{y} \le \mathbf{c}$, weak duality implies $f(\lambda) \ge \mathbf{b}^\top \mathbf{y}$. Hence, \ref{D-max} is a YES-instance if and only if there exists $\lambda \in [\lambda_1, \lambda_2]$ at which the dual program of~\ref{prob:parametric_lp} is feasible with optimal value at least $K$, or~\ref{prob:parametric_lp} is infeasible. Thus, the rest of the proof follows a logic similar to the one for \ref{D-min}, with step~\ref{step:optimalii} being adapted to focus on the optimality of the dual basis and with step~\ref{step:unboundediii}  being adapted to focus on the infeasibility of the lower-level.

We express the dual of $\mathcal{P}(\lambda)$ in standard form by introducing slack variables $\mathbf{v} \in \mathbb{R}^n_{\ge 0}$ and defining $\mathbf{w} = (-\mathbf{y}, \mathbf{v}) \in \mathbb{R}^N_{\ge 0}$, namely
\[
\max_{\mathbf{w} \ge \mathbf{0}} \left\{ -\bar{\mathbf{b}}^\top \mathbf{w} \;\middle|\; \left[-(A+\lambda D)^\top \;\middle|\; I_n \right] \mathbf{w} = \mathbf{c} \right\},
\]
where $\bar{\mathbf{b}} = (\mathbf{b}, \mathbf{0}) \in \mathbb{R}^N$. Let $M_D(\lambda) = [-(A+\lambda D)^\top \mid I_n] \in \mathbb{R}^{n \times N}$ be the dual constraint matrix. 

The nondeterministic polynomial-time machine for \ref{D-max} operates as follows:
\begin{enumerate}[label=(\roman*)]
    \item Nondeterministically guess a dual basis $\mathcal{B}_y \subseteq \{1, \ldots, N\}$ with $|\mathcal{B}_y| = n$, and an indicator $j \in \bar{\mathcal{B}}_y \cup \{\emptyset\}$, where $\bar{\mathcal{B}}_y = \{1, \ldots, N\} \setminus \mathcal{B}_y$.
    \item If $j = \emptyset$, verify whether there exists $\lambda \in [\lambda_1, \lambda_2]$ such that $\mathcal{B}_y$ yields a dual basic feasible solution satisfying $-\bar{\mathbf{b}}^\top \mathbf{w}(\lambda) \ge K$.
    \item If $j \in \bar{\mathcal{B}}_y$, verify whether there exists $\lambda \in [\lambda_1, \lambda_2]$ such that $j$ induces a ray in the homogeneous dual system certifying primal infeasibility via Farkas' Lemma.
    \item Accept if the verification in either step (ii) or step (iii) succeeds.
\end{enumerate}

Let $M_{D,\mathcal{B}_y}(\lambda)$ be the $n \times n$ submatrix of $M_D(\lambda)$ corresponding to $\mathcal{B}_y$. By Cramer's rule, the dual basic variables $\mathbf{w}_{\mathcal{B}_y}(\lambda) = M_{D,\mathcal{B}_y}(\lambda)^{-1}\mathbf{c}$ and the basic components of the dual direction vector $\mathbf{d}_y^{(j)}(\lambda) = -M_{D,\mathcal{B}_y}(\lambda)^{-1}\left[M_D(\lambda)\right]_j$ are rational functions in $\lambda$ with common denominator determinant $q_{\mathcal{B}_y}(\lambda) = \det(M_{D,\mathcal{B}_y}(\lambda))$.

For $j = \emptyset$, step (ii) checks the feasibility over $\lambda \in [\lambda_1, \lambda_2]$ of
\begin{equation} \label{sys:dmax_bounded}
\mathbf{w}_{\mathcal{B}_y}(\lambda) \ge \mathbf{0}, \qquad \mathbf{w}_{\bar{\mathcal{B}}_y}(\lambda) = \mathbf{0}, \qquad -\bar{\mathbf{b}}^\top \mathbf{w}(\lambda) \ge K, \qquad q_{\mathcal{B}_y}(\lambda) \neq 0.
\end{equation}
For $j \in \bar{\mathcal{B}}_y$, step (iii) sets $d_{y,j}^{(j)}(\lambda) = 1$ and $d_{y,k}^{(j)}(\lambda) = 0$ for $k \in \bar{\mathcal{B}}_y \setminus \{j\}$, certifying primal infeasibility by checking the feasibility over $\lambda \in [\lambda_1, \lambda_2]$ of
\begin{equation} \label{sys:dmax_infeasible}
\mathbf{d}_{y,\mathcal{B}_y}^{(j)}(\lambda) \ge \mathbf{0}, \qquad -\bar{\mathbf{b}}^\top \mathbf{d}_y^{(j)}(\lambda) \ge 1, \qquad q_{\mathcal{B}_y}(\lambda) \neq 0.
\end{equation}

Write $K=\frac{u}{v}$, where $u\in\mathbb{Z}$ and $v\in\mathbb{Z}^+_0$.
Multiplying the objective inequality by $v$ and then the inequalities in \eqref{sys:dmax_bounded} and \eqref{sys:dmax_infeasible} by $\left(q_{\mathcal{B}_y}(\lambda)\right)^2$ clears all denominators without altering inequality signs, and enforcing $\left(q_{\mathcal{B}_y}(\lambda)\right)^2 > 0$ guarantees non-singularity. This converts both conditions into equivalent systems of univariate polynomial inequalities in $\lambda$ of degree at most $2n$ with coefficients of polynomial encoding length.
By Lemma~\ref{lem:rational-to-integer} in Appendix~\ref{appendix:np}, the resulting system can be transformed in polynomial time into an equivalent system of univariate polynomial constraints with integer coefficients. Therefore, Lemma~\ref{lem:univariate} in Appendix~\ref{appendix:np} applies and feasibility over $\left[\lambda_1, \lambda_2\right]$ is decidable in polynomial time. This completes the proof that \ref{D-max} belongs to NP.

<!-- proposition-1: body_md -->
\label{prop:uniform_sign}
Assume that all entries of $D$ have the same sign.
\begin{itemize}
    \item If $D_{ij} \geq 0 \ \forall i,j$, then $f(\lambda)$ is nondecreasing on $[\lambda_1,\lambda_2]$.
    
    \item If $D_{ij} \leq 0 \ \forall i,j$, then $f(\lambda)$ is nonincreasing on $[\lambda_1,\lambda_2]$.
\end{itemize}
<!-- end proposition-1 -->
<!-- proof-5: body_md -->
Let $\lambda',\lambda'' \in [\lambda_1,\lambda_2]$ with $\lambda' \le \lambda''$.

\medskip
\noindent
Assume first that $D_{ij} \geq 0 \ \forall i,j$.
Since $\mathbf x \ge \mathbf 0$, we have $D\mathbf x \ge \mathbf 0$ and therefore
\[
\lambda' D\mathbf x \le \lambda'' D\mathbf x.
\]
Hence,
\[
A\mathbf x + \lambda' D\mathbf x
\le
A\mathbf x + \lambda'' D\mathbf x.
\]
It follows that every feasible solution for $\lambda''$ is also feasible for $\lambda'$, i.e.,
\[
F(\lambda'') \subseteq F(\lambda'),
\]
where $F(\lambda)$ denotes the feasible set of $\mathcal P(\lambda)$.
Therefore,
\[
f(\lambda') \le f(\lambda''),
\]
showing that $f$ is nondecreasing on $[\lambda_1,\lambda_2]$.

\medskip
\noindent
The case $D_{ij} \leq 0 \ \forall i,j$ follows similarly and yields that $f$ is nonincreasing on $[\lambda_1,\lambda_2]$.
<!-- end proof-5 -->
<!-- corollary-1: body_md -->
Assume that all entries of $D$ have the same sign.
\begin{itemize}
    \item If $D_{ij} \geq 0 \ \forall i,j$, then
    \[
    \min_{\lambda \in [\lambda_1,\lambda_2]} f(\lambda)=f(\lambda_1),
    \qquad
    \max_{\lambda \in [\lambda_1,\lambda_2]} f(\lambda)=f(\lambda_2).
    \]

    \item If $D_{ij} \leq 0 \ \forall i,j$, then
    \[
    \min_{\lambda \in [\lambda_1,\lambda_2]} f(\lambda)=f(\lambda_2),
    \qquad
    \max_{\lambda \in [\lambda_1,\lambda_2]} f(\lambda)=f(\lambda_1).
    \]
\end{itemize}

Consequently, problems~\eqref{min--min} and~\eqref{max--min} are polynomially solvable,
since computing $f(\lambda)$ for a single value of $\lambda$ can be done in polynomial time.
<!-- end corollary-1 -->
<!-- head-1: body_md -->
Having all coefficients of $D$ with the same sign cannot occur for models with parametric equality constraints, since such an equality is written as two inequalities whose coefficients have opposite signs to match the canonical model we consider in this paper.

<!-- definition-1: body_md -->
A function $f$ is quasi-concave on an interval if for any $\lambda', \lambda''$ in the interval and any $\alpha \in [0,1]$,
\[
f(\alpha \lambda' + (1-\alpha)\lambda'') \geq \min\{f(\lambda'), f(\lambda'')\}.
\]
<!-- end definition-1 -->
<!-- proposition-2: body_md -->
\label{prop:one_row}
If $D$ has exactly one non-zero row, then the value function $f(\lambda)$ is quasi-concave on $[\lambda_1, \lambda_2]$.
<!-- end proposition-2 -->
<!-- proof-6: body_md -->
Let $\lambda',\lambda''\in[\lambda_1,\lambda_2]$ and define
\[
\hat\lambda=\alpha\lambda'+(1-\alpha)\lambda'', \qquad \alpha\in[0,1].
\]
Let $\mathbf x^{\star}\geq \mathbf 0$ be an optimal solution at $\hat\lambda$. Then,
\[
B\mathbf x^{\star}\le \boldsymbol{\beta},\quad (\mathbf{a_{i_0}} + \hat{\lambda} \mathbf{d_{i_0}}) \mathbf x^{\star} \leq b_{i_0}.
\]
Suppose, by contradiction, that $\mathbf x^{\star}$ is infeasible for both $\lambda'$ and $\lambda''$. Then,
\[
(\mathbf{a_{i_0}} + \lambda' \mathbf{d_{i_0}}) \mathbf x^{\star} > b_{i_0}
\qquad\text{and}\qquad
(\mathbf{a_{i_0}} + \lambda'' \mathbf{d_{i_0}}) \mathbf x^{\star} > b_{i_0}.
\]
Multiplying the first inequality by $\alpha \geq 0$ and the second by $1-\alpha \geq 0$, and summing, leads to
\[
(\mathbf{a_{i_0}} + \hat{\lambda} \mathbf{d_{i_0}}) \mathbf x^{\star} > b_{i_0},
\]
which contradicts feasibility at $\hat\lambda$.

Hence, $\mathbf x^{\star}$ is feasible for at least one of the problems associated with $\lambda'$ or $\lambda''$.
If it is feasible for $\lambda'$, then by optimality of $f(\lambda')$ in a minimization problem,
\[
f(\lambda')\le \mathbf c^\top \mathbf x^{\star}=f(\hat\lambda).
\]
Similarly, if it is feasible for $\lambda''$, then
\[
f(\lambda'')\le \mathbf c^\top \mathbf x^{\star}=f(\hat\lambda).
\]
Therefore,
\[
f(\hat\lambda)\ge \min\{f(\lambda'),f(\lambda'')\},
\]
which proves quasi-concavity.
<!-- end proof-6 -->
<!-- head-2: body_md -->
the proof extends to the mixed-integer case since it relies on the same solution $\mathbf{x}^{\star}$ and does not construct new solutions through convex combinations. Hence, any integrality constraints satisfied by $\mathbf{x}^{\star}$ are preserved.
<!-- end head-2 -->
<!-- corollary-2: body_md -->
Suppose $D$ has exactly one non-zero row. Then, the minimum value of $f$ over $[\lambda_1,\lambda_2]$ is attained at one of the endpoints, namely
\[
\min_{\lambda\in[\lambda_1,\lambda_2]} f(\lambda)
=
\min\{f(\lambda_1),f(\lambda_2)\}.
\]
<!-- end corollary-2 -->
<!-- corollary-3: body_md -->
Suppose $D$ has exactly one non-zero row.
Then, a maximum of $f$ over $[\lambda_1,\lambda_2]$ can be approximated within accuracy $\varepsilon$ by applying the standard bisection method for quasiconvex optimization to $-f$ (see \cite{boyd2004convex}).
In particular, given an initial interval $[l,u]$ containing the optimal value of $\min_{\lambda\in[\lambda_1,\lambda_2]} -f(\lambda)$, the method terminates in
\begin{equation*}
O\!\left(
\log\!\left(
\frac{u-l}{\varepsilon}
\right)
\right)
\end{equation*}
iterations, each iteration requiring to solve one LP.
<!-- end corollary-3 -->
<!-- head-3: body_md -->
In this case again, having exactly one non-zero row in $D$ cannot occur for models with parametric equality constraints, since such an equality is written as two inequalities to match the canonical model considered in this paper.

<!-- definition-2: body_md -->
A function $f : \mathbb{R}^n \to \mathbb{R}$ is called \emph{quasi-convex}
if its domain $\mathrm{dom}(f)$ and all its sublevel sets
\[
S_t = \{x \in \mathrm{dom}(f) \mid f(x) \le t\},
\qquad \forall t \in \mathbb{R},
\]
are convex.
<!-- end definition-2 -->
<!-- proposition-3: body_md -->
\label{prop:one_col}
If $D$ has exactly one non-zero column, then the value function $f(\lambda)$ is quasi-convex on $[\lambda_1, \lambda_2]$.
<!-- end proposition-3 -->
<!-- proof-7: body_md -->
Let
\[
S_t = \{\lambda \in [\lambda_1,\lambda_2] \mid f(\lambda)\le t\}.
\]
We show that $S_t$ is convex.

Take $\lambda',\lambda'' \in S_t$.
Then, $ f(\lambda')\le t \quad \text{and} \quad f(\lambda'')\le t $.
Hence, there exist feasible solutions $(\mathbf x',x_0')$ for $\lambda'$ and $(\mathbf x'',x_0'')$ for $\lambda''$,
such that
\[
\mathbf c^\top \mathbf x' + c_0 x_0' \le t,
\qquad
\mathbf c^\top \mathbf x'' + c_0 x_0'' \le t,
\]
and
\[
A\mathbf x' + (A_0+\lambda' \mathbf d)x_0' \le \mathbf b,
\qquad
A\mathbf x'' + (A_0+\lambda'' \mathbf d)x_0'' \le \mathbf b.
\]

Let $\alpha \in [0,1]$ and let $\hat\lambda = \alpha \lambda' + (1-\alpha)\lambda''$.
To prove that $\hat\lambda\in S_t$, we must find a feasible solution for parameter $\hat\lambda$ with objective value at most $t$.

The idea is to construct such a solution from the feasible solutions associated with \(\lambda'\) and \(\lambda''\).
For this purpose, we introduce the coefficient

\[
\theta
=
\frac{\alpha x_0''}
{\alpha x_0'' + (1-\alpha)x_0'},
\quad \text{and \ therefore} \quad
1-\theta
=
\frac{(1-\alpha)x_0'}
{\alpha x_0'' + (1-\alpha)x_0'}.
\]
Set
\[
\hat{\mathbf x}
=
\theta \mathbf x'
+
(1-\theta)\mathbf x''
\quad \text{and} \quad
\hat x_0
=
\theta x_0'
+
(1-\theta)x_0''.
\]
By construction,
\[
\hat x_0
=
\frac{x_0' x_0''}
{\alpha x_0'' + (1-\alpha)x_0'}.
\]
Moreover,
\[
\hat\lambda \hat x_0
=
(\alpha\lambda' + (1-\alpha)\lambda'')
\frac{x_0'x_0''}
{\alpha x_0'' + (1-\alpha)x_0'}.
\]
On the other hand,
\[
\theta \lambda' x_0'
+
(1-\theta)\lambda'' x_0''
=
\frac{\alpha\lambda' x_0' x_0''
+
(1-\alpha)\lambda'' x_0' x_0''}
{\alpha x_0'' + (1-\alpha)x_0'}.
\]
Thus,
\[
\hat\lambda \hat x_0
=
\theta \lambda' x_0'
+
(1-\theta)\lambda'' x_0''.
\]
This identity allows the nonlinear term $\lambda x_0$ to be expressed as a convex combination of the corresponding terms associated with $\lambda'$ and $\lambda''$.

\noindent
We now verify feasibility, i.e.,
\[
A\hat{\mathbf x}
+
(A_0+\hat\lambda \mathbf d)\hat x_0.
\]
Using the definitions above,
\[
\begin{aligned}
A\hat{\mathbf x}
+
(A_0+\hat\lambda \mathbf d)\hat x_0
&=
A(\theta\mathbf x' + (1-\theta)\mathbf x'')
\\
&\quad
+
A_0(\theta x_0' + (1-\theta)x_0'')
\\
&\quad
+
\mathbf d\, \hat\lambda \hat x_0.
\end{aligned}
\]
Since
\[
\hat\lambda \hat x_0
=
\theta \lambda' x_0'
+
(1-\theta)\lambda'' x_0'',
\]
we obtain
\[
\begin{aligned}
A\hat{\mathbf x}
+
(A_0+\hat\lambda \mathbf d)\hat x_0
&=
\theta
\left(
A\mathbf x'
+
(A_0+\lambda'\mathbf d)x_0'
\right)
\\
&\quad
+
(1-\theta)
\left(
A\mathbf x''
+
(A_0+\lambda''\mathbf d)x_0''
\right).
\end{aligned}
\]
Because both vectors inside the parentheses satisfy the constraints
\[
A\mathbf x' + (A_0+\lambda'\mathbf d)x_0' \le \mathbf b,
\quad \text{and} \quad
A\mathbf x'' + (A_0+\lambda''\mathbf d)x_0'' \le \mathbf b,
\]
their convex combination also satisfies them. Therefore,
\[
A\hat{\mathbf x}
+
(A_0+\hat\lambda \mathbf d)\hat x_0
\le \mathbf b.
\]
Hence, $(\hat{\mathbf x},\hat x_0)$ is feasible for $\hat\lambda$.

\noindent
Finally,
\[
\begin{aligned}
\mathbf c^\top \hat{\mathbf x}
+
c_0 \hat x_0
&=
\mathbf c^\top
(\theta\mathbf x' + (1-\theta)\mathbf x'')
+
c_0(\theta x_0' + (1-\theta)x_0'')
\\
&=
\theta
(\mathbf c^\top\mathbf x' + c_0x_0')
+
(1-\theta)
(\mathbf c^\top\mathbf x'' + c_0x_0'')
\\
&\le
\theta t + (1-\theta)t
\\
&= t.
\end{aligned}
\]
Thus, $(\hat{\mathbf x}, \hat x_0)$ is a feasible solution for $\hat{\lambda}$ whose objective does not exceed $t$.
Consequently,
\[
f(\hat\lambda)\le t,
\]
which implies that $\hat\lambda \in S_t$.

Since every convex combination of two points in $S_t$ remains in $S_t$, the set $S_t$ is convex.
Therefore every sublevel set of $f$ is convex, and hence $f$ is quasi-convex.
\medskip

\noindent
\textit{Remark.}
If $x_0'=x_0''=0$, the definition of $\theta$ is not valid.
However, in this case the bilinear term disappears and feasibility follows directly by convexity of the remaining linear constraints.

<!-- lemma-1: body_md -->
\label{lem:parametric-det}
Let $M(\lambda)$ be an $m\times m$ matrix whose entries are affine functions
of $\lambda$ with rational coefficients. Then, the following holds:
\begin{enumerate}[label=(\roman*)]
\item\label{lemma:assertion1} $\det(M(\lambda))$ is a polynomial of degree at most $m$;
\item\label{lemma:assertion3} the coefficients of $\det(M(\lambda))$ can be computed in polynomial time in the encoding length of $M$;
\item\label{lemma:assertion2} the coefficients of $\det(M(\lambda))$ have encoding length polynomial in the encoding length of $M$.
\end{enumerate}
<!-- end lemma-1 -->
<!-- proof-8: body_md -->
By the Leibniz formula, we have
\begin{equation*}
\det(M(\lambda))
=
\sum_{\sigma\in S_m}
\operatorname{sgn}(\sigma)
\prod_{i=1}^m
M_{i,\sigma(i)}(\lambda),
\end{equation*}
where $S_m$ is the permutation group. Each factor $M_{i,\sigma(i)}(\lambda)$ is affine in $\lambda$, hence every product has degree at most
$m$, proving assertion~\ref{lemma:assertion1}.

Next, we prove assertions~\ref{lemma:assertion3} and~\ref{lemma:assertion2}. By assertion~\ref{lemma:assertion1}, the polynomial $\det(M(\lambda))$ has degree at most $m$. Hence, it is uniquely determined by its values at any $m+1$ distinct points. For each integer $\lambda^\prime \in\{0,1,\ldots,m\}$, the matrix $M(\lambda^\prime)$ can be constructed in polynomial time and has integer entries of polynomial encoding length. Therefore, each determinant $\det(M(\lambda^\prime))$ can be computed in polynomial time using any polynomial-time algorithm for the determinant of an integer matrix.

Having computed the values $\det(M(0)), \det(M(1)),\ldots, \det(M(m))$, the coefficients of the unique polynomial $\det(M(\lambda))$ of degree at most $m$ are recovered by polynomial interpolation on these $m+1$ evaluation points, which can be performed in polynomial time using standard polynomial interpolation algorithms. Hence, these coefficients have encoding length polynomial in the encoding length of M.

<!-- lemma-2: body_md -->
\label{lem:rational-to-integer}
Let
\begin{equation*}
f_1(\lambda),\ldots,f_s(\lambda)
\in
\mathbb{Q}[\lambda]
\end{equation*}
be univariate polynomials.
Then there exist other univariate polynomials
\begin{equation*}
g_1(\lambda),\ldots,g_s(\lambda)
\in
\mathbb{Z}[\lambda] 
\end{equation*}
that can be computed in polynomial time such that, for every $i=1, \ldots, s$,
\begin{equation*}
f_i(\lambda)\;\Delta_i\;0 \quad\Longleftrightarrow\quad g_i(\lambda)\;\Delta_i\;0,
\qquad
\Delta_i\in
\{<,\le,=,\ge,>\},
\end{equation*}
<!-- end lemma-2 -->
<!-- proof-9: body_md -->
Fix a polynomial
\begin{equation*}
    f(\lambda)=\sum_{k=0}^{d}\frac{a_k}{b_k}\lambda^k 
\end{equation*}
 where $a_k\in\mathbb{Z}$ and $b_k\in\mathbb{Z}^+_0$.
 Let 
 \begin{equation*}
    D=\prod_{k=0}^{d} b_k     
 \end{equation*}
 and define
 \begin{equation*}
     g(\lambda)=D\,f(\lambda)\in\mathbb{Z}[\lambda].
 \end{equation*}
 Since $D>0$, multiplication by $D$ preserves all comparison relations with zero.
 Moreover, the binary encoding length of $D$ is the sum of the encoding lengths of the denominators $b_k$, and therefore is polynomial in the encoding length of $f$. Consequently, the coefficients of $g$ also have polynomial encoding length.

\label{lem:univariate}

Let

\qquad
\Delta_i\in
