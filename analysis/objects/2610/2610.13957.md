<!-- head-1: body_md -->
\cite[Lemma 1]{LXYZ} \textit{If \(M\) is a Lipschitz star body in \(\mathbb{R}^n\) and
	\(1\leq p<n\), then}
		\begin{equation}\label{1.1}
		S_p(M)
		\geq
		\left(\frac{p-1}{n-p}\right)^{p-1}C_p(M)
		\geq
		n\omega_n^{\frac{p}{n}}
		V(M)^{\frac{n-p}{n}},
	\end{equation}
	\textit{ and}
	\begin{equation}\label{1.2}
		S_p(M)
		\geq
		\Phi_p(M)
		\geq
		n\omega_n^{\frac{p}{n}}
		V(M)^{\frac{n-p}{n}}.
	\end{equation}
	\textit{Equality in \eqref{1.1} holds if \(M\) is a ball centered at the
	origin. Equality in the right-hand inequality of \eqref{1.2} holds
	for \(p>1\) if and only if \(M\) is an ellipsoid centered at the
	origin, and for \(p=1\) if and only if \(M\) is an ellipsoid
	containing the origin.}

<!-- theorem-1: body_md -->
\label{thm:main}
		For every integer \(n\geq2\) and every \(1<p<n\), the geometric
		inequality~\eqref{1.6} and the analytic inequality~\eqref{1.7} both
		fail. More precisely:
		\begin{enumerate}
			\renewcommand{\textup{(\roman{enumi})}}{\textup{(\roman{enumi})}}
			
			\item there exists an origin-symmetric convex body
			\(K\in\mathcal{K}_o^n\) with \(C^\infty\) boundary and everywhere
			positive Gauss curvature such that
			\[
			\Phi_p(K)>\left(\frac{p-1}{n-p}\right)^{p-1}C_p(K);
			\]
			
			\item there exists a nonnegative function
			\(f\in C_c^\infty(\mathbb{R}^n)\) such that
			\[
			\int_0^\infty
			\Phi_p\bigl(\langle f\rangle_t\bigr)\,dt
			>
			\left(\frac{p-1}{n-p}\right)^{p-1}
			\int_0^\infty
			C_p\bigl(\langle f\rangle_t\bigr)\,dt.
			\]
		\end{enumerate}

<!-- lemma-1: body_md -->
\label{lem:affine-deficit}
	Let \(\ell=2j\geq2\) be a fixed even integer. Define
		\begin{equation}\label{eq:cosine-eigenvalue}
			a_{p,2j}
			=
			\prod_{r=0}^{j-1}
			\frac{p-2r}{n+p+2r},
		\ \
			A_\ell
			=
			(k-\lambda_\ell)a_{p,\ell}.
		\end{equation}
	Then as \(\varepsilon\to0\),
		\begin{equation}\label{eq:affine-deficit}
			S_p(K_\varepsilon)-\Phi_p(K_\varepsilon)
			=
			\frac{n+p}{2p}A_\ell^2\varepsilon^2
			\int_{\mathbb{S}^{n-1}}Y_\ell^2\,d\sigma
			+o(\varepsilon^2).
		\end{equation}

<!-- proof-1: body_md -->
Take
			\[
			H
			=
			\nabla_{\mathbb{S}^{n-1}}^2Y_\ell+Y_\ell I,
			\ \
			\tau
			=
			\operatorname{tr}H
			=
			(n-1-\lambda_\ell)Y_\ell.
			\]
			Since \(Y_\ell\) and \(H\) are bounded on
			\(\mathbb{S}^{n-1}\), there exists \(\varepsilon_0>0\) such that whenever \(\lvert\varepsilon\rvert<\varepsilon_0\),
			\[
			h_\varepsilon\geq\frac12,
			\ \
			I+\varepsilon H\geq \frac12 I
			\quad\text{on }\mathbb{S}^{n-1}.
			\]
	It follows that all expansions below are uniform for
		\(u\in\mathbb{S}^{n-1}\). Using the Gauss parametrization, the
		\(L_p\) surface area measure of \(K_\varepsilon\) can be written as
		\[
		dS_p(K_\varepsilon,u)
		=
		q_\varepsilon(u)\,d\sigma(u),
		\ \ 
		q_\varepsilon(u)
		:=
		h_\varepsilon(u)^{1-p}
		\det\bigl(I+\varepsilon H(u)\bigr).
		\]
Using the expansions
		\begin{align*}
			h_\varepsilon^{1-p}
			&=1+(1-p)\varepsilon Y_\ell
			+\frac{p(p-1)}2\varepsilon^2Y_\ell^2
			+O_{C^0}(\varepsilon^3),
		\end{align*}
		and
	 \begin{align*}
			\det(I+\varepsilon H)
			&=1+\varepsilon\tau
			+\frac{\varepsilon^2}{2}
			\bigl(\tau^2-\operatorname{tr}(H^2)\bigr)
			+O_{C^0}(\varepsilon^3),
		\end{align*}
		we obtain
		\begin{equation}\label{eq:surface-density-first}
			q_\varepsilon
			=1+\varepsilon(k-\lambda_\ell)Y_\ell
			+\varepsilon^2R_0+O_{C^0}(\varepsilon^3),
		\end{equation}
		where
		\[
		R_0=
		\frac12\bigl(\tau^2-\operatorname{tr}(H^2)\bigr)
		+(1-p)Y_\ell\tau+\frac{p(p-1)}2Y_\ell^2.
		\]
		
		The Funk-Hecke formula \cite[Chapter~3]{Groemer}
		(or a direct Fourier computation when \(n=2\)) shows that
		\(Y_{2j}\) is an eigenfunction of \(\mathcal{T}_p\) with eigenvalue
		\(a_{p,2j}\). That is,
		\begin{equation*}\label{eq:Funk-Hecke}
			\mathcal{T}_pY_{2j}
			=
			a_{p,2j}Y_{2j}.
		\end{equation*}
		For \(j\geq0\), a direct computation of the corresponding beta integrals gives
		\[
		a_{p,0}=1,
	\ \
		a_{p,2j+2}
		=
		\frac{p-2j}{n+p+2j}\,a_{p,2j}.
		\]
		Repeated application of this recurrence yields the product formula
		in \eqref{eq:cosine-eigenvalue}. In particular,
		\[
		a_{p,2}=\frac{p}{n+p}.
		\]
		Applying \(\mathcal{T}_p\) to the expansion in
		\eqref{eq:surface-density-first}, and using the boundedness of
		\(
		\mathcal{T}_p:
		C^0(\mathbb S^{n-1})\longrightarrow C^0(\mathbb S^{n-1}),
		\)
		we obtain
		\[
		v_{K_\varepsilon}
		=
		1+\varepsilon A_\ell Y_\ell
		+\varepsilon^2B_0
		+O_{C^0}(\varepsilon^3),
		\ \
		B_0:=\mathcal{T}_pR_0.
		\]
		Since \(\mathcal{T}_p\) preserves normalized spherical averages, we have
		\[
		\left\langle B_0\right\rangle_{\!\sigma}=\left\langle R_0\right\rangle_{\!\sigma}.
		\]
		Moreover, \(\left\langle Y_\ell\right\rangle_{\!\sigma}=0\). Therefore, the formula for \(S_p\) in
		\eqref{eq:normalized-functionals} gives
		\[
		\frac{S_p(K_\varepsilon)}{M}
		=
		1+\varepsilon^2\left\langle B_0\right\rangle_{\!\sigma}
		+O(\varepsilon^3).
		\]
		Let $\alpha=n/p$.
		Taylor expanding \(v_{K_\varepsilon}^{-\alpha}\), and applying the
		power \(-1/\alpha\) to its spherical average as in
		\eqref{eq:normalized-functionals}, we obtain
		\begin{align*}
			\left\langle v_{K_\varepsilon}^{-\alpha}\right\rangle_{\!\sigma}
			&=
			1-\alpha\varepsilon^2\left\langle B_0\right\rangle_{\!\sigma}
			+\frac{\alpha(\alpha+1)}{2}
			A_\ell^2\varepsilon^2\left\langle Y_\ell^2\right\rangle_{\!\sigma}
			+O(\varepsilon^3),
		\end{align*}
		and
		\begin{align*}
			\frac{\Phi_p(K_\varepsilon)}{M}
			&=
			1+\varepsilon^2
			\left(
			\left\langle B_0\right\rangle_{\!\sigma}
			-\frac{\alpha+1}{2}
			A_\ell^2\left\langle Y_\ell^2\right\rangle_{\!\sigma}
			\right)
			+O(\varepsilon^3).
		\end{align*}
		Combining the second expansion with the preceding expansion of
		\(S_p(K_\varepsilon)/M\), and using
		\[
		\alpha+1=\frac{n+p}{p} \ \
		\text{and} \ \
		M\left\langle Y_\ell^2\right\rangle_{\!\sigma}
		=
		\int_{\mathbb S^{n-1}}Y_\ell^2\,d\sigma,
		\]
		we give \eqref{eq:affine-deficit}, in fact with an $O(\varepsilon^3)$ remainder
		for fixed $\ell$.

\label{lem:capacity-deficit}
		Let $s_\ell>\mu$ be defined by
		\begin{equation}\label{eq:s-ell}
			(p-1)s_\ell(s_\ell-\mu)=\lambda_\ell,
			\ \
			s_\ell=
			\frac{\mu+\sqrt{\mu^2+4\lambda_\ell/(p-1)}}2.
		\end{equation}
		Then
		\begin{equation}\label{eq:capacity-deficit}
			S_p(K_\varepsilon)-\gamma_{n,p}\,C_p(K_\varepsilon)
			\geq
			\frac{p(p-1)}2(s_\ell-\mu)^2\varepsilon^2
			\int_{\mathbb S^{n-1}}Y_\ell^2\,d\sigma
			+o(\varepsilon^2).
		\end{equation}

<!-- proof-2: body_md -->
For $\theta\in\mathbb S^{n-1}$ and $t\geq1,$ we parametrize the exterior of \(K_\varepsilon\) by
		\[
		x=\rho_\varepsilon(\theta)t\theta.
		\]
		Under this parametrization, write
		\[
		v\bigl(\rho_\varepsilon(\theta)t\theta\bigr)
		=
		F(t,\theta).
		\]
		The corresponding Jacobian is
		\[
		dx
		=
		\rho_\varepsilon(\theta)^n t^{n-1}
		\,dt\,d\sigma(\theta),
		\]
		while the chain rule gives
		\[
		\ \ \nabla v
		=
		\rho_\varepsilon(\theta)^{-1}
		\left[
		F_t e_r
		+\frac{1}{t}
		\left(
		\nabla_{\mathbb S}F
		-tF_tb_\varepsilon
		\right)
		\right],
		\ \ e_r=\theta.
		\]
		Thus a direct polar-coordinate calculation gives
		\begin{align}
			E_\varepsilon(F)
			&:=
			\int_{\mathbb{R}^n\setminus K_\varepsilon}|\nabla v|^p\,dx \nonumber\\
			&=
			\int_{\mathbb S^{n-1}}\int_1^\infty
			\rho_\varepsilon^k t^{n-1}
			\left[
			F_t^2+t^{-2}
			\left|\nabla_{\mathbb S}F-tF_tb_\varepsilon\right|^2
			\right]^{p/2}
			dt\,d\sigma.
			\label{eq:exact-energy}
		\end{align}
		Consider the function
		\[
		F_T(t,\theta):=t^{-\mu},
		\ \ t\geq1,
		\]
		extended by the constant \(1\) on \(K_\varepsilon\). It
		satisfies
		\begin{equation}\label{eq:gauge-energy}
			E_\varepsilon(F_T)
			=
			\mu^{p-1}S_p(K_\varepsilon).
		\end{equation}
		To verify this, observe that on \(\partial K_\varepsilon\),
		\[
		\left|
		\nabla r_{K_\varepsilon}(x)
		\right|
		=
		h_{K_\varepsilon}
		\bigl(\nu_{K_\varepsilon}(x)\bigr)^{-1}.
		\]
		Therefore, the radial representation of
		\(\partial K_\varepsilon\) yields
		\[
		S_p(K_\varepsilon)
		=
		\int_{\mathbb S^{n-1}}
		\rho_\varepsilon^k
		\left(1+|b_\varepsilon|^2\right)^{p/2}
		\,d\sigma.
		\]
		Substituting \(F_T\) into \eqref{eq:exact-energy} and evaluating the
		radial integral,
		\[
		\mu^p
		\int_1^\infty
		t^{n-1-p(\mu+1)}\,dt
		=
		\mu^{p-1},
		\]
		we obtain \eqref{eq:gauge-energy}.
		
	We refine the preceding function by defining
	\begin{equation*}\label{eq:improved-competitor}
		F_*(t,\theta)
		=
		t^{-\mu}
		+\varepsilon\mu Y_\ell(\theta)
		\bigl(t^{-s_\ell}-t^{-\mu}\bigr).
	\end{equation*}
	Since \(F_*(1,\theta)=1\), we extend \(F_*\) to \(K_\varepsilon\)
	by defining
	\(
	F_*\equiv1
	\text{ on }K_\varepsilon.
	\)
The extended function has trace \(1\) on
\(\partial K_\varepsilon\) and is an admissible finite-energy test
function in the corresponding homogeneous Sobolev space. To approximate \(F_*\) by admissible functions in the capacity
definition \eqref{2.9}, fix \(\delta>0\) and modify \(F_*\) near
the boundary by setting it equal to \(1\) for
\(1\leq t\leq1+\delta\). In the transition region
\(1+\delta<t<1+2\delta\), interpolate with respect to \(t\) so that
the modified function equals \(1\) at \(t=1+\delta\) and
\(F_*(1+2\delta,\theta)\) at \(t=1+2\delta\). 
Since
\[
F_*(t,\theta)-1=O(t-1),
\ \ \text{as }t\downarrow1,
\]
uniformly in \(\theta\), the interpolation can be chosen with bounded
gradient. Its contribution to the energy is therefore \(O(\delta)\).

Next, choose a smooth radial cutoff that equals \(1\) on
\(\{|x|\leq R\}\) and vanishes on \(\{|x|\geq2R\}\). The energy of the
discarded tail and the additional term arising from the gradient of
the cutoff are both bounded by
\[
O\bigl(R^{\,n-p(\mu+1)}\bigr)
=
O(R^{-\mu})
=
o(1),
\ \ \text{as }R\to\infty.
\]
The truncated Sobolev function is compactly supported and remains
identically equal to \(1\) on a neighborhood of \(K_\varepsilon\).
Mollifying it at a scale smaller than the width of this neighborhood
produces a function in \(C_c^\infty(\mathbb{R}^n)\) that is still equal to
\(1\) on \(K_\varepsilon\), while its energy converges to that of the
truncated function. Let the mollification scale tend to zero. Then \(R\to\infty\), and
finally \(\delta\to0\). Thus we obtain
\begin{equation}\label{eq:capacity-upper-competitor}
	C_p(K_\varepsilon)
	\leq
	E_\varepsilon(F_*).
\end{equation}
This approximation argument is the standard passage between the
homogeneous Sobolev and smooth definitions of capacity; see
\cite[Chapter~2]{Mazya}.
		
We next compute the second-order change in energy. For \(a>0\), let
\[
u(t)=t^{-\mu},
\ \
w_a(t,\theta)=\mu t^{-a}Y_\ell(\theta),
\ \
T_a=t^{-1}\nabla_{\mathbb S}w_a.
\]
Define
\[
Z=w_{s_\ell}-w_\mu.
\]
Then the improved function can be written as
\[
F_*=u+\varepsilon Z.
\]		
Let \(c\neq0\), and decompose
\[
z=z_re_r+z_{\mathrm{tan}}
\ \ \text{and} \ \
z_{\mathrm{tan}}\perp e_r,
\]
with \(|z|\leq C|c|\). After factoring out \(|c|\), the normalized
argument remains in a fixed compact set away from the origin for all
sufficiently small \(|\varepsilon|\). Taylor's theorem then yields the following uniform expansion:
\begin{align}\label{eq:vector-taylor}
	|ce_r+\varepsilon z|^p
	={}&
	|c|^p
	+p|c|^{p-2}cz_r\varepsilon
	\nonumber\\
	&+
	\frac{p}{2}|c|^{p-2}
	\left(
	(p-1)z_r^2+|z_{\mathrm{tan}}|^2
	\right)\varepsilon^2
	+O\bigl(\varepsilon^3|c|^p\bigr).
\end{align}		
To isolate the second-order angular terms, define
\[
\widetilde b_\varepsilon
=
b_\varepsilon
-\varepsilon\nabla_{\mathbb S}Y_\ell
=
O_{C^1}(\varepsilon^2).
\]
The gradient vectors appearing in the integrand of
\eqref{eq:exact-energy} can then be written as
\begin{align*}
	\mathcal{A}_T
	&=
	\left(
	u',
	\varepsilon T_\mu-u'\widetilde b_\varepsilon
	\right),\\
	\mathcal{A}_*
	&=
	\left(
	u'+\varepsilon Z_t,
	\varepsilon T_{s_\ell}
	-\varepsilon^2Z_t\nabla_{\mathbb S}Y_\ell
	-u'\widetilde b_\varepsilon
	+O(\varepsilon^3|u'|)
	\right).
\end{align*}
Thus, relative to the radial base vector \((u',0)\), their respective
first-order perturbations are
\[
(0,T_\mu)
\ \ \text{and} \ \
(Z_t,T_{s_\ell}).
\]
Moreover, the radial factor satisfies
\[
\rho_\varepsilon^k
=
1+k\varepsilon Y_\ell
+O(\varepsilon^2).
\]
The order-\(\varepsilon\) term in
\(E_\varepsilon(F_T)-E_\varepsilon(F_*)\) vanishes, since for each
fixed \(t\), \(Z_t(t,\cdot)\) is a scalar multiple of \(Y_\ell\),
whose spherical mean is zero.

Let
\[
W(t)=t^{n-1}\lvert u'(t)\rvert^{p-2}.
\]
Applying the vector expansion \eqref{eq:vector-taylor} to the exact
energy formula \eqref{eq:exact-energy}, we obtain the following
unreduced second-order expansion:
\begin{align}
	E_\varepsilon(F_T)-E_\varepsilon(F_*)
	={}&
	\frac{p}{2}\varepsilon^2
	\int_{\mathbb S^{n-1}}\int_1^\infty
	W(t)
	\left(
	\lvert T_\mu\rvert^2
	-\lvert T_{s_\ell}\rvert^2
	-(p-1)Z_t^2
	\right)
	\,dt\,d\sigma
	\nonumber\\
	&\quad
	-pk\varepsilon^2
	\int_{\mathbb S^{n-1}}\int_1^\infty
	W(t)u'Y_\ell Z_t
	\,dt\,d\sigma
	+o(\varepsilon^2).
	\label{eq:raw-energy-difference}
\end{align}
The second integral arises precisely from the interaction between the
first-order term \(k\varepsilon Y_\ell\) in the expansion of
\(\rho_\varepsilon^k\) and the first-order radial variation of the
energy density \(\lvert\nabla F_*\rvert^p\). The terms involving the
second-order contribution to \(\rho_\varepsilon^k\) are identical for
the two energies and therefore cancel. Moreover, the second-order
correction vectors
\[
-u'\widetilde b_\varepsilon
\ \ \text{and}\ \
-\varepsilon^2Z_t\nabla_{\mathbb S}Y_\ell
\]
are both tangential and hence orthogonal to the radial base gradient.
Their linear pairings with the base gradient therefore vanish. Hence, they
produce no additional contribution at order \(\varepsilon^2\).
		

		
		
		Define 
		\begin{align}
			Q(w)
			=
			\int_{\mathbb S^{n-1}}\int_1^\infty
			&\mu^{p-2}
			t^{n-1-(\mu+1)(p-2)}
			\nonumber\\[-2mm]
			&\times
			\left[(p-1)w_t^2+t^{-2}|\nabla_{\mathbb S}w|^2\right]
			dt\,d\sigma.
			\label{eq:Q}
		\end{align}
		Since the weight in \eqref{eq:Q} is $W$, the difference between the
		coefficient in \eqref{eq:raw-energy-difference} and
		$\frac p2[Q(w_\mu)-Q(w_{s_\ell})]$ equals
		\[
		p(p-1)\int_{\mathbb S^{n-1}}\int_1^\infty
		W(w_\mu)_tZ_t\,dt\,d\sigma
		-pk\int_{\mathbb S^{n-1}}\int_1^\infty
		Wu'Y_\ell Z_t\,dt\,d\sigma.
		\]
		This vanishes pointwise in the remaining factors since
		\[
		(w_\mu)_t=\mu Y_\ell u',
		\ \ k=(p-1)\mu.
		\]
		Therefore, we have proved
		\begin{equation}\label{eq:energy-difference-Q}
			E_\varepsilon(F_T)-E_\varepsilon(F_*)
			=
			\frac p2\varepsilon^2
			\bigl[Q(w_\mu)-Q(w_{s_\ell})\bigr]
			+o(\varepsilon^2).
		\end{equation}
		
		The same calculation, with the elementary $t$-integrals performed, displays
		the complete second-order coefficient as
		\begin{align*}
			E_\varepsilon(F_T)-E_\varepsilon(F_*)
			={}&
			\frac p2\mu^p\varepsilon^2
			\int_{\mathbb S^{n-1}}Y_\ell^2\,d\sigma
			\nonumber\\
			&\times
			\left\{
			\lambda_\ell
			\left(\frac1\mu-\frac1{2s_\ell-\mu}\right)
			-(p-1)
			\left(\frac{s_\ell^2}{2s_\ell-\mu}-\mu\right)
			\right\}
			+o(\varepsilon^2).
			\label{eq:raw-energy-coefficient}
		\end{align*}
		Since \(t^{\mu-s_\ell}\leq1\) for \(t\geq1\), the ratios of all
		perturbation fields to \(\lvert u'\rvert\) are uniformly bounded.
		Thus, for fixed \(\ell\) and sufficiently small
		\(\lvert\varepsilon\rvert\), every line segment joining the base
		gradient to the corresponding perturbed gradient remains at least
		\(\lvert u'\rvert/2\) away from the origin.
		
		The third derivative of $z\mapsto|z|^p$ on this region
		is bounded by $C|u'|^{p-3}$. Hence, the pointwise remainder is at most
		$C|\varepsilon|^3|u'|^p$.  Thus, we have
		\[
		\int_1^\infty t^{n-1}|u'|^p\,dt<\infty.
		\]
		The dominated convergence theorem shows that the remainder in
		\eqref{eq:raw-energy-difference} is \(o(\varepsilon^2)\). The same
		argument remains valid when \(1<p<2\).
		
		Using
		\[
		\int_{\mathbb S^{n-1}}
		\lvert\nabla_{\mathbb S}Y_\ell\rvert^2\,d\sigma
		=
		\lambda_\ell
		\int_{\mathbb S^{n-1}}Y_\ell^2\,d\sigma,
		\]
		and evaluating the radial integral in \eqref{eq:Q}, we obtain
		\begin{equation*}\label{eq:Q-mode}
			Q(w_\tau)
			=
			\mu^p
			\frac{(p-1)\tau^2+\lambda_\ell}{2\tau-\mu}
			\int_{\mathbb S^{n-1}}Y_\ell^2\,d\sigma.
		\end{equation*}
		Using \eqref{eq:s-ell}, it follows that
		\begin{equation}\label{eq:Q-gap}
			Q(w_\mu)-Q(w_{s_\ell})
			=
			\mu^{p-1}(p-1)(s_\ell-\mu)^2
			\int_{\mathbb S^{n-1}}Y_\ell^2\,d\sigma.
		\end{equation}
		Combining \eqref{eq:capacity-upper-competitor} with
		\(\gamma_{n,p}=\mu^{1-p}\), we obtain
		\[
		S_p(K_\varepsilon)
		-\gamma_{n,p}C_p(K_\varepsilon)
		\geq
		\mu^{1-p}
		\left[
		E_\varepsilon(F_T)-E_\varepsilon(F_*)

\label{prop:symmetric-counterexample}
		For every $n\geq2$ and $1<p<n$, there is a sufficiently high even degree
		$\ell$ such that for sufficiently small \(\varepsilon\neq0\),

			\Phi_p(K_\varepsilon)>
			\gamma_{n,p}\,C_p(K_\varepsilon).

<!-- end proposition-1 -->
<!-- proof-3: body_md -->
Combining Lemmas~\ref{lem:affine-deficit} and
		\ref{lem:capacity-deficit}, we obtain
		\begin{align}
			\Phi_p(K_\varepsilon)
			-\gamma_{n,p}C_p(K_\varepsilon)
			\geq{}&
			\frac{\varepsilon^2}{2}
			\left[
			p(p-1)(s_\ell-\mu)^2
			-\frac{n+p}{p}A_\ell^2
			\right]
			\nonumber\\
			&\times
			\int_{\mathbb S^{n-1}}Y_\ell^2\,d\sigma
			+o(\varepsilon^2).
			\label{eq:master-gap}
		\end{align}
		It therefore suffices to choose \(\ell\) so that the coefficient in
		brackets is positive.
		
		If \(p=2q\) is an even integer, then
		\eqref{eq:cosine-eigenvalue} immediately yields
		\[
		a_{p,2j}=0
		\ \ \text{for every }j>q.
		\]
		If \(p\) is not an even integer, the product formula can instead be
		expressed in terms of Gamma functions as
		\begin{equation*}\label{eq:gamma-eigenvalue}
			a_{p,2j}
			=
			(-1)^j
			\frac{
				\Gamma(j-p/2)\Gamma((n+p)/2)
			}{
				\Gamma(-p/2)\Gamma(j+(n+p)/2)
			}.
		\end{equation*}
		The standard asymptotic formula for ratios of Gamma functions yields
		\[
		\lvert a_{p,2j}\rvert
		=
		O\bigl(j^{-p-n/2}\bigr),
		\ \
		A_{2j}
		=
		O\bigl(j^{2-p-n/2}\bigr)
		=
		o(1),
		\]
		where the last relation follows from \(n+2p>4\). By contrast, the definition of \(s_{2j}\) gives
		\[
		s_{2j}-\mu
		=
		\frac{
			\sqrt{\mu^2+4\lambda_{2j}/(p-1)}-\mu
		}{2}
		\sim
		\frac{2j}{\sqrt{p-1}}
		\ \ \text{as }j\to\infty.
		\]
		Therefore, when \(p\) is not an even integer, the expression in
		brackets in \eqref{eq:master-gap} satisfies
		\[
		p(p-1)(s_{2j}-\mu)^2
		-\frac{n+p}{p}A_{2j}^2
		=
		4pj^2\bigl(1+o(1)\bigr).
		\]
		When \(p=2q\) is an even integer, \(A_{2j}=0\) for \(j>q\), while the
		first term in the bracket is strictly positive. Hence, in either case,
		the bracket is positive for every sufficiently large \(j\). Fix such
		a \(j\), and let \(\ell=2j\). Since the remainder in
		\eqref{eq:master-gap} is \(o(\varepsilon^2)\) for this fixed degree,
		the desired strict inequality holds for every sufficiently small
		\(\varepsilon\neq0\).

\label{prop:analytic-counterexample}
		Let \(K\in\mathcal{K}_o^n\) be a \(C^\infty\), origin-symmetric convex body
		satisfying
		\[
		\Phi_p(K)
		>
		\gamma_{n,p}\,C_p(K).
		\]
		Define
		\begin{equation}\label{eq:analytic-test}
			f(x):=g\bigl(r_K(x)\bigr).
		\end{equation}
		Then \(f\) is a nonnegative function in \(C_0^\infty(\mathbb{R}^n)\) and
		satisfies
		\begin{equation}\label{eq:analytic-reverse}
			\int_0^\infty
			\Phi_p\bigl(\langle f\rangle_{t}\bigr)\,dt
			>
			\gamma_{n,p}
			\int_0^\infty
			C_p\bigl(\langle f\rangle_{t}\bigr)\,dt.
		\end{equation}
<!-- end proposition-2 -->
<!-- proof-5: body_md -->
The function \(r_K\) is smooth on
		\(\mathbb{R}^n\setminus\{0\}\). Since \(g\equiv1\) on
		\((-\infty,1]\) and vanishes to infinite order at \(s=2\), the function \(f=g\circ r_K\) is smooth at the origin and matches
		smoothly with the zero function along the boundary of its support. Hence, the function $f$ defined by \eqref{eq:analytic-test} belongs to $C_0^\infty(\mathbb{R}^n)$.
		
		Fix \(t\in(0,1)\). Since \(g\) is strictly decreasing on \((1,2)\),
		there exists a unique \(s=s(t)\in(1,2)\) such that \(t=g(s)\).
		Applying the computation from the proof of
		\cite[Lemma~8]{LXYZ} to this regular level yields
		\begin{equation}\label{eq:level-dilate}
			\langle f\rangle_{t}
			=
			c(t)K,
			\ \
			c(t)^{n-p}
			=
			\lvert g'(s)\rvert^{p-1}s^{n-1}.
		\end{equation}
		Indeed, the level surface \(\{f=t\}\) is \(s\partial K\), and on this
		surface,
		\[
		\lvert\nabla f\rvert
		=
		\lvert g'(s)\rvert\,\lvert\nabla r_K\rvert.
		\]
		Moreover, the \((n-1)\)-dimensional surface measure acquires the factor
		\(s^{n-1}\) under dilation. Thus, the Gauss map \(\nu_K\) satisfies
		\[
		(\nu_K)_*\!\left(
		\lvert\nabla r_K\rvert^{p-1}
		\,\mathcal{H}^{n-1}\!\restriction_{\partial K}
		\right)
		=
		S_p(K,\cdot).
		\]
		Therefore, the measure appearing in
		\eqref{eq:convexification-definition} is
		\[
		\lvert g'(s)\rvert^{p-1}s^{n-1}\,dS_p(K,\cdot).
		\]
		Since
		\[
		dS_p(cK,\cdot)=c^{n-p}\,dS_p(K,\cdot),
		\]
		the choice of \(c=c(t)\) in \eqref{eq:level-dilate} gives precisely
		this measure. Uniqueness for the even \(L_p\)-Minkowski problem
		implies
		\[
		\langle f\rangle_{t}=c(t)K.
		\]
		
		
		Both \(\Phi_p\) and \(C_p\) are homogeneous of degree \(n-p\).
		Hence, for either
		\(
		\Psi=\Phi_p
		\ \text{or}\
		\Psi=C_p,
		\)
		equation \eqref{eq:level-dilate} and the substitution \(t=g(s)\)
		yield
		\begin{align}
			\int_0^\infty
			\Psi\bigl(\langle f\rangle_{t}\bigr)\,dt
			=
			A_{g,p}\,\Psi(K),
			 \ \ \text{with} \ \ 
			A_{g,p}
			=
			\int_1^2
			\lvert g'(s)\rvert^p s^{n-1}\,ds
			>0.\label{eq:common-factor}
		\end{align}
		Since \(0\leq f\leq1\), the relevant level set is empty for \(t>1\),
		and the endpoint levels have measure zero.
		Therefore, because \(A_{g,p}>0\), combining
		\eqref{eq:common-factor} with
		\[
		\Phi_p(K)>\gamma_{n,p}\,C_p(K)
		\]
		proves \eqref{eq:analytic-reverse}.

\label{prop:explicit}
		For an even integer $q\geq2$,

			\Phi_2(B_q^3)=S_2(B_q^3)
			=
			\frac{24\pi(q-1)}
			{q(q+1)\sin(\pi/q)}.

		In particular,
		\[
		\Phi_2(B_{24}^3)>C_2(B_{24}^3).
		\]
<!-- end proposition-3 -->
<!-- proof-7: body_md -->
For a smooth convex body \(K\in\mathcal K_o^3\), define the symmetric
		matrix
		\[
		M_K
		=
		\int_{\partial K}
		\frac{\nu_K(x)\otimes\nu_K(x)}
		{x\cdot\nu_K(x)}
		\,d\mathcal{H}^2(x).
		\]
		Note that for
		\(a,b\in\mathbb{R}^3\), the tensor product \(a\otimes b\) is the matrix defined
		by
		\[
		(a\otimes b)_{ij}:=a_i b_j,
		\ \ 1\leq i,j\leq3.
		\]
		Thus,
		\[
		(M_K)_{ij}
		=
		\int_{\partial K}
		\frac{\nu_{K,i}(x)\nu_{K,j}(x)}
		{x\cdot\nu_K(x)}
		\,d\mathcal{H}^2(x).
		\]
		Since \(K\in\mathcal K_o^3\), we have
		\(x\cdot\nu_K(x)>0\) on \(\partial K\). Hence, the matrix \(M_K\) is
		well defined and positive semidefinite.
	Equation \eqref{eq:Vp} yields
	\[
	V_2(K,\bar u)
	=
	\frac{1}{12}\,u^{\mathsf T}M_Ku.
	\]
	The body \(B_q^3\) is invariant under arbitrary sign changes and
	permutations of its coordinates. These symmetries imply that
	\[
	M_{B_q^3}=m_q I
	\]
	for some \(m_q>0\). Consequently,
	\[
	V_2(B_q^3,\bar u)=\frac{m_q}{12}
	\]
	is independent of \(u\in\mathbb S^2\). The normalization formulas
	\eqref{eq:Sp-average} and \eqref{eq:Phi} thus give
	\[
	\Phi_2(B_q^3)=S_2(B_q^3).
	\]
		
	Define the vector field
	\[
	X(x)
	=
	\bigl(x_1^{q-1},x_2^{q-1},x_3^{q-1}\bigr).
	\]
	For \(x\in\partial B_q^3\), we have
	\[
	\nu(x)=\frac{X(x)}{\lvert X(x)\rvert},
	\ \ 
	x\cdot X(x)=1.
	\]
	It follows that
	\[
	\frac{1}{x\cdot\nu(x)}
	=
	\lvert X(x)\rvert
	=
	X(x)\cdot\nu(x).
	\]
	Therefore, by the divergence theorem and coordinate symmetry,
	\begin{align}
		S_2(B_q^3)
		&=
		\int_{\partial B_q^3}
		X\cdot\nu\,d\mathcal{H}^2
		=
		\int_{B_q^3}\operatorname{div}X\,dx
		\nonumber\\
		&=
		3(q-1)
		\int_{B_q^3}
		\lvert x_1\rvert^{q-2}\,dx.
		\label{eq:divergence}
	\end{align}
	To compute the remaining integral, decompose \(B_q^3\) into its eight
	orthants and introduce the variables
	\[
	y_i=\lvert x_i\rvert^q,
	\ \ i=1,2,3.
	\]
	The Dirichlet simplex integral then gives
	\[
	\int_{B_q^3}
	\lvert x_1\rvert^{q-2}\,dx
	=
	\frac{8}{q^3}
	\frac{
		\Gamma(1-1/q)\Gamma(1/q)^2
	}{
		\Gamma(2+1/q)
	}.
	\]	Inserting this expression into \eqref{eq:divergence} and applying
	Euler's reflection formula
	\[
	\Gamma(z)\Gamma(1-z)
	=
	\frac{\pi}{\sin(\pi z)}
	\]
	yields \eqref{eq:exact-S2}.
		
	For \(q=24\), it follows that
	\[
	\Phi_2(B_{24}^3)
	=
	\frac{23\pi}{25\sin(\pi/24)}
	>
	\frac{552}{25}
	>
	22,
	\]
	where the first strict inequality follows from
	\(\sin x<x\) for \(x>0\). On the other hand,
	\[
	B_{24}^3
	\subset
	[-1,1]^3
	\subset
	\sqrt{3}\,B^3.
	\]
	By monotonicity and homogeneity of the \(2\)-capacity, together with
	\(C_2(B^3)=4\pi\), we obtain
	\[
	C_2(B_{24}^3)
	\leq
	C_2(\sqrt{3}\,B^3)
	=
	4\pi\sqrt{3}
	<
	22.
	\]
	The final strict inequality follows from
	\[
	\pi<\frac{22}{7},
	\ \
	\sqrt{3}<\frac{7}{4}.
	\]
