<!-- proposition-1: body_md -->
\label{thm1}
    Consider the Motsch--Tadmor model \eqref{eq:MTmodel} with initial data \(\{(X_i^N(0), V_i^N(0))\}_{i=1}^N\) and assume that the influence function \(\varphi\) satisfies the assumption in \eqref{in:lowupbound}. Introducing the quantity
    \begin{equation}
    \label{eq:weightedN}
    \langle \boldsymbol{V}^N(t)\rangle_{N}:= \sum_{i=1}^N \pi_i^N(t) V_i^N(t),
    \end{equation}
    where \(\{\pi_i^N(t)\}_{i=1}^N\) is given in \eqref{def:eqmeas}, there exists \(\mathbf{V}^N_\infty\in \mathbb{R}^d\), characterized in \eqref{eq:vinf}, such that \( \langle \boldsymbol{V}^N(t)\rangle _{N} \to \mathbf{V}^N_\infty\) as \(t \uparrow \infty\). Moreover, there exists \(C>0\) such that
    \[
    \max_{i\in \{1,\dots,N\}}\left|V_i^N(t)-\mathbf{V}^N_\infty\right|\le C \, \mathrm e^{-\eta t}, \quad \eta:= \frac mM.
    \]
    Similarly, consider a regular solution \(f_t(x,v)\) of the Motsch--Tadmor model \eqref{eq:mesoVlasov} with compactly supported initial datum \(f_{\mathrm{in}} \in C^1_{\mathrm c}(\mathbb{R}^d \times \mathbb{R}^d)\)
    and introduce the quantity
    \begin{equation}
    \label{eq:weightedf}
    \langle v(t)\rangle_{f}:= \displaystyle \int_{\mathbb{R}^d\times \mathbb{R}^d}\pi^\rho(t,x) v\,  \mathrm df_t(x,v),
    \end{equation}
    where \(\pi^\rho(t,x)\) is given in \eqref{def:eqmeas}. Then, there exists \(\mathbf{v}^f_\infty\in \mathbb{R}^d\), characterized in \eqref{eq:vinff}, such that \( \langle v(t)\rangle _{f} \to \mathbf{v}^f_\infty\) as \(t \uparrow \infty\). Moreover, there exists \(C>0\), such that
    \[
    \left|v-\mathbf{v}^f_\infty\right|\le C \, \mathrm e^{-\eta t}, \quad \eta:= \frac mM,
    \]
    for any \(v \in \text{supp}f_t\).

<!-- theorem-1: body_md -->
\label{thm2}
Assume \eqref{in:lowupbound}, and suppose \(f_{\mathrm{in}}\in C_{\mathrm c}^{1}(\mathbb{R}^{d}\times \mathbb{R}^d)\). 
Let \(f_t\) be the regular solution of the kinetic Motsch--Tadmor model \eqref{eq:mesoVlasov} with initial datum \(f_{\mathrm{in}}\) and let \(F_{t}^N\) be the Liouville solution to \eqref{linearLiouville} with initial datum \(F^N_{\mathrm{in}}=f_{\mathrm{in}}^{\otimes N}\).

Then there exists a constant \(C>0\), depending only on \(d\), \(m\), \(M\), \(\|\varphi\|_{\mathrm{Lip}}\), and the diameter of the velocity support of \(f_{\mathrm{in}}\), such that
\begin{equation}
\label{in:propchaos}
\sup_{t\geq0}\,
\mathcal{W}_2\left(
\widetilde F_{k,t}^N,
\widetilde f_t^{\otimes k}
\right)
\leq
C\sqrt{\frac{k}{N}},
\end{equation}
where
\(\widetilde f_t:= (\mathcal{G}_t)_\# f_t\)
and, for \(1\leq k\leq N\), 
\(\widetilde F_{k,t}^N\) is the \(k\)-particle marginal of 
\((\mathcal{G}_t^N)_{\#}F^N_{t}\) where
$
\mathcal G_t(x,v)
:=
\left(
x-t\mathbf v^f_\infty,\,
v-\mathbf v^f_\infty
\right)
$ and
\begin{equation}
    \label{eq:movingframe}
    \mathcal{G}^N_{t}(X_N,V_N):=(x_1-t \mathbf V_\infty^N,\dots, x_N -t \mathbf V_\infty^N, v_1- \mathbf V^N_\infty, \dots, v_N-\mathbf V_\infty^N).
\end{equation}

<!-- corollary-1: body_md -->
\label{cor1}
Assume \eqref{in:lowupbound}, and let \(f_{\mathrm in}\in \mathcal C^1_{\mathrm c}(\mathbb{R}^d\times\mathbb{R}^d)\) be compactly supported. 
Let \(f_t\) be the solution to \eqref{eq:mesoVlasov}, and let
\(
\widetilde f_t=(\mathcal G_t)_\# f_t.
\)
Then there exist \(\rho_\infty\in\mathcal P(\mathbb{R}^d)\) and a constant \(C>0\) such that
\begin{equation}
\label{eq:kinetic-asy}
\mathcal W_2\left(
\widetilde f_t,\,
\rho_\infty\otimes\delta_0
\right)
\leq
C\,\mathrm e^{-\eta t}.
\qquad
\eta=\frac mM,
\end{equation}
Similarly, let
\(
F_t^N=(\Phi_t^N)_\#F_{\rm in}^N
\)
be a solution of the Liouville equation \eqref{linearLiouville}, and let
\(
\widetilde F_{k,t}^N
\) be the \(k\)-th marginal of 
\((\mathcal G_t^N)_\#F_t^N
\)
where
\(
\mathcal G_t^N(X_N,V_N)
\)
is defined in \eqref{eq:movingframe}.
Then there exist
\(
\rho_{k,\infty}^N
\in\mathcal P(\mathbb{R}^{dk})
\)
and a constant \(C>0\) such that
\begin{equation}
\label{eq:particle-asy}
\mathcal W_2\left(
\widetilde F_{k,t}^N,\,
\rho_{k,\infty}^N\otimes\delta_0^{\otimes k}
\right)
\leq
C\,\sqrt k\,\mathrm e^{-\eta t}.
\end{equation}

<!-- proof-1: body_md -->
We give a proof for the discrete case.
By Proposition~\ref{thm1}, for any \(i=1,\ldots,N\) there exists a constant \(C>0\) such that
\[
\left|
V_i^N(t)-\mathbf V_\infty^N
\right|
\leq
C\,\mathrm e^{-\eta t}.
\]
Since
\[
\frac{\mathrm d}{\mathrm dt}
\left(
X_i^N(t)-t\mathbf V_\infty^N
\right)
=
V_i^N(t)-\mathbf V_\infty^N,
\]
we deduce that
\[
X_{i,\infty}^N
:=
X_i^N(0)
+
\int_0^\infty
\left(
V_i^N(s)-\mathbf V_\infty^N
\right)\,\mathrm ds
\]
exists and
\[
\left|
X_i^N(t)
-t\mathbf V_\infty^N
-X_{i,\infty}^N
\right|
\leq
\frac C \eta\,\mathrm e^{-\eta t}.
\]
This yields to 
\[
\left|
\mathcal G^N_t\bigl(\boldsymbol{X}^N(t),\boldsymbol{V}^N(t)\bigr)
-
\bigl(X^N_\infty,0\bigr)
\right|^2
\leq
C^2\, N \left(1 +\frac{1}{\eta^2}\right)\mathrm e^{-2\eta t},
\]
where \(
X_{\infty}^N
:=
\bigl(
X_{1,\infty}^N,\ldots,X_{N,\infty}^N
\bigr).
\)
Setting \(
 \rho_\infty^N:=(X_{\infty}^N)_\#F_{\rm in}^N\), 
the definition of Wasserstein distance implies that
\[
\mathcal W_2^2 \left(
\widetilde F_t^N,\,
\rho_\infty^N\otimes\delta_0^{\otimes N}
\right)
\leq 
\int_{(\mathbb{R}^d \times \mathbb{R}^d)^N} \left|
\mathcal G^N_t\bigl(\boldsymbol{X}^N(t),\boldsymbol{V}^N(t)\bigr)
-
\bigl(X^N_\infty,0\bigr)
\right|^2\mathrm d f_{\mathrm{in}}^{\otimes N}
\leq
C\,N\, \mathrm e^{-2\eta t},
\]
which concludes the proof.

<!-- lemma-1: body_md -->
\label{lemma:lemmapi}
Consider the Motsch--Tadmor dynamics in \eqref{eq:MTmodel} and \eqref{eq:mesoVlasov} and let \(\{\pi_i^N(t)\}_{i=1}^N\) and \(\pi^\rho(t,x)\) as in \eqref{def:eqmeas}. 
Under the assumption \eqref{in:lowupbound}, the following results hold:
\begin{enumerate}
    \item the weights \(\{a_{ij}^N\}\) defined in \eqref{eq:relinf} are reversible w.r.t. \(\{\pi_i^N(t)\}_{i=1}^N\), i.e. \(\sum_{i=1}^N \pi^N_i(t)=1\) and
    \begin{equation}
    \label{eq:detbal}
    \pi^N_i(t) a_{ij}^N(t)=\pi^N_j(t) a_{ji}^N(t).
    \end{equation}
    Similarly, \(a^{\rho}(t)\) defined in \eqref{eq:meanforce} is reversible w.r.t. \(\pi^\rho(t)\), i.e. \(\int \pi^\rho(t,x)\mathrm d\rho_t(x)=1\) and
    \begin{equation}
    \label{eq:detbal2}
    \pi^\rho(t,x) a^{\rho}(t,x,y)=\pi^\rho(t,y) a^{\rho}(t,y,x).
    \end{equation}
    \item It holds that
    \begin{equation}
\label{in:boundpi}
\max_{1\le i\le N}\left|\dfrac{\dot \pi_i^N(t)}{\pi_i^N(t)}\right|\le \dfrac{2 \text{Lip}(\varphi)}{m}\, D_V^N(t),\quad
\left|\dfrac{\partial_t \pi^\rho+v\cdot \nabla_x \pi^\rho}{\pi^\rho}\right| \le  \frac{2 \text{Lip}(\varphi)}{m} D_V^f(t)
\end{equation}
where
\(D^N_V(t)\) is introduced in \eqref{def:suppdv}, while \(
D^f_V(t)
\) 
in \eqref{def:suppdvkin}.
\item For any \(i \in \{1,\dots,N\}\), it holds that
\[
\eta\le\min \left\{N\pi^N_i(t), \pi^\rho(t,x)\right\} \le \max \left\{N\pi^N_i(t), \pi^\rho(t,x)\right\}\le \dfrac{1}{\eta}, \quad \eta=\dfrac mM.
\]
\end{enumerate}
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
We prove the result at the discrete level, a similar argument works for the kinetic quantities.
If \(\varphi_{ij}:=\varphi(|X_i^N-X_j^N|)\), we get
\[
\pi^N_i a^N_{ij}=\dfrac{\sum_\ell \varphi_{i\ell}}{\sum_{\ell h}\varphi_{\ell h}} \dfrac{\varphi_{i j}}{\sum_{\ell} \varphi_{i \ell}}=
\dfrac{\sum_\ell \varphi_{j \ell}}{\sum_{\ell h}\varphi_{\ell h}} \dfrac{\varphi_{j i}}{\sum_{\ell} \varphi_{j \ell}}=
\pi^N_j\, a^N_{j i}\,
\]
since \(\varphi_{ij}=\varphi_{ji}\), this proves the first item. Concerning \eqref{in:boundpi}, we have
\begin{align}
\log \pi_i^N(t)=\log\left(\sum_\ell \varphi(|X_i^N(t)-X_\ell^N(t)|)\right)- \log\left( \sum_{\ell,h} \varphi(|X_\ell^N(t)-X_h^N(t)|\right).
\end{align}
Taking time derivatives, using \eqref{in:lowupbound} and the definition of \(D_V^N(t)\), we get \eqref{in:boundpi}.
Finally, last item follows by the uniform bounds in \eqref{in:lowupbound}.

<!-- proof-3: body_md -->
We consider the weighted average of the velocities introduced in \eqref{eq:weightedN}
and we show that the limit in time of this quantity exists and corresponds to the consensus velocity of the model. Taking the time derivative
\[
\dfrac{{\mathrm d}}{\mathrm dt} \langle \boldsymbol{V}^N(t)\rangle_{N}= \sum_{i=1}^N   \dot \pi_i^N(t) V_i^N(t)+  \sum_{i=1}^N \pi_i^N(t) \dot V_i^N(t),
\]
by a straightforward computation, the second term is identically equal to zero thanks to \eqref{eq:detbal}.
Hence
\[
\dfrac{{\mathrm d}}{\mathrm dt} \langle \boldsymbol{V}^N(t)\rangle_{N}= \sum_{i=1}^N   \dot \pi_i^N(t) V_i^N(t)= \sum_{i=1}^N   \dot \pi_i^N(t)\left[ V_i^N(t)-  \langle \boldsymbol{V}^N(t)\rangle_{N}\right],
\]
since \(\sum_i \dot \pi_i=0\). By \eqref{in:boundpi}, it follows that
\[
\left| \dfrac{{\mathrm d}}{\mathrm dt} \langle \boldsymbol{V}^N(t)\rangle_{N}\right| \le \sum_{i=1}^N \pi_i^N(t)\left|\dfrac{\dot \pi_i^N(t)}{\pi_i^N(t)}\right|\sum_{j=1}^N \pi_j^N(t)|V_i^N(t)-V_j^N(t)|\le \dfrac{2\text{Lip}(\varphi)}{m}[D^N_V(t)]^2.
\]
By the asymptotic flocking \eqref{in:asymptfl}, we conclude that
\[
\left| \dfrac{{\mathrm d}}{\mathrm dt} \langle \boldsymbol{V}^N(t)\rangle_{N}\right|  \le  \dfrac{2 \text{Lip}(\varphi)}{m} [D^N_V(0)]^2 \mathrm{e}^{-2\eta t}.
\]
This implies that the limit in time of \(\langle \boldsymbol{V}^N(t) \rangle_{N}\) exists and is given by 
\begin{equation}
\label{eq:vinf}
\mathbf{V}^N_\infty:= \langle \boldsymbol{V}^N(0) \rangle_{N}+\int_0^\infty \sum_{i=1}^N \dot \pi_i^N(t) V_i^N(t)\, {\mathrm d}t.
\end{equation}
Moreover
\begin{align*}
|V_i^N(t)- \mathbf{V}_\infty^N| &\le |V_i^N(t)- \langle \boldsymbol{V}^N(t) \rangle_{N}| + |\langle \boldsymbol{V}^N(t) \rangle_{N}- \mathbf{V}_\infty^N|\\
&\le \sum_{j=1}^N \pi_j^N(t) |V_i^N(t)- V_j^N(t)| + |\langle \boldsymbol{V}^N(t) \rangle_{N}- \mathbf{V}_\infty^N|\le C{\mathrm e}^{-\eta t},
\end{align*}
with \(
C:=D^N_V(0)+\text{Lip}(\varphi) [D^N_V(0)]^2/ \eta m.
\)

Similarly, at the kinetic level we can
consider the average velocity introduced in \eqref{eq:weightedf}.
Taking the time derivative and integrating by parts, we get
\begin{align}
\dfrac{{\mathrm d}}{\mathrm dt} \langle v(t)\rangle_{f}&=\int \partial_t \pi^\rho(t,x) v\, \mathrm df_t(x,v)+  \int  \pi^\rho(t,x) v\, \mathrm \partial_tf_t(x,v) \mathrm dx \mathrm dv\nonumber\\
&=\int \left(\partial_t \pi^\rho(t,x) + v \cdot \nabla_x \pi^\rho(t,x)\right) v \, \mathrm d f_t(x,v) \nonumber\\
&\quad+ \int \pi^\rho(t,x) \left[\,\int a^\rho(t,x,y)(w-v)\, \mathrm df_t(y,w)\right] \mathrm df_t(x,v),
\end{align}
by a straightforward computation, the last term is identically equal to zero thanks to \eqref{eq:detbal2}.
Hence
\begin{align}
\dfrac{{\mathrm d}}{\mathrm dt} \langle v(t)\rangle_{f}=\int \left(\partial_t \pi^\rho(t,x) + v \cdot \nabla_x \pi^\rho(t,x)\right) [v-\langle v(t) \rangle_{f}] \, \mathrm d f_t(x,v)
\end{align}
since \(\partial_t \int \pi^\rho(t,x) \mathrm d\rho_t(x)=0\).
By \eqref{in:boundpi}, it follows that
\begin{align}
\left| \dfrac{{\mathrm d}}{\mathrm dt} \langle v(t)\rangle_{f}\right| &\le \int \, \mathrm df_t(x,v)\pi^\rho(t,x)\left|\dfrac{\partial_t\pi^\rho(t)+v\cdot \nabla_x \pi^\rho(t)}{\pi^\rho(t)}\right|\int \pi^\rho(t,y)|v-w|\, \mathrm df_t(y,w)\nonumber\\
&\le \dfrac{2\text{Lip}(\varphi)}{m}[D^f_V(t)]^2.
\end{align}
By the asymptotic flocking \eqref{in:asymptflkin}, we conclude that
\[
\left| \dfrac{{\mathrm d}}{\mathrm dt} \langle v(t)\rangle_{f}\right|  \le  \dfrac{2 \text{Lip}(\varphi)}{m} [D^f_V(0)]^2 \mathrm{e}^{-2\eta t}.
\]
This implies that the limit in time of \(\langle v(t) \rangle_{f}\) exists and is given by 
\begin{equation}
\label{eq:vinff}
\mathbf{v}^f_\infty:= \langle v(0) \rangle_{f}+\int_0^\infty \int(\partial_t \pi^\rho(t)+v\cdot \nabla_x \pi^\rho(t)) v \, \mathrm df_t(x,v)\, {\mathrm d}t.
\end{equation}
Moreover, for any \(v \in \text{supp} f_t\)
\[
|v- \mathbf{v}_\infty^f| \le |v- \langle v(t) \rangle_{f}| + |\langle v(t) \rangle_{f}- \mathbf{v}_\infty^f|\le C{\mathrm e}^{-\eta t},
\]
with \(C:=D^f_V(0)+\text{Lip}(\varphi) [D^f_V(0)]^2/ \eta m\).
