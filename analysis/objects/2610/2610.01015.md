\label{th:linearpde}
For each $v\in X_T$, there exists a unique solution of the linearized equation \eqref{eq: linear pde} in $X_T$, under the condition \eqref{eq: nondeg}.

Now, we have the following weak form for a.e. $t$ and all $\varphi \in H^1_0(\Omega)$
\begin{equation}\label{eq:Weak form}
    \begin{aligned}
        & \int_{\Omega}(1-2\eta v)\, u_{tt} \, \varphi \,dx + \int_{\Omega} c^2(x)\nabla u \cdot   \nabla\varphi dx  + \int_{\Omega} (\nabla c^2 \cdot \nabla u ) \,\varphi \,dx +\int_{\Omega} d(x,t) \nabla (\mathcal{M} *  u_{t}) \cdot \nabla  \varphi \, dx\\
        & + \int_{\Omega} \nabla d(x,t) \cdot \nabla (\mathcal{M} *  u_{t}) \,\varphi \, dx =  \int_{\Omega} 2 \, \eta \, v_t^2 \, \varphi \, dx
    \end{aligned}
\end{equation}
Let $\{\phi_k\}_{k=1}^{\infty}$ be eigenfunctions of Dirichlet Laplacian, i.e.,
\begin{equation} \label{eq:Galerkin}
    \begin{aligned}
    -\Delta \phi_k &= \lambda_k \phi_k \quad \text{in}~ \Omega,    \\
    \phi_k  & =0 \,\,\,\qquad \text{on}~ \partial \Omega,
    \end{aligned}
\end{equation}
with $0< \lambda_1\leq \lambda_2 \leq \hdots$. 
For $N \in \mathbb{N}$, define the finite dimensional subspace 
\[
V_N := \text{span} \{\phi_1,\phi_2,\dots, \phi_N\} \subset V,
\]
and let $u_N$ be an approximate solution of \eqref{eq: linear pde} of the form
\[
u_N(x,t) = \sum_{i=1}^N q_k^N(t) \phi_k(x)
\]
where the scaler functions $q_k^N$ are determined by requiring that the weak form of \eqref{eq: linear pde} holds for all test functions $\varphi \in V_N$.  This provides an \textit{N}-dimensional system of second-order ODE for $q^N_k$, with initial conditions obtained by projecting $u_0$ onto $V_N$. Thus, we obtain a finite-dimensional Volterra integro-differential system, which has a unique local solution in time by standard ODE theory\cite{conti2014well,landes1980galerkin}.

    \end{aligned}
\end{equation}
On the boundary $\partial \Omega$, we get $w(y,t)= u(y,t)-g(y,t)=0$, and hence $w$ satisfies homogeneous Dirichlet data. At $t=0$, we have
\[w(x,0)=f(x)- E(g|_{t=0}), w_t(x,0)=-E(g_t|_{t=0}), \quad x\in B.
\]
Let $\Delta_{D}$ be the dirichlet laplacian in $B$ which is unbounded self adjoint operator in $L^2(B)$, $\left\{\Phi_n(x)\right\}_{n=1}^{\infty} $ be orthonormal basis of eigen functions of $-\Delta_{D}$ in $L^2(B)$ and $\left\{\lambda_n\right\}_{n=1}^{\infty} $ be the corresponding spectrum where we assume $\lambda_n>0$ such that

\[\Phi_n(y) = 0, \quad \forall y \in \partial \Omega.\]

    w(x,t)= \sum\limits_{n=1}^{\infty}w_n(t)\phi_n(x),\\
\end{equation}

\end{equation}

&\int\limits_{B}\Delta w\phi_n(x)dx=-\lambda_n^2w_n(t) \quad \text{(using Green's Identity)}, \\
&\int\limits_{B}\mathcal{M}* \Delta w_t\phi_n(x)dx=(-\lambda_n^2)\mathcal{M}*w_n'(t), \\
& -\int\limits_{B}E(g_{tt})\phi_n(x)dx = Q_n(t).
\end{aligned}
\]

\end{aligned}

\end{aligned}   
\end{equation}

\end{equation}

\]
which gives us
\[ f(x)=\sum\limits_{n=1}^{\infty}w_n(0)\phi_n(x)+E(g|_{t=0}).

is equivalent to solving
\[
\mathcal W_{\mathrm{l}}(f)+\mathcal K(f)=g.
\]
Using Theorem 5, let \(\mathcal R_{\mathrm{l}}\) denote the local linear reconstruction operator satisfying
\[
\mathcal R_{\mathrm{l}}\mathcal W_{\mathrm{l}}(f)=f.
\]
Then the nonlinear inverse problem can be rewritten as the fixed-point problem
\[
f
=
\mathcal R_{\mathrm{l}}\big(g-\mathcal K(f)\big).
\]
Define
\[
\mathcal T(f):=\mathcal R_{\mathrm{l}}\big(g-\mathcal K(f)\big).
\]
Since \(\mathcal R_{\mathrm{l}}g=f_{\mathrm{l}}\), we obtain
\[
\mathcal T(f)
=
f_{\mathrm{l}}-\mathcal R_{\mathrm{l}}\mathcal K(f).
\]
We now show that \(\mathcal T\) is a contraction on a sufficiently small ball
\[
B_\rho(f_{\mathrm{l}})
=
\{f\in H^1(\Omega):\|f-f_{\mathrm{l}}\|_{X}\le \rho\}.
\]
By the local well-posedness theorem for the nonlinear damped wave equation, the solution satisfies an estimate of the form
\[
\|u\|_Z
\le C\|f\|_{X}
\]
for \(f\) sufficiently small. Moreover, since
\[
(u^2)_{tt}=2u_t^2+2uu_{tt},
\]
standard product estimates give
\[
\|\eta(u^2)_{tt}\|_{L^2(0,T;L^2(\Omega))}
\le
C\|\eta\|_{L^\infty}\|u\|_Z^2.
\]
Therefore,
\[
\|\mathcal K(f)\|_{L^2(0,T;H^{1/2}(\partial\Omega))}
\le
C\|\eta\|_{L^\infty}\|f\|_{X}^2.
\]
Applying the boundedness of \(\mathcal R_{\mathrm{l}}\), we get
\[
\|\mathcal T(f)-f_{\mathrm{l}}\|_{X}
\le
C\|\eta\|_{L^\infty}\|f\|_{X}^2.
\]
Hence, if \(\rho\), \(\|f_{\mathrm{l}}\|_{X}\), and \(\|\eta\|_{L^\infty}\) are sufficiently small, then
\[
\mathcal T(B_\rho(f_{\mathrm{l}}))
\subset B_\rho(f_{\mathrm{l}}).
\]
Next, let \(f_1,f_2\in B_\rho(f_{\mathrm{l}})\), and let \(u_1,u_2\) be the corresponding nonlinear
solutions. Then
\[
u_1^2-u_2^2=(u_1-u_2)(u_1+u_2).
\]
Using the stability estimate for the nonlinear forward problem, given in \eqref{eq:stability}, we obtain
\[
\|\mathcal K(f_1)-\mathcal K(f_2)\|
\le
C\|\eta\|_{L^\infty}
\big(\|u_1\|_Z+\|u_2\|_Z\big)
\|u_1-u_2\|_Z.
\]
Furthermore, the same stability estimate gives us
\[
\|u_1-u_2\|_Z
\le
C\|f_1-f_2\|_{X}.
\]
Therefore,
\[
\|\mathcal T(f_1)-\mathcal T(f_2)\|_{X}
\le
C\|\eta\|_{L^\infty}
\big(\|f_1\|_{X}+\|f_2\|_{H^1}\big)
\|f_1-f_2\|_{X}.
\]
Since \(f_1,f_2\in B_\rho(f_{\mathrm{l}})\), we have
\[
\|f_j\|_{X}\le \|f_{\mathrm{l}}\|_{X}+\rho,
\qquad j=1,2.
\]
Thus,
\[
\|\mathcal T(f_1)-\mathcal T(f_2)\|_{X}
\le
C\|\eta\|_{L^\infty}
\big(\|f_{\mathrm{l}}\|_{X}+\rho\big)
\|f_1-f_2\|_{X}.
\]
Choose \(\rho\), \(T\), and \(\|\eta\|_{L^\infty}\) sufficiently small so that
\[
C\|\eta\|_{L^\infty}
\big(\|f_{\mathrm{l}}\|_{X}+\rho\big)<1.
\]
Then \(\mathcal T\) is a contraction on \(B_\rho(f_{\mathrm{l}})\). By Banach's fixed-point theorem,
there exists a unique \(f\in B_\rho(f_{\mathrm{l}})\) such that
\[
f=\mathcal T(f).
\]
Equivalently,
\[
\mathcal W(f)=g,
\]
which proves the result.

<!-- prop-2: body_md -->
\label{th:coercive}
The objective functional \( J \), defined in \eqref{eq:minproblem}, is sequentially weakly lower semi-continuous (w.l.s.c.), bounded from below, and coercive on the set \( U_{ad} \).
<!-- end prop-2 -->
<!-- proof-7: body_md -->
The bounded from below and coercivity property of $J$ follows from the non-negativity and the fact that $J$ contains the $L^2$ regularization of $u_0$. For proving the w.l.s.c property, we first note that $J$ is not convex in $u_0$. However, $U_{ad}$ is convex and closed and also weakly sequentially compact. This is because if we choose any $\{u_0^n\}\in U_{ad}$ such that $||u_0^n||_{L^2}\leq D,$ for all $n$, then the sequence is bounded. Thus, there is a  subsequence $\{u_0^{n_k}\}$ such that $u_0^{n_k}\rightharpoonup y\in U_{ad}$. Also, we have that $||u_0||_{L^2}\leq \lim\inf||u_0^n||_{L^2}\leq D$. Thus, we obtain $u_0\in U_{ad}$. Next, we consider the set 
\[
U_{\eta} = \lbrace u_0 \in U_{ad}: J(u_0) \leq \zeta \rbrace.
\]
Since $J$ is continuous, the set $U_{\eta}$ is closed for all $\zeta \in \mathbb{R}$. Thus, $U_{\eta}$ is a closed subset of a weakly sequentially compact spaces $U_{ad}$ and is weakly sequentially closed for all $\zeta\in\mathbb{R}.$
This implies that $J$ is weakly sequentially lower semi-continuous.

\label{th:SQH} Let $\left(u_0^{k},u^{k}\right)$ and $\left(u_0^{k+1},u^{k+1}\right)$ be generated by the SQH method (Algorithm \ref{algSQHmethod}) applied to \eqref{eq:minproblem}, with $u_0^{k+1}, u^k \in U_{ad}$. \end{equation*}
Hence, we obtain the inequality
\begin{equation}\label{ePartialMax}
{H} \left(x,u_0^{k+1}, p^k\right)+\epsilon \, | u_0^{k+1} - u_0^k |^2 \le {H} \left(x,u_0^k, p^k\right) .

<!-- thm-10: body_md -->
\label{Tsqh2}
Under the assumptions of Theorem \ref{th:SQH}, if in Algorithm \ref{algSQHmethod}, at every $k$-th iterate, $\epsilon=C+\eta$ 
is chosen (and tolerance  $\kappa=0$, so that the algorithm never stops), then the following holds
\begin{enumerate}[(a)]
\item $\lim\limits_{k \to \infty} | J(u_0^{k+1},p^{k+1}) - J(u_0^{k},p^k)|=0$
	\label{sqhac}
\item $\lim\limits_{k \to \infty} \| u_0^{k+1} - u_0^{k} \|_{ L^2 \left( \Omega \right) }=0$. 
	\label{sqhbc}
\item Along a subsequence, $
u_0^k \rightharpoonup \bar u_0,  
~ \text{in } L^2(\Omega), $
with corresponding state and adjoint limits
\[
u^k \to \bar u, \quad \text{in } L^2(0,T; L^2(\Omega)) 
\qquad
p^k \to \bar p, \quad \text{in } L^2(0,T; L^2(\Omega)), 
\]
and the pair $(\bar u_0,\bar p)$ satisfies the PMP criterion
\[
H(x,\bar u_0(x),\bar p(x,:))
=
\min_{v\in [u_l,u_r]}
H(x,v,\bar p(x,:)),
\qquad \text{for a.e. } x\in \Omega .
\]
\end{enumerate}

<!-- proof-13: body_md -->
The first statement follows from \eqref{eq:ineq}, since $\epsilon$ can be chosen greater than $C$, which would imply that the sequence $\lbrace J(u_0^{k},u^k)\rbrace $ is monotonically decreasing the $\mathbb{R}$ and is a Cauchy sequence, hence convergent. To prove \eqref{sqhbc}, we rewrite 
\eqref{eq:ineq} as follows 
$$
 \| u_0^{k+1} - u_0^{k} \|^{2}_{ L^2 \left( \Omega\right) } \le \frac{1}{\eta}
 \Big[ J\left(u_0^k, p^{k}\right)   - J\left(u_0^{k+1},p^{k+1}\right) \Big] . 
$$
Therefore we have the partial sum
$$
 \sum_{k=0}^K \| u_0^{k+1} - p^{k} \|^{2}_{ L^2 \left( \Omega \right) } \le \frac{1}{\eta}
 \Big[ J\left(u_0^0,p^{0}\right)   - J\left(u_0^{K+1},p^{K+1}\right) \Big].
$$
This shows that in the limit $K \to  \infty$, the series with positive elements 
$\| u_0^{k+1} - u_0^{k} \|^{2}_{ L^2(\Omega)}$ is convergent, which proves the result. 

For the final statement, we note that since $U_{ad}$ is weakly sequentially compact in $L^2(\Omega)$, there exists a subsequence, still denoted by $\{u_0^k\}$, and an element $\bar u_0\in U_{ad}$ such that
\[
u_0^k\rightharpoonup \bar u_0
\quad \text{in }L^2(\Omega).
\]
Let
\[
u^k=\Lambda(u_0^k)
\]
be the corresponding sequence of states. By the compactness argument used in Theorem~\ref{th:existence}, together with the Aubin--Lions lemma, there exists a further subsequence such that
\[
u^k\to \bar u
\quad \text{strongly in }L^2(0,T;L^2(\Omega)).
\]
By the stability of the forward problem and the continuity of the control-to-state map, the limit satisfies
\[
\bar u=\Lambda(\bar u_0).
\]
Therefore, $(\bar u_0,\bar u)$ is an admissible state-initial condition pair. It remains to identify the limit as a stationary point. At the $k$-th SQH iteration, the updated initial condition $u_0^{k+1}$ is obtained by pointwise minimization of the Hamiltonian. Hence,
\[
H(x,u_0^{k+1}(x),p^k(x,:))
=
\min_{v\in[u_l,u_r]}
H(x,v,p^k(x,:)),
\qquad \text{for a.e. }x\in\Omega.
\]
Equivalently, for every $v\in[u_l,u_r]$,
\[
H(x,u_0^{k+1}(x),p^k(x,:))
\le
H(x,v,p^k(x,:)),
\qquad \text{for a.e. }x\in\Omega.
\]
Passing to the limit along the convergent subsequence, using the convergence of $p^k$ to $\bar p$, the weak convergence of $u_0^{k+1}$ to $\bar u_0$, and the continuity of the Hamiltonian with respect to its arguments, we obtain
\[
H(x,\bar u_0(x),\bar p(x,:))
\le
H(x,v,\bar p(x,:)),
\qquad \forall v\in[u_l,u_r],
\]
for a.e. $x\in\Omega$. Therefore,
\[
H(x,\bar u_0(x),\bar p(x,:))
=
\min_{v\in[u_l,u_r]}
H(x,v,\bar p(x,:)),
\qquad \text{for a.e. }x\in\Omega.
\]
which implies that $\bar u_0$ satisfies the PMP optimality condition.
