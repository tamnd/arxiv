<!-- theorem-1: body_md -->
\label{thm:main}
For every $0<s<1/2$, there exists an $s$-Ahlfors regular compact set
$K\subset\mathbb{R} \subset {\mathbb C}$ such that, if $\omega^\infty$ denotes the harmonic
measure of ${\mathbb C} \setminus K$ with pole at infinity, 
\[
    \dim_{\mathcal H}\omega^\infty=s.
\]
Moreover, $\omega^\infty$ and $\mathcal{H}^s|_K$ are equivalent, i.e., there is some constant $C \ge 1$ for which 
\begin{align*}
	C^{-1} \, \mathcal{H}^s|_K(F)\le \omega^\infty(F) \le C \mathcal{H}^s|_K(F) \text{ for all } F \subset K \text{ Borel.}
\end{align*}

<!-- lemma-1: body_md -->
\label{lem:cheb0}
For each $N \geq2$, we have 
\[
 \sum_{j=1}^Nq_j=0\quad \text{ and } \quad
 \frac1N\sum_{j=1}^Nq_j^2=\frac12.
\]
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
The fact that $\sum_{j=1}^Nq_j=0$ follows because the set of zeros is symmetric about the origin.
To show the second identity, we use that
$q_j=\cos\theta_j$ and
\[
 \cos^2\theta=\frac{1+\cos2\theta}{2},
\]
while the equally spaced angles $2\theta_j$ in $[0,2\pi)$ make the sum of $\cos2\theta_j$ vanish.

<!-- lemma-2: body_md -->
\label{lem:cheb}
For $1\le j\le N$,
\begin{equation}\label{eqTX20}
 b_j=\log N-(N-1)\log2-\log\sin\theta_j,
 \qquad \beta_j=\log\frac{\sin\theta_1}{\sin\theta_j}\le0.
\end{equation}
Moreover,
\begin{equation}\label{eqTX21}
 \sum_{k\ne j}\frac1{q_j-q_k}=\frac{q_j}{2(1-q_j^2)},
\end{equation}
\begin{equation}\label{eqTX22}
 \sum_{k\ne j}\frac1{|q_j-q_k|^2}\lesssim\frac{N^2}{\sin^2\theta_j},
\end{equation}
and
\begin{equation}\label{eqTX23}
 d_N:=\min_{j\ne k}|q_j-q_k|\simeq N^{-2}, \qquad  \left \lvert { q_j - q_{j+1}  }  \right \rvert    \gtrsim \sin(\theta_j)/ N \simeq \sin(\theta_{j+1})/N.
\end{equation}
The implicit constants are absolute.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
The first identity follows immediately from \eqref{eqTX5}:
$$b_j=\sum_{k\ne j}\log|q_j-q_k|=\log N-(N-1)\log2-\log\sin\theta_j.$$
This implies that
$$\beta_j=b_j-b_* =  b_j -\max_kb_k = b_j - b_1 = \log\frac{\sin\theta_1}{\sin\theta_j}\le0.
$$ 

To deduce the identity involving the reciprocal sums, notice that,
for a polynomial $P(x)=c\prod_k(x-x_k)$ with simple zeros,
\[
 \frac{P''(x_j)}{2P'(x_j)}=\sum_{k\ne j}\frac1{x_j-x_k}.
\]
Applying this to $T_N$ and using the differential equation \eqref{eqdiff} at $q_j$ gives
\begin{equation}\label{eqTX6}
 \sum_{k\ne j}\frac1{q_j-q_k} = \frac{T_N''(q_j)}{2\,T_N'(q_j)}
 =\frac{q_j}{2(1-q_j^2)}.
\end{equation}

To prove the inequality about the squared reciprocal terms,
write $T_N(x)=(x-q_j)Q_j(x)$. Then
\[
 \frac{Q_j'(q_j)}{Q_j(q_j)}=\sum_{k\ne j}\frac1{q_j-q_k},
 \qquad
 \left(\frac{Q_j'}{Q_j}\right)'(q_j)=-\sum_{k\ne j}\frac1{(q_j-q_k)^2}.
\]
Since $T_N'(q_j)=Q_j(q_j)$, $T_N''(q_j)=2Q_j'(q_j)$, and the third derivative satisfies $T_N'''(q_j)=3Q_j''(q_j)$,
\[
 \sum_{k\ne j}\frac1{(q_j-q_k)^2} = -\frac{Q_j''(q_j)}{Q_j(q_j)} + \frac{Q_j'(q_j)^2}{Q_j(q_j)^2}
 =\left(\frac{T_N''(q_j)}{2T_N'(q_j)}\right)^2-\frac{T_N'''(q_j)}{3T_N'(q_j)}.
\]
Differentiating the Chebyshev differential equation gives
\[
 (1-x^2)T_N'''(x)-3xT_N''(x)+(N^2-1)T_N'(x)=0.
\]
At $x=q_j$ and using also \eqref{eqTX6}, this yields
\[
 \frac{T_N'''(q_j)}{T_N'(q_j)}= \frac{3q_jT_N''(q_j)}{(1-q_j^2)T_N'(q_j)}-\frac{N^2-1}{1-q_j^2}
 =\frac{3q_j^2}{(1-q_j^2)^2}-\frac{N^2-1}{1-q_j^2}.
\]
Therefore
\[
 \sum_{k\ne j}\frac1{(q_j-q_k)^2}=
 \biggl(\frac{q_j}{2(1-q_j^2)}\biggr)^2 - 
 \frac13\biggl(\frac{3q_j^2}{(1-q_j^2)^2}-\frac{N^2-1}{1-q_j^2}\biggr)=
 \frac{N^2-1}{3(1-q_j^2)}-\frac{3q_j^2}{4(1-q_j^2)^2}.
\]
In particular,
\[
 \sum_{k\ne j}\frac1{|q_j-q_k|^2}
 \lesssim\frac{N^2}{\sin^2\theta_j}.
\]

To prove that $d_N\simeq N^{-2}$, 
observe that the gaps between consecutive Chebyshev nodes are smallest at the endpoints. Hence,
\[
 \min_{j\ne k}|q_j-q_k| = |q_1-q_2|
 =\cos\frac{\pi}{2N}-\cos\frac{3\pi}{2N}
 \simeq N^{-2}.
\]

<!-- lemma-3: body_md -->
\label{lem:weighted}
For every integer $m\ge2$ and every $0\le a\le m$,
\[
 \sum_{\ell\ne j}\frac{\sin^a\theta_\ell}{|q_j-q_\ell|^m}
 \lesssim_m \frac{N^m}{\sin^{m-a}\theta_j}.
\]
The same estimate holds, up to a constant depending on $m$ and the bi-Lipschitz constant, if $q_j-q_\ell$ is replaced by $Q(q_j)-Q(q_\ell)$ for a uniformly bi-Lipschitz map $Q$ on $[-1,1]$.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Use
\[
 |q_j-q_\ell|=2\sin\frac{\theta_j+\theta_\ell}{2}
 \left|\sin\frac{\theta_j-\theta_\ell}{2}\right|.
\]
We shall use the symmetric estimate
\[
 \sin\frac{\theta_j+\theta_\ell}{2}
 \ge \frac12\max\{\sin\theta_j,\sin\theta_\ell\}.
\]
Indeed, for fixed $u\in(0,\pi)$, the minimum of
\(\sin((u+v)/2)\) over \(v\in[0,\pi]\) is attained in one of the end-points. Then it follows that
\[
 \sin\frac{u+v}{2}\ge \min\left\{\sin\frac u2,\sin\frac{u+\pi}2\right\}= 
 \min\left\{\sin\frac u2,\cos\frac u2\right\}\geq \sin\frac u2\,\cos\frac u2
 \ge\frac12\sin u,
\]
and the same argument with $u$ and $v$ interchanged gives the claim. 

Since $|\sin\frac x2|\gtrsim |x|$ for $x\in[-\pi,\pi]$, we also have
\[
 \left|\sin\frac{\theta_j-\theta_\ell}{2}\right|\gtrsim\frac{|j-\ell|}{N}.
\]
Thus
$$|q_j-q_\ell| \gtrsim \max\{\sin\theta_j,\sin\theta_\ell\}\,\frac{|j-\ell|}{N},$$

\leq

\end{aligned}

|y-v|
\leq

\leq

|u-v|

\end{aligned}

This proves the desired estimates on $|x-y|$.

For $\ell \ge 1$, recall the notation $\mathcal Y_\ell(p)$, which denotes the set of points
$y\in K_{n-1}$ whose nearest common ancestor with $p$ is $\ell$ generations above $p$. We split 
\begin{align*}
	\Delta_2 & = N \sum_{y \in K_{n-1} \setminus\{p\}} \left( \log \left \lvert {x-y}  \right \rvert    - \log \left \lvert {p-y}  \right \rvert   \right) =  \sum_{ \ell = 1  }^{n-1} \Delta_2^{(\ell)}, \\
	\Delta_2^{(\ell)} & \coloneqq N \sum_{  y \in \mathcal{Y}_\ell(p) } \left( \log \left \lvert {x-y}  \right \rvert    - \log \left \lvert {p-y}  \right \rvert   \right).
\end{align*}
Recall from Lemma \ref{lem:ahlf_reg} that
\begin{equation}\label{eq:corona-cardinality-delta2}
	\#\mathcal Y_\ell(p)
	=
	(N-1)N^{\ell-1}.
\end{equation}
Let us first estimate $\Delta_2^{(1)}$, which corresponds to the siblings of $p$ in $K_{n-1}$. 

Suppose that \(p\) has index $j$ and parent \(q\in K_{n-2}\), and recall that \(x\in K_n\) has index $k$ and parent \(p\), so that we may write
\begin{align*}
p =q+a_{n-2}(q)r^{n-2}q_j,  \qquad  x=p+a_{n-1}(p)r^{n-1}q_k.
\end{align*}
The elements of $\mathcal{Y}_1(p)$ are given by $y_\ell = q + a_{n-2}(q)r^{n-2} q_\ell$ for $\ell \ne j$, and so direct computation gives us 
\begin{align*}
	\frac{  \left \lvert {x-y_\ell}  \right \rvert    }{ \left \lvert {p - y_\ell}  \right \rvert   } =  \left \lvert { 1 + r \frac{a_{n-1}(p)}{a_{n-2}(q)}  \frac{ q_k  }{q_j - q_\ell}  }  \right \rvert    \eqqcolon  \left \lvert { 1 +  \frac{ t_x  }{q_j - q_\ell}  }  \right \rvert   .
\end{align*}
Since $a_{n-2}(q)\ge1$, $|q_k|\le1$, and $a_{n-1}(p)\lesssim_\eta N\sin\theta_j$, we see
\begin{equation}\label{eq:t-estimate} |t_x| \lesssim_\eta rN\sin\theta_j,
\end{equation} 
and so with the estimate \eqref{eqTX23} we further obtain
\begin{align*}
	 \left \lvert {\frac{t_x}{q_j - q_\ell}  }  \right \rvert    \lesssim_\eta r N^3 \sin\theta_j \ll 1,
\end{align*}
provided that $N$ is sufficiently large. We may then use $\log \left \lvert {1 + u}  \right \rvert    = u + O(u^2)$ for $ \left \lvert {u}  \right \rvert    \le 1/2$ to obtain 
\begin{align*}\label{eq:Delta2-first-exact} \Delta_2^{(1)} =  N\sum_{y \in \mathcal{Y}_1(p)}
	\log\frac{|x-y|}{|p-y|} & = N\sum_{\ell\ne j} \log\left| 1+\frac{t_x}{q_j-q_\ell} \right|  \\
	& = Nt_x\sum_{\ell\ne j}\frac1{q_j-q_\ell} 
	+ O\left( Nt_x^2 \sum_{\ell\ne j} \frac1{|q_j-q_\ell|^2} \right).
\end{align*} 
By applying Lemma \ref{lem:cheb}, \eqref{eq:t-estimate}, and recalling that $(1 - q_j^2) = \sin^2 \theta_j$,  we deduce that 
\begin{equation}\label{eq:Delta2-first-final}

	\end{split}
\end{equation}
since $\sin\theta_j \ge \sin \theta_1 \simeq N^{-1}$.

We now estimate the contribution of the $\mathcal{Y}_\ell(p)$ for $\ell \ge 2$, which is much simpler. Fix $y \in \mathcal{Y}_\ell(p)$, and recall from Lemma \ref{lem:ahlf_reg} that for $N$ large enough, we have 
\begin{align*}

\end{align*}

\end{align*}

\begin{align*}

\end{align*}
since $Nr \ll 1$ for $N$ large. Along with \eqref{eq:Delta2-first-final} this completes the proof of the estimate on $\Delta_2$.

	For every $0 < s< 1/3$, there exists an $s$-Ahlfors regular compact set $K \subset {\mathbb R} \subset {\mathbb C}$ so that the harmonic measure in ${\mathbb C} \setminus K$ with pole at infinity $\omega^\infty$ is equivalent to $\mathcal{H}^s|_K$, i.e., for some constant $C \ge 1$,

Since the proof of this Theorem follows from the arguments in \cite{DJJ23}, we outline just the steps. As long as $N$ is chosen large enough, Lemma \ref{lem:ahlf_reg} shows that $K_n$ converge in the Hausdorff metric to a compact, $s$-Ahlfors regular subset $K \subset {\mathbb R}\subset {\mathbb C}$. 	\begin{align*}
		\mu_n = N^{-n} \sum_{x \in K_n} \delta_x, 
	\end{align*}
	then we know that $\mu$ is an $s$-Ahlfors regular probability measure supported on $K$, and thus is equivalent to $\mathcal{H}^s|_K$. Arguing just as in \cite[Lemma 3.3]{DJJ23}, but now using Lemma \ref{lem:basic-ind}, one can show that the potential

	\end{align*}

Since $\mu$ is equivalent to $\mathcal{H}^s|_K$, this completes the proof.

We start with the proof of \eqref{eq:relative-scale}. If $p, y \in K_{n-1}$ are siblings with indices $j$ and $\ell$ respectively, we use the estimate on $a_{n-1}(p) + a_{n-1}(y) \lesssim_{M, \eta} N (\sin \theta_j + \sin \theta_\ell)$ and obtain

	\begin{align*}

	for $N$ large. This completes the proof of the first claim.
	
	As for the second claim, suppose again that $p$ has index $j$ and $\hat{p} = q \in K_{n-2}$. We split the sum in \eqref{eq:geometric-sum} by $y \in \mathcal{Y}_\ell(p)$ for $1 \le \ell \le n-1$. 	\end{align*}
	since $d \ge 2$ and $\tau \in (0,1)$ for $N$ sufficiently large. This completes the proof.

and $|x_k-y|\approx |p-y|$, the remainder satisfies
\[
\begin{aligned}
    |\mathcal R_y|
    &\lesssim_M
    \sum_{\ell=1}^N
    \left|\frac{u_\ell}{x_k-y}\right|^M\lesssim_{M,\eta}
    N\frac{\rho_y^M}{|p-y|^M}.
\end{aligned}
\]
Consequently, by Lemma~\ref{lem:prev_estimates}
with $d=a=M$, we obtain
\[
\begin{aligned}
    \sum_{\substack{y\in K_{n-1}\\y\ne p}}
    |\mathcal R_y|
    &\lesssim_{M,\eta}
    N\sum_{\substack{y\in K_{n-1}\\y\ne p}}
    \frac{\rho_y^M}{|p-y|^M}\lesssim_{M,\eta}N\tau^M.
\end{aligned}
\]
Thus the total contribution of these remainders
is of the required order, uniformly in the generation.

We next expand
\[
    (x_k-y)^{-a}
    =
    (p-y+h_k)^{-a}
\]
around $p-y$. Up to terms of total order at least $M$,
this produces terms of the form
\begin{equation}
\label{eq:cluster-general-term}
    c_{a,b}\,
    \frac{
        \rho_y^a\rho_p^b
    }{
        (p-y)^{a+b}
    }
    \left(
        \sum_{\ell=1}^N Q_y(q_\ell)^a
    \right)
    Q_p(q_k)^b,
\end{equation}
where
\[
    a\ge2,
    \qquad b\ge0,
\]
and $c_{a,b}$ depends only on $a$ and $b$.
The case $b=0$ is independent of $q_k$ and will be
retained as the constant part of the corresponding profile.


We now expand the polynomial perturbations in both $Q_y^a$ and
$Q_p^b$. A typical monomial in
\[
    Q_y(q_\ell)^aQ_p(q_k)^b
\]
has the form
\begin{equation}
\label{eq:cluster-corrected-monomial}
    q_\ell^{a-r}
    q_k^{b-s}
    \prod_{i=1}^r
        P_{\nu_i+1,y}(q_\ell)
    \prod_{j=1}^s
        P_{\mu_j+1,p}(q_k),
\end{equation}
where
\[
    0\le r\le a,
    \qquad
    0\le s\le b, \qquad \text{ and } 1\le\nu_i,\mu_j\le M-1.
\]
Set
\[
    \sigma
    =
    \sum_{i=1}^r\nu_i
    +
    \sum_{j=1}^s\mu_j.
\]
Since \eqref{eq:new-correction-bounds} holds both at $y$ and at $p$,
the product of the polynomials in
\eqref{eq:cluster-corrected-monomial} is bounded by
\[
    C_M\tau^\sigma.
\]

We claim that the contribution obtained from
\eqref{eq:cluster-general-term} and
\eqref{eq:cluster-corrected-monomial}, after summing over
$y\in K_{n-1}\setminus\{p\}$, is bounded by
\begin{equation}
\label{eq:cluster-general-bound}
    C_MN\tau^{a+b+\sigma}.
\end{equation}
Indeed, the sum over the $N$ children of $y$ contributes at most a
factor $C_MN$, and hence the absolute value of the corresponding term
is bounded by
\[
    C_MN\tau^\sigma
    \sum_{\substack{y\in K_{n-1}\\y\ne p}}
    \frac{\rho_y^a\rho_p^b}
         {|p-y|^{a+b}}.
\]
Applying Lemma~\ref{lem:prev_estimates} with exponent $a+b$ and with
the power of $\rho_y$ equal to $a$, we obtain
\[
    \sum_{\substack{y\in K_{n-1}\\y\ne p}}
    \frac{\rho_y^a\rho_p^b}
         {|p-y|^{a+b}}
    \lesssim_{M, \eta}
    \tau^{a+b},
\]
since $a\ge2$. This proves
\eqref{eq:cluster-general-bound}.

We now organize the expansion according to its total order. Set
\begin{equation}
\label{eq:cluster-total-order}
    m
    =
    a+b+\sigma.
\end{equation}
By \eqref{eq:cluster-general-bound}, every term of total order $m$ is
bounded by $C_MN\tau^m$. Moreover, its degree in the child variable $q_k$ is also at most $m$. Indeed, the
only factor depending on $q_k$ in
\eqref{eq:cluster-corrected-monomial} is
\[
    q_k^{b-s}
    \prod_{j=1}^sP_{\mu_j+1,p}(q_k),
\]
whose degree is at most
\[
    (b-s)
    +
    \sum_{j=1}^s(\mu_j+1)
    =
    b+\sum_{j=1}^s\mu_j
    \le
    a+b+\sigma
    =
    m.
\]
Thus all terms of total order $m$ form a polynomial in $q_k$ of
degree at most $m$.

The same bookkeeping gives the triangular dependence needed later.
Since $a\ge2$, from
\[
    m=a+b+\sigma
\]
we have terms of the form \eqref{eq:cluster-corrected-monomial} must satisfy $\nu_i, \mu_j \le \sigma \le m-2$. Consequently the terms of total order $m$ depend only on
\[
    P_{\nu+1,z},
    \qquad
    z\in K_{n-1},
    \qquad
    1\le\nu\le m-2.
\]

For $2\le m\le M-1$, let $C_{m,p}^{(3)}$ be the sum of all explicit
terms for which
\[
    a+b+\sigma=m.
\]
Then $C_{m,p}^{(3)}
\in
\mathcal P_m$ and 
\[
    \|C_{m,p}^{(3)}\|_{\mathrm{coeff}}
    \lesssim_M
    N\tau^m.
\]
The terms with $b=0$ are independent of $q_k$ and are included as the
constant terms of $C_{m,p}^{(3)}$. A nonconstant term must have $b\ge1$. Since $a\ge2$, every
nonconstant term has total order
\[
    m=a+b+\sigma\ge3.
\]
Thus $C_{2,p}^{(3)}$ is constant, and the first possible non-constant
cluster polynomial occurs with $m \ge 3$, which corresponds to $\tau^3$.

It remains to estimate the discarded terms. All explicit terms for which the total order satisfies
\[
    a+b+\sigma\ge M
\]
are bounded, after summation, by
\[
    C_MN\tau^M
\]
using \eqref{eq:cluster-general-bound} and $\tau<1$. The Taylor remainder arising from the expansion of
$(p-y+h_k)^{-a}$ is estimated using Lemma \ref{lem:prev_estimates}. Namely, every such term
contains a factor of the form
\[
\frac{\rho_y^a\rho_p^b}
{|p-y|^{a+b}}
\]
with $a+b\ge M$, and Lemma~\ref{lem:prev_estimates} gives again an
$O_M(N\tau^M)$ contribution. Collecting all discarded terms into $E_{3,M,p}$ (including the first Taylor remainder from the first expansion of the $\log$) we conclude that
\[
    |E_{3,M,p}(q_k)|
    \lesssim_M
    N\tau^M,
    \qquad
    1\le k\le N.
\]
This proves \eqref{eq:Delta3-profile-expansion}.

Suppose that $p$ is the $j$-th child of $q$.  Since the children of
$q$ were constructed using $Q_q$,
\[
S_{n,d}(p)
=
\rho_q^{-d}
\sum_{\ell\ne j}
\frac1{(Q_q(q_j)-Q_q(q_\ell))^d}.
\]
The scale bound gives
\begin{equation}
\label{eq:rho-child-parent}
\frac{\rho_p}{\rho_q}
=
r\frac{a_n(p)}{a_{n-1}(q)}
\lesssim_{M,\eta}
Nr\sin\theta_j.
\end{equation}

We first consider $d\ge2$. Since $Q_q$ is bi-Lipschitz as in \eqref{eq:Q-bilip}, we apply Lemma \ref{lem:weighted} with $a = 0$ to see that 
\[
\sum_{\ell\ne j}
\frac1{|Q_q(q_j)-Q_q(q_\ell)|^d}
\lesssim_{d,M}
\frac{N^d}{\sin^d\theta_j}.
\]
Hence we conclude 
\begin{align*}
N\rho_p^d|S_{n,d}(p)|
&\lesssim_{M,\eta}
N(Nr)^d\sin^d\theta_j
\frac{N^d}{\sin^d\theta_j} = C_{M,\eta}N\tau^d.
\end{align*}

\[
Q_q(q_j)-Q_q(q_\ell)
=
(q_j-q_\ell)(1+\varepsilon_{j\ell}),
\qquad
|\varepsilon_{j\ell}|\lesssim_M\tau.
\]
For $\tau$ sufficiently small,
\[
\frac1{Q_q(q_j)-Q_q(q_\ell)}
=
\frac1{q_j-q_\ell}
+
O_M\left(\frac{\tau}{|q_j-q_\ell|}\right).

\[
\left|\sum_{\ell\ne j}\frac1{q_j-q_\ell}\right|
\lesssim

\sum_{\ell\ne j}\frac1{|q_j-q_\ell|}
\lesssim
\frac{N\log N}{\sin\theta_j}.

|S_{n,1}(p)|
\lesssim_M

\left(
\frac1{\sin^2\theta_j}
+
\frac{\tau N\log N}{\sin\theta_j}
\right).

N\rho_p|S_{n,1}(p)|

N^{1-d}\tau^dX_{n-1,d}\\
&\le
C_{M,\eta}\tau X_{n-1,d},
\end{align*}
because $d\ge1$ and $0<\tau<1$.

$m$ in place of $n$, by $N\rho_p^d$, divide by $N\tau^d$, and use
Lemmas~\ref{lem:field-local}, \ref{lem:field-evaluation}, and
\ref{lem:field-cluster}.  We obtain
\begin{align}
X_{m,d}
\le{}&
C_{M,\eta}
+
C_{M,\eta}\tau X_{m-1,d}
+
C_{M,\eta}
\sum_{t=1}^{M-d}
\tau^{t+1}X_{m-1,d+t}
+
C_{M,\eta}\tau^2
+
C_{M,\eta}\tau^{M+1}.
\label{eq:Xnd-recursion}
\end{align}
When $d=M$, the sum in $t$ is empty.

Since
\[
    X_{m-1,d+t}\le X_{m-1}
\]
and, for $\tau\le1/2$,
\[
\sum_{t=1}^{M-d}\tau^{t+1}
\le
\frac{\tau^2}{1-\tau}
\le
2\tau^2,
\]
we obtain, after enlarging the constant,
\begin{equation}
\label{eq:Xn-recursion}
    X_m
    \le
    C_{M,\eta}
    +
    C_{M,\eta}
    \bigl(\tau+2\tau^2\bigr)X_{m-1}.
\end{equation}

Since $ \tau=N^2r=N^{2-1/s}\rightarrow0$ as $N\rightarrow \infty$ whenever $s<1/2$, we may choose $N$ so large that
\[
    C_{M,\eta}\bigl(\tau+2\tau^2\bigr)
    \le\frac12.
\]
Then
\[
    X_m
    \le
    C_{M,\eta}
    +
    \frac12X_{m-1}.
\]
Together with the bound for $X_1$, iteration gives
\[
    X_m\le C_{M,\eta},
    \qquad 1\le m\le n.
\]
Equivalently,
\[
|\Lambda_{m,d}(p)|
\le
C_{M,\eta}N\tau^d,
\qquad
1\le d\le M,
\]
for every $1\le m\le n$ and every $p\in K_m$.

The proof proceeds by induction on $1 \le m \le M-1$, as follows. For $m=1$ (which corresponds to choosing a single quadratic $P_{2,p}$), we observe that Lemma \ref{lem:higher-basic-error} (with each $Q_p \equiv \mathrm{Id})$ gives us that 
	\begin{align*}
		\Delta_1(k) + \Delta_2(k) + \Delta_3(k) = (N-1) \log \rho_p + b_* + \beta_k + R_{1,p}(q_k) + E_{2,p}(q_k)
	\end{align*}
	with $\| R_{1,p} \|_{\mathrm{coeff}} \lesssim_{M} N \tau = N^3 r$, and $ \left \lvert {E_{2,p}}  \right \rvert    \lesssim_{M, \eta } N \tau^2 = N^5 r^2$. The reader should compare this estimate with Lemma \ref{lem:basic-error}; the additional information we have gained, is that the low order error in $N^3 r$ is a polynomial in $q_k$. By Lemma \ref{lem:L_prop}, we may choose a polynomial $P_{2,p}$ of degree at most $2$ for which 
	\begin{align*}
		\sum_{k=1}^N P_{2,p}(q_k) = 0, \; \; \; LP_{2,p} = -R_{1,p},  \; \; \; \text{ and }  \| P_{2,p} \|_{C^1([-1,1])} \lesssim_M N^{-1} \| R_{1,p} \|_{\mathrm{coeff}} \lesssim_M \tau.
	\end{align*}
	Notice that we may guarantee the first cancellation condition above, since $L(\mathrm{const}) = 0$. Once again applying Lemma \ref{lem:higher-basic-error}, but now with $Q_p(t) = t + P_{2,p}(t)$, (which indeed satisfies \eqref{eq:new-correction-bounds} and \eqref{eq:new-correction-normalization}, by the above), we then obtain the new errors
	\begin{align*}
		\Delta_1(k) & + \Delta_2(k) + \Delta_3(k) \\
		& = (N-1) \log \rho_p + b_* + \beta_k + L P_{2,p}(q_k) + R_{1,p}(q_k)   + E_{2,p}(q_k) \\
		& = (N-1) \log \rho_p + b_* +\beta_k  +  E_{2,p}(q_k),
	\end{align*}
	by the choice of $P_{2,p}$, and from the fact that $R_{1,p}$ is a linear polynomial which is independent of $P_{2,p}$. Moreover, that same Lemma yields the bounds $  \left \lvert {E_{2,p}(q_k)}  \right \rvert    \lesssim_{M, \eta} N \tau^2$. This is exactly the desired base case for $m = 1$. 
	
	Assume now that for some $1 \le m \le M-2$, we have chosen the polynomials $P_{2,p}, \dotsc, P_{m+1, p}$ in such a way that for each $1 \le j \le m$, 
	\begin{align}\label{eq:desired_bds}
		\sum_{k=1}^{N} P_{j+1, p}(q_k) = 0, \; \; \; L P_{j+1,p} = - R_{j,p}, \; \; \; \text{ and } \| P_{j+1,p}  \|_{C^1([-1,1])} \lesssim_M \tau^j.
	\end{align}
	Here, $R_{1,p} \in \mathcal{P}_1$ is as in the base case, and for $j \ge 2$, $R_{j,p} \in \mathcal{P}_{j}$ is the remainder polynomial given by Lemma \ref{lem:higher-basic-error} with the deformation $Q_p(t) = t + P_{2,p}(t) + \cdots +  P_{j, p}(t)$. 
	Since each $R_{j,p}$ depends \textit{just} on the polynomials $P_{\ell, y}$ for $y \in K_{n-1}$ and $1 \le \ell \le j$, we have that $R_{j, p} \in \mathcal{P}_{j}$ is also the same remainder polynomial as in Lemma \ref{lem:higher-basic-error} for the deformation $Q_p(t) = t + P_{2,p}(t) + \cdots + P_{m+1,p} (t) + P_{m+2, p}(t)$ whenever $P_{m+2,p}(t)$ has degree at most $m+2$ and satisfies $\| P_{m+2,p} \|_{C^1([-1,1])} \lesssim_M \tau^{m+1}$. In particular, if we apply Lemma \ref{lem:higher-basic-error} for this deformation, we obtain the error	 
	\begin{align*}
		\Delta_1(k) & + \Delta_2(k) + \Delta_3(k) \\
		& = (N-1) \log \rho_p + b_* + \beta_k + \sum_{j=1}^{m+1} \left( LP_{j+1,p}(q_k) + R_{j,p}(q_k) \right) + E_{m+2, p}(q_k) \\
		& = (N-1) \log \rho_p + b_* + \beta_k + LP_{m+2,p}(q_k) + R_{m+1, p}(q_k)   +  E_{m+2, p}(q_k),
	\end{align*}	
	since the sum over $1 \le j \le m$ cancels exactly. Using Lemma \ref{lem:L_prop}, we then choose for each $p \in K_{n-1}$ some polynomial $P_{m+2, p} \in \mathcal{P}_{m+2}$ satisfying
$$
		\sum_{k=1}^N P_{m+2,p}(q_k) = 0, \qquad LP_{m+2,p} = -R_{m+1,p},$$
and		
		   $$ \| P_{m+2,p} \|_{C^1([-1,1])} \lesssim_M N^{-1} \| R_{m+1,p} \|_{\mathrm{coeff}} \lesssim_M \tau^{m+1},
$$
	where the last inequality is a consequence of Lemma \ref{lem:higher-basic-error}. This proves the induction claim on the choice of the polynomials $P_{j+1, p}$. Moreover, from the above inequality, we deduce also that with the choice of deformation $Q_p(t) = t + P_{2,p}(t) + \cdots + P_{m+2,p}(t)$, the error satisfies
	\begin{align*}
		\Delta_1(k) + \Delta_2(k) + \Delta_3(k) = (N-1) \log \rho_p + b_* + \beta_k + E_{m+2,p}(q_k),
	\end{align*}
	which for $m+1 = M-1$, is the desired conclusion of our Lemma. What is left to check are \eqref{eq:renormalized-Q-derivative} and \eqref{eq:renormalized-barycenter}, but these are clear for $N$ large by our choice of the polynomials $P_{j+1, p}$ and the bounds \eqref{eq:desired_bds}.

Fix $s \in (0,1/2)$ and $\eta >0$ small. Recall that for $N \ge 1$, we set $N r^s = 1$ and $\tau = N^2 r$. Since $s < 1/2$, notice that we may choose some fixed $M \ge 2$ large enough so that 
\begin{equation}
	\label{eq:finite-order-smallness}
	C_MN\tau^M  = C_M N^{2M+1} r^M = C_M N^{1 + M (2 -1/s)} \le \eta,
\end{equation} 
for all $N$ large enough depending on $M$ and $\eta$. It suffices to choose, for example, $M > s/(1-2s)$. In what follows, we will choose $N$ large enough, depending on $s$ and $\eta$ without explicitly stating, so as to apply our previous Lemmas where needed.



At generation one, choosing $a_0(0) = 1$, we may write for $x = q_k \in K_1$,
\begin{align*}
	g_1(x) = g_1(q_k) = N^{-1} \sum_{j \ne k} \log  \left \lvert {q_k -q_j}  \right \rvert    = \frac{b_k}N = \frac{b^*}N + \frac{\beta_k}N
\end{align*}
With the constant $c_1 = b^*/N$, we see that $e_1(x) = e_1(q_k) = 0$.
                   
               
                                     
             
                                                       
Of course the first choice, $a_0(0) =1$ satisfies $1 \le a_0(0) \lesssim_\eta N$.

\label{thm:main-general-new}
	For every $0 < s< 1/2$, there exists an $s$-Ahlfors regular compact set $K \subset {\mathbb R} \subset {\mathbb C}$ so that the harmonic measure in ${\mathbb C} \setminus K$ with pole at infinity $\omega^\infty$ is equivalent to $\mathcal{H}^s|_K$, i.e., there is some $C \ge 1$ so that 
	\begin{align*}
		 C^{-1} \, \mathcal{H}^s|_K(F)\le \omega^\infty(F) \le C \mathcal{H}^s|_K(F) \text{ for all } F \subset K \text{ Borel.}
	\end{align*}
	In particular, $\dim \omega^\infty = s$.
