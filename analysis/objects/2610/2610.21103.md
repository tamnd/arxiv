<!-- assumption-1: body_md -->
\label{ass:McDiarmid's}
There exists $p\in(1,2]$ such that
\begin{equation*}
V\coloneqq
\left\|
\sum_{k=1}^n
\|D_k(\bm{X})\|_{L_p}^{p}(\bm{X})
\right\|_{L_{\infty}(P_{\bm{X}})}
=
\left\|
\sum_{k=1}^n
\mathbb{E}^{(k)}\!\left[D_k(\bm{X})(\bm{X})^p\right]
\right\|_{L_{\infty}(P_{\bm{X}})}
<\infty.
\end{equation*}
<!-- end assumption-1 -->
<!-- theorem-1: body_md -->
\label{thm:heavy-tail}
Under \Cref{ass:McDiarmid's}, for every $t>0$,
\begin{equation}\label{eq:tail-p}
\mathbb P\!\left(f(\bm{X})-\mathbb{E}[f(\bm{X})]\ge t\right)
\le
\exp\!\left(
-\frac{p-1}{p}
\frac{t^{p/(p-1)}}{V^{1/(p-1)}}
\right).
\end{equation}
If $V=0$, then $f(\bm{X})$ is almost surely constant and the deviation probability is zero.

\noindent Equivalently, for every $\delta\in(0,1)$, with probability at least $1-\delta$,
\begin{equation}\label{eq:confidence-heavy-tail}
f(\bm{X})-\mathbb{E}[f(\bm{X})]
\le
V^{1/p}
\left(
\frac{p}{p-1}\ln\frac1\delta
\right)^{(p-1)/p}.
\end{equation}
<!-- end theorem-1 -->
<!-- proposition-1: body_md -->
\label{prop:McDiarmid's}
Under \Cref{ass:McDiarmid's}, $\mathbb{E}[|f(\bm{X})|^p]<\infty$. However, for every $q>p$, there exist independent inputs and a function $f$ satisfying this assumption with $\mathbb{E}[|f(\bm{X})|^q]=\infty$. Specifically, let $X_1,\ldots,X_n$ be i.i.d.\ Pareto random variables with scale parameter $1$ and shape parameter $q$, and set
$
f(\bm{X})=-\frac1n\sum_{k=1}^n X_k.
$
Then \Cref{ass:McDiarmid's} holds with exponent $p$, while the $q$-th absolute moment of $f(\bm{X})$, with $q>p$, is infinite.
<!-- end proposition-1 -->
<!-- remark-1: body_md -->
\label{rmk:comparison}
    The General $f$ block in \Cref{tab:comparison} compares our assumption and results with other generalizations of McDiarmid's inequality. Existing works assume that $f_k(\bm{X})(x)$ has a bounded norm. We instead assume that $D_k(\bm{X})(x)$ has a bounded $L_p$-norm, which allows $f$ to have infinite variance while requiring $f$ to be bounded above. In exchange for this restriction, our bound on the upper tail decays exponentially in $t$. In contrast, the bound of \citet[Theorem 2.6]{lei2026stability}, the only existing result, to our knowledge, that allows $f$ to have infinite variance, decays only polynomially.
<!-- end remark-1 -->
<!-- proposition-2: body_md -->
\label{thm:reverse-kl}
Under the assumptions $h(\bm{X})\le0\text{ a.s.}$ and
  $\mathbb E[|h(\bm{X})|]<\infty$,
\begin{equation*}
  D_{\mathrm{KL}}(P_{\bm{X}}\Vert P_h)
  \le
  \operatorname*{ess\,sup}_{x\sim P_{\bm{X}}}
  \sum_{k=1}^n
  D_{\mathrm{KL}}\!\left(P_{X_k}\Vert P_{h,k}^{x_{-k}}\right).
\end{equation*}
Equivalently,
\begin{equation*}
  \ln\mathbb E[e^{h(\bm{X})}]-\mathbb E[h(\bm{X})]
  \le
  \left\|
    \sum_{k=1}^n
    \left(
      \ln\mathbb E^{(k)}[e^{h(\bm{X})}]
      -\mathbb E^{(k)}[h(\bm{X})]
    \right)
  \right\|_{L_{\infty}(P_{\bm{X}})}.
\end{equation*}

<!-- proofof-1: body_md -->
For $u\ge0$ and $1<p\le2$,
\begin{equation}\label{eq:scalar}
  e^{-u}\le1-u+\frac{u^p}{p}
\end{equation}
as shown in \citet[Lemma B.8]{behnamnia2025logsumexponential}. Indeed, the two sides are equal at $u=0$ and 
\[
\frac{d}{du}\left(1-u+\frac{u^p}{p}-e^{-u}\right)=-1+u^{p-1}+e^{-u}\ge0.
\]
For $0\le u\le1$, we use $u^{p-1}\ge u$ and $e^{-u}\ge1-u$. For $u\ge1$, we use $u^{p-1}\ge1$. 

Fix $\beta>0$. Since $f(\bm{X})=\sup_kf(\bm{X})-D_k(\bm{X})(\bm{X})$ and $\sup_kf(\bm{X})$ does not depend on $X_k$,
\begin{align}
  \ln\mathbb E^{(k)}[e^{\beta f(\bm{X})}]
  -\beta\mathbb E^{(k)}[f(\bm{X})]
  &=
  \ln\mathbb E^{(k)}[e^{-\beta D_k(\bm{X})(\bm{X})}]
  +\beta\mathbb E^{(k)}[D_k(\bm{X})(\bm{X})]\notag\\
  &\le
  \frac{\beta^p}{p}
  \mathbb E^{(k)}[D_k(\bm{X})(\bm{X})^p]\text{ almost surely}.
  \label{eq:local-heavy-tail}
\end{align}
For the last inequality \eqref{eq:local-heavy-tail}, we apply \eqref{eq:scalar} with $u=\beta D_k(\bm{X})(\bm{X})\ge0$ a.s. and take expectations
\begin{equation*}
  \mathbb{E}^{(k)}\big[e^{-\beta D_k(\bm{X})(\bm{X})}\big]\le1-\beta\,\mathbb{E}^{(k)}[D_k(\bm{X})(\bm{X})]+\frac{\beta^p}{p}\,\mathbb{E}^{(k)}[D_k(\bm{X})(\bm{X})^p].
\end{equation*}
The left-hand side is positive, so the right-hand side is also positive. Since $\ln y\le y-1$ for $y>0$, 
\begin{align*}
    \ln\mathbb E^{(k)}[e^{-\beta D_k(\bm{X})(\bm{X})}]&\le\ln\left(1-\beta\,\mathbb{E}^{(k)}[D_k(\bm{X})(\bm{X})]+\frac{\beta^p}{p}\,\mathbb{E}^{(k)}[D_k(\bm{X})(\bm{X})^p]\right)\\
    &\le-\beta\,\mathbb{E}^{(k)}[D_k(\bm{X})(\bm{X})]+\frac{\beta^p}{p}\,\mathbb{E}^{(k)}[D_k(\bm{X})(\bm{X})^p].
\end{align*}
Adding $\beta\mathbb{E}^{(k)}[D_k(\bm{X})(\bm{X})]$ to both sides proves \eqref{eq:local-heavy-tail}. By \Cref{prop:bound}, there is a constant $b$ with $\beta (f-b)\le0$ a.s. Applying the reverse KL tensorization (\Cref{thm:reverse-kl}) to $h=\beta (f-b)$,
\begin{align*}
    \ln\mathbb E[e^{\beta(f(\bm{X})-\mathbb E[f(\bm{X})])}]&=
\ln\mathbb E[e^{\beta\left((f(\bm{X})-b)-\mathbb E[f(\bm{X})-b]\right)}]\\
  &\le\Big\|\sum_{k=1}^n\Big(\ln\mathbb{E}^{(k)}\big[e^{\beta f(\bm{X})}\big]-\beta\,\mathbb{E}^{(k)}[f(\bm{X})]\Big)\Big\|_{L_{\infty}(P_{\bm{X}})}\\
  &\le\frac{\beta^p}{p}\Big\|\sum_{k=1}^n\mathbb{E}^{(k)}\big[D_k(\bm{X})(\bm{X})^p\big]\Big\|_{L_{\infty}(P_{\bm{X}})}
  =\frac{V\beta^p}{p}.
\end{align*}
The last inequality is by \eqref{eq:local-heavy-tail}. Combined with Chernoff's inequality,
\begin{equation}\label{eq:reverse_chernoff}
  \mathbb P(f(\bm{X})-\mathbb E[f(\bm{X})]\ge t)
  \le
   \inf_{\beta>0}
  \exp\!\left(-\beta t+\ln\mathbb E[e^{\beta(f(\bm{X})-\mathbb E[f(\bm{X})])}]\right)
  \le
  \inf_{\beta>0}
  \exp\!\left(-\beta t+\frac{V\beta^p}{p}\right).
\end{equation}
If $V=0$, $f$ is almost surely constant, so the tail probability is zero. For $V>0$, taking $\beta=(t/V)^{1/(p-1)}$ proves \eqref{eq:tail-p}. The equivalent form \eqref{eq:confidence-heavy-tail} follows by solving for $t$.

<!-- corollary-1: body_md -->
\label{col:gen_empirical}
Assume $\overline M_n>0$. For every $t>0$, 
 
\begin{equation}\label{eq:mean-non-iid}
  \mathbb P\!\left(\overline X_n-\mathbb E[\overline X_n]\ge t\right)
  \le
  \exp\!\left(
    -\frac{p-1}{p}
    \frac{nt^{p/(p-1)}}{\overline M_n^{1/(p-1)}}
  \right).
\end{equation}
 
Equivalently, for every $\delta\in(0,1)$, with probability at least $1-\delta$,
\begin{equation}
\label{eq:non-iid-empirical-confidence}
  \overline X_n-\mathbb E[\overline X_n]
  \le \overline M_n^{1/p}
  \left(\frac{p}{p-1}\frac{\ln(1/\delta)}{n}\right)^{(p-1)/p}.
\end{equation}

<!-- corollary-3: body_md -->
\label{col:mom_rate}
Let $X_1,\ldots,X_n$ be i.i.d.\ copies of $X$. For $t>\left(\frac{p}{p-1}\right)^{\frac{p-1}{p}} M_p^{1/p}\left(\frac{1}{m}\ln\frac{2K}{K+1}\right)^{\frac{p-1}{p}},$
\begin{equation*}
  \mathbb P\!\left(\widehat\mu_{\mathrm{MoM}}-\mu\ge t\right)
  \le
  \exp\!\left(-K d\left(\frac{K+1}{2K} \,\middle\Vert\, \overline q_t\right)\right),
\end{equation*}
where $d(\alpha\Vert \beta)\coloneqq \alpha\ln\frac{\alpha}{\beta}+(1-\alpha)\ln\frac{1-\alpha}{1-\beta}$ is the binary KL divergence, and $\overline q_t
  \coloneqq\exp\!\left(
    -\frac{p-1}{p}
    \frac{mt^{p/(p-1)}}{M_p^{1/(p-1)}}
  \right)$. If $K\ge\frac{\ln(1/\delta)}{\ln(2/\sqrt3)}$, with probability at least $1-\delta$,
\begin{equation}\label{eq:mom-confidence}
  \widehat\mu_{\mathrm{MoM}}-\mu
  \le
  M_p^{1/p}
  \left(
    \frac{p\ln4}{p-1}\frac{K}{n}
  \right)^{(p-1)/p}.
\end{equation}

<!-- proposition-3: body_md -->
\label{prop:bound}
\Cref{ass:McDiarmid's}  implies that $\exists b\in\mathbb{R}~\text{s.t.}~f(\bm{X})\le b~\text{a.s.}$

<!-- proofof-2: body_md -->
Let $\bm{Y}=(Y_1,\ldots,Y_n)$ be an independent copy of $\bm{X}$. For $k=0,\ldots,n$, define $H^{(k)}\colon \mathcal X^n\times\mathcal X^n\to\mathcal X^n$ by
\[
H^{(k)}(x,y)
\coloneqq
(y_1,\ldots,y_k,x_{k+1},\ldots,x_n),
\qquad x,y\in\mathcal X^n .
\]
We note that $H^{(0)}(x,y)=x$ and $H^{(n)}(x,y)=y$.
 For simplicity, we write 
 \[s_k(x_{-k})\coloneqq
\operatorname*{ess\,sup}_{z\sim P_{X_k}} f(x_{-k},z).\] $s_k(x_{-k})$ is $\sup_k f(x)$ in the main body. We make the independence of $\sup_i f(x)$ and $x_k$ explicit. With this notation, we can rewrite $D_k(x)$ as $D_k(x)=s_k(x_{-k})-f(x)$.

For each $k$, the vectors $H^{(k-1)}(x,y)$ and $H^{(k)}(x,y)$
agree in every coordinate except the $k$-th coordinate. By writing $H^{(k)}_{-k}(x,y)$ to denote the elements of $H^{(k)}(x,y)$ except for the $k$-th element, $H^{(k)}_{-k}(x,y)=H^{(k-1)}_{-k}(x,y)$. Hence, for
almost every $(x,y)$, by the definition of $s_k$,
\begin{align*}
f(H^{(k-1)}(x,y))
&\leq s_k\!\left(H^{(k-1)}_{-k}(x,y)\right)=
s_k\!\left(H^{(k)}_{-k}(x,y)\right),\\
f(H^{(k)}(x,y))
&=
s_k\!\left(H^{(k)}_{-k}(x,y)\right)
-D_k(H^{(k)}(x,y)).
\end{align*}
By adding both sides,
\[
f(H^{(k-1)}(x,y))-f(H^{(k)}(x,y))
\leq D_k(H^{(k)}(x,y)).
\]
Summing over $k\in[n]$ gives
\[
f(x)-f(y)
\leq
\sum_{k=1}^n D_k(H^{(k)}(x,y)).
\]
Since $D_k\geq 0$ almost surely, taking the positive part yields, for almost every $(x,y)$,
\[
(f(x)-f(y))_+
\leq
\sum_{k=1}^n D_k(H^{(k)}(x,y)).
\]

We then define \[q_k(x_{-k})
\coloneqq
\mathbb E^{(k)}[D_k(\bm{X})^p|\bm{X}_{-k}=x_{-k}].\] 
Each $q_j(x_{-j})$ is nonnegative. Also, by \Cref{ass:McDiarmid's}, 
$\sum_{j=1}^n q_j(\bm{X}_{-j})\leq V$ $P_{\bm{X}}\text{-a.s.}$. It follows that, for every $k$,
\[
q_k(\bm{X}_{-k})
\leq
\sum_{j=1}^n q_j(\bm{X}_{-j})
\leq V
\qquad P_{\bm{X}}\text{-a.s.}
\]
Thus, by Fubini's theorem, for $P_{\bm{X}}$-almost every
$x$,
\begin{align*}
\mathbb E_{\bm{Y}}\!\left[D_k(H^{(k)}(x,\bm{Y}))^p\right]&=\mathbb E_{Y_1,\ldots,Y_{k-1}}\mathbb{E}_{Y_k}
 \left[
 D_k(H^{(k)}(x,\bm{Y}))^p
 \right]\\
&=
\mathbb E_{Y_1,\ldots,Y_{k-1}}
 \left[
 q_k(Y_1,\ldots,Y_{k-1},x_{k+1},\ldots,x_n)
 \right]
\leq V.
\end{align*}
Minkowski's inequality and monotonicity of $L_p$ norm now yields
\[
\left\|(f(x)-f(\bm{Y}))_+\right\|_{L_p(P_{\bm{Y}})}
\leq
\sum_{k=1}^n
\left\|D_k(H^{(k)}(x,\bm{Y}))\right\|_{L_p(P_{\bm{Y}})}
\leq nV^{1/p}.
\]
Equivalently,
\[
\mathbb E_{\bm{Y}}\!\left[(f(x)-f(\bm{Y}))_+^p\right]
\leq n^pV.
\]
Since $f(\bm{Y})$ is finite almost surely, there exists $M\in\mathbb R$
such that $\rho\coloneqq\mathbb P(f(\bm{Y})\leq M)>0$.
On the event $\{f(\bm{Y})\leq M\}$,
\[
(f(x)-f(\bm{Y}))_+\geq (f(x)-M)_+.
\]
It follows that, for $P_{\bm{X}}$-almost every $x$,
\begin{align*}
\rho(f(x)-M)_+^p
\leq
\mathbb E_{\bm{Y}}\!\left[(f(x)-f(\bm{Y}))_+^p\right]
\leq n^pV.
\end{align*}
Therefore,
\[
f(x)
\leq
M+n\left(\frac{V}{\rho}\right)^{1/p}
\]
for $P_{\bm{X}}$-almost every $x$. This proves that $f(\bm{X})$ is essentially
bounded above.

Let $\bm{Y}=(Y_1,\ldots,Y_n)$ be an independent copy of $\bm{X}$. Define random vectors $H^{(k)}$, for $k=0,\dots,~n$, taking values in $\mathcal X^n$ as 
\[
H^{(k)}
\coloneqq H^{(k)}(\bm{Y},\bm{X}) = 
(X_1,\ldots,X_k,Y_{k+1},\ldots,Y_n),
\qquad k=0,\ldots,n,
\]
so that $H^{(0)}=\bm{Y}$ and $H^{(n)}=\bm{X}$. Then, 
\[
f(\bm{X})-f(\bm{Y})
=
\sum_{k=1}^n\Delta_k,~~\text{where}~~
\Delta_k\coloneqq f(H^{(k)})-f(H^{(k-1)}).
\]

Fix $k$ and write the elements of $H^{(k)}$ except for the $k$-th element as 
\[
\bm{U}_k\coloneqq(X_1,\ldots,X_{k-1},Y_{k+1},\ldots,Y_n).
\]
Conditionally on $\bm{U}_k$, the variables $X_k$ and $Y_k$ are
independent and have the same distribution. Define a random variable
\[
s_k(\bm{U}_k)
\coloneqq
\operatorname*{ess\,sup}_{z\sim P_{X_k}} f(\bm{U}_k,z).
\]
With this notation, we can write $D_k(\bm{U}_k,X_k)$ as $D_k(\bm{U}_k,X_k)=s_k(\bm{U}_k)-f(\bm{U}_k,X_k)$.\\
Since both $f(\bm{U}_k,X_k)$ and $f(\bm{U}_k,Y_k)$ are at most $s_k(\bm{U}_k)$ almost surely, by triangle inequality,
\begin{align*}
|\Delta_k|
&=
|f(\bm{U}_k,X_k)-f(\bm{U}_k,Y_k)| \\
&\leq
\bigl(s_k(\bm{U}_k)-f(\bm{U}_k,X_k)\bigr)
+
\bigl(s_k(\bm{U}_k)-f(\bm{U}_k,Y_k)\bigr) \\
&=
D_k(\bm{U}_k,X_k)+D_k(\bm{U}_k,Y_k).
\end{align*}
Minkowski's inequality therefore gives
\begin{align*}
\left(
\mathbb E\!\left[|\Delta_k|^p\mid \bm{U}_k\right]
\right)^{1/p}
&\leq
\left(
\mathbb E\!\left[D_k(\bm{U}_k,X_k)^p\mid \bm{U}_k\right]
\right)^{1/p} +\left(
\mathbb E\!\left[D_k(\bm{U}_k,Y_k)^p\mid \bm{U}_k\right]
\right)^{1/p} \\
&=
2\left(
\mathbb{E}^{(k)}[D_k(\bm{X})^p](\bm{U}_k)
\right)^{1/p}.
\end{align*}
Since $\bm{U}_k$ has the same distribution as $\bm{X}_{-k}$,
\begin{align*}
\left(
\mathbb E\!\left[|\Delta_k|^p\mid \bm{U}_k\right]
\right)^{1/p}=
\left(
\mathbb E\!\left[|\Delta_k|^p\mid \bm{X}_{-k}\right]
\right)^{1/p}
\leq
2\mathbb{E}^{(k)}[D_k(\bm{X})^p\mid\bm{X}_{-k}]
^{1/p}
=2\mathbb{E}^{(k)}[D_k(\bm{X})^p]
^{1/p}.
\end{align*}
By non-negativity, for every $k$,
\[
\mathbb{E}^{(k)}[D_k(\bm{X})^p]
\leq
\sum_{j=1}^n \mathbb{E}^{(j)}[D_j(\bm{X})^p]
\leq V
\qquad P_{\bm{X}}\text{-a.s.},
\]
Therefore, we obtain
\[
\|\Delta_k\|_{L_p(P_{\bm{X}}\otimes P_{\bm{Y}})}\leq 2V^{1/p}.
\]
Hence, by Minkowski's inequality,
\begin{equation}\label{eq:independent-copy-Lp}
\|f(\bm{X})-f(\bm{Y})\|_{L_p(P_{\bm{X}}\otimes P_{\bm{Y}})}
\leq
\sum_{k=1}^n\|\Delta_k\|_{L_p(P_{\bm{X}}\otimes P_{\bm{Y}})}
\leq
2nV^{1/p}.
\end{equation}

We now show that \eqref{eq:independent-copy-Lp} implies
$f(\bm{X})\in L_p(P_{\bm{X}})$. Since $f(\bm{Y})$ is finite almost surely, there exists
$M<\infty$ such that $\rho\coloneqq\mathbb P\bigl(|f(\bm{Y})|\leq M\bigr)>0$.
On the event $\{|f(\bm{Y})|\leq M\}$,
\[
|f(\bm{X})-f(\bm{Y})|
\geq
\bigl(|f(\bm{X})|-M\bigr)_+.
\]
Using the independence of $\bm{X}$ and $\bm{Y}$, we obtain
\begin{align*}
\mathbb E|f(\bm{X})-f(\bm{Y})|^p
&\geq
\mathbb E\!\left[
\mathbf 1_{\{|f(\bm{Y})|\leq M\}}
\bigl(|f(\bm{X})|-M\bigr)_+^p
\right] \\
&=
\rho\,
\mathbb E\bigl(|f(\bm{X})|-M\bigr)_+^p.
\end{align*}
The left-hand side is finite by
\eqref{eq:independent-copy-Lp}, so 
\[
\mathbb E\bigl(|f(\bm{X})|-M\bigr)_+^p<\infty.
\]
Since $|f(\bm{X})|\leq M+\bigl(|f(\bm{X})|-M\bigr)_+$, it follows that 
\[
\mathbb E|f(\bm{X})|^p<\infty.
\]
Thus $f(\bm{X})\in L_p(P_{\bm{X}})$, and therefore for every $0<r\leq p$,
\[
f(\bm{X})\in L_r(P_{\bm{X}}).
\]

Now we prove that $f(\bm{X})$ can have an infinite variance. For $1<p<q$, let $X$ have the Pareto distribution with scale parameter one and shape parameter $q$, so
\[
  \mathbb P(X>t)=t^{-q},\qquad t\ge 1.
\]
Let  \(X_1,\ldots,X_n\) be i.i.d.\ copies of the Pareto random variable \(X\). We consider $f(\bm{X})=-\frac{1}{n}\sum_{i=1}^n X_i$. Then, for any fixed \(x\in\mathcal X^n\), we have
\[
f(x_1,\ldots,x_{k-1},z,x_{k+1},\ldots,x_n)
=
-\frac{1}{n}
\left(
\sum_{i\ne k}x_i+z
\right).
\]
Since \(z\ge 1\) almost surely and the above expression is decreasing in \(z\),
\[
\begin{aligned}
\sup_k f(x)
&=
\operatorname*{ess\,sup}_{z\sim P_{X_k}}
f(x_1,\ldots,x_{k-1},z,x_{k+1},\ldots,x_n) \\
&=
-\frac{1}{n}
\left(
\sum_{i\ne k}x_i+1
\right).
\end{aligned}
\]
Consequently,
\[
\begin{aligned}
D_k(\mathbf X)
&=
\sup_k f(\mathbf X)-f(\mathbf X) \\
&=
-\frac{1}{n}
\left(
\sum_{i\ne k}X_i+1
\right)
+
\frac{1}{n}\sum_{i=1}^n X_i =
\frac{X_k-1}{n}.
\end{aligned}
\]
Therefore,
\[
D_k(\mathbf X)^p
=
\frac{(X_k-1)^p}{n^p},
\]
and hence
\[
\mathbb E^{(k)}
\left[
D_k(\mathbf X)^p
\right]
=
\frac{1}{n^p}
\mathbb E\left[(X-1)^p\right]\quad P_{\bm{X}}\text{-a.s.}
\]
Since \(p<q\), the Pareto random variable \(X\) has a finite \(p\)-th
moment, and therefore
\[
\mathbb E[(X-1)^p]<\infty.
\]
It follows that
\[
\begin{aligned}
V
=
\left\|
\sum_{k=1}^n
\mathbb E^{(k)}
\left[
D_k(\mathbf X)^p
\right]
\right\|_{L_{\infty}(P_{\mathbf X})} =
\left\|
\frac{n}{n^p}
\mathbb E[(X-1)^p]
\right\|_{L_{\infty}(P_{\mathbf X})} =
n^{1-p}\mathbb E[(X-1)^p]
<\infty.
\end{aligned}
\]
Thus \Cref{ass:McDiarmid's} is satisfied.

On the other hand,
\[
|f(\mathbf X)|
=
\frac{1}{n}\sum_{i=1}^n X_i
\ge
\frac{X_1}{n},
\]
and therefore
\[
|f(\mathbf X)|^q
\ge
\frac{X_1^q}{n^q}.
\]
Taking expectations gives
\[
\mathbb E\left[|f(\mathbf X)|^q\right]
\ge
\frac{1}{n^q}
\mathbb E[X_1^q]
=
\frac{1}{n^q}
\mathbb E[X^q]
.
\]
For a Pareto random variable with tail index \(q\),
$\mathbb E[X^r]<\infty\Longleftrightarrow r<q$, and hence $\mathbb E[X^q]=\infty$. Therefore,
\[
\mathbb E\left[|f(\mathbf X)|^q\right]=\infty.
\]
In particular, if $1<p<q\le 2$, 
\[
\mathbb E\left[f(\mathbf X)^2\right]=\infty.
\]
Hence, for \(1<p<q\le 2\), we obtain an example for which $V<\infty$ i.e. \Cref{ass:McDiarmid's} is satisfied, while simultaneously $f(\mathbf X)\notin L_q(P_{\bm{X}})$ and $f(\mathbf X)\notin L_2(P_{\bm{X}}).$

For $S\subseteq[n]$, we write $X_S\coloneqq\{X_i:i\in S\}$.
   
   Define the projection $\Pi_S$ of integrable $\sigma(X_{[n]})$-measurable random variables onto $\sigma(X_S)$-measurable ones, and a $\sigma(X_S)$-measurable random variable $g_S$, by
\[
  \Pi_S\coloneqq\prod_{j\notin S}\mathbb E^{(j)},
  \qquad
  g_S\coloneqq\Pi_S\!\left(\frac{e^{h(\bm{X})}}{Z_h}\right)=\mathbb E\!\left[\tfrac{e^{h(\bm{X})}}{Z_h}\,\Big|\,X_S\right],
  \qquad
  \text{where}\quad Z_h\coloneqq\mathbb{E}[e^{h(\bm{X})}]
  .

Using the tower property, 
\begin{equation}\label{later_density}
  \mathbb E[g_S]
  =\mathbb E\!\left[\mathbb E\!\left[\tfrac{e^{h(\bm{X})}}{Z_h}\,\Big|\,X_S\right]\right]
  =\mathbb E\!\left[\tfrac{e^{h(\bm{X})}}{Z_h}\right]=1.
\end{equation}
Also, the random variable $g_S$ is nonnegative. Thus, $g_S$ is a probability density with respect to $P_{\bm{X}}$. Since $P_{\bm{X}}$ is a product measure,
\[
  \mathbb E^{(i)}\mathbb E^{(j)}=\mathbb E^{(j)}\mathbb E^{(i)}.
\]
Moreover,
\[
  \mathbb E^{(k)}\mathbb E^{(k)}
  =
  \mathbb E^{(k)},
\]
because for any measurable and integrable $u$, $\mathbb E^{(k)}[u]$ does not depend on $x_k$.  Therefore, the
operators $\mathbb E^{(1)},\ldots,\mathbb E^{(n)}$ are commuting
projections. Consequently, for $S\subseteq[n]$, the operator $\Pi_S=\prod_{j\notin S}\mathbb E^{(j)}$ is independent of the order of the factors.  
This leads to
\[
  \Pi_{[n]}=I,
  \qquad
  \Pi_\varnothing u=\mathbb E[u].
\]
In particular,
\begin{equation}\label{later_ent}
    g_{[n]}=\frac{e^{h(\bm{X})}}{Z_h},
  \qquad
  g_\varnothing=1.
\end{equation}
The projections also satisfy
\[
  \Pi_S\Pi_T=\Pi_{S\cap T},
\]
because applying both operators integrates all the coordinates in $S^c\cup T^c=(S\cap T)^c$. If $k\notin S$, then the commutation of the coordinate projections gives
\begin{align}\label{later_eq}
  \mathbb E^{(k)}[g_{S\cup\{k\}}]=
  \mathbb E^{(k)}
  \Pi_{S\cup\{k\}}
  \left(\frac{e^{h(\bm{X})}}{Z_h}\right)
  =
  \prod_{j\notin S}\mathbb E^{(j)}
  \left(\frac{e^{h(\bm{X})}}{Z_h}\right)
  =
  \Pi_S\left(\frac{e^{h(\bm{X})}}{Z_h}\right)
  =g_S.
\end{align}

We now record some identities for KL divergences. Write $\rho\coloneqq e^{h(\bm{X})}/Z_h$, so $P_h=\rho\,P_{\bm{X}}$. We have 
\[
\mathbb E[\rho]=1,\quad\ln\rho=h(\bm{X})-\ln Z_h,\quad\rho>0~\text{a.s.}
\]
By definition, $D_{\mathrm{KL}}(P_h\Vert P_{\bm{X}})=\mathbb E_{P_h}[\ln\frac{dP_h}{dP_{\bm{X}}}]=\mathbb E[\rho\ln\rho]$. 
Substituting $\ln\rho=h(\bm{X})-\ln Z_h$,
\[
D_{\mathrm{KL}}(P_h\Vert P_{\bm{X}})=\mathbb E[\rho\,h(\bm{X})]-\ln Z_h\,\mathbb E[\rho]=\mathbb E[\rho\,h(\bm{X})]-\ln Z_h.
\]
By definition, $D_{\mathrm{KL}}(P_{\bm{X}}\Vert P_h)=\mathbb E_{P_{\bm{X}}}[\ln\frac{dP_{\bm{X}}}{dP_h}]=\mathbb{E}[\ln(1/\rho)]$. Again, by $\ln\rho=h(\bm{X})-\ln Z_h$,
\[
D_{\mathrm{KL}}(P_{\bm{X}}\Vert P_h)=\mathbb E[\ln Z_h-h(\bm{X})]=\ln Z_h-\mathbb E[h(\bm{X})].
\]
Using these identities,
\begin{align} \notag
  \mathbb E
  \left[
    \left(\frac{e^{h(\bm{X})}}{Z_h}-1\right)h(\bm{X})
  \right]
  -D_{\mathrm{KL}}(P_h\Vert P_{\bm{X}})
  &=
  \mathbb{E}[\rho h(\bm{X})-h(\bm{X})]-(\mathbb E[\rho\,h(\bm{X})]-\ln Z_h)\\ \notag
  &=
  \ln Z_h-\mathbb E[h(\bm{X})]\\ \label{rough_key}
  &=
  D_{\mathrm{KL}}(P_{\bm{X}}\Vert P_h).
\end{align}

Our goal is to bound the reverse KL in \eqref{rough_key}. Our strategy is to first bound $\mathbb E
  \left[
    \left(\frac{e^{h(\bm{X})}}{Z_h}-1\right)h(\bm{X})
  \right]$. Then, we subtract the forward KL term $D_{\mathrm{KL}}(P_h\Vert P_{\bm{X}})$ from both sides to obtain a bound on the reverse KL $D_{\mathrm{KL}}(P_{\bm{X}}\Vert P_h)$.

The first step is to decompose $\mathbb E
  \left[
    \left(\frac{e^{h(\bm{X})}}{Z_h}-1\right)h(\bm{X})
  \right]$.
Fix a permutation $\pi$ of $\{1,\dots,n\}$. We define $T_i\coloneqq\{\pi(1),\dots,\pi(i)\}$, with $T_0\coloneqq\varnothing$. By conditioning successively on the independent random variables and using a telescoping sum,
\begin{equation*}
    \frac{e^{h(\bm{X})}}{Z_h}-1
    =\mathbb{E}\left[\frac{e^{h(\bm{X})}}{Z_h} \;\middle|\;X_{[n]}\right]-\mathbb{E}\left[\frac{e^{h(\bm{X})}}{Z_h} \;\middle|\;X_{\varnothing}\right]
    =\sum_{i=1}^{n}\mathbb{E}\left[\frac{e^{h(\bm{X})}}{Z_h} \;\middle|\;X_{T_i}\right]-\mathbb{E}\left[\frac{e^{h(\bm{X})}}{Z_h} \;\middle|\;X_{T_{i-1}}\right].
\end{equation*}
Denote the set of indices preceding $k$ in $\pi$ by $S_k(\pi)$, so taking $k=\pi(i)$ gives $S_{\pi(i)}(\pi)=T_{i-1}$. Using this notation,
\begin{equation*}
    \frac{e^{h(\bm{X})}}{Z_h}-1
    =\sum_{k=1}^{n}\mathbb{E}\left[\frac{e^{h(\bm{X})}}{Z_h} \;\middle|\;X_{S_k(\pi)\cup\{k\}}\right]-\mathbb{E}\left[\frac{e^{h(\bm{X})}}{Z_h} \;\middle|\;X_{S_k(\pi)}\right].
\end{equation*}
Now let $\pi$ be uniform on the symmetric group $\mathfrak{S}_n\coloneqq \{\text{all permutations of $\{1,\ldots,n\}$}\}$. Averaging over $\pi$ gives
\begin{equation}\label{arc_summand}
    \frac{e^{h(\bm{X})}}{Z_h}-1
    =\sum_{k=1}^n\sum_{S\subseteq[n]\setminus\{k\}}\mathbb P \bigl(S_k(\pi)=S\bigr)
    \left(\mathbb{E}\left[\frac{e^{h(\bm{X})}}{Z_h} \;\middle|\;X_{S\cup\{k\}}\right]-\mathbb{E}\left[\frac{e^{h(\bm{X})}}{Z_h} \;\middle|\;X_{S}\right]\right).
\end{equation}
For fixed $k\in[n]$ and fixed $S\subseteq[n]\setminus\{k\}$,
\begin{align*}
    \mathbb P&\bigl(S_k(\pi)=S\bigr)\\
  &=\frac{\#\text{ all the permutations of }\{1,\dots,n\} \text{ such that the $(|S|+1)$-th element is $k$ and $S_k(\pi)=S$}}{\#\text{ all the permutations of }\{1,\dots,n\}}\\
  &=\frac{|S|!\,(n-1-|S|)!}{n!}
\end{align*}
where the probability is over the random permutation $\pi$. As a check, for fixed $k$,
\[
\sum_{S\subseteq[n]\setminus\{k\}}\mathbb P\bigl(S_k(\pi)=S\bigr)
=
\sum_{l=0}^{n-1}\sum_{S\subseteq[n]\setminus\{k\},~|S|=l}\frac{1}{n}\binom{n-1}{l}^{-1}=1.
\]
For $k\in[n]$ and $S\subseteq[n]\setminus\{k\}$, define
\[
  w_{k,S}\coloneqq\frac{|S|!(n-1-|S|)!}{n!}=\mathbb P\bigl(S_k(\pi)=S\bigr).
\]
To express the summands in \eqref{arc_summand}, we define the random variables
\begin{align*}
  \ell_k
  &\coloneqq
  \sum_{S\subseteq[n]\setminus\{k\}}  \mathbb P\bigl(S_k(\pi)=S\bigr)\mathbb E\!\left[\tfrac{e^{h(\bm{X})}}{Z_h}\,\Big|\,X_S\right]
  =\sum_{S\subseteq[n]\setminus\{k\}}w_{k,S}g_S,\\
  b_k
  &\coloneqq
  \sum_{S\subseteq[n]\setminus\{k\}}  \mathbb P\bigl(S_k(\pi)=S\bigr)\mathbb E\!\left[\tfrac{e^{h(\bm{X})}}{Z_h}\,\Big|\,X_{S\cup\{k\}}\right]
  =\sum_{S\subseteq[n]\setminus\{k\}}w_{k,S}g_{S\cup\{k\}},\\
  a_k&\coloneqq b_k-\ell_k.
\end{align*}
Using this notation, from \eqref{arc_summand}, we obtain
\begin{align} \label{eq:ak-sum}
  \sum_{k=1}^n a_k=\frac{e^{h(\bm{X})}}{Z_h}-1.
\end{align}
Thus,
\begin{align*}
  \mathbb E
  \left[
    \left(\frac{e^{h(\bm{X})}}{Z_h}-1\right)h(\bm{X})
  \right]
  =\sum_{k=1}^n\mathbb E[(b_k-\ell_k)h(\bm{X})].
\end{align*}

The next step is to apply the conditional Donsker--Varadhan inequality to bound $\mathbb E[a_kh(\bm{X})]=\mathbb E[(b_k-\ell_k)h(\bm{X})]$. Fix $x_{-k}$. We apply the Donsker--Varadhan variational formula (\Cref{lem:DV} in Appendix) to the $k$-th
coordinate, with reference measure $P_{X_k}$, test function $z\mapsto h(x_{-k},z)$, and density $\rho\coloneqq b_k/\ell_k$. To confirm that $\rho$ is admissible, we can verify
\begin{equation*}
    \mathbb E^{(k)}
  \left[\frac{b_k}{\ell_k}\right]
  =
  \frac{\mathbb E^{(k)}[b_k]}{\ell_k}
  =\frac{\ell_k}{\ell_k}
  =1
\end{equation*}
since $\ell_k$ is $\sigma(\bm{X}_{-k})$-measurable and \eqref{later_eq} implies $\mathbb E^{(k)}[b_k]=\ell_k$.
Applying the Donsker--Varadhan variational formula (\Cref{lem:DV} in Appendix) with these choices gives, $P_{\bm{X}_{-k}}$-a.s.\ ,
\begin{equation*}\label{rough_varadhan}
    \mathbb E^{(k)}\left[\frac{b_k}{\ell_k}h(\bm{X})\right]
  -\mathbb E^{(k)}\left[\frac{b_k}{\ell_k}\ln\frac{b_k}{\ell_k}\right]
  \le\ln\mathbb E^{(k)}\big[e^{h(\bm{X})}\big].
\end{equation*}
Subtracting $\mathbb E^{(k)}[h(\bm{X})]$ from both sides gives, $P_{\bm{X}_{-k}}$-a.s.\ ,
\begin{align}
  \mathbb E^{(k)}
  \left[
    \left(\frac{b_k}{\ell_k}-1\right)h(\bm{X})
  \right]
  &\le
  \mathbb E^{(k)}
  \left[
    \frac{b_k}{\ell_k}
    \ln\frac{b_k}{\ell_k}
  \right]
  +\ln\mathbb E^{(k)}[e^{h(\bm{X})}]
  -\mathbb E^{(k)}[h(\bm{X})].
  \label{eq:gibbs-fibre}
\end{align}
Multiplying \eqref{eq:gibbs-fibre} by $\ell_k$ gives, $P_{\bm{X}_{-k}}$-a.s.\ ,
\begin{align*}

  \mathbb E^{(k)}
  \left[
    \left(\frac{b_k}{\ell_k}-1\right)h(\bm{X})
  \right]\le
  \ell_k
  \mathbb E^{(k)}
  \left[
    \frac{b_k}{\ell_k}
    \ln\frac{b_k}{\ell_k}
  \right]
  +
  \ell_k
  \left(
    \ln\mathbb E^{(k)}[e^{h(\bm{X})}]
    -\mathbb E^{(k)}[h(\bm{X})]
  \right).
\end{align*}
Since $\ell_k$ is $\sigma(\bm{X}_{-k})$-measurable, it can be moved inside
$\mathbb E^{(k)}$.  Hence, $P_{\bm{X}_{-k}}$-a.s.\ ,
\begin{align*}
  \ell_k
  \mathbb E^{(k)}

    \left(\frac{b_k}{\ell_k}-1\right)h(\bm{X})
  \right]
  &=
  \mathbb E^{(k)}

    \ell_k
    \left(\frac{b_k}{\ell_k}-1\right)h(\bm{X})
  \right]\\
  &=
  \mathbb E^{(k)}

    (b_k-\ell_k)h(\bm{X})
  \right]\\
  &=

Similarly, $P_{\bm{X}_{-k}}$-a.s.\ ,

  \right]
  =
  \mathbb E^{(k)}
  \left[
    \ell_k\frac{b_k}{\ell_k}
    \ln\frac{b_k}{\ell_k}
  \right]=
  \mathbb E^{(k)}
  \left[
    b_k\ln\frac{b_k}{\ell_k}

Consequently, $P_{\bm{X}_{-k}}$-a.s.\ ,
\begin{align*}
  \mathbb E^{(k)}[a_kh(\bm{X})]

    b_k\ln\frac{b_k}{\ell_k}

Taking expectation with respect to the remaining variables $\bm{X}_{-k}$,
or equivalently taking the full expectation with respect to $P_{\bm{X}}$, gives
\begin{align*}
  \mathbb E[a_kh(\bm{X})]
  &\le
  \mathbb E
  \left[
    b_k\ln\frac{b_k}{\ell_k}
  \right]+
  \mathbb E

    \ell_k
    \left(
      \ln\mathbb E^{(k)}[e^{h(\bm{X})}]
      -\mathbb E^{(k)}[h(\bm{X})]
    \right)
  \right].

By summing over $k$,
\begin{align}
 \mathbb E
  \left[
    \left(\frac{e^{h(\bm{X})}}{Z_h}-1\right)h(\bm{X})
  \right]
  \le

\end{align}

We then bound the first term of \eqref{arc_dv}. Using the property \eqref{later_density} and the non-negativity of $g_S$, we write $g_SP_{\bm{X}}$ for the probability measure on $\mathcal X^n$ given by $(g_SP_{\bm{X}})(A)\coloneqq\int_A g_S\,dP_{\bm{X}}$, so that $\frac{d(g_SP_{\bm{X}})}{dP_{\bm{X}}}=g_S$. Since $e^{h(\bm{X})}/Z_h>0$, each $g_S$ is strictly positive almost surely. Thus, each measure $g_SP_{\bm{X}}$ satisfies $P_{\bm{X}}\ll g_SP_{\bm{X}}$. This also defines $\ell_kP_{\bm{X}}$ and $b_kP_{\bm{X}}$ accordingly. We write $\ell_k P_{\bm{X}}$ for the probability measure on $\mathcal X^n$ given by $(\ell_k P_{\bm{X}})(A)\coloneqq\int_A \ell_k\,dP_{\bm{X}}$, so that $\frac{d(\ell_k P_{\bm{X}})}{dP_{\bm{X}}}=\ell_k$, and similarly for $b_k P_{\bm{X}}$. Using this notation, each summand in the first term of \eqref{arc_dv} is
\[
  \mathbb E
  \left[
    b_k\ln\frac{b_k}{\ell_k}
  \right]
  =
  D_{\mathrm{KL}}(b_kP_{\bm{X}}\Vert\ell_kP_{\bm{X}}).
\]
We bound this quantity by the joint convexity of the KL divergence,
\begin{equation}\label{eq:Ck}
  D_{\mathrm{KL}}(b_kP_{\bm{X}}\Vert\ell_kP_{\bm{X}})
  \le
  \sum_{S\subseteq[n]\setminus\{k\}}w_{k,S}
  D_{\mathrm{KL}}\!\left(g_{S\cup\{k\}}P_{\bm{X}}\Vert g_SP_{\bm{X}}\right).
\end{equation}
For $S\subseteq[n]\setminus\{k\}$, by applying the Pythagorean identity for conditional projections (\Cref{lem:pythagoras} in Appendix) with $\mathcal G=\sigma(X_S)$ and
$f=g_{S\cup\{k\}}$,
\begin{align}
  D_{\mathrm{KL}}\!\left(g_{S\cup\{k\}}P_{\bm{X}}\Vert g_SP_{\bm{X}}\right)
  &={\rm Ent}_{P_{\bm{X}}}(g_{S\cup\{k\}})
    -{\rm Ent}_{P_{\bm{X}}}(g_S),
  \label{eq:entropy-difference}
\end{align}
where the entropy ($\rm Ent$) is defined in \Cref{lem:pythagoras}.
Note that, we can apply \Cref{lem:pythagoras} because, by the tower property,
\[
  \mathbb E\!\left[g_{S\cup\{k\}}\mid X_S\right]
  =\mathbb E\!\left[\,\mathbb E\!\left[\tfrac{e^{h(\bm{X})}}{Z_h}\,\Big|\,X_{S\cup\{k\}}\right]\Big|\,X_S\right]
  =\mathbb E\!\left[\tfrac{e^{h(\bm{X})}}{Z_h}\,\Big|\,X_S\right]=g_S.
\]
(This is the same identity $\mathbb E^{(k)}[g_{S\cup\{k\}}]=g_S$ established earlier.) Since $h\le0$ a.s., $e^{h(\bm{X})}/Z_h\le Z_h^{-1}$ a.s. Conditional expectation preserves the bound, so $0<g_S\le Z_h^{-1}$ a.s. In addition, the function $x\mapsto x|\ln x|$ is bounded on $(0,Z_h^{-1}]$. Therefore, $\mathbb E[|g_S\ln g_S|]<\infty$. If moreover $S\subseteq T$, then $\ln g_S$ is $\sigma(X_S)$-measurable and $g_S=\mathbb E[g_T\mid X_S]$, so $\mathbb E[g_T|\ln g_S|]=\mathbb E[g_S|\ln g_S|]<\infty$. Consequently, for $S\subseteq T$,
\[
  D_{\mathrm{KL}}(g_TP_{\bm{X}}\Vert g_SP_{\bm{X}})
  =\mathbb E[g_T\ln g_T]-\mathbb E[g_T\ln g_S]<\infty .
\]
Summing \eqref{eq:Ck} over $k$ and substituting \eqref{eq:entropy-difference},
\begin{align*}
    \sum_{k=1}^nD_{\mathrm{KL}}(b_kP_{\bm{X}}\Vert\ell_kP_{\bm{X}})
  &\le\sum_{k=1}^n\sum_{S\subseteq[n]\setminus\{k\}}w_{k,S}
     \Bigl[{\rm Ent}_{P_{\bm{X}}}(g_{S\cup\{k\}})-{\rm Ent}_{P_{\bm{X}}}(g_S)\Bigr]\\
     &=\mathbb E_\pi\!\left[\sum_{k=1}^n
    \Bigl({\rm Ent}_{P_{\bm{X}}}\bigl(g_{S_k(\pi)\cup\{k\}}\bigr)
         -{\rm Ent}_{P_{\bm{X}}}\bigl(g_{S_k(\pi)}\bigr)\Bigr)\right]\\
        &=\mathbb E_\pi\!\left[{\rm Ent}_{P_{\bm{X}}}(g_{[n]})-{\rm Ent}_{P_{\bm{X}}}(g_\varnothing)\right]\\
        &={\rm Ent}_{P_{\bm{X}}}(g_{[n]})-{\rm Ent}_{P_{\bm{X}}}(g_\varnothing).
\end{align*}
By \eqref{later_ent}, we have
\begin{align*}
    &{\rm Ent}_{P_{\bm{X}}}(g_{[n]})
    =\mathbb{E}\left[\frac{e^{h(\bm{X})}}{Z_h}\ln\left(\frac{e^{h(\bm{X})}}{Z_h}\right)\right]-\mathbb{E}\left[\frac{e^{h(\bm{X})}}{Z_h}\right]\ln\left(\mathbb{E}\left[\frac{e^{h(\bm{X})}}{Z_h}\right]\right)=\mathbb{E}\left[\frac{e^{h(\bm{X})}}{Z_h}\ln\left(\frac{e^{h(\bm{X})}}{Z_h}\right)\right]\\
    &\qquad\quad\quad\quad=D_{\mathrm{KL}}(P_h\Vert P_{\bm{X}})\\
    &{\rm Ent}_{P_{\bm{X}}}(g_\varnothing)=\mathbb{E}\left[\ln(1)\right]-\ln(1)=0.
\end{align*}
Therefore,
\begin{align}\label{eq:Ck-sum}
  \sum_{k=1}^nD_{\mathrm{KL}}(b_kP_{\bm{X}}\Vert\ell_kP_{\bm{X}})
  \le D_{\mathrm{KL}}(P_h\Vert P_{\bm{X}}).
\end{align}

We now bound the second term of $\eqref{arc_dv}$.
For $T\subseteq[n]$, define
\[
  c_T
  \coloneqq\frac{|T|!(n-|T|)!}{(n+1)!},

\]
For $S\subseteq[n]\setminus\{k\}$,
\begin{equation}\label{later_third}
    c_S+c_{S\cup\{k\}}=\frac{|S|!(n-|S|)!}{(n+1)!}+\frac{(|S|+1)!(n-|S|-1)!}{(n+1)!}

\end{equation}

The proof of \Cref{col:mom_rate} relies on the next lemma.
\begin{lemma}\label{lem:mom}
Let $X_1,\ldots,X_n$ be i.i.d.\ copies of $X$. For $t>0$, define $\overline q_t\coloneqq\exp\!\left( -\frac{p-1}{p}\frac{mt^{p/(p-1)}}{M_p^{1/(p-1)}}\right).$\\
If $\overline q_t<\frac{K+1}{2K}$,
\begin{equation*}
  \mathbb P\!\left(\widehat\mu_{\mathrm{MoM}}-\mathbb{E}[X]\ge t\right)
  \le
  \exp\!\left(-K d((K+1)/2K \Vert\overline q_t)\right).
\end{equation*}
Also,
\begin{equation*}
  \mathbb P\!\left(
    \widehat\mu_{\mathrm{MoM}}-\mathbb{E}[X]
    \ge
    M_p^{1/p}
    \left(
      \frac{p\ln4}{p-1}\frac{K}{n}
    \right)^{(p-1)/p}
  \right)
  \le
  \left(\frac{\sqrt{3}}{2}\right)^K.
\end{equation*}
\end{lemma}

\begin{align}

  &\coloneqq\exp\!\left(
    -\frac{p-1}{p}
    \frac{mt^{p/(p-1)}}{M_p^{1/(p-1)}}
  \right).
\end{align}

  &\le
    \exp
  \left(
    -\frac{p-1}{p}
    \frac{mt^{p/(p-1)}}{M_p^{1/(p-1)}}
  \right) = \overline q_t.
\end{align*}

\begin{equation*}
\{\widehat\mu_{\mathrm{MoM}}-\mu\ge t\}
=\{\#\{j\in[K]: Y_j-\mu\ge t\}\ge k\}
=\bigcup_{\substack{S\subseteq[K]\\ |S|=k}}\bigcap_{j\in S}\{Y_j-\mu\ge t\}.
\end{equation*}

\begin{equation}
    \mathbb P \left(\widehat\mu_{\mathrm{MoM}}-\mu \geq t\right) = \mathbb P \left(\sum_{j=1}^K\mathbf 1_{\{Y_j-\mu\ge t\}}\ge K \frac{K+1}{2K}\right)\le \inf_{\lambda>0}\exp
  \left(
    -\lambda K\frac{K+1}{2K}
  \right)\prod_{j=1}^K
  \mathbb E
  \left[
    e^{\lambda\mathbf 1_{\{Y_j-\mu\ge t\}}}
  \right].
\end{equation}

\begin{align*}

  \left[
    e^{\lambda\mathbf 1_{\{Y_j-\mu\ge t\}}}
  \right]

  1+
  (e^\lambda-1)
  \mathbb P(Y_j-\mu\ge t)
  \le
  1-\overline q_t+\overline q_te^\lambda.
\end{align*}

\begin{align}
  \mathbb P
  \left(
    \widehat\mu_{\mathrm{MoM}}-\mu\ge t
  \right)
  \le
  \inf_{\lambda>0}

  \left[
    K
    \left(
      -\lambda \frac{K+1}{2K}

By the Chernoff bound and the Herbst argument \eqref{eq:Herbst},
    \begin{equation}\label{eq:ent_chernoff}
   \mathbb{P}(f(\bm{X})-\mathbb{E}[f(\bm{X})]\ge t)\le\inf_{\beta>0} \exp (\beta\int_0^\beta
\frac{D_{\mathrm{KL}}\!\left(P_{tf} \,\middle\|\, P_{\bm{X}}\right)}{t^2}\,dt -\beta t),
   \end{equation}
   where $P_{tf}$ is the tilted measure defined in \eqref{eq:tilted_measure}. For $i\in[n]$, we define the conditional counterpart of a tilted measure $P_h$ by tilting the $i$-th coordinate only. For fixed \(x_{-i}\in\mathcal{X}^n\), let \(P^{(i)}_h(\cdot\mid x_{-i})\)
be the probability measure on the \(i\)-th coordinate space $\mathcal{X}$ with
\begin{equation*}
\frac{dP^{(i)}_h(\cdot\mid x_{-i})}{dP_{X_i}}(x_i)
\coloneqq\frac{e^{h(x_i,x_{-i})}}{\mathbb{E}^{(i)}\bigl[e^{h}\bigr](x_{-i})} .
\end{equation*}
We write the KL divergence between conditional tilted measure and the marginal distribution as a function on $\mathcal{X}^{n-1}$ as
\begin{equation*}
D_{\mathrm{KL}}^{(i)}\!\left(h\right)(x_{-i})
\coloneqqD_{\mathrm{KL}}\!\left(P^{(i)}_h(\cdot\mid x_{-i})\,\middle\Vert\,P_{X_i}\right).
\end{equation*}
For the conditional KL divergence
averaged against the marginal of \(P_h\),
\begin{equation*}
D_{\mathrm{KL}}\!\left(P_{h,\,X_i\mid \bm{X}_{-i}}\,\middle\Vert\,P_{X_i}\,\middle|\,P_{h,\,\bm{X}_{-i}}\right)
\coloneqq\int D_{\mathrm{KL}}^{(i)}\!\left(h\right)(x_{-i})\,dP_{h,\,\bm{X}_{-i}}(x_{-i})
=\mathbb{E}_{P_h}\bigl[D_{\mathrm{KL}}^{(i)}\!\left(h\right)(\bm{X}_{-i})\bigr].
\end{equation*}
By subadditivity of entropy \citep[Theorem 4.22]{boucheron2013concentration}, \citep[Proposition 4.1]{posta1997spectral}, also known as KL tensorization,
\begin{equation*}
D_{\mathrm{KL}}(P_{t f}\Vert P_{\bm{X}})
\le
\sum_{i=1}^n
D_{\mathrm{KL}}\!\left(
P_{t f,\,X_i\mid \bm{X}_{-i}}\,\middle\Vert\,P_{X_i}\,\middle|\,P_{t f,\,\bm{X}_{-i}}
\right)
=
\mathbb{E}_{P_{t f}}
\left[
\sum_{i=1}^n
D_{\mathrm{KL}}^{(i)}\!\left(t f\right)
\right].
\end{equation*}
Denote a function $D_i$ on $\mathcal X^n$ defined as $D_i(x_1^n)=\sup_i f(x_1^n)-f(x_1^n)$,
\begin{align*}
    D_{\mathrm{KL}}^{(i)}\!\left(tf\right)(x_{-i})&=D_{\mathrm{KL}}\!\left(P^{(i)}_{tf}(\cdot\mid x_{-i})\,\middle\Vert\,P_{X_i}\right)
    =\int \frac{e^{tf(x_i,x_{-i})}}{\mathbb{E}^{(i)}\bigl[e^{tf}\bigr](x_{-i})} \ln \left(\frac{e^{tf(x_i,x_{-i})}}{\mathbb{E}^{(i)}\bigl[e^{tf}\bigr](x_{-i})}\right) dP_{X_i}\\&
    =\frac{\mathbb{E}^{(i)}[tfe^{tf}](x_{-i})}{\mathbb{E}^{(i)}\bigl[e^{tf}\bigr](x_{-i})}-\ln \left(\mathbb{E}^{(i)}\bigl[e^{tf}\bigr](x_{-i})\right)\\ &=\frac{\mathbb{E}^{(i)}[-tD_ie^{-tD_i}](x_{-i})}{\mathbb{E}^{(i)}\bigl[e^{-tD_i}\bigr](x_{-i})}-\ln \left(\mathbb{E}^{(i)}\bigl[e^{-tD_i}\bigr](x_{-i})\right)\\&
    =-\frac{\mathbb{E}^{(i)}[tD_ie^{-tD_i}](x_{-i})}{\mathbb{E}^{(i)}\bigl[e^{-tD_i}\bigr](x_{-i})}-\ln \left(\mathbb{E}^{(i)}\bigl[e^{-tD_i}\bigr](x_{-i})\right)\\&
    \leq-\ln\left(1+\frac{\mathbb{E}^{(i)}[tD_ie^{-tD_i}](x_{-i})}{\mathbb{E}^{(i)}\bigl[e^{-tD_i}\bigr](x_{-i})}\right)-\ln \left(\mathbb{E}^{(i)}\bigl[e^{-tD_i}\bigr](x_{-i})\right)\\&
    =-\ln\left(\mathbb{E}^{(i)}\bigl[(1+tD_i)e^{-tD_i}\bigr](x_{-i})\right)\\&
    \leq -\mathbb{E}^{(i)}\bigl[\ln\left((1+tD_i)e^{-tD_i}\right)\bigr](x_{-i})\\&
    =-\mathbb{E}^{(i)}\bigl[\ln\left(1+tD_i\right)-tD_i\bigr](x_{-i})
    \le\frac{t^p\mathbb{E}^{(i)}\bigl[ D_i^p\bigr](x_{-i})}{p}.
\end{align*}
The last inequality holds because $x-\ln(1+x)\le\frac{x^p}{p}$ for $p\in(1,2]$ and $x\ge0$.\\
Thus,
\begin{align*}
&D_{\mathrm{KL}}(P_{t f}\Vert P_{\bm{X}})
\le
\frac{t^p}{p}\mathbb{E}_{P_{t f}}
\left[
\sum_{i=1}^n
\mathbb{E}^{(i)}\bigl[ D_i(\bm{X})^p\bigr]
\right]\le\frac{t^p}{p}V,\\
&\beta\int_{0}^{\beta}\frac{D_{\mathrm{KL}}\!\left(P_{tf} \,\middle\|\, P_{\bm{X}}\right)}{t^2}\,dt\leq\frac{V\beta^p}{p(p-1)}.
\end{align*}
Combined with \eqref{eq:ent_chernoff},

\mathbb P(f(\bm{X})-\mathbb E[f(\bm{X})]\ge t)

  \inf_{\beta>0}

\]
The bound is worse than \eqref{eq:reverse_chernoff}. To achieve the same bound as \eqref{eq:reverse_chernoff} by bounding the conditional KL divergence of the form $D_{\mathrm{KL}}^{(i)}\!\left(-t D_i\right)(x_{-i})

\label{lem:pythagoras}
Let $f\ge0$ with $\mathbb E[f]=1$, let $\mathcal G\subseteq\mathcal F$ be a
sub-$\sigma$-algebra, and set $f_{\mathcal G}\coloneqq\mathbb E[f\mid\mathcal G]$.
For a probability measure $\mu$ and a measurable $f\ge0$ with
$\mathbb E_\mu[f]<\infty$, define the entropy of $f$ relative to $\mu$ by
\begin{equation*}
  {\rm Ent}_\mu(f)
  \coloneqq\mathbb E_\mu[f\ln f]-\mathbb E_\mu[f]\ln\mathbb E_\mu[f],
\end{equation*}
with the convention $0\ln0=0$. If $\mathbb E[f|\ln f|]<\infty$, then $fP_{\bm{X}}\ll f_{\mathcal G}P_{\bm{X}}$ and
\[
  D_{\mathrm{KL}}\!\left(fP_{\bm{X}}\,\Vert\,f_{\mathcal G}P_{\bm{X}}\right)
  ={\rm Ent}_{P_{\bm{X}}}(f)-{\rm Ent}_{P_{\bm{X}}}(f_{\mathcal G}).
\]

Both $f$ and $f_{\mathcal G}$ are probability densities with respect to $P_{\bm{X}}$,
so ${\rm Ent}_{P_{\bm{X}}}(f)=\mathbb E[f\ln f]$ and likewise for $f_{\mathcal G}$.
Since $f_{\mathcal G}$ is $\mathcal G$-measurable, the set
$N\coloneqq\{f_{\mathcal G}=0\}$ lies in $\mathcal G$, so it is an admissible test set
in the defining property of conditional expectation,
$\int_A f\,d\mu=\int_A\mathbb E[f\mid\mathcal G]\,d\mu$ for all $A\in\mathcal G$.
Taking $A=N$ gives
\[
  \int_N f\,d\mu
  = \int_N \mathbb E[f\mid\mathcal G]\,d\mu
  =\mathbb E\big[\mathbf 1_N\,\mathbb E[f\mid\mathcal G]\big]
  =\mathbb E\big[\mathbf 1_N f_{\mathcal G}\big]=0 ,
\]
because $f_{\mathcal G}$ vanishes on $N$.
As $f\ge0$, this forces $f=0$ $\mu$-a.e.\ on $N$. Consequently, if
$A\in\mathcal F$ satisfies $\int_A f_{\mathcal G}\,d\mu=0$, then
$f_{\mathcal G}=0$ $\mu$-a.e.\ on $A$, i.e.\ $\mu(A\setminus N)=0$, and hence
$\int_A f\,d\mu=0$. Therefore $f\mu\ll f_{\mathcal G}\mu$, and for every
$A\in\mathcal F$,
\[
  \int_A\frac{f}{f_{\mathcal G}}\mathbf 1_{\{f_{\mathcal G}>0\}}f_{\mathcal G}\,d\mu
  =\int_{A\setminus N}f\,d\mu=\int_A f\,d\mu,
\]
so that
$\frac{d(f\mu)}{d(f_{\mathcal G}\mu)}=f/f_{\mathcal G}$ holds
$f_{\mathcal G}\mu$-a.e., the values on $N$ being irrelevant. Thus
\[
  D_{\mathrm{KL}}\!\left(f\mu\Vert f_{\mathcal G}\mu\right)
  =\mathbb E_\mu\!\left[f\ln\frac{f}{f_{\mathcal G}}\right],
  \qquad 0\ln 0\coloneqq0 .
\]
and
\[
  D_{\mathrm{KL}}\!\left(fP_{\bm{X}}\Vert f_{\mathcal G}P_{\bm{X}}\right)
  =\mathbb E\!\left[f\ln\frac{f}{f_{\mathcal G}}\right]
  =\mathbb E[f\ln f]-\mathbb E\!\left[f\ln f_{\mathcal G}\right].
\]
For the second term, note that $(\ln f_{\mathcal G})^{\pm}$ is
$\mathcal G$-measurable and nonnegative, so pulling it out of the conditional
expectation (valid for nonnegative factors, without integrability assumptions)
and using the tower property gives
\[
  \mathbb E\!\left[f(\ln f_{\mathcal G})^{\pm}\right]
  =\mathbb E\!\left[\mathbb E\!\left[f(\ln f_{\mathcal G})^{\pm}\mid\mathcal G\right]\right]
  =\mathbb E\!\left[(\ln f_{\mathcal G})^{\pm}\,\mathbb E[f\mid\mathcal G]\right]
  =\mathbb E\!\left[f_{\mathcal G}(\ln f_{\mathcal G})^{\pm}\right].
\]
$\mathbb E[f_{\mathcal G}\ln f_{\mathcal G}]
\le\mathbb E[f\ln f]<\infty$ by Jensen applied to the convex function
$x\mapsto x\ln x$. Also, $\mathbb E[-f_{\mathcal G}\ln f_{\mathcal G}]\le\frac{1}{e}$ since $\forall x\ge0$, $x\ln x\ge-\frac1e$. Subtracting,
$\mathbb E[f\ln f_{\mathcal G}]=\mathbb E[f_{\mathcal G}\ln f_{\mathcal G}]
={\rm Ent}_{P_{\bm{X}}}(f_{\mathcal G})$, which proves the claim.
