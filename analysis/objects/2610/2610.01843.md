<!-- assumption-1: body_md -->
\label{ass:base}
The following conditions hold for constants $\ell_f,\ell_g,\rho_1,\mu>0$:
\begin{enumerate}[leftmargin=1.5em,label=(\roman*)]
  \item $f$ is differentiable with a jointly $\ell_f$-Lipschitz gradient,
  and $g$ is twice differentiable with a jointly $\ell_g$-Lipschitz
  gradient.
  \item $g(x,\cdot)$ is $\mu$-strongly convex for every $x$.
  \item The Hessian block $\nabla^2_{yy}g$ and the mixed derivative
  $\nabla^2_{xy}g$ are each $\rho_1$-Lipschitz in the joint variable
  $(x,y)$ with respect to the operator norm.
  \item $\left\lVert \nabla_y f(x,y)\right\rVert\le B_f$ for all $(x,y)$, and $\Phi$ is bounded below.
\end{enumerate}

<!-- assumption-2: body_md -->
\label{ass:order-p}
Fix an integer $p\ge1$ and a constant $L_p>0$.
For each $\ell\in\{x,y\}$, the map $\nabla_\ell g(x,\cdot)$ is $p$ times
differentiable for every $x$ and satisfies
\begin{equation}
\label{eq:p-smooth}
\left\lVert \nabla_y^p\nabla_\ell g(x,y)-\nabla_y^p\nabla_\ell g(x,y')\right\rVert_{\mathrm{op}}
\le L_p\left\lVert y-y'\right\rVert, \text{ for all } x,y,y'.
\end{equation}
<!-- end assumption-2 -->
<!-- assumption-3: body_md -->
\label{ass:oracle}
The query rules and oracle responses are measurable.
Let $\mathcal H$ denote the history before an oracle call, and let
$(x,y)$ be an $\mathcal H$-measurable query point.
For each $\ell\in\{x,y\}$, the returned estimates satisfy
\begin{gather}
\mathbb{E}[\hat F_\ell(x,y)\mid\mathcal H]=\nabla_\ell f(x,y),\qquad
\mathbb{E}[\hat G_\ell(x,y)\mid\mathcal H]=\nabla_\ell g(x,y),
\label{eq:oracle-mean}\\
\sum_{\ell\in\{x,y\}}
\mathbb{E}\!\left[
\|\hat F_\ell(x,y)-\nabla_\ell f(x,y)\|^2
+\|\hat G_\ell(x,y)-\nabla_\ell g(x,y)\|^2
\,\middle|\,\mathcal H
\right]
\le \sigma^2,
\label{eq:oracle-variance}
\end{gather}
almost surely, for a constant $\sigma\ge0$.
Each call uses fresh randomness.
Calls at query points fixed by a common conditioning history are
conditionally independent given that history.
No independence is assumed among the four estimates
$\hat F_x,\hat F_y,\hat G_x,\hat G_y$ returned by a single call.

\label{thm:main}

There exist constants $\alpha>0$ and $C_\star\ge1$, depending only on

<!-- lemma-1: body_md -->
\label{lem:app-geometry}
Under Assumption~\ref{ass:base}, the lower solution exists uniquely for every
$x$.  Moreover,
\begin{align}
\|y^\star(x)-y^\star(x')\|
 &\le L_y\|x-x'\|, \label{eq:app-y-lipschitz}\\
\|z(x,y)\|,\ \|z^\star(x)\|&\le q_0, \label{eq:app-z-bounded}\\
\|z^\star(x)-z^\star(x')\|
 &\le L_z\|x-x'\|. \label{eq:app-z-lipschitz}
\end{align}
The hypergradient identity (\ref{eq:hypergradient}) holds, and $\nabla\Phi$
is $L_\Phi$-Lipschitz.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Strong convexity gives a coercive quadratic lower bound for $g(x,\cdot)$,
so continuity gives existence and strong convexity gives uniqueness.
The first-order condition is $\nabla_y g(x,y^\star(x))=0$.
For $a=y^\star(x)$ and $a'=y^\star(x')$, strong monotonicity implies
\[
\mu\|a-a'\|^2
\le
\langle \nabla_y g(x,a)-\nabla_y g(x,a'),a-a'\rangle.
\]
Substitute the two first-order conditions and apply the Lipschitz bound in
$x$.  The right-hand side is at most
$L\|x-x'\|\|a-a'\|$, proving (\ref{eq:app-y-lipschitz}), including the
case $a=a'$.

Differentiability of the Lipschitz gradient and lower strong convexity give
\[
\mu I\preceq H(x,y)\preceq LI,\qquad
\|J(x,y)\|_{\mathrm{op}}\le L,\qquad
\|H(x,y)^{-1}\|_{\mathrm{op}}\le\mu^{-1}.
\]
The bound on $\nabla_y f$ now proves (\ref{eq:app-z-bounded}).
The derivatives of $(x,y)\mapsto\nabla_y g(x,y)$ are $J(x,y)^\top$ and
$H(x,y)$; their continuity follows from the matrix Lipschitz bounds in
Assumption~\ref{ass:base}(iii).
The implicit-function theorem therefore applies locally at each lower
solution.  Uniqueness makes these local solution maps agree.  It follows
that
\[
D_xy^\star(x)=-H(x,y^\star(x))^{-1}J(x,y^\star(x))^\top.
\]
The chain rule proves (\ref{eq:hypergradient}).

For brevity in the remaining estimates, let
$s=(x,y^\star(x))$, $s'=(x',y^\star(x'))$, $q=z^\star(x)$, and
$q'=z^\star(x')$.  Subtracting the two auxiliary systems gives
\[
H(s)(q-q')
=b(s)-b(s')-[H(s)-H(s')]q'.
\]
By Assumption~\ref{ass:base}(iii),
$\|H(s)-H(s')\|_{\mathrm{op}}\le\rho_1\|s-s'\|$.
Use this bound, $\|q'\|\le q_0$, $\|H(s)^{-1}\|_{\mathrm{op}}\le\mu^{-1}$, and
\[
\|s-s'\|\le\|x-x'\|+\|y^\star(x)-y^\star(x')\|
\le(1+L_y)\|x-x'\|.
\]
The resulting bound is
$\|q-q'\|\le B(1+L_y)\|x-x'\|/\mu$, proving
(\ref{eq:app-z-lipschitz}).

Finally, the difference of the hypergradients equals
\[
\nabla_x f(s)-\nabla_x f(s')
-[J(s)-J(s')]q'-J(s)(q-q').
\]
The first two terms have combined norm at most
$B(1+L_y)\|x-x'\|$, and the last term has norm at most
$LL_z\|x-x'\|$.  Their sum gives $L_\Phi$.

<!-- lemma-3: body_md -->
\label{lem:app-fd}
The weights in (\ref{eq:stencil-moments}) exist uniquely for every integer
$p\ge1$.  Under Assumptions~\ref{ass:order-p} and~\ref{ass:oracle}, the
finite differences and their sampled estimates satisfy
(\ref{eq:fd-bias})--(\ref{eq:fd-variance}) with
$C_p,A_p$ from (\ref{eq:app-stencil-constants}).
More precisely, if $x,y,z$ are measurable with respect to a history
$\mathcal F$ and the endpoint batches use fresh calls after that history,
then, for $\ell\in\{x,y\}$,
\begin{align}
\mathbb{E}[\widehat{\mathrm{FD}}_\ell(x,y;z)\mid\mathcal F]
 &=\mathrm{FD}_\ell(x,y;z), \label{eq:app-fd-mean}\\
\|\mathrm{FD}_\ell(x,y;z)-\nabla^2_{\ell y}g(x,y)z\|
 &\le C_pL_p\tau^p\|z\|, \label{eq:app-fd-bias}\\
\mathbb{E}[\|\widehat{\mathrm{FD}}_\ell(x,y;z)-\mathrm{FD}_\ell(x,y;z)\|^2
       \mid\mathcal F]
 &\le \frac{A_p\sigma^2\|z\|^2}{m\tau^2}.
\label{eq:app-fd-var}
\end{align}
These statements include $z=0$.

<!-- proof-3: body_md -->
The set $\mathcal I_p$ has $p+1$ distinct nodes, including for $p=1$.
Let $l_j(t)$ be its Lagrange basis polynomials.  Differentiating the
interpolation identity for a polynomial of degree at most $p$ shows that
$a_j=l_j'(0)$ satisfies (\ref{eq:stencil-moments}).
The Vandermonde matrix of distinct nodes is invertible, giving uniqueness.

Suppose first that $z\ne0$ and condition on $\mathcal F$.
Write $u=z/\|z\|$ and
$\varphi(t)=\nabla_\ell g(x,y+tu)$.
By Assumption~\ref{ass:order-p}, $\nabla_y^p\nabla_\ell g(x,\cdot)$
is $L_p$-Lipschitz.  Since $\|u\|=1$, the $p$th derivative of
$\varphi$ is also $L_p$-Lipschitz.
Taylor's formula with an integral remainder gives, for every real $t$,
\begin{align*}
\varphi(t)
 &=\sum_{r=0}^{p}\frac{t^r}{r!}\varphi^{(r)}(0)+\mathcal R_p(t),\\
\mathcal R_p(t)
 &=\frac{t^p}{(p-1)!}\int_0^1(1-s)^{p-1}
       [\varphi^{(p)}(st)-\varphi^{(p)}(0)]\,ds.
\end{align*}
In particular, including when $t<0$,
\[
\|\mathcal R_p(t)\|
\le\frac{L_p|t|^{p+1}}{(p-1)!}
     \int_0^1s(1-s)^{p-1}\,ds
=\frac{L_p|t|^{p+1}}{(p+1)!}.
\]
Thus this argument does not require a $(p+1)$st derivative.
Insert the expansion at $t=j\tau$ into (\ref{eq:stencil}).
All polynomial terms except the first derivative vanish by the moment
identities.  The retained term is
$\|z\|\varphi'(0)=\nabla^2_{\ell y}g(x,y)z$.
The remainder has norm at most
\[
\frac{\|z\|}{\tau}
\sum_j|a_j|\frac{L_p|j\tau|^{p+1}}{(p+1)!}
=C_pL_p\tau^p\|z\|.
\]

For each node $j$, set
$\nu_j:=\bar G_{\ell,j}(x,y+j\tau u)-\nabla_\ell g(x,y+j\tau u)$.
Every summand in (\ref{eq:oracle-variance}) is nonnegative, so each block
$\hat G_\ell(x,y)$ has conditional variance at most $\sigma^2$.
Averaging $m$ independent calls reduces this bound to $\sigma^2/m$.
Together with (\ref{eq:oracle-mean}), this gives
\[
\mathbb{E}[\nu_j\mid\mathcal F]=0,\qquad
\mathbb{E}[\|\nu_j\|^2\mid\mathcal F]\le\sigma^2/m,\qquad
\mathbb{E}[\langle\nu_i,\nu_j\rangle\mid\mathcal F]=0\quad(i\ne j).
\]
The cross terms vanish because distinct nodes use distinct calls; no
independence between blocks within a single call is required.  Consequently,
\begin{align*}
\mathbb{E}[\|\widehat{\mathrm{FD}}_\ell-\mathrm{FD}_\ell\|^2\mid\mathcal F]
 &=\frac{\|z\|^2}{\tau^2}
       \sum_j a_j^2\mathbb{E}[\|\nu_j\|^2\mid\mathcal F]\\
 &\le\frac{A_p\sigma^2\|z\|^2}{m\tau^2},
\end{align*}
and the same centered-noise representation proves
(\ref{eq:app-fd-mean}).
If $z=0$, both differences and both exact matrix--vector products are
zero by definition, so every assertion follows without endpoint calls.

\label{lem:app-auxiliary}
With $\eta_z=\eta=1/(2L)$, the auxiliary update satisfies
\begin{equation}
\label{eq:app-auxiliary-bound}
\mathbb{E}_k\|\bar e_k\|^2
\le (1-r)\|e_k\|^2
   +\frac{6\eta}{\mu}\bigl(B^2\|\delta_k\|^2+\beta^2\bigr)

The target in $\bar e_k$ is $z^\star(x_k)$, before the outer point moves.

<!-- proof-5: body_md -->
Set $q=z^\star(x_k)$ and
$w_k=H_kq-b_k+d_k^z$.  These are $\mathcal F_k$-measurable, and
Lemmas~\ref{lem:app-mismatch} and~\ref{lem:app-sampling} give
\begin{equation}
\label{eq:app-w-bound}
\|w_k\|^2\le2B^2\|\delta_k\|^2+2\beta^2.
\end{equation}
Because $q\in Q$ and Euclidean projection is nonexpansive,
\[
\|\bar e_k\|^2
\le\|(I-\eta H_k)e_k-\eta w_k-\eta\xi_k^z\|^2.
\]
Take conditional expectation only after this inequality.  The noise is
centered before projection, so
\begin{equation}
\label{eq:app-preprojection}
\mathbb{E}_k\|\bar e_k\|^2
\le\|(I-\eta H_k)e_k-\eta w_k\|^2+\eta^2v.
\end{equation}
No unbiasedness property of the projected iterate is needed.

(\ref{eq:app-preprojection}).  Since $0<r\le1/2$,

(\ref{eq:app-auxiliary-bound}).

We divide the proof into four steps.

\paragraph{Step 1: dimension-free derivative bounds for the weights.}
Let
\[
  {C_{\Gamma,1}}:=\left\lVert \Gamma'\right\rVert_\infty,
  \qquad
  {C_{\Gamma,2}}:=\left\lVert \Gamma''\right\rVert_\infty.
\]
Fix $i$ and set $w_j(v):=\gamma(v_j)$ for $j=i,\ldots,T$.  Write
\[
  w(v):=(w_i(v),\ldots,w_T(v)),
  \qquad
  r(v):=\left\lVert w(v)\right\rVert_2.
\]
Then $\Theta_i(v)=\Gamma(1-r(v))$.  The composition is smooth at
$r=0$ because it is constant whenever $r\le1/2$.  Its first and second
derivatives vanish for $r\le1/2$ and for $r\ge3/4$; at the two endpoints
this follows from the smooth flatness of $\Gamma$, rather than from a
two-sided neighborhood of constancy.  Therefore derivatives can be nonzero
only on
\begin{equation}\label{lb:eq:r-transition}
  \frac12<r<\frac34.
\end{equation}
On this region,
\[
  \nabla r=Dw(v)^\top\frac{w(v)}{r},
  \qquad
  \left\lVert \nabla r\right\rVert\le {C_{\Gamma,1}}.
\]
Moreover,
\[
  \nabla^2r
  =\frac1rDw(v)^\top
  \left(I-\frac{w(v)w(v)^\top}{r^2}\right)Dw(v)
  +\sum_{j=i}^T\frac{w_j(v)}{r}\nabla^2w_j(v).
\]
The first term has operator norm at most $2{C_{\Gamma,1}}^2$ by
(\ref{lb:eq:r-transition}).  The second term is diagonal, and its operator norm
is at most ${C_{\Gamma,2}}$.  Hence
\begin{equation}\label{lb:eq:r-derivative-bounds}
  \left\lVert \nabla r\right\rVert\le {C_{\Gamma,1}},
  \qquad
  \left\lVert \nabla^2r\right\rVert_{\mathrm{op}}\le2{C_{\Gamma,1}}^2+{C_{\Gamma,2}}

$(\mathsf S_j^{(i)})^\perp$, and put
$\Pi_j^{(i)}:=I-P_j^{(i)}$.  For $i\in[n]$ and $j\in[T]$, define

  :=\left\{

\end{equation}

For fixed $j$, the spaces $\mathsf S_{j-1}^{(q)}$ are nested in $q$.

  \Pi_{j-1}^{(q-1)}
  +P_{j-1}^{(q-1)}\Pi_{j-1}^{(q)}P_{j-1}^{(q-1)}
  =\Pi_{j-1}^{(q)}.
\]
Iterating this identity gives

  \Pi_{j-1}^{(i)}
  =\sum_{r=1}^{j-1}u_ru_r^\top
  +\sum_{q=1}^i

\end{equation}

(\ref{lb:eq:projection-telescope}) and Cauchy--Schwarz yield

  |\left\langle u_j,x_i\right\rangle|
  &\le
  \left(
  \sum_{q=1}^i
  \left\lVert \Pi_{j-1}^{(q)}P_{j-1}^{(q-1)}u_j\right\rVert^2
  \right)^{1/2}
  \nonumber\\

  \sum_{q=1}^i
  \left\lVert \Pi_{j-1}^{(q)}P_{j-1}^{(q-1)}x_i\right\rVert^2
  \right)^{1/2}.

\end{align}

The second squared factor in (\ref{lb:eq:projection-cs}) is therefore at most

\label{lb:lem:invariant-posterior}
Let $\mathsf C,\mathsf X,\mathsf Z$ be standard Borel spaces, let
$G$ be a compact second-countable group with normalized Haar probability
$\lambda_G$, and let
\[
  \tau:\mathsf C\times G\times\mathsf X\to\mathsf X
\]
be a Borel map such that $g\mapsto\tau_{c,g}$ is a group action for every
$c$.  Let $K(c,dx)$ be a Borel probability kernel satisfying
\begin{equation}\label{lb:eq:prior-kernel-invariance}
  K(c,\tau_{c,g}^{-1}A)=K(c,A)
  \qquad(c\in\mathsf C,\ g\in G)
\end{equation}
for every Borel $A\subseteq\mathsf X$.  Suppose that the Borel observation
$H:\mathsf C\times\mathsf X\to\mathsf Z$ is invariant:
\begin{equation}\label{lb:eq:observation-invariance}
  H(c,\tau_{c,g}x)=H(c,x).
\end{equation}
Then there is a Borel regular conditional kernel
$L(c,v,dx)$ for $x$ given $v=H(c,x)$ under $K(c,dx)$ such that
\begin{equation}\label{lb:eq:posterior-kernel-invariance}
  L(c,v,\tau_{c,g}^{-1}A)=L(c,v,A)
\end{equation}
for every $g\in G$ and for $H(c,\cdot)_\#K(c)$-almost every $v$.

<!-- proof-15: body_md -->
Let $L_0(c,v,dx)$ be any Borel disintegration of $K(c,dx)$ over the map
$H(c,\cdot)$, and define
\begin{equation}\label{lb:eq:averaged-posterior}
  L(c,v,A)
  :=\int_G L_0(c,v,\tau_{c,g}^{-1}A)\,\lambda_G(dg).
\end{equation}
This is a Borel kernel.  If
$M(c,dv):=H(c,\cdot)_\#K(c)$, then for Borel
$A\subseteq\mathsf X$ and $B\subseteq\mathsf Z$,
\begin{align*}
\int_B L(c,v,A)M(c,dv)
&=\int_G K\bigl(c,\tau_{c,g}^{-1}A\cap H(c,\cdot)^{-1}B\bigr)
  \,\lambda_G(dg)\\
&=\int_G K\bigl(c,\tau_{c,g}^{-1}
  (A\cap H(c,\cdot)^{-1}B)\bigr)\,\lambda_G(dg)\\
&=K\bigl(c,A\cap H(c,\cdot)^{-1}B\bigr),
\end{align*}
where the second line uses (\ref{lb:eq:observation-invariance}) and the third
uses (\ref{lb:eq:prior-kernel-invariance}).  Hence $L$ is a regular
conditional kernel.  Equation (\ref{lb:eq:posterior-kernel-invariance})
follows directly from (\ref{lb:eq:averaged-posterior}) and left/right
invariance of Haar probability on the compact group $G$.

\label{lb:lem:sequential-posterior}
Let $U=[u_1,\ldots,u_T]$ have the uniform invariant Stiefel distribution on
$\operatorname{Ortho}(d,T)$, and let $\vartheta$
be an independent random element of a standard Borel space.  Fix
$j\in[T]$ and an integer $r\ge0$.  Suppose that Borel maps $\chi_q$ define
recursively
\begin{equation}\label{lb:eq:sequential-query-recursion}
  x_q=\chi_q(\vartheta,u_1,\ldots,u_{j-1},v_1,\ldots,v_{q-1}),
  \qquad
  v_q=U^\top x_q,
  \qquad q\in[r].
\end{equation}
Let
\[
  C_q:=(\vartheta,u_1,\ldots,u_{j-1},v_1,\ldots,v_q),
\]
with $C_0=(\vartheta,u_1,\ldots,u_{j-1})$, and set
\[
  \mathsf S_q
  :=\operatorname{span}\{u_1,\ldots,u_{j-1},x_1,\ldots,x_q\}.
\]
Regular conditional kernels $K_q(C_q,dU)$ of $U$ given $C_q$ may be
chosen so that, for almost every $C_q=c_q$, the kernel is invariant under
every orthogonal map fixing $\mathsf S_q(c_q)$ pointwise:
\begin{equation}\label{lb:eq:sequential-posterior-invariance}
  K_q(c_q,W^{-1}A)=K_q(c_q,A)
  \quad
  \text{whenever }Ws=s\text{ for all }s\in\mathsf S_q(c_q).
\end{equation}

We construct the kernels inductively.  Conditional on
$u_1,\ldots,u_{j-1}$, the remaining columns of a uniform Stiefel frame have
the standard invariant-completion law.  Explicitly, after choosing a Borel
orthonormal frame $Q_\perp(u_{<j})$ for
$\operatorname{span}(u_1,\ldots,u_{j-1})^\perp$, this law is the image of
the uniform invariant probability measure on the Stiefel manifold
$\operatorname{Ortho}(d-j+1,T-j+1)$ under
\[
  V\longmapsto [u_1,\ldots,u_{j-1},Q_\perp(u_{<j})V].
\]
It is invariant under every orthogonal map fixing
$\mathsf S_0=\operatorname{span}\{u_1,\ldots,u_{j-1}\}$ pointwise.
Independence of $\vartheta$ gives the required kernel $K_0$.

Assume that $K_{q-1}$ has been chosen.  Let
$\mathsf C_{q-1}^{\rm good}$ be a Borel full-measure subset of the
conditioning space on which the induction hypothesis holds.  We first
restrict the conditioning parameter and the kernel to this good set; only
after constructing the next posterior there will we extend it arbitrarily
over the null complement.  Once $C_{q-1}=c_{q-1}\in
\mathsf C_{q-1}^{\rm good}$ is fixed, the vector $x_q$ in
(\ref{lb:eq:sequential-query-recursion}) is fixed.  Put
\[
  \mathsf S_q
  =\operatorname{span}(\mathsf S_{q-1},x_q).
\]
On each stratum on which $\dim\mathsf S_q=s$, choose a Borel orthogonal
completion $Q_\perp(c_{q-1})$ of $\mathsf S_q$.  For
$R\in O(d-s)$ define
\[
  W_{c_{q-1},R}
  :=P_{\mathsf S_q}
    +Q_\perp(c_{q-1})R Q_\perp(c_{q-1})^\top.
\]
This is a Borel action of the fixed compact group $O(d-s)$ on the
conditional Stiefel fiber.  Because it fixes $\mathsf S_q$ pointwise, it
belongs to the stabilizer from the induction hypothesis.  Moreover, the
new observation
\[
  H_q(c_{q-1},U):=U^\top x_q
\]
is invariant under this action:
\[
  (W_{c_{q-1},R}U)^\top x_q
  =U^\top W_{c_{q-1},R}^\top x_q
  =U^\top x_q.
\]
Apply Lemma~\ref{lb:lem:invariant-posterior}, separately on each Borel rank
stratum of $\mathsf C_{q-1}^{\rm good}$, to disintegrate $K_{q-1}$ over
$H_q$.  The finitely many resulting kernels paste into a Borel kernel on the
good set; extend that kernel arbitrarily over the null complement.  The
resulting $K_q$ is a regular conditional law given $C_q$ and, for almost
every $C_q$, is invariant under the full pointwise stabilizer of
$\mathsf S_q$.  This closes the induction.
Every orthogonal transformation used in this argument is fixed only
after $C_{q-1}$ has been conditioned upon; no state-dependent global
change of variables is invoked.

\label{lb:lem:conditional-haar}
Fix $i\in[n]$ and $j\in[T]$.  On the event
$\mathcal T_j^{(i-1)}$, the past transcript and the current query
$(x_i,y_i)$ are measurable from $\mathcal U_{j-1}^{(i-1)}$.  In particular,
$\mathcal T_j^{(i-1)}$, the space
\[
  \mathsf S
  :=\operatorname{span}\{u_1,\ldots,u_{j-1},x_1,\ldots,x_{i-1}\},
\]
and its orthogonal projector are measurable under this conditioning.
There is an $\mathcal U_{j-1}^{(i-1)}$-measurable radius
$r_{i,j}\in[0,1]$ such that, outside a null event and on
$\mathcal T_j^{(i-1)}$,
\begin{equation}\label{lb:eq:conditional-haar-law}
\mathcal L\!\left(
P_{\mathsf S^\perp}u_j
\,\middle|\,
\mathcal U_{j-1}^{(i-1)}
\right)
=
\begin{cases}
\delta_0,&r_{i,j}=0,\\[1mm]
\operatorname{Unif}\!\left(
\{v\in\mathsf S^\perp:\left\lVert v\right\rVert=r_{i,j}\}
\right),&r_{i,j}>0.
\end{cases}
\end{equation}
Moreover,
$r_{i,j}=\left\lVert P_{\mathsf S^\perp}u_j\right\rVert$ almost surely.  Thus normalization
is used only on $\{r_{i,j}>0\}$.

$\mathcal T_j^{(i-1)}$.  We have therefore obtained, by a sequential

pointwise stabilizer of $\mathsf S$.  In particular, the conditional law of
$u_j$ is invariant under that stabilizer.

second case of (\ref{lb:eq:conditional-haar-law}); for $r_{i,j}=0$, the
sphere is the singleton $\{0\}$ and the law is $\delta_0$.

\label{lb:lem:finite-random-rotation}
Let $U$ have the uniform invariant Stiefel distribution on $\operatorname{Ortho}(d,T)$,
independently of i.i.d. seeds
$\xi_1,\xi_2,\ldots$ and the private randomness of an adaptive randomized
algorithm.  Suppose the algorithm makes queries $(x_t,y_t)$ satisfying
$\left\lVert x_t\right\rVert\le R$ almost surely and observes the canonical responses
(\ref{lb:eq:canonical-rotated-oracle}).  Put
\[
  \pi_N:=\max_{1\le t\le N}\operatorname{prog}_{1/4}(U^\top x_t).
\]
If the latent kernel is a uniform finite-channel probability-$\varpi$ zero-chain
and $d>N+T$, then
\begin{equation}\label{lb:eq:finite-rotation-bound}
  \mathbb P(\pi_N\ge T)
  \le e^{2\varpi N-T}
  +2NT\exp\!\left(-\frac{d-N-T}{32R^2N}\right).
\end{equation}
The bound does not depend on the finite number $m$ of vector channels.

We separate rare progress from geometric guessing.

\paragraph{Rare progress.}
For $t\in[N]$, let
\[
  \mathcal V_t
  :=\{\operatorname{prog}_{1/4}(v_t)\le\gamma_{t-1}\}.
\]
Use the filtration
\begin{equation}\label{lb:eq:progress-filtration}
  \mathcal H_t:=\sigma(U,\xi_0,\xi_1,\ldots,\xi_t),
  \qquad
  \mathcal H_0:=\sigma(U,\xi_0).
\end{equation}
Because the algorithm is deterministic after fixing $\xi_0$, $U$, and the
past seeds, $(x_t,y_t)$, $v_t$, $\gamma_{t-1}$, and $\mathcal V_t$ are
$\mathcal H_{t-1}$-measurable.  On $\mathcal V_t$,
(\ref{lb:eq:finite-chain-one-step}) gives
$\gamma_t-\gamma_{t-1}\in\{0,1\}$.  If additionally $\xi_t\notin E$, then
(\ref{lb:eq:finite-chain-quiet}) gives $\gamma_t=\gamma_{t-1}$.  Equivalently,
\begin{equation}\label{lb:eq:progress-gate}
  0\le\gamma_t-\gamma_{t-1}\le\mathbf 1_{\{\xi_t\in E\}}
  \qquad\text{on }\mathcal V_t.
\end{equation}
For the Bernoulli constructions, the right side is $\zeta_t$.
Thus an open gate is necessary for progress on $\mathcal V_t$, but
need not be sufficient.  Freshness of $\xi_t$ implies
\begin{equation}\label{lb:eq:progress-one-step-rigorous}
  \mathbb P\left(
  \gamma_t-\gamma_{t-1}=1,\mathcal V_t
  \mid\mathcal H_{t-1}
  \right)\le \varpi.
\end{equation}
Set $\iota_t:=\gamma_t-\gamma_{t-1}$.  Iterated conditioning gives
\begin{align*}
  \mathbb{E}\left[
  e^{\sum_{t=1}^N\iota_t}
  \mathbf 1_{\cap_{t=1}^N\mathcal V_t}
  \right]
  &\le(1-\varpi+\varpi e)^N
  \le e^{2\varpi N}.
\end{align*}
Hence
\begin{equation}\label{lb:eq:progress-chernoff}
  \mathbb P\left(
  \gamma_N\ge T,\bigcap_{t=1}^N\mathcal V_t
  \right)
  \le e^{2\varpi N-T}.
\end{equation}
On $\cap_t\mathcal V_t$, the event $\{\pi_N\ge T\}$ implies
$\{\gamma_N\ge T\}$.

\paragraph{Geometric guessing.}
Apply Lemmas~\ref{lb:lem:projection-accumulation} and
\ref{lb:lem:one-pair-guessing} with horizon $n=N$.  Since $\gamma_t$ is
nondecreasing,
\begin{equation}\label{lb:eq:G-implies-V}
  \bigcap_{i=1}^N\bigcap_{j=1}^T
  \left(\mathcal G_j^{(i)}\cup
  (\mathcal T_j^{(i-1)})^c\right)
  \subseteq
  \bigcap_{i=1}^N\mathcal V_i.
\end{equation}
Indeed, fix $i$ and $j>\gamma_{i-1}$.  Then
$j>\gamma_{q-1}$ for every $q\le i$, so the event on the left of
(\ref{lb:eq:G-implies-V}) supplies
$\cap_{q=1}^i\mathcal G_j^{(q)}$.  Lemma~\ref{lb:lem:projection-accumulation}
then gives $|\left\langle u_j,x_i\right\rangle|\le1/4$.  This holds for every
$j>\gamma_{i-1}$, which is exactly $\mathcal V_i$.

A union bound and Lemma~\ref{lb:lem:one-pair-guessing} now give
\begin{equation}\label{lb:eq:no-guessing-geometric}
  \mathbb P\left(\left[\bigcap_{i=1}^N\mathcal V_i\right]^c\right)
  \le
  2NT\exp\!\left(-\frac{d-N-T}{32R^2N}\right).
\end{equation}
Combining (\ref{lb:eq:progress-chernoff}) and
(\ref{lb:eq:no-guessing-geometric}) proves (\ref{lb:eq:finite-rotation-bound}).
The argument never counts the number of vector channels: on
$\mathcal T_j^{(i-1)}$, every one of them lies in
$\operatorname{span}\{u_1,\ldots,u_{j-1}\}$, and all non-ambient channels
are already determined by the conditioned latent queries and seeds.

  32R^2(N_{\mathrm s}+1)\log\bigl(64(N_{\mathrm s}+1)T\bigr)\right\rceil.
\end{equation}
Draw $U$ uniformly from $\operatorname{Ortho}(d,T)$, independently of the algorithm
seed and oracle draws.  The functions are smooth; their Lipschitz,
value, and gap bounds follow from
(\ref{lb:eq:scaled-u-bounds})--(\ref{lb:eq:scaled-gap}).

At a bounded query $\bar r\in\mathbb{R}^d$, write $v=U^\top\bar r$ and consider
\begin{equation}
\label{lb:eq:comparison-canonical}
  (w,v_B,v_W)
  :=\left(\frac\zeta\varpi W_T(v),\,
           U\nabla B_T(v),\,
           U\frac\zeta\varpi\nabla W_T(v)\right).
\end{equation}
By Corollary~\ref{lb:cor:two-channel-zero-chain}, its two vector
channels satisfy Definition~\ref{lb:def:finite-chain} with exceptional
event $\{\zeta=1\}$.  The scalar channel is a Borel function of the
latent query $v$ and the seed, as allowed by that definition.
Hence the full response, including the scalar value, is covered by
Corollary~\ref{lb:cor:terminal-output}.

One comparison call at an arbitrary $x$ is obtained by querying
(\ref{lb:eq:comparison-canonical}) at
$\bar r=\rho_R(x/\lambda_{\rm lb})$ and returning
\begin{equation}
\label{lb:eq:comparison-postprocessing}
\begin{aligned}
  r_A&:=\frac a{\lambda_{\rm lb}}
    J_R(x/\lambda_{\rm lb})^\top v_B
    +\frac{a\eta_{\rm r}}{\lambda_{\rm lb}^2}x,\\
  r_u&:=aw,\qquad
  r_{\nabla u}:=\frac a{\lambda_{\rm lb}}
    J_R(x/\lambda_{\rm lb})^\top v_W.
\end{aligned}
\end{equation}
Differentiating (\ref{lb:eq:scaled-Phi})--(\ref{lb:eq:scaled-u}) shows
that $(r_A,r_u,r_{\nabla u})$ is exactly
$\mathcal R_\varpi(x)$ in (\ref{eq:lb-comparison-oracle}).
These transformations are jointly continuous and independent of $U$.
The compressed final output
$\bar r_{\rm out}=\rho_R(x_{\rm out}/\lambda_{\rm lb})$ has norm at most
$R$ and costs no oracle call.

For any algorithm using at most $N_{\mathrm s}$ comparison calls,
pad a shorter run with ignored calls.
Corollaries~\ref{lb:cor:postprocessing} and \ref{lb:cor:terminal-output}
and the dimension choice give
\begin{equation}
\label{lb:eq:comparison-unfinished}
  \mathbb P\bigl(\operatorname{prog}_{1/4}(U^\top\bar r_{\rm out})\ge T\bigr)
  \le e^{2\varpi N_{\mathrm s}-T}+\frac1{32}
  \le e^{-3T/4}+\frac1{32}<\frac14,
\end{equation}
since $T\ge4$.  Equation~(\ref{lb:eq:scaled-stationarity}) now gives
(\ref{eq:lb-common-progress}).  Thus $c=1/8$ is admissible, and a
common numerical $C$ bounds the remaining quantities.

\label{lem:lb-gradient-variance}
For each pair in Proposition~\ref{prop:lb-comparison},
(\ref{eq:lb-first-order-oracle}) is conditionally unbiased and
\begin{equation}
\label{eq:lb-signal-variance}
  \mathbb{E}[\|\widehat{\nabla P}(x)-\nabla P(x)\|^2\mid\mathcal H]
  =\frac{1-\varpi}{\varpi}\|\nabla u(x)\|^2
  \le\frac{C^2{\epsilon}^2}{\varpi}.
\end{equation}
Choosing $\varpi=2(1+C^2){\epsilon}^2\le1/2$ gives a family with variance at most one for which guaranteeing
$\mathbb{E}\|\nabla P(x_{\rm out})\|^2\le{\epsilon}^2$ requires
$\Omega({\epsilon}^{-4})$ queries.
<!-- end lemma-18 -->
<!-- proof-23: body_md -->
Fix $U$ and condition on the history $\mathcal H$, which fixes the query
$x$.  The fresh Bernoulli seed satisfies $\mathbb{E}[\zeta/\varpi]=1$, so
(\ref{eq:lb-first-order-oracle}) is conditionally unbiased.  Moreover,
\[
  \mathbb{E}[\|\widehat{\nabla P_U}(x)-\nabla P_U(x)\|^2\mid\mathcal H]
  =\frac{1-\varpi}{\varpi}\|\nabla u_U(x)\|^2
  \le\frac{C^2{\epsilon}^2}{\varpi},
\]
where $C$ is the constant in Proposition~\ref{prop:lb-comparison}.
If $\mathcal R_\varpi(x)=(r_A,r_u,r_{\nabla u})$, then
$r_A-r_{\nabla u}$ has exactly this stochastic-gradient law, with one
comparison call per gradient call.

in $(0,1/2]$, and the conditional variance is at most $1/2$.
The objective has gradient Lipschitz constant at most $2C$ and initial
gap at most $1/4$, by Proposition~\ref{prop:lb-comparison}.

$T/(8\varpi)$ calls therefore gives a comparison-oracle algorithm with
the same budget and output law.  Proposition~\ref{prop:lb-comparison}

  \mathbb P\bigl(\|\nabla P_U(x_{\rm out})\|\ge2{\epsilon}\bigr)\ge\frac34.

Averaging this probability over $U$ gives a fixed rotation on which the
probability, now over algorithm and oracle randomness, is at least
$3/4$.  On that instance,
$\mathbb{E}\|\nabla P_U(x_{\rm out})\|^2\ge3{\epsilon}^2>{\epsilon}^2$.
Finally, $T/\varpi=\Theta({\epsilon}^{-4})$.

\label{lb:lem:primitive-simulation}
For every fixed $U$ and selected query $(x,y)$, the four estimates are
conditionally unbiased and their joint conditional variance is at most
$C_{\rm var}\delta^2{\epsilon}^2/\varpi$.
One primitive call is a known measurable transformation of one

<!-- end lemma-19 -->
<!-- proof-24: body_md -->
Condition on the history, fixing $(x,y)$.  Unbiasedness follows from
$\mathbb{E}[\zeta/\varpi]=1$ and direct differentiation of
(\ref{lb:eq:lift-f})--(\ref{lb:eq:lift-g}).
The upper estimates are exact.  The sum of the conditional squared
errors in the lower estimates is

 &\frac{1-\varpi}{\varpi}
 \left(\delta^2\sin^2(y/\delta)\|\nabla u_U(x)\|^2
       +\cos^2(y/\delta)|u_U(x)|^2\right)\nonumber\\
 &\qquad\le
 \frac{C_{u,1}^2\delta^2{\epsilon}^2+C_{u,0}^2{\epsilon}^4}{\varpi}
 \le\frac{C_{\rm var}\delta^2{\epsilon}^2}{\varpi}.
 \label{lb:eq:primitive-variance}
\end{align}
Here Proposition~\ref{lb:prop:scaled-pair} supplies the uniform bounds,

\label{lb:eq:primitive-from-comparison}

<!-- lemma-20: body_md -->
\label{lb:lem:lower-solution}
For $|s|/\delta\le1/4$, the response $Y_\delta(s)$ is continuously
differentiable and obeys
\begin{align}
  |Y_\delta(s)|&\le |s|,
  \label{lb:eq:Y-size}\\
  |Y_\delta(s)+s|&\le\frac{|s|^3}{2\delta^2},
  \label{lb:eq:Y-value-error}\\
  |Y_\delta'(s)+1|&\le\frac{2s^2}{\delta^2}.
  \label{lb:eq:Y-derivative-error}
\end{align}
For all sufficiently small ${\epsilon}$, uniformly over ${\epsilon}\le\delta\le1$, the lower
solution of (\ref{lb:eq:lift-g}) is
\begin{equation}\label{lb:eq:y-star}
  y_U^\star(x)=Y_\delta(u_U(x)),
\end{equation}
and $\nabla^2_{yy}g_U(x,y)\ge3/4$ globally.
<!-- end lemma-20 -->
<!-- proof-25: body_md -->
The derivative in $Y$ of the left side of (\ref{lb:eq:Y-equation}) is
\[
  1-(s/\delta)\sin(Y/\delta)\ge\frac34.
\]
Hence that function is strictly increasing and crosses zero exactly once;
the implicit-function theorem gives differentiability.  Equation
(\ref{lb:eq:Y-equation}) gives $|Y_\delta(s)|\le|s|$, and
\[
  |Y_\delta(s)+s|
  =|s|\,|1-\cos(Y_\delta(s)/\delta)|
  \le\frac{|s|^3}{2\delta^2}.
\]
Implicit differentiation yields
\[
  Y_\delta'(s)
  =-\frac{\cos(Y_\delta(s)/\delta)}
  {1-(s/\delta)\sin(Y_\delta(s)/\delta)}.
\]
With $t=Y_\delta(s)/\delta$ and $\beta=s/\delta$,
\[
  |Y_\delta'(s)+1|
  =\left|\frac{1-\cos t-\beta\sin t}{1-\beta\sin t}\right|
  \le\frac{\tfrac12\beta^2+\beta^2}{3/4}
  =2\beta^2,
\]
where we used $|t|\le|\beta|$, $|1-\cos t|\le t^2/2$, and
$|\sin t|\le|t|$.  This proves (\ref{lb:eq:Y-derivative-error}).  Finally,
Proposition~\ref{lb:prop:scaled-pair} and $\delta\ge{\epsilon}$ imply
\[
  \frac{\left\lVert u_U\right\rVert_\infty}{\delta}
  \le C_{u,0}{\epsilon}\le\frac14
\]
for a universal choice of ${\epsilon}_0$.  Since
\[
  \nabla_y g_U=y+\cos(y/\delta)u_U,
  \qquad
  \nabla^2_{yy}g_U=1-\delta^{-1}\sin(y/\delta)u_U,
\]
the claims follow.

<!-- corollary-4: body_md -->
\label{lb:cor:lift-perturbation}
There are universal constants $C_F,C_\nabla>0$ such that
\begin{align}
  \sup_x|\Phi_U(x)-P_U(x)|
  &\le C_F\frac{{\epsilon}^6}{\delta^2}
  \le C_F{\epsilon}^4,
  \label{lb:eq:lift-value-perturb}\\
  \sup_x\left\lVert \nabla \Phi_U(x)
  -\nabla P_U(x)\right\rVert
  &\le C_\nabla\frac{{\epsilon}^5}{\delta^2}
  \le C_\nabla{\epsilon}^3.
  \label{lb:eq:lift-gradient-perturb}
\end{align}
After decreasing ${\epsilon}_0$ universally if necessary,
\begin{equation}\label{lb:eq:lift-gap}
  \Phi_U(0)-\inf_x\Phi_U(x)\le1,\qquad |y_U^\star(0)|\le1,
\end{equation}
and
\begin{equation}\label{lb:eq:lift-stationarity}
  \|\nabla P_U(x)\|\ge2{\epsilon}
  \quad\Longrightarrow\quad
  \left\lVert \nabla \Phi_U(x)\right\rVert\ge\frac32{\epsilon}.
\end{equation}
<!-- end corollary-4 -->

The first estimate follows from (\ref{lb:eq:Y-value-error}),
$P_U={A_U}-u_U$, and
(\ref{lb:eq:scaled-u-bounds}).  The chain rule and
(\ref{lb:eq:Y-derivative-error}) give
\[
  \nabla \Phi_U
  =\nabla{A_U}
   +Y_\delta'(u_U)\nabla u_U,
\]
and hence the second estimate.  Here $\delta\ge{\epsilon}$ makes both displayed
upper bounds uniform over ${\epsilon}\le\delta\le1$.  Combining the value estimate with

<!-- proof-28: body_md -->
Proposition~\ref{prop:lb-scalar-transfer} proves the base regularity.
For $p\ge2$, write $t=y/\delta$ and denote derivatives of sine by
$\sin^{(j)}$.  Direct differentiation gives
\begin{equation}\label{lb:eq:high-g-formula}
\begin{aligned}
  \nabla_y^p\nabla_x g_U(x,y)
  &=\delta^{1-p}\sin^{(p)}(t)\nabla u_U(x),\\
