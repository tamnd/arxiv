<!-- theorem-1: body_md -->
\label{thm.Hessian-estimate}
Let $2\leq k\leq n$, $0\leq\ell<k$, and $k-\ell\in\{1,2\}$.
Suppose that $u\in C^\infty(B_1)$ is a $k$-convex solution of \eqref{eqn.sigma-quotient} in $B_1\subset\mathbb R^n$.
Then
\[
|D^2u(0)|\leq C\left(n,k,\|u\|_{C^1(B_1)}\right).
\]

From the interior gradient estimate for Hessian quotient equations by Chen \cite{C15},
we further obtain for $k$-convex solutions of \eqref{eqn.sigma-quotient} in $B_2$ that
\[
|D^2u(0)|
\leq C\left(n,k,\left\Vertu\right\Vert_{L^\infty(B_2)}\right).
\]
<!-- end theorem-1 -->
<!-- remark-1: body_md -->
The restriction $k-\ell\in\{1,2\}$ is sharp
for interior Hessian estimates controlled by the $C^1$ norm,
even for smooth convex solutions with constant right-hand side.

<!-- property-1: body_md -->
\label{prop.inequality}
Let $0\leq \ell<k\leq n$ and $\lambda\in\Gamma_k$. Then
\begin{enumerate}[\qquad\quad(1)]
\item
$\displaystyle F_i(\lambda)
=\frac{\sigma_{k-1}(\lambda|i)}{\sigma_\ell(\lambda)}
-\frac{\sigma_k(\lambda)\sigma_{\ell-1}(\lambda|i)}{\sigma_\ell(\lambda)^2}
>0$,

\item
$\displaystyle\sum_{i=1}^nF_i
=(n-k+1)\frac{\sigma_{k-1}(\lambda)}{\sigma_\ell(\lambda)}
-(n-\ell+1)
\frac{\sigma_k(\lambda)\sigma_{\ell-1}(\lambda)}{\sigma_\ell(\lambda)^2}$.

\item
$\displaystyle\sum_{i=1}^n\lambda_iF_i=(k-\ell)F$,
$\displaystyle\sum_{i=1}^n\lambda_iF_{ij}=(k-\ell-1)F_j$ for any $j\in\mathbb Z_{[1,n]}$.
\end{enumerate}

<!-- lemma-1: body_md -->
\label{lem.Fj-lambdaj}
Let $2\leq k\leq n$, $0\leq\ell<k$, $k-\ell\in\left\{1,2\right\}$.
Let $\lambda\in\Gamma_k$ satisfy $3/4\leq F^{1/(k-\ell)}(\lambda)\leq1$.
Then
\begin{equation}\label{eqn.Fj-lambdaj>}
F_j(\lambda)(1+\lambda_j^2)
\geq c_1(n,k)\sum_{i=1}^nF_i(\lambda)
\quad\text{for any}\ j\in\mathbb Z_{[1,n]}.
\end{equation}
Moreover, if $\lambda_j<0$, then
\begin{equation}\label{eqn.Fj>}
F_j(\lambda)\geq c_2(n,k)\sum_{i=1}^nF_i(\lambda),
\end{equation}
where $c_1=c_1(n,k)$ and $c_2=c_2(n,k)$
are positive constants.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
On the level set $\{F=1\}$,
the estimates for $1\leq\ell\leq n$ follow from the proof of \cite[Lemma 2.2]{W26}.
  
When $\ell=0$, we have $k=2$ and $F=\sigma_2$,
so the estimates follow from \cite[Lemma 2.1]{LW26b}
and the identity $\sum_{i=1}^nF_i=(n-1)\sigma_1$.
  
Let $c_1^*=c_1^*(n,k)$ and $c_2^*=c_2^*(n,k)$
denote the corresponding positive constants on $\{F=1\}$.
  
Let
\[
A:=F(\lambda)^{1/(k-\ell)},
\quad
\mu:=\frac{\lambda}{A}.
\]
Since $F$ is homogeneous of degree $k-\ell$, we have
\[
\mu\in\Gamma_k,
\quad
F(\mu)=1,
\quad
F_i(\mu)=A^{1-(k-\ell)}F_i(\lambda).
\]
Applying \cite[Lemma 2.2]{W26} to $\mu$, we obtain
\[
F_j(\mu)(1+\mu_j^2)
\geq c_1^*\sum_{i=1}^nF_i(\mu)
\quad\text{and}\quad
F_j(\mu)\geq c_2^*\sum_{i=1}^nF_i(\mu)
\quad\text{if}\ \lambda_j<0.
\]
  
Rescaling back to $\lambda$ and using $3/4\leq A\leq1$, we get
\begin{align*}
F_j(\lambda)(1+\lambda_j^2)
&\geq F_j(\lambda)(A^2+\lambda_j^2)
=A^{k-\ell+1}F_j(\mu)(1+\mu_j^2)
\\
&\geq c_1^*A^2\sum_{i=1}^nF_i(\lambda)
\geq\frac9{16}c_1^*\sum_{i=1}^nF_i(\lambda).
\\
F_j(\lambda)
&\geq c_2^*\sum_{i=1}^nF_i(\lambda)
\quad\text{if}\ \lambda_j<0.
\end{align*}
This completes the proof with $c_1:=\frac9{16}c_1^*$ and $c_2:=c_2^*$.

<!-- lemma-4: body_md -->
\label{lem.seed-F}
There exist constants $r=r(n,k,K)\in(0,1/32)$, $\varepsilon_0=\varepsilon_0(n,k,K)>0$,
and a point $x_*\in\overline{B_{1/32}}$ such that

<!-- end lemma-4 -->
<!-- proof-2: body_md -->
Fix $\rho:=1/32$.
Since $\zeta=1$ in $B_\rho$, we have $F(D^2v)=\left(3/4\right)^{k-\ell}$ in $B_\rho$.
  
By the concavity of $F^{1/(k-\ell)}$, we obtain
\begin{align}
\sum_{i,j=1}^nF^{ij}(D^2v)(v-u)_{ij}
&\leq(k-\ell)F(D^2v)^{1-\frac{1}{k-\ell}}
\left(F(D^2v)^{\frac1{k-\ell}}-F(D^2u)^{\frac1{k-\ell}}\right)
\nonumber\\
&=-\frac{k-\ell}{4}\left(\frac34\right)^{k-\ell-1}
=:-\varsigma<0.
\label{eqn.polarized}
\end{align}
  
Write $\sigma_m^{ij}(M):=\frac{\partial\sigma_m(M)}{\partial M_{ij}}$ for any $m\in\mathbb Z_{[0,k]}$ and $\sigma_0^{ij}:=0$.
Set
\[
\widetilde F^{ij}
:=\sigma_\ell(D^2v)F^{ij}(D^2v)
=\sigma_k^{ij}(D^2v)-F\sigma_\ell^{ij}(D^2v)
\quad\text{in}\ B_\rho.
\]
Then $(\widetilde F^{ij})>0$.
  
Differentiating and using the symmetry of the third derivatives,
we obtain
\begin{equation}\label{eqn.DwtF=0}
\sum_{i=1}^n\partial_i\sigma_m^{ij}(D^2v)=0,
\quad
\sum_{i=1}^n\partial_i\widetilde F^{ij}=0\quad\text{for all}\ j\in\mathbb Z_{[1,n]}.
\end{equation}
  
Moreover,
by Property \ref{prop.inequality}, it follows that
\begin{equation}\label{eqn.Fii-upper}
0<\sum_{i=1}^n\widetilde F^{ii}\leq(n-k+1)\sigma_{k-1}(D^2v).
\end{equation}
  
The Newton--Maclaurin inequalities and $F(D^2v)=\left(3/4\right)^{k-\ell}$
imply
\begin{equation}\label{eqn.sigma-k-1>}
\sigma_\ell(D^2v)\geq c(n,k)>0\quad\text{in}\ B_\rho.
\end{equation}

Next, choose $\eta\in C_c^\infty(B_1)$ satisfying
$\eta\geq0$ in $B_1$, $\int_{B_1}\eta=1$ and $\left\VertD^2\eta\right\Vert_{L^\infty(B_1)}\leq C(n)$,
and set $\eta_\rho(x):=\rho^2\eta(x/\rho)$.
Then $\eta_\rho\in C_c^\infty(B_\rho)$, $\int_{B_\rho}\eta_\rho=\rho^{n+2}$
and $\left\VertD^2\eta_\rho\right\Vert_{L^\infty(B_\rho)}\leq C(n)$.
  
Multiplying \eqref{eqn.polarized} by $\eta_\rho\sigma_\ell(D^2v)$,
integrating by parts twice,
and using \eqref{eqn.Fii-upper}, \eqref{eqn.DwtF=0} and \eqref{eqn.sigma-k-1>}, we obtain
  
\begin{align*}
c\varsigma\rho^{n+2}
&\leq\varsigma\int_{B_\rho}\eta_\rho\sigma_\ell(D^2v)
\leq-\int_{B_\rho}\eta_\rho\sum_{i,j=1}^n\widetilde F^{ij}(v-u)_{ij}
\\
&=\int_{B_\rho}\sum_{i,j=1}^n\widetilde F^{ij}(\eta_\rho)_j(v-u)_i
=-\int_{B_\rho}(v-u)\sum_{i,j=1}^n\widetilde F^{ij}(\eta_\rho)_{ij}
\\
&\leq C(n)\sup_{B_\rho}(v-u)
\int_{B_\rho}\sum_{i=1}^n\widetilde F^{ii}
\\
&\leq C(n,k)\sup_{B_\rho}(v-u)
\int_{B_\rho}\sigma_{k-1}(D^2v).
\end{align*}
  
We now estimate $\int_{B_\rho}\sigma_{k-1}(D^2v)$.
  
Set $\rho_j:=\rho+\frac{j\rho}{k-1}$ for $j\in\mathbb Z_{[0,k-1]}$.
Then $\rho_0=\rho$, $\rho_{k-1}=2\rho$, and $B_{2\rho}\subset B_1$.
  
For each $j\in\mathbb Z_{[0,k-2]}$, let $m:=k-1-j\geq1$
and choose $\xi_j\in C_0^\infty(B_{\rho_{j+1}})$
such that $0\leq\xi_j\leq1$ in $B_{\rho_{j+1}}$, $\xi_j=1$ in $B_{\rho_j}$,
and $\left\vertD\xi_j\right\vert\leq\frac{C(n,k)}{\rho}$ in $B_{\rho_{j+1}}$.
  
Using the Newton-tensor identities and $|Dv|\leq K$ in $B_1$,
we obtain
\begin{align*}
\int_{B_{\rho_j}}\sigma_m(D^2v)
&\leq\int_{B_{\rho_{j+1}}}\xi_j\sigma_m(D^2v)
=\frac{1}{m}\int_{B_{\rho_{j+1}}}\xi_j\sum_{p,q=1}^n\sigma_m^{pq}(D^2v)v_{pq}
\\
&=-\frac{1}{m}\int_{B_{\rho_{j+1}}}\sum_{p,q=1}^n(\xi_j)_p\sigma_m^{pq}(D^2v)v_q
\leq\frac{C(n,k)K}{\rho}\int_{B_{\rho_{j+1}}}\sum_{p=1}^n\sigma_m^{pp}(D^2v)
\\
&=\frac{C(n,k)K(n-m+1)}{\rho}\int_{B_{\rho_{j+1}}}\sigma_{m-1}(D^2v).
\end{align*}
  
Iterating gives
\begin{align*}
\int_{B_\rho}\sigma_{k-1}(D^2v)
\leq\frac{C}{\rho}\int_{B_{\rho_1}}\sigma_{k-2}(D^2v)
\leq\cdots
\leq\left(\frac{C}{\rho}\right)^{k-1}\int_{B_{2\rho}}\sigma_0(D^2v)
\leq C(n,k,K)\rho^{n-k+1}.
\end{align*}
  
Combining this with the preceding estimate gives
\[
\sup_{B_\rho}(v-u)
\geq c(n,k,K)\rho^{k+1}=:c_0>0.
\]
Choose $x_*\in\overline{B_\rho}$ such that $v(x_*)-u(x_*)\geq c_0$.
Take $r=r(n,k,K)\in(0,1/32)$ sufficiently small that
$Kr\leq c_0/2$, and set $\varepsilon_0:=c_0/2$.
Then $B_r(x_*)\subset B_{1/16}$, and the gradient bound \eqref{eqn.gradient-v-u} gives
\begin{align*}
v(x)-u(x)
&\geq v(x_*)-u(x_*)
-\left\VertD(v-u)\right\Vert_{L^\infty(B_1)}|x-x_*|
\\
&\geq c_0-Kr\geq\varepsilon_0
\quad\text{for any}\ x\in B_r(x_*).
\end{align*}
This completes the proof.

The seed lemma also follows from a compactness argument,
without using the divergence-free structure of the Newton tensors.
  
By the gradient estimate,

\quad\text{in}\ B_\rho.

after passing to a subsequence,
$u_j\to u_\infty$ locally uniformly in $B_2$.

F(D^2u_\infty)=1,
\quad
F(D^2u_\infty)=\left(3/4\right)^{k-\ell}
\quad\text{in}\ B_\rho\ \text{in the viscosity sense}.
\]

<!-- lemma-5: body_md -->
\label{lem.barrier}
For any $y\in B_{1/2}$,
the parameters $\alpha,\beta,\gamma$ can be chosen depending only on $n,k,K$
such that
\begin{equation}\label{eqn.barrier-positive}
L_v\psi_y>0\quad\text{in}\ \Omega_y\setminus B_r(y).
\end{equation}

<!-- proof-3: body_md -->
After a rotation of coordinates, we may assume that
\[
D^2v(x)
=\operatorname{diag}(\lambda_1,\lambda_2,\ldots,\lambda_n),
\quad
\lambda_1\geq\lambda_2\geq\cdots\geq\lambda_n.
\]
All quantities below are evaluated at $x$.
  
By the homogeneity and differentiation of equation $F(D^2v)=\left(1-\zeta/4\right)^{k-\ell}$,
we have
\begin{gather*}
L_vv=(k-\ell)\left(1-\frac\zeta4\right)^{k-\ell},
\\
\sum_{i,j=1}^nF^{ij}(D^2v)v_{ijl}
=-(k-\ell)\left(1-\frac\zeta4\right)^{k-\ell-1}\frac{\zeta_l}{4}
\quad\text{for all}\ l\in\mathbb Z_{[1,n]}.
\end{gather*}
Consequently,
\begin{align*}
(\varphi_y)_i
&=\left(x_i-y_i-2\beta v_i\right)\lambda_i+\alpha(x_i-y_i),
\\
(\varphi_y)_{ij}
&=v_{ij}+\sum_{l=1}^n(x_l-y_l-2\beta v_l)v_{ijl}+\alpha\delta_{ij}-2\beta\sum_{l=1}^n v_{li}v_{lj},
\\
L_v\psi_y
&=\frac{\psi_y+1}{\gamma}
\bigg[\frac{1}{\gamma}\sum_{i=1}^nF_i(\varphi_y)_i^2
+2\beta\sum_{i=1}^nF_i\lambda_i^2
-\alpha\sum_{i=1}^nF_i
\\
&\qquad
-(k-\ell)\left(1-\frac{\zeta}{4}\right)^{k-\ell-1}
\left(1-\frac{\zeta}{4}
-\frac{1}{4}(x-y-2\beta Dv)\cdot D\zeta\right)
\bigg].
\end{align*}
  
Set $\mathds 1:=(1,1,\ldots,1)\in\mathbb R^n$.
  
Since $F^{1/(k-\ell)}$ is concave and homogeneous in $\Gamma_k$,
we have
\begin{align*}
\frac{1}{c_*}:=F^{1/(k-\ell)}(\mathds1)
&\leq F^{1/(k-\ell)}(\lambda)+DF^{1/(k-\ell)}(\lambda)\cdot(\mathds1-\lambda)
\\
&=\sum_{i=1}^nF^{1/(k-\ell)}_i(\lambda)
=\frac{\sum_{i=1}^nF_i}{(k-\ell)\left(1-\zeta/4\right)^{k-\ell-1}},
\end{align*}
and hence
\[
\sum_{i=1}^nF_i\geq\frac{(k-\ell)}{c_*}\left(1-\frac{\zeta}{4}\right)^{k-\ell-1}.
\]
 
Moreover, the bounds $0\leq\zeta\leq1$, $|Dv|+|D\zeta|\leq K$ give
\[
1-\frac{\zeta}{4}
-\frac14(x-y-2\beta Dv)\cdot D\zeta
\leq1+\left(r+\frac{\beta K}{2}\right)K=:c_3.
\]
Consequently, setting $\alpha_1:=\alpha+c_*c_3$,
we obtain
\begin{equation}\label{eqn.Lvpsi-lower}
L_v\psi_y
\geq\frac{\psi_y+1}{\gamma}
\left(\frac{1}{\gamma}\sum_{i=1}^nF_i(\varphi_y)_i^2+2\beta\sum_{i=1}^nF_i\lambda_i^2
-\alpha_1\sum_{i=1}^nF_i\right).
\end{equation}
Therefore, to prove $L_v\psi_y>0$, it suffices to prove
\begin{equation}\label{eqn.Q>Fi}
Q:=\frac{1}{\gamma}\sum_{i=1}^nF_i(\varphi_y)_i^2+2\beta\sum_{i=1}^nF_i\lambda_i^2
>\alpha_1\sum_{i=1}^nF_i.
\end{equation}
  
Since $|x-y|\geq r$, there exists $j\in\mathbb Z_{[1,n]}$ such that
\[
|x_j-y_j|\geq|x-y|/\sqrt n\geq r/\sqrt n.
\]
Set $z_j:=x_j-y_j-2\beta v_j$.
Then $(\varphi_y)_j=z_j\lambda_j+\alpha(x_j-y_j)$.
  
Choosing $\beta:=\frac{r}{16K\sqrt n}$, we have
\[
2\beta|v_j|\leq2\beta K
=\frac{r}{8\sqrt n}
\leq\frac18|x_j-y_j|.
\]
Hence
\begin{gather}
(x_j-y_j)z_j
=(x_j-y_j)^2-2\beta\left(x_j-y_j\right)v_j
\geq\frac78|x_j-y_j|^2>0,
\label{eqn.zj-bound}
\\
\frac78|x_j-y_j|\leq|z_j|
\leq\frac98|x_j-y_j|.
\nonumber
\end{gather}
  
Next, we distinguish two cases.

\textit{Case 1. $\lambda_j\geq0$}.
By \eqref{eqn.zj-bound},
the terms $z_j\lambda_j$ and $\alpha(x_j-y_j)$ have the same sign.
  
By $\alpha\geq1$ and Lemma \ref{lem.Fj-lambdaj},
there exists $c_1=c_1(n,k)>0$ such that
\[
Q
\geq\frac{F_j}{\gamma}(\varphi_y)_j^2
\geq\frac{F_j}{\gamma}\left(\frac78\lambda_j+\alpha\right)^2|x_j-y_j|^2
\geq\frac{49r^2}{64n\gamma}F_j(1+\lambda_j^2)
\geq\frac{49r^2c_1}{64n\gamma}\sum_{i=1}^nF_i.
\]
Hence \eqref{eqn.Q>Fi} holds whenever $\gamma<\frac{49r^2c_1}{64n\alpha_1}$.

\textit{Case 2. $\lambda_j<0$}.
Take $\gamma$ with $\gamma\leq\frac{r^2}{2\beta n}=\frac{8Kr}{\sqrt n}$.
Then $2\beta\gamma\leq(x_j-y_j)^2$.
By Lemma \ref{lem.Fj-lambdaj},
there exists $c_2=c_2(n,k)>0$ such that
\begin{align*}
Q
\geq\frac{1}{\gamma}F_j(\varphi_y)_j^2+2\beta F_j\lambda_j^2
\geq\frac{2\beta\alpha^2(x_j-y_j)^2}{z_j^2+2\beta\gamma}F_j
\geq\frac{1}{2}\beta\alpha^2F_j
\geq\frac{1}{2}\beta\alpha^2 c_2\sum_{i=1}^nF_i.
\end{align*}
Choose $\alpha\geq\max\left\{1,c_*c_3,\frac{8}{\beta c_2}\right\}$.
Then $\frac{1}{2}\beta\alpha^2 c_2>2\alpha\geq\alpha+c_*c_3=\alpha_1$, and hence \eqref{eqn.Q>Fi} holds.
  
Since $\sum_{i=1}^nF_i>0$,
\eqref{eqn.Lvpsi-lower} and \eqref{eqn.Q>Fi} yield $L_v\psi_y(x)>0$.

\partial U_y\subset\partial\Omega_y\cup\partial B_r(y).

$\delta:=\varepsilon_1/2$,

<!-- proof-7: body_md -->
Choose $d_0=d_0(n,K)\in(0,1/4)$ sufficiently small
such that
\[
C(n,K)d_0^{1/2}\leq\varepsilon_1/4,
\]
where $C(n,K)$ comes from Lemma \ref{lem.boundary-decay}.
  
By Lemma \ref{lem.separation},
$v-u\geq\varepsilon_1>\delta$ in $B_{1/2}$,
hence $B_{1/2}\subset\left\{v-u>\delta\right\}=\left\{w>u\right\}$.
  
For any $x\in B_{3/2}$,
if $d_x\leq d_0$, then $|x|=\frac32-d_x\geq\frac32-d_0>\frac54$.
Hence it follows from Lemma \ref{lem.boundary-decay} that
\begin{equation}
(v-u)(x)\leq\frac{\varepsilon_1}{4},
\quad
\text{whenever}\ d_x\leq d_0.
\end{equation}
  
Thus, the set $\left\{v-u>\delta\right\}=\left\{w>u\right\}$ cannot intersect $\left\{d_x\leq d_0\right\}$.
This shows that $\Omega\Subset B_{3/2-d_0}$,
and hence $B_{d_0/2}(x)\subset B_{3/2}$ for any $x\in\Omega$.
  
Therefore,
\[
w>u\quad\text{in}\ \Omega,\
\quad
w=u\quad\text{on}\ \partial\Omega,\
\quad
w-u\geq\delta\quad\text{in}\ B_{1/2}.
\]
   
Applying the gradient estimate (Lemma \ref{lem.gradient-estimate}) in $B_{d_0/2}(x)$, we get
\[
|Dv(x)|
\leq C(n,k,K_0,d_0)\left\Vertv\right\Vert_{L^\infty(B_{d_0/2}(x))}
\leq C(n,k,K).
\]
Thus
\[
\left\VertDw\right\Vert_{L^\infty(\Omega)}=\left\VertDv\right\Vert_{L^\infty(\Omega)}\leq C(n,k,K).
\]
This completes the proof.

<!-- proof-8: body_md -->
Set $\varphi:=w-u$ in $\Omega$.
Consider the auxiliary function
(similar to those in \cite[(7.4)]{QY26b} and \cite[page 10]{LW26b})
\[
\Phi(x,\xi):=\varphi^2(x)
\exp\left(\frac\tau2|Du(x)|^2\right)u_{\xi\xi}(x)
\quad\text{for any}\ (x,\xi)\in\overline\Omega\times\mathbb S^{n-1},
\]
where $\tau\in(0,1]$ will be chosen later.
Since $\Phi=0$ on $\partial\Omega\times\mathbb S^{n-1}$,
$\Phi$ attains a positive maximum
at some $(x_0,\xi_0)\in\Omega\times\mathbb S^{n-1}$.
Next, we distinguish two cases.

\textit{Case 1. $x_0\in\overline{B_{1/4}}$.}
Since $w-u\leq2K_0$ and $|Du|\leq K$,
\begin{equation}\label{eqn.inner-maximum}
\Phi(x_0,\xi_0)\leq C(n,k,K)\sup_{B_{1/4}}b.
\end{equation}

\textit{Case 2. $x_0\notin\overline{B_{1/4}}$.}
Since $\operatorname{supp}\zeta\Subset B_{1/4}$, we have $\zeta(x_0)=0$.
The concavity of $F^{1/(k-\ell)}$ gives
\begin{align}
L_u\varphi
=L_u(w-u)
&=L_u(v-u)
=F^{ij}(D^2u)(v-u)_{ij}
\nonumber\\
&\geq(k-\ell)\left(F(D^2u)\right)^{1-\frac1{k-\ell}}
\left(F^{\frac1{k-\ell}}(D^2v)-F^{\frac1{k-\ell}}(D^2u)\right)
\nonumber\\
&=-\frac{k-\ell}{4}\zeta(x_0)
=0.
\label{eqn.Luvp>=0}
\end{align}
  
After a rotation of coordinates,
we may assume that $\xi_0=e_1$ and
\[
D^2u(x_0)
=\operatorname{diag}(\lambda_1,\lambda_2,\ldots,\lambda_n),
\quad
\lambda_1=\lambda_2=\cdots=\lambda_m>\lambda_{m+1}\geq\cdots\geq\lambda_n.
\]
  
All subsequent quantities are evaluated at $x_0$.
Since $u_{11}(x_0)>0$, the smooth function
\[
\Psi(x):=\log\Phi(x,e_1)
=2\log\varphi+\frac\tau2|Du|^2+\log u_{11}
\]
attains a local maximum at $x_0$.
  
Thus
\begin{align}
0
&=\Psi_i
=2\frac{\varphi_i}{\varphi}+\tau u_i \lambda_i+\frac{u_{11i}}{\lambda_1}
\quad
\text{for all}\ i\in\mathbb Z_{[1,n]},
\label{eqn.Pogorelov-first}
\\
0
\geq F^{ij}\Psi_{ij}
&=\frac{2F^{ii}\varphi_{ii}}{\varphi}
-\frac{2F^{ii}\varphi_i^2}{\varphi^2}
+\tau u_kF^{ii}u_{kii}
+\tau F^{ii}\lambda_i^2
+\frac{F^{ii}u_{11ii}}{\lambda_1}
-\frac{F^{ii}u_{11i}^2}{\lambda_1^2}
\nonumber\\
&\geq
-\frac{2F^{ii}\varphi_i^2}{\varphi^2}
+\tau F^{ii}\lambda_i^2
+\frac{F^{ii}u_{11ii}}{\lambda_1}
-\frac{F^{ii}u_{11i}^2}{\lambda_1^2},
\label{eqn.Pogorelov-second}
\end{align}
where the last inequality follows from \eqref{eqn.Luvp>=0}
and $F^{ii}u_{iil}=\partial_{x_l}F=0$ for all $l\in\mathbb Z_{[1,n]}$.
  
Differentiating the equation $F(D^2u)=1$ twice in the $x_1$-direction
and using the concavity of $F^{1/(k-\ell)}$,
we obtain
\begin{align}
F^{ii}u_{ii11}
=-F^{ij,rs}u_{ij1}u_{rs1}
&=-F^{ii,jj}u_{ii1}u_{jj1}
+2\sum_{i<j}\frac{F^{jj}-F^{ii}}{\lambda_i-\lambda_j}u_{ij1}^2
\geq2\sum_{i>m}\frac{F^{ii}-F^{11}}{\lambda_1-\lambda_i}u_{11i}^2.
\label{eqn.Fii-uii11}
\end{align}
  
For any $i>m$, since $\lambda_1>0$ and $\lambda_1-\lambda_i>0$, we have
\begin{equation}\label{eqn.lambda-ratio}
\frac{2\lambda_1}{\lambda_1-\lambda_i}
\geq2+\frac{2\lambda_i}{\lambda_1}
\geq\frac{7}{4}-\frac{4\lambda_i^2}{\lambda_1^2}.
\end{equation}
Thus, using $F^{ii}\geq F^{11}>0$ for all $i>m$
and $F^{11}=\frac{F^{ii}\lambda_i^2}{\lambda_1^2}$ for all $i\leq m$,
we have
\begin{align*}
2\sum_{i>m}\frac{F^{ii}-F^{11}}{\lambda_1(\lambda_1-\lambda_i)}u_{11i}^2
-\sum_{i=1}^nF^{ii}\frac{u_{11i}^2}{\lambda_1^2}
\geq\sum_{i=1}^n\left(\frac{3}{4}F^{ii}-\frac{7}{4}F^{11}-\frac{4}{\lambda_1^2}F^{ii}\lambda_i^2\right)
\frac{u_{11i}^2}{\lambda_1^2}.
\end{align*}
  
Substituting this inequality and \eqref{eqn.Fii-uii11} into \eqref{eqn.Pogorelov-second},
we obtain
\begin{align}
0
\geq-\frac{2F^{ii}\varphi_i^2}{\varphi^2}
+\tau F^{ii}\lambda_i^2
+\frac{3}{4}F^{ii}\frac{u_{11i}^2}{\lambda_1^2}
-\frac{7}{4}F^{11}\sum_{i=1}^n\frac{u_{11i}^2}{\lambda_1^2}
-\frac{4}{\lambda_1^2}F^{ii}\lambda_i^2\frac{u_{11i}^2}{\lambda_1^2}.
\label{eqn.Pogorelov-third}
\end{align}
By the equality \eqref{eqn.Pogorelov-first},
for any $i\in\mathbb Z_{[1,n]}$,
\begin{gather*}
\frac{3}{4}\frac{u_{11i}^2}{\lambda_1^2}-2\frac{\varphi_i^2}{\varphi^2}
=\frac{3}{4}\frac{u_{11i}^2}{\lambda_1^2}-\frac{1}{2}\left(\tau u_i \lambda_i+\frac{u_{11i}}{\lambda_1}\right)^2
=\frac{1}{4}\left(\frac{u_{11i}}{\lambda_1}-2\tau u_i\lambda_i\right)^2
-\frac{3\tau^2}{2}u_i^2\lambda_i^2
\geq-\frac{3\tau^2K^2}{2}\lambda_i^2,
\\
\frac{u_{11i}^2}{\lambda_1^2}
=\left(2\frac{\varphi_i}{\varphi}+\tau u_i \lambda_i\right)^2
\leq\frac{8\varphi_i^2}{\varphi^2}+2\tau^2K^2\lambda_i^2.
\end{gather*}
  
Moreover, since $\lambda(D^2u)\in\Gamma_k\subset\Gamma_2$,
\[
\sum_{i=1}^n\lambda_i^2=(\triangle u)^2-2\sigma_2(D^2u)<(\triangle u)^2\leq n^2\lambda_1^2.
\]
  
Therefore
\begin{align*}
\frac{7}{4}F^{11}\sum_{i=1}^n\frac{u_{11i}^2}{\lambda_1^2}
&\leq\left(\frac{14|D\varphi|^2}{\varphi^2}+\frac{7}{2}n^2\tau^2K^2\lambda_1^2\right)F^{11}
\leq \left(\frac{14|D\varphi|^2}{(\varphi\lambda_1)^2}+\frac{7}{2}n^2\tau^2K^2\right)F^{ii}\lambda_i^2,
\\
\frac{4}{\lambda_1^2}F^{ii}\lambda_i^2\frac{u_{11i}^2}{\lambda_1^2}
&\leq\left(\frac{32|D\varphi|^2}{(\varphi\lambda_1)^2}+8n^2\tau^2K^2\right)F^{ii}\lambda_i^2.
\end{align*}
Combining these estimates with \eqref{eqn.Pogorelov-third}
and choosing $\tau:=\frac{1}{(3+23n^2)(1+K^2)}$,
we obtain
\[
0\geq\left(\tau-\left(\frac{3+23n^2}{2}\right)\tau^2K^2
-\frac{46|D\varphi|^2}{(\varphi\lambda_1)^2}\right)
\sum_{i=1}^nF^{ii}\lambda_i^2
\geq\left(\frac{\tau}{2}-\frac{46|D\varphi|^2}{(\varphi\lambda_1)^2}\right)
\sum_{i=1}^nF^{ii}\lambda_i^2.
\]
Consequently, using the gradient bound on $\Omega$ in Lemma \ref{lem.comparison-domain},
we obtain
\[
\varphi(x_0)\lambda_1(x_0)\leq\sqrt{\frac{92}{\tau}}|D\varphi(x_0)|
\leq C(n,k,K),
\]
and thus
\[
\Phi(x_0,\xi_0)
=\varphi^2(x_0)\exp\left(\frac\tau2|Du(x_0)|^2\right)\lambda_1(x_0)
\leq C(n,k,K_0).
\]
Combining this with \eqref{eqn.inner-maximum}
and $\varphi=w-u\geq\delta$ in $B_{1/2}$ by Lemma \ref{lem.comparison-domain},
we obtain
\begin{equation}\label{eqn.b1/2<b1/4}
\sup_{B_{1/2}}b
\leq\delta^{-2}\max_{\overline\Omega\times\mathbb S^{n-1}}\Phi
=\delta^{-2}\Phi(x_0,\xi_0)
\leq C(n,k,K)\left(1+\sup_{B_{1/4}}b\right)
\leq C(n,k,K)\sup_{B_{1/4}}b,
\end{equation}
where the last inequality follows from $1=F(\lambda)\leq F(b\mathds 1)=b^{k-\ell} F(\mathds1)$.

Finally, for any $y\in B_{1/2}$ and $r\in(0,1/8)$, set
\[
\widetilde u(x)
:=\frac{u(y+4rx)-u(y)}{(4r)^2}
\quad\text{for any}\ x\in B_2.
\]
Then $D^2\widetilde u(x)=D^2u(y+4rx)$, $F(D^2\widetilde u)=1$ in $B_2$,
and $\left\Vert\widetilde u\right\Vert_{C^1(B_2)}\leq K_0/(2r)$.
Applying the inequality \eqref{eqn.b1/2<b1/4} to $\widetilde u$,
we obtain
\[
\sup_{B_{2r}(y)}b
=\sup_{B_{1/2}}\lambda_{\max}(D^2\widetilde u)
\leq
C(n,k,K,r)\sup_{B_{1/4}}\lambda_{\max}(D^2\widetilde u)
=C(n,k,K_0,r)\sup_{B_r(y)}b.
\]
This completes the proof.

<!-- proposition-1: body_md -->
\label{prop.general-Hessian}
Let $n\geq2$,
and let $\Gamma\subset\mathbb R^n$ be an open symmetric convex cone with vertex at the origin
satisfying $\Gamma_n\subset\Gamma$.
  
Assume that $f\in C^2(\Gamma)\cap C^0(\overline\Gamma)$ is a symmetric function
and satisfies the following conditions:
  
\begin{gather}
f>0\ \text{in}\ \Gamma,
\quad
f=0\ \text{on}\ \partial\Gamma,
\label{eqn.nonnegative}
\\
f_i:=\frac{\partial f}{\partial\lambda_i}>0\quad\text{in}\ \Gamma,\
\label{eqn.elliptic}
\\
f\ \text{is a concave function},
\label{eqn.sym-concave}
\\
f\ \text{is homogeneous of degree one}.
\label{eqn.homogeneous}
\end{gather}
Suppose also that there exists $c>0$ such that,
for every $\lambda\in\Gamma$ with $f(\lambda)=1$,
\begin{gather}
f_i(\lambda)\geq c\sum_{j=1}^nf_j(\lambda)
\quad\text{whenever}\ \lambda_i<0,
\label{eqn.general-negative}
\\
f_i(\lambda)\left(1+\lambda_i^2\right)\geq c\sum_{j=1}^nf_j(\lambda)
\quad\text{for all}\ i\in\mathbb Z_{[1,n]}.
\label{eqn.general-coefficient}
\end{gather}
  
If $u\in C^4(B_2)$ is a solution of $F(D^2u):=f(\lambda(D^2u))=1$ with $\lambda(D^2u)\in\Gamma$ in $B_2$,
then
\[
\sup_{B_{1/2}}|D^2u|
\leq C\left(n,f,\Gamma,c,\|u\|_{L^\infty(B_2)}\right).
\]
<!-- end proposition-1 -->
<!-- remark-6: body_md -->
The assumptions on $\Gamma$
and conditions \eqref{eqn.nonnegative}--\eqref{eqn.sym-concave}
are standard in the Dirichlet solvability theory of Hessian equations and curvature equations;
see \cite{CNS85,Tru90,Tru95}.
  
Homogeneity \eqref{eqn.homogeneous}
provides the radial growth required for solvability,
gives Euler's identity,
and allows conditions \eqref{eqn.general-negative} and \eqref{eqn.general-coefficient} on $\{f=1\}$
to be extended by rescaling to $\{3/4\leq f\leq1\}$.
  
The condition \eqref{eqn.general-negative}
is a standard hypothesis used in gradient estimates;
see \cite{L91,SUW04,D20}.
  
The condition \eqref{eqn.general-coefficient}
supplies the additional control needed for the propagation barrier.

By homogeneity, \eqref{eqn.general-negative} holds throughout $\Gamma$,
and \eqref{eqn.general-coefficient} holds on $3/4\leq f\leq1$
after decreasing $c$.
  
Furthermore, conditions \eqref{eqn.nonnegative}--\eqref{eqn.homogeneous}
and \eqref{eqn.general-negative}
allow us to apply the interior gradient argument in
\cite[proof of Theorem  1.1]{D20} to $u$.
Thus, by translation and scaling,
it suffices to prove the corresponding estimate in terms of $\left\Vertu\right\Vert_{C^1(B_2)}$.

Set $K_0:=1+\left\Vertu\right\Vert_{C^1(B_2)}$
and $L_u:=F^{ij}(D^2u)\partial_{ij}$,
with $L_v$ defined similarly.
  
By conditions \eqref{eqn.nonnegative}--\eqref{eqn.homogeneous} and $\Gamma_n\subset\Gamma$,
the solvability theorem \cite[Theorem  1.2]{Tru95} gives an admissible solution
$v\in C^{3,\alpha}(\overline{B_{3/2}})$, for some $\alpha\in(0,1)$, of
\[
F(D^2v)=1-\zeta/4\quad\text{in}\ B_{3/2},
\quad v=u\quad\text{on}\ \partial B_{3/2},
\]
where $\zeta$ is the fixed cutoff of Section  3.
  
By convexity and symmetry of $\Gamma$,
together with $\Gamma_n\subset\Gamma$ and $0\in\partial\Gamma$,
we have $\Gamma\subset\Gamma_1$.
  
The comparison principle and $\triangle v>0$
give $u\leq v$ in $B_{3/2}$.
  
Applying the same gradient argument to $v$,
we choose $K=K(n,f,\Gamma,c,K_0)\geq1$ such that
\begin{equation}\label{eqn.gradient-F-f}
\left\VertDv\right\Vert_{L^\infty(B_1)}
+\left\Vertu\right\Vert_{C^1(B_2)}
+\left\Vert\zeta\right\Vert_{C^1(B_{3/2})}\leq K.
\end{equation}
  
For admissible solutions of $F(D^2u_j)=1$,
the inclusion $\Gamma\subset\Gamma_1$
gives Alexandrov-type differentiability of locally uniform limits,
as in \cite[proof of Lemma  5.1]{W26}.
  
Together with the closedness of viscosity solutions,
this yields the seed separation in Lemma  \ref{lem.seed-F}
by the compactness argument in Remark \ref{rem.seed-compactness}.

Next, Euler's identity and concavity imply
\[
\sum_{i=1}^n f_i(\lambda)\lambda_i=f(\lambda),
\quad
\sum_{i=1}^n f_i(\lambda)\geq f(\mathds1)>0,
\quad
L_v(v-u)\leq-\zeta/4\leq L_u(v-u).
\]
Using these inequalities,
the seed separation,
the inequalities \eqref{eqn.general-coefficient}, \eqref{eqn.general-negative} and $\triangle v>0$,
we can repeat the propagation and comparison-domain arguments
in Subsections  \ref{subsec.barrier}--\ref{subsec.comparison}.
  
Furthermore,
the doubling inequality in Lemma  \ref{lem.doubling}
follows from the same calculation,
using symmetry, ellipticity and concavity, $L_u(v-u)\geq0$ outside $\operatorname{supp}\zeta$,
$|D^2u|\leq n\lambda_{\max}(D^2u)$, and the inequality
\[
1=F(D^2u)
\leq f\left(\lambda_{\max}(D^2u)\mathds1\right)
=\lambda_{\max}(D^2u)f(\mathds1).
\]

Finally, using the differentiability above,
Savin's theorem \cite[Theorem  1.3]{Sav07},
and the doubling inequality,
we complete the compactness argument
as in the proof of Theorem  \ref{thm.Hessian-estimate}.
