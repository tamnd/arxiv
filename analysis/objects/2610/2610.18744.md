\label{lem:regularity}
For every nonzero finite measure $\lambda_N$ supported on $\mathsf{B}^N$ and all $(t,q),(t',q')\in\mathbb{R}_+\times\mathcal{Q}_1$, we have
\begin{equation}\label{eq:regularity}
 |\overline F_{N,\lambda_N}(t,q)-\overline F_{N,\lambda_N}(t',q')|
 \leqslant M_0|t-t'|+|q-q'|_{L^1}.
\end{equation}
The restriction of $\overline F_{N,\lambda_N}$ to $\mathbb{R}_+\times\mathcal{Q}_2$ is Fr\'echet differentiable relative to its domain. 
The derivative $\partial_q\overline F_{N,\lambda_N}(t,q)$ belongs to $\mathcal Q_\infty$ viewed as a path
\begin{align}\label{e.path_view_of_dqF}
    [0,1)\ni s \quad\mapsto \quad \partial_q\overline F_{N,\lambda_N}(t,q,s) \in S^D_+
\end{align}
and satisfies $\left|\partial_q\overline F_{N,\lambda_N}(t,q)\right|_{L^\infty}\leqslant 1$.
For every $t\in\mathbb{R}_+$, $q\in\mathcal{Q}_\infty$, and $\kappa\in L^2([0,1];S^D)$, we have
\begin{equation}\label{eq:replica-derivatives}
  \partial_t\overline F_{N,\lambda_N}(t,q)
  =\mathbb{E}\langle\xi(\mathsf R_N(\sigma,\sigma'))\rangle \quad\text{and}\quad
  \langle\kappa,\partial_q\overline F_{N,\lambda_N}(t,q)\rangle_{L^2}
  =\mathbb{E}\langle\kappa(\alpha\wedge\alpha')\cdot \mathsf R_N(\sigma,\sigma')\rangle.
\end{equation}
For fixed $N\in\mathbb{N}$ and $t\in\mathbb{R}_+$, we have
\begin{equation}\label{eq:gradient-continuity}
 |\partial_q\overline F_{N,\lambda_N}(t,q)-\partial_q\overline F_{N,\lambda_N}(t,q')|_{L^1}
 \leqslant16N|q-q'|_{L^1},
 \qquad q,q'\in\mathcal{Q}_2.
\end{equation}
For fixed $N\in\mathbb{N}$ and $t\in\mathbb{R}_+$, the function $q\mapsto\overline F_{N,\lambda_N}(t,q)$ is $\mathcal{Q}_2^*$-increasing on $\mathcal{Q}_2$.

<!-- lemma-3: body_md -->
\label{lem:coupling}
Let $\lambda_N$ and $\lambda_N'$ be probability measures on $\mathsf{B}^N$. Suppose that they admit a coupling supported on pairs $(\sigma,\tau)$ satisfying $N^{-1}\sum_{i=1}^N|\sigma_{\cdot,i}-\tau_{\cdot,i}|\leqslant a$. Then, for every $t\in\mathbb{R}_+$ and $q\in\mathcal{Q}_\infty$, we have
\begin{equation}\label{eq:coupling-bound}
 |\overline F_{N,\lambda_N}(t,q)-\overline F_{N,\lambda_N'}(t,q)|
 \leqslant2a\bigl(tM_1+|q|_{L^\infty}\bigr).
\end{equation}
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
Let $\Pi_N$ be a coupling of $\lambda_N$ and $\lambda_N'$ satisfying the assumption of the lemma. We use $x=(\sigma,\tau,\alpha)$ and $x'=(\sigma',\tau',\alpha')$ for two points in $\mathsf{B}^N\times\mathsf{B}^N\times\operatorname{supp}\mathfrak R$. Conditionally on $\mathfrak R$, let $X_0$ and $X_1$ be independent centered Gaussian processes indexed by this space with covariance kernels
\begin{equation}\label{eq:coupling-covariances}
 \begin{aligned}
  \mathsf C_0(x,x')&:=2tN\xi\bigl(\mathsf R_N(\sigma,\sigma')\bigr)+2Nq(\alpha\wedge\alpha')\cdot \mathsf R_N(\sigma,\sigma'),\\
  \mathsf C_1(x,x')&:=2tN\xi\bigl(\mathsf R_N(\tau,\tau')\bigr)+2Nq(\alpha\wedge\alpha')\cdot \mathsf R_N(\tau,\tau').
 \end{aligned}
\end{equation}
These are the covariance kernels of the centered Gaussian parts of the enriched Hamiltonians evaluated respectively at $\sigma$ and $\tau$. In particular, we have
\begin{align*}
 \frac12\mathsf C_0(x,x)=tN\xi\bigl(\mathsf R_N(\sigma,\sigma)\bigr)+q(1)\cdot\sigma\sigma^\intercal \qquad\text{and}\qquad 
 \frac12\mathsf C_1(x,x)=tN\xi\bigl(\mathsf R_N(\tau,\tau)\bigr)+q(1)\cdot\tau\tau^\intercal.
\end{align*}
Thus the diagonal terms in the enriched Hamiltonian are exactly $-\mathsf C_0(x,x)/2$ and $-\mathsf C_1(x,x)/2$ at the two endpoints of the interpolation.

For $\theta\in[0,1]$, define $\widetilde X_\theta:=\sqrt\theta\,X_0+\sqrt{1-\theta}\,X_1$, $\widetilde{\mathsf C}_\theta:=\theta \mathsf C_0+(1-\theta)\mathsf C_1$, and define $\phi:[0,1]\to\mathbb{R}$ by
\begin{equation*}
 \phi(\theta):=-\frac1N\mathbb{E}\log\iint
 \exp\left(\widetilde X_\theta(x)-\frac12\widetilde{\mathsf C}_\theta(x,x)\right)
 \Pi_N(\mathrm d\sigma,\mathrm d\tau)\mathfrak R(\mathrm d\alpha).
\end{equation*}
Since the first and second marginals of $\Pi_N$ are $\lambda_N$ and $\lambda_N'$, respectively, the endpoint values are
\begin{equation*}
 \phi(1)=\overline F_{N,\lambda_N}(t,q)
 \quad\text{and}\quad
 \phi(0)=\overline F_{N,\lambda_N'}(t,q).
\end{equation*}
Let $\langle\cdot\rangle_\theta$ denote the Gibbs expectation associated with $\phi(\theta)$, and let $x^1,x^2$ be two replicas. Gaussian integration by parts, together with the derivative of the compensation $-\widetilde{\mathsf C}_\theta(x,x)/2$, gives
\begin{equation}\label{eq:coupling-interpolation-derivative}
 \phi'(\theta)=\frac1{2N}\mathbb{E}\left\langle
 (\mathsf C_0-\mathsf C_1)(x^1,x^2)
 \right\rangle_\theta.
\end{equation}

We next bound the covariance difference in~\eqref{eq:coupling-interpolation-derivative}. If $(\sigma,\tau)$ and $(\sigma',\tau')$ belong to the support of $\Pi_N$, then
\begin{align*}
 &|\mathsf R_N(\sigma,\sigma')-\mathsf R_N(\tau,\tau')|\leqslant\frac1N\sum_{i=1}^N
 \left|\sigma_{\cdot,i}(\sigma'_{\cdot,i})^\intercal-\tau_{\cdot,i}(\tau'_{\cdot,i})^\intercal\right|\leqslant\frac1N\sum_{i=1}^N
 \left(|\sigma_{\cdot,i}-\tau_{\cdot,i}|+|\sigma'_{\cdot,i}-\tau'_{\cdot,i}|\right)
 \leqslant2a.
\end{align*}
Every overlap in this display belongs to the unit ball of $\mathbb{R}^{D\times D}$. Hence, the definition of $M_1$ gives
\begin{equation*}
 \left|\xi\bigl(\mathsf R_N(\sigma,\sigma')\bigr)-\xi\bigl(\mathsf R_N(\tau,\tau')\bigr)\right|
 \leqslant2M_1a\quad\text{and}\quad \left|q(u)\cdot\bigl(\mathsf R_N(\sigma,\sigma')-\mathsf R_N(\tau,\tau')\bigr)\right|
 \leqslant2a|q|_{L^\infty}.
\end{equation*}
Using~\eqref{eq:coupling-covariances}, we therefore obtain $|(\mathsf C_0-\mathsf C_1)(x,x')|
 \leqslant4Na\bigl(tM_1+|q|_{L^\infty}\bigr)$.
Substituting this estimate into~\eqref{eq:coupling-interpolation-derivative} yields $|\phi'(\theta)|\leqslant2a\bigl(tM_1+|q|_{L^\infty}\bigr)$ for every $\theta\in[0,1]$.
Integrating over $\theta\in[0,1]$ proves~\eqref{eq:coupling-bound}.

<!-- lemma-4: body_md -->
\label{lem:type-probability}
For every $N\in\mathbb{N}$ and $\nu\in\Delta_{m,N}$,
\begin{equation}\label{eq:stirling}
 0\leqslant-\frac1N\log\mu^{\otimes N}(\Sigma_N(\nu))-\operatorname{Ent}(\nu\mid\mu)
 \leqslant\frac{m\log(N+1)}N.
\end{equation}
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
We have the exact identity 
\begin{equation}\label{eq:type-probability-identity}
 \mu^{\otimes N}(\Sigma_N(\nu))
 =e^{-N\operatorname{Ent}(\nu\mid\mu)}\nu^{\otimes N}(\Sigma_N(\nu)).
\end{equation}
 
To conclude, it suffices to show that $\nu^{\otimes N}(\Sigma_N(\nu)) \geqslant (N+1)^{-m}$. First observe that by multinomial factorization, we have 
 
\begin{equation}
    \sum_{\ell} \frac{N!}{\prod_{j=1}^m \ell_j!} \prod_{j  =1}^m \nu_j^{\ell_j}= (\nu_1+\dots+\nu_m)^N = 1,
\end{equation}
 
where the sum is taken over $\ell \in \{0,\dots,N\}^m$ such that $\sum_j \ell_j = N$. Also observe that the map $\ell \mapsto (N\nu_j)^\ell/\ell!$ is maximized over $\mathbb{N}$ at $\ell = N \nu_j$. Hennce, each term in the sum of the previous display is $\leqslant \nu^{\otimes N}(\Sigma_N(\nu))$. Since there is less than $(N+1)^m$ terms in the sum, we obtain 
 
\begin{equation}
    (N+1)^m \nu^{\otimes N}(\Sigma_N(\nu)) \geqslant \sum_{\ell} \frac{N!}{\prod_{j=1}^m \ell_j!} \prod_{j  =1}^m \nu_j^{\ell_j}= (\nu_1+\dots\nu_m)^N = 1,
\end{equation}
 
thus the result is proven.

<!-- lemma-5: body_md -->
\label{lem:type-continuity}
If $\nu=N^{-1}\sum_{i=1}^N\delta_{x_i}$ and $\nu'=N^{-1}\sum_{i=1}^N\delta_{y_i}$ for $x_i,y_i\in\mathsf{B}$, then for every $(t,q) \in \mathbb{R}_+ \times \mathcal Q_\infty$,

 \leqslant2\bigl(tM_1+|q|_{L^\infty}\bigr)W_1(\nu,\nu').

<!-- lemma-6: body_md -->
\label{lem:Parisi-diffusion}
The equation~\eqref{eq:Parisi-PDE} has a unique classical solution $\Phi$ with bounded spatial derivatives of positive order, and $|\nabla_{\mathbf{x}_{\cdot,i}}\Phi|\leqslant1$ for every $i\in\{1,\dots,N\}$. Moreover, we have
\begin{equation}\label{e.F_N=-1/NEPhi}
 \overline F_N^\nu(t,q)=-\frac1N\mathbb{E}\Phi(0,\sqrt{2q(0)}\mathbf{z}_0;\mathbf{g}).
\end{equation}
The stochastic differential equation
\begin{equation}\label{eq:diffusion}
 \mathrm d\mathbf{X}_s=2s 
\overset{\mbox{\large\bfseries .}}{q}(s)\nabla_{\mathbf{x}}\Phi(s,\mathbf{X}_s;\mathbf{g})\,\mathrm d s+\sqrt{2 
\overset{\mbox{\large\bfseries .}}{q}(s)}\,\mathrm dB_s,
 \quad\text{and}\quad \mathbf{X}_0=\sqrt{2q(0)}\mathbf{z}_0
\end{equation}
admits a unique strong solution on $[0,1]$.
<!-- end lemma-6 -->
<!-- proof-5: body_md -->
In column-stacked coordinates,~\eqref{eq:Parisi-PDE} has covariance path $I_N\otimes q$, coefficient $\alpha(s)=s$, and terminal measure $\Lambda_{\mathbf{g}}$. Existence and regularity follow from~\cite[Proposition 4.7(1), (3), and (4)]{chen2024simultaneous}. Its cascade representation~\cite[Lemma 4.5, (4.10)]{chen2024simultaneous} gives~\eqref{e.F_N=-1/NEPhi} after averaging over $(\mathbf{g},\mathbf{z}_0)$.
                                                                                                                                                                              
Differentiating in $\mathbf{x}_{\cdot,i}$ gives a linear backward equation whose terminal gradient lies in the unit ball. The maximum principle gives the column bound. The bounded Hessian makes the drift of~\eqref{eq:diffusion} globally Lipschitz, giving a unique strong solution; see also~\cite[Proposition 3.2]{chen2023parisi}.
The finite terminal sum and the same regularity estimates justify the parameter differentiations below: derivatives in $(\mathbf{x},\mathbf{g})$ and in bounded terminal perturbations are bounded, while the $t$ derivative has at most linear growth in $\mathbf{g}$ locally on $t>0$.

<!-- lemma-7: body_md -->
\label{lem:Parisi-Hessian}
The following statements hold.
\begin{enumerate}
\item 
The function $\Phi(s,\cdot\,;\cdot)$ is jointly convex in $(\mathbf{x},\mathbf{g})$. 
\item The process $\mathbb{E}[\sigma_1\mid\mathcal G_s]$ is a bounded martingale, and for every $s\in[0,1]$,
\begin{align*}
 \mathbb{E}[\sigma_1\mid\mathcal G_s]=\nabla_{\mathbf{x}}\Phi(s,\mathbf{X}_s;\mathbf{g}).
\end{align*}
\item We have
\begin{align*}
 \nabla_{\mathbf{x}}^2\Phi(s,\mathbf{X}_s;\mathbf{g})=s\mathrm{Cov}(\operatorname{vec}(\sigma_1)\mid\mathcal G_s)+\mathbb{E}\left[\int_s^1\mathrm{Cov}(\operatorname{vec}(\sigma_1)\mid\mathcal G_r)\,\mathrm d r\,\middle|\,\mathcal G_s\right].
\end{align*}
\item We have
\begin{equation*}
    \int_0^1\mathbb{E}\bigl|\nabla_{\mathbf{x}}^2\Phi(s,\mathbf{X}_s;\mathbf{g})\bigr|^2\,\mathrm d s\leqslant\frac{N}{2\varepsilon}.
\end{equation*}
\end{enumerate}
<!-- end lemma-7 -->
<!-- proof-6: body_md -->
For a bounded $v:\Sigma_N(\nu)\to\mathbb{R}$, let $\Phi^y$ have terminal measure $e^{yv}\Lambda_{\mathbf{g}}$. All parameter derivatives below are evaluated at $y=0$ along the unperturbed diffusion. Differentiating~\eqref{eq:Parisi-PDE} once in $y$ and applying It\^o's formula show that the first derivative is a martingale with terminal value $\mathbb{E}[v(\sigma_1)\mid\mathcal G_1]$. Hence
\begin{equation*}
 \left.\partial_y\Phi^y(s,\mathbf{X}_s;\mathbf{g})\right|_{y=0}
 =\mathbb{E}[v(\sigma_1)\mid\mathcal G_s].
\end{equation*}
Differentiating twice shows that the second derivative has drift $-s\,\mathrm d[\mathbb{E}[v(\sigma_1)\mid\mathcal G_\cdot]]_s$, whereas the conditional variance $\operatorname{Var}(v(\sigma_1)\mid\mathcal G_s)$ has drift $-\mathrm d[\mathbb{E}[v(\sigma_1)\mid\mathcal G_\cdot]]_s$. Subtracting $s\operatorname{Var}(v(\sigma_1)\mid\mathcal G_s)$ from the second derivative therefore gives a process with drift $-\operatorname{Var}(v(\sigma_1)\mid\mathcal G_s)\,\mathrm d s$ and zero terminal value. Integration and conditioning give
\begin{equation}\label{eq:scalar-Hessian-identity}
 \begin{aligned}
 \left.\partial_y^2\Phi^y(s,\mathbf{X}_s;\mathbf{g})\right|_{y=0}
 =s\operatorname{Var}(v(\sigma_1)\mid\mathcal G_s)
 +\mathbb{E}\left[\int_s^1\operatorname{Var}(v(\sigma_1)\mid\mathcal G_r)\,\mathrm d r\,\middle|\,\mathcal G_s\right].
 \end{aligned}
\end{equation}
We now prove the four items.

\smallskip
\noindent\emph{Proof of (1):}
For $a\in\mathbb{R}^{D\times N}$ and $b\in\mathbb{R}^{(\operatorname{supp}\mu)^N}$, take
\begin{equation*}
 v(\sigma)=a\cdot\sigma+\sqrt{2t}\,b\cdot\mathbf h_N(\sigma).
\end{equation*}
Uniqueness gives $\Phi^y(s,\mathbf{x};\mathbf{g})=\Phi(s,\mathbf{x}+ya;\mathbf{g}+yb)$, so $\left.\partial_y^2\Phi^y(s,\mathbf{X}_s;\mathbf{g})\right|_{y=0}$ is the second directional derivative in direction $(a,b)$. By~\eqref{eq:scalar-Hessian-identity}, it is nonnegative. Restarting the diffusion from any deterministic $(s,\mathbf{x})$, with $\mathbf{g}$ fixed, proves joint convexity at every $(s,\mathbf{x},\mathbf{g})$.

\smallskip
\noindent\emph{Proof of (2):}
For $v(\sigma)=a\cdot\sigma$ with $a\in\mathbb{R}^{D\times N}$, uniqueness gives $\Phi^{\mathbf y}(s,\mathbf{x};\mathbf{g})=\Phi(s,\mathbf{x}+\mathbf ya;\mathbf{g})$. Thus
\begin{equation*}
 a\cdot\nabla_{\mathbf{x}}\Phi(s,\mathbf{X}_s;\mathbf{g})=\left.\partial_{\mathbf y}\Phi^{\mathbf y}(s,\mathbf{X}_s;\mathbf{g})\right|_{\mathbf y=0}
 =\mathbb{E}[a\cdot\sigma_1\mid\mathcal G_s].
\end{equation*}
Since $a$ is arbitrary, this proves the gradient identity. The conditional-expectation process is a martingale and is bounded because $|\sigma_1|^2\leqslant N$.

\smallskip
\noindent\emph{Proof of (3):}
For the same spatial perturbation, the second parameter derivative is the Hessian quadratic form in direction $a$. Applying~\eqref{eq:scalar-Hessian-identity} with $v(\sigma)=a\cdot\sigma$ gives
\begin{align*}
 \operatorname{vec}(a)^\intercal \nabla_{\mathbf{x}}^2\Phi(s,\mathbf{X}_s;\mathbf{g})\operatorname{vec}(a)
 =s\operatorname{Var}(a\cdot\sigma_1\mid\mathcal G_s)
 +\mathbb{E}\left[\int_s^1\operatorname{Var}(a\cdot\sigma_1\mid\mathcal G_r)\,\mathrm d r\,\middle|\,\mathcal G_s\right].
\end{align*}
Since $a$ is arbitrary, this proves the conditional covariance representation of $\nabla_{\mathbf{x}}^2\Phi(s,\mathbf{X}_s;\mathbf{g})$.

\smallskip
\noindent\emph{Proof of (4):}
Differentiating~\eqref{eq:Parisi-PDE} once in $\mathbf{x}$, applying It\^o's formula, and using Lemma~\ref{lem:Parisi-Hessian}(2) give
\begin{equation}\label{eq:Parisi-gradient-Ito}
 \mathrm d\operatorname{vec}(\mathbb{E}[\sigma_1\mid\mathcal G_s])
 =\nabla_{\mathbf{x}}^2\Phi(s,\mathbf{X}_s;\mathbf{g})\sqrt{2(I_N\otimes 
\overset{\mbox{\large\bfseries .}}{q}(s))}\,\mathrm d\operatorname{vec}(B_s);
\end{equation}
see~\cite[Lemma 3.3]{chen2023parisi} and~\cite[Lemma 5.3]{chen2024simultaneous}. Since $(I_N\otimes 
\overset{\mbox{\large\bfseries .}}{q}(s))\succeq\varepsilon I_{DN}$, It\^o's isometry gives
\begin{equation*}
 2\varepsilon\int_0^1\mathbb{E}\bigl|\nabla_{\mathbf{x}}^2\Phi(s,\mathbf{X}_s;\mathbf{g})\bigr|^2\,\mathrm d s
 \leqslant\mathbb{E}|\mathbb{E}[\sigma_1\mid\mathcal G_1]|^2-\mathbb{E}|\mathbb{E}[\sigma_1\mid\mathcal G_0]|^2\leqslant\mathbb{E}|\sigma_1|^2\leqslant N.
\end{equation*}
Dividing by $2\varepsilon$ proves the estimate.

Let $\sigma_s^1,\sigma_s^2$ be conditionally independent copies of $\sigma_s$ conditioned on $\mathcal G_s$. 
Let $\mathsf R_s:=N^{-1}\sigma_s^1(\sigma_s^2)^\intercal$. 
Then,

 \partial_q\overline F_N^\nu(t,q,s)=\mathbb{E}\mathsf R_s
 \quad\text{and}\quad

We first prove the time identity. Differentiating~\eqref{eq:Parisi-PDE} in $t$, applying the diffusion representation, and using~\eqref{eq:orbit-realization} and~\eqref{e.F_N=-1/NEPhi} give
\begin{equation*}
 \partial_t\overline F_N^\nu(t,q)
 =\frac1N\mathbb{E}\left[|\mathbf h_N(\sigma_1)|^2-\frac{\mathbf{g}\cdot\mathbf h_N(\sigma_1)}{\sqrt{2t}}\right].
\end{equation*}
Since $\mathbf{X}_0$ is independent of $\mathbf{g}$, Gaussian integration by parts and~\eqref{eq:scalar-Hessian-identity} in the proof of Lemma~\ref{lem:Parisi-Hessian} at $s=0$, with $v=\sqrt{2t}\,\mathbf h_{N,\ell}$, yield
\begin{align*}
 \frac1{\sqrt{2t}}\mathbb{E}[\mathbf{g}\cdot\mathbf h_N(\sigma_1)]
 &=\frac1{2t}\sum_\ell\mathbb{E}\partial_{\mathbf{g}_\ell}^2\Phi(0,\mathbf{X}_0;\mathbf{g})\\
 &=\int_0^1\mathbb{E}\left[|\mathbf h_N(\sigma_1)|^2
 -\left|\mathbb{E}[\mathbf h_N(\sigma_1)\mid\mathcal G_s]\right|^2\right]\,\mathrm d s.
\end{align*}
The self-overlap terms cancel, and conditional independence and~\eqref{eq:orbit-realization} give
\begin{equation*}
 \partial_t\overline F_N^\nu(t,q)
 =\frac1N\int_0^1\mathbb{E}\left|\mathbb{E}[\mathbf h_N(\sigma_1)\mid\mathcal G_s]\right|^2\,\mathrm d s
 =\int_0^1\mathbb{E}\xi(\mathsf R_s)\,\mathrm d s.
\end{equation*}

                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                
For the path derivative, we first take $q$ smooth; the approximation argument of~\cite[Lemma 5.5, Steps 1--5]{chen2024simultaneous}, together with Lemma~\ref{lem:regularity}, gives the general case. Fix $\kappa\in C_c^\infty((0,1);S^D)$. Since $ 
\overset{\mbox{\large\bfseries .}}{q}\succeq\varepsilon I_D$, the paths $q+h\kappa$ are admissible for small $|h|$. Summing the diagonal $D$-by-$D$ blocks in Lemma~\ref{lem:Parisi-Hessian}(3) and using conditional independence give
\begin{equation*}
 \sum_{i=1}^N\mathbb{E}\nabla^2_{\mathbf{x}_{\cdot,i}}\Phi(s,\mathbf{X}_s;\mathbf{g})+Ns\mathbb{E}\mathsf R_s
 =\mathbb{E}[\sigma_1\sigma_1^\intercal]-N\int_s^1\mathbb{E}\mathsf R_r\,\mathrm d r.
\end{equation*}
Differentiating~\eqref{eq:Parisi-PDE} in $h$, applying It\^o's formula along $\mathbf{X}$, and using Lemma~\ref{lem:Parisi-Hessian}(2) therefore yield
\begin{equation}\label{eq:path-variation-Phi}
 \begin{aligned}
 \left.\partial_h\overline F_N^\nu(t,q+h\kappa)\right|_{h=0}
 &=-\frac1N\int_0^1 
\overset{\mbox{\large\bfseries .}}{\kappa}(s)\cdot
 \left(\mathbb{E}[\sigma_1\sigma_1^\intercal]-N\int_s^1\mathbb{E}\mathsf R_r\,\mathrm d r\right)\,\mathrm d s\\
 &=\int_0^1\kappa(s)\cdot\mathbb{E}\mathsf R_s\,\mathrm d s.
 \end{aligned}
\end{equation}
The last equality follows by integration by parts, since $\kappa(0)=\kappa(1)=0$. Density of these directions in $L^2([0,1];S^D)$ and Lemma~\ref{lem:regularity} identify the path derivative almost everywhere.

<!-- lemma-9: body_md -->
\label{lem:overlap-concentration}
There is a constant $C_D<\infty$, depending only on $D$, such that
\begin{equation}\label{eq:total-concentration}
 \int_0^1\mathbb{E}|\mathsf R_s-\mathbb{E}\mathsf R_s|^2\,\mathrm d s
 \leqslant C_D(N\varepsilon)^{-1/3}.
\end{equation}
<!-- end lemma-9 -->
<!-- proof-8: body_md -->
Since $|\mathsf R_s|\leqslant1$, the left-hand side of~\eqref{eq:total-concentration} is at most $1$. It therefore suffices to consider $N\varepsilon>1$. Fix $\delta\in(0,1)$.

\smallskip
\noindent\emph{Step 1: Thermal fluctuations.}
For a deterministic $v\in\mathbb{R}^{DN}$, Lemma~\ref{lem:Parisi-Hessian}(3) gives
\begin{equation}\label{eq:linear-thermal-variance}
 \operatorname{Var}(v\cdot\operatorname{vec}(\sigma_1)\mid\mathcal G_s)
 =v^\intercal \mathrm{Cov}(\operatorname{vec}(\sigma_1)\mid\mathcal G_s)v\leqslant\frac1s v^\intercal \nabla_{\mathbf{x}}^2\Phi(s,\mathbf{X}_s;\mathbf{g})v
 \leqslant\frac{|v|^2}{s}\bigl|\nabla_{\mathbf{x}}^2\Phi(s,\mathbf{X}_s;\mathbf{g})\bigr|.
\end{equation}
With one replica fixed at $\tau$, each entry of $\mathsf R_s$ is a linear form in the other replica. The sum of the squared coefficient norms is
\begin{equation}\label{eq:coefficient-sum}
 \frac1{N^2}\sum_{\alpha,\beta=1}^D\sum_{i=1}^N\tau_{\alpha i}^2
 =\frac D{N^2}|\tau|^2\leqslant\frac DN.
\end{equation}
Conditional variance tensorization over the two replicas, followed by summation over the entries, therefore gives
\begin{equation}\label{eq:thermal-pointwise}
 \mathbb{E}[|\mathsf R_s-\mathbb{E}[\mathsf R_s\mid\mathcal G_s]|^2\mid\mathcal G_s]
 \leqslant\frac{2D}{Ns}\bigl|\nabla_{\mathbf{x}}^2\Phi(s,\mathbf{X}_s;\mathbf{g})\bigr|.
\end{equation}
Taking expectations, integrating, and applying Cauchy--Schwarz and Lemma~\ref{lem:Parisi-Hessian}(4), we obtain
\begin{align}\label{eq:thermal-concentration}
 \int_\delta^1\mathbb{E}|\mathsf R_s-\mathbb{E}[\mathsf R_s\mid\mathcal G_s]|^2\,\mathrm d s
 &\leqslant\frac{2D}{N}
 \left(\int_\delta^1s^{-2}\,\mathrm d s\right)^{1/2}
 \left(\int_\delta^1\mathbb{E}\bigl|\nabla_{\mathbf{x}}^2\Phi(s,\mathbf{X}_s;\mathbf{g})\bigr|^2\,\mathrm d s\right)^{1/2}
 \leqslant\frac{C_D}{\sqrt{N\varepsilon\delta}}.
\end{align}

\smallskip
\noindent\emph{Step 2: Concentration of the conditional mean.}

Fix $a\in S^D$ with $|a|=1$ and let $T_a:=I_N\otimes a$, a self-adjoint contraction in column-stacked coordinates.

For $s\in(\delta,1]$, define
\begin{equation}\label{eq:U-definition}
 U_s:=\frac1{N\delta}\int_{s-\delta}^s
 a\cdot\left(\sigma_1 \mathbb{E}[\sigma_1\mid\mathcal G_r]^\intercal-\frac12\mathbb{E}[\sigma_1\mid\mathcal G_r]\mathbb{E}[\sigma_1\mid\mathcal G_r]^\intercal\right)\,\mathrm d r.
\end{equation}
Since $\mathbb{E}[\sigma_1\mid\mathcal G_r]$ is $\mathcal G_s$-measurable for $r\leqslant s$, the tower property and symmetry of $a$ give
\begin{equation}\label{eq:A-from-U}
 a\cdot \mathbb{E}[\mathsf R_s\mid\mathcal G_s]-2\mathbb{E}[U_s\mid\mathcal G_s]
 =\frac1{N\delta}\int_{s-\delta}^s
 a\cdot\bigl((\mathbb{E}[\sigma_1\mid\mathcal G_s]-\mathbb{E}[\sigma_1\mid\mathcal G_r])(\mathbb{E}[\sigma_1\mid\mathcal G_s]-\mathbb{E}[\sigma_1\mid\mathcal G_r])^\intercal\bigr)\,\mathrm d r.
\end{equation}
The martingale identity from Lemma~\ref{lem:Parisi-Hessian}(2), 
\[\mathbb{E}|\mathbb{E}[\sigma_1\mid\mathcal G_s]-\mathbb{E}[\sigma_1\mid\mathcal G_r]|^2=\mathbb{E}|\mathbb{E}[\sigma_1\mid\mathcal G_s]|^2-\mathbb{E}|\mathbb{E}[\sigma_1\mid\mathcal G_r]|^2,\]  
the bounds $0\leqslant\mathbb{E}|\mathbb{E}[\sigma_1\mid\mathcal G_r]|^2\leqslant N$ and $|\mathbb{E}[\sigma_1\mid\mathcal G_s]-\mathbb{E}[\sigma_1\mid\mathcal G_r]|^2\leqslant4N$, and time translation imply
\begin{align}\label{eq:integrated-martingale-increments}
 \int_{\delta}^1\mathbb{E}|a\cdot \mathbb{E}[\mathsf R_s\mid\mathcal G_s]-2\mathbb{E}[U_s\mid\mathcal G_s]|^2\,\mathrm d s
 &\leqslant\frac4{N\delta}\int_0^\delta\int_{\delta}^1
 \bigl(\mathbb{E}|\mathbb{E}[\sigma_1\mid\mathcal G_s]|^2-\mathbb{E}|\mathbb{E}[\sigma_1\mid\mathcal G_{s-v}]|^2\bigr)\,\mathrm d s\,\mathrm d v\nonumber\\
 &\leqslant\frac4\delta\int_0^\delta v\,\mathrm d v=2\delta.
\end{align}

Equation~\eqref{eq:orbit-equivariance} and invariance of $\lambda_N(\nu)$ give the terminal equivariance of $\Phi$. Uniqueness for the PDE then gives
\begin{equation}\label{eq:Phi-equivariance}
 \Phi(r,\mathbf{x}^\pi;Q_\pi \mathbf{g})=\Phi(r,\mathbf{x};\mathbf{g}),\qquad \pi\in\mathfrak S_N.
\end{equation}

Consequently the joint law is invariant under
\begin{equation}\label{e.(g,z,B,sigma)mapsto}
 (\mathbf{g},\mathbf{z}_0,B,\sigma_1)\longmapsto(Q_\pi \mathbf{g},\mathbf{z}_0^\pi,B^\pi,\sigma_1^\pi).
\end{equation}
The Gaussian inputs keep their laws, the SDE solution is column-permuted, and the terminal Gibbs law is carried to its corresponding permutation. Thus $\mathbb{E}[\sigma_1\mid\mathcal G_r]$ is also column-permuted and $U_s$ is unchanged. Since $\mathfrak S_N$ acts transitively on $\Sigma_N(\nu)$, $\mathbb{E}[U_s\mid\sigma_1]$ is constant, so
\begin{equation}\label{eq:conditional-centering}
 \operatorname{Var}(U_s)=\mathbb{E}\operatorname{Var}(U_s\mid\sigma_1).
\end{equation}
                                                                                                                                                                                            
Centering~\eqref{eq:A-from-U}, applying Jensen's inequality, and using $|\mathbb{E}[\mathsf R_s\mid\mathcal G_s]|\leqslant1$ on $[0,\delta]$ yield
\begin{equation}\label{eq:mean-average-reduction}
 \int_0^1\operatorname{Var}(a\cdot \mathbb{E}[\mathsf R_s\mid\mathcal G_s])\,\mathrm d s
 \leqslant C\delta+C\int_{\delta}^1\mathbb{E}\operatorname{Var}(U_s\mid\sigma_1)\,\mathrm d s.
\end{equation}

We next bound the conditional variance in~\eqref{eq:mean-average-reduction}. 
Let $\Gamma$ be the product law of the independent inputs $(\mathbf{g},\mathbf{z}_0,B)$, and put
\begin{equation*}
 \mathbf{Y}_r:=\sqrt{2q(0)}\mathbf{z}_0+\int_0^r\sqrt{2 
\overset{\mbox{\large\bfseries .}}{q}(u)}\,\mathrm dB_u.
\end{equation*}
It\^o's formula for $r\Phi(r,\mathbf{Y}_r;\mathbf{g})$ identifies
\begin{equation}\label{eq:Girsanov-density}
 \exp\left(\Phi(1,\mathbf{Y}_1;\mathbf{g})-\int_0^1\Phi(r,\mathbf{Y}_r;\mathbf{g})\,\mathrm d r\right)
\end{equation}
as the stochastic exponential with integrand $r\sqrt{2 
\overset{\mbox{\large\bfseries .}}{q}(r)}\nabla_{\mathbf{x}}\Phi(r,\mathbf{Y}_r;\mathbf{g})$. The bounded gradient satisfies Novikov's condition~\cite[Chapter 3, Corollary 5.13]{karatzas1991brownian}, and Girsanov's theorem identifies $(\mathbf{g},\mathbf{z}_0,\mathbf{Y})$ under this change of measure with $(\mathbf{g},\mathbf{z}_0,\mathbf{X})$ in~\eqref{eq:diffusion}. Multiplying by the terminal Gibbs mass at $\sigma\in\Sigma_N(\nu)$ cancels $\exp(\Phi(1,\mathbf{Y}_1;\mathbf{g}))$. Hence the conditional law $\mathbb{P}_\sigma$ of the reference inputs given $\sigma_1=\sigma$ has density
\begin{equation}\label{eq:conditional-density}
 \frac{\mathrm d\mathbb{P}_\sigma}{\mathrm d\Gamma}
 \propto\exp\left(
 \sqrt{2t}\,\mathbf h_N(\sigma)\cdot \mathbf{g}+\mathbf{Y}_1\cdot\sigma
 -\int_0^1\Phi(r,\mathbf{Y}_r;\mathbf{g})\,\mathrm d r\right).
\end{equation}
The normalizing constant is positive and finite; terms depending only on $\sigma$ are absorbed into it. In these reference coordinates, evaluate $U_s$ by representing $\mathbb{E}[\sigma_1\mid\mathcal G_r]$ as $\nabla_{\mathbf{x}}\Phi(r,\mathbf{Y}_r;\mathbf{g})$ using Lemma~\ref{lem:Parisi-Hessian}(2); the change of measure identifies this with the original conditional mean.

Choose an orthonormal basis of $L^2([0,1];\mathbb{R}^{D\times N})$ and write $(\eta_k)_{k\in\mathbb{N}}$ for the corresponding independent standard Gaussian coordinates of the Brownian noise. Put $\mathcal F_n:=\sigma(\eta_{n+1},\eta_{n+2},\ldots)$.
Let $\zeta$ collect $\mathbf{g},\mathbf{z}_0,\eta_1,\ldots,\eta_n$, and condition on $\mathcal F_n$. Under $\Gamma$, $\zeta$ is standard Gaussian and $(\mathbf{Y}_r,\mathbf{g})$ is affine in $\zeta$. Under $\mathbb{P}_\sigma$, its conditional Lebesgue density is proportional to $\exp(-V_{\sigma,n})$, where
\begin{equation*}
 V_{\sigma,n}(\zeta)=\frac12|\zeta|^2-\sqrt{2t}\,\mathbf h_N(\sigma)\cdot \mathbf{g}-\mathbf{Y}_1\cdot\sigma
 +\int_0^1\Phi(r,\mathbf{Y}_r;\mathbf{g})\,\mathrm d r.
\end{equation*}
The middle terms are affine, and Lemma~\ref{lem:Parisi-Hessian}(1) gives
\begin{equation}\label{eq:conditional-potential-Hessian}
 \nabla_\zeta^2V_{\sigma,n}
 =I+\int_0^1\nabla_\zeta^2[\Phi(r,\mathbf{Y}_r;\mathbf{g})]\,\mathrm d r\succeq I.
\end{equation}

For fixed $\sigma$, the derivative of $a\cdot(\sigma\mathbb{E}[\sigma_1\mid\mathcal G_r]^\intercal-\frac12\mathbb{E}[\sigma_1\mid\mathcal G_r]\mathbb{E}[\sigma_1\mid\mathcal G_r]^\intercal)$ with respect to its conditional-mean argument is $a(\sigma-\mathbb{E}[\sigma_1\mid\mathcal G_r])$. Cauchy--Schwarz for the joint Hessian of $\Phi$ and then in time therefore gives, for every $\zeta$-direction $w$,
\begin{equation}\label{eq:gradient-bar-U}
 \begin{gathered}
 |w\cdot\nabla_\zeta U_s|^2\leqslant b_s\,w^\intercal (\nabla_\zeta^2V_{\sigma,n})w,\\
 b_s:=\frac1{N^2\delta^2}\int_{s-\delta}^s
 \operatorname{vec}(a(\sigma-\mathbb{E}[\sigma_1\mid\mathcal G_r]))^\intercal \nabla_{\mathbf{x}}^2\Phi(r,\mathbf{Y}_r;\mathbf{g})\operatorname{vec}(a(\sigma-\mathbb{E}[\sigma_1\mid\mathcal G_r]))\,\mathrm d r.
 \end{gathered}
\end{equation}
Indeed, the first quadratic factor is $w^\intercal\nabla_\zeta^2[\Phi(r,\mathbf{Y}_r;\mathbf{g})]w$, whose time integral is bounded by $w^\intercal (\nabla_\zeta^2V_{\sigma,n})w$. Taking $w=(\nabla_\zeta^2V_{\sigma,n})^{-1}\nabla_\zeta U_s$ and applying the Brascamp--Lieb variance inequality~\cite[Section 4]{brascampLieb1976} give
\begin{equation}\label{eq:BL-finite}
 \operatorname{Var}_{\mathbb{P}_\sigma}(U_s\mid\mathcal F_n)
 \leqslant\mathbb{E}_{\mathbb{P}_\sigma}[\nabla_\zeta U_s^\intercal (\nabla_\zeta^2V_{\sigma,n})^{-1}\nabla_\zeta U_s\mid\mathcal F_n]
 \leqslant\mathbb{E}_{\mathbb{P}_\sigma}[b_s\mid\mathcal F_n].
\end{equation}
Taking $\mathbb{P}_\sigma$-expectations first and using total variance yields
\begin{equation*}
 \operatorname{Var}_{\mathbb{P}_\sigma}(U_s)
 -\operatorname{Var}_{\mathbb{P}_\sigma}\bigl(\mathbb{E}_{\mathbb{P}_\sigma}[U_s\mid\mathcal F_n]\bigr)
 \leqslant\mathbb{E}_{\mathbb{P}_\sigma}b_s.
\end{equation*}
The tail sigma-algebra is trivial under $\Gamma$ and hence under the equivalent measure $\mathbb{P}_\sigma$. 
Reverse martingale convergence
(see Theorem 4.7.3~in~\cite{durrett2019probability}) gives convergence in $L^1(\mathbb{P}_\sigma)$. Since $U_s$ and its conditional expectations are uniformly bounded, squared differences are bounded by a constant times absolute differences, so convergence also holds in $L^2(\mathbb{P}_\sigma)$ and the second variance vanishes. The right-hand side is independent of $n$ and integrable, since the spatial Hessian is bounded. Returning to the original diffusion $\mathbf{X}$, with $\mathbf{Y}$ replaced by $\mathbf{X}$ in $b_s$, gives
\begin{equation}\label{eq:BL-conditional}
 \operatorname{Var}(U_s\mid\sigma_1=\sigma)\leqslant\mathbb{E}[b_s\mid\sigma_1=\sigma].
\end{equation}
Conditioning the quadratic integrand on $\mathcal G_r$ gives
\begin{equation}\label{eq:quadratic-covariance-bound}
 \begin{aligned}
 &\mathbb{E}[\operatorname{vec}(a(\sigma_1-\mathbb{E}[\sigma_1\mid\mathcal G_r]))^\intercal \nabla_{\mathbf{x}}^2\Phi(r,\mathbf{X}_r;\mathbf{g})\operatorname{vec}(a(\sigma_1-\mathbb{E}[\sigma_1\mid\mathcal G_r]))\mid\mathcal G_r]\\
 &=\operatorname{tr}(\nabla_{\mathbf{x}}^2\Phi(r,\mathbf{X}_r;\mathbf{g})T_a\mathrm{Cov}(\operatorname{vec}(\sigma_1)\mid\mathcal G_r)T_a)\leqslant\frac1r\bigl|\nabla_{\mathbf{x}}^2\Phi(r,\mathbf{X}_r;\mathbf{g})\bigr|^2.
 \end{aligned}
\end{equation}
Here Lemma~\ref{lem:Parisi-Hessian}(3) gives
\begin{equation*}
 \mathrm{Cov}(\operatorname{vec}(\sigma_1)\mid\mathcal G_r)
 \preceq r^{-1}\nabla_{\mathbf{x}}^2\Phi(r,\mathbf{X}_r;\mathbf{g}).
\end{equation*}
Since $T_a$ is a contraction,
\begin{align*}
 \operatorname{tr}(\nabla_{\mathbf{x}}^2\Phi(r,\mathbf{X}_r;\mathbf{g})T_a\nabla_{\mathbf{x}}^2\Phi(r,\mathbf{X}_r;\mathbf{g})T_a)
 &\leqslant|\nabla_{\mathbf{x}}^2\Phi(r,\mathbf{X}_r;\mathbf{g})|\,|T_a\nabla_{\mathbf{x}}^2\Phi(r,\mathbf{X}_r;\mathbf{g})T_a|\\
 &\leqslant\bigl|\nabla_{\mathbf{x}}^2\Phi(r,\mathbf{X}_r;\mathbf{g})\bigr|^2.
\end{align*}
Averaging~\eqref{eq:BL-conditional} over $\sigma$, applying Fubini and using Lemma~\ref{lem:Parisi-Hessian}(4) yield
\begin{align}\label{eq:U-variance}
 \int_{\delta}^1\mathbb{E}\operatorname{Var}(U_s\mid\sigma_1)\,\mathrm d s
 &\leqslant\frac1{N^2\delta^2}\int_{\delta}^1\int_{s-\delta}^s\frac1r\mathbb{E}\bigl|\nabla_{\mathbf{x}}^2\Phi(r,\mathbf{X}_r;\mathbf{g})\bigr|^2\,\mathrm d r\,\mathrm d s
 \nonumber \\
 &\leqslant\frac1{N^2\delta^2}\int_0^1\frac{\min\{r,\delta\}}r\mathbb{E}\bigl|\nabla_{\mathbf{x}}^2\Phi(r,\mathbf{X}_r;\mathbf{g})\bigr|^2\,\mathrm d r
 \nonumber \\
 &\leqslant\frac1{2N\varepsilon\delta^2}.
\end{align}
Here we used $|[\delta,1]\cap[r,r+\delta]|\leqslant\min\{r,\delta\}$ for $r>0$. Substituting into~\eqref{eq:mean-average-reduction} and summing over an orthonormal basis of $S^D$ give
\begin{equation}\label{eq:mean-concentration}
 \int_0^1\mathbb{E}|\mathbb{E}[\mathsf R_s\mid\mathcal G_s]-\mathbb{E}\mathsf R_s|^2\,\mathrm d s
 \leqslant C_D\left(\delta+\frac1{N\varepsilon\delta^2}\right).
\end{equation}

\smallskip
\noindent\emph{Step 3: Combination of the estimates.}
Conditional variance decomposition gives
\begin{equation}\label{eq:overlap-orthogonal}
 \mathbb{E}|\mathsf R_s-\mathbb{E}\mathsf R_s|^2=\mathbb{E}|\mathsf R_s-\mathbb{E}[\mathsf R_s\mid\mathcal G_s]|^2+\mathbb{E}|\mathbb{E}[\mathsf R_s\mid\mathcal G_s]-\mathbb{E}\mathsf R_s|^2.
\end{equation}
Since $|\mathsf R_s|,|\mathbb{E}[\mathsf R_s\mid\mathcal G_s]|\leqslant1$, 
\eqref{eq:thermal-concentration} in Step 1 and \eqref{eq:mean-concentration} 
in Step 2
imply
\begin{equation*}
 \int_0^1\mathbb{E}|\mathsf R_s-\mathbb{E}\mathsf R_s|^2\,\mathrm d s
 \leqslant C_D\left(\delta+\frac1{N\varepsilon\delta^2}+\frac1{\sqrt{N\varepsilon\delta}}\right)
 \leqslant \frac{3 C_D}{2}\left(\delta+\frac1{N\varepsilon\delta^2}\right).
\end{equation*}
Taking $\delta=(N\varepsilon)^{-1/3}$ proves~\eqref{eq:total-concentration}.

Suppose that $\mu$ has finite support. For $T,Q\in\mathbb{R}_+$, put $C:=TM_0+(1+\sqrt D)Q$. For every $N\in\mathbb{N}$, $t\in[0,T]$, $q\in\mathcal{Q}_\infty$ with $|q|_{L^\infty}\leqslant Q$, and $s_0\in(0,1/2]$, we have

Restriction decreases the partition function, which proves the first inequality. By smooth approximation and Lemma~\ref{lem:regularity}, it suffices to prove the second inequality for smooth $q$.

In this proof, $\Phi$ and $\Phi_\nu$ denote the solutions of~\eqref{eq:Parisi-PDE} with reference measures $\mu^{\otimes N}$ and $\1_{\Sigma_N(\nu)}\mu^{\otimes N}$, respectively, using the Gaussian process fixed in~\eqref{eq:orbit-realization}. The restrictions retain their original masses as in~\eqref{eq:restricted-reference-normalization}. Since the sets $\Sigma_N(\nu)$ partition $(\operatorname{supp}\mu)^N$,
\begin{equation}\label{eq:type-terminal-softmax}
 \Phi(1,\mathbf{x};\mathbf{g})=\log\sum_{\nu\in\Delta_{m,N}}\exp\bigl(\Phi_\nu(1,\mathbf{x};\mathbf{g})\bigr).
\end{equation}
The representation~\eqref{e.F_N=-1/NEPhi} gives
\begin{align}\label{eq:type-Parisi-representation}
 \overline F_{N,\mu^{\otimes N}}(t,q)
 &=-\frac1N\mathbb{E}\Phi(0,\sqrt{2q(0)}\mathbf{z}_0;\mathbf{g}),\nonumber\\
 \overline F_{N,\mu^{\otimes N},\Sigma_N(\nu)}(t,q)
 &=-\frac1N\mathbb{E}\Phi_\nu(0,\sqrt{2q(0)}\mathbf{z}_0;\mathbf{g}),
\end{align}
where $\mathbf{z}_0$ is standard Gaussian in $\mathbb{R}^{D\times N}$ and independent of $\mathbf{g}$.

\smallskip
\noindent\emph{Comparison on $[s_0,1]$.}
For $s\in[s_0,1]$, let
\begin{equation*}
 \widehat\Phi(s,\mathbf{x};\mathbf{g}):=
 \frac1s\log\sum_{\nu\in\Delta_{m,N}}e^{s\Phi_\nu(s,\mathbf{x};\mathbf{g})},
 \qquad
 p_\nu:=\frac{e^{s\Phi_\nu}}{\sum_{\nu'}e^{s\Phi_{\nu'}}}.
\end{equation*}
We claim that for all $(s,\mathbf{x})\in[s_0,1]\times \mathbb{R}^{D\times N}$
\begin{equation}\label{eq:type-softmax-comparison}
 \Phi(s,\mathbf{x};\mathbf{g})\leqslant\widehat\Phi(s,\mathbf{x};\mathbf{g}).
\end{equation}
This follows from the following standard maximum principle argument.

We have $\nabla_{\mathbf{x}}\widehat\Phi=\sum_\nu p_\nu\nabla_{\mathbf{x}}\Phi_\nu$, and differentiation gives
\begin{align*}
 \partial_s\widehat\Phi
 &=\sum_\nu p_\nu\partial_s\Phi_\nu+\frac1{s^2}\sum_\nu p_\nu\log p_\nu,\\
 \sum_{i=1}^N 
\overset{\mbox{\large\bfseries .}}{q}(s)\cdot\nabla^2_{\mathbf{x}_{\cdot,i}}\widehat\Phi
 &=\sum_\nu p_\nu\sum_{i=1}^N 
\overset{\mbox{\large\bfseries .}}{q}(s)\cdot\nabla^2_{\mathbf{x}_{\cdot,i}}\Phi_\nu
 +s 
\overset{\mbox{\large\bfseries .}}{q}(s)\cdot\Bigl(\sum_\nu p_\nu\nabla_{\mathbf{x}}\Phi_\nu(\nabla_{\mathbf{x}}\Phi_\nu)^\intercal
 -\nabla_{\mathbf{x}}\widehat\Phi(\nabla_{\mathbf{x}}\widehat\Phi)^\intercal\Bigr).
\end{align*}
Substituting the equations for $\Phi_\nu$, the quadratic terms cancel:
\begin{equation}\label{eq:type-softmax-PDE}
 \partial_s\widehat\Phi
 +\sum_{i=1}^N 
\overset{\mbox{\large\bfseries .}}{q}(s)\cdot\nabla^2_{\mathbf{x}_{\cdot,i}}\widehat\Phi
 +s 
\overset{\mbox{\large\bfseries .}}{q}(s)\cdot\nabla_{\mathbf{x}}\widehat\Phi(\nabla_{\mathbf{x}}\widehat\Phi)^\intercal
 =\frac1{s^2}\sum_\nu p_\nu\log p_\nu\leqslant0.
\end{equation}
Let $\widetilde\Phi:=\widehat\Phi-\Phi$. By~\eqref{eq:type-terminal-softmax}, 
the terminal condition equals $\widetilde\Phi(1,\cdot\,;\mathbf{g})=0$. Subtracting the equation for $\Phi$ from~\eqref{eq:type-softmax-PDE} gives
\begin{equation*}
 \partial_s\widetilde\Phi
 +\sum_{i=1}^N 
\overset{\mbox{\large\bfseries .}}{q}(s)\cdot\nabla^2_{\mathbf{x}_{\cdot,i}}\widetilde\Phi
 +\sum_{i=1}^N b_i\cdot\nabla_{\mathbf{x}_{\cdot,i}}\widetilde\Phi\leqslant0,
 \qquad
 b_i:=s 
\overset{\mbox{\large\bfseries .}}{q}(s)\bigl(\nabla_{\mathbf{x}_{\cdot,i}}\widehat\Phi+\nabla_{\mathbf{x}_{\cdot,i}}\Phi\bigr).
\end{equation*}
After the change of variables $\tau=1-s$, the parabolic maximum principle, together with the zero terminal value of $\widetilde\Phi$, gives $\widetilde\Phi\geqslant0$ on $[s_0,1]$.
This proves \eqref{eq:type-softmax-comparison}.

\smallskip
\noindent\emph{Control of $[0,s_0]$.}
Let $B$ be a standard Brownian motion in $\mathbb{R}^{D\times N}$ independent of $(\mathbf{g},\mathbf{z}_0)$, and let
\begin{equation*}
 \mathbf{Y}_s:=\sqrt{2q(0)}\mathbf{z}_0+\int_0^s\sqrt{2 
\overset{\mbox{\large\bfseries .}}{q}(r)}\,\mathrm dB_r.
\end{equation*}
For $\Psi\in\{\Phi\}\cup\{\Phi_\nu:\nu\in\Delta_{m,N}\}$, It\^o's formula, the Parisi PDE, and the column bound in~\eqref{eq:diffusion-gradient-bounds} give
\begin{align}\label{eq:type-short-time}
 0&\leqslant\mathbb{E}\Psi(0,\mathbf{Y}_0;\mathbf{g})-\mathbb{E}\Psi(s_0,\mathbf{Y}_{s_0};\mathbf{g})\nonumber\\
 &=\mathbb{E}\int_0^{s_0} r 
\overset{\mbox{\large\bfseries .}}{q}(r)\cdot\nabla_{\mathbf{x}}\Psi(r,\mathbf{Y}_r;\mathbf{g})\bigl(\nabla_{\mathbf{x}}\Psi(r,\mathbf{Y}_r;\mathbf{g})\bigr)^\intercal\,\mathrm dr
 \leqslant Ns_0\operatorname{tr}q(1)\leqslant Ns_0\sqrt D\,Q.
\end{align}
Moreover, $\mathbf{Y}_{s_0}$ has the same law as $\sqrt{2q(s_0)}\mathbf{z}$, with $\mathbf{z}$ standard Gaussian and independent of $\mathbf{g}$.

\smallskip
\noindent\emph{Gaussian concentration.}
By~\eqref{eq:diffusion-gradient-bounds} and the chain rule,
\begin{equation*}
 \left|\nabla_{(\mathbf{g},\mathbf{z})}\bigl[\Phi_\nu(s_0,\sqrt{2q(s_0)}\mathbf{z};\mathbf{g})\bigr]\right|^2
 \leqslant2tNM_0+2N\|q(s_0)\|_{\mathrm{op}}
 \leqslant2N(TM_0+Q).
\end{equation*}
Gaussian concentration therefore gives
\begin{align*}
 \log\mathbb{E} e^{s_0\Phi_\nu(s_0,\sqrt{2q(s_0)}\mathbf{z};\mathbf{g})}
 \leqslant s_0\mathbb{E}\Phi_\nu(s_0,\sqrt{2q(s_0)}\mathbf{z};\mathbf{g})
 +Ns_0^2(TM_0+Q).
\end{align*}
By Jensen's inequality,
\begin{align}\label{eq:type-random-softmax}
 \mathbb{E}\widehat\Phi(s_0,\mathbf{Y}_{s_0};\mathbf{g})
 &\leqslant\frac1{s_0}\log\sum_{\nu\in\Delta_{m,N}}\mathbb{E} e^{s_0\Phi_\nu(s_0,\sqrt{2q(s_0)}\mathbf{z};\mathbf{g})}\nonumber\\
 &\leqslant\max_{\nu\in\Delta_{m,N}}\mathbb{E}\Phi_\nu(s_0,\sqrt{2q(s_0)}\mathbf{z};\mathbf{g})
 +\frac{\log|\Delta_{m,N}|}{s_0}+Ns_0(TM_0+Q).
\end{align}
Combining~\eqref{eq:type-softmax-comparison}--\eqref{eq:type-random-softmax}, and using $\mathbb{E}\Phi_\nu(s_0,\sqrt{2q(s_0)}\mathbf{z};\mathbf{g})\leqslant\mathbb{E}\Phi_\nu(0,\mathbf{Y}_0;\mathbf{g})$ from~\eqref{eq:type-short-time}, we obtain
\begin{equation}\label{eq:type-Parisi-max}
 \mathbb{E}\Phi(0,\mathbf{Y}_0;\mathbf{g})
 \leqslant\max_{\nu\in\Delta_{m,N}}\mathbb{E}\Phi_\nu(0,\mathbf{Y}_0;\mathbf{g})
 +\frac{\log|\Delta_{m,N}|}{s_0}
 +Ns_0\bigl(TM_0+(1+\sqrt D)Q\bigr).
\end{equation}
Together with~\eqref{eq:type-Parisi-representation} and $|\Delta_{m,N}|\leqslant(N+1)^m$, this proves the second inequality in~\eqref{eq:type-minimum}.

We first fix $q\in\mathcal{Q}_\infty$.

\smallskip
\noindent\emph{Step 1: Compactness on $\Delta_{m,N}$ and the dual identity.}
Define $G_{N,q}:\Delta_{m,N}\to\mathbb{R}$ by
\begin{equation*}
 G_{N,q}(\nu):=\overline F_{N,\mu^{\otimes N},\Sigma_N(\nu)}(0,q).
\end{equation*}
Combining~\eqref{eq:normalization}, Lemma~\ref{lem:type-continuity}, and~\eqref{eq:stirling}, we obtain, for $\nu,\nu'\in\Delta_{m,N}$,
\begin{align}\label{eq:finite-equicontinuity}
 |G_{N,q}(\nu)-G_{N,q}(\nu')|
 &\leqslant 2|q|_{L^\infty}W_1(\nu,\nu')
 +|\operatorname{Ent}(\nu\mid\mu)-\operatorname{Ent}(\nu'\mid\mu)|
 +\frac{m\log(N+1)}N.
\end{align}
The map $\nu\mapsto\operatorname{Ent}(\nu\mid\mu)=\sum_{j=1}^m\nu_j\log(\nu_j/\mu_j)$ is continuous on the compact set $\mathcal P(\operatorname{supp}\mu)$, with the convention $0\log0:=0$. Thus~\eqref{eq:finite-equicontinuity} gives a common asymptotic modulus of continuity. Moreover, Lemma~\ref{lem:regularity}, applied to the probability measure $\lambda_N(\nu)$ at $t=0$, gives $|\overline F_N^\nu(0,q)|\leqslant |q|_{L^1}$. Together with~\eqref{eq:normalization} and~\eqref{eq:stirling}, this shows that $(G_{N,q})_{N\in\mathbb{N}}$ is uniformly bounded on its varying domains.

The grids $\Delta_{m,N}$ become dense in $\mathcal P(\operatorname{supp}\mu)$ for $W_1$: rounding the masses of any $\nu\in\mathcal P(\operatorname{supp}\mu)$ to multiples of $N^{-1}$ gives $\nu_N\in\Delta_{m,N}$ with $\lim_{N\to\infty}W_1(\nu_N,\nu)=0$. Consequently, a finite-net and diagonal argument using~\eqref{eq:finite-equicontinuity} shows that every subsequence of $(G_{N,q})$ has a further subsequence, still indexed by $N$, and a continuous function $g_q:\mathcal P(\operatorname{supp}\mu)\to\mathbb{R}$ such that
\begin{equation}\label{eq:finite-subsequence}
 \lim_{N\to\infty}\sup_{\nu\in\Delta_{m,N}}|G_{N,q}(\nu)-g_q(\nu)|=0.
\end{equation}

We next identify $g_q$. Fix $h\in\mathbb{R}^m$ and let
\begin{equation*}
 Z_h:=\sum_{j=1}^m\mu_j e^{h_j}
 \qquad\text{and}\qquad
 \mu_h:=Z_h^{-1}e^h\mu.
\end{equation*}
The measure $\mu_h$ is a probability measure with support $\operatorname{supp}\mu$. For every $\nu\in\Delta_{m,N}$, by~\eqref{eq:type-set}, every $\sigma\in\Sigma_N(\nu)$ has exactly $N\nu_j$ coordinates equal to $a_j$. Hence the density of $\mu_h^{\otimes N}$ relative to $\mu^{\otimes N}$ is constant on $\Sigma_N(\nu)$ and equals $Z_h^{-N}e^{Nh\cdot\nu}$. Therefore,
\begin{equation}\label{eq:tilted-restricted}
 \overline F_{N,\mu_h^{\otimes N},\Sigma_N(\nu)}(0,q)
 =G_{N,q}(\nu)+\log Z_h-h\cdot\nu.
\end{equation}
Equation~\eqref{e.tensorize} also gives
\begin{equation}\label{e.F=psi+logZ}
 \overline F_{N,\mu_h^{\otimes N}}(0,q)=\psi_{\mu,h}(q)+\log Z_h.
\end{equation}
Applying Lemma~\ref{lem:type-minimum} to the reference measure $\mu_h$, with $t=0$ and $s_{0,N}:=\min\{\frac12,\sqrt{\log(N+1)/N}\}$, and using~\eqref{eq:tilted-restricted} and~\eqref{e.F=psi+logZ}, we obtain
\begin{equation*}
 \lim_{N\to\infty}
 \left|
 \min_{\nu\in\Delta_{m,N}}\{G_{N,q}(\nu)-h\cdot\nu\}
 -\psi_{\mu,h}(q)
 \right|=0.
\end{equation*}
Passing along the subsequence in~\eqref{eq:finite-subsequence}, using the density of $\Delta_{m,N}$ in $\mathcal P(\operatorname{supp}\mu)$ and the continuity of $g_q$, gives
\begin{equation}\label{eq:finite-duality}
 \psi_{\mu,h}(q)=\min_{\nu\in\mathcal P(\operatorname{supp}\mu)}\{g_q(\nu)-h\cdot\nu\},
 \qquad h\in\mathbb{R}^m.
\end{equation}

\smallskip
\noindent\emph{Step 2: Identification of $g_q$ by convex duality.}
For bounded $q$, differentiation with respect to $h$ gives
\begin{equation}\label{eq:psi-h-gradient}
 \partial_{h_j}\psi_{\mu,h}(q)
 =-\mathbb{E}\langle\1_{\{\sigma=a_j\}}\rangle,
 \qquad j\in\{1,\ldots,m\}.
\end{equation}
A second differentiation expresses the Hessian as the negative of an expected covariance matrix of the indicators $(\1_{\{\sigma=a_j\}})_{j=1}^m$. Since these indicators are bounded by one, the Hessian is uniformly bounded. Thus $h\mapsto\psi_{\mu,h}(q)$ is continuously differentiable.

Fix $h\in\mathbb{R}^m$, and let $\eta\in\mathcal P(\operatorname{supp}\mu)$ be a minimizer in~\eqref{eq:finite-duality}; such a minimizer exists because $\mathcal P(\operatorname{supp}\mu)$ is compact and $g_q$ is continuous. For every $v\in\mathbb{R}^m$ and $\varepsilon\in\mathbb{R}$, we have
\begin{equation*}
 \psi_{\mu,h+\varepsilon v}(q)
 \leqslant g_q(\eta)-(h+\varepsilon v)\cdot\eta
 =\psi_{\mu,h}(q)-\varepsilon v\cdot\eta.
\end{equation*}
Dividing by $\varepsilon$ for $\varepsilon>0$ and $\varepsilon<0$ and then sending $\varepsilon\to0$ gives
\begin{equation}\label{eq:minimizer-gradient}
 \eta=-\nabla_h\psi_{\mu,h}(q).
\end{equation}

We now fix $\nu$ in the relative interior of $\mathcal P(\operatorname{supp}\mu)$ and show that there is an $h$ for which $\nu$ is the minimizer in~\eqref{eq:finite-duality}. Equation~\eqref{eq:regularity} and the identity $\psi_{\mu,h}(0)=-\log\sum_{j=1}^m\mu_j e^{h_j}$ give
\begin{equation}\label{eq:psi-h-zero-comparison}
 \bigg|\psi_{\mu,h}(q)+\log\sum_{j=1}^m\mu_j e^{h_j}\bigg|
 \leqslant |q|_{L^1}.
\end{equation}
Define $J_\nu(h):=\psi_{\mu,h}(q)+h\cdot\nu$. Translation invariance, $J_\nu(h+c\1)=J_\nu(h)$, allows us to normalize $\max_jh_j=0$. Put $\mu_*:=\min_j\mu_j>0$ and $\nu_*:=\min_j\nu_j>0$. On this slice,~\eqref{eq:psi-h-zero-comparison} gives
\begin{equation*}
 J_\nu(h)\leqslant-\log\mu_*+|q|_{L^1}-\nu_*|h|_\infty.
\end{equation*}
Thus $J_\nu$ attains a maximum on the slice, and translation invariance makes the maximizer $h_*$ global on $\mathbb{R}^m$. This also applies when $m=1$, for which the slice is the single point $0$. Differentiability at the global maximum gives
\begin{equation*}
 \nu=-\nabla_h\psi_{\mu,h_*}(q).
\end{equation*}
This is precisely~\eqref{eq:minimizer-gradient} with $\eta=\nu$. Every minimizer in~\eqref{eq:finite-duality} at $h=h_*$ must therefore equal $\nu$. Consequently,
\begin{equation}\label{eq:g-legendre-interior}
 g_q(\nu)=\psi_{\mu,h_*}(q)+h_*\cdot\nu
 =\sup_{h\in\mathbb{R}^m}\{\psi_{\mu,h}(q)+h\cdot\nu\}
\end{equation}
for every $\nu$ in the relative interior of $\mathcal P(\operatorname{supp}\mu)$.

Equation~\eqref{eq:g-legendre-interior} shows that $g_q$ is convex on the relative interior. Since $g_q$ is continuous on $\mathcal P(\operatorname{supp}\mu)$, it is convex on all of $\mathcal P(\operatorname{supp}\mu)$. Extend $g_q$ to $\mathbb{R}^m$ by setting it equal to $+\infty$ outside $\mathcal P(\operatorname{supp}\mu)$. The resulting function is proper, lower semicontinuous, and convex. Equation~\eqref{eq:finite-duality} says that its convex conjugate is $h\mapsto-\psi_{\mu,h}(q)$. The finite-dimensional Fenchel--Moreau theorem therefore gives
\begin{equation}\label{eq:g-legendre}
 g_q(\nu)=\sup_{h\in\mathbb{R}^m}\{\psi_{\mu,h}(q)+h\cdot\nu\},
 \qquad \nu\in\mathcal P(\operatorname{supp}\mu).
\end{equation}
The right-hand side of~\eqref{eq:g-legendre} is independent of the subsequence chosen in~\eqref{eq:finite-subsequence}. Hence every subsequence has the same subsequential limit, and the whole sequence satisfies~\eqref{eq:finite-initial-convergence}. Equation~\eqref{eq:g-legendre} proves the first equality in~\eqref{eq:finite-initial-identification}.

\smallskip
\noindent\emph{Step 3: Identification with $\varphi^\nu$ and uniformity in $q$.}
If $\nu_j>0$ for every $j$, change the tilt by
\begin{equation*}
 \widehat h_j:=h_j+\log(\mu_j/\nu_j).
\end{equation*}
Then $e^h\mu=e^{\widehat h}\nu$ and $h\cdot\nu=\widehat h\cdot\nu+\operatorname{Ent}(\nu\mid\mu)$, so~\eqref{eq:g-legendre} and~\eqref{eq:initial-definition} give $g_q(\nu)=\operatorname{Ent}(\nu\mid\mu)+\varphi^\nu(q)$ directly.

For a boundary measure $\nu$, restrict first to $\operatorname{supp}\nu$. For any $h\in\mathbb{R}^m$, removing the mass outside $\operatorname{supp}\nu$ increases the free energy, and the same tilt change on $\operatorname{supp}\nu$ gives
\begin{equation*}
 \psi_{\mu,h}(q)+h\cdot\nu\leqslant\operatorname{Ent}(\nu\mid\mu)+\varphi^\nu(q).
\end{equation*}
Conversely, fix a tilt $k$ on $\operatorname{supp}\nu$ and let $h_j=k_j+\log(\nu_j/\mu_j)$ for $a_j\in\operatorname{supp}\nu$, and $h_j=-M$ for $a_j\notin\operatorname{supp}\nu$. As $M\to\infty$, the partition functions decrease to those with reference measure $e^k\nu$. Their logarithms lie between the integrable logarithms for $M=0$ and the restricted measure, so dominated convergence gives $\psi_{\mu,h}(q)\to\psi_{\nu,k}(q)$. Taking the supremum over $k$ proves
\begin{equation}\label{e.g_q=Ent+varphi}
 g_q(\nu)=\operatorname{Ent}(\nu\mid\mu)+\varphi^\nu(q),
 \qquad\nu\in\mathcal P(\operatorname{supp}\mu).
\end{equation}
This proves the second equality in~\eqref{eq:finite-initial-identification}. The normalization identity~\eqref{eq:normalization} and the type-probability bound~\eqref{eq:stirling} transfer the uniform convergence in~\eqref{eq:finite-initial-convergence} to the normalized free energies. Since $g_q$ and entropy are continuous,
\begin{equation}\label{eq:normalized-constrained-limit}
 \lim_{N\to\infty}\overline F_N^{\nu_N}(0,q)=\varphi^\nu(q)
 \quad\text{whenever }\nu_N\in\Delta_{m,N},\quad\nu_N\to\nu\text{ in }W_1.
\end{equation}

By Lemma~\ref{lem:initial-regularity}, $\varphi^\nu$ is $1$-Lipschitz in $L^1$. Together with Lemma~\ref{lem:regularity}, this extends~\eqref{eq:finite-initial-identification} and~\eqref{eq:finite-initial-convergence} from bounded paths to every $q\in\mathcal{Q}_1$ by $L^1$ approximation.

Finally, fix $Q\in\mathbb{R}_+$ and define
\begin{equation*}
 \mathcal Q_{\infty, \leqslant Q}:=\{q\in\mathcal{Q}_\infty:|q|_{L^\infty}\leqslant Q\}.
\end{equation*}
The set $\mathcal Q_{\infty, \leqslant Q}$ is relatively compact in $L^1$. Indeed, for every partition $0=s_0<\cdots<s_k<1$, positivity of the increments gives
\begin{equation*}
 \sum_{\ell=1}^k|q(s_\ell)-q(s_{\ell-1})|
 \leqslant\sum_{\ell=1}^k\operatorname{tr}\bigl(q(s_\ell)-q(s_{\ell-1})\bigr)
 \leqslant\sqrt D\,Q.
\end{equation*}
Helly's selection argument and dominated convergence therefore give $L^1$ compactness after taking right-continuous representatives. By~\eqref{eq:normalization} and Lemma~\ref{lem:regularity}, the map $q\mapsto\overline F_{N,\mu^{\otimes N},\Sigma_N(\nu)}(0,q)$ is $1$-Lipschitz in $L^1$, uniformly in $N$ and $\nu$. By~\eqref{e.g_q=Ent+varphi} and the $L^1$-Lipschitz bound for $\varphi^\nu$, the map $q\mapsto g_q(\nu)$ has the same bound, uniformly in $\nu$.

Fix $\varepsilon>0$ and choose $q^1,\ldots,q^R\in \mathcal Q_{\infty, \leqslant Q}$ such that every $q\in \mathcal Q_{\infty, \leqslant Q}$ satisfies $\min_{r\in\{1,\ldots,R\}}|q-q^r|_{L^1}<\varepsilon$. By~\eqref{eq:finite-initial-convergence}, for all sufficiently large $N$ we have
\begin{equation*}
 \max_{r\in\{1,\ldots,R\}}
 \sup_{\nu\in\Delta_{m,N}}
 |\overline F_{N,\mu^{\otimes N},\Sigma_N(\nu)}(0,q^r)-g_{q^r}(\nu)|
 <\varepsilon.
\end{equation*}
For $q\in \mathcal Q_{\infty, \leqslant Q}$, choose $r$ with $|q-q^r|_{L^1}<\varepsilon$. The two Lipschitz estimates and the preceding display give
\begin{equation*}
 \sup_{\nu\in\Delta_{m,N}}
 |\overline F_{N,\mu^{\otimes N},\Sigma_N(\nu)}(0,q)-g_q(\nu)|
 <3\varepsilon.
\end{equation*}
Since $\varepsilon>0$ is arbitrary, we obtain
\begin{equation*}
 \lim_{N\to\infty}
 \sup_{\substack{q\in\mathcal{Q}_\infty\\|q|_{L^\infty}\leqslant Q}}
 \sup_{\nu\in\Delta_{m,N}}
 |\overline F_{N,\mu^{\otimes N},\Sigma_N(\nu)}(0,q)-g_q(\nu)|=0.
\end{equation*}
The proof is complete.

 &\leqslant K\int_{I_k}\int_{I_k}|p(s)-p(r)|\,\mathrm d r\,\mathrm d s\leqslant\frac1K\biggl(\sup_{s\in I_k}\operatorname{tr} p(s)-\inf_{s\in I_k}\operatorname{tr} p(s)\biggr).
\end{align*}
The increments in the last line sum to at most $\sup_{s\in[0,1)}\operatorname{tr} p(s)$, because $\operatorname{tr} p$ is nonnegative and increasing. Consequently, for every $p\in\mathcal{Q}_\infty$, we have
\begin{equation}\label{eq:projection-L1-bound}
 |p-L_KP_Kp|_{L^1}
 \leqslant\frac1K\sup_{s\in[0,1)}\operatorname{tr} p(s)
 \leqslant\frac{\sqrt D}{K}|p|_{L^\infty}.
\end{equation}
This proves the first assertion of the lemma when $|p|_{L^\infty}\leqslant1$.

In this case, both $p(s)$ and $(L_KP_Kp)(s)$ belong to the unit ball of $S^D$. By~\eqref{e.M_i=}, the mean-value theorem gives
\begin{align*}
 \left|\int_0^1\xi(p(s))\,\mathrm d s-\xi_K(P_Kp)\right|
 &\leqslant M_1|p-L_KP_Kp|_{L^1}
 \stackrel{\eqref{eq:projection-L1-bound}}{\leqslant}\frac{M_1\sqrt D}{K}.
\end{align*}
Finally, for $p,p'\in\mathcal{Q}^K$ satisfying $|p|_{\ell^\infty}\vee|p'|_{\ell^\infty}\leqslant1$, the line segment between $p_k$ and $p_k'$ lies in the unit ball. The definition of $M_1$ gives
\begin{align*}
 |\xi_K(p)-\xi_K(p')|
 &\leqslant\frac1K\sum_{k=1}^K|\xi(p_k)-\xi(p_k')|\leqslant\frac{M_1}{K}\sum_{k=1}^K|p_k-p_k'|
 =M_1|p-p'|_{\ell^1_K}.
\end{align*}
This proves~\eqref{eq:finite-h-Lipschitz}.

\begin{equation*}
 h_\delta(a_j):=\begin{cases}
 \displaystyle\log\left(\frac1{\nu(C_j)}\int_{C_j}e^{h(a)}\,\nu(\mathrm da)\right),&\nu(C_j)>0,\\
 0,&\nu(C_j)=0.
 \end{cases}
\end{equation*}

\begin{equation*}

 \quad\text{and}\quad
 \int e^h\,\mathrm d\nu=\int e^{h_\delta}\,\mathrm d\nu_\delta.
\end{equation*}

\begin{equation}\label{eq:tilt-discretization}

 \leqslant2|q|_{L^\infty}\delta.

 \int_{C_j}h\,\mathrm d\nu\leqslant\nu(C_j)h_\delta(a_j),

\end{equation*}

 |\varphi^\nu(L_Kb)-\varphi^{\nu_\delta}(L_Kb)|\leqslant2|b|_{\ell^\infty}\delta.
\end{equation*}
Apply Lemma~\ref{lem:finite-local-comparison} in both directions to the already-defined solutions $f^{\nu,K}$ and $f^{\nu_\delta,K}$. For $t\leqslant T$ and $|a|_{\ell^\infty}\leqslant Q$,
\begin{equation}\label{eq:general-finite-comparison}
 |f^{\nu,K}(t,a)-f^{\nu_\delta,K}(t,a)|\leqslant2(Q+M_1T+1)\delta.
\end{equation}
Taking $a=P_Kq$ and letting $K\to\infty$ in~\eqref{eq:HJ-discrete-limit}, we can deduce from the previous display that $f^{\nu_\delta}\to f^\nu$ uniformly on $[0,T]\times\mathcal Q_{\infty,\leqslant Q}$. This means that if we approximate the measures $\nu$ and $\nu'$ by $\nu_\delta$ and $\nu'_{\delta'}$, apply Lemma~\ref{lem:finite-measure-continuity}, and let $\delta,\delta'\downarrow0$, we obtain~\eqref{eq:measure-continuity}.

For $q\in\mathcal{Q}_2$, put $q^{(s)}(r):=q(r\wedge s)$ for $s<1$. If $|q|_{L^2}\leqslant R$, monotonicity and the Cauchy--Schwarz inequality give
\begin{equation}\label{eq:uniform-truncation}
 |q^{(s)}|_{L^\infty}\leqslant\frac R{\sqrt{1-s}},
 \qquad |q-q^{(s)}|_{L^1}\leqslant2R\sqrt{1-s}.
\end{equation}
Indeed, $(1-s)|q(s)|^2\leqslant\int_s^1|q(r)|^2\,\mathrm d r$, and $|q-q^{(s)}|_{L^1}\leqslant\int_s^1|q(r)|\,\mathrm d r+(1-s)|q(s)|$. Apply~\eqref{eq:solution-regularity} at $q$ and $q^{(s)}$, and~\eqref{eq:measure-continuity} at the bounded path. We obtain, for all target measures,
\begin{equation}\label{eq:L2-measure-continuity}
 \sup_{\substack{0\leqslant t\leqslant T,\ q\in\mathcal{Q}_2\\|q|_{L^2}\leqslant R}}
 |f^\nu(t,q)-f^{\nu'}(t,q)|
 \leqslant4R\sqrt{1-s}+2\left(TM_1+\frac R{\sqrt{1-s}}\right)W_1(\nu,\nu').
\end{equation}
First choose $s$ close to $1$ and then let $W_1(\nu,\nu')\to0$. This proves continuity and also uniform convergence of the finite discretizations on bounded subsets of $\mathcal{Q}_2$.

Write $\operatorname{supp}\mu_\delta=\{a_1,\ldots,a_m\}$ and define the cells $C_j:=\mathcal T_\delta^{-1}(\{a_j\})$. Then $\mu(C_j)=\mu_\delta(\{a_j\})>0$, and these cells cover $\operatorname{supp}\mu$ up to a $\mu$-null set.

\smallskip
\noindent\emph{Entropy contraction.}
Fix $\nu\in\mathcal P(\operatorname{supp}\mu)$ with $\operatorname{Ent}(\nu\mid\mu)<\infty$ and let $\eta:=(\mathcal T_\delta)_\#\nu$. 
The coupling $(a,\mathcal T_\delta(a))$ gives $W_1(\nu,\eta)\leqslant\delta$. In addition, relative entropy decreases under the measurable map $\mathcal T_\delta$:
\begin{equation}\label{eq:entropy-contraction}
 \operatorname{Ent}(\eta\mid\mu_\delta)\leqslant\operatorname{Ent}(\nu\mid\mu).
\end{equation}
Indeed, write $r:=\mathrm d\nu/\mathrm d\mu$. On every cell with $\mu(C_j)>0$, Jensen's inequality for $x\mapsto x\log x$ gives
\begin{equation*}
 \int_{C_j}r\log r\,\mathrm d\mu
 \geqslant\nu(C_j)\log\frac{\nu(C_j)}{\mu(C_j)}.
\end{equation*}
Summing these inequalities proves~\eqref{eq:entropy-contraction}. Combining it with~\eqref{eq:measure-continuity}, we obtain
\begin{align*}
 \operatorname{Ent}(\eta\mid\mu_\delta)+f^\eta(t,q)
 &\leqslant\operatorname{Ent}(\nu\mid\mu)+f^\nu(t,q)
 +2\bigl(tM_1+|q|_{L^\infty}\bigr)\delta.
\end{align*}
Taking the infimum over such $\nu$ proves the first inequality.

\smallskip
\noindent\emph{Entropy-preserving lift.}
Fix $\eta\in\mathcal P(\operatorname{supp}\mu_\delta)$ and let $\beta_j:=\eta(\{a_j\})$. Define $\widetilde\nu\in\mathcal P(\operatorname{supp}\mu)$ by
\begin{equation}\label{eq:entropy-preserving-lift}
 \widetilde\nu:=\sum_{j=1}^m\beta_j\,\mu(\,\cdot\mid C_j).
\end{equation}
The pushforward $(\mathcal T_\delta)_\#\widetilde\nu$ equals $\eta$, so $W_1(\widetilde\nu,\eta)\leqslant\delta$. Moreover, the density of the measure in~\eqref{eq:entropy-preserving-lift} with respect to $\mu$ equals $\beta_j/\mu(C_j)$ on $C_j$. Consequently, we have
\begin{equation}\label{eq:entropy-preserving-identity}
 \operatorname{Ent}(\widetilde\nu\mid\mu)
 =\sum_{j=1}^m\beta_j\log\frac{\beta_j}{\mu(C_j)}
 =\operatorname{Ent}(\eta\mid\mu_\delta).
\end{equation}
Using~\eqref{eq:entropy-preserving-identity} and~\eqref{eq:measure-continuity}, we therefore obtain
\begin{align*}
 \operatorname{Ent}(\widetilde\nu\mid\mu)+f^{\widetilde\nu}(t,q)
 &\leqslant\operatorname{Ent}(\eta\mid\mu_\delta)+f^\eta(t,q)
 +2\bigl(tM_1+|q|_{L^\infty}\bigr)\delta.
\end{align*}
Taking the infimum over $\eta$ completes~\eqref{eq:envelope-discretization}.

\smallskip
\noindent\emph{Step 1: Uniform convergence on bounded paths.}
Fix $T,Q\in\mathbb{R}_+$. Since $\mu_\delta$ has finite support, Corollary~\ref{cor:finite-main} gives
\begin{equation}\label{eq:quantized-free-energy-limit}
 \lim_{N\to\infty}\sup_{\substack{0\leqslant t\leqslant T,\ q\in\mathcal Q_{\infty,\leqslant Q}}}
 |\overline F_{N,\mu_\delta^{\otimes N}}(t,q)-J_\delta(t,q)|=0.
\end{equation}
For every $(t,q)\in[0,T]\times\mathcal Q_{\infty,\leqslant Q}$, the triangle inequality gives
\begin{align*}
 |\overline F_{N,\mu^{\otimes N}}(t,q)-J(t,q)|
 &\leqslant|\overline F_{N,\mu^{\otimes N}}(t,q)-\overline F_{N,\mu_\delta^{\otimes N}}(t,q)|\\
 &\quad+|\overline F_{N,\mu_\delta^{\otimes N}}(t,q)-J_\delta(t,q)|
 +|J_\delta(t,q)-J(t,q)|.
\end{align*}
By~\eqref{eq:reference-discretization}, Lemma~\ref{lem:envelope-discretization}, and~\eqref{eq:quantized-free-energy-limit}, we therefore obtain
\begin{equation}\label{eq:general-uniform-error}
 \limsup_{N\to\infty}\sup_{\substack{0\leqslant t\leqslant T,\ q\in\mathcal Q_{\infty,\leqslant Q}}}
 |\overline F_{N,\mu^{\otimes N}}(t,q)-J(t,q)|
 \leqslant4\delta(TM_1+Q).
\end{equation}
Since $\delta>0$ is arbitrary, sending $\delta\downarrow0$ in~\eqref{eq:general-uniform-error} proves uniform convergence on $[0,T]\times\mathcal Q_{\infty,\leqslant Q}$.

\smallskip
\noindent\emph{Step 2: Extension to every $q\in\mathcal{Q}_2$.}
Fix $t\in\mathbb{R}_+$ and $q\in\mathcal{Q}_2$. For $s\in[0,1)$, take $q^{(s)}$ from~\eqref{eq:uniform-truncation}. Lemma~\ref{lem:regularity} and~\eqref{eq:solution-regularity} imply that both $q\mapsto\overline F_{N,\mu^{\otimes N}}(t,q)$ and $q\mapsto J(t,q)$ are $1$-Lipschitz in $L^1$; for $J$, this follows by taking the infimum of functions with the same Lipschitz constant. Hence
\begin{align*}
 |\overline F_{N,\mu^{\otimes N}}(t,q)-J(t,q)|
 &\leqslant2|q-q^{(s)}|_{L^1}
 +|\overline F_{N,\mu^{\otimes N}}(t,q^{(s)})-J(t,q^{(s)})|.
\end{align*}
The second term tends to zero as $N\to\infty$ for fixed $s$, by Step 1. By~\eqref{eq:uniform-truncation}, the first term tends to zero as $s\uparrow1$. This proves~\eqref{eq:main} with the infimum in place of the minimum.

\smallskip
\noindent\emph{Step 3: Attainment of the infimum.}
The space $\mathcal P(\operatorname{supp}\mu)$ is compact in $W_1$, since $\operatorname{supp}\mu$ is compact. Relative entropy is lower semicontinuous in $W_1$. By~\eqref{eq:L2-measure-continuity}, the map $\nu\mapsto f^\nu(t,q)$ is continuous in $W_1$ for every $q\in\mathcal{Q}_2$. Therefore, the objective
\begin{equation*}
 \nu\longmapsto\operatorname{Ent}(\nu\mid\mu)+f^\nu(t,q)
\end{equation*}
is lower semicontinuous on $\mathcal P(\operatorname{supp}\mu)$. It is finite at $\nu=\mu$, since $\operatorname{Ent}(\mu\mid\mu)=0$ and $f^\mu(t,q)$ is finite. Compactness therefore yields a minimizer, proving~\eqref{eq:main} as stated.

Fix $K\in\mathbb{N}$. Use the weighted inner product $\langle\cdot,\cdot\rangle_K$ and let $|a|_K:=\langlea,a\rangle_K^{1/2}$. Then
\begin{equation*}
 |a|_{\ell^1_K}\leqslant|a|_K\leqslant|a|_{\ell^\infty},
 \qquad a\in E_K,
\end{equation*}
and $|\cdot|_{\ell^1_K}$ and $|\cdot|_{\ell^\infty}$ are dual norms. By assumption, for some $C\in\mathbb{R}_+$,
\begin{equation}\label{eq:app-time-growth}
 |u(t,a)-u(0,a)|\vee|v(t,a)-v(0,a)|\leqslant Ct,
 \qquad(t,a)\in\mathbb{R}_+\times\mathcal{Q}^K.
\end{equation}

\smallskip
\noindent\emph{Localization and the boundary barrier.}
Put $R:=Q+M_1T+1$ and
\begin{equation*}
 S_R:=\sup_{a\in\mathcal{Q}^K:\,|a|_{\ell^\infty}\leqslant R}(u(0,a)-v(0,a)).
\end{equation*}
Radial projection onto this ball, together with the spatial Lipschitz bounds, gives
\begin{equation}\label{eq:app-initial-radial}
 u(0,a)-v(0,a)\leqslant S_R+2(|a|_{\ell^\infty}-R)_+,
 \qquad a\in\mathcal{Q}^K.
\end{equation}
For $\kappa\in(0,1/4)$, define the smooth function
\begin{equation}\label{eq:app-smoothed-norm}
 \rho_\kappa(a):=\sqrt{|a_K|^2+\kappa^2}.
\end{equation}
On $\mathcal{Q}^K$, positivity of the increments implies $|a|_{\ell^\infty}=|a_K|$, so
\begin{equation*}
 |a|_{\ell^\infty}\leqslant\rho_\kappa(a)\leqslant|a|_{\ell^\infty}+\kappa,
 \qquad a\in\mathcal{Q}^K.
\end{equation*}
Moreover, $\nabla_K\rho_\kappa$ has only its last block nonzero, equal to $Ka_K/\rho_\kappa(a)$, and hence $|\nabla_K\rho_\kappa|_{\ell^1_K}\leqslant1$ on $E_K$. Choose a smooth nondecreasing $\theta_\kappa:\mathbb{R}\to\mathbb{R}_+$ with $0\leqslant\theta_\kappa'\leqslant1$, $\theta_\kappa(r)=0$ for $r\leqslant0$, and $\theta_\kappa(r)\geqslant(r-\kappa)_+$. Let
\begin{equation}\label{eq:app-local-cutoff}
 \chi(t,a):=3\theta_\kappa(\rho_\kappa(a)+M_1t-R).
\end{equation}
The preceding bounds give
\begin{align}\label{eq:app-cutoff-properties}
 \partial_t\chi&\geqslant M_1|\nabla_K\chi|_{\ell^1_K},\nonumber\\
 \chi(t,a)&=0,\qquad 0\leqslant t\leqslant T,\quad a\in\mathcal{Q}^K,\quad |a|_{\ell^\infty}\leqslant Q,\nonumber\\
 \chi(t,a)&\geqslant3(|a|_{\ell^\infty}-R-2\kappa)_+,
 \qquad(t,a)\in\mathbb{R}_+\times\mathcal{Q}^K.
\end{align}
In particular,
\begin{equation}\label{eq:app-initial-cutoff}
 \sup_{a\in\mathcal{Q}^K}(u(0,a)-v(0,a)-\chi(0,a))\leqslant S_R+6\kappa.
\end{equation}

For $a\in\operatorname{int}\mathcal{Q}^K$, put $a_0:=0$ and define
\begin{equation}\label{eq:app-boundary-function}
 b(a):=\sum_{k=1}^K\operatorname{tr}[(a_k-a_{k-1})^{-1}].
\end{equation}
This smooth positive function diverges at the boundary. For $c\in\mathcal{Q}^K$, differentiation gives
\begin{equation}\label{eq:app-boundary-monotone}
 \langle\nabla_Kb(a),c\rangle_K
 =-\sum_{k=1}^K\operatorname{tr}[(a_k-a_{k-1})^{-2}(c_k-c_{k-1})]\leqslant0,
 \qquad c_0:=0.
\end{equation}
Thus $-\nabla_Kb(a)\in(\mathcal{Q}^K)^*$.

\smallskip
\noindent\emph{Doubled variables and interior contact points.}
Suppose the comparison fails. There are $t_*\in(0,T]$ and $a_*\in\mathcal{Q}^K$, with $|a_*|_{\ell^\infty}\leqslant Q$, such that $u(t_*,a_*)-v(t_*,a_*)>S_R$. Fix $T_*>T$. Choose $\kappa$, and then $\gamma,\beta>0$, small enough that
\begin{equation*}
 u(t_*,a_*)-v(t_*,a_*)-\chi(t_*,a_*)-\gamma t_*
 -\frac{2\beta}{T_*-t_*}>S_R+12\kappa.
\end{equation*}
By continuity and density of $\operatorname{int}\mathcal{Q}^K$, choose $\widehat a_*\in\operatorname{int}\mathcal{Q}^K$ close to $a_*$, and then $\sigma>0$ small enough that
\begin{equation}\label{eq:app-diagonal-gap}
 u(t_*,\widehat a_*)-v(t_*,\widehat a_*)-\chi(t_*,\widehat a_*)-\gamma t_*
 -\frac{2\beta}{T_*-t_*}-\sigma b(\widehat a_*)>S_R+12\kappa.
\end{equation}
Keep these parameters fixed. For $\alpha>1$, define on $[0,T_*)^2\times\operatorname{int}\mathcal{Q}^K\times\mathcal{Q}^K$
\begin{align}\label{eq:app-doubled-function}
 \Psi_\alpha(t,t',a,a')
 &:=u(t,a)-v(t',a')-\frac\alpha2(|t-t'|^2+|a-a'|_K^2)\nonumber\\
 &\quad-\chi(t,a)-\gamma t-\frac\beta{T_*-t}-\frac\beta{T_*-t'}-\sigma b(a).
\end{align}
The diagonal point from~\eqref{eq:app-diagonal-gap} gives
\begin{equation}\label{eq:app-max-lower}
 \sup\Psi_\alpha>S_R+12\kappa.
\end{equation}
The time-growth and spatial Lipschitz assumptions imply
\begin{equation}\label{eq:app-growth-bound}
 u(t,a)-v(t',a')\leqslant S_R+2(|a|_{\ell^\infty}-R)_++2CT_*+|a-a'|_{\ell^1_K}.
\end{equation}
The cutoff with coefficient $3>2$ controls $a$, the quadratic penalty controls $a'$, the time penalties exclude $T_*$, and $\sigma b(a)$ excludes $\partial\mathcal{Q}^K$. Hence $\Psi_\alpha$ attains a maximum at $(t_\alpha,t'_\alpha,a_\alpha,a'_\alpha)$. Equations~\eqref{eq:app-growth-bound} and~\eqref{eq:app-max-lower}, using $x-\alpha x^2/2\leqslant1/(2\alpha)$, give a constant $C_*<\infty$, independent of $\alpha>1$, such that
\begin{equation}\label{eq:app-compactness}
 |a_\alpha|_K\vee|a'_\alpha|_K\leqslant C_*,
 \qquad
 \alpha(|t_\alpha-t'_\alpha|^2+|a_\alpha-a'_\alpha|_K^2)+b(a_\alpha)\leqslant C_*.
\end{equation}
Since each inverse trace is bounded by $C_*$, every increment of $a_\alpha$ has smallest eigenvalue at least $C_*^{-1}$. Quadratic closeness gives $|a_\alpha-a'_\alpha|_K\to0$; for fixed $K$, Weyl's inequality then shows $a'_\alpha\in\operatorname{int}\mathcal{Q}^K$ for all sufficiently large $\alpha$.

If either contact time is zero, the time-growth assumption gives
\begin{equation*}
 u(t_\alpha,a_\alpha)-v(t'_\alpha,a'_\alpha)
 \leqslant u(0,a_\alpha)-v(0,a_\alpha)+C|t_\alpha-t'_\alpha|+|a_\alpha-a'_\alpha|_K.
\end{equation*}
Since $\chi(t_\alpha,a_\alpha)\geqslant\chi(0,a_\alpha)$, completing squares and using~\eqref{eq:app-initial-cutoff} yields
\begin{equation*}
 \Psi_\alpha(t_\alpha,t'_\alpha,a_\alpha,a'_\alpha)
 \leqslant S_R+6\kappa+\frac{C^2+1}{2\alpha},
\end{equation*}
contradicting~\eqref{eq:app-max-lower} for large $\alpha$. Fix such an $\alpha$, so both contact points lie in $(0,T_*)\times\operatorname{int}\mathcal{Q}^K$.

\smallskip
\noindent\emph{Hamiltonian comparison and viscosity inequalities.}
The test gradients for $u$ and $v$ are
\begin{equation}\label{eq:app-contact-gradients}
 p^v:=\alpha(a_\alpha-a'_\alpha),
 \qquad
 p^u:=p^v+\nabla_K\chi(t_\alpha,a_\alpha)+\sigma\nabla_Kb(a_\alpha).
\end{equation}
At interior contact points, the spatial Lipschitz and $(\mathcal{Q}^K)^*$-monotonicity assumptions give
\begin{equation}\label{eq:app-gradient-cone}
 p^u,p^v\in\mathcal{Q}^K,
 \qquad |p^u|_{\ell^\infty}\vee|p^v|_{\ell^\infty}\leqslant1.
\end{equation}
For example, test monotonicity in a direction $c\in(\mathcal{Q}^K)^*$ at $a_\alpha+hc$ for the upper test and at $a'_\alpha-hc$ for the lower test. Letting $h\downarrow0$ gives nonnegative pairing with every $c\in(\mathcal{Q}^K)^*$, hence membership in $((\mathcal{Q}^K)^*)^*=\mathcal{Q}^K$. Testing arbitrary directions gives the norm bounds.

The monotonicity of $\xi_K$ and the definition of $M_1$ imply
\begin{equation}\label{eq:app-h-gradient}
 \nabla_K\xi_K(p)\in\mathcal{Q}^K,
 \qquad |\nabla_K\xi_K(p)|_{\ell^\infty}\leqslant M_1,
 \qquad p\in\mathcal{Q}^K,\quad |p|_{\ell^\infty}\leqslant1.
\end{equation}
The first inclusion follows by differentiating monotonicity in directions of $(\mathcal{Q}^K)^*$ at interior points and extending by continuity; the norm bound follows from $(\nabla_K\xi_K(p))_k=\nabla\xi(p_k)$. By~\eqref{eq:app-boundary-monotone},
\begin{equation}\label{eq:app-boundary-dual}
 r:=-\sigma\nabla_Kb(a_\alpha)\in(\mathcal{Q}^K)^*,
 \qquad p^u-p^v=\nabla_K\chi(t_\alpha,a_\alpha)-r.
\end{equation}
Integrating the derivative of $\xi_K$ along the segment between $p^v$ and $p^u$, which lies in the set in~\eqref{eq:app-gradient-cone}, gives
\begin{align}\label{eq:app-h-difference}
 \xi_K(p^u)-\xi_K(p^v)
 &=\int_0^1\langle\nabla_K\xi_K(p^v+s(p^u-p^v)),\nabla_K\chi-r\rangle_K\,\mathrm d s\nonumber\\
 &\leqslant M_1|\nabla_K\chi(t_\alpha,a_\alpha)|_{\ell^1_K}.
\end{align}
The time derivatives of the upper and lower tests are respectively
\begin{equation*}
 \alpha(t_\alpha-t'_\alpha)+\partial_t\chi(t_\alpha,a_\alpha)+\gamma
 +\frac\beta{(T_*-t_\alpha)^2},
 \qquad
 \alpha(t_\alpha-t'_\alpha)-\frac\beta{(T_*-t'_\alpha)^2}.
\end{equation*}
Subtracting the viscosity supersolution inequality from the subsolution inequality, and discarding the positive time-penalty terms, yields
\begin{equation*}
 \gamma\leqslant \xi_K(p^u)-\xi_K(p^v)-\partial_t\chi(t_\alpha,a_\alpha)
 \leqslant M_1|\nabla_K\chi|_{\ell^1_K}-\partial_t\chi\leqslant0,
\end{equation*}
by~\eqref{eq:app-cutoff-properties}. This contradicts $\gamma>0$ and proves the comparison.
