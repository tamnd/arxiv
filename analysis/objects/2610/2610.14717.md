\label{vp_hamel}
The following statements are equivalent:
	\begin{enumerate}
		\item[(i)] The map \(\phi\) extremizes the action functional
		\begin{align*}
		    \int_U L(\phi,\mathbf{d}\phi)\,\mathrm{d}s\,\mathrm{d}t,
		\end{align*}
		for all variations \(\delta\phi\) vanishing on \(\partial U\).
        
		\item[(ii)] The map \(\phi\) satisfies the Euler-Lagrange equations
		\begin{align*}
		  \frac{\partial}{\partial t}\frac{\partial L}{\partial\phi_t} + \frac{\partial}{\partial s}\frac{\partial L}{\partial\phi_s} - \frac{\partial L}{\partial\phi} = 0.
		\end{align*}
        
		\item[(iii)] The map \((\phi,\xi)\) extremizess the functional
		\begin{align*}
		  \int_U l(\phi,\xi)\,\mathrm{d}s\,\mathrm{d}t
		\end{align*}
		with respect to variations $\delta\xi$, induced by the variations $\delta\phi = \Psi_{\phi}\eta$, and given by $\delta\xi = \mathbf{d}\eta + [\xi,\eta]_{\phi}$, where $\eta: U\rightarrow V$ vanishes on $\partial U$. 

		\item[(iv)] The map $(\phi,\xi)$ satisfies field-theoretic Hamel's equations
		\begin{align*}
		\frac{\partial}{\partial t}\frac{\partial l}{\partial\zeta}
		+ \frac{\partial}{\partial s}\frac{\partial l}{\partial\gamma}
		- \left[\zeta,\frac{\partial l}{\partial\zeta}\right]_{\phi}^{*}
		- \left[\gamma,\frac{\partial l}{\partial\gamma}\right]_{\phi}^{*}
		- \Psi_{\phi}^{*}\frac{\partial l}{\partial\phi} = 0
		\end{align*}
		coupled with the kinematic equation $\mathbf{d}\phi = \Psi_{\phi}\xi$.
	\end{enumerate}

Consider the linear wave equation, with state  $\phi\in\mathbb{R}$, $t\in\mathbb{R}$ being the temporal variable and $s\in[0,S]$ being the spatial variable. The covariant configuration bundle is therefore $\pi_{BE}: B\times G\rightarrow B$, where the base space $B=\mathbb{R}\times[0,S]$ and the fiber $G=\mathbb{R}$. The Lagrangian of the wave motion is the kinetic energy minus the potential energy
\begin{equation}
\ell(g,\zeta,\gamma) = \frac{1}{2}\left(\zeta^{2} - \gamma^{2}\right).
\label{L_1}
\end{equation}
Similar to the geometrically exact beam, the fiber of this system is also a Lie group, but the trivial one $Q=\mathbb{R}$. By selecting moving frames to be the identity map $T_{e}L_{\phi} = \mathrm{Id}_{\mathbb{R}}$, the operators $\mathrm{ad}$ and $\mathrm{ad}^{*}$ are trivial, that is, $\mathrm{ad}_{\zeta} = 0$ and $\mathrm{ad}^{*}_{\zeta} = 0$. Hence, the stochastic equations of motion \eqref{sto_Hamel}-\eqref{sto_com} for the wave system read
\begin{equation}
\begin{aligned}
d_{t}\zeta &= \gamma_{s}dt, \\
d_{t}\phi &= \zeta dt + \Theta\circ dW,  \\
\phi_{s} &= \gamma,  \\
d_{t}\gamma &= \zeta_{s}dt + \Theta_{s}\circ dW,
\label{wave1}
\end{aligned}
\end{equation}
where $\Theta: \mathbb{R}\times[0,S]\rightarrow\mathbb{R}$ is the diffusivity vector field, and $(\cdot)_{s} = \partial/\partial s$ represents taking the spatial derivative. If $\Theta$ is chosen to be zero, the above equations \eqref{wave1} specialize to
\begin{align*}
\zeta_{t} = \gamma_{s}, 
\quad
\phi_{t} = \zeta, 
\quad
\phi_{s} = \gamma, 
\quad
\gamma_{t} = \zeta_{s},
\end{align*}
which recovers the deterministic wave equation $\left(-\frac{\partial^{2}}{\partial t^{2}} + \triangle\right)\phi = 0$ in \cite{evans2022partial}.

\label{equi_continuous}

\label{equivalence}
\end{align}

\label{equivalence}
\end{align*}
where $C\in\mathbb{R}$ is a constant.

\label{sto_wave_phi}
For the stochastic wave motion, the stochastic field-theoretic equations \eqref{e8}-\eqref{e7} and \eqref{e9} specialize to

\mu = \zeta, 
\quad
\sigma = -\gamma, 
\quad
\gamma_{t} = \zeta_{s},
\end{align*}
which recovers the deterministic wave equation \citep{evans2022partial}.

\label{Coro_Energy}
If the Hamiltonian $h$ Poisson-commutes with the stochastic potential $\Phi$, that is, $\left\{h,\Phi\right\} = 0$, then the total energy of the stochastic system is a conserved quantity.

<!-- rmk-2: body_md -->
\label{Choice_Phi}
There are some specific choices of $\Phi$ such that the energy is conserved. (a) A trivial situation occurs when $\Phi$ is a Casimir function \citep{marsden1999}, which Poisson-commutes with $\mu$ and $\gamma$ and therefore makes the equations of motion implicitly stochastic according to \eqref{SLP}. (b) Other interesting but nontrivial candidates of $\Phi$ that conserve energy include smooth functions of the Hamiltonian $h$ (that is, $\Phi = \Psi(h)$, where $\Psi$ is a differentiable function), then the affine Lie-Poisson bracket $\left\{h,\Phi\right\}
= D\Psi(h)\left\{h, h\right\}
= 0$, which implies that the total energy is conserved.

\label{ChenShi2025}
Given a field-theoretic mechanical system with a $G$-invariant Lagrangian $\ell\left(\zeta,\gamma\right) = L(g,g\zeta,g\gamma)$, the Lie-Poisson bracket is given by
\begin{equation*}
\left\{f,g\right\}(\gamma,\mu)
=
\left<
\frac{\delta f}{\delta\gamma},
\frac{\partial}{\partial s}\frac{\delta g}{\delta\mu} 
+ \mathrm{ad}_{\gamma}\frac{\delta g}{\delta\mu}
\right>
+
\left<
\mathrm{ad}^{*}_{\frac{\delta g}{\delta\mu}}\mu
+ \frac{\partial}{\partial s}\frac{\delta g}{\delta\gamma}
- \mathrm{ad}^{*}_{\gamma}\frac{\delta g}{\delta\gamma},
\frac{\delta f}{\delta\mu}
\right>.
\end{equation*}

Using the Stratonovich chain rule and \eqref{sto_Hamel_Ham}-\eqref{sto_com_Ham}, we have
\begin{align*}
d_{t}f(\gamma,\mu) &= \left<\frac{\delta f}{\delta\gamma},d_{t}\gamma\right> + \left<d_{t}\mu, \frac{\delta f}{\delta\mu}\right> \\
&=
\left<
\frac{\delta f}{\delta\gamma},
\left(
\frac{\partial}{\partial s}\frac{\delta h}{\delta\mu}
+ \mathrm{ad}_{\gamma}\frac{\delta h}{\delta\mu}
\right)dt
+ \left(
\mathrm{ad}_{\gamma}\frac{\delta\Phi}{\delta\mu}
+ \frac{\partial}{\partial s}\frac{\delta\Phi}{\delta\mu}
\right)\circ dW
\right>  \\
&\quad
+
\left<
\left(
\mathrm{ad}^{*}_{\frac{\delta h}{\delta\mu}}\mu
+ \frac{\partial}{\partial s}\frac{\delta h}{\delta\gamma}
- \mathrm{ad}^{*}_{\gamma}\frac{\delta h}{\delta\gamma}
\right)dt
+ \mathrm{ad}^{*}_{\frac{\delta\Phi}{\delta\mu}}\mu\circ dW,
\frac{\delta f}{\delta\mu}
\right>  \\
&=
\left(
\left<
\frac{\delta f}{\delta\gamma},
\frac{\partial}{\partial s}\frac{\delta h}{\delta\mu} 
+ \mathrm{ad}_{\gamma}\frac{\delta h}{\delta\mu}
\right>
+
\left<
\mathrm{ad}^{*}_{\frac{\delta h}{\delta\mu}}\mu
+ \frac{\partial}{\partial s}\frac{\delta h}{\delta\gamma}
- \mathrm{ad}^{*}_{\gamma}\frac{\delta h}{\delta\gamma},
\frac{\delta f}{\delta\mu}
\right>
\right)dt  \\
&\quad
+
\left(
\left<
\frac{\delta f}{\delta\gamma},
\mathrm{ad}_{\gamma}\frac{\delta\Phi}{\delta\mu}

\right>
+

\mathrm{ad}^{*}_{\frac{\delta\Phi}{\delta\mu}}\mu,

\right)\circ dW  \\

\left\{f,h\right\}dt + \left\{f,\Phi\right\}\circ dW,
