<!-- proof-1: body_md -->
For $t\in\mathbb R$, set $\psi(t)=t-\lfloor t\rfloor-1/2$, and fix
		a nonintegral $X\ge4$.
		The classical reduction \cite[proof of \mbox{\hyperlink{cite.GK}{Theorem~4.8}}]{GK}, followed by
		dyadic subdivision, expresses $E(X)$, up to $O(1)$, as a linear
		combination with coefficients $\pm4$ of $O(\log X)$ sums
		\[
		\sum_{n\in I\cap\mathbb Z}\psi\!\left(\frac XM F(n/M)\right),
		\qquad I\subseteq[M,2M],\quad 1\le M\le\sqrt X,
		\]
		where $I$ is an interval and $F$ is as in \eqref{eq:S}, with
		\[
		(\alpha,\kappa)\in
		\left\{\left(\pm\frac1{4M},0\right),
		\left(0,\pm\frac M{4X}\right)\right\}.
		\]
		Thus the four arguments are exactly $X/(4n\pm1)$ and
		$X/(4n)\pm1/4$. These are the four residue-class and sawtooth phases
		in the cited circle reduction. This identifies that standard reduction
		with the reciprocal family \eqref{eq:S} without adding a new analytic
		input. The pointwise Vaaler majorant applies to the convention
		$\psi(t)=\{t\}-1/2$ used here, including integral arguments.
		For any integer $J\ge2$, Vaaler's approximation
		\cite[\mbox{\hyperlink{cite.Vaaler85}{Theorem~18}}]{Vaaler85} and partial summation in $h$ give
		\begin{equation}\label{eq:Vaaler}
			\begin{split}
				&\left|\sum_{n\in I\cap\mathbb Z}\psi\!\left(\frac XM F(n/M)\right)\right|\\
				&\quad\lesssim\frac MJ+
				\sum_{\substack{H=2^j<J\\j\ge0}}\frac1H
				\sup_{H\le H'\le\min(2H,J)}
				\left|\sum_{H\le h<H'}\sum_{n\in I\cap\mathbb Z}
				e\!\left(\frac{hX}{M}F(n/M)\right)\right|.
			\end{split}
		\end{equation}
		For $M\ge2X^\beta$, choose $J=\lfloor MX^{-\beta}\rfloor$.
		Then $J\ge2$, $M/J\lesssim X^\beta$, and every $H<J$ satisfies
		$H\le MX^{-\beta}$. \mbox{\hyperref[thm:recip]{Theorem~\ref*{thm:recip}}}, with $T=X$ and
		$g=G=1$, bounds each supremum in \eqref{eq:Vaaler} by
		$O_\varepsilon(HX^{\beta+\varepsilon})$. After multiplication by
		$H^{-1}$, each dyadic $h$-block therefore contributes
		$O_\varepsilon(X^{\beta+\varepsilon})$. There are only
		$O(\log X)$ such blocks. The dyadic $n$-blocks with
		$M<2X^\beta$ have total length
		$O(\sum_{M<2X^\beta}M)=O(X^\beta)$, so their contribution is
		trivial. Absorbing the logarithm into $X^\varepsilon$ proves the claim
		for nonintegral $X$. Finally, the counting function in the definition of
		$E(X)$ is right-continuous, hence so is $E(X)$. Letting a nonintegral
		argument decrease to an integral $X$ gives the same estimate there.

<!-- proposition-2: body_md -->
\label{prop:preliminary}
		For the reciprocal sum \eqref{eq:S}, uniformly on subintervals,
		\begin{equation}\label{eq:preliminary}
			|S|/H\lesssim (HT)^{2/7}+\frac{M^2}{HT},\qquad
			|S|/H\lesssim \rho^{13/31}M^{3/31}+\frac{M^2}{HT}.
		\end{equation}
		Consequently $|S|/H\lesssim T^\beta$ if either $H\le T^s$ or
		$p\le42/67$.
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
The exponent pairs $(k,l)=(2/7,4/7)$ and $(13/31,16/31)$
		\cite[Chapter~3, (3.3.4)]{GK} give, for every integer $h\approx H$
		and every interval $I\subseteq[M,2M]$,
		\[
		\left|\sum_{n\in I\cap\mathbb Z}e\!\left(\frac{hT}{4(n+M\alpha)}\right)\right|
		\lesssim (hT/M^2)^kM^l+\frac{M^2}{hT}.
		\]
		The constants are uniform in $\alpha$. Put
		$u=n+\lfloor M\alpha\rfloor$ and $a=\{M\alpha\}$. Then
		$n+M\alpha=u+a$, with $u\in\mathbb Z$, $u\eqsim M$ and $0\le a<1$.

		\lesssim\Delta L\qquad(\Delta\ge L+Z).
		\]
		This proves \eqref{eq:coarse-energy}.
		
		For every pair of projected summands, Young's and H\"older's
		inequalities give
		\[
		\|\psi_\vartheta*(u_{\theta,kl}\overline{u_{\theta,k'l'}})\|_2
		\le\|u_{\theta,kl}\|_4\|u_{\theta,k'l'}\|_4\lesssim V^{1/2}.
		\]
		Each term in the expansion of $\int_{\mathbb R^3}T_\vartheta(x)^2\,dx$
		is therefore $O(V)$. In a nonzero term, the integers $k_i,k_i'$
		lie in the fixed interval for $\vartheta$, and $l_i,l_i'\approx L$
		for $i=1,2$. Fourier support gives
		\[
		\left|\sum_{i=1}^2(l_i'-l_i)\right|\le4c_0/K<1,\qquad
		\left|\sum_{i=1}^2(k_i'l_i'-k_il_i)\right|\le4c_0<1.
		\]
		Both sums vanish. Define
		$d_i=k_i'-k_i$ for $i=1,2$, and $v=l_1'-l_1$. Then
		\begin{equation}\label{eq:paired}
			l_2'-l_2=-v,\qquad
			l_1d_1+l_2d_2=v(k_2'-k_1'),\qquad
			|d_i|\lesssim Z/\Delta\quad(i=1,2).
		\end{equation}
		For this upper bound, discard the $l$-difference restriction in
		\eqref{eq:unrounded-pairs}.
		
		If $(d_1,d_2)\ne(0,0)$, interchange the two pairs so that $d_1\ne0$.
		Choose $k_1',k_2',v,d_2,l_2$ first. The number of choices is
		\[
		O(\Delta^2)\,O(L)\,O(1+Z/\Delta)\,O(L)
		=O\bigl(\Delta^2L^2(1+Z/\Delta)\bigr).
		\]
		For an admissible choice, the nonzero integer
		$D=v(k_2'-k_1')-l_2d_2$ satisfies
		\[
		l_1d_1=D,\qquad
		0<|D|=|l_1d_1|\lesssim L(1+Z/\Delta)\lesssim KL.
		\]
		The divisor bound \cite[\mbox{\hyperlink{cite.HW}{Theorem~315}}]{HW} gives
		$O_\varepsilon((KL)^\varepsilon)$ choices of $(l_1,d_1)$.
		All remaining indices are determined by
		\[
		k_i=k_i'-d_i\quad(i=1,2),\qquad
		l_1'=l_1+v,\qquad l_2'=l_2-v.
		\]
		If $d_1=d_2=0$, \eqref{eq:paired} reduces to
		$v(k_2'-k_1')=0$. The cases $v=0$ and $v\ne0$ give, respectively,
		\[
		O(\Delta^2L^2),\qquad O(\Delta L^3).
		\]
		Consequently
		\[
		\begin{aligned}
			V^{-1}\|T_\vartheta\|_2^2
			&\lesssim_\varepsilon(KL)^\varepsilon
			\bigl(\Delta^2L^2(1+Z/\Delta)+\Delta L^3\bigr)\\
			&=(KL)^\varepsilon\Delta L^2(\Delta+L+Z),
		\end{aligned}
		\]
		which is \eqref{eq:envelope-counts}.

\label{lem:finite-separation}
		Let $\Omega$ and $\mathcal I$ be nonempty finite sets, let
		$c_1,c_2:\Omega\to\mathbb R$, and let $J_{I,1},J_{I,2}\subseteq\mathbb R$
		be intervals for $I\in\mathcal I$.
		There are finite families of coefficients $a_{I,j}$ and functions
		$b_j:\Omega\to\mathbb C$ such that, for every $I\in\mathcal I$
		and $\nu\in\Omega$,
		\begin{equation}\label{eq:finite-separation}
			\begin{gathered}
				\mathbf1_{J_{I,1}}(c_1(\nu))\mathbf1_{J_{I,2}}(c_2(\nu))
				=\sum_j a_{I,j}b_j(\nu),\\
				\sum_j\sup_I|a_{I,j}|\sup_\nu|b_j(\nu)|\lesssim\log^2(2+\#\Omega).
			\end{gathered}
		\end{equation}
		For one interval restriction the bound is $O(\log(2+\#\Omega))$.

\label{lem:elementary-sieve}
		Let $d,n\in\mathbb N$, let $D_i,E_i>0$ for $1\le i\le d$, and let
		$\mathcal I$ be finite. Suppose
		$x_j=(x_{j,1},\ldots,x_{j,d})\in\mathcal X=\prod_{i=1}^d[-D_i,D_i]$
		for $1\le j\le n$, and
		$y_\nu=(y_{\nu,1},\ldots,y_{\nu,d})\in\prod_{i=1}^d[-E_i,E_i]$
		for $\nu\in\mathcal I$.
		For $x\in\mathbb R^d$ and complex coefficients $b=(b_\nu)_{\nu\in\mathcal I}$
		with $|b_\nu|\le1$, put
		\[
		\begin{gathered}
			F_b(x)=\sum_{\nu\in\mathcal I}b_\nu e(x\cdot y_\nu),\qquad
			P=\prod_{i=1}^d(1+D_iE_i),\\
			B=\#\{(j,j')\in\{1,\ldots,n\}^2:
			|x_{j,i}-x_{j',i}|\le E_i^{-1}\ (1\le i\le d)\}.
		\end{gathered}
		\]
		For $q>2$, set $\mathcal A_q=\sup_{|b_\nu|\le1}
		\|F_b\|_{L^q_\#(\mathcal X)}$. For complex coefficients $a_1,\ldots,a_n$
		with $|a_j|\le1$,
		\begin{equation}\label{eq:elementary-sieve}
			\left|\sum_{j=1}^na_jF_b(x_j)\right|
			\lesssim_{d,q}P^{1/q}n^{1-2/q}B^{1/q}\mathcal A_q.
		\end{equation}
		If $c_4,c_6\ge0$ satisfy, for every $\lambda>0$,
		\[
		\sup_{|b_\nu|\le1}
		\frac{|\{x\in\mathcal X:|F_b(x)|>\lambda\}|}{|\mathcal X|}
		\le c_4\lambda^{-4}+c_6\lambda^{-6},
		\]
		then
		\begin{equation}\label{eq:distributional-sieve}
			\left|\sum_{j=1}^na_jF_b(x_j)\right|
			\lesssim_d\sum_{q\in\{4,6\}}P^{1/q}n^{1-2/q}B^{1/q}c_q^{1/q}.
		\end{equation}
		The count $B$ includes all ordered pairs of indices, with their multiplicities.

<!-- proof-11: body_md -->
The moment estimate \eqref{eq:elementary-sieve} is
		\cite[\mbox{\hyperlink{cite.BW18}{Section~5}}, (5.5)]{BW18}, with bounded coefficients as in
		\cite[\mbox{\hyperlink{cite.BW18}{Section~9}}, (9.1)--(9.4)]{BW18}.
		For \eqref{eq:distributional-sieve}, partition $\mathcal X$ into
		axis-parallel boxes $\mathcal Q$ of side lengths
		\[
		\delta_i=\frac{2D_i}{\max\{1,\lceil2D_iE_i\rceil\}},\qquad
		\delta_iE_i\le1,\qquad |\mathcal Q|^{-1}\lesssim_d P/|\mathcal X|.
		\]
		Assign each $x_j$ to one box containing it, and let $n_{\mathcal Q}$ be the
		number of assigned indices. Then
		\[
		\sum_{\mathcal Q} n_{\mathcal Q}=n,\qquad \sum_{\mathcal Q}n_{\mathcal Q}^2\le B.
		\]
		Fix the coefficient sequence $b$.
		For $\mu\in\{0,1\}^d$, define
		\[
		F_{b,\mu}(x)=\sum_{\nu\in\mathcal I}b_\nu
		\prod_{i=1}^d(y_{\nu,i}/E_i)^{\mu_i}e(x\cdot y_\nu).
		\]
		These coefficients have modulus at most one. With
		$\partial^\mu=\partial_{x_1}^{\mu_1}\cdots\partial_{x_d}^{\mu_d}$,
		\[
		\partial^\mu F_b=\left(\prod_{i=1}^d(2\pi\mathrm i E_i)^{\mu_i}\right)F_{b,\mu}.
		\]
		Applying the fundamental theorem of calculus in each coordinate gives,
		for $x_j\in \mathcal Q$,
		\[
		|F_b(x_j)|\le\frac1{|\mathcal Q|}\sum_{\mu\in\{0,1\}^d}
		\prod_{i=1}^d\delta_i^{\mu_i}\int_{\mathcal Q}|\partial^\mu F_b|
		\lesssim_d\frac1{|\mathcal Q|}\sum_{\mu\in\{0,1\}^d}\int_{\mathcal Q}|F_{b,\mu}|.
		\]
		For $\lambda>0$, integration of the distribution function gives
		\[
		\begin{aligned}
			\int_{\mathcal X\cap\{|F_{b,\mu}|>\lambda\}}|F_{b,\mu}|^2
			&=\lambda^2|\mathcal X\cap\{|F_{b,\mu}|>\lambda\}|
			+2\int_\lambda^\infty t|\mathcal X\cap\{|F_{b,\mu}|>t\}|\,dt\\
			&\lesssim|\mathcal X|\bigl(c_4\lambda^{-2}+c_6\lambda^{-4}\bigr).
		\end{aligned}
		\]
		Split each integral over $\mathcal Q$ at $|F_{b,\mu}|=\lambda$ and use
		Cauchy--Schwarz. For every $\lambda>0$,
		\begin{equation}\label{eq:sieve-amplitude-split}
			\begin{split}
				\left|\sum_{j=1}^na_jF_b(x_j)\right|
				&\lesssim_d n\lambda+
				\left(\sum_{\mathcal Q}\frac{n_{\mathcal Q}^2}{|\mathcal Q|}\right)^{1/2}
				\sum_{\mu\in\{0,1\}^d}
				\left(\int_{\mathcal X\cap\{|F_{b,\mu}|>\lambda\}}
				|F_{b,\mu}|^2\right)^{1/2}\\
				&\lesssim_d n\lambda+(PB)^{1/2}
				\big(c_4\lambda^{-2}+c_6\lambda^{-4}\big)^{1/2}.
			\end{split}
		\end{equation}
		If $c_4=c_6=0$, let $\lambda\downarrow0$ in
		\eqref{eq:sieve-amplitude-split}. Otherwise take
		$\lambda=(PBc_4/n^2)^{1/4}+(PBc_6/n^2)^{1/6}>0$.
		For $q\in\{4,6\}$, $PBc_q\lambda^{2-q}\le n^2\lambda^2$,
		which proves \eqref{eq:distributional-sieve}.
