<!-- definition-1: body_md -->
\label{def:fxISS}
System \eqref{eq:dynamics} is said to render the set $\mathcal{A}$ fixed-time input-to-state stable (ISS) if there exists a class $\mathcal{K}\mathcal{L}_{\text{FxT}}$ function $\beta$ and a class $\mathcal{K}$ function $\gamma(\cdot)$, such that for all $x_0\in\mathbb{R}^n$ and all inputs $u\in\ell^{\infty}$, all solutions of \eqref{eq:dynamics} satisfy the following inequality
 
\begin{equation}\label{eq:general_fxtiss_estimate}
    |x(j)|_{\mathcal A}\leq\max\left\{\beta(|x(0)|_{\mathcal A},j),\gamma\left(|u|_{[0,j-1],\infty}\right)\right\},
\end{equation}
 
for all $j\geq0$, where $|u|_{[0,j-1],\infty}=0$ for $j=0$.  \hfill $\square$

<!-- definition-2: body_md -->
A continuous function
$V:\mathbb R^n\to\mathbb R_{\geq0}$ is said to be a \emph{Finite-Horizon FxT-ISS Lyapunov function} for system \eqref{eq:dynamics} if there exist functions $\alpha_1,\alpha_2,\sigma,\gamma
\in\mathcal K_\infty$, a constant $\rho\in[0,1)$, and an integer
$M\geq1$ such that the following
conditions hold:
 
\begin{enumerate}[(a)]
 
\item  For every $x\in\mathbb R^n$,
\begin{equation}
    \alpha_1(|x|_{\mathcal A})
    \leq V(x)
    \leq \alpha_2(|x|_{\mathcal A}).
    \label{eq:fh_comparison}
\end{equation}
 
\item For every $(x,u)\in\mathbb R^n
\times\mathbb R^m$,
\begin{equation}
    V(x^+)-V(x)
    \leq -(1-\rho) V(x)+\sigma(\|u\|).
    \label{eq:fh_dissipation}
\end{equation}
 
\item Let $G_j$ be the $j$-step transition map defined recursively with $G_0(x) := x$ as 
\[
G_{j+1}(x,u_0,\ldots,u_j)
:=
g\!\left(G_j(x,u_0,\ldots,u_{j-1}),u_j\right).
\]
 
Then, the following bound holds
 
\begin{equation}\label{finitehorizonbound}
V\!\left(G_M(x,u_0,\ldots,u_{M-1})\right)
\leq
\gamma\!\left(
\max_{0\leq j\leq M-1}\|u_j\|
\right),
\end{equation}
 
for every $x\in\mathbb{R}^n$ and every
$u_0,\ldots,u_{M-1}\in\mathbb{R}^m$, 
 
\end{enumerate}

<!-- example-1: body_md -->
The class of finite-horizon FxT-ISS Lyapunov functions is strictly broader than the class of extinction-based FxT-SS Lyapunov functions. To see this, we can focus on the case when $u=0$ and consider the discrete-time linear system $x^+=Ax$ with $A=[0,a;0,0]$ (using ``Matlab" notation) with $a\in(0,1)$, with $\mathcal{A}=\{0\}$. Consider the Lyapunov function $V(x)=\|x\|$, which clearly satisfies \eqref{eq:fh_comparison}. Since $\|A\|=a$, it follows that $ V(x^{+})\leq aV(x)$, and therefore \eqref{eq:fh_dissipation} holds with zero input gain. Moreover, $A^2=0$, and hence every solution satisfies $ x(j+2)=0$ for every $j\in\mathbb{Z}_{\geq0}$. Therefore, $V(x(j+2))-V(x(j)) =-V(x(j))$, which implies that \eqref{finitehorizonbound} holds with $M=2$. Therefore, $V$ is a finite-horizon FxT (ISS) Lyapunov function. Next, we show that $V$ cannot be an extinction-based FxT Lyapunov
function. Suppose, by contradiction, that there exist
$N\in\mathbb{Z}_{>0}$ and
$\psi\in\mathcal{E}_{\mathrm{FxT}}^{N}$ such that \eqref{eq:additive_dissipation} holds with $\rho=0$. For any $s>0$, let $x=(0,s)$. Then, $V(x)=s$, $x^{+}=[as;0]$, and  $V(x^{+})=as$. Substituting these expressions into \eqref{eq:additive_dissipation} gives $-(1-a)s \leq-\min\{\psi(s),s\}$, or, equivalently, $\min\{\psi(s),s\}\leq(1-a)s$. Since $a\in(0,1)$, we have $(1-a)s<s$ for every $s>0$. Thus, \eqref{eq:additive_dissipation} necessarily implies $\psi(s)\leq(1-a)s$, for all $s>0$. Let $g(s):=[s-\psi(s)]_{+}$. It follows that $g(s)\geq as$ for all $s>0$. Therefore, its monotone envelope $h(s):=\max_{0\leq r\leq s}g(r)$ satisfies $h(s)\geq as$ for all $s>0$. In particular, $h$ is unbounded. This contradicts
$\psi\in\mathcal{E}_{\mathrm{FxT}}^{N}$, since the fixed-time
extinction property requires $h^{N}(s)=0$, for all $s\geq0$, which, in particular, implies that $h$ is globally bounded. Hence, there is no $\psi\in\mathcal{E}_{\mathrm{FxT}}^{N}$, for any finite $N$, such that \eqref{eq:additive_dissipation} holds. \hfill $\square$

\end{enumerate}

<!-- example-2: body_md -->
Consider the discrete-time linear system
 
\begin{equation}
\label{eq:deadbeat_system}
x^+=Ax+Bu,
\end{equation}
 
where $x\in\mathbb R^n$, $u\in\mathbb R^m$, and suppose that $A$ is nilpotent, i.e., there exists $M\in\mathbb Z_{\geq1}$ such that $A^M=0$. Such dynamics arise in deadbeat control. Indeed, for a controllable system
\[
z^+=A_0z+B_0v,
\]
a state-feedback law $v=Kz+u$ can be selected so that
$A:=A_0+B_0K$ has all its eigenvalues at the origin and is therefore
nilpotent. Let $\mathcal A=\{0\}$ and fix any $\rho\in(0,1)$. Define $V(x):=\sum_{\ell=0}^{M-1}\rho^{-\ell}\|A^\ell x\|$. We show that $V$ is a finite-horizon Lyapunov function for FxT-ISS. First, since the term corresponding to $\ell=0$ is $\|x\|$, then $V(x)\geq\|x\|$. Moreover, $V(x)\leq\left(\sum_{\ell=0}^{M-1}\rho^{-\ell}\|A^\ell\|\right)\|x\|$. Defining $c_A:=\sum_{\ell=0}^{M-1}
\rho^{-\ell}\|A^\ell\|$, the comparison bounds \eqref{eq:fh_comparison} hold with $\alpha_1(s)=s$ and $\alpha_2(s)=c_As$. Next, using the fact that $A$ is nilpotent
 
\begin{align*}
V(Ax)&=
\sum_{\ell=0}^{M-1}
\rho^{-\ell}\|A^{\ell+1}x\|=
\rho
\sum_{\ell=1}^{M}
\rho^{-\ell}\|A^\ell x\|
\\
&=
\rho
\sum_{\ell=1}^{M-1}
\rho^{-\ell}\|A^\ell x\|\leq
\rho V(x).
\end{align*}
 
Using \eqref{eq:deadbeat_system} and the triangle
inequality gives
 
\begin{align*}
V(x^+)
&\leq
V(Ax)+V(Bu)\leq
\rho V(x)
+
\sum_{\ell=0}^{M-1}
\rho^{-\ell}\|A^\ell Bu\|
\\
&\leq
\rho V(x)
+
\left(
\sum_{\ell=0}^{M-1}
\rho^{-\ell}\|A^\ell B\|
\right)\|u\|.
\end{align*}
 
Hence, defining $c_B:=\sum_{\ell=0}^{M-1}\rho^{-\ell}\|A^\ell B\|$, we obtain that \eqref{eq:fh_dissipation} holds:
 
\begin{equation}
\label{eq:deadbeat_iss_dissipation}
V(x^+)
\leq
\rho V(x)+\sigma(\|u\|),
\qquad
\sigma(r):=(1+c_B)r.
\end{equation}
 
Finally, to prove \eqref{finitehorizonbound}, we iterate \eqref{eq:deadbeat_system} for $M$ steps, obtaining
 
\begin{equation}
x(M)=A^Mx(0)+\sum_{j=0}^{M-1}A^{M-1-j}Bu(j).
\end{equation}
 
Since $A^M=0$, it follows that $x(M)
= \sum_{j=0}^{M-1}A^{M-1-j}Bu(j).$
 
                 
                            
     
  
                 
                
               
Consequently,
\begin{align*}
V(x(M))
&\leq
\sum_{j=0}^{M-1}
V\!\left(A^{M-1-j}Bu(j)\right)
\\
&\leq
\left(
\sum_{j=0}^{M-1}
\sum_{\ell=0}^{M-1}
\rho^{-\ell}
\|A^{\ell+M-1-j}B\|
\right)
\|u\|_{[0,M-1],\infty}.
\end{align*}
 
Using $c_M:=
\sum_{j=0}^{M-1}
\sum_{\ell=0}^{M-1}
\rho^{-\ell}
\|A^{\ell+M-1-j}B\|$, we obtain $V(x(M))\leq c_M\|u\|_{[0,M-1],\infty}$. Since the same argument applies from any initial time, we obtain:
 
\begin{equation}
V\!\left(
G_M(x,u_0,\ldots,u_{M-1})
\right)
\leq
\gamma\!\left(
\max_{0\leq j\leq M-1}\|u_j\|
\right),
\end{equation}
 
with $\gamma(r):=(1+c_M)r$.
These choices ensure $\sigma,\gamma\in\mathcal K_\infty$
even when $B=0$.
Hence, $V$ is a finite-horizon Lyapunov function.
By Theorem~\ref{theoremLyapu}, the set $\mathcal A$
is FxT-ISS. \hfill\ensuremath{\square}

Consider a strongly connected directed graph
$\mathcal G=(\mathcal V,\mathcal E)$, with
$\mathcal V=\{1,\ldots,N\}$ and $N\geq2$.
Let $D:=\max_{i,j\in\mathcal V}d(i,j)\geq1$
denote its directed diameter, where $d(i,j)$ is
the length of the shortest directed path from
node $i$ to node $j$. Each agent $i$ has a scalar state $x_i\in\mathbb R$ and evolves according to the dynamics:
 
\begin{equation}
\label{eq:min_consensus}
x_i^+
=
\min_{j\in\mathcal N_i\cup\{i\}} x_j+u_i,
\qquad i\in\mathcal V,
\end{equation}
where $u_i\in\mathbb R$ is an additive disturbance and $\mathcal{N}_i$ denotes the set of neighbors of agent \(i\). Define the nominal min-consensus map $T:\mathbb R^N\to\mathbb R^N$ componentwise as $T_i(x):=\min_{j\in\mathcal N_i\cup\{i\}}x_j$. Then \eqref{eq:min_consensus} can be written compactly as:
 
\begin{equation}\label{overallsystem}
x^+=T(x)+u.
\end{equation}
 
Let $\mathcal A:=\{c\mathbf 1:c\in\mathbb R\}$ be the consensus set, and define the disagreement measure $W(x):=\max_i x_i-\min_i x_i$. Since $\mathcal G$ is strongly connected and has directed diameter $D$, every node receives information originating from every other node after at most $D$ iterations. Hence,
 
\begin{equation}
\label{eq:min_consensus_D}
T^D(x)
=
\left(\min_i x_i\right)\mathbf 1,
\end{equation}
 
and therefore $W(T^D(x))=0$, for all $x\in\mathbb R^N$.  Fix any $\rho\in(0,1)$ and let
 
\begin{equation}
\label{eq:min_consensus_V}
V(x)
:=
\sum_{\ell=0}^{D-1}
\rho^{-\ell}W(T^\ell(x)).
\end{equation}
 
We show that $V$ is a finite-horizon Lyapunov function for system \eqref{overallsystem}, with horizon $M=D$. To see this, first note that
$W(T(x))\leq W(x)$ and therefore $W(T^\ell(x))\leq W(x)$ for all
$\ell\geq0$. Therefore, $W(x)\leq V(x)\leq c_DW(x),$
 
                 
                          
     
          
              
               
where $c_D:=\sum_{\ell=0}^{D-1}\rho^{-\ell}$. Since $|x|_{\mathcal A}=\min_{c\in\mathbb R}\|x-c\mathbf 1\|_2$, we have $W(x)\leq\sqrt{2}\,|x|_{\mathcal A}$. Moreover, if $\bar x:=\frac{1}{N}\mathbf 1^\top x$, then $|x|_{\mathcal A}=
\|x-\bar x\mathbf 1\|_2
\leq
\sqrt{N}\,W(x)$, so that $\frac{1}{\sqrt N}|x|_{\mathcal A}
\leq
W(x)
\leq
\sqrt{2}|x|_{\mathcal A}$. Combining the above inequalities gives $\frac{1}{\sqrt N}|x|_{\mathcal A}
\leq
V(x)
\leq
\sqrt{2}c_D|x|_{\mathcal A}$. Thus, $V$ satisfies \eqref{eq:fh_comparison} with $\alpha_1(s)=\frac{s}{\sqrt N}$ and $\alpha_2(s)=\sqrt{2}c_Ds$. Next, using \eqref{eq:min_consensus_D}, we get
 
\begin{align*}
V(T(x))&=
\sum_{\ell=0}^{D-1}
\rho^{-\ell}W(T^{\ell+1}(x))=
\rho
\sum_{\ell=1}^{D}
\rho^{-\ell}W(T^\ell(x))
\\
&=
\rho
\sum_{\ell=1}^{D-1}
\rho^{-\ell}W(T^\ell(x))\leq
\rho V(x).
\end{align*}
 
Since the min-consensus map is nonexpansive in the infinity norm, we have $\|T(x)-T(y)\|_\infty\leq\|x-y\|_\infty$. Also, we have that $|W(x)-W(y)|\leq2\|x-y\|_\infty$. Using the nonexpansiveness of $T$, it follows that, for every $\ell\geq0$,
 
\begin{align*}
W(T^\ell(T(x)+u))
&\leq
W(T^\ell(T(x)))
+
2\|T^\ell(T(x)+u)\\
&~~~~-T^\ell(T(x))\|_\infty
\\
&\leq
W(T^{\ell+1}(x))+2\|u\|_\infty.
\end{align*}
Consequently,
\begin{align*}
V(T(x)+u)
&=
\sum_{\ell=0}^{D-1}
\rho^{-\ell}W(T^\ell(T(x)+u))
\\
&\leq
\sum_{\ell=0}^{D-1}
\rho^{-\ell}W(T^{\ell+1}(x))
+
2\sum_{\ell=0}^{D-1}\rho^{-\ell}\|u\|_\infty
\\
&=
V(T(x))+2c_D\|u\|_\infty\leq
\rho V(x)+2c_D\|u\|_\infty.
\end{align*}
 
Thus, inequality \eqref{eq:fh_dissipation} holds with $\sigma(r):=2c_Dr$. Finally, we establish condition \eqref{finitehorizonbound}. Let $\bar x(0)=x(0)$, $x$ be a solution to \eqref{overallsystem}, and let $\bar x$ be a solution to \eqref{overallsystem} with $u=0$. By nonexpansiveness of $T$, we get
 
\begin{align*}
\|x(j+1)-\bar x(j+1)\|_\infty
&\leq
\|T(x(j))-T(\bar x(j))\|_\infty+\|u(j)\|_\infty
\\
&\leq
\|x(j)-\bar x(j)\|_\infty+\|u(j)\|_\infty.
\end{align*}
 
Iterating this inequality gives
\[
\|x(D)-\bar x(D)\|_\infty
\leq
\sum_{j=0}^{D-1}\|u(j)\|_\infty
\leq
D\|u\|_{[0,D-1],\infty}.
\]
By \eqref{eq:min_consensus_D}, we have $\bar x(D)
=\left(\min_i x_i(0)\right)\mathbf 1\in\mathcal A$. Since $W(\bar x(D))=0$, we get
 
\begin{align*}
W(x(D))&=|W(x(D))-W(\bar x(D))|\leq
2\|x(D)-\bar x(D)\|_\infty\\
&\leq
2D\|u\|_{[0,D-1],\infty}.
\end{align*}
 
Using the bounds on $W$, we get $V(x(D))\leq c_DW(x(D)) \leq 2Dc_D\|u\|_{[0,D-1],\infty}$. Since the same argument applies from any initial time $k$, we obtain
\[
V\!\left(
G_D(x,u_0,\ldots,u_{D-1})
\right)
\leq
\gamma\!\left(
\max_{0\leq j\leq D-1}\|u_j\|_\infty
\right),
\]
where $\gamma(r):=2Dc_Dr$. Hence, $V$ satisfies all the conditions of a finite-horizon FxT-ISS Lyapunov function. By Theorem \ref{theoremLyapu}, the set $\mathcal{A}$ is FxT-ISS. \hfill\ensuremath{\square}
