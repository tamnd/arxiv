<!-- assumption-1: body_md -->
\label{asm:MDPG}
  We assume that there exist positive constants $c_B$, $c_U$, $c_V$, and
  $c_{\rm Lip}$ such that for all ${\mathcal{{z}}}, {\mathcal{{z}}}_1, {\mathcal{{z}}}_2 \in {\mathcal{{U}}_\alpha}$,
  the operators $\mathcal{{B}}_\alpha$ and $\mathcal{{C}}_\alpha$ of the
  reformulation~\eqref{eq:A-split-BandC} satisfy
  \begin{align}
    \label{eq:S4-B-surjective}
    &\operatorname*{range}(\mathcal{{B}}_\alpha) = {\mathcal{{V}}'_\alpha}, \text{ i.e., $\mathcal{{B}}_\alpha$ is surjective,}
    \\
    \label{eq:S4-infsup}
    &\| {\mathcal{{z}}} - P_\alpha {\mathcal{{z}}} \|_{{\mathcal{{U}}_\alpha}}^2 \le c_B \| \mathcal{{B}}_\alpha {\mathcal{{z}}} \|_{{\mathcal{{V}}'_\alpha}}^2,
    \\
    \label{eq:S4-mono}
    &c_U \| P_\alpha({\mathcal{{z}}}_1 - {\mathcal{{z}}}_2) \|_{{\mathcal{{U}}_\alpha}}^2
    \le \langle \mathcal{{C}}_\alpha({\mathcal{{z}}}_1) - \mathcal{{C}}_\alpha({\mathcal{{z}}}_2), {\mathcal{{z}}}_1 - {\mathcal{{z}}}_2 \rangle
    + c_V \| \mathcal{{B}}_\alpha({\mathcal{{z}}}_1 - {\mathcal{{z}}}_2) \|_{{\mathcal{{V}}'_\alpha}}^2,
    \\
    \label{eq:C-zero}
    &\langle \mathcal{{C}}_\alpha({\mathcal{{z}}}), {\mathcal{{z}}}_0 \rangle = 0  \text{  for all }
    {\mathcal{{z}}}_0 \in \ker(\mathcal{{B}}_\alpha) \implies
    \mathcal{{C}}_\alpha({\mathcal{{z}}}) = 0 \text{ in } {\mathcal{{U}}'_\alpha},      \text { and } 
    \\
    \label{eq:S4-Lip}
    &\| \mathcal{{C}}_\alpha({\mathcal{{z}}}_1) - \mathcal{{C}}_\alpha({\mathcal{{z}}}_2) \|_{{\mathcal{{U}}'_\alpha}}
    \le c_{\rm Lip} \| {\mathcal{{z}}}_1 - {\mathcal{{z}}}_2 \|_{{\mathcal{{U}}_\alpha}}.
  \end{align}

<!-- assumption-3: body_md -->
\label{asm:N}
  We assume that $\mathcal{{M}}_\alpha({\cdot}) : {\mathcal{{U}}_\alpha} \to [0, \infty)$ is a functional
  satisfying, for some constant $c_M > 0$,
  \begin{align}
    \label{eq:N-pairing}
    \langle{ \mathcal{{C}}_\alpha({\mathcal{{z}}}), {\mathcal{{w}}} }\rangle  \le\;
    & \mathcal{{M}}_\alpha({{\mathcal{{z}}}}) \, \| {\mathcal{{w}}} \|_{{\mathcal{{U}}_\alpha}},
    \\
    \label{eq:N-lip}
    & \mathcal{{M}}_\alpha({{\mathcal{{z}}}})   \le c_M \| {\mathcal{{z}}} - \bar{{\mathcal{{z}}}} \|_{{\mathcal{{U}}_\alpha}}

      \mathcal{{C}}_\alpha(\bar{\mathcal{{z}}}) = 0,
  \end{align}
  for all ${\mathcal{{z}}}, {\mathcal{{w}}} \in {\mathcal{{U}}_\alpha}$.

<!-- definition-1: body_md -->
\label{defn:mdpg}
  Given a closed subspace $\mathcal{{V}}_{\alpha, h} \subseteq {\mathcal{{V}}_\alpha}$ and a functional
  $\mathcal{{M}}_\alpha({\cdot})$ satisfying Assumption~\ref{asm:N}, the generated

    \mathcal{{L}}(\alpha, {\mathcal{{z}}}_h)

    + \mathcal{{M}}_\alpha({{\mathcal{{z}}}_h})^2,
    \qquad {{\mathcal{{z}}}_h} \in {\mathcal{{U}}_{\alpha, h}}, \; \alpha \in \mathcal{{Q}}.
  \end{equation}

<!-- theorem-1: body_md -->
\label{thm:gen-nonlinear-dpg}
  Suppose Assumptions~\ref{asm:MDPG}, \ref{asm:fortin-b},
  and~\ref{asm:N} hold and
  $\kappa$ satisfies~\eqref{eq:S4-kappa}. Then \eqref{eq:mixedC&H} has
  a unique solution whose ${\mathcal{{u}}}$-component solves $\mathcal{{B}}_\alpha {\mathcal{{u}}} = \ell$
  and $\mathcal{{C}}_\alpha({\mathcal{{u}}}) = 0$.  Moreover, the loss function $\mathcal{{L}}$
  of Definition~\ref{defn:mdpg} is
  variationally correct, i.e., there are positive constants $c$ and
  $C$ such that for any $\alpha \in \mathcal{{Q}}$ and ${{\mathcal{{w}}}_h} \in {\mathcal{{U}}_{\alpha, h}}$,
  \begin{align}
    \label{eq:reliability-L}
    c & \| {\mathcal{{u}}}  - {\mathcal{{w}}}_h \|_{\mathcal{{U}}_\alpha}^2  \le
    \mathcal{{L}}(\alpha,{\mathcal{{w}}}_h)\,
          \| \varPi \|_{L({\mathcal{{V}}_\alpha})}^2
          + \operatorname{{\ensuremath{\text{osc}}}}(\ell)^2,
    \\
    \label{eq:efficiency-L}
    \mathcal{{L}}(\alpha,{\mathcal{{w}}}_h) \le C & \| {\mathcal{{u}}} - {\mathcal{{w}}}_h \|_{\mathcal{{U}}_\alpha}^2.
  \end{align}
  We may choose $C = \kappa \| \mathcal{{B}}_\alpha \|_{L({\mathcal{{U}}_\alpha}, {\mathcal{{V}}'_\alpha})}^2 + c_M^2$,
  with $c_M$ from \eqref{eq:N-lip}, and $c$, defined
  in~\eqref{eq:8} below, is determined by the $c_U$ from~\eqref{eq:S4-mono}.

<!-- proof-1: body_md -->
Conditions~\eqref{eq:S4-B-surjective} and~\eqref{eq:C-zero} of
  Assumption~\ref{asm:MDPG} imply that \eqref{eq:mixedC&H} has a
  unique solution and component ${\mathcal{{u}}}$ of the unique solution
  $({\mathcal{{v}}}, {\mathcal{{u}}}) \in {\mathcal{{V}}_\alpha} \times {\mathcal{{U}}_\alpha}$ of~\eqref{eq:mixedC&H}
  satisfies~\eqref{eq:A-split-BandC} (see~\cite[Theorem~2.3 and
  Proposition~2.4]{CH}). We now focus on proving the new 
  estimates~\eqref{eq:reliability-L} and~\eqref{eq:efficiency-L}.


  {\em Step 1.}  Let us first prove the reliability
  estimate~\eqref{eq:reliability-L}. Since $\mathcal{{B}}_\alpha {\mathcal{{u}}} = \ell$ and
  $\mathcal{{C}}_\alpha({\mathcal{{u}}}) = 0$, applying~\eqref{eq:S4-key} with ${\mathcal{{z}}}_1 = {\mathcal{{u}}}$ and
  ${\mathcal{{z}}}_2 = {{\mathcal{{w}}}_h}$,
  \begin{equation}
    \label{eq:rel-basic}
    c_U \| {\mathcal{{u}}} - {{\mathcal{{w}}}_h} \|_{{\mathcal{{U}}_\alpha}}^2
    \le \langle \mathcal{{C}}_\alpha({\mathcal{{u}}}) - \mathcal{{C}}_\alpha({{\mathcal{{w}}}_h}), {\mathcal{{u}}} - {{\mathcal{{w}}}_h} \rangle
    + \kappa \| \mathcal{{B}}_\alpha({\mathcal{{u}}} - {{\mathcal{{w}}}_h}) \|_{{\mathcal{{V}}'_\alpha}}^2.
  \end{equation}
  We proceed to bound the two terms on the right hand side, beginning
  with the last. Since $\mathcal{{B}}_\alpha {\mathcal{{u}}} = \ell$, for any ${\mathcal{{v}}} \in {\mathcal{{V}}_\alpha}$,
  \begin{align*}
    \big( \mathcal{{B}}_\alpha( {\mathcal{{u}}} - {{\mathcal{{w}}}_h}) \big)({\mathcal{{v}}})
    & = \big( \ell - \mathcal{{B}}_\alpha {{\mathcal{{w}}}_h} \big)({\mathcal{{v}}})
     = \ell({\mathcal{{v}}}) - b_\alpha({{\mathcal{{w}}}_h}, {\mathcal{{v}}})
    \\
    & = \ell({\mathcal{{v}}} - \varPi {\mathcal{{v}}}) - b_\alpha({{\mathcal{{w}}}_h}, {\mathcal{{v}}} - \varPi {\mathcal{{v}}})
      + \ell(\varPi {\mathcal{{v}}}) - b_\alpha({{\mathcal{{w}}}_h}, \varPi {\mathcal{{v}}})
    \\
    & = \ell({\mathcal{{v}}} - \varPi {\mathcal{{v}}}) + \big( \ell - \mathcal{{B}}_\alpha {{\mathcal{{w}}}_h} \big)( \varPi {\mathcal{{v}}})
    \\
    & \le \big( \operatorname{{\ensuremath{\text{osc}}}}(\ell)
      + \| \ell - \mathcal{{B}}_\alpha {{\mathcal{{w}}}_h} \|_{(\mathcal{{V}}_{\alpha, h})'} \| \varPi \|_{L({\mathcal{{V}}_\alpha})}  \big)
      \| {\mathcal{{v}}} \|_{{\mathcal{{V}}_\alpha}},
  \end{align*}
  where we have used~\eqref{eq:Fortin-b} in the third step and
  $\varPi {\mathcal{{v}}} \in \mathcal{{V}}_{\alpha, h}$ in the last.  Taking the supremum
  over $0 \ne {\mathcal{{v}}} \in {\mathcal{{V}}_\alpha}$ and using $(a+b)^2 \le 2 a^2 + 2 b^2$, yields
  \begin{equation}
    \label{eq:rel-bound-B}
    \kappa \| \mathcal{{B}}_\alpha({\mathcal{{u}}} - {{\mathcal{{w}}}_h}) \|_{{\mathcal{{V}}'_\alpha}}^2
    \le 2 \kappa \operatorname{{\ensuremath{\text{osc}}}}(\ell)^2
    + 2 \kappa \| \varPi \|_{L({\mathcal{{V}}_\alpha})}^2 \| \ell - \mathcal{{B}}_\alpha {{\mathcal{{w}}}_h} \|_{(\mathcal{{V}}_{\alpha, h})'}^2.
  \end{equation}

  {\em Step 2.}  Next, to bound the term with the nonlinear operator,
  since $\mathcal{{C}}_\alpha({\mathcal{{u}}}) = 0$, by~\eqref{eq:N-pairing},
  \[
    | \langle \mathcal{{C}}_\alpha({\mathcal{{u}}}) - \mathcal{{C}}_\alpha({{\mathcal{{w}}}_h}), {\mathcal{{u}}} - {{\mathcal{{w}}}_h} \rangle |
    = | \langle \mathcal{{C}}_\alpha({{\mathcal{{w}}}_h}), {\mathcal{{u}}} - {{\mathcal{{w}}}_h} \rangle |
    \le \mathcal{{M}}_\alpha({{{\mathcal{{w}}}_h}}) \, \| {\mathcal{{u}}} - {{\mathcal{{w}}}_h} \|_{{\mathcal{{U}}_\alpha}}.
  \]
  Applying Young's inequality,
  \begin{equation}
    \label{eq:rel-bound-C}
    | \langle \mathcal{{C}}_\alpha({\mathcal{{u}}}) - \mathcal{{C}}_\alpha({{\mathcal{{w}}}_h}), {\mathcal{{u}}} - {{\mathcal{{w}}}_h} \rangle |
    \le \frac{1}{2 c_U} \mathcal{{M}}_\alpha({{{\mathcal{{w}}}_h}})^2
    + \frac{c_U}{2} \| {\mathcal{{u}}} - {{\mathcal{{w}}}_h} \|_{{\mathcal{{U}}_\alpha}}^2.
  \end{equation}
  Substituting~\eqref{eq:rel-bound-B}
  and~\eqref{eq:rel-bound-C} into~\eqref{eq:rel-basic}, and absorbing
  the last term of~\eqref{eq:rel-bound-C} into the left hand side,
  \begin{equation}
    \label{eq:7}
    \begin{aligned}
    \frac{c_U}{2} \| {\mathcal{{u}}} - {{\mathcal{{w}}}_h} \|_{{\mathcal{{U}}_\alpha}}^2
   & \le \frac{1}{2 c_U} \mathcal{{M}}_\alpha({{{\mathcal{{w}}}_h}})^2
    + 2 \kappa \| \varPi \|_{L({\mathcal{{V}}_\alpha})}^2 \| \ell - \mathcal{{B}}_\alpha {{\mathcal{{w}}}_h} \|_{(\mathcal{{V}}_{\alpha, h})'}^2
    + 2 \kappa \operatorname{{\ensuremath{\text{osc}}}}(\ell)^2\\
    & \le\max\left(2, \frac{1}{2c_U\|\varPi\|^2_{L({\mathcal{{V}}_\alpha})}}\right)\|\varPi\|^2_{L({\mathcal{{V}}_\alpha})}
    \mathcal{{L}}(\alpha,{\mathcal{{w}}}_h) +2 \kappa \operatorname{{\ensuremath{\text{osc}}}}(\ell)^2.
    \end{aligned}
  \end{equation}
  Thus, defining 
  \begin{equation}
    \label{eq:8}
    \gamma =
    \max\left( 2, \; \frac{1}{2 c_U \| \varPi \|_{L({\mathcal{{V}}_\alpha})}^2}  \right),
    \qquad
    c = \frac{c_U }{2 \max(\gamma, 2 \kappa)}, 
  \end{equation}
  the reliability estimate~\eqref{eq:reliability-L} holds with the $c$
  in~\eqref{eq:8}.

  {\em Step 3.}  To prove the efficiency
  estimate~\eqref{eq:efficiency-L}, note that since
  $\mathcal{{V}}_{\alpha, h} \subseteq {\mathcal{{V}}_\alpha}$ and $\mathcal{{B}}_\alpha {\mathcal{{u}}} = \ell$,
  \[
    \| \ell - \mathcal{{B}}_\alpha {{\mathcal{{w}}}_h} \|_{(\mathcal{{V}}_{\alpha, h})'}
    \le \| \ell - \mathcal{{B}}_\alpha {{\mathcal{{w}}}_h} \|_{{\mathcal{{V}}'_\alpha}}
    = \| \mathcal{{B}}_\alpha({\mathcal{{u}}} - {{\mathcal{{w}}}_h}) \|_{{\mathcal{{V}}'_\alpha}}
    \le \| \mathcal{{B}}_\alpha \|_{L({\mathcal{{U}}_\alpha}, {\mathcal{{V}}'_\alpha})} \| {\mathcal{{u}}} - {{\mathcal{{w}}}_h} \|_{{\mathcal{{U}}_\alpha}},
  \]
  while $\mathcal{{C}}_\alpha({\mathcal{{u}}}) = 0$ and~\eqref{eq:N-lip} give
  \[
    \mathcal{{M}}_\alpha({{{\mathcal{{w}}}_h}})
    \le c_M \| {\mathcal{{u}}} - {{\mathcal{{w}}}_h} \|_{{\mathcal{{U}}_\alpha}}.
  \]
  Using these two bounds in the definition~\eqref{eq:C&H-lossh} of
  $\mathcal{{L}}$ proves the efficiency estimate~\eqref{eq:efficiency-L}
  with $C = \kappa \| \mathcal{{B}}_\alpha \|_{L({\mathcal{{U}}_\alpha}, {\mathcal{{V}}'_\alpha})}^2 + c_M^2$.

\label{rem:quasioptimality}
  As is well known (see, e.g., \cite[Remark~2.5]{CarstDemkoGopal14})
  {\it a posteriori} efficiency and reliability estimates like that of
  Theorem~\ref{thm:gen-nonlinear-dpg} imply {\it a priori}
  quasi-optimal error estimates for any loss minimizer
  ${\mathcal{{u}}}_h \in \arg \min_{{\mathcal{{w}}} \in {\mathcal{{U}}_{\alpha, h}}}\mathcal{{L}}(\alpha, {\mathcal{{w}}})$ when
  $\operatorname{{\ensuremath{\text{osc}}}}(\ell)$ vanishes or is comparable to the best approximation error.
  The argument is
  \begin{align}
    \nonumber
    \| {\mathcal{{u}}} - {\mathcal{{u}}}_h \|_{{\mathcal{{U}}_\alpha}}^2
    &\le \frac 1 c
      \mathcal{{L}}(\alpha, {\mathcal{{u}}}_h) \| \varPi \|_{L({\mathcal{{V}}_\alpha})}^2 +
      \frac 1 c\operatorname{{\ensuremath{\text{osc}}}}(\ell)^2 
    \le \frac 1 c \mathcal{{L}}(\alpha, {\mathcal{{w}}}_h) \| \varPi \|_{L({\mathcal{{V}}_\alpha})}^2 +
      \frac 1 c \operatorname{{\ensuremath{\text{osc}}}}(\ell)^2 \\
    \label{eq:quasioptimality}
    &\le \left(\frac C c \| \varPi \|_{L({\mathcal{{V}}_\alpha})}^2 \right)
      \| {\mathcal{{u}}} - {\mathcal{{w}}}_h \|_{{\mathcal{{U}}_\alpha}}^2 
      + \frac 1 c \operatorname{{\ensuremath{\text{osc}}}}(\ell)^2,
  \end{align}
  for every ${\mathcal{{w}}}_h \in {\mathcal{{U}}_{\alpha, h}}$, using
  reliability~\eqref{eq:reliability-L}, minimality, and
  efficiency~\eqref{eq:efficiency-L} in succession.

\label{rem:mdpg-rho-formulation}
  The MDPG formulation above uses the auxiliary variable $p$ and enforces
  the closure relation $q = \sigma_\alpha(p)$. An alternative formulation
  replaces $p$ with an auxiliary variable $z \in L_2({\varOmega_h})^d$ and enforces
  the inverse closure relation $z = \rho_\alpha(q)$, where
  $\rho_\alpha = \sigma_\alpha^{-1}$. The trial space becomes
  ${\mathcal{{U}}_\alpha} = L_2({\varOmega_h})^d \times L_2({\varOmega_h}) \times L_2({\varOmega_h})^d
  \times \mathring{H}^{1/2}(\partial{\varOmega_h})  \times H^{-1/2}(\partial{\varOmega_h})$ with variables
  $(q, u, z, \hat u, \hat q_n)$, and the closure operator is
  \[
    c_\alpha({\mathcal{{u}}}; \tilde{\mathcal{{u}}})
    := \kappa_1^{-2}(z - \rho_\alpha(q), \tilde z)_{L_2}
    + \kappa_1^{-2}\kappa_2(z - \rho_\alpha(q), \tilde q)_{L_2}.
  \]
  The bilinear form $b_\alpha$ remains unchanged. Both formulations
  share the same discrete subspaces~\eqref{eq:mdpg-discrete} and yield
  identical variational correctness constants (see
  Remark~\ref{rem:mdpg-rho-verification}).

\label{ex:canonical}
  Set $\phi(t) = 2 + (1 + t)^{-2}$ for $t\ge 0$.  Then, since
  $\phi'(t) = -2(1+t)^{-3}$, we have $2 < \phi(t) \le 3$ and

  in~\cite{CH}.  The numerical experiments of~\cite{CH} use the
  variant $\phi(t) = 2 - (1+t)^{-2}$ for which a similar analysis

\label{ex:invertible}
  Set $\phi(t) = 2 + (1+t)^{-1}$ for $t \ge 0$. Then $\phi'(t) = -(1+t)^{-2}$,
  so $2 < \phi(t) \le 3$ and
  \[
    \phi(t) + t\phi'(t)
    = 2 + \frac{1}{1+t} - \frac{t}{(1+t)^2}
    = 2 + \frac{1+t-t}{(1+t)^2}
    = 2 + (1+t)^{-2}.
  \]
  Hence $2 < \phi(t) + t\phi'(t) \le 3$, and since $t\phi'(t) \to 0$
  when $t \to 0^+$, \eqref{eq:phi-props} holds with
  $\beta_1 = 2$ and $\beta_2 = 3$.

  This profile has the additional property that $\sigma_\alpha$ is explicitly
  invertible. To see this, observe that the monotone function
  $m(t) := t\phi(t) = t\left(2 + \frac{1}{1+t}\right) = 2t + \frac{t}{1+t}$
  can be rewritten as
  \[
    m(t) = 2t + 1 - \frac{1}{1+t},
  \]
  from which $m(t) = s$ gives $(1+t)^{-1} = 2t + 1 - s$, or equivalently
  \[
    2t^2 + (3-s)t - s = 0.
  \]
  Solving this quadratic equation yields
  \[
    m^{-1}(s)
    = \frac{s-3+\sqrt{(3-s)^2+8s}}{4}
    = \frac{s-3+\sqrt{s^2+2s+9}}{4}
  \]
  for $s \ge 0$ (taking the positive root since $t \ge 0$). One can verify
  that $m(m^{-1}(s)) = s$ by direct substitution. To summarize, 
  if $p = \sigma_\alpha(q)$ then $|p| = \alpha^{-1}m(|q|)$,
  so $|q| = m^{-1}(\alpha|p|)$ and the direction reversal gives  the
  explicit inverse
  \begin{equation}
    \label{eq:rho-explicit}
    \rho_\alpha(p) = \sigma_\alpha^{-1}(p)
    = -\frac{m^{-1}(\alpha|p|)}{|p|}\, p
    = -\frac{\alpha|p| - 3 + \sqrt{\alpha^2|p|^2 + 2\alpha|p| + 9}}{4|p|}\, p.
  \end{equation}
  The explicit inverse enables the alternative MDPG formulation of
  Remark~\ref{rem:mdpg-rho-formulation} for this profile.

<!-- lemma-1: body_md -->
\label{lem:phi}
  In the setting of~\eqref{eq:phi-asm}, the nonlinear flux function
  $\sigma_\alpha$ of~\eqref{eq:5} which maps
  $q \mapsto -\phi_\alpha(|q|)q$ is $C^1$ on $\mathbb{{R}}^d$, its Jacobian matrix 
  at any $0 \ne q \in \mathbb{{R}}^d$ takes the form 
  \begin{equation}
    \label{eq:Dsigma}
    d\sigma_\alpha(q)
    = -\alpha^{-1} \left( \phi(|q|)\, I
      + \phi'(|q|)\, |q| \; \frac{q}{|q|} \otimes \frac{q}{|q|}\right),
  \end{equation}
  which is extendable continuously to all $q \in \mathbb{{R}}^d$  including $q = 0$ and 
  has its spectrum
  contained in the real interval  $[-\alpha^{-1}\beta_2, -\alpha^{-1}\beta_1]$ for every
  $q \in \mathbb{{R}}^d$.
  Moreover, for all $q, r \in \mathbb{{R}}^d$,
  \begin{equation}
    \label{eq:sigma-mono-lip}
    \begin{gathered}
      \alpha^{-1}\beta_1 |q - r|^2
      \;\le\; -\big(\sigma_\alpha(q) - \sigma_\alpha(r)\big)\cdot(q-r)
      \;\le\;\alpha^{-1}\beta_2 |q-r|^2,
      \\
      \alpha^{-1}\beta_1 |q-r|
      \;\le\; |\sigma_\alpha(q) - \sigma_\alpha(r)|
      \;\le\; \alpha^{-1}\beta_2 |q - r|.
    \end{gathered}
  \end{equation}
  Consequently, the inverse $\rho_\alpha = \sigma_\alpha^{-1}$ satisfies
  \begin{equation}
    \label{eq:rho-bilip}
    \alpha\beta_2^{-1}|p - s|
    \;\le\; |\rho_\alpha(p) - \rho_\alpha(s)|
    \;\le\; \alpha\beta_1^{-1}|p - s|
    \qquad \text{for all } p, s \in \mathbb{{R}}^d.
  \end{equation}

<!-- proof-2: body_md -->
The derivative in~\eqref{eq:Dsigma} is computed easily by chain and
  product rules.  Since $\phi'(|q|)|q|$ vanishes as $q \to 0$
  by~\eqref{eq:phi-props},
  $d\sigma_\alpha$ is continuous on all of $\mathbb{{R}}^d$.
  The symmetric matrix in~\eqref{eq:Dsigma}
  equals $-\alpha^{-1}$ times a matrix with eigenvalue
  $\phi(|q|) + |q| \phi'(|q|)$ in the direction of $q$ and eigenvalue
  $\phi(|q|)$ on the orthogonal complement, both of which lie in
  $[\beta_1,\beta_2]$ by~\eqref{eq:phi-props},  so 
  $d\sigma_\alpha(q)$ has spectrum contained in
  $[-\alpha^{-1}\beta_2, -\alpha^{-1}\beta_1]$ as claimed.  The
  bound~\eqref{eq:sigma-mono-lip} follows from this and the
  fundamental theorem of calculus along the segment from $r$ to $q$,
  since
  \[
    \big( \sigma_\alpha(q) - \sigma_\alpha(r) \big) \cdot (q - r)
    = \int_0^1 (q-r) \cdot d\sigma_\alpha(r + s(q-r))\, (q-r)\, ds
  \]
  lies in $[-\alpha^{-1}\beta_2 |q-r|^2, -\alpha^{-1}\beta_1|q-r|^2]$.
  The upper Lipschitz bound $|\sigma_\alpha(q)-\sigma_\alpha(r)| \le
  \alpha^{-1}\beta_2|q-r|$ follows since $\|d\sigma_\alpha(\cdot)\|
  \le \alpha^{-1}\beta_2$ everywhere
  (cf.\ \cite[Remark~3.1 and~(4.5)]{carstensen2017}).
  The lower Lipschitz bound follows by combining the strong monotonicity
  inequality with the Cauchy--Schwarz inequality
  $-(\sigma_\alpha(q)-\sigma_\alpha(r))\cdot(q-r) \le
  |\sigma_\alpha(q)-\sigma_\alpha(r)|\,|q-r|$ and dividing by $|q-r|$.
  The bounds~\eqref{eq:rho-bilip} on $\rho_\alpha$ follow
  from~\eqref{eq:sigma-mono-lip}
  upon setting $q=\rho_\alpha(p)$ and $r=\rho_\alpha(s)$ and 
  then rearranging.

  (i) By definition, $\mathcal{{B}}_\alpha' {\mathcal{{v}}} \in {\mathcal{{U}}'_\alpha}$ acts by
  $(\mathcal{{B}}_\alpha' {\mathcal{{v}}})({\mathcal{{z}}}) = b_\alpha({\mathcal{{z}}},{\mathcal{{v}}}) = \langle{\hat q_n, w}\rangle_{\partial{\varOmega_h}}
  - \langle{r\cdot n, \hat u}\rangle_{\partial{\varOmega_h}} + (p,r)_{\varOmega_h} + (u,\operatorname{{\ensuremath{\text{div}}}} r)_{\varOmega_h}

  using~\eqref{eq:globalized-sigma} and the value of $\kappa_2$
  in~\eqref{eq:kappa-alpha}. Expressing the cross term as
  $({q_{12}},{p_{12}})_{L_2} = (q_0, p_0)_{L_2} + (q_{12} - q_0,
  p_0)_{L_2} + (q_0, p_{12} - p_0) + (q_{12} - q_0, p_{12} - p_0)$, 
  note that the first term satisfies
  $(q_0, p_0)_{L_2} = (q_0, \nabla u_0)_{L_2} = -(\operatorname{{\ensuremath{\text{div}}}} q_0, u_0) = 0$.
  The last term can be bounded by the inequality of arithmetic and
  geometric means with weight $\kappa_1^{-1}$,
  \begin{align*}
    \frac{\kappa_2}{\kappa_1^{2}}\,|(q_{12} - q_0, p_{12} - p_0)_{L_2}|
    &\le
      \frac{\kappa_2}{2\kappa_1}\!\Big(
      \kappa_1^{-2}\|q_{12}-q_0\|_{L_2}^2 + \|p_{12}-p_0\|_{L_2}^2
      \Big)
    \le
      \frac{c_B\kappa_2}{2\kappa_1}\,\mathcal{{B}}_{12}
  \end{align*}
  by virtue of~\eqref{eq:S4-infsup}. For the remaining terms, we use
  Young's inequality, 
  \begin{align*}
    \frac{\kappa_2}{\kappa_1^{2}}\,|(q_{12} - q_0, p_0)_{L_2}|
    &\le
      \frac{\kappa_2}{2\varepsilon_1}\,\kappa_1^{-2}\|q_{12}-q_0\|_{L_2}^2
      + \frac{\kappa_2\varepsilon_1}{2}\,\kappa_1^{-2}\|p_0\|_{L_2}^2
      \le
      \frac{c_B\kappa_2}{2\varepsilon_1}\,\mathcal{{B}}_{12}
      + \frac{\kappa_2\varepsilon_1}{2\kappa_1^2}\,P_{12},
  \end{align*}
  using \eqref{eq:S4-infsup}, where~$\varepsilon_1$ is a free
  parameter. Similarly using a second free parameter~$\varepsilon_2$,
  \begin{align*}
    \frac{\kappa_2}{\kappa_1^{2}}\,|(q_0, p_{12} - p_0)_{L_2}|
    &\le
      \frac{\kappa_2\varepsilon_2}{2}\,P_{12}  
      + \frac{c_B\kappa_2}{2\varepsilon_2\kappa_1^2}\, 
      \mathcal{{B}}_{12}.  
  \end{align*}
  Summing these cross-term 
  contributions and setting
  \[
    \tilde{c}_B(\varepsilon_1,\varepsilon_2) = \frac 1 2 {c_B\kappa_2}
    \!\left( \kappa_1^{-1} + \varepsilon_1^{-1} + \varepsilon_2^{-1}\kappa_1^{-2}\right),
  \]
  the estimate of~\eqref{eq:211} implies
  \begin{align}
    \label{eq:171}
    \mathcal{{C}}_{12}
    +
    \tilde{c}_B(\varepsilon_1,\varepsilon_2)\,
    \mathcal{{B}}_{12}  
    +
    \left(\frac{\varepsilon_1\kappa_2}{2\kappa_1^2} + \frac{\kappa_2\varepsilon_2}{2}\right)
    P_{12}  
    \ge  T,
  \end{align}
  where
  $T = \kappa_1^{-2} \big[(\kappa_2^2/\mu)\|{p_{12}}\|^2_{L_2} +
  \|{q_{12}}\|^2_{L_2}
  -\sqrt{\kappa_1}\|{p_{12}}\|_{L_2}\|{q_{12}}\|_{L_2}\big].$

  We can  view $T$ as a quadratic form of the matrix $M(t)$ defined below:
  \[
    r = \frac{\mu}{\kappa_2^2},
    \quad
    t = \frac{ \mu \kappa_1}{\kappa_2^2},
    \quad 
    M(t) =
    \begin{bmatrix}
      1     & -\frac 1 2 \sqrt{r/t}
      \\
      -\frac 1 2 \sqrt{r/t} & r/t^2     
    \end{bmatrix},
    \quad
    y =
    \begin{bmatrix}
      \kappa_1^{-1}\| q_{12} \|_{L_2} \\
      \phantom{\kappa_1^{-1}}\| p_{12} \|_{L_2}
    \end{bmatrix}.
  \]
  Clearly $T = y \cdot M(t) y$. By \eqref{eq:20},
  $0 < \kappa_1 < \kappa_2^2/\mu$, so $ 0 < t < 1$. The smallest
  eigenvalue of $M(t)$, namely
  $\lambda_-(t) = \frac 1 2 \big[ 1 + r/t^2 - \sqrt{ (1 + r/t^2)^2 -
    r (4-t)/t^2 } \big],$ is monotonically decreasing in
  $0 < t \le 1$ (see Remark~\ref{rem:mon-detivative-ew} below), so 
  $T = y \cdot M(t) y \ge \lambda_-(1) \,y\cdot y$. Thus, letting
  $c_r = \lambda_-(1),$ the estimate of \eqref{eq:171} implies
  \begin{align*}
    c_r\Big( \| p_{12}\|_{L_2}^2  +  \kappa_1^{-2} \| q_{12}\|_{L_2}^2\Big)
    \le T \le 
    \mathcal{{C}}_{12}
    + \tilde{c}_B(\varepsilon_1,\varepsilon_2)\,
    \mathcal{{B}}_{12}  
    +
    \left(\frac{\varepsilon_1\kappa_2}{2\kappa_1^2} + \frac{\kappa_2\varepsilon_2}{2}\right)
    P_{12}.
  \end{align*}
  Returning to \eqref{eq:cE-PBdelta}, this gives 
  \begin{align}
    \label{eq:17}
    \left(\frac{c_Ec_r}{2} - \frac{\varepsilon_1\kappa_2}{2\kappa_1^2}
    - \frac{\kappa_2\varepsilon_2}{2}\right)
    P_{12}  
    \;\le\;
    \mathcal{{C}}_{12}
    + \Big(c_Bc_r + \tilde{c}_B(\varepsilon_1,\varepsilon_2)\Big)
    \, \mathcal{{B}}_{12}.
  \end{align}
  Choosing $\varepsilon_1 = \frac{1}{4} c_Ec_r\kappa_1^2/\kappa_2$ and
  $\varepsilon_2 = \frac{1}{4}c_Ec_r / \kappa_2,$ the left hand side becomes
  $\frac 1 4 {c_Ec_r} P_{12}$ and we obtain~\eqref{eq:S4-mono}.  (A similar
  estimate with different constants is stated
  in~\cite[Lemma~4.6]{CH}.)

    

  (iv) Suppose ${\mathcal{{z}}}=(q,u,p,\hat u,\hat q_n) \in {\mathcal{{U}}_\alpha}$ satisfies
  $\langle \mathcal{{C}}_\alpha({\mathcal{{z}}}), {\mathcal{{z}}}_0 \rangle = 0$ for all
  ${\mathcal{{z}}}_0 \in \ker(\mathcal{{B}}_\alpha)$.  Set $g := q - \sigma_\alpha(p) \in
  L_2(\varOmega)^d$.  Using~\eqref{eq:E-map-diffus}
  and~\eqref{eq:C-mdpg'}, for every $u_0 \in \mathring{H}^1(\varOmega)$ and
  $q_0 \in H(\text{div0})$,
  \[
    0 = \big\langle \mathcal{{C}}_\alpha({\mathcal{{z}}}), E(q_0, u_0) \big\rangle
    = \kappa_1^{-2} \big( g, \; q_0 + \kappa_2 \nabla u_0 \big)_{\varOmega_h}.
  \]
  Choosing $q_0 = 0$ gives $g \perp \nabla \mathring{H}^1(\varOmega)$, and choosing
  $u_0 = 0$ gives $g \perp H(\text{div0})$.  Since
  $\nabla \mathring{H}^1(\varOmega) \oplus H(\text{div0})$ equals  all 
  $L_2(\varOmega)^d$, by the Helmholtz decomposition,  we conclude that 
  $g = 0$, and hence
  $\mathcal{{C}}_\alpha({\mathcal{{z}}}) = 0$ in ${\mathcal{{U}}'_\alpha}$, which is~\eqref{eq:C-zero}.

  
  (v) For ${\mathcal{{z}}}_1,{\mathcal{{z}}}_2 \in {\mathcal{{U}}_\alpha}$ and any ${\mathcal{{z}}} \in {\mathcal{{U}}_\alpha}$, the
  Cauchy--Schwarz inequality and the global Lipschitz bound
  $\|\sigma_\alpha(p_1)-\sigma_\alpha(p_2)\|_{L_2} \le \sqrt{\kappa_1}
  \, \|p_1-p_2\|_{L_2}$ (that follows from~\eqref{eq:sigma-mono-lip}),
  give
  \[
    \big| \langle \mathcal{{C}}_\alpha({\mathcal{{z}}}_1) - \mathcal{{C}}_\alpha({\mathcal{{z}}}_2), {\mathcal{{z}}} \rangle \big|
    = \kappa_1^{-2}
    \big| \big( {q_{12}}-{\sigma_{12}}, \, \kappa_2 p + q
    \big)_{L_2} \big|
    \le \kappa_1^{-2}
    \big( \sqrt{\kappa_1}\,\|{p_{12}}\|_{L_2} + \|{q_{12}}\|_{L_2} \big)
    \, \| \kappa_2 p + q \|_{L_2}.
  \]
  Both factors are now expressed in the weighted
  norm~\eqref{eq:mdpg-norms}. By the Cauchy--Schwarz inequality
  in~$\mathbb{{R}}^2$,
  \[
    \sqrt{\kappa_1}\,\|p_{12}\|_{L_2} + \|q_{12}\|_{L_2}
    = \sqrt{\kappa_1}\,\|p_{12}\|_{L_2} + \kappa_1\,\kappa_1^{-1}\|q_{12}\|_{L_2}
    \le \big(\kappa_1 + \kappa_1^2\big)^{1/2}
    \|{\mathcal{{z}}}_1 - {\mathcal{{z}}}_2\|_{{\mathcal{{U}}_\alpha}},
  \]
  and similarly
  $\|\kappa_2 p + q\|_{L_2} \le \kappa_2\|p\|_{L_2} + \kappa_1\,\kappa_1^{-1}\|q\|_{L_2}
  \le (\kappa_1^2 + \kappa_2^2)^{1/2}\|{\mathcal{{z}}}\|_{{\mathcal{{U}}_\alpha}}$. Combining these
  bounds yields~\eqref{eq:S4-Lip} with the stated $c_{\rm Lip}$.

Writing ${\mathcal{{z}}} = (q, u, p)$ for a generic element of ${\mathcal{{U}}_\alpha}$, we prove
  each item.

  (i)--(ii) Given ${\mathcal{{z}}} \in {\mathcal{{U}}_\alpha}$, let $\varphi \in \mathring{H}^1(\varOmega)$ solve
  the Dirichlet problem
  $(\nabla \varphi, \nabla v)_{L_2} = -(\operatorname*{div} q, v)_{L_2}$ for all
  $v \in \mathring{H}^1(\varOmega)$.  Then $\nabla \varphi \in H(\operatorname*{div})$ with
  $\operatorname*{div} \nabla\varphi = \operatorname*{div} q$, and testing with $v = \varphi$, the
  Friedrichs inequality~\eqref{eq:Friedrichs-std} gives
  $\| \nabla \varphi \|_{L_2} \le c_P^{-1/2} \| \operatorname*{div} q \|_{L_2}$.  By
  \eqref{eq:E-map-fosls}, the function
  ${\mathcal{{z}}}_0 := (q - \nabla\varphi, \, u, \, \nabla u)$ belongs to
  $\ker(\mathcal{{B}}_\alpha)$, and ${\mathcal{{z}}} - {\mathcal{{z}}}_0 = (\nabla\varphi, 0, p - \nabla u)$. This gives
  \[
    \| {\mathcal{{z}}} - P_\alpha{\mathcal{{z}}} \|_{{\mathcal{{U}}_\alpha}}^2
    \le \| {\mathcal{{z}}} - {\mathcal{{z}}}_0 \|_{{\mathcal{{U}}_\alpha}}^2
    = \kappa_1^{-2} \big( \| \nabla \varphi \|_{L_2}^2
    + \| \operatorname*{div} q \|_{L_2}^2 \big) + \| p - \nabla u \|_{L_2}^2
    \le ( 1 + c_P^{-1} ) \, \| \mathcal{{B}}_\alpha {\mathcal{{z}}} \|_{{\mathcal{{V}}'_\alpha}}^2,
  \]
  where the last step uses
  $\| \mathcal{{B}}_\alpha {\mathcal{{z}}} \|_{{\mathcal{{V}}'_\alpha}}^2 = \kappa_1^{-2} \| \operatorname*{div} q \|_{L_2}^2 + \|
  p - \nabla u \|_{L_2}^2$, a consequence of the self-duality of the
  weighted $L_2$ norms in~\eqref{eq:mfosls-norms}.  This
  proves~(ii).  Regarding~(i), given any $(g_1, g_2) \in L_2(\varOmega) \times L_2(\varOmega)^d
  \cong {\mathcal{{V}}'_\alpha}$, solving the same Dirichlet problem with $-\operatorname*{div} q$
  replaced by $-g_1$ and setting ${\mathcal{{z}}} = (\nabla\varphi, 0, g_2)$,
  yields $\mathcal{{B}}_\alpha {\mathcal{{z}}} = (g_1, g_2)$.

  (iii) The proof of Proposition~\ref{prop:mdpg-verif}(iii) applies

  $\operatorname*{div} q_0 = 0$, and $u_0 \in \mathring{H}^1(\varOmega)$), as well as manipulations of the

  As for (iv), suppose ${\mathcal{{z}}} \in {\mathcal{{U}}_\alpha}$ satisfies
  $\langle{\mathcal{{C}}_\alpha({\mathcal{{z}}}), {\mathcal{{z}}}_0}\rangle = 0$ for all ${\mathcal{{z}}}_0 \in \ker(\mathcal{{B}}_\alpha)$.  Set
  $g := q - \sigma_\alpha(p) \in L_2(\varOmega)^d$.
  Using~\eqref{eq:E-map-fosls} and~\eqref{eq:C-mdpg'}, for every
  $u_0 \in \mathring{H}^1(\varOmega)$ and $q_0 \in H(\text{div0})$, we have 
  $    0 = \big\langle \mathcal{{C}}_\alpha({\mathcal{{z}}}), E(q_0, u_0) \big\rangle
    = \kappa_1^{-2} \big( g, \; q_0 + \kappa_2 \nabla u_0 \big)_{\varOmega_h}.
  $
  Choosing $q_0 = 0$, gives $g \perp \nabla \mathring{H}^1(\varOmega)$, and choosing

  To see (v), as in the proof of Proposition~\ref{prop:mdpg-verif}(v),  the global Lipschitz bound
  $\|\sigma_\alpha(p_1)-\sigma_\alpha(p_2)\|_{L_2} \le \sqrt{\kappa_1}

    \big| \langle \mathcal{{C}}_\alpha({\mathcal{{z}}}_1) - \mathcal{{C}}_\alpha({\mathcal{{z}}}_2), {\mathcal{{z}}} \rangle \big|
    \le \kappa_1^{-2}
    \big( \sqrt{\kappa_1}\,\|{p_{12}}\|_{L_2} + \|{q_{12}}\|_{L_2} \big)
    \, \| q + \kappa_2 p \|_{L_2}.
  \end{equation}
  In view the norm~\eqref{eq:mfosls-norms},   the
  Cauchy--Schwarz inequality in $\mathbb{{R}}^2$ provides
  $
      \sqrt{\kappa_1}\, \| p_{12} \|_{L_2} + \| q_{12} \|_{L_2}
    \le \big( \kappa_1 + \kappa_1^2 \big)^{1/2}
    \big( \| p_{12} \|_{L_2}^2
    + \kappa_1^{-2} \| q_{12} \|_{L_2}^2 \big)^{1/2}.
    $
    Hence 
    \begin{gather}
      \label{eq:mfosls-CS2}
      \sqrt{\kappa_1}\, \| p_{12} \|_{L_2} + \| q_{12} \|_{L_2}
      \le \big( \kappa_1 + \kappa_1^2 \big)^{1/2}
      \| {\mathcal{{z}}}_1 - {\mathcal{{z}}}_2 \|_{{\mathcal{{U}}_\alpha}}.
    \end{gather}
    Similarly, one obtains
  \begin{gather}
    \label{eq:mfosls-CS1}
    \| q + \kappa_2 p \|_{L_2}
    \le \kappa_1 \, \kappa_1^{-1} \| q \|_{L_2}
    + \kappa_2 \| p \|_{L_2}
    \le \big( \kappa_1^2 + \kappa_2^2 \big)^{1/2} \| {\mathcal{{z}}} \|_{{\mathcal{{U}}_\alpha}}.
  \end{gather}
  Combining~\eqref{eq:C12-mfosls},~\eqref{eq:mfosls-CS2} and~\eqref{eq:mfosls-CS1},  proves~\eqref{eq:S4-Lip} with the
  stated constant $c_{\rm Lip}$.  Condition~\eqref{eq:N-pairing} also follows
  from~\eqref{eq:mfosls-CS1}, since
  $\langle{ \mathcal{{C}}_\alpha({\mathcal{{z}}}_1), {\mathcal{{z}}} }\rangle
  = \kappa_1^{-2} ( q_1 - \sigma_\alpha(p_1),
  \, q + \kappa_2 p )_{L_2}
  \le \mathcal{{M}}_\alpha({{\mathcal{{z}}}_1}) \| {\mathcal{{z}}} \|_{{\mathcal{{U}}_\alpha}}$,
  by~\eqref{eq:mfosls-N}.  Finally, if $\bar{\mathcal{{z}}} \in {\mathcal{{U}}_\alpha}$ satisfies
  $\mathcal{{C}}_\alpha(\bar{\mathcal{{z}}}) = 0$, then testing with tuples ${\mathcal{{z}}} = (0, 0, p)$
  shows $(\bar q - \sigma_\alpha(\bar p), p)_{L_2} = 0$ for all
  $p \in L_2^d$, i.e., $\bar q = \sigma_\alpha(\bar p)$.  Hence, for
  any ${\mathcal{{z}}}_1 \in {\mathcal{{U}}_\alpha}$, writing
  $q_1 - \sigma_\alpha(p_1) = (q_1 - \bar q) -
  (\sigma_\alpha(p_1) - \sigma_\alpha(\bar p))$, the Lipschitz bound
  and~\eqref{eq:mfosls-CS2} (with ${\mathcal{{z}}}_2 = \bar{\mathcal{{z}}}$) give
  \[
    \mathcal{{M}}_\alpha({{\mathcal{{z}}}_1})
    \le \kappa_1^{-2} \big( \kappa_1^2 + \kappa_2^2 \big)^{1/2}
    \big( \| q_1 - \bar q \|_{L_2}
    + \sqrt{\kappa_1} \| p_1 - \bar p \|_{L_2} \big)
    \le c_M \| {\mathcal{{z}}}_1 - \bar{\mathcal{{z}}} \|_{{\mathcal{{U}}_\alpha}},
  \]
