<!-- defi-1: body_md -->
Let $\lambda\in \mathbb C\setminus \sigma(-\Delta_\xi)$. We define the Green function associated with the fiber operator $-\Delta_\xi$ by
\begin{equation}\label{equ:Gfourier}
G_\lambda^\xi(x,y)
:=
\sum_{k\in\Lambda^*}
\frac{e_k(x)\overline{e_k(y)}}{|k+\xi|^2-\lambda}
=
\frac{1}{|Y|}\sum_{k\in\Lambda^*}
\frac{e^{i(k+\xi)\cdot (x-y)}}{|k+\xi|^2-\lambda}
,
\qquad x,y\in Y.
\end{equation}
The series is understood in \(L^2(Y\times Y)\), as justified in
Proposition~\ref{prop:resolvent_kernel}.

<!-- defi-2: body_md -->
An operator \(T\in\mathcal L(H)\) is Hilbert--Schmidt if
\(\|T\|_2^2:=\operatorname{tr}(T^*T)<\infty\). The class of such
operators is denoted by \(\mathfrak S_2\).

<!-- thm-1: body_md -->
An operator \(T\in\mathcal L(L^2(Y))\) is Hilbert--Schmidt if and only if
there is a kernel \(K\in L^2(Y\times Y)\) such that
\[
Tf(x)=\int_Y K(x,y)f(y)\,dy
\qquad\text{for every }f\in L^2(Y).
\]
In that case,
\[
\|T\|_2^2=\int_{Y\times Y}|K(x,y)|^2\,dx\,dy.
\]

<!-- prop-1: body_md -->
\label{prop:resolvent_kernel}
Let $\lambda\in \mathbb C\setminus \sigma(-\Delta_\xi)$. Then
\[
G_\lambda^\xi \in L^2(Y\times Y).
\]
Moreover, for every $f\in L^2(Y)$, one has
\begin{equation}\label{equ:resolvent_kernel}
(-\Delta_\xi-\lambda)^{-1}f(x)
=
\int_Y G_\lambda^\xi(x,y)f(y)\,dy
\qquad \text{in }L^2(Y).
\end{equation}
In other words, $G_\lambda^\xi$ is the resolvent kernel of $(-\Delta_\xi-\lambda)^{-1}$.
<!-- end prop-1 -->
<!-- proof-1: body_md -->
The functions \(e_k(x)\overline{e_k(y)}\), \(k\in\Lambda^*\), are
orthonormal in \(L^2(Y\times Y)\). Since
\[
\sum_{k\in\Lambda^*}\frac{1}{\bigl||k+\xi|^2-\lambda\bigr|^2}<\infty
\]
in two dimensions, the series defining \(G_\lambda^\xi\) converges in
\(L^2(Y\times Y)\). For each finite partial sum, integration against
\(f\) gives the corresponding partial sum of the Fourier expansion of
\((-\Delta_\xi-\lambda)^{-1}f\). By the Cauchy--Schwarz inequality, the
\(L^2(Y\times Y)\) convergence of the kernels implies convergence of
these integrals in \(L^2(Y)\), which proves the claimed kernel
representation.

<!-- thm-2: body_md -->
\label{thm:LSequation}
Fix \(\xi\in B\) and \(\lambda\notin\sigma(-\Delta_\xi)\). Then
\(\lambda\in\sigma(P_\xi)\) if and only if the fiber
Lippmann--Schwinger equation
\begin{equation}\label{equ:LS_equation}
    \big(I-\tau\lambda\mathcal T_{D_\eta}^{\xi,\lambda}\big)[u](x)=0,\quad\text{for a.e. }x\in D_\eta,
\end{equation}
admits a nontrivial solution in \(L^2(D_\eta)\), where
\[
\mathcal T_{D_\eta}^{\xi,\lambda}[u](x)
:=\int_{D_\eta}G_\lambda^\xi(x,y)u(y)\,dy.
\]
By Proposition~\ref{prop:resolvent_kernel} and the kernel criterion,
\(\mathcal T_{D_\eta}^{\xi,\lambda}\) is Hilbert--Schmidt on
\(L^2(D_\eta)\). Equivalently, \(\lambda\) is a characteristic value
of the operator function
\[
\mathcal A(\mu,\xi;\tau,\eta):=I-\tau\mu\mathcal T_{D_\eta}^{\xi,\mu}.
\]
<!-- end thm-2 -->
<!-- proof-2: body_md -->
The form definitions and \(1\leq n_\eta\leq1+\tau\) imply
\[
D(P_\xi)=D(-\Delta_\xi),\qquad -\Delta_\xi=n_\eta P_\xi.
\]
Suppose first that \(\lambda\in\sigma(P_\xi)\). Since \(P_\xi\) has
compact resolvent, there is a nonzero \(u\in D(P_\xi)\) such that
\[
P_\xi u=\lambda u.
\]
By the definition of \(P_\xi\),
\[
\int_Y \overline{\nabla v}\cdot\nabla u\,dx
=
\lambda\int_Y(1+\tau\chi_{D_\eta})\overline v u\,dx
\qquad \forall v\in H_\xi^1(Y).
\]
Thus, in the distributional sense,
\begin{align} \label{equ:LS1}
(-\Delta_\xi-\lambda)u=\lambda\tau\chi_{D_\eta}u.
\end{align}

Since \(\lambda\notin\sigma(-\Delta_\xi)\), applying the free resolvent
and using \eqref{equ:resolvent_kernel} gives
\begin{equation}\label{equ:LS2}
u(x)
=
\lambda\tau\int_{D_\eta} G_\lambda^\xi(x,y)u(y)\,dy,
\qquad\text{for a.e. }x\in Y.
\end{equation}
The restriction of \(u\) to \(D_\eta\) cannot vanish, since otherwise
\eqref{equ:LS2} would give \(u=0\). Restricting \eqref{equ:LS2} to
\(D_\eta\) proves one implication.

Conversely, suppose \(0\neq u\in L^2(D_\eta)\) solves
\eqref{equ:LS_equation}. Define
\begin{equation}\label{equ:LS3}
U(x):=\lambda\tau\int_{D_\eta} G_\lambda^\xi(x,y)u(y)\,dy,
\qquad x\in Y.
\end{equation}
Since \(U=u\) almost everywhere on \(D_\eta\), it is nonzero. The
resolvent representation gives
\[
U=\lambda\tau(-\Delta_\xi-\lambda)^{-1}(\chi_{D_\eta}U).
\]
Hence \(U\in D(-\Delta_\xi)\) and
\((-\Delta_\xi-\lambda)U=\lambda\tau\chi_{D_\eta}U\).
This is the fiber eigenvalue equation \(P_\xi U=\lambda U\), so
\(\lambda\in\sigma(P_\xi)\).
Restriction to \(D_\eta\) and the extension \eqref{equ:LS3}
are inverse linear maps between the two eigenspaces.

\label{thm:two_characteristic_values}
Let $\Xi\Subset\mathbb R^2$ satisfy
\[
\inf_{\xi\in\Xi}\inf_{k\in\Lambda^*}|k+\xi|>0,
\]
and set
\[
\epsilon=\frac1\tau,
\qquad
\widehat{\mathcal A}(\widehat\lambda,\xi;\epsilon,\eta)
:=
\mathcal A(\epsilon\widehat\lambda,\xi;\epsilon^{-1},\eta).
\]
If
\[
\eta\to0,
\qquad
\frac{\epsilon}{\eta^2|\log\eta|}\to0,
\]
then, for every $\xi\in\Xi$, there exist two characteristic values
$\widehat\lambda_1(\xi)$ and $\widehat\lambda_2(\xi)$, counted with
algebraic multiplicity, near
\[
\frac{2\pi}{|D|\eta^2|\log\eta|},
\]
such that
\[
\widehat\lambda_{1,2}(\xi)
=
\frac{2\pi}{|D|\eta^2|\log\eta|}
\left[
1
+
O\left(\frac1{|\log\eta|}\right)
+
O\left(
\frac{\epsilon}{\eta^2|\log\eta|^2}
\right)
\right].
\]
The estimates are uniform with respect to $\xi\in\Xi$.

\frac{\epsilon}

\right)
\right].

characteristic values and their algebraic multiplicities, the same
conclusion holds for
\(\widehat{\mathcal A}
(\widehat\lambda,\xi;\epsilon,\eta)\).

\label{prop:general-xi-effective}
Let \(\Xi\Subset\mathbb R^2\) satisfy
\(\inf_{\xi\in\Xi}\inf_{k\in\Lambda^*}|k+\xi|>0\), and set
\[
L:=|\log\eta|,
\qquad
\delta_{\epsilon,\eta}
:=
\frac1L+\frac{\epsilon}{\eta^2L^2}.
\]
Assume that \(\eta\to0\) and \(\epsilon/(\eta^2L)\to0\).
Then, uniformly for \(\xi\in\Xi\), the two characteristic values given by
Theorem~\ref{thm:two_characteristic_values}, counted with algebraic
multiplicity and labeled so that \(\widehat\lambda_-(\xi)\leq
\widehat\lambda_+(\xi)\), satisfy
\begin{equation}\label{equ:general-xi-refined-eigenvalues}
\widehat\lambda_\pm(\xi)
=
\frac{2\pi}{|D|\eta^2L}
\left[
1-\frac{2\pi\mu_\mp(\xi)}{|D|L}
+
O\left(
\frac1{L^2}
+\frac{\epsilon}{\eta^2L^2}
\right)
\right],
\end{equation}
where
\begin{equation}\label{equ:mu-pm-xi}
\mu_\pm(\xi)
=\kappa_\xi\pm|g(\xi)|,
\end{equation}
and
\begin{align}
\kappa_\xi
&:=
\frac1{|D|}
\int_D\int_D
\left[
-\frac1{2\pi}\log|x-y|+R_\xi
\right]dx\,dy,
\label{equ:kappa-xi}
\\
g(\xi)
&:=|D|G_0^\xi(z_1,z_2).
\label{equ:g-xi}
\end{align}
In particular, \(\kappa_{-\xi}=\kappa_\xi\).
Moreover, the associated normalized characteristic functions are,
after applying \(U_\eta\), within
\(O(\delta_{\epsilon,\eta})\) of the two-dimensional constant-mode
space in \(L^2(D)\oplus L^2(D)\).

<!-- proof-4: body_md -->
Let
\[
u_0:=|D|^{-1/2}\mathbf 1_D,
\qquad
e_1:=\begin{pmatrix}u_0\\0\end{pmatrix},
\qquad
e_2:=\begin{pmatrix}0\\u_0\end{pmatrix},
\]
and let \(\Pi\) be the orthogonal projection onto
\(\mathcal E_0:=\operatorname{span}\{e_1,e_2\}\).
Set \(Q:=I-\Pi\).  By Theorem~\ref{thm:two_characteristic_values}, the
two characteristic values under consideration satisfy
\(\widehat\lambda\eta^2=O(L^{-1})\).

Following \eqref{equ:LS_operator_matrix}, define the scaled
two-inclusion integral operator by
\[
\mathbb T_\eta^\xi(\epsilon\widehat\lambda)
:=
\begin{pmatrix}
\mathcal T_D^{\xi,\epsilon\widehat\lambda,\eta}
&\mathcal R_+
\\
\mathcal R_-
&\mathcal T_D^{-\xi,\epsilon\widehat\lambda,\eta}
\end{pmatrix}.
\]
Thus the conjugated characteristic operator is
\(U_\eta\widehat{\mathcal A}
(\widehat\lambda,\xi;\epsilon,\eta)U_\eta^{-1}
=I-\widehat\lambda\eta^2\mathbb T_\eta^\xi(\epsilon\widehat\lambda)\).

The diagonal and off-diagonal kernel expansions give
\begin{equation}\label{equ:T-effective-simple}
\mathbb T_\eta^\xi(\epsilon\widehat\lambda)
=
\frac{|D|L}{2\pi}\Pi+\mathbb B(\xi)
+
O\left(
\eta+\frac{\epsilon}{\eta^2L}
\right),
\end{equation}
where
\[
\mathbb B(\xi)
:=
\begin{pmatrix}
\mathcal K_\xi
&G_0^\xi(z_1,z_2)\mathcal P
\\
G_0^\xi(z_2,z_1)\mathcal P
&\mathcal K_\xi
\end{pmatrix}.
\]
Consequently, in the basis \(\{e_1,e_2\}\),
\begin{equation}\label{equ:B-xi}
\Pi\mathbb B(\xi)\Pi
=
\mathbb K(\xi)
:=
\begin{pmatrix}
\kappa_\xi&g(\xi)\\
\overline{g(\xi)}&\kappa_\xi
\end{pmatrix}.
\end{equation}

Write the characteristic operator with respect to
\(\mathcal E_0\oplus\operatorname{Ran}Q\) as
\[
\mathbb A(\widehat\lambda,\xi)
=I-\widehat\lambda\eta^2\mathbb T_\eta^\xi(\epsilon\widehat\lambda)
=
\begin{pmatrix}
A_{\Pi\Pi}&A_{\Pi Q}\\
A_{Q\Pi}&A_{QQ}
\end{pmatrix}.
\]
Since \(\widehat\lambda\eta^2=O(L^{-1})\), formula
\eqref{equ:T-effective-simple} implies
\[
A_{QQ}=Q+O(L^{-1}),
\qquad
A_{QQ}^{-1}=Q+O(L^{-1}).
\]
Thus \(A_{QQ}\) is invertible on \(\operatorname{Ran}Q\).

If \(f=f_\Pi+f_Q\) satisfies \(\mathbb Af=0\), the \(Q\)-equation gives
\(f_Q=-A_{QQ}^{-1}A_{Q\Pi}f_\Pi\). Substitution into the \(\Pi\)-equation
gives the Schur-complement equation
\begin{equation}\label{equ:Feshbach-equation}
\left(
A_{\Pi\Pi}-A_{\Pi Q}A_{QQ}^{-1}A_{Q\Pi}
\right)f_\Pi=0.
\end{equation}
Conversely, every solution of \eqref{equ:Feshbach-equation} gives a
solution of the full equation through the preceding formula for
\(f_Q\).

The off-diagonal blocks are \(O(L^{-1})\), so
\(A_{\Pi Q}A_{QQ}^{-1}A_{Q\Pi}=O(L^{-2})\).
Using \eqref{equ:T-effective-simple} and \eqref{equ:B-xi}, the
two-dimensional equation \eqref{equ:Feshbach-equation} becomes
\begin{equation}\label{equ:Feshbach-expansion}
\det\left[
\left(
1-\frac{|D|L}{2\pi}\widehat\lambda\eta^2
\right)I_2
-\widehat\lambda\eta^2\mathbb K(\xi)
+
O\left(
\frac1{L^2}
+\frac{\epsilon}{\eta^2L^2}
\right)
\right]=0.
\end{equation}

The eigenvalues of \(\mathbb K(\xi)\) are \(\mu_\pm(\xi)\) in
\eqref{equ:mu-pm-xi}.
For real \(\widehat\lambda\) in the above spectral window, the Schur
complement is Hermitian and strictly decreasing in \(\widehat\lambda\).
Indeed, the Fourier representation gives
\(\partial_{\widehat\lambda}\widehat{\mathcal A}
=-\mathcal T-\epsilon\widehat\lambda\partial_\omega\mathcal T<0\),
with \(\mathcal T=\mathcal T_{D_\eta}^{\xi,\epsilon\widehat\lambda}\),
and differentiation of the Schur complement preserves this inequality.
The min--max principle and the two-root count therefore pair its zeros
with the ordered eigenvalues of the leading matrix, including when
\(\mu_+=\mu_-\). The matrix perturbation bound in
\eqref{equ:Feshbach-expansion} gives
\[
1-\widehat\lambda_\pm(\xi)\eta^2
\left(
\frac{|D|L}{2\pi}+\mu_\mp(\xi)
\right)
=
O\left(
\frac1{L^2}
+\frac{\epsilon}{\eta^2L^2}
\right).
\]
Solving for \(\widehat\lambda_\pm(\xi)\) and expanding the denominator
gives \eqref{equ:general-xi-refined-eigenvalues}.

Finally, the preceding formula for \(f_Q\) implies
\(\|f_Q\|\le C\delta_{\epsilon,\eta}\|f_\Pi\|\), which proves the
characteristic-function estimate.

Threefold rotational symmetry gives \(\kappa_{R\xi}=\kappa_\xi\),
with quasi-momenta understood modulo \(\Lambda^*\). Since
\(RK=K\pmod{\Lambda^*}\), the vector \(\nabla_\xi\kappa_K\) must be
fixed by the rotation \(R\), and hence it vanishes. Using
\(\kappa_{-\xi}=\kappa_\xi\), we obtain
\[
\kappa_{K+h}
\;=\;\kappa_{-K-h}
\;=\;
\kappa_K+O(|h|^2).
\]
Set \(F(\xi):=G_0^\xi(z_1,z_2)\) and \(d:=z_1-z_2\).
Since \(R^{-1}d=d-\mathbf e_1\), rotation covariance and

\[
 F(R\xi)=e^{-i\xi\cdot\mathbf e_1}F(\xi).
\]
Differentiating at \(K\), using \(F(K)=0\) and
\(RK\equiv K\pmod{\Lambda^*}\), yields
\(R^T\nabla F(K)=\kappa^2\nabla F(K)\).
The corresponding eigenspace is \(\mathbb C(1,i)^T\), proving
the asserted form \(\nabla F(K)=c(1,i)^T\).

\label{thm:eta-global-continuation}
Fix \(\tau>0\), and suppose \((z_s,\eta_s)\) is good for some
\(\eta_s\in(0,\eta_{\mathrm{c}})\).
There is a real-analytic function
\(\lambda:(0,\eta_{\mathrm{c}})\to\mathbb R\), with
\(\lambda(\eta_s)=z_s\), which is a sector-\(1\) eigenvalue for every
\(\eta\).  The set
\[
 \mathcal C_\tau
 :=\{\eta\in(0,\eta_{\mathrm{c}}):(\lambda(\eta),\eta)
                                  \text{ is not good}\}
\]
is relatively closed and discrete. The ordered band indices may change at spectral crossings.
<!-- end thm-5 -->
<!-- proof-7: body_md -->
\textit{Step 1: start a local analytic branch.}
Since \((z_s,\eta_s)\) is good, \(z_s\) is a simple eigenvalue
of \(H_1(\eta_s)\).  The local analytic theory gives a
real-analytic eigenvalue branch \(\lambda(\eta)\) with
\(\lambda(\eta_s)=z_s\), together with a normalized analytic
eigenfunction.  Spectral isolation and continuity of the conical
coefficient ensure, after shrinking the neighborhood, that this
branch is good throughout it.

\medskip\noindent
\textit{Step 2: control the eigenvalue along a continuation.}
Cover a compact interval \(I\Subset(0,\eta_{\mathrm{c}})\) by finitely
many smaller neighborhoods on which the fixed-reference
constructions are valid.  Boundedness of the form derivatives
and uniform coercivity on these neighborhoods give a common
constant \(C_I\) such that, in each representation,
\[
 |\partial_\eta a_{K,\eta}(u,u)|
 \le C_I\bigl(a_{K,\eta}(u,u)+b(u,u)\bigr),
 \qquad u\in H^1_{\mathrm{per}}(Y).
\]
For a local normalized Rellich eigenpair, differentiation gives
\[
 \lambda'(\eta)=\partial_\eta a_{K,\eta}(u,u),\qquad
 |\lambda'(\eta)|\le C_I(1+\lambda(\eta)).
\]
Indeed, the terms involving \(u'\) cancel by the eigenvalue
equation and \(\partial_\eta b(u(\eta),u(\eta))=0\), since
\(b\) is fixed within each representation.  
Since \(\lambda\ge0\), integration yields
\begin{equation}
 1+\lambda(\eta)
 \le (1+\lambda(\eta_0))e^{C_I|\eta-\eta_0|}
 \label{equ:continued-eigenvalue-bound}
\end{equation}
whenever the branch is defined between \(\eta_0\) and \(\eta\)
in \(I\).  Thus at an interior endpoint the eigenvalue remains
bounded, and the derivative bound gives a finite limit.

\medskip\noindent
\textit{Step 3: extend across an interior endpoint.}
Suppose the initial branch has been continued to an open interval
containing \(\eta_s\), remaining good except on a locally discrete
set, and let \(r<\eta_{\mathrm{c}}\) be its right endpoint.
Choose a compact admissible interval containing \([\eta_s,r]\)
in its interior.  Step 2 bounds both \(\lambda\) and
\(\lambda'\) up to \(r\), so
\[
 z_c:=\lim_{\eta\uparrow r}\lambda(\eta)<\infty.
\]
We claim that \(z_c\in\sigma(H_1(r))\).
Otherwise, choose \(\delta_0>0\) such that
\([z_c-\delta_0,z_c+\delta_0]\) is disjoint from
\(\sigma(H_1(r))\).  In a fixed reference representation near
\(r\), norm-resolvent continuity ensures that this interval
remains disjoint from \(\sigma(H_1(\eta))\) for \(\eta\) close
to \(r\).  But \(\lambda(\eta)\to z_c\), so the eigenvalue
\(\lambda(\eta)\in\sigma(H_1(\eta))\) lies in this interval
for \(\eta<r\) sufficiently close to \(r\), a contradiction.

Apply the local Rellich construction at \((z_c,r)\), obtaining
branches \(\lambda_1,\ldots,\lambda_N\) on both sides of \(r\)
that list all sector-\(1\) eigenvalues near \(z_c\).
Since \(\lambda(\eta)\to z_c\), the incoming branch belongs to
this list on \((r-\delta,r)\).  By real analyticity, it agrees
throughout this overlap with one fixed Rellich branch
\(\lambda_j\).  Use \(\lambda_j\) to define its continuation
at \(r\) and to the right; the agreement on the overlap makes
this an analytic continuation.

There is a good point in the overlap, since the incoming bad set
is locally discrete.  Hence the corresponding \(g_j\) is nonzero
at that point and is not identically zero near \(r\).
Its zeros are isolated, so after shrinking \(\delta\),
\[
 g_j(\eta)\ne0,\qquad 0<|\eta-r|<\delta.
\]
By \eqref{equ:rellich-good-equivalence}, the extended branch is
good on this punctured neighborhood; only \(r\) itself may be
bad.  The same construction applies at a left interior endpoint.

\medskip\noindent
\textit{Step 4: continue over the full parameter interval.}
Consider all analytic continuations of the initial branch to
open intervals containing \(\eta_s\), each good except on a
locally discrete set.  Any two agree near \(\eta_s\), and hence,
by real analyticity, throughout their overlap.  Their union
therefore defines a single analytic branch on a maximal open
interval \(J\subset(0,\eta_{\mathrm{c}})\).  The bad set remains locally
discrete, since each point of \(J\) belongs to one of these
continuation intervals.  If \(J\) had an interior endpoint,
Step 3 would extend the branch beyond it while preserving this
property, contradicting maximality.  Consequently,
\[
 J=(0,\eta_{\mathrm{c}}).
\]

\medskip\noindent
\textit{Step 5: identify the exceptional set.}
Fix any interior parameter \(\eta_0\).  By real analyticity,
\(\lambda\) agrees near \(\eta_0\) with one of the local Rellich
branches \(\lambda_j\).  There is a nearby good point, so the
associated \(g_j\) is not identically zero.
Equation \eqref{equ:rellich-good-equivalence} identifies
\(\mathcal C_\tau\) locally with the zero set of \(g_j\).
It is therefore relatively closed and discrete, and its
intersection with any compact admissible interval is finite.
