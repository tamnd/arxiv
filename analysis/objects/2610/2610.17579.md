\label{prop:eta-obj-nonconvex}
Suppose that the distribution of $c(x_0)$ under $q_0$
admits a continuous density $h$. 
Define
\begin{align*}
    A_\lambda(\eta)
    &:= \mathbb{E}_{q_0}\left[ 
        e^{-c_{\lambda,\eta}(x_0)/\alpha}
        \mathds{1}\left\{ c(x_0)\leq\eta \right\}
     \right], \\
    B_\lambda(\eta)
    &:= \mathbb{E}_{q_0}\left[ 
        e^{-c_{\lambda,\eta}(x_0)/\alpha}
        \mathds{1}\left\{ c(x_0)>\eta \right\}
     \right],
\end{align*}
and assume that
$Z_\lambda(\eta):=A_\lambda(\eta)+B_\lambda(\eta)$
is finite for every $\eta\in\mathbb{R}$.
Then $G_\lambda$ in \eqref{eq:obj-eta} is convex
if and only if
\begin{equation}
\label{eq:eta-convexity-condition-general-lambda}
    e^{-\eta/\alpha}h(\eta)Z_\lambda(\eta)
    \geq
    \frac{\lambda}{\alpha(1-\beta)}
    A_\lambda(\eta)B_\lambda(\eta)
\end{equation}
for every $\eta\in\mathbb{R}$. Moreover, we have $G_\lambda'(\eta) =\lambda-\frac{\lambda}{1-\beta} \cdot    \frac{B_\lambda(\eta)}{Z_\lambda(\eta)}$ and $G_\lambda''(\eta)  \geq -
\frac{\lambda^2}{4\alpha(1-\beta)^2}$ for all $\eta\in\mathbb{R}$.

\label{thm:eta-optimization}
Suppose \Cref{assum:bounded-cost} holds.
Let $\eta_c \in [l,u]$ be arbitrary, and suppose that
$G_{\lambda,\kappa}$ is $m_\kappa$-strongly convex on
$[l,u]$ for some $m_\kappa>0$.

\begin{equation}
\label{eq:obj-L-Lkappa-def}
    L := \lambda\max\left\{1,\frac{\beta}{1-\beta}\right\},\quad L_\kappa := L + \kappa (u-l).
\end{equation}
For any $\delta\in(0,1)$ and sufficiently small
$\varepsilon_\eta>0$, define
\[
  K:=\Bigl\lceil\log_{3/2}\Bigl(\frac{8L_\kappa(u-l)}{m_\kappa\varepsilon_\eta^2}\Bigr)\Bigr\rceil,
  \ 
  J:=\Bigl\lceil\frac{32L(u-l)}{m_\kappa\varepsilon_\eta^2}\Bigr\rceil+1.
\]

Then, with probability at least $1-\delta$, the output $\hat\eta$ of \Cref{alg:onedopt} (with Lipschitz bound $L_\kappa$ and target error $m_\kappa \varepsilon_\eta^2/8$) satisfies
$|\hat\eta-\eta_\kappa^*|\le\varepsilon_\eta$
as long as
\[
  N\ \ge\ \frac{4\,C_G}{m_\kappa^2\,\varepsilon_\eta^4}\log\frac{2J}{\delta}
\]
where $C_G>0$ is a constant depending on
$\alpha,\beta,\lambda,l,u$. Furthermore, the algorithm calls the Monte Carlo oracle $\widehat G_{\lambda,\kappa,N}$ at most $1+2K$ times.

\label{thm:paired-value-estimation}
Suppose that \Cref{assum:bounded-cost} holds.
Define the loss $\widehat{\mathcal L}_{t,\eta}(g)
    :=\frac1N\sum_{i=1}^N
    \left( g(x_t^i)-Y_\eta(x_0^i) \right)^2$. For any $\delta\in(0,1)$ and sufficiently small

\[
    \widehat{\mathcal L}_{t,\eta}(\widehat f_{t,\eta})
    \leq \min_{g\in\mathcal F_t} \widehat{\mathcal L}_{t,\eta}(g) + \frac{\varepsilon_v^2}{8(1+\alpha^2)}e^{-\xi}
\]
Now construct value function estimates with the approximate ERM $\widehat f_{t,\eta}$ as follows:
\begin{align}
    \widehat V_{t,\eta,N}
    := a_\eta-\alpha\log\widehat f_{t,\eta},\ \forall t=1,\dots,T.\label{eq:value-function-estimate}
\end{align}
Then, with probability at least $1-\delta$, we have $\sup_{\eta\in[l,u]} \max_{t \in \{1,\dots,T\}} \left\lVert \widehat V_{t,\eta,N}-V_{t,\eta}^* \right\rVert_{L_2(q_t)} \le\varepsilon_v$, as long as
\[
    N\ge
    \frac{C}{\varepsilon_v^2}
    \log\frac{T}{\delta\varepsilon_v},
\]
where $C>0$ is a constant depending on $\alpha,\beta,\lambda,l,u,F$.

Writing $C:=c(x_0)$ and splitting the expectation at $C=\eta$, we have
\begin{align}
Z_{\lambda}(\eta) &= A_{\lambda}(\eta) + B_{\lambda}(\eta) \nonumber \\
&= \int_{-\infty}^{\eta} \exp\left(  -\frac{(1-\lambda)C+\lambda\eta}{\alpha} \right) h(C)\,dC + \int_{\eta}^{\infty} \exp\left(  -\frac{(1-\lambda)C +\lambda\eta +\frac{\lambda}{1-\beta}(C-\eta)}{\alpha} \right) h(C)\,dC.
\label{eq:Z-lambda-eta-decomposition}
\end{align}

Since $c(x_0)$ has continuous density $h$, differentiating the first term in \eqref{eq:Z-lambda-eta-decomposition} yields
\begin{align}
    &\frac{d}{d\eta} A_\lambda(\eta) \nonumber \\
    &=\frac{d}{d\eta} \mathbb{E}_{q_0} \left[  \exp\left(  -\frac{(1-\lambda)c(x_0)+\lambda\eta}{\alpha}  \right) \mathds{1}\left\{ c(x_0)\leq\eta \right\} \right] \nonumber \\
    &= \mathbb{E}_{q_0} \left[  \left(  \exp\left(  -\frac{(1-\lambda)c(x_0)+\lambda\eta}{\alpha} \right) \right)' \mathds{1}\left\{ c(x_0)\leq\eta \right\} \right] + \mathbb{E}_{q_0} \left[  \exp\left(  -\frac{(1-\lambda)c(x_0)+\lambda\eta}{\alpha}  \right) \left(  \mathds{1}\left\{ c(x_0)\leq\eta \right\}  \right)' \right]
    \nonumber\\
    &=
    -\frac{\lambda}{\alpha}
    \mathbb{E}_{q_0}
    \left[ 
        \exp\left( 
            -\frac{(1-\lambda)c(x_0)+\lambda\eta}{\alpha}
         \right)
        \mathds{1}\left\{ c(x_0)\leq\eta \right\}
     \right]
    +
    e^{-\eta/\alpha}h(\eta) \nonumber\\
    &= -\frac{\lambda}{\alpha} A_\lambda(\eta) + e^{-\eta/\alpha}h(\eta),
    \label{eq:lower-term-derivative}
\end{align}
where the derivative of the indicator function follows again by the sifting property of the delta function.

Similarly, differentiating the second term in \eqref{eq:Z-lambda-eta-decomposition} gives
\begin{align}
    &\frac{d}{d\eta}
    B_\lambda(\eta) \nonumber \\
    &=\frac{d}{d\eta}
    \mathbb{E}_{q_0}
    \left[ 
        \exp\left( 
            -\frac{(1-\lambda)c(x_0)+\lambda\eta
            +\frac{\lambda}{1-\beta}(c(x_0)-\eta)}{\alpha}
         \right)
        \mathds{1}\left\{ c(x_0)>\eta \right\}
     \right]
    \nonumber\\
    &=
    \mathbb{E}_{q_0}
    \left[ 
        \left( 
            \exp\left( 
                -\frac{(1-\lambda)c(x_0)+\lambda\eta
                +\frac{\lambda}{1-\beta}(c(x_0)-\eta)}{\alpha}
             \right)
         \right)'
        \mathds{1}\left\{ c(x_0)>\eta \right\}
     \right]
    \nonumber\\
    &\quad+
    \mathbb{E}_{q_0}
    \left[ 
        \exp\left( 
            -\frac{(1-\lambda)c(x_0)+\lambda\eta
            +\frac{\lambda}{1-\beta}(c(x_0)-\eta)}{\alpha}
         \right)
        \left( 
            \mathds{1}\left\{ c(x_0)>\eta \right\}
         \right)'
     \right]
    \nonumber\\
    &=
    \frac{\lambda\beta}{\alpha(1-\beta)}
    \mathbb{E}_{q_0}
    \left[ 
        \exp\left( 
            -\frac{(1-\lambda)c(x_0)+\lambda\eta
            +\frac{\lambda}{1-\beta}(c(x_0)-\eta)}{\alpha}
         \right)
        \mathds{1}\left\{ c(x_0)>\eta \right\}
     \right]
    -
    e^{-\eta/\alpha}h(\eta) \nonumber \\
    &= \frac{\lambda\beta}{\alpha(1-\beta)} B_\lambda(\eta) -
    e^{-\eta/\alpha}h(\eta).
    \label{eq:upper-term-derivative}
\end{align}
Adding \eqref{eq:lower-term-derivative} and \eqref{eq:upper-term-derivative}, the two boundary terms cancel, and hence
\begin{equation}
\label{eq:Z-lambda-eta-first-derivative}
    Z_{\lambda}'(\eta) = -\frac{\lambda}{\alpha} A_\lambda(\eta) + \frac{\lambda\beta}{\alpha(1-\beta)} B_\lambda(\eta).
\end{equation}
Differentiating \eqref{eq:Z-lambda-eta-first-derivative}
and using \eqref{eq:lower-term-derivative}
and \eqref{eq:upper-term-derivative} yields
\begin{align}
    &Z_{\lambda}''(\eta) \nonumber \\
    &= -\frac{\lambda}{\alpha} \left(  -\frac{\lambda}{\alpha} A_\lambda(\eta) + e^{-\eta/\alpha} h(\eta)  \right) + 
    \frac{\lambda\beta}{\alpha(1-\beta)} \left(  \frac{\lambda\beta}{\alpha(1-\beta)} B_\lambda(\eta) - e^{-\eta/\alpha}h(\eta)  \right) \nonumber \\
    &= \frac{\lambda^2}{\alpha^2} A_\lambda(\eta) + \frac{\lambda^2\beta^2}{\alpha^2(1-\beta)^2} B_\lambda(\eta) - \frac{\lambda}{\alpha(1-\beta)} e^{-\eta/\alpha}h(\eta). \label{eq:Z-lambda-eta-second-derivative}
\end{align}

Now, since $G_\lambda(\eta) = -\alpha \log Z_{\lambda}(\eta)$, we have 
\begin{align*}
G_\lambda'(\eta) =-\alpha\frac{Z'_{\lambda}(\eta)}{Z_{\lambda}(\eta)}=\lambda-\frac{\lambda}{1-\beta} \cdot    \frac{B_\lambda(\eta)}{Z_\lambda(\eta)}
\end{align*}
and
\begin{equation}
\label{eq:G-lambda-second-derivative-in-Z}
    G_\lambda''(\eta) = -\alpha \frac{ Z_{\lambda}''(\eta) Z_{\lambda}(\eta) -(Z_{\lambda}'(\eta))^2}{Z_{\lambda}(\eta)^2}.
\end{equation}
Substituting \eqref{eq:Z-lambda-eta-first-derivative} and \eqref{eq:Z-lambda-eta-second-derivative} into \eqref{eq:G-lambda-second-derivative-in-Z} and simplifying yields
\begin{equation}
\label{eq:G-lambda-second-derivative}
G_\lambda''(\eta) = \frac{\lambda}{(1-\beta)Z_{\lambda}(\eta)^2}
\left( 
e^{-\eta/\alpha}h(\eta)Z_{\lambda}(\eta) -
\frac{\lambda}{\alpha(1-\beta)}
A_\lambda(\eta)
B_\lambda(\eta)
 \right).
\end{equation}
Note that here the coefficient $\frac{\lambda}{(1-\beta)Z_{\lambda}(\eta)^2}$ is strictly positive. Hence, $G_\lambda(\eta)$ is convex if and only if
\[
e^{-\eta/\alpha}h(\eta)Z_{\lambda}(\eta)
\geq
\frac{\lambda}{\alpha(1-\beta)}
A_\lambda(\eta)
B_\lambda(\eta),\quad \forall \eta \in \mathbb{R},
\]
which is exactly \eqref{eq:eta-convexity-condition-general-lambda}.
Before we end the proof, we also note that, since the first term in \eqref{eq:G-lambda-second-derivative} is nonnegative,
\begin{equation}
\label{eq:G-weak-convex}
G_\lambda''(\eta) \geq -
\frac{\lambda^2}{\alpha(1-\beta)^2}
\frac{A_\lambda(\eta)
B_\lambda(\eta)}{Z_\lambda(\eta)^2} \geq -
\frac{\lambda^2}{4\alpha(1-\beta)^2}, 
\end{equation}
since $0 \leq (A-B)^2 = (A+B)^2 - 4 A B$ for $A,B \geq 0$. 
The proof is now complete.

fix $\kappa\geq0$ and $\eta_c\in\mathbb{R}$.

\propto
\mu_t(x_{t-1}\mid x_t)
e^{-V_{t-1,\eta_\kappa^*}^*(x_{t-1})/\alpha},
