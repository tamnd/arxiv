<!-- lemma-1: body_md -->
\label{prop:ball}
Let $T_x$ be a measurable bijection with measurable inverse, $\mathcal{U}_x(\varepsilon)$ be defined in \eqref{eq:base-ball}. Set $\mathbb P_x^T=(T_x)_\#\mathbb P_{b,x}$ and
\begin{equation}
\label{eq:metric}
 c_{T_x}(y,y')=c(T_x^{-1}(y),T_x^{-1}(y')).
\end{equation}
Then \eqref{eq:udt} can be equivalently expressed as
\begin{equation}
\label{eq:ball}
 \mathcal U_x^{\rm DT}(\varepsilon)
 =\{\mathbb Q:\mathsf D_{c_{T_x}}(\mathbb Q,\mathbb P_x^T)\le\varepsilon\}.
\end{equation}
<!-- end lemma-1 -->
<!-- theorem-1: body_md -->
\label{cor:lp-reduction}
Let $\mathcal Y=[a,b]$, $c(y,y')=|y-y'|$, and $\mathcal Z$ be polyhedral. Suppose $T_x$ is continuous piecewise affine with knots $a=\tau_0<\cdots<\tau_K=b$, and the decision loss function is
\[
\ell(z,y)=\max_{r\in[R]}\{\alpha_r^\top z+\beta_r y+\gamma_r\}.
\]
For source scenarios $\nu_1,\ldots,\nu_m\sim\mathbb P_{b,x}$, define $ \mathcal K_i := \{\tau_0,\ldots,\tau_K,\nu_i\}. $ Then, the sample average approximation of \eqref{eq:dual} is a finite linear program
\begin{equation}
\label{eq:finite-lp}
\begin{aligned}
\min_{\substack{z\in\mathcal Z,\lambda\ge0\\h\in\mathbb R^m}}
&\quad
\lambda\varepsilon+\frac{1}{m}\sum_{i=1}^m h_i\\
\mathrm{s.t.}
&\quad
h_i
\ge
\alpha_r^\top z
+\beta_r T_x(v)
+\gamma_r
-\lambda|v-\nu_i|,\\
&\quad
\forall v\in\mathcal K_i,\
\forall r\in[R],\
\forall i\in[m].
\end{aligned}
\end{equation}

<!-- lemma-2: body_md -->
\label{prop:minimum-displacement}
Fix $x$ and suppose the base and true conditional CDFs are continuous and strictly increasing on $\mathcal Y=[a,b]$. Define the base-model conditional PIT and its CDF by
\begin{equation}
\label{eq:pit-map}
 U=F_b(Y\mid X),\quad
 G_x^\star(u)=\Pr(U\le u\mid X=x).
\end{equation}
Then an optimal solution for \eqref{eq:center-ot} is
\begin{equation}
\label{eq:tx-oracle}
 T_x^\star(y)
 =F_b^{-1}\!\left((G_x^\star)^{-1}(F_b(y\mid x))\mid x\right),
\end{equation}
where $G_x^\star:[0,1]\to[0,1]$ is the conditional PIT-CDF.
<!-- end lemma-2 -->
<!-- assumption-1: body_md -->
\label{ass:diagnostic}
$~$
    \begin{enumerate}
\item[($a$)] The outcome space is a bounded interval $\mathcal Y=[a,b]$, with $\Delta_{\mathcal Y}=b-a<\infty$. \item[($b$)] For each fixed context $x$, the base CDF $F_b(\cdot\mid x)$ and population diagnostic CDF $G_x^\star$ are continuous and strictly increasing, and the fitted diagnostic $\widehat G_x$ is an increasing homeomorphism of $[0,1]$. \item[($c$)] The calibration pairs are i.i.d. and independent of the data used to fit the base model.
    \end{enumerate}

<!-- theorem-2: body_md -->
\label{thm:radius}
Under Assumption~\ref{ass:diagnostic}, set
    \begin{equation}
    \label{eq:radius}
     \varepsilon=\Delta_{\mathcal Y}
     \left[r_n(x,\alpha/2)+\sqrt{\frac{\log(4/\alpha)}{2m}}\right],
    \end{equation}
then for $\widehat{\mathcal U}_x^{\rm DT}(\cdot)$ defined in \eqref{eq:udt-hat}, there is
    \begin{equation*}
        \Pr\left\{ \mathbb P_x^\star \in \widehat{\mathcal U}_x^{\rm DT}(\varepsilon) \right\} \geq 1- \alpha.
    \end{equation*}
<!-- end theorem-2 -->
<!-- assumption-2: body_md -->
\label{ass:loss}
The feasible set $\mathcal Z$ is nonempty and compact, $\ell$ is continuous in $(z,y)$, and there exists $L_y<\infty$ such that
    \begin{equation}
        \label{eq:loss-lipschitz-theory}
        \left|\ell(z,y)-\ell(z,y')\right|
        \le L_y\left|y-y'\right|
    \end{equation}
for all $z\in\mathcal Z$ and $y,y'\in\mathcal Y$.
<!-- end assumption-2 -->
<!-- theorem-3: body_md -->
\label{thm:value-certificate}
Let $\widehat z$, $\widehat v$ be defined in \eqref{eq:robust-value-estimator}, $v^\star$ be defined in \eqref{eq:pop-opt-val}. For $\varepsilon$ defined in \eqref{eq:radius}, write
    $$
     \omega_T(h)=\sup_{|s-s'|\le h}|T(s)-T(s')|,\quad B_{n,m}=\varepsilon+\overline\omega_{\widehat T_x}(\varepsilon),
    $$
where $\overline\omega_T$ is the least nondecreasing concave majorant of $\omega_T$, extended constantly beyond $\Delta_{\mathcal Y}$. Under Assumptions~\ref{ass:diagnostic} and \ref{ass:loss}, with probability at least $1-\alpha$,
    \begin{equation}
    \label{eq:regret}
     0\le
     \mathcal{L}^\star(\widehat z) - v^\star
     \le\widehat v-v^\star\le L_yB_{n,m}.
    \end{equation}

<!-- proposition-1: body_md -->
\label{prop:geometry-gain}
If
\begin{equation}
\label{eq:geometry-condition}
 \varepsilon_C<\frac{c_pb-v_0}{c_s},
 \qquad \gamma_T(z_0)\varepsilon_T<\varepsilon_C,
\end{equation}
then
\begin{equation}
\label{eq:geometry-gain}
 v_C(\varepsilon_C)-v_{\rm DT}(\varepsilon_T)
 \ge c_s[\varepsilon_C-\gamma_T(z_0)\varepsilon_T]>0.
\end{equation}
The comparison uses any radii for which the two ambiguity sets are valid on a common event. In particular, \Cref{app:containment} supplies the same radius for both sets, in which case $\gamma_T(z_0)<1$ suffices together with the first condition.
