<!-- proposition-1: body_md -->
\label{prop:ternaryGCD_to_binaryGCD}
  Let $x,y,z\in\mathbb{Z}$ with $x\ne 0$. There exists $m\in\mathbb{N},m\ll\log^2|x|$ such that
    \[\gcd(x,y+mz)=\gcd(x,y,z).\]
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
If $x=\pm1$ then the result holds with $m=0$. Suppose instead that $|x|>1$ and let $p_1^{r_1}\cdots p_k^{r_k}$ be the prime factorisation of $x$. Because $\gcd(x,y,z)\mid x$ we have 
    \begin{equation}
        \gcd(x,y,z)=p_1^{s_1}\cdots p_k^{s_k} \quad\text{for some}\quad s_1,\dotsc,s_k\in\mathbb{N} \quad\text{where each}\quad s_i\leqslant r_i.
    \end{equation}
    Similarly, because $\gcd(x,y,z)\mid \gcd(x,y+mz)\mid x$ for all $m\in\mathbb{N}$ we have
    \[
        \gcd(x,y+mz)=p_1^{t_{m,1}}\cdots p_k^{t_{m,k}} \quad\text{for some}\quad t_{m,1},\dotsc,t_{m,k}\in\mathbb{N} \quad\text{where each}\quad s_i\leqslant t_{m,i}\leqslant r_i.
    \] 
    Hence it suffices to show that there exists $m\ll\log^2|x|$ such that for all $i\in[k]$ we have $t_{m,i}=s_i$ (or, equivalently, $p_i^{s_i+1}\nmid x$ or $p_i^{s_i+1}\nmid y+mz$). There are three cases for $i\in[k]$.\medskip
    \begin{description}
        \item[Case 1] If $s_i=r_i$ then $p_i^{s_i+1}\nmid x$ for all $m\in\mathbb{N}$.\smallskip
        \item[Case 2] Suppose $s_i<r_i$ and $p_i^{s_i+1}\mid z$. The former implies $p_i^{s_i+1}\mid x$, which combined with the latter implies $p_i^{s_i+1}\nmid y$ since $p_i^{s_i+1}$ is not a factor of $\gcd(x,y,z)$. Therefore $y+mz\equiv_{p_i^{s_i+1}}y\not\equiv_{p_i^{s_i+1}}0$ and thus $p_i^{s_i+1}\nmid y+mz$ for all $m\in\mathbb{N}$ in this case.\smallskip
        \item[Case 3] Suppose $s_i<r_i$ and $p_i^{s_i+1}\nmid z$. Then $y=a_ip_i^{s_i}$ and $z=b_ip_i^{s_i}$ for some $a_i,b_i\in\mathbb{Z}$ with $p_i\nmid b_i$. This means $b_i$ has a multiplicative inverse $b_i^{-1}$ modulo $p_i$ and thus
    \begin{equation}
    \begin{array}{rclll}
         m \not\equiv_{p_i} -b_i^{-1}a_i  \quad\Rightarrow\quad a_i+mb_i \not\equiv_{p_i} 0 & \Rightarrow & a_i+mb_i \ne qp_i & & \text{for all $q\in\mathbb{Z}$} \\
         & \Rightarrow & a_ip_i^{s_i}+mb_ip_i^{s_i} \ne qp_i^{s_i+1} & & \text{for all $q\in\mathbb{Z}$} \\
         & \Rightarrow & y+mz \ne qp_i^{s_i+1} & & \text{for all $q\in\mathbb{Z}$} \\
         & \Rightarrow & p_i^{s_i+1}\nmid y+mz.
    \end{array}
    \end{equation}
    \end{description}
    \noindent In the first two cases we have $p_i^{s_i+1}\nmid x$ or $p_i^{s_i+1}\nmid y+mz$ for all $m\in\mathbb{N}$, so we need only find $m\ll\log^2|x|$ such that $m \not\equiv_{p_i} -b_i^{-1}a_i$ for all $i\in I:=\{i\in[k]:s_i<r_i \ \text{and} \  p_i^{s_i+1}\nmid z\}$ as per the third case. By the CRT there exists $n\in\mathbb{N}$ such that $-n\equiv_{p_i}-b_i^{-1}a_i$ for all $i\in I$. Choose
    \begin{equation}
        m=\min\{m\in\mathbb{N}:p_i\nmid n+m \text{ for all $i\in I$}\}
    \end{equation}
    and note that $m\leqslant C(|I|)\leqslant C(\omega(x))\ll\log^2|x|$ by \cref{lem:logSquared}. Then
    \[
        n+m\not\equiv_{p_i} 0 \quad\Rightarrow\quad m\not\equiv_{p_i}-n\equiv_{p_i}-b_i^{-1}a_i \qquad\text{for all}\ i\in I.\qedhere
    \]

\label{thm:EPAD_NEXP}
If $\exists\bm{x}\in\mathbb{Z}^n:\Phi(\bm{x})$ is a sentence in $\ensuremath{\textsc{EPAD}}\xspace$ then there exists a solution $\bm{a}\in\mathbb{Z}^n$ to $\Phi$ such that $\|\bm{a}\|\leqslant 2^{poly(\|\Phi\|)}$.

<!-- lemma-3: body_md -->
\label{lem:EPAG-C_NEXP}
    If $\exists\bm{x}\in\mathbb{Z}^n:\Phi(\bm{x})$ is a sentence in $\ensuremath{\textsc{EPAG-N}}\xspace$ then there exists a solution $\bm{a}\in\mathbb{Z}^n$ to $\Phi$ such that $\|\bm{a}\|\leqslant 2^{poly(\|\Phi\|)}$.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
We prove the claim by showing that there exists a sentence $\exists\bm{x}\in\mathbb{Z}^n,\exists\bm{y}\in\mathbb{Z}^m:\Psi(\bm{x},\bm{y})$ in $\ensuremath{\textsc{EPAD}}\xspace$ such that $\bm{a}\in\mathbb{Z}^n$ is a solution to $\Phi$ if and only if $(\bm{a},\bm{b})\in\mathbb{Z}^{n}\times\mathbb{Z}^m$ is a solution to~$\Psi$ for some $\bm{b}\in\mathbb{Z}^m$, and $\|\Psi\|=poly(\|\Phi\|)$.

    First, let $\Phi'$ be the logically equivalent formula obtained from $\Phi$ by using De Morgan's law to push all negations inwards until they apply directly to atoms. Note that by definition no negations apply to atoms of the form $\gcd(a_1,\dots,a_p)\sim d$, $p>2$, in $\Phi'$.

    Next we show that each literal of the form
    \begin{equation}\label{eqn:EPAG-C_atoms}
        \neg(\gcd(a,b)\sim d) \qquad\text{and}\qquad \gcd(a_1,\dotsc,a_p)\sim d \quad\text{where}\quad p>2
    \end{equation}
    can be replaced by \ensuremath{\textsc{EPAD}}\xspace formulae. For this, we note the following equivalences:
    \begin{equation}\label{eqn:EPAG-C_EPAD_equiv1}
    \begin{array}{rcl}
        \neg(\gcd(a,b)= d) & \Leftrightarrow &  \exists c\in\mathbb{Z}: \neg(c=d)\land\gcd(a,b)=c, \\
        \neg(\gcd(a,b)\leqslant d) & \Leftrightarrow &  \exists c\in\mathbb{Z}: \neg(c\leqslant d)\land\gcd(a,b)=c, \\
        \gcd(a_1,\dotsc,a_p)\leqslant d & \Leftrightarrow &  \exists c\in\mathbb{Z}: c\leqslant d\land\gcd(a_1,\dotsc,a_p)=c.
    \end{array}
    \end{equation}
    Next, recall that if $c$ is a common divisor of $a_1,\dotsc,a_p\in\mathbb{Z}$ then $c$ divides any linear combination of $a_1,\dotsc,a_p$. Hence if $d\in\mathbb{N}$ is such that $d\mid a_i$ for all $i\in[p]$ and if there exist $m_1,\dotsc,m_p\in\mathbb{Z}$ for which $m_1a_1+\cdots+m_pa_p=d$ then $d=\gcd(a_1,\dotsc,a_p)$. Bezout's identity gives the converse. That is,
    \begin{equation}\label{eqn:EPAG-C_EPAD_equiv2}
    \begin{array}{rcl}
         \gcd(a_1,\dotsc,a_p)=d & \Leftrightarrow & \exists u_1,\dotsc,u_p\in\mathbb{Z}:\bigwedge_{i=1}^p d\mid a_i \land \bigwedge_{i=1}^p a_i\mid u_i \land \sum_{i=1}^p u_i=d. 
    \end{array}
    \end{equation}
    By applying to $\Phi'$ the equivalences in \eqref{eqn:EPAG-C_EPAD_equiv1}, then those in \eqref{eqn:EPAG-C_EPAD_equiv2} (introducing fresh variables in each case) we obtain a logically equivalent formula $\Phi''$ whose atoms are all \ensuremath{\textsc{EPAD}}\xspace atoms. We then pull out the newly introduced quantifiers (which remain existential since negations apply only to atoms in $\Phi''$) to obtain the sentence $\exists\bm{x}\in\mathbb{Z}^n,\exists\bm{y}\in\mathbb{Z}^m:\Psi(\bm{x},\bm{y})$ where $\bm{y}$ is a tuple containing the newly introduced variables, which is in $\ensuremath{\textsc{EPAD}}\xspace$ since $\Phi$ is feasible by assumption. Clearly the sentence $\exists\bm{x}\in\mathbb{Z}^n,\exists\bm{y}\in\mathbb{Z}^m:\Psi(\bm{x},\bm{y})$ fulfills its requirements.

    By \cref{thm:EPAD_NEXP}, there exists a solution $(\bm{a},\bm{b})$ to $\Psi$ with $\|\bm{a}\|,\|\bm{b}\|\leqslant2^{poly(\|\Psi\|)}=2^{poly(\|\Phi\|)}$. Therefore $\bm{a}$ is a solution to $\Phi$ with $\|\bm{a}\|\leqslant2^{poly(\|\Phi\|)}$, as required.
<!-- end proof-3 -->
<!-- theorem-2: body_md -->
\label{thm:EPAG_NP}
If $\exists\bm{x}\in\mathbb{Z}^n:\Phi(\bm{x})$ is a sentence in $\ensuremath{\textsc{EPAG}}\xspace$ then there exists a solution $\bm{a}\in\mathbb{Z}^n$ to $\Phi$ such that $\|\bm{a}\|\leqslant poly(\|\Phi\|)$.
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm:IP-GCD_NP}
    If $\exists\bm{x}:\Phi(\bm{x})\in\ensuremath{\textsc{IP-GCD}}\xspace$ then there exists a solution $\bm{a}\in\mathbb{Z}^n$ to $\Phi$ such that $\|\bm{a}\|\leqslant poly(\|\Phi\|)$.
<!-- end theorem-3 -->
<!-- proof-4: body_md -->
We begin by transforming $\Phi$ into an equivalent formula $\Psi$ without negations and with atoms of the form
	\begin{equation}\label{eqn:PAG_NP}
		f(\bm{x})\leqslant 0 \qquad\text{or}\qquad \gcd(f(\bm{x}),g(\bm{x}))\approx d
	\end{equation}
	where ${\approx}\in\{=,\ne,\geqslant,\leqslant\}$ and $d\in\mathbb{N}_+$. We do this by using De Morgan's law to push negations inward until they apply directly to atoms, then using the following logical equivalences:
    \begin{equation}
        \begin{array}{rcl}
            a = b & \Leftrightarrow &  a-b\leqslant 0\land b-a\leqslant 0, \\
            \neg(a=b) & \Leftrightarrow & a-b+1\leqslant0\lor b-a+1\leqslant0,\\
            a\leqslant b & \Leftrightarrow & a-b\leqslant0, \\
            \neg(a\leqslant b) & \Leftrightarrow & b-a+1\leqslant0, \\
            \neg(\gcd(a,b)=d) & \Leftrightarrow & \gcd(a,b)\ne d, \\
            \neg(\gcd(a,b)\leqslant d) & \Leftrightarrow & \gcd(a,b)\geqslant d+1.
        \end{array}
    \end{equation}
    Clearly $\|\Psi\|$ is linear in $\|\Phi\|$.

    Now, suppose $\bm{a}\in\mathbb{Z}^n$ is a solution to $\Phi$ and thus also to $\Psi$. The formula $\Psi$ consists of a finite number of nested conjuctive and disjunctive clauses. That is, 
    \begin{equation}
        \Psi=\bigwedge_{i=1}^{k_1}\bigvee_{j=1}^{\ell_{1,i}}\Theta_{1,i,j} \quad\text{where each}\quad \Theta_{1,i,j}=\bigwedge_{i=1}^{k_2}\bigvee_{j=1}^{\ell_{2,i}}\Theta_{2,i,j} \quad\text{where each}\quad \Theta_{2,i,j}=\cdots
    \end{equation}
    and so on, until eventually
    \begin{equation}
        \Theta_{m-1,i,j}=\bigwedge_{i=1}^{k_m}\bigvee_{j=1}^{\ell_{m,i}}\Theta_{m,i,j} \quad\text{where each $\Theta_{m,i,j}$ is an atom of the form \eqref{eqn:PAG_NP}.}
    \end{equation}
    Consider the outermost layer of disjunctive clauses. Because $\Psi(\bm{a})$ is true, for each $i\in[k_1]$ we have that $\Theta_{1,i,r_i}(\bm{a})$ is true for some $r_{i}\in[\ell_{1,i}]$. Therefore $\Psi'(\bm{a}):=\bigwedge_{i=1}^{k_1}\Theta_{1,i,r_i}(\bm{a})$ is also true. We continue moving inwards, replacing each disjunctive clause $\bigvee_{j=1}^{\ell_{x,i}}\Theta_{x,i,j}$ with a single formula $\Theta_{x,i,j}$ from that clause such that $\Theta_{x,i,j}(\bm{a})$ is true, until we obtain a formula~$\Upsilon$ which is a conjunction of atoms of the form \eqref{eqn:PAG_NP}. Note that $\exists\bm{x}:\Upsilon(\bm{x})$ is an $\ensuremath{\textsc{IP-GCD}}\xspace$ sentence with $\|\Upsilon\|=poly(\|\Phi\|)$. By \cref{thm:IP-GCD_NP} there exists a solution $\bm{b}\in\mathbb{Z}^n$ to $\Upsilon$ with $\|\bm{b}\|\leqslant poly(\|\Upsilon\|)=poly(\|\Phi\|)$. By construction, $\bm{b}$ is also a solution to $\Psi$ (and thus to $\Phi$ as well), completing the proof.
<!-- end proof-4 -->
<!-- proof-5: body_md -->
Let $\exists\bm{x}\in\mathbb{Z}^n:\Phi(\bm{x})$ be a sentence in $\ensuremath{\textsc{EPAG-N}}\xspace$ with 
    \begin{equation}
        \Phi(\bm{x}):= \Psi(\bm{x}) \land \bigwedge_{i=1}^k \gcd\!\big(f_{i,1}(\bm{x}),\dotsc,f_{i,p_i}(\bm{x})\big)\sim_i d_i.
    \end{equation}
    where $\Psi(\bm{x})$ is a quantifier-free \ensuremath{\textsc{EPAG}}\xspace formula. Write $N=\|\Phi\|$. By \cref{lem:EPAG-C_NEXP} there exists a solution $\bm{a}\in\mathbb{Z}^n$ to $\Phi$ with $\|\bm{a}\|\leqslant 2^{poly(N)}$. We will construct an \ensuremath{\textsc{EPAG}}\xspace formula $\Theta$ from~$\Phi$ such that: $\bm{a}$ is a solution to $\Theta$, any solution $\bm{b}\in\mathbb{Z}^n$ to $\Theta$ is a solution to $\Phi$, and with $\|\Theta\|=poly(N)$.
    
    Let $i\in[k]$. There are two cases.\medskip
    \begin{description}
        \item[Case 1] Suppose $f_{i,j}(\bm{a})=0$ for all $j\in[p_i]$. There are two subcases.\smallskip
        \item[Subcase 1.1] If $\sim_i$ denotes $=$ then $\gcd\!\big(f_{i,1}(\bm{a}),\dotsc,f_{i,p_i}(\bm{a})\big)=d_i$ is false since $d_i>0$. This contradicts $\bm{a}$ being a solution to $\Phi$, so we can exclude this case.\smallskip
        \item[Subcase 1.2] If $\sim_i$ denotes $\leqslant$ then we use the substitution
        \begin{equation}
            \gcd\!\big(f_{i,1}(\bm{x}),\dotsc,f_{i,p_i}(\bm{x})\big)\leqslant d_i \quad\rightarrow\quad \bigwedge_{j=1}^{p_i}f_{i,j}(\bm{x})=0.
        \end{equation}
        By assumption, $\bm{a}$ is a solution to $\bigwedge_{j=1}^{p_i}f_{i,j}(\bm{x})=0$ in this case, and if $\bm{b}\in\mathbb{Z}^n$ is a solution to $\bigwedge_{j=1}^{p_i}f_{i,j}(\bm{x})=0$ then $\gcd\!\big(f_{i,1}(\bm{b}),\dotsc,f_{i,p_i}(\bm{b})\big)=0\leqslant d_i$.\smallskip
        \item[Case 2] Suppose $f_{i,j}(\bm{a})\ne0$ for some $j\in[p_i]$. Without loss of generality we assume $f_{i,1}(\bm{a})\ne0$. By \cref{thm:MAINC_k_aryGCD_to_binaryGCD}, there exists
        \begin{equation}
            g_i(\bm{x}):=f_{i,2}(\bm{x}) + m_{i,3}f_{i,3}(\bm{x}) + \cdots + m_{i,p_i}f_{i,p_i}(\bm{x})
        \end{equation}
        such that 
        $\gcd\!\big(f_{i,1}(\bm{a}), g_i(\bm{a})\big) = \gcd\!\big(f_{i,1}(\bm{a}),\dotsc,f_{i,p_i}(\bm{a})\big)$ and where each
        \begin{equation}
    		m_{i,j} \ll \log^2|f_{i,1}(\bm{a})| \ \ \Rightarrow\ \  \|m_{i,j}\| \leqslant poly\big(\log\|f_{i,1}(\bm{a})\|\big) \leqslant poly\big(\log 2^{poly(N)}\big) = poly(N),
    	\end{equation}
        implying $\|g_i(\bm{x})\|\leqslant poly(N)$. We consider two subcases.\smallskip
        \item[Subcase 2.1] If $\sim_i$ denotes $\leqslant$ then we use the substitution
        \begin{equation}
            \gcd\!\big(f_{i,1}(\bm{x}),\dotsc,f_{i,p_i}(\bm{x})\big)\leqslant d_i \quad\rightarrow\quad \gcd\!\big(f_{i,1}(\bm{x}), g_i(\bm{x})\big) \leqslant d_i.
        \end{equation}
        Then $\bm{a}$ remains a solution to the latter. Conversely, if $\bm{b}\in\mathbb{Z}^n$ is a solution to $\gcd\!\big(f_{i,1}(\bm{x}), g_i(\bm{x})\big) \leqslant d_i$ then there exists $c\leqslant d_i$ such that $\gcd\!\big(f_{i,1}(\bm{b}), g_i(\bm{b})\big)=c$. By Bezout's identity there exists $s,t\in\mathbb{Z}$ such that
        \begin{equation}
            c \ = \ sf_{i,1}(\bm{b})+tg_i(\bm{b}) \ = \ sf_{i,1}(\bm{b}) +tf_{i,2}(\bm{b}) + sm_{i,3}f_{i,3}(\bm{b}) + \cdots + sm_{i,p_i}f_{i,p_i}(\bm{b}).
        \end{equation}
        But this implies $\gcd\!\big(f_{i,1}(\bm{b}),\dotsc,f_{i,p_i}(\bm{b})\big)\mid c\leqslant d_i$. Hence $\bm{b}$ is a solution to 
        \begin{equation}
            \gcd\!\big(f_{i,1}(\bm{x}),\dotsc,f_{i,p_i}(\bm{x})\big)\leqslant d_i.
        \end{equation}
        \item[Subcase 2.2] If $\sim_i$ denotes $=$ then we use the substitution
        \[
            \gcd\!\big(f_{i,1}(\bm{x}),\dotsc,f_{i,p_i}(\bm{x})\big)= d_i \quad\rightarrow\quad \gcd\!\big(f_{i,1}(\bm{x}), g_i(\bm{x})\big) = d_i\land\bigwedge_{j=1}^{p_i}\gcd(f_{i,j}(\bm{x}),d_i)=d_i.
        \]
        Note that $\gcd(f_{i,j}(\bm{x}),d_i)=d_i\ \Leftrightarrow\ d_i\mid f_{i,j}(\bm{x})$. As in the previous case, $\bm{a}$ remains a solution to the substituted formula. We also have that any solution $\bm{b}\in\mathbb{Z}^n$ to the substituted formula satisfies $\gcd\!\big(f_{i,1}(\bm{b}),\dotsc,f_{i,p_i}(\bm{b})\big)\mid d_i$ and $d_i\mid f_{i,j}(\bm{b})$ for all $j\in[p_i]$ and is thus also a solution to $\gcd\!\big(f_{i,1}(\bm{x}),\dotsc,f_{i,p_i}(\bm{x})\big)= d_i$.
    \end{description}
    
    Let $\Theta$ be the formula obtained from $\Phi$ by applying the above substitutions for each $i\in[k]$. Then $\exists\bm{x}\in\mathbb{Z}^n:\Theta(\bm{x})$ is a sentence in $\ensuremath{\textsc{EPAG}}\xspace$ since $\bm{a}$ is a solution to $\Theta$, and $\|\Theta\|=poly(N)$. By \cref{thm:EPAG_NP}, there exists a solution $\bm{b}\in\mathbb{Z}^n$ to $\Theta$ with $\|\bm{b}\|\leqslant poly(N)$. But any solution to~$\Theta$ is also a solution to $\Phi$. 


    It follows immediately that deciding membership in \ensuremath{\textsc{EPAG-N}}\xspace is in \ensuremath{\mathsf{NP}}\xspace: given a $\Phi$ of size~$N$ we can guess a solution $\|\bm{b}\|\leqslant poly(N)$ and verify it by evaluating the truth value of~$\Phi(\bm{b})$ in $poly(N)$ time. \ensuremath{\mathsf{NP}}\xspace-completeness then follows from the hardness of \ensuremath{\textsc{EPA}}\xspace.

($\Rightarrow$): If $\operatorname{Inv}\langle B\rangle=\mathcal{B}$ then there exist elements $x_1,\dotsc,x_\ell\in {B}^{\pm1}$ such that

		x_1\cdots x_\ell = (1,0).
	\end{equation}

	\begin{equation}
		\rho(bc) \ = \ \rho(c) - \lambda(c) + \max\big(\rho(b),\lambda(c)\big) \ \geqslant \ \rho(c) - \lambda(c) + \lambda(c) \ = \ \rho(c);
	\end{equation}

	\end{equation}

Write $\bm{X}=\big(X\cup \widehat{X}\big)\cup W_{\Theta(R)}$. Suppose $\tau$ is a solution to $R$ over $\mathcal{B}$. By \cref{lem:rem_inv_ele_in_eqns,lem:split_inv_eqns} there is a corresponding solution 
	\begin{equation}
		\sigma:\bm{X}\rightarrow \mathcal{B} \quad\text{satisfying}\quad x\sigma=x\tau \quad\text{to $\Xi(\Theta(R))$.}
	\end{equation}
	For brevity, we will write $\overline{x}=x\overline\sigma$ for $x\in \bm{X}$. Define the mapping
	\begin{equation}
		\varsigma:V\rightarrow\mathbb{N}:\left\{\begin{array}{ll}
			x_\lambda\mapsto \lambda(\overline{x}) \\
			x_\rho\mapsto \rho(\overline{x})
		\end{array}\right.
		\quad\text{for}\quad x\in \bm{X} \quad\text{and write}\quad \widetilde{v}=v\varsigma \quad\text{for}\quad v\in V
	\end{equation}
	Hence $(\widetilde{x}_\lambda,\widetilde{x}_\rho)=\overline{x}$ for all $x\in\bm{X}$. Each $\eta\in\Xi(\Theta(R))$ falls into one of three cases.\medskip
	\begin{description}
		\item[Case 1] If $\eta=(x,1)$ for some $x\in\bm{X}$ then $(\widetilde{x}_\lambda,\widetilde{x}_\rho) = \overline{x} = \overline{1} = (0,0)$.
		\item[Case 2] If $\eta=(x,y)$ for some $x,y\in\bm{X}$ then $(\widetilde{x}_\lambda,\widetilde{x}_\rho)= \overline{x} = \overline{y} = (\widetilde{y}_\lambda,\widetilde{y}_\rho)$.
		\item[Case 3] If $\eta=(x,yz)$ for some $x,y,z\in \bm{X}$ then
		\begin{equation}
			\setlength{\arraycolsep}{3pt}
			\begin{array}{rclll}
				(\widetilde{x}_\lambda,\widetilde{x}_\rho) & = & \overline{x} \ = \ \overline{y}\,\overline{z} \\[1mm]
				& = & \Big(\lambda(\overline{y})-\rho(\overline{y})+\max\big(\rho(\overline{y}), \lambda(\overline{z})\big),\ \rho(\overline{z})-\lambda(\overline{z})+\max\big(\rho(\overline{y}),\lambda(\overline{z})\big)\Big) \\[1.5mm]
				& = & \big(\widetilde{y}_\lambda-\widetilde{y}_\rho+\max(\widetilde{y}_\rho, \widetilde{z}_\lambda),\ \widetilde{z}_\rho-\widetilde{z}_\lambda+\max(\widetilde{y}_\rho,\widetilde{z}_\lambda)\big) \\[1.5mm]
				& = & \left\{\begin{array}{lll}
					(\widetilde{y}_\lambda,\ \widetilde{z}_\rho-\widetilde{z}_\lambda+\widetilde{y}_\rho) & & \text{if $\widetilde{y}_\rho\geqslant \widetilde{z}_\lambda$} \\[1mm]
					(\widetilde{y}_\lambda-\widetilde{y}_\rho+\widetilde{z}_\lambda,\ \widetilde{z}_\rho) & & \text{if $\widetilde{y}_\rho< \widetilde{z}_\lambda$.}
				\end{array}\right.
			\end{array}
		\end{equation}
	\end{description}
	In each case, the assignment $\varsigma$ satisfies $\psi(\eta)$. Clearly $v\geqslant0$ for all $v\in V$ since $\operatorname{im}(\varsigma)=\mathbb{N}$. Therefore $\varsigma$ satisfies $\Psi(\Xi(\Theta(R)))$.
	
	For the converse, suppose $\varsigma$ is a satisfying assignment for $\Psi(\Xi(\Theta(R)))$. Define the mapping
	\begin{equation}
		\sigma:\bm{X}\rightarrow \mathcal{B}:x\mapsto (\widetilde{x}_\lambda,\widetilde{x}_\rho) \quad\text{where}\quad \widetilde{v}=v\varsigma \quad\text{for}\quad v\in V. 
	\end{equation}
    Note $\widetilde{v}\in\mathbb{N}$ for all $v\in V$ since $\varsigma$ is a solution to $\Psi(\Xi(\Theta(R)))$, so we indeed have $\operatorname{im}(\sigma)\subseteq\mathbb{N}\times\mathbb{N}$. As before, we write $\overline{x}=x\overline\sigma$ for $x\in\bm{X}$ and consider three cases for $\eta\in\Xi(\Theta(R))$. \medskip
	\begin{description}
		\item[Case 1] If $\eta=(x,1)$ then the truth of $\psi(\eta)$ implies $\overline{x}=(\widetilde{x}_\lambda,\widetilde{x}_\rho)=(0,0)=\overline{1}$.
		\item[Case 2] If $\eta=(x,y)$ then $\overline{x}=(\widetilde{x}_\lambda,\widetilde{x}_\rho)=(\widetilde{y}_\lambda,\widetilde{y}_\rho)=\overline{y}$.
		\item[Case 3] If $\eta=(x,yz)$ then $\overline{x} = \big(\widetilde{y}_\lambda-\widetilde{y}_\rho+\max(\widetilde{y}_\rho,\widetilde{z}_\lambda), \widetilde{z}_\rho-\widetilde{z}_\lambda+\max(\widetilde{y}_\rho,\widetilde{z}_\lambda)\big) = \overline{y}\,\overline{z}$.
	\end{description} \medskip
	Therefore $\sigma$ is a solution for $\Xi(\Theta(R))$ over $\mathcal{B}$. By \cref{lem:split_inv_eqns}, $\sigma|_{X\cup \widehat{X}}$ is a solution to $\Theta(R)$, which by \cref{lem:rem_inv_ele_in_eqns} implies $\tau=\big(\sigma|_{X\cup \widehat{X}}\big)\big|_X=\sigma|_{X}$ is a solution to $R$.

<!-- lemma-8: body_md -->
\label{thm:epi_iff_PA-GCD-formula}
	Let $\mathcal{I}=\operatorname{Inv}\left\langle{X}\mid{R}\right\rangle$. There exists an epimorphism $\mathcal{I}\rightarrow\mathcal{B}$ if and only if there exists a satisfying assignment for the formula $\Phi(X,R)$.
<!-- end lemma-8 -->
<!-- proof-12: body_md -->
Suppose there exists an epimorphism $\mathcal{I}\rightarrow\mathcal{B}$. By \cref{thm:von-dyck}(ii), this implies that there is a solution $\tau:X\rightarrow\mathcal{B}$ such that $\operatorname{Inv}\langle X\tau\rangle=\mathcal{B}$. By \cref{thm:eqn_sol_iff_formula_satisfied}, there exists a satisfying assignment $\varsigma:V\rightarrow\mathbb{N}$ for $\Psi(\Xi(R))$ such that $(x_\lambda\varsigma,x_\rho\varsigma)=x\tau$. By \cref{thm:gen_set_B}(i), $\operatorname{Inv}\langle X\tau\rangle=\mathcal{B}$ implies that there exists $x\in X$ such that $x\tau=(0,m)$ or $(m,0)$ for some $m>0$. Hence this $x_\lambda\varsigma=\lambda(x\tau)$ and $x_\rho\varsigma=\rho(x\tau)$ satisfy $\Omega(X)$. Finally, \cref{thm:gen_set_B}(ii), implies that 
	\begin{equation}
		\gcd\{x_\lambda\varsigma-x_\rho\varsigma:x\in X\} = \gcd\{\lambda(x\tau)-\rho(x\tau):x\in X\} = \gcd\{\delta(x\tau):x\in X\} = 1.
	\end{equation}
	
	For the converse, suppose that there exists a satisfying assignment $\varsigma:V\rightarrow\mathbb{N}$ to $\Phi(X,R)$. Then, according to \cref{thm:eqn_sol_iff_formula_satisfied}, $\varsigma$ satisfying $\Psi(\Xi(R))$ implies the mapping $\tau:X\rightarrow\mathcal{B}$ given by $x\mapsto(x_\lambda\varsigma,x_\rho\varsigma)$ is a solution to $R$ over $\mathcal{B}$. Since $\varsigma$ satisfies $\Omega(X)$ there exists $x\in X$ such that $x\tau=(x_\lambda\varsigma,x_\rho\varsigma)=(0,m)$ or $(m,0)$ for some $m>0$. Together with
	\begin{equation}
		\gcd\{x_\lambda\varsigma-x_\rho\varsigma:x\in X\} = 1 \quad\Rightarrow\quad \gcd\{\delta(x\tau):x\in X\} = 1
	\end{equation}
	we have by \cref{thm:gen_set_B} that $X\tau$ generates $\mathcal{B}$. Therefore there exists an epimorphism $\mathcal{I}\rightarrow\mathcal{B}$ by \cref{thm:von-dyck}(i).
