\label{lem:proper}
	Algorithm \ref{al:sppa} is well-defined, i.e., if $x_{k+1}\notin\mathrm{zero}(A)$, there exists $\varphi_{k+1}\geq\varphi_k+1$ such that $f_{k+1}(\varphi_{k+1})=0$.

<!-- proof-1: body_md -->
First, according to the definition of $g_{k+1}^1$, $g_{k+1}^2$, $g_{k+1}^3$, the function $f_{k+1}$ can be written as
	\begin{align*}
		&\ f_{k+1}(\varphi_{k+1})\\
		=&\ \left(\frac{\varphi_{k+1}(\varphi_{k+1}+r)}{2r^2}+\frac{\varphi_{k+1}(\varphi_{k+1}-\varphi_k)c}{r^2}+\frac{(r-1)(\varphi_{k+1}-\varphi_k)^2c}{2r^2}-\frac{(\varphi_k+2r)(\varphi_k+r)}{2r^2}\right)\left\|a_{k+1}\right\|^2 \\
		&\ +(\varphi_{k+1}-\varphi_k-1)\left\langlea_{k+1},x_{k+1}-z_k\right\rangle-\frac{\varphi_k(\varphi_k+r)}{r^2}\left\langlea_{k+1}-a_k,x_{k+1}-x_k\right\rangle\\
		&\ -\frac{\varphi_k(\varphi_k+r)}{2r^2}\left\|a_{k+1}-a_k\right\|^2+\frac{\alpha[2(r-1-c)(\varphi_{k+1}-\varphi_k)\varphi_k+2r^2-(r+1)(c+1)]}{2r^2}\left\|a_{k+1}\right\|^2.
	\end{align*}
	Taking $\varphi_{k+1}=\varphi_k+1$, we obtain
	\begin{align*}
		f_{k+1}(\varphi_k+1)
		=&\ \left(\frac{(\varphi_k+1)(\varphi_k+r+1)}{2r^2}+\frac{(\varphi_k+1)c}{r^2}+\frac{(r-1)c}{2r^2}-\frac{(\varphi_k+2r)(\varphi_k+r)}{2r^2}\right)\left\|a_{k+1}\right\|^2\\
		&\ -\frac{\varphi_k(\varphi_k+r)}{2r^2}\left\|a_{k+1}-a_k\right\|^2-\frac{\varphi_k(\varphi_k+r)}{r^2}\left\langlea_{k+1}-a_k,x_{k+1}-x_k\right\rangle\\
		&\ +\frac{\alpha[2(r-1-c)\varphi_k+2r^2-(r+1)(c+1)]}{2r^2}\left\|a_{k+1}\right\|^2\\
		=&\ -\frac{(1-\alpha)[2(r-1-c)\varphi_k+2r^2-(r+1)(c+1)]}{2r^2}\left\|a_{k+1}\right\|^2\\
		&\ -\frac{\varphi_k(\varphi_k+r)}{2r^2}\left\|a_{k+1}-a_k\right\|^2-\frac{\varphi_k(\varphi_k+r)}{r^2}\left\langlea_{k+1}-a_k,x_{k+1}-x_k\right\rangle.
	\end{align*}
	Since $c<r-1$, we have $r-1-c>0$ and
	\[
	2r^2-(r+1)(c+1)>2r^2-(r+1)r=r^2-r>0.
	\]
	As $\alpha\in[0,1)$, we have $1-\alpha>0$, and hence
	\[
	-\frac{(1-\alpha)[2(r-1-c)\varphi_k+2r^2-(r+1)(c+1)]}{2r^2}<0.
	\]
	If $k\geq 1$, then from $x_{k+1}=J_A(\tilde{x}_{k+1})$ we have $a_{k+1}=\tilde{x}_{k+1}-x_{k+1}\in A(x_{k+1})$, and similarly $a_k\in A(x_k)$. Since $A$ is monotone,
	\[
	\left\langlea_{k+1}-a_k,x_{k+1}-x_k\right\rangle\geq 0.
	\]
	Therefore the last two terms in $f_{k+1}(\varphi_k+1)$ are non-positive, and the first term is strictly negative. Since $x_{k+1}\notin\mathrm{zero}(A)$, we have $a_{k+1}\neq 0$, which implies
	\[
	f_{k+1}(\varphi_k+1)<0.
	\]
	If $k=0$, then $\varphi_0=0$ and $a_0=0$, so
	\[
	f_{1}(1)=-\frac{(1-\alpha)[2r^2-(r+1)(c+1)]}{2r^2}\left\|a_1\right\|^2<0.
	\]
	In both cases, since $g_{k+1}^1=\frac{(r+1)c+1}{2r^2}\left\|a_{k+1}\right\|^2>0$, the function $f_{k+1}$ is a quadratic function opening upwards. Together with $f_{k+1}(\varphi_k+1)<0$ and $f_{k+1}(\varphi)\to+\infty$ as $\varphi\to+\infty$, there exists $\varphi_{k+1}\geq\varphi_k+1$ such that $f_{k+1}(\varphi_{k+1})=0$. This completes the proof.

\label{lem:lyapunovdiscretenonnegative}
	Let $A$ be a maximally monotone operator, and let $\{x_k\}$ and $\{z_k\}$ be the sequences generated by Algorithm \ref{al:sppa}. The Lyapunov function $\mathcal{E}(k)$ given in \eqref{eq:lyapunovdiscrete} can be represented as
	\begin{align*}
		\mathcal{E}(k)=&\ \frac{(r-1-c)\varphi_k^2+r(r-1)\varphi_k}{2r^2(r-1)}\left\|a_k\right\|^2+\frac{\varphi_k}{r}\left\langlea_k,x_k-x^*\right\rangle\\
		&\ +\frac{1}{2}\left\|\frac{\varphi_k}{r}\sqrt{\frac{c}{r-1}}a_k+\sqrt{\frac{r-1}{c}}(x^*-z_k)\right\|^2.
	\end{align*}
	If $r>1$ and $0<c<r-1$, then $\mathcal{E}(k)$ is nonnegative.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Using the equality
	\[
	\left\langlex,y\right\rangle=\frac{1}{2}\left\|x+y\right\|^2-\frac{1}{2}\left\|x\right\|^2-\frac{1}{2}\left\|y\right\|^2
	\]
	with
	\begin{align*}
		x &=  \sqrt{\frac{c\varphi_k^2}{r^2(r-1)}}a_k, \\
		y &= \sqrt{\frac{r-1}{c}}(x^*-z_k),
	\end{align*}
	we obtain
	\begin{align*}
		\mathcal{E}(k)= &\ \frac{\varphi_k(\varphi_k+r)}{2r^2}\left\|a_k\right\|^2+\frac{\varphi_k}{r}\left\langlea_k,x_k-x^*\right\rangle+\frac{\varphi_k}{r}\left\langlea_k,x^*-z_k\right\rangle+\frac{r-1}{2c}\left\|z_k-x^*\right\|^2\\
		=&\ \frac{(r-1-c)\varphi_k^2+r(r-1)\varphi_k}{2r^2(r-1)}\left\|a_k\right\|^2+\frac{\varphi_k}{r}\left\langlea_k,x_k-x^*\right\rangle\\
		&\ +\frac{1}{2}\left\|\sqrt{\frac{c\varphi_k^2}{r^2(r-1)}}a_k+\sqrt{\frac{r-1}{c}}(x^*-z_k)\right\|^2.
	\end{align*}
	Since $0<c<r-1$,
	\[
	\frac{(r-1-c)\varphi_k^2+r(r-1)\varphi_k}{2r^2(r-1)}\geq 0,\quad\forall k\geq 0.
	\]
	Moreover, because $x^*\in\mathrm{zero}(A)$ and $A$ is monotone, we have
	\[
	\left\langlea_k,x_k-x^*\right\rangle=\left\langlea_k-0,x_k-x^*\right\rangle\geq 0.
	\]
	Therefore $\mathcal{E}(k)$ is a sum of three nonnegative terms, which implies $\mathcal{E}(k)\geq 0$.

<!-- theorem-1: body_md -->
\label{thm:lyapunovdiscrete}
	Let $A$ be a maximally monotone operator, and let $\{x_k\}$ and $\{z_k\}$ be the sequences generated by Algorithm \ref{al:sppa}. If $r>1$, $0<c<r-1$, and $\alpha\in[0,1)$, then
	\begin{equation}
		\label{eq:differenceestimation}
		\begin{split}
			&\ \mathcal{E}(k+1)-\mathcal{E}(k)\\
			=&\ -\frac{\alpha[2(r-1-c)(\varphi_{k+1}-\varphi_k)\varphi_k+2r^2-(r+1)(c+1)]}{2r^2}\left\|a_{k+1}\right\|^2\\
			&\ -\frac{(r-1)(\varphi_{k+1}-\varphi_k)}{r}\left\langlea_{k+1},x_{k+1}-x^*\right\rangle\\
			\leq&\ 0.
		\end{split}
	\end{equation}
	Also,
	\begin{equation}
		\left\|a_k\right\|^2\leq\frac{r^2(r-1)^2\left\|x_0-x^*\right\|^2}{c[(r-1-c)\varphi_k^2+r(r-1)\varphi_k]}.
		\label{eq:lastiterate}
	\end{equation}
	Moreover, if $\alpha>0$, then
	\begin{equation}
		\sum_{k=0}^\infty(\varphi_{k+1}-\varphi_k)\varphi_k\left\|a_{k+1}\right\|^2<+\infty.
		\label{eq:sumalpha}
	\end{equation}

<!-- proof-3: body_md -->
Step 1: Decompose the difference $\mathcal{E}(k+1)-\mathcal{E}(k)$ into three parts:
	\begin{align*}
		\mathcal{E}(k+1)-\mathcal{E}(k)= &\  \frac{\varphi_{k+1}(\varphi_{k+1}+r)}{2r^2}\left\|a_{k+1}\right\|^2-\frac{\varphi_k(\varphi_k+r)}{2r^2}\left\|a_k\right\|^2\\
		&\ +\underbrace{\frac{\varphi_{k+1}}{r}\left\langlea_{k+1},x_{k+1}-z_{k+1}\right\rangle-\frac{\varphi_k}{r}\left\langlea_k,x_k-z_k\right\rangle}_{\text{I}}\\
		&\ +\underbrace{\frac{r-1}{2c}\left(\left\|z_{k+1}-x^*\right\|^2-\left\|z_k-x^*\right\|^2\right)}_{\text{II}}.
	\end{align*}
	
	Step 2: Transform I and II. First, for I, using $z_{k+1}=z_k-\frac{c(\varphi_{k+1}-\varphi_k)}{r}a_{k+1}$, we have
	\begin{align*}
		\text{I}= &\ \frac{\varphi_{k+1}}{r}\left\langlea_{k+1},x_{k+1}-z_k\right\rangle-\frac{\varphi_k}{r}\left\langlea_k,x_k-z_k\right\rangle+\frac{\varphi_{k+1}(\varphi_{k+1}-\varphi_k)c}{r^2}\left\|a_{k+1}\right\|^2\\
		=&\ \frac{\varphi_{k+1}-\varphi_k}{r}\left\langlea_{k+1},x_{k+1}-z_k\right\rangle+\frac{\varphi_k}{r}\left\langlea_{k+1},x_{k+1}-x_k\right\rangle+\frac{\varphi_k}{r}\left\langlea_{k+1}-a_k,x_k-z_k\right\rangle\\
		&\ +\frac{\varphi_{k+1}(\varphi_{k+1}-\varphi_k)c}{r^2}\left\|a_{k+1}\right\|^2.
	\end{align*}
	By using the equality
	\[
	\varphi_k(x_{k+1}-x_k)=r(z_k-x_{k+1})-(\varphi_k+r)a_{k+1},
	\]
	we have
	\[
	\frac{\varphi_k}{r}\left\langlea_{k+1}, x_{k+1}-x_k\right\rangle=\frac{1}{r}\left\langlea_{k+1},-(\varphi_k+r)a_{k+1}+r(z_k-x_{k+1})\right\rangle.
	\]
	Also, from the equality
	\[
	r(z_k-x_k)=(\varphi_k+r)(a_{k+1}+x_{k+1}-x_k),
	\]
	it follows that
		\[
	\frac{\varphi_k}{r}\left\langlea_{k+1}-a_k,x_k-z_k\right\rangle=-\frac{\varphi_k(\varphi_k+r)}{r^2}\left\langlea_{k+1}-a_k,a_{k+1}+x_{k+1}-x_k\right\rangle.
	\]
	Based on the previous discussion, we obtain
	\begin{align*}
		\text{I}= &\  \frac{\varphi_{k+1}-\varphi_k-r}{r}\left\langlea_{k+1},x_{k+1}-z_k\right\rangle-\frac{\varphi_k+r}{r}\left\|a_{k+1}\right\|^2-\frac{\varphi_k(\varphi_k+r)}{r^2}\left\langlea_{k+1}-a_k,a_{k+1}\right\rangle\\
		&\ -\frac{\varphi_k(\varphi_k+r)}{r^2}\left\langlea_{k+1}-a_k,x_{k+1}-x_k\right\rangle+\frac{\varphi_{k+1}(\varphi_{k+1}-\varphi_k)c}{r^2}\left\|a_{k+1}\right\|^2\\
		=&\ (\varphi_{k+1}-\varphi_k-1)\left\langlea_{k+1},x_{k+1}-z_k\right\rangle-\frac{(r-1)(\varphi_{k+1}-\varphi_k)}{r}\left\langlea_{k+1},x_{k+1}-z_k\right\rangle\\
		&\ -\left(\frac{\varphi_k(\varphi_k+r)}{2r^2}+\frac{\varphi_k+r}{r}-\frac{\varphi_{k+1}(\varphi_{k+1}-\varphi_k)c}{r^2}\right)\left\|a_{k+1}\right\|^2+\frac{\varphi_k(\varphi_k+r)}{2r^2}\left\|a_k\right\|^2\\
		&\ -\frac{\varphi_k(\varphi_k+r)}{2r^2}\left\|a_{k+1}-a_k\right\|^2-\frac{\varphi_k(\varphi_k+r)}{r^2}\left\langlea_{k+1}-a_k,x_{k+1}-x_k\right\rangle.
	\end{align*}
	Now for II:
	\begin{align*}
		\text{II} & =\frac{r-1}{c}\left\langlez_{k+1}-z_k,z_k-x^*\right\rangle+\frac{r-1}{2c}\left\|z_{k+1}-z_k\right\|^2\\
		&=-\frac{(r-1)(\varphi_{k+1}-\varphi_k)}{r}\left\langlea_{k+1},z_k-x^*\right\rangle+\frac{(r-1)(\varphi_{k+1}-\varphi_k)^2c}{2r^2}\left\|a_{k+1}\right\|^2.
	\end{align*}
	
	Step 3: Estimate the upper bound of $\mathcal{E}(k+1)-\mathcal{E}(k)$. Combining the above calculations yields
	\begin{align*}
		&\ \mathcal{E}(k+1)-\mathcal{E}(k)\\
		=&\ \left(\frac{\varphi_{k+1}(\varphi_{k+1}+r)}{2r^2}+\frac{\varphi_{k+1}(\varphi_{k+1}-\varphi_k)c}{r^2}+\frac{(r-1)(\varphi_{k+1}-\varphi_k)^2c}{2r^2}-\frac{(\varphi_k+2r)(\varphi_k+r)}{2r^2}\right)\left\|a_{k+1}\right\|^2 \\
		&\ +(\varphi_{k+1}-\varphi_k-1)\left\langlea_{k+1},x_{k+1}-z_k\right\rangle-\frac{(r-1)(\varphi_{k+1}-\varphi_k)}{r}\left\langlea_{k+1},x_{k+1}-x^*\right\rangle\\
		&\ -\frac{\varphi_k(\varphi_k+r)}{r^2}\left\langlea_{k+1}-a_k,x_{k+1}-x_k\right\rangle-\frac{\varphi_k(\varphi_k+r)}{2r^2}\left\|a_{k+1}-a_k\right\|^2.
	\end{align*}
	Using the definition of $\varphi_{k+1}$, we obtain
	\begin{align*}
		&\ \mathcal{E}(k+1)-\mathcal{E}(k)\\
		=&\ -\frac{\alpha[2(r-1-c)(\varphi_{k+1}-\varphi_k)\varphi_k+2r^2-(r+1)(c+1)]}{2r^2}\left\|a_{k+1}\right\|^2\\
		&\ -\frac{(r-1)(\varphi_{k+1}-\varphi_k)}{r}\left\langlea_{k+1},x_{k+1}-x^*\right\rangle\\
		\leq&\ 0.
	\end{align*}

	\[

	Dividing both sides by $\frac{(r-1-c)\varphi_k^2+r(r-1)\varphi_k}{2r^2(r-1)}$, we obtain \eqref{eq:lastiterate}. Also, by summing the inequality in \eqref{eq:differenceestimation}, we have
	\[

	\]

<!-- corollary-1: body_md -->
\label{coro:1}
	Let $A$ be a maximally monotone operator, and let $\{x_k\}$ and $\{z_k\}$ be the sequences generated by Algorithm \ref{al:sppa}. If $r>1$, $0<c<r-1$, and $\alpha\in(0,1)$, then
	\begin{equation}
		\liminf_{k\to\infty}k^2\left\|a_k\right\|^2=0.
	\end{equation}
<!-- end corollary-1 -->
<!-- proof-4: body_md -->
From \eqref{eq:sumalpha}, we have
	\[
	\sum_{k=0}^\infty(\varphi_{k+1}-\varphi_k)\varphi_k\left\|a_{k+1}\right\|^2<+\infty.
	\]
	Since $\varphi_{k+1}\geq\varphi_k+1$ and $\varphi_0=0$, we have $\varphi_k\geq k$. Hence
	\[
	(\varphi_{k+1}-\varphi_k)\varphi_k\geq\varphi_k\geq k,
	\]
	which implies
	\[
	\sum_{k=0}^\infty k\left\|a_{k+1}\right\|^2
	\leq\sum_{k=0}^\infty(\varphi_{k+1}-\varphi_k)\varphi_k\left\|a_{k+1}\right\|^2
	<+\infty.
	\]
	Suppose, for contradiction, that $\liminf_{k\to\infty}k^2\left\|a_k\right\|^2>0$. Then there exist $\varepsilon>0$ and $K\geq 1$ such that
	\[
	k^2\left\|a_k\right\|^2\geq\varepsilon,\quad\forall k\geq K.
	\]
	Thus
	\[
	k\left\|a_k\right\|^2\geq\frac{\varepsilon}{k},\quad\forall k\geq K,
	\]
	and therefore
	\[
	\sum_{k=K}^\infty k\left\|a_k\right\|^2

	By Proposition \ref{prop:sumprop}, $\lim_{k\to\infty}\frac{1}{2}\left\|x_k-x^*\right\|^2$ exists for all $x^*\in\mathrm{zero}(A)$.
	
	For any weak cluster point $x_\infty$ of $\{x_k\}$, let $\{x_{k_j}\}$ be a subsequence such that $x_{k_j}\rightharpoonup x_\infty$. Since $a_k\to 0$, Proposition \ref{prop:nonexpansiveweak} implies $x_\infty\in\mathrm{zero}(A)$. By Opial's lemma, $x_k\rightharpoonup x_\infty\in\mathrm{zero}(A)$.
	
	Case 2: $\sum_{k=1}^\infty\frac{1}{\varphi_k}=+\infty$ and $\alpha>0$. We first prove that $\lim_{k\to\infty}\left\|z_k-x^*\right\|^2$ exists for all $x^*\in\mathrm{zero}(A)$. Using $z_{k+1}=z_k-\frac{c(\varphi_{k+1}-\varphi_k)}{r}a_{k+1}$, we have
	\begin{align*}
		&\ \frac{1}{2}\left\|z_{k+1}-x^*\right\|^2\\
		=&\ \frac{1}{2}\left\|z_k-x^*\right\|^2-\frac{c(\varphi_{k+1}-\varphi_k)}{r}\left\langlea_{k+1},x_{k+1}-x^*\right\rangle\\
		&\ +\frac{c(\varphi_{k+1}-\varphi_k)}{r}\left\langlea_{k+1},x_{k+1}-z_{k+1}\right\rangle-\frac{1}{2}\left\|z_{k+1}-z_k\right\|^2.
	\end{align*}
	Since $\varphi_{k+1}-\varphi_k\geq 1$ and $x^*\in\mathrm{zero}(A)$, we have
	\[
	-\frac{c(\varphi_{k+1}-\varphi_k)}{r}\left\langlea_{k+1},x_{k+1}-x^*\right\rangle\leq 0.
	\]
	By Young's inequality,
	\begin{align*}
		&\ \frac{c(\varphi_{k+1}-\varphi_k)}{r}\left\langlea_{k+1},x_{k+1}-z_{k+1}\right\rangle\\
		\leq&\ \frac{c^2(\varphi_{k+1}-\varphi_k)^2\varphi_k}{r^2}\left\|a_{k+1}\right\|^2+\frac{\left\|x_{k+1}-z_{k+1}\right\|^2}{\varphi_k}\\
		=&\ \frac{c^2\varphi_k[(\varphi_{k+1}-\varphi_k-1)^2+2(\varphi_{k+1}-\varphi_k-1)\varphi_k+\varphi_k]}{r^2}\left\|a_{k+1}\right\|^2+\frac{\left\|x_{k+1}-z_{k+1}\right\|^2}{\varphi_k}.
	\end{align*}
	Using \eqref{eq:sumalpha}, \eqref{eq:sumbeta} and \eqref{eq:xzsum}, we obtain
	\[
	\sum_{k=1}^\infty\frac{c(\varphi_{k+1}-\varphi_k)}{r}\left\langlea_{k+1},x_{k+1}-z_{k+1}\right\rangle\leq\sum_{k=1}^\infty\frac{c(\varphi_{k+1}-\varphi_k)}{r}|\left\langlea_{k+1},x_{k+1}-z_{k+1}\right\rangle|<+\infty.
	\]
	By Proposition \ref{prop:sumprop}, $\lim_{k\to\infty}\left\|z_k-x^*\right\|^2$ exists.
	
	For any weak cluster point $z_\infty$ of $\{z_k\}$, let $\{z_{k_j}\}$ be a subsequence such that $z_{k_j}\rightharpoonup z_\infty$. Since $\lim_{k\to\infty}\left\|x_k-z_k\right\|^2=0$, we have $x_{k_j}\rightharpoonup z_\infty$. Since $a_k\to 0$, Proposition \ref{prop:nonexpansiveweak} implies $z_\infty\in\mathrm{zero}(A)$. By Opial's lemma, $z_k\rightharpoonup z_\infty\in\mathrm{zero}(A)$ and $x_k\rightharpoonup z_\infty\in\mathrm{zero}(A)$.
	
	Next, we show that $\lim_{k\to\infty}k^2\left\|a_k\right\|^2=0$. Since $\{\mathcal{E}(k)\}$ is non-increasing and nonnegative, $\lim_{k\to\infty}\mathcal{E}(k)$ exists. Due to \eqref{eq:xzsum}, \eqref{eq:xzlimit} and the assumption that is $\sum_{k=1}^{+\infty}1/\varphi_k<+\infty$, we can prove $\lim_{k\to\infty}\left\|x_k-z_k\right\|^2=0$. Since $\lim_{k\to\infty}\left\|x_k-z_k\right\|^2=0$ and $\lim_{k\to\infty}\left\|z_k-x^*\right\|^2$ exists, we have that $\lim_{k\to\infty}\varphi_k^2\left\|a_k\right\|^2$ exists. Moreover, since
	\[
	\sum_{k=0}^\infty(\varphi_{k+1}-\varphi_k)\varphi_k\left\|a_k\right\|^2<+\infty,
	\]
	the limit must be zero. Therefore,
	\[
	0\leq\lim_{k\to\infty}k^2\left\|a_k\right\|^2\leq\lim_{k\to\infty}\varphi_k^2\left\|a_k\right\|^2=0.
	\]
