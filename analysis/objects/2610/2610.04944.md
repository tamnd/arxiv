\label{ass:as}

 \frac1n\sum_{i=1}^{n}\lVert\nabla f_i(z)-\nabla f_i(z')\rVert^2
 \le L^2\lVertz-z'\rVert^2.
\end{equation}

<!-- assumption-2: body_md -->
\label{ass:is}
For all $i\in[n]$ and all $z,z'$, $\lVert\nabla f_i(z)-\nabla f_i(z')\rVert\le L\lVertz-z'\rVert$.

<!-- assumption-3: body_md -->
\label{ass:init}
For every $x$, the function $f(x,\cdot)$ is $\mu$-strongly concave. The primal function is bounded below and satisfies
\begin{equation}\label{eq:init}
 \Phi(0)-\inf_x\Phi(x)\le\Delta,
 \qquad y^*(0):=\arg\max_y f(0,y)=0.
\end{equation}

<!-- definition-1: body_md -->
Given an index $i\in[n]$ and a point $(x,y)$, an IFO query returns $\nabla f_i(x,y)$. Each component-gradient evaluation counts as one query; computing a full gradient costs $n$ queries.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:zr}
Write $(g_x^t,g_y^t)=\nabla f_{i_t}(x^t,y^t)$ for the response to query $t$. The first query is at $(0,0)$. For every $t\ge1$, a zero-respecting algorithm satisfies
\begin{equation}\label{eq:zr}
 \operatorname{supp}(x^t)\subseteq\bigcup_{s<t}\operatorname{supp}(g_x^s),\quad
 \operatorname{supp}(y^t)\subseteq\bigcup_{s<t}\operatorname{supp}(g_y^s),\quad
 \operatorname{supp}(\hat x^t)\subseteq\bigcup_{s\le t}\operatorname{supp}(g_x^s),
\end{equation}
where $\hat x^t$ is an output after $t$ queries, and $\hat x^0=0$.
<!-- end definition-2 -->
<!-- theorem-1: body_md -->
\label{thm:matching}
There is a universal constant $c_0>0$ such that, for $n\ge2$, $\kappa=L/\mu\ge1$, and $0<\varepsilon^2\le c_0L\Delta$,
\begin{equation}\label{eq:matchedcomplexity}
 \mathfrak N^{\mathrm{exp}}_\varepsilon\bigl(\mathcal F_{\mathrm{AS}}^0(n,L,\mu,\Delta)\bigr)=
 \begin{cases}
 \widetilde{\Theta}(n+\sqrt n\,\kappa L\Delta\varepsilon^{-2}),&1\le\kappa\le\sqrt n,\\
 \widetilde{\Theta}(n+n^{3/4}\sqrt\kappa\,L\Delta\varepsilon^{-2}),&\kappa\ge\sqrt n.
 \end{cases}
\end{equation}
For $1\le\kappa\le\sqrt n$, the same statement holds for the fixed-budget complexity $\mathfrak N_\varepsilon$. The lower bound in \eqref{eq:matchedcomplexity} is $\Omega$ without logarithmic factors.
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
Theorem~\ref{thm:as} gives the lower bound in both regimes. For $\kappa\le\sqrt n$, RPDA attains the matching upper bound with a fixed query budget by Theorem~\ref{thm:smallupper}. For $\kappa\ge\sqrt n$, the expected-query Catalyst bound \cite[Corollary 4.1 and Section 4.2]{zhang} matches the lower bound because $n^{3/4}\sqrt\kappa\ge n$ and the accuracy restriction ensures $L\Delta\varepsilon^{-2}\ge1$. Both methods are zero-respecting; Appendix~\ref{app:catalyst} records the imported accuracy guarantee and oracle implementation.

<!-- theorem-2: body_md -->
\label{thm:as}
There are universal constants $c_0,c_1>0$ such that, for $n\ge2$, $\kappa=L/\mu\ge1$, and $0<\varepsilon^2\le c_0L\Delta$,
\begin{equation}\label{eq:mainlower}
 \mathfrak N_\varepsilon\bigl(\mathcal F_{\mathrm{AS}}^0(n,L,\mu,\Delta)\bigr)
 \ge c_1\left[n+\min\{\sqrt n\,\kappa,n^{3/4}\sqrt\kappa\}\,L\Delta\varepsilon^{-2}\right].
\end{equation}
Every component of the hard instances is $\mu$-strongly concave in $y$, with a quadratic dual term and at most bilinear coupling. The same order holds for worst-case expected query complexity, for $\mathbb E\lVert\nabla\Phi(\hat x)\rVert\le\varepsilon$, and for success probability at least $2/3$ under a fixed budget.

<!-- lemma-1: body_md -->
\label{lem:endpoint}
For the parameters in \eqref{eq:parameters} and endpoints in \eqref{eq:endpoints}, with $n\ge2$, $\kappa\ge2$, and $\vartheta:=\min\{1,\kappa/\sqrt n\}$, we have $\rho\ge1$ and
\begin{equation}\label{eq:endpointbounds}
 \vartheta(\rho-1)\le16,\qquad \vartheta b^{-1}\le9L,\qquad
 m-1\ge\frac1{12}\sqrt{\frac\kappa{\sqrt n}}.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
Set $r=\mu/w$ and $t=\operatorname{arcosh}(1+r/2)$. If $u=H^{-1}e_m$, the first boundary equation and the interior recurrence are
\[
 u_2=(1+r)u_1,\qquad u_{j+1}=(2+r)u_j-u_{j-1}\quad(2\le j<m).
\]
Consequently,
\begin{equation}\label{eq:cosh}
 \frac{u_j}{u_1}=
 \frac{\cosh((j-\tfrac12)t)}{\cosh(t/2)},\qquad
 \rho=\frac{\cosh((m-\tfrac12)t)}{\cosh(t/2)}.
\end{equation}
Since $\mathbf1^\top H=\mu\mathbf1^\top$, we have $\sum_j u_j=1/\mu$. Formula~\eqref{eq:cosh} then gives $b=u_1>0$ and $b\le u_j\le a$.

If $m\ge3$, then $m\le\sqrt{w/\mu}$. The inequality $t\le\sqrt r$ follows from $\cosh(\sqrt r)\ge1+r/2$. Hence $\rho\le\cosh(1)<2$, and
\[
 \frac1\mu=\sum_j u_j\le2mb,
 \qquad b^{-1}\le2\mu m\le2\sqrt{\mu w}\le2L.
\]
If $m=2$, direct inversion gives
\[
 \rho=1+r,\qquad b^{-1}=\mu(r+2).
\]
Here $\vartheta r\le16$ and $\mu\le L/2$, so $\vartheta(\rho-1)\le16$ and $\vartheta b^{-1}=\mu(\vartheta r+2\vartheta)\le9L$. For $m\ge3$, the already proved bounds $\rho<2$, $b^{-1}\le2L$, and $\vartheta\le1$ imply the same estimates.

For any $q>0$, $\max\{2,\lfloor q\rfloor\}-1\ge q/3$. Indeed, when $q<3$ the left side is at least one; when $q\ge3$ it is at least $q-2\ge q/3$. Taking $q=\sqrt{w/\mu}=\tfrac14\sqrt{\kappa/\sqrt n}$ proves the last bound.
<!-- end proof-2 -->
<!-- lemma-2: body_md -->
\label{lem:search}
Before exposing $x_d$, an algorithm satisfying Definition~\ref{def:zr} must traverse $J=(d-1)(m-1)$ internal dual edges. The number of required IFO calls stochastically dominates $S=\sum_{q=1}^{J}R_q$, where the $R_q$ are independent and uniform on $\{1,\ldots,n\}$ at the first edge of each block, or on $\{1,\ldots,n-1\}$ at a subsequent edge. Consequently,
\begin{equation}\label{eq:searchmoments}
 \mathbb E S\ge\frac{nJ}{2},\qquad
 \mathbb E S^2\le\frac43(\mathbb E S)^2,\qquad
 \mathbb P\{S>nJ/8\}\ge\frac{27}{64}.
\end{equation}
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
Give previously traversed owners to the algorithm. Let $\mathcal F_t$ be the filtration generated by its private random seed, the first $t$ queries and responses, and these revealed owners. Enumerate the $J$ internal edges in their path order. For edge $q$, let $\tau_q$ be the first response count at which its upstream endpoint is exposed, and let $\eta_q$ be the first at which its downstream endpoint is exposed. These are stopping times, with $\eta_q\le\tau_{q+1}$. Complete a stopped algorithm by an admissible search through the remaining candidates; if a frontier is never reached, its infinite hitting time already dominates the claimed bound.

On reaching frontier $q$, its owner is uniform conditional on $\mathcal F_{\tau_q}$ among $N_q=n$ candidates at the start of a block and $N_q=n-1$ otherwise. A failed informative query excludes exactly its tested label; every untested owner gives the same response. Let $R_q$ be the rank of the successful label among distinct informative candidate tests. Adaptive selection therefore gives
\[
 \mathbb P(R_q=r\mid\mathcal F_{\tau_q})=\frac1{N_q},
 \qquad r=1,\ldots,N_q.
\]
The earlier ranks are $\mathcal F_{\tau_q}$-measurable, and $N_q$ is deterministic, so iterated conditioning proves that the $R_q$ are independent. Repeated or uninformative calls only increase $\eta_q-\tau_q$, giving $\eta_q-\tau_q\ge R_q$ pathwise on this coupling. The intervals are disjoint, hence exposing $x_d$ requires at least $\sum_qR_q$ calls. For unvisited frontiers, extend the coupling with independent fresh ranks. Inferring the final candidate still requires a successful oracle call to expose the next coordinate; this also covers $n=2$.

For a variable uniform on $\{1,\ldots,N\}$, the mean is $(N+1)/2$ and the variance is $(N^2-1)/12$. Thus
\[
 \mathbb E S\ge nJ/2,\qquad
 \operatorname{Var}(S)\le Jn^2/12\le(\mathbb E S)^2/(3J)\le(\mathbb E S)^2/3.
\]
Paley--Zygmund with parameter $1/4$ gives
\[
 \mathbb P\{S>\mathbb E S/4\}
 \ge\frac{(3/4)^2(\mathbb E S)^2}{\mathbb E S^2}\ge\frac{27}{64}.
\]
Since $\mathbb E S/4\ge nJ/8$, this proves \eqref{eq:searchmoments}.

\label{prop:iscomparison}
There is a universal constant $c_{\mathrm{IS}}\in(0,c_0]$ such that, for $n\ge2$, $\kappa\ge1$, and $0<\varepsilon^2\le c_{\mathrm{IS}}L\Delta$,
\begin{equation}\label{eq:iscombined}
 \begin{aligned}
 \mathfrak N^{\mathrm{exp}}_\varepsilon(\mathcal B_{\mathrm{IS}}^0)
 &=\Omega\!\left(n+\max\{\sqrt n,\min\{\kappa,\sqrt{n\kappa}\}\}
 L\Delta\varepsilon^{-2}\right),\\
 \mathfrak N^{\mathrm{exp}}_\varepsilon(\mathcal B_{\mathrm{IS}}^0)
 &=\widetilde{O}\!\left(n+\min\{\sqrt n\,\kappa,n+\sqrt{n\kappa}\}
 L\Delta\varepsilon^{-2}\right),
 \end{aligned}
\end{equation}
where $\mathcal B_{\mathrm{IS}}^0=\mathcal B_{\mathrm{IS}}^0(n,L,\mu,\Delta)$. The lower bound also holds for fixed budgets. At $\kappa=1$, both fixed-budget and expected-query complexities are $\Theta(n+\sqrt n\,L\Delta\varepsilon^{-2})$, without logarithmic factors.

Let $G=n^{-1}\sum_i g_i$ be an individually $L$-smooth minimization instance with $G(0)-\inf G\le\Delta$. Set
\[
 f_i(x,y)=g_i(x)-\frac\mu2\lVerty\rVert^2.
\]
The joint smoothness constant is at most $\max\{L,\mu\}=L$, and $A_i=0$, $h_i(y)=\mu\lVerty\rVert^2/2$, $y^*(x)=0$, and $\Phi=G$. The instance is in $\mathcal B_{\mathrm{IS}}^0$. Its dual gradient is known, so each joint IFO response can be simulated with one minimization query.

For all sufficiently large $n$, \citet[Theorem 3.1 and Appendix F.3, Equation (305)]{densehiding} give the lower bound $\Omega(n+\sqrt n\,L\Delta\varepsilon^{-2})$ for unrestricted randomized IFO algorithms, including worst-case expected query budgets and expected gradient norm. Their oracle also returns function values, which only strengthens that lower bound for our gradient-only oracle. Jensen's inequality transfers it to \eqref{eq:criterion}. Choose $c_{\mathrm{IS}}$ small enough to satisfy their accuracy restriction.

For the finitely many smaller $n$, use the decoupled chain \eqref{eq:isdecoupledchain}, which is admissible whenever $\mu\le L$. Appendix~\ref{app:issmallkappa} proves its $\Omega(L\Delta\varepsilon^{-2})$ bound, including expected budgets. The factor $\sqrt n$ is bounded by a universal constant in this finite range; Appendix~\ref{app:additive} supplies $\Omega(n)$. Thus the minimization lower bound used here holds for every $n\ge2$ after adjusting universal constants.

Taking the worst case over this family and Theorem~\ref{thm:is} proves the lower bound in \eqref{eq:iscombined} for every $\kappa\ge1$. This combination takes a maximum of two lower bounds; it does not multiply their coefficients.

For $\kappa\le\sqrt n$, apply RPDA, since $\mathcal B_{\mathrm{IS}}^0\subseteq\mathcal F_{\mathrm{AS}}^0$. Here $\sqrt n\,\kappa\le n$, so the minimum in the upper bound of \eqref{eq:iscombined} equals $\sqrt n\,\kappa$. For $\kappa\ge\sqrt n$, use Proposition~\ref{prop:upper}. In this range $\sqrt n\,\kappa$ is at least both $n$ and $\sqrt{n\kappa}$, and hence

 n+\sqrt{n\kappa}
 \le2\min\{\sqrt n\,\kappa,n+\sqrt{n\kappa}\}.
\]
Since $L\Delta\varepsilon^{-2}\ge1$, adding the one-time $n$ term does not change this upper bound. Both implementations are zero-respecting. Finally, at $\kappa=1$, Proposition~\ref{prop:boundary} gives the upper bound without logarithms, and the minimization family gives the matching lower bound.

<!-- proposition-3: body_md -->
\label{prop:iscap}
Let $m\ge2$ and
\[
 H=\mu I+\sum_{k=1}^{m-1}w_k(e_k-e_{k+1})(e_k-e_{k+1})^\top,
 \qquad w_k>0.
\]
Suppose edge $k$ occurs in $r_k\in[n]$ components, with weight $nw_k/r_k$ in each owner. Assume that the component dual Hessians are positive sums of these edges and $\mu I$, and have norm at most $L$. Write $u=H^{-1}e_1$ and $\rho=u_1/u_m$. Starting with the first endpoint exposed, let $Q_{\mathrm{link}}$ be the minimum expected number of zero-respecting component queries needed to expose the last endpoint. If the primal amplitude and diagonal-cancellation budgets satisfy $h\le L$ and $h(\rho-1)\le L$, then
\begin{equation}\label{eq:iscap}
 hQ_{\mathrm{link}}=O\!\left(L\min\{\kappa,\sqrt{n\kappa}\}\right).
\end{equation}
The same conclusion holds if either budget is enlarged by a universal constant.
<!-- end proposition-3 -->
<!-- proof-5: body_md -->
Positivity and the component norm bound give $w_k\le r_kL/n$. The inverse of $H$ is entrywise positive. Summing the node equations $(Hu)_j=0$ for $j=k+1,\ldots,m$ gives
\[
 w_k(u_k-u_{k+1})=\mu\sum_{j=k+1}^{m}u_j.
\]
Thus $u_1>\cdots>u_m>0$. Summing the increments and using $u_j\ge u_m$ yields
\begin{equation}\label{eq:isweightedresistance}
 \rho-1\ge\mu\sum_{k=1}^{m-1}\frac{m-k}{w_k}
 \ge\frac n\kappa\sum_{k=1}^{m-1}\frac{m-k}{r_k}.
\end{equation}
Set $S=\sum_k r_k^{-1}$. Since $0<r_k^{-1}\le1$, we have
\[
 \sum_{k=1}^{m-1}\frac{m-k}{r_k}\ge\max\{S,S^2/2\}.
\]
For the second inequality, allocate mass $S$ to the smallest integer weights first. If $S=a+t$ with $a=\lfloor S\rfloor$ and $0\le t<1$, the resulting minimum is $a(a+1)/2+(a+1)t\ge S^2/2$.

At a frontier, query independent uniform component labels, putting a nonzero value at the exposed endpoint. An owner is hit with probability $r_k/n$ on each call, so the expected cost to expose the next coordinate is at most $n/r_k$. This provides an upper bound on exposure cost, regardless of the owner assignment. Hence $Q_{\mathrm{link}}\le nS$. Including the two common primal--dual coupling edges changes this bound only by a constant factor, because they require at most two extra calls and $nS\ge m-1\ge1$.

The cancellation budget and \eqref{eq:isweightedresistance} imply
\[
 hnS\le\kappa L,
 \qquad hnS\le\frac{2\kappa L}{S}.
\]
The amplitude budget also gives $hnS\le nLS$. Therefore
\[
 hnS\le\min\{\kappa L,nLS,2\kappa L/S\}
 \le L\min\{\kappa,\sqrt{2n\kappa}\},
\]
which proves \eqref{eq:iscap}.

<!-- lemma-3: body_md -->
\label{lem:upperinner}
Let $W=W_a(x_{\mathrm{in}},y_{\mathrm{in}})$, and suppose every gradient-estimation error during an inner call has norm at most $e$. Set
\[
 M=\lceil4\kappa\log2048\rceil,\qquad \sigma=1/2048.
\]
For the output potential $P=W_a(x_{\mathrm{out}},y_{\mathrm{out}})$ and the sum $V$ of squared distances between successive queried points, including the input, we have
\begin{equation}\label{eq:upperinner}
 P\le\sigma W+\frac{17\kappa e^2}{L},\qquad
 V\le C_0\left(\frac WL+\frac{Me^2}{L^2}\right),\qquad C_0=400.
\end{equation}
These inequalities are pathwise and allow arbitrary, dependent errors.
<!-- end lemma-3 -->
<!-- proof-6: body_md -->
Fix $y$ and write $\psi(x)=q_a(x,y)$. Since $\psi$ is $L$-strongly convex and $3L$-smooth, $\zeta(x)=\psi(x)-L\lVertx\rVert^2/2$ is convex and $2L$-smooth. Cocoercivity of $\nabla\zeta$ makes $I-\nabla\zeta/(2L)$ nonexpansive, so
\[
 \left\|u-v-\frac{\nabla\psi(u)-\nabla\psi(v)}{3L}\right\|
 =\frac23\left\|u-v-\frac{\nabla\zeta(u)-\nabla\zeta(v)}{2L}\right\|
 \le\frac23\lVertu-v\rVert.
\]
This argument uses only differentiability and Lipschitz gradients. Taking $v=p_a(y)$ shows that an inexact primal step contracts distance by $2/3$, apart from an additive error $e/(3L)$. After six steps the distance is at most $r$ times its initial value plus $e/L$, where $r=(2/3)^6<1/8$.

Index the points after each group of six primal steps by $j=0,\ldots,M$. Write $a_j=\lVertx_j-p_a(y_j)\rVert$, $G_j=G_a(y_j)$, $P_j=G_j+La_j^2$, and $d_j=y_{j+1}-y_j$. The gradient used for the dual update differs from $\nabla D_a(y_j)$ by a vector of norm at most $La_j+e$. Smoothness and strong concavity of $D_a$ imply
\begin{equation}\label{eq:upperonestep}
 \begin{aligned}
 G_{j+1}&\le\left(1-\frac1{4\kappa}\right)G_j
             -L\lVertd_j\rVert^2+\frac L4a_j^2+\frac{e^2}{4L},\\
 a_{j+1}^2&\le3r^2a_j^2+3r^2\lVertd_j\rVert^2+\frac{3e^2}{L^2}.
 \end{aligned}
\end{equation}
For clarity, the first inequality in \eqref{eq:upperonestep} follows from the smooth ascent identity with step $\eta=1/(4L)$: the decrease in the dual gap is at least $\eta\lVert\nabla D_a(y_j)\rVert^2/2+\eta\lVertd_j/\eta\rVert^2/4-\eta(La_j+e)^2/2$. Use $\lVert\nabla D_a(y_j)\rVert^2\ge2\mu G_j$.
Combining the two inequalities in \eqref{eq:upperonestep} gives
\begin{equation}\label{eq:upperlyapunov}
 P_{j+1}\le\left(1-\frac1{4\kappa}\right)P_j
       -\frac L4a_j^2-\frac L2\lVertd_j\rVert^2+\frac{4e^2}{L}.
\end{equation}
The first six primal steps give $P_0\le W+2e^2/L$. Iterating \eqref{eq:upperlyapunov} proves the first inequality in \eqref{eq:upperinner}, since the geometric error sum is at most $16\kappa e^2/L$.

Summing \eqref{eq:upperlyapunov} also gives
\[
 \sum_{j<M}a_j^2\le\frac{4P_0}L+\frac{16Me^2}{L^2},\qquad
 \sum_{j<M}\lVertd_j\rVert^2\le\frac{2P_0}L+\frac{8Me^2}{L^2}.
\]
A group of six primal steps starting at distance $a$ from its minimizer has total squared movement at most $4a^2+22e^2/L^2$: sum the bounds $2(2/3)^{2j}a^2+32e^2/(9L^2)$ for its six steps. After a dual step, the next group's starting distance is at most $a_j+\lVertd_j\rVert$. Including the dual steps, the total is therefore at most
\[
 4\lVertx_{\mathrm{in}}-p_a(y_{\mathrm{in}})\rVert^2
 +8\sum_{j<M}a_j^2+9\sum_{j<M}\lVertd_j\rVert^2
 +\frac{22(M+1)e^2}{L^2}.
\]
Substituting the preceding bounds proves the second inequality with $C_0=400$.

<!-- lemma-4: body_md -->
\label{lem:upperwarm}
For any centers $a,a'$ and any $(x,y)$,
\begin{equation}\label{eq:upperwarm}
 W_{a'}(x,y)\le2W_a(x,y)+12L\lVerta'-a\rVert^2.
\end{equation}
<!-- end lemma-4 -->
<!-- proof-7: body_md -->
Put $d=a'-a$ and $x^*=x_a^*$. The difference $q_{a'}-q_a$ is $-2L\langle d,x-x^*\rangle+c$, with $c$ independent of $(x,y)$. Testing the new primal objective at $x^*$ gives $v_{a'}\le v_a+c$. Strong convexity of $q_a(\cdot,y)$ gives
\[
 D_{a'}(y)\ge D_a(y)+c-2L\langle d,p_a(y)-x^*\rangle-2L\lVertd\rVert^2.
\]
By \eqref{eq:upperdistance},
\[
 G_{a'}(y)\le\bigl(\sqrt{G_a(y)}+\sqrt{2L}\lVertd\rVert\bigr)^2
 \le2G_a(y)+4L\lVertd\rVert^2.
\]
The primal minimizers satisfy $\lVertp_{a'}(y)-p_a(y)\rVert\le2\lVertd\rVert$, because the change in their primal gradients is $-2Ld$. Thus $L\lVertx-p_{a'}(y)\rVert^2\le2L\lVertx-p_a(y)\rVert^2+8L\lVertd\rVert^2$. Adding proves the claim.
