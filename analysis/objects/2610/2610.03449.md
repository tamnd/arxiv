<!-- proposition-1: body_md -->
\label{Prop.203}
  For any $t> 0$ and $x,y> 0$, we always have 
  \begin{eqnarray*}
 	\underline{n}\bigg(  \sum_{0<s<r\leq t\wedge \zeta} \mathbf{1}_{\{\Delta \epsilon_s > x ,\, \Delta \epsilon_r > y \}} \bigg) 
 	\leq t \cdot \underline{n}(\zeta ) \cdot \overline{\nu}(x)  \cdot \overline{\nu}(y).   
  \end{eqnarray*}
<!-- end proposition-1 -->
<!-- head-1: body_md -->
By  the Markov property \eqref{eqn.n-Markov}, we first have 
 \begin{eqnarray*}
 \underline{n}\bigg(  \sum_{0<s<r\leq t\wedge \zeta} \mathbf{1}_{\{\Delta \epsilon_s > x ,\, \Delta \epsilon_r > y \}} \bigg) 
 = \underline{n}\bigg( \sum_{0<s\leq t\wedge \zeta} \mathbf{1}_{\{\Delta \epsilon_s > x  \}} \cdot \mathbf{E}_{\epsilon_{s-} +\Delta\epsilon_s} \Big[ \sum_{0<r\leq t-s} \mathbf{1}_{\{ \tau_0^->r,\,\Delta X_r > y  \}}\Big] \bigg).
 \end{eqnarray*}
 Using the compensation formulas \eqref{eqn.Compensation} and  \eqref{eqn.CompenstationX} one-by-one yields that  the right-hand side equals
 \begin{eqnarray*}
 \lefteqn{ \int_0^t ds \int_0^\infty \underline{n}_s(dz) \int_x^\infty  \mathbf{E}_{z+w} \bigg[ \sum_{0<r\leq t-s} \mathbf{1}_{\{ \tau_0^->r,\,\Delta X_r > y \}}\bigg]  \, \nu(dw)   } \cr
 \!\!\!&=\!\!\!& \int_0^t ds \int_0^\infty \underline{n}_s(dz) \int_x^\infty \nu(dw)  \int_0^{t-s}  \mathbf{P}_{z+w} \big(   \tau_0^->r \big)   \, dr  \cdot \overline{\nu}(y)
 \leq t\cdot \int_0^t   \underline{n}(\zeta>s) \, ds   \cdot \overline{\nu}(x)  \cdot \overline{\nu}(y)  , 
 \end{eqnarray*}
 which is smaller than $ t \cdot \underline{n}(\zeta ) \cdot \overline{\nu}(x)  \cdot \overline{\nu}(y)$. The proof ends.
 \hfill$\Box$\medskip

<!-- proposition-2: body_md -->
\label{Prop.206}
  	There exists a constant $C>0$ such that for all $t\geq 0$, 
  	\begin{eqnarray*}
  	\mathbf{E}\bigg[ \sup_{s\in[0,t]} \big|X_s\big| \bigg] \leq C\cdot \big( \sqrt{t} +t  \big).
  	\end{eqnarray*}
<!-- end proposition-2 -->
<!-- head-2: body_md -->
By the L\'evy-It\^o decomposition, we have $X_t= -\beta t +Y_t+M_t$ with
  \begin{eqnarray*}
  Y_t:=  \sigma B_t+ \int_0^t \int_{|y|\leq 1} y \,\widetilde{N}(ds,dy)
  \quad \mbox{and}\quad 
  M_t:=   \int_0^t \int_{|y|>1} y \widetilde{N}(ds,dy),\quad t\geq 0.
  \end{eqnarray*} 
  Here $B$ is a standard Brownian motion and $\widetilde{N}(ds,dy)$ is a compensated Poisson random measure on $(0,\infty)\times \mathbb{R}$ with intensity $ds\,\nu(dy)$. 
  By the triangle inequality, 
  \begin{eqnarray*}
  \mathbf{E}\bigg[ \sup_{s\in[0,t]} \big|X_s\big| \bigg] \leq \beta t+ 	\mathbf{E}\bigg[ \sup_{s\in[0,t]} \big|Y_s\big| \bigg] + 	\mathbf{E}\bigg[ \sup_{s\in[0,t]} \big|M_s\big| \bigg] .
  \end{eqnarray*}  
  By using Jensen's inequality, then the Burkholder–Davis–Gundy inequality and finally \eqref{eqn.LevyMeasure}, there exists a constant $C>0$ such that for all $t\geq 0$, 
  \begin{eqnarray}\label{eqn.3044}
  \mathbf{E}\Big[ \sup_{r\in[0,t]}  |Y_r | \Big] 
  \!\!\!&\leq\!\!\!& \bigg(\mathbf{E}\Big[ \sup_{r\in[0,t]}  |Y_r |^2 \Big] \bigg)^{1/2}
  \leq C\cdot \Big( \sigma^2 + \int_{|y|\leq 1} |y|^2 \,\nu(dy) \Big)^{1/2}\cdot \sqrt{t}  \leq C\cdot \sqrt{t}. 
  \end{eqnarray}
  Additionally, by the definition of $\widetilde{N}(ds,dy)$ and then using \eqref{eqn.LevyMeasure} again, we also have uniformly in $t\geq 0$, 
  \begin{eqnarray*}
  \sup_{r\leq t}\big|M_r\big| \leq   \int_{|y|>1} |y|\,\nu(dy) \cdot  t+\int_0^t \int_{|y|>1} |y| \,N(ds,dy) 
  \quad \mbox{and then}\quad
  \mathbf{E}\Big[ \sup_{r\in[0,t]}  |M_r | \Big] \leq C\cdot t,
  \end{eqnarray*}
  The desired uniform upper bound follows by putting these two estimates together.

<!-- proposition-3: body_md -->
\label{Lemma.205}
 	For any $x\in \mathbb{R}$ and $\delta >0$, there exists a constant $C>0$ such that for all $t\geq 0$,
 	\begin{eqnarray}\label{eqn.2051}
 	\mathbf{P}\big( X_t\in x+\varDelta_\delta\big)\leq C\cdot  \big(\overline{\nu}(\beta t) \wedge 1\big).
 	\end{eqnarray}
<!-- end proposition-3 -->
<!-- head-4: body_md -->
Note that $\mathbf{P}\big( X_t\in x+\varDelta_\delta\big) \leq \mathbf{P}\big( X_t\in [x]-1+\varDelta_{[\delta]+3}\big)$ for all $x\in \mathbb{R}$, it suffices to prove \eqref{eqn.2051} with $x,\delta\in \mathbb{Z}$. 
 By the monotonicity of $\overline{\nu}$, it suffices to prove that the desired uniform upper bound holds for all large $t$.  
 Firstly, we have 
 \begin{eqnarray}\label{eqn.213}
 \frac{\mathbf{P}\big( X_t\in x+\varDelta_\delta\big)}{\overline{\nu}(\beta t)}
 \!\!\!&=\!\!\!&  \frac{t }{\overline{\nu}(\beta t)} \sum_{k=x}^{x+\delta-1} \frac{\mathbf{P}(X_t\in  k+\varDelta_1)}{t\cdot\nu(\beta t+k+\varDelta_1)}\cdot   \nu\big(\beta t+k+\varDelta_1\big) \cr
 \!\!\!&\leq\!\!\!&  \frac{t  }{\overline{\nu}(\beta t)} \cdot   \nu\big(\beta t+x+\varDelta_{\delta}\big) \cdot\sup_{k=x,\cdots,x+\delta-1} \frac{\mathbf{P}(X_t\in  k+\varDelta_1)}{t\cdot\nu(\beta t+k+\varDelta_1)} .
 \end{eqnarray}
 By Lemma~\ref{Lemma.203} with $\lambda=\beta/2$,   there exist two constants $C,t_0>0$  such that for all $t\geq t_0$,
 \begin{eqnarray*}
  \sup_{k=x,\cdots,x+\delta-1} \frac{\mathbf{P}(X_t\in  k+\varDelta_1)}{t\cdot\nu(\beta t+k+\varDelta_1)}
 \leq    \sup_{z\geq \beta t/2}\frac{\mathbf{P}(X_t+\beta t\in z+\varDelta_1)}{t\cdot \nu\big(z+\varDelta_1\big)}  \leq C. 
 \end{eqnarray*}
 Moreover, by \eqref{eqn.30} and enlarging $t_0$ if needed, we obtain that for all $t\geq t_0$,
 \begin{eqnarray*}
 \frac{t  }{\overline{\nu}(\beta t)} \cdot   \nu\big(\beta t+x+\varDelta_{\delta}\big)
 \leq   \frac{2\alpha}{\beta} \cdot (1+\delta) . 
 \end{eqnarray*} 
 Then \eqref{eqn.2051} follows by taking these two estimates back into the right-hand side of \eqref{eqn.213}. 
 \hfill$\Box$\medskip

<!-- proposition-5: body_md -->
\label{Lemma.208}
 	For any $x>0$, there exist two constants $C,t_0>0$ such that for all $t\geq t_0$, 
 	\begin{eqnarray}\label{eqn.214}
 	\sup_{y\geq 0} \mathbf{P}_{-y}\big(X_t\in(-x,0],\, 	\tau_0^+>t \big)  \leq C\cdot \frac{\overline\nu(\beta t)}{t}. 
 	\end{eqnarray}
<!-- end proposition-5 -->
<!-- head-6: body_md -->
Consider a three-parameter function $F_t(u,v) := \mathbf P\big( \overline{X}_t\leq u,\, \overline{X}_t-X_t\leq v \big)$ with $t\geq 0$ and $u,v\geq 0$. 
 By (3.32) and (3.38) in \cite{Xu2021a},  we have 
 \begin{eqnarray}\label{eqn.2031}
 \int_0^\infty F_s(u,v)\,ds = V(u)\cdot \widehat{V}(v)
 \end{eqnarray}
 and
 \begin{eqnarray}\label{eqn.203}
 F_t(u,v) =  \frac{1}{t}	\int_0^t ds \int_0^u F_s(u-z,v) \, \mathbf P\big(X_{t-s}\in dz
 \big) 
 + \frac{1}{t}	\int_0^t ds \int_0^v  F_s(u,v-z) \, \mathbf P\big( -X_{t-s}\in dz
 \big). 
 \end{eqnarray} 
 The spatial homogeneity of $X$ yields that
 \begin{eqnarray*}
 \mathbf P_{-y}\big( X_t\in(-x,0],\, \tau_0^+>t \big) = \mathbf P\big( X_t\in(y-x,y],\, \overline{X}_t\leq y \big).
 \end{eqnarray*}
 The fact $\big\{ X_t\in(y-x,y],\, \overline{X}_t\leq y \big\} \subset
 \big\{ \overline{X}_t \in (y-x-1,y],\, \overline{X}_t-X_t\leq x \big\}$ induces that 
 \begin{eqnarray*}
 \mathbf P_{-y}\big( X_t\in(-x,0],\, \tau_0^+>t \big) 
 \leq  F_t(y,x)-F_t(y-x-1,x) ,
 \end{eqnarray*}
 which, by \eqref{eqn.203}, can be decomposed into the following parts
 where 
 \begin{eqnarray}
 I_1(t,y) \!\!\!&:=\!\!\!&   \frac{1}{t}\int_0^{t/2} ds 
 \int_0^y \big( F_s(y-z,x)-F_s(y-x-1-z,x) \big) \, 
 \mathbf P\big( X_{t-s}\in dz \big), \label{eqn.204} \\
 I_2(t,y) \!\!\!&:=\!\!\!&   \frac{1}{t}\int_{t/2}^t ds 
 \int_0^y \big( F_s(y-z,x)-F_s(y-x-1-z,x) \big) \, 
 \mathbf P\big( X_{t-s}\in dz \big), \label{eqn.205}\\
 I_3(t,y) \!\!\!&:=\!\!\!&  \frac{1}{t}\int_0^t ds \int_0^x \big( F_s(y,x-z)-F_s(y-x-1,x-z) \big) \,
 \mathbf P\big( -X_{t-s}\in dz \big) .  \label{eqn.206}
 \end{eqnarray}
 Thus \eqref{eqn.214}   follows if we can prove that there exist constants $C,t_0>0$ such that for all $t\geq t_0$, 
 \begin{eqnarray}\label{eqn.2201}
 \sup_{y\geq 0} I_i(t,y) \leq C\cdot \frac{\overline\nu(\beta t)}{t},
 \quad i=1,2,3.
 \end{eqnarray}
 
 \textbf{Case $i=1$.}
 The increase of $F_s(u,v)$ in $u$ yields that
 \begin{eqnarray}
 0\leq F_s(y-z,x)-F_s(y-x-1-z,x) \leq F_s\big(y-[z],x)-F_s(y-x-[z]-2,x\big) , 
 \end{eqnarray}
 for all $z\in(0,y]$, which follows that
 \begin{eqnarray}\label{eqn.207}
 \lefteqn{\int_0^y \big( F_s(y-z,x)-F_s(y-x-1-z,x) \big) \, 
 	\mathbf P\big( X_{t-s}\in dz \big)}\!\!\!&\!\!\!&\cr
 \!\!\!&\leq\!\!\!& \sum_{k=0}^{[y]}\int_k^{k+1} \big( F_s(y-z,x)-F_s(y-x-1-z,x) \big) \, 
 \mathbf P\big( X_{t-s}\in dz \big)\cr
 \!\!\!&\leq\!\!\!&  \sum_{k=0}^{[y]} \big( F_s(y-k,x)-F_s(y-x-k-2,x) \big) \cdot  
 \mathbf P\big( X_{t-s}\in k+\varDelta_1 \big)\cr
 \!\!\!&\leq\!\!\!& \sup_{u\geq 0} 	\mathbf P\big( X_{t-s}\in u+\varDelta_1  \big) \cdot \sum_{k=0}^{[y]} \big( F_s(y-k,x)-F_s(y-x-k-2,x) \big).
 \end{eqnarray}
 Plugging this into the right-hand sides of \eqref{eqn.204}, 
 \begin{eqnarray}\label{eqn.208}
 \lefteqn{\frac{I_{1}(t,y)}{\overline{\nu}(\beta t)} 
 	\leq \frac{1}{t}
 	\int_0^{t/2} \sum_{k=0}^{[y]} \big(F_s(y-k,x) - F_s(y-x-k-2,x) \big)\cdot  \sup_{u\geq0} \frac{\mathbf P\big( X_{t-s}\in  u+\varDelta_1 \big)}{\overline{\nu}(\beta t)} \,ds } 
 \!\!\!&\!\!\!&\cr
 \!\!\!&\leq\!\!\!& \frac{1}{t}\sup_{r\in[\frac{t}{2},t]}\sup_{u\geq0} \frac{\mathbf P\big( X_r\in  u+\varDelta_1 \big)}{\overline{\nu}(\beta t)}\cdot  \int_0^{t/2} \sum_{k=0}^{[y]} \big(F_s(y-k,x) - F_s(y-x-k-2,x) \big)ds.   \qquad
 \end{eqnarray}
 By using Lemma~\ref{Lemma.203} with $\lambda =\beta/2$, then  \eqref{eqn.30} and the monotonicity of $\overline{\nu}$, we obtain that uniformly for all sufficiently  large $t$,
 \begin{eqnarray*}
 \sup_{r\in[\frac{t}{2},t]}\sup_{u\geq0} \frac{\mathbf P\big( X_r\in  u+\varDelta_1 \big) }{r\cdot \nu\big(\beta r+u +\varDelta_1\big)} 
 \leq\sup_{r\in[\frac{t}{2},t]}\sup_{u\geq \beta t/2} \frac{\mathbf P\big( X_r+\beta r\in  u+\varDelta_1 \big) }{r\cdot \nu\big(u +\varDelta_1\big)} \leq C
 \end{eqnarray*}
 and 
 \begin{eqnarray*}
 \sup_{r\in[\frac{t}{2},t]}\sup_{u\geq0} \frac{r\cdot \nu\big(\beta r+u +\varDelta_1\big)}{\overline{\nu}(\beta t) } 
 \leq C\cdot    \frac{ \overline{\nu}(\beta t/2)}{  \overline{\nu}(\beta t) } \leq C ,
 \end{eqnarray*}
 which follows that 
 \begin{eqnarray}\label{eqn.222}
  \sup_{r\in[\frac{t}{2},t]}\sup_{u\geq0} \frac{\mathbf P\big( X_r\in  u+\varDelta_1 \big) }{\overline{\nu}(\beta t) } \leq C.
 \end{eqnarray}
 Moreover, applying Fubini's theorem to the last integral in \eqref{eqn.208} and then using \eqref{eqn.2031},  
 \begin{eqnarray*}
 \lefteqn{\int_0^{t/2} \sum_{k=0}^{[y]} \big(F_s(y-k,x) - F_s(y-x-k-2,x) \big)ds}\!\!\!&\!\!\!&\cr
 \!\!\!&\leq\!\!\!& \sum_{k=0}^{[y]}   \int_0^\infty F_s(y-k,x)\,ds - \sum_{k=0}^{[y]}  \int_0^\infty F_s(y-x-k-2,x)\,ds\cr
 \!\!\!&=\!\!\!& \sum_{k=0}^{[y]}    V(y-k) \cdot \widehat{V}(x) 
 -  \sum_{k=0}^{[y]} V(y-x-k-2)\cdot \widehat{V}(x)  . 
 \end{eqnarray*} 
 which, by  the increase of $V$ and the fact that $V(\infty)<\infty$, can be bounded by 
 \begin{eqnarray*}
 \lefteqn{\widehat{V}(x)\sum_{k=0}^{[y]}    V\big([y]+1-k\big)  - \widehat{V}(x)\sum_{k=0}^{[y]}   V\big([y]-[x]-k-3\big)  }\!\!\!&\!\!\!&\cr
 \!\!\!&=\!\!\!& \widehat{V}(x)\bigg(\sum_{k=1}^{[y]+1}    V(k)  -  \sum_{k=-[x]-3}^{[y]-[x]-3}  V(k) \bigg)
 \leq  \widehat{V}(x)\cdot \sum_{k=[y]-[x]-2}^{[y]+1} V(k)  \leq   \widehat{V}(x) \cdot (x+4) \cdot V(\infty) ,
 \end{eqnarray*}
 uniformly in $y\geq 0$.  
 Taking this and \eqref{eqn.222} back into \eqref{eqn.208},  there exist some constants $C,t_0>0$ such that for all $t\geq t_0$,
 \begin{eqnarray*}
 \sup_{y\geq 0}\frac{I_{1}(t,y)}{\overline{\nu}(\beta t)}  \leq \frac{C}{t}
 \quad \mbox{and then}\quad 
 \sup_{y\geq 0} I_{1}(t,y) \leq C\cdot \frac{\overline{\nu}(\beta t)}{t}. 
 \end{eqnarray*} 
 
 \textbf{Case $i=2$.} We now consider $I_2(t,y)$.  
 By the definition of  $F_s(u,v)$, we have 
 \begin{eqnarray*}
 \sum_{k=0}^{[y]} \big(F_s(y-k,x) - F_s(y-x-k-2,x) \big)
 \!\!\!&=\!\!\!& \sum_{k=0}^{[y]} \mathbf{P}\big( y-x-k-2<\overline{X}_s\leq y-k,\, \overline{X}_s-X_s\leq x \big) \cr
 \!\!\!&=\!\!\!& \mathbf{E}\bigg[ \mathbf{1}_{\{\overline{X}_s-X_s\leq x \}}\cdot \sum_{k=0}^{[y]}  \mathbf{1}_{\{ y-x-k-2<\overline{X}_s\leq y-k \}} \bigg], 
 \end{eqnarray*}
 which  can be bounded by $\big([x]+4\big) \cdot \mathbf{P} \big(  \overline{X}_s-X_s\leq x\big) \leq (x+4 ) \cdot \mathbf{P}_x \big(\tau_0^->s\big)$ for all $y\geq 0$, since 
 \begin{eqnarray*}
 \sum_{k=0}^{[y]}  \mathbf{1}_{\{ y-x-k-2<z\leq y-k \}} \leq x+4,\quad 
 \forall\, y,z\geq 0.
 \end{eqnarray*} 
 Taking this back into the right-hand side of the last inequality in \eqref{eqn.207} and then plugging \eqref{eqn.207} into $I_2(t,y)$, we get

 \textbf{Case $i=3$.}

 \end{eqnarray*}

 F_r(y,x)-F_r(y-x-1,x) = \mathbf{P}\big(y-x-1<\overline{X}_r\leq y,\,

 \sup_{y\geq 0} I_{32}(t,y) \leq C \cdot \frac{\overline{\nu}(\beta t)}{t}.  
 \end{eqnarray*} 
 This along with \eqref{eqn,221} immediately yields the upper bound \eqref{eqn.2201} with $i=3$.

 	\begin{eqnarray*}
 	\bigg|
 	\frac{\underline{n}_t\big(x-y+\varDelta_\delta \big)}
 	{\nu\big(x+\varDelta_\delta\big)}
 	-\underline{n}(\zeta\wedge t)
 	\bigg|
 	\!\!\!&\leq\!\!\!&
 	\bigg|
 	\frac{\underline{n}_t\big(x-y+\varDelta_\delta \big)}
 	{\nu\big(x-y+\varDelta_\delta\big)}
 	-\underline{n}(\zeta\wedge t)
 	\bigg|
 	\cdot
 	\frac{\nu\big(x-y+\varDelta_\delta\big)}
 	{\nu\big(x+\varDelta_\delta\big)}
 	\cr\cr
 	\!\!\!&\!\!\!&
 	+
 	\bigg|
 	\frac{\nu\big(x-y+\varDelta_\delta\big)}
 	{\nu\big(x+\varDelta_\delta\big)}
 	-1
 	\bigg|
 	\cdot \underline{n}(\zeta\wedge t),
 	\end{eqnarray*}
 	which vanishes as $x\to\infty$ by Theorem~\ref{MainThm00} and Condition~\ref{Main.Condition}.

 	\begin{eqnarray}\label{eqn.3008}
 	\lim_{x\to\infty}
 	\bigg|
 	\frac{\underline{n}_{t-s}\big( x-y+z+\varDelta_\delta \big)}
 	{\nu\big(x+\varDelta_\delta\big)}
 	-\underline{n}\big( \zeta\wedge(t-s)\big)
 	\bigg|
 	=0 .
 	\end{eqnarray}
 	In view of \eqref{eqn.312}-\eqref{eqn.4020} and then using Condition~\ref{Main.Condition}, we have that for all sufficiently large $x$,
 	\begin{eqnarray*}
 	\lefteqn{
 		\sup_{s\in[0,t]} \sup_{z\in [0,y]}
 		\underline{n}\big(
 		\epsilon_{t-s}\in x-y+z+\varDelta_{\delta},\,
 		\mathcal{J}^{\lambda (x-y+z)}\leq t-s,\,
 		\zeta> t-s
 		\big)
 	}\cr\cr
 	\!\!\!&=\!\!\!&
 	\sup_{s\in[0,t]} \sup_{z\in [0,y]}
 	\underline{n}\big(
 	\epsilon_s\in x-z+\varDelta_{\delta},\,
 	\mathcal{J}^{\lambda (x-z)}\leq s,\,
 	\zeta> s
 	\big)
 	\cr
 	\!\!\!&\leq\!\!\!&
 	\sup_{s\in[0,t]} \sup_{z\in [0,y]}
 	\underline{n}\big(
 	\#\mathcal{B}_t^{\lambda (x-z)},\,
 	\epsilon_s\in x-z+\varDelta_{\delta},\,
 	\zeta>s
 	\big)
 	\leq C\cdot \nu\big(x+\varDelta_{\delta}\big).
 	\end{eqnarray*}
 	On the other hand, similarly as in the proof of Proposition~\ref{Prop.501}, we also can prove that for all sufficiently large $x$,
 	\begin{eqnarray*}
 	\sup_{s\in[0,t]} \sup_{z\in [0,y]}
 	\underline{n}\big(
 	\epsilon_{t-s}\in x-y+z+\varDelta_{\delta},\,
 	\mathcal{J}^{\lambda (x-y+z)}> t-s,\,
 	\zeta> t-s
 	\big)
 	\leq
 	C\cdot \nu\big(x+\varDelta_{\delta}\big)
 	\cdot \underline{n}(\overline\epsilon \geq \eta x/2).
 	\end{eqnarray*}
 	Note $\underline{n}(\overline\epsilon \geq \eta x/2)\to0$ as $x\to\infty$. Combining these two upper bounds gives that for all sufficiently large $x$,
 	\begin{eqnarray*}
 	\sup_{s\in[0,t]} \sup_{z\in [0,y]}
 	\underline{n}\big(
 	\epsilon_{t-s}\in x-y+z+\varDelta_{\delta},\,
 	\zeta> t-s
 	\big)
 	\leq C\cdot \nu\big(x+\varDelta_{\delta}\big).
 	\end{eqnarray*}
 	Armed with this and \eqref{eqn.3008}, we use the dominated convergence theorem to obtain
 	\begin{eqnarray*}
 	\lefteqn{\lim_{x\to\infty}\int_0^t ds\int_0^y \bigg|\frac{\underline{n}_{t-s}\big( x+z-y+\varDelta_\delta \big)}{\nu\big(x+\varDelta_\delta\big)}- \underline{n}\big( \zeta\wedge(t-s)\big)\bigg|\, \overline{n}_s(dz) }\cr 
 	\!\!\!&\leq\!\!\!&  \int_0^t ds\int_0^y \lim_{x\to\infty}\bigg|\frac{\underline{n}_{t-s}\big( x+z-y+\varDelta_\delta \big)}{\nu\big(x+\varDelta_\delta\big)}- \underline{n}\big( \zeta\wedge(t-s)\big)\bigg|\, \overline{n}_s(dz) =0.
 	\end{eqnarray*}
    Substituting this and \eqref{eqn.3008} into the right-hand side of \eqref{eqn.3005} proves the desired convergence for any fixed $\delta>0$. 
    We now extend this to hold uniformly for $\delta\in[\delta_0,\infty]$. 
    For any large integer $n\geq1/\delta_0$,

    \mathbf{P}_y\big(
    X_t\in x+\varDelta_\delta,\tau_0^->t
    \big) 
    \!\!\!&\leq\!\!\!&
    \sum_{k=0}^{[n\delta]}
    \mathbf{P}_y\big(
    X_t\in x+k/n+\varDelta_{1/n},\tau_0^->t
    \big).
    \end{eqnarray*}
    By the preceding convergence with $\delta=1/n$, there exists  a positive function $\rho_n$ on $\mathbb{R}_+$ decreasing to $0$ at infinity such that, for all $x>y$ and $k\geq0$,
    \begin{eqnarray*}
                            
                                            
                                       
               
    \mathbf{P}_y\big(
    X_t\in x+k/n+\varDelta_{1/n},\tau_0^->t
    \big) 
    \!\!\!&\leq\!\!\!&
    \big(1+\rho_n(x)\big)\,
    \mathbf{E}_y\big[t\wedge\tau_0^-\big]\,
    \nu\big(x+k/n+\varDelta_{1/n}\big).
    \end{eqnarray*}
    Putting these two estimates together yields
    \begin{eqnarray}\label{eqn.3022}
    \frac{
    	\mathbf{P}_y\big(
    	X_t\in x+\varDelta_\delta,\tau_0^->t
    	\big)}
    {\mathbf{E}_y\big[t\wedge\tau_0^-\big]\,
    	\nu\big(x+\varDelta_\delta\big)}
    \!\!\!&\leq\!\!\!&
    \big(1+\rho_n(x)\big)
    \frac{
    	\nu\big(x+\varDelta_{\delta+1/n}\big)}
    {\nu\big(x+\varDelta_\delta\big)}. 
    \end{eqnarray}
  Since $\delta\to \nu\big(x+\varDelta_\delta\big)$ is non-decreasing, by \eqref{eqn.30} we have 
    \begin{eqnarray*}
    	\limsup_{x\to\infty}
    	\sup_{\delta\in[\delta_0,\infty)}
    	\frac{
    		\mathbf{P}_y\big(
    		X_t\in x+\varDelta_\delta,\tau_0^->t
    		\big)}
    	{\mathbf{E}_y\big[t\wedge\tau_0^-\big]\,
    		\nu\big(x+\varDelta_\delta\big)} 
    \!\!\!&\leq\!\!\!&
    1+
    \limsup_{x\to\infty}
    \sup_{\delta\in[\delta_0,\infty)}
    \frac{
    	\nu\big(x+\delta+\varDelta_{1/n}\big)}
    {\nu\big(x+\varDelta_{\delta_0}\big)}
    \cr\cr
    \!\!\!&\leq\!\!\!&
    1+
    \limsup_{x\to\infty}
    \sup_{u\geq x}
    \frac{\nu\big(u+\varDelta_{1/n}\big)}
    {\nu\big(x+\varDelta_{1/n}\big)}
    \cdot
    \frac{\nu\big(x+\varDelta_{1/n}\big)}
    {\nu\big(x+\varDelta_{\delta_0}\big)}
    \leq 
    1+\frac{1}{n\delta_0}.
    \end{eqnarray*}
    On the other hand, an analogous argument gives
    \begin{eqnarray*}
    	\liminf_{x\to\infty}
    	\inf_{\delta\in[\delta_0,\infty)}
    	\frac{
    		\mathbf{P}_y\big(
    		X_t\in x+\varDelta_\delta,\tau_0^->t
    		\big)}
    	{\mathbf{E}_y\big[t\wedge\tau_0^-\big]\,
    		\nu\big(x+\varDelta_\delta\big)}
    \geq
    \liminf_{x\to\infty}
    \inf_{\delta\in[\delta_0,\infty)}
    \frac{
    	\nu\big(x+\varDelta_{\delta-1/n}\big)}
    {\nu\big(x+\varDelta_\delta\big)}
    \geq
    1-\frac{1}{n\delta_0}.
    \end{eqnarray*}
    Combining these two estimates together and then letting $n\to\infty$ yields the desired uniform asymptotic equivalence for $\delta\in[\delta_0,\infty)$. 
    For $\delta=\infty$, the right-hand side of \eqref{eqn.3022} reduces to $1+\rho_n(x)$ which goes to $1$ as $x\to\infty$.  
    The corresponding lower bound follows by the same argument.
    This completes the whole proof.

We only provide the detailed proof of the first uniform asymptotic equivalence in \eqref{eqn.MainThm01}. The second follows, uniformly in $x\geq 0$ and $\delta\in[\delta_0,\infty]$, from the standard discretization argument, together with \eqref{eqn.30} and the monotonicity and boundedness of $V$. We omit the details.
 
 We first prove that the first asymptotic equivalence \eqref{eqn.MainThm01} holds uniformly in $x\geq0$ for each fixed $\delta>0$. 
 By \eqref{eqn.30}, we have as $t\to\infty$,
 \begin{eqnarray*}
  \sup_{x\geq 0}\sup_{z\in x+\varDelta_\delta}\bigg|  \frac{\overline{\nu}(\beta t +z)/(\beta t +z)}{\overline{\nu}(\beta t +x)/(\beta t +x)} -1\bigg| \to 0.
 \end{eqnarray*}
 Since $V(x)>0$ when $x>0$ and $V(\infty)<\infty$; see \eqref{eqn.2011},  then
 \begin{eqnarray}\label{eqn.201}
 0<\inf_{x\geq 0} \int_x^{x+\delta} V(z)\,dz \leq \sup_{x\geq 0} \int_x^{x+\delta} V(z)\,dz\leq V(\infty)\cdot \delta<\infty. 
 \end{eqnarray}
 These two results indicate that we just need to prove that as $t\to\infty$, 
\begin{eqnarray*}
\sup_{x\geq 0}\bigg| \frac{\underline{n}_t\big( x+\varDelta_\delta\big)}{\alpha\cdot  \overline{\nu}(\beta t +x)/(\beta t +x) }-   \int_x^{x+\delta} V(z)\,dz\bigg|\to 0. 
\end{eqnarray*}
For any  $T>0$ and $\eta \in \big(0, \frac{\alpha\wedge 2+1}{8(\alpha+1)} \big)$, by the triangle inequality this supremum can be bounded by 
\begin{eqnarray*}
\sup_{x\geq 0} \frac{ \underline{n}\big(\epsilon_t\in x+\varDelta_\delta,\,\mathcal{J}^{\eta(\beta t+x)}>T,\, \zeta >t\big)}{\alpha\cdot  \overline{\nu}(\beta t +x)/(\beta t +x) } 
+ \sup_{x\geq 0}\bigg| \frac{\underline{n}\big(\epsilon_t\in x+\varDelta_\delta,\,\mathcal{J}^{\eta(\beta t+x)}\leq T,\, \zeta >t\big)}{\alpha\cdot  \overline{\nu}(\beta t +x)/(\beta t +x) }-\int_x^{x+\delta} V(z)\,dz\bigg| ,
\end{eqnarray*}
which goes to $0$ as $t\to\infty$ and then $T\to \infty$ by Lemma~\ref{Lemma.401} and \ref{Lemma.402}. 
Hence \eqref{eqn.MainThm01} holds uniformly in $x\geq0$ for each fixed $\delta>0$. 
We now extend it to hold uniformly for  $\delta\in [\delta_0,\infty]$. 
For any large integer $n\geq 1/\delta_0$, we have 
\begin{eqnarray}\label{eqn.4001}
\sum_{k=0}^{[n\delta]-1} \underline{n}_t\big( x+k/n+\varDelta_{1/n}\big)\leq \underline{n}_t\big( x+\varDelta_\delta\big) 
\!\!\!&\leq\!\!\!& \sum_{k=0}^{[n\delta]} \underline{n}_t\big( x+k/n+\varDelta_{1/n}\big).
\end{eqnarray}
In view of our previous result, there exists a positive function $\rho_n$ on $\mathbb{R}_+$ that decreases to $0$ at infinity and satisfies,   for all $x\geq 0$ and $k\geq 0$, 
\begin{eqnarray*}
\underline{n}_t\big(x+k/n+\varDelta_{1/n}\big) 
\!\!\!&\geq\!\!\!& \big( 1-\rho_n(t) \big)  \cdot  \alpha\cdot\int_{x+k/n}^{x+k/n+1/n} \frac{\overline{\nu}(\beta t+z)}{\beta t+z} \cdot V(z)\, dz 
\end{eqnarray*}
and 
\begin{eqnarray*}
\underline{n}_t\big(x+k/n+\varDelta_{1/n}\big) 
\!\!\!&\leq\!\!\!& \big( 1+\rho_n(t) \big)  \cdot  \alpha\cdot\int_{x+k/n}^{x+k/n+1/n} \frac{\overline{\nu}(\beta t+z)}{\beta t+z} \cdot V(z)\, dz.
\end{eqnarray*}
Substituting these two estimates into the lower and upper bounds in \eqref{eqn.4001} respectively yields that 
\begin{eqnarray*}
\underline{n}_t\big( x+\varDelta_\delta\big) 
\!\!\!&\geq\!\!\!&  \alpha\cdot \big( 1-\rho_n(t) \big)  \cdot \sum_{k=0}^{[n\delta]-1} \int_{x+k/n}^{x+k/n+1/n} \frac{\overline{\nu}(\beta t+z)}{\beta t+z} \cdot V(z)\, dz\cr
\!\!\!&\geq\!\!\!& \alpha\cdot \big( 1-\rho_n(t) \big)  \cdot \bigg( \int_{x}^{x+\delta} \frac{\overline{\nu}(\beta t+z)}{\beta t+z} \cdot V(z)\, dz - \int_{x+\delta-1/n}^{x+\delta} \frac{\overline{\nu}(\beta t+z)}{\beta t+z} \cdot V(z)\, dz \bigg)
\end{eqnarray*}
and
\begin{eqnarray*}
\underline{n}_t\big( x+\varDelta_\delta\big) 
\!\!\!&\leq\!\!\!&  \alpha\cdot \big( 1+\rho_n(t) \big)  \cdot \sum_{k=0}^{[n\delta]} \int_{x+k/n}^{x+k/n+1/n} \frac{\overline{\nu}(\beta t+z)}{\beta t+z} \cdot V(z)\, dz\cr
\!\!\!&\leq\!\!\!& \alpha\cdot \big( 1+\rho_n(t) \big)  \cdot  \bigg(   \int_{x}^{x+\delta} \frac{\overline{\nu}(\beta t+z)}{\beta t+z} \cdot V(z)\, dz
+  \int_{x+\delta}^{x+\delta+1/n} \frac{\overline{\nu}(\beta t+z)}{\beta t+z} \cdot V(z)\, dz\big).
\end{eqnarray*}
Since $\rho_n(t)\to0$ as $t\to\infty$, these two estimates yield that the desired asymptotic equivalence \eqref{eqn.MainThm01} with $\delta=\infty$ holds uniformly for $x\geq0$. 
Moreover, since $\delta\mapsto  \underline{n}_t\big( x+\varDelta_\delta\big)$ is non-decreasing, our first result shows that for all sufficiently large $t$ and all $x\geq 0$,
\begin{eqnarray*}
\inf_{\delta\in [\delta_0,\infty)} \underline{n}_t\big( x+\varDelta_\delta\big)
 =  \underline{n}_t\big( x+\varDelta_{\delta_0}\big) \geq C\cdot  \frac{\overline{\nu}(\beta t+x)}{\beta t+x}. 
\end{eqnarray*} 
Hence the preceding lower and upper bounds also indicate that \eqref{eqn.MainThm01} holds uniformly in $x\geq0$ and  $\delta\in [\delta_0,\infty)$ if we can identify
\begin{eqnarray}\label{eqn.4002}
\limsup_{n\to\infty} \limsup_{t\to\infty} \sup_{x\geq 0}\sup_{\delta\in [\delta_0,\infty)}
\frac{\beta t+x}{\overline{\nu}(\beta t+x)} \int_{x+\delta-1/n}^{x+\delta+1/n} \frac{\overline{\nu}(\beta t+z)}{\beta t+z} \cdot V(z)\, dz =0.
\end{eqnarray}
Actually, since $V(z)\leq V(\infty)<\infty$ and $\overline\nu \in \mathrm{RV}^\infty_{-\alpha}$ is non-increasing, we have as $t\to\infty$, 
\begin{eqnarray*}
\sup_{x\geq 0}\sup_{\delta\in [\delta_0,\infty)}
\frac{\beta t+x}{\overline{\nu}(\beta t+x)} \int_{x+\delta-\frac{1}{n}}^{x+\delta+\frac{1}{n}} \frac{\overline{\nu}(\beta t+z)}{\beta t+z} \cdot V(z)\, dz  
\leq \frac{2V(\infty)}{n} \cdot \sup_{x\geq \beta t} \frac{x}{\overline{\nu}(x)}\frac{\overline{\nu}(x-1)}{x-1}
\to \frac{2V(\infty)}{n},
\end{eqnarray*}
which vanishes as $n\to\infty$  and then the limit \eqref{eqn.4002} holds. This completes the proof.

For each integer $n\geq1$, define the step-function approximation of $f$:
\begin{eqnarray*}
f_n(z):= \sum_{k=0}^\infty f(k/n) \cdot \mathbf{1}_{\{ z\in(\frac{k}{n},\frac{k+1}{n}] \}},\quad z\geq0.

For any $R>0$, recall the three quantities   $\widetilde{I}_1^{y,\delta}(x)$,  $\widetilde{I}_2^{y,\delta}(R,x)$ and $\widetilde{I}_3^{y,\delta}(R,x)$ defined in Lemma~\ref{Lemma.601}-\ref{Lemma.603}, and also define 
\begin{eqnarray*} 
\widetilde{I}_4^{y,\delta}(R,x)  \!\!\!&:=\!\!\!& \alpha \cdot  \int_R^\infty ds\int_0^y    \overline{n}_s(dz) \int_{(x-y+z)^+}^{(x-y+z+\delta)^+}V(u)\,du,\cr
\widetilde{I}_5^{y,\delta}(R,x)  \!\!\!&:=\!\!\!& \alpha \cdot   \int_0^y\widehat{V}(u) \,du  \int_R^\infty\underline{n}_s\big( ((x-y+u)^+,(x-y+u+\delta)^+]\big)\,ds. 
\end{eqnarray*}
 Consider the following function 
\begin{eqnarray*}
F(z) := \alpha	\int_{(x-y+z)^+}^{(x-y+z+\delta)^+}V(u)\, du, \quad  0\leq z\leq y.
\end{eqnarray*}
One can identify that $F$ is absolutely continuous on $(0,y)$   with   Radon--Nikodym derivative 
\begin{eqnarray*}
 F'(z) \!\!\!&=\!\!\!& \alpha\Big(V\big((x-y+z+\delta)^+\big) -V\big( (x-y+z)^+\big)\Big)\cr 
 \!\!\!&=\!\!\!& \alpha\int_0^\infty \underline{n}_s\big( ((x-y+z)^+,(x-y+z+\delta)^+] \big)\,ds.
\end{eqnarray*}
This induces that 
\begin{eqnarray*}
\widetilde{I}_3^{y,\delta}(R,x) + \widetilde{I}_5^{y,\delta}(R,x)
\!\!\!&=\!\!\!&  \int_0^y \widehat V(z)F'(z)\,dz  = \int_0^y \widehat V(z)\,dF(z).
\end{eqnarray*}
In addition,  in view of \eqref{eqn.447},  we have
\begin{eqnarray*}
\overline{\tt d}\cdot  \widetilde{I}_1^{y,\delta}(x) +  \widetilde{I}_2^{y,\delta}(R,x) +  \widetilde{I}_4^{y,\delta}(R,x)	
\!\!\!&=\!\!\!& \overline{\tt d}\cdot F(0)+ \int_0^\infty ds 	\int_0^y F(z)\,\overline{n}_s(dz)
=   \int_{[0,y]}F(z)\widehat V(dz).
\end{eqnarray*} 
Combining these two equalities  together  yields 
\begin{eqnarray*}
\lefteqn{\overline{\tt d}\cdot  \widetilde{I}_1^{y,\delta}(x) + \widetilde{I}_2^{y,\delta}(R,x) + \widetilde{I}_3^{y,\delta}(R,x) +  \widetilde{I}_4^{y,\delta}(R,x)  +  \widetilde{I}_5^{y,\delta}(R,x) }\cr\cr
\!\!\!&=\!\!\!&  \int_{[0,y]}F(z)\widehat V(dz)+\int_0^y\widehat V(z)\,dF(z)
= \widehat V(y)\cdot F(y) =\alpha \cdot \widehat{V}(y)\cdot  \int_x^{x+\delta}V(u)du.
\end{eqnarray*}
Comparing this with \eqref{eqn.605} and appealing to Lemmas~\ref{Lemma.601}--\ref{Lemma.604}, we conclude that the limit \eqref{eqn.602} follows immediately provided the following two limits are proved, 
\begin{eqnarray}\label{eqn.603}
\limsup_{R\to\infty}\sup_{x\geq 0} \widetilde{I}_4^{y,\delta}(R,x)   =0 
\quad \mbox{and}\quad 
\limsup_{R\to\infty}\sup_{x\geq 0} \widetilde{I}_5^{y,\delta}(R,x)  = 0 . 
\end{eqnarray}
Indeed, since $V(z)\le V(\infty)<\infty$ for all $z\ge0$, we have
\begin{eqnarray*}
\sup_{x\geq 0} \widetilde{I}_4^{y,\delta}(R,x)  \!\!\!&\leq\!\!\!& \alpha \delta V(\infty) \int_R^\infty     \overline{n}_s\big([0,y]\big) \, ds,
\end{eqnarray*}
which tends to $0$ as $R\to\infty$, since $\int_0^\infty \overline{n}_s([0,y])\,ds<\infty$; see \eqref{eqn.447}. 
Moreover, we also have 
\begin{eqnarray*}
\sup_{x\geq 0} \widetilde{I}_5^{y,\delta}(R,x)  \leq \alpha  \int_0^y\widehat{V}(u) \,du \cdot \int_R^\infty     \underline{n}(\zeta >s) \, ds
\end{eqnarray*}
which also tends to $0$ as $R\to\infty$, since $\int_0^\infty \underline{n}(\zeta>s)\,ds = \underline{n}(\zeta)<\infty$; see \eqref{eqn.501}.
Thus the two limits in \eqref{eqn.603} hold, and consequently the limit \eqref{eqn.602} follows. This completes the proof.

Let $\widetilde{X}_t=X_t+\beta t$ for $t\geq 0$. 
The L\'evy--It\^{o}
decomposition theorem allows us to represent $\widetilde{X}$ as  
\begin{eqnarray*}
\widetilde X_t= M_t+Q_t^{(x)}  + \int_0^t \int_{\eta(\beta t+x)}^{\infty}
y\,N(ds,dy) - t\cdot \int_{\eta(\beta t+x)}^\infty y\,\nu(dy),
\end{eqnarray*}
where the two processes $M$ and $Q^{(x)}  $ are given by
\begin{eqnarray*}
M_t :=  \sigma B_t + \int_0^t
\int_{-\infty}^1 y\,\widetilde N(ds,dy)
\quad\mbox{and}\quad
Q_t^{(x)}  := \int_0^t \int_1^{\eta(\beta t+x)} y\,\widetilde{N}(ds,dy),\quad t\geq 0 . 
\end{eqnarray*} 
Here $B$ is a standard Brownian motion, $N(ds,dy)$ is a Poisson random measure on $(0,\infty)\times \mathbb{R}$ with intensity $ds\,\nu(dy)$ and $\widetilde{N}(ds,dy):=N(ds,dy)-ds\,\nu(dy)$. 
For any $t> 0$, conditioned on the event $\{\mathcal{J}^{\eta(\beta t+x)}>t\}$, there is no positive jump larger than $\eta(\beta t+x)$ before time $t$. 
This implies that $\widetilde{X}_t\le M_t+Q_t^{(x)}$, and therefore 
\begin{eqnarray*}
\mathbf{P}\big(X_t\geq x, \mathcal{J}^{\eta(\beta t+x)}>t \big)=
\mathbf{P}\big(\widetilde{X}_t\geq \beta t +x,\mathcal{J}^{\eta(\beta t+x)}>t\big) 
\leq 
\mathbf P\Big( M_t\geq \frac{\beta t+x}{2} \Big)
+ \mathbf P\Big(Q_t^{(x)}  \geq \frac{\beta t+x}{2}\Big).
\end{eqnarray*}
Consequently, it suffices to prove that as $t\to\infty$,
\begin{eqnarray}\label{eqn.3001}
\sup_{x\geq-(\beta-\lambda)t} \frac{\mathbf P\big( M_t\geq (\beta t+x)/2 \big)}{\overline{\nu}(\beta t+x)/(\beta t +x)} \to 0
\quad \mbox{and} \quad
\sup_{x\geq-(\beta-\lambda)t} \frac{\mathbf P\big( Q_t^{(x)}  \geq (\beta t+x)/2 \big)}{\overline{\nu}(\beta t+x)/(\beta t +x)}\to 0. 
\end{eqnarray}

\textbf{(i)} Since $M$ is a  L\'evy process with zero mean and all jumps bounded above by $1$, Theorem~25.17 in \cite[ p.~165]{Sato1999} implies that it has finite exponential moments of all non-negative orders, i.e., 
\begin{eqnarray*}
\varPsi_M(u):= \log\mathbf E\big[\exp\{uM_1\}\big] <\infty,\quad \forall\, u \geq 0
\end{eqnarray*}
and $\varPsi_M(0+)=\varPsi_M'(0+)=0$. Thus there exists a constant $u_\lambda>0$ such that $\varPsi_M(u_\lambda)\leq \lambda u_\lambda/4$. 
By Chebyshev's inequality, for all  $x\geq -(\beta-\lambda)t$,
\begin{eqnarray*}
\mathbf P\big( M_t\geq (\beta t+x)/2 \big)
\!\!\!&\leq\!\!\!& \mathbf{E}\Big[ \exp\big\{	u_\lambda M_t \big\} \Big] \cdot  \exp\Big\{	  - \frac{u_\lambda}{2}\cdot (\beta t+x)\Big\} \cr\cr
\!\!\!&=\!\!\!&  
\exp\Big\{	t\cdot \varPsi_M(u_\lambda) - \frac{u_\lambda}{2}\cdot (\beta t+x)\Big\}\cr \cr
\!\!\!&\leq\!\!\!&
\exp\Big\{ \frac{\lambda\cdot u_\lambda }{4}\cdot t
-\frac{u_\lambda}{2}\cdot (\beta t+x) \Big\} 
\leq 
\exp\Big\{
-\frac{u_\lambda }{4}\cdot (\beta t+x)
\Big\}.
\end{eqnarray*} 
Since $\beta t+x\ge\lambda t$ and $\overline{\nu}\in\mathrm{RV}_{-\alpha}^\infty$, we have, as $t\to\infty$, 
\begin{eqnarray*}
\sup_{x\geq-(\beta-\lambda)t}\frac{\mathbf{P}\big( M_t\geq (\beta t+x)/2 \big)}{ \overline{\nu}(\beta t+x)/(\beta t +x) } 
\leq   \sup_{x\geq-(\beta-\lambda)t}\frac{	\exp\big\{- u_\lambda/4\cdot (\beta t+x) \big\}}{ \overline{\nu}(\beta t+x)/(\beta t +x) }  \to 0 . 
\end{eqnarray*} 
                                                 

\textbf{(ii)}  For any integer $n\geq 1$,  the time-orthogonality of $\widetilde{N}(ds,dy)$ allows us to decompose $Q_t^{(x)}  $ into the following $n$ i.i.d. parts:
\begin{eqnarray*}
Q_{t,n,j}^{(x)}  := \int_{(j-1)t/n}^{jt/n} \int_1^{\eta(\beta t+x)} y\, \widetilde{N}(ds,dy),\quad j=1,\cdots, n.  
\end{eqnarray*}
Choose a constant $\kappa \in \big(4\eta(\alpha+1)+1,2\wedge \alpha\big)$. 
By the Fuk--Nagaev inequality; see Corollary 1.6 in
\cite{Nagaev1979},  there exists a constant $C>0$ such that for all $n\geq 1$, all sufficiently large $t$ and $x\geq -(\beta-\lambda)t$,
\begin{eqnarray}\label{eqn.421}
\mathbf P\Big(Q_t^{(x)}  \geq \frac{\beta t+x}{2}\Big)
\!\!\!&=\!\!\!& \mathbf{P}\bigg(  \sum_{j=1}^nQ_{t,n,j}^{(x)}  \geq \frac{\beta t+x}{2} \bigg)\cr
\!\!\!&\leq\!\!\!& \sum_{j=1}^n \mathbf{P}\big(  Q_{t,n,j}^{(x)}  \geq \eta(\beta t+x) \big)
+ C\cdot \bigg( \frac{ 1}{ (\beta t+x)^\kappa} \cdot \sum_{j=1}^n \mathbf{E}\Big[ \big|Q_{t,n,j}^{(x)} \vee 0\big|^\kappa\Big] \bigg)^{1/(4\eta)} \cr
\!\!\!&=\!\!\!& n\cdot \mathbf{P}\big(  Q_{t,n,1}^{(x)} \geq \eta(\beta t+x) \big)
+  \frac{C}{ (\beta t+x)^{\frac{\kappa}{4\eta}}} \cdot \bigg(n\cdot \mathbf{E}\Big[ \big|Q_{t,n,1}^{(x)}\big|^\kappa\Big]\bigg)^{1/(4\eta)} . 
\end{eqnarray}
Using the Burkholder-Davis-Gundy inequality, then the inequality $(a+b)^{\kappa/2}\leq a^{\kappa/2}+b^{\kappa/2}$ for all $a,b\geq 0$ and finally \eqref{eqn.LevyMeasure}, there exists a constant $C>0$ such that, for all $n\geq 1$, all sufficiently large $t$ and $x\geq -(\beta-\lambda)t$,
\begin{eqnarray*}
n\cdot \mathbf{E}\Big[ \big|Q_{t,n,1}^{(x)}\big|^\kappa\Big] 
\!\!\!&\leq\!\!\!& C\cdot  n\cdot \mathbf E\bigg[ \Big(  \int_0^{t/n} \int_1^{\eta(\beta t+x)} y^2\,N(ds,dy)\Big)^{\kappa/2} \bigg]\cr
\!\!\!&\leq\!\!\!& C\cdot n\cdot \mathbf E\bigg[  \int_0^{t/n} \int_1^{\eta(\beta t+x)} y^\kappa\,N(ds,dy) \bigg] \cr
\!\!\!&=\!\!\!& C\cdot t\cdot \int_1^{\eta(\beta t+x)}y^\kappa\,\nu(dy) \cr\cr
\!\!\!&\leq\!\!\!& C\cdot  t . 
\end{eqnarray*}
Moreover, the definition of $ Q_{t,n,1}^{(x)}$ and the fact $\nu\big((1,\eta(\beta t+x)] \big)<\infty$ show that 
\begin{eqnarray*}
Q_{t,n,1}^{(x)} = \int_0^{t/n} \int_1^{\eta(\beta t+x)} y\,N(ds,dy) - \frac{t}{n}\cdot  \int_1^{\eta(\beta t+x)} y\,\nu(dy). 
\end{eqnarray*}
This implies that, for all $n\geq 1$, all sufficiently large $t$ and $x\geq -(\beta-\lambda)t$,
\begin{eqnarray*}
\big\{ Q_{t,n,1}^{(x)} \geq \eta(\beta t+x)\big\} \subset \big\{ N\big((0,t/n],(1,\eta(\beta t+x)] \big) \geq 2 \big\} \subset \big\{ N\big((0,t/n],(1, \infty) \big) \geq 2 \big\} .
\end{eqnarray*} 
Since $N((0,t/n],(1,\infty))$ is Poisson distributed with rate $(t/n)\overline{\nu}(1)$, we have
\begin{eqnarray*}
\sup_{x\geq -(\beta-\lambda)t}  n\cdot \mathbf{P}\big(  Q_{t,n,1}^{(x)}\geq \eta(\beta t+x) \big)
\!\!\!&\leq\!\!\!&    n\cdot \mathbf{P}\big(  N \big((0,t/n],(1,\infty) \big) \geq 2\big)
\leq n\cdot \big(t/n \cdot \overline{\nu}(1) \big)^2 .
\end{eqnarray*} 
Substituting these two estimates into \eqref{eqn.421} and then using the arbitrariness of $n$, we obtain that, uniformly in  sufficiently large $t$ and $x\geq -(\beta-\lambda)t$,
\begin{eqnarray*}
\mathbf P\Big(Q_t^{(x)}  \geq \frac{\beta t+x}{2}\Big)
\leq C\cdot (\beta t+x)^{-\frac{\kappa-1}{4\eta}}.
\end{eqnarray*}
This induces that 
\begin{eqnarray*}
\sup_{x\geq-(\beta-\lambda)t} \frac{\mathbf P\big( Q_t^{(x)}\geq (\beta t+x)/2 \big)}{\overline{\nu}(\beta t+x)/(\beta t +x)} \leq C\cdot  \sup_{x\geq-(\beta-\lambda)t}  \frac{(\beta t+x)^{-\frac{\kappa-1}{4\eta}+1}}{\overline{\nu}(\beta t+x)},
\end{eqnarray*}
which goes to $0$ as $t\to\infty$, since $\overline{\nu}(z)\in\mathrm{RV}_{-\alpha}^\infty$ and $\alpha<\frac{\kappa-1}{4\eta}-1$. The second limit in \eqref{eqn.3001}  is proved.

By the Markov property, the first supremum in \eqref{eqn.3004} equals

\begin{eqnarray}\label{eqn.3003}

\quad \mbox{and}\quad
x-z \geq -\frac{1+\eta}{2}\cdot \beta (t-s) .
\end{eqnarray}
This, together with $\overline{\nu}\in\mathrm{RV}_{-\alpha}^\infty$, yields that, uniformly for sufficiently large $t$, all $x\ge0$, $s\in[0,T]$, and $1/2\le z\le 1+\eta(\beta t+x)$,
\begin{eqnarray*}

and

which can be bounded by

An argument analogous to (but simpler than) that in \textbf{(i)} shows that the preceding supremum tends to $0$ as $t\to\infty$. 
Consequently, the second supremum in \eqref{eqn.3004} also vanishes as $t\to\infty$. 
Combining the limits obtained in these two steps yields the limit \eqref{eqn.4005}. 


\textbf{Part~II.} We now prove the limit \eqref{eqn.4006}. 
On the event $\{T<\mathcal{J}^{\eta(\beta t+x)}\le t,\ \zeta>t\}$, at least one jump of size larger than $\eta(\beta t+x)$ occurs during $[T,t]$. 
Hence 
\begin{eqnarray*}
\underline{n}\big(\epsilon_t\in x+\varDelta_\delta, T<\mathcal{J}^{\eta(\beta t+x)}\leq t,\zeta >t\big) 
\!\!\!&\leq\!\!\!& \underline{n}\bigg( \sum_{T<s\leq t} \mathbf{1}_{\{ \Delta\epsilon_s> \eta(\beta t+x),\, \epsilon_t\in x+\varDelta_\delta,\, \zeta>t \}}  \bigg).
\end{eqnarray*}
Using the Markov property and the compensation formula \eqref{eqn.Compensation} to the right-hand side yields that  
\begin{eqnarray}\label{eqn.3006}
\lefteqn{\underline{n}\bigg( \sum_{T<s\leq t} \mathbf{1}_{\{ \Delta\epsilon_s> \eta(\beta t+x),\, \epsilon_t\in x+\varDelta_\delta,\, \zeta>t \}}  \bigg)}\cr
\!\!\!&=\!\!\!& \int_T^tds
\int_0^\infty \underline{n}_s(dz) \int_{\eta(\beta t+x)}^\infty   \mathbf{P}_{z+w}\big( X_{t-s}\in x+\varDelta_\delta,\,\tau_0^->t-s \big) \,\nu(dw)  \cr
\!\!\!&\leq\!\!\!& \int_T^tds
\int_0^\infty \underline{n}_s(dz) \int_{\eta(\beta t+x)}^\infty   \mathbf{P} \big( X_{t-s}\in x-z-w+\varDelta_\delta  \big) \,\nu(dw)  \cr
\!\!\!&\leq \!\!\!&  \int_T^\infty \underline{n}(\zeta >s)\,ds \cdot \sup_{r\in[0,t]} \sup_{z\geq 0}  \int_{\eta(\beta t+x)}^\infty   \mathbf{P} \big( X_r-x+z\in -w+\varDelta_\delta \big) \,\nu(dw)  .
\end{eqnarray}
By Proposition~\ref{Prop.403},  there exist two constants $C,t_0>0$ such that for all $t\geq t_0$, 
\begin{eqnarray*}
\sup_{r\in[0,t]} \sup_{z\geq 0}  \int_{\eta(\beta t+x)}^\infty   \mathbf{P} \big( X_r-x+z\in -w+\varDelta_\delta \big) \,\nu(dw)  
\leq  C\cdot \frac{\overline\nu\big(\eta(\beta t+x) \big)}{\eta(\beta t+x) } \cdot \delta .
\end{eqnarray*}

\sup_{x\geq 0} \frac{\underline{n}\big(\epsilon_t\in x+\varDelta_\delta, T<\mathcal{J}^{\eta(\beta t+x)}\leq t,\zeta >t\big)}{ \overline{\nu}(\beta t+x)/ (\beta t +x)  } 
\leq C\cdot  \int_T^\infty \underline{n}(\zeta >s)\,ds,

 which goes to $0$ as $T\to\infty$ since $\underline{n}(\zeta)<\infty$; see \eqref{eqn.501}.

Since all jumps occurring before $\mathcal{J}^{\eta(\beta t+x)}$ are  not larger than $\eta(\beta t+x) $, we have  for any $t>T$,
 \begin{eqnarray*}
 \underline{n}\big( \epsilon_t\in x+\varDelta_\delta,\mathcal{J}^{\eta(\beta t+x)}\leq T,\zeta>t \big) 
 \!\!\!&=\!\!\!&  \underline{n}\bigg( \sum_{0<s\leq T} \mathbf{1}_{\{ \Delta\epsilon_s> \eta(\beta t+x),\,  \sup_{0<r<s}\Delta\epsilon_r \leq\eta(\beta t+x),\, \epsilon_t\in x+\varDelta_\delta,\,\zeta >t \}}   \bigg)  ,
 \end{eqnarray*}
 which can be further decomposed into the following two terms
 \begin{eqnarray*}
 \quad \underline{n}\bigg( \sum_{0<s\leq T} \mathbf{1}_{\{ \Delta\epsilon_s > \eta(\beta t+x),\,   \epsilon_t\in x+\varDelta_\delta,\,\zeta >t \}}   \bigg) 
 -\underline{n}\bigg( \sum_{0<s\leq T} \mathbf{1}_{\{ \Delta\epsilon_s > \eta(\beta t+x),\,  \sup_{0<r<s}\Delta\epsilon_r >\eta(\beta t+x),\, \epsilon_t\in x+\varDelta_\delta,\,\zeta >t \}}   \bigg) .
 \end{eqnarray*}
	 We first consider the second term. Note that on the event $\{\Delta\epsilon_s> \eta(\beta t+x),\,  \sup_{0<r<s}\Delta\epsilon_r >\eta(\beta t+x) \}$,  there always exist at least two jumps larger than $\eta(\beta t+x)$ occurring during $(0,T\wedge \zeta]$. 
 Hence it can be bounded by   
 \begin{eqnarray*}
 \underline{n}\bigg( \sum_{0<s<r\leq T\wedge \zeta} \mathbf{1}_{\{ \Delta\epsilon_s > \eta(\beta t+x),\,   \Delta\epsilon_r >\eta(\beta t+x)   \}}   \bigg) 
 \leq  T\cdot \underline{n}(\zeta)\cdot \Big|\overline{\nu}\big( \eta(\beta t+x) \big)\Big|^2 ,
 \end{eqnarray*}
 which is  $o\big( \overline\nu(\beta t+x)/(\beta t+x) \big)$.
 Here this inequality follows from Proposition~\ref{Prop.203}. 
 Thus the contribution of the second term can be ignored. 
 Consequently, it suffices to prove that as $t\to\infty$, 
 \begin{eqnarray}\label{eqn.3007}
  \sup_{x\geq0} \bigg| \frac{   \underline{n}\big( \sum_{0<s\leq T} \mathbf{1}_{\{ \Delta\epsilon_s > \eta(\beta t+x),\,   \epsilon_t\in x+\varDelta_\delta,\,\zeta >t \}}   \big)}{\alpha \cdot \overline\nu(\beta t+x)/(\beta t+x)} -\frac{\underline n(\zeta\wedge T)}{ \underline n(\zeta)  } \int_x^{x+\delta}V(z)\, dz  \bigg| \to 0. 
 \end{eqnarray}
By using the Markov property and the compensation formula \eqref{eqn.Compensation} again, we have 
\begin{eqnarray*}
\lefteqn{ \underline{n}\bigg( \sum_{0<s\leq T} \mathbf{1}_{\{ \Delta\epsilon_s > \eta(\beta t+x),\,   \epsilon_t\in x+\varDelta_\delta,\,\zeta >t \}}   \bigg) }\cr
\!\!\!&=\!\!\!&  \int_0^T ds
\int_0^\infty   \underline{n}_s(dz) 
\int_{\eta(\beta t+x)}^\infty \mathbf P_{z+w}\big( X_{t-s}\in x+\varDelta_\delta,\tau_0^->t-s \big)  \nu(dw).
\end{eqnarray*} 
For any fixed $\kappa\in(0,1/3)$, we decompose the right-hand side into the following two parts:
\begin{eqnarray*}
I_1(t,x,\kappa)\!\!\!&:=\!\!\!&  \int_0^T ds \int_0^\infty  \underline{n}_s(dz) \int_{B(t,x,\kappa)}  \mathbf P_{z+w}\big( X_{t-s}\in x+\varDelta_\delta,\tau_0^->t-s \big) \, \nu(dw), \cr
I_2(t,x,\kappa)\!\!\!&:=\!\!\!&    \int_0^T ds \int_0^\infty   \underline{n}_s(dz) \int_{(1-\kappa)(\beta t+x)}^{(1+\kappa)(\beta t+x)} \mathbf P_{z+w}\big( X_{t-s}\in x+\varDelta_\delta,\tau_0^->t-s \big) \, \nu(dw). 
\end{eqnarray*}
where $B(t,x,\kappa):= \big(\eta(\beta t+x), (1-\kappa)(\beta t+x)\big]\cup \big((1+\kappa)(\beta t+x),\infty\big)$. 
It is obvious that \eqref{eqn.3007} and then \eqref{eqn4.26}  hold once we prove the next two limits:
\begin{eqnarray}\label{eqn.3009}
\limsup_{t\to\infty}\sup_{x\geq 0}\frac{I_1(t,x,\kappa)}{\overline\nu(\beta t+x)/(\beta t+x)} =0 
\end{eqnarray}
and 
\begin{eqnarray}\label{eqn.3010}
\limsup_{\kappa\to0+}\limsup_{t\to\infty}\sup_{x\geq 0}\bigg|\frac{I_2(t,x,\kappa)}{\alpha\cdot\overline\nu(\beta t+x)/(\beta t+x)} -\frac{\underline n(\zeta\wedge T)}{ \underline n(\zeta)  } \int_x^{x+\delta}V(z)\, dz \bigg|=0 . 
\end{eqnarray}


\textbf{Case $i=1$.} Firstly, by the spatial homogeneity of $X$, we have 
\begin{eqnarray*}
I_1(t,x,\kappa)\!\!\!&\leq \!\!\!&  \int_0^T ds \int_0^\infty  \underline{n}_s(dz) \int_{B(t,x,\kappa)}  \mathbf{P}\big( X_{t-s}\in x-z-w+\varDelta_\delta  \big) \, \nu(dw).
\end{eqnarray*}
For any $K>0$, we decompose the right-hand side into the following two terms:
\begin{eqnarray*}
I_1^{\leq K}(t,x,\kappa)\!\!\!&:=\!\!\!&  \int_0^T ds \int_0^K \underline{n}_s(dz) \int_{B(t,x,\kappa)}  \mathbf{P}\big( X_{t-s}\in x-z-w+\varDelta_\delta  \big) \, \nu(dw),\cr
I_1^{>K}(t,x,\kappa)\!\!\!&:=\!\!\!&  \int_0^T ds \int_K^\infty  \underline{n}_s(dz) \int_{B(t,x,\kappa)}  \mathbf{P}\big( X_{t-s}\in x-z-w+\varDelta_\delta  \big) \, \nu(dw). 
\end{eqnarray*}
For the second term, notice that $B(t,x,\kappa) \subset\big(  \eta(\beta t+x), \infty\big)$, we have 
\begin{eqnarray*}
I_1^{>K}(t,x,\kappa) 
\!\!\!&\leq\!\!\!&  \sup_{r\in[0,T]} \sup_{z\geq 0}\int_{\eta(\beta t+x)}^{\infty}   \mathbf{P}\big( X_{t-r}\in x-z-w+\varDelta_\delta  \big) \, \nu(dw) \cdot \int_0^T\underline{n}(\epsilon_s>K,\,\zeta>s)\,ds 
\end{eqnarray*}
By Proposition~\ref{Prop.403}, there exist two constants $C,t_0>0$ such that for all $t\geq t_0$, 
\begin{eqnarray*}
I_1^{>K}(t,x,\kappa)  \!\!\!&\leq\!\!\!& C\cdot  \frac{\overline\nu\big(\beta t+x\big)}{\beta t+x} \cdot \int_0^T\underline{n}(\epsilon_s>K,\,\zeta>s)\,ds . 
\end{eqnarray*}  
This induces that
\begin{eqnarray}\label{eqn.4009}
\limsup_{t\to\infty}\sup_{x\geq 0}\frac{ I_1^{>K}(t,x,\kappa)}{\overline\nu(\beta t+x)/(\beta t+x)} \leq C\cdot   \int_0^T\underline{n}(\epsilon_s>K,\,\zeta>s)\,ds ,
\end{eqnarray}
which tends to $0$ as $K\to\infty$ since  $\underline{n}(\zeta)<\infty$; see \eqref{eqn.501}. 
We now consider $I_1^{\leq K}(t,x,\kappa)$. Firstly,  
\begin{eqnarray*}
I_1^{\leq K}(t,x,\kappa)\!\!\!&\leq \!\!\!&  \underline{n}(\zeta)\cdot \sup_{s\in[0,T]} \sup_{z\in[0,K]}  \int_{B(t,x,\kappa)}  \mathbf{P}\big( X_{t-s}\in x-z-w+\varDelta_\delta  \big) \, \nu(dw).   
\end{eqnarray*}
Using Proposition~\ref{Prop.403} along with the fact that $B(t,x,\kappa) \subset \big(  \eta(\beta t+x), \infty\big)$ again yields that uniformly in sufficiently large $t$,
\begin{eqnarray*}
I_1^{\leq K}(t,x,\kappa)\!\!\!&\leq \!\!\!& C\cdot  \frac{\overline\nu\big(\beta t+x\big)}{\beta t+x} \cdot \sup_{s\in[0,T]} \sup_{z\in[0,K]}  \mathbf{P}\bigg( X_{t-s}\in \bigcup_{w\in B(t,x,\kappa) }\big(x-z-w+\varDelta_\delta\big)  \bigg).
\end{eqnarray*}
A simple calculation shows that for any sufficiently large $t$, 
\begin{eqnarray*}
\big|X_{t-s}+\beta(t-s)\big| \geq \frac{\kappa}{2}\cdot \beta (t-s) ,
\end{eqnarray*}
uniformly in $s\in[0,T]$, $w \in B(t,x,\kappa)$, $z\in [0,K]$ and $X_{t-s} \in x-z-w+\varDelta_\delta$. 
Consequently, 
\begin{eqnarray*}
\sup_{x\geq 0}\frac{ I_1^{\leq K}(t,x,\kappa)}{\overline\nu(\beta t+x)/(\beta t+x)}  \!\!\!&\leq \!\!\!& C\cdot  \sup_{s\in[0,T]}  \mathbf{P}\Big(  \big|X_{t-s}+\beta(t-s)\big| \geq \frac{\kappa}{2}\cdot  \beta (t-s) \Big),
\end{eqnarray*}
which goes to $0$ as $t\to\infty$, since $(X_t+\beta t)/t\overset{\rm a.s.}\to 0$ by the law of large numbers. 
Combining this together with \eqref{eqn.4009} yields the limit \eqref{eqn.3009} directly.


\textbf{Case $i=2$.} We now prove the limit  \eqref{eqn.3010}. We first introduce two index sets:
\begin{eqnarray*}
\underline{\mathcal{K}}(t,x,\kappa)\!\!\!&:=\!\!\!&  \big\{j\in \mathbb{N}:   (1-\kappa)\cdot (\beta t+x) < j\kappa\delta + \eta(\beta t+x) \leq (1+\kappa)\cdot (\beta t+x) -\kappa\delta \big\}, \cr\cr
\overline{\mathcal{K}}(t,x,\kappa)\!\!\!&:=\!\!\!&  \big\{j\in \mathbb{N}:   (1-\kappa)\cdot (\beta t+x) -\kappa\delta < j\kappa\delta + \eta(\beta t+x) \leq (1+\kappa)\cdot (\beta t+x)   \big\}. 
\end{eqnarray*}
The interval $\big((1-\kappa)\cdot (\beta t+x),(1+\kappa)\cdot (\beta t+x)\big]$ contains $ \cup_{j\in\underline{\mathcal{K}}(t,x,\kappa)} \big(  \eta(\beta t+x) + j\kappa\delta +\varDelta_{\kappa\delta} \big)$ and is covered by  $\cup_{j \in \overline{\mathcal{K}}(t,x,\kappa)} \big(  \eta(\beta t+x) + j\kappa\delta +\varDelta_{\kappa\delta} \big)$. 
Hence $ I_{21}(t,x,\kappa) \leq I_2(t,x,\kappa) \leq I_{22}(t,x,\kappa)$, where
\begin{eqnarray*}
I_{21}(t,x,\kappa) \!\!\!&:=\!\!\!&  \int_0^T ds \int_0^\infty   \underline{n}_s(dz)  \sum_{j\in\underline{\mathcal{K}}(t,x,\kappa)}\int_{  \eta(\beta t+x) + j\kappa\delta }^{ \eta(\beta t+x) + (j+1)\kappa\delta } \mathbf P_{z+w}\big( X_{t-s}\in x+\varDelta_\delta,\tau_0^->t-s \big) \, \nu(dw),\cr
I_{22}(t,x,\kappa) \!\!\!&:=\!\!\!&  \int_0^T ds \int_0^\infty   \underline{n}_s(dz)  \sum_{j\in\overline{\mathcal{K}}(t,x,\kappa)} \int_{  \eta(\beta t+x) + j\kappa\delta }^{ \eta(\beta t+x) + (j+1)\kappa\delta } \mathbf P_{z+w}\big( X_{t-s}\in x+\varDelta_\delta,\tau_0^->t-s \big) \, \nu(dw) . 
\end{eqnarray*}
Consequently, the limit \eqref{eqn.3010} follows immediately when we establish the following lower and upper bounds hold:
\begin{eqnarray}\label{eqn.3011}
\liminf_{\kappa\to0+}\liminf_{t\to\infty} \inf_{x\geq 0}
\bigg(  \frac{ I_{21}(t,x,\kappa)}{\alpha \cdot \overline\nu(\beta t+x)/(\beta t+x)} -\frac{\underline n(\zeta\wedge T)}{ \underline n(\zeta)  } \int_x^{x+\delta}V(z)\, dz  \bigg)  \geq 0
\end{eqnarray}
and
\begin{eqnarray}\label{eqn.3012}
\limsup_{\kappa\to0+}\limsup_{t\to\infty} \sup_{x\geq 0}
\bigg(  \frac{ I_{22}(t,x,\kappa)}{\alpha \cdot  \overline\nu(\beta t+x)/(\beta t+x)} -\frac{\underline n(\zeta\wedge T)}{ \underline n(\zeta)  } \int_x^{x+\delta}V(z)\, dz  \bigg)  \leq 0.
\end{eqnarray} 
\medskip

\textbf{(i) Lower bound \eqref{eqn.3011}.}  Firstly, for any  $j\geq0$ and $w\in \eta(\beta t+x) + j\kappa\delta +\varDelta_{\kappa \delta}$, we have 
\begin{eqnarray*}
x-z-w+\varDelta_\delta  \supset x-z-\eta(\beta t+x) - j\kappa\delta + \varDelta_{(1-\kappa)\delta}=: A_j(t,x,z),\quad \forall \, x,z\geq 0,
\end{eqnarray*}
which along with the spatial-homogeneity of $X$ yields that 
\begin{eqnarray}\label{eqn.4010}
\mathbf P_{z+w}\big( X_{t-s}\in x+\varDelta_\delta,\tau_0^->t-s\big)
\!\!\!&=\!\!\!& \mathbf P\big( X_{t-s}\in x-z-w+\varDelta_\delta,\tau_{-z-w}^->t-s\big)\cr\cr
\!\!\!&\geq\!\!\!&  \mathbf{P}\Big( X_{t-s}\in  A_j(t,x,z),\,\tau^-_{-z-\eta(\beta t+x)-j\kappa\delta}>t-s  \Big).
\end{eqnarray}
Substituting this into the definition of $I_{21}(t,x,\kappa)$ yields 
\begin{eqnarray*}
\frac{ I_{21}(t,x,\kappa)}{\alpha \cdot \overline\nu(\beta t+x)/(\beta t+x)}
\geq \widehat{I}_{21}(t,x,\kappa)\cdot  \inf_{j\in\underline{\mathcal{K}}(t,x,\kappa)}\frac{ \nu\big(\eta(\beta t+x) + j\kappa\delta +\varDelta_{\kappa \delta}\big) }{\alpha \cdot \overline\nu(\beta t+x)/(\beta t+x)\cdot \kappa\delta} ,
\end{eqnarray*}
with
\begin{eqnarray*} 
\widehat{I}_{21}(t,x,\kappa)\!\!\!&:=\!\!\!&  \int_0^T ds \int_0^\infty   \kappa\delta \sum_{j\in\underline{\mathcal{K}}(t,x,\kappa)} \mathbf{P}\Big( X_{t-s}\in  A_j(t,x,z),\,\tau^-_{-z-\eta(\beta t+x)-j\kappa\delta}>t-s  \Big)\,   \underline{n}_s(dz).
\end{eqnarray*} 
By the definition of $\underline{\mathcal{K}}(t,x,\kappa)$ and \eqref{eqn.30}, we have as $t\to\infty$,
\begin{eqnarray}\label{eqn.4011}
\inf_{x\geq 0}  \inf_{j\in\underline{\mathcal{K}}(t,x,\kappa)}\frac{ \nu\big(\eta(\beta t+x) + j\kappa\delta +\varDelta_{\kappa \delta}\big) }{\alpha \cdot \overline\nu(\beta t+x)/(\beta t+x)\cdot \kappa\delta} 
\!\!\!&\geq\!\!\!&   \inf_{x\geq 0}\inf_{|\lambda|\leq \kappa} \frac{ \nu\big( (1+\lambda)\cdot (\beta t+x) +\varDelta_{\kappa \delta}\big) }{\alpha \cdot \overline\nu(\beta t+x)/(\beta t+x)\cdot \kappa\delta}\cr
\!\!\!&=\!\!\!&  \inf_{z\geq \beta t}\inf_{|\lambda|\leq \kappa} \frac{ \nu\big( (1+\lambda)\cdot z+\varDelta_{\kappa \delta}\big) }{\alpha \cdot \overline\nu(z)/z\cdot \kappa\delta} \to (1+\kappa)^{-\alpha-1},
\end{eqnarray}
which goes to $1$ as $\kappa\to 0+$. 
Consequently, the limit \eqref{eqn.3011} follows if we can prove 
\begin{eqnarray}\label{eqn.4014}
\liminf_{\kappa\to0+}\liminf_{t\to\infty} \inf_{x\geq 0}
\bigg(    \widehat{I}_{21}(t,x,\kappa)  -\frac{\underline n(\zeta\wedge T)}{ \underline n(\zeta)  } \int_x^{x+\delta}V(z)\, dz  \bigg)  \geq 0.
\end{eqnarray}
For any large constant $M>0$, we have 
\begin{eqnarray}\label{eqn.4013}
\widehat{I}_{21} (t,x,\kappa)  \geq \int_0^T ds \int_0^M\kappa\delta  \sum_{j\in\underline{\mathcal{K}}(t,x,\kappa)} \mathbf{P}\Big( X_{t-s}\in  A_j(t,x,z),\,\tau^-_{-z-\eta(\beta t+x)-j\kappa\delta}>t-s  \Big)\,   \underline{n}_s(dz) .
                                                                                                                                                                                                                                                                   
\end{eqnarray}
The definition of $\tau_y^-$ tells that 
\begin{eqnarray*}
\lefteqn{\kappa\delta  \inf_{z\in [0,M]}  \sum_{j\in\underline{\mathcal{K}}(t,x,\kappa)} \mathbf{P}\Big( X_{t-s}\in  A_j(t,x,z),\,\tau^-_{-z-\eta(\beta t+x)-j\kappa\delta}>t-s  \Big)}\cr
\!\!\!&=\!\!\!&  \kappa\delta  \inf_{z\in [0,M]}  \sum_{j\in\underline{\mathcal{K}}(t,x,\kappa)} \mathbf{P}\Big( X_{t-s}\in  A_j(t,x,z),\,\underline{X}_{t-s}\geq -z-\eta(\beta t+x)-j\kappa\delta  \Big) \cr
\!\!\!&\geq\!\!\!&  \mathbf{E}\bigg[ \kappa\delta  \inf_{z\in [0,M]} \sum_{j\in\underline{\mathcal{K}}(t,x,\kappa)} \mathbf{1}_{\{ X_{t-s}\in  A_j(t,x,z),\,\underline{X}_{t-s}\geq -z-\eta(\beta t+x)-j\kappa\delta\}}  \bigg].
\end{eqnarray*}
We claim the following, to be proved later: for any $u,v\in\mathbb R$ and $s\in[0,T]$,
\begin{eqnarray}\label{eqn.4012}
\lefteqn{ \inf_{z\in [0,M]} \sum_{j\in\underline{\mathcal{K}}(t,x,\kappa)} \mathbf{1}_{\{ u\in  A_j(t,x,z),\,v\geq -z-\eta(\beta t+x)-j\kappa\delta\}} }\cr
\!\!\!&\geq\!\!\!& \Big( \frac{1}{\kappa\delta} \int_x^{x+\delta} \mathbf{1}_{\{ u-v\leq w \}}\,dw -2\Big) \cdot \mathbf{1}_{\{ |u+\beta (t-s)|\leq \kappa(\beta t+x)/2\}}.
\end{eqnarray}
Combining these estimates together yields that 
\begin{eqnarray*}
\lefteqn{ \kappa\delta \inf_{z\in [0,M]}  \sum_{j\in\underline{\mathcal{K}}(t,x,\kappa)} \mathbf{P}\Big( X_{t-s}\in  A_j(t,x,z),\,\tau^-_{-z-\eta(\beta t+x)-j\kappa\delta}>t-s  \Big)}\cr
\!\!\!&\geq \!\!\!&  \mathbf{E}\bigg[\Big(  \int_x^{x+\delta} \mathbf{1}_{\{ X_{t-s}-\underline{X}_{t-s}\leq z \}}\,dz -2\kappa\delta  \Big) \cdot \mathbf{1}_{\{ |X_{t-s}+\beta (t-s)|\leq \kappa(\beta t+x)/2  \}}\bigg]\cr
\!\!\!&\geq \!\!\!& \int_x^{x+\delta} \mathbf{P} \Big(  X_{t-s}-\underline{X}_{t-s}\leq z ,\,   \big|X_{t-s}+\beta (t-s)\big|\leq \frac{\kappa(\beta t+x)}{2}   \Big) \,dz - 2\kappa\delta \cr
\!\!\!&\geq \!\!\!& \int_x^{x+\delta} \mathbf{P} \big(  X_{t-s}-\underline{X}_{t-s}\leq z   \big) \,dz - \delta \cdot \mathbf{P} \Big(    \big|X_{t-s}+\beta (t-s)\big|> \frac{\kappa(\beta t+x)}{2}   \Big) - 2\kappa\delta\cr
\!\!\!&\geq\!\!\!& \int_x^{x+\delta} \mathbf{P} \big(   \overline{X}_{t-s}\leq z  \big) \,dz - \delta \cdot \mathbf{P} \Big(    \big|X_{t-s}+\beta (t-s)\big|> \frac{\kappa\beta (t-s)}{2}   \Big) - 2\kappa\delta. 
\end{eqnarray*}
Here the last inequality follows from $\beta t+x\geq \beta(t-s)$ and $X_{t-s}-\underline{X}_{t-s}\overset{\rm d}= \overline{X}_{t-s}$ by the duality lemma. 
Substituting it into the right-hand side of \eqref{eqn.4013} yields
\begin{eqnarray*}
\lefteqn{\widehat{I}_{21} (t,x,\kappa)   -\frac{\underline n(\zeta\wedge T)}{ \underline n(\zeta)  } \int_x^{x+\delta}V(z)\, dz}\cr\cr
\!\!\!&\geq\!\!\!& \int_0^T \underline{n}\big(\epsilon_s\leq M,\,\zeta >s\big)\, ds \int_x^{x+\delta} \mathbf{P} \big(   \overline{X}_{t-s} \leq z \big) \,dz  -  2\kappa\delta \int_0^T \underline{n}\big( \epsilon_s\leq M,\,\zeta >s\big)\, ds \cr\cr
\!\!\!&\!\!\!& - \delta  \int_0^T \underline{n}\big( \epsilon_s\leq M,\,\zeta >s\big)\cdot \mathbf{P} \Big(  \big|X_{t-s}+\beta (t-s)\big|> \frac{\kappa\beta (t-s)}{2} \Big)\, ds -\frac{\underline n(\zeta\wedge T)}{ \underline n(\zeta)  } \int_x^{x+\delta}V(z)\, dz   \cr\cr
\!\!\!&\geq\!\!\!& \int_0^T \underline{n}\big(\epsilon_s\leq M,\,\zeta >s\big)\, ds \cdot \int_x^{x+\delta}   \inf_{r\in[0,T]}\mathbf{P} \big(   \overline{X}_{t-r}\leq z  \big) \,dz  -\frac{\underline n(\zeta\wedge T)}{ \underline n(\zeta)  } \int_x^{x+\delta}V(z)\, dz  \cr\cr
\!\!\!&\!\!\!& - \delta  \int_0^T \underline{n}\big(  \zeta >s\big)\, ds\cdot \sup_{r\in[0,T]} \mathbf{P} \Big(    \big|X_{t-r}+\beta (t-r)\big|>  \frac{\kappa\beta (t-r)}{2}    \Big) -  2\kappa\delta \int_0^T \underline{n}\big( \zeta >s\big)\, ds . 
\end{eqnarray*}
Firstly, the last term on the right-hand side of the last inequality goes to $0$ as $\kappa \to 0$. 
Secondly, the law of large numbers  yields that the third term  also goes to $0$ as $t\to\infty$. Consequently,
\begin{eqnarray*}
\lefteqn{\liminf_{\kappa\to0+}\liminf_{t\to\infty} \inf_{x\geq 0}
	\bigg(    \widehat{I}_{21} (t,x,\kappa)  -\frac{\underline n(\zeta\wedge T)}{ \underline n(\zeta)  } \int_x^{x+\delta}V(z)\, dz  \bigg)}\cr
\!\!\!&\geq\!\!\!&  \liminf_{t\to\infty} \inf_{x\geq 0} \bigg( \int_0^T \underline{n}\big(\epsilon_s\leq M,\,\zeta >s\big)\, ds \int_x^{x+\delta}   \inf_{r\in[0,T]}\mathbf{P} \big(    \overline{X}_{t-r} \leq z  \big) \,dz-\frac{\underline n(\zeta\wedge T)}{ \underline n(\zeta)  } \int_x^{x+\delta}V(z)\, dz \bigg).
\end{eqnarray*}
Using the triangle inequality, then $\underline{n}(\zeta)<\infty$ (see \eqref{eqn.501}) and  $V(z)\leq V(\infty)<\infty$ to the right-hand side yields
\begin{eqnarray*} \lefteqn{\liminf_{\kappa\to0+}\liminf_{t\to\infty} \inf_{x\geq 0}
	\bigg(  \widehat{I}_{21} (t,x,\kappa)  -\frac{\underline n(\zeta\wedge T)}{ \underline n(\zeta)  } \int_x^{x+\delta}V(z)\, dz  \bigg)}\cr
\!\!\!&\geq\!\!\!& \inf_{x\geq 0} \bigg(\int_0^T \underline{n}\big(\epsilon_s\leq M,\,\zeta >s\big)\, ds - \underline n(\zeta\wedge T) \bigg) \cdot   \int_x^{x+\delta} \frac{V(z)}{ \underline n(\zeta)  }\, dz \cr
\!\!\!&\!\!\!& +  \int_0^T \underline{n}\big(\epsilon_s\leq M,\,\zeta >s\big)\, ds  \cdot \liminf_{t\to\infty} \inf_{x\geq 0}\int_x^{x+\delta}   \Big(\inf_{r\in[0,T]}\mathbf{P} \big(    \overline{X}_{t-r} \leq z  \big) - \frac{V(z)}{\underline n(\zeta)}\Big)\,dz \cr
\!\!\!&\geq \!\!\!& - \sup_{x\geq 0}  \int_x^{x+\delta} \frac{V(z)}{ \underline n(\zeta)  }\, dz  \cdot  \bigg|\int_0^T \underline{n}\big(\epsilon_s\leq M,\,\zeta >s\big)\, ds - \underline n(\zeta\wedge T)\bigg| \cr
\!\!\!&\!\!\!& - \int_0^T \underline{n}\big(\epsilon_s\leq M,\,\zeta >s\big)\, ds  \cdot \limsup_{t\to\infty} \sup_{x\geq 0}\int_x^{x+\delta}   \Big|\inf_{r\in[0,T]}\mathbf{P} \big(    \overline{X}_{t-r} \leq z  \big) - \frac{V(z)}{\underline n(\zeta)}\Big|\,dz\cr
\!\!\!&\geq\!\!\!& - C \cdot  \bigg|\int_0^T \underline{n}\big(\epsilon_s\leq M,\,\zeta >s\big)\, ds - \underline n(\zeta\wedge T)\ \bigg|  - C\cdot \limsup_{t\to\infty} \sup_{r\in[0,T]} \sup_{z\geq 0}   \bigg| \mathbf{P} \big(    \overline{X}_{t-r} \leq z  \big) -    \frac{V(z)}{\underline n(\zeta)}\bigg| ,
\end{eqnarray*} 
for some constant $C>0$ independent of $M$. 
Obviously, the first term on the right-hand side of the last inequality goes to $0$ as $M\to\infty$. 
Furthermore, since $ \overline{X}_{t-r} \overset{\rm a.s.}\to \overline{X}_\infty<\infty$ as $t\to\infty$, and  the Wiener--Hopf factorization tells that  $\mathbf{P}\big(\overline X_\infty\leq z \big)=V(z)/\underline n(\zeta)$, we have as $t\to\infty$ 
\begin{eqnarray}\label{eqn.4015}
\sup_{z\geq 0}\bigg| \mathbf{P} \big(    \overline{X}_{t} \leq z  \big) -    \frac{V(z)}{\underline n(\zeta)}\bigg| \to 0,
\end{eqnarray}
and then the second term is identically zero. 
Consequently, to complete the proof of \eqref{eqn.4014} (and therefore of \eqref{eqn.3011}), it remains only to establish the inequality \eqref{eqn.4012}.
 Fix $M,T>0$. For $x\geq0$, $s\in[0,T]$, $u,v\in\mathbb R$, and $z\in[0,M]$, let $j\in\mathbb Z$  satisfy
 \begin{eqnarray}\label{eqn5.44}
 \big|u+\beta(t-s)\big| \leq \frac{\kappa}{2}(\beta t+x)
 \quad\mbox{and}\quad
 \max\{x,u-v\} 	< u+z+\eta(\beta t+x)+j\kappa\delta \leq x+(1-\kappa)\delta.
 \end{eqnarray}
 Then, for all sufficiently large $t$, uniformly in $x\geq0$ and $s\in[0,T]$,
 \begin{eqnarray*}
  \eta(\beta t+x)+j\kappa\delta > x-u-z	\geq \Big(1-\frac{\kappa}{2}\Big)(\beta t+x)-\beta T-M>	(1-\kappa)(\beta t+x),
 \end{eqnarray*}
 and
 \begin{eqnarray*}
  \eta(\beta t+x)+j\kappa\delta \leq 	x+(1-\kappa)\delta-u-z
  \leq \Big(1+\frac{\kappa}{2}\Big)(\beta t+x)+(1-\kappa)\delta
  <(1+\kappa)(\beta t+x)-\kappa\delta.
 \end{eqnarray*}
 Since $\eta<1-\kappa$, the preceding estimates, together with \eqref{eqn5.44}, give
 \begin{eqnarray*}
 j\in\underline{\mathcal K}(t,x,\kappa),\quad u\in A_j(t,x,z),
 \quad 
 v>-z-\eta(\beta t+x)-j\kappa\delta.
 \end{eqnarray*}
 Consequently,
 \begin{eqnarray*}
 \lefteqn{ \sum_{j\in\underline{\mathcal K}(t,x,\kappa)} \mathbf 1_{\{u\in A_j(t,x,z),\, v\geq-z-\eta(\beta t+x)-j\kappa\delta \}}}\cr
 \!\!\!&\geq\!\!\!&
 \#\big\{j\in\mathbb Z: \max\{x,u-v\} < u+z+\eta(\beta t+x)+j\kappa\delta \leq x+(1-\kappa)\delta \big\}\cdot \mathbf 1_{\{|u+\beta(t-s)| \leq \kappa(\beta t+x)/2\}}.
 \end{eqnarray*}
 Since consecutive points $u+z+\eta(\beta t+x)+j\kappa\delta$ are spaced by $\kappa\delta$, the cardinality factor on the right-hand side of the preceding estimate is bounded below by
 \begin{eqnarray*}
 \frac{1}{	\kappa\delta}\cdot	\big[x+(1-\kappa)\delta-\max\{x,u-v\}\big]^+-1
 \!\!\!&\geq\!\!\!& \frac{1}{	\kappa\delta}\cdot  \big[x+\delta-\max\{x,u-v\}\big]^+ -2\cr\cr
 \!\!\!&=\!\!\!& \frac{1}{\kappa\delta} \int_x^{x+\delta} \mathbf 1_{\{u-v\leq w\}}\,dw -2.
 \end{eqnarray*}
 Combining these  estimates and taking the infimum over $z\in[0,M]$ proves \eqref{eqn.4012}.  
 \medskip

 \textbf{(ii) Upper bound \eqref{eqn.3012}.} 
 Similarly as in \eqref{eqn.4010},   for any $j\geq 0$ and $w\in \eta(\beta t+x) + j\kappa\delta +\varDelta_{\kappa \delta}$ we have
 \begin{eqnarray*}
 \mathbf{P}_{z+w}\big( X_{t-s}\in x+\varDelta_\delta,\tau_0^->t-s \big)   \leq \mathbf{P}\Big( X_{t-s}\in U_j(x,z,\kappa),\, \tau_{-z-\eta(\beta t+x)-(j+1)\kappa\delta}^->t-s \Big),
 \end{eqnarray*}
 with $U_j(x,z,\kappa):= x-z - \eta(\beta t+x)-(j+1)\kappa\delta +\varDelta_{(1+\kappa)\delta} $. 
                                                                                              
        
                                                                                                                     
        
          
        
                                                                          
                                                                                    
        
                                                                                                                        
                                                                                                                           
        
 This yields that 
 \begin{eqnarray*}
  \frac{ I_{22}(t,x,\kappa)}{\alpha \cdot \overline\nu(\beta t+x)/(\beta t+x)}
  \leq  \widehat{I}_{22}(t,x,\kappa)\cdot \sup_{j\in\overline{\mathcal{K}}(t,x,\kappa)}\frac{ \nu\big(\eta(\beta t+x) + j\kappa\delta +\varDelta_{\kappa \delta}\big) }{\alpha \cdot \overline\nu(\beta t+x)/(\beta t+x)\cdot \kappa\delta}
 \end{eqnarray*}
 where 
 \begin{eqnarray*}
  \widehat{I}_{22}(t,x,\kappa)\!\!\!&:=\!\!\!& \int_0^T ds \int_0^\infty    \kappa\delta\sum_{j\in\overline{\mathcal{K}}(t,x,\kappa)}  \mathbf{P}\Big( X_{t-s}\in U_j(x,z,\kappa),\, \tau_{-z-\eta(\beta t+x)-(j+1)\kappa\delta}^->t-s \Big) \,   \underline{n}_s(dz).
 \end{eqnarray*} 
 An argument analogous to \eqref{eqn.4011} gives 
 \begin{eqnarray*}
  \limsup_{\kappa\to0+}  \limsup_{t\to\infty} \sup_{x\geq 0} \sup_{j\in\overline{\mathcal{K}}(t,x,\kappa)}\frac{ \nu\big(\eta(\beta t+x) + j\kappa\delta +\varDelta_{\kappa \delta}\big) }{\alpha \cdot \overline\nu(\beta t+x)/(\beta t+x)\cdot \kappa\delta} \leq \limsup_{\kappa\to0+}  (1-\kappa)^{-\alpha-1} =1. 
 \end{eqnarray*}
 Consequently, we just need to prove that 
 \begin{eqnarray}\label{eqn.4017}
  \limsup_{\kappa\to0+}\limsup_{t\to\infty} \sup_{x\geq 0} \bigg(    \widehat{I}_{22}(t,x,\kappa) -\frac{\underline n(\zeta\wedge T)}{ \underline n(\zeta)  } \int_x^{x+\delta}V(z)\, dz  \bigg)  \leq 0.
 \end{eqnarray}
 To analyze $\widehat{I}_{22}(t,x,\kappa)$, we first claim, and will prove later, that for any $u,v\in\mathbb R$ and $z\geq 0$,
 \begin{eqnarray}\label{eqn.4016}
 \sum_{j\in\overline{\mathcal{K}}(t,x,\kappa)} \mathbf 1_{\{u\in U_j(x,z,\kappa),\,	v\geq -z-\eta(\beta t+x)-(j+1)\kappa\delta \}} \leq  \frac{1}{\kappa\delta} \int_x^{x+\delta} \mathbf 1_{\{u-v\leq w\}}\, dw+2.
 \end{eqnarray}
 This  induces that 
 \begin{eqnarray*}
 \lefteqn{  \sum_{j\in\overline{\mathcal{K}}(t,x,\kappa)}  \mathbf{P}\Big( X_{t-s}\in U_j(x,z,\kappa),\, \tau_{-z-\eta(\beta t+x)-(j+1)\kappa\delta}^->t-s \Big) }\cr
 \!\!\!&=\!\!\!&  \sum_{j\in\overline{\mathcal{K}}(t,x,\kappa)}  \mathbf{P}\Big( X_{t-s}\in U_j(x,z,\kappa),\, \underline{X}_{t-s}\geq -z-\eta(\beta t+x)-(j+1)\kappa\delta \Big)\cr
 \!\!\!&=\!\!\!&  \mathbf{E}\bigg[ \sum_{j\in\overline{\mathcal{K}}(t,x,\kappa)} \mathbf{1}_{\{ X_{t-s}\in U_j(x,z,\kappa),\, \underline{X}_{t-s}\geq -z-\eta(\beta t+x)-(j+1)\kappa\delta\}} \bigg]\cr
 \!\!\!&\leq\!\!\!& \frac{1}{\kappa\delta}\int_x^{x+\delta} \mathbf{P}\big(  X_{t-s}-\underline{X}_{t-s} \leq  w\big) dw+2 
 = \frac{1}{\kappa\delta}\int_x^{x+\delta} \mathbf{P}\big(  \overline{X}_{t-s}  \leq  w\big) dw+2 .
 \end{eqnarray*}
 Here the last equality follows from the identity $X_{t-s}-\underline{X}_{t-s}\overset{\rm d}= \overline{X}_{t-s} $ by the duality lemma. 
 Substituting this estimate into the definition of $ \widehat{I}_{22}(t,x,\kappa)$ yields that 
 \begin{eqnarray*}
 \lefteqn{ \sup_{x\geq 0}
	\bigg(    \widehat{I}_{22}(t,x,\kappa) -\frac{\underline n(\zeta\wedge T)}{ \underline n(\zeta)  } \int_x^{x+\delta}V(z)\, dz  \bigg)  }\cr 
 \!\!\!&\leq\!\!\!&  2 \underline{n}(\zeta) \kappa\delta    +  \underline{n}(\zeta \wedge T) \cdot \sup_{x\geq 0} \bigg(  \int_x^{x+\delta} \sup_{s\in[0,T]}\mathbf{P}\big(  \overline{X}_{t-s} \leq  z\big) \,dz -\int_x^{x+\delta} \frac{V(z) }{ \underline n(\zeta)  } \, dz\bigg)\cr
 \!\!\!&\leq\!\!\!& 2 \underline{n}(\zeta) \kappa\delta    +  \underline{n}(\zeta \wedge T) \cdot \delta \cdot \sup_{z\geq 0} \sup_{s\in[0,T]}\bigg| \mathbf{P}\big(  \overline{X}_{t-s} \leq  z\big)- \frac{V(z) }{ \underline n(\zeta)  }\bigg|  ,
 \end{eqnarray*}
 which goes to $0$ as $t\to\infty$ and then $\kappa\to 0+$ by \eqref{eqn.4015}.
 Consequently, it remains to identify the inequality  \eqref{eqn.4016} to complete the proof of \eqref{eqn.4017}  and  therefore of \eqref{eqn.3012}. 
 Fix $u,v\in\mathbb R$ and $z\geq0$.
 Let $j\in\overline{\mathcal K}(t,x,\kappa)$ be any index satisfying $u\in U_j(x,z,\kappa) $ and $v\geq-z-\eta(\beta t+x)-(j+1)\kappa\delta$.
 By the definition of $U_j(x,z,\kappa)$ and the preceding inequality,
 \begin{eqnarray*}
 x-\kappa\delta<u+z+\eta(\beta t+x)+j\kappa\delta\leq x+\delta
 \quad \mbox{and}\quad 
 u-v-\kappa\delta\leq u+z+\eta(\beta t+x)+j\kappa\delta.
 \end{eqnarray*}
 Consequently,
 \begin{eqnarray*}
 \lefteqn{ \sum_{j\in\overline{\mathcal K}(t,x,\kappa)} \mathbf 1_{\{u\in U_j(x,z,\kappa),\,v\geq -z-\eta(\beta t+x)-(j+1)\kappa\delta\}}}\cr\cr
 \!\!\!&\leq\!\!\!& \#\big\{j\in\mathbb Z:\max\{x,u-v\}-\kappa\delta< u+z+\eta(\beta t+x)+j\kappa\delta \leq x+\delta \big\}.
\end{eqnarray*}
Since consecutive points $u+z+\eta(\beta t+x)+j\kappa\delta$ are spaced by $\kappa\delta$, the cardinality on the right-hand side is bounded above by
\begin{eqnarray*}
 \frac{1}{ \kappa\delta }\cdot 	\big[x+(1+\kappa)\delta-\max\{x,u-v\}\big]^+ +1 \!\!\!&\leq\!\!\!& \frac{1}{	\kappa\delta}\cdot 	\big[x+\delta-\max\{x,u-v\}\big]^+ +2 \cr
 \!\!\!&=\!\!\!& \frac{1}{\kappa\delta} \int_x^{x+\delta} \mathbf 1_{\{u-v\leq w\}}\,dw +2.
 \end{eqnarray*}
 Combining the preceding estimates proves \eqref{eqn.4016}. 
 This completes the whole proof.
 \hfill$\Box$\medskip

Firstly, a simple calculation shows that as $t\to\infty$,

 where 
 \begin{eqnarray*}

 \end{eqnarray*}

 \begin{eqnarray}\label{eqn.607}

 \end{eqnarray}
 When $(x-u+\delta)^+ -(x-u)^+\leq \lambda$, we have 
 \begin{eqnarray*}
  \sup_{s\in[0,R]} \sup_{x\geq 0}  \sup_{u\in[0,y]} \alpha   \int_{(x-u)^+}^{(x-u+\delta)^+} V(z)\, dz  
 \leq \alpha    V(\infty)\cdot\lambda,
 \end{eqnarray*}
 Moreover, since $\delta \mapsto \underline{n}_t(x+\varDelta_\delta)$ is non-decreasing, we use Theorem~\ref{MainThm01}  and obtain that for sufficiently large $t$, 
 \begin{eqnarray*}
 \lefteqn{\sup_{s\in[0,R]}  \sup_{x\geq 0}   \sup_{u\in[0,y]} \frac{\beta (t-s) +(x-u)^+}{\overline\nu\big(\beta (t-s)+(x-u)^+\big)}\cdot \underline{n}_{t-s}\big(  \big((x-u)^+,(x-u+\delta)^+] \big)}\cr

 \end{eqnarray*}
 Combining this together with \eqref{eqn.607} directly yields \eqref{eqn.606}. This completes the proof.

By using the change of variables $r=t-s$ and then Fubini's theorem,

 I_3^{y,\delta}(t,R,x)\!\!\!&=\!\!\!& \frac{\beta t +x}{\overline\nu(\beta t+x)} \int_0^R dr\int_0^y \underline{n}_{r}\big( \big((x-y+z)^+, (x-y+z+\delta)^+\big] \big) \, \overline{n}_{t-r}(dz) \cr

 which along with the fact that $B_y^\delta(x,z)\neq \emptyset$ only if $z\in x-y +\varDelta_{y+\delta}$ induces that

 \end{eqnarray*}

 \begin{eqnarray*}
 \sup_{x\geq M} I_3^{y,\delta}(t,R,x)  
 \!\!\!&\leq\!\!\!&  C\cdot  \frac{\overline\nu(\beta t)}{\beta t} \cdot  \sup_{x\geq M} \frac{\beta t +x}{\overline\nu(\beta t+x)}\cdot  \frac{\overline\nu(x)}{x}\cr

 \end{eqnarray*}
  By $\overline{\nu} \in \mathrm{RV}^\infty_{-\alpha}$ and Potter's theorem for regularly varying functions; see e.g. Theorem 1.5.6 in \cite[p.25]{BinghamGoldieTeugels1987}, the last expression can be bounded by
  \begin{eqnarray*}
  C\cdot  \frac{\overline\nu(\beta t)}{\beta t} \cdot    \frac{\beta t +t}{\overline\nu(\beta t+t)}\cdot  \frac{\overline\nu(M)}{M} + C\cdot  \frac{\overline\nu(\beta t)}{\beta t}, 
  \end{eqnarray*}
  which goes to $0$ as $t\to\infty$ and then $M\to \infty$. Combining these estimates together, we obtain 
  \begin{eqnarray*}
  \limsup_{M\to\infty} \limsup_{t\to\infty}  \sup_{x\geq M} I_3^{y,\delta}(t,R,x)   =0. 
  \end{eqnarray*}
 Combining this, \eqref{eqn.4022} together with \eqref{eqn.4023} yields   $\sup_{x\geq 0}\big|  I_3^{y,\delta}(t,R,x)- \widetilde{I}_3^{y,\delta}(R,x) \big| \to0$ as $t\to\infty$. 
 This completes the proof.
 \hfill$\Box$\medskip

Recall the notation $I_4^{y,\delta}(t,R,x)$ defined in the proof of Corollary~\ref{Main.Corollary02}.   For $t>2R$, we may decompose $I_4^{y,\delta}(t,R,x)$ into the following two parts:
 \begin{eqnarray}
 I_{41}^{y,\delta}(t,R,x)\!\!\!&:=\!\!\!&\frac{\beta t +x}{\overline\nu(\beta t+x)}\int_R^{t/2} ds\int_0^y \underline{n}_{t-s}\big( \big((x-y+z)^+, (x-y+z+\delta)^+\big] \big) \, \overline{n}_s(dz) , \label{eqn.4024}\\
 I_{42}^{y,\delta}(t,R,x)\!\!\!&:=\!\!\!&\frac{\beta t +x}{\overline\nu(\beta t+x)}\int_{t/2}^{t-R} ds\int_0^y \underline{n}_{t-s}\big( \big((x-y+z)^+, (x-y+z+\delta)^+\big] \big) \, \overline{n}_s(dz)  .  \label{eqn.4025}
 \end{eqnarray}
 By Theorem~\ref{MainThm01}, there exists a constant $C>0$ such that for all sufficiently large $t$ and  all $x\geq0$,
 \begin{eqnarray*}
  \sup_{s\in[R,t/2]} \sup_{z\in [0,y]}\underline{n}_{t-s}\big( \big((x-y+z)^+, (x-y+z+\delta)^+\big] \big) 
  \leq C \cdot\frac{\overline\nu(\beta t+x)}{\beta t+x} . 
 \end{eqnarray*}
 Substituting this into the right-hand side of \eqref{eqn.4024}, then using \eqref{eqn.447},  
 \begin{eqnarray}\label{eqn.4026}
  \limsup_{R\to\infty} \limsup_{t\to\infty} \sup_{x\geq 0} I_{41}^{y,\delta}(t,R,x) 
  \leq C \cdot   \limsup_{R\to\infty} \int_R^\infty \overline{n}\big(\epsilon_s\in(0,y],\,\zeta >s\big)\,ds   =0. 
 \end{eqnarray}
 Applying the change of variables $r=t-s$ to  $I_{42}^{y,\delta}(t,R,x)$ gives 
 \begin{eqnarray*}
 I_{42}^{y,\delta}(t,R,x)
 \!\!\!&=\!\!\!& \frac{\beta t +x}{\overline\nu(\beta t+x)}\int_{R}^{t/2} dr\int_0^y \underline{n}_r\big( \big((x-y+z)^+, (x-y+z+\delta)^+\big] \big) \, \overline{n}_{t-r}(dz) \cr
 \!\!\!&\leq\!\!\!& \frac{\beta t +x}{\overline\nu(\beta t+x)}\int_{R}^{t/2} \sup_{z\in[0,y]} \underline{n}_r\big( \big((x-y+z)^+, (x-y+z+\delta)^+\big] \big) \cdot \overline{n}_{t-r}\big([0,y]\big)\, dr . 
 \end{eqnarray*}
 Using Theorem~\ref{MainThm01} again yields that   for all sufficiently large $r>R$ and  all $x\geq0$,
 \begin{eqnarray*}
  \sup_{z\in[0,y]} \underline{n}_r\big( \big((x-y+z)^+, (x-y+z+\delta)^+\big] \big)
  \leq C\cdot \frac{\overline\nu(\beta r+x)}{\beta r+x}. 
 \end{eqnarray*}
 In addition,  by Lemma~\ref{Lem5.4} we also have for all sufficiently large $t$ and $r\in [R,t/2]$,
 \begin{eqnarray*}
 \overline{n}_{t-r}\big([0,y]\big) \leq C\cdot \frac{\overline\nu(\beta(t-r))}{t-r}. 
 \end{eqnarray*}
 Combining these estimates together yields that for all sufficiently large $R$, $t>2R$ and  all $x\geq0$,
 \begin{eqnarray*}
 \sup_{x\geq 0} I_{42}^{y,\delta}(t,R,x)
  \!\!\!&\leq\!\!\!& C\cdot  \sup_{x\geq 0} \frac{\beta t +x}{\overline\nu(\beta t+x)}\int_{R}^{t/2}   \frac{\overline\nu(\beta r+x)}{\beta r+x} \cdot  \frac{\overline\nu(\beta(t-r))}{t-r}  \, dr \cr
  \!\!\!&\leq\!\!\!& C\cdot  \sup_{x\in[0,t]} \frac{\beta t +x}{\overline\nu(\beta t+x)}\int_{R}^{t/2}   \frac{\overline\nu(\beta r+x)}{\beta r+x} \cdot  \frac{\overline\nu(\beta(t-r))}{t-r}  \, dr\cr
  \!\!\!&\!\!\!& +C\cdot  \sup_{x\geq t} \frac{\beta t +x}{\overline\nu(\beta t+x)}\int_{R}^{t/2}   \frac{\overline\nu(\beta r+x)}{\beta r+x} \cdot  \frac{\overline\nu(\beta(t-r))}{t-r}  \, dr\cr
  \!\!\!&\leq\!\!\!& C\cdot \Big( \int_R^\infty \frac{\overline{\nu}(\beta r)}{\beta r}\,dr + \int_R^{t/2} \frac{\overline\nu(\beta(t-r))}{t-r}  \, dr \Big) \cr
  \!\!\!&\leq \!\!\!& C\cdot \Big( \int_R^\infty \frac{\overline{\nu}(\beta r)}{\beta r}\,dr + \int_{t/2}^\infty \frac{\overline\nu(\beta r)}{r}  \, dr \Big),
 \end{eqnarray*}
 which goes to $0$ as $t\to\infty$ and then $R\to\infty$. 
 Combining this together with \eqref{eqn.4026} yields the desired uniform limit. 
 This completes the proof.
