<!-- proposition-1: body_md -->
\label{pro3.1}
	Assume that $U\subseteq\mathbb{R}^m$ is a nonempty set, and $F: U\to\mathbb{R}^m$ is an operator. Then,
	\begin{enumerate}
	\item[$\mathbf{(i)}$] $F$ is $\alpha$-symmetric smooth with $\alpha\in(0,1)$ and constants $L_0, L_1\geq0$, if and only if,	for all $y,y'\in U$,
	\begin{equation*}
		\|F(y)-F(y')\|\leq\|y-y'\|\left(K_0+K_1\|F(y')\|^\alpha
		+K_2\|y-y'\|^{\frac{\alpha}{1 - \alpha}}\right),
	\end{equation*}
	where ${K}_{0} = {L}_{0}\left( 2^{\frac{\alpha^2}{1 - \alpha}} + 1\right)$, ${K}_{1}=L_1{2}^{\frac{\alpha^2}{1 - \alpha}}$, and ${K}_2= {L}_1^{\frac{1}{1 - \alpha}}{2}^{\frac{\alpha^2}{1 - \alpha}}3^{\alpha}(1 - \alpha)^{\frac{\alpha}{1 - \alpha}}$.
	
	\item[$\mathbf{(ii)}$] $F$ is $\alpha$-symmetric smooth with $\alpha=1$ and constants $L_0, L_1\geq0$, if and only if,  for all $y,y'\in U$,
	\begin{equation*}
		\|F(y)-F(y')\|\leq\|y-y'\|\bigl(L_0+L_1\|F(y')\|\bigr)\exp\bigl(L_1\|y-y'\|\bigr).
	\end{equation*}
\end{enumerate}

<!-- lemma-6: body_md -->
\label{lem31}
	Let $\{\lambda_k\}$ be the stepsize sequence generated by Algorithm~\ref{alg3}. Then the following statements hold.
	\begin{enumerate}
		\item[$\mathbf{(i)}$] The sequence is uniformly bounded above:
		\begin{equation}\label{eq:lambda-upper}
			0<\lambda_k\le \bar\lambda
			:=\lambda_0\exp\!\left(\sum_{j=0}^{\infty}\eta_j\right),
			\qquad k\ge0.
		\end{equation}
		
		\item[$\mathbf{(ii)}$] The stepsize sequence converges:
		\begin{equation}\label{eq:lambda-limit}
			\lambda_k\to\lambda_\infty
			\qquad\text{for some }\lambda_\infty\ge0.
		\end{equation}
		
		\item[$\mathbf{(iii)}$] The positive and negative variations are summable:
		\begin{equation}\label{eq:lambda-variation}
			\sum_{k=0}^{\infty}(\lambda_{k+1}-\lambda_k)_+<\infty,
			\qquad
			\sum_{k=0}^{\infty}(\lambda_{k+1}-\lambda_k)_-<\infty,
		\end{equation}
		where
		\begin{equation*}
			(a)_+:=\max\{a,0\},
			\qquad
			(a)_-:=\max\{-a,0\}.
		\end{equation*}
		Consequently,
		\begin{equation}\label{eq:lambda-total-variation}
			\sum_{k=0}^{\infty}|\lambda_{k+1}-\lambda_k|<\infty.
		\end{equation}
	\end{enumerate}

<!-- proof-1: body_md -->
By the first term in the stepsize rule \eqref{eq:pf-aeg-stepsize}, $\lambda_{k+1}\le(1+\eta_k)\lambda_k$. 	Iterating this inequality and using $\log(1+x)\le x$ for $x\ge0$, we obtain
	\begin{align*}
		\lambda_{k+1}
		&\le \lambda_0\prod_{j=0}^{k}(1+\eta_j)
		\le \lambda_0\exp\!\left(\sum_{j=0}^{k}\eta_j\right)
		\le \lambda_0\exp\!\left(\sum_{j=0}^{\infty}\eta_j\right)
		=\bar\lambda,
	\end{align*}
	which proves $\mathbf{(i)}$.
	
	We next prove $\mathbf{(iii)}$. The stepsize rule \eqref{eq:pf-aeg-stepsize} gives
	$\lambda_{k+1}-\lambda_k\le\eta_k\lambda_k\le\bar\lambda\eta_k$. Hence
	\begin{equation*}
		(\lambda_{k+1}-\lambda_k)_+
		\le \bar\lambda\eta_k.
	\end{equation*}
	Since $\sum_{k=0}^{\infty}\eta_k<\infty$, it follows that
	\begin{equation*}
		\sum_{k=0}^{\infty}(\lambda_{k+1}-\lambda_k)_+<\infty.
	\end{equation*}
	Moreover, for every $N\ge0$,
	\begin{align*}
		\lambda_{N+1}-\lambda_0
		&=\sum_{k=0}^{N}(\lambda_{k+1}-\lambda_k)_+
		-\sum_{k=0}^{N}(\lambda_{k+1}-\lambda_k)_-.
	\end{align*}
	Since $\lambda_{N+1}\ge0$,
	\begin{equation*}
		\sum_{k=0}^{N}(\lambda_{k+1}-\lambda_k)_-
		\le
		\lambda_0+\sum_{k=0}^{N}(\lambda_{k+1}-\lambda_k)_+.
	\end{equation*}
	Letting $N\to\infty$ proves the second assertion in \eqref{eq:lambda-variation}, and therefore \eqref{eq:lambda-total-variation} follows.
	
	Finally, \eqref{eq:lambda-total-variation} implies that $\{\lambda_k\}$ is a Cauchy sequence in $\mathbb{R}$. Thus it converges to a finite limit $\lambda_\infty\ge0$, proving $\mathbf{(ii)}$.

\label{th61}

	$\{t_k\}$ and $\{z_k\}$ be generated by Algorithm~\ref{alg3}. Then:
	\begin{enumerate}

			\|z_{k+1}-t^*\|^2\le\|z_k-t^*\|^2-\frac12\|z_k-t_k\|^2

				+\bar{\lambda}^{\frac{2-\alpha}{2(1-\alpha)}}K_2

				\left(

\textbf{(i)}
	By \eqref{eq:operator-norm-control} and the nonexpansiveness of $P_U$, for every $k\ge1$,
	\begin{align}
		\|z_k-t_k\|
		&=\left\|P_U(z_k)-P_U\bigl(z_k-\lambda_kF(z_k)\bigr)\right\| \notag\\
		&\le
		\lambda_k\|F(z_k)\|=
		\sqrt{\lambda_k}
		\bigl(\lambda_k\|F(z_k)\|^2\bigr)^{1/2}
		\le	\sqrt{\bar{\lambda}}.
		\label{eq:zk-tk-bound}
	\end{align}
	Suppose first that $\alpha\in(0,1)$. Applying Proposition~\ref{pro3.1} with $y=t_k$ and $y'=z_k$ gives
	\begin{equation}\label{eq:operator-difference-alpha-pre}
		\|F(z_k)-F(t_k)\|\le\|z_k-t_k\|
		\left(K_0+K_1\|F(z_k)\|^\alpha
		+K_2\|z_k-t_k\|^{\frac{\alpha}{1-\alpha}}\right).
	\end{equation}
	Moreover, \eqref{eq:zk-tk-bound} implies
	\begin{equation}\label{eq:zk-tk-bound-1}
		\|z_k-t_k\|^{\frac{\alpha}{1-\alpha}}
		\le
		\bar{\lambda}^{\frac{\alpha}{2(1-\alpha)}}.
	\end{equation}
	Using \eqref{eq:operator-norm-control}, \eqref{eq:zk-tk-bound}, and \eqref{eq:zk-tk-bound-1}, we obtain
	\begin{align*}
		\sqrt{\lambda_k}\|F(z_k)-F(t_k)\|
		&\le
		\bar{\lambda}K_0
		+\sqrt{\bar{\lambda}}K_1
		\lambda_k^{\frac{1-\alpha}{2}}
		\bigl(\lambda_k\|F(z_k)\|^2\bigr)^{\alpha/2}
		+K_2\bar{\lambda}^{\frac{2-\alpha}{2(1-\alpha)}}\\
		&\le
		\bar{\lambda}K_0
		+\bar{\lambda}^{\frac{2-\alpha}{2}}K_1
		+\bar{\lambda}^{\frac{2-\alpha}{2(1-\alpha)}}K_2.
	\end{align*}
	Similarly, when $\alpha=1$, 
	\begin{align*}
		\sqrt{\lambda_k}\|F(z_k)-F(t_k)\|
		&\le
		\sqrt{\lambda_k}\|z_k-t_k\|
		\bigl(L_0+L_1\|F(z_k)\|\bigr)
		e^{L_1\|z_k-t_k\|}\\
		&\le
		\left(\bar{\lambda}L_0
		+\sqrt{\bar{\lambda}}L_1\right)
		e^{L_1\sqrt{\bar{\lambda}}}.
	\end{align*}
	Hence, in both cases,
	\begin{equation}\label{eq:operator-difference-bound}
		\lambda_k\|F(z_k)-F(t_k)\|^2
		\le M,
		\qquad k\ge1.
	\end{equation}
	Since
	\begin{equation*}
		\lambda_k^2\le\lambda_k\lambda_{k+1}
		+\lambda_k(\lambda_k-\lambda_{k+1})+,
	\end{equation*}
	Lemma~\ref{lem32}, \eqref{eq:local-difference-control}, and \eqref{eq:operator-difference-bound} imply that, for every $y\in U$,

		\|z_{k+1}-y\|^2\le&
		\|z_k-y\|^2-\frac12\|z_k-t_k\|^2
		-2\lambda_k\langle F(t_k),t_k-y\rangle
		+M(\lambda_k-\lambda_{k+1})_+.
	\end{align*}
	Taking $y=t^*\in U^*$ and using $p$-quasi-sharpness proves \eqref{eq:descent}.
	
	\medskip
	\noindent\textbf{(ii)}
	Fix any $t^*\in U^*$. Summing \eqref{eq:descent} from $k=1$ to $n$ and dropping the nonpositive terms gives
	\begin{equation*}
		\|z_{n+1}-t^*\|^2\le\|z_1-t^*\|^2+
		M\sum_{k=1}^{n}(\lambda_k-\lambda_{k+1})+.
	\end{equation*}
	By Lemma~\ref{lem31}, the series on the right-hand side is convergent. Hence $\{z_k\}$ is bounded. Since $U$ is closed and $F$ is continuous, there exists $C>0$ such that
	\begin{equation}\label{eq:Fzk-bound}
		\|F(z_k)\|\le C,
		\qquad k\ge0.
	\end{equation}
	If $\alpha\in(0,1)$, then \eqref{eq:operator-difference-alpha-pre}, \eqref{eq:zk-tk-bound-1}, and \eqref{eq:Fzk-bound} give
	\begin{equation*}\label{eq:operator-difference-alpha}
		\|F(z_k)-F(t_k)\|\le\|z_k-t_k\|
		\left(K_0+K_1C^\alpha
		+K_2\bar{\lambda}^{\frac{\alpha}{2(1-\alpha)}}\right),
		\qquad k\ge1.
	\end{equation*}
	Similarly, if $\alpha=1$,
	\begin{equation*}\label{eq:operator-difference-one}
		\|F(z_k)-F(t_k)\|\le\|z_k-t_k\|
		(L_0+L_1C)e^{L_1\sqrt{\bar{\lambda}}},
		\qquad k\ge1.
	\end{equation*}
	Therefore, whenever $F(z_k)\ne F(t_k)$,
	\begin{equation*}
		\frac{\|z_k-t_k\|^2}
		{2\lambda_k\|F(z_k)-F(t_k)\|^2}
		\ge
		\begin{cases}
			\displaystyle
			\frac{1}{
				2\bar{\lambda}
				\left(
				K_0+K_1C^\alpha
				+K_2\bar{\lambda}^{\frac{\alpha}{2(1-\alpha)}}
				\right)^2},
			& \alpha\in(0,1),\\[4mm]
			\displaystyle
			\frac{1}{
				2\bar{\lambda}(L_0+L_1C)^2
				e^{2L_1\sqrt{\bar{\lambda}}}},
			& \alpha=1.
		\end{cases}
	\end{equation*}
	If $F(z_k)=F(t_k)$, the corresponding candidate is interpreted as $+\infty$.
	By the definition of $\underline{\lambda}$, 
	$\lambda_0\ge\underline{\lambda}$ and
	$\lambda_1\ge\underline{\lambda}$. Suppose that
	$\lambda_k\ge\underline{\lambda}$ for some $k\ge1$. The first
	term in \eqref{eq:pf-aeg-stepsize} satisfies
	$(1+\eta_k)\lambda_k\ge\underline{\lambda}$. The preceding estimates show that the second candidate in \eqref{eq:pf-aeg-stepsize} is also no smaller than $\underline{\lambda}$.  Finally, by \eqref{eq:Fzk-bound}, the third candidate satisfies
	$1/\|F(z_{k+1})\|^2\ge1/C^2\ge\underline{\lambda}$, with the
	convention $1/0=+\infty$. Hence
	$\lambda_{k+1}\ge\underline{\lambda}$. By induction,
	\eqref{eq:lambda-lower} holds for all $k\ge0$.
	
	
	\medskip
	\noindent\textbf{(iii)}
	It follows from \eqref{eq:local-difference-control} and
	Lemma~\ref{lem32} that, for every $y\in U$,
	\begin{equation}\label{6.9}
		                  
			\|z_{k+1}-y\|^2\le\|z_k-y\|^2
			-\left(1-\frac{\lambda_k}{2\lambda_{k+1}}\right)\|z_k-t_k\|^2
			-2\lambda_k\langle F(t_k),t_k-y\rangle.
			                
	\end{equation}
By Lemma~\ref{lem31} and \eqref{eq:lambda-lower}, $\{\lambda_k\}$ converges to a positive limit. Hence $\frac{\lambda_k}{\lambda_{k+1}}\to1$.
	                   
		                                       
		                 
	Therefore, there exists $k_0\ge1$ such that
	\begin{equation*}\label{eq:ratio-control}
		1-\frac{\lambda_k}{2\lambda_{k+1}}
		\ge\frac14,
		\qquad k\ge k_0.
	\end{equation*}
	Fix an arbitrary $t^*\in U^*$. Taking $y=t^*\in U^*$ in \eqref{6.9} and applying $p$-quasi-sharpness and \eqref{eq:lambda-lower} give
	\begin{equation}\label{6.30}
		                  
			\|z_{k+1}-t^*\|^2\le\|z_k-t^*\|^2
			-\frac14\|z_k-t_k\|^2
			-2\mu\underline{\lambda}
			\operatorname{dist}^p(t_k,U^*),
			\qquad k\ge k_0.
			                
	\end{equation}
	Applying Lemma~\ref{lem23} to the tail sequence in \eqref{6.30} shows that $\{\|z_k-t^*\|^2\}$ converges and
	\begin{equation*}\label{eq:summability-tail}
		\sum_{k=k_0}^{\infty}\|z_k-t_k\|^2<\infty,
		\qquad
		\sum_{k=k_0}^{\infty}\operatorname{dist}^p(t_k,U^*)<\infty.
	\end{equation*}
	Since only finitely many indices precede $k_0$, we get
	\begin{equation*}\label{eq:summability-global}
		\sum_{k=0}^{\infty}\|z_k-t_k\|^2<\infty,
		\qquad
		\sum_{k=0}^{\infty}\operatorname{dist}^p(t_k,U^*)<\infty.
	\end{equation*}
	Consequently,
	\begin{equation}\label{6.31}
		\|z_k-t_k\|\to0,
		\qquad
		\operatorname{dist}(t_k,U^*)\to0.
	\end{equation}
	
	We next show that every accumulation point of
	$\{z_k\}$ belongs to $U^*$.
	By part $\mathbf{(ii)}$, the sequence $\{z_k\}$ is bounded.
	Hence it has an accumulation point in $\mathbb{R}^m$.
	Let $\{z_{k_i}\}$ be a convergent subsequence such that $z_{k_i}\to\bar t$. 
	Since $z_{k_i}\in U$ and $U$ is closed, we have
	$\bar t\in U$.
	Moreover, \eqref{6.31} gives $\|t_{k_i}-z_{k_i}\|\to0$, 
	and therefore $t_{k_i}\to\bar t$. 
	Since the distance function is continuous,
	\eqref{6.31} yields
	\begin{equation*}
		\operatorname{dist}(\bar t,U^*)=
		\lim_{i\to\infty}\operatorname{dist}(t_{k_i},U^*)=0.
	\end{equation*}
	Since $U^*$ is closed, $\bar t\in U^*$. Thus every accumulation point of $\{z_k\}$ belongs to $U^*$.
	
	Finally, \eqref{6.30} implies that the tail sequence $\{z_k\}_{k\ge k_0}$ 
	is Fej\'er monotone with respect to $U^*$.
	Applying Lemma~\ref{lem:Fejer} to this tail sequence, we conclude
	that there exists $\bar t\in U^*$ such that $z_k\to\bar t$. 
	Together with $\|z_k-t_k\|\to0$, this further gives $t_k\to\bar t$. 
	Hence both $\{z_k\}$ and $\{t_k\}$ converge to the same
	solution $\bar t\in U^*$.

By Theorem~\ref{th61}(iii), there exists $\bar t\in U^*$ such that $t_k\to\bar t$. Hence $\operatorname{dist}(t_k,U^*)\to0$.
                   
                                    
                 
	Therefore, there exists $B>0$ such that
	\begin{equation}\label{eq:B-bound}
		\operatorname{dist}(t_k,U^*)\le B,
		\qquad k\ge0.
	\end{equation}
	Let $k_0\ge1$ be such that \eqref{6.30} holds for all $k\ge k_0$. For each such $k$, let $z_k^*$ be the projection of $z_k$ onto	$U^*$, i.e., 
                                                                                           
	\begin{equation*}
		\|z_k-z_k^*\|
		=
		\operatorname{dist}(z_k,U^*).
	\end{equation*}
	Taking $t^*=z_k^*$ in \eqref{6.30} gives
	\begin{equation}\label{eq:rate-basic}
		\operatorname{dist}^2(z_{k+1},U^*)
		\le
		\operatorname{dist}^2(z_k,U^*)
		-\frac14\|z_k-t_k\|^2
		-2\mu\underline{\lambda}
		\operatorname{dist}^p(t_k,U^*),
		\qquad k\ge k_0.
	\end{equation}
	Moreover, the triangle inequality yields
	\begin{equation}\label{eq:distance-split}
		\operatorname{dist}(z_k,U^*)
		\le\|z_k-t_k\|+\operatorname{dist}(t_k,U^*).
	\end{equation}
	
	\medskip
	\noindent\textbf{(i)}
	Suppose $0<p\le2$. Squaring \eqref{eq:distance-split} and using
	\eqref{eq:B-bound}, we obtain
	\begin{align*}
		\operatorname{dist}^2(z_k,U^*)
		&\le2\|z_k-t_k\|^2
		+2\operatorname{dist}^2(t_k,U^*)
		\le2\|z_k-t_k\|^2+2B^{2-p}\operatorname{dist}^p(t_k,U^*).
	\end{align*}
	By the definition of $\rho$ in \eqref{eq:rho},
	\begin{equation*}
		2(1-\rho)\le\frac14,
		\qquad
		2(1-\rho)B^{2-p}\le2\mu\underline{\lambda}.
	\end{equation*}
	Hence
	\begin{equation*}
		(1-\rho)\operatorname{dist}^2(z_k,U^*)
		\le\frac14\|z_k-t_k\|^2
		+2\mu\underline{\lambda}
		\operatorname{dist}^p(t_k,U^*).
	\end{equation*}
	Combining this inequality with \eqref{eq:rate-basic} yields
	\begin{equation*}
		\operatorname{dist}^2(z_{k+1},U^*)
		\le\rho\,\operatorname{dist}^2(z_k,U^*),
		\qquad k\ge k_0,
	\end{equation*}
	which implies
	\begin{equation*}\label{eq:linear-tail-rate}
		\operatorname{dist}^2(z_k,U^*)
		\le\rho^{\,k-k_0}
		\operatorname{dist}^2(z_{k_0},U^*),
		\qquad k\ge k_0.
	\end{equation*}
                                                                            
                                                        
	Then \eqref{eq:global-linear-rate} holds for every $k\ge0$ by setting 
	\begin{equation*}
		c:=\max_{0\le j\le k_0}\rho^{-j}\operatorname{dist}^2(z_j,U^*).
	\end{equation*}
		
		
	\medskip
	\noindent\textbf{(ii)}
	Suppose $p>2$. By \eqref{eq:distance-split}, the fact that ${\left(a + b\right)}^{p} \leq  {2}^{p - 1}\left(a^{p} + b^{p}\right)$, and
	$\|z_k-t_k\|\le\sqrt{\bar{\lambda}}$, we have
	\begin{align*}
		\operatorname{dist}^p(z_k,U^*)
		&\le\left(\|z_k-t_k\|+\operatorname{dist}(t_k,U^*)\right)^p\\
		&\le2^{p-1}\left(\|z_k-t_k\|^p+\operatorname{dist}^p(t_k,U^*)
		\right)\\
		&\le2^{p-1}\left(\bar{\lambda}^{\frac{p-2}{2}}
		\|z_k-t_k\|^2+\operatorname{dist}^p(t_k,U^*)\right).
	\end{align*}
	Thus
	\begin{align*}
		&2^{2-p}
		\min\left\{
			\mu\underline{\lambda},
			\frac{1}{8\bar{\lambda}^{(p-2)/2}}
			\right\}
		\operatorname{dist}^p(z_k,U^*)
		\le\frac14\|z_k-t_k\|^2
		+2\mu\underline{\lambda}
		\operatorname{dist}^p(t_k,U^*).
	\end{align*}
	Combining this estimate with \eqref{eq:rate-basic} gives
	\begin{equation}\label{eq:sublinear-tail-recursion}
		\operatorname{dist}^2(z_{k+1},U^*)
		\le\operatorname{dist}^2(z_k,U^*)
		-2^{2-p}\min\left\{
			\mu\underline{\lambda},,
			\frac{1}{8\bar{\lambda}^{(p-2)/2}}
			\right\}
		\operatorname{dist}^p(z_k,U^*),
		\qquad k\ge k_0.
	\end{equation}
	Applying Lemma~\ref{lem25} to \eqref{eq:sublinear-tail-recursion} with $q=(p-2)/2$ yields
	\begin{equation}\label{eq:sublinear-tail-rate}
		\operatorname{dist}^2(z_k,U^*)
		\le\frac{\operatorname{dist}^2(z_{k_0},U^*)}{
			\left[1+a_p(k-k_0)
			\operatorname{dist}^{p-2}(z_{k_0},U^*)
			\right]^{2/(p-2)}
		},
		\qquad k\ge k_0,
	\end{equation}
	where
	\begin{equation*}
		a_p=2^{1-p}(p-2)
		\min\left\{
			\mu\underline{\lambda},
			\frac{1}{8\bar{\lambda}^{(p-2)/2}}
			\right\}.
	\end{equation*}
	If $\operatorname{dist}(z_{k_0},U^*)=0$, then
	\eqref{eq:sublinear-tail-recursion} implies
	$\operatorname{dist}(z_k,U^*)=0$ for all $k\ge k_0$, and the
	result follows immediately. 
	Otherwise, \eqref{eq:sublinear-tail-rate} implies, for $k\ge k_0+1$,
	\begin{equation*}
		\operatorname{dist}^2(z_k,U^*)
		\le\frac{1}{[a_p(k-k_0)]^{2/(p-2)}}.
	\end{equation*}
	For $k\ge k_0+1$, $k-k_0\ge\frac{k+1}{k_0+2}$,
	and hence
	\begin{equation*}
		\operatorname{dist}^2(z_k,U^*)\le
		\left(\frac{k_0+2}{a_p}\right)^{2/(p-2)}
		\frac{1}{(k+1)^{2/(p-2)}}.
	\end{equation*}
	Thus \eqref{eq:global-sublinear-rate} holds for every $k\ge0$ with
	\begin{equation*}
		c_p=\max\left\{
			\max_{0\le j\le k_0}
			(j+1)^{2/(p-2)}
			\operatorname{dist}^2(z_j,U^*),
			\left(\frac{k_0+2}{a_p}\right)^{2/(p-2)}
			\right\}.
	\end{equation*}

	Since $\bar t^k=\beta t^k+(1-\beta)s^k$, the standard convex-combination identity gives, for every $u\in\mathbb R^m$,
	\begin{equation*}
		\|\bar t^k-u\|^2=\beta\|t^k-u\|^2
		+(1-\beta)\|s^k-u\|^2
		-\beta(1-\beta)\|t^k-s^k\|^2.
	\end{equation*}
	Applying this identity with $u=t^*$ and
	$u=t^{k+\frac12}$, and subtracting the resulting equalities, yields
	\begin{align*}
		\|\bar t^k-t^*\|^2
		-\|\bar t^k-t^{k+\frac12}\|^2
		={}&\beta\|t^k-t^*\|^2
		+(1-\beta)\|s^k-t^*\|^2\\
		&-\beta\|t^k-t^{k+\frac12}\|^2
		-(1-\beta)\|s^k-t^{k+\frac12}\|^2.
	\end{align*}
	Substituting this equality into
	\eqref{eq:vr-after-young} gives
	\begin{align}
		\|t^{k+1}-t^*\|^2
		\le{}&\beta\|t^k-t^*\|^2
		+(1-\beta)\|s^k-t^*\|^2
		-2\lambda_k\left\langle
		t^{k+\frac12}-t^*,\widehat F_k\right\rangle
		\notag\\
		&-(1-\beta)
		\|t^{k+1}-t^{k+\frac12}\|^2
		-(1-\beta)\|s^k-t^{k+\frac12}\|^2
		-\beta\|t^k-t^{k+\frac12}\|^2
		\notag\\
		&+\frac{\lambda_k^2}{\beta}\|F_{\xi_k}(t^{k+\frac12})-F_{\xi_k}(s^k)\|^2.
		\label{eq:vr-prelocal}
	\end{align}
	
	
	
	We next control the last term. By the second candidate in
	\eqref{eq:aeg-vr-stepsize},
	\begin{equation*}
		\frac{\lambda_{k+1}^2}{\beta}
		\|F_{\xi_k}(t^{k+\frac12})-F_{\xi_k}(s^k)\|^2
		\le\beta\|t^{k+\frac12}-s^k\|^2.
	\end{equation*}
	Using $\lambda_k^2\le\lambda_{k+1}^2+2\lambda_k(\lambda_k-\lambda_{k+1})_+$, we have
	\begin{align}
		\frac{\lambda_k^2}{\beta}
		\|F_{\xi_k}(t^{k+\frac12})-F_{\xi_k}(s^k)\|^2
		\le{}&\beta\|t^{k+\frac12}-s^k\|^2
		+\frac{2\lambda_k}{\beta}
		\|F_{\xi_k}(t^{k+\frac12})-F_{\xi_k}(s^k)\|^2
		(\lambda_k-\lambda_{k+1})_+.
		\label{eq:vr-error-split}
	\end{align}
	Since $s^k\in U$,	nonexpansiveness of $P_U$, \eqref{eq:snapshot-gap}, and
	\eqref{eq:stochastic-operator-control} yield
	\begin{align}
		\|t^{k+\frac12}-s^k\|
		&=\left\|P_U\!\left(\bar t^k-\lambda_kF(s^k)\right)-P_U(s^k)\right\|
		\notag\\
		&\le\beta\|t^k-s^k\|+\lambda_k\|F(s^k)\|
		\notag\\
		&\le\beta w+\lambda_k\max_{i\in[N]}\|F_i(s^k)\|
		\notag\\
		&\le\beta w+\sqrt{\bar\lambda}=R.
		\label{eq:vr-prediction-gap}
	\end{align}
	If $\alpha\in(0,1)$, by
	Assumption~\ref{ass4}, Proposition~\ref{pro3.1}, and
	\eqref{eq:vr-prediction-gap},
	\begin{align*}
		&\sqrt{\frac{2\lambda_k}{\beta}}\,
		\|F_{\xi_k}(t^{k+\frac12})-F_{\xi_k}(s^k)\|\\
		&\quad\le\sqrt{\frac{2}{\beta}}\,
		\|t^{k+\frac12}-s^k\|
		\Bigl[\sqrt{\lambda_k}
		\left(K_0+K_2\|t^{k+\frac12}-s^k\|^{\frac{\alpha}{1-\alpha}}\right)
		+K_1\sqrt{\lambda_k}\,\|F_{\xi_k}(s^k)\|^\alpha\Bigr]\\
		&\quad\le\sqrt{\frac{2}{\beta}}\,R
		\left[\sqrt{\bar\lambda}
		\left(K_0+K_2R^{\frac{\alpha}{1-\alpha}}\right)
		+\bar\lambda^{\frac{1-\alpha}{2}}K_1\right].
	\end{align*}
	If $\alpha=1$, Proposition~\ref{pro3.1} similarly gives
	\begin{align*}
		&\sqrt{\frac{2\lambda_k}{\beta}}\,
		\|F_{\xi_k}(t^{k+\frac12})-F_{\xi_k}(s^k)\|
		\le\sqrt{\frac{2}{\beta}}\,R
		\left(\sqrt{\bar\lambda}L_0+L_1\right)e^{L_1R},
	\end{align*}
	where we used $\sqrt{\lambda_k}\|F_{\xi_k}(s^k)\|
	\le1$. Thus, in both cases, by the definition of $\bar M$,
	\begin{equation}\label{eq:vr-component-bound}
		\frac{2\lambda_k}{\beta}
		\|F_{\xi_k}(t^{k+\frac12})-F_{\xi_k}(s^k)\|^2
		\le\bar M.
	\end{equation}
	Combining \eqref{eq:vr-error-split} and
	\eqref{eq:vr-component-bound} gives
	\begin{align*}
		\frac{\lambda_k^2}{\beta}
		\|F_{\xi_k}(t^{k+\frac12})-F_{\xi_k}(s^k)\|^2
		\le&\beta\|t^{k+\frac12}-s^k\|^2
		+\bar M(\lambda_k-\lambda_{k+1})_+.
	\end{align*}
	Substituting this estimate into \eqref{eq:vr-prelocal} yields
	\begin{align*}
		\|t^{k+1}-t^*\|^2
		\le{}&\beta\|t^k-t^*\|^2
		+(1-\beta)\|s^k-t^*\|^2-2\lambda_k
		\langle t^{k+\frac12}-t^*,\widehat F_k\rangle\\
		&-(1-\beta)\|t^{k+1}-t^{k+\frac12}\|^2
		-(1-2\beta)\|s^k-t^{k+\frac12}\|^2
		-\beta\|t^k-t^{k+\frac12}\|^2\\
		&+\bar M(\lambda_k-\lambda_{k+1})_+.
	\end{align*}
	
	Finally, taking conditional expectation with respect to
	$\mathcal F_k$, using \eqref{eq:svrg-unbiased}, gives
	\begin{equation*}

		\langle t^{k+\frac12}-t^*,\widehat F_k\rangle
		=-\langle t^{k+\frac12}-t^*,F(t^{k+\frac12})\rangle
		\le-\mu\operatorname{dist}^p(t^{k+\frac12},U^*),
	\end{equation*}
	where the last inequality is due to $p$-quasi-sharpness.
	Substituting this inequality into the preceding estimate proves
	\eqref{eq:vr-one-step}.

		\qquad k\ge1.
	\end{equation}

	index $k=0$ does not affect the conclusion.

	Moreover, $\lambda_0\ge\underline{\lambda}$ and $\lambda_1\ge\underline{\lambda}$.	Suppose that
	$\lambda_k\ge\underline{\lambda}$ for some $k\ge1$.
	Then the first candidate in \eqref{eq:aeg-vr-stepsize} satisfies $(1+\eta_k)\lambda_k\ge\lambda_k\ge\underline{\lambda}$.
	By \eqref{eq:vr-local-candidate-alpha} or
	\eqref{eq:vr-local-candidate-one}, the second candidate is also no smaller
	than $\underline{\lambda}$. Finally,
	\begin{equation*}
		\frac{1}{\max_{i\in[N]}\|F_i(s^{k+1})\|^2}
		\ge\frac1{H_1^2}
		\ge\underline{\lambda}.
	\end{equation*}
	Thus every candidate in \eqref{eq:aeg-vr-stepsize} is bounded below by
	$\underline{\lambda}$, and hence $\lambda_{k+1}\ge\underline{\lambda}$.
	                  
	                                      
	                
	By induction, \eqref{eq:vr-lambda-lower} holds almost surely. This proves part~$\mathbf{(ii)}$.


	\medskip
	\noindent\textbf{(iii)}
	We finally prove the convergence of the iterates. By
	\eqref{eq:vr-tail-summability} and
	\eqref{eq:vr-lambda-lower},
	\begin{equation*}
		\sum_{k=1}^{\infty}
		\operatorname{dist}^p(t^{k+\frac12},U^*)
		\le\frac1{\underline{\lambda}}
		\sum_{k=1}^{\infty}
		\lambda_k\operatorname{dist}^p(t^{k+\frac12},U^*)
		<\infty
	\end{equation*}
	almost surely. Hence
	\begin{equation}\label{eq:vr-residual-limits}
		\operatorname{dist}(t^{k+\frac12},U^*)\to0,
		\qquad
		\|s^k-t^{k+\frac12}\|\to0,
		\qquad
		\|t^k-t^{k+\frac12}\|\to0
	\end{equation}
	almost surely, where the last two limits follow directly from
	\eqref{eq:vr-tail-summability}. Consequently,
	\begin{equation}\label{eq:tk-sk-zero}
		\|t^k-s^k\|\le\|t^k-t^{k+\frac12}\|
		+\|t^{k+\frac12}-s^k\|\to0
		\qquad\text{almost surely}.
	\end{equation}
	
	We next show that every accumulation point of $\{t^k\}$ belongs to
	$U^*$. Fix a sample path on which the preceding properties hold.
	By part~$\mathbf{(i)}$, $\{t^k\}$ is bounded. Let
	$\{t^{k_j}\}$ be a convergent subsequence such that $t^{k_j}\to u$. 
	By \eqref{eq:vr-residual-limits}, $\|t^{k_j+\frac12}-t^{k_j}\|\to0$, and hence $t^{k_j+\frac12}\to u$.	Since the distance function is continuous,
	\eqref{eq:vr-residual-limits} yields
	\begin{equation*}
		\operatorname{dist}(u,U^*)
		=\lim_{j\to\infty}
		\operatorname{dist}(t^{k_j+\frac12},U^*)=0.
	\end{equation*}
	Since $U^*$ is closed, $u\in U^*$. Thus every accumulation point of
	$\{t^k\}$ belongs to $U^*$.
	
	It remains to show that the accumulation point is unique. Since
	$U^*\subseteq\mathbb R^m$ is separable, let
	$\{u_j^*\}_{j\ge1}$ be a countable dense subset of  $U^*$.
	By part~$\mathbf{(i)}$, for every $j\ge1$,
	$\Phi_k(u_j^*)$ converges almost surely. Hence there exists an event of
	probability one on which
	\begin{equation*}\label{eq:Phi-dense-convergence}
		\Phi_k(u_j^*)
		\quad\text{converges for every }j\ge1.
	\end{equation*}
	Fix a sample path in this event, and let $u',u''\in U^*$ be two
	accumulation points of $\{t^k\}$. Then there exist subsequences such that
	\begin{equation*}
		t^{k_\ell}\to u',
		\qquad
		t^{m_\ell}\to u''.
	\end{equation*}
	By \eqref{eq:tk-sk-zero}, we have
	\begin{equation*}
		s^{k_\ell}\to u',
		\qquad
		s^{m_\ell}\to u''.
	\end{equation*}
	For every $j\ge1$, the convergence of
	$\Phi_k(u_j^*)$ implies that its limits along these two subsequences
	coincide. Therefore,
	\begin{equation}\label{eq:cluster-equality}
		\left(\beta+\frac{1-\beta}{P_1}\right)\|u'-u_j^*\|^2
		=\left(\beta+\frac{1-\beta}{P_1}\right)\|u''-u_j^*\|^2,
		\qquad j\ge1.
	\end{equation}
	By density, there exists a subsequence
	$\{u_{j_\ell}^*\}$ such that $u_{j_\ell}^*\to u'$. Passing to the limit in \eqref{eq:cluster-equality} along this subsequence
	gives
	\begin{equation*}
		\left(\beta+\frac{1-\beta}{P_1}\right)\|u''-u'\|^2=0.
	\end{equation*}
	Since $\beta+\frac{1-\beta}{P_1}>0$, we obtain $u''=u'$. Thus $\{t^k\}$ has a unique accumulation point.	Since it is bounded, there exists $\bar t\in U^*$ such that
	\begin{equation*}
		t^k\to\bar t
		\qquad\text{almost surely}.
	\end{equation*}
	Finally, \eqref{eq:tk-sk-zero} implies
	\begin{equation*}
		s^k\to\bar t
		\qquad\text{almost surely}.
	\end{equation*}
	This proves \eqref{eq:vr-as-convergence} and part~$\mathbf{(iii)}$.

By \eqref{eq:vr-rate-uniformity},
	Theorem~\ref{th41}(ii), and
	Lemma~\ref{lem:vr-eventual-stability}, for every
	$k\ge k_{\rm r}$, almost surely,
	\begin{equation}\label{eq:vr-rate-tail-bounds}
		\lambda_k\ge\lambda_{\min},
		\qquad
		\|t^{k+1}-s^k\|\le w,
		\qquad
		1-\beta-\beta\frac{\lambda_k^2}{\lambda_{k+1}^2}
		\ge\frac{1-2\beta}{2}.
	\end{equation}
	Fix $k\ge k_{\rm r}$. The second inequality in
	\eqref{eq:vr-rate-tail-bounds} implies that the snapshot is updated
	according to the Bernoulli rule. Hence, for every $u^*\in U^*$,
	\begin{equation}\label{eq:vr-rate-snapshot}
		\frac{1-\beta}{P_1}
		\mathbb E_k^+\|s^{k+1}-u^*\|^2
		=(1-\beta)\|t^{k+1}-u^*\|^2
		+(1-\beta)\left(\frac1{P_1}-1\right)\|s^k-u^*\|^2.
	\end{equation}
	By the second candidate in \eqref{eq:aeg-vr-stepsize},
	\begin{equation}\label{eq:vr-rate-local-control}
		\frac{\lambda_k^2}{\beta}
		\|F_{\xi_k}(t^{k+\frac12})-F_{\xi_k}(s^k)\|^2
		\le\beta\frac{\lambda_k^2}{\lambda_{k+1}^2}
		\|t^{k+\frac12}-s^k\|^2.
	\end{equation}
	If the operator difference vanishes, this inequality holds trivially.
	Substituting \eqref{eq:vr-rate-local-control} into
	\eqref{eq:vr-prelocal} gives
	\begin{align}
		\|t^{k+1}-u^*\|^2
		\le{}&
		\beta\|t^k-u^*\|^2
		+(1-\beta)\|s^k-u^*\|^2
		-2\lambda_k
		\langle t^{k+\frac12}-u^*,\widehat F_k\rangle
		\notag\\
		&-(1-\beta)\|t^{k+1}-t^{k+\frac12}\|^2
		-\left(1-\beta-\beta\frac{\lambda_k^2}{\lambda_{k+1}^2}\right)
		\|s^k-t^{k+\frac12}\|^2
		\notag\\
		&-\beta\|t^k-t^{k+\frac12}\|^2.
		\label{eq:vr-rate-pre}
	\end{align}
	Using \eqref{eq:vr-rate-tail-bounds} in
	\eqref{eq:vr-rate-pre} and dropping the nonpositive term involving
	$\|t^{k+1}-t^{k+\frac12}\|^2$, we obtain
	\begin{align*}
		\|t^{k+1}-u^*\|^2
		\le{}&
		\beta\|t^k-u^*\|^2
		+(1-\beta)\|s^k-u^*\|^2
		-2\lambda_k
		\langle t^{k+\frac12}-u^*,\widehat F_k\rangle
		\notag\\
		&-
		\frac{1-2\beta}{2}
		\|s^k-t^{k+\frac12}\|^2
		-\beta\|t^k-t^{k+\frac12}\|^2.
                                     
	\end{align*}
	Taking conditional expectation with respect to $\mathcal F_k$, using
	\eqref{eq:svrg-unbiased}, $p$-quasi-sharpness, and
	$\lambda_k\ge\lambda_{\min}$, gives
	\begin{align}
		\mathbb E_k\|t^{k+1}-u^*\|^2
		\le{}&
		\beta\|t^k-u^*\|^2
		+(1-\beta)\|s^k-u^*\|^2
	-2\mu\lambda_{\min}
		\operatorname{dist}^p(t^{k+\frac12},U^*)
		\notag\\
		&-\frac{1-2\beta}{2}
		\|s^k-t^{k+\frac12}\|^2
		-\beta\|t^k-t^{k+\frac12}\|^2.
		\label{eq:vr-rate-t}
	\end{align}
	Taking conditional expectation with respect to $\mathcal F_k$ in
	\eqref{eq:vr-rate-snapshot} and using the tower property, we have
	\begin{align*}

		\left[\beta\|t^{k+1}-u^*\|^2
		+\frac{1-\beta}{P_1}
		\|s^{k+1}-u^*\|^2\right]
		=\mathbb E_k\|t^{k+1}-u^*\|^2
		+(1-\beta)\left(\frac1{P_1}-1\right)

		-\frac{1-2\beta}{2}
		\|s^k-t^{k+\frac12}\|^2
		-\beta\|t^k-t^{k+\frac12}\|^2.
                               
	\end{align*}

		\inf_{u^*\in U^*}\Phi_{k+1}(u^*)\right]

	we obtain
	\begin{align}
		\mathbb E_k
		\left[\inf_{u^*\in U^*}\Phi_{k+1}(u^*)\right]
		\le{}&
		\inf_{u^*\in U^*}\Phi_k(u^*)
		-2\mu\lambda_{\min}
		\operatorname{dist}^p(t^{k+\frac12},U^*)
		\notag\\
		&-\frac{1-2\beta}{2}

		\qquad k\ge k_{\rm r}.

		\le B^{2-p}
		\operatorname{dist}^p(t^{k+\frac12},U^*).

	For $p=2$, the same relation holds with $B^{2-p}=1$. 
	Combining the definition of $\rho$ in \eqref{eq:vr-rho} and

		\|s^k-t^{k+\frac12}\|^2

		\mathbb E
		\left[\inf_{u^*\in U^*}\Phi_k(u^*)\right]
		\le\rho^{k-k_{\rm r}}

	\end{equation*}

	\noindent\textbf{(ii)}

		-P_U\!\left(\bar t^k-\lambda_kF(s^k)\right)\right\|\\
		&\le(1-\beta)\|t^k-s^k\|+\lambda_k\|F(s^k)\|\\
		&\le(1-\beta)w+\sqrt{\bar\lambda},
		\qquad k\ge1,
	\end{align*}
	which together with \eqref{eq:vr-prediction-gap} gives
	\begin{equation}\label{eq:vr-weighted-gap-bound}
		\beta\|t^k-t^{k+\frac12}\|^2
		+\frac{1-\beta}{P_1}
		\|s^k-t^{k+\frac12}\|^2
		\le	R_0^2,
		\qquad k\ge k_{\rm r},
	\end{equation}
	where
		\begin{equation*}
		R_0^2:=\beta
		\left[(1-\beta)w+\sqrt{\bar\lambda}\right]^2
		+\frac{1-\beta}{P_1}R^2.
	\end{equation*}
	Let $u_k^*\in U^*$ be a nearest point to
	$t^{k+\frac12}$. By the triangle inequality,
	\begin{align*}
		\Phi_k(u_k^*)^{1/2}
		\le{}&
		\left[\beta\|t^k-t^{k+\frac12}\|^2
		+\frac{1-\beta}{P_1}
		\|s^k-t^{k+\frac12}\|^2\right]^{1/2}+
		\left(\beta+\frac{1-\beta}{P_1}\right)^{1/2}
		\operatorname{dist}(t^{k+\frac12},U^*).
	\end{align*}
	Using $(a+b)^p\le2^{p-1}(a^p+b^p)$ and
	\eqref{eq:vr-weighted-gap-bound}, we obtain
	\begin{align*}
		\left(\inf_{u^*\in U^*}\Phi_k(u^*)\right)^{p/2}
		\le{}&
		2^{p-1}R_0^{p-2}
		\left[\beta\|t^k-t^{k+\frac12}\|^2
		+\frac{1-\beta}{P_1}\|s^k-t^{k+\frac12}\|^2\right]
		\notag\\
		&+
		2^{p-1}\left(\beta+\frac{1-\beta}{P_1}\right)^{p/2}
		\operatorname{dist}^p(t^{k+\frac12},U^*),
                                         
	\end{align*}
	which implies
	\begin{align*}
		a_p\left(\inf_{u^*\in U^*}\Phi_k(u^*)\right)^{p/2}
		\le{}&
		\beta\|t^k-t^{k+\frac12}\|^2
		+\frac{1-2\beta}{2}
		\|s^k-t^{k+\frac12}\|^2
		+2\mu\lambda_{\min}
		\operatorname{dist}^p(t^{k+\frac12},U^*),
		\label{eq:vr-sublinear-control}
	\end{align*}
	where 
	\begin{equation*}\label{eq:vr-ap}
		a_p:=
		\min\left\{
		2^{1-p}R_0^{2-p}
		\min\left\{1,\,
		\frac{(1-2\beta)P_1}{2(1-\beta)}\right\},
		\frac{2^{2-p}\mu\lambda_{\min}}{
			\left(\beta+\frac{1-\beta}{P_1}\right)^{p/2}}
		\right\}.
	\end{equation*}
	Combining this estimate with
	\eqref{eq:vr-rate-descent} gives
	\begin{equation}
		\mathbb E_k
		\left[\inf_{u^*\in U^*}\Phi_{k+1}(u^*)\right]
		\le\inf_{u^*\in U^*}\Phi_k(u^*)
		-a_p\left(\inf_{u^*\in U^*}\Phi_k(u^*)\right)^{p/2},
		\qquad k\ge k_{\rm r}.
		\label{eq:vr-sublinear-recursion}
	\end{equation}
	Taking total expectations in
	\eqref{eq:vr-sublinear-recursion} and applying Jensen's inequality,
	since $p/2>1$, yields
	\begin{equation*}\label{eq:vr-sublinear-expectation-recursion}
		\mathbb E
		\left[\inf_{u^*\in U^*}\Phi_{k+1}(u^*)\right]
		\le\mathbb E
		\left[\inf_{u^*\in U^*}\Phi_k(u^*)\right]-a_p\left(\mathbb E
		\left[\inf_{u^*\in U^*}\Phi_k(u^*)\right]\right)^{p/2},
		\qquad k\ge k_{\rm r}.
	\end{equation*}
	Applying Lemma~\ref{lem25} with $q=(p-2)/2$ gives
	\begin{equation}\label{eq:vr-sublinear-tail}
		\mathbb E\left[\inf_{u^*\in U^*}\Phi_{k}(u^*)\right]
		\le\frac{\mathbb E
			\left[\displaystyle\inf_{u^*\in U^*}\Phi_{k_{\rm r}}(u^*)\right]}
			{\left[1+\frac{p-2}{2}
			a_p\left(\mathbb E
			\left[\displaystyle
			\inf_{u^*\in U^*}\Phi_{k_{\rm r}}(u^*)
			\right]\right)^{(p-2)/2}
			(k-k_{\rm r})\right]^{2/(p-2)}},
		\qquad k\ge k_{\rm r}.
	\end{equation}
	If $\mathbb E\left[\displaystyle\inf_{u^*\in U^*}\Phi_{k_{\rm r}}(u^*)\right]=0$, the result follows immediately. Otherwise, for $k\ge k_{\rm r}+1$,
\begin{equation*}
	\mathbb E\left[\inf_{u^*\in U^*}\Phi_k(u^*)\right]
	\le\left[\frac{p-2}{2}a_p(k-k_{\rm r})\right]^{-2/(p-2)}.
\end{equation*}
Since $k-k_{\rm r}\ge\frac{k+1}{k_{\rm r}+2}$ for $k\ge k_{\rm r}+1$, we obtain
\begin{equation*}
	\mathbb E\left[\inf_{u^*\in U^*}\Phi_k(u^*)\right]
	\le\left[\frac{2(k_{\rm r}+2)}{(p-2)a_p}\right]^{2/(p-2)}
	\frac1{(k+1)^{2/(p-2)}}.
\end{equation*}
Thus \eqref{eq:vr-sublinear-rate} holds for every $k\ge0$ by choosing
\begin{equation*}
	c_p:=\max\left\{
	\max_{0\le j\le k_{\rm r}}(j+1)^{2/(p-2)}
	\mathbb E
	\left[\inf_{u^*\in U^*}\Phi_j(u^*)\right],
	\left[\frac{2(k_{\rm r}+2)}{(p-2)a_p}\right]^{2/(p-2)}
	\right\}.
\end{equation*}
