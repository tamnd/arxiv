<!-- definition-1: body_md -->
A weakly holomorphic function $f^{+}:\mathbb{H} \rightarrow {\mathbb C}[{\mathbb Z}_{2p}]  $ is called a \textbf{vector-valued mock modular form} of weight $k \in \frac{1}{2}{\mathbb Z}$ with respect to $\rho_{p}$ and with shadow $g$ if it is the holomorphic part of a harmonic Maass form $\widehat{f}$ of weight $k$ and with respect to $\rho_{p}$ which satisfies $g = \xi_{k}\widehat{f}$.
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{equivalenceofbridges}
    For a positive integer $p$ and $a \in {\mathbb Z}_{2p}$, a vector-valued function $\psi_{p, a}^{\vee}$ satisfies the resurgent characterization of dual false theta functions if and only if it satisfies the modular characterization of dual false theta functions.
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
\textbf{Modular $\implies$ resurgent.} Suppose $\psi_{p,a}^{\vee}$ is a vector-valued mock modular form of weight $\frac{1}{2}$ with respect to $\rho_{-p}$, with shadow $\tfrac{1}{\sqrt{p}}\theta^{1}_{p,a}$ and with Fourier coefficients in $\tfrac{1}{\mathcal{N}_{a}}{\mathbb Z}$ for some integer $\mathcal{N}_{a}$.

Let us denote by $\widehat{\psi}_{p,a}^{\vee}$ the harmonic Maass form (modular completion) associated with $\psi_{p,a}^{\vee}$. It is given by 
\begin{equation} 
\widehat{\psi}_{p,a}^{\vee}(\tau, \bar{\tau}) = \psi_{p,a}^{\vee}(\tau) - \sqrt{\frac{-i}{2p}} \int_{- \bar{\tau}}^{\infty}  \frac{\overline{\theta_{p,a}^{1}(-\bar{z})}}{ \sqrt{z+ \tau} } \dd z.
\end{equation} 
\textit{$T$-transformation.} Under $\tau \mapsto \tau +1$, the completion $\widehat{\psi}_{p,a}^{\vee}$ transforms as $\widehat{\psi}_{p,a}^{\vee}(\tau +1) = e^{- 2 \pi i \tfrac{a^{2}}{4p}} \widehat{\psi}_{p,a}^{\vee}(\tau)$. The non-holomorphic part transforms in the same way:
\begin{equation} 
 \int_{- \bar{\tau}-1}^{\infty}  \frac{\overline{\theta_{p,a}^{1}(-\bar{z})}}{ \sqrt{z+ \tau+1} } \dd z  =   \int_{- \bar{\tau}}^{\infty}  \frac{\overline{\theta_{p,a}^{1}(-\bar{z}+1)}}{ \sqrt{z+ \tau} } \dd z =  e^{- 2 \pi i \tfrac{a^{2}}{4p}} \int_{- \bar{\tau}}^{\infty}  \frac{\overline{\theta_{p,a}^{1}(-\bar{z})}}{ \sqrt{z+ \tau} } \dd z.
\end{equation} 
Hence the holomorphic part satisfies $\psi_{p,a}^{\vee}(\tau+1) = e^{- 2 \pi i \tfrac{a^{2}}{4p}} \psi_{p,a}^{\vee}(\tau)$. Together with the integrality assumption, this means that $\psi^{\vee}_{p, a}(\tfrac{1}{k}) \in \tfrac{q^{\Delta^{\vee}_{a}}}{\mathcal{N}_{a}} {\mathbb Z}[[q]] $ with $\Delta^{\vee}_{a} = - \frac{a^{2}}{4 p} \mod {\mathbb Z}$, which is the first resurgent condition.\footnote{We would like to thank Miranda Cheng for pointing this out.} 


\textit{$S$-transformation.} Under $\tau \mapsto -1/\tau$, $\widehat{\psi}_{p,a}^{\vee}$ transforms as
\begin{equation} 
\widehat{\psi}_{p,a}^{\vee}(\tau, \bar{\tau}) = - \sqrt{\frac{i}{\tau}} \sum_{b =1}^{p-1} \sqrt{\frac{2}{p}} \sin\left(\frac{\pi a b}{p}\right)  \widehat{\psi}_{p,b}^{\vee}(-1/\tau,-1/\bar{\tau}). 
\end{equation} 
Collecting the holomorphic parts on the left-hand side and the non-holomorphic part on the right-hand side, we get 
\begin{align} \label{stransformdftholnonholsep}
& \psi_{p,a}^{\vee}(\tau) + \sqrt{\frac{i}{\tau}} \sum_{b =1}^{p-1} \sqrt{\frac{2}{p}} \sin\left(\frac{\pi a b}{p}\right)  \psi_{p,b}^{\vee}(-1/\tau) \endline  = &  \sqrt{\frac{- i}{2p}} \left[ \int_{- \bar{\tau}}^{\infty}  \frac{\overline{\theta_{p,a}^{1}(-\bar{z})}}{ \sqrt{z+ \tau} } \dd z + \sqrt{\frac{i}{\tau}} \sum_{b =1}^{p-1} \sqrt{\frac{2}{p}} \sin\left(\frac{\pi a b}{p}\right)   \int_{1/\bar{\tau}}^{\infty}  \frac{\overline{\theta_{p,b}^{1}(-\bar{z})}}{ \sqrt{z-1/ \tau} } \dd z  \right].
\end{align}
We now show that the right-hand side is a Borel--Mordell integral. Changing variables $z \mapsto - 1/z$ in the last integral gives
\begin{equation} 
\sqrt{\frac{i}{\tau}}\int_{1/\bar{\tau}}^{\infty}  \frac{\overline{\theta_{p,b}^{1}(-\bar{z})}}{ \sqrt{z-1/ \tau} } \dd z = \sqrt{\frac{i}{\tau}} \int_{- \bar{\tau}}^{0}  \frac{\overline{\theta_{p,b}^{1}(1/\bar{z})}}{ \sqrt{-1/z -1/ \tau} } \frac{1}{z^{2}} \dd z =  \int_{0}^{- \bar{\tau}} \frac{ \overline{\theta_{p,b}^{1}(1/\bar{z})}   }{\sqrt{z + \tau}} \left(\frac{i}{z}\right)^{\frac{3}{2}} \dd z,
\end{equation} 
so the bracket in \eqref{stransformdftholnonholsep} becomes, 
\begin{align}
    & \int_{- \bar{\tau}}^{\infty}  \frac{\overline{\theta_{p,a}^{1}(-\bar{z})}}{ \sqrt{z+ \tau} } \dd z +  \int_{0}^{- \bar{\tau}} \frac{ 1}{\sqrt{z + \tau}}  \left(\frac{i}{z}\right)^{\frac{3}{2}} \sum_{b =1}^{p-1} \sqrt{\frac{2}{p}} \sin\left(\frac{\pi a b}{p}\right) \overline{\theta_{p,b}^{1}(1/\bar{z})}   \dd z.
\end{align}
Since $\theta_{p,a}^{1}$ is modular of weight $3/2$ with respect to $\rho_{p}$, the sum over $b$ in the second integrand equals $\overline{\theta_{p,a}^{1}(-\overline{z})}$. The two integrals then combine into a single integral over ${\mathbb R}_{>0}$:
\begin{align}
     \int_{- \bar{\tau}}^{\infty}  \frac{\overline{\theta_{p,a}^{1}(-\bar{z})}}{ \sqrt{z+ \tau} } \dd z +  \int_{0}^{- \bar{\tau}} \frac{ \overline{\theta_{p,a}^{1}(-\bar{z})}}{\sqrt{z + \tau}}  \dd z =  &\int_{0}^{\infty}  \frac{\overline{\theta_{p,a}^{1}(-\bar{z})}}{ \sqrt{z+ \tau} } \dd z = \sum_{n \in a + 2 p {\mathbb Z}}  \int_{0}^{\infty}  \frac{ n e^{ i \pi z \frac{n^{2}}{2p}    }}{ \sqrt{z+ \tau} } \dd z,
\end{align}
where in the last step we used the definition of $\theta_{p, a}^{1}$ and exchanged integration and summation.

Substituting $z= \frac{2 i  u}{ n} - \frac{u^{2}}{n^{2}\tau} $ and summing the resulting geometric series, we find
\begin{align}
    \sum_{n \in a + 2 p {\mathbb Z}}  \int_{0}^{\infty}  \frac{ n e^{ i \pi z \frac{n^{2}}{2p}    }}{ \sqrt{z+ \tau} } \dd z  = & \sqrt{-1/\tau}  \sum_{n \in a + 2 p {\mathbb Z}} 2 \int_{0}^{\infty} \mathrm{Sign}(n) e^{- \frac{i \pi u^{2}}{2 p \tau}} e^{- \frac{\pi u |n|}{p}} \dd u \endline
    = &2 \sqrt{-1/\tau}    \int_{0}^{\infty}  e^{- \frac{i \pi u^{2}}{2 p \tau}} \sum_{n \in a + 2 p {\mathbb Z}} \mathrm{Sign}(n)e^{- \frac{\pi u |n|}{p}} \dd u \endline
    = &2 \sqrt{-1/\tau}    \int_{0}^{\infty}  e^{- \frac{i \pi u^{2}}{2 p \tau}} \frac{\sinh(\pi u \tfrac{p-a}{p})}{\sinh(\pi u)} \dd u .
\end{align}
Therefore,
\begin{align} \label{psidualstransform1}
\psi_{p,a}^{\vee}(\tau) + \sqrt{\frac{i}{\tau}} \sum_{b =1}^{p-1} \sqrt{\frac{2}{p}} \sin\left(\frac{\pi a b}{p}\right)  \psi_{p,b}^{\vee}(-1/\tau)  = &  2\sqrt{\frac{i}{2 p \tau}} \int_{0}^{\infty}  e^{- \frac{i \pi u^{2}}{2 p \tau}} \frac{\sinh(\pi u \tfrac{p-a}{p})}{\sinh(\pi u)} \dd u 
\end{align}
Setting $\tau = \tfrac{1}{k}$, using that the integrand is even in $u$, and rotating the contour, this becomes
\begin{align}
    \psi_{p,a}^{\vee}(\frac{1}{k}) + \sqrt{ik} \sum_{b =1}^{p-1} \sqrt{\frac{2}{p}} \sin\left(\frac{\pi a b}{p}\right)  \psi_{p,b}^{\vee}(-k)  = &  \sqrt{\tfrac{i k}{2p}} \int_{e^{i \phi} {\mathbb R}}  e^{- 2 \pi i k \tfrac{u^{2}}{4p}} \frac{\sinh(\pi u\frac{p-a}{p})}{ \sinh(\pi u )} \dd u,
\end{align}
which is the second resurgent condition.

\textbf{Resurgent $\implies$ modular.} Now suppose $\psi_{p, a}^{\vee}$ satisfies the resurgent characterization. Let $\chi_{p,a}$ be any vector-valued mock modular form of weight $\frac{1}{2}$ with respect to $\rho_{-p}$, with shadow $\tfrac{1}{\sqrt{p}}\theta^{1}_{p,a}$ and with Fourier coefficients in $\tfrac{1}{\mathcal{N}_{a}}{\mathbb Z}$ for some integer $\mathcal{N}_{a}$; as we will see in Section \ref{dualfalseformulas}, such a function exists for all $p$, $a$. Set $f_{p,a} := \psi_{p,a}^{\vee} - \chi_{p,a}$.

By the first resurgent condition, $\psi_{p,a}^{\vee}(\tau+1) = e^{- 2 \pi i \tfrac{a^{2}}{4p}} \psi_{p,a}^{\vee}(\tau)$, and $\chi_{p,a}$ transforms in the same way, so 
$$f_{p,a}(\tau+1) = e^{- 2 \pi i \tfrac{a^{2}}{4p}}f_{p,a}(\tau).$$

By the first part of the proof, $\chi_{p, a}$ satisfies \eqref{psidualstransform1}, and by the second resurgent characterization so does $\psi_{p, a}^{\vee}$. Subtracting the two, the Borel--Mordell integrals cancel, and under $\tau \mapsto - 1/\tau$ we obtain
\begin{align} 
f_{p,a}(\tau) = - \sqrt{\frac{i}{\tau}} \sum_{b =1}^{p-1} \sqrt{\frac{2}{p}} \sin\left(\frac{\pi a b}{p}\right)  f_{p,b}(-1/\tau).
\end{align}
Since $S$ and $T$ generate $\mathrm{Mp}(2,{\mathbb Z})$, $f_{p, a}$ is a weakly holomorphic modular form of weight $\frac{1}{2}$ with respect to $\rho_{-p}$. Adding a weakly holomorphic modular form to a mock modular form gives another mock modular form with the same shadow. Hence, $\psi_{p, a}^{\vee} = \chi_{p, a}+ f_{p, a}$ is a vector-valued mock modular form of weight $\frac{1}{2}$ with respect to $\rho_{-p}$, with shadow $\tfrac{1}{\sqrt{p}}\theta^{1}_{p, a}$, and its coefficients are in $\frac{1}{\mathcal{N}_{a}}{\mathbb Z}$ by the first resurgent condition.
<!-- end proof-1 -->
<!-- theorem-2: body_md -->
\label{arthfamthm}
$ \psi^{\vee, \eta}_{p,a}$ is a vector-valued mock modular form of weight $\frac{1}{2}$ with respect to $\rho_{-p}$, with shadow $-\frac{2}{\sqrt{p}}\theta^{1}_{p,a}$.
<!-- end theorem-2 -->
<!-- proof-2: body_md -->
We first compute the shadow of $\psi^{\vee, \eta}_{p,a}$ and then show that the modular completion of $\psi^{\vee, \eta}_{p,a}$ is modular of weight $\frac{1}{2}$ with respect to $\rho_{-p}$.

Note that $B(c_{i}, c_{i}) = -2p$ for $i \in \{1,2\}$. The sublattice $\{ \lambda \in {\mathbb Z}^{2} \vert B(c_1, \lambda) =0 \}$ is generated by $c_{1, \perp}=(0,1)$, and $\{ \lambda \in {\mathbb Z}^{2} \vert B(c_2, \lambda) =0 \}$ is generated by $c_{2, \perp} = (3v,u)$. Since $u^{2}- 6 p v^{2}=1$, the quotient ${\mathbb Z}^{2}/ (c_{i}{\mathbb Z} \oplus c_{i, \perp}{\mathbb Z})$ is trivial for both $i =1 $ and $i =2$, which makes the shadow computation particularly simple.

\noindent We denote by $\widehat{\psi}^{\vee, \eta}_{p,a}$ the modular completion of $\psi^{\vee, \eta}_{p,a}$. It is given by 
\begin{align}
   \widehat{\psi}^{\vee, \eta}_{p,a}(\tau, \bar{\tau}) = &  \psi^{\vee, \eta}_{p,a}(\tau) + G_{p,a,c_{1}}(\tau, \bar{\tau} ) -G_{p,a,c_{2}}(\tau, \bar{\tau}),
\end{align}
where 
\begin{align}
    G_{p,a,c_{1}}(\tau, \bar{\tau} ) = \frac{-i  e^{-\frac{i \pi }{6}} }{2\eta(\tau) \sqrt{2p} } \sum_{s \in \{0,2\}} e^{\frac{i \pi s}{6}} \sum_{\ell \in \frac{2 s +1}{6} + {\mathbb Z}  } e^{i \pi \ell } q^{\frac{3 \ell^{2}}{2}}   \int_{- \bar{\tau}}^{\infty }  \frac{\theta_{p,a}^{1}(y)}{\sqrt{-i(y + \tau)}} \dd y, 
\end{align}
and 
\begin{align}
     G_{p,a,c_{2}}(\tau, \bar{\tau} ) = &  \frac{-i  e^{-\frac{i \pi }{6}} }{2\eta(\tau) \sqrt{2p} } \sum_{s \in \{0,2\}} e^{\frac{i \pi s}{6}} \sum_{\ell \in \frac{2 s +1}{6}u + {\mathbb Z}  } e^{i \pi \ell u} q^{\frac{3}{2}\ell^{2}}  e^{ - \pi i v ( a u - (2 s +1)p v )} \int_{- \bar{\tau}}^{\infty }  \frac{\theta_{p,a u -(2s+1)p v}^{1}(y)}{\sqrt{-i(y + \tau)}} \dd y. 
 \end{align}
Using that $v$ is even and $\eta(\tau) = \sum_{n \in {\mathbb Z}} e^{i \pi n} q^{\frac{3}{2} (n - \frac{1}{6})^{2} }$, these simplify to 
\begin{align}
    G_{p,a,c_{1}}(\tau, \bar{\tau} ) =& \frac{-i   }{ \sqrt{2p} } \int_{- \bar{\tau}}^{\infty }  \frac{\theta_{p,a}^{1}(y)}{\sqrt{-i(y + \tau)}} \dd y, \endline
         G_{p,a,c_{2}}(\tau, \bar{\tau} ) = &  \frac{-i   }{ \sqrt{2p} }\left(\frac{12}{u}\right) \int_{- \bar{\tau}}^{\infty }  \frac{\theta_{p,a u }^{1}(y)}{\sqrt{-i(y + \tau)}} \dd y. 
\end{align}
Since $\left(\frac{12}{u}\right) u = -1 \mod 2p$ and $\theta_{p,a}^{1}=- \theta_{p,-a}^{1} $, we obtain 
\begin{equation} 
G_{p,a,c_{1}}(\tau, \bar{\tau}) - G_{p,a,c_{2}}(\tau, \bar{\tau}) =   \sqrt{\frac{-2i}{p}} \int_{- \bar{\tau}}^{\infty }  \frac{\theta_{p,a}^{1}(y)}{\sqrt{(y + \tau)}} \dd y.
\end{equation} 
Noting that $\overline{\theta_{p,a}^{1}(- \overline{y})} = \theta_{p,a}^{1}(y)$ since $\theta_{p,a}^{1}(y)$ has real coefficients, we see that $\widehat{\psi}^{\vee, \eta}_{p,a}$ has shadow $-\frac{2}{\sqrt{p}}\theta^{1}_{p,a}$.

The $T$ transformation of $\widehat{\psi}^{\vee, \eta}_{p,a}$ follows from those of Zwegers' theta function and the Dedekind eta function:
$$\widehat{\psi}^{\vee, \eta}_{p,a}(\tau +1, \bar{\tau} +1) = e^{- 2 \pi i\frac{a^{2}}{4p}} \widehat{\psi}^{\vee, \eta}_{p,a}(\tau, \bar{\tau}).$$
 
For the $S$ transformation, we can combine the transformation of the modular completion of the indefinite theta functions (see Appendix A of \cite{Harichurn:2025suf}, and Corollary 2.9 of \cite{zwegers2008mock}) with that of the Dedekind eta function:
\begin{align}
   & \widehat{\psi}^{\vee, \eta}_{p,a}(\tau, \bar{\tau}) \endline
   = &\frac{e^{- \frac{i \pi }{6}}}{2\eta(\tau)} \sum_{s \in \{0,2\}} e^{\frac{i \pi  s}{6}} \frac{e^{\frac{i \pi (2s +1)}{6}} (-\tau)^{-1}}{\sqrt{6p}} \sum_{ b =1}^{2p} \sum_{r = 0}^{2} e^{\frac{i \pi a b}{p} } e^{-\frac{i \pi (2s +1)(2r+1)}{6}} e^{- \frac{i \pi (2r+1)}{6}} \widehat{\theta}_{A,(\frac{b}{2p}, \frac{2r+1}{6}), \beta, c_{1}, c_{2}} (\tfrac{-1}{\tau}, \tfrac{-1}{\bar{\tau}}) \endline
   = &\frac{e^{- \frac{i \pi }{6}}}{2\eta(\tfrac{-1}{\tau})}  \frac{\sqrt{-i \tau}(-\tau)^{-1}}{\sqrt{2p}} \sum_{ b =1}^{2p} \sum_{r \in \{0,2\}} e^{\frac{i \pi a b}{p} } e^{\frac{i \pi r}{6}} \widehat{\theta}_{A,(\frac{b}{2p}, \frac{2r+1}{6}), \beta, c_{1}, c_{2}} (\tfrac{-1}{\tau}, \tfrac{-1}{\bar{\tau}})\endline
   = & \frac{ 1}{\sqrt{-i \tau}} \sum_{ b =1}^{2p} \sqrt{\frac{2}{p}} \frac{i e^{\frac{i \pi a b}{p} }}{2} \widehat{\psi}_{p,b}^{\vee, \eta}(\tfrac{-1}{\tau}, \tfrac{-1}{\bar{\tau}}),
\end{align}
where $\widehat{\theta}_{A,(\frac{b}{2p}, \frac{2r+1}{6}), \beta, c_{1}, c_{2}}$ is the modular completion of $\theta_{A,(\frac{b}{2p}, \frac{2r+1}{6}), \beta, c_{1}, c_{2}}$. By Corollary 2.9 from \cite{zwegers2008mock}, $\psi_{p,a}^{\vee, \eta} =- \psi_{p,-a}^{\vee, \eta}  $, and hence,
\begin{equation} 
\widehat{\psi}^{\vee, \eta}_{p,a}(\tau, \bar{\tau})  = -(-i \tau)^{-\frac{1}{2}} \sum_{b =1}^{p-1} \sqrt{\frac{2}{p}} \sin(\frac{\pi a b }{p}) \widehat{\psi}_{p,b}^{\vee, \eta}(\tfrac{-1}{\tau}, \tfrac{-1}{\bar{\tau}}).
\end{equation} 
Since $S$ and $T$ generate $\mathrm{Mp}(2,{\mathbb Z})$, $\widehat{\psi}^{\vee, \eta}_{p,a}$ is a harmonic Maass form of weight $\frac{1}{2}$ with respect to $\rho_{-p}$. Hence $ \psi^{\vee, \eta}_{p,a}$ is a vector-valued mock modular form of weight $\frac{1}{2}$ with respect to $\rho_{-p}$, with shadow $-\frac{2}{\sqrt{p}}\theta^{1}_{p,a}$.
