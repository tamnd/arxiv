<!-- definition-1: body_md -->
Let  $\tau\in{\mathcal  T}$, $\zeta$ be a ${\mathcal  G}_{\tau}$-measurable  random variable,
$V$ be a finite-variation ${\mathbb G}$-adapted c\`adl\`ag process with $V_0=0$,
and $X$ be a ${\mathbb G}$-adapted c\`adl\`ag process such that $\mathbb P(X_t\in D,\, t\in [0,\tau))=1$.
We say that  a pair $(Y,M)$ of $\mathbb G$-adapted c\`adl\`ag processes is a solution of the backward
stochastic differential equation with terminal condition $\zeta$  at
terminal time $\tau$  and coefficient $f+dV$
(BSDE$^\tau(\zeta,f+dV,X,{\mathbb G},{\mathbb P})$ for short) if
\begin{enumerate}
\item[(a)]$Y$ is of class D  and  $M$ is a local martingale with $M_0=0$,
\item[(b)]  ${\mathbb P}\left( \int_0^\tau|f(X_t,Y_t)|\,dt<\infty\right)=1$ 
and
\[
Y_t=\zeta+\int^{\tau}_{t}f(X_s,Y_s)\,ds+(V_\tau-V_t)
-\int^{\tau}_{t}dM_s,\quad t\in[0,\tau],\quad
{\mathbb P}\mbox{-a.s.}
\]
\end{enumerate}
<!-- end definition-1 -->
<!-- remark-1: body_md -->
It is known (see, e.g., \cite{KR:NoD}) that if a pair $(Y,M)$ is a solution of 
 BSDE$^\tau(\zeta,f+dV,X,{\mathbb G},\mathbb P)$ such that  
\begin{equation}
\label{eq2.18}
{\mathbb E}\int^{\tau}_0|f(X_s,Y_s)|\,ds<\infty,
\end{equation} 
and $\mathbb E|V|_\tau<\infty$,   then $M$ is a uniformly integrable martingale. We write \(|V|_t\) for the total variation of \(V\) on \([0,t]\).
<!-- end remark-1 -->
<!-- theorem-1: body_md -->
\label{th2.2}
Let  $\tau\in\mathcal T$, $\zeta$ be an ${\mathcal  G}_{\tau}$-measurable  random variable,
$V$ be a finite-variation $\mathbb G$-adapted c\`adl\`ag process with $V_0=0$
and $f:D\times{\mathbb R}\to{\mathbb R}$ be a measurable function 
satisfying \mbox{\rm(H3)}. Then 
\begin{enumerate}[\rm(i)]
\item There exists at most one solution to $\operatorname{BSDE}^\tau(\zeta,f+dV,X,{\mathbb G},{\mathbb P})$.

\item If additionally  $f(x,\cdot)$ is continuous for every $x\in D$, ${\mathbb E}\int^{\tau}_0|f(X_t,y)|\,dt<\infty$
for every $y\in{\mathbb R}$ and ${\mathbb E}|\zeta|+\mathbb E|V|_\tau<\infty$, then there exists a  solution $(Y,M)$ 
of $\operatorname{BSDE}^\tau(\zeta,f+dV,X,{\mathbb G},{\mathbb P})$. 
Moreover, $M$ is a uniformly integrable martingale and
\[
{\mathbb E}\int^{\tau}_0|f(X_t,Y_t)|\,dt\le \mathbb E|\zeta|+2{\mathbb E}\int^{\tau}_0|f(X_t,0)|\,dt
+{\mathbb E}|V|_\tau.
\]
\end{enumerate}
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
The desired result follows from
\cite[Propositions 2.4 and 2.8]{K:SPA} with $S=0$ (or   
\cite[Theorem 3.7]{KR:CM} in the case $\zeta=0$).

<!-- proposition-1: body_md -->
\label{prop2.4}
Let  $\tau\in{\mathcal  T}$, $\zeta,\zeta'$ be ${\mathcal  G}_\tau$-measurable random variables such that
${\mathbb E}|\zeta|+{\mathbb E}|\zeta'|<\infty$, $V,V'$ be  finite-variation $\mathbb G$-adapted c\`adl\`ag processes with $V_0=V'_0=0$, and $f,f':D\times{\mathbb R}\to{\mathbb R}$ 
be  measurable functions such that $f(x,\cdot),f'(x,\cdot)$ are continuous for every $x\in D$ and  $f$ satisfies \mbox{\rm(H3)}. 
Let $(Y,M)$ be a solution to $\operatorname{BSDE}^\tau(\zeta,f+dV,X,\mathbb G,{\mathbb P})$.  
Then
\begin{enumerate}[\rm(i)]
\item For every moderate convex function
$F:[0,\infty)\to[0,\infty)$ there exists a
constant $C_F>0$ depending only on $F$ such that for any solution $(Y',M')$ of  
$\operatorname{BSDE}^\tau(\zeta',f'+dV',X,\mathbb G,{\mathbb P})$,
\begin{align*}
&{\mathbb E}\Big[
F\Big(\int_0^\tau |f(X_s,Y_s)-f'(X_s,Y'_s)|\,ds\Big)\Big]
\\&\quad\quad\quad\leq
C_F\,
{\mathbb E}\Big[F\Big(
|\zeta-\zeta'|+\int_0^\tau |f(X_s,Y'_s)-f'(X_s,Y'_s)|\,ds+|V-V'|_\tau\Big)\Big].
\end{align*}

\item For any solution $(Y',M')$ of  $\operatorname{BSDE}^\tau(\zeta',f'+dV',X,{\mathbb G},{\mathbb P})$ and any $q\in (0,1)$ one has
\[
{\mathbb E}\sup_{t\le\tau}|Y_t-Y'_t|^q\le c_q \Big( \mathbb E\Big[|\zeta-\zeta'|+\int_0^\tau |f(X_s,Y'_s)-f'(X_s,Y'_s)|\,ds+|V-V'|_\tau\Big]\Big)^q.
\]

\item For any solution $(Y',M')$ of  $\operatorname{BSDE}^\tau(\zeta',f'+dV',X,\mathbb G,{\mathbb P})$ and any $p>1$ one has
\[
{\mathbb E}\sup_{t\le\tau}|Y_t-Y'_t|^p\le c_p  \mathbb E\Big(|\zeta-\zeta'|+\int_0^\tau |f(X_s,Y'_s)-f'(X_s,Y'_s)|\,ds+|V-V'|_\tau\Big)^p.
\]
\end{enumerate}
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
We assume that ${\mathbb E}(|\zeta-\zeta'|+\int_0^\tau |f(X_s,Y'_s)-f'(X_s,Y'_s)|\,ds+|V-V'|_\tau)<\infty$, 
otherwise the asserted inequalities trivially 
hold. By (H3) and the Tanaka--Meyer  formula (see, e.g., \cite[p. 216]{P}) applied  to $|Y-Y'|$  we get, after a standard localization argument, that for any ${\mathbb G}$-stopping time $\sigma$,
\begin{align*}
&|Y_{\sigma\wedge\tau}-Y'_{\sigma\wedge\tau}|\\
&\qquad\le {\mathbb E}\big( |\zeta-\zeta'|+\int_{\sigma\wedge\tau}^\tau 
\operatorname{sgn}(Y_s-Y_s')(f(X_s,Y_s)-f'(X_s,Y'_s))\,ds
+|V-V'|_\tau\big|\,{\mathcal  G}_\sigma\Big)\\
&\qquad=
{\mathbb E}\Big(|\zeta-\zeta'|-\int_{\sigma\wedge\tau}^\tau \left|f(X_s,Y_s)-f'(X_s,Y'_s)
+[f'(X_s,Y'_s)-f(X_s,Y'_s)]\right|\,ds\big|\,{\mathcal  G}_\sigma\Big)\\
&\qquad\quad-{\mathbb E}\Big(\int_{\sigma\wedge\tau}^\tau  \operatorname{sgn}(Y_s-Y_s')[f'(X_s,Y'_s)-f(X_s,Y'_s)]\,ds-|V-V'|_\tau\,\big|\, {\mathcal  G}_\sigma\Big)\\
&\qquad\le 
{\mathbb E}\Big( |\zeta-\zeta'|-\int_{\sigma\wedge\tau}^\tau \left|f(X_s,Y_s)-f'(X_s,Y'_s)\right|\,ds\,\big|\,{\mathcal  G}_\sigma\Big)\\
&\quad\qquad+2{\mathbb E} \Big(\int_{0}^\tau |f'(X_s,Y'_s)-f(X_s,Y'_s)|\,ds+|V-V'|_\tau\,\big|\, {\mathcal  G}_\sigma\Big).
\end{align*}
Hence
\begin{align}
\label{eq2.1}
&|Y_{\sigma\wedge\tau}-Y'_{\sigma\wedge\tau}|
+{\mathbb E}\Big(\int_{\sigma\wedge\tau}^\tau \left|f(X_s,Y_s)-f'(X_s,Y'_s)\right|\,ds\,\big|\, {\mathcal  G}_\sigma\Big)\nonumber\\
 &\qquad\le \mathbb E\Big(|\zeta-\zeta'|+2\int_{0}^\tau |f'(X_s,Y'_s)-f(X_s,Y'_s)|\,ds+2|V-V'|_\tau\,\big|\,{\mathcal  G}_\sigma\Big).
\end{align}
Define
\[
A_t=\int_0^{t\wedge\tau}|f(X_s,Y_s)-f'(X_s,Y'_s)|\,ds,
\qquad t\ge 0,
\]
and
\[
U=|\zeta-\zeta'|+2\int_0^\tau |f'(X_s,Y'_s)-f(X_s,Y'_s)|\,ds+2|V-V'|_\tau.
\]
Then $A$ is continuous, adapted and increasing, $A_0=0$, and
\eqref{eq2.1} implies that
\[
{\mathbb E}\left[A_\infty-A_\sigma\right]\leq\mathbb E({\mathbf{1}}_{\{\sigma<\infty\}}U).
\]
The Garsia--Neveu lemma
\cite[Lemma 1.2(b), p.~31]{LLP}
therefore gives (i).  Furthermore, 
\eqref{eq2.1}  when combined with  \cite[Lemma 6.1]{BDHPS} yields (ii), and
when combined with  Doob's $L^p$-inequality  yields (iii).

{\em Step 1.} We assume additionally that $g$ is bounded.
By  \cite[Theorem 2.7]{KR:JFA},
for any $n\geq1$  and $T>0$ there exists a unique solution $(Y^{n,T},M^{n,T})$  of  the problem 
BSDE$^{T\wedge\tau_n}(g(X^n_{T\wedge\tau_n}),f,X^n,{\mathbb G}^n,{\mathbb P})$,
and a  unique solution $(Y^{T},M^{T})$  of  the problem
BSDE$^{T\wedge\tau}(g(X_{T\wedge\tau}),f,X,{\mathbb G},{\mathbb P})$. 
Set
\[
\tilde x=(x,x_{d+1}),\qquad \tilde f(\tilde x,y)=x_{d+1}\cdot f(x,y),\quad x_{d+1}\in{\mathbb R}^+,\,x\in D,
\]
and
\[
X^{d+1}_t={\mathbf{1}}_{\{t<\tau\}},\quad \tilde X=(X,X^{d+1}),\quad X^{n,d+1}_t={\mathbf{1}}_{\{t<\tau_n\}},\quad \tilde X^n=(X^n,X^{n,d+1}).
\]
Then $\tilde f$ satisfies hypotheses \mbox{\rm(H2), (H3)}. Set
$\sigma_n= T\wedge\tau_n$ and $\sigma=T\wedge\tau$. Then
\[
 Y^{n,T,(\sigma_n)}_t=g(X^n_{T\wedge\tau_n})
+\int^{T}_{t}\tilde f(\tilde X^n_s, Y^{n,T,(\sigma_n)}_s)\,ds
-\int^{T}_{t}d M^{n,T,(\sigma_n)}_s,\quad t\in[0,T],
\]
and
\[
 Y^{T,(\sigma)}_t=g(X_{T\wedge\tau})
+\int^T_t\tilde f(\tilde X_s, Y^{T,(\sigma)}_s)\,ds-\int^T_td M^{T,(\sigma)}_s,  \quad t\in[0,T].
\]
Since $\tau_n\mathop{\rightarrow}_P \tau$, $X^{n,d+1}\mathop{\rightarrow}_P X^{d+1}$ in $D([0,T];{\mathbb R})$ for every $T>0$ such that ${\mathbb P}(\tau=T)=0$. 
Since we also assume that ${\mathbb P}(\Delta X_T=0)=1$, 
it follows from  the first part of assumption  (\ref{eq2.2}) and  \eqref{eq2.9} that
\[
\tilde X^n\mathop{\longrightarrow}_P \tilde X\quad\mbox{in }[D([0,T];{\mathbb R})]^{d+1},\qquad X^n_{T\wedge\tau_n}\mathop{\longrightarrow}_P X_{T\wedge\tau}\quad\mbox{in }{\mathbb R}^d.
\]
By \cite[Theorem 2.6]{RS2}, for $T>0$ having the properties stated above,
\begin{equation}
\label{eq2.5}
( Y^{n,T,(\sigma_n)}, M^{n,T,(\sigma_n)})\mathop{\longrightarrow}_P( Y^{T,(\sigma)}, M^{T,(\sigma)})\quad\mbox{in }D([0,T];{\mathbb R}^2).
\end{equation}
By Proposition \ref{prop2.4} and Remark \ref{rem2.6}, for every $q\in (0,1)$,
\begin{align}
\label{eq2.6}
E\sup_{t\ge 0}|Y^{T,[\sigma]}_t- Y^{[\tau]}_t|^q&\le c_q \Big({\mathbb E}\int_{T\wedge \tau}^\tau|f(X_s,0)|\,ds+{\mathbb E}({\mathbf{1}}_{\{\tau>T\}}(|g(X_{T\wedge \tau})|+|g(X_\tau)|))\Big)^q \nonumber\\
&\le c_q(\|f(\cdot,0)\|_\infty {\mathbb E}(\tau-T)^++\|g\|_\infty P(T<\tau))^q
\end{align}
and
\[
{\mathbb E}\int^{\tau}_{0}|f(X_s, Y^{[\tau]}_s)-f(X_s,Y^{T,[\sigma]}_s)|\,ds\le c (\|f(\cdot,0)\|_\infty {\mathbb E}(\tau-T)^++\|g\|_\infty {\mathbb P}(T<\tau)).
\]
From this, by using the formulas for $Y^{T,[\sigma]}$ and $Y^{[\tau]}$ (see Remark \ref{rem2.6}), we infer that for $q\in (0,1)$,
\begin{equation}
\label{eq2.8}
\lim_{T\rightarrow\infty}{\mathbb E}\sup_{t\ge 0}|M^{T,(\sigma)}_t- M^{(\tau)}_t|^q=0.
\end{equation}
Analogously,
\begin{align}
\label{eq2.10}
&{\mathbb E}\sup_{t\ge0}|Y^{n,T,[\sigma_n]}_t- Y^{n,[\tau_n]}_t|^{q}\le c_q(\|f(\cdot,0)\|_\infty {\mathbb E}(\tau_n-T)^++\|g\|_\infty {\mathbb P}(T<\tau_n))^q
\end{align}
and 
\begin{align*}
&{\mathbb E}\int^{\tau_n}_{0}
|f(X^n_s,Y^{n,[\tau_n]}_s)-f(X^n_s,Y^{n,T,[\tau_n\wedge T]}_s)|\,ds\\
&\qquad\le c(\|f(\cdot,0)\|_\infty {\mathbb E}(\tau_n-T)^++\|g\|_\infty {\mathbb P}(T<\tau_n)).
\end{align*}
Furthermore, by  (\ref{eq2.9}),  
\begin{equation} 
\label{eq2.12}
\lim_{T\rightarrow\infty}\limsup_{n\rightarrow\infty}({\mathbb P}(T<\tau_n)+{\mathbb E}(\tau_n-T)^+)=0.
\end{equation}
As a result we get
\[
\lim_{T\rightarrow\infty}\limsup_{n\rightarrow\infty}
{\mathbb E}\Big(\sup_{t\ge 0}|Y^{n,T,[\sigma_n]}_t- Y^{n,[\tau_n]}_t|^q+
\int^{\tau_n}_{0}|f(X^n_s,Y^{n,[\tau_n]}_s)-f(X^n_s,Y^{n,T,[\sigma_n]}_s)|\,ds\Big)=0.
\]
From this, by using the formulas for $Y^{n,T,[\sigma_n]}$ and $Y^{n,[\tau_n]}$ (see Remark \ref{rem2.6}), we deduce that
\[
\lim_{T\rightarrow\infty}\limsup_{n\rightarrow\infty}
{\mathbb E}\sup_{t\ge 0}|M^{n,T,(\sigma_n)}_t- M^{n,(\tau_n)}_t|^{q}=0.
\]
Observe that \eqref{eq2.6} and \eqref{eq2.10} hold with $[\sigma],[\tau]$ and $[\sigma_n],[\tau_n]$ replaced
by $(\sigma),(\tau)$ and $(\sigma_n),(\tau_n)$, respectively, with  possibly different constants. 
Consequently, these two estimates combined with   (\ref{eq2.5}), \eqref{eq2.8} and the above equality give the desired convergence.

g_k(x):= (g(x)\wedge k)\vee (-k),\quad x\in {\mathbb R}^d.

{\mathbb E}\sup_{t\ge 0}| Y^{[\tau]}_t- Y^{k,[\tau]}_t|^q \le c_q\big({\mathbb E}({\mathbf{1}}_{\{|g(X_{\tau})|>k\}}|g(X_{\tau})|)\big)^q.

\[
{\mathbb E}\sup_{t\ge 0}| Y^{n,[\tau_n]}_t- Y^{n,k,[\tau_n]}_t|^q
+{\mathbb E}\sup_{t\ge 0}| Y^{[\tau]}_t- Y^{k,[\tau]}_t|^q \le \varepsilon,\quad n\ge 1.

From this and the previous step of the proof one easily concludes the result.

\label{th4.2}
Let $g:D^c\rightarrow{\mathbb R}$  and $f:D\times{\mathbb R}\to{\mathbb R}$ be  measurable functions.
\begin{enumerate}
\item[\rm(i)]  If $u$ is a probabilistic solution to \eqref{eq1.1} with exceptional set $N$, then for every $x\in D\setminus N$ the pair  $(Y,L^x)$, where
\[
Y_t=u(X_{t\wedge\tau_D}),\quad L^x_t= {\mathbb E}_x\Big(g(X_{\tau_D})+\int_0^{\tau_D}f(X_s,u(X_s))\,ds\,\big|\,{\mathcal  F}_t\Big)-u(x),\quad  t\ge 0,
\]
is a solution of  $\operatorname{BSDE}^{\tau_D}(g(X_{\tau_D}),f,X,\mathbb F,\mathbb P_x)$, and  the pair $(Y^x_t,M^x_t)$, where
\[
Y^x_t=u(X^x_{t\wedge\tau^x_D}),\quad M^x_t= {\mathbb E}\Big(g(X^x_{\tau^x_D})
+\int_0^{\tau^x_D}f(X^x_s,u(X^x_s))\,ds\,\big|\,{\mathcal  G}_t\Big)-u(x),\quad t\ge 0,
\]
 is a solution of   $\operatorname{BSDE}^{\tau^x_D}(g(X^x_{\tau^x_D}),f,X^x,{\mathbb G},{\mathbb P})$.

\item[\rm(ii)]
If {\rm(H3)} holds, then the probabilistic solution of
\eqref{eq1.1} is unique up to an $m$-inessential set.

\item[\rm(iii)] If \mbox{\rm (H2), (H3)} hold and $\int_{D^c} |g(z)|\,P_D(x,dz)<\infty$   for a.e. $x\in D$, then there exists a probabilistic solution of \eqref{eq1.1}.
\end{enumerate}

The first assertion in (i)  easily follows from the Markov property of $(X_t)$.
The second one follows then from the fact that $Y$ and $Y^x$ have the same law. Indeed, we have
\[
u(X_{t\wedge\tau_D})={\mathbb E}_x\Big(g(X_{\tau_D})+\int_{t\wedge\tau_D}^{\tau_D} f(X_s,u(X_s))\,ds\,\big|\,{\mathcal  F}_t\Big).
\]
Hence,  for any Borel sets $B_1,\dots, B_n$ and $0\le t_1,\dots,t_n\le t$, 
\begin{align*}
&{\mathbb E}_x({\mathbf{1}}_{\{X_{t_1}\in B_1,\dots,X_{t_n}\in B_n\}}u(X_{t\wedge\tau_D}))\\
&\qquad=
{\mathbb E}_x\Big({\mathbf{1}}_{\{X_{t_1}\in B_1,\dots, X_{t_n}\in B_n\}}(g(X_{\tau_D})+\int_{t\wedge\tau_D}^{\tau_D} f(X_s,u(X_s))\,ds)\Big).
\end{align*}
Since the law of $X$ under ${\mathbb P}_x$ equals the law of $X^x$ under ${\mathbb P}$, we deduce from the above equation that
\begin{align*}
&{\mathbb E}({\mathbf{1}}_{\{X^x_{t_1}\in B_1,\dots, X^x_{t_n}\in B_n\} }u(X^x_{t\wedge\tau_D^x}))\\
&\qquad={\mathbb E}\Big({\mathbf{1}}_{\{X^x_{t_1}\in B_1,\dots, X^x_{t_n}\in B_n\}}(g(X^x_{\tau^x_D})
+\int_{t\wedge \tau^x_D}^{\tau^x_D} f(X^x_s,u(X^x_s))\,ds)\Big).
\end{align*}
From this and the fact that ${\mathbb G}$ is the completion of the natural filtration generated by $(X^0_t)$ one easily obtains
\[
u(X^x_{t\wedge \tau^x_D})={\mathbb E}\Big(g(X^x_{\tau^x_D})+\int_{t\wedge \tau^x_D}^{\tau^x_D} f(X^x_s,u(X^x_s))\,ds\,\big|\,{\mathcal  G}_t\Big),
\]
which in turn implies the second assertion of (i). Assertion (ii) follows from Theorem \ref{th2.2}.
As for (iii), first note that, by Remark \ref{rem3.2} and the monotonicity of $f$, 
there exists an $m$-inessential set $N\subset \mathbb R^d$ such that
$\int_{D^c} |g(z)|\,P_D(x,dz)<\infty$ for $x\in D\setminus N$.
Then, by Theorem \ref{th2.2},  for any $x\in D\setminus N$ there exists a unique solution 
$(Y^x,M^x)$ to $\operatorname{BSDE}^{\tau_D}(g(X_{\tau_D}),f,X,\mathbb F,{\mathbb P}_x)$.
By \cite[Proposition 3.5]{KR:JFA}, $Y^x_t=u(X_t),\, t\in [0,\tau_D]$,  ${\mathbb P}_x$-a.s. for 
$x\in D\setminus N$, where  $u(x):={\mathbb E}_x Y^x_0,\, x\in D\setminus N$. This completes the proof of (iii).

 \|v\|_{L^2(D)}^2 \leq C\,\Psi(v,v),\quad v\in H^\psi_0(D).
\]
To see this, consider a sequence  $(u_n)$ bounded in \(H_0^\psi(D)\). Since
\(u_n=0\) q.e. on \(D^c\), in particular \(u_n=0\) a.e. on \(D^c\). As \(D\) is
bounded, the sequence is tight in $L^2(\mathbb R^d)$. It remains to check the
uniform continuity with respect to translations. By Plancherel's theorem,
\[
 \|u_n(\cdot+h)-u_n\|_{L^2(\mathbb R^d)}^2
 =
\int_{{\mathbb R}^d}|e^{ih\cdot \xi}-1|^2 |\widehat u_n(\xi)|^2\,d\xi,
\]
and consequently,
\begin{align*}
 \|u_n(\cdot+h)-u_n\|_{L^2({\mathbb R}^d)}^2
 &\leq \int_{|\xi|\le R}
 |e^{ih\cdot \xi}-1|^2 |\widehat u_n(\xi)|^2\,d\xi\\
 &\quad+\frac{4}{\inf_{|\xi|>R}|\psi(\xi)|}
 \int_{\mathbb R^d} |\psi(\xi)| |\widehat u_n(\xi)|^2\,d\xi.
\end{align*}
Since \(|\psi(\xi)|\to\infty\) as $|\xi|\rightarrow\infty$, the second term on the  right-hand side of the above inequality is uniformly small in \(n\) for \(R\) sufficiently large,
while the first term tends to zero uniformly in \(n\) as \(h\to0\). Hence, by
the Fr\'echet--Kolmogorov compactness criterion, \((u_n)\) is relatively compact
in \(L^2({\mathbb R}^d)\), and therefore also in \(L^2(D)\). Now,  a standard contradiction argument gives the asserted Poincar\'e inequality.

<!-- proposition-4: body_md -->
If $u\in H^\psi$ is a weak solution of \eqref{eq1.1}, then for any quasi-continuous Borel measurable  $m$-version $g_0$ of $g$
there exists a Borel version $\tilde u$ of $u$ which is a probabilistic solution of \eqref{eq1.1} with $g=g_0$.
<!-- end proposition-4 -->
<!-- proof-7: body_md -->
If $g\in H^{\psi}({\mathbb R}^d)$, then $H^\alpha_Dg\in H^{\psi}({\mathbb R}^d)$ and 
$\Psi_\alpha(H^\alpha_Dg,v)=0$ for $v\in H^{\psi}_0(D)$.  
Hence, for every $v\in H^{\psi}_0(D)$,
\[
\Psi_\alpha(u-H^\alpha_Dg,v)=\int_Df(x,u(x))v(x)\,dx+\alpha\int_D u(x)v(x)\,dx.
\]
Thus $u-H^\alpha_Dg=G^D_\alpha f(\cdot,u)+\alpha G^D_\alpha u$. Let $u_0$ be a Borel measurable and quasi-continuous 
function that belongs to the $m$-equivalence class of $u$ (see \cite[Theorem 3.5.7]{Oshima}). We may assume that $g$ is quasi-continuous
and Borel measurable.
Then, by the comments preceding the proposition,   
\[
u_0(x)={\mathbb E}_xe^{-\alpha\tau_D}g(X_{\tau_D})+{\mathbb E}_x\int_0^{\tau_D} e^{-\alpha s} f(X_s,u_0(X_s))\,ds+\alpha {\mathbb E}_x\int_0^{\tau_D}  e^{-\alpha s} u_0(X_s)\,ds
\] 
for a.e. $x\in D$. Letting $\alpha\rightarrow0$ we get that 
\begin{equation}
\label{eq4.3}
u_0(x)={\mathbb E}_xg(X_{\tau_D})+{\mathbb E}_x\int_0^{\tau_D}f(X_s,u_0(X_s))\,ds
\end{equation}
for a.e. $x\in D$ provided we show that the set 
\[
N_0:=D\setminus \Big\{x\in D:{\mathbb E}_x\Big(|g(X_{\tau_D})|+\int_0^{\tau_D}|f(X_s,u_0(X_s))|\,ds
+\int_0^{\tau_D}|u_0(X_s)|\,ds\Big)<\infty \Big\}
\]
is of Lebesgue measure zero. By the Poincar\'e inequality (see Remark \ref{rem4.3}),
\begin{align*}
&\Big\|{\mathbb E}_\cdot \int_0^{\tau_D} e^{-\alpha s} |f(X_s,u_0(X_s))|\,ds+{\mathbb E}_\cdot\int_0^{\tau_D}  e^{-\alpha s} |u_0(X_s)|\,ds\Big\|_{L^2(D)}\\
&\qquad\le c(\|f(\cdot,u_0)\|_{L^2(D)}+\|u_0\|_{L^2(D)}).
\end{align*}
On the other hand, by transience of $\Psi$ (see \cite[Theorem 3.5.33]{J2} and \cite[Lemma 2.4.1]{Oshima}),
for some strictly positive function $\rho$ we have
\[
\int_{{\mathbb R}^d}{\mathbb E}_x e^{-\alpha\tau_D}|g(X_{\tau_D})| \rho(x)\,dx
\le \sqrt{\Psi_\alpha(H^\alpha_D|g|,H^\alpha_D|g|)}\le c\sqrt{\Psi_\alpha(|g|,|g|)}\le c\sqrt{\Psi_\alpha(g,g)}.
\]
Consequently, by Fatou's lemma, $m(N_0)=0$. Now, by Remark \ref{rem3.2} and \cite[Theorem 3.2.9, Theorem 3.5.6, Corollary 3.5.3]{Oshima}, 
we infer that \eqref{eq4.3}
holds q.e. in $D$. Since $H^\alpha_Dg=g$ q.e. in $D^c$, we obtain that  $u=g$ q.e. in $D^c$.
Finally, by Proposition \ref{prop3.1} we get the result.

<!-- lemma-1: body_md -->
\label{lem4.7}
Let $U$ be an open subset of ${\mathbb R}^d$ and $u,h\in C_b({\mathbb R}^d)$. Assume that $u$ is a viscosity solution of
\[
-Lu=h\quad\text{in }U.
\]
For $\varepsilon>0$ put  $u_\varepsilon=u*\rho_\varepsilon$, $h_\varepsilon=h*\rho_\varepsilon$ and define
\[
U_\varepsilon=
\{x\in U:\operatorname{dist}(x,U^c)>\varepsilon\}.
\]
Then
\[
-Lu_\varepsilon=h_\varepsilon
\quad\text{pointwise in }U_\varepsilon.
\]
<!-- end lemma-1 -->
<!-- proof-8: body_md -->
For $y\in B(0,\varepsilon)$ define
$u^y(x):=u(x-y)$, $h^y(x):=h(x-y)$.
Since $L$ is translation invariant, $u^y$ is a viscosity solution of
\[
-Lu^y=h^y
\]
in the translated domain $U+y$. In particular, $u^y$ is a viscosity
solution of this equation in $U_\varepsilon$.
Since the  equation is linear,  
finite convex combinations of viscosity solutions are viscosity solutions
(by the nonlocal Jensen–Ishii lemma, see \cite[Lemma 1]{BarlesImbert2008}).  Approximating the
integral defining $h_\varepsilon$ by finite convex combinations of the
functions $h^y$ and using the stability of viscosity inequalities under
locally uniform convergence (see \cite[Theorem~1]{BarlesImbert2008}), we conclude that $u_\varepsilon$ is a
viscosity solution of
\[
-Lu_\varepsilon=h_\varepsilon
\quad\text{in }U_\varepsilon.
\]
From this one readily obtains the desired result.

<!-- theorem-4: body_md -->
Let $h,g\in C_b({\mathbb R}^d)$ and $u\in C_b({\mathbb R}^d)$ be a viscosity solution of \eqref{eq4.6}.
Then 
\[
u(x)={\mathbb E}_x\Big\{g(X_{\tau_D})+\int_0^{\tau_D}h(X_s)\,ds\Big\},
\quad x\in D.
\]
<!-- end theorem-4 -->
<!-- proof-9: body_md -->
For $n\ge 1$ let
\[
D_n=\{x\in D:\operatorname{dist}(x,D^c)>2/n\}.
\]
Then $D_n\subset\subset D$ and  $D_n\uparrow D$. 
Fix $x\in D_n$ and $0<\varepsilon<1/n$. By Lemma~\ref{lem4.7},
\[
-Lu_\varepsilon=h_\varepsilon\quad \text{pointwise in }D_\varepsilon.
\]
Since $D_n\subset D_\varepsilon$, Dynkin's formula (see \cite[Theorem~3.9.4, pp.~145--146]{Kolokoltsov2011}) yields
\[
{\mathbb E}_xu_\varepsilon(X_{\tau_{D_n}})-u_\varepsilon(x)=
{\mathbb E}_x\int_0^{\tau_{D_n}}Lu_\varepsilon(X_s)\,ds=
-\mathbb E_x
\int_0^{\tau_{D_n}}h_\varepsilon(X_s)\,ds.
\]
Letting $\varepsilon\downarrow0$ and using the dominated convergence theorem we get
\begin{equation}
\label{eq4.9}
u(x)={\mathbb E}_x\Big\{u(X_{\tau_{D_n}})+\int_0^{\tau_{D_n}}h(X_s)\,ds\Big\}.
\end{equation}
Since $\tau_{D_n}\uparrow\tau_D$, it follows from the quasi-left-continuity of $X$ (see \cite[Chapter~I, Proposition~7, p.~21]{Bertoin1996}) that
\[
X_{\tau_{D_n}}\rightarrow X_{\tau_D}
\quad\text{almost surely}.
\]
Letting $n\to\infty$ in \eqref{eq4.9} and using
dominated convergence once again we obtain the desired representation.

<!-- theorem-5: body_md -->
\label{th3.4}
Let $u$ be a probabilistic solution of \eqref{eq1.1} with exceptional set $N$ and for $n\ge1$ let $u_n$ be a probabilistic solution of 
\eqref{eq1.1} with $L$ replaced by $L^n$ with exceptional set $N_n$. Set $N_\infty=\bigcup_{n} N_n$. Suppose that \mbox{\rm(H1)} is satisfied and
\begin{enumerate}[\rm(i)]
\item condition    \mbox{\rm(\ref{eq5.1})} is satisfied for every $x\in D\setminus N$;
\item $f,g$ satisfy  \mbox{\rm(H2)--(H3)};
\item  \mbox{\rm(H4$_x$)} holds for $x\in D\setminus N$, \mbox{\rm (H5$_x$)} holds for $x\in D\setminus N_\infty$;
\item there exists $\varphi\in {\mathcal  V}$ such that
\[
\sup_{n\ge1} \int_{D^c}\varphi(|g(y)|)P^n_D(x,dy)<\infty,\quad x\in D\setminus N_\infty.
\]
\end{enumerate}
Then $u_n(x)\rightarrow u(x)$ for every $x\in D\setminus (N\cup N_\infty)$.

<!-- proof-10: body_md -->
Fix $x\in D\setminus (N\cup N_\infty)$.  By Theorem \ref{th4.2},  it suffices to show that
\begin{equation}
\label{eq5.3}
Y^{x,n}_0\mathop{\longrightarrow}_d Y^{x}_0\quad\mbox{in }{\mathbb R},
\end{equation}
where $Y^{x}$ (resp. $Y^{x,n}$) is the first component of the solution of BSDE${}^{\tau^x_D}(\zeta^x,f,X^x,{\mathbb G},{\mathbb P})$ with  $\zeta^x=g(X^x_{\tau^x_D})$ 
(resp. BSDE${}^{\tau^{x,n}_D}(\zeta^{x,n},f,X^{x,n},{\mathbb G}^n,{\mathbb P})$ with $\zeta^{x,n}=g(X^{x,n}_{\tau^{x,n}_D})$).
To show (\ref{eq5.3}) we will use Theorem \ref{th2.4} with $X=X^x$,   $X^n=X^{x,n}$, $\tau=\tau^x_D$,

(see Remark \ref{rem2.8}). Hypothesis (H1) implies that $\lim_{n\rightarrow\infty}
\sup_{s\le t\le q} |e^{-(t-s)\psi_n(\xi)}-e^{-(t-s)\psi(\xi)}|=0$ for all $\xi\in{\mathbb R}^d,q>0$,

Since the processes $X^{x,n}$ have independent increments, \cite[Proposition 2]{CMS} implies that in fact  condition (\ref{eq2.2})

Furthermore, by  Propositions \ref{prop6.1}, \ref{prop4.3} again, also the fourth condition in (\ref{eq2.9}) is satisfied. 
Next, \eqref{eq2.15} holds thanks to (iv) while  (a), (b) of Theorem \ref{th2.4} hold thanks to (iii).
Consequently, the assertion follows from  Theorem \ref{th2.4}.

<!-- remark-8: body_md -->
\label{rem5.2}
There is a qualitative difference between condition \textup{(i)} and conditions
\textup{(ii)--(iv)} of Theorem \ref{th3.4}. The latter concern only the
regularity, monotonicity, growth, and integrability properties of the data
$f$ and $g$ and are usually straightforward to verify. For example, if
\textup{(H2)}, \textup{(H3)} hold and
\[
 g \in C_b({\mathbb R}^d),
\]
then conditions \textup{(ii)--(iv)} are satisfied. By contrast, condition \textup{(i)} is of a genuinely different nature. It
concerns the interplay between the geometry of the boundary $\partial D$ and
the path behaviour of the limiting process $X$, in particular the way in
which $X$ may approach, hit or jump across the boundary.

<!-- theorem-6: body_md -->
\label{th3.10}
Let $p\ge 1$. Let $u$ be a probabilistic solution to \eqref{eq1.1}  and $u_n$ be a probabilistic solution to 
\eqref{eq1.1} with $L$ replaced by $L^n$.                                   
Suppose that \mbox{\rm(H1)} holds and:
\begin{enumerate}[\rm(a)]
\item there exists an $m$-inessential set $N\subset{\mathbb R}^d$ such that     \mbox{\rm(\ref{eq5.1})} is satisfied for $x\in D\setminus N$;
\item $f$ satisfies  \mbox{\rm(H3)} and $g\in C({\mathbb R}^d)$;
\item $P_D[|g|^p],P_D^n[|g|^p]\in L^1(D)$ and $f(\cdot,y)\in L^1(D)$ for every $y\in{\mathbb R}$;
\item $f-f(\cdot,0)\in C(\bar D\times{\mathbb R})$, $f(\cdot,0)\in L^p(D)$;
\item there exists $\varphi\in \mathcal V$ such that
$\sup_{n\ge 1} \int_D\int_{D^c}\varphi(|g(y)|^p)P^n_D(x,dy)\,dx<\infty$.
\end{enumerate}
 Then
$u_n\rightarrow u$ in $L^p(D)$.
<!-- end theorem-6 -->
<!-- proof-12: body_md -->
We denote by $N_n$ the exceptional set for $u_n$ and by  $N_0$ the exceptional set for $u$.
Since the union of $m$-inessential sets is an $m$-inessential set,
we may assume that $N_0=N$. 
By Theorem \ref{th4.2}, for any $x\in D\setminus N$, $u(X)$ is the first component of the solution of 
$\operatorname{BSDE}^{\tau_D}(g(X_{\tau_D}),f,X,\mathbb F,{\mathbb P}_x)$,
and for any $x\in D\setminus N_n$, $u_n(X)$ is the first component of the solution of $\operatorname{BSDE}^{\tau_D}(g(X_{\tau_D}),f,X,{\mathbb F},{\mathbb P}^n_x)$.
Next, notice that   thanks to (c), we have $u,u_n\in L^p(D)$.
Indeed, by Theorem \ref{th2.2},
\[
|u(x)|\le P_D|g|(x)+\int_D|f(y,0)|\,G_D(x,dy),\quad x\in D\setminus N,
\]
and by Fubini's theorem and Proposition \ref{prop4.2},
\[
\int_D\Big(\int_D|f(y,0)|\,G_D(x,dy)\Big)\,dx\le C\|f(\cdot,0)\|_{L^1(D)},
\]
which when combined with $P_D[|g|^p]\in L^1(D)$ shows that $u\in L^1(D)$ if $p=1$.
If $p>1$, then
by It\^o's formula (see, e.g., \cite[Theorem IV.66]{P}),
\begin{equation}
\label{eq5.4}
|u(x)|^p\le {\mathbb E}_x|g(X_{\tau_D})|^p
+p{\mathbb E}_x\int^{\tau_D}_0\operatorname{sgn}(u(X_t))|u(X_t)|^{p-1}f(X_t,u(X_t))\,dt.
\end{equation}
By using (H3) and  the elementary inequality 
\begin{equation}
\label{eq5.5}
ab\le \frac{(\varepsilon a)^{p/(p-1)}}{p/(p-1)}
+\frac{(\varepsilon^{-1}b)^{p}}{p},\quad a,b,\varepsilon>0,
\end{equation}
we show that the right-most term in \eqref{eq5.4} does not exceed
\begin{align}
\label{eq5.6}
&p{\mathbb E}_x\int^{\tau_D}_0|u(X_t)|^{p-1}|f(X_t,0)|\,dt\nonumber\\
&\qquad\le  {\mathbb E}_x\int^{\tau_D}_0
\big\{(p-1)\varepsilon^{p/(p-1)}|u(X_t)|^{p}
+\varepsilon^{-p}|f(X_t,0)|^p\big\}\,dt \nonumber\\
&\qquad=\int_D\big\{(p-1)\varepsilon^{p/(p-1)}|u(y)|^{p}
+\varepsilon^{-p}|f(y,0)|^p\big\}\,G_D(x,dy).
\end{align}
Taking now $\varepsilon=(2(p-1)C)^{-(p-1)/p}$, where $C$ is the constant from Proposition \ref{prop4.2}
applied to the dual operator $L^*$ and with  $q=1$, we have  $(p-1)\varepsilon^{p/(p-1)}=1/(2C)$. Therefore from 
Proposition \ref{prop4.2} and (\ref{eq5.4}) it follows that
\[
\|u\|^p_{L^p(D)}\le 2\|P_D|g|^p\|_{L^1(D)}+Cc(p,C)\|f(\cdot,0)\|^p_{L^p(D)},\quad 
c(p,C)=2(2(p-1)C)^{p-1}.
\]
In the same way we show that the above inequality is satisfied with $u$ replaced by $u_n$. To show the convergence of $u_n$ to $u$, we 
choose a sequence  $(\gamma_k)\subset C(\bar D)$ such that  $\gamma_k\rightarrow f(\cdot,0)$ in $L^p(D)$. 
For  $k\ge1$ define $f_k:D\times{\mathbb R}\rightarrow{\mathbb R}$ and $g_k:\mathbb R^d\to\mathbb R$ by 
\[
f_k(x,y)=f(x,y)-f(x,0)+\gamma_k(x),\quad g_k(x):=(g(x)\wedge k)\vee (-k).              
\]
By (c) and Remark \ref{rem3.2}, there exist  $m$-inessential sets $N',N'_n\subset D$ (with respect to $L$ and $L^n$, respectively)
such that (H4$_x$) holds for $x\in D\setminus N'$ and $\int_{D^c} |g(z)|\,P^n_D(x,dz)<\infty$ for $x\in D\setminus N'_n$. 
Clearly, we may assume that $N=N'$ and $N'_n=N_n$.
Notice that each  function $f_k$ satisfies (H2) and (H3). We denote by $u^k$   the solution of (\ref{eq1.1}) with $(f,g)$ replaced by $(f_k,g_k)$
and with an exceptional set $N$, 
while by $u^k_n$ we denote the solution of (\ref{eq1.1}) with $(f,g,L)$ replaced by $(f_k,g_k,L^n)$ and with an exceptional set $N_{k,n}$
(the existence follows from Theorem \ref{th4.2}).
By Theorem \ref{th3.4}, for a.e. $x\in D$,
\begin{align}
\label{eq5.7}
u^k_n(x)\rightarrow u^k(x)
\end{align}
as $n\rightarrow\infty$. Since $g_k,\gamma_k$ are bounded, it follows from Proposition \ref{prop2.4} and 
Proposition \ref{prop4.2} that $u^k$ is bounded and $u^k_n$
are bounded uniformly in $n\ge1$. Therefore (\ref{eq5.7}) implies that for each $k\ge1$,
\begin{equation}
\label{eq5.8}
\lim_{n\rightarrow\infty}\|u^k_n-u^k\|_{L^p(D)}=0
\end{equation}
for $p\ge1$. 
The rest of the proof is divided into two cases: $p=1$ and $p>1$. Let $p=1$. We shall show that
\begin{equation}
\label{eq5.9}
\lim_{k\rightarrow\infty}\|u^k-u\|_{L^1(D)}=0.
\end{equation}
By Proposition \ref{prop2.4},
\begin{align}
\label{eq5.10}
|u(x)-u^k(x)|\le c\Big(P_D|g-g_k|(x)+\int_D|f(y,0)-\gamma_k(y)|\,G_D(x,dy)\Big).
\end{align}
By this and  Proposition \ref{prop4.2},  
\[
\|u-u^k\|_{L^1(D)}\le C(\|P_D|g-g_k|\|_{L^1(D)}+\|f(\cdot,0)-\gamma_k\|_{L^1(D)}).
\]
This proves (\ref{eq5.9}). We next show that
\begin{equation}
\label{eq5.11}
\lim_{k\rightarrow\infty}\limsup_{n\rightarrow\infty}
\|u^k_n-u_n\|_{L^1(D)}=0.
\end{equation}
In much the same way, by using  Proposition \ref{prop2.4} and Proposition \ref{prop4.2}, we get 
\begin{align}
\label{eq5.12}
|u_n(x)-u_n^k(x)|\le c\Big(P^n_D|g-g_k|(x)
+\int_D|f(y,0)-\gamma_k(y)|\,G^n_D(x,dy)\Big),
\end{align}
and hence
\begin{align*}
\|u^k_n-u_n\|_{L^1(D)} 
\le C(\|P^n_D|g-g_k|\|_{L^1(D)}+ \|f(\cdot,0)-\gamma_k\|_{L^1(D)})
\end{align*}
for all  $n$, which  combined with (v) proves  (\ref{eq5.11}). From  (\ref{eq5.8})--(\ref{eq5.11}) we get the desired convergence in case  $p=1$. 

Suppose now that  $p>1$. Then by It\^o's formula (see, e.g., \cite[Theorem IV.66]{P}),
\begin{align*}
|u(x)-u^k(x)|^p&\le {\mathbb E}_x|g(X_{\tau_D})-g_k(X_{\tau_D})|^p\\
&\quad+p{\mathbb E}_x\int^{\tau_D}_0
\mbox{sgn}(u(X_t)-u^k(X_t))
|u(X_t)-u^k(X_t)|^{p-1}\\
&\qquad\qquad\qquad\qquad\qquad
\times(f(X_t,u(X_t))-f_k(X_t,u^k(X_t)))\,dt\\
&\le {\mathbb E}_x|g(X_{\tau_D})-g_k(X_{\tau_D})|^p\\
&\quad+p{\mathbb E}_x\int^{\tau_D}_0
|u(X_t)-u^k(X_t)|^{p-1}|f(X_t,u(X_t))-f_k(X_t,u(X_t))|\,dt.
\end{align*}
By using (\ref{eq5.5})  with $\varepsilon$ defined right after (\ref{eq5.6})  we get  
\begin{align*}
&|u(x)-u^k(x)|^p\le{\mathbb E}_x|g(X_{\tau_D})-g_k(X_{\tau_D})|^p\\
&\quad\quad+  {\mathbb E}_x\int^{\tau_D}_0
\Big\{\frac{1}{2C}|u(X_t)-u^k(X_t)|^{p}
+\frac{c(p,C)}{2}|f(X_t,u(X_t))-f_k(X_t,u(X_t))|^p\Big\}\,dt\\
&\quad=P_D|g-g_k|^p(x)+\int_D\Big\{\frac{1}{2C}|u(y)-u^k(y)|^{p}
+\frac{c(p,C)}{2}|f(y,0)-\gamma_k(y)|^p\Big\}\,G_D(x,dy).
\end{align*}
Integrating both sides of the above inequality with respect to $x$ we get
\begin{equation}
\label{eq5.13}
\|u-u^k\|^{p}_{L^p(D)}\le C c(p,C)\left(\|P_D|g-g_k|^p\|_{L^1(D)}+\|f(\cdot,0)-\gamma_k\|^p_{L^p(D)}\right).
\end{equation}
In the same way we get
\begin{equation}
\label{eq5.14}

<!-- lemma-2: body_md -->
\label{lem6.3}
Let $D\subset{\mathbb R}^d$ be an open bounded set such that $m(\partial D)=0$ and
$X$ be a L\'evy process in ${\mathbb R}^d$ satisfying condition \mbox{\rm(ACP)}. Then, for every $x\in{\mathbb R}^d\setminus \partial D$,
\begin{equation}
\label{eq6.2}
{\mathbb P}_x(\exists s>0:\ X_{s-}\in {\mathbb R}^d\setminus \partial D,\ X_s\in \partial D)=0.
\end{equation}
<!-- end lemma-2 -->
<!-- proof-14: body_md -->
Set $A=\partial D$ and define  $h(y)={\mathbf{1}}_{A^c}(y)\,\nu(A-y)$, $y\in{\mathbb R}^d$,
\[
N_A=\sum_{s>0}{\mathbf{1}}_{\{X_{s-}\notin A,\ X_s\in A\}}.
\]
By using the L\'evy system (\ref{eq6.1}) for $X$ and (\ref{eq3.5}) we obtain 
\begin{equation}
\label{eq6.3}
{\mathbb E}_xN_A
={\mathbb E}_x\int_0^\infty \mathbf{1}_{\{X_t\notin A\}}\,\nu(A-X_t)\,dt
={\mathbb E}_x\int_0^\infty h(X_t)\,dt.
\end{equation}
By Tonelli's theorem,
\[
\int_{{\mathbb R}^d} h(y)\,dy=\int_{{\mathbb R}^d}
\Big(\int_{{\mathbb R}^d}{\mathbf{1}}_{A^c}(y){\mathbf{1}}_{A}(y+z)\,dy\Big)\nu(dz).
\]
For each fixed $z\in{\mathbb R}^d\setminus\{0\}$ we have
\[
\int_{{\mathbb R}^d}{\mathbf{1}}_{A^c}(y){\mathbf{1}}_{A}(y+z)\,dy\le m(A-z)=0.
\]
Hence  $h=0$ a.e. in ${\mathbb R}^d$. Since (ACP) is satisfied, it follows from this and (\ref{eq6.3}) that ${\mathbb E}_x N_A=0$, which proves \eqref{eq6.2}.

<!-- lemma-3: body_md -->
\label{lem6.4}
Let $X$ be a L\'evy process in ${\mathbb R}^d$ satisfying condition \mbox{\rm(ACP)} and 
$U$ be an  open subset of ${\mathbb R}^d$ of the  form
\[	
U=T(V)\cap B(z,r),
\]
where $T:{\mathbb R}^d\to {\mathbb R}^d$ is an isometry, $V$ is a hypograph of the form 
$V=\{(x,y): y<f(x)\}$   for  some $f\in C({\mathbb R}^{d-1})$
and  $B(z,r)=\{x\in{\mathbb R}^d:|x-z|<r\}$ for some $z\in{\mathbb R}^d$, $r>0$.
Then 
\[
{\mathbb P}(\tau^x_U=\tau^x_{\bar U})=1,\quad x\in U,
\]
where $\tau^x_U=\inf\{t>0:X^x_t\notin U\}$, 
$\tau^x_{\bar U}=\inf\{t>0:X^x_t\notin\bar U\}$.
<!-- end lemma-3 -->
<!-- proof-15: body_md -->
First observe that for any open sets $V, W$ such that $\tau^x_V=\tau^x_{\bar V}$ for $x\in V$ and $\tau^x_W=\tau^x_{\bar W}$ for $x\in W$ we have 
\begin{equation}
\label{eq6.4}
\tau^x_{V\cap W}=\tau^x_{\overline{V\cap W}},\quad x\in V\cap W.
\end{equation}
Indeed, $\overline{V\cap W}\subset \bar V\cap\bar W$, which implies that
\[
\tau^x_{V\cap W}\le \tau^x_{\overline{V\cap W}}\le
\tau^x_{\bar V\cap\bar W}=\tau^x_{\bar V}\wedge \tau^x_{\bar W}
=\tau^x_V\wedge \tau^x_W=\tau^x_{V\cap W},\quad x\in V\cap W.
\]
Since for any L\'evy process $X$ satisfying (ACP) and any isometry $T$, the process $Y=T(X)$ is again a L\'evy process satisfying (ACP), 
and since each open ball is the intersection of a finite number of rotated open hypographs of continuous functions, it follows from (\ref{eq6.4}) 
that we are reduced to proving that $\tau^x_U=\tau^x_{\bar U}$ for every $x\in U$, where $U$ is an open hypograph of a continuous function 
$f:{\mathbb R}^{d-1}\to {\mathbb R}$. Let $U$ be of this form.
Let $\vec  r=(0,\dots,0,r)\in {\mathbb R}^d$. Note that 
\begin{equation}
\label{eq6.5}
\tau^x_{U}=\tau^{x-\vec r}_{U-\vec r},\qquad \tau^x_{\bar U}
=\tau^{x-\vec r}_{\bar U-\vec r}\,,\quad x\in U.
\end{equation}
From the proof of \cite[Proposition (2.5)]{D} it follows that  there exists $r>0$ such that  $\tau^{y}_{U-\vec r}
=\tau^{y}_{\bar U-\vec r}$ for a.e. $y\in U-\vec r$.
This together with (\ref{eq6.5})  yields  
$\tau^x_{U}= \tau^x_{\bar U}$ for a.e. $x\in U$, so the functions $w,\bar w$  defined by $w(x)={\mathbb E} e^{-\tau^x_{U}}$, $\bar w(x)={\mathbb E} e^{-\tau^x_{\bar U}}$ 
are equal a.e. in $U$. The functions  $w,\bar w$ are $1$-excessive with respect to  the semigroup $(P^U_t)$  (see, e.g., \cite[Proposition II.(2.8)]{BG}).
Since (ACP)  is satisfied, in fact
$w(x)=\bar w(x)$ for every  $x\in U$ by \cite[Proposition VI.(1.3)]{BG}. Thus ${\mathbb E}(e^{-\tau^x_{U}}-e^{-\tau^x_{\bar U}})=0$ for $x\in U$, 
which proves the lemma since $\tau^x_{\bar U}\ge\tau^x_U\,$a.s.

<!-- proposition-6: body_md -->
\label{prop6.5}
Let $D$ be a bounded open subset of ${\mathbb R}^d$ of class $C^0$.
If  $X$ is a L\'evy process satisfying condition \mbox{\rm(ACP)},  then \eqref{eq5.1} is satisfied.
<!-- end proposition-6 -->
<!-- proof-16: body_md -->
We  let $\tau=\tau_{D}$ and $\bar\tau=\tau_{\bar D}$. Fix $x\in D$. Since  ${\mathbb P}_x(\tau\le \bar\tau)=1$, it suffices to prove that 
\begin{equation}
\label{eq6.6}
\mathbb P_x(\tau<\bar\tau)=0.
\end{equation}
By Lemma \ref{lem6.3},
\begin{equation}
\label{eq6.7}
{\mathbb P}_x(\exists{s>0}:X_{s-}\in D,\, X_s \in \partial D)=0.
\end{equation}
Since  $\tau$ is the first exit time from $D$, $X_\tau\notin D$. Hence $X_\tau\in\partial D$ on the set $\{\tau<\bar\tau\}$. From this,  \eqref{eq6.7} and 
Proposition \ref{prop4.3} we get 
\begin{equation}
\label{eq6.8}
{\mathbb P}_x(\tau<\bar\tau)={\mathbb P}_x(A),\quad\mbox{where}\quad A=\{\tau<\bar\tau,\,  X_{\tau-}=X_\tau,\, X_\tau\in \partial D\}.
\end{equation}
Let ${\mathcal  R}$ denote the family of sets $T(V)$, where
$T:{\mathbb R}^d\to{\mathbb R}^d$ is an isometry and
$V=\{(x',x_d):x_d<F(x')\}$ for some
$F\in C({\mathbb R}^{d-1})$.
For every $y\in \partial D$ there exist $r_y>0$ and $V_y\in{\mathcal  R}$ such that
\[
D\cap B(y,3r_y)=V_y\cap B(y,3r_y).
\]	
Since $\partial D$ is compact, there exists a  finite set  
$\{y_1,\dots,y_N\}\subset \partial D$  such that the  balls $B(y_k,r_{y_k})$ cover $\partial D$. To shorten notation, set 
\[
b_k=B(y_k,r_{y_k}),\quad B_k= B(y_k,2r_{y_k}),\quad V_k= V_{y_k},\quad  U_k= V_k\cap B(y_k,3r_{y_k}).
\]
For each $k\in\{1,\dots,N\}$  define recursively the entrance times $\sigma_{k,n}$  and exit times $\eta_{k,n}$ by 
\[
\sigma_{k,1}=\inf\{t>0:X_t\in D\cap b_k\},\qquad
\eta_{k,1}=\inf\{t>\sigma_{k,1}:X_t\notin B_k\},
\]
and, for $n\ge1$,
\[
\sigma_{k,n+1}=\inf\{t>\eta_{k,n}:X_t\in D\cap b_k\},\qquad
\eta_{k,n+1}=\inf\{t>\sigma_{k,n+1}:X_t\notin B_k\}.
\]
On the set $A$  we have $X_\tau\in \partial D$, so $X_\tau\in b_k$ for some $k\ge1$. Consequently,
\[	
A\subset \bigcup_{k=1}^N\bigcup_{n=1}^{\infty}\{X_\tau\in b_k,\,\sigma_{k,n}<\tau<\eta_{k,n}\}.
\]
Observe now that 
\[
A\cap\{X_\tau\in b_k,\ \sigma_{k,n}<\tau<\eta_{k,n}\}\subset
\{\tau_{U_k}\circ\theta_{\sigma_{k,n}}<
\tau_{\bar U_k}\circ\theta_{\sigma_{k,n}}\},
\]
and
\[
\{\sigma_{k,n}<\tau\}
\subset
\{X_{\sigma_{k,n}}\in U_k\}.
\]
Therefore, by Lemma \ref{lem6.4} and the
strong Markov property at $\sigma_{k,n}$,
\[
{\mathbb P}_x(A\cap\{X_\tau\in b_k,\,\sigma_{k,n}<\tau<\eta_{k,n}\})
\le {\mathbb E}_x\Big({\mathbf{1}}_{\{\sigma_{k,n}<\tau\}}{\mathbb P}_{X_{\sigma_{k,n}}}\left(\tau_{U_k}<\tau_{\bar U_k}\right)\Big)
=0.
\]
Summing over $k,n$ we get  ${\mathbb P}_x(A)=0$, which together with \eqref{eq6.8} proves \eqref{eq6.6}.

\label{prop7.11}
Let $({\mathcal  E},{\mathcal  F})$, $({\mathcal  E}^{(n)},{\mathcal  F}^{(n)})$, $n\ge1$, be closed quadratic forms on $H$. Then  $({\mathcal  E}^{(n)},{\mathcal  F}^{(n)})$ converges to $({\mathcal  E},{\mathcal  F})$ in the sense of 
Mosco if and only if for any $\alpha>0$ and $f\in H$, $G^{(n)}_{\alpha}f$ converges to $G_{\alpha}f$ in $H$.

<!-- theorem-7: body_md -->
\label{th7.12}
Let $D$ be an open bounded subset of ${\mathbb R}^d$  of class $C^0$.
If \mbox{\rm(H1)} is satisfied and 
$X$ is a L\'evy process satisfying condition \mbox{\rm(ACP)}, then	
the Dirichlet forms $(\Psi^{(n)}, H^{\psi_n}_0(D))$ converge to  $(\Psi, H^{\psi}_0(D))$ in the sense of Mosco.
<!-- end theorem-7 -->
<!-- proof-18: body_md -->
Fix $\alpha>0$ and let $h$ be a  Borel function in $L^2(D)$.
The functions $R^{D,n}_\alpha h$ and $R^D_\alpha h$ are the
probabilistic solutions corresponding to
$f(x,y)=h(x)-\alpha y$ and $g=0$.
Thus Corollary \ref{cor5.5}, applied with $p=2$, and the
identification of resolvents given above yield
\[
G^{D,n}_\alpha h=R^{D,n}_\alpha h
\longrightarrow R^D_\alpha h=G^D_\alpha h
\quad\text{in }L^2(D).
\]
The conclusion follows from Proposition \ref{prop7.11}.

<!-- example-10: body_md -->
\label{ex7.13}
Let $D=(-1,0)\cup(0,1)$. Let $X$ be a standard Brownian motion and for $n\ge1$ let $X^n$ be a symmetric L\'evy process in ${\mathbb R}$ with generator
\[
L^n \varphi(x)=\frac{n^2}{2}\Big(\varphi\Big(x+\frac1n\Big)-2\varphi(x)
+\varphi\Big(x-\frac1n\Big)\Big),
\]
i.e. $X^n$ is a compound Poisson process with L\'evy measure $(n^2/2)(\delta_{1/n}+\delta_{-1/n})$ (see Example \ref{ex5.7}). 
An elementary computation shows that $\psi_n(\xi)=n^2(1-\cos(\xi/n))\rightarrow|\xi|^2/2$ as $n\rightarrow\infty$. The corresponding  Dirichlet forms on $L^2({\mathbb R})$ are 
\[
{\mathcal  E}(u,u)=\frac12\int_{{\mathbb R}}|u'(x)|^2\,dx,
\quad {\mathcal  F}=H^1({\mathbb R}),
\]
and
\[
{\mathcal  E}^{(n)}(u,u)=
\frac{n^2}{2}\int_{{\mathbb R}}\Big(u\Big(x+\frac1n\Big)-u(x)\Big)^2\,dx,
\quad
{\mathcal  F}^{(n)}=L^2({\mathbb R}).
\]
The domains of parts of these forms on  $D$ are of the form  
\[
{\mathcal  F}^D=H_0^1(D)=H_0^1(-1,0)\oplus H_0^1(0,1),\qquad {\mathcal  F}^{(n),D}=L^2(D).
\]	
Define
\[
v(x)=(1-|x|)^+,\qquad v_n=v|_D \in L^2(D).
\]
Then $v_n\to v|_D$ strongly in $L^2(D)$. However, $v|_D\notin H_0^1(D)$.
Hence ${\mathcal  E}^D(v,v)=\infty$. On the other hand,
\[
\Big| v\Big(x+\frac1n\Big)- v(x)\Big|\le \frac1n
\quad\text{for }x\in{\mathbb R},\qquad  v(x)= v\Big(x+\frac1n\Big),\quad |x|\ge 1+1/n.
\]
Consequently,
\[
{\mathcal  E}^{(n),D}(v_n,v_n)=\frac{n^2}{2}\int_{{\mathbb R}}
\Big( v\Big(x+\frac1n\Big)- v(x)\Big)^2\,dx
\le
\frac{n^2}{2}\cdot \frac1{n^2}\cdot\Big(2+\frac2n\Big)
=1+\frac1n.
\]
In particular, $\sup_{n\ge1}{\mathcal  E}^{(n),D}(v_n,v_n)<\infty$. We thus have 
\[
v_n\rightharpoonup v \quad\text{in }L^2(D),
\qquad
\liminf_{n\to\infty}{\mathcal  E}^{(n),D}(v_n,v_n)<\infty,
\qquad {\mathcal  E}^D(v,v)=\infty,
\]
so the forms ${\mathcal  E}^{(n),D}$ and ${\mathcal  E}^D$ do not satisfy (M1).

<!-- proposition-8: body_md -->
\label{prop8.1}
Let $((X_t)_{t\ge 0},({\mathbb P}_x)_{x\in E})$ be a strong Markov process with state space  $E$, $D$ be a Borel subset of $E$, and $\tau=\inf\{t>0:X_t\notin D\}$.
Assume that there exist $q>0$ and $p\in(0,1)$ such that
\begin{equation}
\label{eq8.1}
\inf_{x\in D} {\mathbb P}_x(\tau\le q)\ge p.
\end{equation}
Then, for every $n\in{\mathbb N}$,
\begin{equation}
\label{eq8.2}
\sup_{x\in D}{\mathbb P}_x(\tau>n q)\le(1-p)^n.
\end{equation}
Consequently, for every $m\in{\mathbb N}$,
\begin{equation}
\label{eq8.3}
\sup_{x\in D}{\mathbb E}_x|\tau|^m\le mq^m \sum_{n=0}^\infty(n+1)^{m-1} (1-p)^n<\infty,
\end{equation}
and moreover, $\sup_{x\in D}{\mathbb E}_x[e^{\theta \tau}]<\infty$ for every
$\theta<-(1/q)\ln(1-p)$.
<!-- end proposition-8 -->
<!-- proof-19: body_md -->
By \eqref{eq8.1},
\begin{equation}
\label{eq8.4}
\sup_{x\in D}{\mathbb P}_x(\tau>q)\le 1-p.
\end{equation}
Fix $n\in{\mathbb N}$ and $x\in D$. Using the strong Markov property  we get
\[
{\mathbb P}_x(\tau>(n+1)q)={\mathbb E}_x\big({\mathbf{1}}_{\{\tau>q\}}
{\mathbb P}_x(\tau>(n+1)q|{\mathcal  F}_{q})\big)
={\mathbb E}_x\big({\mathbf{1}}_{\{\tau>q\}}\,{\mathbb P}_{X_{q}}(\tau>n q)\big).
\]
On the event $\{\tau>q\}$ we have $X_{q}\in D$. Hence
\[
{\mathbb P}_{X_{q}}(\tau>n q) \le\sup_{y\in D}{\mathbb P}_y(\tau>n q).
\]
Therefore,
\[
{\mathbb P}_x(\tau>(n+1)q)\le{\mathbb P}_x(\tau>q)\cdot \sup_{y\in D}{\mathbb P}_y(\tau>n q).
\]
From this and  \eqref{eq8.4} we obtain
\[
a_{n+1}\le(1-p)\cdot a_n,\quad\mbox{where}\quad
a_n=\sup_{x\in D}{\mathbb P}_x(\tau>nq).
\]
Since $a_1\le 1-p$ by \eqref{eq8.4}, induction yields $a_n\le(1-p)^n$, i.e.
\eqref{eq8.2} holds true.
To show (\ref{eq8.3}) we first note that  ${\mathbb P}_x(\tau>t)\le{\mathbb P}_x(\tau>nq)\le(1-p)^n$ for $t\in[nq,(n+1)q)$. Hence, for every $x\in D$,
\begin{align*}
{\mathbb E}_x|\tau|^m
= m\int_0^\infty t^{m-1}{\mathbb P}_x(\tau>t)\,dt 
&\le m\sum_{n=0}^\infty \int_{nq}^{(n+1)q} t^{m-1}(1-p)^n\,dt\\
&\le mq^m \sum_{n=0}^\infty (n+1)^{m-1}(1-p)^n <\infty,
\end{align*}
which implies (\ref{eq8.3}). Finally, for every 
$\theta\ge 0$, we have
\begin{align*}
{\mathbb E}_xe^{\theta \tau}
\le 1+ \sum_{n=0}^\infty
{\mathbb E}_x\big(e^{\theta \tau}{\mathbf{1}}_{\{nq<\tau\le (n+1)q\}}\big)
&\le1+ \sum_{n=0}^\infty e^{\theta (n+1)q}\,{\mathbb P}_x(\tau>nq)\\
&\le 1+e^{\theta q} \sum_{n=0}^\infty \big((1-p) e^{\theta q}\big)^n,
\end{align*}
which is finite whenever $(1-p)e^{\theta q}<1$, i.e.
$\theta<-(\ln(1-p))/q$.

<!-- lemma-4: body_md -->
\label{lem4.2}
Assume that $X$ is a nonconstant L\'evy process in ${\mathbb R}^d$. Let
$R>0$ and  $\tau_R=\inf\{t>0:|X_t|\ge R\}$. Then for every $t>0$,
\[
\inf_{x\in B(0,R)}{\mathbb P}_x(\tau_R\le t) \ge p,
\quad\mbox{where}\quad p={\mathbb P}_0(|X_{t}|\ge 2R).
\]
<!-- end lemma-4 -->
<!-- proof-20: body_md -->
Fix $t>0$.  For every  $x$ such that $|x|<R$ we have
\[
\{|X_{t}^0|\ge 2R\}\ \subseteq\ \{|x+X^0_{t}|\ge R\}
\ \subseteq\ \{\tau_R\le t\}.
\]
This proves the lemma.

<!-- proposition-9: body_md -->
\label{prop4.2}
Let $X, X^n$, $n\ge1$, be nonconstant L\'evy processes in ${\mathbb R}^d$ and $D\subset \mathbb R^d$ be open and bounded.
If $X^n_1\mathop{\rightarrow}_d X_1$, then for every $q\ge 1$, there exists $C_q>0$ such that 
\begin{equation}
\label{eq8.5}
{\mathbb E}^n_x|\tau_D|^q\le C_q,\quad x\in D,\, n\ge 1.
\end{equation}
<!-- end proposition-9 -->
<!-- proof-21: body_md -->
Choose $R, t_0>0$ so that $D\subset B(0,R):=\{x:|x|<R\}$,  ${\mathbb P}_0(|X_{t_0}|=2R)=0$,
and $p:={\mathbb P}_0(|X_{t_0}|>2R)>0$ (see  \cite[Theorem 24.3]{Sa}).
Define $\tau_R$ as in Lemma \ref{lem4.2}  and set $\tau=\tau_D$ and $p_n={\mathbb P}^n_0(|X_{t_0}|>2R)$. 
Clearly  $\{\tau_R\le t_0\}\subset\{\tau\le t_0\}$, so by Lemma \ref{lem4.2},
\[
\inf_{x\in D}{\mathbb P}^n_x(\tau\le t_0)\ge\inf_{x\in B(0,R)}{\mathbb P}^n_x(\tau_R\le t_0)\ge p_n.
\]
Since $X^n_{t_0}\mathop{\rightarrow}_d X_{t_0}$, we have $p_n\to p$. Thus, by Proposition \ref{prop8.1} and Lemma \ref{lem4.2}
we get \eqref{eq8.5} with a constant $C^0_q>0$ for $n\ge n_0$, where $n_0$ is such that $p_n>p/2$, $n\ge n_0$.  
By  Proposition \ref{prop8.1} and Lemma \ref{lem4.2}, for any $k\in \{1,\dots,n_0-1\}$, there exists $C^k_q$
such that ${\mathbb E}^k_x|\tau_D|^q\le C^k_q$, $x\in D$. Therefore the asserted inequality holds with $C_q:= \max\{C^0_q,C^1_q,\dots, C^{n_0-1}_q\}$.

<!-- proposition-10: body_md -->
\label{prop6.1}
Assume that $x^n\rightarrow x$ in $D({\mathbb R}_+;{\mathbb R}^d)$ as $n\rightarrow\infty$, $x_0\in D$, and moreover
\begin{enumerate}[\rm(a)]
\item for every $0<s<t $, if $\Delta x_s\neq0$, then $x_{s-}\notin\partial D$; 
		
\item	if $t<\infty$, then $t=\bar t$, where $\bar t=\inf\{s>0:x_s\notin \bar D\}$.
\end{enumerate}
Then $t_n\rightarrow t$. If, in addition $t<\infty$ and $x_{t-}\in D$ when $\Delta x_t\neq0$, then $x^n_{t_n}\rightarrow x_t$.
<!-- end proposition-10 -->
<!-- proof-22: body_md -->
We first prove the first assertion in the case  $t=\infty$. Striving for a contradiction, suppose that $t_{n'}\rightarrow t'<\infty$ 
for some subsequence $\{t_{n'}\}$.  Then $x_{t'}$, $x_{t'-}$ are the only possible limit points of the sequence $\{x^{n'}_{t_{n'}}\}$. 
Since $x^{n'}_{t_{n'}}\in D^c$ for each $n'$, the limit points belong to $D^c$. If $x_{t'}\in D^c$ is the limit point, then clearly $t\leq t'$ which is a contradiction.
Consider now the case where  $x_{t'-}\in D^c$ is the limit point.
Then, by assumption (a), $x_{t'}\in D^c$, which means as before that $t\leq t'$ and leads to a contradiction.
	
Assume now that $t<\infty$.
As in  the first   part of the proof one can show that there is no subsequence $\{t_{n'}\}$ such that $t_{n'}\rightarrow t'<t$. Suppose that there is a 
subsequence  $\{t_{n''}\}$ such that $t_{n''}\rightarrow t''>t$. Then, by assumption (b) and
right-continuity of $x$, there are  $t_1,t_2\in(t,t'')$
such that $x_s\in(\bar D)^c$ for $s\in(t_1,t_2]$. Let $s^*$ be such that
\begin{equation}
\label{eq8.6}
t<t_1<s^*<t_2<t''\quad\mbox{and}\quad \Delta x_{s^*}=0.
\end{equation}
Then $x^{n''}_{s^*}\rightarrow x_{s^*}$ and $x^{n''}_{s^*}\in(\bar D)^c$ for all sufficiently large $n''$. This implies that $s^*\ge t_{n''}$ for a 
sufficiently large $n''$. Hence $s^*\ge t''$, which contradicts (\ref{eq8.6}). 

We now turn to the proof of the second assertion. We already know that $t_n\rightarrow t$, so $x_t,x_{t-}$ are the only 
possible limit points of the sequence $\{x^n_{t_n}\}$. Since $x^n_{t_n}\in D^c$, $n\ge1$, these limit points belong to $D^c$. 
Hence, in fact, $x_t$ is the only limit point.
