<!-- theorem-1: body_md -->
\label{thm:main}
Let $\mu$ be a probability measure supported on finitely many contracting similarities of $\mathbb{R}$. Then the self-similar measure $\nu$ of $\mu$ satisfies
\[
 \dim\nu=\min\left\{1,\frac{h_{\mu}}{|\chi_{\mu}|}\right\}.
\]
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{cor:self-similar-sets}
Let $g_1,\ldots , g_k$ be contracting similarities on $\mathbb{R}$ with contraction rates $r_i$ satisfying $0 < |r_i| < 1$ for $1 \leq i \leq k$. Let $s\geq 0$ be the unique non-negative solution of $\sum_{i=1}^k |r_i|^s=1$. If $g_1,\ldots , g_k$ have no exact overlaps, then the self-similar set $K$ satisfies
\[
    \dim K=\min\{1,s\}.
\]
<!-- end corollary-1 -->
<!-- theorem-2: body_md -->
\label{prop:addition}
For $m\geq 1$ and independent finitely supported real random variables $X_1,\ldots,X_m$, it holds that
\begin{equation}\label{eq:addition}
 \sum_{i=1}^m H(X_i)-H\left(\sum_{i=1}^mX_i\right)
 \le 30 m\sum_{i=1}^m\mathcal E(X_i).
\end{equation}

<!-- lemma-1: body_md -->
\label{lem:variance-basic}
The functionals $F_\rho(I)$, $V_\rho(r)$ and $\mathcal E_R(\rho)$ are
concave in $\rho$. Moreover, $0\le V_\rho(r)\le1$ and, for $a,b \in \mathbb{R}$ with $a\ne0$,
\[
 V_{aY+b}(r)=V_Y(r/|a|)\quad \text{ and } \quad \mathcal E(aY+b)=\mathcal E(Y).
\]
If $\rho$ is finitely supported and its support has diameter at most $D$,
then $\mathcal E(\rho)<\infty$ and
\begin{equation}\label{eq:energy-tail}
 \mathcal E(\rho)-\mathcal E_R(\rho)\le\frac{D^2}{2R^2}.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
$F_\rho(I)$ is defined by an infimum over integrals over $\rho$ and is therefore concave. The same follows therefore for
$V_\rho(r)$ and $\mathcal E_R$. Taking the midpoint $c=a+r/2$ in the definition of $F_\rho(I(a,r))$ gives
\[
 0\le F_\rho(I(a,r))\le\frac{r^2}{4}\rho(I(a,r)).
\]
Since $\int_\mathbb R\rho(I(a,r))\,da=r$, the factor $4/r^3$ in
\eqref{eq:local-variance} gives $0\le V_\rho(r)\le1$. Substitution gives the affine change of variables formula.

If the support of $\rho$ has diameter at most $D$, then
every conditional law $\rho\mid I$ also has support of
diameter at most $D$, so $\operatorname{Var}(\rho\mid I)\le D^2/4$. Since
$\int_\mathbb R\rho(I(a,r))\,da=r$, we have $V_\rho(r)\le D^2/r^2$.
Hence
\[
 \mathcal E(\rho)-\mathcal E_R(\rho)
 \le D^2\int_R^\infty\frac{dr}{r^3}
 =\frac{D^2}{2R^2}.
\]
For a finitely supported law, $V_\rho(r)$ also vanishes below the smallest
spacing between atoms. Thus the full energy $\mathcal{E}(\rho)$ is finite.

<!-- lemma-2: body_md -->
\label{lem:cpe}
If $X,Y$ are independent finitely supported real random variables, then
\begin{equation}\label{eq:cpe}
 \mathcal A(X\mid X+Y)\le9\bigl(\mathcal D(X)+\mathcal D(Y)\bigr).
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
We begin the proof with the following inequality. Suppose that a random variable $Z$ satisfying $Z\in[0,1]$ almost surely
dominates a uniform variable. Write $T=\mathbb{E}[1-Z]$ and we then claim that
\begin{equation}\label{eq:quantile}
 \mathbb{E}\left[\sqrt{\frac{1-Z}{Z}} \right]\le3\sqrt{2T}.
\end{equation}

We prove this by splitting into the two cases $\{Z\ge1/2\}$ and $\{Z<1/2\}$.
On $\{Z\ge1/2\}$, the integrand is at most $\sqrt{2(1-Z)}$,
whose expectation is at most $\sqrt{2T}$ by Cauchy--Schwarz.

Now we deal with $\{Z<1/2\}$. Write $q=\mathbb{P}[Z<1/2]$ and note $q\le2T$. For $Q(u) = \inf\{z \,:\, \mathbb{P}[Z \leq z] \geq u \}$ with $u \in (0,1)$ note that stochastic domination implies $Q(u) \geq u$. Moreover, $Q(U)$ has the same distribution as $Z$ when $U$ is uniform on $(0,1)$. Therefore
\[
 \mathbb{E} \left[\mathbf 1_{\{Z<1/2\}} \sqrt{\frac{1-Z}{Z}} \right] \le \mathbb{E}[Z^{-1/2}\mathbf 1_{\{Z<1/2\}}]
 \le\int_0^q u^{-1/2}\,du=2\sqrt q\le2\sqrt{2T}.
\] 
The first inequality is trivial and the second inequality holds since $Q(U)$ has the same distribution as $Z$ and therefore $\mathbb{E}[Z^{-1/2}\mathbf 1_{\{Z<1/2\}}] = \int_0^1 Q(u)^{-1/2}1_{\{Q(u)<1/2\}} du$ and then apply $Q(u) \geq u$ and $1_{\{Q(u) < 1/2\}} \leq 1_{\{u \leq q\}}$.

We now write $F_X(a)=\mathbb{P}[X\le a]$ and
$T_X(u)=\mathbb{P}[X'>X+u]$ for a fixed $u>0$ and $X'$ an independent copy of $X$.
Note that $F_X(X)$ stochastically dominates a uniform variable and therefore the same
holds for the larger variable $F_X(X+u)$. Its expected complement
is $\mathbb{P}(X'>X+u)=T_X(u)$, and thus by \eqref{eq:quantile},
\begin{equation}\label{eq:tail-odds}
 \mathbb{E}\left[\sqrt{\frac{1-F_X(X+u)}{F_X(X+u)}}\right]
 \le3\sqrt{2T_X(u)}.
\end{equation}
The reflected estimate follows from
$1-F_Y(Y-u)\ge\mathbb{P}(Y'\ge Y\mid Y)$ for $u>0$; the latter variable
is the distribution function of $-Y$ evaluated at $-Y$ and therefore
also dominates uniform. The expected complement of $1-F_Y(Y-u)$ is
$\mathbb{P}(Y'\le Y-u)=T_Y(u)$ outside the finite set of positive
differences between atoms. Thus the analogous estimate holds for
$F_Y(Y-u)/(1-F_Y(Y-u))$ for almost every $u>0$.

Let $p_s(a) := \mathbb{P}[X\le a | X+Y=s]$ and let
\[
 q_s(a):=\frac{F_X(a)(1-F_Y(s-a))}
 {F_X(a)(1-F_Y(s-a))+(1-F_X(a))F_Y(s-a)}.
\]

For almost every $a$ we have
\begin{align*}
 H(\mathbf 1_{\{X\le a\}}\mid S) & = \mathbb{E}\bigl[-\mathbf 1_{\{X\le a\}}\log p_S(a)
             -\mathbf 1_{\{X>a\}}\log(1-p_S(a))\bigr] \\
             & = \mathbb{E}\bigl[-\mathbf 1_{\{X\le a\}}\log q_S(a)
             -\mathbf 1_{\{X>a\}}\log(1-q_S(a))\bigr] \\ & \quad\quad\quad\quad - \mathbb{E}[\mathcal{D}_{KL} (p_S(a) || q_S(a))]\\
 &\le\mathbb{E}\bigl[-\mathbf 1_{\{X\le a\}}\log q_S(a)
             -\mathbf 1_{\{X>a\}}\log(1-q_S(a))\bigr]
\end{align*}
where $$\mathcal{D}_{KL}(p || q) := p \log \frac{p}{q} + (1-p) \log \frac{1-p}{1-q} \geq 0.$$
Write $$L_a(X, S) := -\mathbf 1_{\{X\le a\}}\log q_S(a) -\mathbf 1_{\{X>a\}}\log(1-q_S(a)).$$
By the above and Tonelli's theorem we have
$$
\mathcal A(X\mid X+Y) \le \mathbb{E} \left[ \int_{\mathbb R} L_{X+u}(X, X+Y) \, du \right] .
$$

We therefore wish to estimate the integrals
$$
\int_{\mathbb R} L_a(x, x+y) \, da.
$$ On $a>x$, write $a=x+u$ with $u>0$. We have
\[
 L_{x+u}(x, x+y) = \log\left(1+
 \frac{1-F_X(x+u)}{F_X(x+u)}
 \frac{F_Y(y-u)}{1-F_Y(y-u)}\right).
\]
Using $\log(1+z)\le\sqrt z$, independence, and
\eqref{eq:tail-odds}, its expectation $\mathbb{E}[L_{X+u}(X, X+Y)]$ is at most
\[
 18\sqrt{T_X(u)T_Y(u)}\le9(T_X(u)+T_Y(u)).
\]
Similarly on $a<x$, write $a=x-u$. We have
\[
 L_{x-u}(x, x+y) = \log\left(1+
 \frac{F_X(x-u)}{1-F_X(x-u)}
 \frac{1-F_Y(y+u)}{F_Y(y+u)}\right).
\]
The reflected estimates give the same bound. Since
\[
 \int_0^\infty T_X(u)\,du=\tfrac12\mathcal D(X),
\]
integration over both halves proves \eqref{eq:cpe}.

<!-- lemma-3: body_md -->
\label{lem:d_to_v}
      Suppose that $Y$ is a random variable. Define
      \[
            v_Y(t) := \mathbb{E} [ \operatorname{Var} (Y | P_t(Y)) ].\]
      Then $$\mathcal{D}(Y) \leq 2 \int_{0}^{\infty} v_Y(t) \, dt.$$
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Write $\rho = \mathcal{L}(Y)$ and let $Y'$ be an independent copy of $Y$ also independent of $P_t$. Then we have $$\mathbb{E}[(Y-Y')^2 \mathbf 1_{P_t(Y)=P_t(Y')} | P_t] = 2 \sum_{C \in P_t} \rho(C)^2 \operatorname{Var} (\rho | C) \leq  2 \sum_{C \in P_t} \rho(C) \operatorname{Var} (\rho | C).$$
      Two points at distance $d$ lie in the same cell of $P_t$ with probability $e^{-dt}$. Therefore averaging over $P_t$ we get $$ \mathbb{E} [(Y-Y')^2 e^{- t |Y-Y'|}] \leq 2 v_Y(t).$$
      Integrating over $t$ and applying Tonelli's theorem gives the required inequality.

<!-- lemma-4: body_md -->
\label{lem:dispersion}
For every finitely supported real random variable $Y$,
\begin{align}\label{eq:poisson-energy}
 \int_0^\infty\mathbb{E}_{P}[\mathcal D(Y\mid P_t(Y))]\,dt
 &\le2\int_0^\infty t\mathbb{E}_{P}[\operatorname{Var}(Y\mid  P_t(Y))]\,dt\notag\\
 &=3\int_0^\infty V_Y(r)\,\frac{dr}{r}=3\mathcal E(Y).
\end{align}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Let $P_t'$ be an independent process of Poisson cuts. By Lemma \ref{lem:d_to_v} we have
$$ \mathbb{E}_P[\mathcal D (Y | P_t(Y))] \leq 2 \int_{s=0}^{\infty} \mathbb{E}_{P, P'} [\operatorname{Var} (Y | P_t(Y), P_s'(Y)) ]\,ds.$$
Since the refinement of two independent Poisson partitions with intensities $t$ and $s$ is a Poisson partition with intensity $t+s$ the right hand side is $2 \int_{s=0}^{\infty} v_{Y}(s+t) \, ds$. Therefore
\begin{align*}
      \int_0^\infty\mathbb{E}_{P}[\mathcal D(Y\mid P_t(Y))]\,dt & \leq 2 \int_0^\infty t\mathbb{E}_{P}[\operatorname{Var}(Y\mid  P_t(Y))]\,dt.
\end{align*}

At intensity $t$, cells with left endpoint $a$
and length $r$ have intensity $t^2e^{-tr}\,da\,dr$. This is because left endpoints
have intensity $t\,da$, and the distance to the next cut is exponential
of rate $t$. Tonelli's theorem and
$\int_0^\infty t^3e^{-tr}\,dt=6r^{-4}$ therefore give
\begin{align*}
 \int_0^\infty t\mathbb{E}_{P}\operatorname{Var}(Y\mid P_t(Y))\,dt
 &=\int_0^\infty\int_0^\infty\int_\mathbb R
       t^3e^{-tr}F_\rho(I(a,r))\,da\,dr\,dt\\
 &=\frac32\int_0^\infty V_\rho(r)\,\frac{dr}{r}.
\end{align*}
This proves
\eqref{eq:poisson-energy}.

<!-- proof-5: body_md -->
Use the notation $X$, $S$, $P_t$ and $J(t)$ introduced at the
beginning of this section. We have $J(0)=0$. The partitions
eventually distinguish all support points almost surely, so bounded
convergence gives
\[
 \lim_{t\to\infty}J(t)
 =H(X\mid S)=\sum_{i=1}^m H(X_i)-H(S).
\]

To justify \eqref{eq:poisson-derivative}, restrict cuts to a compact
interval $K$ containing all supports. This does not change the
information carried by the cell labels. The restricted cuts form
a Poisson process $\eta_t$ on $E=\{1,\ldots,m\}\times K$ with
intensity $t\nu$, where
$\nu=\sum_i\delta_i\otimes(da|_K)$ is finite.
For a fixed configuration $\eta$, let $L_\eta(X)$ denote its tuple
of cell labels and set $f(\eta)=H(L_\eta(X)\mid S)$.
Then $f$ is measurable, $0\le f\le H(X)$, and
$J(t)=\mathbb{E}[f(\eta_t)]$. Thus \cite{LP17}*{Theorem 19.1}, applied
with base intensity $\lambda=0$, perturbation measure $\nu$ and
$n=1$, gives continuous differentiability and
\[
 J'(t)=\int_E
 \mathbb{E}\bigl[f(\eta_t+\delta_z)-f(\eta_t)\bigr]\,\nu(dz).
\]
At $t=0$ the derivative is understood from the right.
For $z=(i,a)$, adding the cut reveals the threshold test
$\mathbf 1_{\{X_i\le a\}}$, so the entropy chain rule gives, for almost
every $a$,
\[
 f(\eta+\delta_{(i,a)})-f(\eta)
 =H(\mathbf 1_{\{X_i\le a\}}\mid S,L_\eta(X)).
\]
Expanding over the cell labels and integrating proves
\eqref{eq:poisson-derivative}, since the threshold test is constant
on $X\in C$ whenever $a\notin C_i$.

Fix a product cell $C=C_1\times\cdots\times C_m$ of positive
probability. Conditional on $X\in C$, the coordinates remain
independent. Applying Lemma~\ref{lem:cpe} to $X_i$ and
$\sum_{j\ne i}X_j$ under this conditional law, and using the
triangle inequality for independent copies, gives
\[
 \int_{C_i}
 H(\mathbf 1_{\{X_i\le a\}}\mid S,X\in C)\,da
 \le 9\sum_{j=1}^m\mathcal D(X_j\mid X_j\in C_j).
\]
Substitute this into \eqref{eq:poisson-derivative}, sum over $i$
and average over the cells. The marginal cell weights give
\[
 J'(t)\le9m\sum_{i=1}^m
 \mathbb{E}_{P_t}\left[\mathcal D(X_i\mid P_t^{(i)}(X_i))\right].
\]
Integrating and applying Lemma~\ref{lem:dispersion}, we conclude
\[
 \sum_{i=1}^m H(X_i)-H(S)
 =\int_0^\infty J'(t)\,dt
 \le27m\sum_{i=1}^m\mathcal E(X_i),
\]
which proves the theorem since $27\le30$.

<!-- lemma-6: body_md -->
\label{lem:hochman}
If $\dim \nu<1$, then, for every $C>e^{|\chi_{\mu}|}$,
\[
 H(B_n;C^{-n})=n|\chi_{\mu}| \dim \nu+o(n).
\]
<!-- end lemma-6 -->
<!-- proof-6: body_md -->
Let $n'=n|\chi_\mu|/\log 2$. Hochman's theorem
\cite{H}*{Theorem~1.4 and the discussion following it}, after converting
to natural logarithms, gives
\[
 H\bigl(\lfloor2^{qn'}B_n\rfloor,R_n\bigr)
 =n|\chi_\mu|\dim\nu+o(n)\qquad(q>1).
\]
Integer parts of the partition levels change the entropy by $O(1)$.
Clearly $H(R_n)\le k\log(n+1)=o(n)$. We may therefore discard $R_n$.
Any two translated grids of the same mesh have entropies differing
by at most $\log 2$, since each cell of either grid meets at most two
cells of the other. Averaging over the translation and taking
$q=\log C/|\chi_\mu|$ proves the claim.

<!-- proposition-1: body_md -->
\label{prop:finite-sum}
Suppose $Y=\sum_{j=1}^L a_jD_j$, where the $a_j$ are real constants
and the $D_j$ are independent real random variables, each taking at
most $K$ values. For every $R>0$ and positive integer $m$,
\begin{equation}\label{eq:finite-sum}
 \mathcal E_R(Y)\ge\frac1{30m}\left(
 H(Y)-H(Y;R)-\frac{L(K-1)\log(m+1)}m\right)-\frac12.
\end{equation}
<!-- end proposition-1 -->
<!-- proof-7: body_md -->
Take independent copies of the digit vector and put
$Y_i=\sum_{j=1}^L a_jD_j^{(i)}$ and $S=\sum_{i=1}^mY_i$.
For each $j$, the sum $\sum_iD_j^{(i)}$ is determined by the
multiplicities of its possible values. The first $K-1$ multiplicities
determine the last, so
\[
 H(S)\le L(K-1)\log(m+1).
\]
Fix $t\in[0,R)$ and write $Q_i=\lfloor(Y_i+t)/R\rfloor$ and
$Q=\lfloor(Y+t)/R\rfloor$. Conditioned on $(Q_1,\ldots,Q_m)$,
the $Y_i$ remain independent. Apply Theorem~\ref{prop:addition}, average over
the $Q_i$, and use $H(S\mid Q_1,\ldots,Q_m)\le H(S)$ to obtain
\[
 mH(Y\mid Q)\le H(S)+30m^2\mathbb{E}\bigl[\mathcal E(\mathcal L(Y\mid Q))\bigr].
\]
Each conditional law is supported on an interval of length $R$.
The tail bound and concavity from Lemma~\ref{lem:variance-basic} with $D = R$ therefore give
\[
 \mathbb{E}\bigl[\mathcal E(\mathcal L(Y\mid Q))\bigr]
 \le\mathbb{E}\bigl[\mathcal E_R(\mathcal L(Y\mid Q))\bigr]+\frac12
 \le\mathcal E_R(Y)+\frac12.
\]
Average over $t$, use \eqref{eq:conditional-entropy}, and divide by
$m$. Rearranging gives \eqref{eq:finite-sum}.

<!-- corollary-2: body_md -->
\label{cor:fine-variance}
Suppose that $\dim \nu<\min\{1,h_\mu/|\chi_{\mu}|\}$. Then there is $\eta>0$, depending
only on $\mu$, such that, for every $C>e^{|\chi_{\mu}|}$ and all sufficiently
large (in terms of $C$) $n$,
\begin{equation}\label{eq:fine-variance}
 \mathbb{E}\bigl[\mathcal E_{C^{-n}}(\mathcal L(B_n\mid R_n))\bigr]\ge\eta n.
\end{equation}
<!-- end corollary-2 -->
<!-- proof-8: body_md -->
Put $\delta=h_\mu-|\chi_{\mu}| \dim \nu >0$. Choose a block length $N$ such that
$k\log(N+1)/N<\delta/4$. To reveal only little entropy, split the first $n$ increments into
$L=\lceil n/N\rceil$ consecutive blocks of length $N$, except
possibly the last, and write their products as $x\mapsto A_jx+D_j$.
Set $T=(A_1,\ldots,A_L)$. Each $A_j$ takes at most $(N+1)^k$ values,
and $R_n$ is determined by $T$. Hence
\begin{align*}
 H(B_n\mid T)
 &=H(G_n\mid T)\ge H(G_n)-H(T)\\
 &\ge nh_\mu-Lk\log(N+1)
 \ge n(h_\mu-\delta/4)-O(1),
\end{align*}
where $H(G_n)\ge nh_\mu$ follows from subadditivity of random-walk
entropy. Concavity of entropy at a scale and Lemma~\ref{lem:hochman}
give
\[
 \mathbb{E}[H(\mathcal L(B_n\mid T);C^{-n})]
 \le H(B_n;C^{-n})=n|\chi_{\mu}| \dim \nu +o(n).
\]
Thus the average conditional entropy gap is at least $\delta n/2$
for large $n$.

Given $T$, the $D_j$ remain independent and
\[
 B_n=D_1+A_1D_2+\cdots+(A_1\cdots A_{L-1})D_L
\]
has fixed coefficients. Each digit has at most $K=k^N$ values.
Choose $m$ so large that
$(K-1)\log(m+1)/m<\delta N/8$.
Apply Proposition~\ref{prop:finite-sum} conditionally and average.
For large $n$, this gives
\[
 \mathbb{E}\bigl[\mathcal E_{C^{-n}}(\mathcal L(B_n\mid T))\bigr]
 \ge\frac{\delta n}{120m}-\frac12.
\]
Finally, $R_n$ is a function of $T$, so concavity of $\mathcal E_{C^{-n}}$
allows us to replace $T$ by $R_n$. Decreasing the constant proves
\eqref{eq:fine-variance}, for example with $\eta=\delta/(240m)$.

<!-- definition-1: body_md -->
\label{def:convolution-variance}
For a compactly supported probability measure $\lambda$ and $r>0$,
a scale-$r$ convolution disintegration is a representation
\[
 \lambda=\int_I \lambda_i^{(1)}*\cdots*
 \lambda_i^{(m_i)}\,d\theta(i),
\]
where $(I,\theta)$ is a probability space, $m_i$ is a finite positive
integer, and each factor is supported on an interval of length at
most $r$. All families are required to be measurable. Define
\begin{equation}\label{eq:convolution-variance}
 W(\lambda;r)=\inf\int_I
 \exp\left(-\frac4{r^2}\sum_{j=1}^{m_i}
 \operatorname{Var}(\lambda_i^{(j)})\right)\,d\theta(i),
\end{equation}
with the infimum over these disintegrations.
<!-- end definition-1 -->
<!-- proposition-2: body_md -->
\label{prop:w_to_zero}
Suppose that $\nu$ is a self-similar measure on $\mathbb R$ and that there is
a sequence $r_n\to0$ such that $W(\nu;r_n)\to0$. Then $\dim\nu=1$.
<!-- end proposition-2 -->
<!-- lemma-7: body_md -->
\label{lem:convolution-properties}
The functional $W$ takes values in $[0,1]$ and is convex in its
measure argument. Moreover,
\begin{align}
 W(aY+b;r)&=W(Y;r/|a|) &&(a\ne0),\label{eq:affine-W}\\
 W(\lambda*\theta;r)&\le W(\lambda;r)W(\theta;r),\label{eq:product-W}\\
 W(\lambda;s)&\le W(\lambda;r)^{r^2/s^2} &&(s\ge r),\label{eq:scale-W}\\
 W(\lambda;r)&\le1-(1-e^{-1})V_\lambda(r).\label{eq:local-to-convolution}
\end{align}
<!-- end lemma-7 -->
<!-- proof-9: body_md -->
Clearly $W\le1$. Convexity follows by combining disintegrations, affine changes of variables give \eqref{eq:affine-W}, and
concatenating two independent disintegrations gives
\eqref{eq:product-W}. A scale-$r$ convolution disintegration is also a
scale $s$ convolution disintegration for $s\ge r$. From the definition it is immediate that if its integrand at scale $r$ is $Z\in[0,1]$,
its integrand at scale $s$ is $Z^{r^2/s^2}$. Jensen's inequality
followed by the infimum gives \eqref{eq:scale-W}.

For \eqref{eq:local-to-convolution}, condition $\lambda$ on a shifted
grid of mesh $r$, taking one factor in each component, and average
over a uniform shift. If $Q$ denotes the resulting cell label, every
normalized component variance
$u=4\operatorname{Var}(\lambda\mid Q)/r^2$ lies in $[0,1]$. The inequality
$e^{-u}\le1-(1-e^{-1})u$ and the definition of $V_\lambda(r)$
give the result.

<!-- lemma-8: body_md -->
\label{lemma:approx_normal}
There is an absolute constant $C>0$ such that the following holds.
Suppose that $Y_1,\ldots,Y_n$ are independent real-valued random
variables with finite third moments and mean zero. Let
\[
 \omega_i^2=\mathbb{E}[Y_i^2],\qquad \gamma_i^3=\mathbb{E}[|Y_i|^3],
 \qquad r^2=\sum_{i=1}^n\omega_i^2>0.
\]
Then
\[
 \mathcal W_1\left(\sum_{i=1}^nY_i,\mathcal N(0,r^2)\right)
 \le C\frac{\sum_{i=1}^n\gamma_i^3}{r^2}.
\]
In particular, if $|Y_i|\le\varepsilon$ almost surely for every $i$,
then
\[
 \mathcal W_1\left(\sum_{i=1}^nY_i,\mathcal N(0,r^2)\right)
 \le C\varepsilon.
\]
<!-- end lemma-8 -->
<!-- proof-10: body_md -->
Many estimates of this form are available in the literature and we
    cite \cite{Gol10}*{Theorem~1.1} as one convenient version. In one
    dimension, the Wasserstein distance is the $L^1$ distance from \cite{Gol10}*{Theorem~1.1}. So \cite{Gol10}*{Theorem~1.1}, after
    rescaling by $r$, gives the first assertion with $C=1$. For the
    second assertion simply note that
    $\gamma_i^3\leq\varepsilon\omega_i^2$ and sum over $i$.

<!-- lemma-9: body_md -->
\label{lemma:wasserstein_enough}
For every $0<a<b<\infty$ and $\varepsilon>0$, there is $\delta>0$
such that the following holds. If $r>0$, $\sigma\in[ar,br]$, and $Y$
is a mean-zero random variable satisfying
\[
 \mathbb{E}[Y^2]\le b^2r^2,
 \qquad
 \mathcal W_1(Y,\mathcal N(0,\sigma^2))<\delta r,
\]
then
\[
 \left|H(Y;r)-H(\mathcal N(0,\sigma^2);r)\right|<\varepsilon.
\]
<!-- end lemma-9 -->
<!-- proof-11: body_md -->
By scaling, it suffices to take $r=1$. If the assertion failed, there
would be mean-zero random variables $Y_n$ and numbers
$\sigma_n\in[a,b]$ such that
\[
 \mathbb{E}[Y_n^2]\le b^2,\qquad
 \mathcal W_1(Y_n,\mathcal N(0,\sigma_n^2))\longrightarrow0,
\]
but the asserted entropy convergence fails. Passing to a subsequence,
we may assume that $\sigma_n\to\sigma\in[a,b]$. It follows that
$Y_n$ converges in distribution to $\mathcal N(0,\sigma^2)$.

For $k\in\mathbb Z$ and $t\in[0,1]$, write
\[
 A_{k,t}^{(n)}=\mathbb{P}[Y_n\in[k-t,k+1-t)],
 \quad \text{and} \quad
 B_{k,t}=\mathbb{P}[\mathcal N(0,\sigma^2)\in[k-t,k+1-t)].
\]
Then $A_{k,t}^{(n)}\to B_{k,t}$ for every $k$ and almost every $t$.
Moreover, Chebyshev's inequality gives, uniformly in $n$ and $t$, $A_{k,t}^{(n)}\ll_b(1+|k|)^{-2}.$ Consequently the terms $-A_{k,t}^{(n)}\log A_{k,t}^{(n)}$ admit a
summable dominating function. Dominated convergence gives $H(Y_n;1)\rightarrow H(\mathcal N(0,\sigma^2);1)$ as $n\to \infty$, which is
a contradiction.

<!-- lemma-10: body_md -->
\label{lemma:variance_sum_to_entropy}
For every integer $C>1$ and $\varepsilon>0$, there are $\delta>0$
and $A>0$ such that the following holds. If $r>0$ and
$Y_1,\ldots,Y_n$ are independent mean-zero real-valued random
variables satisfying
\[
 |Y_i|\le\delta r\quad\text{almost surely},
 \qquad
 \sum_{i=1}^n\operatorname{Var}(Y_i)\ge Ar^2,
\]
then
\[
 H\left(\sum_{i=1}^nY_i;r\mid Cr\right)>\log C-\varepsilon.
\]
<!-- end lemma-10 -->
<!-- proof-12: body_md -->
By scale invariance and the smoothness of the Gaussian density, we may
choose $A$ sufficiently large such that whenever $Ar^2\le\sigma^2\le(A+1)r^2$ it holds that
\[
 H(\mathcal N(0,\sigma^2);r\mid Cr)>
 \log C-\frac{\varepsilon}{2}.
\]
Choose $\delta<1$ sufficiently
small for Lemmas~\ref{lemma:approx_normal} and
\ref{lemma:wasserstein_enough}, applied at both scales $r$ and $Cr$,
to give a total entropy error of less than $\varepsilon/2$.

Take the shortest initial subsum $S=Y_1+\cdots+Y_\ell$ whose variance
is at least $Ar^2$. Writing $\sigma^2=\operatorname{Var}(S)$, minimality and
$\operatorname{Var}(Y_i)\le\delta^2r^2$ give
\[
 Ar^2\le\sigma^2\le(A+\delta^2)r^2\le(A+1)r^2.
\]
Lemma~\ref{lemma:approx_normal} gives
\[
 \mathcal W_1(S,\mathcal N(0,\sigma^2))\ll\delta r.
\]
It follows from Lemma~\ref{lemma:wasserstein_enough}, used at the two
scales, that
\[
 H(S;r\mid Cr)>\log C-\varepsilon.
\]
Finally, adding the remaining independent summands cannot decrease
entropy between these scales by, for example, Lemma~\ref{lemma:entropy_concave} and the translation invariance of entropy between two scales.

<!-- proof-13: body_md -->
We first show $W(\nu;s)\to0$ as $s\to0$. For $0<s\le t$,
let $\sigma=\min\{n\ge0:|R_n|\le s/t\}$. This stopping time is
bounded, and for $\rho_{\min} = \min_{1\leq i \leq k} |r_i|$ we have
\[
 \rho_{\min} s/t<|R_\sigma|\le s/t.
\]
By \eqref{eq:stationarity}, convexity, affine invariance and
\eqref{eq:scale-W},
\begin{equation}\label{eq:propagate}
 W(\nu;s)\le\mathbb{E}[W(\nu;s/|R_\sigma|)]
 \le W(\nu;t)^{\rho_{\min}^2}.
\end{equation}
Passing to a decreasing subsequence of $(r_n)$ and applying
\eqref{eq:propagate} with $t=r_n$ shows that $W(\nu;s)\to0$ as
$s\to0$.

Next we show that $H(\nu;r\mid2r)\to\log2$ as $r\to0$. Let
$\varepsilon>0$, and choose $\delta$ and $A$ from
Lemma~\ref{lemma:variance_sum_to_entropy}, with $C=2$. Since
$W(\nu;\delta r)\to0$, for all sufficiently small $r$ there is a
scale-$\delta r$ convolution disintegration
\[
 \nu=\int_I\lambda_i^{(1)}*\cdots*\lambda_i^{(m_i)}\,d\theta(i)
\]
such that, writing
$S_i=\sum_{j=1}^{m_i}\operatorname{Var}(\lambda_i^{(j)})$,
\[
 \int_I\exp\left(-\frac{4S_i}{\delta^2r^2}\right)d\theta(i)
 <\varepsilon\exp\left(-\frac{4A}{\delta^2}\right).
\]
It follows that $\theta(\{i:S_i<Ar^2\})<\varepsilon$. For every other $i$, Lemma~\ref{lemma:variance_sum_to_entropy} gives
\[
 H\left(\lambda_i^{(1)}*\cdots*\lambda_i^{(m_i)};
 r\mid2r\right)\ge\log2-\varepsilon.
\]
The concavity of entropy between two scales therefore gives
\[
 H(\nu;r\mid2r)\ge\log2-O(\varepsilon).
\]
Consequently, as $r \to 0$,
\[
 H(\nu;r)=\log r^{-1}+o(\log r^{-1}),
\]
and hence $\dim\nu=1$.

<!-- lemma-11: body_md -->
\label{lem:concatenation}
Let $0<s_1<\cdots<s_m$ and let $\tau_1,\ldots,\tau_m$ be bounded
stopping times for independent walks with independent increments of law
$\mu$.
Write $(C_j,D_j)$ for their signed contraction ratio and translation.
Suppose that, for $j<m$,
\begin{equation}\label{eq:separation}
 s_j\le\rho_{\min} s_{j+1}|C_j|\quad\text{almost surely}.
\end{equation}
For $0<r\le s_1$, these blocks can be concatenated to one walk ending
at a bounded stopping time $T$, such that
\begin{equation}
 \mathbb{E}[W(\mathcal L(B_T\mid R_T);r)]
 \le\prod_{j=1}^m
 \mathbb{E}[W(\mathcal L(D_j\mid C_j);s_j)]^{\rho_{\min}^2},
 \label{eq:concatenation-prefix}
 \end{equation}
 and consequently
 \begin{equation}
     W(\nu;r)
 \le\prod_{j=1}^m
 \mathbb{E}[W(\mathcal L(D_j\mid C_j);s_j)]^{\rho_{\min}^2}.
 \label{eq:concatenation-nu}
 \end{equation} Moreover, $|R_T|>\rho_{\min} r|C_m|/s_m$ almost surely.
<!-- end lemma-11 -->
<!-- proof-14: body_md -->
For $1\leq j \leq m$ define
\[
 S_j=\min\{n\ge0:|R_n|\le r/s_j\} \quad\text{ and } \quad
T_j=S_j+\tau_j^{(S_j)},
\]
where $\tau_j^{(S_j)}$ applies the $j$th stopping rule to the
increments after $S_j$. In other words the law of $(X_{S_j+1}, X_{S_j+2},\dots, X_{S_j+\tau_j^{(S_j)}})$ given $X_1, \dots, X_{S_j}$ is $\mu^{*\tau_j}$. Denote $A_j=R_{S_j}$. By construction
\begin{equation}\label{eq:overshoot}
 \rho_{\min} r/s_j<|A_j|\le r/s_j.
\end{equation}
Thus \eqref{eq:separation} implies
\[
 |R_{T_j}|=|A_jC_j|>r/s_{j+1},
\]
so $T_j<S_{j+1}$. Therefore, the selected blocks are disjoint and their
stopped products are independent. As $\max_i|r_i|<1$ all stopping times
are uniformly bounded.
We take $T=T_m$ and the final assertion that $|R_T|>\rho_{\min} r|C_m|/s_m$ follows from \eqref{eq:overshoot}.

Let $\mathcal A$ include the ratios $C_1,\ldots,C_m$ and the complete words
in all intervening gaps including the initial gap. Then the $A_j$
and $R_T$ are $\mathcal A$-measurable, and
\begin{equation}\label{eq:conditional-blocks}
 B_T=d_{\mathcal A}+\sum_{j=1}^m A_jD_j
\end{equation}
for an $\mathcal A$-measurable translation $d_{\mathcal A}$. Given $\mathcal A$, the $D_j$
are independent and have laws $\mathcal L(D_j\mid C_j)$. 

Put $v_j(c)=W(\mathcal L(D_j\mid C_j=c);s_j)$. From
\eqref{eq:conditional-blocks}, \eqref{eq:product-W} and
\eqref{eq:overshoot},
\[
 W(\mathcal L(B_T\mid\mathcal A);r)
 \le\prod_{j=1}^m
 W(\mathcal L(D_j\mid C_j);r/|A_j|)
 \le\prod_{j=1}^m v_j(C_j)^{\rho_{\min}^2}.
\]
Average and use convexity to pass from $\mathcal A$ to $R_T$. The $C_j$
are independent, so Jensen's inequality yields
\[
 \mathbb{E}\left[\prod_jv_j(C_j)^{\rho_{\min}^2}\right]
 =\prod_j\mathbb{E}[v_j(C_j)^{\rho_{\min}^2}]
 \le\prod_j \mathbb{E}[v_j(C_j)]^{\rho_{\min}^2}.
\]
This proves \eqref{eq:concatenation-prefix}. Adding the
tail $R_TZ'$ with $Z' \sim \nu$ independent of the random walk proves \eqref{eq:concatenation-nu}.

<!-- proposition-3: body_md -->
\label{prop:stopped-block}
Suppose $\dim \nu<\min\{1,h_\mu/|\chi_{\mu}|\}$. Then there is $\kappa>0$ such that,
for every $\varepsilon>0$, there is a bounded stopping time $\tau$
and some $r>0$ satisfying
\begin{equation}\label{eq:stopped-block}
 \mathbb{E}[W(\mathcal L(B_\tau\mid R_\tau);r)]\le e^{-\kappa},
 \qquad r<\varepsilon|R_\tau|\quad\text{almost surely}.
\end{equation}
<!-- end proposition-3 -->
<!-- proof-15: body_md -->
Fix $C>\rho_{\min}^{-1}$ with $\rho_{\min} = \min_{1 \leq i \leq k} |r_i|$ and let $\eta$ be as in
Corollary~\ref{cor:fine-variance}. For sufficiently large $N$, write
\[
 w_N(s)=\mathbb{E}[V_{\mathcal L(B_N\mid R_N)}(s)] \quad \text{ and } \quad
  d_N=(N+1)\log\rho_{\min}^{-1}.
\]
Then $\int_0^{C^{-N}}w_N(s)\,ds/s\ge\eta N$ for a suitable $\eta > 0$.
The conditional laws involved are finitely supported, so $w_N(s)=0$
for all sufficiently small $s$. Averaging the logarithmic grids
$\{e^{u+jd_N}:j\in\mathbb Z\}$ over $u\in[0,d_N)$ gives
\[
 \frac1{d_N}\int_0^{d_N}
 \sum_{e^{u+jd_N}<C^{-N}}w_N(e^{u+jd_N})\,du
 =\frac1{d_N}\int_0^{C^{-N}}w_N(s)\,\frac{ds}{s}
 \ge\frac{\eta}{2\log\rho_{\min}^{-1}}.
\]
Consequently, some finite collection
$0<s_1<\cdots<s_m<C^{-N}$ satisfies
\begin{equation}\label{eq:selected-scales}
 s_{j+1}\ge\rho_{\min}^{-(N+1)}s_j \quad \text{ and } \quad \sum_{j=1}^m w_N(s_j)\ge
 \eta_0:=\frac{\eta}{2\log\rho_{\min}^{-1}}>0.
\end{equation}
Apply Lemma~\ref{lem:concatenation} with $r=s_1$ and with every
block having the fixed length $N$. Condition \eqref{eq:separation}
holds because $|R_N|\ge\rho_{\min}^N$. Using
\eqref{eq:local-to-convolution} and $1-x\le e^{-x}$ gives
\begin{align*}
 \mathbb{E}[W(\mathcal L(B_T\mid R_T);r)]
 &\le\prod_{j=1}^m
 \bigl(1-(1-e^{-1})w_N(s_j)\bigr)^{\rho_{\min}^2}\\
 &\le\exp\bigl(-\rho_{\min}^2(1-e^{-1})\eta_0\bigr).
\end{align*}
Moreover, the last assertion of Lemma~\ref{lem:concatenation} gives
\[
 \frac r{|R_T|}<\frac{s_m}{\rho_{\min}^{N+1}}
 <\rho_{\min}^{-1}(C\rho_{\min})^{-N}.
\]
Since $C\rho_{\min}>1$, choose $N$ sufficiently large that the last
expression is at most $\varepsilon$. Taking $\tau=T$ and
$\kappa=\rho_{\min}^2(1-e^{-1})\eta_0$ proves the proposition.

<!-- proof-16: body_md -->
The upper bound
\[
 \dim \nu \leq \min \left\{1,\frac{h_\mu}{|\chi_\mu|}\right\}
\]
is well known. In particular it follows from Feng--Hu \cite{FH}. Indeed, apply
their projection-entropy bound \cite{FH}*{Theorem~2.2(i)} to the system given by the distinct
similarities in $\operatorname{supp}(\mu^{*n})$. Its symbolic entropy is
$H(\mu^{*n})$ and its Lyapunov exponent is $n\chi_\mu$, so
$\dim\nu\leq\min\{1,
 \frac{H(\mu^{*n})}{n|\chi_\mu|}\}$.
Letting $n\to\infty$ gives the claimed upper bound.
It remains to show that $\dim \nu \geq \min \{ 1, \frac{h_{\mu}}{|\chi_{\mu}|} \}$. Assume for contradiction that this does not hold.

Choose $(\tau_1,r_1)$ from
Proposition~\ref{prop:stopped-block} with $1$ in the role of $\varepsilon$.
Inductively choose $(\tau_{j+1},r_{j+1})$ with $\rho_{\min} r_j$
 in the role of $\varepsilon$. Thus
\begin{equation}\label{eq:recursive-scales}
 r_{j+1}<\rho_{\min} r_j|R_{\tau_{j+1}}|
 \quad \text{ and } \quad
 \mathbb{E}[W(\mathcal L(B_{\tau_j}\mid R_{\tau_j});r_j)]\le e^{-\kappa}.
\end{equation} For each $n$, apply Lemma~\ref{lem:concatenation} at the scale
$r_n$ with the blocks ordered as
\[
 (\tau_n,r_n),(\tau_{n-1},r_{n-1}),\ldots,(\tau_1,r_1).
\]
The separation hypothesis is exactly \eqref{eq:recursive-scales}.
Consequently
\[
 W(\nu;r_n)\le e^{-n\kappa\rho_{\min}^2}\longrightarrow0.
\]
Proposition~\ref{prop:w_to_zero} gives $\dim\nu=1$, a
contradiction. This proves the theorem.
