\label{def:GH}
A diffeomorphism $f\in\mathfrak{D}(\mathbb{B})$ satisfies \emph{generalized
hyperbolicity} if there exist constants $C>0$, $\lambda\in(0,1)$ and a
family of projections
$P_x,Q_x:\mathbb{B}\to\mathbb{B}$, $x\in\mathbb{B}$,
such that, writing
$E_x^s:=P_x(\mathbb{B})$,
$E_x^u:=Q_x(\mathbb{B})$,
the following conditions hold.
\begin{enumerate}[label=\textup{(GH\arabic*)},leftmargin=12mm]
\item \textbf{Splitting.} For every $x\in\mathbb{B}$,
\[
E_x^s\oplus E_x^u=\mathbb{B}, \qquad
 P_x+Q_x=\operatorname{Id},
 \qquad
 \left\lVert P_x\right\rVert,\left\lVert Q_x\right\rVert\leq C.
\]

\item \textbf{Inclusion.} For every $x\in\mathbb{B}$,
\[
 Df(x)E_x^s\subseteq E_{f(x)}^s,
 \qquad
 D(f^{-1})(x)E_x^u\subseteq E_{f^{-1}(x)}^u.
\]

\item \textbf{Exponential estimates.} For every $x\in\mathbb{B}$ and every
integer $n>0$,
\[
 \left|Df^n(x)v_x^s\right|\leq C\lambda^n\left|v_x^s\right|,
 \qquad v_x^s\in E_x^s,
\]
\[
 \left|D(f^{-n})(x)v_x^u\right|\leq C\lambda^n\left|v_x^u\right|,
 \qquad v_x^u\in E_x^u.
\]
\end{enumerate}
No finite-dimensionality assumption is imposed on $E_x^u$.
No continuity, uniform continuity, or measurability of the maps
$x\mapsto P_x,Q_x$ is assumed.

\label{def:stability}
Let $f\in\mathfrak{D}(\mathbb{B})$.  We call $f$ \emph{semi-structurally stable} if
for every $\varepsilon>0$ there exists $d_0>0$ such that every
$g\in\mathfrak{D}(\mathbb{B})$ with $\left\lVert f-g\right\rVert_{C^1}<d_0$ admits continuous maps
$h,k:\mathbb{B}\to\mathbb{B}$ satisfying
$
 \max\{\left\lVert h\right\rVert_\infty,\left\lVert k\right\rVert_\infty\}<\varepsilon
$
and, writing $H=\operatorname{Id}+h$ and $K=\operatorname{Id}+k$,
\[
 g\circ H=H\circ f,
 \qquad
 f\circ K=K\circ g.
\]
Here $H$ and $K$ are called semiconjugacies.\footnote{Surjectivity of either
map is not required: in finite dimensions, $\operatorname{Id}+h$ with $h$ bounded
and continuous is surjective by Brouwer's fixed point theorem, but in infinite-dimensional Banach spaces it
need not be.}

We call $f$ \emph{structurally stable} if, under the same quantifiers,
the maps can be chosen so that
$
 K\circ H=H\circ K=\operatorname{Id}.
$
Thus $H$ and $K$ are mutually inverse homeomorphisms.  


We call $f$ \emph{topologically stable} if for every $\varepsilon>0$
there exists $d_0>0$ such that every homeomorphism $g:\mathbb{B}\to\mathbb{B}$ with
$\sup_{x\in\mathbb{B}}\left|f(x)-g(x)\right|<d_0$ admits a continuous map
$k:\mathbb{B}\to\mathbb{B}$ with $\left\lVert k\right\rVert_\infty<\varepsilon$ such that $K=\operatorname{Id}+k$
satisfies
$
 f\circ K=K\circ g.
$

\label{thm:main}
Assume that $f\in\mathfrak{D}(\mathbb{B})$ satisfies generalized hyperbolicity.  Then
there exist constants $M>0$ and $d_0>0$ such that the following holds.
Let $\Delta:\mathbb{B}\to\mathbb{B}$ be a bounded Lipschitz map satisfying
 $\left\lVert \Delta\right\rVert_{\infty}\leq d$,
 $\operatorname{Lip}(\Delta)\leq d$, with 
 $0<d<d_0$,
and put
$
 g:=f+\Delta$.
Then $g$ is a bi-Lipschitz homeomorphism, and there exist
$h,k\in C_b^0(\mathbb{B},\mathbb{B})$ satisfying

\begin{equation}
\label{eq:main-semiconjugacy-reverse}
  g\circ(\operatorname{Id}+h)=(\operatorname{Id}+h)\circ f, \qquad f\circ(\operatorname{Id}+k)=(\operatorname{Id}+k)\circ g,
\end{equation}
\begin{equation*}
 \left|h(x)\right|,\left|k(x)\right|\leq Md,
 \qquad x\in\mathbb{B}.
\end{equation*}

\label{thm:topological-stability}
Assume that $f\in\mathfrak{D}(\mathbb{B})$ satisfies generalized hyperbolicity.  Then
there exist constants $M>0$ and $d_0>0$ such that the following holds.
Let $g:\mathbb{B}\to\mathbb{B}$ be a homeomorphism satisfying
$
 \sup_{x\in\mathbb{B}}\left|g(x)-f(x)\right|\leq d < d_0.
$
Then there exists $k\in C_b^0(\mathbb{B},\mathbb{B})$ satisfying
\begin{equation}
\label{eq:top-stability-semiconjugacy}
 f\circ(\operatorname{Id}+k)=(\operatorname{Id}+k)\circ g,
 \qquad
 \left|k(x)\right|\leq Md,
 \quad x\in\mathbb{B}.
\end{equation}

<!-- remark-1: body_md -->
\label{rem:trajectory-set-stability}
Walters proved that Anosov diffeomorphisms are topologically stable
\cite{WaltersTopStability} and, more generally, that every expansive
homeomorphism of a compact metric space with the shadowing property is
topologically stable \cite{WaltersPOTP}.  Nitecki \cite{Nitecki} proved
lower semi-stability, a closely related one-sided notion, for
diffeomorphisms of smooth closed manifolds satisfying Axiom~A and the
strong transversality condition; these need not be expansive.
Theorem~\ref{thm:topological-stability} differs from both results in
its setting and in its proof.  The phase space is a Banach space, so
compactness is unavailable; the map need not be expansive, so the
uniqueness of shadowing trajectories, which yields the continuity of
the semiconjugacy in Walters' argument, is unavailable as well; and the
finite decomposition of the nonwandering set used in the theory of
Axiom~A diffeomorphisms does not seem available, as the product of
Subsection~\ref{sec:product-results} shows.  

The semiconjugacies also control complete trajectories.
For a map $q:\mathbb{B}\to\mathbb{B}$, let
\[
 \mathcal T_q:=
 \bigl\{\mathbf x=(x_n)_{n\in\mathbb{Z}}\in\mathbb{B}^{\mathbb{Z}}:
             x_{n+1}=q(x_n)\text{ for all }n\in\mathbb{Z}\bigr\}
\]
be its complete-trajectory set.  Equip $\mathbb{B}^{\mathbb{Z}}$ with the extended
metric
\[
 d_\infty(\mathbf x,\mathbf y):=
       \sup_{n\in\mathbb{Z}}\left|x_n-y_n\right|\in[0,+\infty].
\]
For nonempty $A,B\subset\mathbb{B}^{\mathbb{Z}}$, consider the Hausdorff distance
\[
\operatorname{dist}(A,B):=\sup_{\mathbf x\in A}\inf_{\mathbf y\in B}
                        d_\infty(\mathbf x,\mathbf y),
 \qquad
 d_{H}(A,B):=\max\{\operatorname{dist}(A,B),\operatorname{dist}(B,A)\}.
\]
Under the assumptions of Theorem~\ref{thm:main},
$
 d_{H}(\mathcal T_f,\mathcal T_g)\leq Md.
$
Indeed, writing $H=\operatorname{Id}+h$ and $K=\operatorname{Id}+k$, the semiconjugacy identities imply that
$(H(x_n))_{n\in\mathbb{Z}}\in\mathcal T_g$ for every
$\mathbf x\in\mathcal T_f$, and
$(K(y_n))_{n\in\mathbb{Z}}\in\mathcal T_f$ for every
$\mathbf y\in\mathcal T_g$.  Each image trajectory is at
$d_\infty$-distance at most $Md$ from the original one, proving both
directed estimates.  Under the assumptions of
Theorem~\ref{thm:topological-stability} alone, the semiconjugacy $K$
still gives the one-sided estimate
$\operatorname{dist}(\mathcal T_g,\mathcal T_f)\leq Md$ for every homeomorphism $g$
uniformly close to $f$.

Thus $f$ has a quantitative form of \emph{strong tolerance stability},
relative to the bounded Lipschitz perturbations in
Theorem~\ref{thm:main}; compare \cite{MazurTolerance}.
The comparison is uniform over all time
indices; ordinary tolerance stability compares orbit closures as
subsets of phase space.  In shadowing terminology, the estimate from
$\mathcal T_f$ to $\mathcal T_g$ is inverse shadowing for methods
generated by these perturbations
\cite{CorlessPilyuginInverse,PilyuginInverseContinuous,PilyuginVolfsonTodorov}.
For the related theory of
bishadowing, including applications to delay equations, see
\cite{KloedenOmbachBishadowing,AlNayefBishadowing}.
For dissipative PDEs, related continuity questions are studied through
trajectory attractors in spaces of solutions, typically with
local-in-time topologies and without requiring uniqueness
\cite{VishikChepyzhovTrajectories, ChepyzhovVishikAttractors}.

Since the only fixed
points of $a$ are $-1,0,1$, the fixed-point set of $f$ is
\[
 \{q\in\ell^p(\mathbb{Z}):q_j\in\{-1,0,1\}\text{ for every }j\in\mathbb{Z}\}.
\]
For $p<+\infty$, such sequences have finite support, so this set is

At a fixed point $q$, the derivative is the diagonal operator with
multiplier $\lambda_1$ on the coordinates where $q_j\neq0$, and

attractor would contain all of them.  Since $|e_i-e_j|_p\geq1$ for
$i\neq j$, this contradicts compactness.

\label{def:CL-cocycle}
We say that the pair $(\alpha,A)$ satisfies \emph{cocycle generalized
hyperbolicity} if there exist constants $C>0$, $\lambda\in(0,1)$ and a
family of projections
$P_x,Q_x:\mathbb{B}\to\mathbb{B}$, $x\in\mathbb{B}$,
such that, writing
$
 E_x^s:=P_x(\mathbb{B})$,
$
 E_x^u:=Q_x(\mathbb{B})$,
for every $x\in\mathbb{B}$ the following conditions hold.
\begin{enumerate}[label=\textup{(C-GH\arabic*)},leftmargin=15mm]
\item \textbf{Splitting.}
$
 P_x+Q_x=\operatorname{Id}$,
$ \left\lVert P_x\right\rVert,\left\lVert Q_x\right\rVert\leq C.
$
Hence, $E_x^s\oplus E_x^u=\mathbb{B}$.

\item \textbf{Inclusion.} $
 A(x)E_x^s\subseteq E_{\alpha(x)}^s$,
$ A(x)^{-1}E_{\alpha(x)}^u\subseteq E_x^u.
$ hold.

\item \textbf{Exponential estimates.} For every $n>0$,
\[
 \left|A_\alpha^n(x)v_x^s\right|
 \leq C\lambda^n\left|v_x^s\right|,
 \qquad v_x^s\in E_x^s,
\]
\[
 \left|A_\alpha^{-n}(x)v_x^u\right|
 \leq C\lambda^n\left|v_x^u\right|,
 \qquad v_x^u\in E_x^u.
\]
\end{enumerate}
As in Definition \ref{def:GH}, no finite-dimensionality assumption is imposed on $E_x^u$.
No continuity, uniform continuity, or measurability of the maps
$x\mapsto P_x,Q_x$ is assumed.

 \left\lVert v\right\rVert_\infty\leq K_c\left\lVert w\right\rVert_\infty, \qquad v(\alpha(x))-A(x)v(x)=w(x),\qquad x\in\mathbb{B}.

\label{thm:GH-shadowing}
Let $f\in\mathfrak{D}(\mathbb{B})$ satisfy generalized hyperbolicity in the sense of
Definition~\ref{def:GH}.  There exist constants $M,d_0>0$ such that,
for every $0<d<d_0$, the following hold.
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=10mm]
\item \textbf{Lipschitz shadowing.}
For every $d$-pseudotrajectory $(y_k)_{k\in\mathbb{Z}}$ of $f$, there exists
$x\in\mathbb{B}$ such that
\[
 \sup_{k\in\mathbb{Z}}\left|y_k-f^k(x)\right|\leq Md.

\[
 f^m(x)=x,
 \qquad
 \sup_{k\in\mathbb{Z}}\left|y_k-f^k(x)\right|\leq Md.
\]
\end{enumerate}

\label{lem:sequence-GH-robustness}
Assume that $\mathcal A=(A_k)_{k\in\mathbb{Z}}$ satisfies generalized
hyperbolicity with constants $C>0$, $\lambda\in(0,1)$ and
$
 \left\lVert A_k\right\rVert,\left\lVert A_k^{-1}\right\rVert\leq R$.
For every $\lambda'\in(\lambda,1)$ there exist $C_1>0$ and
$\varepsilon>0$, depending only on $C,\lambda,\lambda',R$, such that
the following hold.
\begin{enumerate}[label=\textup{(\roman*)},ref=(\roman*),leftmargin=10mm]
\item \label{item:sequence-robustness-general}
Every sequence $\mathcal B=(B_k)_{k\in\mathbb{Z}}$ of bounded linear
operators satisfying
\begin{equation}
\label{eq:sequence-robustness-close}
 \left\lVert B_k-A_k\right\rVert<\varepsilon,\qquad k\in\mathbb{Z},
\end{equation}
satisfies generalized hyperbolicity with constants $C_1,\lambda'$.
\item \label{item:sequence-robustness-periodic}
If, in addition, $\mathcal A$ is $m$-periodic and satisfies periodic
generalized hyperbolicity, then every $m$-periodic sequence $\mathcal B$
satisfying \eqref{eq:sequence-robustness-close} satisfies periodic
generalized hyperbolicity with the same constants $C_1,\lambda'$.
\end{enumerate}
The constants are independent of the sequences and of the period $m$.

\label{def:lsc}
Let $X,Y$ be topological spaces.  A set-valued map
$\Theta:X\to2^Y$ is \emph{lower semicontinuous} if, for every open set
$U\subset Y$, the set
$
 \{x\in X:\Theta(x)\cap U\neq\varnothing\}
$
is open in $X$.

<!-- thm-7: body_md -->
\label{thm:Michael}
Let $X$ be a paracompact topological space and let $Y$ be a Banach space.
Suppose that $\Theta:X\to2^Y$ is lower semicontinuous and for every
$x\in X$, the set $\Theta(x)$ is a nonempty closed convex subset of $Y$.
Then $\Theta$ admits a continuous selection: there exists a continuous map
$a:X\to Y$ such that
$a(x)\in\Theta(x)$ for $x\in X$.

\label{lem:strict}
Let $X$ be a topological space, let $Y,Z$ be Banach spaces, let
$N\in\mathbb N$, and let
$V_k:X\times Y\longrightarrow Z$, $k=0,\ldots,N$,
be continuous maps such that $y\mapsto V_k(x,y)$ is affine for every
$x\in X$.  Let $M>0$ and $\alpha>0$.  Assume that, for every $x\in X$,
there exists $y^*=y^*(x)\in Y$ satisfying
\begin{equation}
\label{eq:strict-feasible}
 \left\lVert V_k(x,y^*)\right\rVert\leq M-\alpha,
 \qquad k=0,\ldots,N.
\end{equation}
We refer to \eqref{eq:strict-feasible} as a \emph{uniform Slater-type
condition}: one point satisfies every constraint with the same positive
margin $\alpha$.
Define
\begin{equation*}
 \Theta(x):=
 \{y\in Y:\left\lVert V_k(x,y)\right\rVert\leq M
       \text{ for }k=0,\ldots,N\}.
\end{equation*}
Then every $\Theta(x)$ is nonempty, closed and convex, and the map
$\Theta:X\to2^Y$ is lower semicontinuous.
<!-- end lem-3 -->
<!-- proof-6: body_md -->
Nonemptiness follows from \eqref{eq:strict-feasible}.  For a fixed $x$,
each set
$\{y\in Y:\left\lVert V_k(x,y)\right\rVert\leq M\}$
is closed because $V_k(x,\cdot)$ is continuous, and it is convex because
$V_k(x,\cdot)$ is affine and the closed ball of radius $M$ in $Z$ is
convex.  Their finite intersection $\Theta(x)$ is therefore closed and
convex.

It remains to prove lower semicontinuity.  Fix $x_0\in X$ and an open set
$U\subset Y$ such that $\Theta(x_0)\cap U\neq\varnothing$.  Choose
$y_0\in\Theta(x_0)\cap U$.  By the uniform Slater-type condition,
there is a strictly feasible point $y^*$ at $x_0$ satisfying
\eqref{eq:strict-feasible}.  For
$t\in(0,1)$ put
$y_t:=ty_0+(1-t)y^*$.
For $t$ sufficiently close to $1$, one has $y_t\in U$.  By affinity,
\[
 V_k(x_0,y_t)
 =tV_k(x_0,y_0)+(1-t)V_k(x_0,y^*),
\]
and hence
\begin{align*}
 \left\lVert V_k(x_0,y_t)\right\rVert
 \leq tM+(1-t)(M-\alpha)
 =M-(1-t)\alpha<M,
 \qquad k=0,\ldots,N.
\end{align*}
Thus all the defining inequalities are strict at $(x_0,y_t)$.  Since
there are only finitely many of them and all $V_k$ are continuous, there
exists a neighborhood $W$ of $x_0$ such that
\[
 \left\lVert V_k(x,y_t)\right\rVert<M,
 \qquad x\in W,
 \quad k=0,\ldots,N.
\]
Consequently $y_t\in\Theta(x)\cap U$ for every $x\in W$.  By
Definition~\ref{def:lsc}, $\Theta$ is lower semicontinuous.
<!-- end proof-6 -->
<!-- remark-3: body_md -->
The finiteness of the horizon is essential in this argument: continuity
provides a common neighborhood for finitely many strict inequalities.  The
same proof does not directly give lower semicontinuity for a fiber defined
by infinitely many inequalities.

\label{lem:successive-corrections}
Let $\mathcal X,\mathcal Y$ be Banach spaces and let
$\mathcal L:\mathcal X\to\mathcal Y$ be a bounded linear operator.  Assume
that there exists $K>0$ such that, for every $w\in\mathcal Y$, one can choose
$q\in\mathcal X$ satisfying
\begin{equation}
\label{eq:abstract-preimage}
 \mathcal Lq=w,
 \qquad
 \left\lVert q\right\rVert\leq K\left\lVert w\right\rVert.
\end{equation}
Let $\rho>0$, and let
$\mathcal N:\{u\in\mathcal X:\left\lVert u\right\rVert\leq\rho\}\longrightarrow\mathcal Y$
be Lipschitz with constant $\mu$.  If
\begin{equation}
\label{eq:successive-smallness}
 \theta:=K\mu<1,
 \qquad
 \frac{K\left\lVert \mathcal N(0)\right\rVert}{1-\theta}\leq\rho,
\end{equation}
then there exists $u\in\mathcal X$ such that
 $\mathcal Lu=\mathcal N(u)$,
 $\left\lVert u\right\rVert\leq \frac{K\left\lVert \mathcal N(0)\right\rVert}{1-\theta}\leq\rho$.
No linear or continuous choice of the preimages in
\eqref{eq:abstract-preimage} is required.
<!-- end lem-4 -->
<!-- proof-7: body_md -->
Set $u_0:=0$.  Choose $q_0\in\mathcal X$ according to
\eqref{eq:abstract-preimage} with $\mathcal Lq_0=\mathcal N(u_0)$,
and put $u_1:=q_0$.  For $j\geq1$, once $u_{j-1}$ and $u_j$ have
been defined and lie in the ball of radius $\rho$, choose
$q_j\in\mathcal X$ according to \eqref{eq:abstract-preimage} with
\begin{equation}
\label{eq:abstract-correction}
 \mathcal Lq_j=\mathcal N(u_j)-\mathcal N(u_{j-1}),
\end{equation}
and put $u_{j+1}:=u_j+q_j$.
We now show that the construction continues for all $j\geq0$ and that
\begin{equation}
\label{eq:abstract-geometric}
 \left\lVert q_j\right\rVert\leq K\left\lVert \mathcal N(0)\right\rVert\theta^j,
 \qquad
 \left\lVert u_{j+1}\right\rVert
 \leq K\left\lVert \mathcal N(0)\right\rVert\sum_{i=0}^{j}\theta^i
 \leq\rho.
\end{equation}
The case $j=0$ follows from the choice of $q_0$ and
\eqref{eq:successive-smallness}.  Suppose the estimates hold through
$j-1$.  Then $u_{j-1}$ and $u_j$ lie in the ball of radius $\rho$,
so $q_j$ and $u_{j+1}$ can be defined.  By the choice of $q_j$,
the Lipschitz estimate, and $u_j-u_{j-1}=q_{j-1}$,
\[
 \left\lVert q_j\right\rVert
 \leq K\left\lVert \mathcal N(u_j)-\mathcal N(u_{j-1})\right\rVert
 \leq K\mu\left\lVert q_{j-1}\right\rVert
 \leq K\left\lVert \mathcal N(0)\right\rVert\theta^j.
\]
Since $u_{j+1}=\sum_{i=0}^{j}q_i$, summing these bounds and using
\eqref{eq:successive-smallness} proves the remaining inequalities in
\eqref{eq:abstract-geometric}.  Thus every step is well-defined.
Moreover, induction using \eqref{eq:abstract-correction} gives
\begin{equation}
\label{eq:abstract-approximation}
 \mathcal Lu_{j+1}=\mathcal N(u_j),
 \qquad j\geq0.
\end{equation}

The series $\sum_{j\geq0}q_j$
converges to some $u\in\mathcal X$, and
$
 \left\lVert u\right\rVert\leq\frac{K\left\lVert \mathcal N(0)\right\rVert}{1-\theta}\leq\rho.
$
Since $\mathcal L$ is bounded and $\mathcal N$ is Lipschitz, passing to
the limit in \eqref{eq:abstract-approximation} gives
$\mathcal Lu=\mathcal N(u)$.

\label{lem:cocycle-SBS}
Assume that $(\alpha,A)$ satisfies cocycle generalized hyperbolicity with
constants $C>0$ and $\lambda\in(0,1)$.  There exists
$L_c=L_c(C,\lambda)>0$ such that, for every $x\in\mathbb{B}$, every finite or
infinite integer interval $I$, and every bounded sequence
$(w_{k+1})_{k\in\widetilde I}$, where
$\widetilde I:=\{k\in\mathbb{Z}:k,k+1\in I\}$, there exists a sequence
$(v_k)_{k\in I}$ satisfying $ \sup_{k\in I}\left|v_k\right|
 \leq L_c\sup_{k\in\widetilde I}\left|w_{k+1}\right|$ and
\[
 v_{k+1}=A(\alpha^k(x))v_k+w_{k+1},
 \qquad k\in\widetilde I.
\]
<!-- end lem-5 -->
<!-- remark-4: body_md -->
The bounded-solvability approach goes back to Perron \cite{Perron}
and the classical work of Maizel and Pliss
\cite{Maizel,PlissBounded}; see Coppel \cite{Coppel} for a broader
treatment. For nonautonomous evolution equations,
the relation between exponential dichotomy and bounded mild
solutions is studied in \cite{LatushkinRandolphSchnaubelt}.
The proof below uses the Perron construction in
\cite[proof of Lemma~31]{TikhomirovGH}.
<!-- end remark-4 -->
<!-- proof-8: body_md -->
Fix $x\in\mathbb{B}$ and first take $I=\mathbb{Z}$.  Put $x_j:=\alpha^j(x)$ and
define $v_k$ by the Perron sums
\[
 v_k:=
 \sum_{j\leq k}A_\alpha^{k-j}(x_j)P_{x_j}w_j
 -
 \sum_{j>k}A_\alpha^{k-j}(x_j)Q_{x_j}w_j,
 \qquad k\in\mathbb{Z}.
\]
The convergence of these sums and the identity
$v_{k+1}=A(x_k)v_k+w_{k+1}$ follow exactly as in
\cite[Lemma~31]{TikhomirovGH}, with derivative iterates
replaced by cocycle iterates.  The projection bounds and exponential
estimates give
\[
 \sup_{k\in\mathbb{Z}}\left|v_k\right|
 \leq\left(C+2C^2\sum_{n=1}^{\infty}\lambda^n\right)
       \sup_{j\in\mathbb{Z}}\left|w_j\right|
 =\left(C+\frac{2C^2\lambda}{1-\lambda}\right)
       \sup_{j\in\mathbb{Z}}\left|w_j\right|.
\]
Thus we may take $L_c:=C+2C^2\lambda/(1-\lambda)$.

For a general interval $I$, extend the forcing by setting
$w_{k+1}:=0$ for $k\notin\widetilde I$ and restrict the resulting
solution to $I$.  Neither operation increases the supremum norm,
so the same constant works independently of $x$ and $I$.

\label{lem:finite-cont}
Assume that $(\alpha,A)$ satisfies cocycle generalized hyperbolicity with
constants $C,\lambda$, and let $L_c$ be as in
Lemma~\ref{lem:cocycle-SBS}.  For every integer $N\geq1$ and every
collection
$
 w_1,\ldots,w_N\in X
$
there exist $v_0,\ldots,v_N\in X$ such that $ \max_{0\leq k\leq N}\left\lVert v_k\right\rVert_\infty
 \leq 2L_c\max_{1\leq j\leq N}\left\lVert w_j\right\rVert_\infty$ and
\begin{equation*}
 v_{k+1}=T_{\alpha,A}v_k+w_{k+1},
 \qquad k=0,\ldots,N-1.
\end{equation*}

The constant $L_c$ is independent of $N$ and of the functions $w_j$.

<!-- proof-9: body_md -->
Fix $N\geq1$ and $w_1,\ldots,w_N\in X$.  Put
$
 m:=\max_{1\leq j\leq N}\left\lVert w_j\right\rVert_\infty.
$
The case $m=0$ is immediate, so after rescaling assume that $m=1$.
For $x,y\in\mathbb{B}$, define
$
 V_0(x,y):=y
$
and, recursively,
\begin{equation}
\label{eq:cocycle-Vk}
 V_{k+1}(x,y)
 :=A(\alpha^k(x))V_k(x,y)
   +w_{k+1}(\alpha^{k+1}(x)),
 \qquad k=0,\ldots,N-1.
\end{equation}
Every $V_k:\mathbb{B}\times\mathbb{B}\to\mathbb{B}$ is continuous and is affine in the second
variable.

$I=\{0,\ldots,N\}$, provides an initial vector $y^*(x)$ such that
\[
 \left|V_k(x,y^*(x))\right|\leq L_c,
 \qquad k=0,\ldots,N.
\]
Define
\[
 \Theta_N(x):=
 \{y\in\mathbb{B}:\left|V_k(x,y)\right|\leq2L_c
        \text{ \; for \; }k=0,\ldots,N\}.
\]
The preceding estimate gives strict feasibility with margin $L_c$.
Lemma~\ref{lem:strict} and Theorem~\ref{thm:Michael} therefore give a
continuous selection
$a_N:\mathbb{B}\longrightarrow\mathbb{B}$ with $a_N(x)\in\Theta_N(x)$.
For $k=0,\ldots,N$, define $v_k\in X$ by
 $v_k(\alpha^k(x)):=V_k(x,a_N(x))$.
This is well-defined and continuous because $\alpha$ is a homeomorphism,

$
 \max_{0\leq k\leq N}\left\lVert v_k\right\rVert_\infty\leq2L_c.
$

\[
 v_{k+1}=T_{\alpha,A}v_k+w_{k+1},
 \qquad k=0,\ldots,N-1.
\]
Undoing the normalization gives the finite-horizon bound with right-hand

\label{rem:partition-of-unity}
The continuous selection $a_N$ can also be obtained directly from a
partition of unity, without Theorem~\ref{thm:Michael}.  For $z\in\mathbb{B}$, the set
\[
 W_z:=\{x\in\mathbb{B}:\left|V_k(x,y^*(z))\right|<2L_c\text{ for }k=0,\ldots,N\}
\]
is open, because only finitely many continuous maps are involved, and it
contains $z$, since $\left|V_k(z,y^*(z))\right|\leq L_c$.  Since $\mathbb{B}$ is a metric
space, hence paracompact, there is a partition of unity \cite{Engelking}
$\{\varphi_\gamma\}_{\gamma\in\Gamma}$ subordinate to the open cover
$\{W_z\}_{z\in\mathbb{B}}$: the supports of the functions $\varphi_\gamma$ form a
locally finite family, and
$\operatorname{supp}\varphi_\gamma\subset W_{z_\gamma}$ for some
$z_\gamma\in\mathbb{B}$.  The index set $\Gamma$ is arbitrary and may be
uncountable.  Put
\[
 a_N(x):=\sum_{\gamma\in\Gamma}\varphi_\gamma(x)\,y^*(z_\gamma),
 \qquad x\in\mathbb{B}.
\]
Near each point only finitely many terms of the sum are nonzero, so $a_N$
is well defined and continuous.  If $\varphi_\gamma(x)\neq0$, then
$x\in W_{z_\gamma}$.  Since every $V_k(x,\cdot)$ is affine,
\[
 \left|V_k(x,a_N(x))\right|
 =\left|\sum_{\gamma\in\Gamma}\varphi_\gamma(x)V_k(x,y^*(z_\gamma))\right|
 <2L_c,
 \qquad k=0,\ldots,N,
\]
and hence $a_N(x)\in\Theta_N(x)$.  
This argument is similar to the first step of the proof of
Theorem~\ref{thm:Michael}, in which a partition of unity gives approximate
selections.

\label{lem:Birkhoff}
Let $X$ be a Banach space and let $T:X\to X$ be a bounded linear operator.
Assume that there exists $M_0>0$ such that, for every $w\in X$ and every
integer $N\geq1$, one can find $v_0,\ldots,v_N\in X$ satisfying
\begin{equation}
\label{eq:abstract-rec}
 v_{k+1}=Tv_k+w,
 \qquad k=0,\ldots,N-1,
\end{equation}
and
$
 \max_{0\leq k\leq N}\left\lVert v_k\right\rVert
 \leq M_0\left\lVert w\right\rVert.
$
Then, for every $w\in X$, there exists $u\in X$ such that
\begin{equation*}
 u-Tu=w,
 \qquad
 \left\lVert u\right\rVert\leq2M_0\left\lVert w\right\rVert.
\end{equation*}
No linear or continuous dependence of $u$ on $w$ is asserted.
<!-- end lem-7 -->
<!-- proof-11: body_md -->
Choose once and for all an integer $N>4M_0$, and put
$\kappa:=\frac{2M_0}{N}<\frac12$.
Set $w_0:=w$, apply the hypothesis to $w_0$, and choose
$v_0^{(0)},\ldots,v_N^{(0)}$.  Define the Birkhoff average
$
 u_0:=\frac1N\sum_{k=0}^{N-1}v_k^{(0)}.
$
Using \eqref{eq:abstract-rec},
\begin{align*}
 u_0-Tu_0
 =\frac1N\left(
     \sum_{k=0}^{N-1}v_k^{(0)}
     -\sum_{k=0}^{N-1}(v_{k+1}^{(0)}-w_0)
   \right)
 =w_0+\frac1N\bigl(v_0^{(0)}-v_N^{(0)}\bigr).
\end{align*}
Set
 $w_1:=\frac1N\bigl(v_N^{(0)}-v_0^{(0)}\bigr)$.
Then
\[
 u_0-Tu_0=w_0-w_1,
 \qquad
 \left\lVert u_0\right\rVert\leq M_0\left\lVert w_0\right\rVert,
 \qquad
 \left\lVert w_1\right\rVert\leq\kappa\left\lVert w_0\right\rVert.
\]

We now define the remaining terms inductively, keeping the same $N$.
Suppose that $w_j$ has been defined for some $j\geq1$.  Apply the
hypothesis of Lemma \ref{lem:Birkhoff} with forcing $w_j$ to choose
$v_0^{(j)},\ldots,v_N^{(j)}\in X$ satisfying
\[
 v_{k+1}^{(j)}=Tv_k^{(j)}+w_j,
 \qquad k=0,\ldots,N-1,
 \qquad
 \max_{0\leq k\leq N}\left\lVert v_k^{(j)}\right\rVert\leq M_0\left\lVert w_j\right\rVert.
\]
Define
\[
 u_j:=\frac1N\sum_{k=0}^{N-1}v_k^{(j)},
 \qquad
 w_{j+1}:=\frac1N\bigl(v_N^{(j)}-v_0^{(j)}\bigr).
\]
Thus $u_j$ is the correction obtained by averaging the finite
recurrence solution with forcing $w_j$, and $w_{j+1}$ is its
remaining error.  The same calculation as for $j=0$ gives
\begin{equation}
\label{eq:telescoping}
 u_j-Tu_j=w_j-w_{j+1},
 \qquad j\geq0.
\end{equation}
Moreover, for every $j\geq0$,
$$
 \left\lVert u_j\right\rVert
 \leq\frac1N\sum_{k=0}^{N-1}\left\lVert v_k^{(j)}\right\rVert
 \leq M_0\left\lVert w_j\right\rVert,\qquad
 \left\lVert w_{j+1}\right\rVert
 \leq\frac{\left\lVert v_N^{(j)}\right\rVert+\left\lVert v_0^{(j)}\right\rVert}{N}
 \leq\kappa\left\lVert w_j\right\rVert.
$$
Starting from $w_0=w$, induction therefore gives
\[
 \left\lVert w_j\right\rVert\leq\kappa^j\left\lVert w\right\rVert,
 \qquad
 \left\lVert u_j\right\rVert\leq M_0\kappa^j\left\lVert w\right\rVert,
 \qquad j\geq0.
\]
The series
$u:=\sum_{j=0}^{\infty}u_j$
converges in $X$, and
\[
 \left\lVert u\right\rVert
 \leq \frac{M_0}{1-\kappa}\left\lVert w\right\rVert
 \leq2M_0\left\lVert w\right\rVert.
\]
For every integer $m\geq0$, summing \eqref{eq:telescoping} gives
\[
 (I-T)\sum_{j=0}^{m}u_j
 =\sum_{j=0}^{m}(w_j-w_{j+1})
 =w-w_{m+1}.
\]
Since $\sum_{j=0}^{m}u_j\to u$, $w_{m+1}\to0$, and $I-T$ is bounded,
passing to the limit gives $(I-T)u=w$.

Apply Lemma~\ref{lem:finite-cont} with

\label{thm:g-Df-GH}
Assume that $f\in\mathfrak{D}(\mathbb{B})$ satisfies generalized hyperbolicity with
constants $C>0$ and $\lambda\in(0,1)$.  For every
$\lambda'\in(\lambda,1)$ there exist constants $C_1>0$ and $d_1>0$
such that the following holds.  If $g:\mathbb{B}\to\mathbb{B}$ is a homeomorphism
satisfying
\begin{equation}
\label{eq:g-f-cocycle-small}
 \sup_{x\in\mathbb{B}}\left|g(x)-f(x)\right|<d_1,
\end{equation}
then the pair $(g,Df)$ satisfies cocycle generalized hyperbolicity with
constants $C_1,\lambda'$.
<!-- end thm-8 -->
<!-- proof-14: body_md -->
We use the shadowing statements in Theorem~\ref{thm:GH-shadowing}
and the sequence robustness statement in
Lemma~\ref{lem:sequence-GH-robustness}.

Let \(R\) be as in \eqref{eq:C1-2}. Fix $\lambda'\in(\lambda,1)$ and take $\varepsilon,C_1>0$ from
Lemma~\ref{lem:sequence-GH-robustness} for $C,\lambda,\lambda',R$.
Let $M,d_0>0$ be the constants from Theorem~\ref{thm:GH-shadowing}.
By \eqref{eq:C1-r2}, we may choose $d_1\in(0,d_0)$ so small that
\begin{equation}
\label{eq:cocycle-r-small}
 r(Md_1)<\varepsilon.
\end{equation}

Let $g:\mathbb{B}\to\mathbb{B}$ be a homeomorphism satisfying
\eqref{eq:g-f-cocycle-small}.  Consider the
partition of $\mathbb{B}$ into orbits of $g$ and choose one point $y_0$ on each
orbit.  Put
$y_k:=g^k(y_0)$ for $k\in\mathbb{Z}$.
Then
\[
 \left|y_{k+1}-f(y_k)\right|
 =\left|g(y_k)-f(y_k)\right|<d_1,
 \qquad k\in\mathbb{Z},
\]
so $\{y_k\}$ is a $d_1$-pseudotrajectory of $f$.
Lipschitz shadowing gives an exact trajectory $\{x_k\}_{k\in\mathbb{Z}}$
of $f$ such that
\begin{equation}
\label{eq:cocycle-shadowing}
 \left|x_k-y_k\right|\leq Md_1,
 \qquad k\in\mathbb{Z}.
\end{equation}
If the orbit of $y_0$ has least period $m$, use periodic shadowing to
choose $\{x_k\}$ to be $m$-periodic.

Set
$
 A_k:=Df(x_k)$,
 $B_k:=Df(y_k)$.
The sequence $\{A_k\}$ satisfies generalized hyperbolicity with constants
$C,\lambda$, using the spaces $E_{x_k}^s,E_{x_k}^u$.  On the other hand,
\eqref{eq:C1-r1}, \eqref{eq:cocycle-shadowing}, and
\eqref{eq:cocycle-r-small} give
\[
 \left\lVert A_k-B_k\right\rVert
 \leq r(\left|x_k-y_k\right|)
 \leq r(Md_1)<\varepsilon.
\]

\label{lem:product-remainders}
\begin{enumerate}[label=\textup{(\roman*)},ref=(\roman*),leftmargin=10mm]
\item\label{item:product-remainders-bounds}
\textbf{Bounds and Lipschitz estimates.}
The operators $G$ and $R$ map $\mathcal U_\rho$ into
$\mathcal W_\varepsilon$.  They satisfy
\begin{equation}
 \label{eq:product-Gf}
 |G(h)|_\infty\leq d+\frac M2\rho^2=\varepsilon,
 \qquad
 |G(h)-G(\widetilde h)|_\infty \leq(M\rho+d)|h-\widetilde h|_\infty,
\end{equation}
\begin{equation}
\label{eq:product-R-field-estimates}
 |R(k)|_\infty\leq\varepsilon,
 \qquad
 |R(k)-R(\widetilde k)|_\infty
 \leq M\rho|k-\widetilde k|_\infty.
\end{equation}

\item\label{item:product-remainders-cancellation}
\textbf{Cancellation estimates.}
For arbitrary $h,k\in\mathcal U_\rho$, put
$H=\operatorname{Id}+h$ and $K=\operatorname{Id}+k$.  Then
\begin{equation}
\label{eq:product-remainder-KH}
 |G(h)+R(k)\circ H|_\infty
 \leq M\rho|h+k\circ H|_\infty,
\end{equation}
\begin{equation}
\label{eq:product-remainder-HK}
 |R(k)+G(h)\circ K|_\infty
 \leq(M\rho+d)|k+h\circ K|_\infty.
\end{equation}
These two estimates do not require either semiconjugacy identity.
\end{enumerate}

\eqref{eq:product-s-Lip-v} and \eqref{eq:product-Delta-estimates}
give \eqref{eq:product-Gf}.
To prove \eqref{eq:product-R-field-estimates}, we estimate
$s_f(y+v,-v)$ directly.  For a scalar base point $t$, differentiation
gives
\begin{equation*}
 \frac{d}{dq}\bigl(a(t)-a(t+q)+a'(t+q)q\bigr)
 =a''(t+q)q.
\end{equation*}

\begin{gathered}
 |s_f(y+v,-v)-s_f(y+\widetilde v,-\widetilde v)|_p
 \leq M\rho|v-\widetilde v|_p,\\
 |v|_p,|\widetilde v|_p\leq\rho.
\end{gathered}
\end{equation}
For $k\in\mathcal U_\rho$, the defining formulas give
\begin{equation}\label{eq:Rk-F}

The Taylor bound $|s_f(y+v,-v)|_p\leq M|v|_p^2/2$ and

 |R(k)(y)|_p
 \leq \frac M2|k(y)|_p^2+d
 \leq \frac M2\rho^2+d=\varepsilon.

 |R(k)(y)-R(\widetilde k)(y)|_p
 \leq M\rho|k(y)-\widetilde k(y)|_p.

\label{rem:product-uniform-conjugacy}
Let $BUC(\mathbb{B},\mathbb{B})$ denote the closed subspace of $C_b^0(\mathbb{B},\mathbb{B})$
consisting of bounded uniformly continuous fields.  The corrections in
Theorem~\ref{thm:product-structural-stability} can be chosen with
$h,k\in BUC(\mathbb{B},\mathbb{B})$, with the same bounds.  Thus both $H$ and
$H^{-1}$ are uniformly continuous.
Indeed, the maps $f^{\pm1}$ and $g^{\pm1}$ are Lipschitz; in particular,
the argument in Subsection~\ref{sec:product-cohomological-operators}
gives $\operatorname{Lip}(g^{-1})\leq(\lambda_1-d)^{-1}$.

and Lipschitz.  Hence, if $u,w\in BUC(\mathbb{B},\mathbb{B})$ and $u$ is admissible
for $\alpha\in\{f,g\}$, every finite partial sum of
$\mathcal S_\alpha(u)w$ is uniformly continuous: it uses finitely many
Lipschitz iterates of $\alpha^{\pm1}$ and the uniformly continuous map
$\operatorname{Id}+u$.  Uniform convergence of the Green series gives
$\mathcal S_\alpha(u)w\in BUC(\mathbb{B},\mathbb{B})$.
The operators $G$ and $R$ also preserve bounded uniform continuity
on $\mathcal U_\rho$.  Starting from zero, the iterations defining
$h=O_fG(h)$, $O_gw$, and $k=O_gR(k)$ therefore remain in this closed
subspace and converge uniformly, which gives the asserted regularity
with the same bounds.

An analogous upgrade in the general part of the paper requires
additional reasoning. Continuous selections, whether from Michael's theorem or from a partition of unity, come without a uniform modulus of continuity, so the subsequent
uniform-limit arguments do not by themselves yield semiconjugacies in
$\operatorname{Id}+BUC(\mathbb{B},\mathbb{B})$.

$\operatorname{Lip}(a^{-1})\leq \Lambda$.  Therefore
\begin{equation}
\label{eq:product-backward-error}
 |t_k-a^{-1}(t_{k+1})|
 \leq \Lambda|a(t_k)-t_{k+1}|
 <\Lambda\delta_*.
\end{equation}
If $1-\eta(t_r)\neq0$, then $|t_r|<c-4\gamma$.
The definition of $\alpha_u$, \eqref{eq:product-delta-margin}, and
backward induction give \eqref{eq:product-unstable-margin}.
This proves item~\ref{item:product-scalar-margins}.

Keep the cutoff and margins from Lemma~\ref{lem:product-scalar-trapping},
and recall that $\Lambda=\lambda_1^{-1}$ from \eqref{eq:product-R}.
We prove the subsegment estimates by comparing finite pseudotrajectory
blocks with exact orbits and then concatenating blocks.

\medskip
\noindent\textbf{Step 1: forward products.}
Choose $N\geq1$ so large that
\begin{equation}
\label{eq:product-N}
 C\lambda^N<\frac14.
\end{equation}
Consider a scalar pseudotrajectory segment
$t_0,t_1,\ldots,t_N$ contained in the stable region
$|t_i|>c+2\gamma$.  Let $u_i=a^i(t_0)$.  From
$|a'|\leq \Lambda$ and \eqref{eq:product-scalar-pseudo}, induction gives
\begin{equation}
\label{eq:product-finite-comparison}
 |t_i-u_i|
 \leq\delta_*\sum_{q=0}^{i-1}\Lambda^q,
 \qquad 0\leq i\leq N.
\end{equation}
For arbitrary numbers $\alpha_i,\beta_i$, one has the product telescoping
identity
\begin{equation}
\label{eq:product-product-telescope}
 \prod_{i=0}^{N-1}\alpha_i-
 \prod_{i=0}^{N-1}\beta_i
 =\sum_{q=0}^{N-1}
 \left(\prod_{i<q}\alpha_i\right)
 (\alpha_q-\beta_q)
 \left(\prod_{i>q}\beta_i\right).
\end{equation}
Apply this with $\alpha_i=a'(t_i)$ and $\beta_i=a'(u_i)$.  Using
\eqref{eq:product-a-bounds}, \eqref{eq:product-finite-comparison}, and
$|a''|\leq M$, we obtain
\begin{equation*}
 \left|
 \prod_{i=0}^{N-1}a'(t_i)-
 \prod_{i=0}^{N-1}a'(u_i)
 \right|
 \leq
 M \Lambda^{N-1}N\left(\sum_{q=0}^{N-1}\Lambda^q\right)\delta_*.
\end{equation*}
After decreasing $\delta_*$, the right-hand side is less than $1/4$.
Because $|t_0|>c$, \eqref{eq:product-scalar-estimates} and
\eqref{eq:product-N} imply
$
 \prod_{i=0}^{N-1}a'(u_i)=|(a^N)'(t_0)|<\frac14.
$
Thus every stable pseudotrajectory block of length $N$ has a derivative
product of at most $1/2$.

\medskip
\noindent\textbf{Step 2: backward products.}
Use the same argument with $a^{-1}$ and the factors
 $b(t)=\frac1{a'(t)}$.
By \eqref{eq:product-a-bounds},
\begin{equation}
\label{eq:product-b-bounds}
 |b(t)|\leq \Lambda,
 \qquad
 |b'(t)|=\frac{|a''(t)|}{|a'(t)|^2}
 \leq M\lambda_1^{-2}.
\end{equation}
For a backward pseudotrajectory block $t_{r-N},\ldots,t_r$ contained
in the region $|t|<c-2\gamma$, put $u_{r-q}=a^{-q}(t_r)$, $0\leq q\leq N$.
By \eqref{eq:product-backward-error}, the one-step error is bounded
by $\Lambda\delta_*$, so
\begin{equation*}
 |t_{r-q}-u_{r-q}|
 \leq \Lambda\delta_*\sum_{i=0}^{q-1}\Lambda^i,
 \qquad 1\leq q\leq N.
\end{equation*}
The product telescoping identity \eqref{eq:product-product-telescope}
and \eqref{eq:product-b-bounds} therefore give
\begin{equation*}
 \left|\prod_{k=r-N}^{r-1}b(t_k)
 -\prod_{k=r-N}^{r-1}b(u_k)\right|
 \leq C_N\delta_*,
\end{equation*}
where $C_N$ depends only on the scalar bounds and the chosen block
length $N$.  The exact product satisfies
$ \prod_{k=r-N}^{r-1}b(u_k)
 =|(a^{-N})'(t_r)|
 <\frac14
$
by \eqref{eq:product-scalar-estimates} and \eqref{eq:product-N}.
Decreasing $\delta_*$ so that $C_N\delta_*<1/4$ gives the same
bound $1/2$ for every backward block of length $N$.

\medskip
\noindent\textbf{Step 3: concatenation of blocks.}
Split either type of segment of length $m$ into blocks of length $N$
and a remainder of length less than $N$.  Each full-block product is
at most $1/2$, and each factor in the remainder is at most $\Lambda$.
Thus each product, in its respective region, is bounded by
\[
 \Lambda^{N-1}2^{-\lfloor m/N\rfloor}
 \leq 2\Lambda^{N-1}\bigl(2^{-1/N}\bigr)^m.
\]
We may therefore take
$\theta=2^{-1/N}$ and $C_0=2\Lambda^{N-1}$.
This gives \eqref{eq:product-stable-subsegment} and
\eqref{eq:product-unstable-subsegment}, completing the proof.

\medskip
\noindent\textbf{Step 1: coefficient bounds.}
Let $\widehat C_0$ be the subsegment constant supplied by
Lemma~\ref{lem:product-scalar-pseudotrajectories} before any further
increase, and put $\Lambda=\lambda_1^{-1}$.  If $\eta(t_r)\neq0$, the forward
margin in Lemma~\ref{lem:product-scalar-trapping} and
\eqref{eq:product-stable-subsegment} give
\[
 |b^s_{n,r}(t)|\leq\widehat C_0\theta^{n-r-1},\qquad r<n.
\]
If $1-\eta(t_r)\neq0$, the backward margin and
\eqref{eq:product-unstable-subsegment} control the product from $n$ to
$r-1$.  The last factor $1/a'(t_r)$ is at most $\Lambda$, so
\[
 |b^u_{n,r}(t)|\leq \Lambda\widehat C_0\theta^{r-n},\qquad n\leq r.
\]
When the corresponding cutoff vanishes, the coefficient is zero.
Thus replacing $C_0$ by
$\max\{\widehat C_0,\Lambda\widehat C_0/\theta\}$ proves
\eqref{eq:product-stable-coefficient} and
\eqref{eq:product-unstable-coefficient}, including the empty-product
cases.  The subsegment estimates remain valid with this larger constant.

\medskip
\noindent\textbf{Step 2: Lipschitz dependence.}
Choose $\rho_*\in(0, \beta/2)$.
Let $t,\widetilde t$ satisfy the assumptions of the lemma, and put
$
 \varepsilon:=\sup_{k\in\mathbb{Z}}|t_k-\widetilde t_k|<\rho_*$,
 $L_\eta:=\operatorname{Lip}(\eta).
$
We first check that the subsegment estimates apply to both sequences
whenever at least one of the corresponding cutoff factors is nonzero.
If $\eta(t_r)\neq0$, then \eqref{eq:product-stable-margin} gives, for
$k\geq r+1$,
\[
 |\widetilde t_k|\geq |t_k|-\varepsilon
 >c+2\gamma+\beta-\varepsilon>c+2\gamma.
\]
The same argument with $t,\widetilde t$ interchanged applies if
$\eta(\widetilde t_r)\neq0$.  Similarly, if $1-\eta(t_r)\neq0$, then
\eqref{eq:product-unstable-margin} gives, for $k\leq r$,
\[
 |\widetilde t_k|\leq |t_k|+\varepsilon
 <c-2\gamma-\beta+\varepsilon<c-2\gamma,
\]
and again the roles of the two sequences can be interchanged.
Thus, in each nontrivial case, both sequences lie in the region required
by the corresponding subsegment estimate.

For integers $i\leq j$, abbreviate
\[
 \Pi^s_{i,j}(t):=\prod_{k=i}^{j-1}a'(t_k),
 \qquad
 \Pi^u_{i,j}(t):=\prod_{k=i}^{j-1}b(t_k),
 \qquad b(t)=\frac1{a'(t)}.
\]
Empty products equal $1$.  By
\eqref{eq:product-stable-subsegment} and
\eqref{eq:product-unstable-subsegment}, the relevant products along
either sequence are bounded by $C_0\theta^{j-i}$, including the case $i=j$,
since $C_0\geq1$.

\medskip
\noindent\emph{Stable coefficients.}
Fix $r<n$ and put $m=n-r-1$.  If
$\eta(t_r)=\eta(\widetilde t_r)=0$, the assertion is immediate.
Otherwise, both sequences lie in the stable region for $k\geq r+1$,
and
$$
 b^s_{n,r}(t)-b^s_{n,r}(\widetilde t)
 =\bigl(\eta(t_r)-\eta(\widetilde t_r)\bigr)
       \Pi^s_{r+1,n}(\widetilde t)
 +\eta(t_r)
       \bigl(\Pi^s_{r+1,n}(t)-\Pi^s_{r+1,n}(\widetilde t)\bigr).
$$
The first term has absolute value at most
$L_\eta C_0\theta^m\varepsilon$.
For $m\geq1$, the product telescoping identity
\eqref{eq:product-product-telescope} reads
\[
 \Pi^s_{r+1,n}(t)-\Pi^s_{r+1,n}(\widetilde t)
 =\sum_{j=r+1}^{n-1}
 \Pi^s_{r+1,j}(t)
 \bigl(a'(t_j)-a'(\widetilde t_j)\bigr)
 \Pi^s_{j+1,n}(\widetilde t).
\]
For each summand,
$$
 |\Pi^s_{r+1,j}(t)|\leq C_0\theta^{j-r-1}, \quad
 |a'(t_j)-a'(\widetilde t_j)|\leq M\varepsilon, \quad
 |\Pi^s_{j+1,n}(\widetilde t)|\leq C_0\theta^{n-j-1}.
 $$
The two products are estimated separately along their respective
pseudotrajectories.  Their lengths add up to $m-1$, so each of the
$m$ summands is bounded by $MC_0^2\theta^{m-1}\varepsilon$.
Using $0\leq\eta\leq1$, we obtain
\begin{equation*}
 |b^s_{n,r}(t)-b^s_{n,r}(\widetilde t)|
 \leq
 \bigl(L_\eta C_0\theta^m
       +mMC_0^2\theta^{\max\{m-1,0\}}\bigr)\varepsilon.
\end{equation*}
This also holds for $m=0$: the products are empty, and the difference
is simply $\eta(t_r)-\eta(\widetilde t_r)$.

\medskip
\noindent\emph{Unstable coefficients.}
The argument is analogous, with $a'$ replaced by $b=1/a'$.
For $m=r-n\geq0$, separate the final factor:
$
 b^u_{n,r}(t)=(1-\eta(t_r))b(t_r)\Pi^u_{n,r}(t).
$
In the nontrivial case, the remaining product uses only the controlled
segment from $n$ to $r$.
By \eqref{eq:product-b-bounds}, $|b|\leq\Lambda$ and
$\operatorname{Lip}(b)\leq M\Lambda^2$, so the endpoint factor satisfies
\[
 |(1-\eta)b|_\infty\leq\Lambda,\qquad
 \operatorname{Lip}((1-\eta)b)\leq\Lambda L_\eta+M\Lambda^2.
\]
The same cutoff decomposition and telescoping estimate therefore give
$$
 |b^u_{n,r}(t)-b^u_{n,r}(\widetilde t)|
 \leq\bigl((\Lambda L_\eta+M\Lambda^2)C_0\theta^m
          +mM\Lambda^3C_0^2
             \theta^{\max\{m-1,0\}}\bigr)\varepsilon.
$$
As in the stable case, this includes $m=0$, when the product is empty.

Replacing $\theta$ by any slightly larger $\theta_1\in(\theta,1)$
absorbs the polynomial factors into a constant.
After we also absorb the fixed factor $\theta_1^{-1}$ in the unstable
case, \eqref{eq:product-scalar-Green-Lip-s} and
\eqref{eq:product-scalar-Green-Lip-u} follow with a common constant $C_4$.
All constants depend only on $a$.
