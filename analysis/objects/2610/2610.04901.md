\label{Thm:SupersonicInvariance}
    Let $u_0(r),h_0(r)\in C^1([a,b])$, where $0<a<b$, satisfy the following inequalities, 

    \begin{align*}
        &u_0(r)\geq \frac{2}{\gamma-1}h_0(r), 
        \\
        &h_0(r)=\frac{\gamma-1}{4}(w_0(r)-z_0(r))>0 . 
    \end{align*} 
    Further suppose that their time evolution, $u(t,r),h(t,r)$, remain classical over 
    $\Omega(T,[a,b])$. Then, the following holds for all $(t,r)\in\Omega(T,[a,b])$, 

    \begin{align*}
        u(t,r)\geq \frac{2}{\gamma-1}h(t,r). 
    \end{align*}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Suppose by way of contradiction that there exists some $(t_1,r_1)\in\Omega(T,[a,b])$ such that $u(t_1,r_1)<\frac{2}{\gamma-1}h(t_1,r_1)$. Tracing our characteristics backward, this implies there exists some $t^*,y^*$ such that the following hold,  

    \begin{align*}
        &z\circ\psi(t^*,y^*)=0 ,
        \\
        &\left.\partial_t(z\circ\psi(t,y^*))
        \right|_{t=t^*} \leq 0 .
    \end{align*} 
    However, from the lower bound on our density, we observe the following, 

    \begin{align*}
        \left.\partial_t(z\circ\psi(t,y^*))
        \right|_{t=t^*}&=\left(\frac{\gamma-1}{4r} 
        (w^2-z^2)\right)\circ\psi(t^*,y^*)
        \\
        &= \left(\frac{\gamma-1}{4r}w^2 
        \right)\circ\psi(t^*,y^*)
        \\
        &=\left(\frac{\gamma-1}{4r}\left( 
        z+\frac{4}{\gamma-1}h\right)^2 
        \right)\circ\psi(t^*,y^*)
        \\
        &=\left(\frac{4h^2}
        {(\gamma-1)r}\right)\circ\psi(t^*,y^*)>0.  
    \end{align*} 
    This implies a contradiction.

<!-- theorem-1: body_md -->
\label{Thm:Radial_Rarefactive_Existence}

    Let $1<\gamma< 3$, and let the initial data, 
    $(h_0(r),u_0(r))\in C^1([0,\infty))$ satisfy 

    \begin{align*}
        0<\frac{2}{\gamma-1}h_0(r)\leq u_0(r)\leq M_1 
    \end{align*} 
    over any $(b,\infty)$ for any $b>0$. Further suppose that at zero, the solution satisfies the following, 

    \begin{align*}
        &\rho_0(0)=u_0(0)=0, 
    \end{align*}
    and that the gradient variables, $\alpha_0,\beta_0$ satisfy 

    \begin{align*}
        &\min_{r\in[0,\infty)}\{\alpha_0(r), 
        \beta_0(r)\}\geq 0, 
        \\
        &\max_{r\in[0,\infty)}\{\alpha_0(r), 
        \beta_0(r)\}\leq M_2, 
    \end{align*} 
    where $\alpha_0(0),\beta_0(0)$ are determined in the sense of their respective limits as $r\rightarrow0$. Then, the Euler Equations under radial symmetry admit a global $C^1$ solution satisfying, 

    \begin{align*}
        &\frac{2}{\gamma-1}h(r,t)\leq 
        u(r,t) \leq 2M_1 \ \forall (t,r) 
        \in [0,\infty)\times[0,\infty),  
        \\
        &h(r,t)>0 \ \forall r>0, 
        \\
        &\min_{t\geq0,r\geq0}\{\alpha(t,r), 
        \beta(t,r)\}\geq 0, 
        \\
        &\max_{t\geq0,r\geq0}\{\alpha(t,r), 
        \beta(t,r)\}\leq M_2.  
    \end{align*}
<!-- end theorem-1 -->
<!-- remark-1: body_md -->
From standard theory for quasilinear symmetric-hyperbolic systems, it is known that higher regularity of the initial data persists so long as the $C_b^1$ norm of the solution remains bounded. Thus, if initial data in $C_b^m$ satisfy the conditions of Theorem \ref{Thm:Radial_Rarefactive_Existence} together with additional compatibility conditions at the origin (detailed later in Assumption \ref{Ass:A1}), then the corresponding classical solution retains this regularity on every finite time interval.

\label{Thm:Refined_Upper_Bound}
    Let $h_0(r),u_0(r),\alpha_0(r),\beta_0(r)$ satisfy the assumptions in Theorem 
    \ref{Thm:Radial_Rarefactive_Existence}. Then, the following bound holds for all $t\geq 0$, 

    \begin{align*}
        0\leq \alpha(t,r),\beta(t,r)\leq 
        \frac{M_2}{1+\frac{\gamma+1}{8}M_2t}.
    \end{align*}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
We first define the following, 

    \begin{align*}
        Q(t):=\frac{M_2}{1+\frac{\gamma+1}{8}M_2t}. 
    \end{align*}
    Suppose by way of contradiction that there exists some $(t_1,r_1)$ such that 
    
    \begin{align*}
        \max\{\alpha(t_1,r_1),\beta(t_1,r_1)\}
        >Q(t_1). 
    \end{align*} 
    Let $\Gamma(t_1,r_1)$ denote some closed characteristic triangle with $(t_1,r_1)$ as its top vertex. Since the initial data satisfy 
    \begin{align}
        \alpha_0(r),\beta_0(r)\leq M_2=Q(0), 
    \end{align} 
    whereas at least one of the gradient variables exceeds $Q$ at $(t_1,r_1)$, the compactness of the characteristic triangle, $\Gamma(t_1,r_1)$ implies the existence of a first upward crossing $(t^*,r^*)\in\Gamma(t_1,r_1)$. Without loss of generality, suppose that $\alpha$ crosses first.  

    \begin{align}
        &\alpha(t^*,r^*)=Q(t^*), \nonumber\\
        &\beta(t^*,r^*)\leq Q(t^*), \label{Eqn:Contradictive_Assumption}\\ 
        &\left.\partial_2(\alpha-Q)
        \right|_{(t^*,r^*)}\geq0. \nonumber
    \end{align} 
    We then evaluate the derivative of $\alpha$ along the $2$-characteristics, 

    \begin{align*}
        \partial_2\alpha(t^*,r^*) 
        &=\left.\left(-\frac{\gamma+1}{4} 
        \alpha^2-\frac{3-\gamma}{4}\alpha\beta
        +A_2\beta-B_2\alpha\right)\right|
        _{(t^*,r^*)} 
        \nonumber\\
        &\leq \left.\left(-\frac{\gamma+1}{4}Q^2 
        -\frac{3-\gamma}{4}Q\beta 
        -A_2(Q-\beta) 
        -(B_2-A_2)Q\right)\right|_{(t^*,r^*)}
        \nonumber\\
        &\leq -\frac{\gamma+1}{4}Q(t^*)^2. 
    \end{align*} 
    This allows us to compute the derivative of the difference between $\alpha$ and $Q(t)$, 

    \begin{align*}
        \left.\partial_2\left(\alpha-Q(t)\right) 
        \right|_{(t^*,r^*)} 
        &\leq -\frac{\gamma+1}{4}Q(t^*)^2 
        -(\left.\partial_tQ(t))\right|_{t^*} 
        \nonumber\\
        &=-\frac{\gamma+1}{4}Q(t^*)^2 
        +\frac{\gamma+1}{8}Q(t^*)^2 
        \nonumber\\
        &=-\frac{\gamma+1}{8}Q(t^*)^2 < 0. 
    \end{align*} 
    This contradicts (\ref{Eqn:Contradictive_Assumption}), thus completing the proof.

<!-- corollary-1: body_md -->
\label{Cor:FirstDerivativeDecay}
    Let $h_0(r),u_0(r),\alpha_0(r),\beta_0(r)$ satisfy the assumptions in Theorem 
    \ref{Thm:Radial_Rarefactive_Existence}. Then, the derivatives of the Riemann variables, $w_r$, $z_r$, decay according to  

    \begin{align*}
        \max\{\lVert w_r(t,\cdot)\rVert_{L^\infty}
        ,\lVert z_r(t,\cdot)\rVert_{L^\infty}\} 
        =O(1+t)^{-1}.
    \end{align*}
<!-- end corollary-1 -->
<!-- proof-4: body_md -->
We first note that 
    \begin{align}
        &w_r=\alpha-\frac{2uh}{rc_2},  
        \label{Eqn:w_rCorr}
        \\
        &z_r=\beta+\frac{2uh}{rc_1}. 
        \label{Eqn:z_rCorr}
    \end{align} 
    from the invariance of the supersonic expanding condition under the classical flow, we note that 
    \begin{align}
        0\leq \frac{2}{\gamma-1}h\leq u.  
    \end{align} 
    Therefore, the latter term in (\ref{Eqn:w_rCorr}) is negative, allowing us to bound $w(t,r)$ above. Indeed, 
    \begin{align}
        &w(t,r)= \int_0^r w_r(t,x)dx 
        \leq rQ(t), 
        \\
        &\frac{w(t,r)}{r}\leq Q(t),   
        \label{Eqn:wUpperCorr}
    \end{align} 
    where $Q(t)$ is defined as in the proof of Lemma
    \ref{Thm:Refined_Upper_Bound}. Using this, we obtain 
    \begin{align}
        \frac{h}{r}\leq \frac{\gamma-1}{4r} 
        (w-z)\leq \frac{w}{2r}\leq\frac{1}{2}Q(t). 
        \label{Eqn:hr}
    \end{align} 
    It remains to bound the ratios $u/c_1$, $u/c_2$. The latter is trivial owing to the supersonic flow, 
    \begin{align}
        \frac{u}{c_2}=\frac{u}{u+h} 
        \leq  \frac{u}{u}=1, 
        \label{Eqn:uc2}
    \end{align} 
    whereas the former is bounded owing to the fact that $\gamma<3$. Indeed, 
    \begin{align}
        &c_1=u-h\geq \frac{3-\gamma}{2}u, 
        \\
        &\frac{u}{c_1}\leq\frac{2}{3-\gamma}. 
        \label{Eqn:uc1}
    \end{align} 
    Thus, combining (\ref{Eqn:wUpperCorr})--(\ref{Eqn:uc1}) and Lemma \ref{Thm:Refined_Upper_Bound} yields the desired decay 
    \begin{align*}
        \max\{\lVert w_r(t,\cdot)\rVert_{L^\infty}
        ,\lVert z_r(t,\cdot)\rVert_{L^\infty}\} 
        =O(1+t)^{-1}.
    \end{align*}

    \cap C^1([0,T];C_b^{N+1}(\mathbb R^3)).
    \end{align*}
    Then, for all $0\leq t\leq T$, 

    \begin{align*}
        &\partial_x^\alpha H(t,0) 
        =\partial_x^\alpha \mathbf{U}(t,0) 
        =0 \qquad \forall |\alpha|\leq N+1, 
        \\
        &H(t,x) >0 \qquad \forall x \neq 0. 
    \end{align*}

<!-- proof-5: body_md -->
We first define the ideal, 
    
    \begin{align*}
        \mathfrak{I}_{N+1}:= 
        \{f\in C^{N+1}:\partial_x^\alpha f(0) 
        =0 \ \forall |\alpha|\leq N+1 \}. 
    \end{align*} 
    Given that 

    \begin{align*}
        &H \in \mathfrak{I}_{N+1}(\mathbb{R}^3), 
        \\
        &\mathbf{U} \in \mathfrak{I}_{N+1}(\mathbb{R}^3; 
        \mathbb{R}^3),   
    \end{align*} 
    then we observe the following, 

    \begin{align}
        \mathbf{U}\cdot\nabla H&\in\mathfrak{I}_{N+1} 
        (\mathbb{R}^3), \label{Eqn:UGradH}
        \\
        H\nabla\cdot\mathbf{U}&\in\mathfrak{I}_{N+1} 
        (\mathbb{R}^3), \label{Eqn:HGradU}
        \\
        \mathbf{U}\cdot\nabla\mathbf{U}&\in\mathfrak{I}_{N+1} 
        (\mathbb{R}^3;\mathbb{R}^3), \label{Eqn:UGradU}
        \\
        H\nabla H&\in\mathfrak{I}_{N+1} 
        (\mathbb{R}^3;\mathbb{R}^3), \label{Eqn:HGradH} 
        \\
        H^2\nabla S&\in\mathfrak{I}_{N+1} 
        (\mathbb{R}^3;\mathbb{R}^3), \label{Eqn:H2GradS}
    \end{align} 
    thus implying that 

    \begin{align}
        &\partial_tH\in\mathfrak{I}_{N+1} 
        (\mathbb{R}^3), \label{Eqn:H_ideal}
        \\
        &\partial_t\mathbf{U}\in\mathfrak{I}_{N+1} 
        (\mathbb{R}^3;\mathbb{R}^3).  
        \label{Eqn:U_ideal}
    \end{align} 
    We then define a vector of the form, 

    \begin{align*}
        &X(t):= \left[ \partial_x^\alpha 
        H(t,0), \partial_x^\alpha\mathbf{U}(t,0)\right], 
    \end{align*} 
    where $\alpha$ ranges over $|\alpha|\leq N+1$. Differentiating with respect to time gives us an ODE, 

    \begin{align*}
        \frac{d}{dt}X(t)=F(t,X(t)), 
    \end{align*} 
    By Equations (\ref{Eqn:H_ideal})--(\ref{Eqn:U_ideal}), this ODE satisfies
    $F(t,0)=0$. Since $X(0)=0$, uniqueness for this finite-dimensional ODE
    implies that $X(t)\equiv0$ for all $0\leq t\leq T$.

    It then remains to prove that $H(t,x)>0$ for all $x\neq 0$. Indeed, we consider the flow, 

    \begin{align*}
        &\partial_t \Phi =\mathbf{U}\circ\Phi, 
        \\
        &\Phi(0,y) = y. 
    \end{align*} 
    Since $\mathbf{U}(t,0)=0$, the flow satisfies $\Phi(t,0)=0$. This gives us the time evolution of our sonic speed, 

    \begin{align*}
        &\partial_t(H\circ\Phi) 
        = \left(-\frac{\gamma-1}{2}H 
        \nabla \cdot \mathbf{U}\right)\circ\Phi, 
        \\
        &H\circ\Phi = H_0(y) 
        \exp\left(-\frac{\gamma-1}{2} 
        \int_0^t \nabla\cdot \mathbf{U} \circ\Phi ds\right) 
        > 0 \qquad \forall y\neq 0, 
    \end{align*} 
    thus implying that the sonic speed, $H$, is positive for all $y\neq 0$. Furthermore, because the solution is classical, the flow map is invertible. Since
    $\Phi(t,0)=0$, every $x\neq0$ has a preimage $y\neq0$. Therefore, $H(t,x)>0$ for all $x\neq0$.

<!-- lemma-5: body_md -->
\label{Lemma:CartRadScalar}
    Let $|\alpha|=k\geq1$, and let $f(x)=F(r)\in\mathbb{R}$ where $r=|x|$. Then for all $x\neq0$, 

\begin{align}
    \partial^\alpha f(x) = \sum_{j=1}^k 
    \frac{F^{(j)}(r)}{r^{k-j}}A_{\alpha,j}\left(
    \frac{x}{r}\right), 
    \label{Eqn:CartRadScalar}
\end{align} 
where the $A_{\alpha,j}$ are smooth and bounded with respect to their argument.
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
We first write our radial unit vector component-wise,  

\begin{align*}
    \omega:=\frac{x}{r}=\left[\frac{x_1}{r},
    \frac{x_2}{r},\frac{x_3}{r}\right], 
\end{align*} 
and define the following notation, 

\begin{align*}
    \partial_l:=\frac{\partial}{\partial x_l}. 
\end{align*}
This gives us  

\begin{align*}
    \partial_l r= \omega_l,  
\end{align*} 
and its partial derivative componentwise,  

\begin{align*}
    &\partial_l\omega_i=\frac{\delta_{il}
    -\omega_i\omega_l}{r}. 
\end{align*}
We proceed by induction. The base case is trivial. Indeed, 

\begin{align*}
    \partial_l f(x) = F'(r)\frac{x_l}{r}. 
\end{align*} 
Then, suppose by way of induction that (\ref{Eqn:CartRadScalar}) holds for all $|\alpha|\leq k$.  

\begin{align*}
    \partial^\alpha f(x) = \sum_{j=1}^k 
    \frac{F^{(j)}(r)}{r^{k-j}}A_{\alpha,j}\left(
    \frac{x}{r}\right). 
\end{align*} 
We then differentiate once more, 

\begin{align*}
    \partial_l\partial^\alpha f(x) &= 
    \sum_{j=1}^k
    \biggl((j-k)\frac{F^{(j)}(r)}{r^{(k+1)-j}} 
    \frac{x_l}{r}
    A_{\alpha,j}\left(\frac{x}{r}\right) 
    +\frac{F^{(j+1)}(r)}{r^{(k+1)-(j+1)}}\frac{x_l}{r}A_{\alpha,j} 
    \left(\frac{x}{r} \right) 
    \nonumber\\
    &+\frac{F^{(j)}(r)}{r^{k+1-j}}
    B_{\alpha,j,l}(\omega) \biggr), 
\end{align*}  
where 

\begin{align*}
    B_{\alpha,j,l}(\omega):=\sum_{i=1}^3 
    \frac{\partial A_{\alpha,j}}{\partial\omega_i} 
    (\omega)(\delta_{il}-\omega_i\omega_l) . 
\end{align*} 
Since $|x_l/r|\leq 1$, they can be absorbed into the $A_{\alpha,j}$---as can $B_{\alpha,j,l}$. This gives the desired form and completes the proof.
<!-- end proof-6 -->
<!-- lemma-6: body_md -->
\label{Lemma:CartRadVector}
    Let $|\alpha|=k\geq0$, and let 
    $G(x)=g(r)\omega\in\mathbb{R}^3$. Then for all $x\neq0$, 

\begin{align}
    \partial^\alpha G_i(x) 
    =\sum_{j=0}^k \frac{\partial_r^jg(r)}
    {r^{k-j}}A_{\alpha,i,j}(\omega), 
    \label{Eqn:CartRadVector}
\end{align} 
where the $A_{\alpha,i,j}$ are smooth and bounded with respect to their argument.
<!-- end lemma-6 -->
<!-- proof-7: body_md -->
Here, the base case starts at $k=0$. Indeed, we have trivially, 

    \begin{align*}
        G_i(x)=g(r)\omega_i. 
    \end{align*} 
    We then proceed to the inductive case. Suppose that the following holds for $|\alpha| = k$, 
    
    \begin{align*}
        \partial^\alpha G_i(x) 
        =\sum_{j=0}^k \frac{\partial_r^jg(r)}
        {r^{k-j}}A_{\alpha,i,j}(\omega). 
    \end{align*} 
    Differentiating once more gives 

    \begin{align*}
    &\partial_l\partial^\alpha G_i(x) 
    =\sum_{j=0}^k \partial_l\left(\frac{\partial_r^jg(r)}
    {r^{k-j}}A_{\alpha,i,j}(\omega)\right) 
    \\
    &=\sum_{j=0}^k
    \biggl((j-k)\frac{\partial_r^j g(r)}
    {r^{(k+1)-j}}\frac{x_l}{r}
    A_{\alpha,i,j}\left(\frac{x}{r}\right) 
    +\frac{\partial_r^{j+1} g(r)}{r^{(k+1)-(j+1)}}\frac{x_l}{r}A_{\alpha,i,j} 
    \left(\frac{x}{r} \right) 
    \nonumber\\
    &+\frac{\partial_r^j g(r)}{r^{k+1-j}}
    B_{\alpha,i,j,l}(\omega) \biggr). 
    \end{align*} 
    As in the proof of Lemma \ref{Lemma:CartRadScalar}, terms can be appropriately collected and re-indexed to be of the form in (\ref{Eqn:CartRadVector}). This completes the proof.
