<!-- theorem-1: body_md -->
\label{thm:impossibility}
        Fix $0<\alpha<1$. There is no finite constant $A_\alpha$, independent
        of $n$, such that
        \begin{equation}
            \mathcal M^2(\nu,\mu)
            \leq A_\alpha D_\alpha(\nu\Vert\mu)
            \label{eq:false-marton}
        \end{equation}
        for every product measure $\mu$ and every $\nu\ll\mu$.
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
Fix $s\in(0,1)$ and let
        \begin{equation}
            \mu=\operatorname{Ber}(1/2)^{\otimes n},\qquad
            \nu=(1-s)\mu+s\delta_{1^n}.
            \label{eq:counterexample}
        \end{equation}
        The $i$th marginal of $\nu$ is $\operatorname{Ber}((1+s)/2)$. Hence, for every coupling $\pi\in\Pi(\mu,\nu)$,
        \begin{equation}
            \pi(X_i\neq Y_i)
            \geq \left\|\mu_i-\nu_i\right\|_{\rm TV}
            =\frac{s}{2}.
            \label{eq:marginal-lower}
        \end{equation}
        It follows that $\mathcal M^2(\nu,\mu)\geq ns^2/4$.
        
        This lower bound is attained. Draw $X^n\sim\mu$ and
        $B\sim\operatorname{Ber}(s)$ independently, and set $Y^n=X^n$ if $B=0$ and
        $Y^n=1^n$ if $B=1$. Then $Y^n\sim\nu$ and
        \begin{equation}
            \pi(X_i\neq Y_i)=\Pr(B=1,X_i=0)=\frac{s}{2}.
        \end{equation}
        Consequently,
        \begin{equation}
            \mathcal M^2(\nu,\mu)=\frac{ns^2}{4}.
            \label{eq:counter-cost}
        \end{equation}
        
        The density $\mathrm{d}\nu/\mathrm{d}\mu$ equals $1-s$ away from $1^n$ and $1-s+s2^n$ at $1^n$. Substitution in \eqref{eq:renyi} gives
        $D_\alpha(\nu\Vert\mu)=\frac{1}{\alpha-1}\log\bigl[       (1-2^{-n})(1-s)^\alpha
            +2^{-n}(1-s+s2^n)^\alpha\bigr].$
        Since $0<\alpha<1$, the second term in the brackets is of order $2^{-n(1-\alpha)}$. Therefore
        \begin{equation}
            \lim_{n\to\infty}D_\alpha(\nu\Vert\mu)
            =-\frac{\alpha}{1-\alpha}\log(1-s)<\infty,
            \label{eq:counter-limit}
        \end{equation}
        and hence equations~\eqref{eq:counter-cost} and~\eqref{eq:counter-limit} rule out any constant independent of $n$. For the same pair, $D(\nu\Vert\mu)=sn\log 2+O(1)$, consistent with \eqref{eq:marton-kl}.

\label{prop:boundedDifferencesOfMeans}
            \label{thm:product-transport}
            Let $\mu=\bigotimes_{i=1}^n\mu_i$ and $f(x^n)=\sum_{i=1}^n f_i(x_i)$, with bounded $f_i$ and $C_\alpha$ defined in \eqref{eq:kappa}. Let $\alpha \in (0,1)$. For every measure $\rho=\bigotimes_{i=1}^n\rho_i$ with $\rho_i\ll\mu_i$ for every $i$,

<!-- proposition-2: body_md -->
\label{prop:refined-pinsker}
            Let $0<\alpha<1$, let $\nu\ll\mu$, and let $g$ be a measurable
            function taking values in $[0,1]$. Then
            \begin{align}
                &\bigl(\mathbb{E}_\nu[g]\!-\!\mathbb{E}_\mu[g]\bigr)^2\!\!\leq\!\!
                \Bigl[
                \alpha\psi\bigl(\mathbb{E}_\mu[g]\bigr)
                \!+\!(1-\alpha)\psi\bigl(\mathbb{E}_\nu[g]\bigr)
                \Bigr]\frac{D_\alpha(\nu\Vert\mu)}{\alpha}.
                \label{eq:refined-expectation}
            \end{align}
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
The refined Hoeffding bound~\cite[Section 2.2.4]{raginsky-sason} gives, for any probability measure $\eta$ and any $s\in\mathbb R$,
            \begin{equation}
                \log\mathbb{E}_\eta\!\left[e^{s\big(g-\mathbb{E}_\eta[g]\big)}\right]
                \leq \frac{s^2}{4}\psi\bigl(\mathbb{E}_\eta[g]\bigr).
                \label{eq:refined-hoeffding}
            \end{equation}
            For every $\lambda\in\mathbb R$, the variational representation of R\'enyi divergence (see e.g.~\cite[Lemma 2]{esposito-gastpar-issa}) yields
            \begin{align}
                \frac{D_\alpha(\nu\Vert\mu)}{\alpha}
                &\geq
                \frac{1}{\alpha-1}
                \log\mathbb{E}_\nu\!\left[e^{(\alpha-1)\lambda g}\right]
                -
                \frac{1}{\alpha}
                \log\mathbb{E}_\mu\!\left[e^{\alpha\lambda g}\right].
                \label{eq:renyi-variational-test}
            \end{align}
            Apply \eqref{eq:refined-hoeffding} to both logarithms. Since $\alpha-1<0$, division by $\alpha-1$ reverses the inequality for the first term. Consequently,
            \begin{align}
                \frac{D_\alpha(\nu\Vert\mu)}{\alpha}
                &\geq
                \lambda\bigl(\mathbb{E}_\nu[g]-\mathbb{E}_\mu[g]\bigr)
                \notag\\
                &\quad-\frac{\lambda^2}{4}
                \Bigl[
                \alpha\psi\bigl(\mathbb{E}_\mu[g]\bigr)
                +(1-\alpha)\psi\bigl(\mathbb{E}_\nu[g]\bigr)
                \Bigr].
            \end{align}
            Maximising the right-hand side over $\lambda\in\mathbb R$ proves \eqref{eq:refined-expectation} whenever the coefficient of $\lambda^2$ is positive. If it vanishes, then $\mathbb{E}_\mu[g]\in\{0,1\}$, so $g$ is constant $\mu$-almost surely. Absolute continuity implies that the two expectations coincide, and the claim follows.

<!-- theorem-2: body_md -->
\label{thm:mgf}
            Under the assumptions of Theorem \ref{thm:product-transport}, with $\mathbb{E}_\mu[f]=0$, for every $\lambda>0$,
            \begin{equation}
                K(\lambda)
                \leq\frac{C_\alpha\lambda^2}{2}-E_\alpha(\lambda).
                \label{eq:mgf-bound}
            \end{equation}
            For every $t>0$,
            \begin{equation}
                \mu(f\geq t)
                \leq\inf_{\lambda>0}e^{-\lambda t
                +\frac{C_\alpha\lambda^2}{2}-E_\alpha(\lambda)}.
                \label{eq:tail-bound}
            \end{equation}
<!-- end theorem-2 -->
<!-- proof-4: body_md -->
If $C_\alpha=0$, then $f$ is $\mu$-almost surely constant and the centred claim is immediate. Assume $C_\alpha>0$. Since $f$ is additive, $\nu_\lambda$ is a product measure. Differentiating \eqref{eq:cgf} gives
            \begin{equation}
                \mathbb{E}_{\nu_\lambda}[f]=K'(\lambda).
                \label{eq:tilt-mean}
            \end{equation}
            Substituting \eqref{eq:tilt} into the definition of R\'enyi divergence gives
            \begin{equation}
                D_\alpha(\nu_\lambda\Vert\mu)
                =\frac{\alpha K(\lambda)-K(\alpha\lambda)}{1-\alpha},
                \label{eq:tilt-renyi}
            \end{equation}
            whereas for KL divergence one has
            \begin{equation}
                D(\nu_\lambda\Vert\mu)
                =\lambda K'(\lambda)-K(\lambda).
                \label{eq:tilt-kl}
            \end{equation}
            Subtracting these identities characterises the correction in terms of $K$:
            \begin{equation}
                E_\alpha(\lambda)
                =\lambda K'(\lambda)
                -\frac{K(\lambda)-K(\alpha\lambda)}{1-\alpha}.
                \label{eq:correction}
            \end{equation}
            
            Theorem \ref{thm:product-transport} and \eqref{eq:tilt-mean} imply
            \begin{equation}
                K'(\lambda)^2
                \leq2C_\alpha D_\alpha(\nu_\lambda\Vert\mu).
                \label{eq:mean-divergence}
            \end{equation}
            By \eqref{eq:correction-divergence} and \eqref{eq:tilt-kl},
            \begin{equation}
                D_\alpha(\nu_\lambda\Vert\mu)
                =\lambda K'(\lambda)-K(\lambda)-E_\alpha(\lambda).
            \end{equation}
            Substitution in \eqref{eq:mean-divergence} and completion of the square give
            \begin{align}
                K(\lambda)
                &\leq\lambda K'(\lambda)
                -\frac{K'(\lambda)^2}{2C_\alpha}-E_\alpha(\lambda)\\
                                                                
                                                                     
                &\leq\frac{C_\alpha\lambda^2}{2}-E_\alpha(\lambda).
                \label{eq:complete-square}
            \end{align}
            Applying Markov's inequality to $e^{\lambda f}$ gives
            \begin{equation}
                \mu\bigl(f-\mathbb{E}_\mu[f]\geq t\bigr)
                =\mu(f\geq t)\leq e^{-\lambda t+K(\lambda)}.
            \end{equation}
            Applying \eqref{eq:mgf-bound} and taking the infimum over $\lambda>0$ proves \eqref{eq:tail-bound}.

<!-- proposition-3: body_md -->
\label{prop:strict}
        If $0<p<1/2$ and $0<u<1-2p$, then
        \begin{equation}
            I_{\rm R}(u)>I_D(u).
            \label{eq:strict-improvement}
        \end{equation}
<!-- end proposition-3 -->
<!-- proof-5: body_md -->
We keep the optimiser defining $I_D(u)$ fixed and show that decreasing $\alpha$ slightly below $1$ increases the rate. Let $L=\log((1-p)/p)$, $d=1-2p$, and $
            \lambda_0=\frac{u}{\kappa_1(p)}=\frac{2Lu}{d}.
            \label{eq:lambda-zero}$
        
        We first compute the change in the optimal coordinate constant. For $\varepsilon\geq0$ close to zero, let $F_\varepsilon(q)=(q-p)^2/[2D_{1-\varepsilon}(q\Vert p)]$, with continuous value $F_\varepsilon(p)=p(1-p)/(1-\varepsilon)$. Here $D_1=D$. The function $F_\varepsilon$ and its derivative with respect to $\varepsilon$ are jointly continuous on a neighbourhood of $\{0\}\times[0,1]$. At $\varepsilon=0$, the unique maximiser is $q=1-p$~\cite{ordentlich-weinberger}. Danskin's theorem~\cite{danskin} therefore shows that the first-order change in the maximum is obtained by keeping $q=1-p$ fixed and differentiating $F_\varepsilon(q)$ with respect to $\varepsilon$. Since
       $
            D_{1-\varepsilon}(1-p\Vert p)
            =dL-2\varepsilon p(1-p)L^2+o(\varepsilon),
        $
        we obtain $
            \kappa_{1-\varepsilon}(p)
            =\kappa_1(p)+\varepsilon p(1-p)+o(\varepsilon).
            \label{eq:kappa-expansion}$
        Taylor expansion of \eqref{eq:binary-correction}, using $k_p''(\lambda)=q_\lambda(1-q_\lambda)$, similarly gives
       $
            \frac{E_{1-\varepsilon}(\lambda)}{n}
            =\frac{\varepsilon\lambda^2}{2}q_\lambda(1-q_\lambda)
            +o(\varepsilon).
            \label{eq:correction-expansion}$
        The first expansion determines the increase in the quadratic penalty; the second gives the gain from the correction. Evaluating the objective in \eqref{eq:binary-rate} at $\alpha=1-\varepsilon$ and $\lambda=\lambda_0$ now yields
        \begin{align}
            I_{\rm R}(u)&\geq I_D(u)
            +\frac{\varepsilon\lambda_0^2}{2}
            \bigl[q_{\lambda_0}(1-q_{\lambda_0})-p(1-p)\bigr]+o(\varepsilon).
            \label{eq:rate-expansion}
        \end{align}
        The assumption $0<u<d$ gives $0<\lambda_0<2L$. Since $q_0=p$, $q_{2L}=1-p$, and $q_\lambda$ is strictly increasing, it follows that $p<q_{\lambda_0}<1-p$. Thus $q_{\lambda_0}(1-q_{\lambda_0})>p(1-p)$, and the rate strictly exceeds $I_D(u)$ for sufficiently small $\varepsilon>0$.

<!-- remark-3: body_md -->
A similar approach to the one developed in this paper can also be employed for R\'enyi divergences of order $\alpha>1$, yielding improvements for larger deviations. For instance, in the Bernoulli example with $0<p<1/2$, choosing $\alpha>1$ sufficiently close to $1$ improves the bound induced by KL divergence for the event $\{\overline X_n-p\geq u\}$ when $1-2p<u<1-p$. This complements the improvement obtained with $\alpha<1$ for $0<u<1-2p$. At $u=1-2p$, the bounds obtained using R\'enyi divergence coincide with the bound induced by KL divergence, which
also matches the Chernoff bound.
