\label{Main_Result_1}

{

Consider the system \eqref{main_problems_rewritten}, where $\Omega = \{x \in \mathbb{R}^n: R_1 < \left\|x\right\| < R_2\}$, $M_1 > 0$, $M_2 > 0$, $R > 0$, $0 < R < R_1 < R_2$, $R_1 + R < R_2$, and the function $H_{R}(w)$ is given by \eqref{H1}. Let $(u^{\epsilon}_1, u^{\epsilon}_2) \in C^2(\Omega; \mathbb{R}^2) \cap C^0(\overline{\Omega}; \mathbb{R}^2)$ be a solution of \eqref{main_problems_rewritten} for any $\epsilon > 0$, and suppose that $\{(u^{\epsilon}_1, u^{\epsilon}_2)\}_{\epsilon > 0}$ converges along a subsequence as $\epsilon \rightarrow 0^+$ to $(u_1, u_2)$. Then we have the followings:
\begin{enumerate}
\item There exists a unique $r^* \in [R_1, R_2 -R], r^* = r^*(M_1, M_2, R, R_1, R_2)$ such that 
\begin{equation}
\begin{split}
\partial \{u_1 > 0\} \cap \Omega = \{x \in \mathbb{R}^n: \left\|x\right\| = r^*\}, \\
\partial \{u_2 > 0\} \cap \Omega = \{x \in \mathbb{R}^n: \left\|x\right\| = r^* + R\}.
\end{split}
\end{equation}
In particular, all the free boundary points are regular.
\item For $n = 2$, $r^*$ is the unique solution of the equation 
\begin{equation}
     \frac{\log (\frac{r^*}{R_1})}{\log (\frac{R_2}{r^* + R})} - \frac{M_1}{M_2} = 0
\end{equation}
in the interval $[R_1, R_2]$.
\item For $n \geq 3$, $r^*$ is the unique solution of the equation 
\begin{equation}
    \frac{\frac{1}{R_1^{n-2}} - \frac{1}{(r^*)^{n-2}}}{\frac{1}{(r^* + R)^{n-2}} - \frac{1}{R_2^{n-2}}} - \frac{M_1}{M_2} = 0
\end{equation}
in the interval $[R_1, R_2]$.
\item In any dimensions, for fixed $M_2, R, R_1, R_2$, $r^* = r^*(M_1, M_2, R, R_1, R_2)$ is increasing as a function of $M_1$, and
\begin{equation}
\begin{split}
\lim_{M_1 \rightarrow 0^+}r^*(M_1, M_2, R, R_1, R_2) = R_1, \\
\lim_{M_1 \rightarrow \infty} r^*(M_1, M_2, R, R_1, R_2) = R_2 -R.    
\end{split}
\end{equation}
\item In any dimensions, for fixed $M_1, R, R_1, R_2$, $r^* = r^*(M_1, M_2, R, R_1, R_2)$ is decreasing as a function of $M_2$, and 
\begin{equation}
\begin{split}
\lim_{M_2 \rightarrow 0^+}r^*(M_1, M_2, R, R_1, R_2) = R_2 - R, \\
\lim_{M_2 \rightarrow \infty} r^*(M_1, M_2, R, R_1, R_2) = R_1.  
\end{split}
\end{equation}
\item In any dimensions, for fixed $M_1, M_2, R_1, R_2$, $r^* = r^*(M_1, M_2, R, R_1, R_2)$ is decreasing as a function of $R$, and
\begin{equation}
\begin{split}
\lim_{R \rightarrow 0^+} r^*(M_1, M_2, R, R_1, R_2) = r_0 = r_0(M_1, M_2, R_1, R_2),  
\end{split}
\end{equation}
where 
\begin{equation}
\begin{split}
    r_0 = r_0(M_1, M_2, R_1, R_2) = \begin{cases} R_1^{\frac{M_2}{M_1 + M_2}}R_2^{\frac{M_1}{M_1 + M_2}}, & n = 2, \\
    (\frac{M_1}{M_1 + M_2}\frac{1}{R_2^{n-2}} + \frac{M_2}{M_1 + M_2}\frac{1}{R_1^{n-2}})^{\frac{1}{2-n}}, & n \geq 3, 
    \end{cases}
\end{split}
\end{equation}
\end{enumerate}
}

<!-- remark-1: body_md -->
\label{Remark_about_Main_Result_1}

{\rm

We elaborate more on part (6) of Theorem \ref{Main_Result_1}. Consider the local analogue (by setting $R = 0$) of \eqref{main_problems_rewritten}. That is, we consider
\begin{equation}
\label{main_problems_local}
\begin{cases} \Delta u^{\varepsilon}_{1}(x) = \frac{1}{\varepsilon^2} u^{\varepsilon}_1(x)   u^{\varepsilon}_2(x)\quad & \text{ in } \{x \in \mathbb{R}^n: R_1 < \left\|x\right\| < R_2\},\\
\Delta u^{\varepsilon}_2(x) = \frac{1}{\varepsilon^2} u^{\varepsilon}_2(x)   u^{\varepsilon}_1(x)\quad & \text{ in } \{x \in \mathbb{R}^n: R_1 < \left\|x\right\| < R_2\},\\
        u^{\varepsilon}_1 = M_1 & \text{ on } \{\left\|x\right\| = R_1\}, \\
        u^{\varepsilon}_1 = 0 & \text{ on } \{\left\|x\right\| = R_2\}, \\
        u^{\varepsilon}_2 = 0 & \text{ on } \{\left\|x\right\| = R_1\}, \\
        u^{\varepsilon}_2 = M_2 & \text{ on } \{\left\|x\right\| = R_2\},
       \end{cases}
\end{equation}
Then the difference $u(x) = u_1(x) - u_2(x)$ satisfies 
\begin{equation}
\begin{split}
\begin{cases} \Delta u = 0 & \text{ in } \{x \in \mathbb{R}^n: R_1 < \left\|x\right\| < R_2\}, \\
u = M_1 & \text{ on } \{\left\|x\right\| = R_1\}, \\
u = -M_2 & \text{ on } \{\left\|x\right\| = R_2\}, \\
\end{cases}
\end{split}
\end{equation}
It follows that $\partial \{u_1 > 0\} \cap \Omega = \partial \{u_2 > 0\} \cap \Omega = \{\left\|x\right\| = r_0\}$, which is also the zero level set of the function $u = u_1 - u_2$. Therefore, in part (6) of Theorem \ref{Main_Result_1}, it was asserted that as $R \rightarrow 0^+$, the free boundary for the long range case in \eqref{main_problems_rewritten} converges to the free boundary for the adjacent case \eqref{main_problems_local}. This connects the theory of long range interaction with that of the local interaction in this setting.  
}

<!-- notation-1: body_md -->
\label{Notation_for_Two_Standard_Functions}

Assume that $R_1, R_2 > 0$, and $R$ are such that $R < R_1 < R_1 + R < R_2$.
\begin{enumerate}
\item We define $\phi_{R_1, R_2, R, 2}: [R_1, R_2 -R) \rightarrow \mathbb{R}$ by setting 
\begin{equation}
    \phi_{R_1, R_2, R, 2}(x) : = \frac{\log(\frac{x}{R_1})}{\log(\frac{R_2}{x + R})}
\end{equation}
for all $x \in [R_1, R_2 - R)$.
\item For all $n \geq 3$, we define $\phi_{R_1, R_2, R, n}: [R_1, R_2 -R) \rightarrow \mathbb{R}$ by setting 
\begin{equation}
    \phi_{R_1, R_2, R, n}(x) : = \frac{R_1^{2-n} - x^{2-n}}{(x + R)^{2 - n} - R_2^{2-n}}
\end{equation}
for all $x \in [R_1, R_2 - R)$.
\end{enumerate}

<!-- lemma-3: body_md -->
\label{Phi_is_Strictly_Increasing}
Consider the same setting as in Notation \ref{Notation_for_Two_Standard_Functions}. Then we have the followings:
\begin{enumerate}
\item For any $n \geq 2$, the function $\phi_{R_1, R_2, R, n}(x)$ is a non-negative continuous strictly increasing function on $[R_1, R_2 - R)$, with $\lim_{x \rightarrow (R_2 - R)^-}\phi_{R_1, R_2, R, n}(x) = \infty$ and $\phi_{R_1, R_2, R, n}(R_1) = 0$.
\item If $0 < r_2 < r_1 < R_2 - R_1$, we have $\phi^{-1}_{R_1, R_2, 0, n}(x) \geq \phi^{-1}_{R_1, R_2, r_2, n}(x) \geq \phi^{-1}_{R_1, R_2, r_1, n}(x)$ for all $x \in [0,\infty)$.
\item We have $\lim_{R \rightarrow 0^+}\phi^{-1}_{R_1, R_2, R,n}(x) = \phi^{-1}_{R_1, R_2, 0,n}(x)$ for all $x \in [0,\infty)$.
\end{enumerate}
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
(1) is clear, and (2) follows easily from the definition of $\phi_{R_1, R_2, R, n}(x)$ in Notation \ref{Notation_for_Two_Standard_Functions}. Therefore, we focus on proving (3) only. 
It suffices to show that if $\{r_k\}_{k = 1}^{\infty} \subset [0,\infty)$ is a sequence with $r_1 \geq r_2 \geq r_3 \geq \cdots$ and $r_k \rightarrow 0$, we have $\lim_{k \rightarrow \infty}\phi^{-1}_{R_1, R_2, r_k,n}(x) = \phi^{-1}_{R_1, R_2, 0,n}(x)$ for all $x \in [0, \infty)$. Take $y_k := \phi^{-1}_{R_1, R_2, r_k, n}(x)$, so by (2), $\{y_k\}_{k \in \mathbb{N}}$ is an increasing sequence, so $\{y_k\}_{k \in \mathbb{N}}$ converges to $\sup_{k \in \mathbb{N}} y_k =: y_0 \leq R_2$. Since $\phi_{R_1, R_2, r_k, n}(y_k) = x$, we have
\begin{equation}
\label{The_Equation_Satisfied_by_y_k}
\begin{split}
\begin{cases}
\frac{\log(\frac{y_k}{R_1})}{\log(\frac{R_2}{y_k + r_k})} = x, & n = 2, \\
\frac{R_1^{2-n} - y_k^{2-n}}{(y_k + r_k)^{2 - n} - R_2^{2-n}} = x, & n \geq 3.
\end{cases}
\end{split}
\end{equation}
Pass to the limit as $k \rightarrow \infty$ in \eqref{The_Equation_Satisfied_by_y_k}, and we obtain 
\begin{equation}
\label{The_Equation_Satisfied_by_y_k}
\begin{split}
\begin{cases}
\frac{\log(\frac{y_0}{R_1})}{\log(\frac{R_2}{y_0})} = x, & n = 2, \\
\frac{R_1^{2-n} - y_0^{2-n}}{(y_0)^{2 - n} - R_2^{2-n}} = x, & n \geq 3.
\end{cases}
\end{split}
\end{equation}
This implies that $\phi_{R_1, R_2, 0, n}(y_0) = x$, so $y_0 = \phi_{R_1, R_2, 0, n}^{-1}(x)$. Consequently, $\lim_{k \rightarrow \infty}\phi^{-1}_{R_1, R_2, r_k,n}(x) = \phi^{-1}_{R_1, R_2, 0,n}(x)$, as desired.

<!-- lemma-4: body_md -->
\label{Analysis_of_r^*_in_2D}

Consider the case in which $n = 2$. Then $r^* \in [R_1, R_2 - R]$ is the unique solution to the equation
\begin{equation}
\label{The_Equation_Satisfied_by_r^*_in_2D}
\frac{\log(\frac{r^*}{R_1})}{\log (\frac{R_2}{r^* + R})} - \frac{M_1}{M_2} = 0
\end{equation}
in $[R_1, R_2 - R]$.
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
Take 
\begin{equation}
\begin{split}
A = \{x \in \mathbb{R}^n: R_1 < \left\|x\right\| < r^*\}, \\
B = \{x \in \mathbb{R}^n: r^* + R < \left\|x\right\| < R_2\}.
\end{split}
\end{equation} 


Observe that $u_1(x)$ and the function $\frac{M_1}{\log (\frac{R_1}{r^*})}\log |x| - \frac{M_1\log (r^*)}{\log (\frac{R_1}{r^*})}$ are both harmonic in $A$ and take on the same boundary value on $\partial A$. (To see that they agree on $\partial A$, we note that $\partial A = \{\left\|x\right\| = r^*\} \cup \{\left\|x\right\| = R_1\}$, and they both vanish on $\{\left\|x\right\| = r^*\}$, and they both take on the value $M_1$ on $\{\left\|x\right\| = R_1\}$.) By the standard comparison principle for harmonic functions, we have $u_1(x) = \frac{M_1}{\log (\frac{R_1}{r^*})}\log |x| - \frac{M_1\log (r^*)}{\log (\frac{R_1}{r^*})}$ identically in $A$. 

Now take $\{e_1, e_2\} \subset \mathbb{R}^2$ to be the standard orthonormal basis of $\mathbb{R}^2$, so by a direct computation we have 
\begin{equation}
\label{Normal_Derivative_for_u_1}
\frac{\partial u_1}{\partial \nu}(r^*e_2) = \frac{M_1}{\log (\frac{R_1}{r^*})}\frac{1}{r^*},
\end{equation}
where the outward is the positive direction. 

By a similar argument applied to $u_2(x)$ and $\frac{M_2}{\log (\frac{R_2}{r^* + R})}\log |x| - \frac{M_2\log (r^* + R)}{\log (\frac{R_2}{r^* + R})}$ in the annular domain $B$, we have $u_2(x) = \frac{M_2}{\log (\frac{R_2}{r^* + R})}\log |x| - \frac{M_2\log (r^* + R)}{\log (\frac{R_2}{r^* + R})}$ identically in $B$. Similarly as in the case of $u_1$, we have 
\begin{equation}
\label{Normal_Derivative_for_u_2}
\frac{\partial u_2}{\partial \nu}((r^*+R)e_2) = -\frac{M_2}{\log (\frac{R_2}{r^* + R})}\frac{1}{r^* + R},
\end{equation}
where we have a minus sign as now the inward is the positive direction. 




Now by \eqref{Normal_Derivative_for_u_1}, \eqref{Normal_Derivative_for_u_2}, and \eqref{The_First_Free_Boundary_Condition} in Theorem \ref{The_Free_Boundary_Condition}, we have 
\begin{equation}
\frac{\frac{M_1}{\log (\frac{R_1}{r^*})}\frac{1}{r^*}}{-\frac{M_2}{\log (\frac{R_2}{r^* + R})}\frac{1}{r^* + R}} = \frac{\frac{1}{r^*}}{\frac{1}{r^* +R}}.
\end{equation}
Therefore, 
\begin{equation}
\frac{M_1}{\log (\frac{R_1}{r^*})} = -\frac{M_2}{\log (\frac{R_2}{r^* + R})}.
\end{equation}
Rearrange the equation, and we obtain \eqref{The_Equation_Satisfied_by_r^*_in_2D}. 

Finally, we observed that $r^*$ is the solution of the equation $\phi_{R_1, R_2, R, 2}(r^*) = \frac{M_1}{M_2}$, so in view of Lemma \ref{Phi_is_Strictly_Increasing}, $r^*$ is unique.

\label{The_Equation_Satisfied_by_r^*_in_nD}

\begin{equation}
\begin{split}

\end{equation}

\frac{\partial u_1}{\partial \nu}(r^*e_n) = \frac{M_1}{R_1^{2-n} - (r^*)^{2-n}}(2-n)(r^*)^{1-n},
\end{equation}

\label{Normal_Derivative_for_u_2_n_geq_3}

\end{equation}

\end{equation}

\frac{M_1}{R_1^{2-n} - (r^*)^{2-n}} = -\frac{M_2}{R_2^{2-n} - (r^* + R)^{2-n}}.
\end{equation}

\end{split}

\end{split}

<!-- remark-2: body_md -->
Lemma \ref{Relation_Between_r^*_and_M_1_and_M_2} matches with our intuition. \begin{equation}
\label{The_Main_Computational_Equation}
\left(\frac{d}{dt}r^*_t\right)\left(\frac{1}{x_{2,t}r^*_t} + \frac{x_{1,t}}{x_{2,t}^2(r^*_t + R_t)}\right) = \frac{d}{dt}\left(\frac{M_{1,t}}{M_{2,t}}\right) + \frac{\frac{d}{dt}R_{1,t}}{R_{1,t}}\frac{1}{x_{2,t}} + \frac{x_{1,t}}{x_{2,t}^2}\frac{\frac{d}{dt}R_{2,t}}{R_{2,t}} - \frac{x_{1,t}\frac{d}{dt}R_t}{x_{2,t}^2(r^*_t + R_t)},
\end{equation}
where 
\begin{equation}
    x_{1,t} = \log(\frac{r^*_t}{R_{1,t}}) > 0, x_{2,t} = \log(\frac{R_{2,t}}{r^*_{t} + R_t}) > 0
\end{equation}
for all $t \in I$.
<!-- end lemma-7 -->
<!-- proof-7: body_md -->
We recall that by Theorem \ref{Main_Result_1} (2), one has 
\begin{equation}
    \frac{x_{1,t}}{x_{2,t}} - \frac{M_{1,t}}{M_{2,t}} = 0
\end{equation}
for all $t \in I$. Take the derivative on both sides of the equation with respect to $t$, and we obtain that 
\begin{equation}
\label{Derivative_on_Both_Sides_Dimension_Two}
    \frac{(\frac{dx_{1,t}}{dt})x_{2,t} - x_{1,t}(\frac{x_{2,t}}{dt})}{x_{2,t}^2} - \frac{d}{dt}\left(\frac{M_{1,t}}{M_{2,t}}\right)  = 0.
\end{equation}
By a direct computation, 
\begin{equation}
\label{Derivative_of_x_1,t}
\begin{array}{rl}
\begin{split}
\frac{d}{dt}x_{1,t} & = \frac{1}{\left(\frac{r^*_t}{R_{1,t}}\right)}\frac{d}{dt}\left(\frac{r^*_t}{R_{1,t}}\right)
\\ &= \frac{R_{1,t}}{r^*_t}\frac{(\frac{d}{dt}r^*_t)(R_{1,t}) - (r^*_t)(\frac{d}{dt}R_{1,t})}{R_{1,t}^2}
\\ &= \frac{(\frac{d}{dt}r^*_t)(R_{1,t}) - (r^*_t)(\frac{d}{dt}R_{1,t})}{r^*_tR_{1,t}}
\end{split}
\end{array}
\end{equation} 
Similarly, by a direct computation, 
\begin{equation}
\label{Derivative_of_x_2,t}
\begin{split}
\frac{d}{dt}x_{2,t} = \frac{(\frac{d}{dt}R_{2,t})(r^*_t + R_t) - (R_{2,t})(\frac{d}{dt}(r^*_t + R_t))}{R_{2,t}(r^*_t + R_t)}
\end{split}
\end{equation}
Now plug \eqref{Derivative_of_x_1,t} and \eqref{Derivative_of_x_2,t} into \eqref{Derivative_on_Both_Sides_Dimension_Two}, and we obtain 
\begin{equation}
    \frac{(\frac{(\frac{d}{dt}r^*_t)(R_{1,t}) - (r^*_t)(\frac{d}{dt}R_{1,t})}{r^*_tR_{1,t}})x_{2,t} - x_{1,t}(\frac{(\frac{d}{dt}R_{2,t})(r^*_t + R_t) - (R_{2,t})(\frac{d}{dt}(r^*_t + R_t))}{R_{2,t}(r^*_t + R_t)})}{x_{2,t}^2} - \frac{d}{dt}\left(\frac{M_{1,t}}{M_{2,t}}\right)  = 0. \nonumber 
\end{equation}
Simplify the left hand side of the above equality, and we obtain 
\begin{equation}
    \frac{1}{x_{2,t}}(\frac{\frac{d}{dt}r^*_t}{r^*_t} - \frac{\frac{d}{dt}R_{1,t}}{R_{1,t}}) - \frac{x_{1,t}}{x_{2,t}^2}(\frac{\frac{d}{dt}R_{2,t}}{R_{2,t}} - \frac{\frac{d}{dt}r^*_t + \frac{d}{dt}R_t}{r^*_t + R_t}) - \frac{d}{dt}\left(\frac{M_{1,t}}{M_{2,t}}\right)  = 0. \nonumber 
\end{equation}
Rearrange the above equation, and we obtain
\begin{equation}
    \left(\frac{d}{dt}r^*_t\right)\left(\frac{1}{x_{2,t}r^*_t}\right) + \frac{x_{1,t}}{x_{2,t}^2}\frac{\frac{d}{dt}r^*_t + \frac{d}{dt}R_{t}}{r^*_t + R_{t}} = \frac{d}{dt}\left(\frac{M_{1,t}}{M_{2,t}}\right) + \frac{\frac{d}{dt}R_{1,t}}{R_{1,t}}\frac{1}{x_{2,t}} + \frac{x_{1,t}}{x_{2,t}^2}\frac{\frac{d}{dt}R_{2,t}}{R_{2,t}}. \nonumber
\end{equation}
                                                                                                  
By a simple calculation, we have
\begin{equation}
    \left(\frac{d}{dt}r^*_t\right)\left(\frac{1}{x_{2,t}r^*_t} + \frac{x_{1,t}}{x_{2,t}^2(r^*_t + R_t)}\right) = \frac{d}{dt}\left(\frac{M_{1,t}}{M_{2,t}}\right) + \frac{\frac{d}{dt}R_{1,t}}{R_{1,t}}\frac{1}{x_{2,t}} + \frac{x_{1,t}}{x_{2,t}^2}\frac{\frac{d}{dt}R_{2,t}}{R_{2,t}} - \frac{x_{1,t}\frac{d}{dt}R_t}{x_{2,t}^2(r^*_t + R_t)}. \nonumber
\end{equation}
This yields \eqref{The_Main_Computational_Equation}, as desired.

\label{r^*_t_is_constant_if_ratio_is_constant}

Assume that $n = 2$, and let the setting be as above. Suppose that $R_{1,t}, R_{2,t}$, and $R_t$ are constants in $t \in I$. Then $r^*_t$ is constant in $t \in I$ if $\frac{M_{1,t}}{M_{2,t}}$ is constant in $t \in I$.
<!-- end corollary-1 -->
<!-- proof-8: body_md -->
One can see this immediately from Theorem \ref{Main_Result_1} (2). Here we deduce the result from Lemma \ref{Derivative_of_r^*_t}. By Lemma \ref{Derivative_of_r^*_t}, we have
\begin{equation}
    \left(\frac{dr^*_t}{dt}\right)\left(\frac{1}{x_{2,t}r^*_t} + \frac{x_{1,t}}{x_{2,t}^2r^*_t}\right) = \frac{d}{dt}\left(\frac{M_{1,t}}{M_{2,t}}\right) = 0,
\end{equation}
so $r^*_t$ is constant in $t \in I$.

<!-- proposition-1: body_md -->
\label{Equilibrium_for_R_1_and_R_2}

Assume that $n = 2$, and let the setting be as above. Assume $\frac{M_{1,t}}{M_{2,t}}$ is a constant $\alpha$ for all $t \in I$, and that $R_t$ is also constant in $t \in I$. Then $r^*_t$ is constant in $t \in I$ if 
\begin{equation}
\label{Key_Equation_for_R_1_and_R_2}
    \frac{\frac{d}{dt}(\log R_{1,t})}{\frac{d}{dt}(\log R_{2,t})} = -\alpha
\end{equation}
for all $t \in I$.
<!-- end proposition-1 -->
<!-- proof-9: body_md -->
Observe that by Theorem \ref{Main_Result_1} (2), $\alpha = \frac{M_{1,t}}{M_{2,t}} = \frac{x_{1,t}}{x_{2,t}}$, where 
\begin{equation}
    x_{1,t} = \log(\frac{r^*_t}{R_{1,t}}) > 0, x_{2,t} = \log(\frac{R_{2,t}}{r^*_{t} + R_t}) > 0
\end{equation}
for all $t \in I$. Therefore, we have from \eqref{Key_Equation_for_R_1_and_R_2} that 
\begin{equation}
    \frac{\frac{d}{dt}(\log R_{1,t})}{\frac{d}{dt}(\log R_{2,t})} = -\frac{x_{1,t}}{x_{2,t}}.
\end{equation}
Therefore, 
\begin{equation}
    \frac{\frac{1}{R_{1,t}}\frac{d}{dt}(R_{1,t})}{\frac{1}{R_{2,t}}\frac{d}{dt}( R_{2,t})} = -\frac{x_{1,t}}{x_{2,t}},
\end{equation}
so 
\begin{equation}
\label{Key_Equation_Relating_Derivative_of_R_1_and_R_2_Dimension_Two}
\frac{x_{2,t}}{R_{1,t}}\left(\frac{d}{dt}R_{1,t}\right) + \frac{x_{1,t}}{R_{2,t}}\left(\frac{d}{dt}R_{2,t}\right) = 0.
\end{equation}
Divide the equation \eqref{Key_Equation_Relating_Derivative_of_R_1_and_R_2_Dimension_Two} by $x_{2,t}^2$, and we obtain 
\begin{equation}
\label{Key_Equation_Relating_Derivative_of_R_1t_and_R_2t_Dimension_two}
\frac{1}{R_{1,t}x_{2,t}}\left(\frac{dR_{1,t}}{dt}\right) + \frac{x_{1,t}}{R_{2,t}x_{2,t}^2}\left(\frac{dR_{2,t}}{dt}\right) = 0.
\end{equation}
Now recall that $\frac{M_{1,t}}{M_{2,t}}$ is a constant $\alpha$ for all $t \in I$, and that $R_t$ is a constant in $t \in I$, so by Lemma \ref{Derivative_of_r^*_t} we have
\begin{equation}
\label{Application_of_Lemma_4_1_to_Proposition_4_4_Dimension_Two}
     \left(\frac{d}{dt}r^*_t\right)\left(\frac{1}{x_{2,t}r^*_t} + \frac{x_{1,t}}{x_{2,t}^2(r^*_t + R_t)}\right) = \frac{\frac{d}{dt}R_{1,t}}{R_{1,t}}\frac{1}{x_{2,t}} + \frac{x_{1,t}}{x_{2,t}^2}\frac{\frac{d}{dt}R_{2,t}}{R_{2,t}}.
\end{equation}
Plug \eqref{Key_Equation_Relating_Derivative_of_R_1t_and_R_2t_Dimension_two} into \eqref{Application_of_Lemma_4_1_to_Proposition_4_4_Dimension_Two}, and we obtain 
\begin{equation}
    \left(\frac{dr^*_t}{dt}\right) = 0,
\end{equation}
so $r^*_t$ is a constant in $t \in I$.

\label{Derivative_of_r^*_t_Higher_Dimensions}

Assume that $n \geq 3$, and let the setting be as above. Then the mapping $I \rightarrow \mathbb{R}$ given by $t \mapsto r^*_t$ is also differentiable on $I$, with 
\begin{equation}
\label{The_Main_Computational_Equation_Higher_Dimensions}
\begin{split}
\left(\frac{dr^*_t}{dt}\right)(n-2)[\frac{1}{x_{2,t}}(r^*_t)^{1-n} + \frac{x_{1,t}}{x_{2,t}^2}(r^*_t + R_t)^{1-n}] = \frac{d}{dt}\left(\frac{M_{1,t}}{M_{2,t}}\right) 
\\+ (n-2)\frac{1}{x_{2,t}}R^{1-n}_{1,t}\left(\frac{dR_{1,t}}{dt}\right) + (n-2)\frac{x_{1,t}}{x_{2,t}^2}R^{1-n}_{2,t}\left(\frac{dR_{2,t}}{dt}\right) - (n-2)\frac{x_{1,t}}{x_{2,t}^2}(r^*_t + R_t)^{1-n}\left(\frac{dR_t}{dt}\right),
\end{split}
\end{equation}
where 
\begin{equation}
    x_{1,t} = R_{1,t}^{2-n} - (r^*_t)^{2-n} > 0, x_{2,t} = (r^*_t + R_t)^{2-n} - R_{2,t}^{2-n} > 0
\end{equation}
for all $t \in I$.

<!-- proof-11: body_md -->

\begin{equation}
    \frac{x_{1,t}}{x_{2,t}} - \frac{M_{1,t}}{M_{2,t}} = 0
\end{equation}
for all $t \in I$. Take the derivative on both sides of the equation with respect to $t$, and we obtain that 
\begin{equation}
\label{Derivative_on_Both_Sides}
    \frac{(\frac{dx_{1,t}}{dt})x_{2,t} - x_{1,t}(\frac{x_{2,t}}{dt})}{x_{2,t}^2} - \frac{d}{dt}\left(\frac{M_{1,t}}{M_{2,t}}\right)  = 0.
\end{equation}
By a direct computation, 

\begin{equation}
\label{Derivative_of_x_1,t_Higher_Dimensions}
\begin{split}
\frac{d}{dt}x_{1,t} = (2-n)R_{1,t}^{1-n}(\frac{d}{dt}R_{1,t}) - (2-n)(r^*_t)^{1-n}(\frac{d}{dt}r^*_t),
\end{split}
\end{equation}
\begin{equation}
\label{Derivative_of_x_2,t_Higher_Dimensions}

\label{Equilibrium_for_R_1_and_R_2_Higher_Dimensions}

Assume that $n \geq 3$, and let the setting be as above. Assume $\frac{M_{1,t}}{M_{2,t}}$ is a constant $\alpha$ for all $t \in I$, and that $R_t$ is also constant in $t \in I$. Then $r^*_t$ is constant in $t \in I$ if 
\begin{equation}
\label{Key_Equation_for_R_1_and_R_2_Higher_Dimensions}
\frac{\frac{d}{dt}(R_{1,t}^{2-n})}{\frac{d}{dt}(R_{2,t}^{2-n})} = -\alpha

\begin{equation}
    x_{1,t} = R_{1,t}^{2-n} - (r^*_t)^{2-n} > 0, x_{2,t} = (r^*_t + R_t)^{2-n} - R_{2,t}^{2-n} > 0
\end{equation}
for all $t \in I$. Therefore, we have from \eqref{Key_Equation_for_R_1_and_R_2_Higher_Dimensions} that 
\begin{equation}
    \frac{\frac{d}{dt}(R_{1,t}^{2-n})}{\frac{d}{dt}(R_{2,t}^{2-n})} = -\frac{x_{1,t}}{x_{2,t}}.
\end{equation}
Therefore, 
\begin{equation}
    \frac{(2-n)R_{1,t}^{1-n}\frac{d}{dt}(R_{1,t})}{(2-n)R_{2,t}^{1-n}\frac{d}{dt}(R_{2,t})} = -\frac{x_{1,t}}{x_{2,t}},
\end{equation}

\begin{equation}
\label{Key_Equation_Relating_Derivative_of_R_1_and_R_2}
{x_{2,t}}R_{1,t}^{1-n}\left(\frac{d}{dt}R_{1,t}\right) + {x_{1,t}}R_{2,t}^{1-n}\left(\frac{d}{dt}R_{2,t}\right) = 0.
\end{equation}

\begin{equation}
\label{Key_Equation_Relating_Derivative_of_R_1t_and_R_2t}
\frac{R_{1,t}^{1-n}}{x_{2,t}}\left(\frac{dR_{1,t}}{dt}\right) + \frac{R_{2,t}^{1-n}x_{1,t}}{x_{2,t}^2}\left(\frac{dR_{2,t}}{dt}\right) = 0.
\end{equation}
Now recall that $\frac{M_{1,t}}{M_{2,t}}$ is a constant $\alpha$ for all $t \in I$, and that $R_t$ is a constant in $t \in I$, so by Lemma \ref{Derivative_of_r^*_t_Higher_Dimensions} we have
\begin{equation}
\label{Application_of_Lemma_4_1_to_Proposition_4_4}
\left(\frac{dr^*_t}{dt}\right)[\frac{1}{x_{2,t}}(r^*_t)^{1-n} + \frac{x_{1,t}}{x_{2,t}^2}(r^*_t + R_t)^{1-n}] =  
\\ \frac{1}{x_{2,t}}R^{1-n}_{1,t}\left(\frac{dR_{1,t}}{dt}\right) + \frac{x_{1,t}}{x_{2,t}^2}R^{1-n}_{2,t}\left(\frac{dR_{2,t}}{dt}\right).
\end{equation}
Plug \eqref{Key_Equation_Relating_Derivative_of_R_1t_and_R_2t} into \eqref{Application_of_Lemma_4_1_to_Proposition_4_4}, and we obtain 
\begin{equation}
    \left(\frac{dr^*_t}{dt}\right) = 0,
\end{equation}
so $r^*_t$ is a constant in $t \in I$.
