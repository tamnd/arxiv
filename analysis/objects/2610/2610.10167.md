\label{ass:gaussian_model}
For $k = 1, \ldots, K$,
\[
    \widehat\tau_k
    \overset{\mathrm{ind}}{\sim}
    \mathcal N\left(\tau_k,\frac{\sigma_k^2}{n_k}\right),
\]
where the standard deviations $\sigma_k$ are bounded, i.e., $\underline{\sigma} \leq \sigma_k \leq \overline{\sigma}$ with known constants $\underline{\sigma}, \overline{\sigma} > 0$.

<!-- assumption-2: body_md -->
\label{ass:weak_signal}
There is a constant $H > 0$ such that $\tau_k = h_k / \sqrt{n}$ with $\sup_k |h_k| \leq H$ for all $k$.

<!-- theorem-1: body_md -->
\label{thm:minimax_regret}
   Suppose Assumptions~\ref{ass:gaussian_model} and~\ref{ass:weak_signal} hold. Then,
   \begin{align*}
        &\inf_{\widetilde\pi} \sup_{P_{\bm h, \bm \sigma} \in \mathcal{P}_K} R_K(P_{\bm h, \bm \sigma}, \widetilde\pi) 
        \\
        &=  \max_{\Pi \in \mathfrak P} \inf_{\widetilde\pi} \int R_K(P_{\bm h, \bm \sigma},\widetilde\pi)
\Pi(dP_{\bm h, \bm \sigma})
        \\
        &=\max_{0 \leq t \leq H} t\Phi \left(-\frac{t}{\overline \sigma}\right),
    \end{align*}
   where the infima range over rules $\widetilde\pi: \mathbb{R}^K \times [0, \infty)^K \to [0, 1]^K$ and $\mathfrak P$ is the set of Borel priors on $\mathcal{P}_K$.
    The plug-in rule $\widetilde\pi_k^*(Z, \widehat \sigma)=\mathbf{1}{\left\{ Z_k>0 \right\}}$ is both minimax and Bayes under the least-favorable prior.

<!-- theorem-2: body_md -->
\label{thm:value}
Suppose Assumptions~\ref{ass:gaussian_model}--\ref{ass:weak_signal} hold and that the true standard errors $\bm \sigma$ are known. Suppose further that $\max_k p_k = O(1/K)$ and that $m_K = \left \lfloor \frac{1}{4} \log(K) / (\log(\log(K))) \right \rfloor$. Then,
\begin{align*}
    \sup_{\bm h \in [-H, H]^K} &\mathbb{E}_{\bm h, \bm \sigma}\left[\left(\sqrt{n} \{\widehat V^\ast_{m_K} - V_{\bm h}^\ast\}\right)^2\right]
    \\
    &\qquad \qquad \qquad \lesssim_{H, \overline{\sigma}, \underline{\sigma}} \left(\frac{\log(\log (K))}{\log (K)}\right)^2,
\end{align*}
This rate is minimax optimal in the submodel $p_k = 1/K, \sigma_k = \sigma$ for all $k = 1, \ldots, K$ for some $\sigma > 0$. \footnote{$a\lesssim b$ means $a\leq Cb$ for a universal constant $C$.  Subscripts on $\lesssim$ list the quantities on which $C$ may depend.}

\label{ass:holder_smoothness}
For fixed $\beta\in(0,1]$ and $L,H>0$, the task-level signals satisfy
$h_k=g(x_k)$ for every $k\in[K]$, where
$g\in\mathcal{H}_d(\beta,L,H)$ and

    \\
    &\quad |g(x) - g(y)| \leq L \|x - y\|_2^\beta, \, \forall x, y \in \mathbb{R}^d\Big\}.
\end{align*}

<!-- definition-1: body_md -->
\label{def:linear_smoother}
Let $w_{j} :\mathbb{R}^d \to \mathbb{R}$ be deterministic weight functions
satisfying $\sum_{j=1}^K w_{j}(x)=1.$
Then
\[
    \widehat g_w(x)
    :=
    \sum_{j=1}^K w_{j}(x)Z_j,
\]
is a linear smoother for signal $g$ and induces the (threshold) policy $\widetilde\pi_w(x) = \mathbf{1}{\left\{ \widehat g_w(x)>0 \right\}}$.

<!-- proposition-1: body_md -->
\label{pro:smoothing_tradeoff}
Suppose Assumptions~\ref{ass:gaussian_model}--\ref{ass:holder_smoothness} hold. The policy induced by
any linear smoother $\widehat g_w$ satisfies
    \begin{align*}
         R_K(P_{g, \bm{\sigma}}, \widetilde\pi_w) \leq \min\{H, \mathsf{A}_g(w) + \gamma^\ast \mathsf{E}_{\bm{\sigma}}(w)\},
    \end{align*}
    where $\mathsf{A}_g(w) = \sum_{k = 1}^K p_k |g(x_k) - m_w(x_k)|$ is the \emph{approximation} and $\mathsf{E}_{\bm{\sigma}}(w) = \sum_{k = 1}^K p_k s_w(x_k)$ is the \emph{estimation} error, and $\gamma^\ast =\sup_{t\geq0}t\Phi(-t)$. This bound is sharp over the class of linear smoothers.

\label{pro:welfare_decomposition}
    Suppose Assumptions~\ref{ass:gaussian_model}--\ref{ass:weak_signal} hold. Let $\widehat g_w$ be a linear smoother. Then,
    \begin{align*}
         \sqrt{n} W_{P_{g, \bm{\sigma}}}(\widetilde\pi_w) = \bar g_p \bar \psi_{w, p} + \operatorname{Cov}_p(g, \psi_w), 
    \end{align*}
    where $\bar g_p = \sum_{k = 1}^K p_k g(x_k)$, $\psi_{w, k} = \mathbb{E}_{g, \bm{\sigma}}\left[\widetilde\pi_{w, k}(Z)\right] = \Phi\left(\frac{m_w(x_k)}{s_w(x_k)}\right)$, $\bar \psi_{w, p} = \sum_{k = 1}^K p_k \psi_{w, k}$ and
    \[
        \operatorname{Cov}_p(g, \psi_w) = \sum_{k = 1}^K p_k (g(x_k) - \bar g_p) (\psi_{w, k} - \bar \psi_{w, p}) = \frac{1}{2} \sum_{k = 1}^K \sum_{l = 1}^K p_k p_l (g(x_k) - g(x_l)) (\psi_{w, k} - \psi_{w, l}).
    \]

<!-- proposition-3: body_md -->
\label{pro:cross_validation}
Suppose Assumptions~\ref{ass:gaussian_model}--\ref{ass:weak_signal} hold. Assume further that $\max_k p_k = O(1/K)$. Suppose for each task $x_k$, the data is split into $F$ folds and satisfies 
\[
    Z_k^{(f)} \overset{\mathrm{ind}}{\sim} \mathcal{N}(h_k, F \sigma_k^2), 
\]
for $k \in [K], f \in [F]$. Denote $Z_k^{(-f)} = \frac{1}{F - 1} \sum_{j \neq f} Z_k^{(j)}$ and $Z_k = \frac{1}{F} \sum_{f = 1}^F Z_k^{(f)}$. Let $\Lambda_K$ be a finite set of resolutions. For each $r \in \Lambda_K$, define the linear smoothing policies $\widetilde \pi_{w_r, k}(z) = \mathbf{1}{\left\{ \sum_{j = 1}^K w_{r, j}(x_k) z_j > 0 \right\}}$. Select
\[
    \widehat r \in \operatorname*{arg\,max}_{r \in \Lambda_K} \frac{1}{F} \sum_{f = 1}^F \sum_{k = 1}^K p_k Z_k^{(f)} \widetilde\pi_{w_r,k}(Z^{(-f)}).
\]
Then, 
\[
    R_K(P_{g, \bm{\sigma}}, \widetilde\pi_{w_{\widehat r}}) \lesssim  \min_{r\in\Lambda_K} R_K(P_{g, \bm{\sigma}_F}, \widetilde\pi_{w_r}) + \sqrt{\frac{\log(|\Lambda_K|)}{K}},
\]
where $\bm\sigma_F=\sqrt{F/(F-1)}\bm\sigma$.

<!-- lemma-1: body_md -->
\label{lem:weighted_assouad}
Let $M,K\geq1$, let $\Omega:=\{0,1\}^M$, and let
$\{P_\omega:\omega\in\Omega\}$ be Borel probability measures on
$\mathbb{R}^K$.
For $j\in[M]$, let $\omega^{(j)}$ denote the vector
obtained from $\omega$ by flipping its $j$th coordinate, i.e., $\omega^{(j)} := (\omega_1,\ldots,\omega_{j-1}, 1-\omega_j,\omega_{j+1},\ldots,\omega_M).$
Suppose that
\[
    \max_{\omega\in\Omega}
    \max_{j\in[M]}
    \operatorname{d_{\mathrm{TV}}}\!\left(P_\omega,P_{\omega^{(j)}}\right)
    \leq \Delta,
\]
for some $\Delta\in[0,1)$, where $\operatorname{d_{\mathrm{TV}}}(P,Q) := \sup_{A\in\mathcal B(\mathbb{R}^K)} |P(A)-Q(A)|$ is the total variation distance. Then, for any nonnegative weights $v_1,\ldots,v_M$,
\[
    \inf_{\widetilde\omega:\mathbb{R}^K\to[0,1]^M}
    \max_{\omega\in\Omega}
    \mathbb{E}_\omega\!\left[
        \sum_{j=1}^M
        v_j
        |\widetilde\omega_j(Z)-\omega_j|
    \right]
    \geq
    \frac{1-\Delta}{2}
    \sum_{j=1}^M v_j,
\]
where the infimum ranges over all measurable decision rules
$\widetilde\omega:\mathbb{R}^K\to[0,1]^M$, and $\mathbb{E}_\omega$ denotes expectation under $P_\omega$. The same lower bound holds with the maximum replaced by the average over $\Omega$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
The proof follows from similar arguments as in
\citet[Theorem~2.12]{tsybakov2009introduction}, but extended to
nonnegative weights and randomized decisions.

Fix any measurable decision rule
$\widetilde\omega:\mathbb{R}^K\to[0,1]^M$. Bounding the maximum by the average over $\Omega$ gives
\begin{align*}
    \max_{\omega\in\Omega} \mathbb{E}_\omega \left[ \sum_{j=1}^M v_j|\widetilde\omega_j(Z)-\omega_j| \right]
    &\geq \frac{1}{2^M} \sum_{\omega\in\Omega}
   \mathbb{E}_\omega \left[ \sum_{j=1}^M v_j|\widetilde\omega_j(Z)-\omega_j| \right]
   \\
    &=
    \frac{1}{2^M}
    \sum_{j=1}^M v_j
    \sum_{\omega\in\Omega}
    \mathbb{E}_\omega \left[ |\widetilde\omega_j(Z)-\omega_j| \right].
\end{align*}

For each $j\in[M]$, define $\Omega_{j,0} := \{\omega\in\Omega:\omega_j=0\}$. Using $\widetilde\omega_j(Z)\in[0,1]$,
\begin{align*}
    \max_{\omega\in\Omega} \mathbb{E}_\omega \left[ \sum_{j=1}^M v_j|\widetilde\omega_j(Z)-\omega_j| \right]
    & \geq
    \frac{1}{2^M}
    \sum_{j=1}^M v_j
    \sum_{\omega\in\Omega}
    \mathbb{E}_\omega \left[ |\widetilde\omega_j(Z)-\omega_j| \right]
    \\
    &= \frac{1}{2^M}
    \sum_{j=1}^M v_j
    \sum_{\omega\in\Omega_{j,0}}
    \Big\{
        \mathbb{E}_\omega[|\widetilde\omega_j(Z)|]
        +
        \mathbb{E}_{\omega^{(j)}}[|1-\widetilde\omega_j(Z)|]
    \Big\}
    \\
    &= \frac{1}{2^M}
    \sum_{j=1}^M v_j
    \sum_{\omega\in\Omega_{j,0}}
    \Big\{
        \mathbb{E}_\omega[\widetilde\omega_j(Z)]
        +
        \mathbb{E}_{\omega^{(j)}}[1-\widetilde\omega_j(Z)]
    \Big\}
    \\
    &= \frac{1}{2^M}
    \sum_{j=1}^M v_j
    \sum_{\omega\in\Omega_{j,0}}
    \left[
        1 -
        \Big\{\mathbb{E}_{\omega^{(j)}}[\widetilde\omega_j(Z)] - \mathbb{E}_\omega[\widetilde\omega_j(Z)]
    \Big\} \right]
    \\
    &\geq \frac{1}{2^M}
    \sum_{j=1}^M v_j
    \sum_{\omega\in\Omega_{j,0}}
    \left[
        1 -
        \left|\mathbb{E}_{\omega^{(j)}}[\widetilde\omega_j(Z)] - \mathbb{E}_\omega[\widetilde\omega_j(Z)]
    \right| \right]
\end{align*}
For any measurable $\delta:\mathbb{R}^K\to[0,1]$ and any probability measures $P,Q$ on $\mathbb{R}^K$, 
\begin{align*}
    \left| \mathbb{E}_P[\delta(Z)]-\mathbb{E}_Q[\delta(Z)] \right|
    &=\left| \int_0^1 \left\{ P(\delta(Z)>t)-Q(\delta(Z)>t) \right\} dt \right|
    \\
    &\leq \int_0^1 \left| P(\delta(Z)>t)-Q(\delta(Z)>t) \right| dt    
    \\
    &\leq \int_0^1 \operatorname{d_{\mathrm{TV}}}(P,Q) dt
    \\
    &= \operatorname{d_{\mathrm{TV}}}(P,Q).
\end{align*}
Applying this inequality with $\delta=\widetilde\omega_j$, $P=P_{\omega^{(j)}}$, and $Q=P_\omega$ and recalling that $\operatorname{d_{\mathrm{TV}}}(P_\omega, P_{\omega^{(j)}}) \leq \Delta$ for all $\omega \in \Omega, j \in [M]$,
\begin{align*}
     \max_{\omega\in\Omega} \mathbb{E}_\omega \left[ \sum_{j=1}^M v_j|\widetilde\omega_j(Z)-\omega_j| \right]
    & \geq \frac{1}{2^M}
    \sum_{j=1}^M v_j
    \sum_{\omega\in\Omega_{j,0}}
    \left[
        1 -
        \left|\mathbb{E}_{\omega^{(j)}}[\widetilde\omega_j(Z)] - \mathbb{E}_\omega[\widetilde\omega_j(Z)]
    \right| \right]
    \\
    &\geq \frac{1}{2^M}
    \sum_{j=1}^M v_j
    \sum_{\omega\in\Omega_{j,0}}
    \left[ 1 - \operatorname{d_{\mathrm{TV}}}(P_\omega, P_{\omega^{(j)}}) \right]
    \\
    &\geq \frac{1}{2^M}
    \sum_{j=1}^M v_j
    \sum_{\omega\in\Omega_{j,0}} \left[ 1 - \Delta \right]
    \\
    &= \frac{1 - \Delta}{2} \sum_{j=1}^M v_j,
\end{align*}
where we used $|\Omega_{j,0}|=2^{M-1}$ for every $j\in[M]$. Since this holds for every measurable
$\widetilde\omega:\mathbb{R}^K\to[0,1]^M$, taking the infimum over
$\widetilde\omega$ proves the claim.

<!-- lemma-2: body_md -->
\label{lem:low_norm_packing}
Suppose Assumption~\ref{ass:quasi_uniform_design} holds with constants $0 < c < C < \infty, a_0, r_0 > 0$ and let $r\in[2a_0K^{-1/d},r_0/2]$. Define $\phi_0(x):=(1-\|x\|_2^2)_+^\beta$ and set $\phi_{z,r}(x):=\phi_0\left(\frac{x-z}{r}\right)$. Then there exists $\mathcal{Z}_r=\{z_1,\ldots,z_M\}  \subseteq \{x_k:k\in[K]\}$
satisfying the following properties:
\begin{enumerate}[label=(\alph*)]
    \item \emph{Separation:} $\|z_i-z_j\|_2>2r$ for $i \neq j \in [M]$.
    \item \emph{Cardinality:} $ M\geq\frac{1}{C2^{d+1}r^d}.$
    \item \emph{Task Norm:} For every $j\in[M]$, $\sum_{k=1}^K\phi_{z_j,r}(x_k)^2 \leq 2^{d+1}CKr^d$.
    \item \emph{Mass:} $\sum_{j=1}^M Q_K\left(B(z_j,r/2)\right) \geq \frac{c}{C2^{2d+1}}$.
\end{enumerate}

<!-- proof-2: body_md -->
Let
$\widetilde{\mathcal{Z}}_r
=\{\widetilde z_1,\ldots,\widetilde z_{\widetilde M}\}$
be a maximal $2r$-packing of the distinct task locations, i.e.,
$\|\widetilde z_i-\widetilde z_j\|_2>2r$ for $i\neq j$ and
\[
    \{x_k:k\in[K]\}
    \subseteq
    \bigcup_{j=1}^{\widetilde M}B(\widetilde z_j,2r).
\]
Such a set exists because there are finitely many distinct task locations.
Hence, by Assumption~\ref{ass:quasi_uniform_design},
\[
    1
    =
    Q_K\bigl(\{x_k:k\in[K]\}\bigr)
    \leq
    \sum_{j=1}^{\widetilde M}
    Q_K\bigl(B(\widetilde z_j,2r)\bigr)
    \leq
    \widetilde M C(2r)^d,
\]
and therefore $ \widetilde M \geq \frac{1}{C2^dr^d}$.

Next, since $\operatorname{supp}(\phi_0)=B(0,1)$, we have
$\operatorname{supp}(\phi_{z,r})=B(z,r)$. Thus, because
$\widetilde{\mathcal{Z}}_r$ is a $2r$-packing, the supports of
$\{\phi_{\widetilde z_j,r}:j\in[\widetilde M]\}$ are pairwise disjoint.
Consequently, since $0\leq\phi_{z,r}\leq1$,
\[
    \sum_{j=1}^{\widetilde M} \sum_{k=1}^K
    \phi_{\widetilde z_j,r}(x_k)^2
    \leq K.
\]
Combining this with the lower bound on $\widetilde M$ gives
\[
    \frac{1}{\widetilde M}
    \sum_{j=1}^{\widetilde M}
    \sum_{k=1}^K \phi_{\widetilde z_j,r}(x_k)^2
    \leq
    \frac{K}{\widetilde M}
    \leq
    C2^dKr^d.
\]
Therefore, at least half $\widetilde z_j \in \widetilde{\mathcal{Z}}_r$ satisfies
\[
    \sum_{k=1}^K \phi_{\widetilde z_j,r}(x_k)^2 \leq
    2^{d+1}CKr^d;
\]
otherwise, the preceding average would exceed $C2^dKr^d$.

Only retain these elements and relabel them as
$\mathcal{Z}_r=\{z_1,\ldots,z_M\}$. Then separation still holds and
\[
    M \geq \frac{\widetilde M}{2} \geq \frac{1}{C2^{d+1}r^d},
\]
by construction.
Moreover, for every $j\in[M]$,
\[
    \sum_{k=1}^K\phi_{z_j,r}(x_k)^2
    \leq
    2^{d+1}CKr^d.
\]

Finally, the balls $\{B(z_j,r/2):j\in[M]\}$ are pairwise disjoint.
Hence, by Assumption~\ref{ass:quasi_uniform_design},
\[
    \sum_{j=1}^M Q_K\left(B(z_j,r/2)\right)
    \geq
    Mc(r/2)^d
    \geq
    \frac{c}{C2^{2d+1}}.
\]

\label{lem:hoelder_regularity}
Fix $d\geq1$ and $0<\beta\leq1$. Let $x, z \in \mathbb{R}^d, r > 0$. Define $\phi_0(x):=(1-\|x\|_2^2)_+^\beta$ and set $\phi_{z,r}(x) := \phi_0\left(\frac{x-z}{r}\right)$. Then, for every $x,y\in\mathbb{R}^d$,
\[
    |\phi_{z,r}(x)-\phi_{z,r}(y)| \leq \frac{2^\beta}{r^\beta}\|x-y\|_2^\beta.
\]

<!-- proof-3: body_md -->
Define $f(x):=(1-\|x\|_2^2)_+$ such that $\phi_{z,r}(x) = h\left(\frac{x - z}{r}\right)^\beta$. Let $x, y \in \mathbb{R}^d$. We show that $f$ is $2$-Lipschitz.
    
    First, assume $\|x\|_2,\|y\|_2\leq1$. Then by the reverse triangle inequality 
    \[
        |f(x)-f(y)| = \left|\|x\|_2^2-\|y\|_2^2\right| \leq 2\|x-y\|_2.
    \]
    Similarly, if $\|x\|_2,\|y\|_2 > 1$ then $f(x) = f(y) = 0$ and the condition holds trivially. Assume now that, without loss of
    generality, $\|x\|_2\leq1<\|y\|_2$. Then by the reverse triangle inequality again
    \[
        |f(x)-f(y)| = f(x) = (1+\|x\|_2) (1-\|x\|_2) \leq 2 (1-\|x\|_2)
        \leq 2(\|y\|_2-\|x\|_2) \leq 2\|x-y\|_2.
    \]
    Thus $f$ is globally $2$-Lipschitz. 
    
    Since $0<\beta\leq1$,
    $|u^\beta-v^\beta|\leq|u-v|^\beta$ for all $u,v\geq0$. Consequently,
    \[
        |\phi_0(x)-\phi_0(y)|
        =
        |f(x)^\beta-f(y)^\beta|
        \leq
        2^\beta\|x-y\|_2^\beta,
    \]
    which shows that $\phi_0$ is $\beta$-H\"older with constant $2^\beta$. Finally, 
    \[
         |\phi_{z,r}(x)-\phi_{z,r}(y)| = \left|\phi_0\left(\frac{x-z}{r}\right) - \phi_0\left(\frac{y-z}{r}\right)\right| \leq 2^\beta \left\| \frac{x - z}{r} - \frac{y - z}{r} \right\|_2^\beta \leq \frac{2^\beta}{r^\beta}\|x-y\|_2^\beta.
    \]
