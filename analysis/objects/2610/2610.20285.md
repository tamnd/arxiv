<!-- definition-1: body_md -->
\label{def:strong-linearization}
A $dn\times dn$ pencil $L(\lambda)=\lambda X+Y$
is a \emph{linearization} of $P$ if
\[
    U(\lambda)L(\lambda)V(\lambda)
    =\operatorname{diag}\bigl(P(\lambda),I_{(d-1)n}\bigr)
\]
for some unimodular matrix polynomials $U$ and $V$,
that is, matrix polynomials whose determinants are
nonzero constants.

A linearization $L(\lambda)$ is \emph{strong} if
$\operatorname{rev}_1L$ also linearizes
$\operatorname{rev}_dP$, where
\[
    \operatorname{rev}_1L(\lambda)=X+\lambda Y,
    \qquad
    \operatorname{rev}_dP(\lambda)
    =\sum_{k=0}^{d}P_{d-k}\lambda^k.
\]

<!-- proposition-1: body_md -->
\label{prop:one-sided-eigenvector-recovery}
Let $\lambda\in\Omega$.
If $L(\lambda)z=0$ and $x=E_{\rm R}z\neq0$, then
$P(\lambda)x=0$.
If $w^*L(\lambda)=0$ and $y=E_{\rm L}^*w\neq0$, then
$y^*P(\lambda)=0$.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Multiplying the first factorization by $z$ gives
\[
    P(\lambda)x
    =H_{\rm R}(\lambda)L(\lambda)z=0.
\]
Multiplying the second by $w^*$ gives
\[
    y^*P(\lambda)
    =w^*L(\lambda)H_{\rm L}(\lambda)=0.
\]

<!-- definition-2: body_md -->
\label{def:weighted-backward-error}
Let $w_0,\ldots,w_d$ be nonnegative weights.
The weighted normwise backward error of a computed eigenpair
$(\lambda,x)$, with $x\neq0$, is
\begin{equation}
\label{eq:weighted-backward-error}
    \eta_{P,w}(\lambda,x)
    =
    \inf\left\{
    \varepsilon\geq0:
    \begin{array}{l}
    (P(\lambda)+\Delta P(\lambda))x=0,\\[1mm]
    \|\Delta P_k\|_2\leq\varepsilon w_k,
        \quad k=0,\ldots,d
    \end{array}
    \right\},
\end{equation}
where
\[
    \Delta P(\lambda)=\sum_{k=0}^{d}\Delta P_k\lambda^k,
    \qquad
    \Delta P_k\in\mathbb C^{n\times n}.
\]
The weights determine how we measure the coefficient
perturbations.
If $w_k=0$, the coefficient $P_k$ cannot change.
<!-- end definition-2 -->
<!-- proposition-2: body_md -->
\label{prop:backward-error-residual}
For $x\neq0$ and $s_w(\lambda)>0$, the weighted normwise
backward error satisfies
\begin{equation}
\label{eq:weighted-residual-formula}
    \eta_{P,w}(\lambda,x)
    =
    \frac{\|P(\lambda)x\|_2}
         {s_w(\lambda)\|x\|_2}.
\end{equation}

<!-- lemma-1: body_md -->
\label{lem:scaling-reversal-backward-error}
Let $P(\lambda)=\sum_{k=0}^{d}P_k\lambda^k$ be a regular
matrix polynomial.
For $\gamma>0$ and
$\delta\in\mathbb C\setminus\{0\}$, define
\[
    Q(t)=\delta P(\gamma t),
    \qquad
    R(t)=\operatorname{rev}_dP(t)
        =\sum_{k=0}^{d}P_{d-k}t^k.
\]
For any matrix polynomial $F(t)=\sum_{k=0}^{d}F_kt^k$,
write
\[
    S_F(r)=\sum_{k=0}^{d}\|F_k\|_2r^k,
    \qquad r\geq0,
\]
with $r^0=1$, including at $r=0$.
Then
\begin{align}
    S_Q(r)&=|\delta|S_P(\gamma r),
    &&r\geq0,
    \label{eq:scaled-polynomial-weight}\\
    S_R(r)&=r^dS_P(r^{-1}),
    &&r>0.
    \label{eq:reversed-polynomial-weight}
\end{align}
Let $\eta_F^{\rm R}$ and $\eta_F^{\rm L}$ denote the
right and left normwise backward errors with weights
$\|F_k\|_2$.
For nonzero vectors $x,y\in\mathbb C^n$,
\begin{align}
    \eta_Q^{\rm R}(x,t)
    &=\eta_P^{\rm R}(x,\gamma t),
    &
    \eta_Q^{\rm L}(y,t)
    &=\eta_P^{\rm L}(y,\gamma t),
    &&t\in\mathbb C,
    \label{eq:scaled-polynomial-backward-error}\\
    \eta_R^{\rm R}(x,t)
    &=\eta_P^{\rm R}(x,t^{-1}),
    &
    \eta_R^{\rm L}(y,t)
    &=\eta_P^{\rm L}(y,t^{-1}),
    &&t\neq0.
    \label{eq:reversed-polynomial-backward-error}
\end{align}
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
Since
\[
    \|\delta\gamma^kP_k\|_2
    =|\delta|\gamma^k\|P_k\|_2,
\]
summing over $k$ gives
\eqref{eq:scaled-polynomial-weight}.
Moreover,
\[
    Q(t)x=\delta P(\gamma t)x,
    \qquad
    y^*Q(t)=\delta y^*P(\gamma t).
\]
Thus, scaling multiplies each residual norm and its
backward-error denominator by the same factor $|\delta|$.

For reversal, reindexing gives
\[
    S_R(r)
    =\sum_{k=0}^{d}\|P_{d-k}\|_2r^k
    =r^dS_P(r^{-1}).
\]
The identity $R(t)=t^dP(t^{-1})$ shows that both the residual norm and its denominator are multiplied by $|t|^d$.
The backward-error identities follow.

<!-- lemma-2: body_md -->
\label{lem:linearization-scaling-reversal}
Let $L(\lambda)$ be a linearization of a matrix
polynomial $P(\lambda)$ of grade $d$.
For $\gamma>0$ and nonzero scalars $\delta,\kappa$,
define
\[
    Q(t)=\delta P(\gamma t),
    \qquad
    M(t)=\kappa L(\gamma t).
\]
Then $M$ is a linearization of $Q$.
If $L$ is strong, then $M$ is strong, and
$\operatorname{rev}_1L$ is a strong linearization
of $\operatorname{rev}_dP$.
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
Suppose
\[
    A(\lambda)L(\lambda)B(\lambda)
    =\operatorname{diag}(P(\lambda),I_{(d-1)n}),
\]
where $A$ and $B$ are unimodular.
Substituting $\lambda=\gamma t$ preserves
unimodularity. Multiplication by the constant
invertible matrix
\[
    \operatorname{diag}
    \bigl((\delta/\kappa)I_n,
          \kappa^{-1}I_{(d-1)n}\bigr)
\]
then gives a unimodular equivalence between $M$
and $\operatorname{diag}(Q,I_{(d-1)n})$.

If $L$ is strong, apply the same argument to
$\operatorname{rev}_1L$ and
$\operatorname{rev}_dP$, using
\[
    \operatorname{rev}_1M(t)
    =\kappa\gamma\,
      \operatorname{rev}_1L(t/\gamma),
    \qquad
    \operatorname{rev}_dQ(t)
    =\delta\gamma^d\,
      \operatorname{rev}_dP(t/\gamma).
\]
Thus $M$ is strong.

Finally, $\operatorname{rev}_1L$ linearizes
$\operatorname{rev}_dP$ by the strong property.
Reversing both again gives $L$ and $P$, so this
linearization is also strong.

<!-- proposition-3: body_md -->
\label{prop:backward-error-transfer}
Suppose the factorizations
\eqref{eq:residual-transfer-identities} hold at $\lambda$,
with $S_P(|\lambda|)>0$ and $W_L(|\lambda|)>0$.
For $z\neq0$ and $x=E_{\rm R}z\neq0$,
\begin{equation}
\label{eq:right-backward-error-transfer}
    \eta_P^{\rm R}(x,\lambda)
    \leq
    \frac{W_L(|\lambda|)\|H_{\rm R}(\lambda)\|_2}
         {S_P(|\lambda|)}
    \frac{\|z\|_2}{\|x\|_2}
    \eta_L^{\rm R}(z,\lambda).
\end{equation}
Similarly, for $w\neq0$ and $y=E_{\rm L}^*w\neq0$,
\begin{equation}
\label{eq:left-backward-error-transfer}
    \eta_P^{\rm L}(y,\lambda)
    \leq
    \frac{W_L(|\lambda|)\|H_{\rm L}(\lambda)\|_2}
         {S_P(|\lambda|)}
    \frac{\|w\|_2}{\|y\|_2}
    \eta_L^{\rm L}(w,\lambda).
\end{equation}

<!-- definition-3: body_md -->
\label{def:backward-error-growth-factors}
For the chosen one-sided factorizations, define the
right and left residual amplification factors by
\[
    g_{\rm R}(\lambda)
    =
    \frac{W_L(|\lambda|)\|H_{\rm R}(\lambda)\|_2}
         {S_P(|\lambda|)},
    \qquad
    g_{\rm L}(\lambda)
    =
    \frac{W_L(|\lambda|)\|H_{\rm L}(\lambda)\|_2}
         {S_P(|\lambda|)}.
\]
For pencil vectors with nonzero selected blocks,
define the recovery norm ratios by
\[
    \rho_{\rm R}(z)
    =\frac{\|z\|_2}{\|E_{\rm R}z\|_2},
    \qquad
    \rho_{\rm L}(w)
    =\frac{\|w\|_2}{\|E_{\rm L}^*w\|_2}.
\]
The corresponding backward-error growth factors are
the products
\[
    \Gamma_{\rm R}(\lambda,z)
    =g_{\rm R}(\lambda)\rho_{\rm R}(z),
    \qquad
    \Gamma_{\rm L}(\lambda,w)
    =g_{\rm L}(\lambda)\rho_{\rm L}(w).
\]

<!-- proposition-4: body_md -->
\label{prop:tropical-dominance}
For an interior index $j$ with $a_j>0$,
\[
    I_j=
    \begin{cases}
        [\ell_j,h_j],&\ell_j\leq h_j,\\
        \varnothing,&\ell_j>h_j.
    \end{cases}
\]
The upper convex hull gives the following description:
\begin{enumerate}
    \par\hangindent 40\p@ If $j=v_i$ is an interior hull vertex, then
    \[
        I_j=[\alpha_i,\alpha_{i+1}].
    \]
    \par\hangindent 40\p@ If $(j,\log a_j)$ lies in the relative interior of the hull segment associated with $\alpha_i$, then
    \[
        I_j=\{\alpha_i\}.
    \]
    \par\hangindent 40\p@ If $(j,\log a_j)$ lies strictly below the
    upper hull, then $I_j=\varnothing$.
\end{enumerate}
The endpoint intervals are
\[
    I_0=(0,\alpha_1],
    \qquad
    I_d=[\alpha_s,\infty).
\]
<!-- end proposition-4 -->
<!-- proof-4: body_md -->
For $k<j$ with $a_k>0$,
\[
    a_kr^k\leq a_jr^j
    \quad\text{ if and only if }\quad
    r\geq\left(\frac{a_k}{a_j}\right)^{1/(j-k)}.
\]
For $k>j$ with $a_k>0$,
\[
    a_kr^k\leq a_jr^j
    \quad\text{ if and only if }\quad
    r\leq\left(\frac{a_j}{a_k}\right)^{1/(k-j)}.
\]
Terms with $a_k=0$ impose no restriction.
Intersecting these inequalities gives the formula for $I_j$.

For an interior hull vertex $v_i$, the line
$\log a_{v_i}+v_i u$ attains the maximum defining
$\phi_P(u)$ precisely between the consecutive breakpoints
$\log\alpha_i$ and $\log\alpha_{i+1}$, including both
endpoints. This gives $I_{v_i}=[\alpha_i,\alpha_{i+1}]$.

A coefficient point inside a hull segment attains the
maximum only at the breakpoint associated with that
segment. A point strictly below the hull never attains
the maximum. Finally, the terms of degree zero and $d$
dominate before the first breakpoint and after the last,
respectively, giving the endpoint intervals.

<!-- corollary-1: body_md -->
\label{cor:dominance-covering}
The dominance intervals associated with the upper hull vertices cover all positive radii:
\[
    \bigcup_{i=0}^{s} I_{v_i}=(0,\infty).
\]
Consequently,
\[
    \bigcup_{i=0}^{s}\mathcal D_{v_i}
    =\mathbb C\setminus\{0\}.
\]
With the convention that $0\in\mathcal D_0$, these
regions cover the whole complex plane.

No other coefficient indices are needed for this
covering.
Points below the hull have empty dominance intervals, and points inside hull segments have singleton intervals already contained in those of
the adjacent hull vertices.
<!-- end corollary-1 -->
<!-- proof-5: body_md -->
The vertex intervals are
\[
    I_{v_0}=(0,\alpha_1],\qquad
    I_{v_i}=[\alpha_i,\alpha_{i+1}]
    \quad(1\leq i\leq s-1),\qquad
    I_{v_s}=[\alpha_s,\infty).
\]
Their union is $(0,\infty)$.
A point inside the $i$th hull segment has dominance interval $\{\alpha_i\}$, which lies in both $I_{v_{i-1}}$ and $I_{v_i}$.

<!-- lemma-3: body_md -->
\label{lem:tropical-coefficient-decay}
Suppose $s\geq2$, and let $j=v_i$ be an interior hull vertex, with $1\leq i\leq s-1$.
Then
\begin{equation}
\label{eq:tropical-coefficient-decay}
    \|Q_{j,j}\|_2=1,
    \qquad
    \|Q_{j,k}\|_2\leq U_j^{-|k-j|},
    \qquad 0\leq k\leq d.
\end{equation}
Moreover, for $U_j^{-1}\leq r\leq U_j$,
\[
    T_{Q_j}(r)=r^j,
    \qquad
    r^j\leq S_{Q_j}(r)\leq(d+1)r^j.
\]
<!-- end lemma-3 -->
<!-- proof-6: body_md -->
For $k<j$, dominance at $\ell_j$ gives
\[
    a_k\ell_j^k\leq a_j\ell_j^j.
\]
Consequently,
\[
    \|Q_{j,k}\|_2
    =\frac{a_k}{a_j}\gamma_j^{k-j}
    \leq
    \left(\frac{\ell_j}{\gamma_j}\right)^{j-k}
    =U_j^{-(j-k)}.
\]
For $k>j$, dominance at $h_j$ similarly gives
\[
    \|Q_{j,k}\|_2
    \leq
    \left(\frac{\gamma_j}{h_j}\right)^{k-j}
    =U_j^{-(k-j)}.
\]
These inequalities also hold when $a_k=0$.
For $k=j$, the normalization gives
$\|Q_{j,j}\|_2=1$.

If $U_j^{-1}\leq r\leq U_j$, then
$\gamma_jr\in I_j$. Therefore
\[
    T_{Q_j}(r)
    =\frac{T_P(\gamma_jr)}{a_j\gamma_j^j}
    =r^j.
\]
The bounds for $S_{Q_j}(r)$ follow from
\eqref{eq:tropical-sum-comparison}.

<!-- proposition-5: body_md -->
\label{prop:interior-strong-linearization}
The pencil $S_{j,U}$ is a strong linearization of $Q$.
<!-- end proposition-5 -->
<!-- proof-7: body_md -->
Multiply block column $h$ by $U^{-\max\{h-j,0\}}$, for $0\leq h\leq d-1$.
The body row becomes
\[
    [Q_{d-1}+tQ_d,\ Q_{d-2},\ldots,Q_0].
\]
For each auxiliary row indexed by $h$, multiply by $-U^{h-j}$ if $h>j$, and by $U$ if $h\leq j$.
Its two nonzero blocks then become $[-I_n,tI_n]$.

These constant invertible row and column scalings transform $S_{j,U}$ into the Frobenius companion pencil $C_Q$. Since $C_Q$ is a strong linearization of $Q$, so is $S_{j,U}$.

<!-- proposition-6: body_md -->
\label{prop:interior-one-sided-factorizations}
For $t\neq0$ and $0\leq h\leq d-1$, there are matrix
functions $F_h(t)$ and $G_h(t)$ such that
\begin{align}
\label{eq:interior-all-right-factorizations}
    G_h(t)S(t)&=\beta_h(t)Q(t)E_h^*,\\
\label{eq:interior-all-left-factorizations}
    S(t)F_h(t)&=E_{\rm b}\,\beta_h(t)Q(t),
\end{align}
where
\[
    \beta_h(t)=U^{\max\{h-j,0\}}t^{-h}.
\]
They satisfy
\[
    E_h^*F_h(t)=I_n,
    \qquad
    G_h(t)E_{\rm b}=I_n.
\]
<!-- end proposition-6 -->
<!-- proof-8: body_md -->
For $t\neq0$, define
\[
    c_k(t)=
    \begin{cases}
        (t/U)^{k-j}, & k>j,\\
        t^{k-j}, & k\leq j.
    \end{cases}
\]
Then
\[
    F_h(t)=
    \begin{bmatrix}
        \dfrac{c_{d-1}(t)}{c_h(t)}I_n\\
        \vdots\\
        \dfrac{c_0(t)}{c_h(t)}I_n
    \end{bmatrix}
    =
    U^{\max\{h-j,0\}}t^{j-h}
    \begin{bmatrix}
        (t/U)^{d-1-j}I_n\\
        \vdots\\
        (t/U)I_n\\
        I_n\\
        t^{-1}I_n\\
        \vdots\\
        t^{-j}I_n
    \end{bmatrix}.
\]
The blocks above $I_n$ are absent when $j=d-1$.
Equivalently, the block indexed by $k$ is
\[
    (F_h(t))_k
    =
    U^{\max\{h-j,0\}-\max\{k-j,0\}}
    t^{k-h}I_n,
    \qquad 0\leq k\leq d-1.
\]



The blocks of $F_h(t)$ satisfy the auxiliary equations,
and its block indexed by $h$ is $I_n$. Hence
\[
    E_h^*F_h(t)=I_n.
\]
Substitution into the body row gives
\[
    S(t)F_h(t)
    =E_{\rm b}\frac{t^{-j}}{c_h(t)}Q(t)
    =E_{\rm b}\,\beta_h(t)Q(t),
\]
where
\[
    \beta_h(t)
    =\frac{t^{-j}}{c_h(t)}
    =U^{\max\{h-j,0\}}t^{-h}.
\]

For the other factorization, let $B_h(t)$ be the
matrix obtained by deleting column $h$ from the
auxiliary rows, and let $C_h(t)$ be the body row
with the same column deleted.

The matrix $B_h(t)$ is invertible. Indeed, a vector
in its kernel extends to an auxiliary solution with
block $h$ equal to zero. The auxiliary equations then
force every block to vanish.

Set
\[
    G_h(t)=[I_n,-C_h(t)B_h(t)^{-1}].
\]
Then $G_h(t)S(t)$ vanishes in every column except $h$.
Since $E_h^*F_h(t)=I_n$, its remaining block is
\[
    G_h(t)S(t)F_h(t)=\beta_h(t)Q(t).
\]
This proves the first factorization.
The first block of $G_h(t)$ is $I_n$, giving
$G_h(t)E_{\rm b}=I_n$.

<!-- proof-10: body_md -->
We first establish the bounds for $S=S_{j,U}$ and a
polynomial $Q$ satisfying
\begin{equation}
\label{eq:interior-coefficient-assumptions}
    \|Q_j\|_2=1,
    \qquad
    \|Q_k\|_2\leq U^{-|k-j|},
    \qquad 0\leq k\leq d.
\end{equation}
Use right recovery from position $d-j$ and left
recovery from position $1$.

Write $r=|t|$ and $S(t)=tX+Y$, with $1\leq r\leq U$.
The $t$-dependent body block and auxiliary part each
have norm at most $U^{-1}$, so $\|X\|_2\leq2/U$.
Every constant body block has norm at most one,
and the constant auxiliary part has norm at most one.
Hence $\|Y\|_2\leq\sqrt d+1$, giving
\begin{equation}
\label{eq:interior-pencil-weight-bound}
    W_S(r)\leq 2r/U+\sqrt d+1\leq A_d.
\end{equation}
Also, $S_Q(r)\geq r^j$.

The one-sided factorizations for these recovery blocks
give
\[
    g_{\rm R}(t)
    =\frac{W_S(r)r^j\|G_j(t)\|_2}{S_Q(r)},
    \qquad
    g_{\rm L}(t)
    =\frac{W_S(r)r^j\|F_j(t)\|_2}{S_Q(r)}.
\]
Every scalar block of $F_j(t)$ has modulus at most
one on this region, so $\|F_j(t)\|_2\leq\sqrt d$.

To bound $G_j$, write
\[
    G_j(t)=[I_n,G_{j,d-1}(t),\ldots,G_{j,1}(t)],
\]
where $G_{j,q}$ corresponds to auxiliary row $q$.
Cancellation of the columns other than $j$ gives
\[
    G_{j,q}(t)
    =
    \begin{cases}
        \displaystyle
        -U\sum_{k=0}^{q-1}Q_kt^{-(q-k)},
        &1\leq q\leq j,\\[6pt]
        \displaystyle
        -\sum_{k=q}^{d-1}
          U^{k-j}Q_k(t/U)^{k-q}
        -tU^{d-1-j}Q_d(t/U)^{d-1-q},
        &j<q\leq d-1.
    \end{cases}
\]
Each summand has norm at most one. For the first case,
this follows from
\[
    U\|Q_k\|_2r^{-(q-k)}
    \leq U^{1-(j-k)}r^{-(q-k)}\leq1.
\]
For the second case, each term in the sum has norm
at most $(r/U)^{k-q}\leq1$. The final term satisfies
\[
    \bigl\|tU^{d-1-j}Q_d(t/U)^{d-1-q}\bigr\|_2
    \leq
    rU^{-1}(r/U)^{d-1-q}
    =(r/U)^{d-q}\leq1.
\]
Therefore,
\[
    \|G_{j,q}(t)\|_2
    \leq
    \begin{cases}
        q, & q\leq j,\\
        d-q+1, & q>j.
    \end{cases}
\]
Consequently,
\[
    \|G_j(t)\|_2^2
    \leq
    1+\sum_{q=1}^{j}q^2
      +\sum_{q=j+1}^{d-1}(d-q+1)^2
    \leq
    1+\sum_{q=1}^{d-1}q^2
    =B_d^2,
\]
where the second inequality follows from
\[
    \sum_{q=1}^{d-1}q^2
    -\sum_{q=1}^{j}q^2
    -\sum_{q=j+1}^{d-1}(d-q+1)^2
    =(j-1)(d-j-1)(d+1)\geq0.
\]
Together with \eqref{eq:interior-pencil-weight-bound}
and $S_Q(r)\geq r^j$, these estimates prove the
direct bounds.

For $L_j^+$, apply these bounds to $Q_j$ with
$t=\lambda/\gamma_j$.
For $L_j^-$, apply them to $\widetilde Q_j$ with
dominant index $d-j$ and $s=\gamma_j/\lambda$.
The respective assigned regions give
$1\leq|t|\leq U_j$ and $1\leq|s|\leq U_j$.
Scaling and reversal preserve the residual amplification
factors, so the same bounds hold in the original variable.

Let $1\leq|t|\leq U$.

<!-- proof-14: body_md -->
Corollary~\ref{prop:interior-eigenvector-recovery}
gives $x,y\neq0$ and
\[
    v=F_j(t)x,
    \qquad
    w^*=y^*G_j(t).
\]
Adding the squared right block norms gives
\[
    \|v\|_2^2
    =
    \left(
        \sum_{k=0}^{p}(|t|/U)^{2k}
        +\sum_{k=1}^{j}|t|^{-2k}
    \right)\|x\|_2^2.
\]
The sum contains $d$ terms, each at most one,
and includes a term equal to one.
This proves the right statement.

The multiplier estimate established in the proof of
Proposition~\ref{prop:half-annulus-residual-amplification}
gives $\|G_j(t)\|_2\leq B_d$.
Hence
\[
    \|w\|_2=\|y^*G_j(t)\|_2\leq B_d\|y\|_2.
\]
Block extraction gives the lower bound.

<!-- proof-15: body_md -->
For the large endpoint, set
$t=\lambda/\gamma_j$ and $r=|t|$.
Since $\alpha_s=\gamma_jU_j$, we have $r\geq U_j$.
The block formulas give
\[
    \|v\|_2^2
    =
    \left(
        \sum_{k=0}^{p}(U_j/r)^{2k}
        +U_j^{2p}\sum_{k=1}^{j}r^{-2(p+k)}
    \right)\|x\|_2^2.
\]
Each of the $d$ terms is at most one:
in the second sum,
\[
    U_j^{2p}r^{-2(p+k)}
    =(U_j/r)^{2p}r^{-2k}\leq1.
\]
The first term is one, giving the right bounds.

With right recovery from position $1$, the corresponding
matrix $G_{d-1}(t)$ gives
\[
    w^*=y^*G_{d-1}(t).
\]
The endpoint-extension proof established
$\|G_{d-1}(t)\|_2\leq B_d$.
This gives the left upper bound.
Nonvanishing follows from the extraction result,
and block extraction gives the lower bound.

The small endpoint uses this same construction for
the reversed polynomial, with dominant index $d-j$
and variable $s=\gamma_j/\lambda$.
Here $|s|\geq U_j$, so the preceding argument applies.
Pencil reversal multiplies the evaluated chart pencil
by a nonzero scalar and therefore leaves its
eigenvectors and recovery ratios unchanged.

<!-- theorem-2: body_md -->
\label{thm:exact-selected-ratios}
Let $\lambda\neq0$ belong to the assigned region
of a pencil in the linearization covering.
For every nonzero right and left pencil eigenvector,
the prescribed recovered vectors are nonzero and
\[
    1\leq\rho_{\rm R}(v)\leq\sqrt d.
\]
The left ratio satisfies
\[
    1\leq\rho_{\rm L}(w)\leq
    \begin{cases}
        B_d,&s\geq2,\\
        \widehat B_d,&s=1.
    \end{cases}
\]
In particular, $\widehat B_d$ is a common left bound
for every case.

The bounds include all region boundaries.
They require neither simple eigenvalues nor
invertible coefficient matrices.
<!-- end theorem-2 -->
<!-- proof-17: body_md -->
The interior, endpoint, and single-root results cover
every assigned region.
Scaling preserves the pencil vectors, and reversal
changes an evaluated pencil only by a nonzero scalar
relative to its reversed chart.
Thus, the chart recovery ratios are also the ratios
for the original-variable pencils.

All arguments apply to every nonzero null vector,
without any simplicity or coefficient-invertibility
assumption. Finally, $B_d\leq\widehat B_d$.

<!-- corollary-3: body_md -->
For every pencil in the covering, at every nonzero
finite eigenvalue in its assigned region, the prescribed
recovery rules satisfy
\[
    \Gamma_{\rm R}(\lambda,v)\leq C_d,
    \qquad
    \Gamma_{\rm L}(\lambda,w)\leq C_d
\]
for all nonzero exact right and left pencil eigenvectors,
where
\[
    C_d=
    \begin{cases}
        A_dB_d\sqrt d, & s\geq2,\\
        A_d\widehat B_d\sqrt d, & s=1.
    \end{cases}
\]
In both cases, $C_d=O(d^{5/2})$.
