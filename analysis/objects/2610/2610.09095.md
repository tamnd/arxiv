\label{thm:cor}
    The following are equivalent.
    \begin{enumerate}
        \item The polygon $\Omega$ is canonical.
        \item All Laplacian eigenfunctions on $\Omega$ are finite sums of plane waves $$\sum_{k=1}^N a_k \cos(m_k x + n_k y) + b_k \sin(p_k x + q_k y) ,$$ with $a_k,b_k,m_k,n_k,p_k,q_k\in\mathbb{R}$.
        \item The ground eigenfunction is smooth on the closure of $\Omega$, i.e., $u_1\in C^\infty(\overline{\Omega})$.
    \end{enumerate}

\label{thm:thm}
     For both $\theta=0$ and $\theta=\alpha$, there exists some $\rho >0$ such that the following holds.
    \begin{enumerate}
        \item For $\alpha\in(0,\pi)$, if $\pi/\alpha\in\mathbb{N}$ then $\partial_\nu u(\cdot,\theta) \in C^\infty([0,\rho))$.
        \item Otherwise, 
        $$\partial_\nu u(\cdot,\theta)\in C^{k(\alpha),\beta(\alpha)}([0,\rho)), \text{ where }k(\alpha) = \lfloor\pi/\alpha\rfloor-1 \text{ and }\beta(\alpha)= \pi/\alpha - \lfloor\pi/\alpha\rfloor\in(0,1).$$
        \item For the first eigenfunction, the choices of $k(\alpha)$ and $\beta(\alpha)$ are sharp, in the sense that $\partial_\nu u_1(\cdot,\theta) \notin C^{\tilde{k},\tilde{\beta}}([0,\rho))$ for any $\tilde{k}>k(\alpha)$ or $\tilde{\beta}>\beta(\alpha)$. Otherwise, if $\alpha\in(\pi,2\pi)$ then the normal derivative of the first eigenfunction cannot be extended to a continuous function. That is,
        $$\lim_{r\to 0} |\partial_\nu u_1(r,\theta)|=\infty.$$
    \end{enumerate}

Let $\tilde{u}:(0,\rho)\times(0,\alpha)\to\mathbb{R}$ denote the restriction of $u$ to $D$ in local polar coordinates. 
    \begin{enumerate}
        \item The function $\tilde{u}$ is given by the series
            \begin{equation}
                \tilde{u}(r,\theta) = \sum_{n = 1}^\infty a_n J_{n\pi/\alpha}(\sqrt{\lambda}r)\sin\left(\frac{n\pi\theta}{\alpha}\right)
                \label{eqn:sectorefunc}
            \end{equation}
            with coefficients
            \begin{equation}
                a_n = \frac{2}{\alpha J_{n\pi/\alpha}(\sqrt{\lambda}\rho)}\int_0^\alpha \tilde{u}(\rho,\phi)\sin\left(\frac{n\pi\phi}{\alpha}\right)\,d\phi,\quad n\in\mathbb{N}.
                \label{eqn:coeffs}
            \end{equation}
        \item The series~\eqref{eqn:sectorefunc} converges uniformly for $r<\rho$.
    \end{enumerate}

<!-- proof-1: body_md -->
\begin{enumerate}
        \item Consider the inhomogeneous problem on $D$
    \begin{equation}
        \begin{cases}
            (\Delta +{\lambda})\varphi (r,\theta) = 0 & (r,\theta)\in(0,\rho)\times(0,\alpha)\\
            \varphi(r,0) = \varphi(r,\alpha) = 0 & r\in (0,\rho)\\
            \varphi(\rho,\theta) = G(\theta)
            \label{eqn:poisson}
        \end{cases}
    \end{equation}
    where ${\lambda}$ and $G(\theta) = \tilde{u}(\rho,\theta)$ are given. Firstly, $\tilde{u}$ is automatically a solution to this problem. Equation~\eqref{eqn:poisson} can also be solved using separation of variables to give
    \begin{equation}
        \psi(r,\theta) = \sum_{n = 1}^\infty a_n J_{n\pi/\alpha}(\sqrt{{\lambda}}r)\sin\left(\frac{n\pi\theta}{\alpha}\right),
        \label{eqn:series}
    \end{equation}
    as in \cite{BanjBoul}, where $J_m$ is an $m$th order Bessel function of the first kind and
    $$a_n = \frac{2}{\alpha J_{n\pi/\alpha}(\sqrt{\lambda} \rho)}\int_0^\alpha G(\phi)\sin(n\pi\phi/\alpha)\,d\phi.$$
    Finally we claim that a solution to this problem is unique. Consider the problem solved by $\tilde{u}-\psi$ where $\psi$ is the series \eqref{eqn:series}:
    \begin{equation*}
        \begin{cases}
            (\Delta + {\lambda})(\tilde{u}-\psi) = 0&\text{in }D\\
            \tilde{u}-\psi = 0 &\text{on }\partial D.
        \end{cases}
    \end{equation*}
    If the operator $\Delta + \lambda$ is injective then $\tilde{u}_1-\psi$ is identically 0. This occurs precisely when $\lambda$ is not a Dirichlet Laplacian eigenvalue on $D$. The eigenvalues on $D$ form a discrete subset of $[0,\infty)$ with the only accumulation point at $\infty$. When we scale $\rho$ by $1-\varepsilon$ for $\varepsilon>0$, all eigenvalues on $D$ are scaled by $(1-\varepsilon)^{-2}$. If $\lambda$ does happen to be an eigenvalue on $D$ then we can choose a smaller, but arbitrarily close, value of $\rho$ and repeat the argument to prove the first statement.
    \item We first bound the coefficients~\eqref{eqn:coeffs} from above:
    \begin{equation*}
        |a_n| = \left| \frac{2}{\alpha J_{n\pi/\alpha}(\sqrt{\lambda} \rho)}\int_0^\alpha \tilde{u}(\rho,\phi)\sin(n\pi\phi/\alpha)\,d\phi\right|\leq \frac{2\|\tilde{u} \|_{L^\infty(D)}}{J_{n\pi/\alpha}(\sqrt{{\lambda }}\rho)}.
    \end{equation*}
    Consider the integral representation for Bessel functions of order $\beta>-1/2$ \cite[Equation 17.3.1, ][]{WhitWats}
    \begin{equation}
        J_\beta(z) = \frac{2(z/2)^\beta}{\sqrt{\pi}\Gamma(\beta+1/2)}\int_0^1 (1-t^2)^{\beta-1/2}\cos(zt)\,dt,\, \text{Re}z>0
        \label{eqn:besselintegral}
    \end{equation}
    For arbitrary $\beta$ we choose the principal branch of $(z/2)^\beta$, which is analytic in $\text{Re} z>0$. However, for $\beta\in\mathbb{N}$ the representation is valid for any $z\in\mathbb{C}$. We use $|\cos(zt)| \le e^{|\Im z|}$ for $z\in\mathbb{C}$ and $t\in(0,1)$ to provide an upper bound
    \begin{equation}
        |J_\beta(z)| \le \underbrace{\frac{\int_0^1 (1-t^2)^{\beta-1/2}\,dt}{\sqrt\pi 2^{\beta-1}\Gamma(\beta+1/2)}}_{=: C_\beta}z^{\beta} e^{|\Im z|}.
        \label{eqn:besselupper}
    \end{equation}
    For $x\in[0,\pi/2]$ we can use $\cos(xt) \ge \cos(x)$ for $t\in(0,1)$ to provide the lower bound
    \begin{equation}
        C_\beta\cos(x)x^\beta \le J_\beta(x).
        \label{eqn:bessellower}
    \end{equation}
    According to \eqref{eqn:besselupper} and \eqref{eqn:bessellower}, each term of the series \eqref{eqn:sectorefunc} is bounded as follows, once we restrict $\sqrt{\lambda}\rho<\pi/2$:
    \begin{align}
        \left|a_n J_{n\pi/\alpha}(\sqrt{\lambda } r)\right | &\leq \frac{2\|\tilde{u} \|_{L^\infty(D)}}{J_{n\pi/\alpha}(\sqrt{\lambda }\rho)} J_{n\pi/\alpha}(\sqrt{\lambda } r)\nonumber\\
        &\le \frac{2\|\tilde{u} \|_{L^\infty(D)}C_{n\pi/\alpha}}{(\sqrt{{\lambda }}\rho)^{n\pi/\alpha}\cos(\sqrt{\lambda }\rho)} \frac{(\sqrt{{\lambda }}r)^{n\pi/\alpha}}{C_{n\pi/\alpha}}\nonumber\\
        &=  \frac{2\|\tilde{u} \|_{L^\infty(D)}}{\cos(\sqrt{{\lambda }}\rho)}\left(\frac r\rho\right)^{n\pi/\alpha}.
        \label{eqn:summandbound}
    \end{align}
    With
    \begin{equation*}
        M_n = \frac{2\|\tilde{u} \|_{L^\infty(D)}}{\cos(\sqrt{{\lambda }}\rho)}\left(\frac {R^{\pi/\alpha}}{\rho^{\pi/\alpha}}\right)^n,
    \end{equation*}
    the Weierstrass M-test implies that the series \eqref{eqn:sectorefunc} converges uniformly for $(r,\theta) \in [0,R]\times[0,\alpha]$ for any $R\leq\frac{\pi}{2\sqrt{{\lambda }}}$
    as
    \begin{equation*}
        \sum_{n=1}^\infty M_n < \infty.
    \end{equation*}
    \end{enumerate}

\label{lemma:2.2}
    Assume the same hypotheses are in Lemma~\ref{thm:series}. Then we have
    \begin{equation}
        \partial_\nu \tilde{u} (r,\theta) =-\frac{\pi}{\alpha}\frac{a_1}{\Gamma(\pi/\alpha+1)}\left(\frac{\sqrt{\lambda }}{2}\right)^{\pi/\alpha}r^{\pi/\alpha-1} + o(r^{\pi/\alpha-1}),
        \label{eqn:asymptotic}
    \end{equation}
    as $r\to 0^+$.

<!-- proof-2: body_md -->
At the point $(r,\theta)$ we use $\{\bf{e_r},\bf{e_\theta}\}$ as a local orthonormal basis for $\mathbb{R}^2$, where the former is the unit vector in the radial direction of $(r,\theta)$ and the latter is the orthogonal tangential vector in the anti-clockwise direction. We find the normal derivative by differentiating \eqref{eqn:sectorefunc} term-wise, thanks to Lemma~\ref{thm:series}, and by using that, in this basis, the outward unit normal vector to the edge $\theta=0$ is $-\mathbf{e_\theta}$:
    \begin{align}
        \partial_\nu \tilde{u} (r,0) &= -\mathbf{e_\theta} \cdot\nabla \tilde{u} (r,\theta)\big|_{\theta=0}\nonumber\\
        &= -\mathbf{e_\theta}\cdot\left(\frac{\partial \tilde{u} }{\partial r} \mathbf{e_r} + \frac{1}{r}\frac{\partial \tilde{u} }{\partial\theta}\mathbf{e_\theta}\right)\Bigg|_{\theta=0}\nonumber\\
        &= -\frac{1}{r} \frac{\partial \tilde{u} (r,\theta)}{\partial\theta}\Bigg|_{\theta=0}\nonumber\\
        &= -\sum_{n = 1}^\infty \frac{a_nn\pi}{\alpha}\frac{J_{n\pi/\alpha}(\sqrt{\lambda }r)}{r}\cos\left(\frac{n\pi\theta}{\alpha}\right)\Bigg|_{\theta=0}\nonumber\\
        &= -\sum_{n = 1}^\infty \frac{a_nn\pi}{\alpha}\frac{J_{n\pi/\alpha}(\sqrt{\lambda }r)}{r}.
        \label{eqn:normal1}
    \end{align}
    The outward unit normal vector to the edge $\theta=\alpha$ is $\mathbf{e_\theta}$ so
    \begin{equation} 
        \partial_\nu \tilde{u} (r,\alpha) = \mathbf{e_\theta} \cdot\nabla \tilde{u} (r,\theta)\big|_{\theta=\alpha} = \sum_{n = 1}^\infty(-1)^n \frac{a_nn\pi}{\alpha}\frac{J_{n\pi/\alpha}(\sqrt{\lambda }r)}{r}.
        \label{eqn:normal2}
    \end{equation}
    We use the series~\eqref{eqn:normal1} to characterise the limit equation~\eqref{eqn:asymptotic} by splitting into a sum of limits, the existence of which is justified \textit{a posteriori}:
    \begin{align}
        &\left(\sum_{n=1}^\infty \frac{n\pi a_n}{\alpha} \frac{J_{n\pi/\alpha}(\sqrt{\lambda }r)}{r} - \frac{a_1 \pi}{\alpha\Gamma(\pi/\alpha+1)}\left(\frac{\sqrt{\lambda }}{2}\right)^{\pi/\alpha}r^{\pi/\alpha-1}\right)\bigg/r^{\pi/\alpha-1} \to 0\nonumber\\
        \iff &\sum_{n=1}^\infty n a_n \frac{J_{n\pi/\alpha}(\sqrt{\lambda }r)}{r^{\pi/\alpha}} -  \frac{a_1}{\Gamma(\pi/\alpha+1)}\left(\frac{\sqrt{\lambda }}{2}\right)^{\pi/\alpha}\to 0\nonumber\\
        \iff& \sum_{n=2}^\infty n a_n \frac{J_{n\pi/\alpha}(\sqrt{\lambda }r)}{r^{\pi/\alpha}} + a_1\frac{J_{\pi/\alpha}(\sqrt{\lambda }r)}{r^{\pi/\alpha}} -  \frac{a_1}{\Gamma(\pi/\alpha+1)}\left(\frac{\sqrt{\lambda }}{2}\right)^{\pi/\alpha}\to 0\nonumber\\
        \Longleftarrow&\underbrace{\sum_{n=2}^\infty n a_n \frac{J_{n\pi/\alpha}(\sqrt{\lambda }r)}{r^{\pi/\alpha}} \to 0}_{\textbf{(I)}} \text{ and }\underbrace{\frac{J_{\pi/\alpha}(\sqrt{\lambda }r)}{r^{\pi/\alpha}} - \frac{1}{\Gamma(\pi/\alpha+1)}\left(\frac{\sqrt{\lambda }}{2}\right)^{\pi/\alpha}\to 0}_{\textbf{(II)}}.\nonumber
    \end{align}
    Claim \textbf{(II)} asserts that the leading order behaviour of the Bessel function comes from the lowest power in its Frobenius expansion and claim $\textbf{(I)}$ precludes any contributions from the tail sum at leading order. We begin with claim \textbf{(I)}, using inequality~\eqref{eqn:summandbound} to bound the summands
    \begin{align*}
        \left|\sum_{n=2}^\infty na_n \frac{J_{n\pi/\alpha}(\sqrt{\lambda }r)}{r^{\pi/\alpha}}\right| 
        \le& \frac{1}{r^{\pi/\alpha}}\sum_{n=2}^\infty n\left|a_nJ_{n\pi/\alpha}(\sqrt{\lambda }r)\right| \nonumber\\
        \le&\frac{2\|u\|_\infty}{r^{\pi/\alpha}\cos(\sqrt{\lambda }\rho)}\sum_{n=2}^\infty n\left(\frac{r^{\pi/\alpha}}{\rho^{\pi/\alpha}}\right)^{n}\\
        =& \frac{2\|u\|_\infty}{r^{\pi/\alpha}\cos(\sqrt{\lambda }\rho)}\left[\frac{(r/\rho)^{\pi/\alpha}}{(1-(r/\rho)^{\pi/\alpha})^2} - (r/\rho)^{\pi/\alpha}\right]\nonumber,
    \end{align*}
    which tends to 0 as $r\to 0$. In the final step we used that, for $|t|<1$,
    $$\sum_{n=1}^\infty nt^n = t\sum_{n=1}^\infty \frac{d}{dt}t^n= t\frac{d}{dt}\frac{1}{1-t} = \frac{t}{(1-t)^2}.$$
    
   Claim \textbf{(II)} follows directly from the Frobenius series for Bessel functions:
    \begin{equation*}
        J_{\pi/\alpha} (r) = \sum_{m=0}^\infty \frac{(-1)^m}{m!\,\Gamma(m+\pi/\alpha+1)}\Big(\frac{r}{2}\Big)^{2m+\pi/\alpha}.
    \end{equation*}
    Indeed,
    \begin{align}
        &\left|\frac{J_{\pi/\alpha}(\sqrt{\lambda }r)}{r^{\pi/\alpha}} -  \frac{1}{\Gamma({\pi/\alpha}+1)}\left(\frac{\sqrt{\lambda }}{2}\right)^{\pi/\alpha}\right|\nonumber\\
        =& \left|\frac{1}{r^{\pi/\alpha}}\left(\sum_{m=0}^\infty \frac{(-1)^m}{m!\Gamma(m+{\pi/\alpha}+1)}\left(\frac{\sqrt{\lambda }r}{2}\right)^{2m+{\pi/\alpha}} \right) -  \frac{1}{\Gamma({\pi/\alpha}+1)}\left(\frac{\sqrt{\lambda }}{2}\right)^{\pi/\alpha}\right|\nonumber\\
        =& \left(\frac{\sqrt{\lambda }}{2}\right)^{\pi/\alpha}\left|\sum_{m=1}^\infty \frac{(-1)^m}{m!\Gamma(m+{\pi/\alpha}+1)}\left(\frac{\sqrt{\lambda }r}{2}\right)^{2m}\right|\nonumber\\
        \le& \left(\frac{\sqrt{\lambda }}{2}\right)^{\pi/\alpha}\sum_{m=1}^\infty \frac{1}{m!}\left(\frac{\lambda r^2}{4}\right)^{m} =  \left(\frac{\sqrt{\lambda }}{2}\right)^{\pi/\alpha}\left(\exp\left(\lambda r^2/4\right)-1\right).\nonumber
    \end{align}
    Taking the limit as $r\to0$ proves the claim. An identical argument proves the claim for the edge $\theta=\alpha$.

We first prove the positive statements. We let $\theta=0$ and the $\theta=\alpha$  case is analogous.
    \begin{enumerate}
        \item If $\pi/\alpha\in\mathbb{N}$ then every Bessel function in \eqref{eqn:normal1} has integer order and is thus infinitely differentiable. However, uniform convergence does not necessarily confer differentiability on the limit function. For this to be true we need complex analyticity, hence we consider a complex extension of the series~\eqref{eqn:normal1}. Let $U = \{ z\in\mathbb{C} : |z|<\pi/2\}$ and
        \begin{equation*}
            F(z) = \frac{1}{ z}\sum_{n=1}^\infty na_n J_{n\pi/\alpha}(z), \quad z\in U,
        \end{equation*}
        so that $\partial_\nu \tilde{u}(r,0) = -\frac{\pi}{\alpha}F(\sqrt{\lambda}r)$ for $r\in[0,\rho)$. The sequence of functions given by the partial sums,
        $$F_N(z) = \frac1z\sum_{n=1}^N a_n n J_{n\pi/\alpha}(z),$$
        converge uniformly to $F$ on compact subsets of $U$ by applying inequality~\eqref{eqn:summandbound} and invoking the Weierstrass M-test. As the integral representation~\eqref{eqn:besselintegral} is valid for all $z\in\mathbb{C}$ when $\pi/\alpha\in\mathbb{N}$, convergence of $F_N$ is uniform on the disc $U$. As Bessel functions of integer order are entire,
        $$\int_\gamma F_N(z)\,dz = 0,$$
        where $\gamma$ is any contour inside $U$. Uniformity of the convergence of $F_N$ to $F$ allows interchanging limits, so
        $$0 = \lim_{N\to \infty} \int_\gamma F_N(z)\,dz = \int_\gamma \lim_{N\to\infty} F_N(z)\,dz = \int_\gamma F(z)\,dz. $$
        As this is true for all contours $\gamma$ inside $U$, Morera's theorem implies that $F$ is analytic on $U$. Hence, $\partial_\nu u$ is infinitely differentiable at $r = 0$.
                
        \item Suppose $\alpha\in(0,\pi)$ and $\pi/\alpha \notin \mathbb{N}$. 
        As all derivatives of $u$ can be extended to the $C^\infty$ portions of the boundary, the only point that remains to check is $r=0$. By the identity~\eqref{eqn:kderiv}, we see that the first $k$ derivatives where $\pi/\alpha-1-k >0$ are continuous and take the value $0$ at the point $r=0$. As for the H\"older continuity, we need only consider the limit as $r\to 0$. Let $\beta\in(0,1)$ and consider
        \begin{align}
            & \frac{1}{r^\beta}\left( \frac{d^{k(\alpha)}}{dr^{k(\alpha)}} \partial_\nu \tilde{u}(r,\theta)  - \frac{d^{k(\alpha)}}{dr^{k(\alpha)}} \partial_\nu \tilde{u}(0,\theta)\right)\nonumber\\
            =& \frac1{r^{\beta}} \left({ -\gamma \left(\frac{\pi}{\alpha}\right)_{(k(\alpha)+1)} r^{\pi/\alpha-1-k(\alpha)} + o\left(r^{\pi/\alpha-1-k(\alpha)}\right)}\right)\nonumber\\
            =& \left({ -\gamma\left(\frac{\pi}{\alpha}\right)_{(k(\alpha)+1)} r^{\pi/\alpha-1-k(\alpha)-\beta} + o\left(r^{\pi/\alpha-1-k(\alpha)-\beta}\right)}\right)\nonumber
        \end{align}
        as $r\to 0$. If $\beta$ is such that $\pi/\alpha-1-k(\alpha)-\beta \ge 0$ then the limit
        $$\lim_{r\to 0}\frac{ \frac{d^{k(\alpha)}}{dr^{k(\alpha)}} \partial_\nu \tilde{u}(r,\theta)  - \frac{d^{k(\alpha)}}{dr^{k(\alpha)}} \partial_\nu \tilde{u}(0,\theta)}{r^\beta}$$
        is finite. Thus $\partial_\nu u \in C^{k(\alpha),\beta(\alpha)}$ where $\beta(\alpha) = \pi/\alpha - 1 - k(\alpha) = \pi/\alpha - \lfloor \pi/\alpha\rfloor$.
    \end{enumerate}
    To prove the negative statements we restrict our attention to the ground eigenfunction.
    \begin{enumerate}[resume]  
        \item Letting $\tilde{k}>k(\alpha)$ implies that the exponent $\pi/\alpha -1-\tilde{k}$ in \eqref{eqn:kderiv} is negative, meaning the $\tilde{k}$th derivative of $\partial_\nu u_1$ grows without bound as $r\to0$.

        Now let $\tilde{\beta}>\beta(\alpha)$. Once again, the limit under consideration has behaviour
        \begin{equation*}
             { -\gamma\left(\frac{\pi}{\alpha}\right)_{(k(\alpha)+1)} r^{\pi/\alpha-1-k(\alpha)-\tilde{\beta}} + o\left(r^{\pi/\alpha-1-k(\alpha)-\tilde{\beta}}\right)}
        \end{equation*}
        as $r\to 0$. As the exponent is negative again, the limit does not exist, meaning $\partial_\nu \tilde{u}_1(r,\theta) \notin C^{k(\alpha),\tilde{\beta}}([0,\rho))$.
        
        For the second part, consider a re-entrant corner of angle $\alpha\in(\pi,2\pi)$. Because $\partial_\nu\tilde{u}_1^j = -\gamma\frac{\pi}{\alpha}r^{\pi/\alpha-1} + o(r^{\pi/\alpha-1})$, the exponent is always negative for this range of angles, meaning
        \begin{equation*}
            \lim_{r\to 0^+}|\partial_\nu u_1^j(r)| = \infty.
        \end{equation*}
    \end{enumerate}
    This concludes the proof.

<!-- proof-4: body_md -->
The eigenfunctions of the canonical polygons are listed in \cite{KuttSig} and they are all readily extended to smooth functions on all of $\mathbb{R}^2$. 

    It remains to show that the ground eigenfunction of a non-canonical polygon is not smooth on $\overline{\Omega}$. When searching for polygons with smooth eigenfunctions, we focus on convex polygons whose angles are all of the form $\pi/k$ with $k\in\mathbb{N}$; any non-convex polygon must have a singularity in its normal derivative at each re-entrant corner, by virtue of Theorem~\ref{thm:thm}. 
    
    As the sum of angles in a convex $n$-gon is $(n-2)\pi$, we seek $n\in\mathbb{N}$ and $\{k_i\}_{i=1}^n\subset\mathbb{N}$ such that
    \begin{equation}
        \sum_{i=1}^n \frac{\pi}{k_i} = (n-2)\pi.
        \label{eqn:dioph}
    \end{equation}
    Convexity of $\Omega$ implies that each $k_i \ge2$, meaning
    \begin{equation*}
        (n-2) = \sum_{i=1}^n \frac{1}{k_i}\le \frac{n}{2}\implies n\le 4.
    \end{equation*}
    First consider the case $n=4$. If, say, $k_1>2$ then we must have $k_i<2$ for some index $i\neq 1$, contradicting convexity. Thus equality in \eqref{eqn:dioph} is attained only by $k_i =2$ for $i=1,2,3,4$, corresponding to a rectangle.

    In the case $n=3$ we seek solutions to
    \begin{equation*}
        \frac{1}{k_1} + \frac{1}{k_2} + \frac{1}{k_3} = 1,\quad k_1,k_2,k_3\in\mathbb{N}.
    \end{equation*}
    At least one of the $k_i$ must be less than 4, otherwise the sum will not exceed $3/4$. Therefore we need only consider the cases when at least one of the integers, say $k_1$, is $2$ or $3$. If $k_1=2$,
    $$\frac{1}{k_2} + \frac{1}{k_3} = \frac{1}{2}\implies 2k_2 + 2l_3 = k_2k_3 \implies (k_2-2)(k_3-2) = 4.$$
    The only integer factorisations with $k_2,k_3>0$ are $4\cdot 1$ and $2\cdot 2$ leading to the solutions $(2,3,6)$ and $(2,4,4)$. If $k_1=3$ then
    $$\frac{1}{k_2} + \frac{1}{k_3} = \frac{2}{3} \implies 3k_2 + 3k_3 = 2k_2k_3 \implies (2k_2-3)(2k_3-3) = 9.$$
    The integer factorisations with $k_2,k_3>0$ are $3\cdot 3$ and $9\cdot 1$, corresponding to the solutions $(2,3,6)$ and $(3,3,3).$ Therefore the only solutions correspond to the canonical polygons.

    This exhaustive search means that a non-canonical polygon has at least one angle not of the form $\pi/n$ for $n\in\mathbb{N}$. Theorem~\ref{thm:thm} implies that neither $\partial_\nu u_1$ nor $u_1$ is infinitely differentiable at that corner.

<!-- remark-1: body_md -->
It is well-known that the eigenfunctions of the Dirichlet Laplacian form a complete set in $L^2(\Omega)$, however Lam\'e did not prove that his given set of eigenfunctions were complete. This was later established by Pinsky \cite{Pinsky}, whose proof uses the known completeness of the eigenfunctions of the rectangle. In \cite{FokKal}, Fokas and Kalimeris indicate that the completeness of the eigenfunctions they derived follows from employing the UTM to solve the heat equation in the interior of an equilateral triangle with Dirichlet boundary conditions and arbitrary initial data $q_0(x,y)$ \cite{HeatEqui}; evaluating the solution at $t=0$ gives the complete spectral decomposition of the arbitrary function $q_0$. However, some regularity of $q_0$ is required for the pointwise evaluation of the solution to be valid.
