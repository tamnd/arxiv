<!-- definition-1: body_md -->
\label{FormalDualDefiinition}
			The formal dual of the transform $R_{\alpha}$ given in Equation \eqref{OurTransformDefinition} is an operator $R_{\alpha}^*$ that takes functions on the affine Grassmannian and gives out functions on $\mathbb{R}^n$ and satisfies
			\begin{equation}
				\label{DualityEquation}
				\int\limits_{G \left( n, k \right)} R_{\alpha}f \left( \xi \right) \varphi \left( \xi \right) \mathrm{d}\xi = \int\limits_{\mathbb{R}^n} f \left( x \right) R_{\alpha}^* \varphi \left( x \right) \mathrm{d}x,
			\end{equation}
			whenever both the sides of the equation are finite in the Lebesgue sense.
<!-- end definition-1 -->
<!-- proposition-1: body_md -->
\label{DualTransform}
			Let $\varphi: G \left( n, k \right) \rightarrow \mathbb{C}$ be a ``nice" function. Then, we have,
			\begin{equation}
				\label{DualTransformEquation}
				R_{\alpha}^* \varphi \left( x \right) = \int\limits_{G_{n,  k}} \varphi \left( \pi, \text{Pr}_{\pi^{\perp}} \left( x \right) \right) \| \text{Pr}_{\pi} \left( x \right) \|^{\alpha - k} \mathrm{d}\pi,
			\end{equation}
			where, $\text{Pr}_{\pi^{\perp}}$ is the orthogonal projection onto $\pi^{\perp}$, $\text{Pr}_{\pi}$ is the orthogonal projection onto $\pi$, and $\mathrm{d}\pi$ is the normalized measure on the compact Grassmannian $G_{n, k}$, of $k$-dimensional subspaces of $\mathbb{R}^n$.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
We have, for all ``nice" functions $f: \mathbb{R}^n \rightarrow \mathbb{C}$, by Equation \eqref{OurTransformDefinition},
			\begin{align*}
				\int\limits_{G \left( n, k \right)} R_{\alpha}f \left( \pi, u \right) \varphi \left( \pi, u \right) \mathrm{d}u \mathrm{d}\pi &= \int\limits_{G_{n, k}} \int\limits_{\pi^{\perp}} \int\limits_{\pi} f \left( x + u \right) \| x \|^{\alpha - k} \varphi \left( \pi, u \right) \mathrm{d}x \mathrm{d}u \mathrm{d}\pi.
			\end{align*}
			We notice that as $x \in \pi$ and $u \in \pi^{\perp}$ varies, $x + u \in \mathbb{R}^n$ varies. In fact, from the inner product structure of $\mathbb{R}^n$, it is clear at once that any point $z \in \mathbb{R}^n$ can be uniquely written as $z = x + u$ for $x \in \pi$ and $u \in \pi^{\perp}$. In this case, it is also clear that $x = \text{Pr}_{\pi} \left( z \right)$ and $u = \text{Pr}_{\pi^{\perp}} \left( z \right)$. Hence, we get
			\begin{align*}
				&\int\limits_{G \left( n, k \right)} R_{\alpha}f \left( \pi, u \right) \varphi \left( \pi, u \right) \mathrm{d}u \mathrm{d}\pi \\
				&= \int\limits_{G_{n, k}} \int\limits_{\mathbb{R}^n} f \left( z \right) \| \text{Pr}_{\pi} \left( z \right) \|^{\alpha - k} \varphi \left( \pi, \text{Pr}_{\pi^{\perp}} \left( z \right) \right) \mathrm{d}z \mathrm{d}\pi.
			\end{align*}
			By using Fubini's theorem, we get
			\begin{equation}
				\label{DualityEquation2}
				\int\limits_{G \left( n, k \right)} R_{\alpha}f \left( \pi, u \right) \varphi \left( \pi, u \right) = \int\limits_{\mathbb{R}^n} f \left( z \right) \left( \int\limits_{G_{n, k}} \varphi \left( \pi, \text{Pr}_{\pi^{\perp}} \left( z \right) \right) \| \text{Pr}_{\pi} \left( z \right) \|^{\alpha - k} \mathrm{d}\pi \right) \mathrm{d}z.
			\end{equation}
			This gives Equation \eqref{DualTransformEquation}.
<!-- end proof-1 -->
<!-- remark-1: body_md -->
\normalfont
			We remark here that the quantity $\| \text{Pr}_{\pi} \left( x \right) \|$ gives the distance of $x$ from $\pi^{\perp}$. Therefore, in some sense, the dual $R_{\alpha}^*$ is a power-weighted analogue of the dual $k$-plane transform. We also notice that the set of singularities of $\| \text{Pr}_{\pi} \left( x \right) \|^{\alpha - k}$ (when $\alpha < k$) is a measure zero set.
<!-- end remark-1 -->
<!-- proposition-2: body_md -->
\label{RadialFormula}
			Let $f: \mathbb{R}^n \rightarrow \mathbb{C}$ be a ``nice" radial function, i.e., there is a function $\tilde{f}: \left[ 0, \infty \right) \rightarrow \mathbb{C}$ such that $f \left( x \right) = \tilde{f} \left( \| x \| \right)$. Then,
			\begin{equation}
				\label{RadialFormulaEquation}
				R_{\alpha}f \left( \pi, u \right) = \left| \mathbb{S}^{k - 1} \right| \int\limits_{\| u \|}^{\infty} \tilde{f} \left( t \right) \left( t^2 - \| u \|^2 \right)^{\frac{\alpha}{2} - 1} t \ \mathrm{d}t.
			\end{equation}
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
By the definition of $R_{\alpha}$ (see Equation \eqref{OurTransformDefinition}), we have,

<!-- proposition-5: body_md -->
\label{DualRadialFormula}
			Let $\varphi: G \left( n, k \right) \rightarrow \mathbb{C}$ be a ``nice" radial function, i.e., there is a function $\tilde{\varphi}: \left[ 0, \infty \right) \rightarrow \mathbb{C}$ such that $\varphi \left( \xi \right) = \tilde{\varphi} \left( d \left( 0, \xi \right) \right)$. Then,
			\begin{equation}
				\label{DualRadialFornulaEquation}
				R_{\alpha}^*\varphi \left( x \right) = \frac{\left| \mathbb{S}^{k - 1} \right| \left| \mathbb{S}^{n - k - 1} \right|}{\left| \mathbb{S}^{n - 1} \right|} \frac{1}{\| x \|^{n - 2}} \int\limits_{0}^{\| x \|} \tilde{\varphi} \left( s \right) \left( \| x \|^2 - s^2 \right)^{\frac{\alpha}{2} - 1} s^{n - k - 1} \mathrm{d}s.
			\end{equation}
<!-- end proposition-5 -->
<!-- proof-5: body_md -->
Since we consider the normalized probability measure on $G_{n, k}$, we can convert the integrals on $G_{n, k}$ to those on $SO \left( n \right)$, without giving any mutliplicative constants. Particularly, we have from Equation \eqref{DualTransformEquation},
			\begin{align*}
				R_{\alpha}^*\varphi \left( x \right) &= \int\limits_{G_{n, k}} \varphi \left( \pi, \text{Pr}_{\pi}^{\perp} \left( x \right) \right) \| \text{Pr}_{\pi} \left( x \right) \|^{\alpha - k} \mathrm{d}\pi \\
				&= \int\limits_{SO \left( n \right)} \varphi \left( g \cdot \mathbb{R}^k, \text{Pr}_{g \cdot \mathbb{R}^{n - k}} \left( x \right) \right) \| \text{Pr}_{g \cdot \mathbb{R}^k} \left( x \right) \|^{\alpha - k} \mathrm{d}g.
			\end{align*}
			Here, $\mathbb{R}^k = \left\lbrace \left( x_1, \cdots, x_k, 0, \cdots, 0 \right) \in \mathbb{R}^n | x_1, \cdots, x_k \in \mathbb{R} \right\rbrace$ and $\mathbb{R}^{n - k} = \left( \mathbb{R}^k \right)^{\perp}$.
			
			If $\xi = \left( g \cdot \mathbb{R}^k, \text{Pr}_{g \cdot \mathbb{R}^{n - k}} \left( x \right) \right)$, then $d \left( 0, \xi \right) = \| \text{Pr}_{g \cdot \mathbb{R}^{n - k}} \left( x \right) \| = \| \text{Pr}_{\mathbb{R}^{n - k}} \left( gx \right) \|$. Therefore, whenever $\varphi$ is radial, we have,
			\begin{align*}
				R_{\alpha}^*\varphi \left( x \right) = \int\limits_{SO \left( n \right)} \tilde{\varphi} \left( \| \text{Pr}_{\mathbb{R}^{n - k}} \left( gx \right) \| \right) \| \text{Pr}_{\mathbb{R}^k} \left( gx \right) \|^{\alpha - k} \mathrm{d}g.
			\end{align*}
			We now notice that as $g \in SO \left( n \right)$, $g \left( \frac{x}{\| x \|} \right)$ covers all of $\mathbb{S}^{n - 1}$, whenever $x \neq 0$. Thus, we obtain,
			\begin{align*}
				R_{\alpha}^*\varphi \left( x \right) = \frac{1}{\left| \mathbb{S}^{n - 1} \right|} \int\limits_{\mathbb{S}^{n - 1}} \tilde{\varphi} \left( \| \text{Pr}_{\mathbb{R}^{n - k}} \left( \| x \| \omega \right) \| \right) \| \text{Pr}_{\mathbb{R}^k} \left( \| x \| \omega \right) \|^{\alpha - k} \mathrm{d}\omega.
			\end{align*}
			Using the bispherical decomposition on $\mathbb{S}^{n - 1}$ (see \cite{GrfakosCFA}), we get
			\begin{align*}
				&R_{\alpha}^*\varphi \left( x \right) \\
				&= \frac{1}{\left| \mathbb{S}^{n - 1} \right|} \int\limits_{\mathbb{S}^{k - 1}} \int\limits_{\mathbb{S}^{n - k - 1}} \int\limits_{0}^{\frac{\pi}{2}} \tilde{\varphi} \left( \| x \| \sin \theta \right) \left( \| x \| \cos \theta \right)^{\alpha - k} \sin^{n - k - 1} \theta \cos^{k - 1} \theta \ \mathrm{d}\theta \mathrm{d}\zeta \mathrm{d}\eta \\
				&= \frac{\left| \mathbb{S}^{k - 1} \right| \left| \mathbb{S}^{n - k - 1} \right|}{\left| \mathbb{S}^{n - 1} \right|} \| x \|^{\alpha - k} \int\limits_{0}^{\frac{\pi}{2}} \tilde{\varphi} \left( \| x \| \sin \theta \right) \sin^{n - k - 1} \theta \cos^{\alpha - 1} \theta \ \mathrm{d}\theta.
			\end{align*}
			Equation \eqref{DualRadialFornulaEquation} is now obtained by substituting $\| x \| \sin \theta = s$.

<!-- theorem-1: body_md -->
\label{ExistenceTheorem}
			Let $\alpha > 0$ and $1 \leq k \leq n - 1$ be fixed. For $\mu > \alpha - n$, the operator $R_{\alpha}$ is well-defined on the weighted spaces $L^p_{\mu} \left( \mathbb{R}^n \right)$, for all $1 \leq p < \frac{\mu + n}{\alpha}$. Further, for $\mu = \alpha - n$, the transform $R_{\alpha}$ is well-defined on $L^1_{\mu} \left( \mathbb{R}^n \right)$. Moreover, the conditions obtained on $\mu$ and $p$ are optimal in the sense that if these conditions are not satisfied, there are radial $L^p_{\mu}$-functions $f$, for which $R_{\alpha}f \equiv \infty$.
<!-- end theorem-1 -->
<!-- proof-8: body_md -->
To prove the existence we shall employ duality of $R_{\alpha}$ and $R_{\alpha}^*$. We first consider the case $1 < p < \frac{\mu + n}{\alpha}$ and $\mu > \alpha - n$. For the same, let us define $\varphi: G \left( n, k \right) \rightarrow \mathbb{C}$ as
			\begin{equation}
				\label{PhiExistence}
				\varphi \left( \xi \right) = \frac{\left( d \left( 0, \xi \right) \right)^{\mu + \gamma_1}}{\left( 1 + \left( d \left( 0, \xi \right) \right)^2 \right)^{\frac{\gamma_2}{2}}},
			\end{equation}
			where, $\gamma_1 > k - n - \mu$, $\gamma_1 > - \frac{\mu + n}{p'} - \alpha + k$ and $k - n - \mu < \gamma_1 - \gamma_2 < k - \alpha - \frac{\mu + n}{p'}$. Clearly, such a choice is possible since $p < \frac{\mu + n}{\alpha}$. Then, using Equations \eqref{DualityEquation} and \eqref{DualRadialExample2}, we get
			\begin{align*}
				&\left| \int\limits_{G \left( n, k \right)} R_{\alpha}f \left( \xi \right) \varphi \left( \xi \right) \mathrm{d}\xi \right| \\
				&\leq C \int\limits_{X} f \left( x \right) \frac{\| x \|^{\gamma_1 + \mu + \alpha - k}}{\left( 1 + \| x \|^2 \right)^{\frac{\gamma_2}{2}}} \left| {}_2F_1 \left( \frac{\gamma_2}{2}, \frac{\alpha}{2}; \frac{\gamma_1 + \mu + \alpha + n - k}{2}; \frac{\| x \|^2}{1 + \| x \|^2} \right) \right| \mathrm{d}x.
			\end{align*}
			Using the H\"{o}lder's inequality and polar decomposition on $\mathbb{R}^n$, we obtain
			\begin{align*}
				&\left| \int\limits_{G \left( n, k \right)} R_{\alpha}f \left( \xi \right) \varphi \left( \xi \right) \mathrm{d}\xi \right| \leq C \| f \|_{L^p_{\mu} \left( \mathbb{R}^n \right)} \times \\
				&\left[ \int\limits_{0}^{\infty} \frac{r^{p' \left( \gamma_1 + \alpha - k \right) + \mu + n - 1}}{\left( 1 + r^2 \right)^{\frac{p' \gamma_2}{2}}} \left| {}_2F_1 \left( \frac{\gamma_2}{2}, \frac{\alpha}{2}; \frac{\gamma_1 + \mu + \alpha + n - k}{2}; \frac{r^2}{1 + r^2} \right) \right|^{p'} \mathrm{d}r \right]^{\frac{1}{p}}.
			\end{align*}
			From Equation 15.4.20 of \cite{Olver}, we know that the hypergeometric function is bounded due to our choice of $\gamma_1$ and $\gamma_2$. Further, due to the choice of $\gamma_1$ and $\gamma_2$, the integral on the right hand side of the above inequality is convergent. Consequently, $R_{\alpha}f$ is finite almost everywhere.
			
			For $p = 1$ and $\mu > \alpha - n$, we consider the following. Again using Equations \eqref{DualityEquation} and \eqref{DualRadialExample1}, we get
			\begin{align*}
				\left| \int\limits_{G \left( n, k \right)} R_{\alpha}f \left( \xi \right) \left( d \left( 0, \xi \right) \right)^{\mu + k - \alpha} \mathrm{d}\xi \right| &\leq C \int\limits_{\mathbb{R}^n} f \left( x \right) \| x \|^{\mu} \mathrm{d}x = C \| f \|_{L^1_{\mu} \left( \mathbb{R}^n \right)}.
			\end{align*}
			Clearly, for $L^1_{\mu} \left( \mathbb{R}^n \right)$ functions, we must have $R_{\alpha}f$ finite almost everywhere.
			
			The proof will be complete once we show the result for $\mu = \alpha - n$ and $p = 1$. For the same, let us consider the function $\varphi: G \left( n, k \right) \rightarrow \mathbb{C}$, given in Equation \eqref{PhiExistence}, with $\gamma_1 > k - \alpha$ and $\gamma_2 > \gamma_1 + \alpha - k$. Then, using Equations \eqref{DualityEquation} and \eqref{DualRadialExample2}, we get
			\begin{align*}
				\left| \int\limits_{G \left( n, k \right)} R_{\alpha}f \left( \xi \right) \varphi \left( \xi \right) \mathrm{d}\xi \right| &\leq C \int\limits_{\mathbb{R}^n} \left| f \left( x \right) \right| \frac{\| x \|^{\mu + \gamma_1 + \alpha - k}}{\left( 1 + \| x \|^2 \right)^{\frac{\gamma_2}{2}}} \times \\
				&\left| {}_2F_1 \left( \frac{\gamma_2}{2}, \frac{\alpha}{2}; \frac{\gamma_1 + \mu + \alpha + n - k}{2}; \frac{\| x \|^2}{1 + \| x \|^2} \right) \right| \mathrm{d}x \\
				&= C \left[ \int\limits_{\| x \| \leq 1} \left| f \left( x \right) \right| \| x \|^{\mu} \frac{\| x \|^{\gamma_1 + \alpha - k}}{\left( 1 + \| x \|^2 \right)^{\frac{\gamma_2}{2}}} \times \right. \\
				&\left. \left| {}_2F_1 \left( \frac{\gamma_2}{2}; \frac{\alpha}{2}; \frac{\gamma_1 + \mu + \alpha + n - k}{2}; \frac{\| x \|^2}{1 + \| x \|^2} \right) \right| \mathrm{d}x \right. \\
				&\left. + \int\limits_{\| x \| > 1} \left| f \left( x \right) \right| \| x \|^{\mu} \frac{\| x \|^{\gamma_1 + \alpha - k}}{\left( 1 + \| x \|^2 \right)^{\frac{\gamma_2}{2}}} \times \right. \\
				&\left. \left| {}_2F_1 \left( \frac{\gamma_2}{2}, \frac{\alpha}{2}; \frac{\gamma_1 + \mu + \alpha + n - k}{2}; \frac{\| x \|^2}{1 + \| x \|^2} \right) \right| \mathrm{d}x \right].
			\end{align*}
			It is clear that for $\| x \| \leq 1$, the term $\frac{\| x \|^{\gamma_1 + \alpha - k}}{\left( 1 + \| x \|^2 \right)^{\frac{\gamma_2}{2}}} \left| {}_2F_1 \left( \frac{\gamma_2}{2}; \frac{\alpha}{2}; \frac{\gamma_1 + \mu + \alpha + n - k}{2}; \frac{\| x \|^2}{1 + \| x \|^2} \right) \right|$ is bounded. On the other hand, by the choice of $\gamma_1$ and $\gamma_2$, we have from Equation 15.4.23 of \cite{Olver} for $\| x \| > 1$,
			$$\left| {}_2F_1 \left( \frac{\gamma_2}{2}, \frac{\alpha}{2}; \frac{\gamma_1 + \mu + \alpha + n - k}{2}; \frac{\| x \|^2}{1 + \| x \|^2} \right) \right| \leq C \frac{1}{\left( 1 + \| x \|^2 \right)^{\frac{\gamma_1 - \gamma_2 + \alpha - k}{2}}}.$$
			Hence, we obtain
			\begin{align*}
				&\left| \int\limits_{G \left( n, k \right)} R_{\alpha}f \left( \xi \right) \varphi \left( \xi \right) \mathrm{d}\xi \right| \\
				&\leq C \left[ \int\limits_{\| x \| \leq 1} \left| f \left( x \right) \right| \| x \|^{\mu} \mathrm{d}x + \int\limits_{\| x \| > 1} \left| f \left( x \right) \right| \| x \|^{\mu} \frac{\| x \|^{\gamma_1 + \alpha - k}}{\left( 1 + \| x \|^2 \right)^{\frac{\gamma_1 + \alpha - k}{2}}} \mathrm{d}x \right] \\
				&\leq C \| f \|_{L^1_{\mu} \left( \mathbb{R}^n \right)}.
			\end{align*}
			Therefore, $R_{\alpha}$ is well-defined on $L^1_{\alpha - n} \left( \mathbb{R}^n \right)$.
			
			The optimality of the conditions on $p$ and $\mu$ is shown in Examples \ref{CounterExample1} and \ref{CounterExample2}.

			\begin{align*}

<!-- remark-2: body_md -->
\label{RemarkLPLR}
			\normalfont
			Certain remarks are in order.
			\begin{enumerate}
				\item First, we notice that from Equation \eqref{ConvolutionFG} it is clear that $R_{\alpha}$ has a convolution structure. Therefore, by the famous Theorem due to H\"{o}rmander, we must have $R_{\alpha}$ to be $L^p$-improving (see \cite{Hormander}). That is, $r \geq p$ is necessary.
				\item The conditions on $\mu$ and $p$ are necessary for the existence of the transforms $R_{\alpha}$, as discussed in Theorem \ref{ExistenceTheorem}.
				\item The necessity of the condition $\frac{1}{p} - \frac{1}{r} \leq \frac{\alpha}{2}$ is not immediate. To see the same, we consider the following, motivated from \cite{KumarRayWtEst}. First, using $t = e^y$ and $s = e^x$ in Equation \eqref{ConvolutionDescription}, we obtain
				$$\| R_{\alpha}f \|_{L^r_{\nu} \left( G \left( n, k \right) \right)} = C \| \varphi * \psi \|_{L^r \left( \mathbb{R} \right)},$$
				where, $\varphi \left( x \right) = \tilde{f} \left( e^x \right) e^{\left( \frac{\mu + n}{p} \right)y}$, and 
				$$\psi \left( x \right) = \left( 1 - e^{2x} \right)^{\frac{\alpha}{2} - 1} e^{\left( \frac{\nu + n - k}{r} \right)x} \chi_{\left( - \infty, 0 \right)} \left( x \right).$$
				We notice that the convolution operator $\varphi \mapsto \varphi * \psi$ is a non-negative operator and Inequality \eqref{RequiredInequality1} implies the boundedness of $\varphi \mapsto \varphi * \psi$. For $\varphi \geq 0$ and $\lambda > 0$, let $\varphi_{\lambda} \left( x \right) = \varphi \left( \lambda x \right)$. Then, it is clear at once that for $\lambda > 1$, we have
				\begin{align*}
					\| \varphi_{\lambda} * \psi \|_{L^r \left( \mathbb{R} \right)} &\geq \left[ \int\limits_{\mathbb{R}} \left( \int\limits_{\mathbb{R}} \varphi \left( e^{\lambda x} \right) \psi \left( e^{x - y} \right) \chi_{\left( -1, 0 \right)} \left( y \right) \right)^r \mathrm{d}x \right]^{\frac{1}{r}} \\
					&\geq C \lambda^{- \frac{\alpha}{2} - \frac{1}{r}} \| \varphi * \psi \|_{L^r \left( \mathbb{R} \right)}.
				\end{align*}
				On the other hand, we also have,
				$$\| \varphi_{\lambda} \|_{L^p \left( \mathbb{R} \right)} = \lambda^{- \frac{1}{p}} \| \varphi \|_{L^p \left( \mathbb{R} \right)}.$$
				Since Inequality \eqref{RequiredInequality1} implies the $L^p$-$L^r$ boundedness of $\varphi \mapsto \varphi * \psi$, we must have for $\lambda > 1$, and a fixed $\varphi \geq 0$ with $\varphi \in L^p \left( \mathbb{R} \right)$,
				$$\lambda^{- \frac{\alpha}{2} - \frac{1}{r}} \| \varphi * \psi \|_{L^r \left( \mathbb{R} \right)} \leq C \lambda^{- \frac{1}{p}} \| \varphi \|_{L^p \left( \mathbb{R} \right)}.$$
				As $\lambda \rightarrow \infty$, the above inequality forces $\frac{1}{p} - \frac{1}{r} \leq \frac{\alpha}{2}$.
			\end{enumerate}

\label{FIRemark}
			\normalfont
			Let us assume that Inequality \eqref{RequiredInequality2} holds for all radial functions in $L^p_{\mu} \left( \mathbb{R}^n \right)$. This implies the following inequality of fractional integrals
			$$\| \rho_2 \left( \cdot \right) I^{\frac{\alpha}{2}}_-\varphi \|_{L^{\infty} \left( 0, \infty \right)} \leq C \| \varphi \|_{L^{\frac{2}{\alpha}} \left( \left( 0, \infty \right), \rho_1 \left( t \right) \mathrm{d}t \right)},$$
			where, $\rho_2 \left( t \right) = t^{\frac{\nu}{2}}$, and $\rho_1 \left( t \right) = t^{\frac{\mu + n}{2} - 1}$. We claim that such an inequality is impossible, and therefore Inequality \eqref{RequiredInequality2} is not possible.
			
			In fact, we show that $I^{\frac{\alpha}{2}}_-: L^{\frac{2}{\alpha}} \left( \left( 0, \infty \right), \rho_1 \left( t \right) \mathrm{d}t \right) \rightarrow L^{\infty} \left( \left( 0, \infty \right), \rho_2 \left( t \right) \mathrm{d}t \right)$ cannot be bounded for any choices of positive power functions $\rho_1$ and $\rho_2$. Here, we use the notation
			\begin{align*}
				L^{\infty} \left( \left( 0, \infty \right), \rho_2 \left( t \right) \mathrm{d}t \right) := \lbrace \psi: \left( 0, \infty \right) \rightarrow \mathbb{C} | &\psi \text{ is measurable and } \\
				&\| \rho_2 \left( \cdot \right) \psi \left( \cdot \right) \|_{L^{\infty} \left( 0, \infty \right)} < +\infty \rbrace.
			\end{align*}
			Throughout the argument, we shall use $\alpha < 2$. If, on the contrary, we have for all $\varphi \in L^{\frac{2}{\alpha}} \left( \left( 0, \infty \right), \rho_1 \left( t \right) \mathrm{d}t \right)$,
			$$\| \rho_2 \left( \cdot \right) I^{\frac{\alpha}{2}}_-\varphi \|_{L^{\infty} \left( 0, \infty \right)} \leq C \| \varphi \|_{L^{\frac{2}{\alpha}} \left( 0, \infty \right)},$$
			then, for any $\psi \in L^1 \left( \left( 0, \infty \right), \frac{\rho_1 \left( t \right)}{\rho_2 \left( t \right)} \mathrm{d}t \right)$, we have,
			\begin{align*}
				\int\limits_{0}^{\infty} I^{\frac{\alpha}{2}}_-\varphi \left( x \right) \psi \left( x \right) \rho_1 \left( x \right) \mathrm{d}x &= \int\limits_{0}^{\infty} \varphi \left( t \right) \int\limits_{0}^{t} \frac{\psi \left( x \right) \rho_1 \left( x \right)}{\left( t - x \right)^{1 - \frac{\alpha}{2}}} \mathrm{d}x \mathrm{d}t \\
				&= \int\limits_{0}^{\infty} \varphi \left( t \right) \frac{1}{\rho_1 \left( t \right)} I^{\frac{\alpha}{2}}_+ \left( \rho_1 \cdot \psi \right) \left( t \right) \rho_1 \left( t \right) \mathrm{d}t.
			\end{align*}
			Therefore, by duality, we have, $\frac{1}{\rho_1} I^{\frac{\alpha}{2}}_+ \left( \rho_1 \cdot \psi \right) \in L^q \left( \left( 0, \infty \right), \rho_1 \left( t \right) \mathrm{d}t \right)$, where $q = \left( \frac{2}{\alpha} \right)'$, whenever $\psi \in L^1 \left( \left( 0, \infty \right), \frac{\rho_1 \left( t \right)}{\rho_2 \left( t \right)} \mathrm{d}t \right)$. However, one can show by producing explicit examples that this is not true. We refer the reader to \cite{SelfLPImprovingKPlane} (Example C.3), where the authors have given an example for the case $\alpha = 1$. For an arbitrary $0 < \alpha < 2$, we consider the function $\psi$ of Example C.3 of \cite{SelfLPImprovingKPlane} with $1 < \gamma \leq 2 - \frac{\alpha}{2}$. Then, it is easy to see that $\psi \in L^1 \left( \left( 0, \infty \right), \rho_1 \right)$ but $I^{\frac{\alpha}{2}}_+ \psi \notin L^{\frac{2}{2 - \alpha}} \left( \left( 0, \infty \right), \rho_2 \right)$, for any choices of $\rho_1$ and $\rho_2$.

<!-- theorem-6: body_md -->
\label{EndPointTheorem}
			Let $\alpha > 0$, $\mu > \alpha - n$ and $p = \frac{\mu + n}{\alpha}$. 
			\begin{enumerate}
				\item[(A)\def#2{(A)}\label{AlphaMoreThan2EndPoint}] For $\alpha \geq 2$ and all radial functions $f \in L^{p, 1}_{\mu} \left( \mathbb{R}^n \right)$, we have,
			\begin{equation}
				\label{EndPointInequality}
				\| \left( d \left( 0, \cdot \right) \right)^{\nu} R_{\alpha}f \|_{L^{\infty} \left( G \left( n, k \right) \right)} \leq C \| f \|_{L^{p, 1}_{\mu} \left( \mathbb{R}^n \right)},
			\end{equation}
			if and only if $\nu = 0$.
			\item[(B)\def#2{(B)}\label{AlphaLessThan2EndPoint}] For $0 < \alpha < 2$, Inequality \eqref{EndPointInequality} holds for all radial functions $f \in L^{p, 1}_{\mu} \left( \mathbb{R}^n \right)$ if and only if $\nu = 0$ and $p \geq \frac{2}{\alpha}$.
			\end{enumerate}

<!-- proof-13: body_md -->
First, we notice that since for characteristic functions of measurable sets, the $L^p$-norm and the Lorentz $L^{p, 1}$-norm coincide, the proof of necessity $\nu = 0$ is verbatim to that of Theorem \ref{NecessaryLPLInfinity}. We only prove the sufficiency of the conditions. To do so, it is enough to prove Inequality \eqref{EndPointInequality} for characteristic functions of radial measurable sets (see \cite{SteinWeissFA}). In this proof, therefore, we shall consider $f = \chi_A$, where $A \subseteq \mathbb{R}^n$ is measurable and radial.
			\begin{enumerate}
				\item[(A)\def#2{(A)}\label{AlphaMoreThan2EndPointProof}] We start with the case $\alpha \geq 2$, which is easy to handle. In this case, we have for $s > \| u \|$, $\left( 1 - \frac{\| u \|^2}{s^2} \right)^{\frac{\alpha}{2} - 1} \leq 1$. Therefore, we get,
				\begin{align*}
					\left| R_{\alpha}f \left( \pi, u \right) \right| & \leq C \int\limits_{\| u \|}^{\infty} \tilde{f} \left( t \right) t^{\alpha - 1} \mathrm{d}t.
				\end{align*}
				Using Lemma \ref{EndPointLemmaKumarRay}, we get
				\begin{align*}
					\left| R_{\alpha}f \left( \pi, u \right) \right| &\leq C \left( \int\limits_{0}^{\infty} \tilde{f} \left( t \right) t^{p \alpha - 1} \right)^{\frac{1}{p}} \\
					&= C \left( \int\limits_{0}^{\infty} \tilde{f} \left( t \right) t^{\mu + n - 1} \mathrm{d}t \right)^{\frac{1}{p}} = C \| f \|_{L^{p, 1}_{\mu} \left( \mathbb{R}^n \right)}.
				\end{align*}
				\item[(B)\def#2{(B)}\label{AlphaLessThan2EndPointProof}] When $0 < \alpha < 2$, the proof is a bit tricky since we can no longer dominate $\left( 1 - \frac{\| u \|^2}{s^2} \right)^{\frac{\alpha}{2} - 1}$ by a constant as in part \ref{AlphaMoreThan2EndPointProof}. To get around this, we substitute $s^2 - \| u \|^2 \left( 1 + x \right)$ in Equation \eqref{RadialFormulaEquation} and simplify to obtain
				\begin{align*}
					\left| R_{\alpha}f \left( \pi, u \right) \right| &= C \| u \|^{\alpha} \int\limits_{0}^{\infty} \tilde{f} \left( \| u \| \sqrt{1 + x} \right) x^{\frac{\alpha}{2} - 1} \mathrm{d}x.
				\end{align*}
				We notice that since $\tilde{f}$ is a characteristic function of a measurable set in $\left( 0, \infty \right)$, so is the function $\tilde{f} \left( \| u \| \sqrt{1 + \cdot} \right)$. Also, since $\alpha > 0$, Lemma \ref{EndPointLemmaKumarRay} is applicable. From Inequality \eqref{PolynomialInequalityLemma}, we get
				\begin{align*}
					\left| R_{\alpha}f \left( \pi, u \right) \right| &= C \| u \|^{\alpha} \left( \int\limits_{0}^{\infty} \tilde{f} \left( \| u \| \sqrt{1 + x} \right) x^{\frac{p \alpha}{2} - 1} \mathrm{d}x \right)^{\frac{1}{p}}.
				\end{align*}
				Resubstituting $t = \| u \| \sqrt{1 + x}$ and simplifying by using the fact that $p = \frac{\mu + n}{\alpha}$, we get
				\begin{align*}
					\left| R_{\alpha}f \left( \pi, u \right) \right| &\leq C \left( \int\limits_{\| u \|}^{\infty} \tilde{f} \left( t \right) \left( t^2 - \| u \|^2 \right)^{\frac{\mu + n}{2} - 1} t \ \mathrm{d}t \right)^{\frac{1}{p}}.
				\end{align*}
				Since $\mu + n = p \alpha \geq 2$, we have $\left( t^2 - \| u \|^2 \right)^{\frac{\mu + n}{2} - 1} \leq t^{\mu + n - 2}$. That is,
				\begin{align*}
					\left| R_{\alpha}f \left( \pi, u \right) \right| &\leq C \left( \int\limits_{0}^{\infty} \tilde{f} \left( t \right) t^{\mu + n - 1} \mathrm{d}t \right)^{\frac{1}{p}} = C \| f \|_{L^{p, 1}_{\mu} \left( \mathbb{R}^n \right)}.
				\end{align*}
			\end{enumerate}

<!-- theorem-7: body_md -->
\label{LPLPBoundednessTheorem}
			Let $\alpha > 0$, $\mu > \alpha - n$ and $1 \leq p < \frac{\mu + n}{\alpha}$. Then, the operator $R_{\alpha}: L^p_{\mu} \left( \mathbb{R}^n \right) \rightarrow L^p_{\nu} \left( G \left( n, k \right) \right)$ is bounded if and only if $\nu = \mu + k - p \alpha$. Moreover, in this case, we have,
			\begin{equation}
				\label{OperatorNorm}
				\| R_{\alpha} \|_{L^p_{\mu} \left( \mathbb{R}^n \right) \rightarrow L^p_{\nu} \left( G \left( n, k \right) \right)} = \left( \frac{\left| \mathbb{S}^{n - k - 1} \right|}{\left| \mathbb{S}^{n - 1} \right|} \right)^{\frac{1}{p}} \frac{\left| \mathbb{S}^{k - 1} \right|}{2} \frac{\Gamma \left( \frac{\alpha}{2} \right) \Gamma \left( \frac{\mu + n}{2p} - \frac{\alpha}{2} \right)}{\Gamma \left( \frac{\mu + n}{2p} \right)}.
			\end{equation}

<!-- proof-15: body_md -->
We first remark that the necessity of $\nu = \mu + k - p \alpha$ follows from Equation \eqref{NecessaryConditionsLPLR} by taking $r = p$. Also, since $p < \frac{\mu + n}{\alpha}$, this condition automatically forces $\nu > k - n$. We shall first prove the boundedness of $R_{\alpha}$ and get an estimate on its operator norm.
			
			Using polar decomposition on $\pi^{\perp}$ (for a given $\pi \in G_{n, k}$), and recalling that we have considered the normalized measure on $G_{n, k}$, we obtain
			\begin{align*}
				&\| R_{\alpha}f \|_{L^p_{\nu} \left( G \left( n, k \right) \right)} \\
				&= \left| \mathbb{S}^{n - k - 1} \right|^{\frac{1}{p}} \left[ \int\limits_{SO \left( n \right)} \int\limits_{0}^{\infty} \left| r^{\alpha} \int\limits_{\mathbb{S}^k_+} f_g \left( \frac{r \theta}{\theta_{k + 1}} \right) \frac{\| \theta' \|^{\alpha - k}}{\theta_{k + 1}^{\alpha + 1}} \mathrm{d}\theta \right|^p r^{\nu + n - k - 1} \mathrm{d}r \mathrm{d}g \right]^{\frac{1}{p}}.
			\end{align*}
			Now, by applying Minkowski's integral inequality, we get
			\begin{align*}
				&\| R_{\alpha}f \|_{L^p_{\nu} \left( G \left( n, k \right) \right)} \\
				&\leq \left| \mathbb{S}^{n - k - 1} \right|^{\frac{1}{p}} \int\limits_{\mathbb{S}^k_+} \frac{\| \theta' \|^{\alpha - k}}{\theta_{k + 1}^{\alpha + 1}} \left[ \int\limits_{SO \left( n \right)} \int\limits_{0}^{\infty} \left| f_g \left( \frac{r \theta}{\theta_{k + 1}} \right) \right|^p r^{\nu + n - k - 1 + p \alpha} \mathrm{d}r \mathrm{d}g \right]^{\frac{1}{p}} \mathrm{d}\theta.
			\end{align*}
			Next, we substitute $r = \theta_{k + 1} t$ in the innermost integral, use the fact that $\nu = \mu + k - p \alpha$, and simplify to obtain
			\begin{align}
				&\| R_{\alpha}f \|_{L^p_{\nu} \left( G \left( n, k \right) \right)} \nonumber \\
				&\leq \left| \mathbb{S}^{n - k - 1} \right|^{\frac{1}{p}} \int\limits_{\mathbb{S}^k_+} \| \theta' \|^{\alpha - k} \theta_{k + 1}^{\frac{\nu + n - k}{p} - 1} \left[ \int\limits_{SO \left( n \right)} \int\limits_{0}^{\infty} \left| f_g \left( t \theta \right) \right|^p t^{\mu + n - 1} \mathrm{d}t \mathrm{d}g \right]^{\frac{1}{p}} \mathrm{d}\theta \nonumber \\
				\label{PenultipmateLPLPBounded}
				&= \left( \frac{\left| \mathbb{S}^{n - k - 1} \right|}{\left| \mathbb{S}^{n - 1} \right|} \right)^{\frac{1}{p}} \| f \|_{L^p_{\mu} \left( \mathbb{R}^n \right)} \int\limits_{\mathbb{S}^k_+} \| \theta' \|^{\alpha - k} \theta_{k + 1}^{\frac{\mu + n}{p} - \alpha - 1} \mathrm{d}\theta.
			\end{align}
			Since $p < \frac{\mu + n}{\alpha}$, it is clear that the above integral on the hemisphere ($\mathbb{S}^k_+$) is finite. Moreover, from the polar decomposition of the sphere, we easily see that
			\begin{equation}
				\label{BetaIntegralHemisphere}
				\int\limits_{\mathbb{S}^k_+} \| \theta' \|^{\alpha - k} \theta_{k + 1}^{\frac{\mu + n}{p} - \alpha - 1} \mathrm{d}\theta = \frac{\left| \mathbb{S}^{k - 1} \right|}{2} \frac{\Gamma \left( \frac{\alpha}{2} \right) \Gamma \left( \frac{\mu + n}{2p} - \frac{\alpha}{2} \right)}{\Gamma \left( \frac{\mu + n}{2p} \right)}.
			\end{equation}
			Equations \eqref{PenultipmateLPLPBounded} and \eqref{BetaIntegralHemisphere}  prove the weighted $L^p$-$L^p$ boundedness of $R_{\alpha}$, and give
			$$\| R_{\alpha} \|_{L^p_{\mu} \left( \mathbb{R}^n \right) \rightarrow L^p_{\nu} \left( \mathbb{R}^n \right)} \leq \left( \frac{\left| \mathbb{S}^{n - k - 1} \right|}{\left| \mathbb{S}^{n - 1} \right|} \right)^{\frac{1}{p}} \frac{\left| \mathbb{S}^{k - 1} \right|}{2} \frac{\Gamma \left( \frac{\alpha}{2} \right) \Gamma \left( \frac{\mu + n}{2p} - \frac{\alpha}{2} \right)}{\Gamma \left( \frac{\mu + n}{2p} \right)}.$$
			
			We now prove Equation \eqref{OperatorNorm}. To do so, we first observe (by using H\"{o}lder's inequality) that
			\begin{equation}
				\label{DualEstimate}
				\left| \int\limits_{G \left( n, k \right)} R_{\alpha}f \left( \xi \right) \varphi \left( \xi \right) \mathrm{d}\xi \right| \leq \| R_{\alpha} \|_{L^p_{\mu} \left( \mathbb{R}^n \right) \rightarrow L^p_{\nu} \left( G \left( n, k \right) \right)} \| f \|_{L^p_{\mu} \left( \mathbb{R}^n \right)} \| \varphi \|_{L^{p'}_{- \frac{p' \nu}{p}} \left( G \left( n, k \right) \right)},
			\end{equation}
			for all nice functions $f: \mathbb{R}^n \rightarrow \mathbb{C}$ and $\varphi: G \left( n, k \right) \rightarrow \mathbb{C}$. When $p = 1$, the norm $\| \varphi \|_{L^{p'}_{- \frac{p' \nu}{p}} \left( G \left( n, k \right) \right)}$ is to be interpreted as $\| \left( d \left( 0, \cdot \right) \right)^{-\nu} \varphi \|_{L^{\infty} \left( G \left( n, k \right) \right)}$.
			
			Now, we consider the function $f: \mathbb{R}^n \rightarrow \mathbb{C}$, defined by
			$$f \left( x \right) = \| x \|^{- \frac{\mu + n}{p} - \delta} \chi_{\mathbb{R}^n \setminus B \left( 0, 1 \right)} \left( \| x \| \right).$$
			Here, $\delta > 0$ is fixed arbitrarily. Then, we have, by using polar decomposition on $\mathbb{R}^n$,
			$$\| f \|_{L^p_{\mu} \left( \mathbb{R}^n \right)}^p = \left| \mathbb{S}^{n - 1} \right| \int\limits_{1}^{\infty} t^{-p \delta - 1} \mathrm{d}t = \frac{\left| \mathbb{S}^{n - 1} \right|}{p \delta}.$$
			We observe that $f$ is a radial function. We denote $\tilde{f}: \left( 0, \infty \right) \rightarrow \mathbb{C}$, the function so that $f \left( x \right) = \tilde{f} \left( \| x \| \right)$. That is, $\tilde{f} \left( t \right) = t^{- \frac{\mu + n}{p} - \delta} \chi_{\left( 1, \infty \right)} \left( t \right)$.
			
			Next, let us consider $\varphi: G \left( n, k \right) \rightarrow \mathbb{C}$, defined by
			$$\varphi \left( \xi \right) = \left( d \left( 0, \xi \right) \right)^{\mu + k - \alpha} \left( \tilde{f} \left( d \left( 0, \xi \right) \right) \right)^{p - 1}.$$
			Then, we see from the polar decomposition on $\pi^{\perp}$, for a given $\pi \in G_{n, k}$, that
			\begin{align*}
				\| \varphi \|_{L^{p'}_{- \frac{p' \nu}{p}} \left( G \left( n, k \right) \right)} &= \left| \mathbb{S}^{n - k - 1} \right|^{\frac{1}{p'}} \left( \int\limits_{0}^{\infty} \left| \tilde{f} \left( t \right) \right|^{p' \left( p - 1 \right)} t^{\mu + n - 1} \mathrm{d}t \right)^{\frac{1}{p'}} \\
				&= \frac{\left| \mathbb{S}^{n - k - 1} \right|^{\frac{1}{p'}}}{\left| \mathbb{S}^{n - 1} \right|^{\frac{1}{p'}}} \| f \|_{L^p_{\mu} \left( \mathbb{R}^n \right)}^{p - 1}.
			\end{align*}
			
			Now, since both $f$ and $\varphi$ are radial, we have from Equation \eqref{SphericalRepresentationRAlphaEquation}
			\begin{align*}
				&\int\limits_{G \left( n, k \right)} R_{\alpha}f \left( \xi \right) \varphi \left( \xi \right) \mathrm{d}\xi \\
				&= \left| \mathbb{S}^{n - k - 1} \right| \int\limits_{0}^{\infty} r^{\alpha + n - k - 1} \tilde{\varphi} \left( r \right) \int\limits_{\mathbb{S}^k_+} \tilde{f} \left( \frac{r}{\theta_{k + 1}} \right) \frac{\| \theta' \|^{\alpha - k}}{\theta_{k + 1}^{\alpha + 1}} \mathrm{d}\theta \mathrm{d}r \\
				&= \left| \mathbb{S}^{n - k - 1} \right| \int\limits_{\mathbb{S}^k_+} \frac{\| \theta' \|^{\alpha - k}}{\theta_{k + 1}^{\alpha + 1}} \left( \int\limits_{0}^{\infty} \tilde{f} \left( \frac{r}{\theta_{k + 1}} \right) \tilde{\varphi} \left( r \right) r^{\alpha + n - k - 1} \mathrm{d}r \right) \mathrm{d}\theta.
			\end{align*}
			Using the definitions of $f$ and $\varphi$, we evaluate the inner integral to be
			$$\int\limits_{0}^{\infty} \tilde{f} \left( \frac{r}{\theta_{k + 1}} \right) \tilde{\varphi} \left( r \right) r^{\alpha + n - k - 1} \mathrm{d}r = \frac{\theta_{k + 1}^{\frac{\mu + n}{p} + \delta}}{p \delta}.$$
			Using these observations in Inequality \eqref{DualEstimate}, we get
			\begin{align*}
				&\left| \mathbb{S}^{n - k - 1} \right| \int\limits_{\mathbb{S}^k_+} \frac{\| \theta' \|^{\alpha - k}}{\theta_{k + 1}^{\alpha + 1}} \frac{\theta_{k + 1}^{\frac{\mu + n}{p} + \delta}}{p \delta} \\
				&\leq \left( \frac{\left| \mathbb{S}^{n - k - 1} \right|}{\left| \mathbb{S}^{n - 1} \right|} \right)^{\frac{1}{p'}} \frac{\left| \mathbb{S}^{n - 1} \right|}{p \delta} \| R_{\alpha} \|_{L^p_{\mu} \left( \mathbb{R}^n \right) \rightarrow L^p_{\nu} \left( G \left( n, k \right) \right)}.
			\end{align*}
			Upon simplifying, we get
			\begin{align*}
				\| R_{\alpha} \|_{L^p_{\mu} \left( \mathbb{R}^n \right) \rightarrow L^p_{\nu} \left( G \left( n, k \right) \right)} &\geq \left( \frac{\left| \mathbb{S}^{n - k - 1} \right|}{\left| \mathbb{S}^{n - 1} \right|} \right)^{\frac{1}{p}} \int\limits_{\mathbb{S}^k_+} \| \theta' \|^{\alpha - k} \theta_{k + 1}^{\frac{\mu + n}{p} + \delta - \alpha - 1} \mathrm{d}\theta \\
				&= \left( \frac{\left| \mathbb{S}^{n - k - 1} \right|}{\left| \mathbb{S}^{n - 1} \right|} \right)^{\frac{1}{p}} \frac{\left| \mathbb{S}^{k - 1} \right|}{2} \frac{\Gamma \left( \frac{\alpha}{2} \right) \Gamma \left( \frac{\mu + n}{2p} - \frac{\alpha}{2} + \frac{\delta}{2} \right)}{\Gamma \left( \frac{\mu + n}{2p} + \frac{\delta}{2} \right)},
			\end{align*}
			for every $\delta > 0$. By taking $\delta \rightarrow 0$ and using the continuity of Gamma function, we obtain
			\begin{align*}
				\| R_{\alpha} \|_{L^p_{\mu} \left( \mathbb{R}^n \right) \rightarrow L^p_{\nu} \left( G \left( n, k \right) \right)} &\geq \left( \frac{\left| \mathbb{S}^{n - k - 1} \right|}{\left| \mathbb{S}^{n - 1} \right|} \right)^{\frac{1}{p}} \frac{\left| \mathbb{S}^{k - 1} \right|}{2} \frac{\Gamma \left( \frac{\alpha}{2} \right) \Gamma \left( \frac{\mu + n}{2p} - \frac{\alpha}{2} \right)}{\Gamma \left( \frac{\mu + n}{2p} \right)}.
			\end{align*}
			This completes the proof!
