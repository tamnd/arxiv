<!-- remark-1: body_md -->
\begin{enumerate}
    		\item Gouv\^{e}a's dimension conjecture \cite[Section~4]{Gouvea2001}
    		allows arbitrary finite sets $S$, i.e., it does not require $S$ to
    		contain the primes of $K$ above $p$.
    		In this broader setting, Bleher and Chinburg
    		\cite[Theorem~1.1]{BleherChinburg2007} constructed counterexamples
    		over infinitely many real quadratic fields, with $p=2$ and
    		$S=\varnothing$. The author also constructed counterexamples over
    		every number field, with ramification sets disjoint from the
    		primes above $p$; see \cite[Theorem~1.7]{Luo2026}. These examples do not provide counterexamples to Conjecture~\ref{conj:unmodified}, since their sets $S$ do not contain all primes above $p$.
    		
    		\item If the absolute irreducibility of $\bar{\rho}$ is replaced by the weaker condition
    		$\operatorname{End}_{k[G_{K,S}]}(\bar\rho)=k$, then a universal deformation ring still exists; see \cite[Proposition~1.3.1(b)]{Boeckle2013}. However, the dimension conjecture can fail for such reducible residual representations, even when $S$ contains all primes above $p$ and all archimedean primes; see \cite[Section~4]{Boeckle1998}.
    		
    		\item On the positive side, Xinyao Zhang recently proved
    		Conjecture~\ref{conj:unmodified} for $K=\mathbb Q$, $n=2$, and $p\ge3$
    		when $\bar\rho$ is odd and its restriction to the local Galois group $G_{\mathbb Q_p}$ at $p$ is reducible;
    		see \cite[Theorem~A.2.2 and Corollary~A.3.3]{Zhang2026}.
    		These results give that $R_{\bar{\rho}}$ is flat over $W(k)$ with $\dim R_{\bar\rho}=4$, and hence $\dim(R_{\bar\rho}/pR_{\bar\rho})=3=\delta(\bar\rho)$.
    	\end{enumerate}

\label{rem:odd-prime}
	\begin{enumerate}
		\item When $n=1$, every $s_v$ is zero and
		$\lfloor(n-2s_v)^2/2\rfloor=0$. Thus $C(\bar\rho)=0$ in rank one.
		If $K$ is totally imaginary, then $S_{\mathbb R}=\varnothing$, and hence $C(\bar\rho)=0$ for every $n$. In either case, Theorem~\ref{thm:main} gives the usual lower bound.
		
		\item At a real prime $v$, the special fiber of the framed deformation ring has dimension $\lfloor n^2/2\rfloor$, while the residual conjugacy orbit has
		dimension $2s_v(n-s_v)$; see Lemma \ref{lem:real-special}. The sum of these differences over the real
		primes is $C(\bar\rho)$. The usual framed local tangent obstruction count gives the orbit
		dimension. The relative global presentation and the surjectivity of
		restriction in degree two to the real primes allow us to use the
		actual local special fiber dimensions instead, yielding the global
		special fiber lower bound $\delta(\bar\rho)+C(\bar\rho)$. See the proof of Theorem \ref{thm:main}.
		
		\item In odd residue characteristic, $2$ is invertible, and hence the $1$-eigenspaces and
		$-1$-eigenspaces of an involution are direct summands whose ranks
		remain fixed under deformation. Choosing bases in these summands
		puts every deformation into the same diagonal form. Hence all framed
		local deformations arise by conjugation, and the local special fiber
		has the same dimension as the residual conjugacy orbit. Their
		dimension difference, and therefore the analogous correction, is zero.
	\end{enumerate}

\label{thm:lift}
	We keep the notation as above. Suppose that $\rho:G_{K,S}\to\operatorname{GL}_n(\mathcal O_F)$ is a continuous representation with reduction $\bar\rho\otimes_k k_F$, where $F$ is a finite extension of the field $W(k)[1/2]$, with ring of integers $\mathcal O_F$
	and residue field $k_F$. For $v\in S_{\mathbb R}$, let $a_v$ and $b_v$ denote the multiplicities of the eigenvalues $1$ and $-1$ of $\rho(c_v)$, respectively. Put
	\[ 	D_{\mathrm{lift}}(\rho)=1+n^2r_2(K)
	+2\sum_{v\in S_{\mathbb R}}a_vb_v. \]
	Then we have the following.
	\begin{enumerate}
		\item One has 
		\[
		\dim(R_{\bar\rho}/2R_{\bar\rho})\ge D_{\mathrm{lift}}(\rho)=\delta(\bar\rho)+2\sum_{v\in S_{\mathbb R}}
		(a_v-s_v)(b_v-s_v).
		\]
		In particular, Conjecture~\ref{conj:unmodified} fails if
		$s_v<\min(a_v,b_v)$ for some $v\in S_{\mathbb R}$.
		\item One has
		\[ 	\dim R_{\bar\rho}-\operatorname{depth}R_{\bar\rho}\geq D(K,n)-D_{\mathrm{lift}}(\rho)-\dim_F H^2(G_{K,S},\operatorname{ad}(\rho\otimes_{\mathcal O_F}F)).  \]
	 where $\operatorname{depth}(R_{\bar{\rho}})$ denotes the depth of the ring $R_{\bar{\rho}}$. Moreover, if
	 \[ D(K,n)-D_{\mathrm{lift}}(\rho)>\dim_F H^2(G_{K,S},\operatorname{ad}(\rho\otimes_{\mathcal O_F}F)), \]
	 then $R_{\bar\rho}$ is not equidimensional, not Cohen--Macaulay, and not a complete
		intersection.
	\end{enumerate}

\label{rem:lift-correction}
	\begin{enumerate}
		\item The correction in Theorem~\ref{thm:lift}\textup{(1)} comes from
		comparing the dimension lower bounds obtained from the residual
		and characteristic-zero Euler characteristic formulas.
		For $V=\rho\otimes_{\mathcal O_F}F$, their common term
		$1+(r_1(K)+r_2(K))n^2$ cancels, giving
		\[
		\begin{aligned}
			D_{\mathrm{lift}}(\rho)-\delta(\bar\rho)
			&=\sum_{v\in S_{\mathbb R}}
			\bigl(\dim_k(\operatorname{ad}\bar\rho)^{c_v=1}
			-\dim_F(\operatorname{ad} V)^{c_v=1}\bigr)\\
			&=2\sum_{v\in S_{\mathbb R}}(a_v-s_v)(b_v-s_v).
		\end{aligned}
		\]
		Thus the correction measures the increase in the real adjoint
		invariant dimensions under reduction. Obstruction theory and the
		residual Euler characteristic formula give
		$\dim(R_{\bar\rho}/2R_{\bar\rho})\ge\delta(\bar\rho)$.
		The characteristic-zero formula at the point defined by $\rho$
		similarly gives $\dim R_{\bar\rho}[1/2]\ge D_{\mathrm{lift}}(\rho)$.
		The inequality $\dim(R_{\bar\rho}/2R_{\bar\rho})\ge\dim R_{\bar\rho}[1/2]$
		transfers the latter bound to the special fiber.
		
		\item For odd prime $p$, the projectors $(I_n\pm\rho(c_v))/2$ preserve the
		eigenspace ranks and hence the adjoint invariant dimensions under
		reduction, and hence the analogous correction vanishes.
		
		\item For every characteristic-zero lift $\rho$ as in Theorem~\ref{thm:lift},
		the equality $a_v+b_v=n$ implies that $D_{\mathrm{lift}}(\rho)\le D(K,n)$.
		Thus the special fiber bound in Theorem~\ref{thm:lift}\textup{(1)}
		is numerically no stronger than that in Theorem~\ref{thm:main},
		and its counterexample criterion implies $C(\bar\rho)>0$.
		The main additional contribution of Theorem~\ref{thm:lift} lies in
		part~\textup{(2)}. 
	\end{enumerate}

<!-- lemma-1: body_md -->
\label{lem:residual-centralizer}
We have $0\le s_v\le\lfloor n/2\rfloor$. The Jordan form of
$\bar\rho_v(c_v)$ has $s_v$ blocks of size two and $n-2s_v$ blocks
of size one, and
\[
\dim_k(\operatorname{ad}\bar\rho_v)^{G_v}
=(n-s_v)^2+s_v^2=n^2-2s_v(n-s_v).
\]
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
The matrix $N_v=\bar\rho_v(c_v)-I_n$ satisfies $N_v^2=0$.
As a $k[T]$-module, with $T$ acting by $N_v$, one has
\[
k^n\simeq\bigl(k[T]/(T^2)\bigr)^{s_v}
\oplus\bigl(k[T]/(T)\bigr)^{n-2s_v}.
\]
The space of homomorphisms from $k[T]/(T^i)$ to $k[T]/(T^j)$ has
dimension $\min(i,j)$. The endomorphism algebra is the centralizer
of $\bar\rho_v(c_v)$, and hence its dimension is
\[
2s_v^2+2s_v(n-2s_v)+(n-2s_v)^2=(n-s_v)^2+s_v^2.
\]

<!-- lemma-2: body_md -->
\label{lem:real-special}
The ring $R_v^{\square}/2R_v^{\square}$ has Krull dimension $d_n$.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Modulo $2$, the substitution $X=I_n+N$ changes $X^2=I_n$
into $N^2=0$. Over an algebraic closure of $k$, the square-zero
matrices of rank $q$, for $0\le q\le m:=\lfloor n/2\rfloor$,
form one conjugacy class, of Jordan type $(2^q,1^{n-2q})$.
Its stabilizer is the open set of units in its centralizer algebra.
The calculation in Lemma~\ref{lem:residual-centralizer} therefore
gives its dimension as
\[
n^2-((n-q)^2+q^2)=2q(n-q).
\]
Every such class lies in the closure of the rank-$m$ class:
multiply the upper-right entries of $m-q$ size-two blocks by a
parameter and specialize it to zero, then conjugate.
Thus the variety of square-zero matrices is irreducible of
dimension $2m(n-m)=d_n$. Since $k$ is perfect, the reduced
coordinate ring
\[
T=(A_n/2A_n)_{\mathrm{red}}
\]
is geometrically integral of dimension $d_n$.

Since $T$ is a finite-type domain over $k$, its localization at
the maximal ideal induced by $J_v$ has dimension $d_n$
\cite[Tag~02IJ]{Stacks}. Reduction modulo $2$ commutes with
Noetherian local completion. Removing nilpotents and completing
preserve Krull dimension \cite[Tag~07NU]{Stacks}, which proves
the assertion.

<!-- lemma-3: body_md -->
\label{lem:balanced-real-lift}
The representation $\bar\rho_v$ has a lift

<!-- proposition-1: body_md -->
\label{prop:real-base}
For every real prime $v$, we have
\[
\dim R_v^{\square}=1+d_n.
\]
For a family $(\bar\rho_v)_{v\in S_{\mathbb R}}$, let
\[
B=\mathop{\widehat{\bigotimes}}\limits_{\substack{v\in S_{\mathbb R}\\W(k)}}R_v^{\square},
\]

the maximal ideals of the factors. If $S_{\mathbb R}=\varnothing$, then we
set $B=W(k)$. Then we have
\begin{equation}\label{eq:real-base-dimensions}
\dim(B/2B)=r_1(K)d_n,\qquad \dim B=1+r_1(K)d_n.
\end{equation}
Moreover, $\operatorname{Spec}(B/2B)$ is equidimensional.

<!-- proof-4: body_md -->
We first prove the assertion for $R_v^{\square}$.
Let $a=\lfloor n/2\rfloor$ and $b=\lceil n/2\rceil$.
Over $F$, the conjugacy class of $\operatorname{diag}(I_a,-I_b)$
is closed and geometrically integral, of dimension
\[
n^2-a^2-b^2=2ab=d_n.
\]
Indeed, every involution is semisimple, and its trace determines
the two multiplicities. Let $\mathfrak p\subset A_n$ be the
contraction of the prime ideal defining this conjugacy class in
$A_n[1/2]$. Then $A_n/\mathfrak p$ is a domain containing $W(k)$,
and its fraction field has transcendence degree $d_n$ over $F$.
Evaluation at the lift in Lemma~\ref{lem:balanced-real-lift}
kills $\mathfrak p$, so $\mathfrak p\subset J_v$.
The dimension formula over the universally catenary discrete
valuation ring $W(k)$ gives
\[
\dim(A_n/\mathfrak p)_{J_v/\mathfrak p}=1+d_n
\]
\cite[Tag~02IJ]{Stacks}. The surjection
\[
R_v^{\square}\twoheadrightarrow
\widehat{(A_n/\mathfrak p)_{J_v/\mathfrak p}}
\]
therefore gives $\dim R_v^{\square}\ge1+d_n$.
Conversely, the principal ideal dimension inequality and
Lemma~\ref{lem:real-special} give
\[
\dim R_v^{\square}
\le1+\dim(R_v^{\square}/2R_v^{\square})=1+d_n.
\]

We now prove the assertions for $B$. They are immediate if
$S_{\mathbb R}=\varnothing$. Otherwise, form the ordinary tensor product
\[
A_{S_{\mathbb R}}=\bigotimes_{v\in S_{\mathbb R},W(k)}A_n,
\qquad
J=\ker\left(A_{S_{\mathbb R}}
\xrightarrow{\,\bigotimes_v\varepsilon_v\,}k\right).
\]
Here each factor is evaluated at its own $\bar\rho_v(c_v)$.
The completed tensor product satisfies
\[
B\simeq\widehat{(A_{S_{\mathbb R}})_J}.
\]
Indeed, both rings represent tuples of framed lifts of the
$\bar\rho_v$. In the notation of the proof of
Lemma~\ref{lem:real-special},
\[
(A_{S_{\mathbb R}}/2A_{S_{\mathbb R}})_{\mathrm{red}}
\simeq\bigotimes_{v\in S_{\mathbb R},k}T.
\]
This is a geometrically integral $k$-algebra of dimension
$r_1(K)d_n$. Localizing at the maximal ideal induced by $J$ and
completing, as in Lemma~\ref{lem:real-special}, gives
$\dim(B/2B)=r_1(K)d_n$.
The same calculation also proves equidimensionality after completion.
Indeed, the localization of
$(A_{S_{\mathbb R}}/2A_{S_{\mathbb R}})_{\mathrm{red}}$ at the
maximal ideal induced by $J$ is a local domain essentially of finite
type over $k$, hence is universally catenary. By Ratliff's theorem,
its completion is equidimensional
\cite[Tags~0AW2 and~0AW6]{Stacks}. The natural surjection from
$B/2B$ to this completion has nilpotent kernel, since completion
is exact on finite modules over a Noetherian local ring.
It therefore induces a homeomorphism on spectra.

For the integral dimension, take the product over $F$ of the
conjugacy classes used in the single-prime argument. It is
geometrically integral of dimension $r_1(K)d_n$.
Let $\mathfrak P\subset A_{S_{\mathbb R}}$ be the contraction
of its defining prime ideal in $A_{S_{\mathbb R}}[1/2]$.
The lifts from Lemma~\ref{lem:balanced-real-lift}, chosen for every
$v$, give an evaluation map $A_{S_{\mathbb R}}\to W(k)$ that
kills $\mathfrak P$ and reduces to the map defining $J$.
Thus $\mathfrak P\subset J$. The same dimension formula gives
\[
\dim(A_{S_{\mathbb R}}/\mathfrak P)_{J/\mathfrak P}
=1+r_1(K)d_n.
\]
Its completion is a quotient of $B$, so
$\dim B\ge1+r_1(K)d_n$. The reverse inequality follows from
$\dim B\le1+\dim(B/2B)$.

Absolute irreducibility implies that the matrices $\bar\rho(g)$ span
$M_n(k)$, by Burnside's theorem after extension to an algebraic closure.
Choose $n^2$ such matrices forming a basis. For every lift over $C$,
their lifts form a $C$-basis of $M_n(C)$ by Nakayama's lemma; hence
its centralizer consists of scalars.
Fix a representative of the universal deformation over $R_{\bar\rho}$.
The choices of frames are then $r_1(K)$ matrices in
$I_n+M_n(\mathfrak m_C)$ modulo simultaneous multiplication by
$1+\mathfrak m_C$. Normalizing the $(1,1)$-entry of one frame to $1$
removes this scalar uniquely and leaves $r_1(K)n^2-1$ free parameters.
This proves \eqref{eq:global-frame-parameters} and gives $B\to A$.

The induced tangent map is
\[
\theta:T_A\longrightarrow T_B
=\bigoplus_{v\in S_{\mathbb R}}Z^1(G_v,\operatorname{ad}\bar\rho).
\]
Varying the frames supplies all local coboundaries, so
\[
\operatorname{coker}\theta\simeq
\operatorname{coker}(\operatorname{res}^1),
\qquad
0\longrightarrow\operatorname{coker}(\operatorname{res}^0)
\longrightarrow\ker\theta
\longrightarrow\ker(\operatorname{res}^1)\longrightarrow0.
\]
Since $H^0(G_{K,S},\operatorname{ad}\bar\rho)=k$ and $S_{\mathbb R}\ne\varnothing$,
$\operatorname{res}^0$ is injective. Hence $\dim_k\ker\theta=t$,
and Lemma~\ref{lem:relative-relations} gives a minimal presentation
\[
\varphi:P=B[[X_1,\ldots,X_t]]\twoheadrightarrow A.
\]

We bound its relations by following the proof of
\cite[Key Lemma~5.2.2, Section~5.6]{Boeckle2013}, using $\operatorname{ad}\bar\rho$
in place of the trace-zero adjoint module.
Put $J=\ker\varphi$, let $\mathfrak m_P$ be the maximal ideal of
$P$, and set $E=J/\mathfrak m_PJ$. Nakayama's lemma gives
$\mu(J)=\dim_k E$. Consider the square-zero extension
\[
0\longrightarrow E\longrightarrow P/\mathfrak m_PJ
\longrightarrow A\longrightarrow0.
\]
For $u\in E^\vee=\operatorname{Hom}_k(E,k)$, pushing out along $u$
gives an extension of complete local $W(k)$-algebras
\[
0\longrightarrow I_u\longrightarrow P_u
\xrightarrow{\,\varphi_u\,}A\longrightarrow0,
\qquad I_u\simeq k,\quad\mathfrak m_{P_u}I_u=0.
\]

$S_{\mathbb R}$-framed deformation over $A$ and a continuous
set-theoretic lift $\widetilde\rho_u:G_{K,S}\to\operatorname{GL}_n(P_u)$.
Such a lift exists: the surjection $\operatorname{GL}_n(P_u)\to\operatorname{GL}_n(A)$
has finite kernel and admits a continuous set-theoretic section,
obtained by choosing coset representatives for an open subgroup
on which it is injective.
Its multiplicative defect defines a continuous cocycle
\[
\widetilde\rho_u(g)\widetilde\rho_u(h)
\widetilde\rho_u(gh)^{-1}=I_n+z_u(g,h),
\qquad z_u\in Z^2(G_{K,S},\operatorname{ad}\bar\rho).
\]
The class $[z_u]$ is independent of the lift and vanishes precisely
when $\rho_A$ lifts to a representation over $P_u$.
These classes are contractions of the obstruction class with
coefficients in $E$, so $u\mapsto[z_u]$ is $k$-linear.
The map $B\to P_u$ supplies local framed lifts; conjugating by
lifts of the $\gamma_{v,A}$ gives lifts of $\rho_A|_{G_v}$.
Thus every local restriction of $[z_u]$ vanishes, and we obtain
\[
\alpha:E^\vee\longrightarrow\ker(\operatorname{res}^2),
\qquad u\longmapsto[z_u].
\]

It remains to bound $\ker\alpha$. Put
\[
K_0=\ker\bigl(\operatorname{ct}(P)\to\operatorname{ct}(A)\bigr).
\]
There is a natural surjection
$\beta:E\twoheadrightarrow K_0$, induced by $J\subset P$.
Since the images of the $X_i$ form a basis of the relative
cotangent space, the decomposition
$\operatorname{ct}(P)=\operatorname{ct}(B)\oplus k^t$ identifies
\[
K_0\simeq\ker\bigl(\operatorname{ct}(B)\to\operatorname{ct}(A)\bigr),
\qquad K_0^\vee\simeq\operatorname{coker}\theta.
\]
For $0\ne u\in\ker\alpha$, lift $\rho_A$ and all its frames
to $P_u$. Universality gives a $W(k)$-algebra section
$s:A\to P_u$ of $\varphi_u$.
The decomposition $P_u=s(A)\oplus I_u$, together with
$\mathfrak m_{P_u}I_u=0$, implies
\[
\mathfrak m_{P_u}^2+2P_u=s(\mathfrak m_A^2+2A).
\]
Hence $I_u$ injects into $\operatorname{ct}(P_u)$.
If $e\in\ker\beta$, its image in $\operatorname{ct}(P_u)$ is zero,
while its image in $I_u\simeq k$ is $u(e)$. Thus $u(e)=0$.
Consequently $u$ factors uniquely through $K_0$, giving a linear
injection $\ker\alpha\hookrightarrow K_0^\vee$.
It follows that
\[
\mu(J)=\dim_k E
\le\dim_k\operatorname{coker}\theta
+\dim_k\ker(\operatorname{res}^2)=o.
\]
Finally, $\dim P=\dim B+t$, and quotienting by at most $o$
elements gives $\dim A\ge\dim B+t-o$.

D_{\mathrm{lift}}(\rho)
\le\dim A\le D_{\mathrm{lift}}(\rho)+h^2_\rho.
\]

\dim(R_{\bar\rho}/2R_{\bar\rho})
\ge\dim R_{\bar\rho}[1/2]\ge\dim A
\ge D_{\mathrm{lift}}(\rho).
\]
Lemma~\ref{lem:involution} gives the stated identity and shows
that its correction is positive exactly when
$s_v<\min(a_v,b_v)$ for some $v\in S_{\mathbb R}$.

\dim(R_{\bar\rho}/\mathfrak q)
=\dim(A/\mathfrak qA)+\dim(R_{\bar\rho}/\mathfrak P)

\]
Every minimal prime is associated \cite[Tag~05BV]{Stacks}.

\end{equation}

has dimension smaller than $\dim R_{\bar\rho}$, and
$\operatorname{depth}R_{\bar\rho}<\dim R_{\bar\rho}$.
Thus $R_{\bar\rho}$ is neither equidimensional nor
Cohen--Macaulay, and hence is not a complete intersection.
