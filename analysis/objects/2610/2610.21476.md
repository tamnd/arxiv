<!-- theorem-1: body_md -->
\label{thm:main}
For every deterministic finite initial state~$\left(\eta_0,\mu_0\right)$:
\begin{enumerate}
\item If~$\theta\leq1$, then~$\mathds{P}\left(\tau_0<\infty\right)=1$.
\item Suppose~$\theta>1$ and, for some~$0<r\leq R$ and~$k_0>0$,
\begin{equation*}
 K\left(x\right)=0\quad\text{for }\left|x\right|>R,
 \qquad K\left(x\right)\geq k_0\quad\text{for }\left|x\right|\leq r.
\end{equation*}
Then~$\;\mathds{P}\left(\tau_0=\infty\right)>0
 \quad\Longleftrightarrow\quad
 \eta_0\left(\mathds{R}^d\right)\geq2\ \text{ or }\ \eta_0\left(\mathds{R}^d\right)=1,\ \mu_0\left(\mathds{R}^d\right)>0$.
\end{enumerate}

<!-- proof-1: body_md -->
Let~$V_t
\mathrel{\mathop:}=
\eta_t\left(\mathds{R}^d\right)+\theta\mu_t\left(\mathds{R}^d\right)$,
and let~$N_t^b$ and~$N_t^d$ denote the numbers of births and deaths by time~$t$. Let~$A_t$ be the total consumption rate at time~$t$, given by~$A_t
\mathrel{\mathop:}= \int_{\mathds{R}^d}\left(K*\eta_t\right)\left(y\right)\,\mu_t(dy)$.
 Between jumps, resource consumption decreases~$V_t$ at rate~$\theta A_t$. A birth increases~$V_t$ by~$1$, while a death removes one particle and adds one unit of resource, so it changes~$V_t$ by~$\theta-1$. Hence,
\begin{equation}\label{eq:V-decomposition}
V_t
=
V_0+N_t^b+(\theta-1)N_t^d
-\theta\int_0^t A_s\,ds.
\end{equation}
By Fubini's theorem and the symmetry of~$K$ we have~$A_t = \int_{\mathds{R}^d}\left(K*\mu_t\right)\left(x\right)\,\eta_t(dx)$, so the process~$\theta A_t$ is also a compensator for~$N_t^b$. Thus
\begin{equation*}
M_t^b
\mathrel{\mathop:}=
N_t^b-\theta\int_0^t A_s\,ds,
\qquad
M_t^d
\mathrel{\mathop:}=
N_t^d-\int_0^t\eta_s\left(\mathds{R}^d\right)\,ds
\end{equation*}
are local martingales. Substituting these in~\eqref{eq:V-decomposition} gives
\begin{align*}
V_t
&=
V_0+M_t^b+(\theta-1)M_t^d
+(\theta-1)\int_0^t\eta_s\left(\mathds{R}^d\right)\,ds,
\end{align*}
so if~$\theta\leq1$, then~$V$ is a nonnegative local supermartingale. Consequently, it converges almost surely.

Every birth produces a jump of size~$1$ in~$V$, so convergence implies that only finitely many births occur. Hence only finitely many particles ever exist, and since all eventually die, the population becomes extinct in finite time almost surely.

<!-- lemma-1: body_md -->
\label{lem:short-time-growth-new}
There exist~$h>0$,~$\lambda>1$,~$\gamma>0$, and~$M_*\in\mathds{N}$ such that the following holds for every~$L\geq 0$. Let~$\tau$ be a finite stopping time, let~$z$ be~$\mathcal F_\tau$-measurable, and let~$\mathsf P\subseteq\mathsf L_\tau$ be an~$\mathcal F_\tau$-measurable collection of~$M\geq M_*$ particles contained in a ball~$B_L\left(z\right)$. Let
\begin{equation*}
\mathsf C
\mathrel{\mathop:}=
\left\{
\mathsf{q}\in \mathsf P \cap \mathsf{D}_{\tau + \left(1-\kappa\right)h}:
\sup_{\tau\leq s\leq d_{\mathsf{q}}}
\left|X_s\left(\mathsf{q}\right)-X_\tau\left(\mathsf{q}\right)\right|
\leq\frac r4
\right\}
\end{equation*}
be the particles which die sufficiently early and remain nearby. Then
\begin{equation*}
\mathds{P}\left(
\eta_{\tau+h}\left(B_{L+R+r/2}\left(z\right)\right)<\lambda M,\text{every }\mathsf{q}\in\mathsf C\text{ is $\alpha$-consumed by }\tau+h
\,\middle|\,
\mathcal F_\tau
\right)
\leq e^{-\gamma M}.
\end{equation*}
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
The argument relies on there being many particles that can make use of the~$\alpha$-consumed corpses from~$\mathsf C$. Let
\begin{equation*}
\mathsf S
\mathrel{\mathop:}=
\left\{
\mathsf{p}\in \mathsf P\cap \mathsf{L}_{\tau+h}:
\sup_{\tau\leq s\leq\tau+h}
\left|X_s\left(\mathsf{p}\right)-X_\tau\left(\mathsf{p}\right)\right|
\leq\frac r4
\right\}.
\end{equation*}
The set~$\mathsf S$ consists of the particles which survive until~$\tau+h$ and do not stray too far. 

The argument requires~$\left|\mathsf S\right|$ and~$\left|\mathsf C\right|$ to be comparable to~$M$. To show this, let~$W$ be a standard Brownian motion started at the origin, and let~$E\sim\operatorname{Exp}\left(1\right)$, independent of each other. The process~$W$ represents the displacement of one particle after time~$\tau$, while~$E$ represents its lifetime. Set
\begin{align*}
s_h
&\mathrel{\mathop:}=
\mathds{P}\left(
E>h,\sup_{0\leq s\leq h}\left|W_s\right|\leq\frac r4
\right)=1-h + o(h),
\\
c_h
&\mathrel{\mathop:}=
\mathds{P}\left(
E\leq(1-\kappa)h,\sup_{0\leq s\leq E}\left|W_s\right|\leq\frac r4
\right)=(1-\kappa)h + o(h).
\end{align*}
Thus~$s_h$ is the probability that a particle belongs to~$\mathsf S$, while~$c_h$ is the probability that it belongs to~$\mathsf C$. The equalities above hold by the reflection principle, which gives~$\mathds{P}\left(\sup_{0\leq s\leq h} \left|W_s\right|>\frac{r}{4}\right)=o(h)$.
Conditional on~$\mathcal F_\tau$, the trajectories and lifespans of the particles in~$\mathsf P$ are independent, with the laws of~$W$ and~$E$. Hence, for every~$\varepsilon<\min\left\{s_h,c_h\right\}$, a Chernoff bound gives, for some~$\gamma'>0$,
\begin{equation}\label{eq:SC-lower-bounds}
\mathds{P}\left(
\left|\mathsf S\right|<(s_h-\varepsilon)M
\text{ or }
\left|\mathsf C\right|<(c_h-\varepsilon)M
\,\middle|\,
\mathcal F_\tau
\right)
\leq
2e^{-\gamma' M}.
\end{equation}

Next we show that if corpses in~$\mathsf C$ are~$\alpha$-consumed then there are many new births. For~$s\in[\tau,\tau+h]$, let
\begin{equation*}
\mathsf N_s
\mathrel{\mathop:}=
\left\{
\mathsf{p}\in\mathsf T_s\setminus \mathsf T_{\tau}:
X_{b_{\mathsf{p}}}\left(\mathsf{p}\right)\in B_{L+R+r/4}\left(z\right)
\right\}
\end{equation*}
be the new particles born in~$B_{L+R+r/4}\left(z\right)$ by time~$s$, and set
\begin{equation*}
\Lambda_s
\mathrel{\mathop:}=
\theta\int_\tau^s\int_{B_{L+R+r/4}\left(z\right)}
\left(K*\mu_u\right)\left(x\right)\,\eta_u(dx)\,du.
\end{equation*}
Since~$z$ is~$\mathcal F_\tau$-measurable, the ball~$B_{L+R+r/4}\left(z\right)$ is fixed conditional on~$\mathcal F_\tau$. A particle at~$x$ gives birth at rate~$\theta\left(K*\mu_t\right)(x)$, so~$\Lambda$ is a compensator of~$\left|\mathsf N\right|$. Hence~$\left|\mathsf N\right|-\Lambda$ is a local martingale.

We obtain the following lower bound on~$\Lambda_{\tau+h}$:
\begin{align*}
\Lambda_{\tau+h}
&=
\theta\int_\tau^{\tau+h}
\int_{B_{L+R+r/4}\left(z\right)}
\int_{\mathds{R}^d}
K\left(x-y\right)\,\mu_s(dy)\,\eta_s(dx)\,ds
\\
&\geq
\theta\int_\tau^{\tau+h}
\int_{B_{L+R+r/4}\left(z\right)}
\sum_{\mathsf q\in\mathsf D_s}
m_{\mathsf{q}}(s)
K\left(x-X_{d_{\mathsf{q}}}\left(\mathsf{q}\right)\right)
\,\eta_s(dx)\,ds
\\
&\geq
\theta\int_\tau^{\tau+h}
\int_{B_{L+R+r/4}\left(z\right)}
\sum_{\mathsf q\in\mathsf C}
\mathbf{1}\left(d_{\mathsf q}\leq s\right)m_{\mathsf q}(s)
K\left(x-X_{d_{\mathsf q}}(\mathsf q)\right)
\,\eta_s(dx)\,ds
\\
&=
\theta\sum_{\mathsf q\in\mathsf C}
\int_{d_{\mathsf q}}^{\tau+h} 
m_{\mathsf q}(s)\int_{B_{L+R+r/4}\left(z\right)}
K\left(X_{d_{\mathsf q}}\left(\mathsf q\right)-x\right)\eta_s(dx)
\,ds
\\
&=
\theta\sum_{\mathsf q\in\mathsf C}
\int_{d_{\mathsf q}}^{\tau+h}
m_{\mathsf q}(s)
\left(K*\eta_s\right)\left(X_{d_{\mathsf q}}(\mathsf q)\right)
\,ds
\\
&=
\theta
\sum_{\mathsf q\in\mathsf C}
\int_{d_{\mathsf q}}^{\tau+h}
\left(-\frac{d}{ds}m_{\mathsf q}(s)\right)\,ds
=
\theta
\sum_{\mathsf q\in\mathsf C}
\left(1-m_{\mathsf q}(\tau+h)\right).
\end{align*}
The first line is the definition of~$\Lambda_{\tau+h}$ with the convolution expanded.
The second line comes from the resource evolution formula~\eqref{eq:resource-evolution} dropping the contribution from~$\mu_0$. The third line is just~$\mathsf{C}\cap\mathsf D_s\subseteq \mathsf D_s$. The fourth line follows from Fubini's theorem and the symmetry of~$K$. The fifth line holds because for~$\mathsf{q}\in\mathsf C$ we have~$X_{d_{\mathsf{q}}}\left(\mathsf{q}\right)\in B_{L+r/4}\left(z\right)$, and~$K$ has range~$R$, so~$K\left(X_{d_{\mathsf q}}(\mathsf q)-x\right)$ vanishes outside~$B_{L+R+r/4}\left(z\right)$. The last two identities follow from~\eqref{eq:corpse-mass-evolution}: differentiating gives~$\frac{d}{ds}m_{\mathsf q}(s)
=
-m_{\mathsf q}(s)
\left(K*\eta_s\right)\left(X_{d_{\mathsf q}}(\mathsf q)\right)$,
and~$m_{\mathsf q}(d_{\mathsf q})=1$.
Therefore, on the event that every corpse in~$\mathsf C$ is~$\alpha$-consumed by~$\tau+h$,
\begin{equation*}
\Lambda_{\tau+h}
\geq \theta\sum_{\mathsf q\in\mathsf C}\left(1-(1-\alpha)\right)\geq  
\alpha\theta\left|\mathsf C\right|.
\end{equation*}
Consequently,
\begin{equation}\label{eq:compensator-lower-bound-event}
\left\{
\left|\mathsf C\right|\geq(c_h-\varepsilon)M
\right\}
\cap
\left\{
\text{every }\mathsf{q}\in\mathsf C
\text{ is $\alpha$-consumed by }\tau+h
\right\}
\subseteq
\left\{
\Lambda_{\tau+h}
\geq
\alpha\theta(c_h-\varepsilon)M
\right\}.
\end{equation}

A lower bound on~$\Lambda$ is useful because it is unlikely that a compensator of a counting process is large and the process small. For~$a>0$, set
\begin{equation*}
Z_s
\mathrel{\mathop:}=
\exp\left\{
-a\left|\mathsf N_s\right|
+
\left(1-e^{-a}\right)\Lambda_s
\right\},
\qquad
s\in[\tau,\tau+h].
\end{equation*}
Since~$\left|\mathsf N\right|-\Lambda$ is a local martingale, we have
\begin{equation*}
dZ_s
=
Z_{s-}\left(e^{-a}-1\right)
\,d\left(\left|\mathsf N_s\right|-\Lambda_s\right).
\end{equation*}
Thus~$Z$ is a nonnegative local martingale, hence a supermartingale, with~$Z_\tau=1$. Choose~$a>0$ sufficiently small that~$\gamma''\mathrel{\mathop:}=1-e^{-a}-a(1-\varepsilon)>0$. On the event where~$\left|\mathsf N_{\tau+h}\right|\leq(1-\varepsilon)v$ and~$\Lambda_{\tau+h}\geq v$, we have~$Z_{\tau+h}\geq e^{\gamma''v}$, so
\begin{equation}\label{eq:compensator-large-process-small-unlikely}
\mathds{P}\left(
\left|\mathsf N_{\tau+h}\right|\leq(1-\varepsilon)v,\
\Lambda_{\tau+h}\geq v
\,\middle|\,
\mathcal F_\tau
\right)
\leq
e^{-\gamma''v}
\mathds{E}\left[Z_{\tau+h}\,\middle|\,\mathcal F_\tau\right]
\leq
e^{-\gamma''v}.
\end{equation}
Thus~\eqref{eq:compensator-lower-bound-event} and~\eqref{eq:compensator-large-process-small-unlikely} with~$v=\alpha\theta(c_h-\varepsilon)M$ give, possibly after decreasing~$\gamma''$,
\begin{equation}\label{eq:N-lower-bound-very-likely}
\mathds{P}\left(
\begin{gathered}
\left|\mathsf N_{\tau+h}\right|
<
(1-\varepsilon)\alpha\theta(c_h-\varepsilon)M,\quad
\left|\mathsf C\right|\geq(c_h-\varepsilon)M,
\\
\text{every }\mathsf{q}\in\mathsf C
\text{ is $\alpha$-consumed by }\tau+h
\end{gathered}
\,\middle|\,
\mathcal F_\tau
\right)
\leq
e^{-\gamma'' M}.
\end{equation}

On the event that~$\left|\mathsf N_{\tau+h}\right|$ is large we have many newborns. We verify that not too many of them die or stray too far away. Let
\begin{equation*}
\mathsf{NS}
\mathrel{\mathop:}=
\left\{
\mathsf{p}\in\mathsf N_{\tau+h}\cap \mathsf{L}_{\tau+h}:
\sup_{b_{\mathsf{p}}\leq s\leq\tau+h}
\left|X_s\left(\mathsf{p}\right)-X_{b_{\mathsf{p}}}\left(\mathsf{p}\right)\right|
\leq\frac r4
\right\}
\end{equation*}
be the newborns in~$\mathsf N_{\tau+h}$ that survive until~$\tau+h$ and stay nearby. Set~\mbox{$n\mathrel{\mathop:}=\lfloor(1-\varepsilon)\alpha\theta(c_h-\varepsilon)M\rfloor$}.
Each newborn, conditionally on the history up to its birth time, survives for the next~$h$ units of time and stays within~$r/4$ of its birth location with probability~$s_h$, independently of the past. Such a newborn necessarily belongs to~$\mathsf{NS}$. Thus, on~$\left\{\left|\mathsf N_{\tau+h}\right|\geq n\right\}$, the lower tail of~$\left|\mathsf{NS}\right|$ is bounded by that of a~$\operatorname{Bin}\left(n,s_h\right)$ random variable. Hence a Chernoff bound gives
\begin{equation}\label{eq:NS-lower-bound-very-likely}
\mathds{P}\left(
\left|\mathsf N_{\tau+h}\right|\geq n,\
\left|\mathsf{NS}\right|<(s_h-\varepsilon)n
\,\middle|\,
\mathcal F_\tau
\right)
\leq
e^{-\gamma''' M}
\end{equation}
for some~$\gamma'''>0$. Thus it is unlikely that there are many newborns but few survivors.

We check how far the particles from~$\mathsf S$ and~$\mathsf{NS}$ could have gone. Every particle in~$\mathsf S$ lies in~$B_{L+r/4}\left(z\right)$. Every particle in~$\mathsf{NS}$ was born in~$B_{L+R+r/4}\left(z\right)$ and moves by at most~$r/4$ after birth, so at time~$\tau+h$ it lies in~$B_{L+R+r/2}\left(z\right)$. Thus all particles of~$\mathsf S\cup\mathsf{NS}$ lie in~$B_{L+R+r/2}\left(z\right)$ at time~$\tau+h$.

The sets~$\mathsf S$ and~$\mathsf{NS}$ are disjoint, because particles in the former existed before time~$\tau$ and those in the latter were born during~$(\tau,\tau+h]$. Therefore
\begin{equation*}
\eta_{\tau+h}\left(B_{L+R+r/2}\left(z\right)\right)
\geq
\left|\mathsf S\right|+\left|\mathsf{NS}\right|.
\end{equation*}

Now we choose the parameters. For small~$h$ we have
\begin{equation*}
s_h+s_h\alpha\theta c_h
=
1+\left((1-\kappa)\alpha\theta-1\right)h+o(h)>1.
\end{equation*}
Hence we can choose~$\varepsilon>0$ and then~$\lambda>1$ with
\begin{equation*}
\lambda
<
(s_h-\varepsilon)
\left[1+(1-\varepsilon)\alpha\theta(c_h-\varepsilon)\right],
\end{equation*}
since the right-hand side tends to~$s_h+s_h\alpha\theta c_h$ as~$\varepsilon\to0$. By the definition of~$n$, for all sufficiently large~$M$ this implies
\begin{equation}\label{eq:deterministic-one-step-growth}
(s_h-\varepsilon)M+(s_h-\varepsilon)n
\geq
\lambda M.
\end{equation}

We finish the proof by combining the estimates. By~\eqref{eq:SC-lower-bounds}, we have~$\left|\mathsf S\right|\geq(s_h-\varepsilon)M$ and~$\left|\mathsf C\right|\geq(c_h-\varepsilon)M$, outside of an event with probability~$2e^{-\gamma' M}$. On the event that every corpse in~$\mathsf C$ is~$\alpha$-consumed by~$\tau+h$, the bound~\eqref{eq:N-lower-bound-very-likely} gives~$\left|\mathsf N_{\tau+h}\right|\geq n$, outside an event of probability at most~$e^{-\gamma''M}$. On the event~$\left|\mathsf N_{\tau+h}\right|\geq n$, the bound~\eqref{eq:NS-lower-bound-very-likely} gives~$\left|\mathsf{NS}\right|\geq(s_h-\varepsilon)n$, outside an event of probability at most~$e^{-\gamma'''M}$. Therefore, by~\eqref{eq:deterministic-one-step-growth},
\begin{equation*}
\eta_{\tau+h}\left(B_{L+R+r/2}\left(z\right)\right)
\geq
\left|\mathsf S\right|+\left|\mathsf{NS}\right|
\geq
\lambda M
\end{equation*}
whenever all corpses in~$\mathsf C$ are~$\alpha$-consumed, except on an event with probability no larger than~$2e^{-\gamma' M}+e^{-\gamma'' M}+e^{-\gamma''' M}$. This proves the lemma for~$0<\gamma<\min\left\{\gamma',\gamma'',\gamma'''\right\}$ and sufficiently large~$M_*$.

Fix~$k\in\mathds{N}$, to be chosen later, and set
\begin{equation*}
T\mathrel{\mathop:}=(k+1)h,
\qquad
B_j
\mathrel{\mathop:}=
B_{r+r/4+j(R+r/2)}\left(z\right),
\qquad
j=0,\ldots,k,
\end{equation*}
where~$h$ is as in Lemma~\ref{lem:short-time-growth-new}. Write~$B\mathrel{\mathop:}= B_k$. Our goal is to show that, except with probability exponentially small in~$H$, we have~$\eta_{S+T}\left(B\right)\gtrsim\lambda^kH$. The radius of~$B$ grows linearly in~$k$, so~$B$ can be covered by~$O\left(k^d\right)$ balls of radius~$r$. Since~$\lambda>1$, we can choose~$k$ so that one of these~$r$-balls contains at least~$2H$ particles.

We prove the lower bound on~$\eta_{S+T}\left(B\right)$ by repeatedly applying Lemma~\ref{lem:short-time-growth-new} to~$k$ successive time intervals of length~$h$. The~$j$th application starts with a dense colony in~$B_j$ and implies the existence of a dense colony in~$B_{j+1}$. Thus all relevant particles and corpses remain in~$B$. To iterate, we must ensure that every corpse in the corresponding sets~$\mathsf C$ is~$\alpha$-consumed. 

Recall that~$K\left(x\right)\geq k_0$ whenever~$\left|x\right|\leq r$. Consider one of the growth intervals~$\left[t,t+h\right]$ and a corpse~$\mathsf{q}\in\mathsf C$, as in Lemma~\ref{lem:short-time-growth-new}. Since~$d_{\mathsf q}\leq t+(1-\kappa)h$, the corpse has at least~$\kappa h$ time to be consumed before the end of the interval. If at least~$n$ particles remain within distance~$r$ of~$X_{d_{\mathsf{q}}}\left(\mathsf{q}\right)$ throughout~$\left[t,t+h\right]$, then
\begin{equation}\label{eq:many-particles-means-fast-consumption}
m_{\mathsf{q}}(t+h)
\leq
e^{-k_0n(t+h-d_{\mathsf{q}})}
\leq
e^{-k_0n\kappa h}.
\end{equation}
Choose~$n\in\mathds{N}$ large enough that~$e^{-k_0n\kappa h}\leq1-\alpha$. Thus~$n$ nearby particles are enough to make a corpse~$\alpha$-consumed within time~$\kappa h$.

To apply Lemma~\ref{lem:short-time-growth-new} repeatedly, we only need to verify that there are at least~$n$ particles within distance~$r$ of any point in~$B$, throughout~$\left[S+h,S+T\right]$.  Choose~$\mathcal F_S$-measurable points~$z_1,\ldots,z_l$ such that
\begin{equation*}
B
\subseteq
\bigcup_{m=1}^l B_{r/2}\left(z_m\right),
\qquad
l\leq C(1+k)^d,
\end{equation*}
where~$C<\infty$ depends only on~$d,r$, and~$R$. For~$m\leq l$, define
\begin{equation*}
\mathsf D_m
\mathrel{\mathop:}=
\left\{
\mathsf{p}\in\mathsf L_S\cap \mathsf L_{S+T}:
X_S\left(\mathsf{p}\right)\in B_r\left(z\right),\
X_s\left(\mathsf{p}\right)\in B_{r/2}\left(z_m\right)
\text{ for every }s\in[S+h,S+T]
\right\}.
\end{equation*}
Thus~$\mathsf D_m$ consists of particles from the initial colony which enter the~$m$th covering ball by time~$S+h$, then remain there and alive until~$S+T$. Let~$\mathcal{D}\mathrel{\mathop:}=\bigcap_{m=1}^l\left\{\left|\mathsf D_m\right|\geq n\right\}$. On~$\mathcal{D}$, every point of~$B$ has at least~$n$ particles within distance~$r$ throughout~$\left[S+h,S+T\right]$.

For fixed~$k$, there is~$p>0$ such that, conditional on~$\mathcal F_S$, every particle in~$B_r\left(z\right)$ belongs to~$\mathsf D_m$ with probability at least~$p$, uniformly in~$m\leq l$. Indeed, the centers~$z_m$ lie at a bounded distance from~$z$, and a Brownian particle has uniformly positive probability to be in~$B_{r/2}\left(z_m\right)$ at time~$S+h$, remain there until~$S+T$, and survive throughout this interval. Conditional on~$\mathcal F_S$, these events are independent for the particles in~$B_r\left(z\right)$. Hence, on~$\left\{\eta_S\left(B_r\left(z\right)\right)\geq H\right\}$, each~$\left|\mathsf D_m\right|$ stochastically dominates~$\operatorname{Bin}\left(H,p\right)$. Since~$n$ is fixed, a binomial lower-tail bound and a union bound give~$\gamma'>0$ such that, for all~$H$ sufficiently large,
\begin{equation}\label{eq:density-background-failure}
\mathds{P}\left(
\mathcal{D}^c
\,\middle|\,
\mathcal F_S
\right)
\leq
e^{-\gamma'H}.
\end{equation}

We now set up the~$k$ applications of Lemma~\ref{lem:short-time-growth-new}. Set
\begin{equation*}
t_j\mathrel{\mathop:}= S+(j+1)h,
\qquad
M_0\mathrel{\mathop:}=\left\lfloor\frac{s_hH}{2}\right\rfloor,
\qquad
M_{j+1}\mathrel{\mathop:}=\lfloor\lambda M_j\rfloor,
\end{equation*}
for~$j\leq k$ in the first definition and~$j\leq k-1$ in the last. Thus~$t_0=S+h$,~$t_k=S+T$, and~$M_j$ is a lower bound on population size after~$j$ growth steps, with~$M_k=\frac{s_h}{2}\lambda^kH+O(1)$. If~$\eta_{t_j}\left(B_j\right)\geq M_j$ then
let~$\mathsf P_j$ be an~$\mathcal F_{t_j}$-measurable set of~$M_j$ particles in~$B_j$; otherwise let~$\mathsf P_j=\varnothing$. 
Let~$\mathsf C_j$ be the set~$\mathsf C$ from Lemma~\ref{lem:short-time-growth-new} corresponding to~$\mathsf P_j$ on the interval~$\left[t_j,t_{j+1}\right]$,
\begin{equation*}
\mathsf C_j
\mathrel{\mathop:}=

\end{equation*}
These are precisely the corpses whose consumption is required in the~$j$th growth step.

Indeed, if~$\mathsf{q}\in\mathsf C_j$, then its death location belongs to~$B_{j+1}\subseteq B$, so there are at least~$n$ particles within distance~$r$ of the corpse throughout~$\left[d_{\mathsf{q}},t_{j+1}\right]$. Now we start the iteration. \eta_{t_0}\left(B_0\right)<\frac{s_h H}{2}
\,\middle|\,
\mathcal F_S
\right)
\leq
e^{-\gamma''H}.
\end{equation}

\mathcal F_S

\mathcal F_S

\mathcal F_S

\mathcal F_S

\cup
\mathcal A_0^c
\cup

\mathcal F_S

For every~$H\in\mathds{N}$ and every finite initial state with at least two particles, or one particle and positive resource,
there exists an~$r$-ball~$B$ such that for any~$T>0$ we have~$\mathds{P}\left(

\right)

We first treat the case of one particle~$\mathsf p$ and~$\mu_0\left(\mathds{R}^d\right)>0$. Then there exists an~$r/2$-ball~$B'$ such that~$\mu_0\left(B'\right)\geq \varepsilon$ for some~$\varepsilon>0$. Let~$B$ be an~$r$-ball with the same center as~$B'$. For any~$x,y\in B'$ we have~$K\left(x-y\right)\geq k_0$. Consider the event that~$\mathsf p$ enters~$B'$ by time~$\frac{T}{2}$ without branching, does not leave~$B'$ before~$T$, survives until~$T$, branches exactly~$H-1$ times while in~$B'$, and its offspring never leave~$B'$, do not branch and do not die before~$T$. Because~$K$ is bounded, on this event we have~$\left(K*\eta_t\right)\left(y\right)\leq H\left\VertK\right\Vert_\infty$, so by~\eqref{eq:resource-between-deaths} the resource available in~$B'$ stays uniformly bounded below throughout~$\left[0,T\right]$: 
\begin{equation*}
\mu_t\left(B'\right)
\geq
\varepsilon e^{-H\left\VertK\right\Vert_\infty t}
\geq
\varepsilon e^{-H\left\VertK\right\Vert_\infty T}.
\end{equation*}
Thus the rate at which~$\mathsf p$ branches while in~$B'$ is at least~$\theta k_0\varepsilon e^{-H\left\VertK\right\Vert_\infty T}$, and it is at most~$\theta\left\VertK\right\Vert_\infty\mu_0\left(\mathds{R}^d\right)$, since on this event there are no deaths in~$\left[0,T\right]$, so~$\mu_t\left(\mathds{R}^d\right)\leq \mu_0\left(\mathds{R}^d\right)$. Since the particle motions are independent of the branching and of each other, the event has positive probability. On it,~$\eta_T\left(B\right)\geq\eta_T\left(B'\right)=H$.

The second case,~$\eta_0\left(\mathds{R}^d\right)\geq2$, reduces to the first with positive probability, via the event that all initial particles but one die before any branching.

Let~$T$,~$\gamma$, and~$H_0$ be as in Proposition~\ref{prop:dense-doubling-new}, and choose~$H\geq H_0$ so large that
\begin{equation*}
\sum_{k=0}^{\infty}e^{-\gamma2^kH}<1.
\end{equation*}
By Lemma~\ref{lem:entrance-new}, there is an~$r$-ball~$B_0$ such that~$\mathds{P}\left(\eta_T\left(B_0\right)\geq H\right)>0$.

On~$\left\{\eta_T\left(B_0\right)\geq H\right\}$, apply Proposition~\ref{prop:dense-doubling-new} successively at times~$T,2T,\ldots$, using at each step the measurable ball supplied by the proposition. If the first~$k$ doublings have succeeded, the~$k$th application starts from an~$r$-ball containing at least~$2^kH$ particles, so its conditional probability of failure is at most~$e^{-\gamma2^kH}$. Hence, by the tower property and a union bound,
\begin{equation*}
\mathds{P}\left(
\eta_{(k+1)T}\left(B_k\right)\geq2^kH
\text{ for every }k\geq0
\,\middle|\,
\eta_T\left(B_0\right)\geq H
\right)
\geq
1-\sum_{k=0}^{\infty}e^{-\gamma2^kH}
>0,
\end{equation*}
where~$B_k$ denotes the ball selected after~$k$ successful steps. Since the entrance event also has positive probability, with positive probability the population is nonempty at all times~$(k+1)T$. Extinction is absorbing, so on this event~$\tau_0=\infty$. Thus~$\mathds{P}\left(\tau_0=\infty\right)>0$.
