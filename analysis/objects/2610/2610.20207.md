Let $\Omega=\Omega^+\subset {\mathbb R}^2$ be a Jordan domain, let $\Omega^-={\mathbb R}^2\setminus \overline{\Omega^+}$, and let $\Gamma=
    \partial\Omega^+$.  Then, up to a set of zero ${\mathcal H}^1$-measure,
  \begin{equation}\label{eqCar00}
    \mbox{$\displaystyle {\mathcal E}_\Omega(x)^2:=\int_0^1 {\varepsilon}_\Omega(x,r)^2\,\frac{dr}r<\infty$\; for $x\in\Gamma$ \quad$\Leftrightarrow$ \quad
      $\Omega$ has a tangent at $x\in\Gamma$.}
  \end{equation}
  In particular, the set
  $G=\{x\in\Gamma:{\mathcal E}(x)<\infty\}$ is $1$-rectifiable.

Let $\Omega^+$ and $\Omega^-$ be two disjoint open subsets of ${\mathbb R}^{n+1}$. Suppose that $\mu$ is an $n$-Ahlfors regular measure with $\operatorname{supp}(\mu) = \partial \Omega^+ \cup \partial \Omega^-$. Then the following are equivalent.
  \begin{enumerate}

    \item There is a constant $C_1$ so that for each ball $B$ centered on $\operatorname{supp}(\mu)$ it holds that
          \begin{equation*} 
            \int_B \int_0^{r(B)} {\varepsilon}_{n,\Omega^\pm} (x,r)^2 \, \tfrac{dr}{r} \, d\mu(x) \leq C_1 r(B)^n.
          \end{equation*}
  \end{enumerate}

<!-- lemma-1: body_md -->
\label{thm:epsilon-to-second-difference}
  Let $f\colon I\to\mathbb{R}$ be $L$-Lipschitz with $L<1$ and $0<\alpha\leq1$. Suppose that
  \[
    {\widetilde{{\varepsilon}}}_f(z,r)\leq C_{\varepsilon} r^\alpha
  \]
  for some $z\in I$ and $r$ as below. If $h>0$,
  \[
    r :=|{\varphi}_f(z+h)-{\varphi}_f(z)| \quad\text{and}\quad [z-r,z+r]\subset I,
  \]
  then
  \begin{equation}
    |\delta_h^2f(z)|\leq M h^{1+\alpha},\qquad
    M=(1+L)(1+L^2)^{(1+\alpha)/2}C_{\varepsilon}.
  \end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
By the choice of $r$, the right intersection point of $\Gamma_f\cap S({\varphi}_f(z),r)$ is $X_+ :={\varphi}_f(z+h)$.
  Write the left one as $X_- :={\varphi}_f(z-k)$, where $k\in (0,r]$, and put
  $X :={\varphi}_f(z)$ and $v_\pm :=X_\pm-X$.

  Let $\theta\in(0,2\pi)$ be the angle of the arc of $\partial B({\varphi}_f(z),r)$ in $\Omega_f^+$. Then
  \[
    {\widetilde{\varepsilon}}_f(z,r)=|\theta-\pi|,
  \]
  and elementary circle geometry gives
  \begin{equation}\label{eq:vector-estimate}
    |v_++v_-|=2r|\cos(\theta/2)|
    =2r\sin\bigl({{\widetilde{\varepsilon}}}_f(z,r)/2\bigr)
    \leq C_{\varepsilon} r^{1+\alpha}.
  \end{equation}

  Recall that
  \[
    v_++v_-=(h-k,\ f(z+h)+f(z-k)-2f(z)).
  \]
  Consequently,
  \begin{align*}
    |\delta_h^2f(z)|
     & \leq |f(z+h)+f(z-k)-2f(z)|+|f(z-h)-f(z-k)| \\
     & \leq |v_++v_-|+L|h-k|                      \\
     & \leq(1+L)|v_++v_-|                         \\
     & \leq(1+L)C_{\varepsilon} r^{1+\alpha}.
  \end{align*}
  Finally,
  \[
    h\leq r=\sqrt{h^2+|f(z+h)-f(z)|^2}\leq\sqrt{1+L^2}\,h,
  \]
  which proves the claim.

<!-- lemma-2: body_md -->
\label{lem:second-difference}
  Let $f\colon I\to\mathbb{R}$ be continuous and $0<\alpha\leq1$. Suppose
  \begin{equation}\label{eq:criterion-hypothesis}
    |\delta_h^2f(z)|\leq M|h|^{1+\alpha}
    \quad\text{whenever }z-h,z+h\in I.
  \end{equation}
  Then $f\in C^1(I)$ and
  \begin{equation}\label{eq:holder-constant}
    |f'(x)-f'(y)|\leq\frac{M}{2^\alpha-1}|x-y|^\alpha
    \qquad\forall x,y\in I.
  \end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
For $t>0$ with $[x,x+t]\subset I$, set
  \[
    q_t^+(x)=\frac{f(x+t)-f(x)}t.
  \]
  The hypothesis, applied at the midpoint of $[x,x+t]$, gives
  \begin{equation}\label{eq:quotient-step}
    |q_t^+(x)-q_{t/2}^+(x)|
    =\frac{|\delta_{t/2}^2f(x+t/2)|}{t}
    \leq\frac{M}{2^{1+\alpha}}t^\alpha.
  \end{equation}
  Fix a compact subinterval $J\subset I$ and choose $t_0>0$ small enough such that $q_{t_0}^+$ is defined on $J$.
  Summing \cref{eq:quotient-step} shows
  that $(q_{2^{-m}t_0}^+)_m$ converges uniformly to a continuous function $v$.

  We now show that $v$ is indeed the derivative.
  For any $a<b$ in the interior of $J$ and $t_m=2^{-m}t_0$,
  \begin{align*}
    \int_a^b q_{t_m}^+(x)\,dx
    =\frac1{t_m}\bigg( \int_b^{b+t_m}f(s)\,\mathrm{d} s - \int_a^{a+t_m}f(s)\,\mathrm{d} s \bigg)
    \longrightarrow f(b)-f(a)
  \end{align*}
  by continuity of $f$. Uniform convergence gives
  $f(b)-f(a)=\int_a^b v(x)\,dx$. Since $v$ is continuous, the fundamental
  theorem of calculus implies $f'=v$. Applying this on compact subintervals
  proves $f\in C^1(I)$.

  Now telescope \cref{eq:quotient-step} starting from any admissible $t$.
  As the quotients converge to $f'(x)$, we obtain
  \begin{equation}
    |f'(x)-q_t^+(x)|
    \leq\frac{M}{2^{1+\alpha}}t^\alpha\sum_{m=0}^\infty2^{-m\alpha}
    =\frac{M}{2(2^\alpha-1)}t^\alpha.
  \end{equation}
  The same argument for backward quotients
  $q_t^-(y)=(f(y)-f(y-t))/t$ gives the identical bound. If $x<y$ and $t=y-x$,
  then
  \[
    q_t^+(x)=\frac{f(y)-f(x)}{y-x}=q_t^-(y).
  \]
  Adding the two bounds proves \cref{eq:holder-constant}.

<!-- proof-5: body_md -->

  independence of the half-space in the definition of $a_{\eta,\Omega_g^+}$, choose $H_x=\{y:y_{n+1}=g(x_0)\}$ and

<!-- lemma-5: body_md -->
\label{thm:difference-of-a-coeff}
  Let $g\colon\mathbb{R}^n\to\mathbb{R}$ satisfy a Lipschitz condition at $x_0$.
  For any $N\in\mathbb{N}$ large enough, we have
  \begin{align*}
    \left|  a_{\eta,\Omega_g^+}(\varphi_g(x_0),r) - a_{\rho,\Omega_g^+}(\varphi_g(x_0),r)  \right|
    \lesssim_N L^2 \sum_{j\ge 1} \mathfrak b_g(x_0,2^jr) 2^{-Nj}.
  \end{align*}
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
We first rewrite $a_\rho(x,r)$:
  Let $K_x^+ := \{ y\in\mathbb{R}^{n+1} : y_{n+1} - x_{n+1} > C_K L |y_0 - x_0| \}$ be the {\em upper opening} of the cone $K_x$.
  If we choose $H_x$ to be contained in the cone $x+K$, we have $K_x^+ \subset \Omega_g^+ \cap H_x^+$, and thus for $x\in \partial \Omega_g^+$ and $r>0$,
  \begin{align*}
    a_{\rho,\Omega_g^+}(x,r) &
    = \frac{1}{r^{n+1}} \bigg|  \left( \int_{\Omega_g^+} \rho\Big(\frac{y-x}{r}\Big) \,\mathrm{d} y - \!\int_{K_x^+} \rho\Big(\frac{y-x}{r}\Big) \,\mathrm{d} y \right) \\
                             & \qquad \quad
    - \left( \int_{H_x^+} \rho\Big(\frac{y-x}{r}\Big) \,\mathrm{d} y  -\! \int_{K_x^+} \rho\Big(\frac{y-x}{r}\Big) \,\mathrm{d} y \right)
    \bigg|                                                                                                                                                              \\
                             & = \frac{1}{r^{n+1}} \left|  \int_{\Omega_g^+\cap K_x} \rho(\frac{y-x}{r}) \,\mathrm{d} y
    - \int_{H_x^+\cap K_x} \rho(\frac{y-x}{r}) \,\mathrm{d} y
    \right|.
  \end{align*}
  We use this to calculate
  \begin{align*}
     & \left|  a_\eta(x,r) - a_\rho(x,r)  \right|                                                                          \\
     & \le \frac{1}{r^{n+1}} \bigg|
    \left( \int_{\Omega_g^+\cap K_x} \psi\bigg(\frac{|y_0-x_0|}{r}\bigg) \,\mathrm{d} y
    - \int_{H_x^+\cap K_x} \psi\bigg(\frac{|y_0-x_0|}{r}\bigg) \,\mathrm{d} y
    \right)                                                                                                                \\
     & \qquad\qquad\qquad -
    \left( \int_{\Omega_g^+\cap K_x} \psi\bigg(\frac{|y-x|}{r}\bigg) \,\mathrm{d} y
    - \int_{H_x^+\cap K_x} \psi\bigg(\frac{|y-x|}{r}\bigg) \,\mathrm{d} y
    \right)
    \bigg|                                                                                                                 \\
     & \le \frac{1}{r^{n+1}} \int_{(\Omega_g^+ \Delta H_x^+) \cap K_x}
    \left|  \psi\bigg(\frac{|y_0-x_0|}{r}\bigg) - \psi\bigg(\frac{|y-x|}{r}\bigg) \right| \,\mathrm{d}\mathcal{H}^{n+1}(y) \\
     & \le \frac{1}{r^{n+1}} \sum_{k\ge 0}
    \int_{A_{k,x,r}\cap (\Omega_g^+ \Delta H_x^+) \cap K_x} \lVert \psi' \rVert_{L^{\infty}((2^k - 1, c2^{k+1}))}
    \left|  \frac{|y_0-x_0| - |y-x| }{r}  \right|  \,\mathrm{d}\mathcal{H}^{n+1}(y)                                        \\
  \end{align*}
  where $A_{0,x,r} := B(x_0, 2r)\times \mathbb{R}$ and
  \begin{align*}
    A_{k,x,r} := \{ y \in \mathbb{R}^{n+1} : |y_0 - x_0| \in (2^kr, 2^{k+1}r) \} ,\qquad\forall k\ge 1.
  \end{align*}
  When applying the mean value theorem, note that $y\in A_{k,x,r}$ implies that $|y_0-x_0|/r \in (2^k - 1,2^{k+1})$.
  Since $y\in K_x$, we also have $|y-x|/r \lesssim |y_0-x_0|/r \lesssim 2^{k+1}$,
  and
  \begin{align*}
    \left|  |y_0-x_0| - |y-x|  \right| \le |1-\cos(\arctan C_K L)| |y-x| \lesssim 2^k L^2 r.
  \end{align*}
  Since $\psi$ is a Schwartz function, we have $\lVert \psi' \rVert_{L^{\infty}((2^k - 1, c2^{k+1}))} \le c(M) 2^{-Mk}$ for any $M\in \mathbb{N}$.

  We use affine approximating functions on consecutive dyadic scales to estimate the Hausdorff measure.
  For $j\ge 1$, let $\ell_j$ denote the affine function minimizing $\mathfrak b_g(x_0, 2^jr)$.
  Let $\ell_0 := \ell_1 - \ell_1(x_0) +g(x_0)$ and $H_x$ be the hyperplane associated to it.
  The Lipschitz condition of $g$ at $x_0$ implies that $|\nabla \ell_1| \le 2L$, and indeed $H_x$ is contained in the cone $x+K$:
  \begin{align*}
    2r|\nabla \ell_1|
    \le \lVert \ell_1 - g(x_0) \rVert_{L^{\infty}(B(x_0,2r))}
    \le \lVert \ell_1 - g \rVert_{L^{\infty}(B(x_0,2r))} + \lVert g - g(x_0) \rVert_{L^{\infty}(B(x_0,2r))}
    \le 4Lr.
  \end{align*}
  We have $\lVert \ell_1 - \ell_0 \rVert_{L^{\infty}(B(x_0,2r))} = |\ell_1(x_0) - g(x_0)| \le 2r \mathfrak b_g(x_0,2r)$.
  For $k+1\ge j\ge 2$,
  \begin{align*}
    \lVert \ell_j - \ell_{j-1} \rVert_{L^{\infty}(B(x_0,2^{k+1}r))}
     & \le \frac{2^{k+1}r}{2^{j-1}r}\lVert \ell_j - \ell_{j-1} \rVert_{L^{\infty}(B(x_0,2^{j-1}r))} \\
     & \le 2^{k+1}r ( \mathfrak b_g(x_0,2^{j-1}r) + 2 \mathfrak b_g(x_0,2^{j}r)).
  \end{align*}
  We have
  \begin{align*}
    \mathcal{H}^{n+1}(A_{k,x,r}\cap (\Omega_g^+ \Delta H_x^+))
     & \le \int_{B(x_0, 2^{k+1}r)} |g - \ell_0| \,\mathrm{d}\mathcal{H}^{n}                                                       \\
     & \lesssim (2^{k+1}r)^n \Big\| g - \ell_{k+1} + \sum_{j=1}^{k+1}  (\ell_j - \ell_{j-1}) \Big\|_{L^{\infty}(B(x_0,2^{k+1}r))} \\
     & \lesssim (2^{k+1}r)^{n+1} \sum_{j=1}^{k+1} \mathfrak b_g(x_0,2^jr).
  \end{align*}

  Combining all estimates yields

  \end{align}

  \begin{align*}

  $$I_1\lesssim \frac{1}{r^{n+1}} \int_0^{\ell_1} \Bigl|  \psi\Bigl(\frac{t}{r}\Bigr) \Bigr| t^{n-1}\epsilon\ell_1\,\mathrm{d} t
    \lesssim
    \frac{\epsilon}{r^{n+1}} \int_0^{\ell_1}  t^{n}\,\mathrm{d} t\lesssim

    \frac{\epsilon\,r}{\ell(Q_0)}.$$
