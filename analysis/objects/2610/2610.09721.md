<!-- theorem-1: body_md -->
\label{thm:main}
Assume that \(a(y)\) and $\bf A$ are given by \eqref{eq:A-form} and satisfy (A1) and (A2). Let \(d\ge 3\), \(\tau>0\), and
\(\phi(r,y) = \frac14 \bigl(\log r\bigr)^2\). Then there exists a constant \(C\),
depending on \(d\), \(\mathbf{A}\), \(a\) and \(\phi\), but not on \(\tau\) or \(u\),
such that for all \(u\in C^\infty_c(X)\) and all sufficiently large $\tau$,
\begin{equation}\label{est:main}
\| e^{\tau \phi} u \|_{L^{\frac{2d}{d-2}}(X)}
\le C \| e^{\tau \phi} H_{A,V} u \|_{L^{\frac{2d}{d+2}}(X)}.
\end{equation}
<!-- end theorem-1 -->
<!-- definition-1: body_md -->
We say that the unique continuation property across $\Sigma$ holds for the operator $P(x, D)$ if for each $x_0 \in \Sigma$ there exists a neighborhood $V$ of $x_0$ such that the following holds: Let $u$ be a solution to $P(x, D)u = 0$ in $V$ such that $u = 0$ in $\Sigma^+ \cap V$. Then $u = 0$ near $x_0$.
<!-- end definition-1 -->
<!-- theorem-2: body_md -->
Let $D_{a,b} = \{a<|x|<b\}$, and let $f\in W^{2,p}_{\mathrm{loc}}(\mathbb R^d)$ solve $H_{A,V}f=0$ in $\mathbb R^d$ and satisfy assumptions \ref{assump:A1}--\ref{assump:A2}. If $f=0$ on the shell $a<|x|<b$, then $f=0$.
 \label{prob:5.2}
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
Let $f\in W^{2,p}_{\mathrm{loc}}(\mathbb R^d)$ be a solution to $H_{A,V} f=0$ in $\mathbb R^d$, and suppose that the assumptions \ref{assump:A1}--\ref{assump:A2} hold.
Assume that $f$ satisfies the following vanishing condition at the origin:
\[
\int_{|x|<R}\big(|f|^p+|\nabla_A f|^p\big)\mathrm dx
= O\big(e^{-N(\log R)^2}\big)\qquad \text{for all } N>0.
\]where $\nabla_A:=i\nabla+A$.
Then $f=0$ almost everywhere in $\mathbb R^d$.
\label{theorem1.3}

<!-- lemma-1: body_md -->
\label{lem:kernel}
Let \(\beta \ge 1\), \(m > 0\), \(h_\beta(t)=\beta t^2/2\), and
\[
Q_{\beta,m}(t)=e^{h_\beta(t)}\bigl(-\partial_t^2 + m^2\bigr)e^{-h_\beta(t)}.
\]
For every \(w\in C_c^\infty(\mathbb R)\) there exists a measurable kernel
\(K_{\beta,m}(t,s)\) such that
\begin{equation}\label{eq:5.8}
w(t)=\int_{\mathbb R} K_{\beta,m}(t,s)\, \big[ Q_{\beta,m} w\big](s)\,\mathrm{d}s.
\end{equation}
Let \(\delta=|t-s|\) and \(u=(t+s)/2\). Then the kernel satisfies that
\begin{equation}\label{eq:5.9}
|K_{\beta,m}(t,s)| \le \frac{C}{\sqrt{\beta}+m} e^{-c\beta \delta^2},\qquad 0< m\le \sqrt{\beta},
\end{equation}
and, for \(m\ge \sqrt{\beta}\),
\begin{equation}\label{eq:5.10}
|K_{\beta,m}(t,s)| \le \frac{C}{\sqrt{\beta}+m} e^{-c\beta \delta^2}
\Bigl(e^{-c|m-\beta u|\delta}+e^{-c|m+\beta u|\delta}\Bigr),
\end{equation}
where the constants \(C,c>0\) are independent of \(\beta,m\).
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{lem:bandsum}
Let \(\gamma=2/d\), \(a_0=1-\gamma\), and \(\beta\ge 1\). Uniformly in \(M\ge 0\)
and \(\delta>0\),

\begin{equation}
e^{-c\beta\delta^2}
\left[
\beta^{-1/2}\sum_{0\le j\le 2\sqrt{\beta}}(1+j)^{a_0}
+\sum_{j>2\sqrt{\beta}} j^{-\gamma} e^{-c|j-M|\delta}
\right]
\le C\delta^{-a_0}.
\label{eq:sum}
\end{equation}

One may verify directly that
\[
K_0(t,s)=\frac{e^{h_\beta(t)-h_\beta(s)-m|t-s|}}{2m}
\]
satisfies
\[
w(t)=\int_{\mathbb R} K_0(t,s)\,Q_{\beta,m} w(s)\,\mathrm{d}s.
\]

Indeed,
\begin{align*}
&\int_{\mathbb R} K_0(t,s)\,Q_{\beta,m} w(s)\,\mathrm{d}s\\
&=\frac{e^{h_\beta(t)-mt}}{2m}\int_{-\infty}^t e^{ms}\bigl(-\partial_s^2+m^2\bigr)f(s)\,\mathrm{d}s
+\frac{e^{h_\beta(t)+mt}}{2m}\int_{t}^{+\infty} e^{-ms}\bigl(-\partial_s^2+m^2\bigr)f(s)\,\mathrm{d}s\\
&=e^{h_\beta(t)}f(t)=w(t),
\end{align*}
where
\[
f(s)=e^{-h_\beta(s)}w(s).
\]

On the other hand, it is not difficult to see that
\[
\int_{\mathbb R} e^{\pm ms}\bigl(-\partial_s^2+m^2\bigr)f(s)\,\mathrm{d}s = 0,
\]
that is,
\[
\int_{\mathbb R} e^{-h_\beta(s)\pm ms}\, Q_{\beta,m} w(s)\,\mathrm{d}s = 0.
\]

Thus we obtain
\[
K_R(t,s)=K_0(t,s)-\frac{1}{2m}e^{h_\beta(t)-mt-h_\beta(s)+ms},
\]
\[
K_L(t,s)=K_0(t,s)-\frac{1}{2m}e^{h_\beta(t)+mt-h_\beta(s)-ms},
\]
both of which are left inverses for \(Q_{\beta,m}\).
More precisely,
\[
K_R(t,s)=
\begin{cases}
0, & s<t,\\[4pt]
-m^{-1}e^{h_\beta(t)-h_\beta(s)}\sinh\bigl(m(s-t)\bigr), & s>t,
\end{cases}
\]
\[
K_L(t,s)=
\begin{cases}
m^{-1}e^{h_\beta(t)-h_\beta(s)}\sinh\bigl(m(s-t)\bigr), & s<t,\\[4pt]
0, & s>t.
\end{cases}
\]

We now select a suitable left inverse kernel to establish estimates \eqref{eq:5.9}
and \eqref{eq:5.10}.

\medskip
\noindent\textbf{Case 1 $m\ge \sqrt{\beta}$.}
Set \(\delta=|t-s|\), \(u=\frac{t+s}{2}\). We first record a simple auxiliary
inequality: if \(A\ge \frac{\beta \delta}{2}\) and \(0<c_0<\frac13\), then
\begin{equation}
e^{-A\delta}\le e^{-c_0\beta\delta^2}e^{-c_0 A\delta},
\label{eq:basic_ineq}
\end{equation}
since \((1-c_0)A\ge c_0\beta\delta\).

We choose the left inverse according to three regimes for \(t\):
\begin{itemize}
\item (Regime A) When \(t<-\dfrac{m}{\beta}\), take \(K_L\) as the left inverse of \(Q_{\beta,m}\).
\item (Regime B) When \(|t|\le \dfrac{m}{\beta}\), take \(K_0\) as the left inverse of \(Q_{\beta,m}\).
\item (Regime C) When \(t>\dfrac{m}{\beta}\), take \(K_R\) as the left inverse of \(Q_{\beta,m}\).
\end{itemize}

\medskip
\noindent\textbf{(Regime B): Case $|t|\le \dfrac{m}{\beta}$.}

\smallskip
\noindent\textit{(Case B1): $t>s$.}
\begin{equation}
\begin{aligned}
h_\beta(t)-h_\beta(s)-m|t-s|
&=(\beta u-m)(t-s)\\
&=-(m-\beta u)\delta,
\end{aligned}
\label{eq:case_B1_tgreater_s}
\end{equation}
Note that
\begin{equation}
m-\beta u
=m-\beta t+\frac{\beta\delta}{2}
\ge \frac{\beta\delta}{2},
\label{eq:lower_m_beta_u_B1}
\end{equation}
Hence, applying \eqref{eq:basic_ineq},
\begin{equation}
\begin{aligned}
K_0=\frac{1}{2m}e^{-(m-\beta u)\delta}
&\le \frac{1}{2m}e^{-c_0(m-\beta u)\delta}e^{-c_0\beta\delta^2}\\
&=\frac{1}{2m}e^{-c_0|m-\beta u|\delta}e^{-c_0\beta\delta^2}.
\end{aligned}
\label{eq:K0_est_B1}
\end{equation}

\smallskip
\noindent\textit{(Case B2): $s>t$.}
\begin{equation}
\begin{aligned}
h_\beta(t)-h_\beta(s)-m|t-s|
&=-(\beta u+m)(s-t)\\
&=-(m+\beta u)\delta,
\end{aligned}
\label{eq:case_B2_sgreater_t}
\end{equation}
Observe that
\begin{equation}
m+\beta u
=m+\beta\frac{t+s}{2}
=m+\beta t+\frac{\beta\delta}{2}
\ge \frac{\beta\delta}{2},
\label{eq:lower_m_plus_beta_u_B2}
\end{equation}
Thus, by \eqref{eq:basic_ineq},
\begin{equation}
K_0=\frac{1}{2m}e^{-(m+\beta u)\delta}
\le \frac{1}{2m}e^{-c_0|m+\beta u|\delta}e^{-c_0\beta\delta^2},
\label{eq:K0_est_B2}
\end{equation}
Recall that \(m\ge\sqrt{\beta}\). Combining Cases B1 and B2,
\begin{equation}
|K_0|\le \frac{C}{m+\sqrt{\beta}}e^{-c_0\beta\delta^2}
\Bigl(e^{-c_0|m-\beta u|\delta}+e^{-c_0|m+\beta u|\delta}\Bigr).
\label{eq:K0_final_B}
\end{equation}

\medskip
\noindent\textbf{(Regime A): Case $t<-\dfrac{m}{\beta}$, $s<t$.}
We use the left-inverse kernel \(K_L\).
\begin{equation}
\begin{aligned}
K_L&=\frac{1}{2m}\Bigl(e^{h_\beta(t)-h_\beta(s)-mt+ms}-e^{h_\beta(t)-h_\beta(s)+mt-ms}\Bigr)\\
&=\frac{1}{2m}\Bigl(e^{\beta u(t-s)-m(t-s)}-e^{\beta u(t-s)+m(t-s)}\Bigr)\\
&=\frac{1}{2m}\Bigl(e^{-(m-\beta u)\delta}-e^{-(-m-\beta u)\delta}\Bigr)
=\frac1{2m}\Bigl(e^{-|m-\beta u|\delta}-e^{-|m+\beta u|\delta}\Bigr).
\end{aligned}
\label{eq:KL_def_A}
\end{equation}

Since \(s<t\), we have \(u=t-\frac{\delta}{2}<-\frac{m}{\beta}-\frac{\delta}{2}\), so
\(\beta u+m<-\frac{\beta\delta}{2}\). Hence \(-\beta u-m>\frac{\beta\delta}{2}>0\),
and \(m-\beta u>-\beta u-m>0\). Therefore \(m-\beta u=|m-\beta u|\),
\(-m-\beta u=|m+\beta u|\), and
\begin{equation}
|m-\beta u|=m-\beta u> -m-\beta u=|m+\beta u|\ge \frac{\beta\delta}{2}.
\label{eq:abs_estimate_A}
\end{equation}

Applying inequality \eqref{eq:basic_ineq} to both exponential terms,
\begin{equation}
|K_L|\lesssim \frac{C}{m+\sqrt{\beta}}e^{-c_0\beta\delta^2}
\Bigl(e^{-c_0|m+\beta u|\delta}+e^{-c_0|m-\beta u|\delta}\Bigr).
\label{eq:KL_est_A}
\end{equation}

\medskip
\noindent\textbf{(Regime C): Case $t>\dfrac{m}{\beta}$, $s>t$.}
We use the left-inverse kernel \(K_R\).

For \(s>t\), \(u=t+\frac{\delta}{2}>\frac{m}{\beta}+\frac{\delta}{2}\), so
\(\beta u-m>\frac{\beta\delta}{2}>0\).

We have
\begin{equation}
\frac{\beta\delta}{2}<|m-\beta u|=\beta u-m<m+\beta u=|m+\beta u|.
\label{eq:abs_estimate_C}
\end{equation}
Thus
\begin{equation}
\begin{aligned}
K_R&=\frac1{2m}\Bigl(e^{h_\beta(t)+mt-h_\beta(s)-ms}-e^{h_\beta(t)-mt-h_\beta(s)+ms}\Bigr)\\
&=\frac1{2m}\Bigl(e^{\beta u(t-s)+m(t-s)}-e^{\beta u(t-s)-m(t-s)}\Bigr)\\
&=\frac1{2m}\Bigl(e^{-(m+\beta u)(s-t)}-e^{-(\beta u-m)(s-t)}\Bigr)
=\frac1{2m}\Bigl(e^{-|m+\beta u|\delta}-e^{-|m-\beta u|\delta}\Bigr).
\end{aligned}
\label{eq:KR_def_C}
\end{equation}

Applying \eqref{eq:basic_ineq} once more,
\begin{equation}
|K_R|\lesssim \frac{C\,e^{-c_0\beta\delta^2}}{m+\sqrt{\beta}}
\Bigl(e^{-c_0|m+\beta u|\delta}+e^{-c_0|m-\beta u|\delta}\Bigr).
\label{eq:KR_est_C}
\end{equation}

\medskip
Collecting Regimes A, B and C, for \(m\ge\sqrt{\beta}\) we obtain a measurable
left-inverse kernel \(K_{\beta,m}\) satisfying
\begin{equation}
|K_{\beta,m}(t,s)|\le \frac{C}{\sqrt{\beta}+m}e^{-c\beta\delta^2}
\Bigl(e^{-c|m-\beta u|\delta}+e^{-c|m+\beta u|\delta}\Bigr),
\label{eq:final_kernel_est}
\end{equation}
which is exactly estimate \eqref{eq:5.10}.

\medskip
\noindent\textbf{Case 2 $0<m\le \sqrt{\beta}$.}

\smallskip
\noindent\textit{(Case 2.1): $t\ge 0$.}
We select \(K_R\), which vanishes for \(s<t\) and is non-zero only when \(s>t\).
Set
\[
x=\sqrt{\beta}\,t\ge 0,\quad \rho=\sqrt{\beta}(s-t)>0,\quad \nu=\frac{m}{\sqrt{\beta}}\in(0,1),
\]
We compute
\[
h_\beta(t)-h_\beta(s)=\beta\Bigl(\frac{t^2-s^2}{2}\Bigr)
=-x\rho-\frac{\rho^2}{2},
\]
Then
\begin{equation}
\begin{aligned}
|K_R|&= m^{-1} e^{h_\beta(t)-h_\beta(s)} \sinh\big(m(s-t)\big) \\
&=\beta^{-\frac12} e^{-x\rho-\frac{\rho^2}{2}} \beta^{\frac12}m^{-1}\sinh(\nu\rho) \\
&=\beta^{-\frac12} e^{-x\rho-\frac{\rho^2}{2}}\frac{\sinh(\nu\rho)}{\nu\rho}\ \rho,
\end{aligned}
\label{eq:KR_lowm_def}
\end{equation}
Recall the elementary inequality
\[
\frac{\sinh x}{x}\le e^{x},\quad \forall x>0,
\]
Hence
\[
|K_R|\le \beta^{-\frac12}\rho\, e^{-x\rho-\frac{\rho^2}{2}+\nu\rho},
\]
Observe the bound
\[
\nu\rho\le \frac{\rho^2}{4}+\nu^2 \le \frac{\rho^2}{4}+1,
\]
Substituting this estimate,
\begin{equation}
\begin{aligned}
|K_R|&\le \beta^{-\frac12}\rho\, e^{-x\rho-\frac{\rho^2}{4}+1}
\le C\,\beta^{-\frac12}\rho\, e^{-\frac{\rho^2}{4}}
\le C\,\beta^{-\frac12} e^{-\frac{\rho^2}{8}},
\end{aligned}
\label{eq:KR_lowm_est}
\end{equation}
Since \(\rho=\sqrt{\beta}\delta\), we have \(e^{-\rho^2/8}=e^{-\beta\delta^2/8}\). Therefore
\begin{equation}
|K_R|\le\frac{C}{m+\sqrt{\beta}}\,e^{-c\beta\delta^2}.
\label{eq:KR_lowm_final}
\end{equation}

\smallskip
\noindent\textit{(Case 2.2): $t< 0$.}
We take \(K_L\), which vanishes for \(s>t\) and is non-zero only for \(s<t\).
Set
\[
x=-\sqrt{\beta}\,t\ge 0,\quad \rho=\sqrt{\beta}(t-s)>0,\quad \nu=\frac{m}{\sqrt{\beta}}\in(0,1).
\]

Using the explicit form of \(K_L\),
\begin{equation}
\begin{aligned}
|K_L|&=\Bigl| m^{-1} e^{h_\beta(t)-h_\beta(s)} \sinh\big(m(t-s)\big) \Bigr| \\
&=\Bigl| m^{-1} e^{-x\rho-\frac{\rho^2}{2}} \sinh(\nu\rho) \Bigr|
=\Bigl| m^{-1} e^{-x\rho-\frac{\rho^2}{2}}  \nu\rho \frac{\sinh(\nu\rho)}{\nu\rho}\Bigr| \\
&=\Bigl| \beta^{-1/2} \rho\, e^{-x\rho-\frac{\rho^2}{2}}  \frac{\sinh(\nu\rho)}{\nu\rho}\Bigr|
\le \beta^{-1/2}\rho\, e^{-x\rho-\frac{\rho^2}{2}+\nu\rho}\\
&\le \beta^{-1/2}\rho\, e^{-x\rho-\frac{\rho^2}{4}+1}
\le C\beta^{-1/2} e^{-\rho^2/8}
\lesssim \frac{C}{m+\sqrt{\beta}}\, e^{-c\beta\delta^2}.
\end{aligned}
\label{eq:KL_lowm_est}
\end{equation}

Combining Case 2.1 and Case 2.2, we obtain the kernel bound \eqref{eq:5.9} for
\(0<m\le\sqrt{\beta}\). This completes the proof of Lemma \ref{lem:kernel}.

\end{align*}

\sum_{j>\frac4\delta} j^{-\gamma}e^{-c|j-M|\delta}
\le \sum_{j>\frac4\delta}\Bigl(\frac4\delta\Bigr)^{-\gamma} e^{-c\frac12 j\delta}
\lesssim \delta^{\gamma}\sum_{j>\frac4\delta}e^{-\frac c2 j\delta}
\lesssim \delta^{\gamma}\cdot \delta^{-1}=\delta^{-a_0}.
\]

Multiplying by \(e^{-c\beta\delta^2}\le 1\), we get
\[
e^{-c\beta\delta^2}\sum_{j>2\sqrt{\beta}} j^{-\gamma}e^{-c|j-M|\delta}\le C\delta^{-a_0}.
\]

\paragraph{(ii) $M>\dfrac{2}{\delta}$.}
We partition the summation range \(j>2\sqrt{\beta}\) into three parts:
\[
\sum_{j>2\sqrt{\beta}}=\sum_{\substack{j>2\sqrt{\beta}\\j<\frac M2}}
+\sum_{\substack{j>2\sqrt{\beta}\\|j-M|\le \frac M2}}
+\sum_{\substack{j>2\sqrt{\beta}\\j>\frac32 M}}.
\]

 Part 1: \(2\sqrt{\beta}<j<\frac M2\). Here \(|j-M|\ge \frac M2\), so
\begin{align*}
e^{-c\beta\delta^2}\sum_{\substack{j>2\sqrt{\beta}\\j<\frac M2}} j^{-\gamma}e^{-c|j-M|\delta}
&\lesssim e^{-c\beta\delta^2}e^{-c\frac M2\delta}\sum_{\substack{j>2\sqrt{\beta}\\j<\frac M2}} j^{-\gamma}\\
&\lesssim e^{-c\beta\delta^2}e^{-c\frac M2\delta} M^{1-\gamma}\\
&\lesssim e^{-c\beta\delta^2}\,\delta M\, e^{-\frac c2 M\delta} M^{-\gamma}\delta^{-1}\\
&\lesssim \delta^{-a_0}.
\end{align*}

 Part 2: \(|j-M|\le \frac M2\). In this region \(j\simeq M\), so \(j^{-\gamma}\lesssim M^{-\gamma}\):
\begin{align*}
e^{-c\beta\delta^2}\sum_{\substack{j>2\sqrt{\beta}\\|j-M|\le \frac M2}} j^{-\gamma}e^{-c|j-M|\delta}
&\lesssim e^{-c\beta\delta^2}M^{-\gamma}
\biggl(\sum_{\frac M2\le j\le M}e^{-c(M-j)\delta}
+\sum_{M<j\le \frac32 M}e^{-c(j-M)\delta}\biggr)\\
&\lesssim e^{-c\beta\delta^2}M^{-\gamma} (1+\delta)^{-1},
\end{align*}
since \(M>\frac2\delta\), \(M^{-\gamma}\lesssim \delta^{\gamma}\), hence
\[
e^{-c\beta\delta^2}M^{-\gamma}(1+\delta)^{-1}
\lesssim e^{-c\beta\delta^2}\delta^{\gamma}(1+\delta)^{-1}
\le C\delta^{-a_0}.
\]

 Part 3: \(j>\frac32 M\). Then \(|j-M|\ge \frac j3\), so
\begin{align*}
e^{-c\beta\delta^2}\sum_{\substack{j>2\sqrt{\beta}\\j>\frac32 M}} j^{-\gamma}e^{-c|j-M|\delta}
&\lesssim e^{-c\beta\delta^2}\sum_{j>\frac32 M} j^{-\gamma}e^{-c\frac13 j\delta}\\
&\lesssim e^{-c\beta\delta^2} M^{-\gamma}\sum_{j>\frac32 M} e^{-\frac c3 j\delta}\\
&\lesssim e^{-c\beta\delta^2} M^{-\gamma}\cdot \delta^{-1},
\end{align*}
again \(M>\frac2\delta\) yields \(M^{-\gamma}\lesssim \delta^{\gamma}\), so
\[
e^{-c\beta\delta^2} M^{-\gamma}\delta^{-1}
\lesssim e^{-c\beta\delta^2}\delta^{\gamma}\delta^{-1}
\le C\delta^{-a_0}.
\]

Combining the three sub-parts for \(M>\frac2\delta\), we obtain
\[
e^{-c\beta\delta^2}\sum_{j>2\sqrt{\beta}} j^{-\gamma}e^{-c|j-M|\delta}\le C\delta^{-a_0}.
\]

\medskip

We point out that when estimating sums by integrals, the two quantities may differ
by finitely many boundary terms. These boundary terms are uniformly bounded
constants. After multiplying by the factor \(e^{-c\beta\delta^2}\), they are
trivially dominated by \(C\delta^{-a_0}\).

Putting the estimates for the first term and the second term together, inequality
\eqref{eq:sum} holds uniformly in \(\beta\ge 1\), \(M\ge 0\), \(\delta>0\).

\label{lemma5.3}
For every $R>0$ the estimate
\begin{equation}\label{eq:Carleman-R}
\bigl\|e^{\tau \phi_R}u\bigr\|_{L^q(\mathbb R^d)}
\le C\bigl\|e^{\tau \phi_R}H_{A,V}u\bigr\|_{L^p(\mathbb R^d)},
\qquad u\in C_c^\infty(\mathbb R^d\setminus\{0\}),
\end{equation}
holds for all $\tau\ge \tau_0$ with the same constant $C$, where
\begin{equation}\label{eq:phi-R}
\phi_R(x)=\frac14\bigl(\log(|x|/R)\bigr)^2.
\end{equation}
\begin{proof}

The proof of this lemma is straightforward and follows immediately from a scaling transformation applied to equation \eqref{eq:E}.

For $g(y)=f(Ry)$ the homogeneity of \eqref{eq:A-form} gives
$A(Ry)=R^{-1}A(y)$ and $V(Ry)=R^{-2}V(y)$, hence
\begin{equation}\label{eq:homog}
  (H_{A,V}g)(y)=R^{2}(H_{A,V}f)(Ry) .
\end{equation}
Applying \eqref{eq:E} to $g$ and changing variables $x=Ry$ yields
\[
  R^{-d/q}\left\lVert e^{\tau\phi_R}f\right\rVert_{L^{q}_x}
  \le C R^{2-d/p}\left\lVert e^{\tau\phi_R}H_{A,V}f\right\rVert_{L^{p}_x} .
\]
By \eqref{def:qp-P},
$d/q=(d-2)/2$ and $2-d/p=2-(d+2)/2=-(d-2)/2=-d/q$, so the powers of $R$ on
the two sides coincide and cancel.  This is \eqref{eq:Carleman-R}.  The
cancellation is precisely the reason the exponents in \eqref{def:qp-P} are
the scale-invariant ones.
\end{proof}

\bigl\|e^{\tau\phi} u_j\bigr\|_{L^q(\mathbb R^d)}
\le \bigl\|e^{\tau\phi}\bigl[H_{A,V},\rho(jx)\bigr]u\bigr\|_{L^p(\mathbb R^d)}
+\bigl\|e^{\tau\phi}\rho(jx)\bigl[H_{A,V},\chi\bigr]f\bigr\|_{L^p(\mathbb R^d)}.
\]

\[
\bigl\|e^{\tau\phi}\rho(jx)\bigl[H_{A,V},\chi\bigr]f\bigr\|_{L^p(\mathbb R^d)}
=\bigl\|e^{\tau\phi}\bigl[H_{A,V},\chi\bigr]f\bigr\|_{L^p(R_0<|x|<2R_0)}.

\[
\begin{aligned}
\bigl\|e^{\tau\phi}\rho(jx)f\bigr\|_{L^q(B_{R_1})}
&\le \bigl\|e^{\tau\phi}\bigl[H_{A,V},\rho(jx)\bigr]f\bigr\|_{L^p\left(\frac1{2j}<|x|<\frac1j\right)}\\
&\quad +\bigl\|e^{\tau\phi}\bigl[H_{A,V},\chi\bigr]f\bigr\|_{L^p(R_0<|x|<2R_0)}.
\end{aligned}

\[
\bigl[H_{A,V},\rho(jx)\bigr]f
=2i j(\nabla\rho)(jx)\cdot\nabla_A f -j^2(\Delta\rho)(jx)f.

Thus
\[
\begin{aligned}

&\lesssim j\, e^{\frac{\tau}{4}(\log 2j)^2}\bigl\|\nabla_A f\bigr\|_{L^p(|x|<\frac1j)}
+j^2 e^{\frac{\tau}{4}(\log2j)^2}\|f\|_{L^p(|x|<\frac1j)}\\
&\lesssim j^2 e^{\frac{\tau}{4}(\log 2j)^2} e^{-\frac{N}{p}(\log j)^2}.
\end{aligned}

\[

\]
This yields
\[
\bigl\|e^{\tau\phi}\rho(jx)f\bigr\|_{L^q(B_{R_1})}
\lesssim j^2 e^{\frac{\tau}{4}(\log 2j)^2}e^{-\frac{N}{p}(\log j)^2}
+\bigl\|e^{\tau\phi}\bigl[H_{A,V},\chi\bigr]f\bigr\|_{L^p(R_0<|x|<2R_0)}.
\]

Set $\varphi_k=\bigl|e^{\tau\phi}\rho(kx)f\bigr|^q$. We have $\varphi_k\to |e^{\tau\phi}f|^q$ almost everywhere in $B_{R_1}$ as $k\to\infty$.
Applying Fatou's lemma,
\[
\begin{aligned}
\bigl\|e^{\tau\phi}f\bigr\|_{L^q(B_{R_1})}
&=\left(\int_{B_{R_1}} \bigl|e^{\tau\phi}f\bigr|^q dx\right)^{\frac1q}
=\left(\int_{B_{R_1}} \lim_{j\to\infty}\!\inf_{k\ge j}\varphi_k\,dx\right)^{\frac1q}\\
&\le \lim_{j\to\infty}\!\inf_{k\ge j}\left(\int_{B_{R_1}}\varphi_k\,dx\right)^{\frac1q}
=\lim_{j\to\infty}\!\inf_{k\ge j}
\bigl\|e^{\tau\phi}\rho(kx)f\bigr\|_{L^q(B_{R_1})}\\
&\le\lim_{j\to\infty}\!\inf_{k\ge j}
k^2 e^{\frac{\tau}{4}(\log 2k)^2}e^{-\frac{N}{p}(\log k)^2}
+\bigl\|e^{\tau\phi}\bigl[H_{A,V},\chi\bigr]f\bigr\|_{L^p(R_0<|x|<2R_0)}
\\
&=\bigl\|e^{\tau\phi}\bigl[H_{A,V},\chi\bigr]f\bigr\|_{L^p(R_0<|x|<2R_0)}.
\end{aligned}

It is straightforward to see that $\bigl[H_{A,V},\chi\bigr]f\in L^p(R_0<|x|<2R_0)$. Letting $\tau\to\infty$, we get $f=0$ a.e.\ in $B_{R_1}$.
 Applying Theorem \ref{prob:5.2}, we conclude $f=0$ a.e.\ in $\mathbb R^d$.
