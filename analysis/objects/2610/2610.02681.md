<!-- lemma-2: body_md -->
\label{lem:rounding}
Let $M$ be a symmetric $n\times n$ matrix with zero diagonal and
$|M_{ij}|\le1$, and let $\beta=\max_{\sigma\in\{-1,1\}^n}\sigma^\top M\sigma$.
For every $X\succeq0$ with $X_{ii}\le1$ for $i=1,\dots,n$,
$\langle M,X\rangle-\lVert X\rVert_F^2\le\frac\pi2\beta$.
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
Let $\widehat{X}=X+\operatorname{diag}(1-X_{11},\dots,1-X_{nn})$. Then $\widehat{X}\succeq 0$ and $\widehat{X}_{ii}=1$ for every $i$. Hence there exists a random sign vector $\sigma\in\{-1,1\}^n$ such that
$\mathbb{E}(\sigma_i\sigma_j)=\frac{2}{\pi}\arcsin X_{ij}$ for $i\ne j$.
Since $M$ has zero diagonal, $|M_{ij}|\le 1$ and $|\arcsin u-u|\le u^2$ for $|u|\le 1$, we have
\begin{align*}
\beta&\ge\mathbb{E}(\sigma^\top M\sigma)=\frac{2}{\pi}\sum_{i\ne j}M_{ij}\arcsin X_{ij}\\
&\ge\frac{2}{\pi}\left(\langle M,X\rangle-\sum_{i\ne j}X_{ij}^2\right)\ge\frac{2}{\pi}\left(\langle M,X\rangle-\|X\|_F^2\right).
\end{align*}
This proves the claimed inequality.

<!-- lemma-3: body_md -->
\label{lem:positivepart}
For every symmetric matrix $M$, let $g(M)=\frac14\lVert M_+\rVert_F^2$. Then 
$g(M)=\max_{X\succeq0}\bigl(\langle M,X\rangle-\lVert X\rVert_F^2\bigr)$,
and the maximizer is uniquely $X=\frac{1}{2}M_+$.
The function $g$ is continuously differentiable, with
$\nabla g(M)=\frac{1}{2}M_+$. In particular, for every symmetric matrix $E$, the directional derivative of $g$ at $M$ in direction $E$ is $\frac{1}{2}\langle M_+,E\rangle$, even when $M$ has zero eigenvalues.

\label{lem:regular}
Let $G$ be a noncomplete $d$-regular graph of order $n\ge2$.
Then $\lambda_2(G)\ge0$,
$\lambda_n(\overline G)=-1-\lambda_2(G)$, and
$\operatorname{disc}^+(G)\le\frac n8\left(\lambda_2(G)+\frac{d}{n-1}\right)$.

<!-- lemma-5: body_md -->
\label{lem:obstruction}
Let $0<\delta<1/16$ and $s\ge0$. Suppose a graph $G$ satisfies $\lambda_n(G)\ge-s$. There cannot be three disjoint vertex sets $X,Y,Z$ with $|X|=|Y|=|Z|>6s$, such that $(X,Y)$ and $(Y,Z)$ are $\delta$-full and $(X,Z)$ is $\delta$-empty.
<!-- end lemma-5 -->
<!-- proof-2: body_md -->
Let $t=|X|=|Y|=|Z|$.
Suppose these sets exist. 
We have $e(X,Y)\ge (1-\delta)t^2$, $e(Y,Z)\ge (1-\delta)t^2$ and $e(X,Z)\le \delta t^2$. 
Let $\mathbf{x}$ be a vector with \[
x_u=\begin{cases}
1&\mbox{ if } u\in X,\\
-2&\mbox{ if }u\in Y,\\
1&\mbox{ if }u\in Z,\\
0&\mbox{ otherwise.}
\end{cases}
\] 
Then, as for $W\in \{X,Y,Z\}$, $e(W)\le {t\choose 2}\le \frac{1}{2} t^2$, we have
\begin{align*}
\mathbf{x}^\top A(G)\mathbf{x}&=2e(X)+8e(Y)+2e(Z)-4e(X,Y)-4e(Y,Z)+2e(X,Z)\\
&\le t^2+4t^2+t^2-4(1-\delta)t^2-4(1-\delta)t^2+2\delta t^2 \\
&=(-2+10\delta)t^2\\
&<-t^2.
\end{align*}
Note that $\mathbf{x}^\top \mathbf{x}=6t$.
So by Rayleigh's principle, \[
\lambda_n\le \frac{\mathbf{x}^\top A(G)\mathbf{x}}{\mathbf{x}^\top \mathbf{x}}<\frac{-t^2}{6t}=-\frac{t}{6}<-s,
\]
a contradiction.

<!-- lemma-6: body_md -->
\label{lem:editing}
For a graph $G$ of order $n$, $\operatorname{ed}_c(G)\le 3\sqrt{np_3(G)}$.
<!-- end lemma-6 -->
<!-- proof-3: body_md -->
If $p_3(G)=0$, then $G$ is already a union of cliques, so $\operatorname{ed}_c(G)=0\le 3\sqrt{np_3(G)}$. 
Suppose that $p_3(G)\ge 1$ and let $b=\sqrt{\frac{p_3(G)}{n^3}}$. 
Let $G_1$ be the subgraph of $G$ induced by the vertices not belonging to any isolated clique.
Then $p_3(G_1)= p_3(G)$. 
For $v\in V(G_1)$, let $p_{3}^{(1)}(v)$ denote the number of sets inducing a $P_3$ in $G_1$ and containing $v$. 

If $e(G_1) \le bn^2$, then deleting all edges of $G_1$ yields a union
of cliques, and the number of edges deleted is at most $bn^2 < 3\sqrt{np_3(G)}$, as desired.

Otherwise, we proceed iteratively. Suppose at some step we have a graph
$G_i$ with $e(G_i) > bn^2$. For $v\in V(G_i)$, let $p_{3}^{(i)}(v)$ denote the number of sets inducing a $P_3$ in $G_i$ and containing $v$. 
Then $\sum_{v \in V(G_i)} p_3^{(i)}(v) = 3p_3(G_i) \le 3p_3(G)$
and $\sum_{v \in V(G_i)} \bigl(d_{G_i}(v)+1\bigr) = 2e(G_i) + |V(G_i)| > 2bn^2$.
Averaging the quantities $\frac{p_3^{(i)}(v)}{d_{G_i}(v)+1}$ over all vertices $v$, we obtain $\frac{\sum_{v \in V(G_i)} p_3^{(i)}(v)}
     {\sum_{v \in V(G_i)} \bigl(d_{G_i}(v)+1\bigr)}$.
Hence there exists a vertex $u_i \in V(G_i)$ such that
\[
\frac{p_3^{(i)}(u_i)}{d_{G_i}(u_i)+1}
\le \frac{\sum_{v \in V(G_i)} p_3^{(i)}(v)}
        {\sum_{v \in V(G_i)} \bigl(d_{G_i}(v)+1\bigr)}
\le \frac{3p_3(G)}{2bn^2} = \frac{3}{2}bn.
\]
Let $C_i = N_{G_i}[u_i]$. We add every missing edge inside $C_i$ and delete every edge between $C_i$ and $V(G_i)\setminus C_i$. The total number of edge additions and deletions is exactly $p_3^{(i)}(u_i)$. We then let $G_{i+1} = G_i - C_i$. One has $p_3(G_{i+1}) \le p_3(G_i)\le p_3(G)$.

This process terminates at some graph $G_t$ with $e(G_t) \le bn^2$.
Deleting all edges of $G_t$ yields a union of cliques. Thus, the total number of edges deleted or added is at most \[
\sum_{i=1}^{t-1} p_3^{(i)}(u_i) + e(G_t)
\le \frac{3}{2}bn \sum_{i=1}^{t-1} \bigl(d_{G_i}(u_i)+1\bigr) + bn^2
\le \frac{5}{2}bn^2\le 3\sqrt{np_3(G)}.
\]
So $\operatorname{ed}_c(G)\le 3\sqrt{np_3(G)}$.

<!-- lemma-7: body_md -->
\label{lem:partition}
Let $r\ge1$ be an integer and $0<\delta<\frac{1}{16}$. There is an integer
$K_{r,\delta}$ such that if $n\ge K_{r,\delta}$ and $A(G)$ has an approximation
$B\succeq0$ satisfying $\operatorname{rk} B\le r$, $\lVert B\rVert_{\mathrm{op}}\le2n$ and $\lVert A(G)-B\rVert_F^2\le\frac{\delta^2}{100}n^2$, 
then there is a partition
$V(G)=V_0\cup V_1\cup\cdots\cup V_K$ such that
$|V_0|\le\delta n$, $|V_1|=\cdots=|V_K|=m$, $\frac1\delta\le K\le K_{r,\delta}$ and $m\ge\frac{n}{K_{r,\delta}}$.
The set $V_0$ may be empty. 
Among the unordered pairs $\{V_i,V_j\}$ with $1\le i<j\le K$, at most $\delta K^2$ are $\delta$-exceptional.

<!-- thm-3: body_md -->
\label{thm:structure}
For every $\eta\in(0,1)$, there is an integer $N_\eta\ge2$,                           
such that for every $n$-vertex graph $G$ with $n\ge N_\eta$, if there exists a real
symmetric matrix $R$ such that $R\succeq0$, $R_{ii}\le2$ for $i=1,\dots,n$, $\lVert R\rVert_F\le n^{2/3}$ and $\lambda_n(A(G)+R)\ge-\frac{\eta^4}{4\cdot10^9}n^{1/4}$, then $\operatorname{ed}_c(G)\le \eta n^2$.
<!-- end thm-3 -->
<!-- proof-4: body_md -->
Let $\delta=\frac{\eta^2}{100}$, $\xi=\frac{\delta^2}{100}$ and $\tau=\frac{\xi}{4}$.
Then $0<\delta<\frac{1}{16}$ and $\tau\in(0,1)$. Let $\kappa=(\frac{\tau}{1024})^4$, $r=\lceil4\kappa^{-2}\rceil$ and 
$a=\frac{\tau}{1000}$.
Let $N^*(\tau)$ be an integer from Proposition~\ref{prop:lowrank}, and let $K_{r,\delta}$ be an integer from Lemma~\ref{lem:partition}. Let
\begin{equation}\label{eq:structure-threshold}
 N_\eta=\max\left\{
  N^*(\tau),\  (13K_{r,\delta})^3,\ \left\lceil\frac{8}{\xi^{3/2}}\right\rceil \right\}.
\end{equation}
                                                                                                                

Let $R$ be a matrix as in the statement of the theorem and $C=A(G)+R$. Then $\lambda_n(C)\ge -an^{1/4}$.
Proposition~\ref{prop:lowrank} gives a matrix $B\succeq0$ of rank at
most $r$, with $\lVert B\rVert_{\mathrm{op}}\le2n$ and $\lVert C-B\rVert_F^2\le\tau n^2$. 
                                                             
Then
\[
 \lVert A(G)-B\rVert_F^2\le2\lVert C-B\rVert_F^2+2\lVert R\rVert_F^2\le2\tau n^2+2n^{4/3}\le\xi n^2=\frac{\delta^2}{100}n^2.
\]
                                                                                                                    
Thus, all hypotheses of Lemma~\ref{lem:partition} hold. Let $V(G)=V_0\cup V_1\cup \dots \cup V_K$ be a partition with $|V_i|=m$ for $i=1,\dots,K$, where $m\ge \frac{n}{K_{r,\delta}}$ and $\frac{1}{\delta}\le K\le K_{r,\delta}$.

Let $s=2n^{2/3}$.
By Lemma~\ref{weyl}, 
\[
 \lambda_n(G)\ge\lambda_n(C)-\lVert R\rVert_{\mathrm{op}}
 \ge-a n^{1/4}-n^{2/3}\ge-2n^{2/3}=-s.
\]
By~\eqref{eq:structure-threshold}, $m\ge \frac{n}{K_{r,\delta}}\ge 13n^{2/3}>6s$.
Thus, Lemma~\ref{lem:obstruction} shows that $G$ does not contain three vertex subsets whose successive pairs are $\delta$-full and whose remaining pair is $\delta$-empty.

Let $Q$ be the graph on vertex set $\{v_1,\dots,v_K\}$ in which $v_iv_j\in E(Q)$ if and only if $(V_i,V_j)$ is $\delta$-full.
Suppose that $v_iv_jv_k$ is an induced $P_3$ of $Q$. Then as $v_iv_k\notin E(Q)$, $(V_i,V_k)$ is not $\delta$-full. Also, $(V_i,V_k)$ is not $\delta$-empty.  It must be $\delta$-exceptional. 
Each $\delta$-exceptional unordered pair can be the pair of endpoints of at most $K-2$ induced $P_3$. 
Lemma~\ref{lem:partition} consequently gives
$p_3(Q)\le\delta K^2\cdot K=\delta K^3$.
By Lemma~\ref{lem:editing}, $\operatorname{ed}_{c}(Q)\le 3\sqrt{\delta}K^2$. 

We obtain a new graph $G'$ from $G$ through the following operations. First, delete all edges incident to $V_0$. Next, add all missing edges inside each $V_i$ for $i=1,\dots,K$. Then, delete all edges between all $\delta$-empty pairs, and add all missing edges between $\delta$-full pairs. Finally, delete all edges between all $\delta$-exceptional pairs.
We now estimate the number of edges added or deleted in each step. 
The first step changes at most $|V_0|n\le\delta n^2$ edges. 
The second changes at most $K\binom m2\le\frac{(Km)^2}{2K}\le\frac{n^2}{2K}\le\frac{\delta}{2}n^2$.
The third changes at most $\delta\binom K2m^2\le\frac{\delta n^2}{2}$.
The final step changes at most $\delta K^2m^2\le\delta n^2$.
Thus, the total number of edges added or deleted is at most $3\delta n^2$.
Now, $G'[V_0]$ is independent, and $G'[V_i]$ is a clique for $i=1,\dots,K$. For $\{i,j\}\subseteq  \{1,\dots,K\}$,  $V_i$ and $V_j$ are completely connected in $G'$ precisely when their corresponding vertices in $Q$ are adjacent.

To conclude the construction, we obtain a new graph from $G'$ by performing the operations corresponding to the transformation of $Q$ into a union of cliques. 
Note that whenever an edge is added to or deleted from $Q$, we change all $m^2$ vertex pairs between the corresponding parts. Recall that $\operatorname{ed}_{c}(Q)\le 3\sqrt{\delta}K^2$.
The number of edges added or deleted is at most $3\sqrt{\delta}K^2 m^2\le 3\sqrt\delta n^2$. 
Together with the steps before, the total number of edges added or deleted is at most 
\[
 (3\delta+3\sqrt\delta)n^2
 =\left(\frac{3\eta^2}{100}+\frac{3\eta}{10}\right)n^2
 <\eta n^2.
\]
This shows that $\operatorname{ed}_{c}(G)\le \eta n^2$.

<!-- lemma-8: body_md -->
\label{lem:balance}
Let $G$ be a graph of order $n$ with average degree $k$.
Suppose that a union of cliques can be obtained from $G$ by $t$ edge additions or deletions. 
Let $\mathcal E=2t+\sum_{v\in V(G)}|d_G(v)-k|$.
If $\mathcal E<\frac{(k+1)n}{4}$, then $\operatorname{ed}_{bc}(G)\le  t+\frac{4\mathcal E n}{k+1}+\frac{n^2}{k+1}$.
<!-- end lemma-8 -->
<!-- proof-5: body_md -->
Let $H$ be a union of cliques on $V(G)$ obtainable with $t$ edge additions or deletions, 
and let $a_1,\ldots,a_\ell$ be the orders of its components. Assume that $a_1,\dots,a_r\ge \frac{k+1}{2}$, and if $r<\ell$,  $a_{r+1},\dots,a_\ell<\frac{k+1}{2}$. 
For every vertex $v\in V(G)$, we have $|d_H(v)-k|=|d_H(v)-d_G(v)|+|d_G(v)-k|$ and so~\eqref{eq:degree-edit} gives \[
\sum_{v\in V(G)}|d_H(v)-k|\le2t+\sum_{v\in V(G)}|d_G(v)-k|\le \mathcal E.
\]
Let $W=\sum_{i=1}^\ell a_i|a_i-(k+1)|$. Then $W=\sum_{v\in V(G)}|d_H(v)-k|\le \mathcal E$.

For every vertex $v$ in a component of order smaller than $\frac{k+1}{2}$, it contributes more than $\frac{k+1}{2}$ to $W$. The number of such vertices is equal to $s=\sum_{i=r+1}^\ell a_i$. Then $s\le \frac{2\mathcal{E}}{k+1}$. As $\mathcal E<\frac{(k+1)n}{4}$, $s<n$. This implies $r\ge 1$.  
As $a_i\ge \frac{k+1}{2}$ for $i\le r$, we have $r\frac{k+1}{2}\le \sum_{i=1}^ra_i\le \sum_{i=1}^\ell a_i=n$ and so $r\le \frac{2n}{k+1}$. Moreover, \[
\frac{k+1}{2}\sum_{i=1}^r |a_i-(k+1)|\le \sum_{i=1}^ra_i|a_i-(k+1)|\le \sum_{i=1}^\ell a_i|a_i-(k+1)|\le \mathcal{E}.
\]
Thus $\sum_{i=1}^r |a_i-(k+1)|\le \frac{2\mathcal{E}}{k+1}$.
As $\sum_{i=1}^r a_i=n-s$, we have $|r(k+1)-n|\le\sum_{i=1}^r|(k+1)-a_i|+s\le\frac{4\mathcal E}{k+1}$.
It follows that
\[
 \sum_{i=1}^r\left|a_i-\frac nr\right|
 \le\sum_{i=1}^r|a_i-(k+1)|+r\left|(k+1)-\frac nr\right|
 \le\frac{6\mathcal E}{k+1}.
\]
Write $n=qr+r_0$, where $q=\lfloor \frac{n}{r}\rfloor$ and $0\le r_0<r$. Let $b_1,\ldots,b_r$ be such that $b_1=\cdots=b_{r_0}=q+1$ and
$b_{r_0+1}=\cdots=b_r=q$. Then \[
 \sum_{i=1}^r|a_i-b_i|\le \sum_{i=1}^r|a_i-\frac{n}{r}|+r\le \frac{6\mathcal E}{k+1}+r. 
\]

Let $H^*$ be a balanced union of cliques with exactly $r$ components. We obtain $H^*$ from $H$ by retaining $\min\{a_i,b_i\}$ vertices in each of the first $r$ components of $H$, and then assigning the remaining vertices among these $r$ components so that the $i$-th component has order $b_i$, then completing each component
and deleting all edges between distinct components. So the number of assigned vertices is \[
s+\sum_{i=1}^r(a_i-b_i)_+=s+\frac{1}{2}\sum_{i=1}^r(|a_i-b_i|+(a_i-b_i))=\frac{1}{2}\left(s+\sum_{i=1}^r|a_i-b_i|\right)\le \frac{4\mathcal E}{k+1}+\frac{r}{2}\le \frac{4\mathcal E}{k+1}+\frac{n}{k+1}. 
\]
From $H$ to $H^*$, the edges added or deleted are only those incident with assigned vertices. Hence the number of such edges is at most $\left(s+\sum_{i=1}^r(a_i-b_i)_+\right)n\le\frac{4\mathcal E n}{k+1}+\frac{n^2}{k+1}$. Thus, $\operatorname{ed}_{bc}(G)\le t+\frac{4\mathcal E n}{k+1}+\frac{n^2}{k+1}$.

<!-- lemma-9: body_md -->
\label{lem:degrees}
For a graph $G$ of order $n\ge 2$,  
$\sum_{v\in V(G)}|d_G(v)-d|\le2n\sqrt{2D}$.
<!-- end lemma-9 -->
<!-- proof-6: body_md -->
Let $\mathbf{z}\in \mathbb R^n$ with $z_v=d_G(v)-d$ for $v\in V(G)$.
Then, as $p(n-1)=d$, $\mathbf{z}=M\mathbf{1}$. Let $Z=\{v\in V(G):z_v<0\}$ and let $s=-\sum_{v\in Z}z_v$. 
We have
$\sum_{v\in V(G)}z_v=\mathbf{1}^\top M\mathbf{1}=0$, and so $\sum_{v\in V(G)}|z_v|=2s$. 

If $s=0$, then $\mathbf{z}$ is a zero vector, proving the lemma. Suppose in the following that $s>0$. Let $t=\frac{s}{n^2}$. As $|z_v|\le n$, $0<t\le 1$. Let $\mathbf{x}=\mathbf{1}-t\mathbf{1}_Z$. Then $0\le x_v\le 1$ for $v\in V(G)$. Let $U\subseteq V(G)$ be the random subset in which each vertex $v$ is included independently with probability $x_v$.  Note that $\frac{1}{2}\mathbf{x}^\top M\mathbf{x}=\mathbb E\operatorname{disc}_G(U)\le D$, where $\mathbb E$ denotes expectation with respect to these independent choices.
On the other hand, \[
\mathbf{x}^\top M\mathbf{x}=-2t\mathbf{1}_Z^\top M\mathbf{1}+t^2\mathbf{1}_Z^\top M\mathbf{1}_Z=2ts+t^2\mathbf{1}_Z^\top M\mathbf{1}_Z\ge 2ts-t^2n^2=\frac{s^2}{n^2},
\]
where the inequality holds as $|\mathbf{1}_Z^\top M\mathbf{1}_Z|\le |Z|^2\le n^2$. So $s\le n\sqrt{2D}$.
Thus, $$\sum_{v\in V(G)}|d_G(v)-d|=\sum_{v\in V(G)}|z_v|=2s\le 2n\sqrt{2D},$$ as required.

<!-- lemma-10: body_md -->
\label{lem:certificate}
There exist a nonnegative diagonal matrix $Y=\operatorname{diag}(y_1,\ldots,y_n)$ and a matrix $R\succeq0$ such that $Y-M+R\succeq0$,  $R_{ii}\le2$ for any $i=1,\dots,n$ and $\sum_{i=1}^n y_i+\frac14\lVert R\rVert_F^2\le4\pi D$.
<!-- end lemma-10 -->
<!-- proof-7: body_md -->
Let $\beta=\max_{\sigma\in\{-1,1\}^n}\sigma^\top M\sigma$. For a sign vector $\sigma$, let $U$ be the set of vertices $v\in V(G)$ with $\sigma(v)=1$, and $U^c=V(G)\setminus U$. 
Write $\mathbf{x}=\mathbf 1_U$ and $\mathbf{z}=\mathbf 1_{U^c}$, so that $\sigma=\mathbf{x}-\mathbf{z}$.
Then
\[
0=(\mathbf{x}+\mathbf{z})^\top M(\mathbf{x}+\mathbf{z})
   =\mathbf{x}^\top M\mathbf{x}+2\mathbf{x}^\top M\mathbf{z}+\mathbf{z}^\top M\mathbf{z}.
\]
For the sign vector $\mathbf{x}-\mathbf{z}$ this gives \[
 (\mathbf{x}-\mathbf{z})^\top M(\mathbf{x}-\mathbf{z})=2\mathbf{x}^\top M\mathbf{x}+2\mathbf{z}^\top M\mathbf{z}=4\bigl(\operatorname{disc}_G(U)+\operatorname{disc}_G(U^c)\bigr)\le8D.
\]
Thus $\beta\le8D$.

Let $g$ be the function from Lemma~\ref{lem:positivepart}. For
$\mathbf{y}=(y_1,\ldots,y_n)$ with all $y_i\ge 0$, define
\[
\Phi(\mathbf{y})=\sum_{i=1}^n y_i+g\bigl(M-\operatorname{diag}(\mathbf{y})\bigr).
\]
By Lemma~\ref{lem:positivepart}, $g$ is continuous. Moreover, $\Phi(\mathbf{y})\ge\sum_i y_i\ge0$. The set $\{\mathbf{y}\in[0,\infty)^n:\Phi(\mathbf{y})\le\Phi(\mathbf{0})\}$ is nonempty, closed and contained in the compact cube $[0,\Phi(\mathbf{0})]^n$.
Hence $\Phi$ attains a minimum on $[0,\infty)^n$.
Fix a minimizing vector $\mathbf{y}$ of $\Phi$ and let $Y=\operatorname{diag}(\mathbf{y})$ and $R=(M-Y)_+$. Clearly, $R\succeq 0$. 
By Lemma~\ref{lem:positivepart}, for each $i$, we have
\[
 \frac{\partial\Phi}{\partial y_i}=1-\frac12R_{ii}.
\]
Note that $R_{ii}$ depends on $\mathbf{y}$. As $\mathbf{y}$ minimizes $\Phi$,  $\frac{\partial\Phi}{\partial y_i}=0$ if $y_i>0$ and $\frac{\partial\Phi}{\partial y_i}\ge 0$ if $y_i=0$. 
Consequently, $R_{ii}\le2$.
Moreover, $Y-M+R=(M-Y)_-\succeq0$.

Recall that $\mathbf{y}$ minimizes $\Phi$, $y_i(1-\frac{1}{2}R_{ii})=0$ for $i=1,\dots,n$. Then $y_iR_{ii}=2y_i$, and so $\langle Y,R\rangle=\sum_{i=1}^ny_iR_{ii}=2\sum_{i=1}^ny_i$. 
Since $M-Y=R-(M-Y)_{-}$ and the positive and negative parts are orthogonal in the Frobenius inner product, $\langle M-Y,R\rangle=\|R\|_F^2$. Hence
\[
 \langle M,R\rangle=\langle Y,R\rangle+\lVert R\rVert_F^2.
\]
Let $X=\frac{1}{2}R$. Then $X\succeq 0$ and $X_{ii}\le 1$ for $i=1,\dots,n$. By Lemma~\ref{lem:rounding}, we have \[
 \sum_{i=1}^n y_i+\frac14\lVert R\rVert_F^2=\frac12\langle Y,R\rangle+\frac14\lVert R\rVert_F^2=\langle M,X\rangle-\lVert X\rVert_F^2 \le\frac\pi2\beta\le4\pi D,
\]
proving the lemma.

\begin{equation}\label{eq:main-threshold}
 \begin{split}
 N_\varepsilon=\left\lceil\max\left\{
 2N_\eta,\ \rho^{-1},\
 \left(\frac{4}{a_\eta}\right)^4,\

 \end{split}

\begin{equation}\label{eqq1}
 A+R+Y+(1-p)I\succeq0.
\end{equation}

The bound $\sum_vy_v\le4\pi D$ yields
\[
 |W|\le\frac{\sum_vy_v}{t_0}
 \le\frac{16\pi D}{a_\eta n^{1/4}}

$m\ge(1-\eta)n\ge n/2\ge N_\eta$ and

 \le2^{-2/3}n^{2/3}\le m^{2/3}.

\[
 \lambda_m(A[U]+R[U])\ge-(t_0+1-p)\ge-(t_0+1)\ge -\frac{a_\eta}{2}n^{1/4}\ge -a_\eta |U|^{1/4}.
\]

A union of cliques can be obtained from $\overline{G}$ by isolating all vertices of $W$ and adding or deleting at most $\eta |U|^2$ edges of $\overline{G}[U]$. Thus, \[
\operatorname{ed}_c(\overline{G})\le |W|n+\operatorname{ed}_c(\overline{G}[U])\le \eta n^2+\eta |U|^2\le 2\eta n^2\le \frac{\rho}{2}n^2.

As $n\ge N_\varepsilon\ge \bigl(2\sqrt{2c_0}/\rho\bigr)^{8/3}$, we have $\sqrt{2c_0}\le (\frac{\rho n}{2})^{3/8}$.

\sum_{v\in V(G)}|d_{\overline{G}}(v)-k|=\sum_{v\in V(G)}|d_G(v)-d|\le 2n\sqrt{2D}\le 2\sqrt{2c_0}n^{13/8}\le \rho n^2.

\]

\label{lem:compression}
Let $n\ge N_*$, $q\in\{2,3\}$ and $t>0$ satisfying $t\ge16s\,n^{1-1/q}$. If $tS(t)\ge n^{9/5}$, then $S\left(\frac{t^q}{(16n)^{q-1}}\right)\ge\frac{S(t)^q}{4n^{q-1}}$.

<!-- proof-10: body_md -->
Let $X=\sum_{i\in I(t)}\theta_i \mathbf{z}_i \mathbf{z}_i^\top$.
Then $0\preceq X\preceq C+sI$. By~\cite[Theorem~4.1]{JMTZ}, $(C+sI)^{\circ q}\succeq X^{\circ q}$. 
Let $W=\operatorname{span}\{\mathbf{z}_{i_1}\circ\cdots\circ \mathbf{z}_{i_q}: i_1,\ldots,i_q\in I(t)\}$ and $d=\dim W$.
Then $d\le|I(t)|^q\le\bigl(\frac{S(t)}{t}\bigr)^q$.
By the identity $(\mathbf{x}\mathbf{x}^\top)\circ(\mathbf{y}\mathbf{y}^\top)=(\mathbf{x}\circ \mathbf{y})(\mathbf{x}\circ \mathbf{y})^\top$, each column of $X^{\circ q}$ is a Hadamard product of $q$ columns of $X$. Hence the range of $X^{\circ q}$ is contained in $W$. 
So H\"older's inequality gives
\[
\operatorname{tr}_W\bigl((C+sI)^{\circ q}\bigr)\ge\operatorname{tr} X^{\circ q}=\sum_{i=1}^n X_{ii}^q\ge\frac{(\operatorname{tr} X)^q}{n^{q-1}}=\frac{S(t)^q}{n^{q-1}}.
\]

Decompose $(C+sI)^{\circ q}$ as $(C+sI)^{\circ q}=C+D+E$, where $D$ is the diagonal part of $(C+sI)^{\circ q}-C$ and $E$ is its zero-diagonal part. 
As $0\le C_{ii}\le2$ and $s\ge1$, we have $0\preceq D\preceq3^qs^qI$ and so $\operatorname{tr}_W D\le3^qs^q d$.
For $i\ne j$, let $f_q(t)=t^q-t$. Then $f_q(A_{ij})=0$ and $E_{ij}=f_q(C_{ij})-f_q(A_{ij})$.
Let $L_q=\sup_{x\in[-2,3]}|f_q'(x)|$. Then $L_2=5$ and $L_3=26$. Hence, by the mean value theorem, $\lVert E\rVert_F\le L_q\lVert R\rVert_F\le L_q n^{2/3}$.  
By Cauchy--Schwarz inequality, $|\operatorname{tr}_W E|\le \lVert P_W\rVert_F\lVert E\rVert_F\le  L_q n^{2/3}\sqrt d$.

Let $K=\frac{t^q}{(16n)^{q-1}}$. 
The proof of~\cite[Lemma~6.2]{JMTZ}, which applies directly to any real symmetric matrix, yields $\operatorname{tr}_W C\le S(K)+Kd$. Thus,
\[
 S(K)\ge\operatorname{tr}_W C-Kd\ge \operatorname{tr}_W\bigl((C+sI)^{\circ q}\bigr)-\operatorname{tr}_W D-\operatorname{tr}_W E-Kd\ge \frac{S(t)^q}{n^{q-1}}
       -(K+3^qs^q)d-L_q n^{2/3}\sqrt d.
\]
As $S(t)>0$, \[
\frac{(K+3^qs^q)d+L_q n^{2/3}\sqrt d}{\frac{S(t)^q}{n^{q-1}}}
\le
\frac{1}{16^{q-1}}+\left(\frac{3}{16}\right)^q
+\frac{L_q n^{q-1+2/3}}{(tS(t))^{q/2}}\le \frac{1}{16}+\frac{9}{256}+\frac{1}{4}<\frac{3}{4}.
\]
                                                                                    
Thus, $S(K)\ge \frac{S(t)^q}{4n^{q-1}}$.

\label{lem:tail}
Let $n\ge N_*$ and $t>0$ satisfying $16n^{7/8}\le t\le2n$. Then $S(t)\le4n\left(\frac{16n}{t}\right)^{3/4}$.

<!-- proof-11: body_md -->
Let $a$ be such that $0<a\le\frac18$ and $t=16n^{1-a}$, and let
$F(x)=\frac{S(x)}{4n}$ for $x>0$.
It suffices to prove that $F(t)\le n^{3a/4}$.
                                                             

Suppose to the contrary that $F(t)>n^{3a/4}$.
We will carry this inequality forward along a sequence of parameter transformations until the parameter lies in $[\frac13,\frac12]$, where it contradicts the known upper bound. To this end, we first choose a sequence of transformations that takes the parameter $a$ to a number in $[\frac13,\frac12]$.

Let $j$ be the least nonnegative integer such that $2^ja\ge\frac13$.
Then $j\ge2$ and $\frac13\le2^ja<\frac23$.
If $2^ja\le\frac12$, then let $b=2^j$.                                       
If $2^ja>\frac12$, then $j\ge3$ and the minimality of $j$ gives
$a<\frac1{3\cdot2^{j-1}}\le\frac1{12}$. 
In this case,  let $b=3\cdot2^{j-2}$.                                                                          
The final parameter is then
$$ba=\frac34\,2^ja\in\left(\frac38,\frac12\right).$$
Thus, in either case, the final parameter $ba$ lies in
$[\frac13,\frac12]$.
For $q=2$, the starting value is at most $\frac{ba}{2}\le\frac14$.
For $q=3$, the starting value is $a<\frac1{12}$.
Thus the required bounds on $a$ hold in all cases.

Now we show by induction that $F(16n^{1-c})>n^{3c/4}$ for every $c$ in the chosen sequence.
The assertion holds at the start $c=a$ by our contradiction hypothesis.
Suppose it holds at some $c$ from which the next step is to be made, and let $x=16n^{1-c}$.
Since $c\le ba\le\frac12$, we have
$$xS(x)=64n^{2-c}F(x)>64n^{2-c/4}
      \ge64n^{15/8}>n^{9/5}.$$
By Lemma~\ref{lem:compression},
\[
F\left(16n^{1-qc}\right)
=\frac{S\left(16n^{1-qc}\right)}{4n}
=\frac{S\left(\frac{(16n^{1-c})^q}{(16n)^{q-1}}\right)}{4n}
\ge \frac{S(16n^{1-c})^q}{16n^q}
=4^{q-2}F(16n^{1-c})^q
\ge F(16n^{1-c})^q.
\]

At the end $ba$, however,~\eqref{eq:basic-spectral} gives $$n^{1/4}\le n^{3ba/4}<F(16n^{1-ba}) \le\frac{3s}{4}\le\frac34 n^{1/4},$$
a contradiction.
It follows that our contradiction hypothesis was false, and hence $F(t)\le n^{3a/4}$.
Therefore, $S(t)\le4n\,n^{3a/4}=4n\left(\frac{16n}{t}\right)^{3/4}$, as required.
