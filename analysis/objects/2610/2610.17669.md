<!-- condition-1: body_md -->
Let $\phi$ be a smooth strictly plurisubharmonic exhaustion function on $X$. We say that $\phi$ satisfies \textit{Condition $\mathfrak{A}$} if
\begin{equation}\label{conditionA}
    \phi \geq 1 \quad \text{and}\quad i\partial\phi\wedge\bar{\partial}\phi \leq B \phi \,i\partial\bar{\partial}\phi
\end{equation}
for some $B<\infty$.
<!-- end condition-1 -->
<!-- maintheo-1: body_md -->
Let $X$ be a Stein manifold with a plurisubharmonic exhaustion function $\phi$ satisfying the Condition $\mathfrak{A}$. Let $\Theta_k$ be the determinantal point process associated with the weighted Bergman space of holomorphic $n$-forms with weight $e^{-k\phi}$, defined in Definition \ref{definitionDPP}. Set 
\begin{equation*}
    L_k=\frac{1}{k^n}\Theta_k.
\end{equation*}
The laws of $L_k$, on the space of positive Radon measures with the vague topology, satisfy a large deviation principle with speed $r_k=k^{n+1}$ and a good rate function
\begin{equation*}
    I_\phi(\mu)=\sup_{u\in C_c^\infty(X)}\biggl\{ \mathcal{E}_\phi(P(\phi+u))-\int_X ud\mu \biggr\}.
\end{equation*}
The unique zero of $I_\phi$ is
\begin{equation*}
    \mu_\phi=\operatorname{MA}(\phi)=\frac{(dd^c \phi)^n}{n!}.
\end{equation*}
<!-- end maintheo-1 -->
<!-- example-1: body_md -->
Let $X=\mathbb{C}^n$ and $\phi_0(z)=1+|z|^2$. Then $\phi_0$ satisfies Condition $\mathfrak{A}$ with $B=1$. Identifying a holomorphic $n$-form with its coefficient relative to $dz^1\wedge\cdots\wedge dz^n$, the space $\mathcal{H}(k\phi_0)$ is a weighted Fock space. After absorbing the Gaussian weight into the kernel, the associated determinantal process $\Theta_k$ has correlation kernel
\begin{equation*}
    \widetilde{K}_k(z,w)=\left(\frac{k}{\pi}\right)^n \exp\left(k z\cdot\overline{w} -\frac{k}{2}|z|^2 -\frac{k}{2}|w|^2 \right)
\end{equation*}
with respect to Lebesgue measure $d\lambda_{2n}$, where $z\cdot\overline{w}=\sum_{j=1}^n z^j\overline{w^j}$. In particular,
\begin{equation*}
    \frac{1}{k^n}\beta_{k\phi_0}=\frac{1}{\pi^n}d\lambda_{2n}=\operatorname{MA}(\phi_0)
\end{equation*}
for every $k>0$. The main theorem therefore gives an LDP for $L_k=k^{-n}\Theta_k$ in the vague topology, with speed $k^{n+1}$ and good rate function $I_{\phi_0}$, whose unique zero is $\pi^{-n}d\lambda_{2n}$.

When $n=1$, $\Theta_1$ is the standard infinite fermion process with exponential kernel \cite{shirai2006large}. If $\mathcal{G}$ denotes its random configuration, then
\begin{equation*}
    \Theta_k \overset{\mathrm{law}}{=} \sum_{\zeta\in\mathcal{G}} \delta_{\zeta/\sqrt{k}}.
\end{equation*}
Equivalently, writing $k=R^2$, the random measures
\begin{equation*}
    \frac{1}{R^2}\sum_{\zeta\in\mathcal{G}} \delta_{\zeta/R}
\end{equation*}
satisfy an LDP as $R\rightarrow\infty$, with speed $R^4$ and unique zero $\pi^{-1}dA$, where $dA$ is planar Lebesgue measure.
<!-- end example-1 -->
<!-- definition-1: body_md -->
A complex manifold $X$ is called a \textit{Stein manifold} if

(1) For any compact subset $K\subset X$, $\widehat{K}:=\{x\in X : |f(x)|\leq \sup_K |f|, \;\forall f\in \mathcal{O}(X)\}$ is compact.

(2) For any $x\neq y\in X$,  there exists $f\in \mathcal{O}(X)$ such that $f(x)\neq f(y)$.
<!-- end definition-1 -->
<!-- proposition-1: body_md -->
Let $X$ be a Stein manifold. Then there is a function $\phi$ satisfying Condition $\mathfrak{A}$. For example, let $F=(F_1,\cdots,F_N):X\hookrightarrow \mathbb{C}^N$ be a proper holomorphic embedding. Then
\begin{equation*}
    \phi_F=1+\sum_{j=1}^N |F_j|^2
\end{equation*}
satisfies the Condition $\mathfrak{A}$ with $B=1$.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Properness of $\phi_F$ follows from properness of $F$. Since $F$ is an immersion, 
\begin{equation*}
    i\partial\bar{\partial} \phi_F = i \sum_j \partial F_j \wedge \bar{\partial}\overline{F_j}=i \sum_j d F_j \wedge d \overline{F_j}=F^*(i\sum_j dz_j\wedge d\bar{z}_j)>0.
\end{equation*}
Hence $\phi_F$ is a strictly psh exhaustion function. Equation \eqref{conditionA} with $B=1$ follows from the Cauchy--Schwarz inequality: for $\xi\in T^{1,0}_x X$,
\begin{equation*}

\end{equation*}

<!-- proof-4: body_md -->
\emph{(1) Proof of the compact support.---}
Choose a smooth nonincreasing function $\eta:\mathbb{R}\rightarrow [0,1]$ with $\eta=1$ on $(-\infty,1]$ and $\eta=0$ on $[2,\infty)$. Let $C:=\|u\|_\infty$. Choose $L$ large enough so that $\operatorname{supp} u \subset \{\phi<L\}$ and 
\begin{equation}\label{Lcondition}
    \frac{3BC}{L}\|\eta''\|_\infty <1.
\end{equation}

Now set $h:=-C\eta(\phi/L)$. On $\operatorname{supp} u$, $\phi<L$ and we have $h=-C \leq u$, while outside $\operatorname{supp} u$ we have $h\leq 0=u$. Hence $h\leq u$ on $X$. We claim that $\phi+h$ is strictly plurisubharmonic. Indeed,
\begin{equation*}
    i\partial\bar\partial (\phi+h)=\left(1-\frac{C}{L}\eta'(\phi/L)\right)  i\partial\bar\partial \phi-\frac{C}{L^2}\eta''(\phi/L)i\partial \phi\wedge \bar\partial \phi.
\end{equation*}
Since $h$ is nonconstant only on $L\leq \phi\leq 2L$, the second term is bounded below by (using \eqref{Lcondition})
\begin{equation*}
-\frac{C}{L^2}\|\eta''\|_\infty B\phi \,i\partial\bar\partial \phi \geq -\frac{1}{3L}\phi \,i\partial\bar\partial \phi \geq -\frac{2}{3}i\partial\bar\partial \phi.
\end{equation*}
This proves $ i\partial\bar\partial (\phi+h)>0$. Hence $\phi+h$ is plurisubharmonic and lies below $\phi+u$, and we have
\begin{equation*}
    \phi+h \leq P(\phi+u) \leq \phi+u,
\end{equation*}
since $\phi+u$ is continuous. On $\{\phi\geq 2L\}$, both $h$ and $u$ vanish and hence $\operatorname{supp} P(\phi+u)-\phi \subset \{\phi \leq 2L\}$. For the uniform statement, replace $C$ by a common sup-norm bound.

\vspace{2mm}
\emph{(2) Proof of the $C^{1,1}$-regularity.---}
We basically adapt the method of Bedford and Taylor on unit ball case \cite{MR445006}. See also \cite[Section 4.3]{klimek1991pluripotential}. It was adapted on $\mathbb{C}^n$ \cite{berman2009poly} and on projective manifolds \cite{berman2009bergman}. The key point of these arguments is using holomorphic vector fields that generate small translations on manifolds in every direction at each point. In our Stein manifold setting, Cartan's Theorem A provides such vector fields. 

\begin{lemma}\label{lemmatransl}
Let $K$ be a compact subset of $X$. Then there are global holomorphic vector fields $V_1,\cdots,V_m$, neighborhoods $K\Subset U_0 \Subset U_1 \Subset X$, a constant $\epsilon>0$, and a map
\begin{equation*}
    S : U_1 \times B_{\mathbb{C}^m}(0,\epsilon) \longrightarrow X
\end{equation*}
such that
\begin{enumerate}[label=\textup{(\roman*)}]
\item $S$ is jointly holomorphic; in particular, $S_\zeta(x):=S(x,\zeta)$ is holomorphic in $x$;
\item $S_0(x)=x$;
\item $D_\zeta S(x,0): \mathbb{C}^m \rightarrow T_xX$ is surjective for $x$ in a neighborhood of $\overline{U}_0$;
\item in any fixed finite coordinate cover of $\overline{U}_0$,
\begin{equation}\label{Slinear}
 S_\zeta(x)=x+A_x\zeta+O(|\zeta|^2), \quad A_x(e_j)=V_j(x),
\end{equation}
uniformly in $x$.
\end{enumerate}
Moreover, after shrinking the neighborhoods, every pair of sufficiently nearby points $x,y \in U_0$ admits $\zeta$ with $S_\zeta(x)=y$ and $|\zeta| \leq C \textrm{dist}(x,y)$.
\end{lemma}

Let $q=\phi+u$ and $p=P(q)$. By the previous result, there is a compact set $C$ such that $\operatorname{supp}(p-\phi)\cup \operatorname{supp}u\subset C$. Thus $p=q=\phi$ on $X\setminus C$. Choose a relatively compact neighborhood $D$ of $C$ and apply Lemma \ref{lemmatransl} with $K=\overline{D}$. Let $U=U_1$ in the Lemma, and choose a collar $W$ around $\partial D$ satisfying $\partial D \subset W, \;\overline{W}\Subset U\setminus C$. Possibly shrink $\epsilon$ in the Lemma so that for all $\zeta$ in the ball,
\begin{equation*}
    S_{\pm\zeta}(\overline{D})\Subset U,\qquad S_{\pm\zeta}(\overline{W})\Subset X\setminus C.
\end{equation*}

We first show that $p$ is locally Lipschitz. Since $q$ and $\phi$ are smooth on $\overline{U}$, by \eqref{Slinear},
\begin{equation}\label{qphi}
    q(S_{\pm\zeta}x)-C_1|\zeta| \leq q(x), \quad x\in D\quad\text{and}\quad \phi(S_{\pm\zeta}x)-C_1|\zeta| \leq \phi(x), \quad x\in W
\end{equation}
for some $C_1$. On $D$, define the psh function
\begin{equation*}
    r_\zeta(x)=p(S_\zeta x)-C_1|\zeta|.
\end{equation*}
By \eqref{qphi}, $r_\zeta \leq q$ on $D$. On $W\cap D$, both $x$ and $S_\zeta x$ lie in $X\setminus C$, where $p=q=\phi$. Thus \eqref{qphi} implies $r_\zeta \leq p$ on $W\cap D$. It follows that
\begin{equation*}
\widetilde{p}_\zeta=
\begin{cases}
\max\{p,r_\zeta\},&\text{on}\;D,\\
p,&\text{on}\;X\setminus D,
\end{cases}
\end{equation*}
is a psh function that lies below $q$; hence $\widetilde{p}_\zeta \leq p$. Thus we have $r_\zeta \leq p$ on $D$, which gives
\begin{equation*}
    p(S_\zeta x)-p(x)\leq C_1 |\zeta|.
\end{equation*}
For $x,y$ in a small coordinate chart in $D$, by the last assertion of Lemma \ref{lemmatransl}, we have $|\zeta|=O(\textrm{dist}(x,y))$ sending $x$ to $y$ and $y$ to $x$. Hence we get
\begin{equation}\label{Lipshitz}
    |p(x)-p(y)|\leq C_2 \textrm{dist}(x,y).
\end{equation}

We next prove semiconcavity of $p$. Since $q$ is smooth, by \eqref{Slinear},
\begin{equation}
    \frac{q(S_\zeta x)+q(S_{-\zeta}x)}{2}-C_3|\zeta|^2 \leq q(x),\quad x\in D,
\end{equation}
for some $C_3$. The analogous estimate holds for $\phi$ on $W$. On $D$, define the psh function
\begin{equation*}
    R_\zeta(x)=\frac{p(S_\zeta x)+p(S_{-\zeta}x)}{2}-C_3|\zeta|^2.
\end{equation*}
As in the preceding argument, this can be pasted with $p$ and gives
\begin{equation}\label{concave}
    p(S_\zeta x)+p(S_{-\zeta}x)-2p(x)\leq 2C_3 |\zeta|^2.
\end{equation}
Fix a coordinate ball around $x$. For small enough $h\in\mathbb{C}^n$, the last assertion of Lemma \ref{lemmatransl} gives $\zeta=\zeta(x,h)$ with
\begin{equation*}
    S_\zeta x=x+h, \quad |\zeta|=O(|h|).
\end{equation*}
By \eqref{Slinear}, $h=A_x\zeta+O(|\zeta|^2)$. This implies
\begin{equation*}
    S_{-\zeta}x=x-A_x\zeta+O(|\zeta|^2)=x-h+O(|\zeta|^2).
\end{equation*}
By the local Lipschitz estimate \eqref{Lipshitz}, we have $p(S_{-\zeta}x)=p(x-h)+O(|h|^2)$. The estimate \eqref{concave} now gives
\begin{equation}\label{differnce}
     p(x+h)+p(x-h)-2p(x)\leq C_4 |h|^2.
\end{equation}

By a standard change of variable argument, for any real unit vector $e$, from \eqref{differnce} we get $C_4d\lambda_{\mathbb{R}^{2n}}-D^2_{e,e}p \geq 0$ in the sense of distributions. Thus $D^2_{e,e}p$ is a signed Radon measure. On the other hand, plurisubharmonicity gives
\begin{equation*}
    D^2_{e,e}p+D^2_{Je,Je}p \geq 0
\end{equation*}
in the sense of distributions. Applying the upper bound to $Je$, we obtain $D^2_{e,e}p \geq -C_4d\lambda_{\mathbb{R}^{2n}}$. By the Radon--Nikodym theorem applied to $2C_4d\lambda_{\mathbb{R}^{2n}}\geq C_4d\lambda_{\mathbb{R}^{2n}}-D^2_{e,e}p \geq 0$, we conclude that $D^2_{e,e}p$ is represented by an $L^\infty_{\mathrm{loc}}$ function. Then by polarization, every entry of the distributional Hessian is an $L^\infty_{\mathrm{loc}}$ function. That is, $p \in W^{2,\infty}_{\mathrm{loc}}=C^{1,1}_{\mathrm{loc}}$. The uniform statement also follows.

Let $q=\phi+u$, $p=P(\phi+u)$, and $p_t=P(q+th)$. We first use the definition of the envelope to show that the envelope operator is a contraction in the sup norm:
\begin{equation}\label{envelopecontraction}
   \|p_t-p\|_\infty \leq |t|\|h\|_\infty.
\end{equation}
Indeed, if $M=\|h\|_\infty$, then $p_t-|t|M \leq q$ and $p-|t|M\leq q+th$. The functions on the left are competitors for the corresponding envelopes. By Theorem \ref{plurimain}, there is a common compact support for $p_t-\phi$ when $|t|\leq1$.

By \eqref{envelopecontraction}, the potentials converge uniformly and are locally bounded as $t\rightarrow 0$. By the continuity property of the MA operator,
\begin{equation}\label{MAcontinuity}
   \operatorname{MA}(p_t) \longrightarrow \operatorname{MA}(p) \qquad \text{locally weakly on} \,X.
\end{equation}
Thus the integrals against every function in $C_c(X)$ converge.

To compute the derivative for $t>0$, apply \eqref{energyinequal} to $p_t$ and $p$. By the argument for the orthogonality relation,
\begin{equation*}
   p_t=q+th \quad \operatorname{MA}(p_t)\text{-almost everywhere}, \quad p=q \quad \operatorname{MA}(p)\text{-almost everywhere}.
\end{equation*}
Since $p \leq q$ and $p_t \leq q+th$, it follows that
\begin{equation*}
   p_t-p \geq th \quad \operatorname{MA}(p_t)\text{-almost everywhere}, \quad p_t-p\leq th \quad \operatorname{MA}(p)\text{-almost everywhere}.
\end{equation*}
Consequently,
\begin{equation}\label{diff1}
   \int_X h\operatorname{MA}(p_t) \leq \frac{\mathcal{E}_\phi(p_t)-\mathcal{E}_\phi(p)}{t} \leq \int_X h\operatorname{MA}(p).
\end{equation}
Letting $t\downarrow 0$ and using \eqref{MAcontinuity}, we obtain the right derivative. For $t<0$, division by $t$ reverses the inequalities and gives
\begin{equation*}
   \int_X h\operatorname{MA}(p) \leq\frac{\mathcal E_\phi(p_t)-\mathcal E_\phi(p)}{t} \leq \int_X h\operatorname{MA}(p_t).
\end{equation*}
The left derivative follows by letting $t\uparrow 0$.

We next prove concavity. For arbitrary $u_0,u_1$, let $p_{u_i}=P(\phi+u_i)$. The right inequality in \eqref{energyinequal}, together with the orthogonality relation, gives
\begin{equation*}
   \mathcal{E}_\phi(p_{u_1})-\mathcal{E}_\phi(p_{u_0})\leq\int_X (p_{u_1}-p_{u_0})\operatorname{MA}(p_{u_0})\leq\int_X (u_1-u_0)\operatorname{MA}(p_{u_0}),
\end{equation*}
since $p_{u_0}=\phi+u_0$ almost everywhere for $\operatorname{MA}(p_{u_0})$ and $p_{u_1}\leq\phi+u_1$. Thus $\operatorname{MA}(p_{u_0})$ is a supergradient and $\mathcal{E}_\phi(P\cdot)$ is concave.

\end{equation*}

\begin{equation*}
    |f_s(z)|^2e^{-ka(z)}=|f_s(z)e^{-kH_x(z)}|^2e^{-ka(0)+O(k|z|^2)}.
\end{equation*}
Apply the submean value property to the local holomorphic function $f_s(z)e^{-kH_x(z)}$ on a ball of radius $\propto k^{-1/2}$ and we get \eqref{uniformupper}.

For the pointwise upper bound, assume $a$ has second derivatives at $x$ and consider the second order Taylor expansion
\begin{equation*}
    a(z)=a(0)+2\operatorname{Re}H(z)+\sum_{j=1}^n \lambda_j|z^j|^2+o(|z|^2).
\end{equation*}
Here $H(z)$ is a holomorphic function that also absorbs the holomorphic quadratic part of the Hessian of $a$, and we perform a unitary coordinate transform to diagonalize the complex Hessian of $a$. Set
\begin{equation*}
    F_{k,s}(z)=f_s(z/\sqrt k)e^{-kH(z/\sqrt k)},\quad Q(z)=a(z)-a(0)-2\operatorname{Re}H(z).
\end{equation*}
Fix $R>0$ and consider the ball of radius $R/\sqrt{k}$ (included in the fixed coordinate chart for large enough $k$). Then for fixed $R$, 
\begin{equation}\label{rho}
    \rho_{k,R}:=\exp\left(\sup_{|z|\leq R}\left|kQ\left(\frac{z}{\sqrt{k}}\right)-\sum_j \lambda_j|z^j|^2\right|\right) \longrightarrow 1
\end{equation}
as $k\rightarrow \infty$. Note that we have
\begin{align}
    k^{-n}\frac{|f_s(0)|^2e^{-ka(0)}}{\|s\|^2_{ka}}&\leq k^{-n}\frac{|f_s(0)|^2e^{-ka(0)}}{\int_{|z|\leq R/\sqrt{k}}|f_s|^2 e^{-ka}d\lambda_{\mathbb{R}^{2n}}}\nonumber\\
    &= \frac{|f_s(0)|^2e^{-ka(0)}}{\int_{|z|\leq R}|f_s(\cdot/\sqrt{k})|^2 e^{-ka(\cdot/\sqrt{k})}d\lambda}\leq \rho_{k,R}\frac{| F_{k,s}(0)|^2}{\int_{|z|\leq R}| F_{k,s}|^2 e^{-\sum \lambda_j|z^j|^2}d\lambda}.\label{berman}
\end{align}
Now we provide an upper bound for the last fraction term in \eqref{berman}. Let $r_1,\cdots,r_n>0$ so that $r_1^2+\cdots+r_n^2<R^2$. By the mean value property of holomorphic functions, we have
\begin{equation*}
    |F_{k,s}(0)|^2=\left| \frac{1}{(2\pi)^n}\int_{[0,2\pi]^n} F_{k,s}(r_1e^{i\theta_1},\cdots,r_ne^{i\theta_n})d\theta \right|^2\leq \frac{1}{(2\pi)^n}\int_{[0,2\pi]^n}\left| F_{k,s}(r_1e^{i\theta_1},\cdots,r_ne^{i\theta_n})\right|^2 d\theta.
\end{equation*}
Multiplying it by $e^{-\sum \lambda_j r_j^2}$ and integrating it with respect to $r_jdr_j$ over all $r_1^2+\cdots+r_n^2<R^2$, we get
\begin{equation*}
    |F_{k,s}(0)|^2\int_{|z|\leq R} e^{-\sum \lambda_j |z^j|^2} d\lambda \leq \int_{|z|\leq R} \left| F_{k,s}\right|^2 e^{-\sum \lambda_j |z^j|^2} d\lambda.
\end{equation*}
This gives for fixed $R$,
\begin{equation*}
    \limsup_{k\rightarrow\infty}k^{-n}B_{ka}(0)\leq \frac{1}{\int_{|z|\leq R} e^{-\sum \lambda_j |z^j|^2} d\lambda}.
\end{equation*}
Letting $R$ tend to infinity proves \eqref{pointwiseupper}.

<!-- theorem-7: body_md -->
\label{thmBergman}
    Let $a\in C^{1,1}_{\mathrm{loc}}(X)$ be a psh function such that $a-\phi \in C^{1,1}_c(X)$. Suppose also that

<!-- proof-10: body_md -->
As in Proposition \ref{localMorse}, choose a coordinate ball around $x$ and write
\begin{equation*}
    a(z)=a(0)+2\operatorname{Re}H(z)+\sum_{j=1}^n \lambda_j|z^j|^2+o(|z|^2).
\end{equation*}
Since $a-\varepsilon\phi$ is psh, $\lambda_j>0$ for all $j$. Hence we get (possibly after shrinking the coordinate ball)
\begin{equation}\label{quadratic}
    Q_x(z):=a(z)-a(0)-2\operatorname{Re}H(z) \geq c_0 |z|^2
\end{equation}
for some $c_0>0$. Choose $\chi\in C_c^\infty$ supported in the coordinate ball with $0\leq \chi\leq 1$, equal to $1$ near $x$. Let 
\begin{equation*}
    \sigma_k=\chi e^{kH}dZ, \qquad g_k=\bar\partial \sigma_k.
\end{equation*}
Here $\sigma_k$ is extended by zero to all of $X$. Put $A=\operatorname{supp}(\bar\partial\chi)$. By \eqref{quadratic}, 
\begin{equation*}
    \eta:=\min_{A} Q_x>0.
\end{equation*}

Let $\psi_x$ be given by Lemma \ref{psix} and set $\Psi_k=ka+\psi_x$. Recall that $\omega_\phi$ is a complete K\"ahler metric (Proposition \ref{complete}). Thus the weight $\Psi_k$ satisfies the condition of Demailly's $L^2$-estimate for large enough $k$ since
\begin{equation*}
    i\partial\bar\partial \Psi_k \geq (k\varepsilon-C_x)\omega_\phi.
\end{equation*}
Choose $k_0$ so that $k_0 \geq (C_x+1)/\varepsilon$. Then for $k\geq k_0$, we can apply Theorem \ref{demailly} with $\delta=1$. Since $g_k$ is supported on $A$, we have
\begin{equation*}
    \int_X |g_k|^2_{\omega_\phi}e^{-\Psi_k}\frac{\omega_\phi^n}{n!}=e^{-ka(x)}\int_A \left| \bar\partial\chi dZ\right|^2_{\omega_\phi} e^{-kQ_x-\psi_x}\frac{\omega_\phi^n}{n!}\leq Ce^{-ka(x)-k\eta}<\infty.
\end{equation*}
By Theorem \ref{demailly}, we have a $(n,0)$-form $v_k$ with $\bar\partial v_k=g_k$ and
\begin{equation}\label{l2bound}
    \|v_k\|^2_{ka}\leq  \|v_k\|^2_{\Psi_k} \leq Ce^{-ka(x)-k\eta}.
\end{equation}
The first inequality uses $\psi_x\leq 0$.

We claim that $v_k(x)=0$. Indeed, on a ball around $x$ where $\chi\equiv 1$, $g_k=0$ and $v_k$ is holomorphic. Write $v_k=f_kdZ$. Since $a$ is bounded near $x$ and $e^{-\psi_x(z)}\propto |z|^{-2(n+1)}$, \eqref{l2bound} implies
\begin{equation*}
    \int_{|z|<r}|f_k|^2|z|^{-2(n+1)}d\lambda(z) <\infty.
\end{equation*}
Hence $f_k$ must vanish at $x$ at least of order $2$. It follows that $s_k:=\sigma_k-v_k \in \mathcal{H}(ka)$ and its local coefficient at $x$ is $1$ since $H(0)=0$.

Now we compute the $L^2$-norm of $s_k$. First, by the change of variables $w=\sqrt{k}z$ and \eqref{rho},
\begin{equation}\label{sigmal2}
    \|\sigma_k\|^2_{ka}=e^{-ka(x)}\int_{\mathbb{C}^{n}} \chi^2 e^{-kQ_x}d\lambda=e^{-ka(x)}\frac{\pi^n}{k^n\det{(\partial_j\bar\partial_k a(x))}}(1+o(1)).
\end{equation}
Moreover, \eqref{l2bound} and \eqref{sigmal2} imply
\begin{equation*}
    \frac{\|v_k\|_{ka}}{\|\sigma_k\|_{ka}}=O(k^{n/2}e^{-k\eta/2})=o(1).
\end{equation*}
By the triangle inequality, we obtain
\begin{equation*}
    \|s_k\|^2_{ka} \leq e^{-ka(x)}\frac{\pi^n}{k^n\det{(\partial_j\bar\partial_k a(x))}}(1+o(1)).
\end{equation*}
Since $s_k(x)=1$ in local coordinates, we have \eqref{liminf}. The convergence of Bergman measures now follows from the uniform upper bound on compact sets \eqref{uniformupper}, almost everywhere pointwise density convergence provided by \eqref{pointwiseupper} and \eqref{liminf}, and the dominated convergence theorem.

        -k\operatorname{Tr}T_{kp,r}\leq \log \mathcal{D}_{k,p}(r)\leq 0.

        -\log B=-\log(P_{kp}M_{e^{-kr}}) \leq -P_{kp}\log(M_{e^{-kr}})=kT_{kp,r}.

\end{proof}

\end{align*}
The last equality follows from the orthogonality relation \eqref{orthogonality}.

<!-- proposition-4: body_md -->
\label{thmexptight}
The family of probability measures $\{\Gamma_k\}$ on $M$ is exponentially tight at speed $r_k$.
<!-- end proposition-4 -->
<!-- proof-15: body_md -->
Choose compact sets $K_j \Subset X$ of $X$ such that $K_j \subset K_{j+1}^\circ$, $\bigcup_{j\geq 0} K_j=X$. Choose $\chi_j\in C_c^\infty(K_{j+1})$ with $\chi_j\geq 0$ and equal to $1$ on a neighborhood of $K_j$. Set
    \begin{equation}\label{bj}
        b_j:=\sup_{k\geq 1}\frac{1}{r_k}\log \mathbb{E}\exp\left(r_k\langle \chi_j,L_k \rangle\right).
    \end{equation}
    We first show that $b_j$ is finite. For fixed $k$, we have (by \eqref{logmomgenformula})
    \begin{equation*}
        \mathbb{E}\exp\left(r_k\langle \chi_j,L_k \rangle\right)= \mathbb{E}\exp\left(k\langle \chi_j,\Theta_k \rangle\right)=\det_{\mathcal{H}(k\phi)} \left(I+T_{k\phi,e^{k\chi_j}-1}\right)<\infty.
    \end{equation*}

Hence they are bounded and $b_j<\infty$ (they are trivially bounded on finite interval $k\in[1,R]$). Moreover, $\chi_j \geq 0$ and $L_k \geq 0$ imply that $b_j \geq 0$. Therefore, we have
\begin{equation}\label{222}
    \mathbb{E}e^{r_k\langle \chi_j,L_k \rangle} \leq e^{r_kb_j}
\end{equation}
for all $k, j$.

For $\alpha>0$, define
\begin{equation*}
    \mathcal{C}_\alpha:=\{ \mu\in M_+ : \langle \chi_j,\mu\rangle \leq b_j+\alpha+j \;\,\text{for all}\;\, j\geq 0 \}.
\end{equation*}
By Theorem \ref{thmcompact}, each $\mathcal{C}_\alpha$ is a compact subset of $M_+$, hence a compact subset of $M$. By Markov's inequality and \eqref{222}, we have 
\begin{align*}
    \Gamma_k(M\setminus \mathcal{C}_\alpha) &\leq \sum_j \mathbb{P}\left[ \langle \chi_j, L_k \rangle >b_j+\alpha+j\right] = \sum_j \mathbb{P}\left[ e^{r_k\langle \chi_j, L_k \rangle} > e^{r_k(b_j+\alpha+j)}\right]\\
    &\leq \sum_j e^{-r_k(b_j+\alpha+j)}\mathbb{E}e^{r_k\langle \chi_j, L_k \rangle}\leq \sum_j e^{-r_k(\alpha+j)}=\frac{e^{-r_k\alpha}}{1-e^{-r_k}}.
\end{align*}
Consequently,
\begin{equation*}
    \limsup_{k\rightarrow\infty}\frac{1}{r_k}\log \Gamma_k(M\setminus \mathcal{C}_\alpha) \leq -\alpha,
\end{equation*}
and this completes the proof.

<!-- theorem-9: body_md -->
\label{thmmain}
Let $X$ be a Stein manifold with a plurisubharmonic exhaustion function $\phi$ satisfying the Condition $\mathfrak{A}$. Let $\Theta_k$ be the determinantal point process associated with the weighted Bergman space of holomorphic $n$-forms with weight $e^{-k\phi}$, defined in Definition \ref{definitionDPP}. Set

\end{equation*}
The laws of $L_k$, on the space of positive Radon measures with the vague topology, satisfy a large deviation principle with speed $r_k=k^{n+1}$ and a good rate function
\begin{equation*}
    I_\phi(\mu)=\sup_{u\in C_c^\infty(X)}\biggl\{ \mathcal{E}_\phi(P(\phi+u))-\int_X ud\mu \biggr\}.
\end{equation*}
The unique zero of $I_\phi$ is
\begin{equation*}
    \mu_\phi=\operatorname{MA}(\phi)=\frac{(dd^c \phi)^n}{n!}.
\end{equation*}
