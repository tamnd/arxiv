<!-- theorem-1: body_md -->
\label{thm:classification} Up to the action by the modular group,
there are three thresholds $0<\alpha_a<\alpha_b<\alpha_c<1$ such that
\begin{equation}\aligned\nonumber
\operatorname*{minima}_{z\in\mathbb H,\,(a,b)\in\mathbb R^2}\Big(\theta(1;z)+\alpha J(z;a,b)\Big)=
\begin{cases}
\;\big(e^{i{\pi}/{3}};1/3,1/3\big), &\hbox{if}\;\; \alpha\in[0,\alpha_a),\\
\;\big(e^{i{\pi}/{3}};1/3,1/3\big)\;\hbox{or}\;\big(e^{i{\varphi_{\alpha_a}}};1/2,1/2\big), &\hbox{if}\;\; \alpha=\alpha_a,\\
\;\big(e^{i{\varphi_\alpha}};1/2,1/2\big),\;\varphi_\alpha\in(\varphi_{\alpha_a},\frac{\pi}{2}),\;\; &\hbox{if}\;\; \alpha\in(\alpha_a,\alpha_b),\\
\;\big(i;1/2,1/2\big),\; &\hbox{if}\;\; \alpha\in[\alpha_b,\alpha_c],\\
\;\big(iy_\alpha;1/2,1/2\big),\; y_\alpha\in(1,\sqrt3],\;\;y_1=\sqrt3,\;\;&\hbox{if}\;\; \alpha\in(\alpha_c,1],
\end{cases}
\endaligned\end{equation}
More precisely, all minimizing configurations are classified as follows:
\begin{enumerate}
\item If $\alpha\in(0,\alpha_a)$, the lattice is triangular and
      $(a,b)=(1/3,1/3)$.
\item If $\alpha=\alpha_a$, the transition

\label{Def}
The fundamental domain associated to group $G$ is a connected domain $\mathcal{D}$ that satisfies:
\begin{itemize}
  \item [(1)] For any $z\in\mathbb{H}$, there exists an element $\pi\in G$ such that $\pi(z)\in\overline{\mathcal{D}}$;

<!-- proposition-1: body_md -->
\label{Prop21}
Let $\alpha\in{\mathbb R}$, $z\in\mathbb H$, and
$(a,b)\in\mathbb R^2$. Then
\begin{equation}\label{Eq21}
\begin{aligned}
 &E_\alpha(z+1;a-b,b)=E_\alpha(z;a,b),\\
 &E_\alpha(-1/z;-b,a)=E_\alpha(z;a,b),\\
 &E_\alpha(-\bar z;a,-b)=E_\alpha(z;a,b).
\end{aligned}
\end{equation}
and
\begin{equation}\label{Eq22}\begin{aligned}
 &E_\alpha(z;a+1,b)=E_\alpha(z;a,b),\\
  &E_\alpha(z;a,b+1)=E_\alpha(z;a,b),\\
  &E_\alpha(z;1-a,1-b)=E_\alpha(z;a,b)
\end{aligned}\end{equation}

Whenever a minimum is attained,
\begin{equation*}\begin{aligned}
\min_{z\in\mathbb{H}, (a,b)\in\mathbb{R}^2} E_\alpha(z;a,b)
=\min_{z\in\overline{\mathcal{D}_{\mathcal{G}}}, (a,b)\in\mathbb{R}^2} E_\alpha(z;a,b)
=\min_{z\in\overline{\mathcal{D}_{\mathcal{G}}}, (a,b)\in[0,1]^2} E_\alpha(z;a,b).
\end{aligned}\end{equation*}
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Write $z=x+iy$. The series
\[
 E_\alpha(z;a,b)=\sum_{m,n\in\mathbb Z}
 e^{-\pi |mz-n|^2/y}
 \bigl[1+\alpha\cos\bigl(2\pi(ma+nb)\bigr)\bigr]
\]
converges absolutely, since $|1+\alpha\cos t|\leq1+|\alpha|$.
Using
\[
 \operatorname{Im}(-1/z)=\frac{y}{|z|^2},\qquad
 \frac{|m(-1/z)-n|^2}{\operatorname{Im}(-1/z)}=\frac{|nz+m|^2}{y},\qquad
 |m(-\bar z)-n|^2=|mz+n|^2,
\]
we obtain
\[
\begin{aligned}
 E_\alpha(z+1;a-b,b)
 &=\sum_{m,n\in\mathbb Z}e^{-\pi |mz-(n-m)|^2/y}
   \bigl[1+\alpha\cos\bigl(2\pi(ma+(n-m)b)\bigr)\bigr]\\
 &\overset{k=n-m}{=}E_\alpha(z;a,b),\\[4pt]
 E_\alpha(-1/z;-b,a)
 &=\sum_{m,n\in\mathbb Z}e^{-\pi |nz+m|^2/y}
   \bigl[1+\alpha\cos\bigl(2\pi(na-mb)\bigr)\bigr]\\
 &\overset{(p,q)=(n,-m)}{=}E_\alpha(z;a,b),\\[4pt]
 E_\alpha(-\bar z;a,-b)
 &=\sum_{m,n\in\mathbb Z}e^{-\pi |mz+n|^2/y}
   \bigl[1+\alpha\cos\bigl(2\pi(ma-nb)\bigr)\bigr]\\
 &\overset{k=-n}{=}E_\alpha(z;a,b).
\end{aligned}
\]
The three changes of indices are bijections of $\mathbb Z^2$;
this proves \eqref{Eq21}. They also give the corresponding
invariances of $\theta(s;z)$ for every $s>0$.

For $p,q\in\mathbb Z$, we have
\[
\begin{aligned}
 \cos\bigl(2\pi(m(a+p)+n(b+q))\bigr)
 &=\cos\bigl(2\pi(ma+nb)\bigr),\\
 \cos\bigl(2\pi(m(1-a)+n(1-b))\bigr)
 &=\cos\bigl(2\pi(m+n)-2\pi(ma+nb)\bigr)\\
 &=\cos\bigl(2\pi(ma+nb)\bigr).
\end{aligned}
\]
These identities prove \eqref{Eq22} term by term.

The fundamental-domain property of \eqref{Fd3} and
\eqref{Eq21}--\eqref{Eq22} imply
\[
\begin{gathered}
 \forall\,(z;a,b)\in\mathbb H\times\mathbb R^2,\quad
 \exists\,(z';a',b')\in
 \overline{\mathcal D_{\mathcal G}}\times[0,1]^2:\\
 E_\alpha(z';a',b')=E_\alpha(z;a,b).
\end{gathered}
\]
Hence the three infima agree for every real $\alpha$, and the
minima agree whenever attained. For $0\leq\alpha\leq1$,
attainment follows from Lemma~\ref{lem:cusp}.

\label{lem:cusp}
For every $0\leq\alpha\leq1$, the unrestricted minimum of $E_\alpha$ is

By Proposition~\ref{Prop21}, it suffices to consider
$z\in\overline{\mathcal D_{\mathcal G}}$ and $(a,b)\in[0,1]^2$.
For $t>0$ and $u\in\mathbb R$, the Jacobi product gives
\[
\begin{aligned}
 \vartheta(t;u)
 &=\prod_{k\geq1}(1-e^{-2\pi kt})
 \bigl[(1-e^{-\pi(2k-1)t})^2
       +4e^{-\pi(2k-1)t}\cos^2(\pi u)\bigr]\\
 &\geq\vartheta_4(t)>0.
\end{aligned}
\]
Poisson summation and \eqref{eq:positive-translated} yield
\[
\begin{aligned}
 \theta(1;x+iy)
 &=\sqrt y\sum_{m\in\mathbb Z}e^{-\pi ym^2}\vartheta(y;mx)\\
 &\geq\sqrt y\,[\vartheta_3(y)+\vartheta_4(y)(\vartheta_3(y)-1)],\\
 J(x+iy;a,b)
 &=\sqrt y\sum_{n\in\mathbb Z}e^{-\pi y(n+b)^2}
       \vartheta\bigl(y;a+x(n+b)\bigr)\\
 &\geq\sqrt y\,\vartheta_4(y)
       \sum_{n\in\mathbb Z}e^{-\pi y(n+b)^2}.
\end{aligned}
\]
Moreover,
\[
 \sum_{n\in\mathbb Z}e^{-\pi y(n+b)^2}
 =\frac1{\sqrt y}\vartheta(1/y;b)
 \geq\frac1{\sqrt y}\vartheta_4(1/y)=\vartheta_2(y).
\]
For $y\geq2$, the bounds
\[
 \vartheta_3(y)\geq1+2e^{-\pi y},\qquad
 \vartheta_4(y)\geq1-2e^{-\pi y}>0,\qquad
 \vartheta_2(y)\geq2e^{-\pi y/4}
\]
give
\[
 E_\alpha(x+iy;a,b)\geq
 \sqrt y\,[1+4e^{-\pi y}(1-e^{-\pi y})
               +2\alpha e^{-\pi y/4}(1-2e^{-\pi y})].
\]
Since
\[
\begin{aligned}
 \frac d{dy}\bigl[(2\pi y-1)e^{-\pi y}\bigr]
 &=\pi e^{-\pi y}(3-2\pi y)<0,\\
 \frac d{dy}\bigl[(\pi y-2)e^{-\pi y/4}\bigr]
 &=\frac\pi4e^{-\pi y/4}(6-\pi y)<0,
\end{aligned}
\]
we obtain, for $0\leq\alpha\leq1$,

\begin{aligned}
 &2\sqrt y\,\frac d{dy}
 \left\{\sqrt y\,[1+4e^{-\pi y}(1-e^{-\pi y})
                 +2\alpha e^{-\pi y/4}(1-2e^{-\pi y})]\right\}\\
 &\quad=1-4(2\pi y-1)e^{-\pi y}+4(4\pi y-1)e^{-2\pi y}\\
 &\qquad\quad-\alpha(\pi y-2)e^{-\pi y/4}
               +2\alpha(5\pi y-2)e^{-5\pi y/4}\\
 &\quad\geq1-4(2\pi y-1)e^{-\pi y}-(\pi y-2)e^{-\pi y/4}\\
 &\quad\geq1-4(4\pi-1)e^{-2\pi}-(2\pi-2)e^{-\pi/2}>\frac1{50}.
\end{aligned}
\]

Thus the displayed lower bound is strictly increasing in $y$.
For $t>0$, the geometric and alternating series give
\[
 \vartheta_3(t)\leq1+\frac{2e^{-\pi t}}{1-e^{-3\pi t}},\qquad
 \vartheta_4(t)\leq1-2e^{-\pi t}+2e^{-4\pi t},\qquad
 \vartheta_2(t)\leq\frac{2e^{-\pi t/4}}{1-e^{-2\pi t}}.
\]

\[
\begin{aligned}
 \theta(1;i\sqrt3)
 &=3^{1/4}\vartheta_3(\sqrt3)^2
 <\frac{71}{50}<\sqrt2\,[1+4e^{-2\pi}(1-e^{-2\pi})],\\
 E_1(i\sqrt3;1/2,1/2)
 &=3^{1/4}\bigl[\vartheta_3(\sqrt3)^2
       +\vartheta_4(\sqrt3)\vartheta_2(\sqrt3)\bigr]\\

 <\sqrt2\,[1+4e^{-2\pi}(1-e^{-2\pi})
                 +2e^{-\pi/2}(1-2e^{-2\pi})].
\end{aligned}
\]

\[
\begin{aligned}

 &\geq\sqrt2\,[1+4e^{-2\pi}(1-e^{-2\pi})
                   +2\alpha e^{-\pi/2}(1-2e^{-2\pi})]\\
 &>(1-\alpha)\theta(1;i\sqrt3)
       +\alpha E_1(i\sqrt3;1/2,1/2)\\
 &=E_\alpha(i\sqrt3;1/2,1/2).
\end{aligned}
\]
The Gaussian series converge uniformly on
\[
 \{z\in\overline{\mathcal D_{\mathcal G}}:\operatorname{Im} z\leq2\}
 \times[0,1]^2.
\]
Continuity gives a minimum, and the strict comparison excludes $y\geq2$.

<!-- lemma-4: body_md -->
\label{lem:four}
Let $0\leq\alpha\leq1$ and $z=x+iy\in\mathbb H$.
For $(m,n)\in\mathbb Z^2$, define
\begin{equation}\label{eq:lattice-quadratic-form}
 Q_{m,n}=Q_{m,n}(x,y):=\frac{|mz-n|^2}{y}
                       =m^2y+\frac{(mx-n)^2}{y},
\end{equation}
and put
\[
 A=e^{-\pi(y+x^2/y)},\quad B=e^{-\pi/y},\quad
 C=e^{-\pi(y+(x-1)^2/y)},\quad D=e^{-\pi(y+(x+1)^2/y)}.
\]
Assume that
\[
 x\geq0,\qquad A^2>4CD,\qquad B^2>4CD.
\]
Then, for every $(a,b)\in\mathbb R^2$,
\begin{align}\label{eq:lower}
 E_\alpha(z;a,b)\geq{}&1+\alpha+2(A+B+C+D)\\
 &+2\alpha\min_{-1\leq t\leq1}
 \left[Bt-\sqrt{A^2+(C-D)^2+2A(C+D)t+4CDt^2}\right]\nonumber\\
 &+(1-\alpha)
 \sum_{\substack{|m|,|n|\leq2\\
 (m,n)\notin\{(0,0),\pm(1,0),\pm(0,1),\pm(1,1),\pm(1,-1)\}}}
 e^{-\pi Q_{m,n}}.\nonumber
\end{align}
The right-hand side is locally Lipschitz in $(x,y)$, and the scalar
minimum is attained uniquely at
\[
\begin{aligned}
 t&=\max\{-1,\min\{1,t_0\}\},\\
 t_0&=
 \frac{A^2(C+D)^2-B^2[A^2+(C-D)^2]}
 {\sqrt{B^2-4CD}
 [B(C-D)\sqrt{A^2-4CD}+A(C+D)\sqrt{B^2-4CD}]}.
\end{aligned}
\]
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
With $u=2\pi a$ and $v=2\pi b$, the four retained pairs contribute
\[
 A\cos u+B\cos v+C\cos(u+v)+D\cos(u-v).
\]
For $t=\cos v$, minimization in $u$ gives precisely the minimum
in \eqref{eq:lower}. Every omitted coefficient of $E_\alpha$ is
$1+\alpha\cos(2\pi(ma+nb))\geq1-\alpha\geq0$.
Retaining the stated finite remainder proves the inequality.

Let $f(t)$ denote the function inside the minimum. The assumptions give
\[
 C/D=e^{4\pi x/y}\geq1,\qquad
 A(C+D)\geq2A\sqrt{CD}>4CD.
\]
If $x=0$, then $C=D$, $A>2C$, and $B>2C$. Thus
\[
 f(t)=(B-2C)t-A,\qquad t_0=-\frac{A}{2C}<-1,
\]
and the unique minimum is at $t=-1$, as asserted.
Assume henceforth that $C>D$. The radicand is positive for
$-1<t<1$, and
\[
 f''(t)=
 \frac{(C-D)^2(A^2-4CD)}
 {[A^2+(C-D)^2+2A(C+D)t+4CDt^2]^{3/2}}>0
 \qquad(-1<t<1).
\]
Thus $f$ is strictly convex on $[-1,1]$.
Solving $f'(t)=0$ on the branch containing $[-1,1]$ gives $t=t_0$.
Strict convexity then gives the unique constrained minimizer
$t=\max\{-1,\min\{1,t_0\}\}$.
The interior stationary value is
\[
 -\frac{A^2B^2+(C-D)^2(A^2+B^2-4CD)}
 {AB(C+D)+(C-D)\sqrt{(A^2-4CD)(B^2-4CD)}}.
\]
If $f'(-1)\geq0$, the minimum equals
$-B-|A-C-D|$. When the unconstrained stationary point lies outside
$[-1,1]$, its value is still a valid lower bound for the constrained
minimum; the projected point gives the exact value.
These formulas avoid cancellation as $D$ tends to zero.

At a minimizing $t$ for which
$A^2+(C-D)^2+2A(C+D)t+4CDt^2>0$, the corresponding cosines are
\[
\begin{aligned}
 \cos u&=-\frac{A+(C+D)t}
 {\sqrt{A^2+(C-D)^2+2A(C+D)t+4CDt^2}},&\cos v&=t,\\
 \cos(u+v)&=\frac{-[A+(C+D)t]t-(C-D)(1-t^2)}
 {\sqrt{A^2+(C-D)^2+2A(C+D)t+4CDt^2}},\\
 \cos(u-v)&=\frac{-[A+(C+D)t]t+(C-D)(1-t^2)}
 {\sqrt{A^2+(C-D)^2+2A(C+D)t+4CDt^2}}.
\end{aligned}
\]
The shape derivative of the minimized contribution is obtained by
multiplying the derivatives of $A,B,C,D$ by
$2(1+\alpha\cos u)$ and the three analogous factors.
The finite remainder is differentiated termwise.

To justify the derivative bounds at changes of branch, the minimum is
taken over a compact displacement torus of smooth functions whose shape
derivatives are bounded on compact shape rectangles. It is therefore
locally Lipschitz. Along any line segment it is absolutely continuous,
and its derivative almost everywhere is bounded by the directional
derivatives at the minimizing displacements. The displayed cosine
formulas enclose those derivatives; when the radicand interval meets
zero, using $[-1,1]$ for every cosine remains valid.
Thus integrating the resulting derivative bounds along segments gives
rigorous shape-box estimates, including boxes meeting a branch change.

 &=1+2\{A\cos U+B\cos V+C\cos(U+V)\}+\mathcal R(a,b)\\
 &=1-P^2-Q^2-R^2+|P+Qe^{iU}+Re^{-iV}|^2
       +\mathcal R(a,b).\label{eq:hex-phasor}
\end{align}
The strict triangle inequalities for $P,Q,R$, verified in
Lemma~\ref{lem:cert-localization}, imply that the last square vanishes
at exactly two points on the displacement torus.
Denote them by $\pm(a_0,b_0)$, choosing
$U_0=2\pi a_0$, $V_0=2\pi b_0$ in $(0,\pi)$. They are
determined by the exact identities
\begin{equation}\label{eq:hex-phasor-centers}
\begin{aligned}
 \cos U_0&=\frac{R^2-P^2-Q^2}{2A},&
 \cos V_0&=\frac{Q^2-P^2-R^2}{2B},\\
 \cos(U_0+V_0)&=\frac{P^2-Q^2-R^2}{2C}.
\end{aligned}
\end{equation}
Indeed, the first two equations determine the angles in $(0,\pi)$;
the triangle identities give $Q\sin U_0=R\sin V_0$ and
$P+Q\cos U_0+R\cos V_0=0$.

For fixed $(x,y)$, set
\[
 \operatorname{osc}_{a,b}\mathcal R
 :=\max_{(a,b)\in[0,1]^2}\mathcal R(a,b)
   -\min_{(a,b)\in[0,1]^2}\mathcal R(a,b),
\]
\[
 \nabla_{a,b}\mathcal R=(\partial_a\mathcal R,\partial_b\mathcal R),
 \qquad
 \|\nabla_{a,b}\mathcal R\|_2
 :=\sqrt{|\partial_a\mathcal R|^2+|\partial_b\mathcal R|^2}.
\]
Lemma~\ref{lem:cert-localization} gives
\begin{equation}\label{eq:hex-cap-tail}
 \operatorname{osc}_{a,b}\mathcal R<\frac{27}{100000},\qquad
 \sup_{(a,b)\in[0,1]^2}\|\nabla_{a,b}\mathcal R(a,b)\|_2<\frac1{625},
\end{equation}
together with
\begin{equation}\label{eq:hex-cap-coarse}
\begin{gathered}
 A>\frac1{40},\quad B>\frac1{38},\quad C>\frac1{44},\qquad
 P<\frac{19}{100},\quad Q<\frac16,\quad R<\frac7{40},\\
 -\frac{16}{25}<\cos U_0<0,\qquad
 -\frac23<\cos V_0<0,\qquad \cos(U_0+V_0)<-\frac29.
\end{gathered}
\end{equation}
Lemma~\ref{lem:cert-localization} establishes these bounds uniformly
on the cap. In particular, $U_0,V_0\in(\pi/2,3\pi/4)$.

 &>\frac{33}{100}\left(\frac7{20}+\frac{33}{2000}\right),\\

 |r_b|<\frac7{2000}+\frac4{1625}<\frac3{500}.

<!-- proof-9: body_md -->
Fix $\alpha\in[4/25,7/40]$, and regard $E_\alpha$ as a function
of the four real variables $(x,y,a,b)$.
It attains its minimum on the closed shape rectangle in the statement
times the displacement torus. Choose such a minimizing configuration
$p=(x,y,a,b)$. Since $\alpha>0$, its displacement minimizes $J$
at the fixed shape. Lemma~\ref{lem:hex-shift-localization} places
$p$, after simultaneous displacement inversion and integer
translations, in the affine prism \eqref{eq:hex-affine-prism}.
This prism is convex and contains
\[
 p_0=\left(\frac12,\frac{\sqrt3}{2},\frac13,\frac13\right).
\]

The triangular configuration is stationary in all four variables.
Indeed, the joint modular symmetry and integer periodicity give
\[
 E_\alpha\circ\Phi=E_\alpha,\qquad
 \Phi(z;a,b)=\left(1-\frac1z;1-a-b,a\right),\qquad
 \Phi(p_0)=p_0.
\]
At $p_0$, the shape derivative is multiplication by
$e^{-2\pi i/3}$, and the displacement derivative is
$\left(\begin{smallmatrix}-1&-1\\1&0\end{smallmatrix}\right)$.
Thus $\det(I-D\Phi(p_0))=9$, and differentiation yields
\[
 (I-D\Phi(p_0)^{\mathsf T})\nabla E_\alpha(p_0)=0,
 \qquad \nabla E_\alpha(p_0)=0.
\]

Suppose that $p\ne p_0$. If the minimum is attained in the
interior of the shape rectangle, then $\nabla E_\alpha(p)=0$.
Lemma~\ref{lem:hex-convexity} and integration give the contradiction
\[
\begin{aligned}
 0&=(p-p_0)\cdot[\nabla E_\alpha(p)-\nabla E_\alpha(p_0)]\\
 &=\int_0^1(p-p_0)^{\mathsf T}
 D^2E_\alpha(p_0+t(p-p_0))(p-p_0)\,dt>0.
\end{aligned}
\]
If $p$ lies on the shape boundary, convexity keeps
$p-s(p-p_0)$ feasible for $0\le s\le1$. Minimality gives
\[
\begin{aligned}
 0&\le\left.\frac d{ds}E_\alpha(p-s(p-p_0))\right|_{s=0^+}\\
 &=-\int_0^1(p-p_0)^{\mathsf T}
 D^2E_\alpha(p_0+t(p-p_0))(p-p_0)\,dt<0,
\end{aligned}
\]
again a contradiction. Every minimum is therefore the triangular
deep-hole configuration, up to inversion and integer translations.
The two deep holes have the same energy by inversion, proving the
comparison and all equality cases.

<!-- proof-10: body_md -->
Since $y\geq\sqrt3/2$, we have $e^{-\pi y}<1/15$. The
alternating series and the estimates
$n^2\geq4+5(n-2)$, $n^2\leq4\,3^{n-2}$ for $n\geq2$ give
\begin{equation}\label{eq:large-alpha-tails}
 \vartheta_4(y)\geq1-2e^{-\pi y},\qquad
 \sum_{n\geq1}n^2e^{-\pi yn^2}
 \leq e^{-\pi y}(1+5e^{-3\pi y}).
\end{equation}
Indeed,
\[
 \sum_{n\geq2}n^2e^{-\pi yn^2}
 \leq\frac{4e^{-4\pi y}}{1-3e^{-5\pi y}}<5e^{-4\pi y}.
\]
The Jacobi product implies that $\vartheta(y;\cdot)$ has its
unique minimum $\vartheta_4(y)$ at $1/2$ modulo $\mathbb Z$.
More precisely,
\begin{equation}\label{eq:quantified-one-d}
 \vartheta(y;t)\geq\vartheta_4(y)
 +4e^{-\pi y}(1-5e^{-3\pi y})\cos^2(\pi t).
\end{equation}
The $k=1$ pair contributes $4e^{-\pi y}\cos^2(\pi t)$ to the
difference. Every odd $k$ contributes a nonnegative quantity,
whereas for even $k$, $|\sin(ku)|\leq k|\sin u|$ bounds its
contribution below by
$-4k^2e^{-\pi yk^2}\cos^2(\pi t)$.
Thus \eqref{eq:large-alpha-tails} proves
\eqref{eq:quantified-one-d}.

Rowwise Poisson summation gives
\[
 \theta(1;x+iy)=\sqrt y\sum_{m,k\in\mathbb Z}
 e^{-\pi y(m^2+k^2)}\cos(2\pi mkx).
\]
Hence
\begin{equation}\label{eq:theta-shear-loss}
\begin{aligned}
 0\leq\theta(1;iy)-\theta(1;x+iy)
 &=8\sqrt y\sum_{m,k\geq1}
 e^{-\pi y(m^2+k^2)}\sin^2(\pi mkx)\\
 &\leq8\sqrt y\,e^{-2\pi y}(1+5e^{-3\pi y})^2
 \sin^2(\pi x).
\end{aligned}
\end{equation}
The translated representation of $J$, followed by horizontal
Poisson summation, yields
\[
 J(x+iy;a,b)=\sqrt y\sum_{n\in\mathbb Z}
 e^{-\pi y(n+b)^2}\vartheta\bigl(y;a+x(n+b)\bigr),
 \quad
 J(iy;1/2,1/2)=\sqrt y\,\vartheta_4(y)\vartheta_2(y).
\]
By periodicity, take $0\leq b\leq1$. The Jacobi product after
one-dimensional Poisson summation shows that
$\sum_ne^{-\pi y(n+b)^2}$ decreases strictly on $[0,1/2]$,
is symmetric about $1/2$, and has minimum $\vartheta_2(y)$.
Retaining the $n=0,-1$ corrections in
\eqref{eq:quantified-one-d} gives
\begin{equation}\label{eq:two-row-penalty}
\begin{aligned}
 &J(x+iy;a,b)-\sqrt y\,\vartheta_4(y)\vartheta_2(y)\\
 &\quad\geq\sqrt y\left[
 \vartheta_4(y)\left(\sum_{n\in\mathbb Z}e^{-\pi y(n+b)^2}
 -\vartheta_2(y)\right)
 +\frac{4e^{-\pi y}(1-5e^{-3\pi y})}
 {e^{\pi yb^2}+e^{\pi y(1-b)^2}}\sin^2(\pi x)\right].
\end{aligned}
\end{equation}
Here, for $\lambda,\mu>0$,
\[
 \min_u\{\lambda\cos^2u+\mu\cos^2(u-\pi x)\}
 =\frac{\lambda+\mu-
 \sqrt{(\lambda+\mu)^2-4\lambda\mu\sin^2(\pi x)}}2
 \geq\frac{\lambda\mu}{\lambda+\mu}\sin^2(\pi x).
\]

Suppose first that $\sqrt3/2\leq y\leq1$. Since
$e^{-\pi/y}<1/23$, Poisson summation and
\eqref{eq:quantified-one-d}, applied at $1/y$, imply
\[
 \sum_ne^{-\pi y(n+b)^2}-\vartheta_2(y)
 \geq\frac{4e^{-\pi/y}(1-5e^{-3\pi/y})}{\sqrt y}
 \sin^2\bigl(\pi(b-1/2)\bigr).
\]
Moreover,
\[
 \frac1{e^{\pi yb^2}+e^{\pi y(1-b)^2}}
 =\frac{e^{-\pi y/4}}2
 \frac{e^{-\pi y(b-1/2)^2}}{\cosh\bigl(\pi y(b-1/2)\bigr)}
 \geq\frac{e^{-\pi y/4}}2
 \cos^2\bigl(\pi(b-1/2)\bigr).
\]
Indeed, $\log\cosh u\leq u^2/2$, $\log\cos u\leq-u^2/2$
for $|u|<\pi/2$, and
$\pi y+\pi^2y^2/2\leq\pi+\pi^2/2<\pi^2$; the endpoints
follow by continuity. Also,
\[
 \frac{4\vartheta_4(y)e^{-\pi/y}(1-5e^{-3\pi/y})}{\sqrt y}
 >2e^{-5\pi y/4}(1-5e^{-3\pi y}),
\]
because
\[
 e^{-\pi(5y/4-1/y)}
 \leq e^{\pi/(8\sqrt3)}<\frac43
 <\frac53<2\frac{13}{15}\left(1-\frac5{23^3}\right).
\]
Inserting these inequalities into \eqref{eq:two-row-penalty} gives
\begin{equation}\label{eq:low-height-penalty}
 J(x+iy;a,b)-\sqrt y\,\vartheta_4(y)\vartheta_2(y)
 \geq2\sqrt y\,e^{-5\pi y/4}(1-5e^{-3\pi y})\sin^2(\pi x).
\end{equation}
The ratio of the bound in \eqref{eq:theta-shear-loss} to that in

\[
 4e^{-3\pi y/4}\frac{(1+5e^{-3\pi y})^2}{1-5e^{-3\pi y}}
 <4\,15^{-3/4}\frac{(1+5/15^3)^2}{1-5/15^3}<\frac8{15}.
\]
This proves the comparison in this height range, strictly if $x\ne0$.

 Q_{m,n}&=\frac{m^2+n^2-2mnx}{\sqrt{1-x^2}},\\
 Q_{m,n}'&=-\frac{2mn}{\sqrt{1-x^2}}
     +\frac{x(m^2+n^2-2mnx)}{(1-x^2)^{3/2}},\\
 Q_{m,n}''&=\frac{-4mnx(1-x^2)+(1+2x^2)(m^2+n^2-2mnx)}
 {(1-x^2)^{5/2}}.
\end{aligned}
\]
For $3/10\le x\le2/5$, we have
\[
 Q_{m,n}\ge\frac9{10}\max\{|m|,|n|\}^2,\qquad
 |Q_{m,n}'|<10\max\{|m|,|n|\}^2,\qquad
 |Q_{m,n}''|<30\max\{|m|,|n|\}^2
\]
for $(m,n)\ne(0,0)$.
Consequently the value, first-derivative, and second-derivative
errors in each truncated theta sum are bounded by
\[
 10^{-27},\qquad10^{-22},\qquad10^{-19},
\]
respectively, by \eqref{eq:common-ring-tail}. At the triangular
shape, the triangular deep-hole factors are
\[
 \cos\frac{2\pi(m+n)}3=
 \begin{cases}1,&m+n\equiv0\pmod3,\\-1/2,&m+n\not\equiv0\pmod3,
 \end{cases}
\]
and the value error is below $10^{-25}$.

At $\alpha=1723/10000,7/40$, the arc derivative is negative
at $x=3/10$ and positive at $x=2/5$. Forty-two rational
bisections enclose each stationary point in an interval of length
$1/(10\cdot2^{42})$. The Luo--Wei classification identifies these
points as the centered minima. Finite-sum evaluation on these
intervals proves the two energy inequalities in the statement.

Finally, substitution in the left-hand side of
\eqref{eq:crossing-residual} gives values below $-5\cdot10^{-16}$
and above $10^{-15}$ at
$x=0.3754117719806$ and $x=0.3754117719808$, respectively.
Moreover,
\[
 \frac d{dx}J(x+i\sqrt{1-x^2};1/2,1/2)>\frac{17}{50}
 \qquad(0.3754117719806\le x\le0.3754117719808).
\]
On the same interval,
\[
 \left|\frac d{dx}\left[
 -\frac{\frac d{dx}\theta(1;x+i\sqrt{1-x^2})}
 {\frac d{dx}J(x+i\sqrt{1-x^2};1/2,1/2)}
 \right]\right|<\frac65.
\]
Midpoint evaluation and this derivative bound give the asserted
quotient enclosure. This proves the lemma.
