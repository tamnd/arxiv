\end{align*}

<!-- remark-3: body_md -->
Polynomial boundary traces reduce the number of local problems by
exciting an entire coarse face at a time. For an edge \((d=2)\) or a face \((d=3)\)
\(F\subset\partial T_i^m\), let
\(\{q_{F,r}\}_{r=1}^{(p+1)^{d-1}}\) be a basis of the tensor-product
polynomial trace space \(\mathbb Q_p(F)\).  For each trace \(q_{F,r}\), we compute
\((\boldsymbol\Phi_{i,F}^r,\psi_{i,F}^r)\) from
\begin{subequations}\label{polynomial-snapshot}
\begin{align}
\boldsymbol\Phi_{i,F}^r+\nabla\psi_{i,F}^r&=0
&&\text{in }T_i^m,\\
-k^2\psi_{i,F}^r+\nabla\!\cdot\boldsymbol\Phi_{i,F}^r&=0
&&\text{in }T_i^m,\\
\psi_{i,F}^r&=q_{F,r}
&&\text{on }F,\\
\frac{\partial\psi_{i,F}^r}{\partial n}-ik\psi_{i,F}^r&=0
&&\text{on }\partial T_i^m\setminus F.
\end{align}
\end{subequations}
The resulting alternative snapshot space is
\[
 W_{i,m}^{\mathrm{snap},p}
 :=\operatorname{span}\{\psi_{i,F}^r:
 F\subset\partial T_i^m,\ 1\le r\le(p+1)^{d-1}\}.
\]

Thus, the two constructions differ in the choice of boundary trace
space: the fine-grid trace construction resolves the boundary data at the
fine-grid level, whereas $W_{i,m}^{\mathrm{snap},p}$ employs a
low-dimensional polynomial trace space. In particular, for a fixed
boundary entity $F$, all polynomial traces share the same local matrix
and can be treated as multiple right-hand sides. Consequently, the
polynomial construction substantially reduces the number of local
factorizations, especially in three dimensions, while the subsequent
spectral reduction can be applied in the same manner as for the
snapshot space.

<!-- remark-4: body_md -->
For a globally PML-truncated problem, the same construction is applied to
the stretched fine-grid operator. We first enlarge the oversampled region
by an auxiliary local PML, solve the boundary-excited problem there, discard
the auxiliary-layer degrees of freedom, and restrict the resulting snapshots
to the target coarse element. The auxiliary layer supplies the absorbing
boundary treatment for local construction, while the global PML remains
part of the fine and reduced operators.

&\le
 k^2 h\|v\|_{0,\partial\Omega}^2
+h^2|v|_{2,\Omega}^2.
\label{eq:boundary-average-imp}
\end{align}

<!-- lemma-7: body_md -->
\label{lem:P0-D}
Let $z\in H^2(\Omega)$ be the solution to problem \eqref{eq:continuous-adjoint}. Then
\begin{align*}
    \|P_h^0z\|_D\le C_{T,k h}k\|w_h\|_D,
\end{align*} where $C_{T,k h}=2\sqrt{3}C_{\rm stab}(k h+1)\sqrt{C_I+\frac43 k h+2}.$
<!-- end lemma-7 -->
<!-- proof-2: body_md -->
We estimate separately the interior-edge, physical-boundary, and mass contributions to the $D$-norm. 

\emph{(i) Interior-edge.} By \cref{lem:flux} and the triangle inequality,
\begin{align*}
\left(\sum_{e\in\mathcal{E}_h^0}|\llbracket P_h^0z\rrbracket _e|^2\right)^{1/2}
&\le
2C_{\rm bh}h|z|_{2,\Omega}
+
\left(
\sum_{e\in\mathcal E_h^0}
\left|
\int_e\partial_{n_e}z\,ds
\right|^2
\right)^{1/2}.
\end{align*}
For each edge, Cauchy--Schwarz inequality gives $\left|\int_e\partial_{n_e}z\,ds\right|^2
\leq h\|\partial_{n_e}z\|_{0,e}^2
\leq h\|\nabla z\|_{0,e}^2.$ Applying  scaled trace inequality\cite[Lemma~1.49]{DiPietroErn2012} elementwise and observing that every interior edge is counted at most twice in the element-boundary sums yields $\sum_{e\in\mathcal{E}_h^0}\left|\int_e\partial_{n_e}z\,ds\right|^2
\le C_{\rm tr}(\|\nabla z\|_{0,\Omega}^2+h^2|z|_{2,\Omega}^2).$ Hence, for a constant $C_I$ depending only on $C_{\rm bh}$ and $C_{\rm tr}$,
\begin{align}
\sum_{e\in\mathcal{E}_h^0}|\llbracket P_h^0z\rrbracket_e|^2
&\leq C_I\left(\|\nabla z\|_{0,\Omega}^2+h^2|z|_{2,\Omega}^2\right).
\label{eq:D-int-bound}
\end{align}
\emph{(ii) Physical-boundary.} By Cauchy--Schwarz inequality, we have 
$$
|z_e^{\rm av}|^2
=\frac1{h^2}\left|\int_ez\,ds\right|^2
\leq \frac1{h^2}|e|\int_e|z|^2\,ds
=\frac1h\|z\|_{0,e}^2.
$$ 
Combining with \cref{cell-edge-average-H1}, for each boundary edge $e\subset\partial t$, 
\begin{align*}
|z_{t(e)}^{\rm av}|^2
\leq 2|z_e^{\rm av}|^2+2|z_{t(e)}^{\rm av}-z_e^{\rm av}|^2
\le \frac{2}{h}\|z\|_{0,e}^2+\frac{2}{3}\|\partial_{n_e}z\|_{0,t}^2.
\end{align*}
Each cell has at most two edges normal to either coordinate direction,
so summing over physical boundary edges gives
\begin{align}
|\alpha|\sum_{e\in\mathcal{E}_h^B}|z_{t(e)}^{\rm av}|^2
&\le \frac{2|\alpha|}{h}\|z\|_{0,\partial\Omega}^2
+\frac{2|\alpha|}{3}
\sum_{e\in\mathcal{E}_h^B}\|\partial_{n_e}z\|_{0,t(e)}^2\notag\\
&\le
\frac{4k}{\sqrt{(k h)^2+4}}\|z\|_{0,\partial\Omega}^2
+\frac{8k h}{3\sqrt{(k h)^2+4}}\|\nabla z\|_{0,\Omega}^2.
\label{physical-b}
\end{align}

\emph{(iii) Mass contribution.}
Since $P_h^0$ is the $L^2$-orthogonal projection, its contractivity gives
\begin{align}
k^2\|P_h^0z\|_{0,\Omega}^2
&\leq k^2\|z\|_{0,\Omega}^2.
\label{eq:D-mass-bound}
\end{align}

Combining \eqref{eq:D-int-bound}, \eqref{physical-b} and \eqref{eq:D-mass-bound}, we have $\|P_h^0z\|_D^2
\le (C_I+\frac43 k h+2) 
\|z\|_{E,k}^2+C_Ih^2|z|_{2,\Omega}^2.$  
Combining \eqref{eq:z-energy-w}, \eqref{eq:z-H2-w}   we obtain $\|P_h^0z\|_D
\le C_{T,k h}k\|w_h\|_D,$ where $C_{T,k h}=2\sqrt{3}C_{\rm stab}(k h+1)\sqrt{C_I+\frac43 k h+2}$.

<!-- lemma-8: body_md -->
\label{lem:cell-identity}
Let $z$ solve \eqref{eq:continuous-adjoint}. 
Then
\begin{align*}
\left|a_h(w_h,P_h^0z)-2k^2\|w_h\|_{0,\Omega}^2\right|
\le C_{\rm con,k h}k^2h\|w_h\|_D^2,
\end{align*}
where $C_{\rm con,k h}
:=4C_{\rm stab}C_{\rm bh}
+\frac{C_{T,k h}}{2}
+2\sqrt{3}C_{\rm stab}(2+k h
). $

<!-- proof-3: body_md -->
Take the complex conjugate of \eqref{eq:continuous-adjoint-a}, multiply it on each cell $t$ by the constant value $w_h|_t$, and integrate. This gives
\begin{align*}
2k^2\int_t|w_h|^2
&=-w_h|_t\int_{\partial t}\partial_{n_t}\overline z\,ds
-k^2w_h|_t\int_t\overline z\,dx.
\end{align*}
Summing over all elements, the contributions from the two cells adjacent to an interior edge combine as
\begin{align*}
-w_{t^-}\int_e\partial_{n_e}\overline z\,ds
+w_{t^+}\int_e\partial_{n_e}\overline z\,ds
&=-\llbracket w_h\rrbracket_e\overline{\int_e\partial_{n_e}z\,ds}.
\end{align*}
$\partial_nz+ik z=0$ implies $\partial_n\overline z-ik\overline z=0, $ 
and therefore $-w_h\int_e\partial_n\overline z\,ds
=-ik h\,\llbracket w_h\rrbracket_e\overline{z_e^{\rm av}}. $ 
Combining the interior and boundary contributions, we obtain
\begin{align}
2k^2\|w_h\|_{0,\Omega}^2
&=-\sum_{e\in\mathcal{E}_h^0}\llbracket w_h\rrbracket_e
\overline{\int_e\partial_{n_e}z\,ds}
-ik h\sum_{e\in\mathcal{E}_h^B}\llbracket w_h\rrbracket_e\overline{z_e^{\rm av}}
-k^2(w_h,z)_\Omega.
\label{eq:cell-identity}
\end{align}
By definition of $a_h(\cdot,\cdot)$, \eqref{eq:mass-exact} and \eqref{eq:cell-identity}, we have
\begin{align*}
 a_h(w_h,P_h^0z)-2k^2\|w_h\|_{0,\Omega}^2&=\sum_{e\in \mathcal{E}_h^0}\llbracket w_h \rrbracket_e \,\overline{\llbracket P_h^0z \rrbracket_e} 
+\alpha\sum_{e\in \mathcal{E}_h^{B}} \llbracket w_h \rrbracket_e \,\overline{\llbracket P_h^0z \rrbracket_e}-k^2( w_h,P_h^0z)_\Omega-2k^2\|w_h\|_{0,\Omega}^2\\
&=:R_1+R_2,
\end{align*}
where 
\begin{align*}
R_1
:=\sum_{e\in\mathcal{E}_h^0}\llbracket w_h\rrbracket_e
\overline{\left(\llbracket P_h^0z\rrbracket_e+\int_e\partial_{n_e}z\,ds\right)},\quad
R_2
:=\sum_{e\in\mathcal{E}_h^B}\llbracket w_h\rrbracket_e
\left(\alpha\overline{P_h^0z|_{t(e)}}+ik h\,\overline{z_e^{\rm av}}\right).
\end{align*}
By \cref{lem:flux} and \eqref{eq:z-H2-w},
\begin{align}
|R_1|
&\le 4C_{\rm stab}C_{\rm bh}k^2h\|w_h\|_D^2.
\label{eq:RI-bound}
\end{align}
For $R_2,$ we split it as
\begin{align*}
R_2
&=(\alpha+i k h)\sum_{e\in\mathcal{E}_h^B}\llbracket w_h\rrbracket_e\overline{P_h^0z|_{t(e)}}
+i k h\sum_{e\in\mathcal{E}_h^B}\llbracket w_h\rrbracket_e
\overline{(z_e^{\rm av}-P_h^0z|_{t(e)})}
\nonumber\\
&=:R_{2,1}+R_{2,2}.
\end{align*}
 
              
               
                                    
  
                            
                  
                          
            
Using  \cref{lem:P0-D},
\begin{align}
|R_{2,1}|
&\leq \frac {k h}2
\left(|\alpha|\sum_{e\in\mathcal{E}_h^B}|\llbracket w_h\rrbracket_e|^2\right)^{1/2}
\left(|\alpha|\sum_{e\in\mathcal{E}_h^B}|P_h^0z|_{t(e)}|^2\right)^{1/2}
\nonumber\\
&\le \frac{C_{T,k h}k^2h}{2}\|w_h\|_D^2.
\label{eq:RB1-bound}
\end{align}
                          
                                  
By \cref{lem:boundary-average}, \eqref{eq:z-energy-w} and \eqref{eq:z-H2-w}, we have
\begin{align*}
\sum_{e\in\mathcal{E}_h^B}|z_e^{\rm av}-P_h^0z|_{t(e)}|^2
&\le 12C_{\rm stab}^2(1+k h)k^3h
\|w_h\|_D^2.
\end{align*}
Using the Cauchy--Schwarz inequality gives
\begin{align}
|R_{2,2}|
&=|i k h\sum_{e\in\mathcal{E}_h^B}\llbracket w_h\rrbracket_e
\overline{(z_e^{\rm av}-P_h^0z|_{t(e)})}|\notag\\
&\leq \frac{k h}{\sqrt{|\alpha|}}\left(|\alpha|\sum_{e\in\mathcal{E}_h^B}|\llbracket w_h\rrbracket_e|^2\right)^{\frac{1}{2}}\left(\sum_{e\in\mathcal{E}_h^B}|z_e^{\rm av}-P_h^0z|_{t(e)}|^2\right)^{\frac{1}{2}}\notag\\
&\le \sqrt{6}C_{\rm stab}((4+(k h)^2)^\frac{1}{2}(1+k h))^{\frac{1}{2}} k^2h\|w_h\|_D^2\\
&\le  \sqrt{6}C_{\rm stab}(2+k h) k^2h\|w_h\|_D^2.
\label{eq:RB2-bound}
\end{align}
               
                           
                                                               
Combining \eqref{eq:RI-bound}, \eqref{eq:RB1-bound}, and \eqref{eq:RB2-bound}, we finish the proof.

<!-- theorem-1: body_md -->
\label{thm:main-infsup}
Assume that $C_{\rm con,k h}k^2h\leq \frac{\sqrt{2}}{4}.$ Then the sesquilinear
form $a_h(\cdot,\cdot)$ satisfies the discrete inf--sup condition
\begin{equation}
\beta_h
:=
\inf_{0\neq w_h\in W_h}
\sup_{0\neq v_h\in W_h}
\frac{|a_h(w_h,v_h)|}
{\|w_h\|_D\|v_h\|_D}
\ge
\frac{\sqrt{2}}{4(1+C_{T,k h}k)}.
\label{eq:fine-grid-infsup}
\end{equation}
<!-- end theorem-1 -->
<!-- proof-4: body_md -->
Fix $0\neq w_h\in W_h$. Let $z$ be the solution of
\eqref{eq:continuous-adjoint} and set
$v_h^\star:=w_h+P_h^0z$. By \cref{lem:cell-identity},
\begin{align*}
a_h(w_h,P_h^0z)
=
2k^2\|w_h\|_{0,\Omega}^2+R_1+R_2,
\quad
|R_1+R_2|
\leq
C_{\rm con,k h}k^2h\|w_h\|_D^2.
\end{align*}
Hence, under the assumption  $C_{\rm con,k h}k^2h\le\frac{\sqrt{2}}{4},$ 
\begin{align}
|a_h(w_h,v_h^\star)|
&=
|b_h(w_h,w_h)+R_1+R_2|
\nonumber\\
&\geq
|b_h(w_h,w_h)|-|R_1+R_2|
\nonumber\\
&\geq
\left(\frac{1}{\sqrt{2}}-C_{\rm con,k h}k^2h\right)
\|w_h\|_D^2
\geq
\frac{\sqrt{2}}{4}\|w_h\|_D^2.
\label{eq:numerator}
\end{align}
On the other hand, \cref{lem:P0-D} gives
\begin{align}
\|v_h^\star\|_D
&\leq
\|w_h\|_D+\|P_h^0z\|_D
\le
(1+C_{T,k h}k)\|w_h\|_D.
\label{eq:denominator}
\end{align}
Combining \eqref{eq:numerator} and \eqref{eq:denominator}, we obtain
\begin{align}
\sup_{0\neq v_h\in W_h}
\frac{|a_h(w_h,v_h)|}{\|v_h\|_D}
&\geq
\frac{|a_h(w_h,v_h^\star)|}{\|v_h^\star\|_D}
\geq
\frac{\sqrt{2}}{4(1+C_{T,k h}k)}\|w_h\|_D.
\end{align}
Dividing by $\|w_h\|_D$ and taking the infimum over all
$0\neq w_h\in W_h$ yields \eqref{eq:fine-grid-infsup}, completing the proof.

<!-- remark-7: body_md -->
The assumption in \cref{thm:main-infsup} takes the form $k^2 h \le C,$ where $C>0$ is independent of $k$ and $h$. Such a mesh-resolution
condition is consistent with classical stability and quasi-optimality
analyses for low-order Helmholtz finite element discretizations
\cite{IhlenburgBabuska1995,Sauter2006}. Refined pre-asymptotic error
estimates can be found in \cite{Wu2014}. 
For the present $RT_0$--$P_0$ scheme, this condition follows directly
from the discrete inf--sup argument.

<!-- lemma-9: body_md -->
\label{first_comp}
Let Assumption~\ref{assumption} hold. Then
\eqref{adjoint-decom-I} is uniquely solvable and
\begin{equation}
\left(
\|{\boldsymbol{r}}_{h,T_i^m}^{\mathrm I}\|_{Q,T_i^m}^{2}
+
k^2\|\eta_{h,T_i^m}^{\mathrm I}\|_{0,T_i^m}^{2}
\right)^{1/2}
\le
\frac{2\sqrt5}{3}\,
k^2(H+2mh)
\|w_{\rm ms}\|_{0,T_i^m}.
\label{eq:first-comp-energy}
\end{equation}
Consequently,
\begin{equation}
\|\eta_{h,T_i^m}^{\mathrm I}\|_{D,T_i^m}
\le
\frac{2\sqrt5}{3}\,
k^2(H+2mh)
\|w_{\rm ms}\|_{0,T_i^m}.
\label{eq:first-comp-D}
\end{equation}

<!-- proof-6: body_md -->
We write ${\boldsymbol{r}}_{h,T_i^m}^{\mathrm I}
=
\sum_{e\in\mathcal E_{T_i^m}}
r_{h,T_i^m,e}^{\mathrm I}{\boldsymbol{v}}_e,$ where $r_{h,T_i^m,e}^{\mathrm I}$ denotes the coefficient associated
with the edge $e$. Testing
\eqref{eq:adjoint-decom-I-a} with ${\boldsymbol{v}}_e$ and conjugating the resulting
scalar identity gives
\[
r_{h,T_i^m,e}^{\mathrm I}
=
\begin{cases}
\llbracket\eta_{h,T_i^m}^{\mathrm I}\rrbracket_e,
&
e\in\mathcal E_{T_i^m}^{0},
\\[1mm]
2\,\eta_{h,T_i^m}^{\mathrm I}|_{t_e},
&
e\in\mathcal E_{T_i^m}^{B},
\end{cases}
\]
where, for $e\in\mathcal E_{T_i^m}^{B}$,
$t_e\in\mathcal T_h$ is the unique fine-grid element contained in
$T_i^m$ such that $e\subset\partial t_e$.

Using the local trapezoidal quadrature rule, an interior edge receives
two contributions of weight $1/2$, whereas a boundary edge receives
only one. Therefore,
\begin{align}
\|{\boldsymbol{r}}_{h,T_i^m}^{\mathrm I}\|_{Q,T_i^m}^{2}
&=
\sum_{e\in\mathcal E_{T_i^m}^{0}}
\left|
\llbracket\eta_{h,T_i^m}^{\mathrm I}\rrbracket_e
\right|^{2}
+
2
\sum_{e\in\mathcal E_{T_i^m}^{B}}
\left|
\eta_{h,T_i^m}^{\mathrm I}|_{t_e}
\right|^{2}.
\label{q_u_condition}
\end{align}

We next derive an estimate for the $L^2$-norm. Let
$v_h\in W_h(T_i^m)$ and denote its values on the fine cells by
$v_{j,r}$, where $1\le j\le n_x$ and $1\le r\le n_y$.
Since both side lengths of $T_i^m$ are bounded by $H+2mh$,
$n_xh,n_yh\le H+2mh$. Along each row,
\[
2v_{j,r}
=
v_{1,r}+v_{n_x,r}
+
\sum_{\ell=1}^{j-1}
(v_{\ell+1,r}-v_{\ell,r})
-
\sum_{\ell=j}^{n_x-1}
(v_{\ell+1,r}-v_{\ell,r}),
\]
and hence the Cauchy--Schwarz inequality gives
\[
|v_{j,r}|^2
\le
|v_{1,r}|^2+|v_{n_x,r}|^2
+
\frac{n_x-1}{2}
\sum_{\ell=1}^{n_x-1}
|v_{\ell+1,r}-v_{\ell,r}|^2.
\]
Multiplying by $h^2$, summing over all cells, repeating the argument
in the vertical direction, and averaging the two estimates yields
\begin{align}
\|v_h\|_{0,T_i^m}^{2}
&\le
\frac{(H+2mh)^2}{4}
\sum_{e\in\mathcal E_{T_i^m}^{0}}
|\llbracket v_h\rrbracket_e|^2+
\frac{h(H+2mh)}{2}
\sum_{e\in\mathcal E_{T_i^m}^{B}}
|v_h|_{t_e}|^2
\notag\\
&\le
\frac{(H+2mh)^2}{4}
\left[
\sum_{e\in\mathcal E_{T_i^m}^{0}}
|\llbracket v_h\rrbracket_e|^2
+
2\sum_{e\in\mathcal E_{T_i^m}^{B}}
|v_h|_{t_e}|^2
\right].
\label{eq:adjoint-dirichlet-poincare}
\end{align}
Applying \eqref{eq:adjoint-dirichlet-poincare} to
$\eta_{h,T_i^m}^{\mathrm I}$ and using
\eqref{q_u_condition}, we obtain
\begin{equation}
\|\eta_{h,T_i^m}^{\mathrm I}\|_{0,T_i^m}
\le
\frac{H+2mh}{2}
\|{\boldsymbol{r}}_{h,T_i^m}^{\mathrm I}\|_{Q,T_i^m}.
\label{eq:first-comp-L2-flux}
\end{equation}
Taking
${\boldsymbol{v}}_h={\boldsymbol{r}}_{h,T_i^m}^{\mathrm I}$ and
$p_h=\eta_{h,T_i^m}^{\mathrm I}$ in
\eqref{adjoint-decom-I}, and taking real parts, gives
\begin{align}
\|{\boldsymbol{r}}_{h,T_i^m}^{\mathrm I}\|_{Q,T_i^m}^{2}
-
k^2\|\eta_{h,T_i^m}^{\mathrm I}\|_{0,T_i^m}^{2}
&=
\operatorname{Re}
\bigl(
\eta_{h,T_i^m}^{\mathrm I},
2k^2w_{\rm ms}
\bigr)_{T_i^m}
\notag\\
&\le
2k^2
\|w_{\rm ms}\|_{0,T_i^m}
\|\eta_{h,T_i^m}^{\mathrm I}\|_{0,T_i^m}.
\label{q_u}
\end{align}
Combining \eqref{eq:first-comp-L2-flux} and \eqref{q_u} gives
\[
\left(
1-\frac{k^2(H+2mh)^2}{4}
\right)
\|{\boldsymbol{r}}_{h,T_i^m}^{\mathrm I}\|_{Q,T_i^m}^{2}
\le

$1-k^2(H+2mh)^2/4\ge3/4$, and therefore
\begin{align*}

\le
\frac43
k^2(H+2mh)
\|w_{\rm ms}\|_{0,T_i^m}.
\end{align*}
Furthermore, \eqref{eq:first-comp-L2-flux} gives
\[
k\|\eta_{h,T_i^m}^{\mathrm I}\|_{0,T_i^m}
\le
\frac{k(H+2mh)}{2}
\|{\boldsymbol{r}}_{h,T_i^m}^{\mathrm I}\|_{Q,T_i^m}

\frac12
\|{\boldsymbol{r}}_{h,T_i^m}^{\mathrm I}\|_{Q,T_i^m}.

Consequently,

\right)^{1/2}

\|{\boldsymbol{r}}_{h,T_i^m}^{\mathrm I}\|_{Q,T_i^m}

\frac{2\sqrt5}{3}
k^2(H+2mh)
\|w_{\rm ms}\|_{0,T_i^m}.
\end{align*}

\[
\|\eta_{h,T_i^m}^{\mathrm I}\|_{D,T_i^m}^{2}
=
\sum_{e\in\mathcal E_{T_i^m}^{0}}
\left|
\llbracket\eta_{h,T_i^m}^{\mathrm I}\rrbracket_e
\right|^2
+
k^2
\|\eta_{h,T_i^m}^{\mathrm I}\|_{0,T_i^m}^{2},

v_{i,T_i^m}^{\mathrm{snap}}:=u_{h,T_i^m}^{\mathrm{II}}
=\sum_{j=1}^{r_i}\widehat{v}_{i,j}\tilde{\psi_i^j}.
\]
Its restriction gives
$v^{\mathrm{snap}}=\sum_{i=1}^{N_T}\sum_{j=1}^{r_i}
\widehat{v}_{i,j}\tilde{\psi_i^j}|_{T_i}$, with each restriction
extended by zero outside $T_i$. We then define
$v^{\mathrm{ms}}\in W_{\mathrm{ms}}$ by
\[
v^{\mathrm{ms}}:=\sum_{i=1}^{N_T}v^{ms}_i:=\sum_{i=1}^{N_T}\sum_{j=1}^{l_i}\widehat{v}_{i,j}\tilde{\psi_i^j}|_{T_i}
\]
and $v_{i,T_i^m}^{\mathrm{ms}}=\sum_{j=1}^{l_i}\widehat{v}_{i,j}\tilde{\psi_i^j}$,
 where $l_i \leq r_i$ is the number of eigenfunctions selected for $T_i$. 
Set $e_i^m=v_{i,T_i^m}^{\mathrm{snap}}
-v_{i,T_i^m}^{\mathrm{ms}}$.
The restriction estimate \eqref{eq:restriction-assembly} and spectral
orthogonality give
\begin{align*}
\|v^{\mathrm{snap}}-v^{\mathrm{ms}}\|_D^2
&\le C_{R,m}H^2\sum_i s_i(e_i^m,e_i^m)\\
&\le \frac{C_{R,m}H^2}{\Lambda_{2,m}}\sum_i a_i(e_i^m,e_i^m)\\
&\le \frac{C_{R,m}H^2}{\Lambda_{2,m}}
\sum_i a_i(u_{h,T_i^m}^{\mathrm{II}},u_{h,T_i^m}^{\mathrm{II}})\\
&\le \frac{2C_{R,m}H^2}{\Lambda_{2,m}}
\left(\sum_i a_i(u_h,u_h)
+\sum_i a_i(u_{h,T_i^m}^{\mathrm I},u_{h,T_i^m}^{\mathrm I})\right),
\end{align*}
where $\Lambda_{2,m}
=
\min\{\lambda_{l_i+1,2},\ i=1,\cdots,N_T\}. $
On an artificial patch-boundary edge,
$kh^2|u_h|_{t_e}|^2=k\|u_h\|_{0,t_e}^2$.
Counting these boundary-cell incidences in $N_{\rm ov}$ therefore yields
\[
\sum_i a_i(u_h,u_h)
\le N_{\rm ov}(h+k^{-1})\|u_h\|_D^2.
\]
Furthermore, \cref{lem:abc-source-energy} and finite overlap give
\[
\sum_i a_i(u_{h,T_i^m}^{\mathrm I},u_{h,T_i^m}^{\mathrm I})
\le \frac{2N_{\rm ov}(H+2mh)h}{k}\|f\|_{0,\Omega}^2.
\]
Thus we have
\begin{align*}
\|v^{\mathrm{snap}}-v^{\mathrm{ms}}\|_{D}
\le
\frac{H}{\sqrt{\Lambda_{2,m}}}
\left(
\sqrt{2C_{R,m}N_{\rm ov}}
(\sqrt{h}+k^{-\frac{1}{2}})\|u_h\|_D
+2\sqrt{\frac{C_{R,m}N_{\rm ov}(H+2mh)h}{k}}\|f\|_{0,\Omega}
\right).
\end{align*}
By the following estimates
$$
\|u_h-(u_f+v_{\rm ms})\|_{D}
\leq
\|u_h-(u_f+v^{\mathrm{snap}})\|_{D}
+\|v^{\mathrm{snap}}-v^{ms}\|_{D},
$$ 
we finish the proof.

Define the $b_h$-projection
$R_{\rm ms}^b:W_h\rightarrow W_{\rm ms}$ by
\[
b_h(R_{\rm ms}^bw_h,v_{\rm ms})
=
b_h(w_h,v_{\rm ms}),
\qquad
\forall v_{\rm ms}\in W_{\rm ms}.
\]
By the coercivity and continuity of $b_h$, for any
$v_{\rm ms}\in W_{\rm ms}$,
\begin{align*}
\frac{1}{\sqrt{2}}\|w_h-R_{\rm ms}^bw_h\|_D^2
&\leq
|b_h(w_h-R_{\rm ms}^bw_h,w_h-R_{\rm ms}^bw_h)|\\
&= 
|b_h(w_h-R_{\rm ms}^bw_h,w_h-v_{\rm ms})|\\
&\leq
\|w_h-R_{\rm ms}^bw_h\|_D
\|w_h-v_{\rm ms}\|_D.
\end{align*}
Hence,
\begin{align*}
\|w_h-R_{\rm ms}^bw_h\|_D
\le
\sqrt{2}
\inf_{v_{\rm ms}\in W_{\rm ms}}
\|w_h-v_{\rm ms}\|_D.
\label{eq:b-projection-best}
\end{align*}

$\widetilde u_h:=u_h-u_f$ and
$e_h:=\widetilde u_h-u_{\rm ms}$, and write

\label{eq:error-b-decomposition}
