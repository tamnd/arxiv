<!-- theorem-1: body_md -->
\label{thm:main} 
The harmonic trinomial $H_{\varepsilon,\delta}$ defined in \eqref{eq:H}
has unimodular zeros if and only if the corresponding condition below holds. The possible values of the $g$th powers of its unimodular zeros are listed in the third column.
\begin{center}
\begin{tabular}{@{}clc@{}}
\toprule
$(\varepsilon,\delta)$ & \multicolumn{1}{c}{condition} & values of $z^g$ \\
\midrule
$(1,1)$
  & $n_0\equiv k_0\pmod{3}$
  & $\mathrm{e}^{\pm \mathsf{i}\, 2\pi /3}$ \\

$(1,-1)$
  & $n_0\equiv k_0\pmod{6}$
  & $\mathrm{e}^{\pm \mathsf{i}\,\pi /3}$ \\

$(-1,1)$
  & $n_0+2k_0\equiv 0\pmod{6}$
  & $\mathrm{e}^{\pm \mathsf{i}\,\pi /3}$ \\

$(-1,-1)$
  & $4n_0-k_0\equiv 0\pmod{6}$
  & $\mathrm{e}^{\pm \mathsf{i}\,\pi /3}$ \\
\bottomrule
\end{tabular}
\end{center}
Moreover, if the condition in the second column holds, then $H_{\varepsilon,\delta}$ has exactly $2g$
distinct unimodular zeros.
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{cor:explicit}
For $\varepsilon,\delta\in\{-1,1\}$, define
\[
 \theta:=\begin{cases}
 2\pi/3,&(\varepsilon,\delta)=(1,1),\\
 \pi/3,&(\varepsilon,\delta)\ne(1,1).
 \end{cases}
\]
Under the corresponding congruence in Theorem~\ref{thm:main},
the unimodular zeros of $H_{\varepsilon,\delta}$ are given by
\begin{equation}\label{eq:explicit-roots}
 z_j^{\pm}
   =\exp\!\left(\mathsf{i}\frac{(\pm\theta+2\pi j)}{g}\right),
 \quad j=0,1,\ldots,g-1.
\end{equation}
<!-- end corollary-1 -->
<!-- proof-1: body_md -->
In all cases covered by Theorem~\ref{thm:main}, the solutions of $z^g=\mathrm{e}^{\pm\mathsf{i}\theta}$ are
\begin{equation}\label{eq:roots}
z_j^{\pm}
=\exp\left(\mathsf{i}\frac{(\pm\theta+2\pi j)}{g}\right),
\quad j=0,\ldots,g-1.
\end{equation}
Each sign gives $g$ distinct solutions. The two sets are disjoint because
$\mathrm{e}^{\mathsf{i}\theta}\ne\mathrm{e}^{-\mathsf{i}\theta}$. Consequently, there are exactly $2g$ unimodular
zeros.
<!-- end proof-1 -->
<!-- lemma-1: body_md -->
\label{lem:reduction}
For $z\in\mathbb{S}^1$, set $w:=z^g$.
Then
\begin{equation}\label{eq:reduced}
H_{\varepsilon,\delta}(z)=0
\quad\textrm{if and only if}\quad
w^{n_0}+\varepsilon w^{-k_0}+\delta=0.
\end{equation}
Every solution $w\in\mathbb{S}^1$ of the equation on the right-hand side of \eqref{eq:reduced} has exactly $g$
preimages in $\mathbb{S}^1$ under the map $z\mapsto z^g$.
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
Recall \eqref{eq:n0k0}.
For $z\in\mathbb{S}^1$, we note that
$z^n=(z^g)^{n_0}=w^{n_0}$ and
$\overline z^{\,k}=z^{-k}=(z^g)^{-k_0}=w^{-k_0}$.
Hence, $H_{\varepsilon,\delta}(z)=
w^{n_0}+\varepsilon w^{-k_0}+\delta$,
which proves \eqref{eq:reduced}.

Now, let $w\in\mathbb{S}^1$ be a solution of the equation on the right-hand side of~\eqref{eq:reduced}. Write $w=\mathrm{e}^{\mathsf{i} t}$ for some $t\in\mathbb{R}$. The solutions of $z^g=w$ are given by
\[
z_j:=
\exp\left(\mathsf{i}\frac{(t+2\pi j)}{g}\right),
\qquad
j=0,\ldots,g-1.
\]
These $g$ points are all different and belong to $\mathbb{S}^1$. Therefore, every solution
$w\in\mathbb{S}^1$ of the reduced equation on the right-hand side of \eqref{eq:reduced} has exactly $g$ preimages in $\mathbb{S}^1$.
<!-- end proof-2 -->
<!-- lemma-2: body_md -->
\label{lem:geometry}
Let $u,v\in\mathbb{S}^1$ and $\zeta:=\mathrm{e}^{2\pi\mathsf{i}/3}$.
Then the following statements hold.
\begin{align*}
\textrm{(a)}\quad u+v+1=0
&\quad\Longleftrightarrow\quad
(u,v)=(\zeta,\zeta^2)\ \text{or}\ (\zeta^2,\zeta),\\
\textrm{(b)}\quad u+v-1=0
&\quad\Longleftrightarrow\quad
(u,v)=(-\zeta,-\zeta^2)\ \text{or}\ (-\zeta^2,-\zeta),\\
\textrm{(c)}\quad u-v+1=0
&\quad\Longleftrightarrow\quad
(u,v)=(\zeta,-\zeta^2)\ \text{or}\ (\zeta^2,-\zeta),\\
\textrm{(d)}\quad u-v-1=0
&\quad\Longleftrightarrow\quad
(u,v)=(-\zeta,\zeta^2)\ \text{or}\ (-\zeta^2,\zeta).
\end{align*}
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
We prove (a); (b), (c), and (d) follow by applying (a) to $(-u,-v)$, $(u,-v)$, and $(-u,v)$, respectively.
Suppose $u+v+1=0$. Then $v=-1-u$, and
\[
1=|v|^2=|1+u|^2=2+2\operatorname{Re}u.
\]
Therefore, $\operatorname{Re}u=-1/2$. Write $u=x+\mathsf{i} y$ with $x,y\in\mathbb R$ and note that
$y=\pm\sqrt3/2$. 
Hence, $u=\zeta$ or $u=\zeta^2$. Since $\zeta^3=1$ and $\zeta\ne1$, we have $1+\zeta+\zeta^2=0$. Therefore, $v=-1-u$ is the other member of $\{\zeta,\zeta^2\}$. Conversely, both ordered pairs have unit-modulus entries and satisfy $u+v+1=0$.
<!-- end proof-3 -->
<!-- remark-1: body_md -->
\label{rem:primitive}
Recall that $\zeta:=\mathrm{e}^{2\pi\mathsf{i}/3}$.
Note that the cube roots of unity are
$1,\zeta,\zeta^2$ and have orders $1,3,3$, respectively.
Moreover, the sixth roots of unity are $1,-\zeta^2,\zeta,-1,\zeta^2,-\zeta$,
of orders $1,6,3,2,3,6$, respectively. 
Indeed, 
for $m\in\mathbb{Z}$, $(\zeta^2)^m=1$ exactly when $3\mid2m$, or, equivalently, when $3\mid m$.
The other orders follow in the same way from divisibility.
Consequently, the only roots of unity of order six are
$-\zeta$ and $-\zeta^2$.
<!-- end remark-1 -->
<!-- proof-4: body_md -->
Write $w=z^g$. By Lemma~\ref{lem:reduction}, it is enough to solve
\begin{equation}\label{eq:weqn}
w^{n_0}+\varepsilon w^{-k_0}+\delta=0,
\quad w\in\mathbb{S}^1.
\end{equation}
Suppose that $(\varepsilon,\delta)=(1,1)$.
The proofs of the other cases proceed similarly, using parts (b), (c), and (d) of 
Lemma~\ref{lem:geometry} and Remark~\ref{rem:primitive}.
Then \eqref{eq:weqn} becomes
$w^{n_0}+w^{-k_0}+1=0$.
By part (a) of Lemma~\ref{lem:geometry}, we obtain
$\{w^{n_0},w^{-k_0}\}=\{\zeta,\zeta^2\}$,
which implies $w^{3n_0}=w^{3k_0}=1$. Since $\gcd(n_0,k_0)=1$, B\'ezout's identity
gives $w^3=1$. Note that $w=1$ does not satisfy $w^{n_0}+w^{-k_0}+1=0$.
Hence, $w\in\{\zeta,\zeta^2\}$.
Assume $w=\zeta$. Using the division Euclides algorithm, write $n_0=3p_0+r_0$ and $k_0=3p'_0+r'_0$ for some $p_0,p'_0\in \mathbb{N}\cup\{0\}$ and $r_0,r'_0\in \{0,1,2\}$. We observe that
\begin{equation}
\begin{split}
0=w^{n_0}+w^{-k_0}+1&=\zeta^{n_0}+\zeta^{-k_0}+1=
\zeta^{3p_0+r_0}+\zeta^{-3p'_0-r'_0}+1=\zeta^{r_0}+\zeta^{-r'_0}+1.
\end{split}
\end{equation}
Since $|\zeta|=1$, this implies $r_0,r'_0\in \{1,2\}$.
If $r_0=r'_0$, the statement is complete.
Assume without loss of generality that $r_0=1$ and $r'_0=2$. Note that
\begin{equation}
\begin{split}
0=\zeta+\zeta^{-2}+1=\zeta+\zeta+1,
\end{split}
\end{equation}
which yields a contradiction because $|\zeta|=1$.

The proof of the case $w=\zeta^2$ follows similarly.
