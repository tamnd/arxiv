\label{rem:reference-connection-sign}
The affine formula~\eqref{eq:identificationCP} is intrinsic, but its local signs
must be distinguished from those of a conventional gauge potential. On the trivialization $P\simeq X\times(G\Join V)$, let
$A,K\in\Omega^1(X,\mathfrak g\Join V)$ be the usual local connection one-forms representing the reference and reduced
connections $\mathcal A,\sigma\in\Gamma(\mathcal C(P))$, respectively.
For a right principal action the corresponding horizontal splittings have
local vertical parts $-A$ and $-K$.  Consequently, the local representative of
the affine difference $\sigma-\mathcal A$ is $\xi=A-K$.
                  
         
                                    
         
                                             
                
Thus, the reconstruction condition
$\operatorname{Curv}(\mathcal A+\xi)=0$ (see~\eqref{eq:reconstructioncompatibility} below), where $+$ denotes affine addition in
$\mathcal C(P)$, reads in conventional local connection forms
\begin{equation}
F_{A-\xi}
=\mathrm d(A-\xi)
+\frac12[A-\xi\wedge A-\xi]=0,
\label{eq:local-reconstruction-sign}
\end{equation}
where $[K\wedge K](u,v)=[K(u),K(v)]-[K(v),K(u)]$ for (local) fields $u,v\in\mathfrak X(X)$.

In particular, for the zero reference connection and the right logarithmic
derivative $\Xi=(\mathrm ds)s^{-1}$, the reduced connection form is
$K=-\Xi$, whence~\eqref{eq:local-reconstruction-sign} becomes
\begin{equation}
\mathrm d\Xi-\frac12[\Xi\wedge\Xi]=0.
       
                                    
\label{eq:right-MC-sign-convention}
\end{equation}

\label{prop:centered-standard-equivalence}
Under the previous conditions, the following statements hold true:
\begin{enumerate}
    \item The map
    \begin{equation*}
    \Phi:G\Join V\longrightarrow G\ltimes_\rho V,
    \qquad
    \Phi(g,v):=(g,v\cdot g^{-1}),
                                             
    \end{equation*}
    where $\ltimes_\rho$ denotes the semi-direct product given by the representation $\rho$, is a Lie group isomorphism, whose tangent map at the identity element is the identity,
    \begin{equation}
    F:=T_{(e,0)}\Phi:
    \mathfrak g\Join V\longrightarrow\mathfrak g\ltimes_{\rho_*}V,
    \qquad
    F(\xi,v)=(\xi,v).
    \label{eq:centered-standard-Lie-isomorphism}
    \end{equation}
    \item Let $P_\Phi$ denote the same total space as $P$, regarded as a principal
    $(G\ltimes_\rho V)$-bundle with the right action
    $p\cdot_\Phi(g,b):=p\cdot\Phi^{-1}(g,b)$ for each $p\in P$ and $(g,b)\in G\ltimes_{\rho}V$. Then the map $F$ induces natural
    isomorphisms of adjoint and connection bundles,
    \begin{equation}
    \operatorname{Ad}_{G\Join V}(P)
    \simeq
    \operatorname{Ad}_{G\ltimes_\rho V}(P_\Phi),
    \qquad
    \mathcal C_{G\Join V}(P)
    \simeq
    \mathcal C_{G\ltimes_\rho V}(P_\Phi).
    \label{eq:centered-standard-bundle-isomorphisms}
    \end{equation}
    \item For each $\xi\in\Omega^1(X,\operatorname{Ad}_{G\Join V}(P))$ and $\eta\in\Gamma(\operatorname{Ad}_{G\Join V}(P))$, we have
    \begin{equation}
    F\big(\nabla^{\mathcal A}\eta-[\xi,\eta]\big)
    =\nabla^{\mathcal A'}\eta'-[\xi',\eta'],
    \label{eq:centered-standard-variation-intertwining}
    \end{equation}
    where $\xi'=F(\xi)$, $\eta'=F(\eta)$, and
    $\mathcal A'=F(\mathcal A)$.
    \item For the reduced Lagrangian on the semidirect product,
    \[\ell':T^*X\otimes\operatorname{Ad}_{G\ltimes_\rho V}(P_\Phi)\to\mathbb R,\qquad\ell'(\xi'):=\ell(F^{-1}\xi'),
    \]
    the momenta satisfy
    \begin{equation}
    \frac{\delta\ell'}{\delta\xi'}=(F^{-1})^*\left(\frac{\delta\ell}{\delta\xi}\right).
    \label{eq:centered-standard-momentum-intertwining}
    \end{equation}
\end{enumerate}

<!-- proof-1: body_md -->
It is readily seen that the inverse of $\Phi$ reads $\Phi^{-1}(g,b)=(g,b\cdot g)$ for each $(g,b)\in G\ltimes_\rho V$, whence $\Phi$ is a diffeomorphism. Using commutativity of the left and right actions, we have
\begin{align*}
\Phi\big((g_1,v_1)(g_2,v_2)\big)
&=\big(g_1g_2,
(g_1\cdot v_2+v_1\cdot g_2)\cdot(g_1g_2)^{-1}\big)\\
&=\big(g_1g_2,
v_1\cdot g_1^{-1}
+\rho(g_1)(v_2\cdot g_2^{-1})\big),
\end{align*}
for each $(g_1,v_1),(g_2,v_2)\in G\Join V$, which is the multiplication in $G\ltimes_\rho V$, thus making $\Phi$ a Lie group isomorphism. By differentiating at the
identity, we obtain~\eqref{eq:centered-standard-Lie-isomorphism}; in particular,
the two Lie brackets coincide on the underlying vector space
$\mathfrak g\oplus V$. Relabelling the principal action by a group
isomorphism gives the bundle maps in
\eqref{eq:centered-standard-bundle-isomorphisms}.  Naturality of the adjoint
connection, bracket, covariant divergence, and curvature under $F$ proves
\eqref{eq:centered-standard-variation-intertwining}.  Finally, the chain rule for
$\ell'=\ell\circ F^{-1}$ gives
\eqref{eq:centered-standard-momentum-intertwining}.

\label{thm:covariant-reduction-centered}
Let $X$ be an oriented manifold with volume form ${\rm vol}_X$, let
$P\to X$ be a right principal $(G\Join V)$-bundle, and let $P\simeq\mathcal U\times(G\Join V)$ be a (possibly local) trivialization of $P\to X$, where $\mathcal U\subset X$ is an
open set.  Let
$\mathcal L=L\,{\rm vol}_X:J^1P\to\Lambda^{n+1}T^*X$ be a
$(G\Join V)$-invariant Lagrangian density and let
$\ell:\mathcal C(P)\to\mathbb R$ be its reduced Lagrangian~\eqref{eq:reducedlagrangian}.  Given a
principal connection $\mathcal A$ on $P$, regard $\ell:T^*X\otimes\operatorname{Ad}(P)\to\mathbb R$ thanks to~\eqref{eq:identificationCP}.  Let
$s=(g,v)\in\Gamma_{\mathcal U}(P)$ be a (local) section and
$\xi=(\xi^{\mathfrak g},\xi^V)=[j^1s]-\mathcal A\in
\Omega^1(U,\operatorname{Ad}(P)|_U)$ its (local) reduced field. 

\medskip

Then the following statements are equivalent:

\begin{enumerate}[(i)]
\item The section $s$ is critical for the Hamilton principle, i.e.,
\[
\delta\mathcal S[s]=\delta\int_U L(j^1 s)\,{\rm vol}_X=0,
\]
for compactly supported vertical variations
$\delta s\in\Gamma_c(s^*VP)$.

\item The section $s$ satisfies the Euler--Lagrange equations for $L$.

\item The reduced section $\xi$ is critical for the reduced variational principle, i.e.,
\[
\delta\mathcal S_{\mathrm{red}}[\xi]=\delta\int_U \ell(\xi)\,{\rm vol}_X=0,
\]
for compactly supported variations of the form \eqref{eq:delta-xi-CGPR}.                                                                                                                 

\item The reduced section $\xi$ satisfies the Euler--Poincar\'e field equations \cite{CaRaSh2000,CaGaRa2001}. In the trivialization, we obtain the \emph{Euler--Poincar\'e equations on the centered semidirect product} \eqref{eq:centeredeqsg} and \eqref{eq:centeredeqsV}.                                                
               
                                                                         
   
                                                       
                                                                              
                                  
   
                                                                                          
             
\end{enumerate}

\label{prop:reduced-MSFF}
Let $s:X\to P$ be a solution of the Euler--Lagrange equations and let

\end{equation}
<!-- end proposition-2 -->
<!-- proof-4: body_md -->
Since ${\rm vol}_X$ is fixed, $\delta_U\widehat{\mathsf P}$ and $\delta_W\widehat{\mathsf P}$
are the $n$-form representatives of $\delta_U\mathsf P$ and
$\delta_W\mathsf P$, respectively. Starting from
\eqref{eq:reduced-symplectic-current} and using
\eqref{eq:momentum-n-form}, together with the compatibility of
$\nabla^{\mathcal A}$ with the duality pairing and the Lie bracket,
we obtain
\begin{align*}
\mathrm d\,\omega(U,W)
={}&
\Big(
\langle
\operatorname{div}^{\mathcal A}(\delta_W\mathsf P),\eta_U
\rangle
+
\langle
\delta_W\mathsf P,\nabla^{\mathcal A}\eta_U
\rangle
\\
&\quad-
\langle
\operatorname{div}^{\mathcal A}(\delta_U\mathsf P),\eta_W
\rangle
-
\langle
\delta_U\mathsf P,\nabla^{\mathcal A}\eta_W
\rangle
\\
&\quad-
\langle
\operatorname{div}^{\mathcal A}\mathsf P,[\eta_U,\eta_W]
\rangle
-
\langle
\mathsf P,\nabla^{\mathcal A}[\eta_U,\eta_W]
\rangle
\Big){\rm vol}_X.
\end{align*}
Using \eqref{eq:intrinsic-EP} and \eqref{eq:linearized-EP},
together with
\[
\nabla^{\mathcal A}\eta_U
=
\delta_U^{\mathcal A}\xi+[\xi,\eta_U],
\qquad
\nabla^{\mathcal A}\eta_W
=
\delta_W^{\mathcal A}\xi+[\xi,\eta_W],
\]
the coadjoint and bracket terms cancel by the convention
$\langle\operatorname{ad}^*_\zeta\alpha,\eta\rangle
=-\langle\alpha,[\zeta,\eta]\rangle
$
and the Jacobi identity. Hence
\begin{equation}\label{eq:MSFF-Hessian-step}
\mathrm d\,\omega(U,W)
=
\Big(
\langle
\delta_W\mathsf P,\delta_U^{\mathcal A}\xi
\rangle
-
\langle
\delta_U\mathsf P,\delta_W^{\mathcal A}\xi
\rangle
\Big){\rm vol}_X.
\end{equation}

\[

\mathbb H_\ell(\xi)[\delta_W^{\mathcal A}\xi],
\delta_U^{\mathcal A}\xi
\rangle
-
\langle
\mathbb H_\ell(\xi)[\delta_U^{\mathcal A}\xi],
\delta_W^{\mathcal A}\xi
\rangle,
\]
which vanishes by symmetry of the fiber Hessian of $\ell$.
Therefore $\mathrm d\,\omega(U,W)=0$.

\label{prop:covariant-balance-laws}
Under the assumptions above, the following statements hold.

\emph{(i)} For every $\eta\in\Gamma(\operatorname{Ad}(P))$,
\begin{equation}\label{eq:reduced-momentum-balance}
\mathrm d J_\eta
=
\left\langle
\mathsf P,
\delta_\eta^{\mathcal A}\xi
\right\rangle
{\rm vol}_X,
\qquad
\delta_\eta^{\mathcal A}\xi
:=
\nabla^{\mathcal A}\eta-[\xi,\eta].
\end{equation}

\emph{(ii)} Let $\chi\in\mathfrak g\Join V$ and consider the infinitesimal
variation generated by the fixed right action of $G\Join V$,
$s_\varepsilon=s\cdot\exp(\varepsilon\chi)$. If $\eta_\chi$ denotes its
reduced variation parameter, then
\begin{equation}\label{eq:fixed-group-transport}
\delta_{\eta_\chi}^{\mathcal A}\xi=0,
\qquad
\mathrm dJ_{\eta_\chi}=0.
\end{equation}
In a local trivialization, write
$\eta_\chi=(\eta_\chi^{\mathfrak g},\eta_\chi^V)$, $\mathsf P=(\mathsf P_{\mathfrak g},\mathsf P_V)$ and $\widehat{\mathsf P}=(\widehat{\mathsf P}_{\mathfrak g},\widehat{\mathsf P}_V)$, and define
\begin{equation}\label{eq:centered-sector-currents}
J_\chi^{\mathfrak g}
:=
\langle
\widehat{\mathsf P}_{\mathfrak g},
\eta_\chi^{\mathfrak g}
\rangle,
\qquad
J_\chi^V
:=
\langle
\widehat{\mathsf P}_V,
\eta_\chi^V
\rangle.
\end{equation}
Then $J_{\eta_\chi}=J_\chi^{\mathfrak g}+J_\chi^V$ and
\begin{equation}\label{eq:centered-sector-balance}
\mathrm dJ_\chi^{\mathfrak g}
=
\mathcal E_\chi\,{\rm vol}_X,
\qquad
\mathrm dJ_\chi^V
=
-\mathcal E_\chi\,{\rm vol}_X,
\end{equation}
where
\begin{equation}\label{eq:centered-exchange-rate}
\mathcal E_\chi
:=
\left\langle
(\mathcal A^V-\xi^V)\diamondsuit\mathsf P_V,
\eta_\chi^{\mathfrak g}
\right\rangle
=
-
\left\langle
\eta_\chi^{\mathfrak g}\heartsuit\mathsf P_V,
\mathcal A^V-\xi^V
\right\rangle.
\end{equation}
Thus the two sectorial currents need not be separately conserved, while their
sum is closed.

\emph{(iii)} Let $\widetilde Z\in\mathfrak X(P)$ be a projectable vector field
over $Z\in\mathfrak X(X)$ whose first prolongation is an infinitesimal
symmetry of $\mathcal L$. Along $s$, let
$U_{\widetilde Z}:=\widetilde Z\circ s-Ts(Z)\in\Gamma(s^*VP)$
and let $\eta_{\widetilde Z}\in\Gamma(\operatorname{Ad}(P))$ be its reduced variation
parameter. The covariant Noether current is
\begin{equation}\label{eq:reduced-Noether-current}
\mathcal J_{\widetilde Z}
:=
(j^1s)^*\!\left(
\iota_{j^1\widetilde Z}\Theta_{\mathcal L}
\right)
=
J_{\eta_{\widetilde Z}}
+
\ell\,\iota_Z{\rm vol}_X.
\end{equation}
and it satisfies
\begin{equation}\label{eq:covariant-Noether-conservation}
\mathrm d\mathcal J_{\widetilde Z}=0.
\end{equation}

In particular, let $Z^{H_{\mathcal A}}$ be the horizontal lift of
$Z\in\mathfrak X(X)$ with respect to the reference connection $\mathcal A$.
Its vertical characteristic has reduced parameter
\begin{equation}\label{eq:horizontal-lift-parameter}
\eta_{Z^{H_{\mathcal A}}}
=
-\iota_Z\xi.
\end{equation}

$\mathcal L$,
\begin{equation}\label{eq:energy-momentum-current}
\mathcal J_{Z^{H_{\mathcal A}}}
= -\iota_{\mathsf T_{\mathcal A}(Z)}{\rm vol}_X.\end{equation}

\begin{equation}\label{eq:canonical-energy-momentum}
\mathsf T_{\mathcal A}(Z)

\left\langle
\mathsf P,\iota_Z\xi
\right\rangle
-\ell Z,
\qquad Z\in\mathfrak X(X),
\end{equation}

\begin{equation}\label{eq:energy-momentum-conservation}
\operatorname{div}\bigl(\mathsf T_{\mathcal A}(Z)\bigr)=0.
\end{equation}

\label{rem:noether-generator-splitting}
The splitting of the sectorial currents in
\eqref{eq:centered-sector-currents} should not be confused with splitting
the fixed infinitesimal generator itself. Indeed, for
$\chi=(\chi^{\mathfrak g},\chi^V)\in\mathfrak g\Join V$, set
\[
\chi_{\mathfrak g}:=(\chi^{\mathfrak g},0),
\qquad
\chi_V:=(0,\chi^V).
\]
Since the Lagrangian is invariant under the full $G\Join V$-action,
each of these generators separately defines an infinitesimal symmetry.
Hence, by Proposition~\ref{prop:covariant-balance-laws} \emph{(ii)},
\[
\mathrm dJ_{\eta_{\chi_{\mathfrak g}}}=0,
\qquad
\mathrm dJ_{\eta_{\chi_V}}=0.
\]
By linearity,
\[
\eta_\chi
=
\eta_{\chi_{\mathfrak g}}+\eta_{\chi_V},
\qquad
J_{\eta_\chi}
=
J_{\eta_{\chi_{\mathfrak g}}}
+
J_{\eta_{\chi_V}}.
\]
These separately conserved currents should not, however, be identified
with $J_\chi^{\mathfrak g}$ and $J_\chi^V$ in
\eqref{eq:centered-sector-currents}. In general,
$\eta_{\chi_{\mathfrak g}}$ and $\eta_{\chi_V}$ need not coincide with
$(\eta_\chi^{\mathfrak g},0)$ and $(0,\eta_\chi^V)$, respectively,
because the centered semidirect-product adjoint action mixes the two
components. Thus splitting the symmetry generator before forming the
reduced current is different from projecting the reduced current into
its $\mathfrak g$- and $V$-sectorial contributions. The latter satisfy
the exchange law \eqref{eq:centered-sector-balance} and therefore need
not be separately conserved.

Let the left action of $\mathrm{SO}(3)$ on $\mathbb R^{3}$ be the standard
representation.  Every smooth linear right action of $\mathrm{SO}(3)$ on $\mathbb R^3$

<!-- proof-8: body_md -->
The commutant of the standard real representation of $\mathrm{SO}(3)$ consists of the
scalar endomorphisms.  Hence a commuting right action has the form
$v\cdot Q=\lambda(Q)v$.  The right-action identity implies that $\lambda$ is a
smooth character of $\mathrm{SO}(3)$ with values in $\mathbb R^{\times}$.  Since
$\mathrm{SO}(3)$ is connected and compact, its image is a connected compact subgroup of
$\mathbb R^{\times}$, and is therefore $\{1\}$.  Thus $v\cdot Q=v$ for every
$Q\in \mathrm{SO}(3)$.

\label{rem:advantages-matrix-formulation}
The three summands of $\operatorname{Mat}(3)$ under conjugation are
pairwise non-isomorphic irreducible real $\operatorname{SO}(3)$-modules,
each occurring with multiplicity one. Consequently, an
$\operatorname{SO}(3)$-equivariant linear constitutive map preserves
each summand and acts on it by a scalar. Such a map cannot couple
distinct summands. Off-diagonal constitutive couplings require
additional anisotropic data or an enlarged representation with
equivalent summands.

There is also a useful dynamical consequence. If $K$ is self-adjoint
and equivariant, invariance of its quadratic form gives
$\langle KW,[\Sigma,W]\rangle=0$ for every
$\Sigma\in\mathfrak{so}(3)$.
Polarization and contraction with the symmetric tensor
$\gamma^{\mu\nu}$ therefore yield
$W_\mu\diamondsuit N^\mu=0$.
Thus the internal torque in the rotational equation vanishes for
this constitutive choice. If $J$ is isotropic as well, the contracted
term $[\Omega_\mu,M^\mu]$ also vanishes.
The general reduced equations retain these terms because anisotropic
constitutive maps need not have these cancellations.

The representation decomposition and covariant derivatives remain
useful in both cases. Unit-director constraints, projective
identification, bulk potentials, and anisotropic Oseen--Frank moduli
are additional constitutive or broken-symmetry data; they are not
imposed by the centered product itself.

<!-- proposition-9: body_md -->
\label{prop:jet1-linear-stability} For every initial state
$(\sigma_0,\sigma_1,\rho_0,\rho_1)\in\mathcal E$,
the Cauchy problem has a unique global energy solution,
with $\sigma$ understood modulo constants, and the solution
depends continuously on the initial state. Moreover,
\begin{equation}
E_{\rm lin}(t)
=\frac{\kappa_1}{2}\int_{\mathbb R}
\left(
\sigma_t^2+\sigma_x^2
+\rho_t^2+\rho_x^2+m^2\rho^2
\right){\rm d}x
=E_{\rm lin}(0).
\label{eq:jet1-linearized-energy}
\end{equation}
Thus the linearized flow preserves the norm
$\|(\sigma,\sigma_t,\rho,\rho_t)\|_{\mathcal E,m}
=(2E_{\rm lin}/\kappa_1)^{1/2}$.
In particular, it controls the tangential derivatives and,
since $m>0$, the full $H^1\times L^2$ norm of the normal
perturbation.

The conversion front has no exponentially growing linear
Fourier modes\footnote{These conclusions concern the linearized
equations and do not assert nonlinear orbital or
asymptotic stability.}.

The wave-map density associated with
\eqref{eq:jet1-fermi-metric} is
\[
\frac{\kappa_1}{2}
\left[-r_t^2+r_x^2
+\cosh^2r\bigl(-s_t^2+s_x^2\bigr)\right],
\]
whose Euler--Lagrange equations are
\eqref{eq:jet1-fermi-r}-\eqref{eq:jet1-fermi-s}.

For the background solution,
$s_{0,x}=k$ and $s_{0,t}=-kc$, so
$s_{0,x}^2-s_{0,t}^2=k^2(1-c^2)=m^2$.
Using
\[
\sinh(\varepsilon\rho)\cosh(\varepsilon\rho)
=\varepsilon\rho+O(\varepsilon^3),
\qquad
\cosh^2(\varepsilon\rho)=1+O(\varepsilon^2),
\]
the normal equation becomes
\[
\varepsilon\bigl(\rho_{tt}-\rho_{xx}+m^2\rho\bigr)
+O(\varepsilon^2)=0.
\]
In the tangential equation, the factor $\cosh^2r$
has no term of first order in $\varepsilon$.
Since $s_0$ solves the wave equation, its linearization is
$\sigma_{tt}-\sigma_{xx}=0$.
This proves \eqref{eq:jet1-linearized-wave-KG}.

For smooth compactly supported initial data, multiply the
two linearized equations by $\kappa_1\sigma_t$ and
$\kappa_1\rho_t$, respectively, and add them. This gives
the local energy identity
\[
\partial_t\left[
\frac{\kappa_1}{2}
\left(\sigma_t^2+\sigma_x^2
+\rho_t^2+\rho_x^2+m^2\rho^2\right)
\right]
=
\kappa_1\partial_x
\left(\sigma_t\sigma_x+\rho_t\rho_x\right).
\]
Finite propagation speed ensures that the boundary terms
vanish upon integration over $\mathbb R$, yielding
\eqref{eq:jet1-linearized-energy}.

To describe the Fourier modes explicitly, consider
complex plane waves
\[
\sigma(t,x)=a\,e^{i(\xi x-\omega t)},
\qquad\text{or}\qquad
\rho(t,x)=b\,e^{i(\xi x-\omega t)},
\]
where $\xi\in\mathbb R$ is the spatial wave number.
Substitution into the respective equations gives
\[
\omega^2=\xi^2
\quad\text{and}\quad
\omega^2=\xi^2+m^2.
\]
Both dispersion relations have real frequencies, so
no such mode grows exponentially in time.
At $\xi=0$, the tangential equation admits constant
and affine functions of time; this is a zero-frequency
mode, not an exponential instability. Spatially constant
nonzero velocities are excluded by the $L^2(\mathbb R)$
condition on $\sigma_t$, and constant displacements are
identified in $\dot H^1(\mathbb R)$.

Finally, the Fourier solution formulas for the wave and
Klein-Gordon equations give global solutions for smooth
data. The conserved energy estimate extends these
solutions by density to all initial states in $\mathcal E$,
and gives uniqueness and continuous dependence in the
energy norm.
