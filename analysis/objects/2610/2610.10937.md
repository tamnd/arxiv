<!-- definition-1: body_md -->
Let $M$ be a smooth manifold together with a  Koszul connection $\nabla$ on $TM$. 
For  $I=(a,b)$, $a<0<b$ let
\begin{equation}\label{E:ueps}
{\mathbb C} I=\{z=\sigma+i\tau\in\mathbb C \mid \text{Re }z\in I\}.
\end{equation}
Let $v\in TM$ and 
$\gamma_v:I\rightarrow M$ be the geodesic
with initial condition $\dot\gamma_v(0)=v$. Define
\begin{equation}\label{F:Omegav}
\begin{aligned}
\Omega_v:{\mathbb C} I&\rightarrow TM\\
\sigma+i\tau&\mapsto \tau\dot{\gamma}(\sigma)
\end{aligned}
\end{equation}
<!-- end definition-1 -->
<!-- definition-2: body_md -->
Let  $N\subset TM$ be an open, fiberwise $*$-domain (i.e. $v\in N$ and $0\le s\le1$ implies $sv\in N$)  with $M\subset N$. An {\it adapted complex structure} (ac-structure for short) on $N$ means a complex manifold structure so that for every
$v\in TM$, the map $\Omega_v$ is holomorphic on $\Omega_v^{-1}(N)$.
$\nabla$ is called {\it entire}, if an ac-structure exists on $TM$.
A  pseudo-Riemannian manifold $(M,g)$    is called {\it entire} if the Lévi-Civita connection of $g$ is entire.\cite[Definition 2.1.]{FSz26}
<!-- end definition-2 -->
<!-- definition-3: body_md -->
Let $N$ be a smooth manifold. An involutive structure  on $N$
 is a smooth, involutive, complex subbundle $P$ of $\mathbb CTN$.  Involutivity means that the Lie bracket of local smooth sections of  $P$  is again
 a section of $P$. An involutive structure $P$ 
 is called a polarization   if $\dim N$
 is even and

\label{thm:x'^2}
    Let $c_1,c_2\in{\mathbb C}$, then the following ODE:

    \begin{equation}\label{eq: x'^2}
     \dot{x}^2(t)=c_1 x^2(t)+c_2 x^4(t),
    \end{equation}
    

    has solutions as follows:
    \begin{enumerate}
        \item If $c_1=0=c_2$, then $x(t)$ is constant.
        \item If $c_1=0$, $c_2\neq 0,$ then $$x(t)=\frac{x(0)}{1\pm x(0)\sqrt{c_2}t}$$
        \item If $c_1\neq 0$, $c_2=0$ then $$x(t)=x(0)e^{\pm\sqrt{c_1}t}$$
        \item If $c_1\neq 0\neq c_2,$ then $$x(t)=\sqrt{\frac{c_1}{c_2}}\frac{2c_3e^{\sqrt{c_1}t}}{1-c_3^2 e^{2\sqrt{c_1}t}},$$
    \end{enumerate}

    where $c_3=\frac{-\sqrt{\frac{c_1}{c_2}}\pm\sqrt{x(0)^2+\frac{c_1}{c_2}}}{x(0)}\in{\mathbb C}$, if $x(0)\neq 0$ and $c_3=0$, if $x(0)=0$. The square root sign can take both possible complex values in each case. Furthermore, under the assumption that $c_1,c_2,c_3\in\mathbb R$, the solution is $i\mathbb R$-complete iff $c_2\geq 0$.
<!-- end theorem-2 -->
<!-- proof-1: body_md -->
The first three cases are elementary and show thatentire solution exists only if $c_2=0$.

    In the case when $c_1\neq0\neq c_2$, note that since equation \eqref{eq: x'^2} is separable, it suffices to solve 

    \begin{equation}\label{eq:integral}
        \int \frac{\dot{x}(t)dt}{x(t)\sqrt{1+\frac{c_2}{c_1} x^2}}=\sqrt{c_1}t+c.
    \end{equation}

    We now consider the following substitution:

    \begin{equation}\label{eq:substitution}\begin{split}
        x(t)&=\sqrt{\frac{c_1}{c_2}}\frac{v(t)-\frac{1}{v(t)}}{2} \\
        \dot{x}(t)&=\dot{v}(t)\sqrt{\frac{c_1}{c_2}}\frac{1+\frac{1}{v^2(t)}}{2}=\dot{v}(t)\sqrt{\frac{c_1}{c_2}}\frac{v(t)+\frac{1}{v(t)}}{2v(t)} \\
        1+\frac{c_2}{c_1}x^2(t)&=1+\left(\frac{v(t)-\frac{1}{v(t)}}{2}\right)^2=\left(\frac{v(t)+\frac{1}{v(t)}}{2}\right)^2
    \end{split}\end{equation}

    The left hand side of equation \eqref{eq:integral} becomes

    \begin{equation}\label{eq:integral sub}
        \begin{split}
        \int \frac{\dot{x}(t)dt}{x(t)\sqrt{1+\frac{c_2}{c_1} x^2(t)}}&=\int \frac{\dot{v}(t)\sqrt{\frac{c_1}{c_2}}\frac{v(t)+\frac{1}{v(t)}}{2v(t)}dt}{\sqrt{\frac{c_1}{c_2}}\frac{v(t)-\frac{1}{v(t)}}{2}\frac{v(t)+\frac{1}{v(t)}}{2}}\\
        &=\int \frac{2\dot{v}(t)dt}{v^2(t)-1}=\int\frac{\dot{v}(t)dt}{v(t)-1}-\int\frac{\dot{v}(t)dt}{v(t)+1}\\
        &=\log\left(\frac{v(t)-1}{v(t)+1}\right)+C
        \end{split}
    \end{equation}

    Using equation \eqref{eq:integral sub} and exponentiating both sides of equation \eqref{eq:integral} we obtain

    \begin{equation}
        \frac{v(t)-1}{v(t)+1}=e^{\sqrt{c_1}t+c}.
    \end{equation}

    By rearranging, we get
    
    \begin{equation}
        v(t)=\frac{1+e^{\sqrt{c_1}t+c}}{1-e^{\sqrt{c_1}t+c}}.
    \end{equation}     

    Using substitution \eqref{eq:substitution} 
    
    \begin{equation}\begin{split}\label{eq:x'^2 x}
        x(t)&=\sqrt{\frac{c_1}{c_2}}\frac{\frac{1+e^{\sqrt{c_1}t+c}}{1-e^{\sqrt{c_1}t+c}}-\frac{1-e^{\sqrt{c_1}t+c}}{1+e^{\sqrt{c_1}t+c}}}{2}\\
        &=\sqrt{\frac{c_1}{c_2}}\frac{2e^{\sqrt{c_1}t+c}}{1-e^{2(\sqrt{c_1}t+c)}}\\
        &=\sqrt{\frac{c_1}{c_2}}\frac{2c_3e^{\sqrt{c_1}t}}{1-c_3^2 e^{2\sqrt{c_1}t}}
        \\
        &=\sqrt{\frac{c_1}{c_2}}\frac{1}{\sinh({\sqrt{c_1}t+c})},
    \end{split}\end{equation}
    where $c_3\in{\mathbb C}$, and $c_3^2\neq 1$. This function has a pole at $t_0$ exactly when 



    \begin{equation}
        \sqrt{c_1}t_0+c=ik\pi,
    \end{equation}

    where $k\in{\mathbb Z}$. So

    \begin{equation}\label{eq:poles}
        t_0=\frac{ik\pi-c}{\sqrt{c_1}}
    \end{equation}
    
    From now on, suppose that $c_1,c_2\in\mathbb R$. For $t_0$ in \eqref{eq:poles} to be purely imaginary, either $c\in\mathbb R$, and $c_1<0$, or $c\in i\mathbb R$, and $c_1>0$. 

    The value of $c$ depends on the initial condition in the following way according to equation \eqref{eq:x'^2 x} 
    
    \begin{equation}
        \sinh(c)=\sqrt\frac{c_1}{c_2x(0)^2}.
    \end{equation}

    The right hand side is either purely imaginary, or real, since by our assumption $c_1,c_2,x(0)\in\mathbb R$. Recall that $\sinh(c)=\sinh(a)\cos(b)+i\cosh(a)\sin(b)$, where $c=a+ib\in{\mathbb C}$. It is easy to see from this that 

    \begin{equation}
        \begin{split}
            \sinh(c)\in i\mathbb R,\  |\sinh(c)|\geq1&\iff b=\frac{\pi}{2}+k\pi,\text{ where } k\in{\mathbb Z},\\
            \sinh(c)\in i\mathbb R,\  |\sinh(c)|\leq 1&\iff a=0 \\
            \sinh(c)\in \mathbb R\setminus\{0\},&\iff b=k\pi,\text{ where } k\in{\mathbb Z},\text{ and } a\neq 0
        \end{split}
    \end{equation}

    If $c_1,c_2>0$, then $b=k\pi$, $a\neq 0$. Hence $t_0$ takes no purely imaginary values.

    If $c_1>0$, $c_2<0$, then $c\in i\mathbb R$ if $|\sqrt\frac{c_1}{c_2x(0)^2}|\leq 1$, which holds for $x(0)^2\geq |\frac{c_1}{c_2}|$. Therefore $x(t)$ will have an imaginary pole if $x(0)$ is big enough.

    If $c_1<0$, $c_2>0$, then $c\not\in\mathbb R$, therefore Equation \eqref{eq:poles} takes no purely imaginary values.

    If $c_1,c_2<0$ then $c\in \mathbb R+ki\pi$, where $k\in\mathbb Z$, therefore Equation \eqref{eq:poles} has purely imaginary solutions.

According to \cite{G96} a pseudo-Riemannian metric $q$ on $E(2)$ is always complete in the real sense.

\textit{Case 1.} $q|_V$ is definite. Then for a trajectory $x(t)=x_0(t)e_0+x_1(t)e_1+x_2(t)e_2$, the Euler-Arnold equations are 

\begin{equation}\begin{split}\label{eq:e2 nondeg EA}
    \dot{x}_0(t)&=\frac{\lambda_2-\lambda_1}{\lambda_0}x_1(t)x_2(t) \\
    \dot{x}_1(t)&=-\frac{\lambda_2}{\lambda_1}x_0(t)x_2(t) \\
    \dot{x}_2(t)&=\frac{\lambda_1}{\lambda_2}x_0(t)x_1(t).
\end{split}\end{equation}

It is easy to see that if $\lambda_1=\lambda_2$ then the trajectories are ${\mathbb C}$-complete. Since then $x_0(t)$ remains constant and $x_1(t)$ and $x_2(t)$ form a harmonic oscillator.

In the following, we will suppose that $\lambda_1\neq\lambda_2$. To disprove $i\mathbb R$-completeness, we will construct a trajectory which is not $i\mathbb R$-complete using first integrals and carefully choosing the initial values.

This system of equations has 2 first integrals:

\begin{equation}\begin{split}\label{eq:e2 nondeg first integrals}
    \lambda_0x_0^2(t)+\lambda_1x_1^2(t)+\lambda_2x_2^2(t)=&e \\
    \lambda_1^2 x_1^2(t)+\lambda_2^2 x_2^2(t)=&m\ 
\end{split}\end{equation}

Since $\lambda_1,\lambda_2, \lambda_0\neq 0$, we can express $x_2(t)$ and $x_1(t)$ from $x_0(t)$ in the following way:

\begin{equation}\begin{split}\label{eq:e2 nondeg x_2 x_0}
    \lambda_1e-m&=\lambda_1\lambda_0x_0^2(t)+(\lambda_1\lambda_2-\lambda_2^2)x_2^2(t)\\
    x_2^2(t)&=\frac{\lambda_1e-m-\lambda_1\lambda_0x_0^2(t)}{\lambda_2(\lambda_1-\lambda_2)}
\end{split}\end{equation}

and

\begin{equation}\begin{split}\label{eq:e2 nondeg x_1 x_0}
    \lambda_2e-m&=\lambda_2\lambda_0x_0^2(t)+(\lambda_2\lambda_1-\lambda_1^2)x_1^2(t)\\
    x_1^2(t)&=\frac{\lambda_2e-m-\lambda_2\lambda_0x_0^2(t)}{\lambda_1(\lambda_2-\lambda_1)}.
\end{split}\end{equation}

Now, to proceed, let's calculate $\dot{x}_0^2(t)$ by substituting equation \eqref{eq:e2 nondeg x_1 x_0} and \eqref{eq:e2 nondeg x_2 x_0} into \eqref{eq:e2 nondeg EA} to yield

\begin{equation}\begin{split}
    \dot{x}_0^2(t)&=\left( \frac{\lambda_2-\lambda_1}{\lambda_0}\right)^2\frac{\lambda_2e-m-\lambda_2\lambda_0x_0^2(t)}{\lambda_1(\lambda_2-\lambda_1)}\frac{\lambda_1e-m-\lambda_1\lambda_0x_0^2(t)}{\lambda_2(\lambda_1-\lambda_2)}=\\
    &=-x_0^4(t)+\frac{(\lambda_2+\lambda_1)e-2m}{\lambda_0}x_0^2(t)-\frac{(\lambda_1e-m)(\lambda_2e-m)}{\lambda_0^2\lambda_1\lambda_2}.
\end{split}\end{equation}

If $(\lambda_1e-m)(\lambda_2e-m)=0$, then we can use Theorem \ref{thm:x'^2} and conclude that a trajectory belonging to the given initial condition is not $i\mathbb R$-complete.

We need to prove that there exist initial conditions such that $\lambda_1e=m$ or $\lambda_2e=m$.



With the choice of

\begin{equation}
    x_0(0)=x_{j}(0)=0,\quad x_{3-j}(0)\neq 0, \quad \text{for } j=1,2
\end{equation}


$(\lambda_1e-m)(\lambda_2e-m)=0$ is satisfied so the corresponding trajectory, and geodesic, is not $i\mathbb R$-complete.



\textit{Case 2.} If $q|_V$ is indefinite, we can form a basis, for which the Euler-Arnold equations are:

\begin{equation}\begin{split}
    \dot{x}_0(t)&=x_0(t)x_2(t) \\
    \dot{x}_1(t)&=-(x_0(t)+x_1(t))x_2(t) \\
    \dot{x}_2(t)&=-x_0^2(t),
\end{split}\end{equation}

where $x(t)=x_0(t)e_0+x_1(t)e_1+x_2(t)e_2$. 

In this case the two first integrals are:

\begin{equation}\begin{split}
    x_2^2(t)-2x_0(t)x_1(t)&=e \\
    x_0^2(t)+x_2^2(t)&=m
\end{split}\end{equation}

From these two expressions, we can easily deduce that

\begin{equation}
    \dot{x}_2(t)=x_2^2(t)-m.
\end{equation}

Thus $x_2(t)$ is not $i\mathbb R$-complete when $m>0$ and $x_2(0)=0$ according to Proposition \ref{P:t^2+a}. The trajectory belonging to an initial condition choice of $x_0(0)\neq 0$ and $x_2(0)=0$ is not $i\mathbb R$-complete, and therefore neither is this family of pseudo-Riemannian metrics.

where $\lambda_0=q(e_0,e_0)$. Our method of proof is similar to the case of $E(2)$ we show that for certain initial conditions the corresponding trajectories are not $i\mathbb R$-complete.

From the two first integrals, one can express $x_1(t)$ and $x_2(t)$ as functions in terms of each other such that

\begin{equation}\label{eq: diag x1 x2}
    x_1^2(t)=\frac{m+\lambda_2^2x_2^2(t)}{\lambda_1^2},\quad \text{ and}\quad x_2^2(t)=\frac{-m+\lambda_1^2x_1^2(t)}{\lambda_2^2}.
\end{equation}

Therefore by combining equation \eqref{eq: diag first integrals} and \eqref{eq: diag x1 x2} we get that

\begin{equation}
    e=\lambda_0x_0^2(t)+\frac{m}{\lambda_1}+\frac{\lambda_2(\lambda_2-\lambda_1)}{\lambda_1}x_2^2(t)=
\lambda_0x_0^2(t)+\frac{m}{\lambda_2}+\frac{\lambda_1(\lambda_2-\lambda_1)}{\lambda_2}x_1^2(t).
\end{equation}

By expressing $x_1^2(t)$ and $x_2^2(t)$ we obtain

\begin{equation}\label{eq: diag xi x0}
\begin{split}
    x_2^2(t)&=\left(e-\frac{m}{\lambda_1}-\lambda_0x_0^2(t)\right)\frac{\lambda_1}{\lambda_2(\lambda_2-\lambda_1)},\\
    &\text{and} \\
    x_1^2(t)&=\left(e-\frac{m}{\lambda_2}-\lambda_0x_0^2(t)\right)\frac{\lambda_2}{\lambda_1(\lambda_2-\lambda_1)}.
\end{split}    
\end{equation}

Combining equation \eqref{eq: diag xi x0} and \eqref{eq: diag EA} we get the following ODE for $x_0(t)$:

\begin{equation}
    \dot{x}_0^2(t)=\left(\frac{e}{\lambda_0}-\frac{m}{\lambda_0\lambda_1}-x_0^2(t)\right)\left(\frac{e}{\lambda_0}-\frac{m}{\lambda_0\lambda_2}-x_0^2(t)\right)
\end{equation}

With the choice of $x_1(0)=\lambda_2$ and $x_2(0)=\lambda_1$, $m=0$ using the continuity of $\dot x_0$ the either

\begin{equation}
    \dot{x}_0(t)=\frac{e}{\lambda_0}-x_0^2(t)\quad\text{or}\quad\dot{x}_0(t)=-\left(\frac{e}{\lambda_0}-x_0^2(t)\right)
\end{equation}

If $\frac{e}{\lambda_0}>0 $ and $x_0(0)=0$, then the corresponding trajectory is not $i\mathbb R$-complete. With these choices of initial conditions, $\frac{e}{\lambda_0}$ becomes

\begin{equation}
    \frac{e}{\lambda_0}=x_0^2(0)+\frac{\lambda_1}{\lambda_0}\lambda_2^2-\frac{\lambda_2}{\lambda_0}\lambda_1^2=\frac{\lambda_1\lambda_2}{\lambda_0}(\lambda_2-\lambda_1).
\end{equation}

If $\frac{\lambda_1(\lambda_1-\lambda_2)}{\lambda_0\lambda_2}>0$, this gives a not $i\mathbb R$-complete trajectory by Proposition \ref{P:t^2+a}. Otherwise, we can combine equation \eqref{eq: diag x1 x2}, \eqref{eq: diag xi x0} and \eqref{eq: diag EA} we arrive at the following ODE for $x_1(t)$

\begin{equation}\label{eq: diag x1}
    \dot{x}_1^2(t)=\frac{\lambda_2^2}{\lambda_1^2}\left(\frac{-m+\lambda_1^2x_1^2(t)}{\lambda_2^2}\right)\left(\frac{\lambda_2e-m}{\lambda_0\lambda_2}+\frac{\lambda_1(\lambda_1-\lambda_2)}{\lambda_0\lambda_2}x_1^2(t)\right)
\end{equation}

If $\frac{\lambda_1(\lambda_1-\lambda_2)}{\lambda_0\lambda_2}<0$, the trajectory belonging to $x_0(0)=x_1(0)=0\neq x_2(0)$ has $\lambda_2e-m=0$, so equation \eqref{eq: diag x1} becomes

\begin{equation}
    \dot{x}_1^2(t)=\left(-m\frac{\lambda_2^2}{\lambda_1^2}+x_1^2(t)\right)\frac{\lambda_1(\lambda_1-\lambda_2)}{\lambda_0\lambda_2}x_1^2(t).
\end{equation}
Using Theorem \ref{thm:x'^2}, the corresponding trajectory is not $i\mathbb R$-complete.

\textit{Case 2}. The matrix of $\phi$ is nondegenerate and nondiagonalizable. Due to \cite{G96} there exists a basis $\{e_1,e_2,e_3\}$, such that for a trajectory $x(t)=x_1(t)e_1+x_2(t)e_2+x_3(t)e_3$, the Euler-Arnold equations become

\begin{equation}\begin{split}\label{eq: nondiag EA}
    \dot{x}_0(t)&=\frac{a}{2\lambda_0}(x_1(t)-x_2(t))^2 \\
    \dot{x}_1(t)&=-\frac{a}{\lambda}x_0(t)x_1(t)+\left(\frac{a}{\lambda}-1\right)x_0(t)x_2(t) \\
    \dot{x}_2(t)&=-\left(\frac{a}{\lambda}+1\right)x_0(t)x_1(t)+\frac{a}{\lambda}x_0(t)x_2(t) 
\end{split}\end{equation}

with first integrals

\begin{equation}\begin{split}
    e&=\lambda_0x_0^2(t)+(a/2+\lambda)x_1^2(t)+(a/2-\lambda)x_2^2(t)-ax_1x_2(t) \\
    m&=\lambda_0x_0^2(t)+\lambda/2(x_1^2(t)-x_2^2(t)),
\end{split}\end{equation}

Notice that if we substitute $m$ into the definition of $e$, we obtain the following identity:

\begin{equation}\label{eq: nondiag e m}
    e=-\lambda_0x_0^2(t)+2m+\frac{a}{2}(x_1(t)-x_2(t))^2.
\end{equation}

Combining equation \eqref{eq: nondiag EA} and \eqref{eq: nondiag e m} we get

\begin{equation}
    \dot{x}_0(t)=\frac{e-2m}{\lambda_0}+x_0^2(t).
\end{equation}

If $\frac{e-2m}{\lambda_0}<0$, $x_0(0)=0$, then using Proposition \ref{P:t^2+a} the corresponding trajectory is not $i\mathbb R$-complete.

\begin{equation}
    \frac{e-2m}{\lambda_0}=\frac{a}{2\lambda_0}(x_1(0)-x_2(0))^2,
\end{equation}

which is negative iff $\frac{a}{2\lambda_0}<0$. In this case, the trajectory belonging to the initial condition $x(0)=0$, $x_1(0)\neq x_2(0)$ is not $i\mathbb R$-complete.

If $\frac{a}{2\lambda_0}>0$ then $x_0(t)$ is $i\mathbb R$-complete. Therefore we can use Proposition \ref{P:l-L-Omega} for the Euler-Arnold equation with the setting $\langle e_0\rangle=W$, $V=V$ and conclude that the connection in this case is $i\mathbb R$-complete.

 
\textit{Case 3}. $\phi$ has complex eigenvalues. Then there exists a basis $\{e_1,e_2,e_3\}$, such that for a trajectory $x(t)=x_1(t)e_1+x_2(t)e_2+x_3(t)e_3$, Euler-Arnold equations are

\begin{equation}\begin{split}\label{eq: complex EA}
    \dot{x}_0(t)&=-\frac{\beta}{\lambda_0}(x_1^2(t)+x_2^2(t)) \\
    \dot{x}_1(t)&=x_0(t)(ax_1(t)+bx_2(t)) \\
    \dot{x}_2(t)&=x_0(t)(bx_1(t)-ax_2(t)),
\end{split}\end{equation}

where $a=\frac{2\alpha\beta}{\alpha^2+\beta^2},$ and $b=\frac{\beta^2-\alpha^2}{\alpha^2+\beta^2}$, with $\alpha, \beta\in\mathbb R$ and $\beta\neq 0$. This system of equations has two first integrals:

\begin{equation}\begin{split}\label{eq: complex e m}
    e&=\lambda_0 x_0^2(t)+\alpha(x_1^2(t)-x_2^2(t))+2\beta x_1x_2 \\
    m&=b\lambda_0 x_0^2(t)+2\beta x_1x_2(t).
\end{split}\end{equation}

First, suppose that $\alpha\neq 0$. Then, by combining equation \eqref{eq: complex e m} and \eqref{eq: complex EA} we obtain the following ODE for $x_0(t)$:

\begin{equation}\begin{split}\label{eq:e(1,1) c-ev x_0}
    \dot{x}_0^2(t)&=\frac{\beta^2}{\lambda_0^2}\left((x_1^2(t)-x_2^2(t))^2+4x_1^2(t)x_2^2(t)\right) \\
    &=\frac{\beta^2}{\lambda_0^2}\left(\left(\frac{e-\lambda_0x_0^2(t)-(m-b\lambda_0x_0^2(t))}{\alpha}\right)^2+4\left(\frac{m-b\lambda_0x_0^2(t)}{2\beta}\right)^2\right).
\end{split}\end{equation}

If $(b-1)m=b(m-e)$, then $\dot{x}_0^2(t)$ is a complete square and \eqref{eq:e(1,1) c-ev x_0} becomes a Riccati-type equation. If we look at a trajectory for which $m=be$ then 

\begin{equation}
    (\dot{x}_0(t)-(x_0^2(t)-\frac{e}{\lambda_0}))(\dot{x}_0(t)+(x_0^2(t)-\frac{e}{\lambda_0}))=0.
\end{equation}

By the continuity of $\dot x_0$, either

\begin{equation}
    \dot{x}_0(t)=x_0^2(t)-\frac{e}{\lambda_0}\quad\text{ or}\quad\dot{x}_0(t)=-(x_0^2(t)-\frac{e}{\lambda_0})
\end{equation}
Using Proposition \ref{P:t^2+a} $x_0$ is not $i\mathbb R$-complete if $\frac{e}{\lambda_0}>0$ and $x_0(0)=0$.

To find a trajectory such that $m=be$ we use their definition \eqref{eq: complex e m} and divide both sides by $\frac{\alpha}{\alpha^2+\beta^2}$, then after simplification we obtain the following:

\begin{equation}\label{eq: m=be}
    4\alpha\beta x_1(t)x_2(t)=(\beta^2-\alpha^2)(x_1^2(t)-x_2^2(t)).    
\end{equation}

If $\frac{\beta}{\lambda_0}>0$, then we can choose $x_1(0)=\alpha+\beta$ and $x_2(0)=\beta-\alpha$. This satisfies equation \eqref{eq: m=be}, so we only need to prove that $\frac e \lambda_0>0$.

\begin{equation}
    \frac{e}{\lambda_0}=\frac{1}{\lambda_0}(4\alpha^2\beta+2\beta^3-2\alpha^2\beta)=2\frac{\beta}{\lambda_0}(\alpha^2+\beta^2),
\end{equation}

so $\frac{e}{\lambda_0}>0$ and $x_0(t)$ is not $i\mathbb R$-complete.

 If $\frac{\beta}{\lambda_0}<0$, then we can choose  $x_1(0)=\alpha-\beta$, $x_2(0)=\alpha+\beta$. Then

\begin{equation}
    \frac{e}{\lambda_0}=\frac{1}{\lambda_0}(-4\alpha^2\beta+2\beta(\alpha^2-\beta^2))=-2\frac{\beta}{\lambda_0}(\alpha^2+\beta^2).
\end{equation}

so $\frac{e}{\lambda_0}>0$ and $x_0(t)$ is not $i\mathbb R$-complete.

Now, let $\alpha=0$. In this case, $e=m$, so we define a new first integral. Notice that
\begin{equation}
    c=\frac{e-m}{\alpha}=x_1^2(t)-x_2^2(t)+\frac{1-b}{\alpha}\lambda_0x_0^2(t)=x_1^2(t)-x_2^2(t)+\frac{2\alpha}{\alpha^2+\beta^2}\lambda_0x_0^2(t)
\end{equation} is a first integral, which is well-defined even for $\alpha=0$.

\begin{equation}\begin{split}\label{eq: complex e c}
    e&=\lambda_0 x_0^2(t)+2\beta x_1(t)x_2(t) \\
    c&=x_1^2(t)-x_2^2(t).
\end{split}\end{equation}

By combining equation $\eqref{eq: complex e c}$ and $\eqref{eq: complex EA}$ we obtain the following ODE for $x_0$:

\begin{equation}\label{eq: alpha0 x0}
    \dot{x}_0^2(t)=\frac{\beta^2}{\lambda_0^2}\left(c^2+4\left(\frac{e-\lambda_0x_0^2(t)}{2\beta}\right)^2\right).
\end{equation}

Now, if we look at a trajectory with initial condition $c=0$, equation \eqref{eq: alpha0 x0} simplifies to:

\begin{equation}
    \dot{x}_0(t)=\pm\left(x_0^2(t)-\frac{e}{\lambda_0}\right)
\end{equation}

If $x_0(0)=0$ and $e/\lambda_0>0$, then by Proposition \ref{P:t^2+a} the corresponding trajectory is not $i\mathbb R$-complete. This means that we need to construct initial conditions such that $c=0$ to prove that these metrics are not $i\mathbb R$-complete. 

If $\beta/\lambda_0>0$, then with the initial condition $x_1(0)=x_2(0)=1$, we have

\begin{equation}
    \frac{e}{\lambda_0}=2\frac{\beta}{\lambda_0}.
\end{equation}

Otherwise, if $\beta/\lambda_0<0$, then we can choose $x_1(0)=-x_2(0)=1$ to have

\begin{equation}
    \frac{e}{\lambda_0}=-2\frac{\beta}{\lambda_0}.
\end{equation}

In both cases, we have not $i\mathbb R$-complete trajectories so neither of the corresponding metric is $i\mathbb R$-complete.

\textit{Case 4}. $\phi$  is degenerate. Then there exists a basis such that the Euler-Arnold equations are

\begin{equation}\begin{split}\label{eq: deg EA}
    \dot{x}_0(t)&=2\lambda x_0(t)x_2(t) \\
    \dot{x}_1(t)&=-x_2(t)(x_0(t)+2\lambda x_1(t)) \\
    \dot{x}_2(t)&=\frac{x_0^2(t)}{2\lambda}
\end{split}\end{equation}

with first integrals

\begin{equation}\begin{split}\label{eq: deg first integral}
    e=&x_0(t)x_1(t)+\lambda x_2^2(t) \\
    m=&x_0^2(t)-4\lambda^2x_2^2(t).
\end{split}\end{equation}

By substituting $m$ into $\dot{x}_2(t)$ we get

\begin{equation}
    \dot{x}_2(t)=\frac{m+4\lambda^2x_2^2(t)}{2\lambda}.
\end{equation}

By using Proposition \ref{P:t^2+a} $x_2$ is not $i\mathbb R$-complete iff $m<0$ and $x_2(0)=0$, but if $x_2(0)=0$, then $m=x_0^2\geq 0$. Therefore $x_2$ is always $i\mathbb R$-complete, hence by applying Propositon \ref{P:l-L-Omega} to equation \eqref{eq: deg EA} we see that all the trajectories are $i\mathbb R$-complete.

    Let $G$ be a solvable, non-unimodular Lie group of dimension 3 with $D=0$ and $\nabla$ a biinvariant torsion-free connection on $TG$. Then $\nabla$ takes the form
    
    \begin{equation}
        \nabla_xy=\frac{1}{2}[x,y]+l(x)y_a+l(y)x_a+(\nabla_x y)_2 f_2.

<!-- proof-5: body_md -->
To prove the statement of the theorem, we only need to determine the class of connections for wich equation \eqref{eq:biinv conn} holds. Suppose we have such a connection $\nabla$. Then by using equation \eqref{eq:lie D0} we obtain
    \begin{equation}\label{eq:biinv D0}
        \begin{split}
            [x,\nabla_y z]&=(x_0(\nabla_y z)_1-(\nabla_y z)_0x_1)f_1 \\
            \nabla_{[x,y]}z+\nabla_y[x,z]&=(x_0y_1-y_0x_1)\nabla_{f_1}z+(x_0z_1-z_0x_1)\nabla_y f_1.
        \end{split}
    \end{equation}

    for $x,y,z\in\mathfrak{g}$. If we choose $x=f_0$ in equation \eqref{eq:biinv D0} then we obtain the following identity

    \begin{equation}\label{eq:D0 x0}
        (\nabla_y z)_1f_1=y_1\nabla_{f_1}z+z_1\nabla_y f_1.
    \end{equation}

    Substituting $y=z=f_1$ we get

    \begin{equation}\label{eq:D0 111}
        (\nabla_{f_1}f_1)_1f_1=2\nabla_{f_1}f_1,
    \end{equation}

     which is therefore equal to $0$. Setting $y=f_1$ $z=f_j$ in equation \eqref{eq:D0 x0} for $j\neq 1$

    \begin{equation}\label{eq:D0 f1}
        (\nabla_{f_1}f_j)_1f_1=\nabla_{f_1}f_j,
    \end{equation}

    hence 
    
    \begin{equation}\label{eq:D0 l0}
        \nabla_{f_1}x=l_0(x)f_1,
    \end{equation} for some $l_0\in\mathfrak{g}^*$, with $l_0(f_1)=0$ because of the equation \eqref{eq:D0 111}. By combining equation \eqref{eq:D0 f1} and torsion-freeness we also get that.
    
    \begin{equation}\label{eq:D0 f1*}
        \nabla_z f_1=[z,f_1]+\nabla_{f_1}z=z_0f_1+l_0(z)f_1
    \end{equation}

    Equations \eqref{eq:D0 l0} and \eqref{eq:D0 f1*} imply that $(\nabla_x y)_0$ and $(\nabla_x y)_2$ do not depend on $x_1$ or $y_1$.
    
    Putting together equations \eqref{eq:D0 x0}, \eqref{eq:D0 l0} and \eqref{eq:D0 f1*} we can see that

    \begin{equation}\label{eq:D0 x1}
        (\nabla_y z)_1f_1=(l_0(y_1z+z_1y)+y_0z_1)f_1.
    \end{equation}


    
    Now, looking at equation \eqref{eq:biinv D0} at $x=f_1$ we get

    \begin{equation}\label{eq:D0 x1 f1}
        (\nabla_y z)_0f_1=y_0\nabla_{f_1}z+z_0\nabla_y f_1=l_0(y_0z+z_0y)f_1+z_0y_0f_1.
    \end{equation}

    Combining equation \eqref{eq:D0 x1} and \eqref{eq:D0 x1 f1} we get that

    \begin{equation}\label{eq:D0 nabla}
        \begin{split}
            \nabla_xy=&(\nabla_x y)_0f_0+(\nabla_x y)_1f_1+(\nabla_x y)_2 f_2 \\
            =&(l_0(x_0y+y_0x)+y_0x_0)f_0+\\&(l_0(x_1y+y_1x)+x_0y_1)f_1+(\nabla_x y)_2 f_2 \\
            =&\frac{1}{2}[x,y]+(l_0(x_0y+y_0x)+y_0x_0)f_0+\\&(l_0(x_1y+y_1x)+\frac{1}{2}x_0y_1+\frac{1}{2}y_0x_1)f_1+
            (\nabla_x y)_2 f_2
        \end{split}        
    \end{equation}

    Denote $\frac{1}{2}f_0^*+l_0\in\mathfrak{g}^*$ by $l$. Then equation \eqref{eq:D0 nabla} simplifies to

    \begin{equation}\label{eq:D0 nabla'}
        \nabla_xy=\frac{1}{2}[x,y]+l(x)y_a+l(y)x_a+(\nabla_x y)_2 f_2.
    \end{equation}

    We only need to prove at this point that for a connection of the form \eqref{eq:D0 nabla'} equation \eqref{eq:biinv conn} always holds.

    \begin{equation}
        \begin{split}
            [x,\nabla_y z]&=\frac{1}{2}[x,[y,z]]+l(y)[x,z]+l(z)[x,y] \\
            \nabla_{[x,y]}z&=\frac{1}{2}[[x,y],z]+l(z)[x,y]\\
            \nabla_y[x,z]&=\frac{1}{2}[y,[x,z]]+l(y)[x,z]
        \end{split}
    \end{equation}

    The sum of the latter two equations clearly equals the first one, thus a connection of the form \eqref{eq:D0 nabla'} is always biinvariant.

The case of $A=I$ was treated in \cite[Theorem 9.1.]{FSz26}. The fact that

    \begin{equation}\label{eq:F-type biinv}
        x_0^2\nabla_{e_0}e_0+x_0(\nabla_{e_0}x_v+\nabla_{x_v}e_0)=x_0(\nabla_{e_0}e_0)_0x
    \end{equation}

    and $(\nabla_{x_v}x_v)_0=0$ can be proven in a similar fashion.

   
    
      To treat $(\nabla_{x_v}x_v)_v$ we choose $x=e_0$, and use equation \eqref{eq:lie e11} and \eqref{eq:biinv conn} to get

    \begin{equation}\label{eq:e11 biinv V}
        A(\nabla_y z)_v=\nabla_{Ay_v} z+\nabla_y Az_v.
    \end{equation}

    

    Using equation \eqref{eq:F-type biinv}

    \begin{equation}\label{eq:general biinv beta}
            \nabla_x x           =x_0(\nabla_{e_0}e_0)_0x+\nabla_{x_v}x_v.     
    \end{equation}

    Since $\nabla$ is torsion-free, the values of $\nabla_xx$ completely determine it completely.

    We also need to show, that the left-invariant connection defined by equation \eqref{eq:general biinv beta} is in fact biinvariant. Denote $(\nabla_{e_0}e_0)_0$ by $a$. To check biinvariance, it suffices to prove that $\nabla$ satisfies equation \eqref{eq:biinv conn}.

    \begin{equation}
        \begin{split}
            [x,\nabla_yz]&=[x,\frac{1}{2}[y,z]+\frac{a}{2}(y_0z+z_0y)+\nabla_{y_v}z_v]\\
            &=\frac{1}{2}[x,[y,z]]+\frac{a}{2}y_0[x,z]+\frac{a}{2}z_0[x,y]+x_0A(\nabla_{y_v}z_v)\\
            \nabla_{[x,y]}z&=\frac{1}{2}[[x,y],z]+\frac{a}{2}z_0[x,y]+\nabla_{x_0Ay_v-y_0Ax_v}z_v\\
            \nabla_y[x,z]&=\frac{1}{2}[y,[x,z]]+\frac{a}{2}y_0[x,z]+\nabla_{y_v}(x_0Az_v-z_0Ax_v).
        \end{split}
    \end{equation}

    The sum of the latter two equations has to equal the first. From Equation \eqref{eq:e11 biinv V} we have that

    \begin{equation}
        \nabla_{y_0Ax_v}z_v+\nabla_{y_v}(z_0Ax_v)=0
    \end{equation}

    for all $x,y,z\in \mathfrak{g}$. With the choice of $y_0=0$, $z_0=1$ and $x_v=A^{-1}z_v$ we obtain

    \begin{equation}
        \nabla_{y_v}z_v=0.
    \end{equation}
    
    And thus, the Euler-Arnold equation is 

    \begin{equation}
        \dot{x}(t)=-a x_0(t)x(t).
    \end{equation}
    
    Proposition~\ref{P:l-L-Omega} implies that such a vector field is always $i\mathbb R$-complete, and by Theorem \ref{T:nabla-i-complete}   the connection is $\mathcal P$-complete.
