<!-- theorem-1: body_md -->
\label{Thurston's theorem}
    (\cite{Culler_Shalen}, 3.2.1.) Let $M$ be a noncompact hyperbolic 3-manifold of finite volume with $d$ cusps and set $\Gamma=\pi_1(M)$. Then any canonical component $X_0(\Gamma)$ is a $d$-dimensional complex affine algebraic variety. <!-- definition-2: body_md -->
\label{various morphisms}
Consider $\phi: X \longrightarrow Y$, then 

\begin{enumerate}
\par\hangindent 40\p@ \textbf{Generically finite.}
Suppose that $\phi$ is dominant, i.e. $\overline{\phi(X)}=Y$. The morphism $\phi$ is \emph{generically finite} if the induced extension of function fields
$\mathbb C(Y)=\operatorname{Frac}(B)\ \longrightarrow \mathbb C(X)=\operatorname{Frac}(A)$
is finite. In our case, under the dominance assumption, the condition $\dim X=\dim Y$ already implies that $\phi$ is generically finite. When $\phi$ is generically finite, the degree of $\phi$ is defined to be $[\mathbb C(X):\mathbb C(Y)]$. 

\par\hangindent 40\p@ \textbf{Finite.}
The morphism $\phi$ is \emph{finite} if $A$ is a finitely generated
$B$-module.

\par\hangindent 40\p@ \textbf{Finite flat.}
The morphism $\phi$ is \emph{finite flat} if $A$ is a finite and
flat $B$-module.

\par\hangindent 40\p@ \textbf{Finite étale.}
The morphism $\phi$ is \emph{finite étale} if it is finite flat and,
for every closed point $y\in Y$, the scheme-theoretic fiber $X_y=\operatorname{Spec}\bigl(A\otimes_B\kappa(y)\bigr)$ is reduced. Since $\kappa(y)=\mathbb C$, this means that $A\otimes_B\kappa(y)\cong\mathbb C^{\,r}$
for some integer $r\geq 0$. 
\end{enumerate}

<!-- definition-3: body_md -->
\label{def of fiber multiplicity and total length}
Let $y\in Y$ be a closed point, corresponding to a
maximal ideal $\mathfrak m_y\subset B$, and suppose that the
set-theoretic fiber $\phi^{-1}(y)$ is finite. Set $R_y:=A/\mathfrak m_yA.$ For $x\in\phi^{-1}(y)$, let $\mathfrak m_x\subset A$ be its
corresponding maximal ideal. The \emph{multiplicity of $x$ in the
fiber over $y$} is
$$
e_x(y):=\dim_{\mathbb C}(R_y)_{\mathfrak m_x/\mathfrak m_yA}.
$$
The \emph{total length of the fiber over $y$} is
$$
L(y):=\sum_{x\in\phi^{-1}(y)}e_x(y)
     =\dim_{\mathbb C}R_y.
$$
If the fiber is empty, we set $L(y)=0$.
<!-- end definition-3 -->
<!-- proposition-1: body_md -->
\label{characterization of morphisms using fiber}
For each closed point $y\in Y$, consider the set-theoretic fiber $\phi^{-1}(y)$. When this fiber is finite, let $e_x(y)$ denote the multiplicity of $x\in\phi^{-1}(y)$, and set $L(y)=\sum_{x\in\phi^{-1}(y)}e_x(y).$ Then:
\begin{enumerate}
\par\hangindent 40\p@ $\phi$ is finite if and only if it is proper and
      $\phi^{-1}(y)$ is a finite set for every closed point $y\in Y$.

\par\hangindent 40\p@ $\phi$ is finite flat if and only if it is finite and
      $L(y)$ is constant as $y$ ranges over the closed points of $Y$.

\par\hangindent 40\p@ $\phi$ is finite étale if and only if it is finite and
      there is an integer $r\geq 1$ such that, for every closed
      point $y\in Y$, $\#\phi^{-1}(y)=r$ and $e_x(y)=1$ for every $x\in\phi^{-1}(y)$.
\end{enumerate}

<!-- definition-5: body_md -->
\label{Cohen--Macaulay}
$X=\operatorname{Spec}(A)$ is \emph{Cohen--Macaulay} if for every prime ideal
$\mathfrak p\subseteq A$, $\operatorname{depth}(A_{\mathfrak p})=\dim(A_{\mathfrak p}).$ Here $\dim(A_{\mathfrak p})$ is the Krull dimension of the local ring $A_{\mathfrak p}$, and $\operatorname{depth}(A_{\mathfrak p})$ is the maximum length of an $A_{\mathfrak p}$-regular sequence contained in
its maximal ideal $\mathfrak pA_{\mathfrak p}$.

<!-- lemma-1: body_md -->
{(trace identities)}\label{trace identities} For any $U,V\in\operatorname{SL}_2(\mathbb C)$, we have

    (1). $\operatorname{tr}(UV)=\operatorname{tr}(VU), ~~\operatorname{tr}(U)=\operatorname{tr}(U^{-1})$

    (2). $\operatorname{tr}(UV)+\operatorname{tr}(UV^{-1})=\operatorname{tr}(U)\operatorname{t}(V)$

    (3). $\operatorname{tr}(U_1U_2\cdots U_k)=\operatorname{tr}(U_2\cdots U_kU_1)$

    (4). $\operatorname{tr}([U,V])=\operatorname{tr}^{2}(U)+\operatorname{tr}^{2}(V)+\operatorname{tr}^{2}(UV) -\operatorname{tr}(U)\operatorname{tr}(V)\operatorname{tr}(UV)-2$

    (5). $\operatorname{tr}(U^k) =2S_{k}(\operatorname{tr}(U)) -\operatorname{tr}(U)S_{k-1}(\operatorname{tr}(U))$. In particular, $\operatorname{tr}(U^2) =\operatorname{tr}^2(U) -2$.

<!-- lemma-2: body_md -->
\label{trace functions of n and l}
Let $\phi: X=X_0(M) \longrightarrow Y=X(T)$ be the boundary restriction map. Then the coordinate maps $m(x,y,z),\ell(x,y,z),n(x,y,z)$ for $\phi$ can be given as follows 
$$
m=\operatorname{tr}(A)=x,
$$
$$
n=\operatorname{tr}(W)=(x-yz)\kappa-x+2yz,
$$
and 
$$
\ell=\operatorname{tr}(WA^{-1})=\kappa^2-y^2\kappa+2y^2-2.
$$
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
We begin with some trace identities. Using \ref{trace identities} (5), we have 
$$
\operatorname{tr}(A^2)=x^2-2,\qquad\operatorname{tr}(B^2)=y^2-2,\qquad 
\operatorname{tr}\bigl((AB)^2\bigr)=z^2-2.
$$
We also use Cayley-Hamilton to derive
$$
A^2=\operatorname{tr}(A)A-I=x \cdot A-I,\qquad B^2=\operatorname{tr}(B)B-I=y \cdot B-I.
$$
Therefore,
$$
\operatorname{tr}(A^2B)=xz-y, \qquad  \operatorname{tr}(AB^2)=yz-x.
$$


The fact that $m=x$ follows easily from $a=\mu$. We then compute $n=\operatorname{tr}(W)$. Write $C=BAB^{-1}A^{-1},~ D=B^{-1}AB.$ Then $W=CD.$ By \ref{trace identities} (3), it is clear that $\operatorname{tr}(D)=\operatorname{tr}(A)=x.$ Using \ref{trace identities} (2), we have $\operatorname{tr}(CD)+\operatorname{tr}(CD^{-1})=\operatorname{tr}(C)\operatorname{tr}(D).$ Thus 
$$
n=x \cdot \operatorname{tr}(C)-\operatorname{tr}(CD^{-1}).
$$ 
By \ref{trace identities} (4), $\operatorname{tr}(C)=x^2+y^2+z^2-xyz-2=\kappa.$ It remains to compute $S:=\operatorname{tr}(CD^{-1}).$ Since $D^{-1}=B^{-1}A^{-1}B,$ we have $CD^{-1}=BAB^{-1}A^{-1}B^{-1}A^{-1}B.$
Using \ref{trace identities} (3) and Cayley-Hamilton with $B^2=\operatorname{tr}(B)B-I$, we have
$$
S=\operatorname{tr}\left(B^2AB^{-1}A^{-1}B^{-1}A^{-1}\right)=y\cdot \operatorname{tr}(BAB^{-1}A^{-1}B^{-1}A^{-1}) -\operatorname{tr}(AB^{-1}A^{-1}B^{-1}A^{-1}).
$$
Then using \ref{trace identities} (2) and (1), we have $\operatorname{tr}(AB^{-1}A^{-1}B^{-1}A^{-1})=\operatorname{tr}(AB^2)=yz-x$. Separate $BAB^{-1}A^{-1}B^{-1}A^{-1}$ into the product of $BAB^{-1}A^{-1}$ and $B^{-1}A^{-1}$. Then applying \ref{trace identities} (2), we can derive $\operatorname{tr}(BAB^{-1}A^{-1}B^{-1}A^{-1})=\kappa z-z$. Collecting all the identities gives 
$$
n=(x-yz)\kappa-x+2yz. 
$$ 

Now, we want to compute $\ell=\operatorname{tr}(WA^{-1})$. Notice that $WA^{-1}=BAB^{-1}A^{-1}B^{-1}ABA^{-1}$. Separate $WA^{-1}$ into the product of $BAB^{-1}A^{-1}$ and $B^{-1}ABA^{-1}$. Then we can apply \ref{trace identities} (2) and (3) to derive
$$
\operatorname{tr}(WA^{-1})=\operatorname{tr}(BAB^{-1}A^{-1})\operatorname{tr}(B^{-1}ABA^{-1}) - \operatorname{tr}(B^{2}AB^{-2}A^{-1})
$$
Each trace term on the right-hand side of the equation above can be evaluated using \ref{trace identities} (4) and the identities $\operatorname{tr}(B^{-1}A)=xy-z,~\operatorname{tr}(B^2)=y^2-2,~\operatorname{tr}(B^2A)=yz-x$. After direct computations, we can derive 
$$
\ell=\kappa^2-y^2\kappa+2y^2-2.
$$
This completes the proof.

<!-- lemma-3: body_md -->
\label{quadratic extension of K over C(x,kappa)}
Let $K$ be the function field of $X_{0}(M)$ and view $x,\kappa$ as elements in $K$. Then 

(1). $[K:\mathbb C(x,\kappa)]=2.$

(2). $\operatorname{trdeg}_{\mathbb C}\mathbb C(x,\kappa)=2$. Therefore, $x$ and $\kappa$ are algebraically independent over $\mathbb C$.
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
We first determine the degree of the function field $K$ over
$\mathbb C(x,\kappa)$. Since $z\kappa=xy$ on $X_0(M)$, we have, in the function field $K$, $y=\frac{\kappa z}{x}.$  Substituting $y=\frac{\kappa z}{x}$ into $\kappa=x^2+y^2+z^2-xyz-2$ and collecting the terms gives
$$
z^2=\frac{x^2(\kappa-x^2+2)}{\kappa^2-x^2\kappa+x^2} \in\mathbb C(x,\kappa). 
$$
Moreover, from $y=\frac{\kappa z}{x}.$ It follows that $K=\mathbb C(x,\kappa)(z),$ and hence $[K:\mathbb C(x,\kappa)]\leq 2.$

We now show that the extension has degree exactly $2$. Consider the involution map $\tau(x,y,z)=(x,-y,-z)$. $X_0(M)$ is defined as the vanishing locus of $F(x,y,z)=z^3-xyz^2+(x^2+y^2-2)z-xy$. Since $F(x,-y,-z)=-F(x,y,z)$, $X_0(M)$ is preserved by $\tau$ and hence $\tau$ induces an automorphism $\tau_{*}$ on $K$. Notice that $\tau_{*}(x)=x$ and $\tau_{*}(\kappa)=\kappa$. Therefore, $\tau_{*}$ fixes $\mathbb C(x,\kappa) \subset K$. On the other hand, $\tau_{*}(z)=-z\neq z$. So $z \notin \mathbb C(x,\kappa)$ and $[K:\mathbb C(x,\kappa)] = 2.$

In particular, $K$ is a finite algebraic extension of $\mathbb C(x,\kappa)$. Since $X_{0}(M)$ is an algebraic surface, $\operatorname{trdeg}_{\mathbb C}K=2.$ It follows that $\operatorname{trdeg}_{\mathbb C}\mathbb C(x,\kappa)=2.$ Thus $x$ and $\kappa$ are algebraically independent over $\mathbb C$.
<!-- end proof-2 -->
<!-- lemma-4: body_md -->
\label{C(m,l,n)=C(x,kappa)}
$\mathbb C(m,\ell,n)=\mathbb C(x,\kappa).$
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
from lemma \ref{trace functions of n and l}, we can write $n,\ell \in \mathbb C(x,y,z)$ as $n(x,y,z)=(x-yz)\kappa-x+2yz$ and $\ell(x,y,z)=\kappa^2-y^2\kappa+2y^2-2$. We now rewrite $n,\ell $ in terms of $x$ and $\kappa$, when view them as elements in the function field of $X_0(M)$. Notice that along $X_0(M)$, we have $y=\frac{\kappa z}{x},~ z^2=\frac{x^2(\kappa-x^2+2)}{\kappa^2-x^2\kappa+x^2}$. Substituting these identities into $n(x,y,z)$ and $\ell(x,y,z)$, we obtain
\begin{equation}\label{n l in terms of x kappa}
    n=x\,\frac{-\kappa^2+4\kappa-x^2}{\kappa^2-x^2\kappa+x^2},\qquad \ell=\frac{(2-x^2)\kappa^2+2x^2\kappa-2x^2}{\kappa^2-x^2\kappa+x^2}.
\end{equation}
Since $m=x,$ it follows that $m,\ell,n\in \mathbb C(x,\kappa).$ Therefore, $\mathbb C(m,\ell,n)\subseteq \mathbb C(x,\kappa).$

We next prove the reverse inclusion $\mathbb C(x,\kappa)\subseteq \mathbb C(m,\ell,n).$ Since $m=x,$ we already have $x\in \mathbb C(m,\ell,n).$ Set $u=x^2,~t=\frac{n}{x}.$ Since $x=m$, both $u$ and $t$ belong to $\mathbb C(m,\ell,n)$. Substituting $u,t$ into $n(x,\kappa)$ of \eqref{n l in terms of x kappa}, we obtain
$$
(t+1)\kappa^2-(tu+4)\kappa+u(t+1)=0.
$$
Similarly, Substituting $u,t$ into $\ell(x,\kappa)$ of \eqref{n l in terms of x kappa}, we obtain
$$
(\ell+u-2)\kappa^2-u(\ell+2)\kappa+u(\ell+2)=0.
$$
Combining the two equations above gives 
$$
\kappa=\frac{u(t+1)(4-u)}{u(t+1)(\ell+2)-(\ell+u-2)(tu+4)}.
$$

The denominator in the equation above is nonzero since its evaluation at the point $(1,0,0) \in X_0(M)$ is not $0$. It follows that $\kappa\in\mathbb C(m,\ell,n).$ Since $x=m$, we conclude that $\mathbb C(x,\kappa)\subseteq \mathbb C(m,\ell,n).$ Combining this with the reverse inclusion proved above gives $\mathbb C(m,\ell,n)=\mathbb C(x,\kappa),$ as desired.

<!-- proof-4: body_md -->
We first prove that $\phi:X\longrightarrow Y$ is dominant. The function field generated by the coordinate functions of the image of $\phi$ is $\mathbb C(m,\ell,n).$ By the previous section, $\mathbb C(m,\ell,n)=\mathbb C(x,\kappa).$ Since $x$ and $\kappa$ are algebraically independent, $\operatorname{trdeg}_{\mathbb C}\mathbb C(m,\ell,n)=2.$ Therefore, the Zariski closure of the image of $\phi$ has dimension
$$
\dim\overline{\phi(X)}=\operatorname{trdeg}_{\mathbb C}\mathbb C(m,\ell,n)=2.
$$
On the other hand, $\phi(X)\subseteq Y,$ and $Y=X(T)$ is the irreducible affine surface $Y=V(m^2+\ell^2+n^2-m\ell n-4).$ Thus $\overline{\phi(X)}\subseteq Y.$ Since $Y$ is irreducible of dimension $2$, every proper closed
subset of $Y$ has dimension strictly less than $2$. Hence $\overline{\phi(X)}=Y.$ Therefore, $\phi:X\longrightarrow Y \text{ is dominant}.$

Because $\phi$ is dominant, it induces an inclusion of function
fields $\phi^*:\mathbb C(Y)\hookrightarrow \mathbb C(X).$ Under the present coordinates, $\phi^*\mathbb C(Y)=\mathbb C(m,\ell,n).$ We have proved that $\mathbb C(m,\ell,n)=\mathbb C(x,\kappa).$ Therefore,
$$
\deg\phi=\left[\mathbb C(X):\phi^*\mathbb C(Y)\right]= \left[K:\mathbb C(m,\ell,n)\right]=\left[K:\mathbb C(x,\kappa)\right]=2.
$$
Thus, $\deg\phi=2.$

<!-- lemma-5: body_md -->
\label{finite locus}
$U_{fin}=Y\setminus\left(V_Y(m^2-4)\cup V_Y(\delta)\right)$, where $Y=X(T)$ and $\delta=2-\ell+mn\in \mathbb C[m,\ell,n]$.
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
We first explain why $V_{Y}(m^2-4)$ must be removed from the finite locus. Since $m=x$, the fiber over a point with $m=2$ is contained in the plane section of $X_0(M)$ with $x=2$. Substituting $x=2$ into $F$, we obtain $F(2,y,z)=(z-y)\bigl(z(z-y)+2\bigr).$ Thus $X_0(M)\cap\{x=2\}=C_{1}\cup C_{2}$, where $C_{1}=V(x-2,y-z)$ and $C_{2}=V(x-2,z(z-y)+2)$. On the other hand, the equation of $Y$ restricted to $m=2$ is $G(2,\ell,n)=(\ell-n)^2.$ Thus $Y\cap\{m=2\}=\{(2,t,t):t\in\mathbb C\}.$

On $C_{1}$, we have $y=z$, and hence $\kappa=2.$ Substituting these values into the expressions of $n,\ell$ in lemma \ref{trace functions of n and l} gives $(m,\ell,n)=(2,2,2).$ On $C_{2}$, we have $z(z-y)+2=0$. Let $B=\kappa^2-x^2\kappa+x^2$. It is straightforward to check that on $C_2$, we must have $B \neq 0$. Therefore, when we restrict $\phi$ to $C_2$, $n(x,y,z)$ and $\ell(x,y,z)$ take the form in \eqref{n l in terms of x kappa}. Since $\phi(C_2) \subset Y\cap\{m=2\}$, equating $n =\ell$ in \eqref{n l in terms of x kappa} with $x=2$ gives $\kappa=1$ and $(m,\ell,n)=(2,-2,-2).$ Therefore, 
$$
\phi(X_0(M))\cap\{m=2\}=\{(2,2,2),(2,-2,-2)\}.
$$
Since a finite dominant morphism over an irreducible open subset has closed image containing the generic point, no open neighborhood of such a point can support a finite degree-two restriction of $\phi$. Hence the entire divisor $Y\cap V(m-2)$ must be removed from the finite locus. Similarly, $V_{Y}(m+2)$ must also be removed from the finite locus.

We then explain why $V_Y(\delta)$ must be excluded from the finite locus. The equation $\delta=2-\ell+mn=0$ cuts out a curve in $Y$. Indeed, substituting $\ell=2+mn$ into $m^2+\ell^2+n^2-m\ell n-4=0$ gives $(m+n)^2=0.$ Thus, set-theoretically,
$$
V_Y(\delta)=\left\{(m,\ell,n):n=-m,\quad\ell=2-m^2\right\}.
$$

Now suppose that $\xi=(m,\ell,n)\in V_Y(\delta)$ has a preimage $P=(x,y,z)\in X_{0}(M)$ under $\phi$, i.e. $\phi(x,y,z)=(m,\ell,n)$. From lemma \ref{trace functions of n and l}, the expressions for $m,\ell,n$ in terms of $x,y,z$ under $\phi$ are
$$
m=x, \qquad n=(x-yz)\kappa-x+2yz, \qquad \ell=\kappa^2-y^2\kappa+2y^2-2.
$$
Using these expressions, it is not hard to compute that $\delta B=x^2(4-x^2)$, where $B=\kappa^2-x^2\kappa+x^2$. Since $\delta=0$ along $V_{Y}(\delta)$, a point of $V_Y(\delta)$ can have a preimage only if $m=0,~2,~\text{or}-2.$ Consequently, every point $(m,2-m^2,-m)$ with $m\notin\{0,\pm2\}$ has empty fiber. For the similar reason as above, the entire divisor $V_Y(\delta)$ must be removed from the maximal finite locus.

                                                                                                                                                                                                                                                                       



We now prove $U=U_{fin}$ where $U=Y\setminus\left(V(m^2-4)\cup V(\delta)\right)$. From the arguments above, it suffices to prove that $\phi:\phi^{-1}(U) \longrightarrow U$ is finite. Let $R$ and $A$ be the coordinate rings of $X$ and $Y$ respectively,i.e. $X=\operatorname{spec}(R), Y=\operatorname{spec}(A)$(Note that we are abusing notations here by identifying an affine scheme with the underlying algebraic variety that corresponds to its closed points). The boundary restriction map $\phi: X \longrightarrow Y$ induces a ring homomorphism $\phi^{*}: A \longrightarrow R$, and hence $R$ can be viewed as an A-module via $ar \mapsto \phi^{*}(a)r$. To simplify the notation, we identify $\phi^{*}(a) \in R$ with $a \in A$. Set $s=(m^2-4)\delta$ and let $A_s,R_s$ be the localizations at $s$ of $A,R$ respectively. Thus $\operatorname{Spec}A_{s}=D_Y\bigl((m^2-4)\delta\bigr)=U$ and $\operatorname{spec}(R_{s})=\operatorname{spec}(R\otimes_A A_{s})=\phi^{-1}(U)$. To prove that $\phi:\phi^{-1}(U) \longrightarrow U$ is finite, it suffices to prove that $R_s$ is a finite $A_s$-module. We emphasize that for the following discussion, $x,y,z,m,\ell,n$ should be viewed as elements in $A_s$ or $R_s$, rather than coordinates in $X$ or $Y$. Moreover, $\phi^{*}(m),\phi^{*}(\ell), \phi^{*}(n)$ are given as in lemma \ref{trace functions of n and l}.

Since $x=m$ already lies in $A_{s}$, the localized coordinate ring $R_{s}$ is generated as an $A_{s}$-algebra by $y$ and $z$, i.e. $R_{s}=A_{s}[y,z]$. Then $R_s$ is a finite $A_s$-module if and only if $y$ and $z$ are integral over $A_s$, which we now prove. We first show that $\kappa \in A_s$. Recall that $\delta B=x^2(4-x^2).$ It can also be computed that $(x+n)B=x(4-x^2)\kappa$. Therefore $\kappa=\frac{x(x+n)}{\delta} \in A_s$, since $\delta$ is invertible in $A_s$. 


We now prove that $z$ is integral over $A_s$. Recall that $Bz^2=x^2(\kappa-x^2+2)$ and $\delta B=x^2(4-x^2).$ Since $m^2-4=x^2-4$ is invertible in $A_s$, it can be computed that 
$$
z^2=\frac{\delta(\kappa-x^2+2)}{4-x^2} \in A_s.
$$
Thus $z$ satisfies the monic polynomial $T^2-c\in A_{s}[T]$, and hence $z$ is integral over $A_{s}$. 

We then prove $y$ is integral over $A_s$. Using the fact that $z\kappa = xy$, the formulas for $n$ and $\ell$ can be rewritten as
$$
-xy^2+2yz=n-x\kappa+x,\qquad(2-x^2)y^2+xyz=\ell+(2-x^2)\kappa+2.
$$
We can regard these as two linear equations in the two quantities $y^2$ and $yz$. The coefficient matrix is
$$
\begin{pmatrix}
-x & 2\\
2-x^2 & x
\end{pmatrix},
$$
whose determinant is $x^2-4.$ Since $m^2-4$ is invertible in $A_{s}$, this matrix is invertible after localization. Solving for $y^2$ gives
$$
y^2=\frac{xn+(x^2-4)\kappa+x^2-2\ell-4}{x^2-4} \in A_s.
$$
Thus $y$ also satisfies a quadratic monic polynomial in $A_{s}[T],$ and therefore $y$ is integral over $A_{s}$. This completes the proof.

Let $b=(m,\ell,n)\in X(T)\setminus \Sigma,$ and let $\rho:\pi_1(T)\longrightarrow \mathrm{SL}_2(\mathbb{C})$ be a representation with character $b$. Because $\pi_1(T)$ is abelian, $\rho(\pi_1(T))$ is is an abelian subgroup of $\mathrm{SL}_2(\mathbb{C})$. Up to conjugation, this means $\rho(\pi_1(T))$ is a subgroup of either $D$ or $S \subset \mathrm{SL}_2(\mathbb{C})$, where $D$ is the group of diagonal matrices and $S$ is the group of upper triangular unipotent matrices and their negatives. Therefore, with proper conjugation, we can assume $\rho(\mu),\rho(\lambda)$ take the forms as in \eqref{eigenvalues of mu lambda}.

R_{-+},&p\equiv0\pmod2,\\

\end{cases}

In particular, if $p$ and $r$ are both even, then $R_{-+}\subset C_{p/q}\cap C_{r/s}.$ Otherwise, $R_{-+} \cap C_{p/q}\cap C_{r/s}=\varnothing$.

Since $\operatorname{tr}(A)=x=2$ and $\operatorname{det}(A)=1$, the characteristic polynomial of $A$ is $(t-1)^2$ and hence $A$ is unipotent. If $A=I$, then $z=\operatorname{tr}(AB)=\operatorname{tr}(B)=y$ and $g=0$. This contradicts the fact that $g \neq 0$ on $I_{+-}$, so $A \neq I$. Every nontrivial unipotent matrix in $\operatorname{SL}(2,\mathbb C)$ is conjugate to $I+N$ where 
$$

\begin{pmatrix}

\end{pmatrix}.

\begin{pmatrix}

\end{pmatrix}.

\rho(\lambda)=\rho(wa^{-1})=
\begin{pmatrix}
-1&yz\\
0&-1
\end{pmatrix}
=-\bigl(I-yzN\bigr).

<!-- remark-6: body_md -->
\label{advantage of the algorithm}
The main advantage of algorithm \ref{Algorithm for computing intersection pattern} is that it breaks down the whole procedure of computing the intersection into two independent stages. The first stage involves the first two steps of the algorithm, which depend only on the slope coefficients $\frac{p}{q},\frac{r}{s}$ and are easy to compute. The second stage is to find the fiber classification table(cf. Table \ref{fiber classification table for whitehead}). This stage relies on a comprehensive understanding of the boundary restriction map $\phi$, including its degree of map, finite(resp. flat and \'{e}tale) locus, singular fibers, etc. All of this information(except for possibly the intersection pattern in the fibers of the 4 singular points in $\Sigma$) is completely determined by the map $\phi$ and is independent of the slope coefficients. Therefore, the table can be computed and stored \textit{a priori}. Once the two stages are completed, the final step of the algorithm is simply tracking the table point by point, which can be easily done with the assistance of a computer.

\label{R_++ contained in all C_gamma}

Based on table \ref{fiber classification table for whitehead}, this point will contribute a common line component $R_{++}=\{(2,y,y):y\in \mathbb C\}$ to $C_\gamma \cap C_{\delta}$. Therefore, $R_{++}$ is contained in all $C_\gamma$ and $C_\gamma$ is a subvariety of dimension 1 in $X_0(M)$(where $M$ is the Whitehead link complement) for any slope $\gamma$.

<!-- theorem-6: body_md -->
\label{Algorithm for computing intersection pattern in general}
Let $M$ be a 2-bridge link in $\mathbb{S}^3$ and $X_0(M)$ a given canonical surface of the character variety of $M$. Fix a boundary component $T$ of $M$ and perform two Dehn surgeries with distinct slopes $\gamma=\frac{p}{q},~\delta=\frac{r}{s}$ on $T$ to obtain two generalized curves $C_\gamma, C_\delta \subset X_0(M)$. Let $\phi:X_0(M) \longrightarrow X(T)$ be the boundary restriction map. Then the intersection pattern $C_\gamma \cap C_\delta$ and the corresponding intersection multiplicity can be computed using the following steps. 
\begin{enumerate}
    \par\hangindent 40\p@ Compute the relevant properties of $\phi$. This includes finding the map degree, the finite(resp. flat, \'{e}tale) locus, and the singular fibers. A table similar to Table \ref{fiber classification table for whitehead} should be obtained. 
    \par\hangindent 40\p@ Compute $\mathcal{S}_{\gamma,\delta}=\{(M,L)\in (\mathbb C^{*})^2 \mid M^p L^q=M^r L^s=1\}$;
    \par\hangindent 40\p@ Take Weyl involution and compute $\mathcal B_{\gamma,\delta}=\epsilon(\mathcal{S}_{\gamma,\delta})\subset X(T)$;
    \par\hangindent 40\p@ For each $b\in \mathcal B_{\gamma,\delta}$, determine the fiber $\phi^{-1}(b)$ and the contribution of the fiber to $C_\gamma \cap C_\delta$, based on the fiber classification of $\phi$ obtained in step (1).
\end{enumerate}
<!-- end theorem-6 -->
<!-- remark-8: body_md -->
Once the degree $k$ of the boundary restriction map $\phi$ is computed for a given 2-bridge link $M$, any possible candidate in $U_{\mathrm{\acute et}}\subset X(T)$ will correspond to $k$ distinct intersection points in $X_0(M)$, each with multiplicity $1$, using similar arguments in lemma \ref{intersection multiplicity computation}. However, for candidates in the branch locus, we need to compute the fiber multiplicities of $\phi$ along these branch points to obtain the correct number of intersection points and multiplicities. This requires some additional effort compared to the Whitehead link case, where the degree of $\phi$ is 2 and hence the fiber multiplicity at a ramification point is easily seen to be 2.

<!-- lemma-12: body_md -->
\label{strongly detected boundary slope}
Let $M$ be an irreducible knot exterior such that $\pi_1(M)$ acts non-trivially on the tree defined by a discrete valuation $v$ of a field $K$ via a representation $\rho: \pi_1(M) \longrightarrow \operatorname{SL}(2,K)$. If $\operatorname{tr}(\rho(\delta)) \in O_v$ for one slope $\delta$ in $\partial M$ and $\operatorname{tr}(\rho(\gamma)) \notin O_v$ for another slope
$\gamma$ in $\partial M$,then $M$ contains an essential surface $S$ associated to the tree such that the boundary of $S$ is nonempty and is of the slope $\delta$,i.e. $\delta$ is a boundary slope.

<!-- lemma-13: body_md -->
\label{slope is -v(l)/v(m)}
Let $W$ be an irreducible knot exterior with a representation $\rho: \pi_1(W) \longrightarrow \operatorname{SL}(2,K)$, where $K$ is a field with a discrete valuation $v$. If $\operatorname{tr}(\rho(\delta)) \in O_v$ for one slope $\delta=\mu^{p}\lambda^{q}$ and $\operatorname{tr}(\rho(\gamma)) \notin O_v$ for another slope $\gamma$. Then $\frac{p}{q}=-\frac{v(L)}{v(M)}$, where $(M,L)$ is a compatible pair of eigenvalues of $\rho(\mu)$ and $\rho(\lambda)$.
<!-- end lemma-13 -->
<!-- proof-14: body_md -->
Since $\rho(\mu)=\operatorname{diag}(M,M^{-1}), ~\rho(\lambda)=\operatorname{diag}(L,L^{-1})$, we have $\rho(\delta)=\rho(\mu^p\lambda^q)=\operatorname{diag}(M^pL^q,M^{-p}L^{-q})$. Let $t=M^pL^q$, then $v(t)=pv(M)+qv(L)$ and $\operatorname{tr}(\rho(\delta))=t+t^{-1}$. If $v(t)\neq0$, then $v(t^{-1)}=-v(t)\neq v(t)$ and hence $v(\operatorname{tr}(\rho(\delta)))=v(t+t^{-1})=\operatorname{min}(v(t),v(t^{-1}))<0$. This contradicts the fact that $\operatorname{tr}(\rho(\delta)) \in O_v$. So we must have $v(t)=0$ and hence $pv(M)+qv(L)=0$. Since $v(\operatorname{tr}(\rho(\gamma)))<0$, $(v(M),v(L))\neq (0,0)$. Therefore, $\frac{p}{q}=-\frac{v(L)}{v(M)}$(we view $\infty = \frac{1}{0}$).
<!-- end proof-14 -->
<!-- remark-9: body_md -->
\label{v(g)<0 is enough}
If $\rho$ is an ANI representation such that $\operatorname{tr}(\rho(\gamma)) \notin O_v$ for some slope $\gamma \in \pi_1(T)$ where $T$ is the boundary torus, then it can be shown that $\operatorname{tr}(\rho(\delta)) \in O_v$ for some other slope $\delta$. Therefore, the hypotheses in the previous two lemmas are automatically satisfied, once $\operatorname{tr}(\rho(\gamma)) \notin O_v$ for some slope $\gamma$.

<!-- lemma-14: body_md -->
\label{traces on T_1 are algebraic integers}
Let $\chi \in C_\gamma \cap C_\delta$ and let $(m_1,\ell_1,n_1)=\phi(\chi)$, where $\phi: X_0(M) \longrightarrow X(T_1)$ is the boundary restriction map onto $T_1$. Then $m_1,\ell_1$ are both algebraic integers.
<!-- end lemma-14 -->
<!-- proof-15: body_md -->
Let $\rho$ be a representative of $\chi$. Write $P_{\mu_1}=\rho(\mu_1),~ P_{\lambda_1}=\rho(\lambda_1)$. These two matrices commute.  The two filling relations give $P_{\mu_1}^\Delta=(P_{\mu_1}^p P_{\lambda_{1}}^q)^s(P_{\mu_1}^r P_{\lambda_1}^s)^{-q}=I$ and $P_{\lambda_1}^\Delta=(P_{\mu_1}^r P_{\lambda_1}^s)^p(P_{\mu_1}^p P_{\lambda_1}^q)^{-r}=I.$ Thus the eigenvalues of $P_{\mu_1}$ and $P_{\lambda_1}$ are roots of unity. Their traces are therefore algebraic integers.

<!-- proposition-3: body_md -->
\label{criterion for ANI}
Let $M$ be the Whitehead link complement in $\mathbb{S}^3$. Let $P=(m_1,\ell_1,n_1) \in \mathcal{B}_{\gamma,\delta} \subset X(T_1)$ such that $P$ lies inside $U_{fin}$, the finite locus of $\phi: X_0(M) \longrightarrow X(T_1)$. Define 
\begin{equation}\label{compute T,H}
    T=\frac{m_1(m_1+n_1)}{2-\ell_1+m_1n_1}, 
\qquad  
H=\begin{cases}
    \frac{\ell_1+2-T^2}{2-T}, ~~\text{if} ~~T\neq 2  \\
    4, ~~\text{if} ~~T=2.
    \end{cases}
\end{equation}
Suppose that $\chi=(x,y,z)$ lies in the fiber $\phi^{-1}(P)$. Then $\chi$ is the character of an ANI representation if and only if $y$ is not an algebraic integer, and $y$ is not an algebraic integer if and only if $H$ is not an algebraic integer.
<!-- end proposition-3 -->
<!-- proof-16: body_md -->
First, from the defining equation $F=0$ of $X_0(M)$(cf. \eqref{defining equation for X_0(M) Whitehead}), $z$ satisfies a polynomial equation with coefficients in $\mathbb Z[x,y]$ and with leading coefficient $1$. Therefore, by the transitivity of integrality, $z$ is an algebraic integer if $x,y$ are. Since $x$ is an algebraic integer by lemma \ref{traces on T_1 are algebraic integers}, $\chi$ is the character of an ANI representation if and only if $y$ is not an algebraic integer. 

Since $P$ lies in the finite locus of $\phi$, the fiber of $P$ is nonempty and we choose $\chi=(x,y,z) \in \phi^{-1}(P)$. From the defining equation of $X_0(M)$, the definition of $\kappa$ and \eqref{trace formulas for m_i, l_i, n_i}, we have 
\begin{equation}\label{equation for a fiber}
    \begin{cases}
        &\kappa=m_1^2+y^2+z^2-m_1yz-2, \\
        & z\kappa=m_1y,\\
        & \ell_1=\kappa^2+(2-\kappa)y^2-2\\
        & n_1=m_1(\kappa-1)+(2-\kappa)yz
    \end{cases}
\end{equation}

From \eqref{equation for a fiber}, we have $2-\ell_1+m_1n_1=m_1^2+(2-\kappa)z^2$ and  $m_1(m_1+n_1)=m_1^2\kappa+(2-\kappa)m_1yz$. Then 
$$
\kappa(2-\ell_1+m_1n_1)=\kappa m_1^2+(2-\kappa)\kappa z^2=m_1^2\kappa+(2-\kappa)m_1yz=m_1(m_1+n_1).
$$
Since the point $P$ lies in the finite locus of $\phi$, $2-\ell_1+m_1n_1 \neq 0$ and hence $\kappa=\frac{m_1(m_1+n_1)}{2-\ell_1+m_1n_1}=T$. When $T\neq 2$, from the third equation of \eqref{equation for a fiber}, we have $y^2=\frac{\ell_1+2-T^2}{2-T}=H$. When $T=\kappa=2$, from the first two equations in \eqref{equation for a fiber}, we have $z=\frac{m_1y}{2},~ 4=m_1^2+y^2-\frac{m_1^2y^2}{4}$. Therefore, $(m_1^2-4)(y^2-4)=0$. Since $P$ lies in the finite locus, $m_1^2-4\neq 0$. So we must have $y^2=4=H$. Collecting these results gives $y^2=H$ and hence $y$ is an algebraic integer if and only if $H$ is.
<!-- end proof-16 -->

So $\chi$ is indeed the character of some $\rho$ where the target of $\rho$ lies in $\operatorname{SL}(2,K)$ for some number field $K$. Therefore, it makes sense to discuss whether $\chi$ is ANI or not for $\chi \in \phi^{-1}(P)$.

<!-- theorem-7: body_md -->
\label{boundary slopes are 0 or 4}
Let $M$ be the Whitehead link complement in $\mathbb{S}^3$. Let $\gamma=\frac{p}{q},~ \delta=\frac{r}{s},~ ps-qr\neq 0$ be two distinct slopes on $T_1$. Suppose that $\chi \in C_{\gamma}\cap C_{\delta} \subset X_0(M)$ is the character of an ANI representation $\rho:\pi_1(M)\longrightarrow \mathrm{SL}_2(K)$
for some number field $K$, i.e. $\rho(\mu_1)^p\rho(\lambda_1)^q=\rho(\mu_1)^r\rho(\lambda_1)^s=I$. Let $v$ be a discrete valuation at which $v(y)<0$(cf. Proposition \ref{criterion for ANI}). Then the common boundary slope detected by $\chi,v$ is either $0$ or $4$. More precisely, writing $\kappa=\operatorname{tr}\rho([a,b]),$ the detected slope is
$$
\begin{cases}
0,& v(\kappa) \geq 0,\\
4,& v(\kappa)<0.
\end{cases}
$$
<!-- end theorem-7 -->
<!-- proof-17: body_md -->
From lemma \ref{slope is -v(l)/v(m)}, the boundary slope detected is of the form $-\frac{v(L_2)}{v(M_2)}$. Therefore, we need to find $v(L_2)$ and $v(M_2)$ respectively. Let $h:=v(y)<0$. Since $v(y)<0$, the matrix $\rho(\mu_2)=\rho(b)$ has distinct eigenvalues. Then, we may simultaneously diagonalize $\rho(\mu_2)$ and $\rho(\lambda_2)$, since they commute:
$$
\rho(\mu_2)=
\begin{pmatrix}
M_2&0\\
0&M_2^{-1}
\end{pmatrix},
\qquad
\rho(\lambda_2)=
\begin{pmatrix}
L_2&0\\
0&L_2^{-1}
\end{pmatrix}.
$$
Thus
\begin{equation}\label{eq:wh-eigenvalue-traces}
m_2=M_2+M_2^{-1},\qquad
\ell_2=L_2+L_{2}^{-1},\qquad
n_2=M_2L_2+(M_2L_2)^{-1}.
\end{equation}
For any nonzero $\xi$, if $v(\xi)\neq 0$, then
\begin{equation}
\label{eq:wh-trace-valuation}
v(\xi+\xi^{-1})=-|v(\xi)|.
\end{equation}
Indeed, the two summands have distinct valuations, so the
valuation of their sum is their minimum. Hence, we choose $M_2$ so that $v(M_2)=v(m_2)=v(y)=h<0.$ 


Suppose that $v(\kappa)\geq 0$, including $\kappa=0$. Since $x$ is integral, the formula $\ell_2=\kappa^2-x^2\kappa+2x^2-2$ shows that $v(\ell_2)\geq 0.$ It follows from \eqref{eq:wh-eigenvalue-traces} and \eqref{eq:wh-trace-valuation} that $v(L_2)=0.$ Indeed, a nonzero value of $v(L_2)$ would force $v(\ell_2)=-|v(L_2)|<0$. Therefore, the detected slope is 0.

Suppose now that $t:=v(\kappa)<0.$ Since $v(2)\geq 0$, we have $v(2-\kappa)=t.$
Consider the equation $\ell_1+2=\kappa^2+(2-\kappa)y^2$ obtained from \eqref{trace formulas for m_i, l_i, n_i}. Its left-hand side is integral since $v(\ell_1)\geq 0$. On the right, $v(\kappa^2)=2t<0,~v((2-\kappa)y^2)=t+2h<0.$ These valuations must be equal. Otherwise the right-hand side would have a unique lowest-valuation term, and hence negative valuation, contradicting the integrality of the left-hand side.
Thus $2t=t+2h,$ and therefore $h=\frac{t}{2}.$

Next, from \eqref{trace formulas for m_i, l_i, n_i}, $\ell_2=\kappa^2-x^2\kappa+2x^2-2$. $v(\kappa^2)=2t$ while $v(x^2\kappa)\geq t>2t$. The remaining terms are integral. Hence $\kappa^2$ is the unique lowest-valuation term, giving $v(\ell_2)=2t.$ To determine the compatible sign of $v(L_2)$, we also need the mixed trace $n_2$. Since $\kappa\neq 0$, we have $z=\frac{xy}{\kappa}.$ Substituting this into \eqref{trace formulas for m_i, l_i, n_i}, we obtain
$$
n_2=y\kappa-xz\kappa-y+2xz=y\left(\kappa-x^2-1+\frac{2x^2}{\kappa}\right).
$$
Inside the parentheses, the term $\kappa$ has valuation $t<0$.
All other terms are integral; in particular, $v\left(\frac{2x^2}{\kappa}\right)=v(2)+2v(x)-t\geq -t>0.$ Therefore $v\left(\kappa-x^2-1+\frac{2x^2}{\kappa}\right)=t.$ Using $h=\frac{t}{2}$, we conclude that
\begin{equation}
\label{eq:wh-U-value}
v(n_2)=h+t=\frac{3t}{2}.
\end{equation}

Since $v(\ell_2)=2t$, the longitude eigenvalue satisfies $v(L_2)\in\{2t,-2t\}.$ We have chosen $v(M_2)=h=\frac{t}{2}.$ If $v(L_2)=2t$, then $v(M_2L_2)=\frac{5t}{2}<0.$ Equation \eqref{eq:wh-eigenvalue-traces} would imply $v(n_2)=\min\left\{\frac{5t}{2},-\frac{5t}{2}\right\}=\frac{5t}{2},$ contradicting \eqref{eq:wh-U-value}. Hence the compatible eigenvalue must satisfy $v(M_2)=\frac{t}{2},~v(L_2)=-2t.$ Then the detected slope is given by $-\frac{v(L_2)}{v(M_2)}=4$.

<!-- remark-11: body_md -->
Although there are only two possibilities for the detected boundary slopes for Dehn fillings of the Whitehead link, there is no restriction on the possible primes that lead to non-integral valuations. For instance, in example \ref{example 2 of section 4}, both the 2-adic and 3-adic valuations give non-integral valuations at some characters, but the detected boundary slopes for both of them are 4.

<!-- remark-12: body_md -->
The $\frac{1}{n}$-fillings on $T_1$ of the Whitehead link will give twist knots in $\mathbb{S}^3$. From \cite{HT1985}, the possible boundary slopes for twist knots can be computed using topological methods. Except for the unknot and the trefoil knot, every twist knot has three distinct boundary slopes: 0,4, and another one. This last boundary slope is distinct for different twist knots. Therefore, the common boundary slopes for twist knots will typically be 0 and 4, which is consistent with our findings(our result is more general in the sense that the filling slopes can be arbitrary). It is surprising that similar conclusions can be reached by two completely different methods(one topological and one arithmetic). This suggests that there is a deeper mathematical connection here that deserves further investigation.

<!-- proposition-4: body_md -->
\label{reducible only detect 0}
Let $M$ be the Whitehead link complement in $\mathbb{S}^3$. Suppose that $\chi \in C_{\gamma}\cap C_{\delta}$ is the character of a reducible and ANI representation. Then the detected slope is exactly $0$.
<!-- end proposition-4 -->
<!-- proof-18: body_md -->
Let $\chi$ be the character of $\rho: \pi_1(M) \longrightarrow \operatorname{SL}(2,K)$ for some number field $K$. Since $\rho$ is reducible, we can assume $\rho$ is upper-triangular, and hence diagonal(setting the upper-right corner to be 0 won't change the character). Therefore, we write 
$$
\rho(g)=
\begin{pmatrix}
\alpha(g) & 0\\
0 & \alpha(g)^{-1}
\end{pmatrix},
$$
for some group homomorphism $\alpha: \pi_1(M) \longrightarrow K^{\times}$. Since $K^{\times}$ is abelian, $\alpha$ factors through the abelianization of $\pi_1(M)$, which is $H_1(M;\mathbb Z) \cong \mathbb Z^2$. This means that there is a group homorphism $\overline{\alpha}: H_1(M;\mathbb Z) \longrightarrow K^{\times}$ such that $\alpha(g)=\overline{\alpha}([g])$ for any $g\in \pi_1(M)$, where $[g] \in H_1(M;\mathbb Z)$ represents the homology class of $g$. Since the linking number for the Whitehead link(between the two cusps) is 0, $[\lambda_1]=[\lambda_2]=0 \in H_1(M;\mathbb Z)$. Therefore, $\alpha(\lambda_2)=1$ and hence $L_2=1$. So $v(L_2)=0$ for any discrete valuation $v$. On the other hand, since $\chi \in C_{\gamma}\cap C_{\delta}$ is ANI, the negative valuation must occur at the character of $\mu_2$ from our previous discussions. Therefore, $v(\operatorname{tr}(\rho(\mu_2))=v(M_2+M_2^{-1})<0$ and hence $v(M_2)\neq 0$. Then the detected slope is $-\frac{v(L_2)}{v(M_2)}=0$.

<!-- lemma-15: body_md -->
\label{relation between chain complex of X_phi and tilde(X)}
Let $\sigma=1$ and regard $\Lambda$ as a right $\mathbb C G$-module by $a\cdot g=a\,t^{\phi(g)},~ a\in\Lambda,~ g\in G$. Then there is a natural isomorphism of chain complexes of left $\Lambda$-modules 
$$
\boxed{C_*(X_\phi;\mathbb C) \cong \Lambda\otimes_{\mathbb C G}C_*(\widetilde X;\mathbb C).}
$$
<!-- end lemma-15 -->
<!-- proof-19: body_md -->
Let $q:\widetilde X\longrightarrow X_\phi$ be the quotient map. If $c$ is a cellular chain in $\widetilde X$, write $\overline c=q_\#(c).$ With our convention for the generator $t$, we have $\overline{gc}=t^{\phi(g)}\overline c.$ Define $\Psi_n: \Lambda\otimes_{\mathbb C G}C_n(\widetilde X;\mathbb C) \longrightarrow C_n(X_\phi;\mathbb C)$ by $\Psi_n(a\otimes c)=a\overline c.$ The balancing relation in the tensor product is $(a\cdot g)\otimes c=a\otimes gc.$ Now,
$$
\Psi_n((a\cdot g)\otimes c) = \Psi_n(at^{\phi(g)}\otimes c) = at^{\phi(g)}\overline c,
$$
whereas $\Psi_n(a\otimes gc)=a\overline{gc}=at^{\phi(g)}\overline c.$ Thus $\Psi_n$ is well-defined. It also commutes with the boundary operators:
$$
\partial\Psi_n(a\otimes c) = \partial(a\overline c) = a\,\partial\overline c
= a\,\overline{\partial c} = \Psi_{n-1}(a\otimes\partial c).
$$
Moreover, the lifts of every cell of $X$ to $X_\phi$ form a free
orbit under $\mathbb Z=\langle t\rangle$. Therefore, $\Psi_n$ is an
isomorphism for every $n$, proving that $C_*(X_\phi;\mathbb C) \cong \Lambda\otimes_{\mathbb C G}C_*(\widetilde X;\mathbb C).$

<!-- lemma-16: body_md -->
\label{condition for rho to be group homo}
The map \(\rho\) is a group homomorphism if and only if $\alpha$ is a character and $u(gh)=u(g)+\alpha(g)^2u(h)$.
<!-- end lemma-16 -->
<!-- proof-20: body_md -->
With the substitution $u(g)=\alpha(g)b(g)$, we clearly have 
$$
\rho(gh)=
\begin{pmatrix}
\alpha(gh) & u(gh)\alpha(gh)^{-1}\\
0 & \alpha(gh)^{-1}
\end{pmatrix}.
$$

Then by direct computation, we get 
$$
\rho(g)\rho(h)
=
\begin{pmatrix}
\alpha(g)\alpha(h) &
\alpha(g)u(h)\alpha(h)^{-1}
+
u(g)\alpha(g)^{-1}\alpha(h)^{-1}\\
0 & \alpha(g)^{-1}\alpha(h)^{-1}
\end{pmatrix}.
$$
Equalize the two matrices entry by entry, it can be verified that $\rho$ is a group homomorphism if and only if $\alpha$ is a character and $u(gh)=\alpha(g)^2 u(h) + u(g)$.

<!-- theorem-9: body_md -->
\label{Cohomological criterion}
Let $\alpha:G\longrightarrow\mathbb C^\times$ be a character. Then there exists a reducible representation $\rho:G\longrightarrow SL(2,\mathbb C)$ with nonabelian image and diagonal characters $\alpha$ and $\alpha^{-1}$ if and only if $\alpha^2\neq 1$
and $H^1(G;\mathbb C_{\alpha^2})\neq 0.$
<!-- end theorem-9 -->
<!-- proof-21: body_md -->
Let $u\in Z^1(G;\mathbb C_{\alpha^2}),$ so that $u(gh)=u(g)+\alpha(g)^2u(h).$ Define
$$
\rho_u(g)=
\begin{pmatrix}
\alpha(g) & u(g)\alpha(g)^{-1}\\
0 & \alpha(g)^{-1}
\end{pmatrix}.
$$
By Lemma \ref{condition for rho to be group homo}, $\rho_u$ is a reducible $SL(2,\mathbb C)$-representation. After a suitable conjugation, we may assume that any reducible representation $\rho$ is of the form $\rho=\rho_u$ for some $u$. Suppose next that $u$ is a coboundary, so that $u(g)=c\bigl(1-\alpha(g)^2\bigr)$
for some $c\in\mathbb C$. Let
$$
P_c=
\begin{pmatrix}
1 & c\\
0 & 1
\end{pmatrix}.
$$
A direct calculation gives
$$
P_c
\begin{pmatrix}
\alpha(g) & 0\\
0 & \alpha(g)^{-1}
\end{pmatrix}
P_c^{-1}
=
\begin{pmatrix}
\alpha(g) &
c\bigl(1-\alpha(g)^2\bigr)\alpha(g)^{-1}\\
0 & \alpha(g)^{-1}
\end{pmatrix}.
$$
The right-hand side is $\rho_u(g)$. Hence $\rho_u$ is conjugate
to the diagonal representation
$$
\rho_\alpha(g)=
\begin{pmatrix}
\alpha(g) & 0\\
0 & \alpha(g)^{-1}
\end{pmatrix}.
$$
In particular, $[u]=0$ implies that $\rho_u(G)$ is abelian.

Conversely, suppose that $\alpha^2\neq 1$ and that $\rho_u(G)$ is abelian. For all $g,h\in G$, we have $\rho_u(g)\rho_u(h)=\rho_u(h)\rho_u(g).$ Directly compute both sides and Compare the upper-right entries, we have 
$$
u(g)+\alpha(g)^2u(h)
=
u(h)+\alpha(h)^2u(g).
$$
Hence $\bigl(1-\alpha(h)^2\bigr)u(g)=\bigl(1-\alpha(g)^2\bigr)u(h).$ Since $\alpha^2\neq 1$, there is $h_0\in G$ such that
$\alpha(h_0)^2\neq 1.$ Set $c=\frac{u(h_0)}{1-\alpha(h_0)^2}.$
Taking $h=h_0$ in the preceding identity gives $u(g)=c\bigl(1-\alpha(g)^2\bigr)$
for all $g\in G$. Thus $u$ is a coboundary, and therefore $[u]=0\in H^1(G;\mathbb C_{\alpha^2}).$
It follows that, when $\alpha^2\neq 1$, $\rho_u(G)$ is nonabelian if and only if $[u]\neq 0.$

Finally, suppose that $\alpha^2=1$. Then the cocycle condition becomes $u(gh)=u(g)+u(h),$ and
$$
\rho_u(g)
=
\alpha(g)
\begin{pmatrix}
1 & u(g)\\
0 & 1
\end{pmatrix}.
$$
Scalar matrices commute with all matrices, and the upper-triangular
unipotent matrices commute with one another. Hence $\rho_u(G)$ is
always abelian when $\alpha^2=1$. This proves the theorem.

<!-- lemma-17: body_md -->
\label{Computing group cohomology using universal cover}
Let $X$ be a path-connected CW complex with $G=\pi_1(X)$, and let $M$ be a left $\mathbb C G$-module. Then 
\begin{equation}\label{group cohomology using universal cover}
    H^1(G;M) \cong H^1\left(\operatorname{Hom}_{\mathbb C G}(C_*(\widetilde X;\mathbb C),M)\right).
\end{equation}
<!-- end lemma-17 -->
<!-- proof-22: body_md -->
Let $P_*=C_*(\widetilde X;\mathbb C)=C_*(\widetilde X;\mathbb Z) \otimes_{\mathbb Z} \mathbb C .$ Let $e^{n}_i$ be a $n$-cell of $X$ and let $\tilde{e}^{n}_{i}$ be a lift of $e^{n}_i$ in $\widetilde{X}$. Since the action of $G$ on the lifted cells is free, every $P_n$ is a free $\mathbb C G$-module with basis $\{ \tilde{e}^{n}_{i}\}$. Hence there is an augmented chain complex
\begin{equation}\label{augmented chain complex}
    \cdots \longrightarrow P_2 \xrightarrow{\partial_2} P_1 \xrightarrow{\partial_1} P_0 \xrightarrow{\varepsilon} \mathbb C \longrightarrow0,
\end{equation}
where $\mathbb C$ is regarded as the trivial $G$-module, and $\epsilon$ is the augmentation map given by $\sum_{\tilde{v}} a_{\tilde{v}}\tilde{v} \mapsto \sum_{\tilde{v}} a_{\tilde{v}}$. 

For degree-one cohomology, since $\widetilde X$ is connected and simply connected, $H_0(\widetilde X;\mathbb C)\cong\mathbb C,~ H_1(\widetilde X;\mathbb C)=0.$ Hence
$$
P_2\longrightarrow P_1\longrightarrow P_0
\longrightarrow\mathbb C\longrightarrow0
$$
is exact at $P_1$, $P_0$, and $\mathbb C$. It is therefore a free
partial resolution sufficient for computing first group
cohomology, where the free resolution definition of group cohomology as presented in section \ref{group cohomology} is used. Consequently, $H^1(G;M) \cong H^1\left(\operatorname{Hom}_{\mathbb C G}(C_*(\widetilde X;\mathbb C),M)\right).$
<!-- end proof-22 -->
<!-- remark-15: body_md -->
If $X$ is a $K(G,1)$, then $\widetilde X$ is contractible. Consequently, the augmented chain complex \eqref{augmented chain complex} is a free $\mathbb C G$-resolution of the trivial $\mathbb C G$-module $\mathbb C$. Therefore, for every left $\mathbb C G$-module $M$, $H^n(G;M) \cong H^n \left(\operatorname{Hom}_{\mathbb C G}(P_*,M)\right)$ for all $n \geq 1$. But since we only need computation for $H^1$, it is not necessary to require $X$ to be a $K(G,1)$.

<!-- lemma-18: body_md -->
\label{Computing group cohomology using infinite cyclic cover}
Let $X$ be a compact connected orientable three-manifold with torus boundary with $G=\pi_1(X)$, and let $X_\phi$ be the infinite cyclic cover of $X$. Then we have
\begin{equation}\label{group cohomology using infinite cyclic cover}
H^1(G;\mathbb C_{\sigma,\lambda})
\cong
H^1\left(\operatorname{Hom}_{\Lambda}(C_*^{\sigma}(X_\phi;\mathbb C),\mathbb C_\lambda)\right).
\end{equation}
<!-- end lemma-18 -->
<!-- proof-23: body_md -->
By definition, $C_{*}^{\sigma}(X_\phi;\mathbb C)=\Lambda_{\sigma}\otimes_{\mathbb C G} C_{*}(\widetilde{X};\mathbb C)$, which is naturally a chain complex of left $\Lambda$-modules through self-multiplication on $\Lambda_{\sigma}$.

For a left $\mathbb C G$-module $P$ and a left $\Lambda$-module $M$,
there is a natural isomorphism called Tensor-Hom adjunction given by 
$$
\operatorname{Hom}_\Lambda\left(\Lambda_{\sigma}\otimes_{\mathbb C G}P,M\right)
\cong
\operatorname{Hom}_{\mathbb C G}(P,M),
$$
where $M$ is regarded as a $\mathbb C G$-module via the composition of $\Phi_{\sigma}:\mathbb C G\longrightarrow\Lambda$ and the action of $\Lambda$. 

Applying this adjunction degree by degree to $P_*=C_*(\widetilde X;\mathbb C),~M=\mathbb C_\lambda,$ gives an isomorphism of cochain complexes
$$
\operatorname{Hom}_\Lambda\left(\Lambda_{\sigma}\otimes_{\mathbb C G}C_*(\widetilde X;\mathbb C),\mathbb C_\lambda\right)
\cong
\operatorname{Hom}_{\mathbb C G}\left(C_*(\widetilde X;\mathbb C),\mathbb C_{\sigma,\lambda}\right).
$$
which gives 
$$
\operatorname{Hom}_\Lambda(C_{*}^{\sigma}(X_\phi;\mathbb C),\mathbb C_\lambda)
\cong \operatorname{Hom}_{\mathbb C G}(C_*(\widetilde X;\mathbb C),\mathbb C_{\sigma,\lambda}).
$$
Then we apply lemma \ref{Computing group cohomology using universal cover} by setting $M=\mathbb C_{\sigma,\lambda}$ in \eqref{group cohomology using universal cover} to give 
$$
\begin{aligned}
H^1(G;\mathbb C_{\sigma,\lambda})
&\cong H^1\left(\operatorname{Hom}_{\mathbb C G}(C_*(\widetilde X;\mathbb C),\mathbb C_{\sigma,\lambda})\right)\\
&\cong H^1\left(\operatorname{Hom}_\Lambda(C_{*}^{\sigma}(X_\phi;\mathbb C),\mathbb C_\lambda)\right).
\end{aligned}
$$
This completes the proof.

<!-- theorem-10: body_md -->
\label{equivalence of H^1 and twisted Alexander}
Let $X$ be a compact connected orientable three-manifold with torus boundary with $G=\pi_1(X)$ and $H_{1}(X) \cong \mathbb Z \oplus T$ where $T$ is the finite torsion part. Let $\chi=(\sigma,\lambda)$ be the decomposition of a character $\chi: G \to \mathbb C^{\times}$, and let $\Delta_{X}^{\sigma}(t)$ be the associated twisted Alexander polynomial. Suppose $\chi$ is nontrivial,i.e. $(\sigma,\lambda) \neq (1,1)$, then we have
$$
H^1(G;\mathbb C_{\sigma,\lambda})\neq 0\quad\Longleftrightarrow\quad
\Delta_{X}^\sigma(\lambda)=0.
$$
<!-- end theorem-10 -->
<!-- proof-24: body_md -->
Set $Q_*=C_*^\sigma(X_\phi;\mathbb C).$ Since $Q_*$ is a chain complex of free $\Lambda$-modules and $\Lambda=\mathbb C[t^{\pm 1}]$ is a PID, the universal coefficient theorem gives a natural short exact sequence
$$
0
\longrightarrow
\operatorname{Ext}^1_\Lambda\left(H_0(Q_*),\mathbb C_\lambda\right)
\longrightarrow
H^1\left(\operatorname{Hom}_\Lambda(Q_*,\mathbb C_\lambda)\right)
\longrightarrow
\operatorname{Hom}_\Lambda\left(H_1(Q_*),\mathbb C_\lambda\right)
\longrightarrow 
0.
$$
Using $H_1(Q_*)=A_X^\sigma$ and lemma \ref{Computing group cohomology using infinite cyclic cover}, this becomes
$$
0
\longrightarrow
\operatorname{Ext}^1_\Lambda\left(H_0\bigl(C_*^\sigma(X_\phi;\mathbb C)\bigr),
\mathbb C_\lambda\right)
\longrightarrow 
H^1(G;\mathbb C_{\sigma,_\lambda})
\longrightarrow
\operatorname{Hom}_\Lambda\left(A_X^\sigma,\mathbb C_\lambda\right)
\longrightarrow 
0.
$$
Therefore, whenever $\operatorname{Ext}^1_\Lambda\left(H_0\bigl(C_*^\sigma(X_\phi;\mathbb C)\bigr),
\mathbb C_\lambda\right)=0,$ there is a natural isomorphism
$$
H^1(G;\mathbb C_{\sigma,_\lambda})
\cong
\operatorname{Hom}_\Lambda\left(A_X^\sigma,\mathbb C_\lambda\right).
$$

We now prove that when $(\sigma,\lambda) \neq (1,1)$, the Ext term indeed vanishes. Suppose that $G$ is generated by $x_1,\ldots,x_m.$ The degree-one boundary map is $\partial_1:\Lambda^m\longrightarrow\Lambda,$ where $\partial_1(e_j)=\sigma(\tau(x_j))t^{\phi(x_j)}-1.$ Consequently,
$$
H_0(C_*^\sigma)
\cong
\frac{\Lambda}{\left(\sigma(\tau(g))t^{\phi(g)}-1\mid g\in G\right)}.
$$


Suppose that $\sigma$ is nontrivial. Choose $u\in T$ such that $\sigma(u)\neq1.$
Since the projection of $G$ onto $H_1(X;\mathbb Z)\cong\mathbb Z\oplus T$ is surjective, there exists $g\in G$ whose homology class is $[g]=(0,u).$ Thus $\phi(g)=0,~\tau(g)=u.$ The corresponding relation in $H_0(C_*^\sigma)$ is $\sigma(\tau(g))t^{\phi(g)}-1=\sigma(u)-1.$
Since $\sigma(u)-1$ is a nonzero complex number, it is a unit in
$\Lambda$. Therefore, $H_0(C_*^\sigma)=0.$ It follows immediately that $\operatorname{Ext}^1_\Lambda\left(H_0(C_*^\sigma),\mathbb C_\lambda\right)=0$ for every $\lambda\in\mathbb C^\times$.

Suppose now that $\sigma$ is trivial. Then

$$
H_0(C_*^1)\cong\frac{\Lambda}{\left(t^{\phi(g)}-1\mid g\in G\right)}.
$$
Every polynomial $t^{\phi(g)}-1$ is divisible by $t-1$. Since $\phi$ is surjective, there exists $g_0\in G$ such that $\phi(g_0)=1.$ For this element,
$t^{\phi(g_0)}-1=t-1.$ Therefore, $H_0(C_*^1)\cong\Lambda/(t-1).$ To calculate the relevant $\operatorname{Ext}$ term, consider the free resolution of $\Lambda/(t-1)$
$$
0\longrightarrow\Lambda\xrightarrow{\,t-1\,}\Lambda\longrightarrow\Lambda/(t-1)\longrightarrow0.
$$
Since Hom functor is left exact, applying $\operatorname{Hom}_\Lambda(-,\mathbb C_\lambda)$ to the short exact sequence above gives
$$
0\longrightarrow\operatorname{Hom}_\Lambda(\Lambda/(t-1),\mathbb C_\lambda)\longrightarrow\mathbb C_\lambda
\xrightarrow{\,\lambda-1\,}\mathbb C_\lambda.
$$
On $\mathbb C_\lambda$, multiplication by $t-1$ is just multiplication by
$\lambda-1$, which is invertible when $\lambda \neq 1$. Hence $\operatorname{Ext}^1_\Lambda(\Lambda/(t-1),\mathbb C_\lambda)\cong\frac{\mathbb C_\lambda}{(\lambda-1)\mathbb C_\lambda}=0$ when $\lambda \neq 1$. 

Now by the definition of twisted Alexander polynomial as presented in section \ref{twisted Alexander module and polynomials}, we write
$$
A_X^\sigma
\cong
\bigoplus_{j=1}^s \Lambda/(f_j(t)),
$$
and hence $\Delta_{X}^\sigma(t) \doteq \prod_{j=1}^s f_j(t).$ For each summand, one has $\operatorname{Hom}_\Lambda\left(\Lambda/(f_j(t)),\mathbb C_\lambda\right) \neq 0$ if and only if $f_j(t)$ acts trivially on $\mathbb C_\lambda$. Since
$t$ acts as $\lambda$, this is equivalent to $f_j(\lambda)=0.$ Therefore,
$\operatorname{Hom}_\Lambda\left(A_X^\sigma,\mathbb C_\lambda\right)\neq 0$
if and only if at least one $f_j(\lambda)$ vanishes. Equivalently,
$$
\operatorname{Hom}_\Lambda\left(A_X^\sigma,\mathbb C_\lambda\right)
\neq 0
\quad\Longleftrightarrow\quad\Delta_{X}^\sigma(\lambda)=0.
$$
Since the Ext term vanishes, we therefore obtain
$$
H^1(G;\mathbb C_{\sigma,\lambda})\neq 0\quad\Longleftrightarrow\quad
\Delta_{X}^\sigma(\lambda)=0.
$$

<!-- theorem-11: body_md -->
\label{twisted Burde de Rham theorem}
Let $X$ be a compact connected three-manifold with torus boundary
and fix a projection of $G=\pi_1(X)$ onto $H_1(X;\mathbb Z)\cong T\oplus \mathbb Z $ given by $(\tau,\phi)$. For any character $\alpha:G \longrightarrow \mathbb C^{\times}$, we write $\alpha=(\sigma,\lambda)$ so that $\alpha(g)=\sigma(\tau(g))\lambda^{\phi(g)}$ for any $g \in G$. Assume $\alpha^2\neq1.$ Then the following are equivalent:

(1). There exists a nonabelian reducible representation $\rho:G\longrightarrow \operatorname{SL}(2,\mathbb C)$ whose diagonal characters are $\alpha, \alpha^{-1}.$

(2). $\Delta^{\sigma^2}_{X}(\lambda^2)=0.$

<!-- theorem-12: body_md -->
\label{Burde de Rham theorem for 2-bridge link}
Let $L$ be a 2-bridge link in $\mathbb{S}^3$ and let $M=\mathbb{S}^3-N(L)$ be the complement of some neighborhood of $L$ in $\mathbb{S}^3$. Fix a projection of $G=\pi_1(M)$ onto $H_1(M;\mathbb Z)\cong \mathbb Z\oplus\mathbb Z $ given by $(\phi_1,\phi_2)$. For any character $\alpha:G \longrightarrow \mathbb C^{\times}$, we write $\alpha=(\lambda_1,\lambda_2)$ so that $\alpha(g)=\lambda_1^{\phi_1(g)}\lambda_2^{\phi_2(g)}$ for any $g \in G$. Assume $\alpha^2\neq1.$ Then the following are equivalent:

(1). There exists a nonabelian reducible representation $\rho:G\longrightarrow \operatorname{SL}(2,\mathbb C)$ whose diagonal characters are $\alpha, \alpha^{-1}.$

(2). $\Delta_{M}(\lambda_1^2,\lambda_2^2)=0.$
<!-- end theorem-12 -->
<!-- proof-26: body_md -->
Theorem \ref{Cohomological criterion} and lemma \ref{Computing group cohomology using universal cover} can be applied directly to the situation where $M$ is a 2-bridge link complement. For lemma \ref{Computing group cohomology using infinite cyclic cover}, set $\Lambda=\mathbb C[t_1^{\pm 1},t_2^{\pm 1}]$ and $C_*(M_\phi;\mathbb C)=\Lambda\otimes_{\mathbb CG} C_*(\widetilde M;\mathbb C)$. Then applying the same arguments in lemma \ref{Computing group cohomology using infinite cyclic cover}, we obtain the similar result for $M$: $H^1(G;\mathbb C_{\lambda_1,\lambda_2})
\cong H^1\left(\operatorname{Hom}_{\Lambda}(C_*(M_\phi;\mathbb C),\mathbb C_{\lambda_1,\lambda_2})\right)$.   For Theorem \ref{equivalence of H^1 and twisted Alexander}, we can apply similar arguments with the following modifications. When $M$ is a 2-bridge link complement, $\Lambda=\mathbb C[t_1^{\pm 1},t_2^{\pm 1}]$ is no longer a PID and we can not directly apply the universal coefficient theorem. Instead, we apply the universal coefficient spectral sequence to obtain the exact sequence 

$$
\begin{aligned}
0\longrightarrow&
\operatorname{Ext}_\Lambda^1(H_0(Q_*),\mathbb C_{\lambda_1,\lambda_2})
\longrightarrow H^1(M;\mathbb C_{\lambda_1,\lambda_2})\\
\longrightarrow&
\operatorname{Hom}_\Lambda(A_M,\mathbb C_{\lambda_1,\lambda_2})
\longrightarrow
\operatorname{Ext}_\Lambda^2(H_0(Q_*),\mathbb C_{\lambda_1,\lambda_2})
\longrightarrow H^2(M;\mathbb C_{\lambda_1,\lambda_2}).
\end{aligned}
$$
where $Q_{*}=C_*(M_\phi;\mathbb C)$. By similar computations in the proof of Theorem \ref{equivalence of H^1 and twisted Alexander}, we can show that the two Ext terms in the sequence vanish when $(\lambda_1,\lambda_2)\neq (1,1)$ and hence  $H^1(M;\mathbb C_{\lambda_1,\lambda_2}) \cong \operatorname{Hom}_\Lambda(A_M,\mathbb C_{\lambda_1,\lambda_2})$. By remark \ref{Alexander poly for 2-bridge link}, $A_M \cong \Lambda/(\Delta_M)$ when $M$ is a 2-bridge link in $\mathbb{S}^3$. Therefore, $\operatorname{Hom}_\Lambda(A_M,\mathbb C_{\lambda_1,\lambda_2}) \neq 0$ if and only if $\Delta_{M}(\lambda_1,\lambda_2)=0$. Combing this result with Theorem \ref{Cohomological criterion} yields the desired result.

\label{reducible intersection and common root}
Let $M$ be a two bridge link complement in $\mathbb{S}^3$ such that the linking number of the two boundary components is 0. Let $\gamma=\frac{p}{q}\neq 0,\delta=\frac{r}{s}\neq0$ and let $C_1\subset C_{\gamma}, C_2 \subset C_{\delta}$ be two irreducible curves, each containing a character of an irreducible representation. If $\chi$ is a reducible character and $\chi \in C_1\cap C_2\setminus \phi^{-1}(\Sigma)$, then $\chi$ detects a common root $\lambda^2$ for $\Delta_{M_\gamma}^{\sigma^2_{\gamma}}$ and $\Delta_{M_\delta}^{\sigma^2_{\delta}}$, where $\sigma_{\gamma}(\overline{1})=\sigma_{\delta}(\overline{1})=\xi$ and $\xi$ is some $d$-th root of unity for $d=\operatorname{gcd}(|p|,|r|)$.
<!-- end proposition-5 -->
<!-- proof-27: body_md -->
Since the linking number is 0, from the discussion above, we can fix the isomorphism $H_1(M_\gamma,\mathbb Z) \cong \mathbb Z/|p|\mathbb Z \oplus \mathbb Z$ with generators $[\overline{\mu_1}], [\overline{\mu_2}]$. Then $\alpha_{\gamma}=(\sigma_\gamma,\lambda_{\gamma})$ where $\lambda_\gamma=\alpha_{\gamma}(\overline{\mu_2})$ and $\sigma_\gamma(\overline{1})=\alpha_{\gamma}(\overline{\mu_1})$. Similar results hold for $\alpha_{\delta}=(\sigma_\delta,\lambda_{\delta})$. Since $\alpha=\alpha_{\gamma}\circ \iota_{\gamma}=\alpha_{\delta}\circ \iota_{\delta}$, we have $\alpha(\mu_2)=\alpha_{\gamma}\circ \iota_{\gamma}(\mu_2)=\alpha_{\delta}\circ \iota_{\delta}(\mu_2)$, i.e. $\lambda_\gamma=\lambda_\delta=\lambda$, and $\alpha(\mu_1)=\alpha_{\gamma}\circ \iota_{\gamma}(\mu_1)=\alpha_{\delta}\circ \iota_{\delta}(\mu_1)$, i.e. $\sigma_\gamma(\overline{1})=\sigma_\delta(\overline{1})=\xi$. Then consider $\rho_\gamma,\rho_\delta$. From the discussion above and Theorem \ref{twisted Burde de Rham theorem}, we have $\Delta_{M_\gamma}^{\sigma^2_{\gamma}}(\lambda^2)=\Delta_{M_\delta}^{\sigma^2_{\delta}}(\lambda^2)=0$ and $\sigma_{\gamma}(\overline{1})=\sigma_{\delta}(\overline{1})=\xi$. $\xi$ is some $d$-th root of unity for $d=\operatorname{gcd}(|p|,|r|)$ because the domain for $\sigma_\gamma, \sigma_\delta$ are $\mathbb Z/|p|\mathbb Z, \mathbb Z/|r|\mathbb Z$ respectively.

<!-- corollary-2: body_md -->
\label{never intersect at reducible for whitehead}
Let $M$ be the Whitehead link complement in $\mathbb{S}^3$ and let $\gamma=\frac{1}{n}, \delta=\frac{1}{m},~n\neq m$ such that $M_\gamma=K_n, M_{\delta}=K_m$ are both hyperbolic. Let $C_1\subset C_{\gamma}, C_2 \subset C_{\delta}$ be the canonical curves of $X(K_n)$ and $X(K_m)$ respectively, when embedded in $X_0(M)$. Then $C_1$ and $C_2$ can never intersect at a reducible character $\chi \notin \phi^{-1}(\Sigma)$.
<!-- end corollary-2 -->
<!-- proof-28: body_md -->
The boundary components of the Whitehead link have linking number 0. Suppose $C_1,C_2$ intersect at a reducible character $\chi \notin \phi^{-1}(\Sigma)$, then from Proposition \ref{reducible intersection and common root}, $\chi$ will detect common roots of the ordinary Alexander polynomials for $K_n,K_m$. But the ordinary Alexander polynomial for $K_n$ is given by $\Delta_{K_n}(t)\doteq -nt^2+(2n+1)t-n$. It is easy to check that $\Delta_{K_n}(t), \Delta_{K_m}(t)$ have no common root, which leads to the contradiction.
