<!-- theorem-1: body_md -->
\label{thm:fitted_replay_value_bound}
Assume $\mathcal A$ finite, $|r(s,a)|\le R_{\max}$, the policy path stability of Assumption~\ref{ass:learned_policy_path_stability_clean} (constant $C_\pi$) and the replay error bounds of Assumption~\ref{ass:replay_error_decomp} (Appendix~\ref{app:theory_setup}), and $C_{\mathrm{cov}}(\pi^\star)<\infty$. Let $\zeta\in(0,1)$ and $C_{\mathrm{loc}}=\frac{(1+\gamma)R_{\max}C_\pi}{(1-\gamma)^2(1-2^{\zeta-1})}$. Then for every stationary comparator $\pi^\star$ satisfying the above assumptions,
$$
\begin{aligned}
\mathcal G_T^{\mathrm{reg}}(\pi^\star)
\le \frac{1}{1-\gamma}\Bigg[\,
&\frac{\beta\ln|\mathcal A|(1+\ln T)}{T-1}
+\frac{2C_{\mathrm{loc}}}{\zeta}\frac{T^\zeta}{T-1}\\[-0.4ex]
&+2\sqrt{C_{\mathrm{cov}}(\pi^\star)}\Big(\bar\epsilon_{\mathrm{fit}}+\epsilon_0+\tfrac{C_{\mathrm{var}}}{1-\zeta/2}\tfrac{T^{1-\zeta/2}}{T-1}\Big)
\Bigg]
=:\mathcal B_T(\pi^\star).
\end{aligned}
$$

\label{cor:unregularized_rate}

<!-- lemma-2: body_md -->
\label{lem:value_advantage_lipschitz_clean}
For any two stationary policies $\pi$ and $\pi'$,
$$
    \left\|V^\pi-V^{\pi'}\right\|_\infty
    \le
    \frac{R_{\max}}{(1-\gamma)^2}
    \sup_s
    \left\|
        \pi(\cdot\mid s)-\pi'(\cdot\mid s)
    \right\|_1.
$$
Moreover,
$$
    \left\|A^\pi-A^{\pi'}\right\|_\infty
    \le
    \frac{(1+\gamma)R_{\max}}{(1-\gamma)^2}
    \sup_s
    \left\|
        \pi(\cdot\mid s)-\pi'(\cdot\mid s)
    \right\|_1.
$$
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
For any state $s$,
$$
\begin{aligned}
V^\pi(s)-V^{\pi'}(s)
&=
\left\langle
    \pi(\cdot\mid s)-\pi'(\cdot\mid s),
    Q^{\pi'}(s,\cdot)
\right\rangle \\
&\qquad+
\mathbb E_{a\sim\pi(\cdot\mid s)}
\left[
    Q^\pi(s,a)-Q^{\pi'}(s,a)
\right].
\end{aligned}
$$
By the Bellman relation,
$$
    Q^\pi(s,a)-Q^{\pi'}(s,a)
    =
    \gamma
    \mathbb E_{s'\sim P(\cdot\mid s,a)}
    \left[
        V^\pi(s')-V^{\pi'}(s')
    \right].
$$
Therefore, by H\"older's inequality,
$$
\begin{aligned}
\left|V^\pi(s)-V^{\pi'}(s)\right|
&\le
\left\|\pi(\cdot\mid s)-\pi'(\cdot\mid s)\right\|_1
\left\|Q^{\pi'}\right\|_\infty
+
\gamma
\left\|V^\pi-V^{\pi'}\right\|_\infty.
\end{aligned}
$$
Taking the supremum over $s$ and using $\left\|Q^{\pi'}\right\|_\infty\le R_{\max}/(1-\gamma)$ gives
$$
    \left\|V^\pi-V^{\pi'}\right\|_\infty
    \le
    \frac{R_{\max}}{1-\gamma}
    \sup_s
    \left\|
        \pi(\cdot\mid s)-\pi'(\cdot\mid s)
    \right\|_1
    +
    \gamma
    \left\|V^\pi-V^{\pi'}\right\|_\infty.
$$
Rearranging proves the value bound. For the advantage bound,
$$
    \left\|Q^\pi-Q^{\pi'}\right\|_\infty
    \le
    \gamma
    \left\|V^\pi-V^{\pi'}\right\|_\infty,
$$
and since $A^\pi=Q^\pi-V^\pi$,
$$
    \left\|A^\pi-A^{\pi'}\right\|_\infty
    \le
    \left\|Q^\pi-Q^{\pi'}\right\|_\infty
    +
    \left\|V^\pi-V^{\pi'}\right\|_\infty
    \le
    (1+\gamma)
    \left\|V^\pi-V^{\pi'}\right\|_\infty.
$$
Substituting the value bound completes the proof.

Applying Assumption~\ref{ass:learned_policy_path_stability_clean} to the right hand side gives the claim with $L_A = (1+\gamma) R_{\max}C_\pi/(1-\gamma)^2$.

<!-- lemma-3: body_md -->
\label{lem:recent_window_tracks_current_advantage_clean}
Under Corollary~\ref{cor:local_advantage_drift_clean}, for every $t\ge2$,
$$
    \left\|A_t-Z_t^A\right\|_\infty
    \le
    C_{\mathrm{loc}}t^{\zeta-1},
$$
where one may take
$$
    C_{\mathrm{loc}}
    =
    \frac{L_A}{1-2^{\zeta-1}}.
$$
Consequently, for every $T\ge2$,
$$
    \frac{2}{T-1}\sum_{t=2}^T
    \left\|A_t-Z_t^A\right\|_\infty
    \le
    \frac{2C_{\mathrm{loc}}}{\zeta}
    \frac{T^\zeta}{T-1}.
$$
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
For $k\in\mathcal W_t$, Corollary~\ref{cor:local_advantage_drift_clean} gives
$$
\begin{aligned}
\left\|A_t-A_k\right\|_\infty
&\le
\sum_{j=k}^{t-1}
\left\|A_{j+1}-A_j\right\|_\infty \\
&\le
L_A\sum_{j=k}^{t-1}\frac1j.
\end{aligned}
$$

Since $k\ge t-W_t$ and $W_t\le t^\zeta$, we have $k\ge t-t^\zeta$. Also, since $t^\zeta/t=t^{\zeta-1}\le2^{\zeta-1}$ for every $t\ge2$,
$$
    t-W_t
    \ge
    t-t^\zeta
    \ge
    \left(1-2^{\zeta-1}\right)t.
$$
Therefore,
$$
    \sum_{j=k}^{t-1}\frac1j
    \le
    \frac{t-k}{t-W_t}
    \le
    \frac{W_t}{(1-2^{\zeta-1})t}
    \le
    \frac{t^{\zeta-1}}{1-2^{\zeta-1}}.
$$
Thus, uniformly over $k\in\mathcal W_t$,
$$
    \left\|A_t-A_k\right\|_\infty
    \le
    C_{\mathrm{loc}}t^{\zeta-1}.
$$
For each fixed $(s,a)$, $Z_t^A(s,a)$ is a convex combination of the values $\{A_k(s,a):k\in\mathcal W_t\}$ (coefficients proportional to $\mu_k(s,a)$ on the support of $\mu_t$, and uniform off it). Hence
$$
\begin{aligned}
\left|A_t(s,a)-Z_t^A(s,a)\right|
&\le
\max_{k\in\mathcal W_t}
\left|A_t(s,a)-A_k(s,a)\right| \\
&\le
C_{\mathrm{loc}}t^{\zeta-1}.
\end{aligned}
$$
Taking the supremum over $(s,a)$ proves the pointwise bound.

<!-- lemma-4: body_md -->
\label{lem:direct_regularized_learned_readout_clean}
For every comparator policy $\pi^\star$, every state $s$, and every $t\ge2$,
$$
\begin{aligned}
&\left\langle
    \pi^\star(\cdot\mid s)-\pi_{\theta_t}(\cdot\mid s),
    A_t(s,\cdot)
\right\rangle
+
\lambda
\left(
    \varphi(\pi_{\theta_t}(\cdot\mid s))
    -
    \varphi(\pi^\star(\cdot\mid s))
\right) \\
&\qquad\le
\frac{\beta}{t-1}\ln|\mathcal A|
+
2\left\|A_t(s,\cdot)-Z_t^A(s,\cdot)\right\|_\infty
+

\end{aligned}
$$

<!-- proof-4: body_md -->
Fix $s$ and $t\ge2$. By definition of the readout, with $\alpha_{t-1}=\lambda+\beta/(t-1)$,
$$
    \pi_{\theta_t}(\cdot\mid s)
    \in
    \arg\max_{p\in\Delta(\mathcal A)}
    \left\{
        \left\langle p,Z_{\theta_{t-1}}(s,\cdot)\right\rangle
        -
        \alpha_{t-1}\varphi(p)
    \right\}.
$$
Since this objective is concave, evaluating it at $\pi_{\theta_t}(\cdot\mid s)$ and at $\pi^\star(\cdot\mid s)$ and using maximality gives
$$
    \left\langle
        \pi^\star(\cdot\mid s)-\pi_{\theta_t}(\cdot\mid s),
        Z_{\theta_{t-1}}(s,\cdot)
    \right\rangle
    +
    \alpha_{t-1}
    \left(
        \varphi(\pi_{\theta_t}(\cdot\mid s))
        -
        \varphi(\pi^\star(\cdot\mid s))
    \right)
    \le 0.
$$
Writing $\alpha_{t-1}=\lambda+\beta/(t-1)$ and carrying the score from $Z_{\theta_{t-1}}$ to $A_t$ through the regression Bayes target $Z_t^A$ (all quantities evaluated at $s$),
$$
\begin{aligned}
&\left\langle
    \pi^\star-\pi_{\theta_t},A_t
\right\rangle
+
\lambda
\left(
    \varphi(\pi_{\theta_t})-\varphi(\pi^\star)
\right) \\
&=
\underbrace{
\left\langle
    \pi^\star-\pi_{\theta_t},Z_{\theta_{t-1}}
\right\rangle
+
\alpha_{t-1}
\left(
    \varphi(\pi_{\theta_t})-\varphi(\pi^\star)
\right)
}_{\le 0}
-
\frac{\beta}{t-1}
\left(
    \varphi(\pi_{\theta_t})-\varphi(\pi^\star)
\right) \\
&\qquad
+
\left\langle
    \pi^\star-\pi_{\theta_t},A_t-Z_t^A
\right\rangle
+
\left\langle
    \pi^\star-\pi_{\theta_t},Z_t^A-Z_{\theta_{t-1}}
\right\rangle.
\end{aligned}
$$
We bound the three remaining terms. Since $\varphi(p)=\sum_a p(a)\ln p(a)\in[-\ln|\mathcal A|,0]$,
$$
    -\frac{\beta}{t-1}
    \left(
        \varphi(\pi_{\theta_t})-\varphi(\pi^\star)
    \right)
    \le
    \frac{\beta}{t-1}\ln|\mathcal A|.
$$
By H\"older's inequality with $\|\pi^\star-\pi_{\theta_t}\|_1\le2$,
$$
    \left\langle
        \pi^\star-\pi_{\theta_t},A_t-Z_t^A
    \right\rangle
    \le
    2\left\|A_t(s,\cdot)-Z_t^A(s,\cdot)\right\|_\infty .
$$
By the Cauchy--Schwarz inequality with $\|\pi^\star-\pi_{\theta_t}\|_2\le\|\pi^\star\|_2+\|\pi_{\theta_t}\|_2\le2$ (each distribution has $\ell_2$ norm at most $1$),
$$
    \left\langle
        \pi^\star-\pi_{\theta_t},Z_t^A-Z_{\theta_{t-1}}
    \right\rangle
    \le
    2\left\|Z_{\theta_{t-1}}(s,\cdot)-Z_t^A(s,\cdot)\right\|_2 .
$$
Summing the three bounds proves the claim.

<!-- assumption-2: body_md -->
\label{ass:replay_error_decomp}
For every $T\ge2$, the following bounds hold.
\begin{enumerate}
    \item Fitting error:
    $$
    \frac1{T-1}\sum_{t=2}^T
    \left\|
        Z_{\theta_{t-1}}-
        \widetilde Z_t
    \right\|_{2,\mu_t}
    \le
    \bar\epsilon_{\mathrm{fit}}.
    $$

    That is, the fitting error is not growing on average.

    \item Finite replay concentration:
    $$
    \left\|
        \widetilde Z_t
        -
        \widehat Z_t
    \right\|_{2,\mu_t}
    \le
    C_{\mathrm{var}}t^{-\zeta/2}
    \qquad\text{for every }t\ge2.
    $$

    This rate is the standard $N^{-1/2}$ concentration for the empirical mean of $N_t\asymp t^\zeta$ bounded labels around its conditional expectation, with $C_{\mathrm{var}}$ depending on the label range $G_A$, the regression function class capacity, and a logarithmic factor from a union bound over $t$.

    \item Critic label bias:
    $$
    \frac1{T-1}\sum_{t=2}^T
    \left\|
        \widehat Z_t
        -
        Z_t^A
    \right\|_{2,\mu_t}
    \le
    \epsilon_0.
    $$

    That is, the critic bias is not growing.

\end{enumerate}

<!-- remark-1: body_md -->
The regularized gap is the natural goal for an entropy-regularized method. It is the quantity the RDA iterates optimize, the entropy floor and the stabilizer control it directly, and it converts to the ordinary value gap at the cost of one explicit entropy bias term (Corollary~\ref{cor:unregularized_rate}).

The bound has three groups of terms: the dual averaging stabilizer (optimization) term, the local replay window drift, and the replay regression error, namely supervised fit $\bar\epsilon_{\mathrm{fit}}$, critic label bias $\epsilon_0$, and finite buffer variance $C_{\mathrm{var}}$, transferred to the comparator through coverage. The proof is given in Appendix~\ref{app:proofs}.

The replay terms expose the window length tradeoff. With $W_t\asymp t^\zeta$, finite buffer variance scales as $T^{-\zeta/2}$ and replay drift as $T^{\zeta-1}$. Balancing the two gives $\zeta=2/3$ and a $T^{-1/3}$ rate. Smaller $\zeta$ uses fresher but noisier data, while larger $\zeta$ averages more data but increases stale replay drift.
