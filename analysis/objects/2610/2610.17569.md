<!-- definition-1: body_md -->
(Upper box dimension)
Define the parabolic distance by
\begin{equation}\label{eq:parabolic-distance}
d_{\mathrm p}\bigl((x,t),(y,s)\bigr)
:=\max\bigl\{|x-y|,|t-s|^{1/2}\bigr\}.
\end{equation}
For a compact set $X$, let $N_{\mathrm p}(X,r)$ be the least number of
parabolic balls of radius $r$ needed to cover $X$, and define the upper box dimension of $X$ as
\begin{equation}\label{eq:box-dim}
\overline{\dim}_{\mathrm B,p} X
:=\limsup_{r\to 0}
\frac{\log N_{\mathrm p}(X,r)}{-\log r}.
\end{equation}
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
(Local $\varepsilon$-criterion at one scale)\label{thm:density}	
	There exist an absolute constant $\varepsilon>0$ and a radius $\rho_0<1$ with the following property. If
	$Q_{4\rho}(z_0)\Subset\Omega\times(-T,0)$ with $\rho\le \rho_0$ and
	\begin{equation}\label{eq:density-criterion}
		\begin{aligned}
			\iint_{Q_{4\rho}(z_0)}\Bigl(
			|\nabla u|^2+|u|^{10/3}
			+|\pi-\overline{\pi}_{B_{4\rho}}(t)|^{5/3}
			+|\nabla\pi|^{5/4}
			\Bigr)\,\mathrm{d} x\,\mathrm{d} t
			\le \varepsilon\rho^{15/13},
		\end{aligned}
	\end{equation}
	then $z_0$ is a regular point. Here the pressure mean is taken only in space:
	\[
	\overline{\pi}_{B_r}(t):=\frac1{|B_r|}\int_{B_r}\pi(x,t)\,\mathrm{d} x.
	\]
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
(Main theorem)\label{thm:main}
Let $(u,\pi)$ be a suitable weak solution of \eqref{eq:NS},
then we have
\begin{equation}\label{eq:main-result}
\overline{\dim}_{\mathrm B,p}(\mathcal{S})\le\frac{15}{13}.
\end{equation}
<!-- end theorem-2 -->
<!-- definition-2: body_md -->
(Suitable weak solution)

A  pair   $(u, \pi)$  is called a suitable weak solution to the Navier--Stokes equations \eqref{eq:NS} if the following conditions are satisfied:
\begin{enumerate}
\item \label{eq:energy-class}
$u\in L^\infty_tL^2_x (\Omega_T)\cap L^2_tH^1_x  (\Omega_T),
\quad \pi\in L^{3/2}_{t,x} (\Omega_T)$;
\item
$(u, \pi)$~solves \eqref{eq:NS} in $\mathbb{R}^{3}\times (-T,\,0) $ in the sense of distributions;
\item
For every nonnegative $\phi\in C_0^\infty(\Omega\times(-T,0))$ and
almost every $t$,
\begin{equation}\label{eq:LEI}
\begin{aligned}
&\int_\Omega |u(x,t)|^2\phi(x,t)\,\mathrm{d} x
+2\int_{-T}^{t}\!\int_\Omega |\nabla u|^2\phi\,\mathrm{d} x\,\mathrm{d} s\\
&\quad\le
\int_{-T}^{t}\!\int_\Omega |u|^2(\partial_s\phi+\Delta\phi)\,\mathrm{d} x\,\mathrm{d} s
+\int_{-T}^{t}\!\int_\Omega (|u|^2+2\pi)
u\cdot\nabla\phi\,\mathrm{d} x\,\mathrm{d} s.
\end{aligned}
\end{equation}
\end{enumerate}

	|u|^{10/3},\quad |\nabla u|^2,\quad

<!-- lemma-1: body_md -->
(\cite{WY}, Scale interpolation for the velocity)\label{lem:velocity-interpolation}
If $0<r\le\rho/2$, then
\begin{equation}\label{eq:E2-interpolation}
E_2(r)
\le C\left(\frac{\rho}{r}\right)E(\rho) 
+C\Bigl(\frac r\rho\Bigr)^2A(\rho).
\end{equation}
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
(\cite{WY}, Pressure-gradient decay)\label{lem:pressure-decay}
For $0<r\le\rho/2$,
\begin{equation}\label{eq:pressure-decay}
P_{1,3/2}(\nabla\pi,r)
\le C \left( \frac\rho r\right)  E(\rho)
+C\left( \frac r\rho\right)  P_{5/4}(\nabla\pi,\rho).
\end{equation}
<!-- end lemma-2 -->
<!-- lemma-3: body_md -->
(\cite{WY}, $\varepsilon$-regularity)\label{lem:epsilon-regularity}
There exists $\varepsilon_0>0$ such that if, for some $r$,
\begin{equation}\label{eq:epsilon-regularity}
E_2(r)+E(r)+P_{1,3/2}(\nabla\pi,r)
\le\varepsilon_0,
\end{equation}
then the center of the cylinder is a regular point.
<!-- end lemma-3 -->
<!-- lemma-4: body_md -->
(Nested-ball averages of pressure control)\label{lem:nested-average}
For $q\geq 1$,
\begin{equation}\label{eq:nested-average}
\int_{B_{2\rho}}|\pi-\overline{\pi}_{B_{2\rho}}|^q\,\mathrm{d} x
\le C_q\int_{B_{2\rho}}|\pi-\overline{\pi}_{B_{4\rho}}|^q\,\mathrm{d} x.
\end{equation}
<!-- end lemma-4 -->
<!-- proof-1: body_md -->
First write
\begin{equation}\label{eq:average-difference}
\overline{\pi}_{B_{2\rho}}-\overline{\pi}_{B_{4\rho}}
=\frac1{|B_{2\rho}|}\int_{B_{2\rho}}
  (\pi-\overline{\pi}_{B_{4\rho}})\,\mathrm{d} x.
\end{equation}
Jensen's inequality gives
\begin{equation}\label{eq:jensen-average}
|B_{2\rho}|\,
|\overline{\pi}_{B_{2\rho}}-\overline{\pi}_{B_{4\rho}}|^q
\le\int_{B_{2\rho}}|\pi-\overline{\pi}_{B_{4\rho}}|^q\,\mathrm{d} x.
\end{equation}
The $L^q$ triangle inequality now proves \eqref{eq:nested-average}.
<!-- end proof-1 -->
<!-- head-1: body_md -->
By  Theorem \ref{thm:density}, for every
$z\in\mathcal{S}$ and every sufficiently small $r$,
\begin{equation}\label{eq:singular-lower-density}
\iint_{Q_{4r}(z)}\Bigl(
|\nabla u|^2+|u|^{10/3}
+|\pi-\overline{\pi}_{B_{4r}(z)}|^{5/3}
+|\nabla\pi|^{5/4}
\Bigr)\,\mathrm{d} x\,\mathrm{d} t
>\varepsilon r^{15/13}.
\end{equation}

In light of known regularity results for suitable weak solutions to N-S equations, there exists 
a compact set  $K\subset\Omega_T$, such that $\mathcal{S}\subset K$.


For every  sufficiently small $\rho>0$, choose the maximal $\rho$-separated family
\[
\{z_j\}_{j=1}^{N_\rho}\subset\mathcal{S},
\]
such that the  parabolic cylinders $Q_{\rho/2}(z_j)$ may be chosen pairwise
disjoint, while $\cup_{j=1}^{N_\rho} Q_{\rho}(z_j)$  cover $\mathcal{S}$. Apply
\eqref{eq:singular-lower-density} with $r=\rho/8$; then
$Q_{4r}(z_j)=Q_{\rho/2}(z_j)$, so
\begin{equation}\label{eq:cover-sum}
\begin{aligned}
N_\rho\,\varepsilon\Bigl(\frac\rho8\Bigr)^{15/13}
&\le\sum_{j=1}^{N_\rho}
\iint_{Q_{\rho/2}(z_j)} \Bigl(|\nabla u|^2+|u|^{10/3}
+|\pi-\overline{\pi}_{B_{4r}(z)}|^{5/3}
+|\nabla\pi|^{5/4}
\Bigr)\,\mathrm{d} x\,\mathrm{d} t\\
&\le \iint_{Q_T} \Bigl(|\nabla u|^2+|u|^{10/3}
+|\pi|^{5/3}
+|\nabla\pi|^{5/4}
\Bigr)\,\mathrm{d} x\,\mathrm{d} t <\infty.
\end{aligned}
\end{equation}
The right-hand side is controlled by disjointness of the cylinders and the
 integrability in \eqref{eq:integrability}. The pressure oscillation is
also controlled by Jensen's inequality:
\begin{equation}\label{eq:pressure-jensen-cover}
\int_B|\pi-\overline{\pi}_B|^{5/3}\,\mathrm{d} x
\le C\int_B|\pi|^{5/3}\,\mathrm{d} x.
\end{equation}
Thus,
\begin{equation}\label{eq:N-rho-bound}
N_\rho\le {C\over\varepsilon} \rho^{-15/13}.
\end{equation}
By the definition \eqref{eq:box-dim} of upper box dimension,
\begin{equation}\label{eq:box-final}
\overline{\dim}_{\mathrm B,p}(\mathcal{S})\le\frac{15}{13}.
\end{equation}
This proves Theorem \ref{thm:main}.
