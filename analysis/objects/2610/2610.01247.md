<!-- defn-1: body_md -->
\citep{Nelsen(2006)}\label{df:qcopula}
		A two-dimensional function $Q$: $[0,1]^2\rightarrow[0,1]$ is said to be a \textit{bivariate quasi-copula} (for simplicity, a \textit{quasi-copula}) if it  satisfies the following conditions:
		\begin{itemize}
			\item[(Q1)] (Boundary conditions) $Q(0,t)=Q(t,0)=0$ and $Q(1,t)=Q(t,1)=t$ for all $t\in[0,1]$;
			\item[(Q2)] (Nondecreasing condition) $Q$ is nondecreasing in each argument, that is, for every fixed $v\in[0,1]$,  the mappings $u\mapsto Q(u, v)$ and $u \mapsto Q(v, u)$ are both nondecreasing; 
			\item[(Q3)] (Lipschitz condition) for
			all $u_1$, $u_2$, $v_1$ and $v_2$ in $[0, 1]$,
			
			\begin{equation}\label{Lp}
				|Q(u_1,v_1)-Q(u_2,v_2)|\le |u_1-u_2|+|v_1-v_2|. 
			\end{equation}	 
		\end{itemize}

<!-- defn-2: body_md -->
\citep{Nelsen(2006)}\label{df:copula}
		A two-dimensional function $C$: $[0,1]^2\rightarrow[0,1]$ is said to be a \textit{bivariate copula} (for simplicity, a   \textit{copula}) if it satisfies the following conditions:
		\begin{itemize}
			\item[(C1)] (Boundary conditions) $C(0,t)=C(t,0)=0$ and $C(1,t)=C(t,1)=t$ for all $t\in[0,1]$;
			\item[(C2)] (2-increasing condition) for all $u_1, u_2, v_1$ and $v_2$ in $[0, 1]$  with  $u_1\le u_2$ and $v_1\le v_2$, \begin{equation}\label{qe:V}V_C([u_1,u_2]\times[v_1,v_2]):=C(u_2,v_2)-C(u_2,v_1)-C(u_1,v_2)+C(u_1,v_1)\ge 0.\end{equation}
			Here, $V_C([u_1,u_2]\times[v_1,v_2])$ is referred to as the $C$-\textit{volume} of the rectangle $[u_1,u_2]\times[v_1,v_2]$.	 
		\end{itemize}

<!-- defn-3: body_md -->
\label{track} 	For any curve $\phi\in\Phi$, the \textit{track section} (or \textit{curvilinear section}) of a quasi-copula  $Q$ along $\phi$ is the function $\Gamma_{Q,\phi}:$ $[0,1]\rightarrow [0,1]$ defined by $\Gamma_{Q,\phi}(t):=Q\big(t,\phi(t)\big)$, $t\in[0,1]$.  In the particular case where  $\phi={\rm id}$, i.e., $\phi(t)=t$ for all $t\in[0,1]$, the track section reduces to the \textit{diagonal section} of $Q$, denoted by  $\Gamma_{Q, {\rm id}}(t)=Q(t,t)=\delta_Q(t)$.
<!-- end defn-3 -->
<!-- defn-4: body_md -->
\label{sub-track}
		Given a quasi-copula $Q$ and $x_0\in[0, 1)$, the \textit{sub-track section} (or \textit{sub-curvilinear section}) of $Q$ along the $x_0$-curve $\phi_{x_0}$  is the function $\Gamma_{Q,\phi_{x_0}}:$ $[0,1-x_0]\rightarrow [0,1-x_0]$ defined by
		\begin{equation}\label{df:sub-track}
		\Gamma_{Q,\phi_{x_0}}(t):=Q\big(x_0+\phi^{-1}_{x_0}(t),t\big). 
		\end{equation}
<!-- end defn-4 -->
<!-- rem-1: body_md -->
\label{4} ($i$) In the special case where $x_0=0$, we have  $\Gamma_{Q,\phi_{0}}(t)=Q\big(\phi^{-1}_{0}(t),t\big)=\Gamma_{Q,\phi}\big(\phi^{-1}(t)\big)$, where $\phi=\phi_0$ and $\Gamma_{Q,\phi}(t)$ is the track section of $Q$ along the curve $\phi$ given in Definition \ref{track}.	
	
	($ii$) Note that for any $x_0\in[0, 1)$, when $\phi_{x_0}(t)=t$ for all $t\in[0,1-x_0]$, the sub-track section  $\Gamma_{Q,\phi_{x_0}}(t)=Q(x_0+t,t)$ reduces to the \textit{sub-diagonal section} $\delta_{Q, x_0}(t)$ of the quasi-copula $Q$ \citep{2008quasi-copulas}.
<!-- end rem-1 -->
<!-- rem-2: body_md -->
Using the equation $v=\phi_{x_0}(u-x_0)$, we obtain the $x_0$-curve  $\{\big(x_0+\phi^{-1}_{x_0}(t), t\big)|t\in[0,1-x_0]\}$ by setting $v=t$. Then, in Definition \ref{sub-track}, the corresponding expression of sub-track section is given by  $Q\big(x_0+\phi^{-1}_{x_0}(t), t\big)$. This formulation differs from the definition of a track section of $Q$ along a curve $\phi\in \Phi$ as presented in Definition \ref{track}, where the expression is instead chosen as $u=t$, resulting in  the curve $\{\big(t, \phi(t)\big)|t\in[0,1]\}$ and the  track section  $Q\big(t, \phi(t)\big)$. In the special case where $x_0=0$, from Remark \ref{4}($i$), we know that the two definitions coincide essentially. Equivalently, the sub-track section of $Q$ along the $x_0$-curve $\phi_{x_0}$ may alternatively be defined as $Q\big(x_0+t, \phi_{x_0}(t)\big)$, $t\in[0,1-x_0]$,   which is essentially consistent with Definition \ref{sub-track}. The expression adopted in Definition \ref{sub-track} is motivated by its convenience for the analytical arguments in the subsequent sections.

\label{sub-track1}
		Given $x_0\in[0, 1)$ and an $x_0$-curve $\phi_{x_0}\in\Phi_{x_0}$, a function $\Gamma_{\phi_{x_0}}$: $[0,1-x_0]\rightarrow$ $[0,1-x_0]$ is called a \textit{$\phi_{x_0}$-function}  if it fulfills   the following conditions:  		\begin{itemize}

			\item[(S2)] $0\le \Gamma_{\phi_{x_0}}(t_{2})-\Gamma_{\phi_{x_0}}(t_{1})\le t_{2}-t_{1}+	\phi^{-1}_{x_0}(t_{2})-\phi^{-1}_{x_0}(t_{1})$ for any $0\le t_{1}\le t_{2}\le 1-x_0$.	 
		\end{itemize}

<!-- thm-1: body_md -->
\label{prop:G}
	For a fixed $x_0\in (0, 1)$, let $\phi_{x_0}\in\Phi_{x_0}$ be an $x_0$-curve satisfying $\phi_{x_0}(t)\le t$ for all $t\in[0,1-x_0]$, and let $\Gamma_{\phi_{x_0}}$ 
	be a $\phi_{x_0}$-function. Take an arbitrary quasi-copula $Q_1$, and let $Q_2$ be a quasi-copula
whose track section along the curve $\phi(t)=\frac{\phi_{x_0}\big(t(1-x_0)\big)}{1-x_0}$, $t\in[0,1]$ is given by  
	\begin{equation}
	\label{eq:G} \Gamma_{\phi}(t)
=\frac{	\Gamma_{\phi_{x_0}}\Big(\phi_{x_0}\big(t(1-x_0)\big)\Big)}{1-x_0},\ t\in[0,1]. 
	\end{equation}
	Then, the $W$-ordinal sum $Q=(\langle 0,x_0, Q_1 \rangle, \langle x_0, 1, Q_2 \rangle)^W$ defined by \eqref{eq:W} is a quasi-copula satisfying  $\Gamma_{Q,\phi_{x_0}}=\Gamma_{\phi_{x_0}}$, i.e., $Q \in \mathbb{Q}_{\Gamma_{\phi_{x_0}}}$. Consequently, for every $\phi_{x_0}$-function  $\Gamma_{\phi_{x_0}}$  with $\phi_{x_0}(t)\le t$, $t\in [0,1-x_0]$, there exists a quasi-copula $Q$ 
	belonging to $\mathbb{Q}_{\Gamma_{\phi_{x_0}}}$.
<!-- end thm-1 -->
<!-- proof-1: body_md -->
It suffices to verify that, for every fixed  $x_0\in(0,1)$, the function $\Gamma_{\phi}$ defined by 	\eqref{eq:G} is indeed a track section, and that $Q\big( x_0+\phi^{-1}_{x_0}(t), t \big)= \Gamma_{\phi_{x_0}}(t)$ for all
	$t\in[0, 1-x_0]$. Since $\Gamma_{\phi_{x_0}}(0)=0$ and	$\Gamma_{\phi_{x_0}}(1-x_0)=1-x_0$, we obtain directly  that
	$\Gamma_{\phi}(0)=0$ and	$\Gamma_{\phi}(1)=1$. From Condition (S1), it also follows that   
	$$
	\Gamma_{\phi}(t)=\frac{	\Gamma_{\phi_{x_0}}\Big(\phi_{x_0}\big(t(1-x_0)\big)\Big)}{1-x_0}\ge \max\{0,\frac{t(1-x_0)+ x_0+\phi_{x_0}\big(t(1-x_0)\big)-1}{1-x_0}\}=\max\{0,t+\phi(t)-1\}
	$$
and 
$$\Gamma_{\phi}(t)=\frac{	\Gamma_{\phi_{x_0}}\Big(\phi_{x_0}\big(t(1-x_0)\big)\Big)}{1-x_0}\le\min\{\frac{\phi_{x_0}\big(t(1-x_0)\big)}{1-x_0},\  t+\frac{x_0}{1-x_0}\}=\min\{\phi(t),\  t+\frac{x_0}{1-x_0}\}\le \phi(t). $$	
From the assumption that $\phi_{x_0}(t)\le t$, $t\in[0, 1-x_0]$, we know that $\phi(t)=\frac{\phi_{x_0}\big(t(1-x_0)\big)}{1-x_0}\le t$, $t\in[0,1]$ such that  $\Gamma_{\phi}(t)\le\min\{t,\phi(t)\}$ for all $t\in[0, 1]$. Moreover, because $\Gamma_{\phi_{x_0}}$ fulfills Condition (S2), it follows immediately that, for all $t_1$, $t_2\in[0, 1]$ with $t_1 \le t_2$,
$$0\le \Gamma_{\phi}(t_2)-\Gamma_{\phi}(t_1)=\frac{\Gamma_{\phi_{x_0}}\Big(\phi_{x_0}\big(t_2(1-x_0)\big)\Big)-\Gamma_{\phi_{x_0}}\Big(\phi_{x_0}\big(t_1(1-x_0)\big)\Big)}{1-x_0}$$
$$\ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \   =\frac{\phi_{x_0}\big(t_2(1-x_0)\big)-\phi_{x_0}\big(t_1(1-x_0)\big)+(t_2-t_1)(1-x_0)}{1-x_0}=t_2-t_1+\phi(t_2)-\phi(t_1).$$
Hence, from \cite{De2019On}, we know that  $\Gamma_{\phi}$ is a track section along the curve $\phi$.

It is obvious that  the $x_0$-curve $\phi_{x_0}$, which connects the points $(x_0, 0)$ and $(1, 1-x_0)$, is entirely contained in the region $[0,1]^2 \backslash S_{x_0}$. It follows, by the property (S1) and the assumption $\phi_{x_0}(t)\le t$ for all $t\in[0,1-x_0]$, that
	$$C_{\Gamma_{\phi_{x_0}}}\big(x_0+\phi^{-1}_{x_0}(t),t\big)=\min\Big\{x_0+\phi^{-1}_{x_0}(t),t,	\frac{\Gamma_{\phi_{x_0}}\Big(\phi_{x_0}\big(x_0+\phi^{-1}_{x_0}(t)-x_0\big)\Big)+\Gamma_{\phi_{x_0}}(t)}{2}\Big\}=\min\big\{t,	\Gamma_{\phi_{x_0}}(t)\big\}=\Gamma_{\phi_{x_0}}(t).$$
 Thus, the last claim of the theorem  is established.

From the expression of $C_{\Gamma_{\phi_{x_0}}}$, it is immediate that   $C_{\Gamma_{\phi_{x_0}}}(u,v)=C_{\Gamma_{\phi_{x_0}}}(v,u)$ for all $(u,v)\in[0,1]^2$, and thus $C_{\Gamma_{\phi_{x_0}}}$ is symmetric on $[0,1]^2$.
	
We now prove that $C_{\Gamma_{\phi_{x_0}}}$ 
	is indeed a copula. The first boundary condition  in Definition  \ref{df:copula} is
	easily verified. The second condition, however, is less straightforward and  requires a separate  argument. 
	\begin{itemize}
		\item To this end, assume first that $v < 1-x_0$. Applying the property (S2), it 
		yields that
		$$\Gamma_{\phi_{x_0}}(1-x_0)-\Gamma_{\phi_{x_0}}(v)\le 1-x_0-v+\phi^{-1}_{x_0}(1-x_0)-\phi^{-1}_{x_0}(v).$$
		Using the equality $\Gamma_{\phi_{x_0}}(1-x_0)=1-x_0$ and the assumption $\phi_{x_0}(t)\le t$, this inequality is rewritten as 
		$$2v\le \phi^{-1}_{x_0}(v)+v\le \Gamma_{\phi_{x_0}}(v) +1-x_0. $$
		Then, 
		$$C_{\Gamma_{\phi_{x_0}}}(1,v)=\min\Big\{1,v, 	\frac{1-x_0+\Gamma_{\phi_{x_0}}(v)}{2}\Big\}=v.$$
		\item Next, we turn to the case $v \ge 1-x_0$. Under this assumption,
		$$C_{\Gamma_{\phi_{x_0}}}(1,v)=	\min\{H_{\phi_{x_0}}(1,v), H_{\phi_{x_0}}(v,1)\}.$$
		
		Notice that for all $(u,v)\in S_{U,x_0}$,
		\begin{equation}\label{ieq:1} H_{\phi_{x_0}}(u,v)\le  H_{\phi_{x_0}}(v,u)\end{equation} if and only if 
		$$u-\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u-x_0)\big)}{2}\le v-\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(v-x_0)\big)}{2}. $$
		As a consequence of the assumption that $t+\hat \Gamma^{I}_{\phi_{x_0}}\big(\phi_{x_0}(t)\big)=2t-\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(t)\big) $ is nondecreasing on  $[0, 1-x_0]$, it follows that,  for all $1-x_0\le u\le v\le 1$, the inequality \eqref{ieq:1} holds. 
		Hence, 	$$C_{\Gamma_{\phi_{x_0}}}(1,v)=H_{\phi_{x_0}}(v,1)= v+\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(1-x_0)\big)-(1-x_0)}{2}=v.$$
	\end{itemize}
	

Since the boundary conditions have been established for  $C_{\Gamma_{\phi_{x_0}}}$, the only remaining requirement is to verify the 2-increasing property, i.e., the condition
(C2) of Definition \ref{df:copula}. To this end,  several cases need to be examined.	
	
\noindent \textit{Case 1}: We now consider a rectangle $[u_1,  u_2]\times [v_1, v_2]$ that is completely contained either in  $T_{L,\phi_{x_0}}$ or in 
 the region bounded by the lines $u=x_0$, $v=1-x_0$,  $v=u$ and the $x_0$-curve $v=\phi_{x_0}(u-x_0)$. Then, the rectangle is entirely contained in  $[0,1]^2 \backslash S_{x_0}$, and thus 
	\begin{eqnarray*}
	&& V_{C_{\Gamma_{\phi_{x_0}}}}([u_1,u_2]\times[v_1,v_2])\nonumber \\
	&=&	\min\Big\{v_1,	\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_1-x_0)\big)+\Gamma_{\phi_{x_0}}(v_1)}{2}\Big\}+\min\Big\{v_2,	\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_2-x_0)\big)+\Gamma_{\phi_{x_0}}(v_2)}{2}\Big\} \nonumber \\
	&&-\min\Big\{v_1,	\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_2-x_0)\big)+\Gamma_{\phi_{x_0}}(v_1)}{2}\Big\}-\min\Big\{v_2,	\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_1-x_0)\big)+\Gamma_{\phi_{x_0}}(v_2)}{2}\Big\},\nonumber
\end{eqnarray*}
since $v_i\le u_j$ for all $i,j\in\{1,2\}$. 

The proof proceeds by considering the following sub-cases:

\noindent (a) $v_1\le \frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_1-x_0)\big)+\Gamma_{\phi_{x_0}}(v_1)}{2}\le \frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_2-x_0)\big)+\Gamma_{\phi_{x_0}}(v_1)}{2}$: 
$$V_{C_{\Gamma_{\phi_{x_0}}}}([u_1,u_2]\times[v_1,v_2])=\min\Big\{v_2,	\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_2-x_0)\big)+\Gamma_{\phi_{x_0}}(v_2)}{2}\Big\}-\min\Big\{v_2,	\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_1-x_0)\big)+\Gamma_{\phi_{x_0}}(v_2)}{2}\Big\}\ge 0$$
	since $\Gamma_{\phi_{x_0}}(\phi_{x_0}(t))$ is nondecreasing with respect to $t$ and $u_2\ge u_1$. 
	
\noindent (b) $ \frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_1-x_0)\big)+\Gamma_{\phi_{x_0}}(v_1)}{2}< v_1\le \frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_2-x_0)\big)+\Gamma_{\phi_{x_0}}(v_1)}{2}$, then the following cases can be distinguished:
\begin{itemize}
	\item [(b1)] It is worth noting that the two inequalities  $\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_1-x_0)\big)+\Gamma_{\phi_{x_0}}(v_1)}{2}< v_1$ and $v_2<\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_1-x_0)\big)+\Gamma_{\phi_{x_0}}(v_2)}{2}$ cannot hold simultaneously. Indeed, these
	two inequalities imply 
	$$v_2-v_1<\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_1-x_0)\big)+\Gamma_{\phi_{x_0}}(v_2)}{2}-\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_1-x_0)\big)+\Gamma_{\phi_{x_0}}(v_1)}{2}=\frac{\Gamma_{\phi_{x_0}}(v_2)-\Gamma_{\phi_{x_0}}(v_1)}{2},$$
	which contradicts the assumption that $t+\hat \Gamma^{II}_{\phi_{x_0}}(t)=2t-\Gamma_{\phi_{x_0}}(t)$ is nondecreasing on  $[0, 1-x_0]$. Hence, the following cases remain:
	\item [(b2)] If $\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_1-x_0)\big)+\Gamma_{\phi_{x_0}}(v_2)}{2}\le v_2<\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_2-x_0)\big)+\Gamma_{\phi_{x_0}}(v_2)}{2}$, then, by the nondecreasing property of   $t+\hat \Gamma^{II}_{\phi_{x_0}}(t)$   on  $[0, 1-x_0]$, it follows that 
	$$V_{C_{\Gamma_{\phi_{x_0}}}}([u_1,u_2]\times[v_1,v_2])=	\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_1-x_0)\big)+\Gamma_{\phi_{x_0}}(v_1)}{2}-	\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_1-x_0)\big)+\Gamma_{\phi_{x_0}}(v_2)}{2}+v_2-v_1\ge 0.$$
		\item [(b3)] If $\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_1-x_0)\big)+\Gamma_{\phi_{x_0}}(v_2)}{2}\le \frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_2-x_0)\big)+\Gamma_{\phi_{x_0}}(v_2)}{2}\le v_2$, then we have
	$$V_{C_{\Gamma_{\phi_{x_0}}}}([u_1,u_2]\times[v_1,v_2])=	\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_2-x_0)\big)+\Gamma_{\phi_{x_0}}(v_1)}{2}-v_1\ge 0.$$
\end{itemize}

\noindent (c) $\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_2-x_0)\big)+\Gamma_{\phi_{x_0}}(v_1)}{2}< v_1$. Then, it must hold that  $\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_2-x_0)\big)+\Gamma_{\phi_{x_0}}(v_2)}{2}\le v_2$. Indeed, otherwise the inequalities $\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_2-x_0)\big)+\Gamma_{\phi_{x_0}}(v_1)}{2}< v_1$ 
and $\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_2-x_0)\big)+\Gamma_{\phi_{x_0}}(v_2)}{2}> v_2$ imply
$$v_2-v_1<\frac{\Gamma_{\phi_{x_0}}(v_2)-\Gamma_{\phi_{x_0}}(v_1)}{2},$$
which contradicts the assumption that $t+\hat \Gamma^{II}_{\phi_{x_0}}(t)$ is nondecreasing on  $[0, 1-x_0]$. Hence, 
	\begin{eqnarray*}

Consequently, in this case we always have   $V_{C_{\Gamma_{\phi_{x_0}}}}([u_1,u_2]\times[v_1,v_2])\ge 0$.	
	
\noindent \textit{Case 2}:  Consider the rectangle $[u_1, u_2] \times[\phi_{x_0}(u_1-x_0), \phi_{x_0}(u_2-x_0)]$, where $u_1$ and $u_2$ satisfy $1\ge u_1\ge x_0\ge u_2-u_1>0$.

\noindent \textit{Case 3}: Consider a square $[u_1,u_2]\times[u_1,u_2]\subseteq[0,1]^2
\textbackslash S_{x_0}$, where $x_0 \le u_1 < u_2 \le 1-x_0$. Then

	\end{eqnarray*}	
where the second equality holds due to that   $\Gamma_{\phi_{x_0}}(t)\le t$ and $\phi_{x_0}(t)\le t$ for every $t\in[0,1-x_0]$, and the second inequality follows from the assumption that $\Gamma_{\phi_{x_0}}(t)-\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(t-x_0)\big)$ is nondecreasing on $[x_0,1-x_0]$.
	
\noindent \textit{Case 4}: Consider a rectangle $[u_1,  u_2]\times [v_1, v_2]\subseteq S_{L, x_0}$ with $u_1\ge v_2$, that is, $0 \le v_1 < v_2 \le u_1 <  u_2 \le x_0$.
                                                                                                                                                                                                                                                                                                                                                   
	Then 
		$$ V_{C_{\Gamma_{\phi_{x_0}}}}([u_1,  u_2]\times [v_1, v_2])=\frac{\Gamma_{\phi_{x_0}}(\min\{u_1,v_1\})}{2}+\frac{\Gamma_{\phi_{x_0}}(\min\{u_2,v_2\})}{2}-\frac{\Gamma_{\phi_{x_0}}(\min\{u_2,v_1\})}{2}-\frac{\Gamma_{\phi_{x_0}}(\min\{u_1,v_2\})}{2}=0.$$

\noindent \textit{Case 5}: Consider a square $[u_1,  u_2]\times [u_1, u_2]\subseteq S_{L, x_0}$. Then
	\begin{eqnarray*}
	V_{C_{\Gamma_{\phi_{x_0}}}}([u_1,  u_2]\times [u_1, u_2])	&=& \frac{\Gamma_{\phi_{x_0}}(\min\{u_1,u_1\})}{2}+\frac{\Gamma_{\phi_{x_0}}(\min\{u_2,u_2\})}{2}-\frac{\Gamma_{\phi_{x_0}}(\min\{u_2,u_1\})}{2}-\frac{\Gamma_{\phi_{x_0}}(\min\{u_1,u_2\})}{2}\nonumber\\
		&=& \frac{\Gamma_{\phi_{x_0}}(u_2)}{2}-\frac{\Gamma_{\phi_{x_0}}(u_1)}{2}\ge 0. 
	\end{eqnarray*}	

\noindent \textit{Case 6}: Consider a rectangle $[u_1,  u_2]\times [v_1, v_2]\subseteq S_{U, x_0}$ with $u_1\ge v_2$, that is, $1-x_0\le v_1 < v_2 \le u_1 < u_2 \le 1$. 	\begin{eqnarray*}
	V_{C_{\Gamma_{\phi_{x_0}}}}([u_1,  u_2]\times [v_1, v_2])	
	&=& H_{\phi_{x_0}}(v_2,u_2)+H_{\phi_{x_0}}(v_1,u_1)-H_{\phi_{x_0}}(v_2,u_1)-H_{\phi_{x_0}}(v_1,u_2)\nonumber \\
&=&	v_2+\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_2-x_0)\big)-(1-x_0)}{2}+v_1+\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_1-x_0)\big)-(1-x_0)}{2}
		\nonumber \\
		&&-v_2-\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_1-x_0)\big)-(1-x_0)}{2}-v_1-\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_2-x_0)\big)-(1-x_0)}{2}=0.\nonumber 
	\end{eqnarray*}	

\noindent \textit{Case 7}: Consider a square $[u_1,  u_2]\times [u_1, u_2]\subseteq  S_{U, x_0}$.
Then, 
	\begin{eqnarray*}
	V_{C_{\Gamma_{\phi_{x_0}}}}([u_1,  u_2]\times [u_1, u_2])	&=& H_{\phi_{x_0}}(u_2,  u_2)+H_{\phi_{x_0}}(u_1,  u_1)-2H_{\phi_{x_0}}(u_1,  u_2)
		\nonumber \\
		&=& u_2-\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_2-x_0)\big)}{2}-u_1+\frac{\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(u_1-x_0)\big)}{2}\ge 0, \nonumber 
	\end{eqnarray*}	
where the  inequality follows from the assumption that
$t+\hat \Gamma^{I}_{\phi_{x_0}}\big(\phi_{x_0}(t)\big)=2t-\Gamma_{\phi_{x_0}}\big(\phi_{x_0}(t)\big)$ is  nondecreasing on  $[0, 1-x_0]$.

2$ whose vertices lie in more than one of the  
considered regions can be divided into a finite number  of rectangles $R_i$, each belonging to one of the seven cases above. Then,  $V_{C_{\Gamma_{\phi_{x_0}}}}(R)$ equals the sum of  $V_{C_{\Gamma_{\phi_{x_0}}}}(R_i)$. Therefore, it holds that 
 $V_{C_{\Gamma_{\phi_{x_0}}}}(R)\ge 0$. 

This completes the proof.

\label{B+}
For a fixed $x_0\in (0,1)$ and a $\phi_{x_0}$-function $\Gamma_{\phi_{x_0}}$ with $\phi_{x_0}\in\Phi_{x_0}$ satisfying   $\phi_{x_0}(t)\le t$ for all $t\in[0,1-x_0]$, the copula $B_{\Gamma_{\phi_{x_0}}}$

As established in Proposition \ref{prop:Bertino copula*}, the function $B_{\Gamma_{\phi_{x_0}}}$ 
	is a copula and therefore a quasi-copula; moreover, $B_{\Gamma_{\phi_{x_0}}}\in \mathbb{C}_{\Gamma_{\phi_{x_0}}}\subset \mathbb{Q}_{\Gamma_{\phi_{x_0}}}$. Now 
	let $Q$ be any (quasi-)copula with the prescribed sub-track section $\Gamma_{\phi_{x_0}}$ along the $x_0$-curve $\phi_{x_0}$. Clearly,  $B_{\Gamma_{\phi_{x_0}}}(u, v)=W(u,v)\le  Q(u, v)$ for every $(u, v)\notin D_{L, x_0}$.  Now consider 
	 $(u,v)\in D_{L, x_0}$ and assume first that $v \ge \phi_{x_0}(u-x_0)$,  
	thus $B_{\Gamma_{\phi_{x_0}}}(u,v)=u-x_0-\hat \Gamma^{I}_{\phi_{x_0}}(t^*)$, where  $$\hat \Gamma^{I}_{\phi_{x_0}}(t^*):=\min\{\phi^{-1}_{x_0}(t)-\Gamma_{\phi_{x_0}}(t)| t\in[\phi_{x_0}(u-x_0),v] \}.$$
	
We recall  \citep[see][Proposition 3]{Genest1999A} that a quasi-copula satisfies the inequality \eqref{qe:V} whenever at least one of the arguments $u_1$, $u_2$, $v_1$ or $v_2$  
equals  either 0 or 1. Applying this inequality for $Q$ to the rectangle $[u, \phi^{-1}_{x_0}(t^*)+x_0]\times[t^*,1]$, it  
 yields that
$$0\le \phi^{-1}_{x_0}(t^*)+x_0-u+Q(u,t^*)-Q\big(\phi^{-1}_{x_0}(t^*)+x_0,t^*\big)=\phi^{-1}_{x_0}(t^*)-(u-x_0)+Q(u,t^*)-\Gamma_{\phi_{x_0}}(t^*),$$
and thus 
$$B_{\Gamma_{\phi_{x_0}}}(u,v)=u-x_0-\hat \Gamma^{I}_{\phi_{x_0}}(t^*)=\Gamma_{\phi_{x_0}}(t^*)-\big(\phi^{-1}_{x_0}(t^*)-(u-x_0)\big)\le Q(u,t^*)\le Q(u,v).$$
	
An analogous argument holds when $v < \phi_{x_0}(u-x_0)$. In this case, $B_{\Gamma_{\phi_{x_0}}}(u,v)=v-\hat \Gamma^{II}_{\phi_{x_0}}(t^\star)$, where  $$\hat \Gamma^{II}_{\phi_{x_0}}(t^\star):=\min\{t-\Gamma_{\phi_{x_0}}(t)|t\in[v, \phi_{x_0}(u-x_0)] \}.$$  
Applying the inequality  \eqref{qe:V} for $Q$ to the rectangle  $[\phi^{-1}_{x_0}(t^\star)+x_0, 1]\times[v,t^\star]$, it  
 yields 
$$0\le t^\star -v+Q\big(\phi^{-1}_{x_0}(t^\star)+x_0,v\big)-Q\big(\phi^{-1}_{x_0}(t^\star)+x_0, t^\star\big)=t^\star -v+Q\big(\phi^{-1}_{x_0}(t^\star)+x_0,v\big)-\Gamma_{\phi_{x_0}}(t^\star),$$
and then 
$$B_{\Gamma_{\phi_{x_0}}}(u,v)=v-\hat \Gamma^{II}_{\phi_{x_0}}(t^\star)=\Gamma_{\phi_{x_0}}(t^\star)-\big(t^\star-v\big)\le Q\big(\phi^{-1}_{x_0}(t^\star)+x_0,v\big)\le Q(u,v).$$
This completes the proof.

If $(u,v)\in T_{U,\phi_{x_0}}$ and either $u=0$ or $v=0$, Lemma \ref{lem24} implies  that $A_{\Gamma_{\phi_{x_0}}}(u,0)=A_{\Gamma_{\phi_{x_0}}}(0,v)=0$, since $v-q^{II}_{\phi_{x_0}}(u,v)\ge 0$  for all $(u,v)\in T_{U,\phi_{x_0}}$. We now turn to the case $(u,v)\in T_{L,\phi_{x_0}}$ with $v=0$, which means $u\ge x_0$ and $v=0$. Then, the inequality  $u-x_0-q^{I}_{\phi_{x_0}}(u,v)\ge 0$ implies $A_{\Gamma_{\phi_{x_0}}}(u,0)=0$.
	
	We now consider the boundary cases in which $u=1$ or $v=1$. 
	\begin{itemize}
	\item First, consider the case $u=1$. 
	\begin{itemize}
		\item In the case where $u=1$ and $v\ge 1-x_0$, from the expression \eqref{eq:G copula1*} of $A_{\Gamma_{\phi_{x_0}}}$, we have $A_{\Gamma_{\phi_{x_0}}}(1,v)=\min\{1, v-q^{II}_{\phi_{x_0}}(1,v)\}=v$, since $q^{II}_{\phi_{x_0}}(1,v)=\max\{t-\Gamma_{\phi_{x_0}}(t)|t\in[1-x_0,1-x_0]\}=1-x_0-\Gamma_{\phi_{x_0}}(1-x_0)=0$.

<!-- prop-4: body_md -->
The function  $A_{\Gamma_{\phi_{x_0}}}$ defined by \eqref{eq:G copula1*} is nondecreasing in each argument
	and satisfies the Lipschitz condition \eqref{Lp}.

<!-- proof-9: body_md -->
In order to show the nondecreasing property and the Lipschitz condition for  $A_{\Gamma_{\phi_{x_0}}}$, 
	it suffices to verify that  $A_{\Gamma_{\phi_{x_0}}}$ 
	is nondecreasing and satisfies 
	Lipschitz condition separately on $T_{U,\phi_{x_0}}$ and $T_{L,\phi_{x_0}}$. The proofs for other cases can be reduced to the two  situations considered above, using either the triangle inequality or basic summation. Accordingly, we distinguish four cases and, in each, prove the required 
	properties. 
	
	\noindent \textit{Case} 1: Consider the case that both $(u_1, v)$ and $(u_2, v)$ are in $T_{U,\phi_{x_0}}$ with $u_1\le  u_2$. Then,  $\phi_{x_0}(\max\{u_1-x_0,0\})\le \phi_{x_0}(\max\{u_2-x_0,0\})\le v$, and consequently $m_{\phi_{x_0}}(u_1,v)\le m_{\phi_{x_0}}(u_2,v)$ while  $M_{\phi_{x_0}}(u_1,v)= M_{\phi_{x_0}}(u_2,v)$. It implies 
$q^{II}_{\phi_{x_0}}(u_1,v)\ge q^{II}_{\phi_{x_0}}(u_2,v)$. Recall that for every   $(u, v)\in T_{U,\phi_{x_0}}$,
$A_{\Gamma_{\phi_{x_0}}}(u,v)=\min\{u, v-q^{II}_{\phi_{x_0}}(u,v)\}$. 

\begin{itemize}
	\item We first show   that, for each fixed $v$, the mapping $u \mapsto A_{\Gamma_{\phi_{x_0}}}(u,v)$ is nondecreasing. If $A_{\Gamma_{\phi_{x_0}}}(u_2,v)=u_2$, then
	$$A_{\Gamma_{\phi_{x_0}}}(u_2,v)-A_{\Gamma_{\phi_{x_0}}}(u_1,v)=u_2-\min\{u_1, v-q^{II}_{\phi_{x_0}}(u_1,v)\}\ge u_2-u_1\ge 0.$$
	If $A_{\Gamma_{\phi_{x_0}}}(u_2,v)=v-q^{II}_{\phi_{x_0}}(u_2,v)$, then
	$$A_{\Gamma_{\phi_{x_0}}}(u_2,v)-A_{\Gamma_{\phi_{x_0}}}(u_1,v)=v-q^{II}_{\phi_{x_0}}(u_2,v)-\min\{u_1, v-q^{II}_{\phi_{x_0}}(u_1,v)\}\ge v-q^{II}_{\phi_{x_0}}(u_2,v)-\big( v-q^{II}_{\phi_{x_0}}(u_1,v)\big)\ge 0.$$
	\item We now turn to the Lipschitz condition. If $A_{\Gamma_{\phi_{x_0}}}(u_1,v)=u_1$, then
	$$A_{\Gamma_{\phi_{x_0}}}(u_2,v)-A_{\Gamma_{\phi_{x_0}}}(u_1,v)=\min\{u_2, v-q^{II}_{\phi_{x_0}}(u_2,v)\}-u_1\le u_2-u_1.$$
	If $A_{\Gamma_{\phi_{x_0}}}(u_1,v)=v-q^{II}_{\phi_{x_0}}(u_1,v)$, then
	$$A_{\Gamma_{\phi_{x_0}}}(u_2,v)-A_{\Gamma_{\phi_{x_0}}}(u_1,v)\le  v-q^{II}_{\phi_{x_0}}(u_2,v)-\big(v-q^{II}_{\phi_{x_0}}(u_1,v)\big)\le q^{II}_{\phi_{x_0}}(u_1,v)-q^{II}_{\phi_{x_0}}(u_2,v).$$
	It therefore remains to verify the inequality $q^{II}_{\phi_{x_0}}(u_1,v)-q^{II}_{\phi_{x_0}}(u_2,v)\le u_2-u_1$.
	
	If $q^{II}_{\phi_{x_0}}(u_1,v)=q^{II}_{\phi_{x_0}}(u_2,v)$, the claim is trivial. Then, suppose that $q^{II}_{\phi_{x_0}}(u_1,v)>q^{II}_{\phi_{x_0}}(u_2,v)$.  It implies that $m_{\phi_{x_0}}(u_2,v)>m_{\phi_{x_0}}(u_1,v)\ge 0$, $m_{\phi_{x_0}}(u_1,v)=\phi_{x_0}(\max\{u_1-x_0,0\})$ and 
	$m_{\phi_{x_0}}(u_2,v)=\phi_{x_0}(u_2-x_0)$. Consequently, there exist $t'$ and $t''$ satisfying  
	$$m_{\phi_{x_0}}(u_1,v)\le t'<m_{\phi_{x_0}}(u_2,v)\le t''\le M_{\phi_{x_0}}(u_2,v)  $$
	such that $q^{II}_{\phi_{x_0}}(u_1,v)=\hat\Gamma^{II}_{\phi_{x_0}}(t')$ and $q^{II}_{\phi_{x_0}}(u_2,v)=\hat\Gamma^{II}_{\phi_{x_0}}(t'')$. Since $\hat\Gamma^{II}_{\phi_{x_0}}(t'')$ is the maximum of $\hat\Gamma^{II}_{\phi_{x_0}}$ 
	on the interval $[m_{\phi_{x_0}}(u_2,v),$
	$ M_{\phi_{x_0}}(u_2,v)]$, we have  $\hat\Gamma^{II}_{\phi_{x_0}}(t'')\ge \hat\Gamma^{II}_{\phi_{x_0}}(t)$ for every $t$ in that interval, and 
	in particular for $t =m_{\phi_{x_0}}(u_2,v)= \phi_{x_0}(u_2-x_0)$. Applying the property (D2) of $\hat\Gamma^{II}_{\phi_{x_0}}$, we obtain 
	$$q^{II}_{\phi_{x_0}}(u_1,v)-q^{II}_{\phi_{x_0}}(u_2,v)=\hat\Gamma^{II}_{\phi_{x_0}}(t')-\hat\Gamma^{II}_{\phi_{x_0}}(t'')\le \hat\Gamma^{II}_{\phi_{x_0}}(t')-\hat\Gamma^{II}_{\phi_{x_0}}\big(\phi_{x_0}(u_2-x_0)\big)\le u_2-x_0-\phi^{-1}_{x_0}(t') $$
	$$\le u_2-x_0-\max\{u_1-x_0,0\}\le u_2-u_1,\ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \   $$
	which establishes the claim in this case. 
	
\end{itemize}





 
 \noindent \textit{Case} 2:   Let both $(u, v_1)$ and $(u, v_2)$ be in $T_{U,\phi_{x_0}}$ with $v_1\le  v_2$, meaning that $\phi_{x_0}(\max\{u-x_0,0\})\le v_1\le v_2$. In this case, we know that 
 $$m_{\phi_{x_0}}(u,v_1)=m_{\phi_{x_0}}(u,v_2)=\phi_{x_0}(\max\{u-x_0,0\})$$
 and 
 $$M_{\phi_{x_0}}(u,v_1)=v_1\le v_2=M_{\phi_{x_0}}(u,v_2).$$
Thus, 
 $q^{II}_{\phi_{x_0}}(u,v_1)\le q^{II}_{\phi_{x_0}}(u,v_2)$. Recall that for any $(u, v)\in T_{U,\phi_{x_0}}$,  $A_{\Gamma_{\phi_{x_0}}}(u,v)=\min\{u, v-q^{II}_{\phi_{x_0}}(u,v)\}$. 
 
 \begin{itemize}
 	\item To verify the nondecreasing property of $A_{\Gamma_{\phi_{x_0}}}$,  assume first that  $A_{\Gamma_{\phi_{x_0}}}(u,v_2)=u$.  Then
 	$$A_{\Gamma_{\phi_{x_0}}}(u,v_2)-A_{\Gamma_{\phi_{x_0}}}(u,v_1)=u-\min\{u, v_1-q^{II}_{\phi_{x_0}}(u,v_1)\}\ge u-u= 0.$$

 	$$A_{\Gamma_{\phi_{x_0}}}(u,v_2)-A_{\Gamma_{\phi_{x_0}}}(u,v_1)\ge v_2-q^{II}_{\phi_{x_0}}(u,v_2)-\big(v_1-q^{II}_{\phi_{x_0}}(u,v_1)\big)=v_2-v_1-\big(q^{II}_{\phi_{x_0}}(u,v_2)-q^{II}_{\phi_{x_0}}(u,v_1)\big) .$$ 
 	It therefore remains to verify the inequality  $v_2-v_1\ge q^{II}_{\phi_{x_0}}(u,v_2)-q^{II}_{\phi_{x_0}}(u,v_1)$. This inequality is obviously satisfied when
 	$q^{II}_{\phi_{x_0}}(u,v_2)=q^{II}_{\phi_{x_0}}(u,v_1)$.  In the case where  $q^{II}_{\phi_{x_0}}(u,v_2)>q^{II}_{\phi_{x_0}}(u,v_1)$,  we have $M_{\phi_{x_0}}(u,v_1) =v_1 < 1-x_0$, and there
 	exist $t'$ and $t''$ satisfying  $m_{\phi_{x_0}}(u,v_1)\le t'\le v_1<t''\le M_{\phi_{x_0}}(u,v_2)$ such that 
 	$$q^{II}_{\phi_{x_0}}(u,v_2)=\hat\Gamma^{II}_{\phi_{x_0}}(t'')>\hat\Gamma^{II}_{\phi_{x_0}}(t')=q^{II}_{\phi_{x_0}}(u,v_1).$$
 	Then, from the property (D2) of  $\hat\Gamma^{II}_{\phi_{x_0}}$
 	and the fact that $\hat\Gamma^{II}_{\phi_{x_0}}(t')$ is the maximum of $\hat\Gamma^{II}_{\phi_{x_0}}$ 
 	on the interval $[m_{\phi_{x_0}}(u,v_1), M_{\phi_{x_0}}(u,v_1)]$, we have
 	$$0\le q^{II}_{\phi_{x_0}}(u,v_2)-q^{II}_{\phi_{x_0}}(u,v_1)=\hat\Gamma^{II}_{\phi_{x_0}}(t'')-\hat\Gamma^{II}_{\phi_{x_0}}(t')\le \hat\Gamma^{II}_{\phi_{x_0}}(t'')-\hat\Gamma^{II}_{\phi_{x_0}}(v_1)$$
 	$$ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \  \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \le t''-v_1\le\min\{v_2,1-x_0\}-v_1\le v_2-v_1.   $$	
 \item	We now turn to the Lipschitz condition of $A_{\Gamma_{\phi_{x_0}}}$. If $A_{\Gamma_{\phi_{x_0}}}(u,v_1)=u$, then

 \end{itemize}
