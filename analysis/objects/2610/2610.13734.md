<!-- proposition-1: body_md -->
\label{P1}Let $A$ be a Banach lattice algebra and $f\in A^{\prime}$. The
following are equivalent.

\begin{enumerate}
\item[(1)] $f\in\mathcal{L}$-\textrm{$wap$}$\mathrm{(}A)$.

\item[(2)] For every norm bounded disjoint sequence $(x_{n})$, one has
$\left\Vert f\circ x_{n}\right\Vert \longrightarrow0$ in $A^{\prime}$.

\item[(3)] For every norm bounded disjoint sequence $(x_{n})_{n\in\mathbb{N}}$
and every positive sequence $(y_{n})_{n\in\mathbb{N}}$ in $B_{A}$, we have
\[
\lim\limits_{n\longrightarrow\infty}f(y_{n}x_{n})=0\text{.} 
\]

\end{enumerate}
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
(1)$\Leftrightarrow$(2) follows from the standard characterization of
$L$-weakly compact subsets of a Banach lattice, see \cite[Theorem
5.63]{Aliprantis} or \cite[Proposition 3.6.3]{Meyer}.

(2)$\Longrightarrow$(3) This implication is straightforward.

(3)$\Longrightarrow$(2) Suppose, for contradiction, that there exists a
disjoint sequence $(x_{n})_{n\in\mathbb{N}}\subset$ $B_{A}$ such that
$(\left\Vert f\circ x_{n}\right\Vert )_{n\in\mathbb{N}}$ does not converges to
zero. Passing to a subsequence if necessary, we may assume that there exists
some $\varepsilon>0$ such that $\left\Vert f\circ x_{n}\right\Vert
\geq\varepsilon$ for all $n\in\mathbb{N}$. Then, for each $n\in\mathbb{N}$, we
can choose $0\leq y_{n}\in B_{A}$ such that $\left\vert f\circ x_{n} 
\right\vert (y_{n})\geq\dfrac{\varepsilon}{2}.$ Using the classical
Kantorovich formula
\[
\frac{\varepsilon}{2}\leq\left\vert f\circ x_{n}\right\vert (y_{n} 
)=\sup\left\{  \left\vert f\circ x_{n}(z)\right\vert :0\leq z\leq
y_{n}\right\}  \text{ for all }n\in\mathbb{N}\text{,} 
\]
there exists $0\leq z_{n}\leq y_{n}$ such that $\left\vert f\circ x_{n} 
(z_{n})\right\vert =\left\vert f(z_{n}x_{n})\right\vert \geq\dfrac
{\varepsilon}{4}$. This contradicts (3).

<!-- theorem-1: body_md -->
\label{Main}Let $A$ be a Banach lattice.

\begin{enumerate}
\item If the dual space $A^{\prime}$ has an order continuous norm, then for
every Banach lattice algebra product $p$ on $A$, \textrm{$wap$}$\mathrm{(} 
A)=A^{\prime}$.

In particular, if $A^{\prime}$ has an order continuous norm then, $A$ equipped
with any product, is Arens regular.

\item If the dual space $A^{\prime}$ has the PSP then, for every Banach
lattice algebra product $p$ on $A$,
\[
\mathcal{L}\text{-}\mathrm{wap(}A)=\mathrm{wap(}A)=A^{\prime}.
\]


\item If for every Banach lattice algebra product $p$ on $A$, $\mathcal{L} 
$-\textrm{$wap$}$\mathrm{(}A)=A^{\prime}$ then $A^{\prime}$ has an order
continuous norm.
\end{enumerate}

\leq1$ such that $g_{n}=f.x_{n}$. Since $A^{\prime}$ has order continuous
norm, it follows from \cite[Theorem 5.28]{Aliprantis} that $(g_{n} 
)_{n\in\mathbb{N}}$ admits a weak Cauchy subsequence. Consequently,
$(g_{n})_{n\in\mathbb{N}}$ has a weakly convergent subsequence. By
Eberlein-Smulian Theorem, $O_{f}$ is relatively weakly compact. Therefore,
$f\in$\textrm{$wap$}$\mathrm{(}A)$.

\item Assume that $A^{\prime}$ has the PSP then $A^{\prime}$ has an order
continuous norm. Let $p$ be a Banach lattice algebra product on $A$, and let
$0\leq f\in A^{\prime}$. By the first point the orbit $O_{f}$ is actually
relatively weakly compact subset. According to \cite[Theorem 4.34] 
{Aliprantis}, if $(g_{n})\subset S(O_{f})$ is a disjoint sequence then it
converges weakly to zero and, by the PSP, it converges to zero in norm. Thus
$f\in\mathcal{L}$-\textrm{$wap$}$\mathrm{(}A)$.

\item Assume that $\mathcal{L}$-\textrm{$wap$}$\mathrm{(}A)=A^{\prime}$ for
every Banach lattice algebra product $p$ on $A$. To show that $A^{\prime}$ has
order continuous norm, it suffices to prove that every norm bounded disjoint
sequence in $A$ converges weakly to zero. Let $(x_{n})_{n\in\mathbb{N}}$ be a
disjoint sequence in $B_{A}$. Consider a positive functional $0\leq\varphi\in
A^{\prime}$ with $\left\Vert \varphi\right\Vert =1$ and define the product
$.:A\times A\longrightarrow A$ by $x.y=\varphi(y)x$. Then $A$, equipped with
this product becomes a Banach lattice algebra. Let $f\in A^{\prime}$, for
every $x\in A$, the action of $f\circ x$ is given by $f\circ x=f(x)\varphi$.
Since $f\in\mathcal{L}$-\textrm{$wap$}$\mathrm{(}A)$, $\left\Vert f\circ
x_{n}\right\Vert =\left\vert f(x_{n})\right\vert \longrightarrow0$. Therefore,
$x_{n}\longrightarrow0$ weakly in $A$. This shows that $A^{\prime}$ has an
order continuous norm.
\end{enumerate}

<!-- proposition-2: body_md -->
Let $A$ be a Banach $f$-algebra. Then \textrm{$wap$}$(A)=A^{\prime}$ , and
consequently $A$ is Arens regular.

<!-- proof-3: body_md -->
Assume first that $A$ has a unit element. Then its dual space $A^{\prime}$ has
an order continuous norm (see \cite[Theorem 3.3]{depagter}). By Theorem
\ref{Main}, this implies that \textrm{$wap$}$(A)=A^{\prime}$.

If $A$ is semiprime $f$-algebra. Then $A$ is continuously embedded in the
unital Banach algebra $\mathcal{Z}(A)$, via the mapping $x\longmapsto\pi_{x}$
where $\pi_{x}:y\longmapsto xy$. Since $\mathcal{Z}(A)$ is Arens regular, $A$
is Arens regular, because every subalgebra of an Arens regular algebra is
Arens regular (\cite[Corollary 1.4.12]{Palmer}).

In the general case, let $N$ denote the set of all nilpotent elements in $A$
and let
\[

\]

10.2]{BD}). Let $f\in A^{\prime}$ and denote by $g=f_{/M}$ its restriction to

y_{m})=\beta\text{ }
