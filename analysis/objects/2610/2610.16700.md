<!-- definition-1: body_md -->
\label{def:gen.sol} \textit{Let $f_0\in L^1(\mathbb{R}^{2d})$ and  $0\leq f_0(x,p)\leq 1$ for almost every $(x,p)\in \mathbb{R}^{2d}$. 
A measurable  function $f: (0, \infty)\times \mathbb{R}^{2d}\to \mathbb{R}$   is called an \emph{entropy solution} of  \eqref{DEq}   with the initial data $f_0$  if: 
\begin{itemize}
\item $0\leq f(t,x,p)\leq 1$ for almost every $(t,x,p)\in (0,\infty)\times \mathbb{R}^{2d};$
\item $f\in L^1((0,T)\times \mathbb{R}^{2d})$, $f\in BV((0,T)\times \mathbb{R}^{2d})$,  and $\nabla_p f\in [L^2((0,T)\times \mathbb{R}^{2d})]^d$ for all $T>0$;
\item For any smooth compactly supported $\phi\in C_c^{\infty}((0, \infty)\times \mathbb{R}^{2d}) $ with $\phi\geq 0$ and any constant $a\in \mathbb{R}$
\begin{align}\label{def}
\int_0^{\infty}\int_{\mathbb{R}^{2d}}\mathrm{sign}(f-a)\big\{(f-a)\partial_t \phi &+(f-f^2-(a-a^2))(p\cdot \nabla_x \phi-p\cdot \nabla_p \phi)\nonumber \\
&-\nabla_p f\cdot\nabla_p \phi+d(a-a^2)\phi \big\}dxdpdt\geq 0;
\end{align}
\item There is a set $\mathcal{E}  $  of zero Lebesgue  measure on $(0,1)$ such that 
$$ \lim_{t \to 0,\, t\in (0,1) \setminus \mathcal{E}} \int_{\mathbb{R}^{2d}}|f(t,x,p)-f_0(x,p)|dxdp=0.$$ 
\end{itemize}
}

<!-- theorem-1: body_md -->
\label{th:unique}
Let $d\in \mathbb{N}.$ Let $f$ and $g$ be two entropy solutions to \eqref{DEq} with the respective initial data $f_0,g_0\in L^1(\mathbb{R}^{2d})$ satisfying  $0\leq f_0(x,p),g_0(x,p)\leq 1$ for almost every $(x,p)\in \mathbb{R}^{2d}$. Then \begin{align*}
\int_{\mathbb{R}^{2d}}|f(t,x,p)-g(t,x,p)| dxdp\leq  \int_{\mathbb{R}^{2d}}|f_0(x,p)-g_0(x,p)| dxdp
\end{align*} 
for almost every $t\in (0,\infty).$
In particular, for any initial data $f_0$ as above, there is at most one entropy solution $f$.

<!-- theorem-2: body_md -->
\label{th:global existence}
Let $d\in \mathbb{N},$  $k>2d$,  $f_0\in L^1_{k}(\mathbb{R}^{2d})\cap W^{1,1}(\mathbb{R}^{2d}) $,   $0\leq f_0(x,p)\leq 1$ for almost every $(x,p)\in \mathbb{R}^{2d},$ $\mathcal{L}[f_0]\in L^1(\mathbb{R}^{2d})$, and $\Delta_x f_0\in L^1(\mathbb{R}^{2d})$. Then there is a unique entropy solution $f: (0, \infty)\times \mathbb{R}^{2d}\to \mathbb{R}$   as in Definition \ref{def:gen.sol}. We have  
 $$  \int_{\mathbb{R}^{2d}}f(t,x,p)dpdx=\int_{\mathbb{R}^{2d}}f_0(x,p)dpdx$$
 for almost every $t\in (0,\infty)$.     For all $ \gamma\in [1, k]$, the $\gamma$-moment of $f$ with respect to $p$, i.e.,  $\int_{\mathbb{R}^{2d}}|p|^{\gamma}f(t,x,p)dxdp,$ is bounded by a positive polynomial $P_{\lceil \gamma/2 \rceil}(t)$   (whose coefficients depend only on  $\int_{\mathbb{R}^{2d}}f_0(x,p)dxdp,$ $\int_{\mathbb{R}^{2d}}|p|^{\gamma}f_0(x,p)dxdp,$ $\gamma$, and $d$) of degree $\lceil \gamma/2 \rceil$ for almost every $t\in (0,\infty)$. In particular, $f(t,\cdot,\cdot)\in L_k^1(\mathbb{R}^{2d})$ for almost every $t\in (0,\infty)$.

<!-- remark-1: body_md -->
$(i)$ The condition $f_0\in L^1_{k}(\mathbb{R}^{2d})$ with $k>2d$ is needed to prove existence of global solutions; see Lemma \ref{lem:p-moment} and Theorem \ref{th:global} below.  We note that $ L_{k}^1(\mathbb{R}^{2d})\subset L_{\gamma}^1(\mathbb{R}^{2d}) $  for all $ \gamma \in [1, k].$\\
$(ii)$ The regularity assumptions $f_0\in  W^{1,1}(\mathbb{R}^{2d}) $, $\mathcal{L}[f_0]\in L^1(\mathbb{R}^{2d})$, and  $\Delta_x f_0\in L^1(\mathbb{R}^{2d})$ are needed to obtain   uniform bounds with respect to $\varepsilon\in (0,1]$ in $W^{1,1}((0, T)\times \mathbb{R}^{2d})$  (with $T>0$) for the solution of \eqref{Eq}, which imply compactness in $BV((0, T)\times \mathbb{R}^{2d});$ see Lemma \ref{l:epsilon} and the proof of Theorem \ref{th:global existence}. In view of $0\leq f_0\leq 1$, the condition $\mathcal{L}[f_0]\in L^1(\mathbb{R}^{2d})$ is satisfied if  $\Delta_p f_0$, $p\cdot \nabla_x f_0$, and $p\cdot \nabla_p f_0$ belong to $ L^1(\mathbb{R}^{2d})$.
\\
$(iii)$ The uniqueness of the solution in Theorem \ref{th:global existence}  follows directly from Theorem \ref{th:unique}.

<!-- lemma-1: body_md -->
Let $f$ and $g$ be two entropy solutions to \eqref{DEq} as in Definition \ref{def:gen.sol}. Then 
\begin{align}\label{I>0}
 \int_0^{\infty}\int_{\mathbb{R}^{2d}}\mathrm{sign}(f-g)\big\{(f-g)\partial_t \phi &+(f-f^2-(g-g^2))(p\cdot \nabla_x \phi-p\cdot \nabla_p \phi)\nonumber \\
&-\nabla_p (f-g)\cdot\nabla_p \phi \big\}dxdpdt\geq 0
\end{align}
for all $\phi \in  C_c^{\infty}((0, \infty)\times \mathbb{R}^{2d})$ with $\phi\geq 0.$
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
We denote the left hand side of \eqref{I>0} by $I(f,g,\phi).$ 
 Since \eqref{DEq} holds in the sense of equality of measures for $f$ and $g$, we have 
\begin{align}\label{new I}
I(f,g,\phi)=\int_0^{\infty}\int_{\mathbb{R}^{2d}}\mathrm{sign}(z)\big\{\partial_t (\phi z) &+p\cdot \nabla_x(\phi z(1-f-g))\nonumber\\ &- \mathrm{div}_p(p\phi z(1-f-g))
-\mathrm{div}_p(\phi \nabla_p z) \big\}dxdpdt,
\end{align}
where  $z\colonequals f-g$.
We use \eqref{nu_p} and \cite[Theorem 1.1]{Khu} to integrate by parts 
\begin{align*}
&I_1(f,g,\phi)\colonequals\int_0^{\infty}\int_{\mathbb{R}^{2d}}\mathrm{sign}(z)\big\{\partial_t (\phi z) +p\cdot \nabla_x(\phi z(1-f-g))- \mathrm{div}_p(p\phi z(1-f-g))\big\}dxdpdt\nonumber \\
&=-\int_{J_{z}}\phi(\mathrm{sign}(z^+)-\mathrm{sign}(z^-))\Big(\frac{z^++z^-}{2}\nu_t+\frac{ z^+(1-f^+-g^+)+ z^-(1-f^--g^-)}{2}p\cdot \nu_x\Big)d\mathcal{H}^{2d}.
\end{align*}
By writing and combining \eqref{+=-} for $f$ and $g$, we get 
\begin{align*}
z^+\nu_t+z^+(1-f^+-g^+)p\cdot \nu_x& =z^-\nu_t+z^-(1-f^--g^-)p\cdot \nu_x\\&=\frac{z^++z^-}{2}\nu_t+\frac{ z^+(1-f^+-g^+)+ z^-(1-f^--g^-)}{2}p\cdot \nu_x.
\end{align*}
We choose $a=g$ in \eqref{+<-} and get 
\begin{align*}
\mathrm{sign}(z^+)(z^+\nu_t+z^+(1-f^+-g^+)p\cdot \nu_x)\leq \mathrm{sign}(z^-)(z^-\nu_t+z^-(1-f^--g^-)p\cdot \nu_x).
\end{align*}
The last two equations imply
\begin{align*}
&(\mathrm{sign}(z^+)-\mathrm{sign}(z^-))\Big(\frac{z^++z^-}{2}\nu_t+\frac{ z^+(1-f^+-g^+)+ z^-(1-f^--g^-)}{2}p\cdot \nu_x\Big)\\
&=\mathrm{sign}(z^+)(z^+\nu_t+z^+(1-f^+-g^+)p\cdot \nu_x)-\mathrm{sign}(z^-)(z^-\nu_t+z^-(1-f^--g^-)p\cdot \nu_x\leq 0
\end{align*}
and so $I_1(f,g,\phi)\geq 0. $ 

The last term of  \eqref{new I} can be written as
$$I_2(f,g,\phi)\colonequals -\int_0^{\infty}\int_{\mathbb{R}^{2d}}\mathrm{sign}(z)  \mathrm{div}_p(\phi \nabla_p z) dxdpdt=-\lim_{\tau\to 0}\int_0^{\infty}\int_{\mathbb{R}^{2d}}\mathrm{sign}_{\tau}(z)  \mathrm{div}_p(\phi \nabla_p z) dxdpdt,$$ 
where $\mathrm{sign}_{\tau}(z) $ is the function defined in \eqref{sign} below. Integration by parts and $\mathrm{sign}_{\tau}'(z)\geq 0$ provide $$-\int_{\mathbb{R}^{2d}}\mathrm{sign}_{\tau}(z)  \mathrm{div}_p(\phi \nabla_p z) dxdpdt=\int_{\mathbb{R}^{2d}}\mathrm{sign}_{\tau}'(z)  \phi |\nabla_p z|^2 dxdpdt\geq 0$$
and so  $I_2(f,g,\phi)\geq 0.$

<!-- proof-2: body_md -->
Equation \eqref{I>0} can written as 
\begin{align*}
\int_0^{\infty}\int_{\mathbb{R}^{2d}}\big\{|f-g|\partial_t \phi &+|f-g|(1-f-g)(p\cdot \nabla_x \phi-p\cdot \nabla_p \phi)\nonumber \\
&-\nabla_p (|f-g|)\cdot\nabla_p \phi \big\}dxdpdt\geq 0.
\end{align*}
This means that the distribution
$$-\partial_t(|f-g|)+(p\cdot \nabla_p-p\cdot\nabla_x)(|f-g|(1-f-g))+\Delta_p(|f-g|)$$
is a nonnegative  measure.  The first two terms are locally finite Radon measures. Hence, the last term is also a locally finite Radon measure.  Let $\phi_R(x,p)$ be a smooth nonnegative function such that $\phi_R(x,p)=1$ if $\sqrt{|x|^2+|p|^2}\leq R$ and $\phi_R(x,p)=0$ if $\sqrt{|x|^2+|p|^2}\geq R+1$.   We use the nonnegativity of this measure, multiply it  by $\phi_R$, and integrate 
over $[0,T]\times \mathbb{R}^{2d}$
\begin{align*}
\int_{\mathbb{R}^{2d}}|f(T)-g(T)|\phi_R dxdp\leq & \int_{\mathbb{R}^{2d}}|f_0-g_0|\phi_R dxdp\\
&+\int_0^{T}\int_{\mathbb{R}^{2d}}|f-g|\Big[(1-f-g)(p\cdot \nabla_x \phi-p\cdot \nabla_p \phi)+\Delta_p \phi\Big]dxdpdt \\
\leq & \int_{\mathbb{R}^{2d}}|f_0-g_0|\phi_R dxdp+C\int_0^{T}\int_{\sqrt{|x|^2+|p|^2}\geq R}|f-g|dxdpdt. 
\end{align*}
The terms $p\cdot \nabla_x \phi-p\cdot \nabla_p \phi$ and $\Delta_p \phi$ are bounded uniformly with respect to $R$ (see \eqref{zeta}-\eqref{gr.z} below for a similar function).
Using  $f,g\in L^1([0,T]\times \mathbb{R}^{2d})$ and  the monotone convergence theorem, we let $R\to \infty$ and obtain 
\begin{align*}
\int_{\mathbb{R}^{2d}}|f(T)-g(T)| dxdp\leq  \int_{\mathbb{R}^{2d}}|f_0-g_0| dxdp. 
\end{align*}

<!-- lemma-2: body_md -->
Let $k\geq 1$.  There is a constant $C_1=C_1(k,d)>0$ such that, for all $u\in  L^k_1$, 
\begin{align}\label{fun1}
||\mathcal{G}[u](t)||_{L^k_1}\leq C_1e^{dt}||u||_{L^k_1},\,\,\,\forall\, t> 0.
\end{align}
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
Using $1+|p|\leq 1+|P|+|e^{-t}p_0|\leq (1+|P|)(1+|e^{-t}p_0|)$ we estimate 
\begin{align}\label{*}
(1+|p|)|\mathcal{G}[u](t,x,p)|\leq \int_{\mathbb{R}^{2d}}&\frac{(1+|P|)}{(4\pi^2 \gamma(t))^{d/2}}\mathrm{exp}\left(-\frac{\Big|\sqrt{\frac{\alpha(t)}{\gamma(t)}}X-\frac{\beta(t)}{\sqrt{\alpha(t)\gamma(t)}}P\Big|^2+\Big|\frac{P}{\sqrt{\alpha(t)}}\Big|^2}{2}\right)\times  \nonumber \\
& \times (1+|e^{-t}p_0|)|u(x_0,p_0)|dx_0dp_0.
\end{align}
Let us denote
$$\xi\colonequals x-(e^t-1)p,\,\eta\colonequals e^t p.$$
Then we have $$X=\xi-x_0+(1-e^{-t})(\eta-p_0),\,P=e^{-t}(\eta-p_0).$$
Hence,  we can consider the right hand side of \eqref{*} 
as a convolution with respect to the variables $\xi$ and $\eta$, i.e., we can write it as $\int_{\mathbb{R}^{2d}} U(t,\xi-x_0, \eta-p_0)V(t, x_0,p_0)dx_0dp_0$ with $$U(t,\xi-x_0, \eta-p_0)\colonequals \frac{(1+|P|)}{(4\pi^2 \gamma(t))^{d/2}}\mathrm{exp}\left(-\frac{\Big|\sqrt{\frac{\alpha(t)}{\gamma(t)}}X-\frac{\beta(t)}{\sqrt{\alpha(t)\gamma(t)}}P\Big|^2+\Big|\frac{P}{\sqrt{\alpha(t)}}\Big|^2}{2}\right)$$ and  $$V(t, x_0,p_0)\colonequals (1+|e^{-t}p_0|)|u(x_0,p_0)|.$$ This allows us to use the Young inequality
\begin{align*}
||\mathcal{G}[u](t)||_{L^k_1}\leq C& \left(\int_{\mathbb{R}^{2d}}(1+|e^{-t}p|)^k|u(x,p)|^kdxdp\right)^{\frac{1}{k}}
\times \\
&\times \int_{\mathbb{R}^{2d}}\frac{(1+|\tilde{P}|)}{(4\pi^2 \gamma(t))^{d/2}}\mathrm{exp}\left(-\frac{\Big|\sqrt{\frac{\alpha(t)}{\gamma(t)}}\tilde{X}-\frac{\beta(t)}{\sqrt{\alpha(t)\gamma(t)}}\tilde{P}\Big|^2+\Big|\frac{\tilde{P}}{\sqrt{\alpha(t)}}\Big|^2}{2}\right)d\xi d\eta,
\end{align*}
where  $\tilde{X}\colonequals \xi+(1-e^{-t})\eta,$  $\tilde{P}\colonequals e^{-t}\eta,$ and  $C$ is a positive constant. By changing the variables $\xi=\tilde{X}-(e^t-1)\tilde{P}$ and $\eta=e^t \tilde{P},$ we compute
\begin{align*}
&\int_{\mathbb{R}^{2d}}\frac{(1+|\tilde{P}|)}{(4\pi^2 \gamma(t))^{d/2}}\mathrm{exp}\left(-\frac{\Big|\sqrt{\frac{\alpha(t)}{\gamma(t)}}\tilde{X}-\frac{\beta(t)}{\sqrt{\alpha(t)\gamma(t)}}\tilde{P}\Big|^2+\Big|\frac{\tilde{P}}{\sqrt{\alpha(t)}}\Big|^2}{2}\right)d\xi d\eta\\
&= e^{dt}\int_{\mathbb{R}^{2d}}\frac{(1+|\tilde{P}|)}{(4\pi^2 \gamma(t))^{d/2}}\mathrm{exp}\left(-\frac{\Big|\sqrt{\frac{\alpha(t)}{\gamma(t)}}\tilde{X}-\frac{\beta(t)}{\sqrt{\alpha(t)\gamma(t)}}\tilde{P}\Big|^2+\Big|\frac{\tilde{P}}{\sqrt{\alpha(t)}}\Big|^2}{2}\right)d\tilde{X} d\tilde{P}.
\end{align*}
We change the variables once more $Z\colonequals \sqrt{\frac{\alpha(t)}{\gamma(t)}}\tilde{X}-\frac{\beta(t)}{\sqrt{\alpha(t)\gamma(t)}}\tilde{P}$ and $Y\colonequals \frac{\tilde{P}}{\sqrt{\alpha(t)}}$ by noting $d\tilde{X}d\tilde{P}=(\gamma(t))^{d/2} dZdY$
\begin{align*}
& e^{dt}\int_{\mathbb{R}^{2d}}\frac{(1+|\tilde{P}|)}{(4\pi^2 \gamma(t))^{d/2}}\mathrm{exp}\left(-\frac{\Big|\sqrt{\frac{\alpha(t)}{\gamma(t)}}\tilde{X}-\frac{\beta(t)}{\sqrt{\alpha(t)\gamma(t)}}\tilde{P}\Big|^2+\Big|\frac{\tilde{P}}{\sqrt{\alpha(t)}}\Big|^2}{2}\right)d\tilde{X} d\tilde{P}\\
&=e^{dt}\int_{\mathbb{R}^{2d}}\frac{(1+|\sqrt{\alpha(t)}{Y}|)}{(4\pi^2)^{d/2}}\mathrm{exp}\left(-\frac{|Z|^2+|Y|^2}{2}\right)dZ dY
.
\end{align*}
The above estimates show
\begin{align*}
||\mathcal{G}[u](t)||_{L^k_1}\leq  Ce^{dt}&\left(\int_{\mathbb{R}^{2d}}(1+|e^{-t}p|)^k|u(x,p)|^kdxdp\right)^{\frac{1}{k}}\times\\
&\times \int_{\mathbb{R}^{2d}}\frac{(1+|\sqrt{\alpha(t)}{Y}|)}{(4\pi^2)^{d/2}}\mathrm{exp}\left(-\frac{|Z|^2+|Y|^2}{2}\right)dZdY.
\end{align*}
Since $1+|e^{-t}p|\leq 1+|p|,$ $1+|\sqrt{\alpha(t)}{Y}|\leq 1+|{Y}|,$ and   $\int_{\mathbb{R}^{2d}}\frac{(1+|\sqrt{\alpha(t)}{Y}|)}{(4\pi^2)^{d/2}}\mathrm{exp}\left(-\frac{|Z|^2+|Y|^2}{2}\right)dZdY$ is finite, we obtain the claimed estimate.

<!-- lemma-3: body_md -->
\label{lem2}
 Let $1\leq q\leq k\leq \infty$ and $T>0$.    There are constants $C_2=C_2(k,q,d, \varepsilon, T)>0$ and $C_3=C_3(k,q,d, \varepsilon, T)>0$ such that 
\begin{align}\label{fun.est11}
||\mathcal{F}[w](t)||_{L^k}\leq \frac{C_2e^{d(1+\frac{1}{k}-\frac{1}{q}-\frac{1}{d})t}}{ t^{d(\frac{1}{q}-\frac{1}{k})+\frac{1}{2}}}||w||_{L^q},\,\,\,\,\forall\,t\in (0,T]
\end{align}
and 
\begin{align}\label{fun.est}
||\mathcal{F}[w](t)||_{L^k_1}\leq \frac{C_3e^{d(1+\frac{1}{k}-\frac{1}{q}-\frac{1}{d})t}}{ t^{d(\frac{1}{q}-\frac{1}{k})+\frac{1}{2}}}||w||_{L^q_1},\,\,\,\,\forall\,t\in (0, T]
\end{align}
hold for all vector functions $w:\mathbb{R}^d\times \mathbb{R}^d\to \mathbb{R}^d$   with $|w|\in L^q_1$.
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
We only prove \eqref{fun.est} and the proof of \eqref{fun.est11} follows similarly. 


 We compute 
 \begin{align*}
 (\nabla_{p_0}-&\nabla_{x_0})G(t,x,p,x_0,p_0)\\
 &=-e^{-t}G(t,x,p,x_0,p_0)\left[\left(\sqrt{\frac{\alpha}{\gamma}}X-\frac{\beta}{\sqrt{\alpha \gamma}}P\right)\left(\sqrt{\frac{\alpha}{\gamma}}+\frac{\beta}{\sqrt{\alpha \gamma}}\right)-\frac{1}{\sqrt{\alpha}}\frac{P}{\sqrt{\alpha}}\right]
 \end{align*}
and so
 \begin{align*}
\mathcal{F}[w](t,x,p)\colonequals -e^{-t} \int_{\mathbb{R}^{2d}}G(t,x,p,x_0,p_0) w(x_0,p_0)\cdot\Big[\Big(\sqrt{\frac{\alpha}{\gamma}}X-\frac{\beta}{\sqrt{\alpha \gamma}}P\Big)&\Big(\sqrt{\frac{\alpha}{\gamma}}+\frac{\beta}{\sqrt{\alpha \gamma}}\Big)\\
&-\frac{1}{\sqrt{\alpha}}\frac{P}{\sqrt{\alpha}}\Big]dx_0dp_0. 
\end{align*}
Using $1+|p|\leq (1+|P|)(1+|e^{-t}p_0|)$ we estimate  
\begin{align}\label{**}
(1&+|p|)\mathcal{F}[w](t,x,p)\leq   e^{-t} \int_{\mathbb{R}^{2d}}(1+|e^{-t}p_0|)|w(x_0,p_0)|\times\nonumber\\
 &\times \left|\left(\sqrt{\frac{\alpha}{\gamma}}X-\frac{\beta}{\sqrt{\alpha \gamma}}P\right)\left(\sqrt{\frac{\alpha}{\gamma}}+\frac{\beta}{\sqrt{\alpha \gamma}}\right)-\frac{1}{\sqrt{\alpha}}\frac{P}{\sqrt{\alpha}}\right| (1+|P|)G(t,x,p,x_0,p_0)dx_0dp_0.
\end{align}
Let us denote
$$\xi\colonequals x-(e^t-1)p,\,\eta\colonequals e^t p.$$
Then we have $$X=\xi-x_0+(1-e^{-t})(\eta-p_0),\,P=e^{-t}(\eta-p_0).$$
Hence,  we can consider
the right hand side of \eqref{**}
as a convolution with respect to the variables $\xi$ and $\eta,$ i.e., we can write it as $\int_{\mathbb{R}^{2d}} U(t,\xi-x_0,\eta-p_0)V(t,x_0,p_0)dx_0dp_0$ with 
$$U(t,\xi-x_0,\eta-p_0)\colonequals \left|\left(\sqrt{\frac{\alpha}{\gamma}}X-\frac{\beta}{\sqrt{\alpha \gamma}}P\right)\left(\sqrt{\frac{\alpha}{\gamma}}+\frac{\beta}{\sqrt{\alpha \gamma}}\right)-\frac{1}{\sqrt{\alpha}}\frac{P}{\sqrt{\alpha}}\right| (1+|P|)G(t,x,p,x_0,p_0)$$
and $$V(t,x_0,p_0)\colonequals (1+|e^{-t}p_0|)|w(x_0,p_0)|.$$
 This allows us to use the Young inequality
\begin{align*}
&||\mathcal{F}[w](t)||_{L^k_1}\leq Ce^{-t} \left(\int_{\mathbb{R}^{2d}}(1+|e^{-t}p|)^q|w|^qdxdp\right)^{\frac{1}{q}}\times\\ 
&\times \left( \int_{\mathbb{R}^{2d}}\left|\left(\sqrt{\frac{\alpha}{\gamma}}\tilde{X}-\frac{\beta}{\sqrt{\alpha \gamma}}\tilde{P}\right)\left(\sqrt{\frac{\alpha}{\gamma}}+\frac{\beta}{\sqrt{\alpha \gamma}}\right)-\frac{1}{\sqrt{\alpha}}\frac{\tilde{P}}{\sqrt{\alpha}}\right|^r (1+|\tilde{P}|)^r \tilde{G}^r(t,\tilde{X},\tilde{P}) d\xi d\eta \right)^{\frac{1}{r}},
\end{align*}
where $1/k+1=1/q+1/r,$ $\tilde{X}\colonequals \xi+(1-e^{-t})\eta,$ $\tilde{P}\colonequals e^{-t}\eta,$ and $$\tilde{G}(t,\tilde{X},\tilde{P})\colonequals \frac{1}{(4\pi^2 \gamma(t))^{d/2}}\mathrm{exp}\left(-\frac{\Big|\sqrt{\frac{\alpha(t)}{\gamma(t)}}\tilde{X}-\frac{\beta(t)}{\sqrt{\alpha(t)\gamma(t)}}\tilde{P}\Big|^2+\Big|\frac{\tilde{P}}{\sqrt{\alpha(t)}}\Big|^2}{2}\right).$$ 
Here, we use $C$ to denote positive constants which can vary from one line to another.
By changing the variables $\xi=\tilde{X}-(e^t-1)\tilde{P}$ and $\eta=e^t \tilde{P},$ we compute
\begin{align*}
&\int_{\mathbb{R}^{2d}}\left|\left(\sqrt{\frac{\alpha}{\gamma}}\tilde{X}-\frac{\beta}{\sqrt{\alpha \gamma}}\tilde{P}\right)\left(\sqrt{\frac{\alpha}{\gamma}}+\frac{\beta}{\sqrt{\alpha \gamma}}\right)-\frac{1}{\sqrt{\alpha}}\frac{\tilde{P}}{\sqrt{\alpha}}\right|^r (1+|\tilde{P}|)^r \tilde{G}^r(t,\tilde{X},\tilde{P}) d\xi d\eta\\&=e^{dt}\int_{\mathbb{R}^{2d}}\left|\left(\sqrt{\frac{\alpha}{\gamma}}\tilde{X}-\frac{\beta}{\sqrt{\alpha \gamma}}\tilde{P}\right)\left(\sqrt{\frac{\alpha}{\gamma}}+\frac{\beta}{\sqrt{\alpha \gamma}}\right)-\frac{1}{\sqrt{\alpha}}\frac{\tilde{P}}{\sqrt{\alpha}}\right|^r (1+|\tilde{P}|)^r \tilde{G}^r(t,\tilde{X},\tilde{P}) d\tilde{X} d\tilde{P}.
\end{align*} 
We change the variables once more $Z\colonequals \sqrt{\frac{\alpha(t)}{\gamma(t)}}\tilde{X}-\frac{\beta(t)}{\sqrt{\alpha(t)\gamma(t)}}\tilde{P}$ and $Y\colonequals \frac{\tilde{P}}{\sqrt{\alpha(t)}}$
\begin{align*}
&e^{dt}\int_{\mathbb{R}^{2d}}\left|\left(\sqrt{\frac{\alpha}{\gamma}}\tilde{X}-\frac{\beta}{\sqrt{\alpha \gamma}}\tilde{P}\right)\left(\sqrt{\frac{\alpha}{\gamma}}+\frac{\beta}{\sqrt{\alpha \gamma}}\right)-\frac{1}{\sqrt{\alpha}}\frac{\tilde{P}}{\sqrt{\alpha}}\right|^r (1+|\tilde{P}|)^r \tilde{G}^r(t,\tilde{X},\tilde{P}) d\tilde{X} d\tilde{P}\\&=\frac{e^{dt}}{(4\pi^2)^{dr/2} (\gamma(t))^{\frac{d(r-1)}{2}}}\int_{\mathbb{R}^{2d}}\left|Z\left(\sqrt{\frac{\alpha}{\gamma}}+\frac{\beta}{\sqrt{\alpha\gamma}}\right)-\frac{1}{\sqrt{\alpha}}Y\right|^r (1+|\sqrt{\alpha}Y|)^r e^{-\frac{r(|Z|^2+|Y|^2)}{2}} dZ dY.
\end{align*} 
We use  $\sqrt{\alpha(t)}\leq 1$ and $\left|Z\left(\sqrt{\frac{\alpha}{\gamma}}+\frac{\beta}{\sqrt{\alpha\gamma}}\right)-\frac{1}{\sqrt{\alpha}}Y\right|^r \leq C\left(\left|\sqrt{\frac{\alpha}{\gamma}}+\frac{\beta}{\sqrt{\alpha\gamma}}\right|^r|Z|^r+\frac{1}{\alpha^{r/2}}|Y|^r \right)$
to estimate

We omit the dependence of the solution $f\colonequals f_{\varepsilon}$ on $\varepsilon$ for notational convenience. We use $C$ to denote positive constants which can vary from one line to another.


   Let $T>0.$ In the following we assume $t\in [0, T].$
We consider the following operator
\begin{align*}
&\Gamma[f](t,x,p)=\int_{\mathbb{R}^{2d}}G(t,x,p,x_0,p_0)f_0(x_0,p_0)dx_0dp_0\nonumber\\
&+\int_0^t\int_{\mathbb{R}^{2d}}f^2(s,x_0,p_0)p_0\cdot(\nabla_{p_0}G(t-s,x,p,x_0,p_0)-\nabla_{x_0}G(t-s,x,p,x_0,p_0))dx_0dp_0ds.
\end{align*}
for functions $f\in \Upsilon_T$. We will  show that there is a fixed point of $\Gamma$ in $\Upsilon_T$.

\textbf{Step 1: Estimates on $\Gamma[f]$ in $\Upsilon_T$:} 
 One can check 
$$\int_{\mathbb{R}^{2d}}G(t,x,p,x_0,p_0)dx_0dp_0=e^{dt}$$
and so $\left|\left|\int_{\mathbb{R}^{2d}}G(t,\cdot,\cdot,x_0,p_0)f_0(x_0,p_0)dx_0dp_0\right|\right|_{L^{\infty}(\mathbb{R}^{2d})}\leq e^{dt}||f_0||_{L^{\infty}(\mathbb{R}^{2d})}.$  This equation and  \eqref{fun.est11} provide 
\begin{align*} 
||\Gamma[f](t)||_{L^{\infty}}&\leq e^{dt}||f_0||_{L^{\infty}}+\int_0^t \frac{Ce^{d(1-\frac{1}{k}-\frac{1}{d})(t-s)}}{ (t-s)^{\frac{d}{k}+\frac{1}{2}}}||\,|p|f^2(s)||_{L^k}ds.
\end{align*}
                                       
Using $|p|^k\leq (1+|p|)^k$ we estimate 
\begin{align*}
||\,|p|f^2(s)||_{L^k}&\leq ||f(s)||_{L^{\infty}}\left(\int_{\mathbb{R}^{2d}} |p|^k|f(s)|^kdxdp\right)^{\frac{1}{k}}\nonumber \\
&\leq ||f(s)||_{L^{\infty}}\left(\int_{\mathbb{R}^{2d}}(1+ |p|)^k|f(s)|^kdxdp\right)^{\frac{1}{k}}\nonumber \\
&=||f(s)||_{L^{\infty}}||f(s)||_{L^k_1}.
\end{align*}
                                                  
 The above two estimates imply
\begin{align}\label{infty} 
||\Gamma[f](t)||_{L^{\infty}}&\leq e^{dt}||f_0||_{L^{\infty}}+\int_0^t \frac{Ce^{d(1-\frac{1}{k}-\frac{1}{d})(t-s)}}{ (t-s)^{\frac{d}{k}+\frac{1}{2}}}||f(s)||_{L^{\infty}}||f(s)||_{L^k_1} ds\nonumber\\
& \leq e^{dt}||f_0||_{L^{\infty}}+\int_0^t \frac{Ce^{d(1-\frac{1}{k}-\frac{1}{d})(t-s)}}{ (t-s)^{\frac{d}{k}+\frac{1}{2}}}||f(s)||^2_{\Upsilon}ds\nonumber \\
&\leq e^{dt}||f_0||_{L^{\infty}}+I_1(t) ||f||^2_{\Upsilon_T},
\end{align}
where $$I_1(t)\colonequals \int_0^t \frac{Ce^{d(1-\frac{1}{k}-\frac{1}{d})(t-s)}}{ (t-s)^{\frac{d}{k}+\frac{1}{2}}} ds$$
is finite because of $k>2d.$

We estimate the $L^k_1$-norm  using \eqref{fun1} and \eqref{fun.est} 
$$
||\Gamma[f](t)||_{L^{k}_1}\leq C e^{dt}||f_0||_{L^{k}_1}+\int_0^t \frac{Ce^{d(1-\frac{1}{k}-\frac{1}{d})(t-s)}}{ (t-s)^{\frac{d}{k}+\frac{1}{2}}}||\,|p|f^2(s)||_{L^{k/2}_1}ds.$$
 We estimate 
 \begin{align*}
 ||\,|p|f^2(s)||_{L^{k/2}_1}=\left(\int_{\mathbb{R}^{2d}} (1+|p|)^{k/2}|p|^{k/2}|f(s)|^kdxdp \right)^{\frac{2}{k}}\leq ||f(s)||_{L_1^k}^2.
 \end{align*}
 The above two estimates imply
\begin{align} \label{km}
||\Gamma[f](t)||_{L^{k}_1}&\leq C e^{dt}||f_0||_{L^{k}_1}+\int_0^t \frac{Ce^{d(1-\frac{1}{k}-\frac{1}{d})(t-s)}}{ (t-s)^{\frac{d}{k}+\frac{1}{2}}}||f(s)||_{L^k_1}^{2}ds \nonumber \\
&\leq  C e^{dt}||f_0||_{L^{k}_1}+\int_0^t \frac{Ce^{d(1-\frac{1}{k}-\frac{1}{d})(t-s)}}{ (t-s)^{\frac{d}{k}+\frac{1}{2}}} ||f(s)||_{\Upsilon}^2ds\nonumber \\
&\leq  C e^{dt}||f_0||_{L^{k}_1}+I_2(t) ||f||_{\Upsilon_T}^2,
\end{align}
where   $$I_2(t)\colonequals \int_0^t \frac{Ce^{d(1-\frac{1}{k}-\frac{1}{d})(t-s)}}{ (t-s)^{\frac{d}{k}+\frac{1}{2}}}ds$$
is finite because of $k>2d.$



We estimate the $L^1_1$-norm using  \eqref{fun.est}
\begin{align*} 
||\Gamma[f](t)||_{L^{1}_1}&\leq C e^{dt}||f_0||_{L^{1}_1}+\int_0^t \frac{Ce^{(d-1)(t-s)}}{ (t-s)^{\frac{1}{2}}}||\,|p|f^2(s)||_{L^1_1}ds.
\end{align*}
Using $k>2d\geq 2$ we estimate by interpolation
\begin{align*}
||\,|p|f^2(s)||_{L^1_1}&=\int_{\mathbb{R}^{2d}}(1+|p|)|p|f^2(s,x,p)dxdp
\\
&\leq \int_{\mathbb{R}^{2d}}(1+|p|)^2 f^2(s,x,p)dxdp\\
&\leq \left(\int_{\mathbb{R}^{2d}}(1+|p|)|f(s,x,p)|dxdp\right)^{\frac{k-2}{k-1}}\left(\int_{\mathbb{R}^{2d}}(1+|p|)^k|f(s,x,p)|^kdxdp\right)^{\frac{1}{k-1}}\\
&\leq C||f(s)||^{\frac{k-2}{k-1}}_{L^1_1}||f(s)||_{L^k_1}^{\frac{k}{k-1}}.
\end{align*}
Consequently, we get 
\begin{align}\label{11} 
||\Gamma[f](t)||_{L^{1}_1}&\leq C e^{dt}||f_0||_{L^{1}_1}+\int_0^t \frac{Ce^{(d-1)(t-s)}}{ (t-s)^{\frac{1}{2}}}||f(s)||^{\frac{k-2}{k-1}}_{L^1_1}||f(s)||_{L^k_1}^{\frac{k}{k-1}}ds\nonumber \\
&\leq C e^{dt}||f_0||_{L^{1}_1}+\int_0^t \frac{Ce^{(d-1)(t-s)}}{ (t-s)^{\frac{1}{2}}}||f(s)||_{\Upsilon}^2ds\nonumber \\

\end{align}
where $I_3(t)\colonequals \int_0^t \frac{Ce^{(d-1)(t-s)}}{ (t-s)^{\frac{1}{2}}}$
is finite.


The above estimates in \eqref{infty}, \eqref{km}, and \eqref{11} show that there are positive constants $C_1(T)\geq 1$ and $C_2(T)>0$ with $C_2(T)\to 0$ as $T\to 0$ such that 
$$||\Gamma[f]||_{\Upsilon_T}\leq C_1(T)||f_0||_{\Upsilon}+C_2(T)||f||^2_{\Upsilon_T}.$$
This shows that $\Gamma$ maps $\Upsilon_T$ to itself. 


\textbf{Step 2: Fixed point in $\Upsilon_T$:}  Let $f,g\in \Upsilon_T$.
We compute
\begin{align*}
&||\Gamma[f]-\Gamma[g]||_{\Upsilon_T}\\&=\left|\left|\int_0^t\int_{\mathbb{R}^{2d}}(f(s)+g(s))(f(s)-g(s))p_0\cdot(\nabla_{p_0}G(t-s)-\nabla_{x_0}G(t-s))dx_0dp_0ds\right|\right|_{\Upsilon_T}\\

\end{align*}
We can use \eqref{fun.est} and repeat the same arguments in  \eqref{infty}, \eqref{km}, and \eqref{11} to get
\begin{align*}
||\Gamma[f]-\Gamma[g]||_{\Upsilon_T}\leq C_2(T) (||f||_{\Upsilon_T}+||g||_{\Upsilon_T})\left|\left|f-g\right|\right|_{\Upsilon_T}. 
\end{align*}
We choose a small $T>0$ such that
 $$C_2(T)< \frac{1}{4C_1(T)||f_0||_{\Upsilon}}.$$   
 We denote $R\colonequals 2C_1(T)||f_0||_{\Upsilon}$ and $\delta\colonequals 2RC_2(T)\in (0,1)$.   We then have $$||\Gamma[f]||_{\Upsilon_T}<R\,\,\,\,\text{and}\,\,\,\,\,||\Gamma[f]-\Gamma[g]||_{\Upsilon_T}\leq \delta ||f-g||_{\Upsilon_T}$$ for all $f,g\in \Upsilon_T$ with $||f||_{\Upsilon_T}\leq R$ and  $||g||_{\Upsilon_T}\leq R$.  This provides  that there is a unique fixed point $f$ of $\Gamma$ in $\Upsilon_T$, and hence a unique solution to the integral equation in \eqref{Duh} for $t\in [0,T]$. 
 
\textbf{Step 3: Extension to $[0,T_{\text{max}})$:}  We note that $T$ depends on $|| f_0||_{\Upsilon}$. From what we have just proved it follows that if $f$ is a  solution on the interval $[0,T]$ it can be extended on the interval $[0, T+T_1]$ with some $T_1>0$ by defining on $[T, T+T_1],$ $f(t)=f_1(t)$ where $f_1(t)$ is the solution of \begin{align*}

&+\int_T^t\int_{\mathbb{R}^{2d}}f_1^2(s,x_0,p_0)p_0\cdot(\nabla_{p_0}G(t-s,x,p,x_0,p_0)-\nabla_{x_0}G(t-s,x,p,x_0,p_0))dx_0dp_0ds.
\end{align*}
Moreover, $T_1 $ depends on $|| f(T)||_{\Upsilon}.$ We can continue this process and extend the existence time interval.  
Let $[0, T_{\mathrm{max}})$ be the maximal interval of  existence of the solution. If $T_{\mathrm{max}}<\infty$ then at least one of the limits
$$\lim_{t\nearrow T_{\mathrm{max}}} ||f(t)||_{L^{\infty}},\,\,\,\,\lim_{t\nearrow T_{\mathrm{max}}} ||f(t)||_{L^{1}_1}, \,\,\,\text{and}\,\,\,\lim_{t\nearrow T_{\mathrm{max}}} ||f(t)||_{L^{k}_1}$$
is infinite. Otherwise there is a sequence $T_n\nearrow T_{\mathrm{max}},$ $n\in \mathbb{N},$ such that $||f(t_n)||_{ \Upsilon}$  is bounded. This would imply what we have just proved that for each $T_n,$ near enough to $T_{\mathrm{max}},$ the solution $f$ on the interval $[0,T_n]$ can be extended to the interval $[0, T_n+\theta],$ where $\theta >0$ is independent of $T_n$ and  hence $f$ can be extended beyond $T_{\mathrm{max}}. $ This contradicts the definition of $T_{\mathrm{max}}.$

\begin{align*}
 \left|\left|\int_0^t\int_{\mathbb{R}^{2d}}2f(s,x_0,p_0)\nabla_{x_0}f(s,x_0,p_0)p_0\cdot(\nabla_{p_0}-\nabla_{x_0})G(t-s,x,p,x_0,p_0)dx_0dp_0ds \right|\right|_{L^1_1}\nonumber \\
 \leq \int_0^t\frac{Ce^{(d-1)(t-s)}}{ (t-s)^{\frac{1}{2}}}||\,|p|f(s)|\nabla_x f(s)|\,||_{L^{1}_1}.
\end{align*} 
This leads to estimate
 \begin{align*}
 ||\, |p|f(s)|\nabla_x f(s)|\,||_{L^1_1}&\leq C||f(s)||_{L^1_1}^{\frac{k-2}{2(k-1)}}||f(s)||_{L^k_1}^{\frac{k}{2(k-1)}}||\nabla_x f(s)||_{L^1_1}^{\frac{k-2}{2(k-1)}}||\nabla_xf(s)||_{L_1^k}^{\frac{k}{2(k-1)}}\\
 &\leq ||f||_{\Upsilon_T}\mathcal{X}(s),
 \end{align*}
where we used the (extended) H\"older inequality \cite[Section 4.2, Remark 2]{Bre}. The above estimate and \eqref{x11} provide
\begin{align}\label{xL^1_1}
 \left|\left|\int_0^t\int_{\mathbb{R}^{2d}}2f(s,x_0,p_0)\nabla_{x_0}f(s,x_0,p_0)p_0\cdot(\nabla_{p_0}-\nabla_{x_0})G(t-s,x,p,x_0,p_0)dx_0dp_0ds \right|\right|_{L^1_1}\nonumber \\
 \leq ||f||_{\Upsilon_T} \int_0^t\frac{Ce^{(d-1)(t-s)}}{ (t-s)^{\frac{1}{2}}}\mathcal{X}(s)ds.
\end{align} 
Equations \eqref{x-derInt}, \eqref{x1} with $k=1$, and the above equation provide 
\begin{align*}
||\nabla_x f(t)||_{L^1_1}&\leq \frac{Ce^{(d-1)t}}{ t^{\frac{1}{2}}}||f_0||_{L^1_1}+||f||_{\Upsilon_T}\int_0^t\frac{Ce^{(d-1)(t-s)}}{ (t-s)^{\frac{1}{2}}}  \mathcal{X}(s)ds\\
 & \leq \frac{Ce^{(d-1)t}}{ t^{\frac{1}{2}}}||f_0||_{L^1_1}+||f||_{\Upsilon_T}\int_0^t\frac{Ce^{(d-1)(T-s)}}{ (t-s)^{\frac{1}{2}}}  \mathcal{X}(s)ds.
\end{align*}
This estimate and \eqref{x33} imply
\begin{align*}
\mathcal{X}(t)\leq \frac{Ce^{(d-1)t}}{ t^{\frac{1}{2}}}\max\{||f_0||_{L^1_1}, ||f_0||_{L^k_1}\}+||f||_{\Upsilon_T}\int_0^t\frac{Ce^{(d-1)(T-s)}}{\min\{ (t-s)^{\frac{1}{2}}, \,(t-s)^{\frac{d}{k}+\frac{1}{2}}\}}  \mathcal{X}(s)ds.
\end{align*}
We apply \cite[Lemma 2.5]{Lin} and obtain
\begin{align*}
& \mathcal{X}(t)\leq \frac{Ce^{(d-1)t}}{ t^{\frac{1}{2}}}\max\{||f_0||_{L^1_1}, ||f_0||_{L^k_1}\}+||f||_{\Upsilon_T} \int_0^t\frac{Ce^{(d-1)T} \mathcal{X}(0)}{ s^{\frac{1}{2}} \min\{ (t-s)^{\frac{1}{2}}, \,(t-s)^{\frac{d}{k}+\frac{1}{2}}\}}  ds.
\end{align*}
 \cite[Lemma 3.13]{Her} shows that $$\int_0^t\frac{1}{ s^{\frac{1}{2}} \min\{ (t-s)^{\frac{1}{2}}, \,(t-s)^{\frac{d}{k}+\frac{1}{2}}\}}  ds\leq \frac{C}{t^{\frac{d}{k}}},\,\,\,\, \forall\, t\in (0,T].$$    
The above inequalities show
\begin{align}\label{x3}
 t^{\frac{1}{2}}\mathcal{X}(t)\leq Ce^{(d-1)t}\max\{||f_0||_{L^1_1}, ||f_0||_{L^k_1}\}+||f||_{\Upsilon_T} Ce^{(d-1)T} \mathcal{X}(0) t^{\frac{1}{2}-\frac{d}{k}}.
\end{align}
Because of   $\frac{1}{2}-\frac{d}{k}>0$ we conclude that $t^{\frac{1}{2}}\mathcal{X}(t)$ is bounded for all $t\in [0,T].$
\\[2mm]
{\textbf{Part 2: Estimates on $\nabla_p f$:}}
Next we take the gradient with respect to $p$ in the integral equation using \eqref{p_i-der} 
\begin{align}\label{p0}
& \nabla_p f(t,x,p)=\nabla_p \int_{\mathbb{R}^{2d}}G(t,x,p,x_0,p_0)f_0(x_0,p_0)dx_0dp_0\nonumber\\
&+2(1-e^t)\int_0^t\int_{\mathbb{R}^{2d}}f(s,x_0,p_0)\nabla_{x_0}f(s,x_0,p_0)p_0\cdot(\nabla_{p_0}-\nabla_{x_0})G(t-s,x,p,x_0,p_0)dx_0dp_0ds\nonumber\\
&+2e^t\int_0^t\int_{\mathbb{R}^{2d}}f(s,x_0,p_0)\nabla_{p_0}f(s,x_0,p_0)p_0\cdot(\nabla_{p_0}-\nabla_{x_0}) G(t-s,x,p,x_0,p_0)dx_0dp_0ds
\nonumber \\
&+de^t\int_0^t\int_{\mathbb{R}^{2d}} f^2(s,x_0,p_0)(\nabla_{p_0}-\nabla_{x_0})G(t-s,x,p,x_0,p_0) dx_0dp_0ds.
\end{align}
Let us denote 
$$\mathcal{Y}(t)\colonequals \max\{||\nabla_pf(t)||_{L^k_1}, ||\nabla_p f(t)||_{L^1_1}\}.$$
We estimate the $L^k_1$-norm of $|\nabla_pf(t)|$:
The first term on the right hand side of \eqref{p0} can be estimated using \eqref{p-der}
\begin{equation}\label{p1}
\left|\left| \nabla_p \int_{\mathbb{R}^{2d}} G(t,x,p,x_0,p_0)f_0(x_0,p_0)dx_0dp_0\right|\right|_{L^k_1}\leq \frac{Ce^{d t}}{ t^{\frac{1}{2}}}||f_0||_{L^k_1}.
\end{equation}
Equations \eqref{x2}, the fact that  $s^{\frac{1}{2}}||\nabla_x f(s)||_{L^k_1}$  is bounded by a constant because of \eqref{x3}, and \cite[Lemma 3.13]{Her} provide 
\begin{align}\label{p22}
2(e^t-1)& \left|\left|\int_0^t\int_{\mathbb{R}^{2d}}f(s,x_0,p_0)\nabla_{x_0}f(s,x_0,p_0)p_0\cdot(\nabla_{p_0}-\nabla_{x_0})G(t-s,x,p,x_0,p_0)dx_0dp_0ds \right|\right|_{L^k_1}\nonumber \\
 &\leq ||f||_{\Upsilon_T}(e^t-1) \int_0^t\frac{Ce^{d(1-\frac{1}{k}-\frac{1}{d})(t-s)}}{ (t-s)^{\frac{d}{k}+\frac{1}{2}}}  ||\nabla_x f(s)||_{L^k_1}ds\nonumber\\
&\leq ||f||_{\Upsilon_T}(e^t-1)\int_0^t\frac{Ce^{d(1-\frac{1}{k}-\frac{1}{d})(T-s)}}{ s^{\frac{1}{2}}(t-s)^{\frac{d}{k}+\frac{1}{2}}}  ds\nonumber \\
&\leq C||f||_{\Upsilon_T}(e^t-1)t^{-\frac{d}{k}}.
\end{align}
The third terms on the right hand side of \eqref{p0} can be estimated similarly to \eqref{x2}
\begin{align}\label{p2}
\left|\left|2e^t\int_0^t\int_{\mathbb{R}^{2d}}f(s,x_0,p_0)\nabla_{p_0}f(s,x_0,p_0)p_0\cdot(\nabla_{p_0}-\nabla_{x_0})G(t-s,x,p,x_0,p_0)dx_0dp_0ds \right|\right|_{L^k_1}\nonumber \\
 \leq ||f||_{\Upsilon_T}e^t\int_0^t\frac{Ce^{d(1-\frac{1}{k}-\frac{1}{d})(t-s)}}{ (t-s)^{\frac{d}{k}+\frac{1}{2}}}  ||\nabla_p f(s)||_{L^k_1}ds.
\end{align}
Using \eqref{fun.est} we estimate the last term on the right hand side of this equation
\begin{align*}
&\left|\left|de^t\int_0^t\int_{\mathbb{R}^{2d}} f^2(s,x_0,p_0)(\nabla_{p_0}-\nabla_{x_0})G(t-s,x,p,x_0,p_0) dx_0dp_0ds\right|\right|_{L^k_1}\\
&\leq  e^t\int_0^t\frac{C e^{(d-1)(t-s)}}{ (t-s)^{\frac{1}{2}}}||f^2(s)||_{L^k_1}ds.
\end{align*}
We can estimate $$||f^2(s)||_{L^k_1}\leq ||f(s)||_{L^{\infty}}||f(s)||_{L^k_1}\leq ||f||_{\Upsilon_T}^2 $$
and $$\int_0^t \frac{ e^{(d-1)(t-s)}}{ (t-s)^{\frac{1}{2}}}ds\leq e^{(d-1)t} \int_0^t\frac{1}{ (t-s)^{\frac{1}{2}}}ds=2e^{(d-1)t}t^{\frac{1}{2}}.$$
These estimates provide 
\begin{align}\label{p3}
\left|\left|de^t\int_0^t\int_{\mathbb{R}^{2d}} f^2(s,x_0,p_0)(\nabla_{p_0}-\nabla_{x_0})G(t-s,x,p,x_0,p_0) dx_0dp_0ds\right|\right|_{L^k_1}\leq  C||f||^2_{\Upsilon_T}e^{dt}t^{\frac{1}{2}}.
\end{align}
Equations \eqref{p0}-\eqref{p3} show 
\begin{align}\label{pL^k_m}
||\nabla_p f(t)||_{L^k_1}\leq & \frac{Ce^{d t}}{t^{\frac{1}{2}}}||f_0||_{L^k_m}+C||f||_{\Upsilon_T}(e^t-1)t^{-\frac{d}{k}}+C||f||^2_{\Upsilon_T}e^{dt}t^{\frac{1}{2}}\nonumber \\
& +||f||_{\Upsilon_T}e^T\int_0^t\frac{Ce^{d(1-\frac{1}{k}-\frac{1}{d})(T-s)}}{ (t-s)^{\frac{d}{k}+\frac{1}{2}}}  ||\nabla_p f(s)||_{L^k_1}ds.
\end{align}


Next we estimate the $L^1_1$-norm of $|\nabla_p f|:$ Equation \eqref{p1} holds with $k=1.$
Equation \eqref{xL^1_1} and the fact that $s^{\frac{1}{2}}\mathcal{X}(s)$ is bounded because of \eqref{x3}, and \cite[Lemma 3.13]{Her}  imply 
\begin{align*}
2(e^t-1)& \left|\left|\int_0^t\int_{\mathbb{R}^{2d}}f(s,x_0,p_0)\nabla_{x_0}f(s,x_0,p_0)p_0\cdot(\nabla_{p_0}-\nabla_{x_0})G(t-s,x,p,x_0,p_0)dx_0dp_0ds \right|\right|_{L^1_1}\nonumber \\
 &\leq ||f||_{\Upsilon_T}(e^t-1) \int_0^t\frac{Ce^{(d-1)(t-s)}}{ s^{\frac{1}{2}}(t-s)^{\frac{1}{2}}} ds\nonumber\\
&\leq C ||f||_{\Upsilon_T}(e^t-1).
\end{align*}
The third terms on the right hand side of \eqref{p0} can be estimated similarly to \eqref{xL^1_1}
\begin{align*}
\left|\left|2e^t\int_0^t\int_{\mathbb{R}^{2d}}f(s,x_0,p_0)\nabla_{p_0}f(s,x_0,p_0)p_0\cdot(\nabla_{p_0}-\nabla_{x_0})G(t-s,x,p,x_0,p_0)dx_0dp_0ds \right|\right|_{L^1_1}\nonumber \\
 \leq ||f||_{\Upsilon_T}e^t\int_0^t\frac{Ce^{(d-1)(t-s)}}{ (t-s)^{\frac{1}{2}}}  \mathcal{Y}(s)ds.
\end{align*}
Equation \eqref{p3} holds for $k=1$
\begin{align*}
\left|\left|de^t\int_0^t\int_{\mathbb{R}^{2d}} f^2(s,x_0,p_0)(\nabla_{p_0}-\nabla_{x_0})G(t-s,x,p,x_0,p_0) dx_0dp_0ds\right|\right|_{L^1_1}\leq  C||f||^2_{\Upsilon_T}e^{dt}t^{\frac{1}{2}}.
\end{align*}
Equation \eqref{p0} and the above estimates provide
\begin{align*}
||\nabla_p f(t)||_{L^1_1}\leq & \frac{Ce^{d t}}{t^{\frac{1}{2}}}||f_0||_{L^1_1}+C||f||_{\Upsilon_T}(e^t-1)+C||f||^2_{\Upsilon_T}e^{dt}t^{\frac{1}{2}}\\
& +||f||_{\Upsilon_T}e^T\int_0^t\frac{Ce^{(d-1)(T-s)}}{ (t-s)^{\frac{1}{2}}}  \mathcal{Y}(s)ds.
\end{align*}
This estimate and \eqref{pL^k_m} yield 
\begin{align*}
\mathcal{Y}(t)\leq & \frac{Ce^{d t}}{t^{\frac{1}{2}}}||f_0||_{\Upsilon}+C||f||_{\Upsilon_T}(e^t-1)t^{-\frac{d}{k}}+C||f||^2_{\Upsilon_T}e^{dt}t^{\frac{1}{2}}\\
& +||f||_{\Upsilon_T}e^T\int_0^t\frac{Ce^{(d-1)(T-s)}}{ (t-s)^{\frac{1}{2}}}  \mathcal{Y}(s)ds.
\end{align*}
We apply \cite[Lemma 2.5]{Lin} to obtain
\begin{align*}
&\mathcal{Y}(t)\leq  \frac{Ce^{d t}}{t^{\frac{1}{2}}}||f_0||_{\Upsilon}+C||f||_{\Upsilon_T}(e^t-1)t^{-\frac{d}{k}}+C||f||^2_{\Upsilon_T}e^{dt}t^{\frac{1}{2}}\\
& +||f||_{\Upsilon_T}e^T\int_0^t\frac{Ce^{(d-1)(T-s)}}{ (t-s)^{\frac{d}{k}+\frac{1}{2}}}  \left(\frac{e^{d s}}{s^{\frac{1}{2}}}||f_0||_{L^k_1}+C||f||_{\Upsilon_T}(e^s-1)s^{-\frac{d}{k}}+C||f||^2_{\Upsilon_T}e^{ds}s^{\frac{1}{2}}\right)ds.
\end{align*}
Using \cite[Lemma 3.13]{Her}  one can show that  
$$t^{\frac{1}{2}}\int_0^t\frac{Ce^{(d-1)(T-s)}}{ (t-s)^{\frac{d}{k}+\frac{1}{2}}}  \left(\frac{e^{d s}}{s^{\frac{1}{2}}}||f_0||_{L^k_1}+C||f||_{\Upsilon_T}(e^s-1)s^{-\frac{d}{k}}+C||f||^2_{\Upsilon_T}e^{ds}s^{\frac{1}{2}}\right)ds$$
is bounded by a positive constant. Therefore, we obtain that $t^{\frac{1}{2}}\mathcal{Y}(t)$ is bounded for all $t\in [0,T].$



\textbf{Part 3: Smoothness:} Equation \ref{Eq} is uniformly parabolic (with $\varepsilon\Delta_x$) with smooth coefficients, the solution $f(t,x,p)$ is in $L^{\infty}([0,T]\times\mathbb{R}^{2d} )$ for every $0<T<T_{\mathrm{max}}$, and  the nonlinearity is smooth and bounded (since $f$ is bounded).  Thus, the solution $f(t,x,p)$ is smooth for all $t>0$ and $(x,p)\in \mathbb{R}^{2d}$ by the regularity results from \cite[Chapter 5, Sections 8 and 10]{Lad} for  quasilinear parabolic equations.

We omit the dependence of the solution $f\colonequals f_{\varepsilon}$ on $\varepsilon$ for notational convenience. By Theorem \ref{th:reg} the solution $f(t,x,p)$ is smooth and satisfies \eqref{Eq} in the classical sense  for all $(x,p)\in \mathbb{R}^{2d}$ and $t\in (0, T_{\mathrm{max}})$.
We multiply \eqref{Eq} by $\zeta_{\sigma}$ and integrate by parts
\begin{align*}
\frac{d}{dt}\int_{\mathbb{R}^{2d}}\zeta_{\sigma}fdpdx=&\int_{\mathbb{R}^{2d}} (\Delta_p \zeta_{\sigma} f-\nabla_p \zeta_{\sigma}\cdot p(f- f^2))dpdx\\

\end{align*}

\begin{align*}
\int_{\mathbb{R}^{2d}}\zeta_{\sigma}f(T)dpdx-\int_{\mathbb{R}^{2d}}\zeta_{\sigma}f_0dpdx&=\int_0^T\int_{\mathbb{R}^{2d}} (\Delta_p \zeta_{\sigma} f-\nabla_p \zeta_{\sigma}\cdot p(f- f^2))dpdxdt\\
&+\int_0^T\int_{\mathbb{R}^{2d}} (\varepsilon \Delta_x \zeta_{\sigma} f+\nabla_x \zeta_{\sigma}\cdot p(f- f^2))dpdxdt.
\end{align*}
 The fact that  $\zeta_{\sigma}\to  1$ pointwise as $\sigma\to 0$  and  the Lebesgue dominated convergence theorem  show
 $$\lim_{\sigma\to 0}\big(\int_{\mathbb{R}^{2d}}\zeta_{\sigma}f(T)dpdx-\int_{\mathbb{R}^{2d}}\zeta_{\sigma}f(0)dpdx\big)=\int_{\mathbb{R}^{2d}}f(T)dpdx-\int_{\mathbb{R}^{2d}}f_0dpdx.$$ 
{The terms $\Delta_p \zeta_{\sigma} f-\nabla_p \zeta_{\sigma}\cdot p(f- f^2)$ and $\varepsilon \Delta_x \zeta_{\sigma} f+\nabla_x \zeta_{\sigma}\cdot p(f- f^2)$  converge  pointwise to zero  as $\sigma \to 0.$} Moreover, $\Delta_{x,p} \zeta_{\sigma} $,  $\nabla_x \zeta_{\sigma}\cdot p$, and $\nabla_p \zeta_{\sigma}\cdot p$ are bounded; see \eqref{gr.z}. Hence, we can apply the Lebesgue dominated convergence theorem 
 $$\lim_{\sigma\to 0}\int_0^T\int_{\mathbb{R}^{2d}} (\Delta_p \zeta_{\sigma} f-\nabla_p \zeta_{\sigma}\cdot p(f- f^2))dpdxdt=0$$
 and 
 $$\lim_{\sigma\to 0}\int_0^T\int_{\mathbb{R}^{2d}} (\varepsilon \Delta_x \zeta_{\sigma} f+\nabla_x \zeta_{\sigma}\cdot p(f- f^2))dpdxdt=0.$$
 These limits show
 $$\int_{\mathbb{R}^{2d}}f(T)dpdx=\int_{\mathbb{R}^{2d}}f_0dpdx$$
for all $T\in (0, T_{\mathrm{max}}).$

\label{lem:L1}
Let $k>2d$,   $\Upsilon \colonequals L^{\infty}(\mathbb{R}^{2d})\cap L^{1}_1(\mathbb{R}^{2d})\cap L^{k}_1(\mathbb{R}^{2d})$ with the norm in \eqref{norm}, and $f_0, g_0\in \Upsilon$.  Assume  $f_{\varepsilon},g_{\varepsilon}\in C([0,T_{\mathrm{max}}); \Upsilon)$ are solutions to \eqref{Eq} with  the initial conditions $f_0$ and $g_0$, respectively. Then 
\begin{equation*}
||f_{\varepsilon}(t)-g_{\varepsilon}(t)||_{L^{1}(\mathbb{R}^{2d})}\leq ||f_0-g_0||_{L^{1}(\mathbb{R}^{2d})}
\end{equation*}
for all $t\in [0, T_{\mathrm{max}}).$ In particular, if $f_0(x,p)\leq g_0(x,p)$ for almost every $(x,p)\in \mathbb{R}^{2d}, $ then $f_{\varepsilon}(t,x,p)\leq g_{\varepsilon}(t,x,p)$ for all $t\in [0,T_{\mathrm{max}})$ and for almost every $(x,p)\in \mathbb{R}^{2d}.$

We omit the dependence of the solutions $f\colonequals f_{\varepsilon}$  and $g\colonequals g_{\varepsilon}$ on $\varepsilon$ for notational convenience.

We compute   
\begin{equation*}
\partial_t(f-g)+p\cdot \nabla_x(f-g-(f^2-g^2))=\varepsilon \Delta_x(f-g)+\Delta_p(f-g)+\text{div}_p(p(f-g) -p (f^2-g^2)). 
\end{equation*}
 By Theorem \ref{th:reg} the solutions $f(t,x,p)$ and $g(t,x,p)$ are smooth and the above equation holds  for all $(x,p)\in \mathbb{R}^{2d}$ and $t\in (0, T_{\mathrm{max}})$. 
 Let $\zeta_{\sigma}$ be the function defined in \eqref{zeta}. We multiply this equation by $\zeta_{\sigma} \text{sign}_{\tau}(f-g)$ and integrate by parts
 \begin{align*}
\frac{d}{dt}& \int_{\mathbb{R}^{2d}}\zeta_{\sigma}|f-g|_{\tau} dpdx\\=&- \int_{\mathbb{R}^{2d}}\zeta_{\sigma}  \text{sign}_{\tau}'(f-g)|\nabla_p(f-g)|^2dpdx- \varepsilon\int_{\mathbb{R}^{2d}}\zeta_{\sigma}  \text{sign}_{\tau}'(f-g)|\nabla_x(f-g)|^2dpdx\\
&- \int_{\mathbb{R}^{2d}}\text{sign}_{\tau}(f-g) \nabla_x\zeta_{\sigma}\cdot (\varepsilon\nabla_x f-pf+pf^2 -\varepsilon\nabla_x  g+pg-pg^2)dpdx\\
&- \int_{\mathbb{R}^{2d}}\text{sign}_{\tau}(f-g) \nabla_p\zeta_{\sigma}\cdot (\nabla_p f+p(f- f^2)- \nabla_p g-p(g- g^2))dpdx\\
&- \int_{\mathbb{R}^{2d}}\zeta_{\sigma} \text{sign}_{\tau}'(f-g)p\cdot\nabla_p(f-g)(f-g)(1- f- g)dpdx\\
&+ \int_{\mathbb{R}^{2d}}\zeta_{\sigma} \text{sign}_{\tau}'(f-g)p\cdot\nabla_x(f-g)(f-g)(1- f- g)dpdx.
 \end{align*}
  We integrate this equation on the interval $[0,T],\, 0<T<T_{\mathrm{max}},$ and  use
 $\zeta_{\sigma}  \text{sign}_{\tau}'(f-g)|\nabla_p(f-g)|^2+\zeta_{\sigma}  \text{sign}_{\tau}'(f-g)|\nabla_x(f-g)|^2 \geq 0$ to estimate
  \begin{align*}
 \int_{\mathbb{R}^{2d}}&\zeta_{\sigma}|f(T)-g(T)|_{\tau} dpdx- \int_{\mathbb{R}^{2d}}\zeta_{\sigma}|f(0)-g(0)|_{\tau} dpdx\\ \leq &-\int_0^T  \int_{\mathbb{R}^{2d}}\text{sign}_{\tau}(f-g) \nabla_x\zeta_{\sigma}\cdot (\varepsilon\nabla_x f-pf+pf^2 -\varepsilon\nabla_x  g+pg-pg^2)dpdxds\\
&-\int_0^T \int_{\mathbb{R}^{2d}}\text{sign}_{\tau}(f-g) \nabla_p\zeta_{\sigma}\cdot (\nabla_p f+p(f- f^2)- \nabla_p g-p(g- g^2))dpdxds\\
&-\int_0^T \int_{\mathbb{R}^{2d}}\zeta_{\sigma} \text{sign}_{\tau}'(f-g)p\cdot\nabla_p(f-g)(f-g)(1- f- g)dpdxds\\
&+\int_0^T \int_{\mathbb{R}^{2d}}\zeta_{\sigma} \text{sign}_{\tau}'(f-g)p\cdot\nabla_x(f-g)(f-g)(1- f- g)dpdxds.
 \end{align*}
 Using the identities
 $$\text{sign}_{\tau}'(f-g)\nabla_x(f-g)(f-g)=\nabla_x((f-g)\text{sign}_{\tau}(f-g)-|f-g|_{\tau}),$$
 $$\text{sign}_{\tau}'(f-g)\nabla_p(f-g)(f-g)=\nabla_p((f-g)\text{sign}_{\tau}(f-g)-|f-g|_{\tau}),$$
 we integrate by parts
 \begin{align*}
 &\int_{\mathbb{R}^{2d}}\zeta_{\sigma}|f(T)-g(T)|_{\tau} dpdx- \int_{\mathbb{R}^{2d}}\zeta_{\sigma}|f(0)-g(0)|_{\tau} dpdx\\ \leq &-\int_0^T  \int_{\mathbb{R}^{2d}}\text{sign}_{\tau}(f-g) \nabla_x\zeta_{\sigma}\cdot (\varepsilon\nabla_x f-pf+pf^2 -\varepsilon\nabla_x  g+pg-pg^2)dpdxds\\
&-\int_0^T \int_{\mathbb{R}^{2d}}\text{sign}_{\tau}(f-g) \nabla_p\zeta_{\sigma}\cdot (\nabla_p f+p(f- f^2)- \nabla_p g-p(g- g^2))dpdxds\\
&+ \int_0^T\int_{\mathbb{R}^{2d}}((f-g)\text{sign}_{\tau}(f-g)-|f-g|_{\tau})\mathrm{div}_p(p\zeta_{\sigma} (1- f- g))dpdxdt\\
&-\int_0^T \int_{\mathbb{R}^{2d}}((f-g)\text{sign}_{\tau}(f-g)-|f-g|_{\tau})\mathrm{div}_x(p\zeta_{\sigma} (1- f- g))dpdxdt.
 \end{align*} 
 Since $|f-g|_{\tau}\to |f-g|,$ $\text{sign}_{\tau}(f-g)\to \text{sign}(f-g),$ and $(f-g)\text{sign}_{\tau}(f-g)-|f-g|_{\tau}\to 0$ pointwise  as $\tau\to 0,$ we conclude that

\label{lem:p-moment}

We omit the dependence of the solution $f\colonequals f_{\varepsilon}$ on $\varepsilon$ for notational convenience. 
We will prove by induction on $\gamma$. First, we will show that the second moment is bounded. Afterwards, we will assume that we can bound the $(\gamma-2)$-moment and from the induction hypothesis  we obtain that the $\gamma$-moment is bounded.

 Let  $\phi:\mathbb{R}\to [0,\infty)$ be a smooth function such that $\phi(s)=\phi(-s),$ $\phi(s)=s^2$ for $|s|\leq 1,$ and $\phi(s)=4$ for $|s|\geq 2.$ We also require that $\phi'(s)\geq 0$ for $1\leq s\leq 2.$  Let $R>0$ and $\phi_R(p)\colonequals R^2\phi(|p|/R). $
 Let $\zeta_{\sigma}$ be the function defined in \eqref{zeta}.  We multiply \eqref{Eq} by $\phi_R\zeta_{\sigma}$ and integrate by parts
\begin{align*}
\frac{d}{dt}\int_{\mathbb{R}^{2d}}\zeta_{\sigma}\phi_R fdxdp=&\varepsilon \int_{\mathbb{R}^{2d}}\Delta_x\zeta_{\sigma}\phi_R fdxdp\\
&+\int_{\mathbb{R}^{2d}}(\Delta_p\zeta_{\sigma}\phi_R+2\nabla_p\zeta_{\sigma}\cdot \nabla_p \phi_R+\Delta_p \phi_R \zeta_{\sigma} ) fdxdp\\
&+ \int_{\mathbb{R}^{2d}}\nabla_x\zeta_{\sigma}\cdot p \phi_R (f-f^2)dxdp\\
&-\int_{\mathbb{R}^{2d}}(\nabla_p\zeta_{\sigma}\cdot p\phi_R+\zeta_{\sigma}\nabla_p \phi_R\cdot p) (f-f^2)dxdp.
\end{align*}
We integrate this equation on the interval $[0,T]$ with $0<T<T_{\mathrm{max}}$
\begin{align*}
\int_{\mathbb{R}^{2d}}&\zeta_{\sigma}\phi_R f(T)dxdp-\int_{\mathbb{R}^{2d}}\zeta_{\sigma}\phi_R f(0)dxdp\\
=&\varepsilon \int_0^T\int_{\mathbb{R}^{2d}}\Delta_x\zeta_{\sigma}\phi_R fdxdpdt\\
&+\int_0^T\int_{\mathbb{R}^{2d}}(\Delta_p\zeta_{\sigma}\phi_R+2\nabla_p\zeta_{\sigma}\cdot \nabla_p \phi_R+\Delta_p \phi_R \zeta_{\sigma} ) fdxdpdt\\
&+ \int_0^T\int_{\mathbb{R}^{2d}}\nabla_x\zeta_{\sigma}\cdot p \phi_R (f-f^2)dxdpdt\\
&-\int_0^T\int_{\mathbb{R}^{2d}}(\nabla_p\zeta_{\sigma}\cdot p\phi_R+\zeta_{\sigma}\nabla_p \phi_R\cdot p) (f-f^2)dxdpdt.
\end{align*}
Equation \eqref{gr.z} shows that  $\Delta_x\zeta_{\sigma}$, $\Delta_p\zeta_{\sigma}$,  $\nabla_x\zeta_{\sigma}$, and $\nabla_p\zeta_{\sigma}$ converge to zero uniformly as $\sigma \to 0$, and $\nabla_x\zeta_{\sigma}\cdot p$ and $\nabla_p\zeta_{\sigma}\cdot p$ are bounded.
Since $0\leq \phi_R\leq R^2$, $f\in \Upsilon_T,$ and $0\leq f\leq 1,$ we can pass to the limit as $\sigma \to 0$ by the Lebesgue dominated convergence theorem
\begin{align*}
\int_{\mathbb{R}^{2d}} \phi_R f(T)dxdp-\int_{\mathbb{R}^{2d}}\phi_R f_0dxdp
=&\int_0^T\int_{\mathbb{R}^{2d}} \Delta_p \phi_R   fdxdpdt-\int_0^T\int_{\mathbb{R}^{2d}}\nabla_p \phi_R\cdot p (f-f^2)dxdpdt.
\end{align*} 
 We have  $\nabla_p \phi_R\cdot p\geq 0$ and $ |\Delta_p \phi_R  |\leq C$ for some constant $C>0$ (independent of $R$).     Using the mass conservation and $f-f^2\geq 0$,  we can estimate
 \begin{align*}
\int_{\mathbb{R}^{2d}} \phi_R f(T)dxdp\leq \int_{\mathbb{R}^{2d}}\phi_R f_0dxdp
+CT\int_{\mathbb{R}^{2d}}    f_0dxdp.
\end{align*}
We now let $R\to \infty.$ Fatou's lemma provides
  \begin{align*}
\int_{\mathbb{R}^{2d}} |p|^2 f(T)dxdp\leq \int_{\mathbb{R}^{2d}}|p|^2 f_0dxdp
+CT\int_{\mathbb{R}^{2d}}    f_0dxdp.
\end{align*}
This show that the second moment $\int_{\mathbb{R}^{2d}} |p|^2 f(T)dxdp$ is finite for all $T\in (0,T_{\mathrm{max}})$  if $ \int_{\mathbb{R}^{2d}}|p|^2 f_0dxdp$ is finite. Having this information we can actually redo the above computations by replacing $\phi_R$ with $|p|^2$ and obtain  
\begin{align*}
\int_{\mathbb{R}^{2d}}|p|^2 f(t)dxdp\leq\int_{\mathbb{R}^{2d}}|p|^2 f_0dxdp+2d t\int_{\mathbb{R}^{2d}} f_0dxdp,\,\,\,\,\forall\, t\in [0,T_{\mathrm{max}}).
\end{align*}
This shows that the second $\int_{\mathbb{R}^{2d}}|p|^2 f(t)dxdp$ is bounded by a linear polynomial.  
In view of $2|p|\leq 1+|p|^2$, we have
$$2\int_{\mathbb{R}^{2d}}|p| f(t)dxdp\leq \int_{\mathbb{R}^{2d}} f_0dxdp+\int_{\mathbb{R}^{2d}}|p|^2 f(t)dxdp.$$
Therefore, $\int_{\mathbb{R}^{2d}}|p| f(t)dxdp$ is also bounded by a linear polynomial. Also,  note $\lceil 1/2\rceil =1.$ Hence, the claimed bound holds for  $\gamma=1,2.$


Next, we assume that $\gamma\in \mathbb{N}$ and  the $(\gamma-2)$-moment is bounded by a polynomial of degree $\lceil \gamma/2-1\rceil.$ As we did before, one can show that the $\gamma$-moment $\int_{\mathbb{R}^{2d}} |p|^{\gamma} f(T)dxdp$ is finite for all $T\in (0,T_{\mathrm{max}})$  if $ \int_{\mathbb{R}^{2d}}|p|^{\gamma} f_0dxdp$ is finite. We compute the $\gamma$-moment 
for any $T<T_{\mathrm{max}}$
\begin{align*}
\int_{\mathbb{R}^{2d}}\zeta_{\sigma}&|p|^{\gamma} f(T)dxdp=\int_{\mathbb{R}^{2d}}\zeta_{\sigma}|p|^{\gamma} f_0dxdp+\varepsilon \int_0^T\int_{\mathbb{R}^{2d}}\Delta_x\zeta_{\sigma}|p|^{\gamma} fdxdpdt
\\&+\int_0^T\int_{\mathbb{R}^{2d}}(\Delta_p\zeta_{\sigma}|p|^{\gamma}+2\gamma|p|^{\gamma-2}\nabla_p\zeta_{\sigma}\cdot p+\gamma(\gamma-2+d)|p|^{\gamma-2} \zeta_{\sigma} ) fdxdpdt\\
&+ \int_0^T\int_{\mathbb{R}^{2d}}\nabla_x\zeta_{\sigma}\cdot p |p|^{\gamma} (f-f^2)dxdpdt\\
&-\int_0^T\int_{\mathbb{R}^{2d}}(\nabla_p\zeta_{\sigma}\cdot p|p|^{\gamma}+\gamma\zeta_{\sigma}|p|^{\gamma}) (f-f^2)dxdpdt.
\end{align*}
Equation \eqref{gr.z} shows that  $\Delta_x\zeta_{\sigma}$, $\Delta_p\zeta_{\sigma}$,  $\nabla_x\zeta_{\sigma}$, and $\nabla_p\zeta_{\sigma}$ converge to zero uniformly as $\sigma \to 0$, and $\nabla_x\zeta_{\sigma}\cdot p$ and $\nabla_p\zeta_{\sigma}\cdot p$ are bounded.
Since $\int_{\mathbb{R}^{2d}} |p|^{\gamma} f(T)dxdp$ is finite and  $0\leq f\leq 1,$ we can pass to the limit as $\sigma \to 0$ by the Lebesgue dominated convergence theorem
\begin{align*}
\int_{\mathbb{R}^{2d}}|p|^{\gamma} f(T)dxdp=&\int_{\mathbb{R}^{2d}}|p|^{\gamma} f_0dxdp+
\gamma(\gamma-2+d)\int_0^T\int_{\mathbb{R}^{2d}}|p|^{\gamma-2}  fdxdpdt\\
&-\gamma\int_0^T\int_{\mathbb{R}^{2d}} |p|^{\gamma} (f-f^2)dxdpdt.
\end{align*}
Using $0\leq f\leq 1$ we get 
\begin{align}\label{Pgam}
\int_{\mathbb{R}^{2d}}|p|^{\gamma} f(T)dxdp\leq &\int_{\mathbb{R}^{2d}}|p|^{\gamma} f_0dxdp+
\gamma(\gamma-2+d)\int_0^T\int_{\mathbb{R}^{2d}}|p|^{\gamma-2}  fdxdpdt.
\end{align}
By induction, $\int_{\mathbb{R}^{2d}}|p|^{\gamma-2}  fdxdp$ is bounded by a polynomial $P_{\lceil \gamma/2-1\rceil}(t)$ of degree $\lceil\gamma/2-1\rceil=\lceil\gamma/2\rceil -1.$ Since its integral on $[0,T]$ gives a polynomial of degree $\lceil \gamma/2\rceil$ with respect to $T$, we obtain the claimed result for all natural numbers  $\gamma\leq k$.  

Let $\gamma$ be not a natural number and $\lceil \gamma\rceil\leq k$. 
 We write $\gamma=\lceil \gamma  \rceil  -\{\gamma\},$ where $0<\{\gamma\}<1.$ Using $|p|^{\gamma}\leq C(1+|p|^{\lceil \gamma \rceil}) $ we have  $$\int_{\mathbb{R}^{2d}}|p|^{\gamma} f(T)dxdp\leq C\int_{\mathbb{R}^{2d}} f_0dxdp+C\int_{\mathbb{R}^{2d}}|p|^{\lceil \gamma \rceil} f(T)dxdp.$$ 
The right hand side of this equation is bounded by a polynomial $P_{\lceil \lceil \gamma \rceil/2\rceil}$. As we have $\lceil \lceil \gamma \rceil/2\rceil=\lceil  \gamma /2\rceil$, we obtain the claimed result for all $\gamma\geq 1$ with $\lceil \gamma\rceil\leq k$. If $k$ is a natural number, the proof is complete.

If $k$ is not  a natural number, then for $\lceil k \rceil -1<\gamma\leq k$,  we have $\lceil \gamma \rceil =\lceil k \rceil +1>k$ and so the above argument cannot be used. In this case we use the fact that  \eqref{Pgam} holds even $\gamma$ is not a natural number.  As we have  $\lceil \gamma-2\rceil=\lceil \gamma\rceil -2 \leq k ,$ we can apply the above argument to the right hand side of  \eqref{Pgam}, implying the claimed result in this case.

 \end{align*}
  We integrate this equation on the interval $[0,T],\, T>0,$ and  use
 $\zeta_{\sigma}  \text{sign}_{\tau}'(u_i)|\nabla_pu_i|^2\geq 0$ and $\zeta_{\sigma}  \text{sign}_{\tau}'(u_i)|\nabla_xu_i|^2 \geq 0$ to estimate

 $$\text{sign}_{\tau}'(u_i)\nabla_xu_iu_i =\nabla_x(u_i\text{sign}_{\tau}(u_i)-|u_i|_{\tau}),$$
 $$\text{sign}_{\tau}'(u_i)\nabla_pu_iu_i=\nabla_p(u_i\text{sign}_{\tau}(u_i)-|u_i|_{\tau}),$$

 \end{align*}

 Since $T>0$ is arbitrary, we obtain the  claimed result.
 
Next, we estimate $||\nabla_p f_{\varepsilon}(t)||_{L^1(\mathbb{R}^{2d})} $: The partial derivatives $v_i\colonequals \partial_{p_i}f$ for $i\in \{1,...,d\}$ satisfy

\end{equation*}
This equation differs from  \eqref{u_i} only by the term $(1-2f)u_i+(1-2f)v_i.$ Hence, we repeat the above computations  by only caring this term and obtain  $$\int_{\mathbb{R}^{2d}}|v_i(T)| dpdx- \int_{\mathbb{R}^{2d}}|v_i(0)| dpdx\leq \int^T_0 \int_{\mathbb{R}^{2d}}\mathrm{sign}(v_i)(1-2f)u_idxdpdt+\int^T_0\int_{\mathbb{R}^{2d}} |v_i|(1-2f)dxdpdt$$
for all $T>0.$  Using $|\mathrm{sign}(v_i)|\leq 1$, $|1-2f|\leq 1$, and $\int_{\mathbb{R}^{2d}}|u_i(t)| dpdx\leq \int_{\mathbb{R}^{2d}}|u_i(0)| dpdx$ for all $t\geq 0,$ we estimate 
$$\int_{\mathbb{R}^{2d}}|v_i(T)| dpdx- \int_{\mathbb{R}^{2d}}|v_i(0)| dpdx\leq  T\int_{\mathbb{R}^{2d}}|u_i(0)|dxdp+\int^T_0 \int_{\mathbb{R}^{2d}}|v_i| dxdpdt.$$
Grönwall's inequality implies 
$$\int_{\mathbb{R}^{2d}}|v_i(T)| dpdx\leq \left(\int_{\mathbb{R}^{2d}}|v_i(0)| dpdx+T\int_{\mathbb{R}^{2d}}|u_i(0)|dxdp\right) e^T.$$
Since $T>0$ is arbitrary, we obtain the  claimed result.

By Theorem \ref{th:global} and Lemma \ref{l:epsilon} 
for every $T>0$, 
$$\int_0^T \int_{\mathbb{R}^{2d}} \big(f_{\varepsilon}(t,x,p)+ |\partial_t f_{\varepsilon}(t,x,p)|+ |\nabla_x f_{\varepsilon}(t,x,p)|+ |\nabla_p f_{\varepsilon}(t,x,p)|\big)dxdpdt$$ is uniformly bounded with respect to $\varepsilon\in (0, 1].$  This fact and  \cite[Proposition 9.3]{Bre} imply 
$$\lim_{s\searrow 0,\, |y|\to 0,\,|p|\to 0}\sup_{\varepsilon\in (0,1]}\int_0^T\int_{\mathbb{R}^{2d}}|f_{\varepsilon}(t+s,x+y, p+q)-f_{\varepsilon}(t,x,p)|dxdpdt=0.$$ Then, by the Kolmogorov-Riesz-Fréchet theorem \cite[Theorem 4.26]{Bre} and a diagonal argument, there exists a sequence $\varepsilon_n\to 0$ such that $ f_{\varepsilon_n}$ converges to a function $f$ in $L^1_{\mathrm{loc}}((0,\infty)\times \mathbb{R}^{2d})$ as $n\to \infty.$ 

We show that the limit $f$ is an entropy solution to \eqref{DEq}, i.e., we show $f$ satisfies the conditions in Definition \ref{def:gen.sol}: We may assume that $f_{\varepsilon_n}$ converges to $f$ pointwise almost everywhere in $(0,\infty)\times \mathbb{R}^{2d}$. This implies 
$$0\leq f\leq 1$$
almost everywhere in $(0,\infty)\times \mathbb{R}^{2d}$.

  The pointwise convergence and Fatou's lemma imply $$f \in L^1([0,T]\times \mathbb{R}^{2d})$$ for  every $T>0$. 
  
  
  


 Next, we show that $f$ is in $BV((0,T)\times \mathbb{R}^{2d})$ for every $T>0$: Let $\phi\in [C^1_c((0,T)\times \mathbb{R}^{2d})]^{2d+1}$ and $||\phi||_{L^{\infty}([0,T]\times \mathbb{R}^{2d})}\leq 1. $ Since $\phi$ has a compact support and $ f_{\varepsilon_n}\to f$ in $L^1_{\mathrm{loc}}((0,\infty)\times \mathbb{R}^{2d}),$ we have
\begin{align*}
\left|\int_0^{T}\int_{\mathbb{R}^{2d}}f\mathrm{div}_{t,x,p}\phi dxdpdt\right|&=\left|\lim_{n\to \infty}\int_0^{T}\int_{\mathbb{R}^{2d}}f_{\varepsilon_n}\mathrm{div}_{t,x,p}\phi dxdpdt\right|\\
&=\left|\lim_{n\to \infty}\int_0^{T}\int_{\mathbb{R}^{2d}}\nabla_{t,x,p}f_{\varepsilon_n}\cdot \phi dxdpdt\right|\\
&\leq \lim_{n\to \infty}\int_0^{T}\int_{\mathbb{R}^{2d}}|\nabla_{t,x,p}f_{\varepsilon_n}| dxdpdt ,
\end{align*}
where the last inequality follows from $||\phi||_{L^{\infty}((0,T)\times \mathbb{R}^{2d})}\leq 1.$
Since the right hand side of this equations is bounded by a constant independent of $\varepsilon_n$, we conclude that the total variation of $f$
\begin{align*}
\sup_{\phi}\left\{\int_0^{T}\int_{\mathbb{R}^{2d}}f\mathrm{div}_{t,x,p}\phi dxdpdt:\,\,\, \phi\in [C^1_c((0,T)\times \mathbb{R}^{2d})]^{2d+1}\,\,\, \text{and}\,\,\, ||\phi||_{L^{\infty}((0,T)\times \mathbb{R}^{2d})}\leq 1\right\}
\end{align*}
is finite. This fact and \cite[Proposition 3.6]{Amb} imply that $f\in BV((0,T)\times \mathbb{R}^{2d})$ for every $T>0$. 


Lemma \ref{l:epsilon} shows that $||\nabla_p f_{\varepsilon_n}||_{L^2((0,T)\times \mathbb{R}^{2d})}$ is uniformly bounded  for every $T>0$.  Without loss generality, we may assume that the entire sequence $\nabla_p f_{\varepsilon_n}$ converges weakly to a vector function $F\in [L^2((0,T)\times \mathbb{R}^{2d})]^d$ . Let $\psi $ be a smooth compactly supported function in  $(0,T)\times \mathbb{R}^{2d}).$ 
Since  $ f_{\varepsilon_n}\to f$ in $L^1_{\mathrm{loc}}((0,\infty)\times \mathbb{R}^{2d})$ and $\nabla_p f_{\varepsilon_n} \rightharpoonup F$ as $n\to \infty$,   we have
\begin{align*}
\int_0^T\int_{\mathbb{R}^{2d}}f \nabla_p \psi dxdt&=\lim_{n\to \infty} \int_0^T\int_{\mathbb{R}^{2d}}f_{\varepsilon_n}\nabla_p \psi dxdt\\
&=-\lim_{n\to \infty} \int_0^T\int_{\mathbb{R}^{2d}}\nabla_p f_{\varepsilon_n} \psi dxdt\\

\end{align*}
Thus, by the definition of weak derivatives
$\nabla_p f=F\in [L^2((0,T)\times \mathbb{R}^{2d})]^d. $

$$\int_{\mathbb{R}^{2d}}f(T)dpdx=\int_{\mathbb{R}^{2d}}f_0dpdx$$
for almost every $T>0.$
