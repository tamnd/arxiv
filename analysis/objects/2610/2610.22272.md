<!-- definition-1: body_md -->
\label{def:encoder}
    Let $\mathcal{T}$ be a finite set. An \emph{encoder} of $X$ is a conditional
    distribution (or channel) $P_{T\mid X}$ from $\mathcal{X}$ to $\mathcal{T}$, i.e., a
    collection $\{P_{T\mid X}(\cdot \mid x)\}_{x \in \mathcal{X}}$ of probability
    distributions on $\mathcal{T}$. The encoder $P_{T\mid X}$ induces a random variable $T$ taking values
    in $\mathcal{T}$ such that $(X,Y,T)$ has joint distribution
    \begin{equation}\label{eq:encoder_joint}
        P_{XYT}(x,y,t) = P_X(x)\, P_{Y\mid X}(y \mid x)\, P_{T\mid X}(t \mid x),
        \qquad \forall (x,y,t) \in \mathcal{X} \times \mathcal{Y} \times \mathcal{T}.
    \end{equation}
    In other words, $T$ and $Y$ are conditionally independent given $X$.
<!-- end definition-1 -->
<!-- proposition-2: body_md -->
Let $\mathcal{T}$ be a finite set. For every encoder $P_{T\mid X}$ defining the random variable $T\sim P_T$, it holds that
    \begin{equation}\label{eq:first_direction_encoder}
        \mathbb{E}_{T}[P_{X\mid T}(x\mid T)]=\sum_{t\in \mathcal{T}} P_T(t)P_{X\mid T}(x\mid t)=P_X(x),\quad\text{for all } x\in \mathcal{X}.
    \end{equation}
    Conversely, let $P$ be a probability distribution on $\mathcal{T}$ and let $\{Q_t\}_{t\in \mathcal{T}}$ be a collection of probability distribution on $\mathcal{X}$ such that
    \begin{equation}\label{eq:posterior_encoder}
        \sum_{t\in \mathcal{T}} P(t)Q_t(x) = P_X(x),\quad\text{for any } x\in\mathcal{X}.
    \end{equation}
    Then the channel $P_{T\mid X}$ defined as
    \begin{equation}\label{eq:encoder_from_mixture}
        P_{T\mid X}(t\mid x)=\frac{P(t)Q_t(x)}{P_X(x)}, \quad\forall x\in\mathcal{X}\text{ and } t\in\mathcal{T},
    \end{equation}
    defines an encoder of $X$ whose induced random variable $T$ satisfies $P_T(t)=P(t)$ for any $t\in \mathcal{T}$ and $P_{X\mid T}(\cdot\mid t)=Q_t$ for every $t\in supp(P)\subseteq\mathcal{T}$.
<!-- end proposition-2 -->
<!-- proof-1: body_md -->
The identity~\eqref{eq:first_direction_encoder} follows from the law of total probability.

    For the converse, we have that $P_{T\mid X}(t\mid x)\geq 0$ for all $x$ and $t$, since the right-hand side of~\eqref{eq:encoder_from_mixture} is non-negative. Moreover, by~\eqref{eq:posterior_encoder}, we have that
    \begin{equation}
        \sum_{t \in \mathcal{T}} P_{T\mid X}(t \mid x) =\sum_{t\in \mathcal{T}}\frac{P(t)Q_t(x)}{P_X(x)}=\frac{P_X(x)}{P_X(x)}=1,
    \end{equation}
    for every $x \in \mathcal{X}$. Hence, $P_{T\mid X}$ is a valid channel from $\mathcal{X}$ to $\mathcal{T}$, and since $T$ depends only on $X$, $T$ is conditionally independent of $Y$ given $X$. Consequently, $P_{T\mid X}$ is an encoder of $X$. Moreover, one can also see that for any $t\in \mathcal{T}$
    \begin{equation}
        P_T(t)=\sum_{x\in \mathcal{X}} P_X(x)P_{T\mid X}(t\mid x)=\sum_{x\in \mathcal{X}}P_X(x)\frac{P(t)Q_t(x)}{P_X(x)}=P(t), 
    \end{equation}
    and that, for $t\in \operatorname{supp}(P)$ and $x\in \mathcal{X}$, by the Bayes' rule, we have that
    \begin{equation}
        P_{X\mid T}(x\mid t)=\frac{P_X(x)P_{T\mid X}(t\mid x)}{P_T(t)}= \frac{P_X(x)P(t)Q_t(x)}{P_X(x)P(t)}=Q_t(x).
    \end{equation}

<!-- lemma-1: body_md -->
\label{lemma:posterior_Y_give_T}
    Let $P_{T\mid X}$ be an encoder of $X$ inducing a random variable $T\sim P_T$ on $\mathcal{T}$, with $P_T(t)>0$ for all $t\in \mathcal{T}$. Then, it holds that
    \begin{equation}
        P_{Y\mid T}(\cdot\mid t)=WP_{X\mid T}(\cdot\mid t),\quad\text{for all } t\in \mathcal{T},
    \end{equation}
    where $W$ is defined in~\eqref{eq:W_def}.
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
For any $y\in \mathcal{Y}$ and $t\in \mathcal{T}$, we have that
    \begin{align*}
        P_{Y\mid T}(y\mid t)&=\frac{P_{YT}(y,t)}{P_T(t)}\\
        &=\frac{\sum_{x\in\mathcal{X}} P_{XYT}(x,y,t)}{P_T(t)}\\
        &=\frac{\sum_{x\in\mathcal{X}} P_X(x)P_{Y\mid X}(y\mid x) P_{T\mid X}(t\mid x)}{P_T(t)}\\
        &=\frac{\sum_{x\in\mathcal{X}} P_{XT}(x,t)P_{Y\mid X}(y\mid x)}{P_T(t)}\\
        &=\frac{\sum_{x\in\mathcal{X}} P_T(t)P_{X\mid T}(x\mid t)P_{Y\mid X}(y\mid x)}{P_T(t)}\\
        &=\sum_{x\in \mathcal{X}}P_{Y\mid X}(y\mid x)P_{X\mid T}(x\mid t)\\
        &=\sum_{x\in \mathcal{X}}W(y\mid x)P_{X\mid T}(x\mid t) = (WP_{X\mid T}(\cdot\mid t))(y).
    \end{align*}

<!-- lemma-2: body_md -->
\label{lemma:Lagrangian_expressed_wt_posteriors}
    Let $P_{T\mid X}$ be an encoder of $X$ inducing a random variable $T\sim P_T$ on $\mathcal{T}$ with $P_T(t)>0$ for all $t\in \mathcal{T}$, and let $W$ be the channel defined in~\eqref{eq:W_def}. Then, we have that
    \begin{equation}\label{eq:lagrangian_cost}
        I(Y;T)-\frac{1}{2}I(X;T)=g_{W}(P_X)-\mathbb{E}_{T}[g_W(P_{X\mid T}(\cdot\mid T))].
    \end{equation}
    In particular,
    \begin{equation}\label{eq:L_1/2_g_def}
        L_{1/2}(P_{XY})=g_W(P_X)-\inf_{P_{T\mid X: T-X-Y}}\mathbb{E}_{T}[g_W(P_{X\mid T}(\cdot\mid T))].
    \end{equation}
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
By definition of mutual information in~\eqref{eq:mutual_information_def} and of conditional entropy in~\eqref{eq:conditional_entropy}, we have that
    \begin{equation}\label{eq:IXT_post}
        I(X;T) = H(X) - H(X \mid T)
        = H(P_X) - \mathbb{E}_{T}\Big[H\big(P_{X\mid T}(\cdot \mid T)\big)\Big],
    \end{equation}
    and similarly that
     \begin{equation}\label{eq:ITY_post}
        I(T;Y) = H(Y) - H(Y \mid T)
        = H(P_Y) - \mathbb{E}_{T}\Big[H\big(P_{Y\mid T}(\cdot \mid T)\big)\Big].
    \end{equation}
    Since $P_Y = W P_X$ and, by Lemma~\ref{lemma:posterior_Y_give_T},
    $P_{Y\mid T}(\cdot \mid t) = W P_{X\mid T}(\cdot \mid t)$ for every
    $t \in \mathcal{T}$, subtracting one half of~\eqref{eq:IXT_post}
    from~\eqref{eq:ITY_post} gives
    \begin{align*}
        I(T;Y) - \frac12 I(X;T)
        &= H(W P_X) - \frac12 H(P_X)
        - \mathbb{E}_{T}\Big[H\big(W P_{X\mid T}(\cdot \mid T)\big)
        - \frac12 H\big(P_{X\mid T}(\cdot \mid T)\big)\Big] \\
        &= g_W(P_X) - \mathbb{E}_{T}\Big[g_W\big(P_{X\mid T}(\cdot \mid T)\big)\Big],
    \end{align*}
    which proves~\eqref{eq:lagrangian_cost}. 
    
    Finally, by~\eqref{eq:L_beta_def} and~\eqref{eq:lagrangian_cost}, we obtain that
    \begin{align}
         L_{1/2}(P_{XY})&=\sup_{P_{T\mid X}:T-X-Y} [I(Y;T)-\frac{1}{2} I(X;T)]\nonumber\\
         &=\sup_{P_{T\mid X}:T-X-Y} \Bigg[ g_W(P_X) - \mathbb{E}_{T}\Big[g_W\big(P_{X\mid T}(\cdot \mid T)\big)\Big]\Bigg]\nonumber\\
         &=g_W(P_X)-\inf_{P_{T\mid X: T-X-Y}}\mathbb{E}_{T}[g_W(P_{X\mid T}(\cdot\mid T))],
    \end{align}
    where the last equality holds because $g_W(P_X)$ does not depend on $T$.

<!-- lemma-3: body_md -->
\label{lemma:cost_clique}
    Let $C$ be a clique of $G$ of size $j \ge 2$. Then $W_G U_C$ is the
    uniform distribution on $C$, that is,
    \begin{equation}\label{eq:W_U_C}
        (W_GU_C)(v)=\begin{cases}
            1/j &\text{if } v\in C,\\
            0&\text{if } v\in V(G)\setminus C.
        \end{cases}
    \end{equation}
    and
    \begin{equation}\label{eq:cost_clique}
        g_G(U_C) = c_j.
    \end{equation}
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
For every $v \in V(G)$, by~\eqref{eq:U_C_def},
    \begin{equation}
        (W_G U_C)(v) = \sum_{(w,u) \in A(G)} W_G\big(v \mid (w,u)\big)\, U_C(w,u)
        = \frac{1}{j(j-1)} \sum_{(w,u) \in A(C)} W_G\big(v \mid (w,u)\big).
    \end{equation}
    By~\eqref{eq:endpoint_channel}, $W_G(v \mid (w,u)) = 1/2$ if
    $v \in \{w,u\}$, and $0$ otherwise. If $v \in C$, exactly $2(j-1)$
    oriented edges of $A(C)$ have $v$ as an endpoint, hence
    \begin{equation}
        (W_G U_C)(v) = \frac{1}{j(j-1)} \cdot 2(j-1) \cdot \frac12 = \frac1j.
    \end{equation}
    If $v \notin C$, no oriented edge of $A(C)$ has $v$ as an endpoint, hence
    $(W_G U_C)(v) = 0$. This proves~\eqref{eq:W_U_C}. Moreover, since
    $H(W_G U_C) = \log j$ and $H(U_C) = \log\big(j(j-1)\big)$, from~\eqref{eq:g_W_G_def} we finally get that
    \begin{equation*}
        g_G(U_C) = \log j - \frac12 \log\big(j(j-1)\big)
        = -\frac12 \log\Big(1 - \frac1j\Big) = c_j,
    \end{equation*}
    which concludes the proof.

<!-- lemma-5: body_md -->
\label{lemma:entropic_turan}
    Let $g_G$ be the function defined in~\eqref{eq:g_W_G_def}. Then, for every probability distribution $P$ on $A(G)$, it holds that
    \begin{equation}\label{eq:entropic_turan}
        g_G(P) \ge c_{\omega(G)},
    \end{equation}
    with equality if $P = U_C$ for a clique $C$ of size $\omega(G)$.
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
Let $\pi = W_G P$, and let $P_1$ and $P_2$ denote the
    distributions on $V(G)$ such that for any $v \in V(G)$
    \begin{equation}
        P_1(v)=\sum_{u: (v,u)\in A(G)} P(v,u)\quad\text{and}\quad P_2(v)=\sum_{u: (u,v)\in A(G)}P(u,v).
    \end{equation}
    By~\eqref{eq:endpoint_channel}, $W_G(v\mid (w,u))=1/2$ if and only if $v\in\{w,u\}$. Hence, we have that
    \begin{align}
        \pi(v)&=\sum_{(w,u)\in A(G)}W_G(v\mid (w, u))P(w,u)\nonumber\\
        &=\frac{1}{2}\sum_{u:(v,u)\in A(G)}P(v,u)+\frac{1}{2}\sum_{w:(w,v)\in A(G)}P(w,v)\nonumber\\
        &=\frac{1}{2}\left(P_1(v)+P_2(v)\right).\label{eq:pi_P_1_P_2}
    \end{align}
    Let us now extend $P$ to a distribution on all $V(G) \times V(G)$ by setting it to
    zero outside $A(G)$.  We observe that if
    $P(u,v) > 0$, then $\pi(u) \ge \frac12 P_1(u) > 0$ and
    $\pi(v) \ge \frac12 P_2(v) > 0$. Hence
    $\operatorname{supp}(P) \subseteq \operatorname{supp}(\pi \otimes \pi)$, and $D(P\|\pi\otimes\pi)$ is finite. Moreover, using~\eqref{eq:pi_P_1_P_2}, we obtain that
    \begin{align}
        D(P\|\pi\otimes\pi)&=-H(P)-\sum_{(u,v)\in V(G)\times V(G)} P(u,v)\log (\pi(v)\pi(u))\nonumber\\
        &=-H(P)-\sum_{(u,v)\in A(G)} P(u,v)\log (\pi(u)\pi(v))\\
        &=-H(P)-\sum_{v:(u,v)\in A(G)} P(u,v)\log\pi(u)-\sum_{u:(u,v)\in A(G)} P(u,v)\log\pi(v)\\
        &=-H(P)-\sum_{u\in V(G)} P_1(u)\log \pi(u)-\sum_{v\in V(G)} P_2(v)\log \pi(v)\\
        &=-H(P)-\sum_{v\in V(G)} (P_1(v)+P_2(v))\log \pi(v)\\
        &=-H(P)-2\sum_{v\in V(G)} \pi(v)\log \pi(v)\\
        &=2H(\pi)-H(P)=2g_G(P),\label{eq:equality_KL_g_G}
    \end{align}
    where the last equality follows from $\pi= W_G P$ and~\eqref{eq:g_W_G_def}.

    Now, by applying Jensen's inequality on the concavity of the logarithm we obtain the following lower bound on $D(P\|\pi\otimes\pi)$:
    \begin{equation}
        D(P\|\pi\otimes\pi)\geq -\log\left(\sum_{(v,u)\in A(G)} \pi(v)\pi(u)\right).\label{eq:lower_bound_KL}
    \end{equation}
    By Lemma~\ref{lemma:motzkin_straus}, \eqref{eq:lower_bound_KL} yields
    \begin{equation}
        D(P\|\pi\otimes\pi)\geq-\log\left(1-\frac{1}{\omega(G)}\right)=2c_{\omega(G)}.\label{eq:bound_KL_2}
    \end{equation}
    Combining~\eqref{eq:equality_KL_g_G} and~\eqref{eq:bound_KL_2} yields~\eqref{eq:entropic_turan}. The equality for $P = U_C$ instead follows from Lemma~\ref{lemma:cost_clique}, since for a clique $C$ of size $\omega(G)$ we have that $g_G(U_C)=c_{\omega(G)}$.

<!-- lemma-7: body_md -->
\label{lemma:gadget}
      The graph ${\bar G}$ satisfies the following properties:
      \begin{enumerate}
          \item Let $E(C_e)$ denote the edges of the clique $C_e$ for any $e\in E(G)$. The collection $\{E(C_e)\}_{e\in E(G)}$ is a partition of $E({\bar G})$, that is,
          \begin{equation}
              \bigcup_{e\in E(G)} E(C_e)=E({\bar G})
          \end{equation}
          and
          \begin{equation}
              E(C_e)\cap E(C_{e'})=\emptyset, \quad\forall e,e'\in E(G) \text{ with }e\neq e'.
          \end{equation}
          
          \item The number of oriented edges of ${\bar G}$ is
          \begin{equation}
              M=|A({\bar G})|=Nm(m-1).
          \end{equation}
          
          \item $\omega({\bar G})=\max\{\omega(G),m\}$.
      \end{enumerate}
<!-- end lemma-7 -->
<!-- proof-7: body_md -->
1. By construction of ${\bar G}$, every edge of ${\bar G}$ belongs to $E(C_e)$ for some $e\in E(G)$. Thus, it suffices to show that for each edge $x\in E({\bar G})$ there exists a unique $e\in E(G)$ such that $x\in E(C_e)$.
    
    Let $x=\{u,v\}$ be an edge of ${\bar G}$. Assume first that $x\in E(G)$. Since for every $e\in E(G)$ $C_e\cap V(G)=e$, we have that $E(C_e)\cap E(G)=\{e\}$ for any $e\in E(G)$. Hence, it follows that $x\in E(C_e)$ if and only if $x=e$.


    If $x=\{u,v\}\notin E(G)$, then $x$ is an edge added in the construction of ${\bar G}$, and at least one vertex between $u$ and $v$, say $u$, belongs to $S_e$ for some $e \in E(G)$. Moreover, $x\in E(C_e)$. Since the sets $S_e$ are
    pairwise disjoint, it follows that $u \notin C_{e'}$ for every $e' \neq e$. Hence, for every $e'\neq e$, we have that
    $x \notin E(C_{e'})$. Consequently, there exists a unique $e\in E(G)$ such that $x\in E(C_e)$.

    Therefore, each edge of ${\bar G}$ lies in exactly one of the sets
    $E(C_e)$, with $e \in E(G)$. Hence,  $\{E(C_e)\}_{e\in E(G)}$ forms a partition of $E({\bar G})$.

    \smallskip
    2. By construction of ${\bar G}$, for each edge $e\in E(G)$, we add a clique of size $m$. Therefore, the total number of oriented edges is
    \begin{equation}
        |A({\bar G})|=2|E(G)|\frac{m(m-1)}{2} = Nm(m-1).
    \end{equation}

    \smallskip
    3. We first show that $\omega({\bar G}) \le \max\{\omega(G), m\}$. 
    
    Let $K$ be
    a clique of ${\bar G}$.
    If $K$ contains a vertex $z \in S_e$ for some $e \in E(G)$, then, by
    construction of ${\bar G}$, all the neighbours of $z$, that are the vertices adjacent to $z$, belong to $C_e$. Therefore, it follows that
    $K \subseteq C_e$, and hence $|K| \le |C_e| = m$.

    If $K$ does not contain a vertex $z\in S_e$ for any $e\in E(G)$, then we have that $K \subseteq V(G)$. Therefore, $K$ is
    a clique of $G$, and $|K| \le \omega(G)$.

    Conversely,  $G$ is a subgraph of ${\bar G}$, so every clique of $G$ is a
    clique of ${\bar G}$, and $\omega({\bar G}) \ge \omega(G)$. Moreover, each $C_e$ is
    a clique of ${\bar G}$ of size $m$, and $E(G) \neq \emptyset$
    by assumption. Thus, $\omega({\bar G}) \ge m$. Putting all together we have that    $\omega({\bar G}) = \max\{\omega(G), m\}$.

<!-- lemma-8: body_md -->
\label{lemma:T_0}
    The encoder $T_0$ defined in~\eqref{eq:T_0_def} satisfies
    \begin{equation}\label{eq:T_0_distribution}
        P_{T_0}(e) = \frac1N
        \qquad \text{and} \qquad
        P_{X \mid T_0}(\cdot \mid e) = U_{C_e},
        \qquad \forall e \in E(G).
    \end{equation}
    Moreover,
    \begin{equation}\label{eq:Lambda_0_def}
        I(Y;T_0) - \frac12 I(X;T_0) = g_{{\bar G}}(P_X) - c_m = \Lambda_0.
    \end{equation}
    In particular, $L_{1/2}(P_{XY}) \ge \Lambda_0$.
<!-- end lemma-8 -->
<!-- proof-8: body_md -->
We first prove~\eqref{eq:T_0_distribution}.
    For every $e \in E(G)$, the set $A(C_e)$ contains $m(m-1)$ oriented
    edges, each of probability $1/M$ under $P_X$. Hence, by~\eqref{eq:T_0_def} we have that
    \begin{align}
        P_{T_0}(e) &= \sum_{x \in A({\bar G})} P_X(x)\, P_{T_0 \mid X}(e \mid x)\nonumber\\
        &=\frac{1}{M}\sum_{x \in A({\bar G})} P_{T_0 \mid X}(e \mid x)\nonumber\\
        &= \frac{|A(C_e)|}{M}=\frac{m(m-1)}{M} = \frac1N,\nonumber
    \end{align}
    where the last equality follows by recalling that $M=m(m-1)N$.

    Moreover, by the Bayes' rule, for every $x \in A({\bar G})$, we have that
    \begin{equation}
        P_{X \mid T_0}(x \mid e)
        = \frac{P_X(x)\, P_{T_0 \mid X}(e \mid x)}{P_{T_0}(e)}
        =
        \begin{cases}
            \dfrac{N}{M} = \dfrac{1}{m(m-1)}, & x \in A(C_e),\\[4pt]
            0, & \text{otherwise},
        \end{cases}
    \end{equation}
    that is, $P_{X \mid T_0}(\cdot \mid e) = U_{C_e}$, where $U_{C_e}$ is the probability distribution on $A({\bar G})$ that is uniform on $A(C_e)$ (cfr.~\eqref{eq:U_C_def}).
    This proves~\eqref{eq:T_0_distribution}.

    Now, we prove~\eqref{eq:Lambda_0_def}. By Lemma~\ref{lemma:Lagrangian_expressed_wt_posteriors}
    and~\eqref{eq:T_0_distribution}, it follows that
    \begin{equation}\label{eq:recall_cost_T_0}
        I(Y;T_0) - \frac12 I(X;T_0)
        = g_{{\bar G}}(P_X) - \frac1N \sum_{e \in E(G)} g_{{\bar G}}(U_{C_e}).
    \end{equation}
    Since each $C_e$ is a clique of ${\bar G}$ of size $m$,
    Lemma~\ref{lemma:cost_clique} gives that $g_{{\bar G}}(U_{C_e}) = c_m$ for every
    $e \in E(G)$. This together with~\eqref{eq:recall_cost_T_0} proves~\eqref{eq:Lambda_0_def}. Finally,
    $L_{1/2}(P_{XY}) \ge \Lambda_0$ immediately follows from the definition of
    $L_{1/2}(P_{XY})$ in~\eqref{eq:L_beta_def} as a supremum over all encoders.

<!-- lemma-9: body_md -->
\label{lemma:no_instance}
    Let $P_{XY}$ be the joint distribution defined in~\eqref{eq:P_XY_explicit}, and assume that $\omega(G) < k$. Then every encoder $T$ of $X$ satisfies
    \begin{equation}\label{eq:no_instance}
        I(Y;T) - \frac12 I(X;T) \le \Lambda_0.
    \end{equation}
    Consequently, $L_{1/2}(P_{XY}) = \Lambda_0$ and
    \begin{equation}\label{eq:no_instance_IB}
        \mathrm{IB}(P_{XY}, R) \le \frac{R}{2} + \Lambda_0, \qquad \forall R \ge 0.
    \end{equation}
<!-- end lemma-9 -->
<!-- proof-9: body_md -->
Since $\omega(G) \le k-1 = m$, property~3 of Lemma~\ref{lemma:gadget}
    gives $\omega({\bar G}) = m$. Hence, by Lemma~\ref{lemma:entropic_turan}
    applied to the graph ${\bar G}$, we have $g_{{\bar G}}(Q) \ge c_m$ for every
    probability distribution $Q$ on $A({\bar G})$. In particular, this holds for
    every posterior distribution $P_{X\mid T}(\cdot \mid t)$ of an encoder
    $T$ of $X$. Thus, by Lemma~\ref{lemma:Lagrangian_expressed_wt_posteriors}, for every encoder $T$ of $X$ we obtain that
    \begin{equation*}
        I(Y;T) - \frac12 I(X;T)
        = g_{{\bar G}}(P_X) - \mathbb{E}_T\Big[g_{{\bar G}}\big(P_{X\mid T}(\cdot \mid T)\big)\Big]
        \le g_{{\bar G}}(P_X) - c_m = \Lambda_0,
    \end{equation*}
    which proves~\eqref{eq:no_instance}. Taking the supremum over all
    encoders gives $L_{1/2}(P_{XY}) \le \Lambda_0$, and by Lemma~\ref{lemma:T_0} we have that $L_{1/2}(P_{XY})\geq \Lambda_0$. Hence, $L_{1/2}(P_{XY}) = \Lambda_0$.

    Now, if $T$ is an encoder with $I(X;T) \le R$,
    then~\eqref{eq:no_instance} gives
    \begin{equation*}
        I(Y;T) \le \frac12 I(X;T) + \Lambda_0 \le \frac{R}{2} + \Lambda_0.
    \end{equation*}
    Taking the supremum over all such encoders yields~\eqref{eq:no_instance_IB}.

<!-- lemma-10: body_md -->
\label{lemma:T_1}
    The encoder $T_1$ defined in~\eqref{eq:T_1_def} satisfies
    \begin{equation}\label{eq:T_1_distribution}
        P_{T_1}(t) =
        \begin{cases}
            \lambda, & t = \star,\\
            (1-\varepsilon)/N, & t \in E(K),\\
            1/N, & t \in E(G) \setminus E(K),
        \end{cases}
        \qquad
        P_{X \mid T_1}(\cdot \mid t) =
        \begin{cases}
            U_K, & t = \star,\\
            V_t, & t \in E(K),\\
            U_{C_t}, & t \in E(G) \setminus E(K).
        \end{cases}
    \end{equation}
<!-- end lemma-10 -->
<!-- proof-10: body_md -->
Recall that $M = N m(m-1) = N/\mu$, so that $P_X(x) = 1/M = \mu/N$ for
    every $x \in A({\bar G})$.

    We first consider the case $T_1=\star$. By~\eqref{eq:T_1_def},
    we recall that $P_{T_1 \mid X}(\star \mid x) = p$ if $x \in A(K)$, and
    $P_{T_1 \mid X}(\star \mid x) = 0$ otherwise. Since $K$ is a clique of
    size $k$, it has $k(k-1) = 2B$ oriented edges. Therefore,
    \begin{equation*}
        P_{T_1}(\star)
        = \sum_{x \in A({\bar G})} P_X(x)\, P_{T_1 \mid X}(\star \mid x)
        = \sum_{x \in A(K)} \frac{\mu}{N}\, p
        = \frac{2B\mu p}{N}
        = \frac{B\varepsilon}{N} = \lambda,
    \end{equation*}
    where the last equality follows from \eqref{eq:mu_eps_lambda}. Moreover, by the Bayes' rule, for
    every $x \in A(K)$, we have that
    \begin{equation*}
        P_{X \mid T_1}(x \mid \star)
        = \frac{P_X(x)\, P_{T_1 \mid X}(\star \mid x)}{P_{T_1}(\star)}
        = \frac{\mu p / N}{2B\mu p / N}
        = \frac{1}{k(k-1)},
    \end{equation*}
    while $P_{X \mid T_1}(x \mid \star) = 0$ for $x \notin A(K)$. By
    definition~\eqref{eq:U_C_def} of $U_K$, this means that
    {$P_{X \mid T_1}(\cdot \mid \star) = U_K$}.

    We now consider $T=e$ for $e \in E(G)$. From~\eqref{eq:T_0_def}
    and~\eqref{eq:T_1_def}, we recall that
    \begin{equation}\label{eq:T_1_on_e}
        P_{T_1 \mid X}(e \mid x) =
        \begin{cases}
            1 - p, & \text{if } x \in A(C_e) \cap A(K),\\
            1, & \text{if } x \in A(C_e) \setminus A(K),\\
            0, & \text{if } x \notin A(C_e).
        \end{cases}
    \end{equation}
    Thus, $T_1$ differs from $T_0$ only on the oriented edges in
    $A(C_e) \cap A(K)$. Let us determine this set. We claim that
    \begin{equation}\label{eq:A_K_cap_A_C_e}
        A(C_e) \cap A(K) =
        \begin{cases}
            A(e), & \text{if } e \in E(K),\\
            \emptyset, & \text{otherwise}.
        \end{cases}
    \end{equation}
     To prove~\eqref{eq:A_K_cap_A_C_e}, let $x = (u,v) \in A(K)$. Since $K$
    is a clique of $G$, the edge $\{u,v\}$ is an edge of $G$. Moreover,
    $x \in A(C_e)$ if and only if $\{u,v\} \in E(C_e)$. By the proof of
    Lemma~\ref{lemma:gadget}, the only edge of $G$ that belongs to
    $E(C_e)$ is $e$. Therefore, $x \in A(C_e)$ if and only if
    $\{u,v\} = e$, that is, if and only if $x$ is one of the two
    orientations of $e$. In other words, $A(C_e) \cap A(K)$ consists of the
    orientations of $e$ that belong to $A(K)$. Therefore, if $e \in E(K)$, both orientations of $e$ belong to $A(K)$, and hence
    $A(C_e) \cap A(K) = A(e)$. If $e \notin E(K)$, no orientation of $e$
    belongs to $A(K)$, and hence $A(C_e) \cap A(K) = \emptyset$, proving~\eqref{eq:A_K_cap_A_C_e}.


    
    Now, if $e \in E(G) \setminus E(K)$, then   $A(C_e) \cap A(K) = \emptyset$ by~\eqref{eq:A_K_cap_A_C_e}. Therefore,~\eqref{eq:T_1_on_e} gives
    $P_{T_1 \mid X}(e \mid x) = P_{T_0 \mid X}(e \mid x)$ for every
    $x \in A({\bar G})$, and, by Lemma~\ref{lemma:T_0} we get that $P_{T_1}(e) = 1/N$ and
    $P_{X \mid T_1}(\cdot \mid e) = U_{C_e}$.

    If $e \in E(K)$, then $A(C_e) \cap A(K) = A(e)$
    by~\eqref{eq:A_K_cap_A_C_e}. Moreover, since $|A(e)|=2$ and $C_e$ is a clique of size $m$, we have that $|A(C_e) \setminus A(e)|=m(m-1) - 2$. Hence,
    by~\eqref{eq:T_1_on_e}, we obtain that
    \begin{equation*}
        P_{T_1}(e)
        = \sum_{x \in A({\bar G})} P_X(x)\, P_{T_1 \mid X}(e \mid x)
        = \frac{\mu}{N} \Big( \big(m(m-1) - 2\big) + 2(1-p) \Big)
        = \frac{1 - 2\mu p}{N}
        = \frac{1 - \varepsilon}{N},
    \end{equation*}
    where the last equalities follow from~\eqref{eq:mu_eps_lambda}. Moreover, by the Bayes' rule, we have that
    \begin{equation*}
        P_{X \mid T_1}(x \mid e)
        = \frac{P_X(x)\, P_{T_1 \mid X}(e \mid x)}{P_{T_1}(e)}
        =
        \begin{cases}
            \dfrac{\mu}{1-\varepsilon} = a_0,
            & \text{if } x \in A(C_e) \setminus A(e),\\[8pt]
            \dfrac{\mu(1-p)}{1-\varepsilon} = a_1,
            & \text{if } x \in A(e),\\[8pt]
            0, & \text{otherwise},
        \end{cases}
    \end{equation*}
    that is, $P_{X \mid T_1}(\cdot \mid e) = V_e$ by~\eqref{eq:V_e_def}.

\label{lemma:perturbation}
    Let $K$ be a clique of size $k$ in ${\bar G}$. Then, for every $e \in E(K)$, we have that
    \begin{equation}\label{eq:perturbation}
        g_{{\bar G}}(V_e) - c_m \le \frac{\varepsilon^2}{4 (\ln 2) \mu (1-\varepsilon)^2}.
    \end{equation}
<!-- end lemma-11 -->
<!-- proof-11: body_md -->
The idea of the proof is to first show that $g_{{\bar G}}(V_e) - c_m$ is at
    most one half of the relative entropy $D(V_e \,\|\, U_{C_e})$, then to
    bound this relative entropy by the $\chi^2$-divergence, and finally to
    compute the $\chi^2$-divergence explicitly.

     We start by rewriting $c_m$ in terms of $\mu=1/(m(m-1))$. By~\eqref{eq:c_j_def}, we have that
     \begin{equation}\label{eq:cm_rewritten}
        c_m = \frac12 \log \frac{m}{m-1}
        = \log m - \frac12 \log\big(m(m-1)\big)
        = \log m - \frac12 \log \frac{1}{\mu}.
    \end{equation}
    Combining~\eqref{eq:cm_rewritten} with the definition~\eqref{eq:g_W_G_def} of $g_{{\bar G}}$, we obtain that
    \begin{equation}\label{eq:g_V_e_split}
        g_{{\bar G}}(V_e) - c_m
        = \big(H(W_{{\bar G}} V_e) - \log m\big)
          + \frac12 \Big(\log \frac{1}{\mu} - H(V_e)\Big).
    \end{equation}
    We bound the two terms on the right-hand side
    of~\eqref{eq:g_V_e_split} separately.

    For the first term, note that $\operatorname{supp}(V_e) \subseteq A(C_e)$
    by~\eqref{eq:V_e_def}, and that every oriented edge in $A(C_e)$ has
    both endpoints in $C_e$. Hence, by~\eqref{eq:endpoint_channel_G_bar},
    the distribution $W_{{\bar G}} V_e$ is supported on $C_e$, which has $m$
    vertices. Consequently, $H(W_{{\bar G}} V_e) \le \log m$, and the first term $\big(H(W_{{\bar G}} V_e) - \log m\big)$
    in~\eqref{eq:g_V_e_split} is non-positive.


    For the second term, we recall that $U_{C_e}(x) = \mu$ for every
    $x \in A(C_e)$. Then, since $\operatorname{supp}(V_e) \subseteq A(C_e)$, we have that
    \begin{equation}\label{eq:KL_V_e}
        D(V_e \| U_{C_e})
        = \sum_{x \in \operatorname{supp}(V_e)} V_e(x) \log \frac{V_e(x)}{\mu}
        = -H(V_e) + \log \frac{1}{\mu}.
    \end{equation}
    Thus, by~\eqref{eq:KL_V_e}, the second term
    in~\eqref{eq:g_V_e_split} equals $\frac12 D(V_e \,\|\, U_{C_e})$, and we obtain that
    \begin{equation}\label{eq:g_V_e_KL}
        g_{{\bar G}}(V_e) - c_m \le \frac12 D(V_e \,\|\, U_{C_e}).
    \end{equation}

    Since $\operatorname{supp}(V_e) \subseteq A(C_e) = \operatorname{supp}(U_{C_e})$, we can apply
    Proposition~\ref{proposition:KL_Chi} to bound the right-hand side of~\eqref{eq:g_V_e_KL} yielding
    \begin{equation}\label{eq:KL_V_e_chi2}
         \frac{1}{2}D(V_e \,\|\, U_{C_e}) \le \frac{\chi^2(V_e \,\|\, U_{C_e})}{2\ln 2}.
    \end{equation}
    Let us compute $\chi^2(V_e \,\|\, U_{C_e})$. By~\eqref{eq:chi_square_def}, we have that
    \begin{equation}\label{eq:chi2_V_e_def}
        \chi^2(V_e \,\|\, U_{C_e})
        = \sum_{x \in A(C_e)} \frac{\big(V_e(x) - \mu\big)^2}{\mu}.
    \end{equation} From~\eqref{eq:V_e_def}, we recall that $V_e(x) = a_0$ for the $m(m-1) - 2$ oriented
    edges $x \in A(C_e) \setminus A(e)$, and $V_e(x) = a_1$ for the $2$
    oriented edges $x \in A(e)$. Thus, using $\mu p = \varepsilon/2$, we have that
    \begin{align*}
        &a_0 - \mu = \frac{\mu}{1-\varepsilon} - \mu
        = \frac{\mu\varepsilon}{1-\varepsilon},
        \\
        &a_1 - \mu = \frac{\mu(1-p) - \mu(1-\varepsilon)}{1-\varepsilon}
        = \frac{\mu\varepsilon - \varepsilon/2}{1-\varepsilon}
        = -\frac{\varepsilon\,(1/2 - \mu)}{1-\varepsilon}.
    \end{align*}
    Substituting these values into~\eqref{eq:chi2_V_e_def}, we obtain that
    \begin{align}
        \chi^2(V_e \,\|\, U_{C_e})
        &= \frac{1}{\mu}\Big( \big(m(m-1) - 2\big)\frac{\mu^2\varepsilon^2}{(1-\varepsilon)^2}
          + 2\,\frac{\varepsilon^2 (1/2 - \mu)^2}{(1-\varepsilon)^2} \Big)\nonumber\\
        &= \frac{\varepsilon^2}{\mu(1-\varepsilon)^2}
          \Big( \big(m(m-1) - 2\big)\mu^2 + 2\Big(\frac12 - \mu\Big)^2 \Big)\label{eq:chi_square_expansion}.
    \end{align}
    Recalling that $\mu=1/(m(m-1))$, we have that $m(m-1)\mu = 1$. Therefore, the expression in the parentheses in \eqref{eq:chi_square_expansion} simplifies to 
    \begin{equation*}
         \Big( \big(m(m-1) - 2\big)\mu^2 + 2\Big(\frac12 - \mu\Big)^2 \Big)=\mu - 2\mu^2 + \frac12 - 2\mu + 2\mu^2 = \frac12 - \mu.
    \end{equation*}
    Hence, we have that
    \begin{equation}\label{eq:chi2_V_e}
        \chi^2(V_e \,\|\, U_{C_e})
        = \frac{\varepsilon^2}{\mu(1-\varepsilon)^2}\Big(\frac12 - \mu\Big)
        \le \frac{\varepsilon^2}{2\mu(1-\varepsilon)^2}.
    \end{equation}
    Finally, combining~\eqref{eq:g_V_e_KL}, \eqref{eq:KL_V_e_chi2}
    and~\eqref{eq:chi2_V_e}, we obtain that
    \begin{equation*}
        g_{{\bar G}}(V_e) - c_m
        \le \frac{\chi^2(V_e \,\|\, U_{C_e})}{2 \ln 2}
        \le \frac{\varepsilon^2}{4 (\ln 2)\mu (1-\varepsilon)^2},
    \end{equation*}
    which proves~\eqref{eq:perturbation}.

\label{lemma:yes_instances}
    Let $P_{XY}$ be the joint distribution defined in~\eqref{eq:P_XY_explicit}.  Assume that $\omega(G) \ge k$, and let $K$ be a $k$-clique of $G$, and let  \begin{equation}\label{eq:delta_def}
        \delta = \frac{1}{32 N m^4}.
    \end{equation}  Then, the encoder $T_1$ defined in~\eqref{eq:T_1_def} satisfies
    \begin{equation}\label{eq:yes_instance}
        I(Y;T_1) - \frac12 I(X;T_1) > \Lambda_0 + \delta,
    \end{equation}
    where $\Lambda_0$ is defined in Lemma~\ref{lemma:T_0}.
    Consequently, $L_{1/2}(P_{XY}) > \Lambda_0 + \delta$.

<!-- proof-12: body_md -->
The idea of the proof is to compare the average cost of the posteriors of $T_1$ with the cost $c_m$ of the posteriors of $T_0$. Indeed, the posterior $U_K$ yields a gain that is linear in $\varepsilon$, while the perturbed posteriors $V_e$ yield a loss that, by Lemma~\ref{lemma:perturbation}, is quadratic in $\varepsilon$. For our choice of $\varepsilon$ in \eqref{eq:mu_eps_lambda}, the gain exceeds the loss by at least $\delta$. 

Let 
\begin{equation}
 \Psi = \mathbb{E}_{T_1}\Big[g_{{\bar G}}\big(P_{X\mid T_1}(\cdot \mid T_1)\big)\Big] 
 \end{equation} 
be the average cost of the posteriors of $T_1$. 
By Lemma~\ref{lemma:T_1}, and since $g_{{\bar G}}(U_K) = c_k$ and $g_{{\bar G}}(U_{C_e}) = c_m$ 
by Lemma~\ref{lemma:cost_clique}, we have that
 \begin{equation*} 
 \Psi = \lambda\, c_k + \frac{1-\varepsilon}{N} \sum_{e \in E(K)} g_{{\bar G}}(V_e) + \frac1N \sum_{e \in E(G) \setminus E(K)} c_m. 
\end{equation*} 
Since the probabilities in~\eqref{eq:T_1_distribution} sum to $1$, we can write $c_m$ as
\begin{equation}\label{eq:cm_rewritten_}
    c_m = \lambda c_m + \frac{1-\varepsilon}{N}\sum_{e \in E(K)}c_m+\frac{1}{N}\sum_{e \in E(G) \setminus E(K)} c_m.
\end{equation} 
Subtracting \eqref{eq:cm_rewritten_} from $\Psi$, the terms corresponding to $e \in E(G) \setminus E(K)$ cancel out, and we obtain that
\begin{equation}\label{eq:Psi_split} 
\Psi - c_m = -\lambda\,(c_m - c_k) + \frac{1-\varepsilon}{N} \sum_{e \in E(K)} \big(g_{{\bar G}}(V_e) - c_m\big). 
\end{equation} 
The first term on the right-hand side of~\eqref{eq:Psi_split} is the gain due to $U_K$, and the second one is the loss due to the perturbed posteriors $V_e$. We bound the two terms separately. 

For the gain, by Lemma~\ref{lemma:clique_gap}, since $k = m+1$, we get that
\begin{equation}\label{eq:gain}
 \lambda\,(c_m - c_k) \ge \frac{\lambda}{2m^2 \ln 2} > \frac{\lambda}{2m^2}.
\end{equation} 
For the loss, we first note that $\varepsilon$ is small. Indeed, since $m \ge 3$, we have that $\mu = \frac{1}{m(m-1)} \le \frac16$ and $m^2 \ge 9$. Consequently, it follows from \eqref{eq:mu_eps_lambda} that $\varepsilon = \frac{\mu}{6m^2} \le \frac{1}{324}$. 
Moreover, since $1 - \varepsilon > \frac12$, and, since $\ln 2 > \frac23$, we have that
\begin{equation}\label{eq:eps_ln2}
 (1-\varepsilon) \ln 2 > \frac13. 
\end{equation} 
Recalling that $|E(K)| = B$, by Lemma~\ref{lemma:perturbation} and~\eqref{eq:eps_ln2} we have that
\begin{equation}\label{eq:bound_second_term_step_1} 
\frac{1-\varepsilon}{N} \sum_{e \in E(K)} \big(g_{{\bar G}}(V_e) - c_m\big) \le \frac{1-\varepsilon}{N} \cdot B \cdot \frac{\varepsilon^2}{4 \ln 2 \cdot \mu (1-\varepsilon)^2} = \frac{B \varepsilon^2}{4 N \mu \cdot (1-\varepsilon)\ln 2} < \frac{3 B \varepsilon^2}{4 N \mu}.
\end{equation} 
Recalling that $\lambda = B\varepsilon/N$ and $\varepsilon/\mu = 1/(6m^2)$, we obtain from \eqref{eq:bound_second_term_step_1} that
\begin{equation}\label{eq:loss} 
\frac{1-\varepsilon}{N} \sum_{e \in E(K)} \big(g_{{\bar G}}(V_e) - c_m\big) < \lambda \cdot \frac{3\varepsilon}{4\mu} = \lambda \cdot \frac{3}{24 m^2} = \frac{\lambda}{8m^2}.
 \end{equation}
 Substituting~\eqref{eq:gain} and~\eqref{eq:loss} into~\eqref{eq:Psi_split}, we get that
\begin{equation}\label{eq:diff} 
\Psi - c_m < -\frac{\lambda}{2m^2} + \frac{\lambda}{8m^2} = -\frac{3\lambda}{8m^2}. 
\end{equation} 
Since $\lambda = \frac{B\varepsilon}{N} = \frac{B\mu}{6Nm^2}$ and $B\mu = \frac{k(k-1)}{2} \cdot \frac{1}{(k-1)(k-2)} = \frac{k}{2(k-2)} \ge \frac12$ by \eqref{eq:mu_eps_lambda}, we can rewrite \eqref{eq:diff} as 
\begin{align} 
\Psi - c_m &< -\frac{3}{8m^2} \cdot \frac{B\mu}{6Nm^2} = -\frac{B\mu}{16 N m^4}\nonumber\\
&\leq -\frac{1}{32 N m^4}=-\delta, \label{eq:Psi_bound} 
\end{align} 
where the last equality follows from~\eqref{eq:delta_def}.

Finally, by Lemma~\ref{lemma:Lagrangian_expressed_wt_posteriors} and~\eqref{eq:Psi_bound}, we obtain that
\begin{equation*}
I(Y;T_1) - \frac12 I(X;T_1) = g_{{\bar G}}(P_X) - \Psi > g_{{\bar G}}(P_X) - c_m + \delta = \Lambda_0 + \delta, 
\end{equation*} 
which proves~\eqref{eq:yes_instance}. The last statement $L_{1/2}(P_{XY})> \Lambda_0 + \delta$ follows from the definition of $L_{1/2}(P_{XY})$ as a supremum over all encoders.

<!-- proof-14: body_md -->
We start by showing that \textup{\textsc{IB-Lagrangian}}{} is NP-hard. To this end, we show that the
    construction of the instance $(P_{XY},\tau)$ of \textup{\textsc{IB-Lagrangian}}{} described above is a polynomial-time
    reduction from \textup{\textsc{Clique}}{} to \textup{\textsc{IB-Lagrangian}}{}.

    Let $(G,k)$ be an instance of \textup{\textsc{Clique}}{} satisfying~\eqref{eq:assumption_on_G}.
    Following the construction in Section~\ref{sec:reduction}, the reduction
    first constructs the extended graph ${\bar G}$ of $G$, and then computes the
    joint distribution $P_{XY}$ defined in~\eqref{eq:P_XY_explicit} and a
    rational threshold $\tau$ satisfying~\eqref{eq:tau_def}. Finally, it
    outputs the instance $(P_{XY}, \tau)$ of \textup{\textsc{IB-Lagrangian}}{}.

    We first show that $(P_{XY}, \tau)$ can be constructed in time polynomial
    in the size of $(G,k)$. Recall that $m = k-1$, $n = |V(G)|$ and $N = |E(G)|$, and assume that the
    graph $G$ is represented by its adjacency matrix. Then the size of $(G,k)$
    is at least $n$, and, therefore, for our aim it suffices to show that the instance of \textup{\textsc{IB-Lagrangian}}{} can
    be constructed in time polynomial in $n$. We notice that $N \le \binom{n}{2}
    \le n^2$, and, by~\eqref{eq:assumption_on_G},
    $\binom{k}{2} = B \le N \le \binom{n}{2}$, which implies $k \le n$ and
    hence $k-1=m < n$. The graph ${\bar G}$ is obtained from $G$ by adding $m-2$ new
    vertices for each edge of $G$, together with the edges of the cliques
    $C_e$. Therefore ${\bar G}$ has
    \begin{equation*}
        |V({\bar G})| = n + N(m-2) \le n + n^3
    \end{equation*}
    vertices, and, by Lemma~\ref{lemma:gadget},
    \begin{equation*}
        M = |A({\bar G})| = N m(m-1) \le n^4
    \end{equation*}
    oriented edges. Therefore, ${\bar G}$ can be constructed in polynomial time in $n$. The
    joint distribution $P_{XY}$ is a table with $|A({\bar G})| \cdot |V({\bar G})| \le
    n^4(n + n^3)$ entries, and, by~\eqref{eq:P_XY_explicit}, each entry is
    either $0$ or $1/(2M)$. Since $M \le n^4$, each entry is a rational
    number whose numerator and denominator have $O(\log n)$ bits.
    Therefore, $P_{XY}$ has polynomial size and can be constructed in
    polynomial time.

    It remains to compute $\tau$. Note that this does not require computing
    $L_{1/2}(P_{XY})$, but only approximating $\Lambda_0 + \delta/2$, where
    $\Lambda_0 = g_{{\bar G}}(P_X) - c_m$ is the value of the explicit deterministic encoder
    $T_0$ defined in~\eqref{eq:T_0_def}. The value $\delta = 1/(32 N m^4)$, chosen in~\eqref{eq:delta_def}, is a rational number, with
    $O(\log n)$ bits. On the other hand, the value $\Lambda_0$ is in general irrational, but
    it is an explicit expression involving polynomially many logarithms of
    rational numbers. Indeed, we have that $H(P_X) = \log M$ and
    $c_m = \frac12 \log\frac{m}{m-1}$. Moreover, let $d_v$ denote the
    degree of a vertex $v$ in ${\bar G}$. Then $v$ is an endpoint of $2d_v$
    oriented edges in $A({\bar G})$, each of which has probability $1/M$ under $P_X$ and is
    mapped to $v$ with probability $1/2$ by $W_{{\bar G}}$ defined in~\eqref{eq:W_def}. Hence, we have that
    $(W_{{\bar G}} P_X)(v) = 2d_v \cdot \frac{1}{M} \cdot \frac12 = \frac{d_v}{M}$,
    and
    \begin{equation*}
        H(W_{{\bar G}} P_X) = \sum_{v \in V({\bar G}) : d_v > 0} \frac{d_v}{M} \log \frac{M}{d_v}.
    \end{equation*}
    Therefore, by Lemma~\ref{lem:log} in Appendix~\ref{app:log}, an expression of this form
    can be approximated by a rational number within any additive error
    $\eta > 0$ in time polynomial in the bit length of its coefficients and
    of $1/\eta$. By taking $\eta = \delta/12$, which has $O(\log n)$ bits, a
    rational $\tau$ satisfying~\eqref{eq:tau_def} can be computed in
    polynomial time. Hence, the whole reduction runs in polynomial time.

    We now show that the reduction is correct, that is, the instance $(P_{XY}, \tau)$ of \textup{\textsc{IB-Lagrangian}}{} is
    a yes-instance if and only if the corresponding instance $(G,k)$ of \textup{\textsc{Clique}}{} is a yes-instance. Recall that $(P_{XY}, \tau)$ is a yes-instance of \textup{\textsc{IB-Lagrangian}}{} if
    there exists an encoder $T$ of $X$ such that
    $I(Y;T) - \frac12 I(X;T) \ge \tau$, and that $(G,k)$ is a yes-instance
    of \textup{\textsc{Clique}}{} if $\omega(G) \ge k$, that is, $G$ contains a clique of size $k$. 
    
    Note that, by~\eqref{eq:tau_def}, we have that
    \begin{equation}\label{eq:tau_window}
        \Lambda_0 + \frac{5\delta}{12} \le \tau \le \Lambda_0 + \frac{7\delta}{12},
    \end{equation}
    that is, $\tau$ lies strictly between $\Lambda_0$ and
    $\Lambda_0 + \delta$.

    Assume first that $(G,k)$ is a yes-instance, that is,
    $\omega(G) \ge k$.

    and the rightmost inequality in~\eqref{eq:tau_window}, the encoder $T_1$ satisfies
    \begin{align}
        I(Y;T_1) - \frac12 I(X;T_1) &> \Lambda_0 + \delta\nonumber\\
        &\ge \tau - \frac{7\delta}{12} + \delta\nonumber\\
        &= \tau+\frac{5\delta}{12} > \tau.
    \end{align}
    Hence $(P_{XY}, \tau)$ is a yes-instance of \textup{\textsc{IB-Lagrangian}}{}. Moreover,
    $L_{1/2}(P_{XY}) > \tau + \frac{5\delta}{12}$.

    Assume now that $(G,k)$ is a no-instance, that is, $\omega(G) < k$.
    Then, by Lemma~\ref{lemma:no_instance} and the leftmost inequality in~\eqref{eq:tau_window}, every
    encoder $T$ of $X$ satisfies
    \begin{equation*}
        I(Y;T) - \frac12 I(X;T) \le \Lambda_0 \le \tau - \frac{5\delta}{12} < \tau.
    \end{equation*}
    Hence, no encoder reaches the threshold $\tau$, and $(P_{XY}, \tau)$ is a
    no-instance of \textup{\textsc{IB-Lagrangian}}{}. Moreover,
    $L_{1/2}(P_{XY}) \le \tau - \frac{5\delta}{12}$.

    Therefore, we have shown that the reduction is correct, and, since \textup{\textsc{Clique}}{} is NP-hard, also
    when restricted to instances satisfying~\eqref{eq:assumption_on_G},
    \textup{\textsc{IB-Lagrangian}}{} is NP-hard.

    We now prove the inapproximability result. The previous argument shows that $L_{1/2}(P_{XY}) > \tau + \frac{5\delta}{12}$ if $(G,k)$ is a yes-instance of \textup{\textsc{Clique}}{}, and $L_{1/2}(P_{XY}) \le \tau - \frac{5\delta}{12}$ if $(G,k)$ is a no-instance.  Hence, any approximation of $L_{1/2}(P_{XY})$ within
    an additive error smaller than $\frac{5\delta}{12}$ allows to decide \textup{\textsc{Clique}}{} in polynomial time. Suppose, for the sake of contradiction, that there
    exists a polynomial-time algorithm that, given $P_{XY}$, outputs a
    value $v$ such that
    \begin{equation*}
        \big|v - L_{1/2}(P_{XY})\big| \le \frac{1}{216 M^{3/2}}.
    \end{equation*}
    By Lemma~\ref{lemma:delta_vs_M}, we have that
    $\frac{1}{216 M^{3/2}} \le \frac{\delta}{3} < \frac{5\delta}{12}$.
    Hence, if $\omega(G) \ge k$, then
    $v \ge L_{1/2}(P_{XY}) - \frac{\delta}{3} > \tau$, while, if
    $\omega(G) < k$, then
    $v \le L_{1/2}(P_{XY}) + \frac{\delta}{3} < \tau$. Thus, running the
    reduction, then the algorithm, and comparing $v$ with $\tau$ would
    decide \textup{\textsc{Clique}}{} in polynomial time, which is impossible unless
    $\mathrm{P} = \mathrm{NP}$.

Let $\beta \in \mathbb{Q} \cap (0,1)$ be fixed, and let
    $\kappa_\beta = 2\min\{\beta, 1-\beta\}$. Given a rational joint
    distribution $P_{XY}$, one can construct in polynomial time a rational

    that

        L_\beta(\tilde P_{XZ}) = \kappa_\beta\, L_{1/2}(P_{XY}).

    $L_\beta(P_{XY})$ within an additive error of

We first prove~\eqref{eq:beta_scaling}. Let 
\begin{equation}\label{eq:rho_def}
    \rho_\beta = \beta - \frac{\kappa_\beta}{2}\quad\text{and}\quad e_\beta =1 - \kappa_\beta - \rho_\beta.
\end{equation}
One can see that $\kappa_\beta, \rho_\beta$ and $e_\beta$ are non-negative and satisfy $\kappa_\beta+ \rho_\beta+e_\beta=1$. Indeed, the equality $\kappa_\beta + \rho_\beta + e_\beta = 1$ holds by definition. Moreover,
if $\beta \le \frac12$, then we have that $\kappa_\beta = 2\beta$, $\rho_\beta = 0$ and
$e_\beta = 1 - 2\beta \ge 0$, while, if $\beta \ge \frac12$, then
$\kappa_\beta = 2(1-\beta)$, $\rho_\beta = 2\beta - 1 \ge 0$ and
$e_\beta = 0$. Hence, $(\kappa_\beta, \rho_\beta,e_\beta)$ is a probability distribution.

Let $J$ be a
random variable independent of $(X,Y)$ taking values in $\{0,1,2\}$ with probabilities $\kappa_\beta$, $\rho_\beta$ and $e_\beta$, respectively, and let $Z$ be the random variable on $\mathcal{Z}= \big(\{0\} \times \mathcal{Y}\big) \cup \big(\{1\} \times \mathcal{X}\big) \cup \{\bot\}$, where $\bot$ is a new symbol, defined as
\begin{equation}\label{eq:Z_def}
        Z =
        \begin{cases}
            (0,Y), & \text{if } J = 0,\\
            (1,X), & \text{if } J = 1,\\
            \bot, & \text{if } J = 2.
        \end{cases}
\end{equation}
In other words, the random variable $Z$ reveals $Y$ with probability $\kappa_\beta$, reveals $X$
with probability $\rho_\beta$, and reveals nothing with probability
$e_\beta$. 

Let $\tilde P_{XZ}$ be the joint distribution of $(X,Z)$. Since $Z$ replaces $Y$ while $X$ is the same random variable as in $(X,Y) \sim P_{XY}$, the marginal distribution of $X$ under $\tilde P_{XZ}$ remains $P_X$. Consequently, since an encoder of $X$  is a channel $P_{T\mid X}$ from $\mathcal{X}$ to $\mathcal{T}$, every encoder of $X$ for $P_{XY}$ is also an encoder of $X$ for $\tilde P_{XZ}$, and conversely. 

Let $P_{T\mid X}$ be an encoder of $X$ inducing $T$. Since $J$ is independent of $(X,Y,T)$ and it is a function of $Z$, by applying the chain rule we obtain that
\begin{align}
    I(Z;T) &=I(Z,J;T)\nonumber\\&= I(J;T)+I(Z;T\mid J)\nonumber\\
    &= \kappa_\beta I(Y;T)+\rho_\beta I(X;T),\label{eq:I_Z_T}
\end{align}
where the last equality follows since $I(J;T) = 0$, and, given $J$, the variable $Z$ is $Y$, $X$ or a
constant, with probabilities $\kappa_\beta$, $\rho_\beta$ and $e_\beta$.

Now, subtracting $\beta I(X;T)$ from both sides of~\eqref{eq:I_Z_T}, and using $\beta - \rho_\beta = \kappa_\beta/2$ by~\eqref{eq:rho_def}, we get that
\begin{equation}\label{eq:beta_scaling_encoder}
    I(Z;T) - \beta I(X;T)
    = \kappa_\beta \Big( I(Y;T) - \frac12 I(X;T) \Big).
\end{equation}
Since the equality~\eqref{eq:beta_scaling_encoder} holds for any encoder, taking the supremum over all encoders proves~\eqref{eq:beta_scaling}.


We now show that \textup{\textsc{IB-Lagrangian}}{}$_\beta$ is NP-hard by reducing \textup{\textsc{IB-Lagrangian}}{} to it.
Given an instance $(P_{XY}, \tau)$ of \textup{\textsc{IB-Lagrangian}}{}, consider the instance
$(\tilde P_{XZ}, \kappa_\beta \tau)$ of \textup{\textsc{IB-Lagrangian}}{}$_\beta$, where $Z$ is defined in~\eqref{eq:Z_def}.
By~\eqref{eq:beta_scaling_encoder}, there exists an encoder $T$ of $X$ with
$I(Y;T) - \frac12 I(X;T) \ge \tau$ if and only if there exists an
encoder $T$ of $X$ with $I(Z;T) - \beta I(X;T) \ge \kappa_\beta \tau$,
that is, $(P_{XY}, \tau)$ is a yes-instance of \textup{\textsc{IB-Lagrangian}}{} if and only if
$(\tilde P_{XZ}, \kappa_\beta \tau)$ is a yes-instance of \textup{\textsc{IB-Lagrangian}}{}$_\beta$.
Therefore, to complete the reduction, it remains to show that the instance
$(\tilde P_{XZ}, \kappa_\beta \tau)$ can be computed in polynomial time. Since $|\mathcal{Z}| = |\mathcal{X}| + |\mathcal{Y}| + 1$, the table
$\tilde P_{XZ}$ has polynomially many entries. Moreover, since $\beta$ is a fixed rational number, we have that $\kappa_\beta$,
$\rho_\beta$ and $e_\beta$ are rational numbers. Hence, by~\eqref{eq:Z_def}, each entry of
$\tilde P_{XZ}$ is a fixed rational multiple of an entry of $P_{XY}$ or
of $P_X$. Thus, $\tilde P_{XZ}$  and the rational threshold $\kappa_\beta \tau$ can
be computed in polynomial time. Therefore, by Theorem~\ref{thm:IBL_hard},  \textup{\textsc{IB-Lagrangian}}{}$_\beta$ is NP-hard.
 
Finally, suppose that a polynomial-time algorithm approximates $L_\beta$
within an additive error of $\kappa_\beta / (216|\mathcal{X}|^{3/2})$. Running
it on $\tilde P_{XZ}$, which has the same input alphabet $\mathcal{X}$ as
$P_{XY}$, and dividing its output by $\kappa_\beta$, we obtain,
by~\eqref{eq:beta_scaling}, an approximation of $L_{1/2}(P_{XY})$ within
an additive error of $1/(216|\mathcal{X}|^{3/2})$. By
Theorem~\ref{thm:IBL_hard}, this is impossible unless
$\mathrm{P} = \mathrm{NP}$.

\label{remark:beta_reale}
   If $\beta$ is irrational, then $\kappa_\beta$, $\rho_\beta$ and
    $e_\beta$ may be irrational, and the joint distribution $\tilde P_{XZ}$ constructed in the proof of Proposition~\ref{prop:other_fixed_beta} is not a valid
    rational distribution. Nevertheless, we can still extend Proposition~\ref{prop:other_fixed_beta},
    with a different constant, to every fixed real $\beta \in (0,1)$ that
    is computable in polynomial time, that is, such that a rational
    number within distance $2^{-r}$ of $\beta$ can be computed in time
    polynomial in $r$.

    The idea is to apply the construction of~\eqref{eq:Z_def} with a
    rational multiplier $b$ close to $\beta$, chosen depending on the
    instance. Replacing $\beta$ with $b$ changes the objective of every
    encoder $T$ by $|\beta - b|\, I(X;T) \le |\beta - b| \log |\mathcal{X}|$. Hence, we have that
    \begin{equation*}
        |L_\beta(\tilde P_{XZ}) - L_b(\tilde P_{XZ})| \le |\beta - b| \log |\mathcal{X}|.
    \end{equation*}
    The instances of Theorem~\ref{thm:IBL_hard} are separated from the
    threshold by a margin $\gamma = \frac{5\delta}{12}$, which becomes
    $\kappa_b \gamma$ after the construction described in the proof of Proposition~\ref{prop:other_fixed_beta}. Therefore, if $b$ is close
    enough to $\beta$ that $|\beta - b| \log |\mathcal{X}|$ is a small fraction of
    $\kappa_b \gamma$, the threshold $\kappa_b \tau$ still correctly separates yes-
    and no-instances. Note that the required precision on $b$ depends on
    $|\mathcal{X}|$: the larger $|\mathcal{X}|$, the closer $b$ must be to $\beta$.
    However, by Lemma~\ref{lemma:delta_vs_M}, $\gamma$ is of order
    $|\mathcal{X}|^{-3/2}$, so the required precision is of order
    $(|\mathcal{X}|^{3/2} \log |\mathcal{X}|)^{-1}$, which corresponds to $O(\log |\mathcal{X}|)$
    bits. Hence, $b$ can be computed in polynomial time, and the arguments
    of Proposition~\ref{prop:other_fixed_beta} show that for every fixed polynomial-time computable real $\beta\in(0,1)$ \textup{\textsc{IB-Lagrangian}}{}$_\beta$ is NP-hard
    and that $L_\beta$ cannot be approximated in polynomial time within an
    additive error of order $|\mathcal{X}|^{-3/2}$, unless
    $\mathrm{P} = \mathrm{NP}$.

We first show that $R^*$ depends only on $N=|E(G)|$ and $k$. We recall that
    $m = k-1$, $B = k(k-1)/2$ and, by Lemma~\ref{lemma:gadget},
    $M = N m(m-1)$. Moreover, by~\eqref{eq:p_def}
    and~\eqref{eq:mu_eps_lambda}, we have that $p = 1/(12m^2)$, $\mu = 1/(m(m-1))$,
    $\varepsilon = 2\mu p$ and $\lambda = B\varepsilon/N$, and,
    by~\eqref{eq:V_e_def}, $a_0 = \mu/(1-\varepsilon)$ and
    $a_1 = \mu(1-p)/(1-\varepsilon)$. Hence, all the quantities appearing
    in~\eqref{eq:R_star_def} depend only on $N$ and $k$. We also note that
    $a_0, a_1 > 0$, since $p < 1$ and $\varepsilon < 1$, so that the
    logarithms in~\eqref{eq:R_star_def} are well defined.

    \begin{equation}\label{eq:I_X_T1}
        I(X;T_1) = H(P_X) - \mathbb{E}_{T_1}\Big[H\big(P_{X\mid T_1}(\cdot \mid T_1)\big)\Big].

    $H(P_X) = \log M$. For the second term, recall from

        \item $U_K$, with probability $\lambda$;
        \item $V_e$ for each $e \in E(K)$, with probability $(1-\varepsilon)/N$;
        \item $U_{C_e}$ for each $e \in E(G) \setminus E(K)$, with
        probability $1/N$.
    \end{itemize}

    the $m(m-1)$ oriented edges of $C_e$, and $H(U_{C_e}) = \log(m(m-1))$.
    While, by~\eqref{eq:V_e_def}, $V_e$ takes the value $a_0$ on the

    every $e \in E(K)$, it holds that
    \begin{equation*}
        H(V_e) = \big(m(m-1)-2\big)\, a_0 \log\frac{1}{a_0} + 2 a_1 \log\frac{1}{a_1}.
    \end{equation*}

We start by showing that \textup{\textsc{IB-Threshold}}{} is NP-hard. To this end, we show that
    the construction of the instance $(P_{XY}, R, \vartheta)$ described
    above is a polynomial-time reduction from \textup{\textsc{Clique}}{} to \textup{\textsc{IB-Threshold}}{}.

    Let $(G,k)$ be an instance of \textup{\textsc{Clique}}{} satisfying~\eqref{eq:assumption_on_G}.
    The reduction first constructs the extended graph ${\bar G}$ and the joint
    distribution $P_{XY}$ defined in~\eqref{eq:P_XY_explicit}, as in the
    proof of Theorem~\ref{thm:IBL_hard}. Then, it computes a rational
    number $\tilde R$ with $|\tilde R - R^*| \le \frac{\delta}{4}$, the
    budget $R$ in~\eqref{eq:R_def}, and a rational threshold $\vartheta$
    satisfying~\eqref{eq:theta_def}. Finally, it outputs the instance
    $(P_{XY}, R, \vartheta)$ of \textup{\textsc{IB-Threshold}}{}.

    We first show that $(P_{XY}, R, \vartheta)$ can be constructed in time
    polynomial in the size of $(G,k)$. As in the proof of
    Theorem~\ref{thm:IBL_hard}, it suffices to show that the construction
    runs in time polynomial in $n=|V(G)|$. We have already shown that ${\bar G}$ and
    $P_{XY}$ can be constructed in polynomial time in the proof of Theorem~\ref{thm:IBL_hard}. Thus, it remains to show that
    $R$ and $\vartheta$ can be computed in polynomial time. By~\eqref{eq:R_star_def}, $R^*$ is a linear
    combination of the five logarithms $\log M$, $\log(k(k-1))$,
    $\log(m(m-1))$, $\log(1/a_0)$ and $\log(1/a_1)$, with rational
    coefficients. Since $M \le n^4$ and $k, m \le n$, all these
    coefficients and the arguments of the logarithms are rational numbers
    with $O(\log n)$ bits. By Lemma~\ref{lem:log} in Appendix~\ref{app:log},
    $R^*$ can be approximated within additive error
    $\frac{\delta}{4}$ in polynomial time. Consequently, $\tilde R$ and $R$ can be computed in polynomial time. Finally,
    $\frac{R}{2} + \Lambda_0 + \frac{3\delta}{8}$ is the sum of the rational
    number $\frac{R}{2} + \frac{3\delta}{8}$ and of $\Lambda_0$, which, as
    shown in the proof of Theorem~\ref{thm:IBL_hard}, is an explicit
    expression involving polynomially many logarithms of rational numbers
    with $O(\log n)$ bits. Hence, again by Lemma~\ref{lem:log} in Appendix~\ref{app:log},
    a rational $\vartheta$ satisfying~\eqref{eq:theta_def} can be computed
    in polynomial time. Therefore, the whole reduction runs in polynomial
    time.

    We now show that the reduction is correct, that is, the instance
    $(P_{XY}, R, \vartheta)$ of \textup{\textsc{IB-Threshold}}{} is a yes-instance if and only if
    $(G,k)$ is a yes-instance of \textup{\textsc{Clique}}{}. Recall that
    $(P_{XY}, R, \vartheta)$ is a yes-instance of \textup{\textsc{IB-Threshold}}{} if there exists an
    encoder $T$ of $X$ such that $I(X;T) \le R$ and
    $I(Y;T) \ge \vartheta$, and that $(G,k)$ is a yes-instance of \textup{\textsc{Clique}}{} if $\omega(G)\geq k$. Note that, by~\eqref{eq:theta_def}, we have that
    \begin{equation}\label{eq:theta_window}
        \frac{R}{2} + \Lambda_0 + \frac{\delta}{4}
        \le \vartheta \le
        \frac{R}{2} + \Lambda_0 + \frac{\delta}{2}.
    \end{equation}

    Assume first that $(G,k)$ is a yes-instance, that is,
    $\omega(G) \ge k$. We show that the encoder $T_1$ satisfies that $I(X;T_1)\leq R$ and $I(Y;T_1)\geq \vartheta$, and therefore $(P_{XY}, R, \vartheta)$ is a yes-instance. By Lemma~\ref{lemma:witness_rate}, $R^* = I(X;T_1)$, and in
    particular $R^* \ge 0$, since mutual information is non-negative. Since
    $|\tilde R - R^*| \le \frac{\delta}{4}$, we have that
    \begin{equation}\label{eq:tilde_R_eq}
        R^* \le \tilde R + \frac{\delta}{4} \le R^* + \frac{\delta}{2}.
    \end{equation}
    In particular, since  $\tilde R + \frac{\delta}{4} \ge R^* \ge 0$, the
    maximum in~\eqref{eq:R_def} equals $\tilde R + \frac{\delta}{4}$, that is, $R = \tilde R + \frac{\delta}{4}$. Therefore, from~\eqref{eq:tilde_R_eq} we have that
    \begin{equation}\label{eq:R_window}
        R^* \le R \le R^* + \frac{\delta}{2}.
    \end{equation}
    The leftmost inequality in~\eqref{eq:R_window} gives $I(X;T_1) = R^* \le R$,
    so the encoder $T_1$ satisfies the rate constraint. Moreover, by
    Lemma~\ref{lemma:yes_instances} and $R^*\geq R-\delta/2$ from the rightmost inequality
    in~\eqref{eq:R_window}, we get that
    \begin{align}
        I(Y;T_1) &> \frac12 I(X;T_1) + \Lambda_0 + \delta
        = \frac{R^*}{2} + \Lambda_0 + \delta\nonumber \\
        &\ge \frac12\Big(R - \frac{\delta}{2}\Big) + \Lambda_0 + \delta
        = \frac{R}{2} + \Lambda_0 + \frac{3\delta}{4}.\label{eq:IBT_yes_instance}
    \end{align}
    By the rightmost inequality in~\eqref{eq:theta_window}, \eqref{eq:IBT_yes_instance} gives
    \begin{equation*}
        I(Y;T_1) > \vartheta + \frac{\delta}{4} > \vartheta.
    \end{equation*}
    Hence, $T_1$ satisfies both $I(X;T_1) \le R$ and
    $I(Y;T_1) \ge \vartheta$, and $(P_{XY}, R, \vartheta)$ is a yes-instance
    of \textup{\textsc{IB-Threshold}}{}. Moreover, $\mathrm{IB}(P_{XY}, R) \ge I(Y;T_1) > \vartheta + \frac{\delta}{4}$.

    Assume now that $(G,k)$ is a no-instance, that is, $\omega(G) < k$.
    Since $R \ge 0$ by~\eqref{eq:R_def}, Lemma~\ref{lemma:no_instance} and

    \begin{equation*}
        \mathrm{IB}(P_{XY}, R) \le \frac{R}{2} + \Lambda_0 \le \vartheta - \frac{\delta}{4} < \vartheta.
    \end{equation*}
    Hence, every encoder $T$ of $X$ with $I(X;T) \le R$ satisfies
    $I(Y;T) \le \mathrm{IB}(P_{XY}, R) < \vartheta$. Thus, $(P_{XY}, R, \vartheta)$ is a no-instance of
    \textup{\textsc{IB-Threshold}}{}.

    Therefore, we have shown that the reduction is correct, and, since
    \textup{\textsc{Clique}}{} is NP-hard, also when restricted to instances
    satisfying~\eqref{eq:assumption_on_G}, \textup{\textsc{IB-Threshold}}{} is NP-hard.

    We now prove the inapproximability result. The previous argument shows
    that $\mathrm{IB}(P_{XY}, R) > \vartheta + \frac{\delta}{4}$ if $(G,k)$ is a
    yes-instance of \textup{\textsc{Clique}}{}, and $\mathrm{IB}(P_{XY}, R) \le \vartheta - \frac{\delta}{4}$
    if $(G,k)$ is a no-instance. Hence, any approximation of
    $\mathrm{IB}(P_{XY}, R)$ within an additive error smaller than
    $\frac{\delta}{4}$ lies on the same side of $\vartheta$ as
    $\mathrm{IB}(P_{XY}, R)$, and thus decides \textup{\textsc{Clique}}{}. Suppose, for
    the sake of contradiction, that there exists a polynomial-time
    algorithm that, given $P_{XY}$ and $R$, outputs a value $v$ such that
    \begin{equation*}
        \big|v - \mathrm{IB}(P_{XY}, R)\big| \le \frac{1}{576 M^{3/2}}.
    \end{equation*}
    By Lemma~\ref{lemma:delta_vs_M},
    $\frac{1}{576 M^{3/2}} \le \frac{\delta}{8} < \frac{\delta}{4}$.
    Hence, if $\omega(G) \ge k$, then
    $v \ge \mathrm{IB}(P_{XY}, R) - \frac{\delta}{8} > \vartheta$, while, if
    $\omega(G) < k$, then
    $v \le \mathrm{IB}(P_{XY}, R) + \frac{\delta}{8} < \vartheta$. Thus, running the
    reduction, then the algorithm, and comparing $v$ with $\vartheta$ would
    decide \textup{\textsc{Clique}}{} in polynomial time, which is impossible unless
    $\mathrm{P} = \mathrm{NP}$.

    \frac{B\varepsilon^2}{4 \ln 2 \cdot N\mu(1-\varepsilon)}
    = \frac{\lambda}{\ln 2} \cdot \frac{\varepsilon}{4\mu(1-\varepsilon)}
    = \frac{\lambda}{m^2 \ln 2} \cdot \frac{1}{8(1-\varepsilon)}.
\end{equation*}
Hence, it follows that
\begin{equation*}
    \Psi - c_m \le -\frac{\lambda}{m^2 \ln 2}
    \Big(\frac12 - \frac{1}{8(1-\varepsilon)}\Big).
\end{equation*}
Since $m \ge 3$, we have $\varepsilon \le \frac{1}{108}$, so the factor in
parentheses is at least $\frac12 - \frac{27}{214} = \frac{40}{107}$, which
is larger than $\frac{\ln 2}{2}$, because $\ln 2 < \frac{7}{10}$.
Therefore $\Psi - c_m < -\frac{\lambda}{2m^2}$, and, since
$B\mu \ge \frac12$, we finally get that
\begin{equation*}
    \Psi - c_m < -\frac{B\mu}{4Nm^4} \le -\frac{1}{8Nm^4}.
\end{equation*}
Thus, Lemma~\ref{lemma:yes_instances} holds with
$\delta = \frac{1}{8Nm^4}$, which is four times larger than
in~\eqref{eq:delta_def}. Repeating the computation in the proof of
Lemma~\ref{lemma:delta_vs_M}, we obtain
that $\delta \ge \frac{1}{18 M^{3/2}}$, and the rest of the proofs is
unchanged. This improves the additive errors in
Theorems~\ref{thm:IBL_hard} and~\ref{thm:IBT_hard} from
$\frac{1}{216|\mathcal{X}|^{3/2}}$ and $\frac{1}{576|\mathcal{X}|^{3/2}}$ to
$\frac{1}{54|\mathcal{X}|^{3/2}}$ and $\frac{1}{144|\mathcal{X}|^{3/2}}$, respectively.

\label{lem:log}
    There is an algorithm that, given rational numbers
    $\beta_0, \dots, \beta_s$, positive rational numbers
    $z_1, \dots, z_s$ and a rational number $\eta > 0$, outputs a rational
    number $\tilde F$ such that
    \begin{equation*}
        \Big|\tilde F - \Big(\beta_0 + \sum_{i=1}^s \beta_i \log z_i\Big)\Big| \le \eta,
    \end{equation*}
    in time polynomial in the total bit length of the input.
