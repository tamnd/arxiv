<!-- definition-1: body_md -->
\label{def:qcol}
Given any $d\in \mathbb{N}$, a \emph{quantum $n$-coloring} of $G$ assigns to each vertex $u\in V(G)$ a
family $\{P_{u,i}\}_{i=1}^{n}\subset \mathbb{C}^{d\times d}$ of projections such that
\[
\text{(completeness)}\quad
\sum_{i=1}^{n}P_{u,i}=I\ \ \ \forall u\in V(G),
\qquad
\text{(orthogonality)}\quad
P_{u,i}P_{v,i}=0\ \text{ whenever }u\sim v .
\]
If there exists a quantum $n$-coloring of $G$, for some $d\in\mathbb{N}$, we say that $G$ is quantum $n$-colorable.
<!-- end definition-1 -->
<!-- lemma-1: body_md -->
\label{lem:reduction}
For every graph $G$ and every $n\ge4$,
\[
G\xrightarrow{\,q\,} K_3
\iff
G'\xrightarrow{\,q\,} K_n .
\]
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
In the following we are using addition of numbers modulo $3$ with representatives $\{1,2,3\}$. Let $\{P_{u,i}\}_{u\in V(G), i\in\{1,2,3\}}$ form a quantum $3$-coloring of $G$. Let
\[
E_{(u,j),i}
=
\begin{cases}
P_{u,\,i+ j}, & i\in\{1,2,3\},\\
0, & 3<i\le n,
\end{cases}
\qquad
F_{\ell,i}
=
\begin{cases}
I, & i=\ell,\\
0, & i\ne\ell.
\end{cases}
\]
We will show that $\{E_{(u,j),i}\}_{(u,j)\in V(G\mathbin{\square} K_3),i\in\{1,2,\dots,n\}}\cup\{F_{\ell,i}\}_{\ell \in \{4,\dots,n\},i\in\{1,2,\dots,n\}}$ forms a quantum $n$-coloring of $G'$.

As $i\mapsto i+ j$ permutes
$\{1,2,3\}$, \textbf{completeness} is immediate:
\[
\sum_{i=1}^{n}E_{(u,j),i}
=
\sum_{i=1}^{3}P_{u,\,i+ j}
=
\sum_{k=1}^{3}P_{u,k}
=
I,
\qquad
\sum_{i=1}^{n}F_{\ell,i}
=
I .
\]

\paragraph{Orthogonality.} Fix a color $i$ and an edge of $G'$. An edge between fibres: $(u,j)\sim(u',j)$ for any $u\sim u'$ gives
\[
E_{(u,j),i}E_{(u',j),i}
=
P_{u,i+ j}P_{u',i+ j}
=
0.
\]
An edge contained in a fibre: $(u,j)\sim(u,j')$ for any $j\ne j'$ has
$i+ j\ne i+ j'$, thus
\[
E_{(u,j),i}E_{(u,j'),i}=P_{u,i+ j}P_{u,i+ j'}=0.
\]
For any other edge in $G'$ and any specified color at least one projection is zero, meaning they trivially are orthogonal. Hence
$\{E_{(u,j),i}\}_{(u,j)\in V(G\mathbin{\square} K_3),i\in\{1,2,\dots,n\}}\cup\{F_{\ell,i}\}_{\ell \in \{4,\dots,n\},i\in\{1,2,\dots,n\}}$ is a quantum $n$-coloring of $G'$, so $G'\xrightarrow{\,q\,} K_n$.
<!-- end proof-1 -->
<!-- proof-2: body_md -->
Let $\{E_{(u,j),i}\}_{(u,j)\in V(G\mathbin{\square} K_3),i\in\{1,2,\dots,n\}} \cup \{F_{\ell,i}\}_{\ell \in \{4,\dots,n\},i\in\{1,2,\dots,n\}}$ be a quantum $n$-coloring of $G'$ with respect to some $\mathbb{C}^{d\times d}$. For each
color $i$ define the projection $R_i:=\sum_{\ell=4}^{n}F_{\ell,i}$. We will show that there exists a color $i^*$ for which \(I-R_{i^*}\) is nonzero. We then recover a $3$-coloring of $G$ from $\{E_{(u,j),i^*}\}_{u\in V(G),j\in \{1,2,3\}}$ on some $\mathbb{C}^{d'\times d'}$.

We show that for any chosen color, the projections of any fibre sum to a projection that is independent of the fibre.\\
Fix $v$. The vertices
\[
(v,1),(v,2),(v,3),4,\dots,n
\]
form a clique in $G'$ of size $n$, so for each color the corresponding $n$ projections
are pairwise orthogonal. Hence
\[
S_i:=\sum_{j=1}^3E_{(v,j),i}+R_i\preceqI.
\]
Summing over the $n$ colors and using completeness at each vertex,
\[
\sum_{i=1}^{n}S_i
=
\sum_{i=1}^{n}
\Bigl(
\sum_{j=1}^{3}E_{(v,j),i}+R_i
\Bigr)
=
3\,I+(n-3)\,I
=
n\,I.
\]
Thus
\[
\sum_{i=1}^n(I - S_i)=0
\]
with every $0\preceq I - S_i$, so each $S_i=I$. Therefore
\begin{equation}
\label{eq:corner}
\sum_{j=1}^{3}E_{(v,j),i}
=
I-R_i
\qquad
\text{for all }v\text{ and }i,
\end{equation}
which is independent of $v$.\\

\noindent We show that $I-R_{i^*} \neq 0$ for some color $i^*$.
\[
\sum_{i=1}^{n}R_i
=
\sum_{\ell=4}^{n}
\sum_{i=1}^{n}F_{\ell,i}
=
\sum_{\ell=4}^{n}I
=
(n-3)I.
\]
Hence
\[
0\prec 3I=\sum_{i=1}^nI-R_i.
\]
Therefore $0\precI-R_{i^\ast}$ for some color $i^\ast$.

\paragraph{Completeness}
Let $\mathcal{H'} = \operatorname{Ran}(I-R_{i^\ast})$ where $d'=\dim(\mathcal{H'}) \leq d$.
Now defining $\{H_{u,i}\}_{u\in V(G), i\in \{1,2,3\}}$ by
\begin{align*}
    H_{u,i} = E_{(u,i),i^*}\big{|}_{\mathcal{H'}}
\end{align*}
it follows that each $H_{u,i}$ is a projection on $\mathcal{H}'$. Consider a linear isometry $\phi:\mathcal{H}'\to \mathbb{C}^{d'}$. Using $\phi$ to define $\{P_{u,i}\}_{u\in V(G), i\in \{1,2,3\}} \subset \mathbb{C}^{d'\times d'}$ by $P_{u,i}=\phi \circ H_{u,i} \circ\phi^{-1}$. It is easy to see that $P_{u,i}$ indeed are projections on $\mathbb{C}^{d'}$, and that $\sum_{i=1}^3 P_{u,i} = I_{d'}$ using~\eqref{eq:corner}.

\paragraph{Orthogonality.}

If $u\sim v$ in $G$, then $(u,i)\sim(v,i)$ in $G'$ for each $i$, so for any color $i\in\{1,2,3\}$,
\[
H_{u,i}H_{v,i} =E_{(u,i),i^*}E_{(v,i),i^*}=0.
\]
It follows $P_{u,i}P_{v,i}=0$ for any $u\sim v$ in $G$. Thus $\{P_{u,i}\}_{u\in V(G),i\in\{1,2,3\}}$ is a quantum $3$-coloring of $G$ in $\mathbb{C}^{d'\times d'}$, so $G\xrightarrow{\,q\,} K_3$.
<!-- end proof-2 -->
<!-- theorem-1: body_md -->
\label{thm:main}
For every fixed integer $n\ge3$ the corresponding quantum $n$-coloring decision problem is undecidable.
<!-- end theorem-1 -->
<!-- proof-3: body_md -->
For $n=3$, this follows from Harris's reduction together with the known
undecidability of perfect-strategy existence~\cite{harris,slofstra}.
For $n\ge4$, the map
\[
G\longmapsto(G\mathbin{\square} K_3)\vee K_{n-3}
\]
is polynomial-time computable and, by Lemma~\ref{lem:reduction}, is a
many-one reduction from quantum $3$-coloring to quantum
$n$-coloring.
<!-- end proof-3 -->
<!-- conjecture-1: body_md -->
For every non-bipartite $H$, quantum $H$-coloring is undecidable.
