		\end{align}
		
		It follows from (\ref{eq:Sijkl-def}) that $S^{I,J}$ is a trace-free symmetric $(0,2)$-tensor. Therefore,
		 $$\left\langle
		 \mathring R(S^{I,J}),
		 S^{I,J}
		 \right\rangle=\left\langle
		 \bar R(S^{I,J}),
		 S^{I,J}
		 \right\rangle.$$
		
		A direct computation using the symmetries of the curvature tensor,
		together with	the first Bianchi identity yields

			=&{\omega }_{si_2\ldots i_p,J}{\omega }_{ti_2\ldots i_p,J}\left\langle \bar R(e_s\odot e_{i_1}),e_t\odot e_{i_1} \right\rangle\nonumber\\

<!-- proof-3: body_md -->
For fixed $i,j,k,l$, if $|\{i,j\}\cap \{k,l\}|=1$, then 
		\begin{equation}\label{eq1}
			(a_i+a_j-a_k-a_l)^2\omega_{ij,kl}^2\le 2\omega_{ij,kl}^2\le\frac83\omega_{ij,kl}^2.
		\end{equation}
		
		If $|\{i,j\}\cap \{k,l\}|=0$, that is, $i,j,k,l$ are pairwise distinct. By the first Bianchi identity,				
		\begin{equation}\label{eq:xyz-bianchi}
			\omega_{ij,kl}+\omega_{ik,lj}+\omega_{il,jk}=0.
		\end{equation}
		Denote
		\[
		\begin{split}
			p&=a_i+a_j-a_k-a_l,\\
			q&=a_i+a_k-a_l-a_j,\\
			r&=a_i+a_l-a_j-a_k.
		\end{split}
		\]
		A direct computation gives
		\begin{equation}\label{eq:pqr}
			\begin{split}
				p^2+q^2+r^2&=4(a_i^2+a_j^2+a_k^2+a_l^2)-(a_i+a_j+a_k+a_l)^2\\
				&\leq4(a_i^2+a_j^2+a_k^2+a_l^2)\leq 4.
			\end{split}
		\end{equation}	
		On the other hand, \eqref{eq:xyz-bianchi} implies
		\[
		\omega_{ij,kl}^2\leq 2(\omega_{ik,lj}^2+\omega_{il,jk}^2),
		\]
		and hence
		\[
		\omega_{ij,kl}^2\leq\frac23(\omega_{ij,kl}^2+\omega_{ik,lj}^2+\omega_{il,jk}^2).
		\]
		The same argument gives
		\[
		\omega_{ik,lj}^2,\omega_{il,jk}^2\leq \frac23(\omega_{ij,kl}^2+\omega_{ik,lj}^2+\omega_{il,jk}^2).
		\]
		Combining this with \eqref{eq:pqr}, we obtain
		\begin{align*}
			p^2\omega_{ij,kl}^2+q^2\omega_{ik,lj}^2+r^2\omega_{il,jk}^2
			&\leq
			\frac23(\omega_{ij,kl}^2+\omega_{ik,lj}^2+\omega_{il,jk}^2)
			(p^2+q^2+r^2)\\
			&\leq
			\frac83(\omega_{ij,kl}^2+\omega_{ik,lj}^2+\omega_{il,jk}^2).
		\end{align*}
		Summing over $\{i,j,k,l\}$ yields
		\[ (a_i+a_j-a_k-a_l)^2\omega_{ij,kl}^2\le \frac83\omega_{ij,kl}^2,\]
		this together with (\ref{eq1}) proves \eqref{eq:key-weight}.

<!-- corollary-5: body_md -->
(Corollary 1.2). Let $(M, g)$ be an $n(\ge 4)$-dimensional complete Riemannian
		manifold with harmonic Weyl tensor. If the curvature operator of the second kind $\mathring{R}$ is$\frac{3(n-1)}{4}$-nonnegative and the Weyl tensor of $(M, g)$ has bounded norm, then M is either globally
		conformally equivalent to a space of positive constant curvature or is isometric to a flat

<!-- proof-5: body_md -->
Riemannian curvature tensor  is divergence free if and only if its Ricci tensor is a Codazzi tensor, and in this case its scalar curvature is constant. Hence the traceless Ricci tensor is harmonic because its Ricci tensor is divergence free. Denote by $E$ its traceless Ricci tensor and by $s$ its scalar curvature.
		
		If $s\equiv0$, then $M$ is flat since $\mathring{R}$ is $\frac{3(n-1)}{4}$-nonnegative.	Otherwise, $s=constant>0$,   i.e.,   $M$ is non-flat. By part $(2)$ of Proposition 4.1 in \cite{Li24}, we have that $Ric(M)\geq\frac{s}{n(n+1)}>0$ because  $\mathring{R}$ with $\frac{3(n-1)}{4}$-nonnegative is $n$-positive. So $M$ is compact by Myers Theorem.
		
		For manifolds with harmonic weyl tensor,  we have
		\[\frac{1}{2}\Delta \left| W \right|^2=\left| \nabla W \right|^2+\frac12g (\operatorname{Ric}_{L} (W),W) \ge 0. \]
		By the maximum principle, from the above we have $\nabla W=0$. According to Derdziński
		and Roter’s result \cite{DR} that Riemannian manifolds with  $\nabla W=0$ are either
		locally conformally flat or locally symmetric, we see that $(M, g)$ is either locally
		conformally flat or locally symmetric.
		Case I, $M$ is locally conformally flat with positive constant scalar curvature. Let ${a_i}$ be the eigenvalue of Ricci tensor, then we obtain that
		\begin{align}\label{laplacian1}
			\left\langle\Delta E,E \right\rangle &=E_{ij}E_{ij,tt}=E_{ij}E_{it,jt}\nonumber\\
			&=E_{ij}(E_{it,tj}+R_{jtsi}E_{st}+R_{jtst}E_{is})\nonumber\\
			&=\frac{n}{n-2}R_{ij}E_{it}E_{jt}-\frac{s}{(n-2)(n-1)}E_{ij}E_{ij}\nonumber\\
			&=\frac{n}{n-2}R_{ij}R_{jt}R_{ti}-\frac{(2n-1)s}{(n-1)(n-2)}R_{ij}R_{ji}+\frac{s^3}{(n-1)(n-2)}\nonumber\\
			&=\frac{n}{n-2}\sum a_i^3-\frac{(2n-1)s}{(n-1)(n-2)}\sum a_i^2+\frac{s^3}{(n-1)(n-2)}\nonumber\\
			&=\frac{n}{n-2}\left(\frac{(2n-1)s}{n(n-1)}\right)^3\left[\sum\left(\frac{n(n-1)a_i}{(2n-1)s}\right)^3-\sum\left(\frac{n(n-1)a_i}{(2n-1)s}\right)^2\right]\nonumber\\
			&+\frac{s^3}{(n-1)(n-2)}.\\\nonumber
		\end{align}		
		It follows from \cite[Lemma 3.3]{FL} and (\ref{laplacian1}) that the minimal points of $\left\langle\Delta E,E \right\rangle$ are $(\frac{s}{n}, \frac{s}{n},\cdots, \frac{s}{n})$ and $(0, \frac{s}{n-1},\cdots, \frac{s}{n-1})$, and that $\left\langle\Delta E,E \right\rangle\geq0$.  Hence  from \begin{equation*}\label{laplacian}
			\frac{1}{2}\Delta \left| E \right|^2=\left| \nabla E \right|^2+\left\langle\Delta E,E \right\rangle,
		\end{equation*} we deduce that the trace-less Ricci tensor $E$ is parallel. 
		Since $(M, g)$ is  locally irreducible by Theorem 1.8 in \cite{Li24}, it follows that $(M, g)$ is isometric to a space of positive constant curvature.
		
		Case II, $M$ is locally symmetric with positive constant scalar curvature. Since $(M, g)$ is  locally irreducible by Theorem 1.8 in \cite{Li24}, $(M, g)$ is an Einstein manifold.		It follows from Theorem B of \cite{NPW23} that $M$ is a rational homology sphere. Wolf \cite[Theorem 1]{Wo} provides a complete classification of compact symmetric spaces that are rational homology spheres. Aside from spheres themselves, the only simply connected example is the space $SU(3)/SO(3)$, which is shown in \cite[Example 4.5]{NPW23} to be $9$-positive. It then follows that the manifold must have positive constant curvature. \\

		\[{{\omega }_{ikI'',J}}{{\omega }_{jlI'',J}}R_{ijkl}={{\omega }_{ikI'',J}}{{\omega }_{jlI'',J}}R_{ilkj}+{{\omega }_{ikI'',J}}{{\omega }_{jlI'',J}}R_{ikjl}\]
		i.e.,
		\[{{\omega }_{ikI'',J}}{{\omega }_{jlI'',J}}R_{ijkl}=\frac{1}{2}{{\omega }_{ijI'',J}}{{\omega }_{klI'',J}}R_{ijkl}.\]
		This together with Proposition~\ref{compute} yields 
		\begin{align*}
			g(\bar{R}({\omega }^{S^2}),{\omega }^{S^2})=&\frac{1}{2}\sum\limits_{ijkl}{g((e_i\bar{\circ}e_j)\omega ,(e_k\bar{\circ }e_l)\omega ))(R_{kijl}+R_{kjil})} \\ 
			=&\frac{p}{2}R_{ij}{\omega }_{iI',J}{\omega }_{jI',J}-\frac{3}{2}\frac{p(p-1)}{2}{\omega }_{ijI'',J}{\omega }_{klI'',J}R_{ijkl}\\

			+{{\omega}_{iI',kJ'}}{{\omega }_{jI',lJ'}})(R_{kijl}+R_{kjil}) \\

			=&\frac{p}{2}R_{ij}{\omega }_{iI',J}{\omega }_{jI',J}-\frac{3}{2}\frac{p(p-1)}{2}{\omega }_{ijI'',J}{\omega }_{klI'',J}R_{ijkl}\\

			&+pq({{\omega }_{iI',lJ'}}{{\omega }_{jI',kJ'}}+{{\omega}_{iI',kJ'}}{{\omega }_{jI',lJ'}})R_{kijl}.

<!-- proposition-7: body_md -->
\label{weighting estimate}
		For a $(p,q)$-double form $\omega$ satisfying Bianchi identity and $S\in S_0^2(V)$,
		\begin{align*}
			&(i)\left| {{\omega }^{S_{0}^{2}}} \right|^2=[\frac{n+2}{2}(p+q)-\frac{n+2}{2n}{{(p+q)}^{2}}+(p+1)q]\left| \omega  \right|^2+pq\left| \boldsymbol{c}\omega  \right|^2-\frac{q}{p+1}|\mathcal{B}\omega|^2,\\
			&(ii){{\left| S\omega  \right|}^{2}}\le \left( p+q+2\min \{p,q\}-\frac{{{\left( p+q \right)}^{2}}}{n} \right){{\left| S \right|}^2}\left| \omega  \right|^2.
		\end{align*}

<!-- proof-10: body_md -->
(i) The norm $\left| {\omega }^{S_0^2}\right|^2$ can be computed by taking $R_{ijkl}=\delta_{ik}\delta_{jl}-\delta_{il}\delta_{jk}$ in Proposition~\ref{proposition2}. We get
		$$\left| {{\omega }^{S_{0}^{2}}} \right|^2=[\frac{n+2}{2}(p+q)-\frac{n+2}{2n}{{(p+q)}^{2}}+pq]\left| \omega  \right|^2+pq\left| \boldsymbol{c}\omega  \right|^2+pq{\omega}_{iI',jJ'}{\omega }_{jI',iJ'}$$
		and notice that from (\ref{Bianchi})
		\begin{align*}
			\left| \mathcal{B}\omega  \right|^2 =(p+1)\left| \omega  \right|^2-p(p+1){\omega }_{iI',jJ'}{\omega }_{jI',ij_2....j_q}.  
		\end{align*}
		
		(ii) For $S\in S_{0}^{2}(V)$, there is an orthonormal basis $e_1,\ldots ,e_n$ for V and $a_1,\ldots ,a_n\in \mathbb{R}$ such that $Se_i=a_ie_i$ for $i=1,\ldots n$. It follows that
		\[{{\left( S\omega  \right)}_{I,J}}=\left( \sum\limits_{i}{c_ia_i} \right){\omega}_{I,J},\]
		where $c_i=\left\{ \begin{aligned}
			& 0\qquad \text{for} \quad i\in {\{1,\ldots n\}}/{I\cup J}\; \\ 
			& 1\qquad \text{for} \quad i\in {I\cup J}/{I\cap J}\; \\ 
			& 2\qquad \text{for} \quad i\in I\cap J 
		\end{aligned} \right.$.
		
		Now we estimate maximum of $\left( \sum\limits_{i}{c_ia_i} \right)^2$ under constraints
		$\sum\limits_{i}{a_i}=0$.
		If we set $\left| I\cap J \right|=r$, then $\sum\limits_{i}{c_i}=p+q$ and $\sum\limits_{i}{c_i^2}=p+q+2r$. Choose $\mu=\frac{p+q}{n}$, then by  the Cauchy-Schwarz inequality
		
		\begin{align*}
			\left( \sum\limits_{i}{c_ia_i} \right)^2=&\left( \sum\limits_{i}{(c_i-\mu )a_i }\right)^2\le \left( \sum\limits_{i}{(c_i-\mu )^2} \right)\left( \sum\limits_{i}{a_i^2} \right) \\ 
			& =\left(\sum\limits_{i}{(c_i-\mu )^2} \right)|S|^2\\ 
			& =\left( p+q+2r-\frac{(p+q)^2}{n} \right)|S|^2\\ 
			& \le \left(p+q+2\min \{p,q\}-\frac{(p+q)^2}{n}\right)|S|^2.
		\end{align*}
		Therefore,
		\[|S\omega|^2=\sum_{I,J}\left( \sum\limits_{i}{c_ia_i} \right)^2{\omega}_{I,J}^2\le \left( p+q+2\min \{p,q\}-\frac{{{\left( p+q \right)}^{2}}}{n} \right){{\left| S \right|}^2}\left| \omega  \right|^2.\]
