<!-- lemma-3: body_md -->
\label{lemma:frequency_count}
  For $\eta \in \mathbb{R}$ sufficiently small, $0 < \delta < 1 $, and $\lambda\in \mathbb{R}$, we have
  \begin{equation*}
    \# \big\{m, n\sim N : m \neq n, \,\big|\phi_\eta(m/N) - \phi_\eta(n/N) - \lambda\big|\leqslant \delta\big\} \ll \delta N^2 + N^{2/3}.                                            
  \end{equation*}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
For $x\asymp 1 $, we record that $\phi'_\eta(x) \asymp \phi_\eta''(x)\asymp 1 $ since $\eta $ is sufficiently small. A direct computation also gives $(\phi_\eta''/(\phi_\eta')^2)'(x) \asymp 1 $.
  
  Denote $\mathcal{S} = \{(m, n)\sim N :  m \neq n,\, |\phi_\eta(n/N) -  \phi_\eta(m/N) - \lambda|\leqslant \delta\} $ and
  $\mathcal{S}_m = \{(m, n)\in \mathcal{S}\} $. We have that $|\mathcal{S}_m| \ll 1 + \delta N $, so  $|\mathcal{S}|\ll N + \delta N^2 $, which is
  sufficient for $\delta \gg 1/N $. 
  
  Suppose that $\delta\leqslant c/N $ for the remainder of this proof, where $c>0$ is sufficiently small. By symmetry, we may assume that $\lambda \geqslant 0 $. We may also suppose that $1/N \ll \lambda \ll 1$, for otherwise we are forced into the case $n = m $. It remains to bound the number of $m $ for which $\mathcal{S}_m $ is nonempty, since the number of $n$ corresponding to each $m$ is $\ll 1 + \delta N \ll 1$.

  Let $g_\lambda(x) = \phi_\eta^{-1}(\phi_\eta(x) + \lambda) $. It can be checked by implicit differentiation and the mean value theorem that for $x\sim 1 $ we have
  \begin{equation*}
    g_\lambda(x) - x \asymp g_\lambda'(x) - 1 \asymp g_\lambda''(x) \asymp \lambda.
  \end{equation*}
  That $|\phi_\eta(n/N) - \phi_\eta(m/N) - \lambda| \leqslant  \delta  $ implies $n = Ng_\lambda(m/N) + O(\delta N) $ by monotonicity, so
  \begin{equation*}
    |\mathcal{S}|\ll  
  \origsum
  \IfValueT{#1}{_{\substack{#1}}}
  \IfValueT{#2}{^{#2}}
_{m\sim N } \text{\usefont{U}{bbold}{m}{n}1}_{  \| N g_\lambda(m/N)  \|\ll \delta N}
    =   
  \origsum
  \IfValueT{#1}{_{\substack{#1}}}
  \IfValueT{#2}{^{#2}}
_{m\sim N } \text{\usefont{U}{bbold}{m}{n}1}_{  \| N (g_\lambda(m/N) - m/N)  \|\ll \delta N}.
  \end{equation*}

  Note that $N(g_\lambda(m/N) - m/N) \asymp N\lambda$ and the corresponding values are spaced by $\gg \lambda$ for $m \sim N$. For $\lambda\leqslant N^{-1/3} $, the desired result follows since
  \begin{equation*}
    |\mathcal{S}| 
    \ll (1 + N\lambda) \biggl( 1 + \frac{\delta N}{\lambda} \biggr)
    \ll \delta N^2 + \lambda N \ll \delta N^2 + N^{2/3}.
  \end{equation*}
  
  Suppose that $\lambda > N^{-1/3} $ from now on. By \cite[Proposition 3.3]{2024arXiv2401.13981}
  (with $\delta $ in the reference taken to be $\delta_1 = \delta N+ (\lambda/N)^{1/3} $ by positivity) we get
  \begin{equation*}
     
  \origsum
  \IfValueT{#1}{_{\substack{#1}}}
  \IfValueT{#2}{^{#2}}
_{m\sim N } \text{\usefont{U}{bbold}{m}{n}1}_{  \| N (g_\lambda(m/N) - m/N)  \|\ll \delta N}
    \ll \delta N^2 + (\lambda/N)^{1/3}N + \sqrt{\frac{N\lambda}{\delta_1}} + \sqrt{\frac{N\delta_1}{\lambda}}.
  \end{equation*}
  It remains to check that the bound is acceptable. Note from $\lambda \ll 1$ that
  \begin{equation*}
    \sqrt{\frac{N\lambda}{\delta_1}} \ll \sqrt{\frac{N\lambda}{(\lambda/N)^{1/3}}} \ll N^{2/3} 
  \end{equation*}
  and from $\delta \ll 1/N$ and $\lambda > N^{-1/3}$ that
  \begin{equation*}
    \sqrt{\frac{N\delta_1}{\lambda}} \ll \sqrt{\frac{\delta N^2}{\lambda}} + \sqrt{\frac{N^{2/3}}{\lambda^{2/3}}}
    \ll \sqrt{N/\lambda} + (N/\lambda)^{1/3}\ll N^{2/3},
  \end{equation*}
  from which the desired result follows.

<!-- lemma-4: body_md -->
\label{lem:F_smoothing}
  Let $A \geqslant 1$ and $\mathcal{S} \subset [-A, A]$ be a finite set. For $\alpha \in \mathbb{R}$, denote
  \begin{equation*}
    C(\alpha) :=  
  \origsum
  \IfValueT{#1}{_{\substack{#1}}}
  \IfValueT{#2}{^{#2}}
_{s\in \mathcal{S}} c_s e(s\alpha),
  \end{equation*}
  where $c_s\in \mathbb{C}$. Then for any $p, \ell \geqslant 1$ we have
  \begin{equation*}
    |C(\alpha)|^p \ll_{p, \ell} A  \int_\mathbb{R} |C(\xi)|^p \cdot \big(1+A|\xi-\alpha|\big)^{-\ell}  \, d\xi.
  \end{equation*}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Let $\psi$ be a Schwartz function with $\psi(x) = 1$ for $x\in [-1, 1]$. For $f(s) = \psi(s/A) e(s\alpha)$, note that $\widehat{f}(\xi) = A\widehat{\psi}(A(\xi-\alpha))$.
  Hence for any $\ell\geqslant 1$, the decay of $\widehat{\psi}$ gives
  \begin{equation}
    |C(\alpha)| = \Big| 
  \origsum
  \IfValueT{#1}{_{\substack{#1}}}
  \IfValueT{#2}{^{#2}}
_{s \in \mathcal{S}}c_s \psi\Big(\frac{s}{A}\Big) e(s\alpha)\Big| = \Big| 
  \origsum
  \IfValueT{#1}{_{\substack{#1}}}
  \IfValueT{#2}{^{#2}}
_{s\in \mathcal{S}} c_s \int_{-\infty}^\infty A\widehat{\psi}(A(\xi-\alpha)) e(\xi s) \, d\xi\Big| 
    \ll_\ell A \int_{-\infty}^\infty  \frac{|C(\xi)|}{\big(1+A|\xi-\alpha|\big)^{\ell} } \, d\xi.
  \end{equation}

  This settles the case $p=1$, and the general case follows from H\"older's inequality.

  \begin{equation}\label{eq:energy_diagonal_split}

    \prec_k

  \origsum

_{m,n\sim N\\ m\ne n\\  |\xi_m-\xi_n|\prec \frac{1}{D}} |F(\xi_m-\xi_n)|^{2k-1}.
  \end{equation}

  By Lemma \ref{lemma:frequency_count}, since $W\leqslant N^{4/3}$ we have 
  \begin{equation*}
    N_W(\lambda) := \#\big\{ m, n \sim N : m\ne n,\  |\xi_m-\xi_n-\lambda|\prec 1/W \big\}
    \prec \frac{N^2}{W}+N^{2/3}
    \prec \frac{N^2}{W}.
  \end{equation*}

  \origsum

    D\int_{|\lambda|\prec \frac{1}{D}} |F(\lambda)|^{2j} \, d\lambda \prec E_{2j}(\mathcal W;D).
  \end{equation}
