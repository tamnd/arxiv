<!-- assumption-1: body_md -->
\label{ass:smooth}
There exists $L\ge0$ such that, for every $x,y\in\mathbb R^d$,
\[
 \left\|\nabla f(x)-\nabla f(y)\right\|\le L\left\|x-y\right\|.
\]
<!-- end assumption-1 -->
<!-- assumption-2: body_md -->
\label{ass:variance}
For every $x\in\mathbb R^d$,
\[
 \mathbb E_\xi[\nabla f(x;\xi)]=\nabla f(x),\qquad
 \mathbb E_\xi\left\|\nabla f(x;\xi)-\nabla f(x)\right\|^2\le\sigma^2.
\]
<!-- end assumption-2 -->
<!-- assumption-3: body_md -->
\label{ass:lower}
The value $f_*:=\inf_{x\in\mathbb R^d}f(x)$ is finite. For the initial point $x_1$, let $\Delta_f=f(x_1)-f_*$.
<!-- end assumption-3 -->
<!-- assumption-4: body_md -->
\label{ass:average}

\begin{equation}\label{eq:average-smoothness}
 \mathbb E_\xi\left\|\nabla f(x;\xi)-\nabla f(y;\xi)\right\|^2
 \le \bar L^2\left\|x-y\right\|^2.
\end{equation}

<!-- theorem-1: body_md -->
\label{thm:serial}
Under Assumptions~\ref{ass:smooth}--\ref{ass:lower}, the output of
Algorithm~\ref{alg:serial} satisfies
\begin{align}\label{eq:serial-standard}
\begin{split}
     \mathbb E\left\|\nabla f(\widehat x_T)\right\|
 &\le C_G n^{-1/4}+\frac{2\sigma}{\sqrt b}\\
 &= O\left(\frac{1+L+\sigma+\Delta_f}{T^{1/4}}\right).
 \end{split}
\end{align}
If Assumption~\ref{ass:average} also holds, the same output satisfies
\begin{align}
\begin{split}
    \mathbb E\left\|\nabla f(\widehat x_T)\right\|
 &\le
 \left[C_G+\frac L2+2\sigma+
       2\sqrt{2(\sigma^2+\bar L^2)}\right]n^{-1/3}
 +\frac{2\sigma}{\sqrt b} \\
 &= O\left(\frac{1+L+\bar{L}+\sigma+\Delta_f}{T^{1/3}}\right).
\end{split}
\end{align} \label{eq:serial-mss-rate}
If the oracle is noiseless, then
\begin{align}
\begin{split}
 \mathbb E\left\|\nabla f(\widehat x_T)\right\|
 &\le\left(\Delta_f+\frac L2\right)n^{-1/2} \\
 &= O\left(\frac{\Delta_f+L}{T^{1/2}}\right).
\end{split}
\end{align} \label{eq:serial-deterministic}

<!-- theorem-2: body_md -->
\label{thm:anytime}
Suppose $f$ is lower bounded and $L$-smooth, and the stochastic gradient
oracle is unbiased with variance at most $\sigma^2$.  At every deterministic
round $\tau\geq 2$, let $n$ be defined by
\eqref{eq:anytime-report-index}.  The reported point of the Algorithm satisfies
\begin{equation}
\label{eq:anytime-general-guarantee}
 \mathbb E\left\|\nabla f(Z_n)\right\|
 \leq
 \sqrt{\frac{M_n(C_A+4L\log M_n)}{n}}
 +\frac{2\sigma}{\sqrt n}.
\end{equation}
If the oracle additionally satisfies mean-square smoothness, then
\begin{equation}
\label{eq:anytime-mss-guarantee}
 \mathbb E\left\|\nabla f(Z_n)\right\|
 \leq
 \min\left\{
 \frac{\Delta_f+2L+2\sqrt K(1+\log n)}{n^{1/3}},\,
 \sqrt{\frac{M_n(C_A+4L\log M_n)}{n}}
 \right\}
 +\frac{2\sigma}{\sqrt n}.
\end{equation}
In particular, for fixed problem parameters, the three rates are respectively
\begin{align}
\label{eq:anytime-three-rates}
 &O\left(\frac{1+\log\tau}{\tau^{1/3}}\right)
 &&\text{under mean-square smoothness},\nonumber\\
 &O\left(\frac{\sqrt{1+\log\tau}}{\tau^{1/4}}\right)
 &&\text{under standard smoothness},\\
 &O(\tau^{-1/2})&&\text{when } \sigma=0 \text{ with standard smoothness}.\nonumber
\end{align}

<!-- lemma-1: body_md -->
\label{lem:normalized-descent}
For any point $x$, vector $v$, and $\eta>0$, the update
$x^+=x-\eta\mathcal N(v)$ satisfies
\begin{equation}
 f(x^+)\le f(x)-\eta\left\|\nabla f(x)\right\|
       +2\eta\left\|v-\nabla f(x)\right\|+\frac L2\eta^2.
 \label{eq:normalized-descent}
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
If $v\ne0$, Cauchy--Schwarz and the triangle inequality give
\begin{align*}
 \left\langle\nabla f(x),\mathcal N(v)\right\rangle
 &=\left\langlev,\frac{v}{\left\|v\right\|}\right\rangle
   +\left\langle\nabla f(x)-v,\frac{v}{\left\|v\right\|}\right\rangle\\
 &\ge\left\|v\right\|-\left\|\nabla f(x)-v\right\|\\
 &\ge\left\|\nabla f(x)\right\|-2\left\|v-\nabla f(x)\right\|.
\end{align*}
For $v=0$, the inequality still holds. Thus,
\begin{align*}
 f(x^+)
 &\le f(x)+\left\langle\nabla f(x),x^+-x\right\rangle
           +\frac L2\left\|x^+-x\right\|^2\\
 &=f(x)-\eta\left\langle\nabla f(x),\mathcal N(v)\right\rangle
           +\frac L2\eta^2\left\|\mathcal N(v)\right\|^2\\
 &\le f(x)-\eta\left\|\nabla f(x)\right\|
           +2\eta\left\|v-\nabla f(x)\right\|+\frac L2\eta^2,
\end{align*}
where the last step uses $\left\|\mathcal N(v)\right\|\le1$.

<!-- lemma-2: body_md -->
\label{lem:selection}
Let $X_1,\ldots,X_m$ be random candidate points. At each point, average
$b$ independent stochastic gradients to obtain $\widehat g_i$.
If $J\in\arg\min_i\left\|\widehat g_i\right\|$, then
\begin{equation}
 \begin{split}
 \mathbb E\left\|\nabla f(X_J)\right\|
 &\le\mathbb E\min_{1\le i\le m}\left\|\nabla f(X_i)\right\|
          +\frac{m\sigma}{\sqrt b}\\
 &\le\min_{1\le i\le m}\mathbb E\left\|\nabla f(X_i)\right\|
          +\frac{m\sigma}{\sqrt b}.
 \end{split}
 \label{eq:selection}
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
For fixed candidates, the validation average is
\[
 \widehat g_i=\frac1b\sum_{j=1}^b g(X_i;\xi_{i,j}).
\]
Expanding the square and then averaging over the candidates yields
\begin{align*}
 \mathbb E\left\|\widehat g_i-\nabla f(X_i)\right\|^2
 =\frac1{b^2}\mathbb E\left\|
       \sum_{j=1}^b[g(X_i;\xi_{i,j})-\nabla f(X_i)]\right\|^2
 \le\frac{\sigma^2}{b}.
\end{align*}
Consequently, Cauchy--Schwarz gives
\[
 \mathbb E\left\|\widehat g_i-\nabla f(X_i)\right\|
 \le\sqrt{\mathbb E\left\|\widehat g_i-\nabla f(X_i)\right\|^2}
 \le\frac{\sigma}{\sqrt b}.
\]

Choose
$i\in\arg\min_{1\le k\le m}\left\|\nabla f(X_k)\right\|$.
If $J=i$, then $\left\|\nabla f(X_J)\right\|=\min_k\left\|\nabla f(X_k)\right\|$.
If $J\ne i$, 
\begin{align*}
 \left\|\nabla f(X_J)\right\|
 &\le\left\|\widehat g_J\right\|
       +\left\|\widehat g_J-\nabla f(X_J)\right\|\\
 &\le\left\|\widehat g_i\right\|
       +\left\|\widehat g_J-\nabla f(X_J)\right\|\\
 &\le\left\|\nabla f(X_i)\right\|
       +\left\|\widehat g_i-\nabla f(X_i)\right\|
       +\left\|\widehat g_J-\nabla f(X_J)\right\|\\
 &\le\min_k\left\|\nabla f(X_k)\right\|
       +\sum_{k=1}^m\left\|\widehat g_k-\nabla f(X_k)\right\|.
\end{align*}
Taking expectations gives
\begin{align*}
 \mathbb E\left\|\nabla f(X_J)\right\|
 &\le\mathbb E\min_k\left\|\nabla f(X_k)\right\|
       +\sum_{k=1}^m\mathbb E\left\|\widehat g_k-\nabla f(X_k)\right\|\\
 &\le\mathbb E\min_k\left\|\nabla f(X_k)\right\|+\frac{m\sigma}{\sqrt b}.
\end{align*}
Finally, for every fixed index $j$,
\[
 \min_k\left\|\nabla f(X_k)\right\|\le\left\|\nabla f(X_j)\right\|
 \quad\Longrightarrow\quad
 \mathbb E\min_k\left\|\nabla f(X_k)\right\|\le\mathbb E\left\|\nabla f(X_j)\right\|.
\]
Taking the minimum over $j$ proves the second inequality in
\eqref{eq:selection}.

<!-- lemma-3: body_md -->
\label{lem:accepted-pair}
Let $G,G'$ be independent random vectors with mean $\mu$ and
$\mathbb E\left\|G-\mu\right\|^2\le\sigma^2$. For every $a\ge0$, let
$p=\Pr\{\left\|G-G'\right\|>a\}$. Then
\begin{equation}
 \mathbb E\bigl[\left\|G-\mu\right\|\mathbf1_{\{\left\|G-G'\right\|\le a\}}\bigr]
 \le a+\sqrt2\,\sigma\sqrt p.
 \label{eq:accepted-pair}
\end{equation}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
By independence, $\mathbb E_{G'}[G-G']=G-\mu$ when $G$ is fixed. Thus Jensen's
inequality gives
\[
 \mathbb E\left\|G-\mu\right\|
 =\mathbb E_G\left\|\mathbb E_{G'}[G-G']\right\|
 \le\mathbb E_G\mathbb E_{G'}\left\|G-G'\right\|
 =\mathbb E\left\|G-G'\right\|.
\]
Moreover,
\begin{align*}
 \mathbb E\left\|G-G'\right\|^2
 &=\mathbb E\left\|(G-\mu)-(G'-\mu)\right\|^2\\
 &=\mathbb E\left\|G-\mu\right\|^2+\mathbb E\left\|G'-\mu\right\|^2
             -2\mathbb E\left\langleG-\mu,G'-\mu\right\rangle\\
 &\le2\sigma^2,
\end{align*}
where
$\mathbb E\left\langleG-\mu,G'-\mu\right\rangle
=\left\langle\mathbb E[G-\mu],\mathbb E[G'-\mu]\right\rangle=0$.
Splitting the expectation at $a$, we obtain
\begin{align*}
 \mathbb E\left\|G-G'\right\|
 &=\mathbb E\bigl[\left\|G-G'\right\|\mathbf1_{\{\left\|G-G'\right\|\le a\}}\bigr]
   +\mathbb E\bigl[\left\|G-G'\right\|\mathbf1_{\{\left\|G-G'\right\|>a\}}\bigr]\\
 &\le a+\sqrt{\mathbb E\left\|G-G'\right\|^2}\,
              \sqrt{\mathbb E\mathbf1_{\{\left\|G-G'\right\|>a\}}}\\
 &\le a+\sqrt2\,\sigma\sqrt p,
\end{align*}
where the second line uses Cauchy--Schwarz. Therefore,
\[
 \mathbb E\bigl[\left\|G-\mu\right\|\mathbf1_{\{\left\|G-G'\right\|\le a\}}\bigr]
 \le\mathbb E\left\|G-\mu\right\|
 \le\mathbb E\left\|G-G'\right\|
 \le a+\sqrt2\,\sigma\sqrt p.
\]

<!-- lemma-4: body_md -->
\label{lem:ema}
Consider $m$ normalized momentum updates with 
parameters $0<\beta\le1$ and $\eta>0$:
\[
 v_1=g(x_1;\xi_1),\qquad
 v_t=(1-\beta)v_{t-1}+\beta g(x_t;\xi_t)\quad(t\ge2),
 \qquad x_{t+1}=x_t-\eta\mathcal N(v_t).
\]
Then, we can ensure
\begin{equation}
 \mathbb E\left\|v_t-\nabla f(x_t)\right\|
 \le\sigma(1-\beta)^{t-1}+\sigma\sqrt\beta+\frac{L\eta}{\beta},
 \qquad 1\le t\le m.
 \label{eq:ema-tracking}
\end{equation}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Write $e_t=v_t-\nabla f(x_t)$ and
$\varepsilon_t=g(x_t;\xi_t)-\nabla f(x_t)$. At $t=1$,
$\mathbb E\left\|e_1\right\|\le\sigma$.
For $t\ge2$,
\begin{align*}
 e_t
 &=(1-\beta)v_{t-1}+\beta g(x_t;\xi_t)-\nabla f(x_t)\\
 &=(1-\beta)[v_{t-1}-\nabla f(x_{t-1})]
       +\beta[g(x_t;\xi_t)-\nabla f(x_t)]+(1-\beta)[\nabla f(x_{t-1})-\nabla f(x_t)]\\
 &=(1-\beta)e_{t-1}+\beta\varepsilon_t
       +(1-\beta)[\nabla f(x_{t-1})-\nabla f(x_t)].
\end{align*}
Since $e_1=\varepsilon_1$, repeated substitution gives
\begin{align*}
 e_t
 =(1-\beta)^{t-1}\varepsilon_1
       +\beta\sum_{k=2}^t(1-\beta)^{t-k}\varepsilon_k+\sum_{k=2}^t(1-\beta)^{t-k+1}
                [\nabla f(x_{k-1})-\nabla f(x_k)].
\end{align*}
For $i<j$, averaging the sample $\xi_j$ first yields
\[
 \mathbb E\left\langle\varepsilon_i,\varepsilon_j\right\rangle
 =\mathbb E\left\langle\varepsilon_i,\mathbb E_{\xi_j}[g(x_j;\xi_j)-\nabla f(x_j)]\right\rangle=0.
\]
Thus the squared norm of the noise sum satisfies
\begin{align*}
 &\mathbb E\left\|(1-\beta)^{t-1}\varepsilon_1
       +\beta\sum_{k=2}^t(1-\beta)^{t-k}\varepsilon_k\right\|^2\\
 &\quad=(1-\beta)^{2(t-1)}\mathbb E\left\|\varepsilon_1\right\|^2
       +\beta^2\sum_{k=2}^t(1-\beta)^{2(t-k)}
                              \mathbb E\left\|\varepsilon_k\right\|^2\\
 &\quad\le\sigma^2\left[(1-\beta)^{2(t-1)}
       +\beta^2\sum_{j=0}^{t-2}(1-\beta)^{2j}\right]\\
 &\quad\le\sigma^2\left[(1-\beta)^{2(t-1)}
       +\frac{\beta^2}{1-(1-\beta)^2}\right]\\
 &\quad=\sigma^2\left[(1-\beta)^{2(t-1)}
       +\frac{\beta}{2-\beta}\right]
 \le\sigma^2[(1-\beta)^{2(t-1)}+\beta].
\end{align*}
For the remaining gradient-difference sum,
smoothness and $\left\|x_k-x_{k-1}\right\|\le\eta$ give
\begin{align*}
 \left\|\sum_{k=2}^t(1-\beta)^{t-k+1}
                       [\nabla f(x_{k-1})-\nabla f(x_k)]\right\|
 &\le\sum_{k=2}^t(1-\beta)^{t-k+1}
                         L\left\|x_{k-1}-x_k\right\|\\
 &\le L\eta\sum_{j=1}^{t-1}(1-\beta)^j
 \le L\eta\sum_{j=0}^{\infty}(1-\beta)^j
 =\frac{L\eta}{\beta}.
\end{align*}
By the triangle inequality and Cauchy--Schwarz,
\begin{align*}
 \mathbb E\left\|e_t\right\|
 &\le\sqrt{\mathbb E\left\|(1-\beta)^{t-1}\varepsilon_1
       +\beta\sum_{k=2}^t(1-\beta)^{t-k}\varepsilon_k\right\|^2}
       +\frac{L\eta}{\beta}\\
 &\le\sigma\sqrt{(1-\beta)^{2(t-1)}+\beta}
       +\frac{L\eta}{\beta}\\
 &\le\sigma(1-\beta)^{t-1}+\sigma\sqrt\beta+\frac{L\eta}{\beta}.
 \qedhere
\end{align*}

<!-- lemma-5: body_md -->
\label{lem:vr}
Consider a phase with $t=1,\ldots,n$ and updates
\[
 \begin{split}
 v_1&=g(x_1;\xi_1),\\
 v_t&=g(x_t;\xi_t)+(1-\beta)
              [v_{t-1}-g(x_{t-1};\xi_t)]\quad(t\ge2),\\
 x_{t+1}&=x_t-\eta\mathcal N(v_t).
 \end{split}
\]
Under Assumption~\ref{ass:average}, with $\beta=\eta=n^{-2/3}$,
\begin{equation}
 \mathbb E\left\|v_t-\nabla f(x_t)\right\|
 \le\sigma(1-\beta)^{t-1}
       +\sqrt{2(\sigma^2+\bar L^2)}\,n^{-1/3}.
 \label{eq:vr-tracking}
\end{equation}
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
Let $e_t=v_t-\nabla f(x_t)$. At $t=1$, the initialization gives
$\mathbb E\left\|e_1\right\|^2\le\sigma^2$, so \eqref{eq:vr-tracking} holds.
For $t\ge2$, the estimator update gives
\begin{align*}
 e_t
 ={}&g(x_t;\xi_t)+(1-\beta)
                   [v_{t-1}-g(x_{t-1};\xi_t)]-\nabla f(x_t)\\
 ={}&(1-\beta)e_{t-1}
       +\beta[g(x_t;\xi_t)-\nabla f(x_t)]\\
 &+(1-\beta)[g(x_t;\xi_t)-g(x_{t-1};\xi_t)
                    -\nabla f(x_t)+\nabla f(x_{t-1})].
\end{align*}
By unbiasedness, we have
\[
 \mathbb E_{\xi_t}[g(x_t;\xi_t)-\nabla f(x_t)]=0,
 \qquad
 \mathbb E_{\xi_t}[g(x_t;\xi_t)-g(x_{t-1};\xi_t)]
       =\nabla f(x_t)-\nabla f(x_{t-1}).
\]
The gradient difference satisfies
\begin{align*}
 &\mathbb E_{\xi_t}\left\|g(x_t;\xi_t)-g(x_{t-1};\xi_t)
                      -\nabla f(x_t)+\nabla f(x_{t-1})\right\|^2\\
 &\quad=\mathbb E_{\xi_t}\left\|g(x_t;\xi_t)-g(x_{t-1};\xi_t)\right\|^2
                      -\left\|\nabla f(x_t)-\nabla f(x_{t-1})\right\|^2\\
 &\quad\le\mathbb E_{\xi_t}\left\|g(x_t;\xi_t)-g(x_{t-1};\xi_t)\right\|^2\\
 &\quad\le\bar L^2\left\|x_t-x_{t-1}\right\|^2
 \le\bar L^2\eta^2.
\end{align*}
Consequently, we can ensure
\begin{align*}
 \mathbb E\left\|e_t\right\|^2
 ={}&(1-\beta)^2\mathbb E\left\|e_{t-1}\right\|^2\\
 &+\mathbb E\left\|
       \beta[g(x_t;\xi_t)-\nabla f(x_t)]+(1-\beta)[g(x_t;\xi_t)-g(x_{t-1};\xi_t)
                    -\nabla f(x_t)+\nabla f(x_{t-1})]
       \right\|^2\\
 \le{}&(1-\beta)^2\mathbb E\left\|e_{t-1}\right\|^2
       +2\beta^2\mathbb E\left\|g(x_t;\xi_t)-\nabla f(x_t)\right\|^2\\
 &+2(1-\beta)^2
     \mathbb E\left\|g(x_t;\xi_t)-g(x_{t-1};\xi_t)
                    -\nabla f(x_t)+\nabla f(x_{t-1})\right\|^2\\
 \le{}&(1-\beta)^2\mathbb E\left\|e_{t-1}\right\|^2
       +2\beta^2\sigma^2+2(1-\beta)^2\bar L^2\eta^2\\
 \le{}&(1-\beta)^2\mathbb E\left\|e_{t-1}\right\|^2
       +2\beta^2\sigma^2+2\bar L^2\eta^2.
\end{align*}
The initialization gives
$\mathbb E\left\|e_1\right\|^2=\mathbb E\left\|g(x_1;\xi_1)-\nabla f(x_1)\right\|^2\le\sigma^2$.
Repeated substitution in the last recursion yields
\begin{align*}
 \mathbb E\left\|e_t\right\|^2
 &\le(1-\beta)^{2(t-1)}\mathbb E\left\|e_1\right\|^2
       +(2\beta^2\sigma^2+2\bar L^2\eta^2)
                  \sum_{j=0}^{t-2}(1-\beta)^{2j}\\
 &\le\sigma^2(1-\beta)^{2(t-1)}
       +\frac{2\beta^2\sigma^2+2\bar L^2\eta^2}
                    {1-(1-\beta)^2}\\
 &\le\sigma^2(1-\beta)^{2(t-1)}
       +\frac{2\beta^2\sigma^2+2\bar L^2\eta^2}{\beta}\\
 &=\sigma^2(1-\beta)^{2(t-1)}
       +2(\sigma^2+\bar L^2)n^{-2/3},
\end{align*}
where $1-(1-\beta)^2=\beta(2-\beta)\ge\beta$ and
$\beta=\eta=n^{-2/3}$. Taking square roots,
\begin{align*}
 \mathbb E\left\|e_t\right\|
 &\le\sqrt{\mathbb E\left\|e_t\right\|^2}\\
 &\le\sqrt{\sigma^2(1-\beta)^{2(t-1)}
                       +2(\sigma^2+\bar L^2)n^{-2/3}}\\
 &\le\sigma(1-\beta)^{t-1}
       +\sqrt{2(\sigma^2+\bar L^2)}\,n^{-1/3},
\end{align*}
which proves \eqref{eq:vr-tracking}.

\label{lem:anytime-vr}
Under mean-square smoothness, the update
\eqref{eq:anytime-vr-parameters}--\eqref{eq:anytime-vr-point} satisfies
\begin{equation}
\label{eq:anytime-tracking}
 \mathbb E\left\|v_t-\nabla f(x^V_t)\right\|^2\leq Kt^{-2/3}.
\end{equation}
Its weighted reservoir candidate satisfies
\begin{equation}
\label{eq:anytime-vr-candidate}
 \mathbb E\left\|\nabla f(V_n)\right\|
 \leq\frac{\Delta_f+2L+2\sqrt K(1+\log n)}{W_n},
 \qquad W_n\geq n^{1/3}.
\end{equation}

<!-- proof-6: body_md -->
We omit the superscript $V$ and write $e_t=v_t-\nabla f(x_t)$.
For $t\geq2$, the estimator gives
\begin{align}
\label{eq:anytime-tracking-expansion}
 e_t
 ={}&(1-\beta_t)e_{t-1}\nonumber\\
 &+(1-\beta_t)\bigl[g(x_t;\xi_t)-g(x_{t-1};\xi_t)
                    -\nabla f(x_t)+\nabla f(x_{t-1})\bigr]\nonumber\\
 &+\beta_t\bigl[g(x_t;\xi_t)-\nabla f(x_t)\bigr].
\end{align}
By unbiasedness, we have
\[
 \mathbb E_{\xi_t}[g(x_t;\xi_t)-g(x_{t-1};\xi_t)]
 =\nabla f(x_t)-\nabla f(x_{t-1}).
\]
Expanding the square gives
\begin{align*}
 &\mathbb E_{\xi_t}\left\|g(x_t;\xi_t)-g(x_{t-1};\xi_t)
                       -\nabla f(x_t)+\nabla f(x_{t-1})\right\|^2\\
 &\quad=\mathbb E_{\xi_t}\left\|g(x_t;\xi_t)-g(x_{t-1};\xi_t)\right\|^2\\
 &\qquad\quad
 -2\left\langle\mathbb E_{\xi_t}[g(x_t;\xi_t)-g(x_{t-1};\xi_t)],\nabla f(x_t)-\nabla f(x_{t-1})\right\rangle\\
 &\qquad\quad +\left\|\nabla f(x_t)-\nabla f(x_{t-1})\right\|^2\\
 &\quad=\mathbb E_{\xi_t}\left\|g(x_t;\xi_t)-g(x_{t-1};\xi_t)\right\|^2
             -\left\|\nabla f(x_t)-\nabla f(x_{t-1})\right\|^2\\
 &\quad\leq\bar L^2\left\|x_t-x_{t-1}\right\|^2
 \leq\bar L^2\eta_{t-1}^2.
\end{align*}
Here we used mean-square smoothness and
$\left\|x_t-x_{t-1}\right\|\leq\eta_{t-1}$.

Both bracketed terms in \eqref{eq:anytime-tracking-expansion} have
zero mean. Their cross terms with $e_{t-1}$ therefore vanish.
Applying $\left\|u+v\right\|^2\leq2\left\|u\right\|^2+2\left\|v\right\|^2$ to their sum gives
\begin{align*}
 \mathbb E_{\xi_t}\left\|e_t\right\|^2
 &\leq(1-\beta_t)^2\left\|e_{t-1}\right\|^2
       +2(1-\beta_t)^2\bar L^2\eta_{t-1}^2
       +2\beta_t^2\sigma^2\\
 &\leq(1-\beta_t)^2\left\|e_{t-1}\right\|^2
       +2\bar L^2\eta_{t-1}^2+2\beta_t^2\sigma^2.
\end{align*}
Averaging over the earlier samples proves
\begin{equation}
\label{eq:anytime-tracking-recursion}
 \mathbb E\left\|v_t-\nabla f(x_t)\right\|^2
 \leq(1-\beta_t)^2\mathbb E\left\|v_{t-1}-\nabla f(x_{t-1})\right\|^2
      +2\bar L^2\eta_{t-1}^2+2\beta_t^2\sigma^2.
\end{equation}

We prove \eqref{eq:anytime-tracking} by induction with
$K=6\bar L^2+2\sigma^2$. At $t=1$,
\[
 \mathbb E\left\|e_1\right\|^2
 =\mathbb E\left\|g(x_1;\xi_1)-\nabla f(x_1)\right\|^2
 \leq\sigma^2\leq K=K\beta_1.
\]
For $t\geq2$, $\beta_t=t^{-2/3}\geq1/t$ and
$u\leq u^{2/3}$ for $0\leq u\leq1$, so
\begin{equation}
\label{eq:anytime-beta-comparison}
 1-\beta_t\leq1-\frac1t
 \leq\left(1-\frac1t\right)^{2/3}
 =\frac{\beta_t}{\beta_{t-1}}.
\end{equation}
Multiplying this inequality by
$(1-\beta_t)\beta_{t-1}\geq0$ yields
\[
 (1-\beta_t)^2\beta_{t-1}
 \leq(1-\beta_t)\beta_t.
\]
Also,
\[
 \frac{\eta_{t-1}^2}{\beta_t^2}
 =\frac{(t-1)^{-4/3}}{t^{-4/3}}
 =\left(\frac{t}{t-1}\right)^{4/3}
 \leq2^{4/3}<3.
\]
If $\mathbb E\left\|e_{t-1}\right\|^2\leq K\beta_{t-1}$, the recursion gives
\begin{align*}
 \mathbb E\left\|e_t\right\|^2
 &\leq K(1-\beta_t)^2\beta_{t-1}
       +6\bar L^2\beta_t^2+2\sigma^2\beta_t^2\\
 &\leq K(1-\beta_t)\beta_t
       +(6\bar L^2+2\sigma^2)\beta_t^2\\
 &=K\beta_t-K\beta_t^2+K\beta_t^2
 =K\beta_t.
\end{align*}
Thus, by Cauchy--Schwarz,
\[
 \mathbb E\left\|e_t\right\|\leq\sqrt{\mathbb E\left\|e_t\right\|^2}
 \leq\sqrt K\,t^{-1/3}.
\]

By Lemma~\ref{lem:normalized-descent},
\[
 \eta_t\left\|\nabla f(x_t)\right\|
 \leq f(x_t)-f(x_{t+1})+2\eta_t\left\|e_t\right\|
                      +\frac L2\eta_t^2.
\]
Taking expectations, summing, and using $f(x_{n+1})\geq f_*$ gives
\begin{align*}
 \sum_{t=1}^n\eta_t\mathbb E\left\|\nabla f(x_t)\right\|
 &\leq f(x_1)-\mathbb E f(x_{n+1})
       +2\sum_{t=1}^n\eta_t\mathbb E\left\|e_t\right\|
       +\frac L2\sum_{t=1}^n\eta_t^2\\
 &\leq\Delta_f+2\sqrt K\sum_{t=1}^n t^{-2/3}t^{-1/3}
                 +\frac L2\sum_{t=1}^n t^{-4/3}\\
 &=\Delta_f+2\sqrt K\sum_{t=1}^n\frac1t
                 +\frac L2\sum_{t=1}^n t^{-4/3}.
\end{align*}
Since both summands decrease with $t$,
\begin{align*}
 \sum_{t=1}^n\frac1t
 &\leq1+\int_1^n\frac{ds}{s}=1+\log n,\\
 \sum_{t=1}^n t^{-4/3}
 &\leq1+\int_1^n s^{-4/3}\,ds
 =1+3(1-n^{-1/3})\leq4.
\end{align*}
Consequently,
\[
 \sum_{t=1}^n\eta_t\mathbb E\left\|\nabla f(x_t)\right\|
 \leq\Delta_f+2\sqrt K(1+\log n)+2L.
\]

The weights $\eta_t$ are deterministic, and the index-selection law in
\eqref{eq:anytime-reservoir-laws} gives
\[
 \mathbb E\left\|\nabla f(V_n)\right\|
 =\mathbb E\left[\sum_{t=1}^n\frac{\eta_t}{W_n}
                           \left\|\nabla f(x_t)\right\|\right]
 =\frac1{W_n}\sum_{t=1}^n\eta_t\mathbb E\left\|\nabla f(x_t)\right\|.
\]
This proves the first bound in \eqref{eq:anytime-vr-candidate}.
For the second bound,
\[
 W_n=\sum_{t=1}^n t^{-2/3}
 \geq\sum_{t=1}^n n^{-2/3}=n^{1/3}.
\]

\label{lem:anytime-adagrad}
Under standard smoothness and bounded variance alone, the update
\eqref{eq:anytime-adagrad-update} satisfies
\begin{align}
\label{eq:anytime-adagrad-two-bounds}
 \mathbb E b_n&\leq M_n,\nonumber\\
 \mathbb E\sum_{t=1}^n\frac{\left\|\nabla f(x^A_t)\right\|^2}{b_{t-1}}
 &\leq C_A+4L\log M_n.
\end{align}
Consequently, its uniform reservoir candidate satisfies
\begin{equation}
\label{eq:anytime-adagrad-candidate}
 \mathbb E\left\|\nabla f(A_n)\right\|
 \leq\sqrt{\frac{M_n(C_A+4L\log M_n)}{n}}.
\end{equation}

<!-- proof-7: body_md -->
We omit the superscript $A$ in this proof. Recall that
\[
 b_0=1,\qquad
 b_t^2=b_{t-1}^2+\left\|g_t\right\|^2,\qquad
 x_{t+1}=x_t-\frac{g_t}{b_t}.
\]
In particular,
\[
 b_t\geq b_{t-1}\geq1,\qquad
 b_t\geq\left\|g_t\right\|,\qquad
 \left\|x_{t+1}-x_t\right\|=\frac{\left\|g_t\right\|}{b_t}\leq1.
\]
Thus $\left\|x_t-x_1\right\|\leq t-1$ and
\[
 \left\|\nabla f(x_t)\right\|
 \leq\left\|\nabla f(x_1)\right\|+L(t-1),\qquad
 \mathbb E\left\|g_t\right\|^2
 \leq\bigl(\left\|\nabla f(x_1)\right\|+L(t-1)\bigr)^2+\sigma^2.
\]
For the proof only, set
$x_0=x_1$ and $g_0=0$ to make
$x_1=x_0-g_0/b_0$ valid.

We first bound the effect of using the current gradient in $b_t$.
Unbiasedness gives
\begin{align*}
 \mathbb E\left\|g_t\right\|^2
 &=\left\|\nabla f(x_t)\right\|^2
   +2\left\langle\nabla f(x_t),\mathbb E g_t-\nabla f(x_t)\right\rangle
   +\mathbb E\left\|g_t-\nabla f(x_t)\right\|^2\\
 &\leq\left\|\nabla f(x_t)\right\|^2+\sigma^2.
\end{align*}
The difference between the two normalizations is
\begin{align*}
 \frac1{b_{t-1}}-\frac1{b_t}
 =\frac{b_t-b_{t-1}}{b_{t-1}b_t}=\frac{b_t^2-b_{t-1}^2}
         {b_{t-1}b_t(b_t+b_{t-1})}=\frac{\left\|g_t\right\|^2}
         {b_{t-1}b_t(b_t+b_{t-1})}\geq0.
\end{align*}
Using this identity, the inner-product inequality, and
$\left\|g_t\right\|\leq b_t$, we obtain
\begin{align}
\label{eq:anytime-adagrad-correlation-start}
 &\mathbb E\left[
   \left\langle\nabla f(x_t),g_t\right\rangle
   \left(\frac1{b_{t-1}}-\frac1{b_t}\right)\right]\nonumber\\
 &\quad\leq
 \mathbb E\left[
   \left|\left\langle\nabla f(x_t),g_t\right\rangle\right|
   \left(\frac1{b_{t-1}}-\frac1{b_t}\right)\right]\nonumber\\
 &\quad\leq
 \left\|\nabla f(x_t)\right\|
 \mathbb E\left[\left\|g_t\right\|
   \left(\frac1{b_{t-1}}-\frac1{b_t}\right)\right]\nonumber\\
 &\quad=
 \frac{\left\|\nabla f(x_t)\right\|}{b_{t-1}}
 \mathbb E\left[\frac{\left\|g_t\right\|^3}{b_t(b_t+b_{t-1})}\right]\nonumber\\
 &\quad\leq
 \frac{\left\|\nabla f(x_t)\right\|}{b_{t-1}}
 \mathbb E\left[\frac{\left\|g_t\right\|^2}{b_t+b_{t-1}}\right].
\end{align}
Cauchy--Schwarz bounds the last expectation by
\begin{align*}
 \left(\mathbb E\frac{\left\|g_t\right\|^2}{b_t+b_{t-1}}\right)^2
 &=\left(\mathbb E\left[
      \left\|g_t\right\|\frac{\left\|g_t\right\|}{b_t+b_{t-1}}\right]\right)^2\\
 &\leq\mathbb E\left\|g_t\right\|^2\,
          \mathbb E\frac{\left\|g_t\right\|^2}{(b_t+b_{t-1})^2}\\
 &\leq\bigl(\left\|\nabla f(x_t)\right\|^2+\sigma^2\bigr)
          \mathbb E\frac{\left\|g_t\right\|^2}{(b_t+b_{t-1})^2}.
\end{align*}
Apply $uv\leq(u^2+v^2)/2$ to the resulting product:
\begin{align*}
 \frac{\left\|\nabla f(x_t)\right\|}{b_{t-1}}
       \mathbb E\frac{\left\|g_t\right\|^2}{b_t+b_{t-1}}
 &\leq
 \frac{\left\|\nabla f(x_t)\right\|}{\sqrt{b_{t-1}}}
 \left(
   \frac{\left\|\nabla f(x_t)\right\|^2+\sigma^2}{b_{t-1}}\,
   \mathbb E\frac{\left\|g_t\right\|^2}{(b_t+b_{t-1})^2}
 \right)^{1/2}\\
 &\leq
 \frac{\left\|\nabla f(x_t)\right\|^2}{2b_{t-1}}
 +\frac{\left\|\nabla f(x_t)\right\|^2+\sigma^2}{2}
   \mathbb E\frac{\left\|g_t\right\|^2}{b_{t-1}(b_t+b_{t-1})^2}.
\end{align*}
Since $b_t+b_{t-1}\geq b_t$,
\[
 \frac{\left\|g_t\right\|^2}{b_{t-1}(b_t+b_{t-1})^2}
 \leq
 \frac{\left\|g_t\right\|^2}{b_{t-1}b_t(b_t+b_{t-1})}
 =\frac1{b_{t-1}}-\frac1{b_t}.
\]
Substituting these bounds into
\eqref{eq:anytime-adagrad-correlation-start} gives
\begin{equation}\label{eq:anytime-adagrad-correlation}
 \mathbb E\left[
   \left\langle\nabla f(x_t),g_t\right\rangle
   \left(\frac1{b_{t-1}}-\frac1{b_t}\right)\right]\leq
 \frac{\left\|\nabla f(x_t)\right\|^2}{2b_{t-1}}
 +\frac{\left\|\nabla f(x_t)\right\|^2+\sigma^2}{2}
       \mathbb E\left[\frac1{b_{t-1}}-\frac1{b_t}\right].
\end{equation}

Next, we rewrite the term containing
$\left\|\nabla f(x_t)\right\|^2$ so that it can be summed over $t$:
\begin{align}
\label{eq:anytime-adagrad-potential}
 &\frac{\left\|\nabla f(x_t)\right\|^2}{2}
       \left(\frac1{b_{t-1}}-\frac1{b_t}\right)\nonumber\\
 &\quad=
 \frac{\left\|\nabla f(x_t)\right\|^2}{2b_{t-1}}
 -\frac{\left\|\nabla f(x_t)\right\|^2}{2b_t}\nonumber\\
 &\quad=
 \frac12\left(
   \frac{\left\|\nabla f(x_{t-1})\right\|^2}{b_{t-1}}
   -\frac{\left\|\nabla f(x_t)\right\|^2}{b_t}\right) 
 +\frac{\left\|\nabla f(x_t)\right\|^2-\left\|\nabla f(x_{t-1})\right\|^2}
        {2b_{t-1}}.
\end{align}
For the last numerator,  smoothness gives
\begin{align*}
 &\left\|\nabla f(x_t)\right\|^2-\left\|\nabla f(x_{t-1})\right\|^2\\
 &\quad=
 2\left\langle\nabla f(x_t),\nabla f(x_t)-\nabla f(x_{t-1})\right\rangle
 -\left\|\nabla f(x_t)-\nabla f(x_{t-1})\right\|^2\\
 &\quad\leq
 2\left\|\nabla f(x_t)\right\|
   \left\|\nabla f(x_t)-\nabla f(x_{t-1})\right\|\\
 &\quad\leq
 2L\left\|\nabla f(x_t)\right\|\left\|x_t-x_{t-1}\right\|\\
 &\quad=
 \frac{2L\left\|\nabla f(x_t)\right\|\left\|g_{t-1}\right\|}{b_{t-1}}.
\end{align*}
Consequently, $uv\leq u^2/4+v^2$ implies
\begin{align}
\label{eq:anytime-adagrad-potential-change}
 \frac{\left\|\nabla f(x_t)\right\|^2-\left\|\nabla f(x_{t-1})\right\|^2}
        {2b_{t-1}}
 &\leq
 \frac{\left\|\nabla f(x_t)\right\|}{\sqrt{b_{t-1}}}
 \frac{L\left\|g_{t-1}\right\|}{b_{t-1}^{3/2}}\nonumber\\
 &\leq
 \frac{\left\|\nabla f(x_t)\right\|^2}{4b_{t-1}}
 +\frac{L^2\left\|g_{t-1}\right\|^2}{b_{t-1}^3}.
\end{align}

The smoothness inequality for the AdaGrad update is
\[
 f(x_{t+1})
 \leq f(x_t)-\frac{\left\langle\nabla f(x_t),g_t\right\rangle}{b_t}
             +\frac L2\frac{\left\|g_t\right\|^2}{b_t^2}.
\]
Separate $1/b_t$ from
$1/b_{t-1}$ and use $\mathbb E g_t=\nabla f(x_t)$:
\begin{align*}
 \mathbb E f(x_{t+1})
 &\leq f(x_t)
 -\frac{\left\langle\nabla f(x_t),\mathbb E g_t\right\rangle}{b_{t-1}}+
 \mathbb E\left[\left\langle\nabla f(x_t),g_t\right\rangle
           \left(\frac1{b_{t-1}}-\frac1{b_t}\right)\right]
 +\frac L2\mathbb E\frac{\left\|g_t\right\|^2}{b_t^2}\\
 &=f(x_t)-\frac{\left\|\nabla f(x_t)\right\|^2}{b_{t-1}}+
 \mathbb E\left[\left\langle\nabla f(x_t),g_t\right\rangle
           \left(\frac1{b_{t-1}}-\frac1{b_t}\right)\right]
 +\frac L2\mathbb E\frac{\left\|g_t\right\|^2}{b_t^2}.
\end{align*}
Insert \eqref{eq:anytime-adagrad-correlation},
\eqref{eq:anytime-adagrad-potential}, and
\eqref{eq:anytime-adagrad-potential-change}. The coefficient of
$\left\|\nabla f(x_t)\right\|^2/b_{t-1}$ is $-1+1/2+1/4=-1/4$.
Now also average over all earlier samples:
\begin{align}
\label{eq:anytime-adagrad-descent}
 \mathbb E f(x_{t+1})
 &\leq\mathbb E f(x_t)
 -\frac14\mathbb E\frac{\left\|\nabla f(x_t)\right\|^2}{b_{t-1}}\nonumber\\
 &\quad+
 \frac12\mathbb E\left[
   \frac{\left\|\nabla f(x_{t-1})\right\|^2}{b_{t-1}}
   -\frac{\left\|\nabla f(x_t)\right\|^2}{b_t}\right]\nonumber\\
 &\quad+
 \frac{\sigma^2}{2}\mathbb E\left[\frac1{b_{t-1}}-\frac1{b_t}\right]
 +L^2\mathbb E\frac{\left\|g_{t-1}\right\|^2}{b_{t-1}^3}
 +\frac L2\mathbb E\frac{\left\|g_t\right\|^2}{b_t^2}.
\end{align}

We next bound the last three sums in
\eqref{eq:anytime-adagrad-descent}. Since $u^{-3/2}$ and $u^{-1}$
are decreasing for $u>0$,
\begin{align*}
 \frac{\left\|g_t\right\|^2}{b_t^3}
 &=\frac{b_t^2-b_{t-1}^2}{(b_t^2)^{3/2}}
 \leq\int_{b_{t-1}^2}^{b_t^2}u^{-3/2}\,du
 =2\left(\frac1{b_{t-1}}-\frac1{b_t}\right),\\
 \frac{\left\|g_t\right\|^2}{b_t^2}
 &=\frac{b_t^2-b_{t-1}^2}{b_t^2}\leq\int_{b_{t-1}^2}^{b_t^2}\frac{du}{u}
 =2\log b_t-2\log b_{t-1}.
\end{align*}
Using $g_0=0$ and $b_0=1$ gives the following bounds, including the
index shift in the first sum:
\begin{align}
\label{eq:anytime-adagrad-scalar-sums}
 \sum_{t=1}^n\frac{\left\|g_{t-1}\right\|^2}{b_{t-1}^3}
 &=\sum_{t=1}^{n-1}\frac{\left\|g_t\right\|^2}{b_t^3}
 \leq2\left(1-\frac1{b_{n-1}}\right)\leq2,\nonumber\\
 \sum_{t=1}^n\frac{\left\|g_t\right\|^2}{b_t^2}
 &\leq2\sum_{t=1}^n(\log b_t-\log b_{t-1})
 =2\log b_n,\nonumber\\
 \sum_{t=1}^n\left(\frac1{b_{t-1}}-\frac1{b_t}\right)
 &=1-\frac1{b_n}\leq1.
\end{align}

The function and gradient terms in
\eqref{eq:anytime-adagrad-descent} telescope as follows:
\begin{align*}
 \sum_{t=1}^n\bigl(f(x_t)-f(x_{t+1})\bigr)
 &=f(x_1)-f(x_{n+1}),\\
 \sum_{t=1}^n\left(
   \frac{\left\|\nabla f(x_{t-1})\right\|^2}{b_{t-1}}
   -\frac{\left\|\nabla f(x_t)\right\|^2}{b_t}\right)
 &=\left\|\nabla f(x_1)\right\|^2
   -\frac{\left\|\nabla f(x_n)\right\|^2}{b_n}.
\end{align*}
Summing \eqref{eq:anytime-adagrad-descent} and moving the gradient sum
to the left therefore yields
\begin{align*}
 \frac14\mathbb E\sum_{t=1}^n
                 \frac{\left\|\nabla f(x_t)\right\|^2}{b_{t-1}}
 &\leq
 f(x_1)-\mathbb E f(x_{n+1})
 +\frac12\left\|\nabla f(x_1)\right\|^2
 -\frac12\mathbb E\frac{\left\|\nabla f(x_n)\right\|^2}{b_n}\\
 &\qquad+
 \frac{\sigma^2}{2}\left(1-\mathbb E\frac1{b_n}\right)
 +L^2\mathbb E\sum_{t=1}^n\frac{\left\|g_{t-1}\right\|^2}{b_{t-1}^3}
 +\frac L2\mathbb E\sum_{t=1}^n\frac{\left\|g_t\right\|^2}{b_t^2}\\
 &\quad\leq
 \Delta_f+\frac12\left\|\nabla f(x_1)\right\|^2
 +\frac{\sigma^2}{2}+2L^2+L\mathbb E\log b_n.
\end{align*}
The last inequality uses $f(x_{n+1})\geq f_*$,
\eqref{eq:anytime-adagrad-scalar-sums}, and the nonnegativity of
$\left\|\nabla f(x_n)\right\|^2/b_n$ and $1/b_n$.
Multiplication by four and concavity of the logarithm give
\begin{align}
\label{eq:anytime-adagrad-weighted}
 \mathbb E\sum_{t=1}^n\frac{\left\|\nabla f(x_t)\right\|^2}{b_{t-1}}
 &\leq
 4\Delta_f+2\left\|\nabla f(x_1)\right\|^2+2\sigma^2\nonumber+8L^2
 +4L\mathbb E\log b_n\nonumber\\
 &=C_A+4L\mathbb E\log b_n\nonumber\\
 &\leq C_A+4L\log(\mathbb E b_n).
\end{align}

We now bound $\mathbb E b_n$. Since $b_{t-1}\leq b_n$ for $t\leq n$,
\[
 \sum_{t=1}^n\left\|\nabla f(x_t)\right\|^2
 =\sum_{t=1}^n b_{t-1}
                   \frac{\left\|\nabla f(x_t)\right\|^2}{b_{t-1}}
 \leq b_n\sum_{t=1}^n
                   \frac{\left\|\nabla f(x_t)\right\|^2}{b_{t-1}}.
\]
Take square roots and apply Cauchy--Schwarz to the expectation:
\begin{align}
\label{eq:anytime-adagrad-square-root-sum}
 \mathbb E\sqrt{\sum_{t=1}^n\left\|\nabla f(x_t)\right\|^2}
 \leq\mathbb E\left[
    \sqrt{b_n}
    \left(\sum_{t=1}^n
              \frac{\left\|\nabla f(x_t)\right\|^2}{b_{t-1}}\right)^{1/2}
    \right]\leq
 \sqrt{\mathbb E b_n}\left(
    \mathbb E\sum_{t=1}^n
              \frac{\left\|\nabla f(x_t)\right\|^2}{b_{t-1}}
              \right)^{1/2}.
\end{align}
To relate $b_n$ to the true gradients, apply the Euclidean triangle inequality:
\begin{align*}
 b_n
 &=\left\|(1,g_1,\ldots,g_n)\right\|\\
 &\leq
 \left\|(1,\nabla f(x_1),\ldots,\nabla f(x_n))\right\|
 +\left\|(0,g_1-\nabla f(x_1),\ldots,g_n-\nabla f(x_n))\right\|\\
 &=\sqrt{1+\sum_{t=1}^n\left\|\nabla f(x_t)\right\|^2}
   +\sqrt{\sum_{t=1}^n\left\|g_t-\nabla f(x_t)\right\|^2}\\
 &\leq
 1+\sqrt{\sum_{t=1}^n\left\|\nabla f(x_t)\right\|^2}
   +\sqrt{\sum_{t=1}^n\left\|g_t-\nabla f(x_t)\right\|^2}.
\end{align*}
Concavity of the square root and the variance bound imply
\[
 \mathbb E\sqrt{\sum_{t=1}^n\left\|g_t-\nabla f(x_t)\right\|^2}
 \leq
 \sqrt{\sum_{t=1}^n\mathbb E\left\|g_t-\nabla f(x_t)\right\|^2}
 \leq\sigma\sqrt n.
\]
Combining these inequalities and then using
$uv\leq(u^2+v^2)/2$ gives
\begin{align*}
 \mathbb E b_n
 &\leq
 1+\sqrt{\mathbb E b_n}
   \left(\mathbb E\sum_{t=1}^n
                \frac{\left\|\nabla f(x_t)\right\|^2}{b_{t-1}}
   \right)^{1/2}
 +\sigma\sqrt n\\
 &\leq
 1+\frac12\mathbb E b_n
 +\frac12\mathbb E\sum_{t=1}^n
                \frac{\left\|\nabla f(x_t)\right\|^2}{b_{t-1}}
 +\sigma\sqrt n.
\end{align*}
Subtract $\mathbb E b_n/2$ and multiply by two. By
\eqref{eq:anytime-adagrad-weighted},
\begin{align*}
 \mathbb E b_n
 &\leq
 2+2\sigma\sqrt n
 +\mathbb E\sum_{t=1}^n
                \frac{\left\|\nabla f(x_t)\right\|^2}{b_{t-1}}\\
 &\leq2+2\sigma\sqrt n+C_A+4L\log(\mathbb E b_n).
\end{align*}
For $u\geq1$, $\log u\leq\sqrt u$. Since $\mathbb E b_n\geq1$,
another application of Young's inequality gives
\[
 4L\log(\mathbb E b_n)\leq4L\sqrt{\mathbb E b_n}
 \leq\frac12\mathbb E b_n+8L^2.
\]
Substituting this inequality and again moving $\mathbb E b_n/2$ to the left,
we obtain
\begin{align*}
 \mathbb E b_n
 &\leq2+2\sigma\sqrt n+C_A+\frac12\mathbb E b_n+8L^2,\\
 \mathbb E b_n
 &\leq4+4\sigma\sqrt n+2C_A+16L^2=M_n.
\end{align*}
Putting this bound into \eqref{eq:anytime-adagrad-weighted} also gives
\[
 \mathbb E\sum_{t=1}^n\frac{\left\|\nabla f(x_t)\right\|^2}{b_{t-1}}
 \leq C_A+4L\log(\mathbb E b_n)
 \leq C_A+4L\log M_n,
\]
which proves both inequalities in
\eqref{eq:anytime-adagrad-two-bounds}.
Finally, the Cauchy--Schwarz inequality and
\eqref{eq:anytime-adagrad-square-root-sum} yield
\begin{align*}
 \mathbb E\left\|\nabla f(A_n)\right\|
 &=\frac1n\mathbb E\sum_{t=1}^n\left\|\nabla f(x_t)\right\|\leq
 \frac1n\mathbb E\left[
      \sqrt n\sqrt{\sum_{t=1}^n\left\|\nabla f(x_t)\right\|^2}\right]\\
 &=\frac1{\sqrt n}
      \mathbb E\sqrt{\sum_{t=1}^n\left\|\nabla f(x_t)\right\|^2}\leq
 \left(
   \frac{\mathbb E b_n}{n}
   \mathbb E\sum_{t=1}^n\frac{\left\|\nabla f(x_t)\right\|^2}{b_{t-1}}
 \right)^{1/2}\\
 &\leq
 \sqrt{\frac{M_n(C_A+4L\log M_n)}{n}}.
\end{align*}
