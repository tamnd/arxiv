<!-- prob-1: body_md -->
\label{problem_1}
    For given values of $n$, $k$ and $q$, compute the value
$$\mathbb{E}_{\textnormal{opt}}[n,k]_q \triangleq \min\{\mathbb{E}[\mathcal C] \, : \,  \mathcal C \subseteq \mathbb{F}_q^n \, \text{ is a code of dimension $k$}\},$$
    and construct
    a code $\mathcal C$ attaining the minimum.
<!-- end prob-1 -->
<!-- thm-1: body_md -->
\label{MDS_bound_thm}
    For any code $\mathcal{C} \subseteq \mathbb{F}_q^n$ of dimension $k$, we have 
    \[
    \mathbb{E}[\mathcal{C}] \ge \sum_{i=0}^{k-1} \frac{n}{n-i} = n(H_n - H_{n-k}).
    \]
    Furthermore, the lower bound is attained with equality if and only if $\mathcal{C}$ is an MDS code.
<!-- end thm-1 -->
<!-- rem-1: body_md -->
Theorem~\ref{MDS_bound_thm} provides a lower bound on the expectation and solves Problem~\ref{problem_1} for any choice of parameters $n, k$ and $q$ for which there exists an $[n,k]_q$ MDS code. In particular, assuming that the MDS conjecture holds, we can write
    \[
    \mathbb{E}_{\text{opt}}[n,k]_q = n(H_n - H_{n-k}) \text{ when } q \ge n-1.
    \]
<!-- end rem-1 -->
<!-- thm-2: body_md -->
Let $\mathcal{C} \subseteq \mathbb{F}_q^n$ be the $q$-ary simplex code of dimension $k$, where $n = (q^k-1)/(q-1)$. We have 
    \begin{equation*} 
    \mathbb{E}[\mathcal{C}] = k + \sum_{i=1}^k \frac{q^{i-1}-1}{q^k - q^{i-1}}.
    \end{equation*}
<!-- end thm-2 -->
<!-- thm-3: body_md -->
Let $\mathcal{C} \subseteq \mathbb{F}_q^n$ be the $q$-ary Hamming code of redundancy $r$, where $n=(q^r-1)/(q-1)$. We have 
    \[
    \mathbb{E}[\mathcal{C}] = nH_n -  \sum_{\ell=1}^{r} \frac{1}{\binom{n-1}{n-\ell}}\frac{\prod_{i = 0}^{\ell-1} \frac{q^r-q^i}{q-1}}{\ell !}.
    \]
<!-- end thm-3 -->
<!-- lem-1: body_md -->
\label{lem:dimension_increasing}
    Let $V \le X$ be a subspace with $\dim(V) = i$, where $0 \le i \le k-1$, and let $U \ge V$ be a subspace with $\dim(U) = i + 1$. 
    Suppose that the drawn columns first span the subspace $V$.
    The probability that $U$ is the next span we obtain is
    \[
    \Pr[V_{i+1} = U \, | \, V_i = V] = \frac{p(U) - p(V)}{1 - p(V)}.
    \]
<!-- end lem-1 -->
<!-- proof-1: body_md -->
For $r \ge 1$, suppose that the $r$-th draw allows us to get $U$. This means that the previous $r-1$ draws were columns inside $V$. This probability is equal to
    \[
    p(V)^{r-1} (p(U)-p(V)).
    \]
    To conclude, we sum over all possible values of $r$, getting
    \[
    \begin{split}
        \Pr[V_{i+1} = U \, | \, V_i = V] &= \sum_{r=1}^\infty p(V)^{r-1} (p(U)-p(V)) \\
        &= (p(U) - p(V)) \sum_{r=0}^\infty p(V)^r = \frac{p(U) - p(V)}{1 - p(V)}.
    \end{split}
    \]
<!-- end proof-1 -->
<!-- cor-1: body_md -->
Let $S = (V_0, V_1, \dots, V_k) \in \mathcal{S}$ be a chain of dimension-increasing subspaces, where $V_i \le X$ for every $0 \le i \le k$. We have
    \[
    \Pr[S] = 
                                              
    \prod_{i=1}^k \frac{p(V_i) - p(V_{i-1})}{1 - p(V_{i-1})}.
    \]
<!-- end cor-1 -->
<!-- thm-4: body_md -->
\label{thm:general_formula} 
     Let $T$ be the random variable that counts the number of draws needed to span $X$. We have
     \[
     \mathbb{E}[T] = \sum_{S = (V_0, \dots, V_k) \in \mathcal{S}} \Pr[S] \sum_{j=1}^k \frac{1}{1-p(V_{j-1})}.
     \]
<!-- end thm-4 -->
<!-- proof-2: body_md -->
The expectation can be expressed by summing over all the possible configurations of $\mathcal{S}$ and all the possible $k$-tuples of waiting times as 
\[
\begin{split}
    \mathbb{E}[T] &= \sum_{S \in \mathcal{S}} \sum_{n_1, \dots, n_k \ge 1} (n_1 + \cdots + n_k) \Pr[S, (n_1, \dots, n_k)] \\
    &= \sum_{S \in \mathcal{S}} \sum_{n_1, \dots, n_k \ge 1} (n_1 + \cdots + n_k) \prod_{i=1}^k p(V_{i-1})^{n_i-1} \left(p(V_i) - p(V_{i-1})\right).
\end{split}
\]
Note that the sums involving the waiting times are independent. Moreover, the $i$-th factor of the product depends only on $n_i$ and is nonnegative, since $0 \le p(V_{i-1}) \le p(V_i) \le 1$ for every $1 \le i \le k$.
We can then exchange the order of sums and products, obtaining
\[
\begin{split}
    \mathbb{E}[T] =& \sum_{S \in \mathcal{S}} \sum_{j=1}^k \sum_{n_1= 1}^\infty \cdots \sum_{n_k=1}^\infty n_j \prod_{i=1}^k p(V_{i-1})^{n_i-1} \left(p(V_i) - p(V_{i-1})\right) \\
    =& \sum_{S \in \mathcal{S}} \sum_{j=1}^k \left( \left(p(V_j) - p(V_{j-1})\right) \sum_{n_j=1}^\infty n_j p(V_{j-1})^{n_j-1} \right)\prod_{\substack{i=1 \\ i \neq j}}^k \left(p(V_i) - p(V_{i-1})\right) \sum_{n_i = 1}^\infty p(V_{i-1})^{n_i-1}.
\end{split}
\]
Since $p(V_i) < 1$ for all $0 \le i \le k-1$, we can apply the convergence properties of geometric series
\[
\sum_{n=1}^\infty x^{n-1} = \frac{1}{1-x} \quad \text{and} \quad \sum_{n=1}^\infty nx^{n-1} = \frac{1}{(1-x)^2}.
\]
Therefore,
\[
\begin{split}
    \mathbb{E}[T] &= \sum_{S \in \mathcal{S}} \sum_{j=1}^k \frac{p(V_j) - p(V_{j-1})}{(1-p(V_{j-1}))^2} \prod_{\substack{i=1 \\ i \neq j}}^k \frac{p(V_i) - p(V_{i-1})}{1-p(V_{i-1})} \\
    &= \sum_{S \in \mathcal{S}} \sum_{j=1}^k \frac{1}{1-p(V_{j-1})} \prod_{i=1}^k \frac{p(V_i) - p(V_{i-1})}{1-p(V_{i-1})} \\
    &= \sum_{S \in \mathcal{S}} \sum_{j=1}^k \frac{\Pr[S]}{1-p(V_{j-1})} = \sum_{S \in \mathcal{S}} \Pr[S] \sum_{j=1}^k \frac{1}{1-p(V_{j-1})}.
\end{split}
\]
<!-- end proof-2 -->
<!-- cor-2: body_md -->
Let $\mathcal{C} \subseteq \mathbb{F}_q^n$ be the first-order $q$-ary Reed-Muller code of dimension $m+1$, where $n = q^m$.
    \[
    \mathbb{E}[\mathcal{C}] = m+1 + \sum_{j=1}^{m} \frac{1}{q^{j}-1}.
    \]
<!-- end cor-2 -->
<!-- proof-3: body_md -->
A natural generator matrix for $\mathcal{C}$ is 
\[
G = 
\begin{pmatrix}
    1 & 1 & \cdots & 1 \\
    v_1 & v_2 & \cdots & v_{q^m} 
\end{pmatrix},
\]
where the $v_i \in \mathbb{F}_q^m$ are pairwise distinct. In other words, each vector of $\mathbb{F}_q^m$ appears exactly once. The columns of $G$ are all and only the points of the affine chart $\mathcal{A} = \{ (1,v) \, : \, v \in \mathbb{F}_q^m \}$. Projectively, these are the normalized representative of the points of $PG(m,q) \setminus PG(H)$, where $H = \{ (0,v) \, : \, v \in \mathbb{F}_q^m\}$.

Let $V \le F_q^{m+1}$ be a $j$-dimensional subspace. We want to count the quantity $m_{G}(V)$, which corresponds to the cardinality of $V \cap \mathcal{A}$. We distinguish two cases based on the containment relation with $H$. If $V \subseteq H$, then $m_{G}(V) = 0$. 

On the other hand, if $V \nsubseteq H$, consider the projection map $\pi_0 : \mathbb{F}_q^{m+1} \rightarrow \mathbb{F}_q$ onto the first coordinate, namely $\pi_0(v) = x_0$ for every vector $v = (x_0, \dots, x_m)$. We have $\ker(\pi_0) = H$ and, since $V \nsubseteq H$, the map restricted to $V$ is nonzero and, therefore, surjective. The rank-nullity theorem yields
\[
\dim(\ker(\pi_0\vert_V)) = \dim(V) - \dim(\operatorname{Im}(\pi_0\vert_V)) = j-1,
\]
or, equivalently, $\dim(V \cap H) = j-1$. 
Let $v' \in V \cap \mathcal{A}$ (such a vector exists since the map $\pi_0\vert_V$ is surjective). It is not difficult to see that
\[
V \cap \mathcal{A} = v'+ (V \cap H),
\]
which means that $V \cap \mathcal{A}$ is an affine translation of $V \cap H$. Since $\dim(V \cap H) = j-1$, we have that 
\[
m_{G}(V) = |V \cap \mathcal{A}| = q^{j-1}.
\]
Therefore, 
\[
p_{G}(V) = \begin{cases}
0 &\text{if } V \subseteq H,\\
q^{j-1-m} &\text{otherwise}.
\end{cases}
\]
Moreover, since every drawn column has the first coordinate equal to $1$, every possible chain of $\mathcal{S}$ is of the form $(0, V_1, \dots, V_{m+1})$, with $V_j \nsubseteq H$ for all $j \ge 1$.
Thus, we have 
\[
p_{G}(V_0) = 0 \quad \mbox{and} \quad p_{G}(V_j) = q^{j-1-m} \quad \mbox{for $1 \le j \le m+1$.}
\]
Combining all of the above yields
\[
\begin{split}
    \mathbb{E}[\mathcal{C}] &= \sum_{S \in \mathcal{S}} \Pr[S] \left( 1 + \sum_{j=1}^{m} \frac{1}{1-q^{j-1-m}} \right) \\
    &= 1 + \sum_{j=1}^{m} \frac{q^{m+1-j}}{q^{m+1-j}-1} = m+1 + \sum_{j=1}^{m} \frac{1}{q^j-1}.
\end{split}
\]

\label{thm:simplex_optimality}
    Let $T$ be the random variable that counts the number of draws needed to span $X$.
    We have
    \begin{equation} \label{eq:simplex_lower_bound}
        \mathbb{E}[T] \ge \sum_{j=0}^{k-1} \frac{q^k-1}{q^k-q^j}.
    \end{equation}

Equivalently, $p(0) = 0$ and $p(L) = 1/\left[\begin{smallmatrix} k \\ 1 \end{smallmatrix}\right]_{q}$ for every $1$-dimensional subspace $L \le X$.

\label{lem:simplex_optimality}
    Let $p$ be a probability distribution on $X$. For every $0 \le j \le k-1$, we have
    \[
    \sum_{\substack{V \le X \\ \dim(V) = j}} \sum_{\substack{S = (V_0, \dots, V_k) \in \mathcal{S} \\ V_j = V}} p(V) \Pr[S] \ge \frac{\left[\begin{smallmatrix} j \\ 1 \end{smallmatrix}\right]_{q}}{\left[\begin{smallmatrix} k \\ 1 \end{smallmatrix}\right]_{q}}.
    \]

From now on denote by 
\[
    A_j = \sum_{\substack{V \le X \\ \dim(V) = j}} \sum_{\substack{S = (V_0, \dots, V_k) \in \mathcal{S} \\ V_j = V}} p(V) \Pr[S].
\]
We proceed by induction on $j$. For $j=0$, the only $0$-dimensional space is $V = 0$. Thus,
\[
\begin{split}
    A_0 &= \sum_{S = (0, V_1, \dots, V_k) \in \mathcal{S}} p(0) \Pr[S] \\
    &= p(0) \sum_{S = (0, V_1, \dots, V_k) \in \mathcal{S}} \Pr[S] \\
    &=p(0) \ge \frac{\left[\begin{smallmatrix} 0 \\ 1 \end{smallmatrix}\right]_{q}}{\left[\begin{smallmatrix} k \\ 1 \end{smallmatrix}\right]_{q}}.
\end{split}
\]
Assume now that $A_j \ge \left[\begin{smallmatrix} j \\ 1 \end{smallmatrix}\right]_{q}/\left[\begin{smallmatrix} k \\ 1 \end{smallmatrix}\right]_{q}$ for some $0 \le j \le k-2$. We fix a $j$-dimensional subspace $V$ as the $j$-th subspace of a chain $(V_0, V_1, \dots, V_k)$ and compute the expected probability mass of the next subspace in the chain, conditional on $V_j = V$. We have
\[
\begin{split}
    \mathbb{E}[p(V_{j+1}) \, | \, V_j = V] &= \sum_{\substack{U \ge V \\ \dim(U) = j+1}} p(U) \Pr[V_{j+1}= U \, | \, V_j = V] \\
    &= \sum_{\substack{U \ge V \\ \dim(U) = j+1}} p(U) \frac{p(U) - p(V)}{1 - p(V)} \\
             
    
                                                                                     
    
                  
                                 
                                                                                                   
    &= \frac{1}{1-p(V)} \sum_{\substack{U \ge V \\ \dim(U) = j+1}} (p(U)-p(V))^2 + \frac{p(V)}{1-p(V)} \sum_{\substack{U \ge V \\ \dim(U) = j+1}} (p(U)-p(V)) \\
    &= \frac{1}{1-p(V)} \sum_{\substack{U \ge V \\ \dim(U) = j+1}} (p(U)-p(V))^2 + p(V) \\
    &\ge \frac{1-p(V)}{\left[\begin{smallmatrix} k-j \\ 1 \end{smallmatrix}\right]_{q}} + p(V),
\end{split}
\]
where we used Lemma~\ref{lem:dimension_increasing} in the second equality, the identity 
\[
\sum_{\substack{U \ge V\\ \dim(U) = j+1}}(p(U)-p(V)) = 1-p(V)
\]
in the second-to-last equality, and the Cauchy-Schwarz inequality
\[
\left( \sum_{\substack{U \ge V \\ \dim(U) = j+1}} (p(U)-p(V))^2 \right) \left( \sum_{\substack{U \ge V \\ \dim(U) = j+1}} 1^2 \right) \ge \left( \sum_{\substack{U \ge V \\ \dim(U) = j+1}} (p(U)-p(V)) \right)^2
\]
in the last step (recall that there are exactly $\left[\begin{smallmatrix} k-j \\ 1 \end{smallmatrix}\right]_{q}$ subspaces of dimension $j+1$ containing a given $j$-dimensional subspace).

From now on, we focus on the inequality
\[
    \sum_{\substack{U \ge V \\ \dim(U) = j+1}} p(U) \frac{p(U) - p(V)}{1 - p(V)} \ge \frac{1-p(V)}{\left[\begin{smallmatrix} k-j \\ 1 \end{smallmatrix}\right]_{q}} + p(V).
\]
Multiplying both sides by $\Pr[V_j = V]$ and then summing over all $j$-dimensional spaces yields
\begin{equation} \label{eq:auxiliary_lemma_simplex}
\begin{split}
    \sum_{\substack{V \le X \\ \dim(V) =j}} \Pr[V_j = V] &\sum_{\substack{U \ge V \\ \dim(U) = j+1}} p(U) \frac{p(U) - p(V)}{1 - p(V)} \\
    &\ge \sum_{\substack{V \le X \\ \dim(V) =j}} \Pr[V_j = V] \left( \frac{1-p(V)}{\left[\begin{smallmatrix} k-j \\ 1 \end{smallmatrix}\right]_{q}} + p(V) \right).
\end{split}
\end{equation}
Let $L(j)$ and $R(j)$ denote the left hand side and the right hand side, respectively, of Equation~\eqref{eq:auxiliary_lemma_simplex}.
The law of total probability gives
\[
\begin{split}
    \Pr[V_{j+1}=U] &= \sum_{\substack{V \le U \\ \dim(V) = j}} \Pr[V_j = V] \Pr[V_{j+1}=U \, | \, V_j = V] \\
    &= \sum_{\substack{V \le U \\ \dim(V) = j}} \Pr[V_j = V] \frac{p(U)-p(V)}{1-p(V)},
\end{split}
\]
which can be applied to $L(j)$ as follows:
\[
\begin{split}
    L(j) &=  \sum_{\substack{U \le X \\ \dim(U) =j+1}} p(U)\sum_{\substack{V \le U \\ \dim(V) = j}} \Pr[V_j = V]  \frac{p(U) - p(V)}{1 - p(V)} \\
    &= \sum_{\substack{U \le X \\ \dim(U) =j+1}} p(U)\Pr[V_{j+1}=U] \\
    &= \sum_{\substack{U \le X\\ \dim(U) =j+1}} \sum_{\substack{S = (V_0, \dots, V_k) \in \mathcal{S} \\ V_{j+1} = U}} p(U)\Pr[S] = A_{j+1}.
\end{split}
\]
On the other hand, 
\[
\begin{split}
    R(j) &=  \frac{1- \sum_{\substack{V \le X \\ \dim(V) =j}} p(V) \Pr[V_j = V]}{\left[\begin{smallmatrix} k-j \\ 1 \end{smallmatrix}\right]_{q}} + \sum_{\substack{V \le X \\ \dim(V) =j}} p(V) \Pr[V_j = V]\\
    &=  \frac{1}{\left[\begin{smallmatrix} k-j \\ 1 \end{smallmatrix}\right]_{q}} + \left( 1 -\frac{1}{\left[\begin{smallmatrix} k-j \\ 1 \end{smallmatrix}\right]_{q}}  \right) \sum_{\substack{V \le X \\ \dim(V) =j}} p(V) \Pr[V_j = V] \\
    &= \frac{1}{\left[\begin{smallmatrix} k-j \\ 1 \end{smallmatrix}\right]_{q}} + \left( 1 -\frac{1}{\left[\begin{smallmatrix} k-j \\ 1 \end{smallmatrix}\right]_{q}}  \right) \sum_{\substack{V \le X \\ \dim(V) =j}} \sum_{\substack{S = (V_0, \dots, V_k) \in \mathcal{S} \\ V_j = V}}  p(V) \Pr[S] \\
    &\ge \frac{1}{\left[\begin{smallmatrix} k-j \\ 1 \end{smallmatrix}\right]_{q}} + \left( 1 -\frac{1}{\left[\begin{smallmatrix} k-j \\ 1 \end{smallmatrix}\right]_{q}}  \right) \frac{\left[\begin{smallmatrix} j \\ 1 \end{smallmatrix}\right]_{q}}{\left[\begin{smallmatrix} k \\ 1 \end{smallmatrix}\right]_{q}} \\
    &= \frac{1-\left[\begin{smallmatrix} j \\ 1 \end{smallmatrix}\right]_{q} / \left[\begin{smallmatrix} k \\ 1 \end{smallmatrix}\right]_{q}}{\left[\begin{smallmatrix} k-j \\ 1 \end{smallmatrix}\right]_{q}} + \frac{\left[\begin{smallmatrix} j \\ 1 \end{smallmatrix}\right]_{q}}{\left[\begin{smallmatrix} k \\ 1 \end{smallmatrix}\right]_{q}}  \\
    &= \frac{q^j}{\left[\begin{smallmatrix} k \\ 1 \end{smallmatrix}\right]_{q}} + \frac{\left[\begin{smallmatrix} j \\ 1 \end{smallmatrix}\right]_{q}}{\left[\begin{smallmatrix} k \\ 1 \end{smallmatrix}\right]_{q}} = \frac{\left[\begin{smallmatrix} j+1 \\ 1 \end{smallmatrix}\right]_{q}}{\left[\begin{smallmatrix} k \\ 1 \end{smallmatrix}\right]_{q}},
\end{split}
\]
where we used the induction hypothesis in the third-to-last step and the combinatorial identity
\[
\left[\begin{smallmatrix} k \\ 1 \end{smallmatrix}\right]_{q} - \left[\begin{smallmatrix} j \\ 1 \end{smallmatrix}\right]_{q} = q^j \left[\begin{smallmatrix} k-j \\ 1 \end{smallmatrix}\right]_{q}
\]
in the second-to-last step.
Combining all of the above gives
\[
A_{j+1} \ge \frac{\left[\begin{smallmatrix} j+1 \\ 1 \end{smallmatrix}\right]_{q}}{\left[\begin{smallmatrix} k \\ 1 \end{smallmatrix}\right]_{q}}.
\]

We can rewrite the formula from Theorem~\ref{thm:general_formula} conditioning on the $j$-th space in the chain $S = (V_0, \dots, V_k)$, namely
\begin{equation} \label{eq:gen_formula_expectation}
    \begin{split}
    \mathbb{E}[T] &= \sum_{j=0}^{k-1} \sum_{S = (V_0, \dots, V_k) \in \mathcal{S}} \frac{\Pr[S]}{1-p(V_j)} \\
    &= \sum_{j=0}^{k-1} \sum_{\substack{V \le X \\ \dim(V) = j}} \sum_{\substack{S = (V_0, \dots, V_k) \in \mathcal{S} \\ V_j = V}} \frac{\Pr[S]}{1-p(V)}.
\end{split}
\end{equation}
Since the function $\varphi(x) = 1/(1-x)$ is convex for $0 \le x <1$, Jensen's inequality gives
\begin{equation} \label{eq:optimization_jensen}
    \mathbb{E}[T] \ge \sum_{j=0}^{k-1}\frac{1}{1-\sum_{\substack{V \le X \\ \dim(V) = j}} \sum_{\substack{S = (V_0, \dots, V_k) \in \mathcal{S} \\ V_j = V}} p(V) \Pr[S]  }.
\end{equation}
Applying Lemma~\ref{lem:simplex_optimality} to Equation~\eqref{eq:optimization_jensen} yields
\[
    \mathbb{E}[T]  \ge \sum_{j=0}^{k-1} \frac{1}{1-\frac{\left[\begin{smallmatrix} j \\ 1 \end{smallmatrix}\right]_{q}}{\left[\begin{smallmatrix} k \\ 1 \end{smallmatrix}\right]_{q}}} = \sum_{j=0}^{k-1} \frac{q^k-1}{q^k-q^j}.
\]

It remains to characterize equality. 
From now on, for $0\leq j\leq k-1$, let 
\[
B_j = \sum_{\substack{V\leq X\\\dim(V)=j}} \sum_{\substack{S\in\mathcal{S}\\V_j=V}} \frac{\Pr[S]}{1-p(V)}. 
\] 
The preceding argument shows, term by term, that 
\begin{equation} \label{eq:inequalities_chain}
    B_j \geq \frac{1}{1-A_j} \geq \frac{1}{ 1- \dfrac{\left[\begin{smallmatrix} j \\ 1 \end{smallmatrix}\right]_{q}} {\left[\begin{smallmatrix} k \\ 1 \end{smallmatrix}\right]_{q}} }. 
\end{equation}
Suppose that equality holds in~\eqref{eq:simplex_lower_bound}. 
Therefore, the inequalities in~\eqref{eq:inequalities_chain} must also be equalities. For $j=0$, we have $V_0=\{0\}$ with probability one and hence
\[ B_0=\frac{1}{1-p(0)}. \] 
The corresponding term in the lower bound is $1$, which implies $p(0)=0$.

We next consider $j=1$. Since $p(0)=0$, the first draw is nonzero with probability one. Thus, for every $L\in\mathcal{P}_k$, we have $\Pr[V_1=L]=p(L).$ 
It follows that 
                                                                    
\[ A_1                       
= \sum_{L\in\mathcal{P}_k}p(L)^2. \] 
            
                                        
                                                                                                                                                              
The fact that the equalities must hold in~\eqref{eq:inequalities_chain} implies that $A_1 = 1/\left[\begin{smallmatrix} k \\ 1 \end{smallmatrix}\right]_{q}.$ 
Therefore, 
\[ \sum_{L\in\mathcal{P}_k}p(L)^2 = \frac{1}{\left[\begin{smallmatrix} k \\ 1 \end{smallmatrix}\right]_{q}}. \] 
On the other hand,                                       
the Cauchy-Schwarz inequality gives
\[ \sum_{L\in\mathcal{P}_k}p(L)^2 \geq \frac{1}{|\mathcal{P}_k|} \left(\sum_{L\in\mathcal{P}_k}p(L)\right)^2 = \frac{1}{\left[\begin{smallmatrix} k \\ 1 \end{smallmatrix}\right]_{q}}, \] 
and equality holds if and only if all the values $p(L)$ are equal. Consequently, $p(L)=1/\left[\begin{smallmatrix} k \\ 1 \end{smallmatrix}\right]_{q}$ for every $L\in\mathcal{P}_k$.

Conversely, suppose that $p(0)=0$ and that $p(L)=1/\left[\begin{smallmatrix} k \\ 1 \end{smallmatrix}\right]_{q}$ 
for every one-dimensional subspace $L\leq X$. If $V\leq X$ has dimension $j$, then $V$ contains exactly $\left[\begin{smallmatrix} j \\ 1 \end{smallmatrix}\right]_{q}$ one-dimensional subspaces. Therefore, 
\[ p(V) = \frac{\left[\begin{smallmatrix} j \\ 1 \end{smallmatrix}\right]_{q}}{\left[\begin{smallmatrix} k \\ 1 \end{smallmatrix}\right]_{q}}. \] 
In particular, $p(V)$ depends only on $\dim(V)$. Substituting back into~\eqref{eq:gen_formula_expectation}, we obtain
\begin{align*} 
\mathbb{E}[T] &= \sum_{j=0}^{k-1} \frac{1}{ 1- \dfrac{\left[\begin{smallmatrix} j \\ 1 \end{smallmatrix}\right]_{q}} {\left[\begin{smallmatrix} k \\ 1 \end{smallmatrix}\right]_{q}} } \sum_{S\in\mathcal{S}}\Pr[S]\\ &= \sum_{j=0}^{k-1} \frac{q^k-1}{q^k-q^j}. \end{align*} 
Thus the lower bound is attained, completing the proof.

\label{cor:simplex-unique-optimal-code} 
Let $n=(q^k-1)/(q-1)$, and let $G\in\mathbb{F}_q^{k\times n}$ have rank $k$. Then 
\[
\mathbb{E}[G] \geq \sum_{j=0}^{k-1} \frac{q^k-1}{q^k-q^j}.

\[
\mathbb{E}_{\text{opt}}\left[\frac{q^k-1}{q-1},k\right]_q = \sum_{j=0}^{k-1} \frac{q^k-1}{q^k-q^j}.
\]
Consequently, every optimal $\left[\frac{q^k-1}{q-1},k\right]_q$ code is monomially equivalent to the $q$-ary simplex code.

Recall that $z_A=\left|\{i\in[n]:a_i=0\}\right|$.
Let $\tau$ be the first time that a nonzero column is sampled. Since a sampled column is nonzero with probability $(n-z_A)/n$, we have
\[
\mathbb{E}[\tau]=\frac{n}{n-z_A}.
\]

Suppose that the first nonzero column sampled is $a_j$. Consider the quotient space
\[
X_j=\mathbb{F}_q^{m+1}/\langle a_j\rangle,
\]
which has dimension $m$, and let
\[
\pi_j:\mathbb{F}_q^{m+1}\longrightarrow X_j
\]
be the natural projection. Once $a_j$ has been sampled, spanning $\mathbb{F}_q^{m+1}$ is equivalent to having the projections of the subsequently sampled columns span $X_j$.

Let
\[
d_j=\left|\{i\in[n]:a_i\in\langle a_j\rangle\}\right|.
\]
A projected column is zero precisely when the sampled column belongs to $\langle a_j\rangle$. Hence, a nonzero projected column is obtained with probability
\[
\frac{n-d_j}{n}.
\]
Moreover, since all zero columns belong to $\langle a_j\rangle$ and $a_j\neq0$, we have
\[
d_j\geq z_A+1.
\]

Conditioned on obtaining a nonzero projection, the projected columns induce a probability distribution on $X_j\setminus\{0\}$. Since $A$ has rank $m+1$, the support of this distribution spans $X_j$. Therefore, by Theorem~\ref{thm:simplex_optimality}, the expected number of nonzero projected columns needed to span $X_j$ is at least the coverage depth of the $q$-ary simplex code of dimension $m$. Denote a generator matrix of this simplex code by $S_m$.

Since a nonzero projected column is obtained with probability $(n-d_j)/n$, it follows that, conditioned on $a_j$ being the first nonzero sampled column, the expected number of additional draws is at least
\[
\frac{n}{n-d_j}\mathbb{E}[S_m].
\]
Combining this with $d_j\geq z_A+1$ yields
\[
\mathbb{E}[A]
\geq
\frac{n}{n-z_A}
+
\frac{n}{n-z_A-1}\mathbb{E}[S_m].
\]
Consequently,
\[
\mathbb{E}[A]
\geq
1+\frac{n}{n-1}\mathbb{E}[S_m].
\]

It remains to show that equality is attained by the first-order Reed--Muller code. A natural generator matrix for this code has columns
\[
r_v=(1,v),\qquad v\in\mathbb{F}_q^m.
\]
In particular, it has no zero columns. Fix a column $r_{v_0}=(1,v_0)$. Since all columns have first coordinate equal to $1$, no other column lies in $\langle r_{v_0}\rangle$, and hence $d_j=1$.

Furthermore, modulo $\langle r_{v_0}\rangle$, the column $r_v=(1,v)$ can be represented by
\[
r_v-r_{v_0}=(0,v-v_0).
\]
As $v$ ranges over $\mathbb{F}_q^m\setminus\{v_0\}$, the vectors $v-v_0$ range over all nonzero elements of $\mathbb{F}_q^m$ exactly once. Thus, conditioned on obtaining a nonzero projection, the induced distribution on the one-dimensional subspaces of $X_j$ is uniform. By the equality condition in Theorem~\ref{thm:simplex_optimality}, the corresponding projected coverage process attains $\mathbb{E}[S_m]$.

Therefore,
\[
\mathbb{E}[R]
=
1+\frac{n}{n-1}\mathbb{E}[S_m].
\]
Since $n=q^m$ and
\[
\mathbb{E}[S_m]
=
\sum_{\ell=0}^{m-1}
\frac{q^m-1}{q^m-q^\ell},
\]
we obtain
\[
\begin{split}
\mathbb{E}[R]
&=
1+\sum_{\ell=0}^{m-1}
\frac{q^m}{q^m-q^\ell}\\
&=
1+\sum_{j=1}^{m}\frac{q^j}{q^j-1} =
m+1+\sum_{j=1}^{m}\frac{1}{q^j-1}.
\end{split}
\]
Hence $\mathbb{E}[A]\geq\mathbb{E}[R]$, completing the proof.

<!-- defn-2: body_md -->
A matroid is a pair \(M=(E,\mathcal{I})\), where \(E\) is a finite set and \(\mathcal{I}\) is a nonempty collection of subsets of \(E\) satisfying the following properties:
\begin{enumerate}
    \item If \(T\in\mathcal{I}\) and \(S\subseteq T\), then \(S\in\mathcal{I}\).
    \item If \(S,T\in\mathcal{I}\) and \(|S|<|T|\), then there exists \(i\in T\setminus S\) such that \(S\cup\{i\}\in\mathcal{I}\).
\end{enumerate}
The sets in \(\mathcal{I}\) are called independent sets. A basis is a maximal independent set, and the rank of \(M\) is the cardinality of any basis.

\[
g_M(\mathbf{z})=\sum_{B\in\mathcal{B}_M}\prod_{e\in B}z_e.

Every linearly independent subset of \(t\) columns of \(G\) consists of nonzero columns with distinct and linearly independent spans. It therefore determines a linearly independent set \(P\subseteq\mathcal{P}_r\) of size \(t\).

Conversely, fix a linearly independent set \(P\subseteq\mathcal{P}_r\) of size \(t\). For each \(p\in P\), one may choose any of the \(m_G(p)\) columns of \(G\) whose span is \(p\). Hence, the number of independent column subsets associated with \(P\) is \(\prod_{p\in P}m_G(p)\). Summing over all linearly independent sets \(P\) of size \(t\) gives
\[
I(G,t)=\sum_{\substack{P\subseteq\mathcal{P}_r\\ |P|=t,\ P\text{ is linearly independent}}}\prod_{p\in P}m_G(p)=g_{r,t}(m_G).
\]

<!-- proof-10: body_md -->
Let \(A\in\mathbb{F}_q^{k\times n}\) have rank \(k\), and let \(B\in\mathbb{F}_q^{r\times n}\) be a parity-check matrix of \(\mathcal{C}_A\). Let \(S\) be a \(q\)-ary simplex matrix of dimension \(r\). The columns of \(S\) contain exactly one nonzero representative of each element of \(\mathcal{P}_r\). Therefore, \(m_S=\mathbf{1}\), and it is sufficient to prove that
\[
g_{r,t}(m_B)\leq g_{r,t}(\mathbf{1})
\]
for every \(1\leq t\leq r\).

The group \(\mathrm{GL}(r,q)\) acts transitively on \(\mathcal{P}_r\). For \(\sigma\in\mathrm{GL}(r,q)\) and \(x\in\mathbb{R}^{\mathcal{P}_r}\), define
\[
(\sigma x)(p)=x(\sigma^{-1}p).
\]
Since every \(\sigma\in\mathrm{GL}(r,q)\) preserves linear independence, it permutes the bases of \(M_{r,t}\). Consequently,
\[
g_{r,t}(\sigma x)=g_{r,t}(x).
\]

Define the orbit average
\[
\overline{m}_B=\frac{1}{|\mathrm{GL}(r,q)|}\sum_{\sigma\in\mathrm{GL}(r,q)}\sigma m_B.
\]
If \(g_{r,t}(m_B)=0\), then the desired inequality is immediate. Assume therefore that \(g_{r,t}(m_B)>0\).

For every \(\varepsilon>0\), the vector \(m_B+\varepsilon\mathbf{1}\) lies in the positive orthant. Since \(\mathbf{1}\) is fixed by the action of \(\mathrm{GL}(r,q)\), the invariance of \(g_{r,t}\) and Jensen's inequality imply
\[
\log g_{r,t}(m_B+\varepsilon\mathbf{1})\leq \log g_{r,t}(\overline{m}_B+\varepsilon\mathbf{1}).
\]
The logarithm is monotonically increasing, and therefore
\[
g_{r,t}(m_B+\varepsilon\mathbf{1})\leq g_{r,t}(\overline{m}_B+\varepsilon\mathbf{1}).
\]
Letting \(\varepsilon\rightarrow0\) and using the continuity of \(g_{r,t}\), we obtain
\[
g_{r,t}(m_B)\leq g_{r,t}(\overline{m}_B).
\]

The transitivity of the action of \(\mathrm{GL}(r,q)\) on \(\mathcal{P}_r\) implies that all coordinates of \(\overline{m}_B\) are equal. Hence,
\[
\overline{m}_B=c\mathbf{1}
\]
for some \(c\geq0\).

The action of \(\mathrm{GL}(r,q)\) only permutes the coordinates of \(m_B\), so
\[
\sum_{p\in\mathcal{P}_r}\overline{m}_B(p)=\sum_{p\in\mathcal{P}_r}m_B(p)=\left|\left\{i\in[n]: b_i\neq0\right\}\right|.
\]
Since \(|\mathcal{P}_r|=n\), it follows that
\[
c=\frac{\left|\left\{i\in[n]: b_i\neq0\right\}\right|}{n}\leq1.
\]
In particular, \(c=1\) if \(B\) has no zero columns, while \(c<1\) otherwise.

Since \(g_{r,t}\) is homogeneous of degree \(t\),
\[
g_{r,t}(\overline{m}_B)=g_{r,t}(c\mathbf{1})=c^t g_{r,t}(\mathbf{1})\leq g_{r,t}(\mathbf{1})=g_{r,t}(m_S).
\]
Combining the two inequalities gives
\[
I(B,t)=g_{r,t}(m_B)\leq g_{r,t}(m_S)=I(S,t)
\]
for every \(1\leq t\leq r\).

Finally, the dual expression for the expected coverage time yields
\[
\mathbb{E}[A]=nH_n-\sum_{t=1}^{r}\frac{I(B,t)}{\binom{n-1}{t-1}}
\geq nH_n-\sum_{t=1}^{r}\frac{I(S,t)}{\binom{n-1}{t-1}}
=\mathbb{E}[H],
\]
which is the desired result.

\label{lem:random-exact}
Let \(A\) be sampled uniformly from \(\mathcal{M}_{k,n}(q)\). Then
\begin{equation}
\label{eq:random-exact}
\begin{aligned}
\mathbb{E}[T]
&=
n\left(H_n-H_{n-k}\right) \\
&\quad+
n\sum_{t=k}^{n-1}
\frac{1}{n-t}
\left(
1-
q^{k(n-t)}
\prod_{i=0}^{k-1}
\frac{q^t-q^i}{q^n-q^i}
\right).
\end{aligned}
\end{equation}

Equivalently,
\begin{equation}
\label{eq:random-exact-product}
\mathbb{E}[T]
=
n\left(H_n-H_{n-k}\right)
+
n\sum_{t=k}^{n-1}
\frac{1}{n-t}
\left(
1-
\prod_{i=0}^{k-1}
\left(
1-
\frac{q^{i-t}-q^{i-n}}
{1-q^{i-n}}
\right)
\right).
\end{equation}

<!-- proof-11: body_md -->
Let \(D\) denote the number of distinct column indices sampled by the
time the sampled columns first attain rank \(k\). Conditioned on \(D=d\), the number of samples required is the
waiting time to observe \(d\) distinct elements in the classical
coupon-collector process. Hence, by the partial coupon-collector
formula \cite[p.~216]{flajolet1992birthday},
\[
\mathbb{E}[T\mid D=d]
=
n\left(H_n-H_{n-d}\right).
\]
By the law of total expectation,
\[
\mathbb{E}[T]
=
\sum_{d=0}^{n}
n\left(H_n-H_{n-d}\right)\Pr(D=d).
\]
Since
\[
H_n-H_{n-d}
=
\sum_{j=n-d+1}^{n}\frac{1}{j},
\]
we obtain
\[
\mathbb{E}[T]
=
\sum_{d=0}^{n}
n
\left(
\sum_{j=n-d+1}^{n}\frac{1}{j}
\right)
\Pr(D=d).
\]

Interchanging the order of summation gives
\[
\mathbb{E}[T]
=
n\sum_{j=1}^{n}
\frac{1}{j}
\Pr(D\geq n-j+1).
\]
Setting \(t=n-j\), we obtain
\[
\mathbb{E}[T]
=
n\sum_{t=0}^{n-1}
\frac{1}{n-t}\Pr(D>t).
\]

Since at least \(k\) distinct columns are required to attain rank \(k\),
\[
\Pr(D>t)=1,
\qquad
0\leq t\leq k-1.
\]
Consequently,
\begin{equation}
\label{eq:random-tail}
\mathbb{E}[T]
=
n\left(H_n-H_{n-k}\right)
+
n\sum_{t=k}^{n-1}
\frac{1}{n-t}\Pr(D>t).
\end{equation}

It remains to compute \(\Pr(D>t)\). The first \(t\) distinct sampled
indices form a uniformly random ordered \(t\)-tuple of distinct
indices. Moreover, the distribution of \(A\) is invariant under column
permutations. Therefore, if \(A_{[t]}\) denotes the submatrix formed by
the first \(t\) columns of \(A\), then
\[
\Pr(D>t)
=
\Pr\left(\operatorname{rank}(A_{[t]})<k\right).
\]
Hence,
\[
\Pr(D>t)
=
1-\Pr\left(\operatorname{rank}(A_{[t]})=k\right).
\]

The number of full-rank matrices in
\(\mathbb{F}_q^{k\times n}\) is
\[
\left|\mathcal{M}_{k,n}(q)\right|
=
\prod_{i=0}^{k-1}(q^n-q^i).
\]
Indeed, after choosing \(i\) linearly independent rows, their span has
cardinality \(q^i\), and the next row can be chosen in
\(q^n-q^i\) ways.

To count the full-rank matrices whose first \(t\) columns already have
rank \(k\), write
\[
A=[\,B\mid C\,],
\]
where
\[
B\in\mathbb{F}_q^{k\times t},
\qquad
C\in\mathbb{F}_q^{k\times(n-t)}.
\]
The matrix \(B\) must have rank \(k\), whereas \(C\) may be arbitrary.
Hence, the number of such matrices is
\[
q^{k(n-t)}
\prod_{i=0}^{k-1}(q^t-q^i).
\]

It follows that
\[
\Pr\left(\operatorname{rank}(A_{[t]})=k\right)
=
q^{k(n-t)}
\prod_{i=0}^{k-1}
\frac{q^t-q^i}{q^n-q^i}.
\]
Substituting this into \eqref{eq:random-tail} proves
\eqref{eq:random-exact}.

Finally,
\[
q^{n-t}
\frac{q^t-q^i}{q^n-q^i}
=
\frac{1-q^{i-t}}{1-q^{i-n}}
=
1-
\frac{q^{i-t}-q^{i-n}}
{1-q^{i-n}}.
\]
Distributing \(q^{k(n-t)}\) over the \(k\) factors in the product gives
\eqref{eq:random-exact-product}.

We have
\[
\begin{split}
    S_{m+1}(q) &= \sum_{j=1}^{m+1} \frac{m+1}{j}q^{j-m-1}
    = 1+ \sum_{j=1}^{m} \frac{m+1}{j}q^{j-m-1} \\
    &= 1+ \frac{m+1}{qm} \sum_{j=1}^{m} \frac{m}{j}q^{j-m} = 1+\frac{m+1}{qm}S_m(q),
\end{split}
\]
which proves \eqref{eq:Sm-recurrence}.

We next show that \(S_m(q)\) is bounded. Setting \(r=m-j\), we obtain
\[
S_m(q)
=
\sum_{r=0}^{m-1}
\frac{m}{m-r}q^{-r}.
\]
Note that, for \(0\leq r\leq m-1\),
\[
\frac{m}{m-r}\leq 1+r.
\]
Equivalently, 
\[
m
\leq
(1+r)(m-r)
=
m+r(m-r-1).
\]

It follows that
\[
S_m(q)
\leq
\sum_{r=0}^{m-1}(1+r)q^{-r}
\leq
\sum_{r=0}^{\infty}(1+r)q^{-r}
<\infty.
\]
Thus, the sequence is bounded.

Let
\[
\overline{S}
=
\limsup_{m\to\infty}S_m(q),
\qquad
\underline{S}
=
\liminf_{m\to\infty}S_m(q).
\]
Since shifting a sequence does not change its limit superior and
\(
\frac{m+1}{qm}\longrightarrow\frac{1}{q},
\)
Equation~\eqref{eq:Sm-recurrence} gives
\(
\overline{S}
=
1+\frac{1}{q}\overline{S}.
\)
Therefore,
\(
\overline{S}
=
\frac{q}{q-1}.
\) Similarly,
\(
\underline{S}
=
1+\frac{1}{q}\underline{S},
\)
and hence
\(
\underline{S}
=
\frac{q}{q-1}.
\)
Thus,
\(
\underline{S}
=
\overline{S}
=
\frac{q}{q-1},
\)
which proves that
\(
S_m(q)\longrightarrow\frac{q}{q-1}.

\begin{equation}
\label{eq:random-gap-bound}
\Delta_{n,k,q}
\leq
\frac{1}{1-q^{-m}}

\cdot
\frac{n}{m}.
\end{equation}

a_i
=
\frac{q^{i-t}-q^{i-n}}{1-q^{i-n}}.

Since \(k\leq t<n\) and \(i<k\), we have $0<a_i<1.$
Using Weierstrass Product Inequality,
    
                  
      
             
    
equation~\eqref{eq:random-exact-product} gives

\label{thm:random-constant-rate}
Let \(q\) and \(R\in(0,1)\) be fixed, and suppose that
\[
\frac{k}{n}=R.
\]
Then, as \(n\to\infty\),
\[
\mathbb{E}[T]
\leq
n\left(H_n-H_{n-k}\right)
+
\frac{q}{(q-1)^2}\cdot\frac{1}{1-R}
+
o(1).
\]
Equivalently, using \(n=k/R\) and the asymptotic expansion of the
harmonic numbers,
\[
\mathbb{E}[T]
\leq
n\log\left(\frac{1}{1-R}\right)
+
\frac{1}{1-R}
\left(
\frac{q}{(q-1)^2}-\frac{R}{2}
\right)
+
o(1).
\]

In particular, the additive gap between the expected coverage depth of
a random code and the MDS benchmark is bounded independently of \(n\).
