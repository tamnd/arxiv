<!-- theorem-1: body_md -->
\label{thm:multipoint}
For every $c\ge0$, $t\ge0$, initial profile $h\in\widehat{\mathcal X}$ as defined in \eqref{eqn:state-space}, and the observation point \(0\leq x_1<\cdots<x_n\) and \(r_1,\ldots,r_n \in \mathbb{Z}\),
\begin{equation}\label{eq:multi-result}
 \mathbb P_h^c\bigl(h(t,x_j)\le r_j,\ 1\le j\le n\bigr)
 =\widehat F_n(t,h)=\operatorname{Pf}(J+JK(t,h)).
\end{equation}

<!-- definition-1: body_md -->
\label{def:regular}
For the fixed positions \(\boldsymbol x = (x_1,\ldots,x_n)\), let $\mathcal R_{\boldsymbol x}$
be the set of pairs $(t,h)$ with $t>0$ and $h\in\widehat{\mathcal X}_{\mathrm{fin}}$

      \item \(t\ne x_j,\qquad 2t\ne|x_i-x_j|\ (i\ne j),\qquad

\end{enumerate}

<!-- lemma-1: body_md -->
\label{lem:firsthit}
For $h\in\widehat{\mathcal X}_{\mathrm{fin}}$, $0\le a\le b$, and $z\in[a,b]$,
\begin{equation}\label{eq:first-hit-change}
 P_{a,b}^{C_zh}-P_{a,b}^h
  =Q_{a,z}^h(\delta_{h(z)+1}\otimes\delta_{h(z)+1})Q_{z,b}^h.
\end{equation}
At the endpoints,
\begin{equation}\label{eq:first-hit-endpoints}
 \begin{split}
 P_{a,b}^{C_ah}-P_{a,b}^h
   &=\delta_{h(a)+1}\otimes Q_{a,b}^h(h(a)+1,\cdot),\\
 P_{a,b}^{C_bh}-P_{a,b}^h
   &=Q_{a,b}^h(\cdot,h(b)+1)\otimes\delta_{h(b)+1}.
 \end{split}
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Since $C_zh\ge h$, the difference
$P_{a,b}^{C_zh}(u,v)-P_{a,b}^h(u,v)$ is the probability
of hitting $C_zh$ but not $h$, and ending at $v$.
The two profiles differ only at $z$, where the height is
increased by one. So this is exactly
\[
 \mathbb{P}_{a,u}(X(y)>h(y)\text{ for all }y\in[a,b],\quad X(z)=h(z)+1,\quad X(b)=v),
\]
which is $Q_{a,z}^h(u,h(z)+1)Q_{z,b}^h(h(z)+1,v)$ by the Markov property. At an endpoint,
$Q_{z,z}^h=1_{h(z)}$ leaves $\delta_{h(z)+1}$ unchanged and gives
\eqref{eq:first-hit-endpoints}.

<!-- proposition-2: body_md -->
\label{prop:rank-two}
For $z\in[0,\infty)$, $p,q\neq 0$, define \(a_z:=Q_{0,z}^h\delta_{h(z)+1}\) and define
$\beta_p$ and $u_p$ by
\[
 \begin{split}
 \beta_p&=
 \begin{cases}
  \delta_{h(z)+1}Q_{z,p}^he^{-p\Delta}H_-^{-1},&p>0,\ z\le p,\\
  0,&\text{otherwise},
 \end{cases}\\
 u_p&=
 \begin{cases}
  (I+T_p^*H)Q_{0,z}^h\delta_{h(z)+1},&p>0,\\
  e^{-p\Delta}Q_{-p,z}^h\delta_{h(z)+1},&p<0,\ -p\le z,\\
  0,&p<0,\ z<-p.
 \end{cases}
 \end{split}
\]
Then,
\begin{equation}\label{eq:rank-two-W}
 \delta_zW_{p,q}=u_p\otimes\beta_q-\beta_p\otimes u_q.
\end{equation}
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
We check the identities by cases. 

      \emph{When $p,q>0$.}
For $z\le p$, Lemma~\ref{lem:firsthit} gives
\[
 \delta_zT_p
 =Q_{0,z}^h(\delta_{h(z)+1}\otimes \delta_{h(z)+1})Q_{z,p}^he^{-p\Delta}H_-^{-1}
 =a_z\otimes\beta_p.
\]
For $z>p$, both sides vanish. By definition of \(W_{p,q}\) in \eqref{eq:Wpq},
\begin{equation}\label{eqn:deltaWpq}
       \delta_zW_{p,q}
  =\delta_zT_q-(\delta_zT_p)^*
       +T_p^*H\delta_zT_q+(\delta_zT_p)^*HT_q+(\delta_zT_p)^*H\delta_zT_q.\\
\end{equation}
Notice that 
\begin{equation}\label{eq:quadratic-jump-zero}
      (\delta_zT_p)^*H\delta_zT_q
 =\left(\sum_{u,v}a_z(u)H(u,v)a_z(v)\right)
                        \beta_p\otimes\beta_q=0.
\end{equation}
The scalar sum is absolutely convergent: the bound
$0\le a_z(u)\le e^{z\Delta}(u,h(z)+1)$ gives $\|a_z\|_1\le1$,
and $H$ has bounded entries. Exchanging $u,v$ negates the sum by
$H(v,u)=-H(u,v)$, so it is zero. The remaining terms in \eqref{eqn:deltaWpq} factor as
\[
 \begin{split}
 \delta_zW_{p,q}
 &=(I+T_p^*H)(a_z\otimes\beta_q)
       +(\beta_p\otimes a_z)(HT_q-I)=u_p\otimes\beta_q-\beta_p\otimes u_q,
 \end{split}
\]
because $(HT_q-I)^*a_z=-(I+T_q^*H)a_z=-u_q$.

\noindent
For $p<0<q$ and $-p\le z\le q$, 
\[
 \delta_zW_{p,q}=\delta_zT_{-p,q}
 =e^{-p\Delta}Q_{-p,z}^h(\delta_{h(z)+1}\otimes \delta_{h(z)+1})
                  Q_{z,q}^he^{-q\Delta}H_-^{-1}
 =u_p\otimes\beta_q.
\]
Since $\beta_p=0$, this is \eqref{eq:rank-two-W}. For \(z\) outside the
interval \([-p,q]\), $u_p=0$ and $\beta_p=0$. If $-p>q$, then for every $z$ either $z<-p$ or $z>q$,
so at least one of $u_p$ and $\beta_q$ vanishes.
Hence $u_p\otimes\beta_q=0$, in agreement with
$W_{p,q}=0$ for an empty interval. The endpoint identities in
Lemma~\ref{lem:firsthit} also cover $-p=z=q$.
For $q<0<p$, skew symmetry gives
$\delta_zW_{p,q}=-(\delta_zW_{q,p})^*=-\beta_p\otimes u_q$,
and $\beta_q=0$. If $p,q<0$, $W_{p,q}=\beta_p=\beta_q=0$.
For a nucleation at the origin, $a_0=Q_{0,0}^h\delta_{h(0)+1}=1_{h(0)}\delta_{h(0)+1}=\delta_{h(0)+1}$;
all negative-endpoint factors vanish, so the same formulas apply at
$z=0$.

<!-- proposition-3: body_md -->
\label{prop:hit-generator}
Let $0\le a<b$. Suppose \(h\in \widehat{\mathcal X}_{\mathrm{fin}}\) has no jump at \(a,b\) and \(h\) has no isolated peaks, i.e. \(h(z) = \max\{h(z^-),h(z^+)\}\). For every $c\ge0$ and $a>0$,
\begin{equation}\label{eq:LP-interior}
 \mathcal L_c P_{a,b}^h=2[P_{a,b}^h,\nabla]+\mathcal E_a+\mathcal E^b.
\end{equation}
For $a=0$,
\begin{equation}\label{eq:LP-origin}
 \mathcal L_c P_{0,b}^h
 =2[P_{0,b}^h,\nabla]+(I-c\tau_-)\mathcal E_0+\mathcal E^b.
\end{equation}
Here $[A,B]=AB-BA$. Jump sizes need not be one.
<!-- end proposition-3 -->
<!-- proof-3: body_md -->
Write $P=P_{a,b}^h$, $Q=Q_{a,b}^h$, and denote the three contributions
in \eqref{eq:hit-generator-expanded}, in their displayed order, by
$\mathsf M$ (motion), $\mathsf B$ (bulk), and $\mathsf W$ (wall).
The endpoints $a,b$ remain fixed throughout.

By our assumption on the height function \(h\) in \eqref{eqn:state-space}. There exists $a=z_0<z_1<\cdots<z_N<z_{N+1}=b$ such that \(h\) is constant $m_j$ on $(z_j,z_{j+1})$. Put
$B_j=Q_{z_{j+1}-z_j}^{(m_j)}$. The Markov property gives
\begin{equation}\label{eq:plateau-product}
 Q:=Q^{h}_{a,b}= B_0\cdots B_N,\qquad P:=P^{h}_{a,b}=e^{(b-a)\Delta}-Q^{h}_{a,b}.
\end{equation}
In the following proof, we will omit the explicit dependence on \(a,b,h\). By our assumption on \(h\), $h(z_j)=\max(m_{j-1},m_j)$, and the adjacent projections
$1_{m_{j-1}}1_{m_j}=1_{\max(m_{j-1},m_j)}$ impose the no-hit restriction.
The endpoint assumptions give $m_0=h(a)$ and $m_N=h(b)$.

We first consider the deterministic shift motion $\mathsf M$. 
At $z_j$, write $m=m_{j-1}$, $n=m_j$,
$\sigma_j=\operatorname{sgn}(n-m)$,
$Q_L=B_0\cdots B_{j-1}$, and $Q_R=B_j\cdots B_N$.
Since $\partial_sQ_s^{(m)}=\Delta_mQ_s^{(m)}=Q_s^{(m)}\Delta_m$,
\[
 \partial_{z_j}Q=Q_L(\Delta_m-\Delta_n)Q_R,
 \qquad -\sigma_j\partial_{z_j}P
       =\sigma_jQ_L(\Delta_m-\Delta_n)Q_R.
\]
The compressed difference is
\begin{equation}\label{eq:stack-generator-difference}
 1_m(\Delta_m-\Delta_n)1_n
 =\begin{cases}
 (1_m-1_n)\Delta1_n=\delta_n\otimes\delta_{n+1},&n>m,\\
 -1_m\Delta(1_n-1_m)=-\delta_{m+1}\otimes\delta_m,&n<m.
 \end{cases}
\end{equation}
Using the fact that $Q_L=Q_L1_m$ and $Q_R=1_nQ_R$, both cases give
\begin{equation}\label{eqn:deterministicshift}
       M_j:=-\sigma_j\partial_{z_j}P
 =Q_L(\delta_{m+1}\otimes\delta_m+
                      \delta_n\otimes\delta_{n+1})Q_R.
\end{equation}
The first summand vanishes when $n>m$, and the second when $n<m$.
Thus $\mathsf M=\sum_{j=1}^NM_j$. The sum is
zero when $N=0$.

Now we consider the bulk nucleation. First consider a constant interval of height $m$ and length $s$.
Set $B_+=1_m\tau_+1_m$ and $B_-=1_m\tau_-1_m$.
The relations $B_+B_-=1_m$ and
$B_-B_+=1_m-\delta_{m+1}\otimes\delta_{m+1}$ imply
\[
 [\Delta_m,\nabla_m]=-[B_+,B_-]
                  =-\delta_{m+1}\otimes\delta_{m+1}.
\]
Consequently,
\[
 \frac{d}{dz}\bigl(Q_z^{(m)}\nabla_mQ_{s-z}^{(m)}\bigr)
 =Q_z^{(m)}[\Delta_m,\nabla_m]Q_{s-z}^{(m)}
 =-Q_z^{(m)}(\delta_{m+1}\otimes\delta_{m+1})Q_{s-z}^{(m)}.
\]
Integrating from $0$ to $s$, using $Q_0^{(m)}=1_m$, gives
\begin{equation}\label{eq:plateau-duhamel}
 2\int_0^s Q_z^{(m)}(\delta_{m+1}\otimes\delta_{m+1})Q_{s-z}^{(m)}\,dz
 =-2[Q_s^{(m)},\nabla_m].
\end{equation}
By Lemma~\ref{lem:firsthit}, this is exactly the bulk nucleation contribution on
the interval. To express it using the full operator $\nabla$, retain the
two edges between heights $m$ and $m+1$:
\[
 [Q_s^{(m)},\nabla]=[Q_s^{(m)},\nabla_m]
 -\tfrac12Q_s^{(m)}(\delta_{m+1}\otimes\delta_m)
 -\tfrac12(\delta_m\otimes\delta_{m+1})Q_s^{(m)}.
\]
Substituting into \eqref{eq:plateau-duhamel} yields
\[
 \begin{split}
 2\int_0^s Q_z^{(m)}(\delta_{m+1}\otimes\delta_{m+1})Q_{s-z}^{(m)}\,dz
 ={}&-2[Q_s^{(m)},\nabla]-Q_s^{(m)}(\delta_{m+1}\otimes\delta_m)
 -(\delta_m\otimes\delta_{m+1})Q_s^{(m)}.
 \end{split}
\]
For the full interval, apply this formula on each constant subinterval and insert
the surrounding factors from \eqref{eq:plateau-product} and take the summation. The commutator
terms telescope:
\[
 -2\sum_{j=0}^N B_0\cdots B_{j-1}[B_j,\nabla]B_{j+1}\cdots B_N
 =-2[Q,\nabla]=2[P,\nabla],
\]
where empty products are identities and
$[e^{(b-a)\Delta},\nabla]=0$. The boundary terms are
\[
 \begin{aligned}
 \text{at }a:\quad&-(\delta_{m_0}\otimes\delta_{m_0+1})Q=\mathcal E_a,\\
 \text{at }b:\quad&-Q(\delta_{m_N+1}\otimes\delta_{m_N})=\mathcal E^b,\\
 \text{at }z_j:\quad&-Q_L(\delta_{m+1}\otimes\delta_m+
                       \delta_n\otimes\delta_{n+1})Q_R=-M_j.
 \end{aligned}
\]
Hence the bulk term in \eqref{eq:hit-generator-expanded} is
\[
 \mathsf B=2[P,\nabla]+\mathcal E_a+\mathcal E^b-\sum_{j=1}^NM_j.
\]
Lastly, for a nucleation at the wall, if $a>0$, creation at zero leaves the hitting interval unchanged, so
$\mathsf W=0$. If $a=0$, put $m=h(0)$ and
$q_b(v)=Q_{0,b}^h(m+1,v)$. Lemma~\ref{lem:firsthit} gives
\[
 \mathsf W=c(P_{0,b}^{C_0h}-P_{0,b}^h)
          =c\delta_{m+1}\otimes q_b=-c\tau_-\mathcal E_0,
\]
because $\mathcal E_0=-\delta_m\otimes q_b$ and
$\tau_-\delta_m=\delta_{m+1}$.

\noindent
Summing the contributions together, the internal terms cancel:
\[
 \mathsf M+\mathsf B
 =\sum_jM_j+2[P,\nabla]+\mathcal E_a+\mathcal E^b-\sum_jM_j
 =2[P,\nabla]+\mathcal E_a+\mathcal E^b.
\]
Adding $\mathsf W=0$ proves \eqref{eq:LP-interior}; adding
$\mathsf W=-c\tau_-\mathcal E_0$ when $a=0$ replaces
$\mathcal E_0$ by $(I-c\tau_-)\mathcal E_0$ and proves
\eqref{eq:LP-origin}.

<!-- proposition-4: body_md -->
\label{prop:moving-endpoints}
Assume $h$ is fixed and it is constant near the positive endpoints
of $0\le a<b$. Then
\begin{equation}\label{eq:endpoint-single}
 -\partial_aP_{a,b}-\Delta P_{a,b}=-\mathcal E_a
 \quad(a>0),\qquad
 \partial_bP_{a,b}-P_{a,b}\Delta=-\mathcal E^b.
\end{equation}
Consequently, along differentiable endpoints $0<a(t)<b(t)$,
\begin{equation}\label{eq:moving-hit-chain-rule}
 \frac{d}{dt}P_{a(t),b(t)}^h
 =-a'(t)\Delta P_{a,b}^h+b'(t)P_{a,b}^h\Delta
       +a'(t)\mathcal E_a-b'(t)\mathcal E^b.
\end{equation}
If $a=0$, only the terms containing $b'$ remain.
<!-- end proposition-4 -->
<!-- proof-4: body_md -->
First assume the profile has the plateau representation
\eqref{eq:plateau-product}.
Write $m_b=h(b)$. Differentiation of the last factor of
\eqref{eq:plateau-product} gives
$\partial_bQ=Q\Delta_{m_b}$. Consequently
\[
 \partial_bP-P\Delta
 =Q(\Delta-\Delta_{m_b})
 =Q(\delta_{m_b+1}\otimes\delta_{m_b})=-\mathcal E^b.
\]
In the second equality, $Q=Q1_{m_b}$ removes every row of
$\Delta-\Delta_{m_b}$ except the downward edge from $m_b+1$ to
$m_b$. Similarly, for $a>0$ and $m_a=h(a)$,
$\partial_aQ=-\Delta_{m_a}Q$, and hence
\[
 -\partial_aP-\Delta P
 =(\Delta-\Delta_{m_a})Q
 =(\delta_{m_a}\otimes\delta_{m_a+1})Q=-\mathcal E_a.
\]
The same argument applies to any constant short interval next to each
endpoint: split it off using the constrained Markov property, leaving
the rest of the profile fixed. Thus internal peaks or stacks do not
affect these endpoint identities. The ordinary chain rule gives
\eqref{eq:moving-hit-chain-rule}. The intervals in $K$ move with
physical time, whereas
\eqref{eq:LP-interior}--\eqref{eq:LP-origin} hold the endpoints fixed.

<!-- proposition-5: body_md -->
\label{prop:folded-residual}
Let $0\le c<1$ and let $p,q\ne0$ vary with $p'(t)=q'(t)=1$. Suppose \(h\in \widehat{\mathcal X}_{\mathrm{fin}}\) has no jump at \(a,b\) and \(h\) has no isolated peaks, i.e. \(h(z) = \max\{h(z^-),h(z^+)\}\). For $b>0$, define
\begin{equation}\label{eq:OD}
 \mathcal D_b:=\mathcal E^b e^{-b\Delta}H_-^{-1}
 =-\bigl(Q_{0,b}^h\delta_{h(b)+1}\bigr)\otimes
       \bigl(\delta_{h(b)}e^{-b\Delta}H_-^{-1}\bigr).
\end{equation} Then
\begin{equation}\label{eq:folded-residual}
 \mathscr D W_{p,q}=
 \begin{cases}
 -2\bigl[(I+T_p^*H)\mathcal D_q+\mathcal D_p^*(HT_q-I)\bigr],
       &p,q>0,\\[2pt]
 -2e^{-p\Delta}(\mathcal E_{-p}+\mathcal E_q)e^{-q\Delta}H_-^{-1},
       &p<0<q\\[2pt]
 2\bigl[e^{-q\Delta}(\mathcal E_{-q}+\mathcal E_p)
                           e^{-p\Delta}H_-^{-1}\bigr]^*,
       &q<0<p\\[2pt]
 0,    &p,q<0.
 \end{cases}
\end{equation}
<!-- end proposition-5 -->
<!-- proof-5: body_md -->
1. We first focus on the case \(p,q>0\). Recall that \(W_{p,q} = T_q-T_p^*+T_p^*HT_q\). For $b>0$ with $b'=1$, let
\[
 \qquad l:=(I-c\tau_-)\delta_{h(0)},\qquad
 o_b:=Q_{0,b}^he^{-b\Delta}H_-^{-1}(h(0)+1,\cdot).
\]
Thus $\mathcal E_0=-\delta_{h(0)}\otimes Q_{0,b}^h(h(0)+1,\cdot)$. The left endpoint is fixed;
differentiating $T_b=P_{0,b}e^{-b\Delta}H_-^{-1}$ and commuting
the free factors with $\nabla$ gives
\begin{align}
 \mathscr D T_b
 &=\bigl[\partial_bP_{0,b}-P_{0,b}\Delta
       -\mathcal L_cP_{0,b}+2[P_{0,b},\nabla]\bigr]
                         e^{-b\Delta}H_-^{-1}\notag\\
 &=\bigl[-\mathcal E^b-(I-c\tau_-)\mathcal E_0
                         -\mathcal E^b\bigr]e^{-b\Delta}H_-^{-1}
   =l\otimes o_b-2\mathcal D_b.\label{eq:T-origin-residual}
\end{align}
The first $-\mathcal E_b$ comes from endpoint differentiation
\eqref{eq:endpoint-single}; the other two terms come from subtracting
the generator identity~\eqref{eq:LP-origin}. Notice that the two \(\mathcal{E}_b\) terms have the same sign rather than cancelling each other.

The definition of $H$ gives
\[
 Hl=(I-c\tau_+)^{-1}\nabla\delta_{h(0)}
   =\tfrac12\sum_{k\ge0}c^k
                      (\delta_{h(0)-1-k}-\delta_{h(0)+1-k}),
 \qquad 1_{h(0)}Hl=-\tfrac12\delta_{h(0)+1}.
\]
Only $-\delta_{h(0)+1}/2$ is supported above $h(0)$. Using
$Q_{0,b}=1_{h(0)}Q_{0,b}$ and $P_{0,b}=e^{b\Delta}-Q_{0,b}$ gives
\[
 P_{0,b}^*Hl=e^{b\Delta}Hl-Q_{0,b}^*1_{h(0)}Hl
            =e^{b\Delta}Hl+\tfrac12 Q_{0,b}^h(h(0)+1,\cdot).
\]
The positive-weight inverse identity in \eqref{eq:inverse-domains}
gives $(H_-^{-1})^*H=-I$. Hence
\begin{equation}\label{eq:wall-dressed-column}
 \begin{split}
 (I+T_b^*H)l
 &=l+(H_-^{-1})^*e^{-b\Delta}P_{0,b}^*Hl\\
 &=l+(H_-^{-1})^*Hl+\tfrac12(H_-^{-1})^*e^{-b\Delta}\big(Q_{0,b}^h(h(0)+1,\cdot)\big)
   =\tfrac12o_b.
 \end{split}
\end{equation}
Since $H^*=-H$, the transposed factor satisfies
\begin{equation}\label{eq:wall-dressed-row}
 (HT_b-I)^*l=-(I+T_b^*H)l=-\tfrac12o_b.
\end{equation}
Time differentiation, deterministic step motion, and the commutator
are derivations. Nucleation obeys
$\delta_z(AB)=(\delta_zA)B+A(\delta_zB)+(\delta_zA)(\delta_zB)$,
so $-\mathcal L_c$ contributes the additional term
\[
 -2\int_0^\infty(\delta_zT_p)^*H(\delta_zT_q)\,dz
       -c(\delta_0T_p)^*H(\delta_0T_q)=0.
\]
Thus
\begin{equation}\label{eq:wall-product-rule}
 \begin{split}
 \mathscr D W_{p,q}
 &=\mathscr D T_q-(\mathscr D T_p)^*
       +(\mathscr D T_p)^*HT_q+T_p^*H\mathscr D T_q\\
 &=(I+T_p^*H)\mathscr D T_q+(\mathscr D T_p)^*(HT_q-I).

\smallskip\noindent
\emph{1. Collisions, wall absorption, and initial peaks.}
For a constant part of height $q$ whose width $\delta$ tends to zero,
\[
 Q_\delta^{(q)}\longrightarrow1_q,\qquad
 1_pQ_\delta^{(q)}1_r
 \longrightarrow1_{\max(p,q,r)}\quad  \text{ in operator norm.}
\]

product kernel (the one in \eqref{eq:plateau-product}) is exactly the restrictions of the limiting
profile. When $s\downarrow0$ and an island shrinks back to an
initial isolated peak, its height restriction remains at that

\emph{2. Contacts with hitting endpoints.}
All kernels in this step are evaluated at the profile $g=\phi_s h$. For $0\le a<b$, let $Q_{a,b^-}^g$ and
$Q_{a^+,b}^g$ denote the no-hit kernels on $[a,b)$ and

$P_{a^+,b}^g$ are defined analogously.
 
Upper semicontinuity gives $g(b)\ge g(b^-)$ and $g(a)\ge g(a^+)$.

         =1_{g(a)}Q_{a^+,b}^g,

 \begin{split}
 P_{a,b}^g-P_{a,b^-}^g
 &=\sum_{g(b^-)<v\le g(b)}
       Q_{a,b^-}^g(\cdot,v)\otimes\delta_v,\\
 P_{a,b}^g-P_{a^+,b}^g

 \end{split}

 \le 1-e^{-(2B+c)t}.
\]
Upper semicontinuity and the discreteness of the heights
imply that $f(\phi_t h)=f(h)$ for all sufficiently small $t$.
The transition probability therefore tends to $f(h)$ as
$t\downarrow0$. Taking this limit in the positive-time
identity proves that the prescribed right-time limiting
Pfaffian at $t=0$ equals $f(h)$, completing the proof.

\label{lem:tails}
For every $\lambda>0$ and $T,B<\infty$, there is a finite constant
such that each kernel
\[
 D\in\{S_{t,x},\ e^{x\Delta},\ L_+(t),\ L_-(t)\},
 \qquad |t|\le T,\quad |x|\le B,
\]
satisfies
\begin{equation}\label{eq:free-tail}
 \sup_u\sum_v|D(u,v)|e^{\lambda|u-v|}
 +\sup_v\sum_u|D(u,v)|e^{\lambda|u-v|}<\infty,
\end{equation}
uniformly over \(t,x\). The factors are defined in
\eqref{eq:free} and \eqref{eq:Lpm}.

For $0\le a\le b\le B$,
\begin{align}
 P_{a,b}^h(u,v)&\le C_{\lambda,B}
 e^{-\lambda((u-h_*)_++(v-h_*)_++|u-v|)},\label{eq:hit-tail}\\
 \sum_{w\ge v}P_{a,b}^h(u,w)&\le C_{\lambda,B}
 e^{-\lambda((u-h_*)_++(v-h_*)_++(v-u)_+)},
 \label{eq:hit-cumulative}
\end{align}
with $C_{\lambda,B}=\exp\{2B(e^{2\lambda}-1)\}$.
The bounds include $a=b$ under the closed-interval convention.

For \(a\in \mathbb{R}\), \(e^{a\tau_{\pm}}(u,u\pm k) = \frac{a^k}{k!}, k>0\). Thus,
      \[\sum_{v}|e^{a\tau_{\pm}}(u,v)|e^{\lambda|u-v|}= \exp\{|a|e^{\lambda}\}.\]
We write \(S_{t,x}, e^{x\Delta}, L_{\pm}\) in terms of \(\tau_{\pm}\):
\[
 \begin{split}
 S_{t,x}=e^{-2x}e^{(x+t)\tau_+}e^{(x-t)\tau_-},\qquad e^{x\Delta}&=e^{-2x}e^{x\tau_+}e^{x\tau_-},\qquad
 L_\pm(t)=e^{2t}e^{-2t\tau_\pm}.
 \end{split}
\]
Thus, \(\sup_u\sum_v|D(u,v)|e^{\lambda|u-v|}\) is bounded respectively by
\[
 e^{2|x|+(|x+t|+|x-t|)e^\lambda},\qquad
 e^{2|x|+2|x|e^\lambda},\qquad
 e^{2|t|+2|t|e^\lambda}.
\]
Translation invariance gives the bounds on \(\sup_v\sum_u|D(u,v)|e^{\lambda|u-v|}\).

Let $N$ be the number of walk jumps on $[a,b]$. It is Poisson with
mean $2(b-a)\le2B$. A path from $u$ to $v$ that hits a profile
bounded above by $h_*$ must satisfy
\[
 N\ge(u-h_*)_++(v-h_*)_+,\qquad N\ge|u-v|.
\]
Let \(E = (u-h_*)_++(v-h_*)_++|u-v|\). Exponential Markov's inequality gives
\[P^{h}_{a,b}(u,v)\leq e^{-\lambda  E}\mathbb{E}(e^{2\lambda N})\leq e^{-\lambda E}\exp\{2B(e^{2\lambda}-1)\}.\]
For a terminal height at least $v$, replace the second lower
bound by $N\ge(v-u)_+$, and the same arguments hold.

Recall the definition of \(G_{i,\pm}\) in \eqref{eq:G-exact}, \(L_{\pm}\) in \eqref{eq:Lpm}. We first establish the following two bounds: for every $\lambda>0$, uniformly on $|c|\le C_0$,
\begin{align}

 &&v>r_j,\label{eq:G-tail}\\
 |(L_+P_{a,b}^hL_-)(u,v)|+|(L_+P_{a,b}^hL_-H_-^{-1})(u,v)|
 &\le C_\lambda e^{-\lambda(u+v)},
 &&u>r_i,\ v>r_j,\label{eq:direct-tail}
\end{align}

Since \(L_-, L_-H^{-1}_-\) are uniformly bounded, we have \(|G(z,v)|\leq C \sum_{w\geq v}P_{0,s}^{h}(z,w)\). The exponent in
\eqref{eq:hit-cumulative} is at least $|z|+v-C'$:
for $z\ge0$, \((z-h_*)_++(v-h_*)_+>|z|+v-2h_*^+\); for $z<0$, $(v-z)_+\ge v+|z|$. 

For \eqref{eq:direct-tail}, the support and boundedness of the outer factors imply that it is bounded by $C\sum_{z\ge u,w\ge v}P_{a,b}^h(z,w)$.

\[\sum_{z\geq u,w\geq v}P_{a,b}^{h}(z,w)\leq C_\lambda e^{2\lambda h_*}(\sum_{z\geq u}e^{-\lambda z})(\sum_{w\geq v}e^{-\lambda w})\leq C_{\lambda,h_*} e^{-\lambda(u+v)}.\]

\[
 \begin{split}
      \sum_{z,w}|G_{i,\alpha}(z,u)H_c(z,w)G_{j,\beta}(w,v)|&\le C_\lambda e^{-\lambda(u+v)}\sum_{z,w}e^{-\lambda|z|}e^{\omega |z-w|}e^{-\lambda |w|}\\
 &\le C_\lambda e^{-\lambda(u+v)}
       \left(\sum_z e^{-(\lambda-\omega)|z|}\right)^2.
 \end{split}
\]
Using \eqref{eq:G-tail} and  \eqref{eq:free-tail},
\[|(S_{-t,-x_i}G_{j,\beta})(u,v)|\leq C_\lambda e^{-\lambda v}\sum_{z}|S(u,z)|e^{-\lambda|z|}\leq C_\lambda e^{-\lambda|u|-\lambda v}\le  C_\lambda e^{-\lambda(u+v)}.\]
Using \eqref{eq:G-tail} and \eqref{eq:H-free-growth},
\[ |((HS_{t,x_i})^*G_{j,\beta})(u,v)|
 \le C_\lambda e^{-\lambda v}
      \sum_z e^{\omega|z-u|-\lambda|z|}
 \le C_\lambda e^{\omega u-\lambda v}.\]
In the last inequality, $u>r_i$ permits replacing $|u|$ by $u$.

 |(J\widetilde K)_{ij}(u,v)|\le C_\lambda
 \begin{pmatrix}
 e^{-\lambda(u+v)}&e^{-\lambda u+\omega v}\\
 e^{\omega u-\lambda v}&e^{\omega u-\lambda v}+e^{-\lambda u+\omega v}
 \end{pmatrix},\qquad u>r_i,\ v>r_j.
\end{equation}

Choose $\lambda>\eta_n>\omega$ in \eqref{eq:scat-tail}. After the conjugation, \(\mathcal{M}(J\tilde{K})\mathcal{M}\) has all the exponents negative.

\end{aligned}

This is where the strict ordering of
$\eta_i$ is used.

Fix the weights $\mathcal M$ and choose $B>x_n+t$. By regularity, there is a neighborhood $\mathcal U$ in the
time and step coordinates on which function heights on each constant interval,
the ordering of steps, hitting endpoints, and the regimes
in \eqref{eq:Wpq} remain unchanged. All profiles in this
neighborhood have a common finite ceiling $H$ on $[0,B]$.

\emph{Differentiability.}
Notice that the derivatives
are sums of translates of $P$ (terms from in \eqref{eq:endpoint-single}) and no-hit products forced
through neighboring heights at most $H+1$ (terms from \eqref{eqn:deterministicshift}). The jump-count argument of Lemma~\ref{lem:tails} therefore gives
\eqref{eq:hit-tail} and \eqref{eq:hit-cumulative} also for
the absolute values of these derivatives, with an enlarged
ceiling and constants, uniformly on $\mathcal U$.
Derivatives of \(S_{\pm t,\pm x}\) are finite sums of
translates and satisfy \eqref{eq:free-tail}; derivatives
of $L_\pm$ also retain their triangular support.

Apply the product rule to the grouped expressions
\eqref{eq:FG-terms}--\eqref{eq:positive-grouped}.
Using the same argument as in the proof of
Proposition~\ref{prop:trace-class}, we have
\[
 \sum_{i,j=1}^n\sum_{\alpha,\beta\in\{-,+\}}
 \sum_{u,v\in\mathbb Z}
 \sup_{\mathcal U}
 \bigl|D A_{ij}^{\alpha\beta}(u,v)\bigr|<\infty.
\]
The entrywise mean-value formula supplies the same summable
majorant for difference quotients. Dominated convergence,
together with \eqref{eq:trace-by-entries}, proves
differentiability in trace norm.

\emph{Integrability of nucleation increments.}
Put $\delta_zA=A(t,C_zh)-A(t,h)$.
Every hitting interval lies in $[0,x_n+t]$, so
$\delta_zA=0$ for $z>x_n+t$.
For $0\le z\le x_n+t$, the profiles $C_zh$ have the common
ceiling $H+1$. The estimates of

restriction at $z$, shows that these entries are piecewise

finitely many steps and hitting endpoints. We also have the bound
\[
 \int_0^\infty\|\delta_zA\|_1\,dz
 \le (x_n+t)\sup_{0\le z\le x_n+t}\|\delta_zA\|_1
 <\infty.
\]
Thus the increment is integrable in trace class.

<!-- lemma-7: body_md -->
\label{lem:pfcalculus}
Let $A$ be a skew trace-class kernel on $\mathcal H$. Let \(\Phi(A):=\operatorname{Pf}(J+A)\). Suppose $A=A(s)$ is differentiable in trace norm and \(I-JA(s)\) is invertible. Then
\begin{equation}\label{eq:pf-derivative}
 \frac{d}{ds}\Phi(A(s))
 =
 -\frac12\Phi(A(s))
 \operatorname{tr}\!\left((I-JA(s))^{-1}J A'(s)\right).
\end{equation}
If $B$ is a skew rank-two operator and  \(I-JA(s)\) is invertible,
then
\begin{equation}\label{eq:pf-affine}
 \Phi(A+B)-\Phi(A)
 =
 -\frac12\Phi(A)\operatorname{tr}\!\left((I-JA)^{-1}JB\right).
\end{equation}
If \(I-JA\) is not invertible, it is still true that
\[A'(s) =B \Rightarrow \frac{d}{ds}\Phi(A(s))= \Phi(A+B)-\Phi(A).\]
<!-- end lemma-7 -->
<!-- proof-23: body_md -->
If $I-JA(s)$ is invertible, \eqref{eq:pf-derivative} follows from the fact that \(\operatorname{Pf}^2(J+K)=\det(I-JK)\) and 
\[
 \frac{d}{ds}\det(I-JA(s))
 =
 -\det(I-JA(s))
 \operatorname{tr}\!\left((I-JA(s))^{-1}J A'(s)\right).
\]

To prove \eqref{eq:pf-affine}, assume \(B\) has the form \( B:=u\otimes v-v\otimes u\). Define the maps $U:\mathbb C^2\to\mathcal H$ and $V:\mathcal H\to\mathbb C^2$
by
\[
 U(a,b)=au+bv,\qquad
 Vf=\begin{pmatrix}v^{\mathsf T}f\\-u^{\mathsf T}f\end{pmatrix}.
\]
Thus, $B=UV$.

Since $I-JA$ is invertible, let
\[
 R=(J+A)^{-1}=-(I-JA)^{-1}J.
\]
Since $J+A$ is skew, $R^{\mathsf T}=-R$. Consequently,
\[
 u^{\mathsf T}Ru=v^{\mathsf T}Rv=0,
 \qquad
 u^{\mathsf T}Rv=-v^{\mathsf T}Ru.
\]
Writing $\alpha=v^{\mathsf T}Ru$, the determinant-square
identity and the finite-rank determinant identity give
\[
 \begin{aligned}
 \Phi(A+B)^2
 &=\det(I-JA-JB)=\Phi(A)^2\det(I+RB)\\
 &=\Phi(A)^2\det(I+VRU)= \Phi(A)^2\det\begin{pmatrix}
  1+v^{\mathsf T}Ru & v^{\mathsf T}Rv\\
  -u^{\mathsf T}Ru & 1-u^{\mathsf T}Rv
 \end{pmatrix}\\
 &=
 \Phi(A)^2\det\begin{pmatrix}
  1+\alpha&0\\
  0&1+\alpha
 \end{pmatrix}
 =\Phi(A)^2(1+\alpha)^2.
 \end{aligned}
\]
So, \(\Phi(A+B)=\Phi(A)(1+\alpha)\). On the other hand, \(
 \operatorname{tr}(RB)
 =v^{\mathsf T}Ru-u^{\mathsf T}Rv
 =2\alpha\), so
\[
 \alpha
 =-\frac12\operatorname{tr}\!\left((I-JA)^{-1}JB\right).
\]
This proves \eqref{eq:pf-affine}.

When \(I-JA\) is not known to be invertible a prior, we use the Fr\'echet derivative; see \cite[Lemma 4.11]{MQR}.
