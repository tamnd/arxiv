<!-- proposition-1: body_md -->
\label{prop:informal_introduction}
Suppose $\bm{A}\in \mathbb{C}^{n \times n}$ is a Hermitian matrix with a leading eigenpair $(\lambda_1, \bm{v}_1)$ and at most $s$ nonzero entries per row/column.
Assume knowledge of a unit-norm vector $\bm{x}_0 \in \mathbb{C}^{n}$ with $\bm{v}_1^* \bm{x}_0 \neq 0$.
For any $\varepsilon \in (0, 1)$, define
\begin{equation*}
    C = \operatorname{nnz}(\bm{x}_0) (s+1)^q, \qquad \text{where } q = \left\lceil\frac{1}{\sqrt{8 \varepsilon}}\log\biggl(\frac{4}{|\bm{v}_1^* \bm{x}_0|^2}\biggr)\right\rceil + 1.
\end{equation*}
Then we can compute an eigenvalue estimate satisfying $(1 - \varepsilon) |\lambda_1| \leq |\hat{\lambda}_1| \leq |\lambda_1|$ by accessing at most $C$ nonzero entries and performing $\mathcal{O}(C)$ additional arithmetic operations.

<!-- theorem-1: body_md -->
\label{thm:main_thm}
Suppose $\bm{A} \in \mathbb{C}^{n \times n}$ is a Hermitian matrix with a leading eigenpair $(\lambda_1, \bm{v}_1)$ that has a second-largest-magnitude eigenvalue satisfying $|\lambda_2| < |\lambda_1|$.
Let $\bm{x}_0 \in \mathbb{C}^n$ be a unit-norm starting vector with $\bm{v}_1^* \bm{x}_0 \neq 0$ and $\operatorname{nnz}(\bm{x}_0) \leq m$, and fix a target accuracy $\varepsilon \in (0, |\bm{v}_1^*\bm{x}_0|^2)$.
Let $k$ be the smallest integer satisfying
\begin{equation*}
    \min_{\operatorname{nnz}(\bm{w}) \leq k} \lVert \bm{v}_1 - \bm{w} \rVert^2 \leq \frac{(1 - |\lambda_2 / \lambda_1|)\,\varepsilon}{10}.
\end{equation*}
Run the sparsified power method \cref{eq:deterministic} with
\begin{equation*}
    m \geq \frac{10k}{(1 - |\lambda_2 / \lambda_1|)^2\, |\bm{v}_1^* \bm{x}_0|^2} \quad \text{nonzeros},
\end{equation*}
starting from $\bm{x}_0$.
When the number of iterations is
\begin{equation*}
    q \geq \frac{5}{2(1 - |\lambda_2/\lambda_1|)} \log\left(\frac{2}{\varepsilon |\bm{v}_1^* \bm{x}_0|^2}\right),
\end{equation*}
the method returns an estimated eigenpair $(\hat{\lambda}_1, \hat{\bm{v}}_1) = (\bm{x}_q^* \bm{A} \bm{x}_q, \bm{x}_q)$ that satisfies
\begin{equation}
    \label{eq:the_guarantee}
    |\bm{v}_1^* \hat{\bm{v}}_1|^2 \geq 1 - \varepsilon/2 \quad \text{and therefore} \quad |\lambda_1 - \hat{\lambda}_1| \leq \varepsilon|\lambda_1|.
\end{equation}

<!-- proposition-2: body_md -->
\label{prop:greedy_negative_thm}
For any relative eigenvalue gap $r \in (0, 1)$ and for every $\delta$ satisfying
\begin{equation*}
    0 < \delta < \frac{2(1-r)^3}{(1+r)^2(2-r)},
\end{equation*}
there exists a symmetric tridiagonal matrix $\bm{A} \in \mathbb{R}^{4 \times 4}$ and unit-norm initialization $\bm{x}_0 \in \mathbb{R}^4$ with the following properties.
\begin{enumerate}
\item \textbf{Prescribed eigenvalue ratio.} The largest-magnitude eigenvalues are
    \begin{equation*}
    \lambda_1 = 1, \qquad \lambda_2 = 1 - r.
\end{equation*}
\item \textbf{Sparse leading eigenvector.} The leading eigenvector $\bm{v}_1$ has support $\{1, 2, 3\}$.
\item \textbf{Prescribed initial overlap.} The initial iterate has nontrivial overlap with the target eigenvector,
    \begin{equation*}
    |\bm{v}_1^* \bm{x}_0|^2 = \delta.
\end{equation*}
\item \textbf{Zero overlap of subsequent iterates.} When we apply the sparsified submatrix method \eqref{eq:recursion_1}--\eqref{eq:recursion_2}, it produces iterates that have zero overlap with the target eigenvector,
    \begin{equation*}
    \bm{v}_1^* \bm{x}_t = 0, \qquad t = 1, 2, \ldots
\end{equation*}
\end{enumerate}

<!-- theorem-2: body_md -->
\label{thm:gap_indep}
Suppose $\bm{A} \in \mathbb{C}^{n \times n}$ is a Hermitian matrix with a leading eigenpair $(\lambda_1, \bm{v}_1)$, and let $\bm{x}_0 \in \mathbb{C}^n$ be a unit-norm starting vector with $\bm{v}_1^* \bm{x}_0 \neq 0$.
Fix a target accuracy $\varepsilon \in (0, 1)$.
For any Krylov depth satisfying
    \begin{equation*}
    q \geq \frac{1}{\sqrt{8 \varepsilon}} \log\Biggl(\frac{4}{|\bm{v}_1^* \bm{x}_0|^2}\Biggr),
\end{equation*}
\cref{alg:lanczos} returns an estimate $\hat{\lambda}_1$ satisfying $(1 - \varepsilon) |\lambda_1| \leq |\hat{\lambda}_1| \leq |\lambda_1|$.
<!-- end theorem-2 -->
<!-- proof-1: body_md -->
The method is exact for $\lambda_1 = 0$, so we focus on the case $\lambda_1 \neq 0$.
Scaling $\bm{A}$ by $1/\lambda_1$ does not change the relative eigenvalue error, so we assume $\lambda_1=1$.
Take an eigendecomposition $\bm{A} = \sum_{i=1}^n \lambda_i \bm{v}_i \bm{v}_i^*$ and introduce coefficients $c_i = |\bm{v}_i^* \bm{x}_0|^2$ for $i = 1, \ldots, n$.
Let $\hat{\lambda}_1$ be the eigenvalue estimate returned at Krylov depth $q$.
Then by the variational argument in the proof of \cite[Thm.~6.4]{Saad:2011},
    \begin{equation*}
    0 \leq 1 - |\hat{\lambda}_1| \leq \frac{\sum_{i=2}^n c_i (1 - \lambda_i) p(\lambda_i)^2}{\sum_{i=1}^n c_i p(\lambda_i)^2},
\end{equation*}
where $p$ is any real polynomial of degree at most $q$ for which $p(\bm{A}) \bm{x}_0 \neq 0$.
The major steps are (i) choosing a polynomial, (ii) bounding $p(1)$ from below, and (iii) completing the argument.

\textbf{(i) Choosing a polynomial.} We set
    \begin{equation*}
    p(x) = W_q\biggl(\frac{2 x + \varepsilon}{2 - \varepsilon}\biggr),
\end{equation*}
where $W_q$ is the $q$th Chebyshev polynomial of the fourth kind.
For $x = \cos \theta$ with $\theta \in (0, \pi]$, it is defined by
    \begin{equation*}
    W_q(x) = W_q(\cos \theta) = \frac{\sin\bigl((q + 1/2) \theta\bigr)}{\sin\bigl(\theta/2\bigr)}
\end{equation*}
with its value at $\theta=0$ defined by continuity, so
    \begin{equation*}
    (1 - x) W_q(x)^2 = (1 - \cos \theta) \frac{\sin^2\bigl((q + 1/2) \theta\bigr)}{\sin^2\bigl(\theta/2\bigr)} = 2 \sin^2\bigl(\theta/2\bigr) \frac{\sin^2\bigl((q + 1/2) \theta\bigr)}{\sin^2\bigl(\theta/2\bigr)} \leq 2.
\end{equation*}
We have chosen the Chebyshev polynomial of the fourth kind because of this property and because it increases quickly for $x > 1$.

\textbf{(ii) Bounding $p(1)$ from below.} Now we make a key estimate.
For $x = (\alpha + \alpha^{-1})/2$ with $\alpha > 1$, the Chebyshev polynomial satisfies
    \begin{equation*}
    W_q(x) = W_q\left(\frac{\alpha + \alpha^{-1}}{2}\right) = \frac{\alpha^{q + 1/2} - \alpha^{-(q + 1/2)}}{\alpha^{1/2} - \alpha^{-1/2}}.
\end{equation*}
We choose the parameter
    \begin{equation*}
    \alpha = \frac{1 + \sqrt{\varepsilon/2}}{1 - \sqrt{\varepsilon/2}}
\end{equation*}
and make the calculation
\begin{align*}
    \frac{\varepsilon}{2 - \varepsilon} p(1)^2 &= \frac{\varepsilon}{2 - \varepsilon} W_q\left(\frac{2 + \varepsilon}{2 - \varepsilon}\right)^2 \\
    &= \frac{\alpha + \alpha^{-1} - 2}{4} W_q\left(\frac{\alpha + \alpha^{-1}}{2}\right)^2 \\
    &= \frac{1}{4} \bigl(\alpha^{q + 1/2} - \alpha^{-(q + 1/2)}\bigr)^2.
\end{align*}
By assumption, the depth parameter $q$ satisfies
    \begin{equation*}
    q \geq \frac{\log(4/c_1)}{\sqrt{8\varepsilon}}.
\end{equation*}
Using the inequality $2s \leq \log((1 + s)/(1-s))$, it follows
    \begin{equation*}
    \log(4/c_1) \leq 2q \sqrt{2\varepsilon} \leq 2q\log\left(\frac{1 + \sqrt{\varepsilon/2}}{1 - \sqrt{\varepsilon/2}}\right) = 2q \log \alpha.
\end{equation*}
Consequently,
    \begin{equation*}
    \alpha^{q + 1/2} \geq \frac{2}{\sqrt{c_1}} \quad \text{and} \quad \alpha^{-q-1/2} \leq \frac{\sqrt{c_1}}{2}.
\end{equation*}
We conclude that
    \begin{equation}
    \label{eq:lower_bound}
    \frac{\varepsilon}{2 - \varepsilon} p(1)^2 = \frac{1}{4} \bigl(\alpha^{q + 1/2} - \alpha^{-(q + 1/2)}\bigr)^2 \geq \frac{1}{4} \left(\frac{2}{\sqrt{c_1}} - \frac{\sqrt{c_1}}{2}\right)^2 = \frac{1}{4}\left(\frac{4}{c_1} - 2 + \frac{c_1}{4}\right) \geq \frac{1-c_1}{c_1}.
\end{equation}
We have succeeded in bounding $p(1)$ from below.

\textbf{(iii) Completing the argument.} Finally we will show that
    \begin{equation*}
    \frac{\sum_{i=2}^n c_i (1 - \lambda_i) p(\lambda_i)^2}{\sum_{i=1}^n c_i p(\lambda_i)^2} \leq \varepsilon
\end{equation*}
or equivalently
    \begin{equation*}
    \sum_{i=2}^n c_i (1 - \lambda_i) p(\lambda_i)^2 - \varepsilon \sum_{i=1}^n c_i p(\lambda_i)^2 \leq 0.
\end{equation*}    
To that end, write
    \begin{equation*}
    \sum_{i=2}^n c_i (1 - \lambda_i) p(\lambda_i)^2 - \varepsilon \sum_{i=1}^n c_i p(\lambda_i)^2 = -\varepsilon c_1 p(1)^2 + \sum_{i=2}^n c_i(1-\varepsilon - \lambda_i) p(\lambda_i)^2.
\end{equation*}
The terms in the right-most sum are nonpositive if $\lambda_i > 1 - \varepsilon$.
They are nonnegative if $\lambda_i \leq 1 - \varepsilon$, but the construction of the Chebyshev polynomial ensures that
    \begin{equation*}
    (1 - \varepsilon - \lambda_i) p(\lambda_i)^2 = \frac{2-\varepsilon}{2}(1-t_i) W_q(t_i)^2 \leq 2 - \varepsilon, \qquad \text{where } t_i = \frac{2 \lambda_i + \varepsilon}{2 - \varepsilon}.
\end{equation*}
Using the lower bound estimate \cref{eq:lower_bound}, we therefore obtain the majorant
\begin{align*}
    &-\varepsilon c_1 p(1)^2 + \sum_{i=2}^n c_i(1-\varepsilon - \lambda_i) p(\lambda_i)^2 \\
    &\quad\leq - \varepsilon c_1 p(1)^2 + (2 - \varepsilon) (1 -c_1) \\
    &\quad\leq 0.
\end{align*}
This completes the proof.

\label{prop:gapped_power}
Suppose $\bm{A} \in \mathbb{C}^{n\times n}$ is a Hermitian matrix with largest-magnitude eigenpair $(\lambda_1, \bm{v}_1)$ and second-largest-magnitude eigenvalue $\lambda_2$.
Let $\bm{x}_0 \in \mathbb{C}^n$ be a unit-norm starting vector with $\bm{v}_1^* \bm{x}_0 \neq 0$, and fix a target accuracy $\varepsilon \in (0, 1)$.
\begin{enumerate}
\item \textbf{Gapless case.} If $\bm{A}$ is positive-semidefinite, then for any number of iterations satisfying
\begin{equation*}
    q \geq \frac{1}{2 \varepsilon}\log\biggl(\frac{1}{|\bm{v}_1^* \bm{x}_0 \vert^2}\biggr),
\end{equation*}
\cref{alg:power} returns an estimate $\hat{\lambda}_1$ satisfying $|\lambda_1 - \hat{\lambda}_1| \leq \varepsilon|\lambda_1|$.
\item \textbf{Gapped case.} If $|\lambda_2| < |\lambda_1|$, then for any number of iterations satisfying
\begin{equation*}

\end{equation*}
\cref{alg:power} returns an estimate $(\hat{\lambda}_1, \hat{\bm{v}}_1)$ satisfying
\begin{equation*}
    |\bm{v}_1^* \hat{\bm{v}}_1|^2 \geq 1 - \varepsilon/2 \quad \text{and therefore} \quad |\lambda_1 - \hat{\lambda}_1| \leq \varepsilon|\lambda_1|.
\end{equation*}
\end{enumerate}

The method is exact for $\lambda_1 = 0$, so we focus on the case $\lambda_1 \neq 0$.
Scaling $\bm{A}$ by $1/\lambda_1$ does not change the relative eigenvalue error, so we assume $\lambda_1=1$.
Take an eigendecomposition $\bm{A} = \sum_{i=1}^n \lambda_i \bm{v}_i \bm{v}_i^\ast$ and introduce coefficients $c_i = \vert \bm{v}_i^\ast \bm{x}_0\vert^2$ for $i = 1,\ldots, n$.
Let $(\hat{\lambda}_1, \hat{\bm{v}}_1)$ be the eigenpair returned after $q$ iterations of the power method.

\textbf{Gapless case.} Define $M_0=1$ and
    \begin{equation*}
    M_t=\sum_{i=1}^n c_i \lambda_i^t, \qquad t>0.
\end{equation*}
By H\"older's inequality, $M_{2q}\leq M_{2q+1}^{2q/(2q+1)}$.
Consequently,
    \begin{equation*}
    \hat{\lambda}_1 =\frac{M_{2q+1}}{M_{2q}} \geq M_{2q+1}^{1/(2q+1)} \geq c_1^{1/(2q+1)}.
\end{equation*}
Using $1-\mathrm{e}^{-x}\leq x$, we obtain
    \begin{equation*}
    0\leq 1-\hat{\lambda}_1 \leq \frac{\log(1/c_1)}{2q+1}.
\end{equation*}
This proves the claimed bound.

\textbf{Gapped case.} We make the calculation
    \begin{equation*}
    1 - |\bm{v}_1^* \bm{x}_q|^2 = \frac{\sum_{i=2}^n c_i \lambda_i^{2q}}{\sum_{i=1}^n c_i \lambda_i^{2q}} \leq \frac{\lambda_2^{2q}}{c_1} \leq \frac{1}{c_1} \mathrm{e}^{-2q(1 - |\lambda_2|)}.
\end{equation*}
We can take
    \begin{equation*}
    q \geq \frac{1}{2(1 - |\lambda_2|)}\log\biggl(\frac{2}{\varepsilon c_1}\biggr)
\end{equation*}
to guarantee $|\bm{v}_1^* \bm{x}_q|^2 \geq 1 - \varepsilon/2$.
Expansion in an orthonormal eigenbasis then gives
    \begin{equation*}
    1-\bm{x}_q^*\bm{A}\bm{x}_q = \sum_{i=2}^n (1-\lambda_i) |\bm{v}_i^*\bm{x}_q|^2 \leq 2\left(1-|\bm{v}_1^*\bm{x}_q|^2\right) \le \varepsilon.
\end{equation*}
This completes the proof.

Scaling $\bm{A}$ by $1/\lambda_1$ does not change the eigenvector error or relative eigenvalue error, so we assume $\lambda_1=1$.
We prove the result in four steps.

\paragraph{Step 1. Effect of truncation.}
We will derive a lower bound on $|\bm{v}_1^*\bm{\psi}_m(\bm{y})|^2/ \lVert\bm{\psi}_m(\bm{y})\rVert^2$, valid for any unit-norm vector $\bm{y}$.

We assume for the moment that $m<n$.
Let $\mathsf{K}$ contain the $k$ largest-magnitude coordinates of $\bm{v}_1$, and let $\mathsf{S}$ contain the $m$ coordinates retained in $\bm{y}$.
The ordering of the retained coordinates gives
\begin{equation*}
    \lVert\bm{y}(\mathsf{K}\setminus\mathsf{S})\rVert^2 \le \frac{|\mathsf{K}\setminus\mathsf{S}|}{|\mathsf{S}\setminus\mathsf{K}|} \lVert\bm{y}(\mathsf{S}\setminus\mathsf{K})\rVert^2 \le \frac{k}{m} \lVert\bm{y}(\mathsf{S}\setminus\mathsf{K})\rVert^2.
\end{equation*}
For the second inequality, we used
\begin{equation*}
    \frac{|\mathsf{K}\setminus\mathsf{S}|}{|\mathsf{S}\setminus\mathsf{K}|} = \frac{k-|\mathsf{K}\cap\mathsf{S}|}{m-|\mathsf{K}\cap\mathsf{S}|} \le \frac{k}{m}.
\end{equation*}
Next, choose the phase of $\bm{v}_1$ so that $\bm{v}_1^*\bm{y}=|\bm{v}_1^*\bm{y}|$.
Then
\begin{equation*}
    \lVert\bm{y}-|\bm{v}_1^*\bm{y}|\bm{v}_1\rVert = \lVert\bm{v}_1-|\bm{v}_1^*\bm{y}|\bm{y}\rVert = \sqrt{1-|\bm{v}_1^*\bm{y}|^2}.
\end{equation*}
Since $\mathsf{S}\setminus\mathsf{K}\subseteq\mathsf{K}^c$, the triangle inequality gives
\begin{equation*}
    \lVert\bm{y}(\mathsf{S}\setminus\mathsf{K})\rVert \le |\bm{v}_1^*\bm{y}| \lVert\bm{v}_1(\mathsf{S}\setminus\mathsf{K})\rVert + \lVert\bm{y}-|\bm{v}_1^*\bm{y}|\bm{v}_1\rVert \le |\bm{v}_1^*\bm{y}| \lVert\bm{v}_1(\mathsf{K}^c)\rVert + \sqrt{1-|\bm{v}_1^*\bm{y}|^2}.
\end{equation*}
For compact notation, write
\begin{equation*}
    \Delta_k = \min_{\operatorname{nnz}(\bm{w})\le k} \lVert\bm{v}_1-\bm{w}\rVert = \lVert\bm{v}_1(\mathsf{K}^c)\rVert \quad \text{and} \quad \eta = \sqrt{\frac{k}{m}}.
\end{equation*}
Combining the preceding bounds gives
\begin{equation}
    \label{eq:simple_bound}
    \lVert\bm{y}(\mathsf{K}\setminus\mathsf{S})\rVert \le \eta \left( \sqrt{1-|\bm{v}_1^*\bm{y}|^2} + |\bm{v}_1^*\bm{y}|\Delta_k \right).
\end{equation}

Now introduce $\bm{w}=\bm{y}-\bm{\psi}_m(\bm{y})$, so $\bm{w}$ equals $\bm{y}$ on $\mathsf{S}^c$ and equals zero elsewhere.
Since $\bm{w}$ and $\bm{\psi}_m(\bm{y})$ have disjoint supports,
\begin{align*}
    1 - \frac{|\bm{v}_1^*\bm{\psi}_m(\bm{y})|^2}{\lVert\bm{\psi}_m(\bm{y})\rVert^2} &= \min_{\gamma\in\mathbb{C}} \lVert\bm{v}_1-\gamma\bm{\psi}_m(\bm{y})\rVert^2 \\
    &\le \lVert\bm{v}_1-|\bm{v}_1^*\bm{y}|\bm{\psi}_m(\bm{y})\rVert^2 \\
    &= \left\lVert \bm{v}_1 - |\bm{v}_1^*\bm{y}|\bm{y} + |\bm{v}_1^*\bm{y}|\bm{w} \right\rVert^2 \\
    &= 1-|\bm{v}_1^*\bm{y}|^2 + 2|\bm{v}_1^*\bm{y}| \operatorname{Re}\left[\bm{v}_1^*\bm{w}\right] - |\bm{v}_1^*\bm{y}|^2\lVert\bm{w}\rVert^2.
\end{align*}
We bound the last two terms separately on $\mathsf{K}^c$ and $\mathsf{K}$.
On $\mathsf{K}^c$, we use
\begin{align*}
    0 &\le \left\lVert \bm{v}_1(\mathsf{K}^c) -|\bm{v}_1^*\bm{y}|\bm{w}(\mathsf{K}^c) \right\rVert^2 \\
    &= \lVert\bm{v}_1(\mathsf{K}^c)\rVert^2 - 2|\bm{v}_1^*\bm{y}| \operatorname{Re}\left[ \bm{v}_1(\mathsf{K}^c)^* \bm{w}(\mathsf{K}^c) \right] + |\bm{v}_1^*\bm{y}|^2 \lVert\bm{w}(\mathsf{K}^c)\rVert^2.
\end{align*}
Since $\lVert\bm{v}_1(\mathsf{K}^c)\rVert=\Delta_k$, rearranging gives
\begin{equation*}
    2|\bm{v}_1^*\bm{y}| \operatorname{Re}\left[ \bm{v}_1(\mathsf{K}^c)^* \bm{w}(\mathsf{K}^c) \right] - |\bm{v}_1^*\bm{y}|^2 \lVert\bm{w}(\mathsf{K}^c)\rVert^2 \le \Delta_k^2.
\end{equation*}
On $\mathsf{K}$, the vector $\bm{w}$ equals $\bm{y}$ on $\mathsf{K}\setminus\mathsf{S}$ and vanishes elsewhere.
Thus,
\begin{align*}
    & 2|\bm{v}_1^*\bm{y}| \operatorname{Re}\left[ \bm{v}_1(\mathsf{K})^* \bm{w}(\mathsf{K}) \right] - |\bm{v}_1^*\bm{y}|^2 \lVert\bm{w}(\mathsf{K})\rVert^2 \\
    &\quad= 2|\bm{v}_1^*\bm{y}| \operatorname{Re}\left[ \bm{v}_1(\mathsf{K}\setminus\mathsf{S})^* \bm{y}(\mathsf{K}\setminus\mathsf{S}) \right] - |\bm{v}_1^*\bm{y}|^2 \lVert\bm{y}(\mathsf{K}\setminus\mathsf{S})\rVert^2 \\
    &\quad= 2|\bm{v}_1^*\bm{y}| \operatorname{Re}\left[ \left( \bm{v}_1(\mathsf{K}\setminus\mathsf{S}) - |\bm{v}_1^*\bm{y}| \bm{y}(\mathsf{K}\setminus\mathsf{S}) \right)^* \bm{y}(\mathsf{K}\setminus\mathsf{S}) \right] + |\bm{v}_1^*\bm{y}|^2 \lVert\bm{y}(\mathsf{K}\setminus\mathsf{S})\rVert^2 \\
    &\quad\le 2|\bm{v}_1^*\bm{y}| \sqrt{1-|\bm{v}_1^*\bm{y}|^2} \lVert\bm{y}(\mathsf{K}\setminus\mathsf{S})\rVert + |\bm{v}_1^*\bm{y}|^2 \lVert\bm{y}(\mathsf{K}\setminus\mathsf{S})\rVert^2.
\end{align*}
The last inequality uses Cauchy--Schwarz and $\lVert\bm{v}_1-|\bm{v}_1^*\bm{y}|\bm{y}\rVert =\sqrt{1-|\bm{v}_1^*\bm{y}|^2}$.
Applying \cref{eq:simple_bound}, we obtain
\begin{align*}
    & 2|\bm{v}_1^*\bm{y}| \sqrt{1-|\bm{v}_1^*\bm{y}|^2} \lVert\bm{y}(\mathsf{K}\setminus\mathsf{S})\rVert + |\bm{v}_1^*\bm{y}|^2 \lVert\bm{y}(\mathsf{K}\setminus\mathsf{S})\rVert^2 \\
    &\quad\le 2\eta|\bm{v}_1^*\bm{y}| \left(1-|\bm{v}_1^*\bm{y}|^2\right) + 2\eta|\bm{v}_1^*\bm{y}|^2 \sqrt{1-|\bm{v}_1^*\bm{y}|^2} \Delta_k \\
    &\qquad\quad + \eta^2|\bm{v}_1^*\bm{y}|^2 \left( \sqrt{1-|\bm{v}_1^*\bm{y}|^2} + |\bm{v}_1^*\bm{y}|\Delta_k \right)^2.
\end{align*}
To bound the cross terms, we use
\begin{equation*}
    2\eta|\bm{v}_1^*\bm{y}|^2 \sqrt{1-|\bm{v}_1^*\bm{y}|^2} \Delta_k \le 3\eta^2|\bm{v}_1^*\bm{y}|^2 \left(1-|\bm{v}_1^*\bm{y}|^2\right) + \frac{1}{3}|\bm{v}_1^*\bm{y}|^2\Delta_k^2
\end{equation*}
due to $2ab = 2(\sqrt{3} a)(b/\sqrt{3}) \le 3a^2+b^2/3$, and
\begin{equation*}
    \left( \sqrt{1-|\bm{v}_1^*\bm{y}|^2} + |\bm{v}_1^*\bm{y}|\Delta_k \right)^2 \le 2\left(1-|\bm{v}_1^*\bm{y}|^2\right) + 2|\bm{v}_1^*\bm{y}|^2\Delta_k^2.
\end{equation*}
Combining these estimates gives
\begin{align*}
    1 - \frac{|\bm{v}_1^*\bm{\psi}_m(\bm{y})|^2}{\lVert\bm{\psi}_m(\bm{y})\rVert^2} &\le \left( 1 + 2\eta|\bm{v}_1^*\bm{y}| + 5\eta^2|\bm{v}_1^*\bm{y}|^2 \right) \left(1-|\bm{v}_1^*\bm{y}|^2\right) + \left( 1 + \frac{1}{3}|\bm{v}_1^*\bm{y}|^2 + 2\eta^2|\bm{v}_1^*\bm{y}|^4 \right) \Delta_k^2 \\
    &\leq \left( 1 + 2\eta|\bm{v}_1^*\bm{y}| + 5\eta^2|\bm{v}_1^*\bm{y}|^2 \right) \left(1-|\bm{v}_1^*\bm{y}|^2\right) + \frac{5}{3} \Delta_k^2,
\end{align*}
where we used $|\bm{v}_1^*\bm{y}|\le1$ and $\eta^2\le\frac{1}{6}$, as guaranteed by the oversampling assumption.

We have established
\begin{equation}
    \label{eq:normalized_truncation}
    \frac{|\bm{v}_1^*\bm{\psi}_m(\bm{y})|^2}{\lVert\bm{\psi}_m(\bm{y})\rVert^2} \ge |\bm{v}_1^*\bm{y}|^2 - \left( 2\eta|\bm{v}_1^*\bm{y}| + 5\eta^2|\bm{v}_1^*\bm{y}|^2 \right) \left(1-|\bm{v}_1^*\bm{y}|^2\right) - \frac{5}{3}\Delta_k^2
\end{equation}
assuming $m < n$.
Additionally, \cref{eq:normalized_truncation} holds when $m\ge n$, because then there is no truncation.

\paragraph{Step 2. Progress from one iteration.}
We consider an iterate $\bm{x}_t$ satisfying
\begin{equation}
    \label{eq:spm_overlap_region}
    |\bm{v}_1^*\bm{x}_t|^2 \ge \min\left\{ |\bm{v}_1^*\bm{x}_0|^2, 1-\frac{\varepsilon}{2} \right\}
\end{equation}
and we will assess the progress from one iteration.

Let $\bm{y}_t$ be the normalized matrix-vector product, $\bm{y}_t = \bm{A} \bm{x}_t / \lVert \bm{A} \bm{x}_t \rVert$ and write
\begin{equation*}
    \lVert\bm{A}\bm{x}_t\rVert^2 = |\bm{v}_1^*\bm{x}_t|^2 + \sigma^2 \left(1-|\bm{v}_1^*\bm{x}_t|^2\right).
\end{equation*}
Here $\sigma$ is the norm amplification of the component orthogonal to $\bm{v}_1$; we take $\sigma=0$ when $|\bm{v}_1^*\bm{x}_t|=1$.
Expansion in an eigenbasis gives $0\le\sigma\le|\lambda_2|$.
Consequently, $\sigma \le \lVert\bm{A}\bm{x}_t\rVert \le1$, and
\begin{equation*}
    |\bm{v}_1^*\bm{y}_t|^2 = \frac{|\bm{v}_1^*\bm{x}_t|^2}{\lVert \bm{A} \bm{x}_t \rVert^2} \quad \text{and} \quad 1-|\bm{v}_1^*\bm{y}_t|^2 = \frac{\sigma^2 \left(1-|\bm{v}_1^*\bm{x}_t|^2\right)}{\lVert \bm{A} \bm{x}_t \rVert^2}.
\end{equation*}
Applying \cref{eq:normalized_truncation} to $\bm{y}_t$ and subtracting the current squared overlap gives
\begin{align*}
    & |\bm{v}_1^*\bm{x}_{t+1}|^2 - |\bm{v}_1^*\bm{x}_t|^2 + \frac{5}{3}\Delta_k^2 \\
    & \quad \geq |\bm{v}_1^*\bm{y}_t|^2 - |\bm{v}_1^*\bm{x}_t|^2 - \left(2 \eta |\bm{v}_1^* \bm{y}_t| + 5 \eta^2 |\bm{v}_1^* \bm{y}_t|^2\right) \left(1 - |\bm{v}_1^* \bm{y}_t|^2\right) \\
    & \quad = \frac{|\bm{v}_1^*\bm{x}_t|^2(1 - |\bm{v}_1^* \bm{x}_t|^2)(1-\sigma^2)}{\lVert \bm{A} \bm{x}_t \rVert^2} - \left(2 \eta |\bm{v}_1^* \bm{y}_t| + 5 \eta^2 |\bm{v}_1^* \bm{y}_t|^2\right) \frac{\sigma^2 \left(1-|\bm{v}_1^*\bm{x}_t|^2\right)}{\lVert \bm{A} \bm{x}_t \rVert^2} \\
    &\quad = \frac{ |\bm{v}_1^*\bm{x}_t|^2 \left(1-|\bm{v}_1^*\bm{x}_t|^2\right) }{\lVert \bm{A} \bm{x}_t \rVert^2} \left( 1-\sigma^2 - \frac{2\eta\sigma^2}{|\bm{v}_1^*\bm{x}_t| \lVert \bm{A} \bm{x}_t \rVert} - \frac{5\eta^2\sigma^2}{\lVert \bm{A} \bm{x}_t \rVert^2} \right).
\end{align*}
The assumption \cref{eq:spm_overlap_region} guarantees that $|\bm{v}_1^*\bm{x}_t|^2 \geq |\bm{v}_1^*\bm{x}_0|^2/2$, and the oversampling assumption then implies
\begin{equation*}
    \eta^2 \le \frac{ (1-|\lambda_2|)^2 |\bm{v}_1^*\bm{x}_0|^2 }{10} \le \frac{ (1-|\lambda_2|)^2 |\bm{v}_1^*\bm{x}_t|^2 }{5}.
\end{equation*}
In particular, we can derive the inequalities
\begin{equation*}
    2\eta \le (1-|\lambda_2|) |\bm{v}_1^*\bm{x}_t| \quad \text{and} \quad 5\eta^2 \le (1-|\lambda_2|)^2 |\bm{v}_1^*\bm{x}_t|^2.
\end{equation*}
Combining these estimates and using $1-\sigma \geq 1-|\lambda_2|$ gives
\begin{align*}
    &|\bm{v}_1^*\bm{x}_{t+1}|^2 - |\bm{v}_1^*\bm{x}_t|^2 + \frac{5}{3}\Delta_k^2 \\
    &\quad\geq \bigl(1-|\lambda_2|\bigr)\frac{ |\bm{v}_1^*\bm{x}_t|^2 \left(1-|\bm{v}_1^*\bm{x}_t|^2\right) }{\lVert \bm{A} \bm{x}_t \rVert^2} \\
    &\qquad\quad\times \left( 1 + \sigma - \frac{\sigma^2}{\lVert \bm{A} \bm{x}_t \rVert} - \frac{(1-\sigma) \sigma^2 |\bm{v}_1^* \bm{x}_t|^2}{\lVert \bm{A} \bm{x}_t \rVert^2} \right).
\end{align*}
To simplify this expression, first suppose $\sigma>0$.
We observe
\begin{equation*}
    \frac{(1-\sigma^2)|\bm{v}_1^*\bm{x}_t|^2}{\lVert \bm{A} \bm{x}_t \rVert^2} = 1-\frac{\sigma^2}{\lVert \bm{A} \bm{x}_t \rVert^2} \le 2\left( 1-\frac{\sigma}{\lVert \bm{A} \bm{x}_t \rVert} \right) \le \frac{1 + \sigma}{\sigma} \left(1-\frac{\sigma}{\lVert \bm{A} \bm{x}_t \rVert}\right).
\end{equation*}
The first inequality uses $1-t^2\le2(1-t)$ for $0\le t\le1$.
The second uses $2\sigma\le1+\sigma$.
Multiplying both sides by $\sigma^2/(1+\sigma)$ yields
\begin{equation*}
    \frac{(1-\sigma) \sigma^2|\bm{v}_1^*\bm{x}_t|^2}{\lVert \bm{A} \bm{x}_t \rVert^2} \le \sigma - \frac{\sigma^2}{\lVert \bm{A} \bm{x}_t \rVert}.
\end{equation*}
When $\sigma=0$, this bound holds because both sides are zero.
Thus, we arrive at
\begin{align*}
    |\bm{v}_1^*\bm{x}_{t+1}|^2 - |\bm{v}_1^*\bm{x}_t|^2 + \frac{5}{3}\Delta_k^2 &\geq \bigl(1-|\lambda_2|\bigr)\frac{ |\bm{v}_1^*\bm{x}_t|^2 \left(1-|\bm{v}_1^*\bm{x}_t|^2\right) }{\lVert \bm{A} \bm{x}_t \rVert^2} \\
    &\geq \bigl(1-|\lambda_2|\bigr) |\bm{v}_1^*\bm{x}_t|^2 \left(1-|\bm{v}_1^*\bm{x}_t|^2\right).
\end{align*}
Finally, the compressibility assumption gives
\begin{equation*}
    \frac{5}{3}\Delta_k^2 \le \frac{(1-|\lambda_2|)\varepsilon}{6}.
\end{equation*}
We have therefore proved the recurrence
\begin{equation}
    \label{eq:spm_overlap_recursion}
    |\bm{v}_1^*\bm{x}_{t+1}|^2 \ge |\bm{v}_1^*\bm{x}_t|^2 + (1-|\lambda_2|) \left( |\bm{v}_1^*\bm{x}_t|^2 \left(1-|\bm{v}_1^*\bm{x}_t|^2\right) - \frac{\varepsilon}{6} \right),
\end{equation}
valid whenever \cref{eq:spm_overlap_region} holds.

\paragraph{Step 3. Reaching and preserving the target.}
Now we put the one-step recursion into action.

Suppose first that
\begin{equation*}
    |\bm{v}_1^*\bm{x}_0|^2 \le |\bm{v}_1^*\bm{x}_t|^2 < 1-\frac{\varepsilon}{2}
\end{equation*}
and recall that $\varepsilon < |\bm{v}_1^*\bm{x}_0|^2 \leq |\bm{v}_1^*\bm{x}_t|^2$.
In the case $|\bm{v}_1^*\bm{x}_t|^2 \le 2/3$, we see that the squared overlap is larger than $\varepsilon$, and its complement is at least $1/3$, so
\begin{equation*}
    |\bm{v}_1^*\bm{x}_t|^2 \left(1-|\bm{v}_1^*\bm{x}_t|^2\right) \geq \frac{\varepsilon}{3}.
\end{equation*}
Otherwise, $|\bm{v}_1^*\bm{x}_t|^2 > 2/3$ and the complement is larger than $\varepsilon/2$, which gives the same bound.
Substituting this estimate into \cref{eq:spm_overlap_recursion} gives
\begin{equation}
    \label{eq:spm_overlap_growth}
    |\bm{v}_1^*\bm{x}_{t+1}|^2 \ge |\bm{v}_1^*\bm{x}_t|^2 + \frac{1-|\lambda_2|}{2} |\bm{v}_1^*\bm{x}_t|^2 \left(1-|\bm{v}_1^*\bm{x}_t|^2\right).
\end{equation}
Thus, until the target is reached, the squared overlap increases and remains at least its initial value.

Next suppose $|\bm{v}_1^*\bm{x}_t|^2\ge1-\varepsilon/2$.
As a function of the current squared overlap, the right-hand side of \cref{eq:spm_overlap_recursion} has derivative at least $|\lambda_2|\ge0$ on $[0,1]$.
To verify that the target is preserved, it therefore suffices to check the boundary $|\bm{v}_1^*\bm{x}_t|^2=1-\varepsilon/2$.
At this boundary, the increment in the recurrence is
\begin{equation*}
    (1-|\lambda_2|) \left( \frac{\varepsilon}{2} - \frac{\varepsilon^2}{4} - \frac{\varepsilon}{6} \right) = (1-|\lambda_2|) \frac{\varepsilon(4-3\varepsilon)}{12} > 0,
\end{equation*}
because $\varepsilon<1$.
We verify that $|\bm{v}_1^* \bm{x}_{t+1}|^2 \geq 1 - \varepsilon/2$ so the target is preserved.

\paragraph{Step 4. Iteration count.}
Now suppose the number of iterations is
\begin{equation*}
    q \geq \frac{5}{2(1 - |\lambda_2|)} \log\left(\frac{2}{\varepsilon |\bm{v}_1^* \bm{x}_0|^2}\right).
\end{equation*}
If the target is reached by iteration $q$, Step 3 shows that it is preserved.
Otherwise, the squared overlap remains below $1-\varepsilon/2$ through iteration $q$, so we may apply \cref{eq:spm_overlap_growth} at each iteration.
Taking ratios gives
\begin{equation*}
    \frac{ |\bm{v}_1^*\bm{x}_{t+1}|^2 }{ 1-|\bm{v}_1^*\bm{x}_{t+1}|^2 } \ge \frac{ 1 + \frac{1-|\lambda_2|}{2} \left(1-|\bm{v}_1^*\bm{x}_t|^2\right) }{ 1 - \frac{1-|\lambda_2|}{2} |\bm{v}_1^*\bm{x}_t|^2 } \frac{ |\bm{v}_1^*\bm{x}_t|^2 }{ 1-|\bm{v}_1^*\bm{x}_t|^2 } \ge \left( 1+\frac{1-|\lambda_2|}{2} \right) \frac{ |\bm{v}_1^*\bm{x}_t|^2 }{ 1-|\bm{v}_1^*\bm{x}_t|^2 }.
\end{equation*}
Thus the ratio of squared overlap to squared error grows by a fixed factor at each iteration.
Using the inequality $\log(1+s)\geq 2s/(2+s)$ for $s\ge0$, we find that
\begin{equation*}
    \log\left( 1+\frac{1-|\lambda_2|}{2} \right) \ge \frac{2(1-|\lambda_2|)}{5-|\lambda_2|} \ge \frac{2}{5}(1-|\lambda_2|).
\end{equation*}
Consequently,
\begin{equation*}
    \left( 1+\frac{1-|\lambda_2|}{2} \right)^q \geq \exp\left(\frac{2q}{5}(1-|\lambda_2|)\right) \geq \frac{2}{\varepsilon |\bm{v}_1^* \bm{x}_0|^2}.
\end{equation*}
After $q$ iterations, the squared error satisfies
\begin{equation*}
    \frac{ 1 }{ 1-|\bm{v}_1^*\bm{x}_q|^2 } \geq \frac{ |\bm{v}_1^*\bm{x}_q|^2 }{ 1-|\bm{v}_1^*\bm{x}_q|^2 } \ge \left( 1+\frac{1-|\lambda_2|}{2} \right)^q \frac{ |\bm{v}_1^*\bm{x}_0|^2 }{ 1-|\bm{v}_1^*\bm{x}_0|^2 } \ge \frac{2}{\varepsilon}.
\end{equation*}
This contradicts the assumption that the target has not been reached.
Therefore, the target is reached by iteration $q$ and preserved thereafter.
We confirm that
\begin{equation*}
    |\bm{v}_1^*\bm{x}_q|^2 \ge 1-\frac{\varepsilon}{2}.
\end{equation*}
Expansion in an orthonormal eigenbasis then gives
\begin{equation*}
    1-\bm{x}_q^*\bm{A}\bm{x}_q = \sum_{j=2}^n (1-\lambda_j) |\bm{v}_j^*\bm{x}_q|^2 \leq 2\left(1-|\bm{v}_1^*\bm{x}_q|^2\right) \le \varepsilon.
\end{equation*}
This completes the proof.

The proof proceeds in three steps.

\emph{Step 1.
Build the starting vector and target matrix.} Choose $q\in(r,1)$ sufficiently close to $r$ so that
\begin{equation*}
    \delta < \frac{2(1-q)^3}{(1+q)^2(2-q)} < \frac{2(1-r)^3}{(1+r)^2(2-r)}
\end{equation*}
and define a starting vector
\begin{equation*}
    \bm{x}_0 = \begin{bmatrix}
    \alpha^{1/2} \\
    0 \\
    0 \\
    (1-\alpha)^{1/2}
    \end{bmatrix}, \qquad \alpha = \frac{(1+q)^2(2-q)}{2(1-q)^3} \delta.
\end{equation*}
Thus, $0 < \alpha < 1$ and $\bm{x}_0$ is supported on coordinates 1 and 4.
Introduce a rotation
\begin{equation*}
    \bm{R} = \begin{bmatrix}
    c & -s \\
    s & c
    \end{bmatrix}, \qquad c = \sqrt{\frac{3-q}{3+3q}}, \qquad s = \sqrt{\frac{4q}{3+3q}}
\end{equation*}
and define a $2 \times 2$ matrix,
\begin{equation*}
    \bm{C} = \begin{bmatrix}
    \bm{C}(1, 1) & \bm{C}(1, 2) \\
    \bm{C}(2, 1) & \bm{C}(2, 2)
    \end{bmatrix} = \bm{R} \begin{bmatrix}
    1-q & 0 \\
    0 & (q-1)/2
    \end{bmatrix} \bm{R}^\top.
\end{equation*}
Last, construct the target matrix
\begin{equation*}
    \bm{A} = \begin{bmatrix}
    \bm{C}(1, 1) & \bm{C}(1, 2) & 0 & 0 \\
    \bm{C}(2, 1) & \bm{C}(2, 2) & d & 0 \\
    0 & d & q & 0 \\
    0 & 0 & 0 & 1-r
    \end{bmatrix}, \qquad d = \sqrt{\frac{1 - q^2}{2}}.
\end{equation*}

\emph{Step 2.
Check the spectral properties.} We carefully check the spectral properties.
By rotating the upper left $3 \times 3$ block, we expose a matrix
\begin{equation*}
    \bm{H} = \begin{bmatrix}
    c & s & 0 \\
    -s & c & 0 \\
    0 & 0 & 1
    \end{bmatrix} \begin{bmatrix}
    \bm{C}(1, 1) & \bm{C}(1, 2) & 0 \\
    \bm{C}(2, 1) & \bm{C}(2, 2) & d \\
    0 & d & q
    \end{bmatrix} \begin{bmatrix}
    c & -s & 0 \\
    s & c & 0 \\
    0 & 0 & 1
    \end{bmatrix} = \begin{bmatrix}
    1-q & 0 & ds \\
    0 & (q-1)/2 & dc \\
    ds & dc & q
    \end{bmatrix}.
\end{equation*}
A direct calculation with determinants shows that
\begin{equation*}
    \det(z \mathbf{I} - \bm{H}) = (z-1)(z-(1-q)/2)(z-(q-1)),
\end{equation*}
so the eigenvalues are $1, (1-q)/2, q-1$.
The remaining eigenvalue of $\bm{A}$ is $1-r$.
Since $r<q<1$, the two largest-magnitude eigenvalues of $\bm{A}$ are therefore $1$ and $1-r$.
Another calculation shows the leading eigenvector of $\bm{H}$ is
\begin{equation*}
    \frac{1}{\sqrt{3(1+q)(2-q)}} \begin{bmatrix}
    \sqrt{2(1-q)(3-q)} \\
    \sqrt{2q(1-q)} \\
    \sqrt{3q(3-q)}
    \end{bmatrix}.
\end{equation*}
Last, rotating back to the coordinate frame of $\bm{A}$, the first coordinate of the leading eigenvector is
\begin{equation*}
    \bm{v}_1(1) = \frac{1}{\sqrt{3(1+q)(2-q)}} \begin{bmatrix}
    c & -s
    \end{bmatrix} \begin{bmatrix}
    \sqrt{2(1-q)(3-q)} \\
    \sqrt{2q(1-q)}
    \end{bmatrix} = \frac{\sqrt{2(1-q)^3}}{\sqrt{(1+q)^2(2-q)}}.
\end{equation*}
This confirms that the initial squared overlap is $|\bm{v}_1^*\bm{x}_0|^2 = \delta$, as desired.
The second and third coordinates are also positive, while the fourth coordinate is zero.
Thus, $\operatorname{supp}(\bm{v}_1)=\{1,2,3\}$.

\emph{Step 3.
Analyze the algorithm.} Since $\bm{C}(1,2)>0$ and the starting vector is supported on coordinates 1 and 4, the first active set is $\mathsf{S}_1=\{1,2,4\}$.
The selected principal submatrix consists of $\bm{C}$, whose spectral norm is $1-q$, and the scalar block $1-r$.
Since $1-r>1-q$, the first eigenvector estimate is $\bm{x}_1=\pm\bm{e}_4$.
Column 4 has no off-diagonal nonzeros, so every subsequent active set is $\{4\}$.
Therefore, $\bm{x}_t=\pm\bm{e}_4$ for all $t\ge1$, and every subsequent iterate has zero overlap with $\bm{v}_1$.
This completes the proof.
