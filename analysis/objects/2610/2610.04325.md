<!-- definition-1: body_md -->
\label{def:multicoloring}
For a nonnegative integer $q$, a proper vertex multi-coloring associated
to $\mathbf m$ is a map
$\tau_{\mathcal G}^{\mathbf m}:I\longrightarrow\mathcal P(\{1,\ldots,q\})$
such that
\[
|\tau_{\mathcal G}^{\mathbf m}(i)|=m_i,
\qquad
\tau_{\mathcal G}^{\mathbf m}(i)\cap\tau_{\mathcal G}^{\mathbf m}(j)=\varnothing
\quad\text{if }e(i,j)\in E.
\]
Let $P_k(\mathbf m,\mathcal G)$ be the set of ordered $k$-tuples
$(P_1,\ldots,P_k)$ of nonempty independent subsets such that every
vertex $i$ belongs to exactly $m_i$ of the sets $P_j$'s.
The set $P_0(\mathbf 0,\mathcal G)$ consists of the empty tuple, and
$P_0(\mathbf m,\mathcal G)=\varnothing$ when $\mathbf m\neq\mathbf 0$.
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{thm:multicoloring-identities}
The number of proper vertex multi-colorings is the value at $q$ of
the polynomial
\begin{equation}\label{eq:chromatic-expansion}
\pi_{\mathcal G}^{\mathbf m}(q)
=\sum_{k=0}^{|\mathbf m|}|P_k(\mathbf m,\mathcal G)|\binom qk.
\end{equation}
Moreover, 
\begin{equation}\label{eq:independence-power}
I(\mathcal G,\mathbf x)^q=\sum_{\mathbf m\geq0}\pi_{\mathcal G}^{\mathbf m}(q)\mathbf x^{\mathbf m}.
\end{equation}
In particular, $\pi_{\mathcal G}^{\mathbf 0}(q)=1$.
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{cor:independent-multinomial}
Let $\mathcal I^*(\mathcal G)=\mathcal I(\mathcal G)\setminus\{\varnothing\}$, and write
$(q)_k=q(q-1)\cdots(q-k+1)$, with $(q)_0=1$. Then
\begin{equation}\label{eq:independent-multinomial}
\pi_{\mathcal G}^{\mathbf m}(q)
=\sum_{\substack{n_P\geq0\ (P\in\mathcal I^*(\mathcal G))\\
\sum\limits_{P: i\in P}n_P=m_i\ (i\in I)}}
\frac{(q)_{\sum_Pn_P}}{\prod_P n_P!}.
\end{equation}
This is a finite expression for an arbitrary graph $\mathcal G$.
<!-- end corollary-1 -->
<!-- proposition-1: body_md -->
\label{prop:blow-up}
 For every $\mathbf m\geq 0$
chromatic polynomials satisfy
\begin{equation}\label{eq:blow-up}
\pi_{\mathcal G}^{\mathbf m}(q)=\frac{\pi_{\mathcal G[\mathbf m]}(q)}{\mathbf m!}.
\end{equation}
Whitney's expansion gives the
following formula.
If $H=\mathcal G[\mathbf m]$ and $c_H(F)$ denotes the number of connected components
of the spanning graph $(V(H),F)$, including isolated vertices, then
\begin{equation}\label{eq:edge-subsets}
\pi_{\mathcal G}^{\mathbf m}(q)
=\frac1{\mathbf m!}\sum_{F\subseteq E(H)}(-1)^{|F|}q^{c_H(F)},
\end{equation}
where for every $\mathbf m$, $c_H(\varnothing)=|V(H)|=|\mathbf m|$ and
when $\mathbf m=\mathbf 0$, $H$ has no vertices, so the only term in
\eqref{eq:edge-subsets} is $q^0=1$.
<!-- end proposition-1 -->
<!-- proposition-2: body_md -->
\label{prop:components}
If the nonempty support $\operatorname{supp}(\mathbf m)$ has components
$S_1,\ldots,S_s$ in $\mathcal G(\operatorname{supp}(\mathbf m))$, then
\begin{equation}\label{eq:component-factorization}
\pi_{\mathcal G}^{\mathbf m}(q)
=\prod_{a=1}^s\pi_{\mathcal G(S_a)}^{\mathbf m|_{S_a}}(q).
\end{equation}
In particular, $q^s$ divides $\pi_{\mathcal G}^{\mathbf m}(q)$.
<!-- end proposition-2 -->
<!-- lemma-1: body_md -->
\label{lem:attachment-character}
For every $P\in\mathcal I(\mathcal G_{\rm{im}})$,
\begin{equation}\label{eq:shifted-character}
\operatorname{Ch}_{\mathfrak g_{\mathrm{re}}}(-\gamma_P)
=x_PH_\lambda^{\mathbf1_{\{v\in P\}}},
\end{equation}
where $\mathbf1_{\{v\in P\}} = 1 ,$ if $v \in P$ and zero otherwise. \qed
<!-- end lemma-1 -->
<!-- proposition-3: body_md -->
\label{cor:S}

\begin{equation}\label{eq:S-independence}
{
S=\sum_{P\in\mathcal I(\mathcal G_{\rm{im}})}(-1)^{|P|}
x_PH_\lambda^{\mathbf1_{\{v\in P\}}}

where $z_v=-x_vH_\lambda$ and $z_j=-x_j$ for $j\neq v$. In particular, $S-1\in(x_1,\ldots,x_n)\widehat R$.

<!-- corollary-2: body_md -->
\label{cor:coefficient}
For $\mu=\eta+\nu(\mathbf m)$ and an integer indeterminate $q$, the following holds.
\begin{equation}
S^q[\mathbf x^{\mathbf m}]
=(-1)^{|\mathbf m|}H_\lambda^{m_v}\pi_{\mathcal G_{\rm{im}}}^{\mathbf m}(q).
\label{eq:imaginary-coefficient}
\end{equation}
\begin{equation}

\label{eq:coefficient}
\end{equation}
\begin{equation}\label{eq:coefficient-edge-sum}

\end{equation}

<!-- proposition-4: body_md -->
\label{prop:kappa-general}
For an arbitrary finite simple graph $\mathcal G$ following expressions holds for $\kappa_{\mathcal G}(\mathbf m).$
\begin{align}
\kappa_{\mathcal G}(\mathbf m)
&=\sum_{k=1}^{|\mathbf m|}\frac{(-1)^{|\mathbf m|+k}}{k}
|P_k(\mathbf m,\mathcal G)|,\label{eq:kappa-tuples}\\
\kappa_{\mathcal G}(\mathbf m)
&=\frac{(-1)^{|\mathbf m|-1}}{\mathbf m!}
\sum_{\substack{F\subseteq E(\mathcal G[\mathbf m])\\
(V(\mathcal G[\mathbf m]),F)\text{ connected}}}(-1)^{|F|}.
\label{eq:kappa-edges}
\end{align}
If $\mathcal G(\operatorname{supp}(\mathbf m))$ is disconnected, then $\kappa_{\mathcal G}(\mathbf m)=0$.
<!-- end proposition-4 -->
<!-- theorem-2: body_md -->
\label{cor:root-general}
If $\mathbf m\neq\mathbf 0$, then
\begin{equation}\label{eq:root-general}
{
\operatorname{mult}\bigl(\eta(\mathbf k)+\nu(\mathbf m)\bigr)
=\sum_{d\mid g(\mathbf k,\mathbf m)}\frac{\mu_{\mathrm{Mob}}(d)}d
M_\lambda\left(\frac\eta d;\frac{m_v}d\right)
\kappa_{\mathcal G_{\mathrm{im}}}\left(\frac\mathbf m d\right),
}
\end{equation}

<!-- lemma-2: body_md -->
\label{lem:starts}
Let $\mathcal G$ be chordal with the chosen perfect elimination ordering,
and let $J=\operatorname{supp}(\mathbf m)\neq\varnothing$. Call $r\in J$ a start if it
has no earlier neighbor in $J$. The number of starts equals the
number of components of $\mathcal G(J)$. In particular, if $\mathcal G(J)$ is
connected and $r_0=\min J$, then
\[
a_{r_0}(\mathbf m)=m_{r_0},\qquad
a_r(\mathbf m)-m_r>0\quad(r\in J\setminus\{r_0\}).
\]
<!-- end lemma-2 -->
<!-- proof-4: body_md -->
Let $C$ be a connected component of $\mathcal G(J)$. Since $C$ is finite and non-empty, it contains a unique minimum vertex $v^*$ with respect to the perfect elimination ordering. Because $v^*$ is the minimum in $C$, it has no earlier neighbors within $C$. Furthermore, since $C$ is a connected component, $v^*$ has no neighbors in $J \setminus C$, making $v^*$ is a start vertex. This shows every connected component contains at least one start vertex.

Now assume $C$ contains another start vertex $u^* \neq v^*$. Because $C$ is connected, there exists a shortest path $P$ between $u^*$ and $v^*$ in $C$, and $P$ can not be an edge. Let $w$ be the highest-ordered vertex on this path. We first note that $w$ cannot be $v^*$ because $v^*$ is the strictly minimum vertex in $C$. Furthermore, $w$ cannot be $u^*$: if it were, $u^*$'s neighbor on the path would precede $u^*$ in the ordering, contradicting the assumption that $u^*$ is a start vertex. Thus, $w$ is an internal vertex of $P$, and its two immediate neighbors on $P$-say, $x$ and $y$-must both precede $w$ in the ordering ($x < w$ and $y < w$).
\begin{center}
\begin{tikzpicture}[
  vertex/.style={circle, fill=black, inner sep=2pt, minimum size=4pt},
  every label/.style={font=\small},
  thick
]
  \node[vertex, label=below:{$u^*$ }] (u) at (0,0) {};
  \node[vertex, label=below:{$x$}] (x) at (2,0.5) {};
  \node[vertex, label=above:{$w$ }] (w) at (3.5,1) {};
  \node[vertex, label=below:{$y$}] (y) at (5,0.5) {};
  \node[vertex, label=below:{$v^*$ }] (v) at (7,0) {};

  \draw (u) -- node[below] {$\dots$} (x);
  \draw (x) -- (w) -- (y);
  \draw (y) -- node[below] {$\dots$} (v);

  \draw[dashed, blue] (x) -- node[below] {Chord} (y);
\end{tikzpicture}
\end{center}
\noindent By the defining property of a perfect elimination ordering, the earlier neighbours of any vertex form a clique. Therefore, $x$ and $y$ must be adjacent in $\mathcal G(J)$. However, this creates a chord in $P$, contradicting the assumption that $P$ is a shortest path. Thus, every connected component of $\mathcal G(J)$ contains exactly one start vertex. Consequently, the number of starts equals the number of components of $\mathcal G(J)$.

If $\mathcal G(J)$ is connected and $r_0 = \min J$. Then, $r_0$ is the unique start vertex in $J$. Thus, every other vertex $r \in J \setminus \{r_0\}$ must have an earlier neighbour in $J$. This forces the strict inequality $a_r(\mathbf m) - m_r > 0$ for all $r \neq r_0$, while the unique start $r_0$ retains $a_{r_0}(\mathbf m) = m_{r_0}$.

<!-- corollary-5: body_md -->
\label{cor:chordal-kappa}
Suppose $\mathcal G$ is chordal with the chosen perfect elimination ordering.
If $\mathbf m\neq\mathbf 0$ and $J=\operatorname{supp}(\mathbf m)$ is connected in $\mathcal G$, let
$r_0=\min J$. Then
\begin{equation}\label{eq:chordal-kappa}
{
\kappa_{\mathcal G}(\mathbf m)
=\frac1{m_{r_0}}
\prod_{r\in J\setminus\{r_0\}}
\binom{a_r(\mathbf m)-1}{m_r}.
}
\end{equation}
For disconnected $J$, $\kappa_{\mathcal G}(\mathbf m)=0$.
<!-- end corollary-5 -->
<!-- proof-5: body_md -->
From \eqref{eq:chordal-pi}, we have expression of $\pi_{\mathbf m}^G(q)$,
\[
\pi_{\mathbf m}^G(q)
=
\binom{q-a_{r_{0}}(\mathbf m)+m_{r_{0}}}{m_{r_{0}}} \times \cdots \times \binom{q-a_{r_{k-1}}(\mathbf m) + m_{r_{k-1}}}{m_{r_{k-1}}} 
\]
\noindent Since $a_{r_{0}}(\mathbf m)=m_{r_{0}}$, the first term is $\binom{q}{m_{r_{0}}}$. For the remaining terms, $a_{r_{i}}(\mathbf m) - m_{r_{i}}>0$, which means evaluating $\binom{q-(a_{r_{i}}(\mathbf m) - m_{r_{i}})}{m_{r_{i}}}$ at $q=0$ yields a non-zero constant. We rewrite the first term as $\binom{q}{m_{r_{0}}}=\frac{q}{m_{r_{0}}}\binom{q-1}{m_{r_{0}}-1}$.
So $q$ divides $\pi_{\mathbf m}^{\mathcal G}(q)$ exactly once. Thus, we can write $\pi_{\mathbf m}^{\mathcal G}(q)=a_{1}q+a_{2}q^{2}+\cdots+ a_{n}q^{n}$. By factoring out $q$, we see that $a_{1}$ is the constant term of $(a_{1}+a_{2}q+\cdots+ a_{n}q^{n-1})$. Therefore, to find the linear coefficient $\pi_{\mathbf m}^{\mathcal G}(q)[q] = a_1$, we factor out $q$ and evaluate the remaining expression at $q=0$:
\begin{align*}
[q]\pi_{\mathbf m}^{\mathcal G}(q) 
&= \left. \frac{1}{m_{r_{0}}}\binom{q-1}{m_{r_{0}}-1} \times \cdots \times \binom{q-a_{r_{k-1}}(\mathbf m) + m_{r_{k-1}}}{m_{r_{k-1}}} \right|_{q=0}\\
&= \frac{1}{m_{r_{0}}}\binom{-1}{m_{r_{0}}-1}\times \cdots \times \binom{-(a_{r_{k-1}}(\mathbf m)-m_{r_{k-1}})}{m_{r_{k-1}}}\\
&= \frac{1}{m_{r_{0}}} (-1)^{m_{r_0}-1} \binom{m_{r_0}-1}{m_{r_0}-1} \prod_{r_{i}\not=r_{0}} (-1)^{m_{r_i}} \binom{(a_{r_{i}}(\mathbf m)-m_{r_{i}})+m_{r_{i}}-1}{m_{r_{i}}}\\
&= \frac{1}{m_{r_{0}}} (-1)^{|\mathbf m|-1}\prod_{r_{i}\not=r_{0}} \binom{a_{r_{i}}(\mathbf m)-1}{m_{r_{i}}}
\end{align*}
Here we used the identity $\binom{-x}{k}=(-1)^{k}\binom{x+k-1}{k}$, and $(m_{r_0}-1) + \sum_{r_i \neq r_0} m_{r_i} = |\mathbf m|-1$.
\\
Finally, substituting this into the definition $\kappa_{\mathcal G}(\mathbf m)=(-1)^{|\mathbf m|-1}\pi_{\mathcal G}^{\mathbf m}(q)[q]$, yields exactly equation (\ref{eq:chordal-kappa}). If $J$ is disconnected, higher power of $q$ divides $\pi_{\mathbf m}^{\mathcal G}$, making $\pi_{\mathbf m}^{\mathcal G}[q]=0$, implying $\kappa_{\mathcal G}(\mathbf m)=0$.

<!-- lemma-3: body_md -->
\label{lem:height-bound}
For every $w\in W_{I^{\mathrm{re}}}$, both $\delta_\lambda(w)$ and $\delta_0(w)$ lie
in $Q_{\mathrm{re}}^+$, and
\[
\operatorname{ht}\delta_\lambda(w)\geq\ell(w),\qquad

For a reduced expression $w=s_{i_1}\cdots s_{i_l}$, it is easy to see that,
\[
\theta-w\theta=\sum_{a=1}^l\theta(h_{i_a})
s_{i_1}\cdots s_{i_{a-1}}(\alpha_{i_a}),
\]
for $\theta=\rho'$ or $\rho'+\lambda$. Now it is a standard fact from

Again $\theta$ is dominant integral and hence $\theta(h_{i_a}) \geq 1$ for all $1 \leq a \leq l$. Therefore $\operatorname{ht} (\theta -w \theta)\geq l,$ this completes the proof.

<!-- proposition-6: body_md -->
\label{prop:finite-Weyl}
Let $\eta\in Q_{\mathrm{re}}^+$, $t\in\mathbb Z_{\geq0}$, $N=\operatorname{ht}(\eta)$, and
$\mathcal W_N=\{w\in W_{I^{\mathrm{re}}}:1\leq\ell(w)\leq N\}$.
For nonnegative integer families $(p_w),(r_w)$ indexed by
$\mathcal W_N$, put $P=\sum_wp_w$ and $R=\sum_wr_w$. Then
\begin{equation}\label{eq:finite-Weyl}
M_\lambda(\eta;t)
=\sum_{\substack{p_w,r_w\geq0\\
\sum_wp_w\delta_\lambda(w)+\sum_wr_w\delta_0(w)=\eta}}
(-1)^{\sum_w\ell(w)(p_w+r_w)}
\frac{(t)_P(-t)_R}{\prod_wp_w!\prod_wr_w!}.
\end{equation}
Here $(z)_a=z(z-1)\cdots(z-a+1)$ is a falling factorial,
including $(z)_0=1$. This sum is finite, including when $W_{I^{\mathrm{re}}}$ is infinite.
<!-- end proposition-6 -->
<!-- proof-13: body_md -->
Note that $M_{\lambda}(\eta;t)
=
H_{\lambda}^{\,t}[e^{-\eta}] = N_{\lambda}^{\,t}N_0^{-t}[e^{-\eta}].$
 Since $\delta_{\lambda}(1)=\delta_0(1)=0,$ we may write
\[
N_{\lambda}=1+V,
\qquad
N_0=1+U,
\]
where
\[
V=
\sum_{\substack{w\in W_{I^{\mathrm{re}}}\\w\neq 1}}
(-1)^{\ell(w)}e^{-\delta_{\lambda}(w)} \text{and  }
U=
\sum_{\substack{w\in W_{I^{\mathrm{re}}}\\w\neq 1}}
(-1)^{\ell(w)}e^{-\delta_0(w)}.
\]
Therefore
\[
H_{\lambda}^{\,t}
=
(1+V)^t(1+U)^{-t}.
\]
By Lemma \ref{lem:height-bound}, $\operatorname{ht}\delta_{\lambda}(w)\geq \ell(w)$ and $
\operatorname{ht}\delta_0(w)\geq \ell(w)$.
Consequently, if $\ell(w)>N$, then
\[
\operatorname{ht}\delta_{\lambda}(w)>N,
\qquad
\operatorname{ht}\delta_0(w)>N.
\]
Since all $\delta_{\lambda}(w)$ and $\delta_0(w)$ belong to $Q_{\mathrm{re}}^+$,
such terms cannot contribute to the coefficient of $e^{-\eta}$.
Hence, for computing coefficient of $e^{-\eta}$, we are restricted
to the finite set
\[
W_N
=
\{\,w\in W_{I^{\mathrm{re}}}:1\leq \ell(w)\leq N\,\}.
\]
Thus we may replace $V$ and $U$ by
\[
V_N=
\sum_{w\in W_N}
(-1)^{\ell(w)}e^{-\delta_{\lambda}(w)},
\qquad
U_N=
\sum_{w\in W_N}
(-1)^{\ell(w)}e^{-\delta_0(w)}.
\]

We now expand
\[
(1+V_N)^t
=
\sum_{P\geq 0}\binom{t}{P}V_N^P.
\]
For a family of nonnegative integers $(p_w)_{w\in W_N}$ satisfying $P=\sum_{w\in W_N}p_w,$ the multinomial theorem gives
\[
V_N^P
=
\sum_{\substack{p_w\geq 0\\\sum_w p_w=P}}
\frac{P!}{\prod_w p_w!}
(-1)^{\sum_w\ell(w)p_w}
e^{-\sum_w p_w\delta_{\lambda}(w)}.
\]
Hence we get
\[
(1+V_N)^t
=
\sum_{(p_w)}
(-1)^{\sum_w\ell(w)p_w}
\frac{(t)_P}{\prod_w p_w!}
e^{-\sum_w p_w\delta_{\lambda}(w)},
\]
since $\binom{t}{P}=\frac{(t)_P}{P!}.$
Similarly, if $(r_w)_{w\in W_N}$ is a family of nonnegative integers with $R=\sum_{w\in W_N}r_w,$ then we get
\[
(1+U_N)^{-t}
=
\sum_{(r_w)}
(-1)^{\sum_w\ell(w)r_w}
\frac{(-t)_R}{\prod_w r_w!}
e^{-\sum_w r_w\delta_0(w)}.
\]
Therefore we have,
\begin{align*}
M_{\lambda}(\eta;t) &=
H_{\lambda}^{\,t}[e^{-\eta}] \cr
&=\sum_{(p_w),(r_w)}
(-1)^{
\sum_w\ell(w)(p_w+r_w)}
\frac{(t)_P(-t)_R}
{\prod_w p_w!\prod_w r_w!}
e^{-
\left(
\sum_w p_w\delta_{\lambda}(w)
+
\sum_w r_w\delta_0(w)
\right)} [e^{-\eta}]\cr
&=\sum_{\substack{p_w,r_w\geq0\\
\sum_w p_w\delta_{\lambda}(w)
+
\sum_w r_w\delta_0(w)=\eta}}
(-1)^{
\sum_w\ell(w)(p_w+r_w)}
\frac{(t)_P(-t)_R}
{\prod_w p_w!\prod_w r_w!},
\end{align*}
where $P=\sum_w p_w,$  $R=\sum_w r_w$ and all sums are taken over $w\in W_N$.\\

\noindent It remains only to verify finiteness. If a family
$(p_w),(r_w)$ contributes, then
\[
\sum_w p_w\delta_{\lambda}(w)
+
\sum_w r_w\delta_0(w)
=
\eta.
\]
Taking heights and using Lemma \ref{lem:height-bound}, we obtain
\[
N
=
\operatorname{ht}(\eta)
\geq
\sum_w p_w+\sum_w r_w
=
P+R.
\]
Hence
\[
P+R\leq N.
\]
Since $W_N$ is finite, only finitely many families
$(p_w),(r_w)$ can occur. This completes prove.

\label{prop:A-character}
For $\eta=\sum_{i=1}^{\ell}k_i\alpha_i$ and $t\in\mathbb Z_{\geq0}$,
the normalized character and its tensor-power coefficients are

For $T=\{j_1<\cdots<j_u\}\in \mathcal{T}_u,$ let $e_T=e_{j_1}\wedge\cdots\wedge e_{j_u}.$  Then $e_T$ is a weight vector of weight $
\operatorname{wt}(e_T)
=\displaystyle\sum_{j\in T}\varepsilon_j.$ The highest-weight vector $
e_1\wedge\cdots\wedge e_u,$
has weight $\omega_u.$
Therefore
\begin{equation}
\omega_u-\operatorname{wt}(e_T)
=
\sum_{j=1}^u\varepsilon_{j}
-
\sum_{j\in T}\varepsilon_{j}.
\label{eq:weight-difference}
\end{equation}
Now we express the right-hand side of \eqref{eq:weight-difference} in terms of the simple roots.
Suppose
\[
\gamma=\sum_{r=1}^{\ell+1}a_r\varepsilon_r
      =\sum_{i=1}^{\ell}c_i\alpha_i.
\]
Then we have
\[
c_i=a_1+\cdots+a_i,
\qquad 1\leq i\leq \ell.
\]
With this observation it is easy to see that the coefficient of $\alpha_i$ in
$\omega_u-\operatorname{wt}(e_T)$ is $b_i(T)$.
Thus
\begin{equation}
\omega_u-\operatorname{wt}(e_T)
=
\sum_{i=1}^{\ell}b_i(T)\alpha_i.
\label{eq:b-coordinates}
\end{equation}
Since the weight spaces of $V_{\mathfrak g_{\mathrm{re}}}(\omega_u)$ have multiplicity one and weights are of the form $e_T$, indexed by
$T\in\mathcal{T}_u$. Hence we have
\[
\operatorname{Ch}V_{\mathfrak g_{\mathrm{re}}}(\omega_u)
=
\sum_{T\in\mathcal{T}_u}
e^{\operatorname{wt}(e_T)}.
\]
Therefore, using \eqref{eq:b-coordinates} we have
\begin{align*}
H_{\omega_u}
&=
e^{-\omega_u}\operatorname{Ch}V_{\mathfrak g_{\mathrm{re}}}(\omega_u)\\
&=\sum_{T\in\mathcal{T}_u}
e^{\operatorname{wt}(e_T)-\omega_u}
\\
&=
\sum_{T\in\mathcal{T}_u}
e^{-\sum_{i=1}^{\ell}b_i(T)\alpha_i} \cr
&=\sum_{T\in\mathcal{T}_u}
\prod_{i=1}^{\ell}y_i^{b_i(T)} =\sum_{T\in\mathcal{T}_u}
\mathbf y^{b(T)}.
\end{align*}
This proves \emph{(9.1)}.
Applying the multinomial theorem we have
\[
H_{\omega_u}^t
=
\left(
\sum_{T\in\mathcal{T}_u}
\prod_{i=1}^{\ell}y_i^{b_i(T)}
\right)^t=
\sum_{\substack{
n_T\geq 0\\
\sum_{T\in\mathcal{T}_u}n_T=t
}}
\frac{t!}{\prod_{T\in\mathcal{T}_u}n_T!}
\prod_{i=1}^{\ell}
y_i^{\sum_{T\in\mathcal{T}_u}n_Tb_i(T)}.
\]
Since $e^{-\eta}=
\prod_{i=1}^{\ell}y_i^{k_i},$ a term contributes to the coefficient of $e^{-\eta}$ precisely when
\[
\sum_{T\in\mathcal{T}_u}n_Tb_i(T)=k_i,
\qquad 1\leq i\leq \ell.
\]
Therefore
\[
M_{\omega_u}(\eta;t)
=
H_{\omega_u}^t[e^{-\eta}]
=\sum_{\substack{
n_T\geq 0\;(T\in\mathcal{T}_u)\\
\sum_T n_T=t\\
\sum_T n_Tb_i(T)=k_i,\;1\leq i\leq \ell
}}
\frac{t!}{\prod_T n_T!}.
\]
This proves \emph{(9.2)} and hence the proposition.

We specialize Proposition \ref{prop:A-character} to the endpoint $u=\ell$. Note that every $T\in\mathcal{T}_{\ell}$ is obtained by deleting exactly one
element from $\{1,\ldots,\ell+1\}$. For
$1\leq r\leq \ell+1$, write
\[
T_r
=
\{1,\ldots,\ell+1\}\setminus\{r\}.
\]
By Proposition~\ref{prop:A-character},
\[
H_{\omega_\ell}(y)
=
\sum_{T\in\mathcal{T}_{\ell}}
\prod_{i=1}^{\ell}y_i^{b_i(T)}, \text{ where } b_i(T)= \min(i,\ell)-|T\cap\{1,\ldots,i\}|.
\]

\noindent Since $1\leq i\leq\ell$, we have $\min(i,\ell)=i$, and hence $b_i(T)=i-|T\cap\{1,\ldots,i\}|.$ From this it clearly follows that $b_i(T_{l+1})=0$ for all $i$. Further, it is easy to see that for $1 \leq i,r \leq l,$
 $$b_i(T_r)= \begin{cases}
     0, \,\,\,\, \text{ if }, \,\, i <r\\
     1,   \,\,\,\, \text{ if }, \,\, i \geq r.
 \end{cases}$$
Consequently, we get
\[
H_{\omega_\ell}(y)
=
1+y_{\ell}
+y_{\ell-1}y_{\ell}
+\cdots
+y_1y_2\cdots y_{\ell}.

This proves the first assertion. Now by definition,
\[
M_{\omega_\ell}(\eta;t)
=
H_{\omega_\ell}(y)^t
\left[ \mathbf y^ \mathbf k\right].
\]
Let $c_0$ denote the number of times the term $1$ is chosen in the
multinomial expansion, and for $1\leq r\leq\ell$, let $c_r$ denote
the number of times the term $y_r y_{r+1}\cdots y_{\ell}$
is chosen. Then we have the following relations:
\[
\displaystyle\sum_{j=0}^{\ell}c_j=t, \,\,\,\,\,  \text{ and} \,\,\,\,\,  k_r=  \displaystyle\sum_{j=1}^{r}c_j, \,\,\,\, 1 \leq r \leq l.

k_1!
\prod_{i=2}^{\ell}(k_i-k_{i-1})!
(t-k_{\ell})!
}.
\]
Therefore $M_{\omega_\ell}(\eta;t)
=K_\ell(\mathbf k;t).$
