<!-- theorem-1: body_md -->
\label{thm:main}
	Under \eqref{eq:assumptions}, for every bounded Lipschitz function
	\(\phi\), every integer \(n\ge1\), and
	\[
	A_G=\frac{8\beta}{\pi\alpha^2},
	\]
	one has
	\begin{equation}\label{eq:main}
		\left|\widehat{\mathbb E}[\phi(W_n)]-\mathcal N_G[\phi]\right|
		\le \frac{\operatorname{Lip}(\phi)}{\sqrt n}
		\left(2A_Gm_3\log n+2\sqrt\beta\right).
	\end{equation}
	In particular, uniformly over bounded tests with Lipschitz constant at
	most one, the error is \(O((1+\log n)/\sqrt n)\).
<!-- end theorem-1 -->
<!-- lemma-1: body_md -->
\label{lem:smoothing}
	Let \(z\in C^{1,2}((s,+\infty)\times\mathbb{R})\cap C_b([s,+\infty)\times\mathbb{R})\) be a bounded classical solution of
	\[
	z_t=a(t,x)z_{xx},\qquad 0<\lambda\le a(t,x)\le\Lambda.
	\]
	
Then for \(t>s\),
	\begin{equation}\label{eq:lineargradient}
		\left\|z_x(t,\cdot)\right\|_\infty
		\le \frac{2\left\|z(s,\cdot)\right\|_\infty}
		{\sqrt{2\pi\lambda(t-s)}}.
	\end{equation}
	The constant does not involve derivatives of \(a\).
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Set \(K=\left\|z(s,\cdot)\right\|_\infty\). The assertion is immediate if \(K=0\).
	For \(h,d>0\), define
	\[
	\Psi(h,d)=2K\operatorname{erf}\left(\frac{d}{2\sqrt{2\lambda h}}\right),
	\qquad
	\operatorname{erf}(r)=\frac2{\sqrt\pi}\int_0^r e^{-q^2}\,dq.
	\]
	Direct differentiation gives
	\[
	\Psi_h=2\lambda\Psi_{dd},\quad \Psi_{dd}\le0,\quad
	\Psi(h,0)=0,\quad \lim_{h\downarrow0}\Psi(h,d)=2K\quad(d>0).
	\]
	We claim that for \(x<y\),
	\begin{equation}\label{eq:twopoint}
		z(s+h,y)-z(s+h,x)\le\Psi(h,y-x).
	\end{equation}
	
	        
                                                                                                                                                                                     
   
                                                   
                                              
   
                               
               
                     
                                                        
                                                                                                
                                                                                            
             
    
	
	Write
	\begin {align}
	F(h,x,y)
	=\begin{cases}
		z(s+h,y)-z(s+h,x)-\Psi(h,y-x), &h>0, x\le y,\\
		z(s,y)-z(s,x)-2K,&h=0, x<y,\\
		0,&h=0, x=y.
	\end{cases}
	\end {align}
	It is easy to check that $F$ is continuous on $D\setminus C$, and upper semicontinuous on $D$, where $D=[0, \infty)\times \{(x,y) \mid x\le y\}$, and $C=\{(0, x,x)\mid x\in \mathbb{R}\}$.
	
	Now, we check the boundary at \(h=0\) or \(x=y\). For \(h>0\), on the diagonal,
	\[
	F(h,x,x)=z(s+h,x)-z(s+h,x)-\Psi(h,0)
	=0.
	\]
	At \(h=0\), for \(x<y\), $F(0,x,y)\le 0$ because
	\[
	z(s,y)-z(s,x)\le2K,
	\] and \(F(0,x,x)\le 0\) by the definition.
	
	On the whole space $D=\{h\ge0\}\times\{x\le y\}$, \(F\) need not attain its supremum. We must first add a penalty that forces a maximum to exist.
	
	Let
	\[
	p(x)=\sqrt{1+x^2}.
	\]
	Then
	\[
	p(x)\longrightarrow\infty\quad(|x|\to\infty),
	\qquad
	0<p''(x)=\frac1{(1+x^2)^{3/2}}\le1.
	\]
	For \(\eta,\delta>0\), define
	\begin {align}
	F_{\eta,\delta}(h,x,y)
	=F(h,x,y)-\eta\bigl(p(x)+p(y)\bigr)
	-(2\Lambda\eta+\delta)h.
	\end {align}
	Fix $H>0.$ Because \(z\) is bounded on \([s,s+H]\times\mathbb R\) and \(\Psi\ge0\),
	\[
	F_{\eta,\delta}(h,x,y)\longrightarrow-\infty
	\quad\text{as }|x|+|y|\longrightarrow\infty,
	\]
	uniformly in \(0<h\le H\). Thus any positive supremum must occur in a bounded spatial region.
	
	
	
	Consequently, if \(F_{\eta,\delta}\) is positive anywhere, upper semicontinuity and spatial localization imply that it attains a positive maximum at some
	\[
	(h_0,x_0,y_0),\qquad 0<h_0\le H,\quad x_0<y_0.
	\]
	
	We shall derive a contradiction at that maximum.
	
	Write
	\[
	a_x=a(s+h_0,x_0),\qquad a_y=a(s+h_0,y_0),
	\]
	and evaluate \(\Psi_{dd}\) at \((h_0,y_0-x_0)\).
	
	The spatial second derivative tests give
	\[
	\begin{aligned}
		0&\ge(F_{\eta,\delta})_{xx}
		=-z_{xx}(s+h_0,x_0)-\Psi_{dd}-\eta p''(x_0),\\
		0&\ge(F_{\eta,\delta})_{yy}
		=z_{xx}(s+h_0,y_0)-\Psi_{dd}-\eta p''(y_0).
	\end{aligned}
	\]
	Hence
	\[
	z_{xx}(s+h_0,x_0)\ge-\Psi_{dd}-\eta p''(x_0),
	\]
	\[
	z_{xx}(s+h_0,y_0)\le\Psi_{dd}+\eta p''(y_0).
	\]
	
	Using \(z_t=a z_{xx}\), we obtain
	
	\begin{align}\label{Fh}
		\begin {split}
		(F_{\eta,\delta})_h
		&=a_yz_{xx}(s+h_0,y_0)-a_xz_{xx}(s+h_0,x_0)
		-\Psi_h-2\Lambda\eta-\delta\\
		&\le(a_x+a_y)\Psi_{dd}-\Psi_h\\
		&\quad+\eta\bigl(a_xp''(x_0)+a_yp''(y_0)\bigr)
		-2\Lambda\eta-\delta\\
		&\le-\delta<0.
		\end {split}
	\end{align}
	Here we used
	\[
	a_x+a_y\ge2\lambda,\quad \Psi_{dd}\le0,\quad
	\Psi_h=2\lambda\Psi_{dd},
	\]
	and \(a_x,a_y\le\Lambda\), \(p''\le1\).
	
	But at a maximum \(0< h_0\le H\),
	\(
	(F_{\eta,\delta})_h\ge0.
	\)
	This contradicts (\ref{Fh}). Therefore
	\[
	F_{\eta,\delta}\le0.
	\]
	
	For each fixed \(h>0\) and \(x<y\), we now have
	\[
	z(s+h,y)-z(s+h,x)
	\le\Psi(h,y-x)+\eta(p(x)+p(y))
	+(2\Lambda\eta+\delta)h.
	\]
	Letting \(\eta,\delta\downarrow0\) gives
	\[
	z(s+h,y)-z(s+h,x)\le\Psi(h,y-x).
	\]
	Applying the same argument to \(-z\),
	\[
	|z(s+h,y)-z(s+h,x)|\le\Psi(h,y-x).
	\]
	Finally, divide by \(y-x\) and let \(y\downarrow x\):
	\[
	|z_x(s+h,x)|
	\le\Psi_d(h,0)
	=\frac{2K}{\sqrt{2\pi\lambda h}}.
	\]

\label{lem:time}
	Under the hypotheses of Lemma \ref{lem:smoothing}, if \(z(s,\cdot)\)
	is \(K_1\)-Lipschitz, then
	\[
	|z(s+h,x)-z(s,x)|\le K_1\sqrt{2\Lambda h}.
	\]

<!-- proof-2: body_md -->
Fix \(x_0\). The function
	\[
	B(h,x)=z(s,x_0)+K_1\mathbb E|x-x_0+\sqrt{2\Lambda}\,W_h|
	\]
	is convex in \(x\), solves \(B_h=\Lambda B_{xx}\), and dominates
	\(z(s,x)\) initially, where $W$ is a standard Brownian motion under $\mathbb{E}$. Thus \(B_h-aB_{xx}\ge0\).
	Comparison gives \(z(s+h,x_0)\le B(h,x_0)\). Therefore,
	\[z(s+h,x_0)-z(s,x_0)\le B(h,x_0)-z(s,x_0)\le K_1\mathbb{E}[\sqrt{2\Lambda}|W_h|].\]
	On the other side,  the heat evolution $\bar{B}(h,x)$ of -\(K_1|x-x_0|+ z(s,x_0)\), is a subsolution. Comparison gives
	
	\[z(s+h,x_0)-z(s,x_0)\ge \bar{B}(h,x_0)-z(s,x_0)\ge- K_1\mathbb{E}[\sqrt{2\Lambda}|W_h|].\]

<!-- proposition-1: body_md -->
\label{prop:smooth}
	For \(t>0\), the smooth solutions satisfy
	\begin{align}
		\left\|u^\varepsilon_x(t)\right\|_\infty&\le L,\label{eq:ux}\\
		\left\|u^\varepsilon_{xx}(t)\right\|_\infty
		&\le\frac{2L}{\sqrt{\pi\alpha t}},\label{eq:qbound}\\
		\left\|u^\varepsilon_t(t)\right\|_\infty
		&\le\frac{\beta L}{\sqrt{\pi\alpha t}},\label{eq:wbound}\\
		[u^\varepsilon_t(t)]_{\mathrm{Lip}}
		&\le\frac{4\beta L}{\pi\alpha t},\label{eq:wlip}\\
		[u^\varepsilon_{xx}(t)]_{\mathrm{Lip}}
		&\le\frac{8\beta L}{\pi\alpha^2t}.\label{eq:qlip}
	\end{align}
	For \(t,s\ge\tau>0\),
	\begin{align}
		|u^\varepsilon_t(t,x)-u^\varepsilon_t(s,x)|
		&\le\frac{4\beta^{3/2}L}{\pi\alpha\tau}|t-s|^{1/2},
		\label{eq:wtime}\\
		|u^\varepsilon_{xx}(t,x)-u^\varepsilon_{xx}(s,x)|
		&\le\frac{8\beta^{3/2}L}{\pi\alpha^2\tau}|t-s|^{1/2}.
		\label{eq:qtime}
	\end{align}
<!-- end proposition-1 -->
<!-- proof-3: body_md -->
Translation invariance and comparison preserve the initial Lipschitz
	constant, proving \eqref{eq:ux}. Put
	\[
	v=u^\varepsilon_x,\qquad w=u^\varepsilon_t,\qquad
	a=G_\varepsilon'(u^\varepsilon_{xx}).
	\]
	Differentiation gives
	\[
	v_t=a v_{xx},\qquad w_t=a w_{xx},\qquad
	\alpha/2\le a\le\beta/2.
	\]
	Apply Lemma \ref{lem:smoothing} to \(v\), starting at time zero,
	to obtain \eqref{eq:qbound}. Since
	\(|G_\varepsilon(r)|\le(\beta/2)|r|\), this gives
	\eqref{eq:wbound}. Apply that lemma to \(w\) between times \(t/2\)
	and \(t\):
	\[
	\left\|w_x(t)\right\|_\infty
	\le\frac{2\left\|w(t/2)\right\|_\infty}{\sqrt{\pi\alpha t/2}}
	\le\frac{4\beta L}{\pi\alpha t}.
	\]
	The inverse of \(G_\varepsilon\) is \(2/\alpha\)-Lipschitz and
	\(u^\varepsilon_{xx}=G_\varepsilon^{-1}(w)\); hence \eqref{eq:qlip}.
	For \(t\ge s\ge\tau\), apply Lemma \ref{lem:time} to \(w\) starting
	at \(s\), using \eqref{eq:wlip} and \(\Lambda=\beta/2\).
	This proves \eqref{eq:wtime}; the inverse Lipschitz bound proves
	\eqref{eq:qtime}.

<!-- theorem-2: body_md -->
\label{thm:regularity}
	For bounded Lipschitz initial data, the solution of \eqref{eq:pde}
	is classical on \((0,\infty)\times\mathbb R\). All the estimates in
	Proposition \ref{prop:smooth} hold with \(u^\varepsilon\) replaced by
	\(u\). In particular,
	\[
	[u_{xx}(t,\cdot)]_{\mathrm{Lip}}\le A_G L/t,
	\]where \[
	A_G=\frac{8\beta}{\pi\alpha^2}.  
	\]
	Both \(u_t\) and \(u_{xx}\) are locally Lipschitz in space and locally
	\(1/2\)-H\"older in time, uniformly on positive-time strips.

<!-- proof-4: body_md -->
Write \(\eta_\varepsilon=\left\|\phi_\varepsilon-\phi\right\|_\infty\) and
	\(\delta_\varepsilon=\left\|G_\varepsilon-G\right\|_\infty\).
	Viscosity test functions show that
	\[
	u(t,x)+\eta_\varepsilon+t\delta_\varepsilon,\qquad
	u(t,x)-\eta_\varepsilon-t\delta_\varepsilon
	\]
	are respectively a supersolution and a subsolution for the
	\(G_\varepsilon\)-equation. Comparison therefore yields
	\begin{equation}\label{eq:uniformconv}
		\sup_{[0,T]\times\mathbb R}|u^\varepsilon-u|
		\le\eta_\varepsilon+T\delta_\varepsilon
		\le(L+\beta T)\varepsilon.
	\end{equation}
	Proposition \ref{prop:smooth} gives local uniform boundedness and
	equicontinuity of \(w^\varepsilon=u^\varepsilon_t\) and
	\(q^\varepsilon=u^\varepsilon_{xx}\) for positive times.
	Arzel\`a--Ascoli gives locally uniformly convergent subsequences with
	continuous limits \(w,q\). For every compactly supported smooth test
	function \(\psi\) in the positive-time region, integration by parts
	and \eqref{eq:uniformconv} imply
	\[
	\int w\psi=-\int u\psi_t,\qquad
	\int q\psi=\int u\psi_{xx}.
	\]
	These continuous distributional derivatives are classical derivatives.
	For clarity, the time statement follows by passing to the limit in
	\(u^\varepsilon(t,x)-u^\varepsilon(s,x)=
	\int_s^t w^\varepsilon(r,x)\,dr\), that is,
	\[u(t,x)-u(s,x)=
	\int_s^t w(r,x)\,dr.\] Thus, $u_t=w.$ For space, fix \(a<b\) and use
	\[
	u^\varepsilon_x(t,a)=
	\frac{u^\varepsilon(t,b)-u^\varepsilon(t,a)
		-\int_a^b(b-r)q^\varepsilon(t,r)\,dr}{b-a}.
	\]
Denote the limit of its right-hand side by $\tilde{u}_{x}(t,a)$, and passing to the limit we get
	\[u(t,b)-u(t,a)=\tilde{u}_{x}(t,a)(b-a)+\int_a^b(b-r)q(t,r)\,dr.\]
	Thus, the
	second-order integral Taylor formula on \([a,b]\) gives \(u_{xx}=q\),
	with \(u_x=\tilde{u}_{x}\) continuous in \((t,x)\). Thus \(u_t=w\), \(u_{xx}=q\),
	and \(w=G(q)\), by the uniform convergence of \(G_\varepsilon\).
	Every subsequential derivative limit is uniquely determined by \(u\),
	so the whole families converge locally uniformly. All the estimates
	pass to the limit.

<!-- remark-1: body_md -->
The following example is a translated and rescaled version of the
	periodic eigenfunction constructed by Song
	\cite[Section 2, equation (2.5) and Proposition 2.1]{Song2015}.
	That construction already gives a bounded function with a uniformly
	Lipschitz second derivative and a separated-variable solution of the
	\(G\)-heat equation. Here we use it to illustrate the failure of
	\(C^3\) regularity across the joining points.
	Assume \(\alpha<\beta\), put \(a_0=\pi\sqrt\alpha\) and
	\(p_0=\pi(\sqrt\alpha+\sqrt\beta)\), and extend periodically the function
	\[
	H(x)=
	\begin{cases}
		\sqrt\alpha\sin(x/\sqrt\alpha),&0\le x\le a_0,\\
		-\sqrt\beta\sin((x-a_0)/\sqrt\beta),&a_0\le x\le p_0.
	\end{cases}
	\]
	The values of \(H,H',H''\) match at the joins, \(H''\) is Lipschitz,
	and \(G(H'')=-H/2\). Hence \(e^{-t/2}H(x)\) solves
	\eqref{eq:pde}. However,
	\(H'''(a_0-)=1/\alpha\ne1/\beta=H'''(a_0+)\).
	The endpoint assertion is Lipschitz continuity of \(u_{xx}\), not
	existence of a continuous third derivative.

<!-- lemma-4: body_md -->
\label{lem:identity}
	For  \(F\in C_b^2(\mathbb{R})\) with Lipschitz
	second derivative,
	define
	\[
	F_s(y)=P_s^GF(\sqrt{1-s}\,y),\qquad
	\mathcal L_GF(y)=G(F''(y))-\tfrac y2F'(y).
	\]
	The nonlinear Stein identity of \cite[Lemma 4.1]{Song2020} is
	\[
	\mathcal N_G[F]-N[F]=
	\int_0^1\frac1{1-s}
	\sup_{\mu\in\Theta_{F_s}}\mu[\mathcal L_GF_s]\,ds.
	\]
	Consequently, whenever the right-hand bound is integrable,
	\begin{equation}\label{eq:steinbound}
		|\mathcal N_G[F]-N[F]|
		\le 2m_3\int_0^1\frac{[F_s'']_{\mathrm{Lip}}}{1-s}\,ds.
	\end{equation}
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
The identity is the result cited from \cite{Song2020}; it also holds
	with the supremum replaced by the infimum. Lemma \ref{lem:defect}
	applied to its maximizing measures gives \eqref{eq:steinbound}.
	In our application below the initial test is already smoothed by a
	strictly positive \(G\)-heat time, and the displayed integrals are finite.

<!-- proof-6: body_md -->
Let \(L=\operatorname{Lip}(\phi)\), \(S_0=0\), and
	\[
	S_i=n^{-1/2}\sum_{k=1}^iX_k,\qquad
	a_i=\widehat{\mathbb E}[u(1-i/n,S_i)]\quad(0\le i\le n).
	\]
	The endpoints are \(a_0=\mathcal N_G[\phi]\) and \(a_n=\widehat{\mathbb E}[\phi(W_n)]\).
	For \(1\le i\le n\), define
	\[
	b_i(x)=N[u(1-i/n,x+\cdot/\sqrt n)],\quad
	c_i(x)=\mathcal N_G[u(1-i/n,x+\cdot/\sqrt n)].
	\]
	Independence gives \(a_i=\widehat{\mathbb E}[b_i(S_{i-1})]\), while the semigroup
	property gives \(a_{i-1}=\widehat{\mathbb E}[c_i(S_{i-1})]\). Sublinearity implies
	\begin{equation}\label{eq:replace}
		|a_i-a_{i-1}|\le\sup_x|b_i(x)-c_i(x)|.
	\end{equation}
	For \(i<n\), fix \(x\) and use
	\(F(y)=u(1-i/n,x+y/\sqrt n)\).
	Positive homogeneity of \(G\), translation invariance and the semigroup
	property identify its Stein interpolation as
	\[
	F_s(y)=u\left(1-\frac in+\frac sn,\,
	x+\sqrt{\frac{1-s}{n}}\,y\right).
	\]
	Thus
	\[
	[F_s'']_{\mathrm{Lip}}=
	\left(\frac{1-s}{n}\right)^{3/2}
	[u_{xx}(1-i/n+s/n,\cdot)]_{\mathrm{Lip}} .
	\]
	Lemma \ref{lem:identity} now gives
	\begin{align*}
		|b_i(x)-c_i(x)|
		&\le\frac{2m_3}{n^{3/2}}\int_0^1
		(1-s)^{1/2}[u_{xx}(1-i/n+s/n)]_{\mathrm{Lip}}\,ds\\
		&\le\frac{2m_3}{\sqrt n}
		\int_{1-i/n}^{1-(i-1)/n}[u_{xx}(t)]_{\mathrm{Lip}}\,dt .
	\end{align*}
	For \(n\ge2\), summing these bounds, using \eqref{eq:replace} and
	Theorem \ref{thm:regularity}, yields
	\[
	\sum_{i=1}^{n-1}|a_i-a_{i-1}|
	\le\frac{2m_3}{\sqrt n}\int_{1/n}^1\frac{A_GL}{t}\,dt
	=\frac{2A_GLm_3\log n}{\sqrt n}.
	\]
	For the last step, Lipschitz continuity gives directly
	\[
	|b_n(x)-\phi(x)|\le\frac L{\sqrt n}N[|x|]
	\le\frac{L\sqrt\beta}{\sqrt n}.
	\]
	Likewise
	\[
	|c_n(x)-\phi(x)|\le
	\frac L{\sqrt n}\mathcal N_G[|x|]\le\frac{L\sqrt\beta}{\sqrt n},
	\]
	because both \(N[x^2]\) and \(\mathcal N_G[x^2]\) equal \(\beta\).
	Consequently \(|a_n-a_{n-1}|\le2L\sqrt\beta/\sqrt n\).
	The triangle inequality proves \eqref{eq:main}.
	For \(n=1\) only this last-step estimate is needed.
