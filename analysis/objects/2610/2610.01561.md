<!-- theorem-1: body_md -->
\label{thm m order HUP}
	For $N,\,m\in\ensuremath{\mathbb{N}}_+$, $m\geq2$ and $u\in\mathcal{H}^m$, we have the following inequality
	\begin{equation}
	\label{eq m order HUP}
	    \int_{\ensuremath{\mathbb{R}}^N}|P_mu|^2\mathrm{\,d} x\int_{\ensuremath{\mathbb{R}}^N}|x|^2|\nabla^{m-1}u|^2\mathrm{\,d} x\geq \mu_m(N)\left(\int_{\ensuremath{\mathbb{R}}^N}|P_{m-1}u|^2\mathrm{\,d} x\right)^2,
	\end{equation}
	where $P_m$ is defined by \eqref{def of Pm} and the constant $\mu_m(N)$ is given by \begin{align}
	\label{def of mu m N}
	    \mu_m(N)=\left(\frac{2+\sqrt{N^2+4(m-2)(N-1)}}{2}\right)^2.
	\end{align}
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{extremal}
For $N\geq1$ and $m\geq2$, the constant $\mu_m(N)$ in \eqref{eq m order HUP} is optimal. Moreover, if $N\geq2$, then  $\mu_m(N)$ is attained by some radially symmetric function $\Phi_{b,m}\in \mathcal{H}^m$ satisfying
\begin{align*}
    P_{m-2}\Phi_{b,m}(x)=\begin{cases}
    \mathrm{e}^{-\frac{br^2}{2}}{_1F_1}\left(\dfrac{N-\sqrt{N^2+4(m-2)(N-1)}}{4},\,\dfrac{N}{2},\,\dfrac{br^2}{2}\right),\,&\text{if $m$ is even,}\\
     x\mathrm{e}^{-\frac{br^2}{2}}{_1F_1}\left(\dfrac{2+N-\sqrt{N^2+4(m-2)(N-1)}}{4},\,\dfrac{N+2}{2},\,\dfrac{br^2}{2}\right),\,&\text{if $m$ is odd,}
    \end{cases}
\end{align*}
\text{where} $r=|x|$ and $b>0$. If $N=1$, then $\mu_m(N    )$ is attained by some function $\Phi_{b,m}\in\mathcal{H}^m$ such that $P_{m-2}\Phi_{b,m}=\mathrm{e}^{-br^2}$ for $b>0$.
<!-- end theorem-2 -->
<!-- remark-1: body_md -->
In the special case $m=2$, the hypergeometric function $_1F_1$ reduces to a constant. Consequently, the extremal functions $\Phi_{b,m}$ are precisely the Gaussian profile. This coincides with the result related to the second-order Heisenberg uncertainty principle in  \cite{MR4470731}.
<!-- end remark-1 -->
<!-- corollary-1: body_md -->
\label{cor m order HUP}
Let $N,\,m\in\ensuremath{\mathbb{N}}_+$ with $m\geq2$, and let $\vec{U}\in C^\infty_0\left(\ensuremath{\mathbb{R}}^N;\,\ensuremath{\mathbb{R}}^N\right)$. If $N\geq2$, we additionally require that  $\mathrm{curl }\,\vec{U}=0$, or equivalently, $U=\sum_{i=1}^N U_i \mathrm{\,d} x_i$ is a closed $1$-form. Then we have the following inequality
	\begin{equation}
	\label{equation of cor m HUP}
	    \int_{\ensuremath{\mathbb{R}}^N}|P_{m-1}\vec{U}|^2\mathrm{\,d} x\int_{\ensuremath{\mathbb{R}}^N}|x|^2|\nabla^{m-2}\vec{U}|^2\mathrm{\,d} x\geq \mu_{m}(N)\left(\int_{\ensuremath{\mathbb{R}}^N}|P_{m-2}\vec{U}|^2\mathrm{\,d} x\right)^2,
	\end{equation}
	where $\mu_m(N)$ is defined by \eqref{def of mu m N} and is optimal in \eqref{equation of cor m HUP}.
<!-- end corollary-1 -->
<!-- corollary-2: body_md -->
\label{cor Mazya}
Let $N=2$, $m\in\ensuremath{\mathbb{N}}_+$ and $m\geq2$. For any $\vec{U}\in C^\infty_0\left(\ensuremath{\mathbb{R}}^2;\,\ensuremath{\mathbb{R}}^2\right)$ satisfying $\mathrm{div}\,\vec{U}=0$, we have
	\begin{equation}
	\label{equation of cor Mazya}
	    \int_{\ensuremath{\mathbb{R}}^N}|P_{m-1}\vec{U}|^2\mathrm{\,d} x\int_{\ensuremath{\mathbb{R}}^N}|x|^2|\nabla^{m-2}\vec{U}|^2\mathrm{\,d} x\geq \mu_m(2)\left(\int_{\ensuremath{\mathbb{R}}^N}|P_{m-2}\vec{U}|^2\mathrm{\,d} x\right)^2.
	\end{equation}
	Moreover, the constant $\mu_m(2)=\left(1+\sqrt{m-1}\right)^2$ in \eqref{equation of cor Mazya} is optimal.
<!-- end corollary-2 -->
<!-- theorem-3: body_md -->
\label{thm m order HUP without tensor}
Let $N,\,m\in \ensuremath{\mathbb{N}}_+$ such that $N\geq2$ and $m\geq2$. Then we have
\begin{align}
     \int_{\ensuremath{\mathbb{R}}^N}|P_mu|^2\mathrm{\,d} x\int_{\ensuremath{\mathbb{R}}^N}|x|^2|P_{m-1}u|^2\mathrm{\,d} x\geq \mu^\star_m(N)\left(\int_{\ensuremath{\mathbb{R}}^N}|P_{m-1}u|^2\mathrm{\,d} x\right)^2\quad\text{for $u\in\mathcal{H}^m_1$},
     \label{equation m order HUP without tensor}
\end{align}
where $\mu^\star_m(N)=\dfrac{(N+2)^2}{4}$ if $m$ is even and $\mu_m^\star(N)=\dfrac{N^2}{4}$ if $m$ is odd. Moreover, the constant $\mu_m^\star(N)$ is attained by some radially symmetric function $\Phi_{b,m}^\star\in\mathcal{H}^m_1$.

<!-- corollary-3: body_md -->
\label{cor m order HUP without tensor}
Let $N,\,m\in \ensuremath{\mathbb{N}}_+$ such that $N\geq2$ and $m\geq2$. Then we have
\begin{align}
     \int_{\ensuremath{\mathbb{R}}^N}|\mathcal{R}_mu|^2\mathrm{\,d} x\int_{\ensuremath{\mathbb{R}}^N}|x|^2|\mathcal{R}_{m-1}u|^2\mathrm{\,d} x\geq \mu^\star_m(N)\left(\int_{\ensuremath{\mathbb{R}}^N}|\mathcal{R}_{m-1}u|^2\mathrm{\,d} x\right)^2\quad\text{for $u\in\mathcal{H}^m_2$},

\end{align}
where $\mathcal{R}_m$ is defined by \eqref{def of Rm} and $\mu^\star_m(N)$ is given in Theorem \ref{thm m order HUP without tensor}. Moreover, the constant $\mu_m^\star(N)$ is attained by some radially symmetric function $\Phi_{b,m}^\star\in\mathcal{H}^m_2$.

<!-- theorem-4: body_md -->
\label{thm m order HyUP}
Let $N,\,m\in\ensuremath{\mathbb{N}}_+$ and $N\geq 2$. Then we have
\begin{align}
\label{equation m order HyUP}
    \int_{\ensuremath{\mathbb{R}}^N}\left|\mathcal{R}_m u\right|^2\mathrm{\,d} x\int_{\ensuremath{\mathbb{R}}^N}\left|\mathcal{R}_{m-1}u\right|^2\mathrm{\,d} x\geq\bar{\mu}_m(N)\left(\int_{\ensuremath{\mathbb{R}}^N}\frac{\left|\mathcal{R}_{m-1}u\right|^2}{|x|}\mathrm{\,d} x\right)^2 \qquad\text{for $u\in\mathcal{H}_3^m$},
\end{align}
where $\bar{\mu}_m(N)=\dfrac{(N+1)^2}{4}$ if $m$ is even and $\bar{\mu}_m(N)=\dfrac{(N-1)^2}{4}$ if $m$ is odd. Moreover, the constant $\bar\mu_m(N)$ is attained by some radially symmetric function $\Psi_{b,m}\in\mathcal{H}^m_3$.

<!-- lemma-1: body_md -->
\label{property of F}
For $\gamma\not=0,\,-1,\,-2,\,\dots$,  the function $_1F_1(\alpha,\,\gamma,\,z)$ satisfies the following property:
\begin{enumerate}
\item Differentiation formula (see \textup{\cite[Chapter 8]{MR3586206}}): $$\dfrac{\mathrm{\,d}}{\mathrm{\,d} z}{_1F_1}(\alpha,\,\gamma,\,z)=\dfrac{\alpha}{\gamma}{_1F_1}(\alpha+1,\,\gamma+1,\,z).$$ 
\item Asymptotic property (see \textup{\cite[Chapter 9]{MR174795}}): If $\alpha$ is either a negative integer or 0, then $_1F_1$ is a polynomial in $z$ of degree not exceeding $|\alpha|$. If $\alpha$ is neither a negative integer nor 0, then $_1F_1$ has the asymptotic representation:
\begin{align*}
_1F_1(\alpha,\,\gamma,\,z)=\dfrac{\Gamma(\gamma)}{\Gamma(\alpha)}\mathrm{e}^{z}z^{\alpha-\gamma}\left[\sum_{k=0}^n\dfrac{(\gamma-\alpha)_k(1-\alpha)_k}{k!}z^{-k}+O\left(|z|^{-n-1}\right)\right],
\end{align*}
for large $|z|$ and $|\mathrm{arg}\,z|\leq \dfrac{\pi}{2}-\delta$, where $n$ is an arbitrary nonnegative integer and $\delta>0$ is arbitrary small.
\item Kummer’s transformation formula  (see \textup{\cite[Chapter 8]{MR3586206}}): \begin{equation*}
_1F_1(\alpha,\,\gamma,\,z)=\mathrm{e}^z {_1F_1}(\gamma-\alpha,\,\gamma,\,-z)
.
\end{equation*}
\end{enumerate}

<!-- lemma-2: body_md -->
\label{integral identity 1}
For $N,\,m\in\ensuremath{\mathbb{N}}_+$ and $u\in C^\infty_0\left(\ensuremath{\mathbb{R}}^N\right)$, we have
\begin{align*}
    \int_{\ensuremath{\mathbb{R}}^N}\left|\nabla^m u\right|^2\mathrm{\,d} x=\int_{\ensuremath{\mathbb{R}}^N}\left|P_m u\right|^2\mathrm{\,d} x.
\end{align*}
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
The case $m=1$ is trivial. Hence, we only consider $m\geq2$ in what follows. In fact, when $m\geq 2$, there holds:
\begin{align*}
    \int_{\ensuremath{\mathbb{R}}^N}\left|\nabla^m u\right|^2\mathrm{\,d} x&=\int_{\ensuremath{\mathbb{R}}^N}\sum_{i_1,\,i_2=1}^N\left(\sum_{i_3,\,\dots,\,i_m=1}^N\left|\frac{\partial }{\partial x_{i_1}}\frac{\partial }{\partial x_{i_2}}\frac{\partial }{\partial x_{i_3}}\cdots\frac{\partial }{\partial x_{i_m}}u\right|^2\right)\mathrm{\,d} x\\
    &=\int_{\ensuremath{\mathbb{R}}^N}\left(\sum_{i_3,\,\dots,\,i_m=1}^N\left|\Delta\left(\frac{\partial }{\partial x_{i_3}}\cdots\frac{\partial }{\partial x_{i_m}}u\right)\right|^2\right)\mathrm{\,d} x\\
    &=\int_{\ensuremath{\mathbb{R}}^N}\left(\sum_{i_3,\,\dots,\,i_m=1}^N\left|\frac{\partial }{\partial x_{i_3}}\cdots\frac{\partial }{\partial x_{i_m}}(\Delta u)\right|^2\right)\mathrm{\,d} x\\
    &=\int_{\ensuremath{\mathbb{R}}^N}\left|\nabla^{m-2}(\Delta u)\right|^2\mathrm{\,d} x.
\end{align*}
Therefore, if $m$ is even and $m=2j$,  then $P_m u=\Delta^j u$ and
\begin{align*}
  \int_{\ensuremath{\mathbb{R}}^N}\left|\nabla^m u\right|^2\mathrm{\,d} x=\int_{\ensuremath{\mathbb{R}}^N}\left|\nabla^{2j-2}\left(\Delta u\right)\right|^2\mathrm{\,d} x=\cdots=\int_{\ensuremath{\mathbb{R}}^N}\left|\nabla^2\left(\Delta^{j-1}u\right)\right|^2\mathrm{\,d} x= \int_{\ensuremath{\mathbb{R}}^N}\left|\Delta^j u\right|^2\mathrm{\,d} x.
\end{align*}
If $m$ is odd and $m=2j+1$, then $P_m u=\nabla \Delta^j u$ and
\begin{equation*}
    \int_{\ensuremath{\mathbb{R}}^N}\left|\nabla^{m} u\right|^2\mathrm{\,d} x=\int_{\ensuremath{\mathbb{R}}^N}\left|\nabla^{2j-1}\left(\Delta u\right)\right|^2\mathrm{\,d} x=\cdots=\int_{\ensuremath{\mathbb{R}}^N}\left|\nabla^{3}\left(\Delta^{j-1}u\right)\right|^2\mathrm{\,d} x=\int_{\ensuremath{\mathbb{R}}^N}\left|\nabla\left(\Delta^{j}u\right)\right|^2\mathrm{\,d} x.
\end{equation*}
<!-- end proof-1 -->
<!-- lemma-3: body_md -->
For $N,\,m\in\ensuremath{\mathbb{N}}_+$, $m\geq2$ and $u\in C^\infty_0\left(\ensuremath{\mathbb{R}}^N\right)$, we have
\begin{align*}
    \int_{\ensuremath{\mathbb{R}}^N}|x|^2\left|\nabla^m u\right|^2\mathrm{\,d} x=\int_{\ensuremath{\mathbb{R}}^N}|x|^2\left|\nabla^{m-2}\left(\Delta u\right)\right|^2\mathrm{\,d} x+2(N-1)\int_{\ensuremath{\mathbb{R}}^N}\left|P_{m-1}u\right|^2\mathrm{\,d} x.
\end{align*}
\label{integral identity 2}
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
For the case $m=2$, there holds
\begin{align}
\label{Bochner 1}
    \int_{\ensuremath{\mathbb{R}}^N}|x|^2\left|\nabla^2 u\right|^2\mathrm{\,d} x&=\sum_{k,\,l=1}^N\int_{\ensuremath{\mathbb{R}}^N}|x|^2\left(\dfrac{\partial^2 u}{\partial x_l\partial x_k}\right)^2\mathrm{\,d} x\nonumber\\
    &=-\sum_{k,\,l=1}^N\int_{\ensuremath{\mathbb{R}}^N}\dfrac{\partial}{\partial x_l}\left(|x|^2\dfrac{\partial^2 u}{\partial x_l\partial x_k}\right)\dfrac{\partial u }{\partial x_k}\mathrm{\,d} x\nonumber\\
    &=-\sum_{k,\,l=1}^N\int_{\ensuremath{\mathbb{R}}^N}\left(2x_l\dfrac{\partial^2 u}{\partial x_l\partial x_k}+|x|^2\dfrac{\partial^3 u}{\partial x_l\partial x_l\partial x_k}\right)\dfrac{\partial u }{\partial x_k}\mathrm{\,d} x.
\end{align}
Notice that
\begin{align}
    \label{Bochner 2}
    -\sum_{k,\,l=1}^N\int_{\ensuremath{\mathbb{R}}^N}2x_l\dfrac{\partial u}{\partial x_k}\dfrac{\partial^2u}{\partial x_l\partial x_k}\mathrm{\,d} x&=-\sum_{k,\,l=1}^N\int_{\ensuremath{\mathbb{R}}^N}x_l\dfrac{\partial}{\partial x_l}\left(\left(\dfrac{\partial u}{\partial x_k}\right)^2\right)\mathrm{\,d} x\nonumber\\
    &=\sum_{k,\,l=1}^N\int_{\ensuremath{\mathbb{R}}^N}\left(\dfrac{\partial u}{\partial x_k}\right)^2\mathrm{\,d} x\nonumber\\
    &=N\int_{\ensuremath{\mathbb{R}}^N}\left|\nabla u\right|^2\mathrm{\,d} x.\\
    -\sum_{k,\,l=1}^N\int_{\ensuremath{\mathbb{R}}^N}|x|^2\dfrac{\partial^3 u}{\partial x_l\partial x_l\partial x_k}\dfrac{\partial u }{\partial x_k}\mathrm{\,d} x&=-\sum_{k=1}^N\int_{\ensuremath{\mathbb{R}}^N}|x|^2\dfrac{\partial(\Delta u)}{\partial x_k}\dfrac{\partial u}{\partial x_k}\mathrm{\,d} x\nonumber\\
    &=2\int_{\ensuremath{\mathbb{R}}^N}(x\cdot\nabla u)\Delta u\mathrm{\,d} x+\int_{\ensuremath{\mathbb{R}}^N}|x|^2|\Delta u|^2\mathrm{\,d} x.\label{Bochner 3}
\end{align}
\\
Moreover, we have
\begin{align}
    \label{Bochner 4}
    2\int_{\ensuremath{\mathbb{R}}^N}(x\cdot\nabla u)\Delta u\mathrm{\,d} x&=-2\sum_{k,\,l=1}^N\int_{\ensuremath{\mathbb{R}}^N}\dfrac{\partial u}{\partial x_l}\dfrac{\partial }{\partial x_l}\left(x_k\dfrac{\partial u}{\partial x_k}\right)\mathrm{\,d} x\nonumber\\
    &=-2\int_{\ensuremath{\mathbb{R}}^N}|\nabla u|^2\mathrm{\,d} x-\sum_{k,\,l=1}^N\int_{\ensuremath{\mathbb{R}}^N}x_k\dfrac{\partial}{\partial x_k}\left(\left(\dfrac{\partial u}{\partial x_l}\right)^2\right)\mathrm{\,d} x\nonumber\\
    &=(N-2)\int_{\ensuremath{\mathbb{R}}^N}|\nabla u|^2\mathrm{\,d} x.
\end{align}
\\
Substituting \eqref{Bochner 2}, \eqref{Bochner 3}, \eqref{Bochner 4} into \eqref{Bochner 1}, we obtain the following result:
\begin{equation*}
    \int_{\ensuremath{\mathbb{R}}^N}\lvert x\rvert^2\lvert \nabla^2 u\rvert^2\mathrm{\,d} x=\int_{\ensuremath{\mathbb{R}}^N}\lvert x\rvert^2\lvert \Delta u\rvert^2\mathrm{\,d} x+2(N-1)\int_{\ensuremath{\mathbb{R}}^N}\lvert \nabla u\rvert^2\mathrm{\,d} x.
\end{equation*}
\\
Using the result for $m=2$, we readily obtain the result for $m\geq3$:
\begin{align*}
    \int_{\ensuremath{\mathbb{R}}^N}|x|^2\left|\nabla^m u\right|^2\mathrm{\,d} x&=\sum_{i_1,\,\dots,\,i_{m-2}=1}^N\int_{\ensuremath{\mathbb{R}}^N}|x|^2\left|\nabla^2\left(\dfrac{\partial^{m-2}u}{\partial x_{i_1}\cdots\partial x_{i_{m-2}}}\right)\right|^2\mathrm{\,d} x\\
    &=\sum_{i_1,\,\dots,\,i_{m-2}=1}^N\left[\int_{\ensuremath{\mathbb{R}}^N}\lvert x\rvert^2\left\lvert \Delta \left(\dfrac{\partial^{m-2}u}{\partial x_{i_1}\cdots\partial x_{i_{m-2}}}\right)\right\rvert^2\mathrm{\,d} x\right.\\
    &\hspace{4cm} \left.+2(N-1)\int_{\ensuremath{\mathbb{R}}^N}\left\lvert \nabla \left(\dfrac{\partial^{m-2}u}{\partial x_{i_1}\cdots\partial x_{i_{m-2}}}\right)\right\rvert^2\mathrm{\,d} x\right]\\
    &=\int_{\ensuremath{\mathbb{R}}^N}|x|^2\left|\nabla^{m-2}(\Delta u)\right|^2\mathrm{\,d} x+2(N-1)\int_{\ensuremath{\mathbb{R}}^N}\left|\nabla^{m-1}u\right|^2\mathrm{\,d} x\\
    &=\int_{\ensuremath{\mathbb{R}}^N}|x|^2\left|\nabla^{m-2}(\Delta u)\right|^2\mathrm{\,d} x+2(N-1)\int_{\ensuremath{\mathbb{R}}^N}\left|P_{m-1}u\right|^2\mathrm{\,d} x,
\end{align*}
where the last equality above follows from Lemma \ref{integral identity 1}.
<!-- end proof-2 -->
<!-- lemma-4: body_md -->
\label{integral identity 3}
For $N,\,m\in\ensuremath{\mathbb{N}}_+$ and $u\in C^\infty_0\left(\ensuremath{\mathbb{R}}^N\right)$, we have
 \begin{align*}
     \int_{\ensuremath{\mathbb{R}}^N}|x|^2\left|\nabla^{m} u\right|^2\mathrm{\,d} x=\int_{\ensuremath{\mathbb{R}}^N}|x|^2\left|P_{m}u\right|^2\mathrm{\,d} x+2(N-1)\left\lfloor \frac{m}{2}\right\rfloor\int_{\ensuremath{\mathbb{R}}^N}\left|P_{m-1}u\right|^2\mathrm{\,d} x,
 \end{align*}
 where $\lfloor m/2\rfloor$ denotes the greatest integer less than or equal to $m/2$.
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
The case $m=1$ is trivial. Therefore, we only need to consider $m\geq2$. Let $j=\lfloor m/2\rfloor$. If $m=2j$ for $j\geq 1$, by Lemma \ref{integral identity 2}, there holds
\begin{align*}
    \int_{\ensuremath{\mathbb{R}}^N}|x|^2\left|\nabla^m u\right|^2\mathrm{\,d} x&=\int_{\ensuremath{\mathbb{R}}^N}|x|^2\left|\nabla^{2j-2}(\Delta u)\right|^2\mathrm{\,d} x+2(N-1)\int_{\ensuremath{\mathbb{R}}^N}\left|P_{m-1}u\right|^2\mathrm{\,d} x\\
    &=\int_{\ensuremath{\mathbb{R}}^N}|x|^2\left|\nabla^{2j-4}\left(\Delta^2 u\right)\right|^2\mathrm{\,d} x+4(N-1)\int_{\ensuremath{\mathbb{R}}^N}\left|P_{m-1}u\right|^2\mathrm{\,d} x\\

\label{expression of v}
For $N\geq 2$, $m\geq2$ and $b>0$, we denote $r=|x|$ and define
\begin{align*}
    v_e(x,\,b)=v_e(r,\,b)=&\mathrm{e}^{-\frac{br^2}{2}}{_1F_1}\left(\dfrac{N-c_e}{2},\,\dfrac{N}{2},\,\dfrac{br^2}{2}\right),\\
    v_o(x,\,b)=v_o(r,\,b)=&-\int_{r}^{+\infty}s\mathrm{e}^{-\frac{bs^2}{2}}{_1F_1}\left(\dfrac{2-c_o}{2},\,\dfrac{N+2}{2},\,\dfrac{bs^2}{2}\right)\mathrm{\,d} s,
\end{align*}
where $c_e$ is defined by \eqref{def of ce}  and $c_o$ is defined by \eqref{def of co}.  Then $v_e$ and $v_o$ are smooth with respect to $x$ such that $\mathcal{E}_{b}\left(v_e\right)=0$ and $\mathcal{O}_b\left(v_o\right)=0$ hold in $\ensuremath{\mathbb{R}}^N$.

We fix $b>0$. By abuse of notation, we write $v_e(x)=v_e(x,\,b)$ and $v_o(x)=v_o(x,\,b)$ whenever no confusion arises. From Lemma \ref{property of F}, we have
\begin{align*}
    s\mathrm{e}^{-\frac{bs^2}{2}}{_1F_1}\left(\frac{2-c_o}{2},\,\frac{N+2}{2},\,\frac{bs^2}{2}\right)=O\left(s^{1-N-c_o}\right)\text{ as }s\to+\infty,
\end{align*}
where $1-N-c_o<-1$. Thus, $v_o(x)$ is well-defined. Next, we rigorously verify that both $v_e$ and $v_o$ are smooth functions on the whole space $\ensuremath{\mathbb{R}}^N$, particularly at the origin $x=0$. Note that a radial function belongs to $C^\infty\left(\ensuremath{\mathbb{R}}^N\right)$ if it can be written as a smooth function of $|x|^2$. For $v_e$, since the map $z\mapsto \mathrm{e}^{-z} {_1F_1}\left(\frac{N-c_e}{2},\,\frac{N}{2},\,z\right)$ is an entire function on $\ensuremath{\mathbb{C}}$, and $z(x)=\frac{b|x|^2}{2}$ is smooth in $x$, their composition is smooth. Thus $v_e\in C^\infty\left(\ensuremath{\mathbb{R}}^N\right)$. For $v_o$, we use the change of variables $\tau=s^2/2$ and the integral can be rewritten as:
\begin{equation*}
    v_o(x)=-\int_{\frac{|x|^2}{2}}^{+\infty}\mathrm{e}^{-{b\tau}}{_1F_1}\left(\dfrac{2-c_o}{2},\,\dfrac{N+2}{2},\,{b\tau}\right)\mathrm{\,d} \tau.
\end{equation*}
The fundamental theorem of calculus ensures that the map $$t\mapsto -\int_{t}^{+\infty}\mathrm{e}^{-{b\tau}}{_1F_1}\left(\dfrac{2-c_o}{2},\,\dfrac{N+2}{2},\,{b\tau}\right)\mathrm{\,d} \tau$$ belongs to $C^\infty\left([0,\,+\infty)\right)$. Since $t(x)=\frac{|x|^2}{2}$ is smooth in $x$, it follows that $v_o(x)\in C^\infty\left(\ensuremath{\mathbb{R}}^N\right)$. 

Next, we show that $\mathcal{E}_{b}(v_e)=0$. Let $z=\dfrac{br^2}{2}$ and $M_{e}(z)={_1F_1}\left(\dfrac{N-c_e}{2},\,\dfrac{N}{2},\,z\right)$. Then $v_e(r)=\mathrm{e}^{-z}M_e(z)$ and
\begin{align*}
    v_e'(r)&=\frac{\mathrm{\,d} v_e}{\mathrm{\,d} z}\frac{\mathrm{\,d} z}{\mathrm{\,d} r}=br\mathrm{e}^{-z}\left(M_e'(z)-M_e(z)\right),\\
    v_e''(r)&=b\mathrm{e}^{-z}\left(2zM_e''(z)+(1-4z)M_e'(z)+(2z-1)M_e(z)\right).
\end{align*}
It follows that
\begin{align*}
    \mathcal{E}_b(v_e)&=v_e''(r)+\left(\frac{N-1}{r}+br\right)v_e'(r)+bc_ev_e(r)\\
    &=b\mathrm{e}^{-z}\left[2zM_e''(z)+(N-2z)M_e'(z)+(c_e-N)M_e(z)\right].
\end{align*}
Notice that $M_e$ satisfies the confluent hypergeometric equation
\begin{equation*}
    zM_e''(z)+\left(\frac{N}{2}-z\right)M_e'(z)-\frac{N-c_e}{2} M_e(z)=0.
\end{equation*}
Hence, we have $\mathcal{E}_b\left(v_e\right)=0$. By a similar argument, one can prove that $\mathcal{O}_b\left(v_o\right)=0$. Denote $z=\dfrac{br^2}{2}$ and $M_o(z)={_1F_1}\left(\dfrac{2-c_o}{2},\,\dfrac{N+2}{2},\,z\right)$, then we have
\begin{align*}
    v_o'(r)&=r\mathrm{e}^{-z}M_o(z),\\
    v_o''(r)&=\mathrm{e}^{-z}\left((1-2z)M_o(z)+2zM_o'(z)\right),\\
    v_o''(r)&=br\mathrm{e}^{-z}\left(2zM_o''(z)+(3-4z)M_o'(z)+(2z-3)M_o(z)\right).
\end{align*}
By a direct computation, we have
\begin{align*}

\end{align*}
As $M_o$ satisfies the confluent hypergeometric equation
\begin{equation*}
    zM_o''(z)+\left(\frac{N+2}{2}-z\right)M_o'(z)-\frac{2-c_o}{2} M_o(z)=0,
\end{equation*}
we have $\mathcal{O}_b\left(v_o\right)=0$.

<!-- definition-1: body_md -->
For $a\in\ensuremath{\mathbb{R}}$ and $\sigma\in \ensuremath{\mathbb{N}}_+\cup\{0\}$, define
\begin{equation*}
    \mathcal{W}(a,\,\sigma)=\left\{f\in C^\infty\left((0,\,+\infty)\right):\,\text{for all }l\in\ensuremath{\mathbb{N}}_+\cup\{0\},\,f^{(l)}(r)=O\left(r^{-a-l}\ln^\sigma r\right)\text{ as }r\to+\infty\right\}.
\end{equation*}

<!-- remark-3: body_md -->
By the definition of the set $\mathcal{W}(a,\,\sigma)$, the following conclusions can be easily verified:
\begin{enumerate}
    \item For any $\sigma_1,\,\sigma_2\in\ensuremath{\mathbb{N}}_+\cup\{0\}$, if $a_1>a_2$, then $\mathcal{W}\left(a_1,\,\sigma_1\right)\subset\mathcal{W}\left(a_2,\,\sigma_2\right)$.
\item For any $a\in\ensuremath{\mathbb{R}}$, if $\sigma_1\leq\sigma_2$, then $\mathcal{W}\left(a,\,\sigma_1\right)\subset\mathcal{W}\left(a,\,\sigma_2\right)$.
\item If $f(r)\in\mathcal{W}(a,\,\sigma)$ and $k\in\ensuremath{\mathbb{R}}$, then we have $r^kf(r)\in \mathcal{W}(a-k,\,\sigma)$.
\end{enumerate}

<!-- lemma-5: body_md -->
Let $N\geq 2$, $a\in\ensuremath{\mathbb{R}}$ and $\sigma\in\ensuremath{\mathbb{N}}_+\cup\{0\}$. Define
\begin{equation*}
    H(r)=\begin{cases}
    r^{2-N},\qquad &\text{if }N\geq3,\\
    \ln r,&\text{if }N=2.
    \end{cases}
\end{equation*}
For $f\in\mathcal{W}(a,\,\sigma)\cap C([0,\,+\infty))$ and large $
r>0$, there holds:
\begin{align*}
\mathcal{I}[f](r)=C_1+C_2H(r)+R(r),
\end{align*}
where $\mathcal{I}[f]$ is defined by \eqref{def of I}, $C_1$ and $C_2$ are constants depending on $f$ and do not depend on $r$. Moreover, the remainder $R\in \mathcal{W}(a-2,\,\sigma+2)$.
\label{estimate of I}
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
Since $f\in\mathcal{W}(a,\,\sigma)$, there exists some $r_0>0$ and $C>0$ such that $|f(r)|\leq Ct^{-a}\ln^\sigma(r)$ for $r\geq r_0$. Denote $\mathcal{J}[f](s)=s^{1-N}\int_0^s t^{N-1}f(t)\mathrm{\,d} t$. We shall prove this lemma by considering the following three cases.
\\
\textbf{Case 1: }$a>N$. As $N-a<0$, the integral $\int_0^{+\infty} t^{N-1}f(t)\mathrm{\,d} t$ converges. It follows that
\begin{align}
    \mathcal{J}[f](s)&=s^{1-N}\left[\int_0^\infty t^{N-1}f(t)\mathrm{\,d} t-\int_s^\infty t^{N-1}f(t)\mathrm{\,d} t\right]\nonumber\\
    &=Cs^{1-N}-s^{1-N}\int_s^{+\infty}t^{N-1}f(t)\mathrm{\,d} t.
    \label{Jf in case 1}
\end{align}
Note that
\begin{align*}
    \left|-\int_s^{+\infty}t^{N-1}f(t)\mathrm{\,d} t\right|\leq C'\int_s^{+\infty}t^{N-1-a}\ln^\sigma(t)\mathrm{\,d} t\leq C' s^{N-a}\ln^\sigma(s)\qquad\text{for large } s.
\end{align*}
It follows that 
\begin{align*}
&\frac{\mathrm{\,d} }{\mathrm{\,d} s}\left(-\int_s^{+\infty}t^{N-1}f(t)\mathrm{\,d} t\right)=s^{N-1}f(s)\in\mathcal{W}(a-N+1,\,\sigma),\\
& -\int_s^{+\infty}t^{N-1}f(t)\mathrm{\,d} t\in\mathcal{W}(a-N,\,\sigma),\\
&R_1(s)\coloneqq -s^{1-N}\int_s^{+\infty}t^{N-1}f(t)\mathrm{\,d} t\in\mathcal{W}(a-1,\,\sigma).
\end{align*}
Then, by \eqref{Jf in case 1}, for large $r>0$, we have
\begin{align}
    \mathcal{I}[f](r)&=\int_0^1\mathcal{J}[f](s)\mathrm{\,d} s+C\int_1^r s^{1-N}\mathrm{\,d} s+\int_1^{+\infty}R_1(s)\mathrm{\,d} s-\int_r^{+\infty}R_1(s)\mathrm{\,d} s\nonumber\\
    &=C_1+C_2 H(r)-\int_r^{+\infty}R_1(s)\mathrm{\,d} s.
    \label{If in case 1}
\end{align}
Similarly, $R(r)=-\int_r^{+\infty}R_1(s)\mathrm{\,d} s$ satisfies:
\begin{align*}
    &\left|-\int_r^{+\infty}R_1(s)\mathrm{\,d} s\right|\leq C'\int_r^{+\infty}s^{1-a}\ln^\sigma(s)\mathrm{\,d} s\leq C'r^{2-a}\ln^\sigma(r),\\
    &R'(r)=R_1(r)\in \mathcal{W}(a-1,\,\sigma).
\end{align*}
Hence, $R(r)\in\mathcal{W}(a-2,\,\sigma)\subset\mathcal{W}(a-2,\,\sigma+2)$.
\\
\textbf{Case 2: }$a=N$. For large $s>r_0$, there holds:
\begin{align*}
    \left|\int_{r_0}^st^{N-1}f(t)\mathrm{\,d} t\right|\leq C\int_{r_0}^st^{N-1}\cdot t^{-N}\ln^{\sigma}(t)\mathrm{\,d} t\leq C\ln^{\sigma+1}(s).
\end{align*}
Notice that
\begin{align*}
    \frac{\mathrm{\,d}}{\mathrm{\,d} s}\left(\int_{r_0}^st^{N-1}f(t)\mathrm{\,d} t\right)&=s^{N-1}f(s)\in\mathcal{W}(1,\,\sigma)\subset\mathcal{W}(1,\,\sigma+1).
\end{align*}
Hence, we have
\begin{align*}
    \int_{0}^st^{N-1}f(t)\mathrm{\,d} t&=\int_{0}^{r_0}t^{N-1}f(t)\mathrm{\,d} t+\int_{r_0}^st^{N-1}f(t)\mathrm{\,d} t\in\mathcal{W}(0,\,\sigma+1)
\end{align*}
Then $\mathcal{J}[f](s)=s^{1-N}\int_0^s t^{N-1}f(t)\mathrm{\,d} t\in\mathcal{W}(N-1,\,\sigma+1)$. There exist some $r_1>0$ and $C>0$ such that
\begin{equation*}
    \left|\mathcal{J}[f](r)\right|\leq Cr^{1-N}\ln^{\sigma+1}(r)\qquad\text{for }r\geq r_1.
\end{equation*}

If $N>2$, then $\int_0^{+\infty}\mathcal{J}[f](s)\mathrm{\,d} s$ converges and we denote its value by $C_1$. Therefore
\begin{align*}
    \mathcal{I}[f](r)=C_1+R(r)\qquad\text{for }R(r)=-\int_{r}^{+\infty}\mathcal{J}[f](s)\mathrm{\,d} s.
\end{align*}
For sufficiently large $r>r_1$, we obtain
\begin{align*}
    &|R(r)|\leq C\int_{r}^{+\infty}s^{1-N}\ln^{\sigma+1}(s)\mathrm{\,d} s\leq C r^{2-N}\ln^{\sigma+1}(r),\\
    &R'(r)=\mathcal{J}[f](s)=s^{1-N}\int_0^s t^{N-1}f(t)\mathrm{\,d} t\in\mathcal{W}(N-1,\,\sigma+1).
\end{align*}
That is, $R(r)\in\mathcal{W}(N-2,\,\sigma+1)\subset\mathcal{W}(a-2,\,\sigma+2)$.

If $N=2$, for sufficiently large $r>r_1$, we have
\begin{align*}
    \left|\mathcal{I}[f](r)\right|&\leq \left|\int_0^{r_1}\mathcal{J}[f](s)\mathrm{\,d} s\right|+C\int_{r_1}^rs^{-1}\ln^{\sigma+1}(s)\mathrm{\,d} s\leq C'\ln^{\sigma+2}(r),\\
    \mathcal{I}[f]'(r)&=\mathcal{J}[f](r)\in\mathcal{W}(1,\,\sigma+1)\subset\mathcal{W}(1,\,\sigma+2).
\end{align*}
Consequently, $\mathcal{I}[f](r)\in\mathcal{W}(0,\,\sigma+2)=\mathcal{W}(a-2,\,\sigma+2)$.
\\
\textbf{Case 3: }$a<N$. Since $f\in\mathcal{W}(a,\,\sigma)$, for large $s$, there holds:
\begin{align*}
    \left|\int_0^st^{N-1}f(t)\mathrm{\,d} t\right|&\leq \left|\int_0^{r_0}t^{N-1}f(t)\mathrm{\,d} t\right|+C\int_{r_0}^st^{N-1-a}\ln^{\sigma}(t)\mathrm{\,d} t\\
    &\leq C's^{N-a}\ln^{\sigma}(s).
 \end{align*}
It follows that
\begin{align*}
&\frac{\mathrm{\,d}}{\mathrm{\,d} s}\left(\int_0^st^{N-1}f(t)\mathrm{\,d} t\right)=s^{N-1}f(s)\in\mathcal{W}(a-N+1,\,\sigma),\\
&\int_0^st^{N-1}f(t)\mathrm{\,d} t\in \mathcal{W}(a-N,\,\sigma).
\end{align*}
Then we have
\begin{align*}
    &\mathcal{J}[f](s)=s^{1-N}\int_0^st^{N-1}f(t)\mathrm{\,d} t\mathrm{\,d} s\in\mathcal{W}(a-1,\,\sigma).
\end{align*}
\\
Thus, there exist some $r_2>0$ and $C>0$ such that
\begin{equation*}
    \left|\mathcal{J}[f](r)\right|\leq Cr^{1-a}\ln^{\sigma}(r)\qquad\text{for }r\geq r_2.
\end{equation*}

If $2<a<N$, then $a-1>1$,  $\int_0^{+\infty}\mathcal{J}[f](s)\mathrm{\,d} s$ converges and we denote its value by $C_1$. Therefore, we arrive at
\begin{align*}
    \mathcal{I}[f](r)=C_1+R(r)\qquad\text{for }R(r)=-\int_{r}^{+\infty}\mathcal{J}[f](s)\mathrm{\,d} s.
\end{align*}
For sufficiently large $r>r_2$, we obtain
\begin{align*}
    &|R(r)|\leq C\int_{r}^{+\infty}s^{1-a}\ln^{\sigma}(s)\mathrm{\,d} s\leq C r^{2-a}\ln^{\sigma}(r),\\
    &R'(r)=\mathcal{J}[f](s)=s^{1-N}\int_0^s t^{N-1}f(t)\mathrm{\,d} t\in\mathcal{W}(a-1,\,\sigma).
\end{align*}
That is, $R(r)\in\mathcal{W}(a-2,\,\sigma)\subset\mathcal{W}(a-2,\,\sigma+2)$.

If $a=2<N$ or $a<2$, for sufficiently large $r>r_2$, we have
\begin{align*}
    \left|\mathcal{I}[f](r)\right|&\leq \left|\int_0^{r_2}\mathcal{J}[f](s)\mathrm{\,d} s\right|+C\int_{r_2}^rs^{1-a}\ln^{\sigma}(s)\mathrm{\,d} s\leq C'r^{2-a}\ln^{\sigma+1}(r),\\
    \mathcal{I}[f]'(r)&=\mathcal{J}[f](r)\in \mathcal{W}(a-1,\,\sigma)\subset\mathcal{W}(a-1,\,\sigma+1).
\end{align*}
This implies that $\mathcal{I}[f](r)\in\mathcal{W}(a-2,\,\sigma+1)\subset\mathcal{W}(a-2,\,\sigma+2)$.

<!-- lemma-6: body_md -->
\label{asymptotic of v}
For $N\geq2$, $m\geq2$ and $b>0$, we have
\begin{align*}
v_e(r,\,b)\in\mathcal{W}\left(c_e,\,0\right),\qquad v_o(r,\,b)\in\mathcal{W}\left(c_o+N-2,\,0\right).
\end{align*}
<!-- end lemma-6 -->
<!-- proof-7: body_md -->
By Kummer’s transformation formula given in Lemma \ref{property of F}, we have
\begin{align*}
v_e(r,\,b)={_1F_1}\left(\dfrac{c_e}{2},\,\dfrac{N}{2},\,-z\right)\qquad\text{for } z=\dfrac{br^2}{2}.
\end{align*}
Together with the differentiation formula from Lemma \ref{property of F} and Fa\`a di Bruno's formula for the derivative of composite functions (see \cite{faadibruno1857}), for $k\in\ensuremath{\mathbb{N}}_+\cup\{0\}$, there holds
\begin{align*}
    \frac{\mathrm{\,d}^k}{\mathrm{\,d}^k r}v_e(r,\,b)&=\sum_{l=0}^{\lfloor k/2\rfloor}C(k,\,l)r^{k-2l}{_1F_1}\left(\dfrac{c_e}{2}+k-l,\,\dfrac{N}{2}+k-l,-z\right)\\
    &=\sum_{l=0}^{\lfloor k/2\rfloor}C(k,\,l)r^{k-2l}\mathrm{e}^{-z}{_1F_1}\left(\dfrac{N-c_e}{2},\,\dfrac{N}{2}+k-l,z\right)\\
    &=O\left(r^{-c_e-k}\right)\text{  as }r\to+\infty,
\end{align*}
where the last equality follows from the asymptotic estimate given in Lemma \ref{property of F}. This implies that $v_e(r,\,b)\in\mathcal{W}\left(c_e,\,0\right)$. Similarly, we have
\begin{align*}
\frac{\mathrm{\,d}}{\mathrm{\,d} r}v_o(r,\,b)&=r\mathrm{e}^{-z}{_1F_1}\left(\dfrac{2-c_o}{2},\,\dfrac{N+2}{2},\,z\right)=r\cdot{_1F_1}\left(\dfrac{N+c_o}{2},\,\dfrac{N+2}{2},\,-z\right),\\
              
                
\frac{\mathrm{\,d}^{k+1}}{\mathrm{\,d}^{k+1} r}v_o(r,\,b)&=r\sum_{l=0}^{\lfloor k/2\rfloor}C(k,\,l)r^{k-2l}{_1F_1}\left(\dfrac{N+c_o}{2}+k-l,\,\dfrac{N+2}{2}+k-l,-z\right)\\
&\qquad+k\sum_{l=0}^{\lfloor (k-1)/2\rfloor}C(k,\,l)r^{k-1-2l}{_1F_1}\left(\dfrac{N+c_o}{2}+k-1-l,\,\dfrac{N+2}{2}+k-1-l,-z\right)\\
&=O\left(r^{1-N-c_o-k}\right) \text{  as }r\to+\infty.
\end{align*}
Hence, we obtain $v_o(r,\,b)\in\mathcal{W}\left(N+c_o-2,\,0\right)$.

<!-- proposition-2: body_md -->
\label{expression of Ik}
Let $N\geq2,\,j\in \ensuremath{\mathbb{N}}_+$ and $j>1$. Denote by  $\mathcal{I}^k$ the $k$-th iteration of $\mathcal{I}$ for $k\in\ensuremath{\mathbb{N}}_+$, i.e. $\mathcal{I}^k[\cdot]=\mathcal{I}\left[\mathcal{I}^{k-1}[\cdot]\right]$. For  large $r>0$, we have
\begin{align*}
    \mathcal{I}^k\left[v_e\right](r)&=\sum_{i=1}^{n_k}f_i(r)+Q_{2k-2}(r)\qquad\text{   for }m=2j,\\
    \mathcal{I}^{k}\left[v_o\right](r)&=\sum_{i=1}^{\tilde n_k}\tilde{f}_i(r)+\tilde{Q}_{2k-2}(r)\qquad\text{   for }m=2j+1,
\end{align*}
where $f_i\in \mathcal{W}\left(N-2k,\,\sigma_i\right)$  and $\tilde{f}_i\in \mathcal{W}\left(N-2k,\,\tilde{\sigma}_i\right)$ with $\sigma_i,\,\tilde{\sigma}_i\in\ensuremath{\mathbb{N}}_+\cup\{0\}$. $Q_{2k-2}$ and $\tilde{Q}_{2k-2}$ are polynomials of degree at most $(2k-2)$  containing only even powers.
<!-- end proposition-2 -->
<!-- proof-8: body_md -->
We proceed by induction. For the base case $k=1$, by Lemma \ref{asymptotic of v}, we have $v_e\in\mathcal{W}\left(c_e,\,0\right)$ and $v_o\in\mathcal{W}\left(N+c_o-2,\,0\right)$. From Lemma \ref{estimate of I}, for large $r>0$, there holds:
\begin{align*}
    \mathcal{I}\left[v_e\right](r)&=C_1+C_2H(r)+R_1(r),\\
    \mathcal{I}\left[v_o\right](r)&=\tilde{C}_1+\tilde{C}_2H(r)+\tilde{R}_1(r),
\end{align*}
where $H\in\mathcal{W}\left(N-2,\,1\right),\,R_1\in\mathcal{W}\left(c_e-2,\,2\right)$ and $\tilde{R}_1\in\mathcal{W}\left(N+c_o-4,\,2\right)$. Since $c_e\geq N$ for $m=2j$ and $N+c_o-2\geq N$ for $m=2j+1$, we have $R_1\in\mathcal{W}\left(N-2,\,2\right)$ and $\tilde{R}_1\in\mathcal{W}(N-2,\,2)$. Thus, the case $k=1$ is proved.

Now, assuming the statement holds for $1\leq k\leq j-2$, we prove it for $k+1$. In the case $m=2j$, suppose that for some $r_0>0$, we have $\mathcal{I}^k\left[v_e\right](r)=\sum_{i=1}^{n_k} f_i(r)+Q_{2k-2}(r)$ for $r\geq r_0$. It follows that
\begin{align*}
    \mathcal{I}^{k+1}\left[v_e\right](r)&=\int_{0}^{r_0}s^{1-N}\int_0^s t^{N-1}\mathcal{I}^k\left[v_e\right](t)\mathrm{\,d} t\mathrm{\,d} s\\
    &\qquad+\int_{r_0}^rs^{1-N}\left[\int_0^{r_0}t^{N-1}\mathcal{I}^k\left[v_e\right](t)\mathrm{\,d} t+\int_{r_0}^st^{N-1}\mathcal{I}^k\left[v_e\right](t)\mathrm{\,d} t\right]\mathrm{\,d} s\\
    &=C'+\int_{r_0}^rs^{1-N}\left[C+\int_{r_0}^st^{N-1}\mathcal{I}^k\left[v_e\right](t)\mathrm{\,d} t\right]\mathrm{\,d} s\\
    &=C'+CH(r)+\int_{r_0}^rs^{1-N}\int_{r_0}^st^{N-1}\mathcal{I}^k\left[v_e\right](t)\mathrm{\,d} t\mathrm{\,d} s.
\end{align*}
Now substitute $\mathcal{I}^k\left[v_e\right](r)=\sum_{i=1}^{n_k} f_i(r)+Q_{2k-2}(r)$ into the above integral. Notice that $Q_{2k-2}$ is a polynomial of degree at most $(2k-2)$  containing only even powers, then $$\int_{r_0}^rs^{1-N}\int_{r_0}^st^{N-1} Q_{2k-2}(t)\mathrm{\,d} t\mathrm{\,d} s$$ is the sum of a polynomial of degree at most $2k$  containing only even powers and a constant multiple of $H(r)$. Moreover, for $f_i\in\mathcal{W}\left(N-2k,\,\sigma_i\right)$, repeating the proof of Lemma  \ref{estimate of I}, except that the lower limits of the two integrals in $\mathcal{I}$ are replaced by $r_0$, we obtain
\begin{align*}
    \int_{r_0}^rs^{1-N}\int_{r_0}^st^{N-1}f_i(t)\mathrm{\,d} t\mathrm{\,d} s
    &=C'+CH(r)+R_{f_i}(r),
\end{align*}
for large $r>0$, where $R_{f_i}\in \mathcal{W}\left(N-2k-2,\,\sigma_i+2\right)$. Therefore, for sufficiently large $r>0$, we obtain
\begin{align*}
    \mathcal{I}^{k+1}\left[v_e\right](r)&=\sum_{i=1}^{n_{k+1}}\bar f_i(r)+Q_{2k}(r),
\end{align*}
for $\bar f_i\in\mathcal{W}(N-2(k+1),\,\sigma_i+2)$ and $Q_{2k}$ is a polynomial of degree at most $2k$  containing only even powers. For the case $m=2j+1$, the computation for $\mathcal{I}^{k+1}\left[v_o\right]$ is similar to the above, so we omit it here. Thus we complete the proof of this proposition.

<!-- proposition-3: body_md -->
\label{estimate of Phi radially}
For $N\geq2$, $m\geq2$ and $b>0$, we denote  $j=\lfloor m/2\rfloor$ and $r=|x|$. Define
\begin{align}
\label{def of extremal}
    \Phi_{b,m}(x)=\Phi_{b,m}(r)=\begin{cases}
    v_e(r,\,b),&\text{ if }m=2,\\
    v_o(r,\,b),&\text{ if }m=3,\\
    \mathcal{I}^{j-1}\left[v_e\right](r)-Q_{2j-4}(r),&\text{ if } m=2j\quad\text{ for }j>1,\\
    \mathcal{I}^{j-1}\left[v_o\right](r)-\tilde{Q}_{2j-4}(r),&\text{ if } m=2j+1\quad\text{ for }j>1,
    \end{cases}
\end{align}
where $Q_{2j-4}$ and $\tilde{Q}_{2j-4}$ are polynomials of degree at most $(2j-4)$  containing only even powers given in Proposition \ref{expression of Ik}. Then the following equation holds on $\ensuremath{\mathbb{R}}^N$:
\begin{equation}
\label{relationship between Phi and v}
    \Delta^{j-1}\Phi_{b,m}=\begin{cases}
    v_e\,,\qquad&\text{if $m=2j$},\\
    v_o\,,&\text{if $m=2j+1$}.
    \end{cases}
\end{equation}
Furthermore, for all $m\geq2$, we have $\Phi_{b,m}(r)\in\mathcal{W}(N-m+2,\,\sigma)$ for some $\sigma\in\ensuremath{\mathbb{N}}_+\cup\{0\}$ depending on $m$.

Notice that
\begin{equation*}
    \Delta \left(\mathcal{I}\left[v_e\right]\right)= \frac{\mathrm{\,d}^2 \mathcal{I}\left[v_e\right]}{\mathrm{\,d}^2 r}(r)+\frac{N-1}{r}\frac{\mathrm{\,d} \mathcal{I}\left[v_e\right]}{\mathrm{\,d} r}(r)=v_e.
\end{equation*}
Additionally, for $j>1$, $\Delta^{j-1}Q_{2j-4}=0$ since the degree of  $Q_{2j-4}$ is less than $2j-2$. Therefore, we obtain $\Delta^{j-1}\Phi_{b,m}=v_e$ for $m=2j$. By a similar argument, we have $\Delta^{j-1}\Phi_{b,m}=v_o$ for $m=2j+1$.

As for the asymptotic properties of $\Phi_{b,m}(r)$, by Lemma \ref{asymptotic of v}, we obtain:
\begin{align*}
    \Phi_{b,2}(r)&=v_e(r,\,b)\in \mathcal{W}\left(c_e,\,0\right)=\mathcal{W}\left(N,\,0\right),\\
    \Phi_{b,3}(r)&=v_o\left(r,\,b\right)\in \mathcal{W}\left(c_o+N-2,\,0\right)\subset \mathcal{W}\left(N-1,\,0\right).
\end{align*}
Here we used the fact that $c_e=N$ and $c_o\in(1,2)$ when $j=1$ and $N\geq2$. In addition, by Lemma \ref{expression of Ik}, for sufficiently large $r>0$, we have the following result for $j>1$:
\begin{align*}
    \Phi_{b,m}(r)=\begin{cases}
    \sum_{i=1}\limits^{n}f_i(r),\qquad&\text{if }m=2j,\\
    \sum_{i=1}\limits^{\tilde n}\tilde{f}_i(r),&\text{if }m=2j+1,
    \end{cases}
\end{align*}
where $f_i\in\mathcal{W}\left(N-2j+2,\,\sigma_i\right)$ and $\tilde{f}_i\in\mathcal{W}\left(N-2j+2,\,\tilde{\sigma}_i\right)$ for some $\sigma_i,\,\tilde{\sigma}_i\in\ensuremath{\mathbb{N}}_+\cup\{0\}$. Thus, we can choose $\sigma=\max\{\sigma_1,\,\dots,\,\sigma_n\}\text{ for }m=2j\text{ and }\sigma= \{\tilde{\sigma}_1,\,\dots,\,\tilde{\sigma}_{\tilde{n}}\}$ for $m=2j+1$ such that
\begin{align*}
\Phi_{b,m}(r)\in\mathcal{W}(N-2j+2,\,\sigma)\subset \mathcal{W}\left(N-m+2,\,\sigma\right).
\end{align*}

<!-- lemma-7: body_md -->
\label{estimate of Lk}
 For any $k\in\ensuremath{\mathbb{N}}_+$, there holds $\mathcal{L}_k(x)=O\left(|x|^{k-1}\right)$ as $|x|\to+\infty$.
<!-- end lemma-7 -->
<!-- proof-10: body_md -->
Notice that
 \begin{align*}
     \mathcal{L}_k(x)&=\frac{1}{(k-1)!}\int_{0}^x(x-t)^{k-1}\mathrm{e}^{-bt^2}\mathrm{\,d} t\\
     &=\frac{1}{(k-1)!}\int_{0}^{-x}(x+t)^{k-1}\mathrm{e}^{-bt^2}(-1)\mathrm{\,d} t\\
     &=\frac{(-1)^k}{(k-1)!}\int_{0}^{-x}(-x-t)^{k-1}\mathrm{e}^{-bt^2}\mathrm{\,d} t\\
     &=(-1)^k \mathcal{L}_k(-x).
 \end{align*}
 Therefore, we only need to consider the case $x\to+\infty$. Expanding $(x-t)^{k-1}$, we obtain
 \begin{align}
 \label{expansion of Lk}
     \mathcal{L}_k(x)&=\sum_{l=0}^{k-1}C(k,\,l)x^{k-1-l}\int_0^x t^l\mathrm{e}^{-bt^2}\mathrm{\,d} t\nonumber\\
     &=\sum_{l=0}^{k-1}C(k,\,l)x^{k-1-l}\int_0^{+\infty} t^l\mathrm{e}^{-bt^2}\mathrm{\,d} t-\sum_{l=0}^{k-1}C(k,\,l)x^{k-1-l}\int_x^{+\infty} t^l\mathrm{e}^{-bt^2}\mathrm{\,d} t.
 \end{align}
 Next, we will prove $\int_x^{+\infty} t^l\mathrm{e}^{-bt^2}\mathrm{\,d} t=O\left(x^{l-1}\mathrm{e}^{-bx^2}\right)$ as $x\to+\infty$ by induction. For $l=0$ and $l=1$, we have
 \begin{align*}
     \int_x^{+\infty}\mathrm{e}^{-bt^2}\mathrm{\,d} t&\leq\frac{1}{2bx}\int_x^{+\infty}2bt\cdot\mathrm{e}^{-bt^2}\mathrm{\,d} t=\frac{1}{2bx}\mathrm{e}^{-bx^2},\\
     \int_x^{+\infty}t\mathrm{e}^{-bt^2}\mathrm{\,d} t&=-\frac{1}{2b}\int_x^{+\infty}\mathrm{\,d} \left(\mathrm{e}^{-bt^2}\right)=\frac{1}{2b}\mathrm{e}^{-bx^2}.
 \end{align*}
 Combining the fact that
 \begin{align*}
     \int_{x}^{+\infty}t^l\mathrm{e}^{-bt^2}\mathrm{\,d} t&=-\frac{1}{2b}\int_x^{+\infty}t^{l-1}\mathrm{\,d} \left(\mathrm{e}^{-bt^2}\right)=\frac{1}{2b}x^{l-1}\mathrm{e}^{-bx^2}+\frac{l-1}{2b}\int_x^{+\infty}t^{l-2}\mathrm{e}^{-bt^2}\mathrm{\,d} t,
 \end{align*}
 we conclude that $\int_x^{+\infty} t^l\mathrm{e}^{-bt^2}\mathrm{\,d} t=O\left(x^{l-1}\mathrm{e}^{-bx^2}\right)$. Substituting this estimate into \eqref{expansion of Lk}, we obtain $\mathcal{L}_k(x)=O\left(|x|^{k-1}\right)$.

<!-- lemma-8: body_md -->
Let $N\geq2$, $m\geq2$ and $b>0$. Then, for any multi-index $\beta$ and $\Phi_{b,m}(x)$ given by \eqref{def of extremal}, as $r=|x|\to+\infty$, we have
\begin{equation*}
    D^\beta\Phi_{b,m}(x)=O\left(r^{m-2-N-|\beta|}\ln^{\sigma}(r)\right),
\end{equation*}
where $\sigma\in\ensuremath{\mathbb{N}}_+\cup\{0\}$ depending on $m$.
\label{estimate of Phi}
<!-- end lemma-8 -->
<!-- proof-11: body_md -->
When $|\beta|=0$, the estimate of $\Phi_{b,m}$ follows directly from  Proposition  \ref{estimate of Phi radially}. When $|\beta|\geq1$, for radially symmetric function $\Phi_{b,m}$, it is easy to prove by induction that
\begin{align}
    D^\beta\Phi_{b,m}(x)=\sum_{l=1}^{|\beta|}f_{\beta,l}(x)\Phi_{b,m}^{(l)}(r).
    \label{derivative of radial function}
\end{align}
In addition, $f_{\beta,l}$ satisfies
\begin{equation*}
    |f_{\beta,l}(x)|\leq C(\beta,\,l)|x|^{l-|\beta|}.
\end{equation*}
The proposition is proved by the fact  that $\Phi_{b,m}(r)\in\mathcal{W}(N-m+2,\,\sigma)$ for some $\sigma\in\ensuremath{\mathbb{N}}_+\cup\{0\}$.

<!-- lemma-9: body_md -->
\label{cut off of Phi}
Let $N\geq2$, $m\geq2$ and $b>0$. For $L>0$, we denote $u_L(x)=\eta_{L}(x)\Phi_{b,m}(x)$, where $\Phi_{b,m}$ is given by \eqref{def of extremal} and $\eta_L$ is given by \eqref{def of eta L}. Define $j=\lfloor m/2\rfloor$ and $v_L=\Delta^{j-1}u_L$. If $m$ is even, then
\begin{equation*}
    \lim_{L\to+\infty}\int_{\ensuremath{\mathbb{R}}^N}\lvert\mathcal{E}_b\left( v_L\right)\rvert^2\mathrm{\,d} x=0.
\end{equation*}
If $m$ is odd, then
\begin{equation*}
    \lim_{L\to+\infty}\int_{\ensuremath{\mathbb{R}}^N}\lvert\mathcal{O}_b \left(v_L\right)\rvert^2\mathrm{\,d} x=0.
\end{equation*}
<!-- end lemma-9 -->
<!-- proof-12: body_md -->
A direct computation shows that
\begin{align}
    v_L=\begin{cases}
    u_L,&\text{if }j=1,\\
    \eta_L\Delta^{j-1}\Phi_{b,m}+w_L,\quad&\text{if }j>1,
    \end{cases}
        \label{expression of vL}
\end{align}
where $w_L$ is a linear combination of products in the form $D^{\beta_1}\eta_L\cdot D^{\beta_2}\Phi_{b,m}$ for $|\beta_1|+|\beta_2|=2j-2$ and $|\beta_1|\geq1$.
\\
\textbf{Case 1: }$m$ is even. In this case,  we have $\Delta^{j-1}\Phi_{b,m}(x)=v_e(x,\,b)$. By Proposition \ref{expression of v} and Lemma \ref{asymptotic of v}, we obtain
\begin{align*}
    \mathcal{E}_b\left(\eta_L v_e\right)&=\eta_L\mathcal{E}_b\left(v_e\right)+v_e\Delta\eta_L+2\nabla v_e\nabla\eta_L+bv_e\left(x\cdot\nabla\eta_L\right)\\
    &=v_e\Delta\eta_L+2\nabla v_e\nabla\eta_L+bv_e\left(x\cdot\nabla\eta_L\right)\\
    &=O\left(|x|^{-c_e}\right),\qquad\text{as }|x|\to+\infty.
\end{align*}
It follows that
\begin{align}
    \int_{\ensuremath{\mathbb{R}}^N}\left|\mathcal{E}_b\left(\eta_L v_e\right)\right|^2\mathrm{\,d} x&=\int_{\left\{x\in\ensuremath{\mathbb{R}}^N:\,L\leq|x|\leq2 L\right\}}\left|\mathcal{E}_b\left(\eta_L v_e\right)\right|^2\mathrm{\,d} x\nonumber\\
    &\leq C L^{-2c_e}\int_L^{2L}t^{N-1}\mathrm{\,d} t\nonumber\\
    &=C L^{N-2c_e}\nonumber\\
    &\to 0,\qquad\text{as }L\to+\infty.
    \label{cut-off for even case 1}
\end{align}
Here we use the fact that $c_e>N/2$. Moreover, when $j>1$, we need to consider the term $w_L$. Since $w_L$ is a linear combination of products of partial derivatives of $\eta_L$ and $\Phi_{b,m}$, from \eqref{estimate of eta} and Lemma \ref{estimate of Phi} we obtain the following estimate:
\begin{align}
    \left|w_L\right|\leq CL^{-N}\ln^{\sigma}(L),\qquad\left|\nabla w_L\right|\leq CL^{-N-1}\ln^{\sigma}(L),\qquad\left|\Delta w_L\right|\leq CL^{-N-2}\ln^{\sigma}(L),
    \label{estimate of wL}
\end{align}
for sufficiently large $L>0$, where $C>0$ and $\sigma\in\ensuremath{\mathbb{N}}_+\cup\{0\}$ are constants independent of $L$. Therefore, we have
\begin{align}
        \int_{\ensuremath{\mathbb{R}}^N}\left|\mathcal{E}_b\left(w_L\right)\right|^2\mathrm{\,d} x&=\int_{\left\{x\in\ensuremath{\mathbb{R}}^N:\,L\leq|x|\leq2 L\right\}}\left|\mathcal{E}_b\left(w_L\right)\right|^2\mathrm{\,d} x\nonumber\\
    &\leq C L^{-2N}\ln^{2\sigma}(L)\int_L^{2L}t^{N-1}\mathrm{\,d} t\nonumber\\
    &=C L^{-N}\ln^{2\sigma}(L)\nonumber\\
    &\to 0,\qquad\text{as }L\to+\infty.
    \label{cut-off for even case 2}
\end{align}
Combining \eqref{cut-off for even case 1} and \eqref{cut-off for even case 2}, we conclude that
\begin{equation*}
    \lim_{L\to+\infty}\int_{\ensuremath{\mathbb{R}}^N}\lvert\mathcal{E}_b\left( v_L\right)\rvert^2\mathrm{\,d} x=0.
\end{equation*}
\\
\textbf{Case 2:} $m$ is odd. In this case, we have $\Delta^{j-1}\Phi_{b,m}(x)=v_o(x,\,b)$.  Then we deduce that
\begin{align*}
    \mathcal{O}_b\left(\eta_Lv_o\right)&=\eta_L\mathcal{O}_b\left(v_o\right)+\Delta v_o\nabla\eta_L+2\nabla\left(\nabla\eta_L\cdot\nabla v_o\right)+\nabla\left(v_o\Delta\eta_L\right)\\
    &\qquad\qquad+b\left(2\nabla v_o\cdot\nabla\eta_L+v_o\Delta\eta_L\right)x+bc v_o\nabla\eta_L.
\end{align*}
From Proposition \ref{expression of v} and Lemma \ref{asymptotic of v}, we get $\mathcal{O}_b\left(v_o\right)=0$ and 
\begin{align*}
    \left|\mathcal{O}_b\left(\eta_Lv_o\right)\right|=O\left(|x|^{1-N-c_o}\right),\qquad\text{as }|x|\to+\infty.
\end{align*}
Notice that $N+2c_o>2$, We deduce
\begin{align}
        \int_{\ensuremath{\mathbb{R}}^N}\left|\mathcal{O}_b\left(\eta_L v_o\right)\right|^2\mathrm{\,d} x&=\int_{\left\{x\in\ensuremath{\mathbb{R}}^N:\,L\leq|x|\leq2 L\right\}}\left|\mathcal{O}_b\left(\eta_L v_o\right)\right|^2\mathrm{\,d} x\nonumber\\
    &\leq C L^{2-2N-2c_o}\int_L^{2L}t^{N-1}\mathrm{\,d} t\nonumber\\
    &=C L^{2-N-2c_o}\nonumber\\
    &\to 0,\qquad\text{as }L\to+\infty.
    \label{cut-off for odd case 1}
\end{align}
As for the term $w_L$, which needs to be considered when $j>1$, we have the estimate \eqref{estimate of wL} together with the following estimate
\begin{equation*}
    \left|\nabla\Delta w_L\right|\leq CL^{-N-3}\ln^{\sigma}(L),
\end{equation*}
for large $L>0$. Then we obtain that
\begin{align}
    \int_{\ensuremath{\mathbb{R}}^N}\left|\mathcal{O}_b\left(w_L\right)\right|^2\mathrm{\,d} x&=\int_{\left\{x\in\ensuremath{\mathbb{R}}^N:\,L\leq|x|\leq2 L\right\}}\left|\mathcal{O}_b\left(w_L\right)\right|^2\mathrm{\,d} x\nonumber\\
    &\leq C L^{-2N-2}\ln^{2\sigma}(L)\int_L^{2L}t^{N-1}\mathrm{\,d} t\nonumber\\

\textbf{Case 1:} $m$ is even and $N\geq2$, or $m$ is odd and $N\geq3$.

Notice that $f(t)=\mathrm{e}^{-bt^2}\in \mathcal{W}(a,\,0)$ for any $a>0$ and $b>0$. Therefore, by repeating the induction argument in Proposition \ref{expression of Ik}, for $k\in\ensuremath{\mathbb{N}}_+$,  we obtain:
\begin{align*}
    \mathcal{I}^k\left[\mathrm{e}^{-bt^2}\right](r)&=\sum_{i=1}^{n_k}f_i(r)+Q_{2k-2}(r) \qquad\text{for large }r>0,
\end{align*}
where $f_i\in \mathcal{W}\left(N-2k,\,\sigma_i\right)$  with $\sigma_i\in\ensuremath{\mathbb{N}}_+\cup\{0\}$ and  $Q_{2k-2}$ is a polynomial of degree at most $(2k-2)$ containing only even powers. For $j=\lfloor m/2\rfloor$, $r=|x|$ and $b>0$, we define
\begin{align}
      \Phi^\star_{b,m}(x)=\Phi^\star_{b,m}(r)=\begin{cases}
    \mathcal{I}^{j-1}\left[\mathrm{e}^{-bt^2}\right](r)-Q_{2j-4}(r),&\text{ if } m=2j,\\
    \mathcal{I}^{j}\left[\mathrm{e}^{-bt^2}\right](r)-{Q}_{2j-2}(r),&\text{ if } m=2j+1.
    \end{cases}
    \label{def of Phi star}
\end{align}
It follows that $P_{m-2}\Phi_{b,m}^\star=\mathrm{e}^{-br^2}$ if $m=2j$ and $P_{m-1}\Phi_{b,m}^\star=\mathrm{e}^{-br^2}$ if $m=2j+1$. Then we have
\begin{align*}

\end{align*}

Next we verify that $\Phi_{b,m}^\star$ can be approximated by smooth, compactly supported functions with respect to the norm $\|\cdot\|_{\mathcal{H}^m_1}$ given in \eqref{def of seminorm 1}. By \eqref{def of Phi star}, there exists some $\sigma\in\ensuremath{\mathbb{N}}_+$ depending on $m$ such that
\begin{align*}
    \Phi_{b,m}^\star(r)&\in \mathcal{W}(N-m+2,\,\sigma),\qquad\text{if }m=2j,\\
    \Phi_{b,m}^\star(r)&\in\mathcal{W}(N-m+1,\,\sigma),\qquad\text{if }m=2j+1.
\end{align*}
For any multi-index $\beta$, by \eqref{derivative of radial function}, as $|x|\to+\infty$, there holds
\begin{equation}
    D^\beta\Phi^\star_{b,m}(x)=\begin{cases}O\left(|x|^{m-2-N-|\beta|}\ln^{\sigma}|x|\right), \qquad&\text{if }m=2j,\\
    O\left(|x|^{m-1-N-|\beta|}\ln^{\sigma}|x|\right), &\text{if }m=2j+1.
    \end{cases}
    \label{estimate of Phi star}
\end{equation}
Define $u^\star_L=\eta_L\Phi^\star_{b,m}\in C^\infty_{0,\text{rad}}\left(\ensuremath{\mathbb{R}}^N\right)$ for $\eta_L$ defined by \eqref{def of eta L} and $L>0$. It suffices to prove the following facts:
\begin{align}
\label{approximate Phi star}
    \begin{cases}
    \lim\limits_{L\to+\infty}\int_{\ensuremath{\mathbb{R}}^N}\left|P_{m}\left(u^\star_L-\Phi_{b,m}^\star\right)\right|^2\mathrm{\,d} x=0,\\
     \lim\limits_{L\to+\infty}\int_{\ensuremath{\mathbb{R}}^N}|x|^2\left|P_{m-1}\left(u^\star_L-\Phi_{b,m}^\star\right)\right|^2\mathrm{\,d} x=0,\\
      \lim\limits_{L\to+\infty}\int_{\ensuremath{\mathbb{R}}^N}\left|P_{m-1}\left(u_L^\star-\Phi_{b,m}^\star\right)\right|^2\mathrm{\,d} x=0.
    \end{cases}
\end{align}
Notice that
\begin{align*}
    P_{m}\left(u^\star_L-\Phi^\star_{b,m}\right)=\left(\eta_L-1\right)P_m\Phi^\star_{b,m}+\rho^\star_L,
\end{align*}
where $\rho^\star_L$ is a linear combination of products in the form $D^{\beta_1}\eta_L\cdot D^{\beta_2}\Phi^\star_{b,m}$ for $|\beta_1|+|\beta_2|=m$ and $|\beta_1|\geq1$. From  \eqref{estimate of eta} and \eqref{estimate of Phi star}, as $L\to+\infty$, we have
\begin{align*}

    C\int_L^{+\infty}t^{-2N-2}\ln^{2\sigma}(t)\cdot t^{N-1}\mathrm{\,d} t\to0,\qquad\text{if }m=2j+1.
    \end{cases}
\end{align*}
Similarly, we have
\begin{align*}
    \int_{\ensuremath{\mathbb{R}}^N}\left|P_{m-1}\left(u^\star_L-\Phi_{b,m}^\star\right)\right|^2\mathrm{\,d} x\leq\begin{cases}
    C\int_L^{+\infty}t^{-2N-2}\ln^{2\sigma}(t)\cdot t^{N-1}\mathrm{\,d} t\to0,\qquad\text{if }m=2j,\\
    C\int_L^{+\infty}t^{-2N}\ln^{2\sigma}(t)\cdot t^{N-1}\mathrm{\,d} t\to0,\qquad\text{if }m=2j+1.
    \end{cases}
\end{align*}
\begin{align*}

    C\int_L^{+\infty}t^{-2N+2}\ln^{2\sigma}(t)\cdot t^{N-1}\mathrm{\,d} t\to0,\qquad\text{if }m=2j+1.

\end{align*}

\end{align*}
where $\bar{\mu}_m(N)=\dfrac{(N+1)^2}{4}$ if $m$ is even and $\bar{\mu}_m(N)=\dfrac{(N-1)^2}{4}$ if $m$ is odd.

\begin{align*}

\end{align*}

\end{align}

Since the higher-order Hydrogen uncertainty principle in $\mathcal{H}^m_3$ has already been proved in Proposition \ref{Prop HyUP}, we only need to show the existence of the extremal functions. Notice that $g_1(r)=\mathrm{e}^{-br}\in \mathcal{W}(a,\,0)$ and $g_2(r)=(1+br)\mathrm{e}^{-br}\in \mathcal{W}(a,\,0)$ for any $a>0$ and $b>0$. Thus, following the induction argument in Proposition \ref{expression of Ik}, we have
\begin{align*}
    \mathcal{I}^k\left[\mathrm{e}^{-bt}\right](r)&=\sum_{i=1}^{n_k}f_i(r)+Q_{2k-2}(r),\\
    \mathcal{I}^k\left[(1+bt)\mathrm{e}^{-bt}\right](r)&=\sum_{i=1}^{\tilde{n}_k}\tilde{f}_i(r)+\tilde{Q}_{2k-2}(r),
\end{align*}
for large $r>0$, where $k\in\ensuremath{\mathbb{N}}_+$, $f_i\in \mathcal{W}\left(N-2k,\,\sigma_i\right)$  with $\sigma_i\in\ensuremath{\mathbb{N}}_+\cup\{0\}$ and $\tilde{f}_i\in \mathcal{W}\left(N-2k,\,\tilde{\sigma}_i\right)$ with $\sigma_i,\,\tilde{\sigma}_i\in\ensuremath{\mathbb{N}}_+\cup\{0\}$. $Q_{2k-2}$ and $\tilde{Q}_{2k-2}$ are polynomials of degree at most $(2k-2)$  containing only even powers. For $j=\lfloor m/2\rfloor$ and $r=|x|$, we define
\begin{align}
      \Psi_{b,m}(x)=\Psi_{b,m}(r)=\begin{cases}
    \mathrm{e}^{-br},&\text{ if }m=1,\\
    (1+br)\mathrm{e}^{-br},&\text{ if }m=2,\\
    \mathcal{I}^{j}\left[\mathrm{e}^{-bt}\right](r)-Q_{2j-2}(r),&\text{ if } m=2j+1\quad\text{ for }j\geq1,\\
    \mathcal{I}^{j-1}\left[(1+bt)\mathrm{e}^{-bt}\right](r)-\tilde{Q}_{2j-4}(r),&\text{ if } m=2j\quad\text{ for }j\geq2,
    \end{cases}
    \label{def of Psi}
\end{align}
It follows that $\mathcal{R}_{m-1}\Psi_{b,m}=\mathrm{e}^{-b|x|}$ if $m$ is odd and $\mathcal{R}_{m-2}\Psi_{b,m}=(1+b|x|)\mathrm{e}^{-b|x|}$ if $m$ is even. Then we have
\begin{align*}
    \frac{\int_{\ensuremath{\mathbb{R}}^N}\left|\mathcal{R}_m\Psi_{b,m}\right|^2\mathrm{\,d} x\int_{\ensuremath{\mathbb{R}}^N}\left|\mathcal{R}_{m-1}\Psi_{b,m}\right|^2\mathrm{\,d} x}{\left(\int_{\ensuremath{\mathbb{R}}^N}|x|^{-1}\left|\mathcal{R}_{m-1}\Psi_{b,m}\right|^2\mathrm{\,d} x\right)^2}=\bar\mu_m(N)=\begin{cases}
    \frac{(N-1)^2}{4},\qquad\text{if $m$ is odd,}\\
    \frac{(N+1)^2}{4},\qquad\text{if $m$ is even.}
    \end{cases}
\end{align*}

Next we verify that $\Psi_{b,m}$ can be approximated by smooth, compactly supported functions. Note that $\Psi_{b,m}\in C^\infty\left(\ensuremath{\mathbb{R}}^N\backslash\{0\}\right)$ is not smooth at the origin. We first derive the asymptotic estimate as $|x|\to+\infty$ and then deal with the smoothness at the origin. By \eqref{def of Psi}, there exists some $\sigma\in\ensuremath{\mathbb{N}}_+$ depending on $m$ such that $\Psi_{b,m}(r)\in \mathcal{W}(N-m+1,\,\sigma).$ For any multi-index $\beta$, by \eqref{derivative of radial function}, as $|x|\to+\infty$, there holds
\begin{equation}
    D^\beta\Psi_{b,m}(x)=
    O\left(|x|^{m-1-N-|\beta|}\ln^{\sigma}|x|\right).
    \label{estimate of Psi}
\end{equation}
Let $\bar u_L=\eta_L\Psi_{b,m}$ for $\eta_L$ defined by \eqref{def of eta L} and $L>0$. We will prove the following facts:
\begin{align}
\label{approximate Psi}
    \begin{cases}
    \lim\limits_{L\to+\infty}\int_{\ensuremath{\mathbb{R}}^N}\left|\mathcal{R}_{m}\left(\bar u_L-\Psi_{b,m}\right)\right|^2\mathrm{\,d} x=0,\\
     \lim\limits_{L\to+\infty}\int_{\ensuremath{\mathbb{R}}^N}\left|\mathcal{R}_{m-1}\left(\bar u_L-\Psi_{b,m}\right)\right|^2\mathrm{\,d} x=0,\\
      \lim\limits_{L\to+\infty}\int_{\ensuremath{\mathbb{R}}^N}|x|^{-1}\left|\mathcal{R}_{m-1}\left(\bar u_L-\Psi_{b,m}\right)\right|^2\mathrm{\,d} x=0.
    \end{cases}
\end{align}
Recall that for radially symmetric function $u$, we have $\left|\mathcal{R}_1u\right|=|\nabla u|$ and $\mathcal{R}_2u=\Delta u$. Then we deduce that
\begin{align*}
    \left|\mathcal{R}_{m}\left(\bar u_L-\Psi_{b,m}\right)\right|=\left|{P}_{m}\left(\bar u_L-\Psi_{b,m}\right)\right|=\left|\left(\eta_L-1\right)P_m\Psi_{b,m}+\bar\rho_L\right|,
\end{align*}
where $\bar\rho_L$ is a linear combination of products in the form $D^{\beta_1}\eta_L\cdot D^{\beta_2}\Psi_{b,m}$ for $|\beta_1|+|\beta_2|=m$ and $|\beta_1|\geq1$. From  \eqref{estimate of eta} and \eqref{estimate of Psi}, as $L\to+\infty$, we have
\begin{align*}
    \int_{\ensuremath{\mathbb{R}}^N}\left|\mathcal{R}_{m}\left(\bar u_L-\Psi_{b,m}\right)\right|^2\mathrm{\,d} x\leq C\int_L^{+\infty}t^{-2N-2}\ln^{2\sigma}(t)\cdot t^{N-1}\mathrm{\,d} t\to0.
\end{align*}
Similarly, we have
\begin{align*}
    \int_{\ensuremath{\mathbb{R}}^N}\left|\mathcal{R}_{m-1}\left(\bar u_L-\Psi_{b,m}\right)\right|^2\mathrm{\,d} x&\leq C\int_L^{+\infty}t^{-2N}\ln^{2\sigma}(t)\cdot t^{N-1}\mathrm{\,d} t\to0.\\
    \int_{\ensuremath{\mathbb{R}}^N}|x|^{-1}\left|\mathcal{R}_{m-1}\left(\bar u_L-\Psi_{b,m}\right)\right|^2\mathrm{\,d} x&\leq C\int_L^{+\infty}t^{-2N-1}\ln^{2\sigma}(t)\cdot t^{N-1}\mathrm{\,d} t\to0.
\end{align*}
Then we obtain \eqref{approximate Psi}. Now, we will analyze the behavior of $\Psi_{b,m}$ near the origin. Since $Q_{2j-2}(r)$ and $\tilde{Q}_{2j-4}(r)$ are  polynomials containing only even powers, they are smooth at the origin. Note that $g_1(t)=\mathrm{e}^{-bt}$ and $g_2(t)=(1+bt)\mathrm{e}^{-bt}$ possess power series expansions in some neighborhood of the origin. Then, in some punctured neighborhood $B_\delta(0)\backslash\{0\}$ of the origin, for $j=\lfloor m/2\rfloor$, we have
\begin{align*}
    \begin{cases}
    \mathcal{I}^j\left[\mathrm{e}^{-bt}\right](r)=\sum\limits_{l=2j}\limits^{+\infty}a_l r^{l},\qquad&\text{if $m=2j+1$ for $j\geq1$},\\
    \mathcal{I}^{j-1}\left[(1+bt)\mathrm{e}^{-bt}\right](r)=\bar a_{2j-2}r^{2j-2}+\sum\limits_{l=2j}\limits^{+\infty}\bar a_l r^{l},&\text{if $m=2j$ for $j\geq2$}.
    \end{cases}
\end{align*}
Consequently, for any multi-index $\beta$ with $|\beta|\leq m$, we have that $D^\beta \left(\mathcal{I}^j\left[\mathrm{e}^{-bt}\right]\right)$ is bounded in $B_\delta(0)\backslash\{0\}$ when $m$ is odd, and $D^\beta \left(\mathcal{I}^{j-1}\left[(1+bt)\mathrm{e}^{-bt}\right]\right)$ is bounded in $B_\delta(0)\backslash\{0\}$ when $m$ is even. Hence, for any $R>0$ and $\phi\in C^\infty_0\left(B_R(0)\right)$, there holds:
\begin{align*}
    \int_{B_R(0)}\Psi_{b,m}D^\beta\phi\mathrm{\,d} x=(-1)^{|\beta|}\int_{B_R(0)}\phi\cdot D^\beta\Psi_{b,m}\mathrm{\,d} x\qquad\text{for }|\beta|\leq m.
\end{align*}
That is, all weak derivatives of $\Psi_{b,m}$ up to order $m$ exist and are bounded on any ball $B_R(0)$. Then we have $\Psi_{b,m}$ belongs to the Sobolev space $W^{m,\infty}_{\text{loc}}\left(\ensuremath{\mathbb{R}}^N\right)$. It follows that $\bar u_L=\eta_L\Psi_{b,m}\in W^{m,\infty}_{\text{loc}}\left(\ensuremath{\mathbb{R}}^N\right)$ and we denote its mollification by $\left(\bar u_L\right)_{\varepsilon}$ for small $\varepsilon>0$ (see \cite[Appendix C]{MR2597943}). Then $\left(\bar u_L\right)_{\varepsilon}$ is radially symmetric and belongs to $C^\infty_0\left(B_{4L}(0)\right)$. For fixed $L>0$, by Theorem 1 in \cite[Section 5.3.1]{MR2597943}, we have
\begin{align}
\label{mollification 1}
    \lim_{\varepsilon\to0}\int_{\ensuremath{\mathbb{R}}^N}\left|\mathcal{R}_m\left(\left(\bar u_L\right)_{\varepsilon}-\bar u_L\right)\right|^2\mathrm{\,d} x&=\lim_{\varepsilon\to0}\int_{\ensuremath{\mathbb{R}}^N}\left|{P}_m\left(\left(\bar u_L\right)_{\varepsilon}-\bar u_L\right)\right|^2\mathrm{\,d} x\nonumber\\
    &=\lim_{\varepsilon\to0}\int_{\ensuremath{\mathbb{R}}^N}\left|\left({P}_m\bar u_L\right)_{\varepsilon}-{P}_m\bar u_L\right|^2\mathrm{\,d} x\nonumber\\
    &=0.
    \end{align}
    \begin{align}
    \lim_{\varepsilon\to0}\int_{\ensuremath{\mathbb{R}}^N}\left|\mathcal{R}_{m-1}\left(\left(\bar u_L\right)_{\varepsilon}-\bar u_L\right)\right|^2\mathrm{\,d} x&=\lim_{\varepsilon\to0}\int_{\ensuremath{\mathbb{R}}^N}\left|{P}_{m-1}\left(\left(\bar u_L\right)_{\varepsilon}-\bar u_L\right)\right|^2\mathrm{\,d} x\nonumber\\
    &=\lim_{\varepsilon\to0}\int_{\ensuremath{\mathbb{R}}^N}\left|\left({P}_{m-1}\bar u_L\right)_{\varepsilon}-{P}_{m-1}\bar u_L\right|^2\mathrm{\,d} x\nonumber\\
    &=0.
    \label{mollification 2}
\end{align}
Moreover, by Theorem 7 in \cite[Appendix C]{MR2597943}, we have $\left({P}_{m-1}\bar u_L\right)_{\varepsilon}\to{P}_{m-1}\bar u_L$ as $\varepsilon \to0$ uniformly on compact subsets of $B_{4L}(0)$. Then, we obtain that
\begin{align}
\label{mollification 3}
    \int_{\ensuremath{\mathbb{R}}^N}|x|^{-1}\left|\mathcal{R}_{m-1}\left(\left(\bar u_L\right)_{\varepsilon}-\bar u_L\right)\right|^2\mathrm{\,d} x &\leq \|\left({P}_{m-1}\bar u_L\right)_{\varepsilon}-{P}_{m-1}\bar u_L\|_{L^\infty\left(B_{3L}(0)\right)}^2\int_{B_{3L}(0)}|x|^{-1}\mathrm{\,d} x\nonumber\\
    &\to 0\qquad\text{as $\varepsilon\to0$}.
\end{align}
Finally, combining \eqref{approximate Psi}, \eqref{mollification 1}, \eqref{mollification 2} and \eqref{mollification 3}, we conclude that there exists a sequence of functions $\left\{\Psi_{n}\right\}\subset C^{\infty}_{0,\text{rad}}\left(\ensuremath{\mathbb{R}}^N\right)$ such that
\begin{align*}
   \lim_{n\to+\infty}\frac{\int_{\ensuremath{\mathbb{R}}^N}\left|\mathcal{R}_m\Psi_{n}\right|^2\mathrm{\,d} x\int_{\ensuremath{\mathbb{R}}^N}\left|\mathcal{R}_{m-1}\Psi_{n}\right|^2\mathrm{\,d} x}{\left(\int_{\ensuremath{\mathbb{R}}^N}|x|^{-1}\left|\mathcal{R}_{m-1}\Psi_{n}\right|^2\mathrm{\,d} x\right)^2}&= \frac{\int_{\ensuremath{\mathbb{R}}^N}\left|\mathcal{R}_m\Psi_{b,m}\right|^2\mathrm{\,d} x\int_{\ensuremath{\mathbb{R}}^N}\left|\mathcal{R}_{m-1}\Psi_{b,m}\right|^2\mathrm{\,d} x}{\left(\int_{\ensuremath{\mathbb{R}}^N}|x|^{-1}\left|\mathcal{R}_{m-1}\Psi_{b,m}\right|^2\mathrm{\,d} x\right)^2}\\
   &=\bar\mu_m(N).
\end{align*}
That is, $\Psi_{b,m}\in \mathcal{H}^m_3\,(b>0)$ are extremal functions of \eqref{equation m order HyUP}.
