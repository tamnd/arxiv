<!-- definition-1: body_md -->
Let $p>1$.
A function $u \in W_{loc}^{1,1}(\Omega)\cap L_{loc}^p(\Omega)$ is called a weak solution of (\ref{Mean Curvature Equation}) if
\begin{align}\label{mean curvature weak equation}
\int_\Omega \frac{\nabla u \cdot \nabla \varphi}{\sqrt{1+|\nabla u|^2}}
= \int_\Omega |u|^{p-1}u \varphi
\qquad \forall \ \varphi \in C_c^\infty(\Omega).
\end{align}
                                                                                                     
                                                           
A stable solution is a weak solution satisfying the following stability inequality
\begin{align}\label{stable}
\int_\Omega p|u|^{p-1}\xi^2
\leq
\int_\Omega
\left(
\frac{|\nabla \xi|^2}{\sqrt{1+|\nabla u|^2}} - \frac{(\nabla u \cdot \nabla \xi)^2}{(1+|\nabla u|^2)^{3/2}}
\right)
\quad \forall \ \xi \in C_c^\infty(\Omega).
\end{align}
<!-- end definition-1 -->

\end{align*}

<!-- lemma-1: body_md -->
\label{Lemma main gradient est}
Let $p>1$ and
\begin{align*}
                                      
a_+(p) := -1+2p+2\sqrt{p(p-1)}.
\end{align*}
Suppose $u\in C^2(\Omega)$ is a stable solution of (\ref{Mean Curvature Equation}).
Then for every
\begin{align*}
a\in\left[1, a_+(p)\right),
\end{align*}
there exists an integer $m = m(p,a)>0$ and a constant $C=C(n,p,a)>0$ such that
\begin{align*}
\int_\Omega |u|^{p+a} \phi^{2m}
+ \int_\Omega |u|^{a-1} \frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}}\phi^{2m}
\leq C \int_\Omega |\nabla\phi|^{\frac{2(p+a)}{p-1}}
\end{align*}
for all $\phi\in C_c^\infty(\Omega)$ with $0\leq \phi\leq 1$.
<!-- end lemma-1 -->
<!-- remark-1: body_md -->
Our gradient estimate differs structurally from the corresponding semilinear inequality \cite[Proposition 1]{Farina JMPA 2007}.
While Farina's estimate contains the second-order quantity $|\Delta\psi|$ on the right-hand side,
ours involves only the first-order term $|\nabla\phi|$.
In Euclidean space this difference is immaterial for the usual radial cut-offs.
On a Riemannian manifold, however, one naturally takes $\phi=\varphi(d(\,\cdot\,,o)/R)$ where $d$ is the distance function.
Controlling $\Delta\phi$ requires a Laplacian comparison estimate for the distance function, which in turn requires suitable geometric assumptions such as a lower Ricci-curvature bound.
In contrast, our estimation only requires an estimate of $|\nabla\phi|$, and when extended to geometric setting, no additional curvature assumptions are needed.
<!-- end remark-1 -->
<!-- proof-1: body_md -->
Let $\eta \in C_c^{\infty}(\Omega)$ be a nonnegative cut-off function.
Since $a\ge1$, the functions $t\mapsto |t|^{a-1}t$ belongs to $C^1(\mathbb R)$.
Hence
\begin{align*}
\varphi = |u|^{a-1} u \eta^2 \in C_c^1(\Omega).
\end{align*}
By a standard density argument,
the weak formulation (\ref{mean curvature weak equation}) remains valid for $C_c^1$ test functions.
Therefore substituting $\varphi$ and its gradient
\begin{align*}
\nabla \varphi = a |u|^{a-1} \eta^2 \nabla u + 2|u|^{a-1}u \eta\nabla \eta
\end{align*}
into (\ref{mean curvature weak equation}) gives
\begin{align}\label{weak eq est 1}
a\int_\Omega \frac{|u|^{a-1}|\nabla u|^2\eta^2}{\sqrt{1+|\nabla u|^2}}
+ 2\int_\Omega |u|^{a-1}u \frac{\nabla u \cdot \nabla\eta}{\sqrt{1+|\nabla u|^2}}\eta
= \int_\Omega |u|^{p+a} \eta^2.
\end{align}

By the same density argument, the function
\begin{align*}
\xi = |u|^{\frac{a-1}{2}} u \eta
\end{align*}
is admissible in the stability inequality (\ref{stable}).
We compute
\begin{align*}
\nabla \xi
= \frac{1+a}{2} |u|^{\frac{a-1}{2}} \eta \nabla u + |u|^{\frac{a-1}{2}} u \nabla\eta,
\end{align*}
\begin{align*}
|\nabla \xi|^2
= \left(\frac{1+a}{2}\right)^2 |u|^{a-1} |\nabla u|^2 \eta^2
+ |u|^{1+a} |\nabla\eta|^2
+ (a+1)|u|^{a-1}u\eta (\nabla u\cdot\nabla\eta),
\end{align*}
and
\begin{align*}
\left(\nabla u \cdot \nabla \xi\right)^2
= \left(\frac{1+a}{2}\right)^2 |u|^{a-1} |\nabla u|^4 \eta^2
+ |u|^{1+a} \left(\nabla u \cdot \nabla\eta\right)^2
+ (a+1) |u|^{a-1}u |\nabla u|^2 \eta (\nabla u\cdot \nabla\eta)
\end{align*}
Inserting these into (\ref{stable}) yields
\begin{align}\label{7e71ce28a}
p \int_\Omega|u|^{p+a} \eta^2
&\leq \left(\frac{1+a}{2}\right)^2 \int_\Omega \frac{|u|^{a-1} |\nabla u|^2 \eta^2}{(1+|\nabla u|^2)^{3/2}}
+ \int_\Omega\frac{|u|^{1+a}|\nabla\eta|^2}{\sqrt{1+|\nabla u|^2}} \notag \\
&\qquad - \int_\Omega\frac{|u|^{1+a}\left(\nabla u \cdot \nabla\eta\right)^2}{(1+|\nabla u|^2)^{3/2}}
+(a+1)\int_\Omega\frac{|u|^{a-1}u \left(\nabla u \cdot \nabla\eta\right)\eta}{(1+|\nabla u|^2)^{3/2}}
\end{align}
Using
\[
\frac{1}{(1+|\nabla u|^2)^{3/2}}
\le
\frac{1}{\sqrt{1+|\nabla u|^2}},
\]
we enlarge the first term on the right-hand side of
(\ref{7e71ce28a}) by
\[
\left(\frac{1+a}{2}\right)^2
\int_\Omega
\frac{|u|^{a-1}|\nabla u|^2\eta^2}
{\sqrt{1+|\nabla u|^2}}.
\]
Next, we add the term
\begin{align*}
\varepsilon \left(\frac{a+1}{2}\right)^2
\int_\Omega \frac{|u|^{a-1}|\nabla u|^2\eta^2}{\sqrt{1+|\nabla u|^2}}
\end{align*}
to both sides, where $\varepsilon >0$ is to be chosen later.
We obtain
\begin{align}\label{stable est 1}
&p \int_\Omega|u|^{p+a} \eta^2
+ \varepsilon \left(\frac{1+a}{2}\right)^2 \int_\Omega\frac{|u|^{a-1}|\nabla u|^2\eta^2}{\sqrt{1+|\nabla u|^2}} \notag\\
&\leq (1+\varepsilon)\left(\frac{1+a}{2}\right)^2 \int_\Omega\frac{|u|^{a-1} |\nabla u|^2 \eta^2}{\sqrt{1+|\nabla u|^2}}
+ \int_\Omega\frac{|u|^{1+a}|\nabla\eta|^2}{\sqrt{1+|\nabla u|^2}} \notag\\
&\qquad - \int_\Omega\frac{|u|^{1+a}\left(\nabla u \cdot \nabla\eta\right)^2}{(1+|\nabla u|^2)^{3/2}}
+(a+1)\int_\Omega |u|^{a-1}u \eta\frac{\left(\nabla u \cdot \nabla\eta\right)}{(1+|\nabla u|^2)^{3/2}}
\end{align}
Using (\ref{weak eq est 1}) to eliminate the first term on the right-hand side of (\ref{stable est 1}), we obtain
\begin{align}\label{071a9e4d4}
&\left(p - \frac{1+\varepsilon}{a} \left(\frac{1+a}{2}\right)^2\right) \int_\Omega |u|^{p+a}\eta^2
+ \varepsilon \left(\frac{1+a}{2}\right)^2 \int_\Omega |u|^{a-1} \frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}} \eta^2 \notag\\
&\leq \int_\Omega \frac{|u|^{1+a}|\nabla\eta|^2}{\sqrt{1+|\nabla u|^2}}
- \int_\Omega \frac{|u|^{1+a}\left(\nabla u \cdot \nabla\eta\right)^2}{(1+|\nabla u|^2)^{3/2}} \notag\\
&\qquad +\int_\Omega \left(\frac{1+a}{(1+|\nabla u|^2)^{3/2}}
- \frac{2(1+\varepsilon)\left(\frac{1+a}{2}\right)^2}{a\sqrt{1+|\nabla u|^2}}\right)|u|^{a-1} u (\nabla u \cdot \nabla\eta)\eta
\end{align}
Under the condition $a>0$, we have
\begin{align*}
p - \frac{1}{a} \left(\frac{1+a}{2}\right)^2>0
&\Longleftrightarrow 4ap >(a+1)^2 \\
&\Longleftrightarrow a^2 + (2-4p)a +1 < 0 \\
&\Longleftrightarrow a \in (a_-(p), a_+(p)),
\end{align*}
where
\begin{align*}
a_-(p) &:= -1+2p-2\sqrt{p(p-1)}.
\end{align*}
Noting $a_-(p) = \left(\sqrt{p} - \sqrt{p-1}\right)^2 <1$,
and recalling that we are working in the sub-interval $a \in [1,a_+(p)) \subset (a_-(p), a_+(p))$,
which means that
\begin{align*}
p - \frac{1}{a} \left(\frac{1+a}{2}\right)^2>0.
\end{align*}
Therefore the coefficient of the first integral on the left of (\ref{071a9e4d4}) is positive for every sufficiently small $\varepsilon>0$.
Fix such a small enough $\varepsilon$.
Consequently,
after absorbing all constants into a generic constant $C=C_{n,p,a}$,
from (\ref{071a9e4d4}) we obtain
\begin{align}\label{est 1}
\int_\Omega |u|^{p+a}\eta^2
+ \int_\Omega |u|^{a-1} \frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}} \eta^2
\leq C_{n,p,a} \int_\Omega \left(\frac{|u|^{1+a}|\nabla\eta|^2}{\sqrt{1+|\nabla u|^2}}
+ \frac{|u|^{a} |\nabla u| |\nabla \eta|\eta}{\sqrt{1+|\nabla u|^2}}\right).
\end{align}
Set $\eta = \phi^m$, where \(m\) is a positive integer to be chosen later.
Then (\ref{est 1}) becomes
\begin{align}\label{est 2}
&\int_\Omega |u|^{p+a}\phi^{2m}
+ \int_\Omega |u|^{a-1} \frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}} \phi^{2m} \notag\\
&\leq C_{n,p,a,m} \int_\Omega \left(\frac{|u|^{1+a}\phi^{2(m-1)}|\nabla\phi|^2}{\sqrt{1+|\nabla u|^2}}
+ \frac{|u|^{a} |\nabla u| \phi^{2m-1}|\nabla\phi|}{\sqrt{1+|\nabla u|^2}}\right).
\end{align}
Apply the following Young's inequality
\begin{align*}
XY \leq \delta X^{\frac{p+a}{1+a}} + C_\delta Y^{\frac{p+a}{p-1}}
\end{align*}
where $\delta$ is a small positive constant chosen later.
Taking
\begin{align*}
X = |u|^{1+a}\phi^{2(m-1)}
\quad \text{and} \quad Y = |\nabla\phi|^2,
\end{align*}
we get
\begin{align*}
\int_\Omega \frac{|u|^{1+a} \phi^{2(m-1)} |\nabla\phi|^2}{\sqrt{1+|\nabla u|^2}}
&\leq \int_\Omega |u|^{1+a}\phi^{2(m-1)}|\nabla\phi|^2 \notag\\
&\leq \delta \int_\Omega |u|^{p+a} \phi^{\frac{2(m-1)(p+a)}{1+a}}
+ C_\delta \int_\Omega |\nabla\phi|^{\frac{2(p+a)}{p-1}}
\end{align*}
Choose integer $m$ so large that $m \geq (p+a)/(p-1)$, which ensures that
\begin{align*}
\frac{2(m-1)(p+a)}{1+a} \geq 2m.
\end{align*}
Since $0 \leq \phi \leq 1$, we deduce
\begin{align}\label{after Youngs eq 1}
\int_\Omega \frac{|u|^{1+a} \phi^{2(m-1)} |\nabla\phi|^2}{\sqrt{1+|\nabla u|^2}}
&\leq \delta \int_\Omega |u|^{p+a} \phi^{2m} + C_\delta \int_\Omega |\nabla\phi|^{\frac{2(p+a)}{p-1}}
\end{align}
Also using the three-factor Young's inequality
\begin{align*}
XYZ \leq \delta X^2 + \delta Y^\frac{2(p+a)}{1+a} + C_\delta Z^\frac{2(p+a)}{p-1},
\end{align*}
and taking
\begin{align*}
X=\frac{|u|^{\frac{a-1}{2}}|\nabla u|\phi^{m}}{(1+|\nabla u|^2)^{1/4}},
\quad Y=|u|^{\frac{a+1}{2}} \phi^{m-1}
\quad\text{and}\quad Z=|\nabla\phi|,
\end{align*}
then
\begin{align}\label{after Youngs eq 2}
&\int_\Omega \frac{|u|^{a} |\nabla u|\phi^{2m-1} |\nabla\phi|}{\sqrt{1+|\nabla u|^2}} \notag\\
&\leq \int_\Omega \frac{|u|^{a} |\nabla u|\phi^{2m-1} |\nabla\phi|}{(1+|\nabla u|^2)^{1/4}} \notag\\
&\leq \delta \int_\Omega |u|^{a-1} \frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}} \phi^{2m}
+\delta \int_\Omega |u|^{p+a} \phi^{\frac{2(m-1)(p+a)}{1+a}}
+C_\delta \int_\Omega |\nabla\phi|^{\frac{2(p+a)}{p-1}} \notag\\
&\leq \delta \int_\Omega |u|^{a-1} \frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}} \phi^{2m}
+ \delta\int_\Omega |u|^{p+a} \phi^{2m}
+C_\delta \int_\Omega |\nabla\phi|^{\frac{2(p+a)}{p-1}}
\end{align}
Let $C_0=C_{n,p,a,m}$ denote the constant in (\ref{est 2}).
Choose $\delta>0$ sufficiently small so that
\[
3C_0\delta<1.
\]
Inserting (\ref{after Youngs eq 1}) and (\ref{after Youngs eq 2}) into (\ref{est 2}) yields
\begin{align*}
\int_\Omega |u|^{p+a} \phi^{2m}
+ \int_\Omega |u|^{a-1} \frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}} \phi^{2m}
\leq C_{n,p,a} \int_\Omega |\nabla\phi|^{\frac{2(p+a)}{p-1}},
\end{align*}
which is the desired estimate.
This proves Lemma \ref{Lemma main gradient est}.

\label{Lemma main gradient est Dirichlet zero boundary}
Let $p>1$ and let $\Omega \subset \mathbb{R}^n$ be a domain with smooth boundary.
Suppose $u\in C^2(\overline{\Omega})$ is a solution of
\begin{align*}
\left\{
\begin{array}{ll}

&\qquad \text{in}\ \Omega, \\
u=0 &\qquad \text{on}\ \partial \Omega
\end{array}
\right.
\end{align*}
that is stable outside a compact set $\mathcal{K}\subset \Omega$.
Then, for every
\begin{align*}
a\in[1,a_+(p)),
\end{align*}
there exist an integer $m=m(p,a)>0$ and a constant
$C=C_{n,p,a}>0$ such that
\begin{align*}
\int_\Omega |u|^{p+a}\phi^{2m}
+
\int_\Omega
|u|^{a-1}
\frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}}
\phi^{2m}
\le

<!-- proof-2: body_md -->
The proof is the same as that of Lemma \ref{Lemma main gradient est}, except that the cut-off function $\phi$ is not required to vanish near $\partial\Omega$. The homogeneous Dirichlet condition allows the test functions below to be approximated by functions in $C_c^\infty(\Omega)$.
Indeed, since $u\in C^2(\overline{\Omega})$,
$\phi\in C_c^\infty(\mathbb R^n \setminus \mathcal{K})$,
and $a\ge1$,
both functions
\begin{align*}
|u|^{a-1}u\phi^{2m}
\quad\text{and}\quad
|u|^{\frac{a-1}{2}}u\phi^m
\end{align*}
belong to $H^1(\Omega)$. <!-- lemma-4: body_md -->
\label{Lemma conservation law}
Let $f\in C^1(\mathbb{R})$ and let $u\in C^2(\Omega)$ be a solution of
\begin{align}\label{Mean Curvature Equation f(u)}
-\operatorname{div} \left(\frac{\nabla u}{\sqrt{1+|\nabla u|^2}}\right) = f(u)
\qquad \text{in}\ \Omega.
\end{align}
Let
\begin{align*}
F(t) = \int_0^t f(s) ds.
\end{align*}
Define the tensor field $T=(T_{ij})$ by
\begin{align*}
T_{ij} = \frac{u_i u_j}{\sqrt{1+|\nabla u|^2}} - \left(\sqrt{1+|\nabla u|^2} - 1 - F(u)\right)\delta_{ij}.
\end{align*}
Then
\begin{align*}
\operatorname{div} T = 0.
\end{align*}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
It suffices to show that
\begin{align*}
\sum_{j=1}^n \partial_j T_{ij} = 0 \qquad \text{for}\ i=1,2,\cdots, n.
\end{align*}
By the equation (\ref{Mean Curvature Equation f(u)}), we have
\begin{align*}
\sum_j \partial_j \left(\frac{u_i u_j}{\sqrt{1+|\nabla u|^2}}\right)
&= \sum_j \frac{u_{ij} u_j}{\sqrt{1+|\nabla u|^2}}
+ u_i \sum_j \partial_j\left(\frac{u_j}{\sqrt{1+|\nabla u|^2}}\right) \notag\\
&=\frac{\partial_i \left(|\nabla u|^2\right)}{2\sqrt{1+|\nabla u|^2}}
-u_i f(u)
\end{align*}
On the other hand, we have
\begin{align*}
\sum_{j=1}^n\partial_j
\left[
\left(
\sqrt{1+|\nabla u|^2}-1-F(u)
\right)\delta_{ij}
\right]
&=
\partial_i
\left(
\sqrt{1+|\nabla u|^2}-1-F(u)
\right)\\
&=
\frac{\partial_i \left(|\nabla u|^2\right)}
{2\sqrt{1+|\nabla u|^2}}
-f(u)u_i,
\end{align*}
where we have used $F'=f$.
Subtracting the two identities gives
\begin{align*}
\sum_j \partial_j T_{ij}
= \sum_j \partial_j \left(\frac{u_i u_j}{\sqrt{1+|\nabla u|^2}}\right)
-\sum_j \partial_j \left(\left(\sqrt{1+|\nabla u|^2} -1 -F(u)\right) \delta_{ij}\right) = 0.
\end{align*}
Since $i=1,\dots,n$ was arbitrary, we conclude that
\begin{align*}
\operatorname{div}T=0
\qquad\text{in}\ \Omega.
\end{align*}

\label{Lemma poho}
Let $p>1$, and $\phi\in C_c^\infty(\Omega)$.
Suppose $u \in C^2(\Omega)$ is a solution of (\ref{Mean Curvature Equation}).
Then
\begin{align*}
                             
&\int_\Omega \frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}} \phi
- n\int_\Omega \left(\sqrt{1+|\nabla u|^2}-1\right)\phi \notag\\
& = - \frac{n}{p+1} \int_\Omega |u|^{p+1}\phi
- \int_\Omega \frac{(x\cdot \nabla u)(\nabla u \cdot \nabla\phi)}{\sqrt{1+|\nabla u|^2}} \notag\\
&\qquad + \int_\Omega\left(\sqrt{1+|\nabla u|^2}-1\right)(x\cdot \nabla\phi)
- \frac{1}{p+1} \int_\Omega |u|^{p+1}(x\cdot \nabla\phi)
\end{align*}

Let
\begin{align}\label{def f F}
f(t) = |t|^{p-1}t \quad \text{and} \quad F(t) = \frac{|t|^{p+1}}{p+1}.
\end{align}
Then from Lemma \ref{Lemma conservation law}, we have $\operatorname{div}T=0$.
Hence
\begin{align*}
\operatorname{div}(\phi Tx)
&= (Tx)\cdot\nabla\phi + \phi ( x\cdot \operatorname{div}T) + \phi \operatorname{tr}T \\
&= (Tx)\cdot\nabla\phi + \phi \operatorname{tr}T
\end{align*}
Integrating over $\Omega$ and using $\phi\in C_c^\infty(\Omega)$ gives

\int_\Omega
(Tx)\cdot\nabla\phi
+
\int_\Omega
\phi\,\operatorname{tr}T
=0.
\end{align}

\begin{align*}
Tx = \frac{(\nabla u\cdot x)\nabla u}{\sqrt{1+|\nabla u|^2}} - \left( \sqrt{1+|\nabla u|^2} -1 - \frac{|u|^{p+1}}{p+1} \right) x
\end{align*}
and
\begin{align*}
\operatorname{tr}T

\label{Lemma poho constant}
Let $X \in \mathbb{R}^n$.
Suppose $\Omega\subset \mathbb{R}^n$ is a domain with smooth boundary and $u\in C^2(\overline{\Omega})$ is a solution of (\ref{Mean Curvature Equation}).
Then
\begin{align*}
&\int_{\partial (\Omega\cap B_R)} \left(\sqrt{1+|\nabla u|^2}-1\right)\left(X\cdot \nu\right) \notag\\
&=\frac{1}{p+1} \int_{\partial (\Omega\cap B_R)} |u|^{p+1}\left(X\cdot \nu\right)
+\int_{\partial (\Omega\cap B_R)} \frac{\left(X\cdot \nabla u\right)\left(\nabla u \cdot \nu\right)}{\sqrt{1+|\nabla u|^2}},
\end{align*}
for a.e. $R>0$,
where $\nu$ denotes the outward unit normal to $\partial(\Omega \cap B_R)$.

Let $f$ and $F$ be the same as in (\ref{def f F}).
Then by Lemma \ref{Lemma conservation law} we have $\operatorname{div}T=0$.

\begin{align*}
\operatorname{div}(TX)=0.
\end{align*}

\int_{\partial(\Omega\cap B_R)}

(TX)\cdot\nu = -\left(

\end{align*}

Let $\eta \in C^\infty([0,\infty))$ be a cut-off function satisfying
\begin{align}\label{def eta}
0\leq \eta\leq 1,
\quad \eta(r) = 1 \ \text{for}\ 0\leq r\leq 1,
\quad |\eta'(r)| \leq C \ \text{for}\ 0\leq r\leq 1,
\quad \eta(r) = 0\ \text{for}\ r\geq 2,
\end{align}
For $R>0$, define
\begin{align*}
\phi_R(x) = \eta\left(\frac{|x|}{R}\right).
\end{align*}
Then
\begin{align*}
\operatorname{supp} \nabla\phi_R \subset B_{2R}\setminus B_{R},
\qquad|\nabla\phi_R(x)| \leq \frac{C}{R}.
                                    
\end{align*}
By Lemma \ref{Lemma main gradient est} with $\phi_R$,
for every
\begin{align*}
a\in\left[1, a_+(p)\right),
\end{align*}
there exists a positive integer $m=m(p,a)$ and a constant $C=C(n,p,a)$ such that
\begin{align*}
\int_{\mathbb R^n}|u|^{p+a}\phi_R^{2m}
+
\int_{\mathbb R^n}
|u|^{a-1}
\frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}}
\phi_R^{2m}
\le
C\int_{\mathbb R^n}|\nabla\phi_R|^{\frac{2(p+a)}{p-1}}.
\end{align*}
Since $\phi_R=1$ in $B_R$, we obtain
\begin{align}\label{08ae787c1}
\int_{B_R} |u|^{p+a}
+ \int_{B_R} |u|^{a-1} \frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}}
\leq C_{n,p,a} R^{n-\frac{2(p+a)}{p-1}}.
\end{align}

Suppose that there exists $a\in\left[1, a_+(p)\right)$ such that
\begin{align}\tag{A1}\label{A1}
\frac{2(p+a)}{p-1} > n.
\end{align}
Then the exponent on the right-hand side of (\ref{08ae787c1}) is negative.
Letting $R\rightarrow \infty$ we obtain
\begin{align*}
\int_{\mathbb{R}^n} |u|^{p+a} = 0,
\end{align*}
and hence $u \equiv 0$.

Therefore, it remains to prove that for every $1 < p < p_{JL}$,
we may choose $a\in\left[1, a_+(p)\right)$ such that (\ref{A1}) is satisfied.
The condition (\ref{A1}) is equivalent to
\begin{align*}
                         
                      
a > \frac{1}{2} (np -2p -n) =: a_0.
\end{align*}
As long as $a_0 < a_+(p)$ holds, the interval $(\max\{1, a_0\}, a_+(p))$ is well-defined and nonempty. Consequently, any choice of $a$ within this interval will simultaneously satisfy both $a \in (1, a_+(p))$ and condition (\ref{A1}).
Thus, it suffices to prove
\begin{align}\tag{A2}\label{A2}
a_0 < a_+(p).
\end{align}
The condition (\ref{A2}) is equivalent to
\begin{align}\tag{A3}\label{A3}
(n-6)p - (n-2) < 4\sqrt{p(p-1)}
\end{align}
We distinguish two cases.
\begin{itemize}
\item
{\bf Case $2\leq n\leq 10$.}
In this range, $p_{JL} = +\infty$,
so it suffices to prove (\ref{A3}) for all $p>1$.
We have
\begin{align*}
(n-6)p - (n-2)
& = (p-1)n -6p+2 \notag\\
&\leq (p-1)10 -6p+2 \notag\\
&= 4p-8.
\end{align*}
We split the proof into two subcases.
\begin{itemize}
\item
If $1<p<2$,
then $4p-8<0$,
and hence $(n-6)p - (n-2)<0$,
whereas $4\sqrt{p(p-1)}>0$.
So (\ref{A3}) holds trivially.
\item
If $p\geq 2$,
then
\begin{align*}
(4p-8)^2 < 16p(p-1)
\end{align*}
because this is equivalent to $p>\frac{4}{3}$.
Since both sides are nonnegative, squaring gives
\begin{align*}
4p-8<4\sqrt{p(p-1)}
\end{align*}
and hence (\ref{A3}) follows.
\end{itemize}
Therefore (\ref{A3}) holds for all $p>1$ when $2\leq n\leq 10$.

\item {Case $n\geq 11$.}
First, we claim that
\begin{align*}
&p_{JL} > \frac{n-2}{n-6}.
\end{align*}
Indeed,
\begin{align*}
p_{JL} - \frac{n-2}{n-6}
&=\frac{(n^2-8n+4+8\sqrt{n-1})(n-6) - (n-2)^2(n-10)}{(n-2)(n-10)(n-6)}\\
&=\frac{8 \left[ n+2+(n-6)\sqrt{n-1} \right]}{(n-2)(n-10)(n-6)} \\
&> 0.
\end{align*}
We split the proof into two subcases.
\begin{itemize}
\item {case $1<p<\frac{n-2}{n-6}$.}
Then the left-hand side of (\ref{A3}) is negative,
while the right-hand side is positive;
therefore, (\ref{A3}) holds immediately.

\item {case $\frac{n-2}{n-6}\leq p< p_{JL}$.}
In this case, both sides of (\ref{A3}) are nonnegative,
so (\ref{A3}) is equivalent to
\begin{align*}
\left((n-6)p - (n-2)\right)^2 < 16 p(p-1)
\end{align*}
or equivalently,
\begin{align}\label{d9323a29c}
(n-2)(n-10)p^2 -2(n^2-8n+4)p +(n-2)^2 <0
\end{align}
A direct computation shows that the discriminant $\Delta$ of the quadratic in (\ref{d9323a29c}) is
\begin{align*}
\Delta = 4(n^2-8n+4)^2-4(n-2)^3(n-10) = 256(n-1)>0
\end{align*}

Hence the quadratic has two distinct real roots,
given by
\begin{align*}
p^- := \frac{n^2-8n+4 - 8\sqrt{n-1}}{(n-2)(n-10)},
\qquad p^+ := p_{JL} = \frac{n^2-8n+4 + 8\sqrt{n-1}}{(n-2)(n-10)}.
\end{align*}
Since the quadratic is negative between its roots,
(\ref{d9323a29c}) holds for
\begin{align*}
p^- < p < p_{JL}.
\end{align*}
It remains to verify that
\begin{align*}
\frac{n-2}{n-6} > p^-
\end{align*}
Indeed,
\begin{align}\label{59f033067}
\frac{n-2}{n-6} - p^-
&= \frac{(n-2)^2(n-10) - (n-6) \left(n^2-8n+4 - 8\sqrt{n-1}\right)}{(n-6)(n-2)(n-10)} \notag\\
& = \frac{8\left[ (n-6)\sqrt{n-1} -(n+2) \right]}{(n-6)(n-2)(n-10)}.
\end{align}
Since
\begin{align*}
(n-6)^2 (n-1) - (n+2)^2 = (n-10)(n-2)^2 > 0,
\end{align*}
for $n\geq 11$,
the numerator on the right-hand side of (\ref{59f033067}) is positive.
Therefore
$\frac{n-2}{n-6} > p^-$.
Hence, for every
\begin{align*}
\frac{n-2}{n-6} \leq p <p_{JL},
\end{align*}
we have $p^- < p < p_{JL}$,
and consequently (\ref{A3}) holds.
\end{itemize}
Combining both subcases,
we have proved that (\ref{A3}) is satisfied for all $1<p<p_{JL}$ when $n\geq 11$.
\end{itemize}

Thus (\ref{A3}) holds for every $1<p<p_{JL}$, and consequently there exists
$a\in[1,a_+(p))$ satisfying (\ref{A1}). Letting $R\to\infty$ in
(\ref{08ae787c1}) then yields $u\equiv0$.
This proves Theorem \ref{Theorem stable}.

Let $\phi \in C_c^\infty(\mathbb{R}^n)$.

which is admissible by a standard density argument,
in the weak equation (\ref{mean curvature weak equation}), we get
\begin{align}\label{b22636d3b}

\end{align}
On the other hand, applying Lemma \ref{Lemma poho} with $\Omega = \mathbb{R}^n$ and the test function $\phi^2$ yields

- \frac{2}{p+1} \int_{\mathbb{R}^n} |u|^{p+1}(x\cdot\nabla\phi) \phi
\end{align}

Next we show that the right-hand side of (\ref{Pohozaev identity with test function}) tends to $0$ by choosing a suitable $\phi=\phi_R$ as $R\rightarrow \infty$.

\end{align*}

\operatorname{supp}\nabla\phi_R
\subset B_{2R}\setminus B_R,
\qquad
|\nabla\phi_R|\leq \frac{C}{R}.

Using the generalized H\"older inequality
\begin{align*}
\int XYZ \leq
\left(\int X^2\right)^\frac{1}{2}
\left(\int Y^{p+1}\right)^\frac{1}{{p+1}}
\left(\int Z^\gamma\right)^\frac{1}{\gamma}
\end{align*}
where
\begin{align*}
\gamma = \frac{2(p+1)}{p-1},
\qquad X = \frac{|\nabla u|}{(1+|\nabla u|^2)^{1/4}},
\qquad Y=|u|
\qquad \text{and}\qquad Z=|\nabla \phi_R|\phi_R,
\end{align*}

&\leq \int_{B_{2R}\setminus B_{R}} \frac{|\nabla u||u| |\nabla\phi_R|\phi_R}{(1+|\nabla u|^2)^{1/4}} \notag\\
&\leq \left(\int_{B_{2R}\setminus B_{R}} \frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}}\right)^\frac{1}{2}
\left(\int_{B_{2R}\setminus B_{R}} |u|^{p+1}\right)^{\frac{1}{p+1}}
\left(\int_{B_{2R}\setminus B_{R}} |\nabla\phi_R|^\gamma\phi_R^\gamma \right)^{\frac{1}{\gamma}}
\end{align*}
For $n>2$, since $p=(n+2)/(n-2)$, we have $\gamma=n$.
                                                                             
Therefore, using (\ref{4daababba}) and $|B_{2R}\setminus B_R|\le CR^n$ we have
\begin{align*}
\left(
\int_{B_{2R}\setminus B_R}
|\nabla\phi_R|^\gamma\phi_R^\gamma
\right)^{1/\gamma}
\le
C R^{\frac n\gamma-1}
\le C,
\end{align*}
Consequently,
\begin{align}\label{921433a99}
\left| \int_{\mathbb{R}^n} \frac{u\phi_R(\nabla u\cdot\nabla\phi_R)} {\sqrt{1+|\nabla u|^2}}\right|
\leq C\left( \int_{B_{2R}\setminus B_R} \frac{|\nabla u|^2} {\sqrt{1+|\nabla u|^2}} \right)^{1/2}
\left( \int_{B_{2R}\setminus B_R}|u|^{p+1} \right)^{\frac1{p+1}}.
\end{align}
Moreover, noting that $x$ is comparable to $R$ in $B_{2R}\setminus B_R$,
we have
\begin{align}\label{7bbdf943b}
\left| \int_{\mathbb{R}^n} \frac{(x\cdot\nabla u)(\nabla u\cdot\nabla\phi_R)\phi_R}{\sqrt{1+|\nabla u|^2}} \right|
\leq C \int_{B_{2R}\setminus B_{R}} \frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}}
\end{align}
Using the elementary inequality
\begin{align}\label{elementary inequality}
\sqrt{1+t}-1\leq \frac{t}{\sqrt{1+t}} \qquad \forall t\geq 0,
\end{align}
we get
\begin{align}\label{00033cd9f}
\left| \int_{\mathbb{R}^n} \left(\sqrt{1+|\nabla u|^2}-1\right)(x\cdot\nabla\phi_R)\phi_R \right|
\leq C \int_{B_{2R}\setminus B_{R}} \frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}}.
\end{align}
Similarly,
\begin{align}\label{b3f00123a}
\left| \int_{\mathbb{R}^n} |u|^{p+1}(x\cdot\nabla\phi_R) \phi_R \right|
\leq C \int_{B_{2R}\setminus B_{R}} |u|^{p+1}.
\end{align}
Since $p = p_S$ and $u$ is stable outside a compact set $\mathcal K\subset\mathbb R^n$,
using Lemma \ref{Lemma integrable}(i) we deduce that $|\nabla u|^2/\sqrt{1+|\nabla u|^2}$ and $|u|^{p+1}$ are integrable near infinity.
Hence the right-hand sides of (\ref{921433a99}) -- (\ref{b3f00123a}) tend to $0$ as $R\rightarrow \infty$.
If we replace $\phi$ in (\ref{Pohozaev identity with test function}) with $\phi_R$,
then the right-hand side of (\ref{Pohozaev identity with test function}) will tend to $0$ as $R \rightarrow \infty$.
Thus, letting $R\rightarrow \infty$ gives
\begin{align}\label{e315d48f6}
\left(1+\frac{n}{p+1}\right) \int_{\mathbb{R}^n} \frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}} = n \int_{\mathbb{R}^n} \left(\sqrt{1+|\nabla u|^2}-1\right)
\end{align}
Define
\begin{align*}
                        
A := 1+\frac{n}{p+1},
\qquad f(t) := (n-A)t + \frac{A}{t} - n, \quad t > 0.
\end{align*}
Then (\ref{e315d48f6}) is equivalent to
\begin{align}\label{d4b6b6633}
\int_{\mathbb{R}^n} f(\sqrt{1+|\nabla u|^2}) = 0
\end{align}
Since $n \ge 3$ and $p = p_S = (n+2)/(n-2)$, we have $A=n/2$.
Consequently,
\begin{align*}
f(t) = \frac{n(t-1)^2}{2t} \geq 0.
\end{align*}
with equality if and only if $t=1$.
Therefore from (\ref{d4b6b6633}) we obtain that
\begin{align*}
\sqrt{1+|\nabla u|^2}=1
\qquad\text{in}\ \mathbb{R}^n,
\end{align*}
and therefore
\begin{align*}
\nabla u\equiv 0\qquad\text{in}\ \mathbb{R}^n.
\end{align*}
It follows that $u$ is constant.
Substituting this into (\ref{Mean Curvature Equation}) gives $|u|^{p-1}u=0$,
and hence $u\equiv0$,
which completes the proof of Theorem \ref{Theorem stable outside compact}.

Let $R>0$.
We apply the Pohozaev-type identity from Lemma \ref{Lemma poho constant} to the domain $\Omega \cap B_R$,
\begin{align}\label{030fb9b7b}
\int_{\partial(\Omega \cap B_R )} \left(\sqrt{1+|\nabla u|^2}-1\right)\left(X\cdot \nu\right)
&=\frac{1}{p+1} \int_{\partial(\Omega \cap B_R )} |u|^{p+1}\left(X\cdot \nu\right) \notag\\
&\qquad +\int_{\partial(\Omega \cap B_R )} \frac{\left(X\cdot \nabla u\right)\left(\nabla u \cdot \nu\right)}{\sqrt{1+|\nabla u|^2}}
\end{align}
Here $\nu$ denotes the outward unit normal to $\partial(\Omega \cap B_R)$.
Now decompose $\partial(\Omega \cap B_R ) = (\partial \Omega \cap B_R) \cup (\Omega \cap \partial B_R)$.
Noting $u=0$ on $\partial \Omega$,
we have
\begin{align*}
\int_{\partial\Omega \cap B_R} |u|^{p+1}\left(X\cdot \nu\right) = 0.
\end{align*}
Moreover,
on $\partial \Omega$,
since $u=0$,
the gradient $\nabla u$ is parallel to the normal $\nu$.
Therefore
\begin{align*}
\int_{\partial\Omega \cap B_R} \frac{\left(X\cdot \nabla u\right)\left(\nabla u \cdot \nu\right)}{\sqrt{1+|\nabla u|^2}}
=\int_{\partial\Omega \cap B_R} \left(X\cdot \nu\right)\frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}}
\end{align*}
Thus, from (\ref{030fb9b7b}), we have
\begin{align*}
&\int_{\partial \Omega \cap B_R} \left(\frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}} -\sqrt{1+|\nabla u|^2} +1\right) \left(X\cdot \nu\right) \notag\\
&= \int_{\Omega \cap \partial B_R} \left(\sqrt{1+|\nabla u|^2}-1\right)\left(X\cdot \nu\right)
-\frac{1}{p+1} \int_{\Omega \cap \partial B_R} |u|^{p+1}\left(X\cdot \nu\right) \notag\\
&\qquad -\int_{\Omega \cap \partial B_R} \frac{\left(X\cdot \nabla u\right)\left(\nabla u \cdot \nu\right)}{\sqrt{1+|\nabla u|^2}}
\end{align*}
Absorbing $X$ and $\nu$ into a constant $C$ and applying elementary inequality (\ref{elementary inequality}) to the right-hand side of the above identity,
we obtain
\begin{align}\label{0ba95f35e}
&\int_{\partial \Omega \cap B_R} \left(\frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}} -\sqrt{1+|\nabla u|^2} +1\right) \left(X\cdot \nu\right) \notag\\
&\leq C \int_{\Omega \cap \partial B_R} \left(\frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}} +|u|^{p+1}\right).
\end{align}
Since $1<p\leq p_S$ and $u$ is stable outside a compact set $\mathcal K\subset \Omega$,
by Lemma \ref{Lemma integrable},
we obtain
$|u|^{p+1}$, $|\nabla u|^2 / (\sqrt{1+|\nabla u|^2)}$ are integrable near infinity.
Consequently, there exists a sequence $R_k\rightarrow \infty$ such that
\begin{align*}
\int_{\Omega \cap \partial B_{R_k}} \left(\frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}} +|u|^{p+1}\right) \rightarrow 0
\end{align*}
Taking $R=R_k$ in (\ref{0ba95f35e}) and letting $k\rightarrow \infty$,
we obtain
\begin{align}\label{6159a508d}
\int_{\partial \Omega} \left(\frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}} -\sqrt{1+|\nabla u|^2} +1\right) \left(X\cdot \nu\right) = 0
\end{align}
Noting $X\cdot \nu\geq 0$ on $\partial \Omega$ and using (\ref{elementary inequality}) again,
we conclude from (\ref{6159a508d}) that
\begin{align}\label{c07f21d6e}
\left(\frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}} -\sqrt{1+|\nabla u|^2} +1\right)\left(X\cdot \nu\right) = 0 \qquad \text{on}\ \partial\Omega
\end{align}
Because $X\cdot\nu\not\equiv 0$ and $\partial\Omega$ is smooth,
there exists a relatively open subset $\Gamma\subset \partial \Omega$ on which $X \cdot \nu>0$.
Then (\ref{c07f21d6e}) implies
\begin{align*}
\frac{|\nabla u|^2}{\sqrt{1+|\nabla u|^2}} -\sqrt{1+|\nabla u|^2} +1 = 0 \qquad \text{on}\ \Gamma,
\end{align*}
which is equivalent to $\nabla u = 0$ on $\Gamma$.
Hence we get both $ u=0$ and $\nabla u = 0$ on $\partial\Omega\cap \Gamma$.
In the following, we show that such an overdetermined boundary condition forces $u\equiv 0$ in $\Omega$.
We use a unique continuation argument.
Rewrite the original equation as a linear elliptic equation
\begin{align}\label{linearized equation}
\operatorname{div}\left(A(x) \nabla u\right) + V(x) u = 0.
\end{align}
where
\begin{align*}
A(x) := \frac{I}{\sqrt{1+|\nabla u|^2}},
\qquad V(x) = |u|^{p-1}.
\end{align*}
Fix a point $y\in \Gamma$.
By reducing $\delta>0$ if necessary, we may assume that
\begin{align*}
\partial\Omega\cap B_\delta(y)\subset\Gamma.
\end{align*}

Choose $x_0\in\Omega\cap B_{\delta/2}(y)$.
Since $\Omega$ is connected, for any fixed $z\in\Omega$, there exists a continuous path $\gamma$ in $\Omega$ joining $x_0$ to $z$.
We may therefore choose a bounded connected open set $D\subset\Omega$ containing both $x_0$ and $z$.
Set
$$
U:=D\cup B_\delta(y).
$$
Then $U$ is a bounded connected open set.
Define the zero extension of $u$ to $U$ by
\begin{align*}
\widetilde{u} =
\left\{
\begin{array}{ll}
u, & x \in U \cap \Omega, \\
0, & x \in U \setminus \Omega.
\end{array}
\right.
\end{align*}

We next show that $\widetilde u$ is $C^2$ across $\Gamma$.
Since $u\in C^2(\overline{\Omega})$ and both $u$ and $\nabla u$ vanish on $\Gamma$,
all first-order tangential derivatives and all second-order derivatives involving at least one tangential direction vanish on $\Gamma$.
To show that the normal second derivative $u_{\nu\nu}$ also vanishes, we expand the original equation.
Using the product rule, the equation can be written as
\begin{align}\label{expanded equation}
-\nabla g \cdot \nabla u - g \Delta u = |u|^{p-1}u,
\qquad \text{where}\ g(x) := \frac{1}{\sqrt{1+|\nabla u|^2}}.
\end{align}
As $x$ approaches the boundary $\Gamma$,
we have $u \to 0$ and $\nabla u \to 0$.
The first term $\nabla g \cdot \nabla u$ vanishes because $\nabla g$ is bounded near $\Gamma$
(as $u$ is $C^2$ and $g$ is a smooth function of the bounded quantity $|\nabla u|$) and $\nabla u \rightarrow 0$.
Meanwhile, $g \to 1$ and $|u|^{p-1}u \to 0$.
Therefore, passing to the limit $x \to \Gamma$ in (\ref{expanded equation}),
the equation reduces to
\begin{align*}
\Delta u = 0 \qquad \text{on}\ \Gamma.
\end{align*}
It follows that $u_{\nu\nu}=0$ on $\Gamma$ and therefore $D^2u=0$ on $\Gamma$.
This ensures that the zero-extension $\widetilde{u}$ is $C^2$ across the boundary $\Gamma$.
Therefore, $\widetilde{u} \in C^2(\overline{U})$, and it satisfies the linear equation
\begin{align*}
                                    
\operatorname{div}\left(\widetilde{A}(x) \nabla \widetilde{u}\right) + \widetilde{V}(x) \widetilde{u} = 0 \qquad \text{in}\ U,
\end{align*}
where $\widetilde{A}(x)$ and $\widetilde{V}(x)$ are defined by replacing $u$ with $\widetilde{u}$ in the definitions of $A(x)$ and $V(x)$.

We now verify the hypotheses of the unique continuation theorem of Garofalo--Lin (Theorem 1.1 in \cite{Garofalo Lin CPAM 1987}) on the connected domain $U$:

\begin{itemize}
\item[1.]
{\bf Uniform ellipticity:}
                
                                                                                                                             
              
Since $\widetilde{u} \in C^2(\overline{U})$,
there exists $M>0$ such that $|\nabla \widetilde{u}| \leq M$.
For every $\xi\in\mathbb{R}^n$,
\begin{align*}
\xi^{T}\widetilde{A}(x)\xi = \frac{|\xi|^2}{\sqrt{1+|\nabla \widetilde{u}|^2}}.
\end{align*}
Therefore
\begin{align*}
\frac{1}{\sqrt{1+M^2}} |\xi|^2 \leq \xi^{T}\widetilde{A}(x)\xi \leq |\xi|^2.
\end{align*}
This shows that $\widetilde{A}$ is uniformly elliptic.

\item[2.]
{\bf Lipschitz continuity of $A$:}
Since $\widetilde{u}\in C^2(\overline{U})$,
we have $\nabla \widetilde{u} \in C^1(\overline{U})$.
The map
\begin{align*}
t \mapsto \frac{1}{\sqrt{1+|t|^2}}
\end{align*}
is smooth.
Consequently, $\widetilde{A}(x) \in C^1(\overline{U})$,
and $\widetilde{A}(x)$ is Lipschitz on the compact set $\overline{U}$.

\item[3.]
{\bf Boundedness of $V$:}
Since $u\in C^2(\overline{U})$,
$V(x)$ is bounded on $\overline{U}$.
The condition (iii) in Garofalo--Lin \cite{Garofalo Lin CPAM 1987} is trivially satisfied for bounded potentials.
\end{itemize}
Therefore, the operator in (\ref{linearized equation}) satisfies the structural assumptions (i), (ii), and (iii) of Theorem 1.1 in \cite{Garofalo Lin CPAM 1987} for $n>2$.
By that theorem, the equation has the strong unique continuation property.
Since $\widetilde{u}\equiv 0$ in the nonempty open set $U \setminus \overline{\Omega}$,
we must have $\widetilde{u}\equiv 0$ in $U$, and hence $u(z) = 0$.
Since $z \in \Omega$ can be chosen arbitrarily,
we obtain $u\equiv 0$ in the whole connected domain $\Omega$.
This completes the proof of Theorem \ref{Theorem epigraph}.
