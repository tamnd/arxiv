<!-- lemma-1: body_md -->
\label{lem: bound number of peelable and addable ribbons}
    Let $\eta\subseteq \lambda$ be two Young diagrams, and let $r\geq 2$. Denote again the diagonal length of $\lambda$ by $\delta = \delta(\lambda)$.
\begin{enumerate}
    \item There are at most $2\delta$ ways to remove an $r$-ribbon from $\eta$.
    \item There are at most $2\delta + r$ ways to add an $r$-ribbon to $\eta$, while staying in $\lambda$.
\end{enumerate}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Each peelable ribbon has either its tail at the top of one of the first $\delta(\eta)$ columns, or its head at the end of one of the first $\delta(\eta)$ rows. Hence there are at most $2\delta(\eta)\leq 2\delta(\lambda)$ such ribbons, proving (a).

Let us now prove (b). There are three different ways to add a ribbon of length $r$. \\
(1) Choose $1\leq i\leq \delta(\eta)$ and place the tail of the ribbon at the end of the $i$-th row. \\
(2) Choose $1\leq j\leq \delta(\eta)$ and place the head of the ribbon at the top of the $j$-th column. \\
(3) Choose a box of the ribbon and place it in position $(\delta(\eta)+1, \delta(\eta)+1)$. 

In all three cases, the chosen box determines the position of the whole ribbon. There are therefore at most $2\delta(\eta) + r\leq 2\delta + r$ such possibilities, which concludes the proof.
<!-- end proof-1 -->
<!-- proposition-1: body_md -->
\label{prop: generalized Bjorner inequality}
Let $\mu \backslash \alpha$ and $\nu\backslash \beta$ be skew Young diagrams. Then $f((\mu\cup \nu) \backslash (\alpha\cup\beta)) \cdot f((\mu\cap\nu)\backslash(\alpha\cap\beta)) \geq f(\mu \backslash \alpha)\cdot f(\nu \backslash \beta)$.
<!-- end proposition-1 -->
<!-- lemma-2: body_md -->
\label{lem: generalized Bjorner inequality rewritten with mu nu lambda R}
    Let $\mu \subseteq \nu \subseteq \lambda \subseteq R$ be four Young diagrams. Then
\begin{equation}
    \frac{f(\lambda \backslash \nu)}{f(\lambda \backslash \mu)}  \leq \frac{f(R \backslash \nu)}{f(R\backslash \mu)}.
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
We have $f((\lambda \cup R)\backslash(\nu\cup \mu)) \cdot f((\lambda \cap R)\backslash(\nu\cap \mu)) \geq f(\lambda \backslash \nu)\cdot f(R \backslash \mu)$ by Proposition~\ref{prop: generalized Bjorner inequality} applied with $ (\nu, \mu, \lambda, R)$ in place of $(\alpha, \beta, \mu, \nu)$.  Since $\mu \subseteq \nu \subseteq \lambda\subseteq R$ by assumption, this simplifies to $f(R\backslash\nu) \cdot f(\lambda \backslash \mu) \geq f(\lambda \backslash \nu)\cdot f(R \backslash \mu)$, which concludes the proof.

<!-- lemma-3: body_md -->
\label{lem: identity Stanley}
    Let $a,b\geq 1$ be integers. Let $R =[a^b]$ be a rectangle, and let $\eta \subseteq R$ be a Young diagram.
    Then
\begin{equation}
    f(R\backslash \eta) = f(R) \frac{d_\eta}{|\eta|!} \prod_{u\in \eta}(b+c(u))(a-c(u)).
\end{equation}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
As in \cite{Stanley2003CharactersRectangularShape}, we let $\tilde{\eta}$ be the diagram $R\backslash \eta$, rotated by $\pi$, and we denote the product of all hook lengths of a diagram $\xi$ by $H_\xi$. Then, combining Equations (4) and (5) from \cite{Stanley2003CharactersRectangularShape} gives $H_\eta H_R =  H_{\tilde{\eta}} \prod_{u\in \eta}(b+c(u))(a-c(u))$.
                 
                                                                              
               
Moreover, we have $d_{R\backslash \eta} = d_{\tilde{\eta}}$ by symmetry. Therefore, by the hook length formula, we have $H_R = |R|!/d_R=1/f(R)$, $H_\eta = |\eta|!/d_\eta$, and $H_{\tilde{\eta}} = (|R|-|\eta|)!/d_{\tilde{\eta}} = (|R|-|\eta|)!/d_{R\backslash \eta}= 1/f(R\backslash \eta)$. The result follows.

<!-- proposition-2: body_md -->
\label{prop: bound on quotients of skew dimensions}
        Let $\mu \subseteq \nu \subseteq \lambda$ be three Young diagrams. Denote $q = |\mu|$, $n=|\lambda|$, $k = |\nu|-|\mu|$, and $p = |\lambda\backslash \nu| = n-k-q$. Then
\begin{equation}
    \frac{d_\mu d_{\lambda \backslash \nu}}{d_\nu d_{\lambda \backslash \mu}} \leq \left( \frac{s+1}{2}\right)^{2k}\frac{p!q!}{(p+k)!(q+k)!} .
\end{equation}
<!-- end proposition-2 -->
<!-- proof-4: body_md -->
Let $R = [a^b]$, where $a$ (resp.\ $b$) is the number of columns (resp.\ rows) of $\lambda$. By Lemmas~\ref{lem: generalized Bjorner inequality rewritten with mu nu lambda R} and~\ref{lem: identity Stanley}, we have 
\begin{equation}
    \frac{(p+k)!}{p!}\frac{d_{\lambda \backslash \nu}}{d_{\lambda \backslash \mu}} =  \frac{f(\lambda \backslash \nu)}{f(\lambda \backslash \mu)}  \leq \frac{f(R \backslash \nu)}{f(R\backslash \mu)} = \frac{d_\nu/(q+k)!}{d_{\mu}/q!} \prod_{u\in \nu\backslash
    \mu}(b+c(u))(a-c(u)).
\end{equation}
Reordering the terms gives 
\begin{equation}\label{eq: bound quotients of skew dimensions with contents}
    \frac{d_\mu d_{\lambda \backslash \nu}}{d_\nu d_{\lambda \backslash \mu}} \leq \frac{p!q!}{(p+k)!(q+k)!} \prod_{u\in \nu\backslash
    \mu}(b+c(u))(a-c(u)).
\end{equation}
But for each box $u \in \nu\backslash\mu$,  setting $L = (a+b)/2=(s+1)/2$ and $x = c(u) +(b-a)/2$, we have 
    $(b+c(u))(a-c(u)) = (L-x)(L+x) \leq L^2 = ((s+1)/2)^2$, so the product is upper bounded by $((s+1)/2)^{2k}$. Plugging this into \eqref{eq: bound quotients of skew dimensions with contents} concludes the proof.

<!-- lemma-4: body_md -->
\label{lem: bound on ribbon covering of the river starting from mu or nu}
  We have
\begin{equation}
    \sum_{\mu, \nu} d_\mu d_{\lambda\backslash \mu} W_{\mu, \nu} \leq  e^{k/(2\delta)}(2\delta)^m d_\lambda \quad \quad \text{ and } \quad \quad  \sum_{\mu, \nu} d_\nu d_{\lambda\backslash \nu} W_{\mu, \nu} \leq (2\delta)^m d_\lambda \, ,
\end{equation}
where the sums are over all $\mu\vdash q$ and $\nu \vdash q+k$ such that $\mu \subseteq \nu \subseteq \lambda$.
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
For each fixed $\mu$, $\sum_\nu W_{\mu,\nu}$ is the number of ways to iteratively add ribbons of length $r_1,\ldots, r_m$ to $\mu$ up to reaching a sub-diagram of $\lambda$ of size $q+k$, which  by Lemma~\ref{lem: bound number of peelable and addable ribbons} is upper bounded by $\prod_{i=1}^m (2\delta + r_i)$. Therefore
\begin{equation}
    \sum_{\mu, \nu} d_\mu d_{\lambda\backslash \mu} W_{\mu, \nu} = \sum_{\mu} d_\mu d_{\lambda\backslash \mu} \sum_\nu W_{\mu, \nu} \leq \left( \prod_{i=1}^m (2\delta + r_i)\right) \sum_{\mu} d_\mu d_{\lambda\backslash \mu}.
\end{equation}
Moreover, $\sum_{\mu} d_\mu d_{\lambda\backslash \mu} = d_\lambda$, and $(2\delta)^{-m}\prod_{i=1}^m (2\delta + r_i) \leq  \prod_{i=1}^m e^{r_i/(2\delta)} = e^{k/(2\delta)}$. This proves the first point. The same arguments prove the second point: we have
\begin{equation}
     \sum_{\mu, \nu} d_\nu d_{\lambda\backslash \nu} W_{\mu, \nu} = \sum_{\nu} d_\nu d_{\lambda\backslash \nu} \sum_\mu W_{\mu, \nu} \leq \left( \prod_{i=1}^m (2\delta)\right) \sum_{\nu} d_\nu d_{\lambda\backslash \nu} =  d_\lambda  (2\delta)^m. \qedhere
\end{equation}

<!-- lemma-5: body_md -->
\label{lem: technical bound p q with factorials}
 We have
 \begin{equation}
      \sqrt{\frac{p!q!}{(p+k)!(q+k)!}} \leq (e/n)^k.
 \end{equation}
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
Rewriting the products, we have $x:= \frac{(p+k)!(q+k)!}{p!q!} = \prod_{j=p+1}^{p+k}j(n+1-j)$. Taking logarithms gives $\ln \sqrt{x} = \sum_{j=p+1}^{p+k}a_j$, where $a_j = \frac{1}{2} \ln j(n+1-j)$ for $1\leq j \leq n$.
But the $a_j$ for $j$ from $p+1$ to $p+k$ are the $k$ maximal terms of the sum $\sum_{j=1}^{n}a_j$. Therefore, lower bounding by the average, we obtain that
\begin{equation}
    \ln \sqrt{x} \geq \frac{k}{n}\sum_{j=1}^n a_j = \frac{k}{n}\ln n!.
\end{equation}
This implies that $\sqrt{1/x} \leq n!^{-k/n}$. Using finally that $n!\geq (n/e)^n$ concludes the proof.

Set $\gamma = (e/2) e^{1/(4\delta)}(1+1/s)$. Then
\begin{equation}
    \frac{M_{p,q}}{d_\lambda} \leq  (\gamma s/n)^k (2\delta)^m.
\end{equation}
<!-- end proposition-3 -->
<!-- proof-7: body_md -->
By definition we have $M_{p,q} = \sum_{\mu, \nu} d_\mu W_{\mu, \nu} d_{\lambda\backslash \nu}$, where the sum is over all $\mu\vdash q$ and $\nu \vdash q+k$ such that $\mu \subseteq \nu \subseteq \lambda$.
Reorganizing this sum, applying Proposition~\ref{prop: bound on quotients of skew dimensions}, and using the Cauchy--Schwarz inequality, we have
\begin{equation}
\begin{split}
    M_{p,q} & = \sum_{\mu, \nu} \sqrt{d_\mu d_{\lambda\backslash \mu} W_{\mu, \nu}} \sqrt{d_\nu d_{\lambda\backslash \nu}W_{\mu, \nu}}   \sqrt{\frac{d_\mu d_{\lambda\backslash \nu}}{d_\nu d_{\lambda\backslash \mu}}} \\

\end{split}
\end{equation}
Applying now Lemmas~\ref{lem: bound on ribbon covering of the river starting from mu or nu} and~\ref{lem: technical bound p q with factorials} and reordering the terms conclude the proof.

<!-- theorem-3: body_md -->
\label{thm: main general form with m}
Let $n\geq 1$, $\lambda\vdash n$ and $\sigma\in {\ensuremath{\mathfrak{S}}} _n$. Then
\begin{equation}\label{eq: main general form with m}
     \left| \chi^\lambda(\sigma) \right| \leq \left( \gamma s/n \right)^{k}(2\delta)^{m},
\end{equation}
where $s$ is the maximal hook length of $\lambda$, $\delta$ is the diagonal length of $\lambda$, $k$ is the support size of $\sigma$, $m$ is the number of non-trivial cycles of $\sigma$, and $\gamma = (e/2) e^{1/(4\delta)}(1+1/s)$. In particular, we have
\begin{equation}
    \left| \chi^\lambda(\sigma) \right| \leq (2\gamma^2 \delta s^2/n^2)^{|\sigma|}.
\end{equation}
<!-- end theorem-3 -->
<!-- proof-8: body_md -->
We may assume that $k\geq 2$, since the bound is trivial for $\sigma = \operatorname{\mathrm{Id}}$. The first bound then follows immediately from \eqref{eq: MN with triangle inequality and min} and Proposition~\ref{prop: bound number river ribbon tableaux directly with specific p and q}. For the second part, observe that $k=|\sigma|+m$, and [$m\leq |\sigma|$ and $2\gamma \delta s/n\geq \delta s/n \geq 1$], so
    \begin{equation}
        \left| \chi^\lambda(\sigma) \right| \leq \left( \gamma s/n \right)^{k}(2\delta)^{m} = \left( \gamma s/n \right)^{|\sigma|}(2\gamma\delta s/n)^{m}\leq (2\gamma^2 \delta s^2/n^2)^{|\sigma|}. \qedhere
    \end{equation}

<!-- proof-9: body_md -->
Our goal is to prove that $\left| \chi^\lambda(\sigma) \right| \leq (5 \delta s^2/n^2)^{|\sigma|}$. Recall that we always have $\left| \chi^\lambda(\sigma) \right|\leq 1$ and $\delta s \geq n$. This implies that the bound is trivial if $s\geq n/5$. We may therefore assume that $s<n/5$. Plugging this into the inequality $\delta s \geq n$ then gives $\delta>5$, so $\delta \geq 6$ since it is an integer. Therefore $s\geq 2\delta - 1 \geq 11$. Using this we obtain a new bound on $2\gamma^2$: we have $2\gamma^2 \leq (1/2)e^{25/12}(12/11)^2 \leq 5$. Applying Theorem~\ref{thm: main general form with m} with this new bound concludes the proof.

<!-- theorem-4: body_md -->
\label{thm: skew diagrams upper bound Dousse Feray conjecture}
Let $C>0$. There exists a constant $K$, depending only on $C$, such that for every (non-empty) $C$-balanced Young diagrams $\lambda$ and $\mu$ such that $\mu \subseteq \lambda$, we have
\begin{equation}
   -K \frac{|\mu|^{3/2}}{|\lambda|^{1/2}}\leq \ln \left( |\mu|! \frac{d_{\lambda\backslash \mu}}{d_\lambda d_\mu} \right) \leq K \frac{|\mu|^{3/2}}{|\lambda|^{1/2}}.
\end{equation}
<!-- end theorem-4 -->
<!-- proof-11: body_md -->
Let $\mu \subseteq \lambda$ be two $C$-balanced Young diagrams. Denote $k = |\mu|$ and $n=|\lambda|$. Applying the triangle inequality and Theorem~\ref{thm: main balanced Moore Russell form} to \cite[Lemma 10]{DousseFeray2019skewdiagramscharacters} (originally from \cite{Stanley2003EnumerationSkewYoungTableaux}), we obtain, setting $B = 5C^2$,
    \begin{equation}
        A_{\lambda, \mu} \leq \sum_{\sigma\in{\ensuremath{\mathfrak{S}}} _k} | \chi^\lambda(\sigma_+) | | \chi^\mu(\sigma) | \leq \sum_{\sigma\in{\ensuremath{\mathfrak{S}}} _k} \left( \frac{B^2}{\sqrt{nk}}\right)^{|\sigma|},
    \end{equation}
where $\sigma_+$ is the extension of $\sigma$ to ${\ensuremath{\mathfrak{S}}} _n$ by fixing $\left\{  k+1, \ldots, n\right\} $. Decomposing according to the value of $|\sigma|$ and applying \cite[Lemma 13]{DousseFeray2019skewdiagramscharacters} (originally from \cite{FeraySniady2011}) then gives
\begin{equation}
    A_{\lambda, \mu} \leq \sum_{i=0}^{k-1} \frac{k^{2i}}{i!} \left( \frac{B^2}{\sqrt{nk}}\right)^{i} \leq \sum_{i=0}^{\infty} \frac{(B^2k^{3/2}/n^{1/2})^i}{i!} = \exp\left( B^2\frac{k^{3/2}}{n^{1/2}} \right). \qedhere
\end{equation}

<!-- lemma-6: body_md -->
\label{lem: lower bound Dousse Feray k order n}
    Let $C>0$. There exists a constant $K_1$, depending only on $C$, such that the following holds. Let $\mu \subseteq \lambda$ be two (non-empty) $C$-balanced Young diagrams such that $|\mu| \geq |\lambda|/C^2$. Then $\ln A_{\lambda, \mu}  \geq - K_1 |\mu|^{3/2}/|\lambda|^{1/2}$.
<!-- end lemma-6 -->
<!-- proof-12: body_md -->
Write $n=|\lambda|$ and $k=|\mu|$. By the Naruse hook length formula and since $\mu\in {\ensuremath{\mathcal E}} (\lambda, \mu)$, we have $\frac{d_{\lambda\backslash\mu}}{d_\lambda} \geq \frac{H(\mu,\mu)}{n^{\downarrow k}}$. It therefore follows from the standard hook length formula and using that $d_\mu^2 \leq \sum_{\nu\vdash k} d_\nu^2 = k!$,  that
\begin{equation}
    A_{\lambda, \mu} \geq \frac{k!}{d_\mu} \frac{H(\mu,\mu)}{n^{\downarrow k}}  = \frac{k!}{d_\mu^2} \frac{k!}{n^{\downarrow k}} \geq \frac{(k/e)^k}{n^k}\geq \left( eC^2\right)^{-k} \geq \left( eC^2\right)^{-Ck^{3/2}/n^{1/2}}.
\end{equation}
This proves the desired result, with $K_1 = C\ln(eC^2)$.

<!-- proposition-4: body_md -->
\label{prop: marked injection Naruse}
    Let $\mu\subseteq\lambda$ be two Young diagrams. Let $M \geq \max(\mu_1, \mu'_1)$ be an integer and set $W = |\lambda^{>M}|$. Assume that $[M^M]\subseteq \lambda$ and $W\geq |\mu|$. Then
    \begin{equation}
        S(\lambda, \mu) \geq d_\mu \binom{W}{|\mu|}.
    \end{equation}
<!-- end proposition-4 -->
<!-- proof-13: body_md -->
Let $k= |\mu|$ and $\delta = \delta(\lambda)$. For $0\leq i \leq \delta - M$, we set $W_i = W-|\lambda^{>M+i}|$: this is the number of boxes on the first $i$ hooks of $\lambda^{>M}$. Let us prove that a set of size $d_\mu \binom{W}{k}$ injects into a set of size $S(\lambda, \mu)$. 

On the one hand, consider the set $\operatorname{ST}_W(\mu)$ of standard tableaux of $\mu$ filled with $k$ numbers from $1$ to $W$ (instead of all numbers from $1$ to $k$). Its cardinality is $d_\mu \binom{W}{k}$.

On the other hand, consider the set ${\ensuremath{\mathcal E}} _{\textup{marked}}(\lambda, \mu)$ of marked excited diagrams of $\mu$ in $\lambda$, that is, the set ${\ensuremath{\mathcal E}} (\lambda, \mu)$ of excited diagrams of $\mu$ in $\lambda$ where each box $u$ is labelled with an integer from 1 to its hook length $H(\lambda, u)$. Its cardinality is $S(\lambda, \mu)$.

Now, consider the map which sends a tableau $T \in \operatorname{ST}_W(\mu)$ to a marked excited diagram, by applying the following to each box $u\in T$. Denote the label of $u$ by $\ell$. Let $i$ be the unique index such that $W_{i-1}<\ell \leq W_i$. Then move $u$ by $i-1$ steps along its diagonal (that is, add $(i-1, i-1)$ to the coordinates) and replace its label by $\ell - W_{i-1}$. We illustrate this process in Figure~\ref{fig: marked Naruse injection}.


Let us prove that it is well defined. Let $T\in \operatorname{ST}_W(\mu)$.
Write $\delta' = \delta-M$. Then each box of $T$ is moved in diagonal by at most $\delta'-1$ by construction. Since it starts in the square $[M^M]$ it ends in the square $[(M+\delta'-1)^{M+\delta'-1}] = [(\delta-1)^{\delta-1}]$, and hence in $\lambda$. Now, due to the standard constraints on the labels in $T$, the boxes higher in the same column or farther right in the same row of a given box $u$ move at least as much as $u$, that is, the displacement of boxes is weakly increasing along rows and columns. This implies that the resulting set of boxes is an excited diagram of $\mu$. Finally, consider a box $u\in T$ with label $\ell$, let $i$ be the unique index such that $W_{i-1}<\ell \leq W_i$. Let $v=u+(i-1, i-1)$ and $z = (M+i, M+i)$, coordinatewise. Then, using that $W_{i-1}<\ell \leq W_i$ and the monotonicity of the hook lengths, we have $1\leq \ell - W_{i-1} \leq W_i-W_{i-1} = H(\lambda, z)\leq H(\lambda, v)$. In particular the new label $\ell' = \ell - W_{i-1}$ is between $1$ and $H(\lambda, v)$. Combining all this shows that the image of $T$ is indeed in ${\ensuremath{\mathcal E}} _{\textup{marked}}(\lambda, \mu)$.

Now, it is easy to recover the original tableau (on the left in Figure~\ref{fig: marked Naruse injection}) from the marked diagram (on the right in Figure~\ref{fig: marked Naruse injection}), by pushing the boxes methodically along their diagonal, as close as possible to the corner, starting from the boxes on $\lambda^1$, then $\lambda^2$, and so on, and adding $W_i$ to the label of the box if it moved back by $(-i,-i)$ coordinatewise. This shows that the process is injective. 

Hence the process that we defined is an injection from $\operatorname{ST}_W(\mu)$ to  ${\ensuremath{\mathcal E}} _{\textup{marked}}(\lambda, \mu)$, which implies that $d_\mu \binom{W}{k} \leq S(\lambda, \mu)$, as desired.

<!-- proof-14: body_md -->
Let $\mu \subseteq \lambda$ be two $C$-balanced Young diagrams. Without loss of generality we may assume that $C\geq 2$. Denote $k = |\mu|$ and $n=|\lambda|$. If $k\geq C^{-8}n$, then the result holds by Lemma~\ref{lem: lower bound Dousse Feray k order n} (applied with $C^4$ instead of $C$).
    
    From now on we assume that $k\leq C^{-8}n$. 
Set $M = \lfloor C\sqrt{k} \rfloor$. The $C$-balance assumptions together with the condition $k\leq C^{-8}n$ then ensure that $\mu \subseteq [M^M] \subseteq \lambda$. We also denote $S = |\lambda^{\leq M}|$ and $W = |\lambda^{>M}|$.

We have $S \leq s(\lambda) M \leq C\sqrt{n} C\sqrt{k} = C^2\sqrt{kn}$. Therefore $W-k\geq n/2$, and we also have $W\geq k$. We deduce that
\begin{equation}
    \frac{n^{\downarrow k}}{W^{\downarrow k}} = \prod_{i=0}^{k-1} \left( 1+\frac{S}{W-i}\right) \leq \left( e^{S/(n/2)} \right)^k \leq e^{2C^2k^{3/2}/n^{1/2}}.
\end{equation}
It follows that 
\begin{equation}
    A_{\lambda, \mu} = k! \frac{d_{\lambda\backslash\mu}}{d_\mu d_\lambda} = \frac{W^{\downarrow k}}{n^{\downarrow k}} \frac{S(\lambda, \mu)}{d_\mu \binom{W}{k}} \geq e^{-2C^2k^{3/2}/n^{1/2}}\frac{S(\lambda, \mu)}{d_\mu \binom{W}{k}}.
\end{equation}
This concludes the proof, since $d_\mu \binom{W}{k} \leq S(\lambda, \mu)$ by Proposition~\ref{prop: marked injection Naruse}.
