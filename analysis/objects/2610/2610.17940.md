<!-- theorem-1: body_md -->
\label{theo1}
The Schur complements $S_k$ satisfy the following relation:
\begin{equation}
    \label{Sk}
  S_k =A_k + B_kA_0^{-1}B_k^{\top} - 
  \underbrace{\sum_{i = 2}^{k} F_{k, i}S_{i-1}^{-1}F_{k, i}^{\top}}_{\text{\normalsize $\Delta_k $}} \equiv A_k + B_kA_0^{-1}B_k^{\top} - \Delta_k
\end{equation}
where $F_{k, 1} =B_k$ and
\begin{equation} \label{Fk}
  F_{k, m} = F_{k, 1} A_0^{-1} F_{m-1, 1}^{\top} - 
	\sum_{j=2}^{m-1} F_{k, j}S_{j-1}^{-1}F_{m-1, j}^{\top}. \end{equation}
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
The matrix $\mathcal{D}\mathcal{L}^{\top}$ can be written as \[
	\mathcal{D}\mathcal{L}^{\top} = 
	\left[\begin{array}{ccccc}
	  A_0 & F_{1, 1}^{\top} &  F_{2, 1}^{\top} &  \cdots &  F_{N, 1}^{\top} \\ 
	  0 & -S_1 & -F_{2, 2}^{\top} &  \cdots &  -F_{N, 2}^{\top} \\ 
	  0 & 0 & -S_2 & & -F_{N, 3}^{\top} \\ 
	  0 & 0 & 0 & \ddots & \vdots \\
	  0 & 0 & 0 & 0 & -S_{N}
	\end{array}\right]
  \] 
  Let $r_i$ and  $c_j$ denote the  $i$-th row of  $\mathcal{L}$ and  $j$-th column of 
  $\mathcal{D}\mathcal{L}^{\top}$ respectively (starting from $0$).
We have $B_k = \mathcal{A}_{k, 0} =  r_{k} \cdot c_0 = F_{k, 1} $.
 For $k > m > 1$ we have  $0 = \mathcal{A}_{k, m-1} = l_{k} \cdot r_{m-1}$. This yields 
 \[
\left(
  F_{k, 1}A_1^{-1}, F_{k, 2}S_1^{-1}, \cdots, 
  F_{k, m}S_{m-1}^{-1}
  ,\cdots, F_{k, k}S_{k-1}^{-1}, I, 0
\right) \cdot \left(
  F_{m-1, 1}^{\top}, - F_{m-1, 2}^{\top}, \cdots, -F_{m-1, m-1}^{\top}, -S_{m-1}, 0
\right) = 0
\] which implies \eqref{Fk}. 
Finally, we see that $-A_k = \mathcal{A}_{k, k} = r_k \cdot c_k$. This yields
 \[\left(
  F_{k, 1}A_1^{-1}, F_{k, 2}S_1^{-1}, \cdots, 
   F_{k, k}S_{k-1}^{-1}, I, 0
\right) \cdot \left(
  F_{k, 1}^{\top}, - F_{k, 2}^{\top}, \cdots, -F_{k, k}^{\top}, -S_{k}, 0
\right) = -A_k \] which gives \eqref{Sk}.

<!-- theorem-2: body_md -->
\label{theo2}
    If $\mathcal A$ is invertible then all Schur complement are SPD.
<!-- end theorem-2 -->
<!-- proof-2: body_md -->
By induction. For $N=0$ this is trivially satisfied since
$\mathcal A \equiv A_0$, which is SPD by assumption.
Let the statement holds for $j < N$, which means that
all $S_i, i \le j$ are SPD. We define the matrix
\begin{equation}
    \label{SP}
 \mathcal  A_j = \begin{bmatrix} A_0 & \mathbf B_j^\top \\ \mathbf B_j & -\mathbf A_j\end{bmatrix}, \ \mathbf B_j = \begin{bmatrix} B_1^\top & \ldots & B_j^\top\end{bmatrix}^\top, \ \mathbf A_j = \text{blkdiag}(A_1, \ldots, A_j).\end{equation}
We also define $\mathbf S_j$ the Schur complement related to matrix $\mathcal A_j$, namely
$\mathbf S_j = \mathbf A_j + \mathbf B_j A_0^{-1} \mathbf B_j^\top$. Since by inductive hypothesis
all block Schur complements $S_1, \ldots, S_j$ are SPD,  $\mathcal A_j$ is invertible and
hence $\mathbf S_j $ is SPD.
We now write $\mathcal A_{j+1}$ in terms of $\mathcal A_j$ as
\[ \mathcal A_{j+1}  = 
 \begin{bmatrix} A_0 & \mathbf B_j^\top & B_{j+1}^\top\\ \mathbf B_j & -\mathbf A_j & \bf 0\\
 B_{j+1} & \mathbf 0 & -A_{j+1}\end{bmatrix} = 
\begin{bmatrix} \text{\Large $\mathcal A_j$} & \begin{array}{l} B_{j+1}^\top\\\mathbf 0 \end{array} \\ \begin{array}{ll} B_{j+1} & \mathbf 0 \end{array}  & -A_{j+1} \end{bmatrix}. \]
Then, by definition of the Schur complement, and exploiting the inverse of a saddle-point matrix, 
\begin{eqnarray*}  S_{j+1} &= &  A_{j+1} + \begin{bmatrix} B_{j+1} & \mathbf 0 \end{bmatrix} \mathcal A_j^{-1}  \begin{bmatrix}  B_{j+1}^\top \\\mathbf 0\end{bmatrix}= A_{j+1} + B_{j+1} (\mathcal A_j^{-1})_{11} B_{j+1}^\top\\
&=& A_{j+1}+
 B_{j+1} \left(A_0^{-1} - A_0^{-1} \mathbf B_j^\top \mathbf S_j^{-1} \mathbf B_j A_0^{-1}\right)
 B_{j+1}^\top = \\
 &=& A_{j+1}+
 B_{j+1} A_0^{-1/2} \underbrace{\left(I - A_0^{-1/2} \mathbf B_j^\top \mathbf S_j^{-1} \mathbf B_j A_0^{-1/2}\right)}_{\text{\large{$Z$}}}A_0^{-1/2}B_{j+1}^\top.
 \end{eqnarray*}
$Z$ is a symmetric orthogonal projector. $Z = I - W W^\top$, where $W = A_0^{-1/2} \mathbf B_j^\top \mathbf S_j^{-1/2}$. $W$ is a rectangular (skinny) matrix. The smallest eigenvalue
of $Z$ is $1 - \rho(W W^\top) = 1 - \rho(W^\top W)$.
Since \[W^\top W=    S_j^{-1/2}  \mathbf B_j A_0^{-1}
 \mathbf B_j^\top \mathbf  S_j^{-1/2} =
  \mathbf S_j^{-1/2}  (\mathbf S_j + A_j)\mathbf  S_j^{-1/2} = I -  \mathbf S_j^{-1/2}  A_j \mathbf S_j^{-1/2}, \]
  we have that $\rho(W W^\top) \le 1$ and therefore $Z$ is positive semidefinite,
  and so is $S_{j+1}$.
 The definiteness follows from invertibility of $\mathcal A$, together with the Sylvester inertia law.

    \textit{Let Y be a symmetric matrix valued function defined in $F\subset\mathbb{R}$ and}
    \[
        0\notin[\min\{\sigma(Y(\zeta))\},\max\{\sigma(Y(\zeta))\}]\quad \quad \textit{for all }\ \zeta\in F.

    \]

<!-- theorem-3: body_md -->
\label{theoremUpsilon}
Let ${\mathcal P}_D$ a Schur complement block diagonal preconditioner. 
Then the eigenvalues of 
${\mathcal P}_D^{-1} \mathcal A$ are located in
$[\lambda_-^{LB}, \lambda_-^{UB}]\ \cup
[\lambda_+^{LB}, \lambda_+^{UB}]
$, where
\begin{eqnarray}
    \lambda_-^{LB} & = &\frac{\alpha_E^{(0)} - \boldsymbol \beta_E
- \sqrt{(\alpha_E^{(0)} + \boldsymbol \beta_E)^2 + 4  \boldsymbol \beta_R}}{2}\\
    \lambda_-^{UB} & =& \frac{\beta_E^{(0)} - \boldsymbol \alpha_E
- \sqrt{(\beta_E^{(0)} + \boldsymbol \alpha_E)^2 + 4  \boldsymbol \alpha_R}}{2}\\
    \lambda_+^{LB} & = & \alpha_E^{(0)} \\
    \lambda_+^{UB} & = &
\frac{\beta_E^{(0)} - \boldsymbol \alpha_E
+ \sqrt{(\beta_E^{(0)} + \boldsymbol \alpha_E)^2 + 4  \boldsymbol \beta_R}}{2} 
\end{eqnarray}

If $\lambda \not \in [\alpha_E^{(0)}, \beta_E^{(0)}]$ then
\[ u_1 = (\lambda I - E_0)^{-1} \mathbf R^\top u_+,\]
and
\[ \left(\mathbf R (\lambda I - E_0)^{-1} \mathbf R^\top - \lambda \mathbf I - \mathbf E_+\right)u_+ = 0.\]
Premultiplying by $-\dfrac{u_+^T}{u_+^\top u_+}$ and applying Lemma \ref{Lemma1}, yield, after defining $s = \mathbf R^\top u_+$:	 
\begin{equation} 
0 = \lambda +\frac{u_+^\top \mathbf E_+ u_+}{u_+^\top u_+} - \frac{1}{\lambda - \dfrac{v^\top E_0 v}{v^\top v}} \frac{u_+^\top \mathbf R \mathbf R^\top u_+}{u_+^\top u_+} = \lambda + \Gamma_E -\frac{\Gamma_R}{\lambda - \gamma_E^{(0)}}, \qquad \text{or, equivalently,}
\end{equation}
\[ \lambda^2 + (\Gamma_E - \gamma_E^{(0)})\lambda - \gamma_E^{(0)}\Gamma_E - \Gamma_R=0,\]
and therefore any eigenvalue of $\mathcal P_D^{-1} \mathcal A$ not lying in $[\alpha_E^{(0)}, \beta_E^{(0)}]$ can be described as

\end{itemize}

<!-- theorem-4: body_md -->
\label{theoremPI}
    The eigenvalues of $\mathcal{P}_D^{-1}\mathcal{A}$ are located in $\displaystyle  [\mu_-^{LB}, -\boldsymbol \alpha_E] 
    \cup [\alpha_E^{(0)}, \mu_+^{UB}] $, where
    \begin{itemize}
    \item $\mu_-^{LB}$ is the negative zero of $\Pi$ with parameters: $\gamma_E^{(0)}=\alpha_E^{(0)}; \gamma_R^{(j)} =\beta_R^{(j)}, \gamma_E^{(j)} = \beta_E^{(j)}, j>0$. 
                                                                                                                                                           
    \item $\mu_+^{UB}$ is the positive zero of $\Pi$ with parameters: $\gamma_E^{(0)}=\beta_E^{(0)}; \gamma_R^{(j)}=\beta_R^{(j)}, \gamma_E^{(j)} =\alpha_E^{(j)}, j>0$. 
    \end{itemize}
<!-- end theorem-4 -->
<!-- proof-4: body_md -->
Let $\lambda \not \in [-\boldsymbol \beta_E, -\boldsymbol \alpha_E]$, then $E_j + \lambda I$
	is definite and hence invertible. Considering \eqref{eq1}, we can write
\[ u_+ = (\lambda \mathbf I + \mathbf E_+)^{-1} \mathbf R^\top u_1. \]
Upon substitution, the first equation reads
\[(  E_0 - \lambda I + \mathbf R^\top(\lambda \mathbf I + \mathbf E_+)^{-1} \mathbf R)u_1  = 0 .\]
Premultiplying by $-\dfrac{u_1^T}{u_1^T u_1}$, defining $s_k = R_k u_1$, and applying Lemma \ref{Lemma1} to $\dfrac{s_k^\top (\lambda I + E_k)^{-1} s_k}{s_k^\top s_k}$, yield:	 
\begin{equation} \label{Pidef} 0 =  -\frac{u_1^\top E_0 u_1}{u_1^\top u_1} + \lambda  - \sum_{k=1}^N \frac{u_1 R_k^\top R_k u_1}{u_1^\top u_1} \frac{1}{\lambda + \dfrac{v_k ^\top E_k v_k}{v_k^\top v_k}} 
= -\gamma_E^{(0)} + \lambda -\sum_{k=1}^{N}\frac{\gamma_R^{(k)}}{\lambda +\gamma_E^{(k)}} \equiv \Pi(\lambda),\end{equation}
with $\Pi: (-\infty, -\boldsymbol \beta_E) \cup (-\boldsymbol \alpha_E, + \infty) \to \mathbb R$.
Since $\displaystyle
\Pi'(\lambda) = 1 +
 \sum_{k=1}^{N}\frac{\gamma_R^{(k)}}{(\lambda +\gamma_E^{(k)})^2}>0$, the function $\Pi(\lambda)$ is increasing in every interval in which it is defined.
 \[ \text{Since also} \quad \lim_{\lambda \to -\infty} \Pi(\lambda) = -\infty\ \
  \lim_{\lambda \to -{\boldsymbol \beta}_E^-} \Pi(\lambda) = +\infty\qquad
   \lim_{\lambda \to -{\boldsymbol \alpha}_E^+} \Pi(\lambda) = -\infty\ \
  \lim_{\lambda \to +\infty} \Pi(\lambda) = +\infty,\]
$\Pi$ has exactly two zeros, separated by the endpoints of the intervals of its definition. 
If $\boldsymbol \alpha_E> 0$, then $-\boldsymbol \alpha_E$ gives a bound for 
the largest negative eigenvalue. In case $\boldsymbol \alpha_E = 0$ the bound is $0$ and hence is not descriptive.
                                                                                                                                                       
                                                      
Easy computation shows
\begin{eqnarray}
    \frac{\partial \Pi}{\partial \gamma_E^{(j)}} &=& \begin{cases}
    -1 & \text{if} \ j =0\\  \dfrac{\gamma_R^{(j)}}{(\lambda + \gamma_E^{(j)})^2} & \text{if} \ j >0 \end{cases}
    \quad \longrightarrow \ \textnormal{sgn}\left( \frac{\partial \Pi}{\partial \gamma_E^{(j)}}\right) = 
    \begin{cases}
    -1 & \text{if} \ j =0\\  +1 & \text{if} \ j >0 \end{cases}
    \\
    \frac{\partial \Pi}{\partial \gamma_R^{(j)}} &=&
    -\frac{1}{\lambda + \gamma_E^{(j)}}.
\end{eqnarray}
By the implicit function theorem \cite[Eq. (17)]{pilotto-et-al2026}, since the function $\Pi(\lambda)$ is locally increasing, the monotonicity of its zeros as a function of the parameters reverts that of the polynomial.
The following table summarizes the monotonicities of
the three bounding roots, with respect to the parameters.

\begin{tabular}{lccc||ccc}
& $\dfrac{\partial \lambda}{\partial \gamma_R^{(j)}}(\xi) $
& $\dfrac{\partial \lambda}{\partial \gamma_E^{(0)}}(\xi) $
& $\dfrac{\partial \lambda}{\partial \gamma_E^{(j)}}(\xi)$ & $\gamma_R^{(j)}$ & $\gamma_E^{(0)} $ &
$\gamma_E^{(j)}, j > 0 $ \\[1.2em] \hline
 Smallest negative root $\xi$&  $-$ &  $+$ & $-$ & $\beta_R^{(j)}$&$\alpha_E^{(0)}$ &  $\beta_E^{(j)}$\\
                                                                 
                                                                        
Smallest positive root $\xi$&  $+$ &  $+$ & $-$&0&$\alpha_E^{(0)}$ &  $\beta_E^{(j)}$\\
 Largest positive root $\xi$&  $+$ &  $+$ & $-$&
 $\beta_R^{(j)}$&$\beta_E^{(0)}$&$\alpha_E^{(j)} $
 \\ 
\end{tabular}

\noindent
The thesis follows, by observing that $\Pi(\xi)$ for
$\gamma_R^{(j)} = 0, \gamma_E^{(0)} = \alpha_E^{(0)}$ simplifies to $\xi - \alpha_E^{(0)}$.

<!-- corollary-1: body_md -->
\label{corollary}
    The eigenvalues of the preconditioned multiple saddle-point matrix are in the following intervals:
\[[\mu_-^{LB}, \lambda_-^{LB}] 
    \cup [\alpha_E^{(0)}, \mu_+^{UB}].\]
<!-- end corollary-1 -->
<!-- proof-5: body_md -->
To show that  $\lambda_-^{LB} < -\boldsymbol \alpha_E$, we evaluate $\Upsilon$ with parameters $\boldsymbol \alpha_R, \beta_E^{(0)}, \boldsymbol \alpha_E$ on $\lambda  = -\boldsymbol \alpha_E$ to get $\Upsilon(-\boldsymbol \alpha_E) = -\Gamma_R < 0$. 
    
    Second, denoting $\xi \equiv \lambda_+^{UB}$  it satisfies $\xi - \beta_E^{(0)} = \dfrac{\boldsymbol \beta_R}{\xi +\boldsymbol \alpha_E}.$ Now we evaluate on $\xi$ the function $\Pi$ with the parameters
    used for computing the upper bound for the positive eigenvalues. With this choice we have
     $\displaystyle \Pi(\xi) = \dfrac{\boldsymbol \beta_R}{\xi + \boldsymbol \alpha_E}- \sum_{k=1}^{N}\frac{\beta_R^{(k)}}{\xi +\alpha_E^{(k)}}.$ Let $j$ such that $\boldsymbol \alpha_E = \min_k \alpha_E^{(k)} = \alpha_E^{(j)}$, then
     \[ \Pi(\xi) = 
     \frac{\boldsymbol\beta_R}{\xi + \alpha_E^{(j)}}- \sum_{k=1}^{N}\frac{\beta_R^{(k)}}{\xi +\alpha_E^{(k)}} \ge
     \frac{\boldsymbol\beta_R}{\xi + \alpha_E^{(j)}}- \sum_{k=1}^{N}\frac{\beta_R^{(k)}}{\xi +\alpha_E^{(j)}} = \frac{\beta_R - \sum_{k=1}^N \beta_R^{(k)}}{\xi + \alpha_E^{(j)}} \ge 0,\]
     showing that $\lambda_+^{UB}  \ge \mu_+^{UB}.$ 
     
     Analogously, setting $\xi = \lambda_-^{LB}$, and evaluating
     the function $\Pi$, with the parameters used for computing the smallest negative eigenvalue, on $\xi$, yields,
      \[ \Pi(\xi) = 
     \frac{-\boldsymbol\beta_R}{-\xi -\beta_E^{(j)}}+ \sum_{k=1}^{N}\frac{\beta_R^{(k)}}{-\xi -\beta_E^{(k)}} \le
     \frac{-\boldsymbol\beta_R}{-\xi -\beta_E^{(j)}}+\sum_{k=1}^{N}\frac{\beta_R^{(k)}}{-\xi -\beta_E^{(j)}} = \frac{-\beta_R+\sum_{k=1}^N \beta_R^{(k)}}{-\xi- \beta_E^{(j)}} \le 0,\]
     where $j$ is such that $\boldsymbol \beta_E = \max_k \beta_E^{(k)} = \beta_E^{(j)}$. This shows that 
$\lambda_-^{LB}  \le \mu_-^{LB}.$
