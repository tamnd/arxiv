<!-- lemma-1: body_md -->
\label{lem:projection}
Let $A$ and $B$ be independent random subsets of $[n]$, set
$a_i=\mathbb P(i\in A)$, $b_i=\mathbb P(i\in B)$, and $Z=A\setminus B$.
Then, for every $J\subseteq[n]$,
\begin{equation}\label{eq:projection}
 \operatorname{TC}(Z)+H(B_J)\ge\sum_{i\in J}g(a_i,b_i).
\end{equation}
Moreover, let $t_1,\ldots,t_n\in[0,1]$, choose a permutation $\pi$ of $[n]$ such that
$t_{\pi(1)}\ge\cdots\ge t_{\pi(n)}$, and define
\[
 c^B_{\pi(k)}=H\bigl(B_{\pi(k)}\mid B_{\pi(1)},\ldots,B_{\pi(k-1)}\bigr)
 \qquad(1\le k\le n).
\]
Then
\begin{equation}\label{eq:weighted-proj}
 \operatorname{TC}(Z)+\sum_{i=1}^n t_i c_i^B
 \ge\sum_{i=1}^n t_i g(a_i,b_i).
\end{equation}
For $n=0$, both assertions are understood as $0\ge0$.

<!-- proof-1: body_md -->
By \eqref{eq:TC-split},
\[
 \operatorname{TC}(Z)=\operatorname{TC}(Z_J)+\operatorname{TC}(Z_{J^c})+I(Z_J;Z_{J^c})
 \ge \operatorname{TC}(Z_J).
\]
For every $U\subseteq J$, on the event $B_J=U$ we have $Z_J=A_{J\setminus U}$. Since $A$ and $B$ are independent,
\begin{align*}
 H(Z_J\mid B_J)
 &=\sum_{U\subseteq J}\mathbb P(B_J=U)H(A_{J\setminus U})\\
 &\le\sum_{U\subseteq J}\mathbb P(B_J=U)
       \sum_{i\in J\setminus U}h(a_i)\\
 &=\sum_{i\in J}\mathbb P(i\notin B)h(a_i)
 =\sum_{i\in J}(1-b_i)h(a_i),
\end{align*}
where the inequality is \eqref{eq:subadditivity}. Moreover,
\[
 H(Z_J)\le H(Z_J,B_J)=H(B_J)+H(Z_J\mid B_J)
\]
by \eqref{eq:chain-two}. Hence
\[
 H(Z_J)\le H(B_J)+\sum_{i\in J}(1-b_i)h(a_i).
\]
Using $H(Z_i)=h(a_i(1-b_i))$ and the definition of $g$, we obtain
\begin{align*}
 \operatorname{TC}(Z_J)
 &=\sum_{i\in J}h(a_i(1-b_i))-H(Z_J)\\
 &\ge\sum_{i\in J}\bigl[h(a_i(1-b_i))-(1-b_i)h(a_i)\bigr]-H(B_J)\\
 &=\sum_{i\in J}g(a_i,b_i)-H(B_J),
\end{align*}
which proves \eqref{eq:projection}.

We now prove \eqref{eq:weighted-proj}. Let $t_{\max}=\max_i t_i$, with $t_{\max}=0$ when $n=0$, and for $0\le u<t_{\max}$ put
\[
 J_u=\{i:t_i>u\}.
\]
The definition is chosen so that coordinate $i$ belongs to $J_u$ for a set of values of $u$ of length $t_i$. Because the weights are ordered decreasingly by $\pi$, each $J_u$ is an initial segment of that order, and \eqref{eq:projection-chain} gives
$H(B_{J_u})=\sum_{i\in J_u}c_i^B$. We can therefore integrate \eqref{eq:projection} over $u\in[0,t_{\max}]$. Since
$\mathbf 1_{\{i\in J_u\}}=\mathbf 1_{\{u<t_i\}}$, the left-hand side becomes
\begin{align*}
 &\int_0^{t_{\max}}\left[\operatorname{TC}(Z)+H(B_{J_u})\right]du\\
 &\qquad=t_{\max}\operatorname{TC}(Z)
   +\sum_i c_i^B\int_0^{t_{\max}}\mathbf 1_{\{u<t_i\}}du
 =t_{\max}\operatorname{TC}(Z)+\sum_i t_i c_i^B,
\end{align*}
and similarly
\[
 \int_0^{t_{\max}}\sum_{i\in J_u}g(a_i,b_i)\,du
 =\sum_i t_i g(a_i,b_i).
\]
Therefore
\[
 t_{\max}\operatorname{TC}(Z)+\sum_i t_i c_i^B
 \ge\sum_i t_i g(a_i,b_i).
\]
Since $0\le t_{\max}\le1$ and $\operatorname{TC}(Z)\ge0$, replacing
$t_{\max}\operatorname{TC}(Z)$ by the larger quantity $\operatorname{TC}(Z)$ proves \eqref{eq:weighted-proj}.

<!-- lemma-2: body_md -->
\label{lem:strongV}
Let $(\mathcal A,\mathcal B)$ be a recovering pair of nonempty families on $[n]$, and let $A,B,T,a_i,b_i$ be as in \eqref{eq:basic-notation}. Then, for every $\lambda\ge0$,
\begin{equation}\label{eq:strongV}
 \log_2T\le\sum_{i=1}^n\widehat V_\lambda(a_i,b_i).
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
The recovering property implies that $A$ is determined by $Z=A\setminus B$ and $B$ is determined by $W=B\setminus A$. Hence $H(A,Z)=H(Z)$ and, by \eqref{eq:chain-two},
\begin{align*}
 H(Z)
 &=H(A)+H(Z\mid A)\\
 &=H(A)+\sum_{S\subseteq[n]}\mathbb P(A=S)H(S\setminus B\mid A=S)\\
 &=H(A)+\sum_{S\subseteq[n]}\mathbb P(A=S)H(B_S).
\end{align*}
The last equality uses independence of $A$ and $B$ and the bijection $B_S\mapsto S\setminus B$. Interchanging $A$ and $B$ gives
\begin{equation}\label{eq:recoveryentropy}
 H(Z)=H(A)+\sum_{S\subseteq[n]}\mathbb P(A=S)H(B_S),\qquad
 H(W)=H(B)+\sum_{S\subseteq[n]}\mathbb P(B=S)H(A_S).
\end{equation}

Choose an order of the coordinates by decreasing
$t_i=(a_i-\lambda)_+$ and let $c_i^B$ be the conditional entropies defined from that order as in \eqref{eq:projection-chain}. For each fixed set $S$, \eqref{eq:projection-chain} gives
$H(B_S)\ge\sum_{i\in S}c_i^B$. Multiplying by $\mathbb P(A=S)$ and summing over $S$ gives
\[
 \sum_{S\subseteq[n]}\mathbb P(A=S)H(B_S)\ge\sum_i a_i c_i^B.
\]
Since $a_i=\lambda+t_i-(\lambda-a_i)_+$,
\begin{align*}
 \sum_i a_i c_i^B
 &=\lambda\sum_i c_i^B+\sum_i t_i c_i^B
   -\sum_i(\lambda-a_i)_+c_i^B\\
 &=\lambda H(B)+\sum_i t_i c_i^B
   -\sum_i(\lambda-a_i)_+c_i^B.
\end{align*}
By \eqref{eq:projection-chain}, $c_i^B\le H(B_i)=h(b_i)$. Since each coefficient $(\lambda-a_i)_+$ is nonnegative,
\[
 -\sum_i(\lambda-a_i)_+c_i^B
 \ge -\sum_i(\lambda-a_i)_+h(b_i).
\]
Combining the last three displays gives
\begin{equation}\label{eq:projection-B}
 \sum_{S\subseteq[n]}\mathbb P(A=S)H(B_S)
 \ge \lambda H(B)+\sum_i t_i c_i^B-\sum_i(\lambda-a_i)_+h(b_i).
\end{equation}
Applying the same argument with $A$ and $B$ interchanged, with
$s_i=(b_i-\lambda)_+$, gives
\begin{equation}\label{eq:projection-A}
 \sum_{S\subseteq[n]}\mathbb P(B=S)H(A_S)
 \ge \lambda H(A)+\sum_i s_i c_i^A-\sum_i(\lambda-b_i)_+h(a_i),
\end{equation}
where $c_i^A$ denotes the conditional entropies associated with the order chosen for the weights $s_i$.

By \eqref{eq:TC} and the definitions of $p_i$ and $q_i$,
\[
 H(Z)=\sum_i h(p_i)-\operatorname{TC}(Z),
 \qquad
 H(W)=\sum_i h(q_i)-\operatorname{TC}(W).
\]
Combining the first identity in \eqref{eq:recoveryentropy} with \eqref{eq:projection-B} therefore gives
\begin{align*}
 H(A)+\lambda H(B)
 \le{}&\sum_i h(p_i)-\operatorname{TC}(Z)-\sum_i t_i c_i^B\\
 &+\sum_i(\lambda-a_i)_+h(b_i).
\end{align*}
Similarly, the second identity in \eqref{eq:recoveryentropy} and \eqref{eq:projection-A} give
\begin{align*}
 H(B)+\lambda H(A)
 \le{}&\sum_i h(q_i)-\operatorname{TC}(W)-\sum_i s_i c_i^A\\
 &+\sum_i(\lambda-b_i)_+h(a_i).
\end{align*}
Adding these two inequalities and using
$H(A)+H(B)=\log_2T$ yields
\begin{align*}
 (1+\lambda)\log_2T\le{}&\sum_i\bigl[h(p_i)+h(q_i)
 +(\lambda-a_i)_+h(b_i)+(\lambda-b_i)_+h(a_i)\bigr]\\
 &-\left[\operatorname{TC}(Z)+\sum_i t_i c_i^B\right]
 -\left[\operatorname{TC}(W)+\sum_i s_i c_i^A\right].
\end{align*}
By \eqref{eq:weighted-proj},
\[
 \operatorname{TC}(Z)+\sum_i t_i c_i^B
 \ge\sum_i t_i g(a_i,b_i),
\]
and the same inequality with $A$ and $B$ interchanged gives
\[
 \operatorname{TC}(W)+\sum_i s_i c_i^A
 \ge\sum_i s_i g(b_i,a_i).
\]
Since $t_i=(a_i-\lambda)_+$ and $s_i=(b_i-\lambda)_+$, we obtain
\begin{align*}
 (1+\lambda)\log_2T
 \le \sum_i\Bigl[&
 h(p_i)+h(q_i)+(\lambda-a_i)_+h(b_i)+(\lambda-b_i)_+h(a_i)\\
 &-(a_i-\lambda)_+g(a_i,b_i)
 -(b_i-\lambda)_+g(b_i,a_i)\Bigr].
\end{align*}
The bracket is
$(1+\lambda)\widehat V_\lambda(a_i,b_i)$ by \eqref{eq:Vhat}. Dividing by
$1+\lambda>0$ proves \eqref{eq:strongV}.

<!-- lemma-3: body_md -->
\label{lem:remainder}
Let $A$ and $B$ be independent random subsets of $[n]$, set
$a_i=\mathbb P(i\in A)$, $b_i=\mathbb P(i\in B)$, and $Z=A\setminus B$.
For every $J\subseteq[n]$,
\begin{align}\label{eq:remainder}
 &\operatorname{TC}(Z)+H(B_J)-\sum_{i\in J}g(a_i,b_i)\notag\\
 &\quad=\operatorname{TC}(Z)-\operatorname{TC}(Z_J)
 +H(B_J\mid Z_J)
 +\sum_{U\subseteq J}\mathbb P(B_J=U)\operatorname{TC}(A_{J\setminus U}).
\end{align}
Every term on the second line is nonnegative. Moreover, for weights $t_i\in[0,1]$ ordered as in Lemma~\ref{lem:projection}, with $t_{\max}=\max_i t_i$ and $J_u=\{i:t_i>u\}$,
\begin{align}\label{eq:weighted-remainder}
 &\operatorname{TC}(Z)+\sum_i t_i c_i^B-\sum_i t_i g(a_i,b_i)\notag\\
 &\quad=(1-t_{\max})\operatorname{TC}(Z)
 +\int_0^{t_{\max}}\Bigl[
 \operatorname{TC}(Z)-\operatorname{TC}(Z_{J_u})
 +H(B_{J_u}\mid Z_{J_u})\notag\\
 &\hspace{27mm}+\sum_{U\subseteq J_u}\mathbb P(B_{J_u}=U)
 \operatorname{TC}(A_{J_u\setminus U})
 \Bigr]du.
\end{align}
For $n=0$, the identity is interpreted with $t_{\max}=0$.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
For a fixed $U\subseteq J$, conditioning on $B_J=U$ makes $Z_J$ the projection of $A$ onto $J\setminus U$. Independence therefore gives
\begin{subequations}\label{eq:conditional-TC-identities}
\begin{equation}\label{eq:conditional-projection}
 H(Z_J\mid B_J)
 =\sum_{U\subseteq J}\mathbb P(B_J=U)H(A_{J\setminus U}).
\end{equation}
Also
\begin{equation}\label{eq:marginal-projection}
 \sum_{i\in J}(1-b_i)h(a_i)
 =\sum_{U\subseteq J}\mathbb P(B_J=U)
   \sum_{i\in J\setminus U}h(a_i).
\end{equation}
Subtracting \eqref{eq:conditional-projection} from \eqref{eq:marginal-projection} gives
\begin{equation}\label{eq:conditional-TC}
 \sum_{i\in J}(1-b_i)h(a_i)-H(Z_J\mid B_J)
 =\sum_{U\subseteq J}\mathbb P(B_J=U)\operatorname{TC}(A_{J\setminus U}).
\end{equation}
\end{subequations}
Now expand the left-hand side of \eqref{eq:remainder}. Since
$H(Z_i)=h(a_i(1-b_i))$,
\begin{align*}
 &\operatorname{TC}(Z)+H(B_J)-\sum_{i\in J}g(a_i,b_i)\\
 &\quad=\operatorname{TC}(Z)+H(B_J)
 -\sum_{i\in J}H(Z_i)+\sum_{i\in J}(1-b_i)h(a_i)\\
 &\quad=\operatorname{TC}(Z)-\operatorname{TC}(Z_J)
 +H(B_J)-H(Z_J)+\sum_{i\in J}(1-b_i)h(a_i).
\end{align*}
The chain rule gives
\[
 H(B_J)-H(Z_J)=H(B_J\mid Z_J)-H(Z_J\mid B_J).
\]
Substituting this identity and then \eqref{eq:conditional-TC} gives \eqref{eq:remainder}. Finally,
\[
 \operatorname{TC}(Z)-\operatorname{TC}(Z_J)
 =\operatorname{TC}(Z_{J^c})+I(Z_J;Z_{J^c})\ge0
\]
by \eqref{eq:TC-split}; the other terms on the right-hand side of \eqref{eq:remainder} are nonnegative by definition or by \eqref{eq:subadditivity}.
Integrate \eqref{eq:remainder} for $J=J_u$ over $0\le u\le t_{\max}$. Since $J_u$ is an initial segment of the chosen order, \eqref{eq:projection-chain} gives
$H(B_{J_u})=\sum_{i\in J_u}c_i^B$. Therefore
\begin{align*}
 \int_0^{t_{\max}}H(B_{J_u})\,du&=\sum_i t_i c_i^B,\\
 \int_0^{t_{\max}}\sum_{i\in J_u}g(a_i,b_i)\,du
 &=\sum_i t_i g(a_i,b_i).
\end{align*}
The integral of $\operatorname{TC}(Z)$ is $t_{\max}\operatorname{TC}(Z)$. Thus the integrated form of \eqref{eq:remainder} has
$t_{\max}\operatorname{TC}(Z)+\sum_i t_i c_i^B-\sum_i t_i g(a_i,b_i)$ on the left. Adding
$(1-t_{\max})\operatorname{TC}(Z)$ to both sides gives \eqref{eq:weighted-remainder}.

<!-- lemma-4: body_md -->
\label{lem:conditional}
Let $(\mathcal A,\mathcal B)$ be a recovering pair of nonempty families on $[n]$, and use the notation in \eqref{eq:basic-notation}. Put
\[
 \Sigma_Z=\sum_{i=1}^n h(p_i),\qquad
 \Sigma_W=\sum_{i=1}^n h(q_i).
\]
Then, for every $J\subseteq[n]$,
\begin{align}\label{eq:conditional}
 H(B_J\mid Z_J)&\ge
 \left[\log_2T-\Sigma_Z-\sum_{i\notin J}(1-a_i)h(b_i)\right]_+,\notag\\
 H(A_J\mid W_J)&\ge
 \left[\log_2T-\Sigma_W-\sum_{i\notin J}(1-b_i)h(a_i)\right]_+.
\end{align}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Because $A$ is determined by $Z$ and $Z$ is a function of $(A,B)$,
\[
 H(B\mid Z)=H(A,B\mid Z)=H(A,B)-H(Z)=\log_2T-H(Z).
\]
Because $A$ is a function of $Z$,
\[
 H(B_{J^c}\mid Z)=H(B_{J^c}\mid A,Z).
\]
Fix a value $A=S$. For $i\in S$, the relation $Z=A\setminus B$ gives
$B_i=1-Z_i$, so these coordinates of $B$ are determined once $A=S$ and $Z$ are known. Consequently, once $A=S$ and $Z$ are fixed, the vector $B_{J^c}$ is determined by its coordinates in $J^c\setminus S$. Thus
\[
 H(B_{J^c}\mid A=S,Z)
 =H(B_{J^c\setminus S}\mid A=S,Z)
 \le H(B_{J^c\setminus S}\mid A=S).
\]
The last inequality is \eqref{eq:conditioning}. Since $A$ and $B$ are independent,
$H(B_{J^c\setminus S}\mid A=S)=H(B_{J^c\setminus S})$, and \eqref{eq:subadditivity} gives
\[
 H(B_{J^c\setminus S})
 \le\sum_{i\in J^c\setminus S}h(b_i).
\]
Averaging over $A$ therefore yields
\begin{align*}
 H(B_{J^c}\mid Z)
 &\le \sum_{S\subseteq[n]}\mathbb P(A=S)
      \sum_{i\in J^c\setminus S}h(b_i)\\
 &=\sum_{i\notin J}\mathbb P(i\notin A)h(b_i)
 =\sum_{i\notin J}(1-a_i)h(b_i).
\end{align*}

Next, the chain rule and \eqref{eq:conditioning} give
\begin{align*}
 H(B\mid Z)
 &=H(B_J\mid Z)+H(B_{J^c}\mid B_J,Z)\\
 &\le H(B_J\mid Z)+H(B_{J^c}\mid Z).
\end{align*}
Hence
\[
 H(B_J\mid Z)\ge H(B\mid Z)-H(B_{J^c}\mid Z).
\]
Since $Z$ contains $Z_J$, conditioning on $Z$ can only decrease entropy, so
\[
 H(B_J\mid Z_J)\ge H(B_J\mid Z).
\]
Combining these inequalities with the bounds for $H(B\mid Z)$ and $H(B_{J^c}\mid Z)$ obtained above gives
\begin{align*}
 H(B_J\mid Z_J)
 &\ge \log_2T-H(Z)-\sum_{i\notin J}(1-a_i)h(b_i)\\
 &\ge \log_2T-\Sigma_Z-\sum_{i\notin J}(1-a_i)h(b_i).
\end{align*}
Let
\[
 R_J=\log_2T-\Sigma_Z-\sum_{i\notin J}(1-a_i)h(b_i).
\]
We have proved $H(B_J\mid Z_J)\ge R_J$. Since conditional entropy is also nonnegative,
\[
 H(B_J\mid Z_J)\ge\max\{R_J,0\}=[R_J]_+,
\]
which is the first inequality in \eqref{eq:conditional}. Interchanging $A$ with $B$ and $Z$ with $W$ gives the second inequality.

\label{lem:integrated}
Let $(\mathcal A,\mathcal B)$ be a recovering pair of nonempty families on $[n]$, and use the notation in \eqref{eq:basic-notation} and the quantities $\Sigma_Z,\Sigma_W$ defined in Lemma~\ref{lem:conditional}. Fix $\lambda\ge0$ and put
\[
 t_i=(a_i-\lambda)_+,
 \qquad s_i=(b_i-\lambda)_+,
 \qquad J_u=\{i:t_i>u\},
 \qquad K_u=\{i:s_i>u\}.
\]
Let $t_{\max}=\max_i t_i$ and $s_{\max}=\max_i s_i$, both interpreted as $0$ when $n=0$. Then
\begin{align}\label{eq:integrated}
 &(1+\lambda)\left[\sum_i\widehat V_\lambda(a_i,b_i)-\log_2T\right]\notag\\
 &\quad\ge
 \int_0^{t_{\max}}
 \left[\log_2T-\Sigma_Z-\sum_{i\notin J_u}(1-a_i)h(b_i)\right]_+du\notag\\
 &\qquad+
 \int_0^{s_{\max}}
 \left[\log_2T-\Sigma_W-\sum_{i\notin K_u}(1-b_i)h(a_i)\right]_+du.

\begin{align}
 (1+\lambda)\log_2T\le{}&\sum_i\bigl[h(p_i)+h(q_i)
 +(\lambda-a_i)_+h(b_i)+(\lambda-b_i)_+h(a_i)\bigr]\notag\\
 &-\left[\operatorname{TC}(Z)+\sum_i t_i c_i^B\right]
 -\left[\operatorname{TC}(W)+\sum_i s_i c_i^A\right].\label{eq:integrated-upper}
\end{align}

 &-t_i g(a_i,b_i)-s_i g(b_i,a_i)\bigr].\label{eq:integrated-Vsum}

 &(1+\lambda)\left[\sum_i\widehat V_\lambda(a_i,b_i)-\log_2T\right]\\
 &\quad\ge
 \left[\operatorname{TC}(Z)+\sum_i t_i c_i^B-\sum_i t_i g(a_i,b_i)\right]\\
 &\qquad+
 \left[\operatorname{TC}(W)+\sum_i s_i c_i^A-\sum_i s_i g(b_i,a_i)\right].

\[
 \operatorname{TC}(Z)+\sum_i t_i c_i^B-\sum_i t_i g(a_i,b_i)
 \ge\int_0^{t_{\max}}H(B_{J_u}\mid Z_{J_u})\,du.

\[
 \operatorname{TC}(W)+\sum_i s_i c_i^A-\sum_i s_i g(b_i,a_i)
 \ge\int_0^{s_{\max}}H(A_{K_u}\mid W_{K_u})\,du.

<!-- theorem-2: body_md -->
\label{thm:W}
Let $(\mathcal A,\mathcal B)$ be a recovering pair of nonempty families on $[n]$, and let $A,B,T,a_i,b_i$ be as in \eqref{eq:basic-notation}. Then, for every $0\le\lambda\le\tau\le1$,
\begin{equation}\label{eq:Wglobal}
 \log_2T\le\sum_{i=1}^n W_{\lambda,\tau}(a_i,b_i).
\end{equation}
<!-- end theorem-2 -->
<!-- proof-6: body_md -->
Let $\Sigma_Z,\Sigma_W$ be as in Lemma~\ref{lem:conditional}, and set
\[
 t_i=(a_i-\lambda)_+,\qquad s_i=(b_i-\lambda)_+,\qquad
 t_{\max}=\max_i t_i,\qquad s_{\max}=\max_i s_i,
\]
with both maxima interpreted as $0$ when $n=0$. For $u\ge0$ put
$J_u=\{i:t_i>u\}$ and $K_u=\{i:s_i>u\}$. Finally, for $v\in[\lambda,1]$ define
\[
 C_B(v)=\sum_{i:a_i\le v}(1-a_i)h(b_i),\qquad
 C_A(v)=\sum_{i:b_i\le v}(1-b_i)h(a_i).
\]
For $u\ge0$,
\[
 i\notin J_u
 \Longleftrightarrow (a_i-\lambda)_+\le u
 \Longleftrightarrow a_i\le\lambda+u,
\]
and the same equivalence holds with $b_i$ and $K_u$ in place of $a_i$ and $J_u$. Hence, whenever $0\le u\le1-\lambda$,
\begin{equation}\label{eq:CA-CB-identities}
 \sum_{i\notin J_u}(1-a_i)h(b_i)=C_B(\lambda+u),
 \qquad
 \sum_{i\notin K_u}(1-b_i)h(a_i)=C_A(\lambda+u).
\end{equation}
Lemma~\ref{lem:conditional} with $J=\varnothing$ gives
\begin{equation}\label{eq:empty-set}
 \log_2T-\Sigma_Z-C_B(1)\le0,
 \qquad
 \log_2T-\Sigma_W-C_A(1)\le0.
\end{equation}
If $t_{\max}\le u\le1-\lambda$, then $J_u=\varnothing$ and $\lambda+u\ge\max_i a_i$, so $C_B(\lambda+u)=C_B(1)$. Using \eqref{eq:empty-set}, the first integrand in \eqref{eq:integrated} is therefore zero. Similarly, the second integrand is zero for $s_{\max}\le u\le1-\lambda$. In formulas,
\begin{equation}\label{eq:integrand-zero}
 \begin{aligned}
 \left[\log_2T-\Sigma_Z-C_B(\lambda+u)\right]_+&=0
 &&(t_{\max}\le u\le1-\lambda),\\
 \left[\log_2T-\Sigma_W-C_A(\lambda+u)\right]_+&=0
 &&(s_{\max}\le u\le1-\lambda).
 \end{aligned}
\end{equation}
Because $\tau-\lambda\le1-\lambda$, equations \eqref{eq:CA-CB-identities} and \eqref{eq:integrand-zero} justify replacing the upper limit of either integral in \eqref{eq:integrated} by $\tau-\lambda$: if the new interval is longer, \eqref{eq:integrand-zero} shows that the added part is zero; if it is shorter, only nonnegative terms are removed. We then use
$[r]_+\ge r$ and substitute $v=\lambda+u$. This gives
\begin{align*}

<!-- lemma-6: body_md -->
\label{lem:direct}
Let $(\mathcal A,\mathcal B)$ be a recovering pair of nonempty families on $[n]$, and let $A,B,T,a_i,b_i$ be as in \eqref{eq:basic-notation}. Then
\begin{equation}\label{eq:direct}

\end{equation}
<!-- end lemma-6 -->
<!-- proof-8: body_md -->
Let $Z=A\setminus B$, and fix $S\subseteq[n]$ with $\mathbb P(Z=S)>0$. By the recovering property, $Z=S$ determines a unique set $C\in\mathcal A$ with $A=C$. Since the event has positive probability, there is $B_0\in\mathcal B$ such that $S=C\setminus B_0$. Choose $U\subseteq S$ as in Proposition~\ref{prop:MSV}.

Because $Z=S$ forces $A=C$, independence of $A$ and $B$ gives
\begin{align*}
 \mathbb P(Z=S)
 &=\mathbb P(A=C)\mathbb P(C\setminus B=S)\\
 &=\frac{1}{|\mathcal A|}
   \frac{|\{B'\in\mathcal B:C\setminus B'=S\}|}{|\mathcal B|}.
\end{align*}
By Proposition~\ref{prop:MSV}, $\psi_{C,S}$ is injective, so the cardinality in the numerator is $|\mathcal B_{C,S}|$. Thus
\begin{equation}\label{eq:atomprob}
 \mathbb P(Z=S)=\frac{|\mathcal B_{C,S}|}{T}.
\end{equation}
The families $\mathcal A_{C,S,U}$ and $\mathcal B_{C,S}$, with $U$ chosen as in Proposition~\ref{prop:MSV}, are nonempty and form a cancellative pair on the $n-|C|$ coordinates of $[n]\setminus C$. Hence Theorem~\ref{thm:FH} gives
\begin{equation}\label{eq:FHfilter}
 |\mathcal A_{C,S,U}|\,|\mathcal B_{C,S}|\le(9/4)^{n-|C|}.
\end{equation}
By \eqref{eq:largestatom},
\[
 \frac1{|\mathcal A_{C,S,U}|}
 \le \frac{2^{\sum_{i\in S}h(a_i)}}{|\mathcal A|}.
\]
Therefore \eqref{eq:FHfilter} implies
\[
 |\mathcal B_{C,S}|
 \le \frac{(9/4)^{n-|C|}}{|\mathcal A|}
      2^{\sum_{i\in S}h(a_i)}.
\]
Using \eqref{eq:atomprob}, we obtain
\[
 \mathbb P(Z=S)
 \le \frac{(9/4)^{n-|C|}}{T|\mathcal A|}
      2^{\sum_{i\in S}h(a_i)}.
\]
Taking negative base-$2$ logarithms gives
\[
 -\log_2\mathbb P(Z=S)
 \ge \log_2T+\log_2|\mathcal A|-(n-|C|)L-\sum_{i\in S}h(a_i).
\]

We now average this inequality over $S=Z$. Since the recovering property determines $A$ from $Z$, the set denoted by $C$ for the realized value $S=Z$ is the random set $A$. Hence
\[
 \mathbb E|C|=\mathbb E|A|=\sum_i a_i.
\]
Moreover,
\[
 \mathbb E\!\left[\sum_{i\in Z}h(a_i)\right]
 =\sum_i \mathbb P(i\in Z)h(a_i)
 =\sum_i p_i h(a_i),
\]
because independence gives
$\mathbb P(i\in Z)=\mathbb P(i\in A,i\notin B)=a_i(1-b_i)=p_i$.
Finally,
\[
 \sum_{\substack{S\subseteq[n]\\ \mathbb P(Z=S)>0}}
 \mathbb P(Z=S)\bigl[-\log_2\mathbb P(Z=S)\bigr]=H(Z).
\]
Therefore
\begin{align*}
 H(Z)
 \ge \log_2T+\log_2|\mathcal A|-
     \left(n-\sum_i a_i\right)L-
     \sum_i p_i h(a_i).
\end{align*}
Entropy subadditivity gives $H(Z)\le\sum_i h(p_i)$. Combining the upper and lower bounds for $H(Z)$ gives
\begin{equation}\label{eq:oneA}
 \log_2T+\log_2|\mathcal A|
 \le \left(n-\sum_i a_i\right)L
     +\sum_i\bigl[p_i h(a_i)+h(p_i)\bigr].
\end{equation}
Interchanging $\mathcal A$ and $\mathcal B$ gives
\begin{equation}\label{eq:oneB}
 \log_2T+\log_2|\mathcal B|
 \le \left(n-\sum_i b_i\right)L
     +\sum_i\bigl[q_i h(b_i)+h(q_i)\bigr].
\end{equation}
Adding \eqref{eq:oneA} and \eqref{eq:oneB}, the left-hand side is
\[
 2\log_2T+\log_2|\mathcal A|+\log_2|\mathcal B|=3\log_2T.
\]
The right-hand side is
\[
 \sum_i\Bigl[
 p_i h(a_i)+q_i h(b_i)+h(p_i)+h(q_i)+L(2-a_i-b_i)
 \Bigr].
\]
Since $p_i=a_i(1-b_i)$ and $q_i=b_i(1-a_i)$, the bracket is
$F(a_i,b_i)+L(2-a_i-b_i)$ by \eqref{eq:F}. This proves \eqref{eq:direct}.

<!-- lemma-7: body_md -->
\label{lem:domain}
Let $\rho>1$. Suppose that $(\mathcal A,\mathcal B)$ is a recovering pair of nonempty families on $[n]$ with
$T=|\mathcal A||\mathcal B|>\rho^n$, and suppose that $n$ is minimal among all such counterexamples to the bound $T\le\rho^n$. Let $A$ and $B$ be independent and uniform on $\mathcal A$ and $\mathcal B$, and put
$a_i=\mathbb P(i\in A)$ and $b_i=\mathbb P(i\in B)$. Then, for every $i\in[n]$,
\begin{equation}\label{eq:domain-strict}
 \max\{a_i,1-a_i\}\max\{b_i,1-b_i\}<\rho^{-1}.
\end{equation}
In particular, $(a_i,b_i)\in\mathcal M_\rho$ for every $i$.
<!-- end lemma-7 -->
<!-- proof-9: body_md -->
The minimal $n$ is at least one, since on the empty ground set both nonempty families are $\{\varnothing\}$. Fix $i\in[n]$ and $s,t\in\{0,1\}$. Restrict $\mathcal A$ to sets with indicator $A_i=s$ and $\mathcal B$ to sets with indicator $B_i=t$, and then delete coordinate $i$. Deletion is injective on each restricted family because the deleted bit is fixed. Separately, the $i$-th coordinates of both differences $A\setminus B$ and $B\setminus A$ are determined by the fixed pair $(s,t)$. Hence equality of either difference after deletion implies equality of the same difference before deletion. Applying the recovering property before deletion shows that every nonempty restricted pair is recovering.

By minimality, the product of the two restricted family sizes is at most $\rho^{n-1}$; if one restricted family is empty, the product is zero. Because $A$ and $B$ are uniform, the restricted family sizes are
\[
 |\mathcal A|\,\mathbb P(A_i=s)
 \qquad\text{and}\qquad
 |\mathcal B|\,\mathbb P(B_i=t),
\]
respectively. Their product is therefore
$T\mathbb P(A_i=s)\mathbb P(B_i=t)$, and hence
\[
 T\mathbb P(A_i=s)\mathbb P(B_i=t)\le\rho^{n-1}<T/\rho.
\]
Divide by $T$ and maximize over $s,t$. The left-hand side becomes
$\max\{a_i,1-a_i\}\max\{b_i,1-b_i\}$, proving \eqref{eq:domain-strict}.

Both $Q$ and $W_{\lambda,\tau}$ are symmetric in their two arguments: this follows directly from \eqref{eq:F}, \eqref{eq:mu}, \eqref{eq:Vhat}, and \eqref{eq:W} by interchanging $x$ and $y$. The finite verification described in the computational supplement uses all eight support points in the table, each with mass $v_i/2$, and checks by rational comparison that every support point lies strictly inside $\mathcal M_{\mu_-}$.

For this measure, $|S_\nu|=10$, so Lemma~\ref{lem:param-reduction} reduces the second inequality in \eqref{eq:obstruction} to the $55$ ordered pairs $\lambda,\tau\in S_\nu$ with $\lambda\le\tau$. The finite verification also establishes by rational arithmetic that the first inequality in \eqref{eq:obstruction} and the second inequality for each of these $55$ pairs hold with the margin stated in \eqref{eq:obstruction}. Lemma~\ref{lem:param-reduction} then gives the second inequality for every $0\le\lambda\le\tau\le1$.

Suppose that a finite convex combination
\[
 R(x,y)=\alpha_0Q(x,y)+\sum_{j=1}^r\alpha_jW_{\lambda_j,\tau_j}(x,y),
 \qquad \alpha_j\ge0,\quad \sum_{j=0}^r\alpha_j=1,
\]
satisfied $R(x,y)\le m_-$ for every $(x,y)\in\mathcal M_{\mu_-}$. Since $\nu$ is supported in $\mathcal M_{\mu_-}$, integration would give $\int R\,d\nu\le m_-$. On the other hand, \eqref{eq:obstruction} gives
\[
 \int R\,d\nu
 =\alpha_0\int Q\,d\nu
  +\sum_{j=1}^r\alpha_j\int W_{\lambda_j,\tau_j}\,d\nu
 >m_-+10^{-5},
\]
a contradiction.
