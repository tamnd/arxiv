Let \(X_1,\ldots,X_k\) be independent centered random variables satisfying
$
    \|X_i\|_{\psi_1}\le K,
    \qquad i=1,\ldots,k.

$

    \frac{1}{k}\sum_{i=1}^k X_i
    \right|
    \le

    \left(

    +

    \right),

\label{lem:subexp_mixed_tail}
Let \(\{X_t\}_{t\in T}\) be a random process indexed by
\(T\subseteq\mathbb{R}^n\). Suppose that, for every \(s,t\in T\),
$
    X_t-X_s
    =
    \sum_{i=1}^k Y_i(s,t),
$
where \(Y_1(s,t),\ldots,Y_k(s,t)\) are independent centered random variables
satisfying
$
    \|Y_i(s,t)\|_{\psi_1}
    \lesssim
    \frac{K}{k}\|t-s\|_2,
    i=1,\ldots,k.
$
Then, for every \(u\geq 1\),
$
\mathbb{P}\left(
    |X_t-X_s|
    >
    CK\|t-s\|_2
    \left[
        \sqrt{\frac{u}{k}}
        +
        \frac{u}{k}
    \right]
\right)
\leq
2e^{-u},
$
where \(C>0\) is a universal constant. Consequently,
\(\{X_t\}_{t\in T}\) satisfies the mixed-tail increment condition
\eqref{eq:mixed_tail} with
$
    d_2(t,s)
    =
    \frac{K}{\sqrt{k}}\|t-s\|_2
    \text{ and }
    d_1(t,s)
    =
    \frac{K}{k}\|t-s\|_2.
$

Let \(\{X_t\}_{t\in T}\) be a random process satisfying the mixed-tail
increment condition \eqref{eq:mixed_tail} with respect to semimetrics

\label{lem:gaussian_linear_process}
Let \(T\subseteq\mathbb{R}^m\) be bounded and let
\(g\sim N(\mathbf 0,I_m)\). Let \(P:\mathbb{R}^m\to\mathbb{R}^m\) be a
linear contraction, i.e.,
$
    \|P\mathbf{x}\|_2\leq \|\mathbf{x}\|_2
    \qquad
    \text{for all } \mathbf{x}\in\mathbb{R}^m.
$
Then, for every \(u\geq 1\), with probability at least
\(1-Ce^{-cu}\),

$$
    \sup_{\mathbf{t}\in T}
    \left|
    \langle Pg,\mathbf{t}\rangle
    \right|
    \lesssim
    \gamma_2(T,\|\cdot\|_2)
    +
    \operatorname{rad}(T)\sqrt{u},
$$

where \(c,C>0\) are universal constants.

\label{lem:prev_dual_eq}
Fix any \(\lambda\in\mathbb{R}^m\) and \(T\subseteq\mathbb{R}^m\). Let
\(G\in\mathbb{R}^{k\times m}\) have i.i.d.\ entries distributed as
\(N(0,1/k)\). Then, for every \(u\geq 1\), with probability at least
\(1-Ce^{-cu}\) for some universal constants \(c,C>0\),
\begin{align*}
\sup_{\mathbf{t}\in T}
\left|
\left\langle
(G^\top G-I_m)\lambda,\mathbf{t}
\right\rangle
\right|
\lesssim
\|\lambda\|_2
\left(
\frac{1}{\sqrt{k}}
+
\frac{\sqrt{u}}{k}
\right)
\left[
\gamma_2(T,\|\cdot\|_2)
+
\operatorname{rad}(T)\sqrt{u}
\right].
\end{align*}

<!-- proof-1: body_md -->
The case \(\lambda=0\) is immediate, so assume that
\(\lambda\neq 0\), and set
$
q:=\frac{\lambda}{\|\lambda\|_2}.
$
Write
$
G=\frac{1}{\sqrt{k}}\Phi,
$
where the rows of \(\Phi\) are
\(\phi_1^\top,\ldots,\phi_k^\top\), with
\(\phi_i\overset{\mathrm{i.i.d.}}{\sim}N(\mathbf 0,I_m)\).

For each \(i\), decompose \(\phi_i\) into its component along \(q\) and its
orthogonal component:
$
\phi_i=\gamma_i q+\zeta_i,
$
where
$
\gamma_i:=\langle \phi_i,q\rangle\sim N(0,1), 
\zeta_i:=P_{q^\perp}\phi_i.
$
By Gaussian orthogonal decomposition, \(\gamma_i\) and \(\zeta_i\) are
independent, and
$
\zeta_i\sim N(0,P_{q^\perp}).
$
Moreover, the pairs \((\gamma_i,\zeta_i)\) are independent across \(i\).

For any \(\mathbf{t}\in T\),
\[
\begin{aligned}
\left\langle
(G^\top G-I_m)\lambda,\mathbf{t}
\right\rangle
&=
\|\lambda\|_2
\left[
\frac1k\sum_{i=1}^k
\langle \phi_i,q\rangle
\langle \phi_i,\mathbf{t}\rangle
-
\langle q,\mathbf{t}\rangle
\right] \\
&=
\|\lambda\|_2
\left[
\frac1k\sum_{i=1}^k
\gamma_i
\langle \gamma_i q+\zeta_i,\mathbf{t}\rangle
-
\langle q,\mathbf{t}\rangle
\right] \\
&=
\|\lambda\|_2
\left[
a\langle q,\mathbf{t}\rangle
+
\langle \ell,\mathbf{t}\rangle
\right],
\end{aligned}
\]
where
$
a:=
\frac1k\sum_{i=1}^k(\gamma_i^2-1)
$
and
$
\ell:=
\frac1k\sum_{i=1}^k\gamma_i\zeta_i.
$

Consequently,
\[
\sup_{\mathbf{t}\in T}
\left|
\left\langle
(G^\top G-I_m)\lambda,\mathbf{t}
\right\rangle
\right|
\leq
\|\lambda\|_2
\left[
|a|\operatorname{rad}(T)
+
\sup_{\mathbf{t}\in T}|\langle \ell,\mathbf{t}\rangle|
\right].
\tag{2}
\label{eq:decomp_dual_cert}
\]

We control the two terms separately. Since
\(\gamma_i^2-1\) are independent centered sub-exponential random variables
with uniformly bounded \(\psi_1\)-norm, Bernstein's inequality
gives, for every \(u\geq1\),
$
|a|
\lesssim
\sqrt{\frac{u}{k}}+\frac{u}{k}
$
with probability at least \(1-2e^{-cu}\). Therefore,
\[
|a|\operatorname{rad}(T)
\lesssim
\operatorname{rad}(T)
\left(
\sqrt{\frac{u}{k}}
+
\frac{u}{k}
\right).
\tag{3}
\label{eq:a_general_bound}
\]

We next control \(\ell\). Conditional on
\(\gamma:=(\gamma_1,\ldots,\gamma_k)\), the vector
\(\sum_{i=1}^k\gamma_i\zeta_i\) is centered Gaussian with covariance
$
\left(\sum_{i=1}^k\gamma_i^2\right)P_{q^\perp}.
$
Hence, conditionally on \(\gamma\), we have
$
\ell
\stackrel{d}{=}
\frac{\|\gamma\|_2}{k}P_{q^\perp}g,
g\sim N(\mathbf 0,I_m).
$

To control \(\|\gamma\|_2\), we again apply Bernstein's inequality to the
independent centered sub-exponential random variables
\(\gamma_i^2-1\). With probability at least \(1-2e^{-cu}\),
\[
\left|
\sum_{i=1}^k(\gamma_i^2-1)
\right|
\lesssim
\sqrt{ku}+u.
\]
Therefore,
\[
\|\gamma\|_2^2
=
\sum_{i=1}^k\gamma_i^2
\lesssim
k+\sqrt{ku}+u
\lesssim
k+u,
\]
and hence
\[
\|\gamma\|_2
\lesssim
\sqrt{k}+\sqrt{u}.
\]
It follows that
\[
\frac{\|\gamma\|_2}{k}
\lesssim
\frac1{\sqrt{k}}
+
\frac{\sqrt{u}}{k}.
\]
Therefore, on this event,
\[
\sup_{\mathbf{t}\in T}|\langle\ell,\mathbf{t}\rangle|
\lesssim
\left(
\frac1{\sqrt{k}}
+
\frac{\sqrt{u}}{k}
\right)
\sup_{\mathbf{t}\in T}
|\langle P_{q^\perp}g,\mathbf{t}\rangle|.
\tag{4}
\label{eq:ell_process_bound}
\]

Consider the centered Gaussian process
\[
X_{\mathbf{t}}:=\langle P_{q^\perp}g,\mathbf{t}\rangle,
\qquad \mathbf{t}\in T.
\]
Its canonical metric satisfies
\[
\begin{aligned}
d_X(\mathbf{t},\mathbf{s})
&:=
\left(
\mathbb E|X_{\mathbf{t}}-X_{\mathbf{s}}|^2
\right)^{1/2} \\
&=
\|P_{q^\perp}(\mathbf{t}-\mathbf{s})\|_2
\leq
\|\mathbf{t}-\mathbf{s}\|_2.
\end{aligned}
\tag{5}
\]
Thus, by Lemma \ref{lem:gaussian_linear_process}, with probability at least \(1-Ce^{-cu}\),
\[
\sup_{\mathbf{t}\in T}
|\langle P_{q^\perp}g,\mathbf{t}\rangle|
\lesssim
\gamma_2(T,\|\cdot\|_2)
+
\operatorname{rad}(T)\sqrt{u}.
\tag{6}
\]
Combining this with \eqref{eq:ell_process_bound} yields
\[
\sup_{\mathbf{t}\in T}|\langle\ell,\mathbf{t}\rangle|
\lesssim
\left(
\frac1{\sqrt{k}}
+
\frac{\sqrt{u}}{k}
\right)
\left[
\gamma_2(T,\|\cdot\|_2)
+
\operatorname{rad}(T)\sqrt{u}
\right]
\tag{7}
\]
with probability at least \(1-Ce^{-cu}\).

Finally, note that
\[
\operatorname{rad}(T)
\left(
\sqrt{\frac{u}{k}}
+
\frac{u}{k}
\right)
\leq
\left(
\frac1{\sqrt{k}}
+
\frac{\sqrt{u}}{k}
\right)
\operatorname{rad}(T)\sqrt{u}.
\]
Thus, substituting \eqref{eq:a_general_bound} and \eqref{eq:ell_process_bound}
into \eqref{eq:decomp_dual_cert}, and taking a union bound over the preceding
events, we obtain
\[
\sup_{\mathbf{t}\in T}
\left|
\left\langle
(G^\top G-I_m)\lambda,\mathbf{t}
\right\rangle
\right|
\lesssim
\|\lambda\|_2
\left(
\frac1{\sqrt{k}}
+
\frac{\sqrt{u}}{k}
\right)
\left[
\gamma_2(T,\|\cdot\|_2)
+
\operatorname{rad}(T)\sqrt{u}
\right]
\]
with probability at least \(1-Ce^{-cu}\), for some universal constants
\(C,c>0\). This completes the proof.

Let

\(N(0,1/k)\). Suppose that strong duality holds for the equality-constrained

\operatorname{\operatorname{RPOPT}}_{=}(k)
\leq
\operatorname{\operatorname{OPT}}_{=},
$
where

\lesssim

\left(

+

<!-- proof-2: body_md -->
The primal bound
\(\operatorname{\operatorname{RPOPT}}_{=}(k)\leq \operatorname{\operatorname{OPT}}_{=}\)
follows directly from feasibility preservation. We focus on the dual side.
By strong duality, there exists an optimal dual
multiplier \(\lambda_\star\in\mathbb{R}^m\) such that
$
\operatorname{\operatorname{OPT}}_{=}
=
\inf_{\mathbf{x}\in S}
\left\{
f(\mathbf{x})
+
\left\langle
\lambda_\star,A\mathbf{x}-\mathbf{b}
\right\rangle
\right\}.
$
The randomly projected problem is
$
\operatorname{\operatorname{RPOPT}}_{=}(k)
=
\inf_{\mathbf{x}\in S}
\left\{
f(\mathbf{x}):
GA\mathbf{x}=G\mathbf{b}
\right\}.
$

We choose \(G\lambda_\star\) as a candidate dual solution for
\(\operatorname{\operatorname{RPOPT}}_{=}(k)\). By weak duality,
$$
\begin{aligned}
\operatorname{\operatorname{RPOPT}}_{=}(k)
&\geq
\inf_{\mathbf{x}\in S}
\left\{
f(\mathbf{x})
+
\left\langle
G\lambda_\star,G(A\mathbf{x}-\mathbf{b})
\right\rangle
\right\}
\\
&=
\inf_{\mathbf{x}\in S}
\left\{
f(\mathbf{x})
+
\left\langle
G^\top G\lambda_\star,A\mathbf{x}-\mathbf{b}
\right\rangle
\right\}.
\end{aligned}
$$

Adding and subtracting \(\lambda_\star\), we obtain
$$
\begin{aligned}
\operatorname{\operatorname{RPOPT}}_{=}(k)
&\geq
\inf_{\mathbf{x}\in S}
\Bigg\{
f(\mathbf{x})
+
\left\langle
\lambda_\star,A\mathbf{x}-\mathbf{b}
\right\rangle
+
\left\langle
(G^\top G-I_m)\lambda_\star,
A\mathbf{x}-\mathbf{b}
\right\rangle
\Bigg\}
\\
&\geq
\inf_{\mathbf{x}\in S}
\left\{
f(\mathbf{x})
+
\left\langle
\lambda_\star,A\mathbf{x}-\mathbf{b}
\right\rangle
\right\}
-
\sup_{\mathbf{x}\in S}
\left|
\left\langle
(G^\top G-I_m)\lambda_\star,
A\mathbf{x}-\mathbf{b}
\right\rangle
\right|
\\
&=
\operatorname{\operatorname{OPT}}_{=}
-
\sup_{\mathbf{x}\in S}
\left|
\left\langle
(G^\top G-I_m)\lambda_\star,
A\mathbf{x}-\mathbf{b}
\right\rangle
\right|.
\end{aligned}
$$

We now apply Lemma~\ref{lem:prev_dual_eq} with
$
T:=AS-\mathbf{b}
=
\{A\mathbf{x}-\mathbf{b}:\mathbf{x}\in S\}.
$
For every \(u\geq1\), with probability at least
\(1-Ce^{-cu}\),
\begin{equation}
\label{eq:form_of_eps}
\begin{aligned}
&
\sup_{\mathbf{x}\in S}
\left|
\left\langle
(G^\top G-I_m)\lambda_\star,
A\mathbf{x}-\mathbf{b}
\right\rangle
\right|
\\
&\qquad\lesssim
\|\lambda_\star\|_2
\left(
\frac{1}{\sqrt{k}}
+
\frac{\sqrt{u}}{k}
\right)
\left[
\gamma_2(AS-\mathbf{b},\|\cdot\|_2)
+
\operatorname{rad}(AS-\mathbf{b})\sqrt{u}
\right].
\end{aligned}
\end{equation}

Since the \(\gamma_2\)-functional is translation invariant,
$
\gamma_2(AS-\mathbf{b},\|\cdot\|_2)
=
\gamma_2(AS,\|\cdot\|_2).
$
Furthermore, by the triangle inequality,
$
\operatorname{rad}(AS-\mathbf{b})
=
\sup_{\mathbf{x}\in S}\|A\mathbf{x}-\mathbf{b}\|_2
\leq
\sup_{\mathbf{x}\in S}\|A\mathbf{x}\|_2+\|\mathbf{b}\|_2
=
\operatorname{rad}(AS)+\|\mathbf{b}\|_2.
$
Substituting these bounds into \eqref{eq:form_of_eps}, we obtain
$$
\sup_{\mathbf{x}\in S}
\left|
\left\langle
(G^\top G-I_m)\lambda_\star,
A\mathbf{x}-\mathbf{b}
\right\rangle
\right|
\lesssim
\|\lambda_\star\|_2
\left(
\frac{1}{\sqrt{k}}
+
\frac{\sqrt{u}}{k}
\right)
\left[
\gamma_2(AS,\|\cdot\|_2)
+
\bigl(
\operatorname{rad}(AS)+\|\mathbf{b}\|_2
\bigr)\sqrt{u}
\right].
$$
Therefore,
$
\operatorname{\operatorname{OPT}}_{=}-\epsilon(k,u)
\leq
\operatorname{\operatorname{RPOPT}}_{=}(k),
$
where
$$
\epsilon(k,u)
\lesssim
\|\lambda_\star\|_2
\left(
\frac{1}{\sqrt{k}}
+
\frac{\sqrt{u}}{k}
\right)
\left[
\gamma_2(AS,\|\cdot\|_2)
+
\bigl(
\operatorname{rad}(AS)+\|\mathbf{b}\|_2
\bigr)\sqrt{u}
\right].
$$
Together with
\(\operatorname{\operatorname{RPOPT}}_{=}(k)\leq\operatorname{\operatorname{OPT}}_{=}\),
this proves the result.

Since
$
\Delta_n\subseteq B_1^n,
$
it suffices for the upper bound to estimate
\(\gamma_1(B_1^n,\|\cdot\|_2)\).
Let
$
e_r(B_1^n,\|\cdot\|_2)
:=
\inf\left\{
\varepsilon>0:
N(B_1^n,\|\cdot\|_2,\varepsilon)
\leq 2^{r-1}
\right\}
$
denote the entropy numbers of \(B_1^n\) under the Euclidean metric.
By Sch\"utt~\cite{schutt1984entropy},
\[
e_r
\asymp
\begin{cases}
1, & r\lesssim \log n,\\[1mm]
\displaystyle
\sqrt{\frac{\log(1+n/r)}{r}},
& \log n\lesssim r\lesssim n,\\[3mm]
\displaystyle
n^{-1/2}2^{-r/n},
& r\gtrsim n.
\end{cases}
\]
Using the entropy bound \cite{van2014probability} for the chaining functional,
$
\gamma_1(T,d)
\lesssim
\sum_{j\geq0}2^j e_{2^j}(T,d),
$
we obtain, since \(\Delta_n\subseteq B_1^n\),
$$
\gamma_1(\Delta_n,\|\cdot\|_2)
\leq
\gamma_1(B_1^n,\|\cdot\|_2)
\lesssim
\sum_{j\geq0}
2^j e_{2^j}(B_1^n,\|\cdot\|_2).
$$
We split the sum according to the three regimes of the entropy numbers.

First, when \(2^j\lesssim \log n\), we have
\(e_{2^j}\asymp 1\), and hence
\[
\sum_{2^j\lesssim\log n}
2^j e_{2^j}
\lesssim
\sum_{2^j\lesssim\log n}2^j
\lesssim
\log n
\lesssim
\sqrt n.
\]

Next, suppose
$
\log n\lesssim 2^j\lesssim n.
$
Then
$
e_{2^j}
\asymp
\sqrt{
\frac{\log(1+n/2^j)}{2^j}
},
$
so that
$
2^j e_{2^j}
\asymp
\sqrt{
2^j\log(1+n/2^j)
}.
$
Let \(J\) satisfy \(2^J\asymp n\) and write \(j=J-\ell\). Then
$
2^j\asymp \frac{n}{2^\ell},
\log\left(1+\frac{n}{2^j}\right)
\lesssim
\ell+1.
$
Therefore,
\begin{align*}
\sum_{\log n\lesssim2^j\lesssim n}
2^j e_{2^j}
&\lesssim
\sqrt n
\sum_{\ell\geq0}
\sqrt{\frac{\ell+1}{2^\ell}} \lesssim
\sqrt n,
\end{align*}
since
$
\sum_{\ell\geq0}
\sqrt{\frac{\ell+1}{2^\ell}}
<\infty.
$

Finally, when \(2^j\gtrsim n\),
$
e_{2^j}
\asymp
n^{-1/2}2^{-2^j/n}.
$
Thus,
$
2^j e_{2^j}
\asymp
\sqrt n
\left(\frac{2^j}{n}\right)
2^{-2^j/n}.
$
Again writing \(2^j/n\asymp 2^\ell\), we obtain
\[
\sum_{2^j\gtrsim n}
2^j e_{2^j}
\lesssim
\sqrt n
\sum_{\ell\geq0}
2^\ell 2^{-c2^\ell}
\lesssim
\sqrt n.
\]
Combining the three regimes gives
$
\gamma_1(\Delta_n,\|\cdot\|_2)
\lesssim
\sqrt n.
$

For the reverse inequality, we use the standard entropy lower bound \cite{van2014probability}
\[
\gamma_1(T,d)
\gtrsim
\sup_{\varepsilon>0}
\varepsilon\log N(T,d,\varepsilon).
\]
Thus, it suffices to exhibit a scale \(\varepsilon\) at which
\(\Delta_n\) has exponentially large covering number.

To this end, by the Varshamov-Gilbert bound \cite{tsybakov2009introduction}, there exists a set
\(\mathcal C\subseteq\{0,1\}^n\) such that
$
|\mathcal C|\geq e^{cn}, 
d_H(\sigma,\tau)\geq cn
$
for all distinct \(\sigma,\tau\in\mathcal C\). For each
\(\sigma\in\mathcal C\) and some constant $c$, define
$
x^\sigma:=\frac{\sigma}{n}.
$
Since
$
\sum_{i=1}^n x_i^\sigma
=
\frac{\|\sigma\|_1}{n}
\leq 1,
$
we have \(x^\sigma\in\Delta_n\). Moreover, for distinct
\(\sigma,\tau\in\mathcal C\),
$
\|x^\sigma-x^\tau\|_2
=
\frac{\sqrt{d_H(\sigma,\tau)}}{n}
\gtrsim
\frac{1}{\sqrt n}.
$
Hence \(\Delta_n\) contains an exponentially large packing at scale
\(1/\sqrt n\), and therefore
$
\log N\left(
\Delta_n,\|\cdot\|_2,\frac{c}{\sqrt n}
\right)
\gtrsim n.
$
Applying the entropy lower bound at
\(\varepsilon\asymp n^{-1/2}\) gives
$
\gamma_1(\Delta_n,\|\cdot\|_2)
\gtrsim
\frac{1}{\sqrt n}\,n
=
\sqrt n.
$
Combining this with the upper bound yields
$
\gamma_1(\Delta_n,\|\cdot\|_2)
\asymp
\sqrt n.
$
