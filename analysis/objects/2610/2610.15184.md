<!-- lemma-1: body_md -->
\label{lem:coh}
Let $c_m(\bm{q})\triangleq\bm{a}_m^{(0)H}\bm{a}_m(\bm{q})/N$ and
$\bm{r}_m(\bm{q})\triangleq\bm{a}_m(\bm{q})-c_m(\bm{q})\bm{a}_m^{(0)}$. Then
$\bm{a}_m^{(0)H}\bm{r}_m(\bm{q})=0$ and
$\|\bm{r}_m(\bm{q})\|^2=N(1-|c_m(\bm{q})|^2)$. Under \textup{(A)}: if
$\eta<\pi/2$ then $|c_m(\bm{q})|\ge\cos\eta>0$ and
$\|\bm{r}_m(\bm{q})\|^2\le N\sin^2\eta$; for every $\eta\ge0$,
$\|\bm{r}_m(\bm{q})\|^2\le N\eta^2$.

<!-- theorem-1: body_md -->
\label{thm:rank1}
Let $\bm{c}(\bm{q})=[c_1(\bm{q}),\ldots,c_M(\bm{q})]^T$ and
$\bm{r}(\bm{q})=[\bm{r}_1(\bm{q});\ldots;\bm{r}_M(\bm{q})]$. For \emph{every}
$\eta\ge0$,
\begin{equation}\label{eq:decomp}
\bm{a}(\bm{q})=\bm{U}\bm{c}(\bm{q})+\bm{r}(\bm{q}),\qquad \bm{U}^H\bm{r}(\bm{q})=\bm0,
\end{equation}
so $\bm{R}_g=\bm{U}\bm{C}\bm{U}^H+\bm{X}+\bm{X}^H+\bm{R}_{\rm dark}$ with $\bm{C}=\langle \bm{c}\bm{c}^H \rangle_{\mathcal{G}}$,
$\bm{X}=\langle \bm{U}\bm{c}\,\bm{r}^H \rangle_{\mathcal{G}}$, and
$\bm{R}_{\rm dark}=\langle \bm{r}\bm{r}^H \rangle_{\mathcal{G}}=\bm{P}_{\rm dark}\bm{R}_g\bm{P}_{\rm dark}$. Cauchy interlacing gives the
exact bounds $\lambda_i(\bm{R}_g)\ge N\lambda_i(\bm{C})$ for $i\le M$ and
\begin{equation}\label{eq:interlace}
\lambda_{M+1}(\bm{R}_g)\ \le\ \lambda_1(\bm{R}_{\rm dark})\ \le\ \operatorname{tr}\bm{R}_{\rm dark}
= MN(1-\bar\kappa),
\end{equation}
with $\bar\kappa=\operatorname{tr}\bm{C}/M\in[0,1]$ the mean squared coherence.
Hence $r_{\rm eff}(\bm{R}_g)\le M$ whenever $\bar\kappa>2M/(2M+1)$. Under
\textup{(A)} with $\eta<\pi/2$, Lemma~\ref{lem:coh} gives
$\lambda_1(\bm{R}_g)\ge N\cos^2\eta$ and
$\lambda_{M+1}(\bm{R}_g)\le MN\sin^2\eta$, so
\begin{equation}\label{eq:suff}
\tan^2\eta<\tfrac1{2M}\ \Longrightarrow\ r_{\rm eff}(\bm{R}_g)\le M
\ (M{=}8{:}\ \eta<0.245).
\end{equation}

<!-- remark-1: body_md -->
\label{rem:perblock}
With per-block references $\bm{q}_{0,m}$, $\bm{U}$ still obeys
$\bm{U}^H\bm{U}=N\bm{I}_M$ and all proofs are unchanged, so each sub-array
may use its own local reference into $\mathcal{G}$, with a per-block
$\eta_m=2\pi R_a\Delta\theta_m/\lambda$: this covers sub-arrays
steered toward \emph{distinct} angular intervals of $\mathcal{G}$.
Sub-arrays protecting physically \emph{different} regions $\mathcal{G}_m$
define a different covariance, outside this model.

<!-- proposition-1: body_md -->
\label{prop:slepian}
Let a sub-array be linear and in its far field with respect to
$\mathcal{G}$, with the direction-cosine offset of $\bm{q}\in\mathcal{G}$ uniform
on $[-\Delta,\Delta]$, $\Delta\le\Delta\theta_{\rm sep}$. Then the
diagonal block is unitarily equivalent to the sinc-kernel (discrete
prolate, DPSS) matrix~\cite{ref:slepian} with
space--bandwidth parameter $c=\pi D_a\Delta/\lambda\le\eta$
\emph{exactly}, with equality when the modeled interval spans
$[-\Delta\theta_{\rm sep},\Delta\theta_{\rm sep}]$, and eigenvalues
$\frac{N\pi}{2c}\lambda_k(c)(1+o_N(1))$, $\lambda_k(c)$ the
Slepian--Pollak eigenvalues. Consequently:
\begin{enumerate}[label=(\roman*),leftmargin=2.25em,itemsep=0.4em,topsep=0.5em]
\item The per-sub-array
Shannon number (the classical count
$(2D_a/\lambda)\sin\theta$ of this
aperture) is $\sum_k\lambda_k(c)=2c/\pi\le2\eta/\pi$, equal to one
at $c=\pi/2$: the design rule
$D_a\Delta\theta_{\rm sep}<\lambda/2$ means \emph{fewer than one
spatial DoF per sub-array}.
\item The half-power block rank is one
iff $\lambda_1(c)/\lambda_0(c)<1/2$, i.e., iff
$c<c^\star=2.2772$ (finite $N$: $c^\star{=}1.99$ and $2.19$ for
$N_x{=}8$ and $27$ elements per side).
\item For a
square panel of side $D_a$ and a disk region the jinc kernel of the
disk band restricted to that \emph{square}~\cite{ref:slepian1964}
replaces the sinc, with $c=\pi D_a\Delta/\lambda=\eta/\sqrt2$,
Shannon number $c^2/\pi$ and threshold
$c^\star_{\square}=2.5227$ ($2.21$, $2.43$ for $N_x{=}8,27$), its
first excited level an exactly degenerate \emph{pair}.
\item For weakly coupled blocks (large baseline) the spectrum of
$\bm{R}_g$ is the union of the block spectra, so with a common $c$,
$r_{\rm eff}(\bm{R}_g)=M$ iff $c<c^\star$; unequal $c_m$ spread the transition
over the per-block thresholds.
\end{enumerate}
For non-negligible coupling, Theorem~\ref{thm:rank1}
instead supplies a coupling-independent upper bound through the
dark-subspace compression; no monotonicity of $r_{\rm eff}$ in the
coupling is claimed or needed.

<!-- corollary-1: body_md -->
\label{cor:invq}
Let $\widetilde\bm{R}=\bm{R}_g+\delta\bm{I}$,
$\delta=\epsilon\lambda_1(\bm{R}_g)$, $\epsilon\ll1$ (loading /
finite-sample floor),
$\rho_\epsilon=\bm{a}_t^H\widetilde\bm{R}^{-1}\bm{a}_t$ ($\rho_\epsilon\to\rho^\star_{\rm avg}$ as
$\delta\to0$), and $\bm\Pi_M$ the
top-$M$ eigenprojector of $\bm{R}_g$. Then
$\rho_\epsilon=\rho_{\rm ill}+\rho_{\rm dark}$ with
$\rho_{\rm dark}\le\|\bm\Pi_M^{\perp}\bm{a}_t\|^2/\delta$, and the dark
projection is
\begin{equation}\label{eq:darkproj}
\|\bm{P}_{\rm dark}\bm{a}_t\|^2=N\textstyle\sum_{m}(1-|g_m|^2),\quad
g_m\triangleq\tfrac1N\bm{a}_m^{(0)H}\bm{a}_{t,m},
\end{equation}
with $g_m$ the target--reference coherence at sub-array $m$
($|g_m|=1$ at $N=1$; the aperture pattern at the offset
$\Delta\bm{u}_m=\hat\bm{u}_m(\bm{q}_t)-\hat\bm{u}_m(\bm{q}_0)$ for large $N$).
\begin{enumerate}[label=(\roman*),leftmargin=2.25em,itemsep=0.6em,topsep=0.5em]
\item \emph{Separated target}: in the weak-coupling regime
$\bm\Pi_M\approx\bm{P}_{\rm ill}$ ($\bm{C}\approx\bm{I}$, large baseline), if
$D_a|\Delta\bm{u}_m|/\lambda\gtrsim1$ for all $m$ (every sub-array
resolves the target from $\mathcal{G}$), then
$|g_m|^2\ll1$, $\|\bm{P}_{\rm dark}\bm{a}_t\|^2=M_{\rm tot}(1-o(1))$, and once the dark
projection lies below the loading floor,
\begin{equation}\label{eq:MoverEps}
\rho_\epsilon\approx
\frac{\sum_m|g_m|^2+\epsilon^{-1}\sum_m(1-|g_m|^2)}{\lambda_1(\bm{C})}
\longrightarrow \frac{M}{\epsilon\lambda_1(\bm{C})},
\end{equation}
i.e., $\rho_\epsilon^{\rm dB}\approx10\log_{10}M+10\log_{10}
\epsilon^{-1}$ when $\lambda_1(\bm{C})\approx1$; $\lambda_1(\bm{C})<1$
raises the ceiling.
\item \emph{In-cone target}: if $|\Delta\bm{u}_m|\ll\lambda/D_a$ for
all $m$ (range-only separation), then $|g_m|\approx1$ and, for
targets whose modal coefficients follow the concentration envelope
$|b_k|^2=O(\lambda_k)$, block mode $k$ carries dark energy
$O(\lambda_k(c))$ and contributes $O(1)$ regardless of $\epsilon$:
$\rho_\epsilon=O(M K_\epsilon)$,
$K_\epsilon=\#\{k:\lambda_k\ge\epsilon\lambda_0\}=O(\log
\epsilon^{-1})$~\cite{ref:landauWidom1980} (no $1/\epsilon$
amplification, for any $N$).
\end{enumerate}

<!-- proposition-2: body_md -->
\label{prop:dual}
Let $\mathcal{P}(\mathcal{G})$ be the probability measures on $\mathcal{G}$,
$\bm{R}_\mu=\int\bm{a}\bm{a}^H d\mu$, and
${\rm SIR}_{\rm wc}(\bm{w})=|\bm{a}_t^H\bm{w}|^2/\max_{\bm{q}\in\mathcal{G}}
|\bm{a}(\bm{q})^H\bm{w}|^2$. For every $\bm{w}$ and every $\mu\in\mathcal{P}(\mathcal{G})$
with $\bm{w}^H\bm{R}_\mu\bm{w}>0$, a supremum dominates a mean, so
\begin{equation}\label{eq:wcbound}
{\rm SIR}_{\rm wc}(\bm{w})\le
\frac{|\bm{a}_t^H\bm{w}|^2}{\bm{w}^H\bm{R}_\mu\bm{w}},
\ \text{hence}\ \ 
{\rm SIR}_{\rm wc}(\bm{w})\le\rho^\star_{\rm avg}
\end{equation}
the second at uniform $\mu$, attained by $\bm{w}\propto\bm{R}_g^{-1}\bm{a}_t$
when $\bm{R}_g\succ0$. No inverse of a general $\bm{R}_\mu$ appears:
$\mathcal{P}(\mathcal{G})$ contains Diracs, for which $\bm{R}_\mu$ is singular.
So $\rho^\star_{\rm avg}$ is the exact average-leakage optimum and a one-sided
upper bound on worst-case SIR, not a value we claim is attained.
Since~\eqref{eq:decomp} is pointwise in $\bm{q}$, the same structural
rank restriction applies to $\bm{R}_\mu$ under any probability
weighting of $\mathcal{G}$.

<!-- corollary-2: body_md -->
\label{cor:agnostic}
\label{cor:baseline}
\begin{enumerate}[label=(\roman*),leftmargin=2.25em,itemsep=0.6em,topsep=0.5em]
\item The bounds of Theorem~\ref{thm:rank1} and the thresholds of
Proposition~\ref{prop:slepian} depend on
$(D_a,\Delta\theta_{\rm sep},\lambda)$, not on $D$: the sufficient
collapse condition, and hence $r_{\rm eff}\le M$ wherever it holds, is
baseline-independent. The ceiling~\eqref{eq:MoverEps} is
baseline-independent in the separated-target regime ($|g_m|\ll1$);
otherwise $D$ enters $\rho_\epsilon$ only through $g_m(D)$.
\item $r_{\rm eff}\le M$ is a property of $\bm{R}_g$ (and of every $\bm{R}_\mu$),
not of the beamformer: MVDR, MMSE, leakage-constrained MRT and the
worst-case optimum of~\eqref{eq:socp} (any
$\bm{w}=f(\bm{R}_\mu,\bm{a}_t)$) face the same $M$-dimensional illuminated
subspace and the same ceilings.
\end{enumerate}
