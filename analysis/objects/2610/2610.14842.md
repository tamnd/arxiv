<!-- thm-1: body_md -->
\label{thm:mainexp}
Let $\Omega\subset\mathbb{R}^d$ be a measurable set of positive and finite measure. Then the space $L^p(\Omega)$, $p>2$, does not admit any unconditional Schauder frame of exponentials, i.e.\ of the form 
$\{(e_\lambda, e_\lambda^*)\}$, $\lambda\in\Lambda$,
where $\Lambda\subset\mathbb{R}^d$ is any countable set
and $\{e_\lambda^*\}$ are arbitrary elements of the dual 
space $(L^p(\Omega))^*$.
<!-- end thm-1 -->
<!-- lemma-1: body_md -->
\label{lem:unc_criterion}
Let $X$ be a reflexive Banach space, and $\{x_n\}_{n=1}^\infty \subset X$. Assume that there is a constant $K$ such that for any $N$ and arbitrary scalars $\{\theta_n\}$ with $|\theta_n|\leqslant 1$, 
\begin{equation}
\label{eq:weakly_unconditionally_convergent}
\Big\| \sum_{n=1}^N \theta_n x_n \Big\| \leqslant K.
\end{equation}
Then the series $\sum_{n=1}^\infty   x_n$ converges unconditionally in $X$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Indeed, the condition \eqref{eq:weakly_unconditionally_convergent} implies that the series  $\sum_{n=1}^\infty x_n$ is weakly unconditionally convergent (see \cite[II.D.3]{Woj91}). On the other hand, the space $X$ being reflexive does not contain any subspace isomorphic to $c_0$ (see \cite[Theorem II.A.14]{Woj91}). Hence every weakly unconditionally convergent series in $X$ is unconditionally convergent (see \cite[Proposition II.D.5]{Woj91}).
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:unc_constant}
Let $\{(u_j, u_j^*)\}_{j=1}^\infty$ be an unconditional Schauder frame in 
a Banach space $X$.  Then there exists a constant $K$ such that for any 
$x\in X$ and for any
sequence of scalars $\{\theta_j\}$ satisfying  $|\theta_j|\leqslant 1$, we have
\begin{equation}
	\label{eqtheta}
	\Big\| \sum_{j=1}^\infty \theta_j u_j^*(x) u_j \Big\| \leqslant K \|x\|
\end{equation}
and the series in \eqref{eqtheta} converges unconditionally.

\label{lem:cotype}
Let $1\leqslant p \leqslant 2$. There exists a positive constant $C_p$ such that for any finite 
sequence of functions $f_1, f_2, \ldots, f_N \in L^p(\Omega)$ we have 
\begin{equation}
\label{eq:cotype}
\Big( \sum_{j=1}^N \|f_j\|_p^2 \Big)^{1/2}\leqslant C_p \, \mathbb{E} \Big\|\sum_{j=1}^N \varepsilon_j f_j\Big\|_p.
\end{equation}

<!-- definition-1: body_md -->
Let $X$ be a separable Banach space with dual space $X^*$. 
A sequence of elements $\{(u_n, u_n^*)\} \subset X\times X^*$ is called an \emph{approximate Schauder frame} for $X$ if for any $x\in X$ the series
\begin{equation}\label{eq:approx_frame}
	Sx = \sum_n u_n^*(x) u_n
\end{equation}
converges in $X$ and defines a bounded and invertible linear operator $S: X \to X$. If additionally the series \eqref{eq:approx_frame} converges unconditionally for every $x\in X$, then the system 
$\{(u_n, u_n^*)\}$ is called an \emph{unconditional approximate Schauder frame}.

\label{lem:approxframe}
If $\{(u_n, u_n^*)\}$ is an 
unconditional approximate Schauder frame for $X$, 
then $\{(u_n, (S^{-1})^* u_n^*)\}$
forms an unconditional Schauder frame for $X$.

<!-- lem-2: body_md -->
\label{lem:perturb}
Let $\Omega\subset\mathbb{R}^d$ be a bounded measurable set of positive measure, and assume that $\{(e_{\lambda_n}, g_{n}^*)\}$ forms an unconditional Schauder frame in $L^p(\Omega)$, $1 < p < + \infty$. There is $\eta > 0$ such that if a sequence $\{\mu_n\} \subset \mathbb{R}^d$ satisfies $|\mu_n-\lambda_n| \leqslant \eta$ for all $n$, then the system $\{(e_{\mu_n}, g_{n}^*)\}$ forms an unconditional approximate  Schauder frame in $L^p(\Omega)$.
<!-- end lem-2 -->
<!-- proof-2: body_md -->
We assume that $\Omega\subset [-R,R]^d$ and that the system $\{(e_{\lambda_n}, g_{n}^*)\}$ forms a $K$-unconditional Schauder frame in the space $L^p(\Omega)$. Let  $\{\mu_n\} \subset \mathbb{R}^d$ be an arbitrary sequence satisfying $|\mu_n-\lambda_n| \leqslant \eta$ for all $n$.

We will use the Taylor series expansion of the exponential function in the form
\begin{equation}
	\label{eq:3.1}
	e_\lambda(x) = \sum_\alpha \frac{(2\pi i)^{|\alpha|}}{\alpha!} \lambda^\alpha  x^\alpha,
	\quad x   \in \mathbb{R}^d,
\end{equation}
where the sum goes over all multi-indices $\alpha$ with nonnegative integer coordinates. As usual, for a multi-index $\alpha=(\alpha_1, \alpha_2, \ldots \alpha_d)$ we denote
\begin{equation}
	x^\alpha = x_1^{\alpha_1} \ldots x_d^{\alpha_d}, \quad
	\alpha! = \alpha_1! \ldots \alpha_d!, \quad 
	|\alpha| = \alpha_1 + \ldots + \alpha_d.
\end{equation}

Let now $\theta_n$ be arbitrary scalars with $|\theta_n|\leqslant 1$, and such that only finitely many $\theta_n$ are nonzero. Given $f\in L^p(\Omega)$, we have
\begin{align}
	& \sum_n \theta_n g_n^*(f) (e_{\mu_n}(x) - e_{\lambda_n}(x)) = \sum_n \theta_n g_n^*(f) e_{\lambda_n}(x)(e_{\mu_n-\lambda_n}(x) - 1) \label{eq:3.2.5}	 \\ 
		& \qquad = \sum_n \theta_n g_n^*(f) e_{\lambda_n}(x) \sum_{\alpha\neq 0} \frac{(2\pi i)^{|\alpha|}}{\alpha!}(\mu_n-\lambda_n)^{\alpha} x^\alpha. \label{eq:3.3}
\end{align}
Here we used the Taylor expansion \eqref{eq:3.1} in order to pass from \eqref{eq:3.2.5} to \eqref{eq:3.3}. The sum over $\alpha$ in \eqref{eq:3.3} converges uniformly on the bounded set $\Omega$.

Applying the triangle inequality and the estimate $|x^\alpha|\leqslant R^{|\alpha|}$ for $x\in\Omega$, we obtain that the $L^p(\Omega)$ norm of the left hand side of \eqref{eq:3.2.5} does not exceed
\begin{align}
	&\sum_{\alpha\neq 0} \Big\|\frac{(2\pi i)^{|\alpha|}}{\alpha!} x^\alpha   \sum_n (\mu_n-\lambda_n)^\alpha \theta_n g_n^*(f)e_{\lambda_n}(x)  \Big\|_p\\
	& \qquad\leqslant \sum_{\alpha\neq 0} \frac{(2\pi R)^{|\alpha|}}{\alpha!}   \Big\| \sum_n (\mu_n-\lambda_n)^\alpha \theta_n g_n^*(f)e_{\lambda_n} \Big\|_p.\label{eq:3.5}
\end{align}
Since $|(\mu_n-\lambda_n)^\alpha|\leqslant \eta^{|\alpha|}$, the $K$-unconditionality of the system $\{(e_{\lambda_n}, g_n^*)\}$ implies that the quantity \eqref{eq:3.5} is not greater than
\begin{equation}
	\label{eq:3.6}
	K \|f\|_p \sum_{\alpha\neq 0} \frac{(2\pi \eta R)^{|\alpha|}}{\alpha!}   = K  (e^{2\pi d \eta R} - 1) \| f \|_p.
\end{equation}

Thus, given $\varepsilon>0$, by choosing $\eta = \eta(K, R,d,\varepsilon) > 0$ sufficiently small, we can make the right hand side of \eqref{eq:3.6} not greater than $\varepsilon \|f\|_p$. We conclude that
\begin{equation}
	\label{eq:3.7}
	\Big\| \sum_n \theta_n g_n^*(f) (e_{\mu_n} - e_{\lambda_n})\Big\|_p\leqslant \varepsilon \|f\|_p 
\end{equation}
for any $f \in L^p(\Omega)$ and for arbitrary scalars $\theta_n$ with $|\theta_n| \leqslant 1$, and such that only finitely many $\theta_n$ are nonzero.

We can now finish the proof by standard arguments. 
Indeed, by \eqref{eq:3.7} and
 Lemma~\ref{lem:unc_criterion},  the series
$\sum_n g_n^*(f) (e_{\mu_n} - e_{\lambda_n})$
converges unconditionally in $L^p(\Omega)$.
Clearly, the series $\sum_n  g_n^*(f) e_{\lambda_n}$ is also unconditionally convergent, 
since the system $\{(e_{\lambda_n}, g_{n}^*)\}$ forms an unconditional Schauder frame in $L^p(\Omega)$. It follows that the series
\begin{equation}
	Sf = 	\sum_n  g_n^*(f) e_{\mu_n}
\end{equation}
converges unconditionally in $L^p(\Omega)$, and defines a bounded linear operator  $S: L^p(\Omega) \to L^p(\Omega)$. Moreover, \eqref{eq:3.7} gives $\|S - I\| \leqslant \varepsilon$, where $I$ is the identity operator. Hence, by choosing $0 < \varepsilon < 1$, we obtain that the operator $S$ is invertible. This means that  $\{(e_{\mu_n}, g_n^*)\}$ forms an unconditional approximate Schauder frame in $L^p(\Omega)$.

<!-- lem-3: body_md -->
\label{lem:torus}
Assume that $\Omega\subset\mathbb{R}^d$ is a bounded, measurable set of positive measure, and that $\{(e_{\lambda_n}, g_{n}^*)\}$ forms an unconditional Schauder frame in $L^p(\Omega)$, $1 < p < + \infty$. Then there exists a measurable set $E \subset \mathbb{T}^d$ of positive measure, and a sequence $\{e^*_m\} \subset L^{p}(E)^*$, such that $\{(e_m, e_m^*)\}$, $m\in\mathbb{Z}^d$, forms an unconditional Schauder frame in $L^p(E)$.
<!-- end lem-3 -->
<!-- proof-3: body_md -->
Let $\eta$ be the number given by Lemma~\ref{lem:perturb}, and
choose a small $\gamma>0$. For each $n$, let $\mu_n$ be a closest point to $\lambda_n$ in the lattice $\gamma\mathbb{Z}^d$. If $\gamma $  is chosen sufficiently small, then we get
 $| \mu_n - \lambda_n| \leqslant \eta$ for all $n$. Hence,
by Lemma~\ref{lem:perturb}, the system $\{(e_{\mu_n}, g_{n}^*)\}$ forms an unconditional approximate Schauder frame in $L^p(\Omega)$. In turn, an application of Lemma~\ref{lem:approxframe} gives us a sequence $\{h^*_n\} \subset L^p(\Omega)^*$ such that the system $\{(e_{\mu_n}, h_n^*)\}$ forms an unconditional Schauder frame in $L^p(\Omega)$.
	
It is possible that certain points of the lattice $\gamma\mathbb{Z}^d$ appear in the sequence $\{\mu_n\}$ several times. However by grouping the summands in the unconditionally convergent expansion
$f = \sum_n h_n^*(f) e_{\mu_n}$, and adding, if necessary, zeros as coefficient functionals, we get an unconditional Schauder frame of the form $\{(e_{\gamma m}, h_m^*)\}$, $m\in\mathbb{Z}^d$, for the space $L^p(\Omega)$.



Finally, after rescaling by $\gamma$, we obtain a new system $\{(e_m, e_m^*)\}$, $m\in\mathbb{Z}^d$, which forms an unconditional Schauder frame in $L^p(E)$, where $E := \gamma \Omega$. To be more precise, the new coefficient functionals $\{e_m^*\}$ are given by $e_m^* = U^* (h_m^*)$, where $U: L^p(E) \to L^p(\Omega)$ is the operator defined by $(Uf)(x) = f(\gamma x)$. 
It remains to notice that if $\gamma$ is sufficiently small then we get $ E \subset[-\frac12, \frac12]^d$, hence $E$ may be identified with a subset of $\mathbb{T}^d$.

<!-- lem-4: body_md -->
\label{lem:extension}
For every $f\in L^p(E)$, the series
\begin{equation}
	\label{eq:Af}
	Af = \sum_{m\in\mathbb{Z}^d} e_m^*(f) e_m
\end{equation}
converges unconditionally in $L^p(\mathbb{T}^d)$. Moreover, 
\begin{equation}
	\label{eq:est_extension}
	\Big\|\sum_{m\in\mathbb{Z}^d} \theta_m e_m^*(f) e_m  \Big\|_{L^p(\mathbb{T}^d)}\leqslant K m(E)^{-1/p}  \|f\|_{L^p(E)}
\end{equation}
for every choice of scalars $\theta_m$ with $|\theta_m|\leqslant 1$.
<!-- end lem-4 -->
<!-- proof-4: body_md -->
Fix a function $f \in L^p(E)$, and let
$\theta_m$ be a sequence of scalars with $|\theta_m|\leqslant 1$, 
and such that only finitely many $\theta_m$ are nonzero. Define
\begin{equation}
\varphi(x) = \sum_{m\in\mathbb{Z}^d} \theta_m e_m^*(f) e_m(x), \quad x \in \mathbb{T}^d,
\end{equation}
then $\varphi$ is a trigonometric polynomial. Using \eqref{eq:unctorus_theta}, for any $t \in \mathbb{T}^d$ we have
\begin{equation}
\int_{E} |\varphi(x-t)|^p dx 
= \Big\| \sum_{m\in\mathbb{Z}^d} \theta_m 
e^{-2\pi i \langle m , t \rangle}
e_m^*(f) e_m\Big\|^p_{L^p(E)} \leqslant K^p \|f\|^p_{L^p(E)}.
\end{equation}
Integrating this inequality with respect to the parameter $t \in \mathbb{T}^d$, we obtain
\begin{equation}
	m(E) \|\varphi\|_{L^p(\mathbb{T}^d)}^p = \int_E  \int_{\mathbb{T}^d}|\varphi(x-t)|^p dt dx = \int_{\mathbb{T}^d}\int_E |\varphi(x-t)|^p dx dt\leqslant K^p \|f\|^p_{L^p(E)}.
\end{equation}
Thus, the estimate \eqref{eq:est_extension} holds
for any $f \in L^p(E)$ and any finitely supported sequence  $\{ \theta_m \}$ with $|\theta_m|\leqslant 1$.
An application of Lemma~\ref{lem:unc_criterion} then implies the  unconditional convergence of the series \eqref{eq:Af}. In turn, this allows us to extend the estimate \eqref{eq:est_extension} from finitely supported sequences $\{\theta_m\}$ to all sequences with $|\theta_m|\leqslant 1$.

The dual operator $A^* : L^q(\mathbb{T}^d) \to L^q(E)$ is given by
\begin{equation}
\label{eq:Astarg}
A^*g = \sum_{m\in\mathbb{Z}^d}  \widehat{g}(-m) e_m^*, \quad g \in L^q(\mathbb{T}^d),
\end{equation}

\begin{equation}

	\Big\|\sum_{m\in\mathbb{Z}^d} \theta_m \widehat{g}(-m) e_m^* \Big\|_{L^q(E)}\leqslant K m(E)^{-1/p}  \|g\|_{L^q(\mathbb{T}^d)}
\end{equation}
for every choice of scalars $\theta_m$ with $|\theta_m|\leqslant 1$.
