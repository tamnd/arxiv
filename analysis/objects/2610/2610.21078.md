<!-- theorem-1: body_md -->
\label{main}
Let $(M,g,F)$ be a nontrivial, complete, steady gradient Yamabe soliton. 
Then it is either isometric to a rotationally symmetric manifold
\[
 ([0,+\infty),dr^2)
 \times_{|\nabla F|}
 (\mathbb{S}^{n-1},\bar g_S),
\]
where $\bar g_S$ is a round metric on $\mathbb{S}^{n-1}$, with its scale chosen as above, or
a manifold with zero scalar curvature.
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
Any nontrivial complete steady gradient Yamabe soliton that is not scalar-flat is rotationally symmetric.
<!-- end corollary-1 -->
<!-- remark-1: body_md -->
There are many nontrivial scalar-flat complete steady gradient Yamabe solitons.
In fact, for any constants $\alpha>0$ and $\beta\in\mathbb{R}$ and any complete Riemannian manifold $(N^{n-1}(0),g_N)$ with a scalar-flat metric, 
\[
(\mathbb{R}\times N^{n-1}(0),dr^2+\alpha^2g_N,F=\alpha r+\beta,\lambda=0)
\]
is a nontrivial scalar-flat complete steady gradient Yamabe soliton. Taking $N^{n-1}$ to be a flat torus gives a non-rotationally symmetric example.
<!-- end remark-1 -->
<!-- theorem-2: body_md -->
\label{CSZ}
Let $(M^n,g,F,\lambda)$ be a nontrivial complete gradient Yamabe soliton. Then $|\nabla F|^2$ is constant on every regular level hypersurface of $F$, and exactly one of the following holds:
\begin{enumerate}
\item[$(1)$]
The potential function $F$ has a unique critical point, and $(M,g)$ is isometric to
\[
 ([0,+\infty),dr^2)
 \times_{|\nabla F|}
 (\mathbb{S}^{n-1},\bar g_S),
\]
where $\bar g_S$ is a round metric on $\mathbb{S}^{n-1}$, with its scale chosen as above.

\item[$(2)$]
The potential function $F$ has no critical point, and $(M,g)$ is isometric to
\[
 (\mathbb{R},dr^2)
 \times_{|\nabla F|}
 (N^{n-1},\bar g),
\]
where $(N^{n-1},\bar g)$ has constant scalar curvature $\bar R$.
\end{enumerate}
Moreover, in Case $(2)$, if $R\geq0$ on $M$, then either $\bar R>0$, or $R=\bar R=0$ and $|\nabla F|\equiv a$ for some constant $a>0$.
In the latter case, $(M,g)$ is isometric to the Riemannian product
\[
(\mathbb{R},dr^2)\times(N^{n-1},a^2\bar g).
\]
<!-- end theorem-2 -->
<!-- lemma-1: body_md -->
\label{zero}
Let $a_1, a_2, a_3, a_4$ be constants. Suppose that $\rho: \mathbb{R} \to \mathbb{R}$ is a smooth positive function satisfying the differential equation
\begin{equation} \label{eq:rho_ode}
\rho\rho'' = a_1- a_2\rho^2 - a_3^2\rho'\rho^2 - a_4^2\rho'^2.
\end{equation}
If $\rho$ satisfies either $\rho' \geq 0$ or $\rho' \leq 0$ on $\mathbb{R}$, and there exists a point $x_0 \in \mathbb{R}$ such that $\rho'(x_0) = 0$, then $\rho' \equiv 0$ on $\mathbb{R}$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
We only need consider Case (2) of Theorem \ref{CSZ}.
In this case, for $a,b,c,d=2,3,\cdots,n,$ the curvature tensors are obtained as follows:
\begin{align*}
R_{1a1b}&=-F'F'''{\bar g}_{ab},\quad R_{1abc}=0,\\
R_{abcd}&=(F')^2{\bar R}_{abcd}+(F'F'')^2(\bar g_{ad}\bar g_{bc}-\bar g_{ac}\bar g_{bd}),\notag
\end{align*}
\begin{align*}
R_{11}&=-(n-1)\frac{F'''}{F'},\quad 
R_{1a}=0,\\
R_{ab}&=\bar R_{ab}-((n-2)(F'')^2+F'F''')\bar g_{ab},\notag
\end{align*}
\begin{align}\label{RT3}
R=(F')^{-2}\bar R-(n-1)(n-2)\Big(\frac{F''}{F'}\Big)^2-2(n-1)\frac{F'''}{F'},
\end{align}
where the curvature tensors with a bar are the curvature tensors of $(N,\bar g)$.
By \eqref{RT3} and the soliton equation $R=F''$, one has the following ODE.
\begin{equation}\label{key}
\rho'\rho^2+(n-1)(n-2)\rho'^2+2(n-1)\rho\rho''=\bar R,
\end{equation}
where one can choose the direction of $r$ such that $\rho(r)=F'(r)>0$.
\\

\noindent
Case 1. $\bar R<0$. 

Assume that there exists $r_0\in\mathbb{R}$ such that $\rho'(r_0)=0$.
By \eqref{key}, we have 
\[
\rho''(r_0)=\frac{\bar R}{2(n-1)\rho(r_0)}<0.
\]
Thus, there is at most one zero of $\rho'$. 
Hence, 
\[
\rho'>0\qquad\text{on}\quad (-\infty,r_0).
\]
Set $s=-r$. We use the same notation $\rho$ for the reparametrized function and $\dot\rho(s)=\frac{d}{ds}\rho(s)$. 
Equation \eqref{key} is
\begin{equation}\label{keys}
-\dot\rho\rho^2+(n-1)(n-2)\dot\rho^2+2(n-1)\rho\ddot\rho=\bar R.
\end{equation}
Combining this, $\dot\rho(s)<0$, and $\rho(s)>0$ on $(-r_0,+\infty)$, we have $\ddot\rho(s)<0$ on $(-r_0,+\infty)$. Since $\rho(s)>0$ on $(-r_0,+\infty)$, we have a contradiction.
Thus, there exists no zero of $\rho'$. 
\\

\noindent
Case 1-a. $\bar R<0$ and $\rho'>0$ on $\mathbb{R}$.

By \eqref{key}, we have $\rho''<0$ on $\mathbb{R}$, which yields a contradiction.
\\

\noindent
Case 1-b. $\bar R<0$ and $\rho'<0$ on $\mathbb{R}$.

Fix $\tilde r\in\mathbb{R}$. Set $a=\rho(\tilde r)$.  
Since $\rho'<0$, $\rho(r)<a$ on $(\tilde r,+\infty)$.

Assume that $\frac{\bar R}{2a^2}\leq\rho'<0$. Then, by \eqref{key}, we have
\begin{align*}
\rho''
&=\frac{1}{2(n-1)\rho}
\Big\{
\bar R-\rho'\rho^2-(n-1)(n-2)\rho'^2
\Big\}\\
&<\frac{1}{2(n-1)\rho}
\Big\{
\bar R-\rho'a^2
\Big\}\\
&\leq\frac{\bar R}{4(n-1)\rho}<0.
\end{align*}
Hence, if $\frac{\bar R}{2a^2}\leq\rho'<0$, then $\rho'$ is strictly decreasing.  
Therefore, we obtain
\[
\rho'
\leq\max
\Big\{
\rho'(\tilde r),\frac{\bar R}{2a^2}\Big\}<0\qquad \text{on}\quad (\tilde r,+\infty),
\]
which is a contradiction, because $\rho$ is a positive function.

Hence, $\bar R<0$ cannot occur.
\\

\noindent
Case 2. $\bar R>0$. 

We show that $\rho'(r)\geq0$ on $\mathbb{R}$.

Assume that there exists $\tilde r\in\mathbb{R}$ such that 
\[
\rho'(\tilde r)<0.
\]
Hence,
\[
\rho'<0\qquad\text{on} \quad (-\infty,r_2)\ni\tilde r,
\]
for some $r_2$. In fact, if there exists a point $r_0\in \mathbb{R}$ such that $\rho'(r_0)=0$, then by \eqref{key}, we have
\[
\rho''(r_0)=\frac{\bar R}{2(n-1)\rho(r_0)}>0.
\]

Set $s=-r$ and $s_2=-r_2$. We show the following.
\begin{claim}
The positive function $\rho$ satisfies $\rho(r)\rightarrow +\infty$ as $r\rightarrow-\infty$.
\end{claim}
\begin{proof}

Assume that $\rho(s)\to c<+\infty$ as $s\to+\infty$.

If $\ddot\rho\geq0$ on $(s_2,+\infty)$, then $\dot\rho$ is nondecreasing.
Hence $\rho$ cannot be bounded from above, which is a contradiction.

Assume that $\ddot\rho\leq0$ on $(s_2,+\infty)$. 
If $\dot\rho\to \alpha>0$ as $s\to+\infty$, then this contradicts $\rho\leq c$. Hence $\dot\rho\to 0$ as $s\to+\infty$.
By \eqref{keys}, 
$\ddot\rho\to \frac{\bar R}{2(n-1)c}>0$ as $s\to+\infty$. 
However, this contradicts the assumption $\ddot\rho\leq0$.

Hence there exists a point $s_0\in(s_2,+\infty)$ such that $\ddot\rho(s_0)=0$.
By differentiating \eqref{keys}, we obtain
\[
\ddot\rho\rho^2+2\rho\dot\rho^2-2(n-1)^2\dot\rho\ddot\rho-2(n-1)\rho\dddot\rho=0.
\]
At $s_0$, 
\[
\dddot\rho=\frac{1}{n-1}\dot\rho^2>0.
\] 
Therefore, there exists exactly one zero of $\ddot\rho$, and 
\begin{equation*}
\left\{
\begin{aligned}
&\ddot\rho<0\qquad\text{on}\quad (s_2,s_0),\\
&\ddot\rho>0\qquad\text{on}\quad (s_0,+\infty).
\end{aligned}
\right.
\end{equation*}
Hence $\rho$ cannot be bounded from above, which is a contradiction.
\end{proof}
Set $G(\rho)=\dot\rho$.
\begin{align*}
\frac{d}{d\rho}
\Big(
\rho^{\frac{n-2}{2}}G(\rho)
\Big)
&=
\frac{1}{2(n-1)G(\rho)}\bar R \rho^{\frac{n-4}{2}}+\frac{1}{2(n-1)}\rho^{\frac{n}{2}}\\
&>\frac{1}{2(n-1)}\rho^{\frac{n}{2}}.
\end{align*}
By integrating both sides of the inequality, we have
\[
G(\rho)>\frac{1}{(n-1)(n+2)}\rho^2+C\rho^{-\frac{n-2}{2}},
\]
for some constant $C$.
For sufficiently large $\rho$, we obtain
\[
G(\rho)>\frac{1}{2(n-1)(n+2)}\rho^2.
\]
Hence, 
\[
\frac{d}{ds}
\Big(
-\rho^{-1}
\Big)
>\frac{1}{2(n-1)(n+2)}.
\]
Take $s_\ell$ sufficiently large. For $s>s_\ell$, integrating both sides of the above inequality, 
we have
\[
-\rho^{-1}(s)+\rho^{-1}(s_\ell)
>\frac{1}{2(n-1)(n+2)}(s-s_\ell).
\]
Taking $s\to +\infty$, we have a contradiction.

Hence there exists no $\tilde r$ such that $\rho'(\tilde r)<0$.
Therefore $\rho'(r)\geq0$ on $\mathbb{R}$.

If there exists $r_0$ such that $\rho'(r_0)=0$, then by Lemma \ref{zero}, we have $\rho'\equiv0$ on $\mathbb{R}$. By \eqref{key}, we have $\bar R=0$, which is a contradiction.
Therefore, we obtain $\rho'(r)>0$ on $\mathbb{R}$.

Set $s=-r$. Fix $s_1$ and set $\alpha=\rho(s_1)$.
Since $\dot\rho(s)=\frac{d}{ds}\rho(s)<0$, we have
\[
\rho<\alpha\qquad\text{on}\quad(s_1,+\infty).
\]
Take a negative constant $\beta$ such that 
\[
\beta\alpha^2-(n-1)(n-2)\beta^2\geq-\frac{\bar R}{2}.
\]
If $\dot\rho<-\varepsilon$ for some $\varepsilon>0$ on $(s_1,+\infty)$, then we have a contradiction, because $\rho>0$. 
Hence we may assume that there exists a maximal interval $(s_3,s_4)\subset(s_1,+\infty)$ on which $\dot\rho\geq\beta$. In this case, by \eqref{keys},
\begin{align*}
\ddot\rho
&\geq \frac{1}{2(n-1)\rho}
\Big(
\bar R+\beta\alpha^2-(n-1)(n-2)\beta^2
\Big)
\geq\frac{\bar R}{4(n-1)\alpha}>0.
\end{align*}
Hence $s_4$ must be $+\infty$. However, since $\ddot\rho>\frac{\bar R}{4(n-1)\alpha}$ on $(s_3,+\infty)$, we have a contradiction, because $\dot\rho<0$. 
\\

\noindent
Case 3. $\bar R=0$.

We show that $\rho'(r)\geq0$ on $\mathbb{R}$.

Differentiating \eqref{key}, we have
\begin{equation}\label{key3}
\rho''\rho^2+2\rho\rho'^2+2(n-1)^2\rho'\rho''+2(n-1)\rho\rho'''=0.
\end{equation}
Assume that there exists $\tilde r\in\mathbb{R}$ such that 
\[
\rho'(\tilde r)<0,
\]
that is, $\rho'(r)<0$ on a maximal interval $(\tilde r_1,\tilde r_2)\ni\tilde r$.
If $\tilde r_1$ and $\tilde r_2$ are finite, then $\rho'(\tilde r_1)=\rho'(\tilde r_2)=0$.
We note that if there exists $\tilde r_0\in(\tilde r_1,\tilde r_2)$ such that $\rho''(\tilde r_0)=0$, by \eqref{key3}, we obtain
\begin{equation}\label{rho3}
\rho'''(\tilde r_0)=-\frac{1}{n-1}\rho'^2(\tilde r_0)<0.
\end{equation}
Hence if there exists a zero of $\rho''$, then it is unique, and $\rho''$ crosses zero from positive to negative.

Assume that $\tilde r_1$ and $\tilde r_2$ are finite. In this case, $\rho(\tilde r_1)>\rho(\tilde r_2)$.
There exists $\tilde r_0\in(\tilde r_1,\tilde r_2)$ such that $\rho''(\tilde r_0)=0$. 
However, since $\rho$ is smooth, the above argument yields a contradiction.

Assume that $\tilde r_1$ is finite and $\tilde r_2=+\infty$. 
In this case we have
$\rho'(r)<0$ on $(\tilde r_1,+\infty)$. 
If there exists $\tilde r_0\in(\tilde r_1,+\infty)$ such that $\rho''(\tilde r_0)=0$, then  
the above argument shows that $\rho''<0$ on $(\tilde r_0,+\infty)$. 
Combining this, $\rho>0$, and $\rho'<0$ on $(\tilde r_1,+\infty)$, we have a contradiction.
Hence there exists no zero of $\rho''$. Since $\rho'(\tilde r_1)=0$, we have $\rho''<0$ on $(\tilde r_1,+\infty)$. Positivity of $\rho$ yields a contradiction.

Assume that $\tilde r_1=-\infty$ and $\tilde r_2$ is finite.
If there exists $\tilde r_0\in(-\infty,\tilde r_2)$ such that $\rho''(\tilde r_0)=0$, then $\rho''<0$ on $(\tilde r_0,\tilde r_2)$. Since $\rho'(\tilde r_2)=0$, we have a contradiction. Hence there exists no zero of $\rho''$ on $(-\infty,\tilde r_2)$.
In this case, since $\rho'(\tilde r_2)=0$, we have $\rho''>0$ on $(-\infty,\tilde r_2)$. Therefore, $\rho\to+\infty$ as $r\to-\infty$. 

Assume that $\tilde r_1=-\infty$ and $\tilde r_2=+\infty$, that is, $\rho'<0$ on $\mathbb{R}$.
If there exists $\tilde r_0\in\mathbb{R}$ such that $\rho''(\tilde r_0)=0$, then we have $\rho''<0$ on $(\tilde r_0,+\infty)$. Since $\rho>0$, we have a contradiction.

Hence we obtain $\rho''>0$ on $\mathbb{R}$ or $\rho''<0$ on $\mathbb{R}$. 
If $\rho''>0$ on $\mathbb{R}$, then we have $\rho\to +\infty$ as $r\to-\infty$.
If $\rho''<0$ on $\mathbb{R}$, then positivity of $\rho$ yields a contradiction.

Therefore, we obtain $\rho'<0$ on $(-\infty, \tilde r_2)$ and $\rho\to+\infty$ as $r\to-\infty$. Set $s=-r$ and $G(\rho)=\dot\rho$. By the same argument as in Case 2, we obtain a contradiction. 

Therefore, there exists no $\tilde r$ such that $\rho'(\tilde r)<0$, and we obtain $\rho'(r)\geq0$ on $\mathbb{R}$.
                                                                                     
                                      

If there exists $r_0\in\mathbb{R}$ such that $\rho'(r_0)=0$, then by Lemma \ref{zero}, we have $\rho'\equiv0$ on $\mathbb{R}$. By \eqref{key} and the soliton equation $\nabla\nabla F=Rg$, we have $R=\bar R=0$.

We consider $\rho'(r)>0$ on $\mathbb{R}$.

By \eqref{key},
\begin{align*}
\rho''
&= -\frac{\rho\rho'}{2(n-1)}-\frac{(n-2)\rho'^2}{2\rho}
<0.
\end{align*}
Since $\rho>0$ on $\mathbb{R}$, we have a contradiction.
