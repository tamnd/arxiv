<!-- theorem-1: body_md -->
\label{thm:incumbent-hardness}
For QKP with integer-valued objective coefficients and strictly positive integer weights, given a feasible incumbent $x^{\mathrm{inc}}$, deciding whether there exists a feasible $x$ with $x^\top Qx>(x^{\mathrm{inc}})^\top Qx^{\mathrm{inc}}$ is NP-complete.
<!-- end theorem-1 -->
<!-- proposition-1: body_md -->
\label{prop:sequence-surjectivity}
Fix sequence length $n$ and a transformer class satisfying the universal approximation property of \citet{yun2020transformers}. For every $x^\star\in\{0,1\}^n$ and $\varepsilon\in(0,1/2)$, there exist parameters $\theta$ such that $\|\tilde{x}_\theta-x^\star\|_\infty<\varepsilon$. Hence threshold decoding at $1/2$ is surjective onto $\{0,1\}^n$.
<!-- end proposition-1 -->
<!-- corollary-1: body_md -->
\label{cor:theta-hardness}
Under Proposition~\ref{prop:sequence-surjectivity} and given $M \in \mathbb{R}$, deciding whether there exists $\theta$ such that $\hat{x}_\theta^\top Q \hat{x}_\theta > M$ subject to $A\hat{x}_\theta\le b$ \footnote{This is the decision version of globally optimizing the objective via the hard-decoded transformer.} is NP-hard.
<!-- end corollary-1 -->
<!-- proposition-2: body_md -->
\label{prop:lambda-bound}
Let $f_{\mathrm{inc}}=(x^{\mathrm{inc}})^\top Qx^{\mathrm{inc}}$ and $U_Q=\sum_{i,j}\max\{Q_{ij},0\}$. If $A,b$ are integer-valued and $\lambda>U_Q-f_{\mathrm{inc}}$, every infeasible binary $x$ satisfies $x^\top Qx-\lambda\Phi(x)<f_{\mathrm{inc}}$; hence no global maximizer of the penalized binary objective is infeasible.
<!-- end proposition-2 -->
<!-- proposition-3: body_md -->
\label{prop:gumbel-gradient}
Let $D_\tau=\partial\tilde{x}_\tau/\partial\Delta$ and let $q$ be the standard logistic density. For finite $\Delta$, as $\tau\to0$,
\begin{equation}
\tilde{x}_\tau\to\mathbf 1\{\Delta+G>0\}\ \text{almost surely},\qquad
\mathbb{E}[D_\tau]\to q(-\Delta),\qquad
\mathrm{Var}(D_\tau)=\Theta(\tau^{-1}),
\label{eq:gumbel_moments}
\end{equation}
with $\mathbb{E}[D_\tau^2]=q(-\Delta)/(6\tau)+o(\tau^{-1})$.
<!-- end proposition-3 -->
<!-- theorem-2: body_md -->
\label{thm:induced-flow}
For continuous-time gradient ascent $\dot\theta=\nabla_\theta F_\lambda(\tilde{x}_\theta)$,
\begin{equation}
\dot{\tilde{x}}=K_\theta\nabla_{\tilde{x}}F_\lambda(\tilde{x}),\qquad K_\theta:=J_\theta J_\theta^\top\succeq0.
\label{eq:induced_kernel_flow}
\end{equation}
<!-- end theorem-2 -->
<!-- proof-1: body_md -->
By the chain rule, $\nabla_\theta F_\lambda=J_\theta^\top\nabla_{\tilde{x}}F_\lambda$ and $\dot{\tilde{x}}=J_\theta\dot\theta$, giving Eq.~\ref{eq:induced_kernel_flow}. Moreover $v^\top K_\theta v=\|J_\theta^\top v\|_2^2\ge0$.
<!-- end proof-1 -->
<!-- theorem-3: body_md -->
\label{thm:path-dependent-informal}
Assume an attention head affects the relaxed rollout nontrivially. For $c>0$, replace $W_Q$ by $cW_Q$ and $W_K$ by $c^{-1}W_K$, leaving all other parameters fixed. The resulting transformer defines exactly the same autoregressive map for every prefix, yet for some $c\ne1$ its kernel $J_\theta J_\theta^\top$ differs. Moreover, there exists a quadratic objective for which the two functionally identical transformers induce different instantaneous velocities in relaxed decision space.
<!-- end theorem-3 -->
<!-- theorem-4: body_md -->
\label{thm:path-dependent}
Let $\theta$ be the parameters of a causal transformer and suppose that, for some attention head, the output of the relaxed autoregressive rollout depends nontrivially on at least one of its query or key matrices, i.e.
\begin{equation}
J_Q
:=
\frac{\partial \tilde{x}_\theta}{\partial W_Q}
\neq 0
\qquad\text{or}\qquad
J_K
:=
\frac{\partial \tilde{x}_\theta}{\partial W_K}
\neq 0.
\label{eq:attention-nondegeneracy}
\end{equation}
For every $c>0$, define a reparametrized transformer $\theta^{(c)}$ by
\begin{equation}
W_Q^{(c)}=cW_Q,
\qquad
W_K^{(c)}=c^{-1}W_K,
\label{eq:qk-rescaling}
\end{equation}
with every other parameter unchanged.

Then:

\begin{enumerate}
\item the two transformers define exactly the same autoregressive map,
\begin{equation}
\tilde{x}_{\theta^{(c)}}(z)
=
\tilde{x}_{\theta}(z)
\qquad
\text{for every input or autoregressive prefix }z;
\label{eq:same-transformer-function}
\end{equation}

\item nevertheless, except in the degenerate case $J_Q=J_K=0$, there exists $c\neq1$ for which
\begin{equation}
    J_{\theta^{(c)}}J_{\theta^{(c)}}^\top
    \neq
    J_{\theta}J_{\theta}^\top.
    \label{eq:different-induced-metric}
\end{equation}

\item there exists a QKP objective for which the two transformers induce different instantaneous velocities in relaxed decision space
\begin{equation}
\dot{\tilde{x}}_{\theta} \neq
\dot{\tilde{x}}_{\theta^{(c)}}.
\end{equation}
\end{enumerate}

In other words there exist two transformers representing exactly the same binary solution but inducing different local optimization geometries in relaxed decision space.
<!-- end theorem-4 -->
<!-- proof-2: body_md -->
Under the rescaling stated in Theorem~\ref{thm:path-dependent}, the query and key matrices become
\begin{equation}
Q^{(c)}
=
H(cW_Q)
=
cQ
\end{equation}
and
\begin{equation}
K^{(c)}
=
H(c^{-1}W_K)
=
c^{-1}K.
\end{equation}
Their product is therefore invariant:
\begin{equation}
Q^{(c)}(K^{(c)})^\top
=
(cQ)(c^{-1}K)^\top
=
QK^\top.
\end{equation}
Hence the matrix entering the causally masked attention softmax is unchanged. Since $W_V$ and every subsequent model parameter are also unchanged, the output of this attention head is identical. Applying the same argument throughout the autoregressive rollout shows that
\begin{equation}
\tilde{x}_{\theta^{(c)}}(z)
=
\tilde{x}_{\theta}(z)
\end{equation}
for every prefix $z$. This proves the first statement.

We now examine the parameter Jacobian. The attention scores depend on $W_Q$ and $W_K$ through
\begin{equation}
W_QW_K^\top.
\end{equation}
For a perturbation $\delta W_Q$ around the reparametrized point,
\begin{equation}
\delta
\left(
W_Q^{(c)}(W_K^{(c)})^\top
\right)
=
\delta W_Q
(c^{-1}W_K)^\top,
\end{equation}
so the sensitivity to $W_Q$ is multiplied by $c^{-1}$. Similarly, the sensitivity to $W_K$ is multiplied by $c$. Consequently,
\begin{equation}
J_Q^{(c)}
=
c^{-1}J_Q,
\qquad
J_K^{(c)}
=
cJ_K.
\label{eq:jacobian-rescaling}
\end{equation}

Let $J_{\mathrm{rest}}$ denote the Jacobian with respect to all remaining parameters. Since the forward computation is unchanged under this rescaling, their contribution is unchanged. Hence
\begin{align}
J_{\theta^{(c)}}J_{\theta^{(c)}}^\top
&=
J_{\mathrm{rest}}J_{\mathrm{rest}}^\top
+
c^{-2}J_QJ_Q^\top
+
c^2J_KJ_K^\top.
\label{eq:metric-c}
\end{align}
At $c=1$,
\begin{equation}
J_{\theta}J_{\theta}^\top
=
J_{\mathrm{rest}}J_{\mathrm{rest}}^\top
+
J_QJ_Q^\top
+
J_KJ_K^\top.
\end{equation}

Suppose, for contradiction, that these two matrices were equal for every $c>0$. Then
\begin{equation}
(c^{-2}-1)J_QJ_Q^\top
+
(c^2-1)J_KJ_K^\top
=
0
\end{equation}
for every $c>0$. Taking $c\rightarrow\infty$ implies
\begin{equation}
J_KJ_K^\top=0,
\end{equation}
and therefore $J_K=0$. Taking $c\rightarrow0$ similarly yields $J_Q=0$. This contradicts the nondegeneracy assumption in Theorem~\ref{thm:path-dependent}.

Therefore, whenever the attention head affects the generated solution nontrivially, there exists $c\neq1$ such that
\begin{equation}\label{eq:jacobian-kernel-diff}
J_{\theta^{(c)}}J_{\theta^{(c)}}^\top
\neq
J_{\theta}J_{\theta}^\top.
\end{equation}

Combining Eq. \ref{eq:jacobian-kernel-diff} with Theorem~\ref{thm:induced-flow}, the two functionally equivalent initializations satisfy
\begin{align}
&\dot{\tilde{x}}_{\theta}
=
K_{\theta}
\nabla_{\tilde{x}}F_\lambda(\tilde{x}),\\&
\dot{\tilde{x}}^{(c)}_\theta
=
K_{\theta^{(c)}}
\nabla_{\tilde{x}}F_\lambda(\tilde{x}),
\end{align}
where
\begin{equation}
K_{\theta}
\neq
K_{\theta^{(c)}}
\end{equation}
for an appropriate $c$. Thus two transformers representing the same current solution can induce different trajectory kernels solely because of how that solution is encoded in weight space.

We next show that there exists a quadratic objective
\begin{equation}
f(\tilde{x})
=
\tilde{x}^\top Q\tilde{x}
\end{equation}
for which the two transformers induce different instantaneous trajectories:
\begin{equation}
K_{\theta}\nabla f(\tilde{x})
\neq
K_{\theta^{(c)}}\nabla f(\tilde{x}).
\end{equation}

Let
\begin{equation}
D
:=
K_{\theta}
-
K_{\theta^{(c)}}.
\end{equation}
Since $D\neq0$, there exists some $g\in\mathbb{R}^n$ such that
\begin{equation}
Dg\neq0.
\label{eq:distinguishing-gradient}
\end{equation}

We show that $g$ can be realized as the gradient of a quadratic objective at $\tilde{x}$. Let
\begin{equation}
r=\tilde{x}^\top\tilde{x}>0
\end{equation}
and define the symmetric matrix
\begin{equation}
S
=
\frac{
g\tilde{x}^\top+\tilde{x}g^\top
}{r}
-
\frac{
\tilde{x}^\top g
}{r^2}
\tilde{x}\tilde{x}^\top.
\label{eq:construct-S}
\end{equation}
A direct calculation gives
\begin{equation}
S\tilde{x}=g.
\end{equation}
Set
\begin{equation}
Q=\frac{1}{2}S.
\end{equation}
Because $S$ is symmetric,
\begin{equation}
\nabla_{\tilde{x}}
\left(
\tilde{x}^\top Q\tilde{x}
\right)
=
(Q+Q^\top)\tilde{x}
=
S\tilde{x}
=
g.
\end{equation}
Therefore
\begin{align}
\dot{\tilde{x}}_{\theta}
-
\dot{\tilde{x}}_{\theta^{(c)}}
=
\left(
K_{\theta}
-
K_{\theta^{(c)}}
\right)g=
Dg
\neq0.
\end{align}
Thus the different geometries produced by the self-attention symmetry can lead to genuinely different trajectories for an objective of exactly the quadratic form considered in this proof.
