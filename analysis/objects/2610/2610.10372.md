\label{thm:main}
Assume \(k\geq1\), \(a\in\mathbb Z\setminus\{0\}\), and
\(\boldsymbol\nu\in\mathbb R_{\geq0}^k\).  Let
\(Q_1,\ldots,Q_k\in\mathbb Z[X]\) be pairwise coprime irreducible
polynomials, put
\[
 Q=Q_1\cdots Q_k,\qquad g=\deg Q,\qquad
 D=\operatorname{Disc}(Q),\qquad \rho=\rho_Q,
\]
and assume that \(Q\) is primitive.  Assume
\(0<\alpha,\delta<1\), \(A,B\geq1\), and let 
\begin{equation*}\label{eq:epsilon-range}
 0<\varepsilon
 \leq\frac{\alpha}{100g(g+1/\delta)}
\end{equation*}
and \(\mathcal{M}_k(A,B,\varepsilon)\) be defined as in Section~\ref{section1}.

If  \(F\in\mathcal M_k(A,B,\varepsilon)\), then, uniformly for
\begin{equation}\label{eq:xyrange}
 x\geq c_0\|Q\|^\delta,\qquad x^\alpha<y\leq x,    
\end{equation}
we have
\begin{equation}\label{eq:main-theorem}
\begin{split}
 &\sum_{\substack{x<n\leq x+y\\Q(n)\neq0}}
 F(|Q_1(n)|,\ldots,|Q_k(n)|)
 \prod_{j=1}^k|\operatorname{Kl}(a;|Q_j(n)|)|^{\nu_j}\\
 &\quad\ll
 y\prod_{g<p\leq x}\left(1-\frac{\rho(p)}p\right)
 \sum_{d_1\cdots d_k\leq x}
 F(d_1,\ldots,d_k)H_{a,\boldsymbol\nu}(d_1,\ldots,d_k).
\end{split}
\end{equation}
Here the constant \(c_0\) and the implied constant depend only on
\(\boldsymbol\nu,g,k,\alpha,\delta,A,B\).

<!-- lemma-1: body_md -->
\label{lem:lambda}
Let \(\nu\geq0\) be fixed. For \(\ell \geq 1\), \(\lambda_{a,\nu}(2^\ell)\leq 2^{\frac52\nu}\) and  
\begin{equation}\label{eq:lambda-power}
 \lambda_{a,\nu}(p^\ell)\leq  2^\nu
 \qquad(p \ \textrm{odd}).
\end{equation}
If \(p\nmid2a\), then
\begin{equation}\label{eq:lambda-prime}
 \lambda_{a,\nu}(p)=\mathfrak s(\nu)+O_\nu(p^{-1/8}),
\end{equation}
uniformly in \(a\bmod p\). For a prime \(p\) and an integer \(a\), we also have
\begin{equation}\label{eq:lambda-prime-upper}
 \lambda_{a,\nu}(p)\leq\mathfrak s(\nu)+O_\nu(p^{-1/8}).
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Averaging \eqref{eq:prime-power-Kl-bound} over the reduced residue
  classes proves \eqref{eq:lambda-power}.

  The upper bound \eqref{eq:lambda-prime-upper} follows from
  \eqref{eq:lambda-prime} when \(p\nmid2a\). If \(p\mid a\), then
  \(\operatorname{Kl}(a\bar h^2;p)=-p^{-1/2}\) for \(h\in\mathbb F_p^\times\), so
  \(\lambda_{a,\nu}(p)=p^{-\nu/2}\). For \(\nu>0\), this is at most
  \(\mathfrak s(\nu)\) for all sufficiently large \(p\); the finitely many
  remaining primes are absorbed in the error constant. The case
  \(\nu=0\) is exact, and \(p=2\) is covered by
  \eqref{eq:lambda-power}.

  It remains to prove \eqref{eq:lambda-prime}. Suppose that
  \(p\nmid2a\). By \eqref{eq:prime-power-Kl-bound}, we may write
  \[
  \operatorname{Kl}(t;p)=2\cos\theta_{p,t},
  \qquad
  \theta_{p,t}\in[0,\pi]
  \qquad(t\in\mathbb F_p^\times).
  \]
  For \(r\geq0\), put
  \[
  \operatorname{sym}_r(\theta)=\frac{\sin((r+1)\theta)}{\sin\theta}.
  \]

  For \(r\geq1\), Katz \cite[Example~13.6]{Katz-1988} proved
  \begin{equation}\label{eq:full-symmetric-power}
   \bigg|\sum_{t\in\mathbb F_p^\times}
   \operatorname{sym}_r(\theta_{p,t})\bigg|
   \leq \frac12(r+1)p^{1/2}.
  \end{equation}
  Fouvry and Michel \cite[Lemma~2.1]{FM-2003} established the corresponding
  estimate for square parameters
  \begin{equation}\label{eq:square-symmetric-power}
   \bigg|\sum_{h\in\mathbb F_p^\times}
   \operatorname{sym}_r(\theta_{p,\bar h^{2}})\bigg|
   \leq 3(r+1)p^{1/2}.
  \end{equation}

  If \(a\) is a quadratic residue modulo \(p\), multiplication by
  \(a\) permutes the nonzero quadratic residues. Thus
  \[
  \sum_{h\in\mathbb F_p^\times}
  \operatorname{sym}_r(\theta_{p,a\bar h^{2}})
  =\sum_{h\in\mathbb F_p^\times}
  \operatorname{sym}_r(\theta_{p,\bar h^{2}}).
  \]
  Otherwise, \(h\mapsto\bar h^2\) and \(h\mapsto a\bar h^2\) are
  two-to-one maps onto the nonzero quadratic residues and nonresidues,
  respectively.
  Hence
  \[
  \sum_{h\in\mathbb F_p^\times}
  \operatorname{sym}_r(\theta_{p,a\bar h^{2}})
  +\sum_{h\in\mathbb F_p^\times}
  \operatorname{sym}_r(\theta_{p,\bar h^{2}})
  =
  2 \sum_{t\in\mathbb F_p^\times}
  \operatorname{sym}_r(\theta_{p,t}).
  \]
  It follows from \eqref{eq:full-symmetric-power} and
  \eqref{eq:square-symmetric-power} that, in either case,
  \begin{equation}\label{eq:a-symmetric-power}
  \begin{aligned}
   \bigg|\sum_{h\in\mathbb F_p^\times}
   \operatorname{sym}_r(\theta_{p,a\bar h^{2}})\bigg|
   &\leq \bigg|\sum_{h\in\mathbb F_p^\times}
   \operatorname{sym}_r(\theta_{p,\bar h^{2}})\bigg|
   +2\bigg|\sum_{t\in\mathbb F_p^\times}
   \operatorname{sym}_r(\theta_{p,t})\bigg|\\
   &\leq 4(r+1)p^{1/2},
  \end{aligned}
  \end{equation}
  uniformly in \(a\in\mathbb F_p^\times\). The argument of
  \cite[Lemmas~2.2 and 2.3]{FM-2003}, together with
  \eqref{eq:a-symmetric-power}, implies
  \begin{equation}\label{eq:a-ST-discrepancy}
    \sup_{0\leq u<v\leq\pi}
    \left|
    \frac1{p-1}
    \#\left\{
    h\in\mathbb F_p^\times:
    u\leq\theta_{p,a\bar h^{2}}\leq v
    \right\}
    -
    \frac2\pi\int_u^v\sin^2t\,dt
    \right|
    \ll p^{-1/8},
  \end{equation}
  uniformly in \(a\in\mathbb F_p^\times\), with an absolute implied constant.

  For \(0\leq t\leq\pi\), put
  \[
  F_{p,a}(t)=\frac1{p-1}
  \#\{h\in\mathbb F_p^\times:\theta_{p,a\bar h^2}\leq t\}.
  \]
  For fixed \(\nu>0\), integration by parts and
  \eqref{eq:a-ST-discrepancy} imply
  \[
  \begin{aligned}
    \frac1{p-1}\sum_{h\in\mathbb F_p^\times}
    |2\cos\theta_{p,a\bar h^2}|^\nu
    &=\int_0^\pi |2\cos t|^\nu\,dF_{p,a}(t)\\
    &=\frac2\pi\int_0^\pi |2\cos t|^\nu\sin^2t\,dt
    +O_\nu(p^{-1/8})\\
    &=\mathfrak s(\nu)+O_\nu(p^{-1/8}).
  \end{aligned}
  \]
  This proves \eqref{eq:lambda-prime} and completes the proof.

<!-- lemma-2: body_md -->
\label{lem:lifting}
Let \(p\) be a prime, and let \(R\in\mathbb Z[X]\) be nonconstant. Suppose
\(p\nmid\operatorname{lc}(R)\operatorname{Disc}(R)\). For any integer \(\ell\geq1\) and real
\(\nu\geq0\),
\begin{equation*}
 \sum_{\substack{r\bmod p^{2\ell}\\p^\ell\parallel R(r)}}
 \left|
 \operatorname{Kl} \left(
 a\,\overline{R(r)/p^\ell}^{2};p^\ell
 \right)
 \right|^\nu
 =\rho_R(p)\varphi(p^\ell)\lambda_{a,\nu}(p^\ell).
\end{equation*}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Fix an integer \(u\) with \(p\nmid u\). The polynomials
\(R(X)-p^\ell u\) and \(R(X)\) have the same reduction modulo \(p\), and hence share the same roots mod $p$ 
which are all simple by the hypothesis
\(p\nmid\operatorname{lc}(R)\operatorname{Disc}(R)\). By Hensel's lemma, each of these
\(\rho_R(p)\) roots, say $r$, lifts uniquely to a solution of
\(R(r)\equiv p^\ell u\,(\text{mod}\,p^{2\ell})\). Thus
\[
 \#\{r\bmod p^{2\ell}:R(r)\equiv p^\ell u\,(\text{mod}\,p^{2\ell})\}
 =\rho_R(p).
\]
Since \(p\nmid u\), the congruence \(R(r)\equiv p^\ell u\,(\text{mod}\,p^{2\ell})\) is equivalent to
\(p^\ell\parallel R(r)\) together with
\(R(r)/p^\ell\equiv u\,(\text{mod}\,p^\ell)\). For the same reason as in \eqref{p-property}, we rewrite the summation over
\(u\bmod p^\ell\) with \(p\nmid u\) to get
\[
\begin{aligned}
 \sum_{\substack{r\bmod p^{2\ell}\\p^\ell\parallel R(r)}}
 \left|\operatorname{Kl}\left(a\,\overline{R(r)/p^\ell}^{\,2};p^\ell\right)\right|^\nu
 &=\rho_R(p)\sideset{}{^*}\sum_{u\bmod p^\ell}
   |\operatorname{Kl}(a\bar u^{\,2};p^\ell)|^\nu\\
 &=\rho_R(p)\varphi(p^\ell)\lambda_{a,\nu}(p^\ell),
\end{aligned}
\]
by \eqref{eq:lambda}.

\label{lem:H-properties}
The function \(H_{a,\boldsymbol\nu}:\mathbb N^k\to\mathbb R_{\geq0}\) is multiplicative,
in the sense that
\[
 H_{a,\boldsymbol\nu}(\mathbf{de})

\label{lem:polynomial-sieve}
	Let \(Q\in\mathbb Z[X]\) be primitive of degree \(g\geq1\), let
	\(M\in\mathbb N\), \(r\in\mathbb Z\), and suppose that
	\[
	 x^\alpha<y\leq x,\qquad
	 M\leq x^{2\varepsilon_1},\qquad
	 2\leq z\leq x^{\varepsilon_3}.
	\]
	Then we have
	\[
		\sum_{\substack{\ x<n\leq x+y\\
				n\equiv r\,({\rm mod}\,M)\\
				p\mid Q(n),\ p\nmid M\ \Longrightarrow\ p>z}}
		1
		\ll
		\frac yM\prod_{\substack{g<p\leq z\\p\nmid M}}
		\left(1-\frac{\rho_Q(p)}p\right),
	\]
	where the implied constant depends only on \(g\).

<!-- proof-6: body_md -->
Write \(\rho=\rho_Q\) and \(n=Mm+r\). The interval for \(m\) is
\[
 I=\left(\frac{x-r}{M},\frac{x+y-r}{M}\right],\qquad
 X=|I|=\frac yM>x^{\alpha-2\varepsilon_1}=x^{19\alpha/25}.
\]
The case \(z\leq g\) follows from
\(\#(I\cap\mathbb Z)\leq X+1\). We may thus suppose that \(z>g\), and define
\[
 \mathfrak P_g=\prod_{\substack{g<p\leq z\\p\nmid M}}p.
\]
The sum in the lemma does not exceed
\[
 \#\{m\in I\cap\mathbb Z:(Q(Mm+r),\mathfrak P_g)=1\}.
\]

For \(d\mid\mathfrak P_g\), the congruence \(Q(Mm+r)\equiv0\, (\text{mod}\,d)\) has \(\rho(d)\) solutions modulo \(d\), since \((d,M)=1\). Hence
\begin{equation}\label{eq:sieve-remainder}
 \#\{m\in I\cap\mathbb Z:d\mid Q(Mm+r)\}
 =X\frac{\rho(d)}d+R_d,
 \qquad |R_d|\leq\rho(d).
\end{equation}

Since \(\rho(p)\leq g\), we have 
\[
 \left(1-\frac{\rho(p)}p\right)^{-1}
 \leq\left(1-\frac1p\right)^{-g}
       \bigl(1+O_g(p^{-2})\bigr)
       \quad \mbox{for \(p>2g\)},
\]
or  \(\le g+1\) for \(g<p\leq2g\).
It follows by Mertens' theorem that
\begin{equation}\label{eq:sieve-dimension-axiom}
 \prod_{\substack{w\leq p<v\\p>g,\ p\nmid M}}
 \left(1-\frac{\rho(p)}p\right)^{-1}
 \ll_g\left(\frac{\log v}{\log w}\right)^g
 \qquad(2\leq w<v),
\end{equation}


We take \(D_0=z^{20g}\). Since \(\log D_0/\log z=20g>9g+1\),
the upper bound in
\cite[Corollary~6.2]{IK-2004}, together with
\eqref{eq:sieve-remainder} and \eqref{eq:sieve-dimension-axiom}, yields
\[
 \#\{m\in I\cap\mathbb Z:(Q(Mm+r),\mathfrak P_g)=1\}
 \ll_g X  \prod_{\substack{g<p\leq z\\p\nmid M}}
 \left(1-\frac{\rho(p)}p\right)
 +\sum_{\substack{d\leq D_0\\d\mid\mathfrak P_g}} \rho(d).
\]
Since \(d\mid\mathfrak P_g\) is squarefree, we have
\(\rho(d)=\prod_{p\mid d}\rho(p)\leq g^{\omega(d)}\). Hence, we get
\[
 \sum_{\substack{d\leq D_0\\d\mid\mathfrak P_g}}\rho(d)
 \leq\sum_{d\leq D_0}g^{\omega(d)}
 \ll_g D_0(\log D_0)^{g-1}.
\]
Since \(X>x^{19\alpha/25}\geq z^{38g}\) and \(D_0=z^{20g}\),
it follows from \eqref{eq:sieve-dimension-axiom} that
\[
 X\prod_{\substack{g<p\leq z\\p\nmid M}}
 \left(1-\frac{\rho(p)}p\right)
 \gg_g X(\log z)^{-g}
 \gg_g D_0(\log D_0)^{g-1}.
\]
This proves the lemma.
