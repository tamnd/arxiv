<!-- lemma-1: body_md -->
\label{lem:max_infec}
Suppose there are $k$ customers in the queue before the arrival of a tagged susceptible customer and $j \geq 1$ of them are infectious.  Then, the probability that the distance between the tagged arriving customer and the closest infectious customer is $j+i$ is given by
\begin{eqnarray*}
\mathbb{P} \left( k+1-D_{I} = j+i |Q=k, |C_I| = j\right) &=& \frac{\binom{j+i-1}{j-1}}{\binom{k}{j}}.
\end{eqnarray*}
\begin{proof}
 Suppose the tagged customer is susceptible, and when they arrive there are $k$ in the queue customers (so they are customer $k+1$), and $j$ of these customers are infectious. Then note that the number of (susceptible) customers between the tagged customer and the closest infectious customer (by definition customer $c_{I_{\max}}$) is $D_I$, such that we are finding the probability that $I_{\max} = j+i$. This requires that there are $j-1$ \textit{other} infectious customers than the closest one, who must all occupy positions between 1 and $j+i-1$ (inclusive) in the queue. It immediately follows that there are $j+i-1 \choose j-1$ possible ways these customers may be arranged in the queue.  Finally, we compute the denominator by using the Hockey Stick identity and summing the possible numerators.  This completes the proof.
\end{proof}

<!-- lemma-2: body_md -->
\label{lem:cond-max-infec} For an $M/M/1$ queue with queue length $k$, the probability for all $1 \leq j \leq k$ that the $j^{th}$ person is the last infectious customer in the queue is equal to
\begin{eqnarray*}
    \mathbb{P}(I_{\max} = j| Q = k) = p(1-p)^{k-j}.
\end{eqnarray*}
               
                         
                
                                         
                                                                         
                          
              
             
                                                                        
\begin{proof}

For $1 \leq m \leq k$, consider a scenario where a tagged susceptible customer arrives to a queue with $k$ customers, where $m$ of them are infectious. We designate customer $c_j$ as the closest infectious individual to the newly arrived susceptible customer, for $j \leq k$. Utilizing the pigeonhole principle, it's evident that we require $m \leq j$.

If we define $i = j - m$, then we can assert that the closest infectious customer is $c_{m+i}$.  Think of $i$ as a measure of this closest infectious customer's ``closeness." As $m$ represents the furthest spot in the queue from the tagged customer that the last infectious customer in the queue could occupy, $i$ quantifies the last infectious customer's additional proximity to the arriving susceptible customer.
Now, using Lemma~\ref{lem:max_infec}, we see that then \begin{eqnarray*}
    \mathbb{P}(I_{\max} = j| Q = k, |C_I| = m) &=& \mathbb{P}(I_{\max} = m+i| Q = k, |C_I| = m)\\ 
    &=& \mathbb{P}(k+1-D_I = m+i| Q = k, |C_I| = m)\\    &=& \frac{{m+i-1 \choose m-1}}{{k \choose m}}= \frac{{j-1 \choose m-1}}{{k \choose m}}.
\end{eqnarray*}

    Using the above observation,
    \begin{align*}
        \mathbb{P}(I_{\max} = j| Q = k)
        &= \sum_{m=1}^j\mathbb{P}(I_{\max} = j| Q = k, |C_I| = m)\mathbb{P}(|C_I|=m)\\
        &= \sum_{m=1}^j \frac{{j-1 \choose m-1}}{{k \choose m}}\cdot p^m(1-p)^{k-m}{k \choose m}\\
        &= \sum_{m=1}^j {j-1 \choose m-1}p^m(1-p)^{k-m}\\
        &= p(1-p)^{k-j}.
    \end{align*}
This completes the proof.
\end{proof}

<!-- lemma-3: body_md -->
\label{lem:cond-upon-infect} Consider a tagged susceptible arrival to the $M/M/1$ queue in steady state with queue length $k$, where for any $1\leq j \leq k$ the $j^{th}$ person waiting in the queue is the last infectious customer in the queue. The probability that the tagged customer gets infected during their time in the system is equal to
$$\mathbb{P}(I_{ua}|I_{\max} = j, Q = k) =\sum_{n=0}^{j-1} e^{-\mu l} \frac{(\mu l)^n}{n!}.$$
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
We begin by noting that for an infection to occur we require both (a) that the tagged customer has sojourn time $(\mathcal{S} = S+W)$ at least $\ell$ and (b) that some infectious customer is also in the system for at least $\ell$. 
    \begin{align*}
\mathbb{P}(I_{ua}|I_{\max} = j, Q = k) 
&= \mathbb{P}\left(S+W \geq l, \max_{i \in C_I}\sum_{m=1}^{i} R_m \geq l|I_{\max} = j, Q = k\right)\\ 
&= \mathbb{P}\left(S+\sum_{m=1}^{k}R_m \geq l, \max_{i \in C_I}\sum_{m=1}^{i} R_m \geq l|I_{\max} = j, Q= k\right) .
\intertext{Since $i \leq k, \forall i \in I$, and $S \geq 0$,}
&= \mathbb{P}\left(\max_{i \in C_I}\sum_{m=1}^{i} R_m \geq l|I_{\max} = j, Q = k\right)\\ 
&= \mathbb{P}\left(\sum_{m=1}^{j} R_m \geq l|I_{\max} = j, Q = k\right)\\ 
&= \mathbb{P}\left(\sum_{m=1}^{j} R_m \geq l\right)\\ 
&=\sum_{n=0}^{j-1} e^{-\mu l} \frac{(\mu l)^n}{n!}.
\end{align*}

\label{thm:upon-arrival}
Let a susceptible customer join the $M/M/1$ queue in steady state, then the probability the tagged customer gets infected by a customer who is in queue upon their arrival is equal to
\begin{equation*}
\mathbb{P}\left( I_{ua} \right) = e^{-(\mu-\lambda) \ell} \frac{p \lambda}{\mu-\lambda(1-p)} .
\end{equation*}
\begin{proof}
\begin{align*}
\mathbb{P}(I_{ua}) &= \sum_{k=0}^{\infty} \mathbb{P}(Q =k) \cdot \mathbb{P}(I_{ua} |Q = k)\\
&= \sum_{k=1}^{\infty}\mathbb{P} \left(Q = k \right) \cdot \mathbb{P}\left(I_{ua} |Q = k\right)\\
&= \sum_{k=1}^{\infty}\mathbb{P}\left(Q = k \right) \cdot \sum_{j=1}^{k}\mathbb{P}\left(I_{ua} | I_{\max} = j, Q = k)\mathbb{P}(I_{\max} = j| Q = k \right).
\intertext{By Lemmas~\ref{lem:cond-max-infec} and~\ref{lem:cond-upon-infect}, respectively, we recognize the above expression to be}
&= e^{-\mu \ell}\left(1-\frac{\lambda}{\mu}\right)\sum_{k=1}^{\infty}\left(\frac{\lambda}{\mu}\right)^k\sum_{j=1}^{k}p(1-p)^{k-j}\sum_{n=0}^{j-1}\frac{(\mu \ell)^n}{n!}\\
&= e^{-\mu \ell} \left(1-\frac{\lambda}{\mu}\right)\sum_{n=0}^{\infty}\frac{(\mu \ell)^n}{n!} \sum_{k=n+1}^{\infty}\left(\frac{\lambda}{\mu} \right)^k\sum_{j=n+1}^{k}p(1-p)^{k-j} \\
&= e^{-\mu \ell}\left(1-\frac{\lambda}{\mu}\right)\sum_{n=0}^{\infty}\frac{(\mu \ell)^n}{n!} \sum_{k=n+1}^{\infty}\left(\frac{\lambda}{\mu}\right)^k\left(1-(1-p)^{k-n}\right)\\
&= e^{-\mu \ell} \left(1-\frac{\lambda}{\mu}\right)\sum_{n=0}^{\infty}\frac{(\mu \ell)^n}{n!}\frac{p\left(\frac{\lambda}{\mu}\right)^{n+1}}{(1-\frac{\lambda}{\mu})(p\frac{\lambda}{\mu}-\frac{\lambda}{\mu}+1)} \\

&= e^{-(\mu-\lambda) \ell}\frac{p\lambda}{\mu-\lambda(1-p)} .
\end{align*}
\end{proof}

\label{lem:integral-prob}
 Consider a susceptible customer arriving to an $M/M/1$ queue in steady state. If $k$ infectious customers arrive while the tagged customer is in the system (that is to say, within the interval $[0,x]$), the probability that the first arrives while the tagged customer has less than $\ell$ sojourn remaining is equal to
    $$\mathbb{P}(A_1>x-\ell|M_I=k) = \left(\frac{\ell}{x}\right)^k.$$
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
We find it easier to calculate the converse, $$\mathbb{P}(A_1>x-\ell|M_I=k) =1 - \mathbb{P}(A_1 \leq x- \ell | M_I = k).$$ By conditioning on $M_I = k$, we now have that the arrival times are distributed uniformly on $[0,x]$. We use the distribution of the CDF for the minimum order statistic of $X \sim U(0,x)$, which is 
        \begin{eqnarray*}
            \sum_{i=1}^k{k \choose i}F_X(x-\ell)^i(1-F_X(x-\ell))^{k-i} &=& \sum_{i=1}^k{k \choose i}\left(\frac{x-\ell}{x}\right)^i\left(\frac{\ell}{x}\right)^{k-i} \\
            &=& 1-\left(\frac{\ell}{x}\right)^k.
             \end{eqnarray*}
It is worth confirming that when $k=0$, that is there are no infectious arrivals within a time window of $x$ after the tagged customer arrived, then $\mathbb{P}(A_1>x-\ell)=1-\left(1-\left(\frac{\ell}{x}\right)^0\right) = 1$.

    e^{-\lambda p x } (\lambda p x )^j }{j!}dx.\intertext{We employ Lemma~\ref{lem:serviceShort} and Lemma~\ref{lem:serviceLong} to substitute for these first and second summations, respectively, thus getting,}
    =& 1-\left(1-e^{-(\mu-\lambda) \ell}
 +\frac{\mu-\lambda}{\mu-\lambda(1-p)}e^{-(\mu-\lambda) \ell}\right)\\
 =& \left(1-\frac{\mu-\lambda}{\mu-\lambda(1-p)}\right)e^{-(\mu-\lambda) \ell}\\
 =& \frac{p\lambda}{\mu-\lambda(1-p)}e^{-(\mu-\lambda) \ell} .
    \end{align*}

<!-- lemma-8: body_md -->
\label{lem:prob-sojourn-x}
A customer arriving to the $M/M/1$ queue in steady state has an exponential sojourn time probability density function i.e.
    $$f_{\mathcal{S}}(x) = (\mu-\lambda)e^{-(\mu-\lambda) x}.$$

<!-- proof-10: body_md -->
The proof is well known in the queueing literature.

<!-- lemma-9: body_md -->
\label{lem:queue-given-sojourn}
For a customer arriving to the $M/M/1$ queue in steady state with sojourn time $x$, the likelihood that there are $r$ customers in the system at their time of arrival is Poisson,
    $$\mathbb{P}(Q=r|\mathcal{S} = x) = \frac{e^{-\lambda x}(\lambda x)^r}{r!}.$$
It is additionally convenient for our purposes to note that this result holds identically for $\mathbb{P}(Q=r|\mathcal{R} = x)$.
<!-- end lemma-9 -->
<!-- proof-11: body_md -->
The proof of this result appears in \citet{ross2014introduction}.  However, we have included a proof for the reader's convenience.
    \begin{align*}
        \mathbb{P}(Q=r|\mathcal{S} = x) 
        &= \frac{\mathbb{P}(\mathcal{S}=x|Q=r)\mathbb{P}(Q=r)}{\mathbb{P}(\mathcal{S} = x)}\intertext{So by Lemma~\ref{lem:prob-sojourn-x},}
        &= \frac{\frac{\mu^{r+1}x^re^{-\mu x}}{r!}\left(\frac{\lambda}{\mu}\right)^r\left(1-\frac{\lambda}{\mu}\right)}{(\mu-\lambda)e^{-(\mu-\lambda) x}}\\
        &=\frac{e^{-\lambda x}(\lambda x)^r}{r!}.
    \end{align*}
By the memoryless property of the exponential inter-arrival distribution, we find that $\mathbb{P}(\mathcal{R}=x)=\mathbb{P}(\mathcal{S} = x)$, so this proof demonstrates that the result holds identically for both remaining and total sojourn time.

<!-- lemma-10: body_md -->
\label{lem:closest-infec-upon-cant-infect}
For a susceptible arrival to the $M/M/1$ queue in steady state with sojourn time $x$, the probability that there are no infectious customers in the queue upon arrival with remaining sojourn time long enough to infect them is,
    $$\mathbb{P}(\mathcal{R}_{I_{\max}} < \ell|\mathcal{S} = x) = e^{-\lambda p(x-\ell)}.$$
<!-- end lemma-10 -->
<!-- proof-12: body_md -->
We recognize that $\mathcal{R}_{I_{\max}}$ is the remaining sojourn time of the closest infectious customer to the tagged customer. We also note from Lemma~\ref{lem:queue-given-sojourn} that for fixed sojourn time $x$, queue length is Poisson. Thus, we utilize the memoryless property of the exponential distribution to recognize that this is the same as confirming that when the tagged customer has remaining sojourn time $x-\ell$ there are no infectious customers ahead of them in the queue.

    \begin{align*}
        \mathbb{P}(\mathcal{R}_{I_{\max}} < \ell|\mathcal{S} = x) 
        &= \sum_{k=0}^{\infty}\mathbb{P}(Q =k |\mathcal{R} = x-\ell)\mathbb{P}(|C_I| = 0|Q =k)\\
        &= \sum_{k=0}^{\infty}e^{-\lambda(x-\ell)}\frac{(\lambda(x-\ell))^k}{k!}(1-p)^k\\
        &= e^{-\lambda(x-\ell)}\sum_{k=0}^{\infty}\frac{(\lambda(1-p)(x-\ell))^k}{k!}\\
        &= e^{-\lambda p(x-\ell)}.
    \end{align*}

<!-- theorem-3: body_md -->
\label{thm:total-infection-prob}
Let $I = I_{ua} \cup I_{aa}$ be the event that a susceptible customer gets infected while in the queue. Suppose a susceptible customer joins the $M/M/1$ queue in steady state, then the probability the customer gets infected is equal to
\begin{eqnarray}\label{eqn:prob-infec}
    \mathbb{P}(I) &=&  \mathbb{P}(I_{ua} \cup I_{aa}) =  \frac{ 2 \lambda p}{\mu - \lambda + 2 \lambda p}  e^{- (\mu - \lambda)  \ell} .
\end{eqnarray}
<!-- end theorem-3 -->
<!-- proof-13: body_md -->
\begin{align*}
    \mathbb{P}(I_{ua} \cup I_{aa}) &=
    1 - \mathbb{P}(I_{ua}^c \cap I_{aa}^c)\\
    &= 1- \Big(\mathbb{P}(I_{ua}^c \cap I_{aa}^c | \mathcal{S} < \ell ) \mathbb{P}( \mathcal{S} < \ell ) + \mathbb{P}(I_{ua}^c\cap I_{aa}^c | \mathcal{S} \geq \ell ) \mathbb{P}( \mathcal{S} \geq \ell )\Big) \\
    &= 1-\Big(\mathbb{P}( \mathcal{S} < \ell ) + \mathbb{P}(I_{ua}^c \cap I_{aa}^c | \mathcal{S} \geq \ell ) \mathbb{P}( \mathcal{S} \geq \ell )\Big) \\
    &=  1-\Big(1 - e^{-(\mu - \lambda) \ell} + \mathbb{P}(I_{ua}^c \cap I_{aa}^c , \mathcal{S} \geq \ell )\Big)  \intertext{We note that $I_{ua}^c$ and $I_{aa}^c$ are independent given $\mathcal{S} =x$. For fixed sojourn time $x$, we also recognize that \textit{not} getting infected upon arrival is dependent on the closest infectious customer having remaining sojourn time less than $\ell$. Similarly, \textit{not} getting infected after arrival is dependent on no infectious customers arriving within $x-\ell$.}
    &=  e^{-(\mu - \lambda) \ell} - \left( \int^{\infty}_{\ell} \mathbb{P}(\mathcal{R}_{I_{\max}} < \ell |\mathcal{S} =x) \cdot \mathbb{P}(N_I(x-\ell ) = 0) \cdot  \mathbb{P}(\mathcal{S} =x) dx\right)  \intertext{By Lemmas~\ref{lem:closest-infec-upon-cant-infect} and~\ref{lem:prob-sojourn-x},}
    &=  e^{-(\mu - \lambda) \ell} - \left( \int^{\infty}_{\ell} e^{-2\lambda p (x-\ell)}  \cdot  (\mu - \lambda) e^{-(\mu - \lambda) x} dx \right)  \\
    &= e^{-(\mu - \lambda) \ell} - e^{2 \lambda p \ell} \left( \int^{\infty}_{\ell} e^{-(2\lambda p + \mu - \lambda ) x}  \cdot  (\mu - \lambda) dx \right)  \\
    &=  e^{-(\mu - \lambda) \ell} - e^{- (\mu - \lambda)  \ell} \frac{ \mu - \lambda}{\mu - \lambda + 2 \lambda p}   \\
    &= \frac{ 2 \lambda p}{\mu - \lambda + 2 \lambda p}e^{- (\mu - \lambda)  \ell}.
\end{align*}

<!-- remark-1: body_md -->
From a practical point of view, a manager of a service system might be interested in understanding what arrival rate would induce a probability of infection $\epsilon$. This has no closed form, but one can give an approximation based on heavy or low traffic.

    For a low traffic setting, where $\lambda \ll \mu$, we can approximate the denominator of Equation \ref{eqn:prob-infec} as $\mu$, such that we have \begin{align*}
        \epsilon &\approx \frac{2p\lambda}{\mu}e^{-(\mu-\lambda)\ell}\intertext{whose terms we rearrange to get,}
        \ell\frac{\mu}{2p} e^{\mu\ell} \epsilon&\approx \ell \lambda e^{\lambda\ell}.\intertext{As the Lambert $W$ function is the inverse of $f(w) = w e^w$, we can thus rewrite this as}
        \lambda(\epsilon) &\approx \frac{1}{\ell}W\left(\frac{\mu \ell \epsilon}{2p} e^{\mu \ell}\right),\intertext{where $W$ is the Lambert $W$ function.}
    \end{align*}

    As the exponential term dominates Equation~\ref{eqn:prob-infec}, this approximation is useful while treating the denominator as $\mu$ is reasonable. In contrast, once we assume $\lambda \approx \mu$, we find it more useful to define $\delta = \mu-\lambda$ such that $\delta \ll \mu$, so that we can rewrite Equation~\ref{eqn:prob-infec} as
    \begin{align*}
        \epsilon &\approx \frac{2p(\mu-\delta)}{\mu +(2p-1)(\mu-\delta)}e^{-\delta\ell}\\
        \epsilon &\approx \frac{2p(\mu-\delta)}{2p(\mu -\delta)+\delta}e^{-\delta\ell}\intertext{The Taylor expansions of the scalar term, $1-\frac{\delta}{2p\mu}-\frac{(2p-1)\delta^2}{(2p\mu)^2}-\frac{(2p-1)^2\delta^3}{(2p\mu)^3}+\cdots$, and of the exponential term, $1-\delta\ell + \frac{(\delta\ell)^2}{2}-\frac{(\delta\ell)^3}{3}+\cdots$, let us rewrite the approximation as}
        \epsilon &\approx 1 - \delta\ell -\frac{\delta}{2p\mu}+O\left(\delta^2\right)\intertext{which we can rearrange to isolate $\delta$ as}
        \delta &\approx \frac{2p\mu(1-\epsilon)}{1+2p\mu\ell}\intertext{So substituting back for $\lambda$ we get our high utility approximation as}
        \lambda(\epsilon) &\approx \mu\left(1-\frac{2p(1-\epsilon)}{1+2p\mu\ell}\right)
    \end{align*}

    Note for these two approximations that their accuracy improves as either $\lambda$ (normalized by $\mu$) or $\delta$, respectively, becomes smaller.

<!-- remark-2: body_md -->
A key observation is that our analytical expressions are initially derived for a 
\emph{fixed} infection transmission threshold $\ell$.  However, this formulation is far more 
general than it may appear at first glance.  If the infection transmission threshold is 
allowed to be random for each susceptible customer, with distribution function $F_\Theta(\ell)$, then the corresponding 
performance measures can be obtained simply by integrating our fixed-threshold 
quantities with respect to $F_\Theta$.  In particular, if $G(\ell)$ denotes any of 
the fixed-threshold expressions derived in our analysis, then the random-threshold 
version is given by
\[
\mathbb{E}\!\left[ G(\Theta) \right] 
    = \int_{0}^{\infty} G(\ell)\, dF_\Theta(\ell).
\]
Because $G(\ell)$ always appears in the form of an exponential evaluated at 
$\mu - \lambda$, this integration step reduces to evaluating the 
Laplace-Stieltjes transform of the threshold distribution at the argument 
$\mu - \lambda$.  That is,
\[
\mathbb{E}\!\left[ e^{-(\mu - \lambda)\Theta} \right]
    = \mathcal{L}_\Theta(\mu - \lambda),
\]
so the random-threshold expressions follow directly by replacing the exponential 
term associated with the fixed threshold by the Laplace-Stieltjes transform of 
the threshold distribution. Consequently, our framework admits 
\emph{arbitrary threshold distributions} for each tagged susceptible customer with no additional structural 
assumptions. This is a manner in which our work can be generalized beyond what is done in \citet{kang2021queueing}. While \citet{kang2021queueing} allows for a pairwise infection threshold for each pair of susceptible-infectious customers it is worth noting two things. Firstly, that their analysis assumes only a single infectious customer in the system at a time, such that it is not substantially different from our consideration of a random threshold for a tagged susceptible customer. Secondly, we conjecture that the random pairwise and random per tagged customer infection threshold cases are not greatly dissimilar, see Tables~\ref{tab:compareaveragemeans} and \ref{tab:compareaveragevars}.
Our approach seamlessly extends to \emph{any} 
distribution, for example, deterministic, Erlang, hyperexponential, heavy-tailed, 
or empirical threshold models merely by substituting the appropriate 
Laplace-Stieltjes transform.  Hence, our results provide a unified treatment 
that accommodates a much broader class of infection mechanisms.

<!-- lemma-12: body_md -->
\label{lem:first-sojourn}
    For the $M/M/1$ queue in steady state, if $k$ people are in the queue, then for all $0<j\leq r$, the probability customer $j$ is the first in the queue with remaining sojourn time at least $\ell$ is $$\mathbb{P}\left(\mathbf{1}_{\tilde{\mathcal{R}_{j}} > \ell}|Q = r\right) = \frac{(\mu\ell)^{j-1}e^{-\mu\ell}}{(j-1)!}.$$
<!-- end lemma-12 -->
<!-- proof-15: body_md -->
First suppose $j = 1$. This is then simply the probability that the first person in the queue has remaining service time at least $\ell$, so $$\mathbb{P}(R_1 \geq \ell|Q=r) =  \mathbb{P}(S_1 \geq \ell) = e^{-\mu\ell}.$$

    Now suppose $j \geq 2$ (for any $j \leq r$). Then we have
    \begin{align*}
        \mathbb{P}\left(\mathbf{1}_{\tilde{\mathcal{R}_{j}} > \ell}|Q = r\right) 
        &= \mathbb{P}\left(\mathbf{1}_{\tilde{\mathcal{R}_{j}} > \ell}\right) \\
        &= \mathbb{P}\left(S_j + \sum_{i=1}^{j-1}R_i \geq \ell, S_{j-1} + \sum_{i=1}^{j-2}R_i < \ell\right)\\
        &= \mathbb{P}\left(\sum_{i=1}^{j}R_i \geq \ell,\sum_{i=1}^{j-1}R_i < \ell\right)\\
        &= \mathbb{P}\left(\sum_{i=1}^{j}R_i \geq \ell\Bigg|\sum_{i=1}^{j-1}R_i < \ell\right)\mathbb{P}\left(\sum_{i=1}^{j-1}R_i < \ell\right)\\
        &= \int_0^{\ell}\mathbb{P}\left(R_j+\sum_{i=1}^{j-1}R_i \geq \ell\Bigg|\sum_{i=1}^{j-1}R_i =x\right)\mathbb{P}\left(\sum_{i=1}^{j-1}R_i  =x\right)dx\\
        &= \int_0^{\ell}\mathbb{P}\left(R_j \geq \ell - x\Bigg|\sum_{i=1}^{j-1}R_i =x\right)\mathbb{P}\left(\sum_{i=1}^{j-1}R_i  =x\right)dx\\
        &= \int_0^{\ell}\mathbb{P}\left(S_j \geq \ell - x\right)\mathbb{P}\left(\sum_{i=1}^{j-1}R_i  =x\right)dx\\
        &= \int_0^{\ell}e^{-\mu(\ell-x)}\frac{\mu^{j-1}x^{j-2}e^{-\mu x}}{(j-2)!}dx\\
        &= \frac{(\mu\ell)^{j-1}e^{-\mu\ell}}{(j-1)!}.
    \end{align*}

<!-- lemma-14: body_md -->
\label{lem:num-upon-given-sojourn}

<!-- proof-19: body_md -->
Here, we recognize that any customer who departs within the first $\ell$ units of time after this tagged customer arrives to the system can not possibly infect them. Thus, utilizing the memoryless property of the exponential distribution, it is useful to only consider a ``snapshot'' of the queue after $\ell$ units of time have already passed, or the state of the queue when the tagged customer's remaining sojourn time is $x-\ell$.

        &=\sum_{r=m}^{\infty}\mathbb{P}(Q=r|\mathcal{R} = x-\ell)\sum_{j=m}^{r}\mathbb{P}(I_{\max} = j|Q=r)\mathbb{P}(N_{ua} = m|I_{\max} = j)\intertext{Because we consider the queue as now only consisting of those customers whose \textcolor{black}{remaining} sojourn time was at least $\ell$ upon the tagged customer's arrival, any infectious customers amongst the new ``first'' $j$ must successfully transmit their infection. Further, we have assumed that customer $j$ is infected. These facts and Lemma~\ref{lem:queue-given-sojourn} give us}
        &=\sum_{r=m}^{\infty}\frac{e^{-\lambda (x-\ell)}(\lambda (x-\ell))^r}{r!}\sum_{j=m}^{r}p(1-p)^{r-j}{m-1 \choose j-1} p^{m-1}(1-p)^{j-m}\\
        &=\left(\frac{p}{1-p}\right)^me^{-\lambda (x-\ell)}\sum_{r=m}^{\infty}\frac{(\lambda (x-\ell)(1-p))^r}{r!}\sum_{j=m}^{r}{m-1 \choose j-1}\\
        &=\left(\frac{p}{1-p}\right)^me^{-\lambda (x-\ell)}\sum_{r=m}^{\infty}\frac{(\lambda (x-\ell)(1-p))^r}{r!}\frac{(r-m+1){r \choose m-1}}{m}\\
        &=\left(\frac{p}{1-p}\right)^me^{-\lambda (x-\ell)}\frac{e^{\lambda(x-\ell)(1-p)}(\lambda(x-\ell)(1-p))^m}{m!}\\
        &=e^{-\lambda p (x-\ell)}\frac{(\lambda p (x-\ell))^m}{m!}.
    \end{align*}

<!-- theorem-6: body_md -->
\label{thm:exactly-k-infections}
Consider any $k\geq 1.$ For a susceptible arrival to the $M/M/1$ queue in steady state, the probability that exactly $k$ infectious customers transmit infection to them is,
\begin{eqnarray}\label{eqn:exactkinfec}
    \mathbb{P}(N_{I} = k) =\frac{\mu-\lambda}{\mu-\lambda+2\lambda p}\left(\frac{2\lambda p}{\mu-\lambda+2\lambda p}\right)^ke^{-(\mu-\lambda)\ell}.
\end{eqnarray}
<!-- end theorem-6 -->
<!-- proof-21: body_md -->
Utilizing the fact that the total number of people who infect the tagged customer is simply the sum of the number who infect them upon arrival and the number who infect them after arrival, since those sets partition all those who infect the tagged customer, we find that
    \begin{align*}
        \mathbb{P}(N_{I} = k) 
        &= \sum_{m=0}^k \mathbb{P}(N_{ua} = m, N_{aa} = k-m)\\
        &= \sum_{m=0}^k\int_{\ell}^{\infty}\mathbb{P}(\mathcal{S} = x)\mathbb{P}(N_{ua} =m, N_{aa} = k-m|\mathcal{S} = x) dx \intertext{So by Lemmas~\ref{lem:num-upon-given-sojourn} and~\ref{lem:num-after-given-sojourn} and Lemma~\ref{lem:prob-sojourn-x},}
        &=\int_{\ell}^{\infty}(\mu -\lambda)e^{-(\mu-\lambda)x}e^{-2\lambda p(x-\ell)}\sum_{m=0}^k\frac{(\lambda p(x -\ell))^{k}}{m!(k-m)!}dx\\
        &=\int_{\ell}^{\infty}(\mu -\lambda)e^{-(\mu-\lambda)x}e^{-2\lambda p(x-\ell)}\frac{(2\lambda p(x -\ell))^{k}}{k!}dx\\
        &=\frac{(\mu-\lambda)(2\lambda p)^k}{k!}\int_{\ell}^{\infty}e^{-(\mu-\lambda)x}e^{-2\lambda p(x-\ell)}(x -\ell)^{k}dx\\
        &=\frac{(\mu-\lambda)(2\lambda p)^k}{k!}\left(\frac{-e^{-(\mu-\lambda)\ell}}{(\mu-\lambda+2\lambda p)^{k+1}}\Gamma(k+1, (\mu-\lambda+2\lambda p)(x-\ell))\right)\Bigg|_{x=\ell}^{x=\infty}\intertext{making use of the fact that $k+1\in\mathbb{Z}^{+}$, we can express the lower incomplete gamma function in summatio form as}
        &=\frac{(\mu-\lambda)e^{-(\mu-\lambda)\ell}}{\mu-\lambda+2\lambda p}\left(\frac{2\lambda p}{\mu-\lambda+2\lambda p}\right)^k\left(\sum_{n=0}^k\frac{((\mu-\lambda+2\lambda p)(x-\ell))^n}{n! e^{(\mu-\lambda+2\lambda p)(x-\ell)}}\right)\Bigg|^{x=\ell}_{x=\infty}\intertext{Recognizing that in this context $0^0 = 1$, we have then,}
        &=\frac{(\mu-\lambda)e^{-(\mu-\lambda)\ell}}{\mu-\lambda+2\lambda p}\left(\frac{2\lambda p}{\mu-\lambda+2\lambda p}\right)^k\left(1 - 0\right)\\
        &= \frac{\mu-\lambda}{\mu-\lambda+2\lambda p}\left(\frac{2\lambda p}{\mu-\lambda+2\lambda p}\right)^ke^{-(\mu-\lambda)\ell}.
    \end{align*}

We also emphasize that all of the results developed in this section naturally extend to the random-threshold setting introduced in Section~\ref{sec:prob-infec}. <!-- proposition-1: body_md -->
\label{prop:mg1bound}
Let $\mathcal{S}$ be a nonnegative random variable and let $\ell \ge 0$.  
Then the following bounds hold:
\[
\left( \mathbb{E}[\mathcal{S}] - \ell \right)^+
\;\le\;
\mathbb{E}\!\left[(\mathcal{S} - \ell)^+\right]
\;\le\;
\mathbb{E}[\mathcal{S}] - \ell\,\mathbb{P}(\mathcal{S} > \ell)
\;\le\;
\mathbb{E}[\mathcal{S}].
\]
\begin{proof}
We start from the identity
\[
(\mathcal{S} - \ell)^+ 
= \mathcal{S} \mathbf{1}_{\{\mathcal{S} > \ell\}} 
- \ell\,\mathbf{1}_{\{\mathcal{S} > \ell\}}.
\]

Taking expectations gives
\[
\mathbb{E}\!\left[(\mathcal{S} - \ell)^+\right]
= \mathbb{E}\!\left[\mathcal{S} \mathbf{1}_{\{\mathcal{S} > \ell\}}\right]
- \ell\,\mathbb{P}(\mathcal{S} > \ell).
\]

Since 
\[
\ell\,\mathbb{P}(\mathcal{S} > \ell) 
\le 
\mathbb{E}\!\left[\mathcal{S} \mathbf{1}_{\{\mathcal{S} > \ell\}}\right]
\le 
\mathbb{E}[\mathcal{S}],
\]
we immediately obtain the upper bound
\[
0 \le 
\mathbb{E}\!\left[(\mathcal{S} - \ell)^+\right]
\le 
\mathbb{E}[\mathcal{S}] - \ell\,\mathbb{P}(\mathcal{S} > \ell).
\]

For a lower bound, note that
\[
\mathbb{E}\!\left[\mathcal{S} \mathbf{1}_{\{\mathcal{S} > \ell\}}\right]
= \mathbb{E}[\mathcal{S}]
- \mathbb{E}\!\left[\mathcal{S} \mathbf{1}_{\{\mathcal{S} \le \ell\}}\right]
\ge 
\mathbb{E}[\mathcal{S}] - \ell\big(1 - \mathbb{P}(\mathcal{S} > \ell)\big),
\]
which implies
\[
\mathbb{E}\!\left[(\mathcal{S} - \ell)^+\right]
\ge 
\mathbb{E}[\mathcal{S}] - \ell.
\]

Combining both results yields the stated inequalities.
\end{proof}
