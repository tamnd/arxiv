<!-- theorem-1: body_md -->
Suppose the initial density is the unit Gaussian with mean $X(0)=0$, and the final destination choices are symmetric about the origin, $y_2=-y_1$. Then, $G$ is odd symmetric about $v=0.5$, i.e., $G(1-v)=1-G(v)$.
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
\textbf{Step 1: Symmetry of the mean $X(t;v)$}

With $y_1=-y_2$, and $X(0)=0$, Eq.\eqref{X mean} yields
\begin{equation}\label{X sym}
X(t;v)=R_2(t)\big(vy_1-(1-v)y_1\big)
      =R_2(t)(2v-1)y_1,
\end{equation}
and 
\begin{equation}
X(t;1-v)=R_2(t)\big((1-v)y_1+v y_2\big)
          =R_2(t)\big((1-v)y_1-vy_1\big)
          =R_2(t)(1-2v)y_1=-X(t;v). \label{eq:X_sym}
\end{equation}

\textbf{Step 2: Symmetry of the Riccati parameters \(s_j\) and \(d_j\)}

Recalling Eq.\eqref{hjb riccati2}, and using Eq. \ref{eq:X_sym}, we have
\begin{equation*}
\dot{s}_1(t;1-v)
=\left(\frac{b^2}{r}p(t)-a\right)s_1(t;1-v)-qX(t;v),
\end{equation*} 
and
\begin{equation*}
\dot{s}_2(t;v)
=\left(\frac{b^2}{r}p(t)-a\right)s_2(t;v)+qX(t;v).
\end{equation*}
The terminal conditions satisfy the relation $s_1(T;1-v)=-Cy_1=-s_2(T;v)$. This implies that $s_1(t;1-v)+s_2(t;v)$ solves a homogeneous linear ODE with zero terminal value, and hence, $s_1(t;1-v)=-s_2(t;v)$. Similar arguments can be employed to show that $d_1(t;1-v)=d_2(t;v)$. 


\textbf{{\color{black}  Step 3: Symmetry under action $v\rightarrow 1-v$ and $x\rightarrow -x$ with label exchange}}

For an arbitrary function $\mathcal{F}(t,x;v)$, we define the group action $\sigma \mathcal{F}(t,x;v):=\mathcal{F}(t,-x;1-v)$. The results of steps 1 and 2, in conjunction with definitions of Section \ref{subsec:map} can be used to conclude that $\sigma F_i=F_{3-i}$ for $V_i,g_i,\gamma_i$, while $\sigma F_i=-F_{3-i}$ for $u_i$ and $X_i$, for $i=1,2$.

\textbf{Step 4: Even symmetry of $f^*(t,x)$ under $\sigma$}

 We say that $\mathcal{F}$ is even under $\sigma$ if $\sigma\mathcal{F}=\mathcal{F}$, and odd if $\sigma\mathcal{F}=-\mathcal{F}$. The results in step 3 imply that $u^*$ is odd. A Fokker-Planck equation of the form given in Eq. \ref{FP f*_v} is equivariant w.r.t. $\sigma$ if the drift is \emph{odd} under $\sigma$. From step 3 and oddness of the linear term, the total drift $ax+bu^*$ is indeed odd. Since the initial condition is even under $\sigma$, i.e., $f^*(0,-x;1-v)=f^*(0,x;v)$, equivariance implies that $f^*(t,-x;1-v)=f^*(t,x;v)$ for all $t>0$. Using this relationship, and integrating over the positive reals,

\begin{equation*}
\int_0^\infty f^*(T,-x;1-v)\,dx
=
\int_0^\infty f^*(T,x;v)\,dx.
\end{equation*}
By a change of variable $x\rightarrow -x$ on the l.h.s, we obtain the desired result $G(1-v)=1-G(v)$.

<!-- lemma-1: body_md -->
For each $T_0\le T$, let $U_{T_0}$ denote the space of smooth controls $u(x,t)$
$$
U_{T_0}
=
\left\{
u(x,t)\;:\;
\sup_{(x,t)} \frac{|u(x,t)|}{1+|x|}<\infty,
\quad
|\partial_x u(x,t)|\le C_{T_0}
\right\},
$$
equipped with the norm
$
\|u\|_{U_{T_0}}
=
\sup_{(x,t)} \dfrac{|u(x,t)|}{1+|x|}.
$ Then for each $T_0<T$, the map $G_2:U_{T_0}\mapsto C([0,T_0],L^1)$ is Fréchet differentiable at $u^*(t,x;v^*,q)$, and the derivative $\hat{f}(t,x;v^*,q) \triangleq DG_2|_{u^*}\hat{u}$ satisfies the following PDE,
\begin{equation}\label{FP fhat}
    \partial_t\hat{f}=\frac{\sigma^2}{2}\,\partial_{xx}\hat{f}
-\partial_x\Bigl((ax+bu^*(t,x;v^*,q))\,\hat{f} +b\hat{u}(t,x;v^*,q)\,f^*(t,x;v^*,q)\Bigr), 
\end{equation}
with $\hat f(0,x)=0.$  \label{lem:fhat}

We mollify the non-smooth terminal data \cite{evans2022partial}, and construct the corresponding classical bounded, smooth solution $\Psi_{\epsilon}$. Direct differentiation yields
\begin{equation*}\label{zdot2}
    \dot{Z}=\int_\mathbb{R} \big((\partial_t\hat{f})\Psi  +\hat{f}\partial_t\Psi\big)\,dx.
\end{equation*}
From Eq. \ref{FP fhat}, $\partial_t\hat{f}=\mathcal{L}\hat{f}-\partial_x(b\hat{u}f^*)$. Hence,
\begin{align*}
\dot{Z}=\int_\mathbb{R} \big(\mathcal{L}[\hat{f}]\Psi  +\hat{f}\partial_t\Psi\big)\,dy-\int_\mathbb{R} \Psi(t,x)\partial_x(b \hat{u}(t,x)f^*(t,x))dx.
\end{align*}
Using Eq. \ref{eq:Psidot} in the first integral, and applying integration by parts on the second integral, 
\begin{align*}
\dot{Z}=\int_\mathbb{R} \big(\mathcal{L}[\hat{f}]\Psi  -\hat{f}\mathcal{L}_A\Psi)\,dx+\int_\mathbb{R} b \hat{u}(t,x)f^*(t,x)\partial_x(\Psi(t,x))dx.
\end{align*}
But the first integral is $0$ by using the property of adjoints, hence we obtain the desired ODE. The initial condition $Z(0)=0$ follows from the fact that $\hat{f}(0,x)=0$.

<!-- theorem-3: body_md -->
For $q>q_0$, the map $G$ has at least two asymmetric fixed points, in addition to the symmetric fixed point $v^*=0.5$.
<!-- end theorem-3 -->
<!-- proof-5: body_md -->
Consider the function $F(v)\triangleq G(v;q)-v$ for any fixed $q>q_0$. Each root of $F$ is a fixed point of $G$. It follows from Theorem \ref{th_main} that $F(v^*)=0$ and $F'(v^*)>0$. This implies for small enough $\delta>0$, $F(v^*-\delta)<0$, and $F(v^*+\delta)>0$. Since $0\leq G(v;q)\leq 1$, there are two possibilities: 
\begin{itemize}
\par\hangindent 40\p@{Case 1: $F(0)>0$ and $F(1)< 0$}\\
In this case, by continuity of $F$, there must exist a fixed point $v_1$ with $0<v_1<v^*-\delta$. Similar argument guarantees the existence of another fixed point $v_2$ with $v^*+\delta\leq v_2 \leq 1$.
\par\hangindent 40\p@{Case 2: $F(0)=0$ and $F(1)=0$}\\
In this case, $v_1=0$ and $v_2=1$ are the two additional fixed points.
\end{itemize}
This concludes the proof. By odd symmetry of $G$, we may choose $v_2=1-v_1$.

<!-- proof-6: body_md -->
Fix $T_0<T$. From Remarks \ref{rem:gamma1}, and $\ref{rem:gammahat1}$, $u^*\in U_{T_0}$, and $\hat{u}\in U_{T}$. Now let $\tilde{f}$ be the solution of the Fokker-Planck Eq. \ref{FP f*_v} with control $\tilde{u}=u^*+\hat{u}$. Hence, $\tilde{u}\in U_{T_0}$, which implies the total drift $ax+b\tilde{u}$ also grows at most linearly in $x$, and therefore, $\tilde{f}$ has finite second moment. 
    
    The evolution of the difference $\delta f\triangleq \tilde{f}-f^*$ is given by  \begin{equation}
        \partial_t\delta f=\mathcal{L}_{v^*}(\delta f){\color{black}  -}\partial_x(b\hat{u}\tilde{f}),\nonumber
    \end{equation}
    with $\delta f(0,x)=0$, and $L_{v^*}$ given by Eq. \ref{eq:L*}.
    Using Duhamel, \begin{equation}
    \delta f(t,x)= {\color{black}  -}\int_0^t \mathcal{U}_*(t,s)\partial_x(b\hat{u}(s,x)\tilde{f}(s,x))ds,\nonumber
     \end{equation}
     where $\mathcal{U}_*$ is the propagator associated with $\mathcal{L}_{v^*}$.
{\color{black}  Standard estimates and the results of Ref.  \cite{menozzi2021density} applied to the controlled drift yield for $m=0,1$
                  
                                                                                              
                     
\begin{align} \nonumber
    \int_{\mathbb R}(1+|x|)^m
\left|[U_*(t,s)\partial_x\phi](x)\right|dx
\le \frac{C}{\sqrt{t-s}}
\int_{\mathbb R}(1+|x|)^m|\phi(x)|dx,\\
\qquad 0\le s<t\le T_0.
\end{align}}

Then for all $t\leq T_0$, 
\begin{align}
    \int_{\mathbb{R}} (1+|x|)|\delta f(t,x)|dx<C\int_{\mathbb{R}} \int_0^t \dfrac{1}{\sqrt{t-s}}\dfrac{|\hat{u}(s,x)|}{1+|x|}(1+|x|)^2\tilde{f}(s,x)dsdx\nonumber\\
    =C\int_{\mathbb{R}}\int_0^t \dfrac{1}{\sqrt{t-s}}\dfrac{|\hat{u}(s,x)|}{1+|x|}(1+|x|)^2\tilde{f}(s,x)dsdx<C\|\hat{u}\|_{U_{T_0}},\label{eq:deltaf}
     \end{align}
where in the last step we have used the fact that $\tilde{f}$ has a finite second moment.

Next, we consider Fréchet remainder $r\triangleq \tilde{f}-f^*-\hat{f}$, whose evolution equation is \begin{align}
     \partial_tr=\mathcal{L}_{v^*}(r){\color{black}  -}\partial_x(b\hat{u}\delta{f}),\nonumber
\end{align}
with $r(0,x)=0$.

Proceeding similarly as above, 
\begin{align}
\|r(t,.)\|_{L^1}=\int_{\mathbb{R}} |r(t,x)|dx<C\int_{\mathbb{R}} \int_0^t \dfrac{1}{\sqrt{t-s}}\dfrac{|\hat{u}(s,x)|}{1+|x|}(1+|x|)|\delta{f}(s,x)|dsdx\nonumber\\  
< C\|\hat{u}\|_{U_{T_0}}\int_{\mathbb{R}}\int_0^t\dfrac{1}{\sqrt{t-s}}(1+|x|)|\delta{f}(s,x)|dsdx <C\|\hat{u}\|_{U_{T_0}}^2\nonumber,
\end{align}
where we have used Eq. \ref{eq:deltaf} in the last step. Since $\|r(t,.)\|_{L^1}=o(\|\hat{u}\|_{U_{T_0}})$, the claim is proved.
