<!-- theorem-1: body_md -->
\label{thm:main}
Let $m\geq 1$ and let $f\not\equiv0$ be analytic in $\mathbb{D}$, having a zero of order $m$ at the origin and satisfying $D(f)\leq 1$. Then
\begin{equation}\label{eq:main}
\int_0^{2\pi} \exp\bigl(|f(e^{i\theta})|^2\bigr) \,\frac{\,\mathrm{d}\theta}{2\pi} \leq \exp\!\left(\frac{D(f)} m\right).
\end{equation}
with equality if and only if $f(z)=cz^m$ for some $c\in\mathbb{C}$ with $0<|c|\leq 1/\sqrt{m}$.
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{cor:sharp}
For every analytic function $f$ in $\mathbb{D}$ with $f(0)=0$ and $D(f)\leq1$,
\begin{equation}\label{eq:sharp}
    \int_0^{2\pi}\exp\bigl(|f(e^{i\theta})|^2\bigr) \, \frac{\,\mathrm{d}\theta}{2\pi}\leq e.
\end{equation}
Equality holds if and only if $f(z)=e^{i\alpha}z$ for some $\alpha\in\mathbb{R}$.
<!-- end corollary-1 -->
<!-- lemma-1: body_md -->
\label{lem:iso}
Let $\gamma$ be a closed continuously differentiable planar curve. Then
\begin{equation}\label{eq:iso}
 4\pi|\mathcal A_\gamma|\leq L_\gamma^2.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
If $L_\gamma=0$, there is nothing to prove. Otherwise, we take a Lipschitz continuous arclength parametrization $\eta:[0,L_\gamma]\to\mathbb{C}$. After translating $\gamma$ if necessary, which does not change $\mathcal A_\gamma$ and $L_\gamma$, we may assume that $\int_0^{L_\gamma} \eta(t)\,\mathrm{d} t=0$. Cauchy--Schwarz and the periodic Wirtinger inequality give
\[
 |\mathcal A_\gamma|
 \leq\frac12\|\eta\|_{L^2(0,L_\gamma)}\|\eta'\|_{L^2(0,L_\gamma)}
 \leq\frac{L_\gamma}{4\pi}\|\eta'\|_{L^2(0,L_\gamma)}^2
 =\frac{L_\gamma^2}{4\pi}.
\]
This gives the claimed inequality.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:geometric}
Suppose that $f$ is analytic in a neighborhood of $\overline\mathbb{D}$, with $f(0)=0$ and $D(f)=1$. Then
\begin{equation}\label{eq:geometric}
 \int_\mathbb{T} e^{|f|^2}\operatorname{Re}(zf'\overline f) \,\mathrm{d}\mu \leq
 \left(\int_\mathbb{T} e^{|f|^2/2} |f'||f| \,\mathrm{d}\mu \right)^2.
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
We set $w(\theta):=f(e^{i\theta})$ and
\begin{equation}\label{eq:phase}
 \phi(\theta) :=\int_0^\theta
       \operatorname{Im}\bigl(\overline{w(\theta')}w'(\theta')\bigr)\,\mathrm{d} \theta' \,.
\end{equation}
Since $w'(\theta)=ie^{i\theta}f'(e^{i\theta})$, Parseval's identity gives
\[
 \phi(2\pi) =2\pi\sum_{n\geq1}n|a_n|^2=2\pi\, D(f) = 2\pi \,.
\]
This, together with $\phi(0)=0$, shows that
\begin{equation}\label{eq:curve}
 \gamma(\theta):=\exp\!\left(\frac{|w(\theta)|^2}2+i\phi(\theta)\right)
 \qquad\text{for all}\ 0\leq\theta\leq2\pi \,,
\end{equation}
defines a closed smooth curve in the complex plane. Moreover,
\begin{equation}\label{eq:curve-derivative}
 \gamma'=\gamma\left(\frac{(|w|^2)'}2+i\phi'\right)
          =\gamma\,\overline w\,w' \,,
\end{equation}
because $(|w|^2)'/2=\operatorname{Re}(\overline w w')$ and $\phi'=\operatorname{Im}(\overline w w')$. The length and signed area of $\gamma$ are therefore
\begin{align}
 L_\gamma &=\int_0^{2\pi} |\gamma'| \,\mathrm{d}\theta =
 \int_0^{2\pi}e^{|w|^2/2}|w||w'|\,\mathrm{d}\theta 
 = 2\pi \int_\mathbb{T} e^{|f|^2/2}|f||f'|\,\mathrm{d}\mu \,,
 \label{eq:length}\\
 \mathcal A_\gamma
 &=\frac12\int_0^{2\pi}\operatorname{Im}(\overline\gamma\gamma')\,\mathrm{d}\theta
 =\frac12\int_0^{2\pi}e^{|w|^2} \operatorname{Im}(\overline w w')\,\mathrm{d}\theta 
 = \pi \int_\mathbb{T} e^{|f|^2}\operatorname{Re}(zf'\overline f) \,\mathrm{d}\mu \,.\label{eq:area}
\end{align}
Applying Lemma~\ref{lem:iso} gives
\[
\int_\mathbb{T} e^{|f|^2}\operatorname{Re}(zf'\overline f) \,\mathrm{d}\mu
= \frac1\pi \mathcal A_\gamma
\leq \frac1\pi |\mathcal A_\gamma|\leq\frac{L_\gamma^2}{4\pi^2}
 =\left(\int_\mathbb{T} e^{|f|^2/2}|f||f'|\,\mathrm{d}\mu\right)^2,
\]
which is the claimed inequality.
<!-- end proof-2 -->
<!-- proposition-1: body_md -->
\label{prop:monotonicity}
Let $f\not\equiv0$ be analytic in $\mathbb{D}$ with $f(0)=0$. Then
\[
r \mapsto J \Bigl( \frac{f(r\,\cdot)}{\sqrt{D(f(r\,\cdot))}} \Bigr)
\]
is nonincreasing on the interval $(0,1)$ and
\[
\lim_{r\downarrow0} J \Bigl( \frac{f(r\,\cdot)}{\sqrt{D(f(r\,\cdot))}} \Bigr) = e^{1/m} \,,
\]
where $m$ is the order of the zero of $f$ at the origin. In particular,
\begin{equation}\label{eq:radial-bound}
 J\Bigl( \frac{f(r\,\cdot)}{\sqrt{D(f(r\,\cdot))}} \Bigr)\leq e^{1/m}\qquad\text{for all}\ 0<r<1 \,.
\end{equation}
<!-- end proposition-1 -->
<!-- proof-3: body_md -->
Writing $f(z)=\sum_{n\geq m}a_nz^n$ with $a_m\neq0$, we have
\begin{equation}
    \label{eq:dfr}
    D(f(r\,\cdot)) =\sum_{n\geq m}n|a_n|^2r^{2n} \,.
\end{equation}
This is finite and positive for every $0<r<1$. For each $0<r<1$, the function $g_r$ defined by
\begin{equation}\label{eq:normalization}
 g_r(z):=\frac{f(rz)}{\sqrt{D(f(r\,\cdot))}}
\end{equation}
is analytic in a neighborhood of $\overline\mathbb{D}$, satisfies $g_r(0)=0$, and has $D(g_r)=1$. 

We compute, using Parseval's identity,
\[
A_r:=\int_\mathbb{T} |g_r'(e^{i\theta})|^2 \,\frac{\,\mathrm{d}\theta}{2\pi} = \frac{\sum_{n\geq m}n^2|a_n|^2r^{2n}}{D(f(r\,\cdot))} \,.
\]
This, together with \eqref{eq:dfr}, gives
\[
r\frac{\mathrm d}{\mathrm dr}D(f(r\,\cdot)) = 2 \, D(f(r\,\cdot))\, A_r \,.
\]
As a consequence, differentiating \eqref{eq:normalization}, we obtain
\begin{equation}\label{eq:radial-equation}
 r\partial_rg_r=zg_r'-A_rg_r \,,
\end{equation}
where $g_r'$ denotes differentiation with respect to $z$. To compute the $r$-derivative of $J(g_r)$ we differentiate under the integral sign, which is justified by analyticity. We obtain
\begin{align}
 r\frac{\mathrm d}{\mathrm dr}J(g_r)
 &=2\int_\mathbb{T} e^{|g_r|^2}
   \operatorname{Re}\bigl(\overline{g_r}(zg_r'-A_rg_r)\bigr) \,\mathrm{d}\mu \,. \label{eq:monotonicity}
\end{align}
It follows from Lemma~\ref{lem:geometric}, applied to $g_r$, together with the Cauchy--Schwarz inequality, that
\begin{align}
    \label{eq:monotonicitycs}
    \int_\mathbb{T} e^{|g_r|^2}\operatorname{Re}(zg_r'\overline{g_r}) \,\mathrm{d}\mu \leq
    \left(\int_\mathbb{T} e^{|g_r|^2/2} |g_r'||g_r| \,\mathrm{d}\mu \right)^2 \leq A_r \int_\mathbb{T} e^{|g_r|^2} |g_r|^2 \,\mathrm{d}\mu \,.
\end{align}
Thus the right side of \eqref{eq:monotonicity} is nonpositive, proving the claimed monotonicity.

Finally, uniformly on $\overline\mathbb{D}$,
\[
 g_r(z)\longrightarrow\frac{a_m}{|a_m|}\frac{z^m}{\sqrt m}
 \qquad\text{as }r\downarrow0.
\]
It follows that $\lim_{r\downarrow0}J(g_r)=e^{1/m}$.
<!-- end proof-3 -->
<!-- proposition-2: body_md -->
\label{prop:monotonicitysharp}
    In the setting of Proposition \ref{prop:monotonicity}, if equality holds in \eqref{eq:radial-bound} for some $r = r_0 \in(0,1)$, then $f(z) = cz^m$ for some $c\in\mathbb{C}\setminus\{0\}$.

<!-- proof-4: body_md -->
We consider again $g_r$, defined by \eqref{eq:normalization}. By Proposition \ref{prop:monotonicity}, the assumed equality implies that
\[
 J(g_r)=e^{1/m}
 \qquad\text{for all}\ 0<r\leq r_0 \,.
\]
    Vanishing of the left side of \eqref{eq:monotonicity} at any $r\in(0,r_0)$ implies equality in the Cauchy--Schwarz inequality in \eqref{eq:monotonicitycs} and therefore gives a constant $C_r>0$ such that
\begin{equation}\label{eq:CS-equality}
 |g_r'|^2=C_r|g_r|^2e^{|g_r|^2}\qquad\text{on }\mathbb{T}.
\end{equation}
The equality holds pointwise because both sides are continuous.

Note that $zf'(z)/f(z)$ tends to $m\neq 0$ as $|z|\to 0$. (Indeed, this follows by writing $f(z)=z^mu(z)$, where $u$ is analytic and $u(0)=a_m\neq0$, and noting that $zf'(z)/f(z) = m + zu'(z)/u(z)$. Consequently, there is a $\rho>0$ such that the function
\[
    T(z):=\frac{zf'(z)}{f(z)}
\]
is analytic and nonzero for $|z|<\rho$. Choose the analytic logarithm with value zero at the origin and write
\begin{equation}\label{eq:log-T}
 \log\frac{T(z)}m=\sum_{k\geq1}\ell_kz^k.
\end{equation}
For $0<r<\min\{r_0,\rho\}$, division of \eqref{eq:CS-equality} by $|g_r|^2$ and use of \eqref{eq:normalization} yield
\begin{equation}\label{eq:log-equality}
 \log|T(re^{i\theta})|^2
 =\log C_r+\frac{|f(re^{i\theta})|^2}{D(f(r\,\cdot))} \,.
\end{equation}
There are no zeros on these sufficiently small circles, so all divisions are legitimate.

Writing
\[
\log|T(re^{i\theta})|^2
= 2\log m+
2\operatorname{Re}\log\frac{T(re^{i\theta})}{m}
=2\log m+\sum_{k\ge1}
\left(\ell_kr^ke^{ik\theta}
+\overline{\ell_k}r^ke^{-ik\theta}\right),
\]
we see that the coefficient of $e^{ik\theta}$ on the left side of \eqref{eq:log-equality} is $\ell_kr^k$. On the right side, we write again $f(z) = \sum_{n\geq m} a_n z^n$ with $a_m\neq 0$ and find that the coefficient of $e^{ik\theta}$ for $k\geq1$ is
\[
 \frac1{D(f(r\,\cdot))}\sum_{n\geq m}a_{n+k}\overline{a_n}\,r^{2n+k} \,.
\]
Thus,
\[
\ell_k \, D(f(r\,\cdot)) = \sum_{n\geq m}a_{n+k}\overline{a_n}\,r^{2n} \,.
\]
Inserting the power series expansion for $D(f(r\,\cdot))$ from \eqref{eq:dfr} and comparing coefficients, we find that
\begin{equation}\label{eq:coefficient-relation}
 a_{n+k}\overline{a_n}
 =n\ell_k|a_n|^2
 \qquad\text{for all}\ n\geq m\ \text{and}\ k\geq1 \,.
\end{equation}
Equivalently, whenever $a_n\neq0$,
\begin{equation}\label{eq:recurrence}
 a_{n+k}=n\ell_ka_n.
\end{equation}

We argue by contradiction and assume that $a_{m+j}\neq0$ for some $j\geq1$. Taking $n=m$ and $k=j$ in \eqref{eq:recurrence} shows that $\ell_j\neq0$. Applying the same relation successively gives
\[
 a_{m+j}=m\ell_ja_m,
 \qquad
 a_{m+2j}=m(m+j)\ell_j^2a_m\neq0 \,.
\]
On the other hand, the relation with $n=m$ and $k=2j$ gives
\[
 \ell_{2j}=(m+j)\ell_j^2 \,.
\]
We can now compute $a_{m+3j}$ in two ways: with $(n,k)=(m+j,2j)$ and with $(n,k)=(m+2j,j)$. These give, respectively,
\[
 a_{m+3j}=m(m+j)^2\ell_j^3a_m,
 \qquad
 a_{m+3j}=m(m+2j)(m+j)\ell_j^3a_m.
\]
Since $j\geq1$ and $\ell_ja_m\neq0$, this is a contradiction. Hence $a_n=0$ for every $n>m$, and $f$ is a monomial of degree $m$, as claimed.
