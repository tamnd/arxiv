<!-- theorem-1: body_md -->
\label{thm:main-results}
	
	Let $\mu>0$ and let $(\beta_n,\gamma_n)$ be the monic recurrence
	coefficients of the generalized Charlier polynomials associated with
	the discrete weight
	\[
	\rho_\mu(k)=\frac{\mu^k}{(k!)^2}, \qquad k\in\mathbb N_0.
	\]
	Define the positive sequence $(c_n)_{n\geq0}$ by the parametrization
	\[
	\beta_n-n=\sqrt{\mu}\,c_nc_{n+1}, \qquad \gamma_n=\mu(1-c_n^2).
	\]
	Then:
	
	\begin{enumerate}
		\item[(i)] The sequence $(c_n)$ satisfies the discrete Painlev\'e II
		equation
		\[
		n c_n=\sqrt{\mu}\,(c_{n+1}+c_{n-1})(1-c_n^2), \qquad n\geq1,
		\]
		and converges to zero as $n\to\infty$.
		
		\item[(ii)] There exists a finite non-negative limit
		\[
		L(\mu)=\lim_{n\to\infty}\frac{c_n}{J_n(2\sqrt{\mu})}.
		\]
		
		\item[(iii)] The connection constant is exactly $L(\mu)=1$, so that
		\begin{equation}
			c_n \sim J_n(2\sqrt{\mu}), \qquad n\to\infty.
			\label{eq:main-connection-formula}
		\end{equation}
		
		\item[(iv)] The recurrence coefficients admit the complete factorial
		asymptotics
		\begin{align}
			\beta_n-n &\sim \frac{\mu^{n+1}}{n!(n+1)!}, \qquad n\to\infty,
			\label{eq:main-beta-asymptotic}\\
			\mu-\gamma_n &\sim \frac{\mu^{n+1}}{(n!)^2}, \qquad n\to\infty.
			\label{eq:main-gamma-asymptotic}
		\end{align}
		
		\item[(v)] The Hankel determinant $\Delta_n(\mu)$ associated with
		$\rho_\mu$ factorizes as
		\[
		\Delta_n(\mu)=\mu^{n(n-1)/2}\mathcal T_n(\mu),
		\]
		where $\mathcal T_n(\mu)=1+\sum_{q=1}^{n}\mathcal S_q(\mu)$ admits a
		sector decomposition with sector contributions $\mathcal S_q(\mu)$
		independent of $n$. Moreover,
		\begin{equation}
			\mathcal S_q(\mu) \sim e^\mu\,\frac{\mu^q}{(q!)^2}, \qquad q\to\infty,
			\label{eq:main-sector-asymptotic}
		\end{equation}
		and
		\begin{equation}
			\mathcal T_n(\mu) \longrightarrow e^\mu, \qquad n\to\infty.
			\label{eq:main-tau-limit}
		\end{equation}
	\end{enumerate}

		index. Suppose that, for some $n\geq1$,
		\[
		c_n\geq c_{n-1},
		\qquad
		c_n\geq c_{n+1}.
		\]
		Then
		\[
		c_{n-1}+c_{n+1}
		\leq
		2c_n.
		\]
		Using \eqref{eq:cn-monotonicity-recurrence}, we obtain
		\[
		n c_n
		=
		\sqrt{\mu}\,(c_{n+1}+c_{n-1})(1-c_n^2)
		\leq
		2\sqrt{\mu}\,c_n(1-c_n^2)
		<
		2\sqrt{\mu}\,c_n.
		\]
		Since $c_n>0$, this implies
		\[
		n<2\sqrt{\mu}.
		\]
		Thus, no local maximum can occur at an index satisfying

			n\geq2\sqrt{\mu}.
			\label{eq:no-large-local-maximum}

		zero. Therefore, after any such nondecreasing step, the sequence must
		eventually decrease. Hence there exists an index $k\geq m+1$ such that

		c_k\geq c_{k-1},
		\qquad
		c_k\geq c_{k+1}.
		\]
		Thus $c_k$ is a local maximum. Taking $m$ sufficiently large gives
		\[
		k\geq2\sqrt{\mu},

		which contradicts \eqref{eq:no-large-local-maximum}.

		\[
		c_{n+1}<c_n,
		\qquad
		n\geq N_\mu.
		\]
		Together with $0<c_n<1$, this proves
		\eqref{eq:eventual-strict-monotonicity}.

By Proposition~\ref{prop:eventual-monotonicity}, after increasing
		$N_\mu$ if necessary, we have
		\[
		0<c_{n+1}<c_n<1,
		\qquad
		n\geq N_\mu.

		In particular,
		\[
		c_{n+1}\leq c_{n-1},
		\qquad
		n\geq N_\mu+1.

		\[
		n c_n
		=
		\sqrt{\mu}\,(c_{n+1}+c_{n-1})(1-c_n^2)
		\leq
		2\sqrt{\mu}\,c_{n-1}.

		\begin{equation}
			r_n
			=

			\frac{c_{n+1}}{c_{n-1}}
			=
			r_nr_{n+1}

		r_n
		\leq
		\frac{\sqrt{\mu}}{n}

		1+r_nr_{n+1}
		\right).

			r_n
			\leq

			n\geq N_\mu+1.

		c_n

	Fix $\mu>0$, and let
	\[
	j_n

		0<c_n\leq A_\mu j_n,

		\label{eq:bounded-Qn-assumption}
	\end{equation}

		\lim_{n\to\infty}W_n=0.
		\label{eq:Wn-limit-zero}
	\end{equation}

	a finite nonnegative limit

		L(\mu)

	\end{equation}

		\leq

		n\geq N_\mu.

	\end{equation}

		c_n
		=

\label{prop:partition-function-representation}

<!-- proof-7: body_md -->
For $0\leq j\leq n-1$ and $k\geq0$, define
	\[
	A_{j,k}
	=
	k^j\sqrt{w_\mu(k)}.
	\]
	The moment matrix is then given by
	\[
	\left[
	M_{j+\ell}(\mu)
	\right]_{j,\ell=0}^{n-1}
	=
	AA^{\mathsf T}.
	\]
	Indeed, for $0\leq j,\ell\leq n-1$,
	\[
	(AA^{\mathsf T})_{j,\ell}
	=
	\sum_{k=0}^{\infty}
	A_{j,k}A_{\ell,k}
	=
	\sum_{k=0}^{\infty}
	k^{j+\ell}w_\mu(k)
	=
	M_{j+\ell}(\mu).
	\]
	
	Applying the Cauchy--Binet formula gives
	\[
	\Delta_n(\mu)
	=
	\sum_{0\leq k_1<\cdots<k_n}
	\det(A_{[k_1,\ldots,k_n]})^2,
	\]
	where $A_{[k_1,\ldots,k_n]}$ denotes the $n\times n$ submatrix of
	$A$ formed by the columns indexed by
	\[
	k_1,\ldots,k_n.
	\]
	Its determinant is
	\[
	\det(A_{[k_1,\ldots,k_n]})
	=
	\left(
	\prod_{\ell=1}^{n}
	\sqrt{w_\mu(k_\ell)}
	\right)
	\det
	\begin{pmatrix}
		1 & 1 & \cdots & 1 \\
		k_1 & k_2 & \cdots & k_n \\
		k_1^2 & k_2^2 & \cdots & k_n^2 \\
		\vdots & \vdots & \ddots & \vdots \\
		k_1^{n-1} & k_2^{n-1} & \cdots & k_n^{n-1}
	\end{pmatrix}.
	\]
	The determinant in the preceding expression is the Vandermonde
	determinant. Hence,
	\[
	\det(A_{[k_1,\ldots,k_n]})
	=
	V(k_1,\ldots,k_n)
	\prod_{\ell=1}^{n}
	\sqrt{w_\mu(k_\ell)}.
	\]
	Therefore,
	\[
	\det(A_{[k_1,\ldots,k_n]})^2
	=
	V(k_1,\ldots,k_n)^2
	\prod_{\ell=1}^{n}
	w_\mu(k_\ell).
	\]
	Substitution into the Cauchy--Binet expansion proves
	\eqref{eq:Hankel-partition-function}.
	
	To obtain \eqref{eq:Hankel-partition-function-symmetric}, observe that
	the summand vanishes whenever two indices coincide, because then
	\[
	V(k_1,\ldots,k_n)=0.
	\]
	For pairwise distinct indices, the summand is invariant under
	permutations of $k_1,\ldots,k_n$. Each strictly increasing
	configuration is therefore counted exactly $n!$ times in the
	unrestricted sum. This proves the symmetric representation.

Fix $q$ and consider the configurations for which
	\[
	\lambda_1=\cdots=\lambda_{n-q}=0,
	\qquad
	\lambda_{n-q+a}=r_a,
	\qquad
	1\leq a\leq q,
	\]
	with
	\[

	\]

	\eqref{eq:reduced-tau-function} is obtained by separating the
	factorial and Vandermonde factors into three parts:
	
	\begin{enumerate}
		\item the factors involving only unexcited particles;
		
		\item the factors involving one unexcited and one excited particle;
		
		\item the factors involving two excited particles.
	\end{enumerate}

	associated with the $a$th excited particle is
	\begin{equation}
		\left(

		\right)^2.

	particles is
	\begin{align}
		\prod_{i=1}^{n-q}
		\left(
		\frac{n-q+a-i+r_a}{n-q+a-i}
		\right)^2

		\left[
		\prod_{d=a}^{n-q+a-1}

		\right]^2
		\nonumber\\
		&=
		\left[
		\frac{(n-q+a-1+r_a)!}
		{(a-1+r_a)!}

		{(n-q+a-1)!}
		\right]^2.
		\label{eq:sector-cross-interaction}

		\frac{(n-q+a-1)!}
		{(n-q+a-1+r_a)!}
		\right)^2
		\left[
		\frac{(n-q+a-1+r_a)!}
		{(a-1+r_a)!}

		{(n-q+a-1)!}
		\right]^2
		\nonumber\\
		&\hspace{4cm}

		\left(
		\frac{(a-1)!}

		\right)^2.
		\label{eq:sector-active-interaction}
	\end{equation}

	\[
	\mathcal T_n(\mu)
	=
	1+\sum_{q=1}^{n}\mathcal S_q(\mu),
	\]
	which proves the claim.
