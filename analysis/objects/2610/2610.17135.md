\label{thm:zeta-intro}
Let $\textrm{\upshape M}$ be a loopless matroid. If there is an integer $j\geqslant2$
such that
\[
Z_\textrm{\upshape M}(j)^2<Z_\textrm{\upshape M}(j-1)Z_\textrm{\upshape M}(j+1),

then for every sufficiently large integer $n$, the polynomials
\[
f(\textrm{\upshape M}^{\oplus n};t),\qquad h(\textrm{\upshape M}^{\oplus n};t),\qquad
\underline{\textrm{\upshape H}}(\textrm{\upshape M}^{\oplus n};t),\qquad \textrm{\upshape H}(\textrm{\upshape M}^{\oplus n};t)
\]
are not log-concave, and hence not real-rooted.

\label{thm:zeta-families}

\quad\Longleftrightarrow\quad q\geqslant81.

Z_{\textrm{\upshape G}_{4,q}}(3)^2
<Z_{\textrm{\upshape G}_{4,q}}(2)Z_{\textrm{\upshape G}_{4,q}}(4)
\quad\Longleftrightarrow\quad q\geqslant81.

\[
Z_{\textrm{\upshape C}(q,q^3)}(3)^2
<Z_{\textrm{\upshape C}(q,q^3)}(2)Z_{\textrm{\upshape C}(q,q^3)}(4)
\quad\Longleftrightarrow\quad q\geqslant149.

\underline{\textrm{\upshape H}}(\textrm{\upshape M}^{\oplus n};t),\qquad \textrm{\upshape H}(\textrm{\upshape M}^{\oplus n};t)

<!-- proposition-1: body_md -->
\label{prop:zeta-chain}
For $n\geqslant1$ and $j\geqslant0$,
\begin{align*}
f_{j-1}(\textrm{\upshape M}^{\oplus n})
&=\sum_{m=1}^{j+1}(-1)^{j+1-m}
\binom{j+1}{m}Z_\textrm{\upshape M}(m)^n,\\
h_j(\textrm{\upshape M}^{\oplus n})
&=\sum_{m=1}^{j+1}(-1)^{j+1-m}
\binom{nr+1}{j+1-m}Z_\textrm{\upshape M}(m)^n.
\end{align*}
<!-- end proposition-1 -->
<!-- proof-3: body_md -->
The first identity follows by binomial inversion from Equation
~\ref{eq:zeta-binomial} and Equation~\ref{eq:zeta-product}. For the second identity, the identity
\[
\sum_{m\geqslant0}\binom{m}{k+1}t^m= \frac{t^{k+1}}{(1-t)^{k+2}}
\] implies that 
\begin{align*}
\sum_{m\geqslant1}Z_{\textrm{\upshape M}^{\oplus n}}(m)t^{m-1}
&=\sum_{k=0}^{nr-1}f_{k-1}(\textrm{\upshape M}^{\oplus n})
  \frac{t^k}{(1-t)^{k+2}}
=\frac{h(\textrm{\upshape M}^{\oplus n};t)}{(1-t)^{nr+1}}.
\end{align*}
Multiply by $(1-t)^{nr+1}$, substitute
$Z_{\textrm{\upshape M}^{\oplus n}}(m)=Z_\textrm{\upshape M}(m)^n$, and compare coefficients.

<!-- lemma-3: body_md -->
\label{lem:constant-terms-multichain}
For any fixed $m \geqslant 1$, and a polynomial $Q_m(\textrm{\upshape M}; t)$ equal to 
\[
(\chi_{\textrm{\upshape M}}^{-m})(\widehat{0},\widehat{1};t) \quad \text{ or } \quad (\zeta_{\textrm{\upshape M}}^{\mathrm{rev}}
  \cdot\chi_{\textrm{\upshape M}}^{-m})(\widehat{0},\widehat{1};t),
\]
we have the constant terms 
\[
Q_m(\textrm{\upshape M};0)=\zeta_{\textrm{\upshape M}}^{m}(\widehat{0},\widehat{1})=Z_{\textrm{\upshape M}}(m),
\qquad
Q_m(\textrm{\upshape M}^{\oplus n};0) = Z_{\textrm{\upshape M}}(m)^n.
\]
<!-- end lemma-3 -->
<!-- proof-5: body_md -->
We first consider the case of a single matroid $\textrm{\upshape M}$. For $(\chi_{\textrm{\upshape M}}^{-m})(\widehat{0},\widehat{1};t)$, evaluation at $t=0$ commutes with convolution and therefore also with
inversion. Since ${\chi_{\textrm{\upshape M}}(0)=\mu_{\textrm{\upshape M}}}$ and
$\mu_{\textrm{\upshape M}}^{-1}=\zeta_{\textrm{\upshape M}}$, we obtain
\[
\chi_{\textrm{\upshape M}}^{-m}(0)
=\mu_{\textrm{\upshape M}}^{-m}
=\zeta_{\textrm{\upshape M}}^{m}.
\]
For $(\zeta_{\textrm{\upshape M}}^{\mathrm{rev}} \cdot\chi_{\textrm{\upshape M}}^{-m})(\widehat{0},\widehat{1};t)$, since
$\zeta_{\textrm{\upshape M}}^{\mathrm{rev}}(0)=\delta_{\textrm{\upshape M}}$, which is the multiplicative identity, the constant term is unchanged: 
\[
(\zeta_{\textrm{\upshape M}}^{\mathrm{rev}} \cdot\chi_{\textrm{\upshape M}}^{-m})(0) = \chi_{\textrm{\upshape M}}^{-m}(0) = \zeta^{m}_{\textrm{\upshape M}}.
\] Therefore, $Q_m(\textrm{\upshape M};0)$ is equal to 
\[
\zeta_{\textrm{\upshape M}}^{m}(\widehat{0},\widehat{1}) = \sum_{\widehat{0}=F_0\leqslant F_1\leqslant\cdots
        \leqslant F_m=\widehat{1}} \, \Big(\prod_{i=1}^{m} \zeta_{\textrm{\upshape M}}(F_{i-1}, F_i)\Big)
=\sum_{\widehat{0}=F_0\leqslant F_1\leqslant\cdots
        \leqslant F_m=\widehat{1}}1
=Z_{\textrm{\upshape M}}(m).
\]

We next apply the same argument to $\textrm{\upshape M}^{\oplus n}$. 
Since $\mathcal{L}(\textrm{\upshape M}^{\oplus n})\cong\mathcal{L}(\textrm{\upshape M})^n$, a multichain in this
product is determined by an independent choice of a multichain
in each coordinate. Therefore,
\[
Q_m(\textrm{\upshape M}^{\oplus n};0)
=Z_{\textrm{\upshape M}^{\oplus n}}(m)
=Z_{\textrm{\upshape M}}(m)^n.
\]

<!-- proposition-3: body_md -->
\label{prop:zeta-amplification}
Let $P(\textrm{\upshape M};t)$ denote any one of
\[
f(\mathcal{L}; t), \qquad f(\overline{\mathcal{L}}; t), \qquad h(\textrm{\upshape M};t),\qquad
\underline{\textrm{\upshape H}}(\textrm{\upshape M};t),\qquad \textrm{\upshape H}(\textrm{\upshape M};t),
\]
and write $P(\textrm{\upshape M};t)=\sum_jP_j(\textrm{\upshape M})t^j$.
For a fixed $j\geqslant1$, 
\begin{equation}\label{eq:zeta-amplification}
\lim_{n \to \infty} P_j(\textrm{\upshape M}^{\oplus n})
=Z_\textrm{\upshape M}(j+1)^n+O\bigl(n^jZ_\textrm{\upshape M}(j)^n\bigr).
\end{equation}
Furthermore, 
\[
\lim_{n \to \infty} \frac{P_j(\textrm{\upshape M}^{\oplus n})}{Z_\textrm{\upshape M}(j+1)^n} = 1. 
\]
<!-- end proposition-3 -->
<!-- proof-6: body_md -->
For $f(\textrm{\upshape M}; t)$ or $h(\textrm{\upshape M}; t)$, since $Z_{\textrm{\upshape M}}(m)$ is strictly increasing, \cref{prop:zeta-chain} implies that as $n$ goes to $\infty$, the $j$th coefficient of $f(\textrm{\upshape M}^{\oplus n}; t)$, which is $f_{j-1}(\textrm{\upshape M}^{\oplus n})$, or that of $h(\textrm{\upshape M}^{\oplus n}; t)$, which is $h_{j}(\textrm{\upshape M}^{\oplus n})$, is dominated by the term $Z_{\textrm{\upshape M}}(j+1)^n$. Therefore, we have the asymptotic ratios 
\[
\lim_{n \to \infty} \frac{f_{j-1}(\textrm{\upshape M}^{\oplus n})}{Z_{\textrm{\upshape M}}(j+1)^n} = 1, \qquad  \lim_{n \to \infty} \frac{h_{j}(\textrm{\upshape M}^{\oplus n})}{Z_{\textrm{\upshape M}}(j+1)^n} = 1.
\]
By \eqref{eq:pL}, the same limit is true for the $j$th coefficient of $f(\mathcal{L}; t)$. 

For $\underline{\textrm{\upshape H}}(\textrm{\upshape M}; t)$ and $\textrm{\upshape H}(\textrm{\upshape M}; t)$, if we write 
\[
Q_m(\textrm{\upshape M}^{\oplus n}; t) = \begin{cases}
    (\chi^{-m})(\widehat{0}, \widehat{1}; t) & \text{ for } \underline{\textrm{\upshape H}} \\ 
    (\zeta^{\mathrm{rev}} \cdot \chi^{-m})(\widehat{0}, \widehat{1}; t) & \text{ for } \textrm{\upshape H}
\end{cases}, 
\] then \cref{prop:zeta-chow} implies that 
\[
P(\textrm{\upshape M}^{\oplus n}; t) = \sum_{m \geqslant 1} (t^{m-1} - t ^m) Q_m(\textrm{\upshape M}^{\oplus n}; t). 
\]
Therefore, for $j \geqslant 1$, the $j$th coefficient of $P(\textrm{\upshape M}^{\oplus n}; t)$ is 
\[
P_j(\textrm{\upshape M}^{\oplus n}) = Q_{j+1}(\textrm{\upshape M}; 0)^n + \sum_{m = 1}^{j} \big(Q_{m, j+1-m}(\textrm{\upshape M}^{\oplus n}) - Q_{m, j-m}(\textrm{\upshape M}^{\oplus n})\big). 
\]
Then Lemma~\ref{lem:constant-terms-multichain} and Lemma~\ref{lem:chow-coefficient-error} imply that, as $n$ goes to infinity, the first term $Q_{j+1}(\textrm{\upshape M}; 0)^n = Z_{\textrm{\upshape M}}(j+1)^n$ dominates, and 
\[
\lim_{n \to \infty} \frac{P_j(\textrm{\upshape M}^{\oplus n})}{Z_{\textrm{\upshape M}}(j+1)^n } = 1.  
\]

<!-- lemma-4: body_md -->
\label{lem:chow-coefficient-error}
For either choice of $Q_m$, and for every fixed $j\geqslant1$, there exists a constant $C = C_{\textrm{\upshape M}, j}$ such that, as $n\to\infty$,
\[
\sum_{m=1}^{j}
\bigl(
Q_{m,j+1-m}(\textrm{\upshape M}^{\oplus n})
-Q_{m,j-m}(\textrm{\upshape M}^{\oplus n})
\bigr)
\leqslant C_{\textrm{\upshape M}, j} n^jZ_{\textrm{\upshape M}}(j)^n.
\]
<!-- end lemma-4 -->
<!-- proof-7: body_md -->
By multiplicativity and Lemma~\ref{lem:constant-terms-multichain},
\[
Q_m(\textrm{\upshape M}^{\oplus n};t)=Q_m(\textrm{\upshape M};t)^n,
\qquad
c_m:=Q_m(\textrm{\upshape M};0)=Z_{\textrm{\upshape M}}(m)>0.
\]
Write $Q_m(\textrm{\upshape M};t)=c_m(1+tB_m(t))$, where $B_m(t)$
is independent of $n$. The binomial theorem implies
\[
Q_{m,d}(\textrm{\upshape M}^{\oplus n})
=c_m^n
\sum_{\ell=0}^{\min(n,d)}
\binom n\ell B_{m, d-\ell}(t)^\ell,
\]
where the coefficients
$B_{m, d -\ell}(t)^\ell$ are independent of $n$.
Since $0\leqslant\ell\leqslant\min(n,d)$,
\[
\binom n\ell\leqslant\frac{n^\ell}{\ell!}
\leqslant\frac{n^d}{\ell!},
\] and  there exists a constant $C_{\textrm{\upshape M},m,d}$ such that
\[
|Q_{m,d}(\textrm{\upshape M}^{\oplus n})|
\leqslant C_{\textrm{\upshape M},m,d}\,n^dZ_{\textrm{\upshape M}}(m)^n.
\]
There are only finitely many pairs
$1\leqslant m\leqslant j$ and $0\leqslant d\leqslant j$,
so these bounds hold with a common constant $C=C(\textrm{\upshape M},j)$.
Therefore,
\[
\begin{aligned}
\left|
\sum_{m=1}^{j}
\bigl(
Q_{m,j+1-m}(\textrm{\upshape M}^{\oplus n})
-Q_{m,j-m}(\textrm{\upshape M}^{\oplus n})
\bigr)
\right| &\leqslant
C\sum_{m=1}^{j}
\bigl(n^{j+1-m}+n^{j-m}\bigr)Z_{\textrm{\upshape M}}(m)^n\\
&\leqslant
2jC\,n^jZ_{\textrm{\upshape M}}(j)^n. 
\end{aligned}
\] Updating the bounding constant $C$ gives the desired statement.

<!-- proposition-4: body_md -->
\label{amp:height}
There is a weight-preserving bijection between height functions on
$G$ and nested families $\mathcal S\subseteq\mathcal{B}(G)$ equipped with
positive integer multiplicities $\mu(T)$, under which
\[
 d(x)=\sum_{T\ni x}\mu(T),\qquad
 w_G(d)=\sum_{T\in\mathcal S}\mu(T).
\]
Consequently
\begin{equation}\label{amp:height-series}
 \Phi_G(t)=\sum_{\mathcal S}
                  \left(\frac{t}{1-t}\right)^{|\mathcal S|}
          =\prod_C\frac{h_C(t)}{(1-t)^{|C|}},
\end{equation}
where $C$ runs over the connected components of $G$. In particular,
for connected $G$,
\[
h_G(t)=(1-t)^{|V(G)|}\Phi_G(t).
\]
<!-- end proposition-4 -->
<!-- proof-9: body_md -->
Given $d$, collect all components of all positive superlevel sets,
and let $\mu(T)$ count the levels at which $T$ occurs. Components at
different levels are comparable or disjoint. Disjoint components
cannot be adjacent: an edge would connect them at the lower level.
Thus they form a nested family. Counting, for each vertex, the levels
at which it occurs gives the formula for $d$, and counting components
gives the formula for $w_H$.

Conversely, start with $(\mathcal S,\mu)$ and define $d$ by the displayed
formula. Put $D(T)=\sum_{U\supseteq T}\mu(U)$, with the sum over
$U\in\mathcal S$. For
$D(T)-\mu(T)<r\leqslant D(T)$, every vertex of $T$ has height at least $r$.
If a vertex outside $T$ is adjacent to $T$, every member of
$\mathcal S$ containing that vertex must properly contain $T$;
its height is therefore at most $D(T)-\mu(T)$. Hence $T$ is a component
of $\{d\geqslant r\}$. For each vertex $x$ with $d(x)\geqslant r$, the members
of $\mathcal S$ containing $x$ form a chain, and exactly one of their
successive multiplicity intervals contains $r$. These components
therefore cover the superlevel set, proving the bijection.

Summing the positive multiplicities gives the first equality in
\eqref{amp:height-series}. Nested families split independently over
components of $H$. For connected $G$, such a family is a tubing,
possibly with $V(G)$ adjoined. Setting $y=t/(1-t)$ therefore gives
$\Phi_G(t)=(1+y)\sum_{\mathcal T}y^{|\mathcal T|}$; now apply
\eqref{amp:h-definition}.

A vertex set containing the common root is connected exactly when
its intersection with every copy is connected. The intersections
recover the set by taking their union, and both operations preserve
inclusion. A multichain in a product is a tuple of multichains.

<!-- lemma-6: body_md -->
\label{amp:layers}
For each chain $\mathcal C:V(G)=S_0\supseteq S_1\supseteq\cdots
\supseteq S_a$ in $\mathcal{F}_v(G)$, put
\[
 U_\ell=S_\ell\setminus N[S_{\ell+1}]\quad(0\leq\ell<a),
 \qquad U_a=S_a\setminus\{v\},
 \qquad U(\mathcal C)=\bigcup_{\ell=0}^aU_\ell.
\]
Then
\begin{equation}\label{amp:layer-series}
 \Psi_a(t)=\sum_{\mathcal C}\Phi_{G[U(\mathcal C)]}(t),
 \qquad
 \Psi_a(0)=Z_G(a+1),\qquad
 \Psi_a(t)\leqslant Z_G(a+1)(1-t)^{-t_G}.
\end{equation}
Moreover $A_a(t)$ is a polynomial of degree at most $n-1$, with
constant term $Z_G(a+1)$.

<!-- proof-11: body_md -->
Use the nested family with multiplicities associated to a height
function by Proposition~\ref{amp:height}. Its members containing $v$
form a chain, and their multiplicities sum to $d(v)=a$. Listing these
members in decreasing order, each repeated with its multiplicity,
gives exactly $S_1,\dots,S_a$.

A connected set $T$ avoiding $v$ is compatible with this chain
exactly when it lies in one of the sets $U_\ell$. Indeed, take the
largest $\ell$ for which $T\subseteq S_\ell$. If $\ell<a$,
compatibility forces $T$ to be disjoint from and nonadjacent to
$S_{\ell+1}$, since $T$ cannot contain the root-containing set
$S_{\ell+1}$. If $\ell=a$, it lies in $S_a\setminus\{v\}$.
The converse follows immediately from these same conditions.
The $U_\ell$ are pairwise disjoint and nonadjacent: for
$\ell<\ell'$, the set $U_{\ell'}$ lies in $S_{\ell+1}$, which has no
vertex or neighbour in $U_\ell$. Thus all the remaining weighted
nested families are precisely those of $G[U(\mathcal C)]$.
Their weights contribute $w_G(d)-a$, proving the series identity.
Reversing $S_1,\ldots,S_a$ and adjoining the bottom $\{v\}$ and
top $V(G)$ gives a multichain with $a+1$ steps. Thus there are
$Z_G(a+1)$ chains, and each summand has constant term $1$.

Every connected set of $G[U(\mathcal C)]$ belongs to $\mathcal{B}(G-v)$.
Forgetting the nestedness restriction bounds its weighted nested
families by all multisets from this set of $t_G$ possibilities.
Their generating series is $(1-t)^{-t_G}$, proving the bound.
Finally, for $U=U(\mathcal C)$, Proposition~\ref{amp:height} gives
\[
 (1-t)^{n-1}\Phi_{G[U]}(t)
       =(1-t)^{n-1-|U|}\prod_C h_C(t).
\]
Since $U\subseteq V(G)\setminus\{v\}$, this is a polynomial, of degree
at most
\[
n-1-|U|+\sum_C(|C|-1)\leqslant n-1.
\]

<!-- thm-1: body_md -->
\label{amp:exact-wedge}
For every integer $q\geq1$,
\begin{equation}\label{amp:wedge-formula}
 h_{W_q(G)}(t)
   =(1-t)^{1+q(n-1)}\sum_{a\geq0}t^a\Psi_a(t)^q
   =(1-t)\sum_{a\geq0}t^aA_a(t)^q.
\end{equation}
These are identities of formal power series, whose left side is a
polynomial.
<!-- end thm-1 -->
<!-- proof-12: body_md -->
Fix a height function on the wedge with common root height $a$, and
let $d_i$ be its restriction to copy $i$. At each of the first $a$
levels, the $q$ components containing the root merge into one.
All other components remain separate. Hence
\[
 w_{W_q(G)}(d)=\sum_{i=1}^q w_G(d_i)-(q-1)a.
\]
The $d_i$ are independent once their root heights are fixed, so their
total generating series is $t^a\Psi_a(t)^q$. Sum over $a$ and apply
the height-function formula to the wedge, which has $1+q(n-1)$
vertices. The definition of $A_a$ gives the second equality.
<!-- end proof-12 -->
<!-- thm-2: body_md -->
\label{amp:error}
For every $j,q\geq1$,
\begin{equation}\label{amp:error-bound}
 \left|\frac{h_j(W_q(G))}{Z_G(j+1)^q}-1\right|\leqslant E_j(q).
\end{equation}
For fixed $G$ and $j$, this gives
\[
 h_j(W_q(G))=Z_G(j+1)^q+O\bigl(q^jZ_G(j)^q\bigr)
 \quad(q\longrightarrow\infty).
\]
<!-- end thm-2 -->
<!-- proof-13: body_md -->
Put $D=1+q(n-1)$. Extracting the coefficient in
\eqref{amp:wedge-formula} yields
\[
 h_j(W_q(G))=\sum_{a=0}^j
                  [t^{j-a}](1-t)^D\Psi_a(t)^q.
\]
The term $a=j$ is $Z_G(j+1)^q$. For $m=j-a\geq1$, the absolute
coefficients of $(1-t)^D$ are those of $(1+t)^D$, and
$(1+t)^D\leq(1-t)^{-D}$ coefficientwise. Since $\Psi_a$ has
nonnegative coefficients, Lemma~\ref{amp:layers} gives
\[
 \begin{aligned}
 \left|[t^m](1-t)^D\Psi_a(t)^q\right|
 &\leqslant Z_G(a+1)^q[t^m](1-t)^{-D-qt_G}\\
 &=Z_G(a+1)^q\binom{q(n-1+t_G)+m}{m}.
 \end{aligned}
\]
Summing and dividing by $Z_G(j+1)^q$ proves the bound. For fixed $j$,
each binomial factor is a polynomial in $q$ of degree at most $j$,
and $Z_G(a+1)\leqslant Z_G(j)$ for $a<j$, proving the asymptotic statement.

<!-- thm-3: body_md -->
\label{amp:amplification}
Suppose $(G,v)$ is a rooted zeta witness at $k$, and put
\[
 \rho=\frac{Z_G(k)^2}{Z_G(k-1)Z_G(k+1)}<1.
\]
If an integer $q\geq1$ satisfies $E_{k-2}(q)<1$, $E_k(q)<1$, and
\begin{equation}\label{amp:certificate}
 \rho^q\frac{(1+E_{k-1}(q))^2}
                 {(1-E_{k-2}(q))(1-E_k(q))}<1,
\end{equation}
then
\[
 h_{k-1}(W_q(G))^2<h_{k-2}(W_q(G))h_k(W_q(G)).
\]
All these conditions hold for every sufficiently large $q$.
As a consequence, the $h$-polynomials of these wedges are not
log-concave, and hence not real-rooted.
<!-- end thm-3 -->
<!-- proof-14: body_md -->
The coefficient estimate places $h_j(W_q(G))$ between
$Z_G(j+1)^q(1-E_j(q))$ and $Z_G(j+1)^q(1+E_j(q))$. The assumed positive
lower bounds at $k-2$ and $k$, together with nonnegativity at $k-1$,
therefore give
\[
 \frac{h_{k-1}(W_q(G))^2}
 {h_{k-2}(W_q(G))h_k(W_q(G))}
 \leq\rho^q\frac{(1+E_{k-1}(q))^2}
                 {(1-E_{k-2}(q))(1-E_k(q))}<1.
\]
Strict increase of $Z_G$ implies $Z_G(a+1)/Z_G(j+1)<1$ whenever $a<j$.
Each summand of $E_j(q)$ is thus a fixed polynomial in $q$ times a
decaying exponential, so $E_j(q)\to0$. Also $\rho^q\to0$, proving
eventual validity of the certificate. Newton's inequalities give the
failure of real-rootedness.

<!-- definition-1: body_md -->
Let $\textrm{\upshape M}$ be a matroid of positive rank, and $k \geqslant 0$. 
The $k$th truncation $\mathrm{T}^k(\textrm{\upshape M})$ of $\textrm{\upshape M}$ is the matroid with rank $r-k$ with lattice of flats 
\[
\mathcal{L}(\mathrm{T}^{k}(\textrm{\upshape M})) = \{F \in \mathcal{L}(\textrm{\upshape M}) \colon \operatorname{rk}_{\textrm{\upshape M}}(F) < k\} \cup \{E\}. 
\]
If $k=1$, the truncation $\mathrm{T}^{k}(\textrm{\upshape M})$ is called the \emph{principal truncation}, or simply \emph{truncation}. 
A matroid $\textrm{\upshape M}$ of rank $r$ is a \emph{detruncation} of a
matroid $\textrm{\upshape T}$ of rank $r-1$, if
the principal truncation $\mathrm{Tr}(\textrm{\upshape M})$ is $\textrm{\upshape T}$.

<!-- lemma-8: body_md -->
\label{lem:trunc}
For a matroid $\textrm{\upshape M}$ of rank $r$ on $E$ and $1\leqslant k\leqslant r$, the following hold.
\begin{enumerate}[(a)]
\item The flats of $\mathrm{Tr}^k(\textrm{\upshape M})$ are the flats of $\textrm{\upshape M}$ of rank at most $k-1$, with their rank in $\textrm{\upshape M}$, and $E$. In
particular $\alpha_{\mathcal{L}(\mathrm{Tr}^k(\textrm{\upshape M}))}(S)=\alpha_{\mathcal{L}(\textrm{\upshape M})}(S)$ for every $S\subseteq[k-1]$ (\cite[Proposition~7.3.10]{oxley}).
\item If $\textrm{\upshape M}$ is loopless and $k<r$, then $\mathrm{Tr}^k(\textrm{\upshape M})$ is connected (\cite[Section~7.3, Exercise~8]{oxley}). 
\item If $\textrm{\upshape M}$ is simple and $k\geqslant2$, then $\mathrm{Tr}^k(\textrm{\upshape M})$ is simple (\cite[Proposition~7.3.10]{oxley}). 
\end{enumerate}

<!-- definition-2: body_md -->
Let $\textrm{\upshape M}$ be a matroid of rank $r\geqslant2$ on $E$, and let
$\mathcal{H}$ be a family of proper spanning subsets of $E$.
A matroid $\textrm{\upshape M}'$ of rank $r$ on $E$ is obtained from $\textrm{\upshape M}$ via
\emph{hyperplane coalescence along $\mathcal{H}$} if:
\begin{enumerate}[(a)]
\item $\textrm{\upshape M}$ and $\textrm{\upshape M}'$ have the same flats of rank at most
$r-2$, with the same ranks;
\item the hyperplanes of $\textrm{\upshape M}'$ are precisely the elements
in $\mathcal{H}$ together with the hyperplanes of $\textrm{\upshape M}$ not contained in
any element in $\mathcal{H}$.
\end{enumerate}
In other words, the collection of hyperplanes of $\textrm{\upshape M}'$ is 
\[
\mathcal{L}^{r-1}(\textrm{\upshape M}')
=
\mathcal{H}\;\cup\;
\bigl\{F\in\mathcal{L}^{r-1}(\textrm{\upshape M}):
F\nsubseteq H\text{ for every }H\in\mathcal{H}\bigr\}.
\]
The members of $\mathcal{H}$ are called \emph{blocks}. We say that $\textrm{\upshape M}'$ is obtained by
\emph{coalescing the blocks $\mathcal{H}$ in $\textrm{\upshape M}$}.
Thus, to \emph{coalesce} a block $H$ is to replace the
hyperplanes contained in $H$ by the single hyperplane $H$.
For each block $H$, the hyperplanes in $\textrm{\upshape M}$ contained in
$H$ are replaced by the single hyperplane $H$ in $\textrm{\upshape M}'$.
<!-- end definition-2 -->

<!-- lemma-9: body_md -->
\label{lem:coalescence}
Let $\textrm{\upshape M}$ have rank $r\geqslant2$ on $E$, and let $\mathcal{H}$ be a
family of proper spanning subsets satisfying
\begin{enumerate}[(a)]
\item $\operatorname{rk}_{\textrm{\upshape M}}(H\cap H')\leqslant r-2$ for distinct
$H,H'\in\mathcal{H}$;
\item $\operatorname{cl}_{\textrm{\upshape M}}(I)\subseteq H$ whenever $I\subseteq H\in\mathcal{H}$
is independent and $|I|=r-1$.
\end{enumerate}
Then $\mathcal{H}$ uniquely determines a hyperplane coalescence $\textrm{\upshape M}'$ of $\textrm{\upshape M}$, which has bases 
\[
\mathcal{B}(\textrm{\upshape M}') = \{B \in \mathcal{B}(\textrm{\upshape M}) \colon B \notin H \in \mathcal{H}\}. 
\]
and rank function
\begin{equation}\label{eq:coalescence-rank}
\operatorname{rk}_{\textrm{\upshape M}'}(S)=
\begin{cases}
r-1,&\operatorname{rk}_{\textrm{\upshape M}}(S)=r\text{ and }S\subseteq H\text{ for some }H\in\mathcal{H},\\
\operatorname{rk}_{\textrm{\upshape M}}(S),&\text{otherwise.}
\end{cases}
\end{equation}
<!-- end lemma-9 -->
<!-- proof-16: body_md -->
This follows from Crapo's criterion \cite[Theorem~12]{pendavingh-vanderpol2017} applied to $\mathrm{Tr}(\textrm{\upshape M})$.

<!-- lemma-11: body_md -->
\label{lem:coalescence-chow}\label{prop:coalescence-chow}\label{prop:local-polynomials}
For $P=f$ or $P=\underline{\textrm{\upshape H}}$,
\begin{equation}\label{eq:coalescence-chow}
P(\textrm{\upshape M}';t)=P(\textrm{\upshape M};t)-\sum_{H\in\mathcal{H}}

<!-- end lemma-11 -->
<!-- proof-18: body_md -->
Summing \eqref{eq:coalescence-flags} by chain length gives
the assertion for $f$. For the Chow polynomial, the weights in \eqref{eq:chow-alpha}
satisfy, for $S\subseteq[r-2]$,
\[
\underline{\tau}(S\cup\{r\})
=t\bigl(\underline{\tau}(S)
+\underline{\tau}(S\cup\{r-1\})\bigr),
\qquad
\underline{\tau}(S\cup\{r-1,r\})=0.
\]
Thus \eqref{eq:chow-alpha} implies that, for any loopless
matroid $\textrm{\upshape M}$ of rank $r$,
\[
\underline{\textrm{\upshape H}}(\textrm{\upshape M};t)-(1+t)\underline{\textrm{\upshape H}}(\mathrm{Tr}(\textrm{\upshape M});t)
=
\sum_{S\subseteq[r-2]}
\bigl(\alpha_{\textrm{\upshape M}}(S\cup\{r-1\})
-\alpha_{\textrm{\upshape M}}(S)\bigr)
\underline{\tau}(S\cup\{r-1\}).
\]
Applying \eqref{eq:coalescence-flags} with these weights
proves the result.

The binomial transform \eqref{eq:zeta-binomial} is a linear map which sends $t^j$ to

\]
The assertion follows by applying this transform to

<!-- proposition-6: body_md -->
\label{prop:absorption-coalescence}
Let $\textrm{\upshape G}$ have rank $r\geqslant2$, and let
$\mathcal{F}_1,\dots,\mathcal{F}_m$ be pairwise disjoint families of hyperplanes
such that
\[
\operatorname{rk}_{\textrm{\upshape G}}(F\cap F')\leqslant r-3
\qquad(F,F'\in\mathcal{F}_i,\ F\ne F').
\]
For a set of new points $X=\{p_1,\dots,p_m\}$, put
\[
\textrm{\upshape M}=\mathrm{Tr}^r(\textrm{\upshape G}\oplus\textrm{\upshape U}_{X,X}),
\qquad
\mathcal{H}=\{F\cup\{p_i\}:F\in\mathcal{F}_i,\ 1\leqslant i\leqslant m\}.
\]
Then the blocks in $\mathcal{H}$ can be coalesced in $\textrm{\upshape M}$.
The resulting matroid $\textrm{\upshape M}'$ restricts to $\textrm{\upshape G}$ and,
for $A\subseteq E(\textrm{\upshape G})$ and $J\subseteq X$, has rank
\begin{equation}\label{eq:absorption-rank}
\operatorname{rk}_{\textrm{\upshape M}'}(A\cup J)=
\begin{cases}
r-1,
&J=\{p_i\}\text{ and }
  \operatorname{cl}_{\textrm{\upshape G}}(A)\in\mathcal{F}_i,\\
\min\{\operatorname{rk}_{\textrm{\upshape G}}(A)+|J|,r\},
&\text{otherwise}.
\end{cases}
\end{equation}
Before coalescence, each block has restriction
\[
\textrm{\upshape M}|(F\cup\{p_i\})
=(\textrm{\upshape G}|F)\oplus\textrm{\upshape U}_{1,1}.
\]
<!-- end proposition-6 -->
<!-- proof-20: body_md -->
Put $\textrm{\upshape Q}
=\textrm{\upshape G}\oplus\textrm{\upshape U}_{X,X}$.
Each block $H=F\cup\{p_i\}$ is a proper rank-$r$ flat of
$\textrm{\upshape Q}$, hence a proper spanning subset of $\textrm{\upshape M}$.
For every independent $(r-1)$-set $I\subseteq H$,
\[
\operatorname{cl}_{\textrm{\upshape M}}(I)=\operatorname{cl}_{\textrm{\upshape Q}}(I)\subseteq H.
\]
Distinct blocks sharing $p_i$ intersect in rank at most
$(r-3)+1=r-2$. Blocks with different auxiliary points intersect
in rank at most $r-2$, since their geometric hyperplanes are
distinct. Thus both conditions of \cref{lem:coalescence} hold.

A set $A\cup J$ is spanning in $\textrm{\upshape M}$ and contained in a block
$F\cup\{p_i\}$ exactly when $J=\{p_i\}$ and
$\operatorname{cl}_{\textrm{\upshape G}}(A)=F$.
Hence \eqref{eq:coalescence-rank} gives
\eqref{eq:absorption-rank}; taking $J=\varnothing$ shows that
$\textrm{\upshape M}'$ restricts to $\textrm{\upshape G}$.
The old block restriction follows because truncation to rank $r$
does not change the restriction to a rank-$r$ subset.

<!-- lemma-12: body_md -->
\label{lem:affine-intersections}
Let $V$ be a vector space over a field, and let
$P=a+D$ and $P'=b+D'$ be distinct affine subspaces of $V$.
\begin{enumerate}[(a)]
\item If $D=D'$, then $P\cap P'=\varnothing$.
\item If $D\cap D'=\{0\}$, then $P\cap P'$ contains
at most one point.
\item If $V=D\oplus D'$, then $P\cap P'$ contains
exactly one point.
\end{enumerate}
In particular, if $V=\mathbb{F}_q^{2k+1}$ and
$\mathbb P(D),\mathbb P(D')$ belong to a
$k$-spread of $\textrm{\upshape PG}(2k,q)$, then the distinct
affine $k$-dimensional subspaces $P$ and $P'$ are
either parallel and disjoint, or have complementary
directions and intersect in exactly one point.
<!-- end lemma-12 -->
<!-- proof-21: body_md -->
Since $a, b$ are distinct vectors, (a) follows immediately. 
For (b), if $x,y \in P\cap P'$, then
$x-y\in D\cap D'= \{0\}$, and $x = y$.
For (c), write $b-a=d+d'$ with $d\in D$ and $d'\in D'$.
Then $a+d=b-d'$ belongs to $P \cap P'$, and (b) implies its uniqueness.

<!-- lemma-13: body_md -->
\label{lem:elliptic-sections}
The Klein correspondence
\[
\kappa:\mathcal{L}_2(\textrm{\upshape PG}(3,q))\longrightarrow\mathcal{K}
\]
induces a bijection between elliptic sections of $\mathcal{K}$ and
regular spreads of $\textrm{\upshape PG}(3,q)$, given by
\[
\mathcal{E}\longmapsto
\mathcal{S}(\mathcal{E}):=\kappa^{-1}(\mathcal{E})
\subseteq\mathcal{L}_2(\textrm{\upshape PG}(3,q)).
\]
<!-- end lemma-13 -->
<!-- proof-22: body_md -->
Let $\mathcal{E}=\mathcal{K}\cap\mathbb P(U)$ be an elliptic section.
If two distinct lines in $\mathcal{S}(\mathcal{E})$ intersected, the
projective line joining their Klein images would lie entirely
on $\mathcal{K}$. It would also lie in $\mathbb P(U)$, hence in
$\mathcal{E}$, contradicting ellipticity. Thus the projective lines of
$\mathcal{S}(\mathcal{E})$ are pairwise skew. An elliptic quadric has $q^2+1$ points, so $\mathcal{S}(\mathcal{E})$
consists of $q^2+1$ pairwise skew lines. Since each projective line contains $q+1$ points, and
\[
(q^2+1)(q+1)=q^3+q^2+q+1=|\textrm{\upshape PG}(3,q)|,
\]
these lines partition the point set of $\textrm{\upshape PG}(3,q)$ and form a spread.
The fact that this spread is regular, and that every regular spread
arises from an elliptic section, is the classical
regular-spread correspondence
\cite[Section~15.4 and Chapter~17]{hirschfeld1985}.
Furthermore, since $\kappa$ is a bijection, the regular spread from an elliptic section is unique.

<!-- construction-1: body_md -->
\label{cons:klein}
Let $q$ be an odd prime power. Choose an external line
\[
\ell_0 = \mathbb P(A_0) \subseteq \mathbb{P}(W) \qquad \text{ with }\dim A_0 = 2, 
\]
and define the orthogonal complement and its associated elliptic section of $\mathcal{K}$, 
\[
U=A_0^{\perp_B},
\qquad
\mathcal{E}_0=\mathcal{K}\cap\mathbb P(U).
\]
Then $\dim U = 4$, $\mathbb P(U)=\ell_0^\perp$,
and $\mathcal{E}_0$ is an elliptic quadric. By \cref{lem:elliptic-involution}, there exists a fixed-point-free projective involution $\tau$ of $\mathbb P(U)$ satisfying
\[
\tau^2=\operatorname{id},
\qquad
\tau(x)\ne x\quad\text{for every }x\in\mathbb P(U),
\qquad
\tau(\mathcal{E}_0)=\mathcal{E}_0.
\]
Let $\{\ell_1, \ldots, \ell_m\}$ denote the distinct lines spanned by $\{x, \tau(x)\}$ for $x \in \mathbb{P}(U)$ which are disjoint from $\mathcal{E}_0$. These lines are external to $\mathcal{K}$,
since they lie in $\mathbb P(U)$ and are disjoint from 
$\mathcal{E}_0=\mathcal{K}\cap\mathbb P(U)$.
We therefore define for each $1 \leqslant i \leqslant m$, 
\[
\mathcal{E}_i \coloneqq \mathcal{K}\cap\ell_i^\perp,
\qquad
\mathcal{S}_i \coloneqq \kappa^{-1}(\mathcal{E}_i).
\]

<!-- lemma-14: body_md -->
\label{lem:klein-design}
\Cref{cons:klein} gives $\frac{q^2+1}{2}$ pairwise disjoint elliptic
sections of $\mathcal{K}$.
<!-- end lemma-14 -->
<!-- proof-23: body_md -->
Each line $\textrm{span}(x,\tau(x))$ is preserved by
$\tau$. Two distinct such lines cannot meet, since their unique
intersection point would be fixed by $\tau$.
Every point lies on its joining line, so these lines form a
spread of $\mathbb P(U)$, consisting of $q^2+1$ lines. Each of these lines meets $\mathcal{E}_0$ in either zero or two points. Indeed, a line containing $x\in\mathcal{E}_0$ also contains the distinct
point $\tau(x)\in\mathcal{E}_0$; it cannot contain more than two points
of the quadric without lying entirely on it.
Since $\mathcal{E}_0$ has $q^2+1$ points, exactly half the spread lines
meet it, and the other half avoid it. Therefore $m=\frac{q^2+1}{2}$.

For $i\ne j$, the lines $\ell_i,\ell_j$ are skew and hence span
$\mathbb P(U)$. Therefore,
\[
\ell_i^\perp\cap\ell_j^\perp
=
\langle\ell_i,\ell_j\rangle^\perp
=
\mathbb P(U)^\perp
=
\ell_0.
\]
Their polar spaces meet precisely along the external line $\ell_0$. Hence
\[
\mathcal{E}_i\cap\mathcal{E}_j
=
\mathcal{K}\cap\ell_0
=
\varnothing.
\]
The corresponding regular spreads $\mathcal{S}_i,\mathcal{S}_j$ therefore
share no common elements.

<!-- lemma-15: body_md -->
\label{lem:elliptic-involution}
For an odd prime power $q$, the projective $3$-space $\mathbb P(U)$ in
\cref{cons:klein} admits a fixed-point-free projective involution
preserving $\mathcal{E}_0$.
<!-- end lemma-15 -->
<!-- proof-24: body_md -->
In diagonal coordinates, we write $Q=Q_1\oplus Q_2$, where
\[
Q_i(s,t)=a_i s^2+b_i t^2,\qquad a_i,b_i\ne0.
\]
Fix a nonsquare $d\in\mathbb{F}_q$. For each $i$, the sets of values
$u^2$ and $d-(a_i/b_i)v^2$ each have $(q+1)/2$ elements,
so they intersect. Thus we can choose $u_i,v_i$ satisfying
\[
u_i^2+\frac{a_i}{b_i}v_i^2=d.
\]
The matrices
\[
J_i=
\begin{pmatrix}
u_i&v_i\\
(a_i/b_i)v_i&-u_i
\end{pmatrix}
\]
satisfy $J_i^2=dI$ and $Q_i(J_i x)=dQ_i(x)$.
Therefore, $J=J_1\oplus J_2$ satisfies
\[
J^2=dI,\qquad Q(Jx)=dQ(x).
\]
Its projectivization $\tau([x])=[Jx]$ is an
involution preserving $\mathcal{E}_0$. A fixed point would give
$Jx=\lambda x$ for some $x \ne 0$ and $\lambda\in\mathbb{F}_q$ with $\lambda^2=d$, a contradiction.

\item rank $2$: the affine lines and the two-element

\item rank $3$: the affine planes outside

The direct sum is simple of rank $9$ and hence its truncation is connected and simple, and its rank-$7$ truncation is simple by \cref{lem:trunc}.

We count flags in
$\mathcal{L}(\textrm{\upshape X})^2\times\mathcal{L}(\textrm{\upshape U}_{1,1})$
by their coordinate ranks, retaining total rank sets in $[6]$.
For each choice of weakly increasing coordinate ranks, the
chains in the factors are chosen independently. 
Substitution in \cref{eq:gamma-stable,eq:rank7-gamma} gives the
Chow coefficients $\underline{\textrm{\upshape H}}_i(q)$ and the cubic $\gamma$-polynomial
$\Gamma_q$. Let
\[
R(q)=\mathrm{disc}(\Gamma_q),\qquad
\Delta(q)=\underline{\textrm{\upshape H}}_2(q)^2-\underline{\textrm{\upshape H}}_1(q)\underline{\textrm{\upshape H}}_3(q).
\]
The Taylor coefficients of $\gamma_3(q)$ at $q_{\min}=3$ for
the affine family, and at $q_{\min}=2$ for the Gabidulin family,
are nonnegative, with positive constant term. Hence $\gamma_3(q)>0$
for $q\geqslant q_{\min}$, and \cref{lem:rank7-criteria} implies that 
\[
\begin{aligned}
\underline{\textrm{\upshape H}}(\textrm{\upshape Y};t)\text{ is not real-rooted}
&\quad\Longleftrightarrow\quad R(q)<0,\\
\underline{\textrm{\upshape H}}(\textrm{\upshape Y};t)\text{ is not log-concave}
&\quad\Longleftrightarrow\quad \Delta(q)<0.
\end{aligned}
\]

We first determine the signs of $R$ and $\Delta$
for real arguments $x\geqslant q_{\min}$. For either $D=R$ or $D=\Delta$, the polynomial $D(q_{\min}+x)$ has one sign variation,
a positive constant term, and a negative leading coefficient.
Descartes' rule therefore implies that $D(q)$ has exactly one zero
$\xi_D>q_{\min}$, with
\[
D(\xi_D)=0,\qquad
D(x)>0\quad(q_{\min}\leqslant x<\xi_D),\qquad
D(x)<0\quad(x>\xi_D).
\]
Thus $\xi_R$ and $\xi_\Delta$ indicate where the discriminant and
the log-concavity defect, respectively, change sign as $q$ varies.
We locate these zeros between consecutive
allowed prime powers by computer computations.

\begin{enumerate}[(a)]
\item For $\textrm{\upshape X}=\textrm{\upshape M}_q$, use \cref{cor:flag-Mq}
with $m=(q^2+1)/2$. The consecutive odd prime powers $43,47$
and $89,97$ satisfy
\[
\begin{aligned}
R(43)&>0>R(47),&43&<\xi_R<47,\\
\Delta(89)&>0>\Delta(97),&89&<\xi_\Delta<97.
\end{aligned}
\]
Hence, for odd prime powers $q$, $R(q)<0$ if and only if
$q\geqslant47$, and $\Delta(q)<0$ if and only if $q\geqslant97$.

\item For $\textrm{\upshape X}=\textrm{\upshape G}_{4,q}$, use the counts following
\cref{prop:structure} with $n=6$, $m=q^3$, and $b=q^9$.
The consecutive prime powers $32=2^5,37$ and $79,81=3^4$ satisfy
\[
\begin{aligned}
R(32)&>0>R(37),&32&<\xi_R<37,\\
\Delta(79)&>0>\Delta(81),&79&<\xi_\Delta<81.
\end{aligned}
\]
Hence, for prime powers $q$, $R(q)<0$ if and only if
$q\geqslant37$, and $\Delta(q)<0$ if and only if $q\geqslant81$.
\end{enumerate}
By the two criteria above, these are the desired
real-rootedness and log-concavity thresholds.

\label{bridge:definition}
Fix a word $w=w_0\cdots w_{L-1}$ in the letters
$\mathrm I,\mathrm O,\mathrm N$, a bridge pattern
$\beta=(\beta_0,\ldots,\beta_{L-1})\in\{-1,0,1,2,3\}^L$, and an integer
$R\geq1$. The graph contains $LR$ layers, indexed by $0\leqslant i<LR$, with
$b_i=\beta_{i\bmod L}$.
\begin{enumerate}
\item Each layer contains vertices $a_i,c_i,c_i'$ and the edge $c_ic_i'$.
If $b_i=-1$, there are no further edges within this layer.
If $b_i=0$, add $a_ic_i$ and $a_ic_i'$.
If $b_i\geq1$, add $b_i$ new vertices to the layer, denoted $s_{i,1},\ldots,s_{i,b_i}$, together with the path
$a_i s_{i,1}\cdots s_{i,b_i}$, and join $s_{i,b_i}$ to both $c_i$ and
$c_i'$. We call this a bridge with $b_i$ additional vertices.
\item Consecutive layers $i-1,i$ are joined by $a_{i-1}a_i$ and by
ladder edges prescribed by $w_{(i-1)\bmod L}$:
\[
\begin{array}{c|l}
\mathrm I & c_{i-1}c_i,\ c_{i-1}'c_i\\
\mathrm O & c_{i-1}'c_i,\ c_{i-1}'c_i'\\
\mathrm N & \text{all four edges between }
              \{c_{i-1},c_{i-1}'\}\text{ and }\{c_i,c_i'\}.
\end{array}
\]
See Figure \ref{fig:ladder-interfaces}.
\item A root $p$ is joined to $a_0,c_0,c_0'$. Optionally, choose $\ell\geq0$, join a new vertex $r$ to
$a_{LR-1},c_{LR-1},c_{LR-1}'$, and attach a path
$r u_1\cdots u_\ell$. The resulting rooted graph is denoted
$X(w,\beta;R,\ell)$. Without this optional attachment, $r$ is absent.
The resulting rooted graph is denoted $X(w,\beta;R, \ell)$.
\end{enumerate}

\label{bridge:matrix}
Fix a design $(w,\beta)$ and a choice of terminal attachment, including
its arm length when present. For each $j\geq1$ there are a finite

\end{equation}

For $j=1$, there is only the chain $\{p\}\subseteq V(X_R)$,
so $Z_{X_R}(1)=1$; take $T_1=(1)$ and $u_1=w_1=(1)$.
For $j\geq2$, group $L$ successive layer transfers into one period.
At every period boundary, the frontier has the same three positions
and the next period has the same edge pattern. Let $A_j(\sigma,\tau)$
count the allowed labelings of one further period taking state $\sigma$
to state $\tau$. Let $v_j(\sigma)$ count initializations through the
first period, including the root edges, and let $\omega_j(\sigma)$ count
accepted terminal extensions, including $r$ and its arm when present.
These are nonnegative integers, and summing over successive states gives
$Z_{X_R}(j)=v_j^{\mathsf T}A_j^{R-1}\omega_j$.

To obtain the exponent $R$ in~\eqref{bridge:count}, add one start state
$*$, put $T_j(*,\sigma)=v_j(\sigma)$ and
$T_j(\sigma,\tau)=A_j(\sigma,\tau)$, and allow no transition into $*$.
Take $u_j$ supported at $*$ with value $1$, and set $w_j(*)=0$ and
$w_j(\sigma)=\omega_j(\sigma)$. Then
$u_j^{\mathsf T}T_j^Rw_j=v_j^{\mathsf T}A_j^{R-1}\omega_j$, as required.

The last inequality of \eqref{cert:inputs} already proves the zeta
failure. It remains to apply the error bound of Theorem~\ref{amp:error}.
Set $n=1540$ and $c=n-1+t_G$. By \eqref{cert:inputs}, $c<2^{882}$.
For $j\in\{35,36,37\}$ and $m=j-a\ge1$, successive multiplication
of the zeta-ratio bounds gives
\[
 \frac{Z_G(j+1-m)}{Z_G(j+1)}<10^{-12m}.
\]
Since $23\cdot2^{882}+37<10^{268}$, the error at $q=23$ satisfies
\begin{align*}
 E_j(23)
 &=\sum_{m=1}^{j}
 \left(\frac{Z_G(j+1-m)}{Z_G(j+1)}\right)^{23}
 \binom{23c+m}{m}\\
 &<\sum_{m=1}^{j}10^{-276m}(23c+m)^m
 <\sum_{m\ge1}10^{-8m}<2\cdot10^{-8}.
\end{align*}
Writing $\varepsilon=2\cdot10^{-8}$, exact rational arithmetic gives
\[
 \rho^{23}\frac{(1+E_{36}(23))^2}
 {(1-E_{35}(23))(1-E_{37}(23))}
 <\left(\frac{9998}{10000}\right)^{23}
 \frac{(1+\varepsilon)^2}{(1-\varepsilon)^2}<1.
\]
This proves the $h$-inequality for $q=23$.

To cover all larger integers, consider an individual error summand
$r^q\binom{qc+m}{m}$, where
$r=Z_G(j+1-m)/Z_G(j+1)<10^{-12m}$. Its ratio at $q+1$ and $q$ is
\[
 r\prod_{s=1}^{m}\frac{(q+1)c+s}{qc+s}
 \le r\left(1+\frac1q\right)^m
 <10^{-12m}\left(\frac{24}{23}\right)^m<1
 \qquad(q\ge23).
\]
Thus every $E_j(q)$ decreases, as does $\rho^q$. The amplification
criterion remains valid for every $q\ge23$. Newton's inequalities
exclude real-rootedness, and the vertex count follows from the
definition of a wedge.
