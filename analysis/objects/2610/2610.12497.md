<!-- definition-1: body_md -->
Let $M$ be a real-analytic hypersurface invariant by $\mathcal F$ and satisfying
$\operatorname{Sing}(M)\subset\operatorname{Sing}(\mathcal F)$. Let
\(
p\in M\cap\operatorname{Sing}(\mathcal F)
\)
be a dicritical singularity, and let
\(
F_p=\frac{f}{g}:U_p\dashrightarrow\mathbb P^1
\)
be a minimal local meromorphic first integral of the germ $(\mathcal F,p)$.
We say that $p$ is a \emph{pencil singularity for $M$} if:
\begin{enumerate}[label=\textup{(\roman*)}]
\item
\[
p\in\overline L
\]
for every Levi leaf $L\subset M_{\mathrm{reg}}$;
\item for every generic global Levi leaf $L\subset M_{\mathrm{reg}}$, the
restriction of $F_p$ to $L\cap U_p$ is constant (where the
restriction is understood off the indeterminacy set of $F_p$).
\end{enumerate}
Equivalently, for every generic global Levi leaf $L$ there exists a
unique $\lambda(L)\in\mathbb P^1$ such that every local component of
$L\cap U_p$ sufficiently near $p$ is contained in
\[
V_{\lambda(L)}=\{f-\lambda(L)g=0\}.
\]
This property is independent of the chosen minimal first integral,
since two minimal generators differ by a M\"obius transformation.

<!-- theorem-1: body_md -->
\label{thm:plane-characterization}
Let $\mathcal F$ be a holomorphic foliation on $\mathbb P^2$. The following
conditions are equivalent.

\begin{enumerate}[label=\textup{(\roman*)}]
\item $\mathcal F$ admits a primitive nonconstant rational first integral
\[
R:\mathbb P^2\dashrightarrow\mathbb P^1.
\]

\item There exists an irreducible real-analytic variety of pure real codimension one 
\(
M\subset\mathbb P^2
\)
invariant by $\mathcal F$, with
\(
\operatorname{Sing}(M)\subset\operatorname{Sing}(\mathcal F),
\)
and all the dicritical singularities
\(
p_1,\ldots,p_r\in M\cap\operatorname{Sing}(\mathcal F)
\)
of $\mathcal F$ on $M$, together with local meromorphic first integrals
\[
H_j:(U_j,p_j)\dashrightarrow\mathbb P^1,
\qquad j=1,\ldots,r,
\]
such that every Levi leaf $L\subset M_{\mathrm{reg}}$ accumulates at least
one of the points $p_j$, and:
\begin{enumerate}[label=\textup{(\alph*)}]
\item for every generic global Levi leaf $L\subset M_{\mathrm{reg}}$ approaching
$p_j$, the restriction $H_j|_{L\cap U_j}$ is constant; denote its value
by $\lambda_j(L)$;
\item there exist transformations
\[
A_{ji}\in\operatorname{PGL}(2,\mathbb C)
\]
satisfying
\[
A_{ii}=\operatorname{id},
\qquad
A_{kj}\circ A_{ji}=A_{ki}
\]
whenever the corresponding continuations are defined;
\item whenever a generic global Levi leaf $L$ accumulates at both
$p_i$ and $p_j$, one has
\[
\lambda_j(L)=A_{ji}\bigl(\lambda_i(L)\bigr).
\]
\end{enumerate}
\end{enumerate}

<!-- corollary-1: body_md -->
\label{cor:single-dicritical-plane}
Let $\mathcal F$ be a codimension-one holomorphic foliation on $\mathbb P^2$
preserving an irreducible real-analytic variety of pure real codimension one  $M$. Assume
\[
\operatorname{Sing}(M)\subset\operatorname{Sing}(\mathcal F).
\]
Suppose that $p\in M\cap\operatorname{Sing}(\mathcal F)$ is the only dicritical
singularity of $\mathcal F$ on $M$ and that $p$ is a pencil singularity for $M$.
Then $\mathcal F$ admits a nonconstant rational first integral on $\mathbb P^2$.

<!-- theorem-2: body_md -->
\label{thm:projective-main}
Let $\mathcal F$ be a codimension-one holomorphic foliation on $\mathbb P^n$,
$n\ge3$, and let $M\subset\mathbb P^n$ be an irreducible real-analytic variety of pure real codimension one invariant by $\mathcal F$. Assume:

\begin{enumerate}[label=\textup{(\roman*)}]
\item
\(
\operatorname{Sing}(M)\subset\operatorname{Sing}(\mathcal F);
\)

\item all dicritical singularities of $\mathcal F$ contained in $M$ belong to
an irreducible codimension-two analytic subset
\(
D\subset M\cap\operatorname{Sing}(\mathcal F),
\)
and $D$ is generically of local product type;

\item there exists a pencil singularity for $M$ in the local-product
locus of $D$;
\end{enumerate}

Then $\mathcal F$ admits a nonconstant rational first integral
\[
R:\mathbb P^n\dashrightarrow\mathbb P^1.
\]

<!-- corollary-2: body_md -->
\label{cor:local-separation}
Let $\mathcal F$ be a codimension-one holomorphic foliation on $\mathbb P^n$,
$n\ge3$, and let $M\subset\mathbb P^n$ be an irreducible real-analytic variety of pure real codimension one invariant by $\mathcal F$. Assume that
\[
\operatorname{Sing}(M)\subset\operatorname{Sing}(\mathcal F),
\]
that all dicritical singularities of $\mathcal F$ contained in $M$ belong to
an irreducible codimension-two analytic subset
\[
D\subset M\cap\operatorname{Sing}(\mathcal F)
\]
which is generically of local product type. Assume that each Levi leaf
$L\subset M_{\mathrm{reg}}$ accumulates at some point in the local-product
locus of $D$. Fix a point $p$ in this locus and let
\[
F_p:U_p\dashrightarrow\mathbb P^1
\]
be a minimal local meromorphic first integral of the germ $(\mathcal F,p)$.
Suppose that, after shrinking $U_p$, there exist a real-analytic
one-dimensional manifold $S$ and a real-analytic map
\[
h:M_{\mathrm{reg}}\cap U_p\longrightarrow S
\]
which separates the nearby Levi leaves at $p$ and has the following
global-leaf compatibility property: for every generic global Levi leaf
$L\subset M_{\mathrm{reg}}$, the restriction $h|_{L\cap U_p}$ is constant.
Then $p$ is a pencil singularity for $M$, and consequently $\mathcal F$ admits a
nonconstant rational first integral on $\mathbb P^n$.

<!-- theorem-3: body_md -->
\label{thm:leviflat-intersection-alternative}
Let $\mathcal F$ be a holomorphic foliation on $\mathbb P^2$ and let
$\{M_\alpha\}_{\alpha\in A}$ be infinitely many pairwise distinct
irreducible invariant real-analytic Levi-flat varieties. Assume that
there is one member $M_0$ for which
\(
(M_0\cap M_\alpha)\setminus\operatorname{Sing}(\mathcal F)\ne\varnothing
\)
for infinitely many $\alpha$. Assume moreover that
\(
\operatorname{Sing}(M_\alpha)\subset\operatorname{Sing}(\mathcal F)
\)
for every $\alpha\in A$. Then either $\mathcal F$ admits a nonconstant
rational first integral, or there is an irreducible invariant algebraic
curve $C$ contained in infinitely many $M_\alpha$. In the second
alternative the virtual holonomy group of every connected regular leaf
component of $C\setminus\operatorname{Sing}(\mathcal F)$ is abelian.
<!-- end theorem-3 -->
<!-- corollary-3: body_md -->
\label{corollary:abelian}
Under (ii) in Theorem~\ref{thm:leviflat-intersection-alternative}, assume, in addition, that the singularities of $\mathcal F$ on $C$ satisfy the following conditions:
\begin{enumerate}
\item every non-dicritical singularity of $\mathcal F$ on $C$ is reduced and nondegenerate, i.e., it is of the form
\[
x\,dy-\lambda y\,dx+(\cdots)=0,
\qquad \lambda\in\mathbb C^*\setminus\mathbb Q_{>0};
\]
\item the dicritical singularities are $p_1,\ldots,p_m\in C\cap\operatorname{Sing}(\mathcal F)$ and, after one blow-up at each $p_j$, every singularity of the transformed foliation lying on the strict transform of $C$ is reduced, non-dicritical, and is not a saddle-node.
\end{enumerate}
 Put
\[
d_j=\operatorname{mult}_{p_j}(C).
\]
If
\[
\sum_{j=1}^m d_j<\deg C,
\]
then $\mathcal F$ is defined on $\mathbb P^2$ by a nonzero closed rational $1$-form.
If, in addition, the reduction of singularities of $\mathcal F$ contains no
saddle-nodes, then $\mathcal F$ is defined by a closed logarithmic rational
$1$-form.

<!-- example-1: body_md -->
\label{ex:pencil}
Let
\[
R=\frac{F}{G}:\mathbb P^n\dashrightarrow\mathbb P^1,
\]
where $F$ and $G$ are homogeneous polynomials of the same degree, and
suppose that $R$ is a rational first integral of a codimension-one
foliation $\mathcal F$.  Let $\gamma\subset\mathbb P^1$ be a real-analytic curve and
set
\[
M=\overline{R^{-1}(\gamma)}.
\]
Away from the critical and indeterminacy loci of $R$, use local
coordinates
\[
(z_1,\ldots,z_{n-1},t),\qquad t=R.
\]
If $\rho(t,\overline t)=0$ is a real-analytic equation for $\gamma$,
then $M$ is locally defined by the same equation and
\[
T^{1,0}M=\ker dR.
\]
Its Levi leaves are the connected components of
\[
R^{-1}(\lambda),\qquad \lambda\in\gamma.
\]

For $\gamma=\mathbb RP^1$ one has
\[
M=\overline{R^{-1}(\mathbb RP^1)},
\]
and, away from the base locus $\{F=G=0\}$,
\[
F\overline G-\overline F\,G=0.
\]
The Levi leaves are
\[
F-\lambda G=0,\qquad \lambda\in\mathbb RP^1.
\]
When $D=\{F=G=0\}$ is irreducible of codimension two and generically
Kupka (hence generically of local product type), this is the basic model for the dicritical configuration in
Theorem~\ref{thm:projective-main}.

The same construction also anticipates the two alternatives for
infinite families of invariant Levi-flats on $\mathbb P^2$. Suppose now that
$n=2$. If two real-analytic curves $\gamma_1,\gamma_2\subset\mathbb P^1$
meet at a regular value $\lambda$, then
\[
\overline{R^{-1}(\gamma_1)}\cap

<!-- lemma-2: body_md -->
\label{lem:period-group-singular-transverse}
Let
\(
\Gamma=
\left\langle
2\pi i\lambda_1,\ldots,2\pi i\lambda_r
\right\rangle
\subset\mathbb C
\) be the additive subgroup. 
Then the following are equivalent:
\begin{enumerate}[label=\textup{(\roman*)}]
\item $\Gamma$ is not dense in $\mathbb C$;
\item there exists a nonempty proper closed real-analytic subset
\(
\gamma\subset\mathbb C
\)
of pure real dimension one satisfying
\(
\gamma+\Gamma=\gamma.
\)
\end{enumerate}
Thus no proper $\Gamma$-invariant real-analytic subset of pure real
dimension one exists if and only if
\(
\overline\Gamma=\mathbb C.
\)
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Assume first that $\overline\Gamma=\mathbb C$, and let
$\gamma\subset\mathbb C$ be nonempty and closed with
$\gamma+\Gamma=\gamma$. Since translations depend continuously on the
translating parameter,
\[
\gamma+\overline\Gamma=\gamma.
\]
Hence $\gamma+a=\gamma$ for every $a\in\mathbb C$. Choosing
$z_0\in\gamma$ gives $z_0+\mathbb C\subset\gamma$, and therefore
$\gamma=\mathbb C$.

Conversely, suppose that $\Gamma$ is not dense. Its closure is a proper
closed subgroup of the real Lie group
$\mathbb C\simeq\mathbb R^2$. Up to a real-linear change of coordinates,
the possibilities are
\[
\{0\},\qquad
\mathbb Z v,\qquad
\mathbb Z v+\mathbb Z w,\qquad
\mathbb R v,\qquad
\mathbb R v+\mathbb Z w,
\]
where $v$ and $w$ are linearly independent over $\mathbb R$ whenever
both occur.

If the group is contained in a real line $\mathbb R v$, an affine line
parallel to $v$ is invariant. If
\[
\overline\Gamma=\mathbb R v+\mathbb Z w,
\qquad w\notin\mathbb R v,
\]
then
\[
\gamma=
\bigcup_{k\in\mathbb Z}\bigl(\mathbb R v+kw\bigr)
\]
is a proper invariant real-analytic set of pure real dimension one; in
suitable real-linear coordinates it is given by
\[
\sin(\pi y)=0.
\]
If $\overline\Gamma$ is a rank-two lattice, then in lattice coordinates
one may take
\[
\gamma=
\{\sin(\pi x)\sin(\pi y)=0\}.
\]
The trivial and discrete cyclic cases are immediate. Thus every proper
closed subgroup of $\mathbb C$ preserves a proper real-analytic subset
of pure real dimension one.

<!-- proposition-1: body_md -->
\label{prop:period-compatible-leviflats}
Assume that there exists a nonempty proper real-analytic subset

\gamma\subset\mathbb C
\]
of pure real dimension one satisfying
\[
\gamma+\Gamma=\gamma.

Then the local sets

nonconstant meromorphic first integral and choose distinct constants

\sum_{j=1}^r

<!-- proposition-2: body_md -->
\label{proposition:logarithmic}
Let $\Pi_\eta\subset\mathbb C$ be the actual period group of $\eta$ on
$X^\circ$. For the logarithmic foliation $\mathcal F_\eta$, the
following are equivalent:
\begin{enumerate}[label=\textup{(\roman*)}]
\item there exists $a\in\mathbb C^*$ such that
\[
\operatorname{Re}(a\tau)=0
\qquad\text{for every }\tau\in\Pi_\eta;
\]
\item after multiplication of $\eta$ by a nonzero constant, the
multivalued primitive has a single-valued real part on $X^\circ$;
\item $\mathcal F_\eta|_{X^\circ}$ admits an infinite one-parameter
family of invariant Levi-flat level hypersurfaces of the form
\[
\{\operatorname{Re}(aF)=c\},
\qquad c\in\mathbb R.
\]
\end{enumerate}
In particular, the collinearity condition
$a\lambda_j\in\mathbb R$ for all $j$ implies these equivalent statements.
When the actual periods are generated by meridians about distinct
reduced polar components with respective residues $\lambda_j$, the
collinearity condition is also necessary.
<!-- end proposition-2 -->
<!-- proof-4: body_md -->
The equivalence of \textup{(i)} and \textup{(ii)} follows from the
actual period group. If all periods become purely imaginary after
multiplication by $a$, then $\operatorname{Re}(aF)$ is single-valued and
its regular levels give \textup{(iii)}. Conversely, the expression in
\textup{(iii)} can be globally single-valued only if every period of
$aF$ has zero real part. If $a\lambda_j$ is real for every $j$,
then every period of $a\eta$ is purely imaginary, since the winding
numbers of the functions $f_j$ are integers. Under the final
meridian-generation assumption, the periods $2\pi i\lambda_j$ occur
among the generators, so the same condition is also necessary.

<!-- definition-2: body_md -->
\label{definition:transvseparation}
Let $p$ be a dicritical singularity of $\mathcal F$ contained in an invariant irreducible real-analytic variety of pure real codimension one $M$ with
$\operatorname{Sing}(M)\subset\operatorname{Sing}(\mathcal F)$. Let $U_p$ be a neighborhood of $p$,
let $S$ be a real-analytic one-dimensional manifold, and let
\[
h:M_{\mathrm{reg}}\cap U_p\longrightarrow S
\]
be a real-analytic map on the regular part near $p$. We say that $h$
\emph{separates the nearby Levi leaves at $p$} if $h$ is constant on
each local Levi leaf sufficiently near $p$ and the values assigned to
distinct such leaves are distinct. Equivalently, the induced map on
the set of nearby local Levi leaves is injective. The target $S$ may,
in particular, be $\mathbb R$ or $\mathbb{RP}^1$.

\quad(\lambda\in\mathbb R),\qquad
L_\infty=\{y=0\}\cap M_{\mathrm{reg}},

<!-- proposition-3: body_md -->
\label{prop:nearMreg}
Let $X$ be a connected complex manifold, let $\mathcal F$ be a codimension-one
holomorphic foliation on $X$, and let $M\subset X$ be a connected irreducible real-analytic variety of pure real codimension one invariant by $\mathcal F$. Let
\[
p\in M\cap\operatorname{Sing}(\mathcal F)
\]
be a pencil singularity for $M$, and let
\[
F_p=\frac{f}{g}:U_p\dashrightarrow\mathbb P^1
\]
be a minimal local meromorphic first integral for the germ $(\mathcal F,p)$.
Then $F_p$ extends by holonomy to a single-valued meromorphic first
integral on an open neighborhood of $M_{\mathrm{reg}}$.
<!-- end proposition-3 -->
<!-- proof-5: body_md -->
Fix $x\in M_{\mathrm{reg}}$ and denote by $L_x$ the Levi leaf through $x$.
By the accumulation hypothesis,
\[
p\in\overline{L_x}.
\]
Hence $L_x$ meets $U_p$.  Choose a regular point
$y\in L_x\cap U_p$ and a path $\gamma\subset L_x$ joining $x$ to
$y$.  Choose small complex transversals $\Sigma_x$ and $\Sigma_y$.
Holonomy along $\gamma$ defines
\[
h_\gamma:\Sigma_x\longrightarrow\Sigma_y.
\]
The meromorphic function
\[
g_\gamma=F_p\circ h_\gamma
\]
on $\Sigma_x$ extends constantly along plaques to a meromorphic first
integral $G_\gamma$ on a flow box around $x$.

Let $\gamma_1,\gamma_2$ be two such choices, and let $G_1,G_2$ be the
corresponding first integrals.  Put
\[
I=\Sigma_x\cap M_{\mathrm{reg}}.
\]
For generic $z\in I$, the two continuations follow the same global
Levi leaf.  By the constancy hypothesis,
\[
G_1(z)=G_2(z)
\]
for generic $z\in I$.  Since $G_1|_{\Sigma_x}$ and
$G_2|_{\Sigma_x}$ are one-variable meromorphic functions and agree on
a real set with an accumulation point,
\[
G_1|_{\Sigma_x}\equiv G_2|_{\Sigma_x}.
\]
Both are constant along plaques, hence $G_1\equiv G_2$ in the flow
box.  The same argument on overlaps gives a meromorphic function
\[
G\in\mathcal M(N_{\mathrm{reg}})
\]
on an open neighborhood $N_{\mathrm{reg}}$ of $M_{\mathrm{reg}}$, and
\[
dG\wedge\omega=0
\]
for every local defining integrable $1$-form $\omega$ of $\mathcal F$.

\label{lem:monodromy}
Let $D$ be an irreducible dicritical component which is generically of
local product type, let $D^\circ$ be its connected generic local-product
locus, and fix $p\in D^\circ$. Suppose that a minimal local meromorphic first integral
\[

\]
has the property that, for every generic global Levi leaf $L$
represented near $p$, the restriction $F_p|_{L\cap U_p}$ is constant. Then the projective
monodromy of the transverse $\mathbb P^1$-parameter along $D^\circ$ is trivial.
<!-- end lemma-4 -->
<!-- proof-7: body_md -->
Normalize the parameter at $p$ by $H_p=F_p$.  Analytic continuation
around a loop $\gamma\in\pi_1(D^\circ,p)$ gives an element
\[
\rho(\gamma)\in\operatorname{PGL}(2,\mathbb C).
\]
For a generic Levi leaf $L$ represented near $p$, let
\[
\lambda(L)=H_p(L).
\]
After continuation around $\gamma$, the same leaf is assigned the
parameter $\rho(\gamma)(\lambda(L))$.  The constancy hypothesis at $p$ gives
\[
\rho(\gamma)(\lambda(L))=\lambda(L)
\]
for every generic Levi leaf.  The local Levi-flat supplies infinitely
many such parameters with an accumulation point.  A nonidentity
M\"obius transformation has at most two fixed points.  Therefore
$\rho(\gamma)=\operatorname{id}$ for every $\gamma$, as claimed.

<!-- proposition-4: body_md -->
\label{prop:dicritical-extension}
The meromorphic first integral $G$ constructed in
Proposition~\ref{prop:nearMreg} extends meromorphically through every
point of $D^\circ$. If, in addition, the no-transverse-degree-jump
condition of Definition~\ref{def:no-degree-jump} holds, then $G$

<!-- proof-8: body_md -->

$N_{\mathrm{reg}}$ the functions $G$ and $\widehat H_q$ are first integrals
of the same foliation and have the same values on generic Levi leaves.

$G$ on a nonempty open set. Uniqueness of meromorphic continuation
therefore gives the extension through $q$.

\label{ex:degree-jump}
The condition in Definition~\ref{def:no-degree-jump} is not automatic.
Let $d\ge2$ and consider in $(\mathbb C^3,0)$ the meromorphic first integral

H(x,y,z)=\frac{x^d}{y^d z}.

The associated foliation is defined, after clearing denominators, by
\[
\omega=d yz\,dx-d xz\,dy-xy\,dz,
\]
and its singular set contains
\[
D=\{x=y=0\}.
\]
At $q=(0,0,0)$ the first integral $H$ is primitive:  For a generic $c\in\mathbb C^*$, the fiber germ

\]

the prime element $z$ \cite[\S 9.4]{DummitFoote}. On the other hand, at a nearby point

\[
G_p(x,y,z)=\frac{x}{a(z)y}

is a primitive local meromorphic first integral and
\[
H=G_p^d.
\]
Thus a primitive first integral at the exceptional point $q$ restricts

<!-- lemma-5: body_md -->
\label{lem:nondicritical}
The meromorphic first integral $G$ extends meromorphically to a
neighborhood of $q$.

<!-- proof-9: body_md -->
Since $q$ is non-dicritical, a minimal local meromorphic first integral

separatrices through $q$. After shrinking $U_q$ and choosing the
appropriate target chart, we may therefore take a minimal holomorphic
first integral

On the regular locus of $H_q$, invariance of $M$ means that $M$ is
saturated by the level hypersurfaces of $H_q$. Hence the image of the
regular part of the relevant local component of $M$ in the one-dimensional
leaf space is a real-analytic curve germ. After selecting that component,
there is a germ of a real-analytic curve $\gamma\subset\mathbb C$, with
$c\in\gamma$, such that
\[
M\cap U_q=\overline{H_q^{-1}(\gamma)}.
\]
In particular,

in $\operatorname{Sing}(\mathcal F)$. We may therefore choose
\[

meromorphic germ $\varphi$ at $c$ such that
\[
G=\varphi\circ H_q
\]
on a neighborhood of $x$. Therefore $\varphi\circ H_q$ is meromorphic
near $q$ and agrees with $G$ on a nonempty open subset of their common
domain. By uniqueness of meromorphic continuation, it extends $G$
through $q$.

\label{prop:pseudoconvex}
Let $X$ be a connected complex manifold and let $M\subset X$ be a singular Levi-flat variety. Assume that, near every point $q\in M$, the Levi foliation is defined by a local meromorphic first integral. Then every local side of $M$ is pseudoconvex, including at singular points of $M$.

<!-- lemma-6: body_md -->
\label{lem:boundary-hartogs}
Let $Y$ be a compact complex manifold, let $M\subset Y$ be compact, and
let $\Omega$ be a connected component of $Y\setminus M$. Assume that
$\Omega$ is Stein and $\dim_\mathbb C\Omega\ge2$. Let $N$ be an open
neighborhood of $M$ in $Y$ and let
\[
g\in\mathcal M(N\cap\Omega).
\]
Then there are a smaller neighborhood $N'\subset N$ of $M$ and a
meromorphic function
\[
\widetilde g\in\mathcal M(\Omega)
\]
such that
\[
\widetilde g=g\qquad\text{on }N'\cap\Omega.
\]
<!-- end lemma-6 -->
<!-- proof-12: body_md -->
Since $Y$ is compact and $N$ contains $\partial\Omega\subset M$, the set
$\Omega\setminus N$ is relatively compact in $\Omega$. A connected
Stein manifold of complex dimension at least two has only one end; see,
for instance, Serre \cite{SerreDuality}. Hence there exists a compact set
$K\Subset\Omega$ containing $\Omega\setminus N$ such that
\[
\Omega\setminus K
\]
is connected. In particular,
\[
\Omega\setminus K\subset N\cap\Omega.
\]
The meromorphic Hartogs theorem
\cite[Theorem~2.2]{MerkerPorten} extends

\label{lem:transverse-rigidity}
Let
\[
G\subset\operatorname{Diff}(\mathbb C,0)
\]
be a group of germs of holomorphic diffeomorphisms. Assume that there
exists an infinite family
\[
\{\gamma_\alpha\}_{\alpha\in A}
\]
of pairwise distinct smooth real-analytic curve germs through the
origin such that
\[
g(\gamma_\alpha)=\gamma_\alpha
\]
for every $g\in G$ and every $\alpha\in A$. Then $G$ is abelian.
<!-- end lemma-8 -->
<!-- proof-16: body_md -->
For a smooth real-analytic curve germ $\gamma\subset(\mathbb C,0)$, denote by
$\sigma_\gamma$ the Schwarz reflection across $\gamma$.  Thus
$\sigma_\gamma$ is the antiholomorphic involution fixing $\gamma$
pointwise.  If $\phi$ is a local holomorphic coordinate straightening
$\gamma$ to $\mathbb R$, then
\[
\sigma_\gamma=\phi^{-1}\circ c\circ\phi,
\qquad c(z)=\overline z.
\]
If $g\in\operatorname{Diff}(\mathbb C,0)$ preserves $\gamma$, then
$g\sigma_\gamma g^{-1}$ is again the antiholomorphic involution fixing
$\gamma$ pointwise.  By uniqueness of Schwarz reflection,
\[
g\circ\sigma_\gamma=\sigma_\gamma\circ g.
\tag{\ref{lem:transverse-rigidity}.1}
\]
Hence every element of $G$ commutes with every reflection associated
with the family.

We distinguish two cases.

Suppose first that the set of tangent lines
\[
\{T_0\gamma_\alpha:\alpha\in A\}
\]
is infinite. Since any $g\in G$ preserves one such real tangent line,
its multiplier is real:
\[
g'(0)\in\mathbb R^*.
\]
If some $f\in G$ has $|f'(0)|\ne1$, then $f$ is analytically
linearizable. In a linearizing coordinate we may write
\[
f(z)=\lambda z,
\qquad \lambda\in\mathbb R^*,\quad |\lambda|\ne1.
\]
Every smooth real-analytic $f$-invariant curve through the origin is a
straight real line. Indeed, after a rotation, and after replacing $f$
by a suitable iterate or inverse, write the curve as
\[
y=\varphi(x),\qquad 0<\lambda<1,
\]
with $\varphi(0)=\varphi'(0)=0$. Invariance gives
\[
\varphi(\lambda x)=\lambda\varphi(x).
\]
Writing $\varphi(x)=\sum_{j\ge2}a_jx^j$ yields
$a_j(\lambda^j-\lambda)=0$, and therefore $\varphi\equiv0$.
Consequently our family contains infinitely many distinct real lines
\[
L_\theta=e^{i\theta}\mathbb R.
\]
If
\[
g(z)=\sum_{j\ge1}b_jz^j\in G,
\]
then $g(L_\theta)=L_\theta$ implies
\[
b_je^{i(j-1)\theta}\in\mathbb R
\]
for every $j$ and infinitely many values of $\theta$ modulo $\pi$.
For $j\ge2$ this forces $b_j=0$. Hence every element of $G$ is a real
homothety, and $G$ is abelian.

It remains, in the present case, to assume that $|g'(0)|=1$ for every
$g\in G$. Since all multipliers are real,
\[
g'(0)\in\{1,-1\}.
\]
If $g'(0)=1$ and $g\ne\operatorname{id}$, write
\[
g(z)=z+az^{k+1}+O(z^{k+2}),
\qquad a\ne0.
\]
Let $\gamma$ be a smooth real-analytic invariant curve with tangent
vector $v\ne0$, and write
\[
\gamma(t)=vt+O(t^2),
\qquad
g(\gamma(t))=\gamma(\rho(t)).
\]
Comparison at the first nonlinear order gives
\[
av^k\in\mathbb R.
\]
Thus a nontrivial germ tangent to the identity admits invariant smooth
real-analytic curves in only finitely many tangent directions, contrary
to the present hypothesis. Hence $g'(0)=1$ implies
$g=\operatorname{id}$. If $g'(0)=-1$, then $(g^2)'(0)=1$, so
$g^2=\operatorname{id}$. Moreover, if $g$ and $h$ both have multiplier
$-1$, then $(gh)'(0)=1$, whence $gh=\operatorname{id}$ and $g=h$.
Thus $G$ has at most two elements and is again abelian.

Suppose now that only finitely many tangent directions occur. Since the
family is infinite, there are two distinct curves $\gamma_1$ and
$\gamma_2$ with
\[
T_0\gamma_1=T_0\gamma_2.
\]
Let $\sigma_1$ and $\sigma_2$ be their Schwarz reflections and set
\[
F=\sigma_2\circ\sigma_1.
\]
Then $F$ is holomorphic and $F'(0)=1$. Moreover $F$ is not the
identity, since otherwise $\sigma_1=\sigma_2$ and therefore
$\gamma_1=\gamma_2$. Hence
\[
F(z)=z+az^{k+1}+\cdots,
\qquad a\ne0.
\]
By \textup{(\ref{lem:transverse-rigidity}.1)}, every element of $G$
commutes with both $\sigma_1$ and $\sigma_2$, and therefore
\[
G\subset Z(F),
\]
where $Z(F)$ denotes the centralizer of $F$ in
$\operatorname{Diff}(\mathbb C,0)$.

We recall the standard formal argument showing that $Z(F)$ is abelian.
There is a unique formal vector field
\[
X=\log F
\]
with $F=\exp X$. If $gF=Fg$, then
\[
g_*X=X.
\]
Writing
\[
X=a(z)\frac{\partial}{\partial z},
\]
consider the formal meromorphic differential
\[
\omega=\frac{dz}{a(z)}.
\]
The identity $g_*X=X$ is equivalent to $g^*\omega=\omega$. If $T$ is a
formal primitive of $\omega$, then for every $g\in Z(F)$ one has
\[
T\circ g=T+c_g
\]
for some constant $c_g$. Therefore, for $g,h\in Z(F)$,
\[
T\circ g\circ h=T+c_h+c_g
=T+c_g+c_h=T\circ h\circ g.
\]
Since $T$ is a nonconstant formal Fatou coordinate, this implies
$g\circ h=h\circ g$. Thus $Z(F)$, and hence $G$, is abelian.

\label{lem:closed-rational-logarithmic}
Let $\mathcal F$ be a holomorphic foliation on $\mathbb P^2$ defined by a
closed rational $1$-form. Assume that the reduction of singularities of

Let $\omega$ be a closed rational $1$-form defining $\mathcal F$. Write
\[
\omega=\alpha+dR,
\]
where $\alpha$ is logarithmic and
\(
R\in\mathbb C(\mathbb P^2).
\)
If $R$ is constant, there is nothing to prove. Also, if $\mathcal F$
admits a rational first integral $H$, then
\[
\frac{dH}{H}
\]
is a closed logarithmic $1$-form defining the same foliation. We may
therefore assume, seeking a contradiction, that $R$ is nonconstant and
that $\mathcal F$ admits no rational first integral.

Start with a reduction of singularities of $\mathcal F$. By hypothesis
all its reduced singularities are nondegenerate. Further point blow-ups
do not create saddle-nodes: blowing up a regular point produces at worst
a reduced nondegenerate singularity, and blowing up a reduced
nondegenerate singularity whose eigenvalue quotient is not a positive
rational again produces only reduced nondegenerate singularities. Thus
the absence of saddle-nodes persists under any additional blow-ups.

We may therefore resolve simultaneously the remaining indeterminacy of
$R$ and the polar divisor of $\alpha$:
\[
\pi:S\longrightarrow\mathbb P^2,
\]
in such a way that
\[
f:=R\circ\pi:S\longrightarrow\mathbb P^1
\]
is a holomorphic map and the relevant divisors have normal crossings.
We have
\[
\pi^*\omega=\widetilde\alpha+df,
\qquad
\widetilde\alpha=\pi^*\alpha,
\]
with $\widetilde\alpha$ logarithmic.

$\widetilde\alpha$ is vertical with respect to $f$. Suppose, on the
contrary, that $H$ is a horizontal polar component. Then $H$ meets the
fiber
\[
f^{-1}(\infty).
\]

\[
p\in H\cap f^{-1}(\infty)
\]
at which $H$ meets a unique irreducible component of that fiber. In

\[

\qquad

\qquad m\geq1,
\]
and
\[

\lambda\frac{dy}{y}
+\mu\frac{dx}{x}
+\beta,
\qquad
\lambda\neq0,
\]
where $\beta$ is holomorphic. Hence
\[
\pi^*\omega
=
-mx^{-m-1}\,dx
+\lambda\frac{dy}{y}
+\mu\frac{dx}{x}
+\beta.
\]
Multiplying by $x^{m+1}y$, which does not change the foliation, gives a

\[

\bigl(-my+\text{higher order terms}\bigr)\,dx
+
\bigl(\lambda x^{m+1}+\text{higher order terms}\bigr)\,dy.
\]
Its associated vector field has one zero eigenvalue and one nonzero

\pi:S=\operatorname{Bl}_{p_1,\ldots,p_m}\mathbb P^2\longrightarrow\mathbb P^2
\]
be the blow-up of the dicritical singularities occurring in the theorem,
and let $\widetilde C$ be the strict transform of $C$. Pull back the
formal form to the regular part of $\widetilde C$. By hypothesis, every
singularity of the transformed foliation on $\widetilde C$ lying over a
dicritical point is reduced, non-dicritical, and is not a saddle-node;
the other singularities were already reduced and non-saddle-node. Since $C$ is
contained in infinitely many invariant real-analytic Levi-flat germs,
the same is true after pull-back along $\widetilde C$. The local theorem
of Cerveau--Lins Neto \cite{CLN}, or Brunella's formulation
\cite{Brunella}, supplies a nonconstant local meromorphic first integral
at these singular points. In the reduced non-dicritical non-saddle-node
case this is the analytically linearizable resonant situation; hence in
suitable local coordinates one has a primitive first integral
\[
H=x^n y^m,\qquad n,m\ge1,
\]
and a closed logarithmic defining form
\[
\eta_0=\frac{dH}{H}
=n\frac{dx}{x}+m\frac{dy}{y}.
\]
On the punctured formal neighborhood of the branch of $\widetilde C$
approaching such a point,
\[
\widehat\eta_L=\Phi\,\eta_0.
\]
Closedness gives $d\Phi\wedge\eta_0=0$, so $\Phi$ is a formal
meromorphic first integral and therefore $\Phi=\varphi(H)$ for a
one-variable formal meromorphic germ $\varphi$. Since $H=x^ny^m$
vanishes on the invariant axes, $\varphi(H)$ is formal meromorphic at
the origin. Thus the form extends formally meromorphically through the
point.

We must also check compatibility when two invariant components meet at
a corner. Let $q$ be such a point and let $\widehat\eta_1$ and
$\widehat\eta_2$ be the closed formal meromorphic forms obtained from
the two sides. Since they define the same foliation, on their common
formal domain one has
\[
\widehat\eta_1=\Psi\,\widehat\eta_2,
\]
where closedness implies
\[
d\Psi\wedge\widehat\eta_2=0.
\]
Hence $\Psi$ is a formal meromorphic first integral. If the germ has no
nonconstant meromorphic first integral, then $\Psi$ is constant and the
two forms glue after a constant normalization. The only remaining
possibility at a reduced non-dicritical corner is the resonant
integrable case, analytically conjugate to
\[
d(x^n y^m)=0.
\]
There the finite-to-finite Dulac correspondence identifies the
transverse dynamics of the two components. Equivalently, one adjoins the
holonomy groups of the adjacent components through the Dulac
correspondence. By construction, this adjoined holonomy group is still
contained in the virtual holonomy group of the leaf under consideration.
Since the latter is abelian in our situation, the adjoined group is
abelian as well. Hence the invariant transverse formal differential is
the same on both sides, up to the same constant normalization, and the
suspended closed forms are compatible. This is the standard adjunction
mechanism for holonomy groups along a reducible invariant divisor; see
Camacho--Lins Neto--Sad~\cite{CamachoLinsNetoSad},
Paul~\cite{Paul1999}, Camacho--Sc\'ardua~\cite{CamachoScarduaLiouvillian},
and Sc\'ardua~\cite{ScarduaBoundedGeometry}.

Consequently the local extensions agree through all corners and all
singular points of $\widetilde C$. Performing this at all points of
$\widetilde C$ gives a global closed formal meromorphic $1$-form
\[
\widehat\eta
\]
on the formal completion $\widehat S_{\widetilde C}$, defining the
transformed foliation formally along $\widetilde C$.

We now use the numerical hypothesis. Write
\[
d=\deg C,\qquad d_j=\operatorname{mult}_{p_j}(C).
\]
On the blown-up surface,
\[
\widetilde C^{\,2}=d^2-\sum_{j=1}^m d_j^2.
\]
Since
\(
\sum_{j=1}^m d_j<d,
\)
we have
\[
\sum_{j=1}^m d_j^2
\le \left(\sum_{j=1}^m d_j\right)^2<d^2,
\]
and therefore
\(
\widetilde C^{\,2}>0.
\)
This is the numerical condition used by Sad in his treatment of
controlled dicritical singularities; radial singularities are among the
allowed examples \cite{Sad1995}.

Hironaka--Matsumura, Proposition~3.6 in \cite{HM}, states that if $D$ is
an effective divisor on a nonsingular projective surface, $D^2>0$, and
every prime component of $D$ has nonnegative self-intersection, then the
field of formal-rational functions along $\operatorname{Supp}(D)$ equals
the rational function field of the surface. Applying this to the
irreducible divisor $D=\widetilde C$ gives
\[
K\!\left(\widehat S_{\widetilde C}\right)=K(S).
\]
Choose rational coordinates $(u,v)$ on a dense affine chart of $S$ and
write
\[
\widehat\eta=\widehat A\,du+\widehat B\,dv.
\]
The coefficients belong to
$K(\widehat S_{\widetilde C})=K(S)$, hence they are the formal germs of
rational functions $A,B\in K(S)$. Therefore
\[
\eta_S=A\,du+B\,dv
\]
is a rational $1$-form on $S$ whose formal completion along
$\widetilde C$ is $\widehat\eta$. Injectivity of the completion map
shows that
\[
d\eta_S=0
\]
and, if $\omega_S$ is a rational defining form of the transformed
foliation, that
\[
\eta_S\wedge\omega_S=0.
\]
Thus $\eta_S$ is a closed rational defining form for the transformed
foliation.

Finally, $\pi$ is birational, so
\[
K(S)=K(\mathbb P^2)
\]
and hence
\[
\Omega^1_{K(S)/\mathbb C}=\Omega^1_{K(\mathbb P^2)/\mathbb C}.
\]
Thus $\eta_S$ is the pull-back, in the rational category, of a unique
rational $1$-form $\eta$ on $\mathbb P^2$. The identities above descend to
\[
d\eta=0,
\qquad
\eta\wedge\omega_{\mathcal F}=0,
\]
for any rational defining form $\omega_{\mathcal F}$ of $\mathcal F$. Consequently
$\eta$ is a nonzero closed rational $1$-form defining $\mathcal F$.

\label{lem:mobius-separated-curves}
Let
\[
G\subset\operatorname{PGL}(2,\mathbb C)
\]
be a group of M\"obius transformations. Assume that there exists an
infinite family
\[
\{\gamma_j\}_{j\in J}
\]
of pairwise disjoint, connected, compact real-analytic subsets of
$\mathbb P^1$, each of pure real dimension one, such that
\[
g(\gamma_j)=\gamma_j
\]
for every $g\in G$ and every $j\in J$. Then there exists a nonzero
rational $1$-form $\theta$ on $\mathbb P^1$ such that
\[
g^*\theta=\theta
\qquad\text{for every }g\in G.
\]
In particular, $\theta$ is closed.

<!-- proof-22: body_md -->
Choose three distinct members
\[
\gamma_1,\gamma_2,\gamma_3
\]
and points $z_j\in\gamma_j$. Since the sets $\gamma_j$ are compact and
pairwise disjoint, their mutual spherical distances are positive.

Let $(g_\nu)$ be any sequence in $G$. Since
\[
g_\nu(z_j)\in\gamma_j,
\qquad j=1,2,3,
\]
after passing to a subsequence we may assume that
\[
g_\nu(z_j)\longrightarrow w_j\in\gamma_j,
\qquad j=1,2,3.
\]
The points $w_1,w_2,w_3$ are pairwise distinct. Since a M\"obius
transformation is uniquely determined by the images of three distinct
points, the sequence $(g_\nu)$ has a convergent subsequence in
$\operatorname{PGL}(2,\mathbb C)$. Hence
\[
K=\overline G
\]
is a compact subgroup of $\operatorname{PGL}(2,\mathbb C)$. Passing to the
limit in $g(\gamma_j)=\gamma_j$, we also have
\[
k(\gamma_j)=\gamma_j
\qquad\text{for every }k\in K\text{ and every }j.
\]

If $K$ is finite, let
\[
q:\mathbb P^1\longrightarrow\mathbb P^1/K
\]
be the quotient map. Since $\mathbb P^1/K\simeq\mathbb P^1$, choose a nonzero
rational $1$-form $\beta$ on the quotient and put
\[
\theta=q^*\beta.
\]
Then $\theta$ is a nonzero rational $K$-invariant $1$-form, hence it is
$G$-invariant.

Assume now that $K$ is infinite. Every compact subgroup of
$\operatorname{PGL}(2,\mathbb C)$ is M\"obius-conjugate to a subgroup of
$\operatorname{PSU}(2)$. Since $K$ preserves the proper one-dimensional
sets $\gamma_j$, it cannot be all of $\operatorname{PSU}(2)$. Hence its
identity component $K^0$ is a circle subgroup. After a M\"obius change
of coordinate we may assume that
\[
K^0=\{z\mapsto e^{it}z;\ t\in\mathbb R\}.
\]

Each $\gamma_j$ is connected, $K^0$-invariant, and of pure real
dimension one. The nontrivial $K^0$-orbits are the circles
\(
|z|=r.
\)
If some $\gamma_j$ contained orbits with more than one value of $r$, it
would contain a real two-dimensional annular subset, contrary to its
pure real dimension one. Thus
\(
\gamma_j=\{|z|=r_j\}
\)
for some $r_j>0$.

It remains to show that $K$ itself is contained in this circle group.
An element of the normalizer of $K^0$ which is not in $K^0$ interchanges
the fixed points $0$ and $\infty$ and has the form
\[
z\longmapsto\frac{a}{z}.
\]
Such a map sends
\[
\{|z|=r\}
\quad\text{to}\quad
\{|z|=|a|/r\}.
\]
It can therefore preserve individually at most one circle of the
family, namely the circle satisfying $r^2=|a|$. Since the family
$\{\gamma_j\}$ is infinite and every member is preserved individually
by every element of $K$, no such element can belong to $K$. Therefore
\[
K\subset\{z\mapsto e^{it}z\}.
\]
Consequently
\[
\theta=\frac{dz}{z}
\]
is invariant under $K$, and hence under $G$. Pulling $\theta$ back by
the initial M\"obius conjugacy gives the required nonzero rational
$G$-invariant $1$-form on $\mathbb P^1$. Since every meromorphic $1$-form on
a Riemann surface is closed, the proof is complete.

\label{thm:stein}
Let $X$ be a Stein manifold of complex dimension $n\ge3$, let $\mathcal F$ be
a codimension-one holomorphic foliation on $X$, and let $M\subset X$
be a connected irreducible real-analytic variety of pure real codimension one 
invariant by $\mathcal F$. Assume:

\begin{enumerate}[label=\textup{(\roman*)}]
\item
\[
\operatorname{Sing}(M)\subset\operatorname{Sing}(\mathcal F);
\]

\item all dicritical singularities of $\mathcal F$ contained in $M$ belong to
an irreducible codimension-two analytic subset
\[
D\subset M\cap\operatorname{Sing}(\mathcal F),
\]
and $D$ is generically of local product type;

\item there exists a pencil singularity for $M$ in the local-product
locus of $D$;

\item the generic transverse parameter along $D^\circ$ has no
transverse degree jump along $D$ in the sense of
Definition~\ref{def:no-degree-jump}.
\end{enumerate}

Then there exist an open neighborhood $N\supset M$ and a nonconstant
meromorphic first integral
\[
G\in\mathcal M(N)
\]
of $\mathcal F|_N$.

If, in addition, there exists a compact set $K\subset X$ such that
\[
X\setminus K\subset N
\qquad\text{and}\qquad
X\setminus K\ \text{is connected},
\]
then $G$ extends meromorphically to $X$, and $\mathcal F$ admits a nonconstant
global meromorphic first integral on $X$.

<!-- proof-25: body_md -->
Let $L\subset M_{\mathrm{reg}}$ be a Levi leaf. Its closure
\(
\overline L\subset M\subset\mathbb P^n
\)
is a nonempty compact $\mathcal F$-invariant set. By a theorem of
Lins Neto on nontrivial minimal sets in projective space
\cite{LinsNeto1999}, a codimension-one holomorphic foliation on
$\mathbb P^n$, $n\ge3$, has no nontrivial minimal set. Consequently
$\overline L$ meets $\operatorname{Sing}(\mathcal F)$. Since every singularity of $\mathcal F$
contained in $M$ is dicritical, every Levi leaf accumulates on the
dicritical locus in $M$.

The finite family $U_1,\ldots,U_r$ therefore meets the accumulation
set of every Levi leaf. The constancy assumption allows the local
parameters $H_j$ to be transported along the Levi foliation, and the
M\"obius compatibility and cocycle relations identify the resulting
parameters on overlaps. Thus the same holonomy argument as in
Proposition~\ref{prop:nearMreg} produces a single-valued meromorphic

near $M$ if necessary extends the first integral through every
complementary component. The resulting global meromorphic function on
$\mathbb P^n$ is necessarily rational.
