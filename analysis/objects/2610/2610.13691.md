<!-- theorem-1: body_md -->
\label{thm:main}
If $r_1,\ldots,r_t$ are positive integers with $r_1^{-1}+\cdots+r_t^{-1}=1$, then for every positive integer $m$, the set of positive odd integers $n$ with $r(n)\ge m$ has a positive lower asymptotic density.
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{cor:criterion}
Let $r_1,\ldots,r_t$ and $m$ be positive integers. Then the set of positive odd integers $n$ with $r(n)\ge m$ has a positive lower asymptotic density if and only if $ r_1^{-1}+\cdots+r_t^{-1}\ge1$.
<!-- end corollary-1 -->
<!-- lemma-1: body_md -->
\cite[Lemma~2.1]{ChenII}\label{lem:Acount}
Let $r_1,\ldots,r_t$ be positive integers. Then, as $x\to+\infty$, we have
\[
 \mathcal A(x)=\left(1+o(1)\right)\frac{1}{w}(\log 2)^{-\frac{1}{r_1}-\cdots-\frac{1}{r_t}}(\log x)^{\frac{1}{r_1}+\cdots+\frac{1}{r_t}}.
\]
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\cite[equation (3.17)]{DingLi}\label{lem:pair}
Assume that $\sigma=r_1^{-1}+\cdots+r_t^{-1}=1$. Let $v$ be a positive integer. Then, as $x\to+\infty$, we have
\[
 \sum_{\substack{b<b'\le x\\b,b'\in\mathcal A}}
 \prod_{\substack{p\mid(b-b')\\2<p\le\log x}}
 \left(1+\frac{v^3}{p}\right)
 \ll_v (\log x)^{2}.
\]
<!-- end lemma-2 -->
<!-- lemma-3: body_md -->
\label{lem:rough}
Assume that $\sigma=r_1^{-1}+\cdots+r_t^{-1}=1$. Let $q$ be a fixed positive integer and let
$0<\rho<1$. Suppose that $X$ is a sufficiently large integer,
$3\le Y\le X$, and $T\in\mathbb Z$. For any integer $X\le n<2X$, let
\[
 U(n)=|\{a\in\mathcal A\cap[1,Y]:\gcd(n+T-a,P(X^\rho))=1\}|.
\]
Then
\begin{equation}\label{eq:roughmoment}
 \sum_{X\le n<2X}U(n)^q\ll_{q,\rho}X.
\end{equation}
<!-- end lemma-3 -->
<!-- proof-1: body_md -->
If $3\le Y<Y_0$, then
\[
 U(n)\le \mathcal A(Y_0),
\]
and hence
\[
 \sum_{X\le n<2X}U(n)^q
 \le X\mathcal A(Y_0)^q
 \ll_q X.
\]
Thus we may assume that $Y$ is a sufficiently large integer.

Let $z=X^\rho$ and $M=\mathcal A(Y)$. If $M=0$, then $U(n)=0$ and so \eqref{eq:roughmoment} is immediate.
For $a\in\mathcal A\cap[1,Y]$, write
\[
 \mathbf 1_a(n)=\mathbf 1_{\gcd(n+T-a,P(z))=1}.
\]
Then
\[
 U(n)=\sum_{a\in\mathcal A\cap[1,Y]}\mathbf 1_a(n)
\]
and
\[
 \sum_{X\le n<2X}U(n)^q=\sum_{a_1,\ldots,a_q\in\mathcal A\cap[1,Y]}\sum_{X\le n<2X}\mathbf 1_{a_1}(n)\cdots\mathbf 1_{a_q}(n).
\]
For a $q$-tuple $(a_1,\ldots,a_q)$, if there are $1\le s\le q$ distinct numbers $b_1,\ldots,b_s$, then
\[
 \mathbf 1_{a_1}(n)\cdots\mathbf 1_{a_q}(n)=\mathbf 1_{b_1}(n)\cdots\mathbf 1_{b_s}(n).
\]
Fix $1\le s\le q$ and choose pairwise distinct
$b_1,\ldots,b_s\in\mathcal A\cap[1,Y]$. Let
\begin{align}
 S(b_1,\ldots,b_s)&=
 \sum_{X\le n<2X}\mathbf 1_{b_1}(n)\cdots\mathbf 1_{b_s}(n)\notag\\
 &=\left|\left\{X\le n<2X:
 \gcd\!\left(\prod_{i=1}^s(n+T-b_i),P(z)\right)=1\right\}\right|.
\end{align}
Then
\begin{equation}\label{eq:U}
 \sum_{X\le n<2X}U(n)^q\ll_q\sum_{s=1}^{q}\sum_{\substack{b_1<\cdots<b_s\\b_i\in \mathcal A\cap[1,Y]}}S(b_1,\ldots,b_s).
\end{equation}
Next, we will obtain an upper bound for $S(b_1,\ldots,b_s)$.

For each prime $p$, let
\[
 I_p=\{n\in\{0,\ldots,p-1\}:n+T-b_i\equiv0\pmod p\text{ for some }1\le i\le s\},
 \qquad
 \nu_p=|I_p|.
\]
If $\nu_p=p$ for some $p\le z$, then $S(b_1,\ldots,b_s)=0$. If $\nu_p<p$ for all primes $p\le z$,
then $\nu_p\le \min\{s,p-1\}$. By the upper bound sieve for residue classes (see Ford \cite[Exercise 2.2]{FordSieve}), we have
\begin{equation}\label{eq:sieveV}
 S(b_1,\ldots,b_s)
 \ll_s
 X V(z),
 \qquad
 V(z)=\prod_{p\le z}
 \left(1-\frac{\nu_p}{p}\right).
\end{equation}
Write
\[
V(z)=\prod_{p\le z}\left(1-\frac1p\right)^s
 \prod_{p\le z}
 \left(1-\frac{\nu_p}{p}\right)
 \left(1-\frac1p\right)^{-s}.
 \]

If $s=1$, then $\nu_p=1$ and so
\[
S(b_1)\ll_{\rho} XV(z)=X\prod_{p\le z}
 \left(1-\frac{1}{p}\right)\ll_{s,\rho}\frac{X}{\log X}.
\]

Now we assume that $s\ge 2$. If $p>2s$ and $p\nmid b_i-b_j$ for any $1\le i<j\le s$, then $n+T-b_1,\ldots,n+T-b_s$ are pairwise distinct modulo $p$, and so
$\nu_p=s$. It follows from Bernoulli's inequality that
\[
 1-\frac{\nu_p}{p}
 =1-\frac{s}{p}
 \le \left(1-\frac1p\right)^s.
\]
Therefore
\[
 \left(1-\frac{\nu_p}{p}\right)
 \left(1-\frac1p\right)^{-s}
 \le1.
\]
If $p>2s$ and $p\mid b_i-b_j$ for some $1\le i<j\le s$, then
\[
 \left(1-\frac{\nu_p}{p}\right)
 \left(1-\frac1p\right)^{-s}
 \le
 \left(1-\frac1p\right)^{-s}
 \le 1+\frac{C_s}{p},
\]
where $C_s>0$ depends only on $s$. For the finitely many primes
$p\le2s$, since $\nu_p<p$, we also have
\[
 \prod_{p\le2s}
 \left(1-\frac{\nu_p}{p}\right)
 \left(1-\frac1p\right)^{-s}
 \ll_s1.
\]
It follows that
\[
\begin{aligned}
 V(z)
 &=
 \prod_{p\le z}\left(1-\frac1p\right)^s
 \prod_{p\le z}
 \left(1-\frac{\nu_p}{p}\right)
 \left(1-\frac1p\right)^{-s} \\
 &\ll_s
 \prod_{p\le z}\left(1-\frac1p\right)^s
 \prod_{1\le i<j\le s}
 \prod_{\substack{p\mid b_i-b_j\\p>2}}
 \left(1+\frac{C_s}{p}\right).
\end{aligned}
\]
By Mertens' theorem and $z=X^\rho$,
\[
 \prod_{p\le z}\left(1-\frac1p\right)^s
 \ll_s\frac1{(\log z)^s}
 \ll_{s,\rho}\frac1{(\log X)^s}.
\]
Then
\[
 V(z)\ll_{s,\rho}\frac1{(\log X)^s}
 \prod_{1\le i<j\le s}
 \prod_{\substack{p\mid b_i-b_j\\p>2}}
 \left(1+\frac{C_s}{p}\right).
\]
Following Erd\H{o}s' idea \cite[Theorem 2]{Erdos}, and applying the inequality of the geometric and arithmetic means
\[
 x_1x_2\cdots x_m \le \frac{x_1^m+\cdots+x_m^m}{m},
\]
we have
\[
 \prod_{1\le i<j\le s}
 \prod_{\substack{p\mid b_i-b_j\\p>2}}
 \left(1+\frac{C_s}{p}\right)
 \le \frac{1}{\binom{s}{2}} \sum_{1\le i<j\le s}
 \prod_{\substack{p\mid b_i-b_j\\p>2}}
 \left(1+\frac{C_s}{p}\right)^{\binom{s}{2}}.
\]
It is clear that there exists a positive integer $v=v(s)$ such that $v^3\ge(1+C_s)^{\binom{s}{2}}-1$ and
\[
\left(1+\frac{C_s}{p}\right)^{\binom{s}{2}}=1+\sum_{k=1}^{\binom{s}{2}}\binom{\binom{s}{2}}{k}\frac{C_s^k}{p^k}\le 1+\frac{(1+C_s)^{\binom{s}{2}}-1}{p}\le1+\frac{v^3}{p}.
\]
Hence,
\begin{equation}\label{eq:Vz-direct}
 V(z)\ll_{s,\rho}\frac1{(\log X)^s}
 \sum_{1\le i<j\le s}
 \prod_{\substack{p\mid b_i-b_j\\p>2}}
 \left(1+\frac{v^3}{p}\right).
\end{equation}
Combining \eqref{eq:sieveV} and \eqref{eq:Vz-direct}, we obtain
\begin{equation}\label{eq:tupleupper}
 S(b_1,\ldots,b_s)
 \ll_{s,\rho}\frac{X}{(\log X)^s}
 \sum_{1\le i<j\le s}
 \prod_{\substack{p\mid b_i-b_j\\p>2}}
 \left(1+\frac{v^3}{p}\right).
\end{equation}
It follows from \eqref{eq:U} that
\begin{align*}
\sum_{X\le n<2X}U(n)^q
&\ll_q \sum_{b_1\in \mathcal A\cap[1,Y]}S(b_1)+\sum_{s=2}^{q}\sum_{\substack{b_1<\cdots<b_s\\b_i\in \mathcal A\cap[1,Y]}}S(b_1,\ldots,b_s)\\
&\ll_{q,\rho}\frac{XM}{\log X}+\sum_{s=2}^{q}\frac{X}{(\log X)^s}\sum_{1\le i<j\le s}\sum_{\substack{b_1<\cdots<b_s\\b_i\in \mathcal A\cap[1,Y]}}
 \prod_{\substack{p\mid b_i-b_j\\p>2}}
 \left(1+\frac{v^3}{p}\right)\\
&\ll_{q,\rho}\frac{XM}{\log X}+\sum_{s=2}^{q}\frac{XM^{s-2}}{(\log X)^s}\sum_{\substack{b<b'\\b,b'\in \mathcal A\cap[1,Y]}}
 \prod_{\substack{p\mid b-b'\\p>2}}
 \left(1+\frac{v^3}{p}\right).
\end{align*}
Since $1\le b'-b<Y$, it follows that the number of distinct prime divisors of
$b'-b$ is at most $\log Y/\log2$, and so
\[
 \prod_{\substack{p\mid b-b'\\p>\log Y}}
 \left(1+\frac{v^3}{p}\right)
 \le
 \left(1+\frac{v^3}{\log Y}\right)^{\log Y/\log2}
 \ll_v1.
\]
By Lemma \ref{lem:pair}, we have
\[
\sum_{X\le n<2X}U(n)^q
\ll_{q,\rho}\frac{XM}{\log X}+\sum_{s=2}^{q}\frac{XM^{s-2}(\log Y)^{2}}{(\log X)^s}.
\]
By Lemma \ref{lem:Acount} and $\sigma=1$, we have $M\ll \log Y\le \log X$. Hence,
\[
\sum_{X\le n<2X}U(n)^q
\ll_{q,\rho}X.
\]
This completes the proof of Lemma \ref{lem:rough}.

<!-- lemma-4: body_md -->
\label{lem:cluster}
For every positive integer $m$, there exist a positive integer $k$
and positive constants $\rho_0,c,X_0$, all depending only on $m$,
such that at least $cX/(\log X)^k$ integers $u\in[X,2X)$ satisfy
\[
 |\{i:1\le i\le k,\ u+h_i\in\mathcal P\}|\ge m
\]
and
\[
 \gcd(u+h_i,P(X^{\rho_0}))=1\qquad(1\le i\le k),
\]
where $X$ is any integer with $X\ge X_0$ and
$\{h_1,\ldots,h_k\}\subseteq[1,X^{1/10}]$ is any admissible
$k$-element set of integers.

<!-- proof-2: body_md -->
This lemma follows from Maynard's result \cite[Theorem~3.1 and its proof]{Maynard}.
For completeness, we give the details of the application. Let
\[
 L_i(u)=u+h_i\qquad(1\le i\le k),\qquad
 \mathcal L=\{L_1,\ldots,L_k\}.
\]
In the notation of \cite[Hypothesis~1]{Maynard}, let
\[
 (A,\mathcal L,P,B,x,\theta)
 =\left(\mathbb N,\mathcal L,\mathcal P,1,X,\frac13\right).
\]
The admissibility of $\{h_1,\ldots,h_k\}$ implies that $\mathcal L$
is admissible. We first verify  \cite[Hypothesis~1]{Maynard}, uniformly for
$1\le h_i\le X^{1/10}$.

Since $X$ is an integer, it follows that
\[
 |\{X\le u<2X:u\equiv a\pmod q\}|=\frac Xq+O(1).
\]
Thus,
\[
 \sum_{q\le X^{1/3}}\max_{0\le a<q}
 \left||\{X\le u<2X:u\equiv a\pmod q\}|-\frac Xq\right|
 \ll X^{1/3}\ll\frac{X}{(\log X)^{100k^2}},
\]
and, for any $q\le X^{1/3}$,
\[
 |\{X\le u<2X:u\equiv a\pmod q\}|
 \le\frac Xq+1\ll\frac{X}{q}.
\]
These are parts (1) and (3) of Hypothesis~1.

For any nonnegative integer $h$, let
\[
 \pi_h=|\mathcal P\cap[X+h,2X+h)|,\qquad
 \pi_h(q,b)=|\{p\in\mathcal P\cap[X+h,2X+h):p\equiv b\pmod q\}|.
\]
By the Bombieri--Vinogradov theorem,
\[
 \sum_{q\le X^{1/3}}\max_{(b,q)=1}
 \left|\pi_0(q,b)-\frac{\pi_0}{\varphi(q)}\right|
 \ll_k\frac{X}{(\log X)^{100k^2+2}}.
\]
Clearly, $|\pi_h(q,b)-\pi_0(q,b)|\le 2h$ and $|\pi_h-\pi_0|\le 2h$. Thus, by the prime number theorem,
\begin{align*}
 \sum_{q\le X^{1/3}}\max_{(b,q)=1}
 \left|\pi_h(q,b)-\frac{\pi_h}{\varphi(q)}\right|
 &\ll_k\frac{X}{(\log X)^{100k^2+2}}+hX^{1/3}\\
 &\ll_k\frac{X}{(\log X)^{100k^2+2}}\\
 &\ll\frac{\pi_h}{(\log X)^{100k^2}},
\end{align*}
uniformly for $1\le h\le X^{1/10}$. This is part (2) of Hypothesis~1.

For $\alpha=1/10$, the coefficient and $B$ bounds in
\cite[Theorem~3.1]{Maynard} hold, since $a_i=1$,
$1\le b_i=h_i\le X^{1/10}$, and $B=1$. Take $\delta=1/2$. Then
\[
 \frac1k\sum_{i=1}^k\pi_{h_i}\ge\delta\frac{X}{\log X}
\]
for all sufficiently large $X$, which is the condition
in \cite[Theorem~3.1]{Maynard}. Let $c_1>0$ be the constant in
\cite[equation~(6.10)]{Maynard}. Choose $k=k(m)$ sufficiently large to
satisfy
\begin{equation}\label{eq:condition610}
 \delta>(\log k)^{-1},\qquad
 3c_1\delta\log k-2m\ge1.
\end{equation}
Following the proof of \cite[Theorem~3.1]{Maynard}, take
\[
 \rho=\frac{c_0}{k^3\log k},
\]
where $c_0>0$ is the sufficiently small constant chosen there.
For a sufficiently large integer $X$, we have
\[
 k\le(\log X)^{1/10},\qquad
 \frac{k(\log\log X)^2}{\log X}\le\rho\le\frac{\theta}{10}=\frac1{30}.
\]
The first inequality also implies $k\le(\log X)^{1/5}$, as required
in the proof of \cite[Theorem~3.1]{Maynard}.

Let $\lambda_u$ be the non-negative weights whose existence is guaranteed by \cite[Proposition~6.1]{Maynard} with $R=X^{\theta/3}=X^{1/9}$, and let
\[
 F(u)=\sum_{i=1}^k\mathbf1_{\mathcal P}(u+h_i)-m
       -k\sum_{i=1}^k\sum_{\substack{p\mid u+h_i\\p<X^\rho}}1,
 \qquad
 S=\sum_{X\le u<2X}F(u)\lambda_u.
\]
Then $S$ is precisely the sum in
\cite[equation~(6.2)]{Maynard}. In the proof of \cite[Theorem~3.1]{Maynard}, Maynard noted that for
any $X\le u <2X$, $F(u)$ is positive only if at least $m+1$ of the $u+h_1,\ldots,u+h_k$ are
primes in $\mathcal P$, and none of the $u+h_1,\ldots,u+h_k$ have any prime factors less than $X^\rho$. It follows from \eqref{eq:condition610} and \cite[equation~(6.10)]{Maynard} that there exists an integer $u\in[X,2X)$ such that $F(u)$ is positive. Note that $P(z)=\prod_{p\le z}p$. Take $\rho_0=\rho/2$ and so this integer $u$ satisfies
\[
 |\{i:1\le i\le k,\ u+h_i\in\mathcal P\}|\ge m
\]
and
\[
 \gcd(u+h_i,P(X^{\rho_0}))=1\qquad(1\le i\le k).
\]
Next, we will show that there are at least $cX/(\log X)^k$ integers $u\in[X,2X)$ such that $F(u)$ is positive.

Let $\mathcal G=\{u\in\mathbb Z:X\le u<2X,\ F(u)>0\}$. It follows from
\cite[equations~(6.10)--(6.11)]{Maynard} that
\[
 S\gg_m X(\log X)^k.
\]
It follows from \cite[equation~(6.3)]{Maynard} that
\[
 \lambda_u\ll_{m}(\log X)^{2k}
 \qquad(u\in\mathcal G).
\]
Since $F(u)\le k$ and $\lambda_u\ge0$,
it follows that
\[
 X(\log X)^k\ll_m S
 \le\sum_{u\in\mathcal G}F(u)\lambda_u
 \ll_m|\mathcal G|(\log X)^{2k}.
\]
Hence
\[
 |\mathcal G|\gg_m\frac{X}{(\log X)^k}.
\]
This completes
the proof of Lemma~\ref{lem:cluster}.

For every positive integer $m$, let $k$ and $\rho_0$ be given by
Lemma~\ref{lem:cluster}, and let
\[
 0<\rho<\min\{\rho_0,1/2\}.
\]
For a sufficiently large integer $X$, let
\[
 Y=X^{1/10},\qquad T=\lfloor Y\rfloor+1,\qquad
 \mathcal B=\mathcal A\cap[1,Y],\qquad W=\prod_{p\le k}p.
\]
By Lemma~\ref{lem:Acount} and $\sigma=1$, we have $|\mathcal B|\asymp\log X$. Let $\mathcal D$ be a largest subset of $\mathcal B$ such that all elements are congruent modulo $W$. Then
\begin{equation}\label{eq:Dsize}
 |\mathcal D|\ge\frac{|\mathcal B|}{W}\gg\log X.
\end{equation}
It is clear that, for every $k$-element subset $H=\{a_1,\ldots,a_k\}\subseteq\mathcal D$, the set
$\{T-a_1,\ldots,T-a_k\}$ is admissible, and
\[
 1\le T-a_i\le Y=X^{1/10}.
\]

For any integer $X\le u<2X$, let $N(u)$ be the number of $k$-element subsets
$H\subseteq\mathcal D$ satisfying
\[
 |\{a\in H:u+T-a\in\mathcal P\}|\ge m
\]
and
\[
 \gcd(u+T-a,P(X^\rho))=1\qquad(a\in H).
\]
Since $\rho<\rho_0$, it follows from Lemma~\ref{lem:cluster} and \eqref{eq:Dsize} that
\begin{equation}\label{eq:Nfirst}
 \sum_{X\le u<2X}N(u)
 \gg\binom{|\mathcal D|}{k}\frac{X}{(\log X)^k}
 \gg X.
\end{equation}
Let
\[
 U(u)=|\{a\in\mathcal B:\gcd(u+T-a,P(X^\rho))=1\}|.
\]
Then
\[
 N(u)\le\binom{U(u)}k\le U(u)^k.
\]
By Lemma~\ref{lem:rough} with $q=2k$,
\begin{equation}\label{eq:Nsecond}
 \sum_{X\le u<2X}N(u)^2
 \le\sum_{X\le u<2X}U(u)^{2k}\ll X.
\end{equation}
By \eqref{eq:Nfirst}, \eqref{eq:Nsecond} and the Cauchy--Schwarz inequality,
\[
 |\{X\le u<2X:N(u)>0\}|
 \ge\frac{\left(\sum_{X\le u<2X}N(u)\right)^2}
            {\sum_{X\le u<2X}N(u)^2}
 \gg X.
\]
If $N(u)>0$, then $r(u+T)\ge m$. Hence,
\[
 |\{n\le3X:r(n)\ge m\}|\ge |\{X\le u<2X:r(u+T)\ge m\}|\gg X.
\]
This completes the proof of Theorem~\ref{thm:main}.
