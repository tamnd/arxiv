<!-- definition-1: body_md -->
A function $\psi\in\mathcal{PD}$ is said to have the \emph{fixed-time extinction property} if there exists $M\in\mathbb{Z}_{\geq1}$ such that the function
 
\begin{equation}
    h_{\psi}(s):=
    \max_{0\leq r\leq s}g(r),~~s\geq0,~~\text{with}~~g(r):=[r-\psi(r)]_{+},
    \label{eq:general_h}
\end{equation}
 
satisfies $h_{\psi}^M(s)=0$ for all $s\geq0$. In this case, we say that $\psi\in\mathcal{E}^M_{\text{FxT}}$. \hfill\ensuremath{\square}
<!-- end definition-1 -->
<!-- lemma-1: body_md -->
\label{lem:general_fxt_extinction}
 Let $\psi\in\mathcal{PD}$. Then there exists
$M\in\mathbb Z_{\geq1}$ such that
$\psi\in\mathcal E_{\mathrm{FxT}}^M$ if and only if there exist $\varepsilon>0$ and $H<\infty$ such that: (a) $\psi(s)\geq s$ for all $0\leq s\leq\varepsilon$; (b) $\psi(s)\geq s-H$ for all $s\geq0$. Moreover, if $\psi\in\mathcal E_{\mathrm{FxT}}^M$
for some $M\in\mathbb Z_{\geq1}$, then
$h_\psi(s)<s$ for every $s>0$.
\hfill $\square$

<!-- definition-2: body_md -->
\label{def:fxISS}
Let $\mathcal{A}\subset\mathbb{R}^n$ be a nonempty closed set. System \eqref{eq:dynamics} is said to render the set $\mathcal{A}$ fixed-time input-to-state stable (ISS) if there exists a class $\mathcal{K}\mathcal{L}_{\text{FxT}}$ function $\beta$ and a class $\mathcal{K}$ function $\gamma(\cdot)$, such that for all $x_0\in\mathbb{R}^n$ and all inputs $u\in\ell^{\infty}$, all solutions of \eqref{eq:dynamics} satisfy the following inequality
 
\begin{equation}\label{eq:general_fxtiss_estimate}
    |x(j)|_{\mathcal A}\leq\max\left\{\beta(|x(0)|_{\mathcal A},j),\gamma\left(|u|_{[0,j-1],\infty}\right)\right\},
\end{equation}
 
for all $j\geq0$, where $|u|_{[0,j-1],\infty}=0$ for $j=0$.  \hfill $\square$
<!-- end definition-2 -->
<!-- definition-3: body_md -->
\label{def:general_fxtiss_lyap}
A continuous function $V:\mathbb R^n\to\mathbb R_{\geq0}$ is called an \emph{Extinction-based FxT-ISS
Lyapunov function} with respect to the nonempty closed set $\mathcal A\subset\mathbb{R}^n$ if there exist $\alpha_1,\alpha_2\in\mathcal K_\infty$, $\rho\in\mathcal K_\infty$, and $\psi\in\mathcal{E}^M_{\text{FxT}}$, such that:
 
\begin{subequations}\label{eq:suffLyap1}
\begin{align}   
    \alpha_1(|x|_{\mathcal A})&\leq V(x)\leq \alpha_2(|x|_{\mathcal A}),
    \qquad \forall x\in\mathbb R^n, \label{eq:general_lyap_bounds}\\
    V(x^+)-V(x)&\leq-\min\{\psi(V(x)),V(x)\}+\rho(|u|)\label{eq:additive_dissipation}
\end{align}
\end{subequations}
 
for every $(x,u)\in\mathbb R^n\times\mathbb R^m$. \hfill $\square$
<!-- end definition-3 -->
<!-- proposition-1: body_md -->
\label{prop:implication_to_dissipation}
Consider the discrete-time system \eqref{eq:dynamics}. Suppose that 
Assumption \ref{assumption1} holds and that $\mathcal{A}$ is compact.  Let $V:\mathbb{R}^n\to\mathbb{R}_{\geq0}$ be continuous and satisfy conditions \eqref{eq:general_lyap_bounds} with $\alpha_1,\alpha_2\in\mathcal{K}_{\infty}$, and suppose there exist $\phi\in\mathcal{K}_{\infty}$ and  $\psi\in\mathcal{E}_{FxT}^M$ such that for all $(x,u)\in\mathbb{R}^n\times\mathbb{R}^m$: 
 
\begin{align}  \label{eq:prop_implication}
V(x)\geq&~\phi(|u|)\implies \notag\\
&V(x^+)-V(x)\leq-\min\{\psi(V(x)),V(x)\}
\end{align}
 
Then, $V$ is an Extinction-Based FxT-ISS Lyapunov function for system \eqref{eq:dynamics}.  \hfill $\hfill\ensuremath{\square}$
<!-- end proposition-1 -->
<!-- example-1: body_md -->
\label{ex:disturbed_newton}
 
Consider the quadratic function $f(x)=\frac{1}{2}x^\top Qx+b^\top x+c$, where $Q=Q^\top\succ 0$. Its unique minimizer is $x^\star=-Q^{-1}b$. Consider the following perturbed Newton optimization algorithm:
 
\begin{equation}
    x^+ =x-\bigl[\nabla^2 f(x)\bigr]^{-1}
        \nabla f(x)+u,
    \label{eq:disturbed_newton}
\end{equation}
 
where $u\in\mathbb R^n$ represents an additive disturbance. For this system, the function $V(x):=\|x-x^\star\|$ is a FxT-ISS Lyapunov function with respect to $\mathcal A=\{x^\star\}$. Indeed, since $\nabla f(x)=Qx+b$ and $\nabla^2 f(x)=Q,$ the iteration \eqref{eq:disturbed_newton}
reduces to $x^+=x-Q^{-1}(Qx+b)+u=-Q^{-1}b+u=x^\star+u$. Consequently, the error $e:=x-x^\star$ satisfies $e^+=u$. By construction, $|x|_{\mathcal A}=V(x)=\|x-x^\star\|$. Thus, condition \eqref{eq:general_lyap_bounds} holds with $\alpha_1(s)=\alpha_2(s)=s$. Moreover, $ V(x^+)=|u|$, and we get $V(x^+)-V(x)= -V(x)+|u|= -\min\{\psi(V(x)),V(x)\}+\rho(|u|)$, where $\psi(s)=s$ and $ \rho(s)=s$.  Thus, $V$ is a Extinction-Based FxT-ISS Lyapunov function for \eqref{eq:disturbed_newton}.  However, the following lemma shows that \eqref{eq:disturbed_newton} does not admit a function $V$ that satisfies the conditions of Proposition \ref{prop:implication_to_dissipation}.  
 
\begin{lemma}\label{prop:newton_implication_impossibility}
Let $Q=Q^\top\succ0$. Then, for the perturbed Newton iteration \eqref{eq:disturbed_newton} there is no function
$V:\mathbb R^n\to\mathbb R_{\geq0}$
(not necessarily continuous), functions $\alpha_1,\alpha_2,\phi\in\mathcal K_\infty$,
and a function $\psi\in\mathcal E_{\mathrm{FxT}}^M$, for any finite $M\geq1$, such that inequalities \eqref{eq:general_lyap_bounds} and \eqref{eq:prop_implication} hold with $\mathcal{A}=\{x^*\}$ for every $(x,u)\in\mathbb R^n\times\mathbb R^n$. \hfill\ensuremath{\square} 
\end{lemma}

<!-- proposition-2: body_md -->
\label{prop:implication_fxt_iss}
Let $\mathcal{A}$ be a nonempty closed set, and consider the discrete-time system \eqref{eq:dynamics}. Let $V:\mathbb R^n\to\mathbb R_{\geq 0}$
be continuous and satisfy \eqref{eq:general_lyap_bounds}
with $\alpha_1,\alpha_2\in\mathcal K_\infty$.
Suppose there exist
$\phi,\kappa\in\mathcal K_\infty$ and
$\psi\in\mathcal E_{\mathrm{FxT}}^M$
such that, for all
$(x,u)\in\mathbb R^n\times\mathbb R^m$,
 
\begin{subequations}
\label{eq:implication_conditions}
\begin{align}
V(x)\geq&~\phi(|u|)
\implies\notag\\
&V(x^+)-V(x)
\leq-\min\{\psi(V(x)),V(x)\},\label{eq:implication_condition}\\
V(x)\leq&~\phi(|u|)
\implies
V(x^+)\leq\kappa(|u|).
\label{eq:subthreshold_condition}
\end{align}
\end{subequations}
Then $V$ is an Extinction-Based FxT-ISS Lyapunov function
for system \eqref{eq:dynamics}. \hfill $\square$

<!-- theorem-1: body_md -->
\label{thm:general_fxtiss}
 
Let $\mathcal{A}\subset\mathbb{R}^n$ be a nonempty closed set, and suppose that $V$ is a FxT-ISS Lyapunov function for system  \eqref{eq:dynamics}. Then, $\mathcal{A}$ is FxT-ISS for system \eqref{eq:dynamics}. \hfill\ensuremath{\square}
<!-- end theorem-1 -->
<!-- remark-1: body_md -->
By Theorem \ref{thm:general_fxtiss}, the perturbed Newton-method \eqref{eq:disturbed_newton} is FxT-ISS.  While the fixed-time (one-step) convergence property of this method is trivial to prove, the uniform $\mathcal{KL}_{\text{FxT}}$ bound \eqref{eq:general_fxtiss_estimate} provides an additional stability characterization and quantifies the effect of $u$ on the dynamics, uniformly across initial conditions and inputs. Moreover, as shown in the next sections, it opens the door for the study of FxT-ISS in other algorithms, as well as interconnected systems via small-gain arguments.
\hfill\ensuremath{\square}

<!-- assumption-2: body_md -->
\label{ass:sg_subsystems}
For every $i\in\{1,2\}$ and $j\neq i$, there exist a continuous
function $V_i:\mathbb R^{n_i}\to\mathbb R_{\geq0}$, functions $\underline{\alpha}_i,\overline{\alpha}_i,
\sigma_i,\chi_i\in\mathcal K_\infty$, a function $\psi_i\in\mathcal E_{\mathrm{FxT}}^{M_i}$, and a continuous, nonnegative, nondecreasing function $\gamma_{ij}:\mathbb R_{\geq0}\to\mathbb R_{\geq0}$, with $\gamma_{ij}(0)=0$, such that
 
\begin{equation}\label{eq:sg_subsystem_bounds}
\underline{\alpha}_i(|x_i|)
\leq V_i(x_i)
\leq\overline{\alpha}_i(|x_i|),~~~\forall~x_i\in\mathbb{R}^{n_i},
\end{equation}
 
and
 
\begin{align}
&V_i\!\left(x_i^+\right)-V_i(x_i)
\notag\\
&\quad\leq
\max\Big\{
-\min\{\psi_i(V_i(x_i)),V_i(x_i)\},
\gamma_{ij}(V_j(x_j))\notag\\
&~~~~~~~~~~~~~~~-V_i(x_i)\Big\}+\chi_i(|u_i|),
\label{eq:sg_subsystem_gain}
\end{align}
 
for all $(x_i,x_j,u_i)$.
\hfill $\square$
<!-- end assumption-2 -->
<!-- remark-2: body_md -->
Note that adding $V_i(x_i)$ to both sides of \eqref{eq:sg_subsystem_gain} and using
$s-\min\{\psi_i(s),s\}=[s-\psi_i(s)]_+$ gives the equivalent inequality
\begin{align}
V_i\!\left(x_i^+\right)
\leq
\max\Big\{
&[V_i(x_i)-\psi_i(V_i(x_i))]_+,
\gamma_{ij}(V_j(x_j))
\Big\}\notag\\
&+\chi_i(|u_i|).
\label{eq:sg_subsystem_max_form}
\end{align}
 
In this sense, Assumption \ref{ass:sg_subsystems} is similar in nature to \cite[Assumption 1]{tang2026smallgain} in the context of FxT-ISS for interconnected continuous-time systems. \hfill $\square$

\label{ass:sg_fxt_condition}
For every $i\in\{1,2\}$ and $j\neq i$, the scaled internal
gain $b_{ij}$ is globally bounded and vanishes on a neighborhood of the origin. \hfill\ensuremath{\square}

<!-- theorem-2: body_md -->
\label{thm:gain_fxt_small_gain}
Suppose that Assumptions~\ref{ass:sg_subsystems}
and~\ref{ass:sg_fxt_condition} hold and that, for every
$i\in\{1,2\}$ and $j\neq i$,
 
\begin{equation}\label{eq:sg_internal_gain_condition}
b_{ij}(s)<s,
\qquad
\forall s>0.
\end{equation}
 
Then the interconnected system \eqref{eq:sg_interconnection} renders the set $\mathcal{A}=\{0\}$ FxT-ISS with respect to the input $u=(u_1,u_2)$.\hfill $\hfill\ensuremath{\square}$
<!-- end theorem-2 -->
<!-- remark-3: body_md -->
Condition \eqref{eq:sg_internal_gain_condition} implies
\[
b_{12}\circ b_{21}(s)<s,
\qquad
b_{21}\circ b_{12}(s)<s,
\qquad s>0,
\]
which is the usual cyclic small-gain condition. The additional boundedness and dead-zone requirements are specific to the discrete-time fixed-time setting, and they ensure exact extinction after a uniformly bounded number of iterations rather than merely asymptotic convergence. \hfill $\square$
