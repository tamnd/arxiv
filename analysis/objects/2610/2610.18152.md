<!-- conj-1: body_md -->
\label{conj:katona}
For every $n\ge3$, $B_n$ admits an AHC.
<!-- end conj-1 -->
<!-- theorem-1: body_md -->
\label{thm:intro-boolean}
Let \(P\) be a finite poset with $|P|\ge 2$.
\begin{enumerate}[label=(\roman*), font=\upshape]
\item If $P$ admits an AHP, then $P\times B_r$ admits an AHC for every $r\ge2$.
\item If $P$ admits an AHC, then $P\times B_r$ admits an AHP for every $r\ge0$.
\end{enumerate}
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{coro:katona}
Conjecture~\ref{conj:katona} is true.
<!-- end corollary-1 -->
<!-- theorem-2: body_md -->
\label{thm:intro-chains}
Let $P$ be a finite poset.
\begin{enumerate}[label=(\roman*), font=\upshape]
\item If $P$ admits an AHP, then $P\times C_m$ admits an AHP for every $m\ne1$.
\item If $P$ admits an AHP, then $P\times C_m$ admits an AHC for every odd $m\ge3$.
\item If $P$ admits an AHC, then $P\times C_m$ admits an AHC for every $m\ne1$.
\item If $P$ admits an AHC, then $P\times C_m$ admits an AHP for every $m\ge0$.
\end{enumerate}
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm:intro-products}
For a positive integer $n$, let $D_n$ be the divisor lattice. Then 
\begin{enumerate}[label=(\roman*), font=\upshape]
\item $D_n$ admits an AHP.
\item $D_n$ admits an AHC if and only if $|D_n|$ is even, $D_n\ncong C_1$ and $D_n\ncong B_2$.
\end{enumerate}
<!-- end theorem-3 -->
<!-- lemma-1: body_md -->
\label{lem:basic-properties}
Let $P$ be a finite poset.
\begin{enumerate}[label=(\roman*), font=\upshape]
\item If $P$ admits an AHC, then it admits an AHP.
\item If $P$ admits an AHC, then $|P|$ is even and $|P|\ge4$.
\item $P$ admits an AHP (resp., AHC) if and only if $P^*$ admits an AHP (resp., AHC).
\item If $P$ admits an AHP (resp., AHC), then so does every refinement of $P$.
                                                                                                             
\end{enumerate}

If $P$ admits an AHP, then $P\times B_2$ admits an AHC.

If $|P|=2$, then $P\cong B_1$ is a two-element chain and
$P\times B_2\cong B_3$, which admits the AHC
\[
\varnothing<\{1,2,3\}>\{1\}<\{1,2\}>\{2\}
<\{2,3\}>\{3\}<\{1,3\}>\varnothing.
\]
We now prove the case $|P|\ge3$. 
First, we suppose that the AHP in $P$ is of the form
\begin{equation}\label{x1-xp}
x_1<x_2>x_3<x_4>\cdots.
\end{equation}

\smallskip
\noindent\textbf{Step 1. Construct four disjoint alternating paths.}
Let $(x_2,x_3,\ldots,x_{|P|-1})$ be an alternating path of $P$ defined in~\eqref{x1-xp}. 
For each $A\in \{\varnothing, \{1\},\{2\}, \{1,2\}\}$, let
\[
I_A=(x_2,A),(x_3,A),\ldots,(x_{|P|-1},A).
\]
Clearly, $I_A$ is an alternating path in $P\times B_2$ from $(x_2,A)$ to $(x_{|P|-1},A)$.
We denote by $\overleftarrow{I_A}$ the path obtained by traversing $I_A$ in reverse order, 
$$
\overleftarrow{I_A}=(x_{|P|-1},A), (x_{|P|-2},A),\ldots,(x_{2},A).
$$
These four paths $I_{\varnothing}, I_{\{1\}},I_{\{2\}}, I_{\{1,2\}}$ are disjoint and traverse
$\{x_2,\ldots,x_{|P|-1}\}\times B_2$.
When $|P|=3$, each $I_A$ consists of a single vertex.

\smallskip
\noindent\textbf{Step 2. The case that $|P|$ is odd.}
Then~\eqref{x1-xp} becomes
$$
x_1<x_2>x_3<x_4>\cdots <x_{|P|-1}>x_{|P|}.
$$
Thus, we construct two joining paths whose endpoints lie in
$\{x_2\}\times B_2$ and whose internal vertices lie in
$\{x_1\}\times B_2$:
\begin{align}
(x_2,\varnothing)
>&(x_1,\varnothing)<(x_2,\{1\}),\label{1-x_1-x_2}\\
(x_2,\{2\})
>(x_1,\{2\})<&(x_1,\{1,2\})
>(x_1,\{1\})<(x_2,\{1,2\}).\label{2-x_1-x_2}
\end{align}
Similarly, we construct two joining paths whose endpoints lie in
$\{x_{|P|-1}\}\times B_2$ and whose internal vertices lie in
$\{x_{|P|}\}\times B_2$:
\begin{align*}
(x_{|P|-1},\varnothing)
>&(x_{|P|},\varnothing)<(x_{|P|-1},\{2\}),\\
(x_{|P|-1},\{1\})
>(x_{|P|},\{1\})<&(x_{|P|},\{1,2\})
>(x_{|P|},\{2\})<(x_{|P|-1},\{1,2\}).
\end{align*}
The four paths visit each vertex $(x_1,A)$ and $(x_{|P|},A)$ 
exactly once.

(P^\ast\times B_2)^\ast
\cong P\times B_2^\ast
\cong P\times B_2

admits an AHC.
This completes the proof.

\label{lem:b3}
Let $P$ be a finite poset with $|P|\ge 2$.
If $P$ admits an AHP, then $P\times B_3$ admits an AHC.
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
If $|P|=2$, then $P\cong B_1$ is a two-element chain and $P\times B_3\cong B_4\cong B_2\times B_2$.
Since $B_2$ admits an AHP, $B_4$ admits an AHC by Lemma~\ref{lem:b2}.

Suppose that $|P|\ge3$. 
As in the proof of Lemma~\ref{lem:b2}, we may assume that the AHP in $P$ is of the form
\[
x_1<x_2>x_3<x_4>\cdots.
\]

\smallskip
\noindent\textbf{Step 1. Construct eight disjoint alternating paths.}
For each 
$$A\in\{\varnothing, \{1\},\{2\}, \{3\}, \{1,2\}, \{1,3\},\{2,3\},\{1,2,3\}\},$$ let
\[
I_A=(x_2,A),(x_3,A),\ldots,(x_{|P|-1},A).
\]
These eight paths are disjoint and traverse $\{x_2,\ldots,x_{|P|-1}\}\times B_3$.

\smallskip
\noindent\textbf{Step 2. The case that $|P|$ is odd.}
We construct four joining paths whose endpoints lie in
$\{x_2\}\times B_3$ and whose internal vertices lie in
$\{x_1\}\times B_3$:
\begin{align}
(x_2,\varnothing)>&(x_1,\varnothing)<(x_2,\{1\}),\label{3-x_1-x_2}\\
(x_2,\{3\})>(x_1,\{3\})<&(x_1,\{1,3\})>(x_1,\{1\})<(x_2,\{1,3\}), \label{4-x_1-x_2}\\
(x_2,\{2\})>&(x_1,\{2\})<(x_2,\{2,3\}),\label{5-x_1-x_2}\\
(x_2,\{1,2\})>(x_1,\{1,2\})<&(x_1,\{1,2,3\})>(x_1,\{2,3\})<(x_2,\{1,2,3\}).\label{6-x_1-x_2}
\end{align}
Similarly, we construct four joining paths whose endpoints lie in
$\{x_{|P|-1}\}\times B_3$ and whose internal vertices lie in
$\{x_{|P|}\}\times B_3$:
\begin{align*}
(x_{|P|-1},\varnothing)
>&(x_{|P|},\varnothing)<(x_{|P|-1},\{2\}),\\
(x_{|P|-1},\{1\})
>(x_{|P|},\{1\})<&(x_{|P|},\{1,2\})
 >(x_{|P|},\{2\})<(x_{|P|-1},\{1,2\}),\\
(x_{|P|-1},\{3\})
>&(x_{|P|},\{3\})<(x_{|P|-1},\{2,3\}),\\
(x_{|P|-1},\{1,3\})
>(x_{|P|},\{1,3\})<&(x_{|P|},\{1,2,3\})
 >(x_{|P|},\{2,3\})<(x_{|P|-1},\{1,2,3\}).
\end{align*}
These eight joining paths visit every vertex $(x_1,A)$ and $(x_{|P|},A)$ exactly once as an internal vertex.

\smallskip
\noindent\textbf{Step 3. The case that $|P|$ is even.}
There are four alternating paths~\eqref{3-x_1-x_2},~\eqref{4-x_1-x_2},~\eqref{5-x_1-x_2} and~\eqref{6-x_1-x_2} in $\{x_1,x_2\}\times B_3$, the same as those when $|P|$ is odd.
For $(x_{|P|-1},A)$ and $(x_{|P|},A)$,
there are four alternating paths
\begin{align*}
(x_{|P|-1},\{1,2,3\})
<&(x_{|P|},\{1,2,3\})>(x_{|P|-1},\{1,3\}),\\
(x_{|P|-1},\{2,3\})
<(x_{|P|},\{2,3\})>&(x_{|P|},\{3\})
<(x_{|P|},\{1,3\})>(x_{|P|-1},\{3\}),\\
(x_{|P|-1},\{1,2\})
<&(x_{|P|},\{1,2\})>(x_{|P|-1},\{1\}),\\
(x_{|P|-1},\{2\})
<(x_{|P|},\{2\})>&(x_{|P|},\varnothing)
<(x_{|P|},\{1\})>(x_{|P|-1},\varnothing).
\end{align*}

\smallskip
\noindent\textbf{Step 4. This construction forms an AHC.}
The eight paths and the joining paths visit all vertices of $P\times B_3$.
When $|P|$ is odd, 
\begin{align*}
(x_1,\varnothing)
&\rightarrow I_\varnothing
 \rightarrow(x_{|P|},\varnothing)
 \rightarrow\overleftarrow{I_{\{2\}}}
 \rightarrow(x_1,\{2\})
 \rightarrow I_{\{2,3\}}\\
&\rightarrow(x_{|P|},\{3\})
 \rightarrow\overleftarrow{I_{\{3\}}}
 \rightarrow(x_1,\{3\})
 <(x_1,\{1,3\})>(x_1,\{1\})
 \rightarrow I_{\{1,3\}}\\
&\rightarrow(x_{|P|},\{1,3\})
 <(x_{|P|},\{1,2,3\})>(x_{|P|},\{2,3\})
 \rightarrow\overleftarrow{I_{\{1,2,3\}}}\\
&\rightarrow(x_1,\{2,3\})
 <(x_1,\{1,2,3\})>(x_1,\{1,2\})
 \rightarrow I_{\{1,2\}}\\
&\rightarrow(x_{|P|},\{2\})
 <(x_{|P|},\{1,2\})>(x_{|P|},\{1\})
 \rightarrow\overleftarrow{I_{\{1\}}}
 \rightarrow(x_1,\varnothing)
\end{align*}
forms an AHC.
Similarly, the connections in Step 3 give an AHC
when $|P|$ is even.

<!-- lemma-4: body_md -->
\label{lem:b1}
Let $P$ be a finite poset.
If $P$ admits an AHC, then $P\times B_1$ admits an AHP.
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
By Lemma~\ref{lem:basic-properties} (ii), $|P|$ is even.
Suppose that the AHC in $P$ is of the form
\[
x_1<x_2>x_3<\cdots<x_{|P|}>x_1.
\]


Identify $B_1$ with the chain $0<1$.
For each $\varepsilon\in\{0,1\}$, define the alternating path
\[
J_\varepsilon=
(x_2,\varepsilon),(x_3,\varepsilon),\ldots,
(x_{|P|},\varepsilon),(x_1,\varepsilon).
\]
These two paths $J_0$ and $J_1$ are disjoint and traverse $P\times B_1$.
Since $x_1<x_2$ and $0<1$, we have $(x_1,0)<(x_2,1)$.
Thus $J_0\rightarrow J_1$ is the desired AHP.
<!-- end proof-3 -->
<!-- proof-4: body_md -->
(i). We proceed by induction on $r$.
By Lemmas~\ref{lem:b2} and~\ref{lem:b3},
we have shown that if $P$ admits an AHP, then $P\times B_r$ admits an AHC for $r=2, 3$.
	Let $r\ge4$ and assume that the result holds for $r-2$.
	Then $P\times B_{r-2}$ admits an AHC.
By Lemma~\ref{lem:basic-properties} (i), $P\times B_{r-2}$ admits an AHP.
	Applying Lemma~\ref{lem:b2} to $P\times B_{r-2}$, we obtain an AHC in
	\[
	(P\times B_{r-2})\times B_2
	\cong P\times B_r.
	\]

(ii). 
Suppose that $P$ admits an AHC. 
Then  by Lemma~\ref{lem:basic-properties} (i), $P$ admits an AHP. 
It is trivial that for $r=0$, the product $P\times B_0\cong P$ admits an AHP.
By Lemma~\ref{lem:b1}, the case $r=1$ holds, that is,
$P\times B_1$ admits an AHP.
For $r\ge2$, since $P$ admits an AHP,
by~(i), $P\times B_r$ admits an AHC, and hence admits an AHP.

	For example, $P=B_1$ admits an AHP, but $P\times B_1\cong B_2$ does not admit an AHC.

\label{B-AHP}

<!-- end corollary-2 -->
<!-- proof-5: body_md -->
It is trivial that the singleton $B_0$ and the chain $B_1$ admit AHPs.
Clearly, $B_2$ admits an AHP of the form

For $n\ge3$, Corollary~\ref{coro:katona} and Lemma~\ref{lem:basic-properties} (i) give an AHP in $B_n$.

<!-- lemma-5: body_md -->
\label{lem:c3}
Let $P$ be a finite poset.
If $P$ admits an AHP, then $P\times C_3$ admits an AHC.
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
We may assume that $|P|\ge2$, since $C_3$ admits an AHC.
By Lemma~\ref{lem:b2}, $P\times B_2$ admits an AHC.
Identify the elements $\varnothing,\{1\},\{2\},\{1,2\}$ of $B_2$
with $1,2,3,4$, respectively.
Under this identification, $C_3$ is a refinement of $B_2$.
Since $P\times C_3$ is a refinement of $P\times B_2$, Lemma~\ref{lem:basic-properties}~(iv) implies that $P\times C_3$ admits an AHC.

<!-- lemma-6: body_md -->
\label{lem:chain-extension}
Let $P$ be a finite poset and let $m\ge0$.
If $P\times C_m$ admits an AHP (resp. AHC), then $P\times C_{m+2}$ admits an AHP (resp. AHC).
<!-- end lemma-6 -->
<!-- proof-7: body_md -->
We first consider the AHP case. 
Let $\mathcal P$ be an AHP in $P\times C_m$.
If $\mathcal P$ consists of a single vertex $(x,1)$, then
$(x,1)<(x,3)>(x,2)$ is the required AHP.
We now assume that $|\mathcal P|\ge 2$.
First, we give an operation on $\mathcal P$.
            
                                              
                                                                                                                           
                                                                      
                        
\begin{itemize}
    \item For $j\ge 2$, replace every vertex $(x,j)$ in $\mathcal {P}$ by $(x,j+2)$.
    \item For $j=1$, replace $(x,1)$ in the following way:
$$
\begin{cases}
(x,1)<(x,3)>(x,2) \qquad \textrm{when}\quad  \cdots>(x,1)<\cdots \quad  \textrm{in } \, \mathcal P;\\
(x,3)>(x,1)<(x,2) \qquad \textrm{when}\quad \cdots<(x,1)>\cdots \quad \textrm{in } \,\mathcal P.
\end{cases}
$$
\end{itemize}
If $(x,1)$ is an endpoint of $\mathcal P$, use the first
replacement when $(x,1)$ is smaller than its unique neighbor,
and use the second replacement when it is larger than its
unique neighbor.
Applying the operation to every vertex of $\mathcal P$ yields a vertex sequence $\mathcal {Q}$ in $P\times C_{m+2}$.

We now show that $\mathcal Q$ is an AHP in $P\times C_{m+2}$.
Let $(x,i)$ and $(y,j)$ be consecutive in $\mathcal P$.
It suffices to show that this operation preserves the alternating order relation of  $(x,i)$ and $(y,j)$.

\smallskip
\noindent\textbf{Case 1: $i,j\ge 2$.}
We obtain $(x,i+2)$ and $(y,j+2)$ in $\mathcal Q$. 
And $(x,i+2)<(y,j+2)$ if and only if $(x,i)<(y,j)$.
The alternating order relation is preserved.

\smallskip
\noindent\textbf{Case 2: $i=j=1$.}
If $(x,1)<(y,1)$, then $x<y$ in $P$, and the operation gives
\[
(x,1)<(x,3)>(x,2)<(y,3)>(y,1)<(y,2).
\]
Similarly, if $(x,1)>(y,1)$, then $x>y$, and the operation gives
\[
(x,3)>(x,1)<(x,2)>(y,1)<(y,3)>(y,2).
\]
The alternating order relation is preserved.

\smallskip
\noindent\textbf{Case 3: $i=1,j\ge 2$ or $i\ge 2,j=1$.}
Consider $(x,1)$ and $(y,j)$, where $j\ge2$.
Then we have  $(x,1)<(y,j)$, which implies that $x\le y$.
Then 
\[
(x,1)<(x,3)>(x,2)<(y,j+2).
\]
Similarly, consider $(x,j)$ and $(y,1)$, where $j\ge2$.
Then $x\ge y$ and
\[
(x,j+2)>(y,1)<(y,3)>(y,2).
\]
In both cases, the alternating order relation is preserved.

For each $x\in P$, the replacements use each chain coordinate $1,\ldots,m+3$ exactly once.
Thus $\mathcal Q$ visits every vertex of $P\times C_{m+2}$ exactly once and is an AHP.

For the AHC case, we operate on $(x,j)$ in an AHC of $P\times C_m$ in the same way.
Similarly, we can prove that $P\times C_{m+2}$ admits an AHC.
This completes the proof.

<!-- corollary-3: body_md -->
\label{cor:product-snakes}
Let $D_n$ be the divisor lattice of a positive integer $n$.
For every integer $k\ge1$, the snake $S_k$ weakly embeds into
$D_n$ if and only if $k\le |D_n|$.
<!-- end corollary-3 -->
<!-- proof-11: body_md -->
If $S_k\hookrightarrow D_n$, then injectivity gives $k\le |D_n|$.
Conversely, assume $k\le |D_n|$.
By Theorem~\ref{thm:intro-products} (i), $D_n$ admits an AHP.
Since $D_n$ is self-dual, we obtain a weak embedding of $S_k$ into $D_n$.
This completes the proof.

<!-- proposition-1: body_md -->
\label{prop:product-snake-free}
For a divisor lattice $Q$ and a finite poset $R$, let $\mathcal A$ be a nonempty antichain of $R$.
Then the family
\[
\mathcal F=Q\times\mathcal A\subseteq Q\times R
\]
is $S_k$-free if and only if $k>|Q|$.
Moreover, for $k>|Q|$,
\[
\operatorname{La}(Q\times R,S_k)
\ge |Q|\,w(R).
\]
<!-- end proposition-1 -->
<!-- proof-12: body_md -->
For each $a\in\mathcal A$, the family $Q\times\{a\}$ is a copy of $Q$.
Since $\mathcal A$ is an antichain, elements from distinct copies are incomparable.

Consecutive elements of $S_k$ are comparable.
Hence, the image of every weak embedding of $S_k$ into $\mathcal F$ lies in a single copy $Q\times\{a\}$.
By Corollary~\ref{cor:product-snakes}, such an embedding
exists if and only if $k\le |Q|$.
Thus $\mathcal F$ is $S_k$-free if and only if $k>|Q|$.

Counting the copies of $Q$ gives $|\mathcal F|=|Q|\,|\mathcal A|$.
Taking $|\mathcal A|=w(R)$ gives the stated lower bound.
<!-- end proof-12 -->
<!-- corollary-4: body_md -->
\label{cor:divisor-snake-bound}
Let $n$ be a positive integer, and let $a,b$ be coprime positive integers with $ab\mid n$.
For every integer $k>|D_a|$, we have
\[
\operatorname{La}(D_n,S_k)\ge |D_a|\,w(D_b).
\]
<!-- end corollary-4 -->
<!-- proof-13: body_md -->
Let $a,b$ be positive integers satisfying
$ab\mid n$, $\gcd(a,b)=1$, and $|D_a|<k$.
By coprimality, we have $D_a\times D_b\cong D_{ab}$.
Moreover, $D_{ab}$ is an induced subposet of $D_n$.

Let $\mathcal A$ be a maximum antichain of $D_b$.
By Proposition~\ref{prop:product-snake-free},
$D_a\times\mathcal A$ is $S_k$-free and has
$|D_a|\,w(D_b)$ elements.
                                                     
                                   
This gives an $S_k$-free subset of $D_n$ with $|D_a|w(D_b)$ elements.
