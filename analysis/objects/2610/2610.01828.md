<!-- assump-1: body_md -->
\label{assump:Markovness}
We assume that the LM outputs a sequence $y^{m+1} = (y_0, y_1, \dots, y_m)\in \mathcal{Y}^{m+1}$ according to a Markov process with an irreducible aperiodic transition matrix $P_{|\mathcal{Y}| \times |\mathcal{Y}|}$, with stationary distribution $\pi \in \Delta_{|\mathcal{Y}|}$. We also assume that the first element of the sequence, $y_0$, is drawn from $\pi$. Thus, the probability distribution of the sequence $y^{m+1}$ is obtained as:
\begin{equation}
    \mathbb{P}_M(y^{m+1}) = Q(y^{m+1}) = \pi_{y_0} \prod_{i = 1}^{m} P_{y_{i - 1} y_{i}}.
    \label{eqn:Q_definition}
\end{equation}
<!-- end assump-1 -->
<!-- assump-2: body_md -->
\label{assump:reward_decomposition}
The reward function decomposes as the sum of the rewards of adjacent terms, i.e. 
\begin{equation}
    r (y^{m+1}) = \sum_{i = 1}^{m} r (y_{i - 1}, y_{i}) = \sum_{i = 1}^{m} r_{y_{i - 1} y_{i}},
\end{equation}
where $r_{ab}$ is a shorthand for $r(a, b): \mathcal{Y} \times \mathcal{Y} \to \mathbb{R}_+$.
<!-- end assump-2 -->

\label{thm:optimization_problem_q_solution}
Under Assumptions~\ref{assump:Markovness}~and~\ref{assump:reward_decomposition}, the solution to~\eqref{eqn:optimization_problem} is    
\begin{equation*}
    q_{\beta}(y^{m + 1}) = \frac{\displaystyle \pi_{y_0} \prod_{i=1}^m {(G_\beta)}_{y_{i - 1} y_i}}{\displaystyle \sum_{\substack{z^{m + 1} \\\in \mathcal{Y}^{m+1}}} \pi_{z_0} \prod_{i=1}^m {(G_\beta)}_{z_{i - 1} z_i}},\ y^{m + 1} \in \mathcal{Y}^{m+1},
\end{equation*}
where $\beta \ge 0$ is chosen such that 
\begin{equation}
    d_{KL}(q_{\beta}, Q) = m \delta.\label{eqn:beta_value_choice}
\end{equation}
\label{thm:q_expression}

<!-- example-1: body_md -->
\label{example:KL_alignment_zero}
Suppose that $Y$ has the p.d.f. $Q(y) = 1,\ y \in (0, 1]$. Let
\begin{equation}
    r(y) = b + \frac{n - 1}{\beta}\cdot \log(y),\  y \in (0, 1],
\end{equation}
where we recall that $\beta$ is such that $d_{KL}(q_{\beta}, Q) = \delta$.

The KL-constrained RL distribution $q_\beta$ obeys, for a suitable normalization constant $C > 0$,
\begin{align}
    q_\beta(y) & = C \cdot 1 \cdot \exp \left(\beta \left(b + \frac{n - 1}{\beta} \log(y)\right) \right)\\
    & = C \cdot \exp \left(\beta b \right) y^{n - 1} \overset{(a)}{=} n y^{n - 1} \ \forall 0 < y \le 1,
\end{align}
where $(a)$ follows since $q_\beta$ must integrate to $1$. It is also possible to show that
$
d_{KL}(q_{\beta}, Q) \le \log (n) - \frac{n - 1}{n} = \delta,
$
where the equality follows from~\eqref{eqn:delta_n_relation}. Finally, note that the best-of-$n${} distribution is simply the maximum of $n$ i.i.d. random variables, each drawn from $Q(y)$. Thus,
\begin{equation}
    q^{\text{best}}_{n, 1}(y) = n y^{n - 1} ,\ 0 < y \le 1.
\end{equation}
Clearly, $q_\beta$ equals $q^{\text{best}}_{n, 1}$, and $d_{KL}(q_\beta,q^{\text{best}}_{n, 1}) = 0$.

We solve this problem by writing a Lagrangian for suitable constants $\beta, c_a, v_b, \eta$ for $a, b \in \mathcal{Y}$, as follows:
\begin{multline}
    \mathcal{L} = \sum_{(a, b) \in \mathcal{Y} \times \mathcal{Y}} \mu_a p_{a b} r_{a b} - \frac{1}{\beta} (d_{KL}(p, P) - \delta) + \sum_{a \in \mathcal{Y}} \mu_a c_a \left(\sum_{b \in \mathcal{Y}} p_{ab}-1\right) \\ + \sum_{b \in \mathcal{Y}} v_b \left(\sum_{a \in \mathcal{Y}} \mu_a p_{ab}-\mu_b\right) + \eta\left(\sum_a\mu_a-1\right).
\end{multline}
Taking gradients with respect to $p_{ab}$ first, we obtain
\begin{equation}
    \mu_a r_{ab} - \frac{1}{\beta} \mu_{a} \left(1 + \log\left( \frac{p_{ab}}{P_{ab}}\right)\right) + \mu_a c_a + v_b \mu_a = 0,
\end{equation}
thereby yielding
\begin{equation}
    p_{ab} = C_{a} D_b P_{ab} \exp(\beta r_{ab}) = C_{a} D_b (G_{\beta})_{ab}, \quad a, b \in \mathcal{Y}, 
\end{equation}
for suitable constants $C_{a}, D_{b} \ge 0$.

Recall from Assumption~\ref{assump:Markovness} that $P$ is aperiodic and irreducible. Using the definition ${(G_\beta)}_{ab} = P_{ab} \exp(\beta r_{ab})$, it follows that $G_{\beta}$ is primitive. From standard arguments (see, for instance~\cite[Page 674, Equation~8.3.10]{meyer_linear_algebra}), it also follows that $G_{\beta}$ will have a unique Perron eigenvalue $\lambda_{\beta}$, and a unique right Perron eigenvector (up to scaling) $h \in \mathbb{R}^{|\mathcal{Y}|}$. Now, take $D_b = h_b$ and $C_a = \frac{1}{\lambda_{\beta} h_a}$ as in the theorem statement, where $h$ is defined as in the theorem statement. Such a $h$ would exist since $G$ is a non-negative matrix. 

It is left to verify all the constraints. First, we verify the row-stochasticity constraint:
\begin{equation}
    \sum_{b \in \mathcal{Y}} p_{ab} = \frac{(G_{\beta} h)_a}{\lambda_\beta h_a} = \frac{\lambda_\beta h_a}{\lambda_\beta h_a} = 1.
\end{equation}
The value of $\mu$ is chosen such that it is the left eigenvector of $p$, with the eigenvalue $1$, and such that $\sum_{a \in \mathcal{Y}} \mu_a = 1$ is satisfied. This choice of $\mu$ ensures that both these constraints are trivially satisfied. Finally, the value of $\beta$ is chosen such that $d_{KL}(p, P) = \delta$.

Next, we argue that $\mu$ is unique. Let $D_h = \textsf{diag}(h_1, h_2, \dots, h_{|\mathcal{Y}|})$. From this, note that
\begin{equation}
    p = D_h^{-1} \frac{G_{\beta}}{\lambda_{\beta}} D_h.
\end{equation}
Thus, $p$ is similar to $\frac{G_{\beta}}{\lambda_{\beta}}$. Since $\frac{G_{\beta}}{\lambda_{\beta}}$ itself has a unique left Perron vector (up to scaling) corresponding to the eigenvalue $1$, $p$, via similarity with $\frac{G_{\beta}}{\lambda_{\beta}}$, also has an unique left Perron eigenvector (say $\mu$) corresponding to the eigenvalue $1$. $\mu$ will hence be that (unique) positive vector satisfying $\mu^T p = \mu^T$, whose elements  sum up to $1$.

First, we relate $\mu$ with $g$ and $h$. Take the quantity 
\begin{equation}

Also recall that $\sum_{j \in \mathcal{Y}} g_j h_j = 1$. Now, differentiating the expression $G_{\beta} h = \lambda_{\beta} h$ with respect to $\beta$ on both sides, we get:
\begin{equation}
    G_{\beta} h' + G_{\beta}'h = \lambda_{\beta}' h + \lambda_{\beta} h',
\end{equation}
where $'$ denotes the derivative with respect to $\beta$. By left-multiplying both sides with $g^T$, we get
\begin{align}
    g^T G_{\beta} h' + g^T G_{\beta}'h &= \lambda_{\beta}' g^T h + \lambda_{\beta} g^T h'. 
\end{align}
Observing that  $g^T G_{\beta} = \lambda_\beta g^T$, we obtain
\begin{align}
    g^T G_{\beta}'h &= \lambda_{\beta}'.
\end{align}

    \Lambda_{\beta}' &= \frac{\lambda_{\beta}'}{\lambda_{\beta}} = \sum_{i, j \in \mathcal{Y}} \frac{g_i (G_{\beta})_{ij} r_{ij} h_j}{\lambda_{\beta}}.
\end{align}

\end{equation}

First, note that $t_\delta$ can be written alternatively as below:
\begin{equation}
    t_{\delta} = \max \Bigg\{
        s: \min_{\substack{\mu \in \Delta_{|\mathcal{Y}|},\\p \in \Delta_{|\mathcal{Y}| \times |\mathcal{Y}|}}} \sum_{i,j \in \mathcal{Y}} \mu_i p_{ij} \log\left(\frac{p_{ij}}{P_{ij}}\right) \leq \delta
        \text{ subject to }\mu^T p = \mu^T, \sum_{i,j \in \mathcal{Y}} \mu_i p_{ij} r_{ij} = s
        \Bigg\}.
\end{equation}

In Lemma~\ref{thm:rate_kl_value}, $\theta_i$ and $b$ are `dummy' variables and we replace them respectively by $\mu$ and $p$:
\begin{equation}
    \tilde{J} (s) = \min_{\mu \in \Delta_{|\mathcal{Y}|}, p \in \Delta_{|\mathcal{Y}| \times |\mathcal{Y}|}} \left\{\sum_{i, j \in \mathcal{Y}} \mu_i p_{ij} \log \left(\frac{p_{ij}}{P_{ij}} \right) : \sum_{i, j \in \mathcal{Y}} r_{ij} \mu_i p_{ij} = s, \quad \sum_{i \in \mathcal{Y}} \mu_i p_{ij} = \mu_j \right\}.
\end{equation}

We show the result by proving $s_\delta \le t_\delta$ and $t_\delta \le s_\delta$. First, let
\((\mu^\star,p^\star)\) be an optimal solution to the optimization problem
in~\eqref{eqn:optimization_problem_with_flow}. Then,
\begin{equation}
    s_\delta = \sum_{i,j \in \mathcal{Y}} \mu_i^\star p_{ij}^\star r_{ij},
\end{equation}
and
\begin{equation}
    \sum_{i,j \in \mathcal{Y}} \mu_i^\star p_{ij}^\star \log\left(\frac{p_{ij}^\star}{P_{ij}}\right) \leq \delta.
\end{equation}
Since \((\mu^\star,p^\star)\) is feasible for the minimization defining
\(\tilde{J}(s_\delta)\), we have
\begin{equation}
    \tilde{J}(s_\delta) \leq \sum_{i,j \in \mathcal{Y}} \mu_i^\star p_{ij}^\star \log\left(\frac{p_{ij}^\star}{P_{ij}}\right) \leq \delta.
\end{equation}
Therefore, \(s_\delta\) is feasible in the definition of \(t_\delta\), and hence $s_\delta \leq t_\delta$.
    
Conversely, let \(s\) be such that
\begin{equation}
    \tilde{J}(s) \leq \delta.
\end{equation}
By the definition of \(\tilde{J}(s)\), there exists a feasible pair
\((\mu^{(s)},p^{(s)})\) satisfying
\begin{equation}
    (\mu^{(s)})^T p^{(s)} = (\mu^{(s)})^T,
\end{equation}
\begin{equation}
    \sum_{i,j \in \mathcal{Y}} \mu_i^{(s)}p_{ij}^{(s)}r_{ij} = s,
\end{equation}
and
\begin{equation}
    \sum_{i,j \in \mathcal{Y}} \mu_i^{(s)}p_{ij}^{(s)} \log\left(\frac{p_{ij}^{(s)}}{P_{ij}}\right) \leq \delta.
\end{equation}

in~\eqref{eqn:optimization_problem_with_flow}. By the definition of
\(s_\delta\), it follows that $s \leq s_\delta$. Since this holds for every \(s\) satisfying \(\tilde{J}(s) \leq \delta\),
we obtain $t_\delta \leq s_\delta$.

Combining the two inequalities gives the desired result.

First, we consider the probability $\mathbb{P}(T_{m, \exp(m \delta)} \ge s_{\delta} + \epsilon)$ for some $\epsilon > 0$. Note from~\eqref{eqn:T_definition}, using the union bound, and using the LDP for $S_m$ from Theorem~\ref{thm:rate_kl_value} that
\begin{multline}
    \mathbb{P}(T_{m, \exp(m \delta)} \ge s_{\delta} + \epsilon) \le n \mathbb{P}(S_m \ge s_{\delta} + \epsilon) \\ \le n \exp(-m \inf_{s \ge s_{\delta} + \epsilon} \tilde{J} (s) + o(m)) = \exp(m (\delta - \inf_{s \ge s_{\delta} + \epsilon} \tilde{J} (s)) + o(m)).
\end{multline}
Recall from Lemma~\ref{lem:s_t_equality} that $s_{\delta} = \sup \{s: \tilde{J} (s) \le \delta\}$. From this, note that for $s > s_{\delta}$, we necessarily have $\tilde{J}(s) > \delta$. Thus, for $s \ge s_{\delta} + \epsilon > s_{\delta}$, we have $\tilde{J}(s) > \delta$. Thus,
\begin{equation}
    \delta - \inf_{s \ge s_{\delta} + \epsilon} \tilde{J} (s) < 0
\end{equation}
and
$\lim_{m \to \infty} \mathbb{P}(T_{m, \exp(m \delta)} \ge s_{\delta} + \epsilon) = 0$.

Next, consider $\mathbb{P}(T_{m, \exp(m \delta)} \le s_{\delta} - \epsilon)$. Note that
\begin{equation}
    \mathbb{P}(T_{m, \exp(m \delta)} \le s_{\delta} - \epsilon) = (1 - \mathbb{P}(S_m > s_{\delta} - \epsilon))^n \le \exp(- n \mathbb{P}(S_m > s_{\delta} - \epsilon)).
    \label{eqn:temp_appendix}
\end{equation}

We next argue that $\tilde{J}(s)$ is a convex function of $s$. To see this, first note that $J(\theta)$ itself is convex since the Markovian KL divergence is convex in the first argument. Next, recall that the map from $\theta$ to $S_m$ is a linear map as in~\eqref{eqn:S_m_continuous_defn}. Now, consider the function
\begin{equation}
    L(s, \theta) = \begin{cases}
    J(\theta), &\phi(\theta) = s, \\
    \infty, &\text{ otherwise.}
\end{cases}.
\end{equation}
Note that $L(s, \theta)$ is convex since $J(\theta)$ is convex. Finally, noting that $\tilde{J}(s) = \min_\theta L(s, \theta)$ gives the desired result that $\tilde{J}(s)$ is convex.

Recall that $\tilde{J}(s_\delta) \le \delta$.
\begin{enumerate}
    \par\hangindent 40\p@ If $\tilde{J}(s_\delta) < \delta$, note that $\inf_{s > s_{\delta} - \epsilon} \tilde{J}(s) \le \tilde{J}(s_\delta) = \delta - \gamma$ for some $\gamma > 0$.
    \par\hangindent 40\p@ The other case has $\tilde{J}(s_\delta) = \delta$.
    Since $\tilde{J}(s)$ is a rate function, we know that at the mean value $s_0 = \mathbb{E}[S_m]$, we have $\tilde{J}(s_0) = 0$. Now, there are two possibilities:
\begin{enumerate}
    \par\hangindent 40\p@ If $s_0 > s_\delta - \epsilon$, we have $\tilde{J}(s_0) = \delta - \gamma$ where $\gamma = \delta > 0$. Thus, we have $\inf_{s > s_{\delta} - \epsilon} \tilde{J}(s) \le \tilde{J}(s_0) = \delta - \gamma$.
    \par\hangindent 40\p@ If $s_0 \le s_\delta - \epsilon$, suppose $s_0 = s_\delta - \beta$ for some $\beta \ge \epsilon$. Now, consider an $s_\delta - \Delta \in (s_\delta - \epsilon, s_\delta)$, where $\Delta < \epsilon$. From the convexity of $\tilde{J}$ and since $0 < \frac{\Delta}{\beta}< 1$, we know that
    \begin{equation}
        \tilde{J}(s_\delta - \Delta) = \tilde{J} \left( \left(1 - \frac{\Delta}{\beta} \right)s_\delta + \frac{\Delta}{\beta}(s_0) \right) \le \left(1 - \frac{\Delta}{\beta} \right) \delta + \frac{\Delta}{\beta} \tilde{J}(s_0) < \delta.
    \end{equation}
    Suppose that $\tilde{J}(s_\delta - \Delta) = \delta - \gamma$ for some $\gamma > 0$. Thus, we have $\inf_{s > s_{\delta} - \epsilon} \tilde{J}(s) \le \tilde{J}(s_\delta - \Delta) = \delta - \gamma$.
\end{enumerate}
\end{enumerate}

In all three of the cases, we have $\inf_{s > s_{\delta} - \epsilon} \tilde{J}(s) \le \delta - \gamma$ for some $\gamma > 0$ and thus
\begin{align}
\mathbb{P}(S_m > s_{\delta} - \epsilon) &\ge \exp \left(-m \inf_{s > s_{\delta} - \epsilon} \tilde{J}(s) + o(m) \right) \nonumber \\
&\ge \exp(-m (\delta - \gamma) + o(m)).
\end{align}
From \eqref{eqn:temp_appendix}, we hence have
\begin{equation}
    \mathbb{P}(T_{m, \exp(m \delta)} \le s_{\delta} - \epsilon) \le \exp(- \exp(m \delta) \exp(- m (\delta - \gamma) + o(m))) \le \exp(-\exp( m \gamma + o(m))).
\end{equation}
Taking the limit as $m\to \infty$ on both sides above gives the desired result.

Recall from Theorem~\ref{thm:p_expression} that
\begin{equation}
p_{ij} = \frac{G_{ij}h_j}{\lambda h_i} = \frac{P_{ij}\exp(\beta R_{ij})h_j}{\lambda h_i}.   
\end{equation}
From this,
\begin{equation}
\log\left(\frac{p_{ij}}{P_{ij}}\right) = \beta R_{ij} + \log h_j - \log h_i -
\log\lambda.    
\end{equation}
Recall that $\mu$ is the stationary distribution for $p$. Hence,
\begin{align}
d_{KL}(p,P) &= \sum_{i,j \in \mathcal{Y}} \mu_i p_{ij} \log\left(\frac{p_{ij}}{P_{ij}}\right) \\
&= \sum_{i,j\in \mathcal{Y}} \mu_i p_{ij} \left(\beta R_{ij} + \log h_j - \log h_i - \log\lambda \right) \\
&= \beta\sum_{i,j \in \mathcal{Y} }\mu_i p_{ij}R_{ij} + \sum_{i,j \in \mathcal{Y}}\mu_i p_{ij}\log h_j - \sum_{i,j \in \mathcal{Y}}\mu_i p_{ij}\log h_i - \log\lambda. \label{eq:temp2}
\end{align}
Since $\mu$ is stationary for $p$,
\begin{equation}
\sum_{i \in \mathcal{Y}} \mu_i p_{ij}=\mu_j \forall j \in \mathcal{Y}.
\end{equation}
Thus, since
\begin{equation}
\sum_{i,j \in \mathcal{Y}}\mu_i p_{ij}\log h_j = \sum_{j \in \mathcal{Y}}\mu_j\log h_j; \qquad \sum_{i,j \in \mathcal{Y}} \mu_i p_{ij}\log h_i = \sum_{i \in \mathcal{Y}} \mu_i\log h_i,
\end{equation}
and the second and third terms in \eqref{eq:temp2} cancel. Putting everything together, we obtain
\begin{align}

Assume that $q^{\text{best}}_{n, 1} = q_\beta$. Since both distributions are absolutely continuous with respect to $Q$, their Radon--Nikodym derivatives with respect to $Q$ must agree almost surely. Hence,
by~\eqref{eqn:rn_tilted} and Lemma~\eqref{lem:rn_best_of_n},
\begin{equation}
n P_L(r(y))^{n-1} = \frac{\exp(\beta r(y))}{A_\beta}
\end{equation}
almost surely. Equivalently,
\begin{equation}
n P_L(l)^{n-1} = \frac{\exp(\beta l)}{A_\beta}
\end{equation}
almost surely. From this,
\begin{equation}
P_L(l) = \exp\left( \frac{\beta(l-b)}{n-1} \right)
\end{equation}
almost surely, where $b=\frac{1}{\beta}\log(nA_\beta) > 0$.

Since $\exp\left( \frac{\beta(l-b)}{n-1} \right)$ is a continuous function of $l$ and it is the CDF of the random variable $L$, it must be that $\mathcal{L}$, the support of $L$, is also continuous. Further, since $P_L$ is a CDF, it must have a value of $1$ for $b \ge l$. From this, we get $\mathcal{L} = (-\infty,b]$. This proves the forward direction of the theorem.

Conversely, suppose that $L$ has the CDF as in the theorem statement. Its density is
\begin{equation}
p_L(l) = \lambda\exp(\lambda(l-b))
\mathbf 1_{\{l\le b\}},
\end{equation}
where $\lambda= \frac{\beta}{n - 1}$. Then,
\begin{equation}
A_\beta = \mathbb{E}_Q[\exp(\beta L)] = \frac{\lambda}{\lambda+\beta}\exp(\beta b) = \frac{\exp(\beta b)}{n}.
\end{equation}
Therefore,
\begin{equation}
\frac{dq_\beta}{dQ}(y) = \frac{\exp(\beta r(y))}{A_\beta} = n\exp(\beta(r(y)-b)) = nP_L(r(y))^{n-1} = \frac{d q^{\text{best}}_{n, 1}}{dQ}(y)
\end{equation}
almost surely. Hence, the converse also follows.

It remains to compute the KL divergence between the tilted distribution $q_\beta$ and the distribution $Q$ to ensure that the condition that $d_{KL}(q_\beta, Q) \le m \delta = \delta$ is satisfied. First, note from~\eqref{eqn:rn_tilted} and this theorem statement that $L'$ also follows a reverse exponential distribution. Namely, for a suitable constant $C > 0$,
\begin{align}
    p_{L'}(l) &= C \exp \left(\frac{\beta}{n - 1}(l-b) \right) \exp(\beta l), \qquad l\le b \\
    &= \exp \left(\frac{\beta n}{n - 1}(l-b) \right), \, \qquad \qquad \qquad l\le b.
\end{align}
Clearly, $L'$ also follows a reverse exponential distribution with parameter $\frac{\beta n}{n - 1}$.

Next, note from Corollary~\ref{corollary:KL_divergence} and using the expression for KL-divergence between two exponential distributions with parameters $\frac{\beta n}{n - 1}$ and $\frac{\beta}{n - 1}$ (see, for instance, \cite[Table 1]{kl_exponential_distributions}) that
\begin{equation}
    d_{KL}(q_\beta, Q) = d_{KL}(p_{L'}, p_L) = \log \left(\frac{n - 1}{\beta} \cdot \frac{\beta n}{n - 1} \right) + \frac 1n - 1 = \log \left(n \right) + \frac {n - 1}n.
\end{equation}

Recalling that the value of $\beta$ is chosen such that $d_{KL}(q_\beta, Q) = \delta$ from~\eqref{eqn:beta_value_choice} gives the desired result.

Recall that we need to calculate the below quantity for $n \ge 4$:
\begin{equation}
d_{KL}(q^{\text{best}}_{n, 1}, q_\beta) = \sum_{i \in [d]} \frac{i^n-(i-1)^n}{d^n} \log \left(\frac{i^n-(i-1)^n}{d^n} \cdot \frac{S_{d,n-1}}{i^{n-1}} \right),
\end{equation}
where $S_{d,n-1}=\sum_{j=1}^d j^{n-1}$.

First, note from the binomial expansion that
\begin{align}
i^n-(i-1)^n &= n i^{n-1} - \frac{n (n-1)}{2} i^{n-2} + O(i^{n-3}), \\
\frac{i^n-(i-1)^n}{i^{n-1}} &= n - \frac{n (n-1)}{2 i} + O \left(\frac 1{i^{2}} \right).
\end{align}
Also note from Faulhaber's formula \cite{faulhaber} that
\begin{align}
    S_{d, n - 1} &= \frac{d^n}{n} + \frac 12 d^{n-1} + O(d^{n - 2}), \\
    \frac{S_{d, n - 1}}{d^{n}} &= \frac{1}{n} + \frac 1{2d} + O \left(\frac{1}{d^2} \right).
\end{align}
Thus, 
\begin{align}
    \left(\frac{i^n-(i-1)^n}{d^n} \cdot \frac{S_{d,n-1}}{i^{n-1}} \right) &= \left(n - \frac{n (n-1)}{2 i} + O \left(\frac 1{i^{2}} \right) \right) \cdot \left(\frac{1}{n} + \frac 1{2d} + O \left(\frac{1}{d^2} \right) \right), \\
    &= 1 + \underbrace{\frac{n}{2d} - \frac{n-1}{2i} + O \left(\frac{1}{i^2}\right)}_{\delta_i} \label{eqn:discrete_proof_1}.
\end{align}
Let 
\begin{align}
    \delta_i = \frac{q^{\text{best}}_{n, 1}(y_i)}{q_\beta(y_i)} - 1 = \frac{n}{2d} - \frac{n-1}{2i} + O \left(\frac{1}{i^2} \right). \label{eqn:discrete_proof_3}
\end{align}

Recall that $q^{\text{best}}_{n, 1}$ and $q_\beta$ are both distributions, and thus satisfy
\begin{equation}
    \sum_{i \in [d]} q_\beta (y_i) \delta_i = 0. \label{eqn:discrete_proof_2}
\end{equation}
Thus, we have
\begin{align}
    d_{KL}(q^{\text{best}}_{n, 1}, q_\beta) = \sum_{i \in [d]} q^{\text{best}}_{n, 1}(y_i) \log\left(1 + \delta_i\right) & \overset{(a)}{=} \sum_{i \in [d]} q_\beta(y_i) (1 + \delta_i) \log\left(1 + \delta_i\right) \\
    &\overset{(b)}{=} \sum_{i \in [d]} q_\beta(y_i) \left(\delta_i +\frac{\delta_i^2}{2} + O(\delta_i^3) \right) \\
    &\overset{(c)}{=} \sum_{i \in [d]} q_\beta(y_i) \left(\frac{\delta_i^2}{2} + O(\delta_i^3) \right).
\end{align}
where $(a)$ uses $q^{\text{best}}_{n, 1}(y_i) = q_\beta(y_i) (1 + \delta_i) \forall i \in [d]$, $(b)$ uses $\log\left(1 + \delta_i\right) = \delta_i - \frac{\delta_i^2}{2} + O(\delta_i^3)$, and $(c)$ uses~\eqref{eqn:discrete_proof_2}.

Next, note from~\eqref{eqn:discrete_proof_3} that
\begin{equation}
    \delta_i^2 = \frac{n^2}{4d^2} + \frac{(n-1)^2}{4i^2} - \frac{2 n (n-1)}{4 i d} + O \left(\frac{1}{i^3}\right)
\end{equation}
and that
\begin{align}
    \sum_{i \in [d]} O(q_\beta(y_i) \delta_i^3) &= \sum_{i \in [d]} O \left( \frac{i^{n - 1}}{S_{d, n - 1}} \frac{1}{i^3} \right) = \sum_{i \in [d]} O \left(\frac{i^{n - 4}}{S_{d, n - 1}} \right) = O \left(\frac{S_{d, n-4}}{S_{d, n - 1}} \right) = O \left(\frac{1}{d^3} \right),
\end{align}
where the last step uses Faulhaber's formula \cite{faulhaber} again.

Next, we perform some algebraic manipulations and use Faulhaber's formula \cite{faulhaber} for the last time to get the desired result.
\begin{align}
    d_{KL}(q^{\text{best}}_{n, 1}, q_\beta) &= \frac 12 \sum_{i \in [d]} \frac{i^{n - 1}}{S_{d, n - 1}} \left(\frac{n^2}{4d^2} + \frac{(n-1)^2}{4i^2} - \frac{n (n-1)}{2 i d}\right) + O \left(\frac{1}{d^3}\right) \\
    &= \frac 12 \sum_{i \in [d]} \frac{i^{n - 1}}{S_{d, n - 1}} \left(\frac{n^2}{4d^2} + \frac{(n-1)^2}{4i^2} - \frac{n (n-1)}{2 i d}\right) + O \left(\frac{1}{d^3}\right) \\
    &= \frac{S_{d, n - 1}}{S_{d, n - 1}} \frac{n^2}{8 d^2} + \frac{(n-1)^2 S_{d, n - 3}}{8 S_{d, n - 1}} - \frac{n (n-1) S_{d, n - 2}}{4 S_{d, n - 1} d} + O \left(\frac{1}{d^3}\right) \\
    &= \frac{n^2}{8d^2} + \frac{(n-1)^2n}{8 d^{2} (n-2)} - \frac{n^2}{4 d^2} + O \left(\frac{1}{d^3}\right) \\
    &= \frac{n}{(n-2) 8d^2} + O \left(\frac{1}{d^3}\right).
\end{align}
