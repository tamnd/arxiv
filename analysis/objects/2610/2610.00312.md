<!-- theorem-1: body_md -->
\label{thm:sp}
Let $n\geq 2$, let $U$ be a proper subdomain of $\mathbb{R}^n$ with $\operatorname{cap} \partial U >0$, let $G$ be a bounded measurable conformal structure on $U$, and let $f:U\to U$ be a non-constant quasiregular map with $f^*G = G$ almost everywhere. Then the invariant modulus metric $d_{G}$ of Definition \ref{def:Gmetric} satisfies
\begin{equation}
\label{eq:sp1} 
d_G (f(x) , f(y) ) \leq d_G(x,y) 
\end{equation}
for all $x,y\in U$. Moreover,
\begin{enumerate}[(a)]
\item if $f$ is a homeomorphism of $U$ onto itself, then $f$ is a $d_G$-isometry;
\item if $f$ is a $d_G$-isometry, then $f$ is injective;
\item if $f$ is not injective, then there are points $x\neq y$ with strict inequality in \eqref{eq:sp1}.
\end{enumerate}

<!-- theorem-2: body_md -->
\label{thm:dw}
Let $f:\mathbb{B}^n \to \mathbb{B}^n$ be uniformly quasiregular and let $d$ be the chain metric of Definition \ref{def:chain}. Then $d$ is a proper metric inducing the Euclidean topology, $(\mathbb{B}^n , d)$ is Gromov hyperbolic with Gromov boundary $\mathbb{S}^{n-1}$, $f$ is $d$-non-expanding, and exactly one of the following holds.
\begin{enumerate}[(a)]
\item Every orbit is relatively compact in $\mathbb{B}^n$. Then either $f^m \to x_0 \in \mathbb{B}^n$ locally uniformly, where $x_0$ is the unique fixed point, attracting or super-attracting, or $f$ is a uniformly quasiconformal homeomorphism of $\mathbb{B}^n$ onto itself, a $d$-isometry, and some subsequence $f^{k_j} \to \operatorname{id}$.
\item There is $w\in \mathbb{S}^{n-1}$ with $f^m \to w$ locally uniformly on $\mathbb{B}^n$. Moreover, $|f^m(x) - f^m(y)| \to 0$ for all $x,y$. For each $x\in \mathbb{B}^n$, the step sizes $s_m = d(f^m(x) , f^{m+1}(x))$ are non-increasing with limit $s\geq 0$, and there is a horofunction $h$ of $(\mathbb{B}^n , d)$ with $h(x) = 0$ such that the orbit of every $y$ lies in $\{ h \leq d(x,y) \}$.
\end{enumerate}

<!-- corollary-1: body_md -->
\label{cor:wolff}
Let $f:\mathbb{B}^n\to\mathbb{B}^n$ be uniformly quasiregular. If $f$ has no fixed point in $\mathbb{B}^n$ and is not a uniformly quasiconformal homeomorphism of $\mathbb{B}^n$ onto itself, then there is $w\in\mathbb{S}^{n-1}$ with $f^m\to w$ locally uniformly on $\mathbb{B}^n$.

<!-- theorem-3: body_md -->
\label{thm:stable}
Let $f:\mathbb{S}^n \to \mathbb{S}^n$ be a uniformly quasiregular map of degree at least $2$ and let $U$ be a stable component of $F(f)$ which is a uniform domain with uniformly perfect boundary (in the spherical metric). Then $U$ is either a (super-)attracting basin, a rotation domain, or a parabolic basin. In the last case, there is a unique $w \in \partial U$ which is a fixed point of $f$, with $f^m \to w$ locally uniformly on $U$.
<!-- end theorem-3 -->
<!-- theorem-4: body_md -->
\label{thm:examples}
For every $n\geq 2$, there exist non-injective uniformly quasiregular self-maps of $\mathbb{B}^n$ in the escaping case, of positive-step type, whose conformal model at the Denjoy-Wolff point $w$ is a hyperbolic M\"obius map, or a parabolic one.

<!-- theorem-5: body_md -->
\label{thm:coarse}
Let $U\subset\mathbb{S}^n$ be a uniform domain with uniformly perfect boundary and let $d$ be a chain metric built from $\mu_U$ as in Definition \ref{def:chain}, with $r$ sufficiently large. Then $(U,d)$ is Gromov hyperbolic, quasi-isometric to $(U,\mu_U)$ and to $U$ with its quasihyperbolic metric, and its Gromov boundary is $\partial U$.

\label{lem:list}
\begin{enumerate}[(a)]
\item There exist constants $b_1,b_2$ depending only on $n$ such that $\mu_U \leq b_1 k_U + b_2$ for every such $U$, see \cite[Lemma 2.39(3)]{Vuo85} or \cite[Lemma 10.7]{HKV}.

\item The Julia set of a uniformly quasiregular map $f:\mathbb{S}^n \to \mathbb{S}^n$ of degree at least $2$ is uniformly perfect, see the paper of the author and Nicks \cite{FN11}.
\end{enumerate}

<!-- theorem-6: body_md -->
\label{thm:calka}
Let $f$ be a nonexpansive mapping of a finitely totally bounded metric space $M$ into itself. If for some $z_0\in M$ the sequence $\{ f^m(z_0) \}$ contains a bounded subsequence then for every $z\in M$ the sequence $\{f^m(z) \}$ is bounded.
<!-- end theorem-6 -->
<!-- lemma-2: body_md -->
\label{lem:null}
If $N\subset \mathbb{R}^n$ has measure zero, then the family $\Gamma_N$ of locally rectifiable paths meeting $N$ in a set of positive length has modulus zero.
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
Enlarge $N$ to a Borel null set. For $k\in \mathbb{N}$ let $\Gamma_{N,k}$ consist of the paths meeting $N$ in length at least $1/k$. The Borel density $k \chi_N$ is admissible for $\Gamma_{N,k}$ with $\int (k\chi_N)^n \: dm = 0$, so $M(\Gamma_{N,k}) = 0$, and $M(\Gamma_N) = 0$ by countable subadditivity.
<!-- end proof-1 -->
<!-- definition-1: body_md -->
\label{def:Gadmiss}
We say that $\eta$ is $G$-admissible for the path family $\Gamma$ if there is a subfamily $\Gamma_0 \subset \Gamma$ with $M(\Gamma_0) = 0$ such that $\int_{\gamma} \eta \: |d\gamma|_G \geq 1$ for every locally rectifiable $\gamma \in \Gamma \setminus \Gamma_0$. We then define the $G$-modulus of $\Gamma$ to be
\[ M_G(\Gamma) = \inf \int_U \eta^n \: dm \]
where the infimum is over all $G$-admissible $\eta$.
<!-- end definition-1 -->
<!-- lemma-3: body_md -->
\label{lem:MGind}
The $G$-modulus $M_G$ does not depend on the choice of Borel representative $\tilde{G}$.
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
Suppose that $\tilde{G_1}$ and $\tilde{G_2}$ are two Borel representatives of $G$ that agree outside a null set $N$. By Lemma \ref{lem:null}, $M(\Gamma_N) = 0$, and off $\Gamma_N$ the two integrals $\int_{\gamma} \eta |d\gamma|_{G_i}$ coincide for every $\eta$. Hence the two notions of $G$-admissibility agree.
<!-- end proof-2 -->
<!-- lemma-4: body_md -->
\label{lem:Gcomp}
For every path family $\Gamma$ in $U$, we have
\[ \lambda^{-n} M(\Gamma) \leq M_G(\Gamma) \leq \lambda^n M(\Gamma).\]
In particular, $M_G$-null families are exactly $M$-null families, and $M_G$ is countably subadditive and satisfies the minorization inequality.
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
If $\eta$ is $G$-admissible, then $\lambda \eta$ is admissible in the classical sense for $\Gamma \setminus \Gamma_0$, since by the convention preceding Lemma \ref{lem:null}, \eqref{eq:vcomp} holds for $\tilde{G}$ everywhere. Since $M(\Gamma) = M(\Gamma\setminus \Gamma_0)$, this gives $M(\Gamma) \leq \lambda^n M_G(\Gamma)$. Symmetrically, we have $M_G(\Gamma) \leq \lambda^n M(\Gamma)$. The remaining properties have the same proofs as for $M$, as exceptional families are absorbed into $\Gamma_0$.
<!-- end proof-3 -->
<!-- definition-2: body_md -->
\label{def:Gmetric}
For $x,y\in U$, we set
\[ d_G(x,y) = \inf \{ M_G(\Delta ( C , \partial U ; U)) : C\subset U \text{ a continuum }, x,y \in C \}.\]
<!-- end definition-2 -->
<!-- proposition-1: body_md -->
\label{prop:dGmetric}
The expression $d_G$ in Definition \ref{def:Gmetric} is a metric on $U$.
<!-- end proposition-1 -->
<!-- proof-4: body_md -->
Symmetry is clear. For positivity, if $x\neq y$, then $d_G(x,y) \geq \lambda^{-n} \mu_U(x,y) > 0$ as $\mu_U$ is a metric. If we take $C = \{x \}$, then $M( \Delta ( \{ x \} , \partial U ; U)) =0$ as we have a family of paths all passing through one point. Then Lemma \ref{lem:Gcomp} implies that $M_G(\Delta ( \{ x \} , \partial U ; U)) =0$ and hence $d_G(x,x) = 0$.

Finally, for the triangle inequality, given $\epsilon >0$, choose continua $C_1 \ni x,y$ and $C_2 \ni y,z$ with $M_G(\Delta(C_i,\partial U ; U))$ within $\epsilon$ of the respective infima. Then $C =C_1 \cup C_2$ is a continuum containing $x,z$ and a path starting in $C$ starts in $C_1$ or in $C_2$, so 
\[ \Delta ( C , \partial U ; U) = \Delta (C_1,\partial U ; U) \cup \Delta ( C_2 , \partial U ; U).\] 
Then subadditivity gives 
\[ d_G(x,z) \leq d_G(x,y) + d_G(y,z) + 2\epsilon.\]
As $\epsilon $ was arbitrary, the triangle inequality holds.
<!-- end proof-4 -->
<!-- proposition-2: body_md -->
\label{prop:top}
We have the following properties.
\begin{enumerate}[(a)]
\item The metric $d_G$ induces the Euclidean topology on $\mathbb{B}^n$. 
\item A set is $d_G$-bounded if and only if it is $\rho$-bounded if and only if it is relatively compact in $\mathbb{B}^n$.
\item The metric space $(\mathbb{B}^n,d_G)$ is proper.
\item If $x\in \mathbb{B}^n$, then $d_G(x,y_j) \to \infty$ if and only if $\operatorname{dist}(y_j , \mathbb{S}^{n-1}) \to 0$.
\end{enumerate}
<!-- end proposition-2 -->
<!-- proof-5: body_md -->
By \eqref{eq:mucomps} and Section \ref{ss:metrics}, we have
\[ \lambda^{-n} \Phi ( \rho(x,y)) \leq d_G(x,y) \leq \lambda^n \Phi (\rho(x,y)),\]
where $\Phi$ is an increasing homeomorphism of $[0,\infty)$. All the claims now follow from the corresponding ones for the hyperbolic metric $\rho$: it induces the Euclidean topology, $\rho$-bounded sets are exactly the relatively compact subsets of $\mathbb{B}^n$, $(\mathbb{B}^n,\rho)$ is proper, and $\rho(x,y_j) \to \infty$ if and only if $\operatorname{dist} (y_j , \mathbb{S}^{n-1}) \to 0$.
<!-- end proof-5 -->
<!-- lemma-5: body_md -->
\label{lem:roughhyp}
For all $x,y\in \mathbb{B}^n$, we have
\[ \lambda^{-n} 2^{n-1} c_n \rho(x,y) \leq d_G(x,y) \leq \lambda^n 2^{n-1}c_n ( \rho(x,y) + \log 4),\]
where $c_n$ is the constant from \eqref{eq:cn} that depends only on $n$.
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
Put $t = \rho(x,y)$ and $s = 1/\tanh(t/2)$ so that $\mu_{\mathbb{B}^n}(x,y) = \gamma_n(s)$ by Section \ref{ss:metrics}. Then
\[ \frac{s+1}{s-1} = \frac{ \cosh(t/2) + \sinh(t/2) }{\cosh(t/2) - \sinh(t/2)} = e^t.\]
Then Anderson's result in \eqref{eq:cn} gives
\[ 2^{n-1} c_n\rho(x,y) \leq \mu_{\mathbb{B}^n}(x,y) \leq 2^{n-1} c_n \rho(x,y) + 2^{n-1}c_n \log 4.\]
The lemma now follows from \eqref{eq:mucomps}. The two-sided estimate for $\mu_{\mathbb{B}^n}$ is \cite[(2.24)]{Vuo85}, with $8\log 2$ in place of $\log 4$, and the identity $(s+1)/(s-1) = e^t$ is \cite[(4.2)]{Lin07}.

<!-- lemma-6: body_md -->
\label{lem:Gpol}
Let $f:U\to U'$ be non-constant quasiregular and $G,G'$ measurable conformal structures on $U,U'$ respectively with $f^*G' = G$ almost everywhere. Then for every path family $\Gamma$ in $U$, we have
\[ M_{G'}(f(\Gamma)) \leq M_G(\Gamma).\]
<!-- end lemma-6 -->
<!-- proof-7: body_md -->
The proof follows Rickman's proof of Poletsky's inequality \cite[Theorem II.8.1]{Rickman}, with the $K_I$-step replaced by an identity. We write $\ell(\gamma)$ for length, $\gamma^0$ for the arclength parametrization of a locally rectifiable path $\gamma$, and $s_{\gamma}$ for its length function, so that $\gamma = \gamma^0 \circ s_{\gamma}$.

{\it Step 1: null sets.} Let $B_f$ be the branch set of $f$. Then $|B_f| = |f(B_f)| = 0$, $f$ is differentiable almost everywhere with $J_f>0$ almost everywhere, and $f$ satisfies Lusin's condition (N), see \cite[Sections I.2 and I.4]{Rickman}. Let $N\subset U$ and $N'\subset U'$ be null sets outside which $\tilde{G} = G$ and $\tilde{G}' = G'$ respectively. Since $\int_{f^{-1}(N')} J_f \: dm \leq \int_{N'} N(y,f,U) \: dm(y) = 0$ by the area formula \cite[Lemma I.4.11]{Rickman} and $J_f>0$ almost everywhere, $f^{-1}(N')$ has measure zero. Let $E\subset U$ be a Borel null set containing $B_f$, $N$, $f^{-1}(N')$, the set where $f$ is not differentiable or $J_f = 0$, and the set where $f^*G' = G$ fails. Then $|E| = 0$ and, by condition (N), $|f(E)| = 0$.

{\it Step 2: the pointwise identity.} For $x\notin E$ and $v\in\mathbb{R}^n$, the invariance $Df(x)^TG'(f(x))Df(x)=J_f(x)^{2/n}G(x)$ gives
\[ |Df(x)v|_{G'(f(x))}^2 = \langle Df(x)^T G'(f(x)) Df(x) v , v \rangle = J_f(x)^{2/n} |v|_{G(x)}^2 .\]
Hence the minimal $G$-stretch
\[ \ell_G(Df(x)):=\min\{|Df(x)v|_{G'(f(x))}:|v|_{G(x)}=1\}\]
satisfies
\begin{equation}
\label{eq:lG}
|Df(x)v|_{G'(f(x))} = \ell_G(Df(x)) |v|_{G(x)} \text{ for all } v, \quad \text{and} \quad \ell_G(Df(x))^n=J_f(x), \qquad x\notin E .
\end{equation}

{\it Step 3: the pushed-forward density.} Let $\eta$ be $G$-admissible for $\Gamma$ with exceptional family $\Gamma_0$, and put $g = \eta \, J_f^{-1/n}$ on $U\setminus E$, a non-negative measurable function. Since $f$ is a local homeomorphism on the open set $U\setminus B_f$, which is second countable, there are open sets $A_1,A_2,\ldots$ covering $U\setminus B_f$ with $f|_{A_i}$ injective. Each $f|_{A_i\setminus E}$ is a continuous injection of a Borel set, so $f(A_i\setminus E)$ is Borel and $(f|_{A_i\setminus E})^{-1}$ is Borel measurable by the Lusin-Souslin theorem. Hence
\[ \eta_i'(y) = \begin{cases} g \big ( (f|_{A_i\setminus E})^{-1}(y) \big ), & y\in f(A_i \setminus E), \\ 0, & \text{otherwise}, \end{cases}\]
is measurable on $U'$, and so is
\[ \eta'(y) := \sup_i \eta_i'(y) = \sup \{ g(x) : x\in f^{-1}(y) \setminus E \} .\]
Choose a Borel function $\eta_0 \geq 0$ on $U'$ with $\eta_0 = \eta'$ outside a Borel null set $N''\subset U'$. Since $\eta'(y)^n \leq \sum_{x\in f^{-1}(y)\setminus E} g(x)^n$, the area formula \cite[Lemma I.4.11]{Rickman} and \eqref{eq:lG} give
\begin{equation}
\label{eq:mass}
\int_{U'} \eta_0^n \: dm = \int_{U'} (\eta')^n \: dm \leq \int_{U'} \sum_{x\in f^{-1}(y)\setminus E} g(x)^n \: dm(y) = \int_{U\setminus E} g^n J_f \: dm = \int_U \eta^n \: dm .
\end{equation}

{\it Step 4: exceptional families for $f(\Gamma)$.} Let $\Gamma_1\subset\Gamma$ be the family of paths $\gamma\in\Gamma$ such that $f\circ\gamma$ is locally rectifiable and $f$ is not absolutely precontinuous on $\gamma$ in the sense of \cite[Section II.5]{Rickman}; recall that this means that the path $\gamma^*$ defined by $\gamma = \gamma^* \circ s_{f\circ\gamma}$ and $f\circ \gamma^* = (f\circ\gamma)^0$ is not locally absolutely continuous; $\gamma^*$ is well defined because $f\circ\gamma$ is constant on an interval only if $\gamma$ is, $f$ being discrete. Poletsky's lemma \cite[Lemma II.5.3]{Rickman} states that $M(f(\Gamma_1)) = 0$. Next, $M(f(\Gamma_0)) \leq K_I(f) M(\Gamma_0) = 0$ by Poletsky's inequality. Finally, let $\Gamma_2'$ be the family of locally rectifiable paths in $U'$ meeting the null set $f(E) \cup N' \cup N''$ in positive length; $M(\Gamma_2') = 0$ by Lemma \ref{lem:null}. Put
\[ \Gamma_0' := f(\Gamma_0) \cup f(\Gamma_1) \cup \Gamma_2' ,\]
a family of modulus zero by countable subadditivity. We claim that $\int_{\beta} \eta_0 |d\beta|_{G'} \geq 1$ for every locally rectifiable $\beta \in f(\Gamma) \setminus \Gamma_0'$, which shows that $\eta_0$ is $G'$-admissible for $f(\Gamma)$; together with \eqref{eq:mass} and the infimum over $\eta$ this proves the lemma.

{\it Step 5: admissibility.} Let $\beta = f\circ \gamma$ be as in the claim, $\gamma\in\Gamma$. As $\beta\notin f(\Gamma_1)$, $f$ is absolutely precontinuous on $\gamma$, so $\gamma^*$ is locally absolutely continuous; in particular $\gamma$ is locally rectifiable, and $\gamma \notin \Gamma_0$ as $\beta \notin f(\Gamma_0)$. Since $\beta^0 = f\circ \gamma^*$ and $\beta \notin \Gamma_2'$, for almost every $t$ we have: $\gamma^*$ is differentiable at $t$, $\beta^0(t) \notin f(E)\cup N'\cup N''$, hence $\gamma^*(t) \notin E$, $f$ is differentiable at $\gamma^*(t)$ with
\[ (\beta^0)'(t) = Df(\gamma^*(t)) (\gamma^*)'(t), \qquad |(\beta^0)'(t)| = 1 ,\]
and $\eta_0(\beta^0(t)) = \eta'(\beta^0(t)) \geq g(\gamma^*(t))$. Using \eqref{eq:lG} at $x = \gamma^*(t)$,
\begin{align*} \int_{\beta} \eta_0 \: |d\beta|_{G'} &= \int_0^{\ell(\beta)} \eta_0(\beta^0(t)) |(\beta^0)'(t)|_{\tilde{G}'(\beta^0(t))} \: dt \geq \int_0^{\ell(\beta)} g(\gamma^*(t)) \, \ell_G ( Df (\gamma^*(t))) \, |(\gamma^*)'(t)|_{\tilde{G}(\gamma^*(t))} \: dt \\ &= \int_0^{\ell(\beta)} \eta(\gamma^*(t)) |(\gamma^*)'(t)|_{\tilde{G}(\gamma^*(t))} \: dt .\end{align*}
For a locally absolutely continuous path $\alpha$ and Borel $\eta\geq 0$ one has $\int_{\alpha} \eta \: |d\alpha|_G = \int \eta(\alpha(t)) |\alpha'(t)|_{\tilde{G}(\alpha(t))} \: dt$, by the change of variables $\alpha = \alpha^0\circ s_{\alpha}$ with $s_\alpha$ absolutely continuous, exactly as in \cite[Theorem 4.1]{Vaisala} for the Euclidean length element. Applied to $\alpha = \gamma^*$, and since $\gamma = \gamma^*\circ s_{f\circ\gamma}$ with $s_{f\circ\gamma}$ continuous and non-decreasing, so that $\gamma$ and $\gamma^*$ have the same arclength parametrization, the last integral equals $\int_{\gamma} \eta \: |d\gamma|_G \geq 1$, because $\gamma\in\Gamma\setminus\Gamma_0$ is locally rectifiable. This proves the claim.

<!-- lemma-7: body_md -->
\label{lem:minlifts}
Let $f:U\to U$ be non-constant quasiregular and let $C\subset U$ be a continuum. Then $\Delta ( f(C) , \partial U ; U)$ is minorized by $f( \Delta ( C,\partial U ; U))$.
<!-- end lemma-7 -->
<!-- proof-8: body_md -->
Let $\beta : [a,b) \to U$ lie in $\Delta ( f(C) , \partial U ; U)$ and pick $x_0 \in C$ with $f(x_0) = \beta (a)$. Let $\alpha :[a,c) \to U$ be a maximal $f$-lift of $\beta$ starting at $x_0$, using the path lifting property of Section \ref{ss:qrmaps}. If $c<b$, then $\operatorname{dist} (\alpha(t) , \partial U) \to 0$. If $c=b$, then also $\operatorname{dist} (\alpha(t) , \partial U) \to 0$, otherwise some $\alpha (t_j)$, where $t_j\to b$, stays in a compact subset $E$ of $U$ and then $\beta(t_j) = f(\alpha(t_j)) \in f(E)$, another compact subset of $U$. This contradicts $\operatorname{dist} (\beta(t) , \partial U) \to 0$. In both cases $\alpha \in \Delta(C,\partial U ; U)$ and $f\circ \alpha = \beta |_{[a,c)}$ is a subpath of $\beta$.
<!-- end proof-8 -->
<!-- proof-9: body_md -->
Given $\epsilon >0$, choose a continuum $C \ni x,y$ with $M_G( \Delta (C , \partial U ; U)) \leq d_G(x,y) + \epsilon$. Then $f(C)$ is a continuum containing $f(x),f(y)$, and by Lemmas \ref{lem:Gcomp}, \ref{lem:Gpol} and \ref{lem:minlifts} we have
\[ d_G(f(x),f(y)) \leq M_G ( \Delta ( f(C) , \partial U ; U)) \leq M_G( f( \Delta ( C,\partial U; U))) \leq M_G( \Delta (C,\partial U;U)) \leq d_G(x,y) + \epsilon.\]
As $\epsilon $ was arbitrary, the first claim follows. Note that $(f^m)^*G = G$ follows from $f^*G = G$ by the chain rule, so every iterate of $f$ is also $d_G$-non-expanding. For~(a), if $f$ is a homeomorphism, then $f^{-1}$ is quasiconformal, hence differentiable almost everywhere and satisfying Lusin's condition (N), so the identity $f^*G = G$ at $x$, transported to $y = f(x)$, gives $(f^{-1})^*G = G$ almost everywhere; both $f$ and $f^{-1}$ are then non-expanding by \eqref{eq:sp1}. For~(b), if $f(x) = f(y)$, then $d_G(x,y) = d_G(f(x),f(y)) = 0$ and hence $x=y$ and $f$ is injective. For~(c), we can take any $x\neq y$ with $f(x) = f(y)$ and obtain the conclusion.

<!-- definition-3: body_md -->
\label{def:chain}
Let $\Psi$ be the class of continuous non-decreasing functions $\psi : [0,\infty) \to [0, \infty)$ with $\psi(0)=0$ and $\psi(u) \geq u/2$ for all $u$. For $r>0$, an $r$-chain from $x$ to $y$ is a finite sequence $x=z_0,z_1,\ldots, z_m = y$ in $\mathbb{B}^n$ with $d_G(z_i,z_{i+1}) \leq r$ for all $i$. For $\psi \in \Psi$ put
\[ d_r^{\psi}(x,y) := \inf \left \{ \sum_{i=0}^{m-1} \psi \left ( d_G(z_i,z_{i+1}) \right ) : (z_i) \text{ is an $r$-chain from $x$ to $y$} \right \}. \]
<!-- end definition-3 -->
<!-- lemma-8: body_md -->
\label{lem:chainprop}
Let $\psi \in \Psi$ and $r\geq r_1$.
\begin{enumerate}[(a)]
\item The expression $d_r^{\psi}$ is a metric, $d_r^{\psi} \geq \tfrac12 d_r \geq \tfrac12 d_G$, and $d_r^{\psi}(x,y) \leq \psi(d_G(x,y))$ whenever $d_G(x,y) \leq r$. For $\psi = \operatorname{id}$, $d_r = d_G$ on such pairs.
\item The metric $d_r^{\psi}$ induces the Euclidean topology, $(\mathbb{B}^n , d_r^{\psi})$ is proper, $d_r^{\psi}$-bounded sets are precisely the relatively compact subsets of $\mathbb{B}^n$.
\item We have 
\[ \tfrac12 \lambda^{-n} 2^{n-1}c_n \rho \leq d_r^{\psi} \leq \left ( \frac{ \rho }{\log 4} + 1 \right ) \psi(r).\]
\item If $g:\mathbb{B}^n \to \mathbb{B}^n$ is non-expanding for $d_G$, then $g$ is non-expanding for $d_r^{\psi}$. If $g$ is a bijective $d_G$-isometry, then it is a $d_r^{\psi}$-isometry.
\end{enumerate}
<!-- end lemma-8 -->
<!-- proof-10: body_md -->
For~(a), symmetry and the triangle inequality come from reversing and concatenating chains. As $\psi(u) \geq u/2$, every chain sum is at least half the corresponding plain sum, which is at least $d_G(x,y)$ by the triangle inequality for $d_G$. This gives $d_r^{\psi} \geq \tfrac12 d_r \geq \tfrac12 d_G$, in particular positivity. When $d_G(x,y) \leq r$, the one-step chain gives the upper bound.

For~(b), by ~(a) and the continuity of $\psi$ at $0$, we have $d_r^{\psi}(x_j,x) \to 0$ if and only if $d_G(x_j,x) \to 0$, so the topologies agree. Since $d_r^{\psi} \geq \tfrac12 d_G$, the $d_r^{\psi}$-bounded sets are $d_G$-bounded and hence relatively compact by Proposition \ref{prop:top}~(b). Closed $d_r^{\psi}$-balls are closed in the Euclidean sense and relatively compact, hence compact.

For~(c), the lower bound is part~(a) and Lemma \ref{lem:roughhyp}. For the upper bound, let $t = \rho(x,y)$ and choose points $z_0 = x,\ldots, z_m=y$ on the $\rho$-geodesic from $x$ to $y$ with $\rho(z_i,z_{i+1}) \leq \log 4$ and $m\leq t/\log 4+1$. By Lemma \ref{lem:roughhyp}, 
\[ d_G (z_i,z_{i+1}) \leq a (\log 4 + \log 4) = 2a \log 4 \leq r,\]
so this is an $r$-chain and each step gives at most $\psi(r)$.

Finally, for~(d), if $(z_i)$ is an $r$-chain from $x$ to $y$ then $(g(z_i))$ is a chain from $g(x)$ to $g(y)$ with $d_G(g(z_i) , g(z_{i+1})) \leq d_G(z_i,z_{i+1}) \leq r$, and hence is an $r$-chain. Taking the infimum and applying this observation to $g$ and $g^{-1}$ gives the result.

<!-- proposition-3: body_md -->
\label{prop:roughgeos}
The space $(\mathbb{B}^n , d_r^{\psi})$ is $h$-roughly geodesic with $h = \psi(r) +1$. That is, for all $x,y$ there is a rough geodesic $\sigma :[0,S] \to \mathbb{B}^n$ with $\sigma(0)=x$, $\sigma(S) = y$, $S\leq d_r^{\psi}(x,y)+1$, and
\[ \left |d_r^{\psi} (\sigma(s) , \sigma(t) ) - |s-t| \right | \leq h,\]
for $s,t\in [0,S]$.
<!-- end proposition-3 -->
<!-- proof-11: body_md -->
Write $d' = d_r^{\psi}$. Choose an $r$-chain $(z_i)_{i=0}^m$ from $x$ to $y$ with 
\[ S:= \sum_{i=0}^{m-1} \psi \left ( d_G(z_i,z_{i+1}) \right ) \leq d'(x,y) +1 ,\]
and let $S_i$ be the partial sums $0=S_0 \leq S_1 \leq \ldots \leq S_m =S$, with $S_{i+1} - S_i \leq \psi(r)$. After discarding repeated consecutive points we may assume $S_i < S_{i+1}$. Define $\sigma(s) = z_i$ for $s\in [S_i,S_{i+1})$ and $\sigma(S) = y$. For $i\leq j$ the sub-chain gives $d'(z_i,z_j) \leq S_j-S_i$, and
\[ d'(x,y) \leq d'(x,z_i) + d'(z_i,z_j) + d'(z_j,y) \leq S_i + d'(z_i,z_j) + (S-S_j) \]
gives $d'(z_i,z_j) \geq S_j-S_i-1$. If $s\in [S_i,S_{i+1})$, $t\in [S_j,S_{j+1})$, $i\leq j$, then $|t-s|$ is within $\psi(r)$ of $S_j-S_i$. Hence
\[ |d'(\sigma(s) , \sigma (t)) - |t-s|| \leq \psi(r)+1.\]
The endpoint $t=S$ is handled the same way.

<!-- theorem-7: body_md -->
\label{thm:gromovchain}
Let $\psi \in \Psi$ and $r\geq r_1$. Then $(\mathbb{B}^n , d_r^{\psi})$ satisfies the four-point Gromov hyperbolic condition, and a sequence $(z_k)$ is a Gromov sequence for $d_r^{\psi}$ if and only if it is one for $\rho$, with two Gromov sequences equivalent for $d_r^{\psi}$ if and only if they are equivalent for $\rho$. Consequently \eqref{eq:gromov} holds and the Gromov boundary of $(\mathbb{B}^n , d_r^{\psi})$ is $\mathbb{S}^{n-1}$ in the sense of Section \ref{ss:metrics}.
<!-- end theorem-7 -->
<!-- proof-12: body_md -->
Write $X = (\mathbb{B}^n,d')$ with $d' = d_r^{\psi}$, $Y = (\mathbb{B}^n , \rho)$, $h = \psi(r)+1$, and let $\delta_0$ be a Gromov hyperbolicity constant for the geodesic space $Y$; for $Y = (\mathbb{B}^n,\rho)$ one may take $\delta_0 = \log 3$.

Let $\sigma:[0,S] \to X$ be an $h$-rough geodesic. Then $\iota \circ \sigma$ satisfies
\[ L^{-1} |s-t| - C' \leq \rho (\sigma(s),\sigma(t) ) \leq L|s-t| + C'\]
with $C' = C+Lh$, that is, it is an $(L,C')$-quasi-geodesic in $Y$, not necessarily continuous. By \cite[Theorem III.H.1.7]{BH99} (the stability of quasi-geodesics, sometimes called the Morse Lemma), its image lies within $\rho$-Hausdorff distance $H = H(L,C',\delta_0)$ of the $\rho$-geodesic joining its endpoints. Hence, given three points and $h$-rough geodesics $\sigma_1,\sigma_2,\sigma_3$ joining them pairwise, each $\iota(\sigma_i)$ lies in the $\rho$-neighbourhood of radius $\delta_0 + 2H$ of $\iota(\sigma_j)\cup \iota ( \sigma_k)$, because the corresponding $\rho$-geodesic triangle is $\delta_0$-slim. Pulling back through the quasi-isometry, each $\sigma_i$ lies in the $d'$-neighbourhood of radius $\delta_1:= L(\delta_0 + 2H) +C$ of $\sigma_j\cup \sigma_k$. So rough geodesic triangles in $X$ are slim.

Let $w\in X$ and let $\sigma$ be an $h$-rough geodesic from $x$ to $y$. Then we claim that
\begin{equation}
\label{eq:gromovchain1}
(x|y)_w - h -1 \leq d'(w,\sigma) \leq (x|y)_w + \kappa, \quad \kappa = 2\delta_1+8h .
\end{equation}
For the lower bound, for $p=\sigma(s)$,
\[ d'(x,p) + d'(p,y) \leq S+2h \leq d'(x,y) + 2h+1,\]
and adding $d'(w,p) \geq d'(w,x) - d'(x,p)$ to $d'(w,p) \geq d'(w,y) - d'(p,y)$ gives $d'(w,p) \geq (x|y)_w - h -\tfrac12$. For the upper bound, let $\sigma_{xw}, \sigma_{wy}$ be $h$-rough geodesics and choose $p\in \sigma$ with $|d'(x,p) - (y|w)_x| \leq 2h$. This is possible as $s\mapsto d'(x,\sigma(s))$ is within $h$ of $s$ and $S \geq d'(x,y) - h$. As rough geodesic triangles in $X$ are slim, $p$ is within $\delta_1$ of some $q$ on $\sigma_{xw}$ or on $\sigma_{wy}$. In the first case, the rough geodesic property of $\sigma_{xw}$ gives 
\[ d'(w,q) \leq d'(x,w) - d'(x,q) +3h \leq d'(x,w) - d'(x,p) +\delta_1 +3h \leq d'(x,w) - (y|w)_x + \delta_1 +5h = (x|y)_w + \delta_1+5h,\]
using $d'(x,w) - (y|w)_x = (x|y)_w$. So $d'(w,p) \leq (x|y)_w + 2\delta_1 + 5h$. In the second case, $q\in \sigma_{wy}$, the same argument with the roles of $x$ and $y$ exchanged uses $d'(y,p) \geq S - s - h \geq d'(x,y) - s - 2h \geq (x|w)_y - 5h$, where $s \leq d'(x,p) + h \leq (y|w)_x + 3h$, and $d'(y,w) - (x|w)_y = (x|y)_w$, giving $d'(w,p) \leq (x|y)_w + 2\delta_1 + 8h$.

Next, let $x,y,z,w \in X$ and choose $p$ on an $h$-rough geodesic $\sigma_{xz}$ with $d'(w,p) \leq (x|z)_w + \kappa+1$ by \eqref{eq:gromovchain1}. Then $p$ is within $\delta_1$ of some $q \in \sigma_{xy} \cup \sigma_{yz}$, say $q\in \sigma_{xy}$. Then by \eqref{eq:gromovchain1},
\[ (x|y)_w \leq d'(w,\sigma_{xy}) + h+1 \leq d'(w,q) + h +1 \leq d'(w,p) +\delta_1 + h+1 \leq (x|z)_w + \kappa +\delta_1 +h+2.\] 
The case $q\in \sigma_{yz}$ gives the same bound for $(y|z)_w$. Hence
\[ (x|z)_w \geq \min \{ (x|y)_w , (y|z)_w \} - \delta, \quad \delta :=\kappa+\delta_1 + h+2,\]
showing the four-point Gromov hyperbolicity condition.

Now, let $(z_k)$ be any sequence in $\mathbb{B}^n$, let $\sigma_{km}$ be an $h$-rough geodesic in $X$ from $z_k$ to $z_m$ and $[z_k,z_m]$ the $\rho$-geodesic. In $Y$, the geodesic case $h=0$, $\delta_1 = \delta_0$ of \eqref{eq:gromovchain1} gives
\[ \rho ( o, [z_k,z_m]) \geq (z_k|z_m)_o^{\rho} \geq \rho ( o, [z_k,z_m]) - 2\delta_0.\]
By the stability of quasi-geodesics, the $\rho$-Hausdorff distance between $[z_k,z_m]$ and $\iota(\sigma_{km})$ is at most $H$, so 
\[ | \rho ( o , [z_k,z_m]) - \rho ( o ,\sigma_{km} ) | \leq H.\]
By the quasi-isometry inequalities and \eqref{eq:gromovchain1}, we have
\[ \frac{ d'(o,\sigma_{km}) - C}{L} \leq \rho(o,\sigma_{km}) \leq L d'(o,\sigma_{km}) + C, \qquad (z_k|z_m)_o^{d'} - h - 1 \leq d'(o,\sigma_{km}) \leq (z_k|z_m)_o^{d'} + \kappa .\]
Combining the three displays,
\[ \frac{ (z_k|z_m)_o^{d'} - h - 1 - C }{L} - H - 2\delta_0 \leq (z_k|z_m)_o^{\rho} \leq L (z_k|z_m)_o^{d'} + L\kappa + C + H ,\]
and $d'(o,z_k) \to \infty$ if and only if $\rho(o,z_k) \to \infty$. Hence $(z_k)$ is a Gromov sequence for $d'$ if and only if it is one for $\rho$, and applying the same estimate to the pairs $(z_k,w_k)$ shows that two Gromov sequences are equivalent for $d'$ if and only if they are equivalent for $\rho$. The remaining claims now follow from the corresponding classical facts for $(\mathbb{B}^n,\rho)$ recalled in Section \ref{ss:metrics}.

<!-- proposition-4: body_md -->
\label{prop:hypchain}
Let $G \equiv I$ and $r\geq r_1$. Then $d_r^h \geq 2^{n-1}c_n \rho$ everywhere and, whenever $\rho(x,y) \geq \log 4$,
\[ d_r^h(x,y) = 2^{n-1}c_n \rho(x,y).\]
<!-- end proposition-4 -->
<!-- proof-13: body_md -->
For $G\equiv I$, we have $d_G = \Phi \circ \rho$. Every step of the chain gives 
\[ \psi_h ( \Phi (\rho_i)) \geq 2^{n-1}c_n \Phi^{-1} ( \Phi ( \rho_i)) = 2^{n-1}c_n \rho_i,\]
so every chain has sum at least $2^{n-1}c_n \sum_i \rho_i \geq 2^{n-1}c_n \rho(x,y)$. For the reverse inequality, first note that $\psi_h(\Phi (s)) = 2^{n-1}c_n s$ as soon as $2^{n-1}c_n s \geq \Phi(s)/2$. By Lemma \ref{lem:roughhyp}, this holds for $s\geq \log 4$. Put $s_r = \Phi^{-1}(r)$. As $\Phi(2\log 4) \leq 3\cdot 2^{n-1}c_n \log 4 \leq r$, we have $s_r \geq 2\log 4$. Let $t = \rho(x,y) \geq \log 4$. If $t\leq s_r$, the one-step chain gives $\psi_h(\Phi(t)) = 2^{n-1}c_n t$. If $t>s_r$, take $m = \lceil t/s_r \rceil \leq 2t/s_r$ equal pieces along the $\rho$-geodesic of length $\ell = t/m \in [s_r/2,s_r] \subset [\log 4, s_r]$. Each is an admissible step yielding $2^{n-1}c_n \ell$ and the total is $2^{n-1}c_n t$.

<!-- proposition-5: body_md -->
\label{prop:notrough}
Let $G\equiv I$, so $d_G = \mu_{\mathbb{B}^n} = \Phi \circ \rho$, and write $\Phi(t) = 2^{n-1}c_n t + E(t)$, where $0\leq E \leq 2^{n-1}c_n\log 4$ by Lemma \ref{lem:roughhyp}. For $r\geq r_1$, put $s_r = \Phi^{-1}(r)$ and $e_r: = \inf_{0<s\leq s_r} E(s)/s \geq 0$. Then
\[ d_r \geq (2^{n-1}c_n + e_r) \rho, \quad \text{while} \quad \mu_{\mathbb{B}^n} \leq 2^{n-1}c_n (\rho + \log 4).\]
Consequently, if $E>0$ on $(0,s_r]$, then $e_r>0$ and $d_r - \mu_{\mathbb{B}^n} \to \infty$ along any sequence with $\rho \to \infty$. Thus the metrics $d_r$ and $d_G$, which are always quasi-isometric, are not roughly isometric whenever the lower bound in \eqref{eq:cn} is strict at $s = 1/\tanh(t/2)$ for all $0<t\leq s_r$.
<!-- end proposition-5 -->
<!-- proof-14: body_md -->
To start, we have $d_G(z_i,z_{i+1}) \leq r$ if and only if $\rho_i := \rho(z_i,z_{i+1}) \leq \Phi^{-1}(r) = s_r$. This justifies taking the infimum of $E(s)/s$ over only $(0,s_r]$. A chain from $x$ to $y$ with $\rho$-steps $\rho_i \leq s_r$ yields 
\[ \sum_i \Phi (\rho_i) = \sum_i (2^{n-1}c_n + E(\rho_i) / \rho_i ) \rho_i \geq (2^{n-1}c_n + e_r) \sum_i \rho_i \geq (2^{n-1}c_n + e_r) \rho(x,y).\]
Taking the infimum over chains gives $d_r \geq (2^{n-1}c_n + e_r)\rho$.

To justify why $e_r >0$, write $E(s)/s = \Phi(s)/s - 2^{n-1}c_n$. As $\Phi(s)$ is comparable to $(\log (1/s) )^{1-n}$ as $s\to 0$, we have $\Phi(s) / s \to \infty$ and $E(s)/s > 2^{n-1}c_n$ for $s<\epsilon$ for some $\epsilon >0$. On $[\epsilon, s_r]$, $E(s)/s$ is continuous and positive by the hypotheses. We conclude that $e_r >0$. Finally,
\[ d_r - \mu_{\mathbb{B}^n} \geq (2^{n-1}c_n + e_r) \rho - 2^{n-1}c_n \rho - 2^{n-1}c_n \log 4 = e_r \rho - 2^{n-1}c_n \log 4 \to \infty\]
as $\rho \to \infty$, so $d_r$ and $\mu_{\mathbb{B}^n}$ are not roughly isometric.

<!-- proposition-7: body_md -->
\label{prop:hmm}
In the bounded case, either $f^m \to x_0 \in \mathbb{B}^n$ locally uniformly, where $x_0$ is the unique fixed point of $f$ in $\mathbb{B}^n$ and is attracting or super-attracting, or $f$ is a uniformly quasiconformal homeomorphism of $\mathbb{B}^n$ onto itself, a $d$-isometry, and some subsequence $f^{m_j} \to \operatorname{id}$ locally uniformly.
<!-- end proposition-7 -->
<!-- proof-16: body_md -->
Every locally uniform limit of a subsequence of $(f^m)$ maps $\mathbb{B}^n$ into $\mathbb{B}^n$ as the orbits are relatively compact. Thus the work of \cite{HMM} applies. If some limit is a constant $x_0$, then $x_0 \in \mathbb{B}^n$ is a (super-)attracting fixed point by \cite[Proposition 4.6]{HMM} and $f^m\to x_0$ locally uniformly. This rules out the existence of a second fixed point in $\mathbb{B}^n$.

If all limits are non-constant, then \cite[Proposition 4.9]{HMM} shows that $f$ is a homeomorphism of $\mathbb{B}^n$ onto itself and some $f^{m_j} \to \operatorname{id}$. This latter fact is because the limit functions form a compact abelian group, which is a Lie group by \cite[Theorem 2.1]{Mar02}. In the second case, $f$ is a $d_G$-isometry by Theorem \ref{thm:sp}~(a) and a $d$-isometry by Lemma \ref{lem:chainprop}~(d).

<!-- proposition-8: body_md -->
\label{prop:escape}
Assume that $\operatorname{dist} ( f^m(x) , \mathbb{S}^{n-1} ) \to 0$ as $m\to \infty$. Then:
\begin{enumerate}[(a)]
\item $a_m \to \infty$ and $|x_m - x_{m+1}| \to 0$.
\item For every $z\in \mathbb{B}^n$, $|f^m(x) - f^m(z)| \to 0$.
\item The set $\Lambda$ of Euclidean accumulation points of $(x_m)$ is a non-empty continuum in $\mathbb{S}^{n-1}$, and it is the same for every orbit.
\item We have $\operatorname{dist} ( f^m(E) , \Lambda) \to 0$ for every $d$-bounded set $E$. In particular, every locally uniform limit of a subsequence of $(f^m)$ is a constant in $\Lambda$, and every point of $\Lambda$ arises in this way. 
\item $s = \lim_k s_k$ exists, and $0\leq s \leq d(x,f(x))$.
\end{enumerate}
<!-- end proposition-8 -->
<!-- proof-17: body_md -->
For~(a), we have $a_m\to \infty$ by Lemma \ref{lem:chainprop}~(b). Since $s_m\leq s_0$, Lemma \ref{lem:chainprop}~(c) gives $\rho(x_m,x_{m+1}) \leq R_0:= 2^{2-n}\lambda^ns_0/c_n$. A hyperbolic ball of fixed radius centred at points tending to $\mathbb{S}^{n-1}$ has Euclidean diameter tending to $0$, so $|x_m - x_{m+1}| \to 0$. For~(b), we run the same argument with $d(f^m(x),f^m(z)) \leq d(x,z)$.

For~(c), a sequence in a compact metric space with $|x_m-x_{m+1}| \to 0$ has a connected set of accumulation points: if the accumulation set were split by two disjoint open sets, the sequence would eventually have to jump between them across a gap of fixed positive width. It is non-empty by the compactness of $\overline{\mathbb{B}^n}$ and lies in $\mathbb{S}^{n-1}$ by part~(a). For~(d), if $E$ is $d$-bounded, then $E \subset\overline{B_d(x,R)}$ for some $R>0$ and $f^m(E) \subset \overline{B_d(x_m,R)}$. The Euclidean diameter of this latter ball tends to $0$ as in part~(a). Finally, part~(e) follows from Theorem \ref{thm:sp} and Lemma \ref{lem:chainprop}~(d).

<!-- proposition-9: body_md -->
\label{prop:horo}
Assume $\operatorname{dist} (f^m(x) ,\mathbb{S}^{n-1} ) \to 0$. There is a $1$-Lipschitz function $h:\mathbb{B}^n \to \mathbb{R}$ with $h(x) = 0$, a locally uniform limit of functions $z\mapsto d(z,y_i) - d(x,y_i)$ with $\operatorname{dist} ( y_i , \mathbb{S}^{n-1}) \to 0$ (a horofunction of $(\mathbb{B}^n,d)$), such that $h(f^m(z)) \leq d(x,z)$ for all $z\in \mathbb{B}^n$ and $m\geq 0$. In particular, $h(f^m(x)) \leq 0$.
<!-- end proposition-9 -->
<!-- proof-18: body_md -->
Since $a_m\to \infty$, the set $\{ m : a_m > \max _{j<m} a_j \}$ is infinite. Enumerate the set by $m_1<m_2 <\ldots$. Then for $1\leq k\leq m_i$,
\begin{equation}
\label{eq:horo1}
a_{m_i-k} \leq a_{m_i}.
\end{equation}
Put $y_i = x_{m_i}$ and $b_i(z) = d(z,y_i) - d(x,y_i)$. Each $b_i$ is $1$-Lipschitz with $|b_i(z)| \leq d(x,z)$, so by Arzel\`a-Ascoli and properness a subsequence converges locally uniformly to some $h$ with the stated properties. For fixed $m$ and $i$ large, non-expansion and \eqref{eq:horo1} gives
\[ b_i(f^m(z)) = d(f^m(z) , f^m(x_{m_i - m}) - a_{m_i} \leq d(z,x_{m_i-m}) - a_{m_i} \leq d(z,x) + a_{m_i-m} - a_{m_i} \leq d(x,z).\]
Now let $i\to \infty$ to complete the proof.

The metric claims are Lemma \ref{lem:chainprop} and Theorem \ref{thm:gromovchain}, the non-expansion is Theorem \ref{thm:sp} with Lemma \ref{lem:chainprop}~(d), the dichotomy is Proposition \ref{prop:dicho}, the bounded case is Proposition \ref{prop:hmm}, and in the escaping case the claims after "moreover" are Propositions \ref{prop:escape} and \ref{prop:horo}. The key inequality is a restatement of \eqref{eq:horo1} in terms of Gromov products based at $x$. For $1\leq k \leq m_i$,
\begin{equation}
\label{eq:dw1}
2 ( x_k | y_i )_x = a_k +a_{m_i} - d(x_k , y_i) \geq a_k+a_{m_i} - a_{m_i-k} \geq a_k,
\end{equation}
where the first inequality is non-expansion, $d(x_k,x_{m_i}) = d(f^k(x),f^k(x_{m_i-k})) \leq d(x,x_{m_i-k})$. For $k,m\geq 1$, choose $i$ with $m_i \geq \max \{ k,m\}$. By Theorem \ref{thm:gromovchain} and \eqref{eq:dw1},
\[ (x_k|x_m)_x \geq \min \{ (x_k|y_i)_x , (y_i|x_m)_x \} - \delta \geq \tfrac12 \min \{ a_k,a_m \} - \delta \to \infty \]

<!-- lemma-9: body_md -->
\label{lem:muupper}
For every proper subdomain $U$ of $\mathbb{S}^n$ with $\operatorname{cap}\partial U >0$ we have $\mu_U \leq \hat{b}_1 \hat{k}_U + \hat{b}_2$, where $\hat{b}_1 = 4\Phi(\log 3)$ and $\hat{b}_2 = 2\Phi(\log 3)$ depend only on $n$.
<!-- end lemma-9 -->
<!-- proof-20: body_md -->
We first show that $\mu_U(x,y) \leq b:=\Phi(\log 3)$ whenever $\hat{k}_U(x,y) \leq 1/4$. Rotations of $\mathbb{S}^n$ are M\"obius maps preserving $\sigma$, hence preserving $\mu_U$ and $\hat{k}_U$, so replacing $U$ by its image under a rotation we may assume $x = 0$. Put $\delta = \sigma(0,\partial U)$; then $0<\delta<\pi$, as $\delta = \pi$ would force $\partial U = \{\infty\}$, contradicting $\operatorname{cap}\partial U >0$. For $z$ in the spherical ball $\{ \sigma(0,z) < \delta/2 \}$ we have $\sigma(z,\partial U) \leq \sigma(z,0) + \delta \leq 3\delta/2$, so every path from $0$ leaving this ball has $\hat{k}_U$-length at least $(\delta/2)/(3\delta/2) = 1/3 > 1/4$. Hence $\sigma(0,y) < \delta/2$. Now the spherical ball $\{ \sigma(0,z) < t \}$, for $0<t<\pi$, is the Euclidean ball $B(0,r_t)$ with $r_t = \tan(t/2)$, and $B(0,r_{\delta}) \subset U$ by the definition of $\delta$. By monotonicity and conformal invariance of $\mu$,
\[ \mu_U(0,y) \leq \mu_{B(0,r_\delta)}(0,y) = \Phi ( \rho_{B(0,r_\delta)}(0,y)) = \Phi \left ( \log \frac{ r_\delta + |y| }{ r_\delta - |y| } \right ) \leq \Phi(\log 3),\]
since $|y| \leq r_{\delta/2} \leq r_\delta / 2$; the last inequality is $\tan(\delta/4)/\tan(\delta/2) = \tfrac12 (1-\tan^2(\delta/4)) \leq \tfrac12$.

For general $x,y$ put $t = \hat{k}_U(x,y)$ and choose a path from $x$ to $y$ of $\hat{k}_U$-length at most $t + 1/8$. Subdividing it into $m\leq 4t+2$ pieces of $\hat{k}_U$-length at most $1/4$ and using the triangle inequality for $\mu_U$ gives $\mu_U(x,y) \leq mb \leq 4bt+2b = \hat{b}_1 t + \hat{b}_2$.

<!-- lemma-10: body_md -->
\label{lem:transfer}
Let $U$ be a proper subdomain of $\mathbb{S}^n$ with $\partial U$ $\alpha$-uniformly perfect. Then $\mathbb{S}^n\setminus U$ is $\alpha'$-uniformly perfect with $\alpha' = \min\{ \alpha/3, 1/8\}$.
<!-- end lemma-10 -->
<!-- proof-21: body_md -->
Write $E' = \mathbb{S}^n\setminus U$, a closed set containing $\partial U$, and let $x\in E'$ and $0<r<\operatorname{diam}_\sigma E'$; put $A = \{ y : \alpha' r \leq \sigma(x,y) \leq r\}$, which is non-empty as $r < \pi$. Let $t = \sigma(x,\partial U)$, attained at $x'\in\partial U$. We first note that $\overline{B}_\sigma(x,t) \subset E'$: if $y\in U$ with $\sigma(x,y)\leq t$, the last point $z$ of a spherical geodesic from $x$ to $y$ that lies in $E'$ belongs to $\partial U$ and satisfies $\sigma(x,z) < \sigma(x,y) \leq t$, a contradiction. 

If $r\leq t$, then $A \subset \overline{B}_\sigma(x,t) \subset E'$. If $\alpha' r \leq t < r$, then $x'\in A\cap E'$. So assume $t<\alpha' r$ and put $s = r-t \in ((1-\alpha')r, r)$.

If $s < \operatorname{diam}_\sigma \partial U$, uniform perfectness of $\partial U$ at $x'$ gives $y\in\partial U$ with $\alpha s \leq \sigma(x',y) \leq s$; then $\sigma(x,y) \leq t+s = r$ and $\sigma(x,y) \geq \alpha s - t \geq \alpha(1-\alpha')r - \alpha' r \geq (\alpha - 2\alpha')r \geq \alpha' r$, so $y\in A\cap E'$.

If $s \geq \operatorname{diam}_\sigma\partial U =: D$, then $\partial U\subset \overline{B}_\sigma(x',D)$, and the connected set $\mathbb{S}^n\setminus \overline{B}_\sigma(x',D)$ (recall $n\geq 2$ and $D\leq s<\pi$) is disjoint from $\partial U$, hence contained in $U$ or in $E'$. In the latter case $E'$, being closed, contains $\mathbb{S}^n\setminus B_\sigma(x',D)$; pick $y$ with $\sigma(x,y) = r$ and $\sigma(x',y)\geq D$, which exists since the sphere $\{ y : \sigma(x,y) = r\}$ contains points at distance $\min\{ r+t , 2\pi - r - t\} \geq r-t = s \geq D$ from $x'$; then $y\in A\cap E'$. In the former case $E'\subset \overline{B}_\sigma(x',D)$, so $r<\operatorname{diam}_\sigma E' \leq 2D$; choose $y\in\partial U$ with $\sigma(x',y)\geq D/2$. Then $\sigma(x,y) \leq D+t \leq r$ and $\sigma(x,y) \geq D/2 - t > r/4 - \alpha' r \geq \alpha' r$, as $\alpha'\leq 1/8$. So $y\in A\cap E'$ in this case too.

<!-- lemma-11: body_md -->
\label{lem:mucomp}
If $U$ is uniform with uniformly perfect boundary, then $\operatorname{cap}\partial U>0$, so $\mu_U$ is a metric, and there is a constant $\hat{c}_1>0$, depending only on $n$, the uniformity constant of $U$, the uniform perfectness constant of $\partial U$ and the spherical diameter of $\partial U$, such that
\[ \hat{c}_1 \hat{k}_U \leq \mu_U \leq \hat{b}_1 \hat{k}_U +\hat{b}_2.\]
<!-- end lemma-11 -->
<!-- proof-22: body_md -->
Uniformly perfect sets have positive capacity by \cite[Theorem 4.1 and Corollary 4.2]{JV96}, and the upper bound is Lemma \ref{lem:muupper}. For the lower bound, by Lemma \ref{lem:transfer} the closed set $E = \mathbb{S}^n\setminus U$ is uniformly perfect, with constant depending only on that of $\partial U$, and $\operatorname{diam}_\sigma E \geq \operatorname{diam}_\sigma \partial U$. By \cite[Theorem 4.4]{JV96}, $E$ satisfies Vuorinen's metric thickness condition \cite[(2.21)]{Vuo87} with parameters $r_0 = \tfrac14 q(E) \geq \tfrac14 q(\partial U)$ and $\delta = \delta(n,\alpha')$. Hence \cite[Theorem 2.27]{Vuo87} gives $\mu_U \geq a_1 j_U$ with $a_1 = a_1(n,r_0,\delta)$, where $j_U$ is defined as $\hat{\jmath}_U$ but with the chordal metric; since $q\leq\sigma\leq\tfrac{\pi}{2}q$ and $\log(1+\lambda u) \leq \lambda\log(1+u)$ for $\lambda\geq 1$, see \cite[(2.12)]{BHK}, we have $\hat{\jmath}_U \leq \tfrac{\pi}{2} j_U$. Finally, in an $A$-uniform metric space the Gehring--Osgood inequality $\hat{k}_U \leq 4A \hat{\jmath}_U$ holds, see \cite[Lemma 2.13, (2.16)]{BHK}. Hence $\hat{c}_1 = a_1/(2\pi A)$ works.

\label{lem:unifgromov}
Let $U$ be uniform. Then $(U,\hat{k}_U)$ is a proper geodesic Gromov hyperbolic space, and every Gromov sequence $(z_k)$ for $\hat{k}_U$ converges in $\mathbb{S}^n$ to a point of $\partial U$, equivalent Gromov sequences having the same limit. <!-- end lemma-12 -->
<!-- proof-23: body_md -->
The last assertion is the elementary inequality $\log ( 1 + \sigma(x,y)/\min\{\sigma(x,\partial U),\sigma(y,\partial U)\}) \leq \hat{k}_U(x,y)$ of \cite[(2.4)]{BHK}.

That $(U,\hat{k}_U)$ is proper, geodesic and Gromov hyperbolic is the first assertion of \cite[Theorem 3.6]{BHK} applied to the uniform space $(U,\sigma)$, which requires no boundedness; see \cite[Chapter 7]{BHK} for domains in $\mathbb{S}^n$. As $(U,\sigma)$ is bounded, \cite[Proposition 3.12 (b),(c),(d)]{BHK} shows that every $\hat{k}_U$-geodesic ray has finite $\sigma$-length and a unique endpoint in $\partial U$, and that ray $\mapsto$ endpoint is a bijection from the Gromov boundary onto $\partial U$. Given a Gromov sequence $(z_k)$ with base point $o$, it defines a point $\xi$ of the Gromov boundary; let $\gamma$ be a geodesic ray from $o$ to $\xi$ with endpoint $w\in\partial U$. The geodesic segments $[o,z_k]$ subconverge to a ray equivalent to $\gamma$, so for large $k$ there are $s_k \to \infty$ with $\hat{k}_U(z_k,\gamma(s_k)) \leq M$, where $M$ depends only on the hyperbolicity constant. Since $\gamma(s) \to w$ we have $\sigma(\gamma(s_k),\partial U) \to 0$, and the estimate of the first paragraph gives $\sigma(z_k,\gamma(s_k)) \to 0$, so $z_k \to w$. Two equivalent Gromov sequences define the same $\xi$, hence the same $w$.

\label{lem:strip}
Let $h>0$. There exists $m_0 = m_0(h)$ such that for every integer $m\geq m_0$ the following holds. Put $a=m^{-1/2}$, $y_3 = \log m/(m-1)$, $y_c = \log m/(2(m-1))$, and let $\beta \in C^{\infty}(\mathbb{R})$ satisfy $0\leq \beta \leq 1$, $\beta = 1$ on $(-\infty, y_3]$, $\beta = 0$ on $[h/2,\infty)$, and $|\beta'| \leq 4/h$. On the strip $T = \{ z \in \mathbb{C} : 0<\operatorname{Im} z < h \}$ define
\[ w(z) = z - i \operatorname{Log} ( 1 + a\beta ( \operatorname{Im} z ) e^{i(m-1)z} ).\]
Then:
\begin{enumerate}[(i)]
\item $w$ is $C^{\infty}$ on $T$, $w(z+2\pi) = w(z) + 2\pi$, and $w(z) = z$ for $\operatorname{Im} z \geq h/2$;
\item $\operatorname{Im} w(z) \geq \operatorname{Im} z - \log (1+a) > -a$;
\item $w$ is $K_m$-quasiregular on $T$ with $K_m \to 1$ as $m\to\infty$;
\item $w$ is holomorphic on $\{ \operatorname{Im} z \leq y_3 \}$ and has critical points there, all on the line $\operatorname{Im} z = y_c$, each of local index $2$; in particular $w$ is not injective.
\end{enumerate}
Let $Z_c = \{ z_c \}$ be the set of critical points of $w$, put 
\[ \epsilon := \tfrac12 \min \{ y_c , \pi/(m-1) \} ,\]
and define $\psi : T \to T$ by
\[ \psi ( z_c + re^{i\vartheta} ) = z_c + \sqrt{\epsilon r}\, e^{i\vartheta} \quad (z_c \in Z_c,\ 0\leq r \leq \epsilon), \qquad \psi = \operatorname{id} \text{ elsewhere},\]
and $W := w \circ \psi$. Then, after decreasing $\epsilon$ if necessary (depending on $h$ and $m$ only):
\begin{enumerate}[(i)]
\setcounter{enumi}{4}
\item $\psi$ is a $2$-quasiconformal homeomorphism of $T$ onto itself commuting with $z\mapsto z+2\pi$, and $W$ is $2K_m$-quasiregular on $T$ with $W = w$ off the disks $D(z_c,\epsilon)$; in particular (i) and (ii) hold with $W$ in place of $w$;
\item there are constants $L$ and $c_0>0$, depending only on $h$ and $m$, such that $|DW| \leq L$ and $J_W \geq c_0$ almost everywhere on $T$;
\item the branch set of $W$ is $Z_c$, each $z_c$ has local index $2$, and near $z_c$ the map $W$ is a winding map: $W(z_c + re^{i\vartheta}) = w(z_c) + \epsilon r e^{2i\vartheta} g_c ( \psi (z_c + re^{i\vartheta}))$ with $g_c$ holomorphic and non-vanishing on $D(z_c,\epsilon)$. \end{enumerate}

\[ \operatorname{Im} w = y - \operatorname{Re} \operatorname{Log} (1+u) \geq y - \log (1+|u|).\]

\[ w'(z) = 1+ \frac{ (m-1)aE}{1+aE} = \frac{ 1+maE}{1+aE},\]

\[ w'(\zeta) = (\zeta - z_c) \left ( 2 g_c(\zeta) + (\zeta - z_c) g_c'(\zeta) \right ), \qquad g_- |\zeta - z_c| \leq |w'(\zeta)| \leq (2g_+ + \epsilon g_+')|\zeta - z_c| .\] Since $DW(z) = Dw(\psi(z)) D\psi(z)$ and $|D\psi(z)| = \sqrt{\epsilon/r}$,

\[ w'(z) = \frac{1+maE(z)}{1+aE(z)}, \qquad E(z) = e^{i(m-1)z}, \quad |E(z)| = e^{-(m-1)\operatorname{Im} z} \leq 1,\]
so $|w'| \leq (1+ma)/(1-a)$ is bounded, while $1+maE$ vanishes exactly on $Z_c$. Put $\delta = \epsilon/2$. For $\operatorname{Im} z \leq y_c - \delta$ we have $ma|E(z)| \geq ma\, e^{-(m-1)(y_c-\delta)} = e^{(m-1)\delta}$, using $ma\,e^{-(m-1)y_c} = 1$, so $|1+maE| \geq e^{(m-1)\delta} - 1$; for $\operatorname{Im} z \geq y_c + \delta$ we have $|1+maE| \geq 1 - e^{-(m-1)\delta}$; and on the set $\{ y_c - \delta \leq \operatorname{Im} z \leq y_c + \delta,\ 0\leq \operatorname{Re} z \leq 2\pi \} \setminus \bigcup_{z_c} D(z_c,\epsilon)$, which is compact and contains no zero of $1+maE$, the continuous function $|1+maE|$ has a positive minimum. Hence $|w'|$, and so $J_w = |w'|^2$, is bounded below off the disks, and~(vi) follows with $L$ and $c_0$ the worst of the constants above.

For~(vii), $W$ is a local homeomorphism wherever it is $C^1$ with $J_W>0$, which by the above is everywhere except at the points $z_c$; near $z_c$ the formula shows that $W - w(z_c)$ is the winding map $re^{i\vartheta} \mapsto \epsilon r e^{2i\vartheta}$ multiplied by the non-vanishing continuous factor $g_c\circ\psi$, so $W$ has local index $2$ at $z_c$ and is $2$-to-$1$ on $D(z_c,\epsilon)\setminus\{z_c\}$.

\label{prop:hypex}
Let $n\geq 2$ and $\theta >1$. Choose $s \in (0,1-\frac{1}{\theta} )$, put $h = 1 - \frac{1}{\theta} - s$, and let $m \geq m_0(h)$ be an integer with $m^{-1/2} < s$. Let $W$ be the map from Lemma \ref{lem:strip} for these $h,m$. Define, on $S = \{0<x_n<1 \} \subset \mathbb{H}^n$,
\[ A(x) = \left ( x_1 , x'' , \frac{1}{\theta} +s+hx_n \right ), \quad \Omega(x) = \left ( \operatorname{Re} W(z) , x'' , \operatorname{Im} W(z) + \frac{1}{\theta} + s \right ), \]
with $z = x_1 + i(x_n - \frac{1}{\theta} -s )$, and
\[ f(x) = \left \{ \begin {array}{cc} \theta x, & x_n \geq 1,\\ \theta \Omega(A(x)), & 0<x_n<1 . \end{array} \right . \]
Then $f:\mathbb{H}^n \to \mathbb{H}^n$ is a non-injective uniformly quasiregular map, $f$ has no fixed point in $\mathbb{H}^n$, and $f^k \to \infty$ locally uniformly on $\mathbb{H}^n$. In the ball model, $f$ is the escaping case of Theorem \ref{thm:dw} with Denjoy-Wolff point the image of $\infty$, the hyperbolic M\"obius map $x\mapsto\theta x$ as conformal model, and of positive-step type: $s_k \geq \tfrac12 \lambda^{-n} 2^{n-1}c_n \log \theta > 0$ for all $k\geq 1$, hence for all $k$ by monotonicity.

Write $V = \{x_n>1\}$, $g(x) = \theta x$, and $T' = A(S) = \{ \frac{1}{\theta} +s < x_n <1 \}$. For $x\in T'$, $z$ lies in the strip $T$ of Lemma \ref{lem:strip} since $0<x_n - \frac{1}{\theta} -s < h$.

As $A$ is the identity on $\{x_n=1\}$, and as for $x\in T'$ with $x_n \geq \frac{1}{\theta} + s +h/2$ we have $\operatorname{Im} z \geq h/2$, it follows that $W(z)=z$ and $\Omega(x)=x$. Hence on $\{\tfrac12 \leq x_n < 1 \}$ we have $f = \theta A$, which equals $\theta x$ at $x_n=1$. We conclude that $f$ is continuous on $\mathbb{H}^n$.

For $x\in S$, Lemma \ref{lem:strip}~(ii) gives $\Omega(A(x))_n = \operatorname{Im} W(z) + \frac{1}{\theta} +s > \frac{1}{\theta} +s-a > \frac{1}{\theta}$, since $a=m^{-1/2} < s$. So $f(S) \subset \{x_n >1 \} = V$ and $f(\overline{V}) = \{x_n \geq \theta \} \subset V$. In particular, $f(\mathbb{H}^n) \subset V \subset \mathbb{H} ^n$.

To show that $f$ is quasiregular, consider first $\Omega$ on $T'$. Writing $z = x_1 + i(x_n - \frac{1}{\theta} - s)$ and $W = W_1 + iW_2$, we have $\Omega(x) = (W_1 , x'', W_2 + \frac{1}{\theta}+s)$, so $D\Omega$ is, after permuting coordinates, the block matrix $\operatorname{diag}(DW, I_{n-2})$. Hence $|D\Omega| = \max \{ |DW|, 1\}$ and $J_{\Omega} = J_W$, and by Lemma \ref{lem:strip}~(vi),
\[ |D\Omega|^n \leq \max\{L,1\}^n \leq \frac{ \max\{L,1\}^n }{c_0} J_{\Omega} \quad \text{a.e. on } T'.\]
Moreover $\Omega$ is locally Lipschitz (since $W$ is), hence in $W^{1,n}_{\rm loc}(T')$, and $J_\Omega \geq c_0 >0$ a.e. Therefore $\Omega$ is $K_{\Omega}$-quasiregular on $T'$ with $K_{\Omega} = \max\{L,1\}^n/c_0$. This is the only place where Lemma \ref{lem:strip}~(vi) is used; see the remark following that lemma for why it is needed. As $A$ is an affine bijection of $S$ onto $T'$ with constant dilatation and $x\mapsto \theta x$ is conformal, $f|_S = \theta\, \Omega\circ A$ is quasiregular on $S$ with dilatation bounded in terms of $K_\Omega$ and $K(A)$.

Next we glue across the hyperplane $\Pi = \{ x_n = 1\}$. On $\{ \tfrac12 \leq x_n < 1 \}$ we have $f = \theta A$, and on $\{ x_n \geq 1\}$ we have $f = \theta\,\operatorname{id}$; these are affine maps agreeing on $\Pi$, so $f$ is locally Lipschitz on the open set $\{ \tfrac12 < x_n < \tfrac32 \}$. Hence $f \in W^{1,n}_{\rm loc}(\mathbb{H}^n)$, and since $\Pi$ has measure zero the inequality $|Df|^n \leq K J_f$ holds a.e. on $\mathbb{H}^n$ with $K$ the larger of the dilatations on $S$ and on $V$ (the latter being $1$). Thus $f:\mathbb{H}^n \to \mathbb{H}^n$ is $K$-quasiregular. 

That $f$ is non-injective follows from Lemma \ref{lem:strip}~(vii): $W$ is $2$-to-$1$ near each $z_c$, and these points correspond to points of $S$, namely those $x$ with $A(x)_n - \frac{1}{\theta} - s = y_c$, i.e. $x_n = y_c/h \in (0,1)$; as $A$ and $x\mapsto\theta x$ are injective, $f$ is $2$-to-$1$ near such points. The branch set of $f$ contains $A^{-1}(\{ z_c\} \times \mathbb{R}^{n-2})$, a union of $(n-2)$-planes, and the local behaviour of $f$ there is that of a winding map about an $(n-2)$-plane.

Since $f(\mathbb{H}^n) \subset \overline{V}$ and $f = g$ on $\overline{V}$ with $g(\overline{V}) \subset V$, we have $f^k = g^{k-1}\circ f$ for every $k\geq 1$; as $g$ is conformal, $K(f^k) = K(f)$ and $f$ is uniformly quasiregular. Every orbit passes through the non-conformal region $S$ at most once, at time $0$, which is the whole content of the conformal trap. Consequently $f^k(x) = \theta^{k-1} f(x)$ for all $x$, and since $f$ maps compact subsets of $\mathbb{H}^n$ to compact subsets of $\mathbb{H}^n$, $f^k \to \infty$ locally uniformly on $\mathbb{H}^n$. There are no fixed points: on $V$, $\theta x = x$ forces $x = 0 \notin \mathbb{H}^n$, and on $S$ we have $f(x) \in V$ while $x\notin V$.

Finally we compute the step type. Let $G$ be an invariant conformal structure for $f$, $\lambda = H^{1/2}$ the constant of \eqref{eq:vcomp}, and $d$ the chain metric, transported to $\mathbb{H}^n$ by a M\"obius map; Lemma \ref{lem:chainprop}~(c) gives $d \geq \tfrac12 \lambda^{-n} 2^{n-1} c_n \rho$, with $\rho$ the hyperbolic metric of $\mathbb{H}^n$. For $k\geq 1$ we have $x_{k+1} = \theta x_k$, and for every $y\in\mathbb{H}^n$, $\rho(y,\theta y) \geq \log \theta$, since $\log\theta$ is the translation length of the hyperbolic M\"obius transformation $y\mapsto \theta y$ (equality holds exactly on its axis $\{ y' = 0\}$; in general $\cosh \rho(y,\theta y) = 1 + (\theta-1)^2|y|^2/(2\theta y_n^2) \geq 1 + (\theta-1)^2/(2\theta) = \cosh \log\theta$). Hence
\[ s_k = d(x_k , \theta x_k) \geq \tfrac12 \lambda^{-n} 2^{n-1} c_n \log \theta > 0 \quad (k\geq 1),\]
so $f$ is of positive-step type. In the ball model, $f$ is a non-injective uniformly quasiregular self-map in the escaping case of Theorem \ref{thm:dw}, with Denjoy-Wolff point the image of $\infty$, and its conformal model $g$ at that point is a hyperbolic M\"obius transformation.

\label{prop:paraex}
Let $g$ be as above. Let $B'$ be any ball with $B\subset B'$, $x_1 \in \partial B'$, and $x_2,\ldots, x_d \notin \overline{B'}$, and let the balls $B_2,\ldots, B_d$ in the construction, which \cite[Theorem 5.2]{HMM} allows to be taken arbitrarily small, be chosen small enough to be disjoint from $B'$.
