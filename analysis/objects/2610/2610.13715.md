<!-- theorem-1: body_md -->
\label{thm: schmidt spitzer magnitude}
    Let $f_m$ be a Laurent polynomial as defined in \eqref{eq: def symbol function banded}, then
    \begin{align}
        \lim_{n \to\infty}\sigma\bigl(\mathbf{T}_n(f_m)\bigr) &= \bigcap_{r \in (0, \infty)} \sigma\Bigl(\mathbf{T}\bigl(f_m(r\mathbb{T})\bigr)\Bigr)\\
        &= \bigl\{ \lambda \in \mathbb{C} ~:~ |z_{m}(\lambda)| = |z_{m+1}(\lambda)| \bigr\} \label{eq: modulus definition},
    \end{align}
    where $z_i(\lambda)$ are the roots of the polynomial $z^m\bigl(f_m(z)-\lambda\bigr)$ sorted in ascending magnitude, that is, $0< |z_1(\lambda)| \leq |z_2(\lambda)| \leq \dots \leq |z_{2m}(\lambda)|$.
<!-- end theorem-1 -->
<!-- definition-1: body_md -->
\label{def: polar curve}
    We define a $C^1$ \emph{polar curve} as the set parametrised by polar coordinates such that
    \begin{equation}\label{eq: def eq of polar curve}
        p(e^{\mathrm{i}\theta}) = r(\theta) e^{\mathrm{i} \theta}, \quad \theta \in [-\pi, \pi],
    \end{equation}
    for some continuously differentiable function $r: [-\pi, \pi] \to (0, \infty)$, such that $r(-\theta) = r(\theta)$ and $r'(\pi) = r'(-\pi) =0$. For ease of notation, we will often denote the polar curve as $p(\mathbb{T})$, where $\mathbb{T}$ is the one-dimensional torus. If $r(\theta)$ is moreover real-analytic as a $2\pi$-periodic function, we call $p(\mathbb{T)}$ an \emph{analytic polar curve}.
<!-- end definition-1 -->
<!-- assumption-1: body_md -->
\label{ass:standing}
    Let $f_m$ be as in \eqref{eq: def symbol function banded}, then we assume that for every $\lambda\in\sigma_{\mathrm{open}}(f_m)$ we have strict modulus separation, that is
    \begin{equation}\label{eq:strictsep}
     |z_{m-1}(\lambda)|<|z_m(\lambda)|=|z_{m+1}(\lambda)|<|z_{m+2}(\lambda)| ,
    \end{equation}
    and that $\sigma_{\mathrm{open}}(f_m)$ is not a single point. For $m = 1$ we use the convention that $|z_0| = 0$ and $|z_{m+2}| = \infty$.
<!-- end assumption-1 -->
<!-- lemma-1: body_md -->
\cite[Lemma 2.6.]{GBZfoundation}\label{lem: separation implies non confluence}
    Suppose that Assumption \ref{ass:standing} holds. Then $f_m'(z) \neq 0$ for every $z \in \mathbf{Z}(f_m)\setminus\mathbb{R}$, and $f''_m(z) \neq 0$ for every $z \in \mathbf{Z}(f_m)\cap\mathbb{R}$ such that $f_m'(z) = 0$.
<!-- end lemma-1 -->
<!-- theorem-2: body_md -->
\cite[Theorem 2.11.]{GBZfoundation}\label{Thm: real spectra criterion}
    Suppose that $\mathbf{Z}(f_m)$ is parametrised by a $C^1$ polar curve $p(\mathbb{T})$ as in \eqref{eq: def eq of polar curve} and that Assumption \ref{ass:standing} holds. Then it holds that
    \begin{equation}\label{eq: spectral limit}
        \lim_{n\to\infty} \sigma\bigl(\mathbf{T}_n(f_m)\bigr) = \bigcup_{\alpha \in[-\pi, \pi]} f_m\bigl(p(e^{\mathrm{i}\alpha})\bigr) \subset\mathbb{R},
    \end{equation}
    where the limit is supposed to be understood in the Hausdorff sense.
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm: exact characterisation}
    Under Assumption \ref{ass:standing} it holds that
    \begin{equation}
        \sigma_{\mathrm{open}}(f_m) \subset \mathbb{R} \implies \text{GBZ is spanned by an analytic  polar curve.} 
    \end{equation}
<!-- end theorem-3 -->

Let us start by noting that since the set $\sigma_{\mathrm{open}}(f_m)$ is connected \cite{bams/1183529098}, reality forces $\sigma_{\mathrm{open}}(f_m) = [A, B]$ for $A<B$ by Assumption~\ref{ass:standing}.
    Under Assumption~\ref{ass:standing} for every $\lambda$ in the interior of $ \sigma_{\mathrm{open}}(f_m) = [A, B]$ there exists an open neighbourhood $U$ such that the roots $z_m$ and $z_{m+1}$ can be analytically extended throughout $U$.    For every $\lambda$ in the open interval $(A,B)$, the central roots are nonreal and form a conjugate pair, so let us without loss of generality assume that $\alpha(\lambda) = \arg z_m(\lambda)\in(0,\pi)$.
    Let us define the \emph{Bloch ratio} as follows
    \begin{equation}
        \Phi(\lambda) := \frac{z_m(\lambda)}{z_{m+1}(\lambda)}=e^{2\mathrm{i}\alpha(\lambda)} .
    \end{equation}
    Then by Theorem~\ref{thm: schmidt spitzer magnitude}
    \begin{equation}
        \sigma_{\mathrm{open}}(f_m)\cap U=\bigl\{\lambda\in U:\ |\Phi(\lambda)|=1\bigr\}.
    \end{equation}
    We first show that on $U\cap\sigma_{\mathrm{open}}$ it holds that $\Phi'(\lambda) \neq 0$.   Indeed if $\Phi'(\lambda)$ has a root of order $k \geq 1$ for some $\lambda^* \in \sigma_{\mathrm{open}}\cap U$, then $\{|\Phi(\lambda)|=1\}$ consists of $2(k+1)\geq 4$ analytic arcs extending from $\lambda^*$ at equal angles, but at most two of them can be real. This would imply that there is a branch of $\{|\Phi(\lambda)|=1\}$ extending into $\operatorname{\mathbb{C}}$, but by assumption $\sigma_{\mathrm{open}}(f_m) \cap U \subset \sigma_{\mathrm{open}}(f_m) \subset \mathbb{R}$.

    Next note that
    \begin{equation}
        \Phi'(\lambda) = 2 \mathrm{i} \alpha'(\lambda) \Phi(\lambda) \neq 0 \implies \alpha'(\lambda) \neq 0.
    \end{equation}
    So $\alpha(\lambda)$ is strictly monotone on $(A, B)$ as well as continuous on $[A, B]$, from which it follows that the GBZ is a polar curve as defined in \eqref{eq: def eq of polar curve}. Let us introduce the radius function as
    \begin{equation}
        r(\theta) := |z_{m}(\alpha^{-1}(\theta))|.
    \end{equation}
    This function is real-analytic on $(0, \pi)$, because $\alpha^{-1}(\theta)$ is real-analytic by the inverse function theorem and the root $z_m$ is analytic by Assumption~\ref{ass:standing}. Hence, the function $r$ is real-analytic as modulus of a non-vanishing real-analytic $\operatorname{\mathbb{C}}$-valued function on $(0, \pi)$. 

    It remains to extend $r$ real-analytically across the endpoint angles $\theta_e=\arg z_e(\lambda_e)\in\{0,\pi\}$, where $z_e$ denotes the real double root at the edge $\lambda_e\in\{A,B\}$. By the edge normal form in the proof of \cite[Thm.~2.10]{GBZfoundation}, the central roots at $\lambda=\lambda_e+\delta$ are $\mu(\delta)\pm\eta(\delta)$ with $\mu,\eta^2$ real-analytic, $\mu(0)=z_e$, and $\eta^2=\bigl(2/f_m''(z_e)\bigr)\delta\bigl(1+\mathcal{O}(\delta)\bigr)$. 
    On the spectral side the pair is non-real, so $\eta=\mathrm{i} t$ with $t\in\mathbb{R}$, and since $\tfrac{\mathrm{d}(t^2)}{\mathrm{d}\delta}(0)=-2/f_m''(z_e)\neq0$ we may invert, that is $\mu$ is real-analytic in $t^2$. Letting $t$ range over $(-\varepsilon,\varepsilon)$, the curve $u(t):=\mu(t^2)+\mathrm{i} t$ is a real-analytic two-sided parametrisation of $\mathbf{Z}(f_m)$ through $z_e$, the sign of $t$ selecting the arc above or below the axis. 
    Its angle $\theta(t)=\theta_e+\arctan\bigl(t/\mu(t^2)\bigr)$ is odd about $\theta_e$ with $\theta'(0)=1/z_e\neq0$, so $t=t(\theta)$ is odd real-analytic on a two-sided interval $I_e\ni\theta_e$, and 
    \begin{equation}
        \widehat r(\theta):=|u(t(\theta))|=\sqrt{\mu\big(t(\theta)^2\big)^2+t(\theta)^2},
    \end{equation} 
    is real-analytic on $I_e$ and, depending on $\theta$ only through $t(\theta)^2$, even about $\theta_e$. In particular $\widehat r(\theta_e)=|z_e|$ and $\widehat r'(\theta_e)=0$. Since $\lambda(\cdot)$ is strictly monotone on $(0,\pi)$ \cite[Lem.~2.8]{GBZfoundation}, each angle in $(0,\pi)$ carries exactly one central root in the upper half-plane, whence $r=\widehat r$ on the interval $I_e\cap(0,\pi)$. By evenness of $\widehat r$, the even extension $r(-\theta):=r(\theta)$ coincides with $\widehat r$ on all of $I_e$ and is therefore real-analytic across $\theta_e$, with $r'(\theta_e)=0$. 
    Performing this at both edges and extending $2\pi$-periodically, $r$ is an even, real-analytic function on the unit circle with $r'(0)=r'(\pi)=0$, so it immediately follows that the GBZ given by the set $\mathbf{Z}(f_m)$ is spanned by an analytic polar curve.

\label{cor:band function}
    Suppose Assumption~\ref{ass:standing} holds and $\sigma_{\mathrm{open}}(f_m)\subset\mathbb{R}$. Let $p(e^{\mathrm{i}\theta})=r(\theta)e^{\mathrm{i}\theta}$ be the analytic polar parametrisation of $\mathbf{Z}(f_m)$ from Theorem~\ref{thm: exact characterisation} and define the band function

        \lambda(\theta):=f_m\bigl(r(\theta)e^{\mathrm{i}\theta}\bigr), \qquad \theta\in\mathbb{R}.
    \end{equation}
    Then $\lambda$ is real-analytic on $\mathbb{R}$, strictly monotone on $(0,\pi)$, and $\lambda''(\theta_e)\neq0$ for $\theta_e\in\{0,\pi\}$.

<!-- lemma-3: body_md -->
\label{lem:trace}
For every symbol $h\in W^0$ and every $n$, $\operatorname{tr}\mathbf{T}_n(h)^2=\sum_{|k|<n}(n-|k|)\widehat h_k\widehat h_{-k}$. Consequently, for $n>m$,
\begin{equation}\label{eq:traceidentity}
 \operatorname{tr}\mathbf{T}_n(f_m)^2-\operatorname{tr}\mathbf{T}_n(g)^2 =\Delta_2-\mathcal{R}_n,
\end{equation}
where the tail $\mathcal{R}_n:=\sum_{|k|\ge n}\bigl(|k|-n\bigr)b_kb_{-k}$ satisfies $|\mathcal{R}_n|\le C_\eta e^{-2\eta n}$ for every $\eta<\gamma_F$.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
We have $\operatorname{tr}\mathbf{T}_n(h)^2=\sum_{j,l=1}^n\widehat h_{j-l}\widehat h_{l-j}$, and the number of index pairs with $j-l=k$ is $n-|k|$, which gives the first identity. It converges absolutely since $h\in W^0$. For $h=f_m$ and $n>m$ the sum is finite and equals $nc_0(f_m^2)-\sum_k|k|a_ka_{-k}$. For $h=g$ the same rearrangement gives $nc_0(g^2)-\sum_k|k|b_kb_{-k}+\mathcal{R}_n$, the tail collecting the terms with $|k|\ge n$, which are absent from the finite section. Subtracting and inserting \eqref{eq:delta2} yields the first equality in \eqref{eq:traceidentity}. The bracket vanishes since $c_0(g^2)=c_0(f_m^2)$ by \cite{GBZfoundation}. Finally, by \eqref{eq: fourier coefficients of g}, $\sum_{|k|\ge n}\bigl(|k|-n\bigr)|b_k||b_{-k}|\le C_\eta e^{-2\eta n}$ for every $\eta<\gamma_F$.

<!-- theorem-6: body_md -->
\label{thm:cinf}
Under Assumption~\ref{ass:standing} and $\sigma_{\mathrm{open}} \subset \mathbb{R}$, there is an $n_0$ such that for all $n\ge n_0$, it holds that:
\begin{itemize}
\item[(i)] For every $k\in\{1,\dots,n\}$,
\begin{equation}
 \bigl|\lambda_k-\tau_k\bigr|\ \le\ \frac{\|\lambda'\Delta\Psi\|_{L^\infty(0,\pi)}}{n} + \frac{C}{n^2}.
\end{equation}
Moreover, with $\alpha_k :=\alpha(\lambda_k)$ and uniformly in $k$, 
\begin{equation}\label{eq:secondorder}
 \lambda_k-\tau_k =\frac{\lambda'(\alpha_k)\Delta\Psi(\alpha_k)}{n} +\mathcal{O}\bigl(n^{-2}\bigr).
\end{equation}
\item[(ii)] Let us define
\begin{equation}\label{eq:cinf}
 c_\infty:=\frac1\pi\int_0^\pi\bigl|\lambda'(\alpha)\bigr| \bigl|\Delta\Psi(\alpha)\bigr|\: \mathrm{d}\alpha = \frac1\pi\int_A^B\Big|\Delta\Psi\big(\alpha(\lambda)\big)\Big|\: \mathrm{d}\lambda  \le \frac{B-A}{\pi}\|\Delta\Psi\|_\infty ,
\end{equation}
then the $\ell^1$ distance defined in \eqref{eq:dsig} satisfies
\begin{equation}
    \bigl|d_\sigma\bigl(\mathbf{T}_n(f_m),\mathbf{T}_n(g)\bigr) -c_\infty\bigr|\le Cn^{-1}.
\end{equation}
\item[(iii)] For every $n>m$ and every bijection $\varsigma$ of $\{1,\dots,n\}$,
\begin{equation}\label{eq:lowerbound}
 \sum_{k}\bigl|\lambda_k-\tau_{\varsigma(k)}\bigr|
 \ \ge\ \frac{\bigl|\Delta_2-\mathcal{R}_n\bigr|}{2\widehat R}
 \ \ge\ \frac{|\Delta_2|-C_\eta e^{-2\eta n}}{2\widehat R};
 \qquad\text{in particular}\qquad
 c_\infty\ \ge\ \frac{|\Delta_2|}{2\widehat R},
\end{equation}
where $\mathcal{R}_n$ is the tail of Lemma~\ref{lem:trace} and $\eta<\gamma_F$.
\end{itemize}
<!-- end theorem-6 -->
<!-- proof-4: body_md -->
We prove each assertion of the theorem separately: 

\begin{itemize}
    \item[(i)] By Theorem~\ref{thm:quant} applied to the banded symbol $f_m$ and Theorem~\ref{thm:quantSL} applied to the dense symbol $g$, both spectra are real and simple for $n\ge n_0$ and, in reduced form,
\begin{align}
    n\alpha(\lambda_k)+\Psi_f(\alpha(\lambda_k)) &=\pi k+\mathcal{O}(e^{-\gamma n})\label{eq: phase f} \\ n\alpha(\tau_k)+\Psi_g(\alpha(\tau_k)) &=\pi k+o(n^{-q}),\label{eq: phase g}
\end{align}
for any $q>0$, and for $k = 1, \dots, n$. Let us define $\delta_k:=\alpha(\lambda_k)-\alpha(\tau_k)$.
Subtracting \eqref{eq: phase f} and \eqref{eq: phase g}, and using that $\Psi_f$ and $\Psi_g = \Psi_f + \Delta \Psi$ are Lipschitz on the compact domain $[0, \pi]$ by Lemma~\ref{lem:H5} (iii), yields
\begin{align}\label{eq:alphadiff}
    n\bigl[\alpha(\lambda_k)-\alpha(\tau_k)\bigr]&=\Psi_g\big(\alpha(\tau_k)\big)-\Psi_f\big(\alpha(\lambda_k)\big) + o(n^{-q})\\
    &=\Psi_g\big(\alpha(\lambda_k)-\delta_k\big)-\Psi_f\big(\alpha(\lambda_k)\big) + o(n^{-q}) \\
    &=\Psi_g\big(\alpha(\lambda_k)\big)-\Psi_f\big(\alpha(\lambda_k)\big) + \mathcal{O}(n^{-1})\\
    &= \Delta\Psi\big(\alpha(\lambda_k)\big) + \mathcal{O}(n^{-1}),
\end{align}
where the above computations hold uniformly in $k$. By Lemma~\ref{cor:band function} we apply the mean value theorem to $\lambda(\cdot)$ on the segment between $\alpha_k=\alpha(\lambda_k)$ and $\alpha(\tau_k)$, which gives $\lambda_k-\tau_k=\lambda'(\eta_k)\delta_k$ for some $\eta_k$ with $|\eta_k-\alpha_k|\le|\delta_k|\le Cn^{-1}$. In particular because of Lemma~\ref{cor:band function} $\lambda'$ is Lipschitz on any compact domain, so $\lambda'(\eta_k)=\lambda'(\alpha_k)+\mathcal O(n^{-1})$, whence
\begin{equation}
 \lambda_k-\tau_k=\frac{\lambda'(\alpha_k)\Delta\Psi(\alpha_k)}{n} +\mathcal O\bigl(n^{-2}\bigr) \qquad\text{uniformly in }k,
\end{equation}
and taking the supremum over $\alpha_k\in(0,\pi)$ yields the first statement of (i). Near the edges the small factor $|\lambda'|$ only strengthens the pointwise bound.

\item[(ii)] Write $h:=|\lambda'||\Delta\Psi|$. By Lemma~\ref{cor:band function}, $h$ is Lipschitz on the closed interval
$[0,\pi]$. By (i), uniformly in $k$,
\begin{equation}\label{eq:dsigmasum}
  d_\sigma\bigl(\mathbf{T}_n(f_m),\mathbf{T}_n(g)\bigr)
  =\frac1n\sum_{k=1}^{n}h(\alpha_k)+\mathcal{O}\bigl(n^{-1}\bigr).
\end{equation}
Since $\Psi_f(0)=0$ and $\Psi_f(\pi)=\pi$, for $n>\|\Psi_f'\|_\infty$ the map
$\Phi_n=n\mathrm{id}+\Psi_f$ is an increasing bijection of $[0,\pi]$ onto $[0, (n+1)\pi]$, and by
\eqref{eq:quant} the nodes $\alpha_k=\Phi_n^{-1}(\pi k)+\mathcal{O}(e^{-\gamma n})$
have consecutive gaps $\pi/n+\mathcal{O}(n^{-2})$, uniformly in $k$. Hence
\eqref{eq:dsigmasum} is a Riemann sum for a Lipschitz integrand with mesh
$\pi/n$, and
\begin{equation}
  \frac1n\sum_{k=1}^{n}h(\alpha_k)=\frac1\pi\int_0^\pi h\: \mathrm{d}\alpha
  +\mathcal{O}\bigl(n^{-1}\bigr),
\end{equation}
which gives $\bigl|d_\sigma-c_\infty\bigr|\le Cn^{-1}$. The equality of the two integrals in \eqref{eq:cinf} is the substitution $\: \mathrm{d}\lambda=\lambda'(\alpha)\: \mathrm{d}\alpha$, and the final inequality uses $\int_0^\pi|\lambda'|\: \mathrm{d}\alpha=B-A$.

\item[(iii)] By Lemma~\ref{lem:trace} and
$\sum_k\lambda_k^2=\operatorname{tr}\mathbf{T}_n(f_m)^2$,
$\sum_k\tau_k^2=\operatorname{tr}\mathbf{T}_n(g)^2$,
\begin{equation}
    \Bigl|\sum_k\bigl(\lambda_k^2-\tau_{\varsigma(k)}^2\bigr)\Bigr| = \bigl|\Delta_2-\mathcal{R}_n\bigr|, \qquad \bigl|\lambda_k^2-\tau_{\varsigma(k)}^2\bigr|\le 2\widehat R\bigl|\lambda_k-\tau_{\varsigma(k)}\bigr|,
\end{equation}
which yields the result.
\end{itemize}

\label{lem:struct}
    Under Assumption~\ref{ass:standing} and $\sigma_{\mathrm{open}} \subset \mathbb{R}$, the function $\Delta\Psi$ is continuous on $[0,\pi]$ and $\Delta\Psi(0)=\Delta\Psi(\pi)=0$. Moreover, its odd $2\pi$-periodic extension is continuous on $\mathbb{R}$. Consequently $g_n$ is real-valued and even, so $\mathbf{T}_n(g_n)$ is a real symmetric Toeplitz matrix with,
    \begin{equation}\label{eq:additivematrix}
      \mathbf{T}_n(g_n)=\mathbf{T}_n(g)+\frac1n\mathbf{T}_n(\lambda'\Delta\Psi)+E_n, \quad \|E_n\|_2\le Cn^{-2}.
    \end{equation}

<!-- proof-5: body_md -->
Since $\Psi_f$ and $\Psi_g$ agree at $0$ and $\pi$ we have $\Delta\Psi(0)=\Delta\Psi( \pi)=0$, so the odd $2\pi$-periodic extension is continuous. Since $\Delta\Psi$ is odd and $\lambda$ is even, $\alpha\mapsto\alpha+\Delta\Psi(\alpha)/n$ is odd and $g_n(e^{-\mathrm{i}\alpha})=\lambda(-\alpha-\Delta\Psi(\alpha)/n)=g_n(e^{\mathrm{i}\alpha})$, hence $\widehat{(g_n)}_k=\widehat{(g_n)}_{-k}\in\mathbb{R}$. To show the matrix expansion in \eqref{eq:additivematrix} we may proceed as follows. Taylor expanding the phase-corrected symbol given by \eqref{eq: phase-corrected symbol} gives,
\begin{equation}\label{eq:symbolexpansion}
  g_n  =  g  +  \frac1n\lambda'\Delta\Psi  +  R_n, \qquad \|R_n\|_{L^\infty(\mathbb{T})}\le\frac{\|\lambda''\|_\infty\|\Delta\Psi\|_\infty^2}{2n^{2}} \leq Cn^{-2},
\end{equation}
where we used $\lambda(\alpha)=g(e^{\mathrm{i}\alpha})$. Since $h\mapsto\mathbf{T}_n(h)$ is linear and $\mathbf{T}_n(h)=P_nL(h)P_n$ is a compression of the Laurent operator, so that $\|\mathbf{T}_n(h)\|_2\le\|h\|_{L^\infty(\mathbb{T})}$, this transfers to the matrices giving \eqref{eq:additivematrix}.
The function $\lambda'\Delta\Psi$ is real and even, being the product of the odd functions $\lambda'$ and $\Delta\Psi$, so $\mathbf{T}_n(\lambda'\Delta\Psi)$ is again real symmetric. Both matrices in \eqref{eq:additivematrix} are real symmetric, which completes the proof.

By \Cref{thm: exact characterisation}, $\zeta(\alpha)=r(\alpha)e^{\mathrm{i}\alpha}$ and $\lambda=f_m\circ\zeta$ are analytic and bounded on $S_{4\eta}$ for some $\eta>0$.

(iii) For $\alpha\in S_{2\eta}$ the roots of $z^m(f_m(z)-\lambda(\alpha))$ are $\zeta(\pm\alpha)$, the inner roots $z_1,\dots,z_{m-1}$ and the outer roots $z_{m+2},\dots,z_{2m}$. By \eqref{eq:strictsep} and compactness, after shrinking $\eta$ we have $|z_i/\zeta|\le e^{-\gamma/2}$ and $|\zeta/z_j|\le e^{-\gamma/2}$ there. Symmetric functions of each group are analytic. Here $\operatorname{Log}$ is the principal branch of the logarithm, which is analytic on $\operatorname{Re} w>0$ and hence on all arguments below, since $|z_i/\zeta|,|\zeta/z_j|<1$. Thus
\begin{equation}
  \Theta(\alpha):=\sum_{i=1}^{m-1}\operatorname{Log}\Bigl(1-\frac{z_i}{\zeta}\Bigr)
  -\sum_{j=m+2}^{2m}\operatorname{Log}\Bigl(1-\frac{\zeta}{z_j}\Bigr)
  =-\sum_{p\ge1}\frac1p\Bigl(\sum_i\bigl(\tfrac{z_i}{\zeta}\bigr)^p-\sum_j\bigl(\tfrac{\zeta}{z_j}\bigr)^p\Bigr)
\end{equation}

is analytic and bounded on $S_{2\eta}$. For real $\alpha$ we have $\zeta(-\alpha)=\bar\zeta(\alpha)$, so $\overline{\Theta(\alpha)}=\Theta(-\alpha)$, and \eqref{eq:PsifWidom} reads $\Psi_f(\alpha)-\alpha=\tfrac{\mathrm{i}}{2}\bigl(\Theta(\alpha)-\Theta(-\alpha)\bigr)$. This gives the odd, periodic extension of $\Psi_f-\mathrm{id}$.

For $\Psi_g-\mathrm{id}=\eta_g$, write $\lambda=\Lambda\circ\cos$. Then $G_\alpha(e^{\mathrm{i}\theta})=\tfrac12\int_0^1\Lambda'\bigl((1-t)\cos\alpha+t\cos\theta\bigr)\: \mathrm{d} t$, and $\Lambda'(\cos\theta)=-\lambda'(\theta)/\sin\theta$ has a fixed sign $\sigma$ and satisfies $\sigma\Lambda'\ge 2c_1>0$ by \Cref{cor:band function}. Hence, after shrinking $\eta$, the function $(\theta,\alpha)\mapsto G_\alpha(e^{\mathrm{i}\theta})$ is analytic on $S_{2\eta}^2$ with $\operatorname{Re}\sigma G_\alpha\ge c_1$, and $\log(\sigma G_\alpha(e^{\mathrm{i}\theta})) =\sum_k c_k(\alpha)e^{\mathrm{i} k\theta}$ is analytic and bounded by some $M$ there. Then $|c_k(\alpha)|\le Me^{-2\eta|k|}$, and
\begin{equation}
  \eta_g(\alpha)=-\mathrm{i}\sum_{k}\operatorname{sgn}(k)c_k(\alpha)e^{\mathrm{i} k\alpha}
\end{equation}
converges uniformly on $S_\eta$, which gives the analytic extension. Since $G_\alpha$ is even in $\theta$ and in $\alpha$, $\eta_g$ is odd. Hence $\Delta\Psi=\eta_g-(\Psi_f-\mathrm{id})$ has the claimed properties, and Lipschitz continuity on $[0,\pi]$ follows from Cauchy's estimate.

(i) By (iii), for $n\ge n_1$ the map $\chi_n$ sends $S_{\eta/2}$ into $S_\eta$ and satisfies $\tfrac12\le\chi_n'\le\tfrac32$ on $\mathbb{R}$. Hence $g_n$ is analytic on $S_{\eta/2}$ with $|g_n|\le\|\lambda\|_{2\eta}$ there, so $|\widehat{(g_n)}_k|\le\|\lambda\|_{2\eta}e^{-\eta|k|/2}$ and $\|g_n\|_{W^\beta}$ is bounded uniformly in $n$ for every $\beta>0$. Since $\chi_n$ is odd and fixes $0$ and $\pi$, the bound $|\lambda'(\alpha)|\ge c|\sin\alpha|$ from \Cref{cor:band function} gives $|\tfrac{\: \mathrm{d}}{\: \mathrm{d}\alpha}g_n(e^{\mathrm{i}\alpha})|\ge c'|\sin\alpha|$. Moreover $\tfrac{\: \mathrm{d}^2}{\: \mathrm{d}\alpha^2}g_n(e^{\mathrm{i}\alpha})\big|_{\alpha_e}=\lambda''(\alpha_e)\chi_n'(\alpha_e)^2$ for $\alpha_e\in\{0,\pi\}$, which is bounded away from $0$. Thus $g_n\in\mathrm{SL}^\beta$ uniformly in $n$ for every $\beta>0$, and the remainder in \cite[Theorem~2.1]{BOGOYA_SLM} depends on the symbol only through these bounds.

(ii) Let $G^{(n)}_\alpha$ denote \eqref{eq:Gdef} for $g_n$. Then $G^{(n)}_\alpha(e^{\mathrm{i}\theta})-G_\alpha(e^{\mathrm{i}\theta})=\frac{d(\theta)-d(\alpha)}{2(\cos\theta-\cos\alpha)}$ with $d(\alpha):=g_n(e^{\mathrm{i}\alpha})-g(e^{\mathrm{i}\alpha})$, which is even and satisfies $\|d\|_{\eta/2}\le\|\lambda'\|_\eta\|\Delta\Psi\|_\eta/n$. Writing $d=D\circ\cos$, Cauchy's formula for the divided difference of $D$ on the ellipse $\cos(S_{\eta/2})$ bounds this difference by $C/n$ on $S_{\eta/4}^2$. Hence, for $n\ge n_1$, $\operatorname{Re}\sigma G^{(n)}_\alpha\ge c_1/2$ and the logarithms differ by at most $C/n$ there. The coefficient bound and series of (iii), applied on $S_{\eta/4}^2$, give $\|\Psi_{g_n}-\Psi_g\|_{\eta/8}\le C/n$, and in particular (ii).

By \Cref{thm:quant} for $f_m$ and by Theorem~\ref{thm:quantSL} together with Lemma~\ref{lem:H5}(i) for $g_n$, both rules hold for $n\ge n_0$ with the same uniform constants: $\lambda_k=\lambda(\theta_k)$ and $\tau^{(n)}_k=\lambda(\sigma_k)$ with $\sigma_k:=\beta_k+\Delta\Psi(\beta_k)/n$, where
    \begin{align} 
      n\theta_k+\Psi_f(\theta_k)&=\pi k+\mathcal{O}(e^{-\gamma n}),\label{eq: rule f} \\
      n\beta_k+\Psi_{g_n}(\beta_k)&=\pi k+o(n^{-q}), \label{eq: rule gn}
    \end{align}
    for any $q>0$, with exactly one solution per $k$, both enumerations sorted by the same monotone $\lambda(\cdot)$. Substituting $\sigma_k$ into the quantisation rule of $f_m$ given in \eqref{eq: rule f} yields
    \begin{equation}
        n \sigma_k + \Psi_f(\sigma_k) = n \beta_k + \Delta\Psi(\beta_k) + \Psi_f\Bigl(\beta_k + \frac{\Delta\Psi(\beta_k)}{n}\Bigr).
    \end{equation}
    Replacing $n\beta_k$ by means of the quantisation rule \eqref{eq: rule gn} and then substituting the definition of $\Delta\Psi$ from \eqref{eq:mismatch}, we obtain
    \begin{align}
        n\sigma_k+\Psi_f(\sigma_k)
        &=\pi k+o(n^{-q})
          -\Psi_{g_n}(\beta_k)+\Delta\Psi(\beta_k)
          +\Psi_f\Bigl(\beta_k+\frac{\Delta\Psi(\beta_k)}{n}\Bigr)\\
        &=\pi k+o(n^{-q})
          -\Psi_{g_n}(\beta_k)+\Psi_g(\beta_k)-\Psi_f(\beta_k)
          +\Psi_f\Bigl(\beta_k+\frac{\Delta\Psi(\beta_k)}{n}\Bigr)\\
        &=\pi k+o(n^{-q})
          +\underbrace{\bigl[\Psi_g-\Psi_{g_n}\bigr](\beta_k)
          +\bigl[\Psi_f(\cdot+\Delta\Psi/n)-\Psi_f\bigr](\beta_k)}_{=O(n^{-1})
      \text{ by Lemma~\ref{lem:H5}(ii),(iii)}}.
        \label{eq: sigma residual}
    \end{align}
    Subtracting \eqref{eq: rule f} from \eqref{eq: sigma residual} and using that $\Psi_f$ is Lipschitz with constant $L$ by Lemma~\ref{lem:H5}(iii),
    \begin{equation*}
      (n-L)|\sigma_k-\theta_k|\le Cn^{-1}+o(n^{-q}),
    \end{equation*}
    so $|\sigma_k-\theta_k|\le Cn^{-2}$ for $n\ge n_0$, uniformly in $k$. Since $\lambda$ is Lipschitz on $[0,\pi]$ by Lemma~\ref{cor:band function}, 
    \begin{align}
       |\lambda_k-\tau^{(n)}_k|&\le\|\lambda'\|_\infty|\theta_k-\sigma_k| \\
       &\le C\|\lambda'\|_\infty n^{-2},
    \end{align}
    which is \eqref{eq:pointwisecorr}. Summing over $k$ gives
    \eqref{eq:l1corr}.

\label{thm:higher}
Define the profiles at the $K$-th iteration
\begin{equation}\label{eq:picard-eps}
  \varepsilon^{(0)}:=0,\qquad
  \varepsilon^{(K+1)}:=\Psi_{g_{\varepsilon^{(K)}}}-\Psi_f\circ\Bigl(\mathrm{id}+\frac{\varepsilon^{(K)}}{n}\Bigr),
  \qquad K=0,1,2,\dots
\end{equation}
Then there is $R>0$ depending on $f_m$ only and, for every $K\in\mathbb{N}$, constants $n_K,L_K>0$ depending on $f_m$ and $K$ only, such that for $n\ge n_K$ the profiles $\varepsilon^{(0)},\dots,\varepsilon^{(K)}$ are odd, $2\pi$-periodic, analytic and bounded by $R$ on the strip $S_{\eta/2}$, and
\begin{equation}\label{eq:respicard}
 \bigl\|\rho\bigl[\varepsilon^{(K)}\bigr]\bigr\|_{\eta/2}\le\Bigl(\frac{L_K}{n}\Bigr)^{K}\|\Delta\Psi\|_\eta.
\end{equation}
Consequently, if $\tau^{(n,K)}_1\le\dots\le\tau^{(n,K)}_n$ are the eigenvalues of the real symmetric matrix $\mathbf{T}_n(g_{\varepsilon^{(K)}})$, there is $C_K$ depending on $f_m$ and $K$ only such that, enlarging $n_K$ if necessary, for $n\ge n_K$
\begin{equation}\label{eq:higher}
 \bigl|\lambda_k-\tau^{(n,K)}_k\bigr|\le\frac{C_K}{n^{K+1}}\ \text{ uniformly in }k,
 \qquad
 d_\sigma\bigl(\mathbf{T}_n(f_m),\mathbf{T}_n(g_{\varepsilon^{(K)}})\bigr)\le\frac{C_K}{n^{K}}.
\end{equation}

<!-- proof-9: body_md -->
Write $\mathcal{F}_n[\tilde\varepsilon]:=\Psi_{g_{\tilde\varepsilon}}-\Psi_f\circ(\mathrm{id}+\tilde\varepsilon/n)$, so that $\rho[\tilde\varepsilon]=\tilde\varepsilon-\mathcal{F}_n[\tilde\varepsilon]$ and the iteration reads $\varepsilon^{(K+1)}=\mathcal{F}_n[\varepsilon^{(K)}]$. For $0<s\le\eta$ let $\mathcal{E}_s$ be the odd, $2\pi$-periodic functions analytic and bounded on $S_s$, normed by $\|\cdot\|_s$ and note $\mathcal{E}_s\subset\mathcal{E}_{s'}$ with $\|\cdot\|_{s'}\le\|\cdot\|_s$ for $s'<s$. Put $R:=2\|\Delta\Psi\|_\eta$ and
\begin{equation}
  \eta_j:=\eta\bigl(\tfrac12+2^{-j-1}\bigr),\qquad j\ge0,
\end{equation}
so $\eta_0=\eta$, $\eta_j\downarrow\eta/2$ and $\eta_j-\eta_{j+1}=\eta2^{-j-2}$.

The map $\mathcal{F}_n$ preserves oddness, but it is not a self-map of a single $\mathcal{E}_s$. The Wiener--Hopf factorisation is bounded only on strictly smaller strips. Let $0<s'<s\le\eta$ and $u_1,u_2\in\mathcal{E}_s$ with $\|u_i\|_s\le R$. For $n\ge R/\eta$ the maps $\mathrm{id}+u_i/n$ send $S_s$ into $S_{2\eta}$, so $g_{u_i}$ is analytic and bounded on $S_s$ with $\|g_{u_1}-g_{u_2}\|_s\le\|\lambda'\|_{2\eta}\|u_1-u_2\|_s/n$. With the stability estimate of Lemma~\ref{lem:H5}(ii) and the Lipschitz continuity of $\Psi_f$ on $S_\eta$ this gives a constant $L$ depending on $f_m$ only, such that
\begin{equation}\label{eq:Flip}
  \bigl\|\mathcal{F}_n[u_1]-\mathcal{F}_n[u_2]\bigr\|_{s'}\le\frac{L}{n(s-s')}\|u_1-u_2\|_s,
  \qquad
  \bigl\|\mathcal{F}_n[u]\bigr\|_{s'}\le\|\Delta\Psi\|_\eta+\frac{L}{n(s-s')}\|u\|_s .
\end{equation}
Write $L_j:=L/(\eta_j-\eta_{j+1})=2^{j+2}L/\eta$ for the constant at the $j$-th scale.

Fix $K$ and let $n\ge n_K:=\max(R/\eta,2L_K)$. Then $L_j/n\le\tfrac12$ for $j\le K$, so the second estimate in \eqref{eq:Flip}, together with $\varepsilon^{(0)}=0$, gives inductively
\begin{equation}  \bigl\|\varepsilon^{(j)}\bigr\|_{\eta_j}\le\|\Delta\Psi\|_\eta+\tfrac12R=R,\qquad j=0,\dots,K .
\end{equation}
In particular each $\varepsilon^{(j)}$, $j\le K$, is bounded by $R$ on $S_{\eta/2}$.

As before define the residual recursively,
\begin{equation}\label{eq: residual identity}
  \rho\bigl[\varepsilon^{(K+1)}\bigr]=\varepsilon^{(K+1)}-\mathcal{F}_n\bigl[\varepsilon^{(K+1)}\bigr]
  =\mathcal{F}_n\bigl[\varepsilon^{(K)}\bigr]-\mathcal{F}_n\bigl[\varepsilon^{(K+1)}\bigr],
\end{equation}
and the increment of the arguments is the previous residual, $\varepsilon^{(K)}-\varepsilon^{(K+1)}=\rho[\varepsilon^{(K)}]$. Both arguments lie in $\mathcal{E}_{\eta_{K+1}}$ with norm at most $R$. Applying the first estimate of \eqref{eq:Flip} at the scale $\eta_{K+1}\to\eta_{K+2}$,
\begin{equation}  \bigl\|\rho\bigl[\varepsilon^{(K+1)}\bigr]\bigr\|_{\eta_{K+2}}\le\frac{L_{K+1}}{n}\bigl\|\rho\bigl[\varepsilon^{(K)}\bigr]\bigr\|_{\eta_{K+1}} .
\end{equation}
Since $\rho[\varepsilon^{(0)}]=-\Delta\Psi$, induction gives $\|\rho[\varepsilon^{(K)}]\|_{\eta_{K+1}}\le n^{-K}\bigl(\prod_{j=1}^{K}L_j\bigr)\|\Delta\Psi\|_\eta$, and \eqref{eq:respicard} follows with $L_K:=\bigl(\prod_{j=1}^{K}L_j\bigr)^{1/K}=\tfrac{4L}{\eta}2^{(K+1)/2}$, because $\eta_{K+1}\ge\eta/2$. Inserting \eqref{eq:respicard} into \eqref{eq:resbound} yields the pointwise bound in \eqref{eq:higher}, summing over $k$ yields the second.

\label{thm: banded approximation}
There exist $c_g>0$ and $n_0\in\mathbb{N}$, depending only on $g$, such that for all $n\ge n_0$, all $M\in\mathbb{N}$ and all $1\le j\le n$:
\begin{enumerate}
    \item[(i)] $\bigl|\lambda_j-\tilde\lambda_j\bigr|\le\varepsilon_M$;
    \item[(ii)] if $\varepsilon_M\le c_g/(2(n+1)^2)$, then $\tilde\lambda_j$ is simple and, with the sign of $\tilde v_j$ chosen such that $\langle v_j,\tilde v_j\rangle\ge0$,
    \begin{equation}\label{eq: eigenvector bound}
        \|v_j-\tilde v_j\|_2\le\frac{2\sqrt{2}(n+1)^2}{c_g}\varepsilon_M .
    \end{equation}
\end{enumerate}
In particular, the eigenvalues of the $M$-banded matrix approximate those of $\mathbf{T}_n(g)$ up to $\mathcal{O}(e^{-\gamma_F M})$ uniformly in $n$, and the eigenvectors converge uniformly in $j$ as soon as $M-\frac{2}{\gamma_F}\log n\to\infty$.

<!-- proof-10: body_md -->
Since $g$ is real-valued on $\mathbb{T}$ and $p$ is symmetric under $\theta\mapsto-\theta$, the coefficients are real and $b_{-k}=b_k$, so that $\mathbf{T}_n(g)$ is real symmetric. Writing $g(\theta):=g(e^{\mathrm{i}\theta})$, Assumption~\ref{ass:standing} together with Lemma~\ref{lem: separation implies non confluence} implies that $g$ is simple-loop. It is strictly monotone on $[0,\pi]$ with $g'\neq0$ on $(0,\pi)$ and $g''(0)\neq0\neq g''(\pi)$.

We start by proving the eigenvalue estimate. The difference $\mathbf{E}:=\mathbf{T}_n(g)-\mathbf{T}_n(g_M)=\mathbf{T}_n(g-g_M)$ is real symmetric. For any $h\in L^\infty(\mathbb{T})$ one has $\|\mathbf{T}_n(h)\|_2\le\|h\|_{L^\infty(\mathbb{T})}$, since $\mathbf{T}_n(h)=\mathbf{P}_n\mathbf{L}(h)\mathbf{P}_n$ is a compression of the Laurent operator $\mathbf{L}(h)$ on $\ell^2(\mathbb{Z})$, which is unitarily equivalent to multiplication by $h$ on $L^2(\mathbb{T})$. Hence
\begin{equation}\label{eq: E bound}
    \|\mathbf{E}\|_2\le\|g-g_M\|_{L^\infty(\mathbb{T})}\le\sum_{|k|>M}|b_k|=\varepsilon_M .
\end{equation}
The pointwise eigenvalue bound established in (i) is therefore a direct consequence of Weyl's inequality for symmetric matrices.

To show the eigenvector estimate given in (ii) we proceed as follows. Following \cite{BOGOYA_SLM} the eigenvalues are sampled from the symbol function, so

\begin{equation}
    \lambda_{j+1} - \lambda_j = C_1 \frac{g'(\theta)}{n} + C_2 \frac{g''(\theta)}{n^2} + \mathcal{O}(n^{-3}) \geq Cn^{-2}.
\end{equation}
Let us define the minimal spectral distance as follows,
\begin{equation}\label{eq: gap}
    \delta_n(g):=\min_{i\neq j}|\lambda_i-\lambda_j|\geq c_g(n+1)^{-2}.
\end{equation}

By assumption of the theorem $\varepsilon_M\le c_g/(2(n+1)^2)\le\delta_n(g)/2$ and by the first part of the proof 
\begin{equation}
    |\tilde\lambda_i-\lambda_j|\ge|\lambda_i-\lambda_j|-\varepsilon_M\ge\delta_n(g)/2, \quad i \neq j.
\end{equation}
In particular the $\tilde\lambda_i$ are pairwise distinct, so each $\tilde v_j$ is determined up to sign. Expand $v_j=\sum_{i}c_i\tilde v_i$ with $\sum_i c_i^2=1$, choosing the sign of $\tilde v_j$ such that $c_j=\langle v_j,\tilde v_j\rangle\ge0$. Using $\mathbf{T}_n(g)v_j=\lambda_jv_j$ and \eqref{eq: E bound},
\begin{equation}
    \varepsilon_M^2\ge\|\mathbf{E}v_j\|_2^2=\|(\mathbf{T}_n(g_M)-\lambda_j)v_j\|_2^2
    =\sum_{i}(\tilde\lambda_i-\lambda_j)^2c_i^2
    \ge\frac{\delta_n(g)^2}{4}\sum_{i\neq j}c_i^2
    =\frac{\delta_n(g)^2}{4}(1-c_j^2).
\end{equation}
Since $c_j\ge0$, the difference of the two unit vectors is controlled by
\begin{equation}
    \|v_j-\tilde v_j\|_2^2=2(1-c_j)\le2(1-c_j)(1+c_j)=2(1-c_j^2).
\end{equation}
Combining the last two displays with \eqref{eq: gap} yields
\begin{equation}
    \|v_j-\tilde v_j\|_2^2\le\frac{8\varepsilon_M^2}{\delta_n(g)^2}\le\frac{8(n+1)^4}{c_g^2}\varepsilon_M^2,
\end{equation}
and \eqref{eq: eigenvector bound} follows by taking square roots. The final assertion follows by inserting \eqref{eq: eps_M} into \eqref{eq: eigenvector bound}.
