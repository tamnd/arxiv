<!-- definition-1: body_md -->
A set $R\subseteq\mathbb Q_{>0}$ is called a
\textit{set of multiplicative recurrence} or \textit{multiplicatively recurrent} if, for every
measure-preserving action $T=(T_n)_{n\in\mathbb N}$ of $(\mathbb N,\times)$ on a
probability space $(X,\mathcal B,\mu)$ and every measurable set
$B\in\mathcal B$ of positive measure, there exist distinct
$m,n \in \mathbb N$ such that $ m/n\in R$ and
\begin{align*}
 \mu(T_m^{-1}B\cap T_n^{-1}B)>0.
\end{align*}
Here $T_1=\rm id$ and $T_{mn}=T_m\circ T_n$ for all $m,n \in \mathbb N$.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:mpr}
A set $R\subseteq\mathbb N$ is called a \textit{multiplicative pseudorandom set} or \textit{multiplicatively pseudorandom}
if, for every sufficiently large $K \in \mathbb N$ and every $Q\in\Phi_K$,
the set $R\cap Q\mathbb N$ is infinite and, for every $f\in\mathcal M$, we have
\begin{align*}
 \lim_{N\to\infty}\left|
 \mathbb E_{\substack{n\leqslant N\\Qn\in R}}f(n)
 -\mathbb E_{\substack{n\leqslant N\\n\in R}}f(n)\right|=0.
\end{align*}
Each expectation is normalized by the cardinality of the set in its
subscript.
<!-- end definition-2 -->
<!-- theorem-1: body_md -->
\label{thm:main}
Every multiplicatively pseudorandom set is multiplicatively recurrent.
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{cor:colour}
Let $R\subseteq\mathbb N$ be a multiplicative pseudorandom set. Then for every
finite partition $\mathbb N=C_1\sqcup\cdots\sqcup \, C_s$, the intersection
$(C_i/C_i)\cap(R\setminus\{1\})$ is non-empty for some integer
$1\leqslant i\leqslant s$. In other words,
the set $R$ is a set of \textit{topological multiplicative recurrence}.
<!-- end corollary-1 -->
<!-- corollary-2: body_md -->
\label{cor:heil}
Let $R\subseteq\mathbb N$ be a multiplicative pseudorandom set. For every
$r \in \mathbb N$ and every $f_1,\ldots,f_r\in\mathcal M$,
we have
\begin{align*}
 \liminf_{\substack{n\to\infty\\n\in R}}
 \max_{1\leqslant j\leqslant r}|f_j(n)-1|=0.
\end{align*}
<!-- end corollary-2 -->
<!-- theorem-2: body_md -->
\label{thm:suff}
Let $R\subseteq\mathbb N$ be a multiplicative pseudorandom set, and let
$T=(T_n)_{n\in\mathbb N}$ be a measure-preserving multiplicative action on a
probability space $(X,\mathcal B,\mu)$. For every measurable set
$B\in\mathcal B$ of positive measure and every $\varepsilon>0$, the set
\begin{align*}
 \mathcal T_R(\mu,T,B;\varepsilon)
 :=\{n\in R:\mu(T_n^{-1}B\cap B)>\mu(B)^2-\varepsilon\}
\end{align*}
is infinite. Suppose in addition that, for every $Q\in\mathbb N$ and $C\geqslant1$, we have
\begin{gather}
 |R\cap Q\mathbb N\cap[1,x]|\gg_{R,Q}R(x),\label{eq:nolocal}\\
 R(x/C)\gg_{R,C}R(x)\label{eq:regular}
\end{gather}
as $x\to\infty$. Then
$\underline d_R(\mathcal T_R(\mu,T,B;\varepsilon))>0$.
<!-- end theorem-2 -->
<!-- definition-3: body_md -->
For a non-integral $c>1$, the set
\begin{align*}
 \mathcal P_c:=\{\lfloor n^c\rfloor:n\in\mathbb N\}
\end{align*}
is called a \textit{Piatetski--Shapiro set}.
<!-- end definition-3 -->
<!-- definition-4: body_md -->
Let $X=G/\Gamma$ and $g:\mathbb Z\to G$ be as above, and suppose that the
orbit is totally equidistributed. For a Borel set $A\subseteq X$
satisfying $m_X(A)>0$ and $m_X(\partial A)=0$, we call
\begin{align*}
 \mathcal B_{X,g}(A):=\{n\in\mathbb N:g(n)\Gamma\in A\}
\end{align*}
a \textit{regular irrational nil-Bohr set}.
<!-- end definition-4 -->
<!-- definition-5: body_md -->
Given $\delta>0$, we put
\begin{align*}
 \mathcal S_\delta:=\{n\in\mathbb N:P^+(n)\leqslant n^\delta\},
\end{align*}
where $P^+(n)$ is the largest prime factor of $n$ and $P^+(1):=1$.
The elements of this set are called \textit{$\delta$-smooth numbers}.
<!-- end definition-5 -->
<!-- theorem-3: body_md -->
\label{thm:ps}
Let $1<c<6/5$. There exists $\delta=\delta(c)>0$ such that the intersection
$\mathcal P_c\cap\mathcal S_\delta$ is multiplicatively pseudorandom.\footnote{Adapting the argument in
\cite[Section~2]{MR3705270}, one can show that without the smoothness
restriction, Theorem~\ref{thm:ps} is false in general.}
In particular, the set $\mathcal P_c$ is multiplicatively
recurrent.
<!-- end theorem-3 -->
<!-- theorem-4: body_md -->
\label{thm:nilbohr}
Every regular irrational nil-Bohr set is multiplicatively pseudorandom,
and hence is a set of multiplicative recurrence.
<!-- end theorem-4 -->
<!-- corollary-3: body_md -->
\label{cor:genpoly}
Let $p:\mathbb Z\to\mathbb R$ be a generalized polynomial such that
$(p(an+b))_{n\in \mathbb N}$ is equidistributed modulo one for every
$a \in \mathbb N$ and every $b \in \mathbb Z$. For an interval
$I \subseteq [0,1)$ of positive length, the set
\begin{align*}
 \mathcal{L}_{p}(I):=\{n\in\mathbb N: \{p(n)\}\in I\}
\end{align*}
is multiplicatively pseudorandom, and hence is a set of multiplicative
recurrence.
<!-- end corollary-3 -->
<!-- example-1: body_md -->
\label{ex:beatty}
Let $\alpha>1$ be irrational and $\beta\geqslant0$. The set

<!-- proposition-1: body_md -->
\label{prop}
Let $R$ be a multiplicatively pseudorandom set and let $\sigma$ be a finite
positive Borel measure on $\mathcal M$. Then
\begin{equation*}
\lim_{K\to\infty}\limsup_{N\to\infty}
\left|
\mathbb E_{Q\in\Phi_K}
\mathbb E_{\substack{n\leqslant N\\Qn\in R}}
\int_{\mathcal M}f(Qn)\,d\sigma(f)-\sigma(\{1\})
\right|=0.
\end{equation*}
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
Write
\begin{align*}
A_Q(N;f)&:=\mathbb E_{\substack{n\leqslant N\\Qn\in R}}f(n),\\
A_1(N;f)&:=\mathbb E_{\substack{n\leqslant N\\n\in R}}f(n).
\end{align*}
Fix a sufficiently large $K$. For each $Q\in\Phi_K$ and $f\in\mathcal M$,
the definition gives $A_Q(N;f)-A_1(N;f)\to0$ as $N\to\infty$.
The integrands are bounded by two, and since $\Phi_K$ is finite, dominated
convergence gives
\begin{gather*}
\mathbb E_{Q\in\Phi_K}
\mathbb E_{\substack{n\leqslant N\\Qn\in R}}
\int_{\mathcal M}f(Qn)\,d\sigma(f)\\
=
\int_{\mathcal M}\left(\mathbb E_{Q\in\Phi_K}f(Q)\right)A_1(N;f)\,d\sigma(f)
+o_{K,\sigma}(1).
\end{gather*}
The contribution of $f\equiv1$ is $\sigma(\{1\})$, while the absolute value
of the remaining integral is at most
\[
\int_{\mathcal M\setminus\{1\}}\left|\mathbb E_{Q\in\Phi_K}f(Q)\right|\,d\sigma(f).
\]
Lemma~\ref{lem:mult} and dominated convergence show that this expression
tends to zero as $K\to\infty$, proving the proposition.

<!-- lemma-2: body_md -->
\label{lem:spectral}
Let $(T_n)_{n\in\mathbb N}$ be a measure-preserving action of
$(\mathbb N,\times)$ on $(X,\mathcal B,\mu)$, and let $B\in\mathcal B$.
There exists a finite positive Borel measure $\sigma$ on $\mathcal M$ such that for every $n \in \mathbb N,$ we have $\sigma(\{1\})\geqslant\mu(B)^2$ and
\begin{align*}
\mu(T_n^{-1}B\cap B)=\int_{\mathcal M}f(n)\,d\sigma(f).
\end{align*}
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
This is the Bochner--Herglotz representation used in
\cite[Section~2.1]{frantzikinakis2023partitionregularitypythagoreanpairs}.
By measure preservation, the map
\begin{align*}
 a/b\longmapsto
 \langle 1_B\circ T_a,1_B\circ T_b\rangle_{L^2(\mu)}
\end{align*}
is well defined and positive definite on $(\mathbb Q_{>0},\times)$.
Applying the representation to $1_B-\mu(B)$ gives a finite positive
measure $\nu$ representing the centered correlations. Then
$\sigma:=\nu+\mu(B)^2\delta_1$ has the asserted representation and
$\sigma(\{1\})\geqslant\mu(B)^2$.

<!-- proof-4: body_md -->
Put $a:=\mu(B)^2$. For $m\in\mathbb N$, write
\begin{align*}
c(m)&:=\mu(T_m^{-1}B\cap B),\\
S&:=\{m\in R:c(m)>a-\varepsilon\}.
\end{align*}
It suffices to consider $0<\varepsilon<a$. Applying
Lemma~\ref{lem:spectral} and Proposition~\ref{prop}, we may fix a
sufficiently large $K$ such that, for every sufficiently large $N$, we have
\[
\mathbb E_{Q\in\Phi_K}
\mathbb E_{\substack{n\leqslant N\\Qn\in R}}c(Qn)>a-\frac\varepsilon2.
\]
For each such $N$, there exists an element $Q=Q_N\in\Phi_K$ such that the
inner average exceeds $a-\varepsilon/2$. Since
$0\leqslant c(m)\leqslant1$ and $c(m)\leqslant a-\varepsilon+1_S(m)$ for $m\in R$, we obtain
\begin{equation} \label{eq:ratio}
|S\cap[1,QN]|
\geqslant\frac\varepsilon2\,|R_Q\cap[1,N]|.
\end{equation}
Each set $R_Q$ is infinite and $\Phi_K$ is finite. Consequently,
the minimum of the counts on the right-hand side tends to infinity with
$N$, proving that $S$ is infinite.

Suppose now that \eqref{eq:nolocal} and \eqref{eq:regular} hold,
and put $L:=\max\Phi_K$. For a sufficiently large real $x$, apply
\eqref{eq:ratio} with $N:=\lfloor x/L\rfloor$. The selected element
satisfies $QN\leqslant x$. Since $QN\geqslant N\geqslant x/(2L)$, the two counting
hypotheses give
\begin{align*}
|R_Q\cap[1,N]|
&=|R\cap Q\mathbb N\cap[1,QN]|\\
&\gg_{R,Q}R(QN)
 \geqslant R(x/(2L))
 \gg_{R,L}R(x).
\end{align*}
The implied constants can be made uniform over the finite set $\Phi_K$.
It follows that $|S\cap[1,x]|\gg_{R,K}\varepsilon|R\cap[1,x]|$ for all
sufficiently large $x$, giving the required positive relative lower density.

<!-- lemma-3: body_md -->
\label{lem:ps-factorization}
Let $U\geqslant1$. Every integer $n>U$ has a unique factorization $n=d\ell$
satisfying
\begin{align*}
 U<d\leqslant U P^-(d),\qquad P^+(\ell)\leqslant P^-(d).
\end{align*}
In this factorization, we have $P^+(n)=P^+(d)$. If $P^+(n)\leqslant y$, then
$d\leqslant Uy$.
<!-- end lemma-3 -->
<!-- proof-5: body_md -->
Arrange the prime factors of $n$, counted with multiplicity, in
non-increasing order, and let $d$ be their shortest initial product exceeding
$U$. The preceding product is $d/P^-(d)\leqslant U$, while every remaining prime
factor is at most $P^-(d)$. Conversely, the two displayed conditions force
$d$ to be this initial product, which proves uniqueness and the remaining
assertions.

<!-- lemma-4: body_md -->
\label{lem:ps-sawtooth}
Let $\frac12<\gamma<1$ and fix $Q\in\mathbb N$. Suppose that $X\geqslant2$ and
$H\geqslant2$ is an integer. For any complex numbers $u_n$ satisfying
$|u_n|\leqslant1$, define
\begin{align*}
 S_h(t):=\sum_{X<n\leqslant t}u_n e\bigl(h(Qn)^\gamma\bigr).
\end{align*}
Then, uniformly for $X\leqslant T\leqslant2X$, we have
\begin{gather*}
\sum_{X<n\leqslant T}u_n
    \bigl(\psi(-(Qn+1)^\gamma)-\psi(-(Qn)^\gamma)\bigr)\\
 \ll_{Q,\gamma}
 X^{\gamma-1}\max_{X\leqslant t\leqslant2X}
       \sum_{1\leqslant|h|\leqslant H}|S_h(t)|
 +\frac{X\log(2H)}H+X^{1-\gamma}+(HX^\gamma)^{1/2}.
\end{gather*}
<!-- end lemma-4 -->
<!-- proof-6: body_md -->
The standard Fourier approximation \cite[p.~245]{MR698168} gives
\begin{align*}
 \psi(t)=-\sum_{0<|h|\leqslant H}\frac{e(ht)}{2\pi ih}+O(g(t,H)),
\end{align*}
where
\begin{align*}
 g(t,H):=\min\left\{1,\frac1{H\|t\|}\right\}.
\end{align*}
Here $g(t,H)=\sum_{h\in\mathbb Z}c_he(ht)$ with
\begin{align*}
 c_0\ll&\frac{\log(2H)}H,\\
 c_h\ll&\min\left\{\frac{\log(2H)}H,\frac1{|h|},
\frac H{|h|^2}\right\}\qquad (h\ne0) .
\end{align*}
For $a\in\{0,1\}$ and $\lambda\neq0$, the derivative estimate in \cite[Lemma~1]{MR698168} gives
\begin{align*}
 \sum_{X<n\leqslant T}e\bigl(\lambda(Qn+a)^\gamma\bigr)
 \ll_{Q,\gamma}|\lambda|^{-1}X^{1-\gamma}
                     +( |\lambda|X^\gamma)^{1/2}
\end{align*}
Summing against the Fourier
coefficients, we obtain
\begin{align*}
 \sum_{X<n\leqslant2X}g(-(Qn+a)^\gamma,H)
 \ll_{Q,\gamma}\frac{X\log(2H)}H+X^{1-\gamma}+(HX^\gamma)^{1/2}.
\end{align*}
To treat the finite Fourier sum, put
\begin{align*}
 \Delta_Q(x)&:=(Qx+1)^\gamma-(Qx)^\gamma,\\
 b_h(x)&:=\frac{1-e(-h\Delta_Q(x))}{2\pi ih}.
\end{align*}
For $X\leqslant x\leqslant2X$, we have
\begin{align*}
 |b_h(x)|\ll_{Q,\gamma}X^{\gamma-1},\qquad
 |b_h'(x)|\ll_{Q,\gamma}X^{\gamma-2}.
\end{align*}
The finite Fourier contribution is
\begin{align*}
\sum_{0<|h|\leqslant H}\sum_{X<n\leqslant T}u_nb_h(n)e(-h(Qn)^\gamma). 
\end{align*}
Partial summation gives the first term in the claimed bound, while the
preceding estimate for $g$ bounds the remainder.

<!-- lemma-5: body_md -->
\label{lem:ps-typeii}
Let $\eta>0$ and $\frac12+\eta\leqslant\gamma<1$. For $Z\geqslant2$, put
\begin{align*}
 H:=\left\lfloor Z^{1-\gamma}(\log Z)^3\right\rfloor,\qquad
 U:=Z^{1-\gamma+\eta},\qquad V:=Z^{5\gamma-4-\eta}.
\end{align*}
For each positive integer $d$, let $\mathcal I_d\subseteq(Z,2Z]$ be an
interval. Suppose that $|a_d|,|b_s|\leqslant1$ and that $a_d=0$ unless
$U<d<V$. Then
\begin{align*}
 \sum_{1\leqslant|h|\leqslant H}
 \left|\sum_{\substack{d,s\geqslant1\\ds\in\mathcal I_d}}
      a_db_s e\bigl(h(ds)^\gamma\bigr)\right|
 \ll_{\gamma,\eta}Z^{1-\eta/7}.
\end{align*}
<!-- end lemma-5 -->
<!-- proof-7: body_md -->
The negative values of $h$ are handled by conjugating the coefficients,
and hence it suffices to consider positive $h$. Choose $|\varepsilon_h|\leqslant1$ to remove the absolute
values in the sum over $h$. After interchanging the sums and using
$|b_s|\leqslant1$, the resulting expression is at most
\begin{align*}
 \sum_s\left|\sum_{\substack{U<d<V\\ds\in\mathcal I_d}}
       a_d\sum_{h\leqslant H}\varepsilon_h e\bigl(h(ds)^\gamma\bigr)\right|.
\end{align*}
We follow the proof of \cite[Lemma~4]{MR698168}, retaining the dependence
of the interval on $d$. Decompose the variables into dyadic ranges satisfying
$d\asymp R$ together with $s\asymp S$ and $h\asymp J$, where
$RS\asymp Z$ and $J\leqslant H$. This decomposition incurs only logarithmic
factors.
For one such choice of ranges, put
\begin{align*}
 L:=\sum_{S<s\leqslant2S}
 \left|\sum_{\substack{d\asymp R,\ h\asymp J\\
                       ds\in\mathcal I_d}}
 a_d\varepsilon_h e\bigl(h(ds)^\gamma\bigr)\right|.
\end{align*}

We first require the spacing bound from \cite[pp.~256--257]{MR698168} that
\begin{gather*}
 \mathcal N(\Delta):=|\{(d,r,h,k):d,r\asymp R,\ h,k\asymp J,
                   \ |hd^\gamma-kr^\gamma|\leqslant\Delta\}|,\\
 \mathcal N(\Delta)\ll_\gamma \Delta JR^{2-\gamma}+JR\log(2JR).
\end{gather*}
To verify it, put $a:=\gcd(d,r)$ and $b:=\gcd(h,k)$. Fix $a,b$. For each
of the $O(J^2/b^2)$ pairs $(h,k)$ with $\gcd(h,k)=b$, the relevant ratios
$d/r$ lie in an interval of length
$O_\gamma(\Delta/(JR^\gamma))$. Distinct such ratios with $(d,r)=a$ are
separated by at least $a^2/(4R^2)$. Hence the number of quadruples is
\begin{align*}
 \ll_\gamma \frac{J^2}{b^2}
          +\frac{\Delta JR^{2-\gamma}}{a^2b^2}.
\end{align*}
Interchanging the two pairs gives the same bound with $R^2/a^2$ in
place of $J^2/b^2$. Summing the smaller bound over $a$ and $b$, and using
\begin{align*}
 \sum_{a\leqslant2R}\sum_{b\leqslant2J}
       \min\left\{\frac{J^2}{b^2},\frac{R^2}{a^2}\right\}
       \ll JR\log(2JR)
\end{align*}
proves the claim.

Let $K\geqslant1$ be an integer, and partition the pairs $(d,h)$ into
sets $\mathcal P_1,\ldots,\mathcal P_m$, where $m\ll K$, such that
$hd^\gamma$ ranges over an interval of length
$O(JR^\gamma/K)$ on each $\mathcal P_\ell$. For $S<s\leqslant2S$, put
\begin{align*}
 F_\ell(s):=
 \sum_{\substack{(d,h)\in\mathcal P_\ell\\
                  ds\in\mathcal I_d}}
 a_d\varepsilon_h e\bigl(hd^\gamma s^\gamma\bigr).
\end{align*}
The Cauchy--Schwarz inequality gives
\begin{align*}
 L^2
 &\ll S\sum_{S<s\leqslant2S}
       \left|\sum_{\ell\leqslant m}F_\ell(s)\right|^2\\
 &\ll KS\sum_{\ell\leqslant m}\sum_{S<s\leqslant2S}|F_\ell(s)|^2.
\end{align*}
Expanding the last square and using $|a_d|,|\varepsilon_h|\leqslant1$, we
obtain
\begin{align*}
 L^2\ll KS\sum_{\ell\leqslant m}
 \sum_{\substack{(d,h),(r,k)\in\mathcal P_\ell}}
 \left|\sum_{s\in(S,2S]\cap d^{-1}\mathcal I_d\cap r^{-1}\mathcal I_r}
 e\bigl((hd^\gamma-kr^\gamma)s^\gamma\bigr)\right|.
\end{align*}
Notice that the range of $s$ in the inner sum is an interval. Put
$\lambda:=hd^\gamma-kr^\gamma$. If $\lambda\neq0$, then
\cite[Lemma~1]{MR698168} gives
\begin{align*}
\sum_{s\in(S,2S]\cap d^{-1}\mathcal I_d\cap r^{-1}\mathcal I_r}
 e\bigl(\lambda s^\gamma\bigr)
 \ll_\gamma
 \min\left\{S,\ |\lambda|^{-1}S^{1-\gamma}
 +( |\lambda|S^\gamma)^{1/2}\right\}.
\end{align*}
For $\lambda=0$, the same sum is trivially bounded by $S$.
Apply the spacing bound to $|\lambda|\leqslant S^{-\gamma}$ and to the
remaining dyadic ranges up to $O(JR^\gamma/K)$. The square-root term is
bounded using the largest possible $|\lambda|$. For every
$\varepsilon>0$, we have
\begin{align*}
 L^2\ll_{\gamma,\varepsilon}Z^\varepsilon KS\bigg\{
 \left(\frac{JZ^\gamma}K\right)^{1/2}
 \left(\frac{J^2R^2}K+JR\right)
 +S^{1-\gamma}\bigl(JR^{2-\gamma}+JRS^\gamma\bigr)\bigg\}.
\end{align*}
Taking
\begin{align*}
 K:=1+\left\lfloor JS^{(\gamma-2)/3}R^{(\gamma+2)/3}\right\rfloor
\end{align*}
and using $S\asymp Z/R$ together with $J\leqslant Z^{1-\gamma}(\log Z)^3$,
we arrive at
the six-term bound in \cite[p.~257]{MR698168}:
\begin{align*}
 L^2\ll_{\gamma,\varepsilon}Z^\varepsilon
 \bigl(&Z^{5/2-\gamma}+Z^{3-2\gamma}+Z^{3-\gamma}R^{-1}
       +Z^{(10-5\gamma)/3}R^{1/3}\\
       &+Z^{(8-4\gamma)/3}R^{2/3}
       +Z^{(10-8\gamma)/3}R^{4/3}\bigr).
\end{align*}
Since $R\gg U$ and $R\ll V$, each term in parentheses is
$\ll Z^{2-\eta/3}$. Taking $\varepsilon<\eta/21$ and absorbing the
dyadic decomposition into the remaining power saving proves the lemma.

Put

 \eta:=\frac{6\gamma-5}{4},\qquad \delta:=\frac\eta{25}.
\end{align*}
Then $\frac12+\eta<\gamma$ and
\begin{align*}
 2\eta+\delta<6\gamma-5,\qquad \frac\eta7>2\delta.

Fix $Q\in\mathbb N$ and take $X$ sufficiently large in terms of $Q$ and $c$.

\begin{align*}

\end{align*}
also holds when either endpoint is an integer. Apply

\end{align*}

\begin{align*}
 \Delta_Q(n)=\gamma Q^{\gamma-1}n^{\gamma-1}

\end{align*}
whose error is summable.

\end{align*}

\begin{align*}
 |\{X<n\leqslant2X:Qn\in\mathcal S_\delta,\ n\notin\mathcal S_\delta\}|

 \ll_{Q,c}N^{\gamma/2}+\frac{N^\gamma}{\log N}
 =o_{Q,c}(N^\gamma).
\end{align*}
Combining these estimates proves the proposition, with all errors uniform
for $f\in\mathcal M$.

\label{lem:nil-orthogonality}
Let $X=G/\Gamma$ be a nilmanifold, and let $g:\mathbb Z\to G$ be a polynomial
sequence whose orbit is totally equidistributed in $X$. If $F:X\to\mathbb C$

\end{align*}

Fix $Q\in\mathbb N$. The polynomial sequence $g_Q(n):=g(Qn)$ is totally
equidistributed. Indeed, its restriction to the progression $a\mathbb N+b$ is
$(g(Qan+Qb)\Gamma)_{n\in\mathbb N}$, which is equidistributed by the total
equidistribution of $g$. Moreover, the function $1_A-\delta$ is Riemann integrable

 \left|\mathbb E_{n\leqslant N}f(n)(1_R(Qn)-\delta)\right|=0.
\end{align*}
Multiplying by $N$ proves \eqref{eq:weighted-main}.

\label{lem:gp-short}
Let $u:\mathbb Z\to\mathbb R$ be a bounded generalized polynomial, and suppose that
there exists $\delta\in\mathbb R$ such that for any $a\in\mathbb N, \ b\in\mathbb Z$, we have
\begin{align*}
 \lim_{N\to\infty}\mathbb E_{n\leqslant N}u(an+b)=\delta.
\end{align*}
Then
\begin{align*}
 \lim_{N\to\infty}
 \sup_{f\in\mathcal M}
 \left|\mathbb E_{n\leqslant N}f(n)(u(n)-\delta)\right|=0.
\end{align*}

After an affine rescaling, the representation theorem of
Bergelson--Leibman \cite[Theorem~A$^*$]{MR2318563} gives
\begin{align*}
 u(n)=F(a(n)\Gamma),
\end{align*}
where $G/\Gamma$ is a connected nilmanifold. Here $a:\mathbb Z\to G$ is
polynomial and $F$ is bounded and piecewise polynomial. By the
factorization theorem \cite[Corollary~1.12]{MR2877065}, there exist a
connected rational subgroup $H\leqslant G$, a constant $\varepsilon\in G$, and
polynomial sequences $h:\mathbb Z\to H$ and $\gamma:\mathbb Z\to G$ such that
\begin{align*}
 a(n)=\varepsilon h(n)\gamma(n).
\end{align*}
The orbit $h(n)(H\cap\Gamma)$ is totally equidistributed in
$H/(H\cap\Gamma)$, while $\gamma(n)\Gamma$ is rational and periodic.
Let $q$ be a period and, for $0\leqslant r<q$, choose a rational element $\gamma_r\in G$
such that
\begin{align*}
 \gamma(n)\Gamma=\gamma_r\Gamma\qquad(n\equiv r\ (\mathrm{mod}\ q)).
\end{align*}

Put
\begin{align*}
 H_r:=\gamma_r^{-1}H\gamma_r,\quad \Gamma_r:=H_r\cap\Gamma,\quad
 h_r(n):=\gamma_r^{-1}h(n)\gamma_r,\quad
 F_r(x\Gamma_r):=F(\varepsilon\gamma_r x\Gamma).
\end{align*}
Then $u(n)=F_r(h_r(n)\Gamma_r)$ whenever $n\equiv r\ (\mathrm{mod}\ q)$.
The orbit $h_r(n)\Gamma_r$ is totally equidistributed in
$X_r:=H_r/\Gamma_r$ by \cite[Corollary~5.5]{MR3556289}. By
\cite[Proposition~2.8]{MR2318563}, the coordinate map defining $F_r$
is piecewise polynomial. Hence $F_r$ is piecewise polynomial and, as
noted in \cite[Section~0.18]{MR2318563}, Riemann integrable. Therefore,
we have
\begin{align*}
 \int_{X_r}F_r\,dm_{X_r}
 =\lim_{N\to\infty}\mathbb E_{n\leqslant N}u(qn+r)=\delta.
\end{align*}
Applied to $F_r-\delta$, the progression estimate
\cite[Theorem~6.1]{MR3556289} gives the required bound for Lipschitz
functions. The $L^1$-approximation argument in \cite[Remark~(3)]{MR3556289} following
\cite[Theorem~2.2]{MR3556289} applies with the progression indicator, since
$|f|\leqslant1$, and gives
\begin{align*}
 \sup_{f\in\mathcal M}
 \left|\frac1N\sum_{\substack{n\leqslant N\\n\equiv r\ (\mathrm{mod}\ q)}}
 f(n)(F_r(h_r(n)\Gamma_r)-\delta)\right|=o(1).
\end{align*}
Summing over $r$ proves the lemma.

\label{prop:equid}
Let $f\in\mathcal M$. The sequence $(f(n))_{n\geqslant1}$ is equidistributed on
$\mathbb T$ if and only if $f$ is strictly non-pretentious. Equivalently, we have
\begin{align}\label{eq:powerpret}
f^j\not\sim n^{it}\qquad\text{for every $j \in \mathbb N$ and every $t\in\mathbb R$.}
\end{align}
<!-- end proposition-4 -->
<!-- proof-16: body_md -->
By Weyl's criterion, Hal\'asz's theorem
\cite[Theorem~2.1.10]{gs}, and its formulation in
\cite[Exercise~2.1.7]{gs}, equidistribution is equivalent to
\eqref{eq:powerpret}. The exceptional case at the prime $2$ in the
general form of Hal\'asz's theorem would require
$f(2)^j=-2^{it}$ and $f(4)^j=-4^{it}$, contrary to complete
multiplicativity.

If $g^j\equiv1$, then
\begin{align*}
\mathbb D(f^j,n^{ijt})\leqslant j \,\mathbb D(f,g\cdot n^{it}).
\end{align*}
Conversely, suppose that $f^j\sim n^{it}$. At each prime $p$, choose
$g(p)$ to be a nearest $j$-th root of unity to $f(p)p^{-it/j}$, and
extend $g$ completely multiplicatively. Write
$f(p)\overline{g(p)}p^{-it/j}=e^{i\theta_p}$, where
$|\theta_p|\leqslant\pi/j$. Then
\begin{align*}
|1-f(p)\overline{g(p)}p^{-it/j}|
=|1-e^{i\theta_p}|\leqslant|1-e^{ij\theta_p}|
=|1-f(p)^jp^{-it}|.
\end{align*}
Consequently, we have $\mathbb D(f,g\cdot n^{it/j})<\infty$, which
proves the claimed equivalence.

<!-- definition-8: body_md -->
Let $T=(T_n)_{n\in\mathbb N}$ be a measure-preserving multiplicative action on
$(X,\mathcal B,\mu)$. Its \textit{invariant $\sigma$-algebra} is
\begin{align*}
 \mathcal I_T:=\{B\in\mathcal B:
 \mu(T_n^{-1}B\mathbin\triangle B)=0
 \text{ for every }n\in\mathbb N\}.
\end{align*}
A function $H\in L^2(\mu)$ is \textit{invariant} if $H\circ T_n=H$ in
$L^2(\mu)$ for every $n\in\mathbb N$. The action is \textit{multiplicatively ergodic} if every
set in $\mathcal I_T$ has measure zero or one, or equivalently, every invariant function in
$L^2(\mu)$ is constant almost everywhere.

<!-- theorem-8: body_md -->
\label{thm:mult-ergodic}
Let $f\in\mathcal M$. The multiplicative action $(T_n^f)_{n\in\mathbb N}$ on
$(\mathbb T,m_{\mathbb T})$ is ergodic if and only if $X_f=\mathbb T$, or equivalently if and only if
$f^k\not\equiv1$ for every $k \in \mathbb N$.
<!-- end theorem-8 -->
<!-- proof-21: body_md -->
The closure of a sub-semigroup of a compact group is a subgroup. Since the
closed subgroups of $\mathbb T$ are $\mathbb T$ and the finite cyclic groups, we have
$X_f=\mathbb T$ precisely when $f$ has infinite order. If $f^k\equiv1$, then
the non-constant function $H(z):=z^k$ is invariant under every $T_n^f$.
Conversely, suppose that $f$ has infinite order and that
$H\in L^2(m_{\mathbb T})$ is invariant. For every $j\in\mathbb Z$ and $n\in\mathbb N$, we have
\begin{align*}
 \widehat H(j)=f(n)^j\widehat H(j) .
\end{align*}
For every  $j \neq 0$, there exists $n\in\mathbb N$ satisfying
$f(n)^j\ne1$. It follows that $\widehat H(j)=0$ for every $j \neq 0$.
Thus $H$ is constant almost everywhere, and the action is ergodic.
