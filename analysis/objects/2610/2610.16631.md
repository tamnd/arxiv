<!-- definition-1: body_md -->
For any copula $C$ its \emph{non-exchangeability} is defined as the supremum norm of the difference $C - C^T$, i.e.,
$$ \mu_T(C) = \max\big\{|C(u,v)-C(v,u)| \colon u,v\in \mathbb{I}\big\}.$$
<!-- end definition-1 -->
<!-- definition-2: body_md -->
For any copula $C$ its \emph{radial asymmetry} is defined as the supremum norm of the difference $C - \widehat{C}$, i.e.,
$$ \mu_R(C) = \max\big\{|C(u,v)-\widehat{C}(u,v)| \colon u,v\in \mathbb{I}\big\}.$$
<!-- end definition-2 -->
<!-- proposition-1: body_md -->
\label{pr:d}
The maximal asymmetry functions are
\begin{align*}
       d_T(u,v) &= \min\big\{u, v, 1-u, 1-v, |u-v|\big\}, \text{ and}\\
       d_R(u,v) &= \min\big\{u, v, 1-u, 1-v, \max\{|u-v|, |u+v-1|\}\big\}.
\end{align*}
<!-- end proposition-1 -->
<!-- theorem-1: body_md -->
\label{th:main}
   The set of all possible pairs $(\mu_R(C),\mu_T(C)), C \in \mathscr C$ is a hexagon $$\Omega_{R,T} = \mathcal{A}\mathcal{B}\mathcal{C}\mathcal{D}\mathcal{E}\mathcal{F}$$ with vertices
   $$\mathcal{A}(0,0), \quad \mathcal{B}(\tfrac13,0), \quad \mathcal{C}(\tfrac13,\tfrac16), \quad \mathcal{D}(\tfrac{3}{10},\tfrac15), \quad \mathcal{E}(\tfrac13,\tfrac13), \quad \mathcal{F}(0,\tfrac14).$$
<!-- end theorem-1 -->
<!-- lemma-1: body_md -->
\label{le:delta}
   \emph{(a)} Let $C \in \mathscr C$ be a copula with $\mu_T(C) = t \in [0, \tfrac13]$. Suppose that  non-exchangeability $t$ is attained at the point $(u,v) \in \mathbb{I}^2$, i.e. $|C(u,v)-{C}^T(u,v)|=t$. Then $(u,v) \in \Delta_t^2 \cup \Delta_t^4$.
   
   \emph{(b)} Let $C \in \mathscr C$ be a copula with $\mu_R(C) = r \in [0, \tfrac13]$. Suppose that radial asymmetry is attained at the point $(u,v) \in \mathbb{I}^2$, i.e. $|C(u,v)-\widehat{C}(u,v)|=r$. Then $(u,v) \in \Delta_r^1 \cup \Delta_r^2 \cup \Delta_r^3 \cup \Delta_r^4$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
(a) Suppose that $|C(u,v)-{C}^T(u,v)|=t\in [0,\frac13]$. Then $d_T(u,v) \ge t$, so by Proposition \ref{pr:d} we have
$$u \ge t, \quad v \ge t, \quad u \le 1-t, \quad v \le 1-t, \quad \text{and} \quad |u-v| \ge t.$$  
Now, if $u\ge v$ then $u-v\ge t$ and $v\ge t$, $u\le 1-t$. So, $(u,v)\in\Delta_t^4$. If $u\le v$ then $v-u\ge t$ and $u\ge t$, $v\le 1-t$. So, $(u,v)\in\Delta_t^2$. 

(b) Suppose that $|C(u,v)-\widehat{C}(u,v)|=r$. Then $d_R(u,v) \ge r$, so by Proposition \ref{pr:d} we have
$$u \ge r, \quad v \ge r, \quad u \le 1-r, \quad v \le 1-r, \quad \text{and} \quad \max\{|u-v|, |u+v-1|\} \ge r.$$
We consider several cases.
If $|u-v| \ge |u+v-1|$ and $u-v \ge 0$, then 
$$\max\{|u-v|, |u+v-1|\} = u-v \ge r,$$
so $(u,v) \in \Delta_r^4$. 
If $|u-v| \ge |u+v-1|$ and $u-v \le 0$, then $v-u \ge r$, so $(u,v) \in \Delta_r^2$. 
Furthermore, if $|u-v| \le |u+v-1|$ and $u+v-1 \ge 0$, then 
$$\max\{|u-v|, |u+v-1|\} = u+v-1 \ge r,$$
thus $u+v \ge 1+r$ and $(u,v) \in \Delta_r^3$. 
Finally, if $|u-v| \le |u+v-1|$ and $u+v-1 \le 0$, then $u+v \le 1-r$, so $(u,v) \in \Delta_r^1$.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{le:simple}
   Let $u, \alpha, \beta \in \mathbb{R}$. Then $\min\{u + \alpha, \beta - u\} \le \tfrac{\alpha+\beta}{2}$.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
If $u \le \tfrac{\beta-\alpha}{2}$ then $u + \alpha \le \beta - u$, so
$$\min\{u + \alpha, \beta - u\} = u + \alpha \le \tfrac{\beta-\alpha}{2} + \alpha = \tfrac{\alpha+\beta}{2}.$$
If $u \ge \tfrac{\beta-\alpha}{2}$ then $u + \alpha \ge \beta - u$, so
$$\min\{u + \alpha, \beta - u\} = \beta - u \le \beta - \tfrac{\beta-\alpha}{2} = \tfrac{\alpha+\beta}{2},$$
which finishes the proof.
<!-- end proof-2 -->
<!-- proposition-2: body_md -->
\label{pr:zgornja1}
   For any copula $C \in \mathscr C$ we have 
   $$\mu_T(C) \le \tfrac14 \mu_R(C) + \tfrac14.$$

<!-- proof-3: body_md -->
We assume that $C\in\mathscr C$ and denote $r=\mu_R(C)\in [0,\tfrac13]$. Then $|C(u,v)-\widehat{C}(u,v)|\le r$ for any $(u,v)\in\mathbb{I}^2$, and hence $C(u,v)-\left(u+v-1+C(1-u,1-v)\right)\le r$ or
    \begin{equation}\label{one_star}
        C(u,v)\leq C(1-u,1-v)+r+u+v-1.
    \end{equation}

    To prove the required inequality, we consider several cases (see Figure~\ref{fig:04}).

    \textbf{Case I.} Suppose first that $u,v\le\frac12$ and $u\leq v$. Then we have
    $$
    C(u,v)-C(v,u)\leq d_T(u,v) \leq \min\{u,v-u\}\leq\tfrac{v}{2}\leq\tfrac14\leq\tfrac{r+1}{4}.
    $$
    Here, the second inequality follows by Proposition \ref{pr:d}, the third inequality by Lemma \ref{le:simple}, the fourth inequality holds since $v\leq\frac12$, and the last one holds since $r\ge 0$.

    \textbf{Case II.} If $u,v\le\frac12$ and $u\geq v$ then, using analogous arguments as in the previous case, we have
    $$
    C(u,v)-C(v,u)\leq \min\{v,u-v\}\leq\tfrac{u}{2}\leq\tfrac14\leq\tfrac{r+1}{4}.
    $$

    \textbf{Case III.} Suppose next that $u,v\ge\frac12$ and $u\leq v$. Then we have
    $$
    C(u,v)-C(v,u)\leq d_T(u,v)\leq \min\{1-v,v-u\}\leq\tfrac{1-u}{2}\leq\tfrac14\leq\tfrac{r+1}{4},
    $$
    where the second inequality follows by Proposition \ref{pr:d}, the third inequality by Lemma \ref{le:simple}, the fourth inequality holds since $u\geq\frac12$, and the last one holds since $r\ge 0$.

    \textbf{Case IV.} If $u,v\ge\frac12$ and $u\geq v$ then, using analogous arguments as in the previous case, we have
    $$
    C(u,v)-C(v,u)\leq \min\{1-u,u-v\}\leq\tfrac{1-v}{2}\leq\tfrac14\leq\tfrac{r+1}{4}.
    $$

    \textbf{Case V.} Suppose now that $u\le\frac12$, $v\ge\frac12$ and $u+v\leq 1$. Then, using \eqref{one_star}, we have
    $$
    C(u,v)-C(v,u)\leq C(1-u,1-v)+r+u+v-1-C(v,u).
    $$
    The $1$-Lipschitz property of $C$ implies
    $$
    C(1-u,1-v)-C(v,u) \leq |(1-u)-v|+|(1-v)-u| = 2(1-u-v),
    $$
    since $u+v\leq 1$. 
    Combining both inequalities we obtain
    \begin{equation}\label{two-star}
        C(u,v)-C(v,u)\leq r+1-u-v.
    \end{equation}
    Finally, we have
    $$
    C(u,v)-C(v,u)\leq\min\{u,v-u,r+1-u-v\}\leq \min\big\{u,\tfrac{r+1-2u}{2}\big\}
    \leq\tfrac{r+1}{4},
    $$
    where the first inequality follows by Proposition \ref{pr:d} and inequality \eqref{two-star}, while the second and the third inequalities follow by Lemma \ref{le:simple}.

    \textbf{Case VI.} If $u\le\frac12$, $v\ge\frac12$ and $u+v\geq 1$ then analogous arguments as in the previous case imply that
    $$
     C(u,v)-C(v,u)\leq r+u+v-1,
    $$
    and also
    $$
    C(u,v)-C(v,u)\leq\min\{1-v,v-u,r+u+v-1\}\leq\min\big\{1-v,\tfrac{2v+r-1}{2}\big\}\leq\tfrac{r+1}{4}.
    $$

    \textbf{Case VII.} Assume that $u\geq\frac12$, $v\leq\frac12$ and $u+v\leq 1$. Then the same argument as in the Case V implies that
    $$
    C(u,v)-C(v,u)\leq r+1-u-v.
    $$
    Combing this inequality with Proposition \ref{pr:d} and Lemma \ref{le:simple} we obtain
    $$
    C(u,v)-C(v,u)\leq \min\{v,u-v,r+1-u-v\}\leq\min\big\{v,\tfrac{r+1-2v}{2}\big\}\leq\tfrac{r+1}{4}.
    $$

    \textbf{Case VIII.} In the final case $u\geq\frac12$, $v\leq\frac12$ and $u+v\geq 1$ we have 
    $$
    C(u,v)-C(v,u)\leq r+u+v-1
    $$
    using the same argument as in Case VI. Combing this inequality with Proposition \ref{pr:d} and Lemma \ref{le:simple} we have
    $$
    C(u,v)-C(v,u)\leq \min\{1-u,u-v,r+u+v-1\}\leq\min\big\{1-u,\tfrac{r-1+2u}{2}\big\}\leq\tfrac{r+1}{4}.
    $$

    We have proved that $C(u,v)-C(v,u)\leq \tfrac{r+1}{4}$ holds for any $(u,v)\in\mathbb{I}^2$. Now,
    \begin{align*}
    \mu_T(C) &= \max\big\{|C(u,v)-C(v,u)| \colon u,v\in \mathbb{I}\big\} \\
    &= \max\big\{C(u,v)-C(v,u) \colon u,v\in \mathbb{I}\big\} \\
    &\leq \tfrac{r+1}{4} = \tfrac14 \mu_R(C) +  \tfrac14,      
    \end{align*}
    which finishes the proof.

\label{pr:spodnja}
   Suppose that for some copula $C \in \mathscr C$ it holds that $\mu_R(C) = r \in [\tfrac14, \tfrac13]$ and that radial asymmetry $\mu_R$ is attained at a point $(u,v) \in \Delta_r^2$. Then $$\mu_T(C) \ge 4 \mu_R(C) - 1.$$

<!-- proof-4: body_md -->
By assumptions of the proposition we have $|C(u,v)-\widehat{C}(u,v)| = r$ for some point $(u,v) \in \Delta_r^2$. Assume first that $C(u,v)-\widehat{C}(u,v) \ge 0$. Then 
   $$ C(u,v) = \widehat{C}(u,v) + r = C(1-u,1-v) + u + v - 1 + r.$$
   
   We consider two cases. Suppose that $u+v \le 1$. Then 
   $$ C(u,v) \le C(1-v,v) \le C(1-v, 1-u),$$
   since vertical and horizontal sections of a copula are non-decreasing. Now,
   \begin{align*}
      \mu_T(C) &\ge C(1-v, 1-u) - C(1-u,1-v) \\
      &\ge C(u,v) - C(1-u,1-v) \\
      &= C(1-u,1-v) + u + v - 1 + r - C(1-u,1-v) \\
      &= u + v - 1 + r \\
      &\ge 2u + 2r - 1  \\
      &\ge 4r - 1, 
   \end{align*}
   since $(u,v) \in \Delta_r^2$, so that $v \ge u+r$ and $u \ge r$. 

   Suppose now that $u+v \ge 1$. Then 
   $$ \widehat{C}(1-u,1-v) \le \widehat{C}(v,1-v) \le \widehat{C}(v,u)$$
   and 
   \begin{align*}
      \mu_T(C) &\ge C(1-v, 1-u) - C(1-u,1-v) \\ 
      &= \widehat{C}(v,u) + 1 - v - u - C(1-u,1-v) \\
      &\ge \widehat{C}(1-u,1-v) + 1 - v - u - C(1-u,1-v) \\
      &= C(u,v) + 2 - 2u - 2v - C(1-u,1-v) \\
      &= C(1-u,1-v) + u + v - 1 + r - C(1-u,1-v) \\
      &= r + 1 - u - v  \\
      &\ge 4r - 1, 
   \end{align*}
    since $(u,v) \in \Delta_r^2$, so that $-u \ge r - v$ and $-v \ge r - 1$.  
    In both cases we have proved that  $\mu_T(C) \ge 4r-1 = 4 \mu_R(C) - 1.$

   In the case that $C(u,v)-\widehat{C}(u,v) \le 0$ we have $\widehat{C}(u,v) - C(u,v) \ge 0$. We interchange the roles of $C$ and $\widehat{C}$ to deduce 
   $\mu_T(\widehat{C}) \ge 4r-1$ in the same way. Now, 
   \begin{align*}
      \mu_T(\widehat{C}) &= \max\big\{|\widehat{C}(u,v)-\widehat{C}(v,u)| \colon u,v\in \mathbb{I}\big\} \\
      &= \max\big\{|u+v-1+C(1-u,1-v)-(v+u-1+C(1-v,1-u))| \colon u,v\in \mathbb{I}\big\} \\
      &= \max\big\{|C(1-u,1-v)-C(1-v,1-u))| \colon u,v\in \mathbb{I}\big\} \\
      &= \max\big\{|C(u,v)-C(v,u))| \colon u,v\in \mathbb{I}\big\} = \mu_T(C),
   \end{align*}
   so again $\mu_T(C) \ge 4r-1 = 4 \mu_R(C) - 1$, which finishes the proof.

    $$
    s=\widehat{C}(u_0,v_0)\geq 0.
    $$
    
    We would like to show that $C(u,v)-C(v,u)\leq \frac12-r$ for all $(u,v)\in\mathbb{I}^2$. To do so, we consider several cases (see Figure~\ref{fig:05}).

    \textbf{Case I.} Assume first that $u\leq v_0$ and $v\leq u_0$. Then the $1$-Lipshitz property implies that 
    $$
    C(u_0,v_0)-C(v,u)\leq u_0-v+v_0-u,
    $$
    and hence
    \begin{equation*}\label{3-star}
        C(v,u)\geq C(u_0,v_0)+u+v-u_0-v_0=r+s+u+v-u_0-v_0.
    \end{equation*}
    Since $C(u,v)\leq u$, $s\geq 0$ and $u_0+v_0\leq 1-r$, this inequality implies that
    $$
    C(u,v)-C(v,u)\leq u_0+v_0-r-s-v\leq u_0+v_0-r-v\leq 1-2r-v.
    $$
    Combing the latter inequality with Proposition \ref{pr:d} and Lemma \ref{le:simple}, respectively, we obtain
    $$
    C(u,v)-C(v,u)\leq \min\{v,1-2r-v\}\leq \tfrac12-r.
    $$

    \textbf{Case II.} Suppose that $u\leq v_0$ and $v\geq u_0$. From these assumptions and the $1$-Lipschitz property it follows that

    $$
    C(u,v)-C(v,u)\leq u_0-r-s\leq u_0-r\leq \tfrac12-r.
    $$

    \textbf{Case IV.} Now, we choose $(u,v)\in [v_0,1-u_0]\times [u_0,1-v_0]$. Observe that then $C(v,u)\geq C(u_0,v_0)=r+s$. 

    Assume first that $u\leq v$. Then $C(u,v)\leq u$ and Proposition \ref{pr:d} imply that
    $$
    C(u,v)-C(v,u)\leq u-r-s
    $$
    and
    $$
    C(u,v)-C(v,u)\leq v-u\leq 1-v_0-u,
    $$
    respectively. Combing these two inequalities with Lemma \ref{le:simple} and using that $s\ge 0$, $v_0\ge r$, we obtain
    $$
    C(u,v)-C(v,u)\leq \min\{u-r-s,1-v_0-u\}\leq \tfrac{1-v_0-r-s}{2}\leq \tfrac{1-v_0-r}{2}\leq \tfrac12-r.

    $$
    C(u,v)-C(v,u)\leq v-r-s
    $$

    $$
    C(u,v)-C(v,u)\leq u-v \leq 1-u_0-v.
    $$
    Then, combing these two inequalities with Lemma \ref{le:simple} and using that $s\ge 0$, $u_0\ge r$, we obtain
    $$
    C(u,v)-C(v,u)\leq \min\{v-r-s,1-u_0-v\}\leq \tfrac{1-u_0-r-s}{2}\leq \tfrac{1-u_0-r}{2}\leq \tfrac12-r.
    $$

    $$

    C(u,v)-C(v,u) \leq \min\{u-v_0-r,1-u\} \leq \tfrac{1-r-v_0}{2}\leq \tfrac12-r.

    Assume first that $u\leq v$. Then, $C(u,v)\leq u$, $C(v,u)\geq u+v-1$ and therefore $C(u,v)-C(v,u)\leq 1-v$. Since $(u_0,v_0)\in\Delta^1_r$ we have $u_0\leq1-r-v_0\leq1-2r$. Then Proposition \ref{pr:d} and Lemma \ref{le:simple} imply that

   so, for any point $\mathcal{P} \in \mathcal{A}\mathcal{B}\mathcal{C}\mathcal{D}\mathcal{E}\mathcal{F}$ there exist a point $\mathcal{T}$ on the union of the edges $\mathcal{B}\mathcal{C} \cup \mathcal{C}\mathcal{D} \cup \mathcal{D}\mathcal{E} \cup \mathcal{E}\mathcal{F}$ and $\lambda \in \mathbb{I}$ such that $\mathcal{P} = \lambda \mathcal{T}$. By Propositions \ref{pr:BC}, \ref{pr:CD}, \ref{pr:DE}, and \ref{pr:EF} there exists a copula $C \in \mathscr C$, such that $(\mu_R(C),\mu_T(C)) = \mathcal{T}$. By Proposition \ref{pr:radial} there exists a copula $C_\lambda \in \mathscr C$, such that $(\mu_R(C_\lambda),\mu_T(C_\lambda)) = \lambda \mathcal{T} = \mathcal{P}$, which finishes the proof.
