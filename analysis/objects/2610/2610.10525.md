\label{rem:multitype-passage}
Let $W_1,\ldots,W_d$ be independent standard Brownian motions and define
\[
    \boldsymbol{\mathcal X}_{\boldsymbol s}
      :=(W_1(s_1),\ldots,W_d(s_d))^{\top}
        -\boldsymbol B\boldsymbol s,
    \qquad \boldsymbol s\in\mathbb{R}_+^d.
\]
Because $\boldsymbol B$ is a nonsingular $M$-matrix, the off-diagonal coordinates of each constituent L\'evy process are nondecreasing.  Write
$\boldsymbol T_{\boldsymbol x}$ for the multivariate first-passage vector of
$\boldsymbol{\mathcal X}$ to level $-\boldsymbol x$, in the sense of (3.2)-(3.3) in 
\cite{ChaumontMarolleau2020}.  To verify the claimed identification, let
$\boldsymbol{\mathcal X}^{(j)}$ be the constituent L\'evy process run by
$s_j$.  Its Laplace exponent, under the convention of that reference, is
\[
 \varphi_j(\boldsymbol q)
 :=\log\mathbb{E}\left[e^{-\left\langle\boldsymbol q,
                    \boldsymbol{\mathcal X}^{(j)}_1\right\rangle}\right]
 =\left(\boldsymbol B^{\top}\boldsymbol q\right)_j
    +\frac12q_j^2,
 \qquad \boldsymbol q\in\mathbb{R}_+^d.
\]
 The first-passage exponent is the nonnegative inverse of $\boldsymbol\varphi=(\varphi_1,\ldots,\varphi_d)^{\top}$ by \cite[Theorem~3.3]{ChaumontMarolleau2020}.
Indeed, the hypotheses of that inversion theorem hold here:
if $\boldsymbol p=\boldsymbol B^{-\mathsf T}\boldsymbol1$, then
$\boldsymbol p>\boldsymbol0$ and
$\boldsymbol\varphi(\boldsymbol p)
 =\boldsymbol1+\boldsymbol p^{\circ2}/2>\boldsymbol0$.
Consequently, for any $\boldsymbol u\in(0,\infty)^d$,
\begin{equation}\label{eq:multitype-first-passage-LT}
 \mathbb{E}\left[e^{-\langle\boldsymbol u,
                       \boldsymbol T_{\boldsymbol x}\rangle}
          \1_{\{\boldsymbol T_{\boldsymbol x}\in\mathbb{R}_+^d\}}\right]
 =e^{-\langle\boldsymbol x,\boldsymbol z(\boldsymbol u)\rangle},
\end{equation}
because $\boldsymbol\varphi(\boldsymbol z(\boldsymbol u))
=\boldsymbol u$ is exactly \eqref{eq:multitype-IG-root}.  Continuity of
$\boldsymbol z$ and monotone convergence theorem extend this identity to the
boundary of $\mathbb{R}_+^d$.  In particular, sending
$\boldsymbol u\downarrow\boldsymbol0$ and using
$\boldsymbol z(\boldsymbol0)=\boldsymbol0$ shows that
$\boldsymbol T_{\boldsymbol x}$ is finite almost surely.  The stationary
independent increment property in the ordered level parameter
\cite[Proposition~3.1(2)]{ChaumontMarolleau2020}, together with the
continuity at zero of its Laplace exponent, gives a c\`adl\`ag
modification of the ray restriction
$(\boldsymbol T_{t\boldsymbol\eta})_{t\ge0}$.
We take this modification as the $\mathbb{R}_+^d$-valued subordinator.
Substituting
$\boldsymbol x=t\boldsymbol\eta$ into \eqref{eq:multitype-first-passage-LT} gives
\eqref{eq:multitype-IG-LT}; hence this ray restriction is a version of
$\boldsymbol Z$.

\label{thm:main}
Under \eqref{eq:multivariate-scaling}, for every $L>0$, as $T\to\infty$,
\begin{equation}\label{eq:main-convergence}
    \left(\boldsymbol X^T,\boldsymbol C^T\right)
       \Longrightarrow(\boldsymbol Z,\boldsymbol Z)
\end{equation}
in $\mathbb{D}([0,L],\mathbb{R}^{2d})$ equipped with the product $M_1$-topology.  Here $\boldsymbol Z$ is the multitype IG subordinator defined by
\eqref{eq:multitype-IG-LT}.  For every coordinate in which $Z_i$ is discontinuous with positive probability, neither corresponding marginal convergence in
\eqref{eq:main-convergence} can hold in the $J_1$-topology.

<!-- corollary-1: body_md -->
\label{cor:diagonal-IG}
If $\boldsymbol B=\operatorname{diag}(b_1,\ldots,b_d)$ with
$b_i>0$, then the coordinates of $\boldsymbol Z$ are independent classical IG subordinators and
\begin{equation}\label{eq:diagonal-IG-LT}
    \mathbb{E}\left[e^{-uZ_{i,t}}\right]
      =\exp\left\{-t\eta_i\left(\sqrt{b_i^2+2u}-b_i\right)\right\},
      \qquad u\ge0.
\end{equation}

Equation \eqref{eq:multitype-IG-root} separates coordinatewise and gives
$z_i(\boldsymbol u)=\sqrt{b_i^2+2u_i}-b_i$.  The exponent in
\eqref{eq:multitype-IG-LT} is therefore a sum of one-dimensional exponents, which proves independence and \eqref{eq:diagonal-IG-LT}.

 (X^T,A^T) \Longrightarrow (Y,Y)

      =\exp\left\{-t\vartheta\left(\sqrt{1+2u}-1\right)\right\},

\begin{align}
    \nu_{\mathrm{IG}}(\,\mathrm{d} x)

        e^{-x/2}\1_{(0,\infty)}(x)\,\mathrm{d} x

         Y_t\sim\operatorname{IG}\left(\vartheta t,\vartheta^2t^2\right),
    \qquad t>0. \label{eq:IG-Levy}
\end{align} 
Neither marginal convergence of \eqref{eq:201} can hold in the $J_1$-topology.

\label{lem:positive-resolvent-limit}
Assume $b_n\to\infty$, 
\eqref{eq:general-Schur-decomposition}-\eqref{eq:general-tilted-stability} and \eqref{eq:general-mass}.  Then the nonnegative measure
$\boldsymbol\Pi$ in \eqref{eq:general-potential} exists uniquely and
\begin{equation}\label{eq:relaxed-resolvent-convergence}
    \boldsymbol R^{(n)}(t)\,\mathrm{d} t \to\boldsymbol\Pi(\,\mathrm{d} t),
\end{equation}
entrywise vaguely on $\mathbb{R}_+$ as $n\to\infty$.  Moreover,  its exponentially tilted transforms satisfy that as $n\to\infty$, 
\begin{equation}\label{eq:positive-resolvent-transform}
 \mathcal L_{\boldsymbol R^{(n)}}(\lambda)
 =\frac1{b_n}
       \left(\boldsymbol I-
          \mathcal L_{\boldsymbol\phi_n}(\lambda/n)\right)^{-1}
       \mathcal L_{\boldsymbol\phi_n}(\lambda/n)
 \to \mathcal L_{\boldsymbol\Pi}(\lambda),
 \qquad \lambda\ge\lambda_0.
\end{equation}

<!-- proof-5: body_md -->
For $\lambda\ge\lambda_0$, let
$A_n(\lambda):=\mathcal L_{\boldsymbol\phi_n}(\lambda/n)$ and
$F_n(\lambda):=b_n(\boldsymbol K-A_n(\lambda))$.
Positivity gives $0\le A_n(\lambda)\le A_n(\lambda_0)$.
The spectral radius is monotone on nonnegative matrices, so
\eqref{eq:general-tilted-stability} and the Neumann series give
$\left(\boldsymbol I-A_n(\lambda)\right)^{-1}\ge0$ for large $n$.
Tonelli's theorem applied to the resolvent series then gives the
identity in \eqref{eq:positive-resolvent-transform}.

Let $\widetilde F_n:=\boldsymbol Q^{\top}F_n(\lambda)\boldsymbol Q$ and
$E_n:=\mathrm{Id}_{d-\ell}-\boldsymbol U_{\mathtt J\mathtt J}
       +b_n^{-1}\widetilde F_{n,\mathtt J\mathtt J}$.
In the $\boldsymbol Q$ coordinates,
\[
 \boldsymbol I-A_n(\lambda)
 \quad\text{has blocks}\quad
 \begin{pmatrix}
 b_n^{-1}\widetilde F_{n,\mathtt I\mathtt I}
   &-\boldsymbol U_{\mathtt I\mathtt J}
          +b_n^{-1}\widetilde F_{n,\mathtt I\mathtt J}\\
 b_n^{-1}\widetilde F_{n,\mathtt J\mathtt I}&E_n
 \end{pmatrix}.
\]
The block $E_n$ is invertible for large $n$.  Its Schur complement
$S_n$ satisfies
\[
 b_n S_n
 =\widetilde F_{n,\mathtt I\mathtt I}
  +\left(\boldsymbol U_{\mathtt I\mathtt J}
             -b_n^{-1}\widetilde F_{n,\mathtt I\mathtt J}\right)
      E_n^{-1}\widetilde F_{n,\mathtt J\mathtt I}
 \longrightarrow\boldsymbol G(\lambda).
\]
The block inverse formula therefore gives
\[
 \frac1{b_n}\left(\boldsymbol I-A_n(\lambda)\right)^{-1}
 \longrightarrow P(\lambda),
\]
where $P(\lambda)$ is the right-hand side of
\eqref{eq:general-potential}.  The case $\ell=d$ follows by
inverting $F_n(\lambda)$ directly.  Since
\[
 \frac1{b_n}\left(\boldsymbol I-A_n(\lambda)\right)^{-1}A_n(\lambda)
 =\frac1{b_n}\left(\boldsymbol I-A_n(\lambda)\right)^{-1}
     -\frac{\boldsymbol I}{b_n},
\]
the transforms in \eqref{eq:positive-resolvent-transform} converge to
$P(\lambda)$.  This function is continuous for $\lambda\ge\lambda_0$
by \eqref{eq:general-admissibility}.

For each entry, consider the finite positive measure
$e^{-\lambda_0 t}R_{ij}^{(n)}(t)\,\mathrm{d} t$.
Its mass converges to $P_{ij}(\lambda_0)$, and its Laplace transform
at $a\ge0$ converges to $P_{ij}(\lambda_0+a)$.
If the limiting mass is positive, normalize these measures by their
masses and apply the extended continuity theorem for Laplace transforms
\cite[Theorem~6.23]{Kallenberg2021}; if it is zero, the measures converge
to zero in total variation.  Thus they converge weakly to a finite
positive measure $\nu_{ij}$.  Let
$\Pi_{ij}(\,\mathrm{d} t):=e^{\lambda_0 t}\nu_{ij}(\,\mathrm{d} t)$.
These measures are locally finite, have transform $P$, and satisfy
\eqref{eq:relaxed-resolvent-convergence}.  Uniqueness follows from
uniqueness of the Laplace transform of the finite tilted measures
\cite[Theorem~6.3]{Kallenberg2021}.

\label{thm:relaxed-Xu}
Assume $b_n\to\infty$ and \eqref{eq:general-mass}--\eqref{eq:general-exogenous},
together with \eqref{eq:general-Schur-decomposition}--\eqref{eq:general-admissibility},
the common tilted-stability condition \eqref{eq:general-tilted-stability},
and the diagonal-atom condition \eqref{eq:general-diagonal-atom}.
Then the scaling conclusions of
\cite[Theorems~2.9, 2.12, and~2.13]{Xu2024Scaling} hold in this setting.
More precisely, there exist a nondecreasing process $\boldsymbol\Xi$ and a
square-integrable martingale $\boldsymbol M$ with the following properties.
\begin{enumerate}[label=(\roman*),leftmargin=*]
\item \emph{Process convergence.}
The rescaled processes satisfy
\begin{equation}\label{eq:general-S-limit}
   \left(\boldsymbol{\mathcal A}^{(n)},
         \boldsymbol{\mathcal N}^{(n)},
         \boldsymbol{\mathcal M}^{(n)}\right)
   \Longrightarrow
   (\boldsymbol\Xi,\boldsymbol\Xi,\boldsymbol M),
\end{equation}
in Jakubowski's $S$-topology.  The finite-dimensional distributions
also converge at every positive continuity point of
$\boldsymbol\Upsilon$.

\item \emph{Riccati equation and regularity.}
For row-vector test functions
\[
 \boldsymbol f\in C^1\left(\mathbb{R}_+;\mathbb C_-^d\right),
 \qquad
 \boldsymbol h\in C^1\left(\mathbb{R}_+;\mathrm i\mathbb{R}^d\right),
 \qquad
 \mathbb C_-:=\left\{z\in\mathbb C:\operatorname{Re}z\le0\right\},
\]
the prelimit solutions in \eqref{eq:local-prelimit-Riccati} converge
in $L^p(0,L)$, for every $L>0$ and $1\le p<\infty$, to the unique global
solution $\boldsymbol V\in\mathbb{D}(\mathbb{R}_+;\mathbb C_-^d)$ of
\begin{equation}\label{eq:general-Riccati}
    \boldsymbol V=\boldsymbol W*\boldsymbol\Pi,
    \qquad
    \boldsymbol W
       =\boldsymbol f+\frac12(\boldsymbol V+\boldsymbol h)^2.
\end{equation}
The solution $\boldsymbol V$ is continuous if
$\boldsymbol f(0)=\boldsymbol h(0)=\boldsymbol0$, or if
$t\mapsto\boldsymbol\Pi([0,t])$ is continuous on $\mathbb{R}_+$.

\item \emph{Characterization of the limiting processes.}
For the test functions and solution in \textup{(ii)}, the Fourier--Laplace
functional satisfies
\begin{equation}\label{eq:general-FLF}
    \mathbb{E}\!\left[
       \exp\left\{\boldsymbol f*\,\mathrm{d}\boldsymbol\Xi(t)
                 +\boldsymbol h*\,\mathrm{d}\boldsymbol M(t)\right\}
       \right]
    =\exp\left\{\boldsymbol W*\,\mathrm{d}\boldsymbol\Upsilon(t)\right\}.
\end{equation}
Here row vectors multiply matrix-valued kernels from the left, and, for
example,
\[
 \boldsymbol f*\,\mathrm{d}\boldsymbol\Xi(t)
 :=\int_{[0,t]}\boldsymbol f(t-s)\,\mathrm{d}\boldsymbol\Xi(s).
\]
The limiting processes are extended by zero to negative times, and
these integrals include any initial jump.  For smooth $\boldsymbol h$,
the martingale integral uses the same convention, as expressed by
the integration-by-parts formula in the proof below.
Moreover,
\begin{equation}\label{eq:general-Volterra}
    \boldsymbol\Xi
       =\boldsymbol\Upsilon+\boldsymbol\Pi*\boldsymbol M.
\end{equation}

\item \emph{Moment bounds and stochastic continuity.}
For every $t\ge0$,
\begin{equation}\label{eq:general-limit-moments}
 \mathbb{E}\left[|\boldsymbol\Xi(t)|_1\right]\le|\boldsymbol\Upsilon(t)|_1,
 \qquad
 \mathbb{E}\left[|\boldsymbol M(t)|_1^2\right]\le d|\boldsymbol\Upsilon(t)|_1.
\end{equation}
If $\boldsymbol\Upsilon$ is continuous on $\mathbb{R}_+$, then
$(\boldsymbol\Xi,\boldsymbol M)$ is stochastically continuous.
If it is continuous on $\mathbb{R}$, then also
$\boldsymbol\Xi(0)=\boldsymbol M(0)=\boldsymbol0$.
\end{enumerate}

\label{lem:local-convolution-vague}
Let $K_n$ and $K$ be nonnegative, locally finite matrix-valued measures on
$[0,\infty)$ such that $K_n\to K$ entrywise vaguely.  For every
row-vector function $g\in L^1_{\mathrm{loc}}(\mathbb{R}_+;\mathbb C^d)$ and
every $L>0$,
\begin{equation}\label{eq:local-fixed-convolution-limit}
    \|g*K_n-g*K\|_{L^1(0,L)}\longrightarrow0.
\end{equation}
The same assertion holds for column-vector measures.
<!-- end lemma-2 -->
<!-- proof-6: body_md -->
Fix $L>0$.  A continuous compactly supported majorant of
$\1_{[0,L]}$, positivity, and vague convergence give
\[
    C_L:=\sup_n m_{K_n}(L)+m_K(L)<\infty.
\]
Choose $q\in C_c((0,L);\mathbb C^d)$ with
$\|g-q\|_{L^1(0,L)}<\varepsilon$, and extend $q$ by zero to $\mathbb{R}$.
For every $t\in[0,L]$, the function $s\mapsto q(t-s)$ is continuous
and compactly supported on $[0,\infty)$.  Thus
$(q*K_n)(t)\to(q*K)(t)$, while
\[
    \left|(q*K_n)(t)-(q*K)(t)\right|_1
       \le C_L\|q\|_{L^\infty(\mathbb{R})}.
\]
The dominated convergence theorem gives convergence in $L^1(0,L)$
for this difference.  Applying \eqref{eq:local-measure-Young} to $g-q$
then yields
\[
 \limsup_{n\to\infty}\left\|g*K_n-g*K\right\|_{L^1(0,L)}
       \le C_L\|g-q\|_{L^1(0,L)}
       \le C_L\varepsilon.
\]
Letting $\varepsilon\downarrow0$ proves
\eqref{eq:local-fixed-convolution-limit}.  For column-vector measures,
the proof uses the same scalar convolution inequality.

                    +\|\boldsymbol h\|_{L^\infty(0,L)}^2\right)

\label{prop:local-Riccati-repair}
Let $b_n>0$ tend to infinity, and assume
\eqref{eq:relaxed-resolvent-convergence}.  For row-vector functions
$\boldsymbol f\in L^\infty_{\mathrm{loc}}(\mathbb{R}_+;\mathbb C_-^d)$ and
$\boldsymbol h\in L^\infty_{\mathrm{loc}}(\mathbb{R}_+;\mathrm i\mathbb{R}^d)$, suppose
\begin{equation}\label{eq:local-prelimit-Riccati}
 \boldsymbol V^{(n)}
       =\boldsymbol W^{(n)}*\boldsymbol R^{(n)},\qquad
 \boldsymbol W^{(n)}
       =\boldsymbol f+b_n^2\mathcal W
           \left(\frac{\boldsymbol V^{(n)}+\boldsymbol h}{b_n}\right),
 \qquad
 \mathcal W(z):=(e^{z_i}-1-z_i)_{i=1}^d,
\end{equation}
where $\boldsymbol V^{(n)}$ takes values in $\mathbb C_-^d$,
$\sup_n\left\|\boldsymbol V^{(n)}\right\|_{L^\infty(0,L)}<\infty$ for every $L>0$,
and $\boldsymbol V^{(n)}\to\boldsymbol V$ almost everywhere.
Let $\boldsymbol W:=\boldsymbol f+(\boldsymbol V+\boldsymbol h)^2/2$.
Then, for every $L>0$ and $1\le p<\infty$,
\begin{equation}\label{eq:local-Riccati-Lp}
 \left\|\boldsymbol V^{(n)}-\boldsymbol V\right\|_{L^p(0,L)}
 +\left\|\boldsymbol W^{(n)}-\boldsymbol W\right\|_{L^p(0,L)}
 \to 0,
\end{equation}
and $\boldsymbol V=\boldsymbol W*\boldsymbol\Pi$ almost everywhere.
Moreover, if the nonnegative column-vector measures
$\nu_n(\,\mathrm{d} s):=\boldsymbol H^{(n)}(s)\,\mathrm{d} s$ converge vaguely to
$\nu(\,\mathrm{d} s):=\,\mathrm{d}\boldsymbol\Upsilon(s)$, then
\begin{equation}\label{eq:local-FLF-L1}
 \left\|\boldsymbol W^{(n)}*\nu_n-\boldsymbol W*\nu\right\|_{L^1(0,L)}
 \to 0.
\end{equation}

first term of \eqref{eq:local-Riccati-Lp}.  The integral remainder in
Taylor's formula is

 e^z-1-z-\frac{z^2}{2}
      =\frac{z^3}{2}\int_0^1(1-s)^2e^{sz}\,\mathrm{d} s.
\]
Since $\operatorname{Re}\left(\boldsymbol V^{(n)}+\boldsymbol h\right)\le0$
coordinatewise, this formula implies
\begin{equation}\label{eq:Riccati-Taylor}
\begin{gathered}
 E_n:=b_n^2\mathcal W
          \left(\frac{\boldsymbol V^{(n)}+\boldsymbol h}{b_n}\right)
             -\frac12\left(\boldsymbol V^{(n)}+\boldsymbol h\right)^2,\\
 \|E_n\|_{L^\infty(0,L)}
   \le \frac{\|\boldsymbol V^{(n)}+\boldsymbol h\|_{L^\infty(0,L)}^3}
               {6b_n}
   \le\frac{C_L}{b_n}.
\end{gathered}
\end{equation}
Factoring the difference of the componentwise squares gives
\[
 \left\|\boldsymbol W^{(n)}-\boldsymbol W\right\|_{L^p(0,L)}
 \le L^{1/p}\|E_n\|_{L^\infty(0,L)}
        +C_L\left\|\boldsymbol V^{(n)}-\boldsymbol V\right\|_{L^p(0,L)} \to 0.
\]
Let $K_n(\,\mathrm{d} s):=\boldsymbol R^{(n)}(s)\,\mathrm{d} s$.

 &\le m_{K_n}(L)
          \left\|\boldsymbol W^{(n)}-\boldsymbol W\right\|_{L^1(0,L)}
       +\left\|\boldsymbol W*K_n-\boldsymbol W*\boldsymbol\Pi\right\|_{L^1(0,L)}
       \notag \to 0.
\end{align}
Combining this with \eqref{eq:local-Riccati-Lp} proves the limiting
Riccati equation almost everywhere.

       \to 0
\end{equation*}  
by \eqref{eq:local-Riccati-Lp} and

Let $J_n(t):=\int_0^t\boldsymbol H^{(n)}(s)\,\mathrm{d} s$.
The mean-intensity identity and Doob's $L^2$ maximal inequality give
\begin{equation}\label{eq:general-compactness-moments}
 \mathbb{E}\left[\boldsymbol{\mathcal A}^{(n)}(t)\right]
   =\mathbb{E}\left[\boldsymbol{\mathcal N}^{(n)}(t)\right]=J_n(t),
 \qquad
 \mathbb{E}\!\left[\sup_{0\le s\le L}
               \left|\boldsymbol{\mathcal M}^{(n)}(s)\right|_1^2\right]
       \le4d|J_n(L)|_1.
\end{equation}
The convergence in \eqref{eq:general-exogenous} and monotonicity bound
$J_n(L)$ uniformly, by first choosing a continuity point of
$\boldsymbol\Upsilon$ larger than $L$.
Thus the uniform-tightness and $S$-compactness arguments in
\cite[Proposition~5.3 and Lemma~5.4]{Xu2024Scaling} apply.
The counts and compensators have the same limit, since
\eqref{eq:count-compensator-difference} and
\eqref{eq:general-compactness-moments} imply
\[
 \mathbb{E}\!\left[\sup_{0\le s\le L}
    \left|\boldsymbol{\mathcal N}^{(n)}(s)
           -\boldsymbol{\mathcal A}^{(n)}(s)\right|_1^2\right]
       \le\frac{4d|J_n(L)|_1}{b_n^2}\to 0.
\]

For completeness, finite-dimensional identification at positive
continuity points of $\boldsymbol\Upsilon$ follows from time averages.
For $0\le s\le t$, the compensator mean and martingale isometry give
\begin{equation}\label{eq:general-mean-increments}
 \mathbb{E}\!\left[\left|\boldsymbol{\mathcal A}^{(n)}(t)
                      -\boldsymbol{\mathcal A}^{(n)}(s)\right|_1\right]
       =|J_n(t)-J_n(s)|_1,
 \qquad
 \mathbb{E}\!\left[\left|\boldsymbol{\mathcal M}^{(n)}(t)
                      -\boldsymbol{\mathcal M}^{(n)}(s)\right|_1^2\right]
       \le d|J_n(t)-J_n(s)|_1.
\end{equation}
For either process $Z^{(n)}$, let
$Z^{(n),\delta}(t):=\delta^{-1}\int_t^{t+\delta}Z^{(n)}(s)\,\mathrm{d} s$.
Monotonicity for the compensator, and Jensen's inequality for the
martingale, bound their differences from these averages by
$|J_n(t+\delta)-J_n(t)|_1$ and $d|J_n(t+\delta)-J_n(t)|_1$,
in first and second mean, respectively.
The averages converge jointly along an $S$-convergent subsequence
\cite[Proposition~C.3(4)]{Xu2024Scaling}.  Choose $t+\delta$ to be a
continuity point of $\boldsymbol\Upsilon$, pass to the limit in $n$,
and then let $\delta\downarrow0$.  The limiting time averages converge
to the path values by right-continuity.  This proves joint
finite-dimensional convergence at every positive continuity point
$t$ of $\boldsymbol\Upsilon$.
Integration by parts,
\[
 \boldsymbol f*\,\mathrm{d}\boldsymbol\Xi(t)
     =\boldsymbol f(0)\boldsymbol\Xi(t)
                       +\boldsymbol f'*\boldsymbol\Xi(t),
 \qquad
 \boldsymbol h*\,\mathrm{d}\boldsymbol M(t)
     =\boldsymbol h(0)\boldsymbol M(t)
                       +\boldsymbol h'*\boldsymbol M(t),
\]
and its prelimit counterparts consequently give convergence of the
smooth-test transforms at these times.
The same integration-by-parts argument applies to smooth test functions of the
counts, because their difference from the compensators tends to zero
uniformly in probability.  These convergences hold jointly for any
finite collection of test functions and times.
If $\boldsymbol f(0)=\boldsymbol h(0)=\boldsymbol0$, the fixed-kernel
convolution consequence of $S$ convergence gives these transforms at
every time, without an evaluation of either path at that time.
The uniform second-moment bound also preserves the martingale property
\cite[Proposition~C.7]{Xu2024Scaling}.

Fix smooth test functions $\boldsymbol f,\boldsymbol h$ and an integer $L\ge1$.
The prelimit solutions exist and take values in $\mathbb C_-^d$ by
\cite[Lemma~4.3(1)]{Xu2024Scaling}, and depend on the kernel and test functions,
independently of the baseline.  Lemma~\ref{lem:uniform-Riccati-bound}
provides a constant $C_{L+1}$ such that
$\left|V_j^{(n)}(t)\right|\le C_{L+1}$ on $[0,L+1]$ for all large $n$ and all $j$.
Choose $\varepsilon>0$ with $\varepsilon C_{L+1}<\pi/4$.
For each $j=1,\ldots,d$, consider an auxiliary Hawkes array with the
same kernel and baseline
\begin{equation}\label{eq:auxiliary-small-immigration}
 \boldsymbol\mu_{n,j}^{\varepsilon}(s)
        :=\varepsilon b_n(\phi_{n,ij}(s))_{i=1}^d,
 \qquad
 \boldsymbol H_{j}^{(n),\varepsilon}(t)
        =\varepsilon\boldsymbol R^{(n)}_{\cdot j}(t).
\end{equation}
The second identity follows from the resolvent equation.
Its cumulative exogenous activity converges to
$\boldsymbol\Upsilon_j^{\varepsilon}(t)
 :=\varepsilon\boldsymbol\Pi_{\cdot j}([0,t])$
at continuity points, by Lemma~\ref{lem:positive-resolvent-limit}.
Applying the preceding compactness and time-average arguments to these
$d$ auxiliary arrays, and extracting a common subsequence, gives
limits $(\boldsymbol\Xi_j^{\varepsilon},\boldsymbol M_j^{\varepsilon})$.
The exact affine formula in \cite[Lemma~4.3(2)]{Xu2024Scaling} gives
\begin{equation}\label{eq:auxiliary-small-transform}
 \exp\left\{\varepsilon V_j^{(n)}(t)\right\}
 =\mathbb{E}\!\left[\exp\left\{
       \boldsymbol f*\,\mathrm{d}\boldsymbol{\mathcal A}_{j}^{(n),\varepsilon}(t)
      +\boldsymbol h*\,\mathrm{d}\boldsymbol{\mathcal M}_{j}^{(n),\varepsilon}(t)
                         \right\}\right]
 \longrightarrow F_j^{\varepsilon}(t)
\end{equation}
outside the countable discontinuity set of
$\boldsymbol\Upsilon_j^{\varepsilon}$, where
\[
 F_j^{\varepsilon}(t)
 :=\mathbb{E}\!\left[\exp\left\{
      \boldsymbol f*\,\mathrm{d}\boldsymbol\Xi_j^{\varepsilon}(t)
      +\boldsymbol h*\,\mathrm{d}\boldsymbol M_j^{\varepsilon}(t)\right\}\right].
\]
This function is c\`adl\`ag: the random exponent has c\`adl\`ag paths by
the integration-by-parts identities, and its exponential has modulus
at most one, so the dominated convergence theorem applies to both
one-sided limits.

On $[0,L+1]$, the prelimit values in
\eqref{eq:auxiliary-small-transform} lie in the closed sector
\[
 \left\{z\in\mathbb C:
       e^{-\varepsilon C_{L+1}}\le|z|\le1,
       \ |\arg z|\le\varepsilon C_{L+1}\right\}.
\]
Their limits lie in the same sector almost everywhere, and
right-continuity extends this property to $[0,L+1)$.
The principal logarithm is continuous on this sector and satisfies
\[
 V_j^{(n)}(t)
    =\varepsilon^{-1}\operatorname{Log}
                            e^{\varepsilon V_j^{(n)}(t)}
 \longrightarrow
 V_j(t):=\varepsilon^{-1}\operatorname{Log}F_j^{\varepsilon}(t)
 \quad\text{almost everywhere on }(0,L+1).
\]
The resulting $\boldsymbol V$ is c\`adl\`ag on $[0,L+1)$ and takes
values in $\mathbb C_-^d$.  Repeating the extraction for $L=1,2,\ldots$ gives a
common diagonal subsequence.  The local representatives agree almost
everywhere on overlapping intervals as limits of the same sequence,
and hence agree everywhere by right-continuity.
Proposition~\ref{prop:local-Riccati-repair} gives
$\boldsymbol V=\boldsymbol W*\boldsymbol\Pi$ almost everywhere.
Both sides are right-continuous, so the equation holds at every time.
The weighted uniqueness estimate above identifies the same global
solution for every extracted subsequence; the subsequence criterion
therefore gives full-sequence convergence in each finite local $L^p$ norm.

For an accumulation point $(\boldsymbol\Xi,\boldsymbol M)$ of the
original array, \eqref{eq:general-exogenous} gives
$\boldsymbol H^{(n)}(s)\,\mathrm{d} s\to\,\mathrm{d}\boldsymbol\Upsilon(s)$ vaguely.
Equation~\eqref{eq:local-FLF-L1} gives a further subsequence of exponents
converging almost everywhere to $\boldsymbol W*\,\mathrm{d}\boldsymbol\Upsilon$.
Combining the exact affine formula with the smooth-test convergence
proved above establishes \eqref{eq:general-FLF} almost everywhere.
Both sides are right-continuous: the random exponential is bounded,
and the dominated convergence theorem applies to convolution of the
zero-extended c\`adl\`ag $\boldsymbol W$ against the locally finite
measure $\,\mathrm{d}\boldsymbol\Upsilon$.  Thus \eqref{eq:general-FLF} holds
at every time.
Taking $\boldsymbol f(t)=\int_0^t g(s)\,\mathrm{d} s$ and
$\boldsymbol h(t)=\int_0^t k(s)\,\mathrm{d} s$ for continuous
$\mathbb C_-^d$- and $\mathrm i\mathbb{R}^d$-valued functions $g,k$ determines
the joint laws of integrals of the paths against continuous test functions.
Right-sided averages recover all path values, so these laws determine
the joint c\`adl\`ag law.  Uniqueness of $\boldsymbol V$ therefore
identifies all accumulation points and proves full-sequence $S$ convergence.

Finally, let $F_n(t):=\int_0^t\boldsymbol R^{(n)}(s)\,\mathrm{d} s$ and
$F(t):=\boldsymbol\Pi([0,t])$.
Lemma~\ref{lem:positive-resolvent-limit}, positivity, and the dominated
convergence theorem give $F_n\to F$ in $L^1(0,L)$ for every $L$.
Integrating the prelimit martingale-resolvent identity
\cite[equations~(5.2) and~(5.13)]{Xu2024Scaling} yields
\begin{equation}\label{eq:integrated-general-Volterra}
 \int_0^t\boldsymbol{\mathcal A}^{(n)}(s)\,\mathrm{d} s
       =\int_0^t J_n(s)\,\mathrm{d} s
                      +F_n*\boldsymbol{\mathcal M}^{(n)}(t).
\end{equation}
Uniform $S$ tightness and the countable family of continuous functions
separating points give a further subsequence with an almost sure $S$
representation on each compact time interval
\cite[Theorem~1.1, Theorem~2.13(i), and Remark~2.16]{Jakubowski1997}.
The represented paths are uniformly bounded on that interval.
For each fixed $t$, the error bound
\[
 |F_n*\boldsymbol{\mathcal M}^{(n)}(t)-F*\boldsymbol M(t)|_1
 \le C_d\|F_n-F\|_{L^1(0,t)}
          \sup_{0\le s\le t}|\boldsymbol{\mathcal M}^{(n)}(s)|_1
      +|F*(\boldsymbol{\mathcal M}^{(n)}-\boldsymbol M)(t)|_1
 \longrightarrow0
\]
uses the convolution consequence of $S$ convergence
\cite[Proposition~C.1(2)]{Xu2024Scaling} for the fixed bounded kernel $F$.
Pass to the limit in \eqref{eq:integrated-general-Volterra}, first at
rational times and then at all times by continuity of the integrals.
Fubini's theorem gives
\[
 \int_0^t\boldsymbol\Xi(s)\,\mathrm{d} s
      =\int_0^t\boldsymbol\Upsilon(s)\,\mathrm{d} s+F*\boldsymbol M(t)
      =\int_0^t\left(\boldsymbol\Upsilon(s)
                                  +\boldsymbol\Pi*\boldsymbol M(s)\right)\,\mathrm{d} s.
\]
Right-continuity of the integrands proves \eqref{eq:general-Volterra}.
Fatou's lemma applied to \eqref{eq:general-compactness-moments}, first
at positive continuity points of $\boldsymbol\Upsilon$ and then along
such points decreasing to any $t\ge0$, gives
\eqref{eq:general-limit-moments}.
Passing \eqref{eq:general-mean-increments} to the limit at continuity
points of $\boldsymbol\Upsilon$, and then approaching arbitrary
$s,t$ from the right, gives
\[
 \mathbb{E}[|\boldsymbol\Xi(t)-\boldsymbol\Xi(s)|_1]
       \le|\boldsymbol\Upsilon(t)-\boldsymbol\Upsilon(s)|_1,
 \qquad
 \mathbb{E}[|\boldsymbol M(t)-\boldsymbol M(s)|_1^2]
       \le d|\boldsymbol\Upsilon(t)-\boldsymbol\Upsilon(s)|_1.
\]
Thus continuity of $\boldsymbol\Upsilon$ implies stochastic continuity
of both processes.  If $\boldsymbol\Upsilon$ is continuous on $\mathbb{R}$,
its zero extension gives $\boldsymbol\Upsilon(0)=\boldsymbol0$, and
\eqref{eq:general-limit-moments} gives zero initial values.

It remains to verify the asserted continuity of $\boldsymbol V$.
If $\boldsymbol f(0)=\boldsymbol h(0)=\boldsymbol0$, the
integration-by-parts formulas make the auxiliary random exponents
continuous in time; dominated convergence makes
$F_j^\varepsilon$ continuous, and its fixed logarithm is continuous.
Alternatively, if $t\mapsto\boldsymbol\Pi([0,t])$ is continuous on
$\mathbb{R}_+$, the increment estimates just proved apply to the auxiliary
arrays and make their limits stochastically continuous at positive
times.  The same integration-by-parts formulas give convergence in
probability of their exponents as time varies.  Their bounded
exponentials therefore have continuous expectations there.
Right-continuity at zero, already established, proves continuity of
$\boldsymbol V$ on $\mathbb{R}_+$ in this case as well.

\]

<!-- theorem-3: body_md -->
\label{thm:monotone-M1}
Let $\boldsymbol Z^n$ and $\boldsymbol Z$ be random elements of
$\mathbb{D}(\mathbb{R}_+,\mathbb{R}^m)$ with nonnegative, coordinatewise nondecreasing paths.
Suppose there is a dense set $D\subset[0,\infty)$ containing $0$ such
that, for every $k\ge1$ and every $t_1,\ldots,t_k\in D$,
\[
 \left(\boldsymbol Z^n_{t_1},\ldots,\boldsymbol Z^n_{t_k}\right)
 \Longrightarrow
 (\boldsymbol Z_{t_1},\ldots,\boldsymbol Z_{t_k}).
\]
Then $\boldsymbol Z^n\Longrightarrow\boldsymbol Z$ in
$\mathbb{D}(\mathbb{R}_+,\mathbb{R}^m)$ equipped with the product $M_1$ topology.
In particular, the restrictions converge in product $M_1$ on $[0,L]$
whenever $L>0$ satisfies
$\mathbb P\big(\Delta Z_L^j=0\text{ for all }j\big)=1$.
<!-- end theorem-3 -->
<!-- proof-10: body_md -->
For each coordinate $j$, let
$Y_t^j:=Z_t^j/(1+Z_t^j)$ and $m_j(t):=\mathbb{E}[Y_t^j]$.
Bounded convergence theorem and the c\`adl\`ag property show that $m_j$ is
c\`adl\`ag; it is also nondecreasing and hence has at most countably
many discontinuities. At a time $t>0$ where $m_j$ has no jump,
\[
 0=m_j(t)-m_j(t-)
   =\mathbb{E}\!\left[Y_t^j-Y_{t-}^j\right].
\]
The difference inside the expectation is nonnegative, so it vanishes
almost surely. Since $z\mapsto z/(1+z)$ is strictly increasing,
$\Delta Z_t^j=0$ almost surely. Let
\[
 \mathcal T:=\{0\}\cup
 \left\{t>0:\mathbb P\left(\Delta Z_t^j=0\text{ for all }j\right)=1\right\}.
\]
The complement of $\mathcal T$ is countable, and $\boldsymbol Z$ is
stochastically continuous at every time in $\mathcal T$.

We extend the assumed finite-dimensional convergence from $D$ to
$\mathcal T$. Fix $t\in\mathcal T\setminus\{0\}$ and choose
$s_k,u_k\in D$ with $s_k\uparrow t$ and $u_k\downarrow t$, where
$s_k<t<u_k$. Coordinatewise monotonicity gives
\[
 \boldsymbol0\le\boldsymbol Z^n_t-\boldsymbol Z^n_{s_k}
       \le\boldsymbol Z^n_{u_k}-\boldsymbol Z^n_{s_k}.
\]
For each fixed $k$, joint convergence at $s_k,u_k$, the continuous-mapping
theorem, and Portmanteau theorem imply, for every $\varepsilon>0$,
\[
 \limsup_{n\to\infty}
 \mathbb P\!\left(\left\|\boldsymbol Z^n_t-\boldsymbol Z^n_{s_k}\right\|_\infty
                  >\varepsilon\right)
 \le
 \mathbb P\!\left(\left\|\boldsymbol Z_{u_k}-\boldsymbol Z_{s_k}\right\|_\infty
                  \ge\varepsilon\right).
\]
The right-hand side tends to zero as $k\to\infty$, because
$\boldsymbol Z$ is almost surely continuous at $t$; also
$\boldsymbol Z_{s_k}\to\boldsymbol Z_t$ in probability. Applying this
approximation simultaneously at finitely many times, and using $0\in D$
when a time is zero, proves that, for every $q\ge1$ and every
$t_1,\ldots,t_q\in\mathcal T$,
\begin{equation}\label{eq:monotone-fdd-continuity}
 \left(\boldsymbol Z^n_{t_1},\ldots,\boldsymbol Z^n_{t_q}\right)
 \Longrightarrow
 \left(\boldsymbol Z_{t_1},\ldots,\boldsymbol Z_{t_q}\right).
\end{equation}

Fix $L\in\mathcal T$ with $L>0$. For each coordinate $j$,
$Z_L^{n,j}\Longrightarrow Z_L^j$, so that
$\sup_{t\le L}Z_t^{n,j}=Z_L^{n,j}$ is tight. The interior scalar $M_1$
oscillation modulus is zero for every nondecreasing path. Choose
$\delta_k\downarrow0$ such that both $\delta_k$ and $L-\delta_k$
belong to $\mathcal T$. The finite-dimensional convergence
\eqref{eq:monotone-fdd-continuity}
and Portmanteau theorem give
\begin{align*}
 \limsup_{n\to\infty}
 \mathbb P\left(Z_{\delta_k}^{n,j}-Z_0^{n,j}>\varepsilon\right)
 &\le \mathbb P\left(Z_{\delta_k}^j-Z_0^j\ge\varepsilon\right),\\
 \limsup_{n\to\infty}
 \mathbb P\left(Z_L^{n,j}-Z_{L-\delta_k}^{n,j}>\varepsilon\right)
 &\le \mathbb P\left(Z_L^j-Z_{L-\delta_k}^j\ge\varepsilon\right).
\end{align*}
Both right-hand sides tend to zero as $k\to\infty$: the first by
right-continuity at $0$, and the second because $L\in\mathcal T$.
The endpoint oscillations are nondecreasing in $\delta$, so control along
$\delta_k\downarrow0$ gives the full endpoint conditions. The scalar
$M_1$ tightness criterion \cite[Theorem~12.12.3]{Whitt2002} now applies
coordinatewise, and finite $m$ gives tightness in product $M_1$.

Joint finite-dimensional convergence at the stochastic continuity times
in $\mathcal T\cap[0,L]$ identifies every subsequential limit as
$\boldsymbol Z|_{[0,L]}$. Thus the restrictions converge in product
$M_1$ on every such interval. Since $\mathcal T$ is unbounded and has
countable complement, these restriction limits give convergence in the
product $M_1$ topology on $\mathbb{D}(\mathbb{R}_+,\mathbb{R}^m)$.

\label{cor:general-M1}
Under the hypotheses of Theorem~\ref{thm:relaxed-Xu}, suppose $\boldsymbol\Upsilon$ is continuous on $\mathbb{R}$.  Then, on every compact interval, as $n\to\infty$,
\begin{equation}\label{eq:general-M1-limit}
    \left(\boldsymbol{\mathcal A}^{(n)},
          \boldsymbol{\mathcal N}^{(n)}\right)
    \Longrightarrow(\boldsymbol\Xi,\boldsymbol\Xi),
\end{equation}
in the product $M_1$ topology.  On any interval $[0,L]$ where a coordinate $\Xi_i$ has discontinuous paths with positive probability, convergence of either corresponding marginal in $J_1$ is impossible whenever $N_{n,i}$ is simple.

<!-- proof-11: body_md -->
By Theorem~\ref{thm:relaxed-Xu}, the pair
$(\boldsymbol{\mathcal A}^{(n)},\boldsymbol{\mathcal N}^{(n)})$
has nonnegative, coordinatewise nondecreasing paths, and its
finite-dimensional distributions converge to those of
$(\boldsymbol\Xi,\boldsymbol\Xi)$ on a dense set
$D_0\subset(0,\infty)$ by the finite-dimensional assertion of
Theorem~\ref{thm:relaxed-Xu}. Both prelimit processes start at
$\boldsymbol0$. Moreover, continuity of $\boldsymbol\Upsilon$,
together with \eqref{eq:general-limit-moments}, gives
$\boldsymbol\Xi(0)=\boldsymbol0$ and stochastic continuity of
$\boldsymbol\Xi$. Hence the same joint finite-dimensional convergence
holds on $D:=D_0\cup\{0\}$. Applying Theorem~\ref{thm:monotone-M1}
to the $2d$-dimensional pair proves \eqref{eq:general-M1-limit}
on every $[0,L]$, since every $L>0$ is a stochastic continuity time
of the limit.

For the $J_1$ assertion, each $\mathcal A_i^{(n)}$ is continuous, whereas simplicity of $N_{n,i}$ gives
\[
    \sup_{0\le t\le L}
       \left|\Delta\mathcal N_i^{(n)}(t)\right|
       \le b_n^{-2}\longrightarrow0.
\]
If either marginal converged in $J_1$, the Skorokhod representation theorem would allow an almost surely $J_1$-convergent realization.  The jump-matching property of $J_1$ then implies that every jump of the limit must be approximated by jumps of the prelimit paths.  Continuous approximants, or approximants whose maximal jump tends to zero, can therefore have only a continuous $J_1$ limit.  This contradicts
\[
    \mathbb P\left(\Xi_i\text{ is discontinuous on }[0,L]\right)>0.
\]
The proof is complete.

The branching property and independence of the Poisson offspring counts give
\begin{equation}\label{eq:multitype-pgf}
    q_{T,j}(\boldsymbol u)
      =\exp\left\{-r_T^2u_j
          +\sum_{i=1}^d A_{T,ij}
             (q_{T,i}(\boldsymbol u)-1)\right\}.
\end{equation}
Taking logarithms, substituting
$\boldsymbol q_T=\boldsymbol1-r_T\boldsymbol z_T$, and dividing by
$r_T^2$ yields the exact vector identity
\begin{equation}\label{eq:multitype-exact-root}
 \boldsymbol u
  =\boldsymbol B_T^{\top}\boldsymbol z_T
    +\boldsymbol g_T(\boldsymbol z_T),
 \qquad
 g_{T,i}(z):=\frac{-\log(1-r_Tz)-r_Tz}{r_T^2}.
\end{equation}

Choose $\boldsymbol p>\boldsymbol0$ such that
$\boldsymbol B\boldsymbol p>\boldsymbol0$, which is possible for a nonsingular $M$-matrix.  For all sufficiently large $T$,
$\boldsymbol B_T\boldsymbol p\ge c\boldsymbol1$ for some $c>0$.
Taking the inner product of \eqref{eq:multitype-exact-root} with
$\boldsymbol p$ and using $g_{T,i}\ge0$ gives
\[
    \langle\boldsymbol p,\boldsymbol u\rangle
      \ge c\|\boldsymbol z_T(\boldsymbol u)\|_1.
\]
Thus $\boldsymbol z_T$ is locally bounded.  Along any convergent subsequence,
$r_T\boldsymbol z_T\to\boldsymbol0$ and
$g_{T,i}(z_{T,i})\to z_i^2/2$, so every subsequential limit solves
\eqref{eq:multitype-IG-root}.  If $\boldsymbol z$ and
$\widetilde{\boldsymbol z}$ are two nonnegative solutions, then
\[
 \left[\boldsymbol B^{\top}
   +\frac12\operatorname{diag}\left(\boldsymbol z+\widetilde{\boldsymbol z}\right)\right]
 \left(\boldsymbol z-\widetilde{\boldsymbol z}\right)=\boldsymbol0.
\]
The bracket is a nonsingular $M$-matrix, hence invertible.  Thus the
nonnegative solution is unique, and every convergent subsequence of
$\boldsymbol z_T(\boldsymbol u)$ has the same limit
$\boldsymbol z(\boldsymbol u)$.  Together with the local boundedness
proved above, this establishes the pointwise convergence
\eqref{eq:multitype-family-limit}.

We next verify continuity of $\boldsymbol z$.  The bound obtained by
taking the inner product with $\boldsymbol p$ also holds for the limiting
equation, so $\boldsymbol z(\boldsymbol u_k)$ is bounded whenever
$\boldsymbol u_k\to\boldsymbol u$.  Every convergent subsequence then
solves \eqref{eq:multitype-IG-root} at $\boldsymbol u$; uniqueness implies
$\boldsymbol z(\boldsymbol u_k)\to\boldsymbol z(\boldsymbol u)$.

To upgrade pointwise convergence to local uniform convergence, fix
$M>0$ and $\varepsilon>0$.  Uniform continuity of $\boldsymbol z$ on
$[0,M]^d$ permits a finite rectangular grid with vertex set $G$ such
that the lower and upper vertices $\boldsymbol v,\boldsymbol w$ of each
cell satisfy
$\|\boldsymbol z(\boldsymbol w)-\boldsymbol z(\boldsymbol v)\|_\infty
<\varepsilon$.  Since $G$ is finite, pointwise convergence gives
\[
 \max_{\boldsymbol a\in G}
 \|\boldsymbol z_T(\boldsymbol a)-\boldsymbol z(\boldsymbol a)\|_\infty
 <\varepsilon
\]
for all sufficiently large $T$.  Each component of $\boldsymbol z_T$ is
nondecreasing in every component of $\boldsymbol u$, as follows from
\eqref{eq:multitype-family-transform}; the same property holds for its
pointwise limit $\boldsymbol z$.  For a point $\boldsymbol u$ in a cell
with $\boldsymbol v\le\boldsymbol u\le\boldsymbol w$, we therefore have,
componentwise,
\[
 \boldsymbol z_T(\boldsymbol v)\le\boldsymbol z_T(\boldsymbol u)
       \le\boldsymbol z_T(\boldsymbol w),
 \qquad
 \boldsymbol z(\boldsymbol v)\le\boldsymbol z(\boldsymbol u)
       \le\boldsymbol z(\boldsymbol w).
\]
The grid-point error and the oscillation of $\boldsymbol z$ on each cell
consequently give
\[
 \sup_{\boldsymbol u\in[0,M]^d}
 \|\boldsymbol z_T(\boldsymbol u)-\boldsymbol z(\boldsymbol u)\|_\infty
 \le2\varepsilon
\]
for all sufficiently large $T$.  Every compact subset of $\mathbb{R}_+^d$ is
contained in such a cube, proving local uniform convergence.

\label{lem:multitype-collapse}
For every fixed $t\ge0$, componentwise,
\begin{equation}\label{eq:multitype-collapse-error}
 \boldsymbol0\le
 \mathbb{E}\left[\overline{\boldsymbol X}_t^T-\boldsymbol X_t^T\right]
 \le
 \boldsymbol B_T^{-1}\boldsymbol D_T
       \boldsymbol B_T^{-1}\boldsymbol\mu_T
 \longrightarrow\boldsymbol0.
\end{equation}
Moreover, for $0\le s\le t\le L$,
\begin{equation}\label{eq:multitype-increment-bound}
 \mathbb{E}\left[\boldsymbol X_t^T-\boldsymbol X_s^T\right]
 \le (t-s)Tr_T\boldsymbol B_T^{-1}\boldsymbol\mu_T.
\end{equation}
Consequently, $(\boldsymbol X^T)$ is asymptotically tight as
$T\to\infty$ in product $M_1$ on every compact interval.

The difference in \eqref{eq:multitype-collapse-error} consists of descendants whose immigrant arrived before $Tt$ but whose own birth occurs after $Tt$.  For any nonnegative locally finite measure $\nu$ on $[0,\infty)$, Tonelli's theorem gives the tail integral identity
\begin{equation}\label{eq:tail-integral-identity}
 \int_0^\infty\nu((v,\infty))\,\mathrm{d} v
 =\int_{[0,\infty)}
      \left(\int_0^\infty\1_{\{v<s\}}\,\mathrm{d} v\right)\nu(\,\mathrm{d} s)
 =\int_{[0,\infty)}s\nu(\,\mathrm{d} s).
\end{equation}
Applying Campbell's formula to each type's immigrant Poisson process
\cite[Proposition~2.7]{LastPenrose2017}, and then applying
\eqref{eq:tail-integral-identity} entrywise to $\boldsymbol U_T$, gives
\begin{align*}
 \mathbb{E}\left[\overline{\boldsymbol X}_t^T-\boldsymbol X_t^T\right]
 &=r_T^2\int_0^{Tt}
       \boldsymbol U_T((Tt-s,\infty))\boldsymbol\mu_T\,\mathrm{d} s\\
 &\le r_T^2\int_0^\infty
       \boldsymbol U_T((v,\infty))\boldsymbol\mu_T\,\mathrm{d} v
       =r_T^2\left(\int_0^\infty s\boldsymbol U_T(\,\mathrm{d} s)\right)
       \boldsymbol\mu_T.
\end{align*}
Using \eqref{eq:matrix-age-moment} and
$(\boldsymbol I-\boldsymbol A_T)^{-1}=r_T^{-1}\boldsymbol B_T^{-1}$
yields the upper bound in \eqref{eq:multitype-collapse-error}.  The matrices
$\boldsymbol B_T^{-1}$ are uniformly bounded, $\boldsymbol D_T$ is bounded by assumption, and
$\boldsymbol\mu_T=O((Tr_T)^{-1})$, so the bound vanishes.

The mean intensity of the process started from empty history is bounded componentwise by
$(\boldsymbol I-\boldsymbol A_T)^{-1}\boldsymbol\mu_T$.  Integrating over $(Ts,Tt]$ and multiplying by $r_T^2$ gives
\eqref{eq:multitype-increment-bound}.  Its right-hand side converges to
$(t-s)\boldsymbol B^{-1}\boldsymbol\eta$.  Each coordinate of
$\boldsymbol X^T$ is nonnegative and nondecreasing, so compact containment and the two endpoint controls follow from this bound and Markov's inequality.  The scalar monotone-path $M_1$ criterion applied coordinatewise proves product-$M_1$ tightness.

\label{prop:forest-passage}
For every fixed $t\ge0$, we have as $T\to\infty$, 
\begin{equation}\label{eq:forest-passage}
    r_T^2\sum_{j=1}^{K_T(t)}S_{T,j}
    \Longrightarrow
    \tau_{\vartheta t}
    :=\inf\{s\ge0:\vartheta t+B_s-s<0\},
\end{equation}
where $B$ is standard Brownian motion.  Moreover,
\begin{equation}\label{eq:forest-passage-LT}
    \mathbb{E}\left[e^{-u\tau_{\vartheta t}}\right]
    =\exp\left\{-\vartheta t
      \left(\sqrt{1+2u}-1\right)\right\}.
\end{equation}
The strict-crossing convention makes $x\mapsto\tau_x$ the
right-continuous inverse of $s\mapsto\sup_{0\le v\le s}(v-B_v)$.
For each fixed $x\ge0$, it agrees almost surely with the first hitting
time of level $x$.  As the initial level varies,
$(\tau_x)_{x\ge0}$ is the Brownian first-passage subordinator; hence
$(\tau_{\vartheta t})_{t\ge0}$ has the same c\`adl\`ag law as $Y$.

Let

\]
The total progeny of the forest is the first $n$ for which $Q_n^T=0$.  By the scaling assumptions of Corollary~\ref{cor:scalar-IG},
\[

   \right]\le

   \lfloor S/r_T^2\rfloor
   \mathbb{E}\left[(\xi_{T,1}-a_T)^4\right]
   \longrightarrow0.

    \Longrightarrow \vartheta t+B_s-s,
\]
locally in $J_1$.

By the strong Markov property, $\widehat B_s:=B_{\widehat\tau_x+s}-B_{\widehat\tau_x}$ is standard Brownian motion.  The reflection principle then gives

\eqref{eq:IG-LT} gives \eqref{eq:forest-passage-LT}.

in the level $x$.  Together with the right-continuous inverse convention,

\label{prop:universality}
For the age-dependent branching cluster process just defined, assume that conditions in Corollary~\ref{cor:scalar-IG} and   \eqref{eq:offspring-expansion} holds.  Then, for every $L>0$, as $T\to\infty$,

\begin{equation}\label{eq:universal-exponent}
    \psi_\sigma(u)
    =\frac{\vartheta}{\sigma^2}
      \left(\sqrt{1+2\sigma^2u}-1\right),
\end{equation}

\begin{equation}\label{eq:universal-Levy}
    \nu_\sigma(\,\mathrm{d} x)

\end{equation}

\begin{equation}\label{eq:universal-mean-check}
