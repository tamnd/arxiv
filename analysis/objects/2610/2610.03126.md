\label{def:o-minimal}
An \emph{o‑minimal structure} on the ordered field \((\mathbb{R},<,+,\cdot,0,1)\) is a sequence 
\(
\mathcal{S}=\bigl(\mathcal{S}_{n}\bigr)_{n\ge 1}
\)
such that for each positive integer \(n\)

\begin{enumerate}
\item \(\mathcal{S}_{n}\) is a Boolean algebra of subsets of \(\mathbb{R}^{n}\) (closed under finite unions, intersections, and complements);
\item if \(A\in\mathcal{S}_{m}\) and \(B\in\mathcal{S}_{n}\), then \(A\times B\in\mathcal{S}_{m+n}\);
\item if \(A\in\mathcal{S}_{n+1}\), the projection
      \(\pi(A)=\{\,x\in\mathbb{R}^{n}\mid\exists y\in\mathbb{R}:(x,y)\in A\}\)
      lies in \(\mathcal{S}_{n}\);
\item \(\mathcal{S}_{n}\) contains every algebraic subset of \(\mathbb{R}^{n}\)
      (i.e.\ every zero set of a real polynomial in \(n\) variables);
\end{enumerate}

A set $A\subset\mathbb{R}^n$ is called \emph{definable} (in $\mathcal{S}$) if $A\in\mathcal{S}_n$.
A function is \emph{definable} if its graph is a definable subset of $\mathbb{R}^{n+1}$.

The structure $\mathcal{S}$ is called \emph{o-minimal} if, in addition,
every set in $\mathcal{S}_1\subset \mathbb{R}$ is a finite disjoint union of open intervals and points.

\label{def:cdcd}
A \emph{cylindrical definable cell decomposition} (cdcd) of $\mathbb{R}^d$ of $\mathbb{R}^d$ is a finite partition $\mathcal{C}_d$ of $\mathbb{R}^d$ into definable sets, called \emph{cells}, satisfying a cylindrical condition defined inductively by dimension $d$. 

For $d=1$, a cdcd of $\mathbb{R}$ is determined by a finite (possibly empty)
set of points $x_1<x_2<\cdots<x_p$ in $\mathbb{R}$. Its cells are the
singletons $\{x_1\},\dots,\{x_p\}$ and the open intervals
\[
  (-\infty,x_1),\quad (x_1,x_2),\quad\dots,\quad (x_{p-1},x_p),
  \quad (x_p,+\infty).
\]
If $p=0$ the unique cell is $\mathbb{R}$ itself.

For $d>1$ a cdcd $\mathcal{C}_d$ of $\mathbb{R}^d$ consists of a cdcd
$\mathcal{C}_{d-1}$ of $\mathbb{R}^{d-1}$ together with, for each cell
$D\in\mathcal{C}_{d-1}$, an integer $m_D\ge 0$ and definable continuous functions
\[
  \zeta_{D,1}<\zeta_{D,2}<\cdots<\zeta_{D,m_D}\;:\;D\longrightarrow\mathbb{R} ,
\]
where the inequalities are required to hold pointwise on $D$. Setting
$\zeta_{D,0}\equiv-\infty$ and $\zeta_{D,m_D+1}\equiv+\infty$, the cells
of $\mathcal{D}$ lying over $D$ are the \emph{graphs}
\[
  \Gamma(\zeta_{D,l}), \qquad 1\le l\le m_D ,
\]
and the \emph{bands}
\[
  (\zeta_{D,l},\zeta_{D,l+1})
  \;=\;\bigl\{(x,y)\in D\times\mathbb{R} \;\bigm|\;
  \zeta_{D,l}(x)<y<\zeta_{D,l+1}(x)\bigr\},

\label{def:ReebQuotient}
Let $A \in Set(d)$ and $v \in S^{d-1}$. Define an equivalence relation $\sim_v$ on $A$ by
declaring that for $x,y \in A$,
\[
x \sim_v y
\quad\Longleftrightarrow\quad
v\cdot x = v\cdot y
\ \text{and}\ 
x,y \text{ lie in the same connected component of } LS(A,v,v\cdot x).
\]
The \emph{Reeb quotient space} of $A$ in direction $v$ is the quotient topological space
\[
\mathcal R_v(A) \ :=\ A/\!\sim_v,
\]
with quotient map $q_v:A\to \mathcal R_v(A)$. The induced height function
$\tilde h_v:\mathcal R_v(A)\to \mathbb{R}$ is defined by $\tilde h_v([x])=v\cdot x$.

<!-- definition-12: body_md -->
\label{def:ReebGraph}

$\mathcal R_v(A)$ from Definition~\ref{def:ReebQuotient}, equipped with the following canonical
graph structure and the function $\tilde h_v$.

For a point $p\in \mathcal R_v(A)$, define its \emph{local degree} $\deg(p)$ to be the number of
connected components of $U\setminus\{p\}$, where $U$ is any sufficiently small neighbourhood of $p$
in $\mathcal R_v(A)$. (For a finite topological graph this number is independent of the choice of
sufficiently small $U$.)

Define the vertex set
\[
V \ :=\ \{\,p\in \mathcal R_v(A)\mid \deg(p)\neq 2\,\},
\]
and define the edge set $E$ as the set of connected components of $\mathcal R_v(A)\setminus V$.
Each edge is homeomorphic to an open interval, and $\tilde h_v$ restricts on each edge to a
homeomorphism to an open interval in $\mathbb{R}$ (possibly unbounded).

<!-- definition-15: body_md -->
\label{def:fd}
Let $(R_f,f)$ and $(R_g,g)$ be Reeb graphs equipped with their Reeb functions
$f:R_f\to \mathbb{R}$ and $g:R_g\to \mathbb{R}$.
For $x,x'\in R_f$, define
\[
d_f(x,x') \ :=\ \inf_{\gamma}\Big(\max_{t\in[0,1]} f(\gamma(t))-\min_{t\in[0,1]} f(\gamma(t))\Big),
\]
where the infimum ranges over all continuous paths $\gamma:[0,1]\to R_f$ with
$\gamma(0)=x$ and $\gamma(1)=x'$. Define $d_g$ analogously on $R_g$.

Given continuous maps $\phi:R_f\to R_g$ and $\psi:R_g\to R_f$, set
\[
G(\phi,\psi)
=\{(x,\phi(x)):\ x\in R_f\}\ \cup\ \{(\psi(y),y):\ y\in R_g\},
\]
and
\[
D(\phi,\psi)
=\sup_{(x,y),(x',y')\in G(\phi,\psi)}\frac12\,\bigl|d_f(x,x')-d_g(y,y')\bigr|.
\]
The \emph{functional distortion distance} between $(R_f,f)$ and $(R_g,g)$ is
\[
d_{\mathrm{FD}}\bigl((R_f,f),(R_g,g)\bigr)
=\inf_{\phi,\psi}\max\Big\{
D(\phi,\psi),\ \|f-g\circ\phi\|_\infty,\ \|f\circ\psi-g\|_\infty
\Big\},
\]
where the infimum ranges over all continuous maps $\phi:R_f\to R_g$ and $\psi:R_g\to R_f$ (see figure \ref{fig:functional-distortion-example}).

<!-- prop-3: body_md -->
\label{prop:single_dir_stability}
Let $A\in Set(d)$ be non-empty, and let $D>0$ be such that
\[
A\subset \overline{B}(0,D)
:=\{x\in\mathbb{R}^d:\|x\|_2\le D\},
\]
where $\overline{B}(0,D)$ denotes the closed Euclidean ball of radius $D$ centered at the origin.
Fix $v,w\in S^{d-1}$. Then
\[
d_{\mathrm{FD}}\!\Bigl(\bigl(\mathcal R_v(A),\tilde h_v\bigr),\ \bigl(\mathcal R_w(A),\tilde h_w\bigr)\Bigr)
\;\le\; D\,\|v-w\|_2
\;=\; 2D\,\sin\!\Bigl(\frac{\theta}{2}\Bigr),
\]
where $\theta=\arccos\langle v,w\rangle$ and $d_{\mathrm{FD}}$ is the functional distortion distance.
In particular, for small $\theta$ this bound is $O(D\,\theta)$.
<!-- end prop-3 -->
<!-- proof-2: body_md -->
Let $h_v,h_w:A\to \mathbb{R}$ be given by $h_v(x)=v\cdot x$ and $h_w(x)=w\cdot x$.
For any $x\in A$,
\[
|h_v(x)-h_w(x)|
=|(v-w)\cdot x|
\le \|v-w\|_2\,\|x\|_2
\le D\,\|v-w\|_2.
\]
Hence $\|h_v-h_w\|_\infty\le D\,\|v-w\|_2$.

By~\cite[Thm.\,4.1]{Bauer2014}, the Reeb graphs of $h_v$ and $h_w$ satisfy
\[
d_{\mathrm{FD}}\bigl((R_{h_v},\bar h_v),(R_{h_w},\bar h_w)\bigr)\ \le\ \|h_v-h_w\|_\infty.
\]
In our notation, $(R_{h_v},\bar h_v)=(\mathcal R_v(A),\tilde h_v)$ and
$(R_{h_w},\bar h_w)=(\mathcal R_w(A),\tilde h_w)$, so the claimed inequality follows.
Finally, since $\|v-w\|_2=\sqrt{2-2\langle v,w\rangle}=2\sin(\theta/2)$, we obtain the stated
trigonometric form.

<!-- lemma-2: body_md -->
\label{lem:collapsing}
Let $A\in Set(d,1)$. The set of collapsing directions of $A$ has measure zero in $S^{d-1}$.
<!-- end lemma-2 -->
<!-- proof-5: body_md -->
Let $\mathcal{C}$ be a $C^1$ cylindrical definable cell decomposition adapted to $A$.
Subdividing the edges if needed, each of the edges in $A$ are a cell in $\mathcal{C}$. This implies each (subdivided part of an) $1$--dimensional stratum
is a definable $C^1$ embedded curve; in particular, there exists a definable $C^1$ embedding
$\gamma_i:(0,\ell)\to \mathbb{R}^d$ whose image is $e_i$ and such that $\gamma_i'(s)\neq 0$ for all $s\in(0,\ell)$.
Choose any $s_0\in(0,\ell)$ and set $w=\gamma_i'(s_0)\neq 0$.

Suppose that $v\in S^{d-1}$ is such that $e_i\subset LS(A,v,t)$ for some $t\in\mathbb{R}$.
Then the function $h(s)=v\cdot \gamma_i(s)$ is constant on $(0,\ell)$, hence $h'(s)=0$ for all $s$.
In particular,
\[
0=h'(s_0)=v\cdot \gamma_i'(s_0)=v\cdot w.
\]
Therefore, any direction $v$ for which the entire edge $e_i$ is contained in a level set must lie in
\[
\{v\in S^{d-1}\mid v\cdot w=0\},
\]
which is a great $(d-2)$--sphere and thus has measure zero in $S^{d-1}$.

Finally, the set of collapsing directions for $A$ is contained in the union of these measure-zero sets
over the finitely many edges $e_1,\dots,e_k$, and hence has measure zero.

<!-- lemma-3: body_md -->
\label{lem:Setd1_inj}
The Reeb Transform is injective when restricted to $Set(d,1)$.
<!-- end lemma-3 -->
<!-- proof-6: body_md -->
By Theorem~\ref{thm:ect-inj}, it suffices to show that
\[
\mathcal{RT}(X)\cong \mathcal{RT}(Y)\quad\Longrightarrow\quad ECT(X)=ECT(Y).
\]

Let $C_X$ and $C_Y$ denote the sets of collapsing directions for $X$ and $Y$
(Definition~\ref{def:collapsing}). By Lemma~\ref{lem:collapsing}, the set
$C_X\cup C_Y\subset S^{d-1}$ has measure zero, hence its complement is dense in $S^{d-1}$.

Fix $v\in S^{d-1}\setminus (C_X\cup C_Y)$. Then no $1$--dimensional stratum of $X$ (or $Y$) lies in a
level set of $h_v$. Consider a connected component $C$ of the level set $LS(X,v,t)$. Suppose that $C$ does not contain any $0$-strata of $X$. This then implies that $C$ is a connected subset of some $1$-strata and thus contractible. If $C$ contains a $0$-strata $x$ then $C$ is a connected subset of $X$ containing $x$ but each of the $1$ strata attached to $x$ can only be partially contained in $C$. Thus no other $0$-strata are in $C$ and $C$ must be homotopic to a star graph which is contractible.

As the fibers are contractible, the Reeb quotient maps
\[
q_v^X:X\to \mathcal R_v(X)
\qquad\text{and}\qquad
q_v^Y:Y\to \mathcal R_v(Y)
\]
are homotopy equivalences. Moreover, by definition of $\tilde h_v$ (Definition~\ref{def:ReebQuotient}),
they satisfy
\[
\tilde h_v^X\circ q_v^X \;=\; h_v|_X
\qquad\text{and}\qquad
\tilde h_v^Y\circ q_v^Y \;=\; h_v|_Y.
\]

Since $\mathcal{RT}(X)\cong \mathcal{RT}(Y)$, for this direction $v$ there exists an isomorphism of
Reeb graphs in the sense of Definition~\ref{def:ReebIso}, i.e., a homeomorphism
\[
\Phi_v:\mathcal R_v(X)\longrightarrow \mathcal R_v(Y)
\]
such that $\tilde h_v^Y\circ \Phi_v=\tilde h_v^X$. Transporting along $q_v^X$ and $q_v^Y$, define
\[
\Psi_v := (q_v^Y)^{-1}\circ \Phi_v\circ q_v^X : X\longrightarrow Y.
\]
Then $\Psi_v$ is a homeomorphism and
\[
h_v|_Y\circ \Psi_v
=\tilde h_v^Y\circ q_v^Y\circ (q_v^Y)^{-1}\circ \Phi_v\circ q_v^X
=\tilde h_v^Y\circ \Phi_v\circ q_v^X
=\tilde h_v^X\circ q_v^X
=h_v|_X.
\]
In particular, for every $t\in\mathbb{R}$ we have
\[
\Psi_v\!\bigl(\{x\in X:\ x\cdot v\le t\}\bigr)=\{y\in Y:\ y\cdot v\le t\},
\]
so the sublevel sets are homeomorphic and hence have the same Euler characteristic. Therefore,
\[
ECT(X)(v,\cdot)=ECT(Y)(v,\cdot)
\qquad\text{for all }v\in S^{d-1}\setminus (C_X\cup C_Y).
\]


Finally, Corollary \ref{cor:inj_dense} yields $X=Y$.

We start our proof by observing that it is sufficient to prove the case when $v=e_1$. This is because we can construct  a change of basis matrix $X$ such that $Xv=e_1$. From $A\inSet(d)$ and $X$ invertible we have $XA\in Set(d)$. From the deformation retract computed for $X(A)$ in direction $e_1$, we can then construct the appropriate one for $v=e_1$ on $A$. This holds as $$\|h_v(x)-t\|=\|h_v(x-tv)\|=\|h_{e_1}(Xx-X(tv)))\|=\|h_{e_1}(Xx)-t\|.$$

definable homeomorphism $\Phi:|K|\to A$ such that
$\pi\circ\Phi=\pi|_{|K|}.$
 

Choose $\epsilon>0$ such that every vertex of $a\in K$ with $\pi|_{|K|}(a)\neq t$ has $\|\pi|_{|K|}(a)-t\|>\epsilon$.

which is the straight line from $x$ to $\sum_{i=0}^k \frac{\lambda_i}{\alpha} a_i$ as $s$ goes from $0$ to $1$.
The function $h^K(x,s)$ is defined for all $x\in K$ with $\pi_{|K|}(x)\in [t-\epsilon, t+\epsilon]$.
Observe that 
$\pi_{|K|}(h^K(x,s))=(1-s)\pi_{|K|}(x)+st$,
so $h^K(\Phi(y),s)$ is defined for all $y\in A$ with $f(y)\in [0,\epsilon]$ and it is continuous for $y\in f^{-1}([0,\epsilon])\times [0,1]$.

Build the deformation retract $h$ from $f^{-1}([0, \epsilon])$ to $f^{-1}(0)$ as 
$h(x,s)=\Phi^{-1}(h^K(\Phi(x),s))$.

\label{prop:degree}
[Euler characteristic of slices from the Reeb graph]\label{prop:chilevel} Let $A\in Surface(3)$ and let $v\in S^{2}$ and $t\in \mathbb{R}$ (see Figure~\ref{fig:injectivity3D}).                                                                                    
                                                                                                                            
Let $(\mathcal R_v(A),\tilde h_v)$ be the Reeb graph 
$A$ in direction $v$
(Definition~\ref{def:ReebGraph}), and let $V(\mathcal R_v(A))$ denote its vertex set.
Then
\begin{equation}\label{eq:chi_deg}
\chi\!\bigl(LS(A,v,t)\bigr)
\;=\;
\sum_{\substack{p\in V(\mathcal R_v(A))\\ \tilde h_v(p)=t}}\bigl(2-\deg(p)\bigr),
\end{equation}
where $\deg(p)$ is the local degree from Definition~\ref{def:ReebGraph}.

Equivalently, if $C(A,v,s)$ denotes the number of connected components of $LS(A,v,s)$, then
\begin{equation}\label{eq:chi}
\chi\!\bigl(LS(A,v,t)\bigr)
\;=\;
2\,C(A,v,t)\;-\;C(A,v,t-\varepsilon)\;-\;C(A,v,t+\varepsilon).
\end{equation}
for $\epsilon>0$ sufficiently small.

<!-- proof-8: body_md -->
Let $q_v:A\to \mathcal R_v(A)$ be the Reeb quotient map. By construction,
the connected components of $LS(A,v,t)$ are precisely the $\sim_v$--equivalence classes at height $t$,
hence are in bijection with the fiber $\tilde h_v^{-1}(t)\subset \mathcal R_v(A)$.
Write
\[
LS(A,v,t)=X_1\sqcup\cdots\sqcup X_r,
\qquad
p_j:=q_v(X_j)\in \tilde h_v^{-1}(t).
\]
Euler characteristic is additive over disjoint unions, so it suffices to compute $\chi(X_j)$ in terms
of the local behavior of $(\mathcal R_v(A),\tilde h_v)$ at $p_j$.

We have $h_v^{-1}[t-\epsilon, t+\epsilon]$ splits into disjoint connected components, $Y_1, \ldots, Y_r$ such that $X_j\subset Y_j$. For sufficiently small $\epsilon$ we have the restriction of $\mathcal R_v(A)$ to heights $[t-\epsilon, t+\epsilon]$ is the disjoint union of the Reeb graphs $\mathcal{R}_v(Y_j)$. 

From now on let us restrict to a single $Y_j$. By Lemma \ref{lem:retract}, for $\epsilon$ small enough we have $Y_j$ deformation retracts to $X_j$ and since both $X_j$ and $Y_j$ are compact this deformation retract implies $\chi(X_j)=\chi(Y_j)$ 
(note that the Euler characteristic used in the ECT and in its injectivity proofs all involve the definable version of Euler Characteristic which thankfully agrees with the standard version when the sets are compact). For $\epsilon$ small, $t-\epsilon$ and $t+\epsilon$ are not critical and we have $\partial Y_j=LS(Y_j, v, t-\epsilon) \cup LS(Y_j, v, t+\epsilon)$ is compact and homotopic to a disjoint union of $deg(p_j)$ copies of $S^1$. 

As the surface $Y_j$ is homotopic to a subset of $\mathbb{R}^2$ we further can infer that it has genus $0$. Thus, $Y_j$ is a surface with boundary with genus $0$ and $\deg(p_j)$ boundary curves, so from the classification of surfaces with boundary $\chi(Y_i)=2-\deg(p_i)$.

<!-- lemma-5: body_md -->
\label{lem:surf_separates}
Let $A\in Surface(3)$ be connected. Then $\mathbb{R}^{3}\setminus A$ has exactly two path-connected components, one bounded and one unbounded.
<!-- end lemma-5 -->
<!-- proof-9: body_md -->
We use Alexander duality for the one-point compactification $S^{3}=\mathbb{R}^{3}\cup\{\infty\}$.
Throughout the proof, the notation $H_i(X;G)$ and $H^i(X;G)$ denotes singular homology and cohomology of $X$ with coefficients in an abelian group $G$. We also write $\widetilde H_i(X;G)$ for \emph{reduced} homology. 
In degree $0$, reduced homology removes the ubiquitous copy of $G$ coming from nonemptiness: if $X$ has $k$ connected components, then
\[
H_0(X;G)\cong G^{\,k}
\qquad\text{and}\qquad
\widetilde H_0(X;G)\cong G^{\,k-1}.
\]
In particular, $\widetilde H_0(X;G)\cong G$ holds if and only if $X$ has exactly two connected components.

Since $A$ is definable and compact, it is locally contractible, so Alexander duality applies.
Using coefficients $G=\mathbb{Z}/2$ (to avoid orientability issues), Alexander duality gives
\[
\widetilde H_{0}(S^{3}\setminus A;\mathbb{Z}/2)\ \cong\ H^{2}(A;\mathbb{Z}/2).
\]
Because $A$ is homeomorphic to a connected compact surface, its top cohomology with $\mathbb{Z}/2$ coefficients satisfies $H^{2}(A;\mathbb{Z}/2)\cong \mathbb{Z}/2$. Therefore
\[
\widetilde H_{0}(S^{3}\setminus A;\mathbb{Z}/2)\cong \mathbb{Z}/2,
\]
which implies that $S^{3}\setminus A$ has exactly two connected components.

Finally, $\mathbb{R}^{3}\setminus A$ is definable and hence locally path-connected, so its connected components coincide with its path-connected components. One component contains $\infty$ and corresponds to the unbounded component of $\mathbb{R}^{3}\setminus A$. The other component is bounded since $A$ is compact.

<!-- definition-18: body_md -->
\label{def:int_ext}
Let $A\in Surface(3)$ be connected. By Lemma~5, the complement $\mathbb{R}^{3}\setminus A$ has exactly two
path-connected components: one bounded and one unbounded. Denote the bounded component by
$\omega_{\mathrm{int}}$ and the unbounded component by $\omega_{\mathrm{ext}}$.

We call $\omega_{\mathrm{int}}$ the \emph{open interior region} of $A$ and $\omega_{\mathrm{ext}}$ the
\emph{open exterior region} of $A$. We define the \emph{closed interior region} and
\emph{closed exterior region} of $A$ by
\[
A^{\mathrm{int}} := \overline{\omega_{\mathrm{int}}}
\qquad\text{and}\qquad
A^{\mathrm{ext}} := \overline{\omega_{\mathrm{ext}}},
\]
where the closure is taken in $\mathbb{R}^{3}$.

<!-- thm-3: body_md -->
\label{theorem:ext_int}
Let $A\in Surface(3)$ be connected, and let $A^{\mathrm{int}}$ and $A^{\mathrm{ext}}$ be the closed
interior and exterior regions from Definition~\ref{def:int_ext}. 
\footnote{Although $A^{\mathrm{ext}}$ is unbounded, it is closed and definable (being the closure of
a definable open set). In this section we use the directional Reeb quotient/graph (and hence the
Reeb Transform) for such closed definable sets whenever the quotient is a definable graph. This holds for $A^{\mathrm{ext}}$ in our setting because $A^{\mathrm{ext}}$ is closed and definable and,
for each $v\in S^{2}$ and $t\in\mathbb{R}$, the slice $A^{\mathrm{ext}}\cap\{x\cdot v=t\}$ is a definable subset
of the plane and hence has finitely many connected components. In particular, the Reeb quotient
$\mathcal R_v(A^{\mathrm{ext}})$ is a definable one-dimensional graph-like space (possibly with unbounded
edges); see~\cite{basu2022reeb} and references therein.}
Then $A$ (as a subset of $\mathbb{R}^3$) is uniquely determined by the pair
$\bigl(\mathcal{RT}(A^{\mathrm{int}}),\,\mathcal{RT}(A^{\mathrm{ext}})\bigr)$(see Figure~\ref{fig:interiorRegions}).

Let $B\in Surface(3)$ be another connected surface such that
\[
\mathcal{RT}(A^{\mathrm{int}})\cong\mathcal{RT}(B^{\mathrm{int}})
\qquad\text{and}\qquad
\mathcal{RT}(A^{\mathrm{ext}})\cong\mathcal{RT}(B^{\mathrm{ext}}).
\]
We will prove $A=B$.

For any set $X\subset \mathbb{R}^3$, direction $w\in S^{2}$, and height $s\in\mathbb{R}$, write
\[
C(X,w,s)\ :=\ \text{the number of connected components of }LS(X,w,s).
\]

\smallskip
\noindent\textbf{Step 1: A component identity valid for all heights.}
Fix $w\in S^{2}$ and $s\in\mathbb{R}$, and let
\[
P:=\{x\in\mathbb{R}^{3}:x\cdot w=s\}\cong \mathbb{R}^{2}.
\]
Set
\[
X_{\mathrm{int}}:=LS(A^{\mathrm{int}},w,s)=A^{\mathrm{int}}\cap P,\qquad
X_{\mathrm{ext}}:=LS(A^{\mathrm{ext}},w,s)=A^{\mathrm{ext}}\cap P,\qquad
K:=LS(A,w,s)=A\cap P.
\]
Since $A^{\mathrm{int}}\cup A^{\mathrm{ext}}=\mathbb{R}^{3}$ and $A^{\mathrm{int}}\cap A^{\mathrm{ext}}=A$,
intersecting with $P$ yields
\[
P = X_{\mathrm{int}}\cup X_{\mathrm{ext}}
\qquad\text{and}\qquad
K = X_{\mathrm{int}}\cap X_{\mathrm{ext}}.
\]

                                                         
                                                                                                      
                                                                              
                                         
    
                                                                 
    
                                                                                       

If $K=\emptyset$, then $P\subset \mathbb{R}^{3}\setminus A$. Since $P$ is connected, it must lie entirely in one of the two connected components $\omega_{\mathrm{int}}$ or $\omega_{\mathrm{ext}}$ of $\mathbb{R}^{3}\setminus A$.
However, $\omega_{\mathrm{int}}$ is bounded, whereas $P\cong\mathbb{R}^{2}$ is unbounded, so $P$ cannot be contained in $\omega_{\mathrm{int}}$. 
Hence
\[
P\subset \omega_{\mathrm{ext}}.
\]
Since $P\cap A=\emptyset$, it follows that
\[
X_{\mathrm{int}}=\emptyset
\qquad\text{and}\qquad
X_{\mathrm{ext}}=P.
\]
Therefore,
\[
C(A,w,s)=0,
\qquad
C(A^{\mathrm{int}},w,s)=0,
\qquad
C(A^{\mathrm{ext}},w,s)=1,
\]
and hence
\[
C(A,w,s)
=
C(A^{\mathrm{int}},w,s)
+
C(A^{\mathrm{ext}},w,s)
-1.
\]
Thus~\eqref{eq:component_identity} holds when $K=\emptyset$.
Henceforth assume $K\neq\emptyset$.



Because $X_{\mathrm{int}},X_{\mathrm{ext}},K$ are closed definable subsets of the plane $P$,
they are locally compact and locally contractible by \cite[Theorem ~4.4]{coste1999introduction}. There is a definable homeomorphism $\Phi$ from $\mathbb{R}^3$ to a bounded subset. The sets $\Phi(X_{\mathrm{int}})$, $\Phi(X_{\mathrm{ext}})$ and $\Phi(K)$ are each homeomorphic to finite simplicial complexes and hence each admits an open $\varepsilon$-neighborhood ($N_\varepsilon(\Phi(X_{\mathrm{ext}}))$, $N_\varepsilon(\Phi(X_{\mathrm{ext}}))$ and $N_\varepsilon(\Phi(K))$ respectively) that strong deformation retracts onto it \cite[Proposition ~A.5]{Hatcher2002}.

Setting $U_{\mathrm{int}} = \Phi^{-1}(N_\varepsilon(\Phi(X_{\mathrm{int}}))$ and $U_{\mathrm{ext}} = \Phi^{-1}(N_\varepsilon(\Phi(X_{\mathrm{ext}})))$ we then, by conjugating the previous deformation retracts with $\Phi$, we also have each of
$X_{\mathrm{int}} \hookrightarrow U_{\mathrm{int}}$ and
$X_{\mathrm{ext}} \hookrightarrow U_{\mathrm{ext}}$ is a strong deformation
retract. Moreover $U_{\mathrm{int}} \cup U_{\mathrm{ext}} \supseteq
X_{\mathrm{int}} \cup X_{\mathrm{ext}} = \mathbb{R}^2 $.
By construction
\begin{align*}
U_{\mathrm{int}} \cap U_{\mathrm{ext}} &=\Phi^{-1}(N_\varepsilon(\Phi(X_{\mathrm{int}})) \cap \Phi^{-1}(N_\varepsilon(\Phi(X_{\mathrm{ext}})))\\
&=\Phi^{-1}(N_\varepsilon(\Phi(X_{\mathrm{int}})\cap (N_\varepsilon(\Phi(X_{\mathrm{ext}}))\\
&=\Phi^{-1}(N_\varepsilon(X_{\mathrm{int}})\cap X_{\mathrm{ext}})\\
&=\Phi^{-1}(N_\varepsilon(K))
\end{align*}
and hence $U_{\mathrm{int}} \cap U_{\mathrm{ext}}$ strong deformation retracts onto $K$.

Apply the Mayer--Vietoris exact sequence in singular homology to the open cover
$P=U_{\mathrm{int}}\cup U_{\mathrm{ext}}$ (see, e.g., \cite[Sec.~2.2]{Hatcher2002}).
Since $P\cong \mathbb{R}^{2}$ is contractible, $H_{1}(P)=0$ and $H_{0}(P)\cong \mathbb{Z}$, so exactness yields a
short exact sequence
\[
0\longrightarrow H_{0}(U_{\mathrm{int}}\cap U_{\mathrm{ext}})
\longrightarrow H_{0}(U_{\mathrm{int}})\oplus H_{0}(U_{\mathrm{ext}})
\longrightarrow H_{0}(P)\longrightarrow 0.
\]
Taking ranks and using $\mathrm{rank}\,H_{0}(Y)=C(Y)$ for any space $Y$ with finitely many connected
components (including $Y=\emptyset$), we obtain
\[
C(U_{\mathrm{int}}\cap U_{\mathrm{ext}})=C(U_{\mathrm{int}})+C(U_{\mathrm{ext}})-1.
\]
By the homotopy equivalences above, we have
\[
C(U_{\mathrm{int}}\cap U_{\mathrm{ext}})=C(K),\qquad
C(U_{\mathrm{int}})=C(X_{\mathrm{int}}),\qquad
C(U_{\mathrm{ext}})=C(X_{\mathrm{ext}}),
\]
and hence
\[
C(K)=C(X_{\mathrm{int}})+C(X_{\mathrm{ext}})-1,
\]
i.e.,
\begin{equation}\label{eq:component_identity}
C(A,w,s)\ =\ C(A^{\mathrm{int}},w,s)\ +\ C(A^{\mathrm{ext}},w,s)\ -\ 1
\end{equation}
for all $w\in S^{2}$ and all $s\in\mathbb{R}$.

\smallskip
\noindent\textbf{Step 2: Recovering $\chi(LS(A,w,t))$ from the pair of transforms.}
Now fix a direction $w$. 
                                                                        
                                                                                                
                                                                                                    
                                                                             
Then Proposition~\ref{prop:chilevel} gives
\[
\chi\!\bigl(LS(A,w,t)\bigr)\;=\;
2\,C(A,w,t)\;-\;C(A,w,t-\varepsilon)\;-\;C(A,w,t+\varepsilon),
\]
and likewise for $B$.

For each $s\in\{t,\,t-\varepsilon,\,t+\varepsilon\}$, the values
$C(A^{\mathrm{int}},w,s)$ and $C(A^{\mathrm{ext}},w,s)$ are determined by the Reeb graphs appearing
in $\mathcal{RT}(A^{\mathrm{int}})$ and $\mathcal{RT}(A^{\mathrm{ext}})$ (as the cardinalities of the
fibers at height $s$), and similarly for $B$. Hence~\eqref{eq:component_identity} determines
$C(A,w,s)$ and $C(B,w,s)$ for these heights. Since the assumed isomorphisms of Reeb transforms imply
\[
C(A^{\mathrm{int}},w,s)=C(B^{\mathrm{int}},w,s)
\quad\text{and}\quad
C(A^{\mathrm{ext}},w,s)=C(B^{\mathrm{ext}},w,s)
\qquad\text{for all such }s,
\]
we obtain $C(A,w,s)=C(B,w,s)$ for $s\in\{t,\,t-\varepsilon,\,t+\varepsilon\}$, and consequently
\begin{equation}\label{eq:slice-eq-on-dense}
\chi\!\bigl(LS(A,w,t)\bigr)=\chi\!\bigl(LS(B,w,t)\bigr)
\qquad\text{for all }t\in\mathbb{R}
\end{equation}
for every direction $w$.                                                                               


Finally, Theorem~\ref{thm:ect-inj} yields $A=B$.
