<!-- question-1: body_md -->
\label{question:yau}
If $\Lambda\subset\mathbb S^n$ is closed, contains at least two points, and
satisfies
\begin{equation}\label{eq:dimension}
 \dim_{\mathrm H}\Lambda<\frac{n-2}{2},
\end{equation}
does $\mathbb S^n\setminus\Lambda$ admit a complete metric conformal to
$g_{\mathrm{rd}}$ with scalar curvature $n(n-1)$?
<!-- end question-1 -->
<!-- theorem-1: body_md -->
\label{thm:main}
Let $n\geq3$, and let $\Lambda\subset\mathbb S^n$ be closed, contain at
least two points, and satisfy \eqref{eq:dimension}. Then
$\Omega=\mathbb S^n\setminus\Lambda$ admits a complete metric
$g=u^{2/a}g_{\mathrm{rd}}$ of scalar curvature $n(n-1)$, with
$u\in C^\infty(\Omega)$ positive. Moreover, $u\in L^p(\mathbb S^n)$ and
$\mathcal A_{\mathrm{rd}}u=\kappa u^p$ on the whole sphere in the sense of distributions.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:insertion}
There exists a universal $0<\eta_*\leq1$ with the following
property. For each fixed $0<\eta\leq\eta_*$ and each choice of data $(U_0,D,K)$ as above, every such family with sufficiently small
$\max_i r_i$ and $Q$ admits a marked nondegenerate finite-ended
solution $U_1$ with singular set $F\cup\{y_i\}$.
On $D\setminus\bigcup_i B(y_i,\rho_i)$,
\begin{equation}\label{eq:core-lower}
 U_1\geq c\left(
 U_0+\sum_i\frac{A_i}{(\rho_i^2+|x-y_i|^2)^a}
 \right),
\end{equation}

<!-- proposition-1: body_md -->
\label{pot:endpoint}
The potentials in \eqref{pot:tails} satisfy
\begin{equation}\label{pot:endpointineq}
 \mathcal I_2(h^p)\le C\delta h
 \qquad\text{on }\mathbb R^n,
\end{equation}
and
\begin{equation}\label{pot:mass}
 \int_{\mathbb R^n}h^p\,\mathrm d x\le C\delta Q.
\end{equation}
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Set $\gamma=(a-\beta)/a\in(0,1)$ and
$D_j(x)=(\rho_j^2+|x-y_j|^2)^{1/2}$.  We first prove
\begin{equation}\label{pot:pointwise}
 h(x)^p\le C\eta^{a+2}\nu^{\gamma-2}
       \sum_j r_j^{a+\gamma}D_j(x)^{-n-\gamma}.
\end{equation}
Suppose $x$ is outside every original ball, so that
$d_j=|x-y_j|\ge r_j$, and write
$$
 S=\sum_j r_j^ad_j^{-2a},\qquad
 T=\sum_j r_j^{a+\gamma}d_j^{-n-\gamma}.
$$
Since $n=2a+2$ and $a\gamma+\beta=a$,
$$
 r_j^ad_j^{-2a}
 =\bigl(r_j^{a+\gamma}d_j^{-n-\gamma}\bigr)^{a/(a+1)}
   \bigl(r_j^\beta d_j^{a-\beta}\bigr)^{1/(a+1)}.
$$
H\"older's inequality and packing give, for $R>0$,
$$
 \sum_{d_j\le R}r_j^ad_j^{-2a}
 \le T^{a/(a+1)}
      \left(\sum_{d_j\le R}r_j^\beta d_j^{a-\beta}\right)^{1/(a+1)}
 \le C(RT)^{a/(a+1)}.
$$
On a shell $2^kR<d_j\le2^{k+1}R$, all $r_j$ are at most
$2^{k+1}R$, and hence
$$
 \sum r_j^a\le(2^{k+1}R)^{a-\beta}\sum r_j^\beta
 \le C(2^kR)^a.
$$
The near--far decomposition, as in \cite{Hedberg1972}, therefore gives
$$
 S\le C(RT)^{a/(a+1)}+CR^{-a}.
$$
If the family is nonempty, take $R=T^{-1/(a+2)}$ to obtain
$S^p\le CT$.  Since $d_j\le D_j\le\sqrt2d_j$ and
$\nu^{\gamma-2}\ge1$, this proves \eqref{pot:pointwise} outside
the original balls.  The empty family is immediate.

If $x\in B(y_i,r_i)$, disjointness permits the same argument
for $H_i$.  For the remaining term, $D_i\ge\nu r_i$ and
$\gamma<2$ imply
$$
 h_i^p
 =2^{a+2}\eta^{a+2}r_i^{a+2}D_i^{-n-2}
 \le C\eta^{a+2}\nu^{\gamma-2}
          r_i^{a+\gamma}D_i^{-n-\gamma}.
$$
Together with $(h_i+H_i)^p\le2^{p-1}(h_i^p+H_i^p)$, this
proves \eqref{pot:pointwise} everywhere.

For $f_\rho(t)=(\rho^2+t^2)^{-(n+\gamma)/2}$, the radial
Newtonian formula bounds $\mathcal I_2f_\rho(D)$ by
$C\rho^{2-n-\gamma}$ for $D\le\rho$, and by
$C\rho^{-\gamma}D^{2-n}$ for $D\ge\rho$. Thus
\begin{equation}\label{pot:convolution}
 \mathcal I_2\bigl((\rho^2+|\cdot-y|^2)^{-(n+\gamma)/2}\bigr)(x)
 \le C\rho^{-\gamma}(\rho^2+|x-y|^2)^{-a}.
\end{equation}
Applying this to \eqref{pot:pointwise}, using $\rho_j=\nu r_j$,
yields
$$
 \mathcal I_2(h^p)
 \le C\eta^{a+2}\nu^{-2}
       \sum_jr_j^a(\rho_j^2+|x-y_j|^2)^{-a}
 =C\delta h.
$$
Multiplying \eqref{pot:endpointineq} by $|x|^{n-2}$,
letting $x$ tend to infinity along a ray, and applying Fatou's
lemma gives $\gamma_n\int h^p\le C\delta Q$, since
$|x|^{n-2}h(x)\to Q$. This proves \eqref{pot:mass}, including
integrability.

Let $w_\varepsilon$ be the periodic profile normalized by
$w_\varepsilon(0)=\varepsilon$ and $w_\varepsilon'(0)=0$; write
$T=T(\varepsilon)$ and $M=M(\varepsilon)$ for its period and maximum.
On the increasing half-period,
$$
 (w_\varepsilon')^2=D_\varepsilon(w_\varepsilon),\qquad
 D_\varepsilon(z)
  =a^2\bigl(z^2-\varepsilon^2-z^{q+2}+\varepsilon^{q+2}\bigr).
$$
Fix a small $c=c(n)>0$ and split the energy integral into
$[\varepsilon,2\varepsilon]$, $[2\varepsilon,c]$, and $[c,M]$.
In the first interval the substitution $z=\varepsilon Z$ gives
$\varepsilon^{-2}D_\varepsilon(\varepsilon Z)\asymp Z^2-1$.
In the middle interval $D_\varepsilon(z)\asymp z^2$. In the last,
$M\to1$ and $|D_\varepsilon'(M)|$ is bounded below; hence the
square-root singularity at $M$ is uniformly integrable. These estimates
give
\begin{equation}\label{in:radial-integrals}
 \int_0^t w_\varepsilon^{q+2}(\tau)\,\mathrm d\tau
       \leq Cw_\varepsilon^{q+2}(t),\qquad
 \int_0^{T/2}w_\varepsilon^q(\tau)\,\mathrm d\tau\leq C
       \quad(0\leq t\leq T/2).
\end{equation}

Set $j=\varepsilon\partial_\varepsilon w_\varepsilon$ and
$h=j/w_\varepsilon$. The variational equation gives
$$
 (w_\varepsilon^2h')'
   =-\kappa q w_\varepsilon^{q+2}h,\qquad h(0)=1,\quad h'(0)=0.
$$
By \eqref{in:radial-integrals},
$$
 |h'(t)|\leq Cw_\varepsilon^q(t)
               \sup_{0\leq\tau\leq t}|h(\tau)|.
$$
Gronwall's inequality therefore bounds $h$ and $h'$ uniformly on the
increasing half-period. Since $|w_\varepsilon'|\leq aw_\varepsilon$,
$$
 |\varepsilon\partial_\varepsilon w_\varepsilon|
  +|\varepsilon\partial_\varepsilon w_\varepsilon'|
       \leq Cw_\varepsilon.
$$
At the maximum, $|w_\varepsilon''(T/2)|$ is uniformly positive.
Differentiating $w_\varepsilon'(T/2)=0$ gives
$|\varepsilon T_\varepsilon|\leq C$. Differentiation of the reflection
identity $w_\varepsilon(t)=w_\varepsilon(T-t)$ extends the relative
bound to the first full period.

The cutoff data satisfy
$$
\begin{aligned}
 v_0:=v_B(0)&\asymp s^a,& |E|/v_0^2&\leq C\vartheta_B,\\
 |\partial_bv_0|/v_0&\leq C\vartheta_B,&
 v_B'(0)/v_0&=a(1+O(s^2+\vartheta_B)).
\end{aligned}
$$
Thus the cutoff lies in the increasing middle interval. If
$w_\varepsilon(\tau_B)=v_0$ with $0<\tau_B<T/2$, then
$w_\varepsilon'(\tau_B)\asymp av_0$ and
$$
 (\tau_B)_b
  =\frac{\partial_bv_0-\varepsilon_b
                   \partial_\varepsilon w_\varepsilon(\tau_B)}
          {w_\varepsilon'(\tau_B)},\qquad |(\tau_B)_b|\leq C.
$$
Also $|T_b|\leq C$, since $|\varepsilon_b|\leq C\varepsilon$.
Now $v_B(t)=w_\varepsilon(t+\tau_B)$. In the $k$th period,
differentiating periodicity introduces the term $-kT_bw_\varepsilon'$;
this is the familiar linear growth of the neck-variation Jacobi field
\cite[Section~5, Lemma~3]{MP1999}; \cite[Section~II]{MPU1996}.
The energy integral on $[c/2,c]$ gives
$T\geq2a^{-1}\log2$ for small $\varepsilon$, so $k\leq C(1+t)$, proving
\eqref{in:radial-loss} for $m=0,1$. The equation and its variational
equation give the higher derivatives; throughout,
$|v_B^{(m)}|\leq C_mv_B$ for each fixed $m$.

Finally, negative energy gives $|v_B'|\leq av_B$ and $v_B\leq1$.
Hence $v_B(t)\leq Cs^ae^{at}$ and
$r^2Q_B=\kappa p v_B^q\leq C\min\{1,s^2e^{2t}\}$.
The parameter estimate follows from \eqref{in:radial-loss}.
One spatial derivative introduces $r^{-1}$, so these pointwise
estimates also give the scaled H\"older bounds.

interior. Specify $K=E_j$ before selecting the radius and coefficient
bounds in the finite gluing statement.

Since $\mathcal H^\beta(E_j)=0$, Lemma~\ref{pot:cover} gives finitely many
disjoint balls $B(y_i,r_i)$, with centers $y_i\in E_j$, whose
open fivefold enlargements cover $E_j$. Their tenfold enlargements
can be required to lie in the interior of the old core and in
$N_{\sigma_j}(\Lambda)$. Their maximum radius and total $\beta$-cost
can be made arbitrarily small. Equation~\eqref{pot:charge}
therefore makes $Q$ satisfy all prescribed coefficient bounds.

Apply the finite gluing statement, requiring in addition that
\begin{align}
 \|u_j-u_{j-1}\|_{C^j(L_j)}&\leq2^{-j},
       \label{eq:absolute-budget}\\
 |u_j-u_{j-1}|&\leq\epsilon_j u_{j-1}
       &&\text{on }L_j\cup\bigcup_{\ell<j}W_\ell.
       \label{eq:relative-budget}
\end{align}
These are finitely many compact approximation requirements; the
relative bounds are possible because $u_{j-1}$ has a positive minimum
on each protected compact set. The output has singular set
$F_j=F_{j-1}\cup\{y_i\}$ and remains marked nondegenerate with
noncylindrical ends. If $E_j=\varnothing$, take $u_j=u_{j-1}$.
By Lemma~\ref{pot:oldrepresentation}, every finite-step factor $u_j$
belongs to $L^p(\mathbb S^n)$ and satisfies the global distributional
equation.

For every new ball choose a smooth cutoff $\chi_i$, equal to one on
$B(y_i,5r_i)$ and zero outside $B(y_i,10r_i)$, with $|D\chi_i|\leq C/r_i$.
Extend all the cutoffs smoothly over the sphere, and set
\begin{equation}\label{eq:barrier-definition}
 f_j=\max\bigl(\{f_z:z\in F_{j-1}\}\cup\{\chi_i\}_i\bigr),
 \qquad W_j=\{1/4\leq f_j\leq3/4\}.
\end{equation}
The old inner balls and new open fivefold balls cover $\Lambda$.
Consequently, $f_j=1$ on an open neighborhood $V_j$ of $\Lambda$,
and both $\mathbb S^n\setminus V_j$ and $W_j$ are compact subsets of
$\Omega$. Since $\operatorname{supp} f_j\subset N_{\sigma_j}(\Lambda)$, the later
supports miss $W_j$; hence the sets $W_j$ are pairwise disjoint.

At a point of $W_j$, every new fivefold ball is avoided. If a new
cutoff $\chi_i$ is active in the maximum, its point lies in $B(y_i,10r_i)$,
and \eqref{eq:shell-lower} gives
$$
 U_j^{1/a}\geq c_1^{1/a}\frac{\eta}{r_i},
 \qquad |\mathrm d\chi_i|_{g_j}\leq\frac{C}{\eta}.
$$
An active old cutoff lies in its logarithmic annulus, in the old core.
Every point of $W_j$ lies outside all the balls $B(y_i,5r_i)$,
and hence outside all the matching balls $B(y_i,\rho_i)$. Thus
\eqref{eq:core-lower} and \eqref{eq:old-cutoff-gradient} give
$|\mathrm d f_z|_{g_j}\leq1/\eta$. The same estimate holds when no new
balls are inserted. Every active cutoff on $W_j$ therefore has
metric gradient bounded by $C/\eta$.

The bounds \eqref{eq:absolute-budget} imply convergence to a smooth
function $u\geq0$ locally on $\Omega$: for any fixed $m,k$, the
$C^k(L_m)$ tail is summable. Passing to the local equations yields
\begin{equation}\label{eq:local-limit-equation}
 \mathcal A_{\mathrm{rd}}u=\kappa u^p\qquad\text{on }\Omega.
\end{equation}
For $x\in L_m$, \eqref{eq:relative-budget} gives
$$
 u(x)\geq u_m(x)\prod_{j>m}(1-\epsilon_j)>0.
$$
Hence $g=u^{2/a}g_{\mathrm{rd}}$ is a smooth metric with scalar
curvature $n(n-1)$. The same product estimate on the protected sets
gives
\begin{equation}\label{eq:protected-limit}
 u\geq\frac34u_j\qquad\text{on }W_j.
\end{equation}

Each $f_j$ is the finite maximum of smooth functions and is therefore
locally Lipschitz. Along a locally absolutely continuous curve,
its derivative agrees almost everywhere with that of an active
cutoff. Together with \eqref{eq:protected-limit}, the gradient
estimates give
\begin{equation}\label{eq:curve-cutoff}
 |(f_j\circ\gamma)'(t)|
 \leq\frac{C'}{\eta}|\dot\gamma(t)|_g
 \quad\text{for almost every }t\text{ with }\gamma(t)\in W_j,
\end{equation}
where $C'$ is independent of $j$.

Let $\gamma$ be a locally absolutely continuous curve leaving
every compact subset of $\Omega$. For every sufficiently large
$j$, its starting point is outside $\operatorname{supp} f_j$, whereas it is
eventually in $V_j$, because $\mathbb S^n\setminus V_j\Subset\Omega$.
Thus it crosses $W_j$ from level $1/4$ to level $3/4$.
Integrating \eqref{eq:curve-cutoff} gives
$$
 \int_{\gamma^{-1}(W_j)}|\dot\gamma(t)|_g\,\mathrm d t
       \ge\frac{\eta}{2C'}.
$$
The $W_j$ are pairwise disjoint, so every such curve has infinite
length. The divergent-curve criterion for Riemannian completeness
therefore proves that $g$ is complete.
Finally, $\mathcal H^a(\Lambda)=0$, so Proposition~\ref{prop:no-defect}
below gives the global equation and the asserted $L^p$ integrability.

\label{prop:no-defect}
Let $\Lambda\subset\mathbb S^n$ be closed, with $\mathcal H^a(\Lambda)=0$.
Suppose that $u_j>0$ belong to $L^p(\mathbb S^n)$, satisfy
$\mathcal A_{\mathrm{rd}}u_j=\kappa u_j^p$ in $\mathcal D'(\mathbb S^n)$, and converge smoothly
locally on $\mathbb S^n\setminus\Lambda$ to $u$. Then
$$
 u_j^p\longrightarrow u^p\quad\text{in }L^1(\mathbb S^n),
 \qquad
 u_j\longrightarrow u\quad\text{in }L^p(\mathbb S^n),
$$
and $\mathcal A_{\mathrm{rd}}u=\kappa u^p$ globally in distributions. Moreover, the source
measures $\mu_j=\kappa u_j^p\,\mathrm d V_{g_{\mathrm{rd}}}$ satisfy
\begin{equation}\label{eq:source-morrey}
 \mu_j(B_{g_{\mathrm{rd}}}(x,r))\leq C_n r^a
 \qquad(0<r\leq r_0),
\end{equation}
where $C_n,r_0>0$ depend only on $n$. In particular, no additional
limiting source measure is supported on $\Lambda$.

Define $u$ arbitrarily on the measure-zero set $\Lambda$.

For the local estimate, we use the test-function method of
Mitidieri and Pohozaev; see \cite[Section~1, equations~(1.2)--(1.5)]{MitidieriPohozaev2004},
with numbering from their March 2004 manuscript. Put
$p'=p/(p-1)=(n+2)/4$ and choose an integer $m>2p'$.
For $B=B_{g_{\mathrm{rd}}}(x,r)$, $0<r\leq r_0\leq1$, choose
$0\leq\psi\leq1$, equal to one on $B$, supported in $2B$, with
$|\nabla\psi|\leq C_n/r$ and $|\Delta\psi|\leq C_n/r^2$.
Then $\varphi=\psi^m$ satisfies
$$
 |\mathcal A_{\mathrm{rd}}\varphi|\leq C_n r^{-2}\psi^{m-2}
             \leq C_n r^{-2}\varphi^{1/p}.
$$
Writing $X_j=\int u_j^p\varphi\,\mathrm d V_{g_{\mathrm{rd}}}$, the equation and H\"older's
inequality give
$$
 \kappa X_j=\int u_j\mathcal A_{\mathrm{rd}}\varphi\,\mathrm d V_{g_{\mathrm{rd}}}
 \leq C_n r^{-2}X_j^{1/p}
           \operatorname{Vol}(2B)^{1/p'}.
$$
If $X_j>0$, divide by $X_j^{1/p}$ and raise to the power $p'$;
the zero case is immediate. Since $n-2p'=a$, this yields
$$
 \int_Bu_j^p\,\mathrm d V_{g_{\mathrm{rd}}}\leq X_j\leq C_n r^a,
$$
which proves \eqref{eq:source-morrey} in every dimension $n\geq3$.
Testing with one gives $\int u_j^p=\int u_j$; H\"older's
inequality implies $\int u_j^p\le\operatorname{Vol}(\mathbb S^n)$.
Fatou's lemma therefore yields $u\in L^p(\mathbb S^n)$.

Given $\varepsilon>0$, cover $\Lambda$ by finitely many open round
balls $B(x_i,r_i)$, with $r_i\leq r_0$ and
$\sum_i r_i^a<\varepsilon$, and let $O$ be their union.
The local estimate gives
$$
 \kappa\int_Ou_j^p\,\mathrm d V_{g_{\mathrm{rd}}}\leq C_n\varepsilon
 \qquad\text{for every }j,
$$
and Fatou's lemma gives the same bound for $u^p$.
On $\mathbb S^n\setminus O\Subset\mathbb S^n\setminus\Lambda$, the sources
converge uniformly. Therefore
$$
 \limsup_{j\to\infty}\kappa
 \int_{\mathbb S^n}|u_j^p-u^p|\,\mathrm d V_{g_{\mathrm{rd}}}\leq2C_n\varepsilon.
$$
Letting $\varepsilon\downarrow0$ proves strong $L^1$ convergence.

For $s,t\geq0$, $|s-t|^p\leq|s^p-t^p|$, since
$(t+h)^p\geq t^p+h^p$ for $h\geq0$. Thus
$$
 \|u_j-u\|_{L^p}^p\leq\|u_j^p-u^p\|_{L^1}\longrightarrow0.
$$
The factors also converge in $L^1$, so for every
$\varphi\in C^\infty(\mathbb S^n)$ we can pass to the limit in
$$
 \int_{\mathbb S^n}u_j\mathcal A_{\mathrm{rd}}\varphi\,\mathrm d V_{g_{\mathrm{rd}}}
 =\kappa\int_{\mathbb S^n}u_j^p\varphi\,\mathrm d V_{g_{\mathrm{rd}}}.
$$
This proves the global distributional equation. Strong $L^1$
convergence of the sources excludes an additional measure on
$\Lambda$.
