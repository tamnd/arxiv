Define
\begin{equation}
V_\rho=\frac12\left[(x-2\rho)^2+y^2+\bigl(z-(1-\rho)\bigr)^2\right].
\label{eq:V}
\end{equation}
Direct differentiation gives
\begin{align}
\dot V_\rho={}&-2(x-\rho)^2
-\left(y-\frac{\rho(3\rho-1)}{2}\right)^2 \nonumber\\
&-5\left(z-\frac{1-\rho}{2}\right)^2+C(\rho),
\label{eq:Vdot}
\end{align}
where
\begin{equation}
C(\rho)=2\rho^2+\frac{\rho^2(3\rho-1)^2}{4}+\frac{5(1-\rho)^2}{4}.
\label{eq:C}
\end{equation}
Let
\begin{align}
a_\rho&=\left(\rho,\frac{\rho(3\rho-1)}{2},\frac{1-\rho}{2}\right)^T,\\
b_\rho&=(2\rho,0,1-\rho)^T,
\end{align}
and define $d(\rho)=\|a_\rho-b_\rho\|$. Since the quadratic form in \eqref{eq:Vdot} is bounded below by $\|\mathbf{x}-a_\rho\|^2$,
\begin{equation}
\dot V_\rho\leq-\|\mathbf{x}-a_\rho\|^2+C(\rho).
\end{equation}
The reverse triangle inequality and $\|\mathbf{x}-b_\rho\|=\sqrt{2V_\rho}$ imply
\begin{equation}
\dot V_\rho\leq-\left(\sqrt{2V_\rho}-d(\rho)\right)^2+C(\rho).
\label{eq:comparison}
\end{equation}
Therefore, with
\begin{equation}
R(\rho)=\frac12\left[d(\rho)+\sqrt{C(\rho)}\right]^2,
\label{eq:R}
\end{equation}
the compact sublevel set
\begin{equation}
\Omega_\rho=\{\mathbf{x}:V_\rho(\mathbf{x})\leq R(\rho)\}
\label{eq:omega}
\end{equation}
is positively invariant because $\dot V_\rho\leq0$ on its boundary. Moreover, $\dot V_\rho<0$ whenever $V_\rho>R(\rho)$, so every trajectory eventually enters $\Omega_\rho$. \hfill$\square$
