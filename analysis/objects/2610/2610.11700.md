\label{lem:jacobian_change_of_coordinates}
		Let $K \subset \mathbb R^n$ be a convex body containing the origin in its interior, and let $f: S^{n-1} \to \partial K$ be the radial map defined by $f(v) = \rho_K(v)v$. Then $f$ is differentiable for almost all $v \in S^{n-1}$ with respect to the $(n-1)$-dimensional Hausdorff measure, and its Jacobian determinant is given by:
		\begin{equation}
			Jf(v) = \frac{\rho_K(v)^{n-1}}{v\cdot n_K\left(\rho_K(v)v\right)}
		\end{equation}

		\begin{equation}

\label{lem:smooth_boundary_implies_smooth_rho}
		Let $K$ be a $C^k$ convex body containing the origin in its interior. Then the function $(x, v) \mapsto \rho_{K_{x, p}}(v)$ is of class $C^k$ in $\operatorname{int} (-K) \times S^{n-1}$. Moreover those derivatives are uniformly bounded on any compact set $L\subset \operatorname{int} (-K) \times S^{n-1}$.
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
It suffices to show that the function $(x, v) \mapsto \rho_{K+x}(v)$ is of class $C^k$, since the function
		\begin{equation}
			y \mapsto \left(\frac{y^{-p} + a^{-p}}{2}\right)^{-1/p}
		\end{equation}
		is in $C^\infty((0,\infty))$, for $a > 0$.
		
		Let $p(v_0) = \rho_K(v_0)v_0$. Since $K$ is of class $C^k$, there exists a $C^k$ defining function $\Phi$ in a neighborhood $U$ of $p(v_0)$ such that
		\begin{equation}
			\partial K \cap U = \{y\in U:\Phi(y) = 0\},\qquad \nabla\Phi(p(v_0))\neq0.
		\end{equation}
		Consider $F(x,v,t) = \Phi(tv-x)$. Note that $F(0, v_0, \rho_K(v_0)) = \Phi(\rho_K(v_0)v_0) = 0$. Moreover, since $\nabla\Phi(p(v_0))$ is normal to $\partial K$ at $p(v_0)$ and the origin is in the interior, the following transversality condition holds:
		\begin{equation}
			\partial_t F(0,v_0,\rho_K(v_0)) = \nabla\Phi(p(v_0))\cdot v_0\neq 0.
		\end{equation}
		By the implicit function theorem, there exists a neighborhood $V\subset \mathbb R^n \times S^{n-1}$ of $(0, v_0)$ and a unique $C^k$ function $g:V\to\mathbb R$ such that $g(0, v_0) = \rho_K(v_0)$ and $F(x, v, g(x, v)) = 0$ for all $(x, v)\in V$. By uniqueness, $\rho_{K+x}(v) = g(x, v)$ near $(0, v_0)$, thus $(x,v)\mapsto\rho_{K+x}(v)$ is $C^k$ near $(0, v_0)$.
		
		By translating $K$, it is enough to show the regularity for $x$ near $0$. Let $x_0$ be such that $0 \in \operatorname{int}(K+x_0)$. Since $K+x_0$ is a convex body containing the origin in its interior, by the previous argument the function
		\begin{equation}
			f(x, v) = \rho_{(K+x_0)+x}(v)
		\end{equation}
		is $C^k$ in a neighborhood of $(0, v_0)$. Now, consider the translation map $\psi(x, v) = (x-x_0, v)$. The composition
		\begin{equation}
			(f \circ \psi)(x, v) = f(x-x_0, v) = \rho_{K+x_0+(x-x_0)}(v) = \rho_{K+x}(v)
		\end{equation}
		is a composition of $C^k$ functions, and therefore is $C^k$ in a neighborhood of $x_0$, as required.

		The uniform boundedness of the derivatives follows directly from the compactness of $L$ and the continuity of the derivatives.

\label{lem:second_derivative_translated}

		Then:
		\begin{equation}
			\left.\frac{\partial^2}{\partial t^2}\rho_{K+tu}(v)\right|_{t=0}
			= -\frac{u^t Q_{x_0(v)} u}{v\cdot n_K(x_0(v))},
		\end{equation}
		where $S_{x_0(v)}$ is the shape operator at $x_0(v)$ and $Q_{x_0(v)} = A_{x_0(v)}^t S_{x_0(v)} A_{x_0(v)}$ is a positive semidefinite symmetric matrix.

Define \(x(t)=\varphi(t)v-tu\). We have
\[
x'(0)=\varphi'(0)v-u=-A_{x_0(v)}u.
\]
Since \(x(t)\in \partial K\), we have
\[
x'(t)\cdot n_K(x(t))=0.
\]
Differentiating this identity with respect to \(t\), and evaluating at \(t=0\),
we obtain
\[
0
=
\left.\frac{d}{dt}
\big(x'(t)\cdot n_K(x(t))\big)\right|_{t=0}.
\]
By the product rule and the chain rule,
\[
0
=
x''(0)\cdot n_K(x_0(v))
+
x'(0)\cdot d n_K{}_{x_0(v)}(x'(0)).
\]
Using the convention \(S_x=d n_K{}_x\), this becomes
\[
0
=
x''(0)\cdot n_K(x_0(v))
+
x'(0)\cdot S_{x_0(v)}x'(0).
\]
Therefore
\[
x''(0)\cdot n_K(x_0(v))
=
-\mathrm{II}_{x_0(v)}(x'(0),x'(0)).
\]
Hence
\[
\begin{aligned}
(v\cdot n_K(x_0(v)))\varphi''(0)
&=
\varphi''(0)v\cdot n_K(x_0(v))  \\
&=
x''(0)\cdot n_K(x_0(v)) \\
&=
-\mathrm{II}_{x_0(v)}(x'(0),x'(0)) \\
&=
-\mathrm{II}_{x_0(v)}(-A_{x_0(v)}u,-A_{x_0(v)}u) \\
&=
-\mathrm{II}_{x_0(v)}(A_{x_0(v)}u,A_{x_0(v)}u) \\
&=
-(A_{x_0(v)}u)\cdot S_{x_0(v)}(A_{x_0(v)}u) \\
&=
-u^tA_{x_0(v)}^tS_{x_0(v)}A_{x_0(v)}u \\
&=
-u^tQ_{x_0(v)}u .
\end{aligned}
\]
Thus

-\frac{u^tQ_{x_0(v)}u}{v\cdot n_K(x_0(v))}.

			\medskip\noindent\textit{Step 1: Computation of $\operatorname{tr}(A_x^t S_x A_x)$.}
			At each $x \in \partial K$, choose an orthonormal basis $\{e_1, \dots, e_{n-1}\}$ for $T_x\partial K$. Since $A_x$ acts as the identity on $T_x\partial K$ and maps $n_K(x)$ to $-x_T/(x \cdot n_K(x))$:

				&= \sum_{i=1}^{n-1} (S_x A_x e_i) \cdot (A_x e_i) + (S_x A_x n_K(x)) \cdot (A_x n_K(x)) \\
				&= \sum_{i=1}^{n-1} (S_x e_i) \cdot e_i + \frac{(S_x x_T) \cdot x_T}{(x \cdot n_K(x))^2} \\
				&= \operatorname{tr}(S_x) + \frac{\mathrm{I\!I}(x_T, x_T)}{(x \cdot n_K(x))^2}.

			\Delta_p(p^\alpha y)\leq p^\alpha.
		\]
By \eqref{eq:comparison-gkp-gk},
		\[
			\Delta_\infty(p^\alpha y)
			\le
			\Delta_p(p^\alpha y)+O(p^{-1})
			\le
			p^\alpha+O(p^{-1}).
		\]
Since \(\alpha>-1\), we have \(p^{-1}=o(p^\alpha)\). Hence
		\[
			\Delta_\infty(p^\alpha y)\le C p^\alpha.
		\]
Using the fact that \(h_{\Pi K}\) is a norm and the expansion \eqref{eq:classical-covariogram-expansion}, it follows
that the sets \(E_p\) are eventually contained in a fixed ball.

For \(y\) in a fixed bounded set, \eqref{eq:comparison-gkp-gk}
 and \eqref{eq:classical-covariogram-expansion} give
		\[
			\begin{aligned}
				\frac{\Delta_p(p^\alpha y)}{p^\alpha}
				&=
				\frac{\Delta_\infty(p^\alpha y)}{p^\alpha}
				+
				O(p^{-(\alpha+1)})  \\
				&=
				h_{\Pi K}(y)+O(p^\alpha)+O(p^{-(\alpha+1)}).
			\end{aligned}
		\]
Since \(\alpha<0\) and \(\alpha>-1\), both error terms tend to zero as \(p\to\infty\). Hence
		\[
			\frac{\Delta_p(p^\alpha y)}{p^\alpha}
			\xrightarrow[p\to\infty]{}
			h_{\Pi K}(y)
		\]
locally uniformly in \(y\). Therefore
		\[
			p^{-\alpha}L(p,p^\alpha)
			\xrightarrow[p\to\infty]{d_H}
			\{y:h_{\Pi K}(y)\le1\}
			=
			\Pi^\circ K.
		\]
This proves part  \eqref{thm:level_sets_coupled_convergence_case1}.

We now prove the critical case \(\alpha=-1\). Define
		\[
			F_p(y)
			=
			p\left(
			\operatorname{vol}(K)-g_{K,p}\left(\frac yp\right)
			\right)
			=
			p\Delta_p\left(\frac yp\right).
		\]
Then
		\[
			pL(p,p^{-1})
			=
			\{y\in\mathbb R^n:F_p(y)\leq 1\}.
		\]

Fix \(y\in\mathbb R^n\). For \(v\in S^{n-1}\), set
	\[
		a(v)=\rho_K(v),
		\qquad
		b_p(v)=\rho_{K+y/p}(v),
		\qquad
		x_0(v)=\rho_K(v)v.
	\]
By Lemma~\ref{lem:first_derivative_translated}, applied with \(u=y\), we have
		\[
			\rho_{K+ty}(v)=\rho_K(v)+t\frac{y\cdot n_K(x_0(v))}{v\cdot n_K(x_0(v))}+O(t^2),
		\]
where the error term is uniform in \(v\in S^{n-1}\), and uniform for \(y\) in
compact subsets of \(\mathbb R^n\). Taking \(t=1/p\), we obtain
		\[
			b_p(v)=a(v)+\frac1p\frac{y\cdot n_K(x_0(v))}{v\cdot n_K(x_0(v))}+O(p^{-2}).
		\]
Since \(x_0(v)=a(v)v\), it follows that
		\[
			\frac{b_p(v)}{a(v)}=1+\frac{q_y(v)}p+O(p^{-2}),
		\]
where
		\[
			q_y(v)=\frac{y\cdot n_K(x_0(v))}{x_0(v)\cdot n_K(x_0(v))}.
		\]
Note that the denominator of $q_y(v)$ is bounded away from zero, since it is exactly the support function of $K$ at $n_K(x_0(v))$, which by our assumption that $K$ contains the origin is non-zero.
Therefore
		\[
			\left(\frac{b_p(v)}{a(v)}\right)^{-p}\xrightarrow[p\to\infty]{}e^{-q_y(v)}
		\]
uniformly in \(v\in S^{n-1}\), and uniformly for \(y\) in compact subsets of \(\mathbb R^n\).
To see this, let $R_p(v, y)$ be the remainder term in the expansion of \(\left(\frac{b_p(v)}{a(v)}\right)^{-p}\), which is the $\mathcal{O}(p^{-2})$ term. Taking the logarithm, we have:
\[
	\log\left(\left(\frac{b_p(v)}{a(v)}\right)^{-p}\right) = -p \log\left(1 + \frac{q_y(v)}{p} + R_p(v, y)\right). 
\]
Now, for $p$ large enough, we have that $\left|\frac{q_y(v)}{p} + R_p(v, y)\right| < 1 $, so we can use the Taylor expansion of the logarithm:
\[
	-p\log\left(1 + \frac{q_y(v)}{p} + R_p(v, y)\right) = q_y(v) + \frac{R_p(v, y)}{p} + \mathcal{O}\left(\frac{1}{p}\left(\frac{q_y(v)}{p} + R_p(v, y)\right)^2\right).
\]
Using the fact that \(R_p(v, y) = \mathcal{O}(p^{-2})\) and that \(q_y(v)\) is uniformly bounded for $v$ in \(S^{n-1}\) and \(y\) in compact sets, we conclude that:

\[
	\log\left(\left(\frac{b_p(v)}{a(v)}\right)^{-p}\right) = -q_y(v) + O(p^{-1}), 
\]
and therefore:

\[
	\left(\frac{b_p(v)}{a(v)}\right)^{-p} = e^{-q_y(v)}e^{O(p^{-1})} = e^{-q_y(v)}(1 + O(p^{-1})) = e^{-q_y(v)} + O(p^{-1}), 
\]
with an error term that is independent of $v$ and $y$.


By the definition of the radial \((-p)\)-combination,
		\[
			\rho_{K_{y/p,p}}(v)^{-p}=\frac{\rho_K(v)^{-p}+\rho_{K+y/p}(v)^{-p}}2.
		\]
Hence
		\[
			\rho_{K_{y/p,p}}(v)^n=a(v)^n\left(\frac{1+\left(b_p(v)/a(v)\right)^{-p}}2\right)^{-n/p}.
		\]
Set
		\[
			A_p(v,y)=\frac{1+\left(b_p(v)/a(v)\right)^{-p}}2.
		\]
Then
		\[
			A_p(v,y)\xrightarrow[p\to\infty]{}A(v,y):=\frac{1+e^{-q_y(v)}}2
		\]
uniformly in \(v\in S^{n-1}\), and uniformly for \(y\) in compact subsets of \(\mathbb R^n\).

Using the polar formula for the volume of a star body,
		\[
			g_{K,p}\left(\frac yp\right)=\frac1n\int_{S^{n-1}}a(v)^nA_p(v,y)^{-n/p}\,dv.
		\]
Therefore
		\[
			F_p(y)=\frac pn\int_{S^{n-1}}a(v)^n\left(1-A_p(v,y)^{-n/p}\right)\,dv.
		\]
Since \(A_p(v,y)\) is bounded away from \(0\) and infinity for \(y\) in compact
sets, we have
		\[
			\frac pn\left(1-A_p(v,y)^{-n/p}\right)\xrightarrow[p\to\infty]{}\log A(v,y)
		\]
uniformly in \(v\in S^{n-1}\), and uniformly for \(y\) in compact subsets of \(\mathbb R^n\). Consequently,
		\[
			F_p(y)\xrightarrow[p\to\infty]{}\int_{S^{n-1}}\rho_K(v)^n\log\left(\frac{1+e^{-q_y(v)}}2\right)\,dv
		\]
locally uniformly in \(y\).

We now rewrite the limit. Using
		\[
			\log\left(\frac{1+e^{-q}}2\right)=-\frac q2+\log\cosh\left(\frac q2\right),
		\]
we get
		\[
			\begin{aligned}\int_{S^{n-1}}\rho_K(v)^n\log\left(\frac{1+e^{-q_y(v)}}2\right)\,dv
				&=-\frac12\int_{S^{n-1}}\rho_K(v)^nq_y(v)\,dv \\
				&\quad+\int_{S^{n-1}}\rho_K(v)^n\log\cosh\left(\frac{q_y(v)}2\right)\,dv.
			\end{aligned}
		\]
The first integral vanishes. Indeed, by Lemma~\ref{lem:jacobian_change_of_coordinates}, with \(x=\rho_K(v)v\), we have
		\[
			\rho_K(v)^n\,dv=(x\cdot n_K(x))\,dS(x).
		\]
Moreover,
		\[
			q_y(v)=\frac{y\cdot n_K(x)}{x\cdot n_K(x)}.
		\]
Therefore
		\[
			\int_{S^{n-1}}\rho_K(v)^nq_y(v)\,dv=\int_{\partial K}y\cdot n_K(x)\,dS(x)=0
		\]
by the divergence theorem. Hence
		\[
			F_p(y)\xrightarrow[p\to\infty]{}\Phi_K(y)
		\]
locally uniformly in \(y\), where
		\[
			\Phi_K(y)=\int_{\partial K}(x\cdot n_K(x))\log\cosh\left(\frac{y\cdot n_K(x)}{2x\cdot n_K(x)}\right)\,dS(x).
		\]

It remains to show that the sets \(pL(p,p^{-1})=\{y:F_p(y)<1\}\) are eventually
contained in a fixed ball. From equation \eqref{eq:taylor_delta_p}, after decreasing \(\rho\) if necessary,
there are constants \(c>0\), \(\eta>0\), and \(p_0\ge1\) such that, whenever
\(p\ge p_0\), \(x\in B_\rho\), and \(p|x|\le\eta\), one has
		\begin{equation}\label{eq:quadratic-lower-bound}
			\Delta_p(x)\ge c p|x|^2.
		\end{equation}
		
		Indeed, the error term \(O(p^2|x|^3)\) is then bounded by a sufficiently small
multiple of \(p|x|^2\).

Using \eqref{eq:monotonicity-delta-p} and \eqref{eq:quadratic-lower-bound}, we obtain the following localization estimate: there are
constants \(c_1>0\) and \(\rho_1>0\) such that, for all \(p\) large enough and
all \(x\in B_{\rho_1}\),
		\begin{equation}\label{eq:localization-lower-bound}
			\Delta_p(x)\ge c_1\min\{p|x|^2,|x|\}.
		\end{equation}To see this, if \(p|x|\le\eta\), then \eqref{eq:quadratic-lower-bound} gives the first bound. If
\(p|x|>\eta\), choose \(q=\eta/|x|\), whenever this is at least \(p_0\), and use
\(\Delta_p(x)\ge\Delta_q(x)\) together with \eqref{eq:quadratic-lower-bound} applied to \(q\). The remaining
range is covered by fixing \(q=p_0\) and using the fact that \(\Delta_{p_0}\) is
positive away from the origin.

Now put \(x=y/p\). From \eqref{eq:localization-lower-bound},
		\[
			F_p(y)=p\Delta_p\left(\frac yp\right)\ge c_1\min\{|y|^2,|y|\}
		\]
whenever \(y/p\in B_{\rho_1}\). If \(y/p\notin B_{\rho_1}\), then by \eqref{eq:monotonicity-delta-p}, with
\(q=p_0\), and by compactness away from the origin, we have
		\[
			F_p(y)\to\infty
		\]
uniformly in that region. Therefore there exists \(R>0\) such that
		\[
			pL(p,p^{-1})\subset B_R
		\]
for all sufficiently large \(p\).

Since \(F_p\to\Phi_K\) locally uniformly and the relevant level sets are
uniformly bounded, the sublevel sets converge in the Hausdorff metric. Thus
		\[
			pL(p,p^{-1})\xrightarrow[p\to\infty]{d_H}\{y\in\mathbb R^n:\Phi_K(y)\le1\}=\mathcal T_K.
		\]
This proves case (\ref{thm:level_sets_coupled_convergence_case3}).

Finally, we prove the case \(\alpha<-1\). Let
		\[
			E_p=p^{\frac{1-\alpha}{2}}L(p,p^\alpha).
		\]
Equivalently,
		\[
			E_p=\left\{y:\Delta_p\left(p^{\frac{\alpha-1}{2}}y\right)<p^\alpha\right\}.
		\]

We first show that the sets \(E_p\) are eventually bounded. Put
		\[
			x=p^{\frac{\alpha-1}{2}}y.
		\]
Using \eqref{eq:localization-lower-bound}, for \(x\) sufficiently small we have
		\[
			\Delta_p(x)\ge c_1\min\{p|x|^2,|x|\}.
		\]
In terms of \(y\),
		\[
			p|x|^2=p^\alpha |y|^2,\qquad |x|=p^{\frac{\alpha-1}{2}}|y|.
		\]
If \(y\in E_p\), then \(\Delta_p(x)<p^\alpha\). The previous lower bound implies
that \(|y|\) must remain bounded. The region where \(x\) is not sufficiently small
is excluded as before by monotonicity and compactness away from the origin. Hence
the sets \(E_p\) are eventually contained in a fixed ball.

For \(y\) in a fixed bounded set, we use the Taylor estimate \eqref{eq:taylor_delta_p}:
		\[
			\Delta_p\left(p^{\frac{\alpha-1}{2}}y\right)=\frac{n+p}{8}p^{\alpha-1}y^t\mathcal{M}_Ky+O\left(p^2p^{\frac{3(\alpha-1)}2}|y|^3\right).
		\]
Dividing by \(p^\alpha\), we obtain
		\[
			\frac{\Delta_p\left(p^{\frac{\alpha-1}{2}}y\right)}{p^\alpha}=\frac18y^t\mathcal{M}_Ky+\frac{n}{8p}y^t\mathcal{M}_Ky+O\left(p^{\frac{\alpha+1}{2}}|y|^3\right).
		\]
Since \(\alpha<-1\), we have
		\[
			p^{\frac{\alpha+1}{2}}\xrightarrow[p\to\infty]{}0.
		\]
Therefore
		\[
			\frac{\Delta_p\left(p^{\frac{\alpha-1}{2}}y\right)}{p^\alpha}\xrightarrow[p\to\infty]{}\frac18y^t\mathcal{M}_Ky
		\]
locally uniformly in \(y\). Consequently,
		\[
			p^{\frac{1-\alpha}{2}}L(p,p^\alpha)\xrightarrow[p\to\infty]{d_H}\left\{y\in\mathbb R^n:\frac18y^t\mathcal{M}_Ky\le1\right\}.
		\]
Since
		\[
			\Pi_2^\circ K=\{y\in\mathbb R^n:y^t\mathcal{M}_Ky\le1\},
		\]
the last set is
		\[
			\sqrt8\,\Pi_2^\circ K.
		\]
This finally proves part  \eqref{thm:level_sets_coupled_convergence_case2}.

<!-- remark-2: body_md -->
The transition body \(\mathcal T_K\) can be interpreted in the framework of
Orlicz projection bodies. Let
\[
\varphi(t)=\log\cosh(t/2).
\]
Then
\[
\varphi(0)=0,\qquad
\varphi'(t)=\frac12\tanh(t/2),\qquad
\varphi''(t)=\frac14\operatorname{sech}^2(t/2)>0.
\]
Hence \(\varphi\) belongs to the strictly convex Orlicz class used in the
definition of Orlicz projection bodies.

With this notation, the function \(\Phi_K\) may be written as
\[
\Phi_K(y)
=
\int_{\partial K}
(x\cdot n_K(x))
\varphi\left(
\frac{y\cdot n_K(x)}
{x\cdot n_K(x)}
\right)
\,dS(x).
\]
This is precisely the Orlicz modular that appears in the definition of the
Orlicz projection body, with the normalization of the level set adapted to the
present problem. Equivalently, if we define the gauge
\[
\|y\|_{\varphi,K}
=
\inf\left\{
\lambda>0:
\int_{\partial K}
(x\cdot n_K(x))
\varphi\left(
\frac{y\cdot n_K(x)}
{\lambda\,x\cdot n_K(x)}
\right)
\,dS(x)
\le 1
\right\},
\]
then
\[
\mathcal T_K
=
\{y\in\mathbb R^n:\|y\|_{\varphi,K}\le1\}.
\]
Thus \(\mathcal T_K\) is the polar unit ball associated with the Orlicz
projection-body construction for the Orlicz function
\(\varphi(t)=\log\cosh(t/2)\), up to the choice of normalization in the
functional (which is different from the one in \cite{LutwakYangZhang2010OrliczProjection}). For more on Orlicz bodies we refer to \cite{GardnerHugWeil2014OrliczBM} and to \cite{GardnerHugWeilYe2015DualOrliczBM,ZhuZhouXu2014DualOrliczBM} for its dual theory where radial functions play an important role, in the same way they do in this manuscript.
