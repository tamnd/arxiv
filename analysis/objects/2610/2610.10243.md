<!-- theorem-1: body_md -->
\label{thm:master}
Let $\Lambda\subseteq \mathcal K\setminus\mathcal K_0$ be a finite set of local reduction data, and let $\Pi$ be a finite set of primes. Suppose that $\operatorname{Re}(s)\ge 0$, that $|u|\le 1$ and $|w|\le 1$, that $|z_\ell|\le 1$ for every $\ell\in\Pi$, and that $|u_K|\le 1$ for every $K\in\Lambda$. Then the limit
\[
\mathcal Z_{\Lambda,\Pi}(s;u,w,\mathbf z,\mathbf u):=
\lim_{X\to\infty}\mathcal Z_{\Lambda,\Pi,X}(s;u,w,\mathbf z,\mathbf u)
\]
exists. Moreover, the Euler product on the right side of \eqref{eq:master-product} converges absolutely and uniformly on every compact subset of the closed parameter domain
\[
\{\operatorname{Re}(s)\ge 0,\ |u|\le 1,\ |w|\le 1,\ |z_\ell|\le 1\ \forall\ell\in\Pi,\ |u_K|\le 1\ \forall K\in\Lambda\},
\]
and
\begin{equation}\label{eq:master-product}
\mathcal Z_{\Lambda,\Pi}(s;u,w,\mathbf z,\mathbf u)
=
\prod_p
\left(
\beta_p+\sum_{K\in\mathcal K\setminus\mathcal K_0}\delta_p(K)\,\Phi_p(K;s,u,w,\mathbf z,\mathbf u)
\right).
\end{equation}

<!-- remark-1: body_md -->
Theorem~\ref{thm:master} gives the generating function we use to study the arithmetic statistics of local reduction data and Tamagawa products for elliptic curves ordered by naive height. In particular:
\begin{enumerate}
\item Setting $\Lambda=\varnothing$, $u=w=1$, and $z_\ell=1$ for all $\ell\in\Pi$ recovers the Euler product for $L_{\mathrm{Tam}}(s)$;
\item Setting $\Lambda=\varnothing$ and $w=1$, $z_\ell=1$ for all $\ell\in\Pi$ gives the joint generating function for $(\omega_{\mathrm{Tam}}(E),\mathrm{Tam}(E))$;
\item Setting $\Lambda=\varnothing$, $u=1$, $s=0$, and $z_\ell=1$ encodes $\Omega(\mathrm{Tam}(E))$;
\item Setting $\Lambda=\varnothing$, $u=w=1$, and $s=0$ encodes the fixed valuations $\bigl(v_\ell(\mathrm{Tam}(E))\bigr)_{\ell\in\Pi}$;
\item Setting $\Pi=\varnothing$, $u=w=1$, and $s=0$ encodes the joint frequency vector $(\omega_K(E))_{K\in\Lambda}$ for local reduction data.
\end{enumerate}
Summing the local densities over the fibers of $c:\mathcal K\to\mathbb{Z}_{\ge1}$ recovers the scalar local factors of \cite{GOT}.

<!-- theorem-2: body_md -->
\label{thm:omegaK}
Let $\Lambda\subseteq\mathcal K\setminus\mathcal K_0$ be a finite set of local reduction data. For each multi-index $\mathbf r=(r_K)_{K\in\Lambda}\in\mathbb{Z}_{\ge 0}^{\Lambda}$, the density
\[
\pi_\Lambda(\mathbf r):=\lim_{X\to\infty}
\frac{\#\{E:\mathrm{ht}(E)\le X,\ \omega_K(E)=r_K\ \text{for all }K\in\Lambda\}}{\mathcal N(X)}
\]
exists. Its joint generating function is
\begin{equation}\label{eq:omegaK-gf}
\sum_{\mathbf r\in\mathbb{Z}_{\ge 0}^{\Lambda}}\pi_\Lambda(\mathbf r)\,\mathbf u^{\mathbf r}
=
\prod_{p\ \text{prime}}\!\left(1-\sum_{K\in\Lambda}\delta_p(K)+\sum_{K\in\Lambda}\delta_p(K)u_K\right)
\qquad (\mathbf u\in\mathbb{C}^\Lambda).
\end{equation}
The marginal distribution of each $\omega_K(E)$ has mean and variance
\begin{equation}\label{eq:omegaK-mean-var}
\mathbf E[\omega_K(E)]=\sum_{p\ \text{prime}}\delta_p(K),\qquad
\mathbf{Var}(\omega_K(E))=\sum_{p\ \text{prime}}\delta_p(K)(1-\delta_p(K)),
\end{equation}
and for distinct $K,K'\in\mathcal K\setminus\mathcal K_0$ we have
\begin{equation}\label{eq:omegaK-cov}
\mathbf{Cov}(\omega_K(E),\omega_{K'}(E))=-\sum_{p\ \text{prime}}\delta_p(K)\delta_p(K').
\end{equation}
All of these sums converge absolutely.

<!-- theorem-3: body_md -->
\label{thm:omegaE}
For each integer $r\ge 0$, the density
\[
\pi_r:=\lim_{X\to\infty}
\frac{\#\{E:\mathrm{ht}(E)\le X,\ \omega_{\mathrm{Tam}}(E)=r\}}{\mathcal N(X)}
\]
exists. Its generating function is
\begin{equation}\label{eq:omegaE-gf}
\sum_{r\ge 0}\pi_r u^r
=
\prod_{p\ \text{prime}}\bigl(\delta_p(1)+(1-\delta_p(1))u\bigr)
\qquad (u\in \mathbb{C}).
\end{equation}
The mean and variance are
\begin{equation}\label{eq:omegaE-mean-var}
\mathbf E[\omega_{\mathrm{Tam}}(E)] = \sum_{p\ \text{prime}} \bigl(1-\delta_p(1)\bigr)
\qquad {\text {and}}\qquad
\mathbf{Var}(\omega_{\mathrm{Tam}}(E)) = \sum_{p\ \text{prime}} \delta_p(1)\bigl(1-\delta_p(1)\bigr),
\end{equation}
and both sums converge absolutely.

<!-- theorem-4: body_md -->
\label{thm:Omega}
For each integer $b\ge 0$, the density
\[
\rho_b:=\lim_{X\to\infty}
\frac{\#\{E:\mathrm{ht}(E)\le X,\ \Omega(\mathrm{Tam}(E))=b\}}{\mathcal N(X)}
\]
exists. Its generating function is
\begin{equation}\label{eq:Omega-gf}
\sum_{b\ge 0}\rho_b w^b
=
\prod_{p\ \text{prime}}\left(\sum_{t\ge 1}\delta_p(t)w^{\Omega(t)}\right)
\qquad (|w|\le 1).
\end{equation}
Moreover, we have
\begin{equation}\label{eq:Omega-mean}
\mathbf E[\Omega(\mathrm{Tam}(E))]
=
\sum_p\sum_{t\ge 1}\delta_p(t)\Omega(t)
<\infty.
\end{equation}
Consequently, we have
\begin{equation}\label{eq:smallomega-bound}
\mathbf E[\omega(\mathrm{Tam}(E))]\le \mathbf E[\Omega(\mathrm{Tam}(E))]<\infty.
\end{equation}

<!-- theorem-5: body_md -->
\label{thm:valuations}
Let $\Pi$ be a finite set of primes. For each multi-index $\mathbf j=(j_\ell)_{\ell\in\Pi}\in\mathbb{Z}_{\ge 0}^{\Pi}$, the density
\[
Q_\Pi(\mathbf j)
:=
\lim_{X\to\infty}
\frac{\#\{E:\mathrm{ht}(E)\le X,\ v_\ell(\mathrm{Tam}(E))=j_\ell\ \text{for all }\ell\in\Pi\}}{\mathcal N(X)}
\]
exists, and the generating function is
\begin{equation}\label{eq:valuation-gf}
\sum_{\mathbf j\in\mathbb{Z}_{\ge 0}^{\Pi}}Q_\Pi(\mathbf j)\mathbf z^{\mathbf j}
=
\prod_{p\ \text{prime}}\left(\sum_{t\ge 1}\delta_p(t)\prod_{\ell\in\Pi}z_\ell^{v_\ell(t)}\right)
\qquad (|z_\ell|\le 1).
\end{equation}
In particular, for any finite set $A$ of primes, we have
\begin{equation}\label{eq:noA-divisibility}
\mathbf P\bigl(\ell\nmid \mathrm{Tam}(E)\ \text{for all }\ell\in A\bigr)
=
\prod_{p\ \text{prime}}\left(\sum_{\substack{t\ge 1 \\ (t,\prod_{\ell\in A}\ell)=1}}\delta_p(t)\right).
\end{equation}
Hence, for a fixed prime $\ell$, we have
\begin{equation}\label{eq:ell-divides-Tam}
\mathbf P\bigl(\ell\mid \mathrm{Tam}(E)\bigr)
=
1-
\prod_p\left(\sum_{\ell\nmid t}\delta_p(t)\right).
\end{equation}
In particular, the density of curves with odd Tamagawa product is
\begin{equation}\label{eq:odd-density}
\mathbf P\bigl(\mathrm{Tam}(E)\text{ is odd}\bigr)
=
\prod_p\left(\sum_{t\text{ odd}}\delta_p(t)\right).
\end{equation}

<!-- theorem-6: body_md -->
\label{thm:covariance-vl}
Suppose that $\ell$ and $\ell'$ are distinct primes. Then the following are true.

\noindent
(1) We have
\begin{displaymath}
\begin{split}
\mathbf{Cov}(v_\ell(\mathrm{Tam}(E)),&v_{\ell'}(\mathrm{Tam}(E)))
\\
&
\ \ \ =\sum_p\Bigl(
\sum_{t\ge 1}\delta_p(t)v_\ell(t)v_{\ell'}(t)
-
\bigl(\textstyle\sum_{t\ge 1}\delta_p(t)v_\ell(t)\bigr)
\bigl(\textstyle\sum_{t\ge 1}\delta_p(t)v_{\ell'}(t)\bigr)
\Bigr),
\end{split}
\end{displaymath}
where the outer sum converges absolutely.

\noindent
(2) This covariance is strictly negative; for example
\[
\mathbf{Cov}(v_2(\mathrm{Tam}(E)),v_3(\mathrm{Tam}(E)))\approx -0.02489.
\]

 \noindent
 (3) There is an absolute constant $C>0$ such that 
$$
|\mathbf{Cov}(v_\ell(\mathrm{Tam}(E)),v_{\ell'}(\mathrm{Tam}(E)))|\le \frac{C}{(\ell\ell')^{2}}.
$$
In particular, the covariance tends to $0$ as $\ell\ell'\to\infty$.

\label{thm:corr}

<!-- theorem-8: body_md -->
\label{thm:tail}
For every integer $k\ge 0$, the $k$th moment
\begin{equation}\label{eq:moment-def}
M_k:=\sum_{m\ge 1}P_{\mathrm{Tam}}(m)m^k
\end{equation}
converges absolutely and is given by the Euler product
\begin{equation}\label{eq:moment-product}
M_k=
\prod_p\left(\sum_{t\ge 1}\delta_p(t)t^k\right).
\end{equation}
Consequently, for every integer $k\ge 1$ and every $M\ge 1$, we have
\begin{equation}\label{eq:tail-bound}
\sum_{m\ge M}P_{\mathrm{Tam}}(m)
\le
\frac{M_k}{M^k}.
\end{equation}
In particular, the cumulative tail $\sum_{m\ge M}P_{\mathrm{Tam}}(m)$ decays faster than every fixed negative power of $M$.
<!-- end theorem-8 -->
<!-- remark-2: body_md -->
Theorem~\ref{thm:master} gives the Euler products for $\omega_{\mathrm{Tam}}(E)$, $\Omega(\mathrm{Tam}(E))$, and the fixed valuation vector $\mathbf v_\Pi(\mathrm{Tam}(E))$. The statistic $\omega(\mathrm{Tam}(E))$, the number of distinct prime divisors of the global integer $\mathrm{Tam}(E)$, is not additive in the local factors $c_p(E)$. As a consequence, there is no one-variable Euler product obtained by inserting a factor $u^{\omega(t)}$ at each prime $p$. We therefore study fixed valuation statistics first, and then recover finite-support divisibility information from them. This gives \eqref{eq:noA-divisibility}--\eqref{eq:ell-divides-Tam}.

\label{thm:GOT-local}
The numbers $\delta_p(t)$ satisfy the following formulas.
\begin{enumerate}
\item For $p=2$, we have
\[
\delta_2(1)=\frac{241}{396},\qquad
\delta_2(2)=\frac{7495}{24552},\qquad
\delta_2(3)=\frac{1153}{16368},\qquad
\delta_2(4)=\frac{171}{10912},
\]
and for $t\ge 5$, we have
\[
\delta_2(t)=\frac{1}{2^{t+1}\cdot 1023}.
\]

\item For $p=3$, we have
\[
\delta_3(1)=\frac{1924841}{2125728},\qquad
\delta_3(2)=\frac{509345}{6377184},\qquad
\delta_3(3)=\frac{30619}{2391444},\qquad
\delta_3(4)=\frac{1193}{652212},
\]
and for $t\ge 5$, we have
\[
\delta_3(t)=\frac{1}{3^{t+1}\cdot 29524}.
\]

\item If $p\ge 5$, then we have
\[
\delta_p(t)=
\begin{cases}
\displaystyle
1-
\frac{p(6p^7+9p^6+9p^5+7p^4+8p^3+7p^2+9p+6)}{6(p+1)^2(p^8+p^6+p^4+p^2+1)}, & t=1,\\[14pt]
\displaystyle
\frac{p(2p^7+2p^6+p^5+p^4+2p^3+p^2+2p+2)}{2(p+1)^2(p^8+p^6+p^4+p^2+1)}, & t=2,\\[14pt]
\displaystyle
\frac{p^2(p^4+1)}{2(p+1)(p^8+p^6+p^4+p^2+1)}, & t=3,\\[14pt]
\displaystyle
\frac{p^3(3p^2-2p+1)}{6(p+1)(p^8+p^6+p^4+p^2+1)}, & t=4,\\[14pt]
\displaystyle
\frac{p^{10}-2p^9+p^8}{2p^t(p^{10}-1)}, & t\ge 5.
\end{cases}
\]
\end{enumerate}
Moreover, we have
\begin{equation}\label{eq:GOT-Euler}
\sum_{m\ge 1}\frac{P_{\mathrm{Tam}}(m)}{m^s}
=
\prod_p\left(\sum_{t\ge 1}\delta_p(t)t^{-s}\right)
\end{equation}
for every $s$ for which the Euler product converges absolutely.

<!-- remark-3: body_md -->
\label{rem:local-normalization}
For each prime $p$, the local densities satisfy
\[
\sum_{t\ge 1}\delta_p(t)=1.
\]
For $p=2,3$ this is immediate from the explicit geometric tails. For $p\ge 5$, this is a direct algebraic check from the formulas in Theorem~\ref{thm:GOT-local}; with the $t=4$ numerator written as $3p^2-2p+1$, multiplying the sum of the five displayed cases by the common denominator $6(p+1)^2(p^{10}-1)$ gives exactly $6(p+1)^2(p^{10}-1)$. Equivalently, the $t=4$ term is the sum of the $I_4$, $I_0^\ast$, and $I_n^\ast$ contributions in Table~5 of \cite{GOT}, and that sum simplifies to the displayed numerator $3p^2-2p+1$. In particular, the numerator $3p^2-2p-1$ printed in Lemma~3.1 of \cite{GOT} for this case is an error. The corresponding values in Table~\ref{tab:cutoff-stability} use this corrected local factor.

<!-- lemma-1: body_md -->
\label{lem:K0-asymptotic}
For every prime $p\ge 5$, we have
\begin{equation}\label{eq:K0-mass}
1-\beta_p=1-\delta_p(I_0)-\delta_p(I_1^{\mathrm{sp}})-\delta_p(I_1^{\mathrm{ns}})=\sum_{K\notin\mathcal K_0}\delta_p(K)\ll \frac{1}{p^2}.
\end{equation}
Moreover, the individual $\mathcal K_0$-densities satisfy
\begin{equation}\label{eq:K0-individual}
\delta_p(I_0)=1-O\!\left(\frac{1}{p}\right),\qquad
\delta_p(I_1^{\mathrm{sp}})\asymp \delta_p(I_1^{\mathrm{ns}})\asymp \frac{1}{p}.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
For every $n\ge 1$ and $p\ge 5$, the split and nonsplit multiplicative densities are
\[
\delta_p(I_n^{\mathrm{sp}})=\delta_p(I_n^{\mathrm{ns}})=\frac{(p-1)^2}{2p^{n+2}}\bigl(1-p^{-10}\bigr)^{-1}.
\]
Indeed, the nodal locus modulo $p$ consists of the $p-1$ residue pairs $(a_4,a_6)\equiv(-3u^2,2u^3)$ with $u\in\mathbb{F}_p^{\times}$. On each of them, the condition $v_p(\Delta)=n$ has relative density $(p-1)/p^{n}$, since $\partial\Delta/\partial a_6=-864a_6$ is a unit there. The quadratic twist $(a_4,a_6)\mapsto(\nu^2a_4,\nu^3a_6)$ by a quadratic non-residue $\nu$ is a measure-preserving bijection that preserves $v_p(\Delta)$ and exchanges the split and nonsplit branches. The factor $(1-p^{-10})^{-1}$ accounts for the non-minimal short models. The additive types contribute a total density $O(p^{-2})$. Summing the multiplicative densities over $n\ge 1$ gives the geometric series
\[
\sum_{n\ge 1}(\delta_p(I_n^{\mathrm{sp}})+\delta_p(I_n^{\mathrm{ns}}))
=
\frac{p-1}{p^2}\bigl(1-p^{-10}\bigr)^{-1}
=
\frac{1}{p}-\frac{1}{p^2}+O(p^{-11}),
\]
so the total bad-reduction density is $1/p+O(p^{-2})$, and \eqref{eq:K0-individual} follows. The $\mathcal K_0$-mass estimate \eqref{eq:K0-mass} then comes from
\[
\sum_{K\notin\mathcal K_0}\delta_p(K)
=
\sum_{n\ge 2}\bigl(\delta_p(I_n^{\mathrm{sp}})+\delta_p(I_n^{\mathrm{ns}})\bigr)+\delta_p(\text{additive types})
=O(p^{-2}),
\]
since the tail of the geometric series is $O(p^{-2})$ and the additive contribution is $O(p^{-2})$.

<!-- lemma-2: body_md -->
\label{lem:local-estimates}
For primes $p\ge 5$ we have
\begin{equation}\label{eq:d1-estimate}
1-\delta_p(1)\ll \frac{1}{p^2}.
\end{equation}
More generally, for every fixed integer $k\ge 0$, we have
\begin{equation}\label{eq:moment-local-estimate}
\sum_{t\ge 1}\delta_p(t)t^k = 1+O_k\!\left(\frac{1}{p^2}\right)
\qquad (p\ge 5).
\end{equation}
Finally, let $\Lambda\subseteq\mathcal K\setminus\mathcal K_0$ be finite, let $\Pi$ be a finite set of primes, and suppose that $|u|\le 1$, $|w|\le 1$, $|z_\ell|\le 1$ for every $\ell\in\Pi$, and $|u_K|\le 1$ for every $K\in\Lambda$. Then we have
\begin{equation}\label{eq:bounded-local-estimate}
\beta_p+\sum_{K\in\mathcal K\setminus\mathcal K_0}\delta_p(K)\,\Phi_p(K;s,u,w,\mathbf z,\mathbf u)
=
1+O\!\left(\frac{1}{p^2}\right)
\end{equation}
for every $s$ with $\operatorname{Re}(s)\ge 0$. Moreover, for every fixed prime $\ell$, we have
\begin{equation}\label{eq:Omega-local-estimate}
\sum_{t\ge 1}\delta_p(t)\Omega(t)\ll \frac{1}{p^2},
\qquad
\sum_{t\ge 1}\delta_p(t)v_\ell(t)\ll \frac{1}{p^2}
\qquad (p\ge 5),
\end{equation}
\begin{equation}\label{eq:second-moment-local-estimate}
\sum_{t\ge 1}\delta_p(t)\Omega(t)^2\ll \frac{1}{p^2},
\qquad
\sum_{t\ge 1}\delta_p(t)v_\ell(t)v_{\ell'}(t)\ll \frac{1}{p^2}
\qquad (p\ge 5),
\end{equation}
and for every fixed $K\in\mathcal K\setminus\mathcal K_0$,
\begin{equation}\label{eq:K-local-estimate}
\delta_p(K)\ll \frac{1}{p^2}
\qquad (p\ge 5).
\end{equation}
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
For \eqref{eq:d1-estimate}, inspect the explicit formula for $\delta_p(1)$ in Theorem~\ref{thm:GOT-local}. The numerator has degree $8$ in $p$ and the denominator has degree $10$, so the difference $1-\delta_p(1)$ is $O(p^{-2})$.

For \eqref{eq:moment-local-estimate}, split the sum into $t=1,2,3,4$ and $t\ge 5$. The $t=1$ term equals $1+O(p^{-2})$, while the terms with $t=2,3,4$ are each $O_k(p^{-2})$ because the corresponding formulas are visibly $O(p^{-2})$. For the tail, Theorem~\ref{thm:GOT-local} gives
\[
\delta_p(t)=\frac{p^8(p-1)^2}{2(p^{10}-1)}\cdot p^{-t}
\ll p^{-t}
\qquad (t\ge 5),
\]
so
\[
\sum_{t\ge 5}\delta_p(t)t^k
\ll
\sum_{t\ge 5}\frac{t^k}{p^t}
\ll_k \frac{1}{p^5}.
\]
This proves \eqref{eq:moment-local-estimate}.

For \eqref{eq:bounded-local-estimate}, recall from \eqref{eq:Phi-def} that $\Phi_p(K)=1$ for every $K\in\mathcal K_0$, so the $\mathcal K_0$ contribution is exactly $\beta_p$ and cancels the leading $1$ up to $1-\beta_p\ll p^{-2}$ by Lemma~\ref{lem:K0-asymptotic}. The hypotheses on the parameters give $|\Phi_p(K)|\le 1$ for every $K$, so
\[
\left|\beta_p+\sum_{K\notin\mathcal K_0}\delta_p(K)\Phi_p(K)-1\right|
=
\left|\sum_{K\notin\mathcal K_0}\delta_p(K)\bigl(\Phi_p(K)-1\bigr)\right|
\le
2(1-\beta_p)
\ll
\frac{1}{p^2}.
\]
This proves \eqref{eq:bounded-local-estimate}.

For \eqref{eq:Omega-local-estimate}, note that $\Omega(1)=v_\ell(1)=0$ and, for every $t\ge 2$,
\[
0\le v_\ell(t)\le \Omega(t)\le t.
\]
Hence
\[
\sum_{t\ge 1}\delta_p(t)\Omega(t)
\le
\sum_{t\ge 2}\delta_p(t)t
=
\left(\sum_{t\ge 1}\delta_p(t)t\right)-\delta_p(1)
=O(p^{-2})
\]
by \eqref{eq:moment-local-estimate} with $k=1$, and the same bound follows for $\sum_t\delta_p(t)v_\ell(t)$.

For \eqref{eq:second-moment-local-estimate}, use $v_\ell(t)v_{\ell'}(t)\le \Omega(t)^2\le t^2$ for $t\ge1$ and $\Omega(1)=0$. Thus
\[
\sum_{t\ge1}\delta_p(t)\Omega(t)^2
\le
\sum_{t\ge2}\delta_p(t)t^2
=
\sum_{t\ge1}\delta_p(t)t^2-\delta_p(1)
=O(p^{-2})
\]
by \eqref{eq:moment-local-estimate} with $k=2$, together with \eqref{eq:d1-estimate}. The mixed valuation bound follows immediately.

Finally, \eqref{eq:K-local-estimate} is immediate from Lemma~\ref{lem:K0-asymptotic}, since $\delta_p(K)\le \sum_{K'\notin\mathcal K_0}\delta_p(K')\ll p^{-2}$.

\label{lem:equidistribution}
Let $M\ge 1$ be fixed. For every residue class $(r,s)\in (\mathbb{Z}/M\mathbb{Z})^2$, we have
\[
\lim_{X\to\infty}
\frac{1}{\mathcal N(X)}
\#\Bigl\{(a_4,a_6)\in\mathbb{Z}^2:
\begin{array}{l}
\mathrm{ht}(E(a_4,a_6))\le X,\ \Delta(E)\neq 0,\\
(a_4,a_6)\equiv (r,s)\!\!\pmod M
\end{array}
\Bigr\}
=
\frac{1}{M^2}.
\]
More generally, if $A\subseteq (\mathbb{Z}/M\mathbb{Z})^2$, then the density of curves whose coefficient pair lies in $A$ modulo $M$ exists and equals $|A|/M^2$.
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
We let
\[
A_X:=\left\lfloor\left(\frac{X}{4}\right)^{1/3}\right\rfloor
\qquad \text{and}\qquad
B_X:=\left\lfloor\left(\frac{X}{27}\right)^{1/2}\right\rfloor.
\]
Then the condition $\mathrm{ht}(E(a_4,a_6))\le X$ is equivalent to $|a_4|\le A_X$ and $|a_6|\le B_X$. Hence the total number of integer pairs in the box is
\[
(2A_X+1)(2B_X+1)=\frac{4}{4^{1/3}27^{1/2}}X^{5/6}+O(X^{1/2}+X^{1/3}).
\]
The singular pairs satisfy $4a_4^3+27a_6^2=0$. For each $a_6$ there is at most one possible value of $a_4$, so the number of singular pairs in the box is $O(B_X)=O(X^{1/2})$. Therefore
\[
\mathcal N(X)=(2A_X+1)(2B_X+1)+O(X^{1/2})
=\frac{4}{4^{1/3}27^{1/2}}X^{5/6}+o(X^{5/6}).
\]
Now fix $(r,s)\bmod M$. The number of pairs in the box satisfying these congruence conditions is
\[
\frac{2A_X+1}{M}\cdot \frac{2B_X+1}{M}+O(A_X+B_X+1)
=
\frac{(2A_X+1)(2B_X+1)}{M^2}+o(X^{5/6}),
\]
and again removing the singular pairs changes the count by only $O(X^{1/2})=o(X^{5/6})$. Dividing by $\mathcal N(X)$ gives the claim for a single residue class, and summing over residue classes in $A$ proves the general statement.

\label{lem:uniform-large-prime-tail}
For every prime $p\ge 5$ and every $X\ge 4$, we have
\[
\frac{1}{\mathcal N(X)}
\#\{E(a_4,a_6): \mathrm{ht}(E)\le X,\ \tau_p(E)\notin\mathcal K_0\}
\ll
\frac{1}{p^2}+\frac{1}{pX^{1/3}}+\frac{1}{X^{1/2}}.
\]
Consequently, for every real number $Y\ge 5$, we have
\[
\limsup_{X\to\infty}
\frac{1}{\mathcal N(X)}
\sum_{\mathrm{ht}(E)\le X}\sum_{p>Y}\1_{\tau_p(E)\notin\mathcal K_0}
\ll
\sum_{p>Y}\frac{1}{p^2},
\]
and the right-hand side tends to $0$ as $Y\to\infty$.

<!-- proof-5: body_md -->
If $4\le X<27$, the only nonsingular models in the height box are $y^2=x^3\pm x$, whose discriminants are $\mp64$. Both have good reduction at every prime $p\ge5$, so the proportion in the first assertion is zero. We may therefore assume that $X\ge27$.

Write
\[
A_X:=\left\lfloor\left(\frac{X}{4}\right)^{1/3}\right\rfloor
\qquad {\text {\rm and}}\qquad
B_X:=\left\lfloor\left(\frac{X}{27}\right)^{1/2}\right\rfloor,
\]
and let
\[
N_p(X):=\#\{E(a_4,a_6): \mathrm{ht}(E)\le X,\ \tau_p(E)\notin\mathcal K_0\}.
\]
If $p\ge 5$ and $\tau_p(E)\notin\mathcal K_0$, then Tate's algorithm shows that the $p$-minimal discriminant has valuation at least $2$. Since the discriminant of the original integral short Weierstrass equation differs from the minimal discriminant by a factor with $p$-adic valuation divisible by $12$, this implies $v_p(\Delta(E))\ge 2$. Indeed, if the minimal discriminant valuation is at most $1$, then $\tau_p(E)$ is $I_0$, $I_1^{\mathrm{sp}}$ or $I_1^{\mathrm{ns}}$, that is, $\tau_p(E)\in\mathcal K_0$. Thus, every curve counted by $N_p(X)$ satisfies
\[
p^2\mid \Delta(E)=-16(4a_4^3+27a_6^2),
\]
so it is enough to count pairs $(a_4,a_6)$ in the height box for which
\[
4a_4^3+27a_6^2\equiv 0 \pmod{p^2}.
\]

\[
27a_6^2\equiv -4a_4^3 \pmod{p^2}
\]
has at most two solutions for $a_6\bmod p^2$, because a nonzero residue modulo $p^2$ has at most two square roots. Hence, the number of such integers $a_6$ with $|a_6|\le B_X$ is
\[
\ll \frac{B_X}{p^2}+1.
\]
If instead $p\mid a_4$, then the same congruence forces $p\mid a_6$, so the number of admissible integers $a_6$ is
\[
\ll \frac{B_X}{p}+1.
\]
Since the number of integers $a_4$ with $|a_4|\le A_X$ and $p\mid a_4$ is $\ll A_X/p+1$, it follows that
\[
N_p(X)
\ll
(A_X+1)\left(\frac{B_X}{p^2}+1\right)
+
\left(\frac{A_X}{p}+1\right)\left(\frac{B_X}{p}+1\right).
\]
Therefore, we find that
\[
N_p(X)
\ll
\frac{A_XB_X}{p^2}+A_X+\frac{B_X}{p}+1.
\]
By Lemma~\ref{lem:equidistribution} we have $\mathcal N(X)\asymp A_XB_X$, with $A_X\asymp X^{1/3}$ and $B_X\asymp X^{1/2}$, so dividing through by $\mathcal N(X)$ gives
\[
\frac{N_p(X)}{\mathcal N(X)}
\ll
\frac{1}{p^2}+\frac{1}{pX^{1/3}}+\frac{1}{X^{1/2}}.
\]

For the second statement, note that $\mathrm{ht}(E)\le X$ implies
\[
|4a_4^3+27a_6^2|\le X+X=2X,
\]
and hence we obtain
\[
|\Delta(E)|\le 32X.
\]
Therefore, if $\tau_p(E)\notin\mathcal K_0$, then $p^2\mid \Delta(E)\neq 0$, so necessarily $p\le (32X)^{1/2}$. Summing the first bound over primes $Y<p\le (32X)^{1/2}$, we obtain
\[
\frac{1}{\mathcal N(X)}
\sum_{\mathrm{ht}(E)\le X}\sum_{p>Y}\1_{\tau_p(E)\notin\mathcal K_0}
\ll
\sum_{Y<p\le (32X)^{1/2}}\frac{1}{p^2}
+
\frac{1}{X^{1/3}}\sum_{Y<p\le (32X)^{1/2}}\frac{1}{p}
+
\frac{\pi((32X)^{1/2})}{X^{1/2}}.
\]
The second term is $O(X^{-1/3}\log X)$, and the third is $O(1/\log X)$ by the standard estimate $\pi(T)\ll T/\log T$. Taking the limit superior as $X\to\infty$ gives
\[
\limsup_{X\to\infty}
\frac{1}{\mathcal N(X)}
\sum_{\mathrm{ht}(E)\le X}\sum_{p>Y}\1_{\tau_p(E)\notin\mathcal K_0}
\ll
\sum_{p>Y}\frac{1}{p^2},
\]
and the right-hand side tends to $0$ as $Y\to\infty$.

\label{lem:local-constancy}
Fix a prime $p$, and let
\[
U_p:=\{(a_4,a_6)\in\mathbb{Z}_p^2:\Delta(a_4,a_6)\neq 0\}.

<!-- proof-6: body_md -->

<!-- proposition-1: body_md -->
\label{prop:finite-prime}
Let $S$ be a finite set of primes. For each $p\in S$, let $f_p:\mathcal K\to\mathbb{C}$ be bounded. Then the natural density
\[
\lim_{X\to\infty}\frac{1}{\mathcal N(X)}\sum_{\mathrm{ht}(E)\le X}\prod_{p\in S} f_p(\tau_p(E))
\]
exists and equals
\begin{equation}\label{eq:finite-prime-factorization}
\prod_{p\in S}\left(\sum_{K\in\mathcal K}\delta_p(K)f_p(K)\right).
\end{equation}
<!-- end proposition-1 -->
<!-- proof-7: body_md -->
Let $\mu_p$ be normalized Haar measure on $\mathbb{Z}_p^2$, and let
\[
K_S:=\prod_{p\in S}\mathbb{Z}_p^2
\qquad {\text {\rm and}}\qquad
\mu_S:=\bigotimes_{p\in S}\mu_p.
\]
For each $X$, define a probability measure $\mu_{S,X}$ on $K_S$ by
\[
\mu_{S,X}:=
\frac{1}{\mathcal N(X)}
\sum_{\mathrm{ht}(E(a_4,a_6))\le X}
\delta_{((a_4,a_6))_{p\in S}},
\]
where the sum runs over nonsingular pairs and $\delta_y$ denotes the unit point mass at $y$.

If $C\subseteq K_S$ is a cylinder set cut out by congruence conditions modulo $p^{\nu_p}$ for $p\in S$, then $C$ corresponds to a subset $A\subseteq (\mathbb{Z}/M\mathbb{Z})^2$ with
\[
M:=\prod_{p\in S}p^{\nu_p}.
\]
Lemma~\ref{lem:equidistribution} therefore gives
\[
\mu_{S,X}(C)\to \mu_S(C)
\qquad (X\to\infty).
\]
Cylinder sets are clopen, so their indicators are continuous. Finite linear combinations of these indicators form the algebra of locally constant functions on $K_S$, which separates points and contains the constants. By the Stone--Weierstrass theorem, this algebra is uniformly dense in $C(K_S)$; see \cite[Theorem~7.33]{RudinPMA}. Hence $\mu_{S,X}$ converges weakly to $\mu_S$.

Now define
\[
G((x_p)_{p\in S}) := \prod_{p\in S} f_p(\tau_p(x_p))
\]
on $\prod_{p\in S}U_p$, where $U_p$ is as in Lemma~\ref{lem:local-constancy}, and extend $G$ arbitrarily to all of $K_S$. By Lemma~\ref{lem:local-constancy}, $G$ is bounded and continuous on $\prod_{p\in S}U_p$. Its discontinuity set is contained in
\[
K_S\setminus \prod_{p\in S}U_p,
\]
which has $\mu_S$-measure $0$. Since $\mathbb{Z}_p^2\setminus U_p$ has Haar measure zero for each $p\in S$, the values of the extension on the singular locus do not affect the integral. The Portmanteau theorem for bounded functions whose discontinuity set has $\mu_S$-measure zero therefore gives
\[
\int_{K_S} G\,d\mu_{S,X}\to \int_{K_S} G\,d\mu_S;
\]
see \cite[Section~2]{Billingsley}.
The left side is exactly
\[
\frac{1}{\mathcal N(X)}\sum_{\mathrm{ht}(E)\le X}\prod_{p\in S} f_p(\tau_p(E)).
\]
Since $\mu_S$ is a product measure and $G$ factors prime-by-prime, the right side equals
\[
\prod_{p\in S}
\left(
\int_{\mathbb{Z}_p^2} f_p(\tau_p(x_p))\,d\mu_p(x_p)
\right).
\]
By the definition of the local densities, each local integral is
\[
\int_{\mathbb{Z}_p^2} f_p(\tau_p(x_p))\,d\mu_p(x_p)
=
\sum_{K\in\mathcal K}\delta_p(K)f_p(K).
\]
This proves \eqref{eq:finite-prime-factorization}.

Fix a finite set of primes $S$. By Lemma~\ref{lem:truncated-gf}, the density generating function for the truncated statistics is exactly the finite product on the right side of \eqref{eq:truncated-identity}.

By Lemma~\ref{lem:local-estimates}, the infinite product in \eqref{eq:master-product} converges absolutely because its local factors are $1+O(p^{-2})$, and $\sum_p p^{-2}<\infty$. In particular, it converges uniformly on every compact subset of the closed parameter set
\[
\{\operatorname{Re}(s)\ge 0,\ |u|\le 1,\ |w|\le 1,\ |z_\ell|\le 1\ \forall\ell\in\Pi,\ |u_K|\le 1\ \forall K\in\Lambda\}.
\]

Let
\[

\]
and
\[
W(E):=u^{\omega_{\mathrm{Tam}}(E)}w^{\Omega(\mathrm{Tam}(E))}\mathbf z^{\mathbf v_\Pi(\mathrm{Tam}(E))}\prod_{K\in\Lambda}u_K^{\omega_K(E)}\mathrm{Tam}(E)^{-s}.
\]
Because $\operatorname{Re}(s)\ge 0$ and all variables have modulus at most $1$, we have
\[
|W_S(E)|\le 1,
\qquad
|W(E)|\le 1.
\]
Moreover $W_S(E)=W(E)$ unless there exists a prime $p\notin S$ with $\tau_p(E)\notin\mathcal K_0$. Hence
\[
|W_S(E)-W(E)|\le 2\sum_{p\notin S}\1_{\tau_p(E)\notin\mathcal K_0}.
\]

Write
\[

\beta_p+\sum_{K\in\mathcal K\setminus\mathcal K_0}\delta_p(K)\,\Phi_p(K;s,u,w,\mathbf z,\mathbf u).
\]

\begin{align*}
\limsup_{X\to\infty}\bigl|\mathcal Z_{\Lambda,\Pi,X}&(s;u,w,\mathbf z,\mathbf u)-\prod_p L_p(s;u,w,\mathbf z,\mathbf u)\bigr| \\
&\le
\limsup_{X\to\infty}\left|\frac{1}{\mathcal N(X)}\sum_{\mathrm{ht}(E)\le X}W(E)-\frac{1}{\mathcal N(X)}\sum_{\mathrm{ht}(E)\le X}W_S(E)\right|\\
&\quad\quad+
\limsup_{X\to\infty}\left|\frac{1}{\mathcal N(X)}\sum_{\mathrm{ht}(E)\le X}W_S(E)-\prod_{p\in S}L_p(s;u,w,\mathbf z,\mathbf u)\right|\\
&\quad \quad+
\left|\prod_{p\in S}L_p(s;u,w,\mathbf z,\mathbf u)-\prod_p L_p(s;u,w,\mathbf z,\mathbf u)\right|.
\end{align*}
The middle term is $0$ by Lemma~\ref{lem:truncated-gf}. For the first term, enlarge $S$ if necessary so that $2,3\in S$, and fix a real number $Y>\max S$ with $Y\ge5$. Then
\begin{align*}
\frac{1}{\mathcal N(X)}\sum_{\mathrm{ht}(E)\le X}|W(E)-W_S(E)|
&\le
2\sum_{\substack{p\le Y\\ p\notin S}}
\frac{\#\{E:\mathrm{ht}(E)\le X,\ \tau_p(E)\notin\mathcal K_0\}}{\mathcal N(X)}\\
&\quad+
2\frac{1}{\mathcal N(X)}
\sum_{\mathrm{ht}(E)\le X}\sum_{p>Y}\1_{\tau_p(E)\notin\mathcal K_0}.
\end{align*}
For each fixed prime $p$, Proposition~\ref{prop:finite-prime} with $S=\{p\}$ and $f_p(K)=\1_{K\notin\mathcal K_0}$ gives
\[
\lim_{X\to\infty}
\frac{\#\{E:\mathrm{ht}(E)\le X,\ \tau_p(E)\notin\mathcal K_0\}}{\mathcal N(X)}
=
1-\beta_p.
\]
Summing over the finitely many primes $p\le Y$ with $p\notin S$, we obtain
\[
\lim_{X\to\infty}
\sum_{\substack{p\le Y\\ p\notin S}}
\frac{\#\{E:\mathrm{ht}(E)\le X,\ \tau_p(E)\notin\mathcal K_0\}}{\mathcal N(X)}
=
\sum_{\substack{p\le Y\\ p\notin S}}(1-\beta_p).
\]
For the tail over $p>Y$, Lemma~\ref{lem:uniform-large-prime-tail} yields
\[
\limsup_{X\to\infty}
\frac{1}{\mathcal N(X)}
\sum_{\mathrm{ht}(E)\le X}\sum_{p>Y}\1_{\tau_p(E)\notin\mathcal K_0}
\ll
\sum_{p>Y}\frac{1}{p^2}.
\]
Hence, we have
\[
\limsup_{X\to\infty}
\frac{1}{\mathcal N(X)}\sum_{\mathrm{ht}(E)\le X}|W(E)-W_S(E)|
\le
2\sum_{\substack{p\le Y\\ p\notin S}}(1-\beta_p)
+
O\!\left(\sum_{p>Y}\frac{1}{p^2}\right).
\]
Letting $Y\to\infty$ and using Lemma~\ref{lem:K0-asymptotic}, which gives $1-\beta_p\ll p^{-2}$, we obtain
\[
\limsup_{X\to\infty}
\frac{1}{\mathcal N(X)}\sum_{\mathrm{ht}(E)\le X}|W(E)-W_S(E)|
\le
2\sum_{p\notin S}(1-\beta_p).
\]
For the third term, absolute convergence of the Euler product and Lemma~\ref{lem:local-estimates} give
\[
\left|\prod_{p\in S}L_p(s;u,w,\mathbf z,\mathbf u)-\prod_p L_p(s;u,w,\mathbf z,\mathbf u)\right|
\to 0
\qquad (S\to\infty).
\]
Since $\sum_p(1-\beta_p)<\infty$ by Lemma~\ref{lem:K0-asymptotic}, letting $S$ increase through finite sets of primes proves \eqref{eq:master-product}. The five specializations are immediate.

\qquad (X\to\infty).

converges uniformly on every polytorus $|z_i|=\rho_i<1$ $(i\in I)$ to

\]
The head sums $A_{p,r}$ admit the lower bounds
\begin{equation}\label{eq:A2A3-lower}
A_{p,2}\ge \frac{1}{2p^2},
\qquad
A_{p,3}\ge \frac{1}{3p^3}
\qquad(p\text{ prime}).
\end{equation}
For $p=2,3$ this is checked directly from the displayed formulas in Theorem~\ref{thm:GOT-local}:
\[
A_{2,2}\ge\delta_2(2)=\frac{7495}{24552}>\frac18,\qquad
A_{2,3}=\delta_2(3)=\frac{1153}{16368}>\frac1{24},
\]
and
\[
A_{3,2}\ge\delta_3(2)=\frac{509345}{6377184}>\frac1{18},\qquad
A_{3,3}=\delta_3(3)=\frac{30619}{2391444}>\frac1{81}.
\]
For $p\ge5$, the first inequality follows from
\[
\delta_p(2)\ge \frac{1}{2p^2},
\]
which is equivalent to
\[
p^{10}-p^8-p^7-p^5-2p^2-2p-1\ge0.
\]
This last polynomial is positive for $p\ge5$ because
$p^5(p^5-p^3-p^2-1)>2p^2+2p+1$. Similarly, the second inequality follows from
\[
\delta_p(3)\ge \frac{1}{3p^3},
\]
which is equivalent to
\[
p^9-2p^8-2p^7-2p^6+p^5-2p^4-2p^3-2p^2-2p-2\ge0;
\]
for $p\ge5$ this is positive since
$p^6(p^3-2p^2-2p-2)>2(p^4+p^3+p^2+p+1)$.

We use the following tail bound. If $r\ne s$ are primes and $q\ge 2$, then
\begin{equation}\label{eq:cross-tail-bound}
\sum_{t\ge 5}q^{-t}v_r(t)v_s(t)
\le
\frac{q^{-rs}}{(1-q^{-rs})^2}.
\end{equation}
The summand vanishes unless $rs\mid t$; and if $i=v_r(t)\ge 1$ and $j=v_s(t)\ge 1$, then $ij\le r^{i-1}s^{j-1}\le t/(rs)$, so
\[
\sum_{t\ge 5}q^{-t}v_r(t)v_s(t)
\le
\sum_{rs\mid t}q^{-t}\frac{t}{rs}
=
\sum_{k\ge1}kq^{-rsk}
=
\frac{q^{-rs}}{(1-q^{-rs})^2}.
\]
For $r\ge 5$ the single tail term $t=r$ gives the complementary lower bound
\begin{equation}\label{eq:B-lower}
B_{p,r}\ge \alpha_p p^{-r}.
\end{equation}

We now prove \eqref{eq:local-cov-negative}. By symmetry assume $\ell<\ell'$.

First let $(\ell,\ell')=(2,3)$. By \eqref{eq:cross-tail-bound},
\[
\sum_{t\ge 1}\delta_p(t)v_2(t)v_3(t)
\le
\alpha_p\frac{p^{-6}}{(1-p^{-6})^2}.
\]
On the other hand, \eqref{eq:A2A3-lower} gives
$A_{p,2}A_{p,3}\ge 1/(6p^5)$. Thus $C_p(2,3)<0$ follows once
\[
\alpha_p\frac{p^{-6}}{(1-p^{-6})^2}<\frac{1}{6p^5}.
\]
For $p\ge5$ this follows from $\alpha_p<1/2$, since
$3/(p(1-p^{-6})^2)<1$. For $p=2$, the required inequality is
\[
\frac{1}{2046}\frac{2^{-6}}{(1-2^{-6})^2}<\frac{1}{6\cdot 2^5},
\]
which is equivalent to $6\cdot 2^5\cdot 2^6<2046\cdot 63^2$. For $p=3$, it is
\[
\frac{1}{88572}\frac{3^{-6}}{(1-3^{-6})^2}<\frac{1}{6\cdot 3^5},
\]
which is equivalent to $6\cdot 3^5\cdot 3^6<88572\cdot 728^2$. Both integer inequalities are immediate.

Next suppose $\ell=2$ and $\ell'\ge5$. By \eqref{eq:cross-tail-bound} and \eqref{eq:B-lower},
\[
\sum_t\delta_p(t)v_2(t)v_{\ell'}(t)
\le
\alpha_p\frac{p^{-2\ell'}}{(1-p^{-2\ell'})^2},
\qquad
A_{p,2}B_{p,\ell'}\ge \frac{\alpha_p}{2p^{\ell'+2}}.
\]
Since
\[
\frac{2p^{2-\ell'}}{(1-p^{-2\ell'})^2}\le
\frac{2\cdot 2^{-3}}{(1-2^{-10})^2}<1,
\]
we have $\sum_t\delta_p(t)v_2(t)v_{\ell'}(t)<A_{p,2}B_{p,\ell'}$, and hence
$C_p(2,\ell')<0$.

The case $\ell=3$ and $\ell'\ge 5$ is analogous: we have
\[
\sum_t\delta_p(t)v_3(t)v_{\ell'}(t)
\le
\alpha_p\frac{p^{-3\ell'}}{(1-p^{-3\ell'})^2},
\qquad
A_{p,3}B_{p,\ell'}\ge \frac{\alpha_p}{3p^{\ell'+3}},
\]
and
\[
\frac{3p^{3-2\ell'}}{(1-p^{-3\ell'})^2}\le
\frac{3\cdot 2^{-7}}{(1-2^{-15})^2}<1,
\]
so $C_p(3,\ell')<0$.

Finally suppose $\ell,\ell'\ge 5$. Then $A_{p,\ell}=A_{p,\ell'}=0$, and \eqref{eq:cross-tail-bound}, \eqref{eq:B-lower}, and \eqref{eq:alpha-bounds} give
\[
\sum_t\delta_p(t)v_\ell(t)v_{\ell'}(t)
\le
\alpha_p\frac{p^{-\ell\ell'}}{(1-p^{-\ell\ell'})^2}
<
\alpha_p^2 p^{-\ell-\ell'}
\le
B_{p,\ell}B_{p,\ell'}.
\]
The middle inequality follows from
\[
\frac{p^{-(\ell\ell'-\ell-\ell')}}{(1-p^{-\ell\ell'})^2}
\le
\frac{2^{-23}}{(1-2^{-35})^2}
<
\frac{1}{88572}
\le \alpha_p,
\]
since for distinct primes $\ell,\ell'\ge 5$ the value $\ell\ell'-\ell-\ell'$ is minimized at $(\ell,\ell')=(5,7)$, where it equals $23$. Summing \eqref{eq:local-cov-negative} over $p$ gives strict negativity of the global covariance.

We now turn to the decay bound in part (3). Write
\[
\mu_{p,r}:=\sum_{t\ge 1}\delta_p(t)v_r(t).
\]
Since $C_p(\ell,\ell')<0$ and both terms in the expression for $C_p$ are nonnegative, we have $|C_p(\ell,\ell')|\le \mu_{p,\ell}\mu_{p,\ell'}$. For $p\ge 5$, the explicit densities and the geometric tail in Theorem~\ref{thm:GOT-local} give
\[
\mu_{p,2}\ll p^{-2},\qquad
\mu_{p,3}\ll p^{-3},\qquad
\mu_{p,r}\ll p^{-r}\quad(r\ge 5).
\]
For $p\in\{2,3\}$, the same geometric tails give $\mu_{p,r}\ll p^{-r}$ for $r\ge 5$, while $\mu_{p,2}$ and $\mu_{p,3}$ are absolute constants. Splitting into the four cases $(\ell,\ell')=(2,3)$, $(2,r)$ with $r\ge 5$, $(3,r)$ with $r\ge 5$, and $\ell,\ell'\ge 5$ distinct, we obtain respectively
\[
\sum_p\mu_{p,2}\mu_{p,3}\ll1,
\]
\[
\sum_p\mu_{p,2}\mu_{p,r}\ll 2^{-r}+3^{-r}+\sum_{p\ge5}p^{-r-2}\ll r^{-2},
\]
\[
\sum_p\mu_{p,3}\mu_{p,r}\ll 2^{-r}+3^{-r}+\sum_{p\ge5}p^{-r-3}\ll r^{-2},
\]
and
\[
\sum_p\mu_{p,\ell}\mu_{p,\ell'}
\ll
2^{-\ell-\ell'}+3^{-\ell-\ell'}+\sum_{p\ge5}p^{-\ell-\ell'}
\ll
(\ell\ell')^{-2}.
\]
After increasing the implied constant to cover the fixed pair $(2,3)$, this proves
part (3).

For $\ell=2$ and $\ell'=3$, direct evaluation of the absolutely convergent series
in part (1), using the explicit formulas of Theorem~\ref{thm:GOT-local} and the
tail bound $\sum_{p>Y}|C_p(2,3)|\ll\sum_{p>Y}p^{-5}$, gives
\[
\mathbf{Cov}(v_2(\mathrm{Tam}(E)),v_3(\mathrm{Tam}(E)))=-0.0248897\ldots,
\]
which is the quoted numerical value.

Set $\Lambda=\varnothing$, $\Pi=\varnothing$, $s=0$ in Theorem~\ref{thm:master} while keeping $u$ and $w$ free. The resulting joint probability generating function of $(\omega_{\mathrm{Tam}}(E),\Omega(\mathrm{Tam}(E)))$ is
\[
G(u,w)
=
\prod_p g_p(u,w),
\qquad
g_p(u,w):=\delta_p(1)+u\sum_{t\ge 2}\delta_p(t)w^{\Omega(t)}.
\]
Each $g_p$ is a bivariate power series with nonnegative coefficients summing to $1$, so $|g_p|\le 1$ on the closed unit bidisc. To justify differentiation in a neighborhood of $(1,1)$, put
\[
a_p:=1-\delta_p(1),\qquad
T_p:=\sum_{t\ge2}\delta_p(t)t^3.
\]
The geometric tails in Theorem~\ref{thm:GOT-local} show that $T_p<\infty$ for every $p$, and Lemma~\ref{lem:local-estimates} gives $T_p\ll p^{-2}$ for $p\ge5$. On the bidisc $|u-1|,|w-1|\le1$, we have $|u|,|w|\le2$ and $2^{\Omega(t)}\le t$. Consequently,
\[
|g_p(u,w)-1|
\le a_p+2\sum_{t\ge2}\delta_p(t)t
\le3T_p.
\]
Moreover, since $\Omega(t)\le t$, the same majorant gives
\[
|\partial_u g_p|+|\partial_w g_p|
+|\partial_u\partial_w g_p|+|\partial_w^2g_p|
\ll T_p
\]
uniformly on this bidisc. The local series and their displayed derivatives converge uniformly there, and their sums over $p$ converge uniformly as well (using $g_p-1$ for the zeroth derivative). For all sufficiently large $p$, the bound $|g_p-1|\le3T_p$ is at most $1/2$ throughout the bidisc. For the remaining finitely many primes, shrink to a common neighborhood of $(1,1)$ on which $|g_p-1|<1/2$, using $g_p(1,1)=1$. The logarithms normalized by $\log g_p(1,1)=0$ and their first and second derivatives then form uniformly convergent series on a smaller closed bidisc. This justifies termwise logarithmic differentiation of $\log G(u,w)=\sum_p\log g_p(u,w)$ at $(1,1)$.

Using $g_p(1,1)=1$, $\partial_u g_p(1,1)=\sum_{t\ge 2}\delta_p(t)=1-\delta_p(1)$, $\partial_w g_p(1,1)=\sum_{t\ge 1}\delta_p(t)\Omega(t)$, and $\partial_u\partial_w g_p(1,1)=\sum_{t\ge 1}\delta_p(t)\Omega(t)$ (since $\Omega(1)=0$), the joint-p.g.f.~identity gives
\begin{align*}
\mathbf{Cov}(\omega_{\mathrm{Tam}}(E),\Omega(\mathrm{Tam}(E)))
&=
\sum_p\Bigl(\partial_u\partial_w g_p-\partial_u g_p\cdot\partial_w g_p\Bigr)_{(1,1)}\\
&=
\sum_p\Bigl(\sum_{t\ge 1}\delta_p(t)\Omega(t)-(1-\delta_p(1))\sum_{t\ge 1}\delta_p(t)\Omega(t)\Bigr)\\
&=
\sum_p\delta_p(1)\sum_{t\ge 1}\delta_p(t)\Omega(t),
\end{align*}
which is \eqref{eq:cov-omega-Omega}. Absolute convergence follows from Lemma~\ref{lem:local-estimates}, since $\sum_{t\ge 1}\delta_p(t)\Omega(t)\ll p^{-2}$ and $\delta_p(1)\le 1$.

The variance of $\omega_{\mathrm{Tam}}(E)$ is given by Theorem~\ref{thm:omegaE}. Applying the same logarithmic differentiation argument to $G(1,w)$ gives
\[
\mathbf{Var}(\Omega(\mathrm{Tam}(E)))
=
\sum_p\Bigl(\sum_{t\ge 1}\delta_p(t)\Omega(t)^2-\Bigl(\sum_{t\ge 1}\delta_p(t)\Omega(t)\Bigr)^{\!2}\Bigr).
\]
This series converges absolutely by Lemma~\ref{lem:local-estimates}. Both global variances are strictly positive: at $p=2$, the local outcomes $t=1$ and $t=2$ have positive probability and give different values of each local statistic. Dividing the covariance by the square root of the product of the variances proves \eqref{eq:corr-omega-Omega}. Truncating these three convergent sums at $p\le 10^8$ (cf.\ Table~\ref{tab:cutoff-stability}) yields the numerical value $\mathbf{Corr}\approx 0.9801$.
