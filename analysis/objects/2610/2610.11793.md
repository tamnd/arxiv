<!-- theorem-1: body_md -->
\label{thm:main}
The set $\mathfrak K$ is closed, connected, and completely invariant under
$F$, and $\pi|_{\mathfrak K}:\mathfrak K\to\mathbb C$ is proper. The set $\mathfrak K_{\mathcal M}$ is
compact and connected.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:stable-main}
Let $(c,z)\in\mathfrak K$. If $c\notin\mathcal M$, then $\mathfrak K$ is not locally
connected at $(c,z)$. If $c\in\mathcal M$ and $f_c$ is hyperbolic, then
$\mathfrak K$ is locally connected at $(c,z)$.
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm:puzzle}
Let $(c_0,z_0)$ be a non-degenerate pair. Then:
\begin{enumerate}
\item[\textup{(i)}] Mixed shrinking is equivalent to simultaneous
parameter and dynamical shrinking:
\[
 \mathfrak I(c_0,z_0)=\{(c_0,z_0)\}
 \quad\Longleftrightarrow\quad
 \left\{\begin{aligned}
  \mathcal I(c_0)&=\{c_0\},\\
  I_{c_0}(z_0)&=\{z_0\}.
 \end{aligned}\right.
\]
\item[\textup{(ii)}] Under these equivalent conditions, $\mathfrak K_{\mathcal M}$ is
locally connected at $(c_0,z_0)$.
\end{enumerate}

<!-- lemma-1: body_md -->
\label{lem:escape}
For every $A\ge0$ there exists $R>0$ such that, for all $|c|\le A$,
\[
 K_c\subset\overline{D(0,R)},\qquad
 K_c=\bigcap_{k\ge0} f_c^{-k}\bigl(\overline{D(0,R)}\bigr).
\]
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Choose $R>2$ with $R^2-A>2R$. For $|c|\le A$ and $|w|\ge R$,
\[
 |f_c(w)|\ge |w|^2-A>2|w|.
\]
Consequently, $|f_c^k(z)|\ge R$ for some $k$ implies $z\notin K_c$.
Conversely, $|f_c^k(z)|\le R$ for every $k\ge0$ implies $z\in K_c$.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:monotone}
Let $p:X\to B$ be a continuous surjection, where $X$ is compact and
$B$ is connected and Hausdorff. If $p^{-1}(\{b\})$ is connected
for every $b\in B$, then $X$ is connected.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Suppose $X=A\sqcup D$ is a separation into nonempty open sets.
Both $A$ and $D$ are compact. Connectedness of each fiber gives
\[
 p(A)\cap p(D)=\varnothing,\qquad p(A)\cup p(D)=B.
\]
Since $B$ is Hausdorff, $p(A)$ and $p(D)$ are nonempty closed sets.
They therefore form a separation of $B$, a contradiction.

<!-- proof-3: body_md -->
The complete invariance of $K_c$ under $f_c$ gives $F^{-1}(\mathfrak K)=\mathfrak K=F(\mathfrak K)$.

Let $(c_j,z_j)\in\mathfrak K$ with $(c_j,z_j)\to(c,z)$. Choose $A>0$ such that $|c|\leq A$ and $|c_j|\leq A$ for every $j$.
Let $R$ be given by Lemma~\ref{lem:escape}.
Continuity of $(c,z)\mapsto f_c^k(z)$ yields
\[
 |f_c^k(z)|=\lim_{j\to\infty}|f_{c_j}^k(z_j)|\le R
 \qquad(k\ge0).
\]
Thus $z\in K_c$, so $\mathfrak K$ is closed. 

If $B\subset\mathbb C$ is compact,
the same lemma gives a uniform $R$ such that
\[
 (\pi|_{\mathfrak K})^{-1}(B)=\mathfrak K_B
 \subset B\times\overline{D(0,R)}.
\]
The left-hand side is closed in the compact set on the right, proving
properness. Every $f_c$ has a fixed point in $K_c$, so
$\pi(\mathfrak K)=\mathbb C$.

Since $\pi|_{\mathfrak K}$ is proper and $\mathcal M$ is compact,
$
 \mathfrak K_{\mathcal M}=(\pi|_{\mathfrak K})^{-1}(\mathcal M)
$
is compact. The continuous surjection
$
 p=\pi|_{\mathfrak K_{\mathcal M}}:\mathfrak K_{\mathcal M}\to\mathcal M
$
has connected fibers $p^{-1}(\{c\})=\{c\}\times K_c$ for all
$c\in\mathcal M$. Lemma~\ref{lem:monotone} therefore implies that $\mathfrak K_{\mathcal M}$ is connected.

To prove that $\mathfrak K$ is connected, we attach each $\mathbf{x}\in\mathfrak K\setminus\mathfrak K_{\mathcal M}$ to $\mathfrak K_{\mathcal M}$ by a continuum. 
Write $\mathbf{x}=(c_0,z_0)$, choose $a\in\mathcal M$ with
$|c_0-a|=\operatorname{dist}(c_0,\mathcal M)=d>0$, and set
\[
 \gamma:[0,1]\to\mathbb C,\qquad
 \gamma(t)=(1-t)c_0+ta.
\]
Thus
$
 \gamma([0,1))\subset\mathbb C\setminus\mathcal M$ and $\gamma(1)=a\in\mathcal M.
$

For each $b\in\gamma([0,1))$, hyperbolic $J$-stability
\cite{MSS} gives a disk
$\mathcal U_b\Subset\mathbb C\setminus\mathcal M$ containing $b$ and a jointly
continuous holomorphic motion
\[
 h_b:\mathcal U_b\times J_b\to\mathbb C,
 \qquad h_b(b,\zeta)=\zeta,
\]
whose slices $h_b(c,\cdot):J_b\to J_c$ are homeomorphisms.
Whenever $\gamma([u,v])\subset\mathcal U_b$ and
$w\in J_{\gamma(u)}$, define
\begin{equation}\label{eq:local-transport}
 z(t)=h_b\!\left(\gamma(t),
    h_b(\gamma(u),\cdot)^{-1}(w)\right),\qquad t \in [u,v].
\end{equation}
Then $z(u)=w$ and $z(t)\in J_{\gamma(t)}$, and $z$ is continuous.

Set $t_k=1-2^{-k}$ for $k\ge0$. Each interval
$[t_k,t_{k+1}]$ admits a finite partition
$t_k=s_0<\cdots<s_\ell=t_{k+1}$ such that
$\gamma([s_{i-1},s_i])\subset\mathcal U_{b_i}$.
Starting from $z(0)=z_0\in K_{c_0}=J_{c_0}$, apply
\eqref{eq:local-transport} successively on these subintervals, with
$w=z(s_{i-1})$. The formulas agree at their endpoints and define
a continuous lift
\[
 L:[0,1)\to\mathfrak K,\qquad
 L(t)=(\gamma(t),z(t)),\qquad L(0)=\mathbf{x}.
\]
By properness,
$
 \mathfrak C:=\overline{L([0,1))}
 \subset\mathfrak K_{\gamma([0,1])}
$
is compact. It is connected because $L([0,1))$ is connected.
A subsequence of $L(t_k)$ converges to some $(a,w)\in\mathfrak C$;
since $a\in\mathcal M$, this limit lies in $\mathfrak C\cap\mathfrak K_{\mathcal M}$.

Only a cluster point of $L(t)$ as $t\to1$ is needed. Therefore
\[
 \mathfrak K=\mathfrak K_{\mathcal M}\cup
 \left(\bigcup_{\mathbf{x}\in\mathfrak K\setminus\mathfrak K_{\mathcal M}}\mathfrak C_{\mathbf{x}} \right),
 \qquad \mathfrak C_{\mathbf{x}}\cap\mathfrak K_{\mathcal M}\ne\varnothing,
\]
where each $\mathfrak C_{\mathbf{x}}$ is connected. Thus the union is connected.

<!-- proposition-1: body_md -->
\label{prop:stable-product}
If $f_{c_0}$ is hyperbolic, there exist a disk $\mathcal U\ni c_0$
and a homeomorphism
\[
 T:\mathcal U\times K_{c_0}\to\mathfrak K_{\mathcal U}
\]
such that $\pi(T(c,z))=c$ and $T(c_0,z)=(c_0,z)$ for all
$(c,z)\in\mathcal U\times K_{c_0}$.
<!-- end proposition-1 -->
<!-- proof-4: body_md -->
By $J$-stability \cite{MSS}, the Julia sets move holomorphically over
a sufficiently small disk $\mathcal U\ni c_0$. Adjoin the fixed
point $\infty$ and apply S\l{}odkowski's extension theorem
\cite{Slodkowski} (see also \cite[Theorem~3.1]{Roesch}).
The $\lambda$-lemma \cite[p.~193]{MSS} gives joint continuity
of the resulting holomorphic motion
\[
 h:\mathcal U\times\widehat\mathbb C\to\widehat\mathbb C,
 \qquad h^c(z):=h(c,z),
\]
normalized by $h^{c_0}=\operatorname{id}$, with each $h^c$ a sphere
homeomorphism fixing $\infty$ and satisfying $h^c(J_{c_0})=J_c$. 
It follows that $h^c(\mathbb C\setminus K_{c_0})=\mathbb C\setminus K_c$, and $ h^c(K_{c_0})=K_c$.
Thus $T:\mathcal U\times K_{c_0}\to\mathfrak K_{\mathcal U}$, defined by
$T(c,z)=(c,h^c(z))$, is a continuous bijection.
To verify continuity of $T^{-1}$, let $(c_j,w_j)\to(c,w)$ in
$\mathfrak K_{\mathcal U}$ and put $z_j=(h^{c_j})^{-1}(w_j)\in K_{c_0}$.
Every convergent subsequence $z_{j_k}\to z_*$ satisfies
\[
 h^c(z_*)=\lim_{k\to\infty}h^{c_{j_k}}(z_{j_k})=w,
\]
so $z_*=(h^c)^{-1}(w)$. Compactness of $K_{c_0}$ therefore gives
$z_j\to (h^c)^{-1}(w)$.

<!-- proof-5: body_md -->
For $c_0\notin\mathcal M$, choose the disk $\mathcal{U}$ in
Proposition~\ref{prop:stable-product} inside $\mathbb C\setminus\mathcal M$.
Let $\pi_2:\mathcal U\times K_{c_0}\to K_{c_0}$ be the
second-coordinate projection. If $\mathfrak V\subset\mathfrak K_{\mathcal U}$
were a connected neighborhood of $(c_0,z)$, then
$\pi_2(T^{-1}(\mathfrak V))$ would be a connected subset of the

\cite[Theorem~3.1 and Corollary~5.4]{MilnorPortrait}. Since
