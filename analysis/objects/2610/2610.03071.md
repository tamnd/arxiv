\label{prop:fk_feasibility}
Fix $\rho\in(0,1)$ and suppose that the Feynman--Kac path weight is nonnegative, integrable, and has a strictly positive normalizing constant, then $\mathbb{Q}_\theta^{\mathrm{FK}}$ is absolutely continuous with respect to $\mathbb{Q}_\theta(\cdot\mid\rho)$, i.e. $\mathbb{Q}_\theta^{\mathrm{FK}}(\cdot\mid\rho) \ll \mathbb{Q}_\theta(\cdot\mid\rho)$. Consequently, the corresponding terminal marginal satisfies
\begin{equation}
    \pi_{\theta,\beta_K}(\cdot\mid\rho)
    \ll
    p_\theta(\cdot\mid\rho).
    \label{eq:terminal_absolute_continuity}

\label{prop:backward_fk}
Suppose the function $h_t$ defined in equation~\ref{eq:app_h_definition} is sufficiently smooth and the standard integrability conditions required for It\^{o}'s formula hold. Then $h_t$ satisfies
\begin{equation}
\label{eq:app_backward_fk}
    \partial_t h_t(x)
    +
    \mathcal{L}_t h_t(x)
    -
    V_t(x)h_t(x)
    =
    0,
    \qquad
    h_T(x)=G(x).
\end{equation}
<!-- end proposition-3 -->
<!-- proof-1: body_md -->
Fix $t\leq u\leq T$ and define
\begin{equation*}
    Z_{t,u}
    \triangleq 
    \exp
    \left(
        -\int_t^u V_s(X_s)\,ds
    \right).
\end{equation*}
By the Markov property and the tower property,
\begin{equation}
\label{eq:app_dynamic_programming_h}
    h_t(x)
    =
    \mathbb{E}_{t,x}
    \left[
        Z_{t,u}h_u(X_u)
    \right].
\end{equation}
Applying It\^{o}'s formula to $Z_{t,u}h_u(X_u)$ yields
\begin{align*}
    d\!\left(
        Z_{t,u}h_u(X_u)
    \right)
    &=
    Z_{t,u}
    \left[
        \partial_u h_u
        +
        \mathcal{L}_u h_u
        -
        V_u h_u
    \right](X_u)\,du +
    Z_{t,u}
    \nabla h_u(X_u)^\top
    \sigma_u(X_u)\,dB_u .
\end{align*}
In fact, the process
\[
    M_u\triangleq Z_{t,u}h_u(X_u),
    \qquad u\in[t,T],
\]
is a martingale under the baseline law $\mathbb{Q}$. Indeed, for $t\leq s\leq u\leq T$, using the multiplicative identity $Z_{t,u}=Z_{t,s}Z_{s,u}$, the Markov property, and the definition of $h_s$, we obtain
\begin{align*}
    \mathbb{E}_{\mathbb Q}
    \left[
        M_u
        \mid
        \mathcal{F}_s
    \right]
    &=
    Z_{t,s}
    \mathbb{E}_{\mathbb Q}
    \left[
        Z_{s,u}h_u(X_u)
        \mid
        \mathcal{F}_s
    \right]=
    Z_{t,s}h_s(X_s)
    =
    M_s .
\end{align*}
Hence $\mathbb{E}_{t,x}[M_u]=M_t=h_t(x)$ is constant in $u$. The stochastic integral has zero conditional expectation, and hence the finite-variation term must vanish. Therefore,
\begin{equation*}
    \partial_u h_u
    +
    \mathcal{L}_u h_u
    -
    V_u h_u
    =
    0.
\end{equation*}
The terminal condition follows immediately from equation~\ref{eq:app_h_definition} by setting $u=T$.

<!-- remark-1: body_md -->
\label{rem:gradient_vs_fk}
The drift correction $a_t\nabla\log h_t$ is one realization of the transformed path law when $h_t$ is sufficiently regular; it is not the definition of the Feynman--Kac transform itself. The fundamental object is the change of measure in equation~\ref{eq:app_fk_path_measure}. For a prescribed marginal evolution, correction can in general be distributed differently among transport, diffusion, and reweighting terms \citep{skreta2025fkc}. This flexibility allows FK transforms to be implemented through weighted particle systems without requiring the transformed law to be represented solely through an additional gradient-based drift.

Fix a risk level $\rho$ and consider the path space $\Omega_K=\mathcal X^{K+1}$, $X_{0:K} =(X_0,\ldots,X_K)$. The path law of the frozen conditional reverse diffusion is
\begin{equation*}
    \mathbb{Q}_\theta(dx_{0:K}\mid\rho) = p_{\mathrm{ref}}(dx_0)
    \prod_{k=1}^{K} M_k^\theta(x_{k-1},dx_k\mid\rho).
    \label{eq:baseline_path_measure_feas}
\end{equation*}
Let $L_K(x_{0:K};\rho) \triangleq \prod_{k=1}^{K} G_k(x_{k-1},x_k;\rho) \geq 0$ denote the Feynman--Kac path weight, and assume that $0<Z_K(\rho)\triangleq\mathbb E_{Q_\theta(\cdot\mid\rho)}\left[L_K(X_{0:K};\rho)\right]<\infty.$ The normalized Feynman--Kac path measure is therefore
\begin{equation*}
    \mathbb{Q}_\theta^{\mathrm{FK}}
    (dx_{0:K}\mid\rho)
    =\frac{L_K(x_{0:K};\rho)}{Z_K(\rho)}
    \mathbb{Q}_\theta(dx_{0:K}\mid\rho) \Rightarrow \frac{
        d\mathbb{Q}_\theta^{\mathrm{FK}}(\cdot\mid\rho)
    }{
        d\mathbb{Q}_\theta(\cdot\mid\rho)
    }
    (X_{0:K})
    =
    \frac{
        L_K(X_{0:K};\rho)
    }{
        Z_K(\rho)
    }.
    \label{eq:fk_path_measure_feas}
\end{equation*}
Since the right-hand side is a well-defined nonnegative density,
\begin{equation}
    \mathbb{Q}_\theta^{\mathrm{FK}}(\cdot\mid\rho)
    \ll
    \mathbb{Q}_\theta(\cdot\mid\rho).
    \label{eq:fk_path_ac}
\end{equation}

We next transfer this absolute continuity from path space to the terminal marginal. Define the measurable terminal evaluation map $e_K:\Omega_K\rightarrow\mathcal X$, $e_K(x_{0:K})=x_K$. By definition,
\begin{equation}
    p_\theta(\cdot\mid\rho)
    =
    (e_K)_{\#}\mathbb{Q}_\theta(\cdot\mid\rho),
    \qquad
    \pi_{\theta,\beta_K}(\cdot\mid\rho)
    =
    (e_K)_{\#}\mathbb{Q}_\theta^{\mathrm{FK}}(\cdot\mid\rho).
    \label{eq:terminal_pushforward}
\end{equation}
where $(e_K)_\#$ denotes the pushforward under the terminal evaluation map $e_K(x_{0:K})=x_K$.

Let $A\subseteq\mathcal X$ be any measurable set satisfying $p_\theta(A\mid\rho)=0$. Using the first identity in equation~\ref{eq:terminal_pushforward}, $\mathbb{Q}_\theta\left(e_K^{-1}(A)\mid\rho\right)=0$. The path-space absolute continuity therefore implies $\mathbb{Q}_\theta^{\mathrm{FK}}\left(e_K^{-1}(A)\mid\rho\right)=0$. Applying the second identity in equation~\ref{eq:terminal_pushforward} gives $\pi_{\theta,\beta_K}(A\mid\rho)=0$. Since this holds for every measurable $A$ with $p_\theta(A\mid\rho)=0$, we conclude that
\begin{equation}
    \pi_{\theta,\beta_K}(\cdot\mid\rho)
    \ll
    p_\theta(\cdot\mid\rho).
    \label{eq:terminal_ac_proof}
\end{equation}

Finally, suppose that the frozen diffusion is chance-feasible almost surely, $p_\theta(\mathcal X_\rho\mid\rho)=1$. Equivalently, $p_\theta(\mathcal X_\rho^c\mid\rho)=0$. Taking $A=\mathcal X_\rho^c$ in equation~\ref{eq:terminal_ac_proof} yields $\pi_{\theta,\beta_K}(\mathcal X_\rho^c\mid\rho)=0$ and therefore
\begin{equation*}
    \pi_{\theta,\beta_K}
    (\mathcal X_\rho\mid\rho)=1.
\end{equation*}
Thus, any almost-sure chance feasibility encoded by the frozen diffusion prior is preserved under Feynman--Kac correction.
