<!-- question-1: body_md -->
\label{mainquest}
When can an equitable $(s_1, p_1)$-edge-coloring of $K_{n_1}$ be embedded in an equitable $(s_2, p_2)$-edge-coloring of $K_{n_2}$?
<!-- end question-1 -->
<!-- theorem-1: body_md -->
\cite{MR1271864, MR3774803, MR3475564} \label{mainlambda1} For $1 \leq p \leq n-1$,
$$
\chi'_p(K_n) =
\begin{cases} 
      p+1 & \text{ if } $p$ \text{ even and } n \equiv p+1 \ (\mathrm{mod}\ 2p)\\
      p & \text{ otherwise.}
   \end{cases}
$$

<!-- observation-1: body_md -->
\cite{MR3764373}
\label{bobservation}
Let $W$ be a $(2t+1,2t)$-edge-coloring of $K_n$ with the colors in $C$ for $n=4tx+2t+1$ for some integer $x$.
Then $b(n, W, u, i)= \frac{n-1}{2t}=2x+1$ for all $u \in V(K_n)$ and $i \in C$.

<!-- question-2: body_md -->
\label{mainquestbpreserved}
Let $1 \leq t_1 < t_2$ be integers and $b$ be an odd integer.
When can an equitable $(2t_1+1, 2t_1)$-edge-coloring of $K_{n_1}$ be embedded in an equitable $(2t_2+1, 2t_2)$-edge-coloring of $K_{n_2}$ where $n_1=2t_1b+1$ and $n_2=2t_2b+1$?
<!-- end question-2 -->
<!-- observation-2: body_md -->
\label{missingone}
In an equitable $(p+1,p)$-edge-coloring of a graph $G$, each $v \in V(G)$ is missing exactly one color.
<!-- end observation-2 -->
<!-- lemma-1: body_md -->
\label{colorvectorlemma} Let $n=2tb+1$ for some odd positive integer $b$ and some positive integer $t$, $p=2t$, and $s=2t+1$. In any equitable $(s,p)$-edge-coloring $W$ of $K_n$, for $1 \leq i \leq s$,
\begin{enumerate}[(i)]
\item $c_i(W)$ must be even,
\item $c_i(W) \geq b +1 = \frac{n-1}{p} +1$, and
\item $c_i(W) \leq n-1$.
\end{enumerate}
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{ineqlem}
    Let $W_1$ be an equitable $(s_1, p_1)$-edge-coloring of $K_{n_1}$. If we can embed $W_1$ into an equitable $(s_2, p_2)$-edge-coloring of $K_{n_2}$, $W_2$, then 
    $$
        n_1(m-1) \leq (n_2-n_1)m-1.
    $$
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
Define $E_{C_N}(V_O,V_N)$ to be the set of edges with one endpoint in $V_O$ and one endpoint in $V_N$ that are colored with a color in $C_N$.
    Recall, by Observation \ref{missingone}, that each vertex is missing exactly $1$ color.
    Let $N$ be the set of vertices in $V_O \cup V_N$ that are missing a color in $C_N$.

    Because each of the vertices in $N$ must be incident with $b$ edges of $m-1$ colors in $C_N$ and each of the vertices in $(V_O \cup V_N) \setminus N$ must be incident with $b$ edges of $m$ colors in $C_N$,
    \begin{equation}
        \label{oldvertineq}
    |E_{C_N}(V_O,V_N)|=|V_O \cap N|(m-1)b + |V_O \setminus N|mb
    \end{equation}
    and
    \begin{equation}
        \label{newvertineq}
        |E_{C_N}(V_O,V_N)| \leq |V_N \cap N|(m-1)b + |V_N \setminus N|mb.
    \end{equation}
    Note that the first equation is an equality because there are no colors from $C_N$ on the edge $\{v,u\}$ for all $v,u \in V_O$, but the second equation is an inequality because there may be colors from $C_N$ on the edge $\{v,u\}$ for some $v,u \in V_N$.
    Immediately, we can conclude from the previous two equations that $n_1(m-1)b \leq |E_{C_N}(V_O,V_N)| \leq (n_2 - n_1)mb$.
    However, this inequality can be improved.

    
    By Lemma \ref{colorvectorlemma} (i), $c_i(W_2)$ must be even for all $1 \leq i \leq s_2$. Also, $n_2$ is odd. So, the number of vertices missing each color $i \in C_N$ is odd.
    Additionally, $|C_N|=m$ is even.
    Therefore, $\vert N \vert$ is even.
    Consider $2$ cases.
    
    \noindent {\bf Case 1:} 
    Assume $|V_O \cap N|$ is odd.
    Therefore, $|V_N \cap N| \geq 1$.
    So, by (\ref{oldvertineq}) and (\ref{newvertineq}), $$n_1(m-1)b \leq |E_{C_N}(V_O,V_N)| \leq (1)(m-1)b+(n_2-n_1-1)mb.$$

    \noindent {\bf Case 2:} 
    Assume $\vert V_O \cap N \vert$ is even. 
    Because $\vert V_O \vert$ is odd, there must be at least one vertex in $V_O$ missing a color in $C_O$.
    Therefore, $|V_O \setminus N| \geq 1$.
     So, by (\ref{oldvertineq}) and (\ref{newvertineq}), $$(n_1-1)(m-1)b + (1)mb \leq |E_{C_N}(V_O,V_N)| \leq (n_2-n_1)mb.$$
     
     Thus, in either case, $n_1(m-1) \leq (n_2-n_1)(m)-1.$
<!-- end proof-1 -->
<!-- corollary-1: body_md -->
\label{equitableembedinequality}
Let $W_1$ be an equitable $(2t_1+1, 2t_1)$-edge-coloring of $K_{n_1}$. If we can embed $W_1$ into an equitable $(2t_2+1, 2t_2)$-edge-coloring of $K_{n_2}$ called $W_2$, then $t_2 \geq 2t_1$.
<!-- end corollary-1 -->
<!-- proof-2: body_md -->
By Lemma \ref{ineqlem}, $n_1(m-1)\leq (n_2-n_1)(m)-1$, so
$$
\begin{aligned}
0  & \leq (n_2-n_1)(m)-n_1m+n_1-1\\
  & = bm^2 -(2t_1b+1)m+2t_1b.\\
\end{aligned}
$$
Suppose for a contradiction that $t_2 \leq 2t_1-1$.
Because $t_2 > t_1$, $t_1 \geq 2$.
Recall, $m=s_2-s_1=2t_2-2t_1$.
Therefore, $m \leq 2t_1-2$.
Thus,
$$
\begin{aligned}
0  & \leq b(2t_1-2)^2 -(2t_1b+1)(2t_1-2)+2t_1b\\
& =4t_1^2b-8t_1b+4b-4t_1^2b+4t_1b-2t_1+2+2t_1b\\
& = -2t_1b+4b-2t_1+2\\
& < (-2t_1+4)(b+1).\\
\end{aligned}
$$
Because $t_1 \geq 2$ and $b \geq 1$, this is a contradiction.
Thus, $t_2 \geq 2t_1$.

\label{equitableembedinequalitywithn}
Let $W_1$ be an equitable $(2t_1+1, 2t_1)$-edge-coloring of $K_{n_1}$. If we can embed $W_1$ into an equitable $(2t_2+1, 2t_2)$-edge-coloring of $K_{n_2}$ called $W_2$, then $n_2 \geq 2n_1-1$.

<!-- lemma-3: body_md -->
\label{thm:b(n)=1}
If $b=1$, an equitable $(2t_1+1, 2t_1)$-edge-coloring of $K_{n_1}$
can be embedded in an equitable $(2t_2+1, 2t_2)$-edge-coloring of $K_{n_2}$ if and only if $t_1 \leq 2t_2$.

<!-- theorem-2: body_md -->
\label{amalgamationtheorem} Let $G'$ be a $k$-edge-colored graph and let $\eta$ be a function from $V(G')$ into $\mathbb{N}$ such that for each $w \in V(G')$, $\eta(w)=1$ implies $\ell_{G'}(w)=0$.
Then there exists a loopless $\eta$-detachment $G$ of $G'$ with amalgamation function $\phi: V(G) \rightarrow V(G')$, $\eta$ being the number function associated with $\phi$, such that $G$ satisfies the following conditions:
\begin{enumerate}[(i)]
    \item $deg_{G(j)}(u) \approx deg_{G'(j)}(w) / \eta(w)$ for each $w \in V(G')$, each $u \in \phi^{-1}(w)$, and for $1 \leq j \leq k$;
    \item $m_G(u, u') \approx \ell_{G'}(w)/ {\eta(w) \choose 2}$ for each $w \in V(G')$ with $\eta(w) \geq 2$ and every pair of distinct vertices $u,u' \in \phi^{-1}(w)$; and
    \item $m_G(u,v) \approx m_{G'}(w,z) / (\eta(w) \eta(z))$ for every pair of distinct vertices $w,z \in V(G')$, each $u \in \phi^{-1}(w)$, and each $v \in \phi^{-1}(z)$.
\end{enumerate}

<!-- lemma-4: body_md -->
\label{obs:Old-color-method-theorem}
Let $t_2 = 2t_1$.
An equitable $(2t_1+1, 2t_1)$-edge-coloring of $K_{n_1}$ can be embedded in an equitable $(2t_2+1, 2t_2)$-edge-coloring of $K_{n_2}$.
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
Let $X = \{x_1, x_2, \dots x_{n_1}\}$.
We will form edge-colored graphs $G_1, G_2, G_3$ on the vertex set $X \cup \{y\}$ as follows.
Once this is done, defining $G'= \cup_{i=1}^3 G_i$, Theorem \ref{amalgamationtheorem} will be used to detach $G'$ to form $G= K_{n_2}$ with our desired equitable $(2t_2+1,2t_2)$-edge-coloring and an arbitrary $(2t_1+1, 2t_1)$-edge-coloring of $K_{n_1}$ as a subgraph.
To do so, $\phi: V(G) \rightarrow V(G')$ will be the amalgamation function with amalgamation numbers $\eta(x_i)= 1$ for $1 \leq i \leq n_1$ and $\eta(y)= n_2 - n_1$.

\begin{enumerate}[i)]
\item Let $G_1=K_{n_1}$ on the vertex set $X$ with any equitable $(2t_1+1, 2t_1)$-edge-coloring, call it $W_1$, using the colors in $C_O$ (such a coloring exists by Theorem \ref{mainlambda1}).
\item Let $S_w$ be the set of vertices in $X$ missing color $w$ in $W_1$ for $1 \leq w \leq 2t_1+1$.
Note that $\vert S_w \vert$ is odd and $\vert S_w \vert \geq 1$ by Lemma \ref{colorvectorlemma} (i) and (iii).
We will form sets $S_w'$ for $1 \leq w \leq 2t_2+1$ by modifying each set $S_w$. Without loss of generality, assume $|S_1|\leq|S_w|$ for $2\leq w\leq 2t_1+1$.
Define $S_1' \coloneqq S_1$. Remove $2t_1$ vertices from some sets $S_w$ for $2 \leq w \leq 2t_1+1$ keeping each $S_w$ odd (repeatedly remove pairs).
Call these new sets (after the vertices have been removed) $S'_w$.
This is possible because

\[
\begin{aligned}
n_1- (2t_1) &= 2t_1b+1-2t_1\\
 &= 2t_1(b-1) +1\\
 &\geq 2t_1+1 &\text{Because $b\geq 3$.}
\end{aligned}
\]
Therefore, $\vert S_w' \vert \geq 1$ for $1 \leq w \leq 2t_1+1$.
Greedily place the removed vertices in singleton sets $S'_w$ for $2t_1+1 \leq w \leq 2t_2+1 = 4t_1+1$.
Let $G_2 = (n_2-n_1)K_{n_1, 1}$ with parts $X$ and $\{y\}$ by attaching each vertex $x \in S_w \cap S'_c$ for $1 \leq w \leq 2t_1+1$ and $1 \leq c \leq 2t_2+1$ to $y$ with $b$ edges colored $k$ for  
$k \in (C_N \cup \{w\}) \setminus \{c\}$.
\item Form $G_3$ by placing $n_2-n_1 \choose 2$ loops on a single vertex, $y$.
For each $k \in C_O \setminus \{1\}$, color $\frac{(n_2-n_1)b-\vert S_k \setminus S'_k \vert b}{2}$ loops with color $k$.
This is well-defined because
\begin{equation*}
\begin{array}{rcl}
\ell_{G'}(y) & = & \sum_{k=2}^{2t_1+1} \frac{(n_2-n_1)b-\vert S_k\setminus S_{k}' \vert b}{2} \\[4pt]
                                           
& = & \sum_{k=2}^{2t_1+1} \frac{((2t_{2}b+1)-(2t_{1}b+1))b-\vert S_k\setminus S_{k}' \vert b}{2}\\[4pt]
                                                        
                                                                                                 
& = & \sum_{k=2}^{2t_1+1} \frac{2t_1b^2}{2}-\sum_{k=2}^{2t_1+1} \frac{\vert S_k\setminus S_{k}' \vert b}{2}\\[4pt]
& = & \frac{
(2t_1b^2)(2t_1)}{2} - \frac{2t_1b}{2}\\[4pt]
                                                 
                                                            
                              
                                            
                              
                            
& = & \frac{2t_1b(2t_1b-1)}{2}\\[4pt]
& = & \frac{(2t_2-2t_1)(b)((2t_2-2t_1)b-1)}{2} \\[4pt]
& = & \frac{((2t_2b+1)-(2t_1b+1))((2t_2b+1)-(2t_1b+1)-1)}{2}\\[4pt]
& = & \frac{(n_2-n_1)(n_2-n_1-1)}{2} \\[4pt]
& = & {n_2-n_1 \choose 2}. \\[4pt]
                                  
\end{array}
\end{equation*}
\end{enumerate}

As stated above, define $G'=\cup_{i=1}^3 G_i$.
We will use Theorem \ref{amalgamationtheorem} to form the detachment $G= K_{n_2}$ with our desired equitable $(2t_2+1,2t_2)$-edge-coloring and an arbitrary $(2t_1+1, 2t_1)$-edge-coloring of $K_{n_1}$ as a subgraph.
Recall that $\phi: V(G) \rightarrow V(G')$ is the amalgamation function with amalgamation numbers $\eta(x_i)= 1$ for $1 \leq i \leq n_1$ and $\eta(y)= n_2 - n_1$.

We first check that there are $b$ edges of each color incident to each vertex $v_i \in V_O$ except exactly $1$ color. Suppose $x_i \in S'_{c}$, then $v_i$ is missing color $c$.

\begin{itemize}
\item For $k \in C_O \setminus \{c\}$ and $1 \leq i \leq n_1$, by (i) or (ii), $deg_{G'(k)}(x_i) = b$.
By Theorem \ref{amalgamationtheorem} (i), $deg_{G(k)}(v_i) = deg_{G'(k)}(x_i)/\eta(x_i) = b/1 = b$ for $v_i \in \phi^{-1} (x_i)$.
\item For $k \in C_N \setminus \{c\}$ and $1 \leq i \leq n_1$, by (ii), $deg_{G'(k)}(x_i) = b$.
By Theorem \ref{amalgamationtheorem} (i), $deg_{G(k)}(v_i) = deg_{G'(k)}(x_i)/\eta(x_i) = b/1 = b$ for $v_i \in \phi^{-1} (x_i)$.
\end{itemize}

We will now check that each vertex detached from $y$ is incident to $b$ edges of each color $k\in (C_O\cup C_N) \setminus \{1\}$.
\begin{itemize}
\item For $2t_1 + 2 \leq k \leq 2t_2 + 1$, by (ii),
\[
\begin{aligned}
deg_{G'(k)}(y) &= (n_1 - 1)b \\
&= 2t_1 b^2 \\
&= (4t_1 b + 1 - (2t_1 b + 1))b \\
&= (2t_2 b + 1 - (2t_1 b + 1))b \\
&= (n_2 - n_1)b.
\end{aligned}
\]
So $deg_{G(k)}(v_i) = deg_{G'(k)}(y)/\eta(y) = (n_2-n_1)b/(n_2-n_1)=b$ for $v_i \in \phi^{-1} (y)$ by Theorem \ref{amalgamationtheorem} (i).
\item For $2 \leq k \leq 2t_1 + 1$, by (ii) and (iii),
\[
\begin{aligned}
deg_{G'(k)}(y) &= |S_k \setminus S_{k}'|b + 2 \left(\frac{(n_2 - n_1)b - (|S_k \setminus S_{k}'|b)}{2}\right) \\
&= (n_2-n_1)b.
\end{aligned}
\]
By Theorem \ref{amalgamationtheorem} (i), $deg_{G(k)}(v_i) = deg_{G'(k)}(y)/\eta(y) = (n_2-n_1)b/(n_2-n_1)=b$ for $v_i \in \phi^{-1} (y)$.
\end{itemize}

We now check that $G=K_{n_2}$, by determining there is exactly $1$ edge between each pair of vertices in $V(G)$.
\begin{itemize}
\item We first show there is exactly $1$ edge between each pair of vertices $v_i,v_j \in V_O$.
For $x_i, x_j \in X$, by (i), $m_{G'}(x_i, x_j) =1$.
By Theorem \ref{amalgamationtheorem} (iii), $m_G(v_i, v_j) = m_{G'}(x_i, x_j)/(\eta(x_i)\eta(x_j)) = 1/ ((1)(1)) = 1$ for $v_i \in \phi^{-1} (x_i)$ and $v_j \in \phi^{-1} (x_j)$.
\item We now show there is exactly $1$ edge between each pair of vertices $v_i,v_j \in V_N$.
By (iii), $\ell_{G'}(y) = {n_2-n_1 \choose 2}={\eta(y) \choose 2}$.
By Theorem \ref{amalgamationtheorem} $(ii)$, $m_G(v_i, v_j) = \ell_{G'}(y)/ {\eta(y) \choose 2}= {\eta(y) \choose 2}/ {\eta(y) \choose 2}=1$.
                                                                                                                                                                                                                                                                        
                                                                                                                                            
\item Finally, we show there is exactly $1$ edge between each pair of vertices $v_i \in V_O$ and $v_j \in V_N$.
By (ii), for each $x_i \in X$, $ m_{G'}(x_i, y)=b \vert (C_N \cup \{w\}) \setminus \{c\}\vert = b(2t_1)$ for some $1 \leq w \leq 2t_1+1$ and some $1 \leq c \leq 2t_2+1$.
So,
\[
\begin{aligned}
m_G(v_i, v_j) &= \frac{m_{G'}(x_i, y)}{(\eta(x_i)\eta(y))} \\[3pt]
&= \frac{2bt_1}{(1)(n_2-n_1)} \\[3pt]
&= \frac{2bt_1}{(2t_2b+1)-(2t_1b+1)}\\[3pt]
&= \frac{2bt_1}{(4t_1b+1)-(2t_1b+1)} \\[3pt]
&= \frac{2bt_1}{2bt_1}\\[3pt]
&= 1 
\end{aligned}
\]
for $v_i \in \phi^{-1}(x_i)$ and $v_j \in \phi^{-1}(y)$ by Theorem $\ref{amalgamationtheorem}$ (iii).
\end{itemize}

<!-- lemma-7: body_md -->
\label{obs:New-color-method-theorem}
Let $2t_1 <t_2 < 4t_1$.
An equitable $(2t_1+1, 2t_1)$-edge-coloring of $K_{n_1}$ can be embedded in an equitable $(2t_2+1, 2t_2)$-edge-coloring of $K_{n_2}$.
<!-- end lemma-7 -->
<!-- proof-5: body_md -->
Recall that $m= \vert C_N \vert = 2t_2-2t_1$.
Let $X = \{x_1, x_2, \dots x_{n_1}\}$ and $Y= \{y_{0}, y_{1}, \dots y_{m-1}\}$.
We will form edge-colored graphs $G_1, G_2, \dots G_6$ on the vertex set $X \cup Y$ as follows.
Once this is done defining $G'= \cup_{i=1}^6 G_i$, Theorem \ref{amalgamationtheorem} will be used to detach $G'$ to form $G= K_{n_2}$ with our desired equitable $(2t_2+1,2t_2)$-edge-coloring and an arbitrary $(2t_1+1, 2t_1)$-edge-coloring of $K_{n_1}$ as a subgraph.

\item For $k\in C_O\setminus \{c\}$, by (i), $deg_{G'(k)}(x_i)=b$. By Theorem \ref{amalgamationtheorem} (i), $deg_{G(k)}(v_i)=deg_{G'(k)}(x_i)/\eta(x_i)=b/1=b$ for $v_i\in \phi^{-1}(x_i)$.
\item For $k\in C_N$, by (ii), $deg_{G'(k)}(x_i)=b$. By Theorem \ref{amalgamationtheorem} (i), $deg_{G(k)}(v_i)=deg_{G'(k)}(x_i)/\eta(x_i)=b/1=b$ for $v_i\in \phi^{-1}(x_i)$.
\end{itemize}

We now check that each vertex $v_i\in V_N$ is incident to $b$ edges of each color $k$, except for exactly one color in $C_N$.

\begin{itemize}
\item For $k\in C_O$ and $y_p\in Y$, by (v) and (vi), $deg_{G'(k)}(y_p)=1+2\left(\frac{b^2-1}{2}\right)=b^2$. By Theorem \ref{amalgamationtheorem} (i), $deg_{G(k)}(v_i)=deg_{G'(k)}(y_p)/\eta(y_p)=b^2/b=b$ for $v_i\in \phi^{-1}(y_p)$.
\item For $k\in C_N\setminus \{p\}$ and $y_p\in Y$, by (ii), (iii), and (iv), $deg_{G'(k)}(y_p)=a+1+(b^2-(a+1))=b^2$. By Theorem \ref{amalgamationtheorem} (i), $deg_{G(k)}(v_i)=deg_{G'(k)}(y_p)/\eta(y_p)=b^2/b=b$ for $v_i\in \phi^{-1}(y_p)$.
\end{itemize}

We now check that $G=K_{n_2}$ by determining there is exactly one edge between each pair of vertices in $V(G)$.

\begin{itemize}
\item We first show there is exactly $1$ edge between each pair of vertices $v_i,v_j\in V_O$. For $x_i,x_j\in X$, by (i), $m_{G'}(x_i,x_j)=1$. By Theorem \ref{amalgamationtheorem} (iii), $m_G(v_i,v_j)=m_{G'}(x_i,x_j)/(\eta(x_i)\eta(x_j))=1/((1)(1))=1$ for $v_i\in \phi^{-1}(x_i)$ and $v_j\in \phi^{-1}(x_j)$.
\item We now show there is exactly $1$ edge between each pair of vertices $v_i\in V_O$ and $v_j\in V_N$. For $x_i\in X$ and $y_p\in Y$, by (ii), $m_{G'}(x_i,y_p)=b$. By Theorem \ref{amalgamationtheorem} (iii), $m_G(v_i,v_j)=m_{G'}(x_i,y_p)/(\eta(x_i)\eta(y_p))=b/((1)(b))=1$ for $v_i\in \phi^{-1}(x_i)$ and $v_j\in \phi^{-1}(y_p)$.
\item We now show there is exactly $1$ edge between each pair of vertices $v_i,v_j\in V_N$, where $v_i\in \phi^{-1}(y_p)$ and $v_j\in \phi^{-1}(y_s)$, and $y_p,y_s\in Y$. By (iii), (iv), (v), and (vi),
\[\begin{aligned}
m_{G'}(y_p,y_s) &= \sum_{i=3}^{6}m_{G_i}(y_p,y_s)\\
&= \sum_{i=3}^{5}m_{G_i}(y_p,y_s) + m_{G_6}(y_p,y_s)\\
&= \sum_{i=3}^{5}m_{G_i}(y_p,y_s) +b^2-\sum_{i=3}^{5}m_{G_i}(y_p,y_s))\\
&= b^2.
\end{aligned}\]
By Theorem \ref{amalgamationtheorem} (iii), $m_G(v_i,v_j)=m_{G'}(y_p,y_s)/(\eta(y_p)\eta(y_s))=b^2/((b)(b))=1$ for $v_i\in \phi^{-1}(y_p)$ and $v_j\in \phi^{-1}(y_s)$.
\item Finally, we show there is exactly $1$ edge between each pair of vertices $v_i,v_j\in V_N$, where $v_i,v_j\in \phi^{-1}(y_p)$ for $y_p \in Y$. By (vi), $\ell_{G'}(y_p)=\frac{b(b-1)}{2}={b \choose 2}={\eta(y_p) \choose 2}$. By Theorem \ref{amalgamationtheorem} (ii), $m_G(v_i,v_j)=\ell_{G'}(y_p)/{\eta(y_p)\choose 2}={\eta(y_p)\choose 2}/{\eta(y_p)\choose 2}=1$ for $v_i,v_j\in \phi^{-1}(y_p)$.
\end{itemize}

\label{mainthm}
An equitable $(2t_1+1, 2t_1)$-edge-coloring of $K_{n_1}$ can be embedded in an equitable $(2t_2+1, 2t_2)$-edge-coloring of $K_{n_2}$ if and only if $t_2 \geq 2t_1$.

By Corollary \ref{equitableembedinequality}, if we can embed an equitable $(2t_1+1, 2t_1)$-edge-coloring of $K_{n_1}$ into an equitable $(2t_2+1, 2t_2)$-edge-coloring of $K_{n_2}$, then $t_2 \geq 2t_1$.

Suppose $t_2 \geq 2t_1$ and $b \geq 3$.
Use the following steps.

\begin{enumerate}
    \item If $t_2=2t_1$, use Lemma \ref{obs:Old-color-method-theorem}.
    \item If $2t_1< t_2 < 4t_1$, use Lemma \ref{obs:New-color-method-theorem}.
