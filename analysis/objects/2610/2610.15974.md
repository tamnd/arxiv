\label{lem:Rio61}
	Let $(Z_n)_{n\in\mathbb{N}}$ be a sequence of $\{0,1\}$-valued random variables, and define
	$S_n = \sum_{k=1}^n \bigl(Z_k - \mathbb{E}[Z_k]\bigr)$, $n \ge 1$.
	Then, for any integers $1 < q < n$ and any $\delta \ge \frac{7q}{2n}$, we have
	\[
	\mathbb{P}\!\left( \frac{1}{n} |S_n| \ge \delta \right)
	\le
	4 \exp\!\left( -\frac{\delta n}{7q}
	\log\!\left( 1 + \frac{(2/7)\delta n}{\,n+q\,} \right) \right)
	+ \frac{28}{\delta}\,\alpha^Z(q+1).
	\]
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Let $1< q<n$ be fixed, and define $Y_k := Z_k\mathds{1}_{\{k\le n\}}$, $k\in\mathbb{N}$.
	We then apply Theorem~6.1 of Rio~\cite{rio2013inequalities} to the sequence of random variables $(Y_k)_{k\in\mathbb{N}}$, whose mixing coefficients do not exceed those of $(Z_k)$; in Rio's notation, $M=1$ and $M(n)=\sum_k\|Y_k\|_\infty\le n$.

	Note that $\|Y_k\|_{\infty}\le 1$ for every $k\ge1$, and $Y_k=0$ for $k>n$.
	For $v_q:=q(n+q)$ we have
	\[
	v_q \ge \Big\lceil\frac{n}{q}\Big\rceil q^2
	\ge \sum_{i>0}\mathbb{E}\big[(Y_{iq-q+1}+\cdots+Y_{iq})^2\big].
	\]
	Hence, by Theorem~6.1 of Rio~\cite{rio2013inequalities}, for any $\lambda\ge q$,
	\[
	\mathbb{P}\!\left(\sup_{1\le k\le n}|S_k|\ge \tfrac{7}{2}\lambda\right)
	\le 4\exp\!\left(-\frac{\lambda}{2q}\log\!\Big(1+\frac{\lambda}{n+q}\Big)\right)
	+ \frac{8n}{\lambda}\,\alpha^Z(q+1),
	\]
	where we have taken into account that in~\cite{rio2013inequalities}
	the definition of the mixing coefficient differs from ours by a factor of~$2$.
	Substituting $\lambda_n = \tfrac{2}{7}\delta n$ and assuming $\delta \ge \tfrac{7q}{2n}$,
	we obtain the desired inequality, which completes the proof.

\label{lem:Hansen}
	Let $(X_n)_{n\in\mathbb{N}}$ be an $\mathbb{R}$-valued sequence of identically distributed variables satisfying $\mathbb{E}[X_0]=0$ and $\mathbb{E}|X_0|^r<\infty$ for some $r>1$, and define
	\[
	S_n = \sum_{k=1}^n X_k, \quad n\ge 1.
	\]
	Assume further that the strong mixing coefficients of $(X_n)_{n\in\mathbb{N}}$ satisfy $\alpha^X(n) = O(n^{-a})$ for some $a>0$, $a\neq 1$.
	Then there exists a constant $c>0$, depending only on $r$, $\mathbb{E}|X_0|^r$ and the sequence $(\alpha^X(n))_{n\in\mathbb{N}}$, such that for all $\delta>0$,
	\[
	\mathbb{P}\!\left(\frac{1}{n}|S_n|\ge \delta\right)
	\le c\,\delta^{-1}\,n^{-\frac{1}{2}\bigl(1-(1-a)_+\bigr)\left(1-\frac{1}{r}\right)}.
	\]
	Note that $1-(1-a)_+=\min(a,1)$. The same bound holds, with the same constant, if $S_n$ is replaced by $X_{m+1}+\cdots+X_{m+n}$ for any $m\in\mathbb{N}$, since the proof only uses the common marginal law and the mixing coefficients.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Following the truncation technique employed by Hansen~\cite{hansen2019weak},
let $n\ge 1$ and $\tau>0$ be arbitrary and fixed.
Define the truncated random variables
\[
X_k' = X_k\mathds{1}_{\{|X_k|< n^\tau\}} - \mathbb{E}[X_k\mathds{1}_{\{|X_k|< n^\tau\}}],
\]
and set $X_k'' = X_k - X_k'$, $k\in\mathbb{N}$.
By the triangle inequality, we arrive at
\begin{equation}\label{eq:decomp}
	\mathbb{E}\!\left[\frac{1}{n}|S_n|\right]
	\le \mathbb{E}\!\left[\frac{1}{n}\left|\sum_{k=1}^n X_k'\right|\right]
	+ \frac{1}{n}\sum_{k=1}^n \mathbb{E}|X_k''|,
\end{equation}
where the second term can be estimated further by using
\begin{align*}
	\mathbb{E}|X_k''|
	&\le 2\,\mathbb{E}[|X_k|\mathds{1}_{\{|X_k|\ge n^\tau\}}]
	\le \frac{2\,\mathbb{E}[|X_0|^r]}{n^{\tau(r-1)}}.
\end{align*}

Furthermore, $X_k'$ satisfies the bound $|X_k'|\le 2n^\tau$, and its mixing coefficients obey
$\alpha^{X'}(m)\le \alpha^{X}(m)$ for all $m\in\mathbb{N}$.
Using the well-known covariance inequality for bounded strongly mixing variables
(cf.\ Theorem~A.5 in~\cite{hall2014martingale} or the proof of Lemma~A.1 in~\cite{lovasCLT}), we obtain
\[
|\mathbb{E}[X_k' X_l']|
= |\operatorname{Cov}(X_k', X_l')|
\le 16\,n^{2\tau}\,\alpha^{X}(|k-l|),\quad 1\le k,l\le n.
\]

Thus, by Jensen's inequality, we can write
\[
\mathbb{E}^2\!\left|\sum_{k=1}^n X_k'\right|
\le \sum_{k,l=1}^n \mathbb{E}[X_k' X_l']
\le 16\,n^{2\tau}\sum_{k,l=1}^n \alpha^X(|k-l|)
\le 32\,n^{2\tau+1}\sum_{k=0}^{n-1}\alpha^X(k).
\]
Since $\alpha^X(n)=O(n^{-a})$, for $a\neq 1$ there exists a constant $c'>0$ such that
\[
\sum_{k=0}^{n-1}\alpha^X(k)\le c'\,n^{(1-a)_+}, \qquad n\ge 1.
\]
Substituting this estimate into~\eqref{eq:decomp}, we obtain
\begin{equation}\label{eq:final-bound}
	\mathbb{E}\!\left[\frac{1}{n}|S_n|\right]
	\le \sqrt{32c'}\,n^{\,\tau-\frac{1}{2}+\frac{(1-a)_+}{2}}
	+ 2\,\mathbb{E}[|X_0|^r]\,n^{-\tau(r-1)}.
\end{equation}
Finally, for
$\tau=\frac{1}{2r}\bigl(1-(1-a)_+\bigr),$
both exponents in~\eqref{eq:final-bound} are equal to $-\frac{1}{2}\bigl(1-(1-a)_+\bigr)\left(1-\frac{1}{r}\right)$, and we arrive at
\[
\mathbb{E}\!\left[\frac{1}{n}|S_n|\right]
\le \bigl(\sqrt{32c'}+2\,\mathbb{E}[|X_0|^r]\bigr)
n^{-\frac{1}{2}\bigl(1-(1-a)_+\bigr)\left(1-\frac{1}{r}\right)},
\]
from which the claimed inequality follows immediately by Markov's inequality.

<!-- theorem-1: body_md -->
\label{thm:main}
	Let $(X_n)_{n\in\mathbb{N}}$ be a sequence of $\mathbb{R}$-valued identically distributed variables such that $\mathbb{E}[X_0]=0$ and $\mathbb{E}|X_0|^r<\infty$ for some $r>1$.
	Assume that the strong mixing coefficients satisfy $\alpha^X(n)=O(n^{-a})$ for some $a>0$, $a\neq 1$.
	Let $c>0$ be the constant of Lemma~\ref{lem:Hansen} (depending only on $r$, $\mathbb{E}|X_0|^r$ and $(\alpha^X(n))_{n\in\mathbb{N}}$), and set
	\begin{equation}\label{eq:Bsize}
	b= \frac{1}{2}[1-(1-a)_+](1-1/r),
	\qquad
	C=(2c)^{1/b}.
	\end{equation}
	Let $\delta>0$, and suppose that the block size $B$ in the block-average process~\eqref{eq:movAvg} satisfies $B > C\,\delta^{-1/b}$. Then
	\[
	\Delta
	:= \frac{1}{2}
	- \frac{c}{\delta}\,
	B^{-b}
	= \frac12\left(1-\Bigl(\frac{C\delta^{-1/b}}{B}\Bigr)^{b}\right)>0,
	\]
	and for every integer $1<q\le\tfrac{2}{7}K\Delta$ the median-of-means estimator $\hat{\mu}$ defined in~\eqref{eq:MoM} obeys
	\[
	\mathbb{P}(|\hat{\mu}|\ge \delta)
	\le \mathcal{R}_{K,q}\bigl(\Delta,\alpha^X(qB)\bigr)
	= 4 \exp\!\left(
	-\frac{\Delta K}{7q}
	\log\!\Bigl(1+\frac{(2/7)\Delta K}{K+q}\Bigr)
	\right)
	+ \frac{28}{\Delta}\,\alpha^X(qB).
	\]
<!-- end theorem-1 -->
<!-- proof-3: body_md -->
As in the classical i.i.d.\ case, the key observation is that the event $\{|\hat{\mu}|\ge \delta\}$ implies that at least half of the block averages $\mu_0,\ldots,\mu_{K-1}$ deviate from zero by at least~$\delta$. Indeed, if $\hat\mu\ge\delta$, $\mu_l\ge\hat\mu\ge\delta$ for at least $K/2$ indices $l$, and the case $\hat\mu\le-\delta$ is symmetric. Hence,
	\begin{align*}
		\mathbb{P}(|\hat{\mu}|\ge \delta)
		&\le
		\mathbb{P}\!\left(\sum_{l=0}^{K-1} \mathds{1}_{\{|\mu_l|\ge \delta\}} \ge \frac{K}{2}\right)
		\le
		\mathbb{P}\!\left(\frac{1}{K}\Big|\sum_{l=0}^{K-1} (Z_l-\mathbb{E} Z_l)\Big|
		\ge \frac{1}{2}-\frac{1}{K}\sum_{l=0}^{K-1}\mathbb{E} Z_l\right),
	\end{align*}
	where $Z_l=\mathds{1}_{\{|\mu_l|\ge \delta\}}$, $0\le l<K$.

	By Lemma~\ref{lem:Hansen}, applied to the $l$-th block, there exists a constant $c>0$, depending only on $r$, $\mathbb{E}|X_0|^r$ and $(\alpha^X(n))_{n\in\mathbb{N}}$, such that for every $0\le l<K$,
	\[
	\mathbb{E} Z_l = \mathbb{P}(|\mu_l|\ge \delta)
	\le c\,\delta^{-1} B^{-b}.
	\]

	Since
	$B > C\,\delta^{-1/b},$
	it follows that
	\[
	0<\Delta = \frac{1}{2}
	- \frac{c}{\delta}\,
	B^{-b} \le \frac{1}{2}-\frac{1}{K}\sum_{l=0}^{K-1}\mathbb{E} Z_l,
	\]
	and therefore
	\[
	\mathbb{P}(|\hat{\mu}|\ge \delta)
	\le
	\mathbb{P}\!\left(
	\frac{1}{K}\Big|\sum_{l=0}^{K-1}(Z_l-\mathbb{E} Z_l)\Big|
	\ge \Delta
	\right).
	\]

	Applying Lemma~\ref{lem:Rio61} to the $\{0,1\}$-valued sequence $(Z_l)_{0\le l<K}$ yields, for every integer $1<q\le \tfrac{2}{7}K\Delta$,
	\[
	\mathbb{P}\!\left(
	\frac{1}{K}\Big|\sum_{l=0}^{K-1}(Z_l-\mathbb{E} Z_l)\Big|
	\ge \Delta
	\right)
	\le
	4 \exp\!\left(
	-\frac{\Delta K}{7q}
	\log\!\Bigl(1+\frac{(2/7)\Delta K}{K+q}\Bigr)
	\right)
	+ \frac{28}{\Delta}\,\alpha^Z(q+1).
	\]
	(The condition $q<K$ of Lemma~\ref{lem:Rio61} holds since $\Delta\le 1/2$.)

	Finally, since $Z_l$ is a function of the $l$-th block, the $\alpha$-mixing coefficients of $(X_n)_{n\in\mathbb{N}}$ and $(Z_l)_{0\le l<K}$ satisfy
	\[
	\alpha^Z(q)\le \alpha^X\bigl((q-1)B+1\bigr)\le\alpha^X\bigl((q-1)B\bigr), \quad q\ge1,
	\]
	and the desired inequality follows.

		\[
		\mathbb{P}(|\hat{\mu}|\ge \delta) = O\!\left(\kappa^{\sqrt{n}}\right)

<!-- lemma-3: body_md -->
\label{lem:meas}
	Assume~\eqref{eq:S}. Then:
	\begin{enumerate}[(i)]
		\item for every $l\in\mathbb{N}$ and $\omega\in\Omega$,
		$\sup_{f\in\mathcal{C}}|\mu_l(f)(\omega)|=\sup_{g\in\mathcal{C}_0}|\mu_l(g)(\omega)|$; in particular $\sup_{f\in\mathcal{C}}|\mu_l(f)|$ is a $[0,\infty]$-valued random variable which is measurable with respect to $\sigma(Y_{lB+1},\ldots,Y_{lB+B})$;
		\item if $\hat\mu(f)=\varphi\bigl(\mu_0(f),\ldots,\mu_{K-1}(f)\bigr)$ for a continuous median rule $\varphi:\mathbb{R}^K\to\mathbb{R}$ (e.g.\ the lower median), then $\sup_{f\in\mathcal{C}}|\hat\mu(f)|=\sup_{g\in\mathcal{C}_0}|\hat\mu(g)|$ is a random variable.
	\end{enumerate}
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
(i) The inequality ``$\ge$'' is trivial. Fix $f\in\mathcal{C}$, $\omega\in\Omega$, and let $g_m\in\mathcal{C}_0$ be as in~\eqref{eq:S}. Since $\mu_l(g_m)(\omega)$ is an average of $B$ values $g_m(Y_j(\omega))$ minus $\mathbb{E} g_m(Y_0)$, it converges to $\mu_l(f)(\omega)$. Hence $|\mu_l(f)(\omega)|=\lim_m|\mu_l(g_m)(\omega)|\le\sup_{g\in\mathcal{C}_0}|\mu_l(g)(\omega)|$. A countable supremum of functions measurable with respect to $\sigma(Y_{lB+1},\ldots,Y_{lB+B})$ is measurable with respect to the same $\sigma$-algebra.

	(ii) Since $\varphi$ is continuous, $\hat\mu(g_m)(\omega)\to\hat\mu(f)(\omega)$, and the argument of (i) applies. (The lower median $\varphi(x)$ of $x\in\mathbb{R}^K$ is nondecreasing in each coordinate and translation equivariant, $\varphi(x+t\mathbf 1)=\varphi(x)+t$; hence $|\varphi(x)-\varphi(y)|\le\max_l|x_l-y_l|$.)

\emph{Step 1 (counting).} If $m$ is a median of $x_0,\ldots,x_{K-1}$ and $|m|\ge\delta$ (resp.\ $|m|>\delta$), then $\#\{l:|x_l|\ge\delta\}\ge K/2$ (resp.\ $\#\{l:|x_l|>\delta\}\ge K/2$): if $m\ge\delta$, for at least $K/2$ indices $l$,
	$x_l\ge\delta$, and the case $m\le-\delta$ is symmetric. The bound $K/2$ cannot be improved to a strict inequality: for $K=2$, $x=(0,2\delta)$ and the midpoint median $m=\delta$, only one block deviates.

	\emph{Step 2 (the supremum).} Let $s_l:=\sup_{f\in\mathcal{C}}|\mu_l(f)|\in[0,\infty]$ and $Z_l:=\mathds{1}_{\{s_l\ge\delta\}}$. We claim that
	\begin{equation}\label{eq:inclusion}
		\Bigl\{\sup_{f\in\mathcal{C}}|\hat\mu(f)|\ge\delta\Bigr\}
		\subseteq
		\Bigl\{\sum_{l=0}^{K-1}Z_l\ge \frac K2\Bigr\}.
	\end{equation}
	Suppose that $\sum_lZ_l<K/2$. Then $L:=\{l:s_l<\delta\}$ is nonempty and $\delta':=\max_{l\in L}s_l<\delta$. For every $f\in\mathcal{C}$ we have $\{l:|\mu_l(f)|>\delta'\}\subseteq\{l:s_l\ge\delta\}$, a set of fewer than $K/2$ elements, so by the strict version of Step~1, $|\hat\mu(f)|\le\delta'$. Hence $\sup_f|\hat\mu(f)|\le\delta'<\delta$. Note that this argument does not require the supremum to be attained.

	\emph{Step 3 (centering).} By Lemma~\ref{lem:meas}, $Z_l$ is a random variable with $\mathbb{E} Z_l\le\varepsilon_{\delta,B}$, hence $\frac12-\frac1K\sum_l\mathbb{E} Z_l\ge\Delta$ and
	\[
	\Bigl\{\sum_{l=0}^{K-1}Z_l\ge\frac K2\Bigr\}
	\subseteq
	\Bigl\{\frac1K\Big|\sum_{l=0}^{K-1}(Z_l-\mathbb{E} Z_l)\Big|\ge\Delta\Bigr\}.
	\]

	\emph{Step 4 (mixing of the indicators).} By Lemma~\ref{lem:meas}, $Z_l$ is $\sigma(Y_{lB+1},\ldots,Y_{lB+B})$-measurable. Consequently $\sigma(Z_j:j\le k)\subseteq\sigma(Y_i:i\le(k+1)B)$ and $\sigma(Z_j:j\ge k+q+1)\subseteq\sigma(Y_i:i\ge(k+q+1)B+1)$; the gap between the two index sets is $qB+1$, whence
	\[
	\alpha^Z(q+1)\le\alpha^Y(qB+1)\le\alpha^Y(qB).
	\]

	\emph{Step 5.} Lemma~\ref{lem:Rio61}, applied to $(Z_{k-1})_{k\ge1}$ with $n=K$ and $\delta=\Delta$ is applicable, because $q\le\frac27K\Delta$ is equivalent to $\Delta\ge\frac{7q}{2K}$, and $q<K$ since $\Delta\le\frac12$. Combining the previous steps with the monotonicity of $\mathcal{R}_{K,q}$ in its second argument completes the proof.

<!-- remark-3: body_md -->
\label{rem:VC}
	Condition~\eqref{eq:S} cannot be dropped from Lemma~\ref{lem:meas}, and it is not implied
	by the Vapnik--Chervonenkis property of $\mathcal{C}$.
	Let $A\subset\mathbb{R}$ be a non-Borel set and consider the class $\mathcal{C}=\{\mathds{1}_{\{a\}}:a\in A\}$ of singleton indicators, whose VC dimension is~$1$. For every $y\in\mathbb{R}$, $\sup_{f\in\mathcal{C}}f(y)=\mathds{1}_A(y)$, which is not Borel measurable. 
	
	Moreover, density in $L^\infty(\mathbb{P}_{Y_0})$ (i.e.\ with respect to the essential supremum) is not sufficient either: if $\mathbb{P}_{Y_0}$ is non-atomic, every element of $\mathcal{C}\cup\{0\}$ equals $0$ almost surely, so $\{0\}$ is dense in $\mathcal{C}\cup\{0\}$ in $L^\infty(\mathbb{P}_{Y_0})$, yet $\sup_{f\in\mathcal{C}\cup\{0\}}f(Y_0)=\mathds{1}_A(Y_0)$ is not a random variable when, for instace, $\Omega=[0,1]$ with the Borel $\sigma$-algebra and the Lebesgue measure, $Y_0(\omega)=\omega$, and $A\subset[0,1]$ is not Borel. The uniform norm in condition (b) above is therefore essential.

<!-- proposition-1: body_md -->
\label{prop:Lipf}
	Let $\mathcal{C}=\{f_\theta:\theta\in\Theta\}$, where $(\Theta,d)$ is a totally bounded metric space, $\mathbb{E}|f_\theta(Y_0)|<\infty$, and
	\[
	|f_\theta(y)-f_{\theta'}(y)|\le d(\theta,\theta')\,\Lambda(y),\qquad \theta,\theta'\in\Theta,\ y\in\mathcal{Y},
	\]
	for a measurable $\Lambda\ge0$ with $\mathbb{E}\Lambda(Y_0)<\infty$. Then $\mathcal{C}$ satisfies~\eqref{eq:S}, and for every $l\in\mathbb{N}$ and $\delta,\eta>0$,
	\[
	\mathbb{P}\Bigl(\sup_{\theta\in\Theta}|\mu_l(f_\theta)|\ge\delta\Bigr)
	\le
	N(\Theta,\eta)\,\sup_{\theta\in\Theta}\mathbb{P}\bigl(|\mu_l(f_\theta)|\ge\delta/2\bigr)
	+\frac{4\eta\,\mathbb{E}\Lambda(Y_0)}{\delta}.
	\]
	If $\Lambda\equiv L$ is constant and $4L\eta<\delta$, the last term can be omitted.
<!-- end proposition-1 -->
<!-- proof-7: body_md -->
The metric space $\Theta$ is separable, and $\theta\mapsto f_\theta(y)$ as well as $\theta\mapsto\mathbb{E} f_\theta(Y_0)$ are Lipschitz, so~\eqref{eq:S} holds by (c). Let $\theta_1,\ldots,\theta_N$ be an $\eta$-net. If $d(\theta,\theta_i)\le\eta$ then $|\mu_l(f_\theta)-\mu_l(f_{\theta_i})|\le W_l:=\eta\bigl(B^{-1}\sum_{j=1}^B\Lambda(Y_{lB+j})+\mathbb{E}\Lambda(Y_0)\bigr)$. If $\max_i|\mu_l(f_{\theta_i})|<\delta/2$ and $W_l<\delta/2$, then $\sup_\theta|\mu_l(f_\theta)|\le\max_i|\mu_l(f_{\theta_i})|+W_l<\delta$. Hence 
	\[
	\{\sup_\theta|\mu_l(f_\theta)|\ge\delta\}\subseteq\bigcup_i\{|\mu_l(f_{\theta_i})|\ge\delta/2\}\cup\{W_l\ge\delta/2\},
	\]
	and Markov's inequality gives $\mathbb{P}(W_l\ge\delta/2)\le 2\mathbb{E} W_l/\delta=4\eta\mathbb{E}\Lambda(Y_0)/\delta$. If $\Lambda\equiv L$, then $W_l=2L\eta<\delta/2$.

<!-- lemma-4: body_md -->
\label{lem:block}
	Let $(X_n)_{n\in\mathbb{N}}$ be identically distributed, $l\in\mathbb{N}$, $\eta>0$, and let $\mu_l(g)=\frac1B\sum_{j=1}^Bg(X_{lB+j})-\mathbb{E} g(X_0)$.
	\begin{enumerate}[(i)]
		\item For a measurable set $S$ and $p=\mathbb{P}(X_0\in S)$,
		\[
		\mathbb{P}\bigl(|\mu_l(\mathds{1}_S)|\ge\eta\bigr)\le\frac{p(1-p)+2\sum_{k=1}^B\min\{\alpha^X(k),\,p(1-p)\}}{B\eta^2}\le\frac{\frac14+2A_B}{B\eta^2}.
		\]
		
		\item If $|g|\le a$, then
		$\displaystyle\mathbb{P}\bigl(|\mu_l(g)|\ge\eta\bigr)\le\frac{a^2(1+8A_B)}{B\eta^2}$.
		
		\item If $\mathbb{E}|g(X_0)|<\infty$, then for every $M\ge0$,
		\[
		\mathbb{P}\bigl(|\mu_l(g)|\ge\eta\bigr)\le\frac{4M^2(1+8A_B)}{B\eta^2}+\frac{4\,\mathbb{E}\bigl(|g(X_0)|-M\bigr)_+}{\eta}.
		\]
	\end{enumerate}
<!-- end lemma-4 -->
<!-- proof-8: body_md -->
All three bounds follow from Chebyshev's inequality and the bound
	\[
	\operatorname{Var}\Bigl(\sum_{j=1}^BW_j\Bigr)\le\sum_{i,j}|\operatorname{Cov}(W_i,W_j)|\le B\Bigl(\gamma_0+2\sum_{k=1}^B\gamma_k\Bigr),
	\]
	valid whenever $|\operatorname{Cov}(W_i,W_j)|\le\gamma_{|i-j|}$.
	
	(i) For two events of probability $p$, $|\mathbb{P}(G\cap H)-p^2|\le p(1-p)$, and $|\mathbb{P}(G\cap H)-\mathbb{P}(G)\mathbb{P}(H)|\le\alpha^X(k)$ if $G\in\sigma(X_i)$, $H\in\sigma(X_{i+k})$, by the very definition~\eqref{eq:mix_coef}.
	
	(ii) The covariance inequality 
	\[
	|\operatorname{Cov}(U,V)|\le4\alpha(\sigma(U),\sigma(V))\|U\|_\infty\|V\|_\infty
	\] 
	for bounded random variables (see Theorem~A.5 in~\cite{hall2014martingale}; an $L^p$ version is due to Davydov~\cite{davydov1968convergence}) gives $\gamma_k=4a^2\alpha^X(k)$, $k\ge1$, and $\gamma_0=a^2$.
	
	(iii) Let $g_M=\max(-M,\min(M,g))$ and $r_M=g-g_M$, so that $|r_M|=(|g|-M)_+$. Then $\mu_l(g)=\mu_l(g_M)+\mu_l(r_M)$, $\mathbb{P}(|\mu_l(g_M)|\ge\eta/2)$ is bounded by (ii), and $\mathbb{P}(|\mu_l(r_M)|\ge\eta/2)\le2\mathbb{E}|\mu_l(r_M)|/\eta\le4\mathbb{E}|r_M(X_0)|/\eta$ by Markov's inequality.

<!-- remark-4: body_md -->
If $\mathbb{E}|g(X_0)|^r<\infty$ for some $r>1$, then $\mathbb{E}(|g(X_0)|-M)_+\le\mathbb{E}|g(X_0)|^rM^{1-r}$, and the choice $M\asymp(B\eta)^{1/(r+1)}$ in Lemma~\ref{lem:block}(iii) gives, when $A_\infty<\infty$, a bound of order $\eta^{-2r/(r+1)}B^{-(r-1)/(r+1)}$. The sharp form of (i) is important for indicators of rare events, e.g.\ when estimating high quantiles in Section~\ref{sec:VaR}.

<!-- theorem-3: body_md -->
\label{thm:NW}
	Assume that $(X_t,Y_t)_{t\ge1}$ is $\alpha$-mixing with coefficients $\alpha(k)$, $A_B=\sum_{k=1}^B\alpha(k)$, and that at most $n_o$ blocks are corrupted. Let $b_h:=\inf_{x\in D}f_h(x)>0$, $M_h:=\sup_{x\in D}|m_h(x)|$, $\delta_1>0$ and $0<\delta_2<b_h$. For $\eta_1>0$ and $M\ge0$, let
	\begin{align*}
		\varepsilon_1&:=N(D,\eta_1)\Bigl[\frac{16M^2(1+8A_B)}{B\delta_1^2}+\frac{8\,\mathbb{E}\bigl(h^{-d}\|\mathcal{K}\|_\infty|\psi(Y_1)|-M\bigr)_+}{\delta_1}\Bigr]+\frac{4\eta_1L_\mathcal{K}\,\mathbb{E}|\psi(Y_1)|}{h^{d+1}\delta_1},\\
		\varepsilon_2&:=N\Bigl(D,\frac{\delta_2h^{d+1}}{8L_\mathcal{K}}\Bigr)\,\frac{4\|\mathcal{K}\|_\infty^2(1+8A_B)}{Bh^{2d}\delta_2^2},
	\end{align*}
	and $\Delta_i:=\frac12-\frac{n_o}K-\varepsilon_i$. If $\Delta_1,\Delta_2>0$, then for every integer $1<q\le\frac27K\min(\Delta_1,\Delta_2)$,
	\[
	\mathbb{P}^*\Bigl(\sup_{x\in D}\bigl|\hat m(x)-m_h(x)\bigr|>\frac{\delta_1+M_h\delta_2}{b_h-\delta_2}\Bigr)
	\le\mathcal{R}_{K,q}\bigl(\Delta_1,\alpha(qB)\bigr)+\mathcal{R}_{K,q}\bigl(\Delta_2,\alpha(qB)\bigr).
	\]
<!-- end theorem-3 -->
<!-- proof-9: body_md -->
\emph{Deterministic step.} If $|a-A|\le\delta_1$ and $|b-F|\le\delta_2<F$, then
	\[
	\Bigl|\frac ab-\frac AF\Bigr|=\Bigl|\frac{(a-A)+\frac AF(F-b)}{b}\Bigr|\le\frac{\delta_1+|A/F|\,\delta_2}{F-\delta_2}.
	\]
	Applied with $a=\hat g(x)$, $A=g_h(x)$, $b=\hat f(x)$, $F=f_h(x)\ge b_h$, it shows that the event on the left-hand side is contained in $\{\sup_x|\hat g(x)-g_h(x)|\ge\delta_1\}\cup\{\sup_x|\hat f(x)-f_h(x)|\ge\delta_2\}$. Since the median is translation equivariant, $\hat g(x)-g_h(x)$ and $\hat f(x)-f_h(x)$ are the MoM processes of the classes $\mathcal{C}_1=\{(u,v)\mapsto\psi(v)\mathcal{K}_h(x-u):x\in D\}$ and $\mathcal{C}_2=\{u\mapsto\mathcal{K}_h(x-u):x\in D\}$.

	\emph{Probabilistic step.} Both classes are Lipschitz in $x$: $|\psi(v)\mathcal{K}_h(x-u)-\psi(v)\mathcal{K}_h(x'-u)|\le\|x-x'\|_\infty L_\mathcal{K} h^{-d-1}|\psi(v)|$, and similarly with $|\psi(v)|$ replaced by $1$. By Proposition~\ref{prop:Lipf} they satisfy~\eqref{eq:S}. For $\mathcal{C}_1$, Proposition~\ref{prop:Lipf} with $\Lambda(u,v)=L_\mathcal{K} h^{-d-1}|\psi(v)|$ and Lemma~\ref{lem:block}(iii) at level $\delta_1/2$ (note $|\psi(v)\mathcal{K}_h(x-u)|\le h^{-d}\|\mathcal{K}\|_\infty|\psi(v)|$) show that $\varepsilon_1$ bounds the single-block probabilities at level $\delta_1$. For $\mathcal{C}_2$, the choice $\eta=\delta_2h^{d+1}/(8L_\mathcal{K})$ makes the Lipschitz term vanish in Proposition~\ref{prop:Lipf}, and Lemma~\ref{lem:block}(ii) with $a=h^{-d}\|\mathcal{K}\|_\infty$ at level $\delta_2/2$ yields $\varepsilon_2$. Corollary~\ref{cor:contamination} applied to both classes completes the proof.

Let $(X_t)_{t\ge1}$ be an $\mathbb{R}^d$-valued, identically distributed, $\alpha$-mixing sequence with coefficients $\alpha(k)$ and common density $f$, and let $A_B=\sum_{k=1}^B\alpha(k)$. In contrast with~\cite{humbert2022robust}, the blocks are formed by \emph{consecutive} observations: a random partition would destroy the dependence structure on which Lemma~\ref{lem:Rio61} relies (for i.i.d.\ data the two constructions have the same distribution). The block estimates and the MoM-KDE are
\[
\hat f_l(x)=\frac1{Bh^d}\sum_{j=1}^B\mathcal{K}\Bigl(\frac{x-X_{lB+j}}{h}\Bigr),\qquad
\hat f_{\rm MoM}(x)=\operatorname{median}\bigl(\hat f_0(x),\ldots,\hat f_{K-1}(x)\bigr)
\]
(lower median, so that $\hat f_{\rm MoM}$ is continuous) and, when a density is required and $\int\hat f_{\rm MoM}>0$, $\tilde f=\hat f_{\rm MoM}/\int\hat f_{\rm MoM}$. (Humbert et al.~\cite{humbert2022robust} write the kernel as $\mathcal{K}((X_i-x)/h)$; for symmetric kernels, $\mathcal{K}((x-X)/h)=\mathcal{K}((X-x)/h)$.) We assume:
\begin{enumerate}[({K}1)]
	\item $\mathcal{K}\ge0$, $\int\mathcal{K}=1$, $\|\mathcal{K}\|_\infty<\infty$, $|\mathcal{K}(u)-\mathcal{K}(v)|\le L_\mathcal{K}\|u-v\|_\infty$, and $\int\mathcal{K}(u)\|u\|^\beta du<\infty$;
	\item $f$ belongs to the H\"older class $\Sigma(\beta,L)$, $0<\beta\le1$: $|f(x)-f(x')|\le L\|x-x'\|^\beta$.
\end{enumerate}
Assumption~(K1) is satisfied by the Gaussian, Epanechnikov, triangular, and many other kernels. Under~(K1)--(K2) the bias satisfies
\begin{equation}\label{eq:KDEbias}
	\bigl|\mathbb{E}\hat f_l(x)-f(x)\bigr|=\Bigl|\int\mathcal{K}(u)\bigl(f(x-hu)-f(x)\bigr)du\Bigr|\le C_2h^\beta,\qquad C_2:=L\int\mathcal{K}(u)\|u\|^\beta du,
\end{equation}
by the change of variables $y=x-hu$ in $\mathbb{E}\mathcal{K}_h(x-X_1)=\int\mathcal{K}_h(x-y)f(y)\,dy$.

If the joint densities of $(X_t,X_{t+k})$, $t,k\ge1$, are bounded by $M_2$ and $\|f\|_\infty<\infty$, then $\operatorname{Var}\mathcal{K}_h(x-X_1)\le\|\mathcal{K}\|_\infty\|f\|_\infty h^{-d}$ and

	|\operatorname{Cov}(\mathcal{K}_h(x-X_1),\mathcal{K}_h(x-X_{1+k}))|\le\min\{M_2+\|f\|_\infty^2,\,4\alpha(k)\|\mathcal{K}\|_\infty^2h^{-2d}\}; 
	\]
	splitting the sum of covariances at $k\asymp h^{-d}$ (Masry's technique, see~\cite{bosq1998nonparametric}) shows that $\operatorname{Var}\hat f_l(x)=O((Bh^d)^{-1})$ under $\alpha(k)=O(k^{-a})$ with $a>2$, and the factor $h^{-2d}$ in $\varepsilon_{\delta,B,h}$ improves to $h^{-d}$. 
	
	Since Theorem~\ref{thm:MoMproc} takes the single-block bound as an input, any sharper single-block bound, e.g.\ one based on the Bernstein inequality for strongly mixing sequences of Merlev\`ede, Peligrad and Rio~\cite{merlevede2009bernstein}, can be inserted directly.

<!-- proposition-2: body_md -->
\label{prop:KDEL1}
	Let $\mathcal S$ be the set of uncorrupted blocks and assume $K>2n_o$, i.e.\ $|\mathcal S|>K/2$.
	\begin{enumerate}[(i)]
		\item Deterministically, $\|\hat f_{\rm MoM}-f\|_1\le\sum_{l\in\mathcal S}\|\hat f_l-f\|_1$ and, if $\|\hat f_{\rm MoM}-f\|_1<1$ (which implies $\int\hat f_{\rm MoM}>0$), $\|\tilde f-f\|_1\le2\|\hat f_{\rm MoM}-f\|_1$.
		\item If $K$ is fixed, $A_\infty<\infty$, $h\to0$ and $Bh^{2d}\to\infty$, then $\|\tilde f-f\|_1\to0$ in outer probability.
	\end{enumerate}
<!-- end proposition-2 -->
<!-- proof-12: body_md -->
(i) Since the uncorrupted blocks form a strict majority, every median lies between 
	the minimum and the maximum of $\hat f_l(x)$ over $l\in\mathcal S$
	(otherwise more than $K/2$ values would lie strictly on one side of it). Hence $|\hat f_{\rm MoM}(x)-f(x)|\le\max_{l\in\mathcal S}|\hat f_l(x)-f(x)|\le\sum_{l\in\mathcal S}|\hat f_l(x)-f(x)|$; integrate. For the normalization, with $c=\int\hat f_{\rm MoM}>0$, $\int|\hat f_{\rm MoM}/c-f|\le|1/c-1|\,c+\int|\hat f_{\rm MoM}-f|=|1-c|+\|\hat f_{\rm MoM}-f\|_1$ and $|1-c|=|\int(f-\hat f_{\rm MoM})|\le\|\hat f_{\rm MoM}-f\|_1$.

	(ii) By (i) it suffices to show $\mathbb{E}\|\hat f_l-f\|_1\to0$ for each uncorrupted $l$. Write $f_h=\mathbb{E}\hat f_l$. Then $\|f_h-f\|_1\to0$ as $h\to0$ ($\mathcal{K}_h$ is an approximate identity), and for $Q_R=[-R,R]^d$, by the Cauchy--Schwarz inequality and the variance bound $\operatorname{Var}\hat f_l(x)\le h^{-2d}\|\mathcal{K}\|_\infty^2(1+8A_\infty)/B$ from the proof of Lemma~\ref{lem:block}(ii),
	\[
	\mathbb{E}\|\hat f_l-f_h\|_1\le(2R)^dh^{-d}\|\mathcal{K}\|_\infty\Bigl(\frac{1+8A_\infty}{B}\Bigr)^{1/2}+2\int_{\mathbb{R}^d\setminus Q_R}f_h.
	\]
	The first term tends to $0$ for every fixed $R$, and the second one is small uniformly in $h\le1$ for large $R$.

<!-- theorem-5: body_md -->
\label{thm:VaR}
	Let $\varepsilon>0$ and assume that $F(q_\alpha+\varepsilon)>\alpha$. Put $p_\pm=F(q_\alpha\pm\varepsilon)$, $\eta_+=p_+-\alpha$, $\eta_-=\alpha-p_->0$,
	\[
	\varepsilon_\pm:=\frac{p_\pm(1-p_\pm)+2\sum_{k=1}^B\min\{\alpha^X(k),\,p_\pm(1-p_\pm)\}}{B\eta_\pm^2},\qquad
	\Delta_\pm:=\frac12-\frac{n_o}K-\varepsilon_\pm.
	\]
	If $\Delta_+,\Delta_->0$, then for every integer $1<q\le\frac27K\min(\Delta_+,\Delta_-)$,
	\[
	\mathbb{P}^*\bigl(|\widehat{\mathrm{VaR}}_\alpha-\mathrm{VaR}_\alpha|>\varepsilon\bigr)\le\mathcal{R}_{K,q}\bigl(\Delta_+,\alpha^X(qB)\bigr)+\mathcal{R}_{K,q}\bigl(\Delta_-,\alpha^X(qB)\bigr).
	\]
	No moment assumption is needed. Without Lemma~\ref{lem:Rio61}, the right-hand side can be replaced by $\sum_\pm\bigl(\frac14+2\sum_{k=0}^{K-1}\alpha^X(kB+1)\bigr)/(K\Delta_\pm^2)$ (Remark~\ref{rem:decrate}, applied to the event $\{\sum_lZ_l\ge\frac K2-n_o\}$).
<!-- end theorem-5 -->
<!-- proof-13: body_md -->
\emph{Quantile inversion.} If $\widehat{\mathrm{VaR}}_\alpha>q_\alpha+\varepsilon$, then $\hat F(q_\alpha+\varepsilon)<\alpha$, since otherwise $q_\alpha+\varepsilon$ would belong to the set whose infimum defines $\widehat{\mathrm{VaR}}_\alpha$. If $\widehat{\mathrm{VaR}}_\alpha<q_\alpha-\varepsilon$, there is $t<q_\alpha-\varepsilon$ with $\hat F(t)\ge\alpha$, and by monotonicity $\hat F(q_\alpha-\varepsilon)\ge\alpha$. Therefore
	\[
	\{|\widehat{\mathrm{VaR}}_\alpha-q_\alpha|>\varepsilon\}\subseteq\{|\hat F(q_\alpha+\varepsilon)-F(q_\alpha+\varepsilon)|\ge\eta_+\}\cup\{|\hat F(q_\alpha-\varepsilon)-F(q_\alpha-\varepsilon)|\ge\eta_-\}.
	\]
	\emph{Deviation of $\hat F$.} The difference $\hat F(t)-F(t)$ is the MoM of the block means $\mu_l(\mathds{1}_{(-\infty,t]})$, so each event is controlled by Theorem~\ref{thm:MoMproc} and Corollary~\ref{cor:contamination} with a one-element class, the single-block bound being given by Lemma~\ref{lem:block}(i). The polynomial version follows from Remark~\ref{rem:decrate}.
<!-- end proof-13 -->
<!-- remark-6: body_md -->
\label{rem:VaRblock}
	If $F$ has a positive density $F'$ near $q_\alpha$, then $\eta_\pm\approx F'(q_\alpha)\varepsilon$ and $p_\pm(1-p_\pm)\approx\alpha(1-\alpha)$. For geometrically mixing sequences $\sum_k\min\{\alpha^X(k),v\}=O(v\log(1/v))$, hence $\varepsilon_\pm\le\frac14$ as soon as
	\[
	B\gtrsim\frac{\alpha(1-\alpha)\log\frac1{\alpha(1-\alpha)}}{F'(q_\alpha)^2\,\varepsilon^2},
	\]
	which, up to the logarithmic factor, is the number of observations needed to resolve the quantile with accuracy $\varepsilon$ (compare with the asymptotic variance $\alpha(1-\alpha)/F'(q_\alpha)^2$ of the empirical quantile, see Bahadur~\cite{bahadur1966note}). For $B$ with $\varepsilon_\pm\le\frac14$, the deviation bound decays as $O(\kappa^{\sqrt n})$ under geometric mixing (Remark~\ref{rem:decrate}). For indicators no moment problem arises, and Lemma~\ref{lem:Rio61} applied to the full sample gives bounds of the same type for the empirical quantile of uncontaminated data; the benefit of the MoM construction here is its robustness against corrupted blocks and its compatibility with the CVaR estimators below, for which the median-of-means principle is essential.

<!-- lemma-6: body_md -->
\label{lem:RU}
	Assume $\mathbb{E}|X_1|<\infty$ and let $\xi$ be any $\alpha$-quantile, i.e.\ $\mathbb{P}(X_1<\xi)\le\alpha\le\mathbb{P}(X_1\le \xi)$. Then for every $c\in\mathbb{R}$,
	\[
	0\le G(c)-G(\xi)\le\frac{|c-\xi|\,|F(c)-\alpha|}{1-\alpha}.
	\]
	In particular, $\min_cG(c)=G(q_\alpha)=\mathrm{CVaR}_\alpha$, and if $|F(c)-\alpha|\le L|c-q_\alpha|$ near $q_\alpha$, then $G(c)-G(q_\alpha)\le L(c-q_\alpha)^2/(1-\alpha)$.
<!-- end lemma-6 -->
<!-- proof-14: body_md -->
The function $c\mapsto(x-c)_+$ is convex, with subgradients $-\mathds{1}_{\{x>\xi\}}$ and $-\mathds{1}_{\{x\ge \xi\}}$ at $\xi$ and $-\mathds{1}_{\{x>c\}}$ at $c$. Hence 
	\begin{equation*}
		(x-c)_+\ge(x-\xi)_+-(c-\xi)\mathds{1}_{\{x>\xi\}}
		\quad\text{and}\quad
		(x-c)_+\ge(x-\xi)_++(\xi-c)\mathds{1}_{\{x\ge \xi\}},
	\end{equation*}
	which after integration give 
	\[
	G(c)-G(\xi)\ge(c-\xi)(\mathbb{P}(X_1\le \xi)-\alpha)/(1-\alpha)\ge0
	\] 
	for $c\ge \xi$ and 
	\[
	G(c)-G(\xi)\ge(\xi-c)(\alpha-\mathbb{P}(X_1<\xi))/(1-\alpha)\ge0
	\] for $c\le \xi$. Similarly, $(x-c)_+\le(x-\xi)_+-(c-\xi)\mathds{1}_{\{x>c\}}$ yields 
	\[G(c)-G(\xi)\le(c-\xi)(F(c)-\alpha)/(1-\alpha).\] 
	The identification of $G(q_\alpha)$ with $\frac1{1-\alpha}\int_\alpha^1q_u\,du$ is classical~\cite{rockafellar2002conditional,acerbi2002coherence}.

<!-- theorem-6: body_md -->
\label{thm:CVaR}
	Assume $\mathbb{E}|X_1|<\infty$, let $a\le q_\alpha\le b$, and set $\widehat{\mathrm{CVaR}}_\alpha:=\min_{c\in[a,b]}\hat G(c)$. Let $\delta>0$ and $M\ge0$, let $N>4(b-a)/\delta$ be an integer, and define
	\[
	\varepsilon^{\rm CV}_{\delta,B}:=(N+1)\Bigl[\frac{16M^2(1+8A_B)}{B\delta^2}+\frac{8\,\mathbb{E}\bigl(|X_1|+\max(|a|,|b|)-M\bigr)_+}{\delta}\Bigr],
	\qquad\Delta:=\frac12-\frac{n_o}K-\varepsilon^{\rm CV}_{\delta,B}.
	\]
	If $\Delta>0$, then for every integer $1<q\le\frac27K\Delta$,
	\[
	\mathbb{P}^*\Bigl(\bigl|\widehat{\mathrm{CVaR}}_\alpha-\mathrm{CVaR}_\alpha\bigr|\ge\frac{\delta}{1-\alpha}\Bigr)\le\mathcal{R}_{K,q}\bigl(\Delta,\alpha^X(qB)\bigr).
	\]
<!-- end theorem-6 -->
<!-- proof-15: body_md -->
By Lemma~\ref{lem:RU}, $\mathrm{CVaR}_\alpha=\min_{[a,b]}G$. For functions with $|\hat G-G|\le s$ on $[a,b]$ the minima differ by at most $s$, and $\hat G(c)-G(c)$ is $(1-\alpha)^{-1}$ times the MoM process of the class $\{x\mapsto(x-c)_+:c\in[a,b]\}$. This class is Lipschitz in $c$ with $\Lambda\equiv1$, hence satisfies~\eqref{eq:S}, and $|(x-c)_+|\le|x|+\max(|a|,|b|)$. With the $\frac{b-a}N$-net $\{a+k(b-a)/N\}_{k=0}^N$ the Lipschitz term of Proposition~\ref{prop:Lipf} vanishes (as $4(b-a)/N<\delta$), and Lemma~\ref{lem:block}(iii) at level $\delta/2$ gives $\varepsilon^{\rm CV}_{\delta,B}$. Corollary~\ref{cor:contamination} completes the proof.

<!-- theorem-7: body_md -->
\label{thm:Bahadur}
	Assume $\mathbb{E}|X_1|<\infty$, $F(q_\alpha)=\alpha$, and $|F(c)-\alpha|\le L|c-q_\alpha|$ for $|c-q_\alpha|\le\varepsilon$. Let $\gamma>0$ and let $\varepsilon^{\rm CV}_{(1-\alpha)\gamma,B}$ be defined as in Theorem~\ref{thm:CVaR} with $[a,b]=[q_\alpha-\varepsilon,q_\alpha+\varepsilon]$ and $\delta=(1-\alpha)\gamma$, and assume that $\Delta_\gamma=\frac12-\frac{n_o}K-\varepsilon^{\rm CV}_{(1-\alpha)\gamma,B}>0$. 	\[
	\mathbb{P}^*\Bigl(\bigl|\widetilde{\mathrm{CVaR}}_\alpha-\mathrm{CVaR}_\alpha\bigr|\ge\gamma+\frac{L\varepsilon^2}{1-\alpha}\Bigr)

	\]

<!-- remark-7: body_md -->
Given a tolerance $\varepsilon$ (or $\delta$) and an admissible risk $p$: (1) choose $B$ such that the single-block bound $\varepsilon_\pm$ (resp.\ $\varepsilon^{\rm CV}$) is at most $\frac14$, using a conservative upper bound for the mixing coefficients of the return model; (2) with $K=n/B$ blocks and $n_o\le K/8$ suspected corrupted blocks, $\Delta\ge\frac18$; (3) choose $q$ as in Corollary~\ref{cor:specmix} and read off $p$ from $\mathcal{R}_{K,q}(\Delta,\alpha^X(qB))$; conversely, the smallest $n$ achieving a prescribed $p$ is obtained by solving for $K$.

<!-- head-2: body_md -->
Let $(\mathbb H,d)$ be a metric space describing the lot (e.g.\ a bounded subset of $\mathbb{R}^3$), and let $C(x)$, $x\in\mathbb H$, be a real-valued random field, $C(x)$ being the result of a measurement taken at the point $x$. We assume that the variables $C(x)$, $x\in\mathbb H$, are identically distributed (not necessarily independent) and integrable, and we wish to estimate $\mu:=\mathbb{E} C(x)$. Identical distribution excludes systematic trends such as segregation; the results below use it only through the common mean $\mu$ and the single-group bounds. For $D\ge0$ and $N\in\mathbb{N}$ we define the mixing coefficients of the field by
\begin{equation}\label{eq:spatial_mixing}
	\alpha^C_N(D):=\sup_{A,A'}\alpha\bigl(\sigma(C(x):x\in A),\,\sigma(C(x):x\in A')\bigr),
\end{equation}
where the supremum is taken over all finite sets $A,A'\subseteq\mathbb H$ with $|A|+|A'|\le N$ and $d(a,a')\ge D$ for all $a\in A$, $a'\in A'$. Clearly, $\alpha^C_N(D)$ is nonincreasing in $D$ and nondecreasing in $N$.

<!-- proposition-3: body_md -->
\label{prop:serial}
	Under~\eqref{eq:spatial_gap}, the sequence~\eqref{eq:serial} satisfies
	\[
	\alpha^Y(m)\le\alpha^C_n(D)\qquad\text{for every } m\ge B,
	\]
	and $\alpha\bigl(\sigma(Y_{lB+1},\ldots,Y_{lB+B}),\,\sigma(Y_{l'B+1},\ldots,Y_{l'B+B})\bigr)\le\alpha^C_{2B}(D)$ for $l\ne l'$.
<!-- end proposition-3 -->
<!-- proof-17: body_md -->
Let $1\le k<n$ and $m\ge B$. If $i\le k$ and $i'\ge k+m$, then $i'-i\ge B$, so $Y_i$ and $Y_{i'}$ belong to different groups, and the corresponding points are at distance at least $D$ by~\eqref{eq:spatial_gap}. Hence the sets $A=\{x_{l,j}:lB+j\le k\}$ and $A'=\{x_{l,j}:k+m\le lB+j\le n\}$ satisfy $d(a,a')\ge D$ for all $a\in A$, $a'\in A'$, and $|A|+|A'|\le n$. Since $\sigma(Y_i:i\le k)=\sigma(C(x):x\in A)$ and $\sigma(Y_i:k+m\le i\le n)=\sigma(C(x):x\in A')$, the first claim follows from~\eqref{eq:spatial_mixing}. The second claim is proved in the same way, with $A$ and $A'$ the sets of increments of the groups $l$ and $l'$.

<!-- corollary-5: body_md -->
\label{cor:spatial}
	Assume~\eqref{eq:spatial_gap}, let $\delta>0$ and $\varepsilon\ge\max_l\mathbb{P}(|\mu_l-\mu|\ge\delta)$, where $\mu_l$ refers to the uncorrupted measurements, and suppose that $\Delta:=\frac12-\frac{n_o}K-\varepsilon>0$. Then, with $\mathbb{P}^*$ denoting outer probability:
	\begin{enumerate}[(i)]
		\item for every integer $1<q\le\frac27K\Delta$,
		\[
		\mathbb{P}^*\bigl(|\hat\mu-\mu|\ge\delta\bigr)\le\mathcal{R}_{K,q}\bigl(\Delta,\alpha^C_n(D)\bigr);
		\]
		in particular, if $K\ge7/\Delta$, then
		\[
		\mathbb{P}^*\bigl(|\hat\mu-\mu|\ge\delta\bigr)\le4\Bigl(1+\frac\Delta4\Bigr)^{-\Delta K/14}+\frac{28}{\Delta}\,\alpha^C_n(D);
		\]
		\item without Lemma~\ref{lem:Rio61},
		\[
		\mathbb{P}^*\bigl(|\hat\mu-\mu|\ge\delta\bigr)\le\frac{\frac14+(K-1)\,\alpha^C_{2B}(D)}{K\Delta^2}\le\frac1{4K\Delta^2}+\frac{\alpha^C_{2B}(D)}{\Delta^2}.
		\]
	\end{enumerate}
<!-- end corollary-5 -->
<!-- proof-18: body_md -->
(i) The proofs of Theorem~\ref{thm:MoMproc} and Corollary~\ref{cor:contamination} only involve $Y_1,\ldots,Y_n$, so they apply to the sequence~\eqref{eq:serial} and the one-element class $\{y\mapsto y\}$. By Proposition~\ref{prop:serial}, $\alpha^Y(qB)\le\alpha^C_n(D)$, which gives the first bound. Since $\alpha^C_n(D)$ does not depend on $q$, we may take $q=2$, which is admissible if $K\ge7/\Delta$; then $K\ge14$, hence $\frac{(2/7)\Delta K}{K+2}\ge\frac27\cdot\frac78\,\Delta=\frac\Delta4$, and the second bound follows.

	(ii) Let $Z_l=\mathds{1}_{\{|\mu_l-\mu|\ge\delta\}}$, computed from the uncorrupted measurements. As in Step~1 of the proof of Theorem~\ref{thm:MoMproc} and in Corollary~\ref{cor:contamination}, $\{|\hat\mu-\mu|\ge\delta\}\subseteq\{\sum_lZ_l\ge\frac K2-n_o\}\subseteq\{\sum_l(Z_l-\mathbb{E} Z_l)\ge K\Delta\}$. Moreover, $\operatorname{Var} Z_l\le\frac14$ and, by Proposition~\ref{prop:serial}, $|\operatorname{Cov}(Z_l,Z_{l'})|\le\alpha^C_{2B}(D)$ for $l\ne l'$. Hence $\operatorname{Var}\sum_lZ_l\le\frac K4+K(K-1)\alpha^C_{2B}(D)$, and Chebyshev's inequality completes the proof.

<!-- remark-8: body_md -->
\label{rem:spatial_block}
	The proof of Lemma~\ref{lem:block}(i)--(iii) bounds the variance of a block sum by the sum of the covariances of pairs of observations, which are controlled by the dependence between two single observations. Therefore it carries over to the group $l$ if $A_B$ is replaced by
	\[
	A^{\rm loc}_l:=\frac1{2B}\sum_{i\ne j}\alpha^C_2\bigl(d(x_{l,i},x_{l,j})\bigr),
	\]
	and, in (i), $\sum_{k=1}^B\min\{\alpha^X(k),p(1-p)\}$ by $\frac1{2B}\sum_{i\ne j}\min\{\alpha^C_2(d(x_{l,i},x_{l,j})),p(1-p)\}$. For instance, by (iii), for every $M\ge0$,
	\[
	\mathbb{P}\bigl(|\mu_l-\mu|\ge\delta\bigr)\le\frac{4M^2(1+8A^{\rm loc}_l)}{B\delta^2}+\frac{4\,\mathbb{E}\bigl(|C(x)|-M\bigr)_+}{\delta}.
	\]
	If the increments of a group are close to each other, they are strongly dependent, $A^{\rm loc}_l$ is of order $B$, and this bound does not improve as $B$ grows: averaging within a group only removes the small-scale variability of the material. A positive $\Delta$ then requires the tolerance $\delta$ to exceed the typical deviation of a group average from $\mu$, and the error probability is driven by the number $K$ of well-separated groups.

<!-- remark-9: body_md -->
\label{rem:spatial_design}
	The two terms in Corollary~\ref{cor:spatial} compete: the first one requires many groups, the second one a large separation $D$, and in a bounded lot the number of groups that are pairwise $D$ apart is limited. If $\mathbb H\subset\mathbb{R}^3$ and $d$ is the Euclidean distance, the open balls of radius $D/2$ centered at one sampling point of each group are disjoint and contained in the $D/2$-neighborhood $\mathbb H^{D/2}$ of $\mathbb H$, hence
	\[
	K\le\frac{\operatorname{Vol}(\mathbb H^{D/2})}{\pi D^3/6}.
	\]
	Thus a small error probability can only be certified if the lot is large compared with the distance at which the spatial dependence becomes negligible. For moderate $K$, bound (ii) is preferable because of the large constants in Lemma~\ref{lem:Rio61}: for $\Delta=\frac14$ it gives $\mathbb{P}^*(|\hat\mu-\mu|\ge\delta)\le\frac4K+16\,\alpha^C_{2B}(D)$, which is below $0.05$ for $K=160$ groups if $\alpha^C_{2B}(D)\le1.5\cdot10^{-3}$, whereas the exponential term of $\mathcal{R}_{K,2}$ in (i) falls below $0.025$ only for $K\ge4122$ (and that of the simplified bound only for $K\ge4689$). Moreover, (i) involves $\alpha^C_n(D)$, i.e.\ sets of $n$ points, whereas (ii) only involves $\alpha^C_{2B}(D)$.
