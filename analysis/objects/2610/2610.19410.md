\label{thm:weighted-maximum}
		Let $1<q\leq2$ and let $\Omega\subset\mathbb R^2$ be a nonempty bounded open
		set. Suppose $A:\Omega\to\operatorname{Sym}(2)$ is measurable and
		$A(x)\in\operatorname{int}\Gamma_q^*$ a.e. If
		$u\in W^{2,q}_{\mathrm{loc}}(\Omega)\cap C(\overline\Omega)$ satisfies
		$\operatorname{tr}(AD^2u)=f \text{  a.e. in }\Omega$,
		then
		\begin{equation}\label{eq:weighted-maximum}
			\sup_\Omega u
			\leq \sup_{\partial\Omega}u
			+C_q\,\operatorname{diam}(\Omega)^{2-2/q}
			\left\|\frac{f^-}{\rho_q^*(A)}\right\|_{L^q(\Omega)},
			\qquad f^-:=\max\{-f,0\}.
		\end{equation}

\label{lem:truncated-integral}

			H_{p,a}(P)=S_p(P)\qquad\text{if }a\geq M(P).
		\end{equation}

		\begin{equation}\label{eq:positive-truncations}
			S_p(P)\geq0\quad\Longrightarrow\quad H_{p,a}(P)\geq0
			\qquad\text{for every }a>0.
		\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
If $P=0$, both conclusions are immediate. Otherwise, write
		$M=M(P)>0$ and $m=m(P)$. At $a=M$, equation~\eqref{eq:Hpa-is-Spa}
		and the signed-singular-value definition of $S_p$ give
		$H_{p,M}(P)=S_{p,M}(M,m)=S_p(P)$. For $a\geq M$, the defining integrand
		vanishes on $(M,a)$, which proves
		\eqref{eq:truncation-equals-Sp}.

		imply $0<m\leq M$. As a function of $a$, $H_{p,a}(P)$ increases
		from zero on $(0,m)$, decreases on $(m,M)$ to
		$H_{p,M}(P)=S_p(P)\geq0$, and is constant on $(M,\infty)$. This proves
		\eqref{eq:positive-truncations}.

<!-- proposition-2: body_md -->
\label{prop:q-variation}
		For fixed $\lambda$ with $\lambda_{\max}>0$, the function
		$q\mapsto\log S_q(\lambda)$ is concave on $[0,q_\lambda)$. Moreover,
		the function
		\[
		q\longmapsto S_q(\lambda)^{1/q}
		\]
		is nonincreasing on $(0,q_\lambda)$, and is strictly decreasing there
		unless $\lambda_1=\lambda_2$.
<!-- end proposition-2 -->
<!-- head-1: body_md -->
If $\lambda_{\max}=\lambda_{\min}$, then
	$S_q(\lambda)=\lambda_{\max}^q$ and both assertions are immediate.
	For $\lambda_{\max}>\lambda_{\min}$, equation
	\eqref{eq:Sq-theta} gives
	\[
	\log S_q(\lambda)
	=q\log\lambda_{\max}+\log(1-q\theta),
	\qquad
	\theta:=\frac{\lambda_{\max}-\lambda_{\min}}
	{2\lambda_{\max}}>0.
	\]
	Consequently,
	\[
	\frac{d^2}{dq^2}\log S_q(\lambda)
	=-\frac{\theta^2}{(1-q\theta)^2}<0,
	\]
	which proves strict concavity. Since $S_0(\lambda)=1$,
	\[
	\log\bigl(S_q(\lambda)^{1/q}\bigr)
	=\frac{\log S_q(\lambda)}q
	=\frac{\log S_q(\lambda)-\log S_0(\lambda)}{q-0}
	\]
	is the secant slope of the strictly concave function
	$q\mapsto\log S_q(\lambda)$ between $0$ and $q$. These secant slopes
	are strictly decreasing as the right endpoint $q$ increases. Hence
	$S_q(\lambda)^{1/q}$ is strictly decreasing unless
	$\lambda_1=\lambda_2$. \qed

<!-- lemma-2: body_md -->
\label{lem:boundary}
		After adding to $W$ a suitable affine map $x\mapsto cJx+v$, where
		$c\in\mathbb R$ and $v\in\mathbb R^2$, the periodic boundary function
		$h(\theta)=W(e(\theta))$, where
		$e(\theta)=(\cos\theta,\sin\theta)$, satisfies
		\begin{equation}\label{eq:boundary-control}
			\|h\|_{L^\infty(\mathbb R/2\pi\mathbb Z)}+
			\|h'\|_{L^\infty(\mathbb R/2\pi\mathbb Z)}\leq C,
		\end{equation}
		with an absolute constant independent of $B$.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Let $h(\theta) = W(e(\theta))$. The chain rule gives $h'(\theta)=DW(e(\theta))e'(\theta) = DW(e(\theta)) Je(\theta)$. 
		Write
		\[
		DW=E+E_0+\omega J,
		\qquad \omega=\tfrac12(\partial_1W_2-\partial_2W_1).
		\]
		We use the same letter
		$\omega(\theta)$ for its restriction to that circle. $E=0$ on $\partial \mathbb D$ and $E_0=(2\pi)^{-1}(Je)\otimes(Je)$, while $J(Je)=-e$. Therefore our earlier formula for $h'$ gives
		\begin{equation}\label{eq:boundary-derivative}
			h'(\theta)=\frac{1}{2\pi}Je(\theta)-\omega(\theta)e(\theta).
		\end{equation}
		We claim that
		\begin{equation}\label{eq:rotation-bound}
			\omega'(\theta)\leq \frac{1}{2\pi}\quad\text{for every }\theta.
		\end{equation}
		By simultaneously rotating the coordinates and vector components,
		it suffices to calculate at $e(0)=(1,0)$. There
		\eqref{eq:rotation-equations} yields
		\[
		\omega'(0)=\partial_2\omega
		=\partial_1(E_{22}+(E_0)_{22})-\partial_2(E_{12}+(E_0)_{12}).
		\]
		Since $E_{12}(e(\theta))=0$ on the boundary, differentiating at $\theta=0$ gives $\partial_2E_{12}(1,0)=0$.
		Since $E\geq0$ inside $\mathbb D$ and $E_{22}(1,0)=0$, its outward derivative
		satisfies $\partial_1E_{22}(1,0)\leq0$. Finally,
		\[
		(E_0)_{22}=\frac{1}{2\pi}\frac{x_1^2}{|x|^2},\qquad
		(E_0)_{12}=-\frac{1}{2\pi}\frac{x_1x_2}{|x|^2}
		\]
		give $\partial_1(E_0)_{22}(1,0)=0$ and
		$\partial_2(E_0)_{12}(1,0)=-\frac{1}{2\pi}$. This proves \eqref{eq:rotation-bound}.
		
		The function $\omega(\theta)$ is single-valued and $2\pi$-periodic.
		Integrating its one-sided derivative bound from a minimum to the next maximum counterclockwise shows that
		\[
		\max\omega-\min\omega\leq2\pi \cdot \frac{1}{2\pi}=1.
		\]
		Subtracting a suitable constant multiple of $Jx$ from $W$, which does
		not change its symmetric gradient, makes $|\omega|\leq1/2$ on the
		whole circle. Equation~\eqref{eq:boundary-derivative} now bounds $h'$.
		Subtracting by $v := W(e(0)) \in \mathbb R^2$ makes $h(0)=0$ so from integrating the bound on $h'$ around the circle, we bound $h$ in $L^{\infty}(\mathbb R/2\pi \mathbb Z)$ as well.

<!-- proof-3: body_md -->
Our goal is to prove \eqref{eq:weighted-maximum}. Dividing $A$ and $f$ by $\operatorname{tr} A$ preserves their weighted quotient. Thus we may assume $\operatorname{tr} A=1$ and $|A|\leq1$ a.e. We may also assume $f^- / \rho_q^*(A) \in L^q(\Omega)$.
		
		\medskip
		\emph{Step 1: smooth coefficients and solutions.}
		First let $A$ and $u$ be smooth on a neighborhood of
		$\overline\Omega$, with $A\in\operatorname{int}\Gamma_q^*$ there.
		Set $D=\operatorname{diam}\Omega$ and fix $x_0\in\Omega$. Then $\overline\Omega\Subset B_D(x_0)$.
		Choose an open set $U$ with
		$\overline\Omega\subset U\Subset B_D(x_0)$ on which $A$ is smooth and
		takes values in $\operatorname{int}\Gamma_q^*$, and choose
		$\chi\in C_c^\infty(U)$ such that $0\leq\chi\leq1$ and
		$\chi=1$ on a neighborhood of $\overline\Omega$. Define
		\[
		\widetilde A=\frac I2+\chi\left(A-\frac I2\right)\quad\text{in }U,
		\qquad \widetilde A=\frac I2\quad\text{in }B_D(x_0)\setminus U.
		\]
		The trace-one slice of $\operatorname{int}\Gamma_q^*$ is convex, so
		\[
		\operatorname{tr}\widetilde A=1,\qquad
		\lambda_{\min}(\widetilde A)>1-q/2,\qquad
		\widetilde A=I/2\quad\text{near }\partial B_D(x_0).
		\]
		In particular, $\widetilde A$ is smooth and uniformly elliptic on the closed
		disk. We henceforth write $A$ for this extension.
		Let $f=\operatorname{tr}(AD^2u)$ in $\Omega$ and solve
		\[
		-\operatorname{tr}(AD^2v)=f^-\mathbf1_\Omega\quad\text{in }B_D(x_0),
		\qquad v=0\quad\text{on }\partial B_D(x_0).
		\]
		The solution is nonnegative and belongs to $W^{2,s}$ for every finite
		$s>1$. Put $b=\sup_{\partial\Omega}u$. On $\Omega$, the function
		$v+b-u$ has nonnegative boundary values and satisfies
		$\operatorname{tr}(A D^2(v+b-u))=-f^--f=-f^+\leq0$. The minimum principle
		therefore gives
		\begin{equation}\label{eq:green-comparison}
			u(x_0)\leq b+v(x_0)
			=b+\int_\Omega G(x_0,x)f^-(x)\,dx.
		\end{equation}
		The tensor $B=GA$ has all the properties in
		Proposition~\ref{prop:dual-burkholder}, after dilation: $B\in L^2$,
		it is smooth away from $x_0$, vanishes on the outer circle, and
		$\operatorname{div}\operatorname{div}B=-\delta_{x_0}$.

		\[
		\int_{B_D(x_0)}G^{q'}\rho_q^*(A)^{q'}\,dx\leq C_qD^2.
		\]
		H\"older inequality gives
		\begin{equation}\label{eq:green-holder}
			\int_\Omega G(x_0,x)f^-(x)\,dx
			\leq \|G\rho_q^*(A)\|_{L^{q^{\prime}}(B_D(x_0))}
			\left\|\frac{f^-}{\rho_q^*(A)}\right\|_{L^q(\Omega)}
			\leq C_qD^{2/q^{\prime}}
			\left\|\frac{f^-}{\rho_q^*(A)}\right\|_{L^q(\Omega)}.
		\end{equation}

		only to construct $G$, not in the final bound.

		\emph{Step 2: measurable coefficients and Sobolev solutions on compact subdomains.}

		$I/2$ outside $\Omega$ and, for $0<\varepsilon<1$, set

		converging a.e. to $A_\varepsilon$. Convexity of the cone and
		\eqref{eq:rho-regularization} give
		\[
		\operatorname{tr} A_{\varepsilon,j}=1,\qquad |A_{\varepsilon,j}|\leq1,
		\qquad \rho_q^*(A_{\varepsilon,j})\geq\varepsilon/2.
		\]
		Indeed, each mollification is $(1-\varepsilon)$ times a
		$\Gamma_q^*$-valued average plus $\varepsilon I/2$.
		Mollify $u$ on a neighborhood of $\overline V$ so that
		$u_j\to u$ uniformly there and $D^2u_j\to D^2u$ in $L^q(V)$.
		For fixed $\varepsilon$, continuity of $\rho_q^*$ and the preceding
		lower bound imply
		\begin{equation}\label{eq:smooth-quotient-limit}
			\frac{\operatorname{tr}(A_{\varepsilon,j}D^2u_j)}{\rho_q^*(A_{\varepsilon,j})}
			\longrightarrow
			\frac{f_\varepsilon}{\rho_q^*(A_\varepsilon)}
			\quad\text{in }L^q(V).
		\end{equation}
		To check this, the part containing $D^2u_j-D^2u$ is bounded in
		absolute value by $2\varepsilon^{-1}|D^2u_j-D^2u|$; the remaining
		part converges a.e. and is dominated by a constant times
		$\varepsilon^{-1}|D^2u|$.
		Since the map $s\mapsto s^-$ is $1$-Lipschitz, the same convergence
		holds after taking negative parts.
		Apply Step 1 on $V$ to $(A_{\varepsilon,j},u_j)$ and let $j\to\infty$.
		
		We next let $\varepsilon\downarrow0$. Since
		$f_\varepsilon=(1-\varepsilon)f+(\varepsilon/2)\Delta u$,
		\eqref{eq:rho-regularization} gives
		\begin{equation}\label{eq:quotient-domination}
			\frac{f_\varepsilon^-}{\rho_q^*(A_\varepsilon)}
			\leq \frac{f^-}{\rho_q^*(A)}+(\Delta u)^-.
		\end{equation}
		The right-hand side is in $L^q(V)$ and
		$\rho_q^*(A)>0$ a.e. Since
		$f_\varepsilon/\rho_q^*(A_\varepsilon)$ converges pointwise to
		$f/\rho_q^*(A)$, dominated convergence gives
		\begin{equation}\label{eq:local-boundary-estimate}
			\sup_V u
			\leq \sup_{\partial V}u+C_q\,\operatorname{diam}(V)^{2-2/q}
			\left\|\frac{f^-}{\rho_q^*(A)}\right\|_{L^q(V)}.
		\end{equation}

		\emph{Step 3: localization at the boundary.}

		$V_\eta=\{x\in\Omega:u(x)>b+\eta\}$ for $\eta>0$.
		If nonempty, $V_\eta\Subset\Omega$ and $u=b+\eta$ on its boundary.

		and source norm are bounded by those on $\Omega$, and let
		$\eta\downarrow0$. This proves \eqref{eq:weighted-maximum} and hence
		the theorem.

Using $\det P=M(P)m(P)$, we can write \eqref{eq:Lt-definition} as
		\begin{equation}\label{eq:Lt-pieces}
			L_t(P)=
			\begin{cases}
				\det P,&M(P)\leq t,\\[1mm]
				t\bigl(M(P)+m(P)\bigr)-t^2,&M(P)>t.
			\end{cases}
		\end{equation}
		Indeed, when $M(P)>t$,
		\[
		\det P-\bigl(t(M(P)+m(P))-t^2\bigr)
		=(M(P)-t)(m(P)-t),
		\]
		and the two branches agree when $M(P)=t$. Equation
		\eqref{eq:Lt-pieces} will be used in both estimates.
		
		Next,
		\begin{equation}\label{eq:singular-sum-real}
			M(P)+m(P)=|\mathbf a(P)+\mathbf b(P)|.
		\end{equation}

		\[
		|\mathbf a(P)+\mathbf b(P)|^2
		=|P|^2+2\det P=(M(P)+m(P))^2,
		\]

		\eqref{eq:Lt-pieces} and \eqref{eq:singular-sum-real} immediately give

			\det P=\mathbf a(P)\cdot \mathbf b(P),\qquad
			|m(P)|\leq |\mathbf a(P)|\leq M(P).
		\end{equation}

		$\psi\in W^{1,2}_{\mathrm{per}}(Q;\mathbb R^2)$, periodicity and the determinant
		calculation used to prove \eqref{eq:moments-disk} give
		\begin{equation}\label{eq:moments}
			\int_Q(A+D\psi)\,dx=|Q|A,
			\qquad
			\int_Q\det(A+D\psi)\,dx=|Q|\det A.
		\end{equation}
		First let $f=Ax+\varphi$ with $\varphi$ smooth and periodic on $Q$,
		and write $f=(u,v)$. If $t<M(A)$, \eqref{eq:pointwise-cutoff}, Jensen's
		inequality for the Euclidean norm, and \eqref{eq:moments} give
		\begin{align*}
			\int_Q L_t(Df)\,dx
			&\geq\int_Q\bigl(t|\mathbf a(Df)+\mathbf b(Df)|-t^2\bigr)\,dx\\
			&\geq t\left|\int_Q\bigl(\mathbf a(Df)+\mathbf b(Df)\bigr)\,dx\right|
			-t^2|Q|\\
			&= |Q|\bigl(t(M(A)+m(A))-t^2\bigr)=|Q|L_t(A).

		\[
		\left|\int_Q\bigl(\mathbf a(Df)+\mathbf b(Df)\bigr)\,dx\right|
		=|Q|\,|\mathbf a(A)+\mathbf b(A)|=|Q|(M(A)+m(A)).
