<!-- definition-1: body_md -->
\label{def:principles}
We use the following principles.
\begin{enumerate}[label=(\roman*)]
 \item $\mathsf{AC}_{\omega}(\mathbb R)$ is countable choice for nonempty sets of reals.
 \item $\mathsf{PSP}_{\mathbb R}$ asserts that every uncountable subset of $2^\omega$
 contains a nonempty perfect subset.
 \item $\mathsf{CU}$ asserts that for every relation
 $S\subseteq2^\omega\times2^\omega$ with
 $\forall X\,\exists Y\ S(X,Y)$, there are a comeager
 $G_\delta$ set $C\subseteq2^\omega$ and a continuous
 $g:C\to2^\omega$ such that $S(X,g(X))$ for every $X\in C$.
\end{enumerate}
Continuity and the $G_\delta$ domain are part of the definition of
$\mathsf{CU}$ in this paper.
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{thm:main}
Work in $\mathsf{ZF}+\mathsf{AC}_{\omega}(\mathbb R)+\mathsf{PSP}_{\mathbb R}+\mathsf{CU}$. Every order-preserving
map $\varphi:\mathcal D_T\to\mathcal D_T$ is constant on a cone or above the
identity on a cone. If $\varphi$ is not constant on any cone:
\begin{enumerate}[label=(\roman*)]
 \item there is $\mathbf k\in\mathcal D_T$ such that
 \begin{equation}\label{eq:main-code}
  \mathbf d\le_T\mathbf k\vee\varphi(\mathbf d)
  \qquad(\mathbf d\ge_T\mathbf k);
 \end{equation}
 \item the range of $\varphi$ is cofinal;
 \item for every $\mathbf a\ge_T\mathbf k$ and every degree
 $\mathbf v$, the set
 \begin{equation}\label{eq:main-fiber}
  \{X\in2^\omega:\varphi(\mathbf a\vee[X]_T)=\mathbf v\}
 \end{equation}
 is at most countable;
 \item there is $\mathbf b\ge_T\mathbf k$ such that
 \begin{equation}\label{eq:main-domination}
  \mathbf d\le_T\varphi(\mathbf d)
  \qquad(\mathbf d\ge_T\mathbf b).
 \end{equation}
 Thus $\varphi$ is above the identity on the upper cone
 $\mathcal C_{\mathbf b}$.
\end{enumerate}
<!-- end theorem-1 -->
<!-- remark-1: body_md -->
\label{rem:LS-consequence}
Under $\mathsf{ZF}+\mathsf{AD}+\mathsf{DC}_{\mathbb R}$, the regularity hypotheses
of Theorem~\ref{thm:main} hold; see Theorem~\ref{cor:AD}.
If $f:2^\omega\to2^\omega$ is Turing invariant and order preserving,
the induced degree map
\[
 \varphi([X]_T)=[f(X)]_T
\]
is well defined and order preserving. Theorem~\ref{thm:main}
therefore shows that $f$ is constant in Turing degree on a cone or
satisfies $X\le_T f(X)$ throughout an upper cone.
Thus Theorem~\ref{thm:main} implies the order-preserving case of
Part~I of Martin's conjecture proved by Lutz--Siskind
\cite[Theorem~4.8]{LutzSiskind2025}.
The cofinality argument in Section~\ref{sec:cofinality} uses
Groszek--Slaman's original perfect-set lemma and relative highness;
it does not invoke the strengthened coding theorem
\cite[Theorem~4.3]{LutzSiskind2025}.
<!-- end remark-1 -->
<!-- theorem-2: body_md -->
\label{thm:consistency}
Let $B$ be any one of $\mathsf{AC}_{\omega}(\mathbb R)$, $\mathsf{DC}_{\mathbb R}$, and $\mathsf{DC}$.
The following theories are equiconsistent:
\begin{enumerate}[label=(\roman*)]
 \item $\mathsf{ZFC}$ plus the existence of a strongly inaccessible cardinal;
 \item $\mathsf{ZF}+B$ plus the assertion that every Turing-invariant,
 order-preserving function $f:2^\omega\to2^\omega$ is constant in
 Turing degree on a cone or satisfies $X\le_T f(X)$ on a cone;
 \item $\mathsf{ZF}+B$ plus the assertion that every order-preserving
 map $\varphi:\mathcal D_T\to\mathcal D_T$ is constant on a cone or above the
 identity on a cone.
\end{enumerate}
Moreover, the theory in~\textup{(ii)} proves that
$\omega_1^V$ is a strongly inaccessible cardinal in $L$.

<!-- lemma-1: body_md -->
\label{lem:code}
If $D\subseteq2^\omega$ and $g:D\to2^\omega$ is continuous in the
subspace topology, then some real $K$ and one oracle program $\Phi_e$ satisfy
\[
 g(X)=\Phi_e^{ K\oplus X}\qquad(X\in D).
\]

<!-- lemma-2: body_md -->
\label{lem:cantor-subset}
Every nonempty perfect subset of $2^\omega$ contains the image of a
continuous injection from $2^\omega$. Every comeager $G_\delta$
subset of $2^\omega$ also contains such an image.

<!-- lemma-3: body_md -->
\label{lem:continuous-restriction}
Every Baire-measurable function $g:X\to2^\omega$, where $X$ is
Polish, is continuous on a comeager $G_\delta$ subset of $X$.

<!-- definition-2: body_md -->
\label{def:saturated-range}
For a degree map $\varphi$, its \emph{saturated real range} is
\[
 R_\varphi=\{Y\in2^\omega:[Y]_T\in\operatorname{ran}(\varphi)\}.
\]
This set is defined without choosing representatives of all
degrees.

<!-- lemma-4: body_md -->
\label{lem:range}
For an order-preserving map $\varphi:\mathcal D_T\to\mathcal D_T$, the following
conditions are equivalent:
\begin{enumerate}[label=(\roman*)]
 \item $\varphi$ is constant on a cone;
 \item its range is bounded by one degree;
 \item $R_\varphi$ is countable.
\end{enumerate}
<!-- end lemma-4 -->
<!-- proof-1: body_md -->
Clearly (i) implies (ii) and (ii) implies (iii).  
 
Assume (iii), and enumerate the nonempty set $R_\varphi$ as
$(Y_n)_{n\in\omega}$, with repetitions if necessary. Choose
$X_n\in2^\omega$ such that
$\varphi([X_n]_T)=[Y_n]_T$, using $\mathsf{AC}_{\omega}(\mathbb R)$.
Put $B=\bigoplus_nX_n$ and $\mathbf b=[B]_T$.
Every attained degree is below $\varphi(\mathbf b)$, by
monotonicity. For $\mathbf d\ge_T\mathbf b$,
\[
 \varphi(\mathbf b)\le_T\varphi(\mathbf d)
 \le_T\varphi(\mathbf b),
\]
which gives (i).

<!-- theorem-3: body_md -->
\label{thm:GS}
Let $P\subseteq2^\omega$ be nonempty and perfect, and let
$B\subseteq P$ be countable and dense, with no member eventually
constant in $P$. Suppose a sequence $(G_i)_{i\in\omega}$ of reals
has range containing $B$. For every $X\in2^\omega$, there are
$Y_0,Y_1\in P$ such that
\begin{equation}\label{eq:GS}
 X\le_T\left(\bigoplus_{i\in\omega}G_i\right)\oplus Y_0\oplus Y_1.
\end{equation}
<!-- end theorem-3 -->
<!-- lemma-6: body_md -->
\label{lem:high-cover}
Suppose $C,D\in2^\omega$ satisfy $C\le_T D$ and $C''\le_T D'$.
There is a sequence $(G_i)_{i\in\omega}$ with
$\bigoplus_iG_i\le_T D$ whose range contains every $C$-computable
real in $2^\omega$.
<!-- end lemma-6 -->
<!-- proof-3: body_md -->
We first give the needed direction of Martin's high domination theorem
\cite{Martin1966}, relativized to $C$. Let
\[
 \mathrm{TOT}^{C}
 =\{e:\Phi_e^C\text{ is a total function from }\omega\text{ to }\omega\}.
\]
Since $\mathrm{TOT}^{C}\le_T C''\le_T D'$, the relativized limit
lemma gives a total $D$-computable function
$p:\omega^2\to\{0,1\}$ such that
\[
 \lim_s p(e,s)=\chi_{\mathrm{TOT}^{C}}(e).
\]
For each $n$, let $s_n$ be the least $s\ge n$ such that
\[
 \forall e\le n\,
 \bigl(p(e,s)=1\ \Longrightarrow\ \Phi_e^C(n)[s]\mathord\downarrow\bigr).
\]
This search is computable from $D$, since $C\le_T D$. It terminates:
the finitely many approximations with $e\le n$ eventually stabilize,
after which the indices still marked $1$ all give total computations.
Define
\[
 h(n)=1+\max\bigl(\{n\}\cup
       \{\Phi_e^C(n):e\le n\text{ and }p(e,s_n)=1\}\bigr).
\]
Then $h\le_T D$. If $\Phi_e^C$ is total, choose $N\ge e$ such
that $p(e,s)=1$ for every $s\ge N$. For $n\ge N$, we have
$s_n\ge n\ge N$, so $h(n)>\Phi_e^C(n)$. Thus $h$ eventually
dominates every total $C$-computable function. This argument places
no computable-enumerability assumption on $C$ or $D$.

For each $e\in\omega$ and $\sigma\in2^{<\omega}$, define
\[
 G_{e,\sigma}(n)=
 \begin{cases}
  \sigma(n),&n<|\sigma|,\\
  \Phi_e^C(n),&n\ge|\sigma|\text{ and }\Phi_e^C(n)
       \text{ halts within }h(n)\text{ steps with a binary output},\\
  0,&\text{otherwise}.
 \end{cases}
\]
These are total binary reals uniformly computable from $D$.
If $Z=\Phi_e^C\in2^\omega$, the running-time function of this
computation is total and $C$-computable. Hence there is $N$ such
that the computation on input $n$ halts within $h(n)$ steps whenever
$n\ge N$. Taking $\sigma=Z\upharpoonright N$ gives
$G_{e,\sigma}=Z$. Enumerate the pairs $(e,\sigma)$ in a fixed
computable order to obtain the required sequence.

<!-- corollary-1: body_md -->
\label{cor:perfect-directed}
If $A\subseteq2^\omega$ contains a nonempty perfect set and every
countable subset of $A$ has a Turing upper bound in $A$, then $A$
is cofinal in the Turing degrees.
<!-- end corollary-1 -->
<!-- proof-4: body_md -->
Choose a perfect $P\subseteq A$. The points eventually constant in
$P$ form a countable set: each is the least or greatest point of
$P\cap[s]$ for some $s\in2^{<\omega}$. Every nonempty
$P\cap[s]$ is perfect and uncountable. Using $\mathsf{AC}_{\omega}(\mathbb R)$, choose a
countable dense set $B\subseteq P$ avoiding all these points.
Let $C\in A$ compute every member of $B$.

\emph{First application of Groszek--Slaman.}
For each oracle-program index $e$, put
\[
 H_e(n)=
 \begin{cases}
  \Phi_e^C(n),&\text{if this computation halts with a binary output},\\
  0,&\text{otherwise}.
 \end{cases}
\]
The sequence $(H_e)$ is uniformly $C'$-computable and contains every
$C$-computable binary real, hence contains $B$. Apply
Theorem~\ref{thm:GS} with this sequence and target $C''$ to obtain
$U_0,U_1\in P$ such that
\begin{equation}\label{eq:GS-highness}
 C''\le_T\left(\bigoplus_eH_e\right)\oplus U_0\oplus U_1
       \le_T C'\oplus U_0\oplus U_1.
\end{equation}
Choose an upper bound $D\in A$ for $C,U_0,U_1$. Then
$C\le_T D$ and $C''\le_T D'$. Lemma~\ref{lem:high-cover}
therefore supplies a uniformly $D$-computable sequence $(G_i)$
containing every $C$-computable real, and in particular containing $B$.

\emph{Second application of Groszek--Slaman.}
Given any target $X\in2^\omega$, apply Theorem~\ref{thm:GS}
again, now with $(G_i)$. There are $V_0,V_1\in P$ such that
\[
 X\le_T\left(\bigoplus_iG_i\right)\oplus V_0\oplus V_1
       \le_T D\oplus V_0\oplus V_1.
\]
An upper bound in $A$ for $D,V_0,V_1$ computes $X$. Thus $A$
is cofinal.
<!-- end proof-4 -->
<!-- remark-3: body_md -->
The first application obtains $C''\le_T D'$, not $C'\le_T D$.
Relative highness nevertheless makes a covering sequence computable
from $D$, which is exactly what the second application needs. The
set $A$ need not be closed under Turing jump or under actual joins;
upper bounds in $A$ suffice. Also, we choose the dense set $B$
before choosing its bound $C$, so the restriction on eventually
constant branches is satisfied. The argument does not require the
stronger coding statement for an arbitrary prescribed dense set.

<!-- proposition-1: body_md -->
\label{prop:factorization}
Assume $\mathsf{PSP}_{\mathbb R}+\mathsf{CU}$, and let $\varphi:\mathcal D_T\to\mathcal D_T$ have
uncountable saturated real range. There are continuous functions
$\alpha,\beta:2^\omega\to2^\omega$, with $\beta$ injective,
such that
\begin{equation}\label{eq:factorization}
 \varphi([\alpha(X)]_T)=[\beta(X)]_T
 \qquad(X\in2^\omega).
\end{equation}
No order-preservation assumption is needed in this proposition.
<!-- end proposition-1 -->
<!-- proof-5: body_md -->
Choose a perfect $P\subseteq R_\varphi$ and a continuous
injection $h:2^\omega\to P$, using
Lemma~\ref{lem:cantor-subset}. The relation
\[
 S(X,U)\quad\Longleftrightarrow\quad
 \varphi([U]_T)=[h(X)]_T
\]
has a nonempty section at every $X$, because $h(X)\in R_\varphi$.
By $\mathsf{CU}$, there are a comeager $G_\delta$ set $C$ and a
continuous $g:C\to2^\omega$ with
\[
 \varphi([g(X)]_T)=[h(X)]_T\qquad(X\in C).
\]
Choose a continuous injection $q:2^\omega\to C$ and set
$\alpha=g\circ q$ and $\beta=h\circ q$. These maps have the
required properties.
<!-- end proof-5 -->
<!-- proposition-2: body_md -->
\label{prop:fixed-code}
Assume $\mathsf{PSP}_{\mathbb R}+\mathsf{CU}$. If $\varphi$ is order preserving and
$R_\varphi$ is uncountable, there is $\mathbf k$ satisfying
\eqref{eq:main-code}.
<!-- end proposition-2 -->
<!-- proof-6: body_md -->
Take $\alpha,\beta$ from Proposition~\ref{prop:factorization}.
Since $2^\omega$ is compact and $\beta$ is continuous and
injective, its inverse is continuous on the compact image
$\beta[2^\omega]$. Apply Lemma~\ref{lem:code} to $\alpha$
and to this inverse, and join the two codes into a single real
$K$. Then
\begin{equation}\label{eq:two-codes}
 \alpha(X)\le_T K\oplus X,
 \qquad X\le_T K\oplus\beta(X)
 \qquad(X\in2^\omega).
\end{equation}
Put $\mathbf k=[K]_T$.

Fix $\mathbf d\ge_T\mathbf k$, and take any representative
$X\in\mathbf d$. The first inequality in \eqref{eq:two-codes}
gives $[\alpha(X)]_T\le_T\mathbf d$. Hence, by monotonicity
and \eqref{eq:factorization},
\[
 [\beta(X)]_T=\varphi([\alpha(X)]_T)
 \le_T\varphi(\mathbf d).
\]
The second inequality in \eqref{eq:two-codes} now yields
\[
 \mathbf d\le_T\mathbf k\vee[\beta(X)]_T
 \le_T\mathbf k\vee\varphi(\mathbf d).
\]
<!-- end proof-6 -->
<!-- corollary-2: body_md -->
\label{cor:fibers}
Under the hypotheses of Proposition~\ref{prop:fixed-code}, fix
$\mathbf a\ge_T\mathbf k$. For every degree $\mathbf v$, the set
in \eqref{eq:main-fiber} is at most countable. In particular, any partial
real representative of $X\mapsto\varphi(\mathbf a\vee[X]_T)$
has countable point fibers.
<!-- end corollary-2 -->
<!-- proof-7: body_md -->
Fix $Y\in\mathbf v$. Whenever
$\varphi(\mathbf a\vee[X]_T)=\mathbf v$, apply
\eqref{eq:main-code} to $\mathbf a\vee[X]_T$ to obtain
\[
 [X]_T\le_T\mathbf a\vee[X]_T
 \le_T\mathbf k\vee\mathbf v.
\]
Thus every such $X$ is computable from $K\oplus Y$, and the
set of those reals is countable.

<!-- lemma-7: body_md -->
\label{lem:absorption}
Suppose $\varphi$ is order preserving,
$\mathbf d\le_T\mathbf k\vee\varphi(\mathbf d)$ whenever
$\mathbf d\ge_T\mathbf k$, and there is a degree $\mathbf e$
with $\mathbf k\le_T\varphi(\mathbf e)$. Then
\[
 \mathbf d\le_T\varphi(\mathbf d)
 \qquad(\mathbf d\ge_T\mathbf e\vee\mathbf k).
\]
<!-- end lemma-7 -->
<!-- proof-8: body_md -->
For such $\mathbf d$, monotonicity gives
$\mathbf k\le_T\varphi(\mathbf e)\le_T\varphi(\mathbf d)$.
Therefore
$\mathbf d\le_T\mathbf k\vee\varphi(\mathbf d)=\varphi(\mathbf d)$.
<!-- end proof-8 -->
<!-- proof-9: body_md -->
If $R_\varphi$ is countable, Lemma~\ref{lem:range} gives
constancy on a cone. Suppose it is uncountable. Proposition
\ref{prop:fixed-code} gives $\mathbf k$ and
\eqref{eq:main-code}; Corollary~\ref{cor:fibers} gives the
fiber assertion. By $\mathsf{PSP}_{\mathbb R}$, the set $R_\varphi$ contains a
perfect subset, and Lemma~\ref{lem:directed} makes it countably
directed. Corollary~\ref{cor:perfect-directed} therefore makes
$R_\varphi$, and hence the degree range of $\varphi$, cofinal.
Choose $\mathbf e$ with $\mathbf k\le_T\varphi(\mathbf e)$ and put
$\mathbf b=\mathbf e\vee\mathbf k$. Then $\mathbf b\ge_T\mathbf k$,
and Lemma~\ref{lem:absorption} proves \eqref{eq:main-domination}
on $\mathcal C_{\mathbf b}$, establishing (iv) and the asserted dichotomy.

<!-- definition-3: body_md -->
\label{def:OP}
The principle $\mathsf{OP}_{\mathrm{real}}$ asserts that for every total Turing-invariant,
order-preserving function $f:2^\omega\to2^\omega$, at least one of
the following holds:
\begin{enumerate}[label=(\roman*)]
 \item there are reals $B,Y$ such that
 $f(X)\equiv_T Y$ for every $X\ge_T B$;
 \item there is a real $B$ such that
 $X\le_T f(X)$ for every $X\ge_T B$.
\end{enumerate}
The principle $\mathsf{OP}_{\mathrm{deg}}$ is the corresponding assertion for every
order-preserving map $\varphi:\mathcal D_T\to\mathcal D_T$.
<!-- end definition-3 -->
<!-- theorem-5: body_md -->
\label{thm:inaccessible-reals}
In $\mathsf{ZF}$,
\begin{equation}\label{eq:inaccessible-reals}
 \mathsf{OP}_{\mathrm{real}}\quad\Longrightarrow\quad
 (\forall r\in2^\omega)\ \omega_1^{L[r]}<\omega_1.
\end{equation}
<!-- end theorem-5 -->
<!-- proof-12: body_md -->
Suppose that $r\in2^\omega$ satisfies
\[
 \kappa=\omega_1^{L[r]}=\omega_1^V,
\]
and put $M=L[r]$. We construct a counterexample to $\mathsf{OP}_{\mathrm{real}}$
on all reals of $V$.

\smallskip\noindent
\emph{The sequence chosen in $M$.}
Work inside $M$. Since every $0<\alpha<\kappa$ is countable
there, use choice in $M$ to fix surjections
$e_\alpha:\omega\to\alpha$ in $L[r]$. Define by transfinite recursion
\[
 A_0=0^\omega,\qquad
 A_\alpha=\left(\bigoplus_{n\in\omega}
                  A_{e_\alpha(n)}\right)'
       \quad(0<\alpha<\kappa),
\]
where the prime denotes the Turing jump. Every earlier real is
computable from the join at stage $\alpha$, and the jump is
strictly above that join. Thus
\begin{equation}\label{eq:old-chain}
 A_\alpha<_T A_\beta\qquad(\alpha<\beta<\kappa).
\end{equation}
Turing reducibility between these fixed reals is arithmetic and
hence absolute between $M$ and $V$. Therefore
\eqref{eq:old-chain} continues to hold in $V$. The entire sequence
belongs to $M$; no choice indexed by the ambient reals is needed.

\smallskip\noindent
\emph{The function on all ambient reals.}
For $X\in(2^\omega)^V$, define
\begin{equation}\label{eq:least-missing}
 \alpha_X=\min\{\alpha<\kappa:A_\alpha\nleq_T X\},
 \qquad F(X)=A_{\alpha_X}.
\end{equation}
The minimum exists due to uncountability of $\kappa$. Definition~\eqref{eq:least-missing} therefore
gives a total real function in $V$, including at inputs outside $M$.
It uses only the fixed sequence and least ordinals.

\smallskip\noindent
\emph{Order preservation and Turing invariance.}
If $X\le_T Y$, every $A_\beta$ with $\beta<\alpha_X$ is computable
from $X$ and hence from $Y$. Thus $\alpha_X\le\alpha_Y$, and the
increasing chain gives $F(X)\le_T F(Y)$. If $X\equiv_T Y$, the
sets of indices computed by the two inputs coincide; in fact
$\alpha_X=\alpha_Y$ and $F(X)=F(Y)$.

\smallskip\noindent
\emph{Failure of both cone conclusions.}
By its definition,
\begin{equation}\label{eq:output-not-below-input}
 F(X)\nleq_T X\qquad(X\in2^\omega).
\end{equation}
If $F$ were constant in degree with value $[Y]_T$ on the cone
above $[B]_T$, take $X=B\oplus Y$. Then
$F(X)\equiv_T Y\le_T X$, contradicting
\eqref{eq:output-not-below-input}.

Suppose next that $X\le_T F(X)$ for every $X\ge_T B$, for some
real $B$. Put $\alpha=\alpha_B$. Evaluating at $B$ gives
$B\le_T F(B)=A_\alpha$. Since
$A_\alpha<_T A_{\alpha+1}$, choose inside $M$ a real
$X\ge_T A_\alpha$ incomparable with $A_{\alpha+1}$ (see \cite{Sacks1961}).
These relations remain true in $V$. In particular, $X\ge_T B$,
every $A_\beta$ with $\beta\le\alpha$ is computable from $X$,
and $A_{\alpha+1}$ is not. Hence
\begin{equation}\label{eq:input-not-below-output}
 \alpha_X=\alpha+1,\qquad F(X)=A_{\alpha+1},\qquad
 X\nleq_T F(X).
\end{equation}
This contradicts the assumed domination on the cone above $B$.
Thus $F$ satisfies neither conclusion of $\mathsf{OP}_{\mathrm{real}}$.

Finally, $\omega_1^{L[r]}\le\omega_1^V$ always holds, since an
ordinal countable in $L[r]$ is countable in $V$. The equality case
having been excluded, \eqref{eq:inaccessible-reals} follows.
