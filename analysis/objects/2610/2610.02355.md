<!-- assumption-1: body_md -->
\label{as:l_smooth}
The objective function $f$ is globally $L$-smooth with $L>0$ such that:
$$
\|\nabla f(x)-\nabla f(y)\|\le L\|x-y\|
\qquad\text{for all }x,y\in\mathbb{R}^d.
$$
<!-- end assumption-1 -->
<!-- assumption-2: body_md -->
\label{as:bg_a_oracle}
Given $a\ge0$, for every $x \in \mathbb{R}^d$, the stochastic gradient oracle is unbiased, and there exist constants $B,G \ge 0$ satisfying:
$$
\mathbb{E}_{\xi}\bigl[\nabla f(x;\xi)\bigr] = \nabla f(x),
\qquad
\mathbb{E}_{\xi}
\left\|\nabla f(x;\xi)-\nabla f(x)\right\|^2
\le B^2\left\lVertx-x_0\right\rVert^a+G^2.
$$
<!-- end assumption-2 -->
<!-- theorem-1: body_md -->
\label{thm:main_lower_bound}
Fix any finite $a\ge0$. Under Assumption~\ref{as:l_smooth}, for all ${\epsilon}>0$ satisfying ${\epsilon}^2\le c\,L\Delta$, where $c>0$ is an absolute constant specified in Appendix~\ref{app:lower-bound}, every algorithm that returns a point $\widehat x$ satisfying $\mathbb{E}\left\lVert\nabla f(\widehat x)\right\rVert^2\le{\epsilon}^2$ for every unbiased BG-$a$ oracle must incur worst-case expected SFO complexity
\begin{equation}\label{eq:main_lower_rate}
    \Omega_a\!\left(
    \frac{L\Delta}{{\epsilon}^2}
    +\frac{G^2L\Delta}{{\epsilon}^4}
    +\frac{B^2L\Delta^{a+1}}{{\epsilon}^{4+a}}
    \right).
\end{equation}
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:main_sgd}
Suppose Assumptions~\ref{as:l_smooth} and~\ref{as:bg_a_oracle} hold with $0\le a\le2$. Fix $\sigma^2>0$ and let $0<{\epsilon}^2\le\min\{2\sigma^2,L\Delta\}$. At iteration $t$, use
\begin{equation}\label{eq:main_sgd_parameters}
    b_t=\max\!\left\{1,
    \left\lceil\frac{B^2\left\lVertx_t-x_0\right\rVert^a+G^2}{\sigma^2}\right\rceil\right\},
    \qquad
    \eta=\frac{{\epsilon}^2}{2L\sigma^2},
    \qquad
    T=\left\lceil\frac{4\Delta}{\eta{\epsilon}^2}\right\rceil,
\end{equation}
and update the iterates using $x_{t+1}=x_t-\eta g_t$, where $g_t$ is the minibatch gradient in equation~\ref{eq:main_minibatch}. Then equation~\ref{eq:main_stationarity} holds with the expected SFO cost given as
\begin{equation}\label{eq:main_sgd_cost}
    \mathbb{E}\!\left[\sum_{t=0}^{T-1}b_t\right]
    =\mathcal{O}\!\left(
    L\Delta(\sigma^2+G^2){\epsilon}^{-4}
    +B^2L\Delta^{a+1}{\epsilon}^{-(4+a)}
    \right).
\end{equation}
In particular, for fixed $\sigma^2$, the expected SFO complexity is $\mathcal{O}({\epsilon}^{-(4+a)})$. The complete theorem statement and proof are provided in Appendix~\ref{sec:convergence_of_sgd}.
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm:main_sgdm}
Suppose Assumptions~\ref{as:l_smooth} and~\ref{as:bg_a_oracle} hold with $0\le a\le2$. Fix $\beta\in[0,1)$ and $\sigma^2>0$, and let $c_{\beta}=\frac{1-\beta}{1+\beta}$ and $0<{\epsilon}^2\le\min\{8\sigma^2(1-\beta^2),L\Delta\}$. At iteration $t$, use
\begin{equation}\label{eq:main_sgdm_parameters}
\begin{aligned}
    b_t&=\max\!\left\{1,
    \left\lceil\frac{c_{\beta}(B^2\left\lVertx_t-x_0\right\rVert^a+G^2)}{\sigma^2}\right\rceil\right\},\\
    \eta&=\frac{c_{\beta}{\epsilon}^2}{16L\sigma^2},
    \qquad
    T=\left\lceil\frac{16\Delta}{\eta{\epsilon}^2}\right\rceil,
\end{aligned}
\end{equation}
and, with $g_t$ as in equation~\ref{eq:main_minibatch}, update the iterates using
\begin{equation}\label{eq:main_sgdm_update}
    m_{-1}=0,
    \qquad
    m_t=\beta m_{t-1}+(1-\beta)g_t,
    \qquad
    x_{t+1}=x_t-\eta m_t.
\end{equation}
Then equation~\ref{eq:main_stationarity} holds, and
\begin{equation}\label{eq:main_sgdm_cost}
    \mathbb{E}\!\left[\sum_{t=0}^{T-1}b_t\right]
    =\mathcal{O}\!\left(
    L\Delta\left(\frac{\sigma^2}{c_{\beta}}+G^2\right){\epsilon}^{-4}
    +B^2L\Delta^{a+1}{\epsilon}^{-(4+a)}
    \right).
\end{equation}
Thus, for fixed $\beta$ and $\sigma^2$, SGDM also uses $\mathcal{O}({\epsilon}^{-(4+a)})$ expected SFO calls. The complete theorem statement and proof are provided in Appendix~\ref{sec:convergence_of_sgdm}.

<!-- definition-1: body_md -->
\label{def:lower_progress}
For $w\in\mathbb{R}^T$ and $\gamma\ge0$, define
$$
\operatorname{prog}_\gamma(w)
:=\max\bigl(\{i\in\{1,\ldots,T\}:|w_i|>\gamma\}\cup\{0\}\bigr).
$$
This is the largest coordinate index whose absolute value exceeds the threshold, and is defined to be zero if no such coordinate exists. E.g., $\operatorname{prog}_1(w)=T$ means that $|w_T|>1$, without imposing any condition on the preceding coordinates.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:lower_zero_respecting}
Let $\operatorname{supp}(v):=\{i:v_i\ne0\}$. Suppose we initialize with $x_0=0$ and $g_t$ is the stochastic gradient returned at query $x_t$. An algorithm is zero-respecting if, almost surely, each query made before termination satisfies
$$
\operatorname{supp}(x_t)
\subseteq\bigcup_{j=0}^{t-1}\operatorname{supp}(g_j).
$$
For the lower bound in Stage 1 below, we also require its output after $\tau$ oracle calls to satisfy
$$
\operatorname{supp}(\widehat x)
\subseteq\bigcup_{j=0}^{\tau-1}\operatorname{supp}(g_j).
$$
These restrictions allow a coordinate to become nonzero only after it has appeared in an oracle response.
<!-- end definition-2 -->
<!-- definition-3: body_md -->
\label{def:lower_chain}
For $T\ge1$ and $w\in\mathbb{R}^T$, define
\begin{equation}\label{eq:lower_chain_definition}
\bar f_T(w):=-\Psi(1)\Phi(w_1)
+\sum_{i=2}^T\bigl[\Psi(-w_{i-1})\Phi(-w_i)-\Psi(w_{i-1})\Phi(w_i)\bigr],
\end{equation}
where
$$
\Psi(r):=\begin{cases}
0,&r\le1/2,\\
\exp\!\left(1-\dfrac1{(2r-1)^2}\right),&r>1/2,
\end{cases}
\qquad
\Phi(r):=\sqrt{e}\int_{-\infty}^r e^{-z^2/2}\,dz.
$$
These scalar functions define the base chain used in the construction of the lower bound.
<!-- end definition-3 -->

\label{lem:lower_chain_properties}

\qquad \left\lVert\nabla\bar f_T(w)\right\rVert_\infty>1.
$$
\end{enumerate}

<!-- definition-4: body_md -->
\label{def:lower_masked_oracle}
Fix $p\in(0,1]$ and draw $\xi\sim\mathrm{Bernoulli}(p)$. For $w\in\mathbb{R}^T$, we define $\bar g_T(w;\xi)\in\mathbb{R}^T$ (coordinate wise) by
\begin{equation}\label{eq:lower_masked_definition}
[\bar g_T(w;\xi)]_i
:=\partial_i\bar f_T(w)
\left[1+\mathbf1_{\{i>\operatorname{prog}_{1/4}(w)\}}(\xi/p-1)\right],
\qquad i=1,\ldots,T.
\end{equation}
Each oracle call uses a fresh independent seed. Coordinates beyond the progress index are suppressed when $\xi=0$ and multiplied by $1/p$ when $\xi=1$.

<!-- lemma-3: body_md -->
\label{lem:lower_masked_properties}
For every $w\in\mathbb{R}^T$, the oracle in Definition~\ref{def:lower_masked_oracle} satisfies
\begin{equation}\label{eq:chainvariance}
\mathbb{E}_\xi[\bar g_T(w;\xi)]=\nabla\bar f_T(w),\qquad
\mathbb{E}_\xi\left\lVert\bar g_T(w;\xi)-\nabla\bar f_T(w)\right\rVert^2
\le\varsigma^2\frac{1-p}{p}.
\end{equation}
Writing $k=\operatorname{prog}_{1/4}(w)$, for both possible seeds its support satisfies
$$
\operatorname{supp}(\bar g_T(w;\xi))
\subseteq\{1,\ldots,\min(T,k+1)\},\qquad
\operatorname{supp}(\bar g_T(w;0))\subseteq\{1,\ldots,k\}.
$$
Thus, a response can extend progress by at most one coordinate, and such an extension requires $\xi=1$. For each fixed seed, these support statements hold simultaneously for every $w$, yielding the probability-$p$ zero-chain property used in \cite{arjevani2023lower}.

Since $\mathbb{E}_\xi[\xi/p]=1$, we achieve unbiasedness by taking expectation in equation~\ref{eq:lower_masked_definition}. Furthermore, $\operatorname{prog}_{1/2}(w)\le k$, hence, Lemma~\ref{lem:lower_chain_properties} implies $\partial_i\bar f_T(w)=0$ for $i>k+1$. Using $\mathbb{E}_\xi[(\xi/p-1)^2]=(1-p)/p$ and applying the gradient bound in Lemma~\ref{lem:lower_chain_properties} gives

\label{thm:lower_bound}
Under fixed $L,\Delta>0$, $B,G\ge0$, and a finite $a\ge0$, let $\ell=155$ and $\varsigma=23$, and suppose
\begin{equation}\label{eq:small}
0<{\epsilon}^2\le\frac{L\Delta}{98304\ell}.
\end{equation}
Any algorithm that returns $\widehat x$ after a random number $\tau$ of SFO calls and satisfies
\begin{equation}\label{eq:criterion}
\mathbb{E}\left\lVert\nabla f(\widehat x)\right\rVert\le{\epsilon}
\end{equation}
for every admissible instance must satisfy
\begin{equation}\label{eq:explicit}
\sup_{(f,g)}\mathbb{E}[\tau]
\ge\frac{L\Delta}{786432\ell{\epsilon}^2}
\left[1+\frac{G^2+B^2(\Delta/(16{\epsilon}))^a}
{64\varsigma^2{\epsilon}^2}\right].
\end{equation}
Consequently, the worst-case expected SFO complexity is
\begin{equation}\label{eq:rate}
\Omega_a\!\left(\frac{L\Delta}{{\epsilon}^2}
+\frac{G^2L\Delta}{{\epsilon}^4}
+\frac{B^2L\Delta^{a+1}}{{\epsilon}^{4+a}}\right).
\end{equation}

\textbf{Proof roadmap and relation to prior works.}
We first explain the roadmap that we follow for the proof and explain its relation to \cite{arjevani2023lower} and PASTA \citep{fazla2026lower}. We follow PASTA's travel coordinate and activation construction. This choice is intuitive, as PASTA's variance assumption (BG-$0$) is a special case of the BG-$a$ noise model with $a=2$, and its extension to general $a$ is our goal. The travel coordinate forces points with sufficiently small gradient to lie at distance of order $\Delta/{\epsilon}$ from initialization, whereas the activation function keeps the noisy chain inactive near initialization and allows it to become active only where the distance-dependent variance budget is sufficiently large. Note that, this is a geometric requirement, as an unrestricted algorithm may jump to a distant point in one query. The established oracle complexity comes from learning the chain coordinates, rather than from the number of steps needed to travel that distance. Note that, the proof strategies in \cite{fazla2026lower} are mostly built on top of \cite{arjevani2023lower}.

In our extension to BG-$a$, we replace PASTA's quadratic activation variance budget by $B^2(D/2)^a+G^2$ and recalibrate the Bernoulli reveal probability $p$. The chain length remains of order $L\Delta/{\epsilon}^2$, while each coordinate revelation requires order $1/p$ calls, with
$$
\frac1p=1+\Theta_a\!\left(\frac{G^2+B^2(\Delta/{\epsilon})^a}{{\epsilon}^2}\right).
$$
Multiplying these two scales gives the claimed rate. We first establish this mechanism for zero-respecting algorithms, then give the compressed random-rotation construction and the random-stopping argument needed for unrestricted randomized algorithms.

\textbf{Stage 1: Construction and lower bound for zero-respecting algorithms.}
Let $D=\Delta/(4{\epsilon})$ and define $f_0(0)=0$ with
\begin{equation}\label{eq:pasta-travel}
f_0'(u)=\begin{cases}
-2{\epsilon},&u\le D,\\
-2{\epsilon}+\frac L2(u-D),&D<u<D+4{\epsilon}/L,\\
0,&u\ge D+4{\epsilon}/L.
\end{cases}
\end{equation}
Writing $f_0^{\inf}:=\inf_u f_0(u)$, integration gives $f_0^{\inf}=-\Delta/2-4{\epsilon}^2/L$. Next, we define the twice continuously differentiable activation
\begin{equation}\label{eq:pasta-activation}
\phi(u)=\begin{cases}
0,&u\le D/2,\\
10t^3-15t^4+6t^5,&D/2<u<D,\quad t=2u/D-1,\\
1,&u\ge D,
\end{cases}
\end{equation}
which satisfies $0\le\phi\le1$, $\phi'\ge0$, and
$$
\left\lVert\phi'\right\rVert_\infty=\frac{15{\epsilon}}{\Delta},\qquad
\left\lVert\phi''\right\rVert_\infty=\frac{640{\epsilon}^2}{\sqrt3\Delta^2}.
$$

Use the chain $\bar f_T$ and oracle $\bar g_T$ from Definitions~\ref{def:lower_chain} and~\ref{def:lower_masked_oracle}, with $M_T=\sup_w\bar f_T(w)$. Lemmas~\ref{lem:lower_chain_properties}--\ref{lem:lower_masked_properties} provide the smoothness, range, gradient, and variance bounds used below.

Using the same scaling as in \cite{fazla2026lower}, we set
\begin{equation}\label{eq:pasta-scaled}
f_{\mathrm{scaled}}(y)=\frac{L\lambda^2}{2\ell_1}
[\bar f_T(y/\lambda)-M_T],\qquad
\lambda=\frac{4\ell_1{\epsilon}}{L},\qquad
T=\left\lfloor\frac{L\Delta}{768\ell_1{\epsilon}^2}\right\rfloor.
\end{equation}
Then $-\Delta/4\le f_{\mathrm{scaled}}\le0$,
$\operatorname{Lip}(\nabla f_{\mathrm{scaled}})\le L/2$, and
$\nabla f_{\mathrm{scaled}}(y)=2{\epsilon}\nabla\bar f_T(y/\lambda)$.
Hence, subtracting $M_T$ makes the scaled chain nonpositive. Consequently, since $\phi'\ge0,$ the activation contribution $\phi'(u)f_{\mathrm{scaled}}(y)$ cannot cancel the negative travel-coordinate gradient.
Defining
\begin{equation}\label{eq:pasta-objective}
f(u,y)=f_0(u)+\phi(u)f_{\mathrm{scaled}}(y),\qquad x_0=(0,0),
\end{equation}
and writing $x=(u,y)$ and $f^{\inf}:=\inf_x f(x)$, we get $f(x_0)=0$ and
$$
f(x_0)-f^{\inf}\le\frac{3\Delta}{4}+\frac{4{\epsilon}^2}{L}\le\Delta.
$$

The block-diagonal part of $\nabla^2f$ has operator norm at most
$L/2+\left\lVert\phi''\right\rVert_\infty\left\lVertf_{\mathrm{scaled}}\right\rVert_\infty$.
The off-diagonal part has operator norm $\left\lVert\phi'\nabla f_{\mathrm{scaled}}\right\rVert$, and
$\left\lVert\nabla f_{\mathrm{scaled}}\right\rVert\le46{\epsilon}\sqrt T$.
Consequently,
\begin{equation}\label{eq:pasta-smoothness}
\left\lVert\nabla^2f\right\rVert_{\mathrm{op}}
\le\frac L2+\frac{160{\epsilon}^2}{\sqrt3\Delta}
+\frac{690{\epsilon}\sqrt L}{\sqrt{768\ell_1\Delta}}
\le L.
\end{equation}
Indeed, ${\epsilon}^2\le L\Delta/(1536\ell_1)$, which follows from equation~\ref{eq:small}, bounds each of the last two terms by $L/4$. The gradient is continuous at the junctions of $f_0'$, so integration along line segments extends the bound to global $L$-smoothness.

Next, observe that the composite gradient is
$$
\nabla f(u,y)=\begin{pmatrix}
f_0'(u)+\phi'(u)f_{\mathrm{scaled}}(y)\\
\phi(u)\nabla f_{\mathrm{scaled}}(y)
\end{pmatrix}.
$$
We consider two cases to identify necessary conditions for $\left\lVert\nabla f(u,y)\right\rVert<2{\epsilon}$.

First, if $u\le D$, then $f_0'(u)=-2{\epsilon}$. Since $\phi'(u)\ge0$ and $f_{\mathrm{scaled}}(y)\le0$, we have
$$
\partial_u f(u,y)
=-2{\epsilon}+\phi'(u)f_{\mathrm{scaled}}(y)
\le-2{\epsilon}.
$$
Therefore, $\left\lVert\nabla f(u,y)\right\rVert\ge|\partial_u f(u,y)|\ge2{\epsilon}$. Thus, a gradient norm strictly below $2{\epsilon}$ requires $u>D$.

Second, suppose $u>D$ but $\operatorname{prog}_1(y/\lambda)<T$. Then $\phi(u)=1$, and the scaling in equation~\ref{eq:pasta-scaled} together with the gradient lower bound in Lemma~\ref{lem:lower_chain_properties} gives
$$
\begin{aligned}
\left\lVert\nabla f(u,y)\right\rVert
&\ge\left\lVert\nabla_y f(u,y)\right\rVert\\
&=\left\lVert\nabla f_{\mathrm{scaled}}(y)\right\rVert\\
&=2{\epsilon}\left\lVert\nabla\bar f_T(y/\lambda)\right\rVert\\
&\ge2{\epsilon}\left\lVert\nabla\bar f_T(y/\lambda)\right\rVert_\infty
>2{\epsilon}.
\end{aligned}
$$
This also rules out a gradient norm strictly below $2{\epsilon}$. Since $\operatorname{prog}_1(y/\lambda)\in\{0,\ldots,T\}$, the two cases imply
\begin{equation}\label{eq:pasta-stationarity}
\left\lVert\nabla f(u,y)\right\rVert<2{\epsilon}
\quad\Longrightarrow\quad u>D\ \text{and}\ \operatorname{prog}_1(y/\lambda)=T.
\end{equation}

We now choose a constant variance budget that is admissible wherever the chain is active. Recall that $x=(u,y)$, $x_0=(0,0)$, and $\phi(u)=0$ for $u\le D/2$. Thus, whenever $\phi(u)>0$,
$$
u>D/2>0,\qquad
\left\lVertx-x_0\right\rVert=\sqrt{u^2+\left\lVerty\right\rVert^2}\ge u>D/2.
$$
Since $r\mapsto r^a$ is nondecreasing for $a\ge0$, the BG-$a$ variance bound throughout this active region satisfies
$$
B^2\left\lVertx-x_0\right\rVert^a+G^2\ge B^2(D/2)^a+G^2.
$$
We therefore define the constant budget
\begin{equation}\label{eq:pasta-variance-budget}
\widetilde\sigma_a^2 :=B^2(D/2)^a+G^2
=B^2\left(\frac{\Delta}{8{\epsilon}}\right)^a+G^2,
\end{equation}
where we use $D=\Delta/(4{\epsilon})$ in the last equality. Note that, this budget is a permissible upper bound for the noise we introduce on the active region, it is not a lower bound on the oracle's actual variance. Thus, we construct the oracle below so that its variance is at most $\phi(u)^2\widetilde\sigma_a^2$. Since $0\le\phi\le1$, the factor $\phi(u)^2$ makes the noise vanish on the inactive region and keeps it within the chosen budget on the active region.

As the next step, we keep the travel-coordinate component of the gradient exact and introduce randomness only through the chain estimator $\bar g_T$. Its Bernoulli seed $\xi\sim\mathrm{Bernoulli}(p)$ permits the next unrevealed chain coordinate to be revealed only when $\xi=1$. The factor $\xi/p$ in the masked estimator preserves its expectation, while a smaller $p$ increases its variance through the factor $(1-p)/p$ in equation~\ref{eq:chainvariance}. We define the composite oracle and choose $p$ to match this variance bound to the budget above:
\begin{equation}\label{eq:pasta-oracle}
\nabla f(u,y;\xi)=\begin{pmatrix}
f_0'(u)+\phi'(u)f_{\mathrm{scaled}}(y)\\
2{\epsilon}\phi(u)\bar g_T(y/\lambda;\xi)
\end{pmatrix},\qquad
p=\frac{4\varsigma^2{\epsilon}^2}
{4\varsigma^2{\epsilon}^2+\widetilde\sigma_a^2}.
\end{equation}
To verify conditional unbiasedness, let $x_t=(u_t,y_t)$ be a query measurable with respect to $\mathcal{F}_t$, and draw $\xi_t\sim\mathrm{Bernoulli}(p)$ independently of $\mathcal{F}_t$. Conditioned on this history, the query is fixed, so equation~\ref{eq:chainvariance} gives
$$
\mathbb{E}_t[\bar g_T(y_t/\lambda;\xi_t)]
=\nabla\bar f_T(y_t/\lambda).
$$
Consequently, using $\nabla f_{\mathrm{scaled}}(y_t)=2{\epsilon}\nabla\bar f_T(y_t/\lambda)$,
$$
\begin{aligned}
\mathbb{E}_t[\nabla f(x_t;\xi_t)]
&=\begin{pmatrix}
f_0'(u_t)+\phi'(u_t)f_{\mathrm{scaled}}(y_t)\\
2{\epsilon}\phi(u_t)\nabla\bar f_T(y_t/\lambda)
\end{pmatrix}\\
&=\nabla f(x_t).
\end{aligned}
$$
The oracle error has a zero travel-coordinate component and can be written explicitly as
$$
e_t:=\nabla f(x_t;\xi_t)-\nabla f(x_t)
=\begin{pmatrix}
0\\
2{\epsilon}\phi(u_t)\bigl[\bar g_T(y_t/\lambda;\xi_t)-\nabla\bar f_T(y_t/\lambda)\bigr]
\end{pmatrix}.
$$
Taking its conditional second moment and applying equation~\ref{eq:chainvariance} yields
$$
\begin{aligned}
\mathbb{E}_t\left\lVerte_t\right\rVert^2
&=4{\epsilon}^2\phi(u_t)^2
\mathbb{E}_t\left\lVert\bar g_T(y_t/\lambda;\xi_t)-\nabla\bar f_T(y_t/\lambda)\right\rVert^2\\
&\le4{\epsilon}^2\phi(u_t)^2\varsigma^2\frac{1-p}{p}.
\end{aligned}
$$
On the other hand, the choice of $p$ in equation~\ref{eq:pasta-oracle} lies in $(0,1]$ and satisfies
$$
\frac{1-p}{p}
=\frac{\widetilde\sigma_a^2}{4\varsigma^2{\epsilon}^2}.
$$
Thus, the conditional variance satisfies the desired bound
$$
\mathbb{E}_t\left\lVert\nabla f(x_t;\xi_t)-\nabla f(x_t)\right\rVert^2
\le\phi(u_t)^2\widetilde\sigma_a^2.
$$
For $u_t\le D/2$, we have $\phi(u_t)=0$, so the conditional variance is zero. For $u_t>D/2$, the active-region bound above gives
$$
\phi(u_t)^2\widetilde\sigma_a^2
\le\widetilde\sigma_a^2
=B^2(D/2)^a+G^2
\le B^2\left\lVertx_t-x_0\right\rVert^a+G^2.
$$
Hence, conditional unbiasedness and the BG-$a$ variance bound hold at every query and for every $a\ge0$, including those in the activation interval $D/2<u_t<D$. To recover the BG-$0$ case, setting $a=2$ in equation~\ref{eq:pasta-variance-budget} gives $\widetilde\sigma_2^2=B^2\Delta^2/(64{\epsilon}^2)+G^2$. Substituting this expression into $p$ and multiplying its numerator and denominator by $64{\epsilon}^2$ gives
$$
p=\frac{256\varsigma^2{\epsilon}^4}
{B^2\Delta^2+64G^2{\epsilon}^2+256\varsigma^2{\epsilon}^4},
$$
recovering the choice in \cite{fazla2026lower}.

We now translate the stationarity requirement into a lower bound on the number of oracle calls. Consider a zero-respecting algorithm as in Definition~\ref{def:lower_zero_respecting}, with output written as $\widehat x=(\widehat u,\widehat y)$, and define the event $A:=\{\left\lVert\nabla f(\widehat x)\right\rVert<2{\epsilon}\}.$
By equation~\ref{eq:criterion} and Markov's inequality, we have
$$
\Pr(A^c)
=\Pr\bigl(\left\lVert\nabla f(\widehat x)\right\rVert\ge2{\epsilon}\bigr)
\le\frac{\mathbb{E}\left\lVert\nabla f(\widehat x)\right\rVert}{2{\epsilon}}
\le\frac12,
$$
so $\Pr(A)\ge1/2$. Within the event $A$, equation~\ref{eq:pasta-stationarity} requires $\operatorname{prog}_1(\widehat y/\lambda)=T$, and in particular the $T$-th chain coordinate of the output must be nonzero. The zero-respecting restriction requires this coordinate to have been revealed before the algorithm outputs it. Starting from $y_0=0$, Lemma~\ref{lem:lower_masked_properties} implies that the masked chain oracle can reveal at most one new chain coordinate per oracle call, and only when the Bernoulli seed equals one. Thus, reaching the $T$-th coordinate requires at least $T$ Bernoulli successes.

Next, we define $\tau$ as the number of oracle calls and denote the number of successes before termination by
$$
S_\tau:=\sum_{t=0}^{\tau-1}\xi_t.
$$
The preceding argument gives $S_\tau\ge T\mathbf1_A$, hence
$$
\mathbb{E}[S_\tau]\ge T\Pr(A)\ge T/2.
$$
Observing that the event $\{\tau>t\}$ means that call $t$ is made and is $\mathcal{F}_t$-measurable and using independence, we obtain
$$
\begin{aligned}
\mathbb{E}[\mathbf1_{\{\tau>t\}}\xi_t]
&=\mathbb{E}\bigl[\mathbf1_{\{\tau>t\}}\mathbb{E}[\xi_t\mid\mathcal{F}_t]\bigr]\\
&=p\Pr(\tau>t).
\end{aligned}
$$
Next, using  Tonelli's theorem and the tail-sum identity for $\tau$ yields
$$
\begin{aligned}
\mathbb{E}[S_\tau]
&=\mathbb{E}\bigl[\sum_{t\ge0}\mathbf1_{\{\tau>t\}}\xi_t\bigr]\\
&=\sum_{t\ge0}\mathbb{E}[\mathbf1_{\{\tau>t\}}\xi_t]\\
&=p\sum_{t\ge0}\Pr(\tau>t)
=p\mathbb{E}[\tau].
\end{aligned}
$$
These identities also hold for infinite expectations, in which case the claimed lower bound is immediate. Otherwise, combining them with $\mathbb{E}[S_\tau]\ge T/2$ and dividing by $p>0$ gives $\mathbb{E}[\tau]\ge T/(2p)$.

For convenience, we weaken this to $\mathbb{E}[\tau]\ge T/(4p)$. The accuracy restriction equation~\ref{eq:small} ensures $L\Delta/(768\ell_1{\epsilon}^2)\ge2$, so the definition of $T$ and $\lfloor r\rfloor\ge r/2$ for $r\ge2$ give
$$
T\ge\frac{L\Delta}{1536\ell_1{\epsilon}^2}.
$$
Furthermore, equation~\ref{eq:pasta-oracle} and equation~\ref{eq:pasta-variance-budget} imply
$$
\frac1p
=1+\frac{\widetilde\sigma_a^2}{4\varsigma^2{\epsilon}^2}
=1+\frac{G^2+B^2(\Delta/(8{\epsilon}))^a}{4\varsigma^2{\epsilon}^2}.
$$
Substituting both bounds yields
\begin{equation}\label{eq:pasta-cost}
\mathbb{E}[\tau]\ge\frac{T}{4p}
\ge\frac{L\Delta}{6144\ell_1{\epsilon}^2}
\left[1+\frac{G^2+B^2(\Delta/(8{\epsilon}))^a}{4\varsigma^2{\epsilon}^2}\right].
\end{equation}

Hence, the expected number of oracle calls is lower bounded by the ${\epsilon}^{-6}$ growth term at $a=2,$ which becomes ${\epsilon}^{-(4+a)}$ for general $a$.

\le\frac{T-\log8}{2p},
$$
where we use $T\ge8>2\log8$ in the last inequality. Thus the bound in equation~\ref{eq:primitive} covers $m$ oracle calls with an additional query at the output point.

First, lets consider any algorithm that makes at most $m$ calls and returns $\widehat x=(\widehat u,\widehat y)$, and simulate these calls using the oracle $(H_U,v_U)$ as described above. If the algorithm stops before $m$ calls, we keep its output fixed and make the remaining queries at the origin, ignoring their responses. We then query $\widehat y/s$ as query $m+1$. This final query is used only to apply equation~\ref{eq:primitive} and its response does not change the output. With probability at least $3/4$, we get
$$
\left\lVert\nabla H_U(\widehat y/s)\right\rVert\ge\frac12.
$$

$$
\left\lVert\nabla f_U(\widehat x)\right\rVert
\ge\left\lVert\nabla_y f_U(\widehat x)\right\rVert
=8{\epsilon}\left\lVert\nabla H_U(\widehat y/s)\right\rVert
\ge4{\epsilon}.
$$
Hence every algorithm with a budget of $m$ calls satisfies
$$
\Pr\bigl(\left\lVert\nabla f_U(\widehat x)\right\rVert\ge4{\epsilon}\bigr)\ge\frac34.
$$

Now let $\mathcal A$ be the algorithm in Lemma \ref{lem:bg_a_unrestricted_reduction}, with stopping time $\tau$ and output $\widehat x$. Its guarantee holds for every fixed $U$, so averaging over $U$ gives $\mathbb{E}\left\lVert\nabla f_U(\widehat x)\right\rVert\le{\epsilon}$. Markov's inequality then yields
$$
\Pr\bigl(\left\lVert\nabla f_U(\widehat x)\right\rVert\ge4{\epsilon}\bigr)
\le\frac{\mathbb{E}\left\lVert\nabla f_U(\widehat x)\right\rVert}{4{\epsilon}}
\le\frac14.
$$
Next, we define a truncated algorithm $\mathcal A_m$ that runs $\mathcal A$ for at most $m$ calls. If $\mathcal A$ stops within this budget, $\mathcal A_m$ returns the same output. Otherwise, it returns $x_0$. Denoting $\widehat x_m$ as this truncated algorithm's output, we run the two algorithms with the same $U$, oracle seeds, and internal randomness. Their outputs agree whenever $\tau\le m$. Consequently,
$$
\begin{aligned}
\Pr\bigl(\left\lVert\nabla f_U(\widehat x_m)\right\rVert\ge4{\epsilon}\bigr)
&\le\Pr\bigl(\left\lVert\nabla f_U(\widehat x)\right\rVert\ge4{\epsilon}\bigr)
+\Pr(\tau>m)\\
&\le\frac14+\Pr(\tau>m).
\end{aligned}
$$
Since $\mathcal A_m$ uses at most $m$ calls, its probability on the left is at least $3/4$. Thus
$$
\Pr(\tau>m)\ge\frac34-\frac14=\frac12.
$$
Finally, $\tau\ge m\mathbf1_{\{\tau>m\}}$, and the supremum over fixed rotations is at least the average over $U$. Therefore,
\begin{equation}\label{eq:expected}
\sup_U\mathbb{E}[\tau\mid U]
\ge\mathbb{E}[\tau]
\ge m\Pr(\tau>m)
\ge\frac m2
\ge\frac{T}{32p},
\end{equation}
which proves the expected SFO lower bound and completes the lemma.

\label{thm:sgd_conv} Suppose Assumptions \ref{as:l_smooth} and \ref{as:bg_a_oracle} hold with $0\le a\le2$. Under a fixed target variance $\sigma^2>0$ and a target accuracy ${\epsilon} > 0$ satisfying ${\epsilon}^2\le\min\{2\sigma^2,L\Delta\}$, denote the batch size $b_t$, step size $\eta$, and total iterations $T$ as

\end{equation}

$g_t=\frac1{b_t}\sum_{i=1}^{b_t}\nabla f(x_t;\xi_{t,i})$

\end{equation}

Let $e_t:=g_t-\nabla f(x_t)$, and $z_{t,i}:=\nabla f(x_t;\xi_{t,i})-\nabla f(x_t)$. Note that the batch size $b_t$ is chosen \emph{before} sampling that batch. Hence, conditioned on $\mathcal{F}_t$, $b_t$ is a fixed positive integer and $z_{t,i}$ are independent with conditional mean zero. Therefore, we have
\begin{align}\label{eq:cap}
\mathbb{E}_t[e_t] = 0, \qquad
\mathbb{E}_t\left\lVerte_t\right\rVert^2 = \frac1{b_t^2}\sum_{i=1}^{b_t}\mathbb{E}_t\left\lVertz_{t,i}\right\rVert^2
\le\frac{B^2\left\lVertx_t-x_0\right\rVert^a+G^2}{b_t}
\le\sigma^2,
\end{align}
where the cross terms vanish by conditional independence and zero means.

                                                                                                                                                                                                                                                                                                                
                                                                       
                                                                                  

\textbf{Step 1: Descent lemma.}
Using $L$-smoothness and with the SGD update iterates $x_t$, we get
\begin{equation}
f(x_{t+1})\le f(x_t)-\eta\langle\nabla f(x_t),g_t\rangle
+\frac{L\eta^2}{2}\left\lVertg_t\right\rVert^2.
\end{equation}
Taking conditional expectation $\mathbb{E}_t[\cdot]$ and using equation~\ref{eq:cap}
\begin{equation}
\mathbb{E}_t\left\lVertg_t\right\rVert^2
=\left\lVert\nabla f(x_t)\right\rVert^2+\mathbb{E}_t\left\lVerte_t\right\rVert^2
\le\left\lVert\nabla f(x_t)\right\rVert^2+\sigma^2.
\end{equation}
Consequently, since $\eta\le1/L$ (as ${\epsilon}^2 \leq 2\sigma^2$),
\begin{align}
\mathbb{E}_t f(x_{t+1})
&\le f(x_t)-\eta\left(1-\frac{L\eta}{2}\right)\left\lVert\nabla f(x_t)\right\rVert^2
+\frac{L\eta^2\sigma^2}{2}\notag\\
&\le f(x_t)-\frac\eta2\left\lVert\nabla f(x_t)\right\rVert^2
+\frac{L\eta^2\sigma^2}{2}.\label{eq:descent}
\end{align}

\textbf{Step 2: Telescoping to obtain the stationarity criterion.} Taking total expectation of each side of equation~\ref{eq:descent}, summing over $t=0,\ldots,T-1$, and since $\mathbb{E}[f(x_T)]\ge f^{\inf},$ we get 
\begin{align}
\frac\eta2\sum_{t=0}^{T-1}\mathbb{E}\left\lVert\nabla f(x_t)\right\rVert^2
&\le f(x_0)-\mathbb{E}[f(x_T)]+\frac{TL\eta^2\sigma^2}{2}\notag\\
&\le\Delta+\frac{TL\eta^2\sigma^2}{2}.\label{eq:telescope}
\end{align}
Dividing both sides by $T\eta/2$ and substituting the parameters from equation~\ref{eq:sgd_convergence_parameters}, we end up with
\begin{equation}\label{eq:accuracy}
\frac1T\sum_{t=0}^{T-1}\mathbb{E}\left\lVert\nabla f(x_t)\right\rVert^2
\le\frac{2\Delta}{T\eta}+L\eta\sigma^2
\le\frac{{\epsilon}^2}{2}+\frac{{\epsilon}^2}{2}={\epsilon}^2.
\end{equation}

\textbf{Step 3: SFO computation.} Telescoping the iterates $x_t$ gives:
\begin{equation}
x_t-x_0=-\eta\sum_{j=0}^{t-1}\nabla f(x_j)-\eta\sum_{j=0}^{t-1}e_j.
\end{equation}
Taking the norm squared into total expectation of both sides gives
\begin{align}
\mathbb{E}\left\lVertx_t-x_0\right\rVert^2 = \mathbb{E}\left\lVert\eta\sum_{j=0}^{t-1}\nabla f(x_j)+\eta\sum_{j=0}^{t-1}e_j\right\rVert^2
\end{align}
Applying the inequality $\left\lVertu+v\right\rVert^2\le2\left\lVertu\right\rVert^2+2\left\lVertv\right\rVert^2$ and Cauchy--Schwarz on the gradient sum gives
\begin{equation}\label{eq:split}
\mathbb{E}\left\lVertx_t-x_0\right\rVert^2
\le2\eta^2t\sum_{j=0}^{t-1}\mathbb{E}\left\lVert\nabla f(x_j)\right\rVert^2
+2\eta^2\mathbb{E}\left\|\sum_{j=0}^{t-1}e_j\right\|^2.
\end{equation}
Now, notice that for $i<j$, $e_i$ is $\mathcal{F}_j$-measurable, hence
\begin{equation}
\mathbb{E}[\langle e_i, e_j\rangle]=\mathbb{E}[\mathbb{E}[ \langle e_i, e_j \rangle\mid \mathcal{F}_j]] = \mathbb{E}[\langle e_i, \mathbb{E}[e_j \mid \mathcal{F}_j]\rangle]=0.
\end{equation}
Thus, martingale orthogonality and equation~\ref{eq:cap} imply
\begin{equation}
\mathbb{E}\left\|\sum_{j=0}^{t-1}e_j\right\|^2
=\sum_{j=0}^{t-1}\mathbb{E}\left\lVerte_j\right\rVert^2\le t\sigma^2.
\end{equation}
Substituting this and equation~\ref{eq:telescope} into equation~\ref{eq:split}, and using $t\le T$, we obtain
\begin{align}
\mathbb{E}\left\lVertx_t-x_0\right\rVert^2
&\le2\eta^2T\left(\frac{2\Delta}{\eta}+TL\eta\sigma^2\right)
+2\eta^2T\sigma^2\notag\\
&=4\eta T\Delta+2L\eta^3T^2\sigma^2+2\eta^2T\sigma^2=:D_T.\label{eq:radius}
\end{align}

\textbf{Step 4: $a$ power growth.}
For $0\le a\le2$, the map $u\mapsto u^{a/2}$ is concave on $[0,\infty)$. Hence, using Jensen's inequality gives
\begin{equation}\label{eq:sgd_power_growth_jensen}
\mathbb{E}\left\lVertx_t-x_0\right\rVert^a
=\mathbb{E}\left[(\left\lVertx_t-x_0\right\rVert^2)^{a/2}\right]
\le\left(\mathbb{E}\left\lVertx_t-x_0\right\rVert^2\right)^{a/2}
\le (D_T)^{a/2}.
\end{equation}
Summing $b_t$, taking expectations and using $\max\{1,\lceil u\rceil\}\le1+u$ for $u\ge0$, we get
\begin{align}
\mathbb{E} \left[ \sum_{t=0}^{T-1} b_t\right]
&\le\sum_{t=0}^{T-1}\left[1+\frac{G^2}{\sigma^2}
+\frac{B^2}{\sigma^2}\mathbb{E}[\left\lVertx_t-x_0\right\rVert^a]\right]\notag\\
&\le T\left(1+\frac{G^2}{\sigma^2}+\frac{B^2}{\sigma^2}(D_T)^{a/2}\right). \label{eq:exp_of_bt}
\end{align}

\textbf{Step 5: Substituting the explicit constants.}
Setting ${\epsilon}^2\le\min\{2\sigma^2,L\Delta\}$ and using the explicit constant values from Theorem \ref{thm:sgd_conv} gives us
\begin{equation}
\eta=\frac{{\epsilon}^2}{2L\sigma^2},\qquad
m:=\frac{4\Delta}{\eta{\epsilon}^2}=\frac{8L\sigma^2\Delta}{{\epsilon}^4}\ge4.
\end{equation}
Since $m\ge4$, the ceiling satisfies $T=\lceil m\rceil\le m+1\le(5/4)m$. Hence
\begin{equation}\label{eq:Kbound}
T\le\frac{10L\sigma^2\Delta}{{\epsilon}^4},\qquad
\eta T\le\frac{5\Delta}{{\epsilon}^2}.
\end{equation}
We now bound the three terms of $D_T$ (equation~\ref{eq:radius}) separately:
\begin{align}
4\eta T\Delta&\le\frac{20\Delta^2}{{\epsilon}^2},\\
2L\eta^3T^2\sigma^2
=2L\eta\sigma^2(\eta T)^2
&\le{\epsilon}^2\frac{25\Delta^2}{{\epsilon}^4}
=\frac{25\Delta^2}{{\epsilon}^2},\\
2\eta^2T\sigma^2
=2\eta\sigma^2(\eta T)
&\le\frac{{\epsilon}^2}{L}\frac{5\Delta}{{\epsilon}^2}
=\frac{5\Delta}{L}\le\frac{5\Delta^2}{{\epsilon}^2},
\end{align}
where the final inequality uses ${\epsilon}^2\le L\Delta$. Combining all three bounds, we get
\begin{equation}\label{eq:sharpD}
D_T\le\frac{50\Delta^2}{{\epsilon}^2}.
\end{equation}
Combining equation~\ref{eq:Kbound}, equation~\ref{eq:sharpD}, and equation~\ref{eq:exp_of_bt}, we arrive at
\begin{equation}\label{eq:sgd_explicitcost}
\mathbb{E} \left[ \text{SFO} \right] \le10L\Delta\left[(\sigma^2+G^2){\epsilon}^{-4}
+50^{a/2}B^2\Delta^a{\epsilon}^{-(4+a)}\right],
\end{equation}
completing the proof.

satisfies, by martingale orthogonality
$$
\mathbb{E}\left\lVertr_t\right\rVert^2
=(1-\beta)^2\sum_{j=0}^t\beta^{2(t-j)}\mathbb{E}\left\lVerte_j\right\rVert^2
\le\frac{(1-\beta)^2}{1-\beta^2}\frac{\sigma^2}{c_{\beta}}
=\sigma^2.
$$
This controls the filtered noise component.

\textbf{Step 1: Descent lemma.}
To handle the components introduced by momentum, we define the auxiliary iterate
$$
\widetilde x_t:=x_t-\frac{\eta\beta}{1-\beta}m_{t-1},
$$
where $\widetilde x_0=x_0$ since $m_{-1}=0$. Expanding the update gives
\begin{align}
\widetilde x_{t+1}
&=x_t-\eta m_t-\frac{\eta\beta}{1-\beta}m_t\notag\\
&=x_t-\frac{\eta}{1-\beta}
 [\beta m_{t-1}+(1-\beta)g_t]
=\widetilde x_t-\eta g_t.
\label{eq:sgdm_virtual_update}
\end{align}
Hence, by using $L$-smoothness at $\widetilde x_t$, we obtain
\begin{equation}\label{eq:L_smooth_at_tilda_x}
f(\widetilde x_{t+1})
\le f(\widetilde x_t)
-\eta\langle\nabla f(\widetilde x_t),g_t\rangle
+\frac{L\eta^2}{2}\left\lVertg_t\right\rVert^2.
\end{equation}
Taking conditional expectation $\mathbb{E}_t[\cdot]$ and using
$$
\mathbb{E}_t[g_t]=\nabla f(x_t),\qquad
\mathbb{E}_t\left\lVertg_t\right\rVert^2=\left\lVert\nabla f(x_t)\right\rVert^2+\mathbb{E}_t\left\lVerte_t\right\rVert^2,
$$
along with the polarization identity and using $L$-smoothness, we have
\begin{align}
-\langle\nabla f(\widetilde x_t),\nabla f(x_t)\rangle
&\le-\frac12\left\lVert\nabla f(x_t)\right\rVert^2
+\frac12\left\lVert\nabla f(\widetilde x_t)-\nabla f(x_t)\right\rVert^2\notag\\
&\le-\frac12\left\lVert\nabla f(x_t)\right\rVert^2
+\frac{L^2\eta^2\beta^2}{2(1-\beta)^2}\left\lVertm_{t-1}\right\rVert^2.
\end{align}
Applying the previous steps to equation~\ref{eq:L_smooth_at_tilda_x}, we arrive at
\begin{align}\label{eq:sgdm_descent}
\mathbb{E}_t[f(\widetilde x_{t+1})]
&\le f(\widetilde x_t)
-\left(\frac\eta2-\frac{L\eta^2}{2}\right)\left\lVert\nabla f(x_t)\right\rVert^2
+\frac{L^2\eta^3\beta^2}{2(1-\beta)^2}\left\lVertm_{t-1}\right\rVert^2
+\frac{L\eta^2}{2}\mathbb{E}_t\left\lVerte_t\right\rVert^2.
\end{align}

\textbf{Step 2: Telescoping to obtain the stationarity criterion.}
We first control the additional momentum term in equation~\ref{eq:sgdm_descent}. Conditional unbiasedness and convexity of the squared norm imply
\begin{align}
\mathbb{E}_t\left\lVertm_t\right\rVert^2
&=\left\lVert\beta m_{t-1}+(1-\beta)\nabla f(x_t)\right\rVert^2
 +(1-\beta)^2\mathbb{E}_t\left\lVerte_t\right\rVert^2\notag\\
&\le\beta\left\lVertm_{t-1}\right\rVert^2
 +(1-\beta)\left\lVert\nabla f(x_t)\right\rVert^2
 +(1-\beta)^2\mathbb{E}_t\left\lVerte_t\right\rVert^2.
\end{align}
Take expectations and sum over $t=0,\ldots,T-2$. The left side is
$\sum_{t=0}^{T-1}\mathbb{E}\left\lVertm_{t-1}\right\rVert^2$, since $m_{-1}=0$;
the momentum sum on the right is at most this same sum multiplied by $\beta$.
Moving it to the left, dividing by $1-\beta$, and enlarging the remaining sums gives
\begin{equation}\label{eq:sgdm_momentum_energy}
\sum_{t=0}^{T-1}\mathbb{E}\left\lVertm_{t-1}\right\rVert^2
\le\sum_{t=0}^{T-1}\mathbb{E}\left\lVert\nabla f(x_t)\right\rVert^2
 +(1-\beta)\sum_{t=0}^{T-1}\mathbb{E}\left\lVerte_t\right\rVert^2.
\end{equation}
For $T=1$, the same inequality holds because the left side is zero.
Next, we take total expectations in equation~\ref{eq:sgdm_descent}, sum over $t=0,\ldots,T-1$, and use
$f(\widetilde x_T)\ge f^{\inf}$ with $\widetilde x_0=x_0$. Substituting equation~\ref{eq:sgdm_momentum_energy} gives
\begin{align}\label{eq:sgdm_telescope}
\left(\frac\eta2-\frac{L\eta^2}{2}-A\right)
\sum_{t=0}^{T-1}\mathbb{E}\left\lVert\nabla f(x_t)\right\rVert^2
\le\Delta+
\left(A(1-\beta)+\frac{L\eta^2}{2}\right)
\sum_{t=0}^{T-1}\mathbb{E}\left\lVerte_t\right\rVert^2,
\end{align}
where we have denoted $A:=\frac{L^2\eta^3\beta^2}{2(1-\beta)^2}$ for convenience. Applying the restriction on the step size $\eta\le(1-\beta)^2/(2L)$, we get
\begin{align*}
\frac{L\eta^2}{2}&\le\frac\eta4,\qquad
A\le\frac{\eta\beta^2(1-\beta)^2}{8}\le\frac\eta8,\\
A(1-\beta)
&=\frac{L\eta^2}{2}\frac{L\eta\beta^2}{1-\beta}
\le\frac{L\eta^2}{2}\frac{\beta^2(1-\beta)}2
\le\frac{L\eta^2}{2}.
\end{align*}
Thus the coefficient on the left of equation~\ref{eq:sgdm_telescope} is at least $\eta/8$, whereas the coefficient of the error sum is at most $L\eta^2$. Dividing and using equation~\ref{eq:sgdm_cap} yields
\begin{equation}\label{eq:sgdm_gradient_sum}
\sum_{t=0}^{T-1}\mathbb{E}\left\lVert\nabla f(x_t)\right\rVert^2
\le\frac{8\Delta}{\eta}
+\frac{8L\eta T\sigma^2}{c_{\beta}}.
\end{equation}
Finally, dividing by $T$ and substituting equation~\ref{eq:sgdm_convergence_parameters}
$$
\frac1T\sum_{t=0}^{T-1}\mathbb{E}\left\lVert\nabla f(x_t)\right\rVert^2
\le\frac{8\Delta}{\eta T}+\frac{8L\eta\sigma^2}{c_{\beta}}
\le\frac{{\epsilon}^2}{2}+\frac{{\epsilon}^2}{2}={\epsilon}^2,
$$
which proves equation~\ref{eq:sgdm_accuracy}.

\textbf{Step 3: SFO computation.}
For $1\le t\le T$, unrolling the momentum recursion and then summing the iterate updates gives
\begin{align}
m_j&=(1-\beta)\sum_{i=0}^j\beta^{j-i}g_i,\notag\\
x_t-x_0
&=-\eta(1-\beta)\sum_{j=0}^{t-1}\sum_{i=0}^j\beta^{j-i}g_i
=-\eta\sum_{i=0}^{t-1}(1-\beta^{t-i})g_i.
\label{eq:sgdm_displacement}
\end{align}
For fixed $t$, denote $w_{t,i}=1-\beta^{t-i}\in[0,1]$, which are deterministic constants. Hence, splitting $g_i=\nabla f(x_i)+e_i$ gives
\begin{equation}\label{eq:sgdm_split}
\mathbb{E}\left\lVertx_t-x_0\right\rVert^2
\le2\eta^2\mathbb{E}\left\|\sum_{i=0}^{t-1}w_{t,i}\nabla f(x_i)\right\|^2
+2\eta^2\mathbb{E}\left\|\sum_{i=0}^{t-1}w_{t,i}e_i\right\|^2.
\end{equation}
Cauchy--Schwarz bounds the first term inside its expectation by
$$
\mathbb{E}\left\|\sum_{i=0}^{t-1}w_{t,i}\nabla f(x_i)\right\|^2
\le t\sum_{i=0}^{t-1}\mathbb{E}\left\lVert\nabla f(x_i)\right\rVert^2
\le T\sum_{i=0}^{T-1}\mathbb{E}\left\lVert\nabla f(x_i)\right\rVert^2.
$$
For $i<j$, $e_i$ is $\mathcal{F}_j$-measurable and
$$
\mathbb{E}\langle e_i,e_j\rangle
=\mathbb{E}\bigl[\langle e_i,\mathbb{E}[e_j\mid\mathcal{F}_j]\rangle\bigr]=0.
$$
The deterministic weights therefore preserve martingale orthogonality, and equation~\ref{eq:sgdm_cap} gives
$$
\mathbb{E}\left\|\sum_{i=0}^{t-1}w_{t,i}e_i\right\|^2
=\sum_{i=0}^{t-1}w_{t,i}^2\mathbb{E}\left\lVerte_i\right\rVert^2
\le\frac{T\sigma^2}{c_{\beta}}.
$$
Substituting these bounds and equation~\ref{eq:sgdm_gradient_sum} into equation~\ref{eq:sgdm_split} yields
\begin{align}\label{eq:sgdm_radius}
\mathbb{E}\left\lVertx_t-x_0\right\rVert^2
&\le2\eta^2T\left(\frac{8\Delta}{\eta}
+\frac{8L\eta T\sigma^2}{c_{\beta}}\right)
+\frac{2\eta^2T\sigma^2}{c_{\beta}}\notag\\
&=16\eta T\Delta
+\frac{16L\eta^3T^2\sigma^2}{c_{\beta}}
+\frac{2\eta^2T\sigma^2}{c_{\beta}}
=:D_T.
\end{align}
The same bound holds at $t=0$ because the displacement is zero.

\textbf{Step 4: $a$ power growth.}
For $0 \leq a \le 2$, the map $u\mapsto u^{a/2}$ is concave on $[0,\infty)$. Hence, Jensen's inequality and equation~\ref{eq:sgdm_radius} imply
\begin{equation}\label{eq:sgdm_power_growth_jensen}
\mathbb{E}\left\lVertx_t-x_0\right\rVert^a
=\mathbb{E}\left[(\left\lVertx_t-x_0\right\rVert^2)^{a/2}\right]
\le\left(\mathbb{E}\left\lVertx_t-x_0\right\rVert^2\right)^{a/2}
\le D_T^{a/2}.
\end{equation}
Using $\max\{1,\lceil u\rceil\}\le1+u$ for $u\ge0$, summing the batch sizes, and taking total expectation gives
\begin{align}
\mathbb{E}\left[\sum_{t=0}^{T-1}b_t\right]
&\le\sum_{t=0}^{T-1}\left[
1+\frac{c_{\beta} G^2}{\sigma^2}
+\frac{c_{\beta} B^2}{\sigma^2}\mathbb{E}[\left\lVertx_t-x_0\right\rVert^a]\right]\notag\\
&\le T\left(1+\frac{c_{\beta} G^2}{\sigma^2}
+\frac{c_{\beta} B^2}{\sigma^2}D_T^{a/2}\right).
\label{eq:sgdm_exp_of_bt}
\end{align}

\textbf{Step 5: Substituting the explicit constants.}
The restriction ${\epsilon}^2\le\min\{8\sigma^2(1-\beta^2),L\Delta\}$ that we set gives
$$
\frac{c_{\beta}{\epsilon}^2}{16L\sigma^2}
\le\frac{c_{\beta}(1-\beta^2)}{2L}
=\frac{(1-\beta)^2}{2L}.
$$
Hence $\eta \leq (1-\beta)^2/(2L)$ as
$$
\eta=\frac{c_{\beta}{\epsilon}^2}{16L\sigma^2},\qquad
\frac{16\Delta}{\eta{\epsilon}^2}
=\frac{256L\sigma^2\Delta}{c_{\beta}{\epsilon}^4}.
$$
Since $\eta\le1/(2L)$ and ${\epsilon}^2\le L\Delta$, the expression inside the ceiling for $T$ is at least $32$, and hence at least $4$. Applying $\lceil u\rceil\le(5/4)u$ for $u\ge4$ yields
\begin{equation}\label{eq:sgdm_Tbound}
T\le\frac{320L\sigma^2\Delta}{c_{\beta}{\epsilon}^4},\qquad
\eta T\le\frac{20\Delta}{{\epsilon}^2}.
\end{equation}
Next, we bound the three terms of $D_T$ separately
\begin{align}
16\eta T\Delta&\le\frac{320\Delta^2}{{\epsilon}^2},\\
\frac{16L\eta^3T^2\sigma^2}{c_{\beta}}
=\frac{16L\eta\sigma^2}{c_{\beta}}(\eta T)^2
&\le{\epsilon}^2\frac{400\Delta^2}{{\epsilon}^4}
=\frac{400\Delta^2}{{\epsilon}^2},\\
\frac{2\eta^2T\sigma^2}{c_{\beta}}
=\frac{2\eta\sigma^2}{c_{\beta}}(\eta T)
&\le\frac{{\epsilon}^2}{8L}\frac{20\Delta}{{\epsilon}^2}
=\frac{5\Delta}{2L}
\le\frac{5\Delta^2}{2{\epsilon}^2}.
\end{align}
The last inequality uses ${\epsilon}^2\le L\Delta$. Therefore,
\begin{equation}\label{eq:sgdm_sharpD}
D_T\le\frac{725\Delta^2}{{\epsilon}^2}.
\end{equation}
Combining equation~\ref{eq:sgdm_Tbound}, equation~\ref{eq:sgdm_sharpD}, and equation~\ref{eq:sgdm_exp_of_bt} yields
\begin{align*}
\mathbb{E}[\mathrm{SFO}]
&\le\frac{320L\sigma^2\Delta}{c_{\beta}{\epsilon}^4}
\left[1+\frac{c_{\beta} G^2}{\sigma^2}
+\frac{c_{\beta} B^2}{\sigma^2}
\left(\frac{725\Delta^2}{{\epsilon}^2}\right)^{a/2}\right]\\
&=320L\Delta\left[
\left(\frac{\sigma^2}{c_{\beta}}+G^2\right){\epsilon}^{-4}
+725^{a/2}B^2\Delta^a{\epsilon}^{-(4+a)}\right].
\end{align*}
This proves equation~\ref{eq:sgdm_explicitcost_} and completes the proof.
