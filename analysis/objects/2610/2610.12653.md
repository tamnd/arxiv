\label{thm:convergence}
Let $D(g) = \max\left(\frac{1}{2-g^2}, \frac43 \frac{1}{g^2} \right)$.
If $0 < g^2 < 2$ such that $g^2 \neq 2 - \frac{2}{k}$ for all $k \geq 1$,
 and $|\alpha_\mathrm{ren}| < \frac{1}{4D(g) (1+g^2D(g))}$,  
then, using the initial condition 
$c_{1,1} = \frac{1}{g^2}$, $c_{1,0} = 0$,
 we can solve~\eqref{eq:recursion} recursively for the coefficients $c_{k,j}$ and the sum in Eq.~\eqref{eq:ansatz} over $k$ and $j$ 
is absolutely convergent.

<!-- proof-1: body_md -->
As explained above, it follows from the assumptions that $B_{k,j} \neq 0$ for all $0 \leq j \leq k$ and $k \geq 2$ that the solution of the recursion~\eqref{eq:recursion} exists uniquely.

For all $k \geq 3/(2-g^2)$, 
we have $B_{k,j}= (2-g^2)k -2 + j^2 g^2 \geq 1$.
Let $h_1 = 1/g^2$ and $h_{k} = \sum_{j =0}^k |B_{k,j} c_{k,j}|$ for $k \geq 2$.
In particular, we have for $k \geq 3/(2-g^2)$ 
\begin{align} \sum_{j=0}^k |c_{k,j} \alpha_\mathrm{ren}^k \cos(j g \varphi) | \leq |\alpha_\mathrm{ren}^k| \sum_{j=0}^k \frac{1}{|B_{k,j}| } |B_{k,j} c_{k,j}| \leq |\alpha_\mathrm{ren}^k| h_k \, . \end{align}
Hence, we need to show that $h_k$ grows at most as $(4D(g) (1+g^2 D(g)))^k$ for $k\rightarrow \infty$ to prove the statement.

If $1 \leq j \leq k$ with $k \geq 2$, we have $B_{k,j} \geq (2-g^2) (k-1) + (j^2-1)g^2 \geq 2-g^2 + (j^2-1)g^2$. So,
\begin{align} \frac{j^2}{B_{k,j}} \leq \frac{1}{ \frac{1}{j^2} (2-g^2) + (1-\frac{1}{j^2})g^2} \leq \max\left(\frac{1}{2-g^2}, \frac43 \frac{1}{g^2} \right) = D(g)\,, \end{align}
where the first term in the maximum bounds the $j=1$ case and the second term the $j \geq 2$ case.

 
Using the above inequality and $c_{1,1} = h_1$, we get the following estimate
\begin{gather} \label{eq:ineq1} \sum_{j = 1}^k j^2 |c_{k,j}| \leq \sum_{j = 1}^k \frac{j^2}{B_{k,j}} |B_{k,j} c_{k,j}| \leq D(g) h_k \, , \end{gather}
for all $k \geq 1$, where we also used $D(g) \geq 1$.
Similarly, we have the estimate
\begin{gather} \label{eq:ineq2} \sum_{j = 0}^k |((2-g^2)k -2) c_{k,j}| = \sum_{j = 0}^k | (B_{k,j} - j^2 g^2) c_{k,j}| \leq h_k + g^2 D(g) h_k \,. \end{gather}
Summing Eq.~\eqref{eq:recursion} over $0\leq j \leq k$ and using 
$\sum_{j=0}^k \delta_{|j_1\pm j_2|,j} \leq 1$,
we find that for $k \geq 2$,
\begin{gather} h_k \leq \frac{1}{2} 2 g^2 \sum_{\substack{k_1, k_2 \geq 1\\ k_1+k_2 = k }} \sum_{j_1=0}^{k_1} \sum_{j_2=0}^{k_2} |((2-g^2)k_1 -2) c_{k_1,j_1}| \cdot |j_2^2 c_{k_2,j_2}| \, . \end{gather}
Using Eqs.~\eqref{eq:ineq1} and \eqref{eq:ineq2}, we get
\begin{gather} h_k \leq g^2 D(g)(1+ g^2 D(g)) \sum_{\substack{k_1, k_2 \geq 1\\ k_1+k_2 = k }} h_{k_1} h_{k_2} \, . \end{gather}
Let $\kappa(g) = g^2 D(g)(1+ g^2 D(g))$ and define 
$H_k = g^{-2k} \kappa(g)^{k-1} C_{k-1}$, where $C_k$ is the Catalan number which fulfills
the recurrence, $C_0=1$ and $C_{k-1} = \sum_{ \substack{k_1, k_2 \geq 1,k_1+k_2 = k } } C_{k_1-1} C_{k_2-1} $ for all $k \geq 2$. The recurrence
$ H_{k} = \kappa(g) \sum_{ \substack{k_1, k_2 \geq 1, k_1+k_2 = k } } H_{k_1} H_{k_2} $
follows and we get $h_k \leq H_k$ for $k\geq 1$.
 
Famously, the Catalan numbers behave as $C_n \sim 4^n/(\sqrt{\pi} n^{3/2})$ for $n\rightarrow \infty$.
