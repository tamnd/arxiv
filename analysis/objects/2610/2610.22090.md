<!-- definition-1: body_md -->
\label{def:strong}
		A strong solution to \eqref{eq:model} on $[0,T]$ is a triple $(h,\mathbf u,p)$ with positive continuous height and
		\begin{gather}
			\mathbf u\in C([0,T];H^1(\Omega_h))\cap L^2(0,T;H^2(\Omega_h)),\ 
			\partial_t\mathbf u\in L^2(0,T;L^2(\Omega_h)),\\
			p\in L^2(0,T;H^1(\Omega_h)),\nonumber\\
			h\in C([0,T];H_\sharp^3(0,L))\cap L^2(0,T;H_\sharp^{7/2}(0,L)),\\
			\partial_th\in C([0,T];H_\sharp^1(0,L))\cap L^2(0,T;H_\sharp^{3/2}(0,L)),\nonumber\\
			\partial_t^2h\in L^2(0,T;H_\sharp^{-1/2}(0,L)).\nonumber
		\end{gather}
		The fluid equations hold in $L^2(0,T;L^2(\Omega_h))$, the beam equation holds in $L^2(0,T;H_\sharp^{-1/2}(0,L))$, and the boundary conditions hold in the trace sense. The prescribed initial values are attained in the indicated continuous spaces. The pressure constant is fixed by the mean of the beam equation.

<!-- theorem-1: body_md -->
\label{thm:separation}
		Let $(h,\mathbf u,p)$ be a solution in Definition~\ref{def:strong} on $[0,T_*)$. Set
		\begin{align}
			\mathcal S(t)=\int_0^L\frac{\,\mathrm dx}{h(t,x)}.
		\end{align}
		There exist $C_1,C_2>0$, depending only on $\mathcal E(0),H,L,\rho_f,\rho_s,\mu,\alpha,\beta$, such that
		\begin{align}
			\mathcal S(t)\leq \mathcal R(t):=C_1(1+\mathcal S(0))e^{C_2t},
			\qquad 0\leq t<T_*, \label{eq:entropy-bound}
		\end{align}
		and
		\begin{align}
			\min_{x\in[0,L]}h(t,x)\geq
			\left(\frac{2\mathcal R(t)}L+\frac{2\mathcal E(0)}{3\alpha}\mathcal R(t)^3\right)^{-1},
			\qquad 0\leq t<T_*.\label{eq:gap-bound}
		\end{align}
		These constants are independent of any assumed positive lower bound for the evolving height.

\label{thm:global}
		Let the initial data $(\mathbf u_0,h_0,h_1)$ satisfy \eqref{eq:data}. Then system~\eqref{eq:model} has a unique global solution in the class of Definition~\ref{def:strong}. It satisfies the energy identity~\eqref{eq:energy-identity} and the distance estimates~\eqref{eq:entropy-bound}--\eqref{eq:gap-bound}. For every finite $T>0$,

			&\sup_{0\leq t\leq T}\left(
			\lVert \mathbf u(t)\rVert_{H^1(\Omega_h(t))}^2+
			\lVert h(t)\rVert_{H_\sharp^3(0,L)}^2+
			\lVert \partial_th(t)\rVert_{H_\sharp^1(0,L)}^2\right)\nonumber\\

			\lVert \partial_t\mathbf u\rVert_{L^2(\Omega_h)}^2+

\label{thm:smoothing}

<!-- proof-1: body_md -->
Integrating incompressibility and using the prescribed traces, we obtain $\int_0^L\partial_th\,\mathrm dx=0$. <!-- lemma-2: body_md -->
\label{lem:geometry}
		Every positive periodic $h\in H_\sharp^2(0,L)$ satisfies
		\begin{align}
			| \partial_x h (x)|^3&\leq\frac{9}{4}\lVert \partial_x^2 h\rVert_{L_\sharp^2(0,L)}^2h(x).\label{eq:slope-cubic}
		\end{align}
		Moreover, it holds
		\begin{align}
			(\min h)^{-1}\leq\frac{2}L \int_0^L \frac1h\,\mathrm dx+\frac{1}{3}\lVert \partial_x^2 h\rVert_{L_\sharp^2(0,L)}^2 \left(\int_0^L \frac1h\,\mathrm dx\right)^3,
			\label{eq:min-entropy0}
		\end{align}
		In particular,
		\begin{align}
			(\min h)^{-1/3}\leq C\left(1+\int_0^L \frac1h\,\mathrm dx\right).
			\label{eq:min-entropy}
		\end{align}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
The one-dimensional Sobolev embedding and positivity of $h$ justify the chain rules and integrations by parts below. We first observe that
		\begin{align*}
			\partial_x\left(\frac{(\partial_x h)^3}{h}\right)
			=3\frac{(\partial_x h)^2}{h}\partial_x^2h
			-\frac{(\partial_x h)^4}{h^2}
			=\frac94|\partial_x^2h|^2
			-\left(\frac{(\partial_x h)^2}{h}
			-\frac32\partial_x^2h\right)^2
			\leq\frac94|\partial_x^2h|^2.
		\end{align*}
		Since $\partial_x h$ has zero mean and is continuous, it vanishes somewhere in each period. Integrating the preceding inequality from such a zero to $x$ bounds $(\partial_x h(x))^3/h(x)$ from above. Integrating from $x$ to the next periodic copy of the same zero bounds it from below. Consequently,
		\begin{align*}
			\frac{|\partial_x h(x)|^3}{h(x)}
			\leq\frac94\int_0^L|\partial_x^2h|^2\,\mathrm dx,
		\end{align*}
		which implies \eqref{eq:slope-cubic}.
		
		For the reciprocal-height estimate, we apply periodic integration by parts and the Cauchy--Schwarz inequality to obtain
		\begin{align*}
			\int_0^L\frac{|\partial_x h|^2}{h^3}\,\mathrm dx
			=\frac12\int_0^L\frac{\partial_x^2h}{h^2}\,\mathrm dx
			& \leq\frac12
			\left(\int_0^L|\partial_x^2h|^2\,\mathrm dx\right)^{1/2}
			\left(\int_0^L\frac1{h^4}\,\mathrm dx\right)^{1/2} \\
			& \leq\frac{\lVert \partial_x^2 h\rVert_{L_\sharp^2(0,L)}}{2} \left(\int_0^L \frac1h\,\mathrm dx\right)^{1/2} (\min h)^{-3/2}.
		\end{align*}
		The periodic $W^{1,1}$ estimate bounds the maximum of $h^{-1}$ by its mean plus one half of the $L^1$ norm of its derivative. Indeed, periodicity implies that its total variation is at least twice its oscillation. Applying this estimate and then the Cauchy--Schwarz and Young inequalities, we obtain
		\begin{align*}
			(\min h)^{-1}
			&\leq\frac 1L \int_0^L \frac1h\,\mathrm dx+\frac12\int_0^L
			\frac{|\partial_x h|}{h^2}\,\mathrm dx\\
			&\leq\frac 1L\int_0^L \frac1h\,\mathrm dx+\frac12
			\left(\int_0^L\frac{|\partial_x h|^2}{h^3}\,\mathrm dx\right)^{1/2}
			\left(\int_0^L\frac1h\,\mathrm dx\right)^{1/2}\\
			&\leq\frac 1L\int_0^L \frac1h\,\mathrm dx
			+ \frac1{2} (\min h)^{-1}
			+ \frac{\lVert \partial_x^2 h\rVert_{L_\sharp^2(0,L)}^2}{6} \left(\int_0^L \frac1h\,\mathrm dx\right)^3,
		\end{align*}
		which shows \eqref{eq:min-entropy0}. Taking cube roots, using $(a+b)^{1/3}\leq a^{1/3}+b^{1/3}$ for $a,b\geq0$, and recalling $\lVert \partial_x^2h\rVert_{L_\sharp^2(0,L)}^2\leq2\mathcal E(0)/\alpha$, we obtain
		\begin{align*}
			(\min h)^{-1/3}
			&\leq\left(\frac2L\right)^{1/3}\left(\int_0^L\frac1h\,\mathrm dx\right)^{1/3}
			+\left(\frac{2\mathcal E(0)}{3\alpha}\right)^{1/3}\int_0^L\frac1h\,\mathrm dx
			\leq C\left(1+\int_0^L\frac1h\,\mathrm dx\right),
		\end{align*}
		which proves \eqref{eq:min-entropy}.

<!-- lemma-3: body_md -->
\label{lem:flux}
		Define the horizontal flux by
		\begin{align*}
			Q(t,x)=\int_0^{h(t,x)}u_1(t,x,y)\,\mathrm dy.
		\end{align*}
		Then
		\begin{align}
			\partial_t h =- \partial_x Q,\qquad
			12\int_0^L\frac{Q^2}{h^3}\,\mathrm dx\leq\mathcal D,\qquad
			\int_0^L\frac{(\partial_t h)^2}{h}\,\mathrm dx\leq\mathcal D.\label{eq:flux-estimates}
		\end{align}
		Moreover,
		\begin{align}
			\lVert \mathbf u(x,\cdot)\rVert_{L^2(0,h(x))}
			\leq h(x)\lVert \partial_y\mathbf u(x,\cdot)\rVert_{L^2(0,h(x))}.\label{eq:column-poincare}
		\end{align}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Differentiating $Q$ and using $u_1(x,h)=0$, we find
		\begin{align*}
			\partial_x Q =\int_0^h \partial_x u_1 \,\mathrm dy=-\int_0^h \partial_y u_2 \,\mathrm dy=- \partial_t h.
		\end{align*}
		The two zero endpoint values of $u_1$ imply
		\begin{align*}
			Q=-\int_0^h\left(y-\frac h2\right) \partial_y u_1 \,\mathrm dy,
			\qquad |Q|^2\leq\frac{h^3}{12}\int_0^h| \partial_y u_1 |^2\,\mathrm dy.
		\end{align*}
		Also $\partial_t h =\int_0^h \partial_y u_2 \,\mathrm dy$, so $(\partial_t h)^2\leq h\int_0^h| \partial_y u_2 |^2\,\mathrm dy$. Integration in $x$ proves \eqref{eq:flux-estimates}. Finally, \eqref{eq:column-poincare} follows from the bottom condition $\mathbf u(x,0)=0$ and the fundamental theorem of calculus.

<!-- lemma-4: body_md -->
The solution of \eqref{eq:elliptic} satisfies
		\begin{align}
			|F|\leq Ch^{1/3},\qquad
			| \partial_x F |\leq Ch^{-1/3},\qquad
			| \partial_x^2 F |\leq Ch^{-1},\label{eq:pointwise}
		\end{align}
		where $C > 0$ does not depend on $\min h$.
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Since $h\in H_\sharp^2(0,L)$ is positive, \eqref{eq:elliptic} implies $F\in H_\sharp^3(0,L)\hookrightarrow C^2([0,L])$. We estimate $F$ and $\ell\partial_xF$ simultaneously, using
		\begin{align*}
			|F|+\ell|\partial_xF|
			=\max\bigl\{|F+\ell\partial_xF|,\,
			|F-\ell\partial_xF|\bigr\}.
		\end{align*}
		The combinations on the right satisfy first-order identities involving only $\partial_xh$ and $\partial_x\ell$. Indeed, since
		\begin{align*}
			\ell^2\partial_x^2F=F-\partial_xh,
			\qquad
			\ell\partial_x(h^{-1/3})
			=-\frac12(\partial_x\ell)h^{-1/3},
		\end{align*}
		direct differentiation yields
		\begin{align*}
			\ell\partial_x\left(
			\frac{F\pm\ell\partial_xF}{h^{1/3}}\right)
			&=
			\pm\frac{F\pm\ell\partial_xF}{h^{1/3}}
			-\frac{\partial_x\ell}{2}
			\frac{F\mp\ell\partial_xF}{h^{1/3}}
			\mp\frac{\partial_xh}{h^{1/3}}.
		\end{align*}
		
		Choose the sign for which the absolute maximum of these two periodic functions is larger, and take a point where this maximum is attained. The derivative of the selected function vanishes there, while the absolute value of the other function is no larger. Taking absolute values in the preceding identity and using \eqref{eq:length-lipschitz}, we obtain
		\begin{align*}
			\max_{\pm}
			\left\|\frac{F\pm\ell\partial_xF}{h^{1/3}}\right\|
			_{L_\sharp^\infty(0,L)}
			&\leq \frac1{32}\max_{\pm}
			\left\|\frac{F\pm\ell\partial_xF}{h^{1/3}}\right\|
			_{L_\sharp^\infty(0,L)}
			+\left\|\frac{\partial_xh}{h^{1/3}}\right\|
			_{L_\sharp^\infty(0,L)}.
		\end{align*}
		Absorbing the first term and recalling the slope bound in \eqref{eq:length-lipschitz}, we conclude that, at every point,
		\begin{align*}
			\frac{|F|+\ell|\partial_xF|}{h^{1/3}}
			&\leq \frac{32}{31}
			\lVert \frac{\partial_xh}{h^{1/3}}\rVert
			_{L_\sharp^\infty(0,L)}
			\leq C.
		\end{align*}
		Consequently, since $\ell=\varepsilon h^{2/3}$,
		\begin{align*}
			|F|\leq Ch^{1/3},
			\qquad
			|\partial_xF|\leq C\varepsilon^{-1}h^{-1/3}.
		\end{align*}
		Finally, the equation itself implies
		\begin{align*}
			|\partial_x^2F|
			\leq \ell^{-2}\bigl(|F|+|\partial_xh|\bigr)
			\leq C\varepsilon^{-2}h^{-1}.
		\end{align*}
		This completes the proof.

<!-- lemma-5: body_md -->
\label{lem:elastic}
		For a positive $h\in H_\sharp^3(0,L)$, it holds
		\begin{align}
			\beta& \int_0^L \partial_x^2 h \partial_x F \,\mathrm dx+\alpha\int_0^L \partial_x^3 h \partial_x^2 F \,\mathrm dx \nonumber\\
			& \geq\frac{15\alpha}{16}
			\bigl(\lVert \partial_x^2 F\rVert_{L_\sharp^2(0,L)}^2+\lVert \ell \partial_x^3 F\rVert_{L_\sharp^2(0,L)}^2\bigr)
			+\beta\bigl(\lVert \partial_x F\rVert_{L_\sharp^2(0,L)}^2+\lVert \ell \partial_x^2 F\rVert_{L_\sharp^2(0,L)}^2\bigr)\geq0.
			\label{eq:elastic-coercivity}
		\end{align}
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
By \eqref{eq:elliptic}, we have 
		\begin{align*}
			\partial_x^2 h = \partial_x F -\partial_x(\ell^2 \partial_x^2 F),\qquad
			\partial_x^3 h = \partial_x^2 F -\partial_x^2(\ell^2 \partial_x^2 F).
		\end{align*}
		Then $F\in H_\sharp^4(0,L)$ because $h\in H_\sharp^3(0,L)$. Integration by parts yields
		\begin{align*}
			\int_0^L \partial_x^2 h \partial_x F \,\mathrm dx
			&=\lVert \partial_x F\rVert_{L_\sharp^2(0,L)}^2+\lVert \ell \partial_x^2 F\rVert_{L_\sharp^2(0,L)}^2,\\
			\int_0^L \partial_x^3 h \partial_x^2 F \,\mathrm dx
			&=\lVert \partial_x^2 F\rVert_{L_\sharp^2(0,L)}^2+\lVert \ell \partial_x^3 F\rVert_{L_\sharp^2(0,L)}^2
			+\int_0^L \partial_x(\ell^2) \partial_x^2 F \partial_x^3 F.
		\end{align*}
		Moreover, 
		\begin{align*}
			\left|\int_0^L \partial_x(\ell^2) \partial_x^2 F \partial_x^3 F \,\mathrm dx\right|
			\leq\frac18\lVert \partial_x^2 F\rVert_{L_\sharp^2(0,L)}\lVert \ell \partial_x^3 F\rVert_{L_\sharp^2(0,L)}
			\leq\frac1{16}\bigl(\lVert \partial_x^2 F\rVert_{L_\sharp^2(0,L)}^2+\lVert \ell \partial_x^3 F\rVert_{L_\sharp^2(0,L)}^2\bigr).
		\end{align*}
		Adding them together with $\alpha > 0,\beta \geq 0$ proves \eqref{eq:elastic-coercivity}.

<!-- lemma-6: body_md -->
\label{lem:time}
		For a strong solution, it follows
		\begin{gather}
			\int_0^L h(\partial_t F)^2\,\mathrm dx\leq C\mathcal D,\qquad
			\int_0^L h(\partial_x\partial_t F)^2\,\mathrm dx\leq C (\min h)^{-2/3}\mathcal D,\label{eq:time-dissipation}\\
			\left|\int_0^L \partial_x\partial_t h \partial_t F \,\mathrm dx\right|\leq C(1+\mathcal S)\mathcal D.
		\end{gather}
<!-- end lemma-6 -->
<!-- proof-6: body_md -->
Testing \eqref{eq:time-resolvent} by $h \partial_t F$ and integrating by parts, we obtain the exact identity
		\begin{align}
			&\int_0^L h(\partial_t F)^2\,\mathrm dx+\int_0^L \ell^2 h(\partial_x\partial_t F)^2\,\mathrm dx
			\nonumber\\
			&\quad=-\int_0^L\partial_x(\ell^2 h)\,\partial_t F\,\partial_x\partial_t F\,\mathrm dx-\int_0^L h\,\partial_t h\,\partial_x\partial_t F\,\mathrm dx
			+\int_0^L\left(\frac43F-\frac73\partial_x h\right)\partial_t h\,\partial_t F\,\mathrm dx.
			\label{eq:first-weighted-identity}
		\end{align}
		Since $\partial_x(\ell^2 h) = \frac72 \partial_x \ell h \ell$, we have 
		\begin{align*}
			\left|\int_0^L\partial_x(\ell^2 h) \partial_t F \partial_x\partial_t F \,\mathrm dx\right|
			\leq\frac7{64}\left(\int_0^L h(\partial_t F)^2\,\mathrm dx+\int_0^L \ell^2 h(\partial_x\partial_t F)^2\,\mathrm dx\right),
		\end{align*}
		where we used $\|\partial_x\ell\|_{L_\sharp^\infty(0,L)} \leq 1/16$.
		Moreover, by Young's inequality, $\ell = \varepsilon h^{2/3}$, \eqref{eq:slope-cubic}, and \eqref{eq:pointwise} one records
		\begin{align*}
			\left|\int_0^L h \partial_t h \partial_x\partial_t F \,\mathrm dx\right|
			&\leq\frac14\int_0^L \ell^2 h(\partial_x\partial_t F)^2\,\mathrm dx+C\int_0^L h^{-1/3}(\partial_t h)^2\,\mathrm dx,\\
			\left|\int_0^L\left(\frac43F-\frac73 \partial_x h \right) \partial_t h \partial_t F \,\mathrm dx\right|
			&\leq\frac14\int_0^L h(\partial_t F)^2\,\mathrm dx+C\int_0^L h^{-1/3}(\partial_t h)^2\,\mathrm dx.
		\end{align*}
		Therefore, by \eqref{eq:energy-geometry} and \eqref{eq:flux-estimates}, we infer from \eqref{eq:first-weighted-identity} that
		\begin{align*}
			\int_0^L h(\partial_t F)^2\,\mathrm dx+\int_0^L \ell^2h(\partial_x\partial_t F)^2\,\mathrm dx
			\leq C\int_0^L h^{-1/3}(\partial_t h)^2\,\mathrm dx
			\leq C\int_0^L h^{-1}(\partial_t h)^2\,\mathrm dx
			\leq C\mathcal D,
		\end{align*}
		which proves the first inequality of \eqref{eq:time-dissipation}.
		
		Now testing instead by $(h/\ell^2) \partial_t F$, one has
		\begin{align}
			&\int_0^L\frac{h}{\ell^2}(\partial_t F)^2\,\mathrm dx+\int_0^L h(\partial_x\partial_t F)^2\,\mathrm dx
			\nonumber\\
			&\qquad=
			-\int_0^L\partial_x h\,\partial_t F\,\partial_x\partial_t F\,\mathrm dx
			-\int_0^L\frac{h}{\ell^2}\,\partial_t h\,\partial_x\partial_t F\,\mathrm dx
			+\int_0^L\frac{\frac43F-\partial_x h}{\ell^2}\,\partial_t h\,\partial_t F\,\mathrm dx.
			\label{eq:second-weighted-identity}
		\end{align}
		Here $h/\ell^2=\varepsilon^{-2}h^{-1/3}$ and $\ell \partial_x h /h=\frac32 \partial_x \ell$ and $\|\partial_x\ell\|_{L^\infty}\leq 1/16$. Thus the first term on the right-hand side is controlled by
		\begin{align*}
			\left|\int_0^L
			(\partial_x h)(\partial_t F)(\partial_x\partial_t F)\,\mathrm dx\right|
			&\leq
			\frac32\|\partial_x\ell\|_{L^\infty}
			\int_0^L\left|\frac{\sqrt h}{\ell}\,\partial_t F\right|\left|\sqrt h\,\partial_x\partial_t F\right|\,\mathrm dx
			\\
			&\quad\leq
			\frac{3}{64}
			\int_0^L\left(\frac{h}{\ell^2}|\partial_t F|^2+h|\partial_x\partial_t F|^2\right)\,\mathrm dx.
		\end{align*}
		Furthermore, by Young's inequality, $\ell = \varepsilon h^{2/3}$, \eqref{eq:slope-cubic}, and \eqref{eq:pointwise}, it holds
		\begin{align*}
			\left|\int_0^L\frac{h}{\ell^2} \partial_t h \partial_x\partial_t F \,\mathrm dx\right|
			&\leq\frac14\int_0^L h(\partial_x\partial_t F)^2\,\mathrm dx+C\int_0^L h^{-5/3}(\partial_t h)^2\,\mathrm dx,\\
			\left|\int_0^L\frac{\frac43F- \partial_x h}{\ell^2}\, \partial_t h \partial_t F \,\mathrm dx\right|
			&\leq\frac14\int_0^L\frac{h}{\ell^2}(\partial_t F)^2\,\mathrm dx+C\int_0^L h^{-5/3}(\partial_t h)^2\,\mathrm dx,
		\end{align*}
		Combining these bounds with the preceding estimates and \eqref{eq:second-weighted-identity}, we obtain
		\begin{align*}
			\int_0^L\frac {h}{\ell^2}(\partial_t F)^2\,\mathrm dx+\int_0^L h(\partial_x\partial_t F)^2\,\mathrm dx
			\leq C\int_0^L h^{-5/3}(\partial_t h)^2\,\mathrm dx
			\leq C (\min h)^{-2/3}\mathcal D.
		\end{align*}
		Here we have used \eqref{eq:flux-estimates}. This shows the second inequality in \eqref{eq:time-dissipation}. 
		
		Finally, integrating by parts with Lemma~\ref{lem:geometry} leads to
		\begin{align*}
			\left|\int_0^L \partial_x\partial_t h \partial_t F \,\mathrm dx\right|
			=\left|\int_0^L \partial_t h \partial_x\partial_t F \,\mathrm dx\right|
			&\leq\left(\int_0^L\frac{(\partial_t h)^2}{h}\,\mathrm dx\right)^{1/2}
			\left(\int_0^L h(\partial_x\partial_t F)^2\,\mathrm dx\right)^{1/2}\\
			&\leq C (\min h)^{-1/3}\mathcal D\leq C(1+\mathcal S)\mathcal D.\qedhere
		\end{align*}

\label{lem:distance-admissibility}
		Let $(h,\mathbf u,p)$ satisfy Definition~\ref{def:strong} on $[0,T]$. The multiplier given by \eqref{eq:elliptic} satisfies
		\begin{align}
			F\in C([0,T];H_\sharp^4(0,L))\cap C^1([0,T];H_\sharp^2(0,L)).\label{eq:multiplier-lower-class}
		\end{align}
		The field in \eqref{eq:stream-function} satisfies
		\begin{align}
			\mathbf w\in C([0,T];H^2(\Omega_h)),\qquad
			\partial_t\mathbf w\in C([0,T];L^2(\Omega_h)).\label{eq:test-lower-class}
		\end{align}
<!-- end lemma-7 -->
<!-- proof-7: body_md -->
The height has a positive minimum on the compact interval. Using the equation $\partial_x^2F=\ell^{-2}(F-\partial_xh)$, the coercive weak formulation, and one-dimensional Sobolev multiplication, we first obtain $F\in H_\sharp^2(0,L)$ and then $F\in H_\sharp^4(0,L)$ because $h\in H_\sharp^3(0,L)$. Resolvent estimates for the difference of two coefficient fields imply continuity with values in $H_\sharp^4(0,L)$.
		
		Subtracting \eqref{eq:elliptic} at $t+\tau$ and $t$, and dividing by $\tau$, we obtain
		\begin{align*}
			&\frac{F(t+\tau)-F(t)}{\tau}
			-\ell(t)^2\partial_x^2\frac{F(t+\tau)-F(t)}{\tau}
			=\partial_x\frac{h(t+\tau)-h(t)}{\tau}
			+\frac{\ell(t+\tau)^2-\ell(t)^2}{\tau}\partial_x^2F(t+\tau).
		\end{align*}
		Since $\partial_th\in C([0,T];H_\sharp^1(0,L))$ and $F\in C([0,T];H_\sharp^4(0,L))$, the right-hand side converges in $L_\sharp^2(0,L)$, uniformly on compact time intervals. Elliptic estimates then show that the time derivative belongs to $C([0,T];H_\sharp^2(0,L))$, which gives \eqref{eq:multiplier-lower-class}. 
		
		The stream function consequently has three spatial derivatives in $L^2$, while its Eulerian time derivative is
		\begin{align*}
			\partial_t\psi=\partial_tF\,\chi
			-F\frac{\partial_th}{h}\frac yh\chi',
		\end{align*}
		which has one spatial derivative in $L^2$ continuously in time. This proves \eqref{eq:test-lower-class}. Also $\partial_xF\in C([0,T];H_\sharp^3(0,L))\cap C^1([0,T];H_\sharp^1(0,L))$, so
		\begin{align*}
			\left\langle\partial_t^2h,\partial_xF\right\rangle_{H_\sharp^{-1/2},H_\sharp^{1/2}}
		\end{align*}
		is integrable and the product rule for $\int_0^L\partial_th\,\partial_xF\,\mathrm dx$ is valid. After pullback by $\chi_h$, the fluid product rule follows from the identity $\partial_t(\mathbf u\circ\chi_h)=(\partial_t\mathbf u)\circ\chi_h+z\partial_th\,(\partial_y\mathbf u)\circ\chi_h\in L^2(0,T;L^2(\Omega_1))$, the preceding regularity of $\mathbf w$, and the Jacobian $h$. The displayed regularity follows from the positive gap and $\partial_th\in C([0,T];H_\sharp^1(0,L))$. Thus Reynolds transport and the integrations by parts used below hold down to zero, by temporal regularization if necessary. In the beam acceleration term of the coupled identity, the integral is understood as the displayed duality pairing.
		
		The positive minimum used here justifies the operations on an existing solution. Every inequality used to estimate the distance is proved separately with the gap dependence displayed explicitly.

\label{lem:stress-identity}
		For the velocity and test above, we have
		\begin{align}
			2\mu& \int_{\Omega_h}D\mathbf u:D\mathbf w\,\mathrm dx\,\mathrm dy \nonumber\\
			& =-6\mu \frac{\mathrm d}{\mathrm dt} \int_0^L \frac{\,\mathrm dx}h 
			- 12\mu\varepsilon^2\int_0^L \frac{Q}{h^{5/3}} \partial_x^2 F \,\mathrm dx+\mu\int_{\Omega_h} \partial_x^2 \psi (\partial_x u_2 -2 \partial_y u_1)\,\mathrm dx\,\mathrm dy.
			\label{eq:distance-viscosity}
		\end{align}
<!-- end lemma-8 -->
<!-- proof-8: body_md -->
First, integrating by parts and recalling $\operatorname{div}\mathbf u=0$, we find
		\begin{align*}
			\int_{\Omega_h}(\nabla\mathbf u)^\top:\nabla\mathbf w\,\mathrm dx\,\mathrm dy
			=\int_0^L \partial_x F (- \partial_x h \partial_y u_1 + \partial_y u_2)(x,h(x))\,\mathrm dx=0.
		\end{align*}
		Indeed, $\partial_y u_2 =- \partial_x u_1$, and differentiation of the trace $u_1(x,h(x))=0$ implies $\partial_x u_1 + \partial_x h \partial_y u_1 =0$. The lower-boundary term vanishes by \eqref{eq:test-traces}, and the lateral terms cancel by periodicity. The same computation with $\mathbf w=\mathbf u$ proves $2\int_{\Omega_h}|D\mathbf u|^2\,\mathrm dx\,\mathrm dy=\int_{\Omega_h}|\nabla\mathbf u|^2\,\mathrm dx\,\mathrm dy$.

		\begin{align*}
			\nabla\mathbf u:\nabla\mathbf w
			=2 \partial_y u_2 \partial_x\partial_y \psi - \partial_y u_1 \partial_y^2 \psi + \partial_x u_2 \partial_x^2 \psi.
		\end{align*}
		Since $u_1|_{y=0} = u_1|_{y=h} = 0$, $\partial_y\chi(z)=\chi'(z)/h$ and $\partial_y^3\chi(z)=\chi'''(z)/h^3=-12/h^3$,

			-\int_{\Omega_h} \partial_y u_1 \partial_y^2 \psi \,\mathrm dx\,\mathrm dy
			=\int_{\Omega_h}u_1 \partial_y^3 \psi \,\mathrm dx\,\mathrm dy
			=-12\int_0^L\frac{QF}{h^3}\,\mathrm dx.
		\end{align*}
		The first term satisfies
		\begin{align*}
			2\int_{\Omega_h} \partial_y u_2 \partial_x\partial_y \psi \,\mathrm dx\,\mathrm dy
			=-2\int_{\Omega_h} \partial_x u_1 \partial_x\partial_y \psi \,\mathrm dx\,\mathrm dy
			=2\int_{\Omega_h}u_1 \partial_x^2\partial_y \psi \,\mathrm dx\,\mathrm dy
			=-2\int_{\Omega_h} \partial_y u_1 \partial_x^2 \psi \,\mathrm dx\,\mathrm dy.
		\end{align*}
		Invoking identities above, we arrive at
		\begin{align}
			2\int_{\Omega_h}D\mathbf u:D\mathbf w\,\mathrm dx\,\mathrm dy
			=-12\int_0^L\frac{QF}{h^3}\,\mathrm dx
			+\int_{\Omega_h} \partial_x^2 \psi (\partial_x u_2 -2 \partial_y u_1)\,\mathrm dx\,\mathrm dy.\label{eq:stress-identity}
		\end{align}
		By $\partial_t h =- \partial_x Q$, we compute
		\begin{align}
			\frac{\mathrm d}{\mathrm dt} \int_0^L \frac{\,\mathrm dx}h  =-\int_0^L\frac{\partial_t h}{h^2}\,\mathrm dx
			=2\int_0^L\frac{Q \partial_x h}{h^3}\,\mathrm dx.\label{eq:entropy-derivative}
		\end{align}
		Substituting $F= \partial_x h +\varepsilon^2h^{4/3} \partial_x^2 F$ into \eqref{eq:stress-identity} with \eqref{eq:entropy-derivative} yields \eqref{eq:distance-viscosity}.

\label{lem:fluid-remainders}

		\begin{align}
			\left|\int_{\Omega_h}\mathbf u\cdot
			(\partial_t \mathbf w +\mathbf u\cdot\nabla\mathbf w)\,\mathrm dx\,\mathrm dy\right|
			\leq C\mathcal D.\label{eq:fluid-inertia-bound}

<!-- proof-10: body_md -->
We first record the second spatial derivative of the stream function. Differentiating at fixed physical coordinates, with the profile and its derivatives evaluated at $z=y/h\in(0,1)$, we obtain
		\begin{align}
			\partial_x^2 \psi &= \partial_x^2 F \chi
			-\left(\frac{2 \partial_x F \partial_x h}{h}+\frac{F \partial_x^2 h}h\right)\frac yh\chi'
			+\frac{F(\partial_x h)^2}{h^2}\Big(2\frac yh\chi'+\frac{y^2}{h^2}\chi''\Big).\label{eq:psi-second}
		\end{align}
		For the time derivative, we use the formula established in the proof of Lemma~\ref{lem:distance-admissibility}. Since the profile and its derivatives are bounded on $[0,1]$, \eqref{eq:time-dissipation} implies
		\begin{align*}
			\lVert \partial_t \psi\rVert_{L^2(\Omega_h)}^2
			\leq C\int_0^L h(\partial_t F)^2\,\mathrm dx+C\int_0^L\frac{F^2(\partial_t h)^2}{h}\,\mathrm dx
			\leq C\mathcal D,
		\end{align*}
		where we used $\lVert F\rVert_{L_\sharp^\infty(0,L)}\leq C$ and \eqref{eq:flux-estimates} for the second term.
		
		By \eqref{eq:length-lipschitz} and \eqref{eq:pointwise}, every term in \eqref{eq:psi-second} other than the one containing $\partial_x^2 h$ is bounded by $Ch^{-1}$. The remaining term is controlled by $C| \partial_x^2 h |h^{-2/3}$. Therefore invoking \eqref{eq:min-entropy}, one has
		\begin{align*}
			\lVert \partial_x^2 \psi\rVert_{L^2(\Omega_h)}^2
			\leq C\left(\mathcal S+\int_0^L \frac{(\partial_x^2 h)^2}{h^{1/3}}\,\mathrm dx\right)
			\leq C(\mathcal S+(\min h)^{-1/3})\leq C(1+\mathcal S).

			\left|\int_{\Omega_h}\mathbf u\cdot \partial_t \mathbf w \,\mathrm dx\,\mathrm dy\right|
			\leq C \mathcal D.

\label{lem:lower-lift}
		For a smooth mean-zero periodic function $\xi$, let $\mathscr{E}(\xi)$ be the stationary Stokes velocity on $\Omega_1$ determined by
		\begin{align*}
			&\operatorname{div}\mathscr{E}(\xi)=0,\qquad
			\mathscr{E}(\xi)|_{z=0}=0,\qquad \mathscr{E}(\xi)|_{z=1}=\xi\mathbf e_2,\\
			&\int_0^L\int_0^1\nabla\mathscr{E}(\xi):\nabla\boldsymbol\psi\,\mathrm dz\,\mathrm dx=0
		\end{align*}
		for every $x$-periodic $\boldsymbol\psi\in H^1(\Omega_1)$ satisfying $\operatorname{div}\boldsymbol\psi=0$ and $\boldsymbol\psi|_{z=0}=\boldsymbol\psi|_{z=1}=0$. The lifting is linear and independent of time, and the estimates for this boundary value problem are
		\begin{align*}
			\lVert \mathscr{E}(\xi)\rVert_{L^2(\Omega_1)}\leq C\lVert \xi\rVert_{H_\sharp^{-1/2}(0,L)},
			\quad
			\lVert \mathscr{E}(\xi)\rVert_{H^1(\Omega_1)}\leq C\lVert \xi\rVert_{H_\sharp^{1/2}(0,L)},\quad
			\lVert \mathscr{E}(\xi)\rVert_{H^2(\Omega_1)}\leq C\lVert \xi\rVert_{H_\sharp^{3/2}(0,L)}.
		\end{align*}
		Moreover, for $\mathbf f\in L^2(0,1;H_\sharp^1(0,L))$,
		\begin{align}
			\left|\int_0^L\int_0^1\mathbf f\cdot\mathscr{E}(\xi)\,\mathrm dz\,\mathrm dx\right|
			\leq C\lVert \mathbf f\rVert_{L^2(0,1;H_\sharp^1(0,L))}\lVert \xi\rVert_{H_\sharp^{-3/2}(0,L)}.
			\label{eq:lower-lift-adjoint}
		\end{align}
		The constants depend only on $L$.

These bounds can be obtained directly from the horizontal Fourier modes. At a nonzero frequency $k\in(2\pi/L)\mathbb Z$, the vertical component $w_k=(\mathscr{E}(\xi)_k)_2$ solves
		\begin{align*}
			(\partial_z^2-k^2)^2w_k=0,\qquad
			w_k(0)=\partial_zw_k(0)=\partial_zw_k(1)=0,\qquad w_k(1)=\xi_k,
		\end{align*}
		and the horizontal component is $(\mathscr{E}(\xi)_k)_1=-(ik)^{-1}\partial_zw_k$. Solving this fourth-order equation with the basis $e^{|k|z}$, $e^{-|k|z}$, $ze^{|k|z}$, and $ze^{-|k|z}$, we integrate the resulting exponential terms and find
		\begin{align}
			\lVert \partial_z^j\mathscr{E}(\xi)_k\rVert_{L^2(0,1)}
			\leq C_j(1+k^2)^{j/2-1/4}|\xi_k|,\qquad j=0,1,2.
			\label{eq:lower-lift-modes}
		\end{align}
		The zero mode vanishes. Multiplying \eqref{eq:lower-lift-modes} by the appropriate powers of $|k|$ and summing by Parseval's identity proves the three estimates. The constructed velocity satisfies the stated Stokes problem. Uniqueness follows by testing the difference of two solutions by itself. The first estimate extends the solution to negative-order boundary data by transposition, cf.~\cite[Appendix~A]{Raymond2007}.
		
		Finally, applying Cauchy--Schwarz to the Fourier series, we obtain
		\begin{align*}
			\left|\int_{\Omega_1}\mathbf f\cdot\mathscr{E}(\xi)\,\mathrm dx\,\mathrm dz\right|
			&\leq C\sum_{k\neq0}\lVert \mathbf f_k\rVert_{L^2(0,1)}(1+k^2)^{-1/4}|\xi_k|\\
			&\leq C\left(\sum_k(1+k^2)\lVert \mathbf f_k\rVert_{L^2(0,1)}^2\right)^{1/2}
			\left(\sum_{k\neq0}(1+k^2)^{-3/2}|\xi_k|^2\right)^{1/2},
		\end{align*}
		which is \eqref{eq:lower-lift-adjoint}. Thus the time derivative of the fluid test can be estimated by duality.

\label{lem:lower-commutator}
		Let $N\geq1$ and $\Lambda_N=(N^2-\partial_x^2)^{1/2}$ on periodic functions. For a scalar periodic $g\in H_\sharp^1(0,L)$ and $f\in L_\sharp^2(0,L)$,
		\begin{align}
			\lVert [\Lambda_N^{-1},g]f\rVert_{L_\sharp^2(0,L)}
			&\leq C N^{-3/2}\lVert \partial_xg\rVert_{L_\sharp^2(0,L)}\lVert f\rVert_{L_\sharp^2(0,L)},\nonumber\\
			\lVert \Lambda_N[\Lambda_N^{-1},g]f\rVert_{L_\sharp^2(0,L)}
			&\leq C N^{-1/2}\lVert \partial_xg\rVert_{L_\sharp^2(0,L)}\lVert f\rVert_{L_\sharp^2(0,L)},\nonumber\\
			\lVert \Lambda_N^{1/2}[\Lambda_N^{-1},g]f\rVert_{L_\sharp^2(0,L)}
			&\leq C N^{-1}\lVert \partial_xg\rVert_{L_\sharp^2(0,L)}\lVert f\rVert_{L_\sharp^2(0,L)}.
			\label{eq:lower-sharp-commutator}
		\end{align}
		For rougher coefficients $g\in H_\sharp^{1/2}(0,L)$,
		\begin{align}
			\lVert \Lambda_N^{1/2}[\Lambda_N^{-1},g]f\rVert_{L_\sharp^2(0,L)}
			\leq C_N\lVert g\rVert_{H_\sharp^{1/2}(0,L)}\lVert f\rVert_{L_\sharp^2(0,L)}.
			\label{eq:lower-rough-commutator}
		\end{align}
		Moreover, if $g\in H_\sharp^1(0,L)$ is a symmetric matrix with $g\geq g_0 I$, where $g_0>0$, then, for every periodic vector field $f\in L_\sharp^2(0,L)$,
		\begin{align}
			\int_0^L(gf)\cdot\Lambda_N^{-1}f\,\mathrm dx
			\geq \left(g_0-CN^{-1/2}\lVert \partial_xg\rVert_{L_\sharp^2(0,L)}\right)
			\lVert \Lambda_N^{-1/2}f\rVert_{L_\sharp^2(0,L)}^2.
			\label{eq:lower-positive-mass}
		\end{align}
		Here the matrix norm in the error can be taken to be the sum of the component norms. The constants $C$ are independent of $N\geq1$.

Indeed, $|g(x)-g(y)|\leq \lVert \partial_xg\rVert_{L_\sharp^2(0,L)}d(x,y)^{1/2}$, where $d$ denotes periodic distance. The periodized kernel of $\Lambda_N^{-1}$ has moment of order $1/2$ bounded by $CN^{-3/2}$ and its derivative has the corresponding moment bounded by $CN^{-1/2}$. These statements follow by scaling the Bessel kernel on the line and then periodizing. Moreover,

		To show \eqref{eq:lower-rough-commutator}, write the commutator as $\Lambda_N^{-1}(gf)-g\Lambda_N^{-1}f$. In one dimension $H^{1/2}\cdot L^2\subset H^{-1/2}$ and $H^{1/2}\cdot H^1\subset H^{1/2}$. 			\leq C d(r,0)^{3/4}\lVert \partial_xg\rVert_{L_\sharp^2(0,L)}.

			\Lambda_N^{1/2}f
			=N^{1/2}f+

			e^{-N^2r}r^{-5/4}(f-e^{r\partial_x^2}f)\,\mathrm dr.
		\end{align*}

			\leq CN^{-1/4}\lVert \Lambda_N^{-1/2} f\rVert_{L_\sharp^2(0,L)},
		\end{align*}
		which proves \eqref{eq:lower-positive-mass}.

		\begin{align}
			&\frac{\mathrm d}{\mathrm dt}\Bigg[
			\frac\mu2\sum_{i,j=1}^2\int_{\Omega_1}(A_h)_{ij}
			\partial_i(B_h^{-\top}\mathbf v)\cdot\partial_j(B_h^{-\top}\Lambda_N^{-1}\mathbf v)\,\mathrm dx\,\mathrm dz\nonumber\\
			&\hspace{22mm}+\int_0^L\left(\alpha\partial_x^2h\,\Lambda_N^{-1}\partial_t\partial_x^2h
			+\beta\partial_xh\,\Lambda_N^{-1}\partial_t\partial_xh\right)\,\mathrm dx\Bigg]\nonumber\\
			&\quad+\rho_f\int_{\Omega_1}(hB_h^{-1}B_h^{-\top}\partial_t\mathbf v)\cdot\Lambda_N^{-1}\partial_t\mathbf v\,\mathrm dx\,\mathrm dz
			+\rho_s\lVert \Lambda_N^{-1/2}\partial_t^2h\rVert_{L_\sharp^2(0,L)}^2\nonumber\\
			&=\alpha\lVert \Lambda_N^{-1/2}\partial_t\partial_x^2h\rVert_{L_\sharp^2(0,L)}^2
			+\beta\lVert \Lambda_N^{-1/2}\partial_t\partial_xh\rVert_{L_\sharp^2(0,L)}^2\nonumber\\
			&\quad+\frac\mu2\sum_{i,j=1}^2\int_{\Omega_1}\Big[
			(\partial_tA_h)_{ij}\partial_i(B_h^{-\top}\mathbf v)\cdot\partial_j(B_h^{-\top}\Lambda_N^{-1}\mathbf v)\nonumber\\
			&\hspace{22mm}+(A_h)_{ij}\partial_i((\partial_tB_h^{-\top})\mathbf v)\cdot\partial_j(B_h^{-\top}\Lambda_N^{-1}\mathbf v)
			+(A_h)_{ij}\partial_i(B_h^{-\top}\mathbf v)\cdot\partial_j((\partial_tB_h^{-\top})\Lambda_N^{-1}\mathbf v)
			\Big]\,\mathrm dx\,\mathrm dz\nonumber\\
			&\quad-\frac\mu2\sum_{i,j=1}^2\int_{\Omega_1}(A_h)_{ij}\Big[
			\partial_i(B_h^{-\top}\mathbf v)\cdot\partial_j(B_h^{-\top}\Lambda_N^{-1}\partial_t\mathbf v)
			-\partial_i(B_h^{-\top}\Lambda_N^{-1}\mathbf v)\cdot\partial_j(B_h^{-\top}\partial_t\mathbf v)
			\Big]\,\mathrm dx\,\mathrm dz\nonumber\\
			&\quad-\rho_f\int_{\Omega_1}h(\partial_tB_h^{-\top})\mathbf v\cdot B_h^{-\top}\Lambda_N^{-1}\partial_t\mathbf v\,\mathrm dx\,\mathrm dz\nonumber\\
			&\quad-\rho_f\int_{\Omega_1}\big[(\widehat\mathbf u-\partial_t\chi_h)\cdot(B_h\nabla)\big]\widehat\mathbf u
			\cdot B_h^{-\top}\Lambda_N^{-1}\partial_t\mathbf v\,\mathrm dx\,\mathrm dz
			\eqqcolon\sum_{i=1}^5 K_i.
			\label{eq:lower-acceleration-identity}
		\end{align}
		Here $K_1$ is the sum of the two structural norms.
		The coefficients $\rho_fhB_h^{-1}B_h^{-\top}$ and the full principal block matrix $\big((A_h)_{ij}B_h^{-1}B_h^{-\top}\big)_{i,j=1}^2$ are uniformly positive and bounded in $L^\infty(0,1;H_\sharp^1(0,L))$ using only the energy and gap. By Lemma~\ref{lem:lower-commutator}, we may choose $N\geq N_0$ depending only on these quantities so that
		\begin{align*}
			&\rho_f\int_{\Omega_1}(hB_h^{-1}B_h^{-\top}\partial_t\mathbf v)\cdot\Lambda_N^{-1}\partial_t\mathbf v\,\mathrm dx\,\mathrm dz
			+\rho_s\lVert \Lambda_N^{-1/2}\partial_t^2h\rVert_{L_\sharp^2(0,L)}^2\\
			&\qquad\geq c\left(\lVert \Lambda_N^{-1/2}\partial_t\mathbf v\rVert_{L^2(\Omega_1)}^2
			+\lVert \Lambda_N^{-1/2}\partial_t^2h\rVert_{L_\sharp^2(0,L)}^2\right).
		\end{align*}
		The same choice ensures positivity of the principal part of the viscous quadratic form inside the time derivative in \eqref{eq:lower-acceleration-identity}.
		
		We now estimate the right-hand side term by term. From the Fourier symbol of $\Lambda_N^{-1/2}$ and \eqref{eq:lower-velocity-trace}, we obtain, uniformly for $N\geq1$,
		\begin{align*}
			0\leq K_1\leq C\lVert \partial_th\rVert_{H_\sharp^{3/2}(0,L)}^2
			\leq C\lVert \partial_x\mathbf v\rVert_{H^1(\Omega_1)}^2.
		\end{align*}
		
		To estimate $K_2$ and $K_3$, we first expand the viscous form. For an arbitrary test $\boldsymbol\varphi$, its expansion is
		\begin{align}
			&\mu\sum_{i,j=1}^2\int_{\Omega_1}(A_h)_{ij}
			\partial_i(B_h^{-\top}\mathbf v)\cdot\partial_j(B_h^{-\top}\boldsymbol\varphi)\,\mathrm dx\,\mathrm dz\nonumber\\
			&=\mu\sum_{i,j=1}^2\int_{\Omega_1}\Big[
			((A_h)_{ij}B_h^{-1}B_h^{-\top}\partial_i\mathbf v)\cdot\partial_j\boldsymbol\varphi
			+((A_h)_{ij}(\partial_jB_h^{-1})B_h^{-\top}\partial_i\mathbf v)\cdot\boldsymbol\varphi\nonumber\\
			&\hspace{27mm}+((A_h)_{ij}B_h^{-1}(\partial_iB_h^{-\top})\mathbf v)\cdot\partial_j\boldsymbol\varphi
			+((A_h)_{ij}(\partial_jB_h^{-1})(\partial_iB_h^{-\top})\mathbf v)\cdot\boldsymbol\varphi
			\Big]\,\mathrm dx\,\mathrm dz.
			\label{eq:lower-coefficient-expansion}
		\end{align}
		Here $\partial_1=\partial_x$ and $\partial_2=\partial_z$. We keep the four coefficient families separate, since only the principal coefficient needs a decaying power of $N$.
		
		The mechanical energy and the positive gap imply
		\begin{align}
			&\lVert \rho_fhB_h^{-1}B_h^{-\top}\rVert_{L^\infty(0,1;H_\sharp^1(0,L))}
			+\sum_{i,j=1}^2\lVert (A_h)_{ij}B_h^{-1}B_h^{-\top}\rVert_{L^\infty(0,1;H_\sharp^1(0,L))}\leq C,
			\\
			&\sum_{i,j=1}^2\Big(
			\lVert \partial_x((A_h)_{ij}B_h^{-1}B_h^{-\top})\rVert_{L^\infty(0,1;H_\sharp^{1/2}(0,L))}
			+\lVert (A_h)_{ij}B_h^{-1}(\partial_iB_h^{-\top})\rVert_{L^\infty(0,1;H_\sharp^{1/2}(0,L))}\Big)\nonumber\\
			&\qquad\leq C\left(1+\lVert \partial_x^3h\rVert_{L_\sharp^2(0,L)}^{1/2}\right).
		\end{align}
		The transposed mixed family has the same bound. A horizontal derivative of a mixed coefficient and an undifferentiated zeroth-order coefficient satisfy
		\begin{align}
			&\sum_{i,j=1}^2\lVert \partial_x((A_h)_{ij}B_h^{-1}(\partial_iB_h^{-\top}))\rVert_{L^\infty(0,1;H_\sharp^{1/2}(0,L))}
			\leq C\left(1+\lVert \partial_x^3h\rVert_{L_\sharp^2(0,L)}+\lVert h\rVert_{H_\sharp^{7/2}(0,L)}\right),\nonumber\\
			&\sum_{i,j=1}^2\lVert (A_h)_{ij}(\partial_jB_h^{-1})(\partial_iB_h^{-\top})\rVert_{L^\infty(0,1;H_\sharp^{1/2}(0,L))}
			\leq C\left(1+\lVert \partial_x^3h\rVert_{L_\sharp^2(0,L)}\right).
		\end{align}
		The same bounds hold after the $z$ derivatives used below. Indeed, these coefficients are finite polynomials in $z$. Their scalar coefficient functions depend smoothly on $h,h^{-1},\partial_xh$, with at most a linear factor of $\partial_x^2h$ in the mixed terms and a quadratic factor in the zeroth-order terms. Smooth functions of $h,h^{-1},\partial_xh$ have uniformly bounded $H_\sharp^1(0,L)$ norms. The only quadratic structural product needing explanation is controlled by
		\begin{align*}
			\lVert (\partial_x^2h)^2\rVert_{H_\sharp^{1/2}(0,L)}
			\leq C\lVert \partial_x^2h\rVert_{L_\sharp^2(0,L)}
			\lVert \partial_x^2h\rVert_{H_\sharp^1(0,L)}.
		\end{align*}
		This follows from the one-dimensional fractional product estimate and interpolation, or from $\lVert f\rVert_{L_\sharp^4(0,L)}\leq C\lVert f\rVert_{L_\sharp^2(0,L)}^{3/4}\lVert f\rVert_{H_\sharp^1(0,L)}^{1/4}$ and $\lVert (1-\partial_x^2)^{1/4}f\rVert_{L_\sharp^4(0,L)}\leq C\lVert f\rVert_{L_\sharp^2(0,L)}^{1/4}\lVert f\rVert_{H_\sharp^1(0,L)}^{3/4}$.
		
		We now estimate $K_2$, the coefficient-time integral in \eqref{eq:lower-acceleration-identity}. In the principal coefficients a time derivative produces $\partial_th$ and $\partial_t\partial_xh$. The mixed coefficients additionally produce $\partial_t\partial_x^2h$, and the zeroth-order coefficients produce $\partial_x^2h\,\partial_t\partial_x^2h$. Integrate the latter two derivatives once in $x$. The resulting products are obtained directly from \eqref{eq:lower-coefficient-expansion}, with at most the following structural factors:
		\begin{align*}
			&\partial_th,\quad \partial_t\partial_xh,\quad
			\partial_th\,\partial_x^2h,\quad
			\partial_t\partial_xh\,\partial_x^2h,\quad
			\partial_th\,(\partial_x^2h)^2,\quad
			\partial_t\partial_xh\,(\partial_x^2h)^2,\quad
			\partial_t\partial_xh\,\partial_x^3h.
		\end{align*}
		They multiply two fluid factors, one of which is acted on by $\Lambda_N^{-1}$. The only second fluid derivatives generated in this integration by parts have a horizontal derivative and are bounded by $\lVert \partial_x\mathbf v\rVert_{H^1(\Omega_1)}$. The estimates use
		\begin{align*}
			&\lVert \partial_th\rVert_{L_\sharp^\infty(0,L)}\leq C \mathcal E_1^{1/4},\qquad
			\lVert \partial_x^2h\rVert_{L_\sharp^\infty(0,L)}\leq C(1+\lVert \partial_x^3h\rVert_{L_\sharp^2(0,L)}^{1/2}),\\
			&\lVert \Lambda_N^{-1}f\rVert_{L_\sharp^\infty(0,L;L^2(0,1))}\leq C_N\lVert f\rVert_{L^2(\Omega_1)},\\
			&\lVert \nabla\mathbf v\rVert_{L_\sharp^\infty(0,L;L^2(0,1))}
			\leq C (1+\mathcal D)^{1/4}((1+\mathcal D)^{1/4}+\lVert \partial_x\mathbf v\rVert_{H^1(\Omega_1)}^{1/2}).
		\end{align*}
		Here the first inequality also uses the energy bound for $\partial_th$ in $L^2$. To make the highest products explicit, their absolute integrals over $\Omega_1$ satisfy
		\begin{align*}
			&\int_{\Omega_1}|\partial_t\partial_xh|\,|\Lambda_N^{-1}\mathbf v|\,
			|\nabla\partial_x\mathbf v|
			\leq C_N\mathcal E_1^{1/2}\lVert \partial_x\mathbf v\rVert_{H^1(\Omega_1)}\leq\delta \lVert \partial_x\mathbf v\rVert_{H^1(\Omega_1)}^2+C_{\delta,N}(1+\mathcal D)\mathcal E_1,\\
			&\int_{\Omega_1}|\partial_t\partial_xh|\,|\partial_x^3h|\,
			|\mathbf v|\,|\Lambda_N^{-1}\mathbf v|
			\leq C_N (1+\mathcal D)^{1/4}\mathcal E_1\leq C_N(1+\mathcal D)\mathcal E_1,\\
			&\int_{\Omega_1}|\partial_t\partial_xh|\,|\partial_x^2h|^2\,
			|\mathbf v|\,|\Lambda_N^{-1}\mathbf v|
			\leq C_N(1+\mathcal D)^{1/4}\mathcal E_1^{3/4}\leq C_N(1+\mathcal D)\mathcal E_1,\\
			&\int_{\Omega_1}|\partial_t\partial_xh|\,|\partial_x^2h|\,
			|\nabla\mathbf v|\,|\Lambda_N^{-1}\mathbf v|
			\leq C_N\mathcal E_1^{1/2}(1+\mathcal D)^{1/4}\left((1+\mathcal D)^{1/4}+\lVert \partial_x\mathbf v\rVert_{H^1(\Omega_1)}^{1/2}\right)\\
			&\phantom{\int_{\Omega_1}|\partial_t\partial_xh|\,|\partial_x^2h|\,
				|\nabla\mathbf v|\,|\Lambda_N^{-1}\mathbf v|}\leq\delta \lVert \partial_x\mathbf v\rVert_{H^1(\Omega_1)}^2+C_{\delta,N}(1+\mathcal D)\mathcal E_1.
		\end{align*}
		Each estimate uses Cauchy--Schwarz in $z$ before applying H\"older's inequality in $x$. The other products have fewer derivatives and satisfy the same bound. Thus, expanding $2K_2$ in \eqref{eq:lower-acceleration-identity}, we obtain
		\begin{align}
			&\mu\Bigg|\sum_{i,j=1}^2\int_{\Omega_1}\Big[
			(\partial_t((A_h)_{ij}B_h^{-1}B_h^{-\top})\partial_i\mathbf v)\cdot\partial_j\Lambda_N^{-1}\mathbf v\nonumber\\
			&\hspace{16mm}+(\partial_t((A_h)_{ij}(\partial_jB_h^{-1})B_h^{-\top})\partial_i\mathbf v)\cdot\Lambda_N^{-1}\mathbf v
			+(\partial_t((A_h)_{ij}B_h^{-1}(\partial_iB_h^{-\top}))\mathbf v)\cdot\partial_j\Lambda_N^{-1}\mathbf v\nonumber\\
			&\hspace{16mm}+(\partial_t((A_h)_{ij}(\partial_jB_h^{-1})(\partial_iB_h^{-\top}))\mathbf v)\cdot\Lambda_N^{-1}\mathbf v
			\Big]\,\mathrm dx\,\mathrm dz\Bigg|\nonumber\\
			&\qquad\leq\delta\lVert \partial_x\mathbf v\rVert_{H^1(\Omega_1)}^2+C_{\delta,N}(1+\mathcal D)\mathcal E_1.
		\end{align}
		Since $K_2$ is one half of this integral, we fix $\delta$, for example $\delta=1$, and obtain
		\begin{align*}
			|K_2|\leq C_N\lVert \partial_x\mathbf v\rVert_{H^1(\Omega_1)}^2+C_N(1+\mathcal D)\mathcal E_1.
		\end{align*}
		
		For $K_3$, we estimate the commutator difference in \eqref{eq:lower-acceleration-identity}. Expanding $-2K_3$ by \eqref{eq:lower-coefficient-expansion}, commuting the self-adjoint operator $\Lambda_N^{-1}$, and integrating by parts once, we obtain
		\begin{align}
			&\mu\sum_{i,j=1}^2\int_{\Omega_1}(A_h)_{ij}\Big[
			\partial_i(B_h^{-\top}\mathbf v)\cdot\partial_j(B_h^{-\top}\Lambda_N^{-1}\partial_t\mathbf v)
			-\partial_i(B_h^{-\top}\Lambda_N^{-1}\mathbf v)\cdot\partial_j(B_h^{-\top}\partial_t\mathbf v)
			\Big]\,\mathrm dx\,\mathrm dz\nonumber\\
			&=\mu\sum_{i,j=1}^2\int_{\Omega_1}\Big\{
			-\partial_j\big([\Lambda_N^{-1},(A_h)_{ij}B_h^{-1}B_h^{-\top}]\partial_i\mathbf v\big)
			-\partial_j\big([\Lambda_N^{-1},(A_h)_{ij}B_h^{-1}(\partial_iB_h^{-\top})]\mathbf v\big)\nonumber\\
			&\hspace{24mm}+[\Lambda_N^{-1},(A_h)_{ij}(\partial_jB_h^{-1})B_h^{-\top}]\partial_i\mathbf v
			+[\Lambda_N^{-1},(A_h)_{ij}(\partial_jB_h^{-1})(\partial_iB_h^{-\top})]\mathbf v
			\Big\}\cdot\partial_t\mathbf v\,\mathrm dx\,\mathrm dz\nonumber\\
			&\quad+\mu\sum_{i=1}^2\int_0^L\Big(
			[\Lambda_N^{-1},(A_h)_{i2}B_h^{-1}B_h^{-\top}]\partial_i\mathbf v
			+[\Lambda_N^{-1},(A_h)_{i2}B_h^{-1}(\partial_iB_h^{-\top})]\mathbf v
			\Big)_2\big|_{z=1}\partial_t^2h\,\mathrm dx.
			\label{eq:lower-commutator-boundary}
		\end{align}
		The bottom contribution vanishes because $\partial_t\mathbf v=0$ there. To proceed, we first expand
		\begin{align*}
			\partial_j\big([\Lambda_N^{-1},(A_h)_{ij}B_h^{-1}B_h^{-\top}]
			\partial_i\mathbf v\big)=[\Lambda_N^{-1},(A_h)_{ij}B_h^{-1}B_h^{-\top}]
			\partial_j\partial_i\mathbf v
			+[\Lambda_N^{-1},\partial_j((A_h)_{ij}B_h^{-1}B_h^{-\top})]
			\partial_i\mathbf v.
		\end{align*}
		The same product rule applies to the differentiated mixed commutator. For the terms with two derivatives on $\mathbf v$, \eqref{eq:lower-sharp-commutator} yields
		\begin{align*}
			\sum_{i,j=1}^2
			\lVert \Lambda_N^{1/2}
				[\Lambda_N^{-1},(A_h)_{ij}B_h^{-1}B_h^{-\top}]
				\partial_j\partial_i\mathbf v\rVert_{L^2(\Omega_1)}
			\leq CN^{-1}\lVert \mathbf v\rVert_{H^2(\Omega_1)}.
		\end{align*}
		Among the other terms, the three containing a first derivative of $\mathbf v$ have a differentiated principal coefficient or an undifferentiated mixed coefficient. Their $H_\sharp^{1/2}(0,L)$ norms are bounded by $C(1+\lVert \partial_x^3h\rVert_{L_\sharp^2(0,L)}^{1/2})$, uniformly in $z$. The two terms containing $\mathbf v$ have a differentiated mixed coefficient or a zeroth-order coefficient, with norms bounded by $C(1+\lVert \partial_x^3h\rVert_{L_\sharp^2(0,L)}+\lVert h\rVert_{H_\sharp^{7/2}(0,L)})$. We apply \eqref{eq:lower-rough-commutator} to these five terms.
		
		At $z=1$, both $\mathbf v$ and $\Lambda_N^{-1}\mathbf v$ have vertical traces, while $\partial_iB_h^{-\top}$ has zero second column. Hence
		\begin{align*}
			[\Lambda_N^{-1},(A_h)_{i2}B_h^{-1}(\partial_iB_h^{-\top})]
			\mathbf v\big|_{z=1}=0.
		\end{align*}
		The principal boundary terms satisfy, by \eqref{eq:lower-sharp-commutator} and the fixed-strip trace theorem,
		\begin{align*}
			\sum_{i=1}^2
			\lVert \left.\Lambda_N^{1/2}
				[\Lambda_N^{-1},(A_h)_{i2}B_h^{-1}B_h^{-\top}]
				\partial_i\mathbf v\right|_{z=1}\rVert_{L_\sharp^2(0,L)}
				\leq CN^{-1}\sum_{i=1}^2
			\lVert \partial_i\mathbf v|_{z=1}\rVert_{L_\sharp^2(0,L)}
			\leq CN^{-1}\lVert \mathbf v\rVert_{H^2(\Omega_1)}.
		\end{align*}
		Pairing the volume and boundary terms with $\Lambda_N^{-1/2}\partial_t\mathbf v$ and $\Lambda_N^{-1/2}\partial_t^2h$, respectively, we obtain
		\begin{align*}
			2|K_3|
			&\leq \bigg[
			CN^{-1}\lVert \mathbf v\rVert_{H^2(\Omega_1)}
			+C_N\left(1+\lVert \partial_x^3h\rVert_{L_\sharp^2(0,L)}^{1/2}\right)
			\lVert \nabla\mathbf v\rVert_{L^2(\Omega_1)}\\
			&\hspace{13mm}
			+C_N\left(1+\lVert \partial_x^3h\rVert_{L_\sharp^2(0,L)}
			+\lVert h\rVert_{H_\sharp^{7/2}(0,L)}\right)
			\lVert \mathbf v\rVert_{L^2(\Omega_1)}
			\bigg]\lVert \Lambda_N^{-1/2}\partial_t\mathbf v\rVert_{L^2(\Omega_1)}\\
			&\quad+CN^{-1}\lVert \mathbf v\rVert_{H^2(\Omega_1)}
			\lVert \Lambda_N^{-1/2}\partial_t^2h\rVert_{L_\sharp^2(0,L)}.
		\end{align*}
		Using $\lVert \mathbf v\rVert_{L^2(\Omega_1)}\leq C$, $\lVert \nabla\mathbf v\rVert_{L^2(\Omega_1)}^2\leq C\mathcal D$, and $1+\lVert \partial_x^3h\rVert_{L_\sharp^2(0,L)}^2\leq C\mathcal E_1$, it follows from Young's inequality that
		\begin{align*}
			2|K_3|
			&\leq\delta\left(
			\lVert \Lambda_N^{-1/2}\partial_t\mathbf v\rVert_{L^2(\Omega_1)}^2
			+\lVert \Lambda_N^{-1/2}\partial_t^2h\rVert_{L_\sharp^2(0,L)}^2\right)
			+C_\delta N^{-2}\lVert \mathbf v\rVert_{H^2(\Omega_1)}^2\\
			&\quad+C_{\delta,N}\left((1+\mathcal D)\mathcal E_1
			+\lVert h\rVert_{H_\sharp^{7/2}(0,L)}^2\right).
		\end{align*}
		
		Finally, since $\lVert \Lambda_N^{-1}\partial_t\mathbf v\rVert_{L^2(\Omega_1)}\leq N^{-1/2}\lVert \Lambda_N^{-1/2}\partial_t\mathbf v\rVert_{L^2(\Omega_1)}$, we apply Young's inequality and \eqref{eq:lower-nonlinear} with its parameter proportional to $N^{-1}$. For $K_4$ and $K_5$, we obtain, for every fixed $\delta>0$,
		\begin{align*}
			|K_4|+|K_5|\leq\delta\lVert \Lambda_N^{-1/2}\partial_t\mathbf v\rVert_{L^2(\Omega_1)}^2
			+C_\delta N^{-2}\lVert \mathbf v\rVert_{H^2(\Omega_1)}^2+C_{\delta,N}(1+\mathcal D)\mathcal E_1.
		\end{align*}
		
		We next estimate the two expressions inside the time derivative in \eqref{eq:lower-acceleration-identity}. The structural squares generated by integration by parts in time constitute $K_1$ and have already been estimated. For the elastic pairing, we have
		\begin{align*}
			\left|\int_0^L\left(\alpha\partial_x^2h\,\Lambda_N^{-1}\partial_t\partial_x^2h
			+\beta\partial_xh\,\Lambda_N^{-1}\partial_t\partial_xh\right)\,\mathrm dx\right|
			\leq C\lVert h\rVert_{H_\sharp^3(0,L)}\lVert \partial_th\rVert_{L_\sharp^2(0,L)}
			\leq \eta\mathcal E_1+C_\eta.
		\end{align*}
		
		For the fluid viscous quadratic form at time $t$, we first consider the terms in which both spatial derivatives fall on the fluid factors. The full principal block is uniformly positive: for every $\boldsymbol\xi_1,\boldsymbol\xi_2\in\mathbb R^2$,
		\begin{align*}
			\sum_{i,j=1}^2
			\big((A_h)_{ij}B_h^{-1}B_h^{-\top}\boldsymbol\xi_i\big)
			\cdot\boldsymbol\xi_j
			&=\sum_{i,j=1}^2(A_h)_{ij}
			(B_h^{-\top}\boldsymbol\xi_i)\cdot
			(B_h^{-\top}\boldsymbol\xi_j)\\
			&\geq c\sum_{i=1}^2
			|B_h^{-\top}\boldsymbol\xi_i|^2
			\geq c_0\sum_{i=1}^2|\boldsymbol\xi_i|^2.
		\end{align*}
		Moreover, its entries have uniformly bounded horizontal $H_\sharp^1(0,L)$ norms, with constants depending only on the energy and gap. Applying \eqref{eq:lower-positive-mass} to this block and the vector $(\partial_x\mathbf v,\partial_z\mathbf v)$ at each fixed $z$, and then integrating in $z$, we obtain
		\begin{align*}
			&\sum_{i,j=1}^2\int_{\Omega_1}
			\big((A_h)_{ij}B_h^{-1}B_h^{-\top}\partial_i\mathbf v\big)
			\cdot\Lambda_N^{-1}\partial_j\mathbf v\,\mathrm dx\,\mathrm dz\\
			&\qquad\geq
			(c_0-CN^{-1/2})
			\lVert \Lambda_N^{-1/2}\nabla\mathbf v\rVert_{L^2(\Omega_1)}^2
			\geq \frac{c_0}{2}
			\lVert \Lambda_N^{-1/2}\nabla\mathbf v\rVert_{L^2(\Omega_1)}^2,
		\end{align*}
		provided $N$ is sufficiently large. This is the positive principal contribution used to absorb the mixed terms below. For either mixed term, move $\Lambda_N^{-1/2}$ to the differentiated fluid factor and use the product estimates underlying \eqref{eq:lower-rough-commutator}. The mixed coefficient has $H_\sharp^{1/2}(0,L)$ norm bounded by $C(1+\lVert \partial_x^3h\rVert_{L_\sharp^2(0,L)}^{1/2})$, while $\lVert \Lambda_N^{-1}\mathbf v\rVert_{L^2(0,1;H_\sharp^1(0,L))}\leq C_N\lVert \mathbf v\rVert_{L^2(\Omega_1)}$ is energy controlled. 
		For the zeroth-order family in \eqref{eq:lower-coefficient-expansion}, set
		\begin{align*}
			\mathbf Z_{ij}=(A_h)_{ij}(\partial_jB_h^{-1})(\partial_iB_h^{-\top}).
		\end{align*}
		The energy bound and one-dimensional interpolation imply
		\begin{align*}
			\lVert \mathbf Z_{ij}\rVert_{L_\sharp^2(0,L;L^\infty(0,1))}
			&\leq C\left(1+\lVert \partial_x^2h\rVert_{L_\sharp^4(0,L)}^2\right)
			\leq C\left(1+\lVert \partial_x^3h\rVert_{L_\sharp^2(0,L)}^{1/2}\right),\\
			\lVert \Lambda_N^{-1}\mathbf v\rVert_{L^\infty(0,L;L^2(0,1))}
			&\leq C_N\lVert \mathbf v\rVert_{L^2(\Omega_1)}.
		\end{align*}
		Applying Cauchy--Schwarz in $z$ and H\"older's inequality in $x$ , we obtain
		\begin{align*}
			\left|\int_{\Omega_1}(\mathbf Z_{ij}\mathbf v)\cdot\Lambda_N^{-1}\mathbf v\,\mathrm dx\,\mathrm dz\right|
			&\leq\lVert \mathbf Z_{ij}\rVert_{L_\sharp^2(0,L;L^\infty(0,1))}
			\lVert \mathbf v\rVert_{L^2(\Omega_1)}
			\lVert \Lambda_N^{-1}\mathbf v\rVert_{L^\infty(0,L;L^2(0,1))}\\
			&\leq C_N\left(1+\lVert \partial_x^3h\rVert_{L_\sharp^2(0,L)}^{1/2}\right).
		\end{align*}
		The product of the two fluid norms in the preceding estimate is bounded by $C_N\lVert \mathbf v\rVert_{L^2(\Omega_1)}^2$, which is controlled by the mechanical energy. Combining the principal, mixed, and zeroth-order terms, we conclude that
		\begin{align}
			&\mu\sum_{i,j=1}^2\int_{\Omega_1}(A_h)_{ij}
			\partial_i(B_h^{-\top}\mathbf v)\cdot\partial_j(B_h^{-\top}\Lambda_N^{-1}\mathbf v)\,\mathrm dx\,\mathrm dz\nonumber\\
			&\geq c\lVert \Lambda_N^{-1/2}\nabla\mathbf v\rVert_{L^2(\Omega_1)}^2
			-C_N\left(1+\lVert \partial_x^3h\rVert_{L_\sharp^2(0,L)}\right)
			\geq -\eta\mathcal E_1-C_{\eta,N}.
		\end{align}
		The absolute value of the expression inside the time derivative in \eqref{eq:lower-acceleration-identity}, evaluated at $t=0$, is bounded by $C_N(1+\mathcal E_1(0))$.
		
		Now we integrate \eqref{eq:lower-acceleration-identity} over $(0,t)$. Choose the small parameters in the estimates of $K_3,K_4,K_5$ so that their acceleration terms are absorbed by the mass coercivity. In view of the estimates of $K_1,K_2$ and the preceding bounds at times $0$ and $t$, we conclude that
		\begin{align*}
			&\int_0^t\left(\lVert \Lambda_N^{-1/2}\partial_t\mathbf v\rVert_{L^2(\Omega_1)}^2
			+\lVert \Lambda_N^{-1/2}\partial_t^2h\rVert_{L_\sharp^2(0,L)}^2\right)\,\mathrm ds\\
			&\quad\leq\eta\mathcal E_1(t)+C_{\eta,N}(1+\mathcal E_1(0))
			+CN^{-2}\int_0^t\lVert \mathbf v\rVert_{H^2(\Omega_1)}^2\,\mathrm ds
			\\&\qquad+C_N\int_0^t\left(\lVert \partial_x\mathbf v\rVert_{H^1(\Omega_1)}^2+(1+\mathcal D)\mathcal E_1+\lVert h\rVert_{H_\sharp^{7/2}(0,L)}^2\right)\,\mathrm ds.
		\end{align*}
		Applying the preceding estimate with $\eta/2$ in place of $\eta$, and then using \eqref{eq:lower-height-bound} with $\varepsilon$ chosen so that $C_N\varepsilon\leq\eta/2$. we record \eqref{eq:lower-filtered-bound}. For $N\geq1$, the Fourier symbols satisfy
		\begin{align*}
			(1+k^2)^{-1/2}\leq N(N^2+k^2)^{-1/2},
			\qquad k\in(2\pi/L)\mathbb Z.
		\end{align*}
		Consequently,
		\begin{align}
			\lVert f\rVert_{H_\sharp^{-1/2}(0,L)}^2
			\leq CN\lVert \Lambda_N^{-1/2}f\rVert_{L_\sharp^2(0,L)}^2.
			\label{eq:lower-norm}
		\end{align}
		which finishes the proof.

We use the higher strong theory of \cite{SchwarzacherSu2025} to construct exact solutions of the undamped system from compatible smooth approximations of the initial data. The separation and lower-order estimates therefore apply directly to these solutions. Alternatively, one could use the global damped solutions of \cite{GrandmontHillairet2016}, with the prescribed initial data at the lower regularity level, and pass to the limit as the damping coefficient $\gamma$ tends to zero. This would require extending Theorem~\ref{thm:separation} and Proposition~\ref{prop:lower-estimates} to the beam equation with the additional term $-\gamma\partial_x^2\partial_th$, with constants independent of $\gamma$. The additional contribution in the distance test can be controlled by the bending coercivity and the damped energy identity. In the lower-order estimates, the horizontal test introduces nonnegative dissipation, while the fractional height test and the acceleration test produce total time derivatives of nonnegative quadratic forms. After integration in time, these quadratic forms are nonnegative at time $t$, and their initial values are uniformly bounded. The same compactness argument then applies.

<!-- proposition-2: body_md -->
\label{prop:weighted-estimates}
		Let $0<T<\infty$ and let $(h,\mathbf u,p)$ be an exact higher strong solution on $[0,T]$ with bounded mechanical energy and $h\geq \min h>0$. Then
		\begin{align}
			&\mathop{\mathrm{ess\,sup}}_{0<t\leq T}t^2\left(
			\lVert \partial_t\mathbf u(t)\rVert_{L^2(\Omega_h(t))}^2
			+\lVert \partial_t^2h(t)\rVert_{L_\sharp^2(0,L)}^2
			+\lVert \partial_th(t)\rVert_{H_\sharp^2(0,L)}^2\right.\nonumber\\
			&\hspace{41mm}\left.
			+\lVert \mathbf u(t)\rVert_{H^2(\Omega_h(t))}^2
			+\lVert p(t)\rVert_{H^1(\Omega_h(t))}^2
			+\lVert h(t)\rVert_{H_\sharp^4(0,L)}^2\right)\nonumber\\
			&\qquad
			+\int_0^Tt^2
			\lVert \nabla\partial_t\mathbf u(t)\rVert_{L^2(\Omega_h(t))}^2
			\,\mathrm dt
			\leq C_T.
			\label{eq:weighted-bound}
		\end{align}
		The constant $C_T>0$ depends only on $T$, $(\min h)^{-1}$, the initial mechanical energy, the mean height, the period, the physical parameters, and the three initial norms in \eqref{eq:data-regularity}. It is independent of the higher initial norms and the initial acceleration.

<!-- proof-20: body_md -->
We first propagate the acceleration from a regular time $s$. By the acceleration estimate in \cite[Section~3.2 \& Theorem~4.4]{SchwarzacherSu2025}, we have, initially for almost every pair of regular times $0\leq s\leq t\leq T$ and then for the corresponding energy representatives,
		\begin{align}
			&1+\lVert \partial_t\widehat\mathbf u(t)\rVert_{L^2(\Omega_1)}^2
			+\lVert \partial_t^2h(t)\rVert_{L_\sharp^2(0,L)}^2
			+\lVert \partial_th(t)\rVert_{H_\sharp^2(0,L)}^2
			+\int_s^t\lVert \nabla\partial_t\widehat\mathbf u(r)\rVert_{L^2(\Omega_1)}^2\,\mathrm dr
			\nonumber\\
			&\qquad\leq C_T\left(1+\lVert \partial_t\widehat\mathbf u(s)\rVert_{L^2(\Omega_1)}^2
			+\lVert \partial_t^2h(s)\rVert_{L_\sharp^2(0,L)}^2
			+\lVert \partial_th(s)\rVert_{H_\sharp^2(0,L)}^2\right).
			\label{eq:weighted-propagation}
		\end{align}
		Here we have used the upper and lower bounds for $h$, together with the zero mean of $\partial_th$ and $\alpha>0$, to write the differentiated physical energy in terms of its equivalent Sobolev norms. The constant $1$ absorbs the lower-order terms in the source estimate. The corrected velocity derivative in the source calculation is precisely
		\begin{align}
			B_h^{-\top}\partial_t\mathbf v=\partial_t\widehat\mathbf u+
			\begin{pmatrix}
				h^{-1}\partial_th\,\widehat u_1\\
				z h^{-1}\partial_th\,\partial_xh\,\widehat u_1
				-z\partial_t\partial_xh\,\widehat u_1
			\end{pmatrix},
			\label{eq:weighted-correction}
		\end{align}
		where all terms can be controlled independently of higher norms at the initial time $s$.
		
		Note that
		\begin{align*}
			\int_0^T2s\int_s^T\lVert \nabla\partial_t\widehat\mathbf u(t)\rVert_{L^2(\Omega_1)}^2\,\mathrm dt\,\mathrm ds
			=\int_0^Tt^2\lVert \nabla\partial_t\widehat\mathbf u(t)\rVert_{L^2(\Omega_1)}^2\,\mathrm dt.
		\end{align*}
		Consequently, on the fixed interval $t \in [0,T]$, integrating \eqref{eq:weighted-propagation} against $s\,\mathrm ds$ on $(0,t)$ leads to
		\begin{align}
			&\mathop{\mathrm{ess\,sup}}_{0<t\leq T}t^2\left(\lVert \partial_t\widehat\mathbf u(t)\rVert_{L^2(\Omega_1)}^2
			+\lVert \partial_t^2h(t)\rVert_{L_\sharp^2(0,L)}^2
			+\lVert \partial_th(t)\rVert_{H_\sharp^2(0,L)}^2\right)
			+\int_0^Tt^2\lVert \nabla\partial_t\widehat\mathbf u\rVert_{L^2(\Omega_1)}^2\,\mathrm dt\nonumber\\
			&\quad\leq C_T+C_T\int_0^Tt\lVert \partial_t\widehat\mathbf u\rVert_{L^2(\Omega_1)}^2\,\mathrm dt
			+C_T\int_0^Tt\lVert \partial_t^2h\rVert_{L_\sharp^2(0,L)}^2\,\mathrm dt
			+C_T\int_0^Tt\lVert \partial_th\rVert_{H_\sharp^2(0,L)}^2\,\mathrm dt.
			\label{eq:weighted-averaging}
		\end{align}
		The fluid acceleration term on the right is controlled by the unweighted lower estimate. It remains to estimate the two beam terms.
		
		To estimate the beam velocity, retain the stationary lifting $\widetilde{\mathbf v}=\mathscr{E}(\partial_th)$ from Section~\ref{sec:lower-fluid}. Its boundary conditions and time derivative are given by Lemma~\ref{lem:lower-lift} and \eqref{eq:lower-lift-time}. Differentiating \eqref{eq:lower-strong-fluid} and beam equation \eqref{eq:beam} in time and testing with $(\boldsymbol\varphi,\zeta)=(\widetilde{\mathbf v},\partial_th)$, we obtain
		\begin{align}
			&\frac{\mathrm d}{\mathrm dt} \mathcal H
			+\alpha\int_0^L|\partial_t\partial_x^2h|^2\,\mathrm dx
			+\beta\int_0^L|\partial_t\partial_xh|^2\,\mathrm dx\nonumber\\
			&=\rho_s\int_0^L|\partial_t^2h|^2\,\mathrm dx
			+\rho_f\int_0^L\int_0^1h(B_h^{-\top}\partial_t\mathbf v)\cdot B_h^{-\top}\partial_t\widetilde{\mathbf v}\,\mathrm dz\,\mathrm dx\nonumber\\
			&\quad+\rho_f\int_0^L\int_0^1h((\partial_tB_h^{-\top})\mathbf v)\cdot B_h^{-\top}\partial_t\widetilde{\mathbf v}\,\mathrm dz\,\mathrm dx \nonumber\\
			&\quad+\rho_f\int_0^L\int_0^1
			\big((\widehat\mathbf u-\partial_t\chi_h)\cdot(B_h\nabla)\big)\widehat\mathbf u
			\cdot B_h^{-\top}\partial_t\widetilde{\mathbf v}\,\mathrm dz\,\mathrm dx\nonumber\\
			&\quad-\mu\sum_{i,j=1}^2\int_0^L\int_0^1(A_h)_{ij}
			\partial_i(B_h^{-\top}\partial_t\mathbf v)\cdot\partial_j(B_h^{-\top}\widetilde{\mathbf v})\,\mathrm dz\,\mathrm dx\nonumber\\
			&\quad-\mu\sum_{i,j=1}^2\int_0^L\int_0^1(\partial_tA_h)_{ij}
			\partial_i(B_h^{-\top}\mathbf v)\cdot\partial_j(B_h^{-\top}\widetilde{\mathbf v})\,\mathrm dz\,\mathrm dx\nonumber\\
			&\quad-\mu\sum_{i,j=1}^2\int_0^L\int_0^1(A_h)_{ij}
			\partial_i((\partial_tB_h^{-\top})\mathbf v)\cdot\partial_j(B_h^{-\top}\widetilde{\mathbf v})\,\mathrm dz\,\mathrm dx\nonumber\\
			&\quad-\mu\sum_{i,j=1}^2\int_0^L\int_0^1(A_h)_{ij}
			\partial_i(B_h^{-\top}\mathbf v)\cdot\partial_j((\partial_tB_h^{-\top})\widetilde{\mathbf v})\,\mathrm dz\,\mathrm dx
			\eqqcolon\sum_{i=1}^8 O_i.
			\label{eq:weighted-virial}
		\end{align}
		Here we have
		\begin{align*}
			\mathcal H & \coloneqq
			\rho_f\int_0^L\int_0^1 h(B_h^{-\top}\partial_t\mathbf v)\cdot B_h^{-\top}\widetilde{\mathbf v}\,\mathrm dz\,\mathrm dx +\rho_f\int_0^L\int_0^1h((\partial_tB_h^{-\top})\mathbf v)\cdot B_h^{-\top}\widetilde{\mathbf v}\,\mathrm dz\,\mathrm dx \\
			& \qquad
			+\rho_f\int_0^L\int_0^1
			\big((\widehat\mathbf u-\partial_t\chi_h)\cdot(B_h\nabla)\big)\widehat\mathbf u
			\cdot B_h^{-\top}\widetilde{\mathbf v}\,\mathrm dz\,\mathrm dx 
			+\rho_s\int_0^L\partial_t^2h\,\partial_th\,\mathrm dx.
		\end{align*}
		Multiplying \eqref{eq:weighted-virial} by $t$ and integrating by parts from $0$ to $T$, we obtain
		\begin{align}
			& \alpha \int_0^Tt\int_0^L|\partial_t\partial_x^2h|^2\,\mathrm dx\,\mathrm dt
			+\beta\int_0^Tt\int_0^L|\partial_t\partial_xh|^2\,\mathrm dx\,\mathrm dt \nonumber\\
			& \quad = - T\mathcal H(T) + \int_0^T \mathcal H(t) \,\mathrm dt + \sum_{i=1}^8 \int_0^TtO_i\,\mathrm dt 
			\label{eq:weighted-t}
		\end{align}
		We estimate the full cross functional using the original coupled equation. Applying \eqref{eq:lower-coupled} with $(\boldsymbol\varphi,\zeta)=(\widetilde{\mathbf v},\partial_th)$ and integrating by parts in the periodic variable, we obtain
		\begin{align*}
			\mathcal H
			&=-\mu\sum_{i,j=1}^2\int_{\Omega_1}(A_h)_{ij}
			\partial_i(B_h^{-\top}\mathbf v)\cdot\partial_j(B_h^{-\top}\widetilde{\mathbf v})\,\mathrm dx\,\mathrm dz\\
			&\quad+\alpha\int_0^L\partial_x^3h\,\partial_t\partial_xh\,\mathrm dx
			-\beta\int_0^L\partial_xh\,\partial_t\partial_xh\,\mathrm dx.
		\end{align*}
		By \eqref{eq:lower-bound}, the upper bound in \eqref{eq:lower-uniform-coercivity}, and Lemma~\ref{lem:lower-lift}, for regular times we have
		\begin{align}
			|\mathcal H(t)|
			\leq C_T\left(
			\lVert \mathbf v(t)\rVert_{H^1(\Omega_1)}
			\lVert \widetilde{\mathbf v}(t)\rVert_{H^1(\Omega_1)}
			+\lVert h(t)\rVert_{H_\sharp^3(0,L)}
			\lVert \partial_th(t)\rVert_{H_\sharp^1(0,L)}
			\right)\leq C_T.
			\label{eq:weighted-cross-bound}
		\end{align}
		Consequently,
		\begin{align*}
			|T\mathcal H(T)|+\int_0^T|\mathcal H(t)|\,\mathrm dt\leq C_T.
		\end{align*}
		Note that $t\mathcal H(t)\to0$ as $t\downarrow0$, uniformly along the higher approximations with common lower-order bounds. 
		
		We now estimate $O_i$, $i=1,\dots,8$ in order. For $O_1$, we apply the trace theorem to the boundary identity $\partial_t\mathbf v|_{z=1}=\partial_t^2h\,\mathbf e_2$ and find
		\begin{align*}
			\lVert \partial_t^2h\rVert_{H_\sharp^{1/2}(0,L)}
			\leq C\lVert \partial_t\mathbf v\rVert_{H^1(\Omega_1)}.
		\end{align*}
		Interpolation between $H_\sharp^{-1/2}(0,L)$ and $H_\sharp^{1/2}(0,L)$, followed by Young's inequality, yields, for every $\delta>0$,
		\begin{align}
			\int_0^TtO_1 \,\mathrm dt
			=\rho_s \int_0^T t\lVert \partial_t^2h\rVert_{L_\sharp^2(0,L)}^2\,\mathrm dt
			\leq\delta \int_0^T t^2\lVert \partial_t\mathbf v\rVert_{H^1(\Omega_1)}^2\,\mathrm dt
			+C_\delta\int_0^T\lVert \partial_t^2h\rVert_{H_\sharp^{-1/2}(0,L)}^2\,\mathrm dt.
			\label{eq:weighted-acceleration}
		\end{align}
		The last term is controlled in \eqref{eq:lower-bound}.
		
		For $O_2$, the bounded geometric coefficients and \eqref{eq:lower-lift-time} imply
		\begin{align*}
			|O_2|
			\leq C_T\lVert \partial_t\mathbf v\rVert_{L^2(\Omega_1)}
			\lVert \partial_t\widetilde{\mathbf v}\rVert_{L^2(\Omega_1)}\leq C_T\lVert \partial_t\mathbf v\rVert_{L^2(\Omega_1)}
			\lVert \partial_t^2h\rVert_{H_\sharp^{-1/2}(0,L)}.
		\end{align*}
		Both factors belong to $L^2(0,T)$ by \eqref{eq:lower-bound}. Therefore by Cauchy--Schwarz in time and $t\leq T$, we have
		\begin{align*}
			\int_0^Tt|O_2|\,\mathrm dt\leq C_T.
		\end{align*}
		
		For $O_3$, the entries of $\partial_tB_h^{-\top}$ contain only bounded factors multiplying $\partial_th$ or $\partial_t\partial_xh$. By the lower-order bounds and the column estimate in Lemma~\ref{lem:lower-coefficients}, we have $\lVert h(\partial_tB_h^{-\top})\mathbf v\rVert_{L^2(\Omega_1)}\leq C_T$. Consequently, by \eqref{eq:lower-lift-time} and \eqref{eq:lower-bound} we infer
		\begin{align*}
			\int_0^Tt|O_3|\,\mathrm dt
			&\leq C_T\int_0^Tt\lVert h(\partial_tB_h^{-\top})\mathbf v\rVert_{L^2(\Omega_1)}
			\lVert \partial_t^2h\rVert_{H_\sharp^{-1/2}(0,L)}\,\mathrm dt
			\leq C_T.
		\end{align*}
		
		For $O_4$, the same lifting estimate with \eqref{eq:lower-nonlinear} and \eqref{eq:lower-bound} yields
		\begin{align*}
			\int_0^Tt|O_4|\,\mathrm dt
			&\leq C_T
			\int_0^Tt\lVert \big[(\widehat\mathbf u-\partial_t\chi_h)\cdot(B_h\nabla)\big]\widehat\mathbf u\rVert_{L^2(\Omega_1)}
			\lVert \partial_t^2h\rVert_{H_\sharp^{-1/2}(0,L)}\,\mathrm dt\\
			&\leq C_T \lVert \big[(\widehat\mathbf u-\partial_t\chi_h)\cdot(B_h\nabla)\big]\widehat\mathbf u\rVert_{L^2(0,T;L^2(\Omega_1))} \lVert \partial_t^2h\rVert_{L^2(0,T;H_\sharp^{-1/2}(0,L))}\leq C_T,
		\end{align*}
		
		For $O_5$, the $H^1$ multiplication bounds for $B_h^{-\top}$ and the stationary lifting estimate in Lemma \ref{lem:lower-lift} imply
		\begin{align}
			\int_0^Tt|O_5|\,\mathrm dt
			&\leq C_T\int_0^Tt\lVert \partial_t\mathbf v\rVert_{H^1(\Omega_1)}
			\lVert \widetilde{\mathbf v}\rVert_{H^1(\Omega_1)}\,\mathrm dt\nonumber\\
			&\leq\delta \int_0^Tt^2\lVert \partial_t\mathbf v\rVert_{H^1(\Omega_1)}^2\,\mathrm dt
			+C_{\delta,T}\int_0^T\lVert \partial_th\rVert_{H_\sharp^{1/2}(0,L)}^2\,\mathrm dt.
		\end{align}
		The last term is finite by \eqref{eq:lower-bound}.
		
		By Lemma \ref{lem:lower-lift} and one-dimensional Sobolev embedding with values in $L^2(0,1)$, one records
		\begin{subequations}
			
			\begin{align}
				\lVert \widetilde{\mathbf v}\rVert_{L_\sharp^\infty(0,L;L^2(0,1))}
				&\leq C\lVert \partial_th\rVert_{H_\sharp^{1/2}(0,L)},\nonumber\\
				\lVert \nabla\widetilde{\mathbf v}\rVert_{L_\sharp^\infty(0,L;L^2(0,1))}
				+\lVert \nabla\partial_x\widetilde{\mathbf v}\rVert_{L^2(\Omega_1)}
				&\leq C\lVert \partial_th\rVert_{H_\sharp^{3/2}(0,L)}.
			\end{align}
		\end{subequations}
		The strong solution bounds $h$ in $L^\infty(0,T;H_\sharp^3(0,L))$ and both $\mathbf v$ and $\widetilde{\mathbf v}$ in $L^\infty(0,T;H^1(\Omega_1))$. In particular, their column suprema and the spatial coefficient bounds used below are uniform on $[0,T]$. Since the entries of $\partial_tA_h$ contain only $\partial_th$ and $\partial_t\partial_xh$, for $O_6$, similar to the proof of Lemma \ref{lem:lower-fluid}, we have
		\begin{align*}
			\int_0^Tt|O_6|\,\mathrm dt
			&\leq C_T\int_0^Tt\left(
			\lVert \nabla\widehat\mathbf u\rVert_{L^2(\Omega_1)}
			+\lVert \nabla\widehat\mathbf u\rVert_{L_\sharp^\infty(0,L;L^2(0,1))}\right)
			\lVert B_h^{-\top}\widetilde{\mathbf v}\rVert_{H^1(\Omega_1)}\,\mathrm dt\\
			&\leq C_T\int_0^Tt\left(1+\lVert \mathbf v\rVert_{H^2(\Omega_1)}^2\right)\,\mathrm dt \leq C_T,
		\end{align*}
		where the second inequality uses the same column estimates and norm conversions as in Lemma~\ref{lem:lower-coefficients}.
		
		For $O_7$, set $a_j=\frac{z(A_h)_{1j}}h,a_1=z,a_2=-\frac{z^2\partial_xh}{h}$.
		The separation bound and \eqref{eq:lower-bound} imply
		\begin{align*}
			\sum_{j=1}^2\left(
			\lVert a_j\rVert_{L^\infty(\Omega_1)}
			+\lVert \partial_xa_j\rVert_{L^\infty(\Omega_1)}
			\right)\leq C_T.
		\end{align*}
		We shall use, for $\mathbf f\in H^2(\Omega_1)$,
		\begin{align*}
			\lVert \mathbf f\rVert_{L_\sharp^\infty(0,L;L^2(0,1))}
			+\lVert B_h^{-\top}\mathbf f\rVert_{H^1(\Omega_1)}
			&\leq C_T\lVert \mathbf f\rVert_{H^1(\Omega_1)},\\
			\lVert \nabla(B_h^{-\top}\mathbf f)\rVert
			_{L_\sharp^\infty(0,L;L^2(0,1))}
			+\lVert \partial_x\nabla(B_h^{-\top}\mathbf f)\rVert
			_{L^2(\Omega_1)}
			&\leq C_T\lVert \mathbf f\rVert_{H^2(\Omega_1)}.
		\end{align*}
		These bounds follow from the coefficient-product argument in Section~\ref{sec:lower-coefficients}, in particular the expansion underlying \eqref{eq:lower-matrix-product}, and the column estimate \eqref{eq:lower-anisotropic}, using the uniform $H_\sharp^3(0,L)$ bound for $h$ from \eqref{eq:lower-bound}.
		
		Explicit differentiation of $B_h^{-\top}$ leads to the pointwise bound
		\begin{align*}
			&|\partial_tB_h^{-\top}|
			+|\partial_z\partial_tB_h^{-\top}|
			+\left|
			\partial_x\partial_tB_h^{-\top}
			-\frac zh\partial_t\partial_x^2h
			(\mathbf e_2\otimes\mathbf e_1)
			\right|
			\leq C_T
			\left(|\partial_th|+|\partial_t\partial_xh|\right).
		\end{align*}
		Expanding $\partial_i((\partial_tB_h^{-\top})\mathbf v)$ and applying Cauchy--Schwarz in $z$ and H\"older's inequality in $x$, we obtain
		\begin{align*}
			\int_0^Tt|O_7|\,\mathrm dt
			\leq
			\mu\sum_{j=1}^2
			\int_0^Tt\left|\int_0^L\int_0^1
			\partial_t\partial_x^2h\,a_jv_1
			\partial_j(B_h^{-\top}\widetilde{\mathbf v})_2
			\,\mathrm dz\,\mathrm dx\right|\,\mathrm dt+C_T.
		\end{align*}
		For the first term, integration by parts in $x$ yields, for $j=1,2$,
		\begin{align*}
			&\left|\int_0^L\int_0^1
			\partial_t\partial_x^2h\,a_jv_1
			\partial_j(B_h^{-\top}\widetilde{\mathbf v})_2
			\,\mathrm dz\,\mathrm dx\right|\\
			&\quad=
			\left|\int_0^L\int_0^1
			\partial_t\partial_xh\,
			\partial_x\left(
			a_jv_1\partial_j(B_h^{-\top}\widetilde{\mathbf v})_2
			\right)\,\mathrm dz\,\mathrm dx\right|\\
			&\quad\leq
			\lVert \partial_t\partial_xh\rVert_{L_\sharp^2(0,L)}
			\bigg(
			\lVert \partial_xa_j\rVert_{L^\infty(\Omega_1)}
			\lVert v_1\rVert_{L_\sharp^\infty(0,L;L^2(0,1))}
			\lVert \partial_j(B_h^{-\top}\widetilde{\mathbf v})_2\rVert_{L^2(\Omega_1)}\\
			&\qquad\quad+
			\lVert a_j\rVert_{L^\infty(\Omega_1)}
			\lVert \partial_xv_1\rVert_{L^2(\Omega_1)}
			\lVert \partial_j(B_h^{-\top}\widetilde{\mathbf v})_2\rVert
			_{L_\sharp^\infty(0,L;L^2(0,1))}\\
			&\qquad\quad+
			\lVert a_j\rVert_{L^\infty(\Omega_1)}
			\lVert v_1\rVert_{L_\sharp^\infty(0,L;L^2(0,1))}
			\lVert \partial_x\partial_j(B_h^{-\top}\widetilde{\mathbf v})_2\rVert
			_{L^2(\Omega_1)}
			\bigg)\\
			&\quad\leq
			C_T\lVert \partial_th\rVert_{H_\sharp^1(0,L)}
			\lVert \mathbf v\rVert_{H^1(\Omega_1)}
			\lVert \widetilde{\mathbf v}\rVert_{H^2(\Omega_1)}.
		\end{align*}
		The remaining terms in the estimate of $O_7$ have the same bound. Consequently, by \eqref{eq:lower-bound} and Lemma \ref{lem:lower-lift}, one infers
		\begin{align*}
			\int_0^Tt|O_7|\,\mathrm dt
			\leq C_T.
		\end{align*}
		
		For $O_8$, in view of the preceding product estimates with $\mathbf v$ and $\widetilde{\mathbf v}$, we find
		\begin{align*}
			\int_0^Tt|O_8|\,\mathrm dt
			&\leq C_T
			\int_0^Tt\lVert \partial_th\rVert_{H_\sharp^1(0,L)}
			\lVert \widetilde{\mathbf v}\rVert_{H^1(\Omega_1)}
			\lVert \mathbf v\rVert_{H^2(\Omega_1)} \,\mathrm dt
			\leq C_T\int_0^Tt\lVert \mathbf v\rVert_{H^2(\Omega_1)}\,\mathrm dt
			\leq C_T.
		\end{align*}
		
		Using \eqref{eq:weighted-t}, \eqref{eq:weighted-cross-bound}, the preceding estimates of $O_1,\ldots,O_8$, and the zero mean of $\partial_th$, we obtain, for every $\delta>0$,
		\begin{align*}
			\int_0^Tt\lVert \partial_th\rVert_{H_\sharp^2(0,L)}^2\,\mathrm dt
			\leq\delta\int_0^Tt^2\lVert \partial_t\mathbf v\rVert_{H^1(\Omega_1)}^2\,\mathrm dt
			+C_{\delta,T}.
		\end{align*}
		Combining this estimate with \eqref{eq:weighted-acceleration} and the unweighted lower bound for $\partial_t\widehat\mathbf u$, and rescaling $\delta$, yields
		\begin{align}
			\int_0^Tt\left(
			\lVert \partial_t\widehat\mathbf u\rVert_{L^2(\Omega_1)}^2
			+\lVert \partial_t^2h\rVert_{L_\sharp^2(0,L)}^2
			+\lVert \partial_th\rVert_{H_\sharp^2(0,L)}^2
			\right)\,\mathrm dt
			\leq\delta\int_0^Tt^2\lVert \partial_t\mathbf v\rVert_{H^1(\Omega_1)}^2\,\mathrm dt
			+C_{\delta,T}.
			\label{eq:weighted-integral-preliminary}
		\end{align}
		To absorb the last integral, differentiating $\mathbf v=B_h^\top\widehat\mathbf u$ and using the global bound \eqref{eq:lower-bound} lead to
		\begin{align}
			\lVert \partial_t\mathbf v\rVert_{H^1(\Omega_1)}^2
			\leq C_T\bigl(1+\lVert \nabla\partial_t\widehat\mathbf u\rVert_{L^2(\Omega_1)}^2
			+\lVert \partial_t\widehat\mathbf u\rVert_{L^2(\Omega_1)}^2
			+\lVert \partial_th\rVert_{H_\sharp^2(0,L)}^2
			+\lVert \mathbf v\rVert_{H^2(\Omega_1)}^2\bigr).
			\label{eq:weighted-piola-derivative}
		\end{align}
		In particular, $\partial_t\partial_x^2h\,\widehat u_1$ is controlled by $\lVert \partial_th\rVert_{H_\sharp^2(0,L)}$ and $\lVert \widehat u_1\rVert_{L_\sharp^\infty(0,L;L^2(0,1))}$. Products of $\partial_t\partial_xh$ with $\nabla\widehat u_1$ are controlled by $\lVert \mathbf v\rVert_{H^2(\Omega_1)}$. These account for all derivatives of the Piola coefficients.
		
		Collecting \eqref{eq:weighted-averaging}, \eqref{eq:weighted-piola-derivative}, and \eqref{eq:weighted-integral-preliminary} together, we conclude that
		\begin{align}
			&\mathop{\mathrm{ess\,sup}}_{0<t\leq T}t^2\left(\lVert \partial_t\widehat\mathbf u(t)\rVert_{L^2(\Omega_1)}^2
			+\lVert \partial_t^2h(t)\rVert_{L_\sharp^2(0,L)}^2
			+\lVert \partial_th(t)\rVert_{H_\sharp^2(0,L)}^2\right)
			\nonumber\\
			&\quad+\int_0^{T}t\left(\lVert \partial_t\widehat\mathbf u\rVert_{L^2(\Omega_1)}^2
			+\lVert \partial_t^2h\rVert_{L_\sharp^2(0,L)}^2
			+\lVert \partial_th\rVert_{H_\sharp^2(0,L)}^2\right)\,\mathrm dt+\int_0^{T}t^2\lVert \nabla\partial_t\widehat\mathbf u\rVert_{L^2(\Omega_1)}^2\mathrm dt\leq C_T,
			\label{eq:weighted-closed}
		\end{align}
		where we have chosen $\delta > 0$ sufficiently small such that all the norms on the right-hand side can be absorbed suitably via the weighted higher energy or dissipation.
		
		We next recover the Eulerian and spatial estimates using the temporal conversion \eqref{eq:lower-eulerian-} and $\widehat\mathbf u=B_h^{-\top}\mathbf v$.
		After taking a spatial derivative, the additional products contain $\partial_th\,\nabla^2\widehat\mathbf u$ and $\partial_t\partial_xh\,\nabla\widehat\mathbf u$. The former uses $H_\sharp^1(0,L)\hookrightarrow L_\sharp^\infty(0,L)$, and the latter satisfies
		\begin{align*}
			\lVert \partial_t\partial_xh\,\nabla\widehat\mathbf u\rVert_{L^2(\Omega_1)}
			\leq\lVert \partial_t\partial_xh\rVert_{L_\sharp^2(0,L)}
			\lVert \nabla\widehat\mathbf u\rVert_{L_\sharp^\infty(0,L;L^2(0,1))}
			\leq C_T\lVert \mathbf v\rVert_{H^2(\Omega_1)}.
		\end{align*}
		Thus \eqref{eq:weighted-closed} also controls $t\partial_t\mathbf u$ in $L^\infty(0,T;L^2(\Omega_h))\cap L^2(0,T;H^1(\Omega_h))$.
		
		For the spatial estimates, use the stationary Stokes estimate and convection absorption from the proof of Lemma~\ref{lem:lower-fluid}, together with the pressure normalization \eqref{eq:lower-pressure}. Combining these with the beam equation \eqref{eq:beam}, the fluid-traction trace estimate, and \eqref{eq:lower-bound}, we obtain
		\begin{align}
			&\lVert \mathbf u\rVert_{H^2(\Omega_h)}^2+\lVert p\rVert_{H^1(\Omega_h)}^2
			+\lVert h\rVert_{H_\sharp^4(0,L)}^2 \nonumber\\
			& \qquad\leq C_T\left(1+\lVert \partial_t\widehat\mathbf u\rVert_{L^2(\Omega_1)}^2
			+\lVert \partial_t^2h\rVert_{L_\sharp^2(0,L)}^2
			+\lVert \partial_th\rVert_{H_\sharp^2(0,L)}^2\right).
			\label{eq:weighted-spatial}
		\end{align}
		Multiplying \eqref{eq:weighted-spatial} by $t^2$ and using \eqref{eq:weighted-closed} shows \eqref{eq:weighted-bound}, which completes the proof.
