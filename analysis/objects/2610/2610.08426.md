<!-- definition-1: body_md -->
\label{autshad}
Let $F\in Homeo(X)$. We say that~$F$ has the \emph{shadowing property} if for every $\epsilon>0$ there exists $\delta>0$ such that every sequence $(x_n)_{n\in\mathbb Z}$ in~$X$ satisfying
\begin{equation*}
\sup_{n\in\mathbb Z}\|x_{n+1}-F(x_n)\|\le \delta
\end{equation*}
is $\epsilon$-shadowed by an exact trajectory of~$F$, i.e., there exists $v\in  X$ such that
\begin{equation*}
\sup_{n\in\mathbb Z}\|x_n-F^n(v)\|\le \epsilon.
\end{equation*}

\label{auttopstab}
Let $F\in Homeo(X)$. We say that~$F$ is \emph{topologically stable} if for every $\epsilon>0$ there exists $\delta>0$ such that, whenever $g\in Homeo(X)$ satisfies
\begin{equation*}
\|F-g\|_\infty\le\delta,
\end{equation*}
there exists a continuous map $h\colon X\to X$ such that
\begin{equation}\label{topconj}
h\circ g=F\circ h
\end{equation}
and
\begin{equation}\label{topclose}
\|h-I\|_\infty\le\epsilon.
\end{equation}

\label{autgph}
Let $T\in GL(X)$. We say that $T$ is \emph{generalized pseudo-hyperbolic} if there exist $C\in H(X)$ and $D, \lambda>0$ such that for all $n\in \mathbb N_0$,
\begin{equation*}
\|T^n C\| \le De^{-\lambda n} \qquad \text{and} \qquad \|T^{-n}(I-C)\|\le De^{-\lambda n}.
\end{equation*}

\label{nonautshad}
Let $(F_n)_{n\in\mathbb Z}$ be a sequence in~$Homeo(X)$. We say that $(F_n)_{n\in\mathbb Z}$ has the \emph{shadowing property} if for every $\epsilon>0$ there exists $\delta>0$ such that every sequence $(x_n)_{n\in\mathbb Z}$ in~$X$ satisfying
\begin{equation*}
\sup_{n\in\mathbb Z}\|x_{n+1}-F_n(x_n)\|\le \delta,
\end{equation*}
there exists $v\in X$ such that 
\begin{equation*}
\sup_{n\in\mathbb Z}\|x_n-F(n,0)(v)\|\le \epsilon,
\end{equation*}
where $F$ is the evolution family associated with $(F_n)_{n\in\mathbb Z}$.

<!-- definition-5: body_md -->
\label{nonauttopstab}
Let $(F_n)_{n\in\mathbb Z}$ be a sequence in~$Homeo(X)$. We say that $(F_n)_{n\in\mathbb Z}$ is \emph{topologically stable} if for every $\epsilon>0$ there exists $\delta>0$ such that, whenever $(g_n)_{n\in\mathbb Z}$ is a sequence in $Homeo(X)$ satisfying
\begin{equation*}
\sup_{n\in\mathbb Z}\|F_n-g_n\|_\infty\le\delta,
\end{equation*}
there exists a sequence $(h_n)_{n\in\mathbb Z}$ of continuous self-maps of~$X$ such that
\begin{equation}\label{topconj_2}
h_{n+1}\circ g_n=F_n\circ h_n,\qquad n\in\mathbb Z,
\end{equation}
and
\begin{equation}\label{topclose_2}
\sup_{n\in\mathbb Z}\|h_n-I\|_\infty\le\epsilon.
\end{equation}

\label{gphdef}
Let $(T_n)_{n\in \mathbb Z}$ be a sequence in $GL(X)$. We say that $(T_n)_{n\in \mathbb Z}$ is \emph{generalized pseudo-hyperbolic} if there exist a sequence $(C_n)_{n\in \mathbb Z}$ in $H(X)$ and constants $D, \lambda>0$ such that for all $m\ge n$,
\begin{equation}\label{909}
\|T(m, n)C_n\|\le De^{-\lambda (m-n)} \quad \text{and} \quad \|T(n, m)(I-C_m)\|\le De^{-\lambda (m-n)}.
\end{equation}

<!-- theorem-1: body_md -->
\label{thm1}
Let $(T_n)_{n\in\mathbb Z}$ be a sequence in $GL(X)$. Then the following statements are equivalent.
\vskip3pt
\begin{enumerate}[label={(\roman*)},itemindent=1em]
\item[\rm (i)] $(T_n)_{n\in \mathbb Z}$ has the shadowing property.
\vskip2pt
\item[\rm (ii)] $(T_n)_{n\in \mathbb Z}$ is generalized pseudo-hyperbolic.
\vskip2pt
\item[\rm (iii)] $(T_n)_{n\in \mathbb Z}$ is topologically stable. 
\end{enumerate}

\label{thm2}
Let $T\in GL(X)$. Then the following statements are equivalent.
\vskip3pt
\begin{enumerate}[label={(\roman*)},itemindent=1em]
\item[\rm (i)] $T$ has the shadowing property.
\vskip2pt
\item[\rm (ii)] $T$ is surjectively hyperbolic.
\vskip2pt
\item[\rm (iii)] $T$ is generalized pseudo-hyperbolic.
\vskip2pt
\item[\rm (iv)] $T$ is topologically stable.
\end{enumerate}

<!-- remark-1: body_md -->
\label{rem1}
In \cite[Corollary~15]{BCDHP}, Bernardes et al. proved that topological stability implies the shadowing property for invertible operators on arbitrary Banach spaces.  For the converse implication, Nguyen~\cite[Theorem~1.5]{N} proved that every limit expansive
invertible linear operator with the shadowing property is topologically stable. More recently, Messaoudi et al.~\cite[Theorem~3.5]{MTS}
established the equivalence between topological stability and the shadowing property under the additional assumption that the eigenspaces $\ker(\lambda I-T)$, $\lambda\in\mathbb T$, are complemented in~$X$. Theorem~\ref{thm2} strengthens these results by showing that, for arbitrary invertible linear operators on complex Banach spaces, shadowing and topological stability are equivalent, without assuming either limit expansiveness or complementability of the unimodular eigenspaces.

\label{thm3}
Let $(T_n)_{n\in\mathbb Z}$ be a sequence in $GL(X)$ which has the shadowing property. Then there exists $\eta>0$ such that if $(f_n)_{n\in\mathbb Z}$ is a sequence of Lipschitz self-maps of~$X$ such that 
\begin{equation*}
\operatorname{Lip}(f_n)<\eta\qquad\text{for all $n\in\mathbb Z$}
\end{equation*}
and
\begin{equation*}
F_n:=T_n+f_n\in Homeo(X)\qquad\text{for all $n\in\mathbb Z$},
\end{equation*}
then the sequence $(F_n)_{n\in\mathbb Z}$ has the shadowing property.

<!-- remark-2: body_md -->
\label{rem2}
Given $n\in\mathbb Z$, a sufficient condition for $F_n=T_n+f_n$ to be a homeomorphism in Theorems~\ref{thm3} and~\ref{thm4} is that
\begin{equation*}
\|T_n^{-1}\|\operatorname{Lip}(f_n)<1.
\end{equation*}
Indeed, since
\begin{equation*}
\operatorname{Lip}(T_n^{-1}f_n)
\le \|T_n^{-1}\|\operatorname{Lip}(f_n)<1, 
\end{equation*}
it follows that $I+T_n^{-1}f_n$ is a self-homeomorphism of~$X$, and hence so is $F_n=T_n\bigl(I+T_n^{-1}f_n\bigr)$.

<!-- remark-3: body_md -->
The assumption about the uniform boundedness of the inverse operators $T_n^{-1}$, $n\in\mathbb Z$, in Corollaries~\ref{cor1} and~\ref{cor2}, in general,  cannot be omitted. To see this, let
$X=\mathbb C$ and define
\begin{equation*}
T_n (z)=\frac{z}{|n|+2},\qquad z\in\mathbb C,\quad n\in\mathbb Z.
\end{equation*}
It is easily seen that $T_n\in GL(\mathbb C)$ for all $n\in\mathbb Z$ and 
\begin{equation*}
T_n^{-1}(z)=(|n|+2)z,\qquad z\in\mathbb C,\quad n\in\mathbb Z.
\end{equation*}
Since
\begin{equation*}
\|T_j\|=\frac{1}{|j|+2}\leq\frac{1}{2},\qquad j\in\mathbb Z,
\end{equation*}
it follows that
\begin{equation*}
\|T(m,n)\|\leq\prod_{j=n}^{m-1}\frac{1}{|j|+2}\le 2^{-(m-n)},\qquad m\ge n,\quad m,n\in\mathbb Z.
\end{equation*}
This implies that the sequence $(T_n)_{n\in\mathbb Z}$ is generalized pseudo-hyperbolic with $D=1$, $\lambda=\log 2$ and 
$C_n=I$ for all $n\in\mathbb Z$. By Theorem~\ref{thm1}, $(T_n)_{n\in\mathbb Z}$ has the shadowing property and is topologically stable.  Given any $\eta>0$, choose $N\in\mathbb Z$ such that
\begin{equation*}
\|T_N\|=\frac{1}{|N|+2}<\eta,
\end{equation*}
and define
\begin{equation*}
f_N=-T_N\qquad\text{and}\qquad f_n=0\quad\text{whenever $n\ne N$ and $n\in\mathbb Z$}.
\end{equation*}
Clearly,
\begin{equation*}
\operatorname{Lip}(f_n)<\eta
\qquad\text{for all }n\in\mathbb Z,
\end{equation*}
whereas $F_N=T_N+f_N=0$. Thus, $F_N\notin Homeo(\mathbb C)$. Of course, in this example, we have
\begin{equation*}
\sup_{n\in\mathbb Z}\|T_n^{-1}\|=\infty.
\end{equation*}

<!-- proposition-1: body_md -->
\label{prop1}
$\mathcal L\colon D(\mathcal L)\subset\ell_\infty(X)\to\ell_\infty(X)$ is a closed linear operator.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
It is easy to verify that $\mathcal L$ is linear. We now prove that it is closed.
Let $(x^k)_{k\in \mathbb N}$ be a sequence in $D(\mathcal L)$ such that $x^k \to x$ and $\mathcal Lx^k \to y$ in $\ell_\infty(X)$. Writing $y^k:=\mathcal L x^k$, for each $n\in\mathbb Z$, we have \begin{equation*}
y_n=\lim_{k\to \infty}y_n^k=\lim_{k\to \infty}(x_{n+1}^k-T_n x_n^k)=\lim_{k\to \infty}x_{n+1}^k-T_n \lim_{k\to \infty}x_n^k=x_{n+1}-T_n x_n 
\end{equation*}
by the continuity of~$T_n$. Thus, $\mathcal L x=y$. Since $y\in\ell_\infty(X)$, it follows that $x\in D(\mathcal L)$. Hence $\mathcal L$ is closed.

<!-- proposition-2: body_md -->
\label{prop2}
Let $(T_n)_{n\in \mathbb Z}$ be a sequence in $GL(X)$. Then the following statements are equivalent.
\vskip3pt
\begin{enumerate}[label={(\roman*)},itemindent=1em]
\item[\rm (i)] $(T_n)_{n\in \mathbb Z}$ has the shadowing property.
\vskip2pt
\item[\rm (ii)] $\mathcal L:(\ell_\infty'(X),\|\cdot\|_\infty')\longrightarrow(\ell_\infty(X),\|\cdot\|_\infty)$ is surjective.
\end{enumerate}
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
Suppose that $(T_n)_{n\in \mathbb Z}$ has the shadowing property. Let $\delta>0$ be as in Definition~\ref{nonautshad} corresponding to $\epsilon=1$. Let $y\in \ell_\infty(X)$ be an arbitrary nonzero sequence. (The case $y=0$ is trivial.) Choose $x_0\in X$ arbitrarily. Since each $T_n$, $n\in\mathbb Z$, is invertible, the recurrence
\begin{equation*}
x_{n+1}-T_n x_n=\frac{\delta}{\|y\|_\infty}y_n, \qquad n\in \mathbb Z,
\end{equation*}
determines a unique bi-infinite sequence $(x_n)_{n\in\mathbb Z}$ in~$X$ with the prescribed~$x_0$.
Then 
\begin{equation*}
\sup_{n\in \mathbb Z}\|x_{n+1}-T_n x_n\|=\delta.
\end{equation*}
By the shadowing property,  there exists $v\in X$ such that
 \begin{equation*}
 \sup_{n\in \mathbb Z}\|x_n-T(n, 0)v\|\le 1.
\end{equation*}
Set
\begin{equation*}
z_n:=\frac{\|y\|_\infty}{\delta}(x_n-T(n, 0)v), \qquad n\in \mathbb Z.
\end{equation*}
Then $z=(z_n)_{n\in\mathbb Z}\in\ell_\infty(X)$. Moreover, using
\begin{equation*}
T(n+1,0)=T_nT(n,0),\qquad n\in\mathbb Z,
\end{equation*}
we obtain
\begin{equation*}
\begin{aligned}
(\mathcal Lz)_n
&=z_{n+1}-T_nz_n\\
&=\frac{\|y\|_\infty}{\delta}
\Bigl(
x_{n+1}-T(n+1,0)v
-T_nx_n+T_nT(n,0)v
\Bigr)\\
&=\frac{\|y\|_\infty}{\delta}
\bigl(x_{n+1}-T_nx_n\bigr)=y_n.
\end{aligned}
\end{equation*}
Thus, $\mathcal Lz=y$. Since $z\in\ell_\infty(X)$ and
$\mathcal L z=y\in\ell_\infty(X)$, it follows that $z\in \ell'_\infty(X)$.
Hence $\mathcal L$ is surjective.


Suppose now that $\mathcal L$ is surjective. By the Open Mapping Theorem (see~\cite[Theorem 2.12.1]{HP}), there exists $K>0$ such that, for each $z\in \ell_\infty(X)$, there exists $y\in \ell_\infty'(X)$ such that 
\begin{equation}\label{259}
\mathcal Ly=z \quad \text{and} \quad 
                 
 \|y\|_\infty'\le K\|z\|_\infty.\end{equation} Given $\epsilon>0$, set $\delta:=\frac{\epsilon}{K}$. Let $(x_n)_{n\in \mathbb Z}$ be an arbitrary sequence in~$X$ such that 
\begin{equation*}    
\sup_{n\in \mathbb Z}\|x_{n+1}-T_n x_n\| \le \delta.
\end{equation*}
Set
\begin{equation}\label{591}
z_n:=x_{n+1}-T_n x_n, \qquad n\in \mathbb Z.
\end{equation}
Then $z=(z_n)_{n\in \mathbb Z}\in \ell_\infty(X)$ with $\|z\|_\infty \le \delta$. Choose $y=(y_n)_{n\in \mathbb Z}\in \ell_\infty'(X)$ such that~\eqref{259} holds. From $\mathcal L y=z$ and~\eqref{591}, we obtain\begin{equation*}
y_{n+1}-T_n y_n=z_n=x_{n+1}-T_n x_n,\qquad n\in \mathbb Z.
\end{equation*}
Hence
\begin{equation*}
x_{n+1}-y_{n+1}=T_n(x_n-y_n), \qquad n\in \mathbb Z,
\end{equation*}
which implies that 
\begin{equation}\label{306}
x_n-y_n=T(n,0)(x_0-y_0),\qquad n\in\mathbb Z.
\end{equation}
Set $v:=x_0-y_0$. By~\eqref{259} and~\eqref{306}, we have for $n\in\mathbb Z$,
\begin{equation*}
\|x_n-T(n, 0)v\|=\|x_n-(x_n-y_n)\|=\|y_n\|\le\|y\|_\infty \le \|y\|'_\infty \le K\|z\|_\infty \le K\delta=\epsilon.
\end{equation*}
Thus, $(T_n)_{n\in \mathbb Z}$ has the shadowing property.

<!-- proposition-3: body_md -->
\label{prop3}
Let $Y_i$, $i=1, 2$, be complex Banach spaces, and let $T\in L(Y_1, Y_2)$ be surjective. Then there exists 
$R_T\in H(Y_2, Y_1)$ such that 
\begin{equation}\label{617}
TR_T=I_{Y_2}.
\end{equation}
 Moreover, if $R_T\in H(Y_2, Y_1)$ satisfies~\eqref{617} and 
\begin{equation*}
0<\alpha <\frac 1 K, \qquad \text{where $K:=\|R_T\|$,}
\end{equation*}
then, for every $S\in L(Y_1, Y_2)$ with $\|T-S\| \le \alpha$, there exists $R_S\in H(Y_2, Y_1)$ such that
\begin{equation}\label{440}
SR_S=I_{Y_2}\qquad\text{and}\qquad \|R_S\| \le \frac{K}{1-\alpha K}.
\end{equation}
<!-- end proposition-3 -->
<!-- proof-3: body_md -->
The existence of $R_T\in H(Y_2, Y_1)$ satisfying~\eqref{617} follows from \cite[Chap.~IV, Sec.~27, Proposition~5~(i)]{Mul}. Define a sequence of functions $F_n\colon Y_2\rightarrow Y_2$, $n\in\mathbb N$, recursively by $F_1(y)=y$ for $y\in Y_2$ and  
\begin{equation}\label{xndef}
	F_{n+1}(y)=(T-S)(R_T(F_n(y)))\qquad\text{for $y\in Y_2$ and $n\in\mathbb N$}.
\end{equation}
It follows by easy induction that the functions~$F_n$, $n\in\mathbb N$, are continuous and homogeneous, i.e., $F_n\in H(Y_2)$ for $n\in\mathbb N$. It follows that 
\begin{equation*}
	\|F_{n+1}(y)\|\leq\|T-S\|\|R_T\|\|F_n(y)\|\leq K\alpha\|F_n(y)\|\qquad\text{for $y\in Y_2$ and $n\in\mathbb N$}. 
\end{equation*}
Hence,
\begin{equation}\label{xnest}
	\|F_n(y)\|\leq (K\alpha)^{n-1}\|y\|\qquad\text{for all $y\in Y_2$ and $n\in\mathbb N$}.
\end{equation}
This yields
\begin{equation}\label{RTest}
	\|R_T(F_n(y))\|\leq \|R_T\|\|F_n(y)\|\leq K(K\alpha)^{n-1}\|y\|\qquad\text{for all $y\in Y_2$ and $n\in\mathbb N$}.
\end{equation}
Since $0< K\alpha<1$, we can define $R_S\colon Y_2 \to Y_1$ by
\begin{equation}\label{RSdef}
	R_S(y)=\sum_{n=1}^\infty R_T(F_n(y))\qquad\text{for $y\in Y_2$}.
\end{equation}
Since $R_T\in H(Y_2, Y_1)$ and $F_n\in H(Y_2)$, $n\in\mathbb N$,  we have that $R_S$ is homogeneous. From~\eqref{RTest}, it follows that, on each bounded subset of~$Y_2$, the convergence of the series in~\eqref{RSdef} is uniform. By the Weierstrass M-test, $R_S$ is continuous and hence $R_S\in H(Y_2, Y_1)$.  Moreover, from~\eqref{RTest} and~\eqref{RSdef}, we find for $y\in Y_2$,
\begin{equation*}
	\|R_S(y)\|\leq K\sum_{n=1}^\infty(K\alpha)^{n-1} \|y\|=\frac{K}{1-K\alpha}\|y\|,
\end{equation*}
which proves the norm estimate in~\eqref{440}. Next, the identity 
\begin{equation*}
	(T-S)R_T=TR_T-SR_T=I_{Y_2}-SR_T,
\end{equation*}
together with~\eqref{xndef}, implies for $y\in Y_2$ and $n\in\mathbb N$,
\begin{equation*}
	F_{n+1}(y)=F_n(y)-S(R_T(F_n(y))).
\end{equation*}
Hence
\begin{equation*}
	S(R_T(F_n(y)))=F_n(y)-F_{n+1}(y)\qquad\text{for all $y\in Y_2$ and $n\in\mathbb N$}.
\end{equation*}
From this and~\eqref{RSdef}, we conclude that for all $y\in Y_2$,
\begin{equation*}
	S(R_S(y))=\sum_{n=1}^\infty S(R_T(F_n(y)))=\sum_{n=1}^\infty(F_n(y)-F_{n+1}(y))=F_1(y)=y.
\end{equation*}
	Thus, \eqref{440} holds.

<!-- proof-4: body_md -->
\textit{(i) $\Rightarrow$ (ii).}
	Suppose that $(T_n)_{n\in \mathbb Z}$ has the shadowing property. Given $z\ge 1$ and $n\in \mathbb Z$, we define  $\mathcal L_{z, n}\colon \ell_\infty'(X)\to \ell_\infty(X)$ by 
\begin{equation*}
(\mathcal L_{z, n}x)_m=\begin{cases}
\frac{1}{z}x_{m+1}-T_m x_m & m\ge n; \\
zx_{m+1}-T_m x_m & m<n,
\end{cases} \quad \text{for $x=(x_m)_{m\in \mathbb Z}\in \ell_\infty'(X)$,}
\end{equation*}
where $\ell'_\infty(X)$ is equipped with the graph norm $\|\cdot\|'_\infty$. It is easily seen that 
\begin{equation*}
\mathcal L_{z,n}\in L(\ell'_\infty(X),\ell_\infty(X)),\qquad z\geq1,\,\,n\in\mathbb Z.
\end{equation*}
Since $|\frac 1 z-1| \le z-1$ for $z\ge 1$, we have for $x\in\ell'_\infty(X)$ and $m\in\mathbb Z$,
\begin{equation*}
\|(\mathcal L_{z, n}x)_m-(\mathcal Lx)_m\| \le (z-1)\|x_{m+1}\|,
\end{equation*}
where $\mathcal L\in L(\ell'_\infty(X),\ell_\infty(X))$ is defined by~\eqref{Ldef}.
Hence
\begin{equation*}
\|\mathcal L_{z, n}x-\mathcal L x\|_\infty \le (z-1)\|x\|_\infty \le (z-1)\|x\|'_\infty
\end{equation*}
whenever $z\geq1$, $n\in\mathbb Z$ and $x\in\ell'_\infty(X)$. Consequently,
\begin{equation}\label{closest}
\|\mathcal L_{z, n}-\mathcal L \| \le z-1,\qquad z\geq1,\,\, n\in\mathbb Z.
\end{equation}
By Proposition~\ref{prop2}, the shadowing property of $(T_n)_{n\in\mathbb Z}$ implies that $\mathcal L$ is surjective.
From this, by Proposition~\ref{prop3}, we conclude that $\mathcal L$ has a right-inverse $\mathcal R\in H(\ell_\infty(X), \ell_\infty'(X))$. Set $K:=\|\mathcal R\|$.  If $z\in (1, \frac{K+1}{K})$, then $0<z-1<\frac{1}{K}$. This, together with~\eqref{closest}, implies that
\begin{equation*}
\|\mathcal L_{z, n}-\mathcal L \| \le z-1<\frac{1}{K}\quad\text{whenever $z\in\biggl(1,\frac{K+1}{K}\biggr)$ and $n\in\mathbb Z$}.
\end{equation*}
It follows from the second part of Proposition~\ref{prop3}  that, for every $z\in (1, \frac{K+1}{K})$ and $n\in \mathbb Z$, $\mathcal L_{z, n}$ has a right-inverse $\mathcal R_{z, n}\in H(\ell_\infty(X), \ell_\infty'(X))$ satisfying 
\begin{equation*}
\|\mathcal R_{z, n}\| \le \frac{K}{1-(z-1)K}.
\end{equation*}
Fix $t\in (\frac{K}{K+1}, 1)$ and $n\in \mathbb Z$. Then $z:=1/t\in(1,\frac{K+1}{K})$. Therefore, $\mathcal L_{1/t, n}$ has a right-inverse $\mathcal R_{1/t, n}\in H(\ell_\infty(X), \ell_\infty'(X))$ satisfying
\begin{equation}\label{Rnormest}
\|\mathcal R_{1/t, n}\| \le M,\qquad\text{where $
M:=\frac{K}{1-(1/t-1)K}>0$}.
\end{equation}
We construct a sequence of maps $G_{m, n}\in H(X)$, $m\in \mathbb Z$, with the following properties:
\begin{equation}\label{900}
\|G_{m, n}(v)\|\le Mt^{|m-n|}\|v\| \quad \text{whenever $v\in X$ and $m\in \mathbb Z$,}
\end{equation}
\begin{equation}\label{906}
G_{n+1, n}-T_n G_{n, n}=I,
\end{equation}
and
\begin{equation}\label{907}
G_{m+1, n}-T_m G_{m, n}=0 \quad\text{whenever $m\in\mathbb Z$ and $m\neq n$}.
\end{equation}
Let $v\in X$. Define $y(v)=(y_m(v))_{m\in \mathbb Z}\in \ell_\infty(X)$ by 
\begin{equation*}
y_m(v)=\begin{cases}
v,\quad &m=n; \\
0,\quad & m\neq n.
\end{cases}
\end{equation*}
For $m\in\mathbb Z$, define $G_{m,n}\colon X\to X$ by
\begin{equation*}
G_{m,n}(v):=t^{|m-n|}\bigl(\mathcal R_{1/t,n}(y(v))\bigr)_m,\qquad v\in X.
\end{equation*}
The map $v\mapsto y(v)$ from~$X$ to $\ell_\infty(X)$ is continuous and homogeneous, $\mathcal R_{1/t,n}$ is continuous and homogeneous and the coordinate evaluation $x\mapsto x_m$, $m\in\mathbb Z$, from $\ell'_\infty(X)$ to~$X$ is linear and continuous. Thus, $G_{m,n}\in H(X)$ for each $m\in\mathbb Z$, as a composition of continuous and homogeneous maps. Clearly, $\|y(v)\|_\infty=\|v\|$ for $v\in X$. This, together with~\eqref{Rnormest}, implies that~\eqref{900} holds. Writing $y=y(v)$ and $x=\mathcal R_{1/t,n}y$, we have $\mathcal L_{1/t,n} x=y$. From this and the definition of $\mathcal L_{1/t,n} x$, we find for $v\in X$ and $m\geq n$,
\begin{equation*}
\begin{aligned}
G_{m+1,n}(v)-T_mG_{m,n}(v)
&=t^{|m+1-n|}x_{m+1}-t^{|m-n|}T_m x_m\\
&=t^{m-n}\bigl(tx_{m+1}-T_mx_m\bigr)\\
&=t^{m-n}\bigl(\mathcal L_{1/t,n}x\bigr)_m\\
&=t^{m-n}y_m,
\end{aligned}
\end{equation*}
while, for $m<n$,
\begin{equation*}
\begin{aligned}
G_{m+1,n}(v)-T_mG_{m,n}(v)
&=t^{|m+1-n|}x_{m+1}-t^{|m-n|}T_m x_m\\
&=t^{n-m}\left(\frac1t x_{m+1}-T_mx_m\right)\\
&=t^{n-m}\bigl(\mathcal L_{1/t,n}x\bigr)_m\\
&=t^{n-m}y_m.
\end{aligned}
\end{equation*}
From this and from the definition of $y=y(v)$, it follows for $v\in X$ and $m\in\mathbb Z$,
\begin{equation*}
G_{m+1,n}(v)-T_m G_{m,n}(v)=\begin{cases}
v,\quad & m=n; \\
0,\quad & m\neq n.
\end{cases}
\end{equation*}
This implies both~\eqref{906} and~\eqref{907}.


Now set
\begin{equation*}
C_n:=G_{n,n-1}\in H(X),\qquad n\in\mathbb Z.
\end{equation*}
From~\eqref{906} and~\eqref{907}, it follows easily by forward and backward induction that
\begin{equation}\label{greenformula}
G_{m,n-1}=
\begin{cases}
T(m,n)C_n, & m\ge n,\\[1mm]
-T(m,n)(I-C_n), & m<n.
\end{cases}
\end{equation}
This, combined with~\eqref{900}, yields for $m\geq n$,
 \begin{equation*}
\|T(m,n)C_n\|=\|G_{m,n-1}\|
\le M t^{m-n+1}\leq\frac{M}{t}t^{m-n}.
\end{equation*}
On the other hand, for $m<n$, we have
\begin{equation*}
\|T(m,n)(I-C_n)\|
=\|G_{m,n-1}\|
\le M t^{|m-n+1|}
=\frac{M}{t}t^{n-m}.
\end{equation*}
Equivalently, after interchanging~$m$ and $n$, we obtain for $m>n$,
\begin{equation*}
\|T(n,m)(I-C_m)\|
\leq\frac{M}{t}t^{m-n}.
\end{equation*}
It remains to  consider the second estimate in the case $m=n$. By~\eqref{900}, 
\begin{equation*}
\|C_n\|=\|G_{n,n-1}\|\leq M t,\qquad n\in\mathbb Z.
\end{equation*}
Hence, 
\begin{equation*}
\|I-C_n\|\le 1+\|C_n\|\leq 1+Mt,\qquad n\in\mathbb Z.
\end{equation*}
Therefore, the exponential estimates in Definition~\ref{gphdef} are satisfied with
\begin{equation*}
D:=\max\biggl\{\frac{M}{t},1+Mt\biggr\}>0\qquad\text{and}\qquad \lambda:=-\log t>0.
\end{equation*}
Thus, $(T_n)_{n\in\mathbb Z}$ is generalized pseudo-hyperbolic, and the proof of $(i)\Rightarrow(ii)$ is complete.

\textit{(ii) $\Rightarrow$ (iii).}
Suppose that $(T_n)_{n\in\mathbb Z}$ is generalized pseudo-hyperbolic. Let 
$C_n\in H(X)$, $n\in\mathbb Z$, and constants $D,\lambda>0$ have the meaning from Definition~\ref{gphdef}.
Define the Green family $G\colon \mathbb Z\times\mathbb Z\to H(X)$ by                                      
\begin{equation}\label{Gmndef}
G_{m,n}=
\begin{cases}
T(m,n+1)C_{n+1},\quad & m\geq n+1,\\[1mm]
-T(m,n+1)(I-C_{n+1}),\quad & m<n+1.
\end{cases}
\end{equation}
From \eqref{Gmndef}, it follows that
\begin{equation}\label{greenident}
G_{m+1,n}-T_mG_{m,n}=
\begin{cases}
I,\quad & m=n,\\
0,\quad & m\ne n.
\end{cases}
\end{equation}
Indeed, if $m\geq n+1$, then  $m+1\geq n+1$, and by the upper equality in~\eqref{Gmndef},
\begin{equation*}
G_{m+1,n}
=T(m+1,n+1)C_{n+1}
=T_mT(m,n+1)C_{n+1}
=T_mG_{m,n}.
\end{equation*}
Hence
\begin{equation*}
G_{m+1,n}-T_m G_{m,n}=0,\qquad m\geq n+1.
\end{equation*}
If $m<n$, then $m<n+1$ and $m+1<n+1$, and by the lower equality in~\eqref{Gmndef},
\begin{equation*}
\begin{aligned}
G_{m+1,n}
&=-T(m+1,n+1)(I-C_{n+1})\\
&=-T_mT(m,n+1)(I-C_{n+1})
=T_mG_{m,n}.
\end{aligned}
\end{equation*}
Hence
\begin{equation*}
G_{m+1,n}-T_m G_{m,n}=0,\qquad m<n.
\end{equation*}
Finally, if $m=n$, then
\begin{equation*}
G_{n+1,n}=C_{n+1},
\end{equation*}
while
\begin{equation*}
T_nG_{n,n}
=-T_nT(n,n+1)(I-C_{n+1})
=-(I-C_{n+1}).
\end{equation*}
Hence
\begin{equation*}
G_{n+1,n}-T_nG_{n,n}
=C_{n+1}+I-C_{n+1}=I.
\end{equation*}
Thus, \eqref{greenident} holds.
The exponential estimates from Definition~\ref{gphdef} imply that
\begin{equation}\label{greenesti}
\|G_{m,n}\|\le De^{-\lambda|m-n-1|},
\qquad m,n\in\mathbb Z.
\end{equation}
Set
\begin{equation*}
\mu:=D\sum_{j\in\mathbb Z}e^{-\lambda|j|}
=D\frac{1+e^{-\lambda}}{1-e^{-\lambda}}>0.
\end{equation*}
Let $\epsilon>0$. Choose $\delta:=\epsilon/\mu$. Let $(g_n)_{n\in\mathbb Z}$ be  an arbitrary sequence in $Homeo(X)$ satisfying
\begin{equation*}
\sup_{n\in\mathbb Z}\|T_n-g_n\|_\infty\le\delta.
\end{equation*}
Put
\begin{equation}\label{fndef}
f_n:=T_n-g_n,\qquad n\in\mathbb Z.
\end{equation}
Then 
\begin{equation}\label{fnest}
\sup_{x\in X}\|f_n(x)\|=\sup_{x\in X}\|T_n(x)-g_n(x)\|=\|T_n-g_n\|_\infty\leq\delta,\qquad n\in\mathbb Z.
\end{equation}
For $n\in\mathbb Z$ and $x\in X$, define
\begin{equation}\label{Hndef}
H_n(x):=\sum_{m\in\mathbb Z}
G_{n,m}\bigl(f_m(g(m,n)(x))\bigr).
\end{equation}
By \eqref{greenesti} and~\eqref{fnest}, we have for $n,m\in\mathbb Z$,
\begin{equation*}
\begin{aligned}
\|G_{n,m}(f_m(g(m,n)(x)))\|
&\le De^{-\lambda|n-m-1|}\,
   \|f_m(g(m,n)(x))\|\\
&\le D\delta e^{-\lambda|n-m-1|}.
\end{aligned}
\end{equation*}
Since
\begin{equation*}
\sum_{m\in\mathbb Z}e^{-\lambda|n-m-1|}
=\sum_{j\in\mathbb Z}e^{-\lambda|j|}<\infty,\qquad n\in\mathbb Z,
\end{equation*}
the series in \eqref{Hndef} converges uniformly in $x\in X$ for every $n\in\mathbb Z$. Hence, each $H_n$, $n\in\mathbb Z$, is continuous. Furthermore,
\begin{equation*}
\begin{aligned}
\|H_n(x)\|
\le D\delta\sum_{m\in\mathbb Z}e^{-\lambda|n-m-1|}=D\delta \sum_{j\in\mathbb Z}e^{-\lambda|j|}=\mu
\delta=\epsilon
\end{aligned}
\end{equation*}
for $n\in\mathbb Z$ and $x\in X$.
Hence
\begin{equation}\label{eq:hbound}
\sup_{n\in\mathbb Z}\|H_n\|_\infty\le\epsilon.
\end{equation}
We now verify the semiconjugacy relation. By the evolution property of the family associated with $(g_n)_{n\in\mathbb Z}$,
\begin{equation*}
g(m,n+1)\circ g_n=g(m,n),
\qquad m,n\in\mathbb Z.
\end{equation*}
Consequently, using \eqref{Hndef},
\begin{equation*}
\begin{aligned}
H_{n+1}(g_n(x))-T_n H_n(x)
&=\sum_{m\in\mathbb Z}
G_{n+1,m}\bigl(f_m(g(m,n+1)(g_n(x)))\bigr)\\
&\qquad-T_n\sum_{m\in\mathbb Z}
G_{n,m}\bigl(f_m(g(m,n)(x))\bigr)\\
&=\sum_{m\in\mathbb Z}
\bigl(G_{n+1,m}-T_nG_{n,m}\bigr)
\bigl(f_m(g(m,n)(x))\bigr).
\end{aligned}
\end{equation*}
From~\eqref{greenident}, it follows that the only nonzero term in the last sum corresponds to $m=n$, and since $g(n,n)=I$, we have for $n\in\mathbb Z$ and $x\in X$,
\begin{equation}\label{Hncorr}
H_{n+1}(g_n(x))-T_n H_n(x)=f_n(x).
\end{equation}
Define
\begin{equation*}
h_n:=I+H_n,\qquad n\in\mathbb Z.
\end{equation*}
Since $f_n=T_n-g_n$, relation \eqref{Hncorr} gives
\begin{equation*}
\begin{aligned}
h_{n+1}(g_n(x))
&=g_n(x)+H_{n+1}(g_n(x))\\
&=g_n(x)+T_n H_n(x)+f_n(x)\\
&=g_n(x)+T_n H_n(x)+T_nx-g_n(x)\\
&=T_n H_n(x)+T_n x\\
&=T_n h_n(x).
\end{aligned}
\end{equation*}
Hence
\begin{equation}\label{topc1}	
h_{n+1}\circ g_n=T_n\circ h_n,
\qquad n\in\mathbb Z.
\end{equation}

Finally, by \eqref{eq:hbound},
\begin{equation}\label{topc2}	
\sup_{n\in\mathbb Z}\|h_n-I\|_\infty
=\sup_{n\in\mathbb Z}\|H_n\|_\infty
\le\epsilon.
\end{equation}


Thus, $(T_n)_{n\in\mathbb Z}$ is topologically stable. This proves $(ii)\Rightarrow(iii)$.

	\leq K\delta\sum_{k=0}^\infty(\eta K)^k=\frac{K\delta}{1-\eta K}=\epsilon.
\end{equation}
Since $r^{(k)}\to0$ by~\eqref{rkest}, and since
\begin{equation*}
z^{(k)}\to z,\qquad
\mathcal Lz^{(k)}\to\mathcal Lz,\qquad\text{and}\qquad
\mathcal H(z^{(k)})\to\mathcal H(z)
\end{equation*}
in $\ell_\infty(X)$, the last convergence being a consequence of~\eqref{Hlip},
letting $k\to\infty$ in~\eqref{baseq} yields
\begin{equation*}
	\mathcal H(z)=d+\mathcal Lz.
\end{equation*}
Equivalently,
\begin{equation*}
f_n(x_n+z_n)-f_n(x_n)=d_n+z_{n+1}-T_n(z_n),\qquad n\in\mathbb Z.
\end{equation*}
By the definition of $d_n$, this implies for $n\in\mathbb Z$,
	\begin{equation*}
f_n(x_n+z_n)-f_n(x_n)=x_{n+1}-T_n(x_n)-f_n(x_n)+z_{n+1}-T_n(z_n).
\end{equation*}
Hence
\begin{equation*}
x_{n+1}+z_{n+1}
=T_n(x_n+z_n)+f_n(x_n+z_n),
\qquad n\in\mathbb Z.
\end{equation*}
Therefore, setting
\begin{equation*}
y_n:=x_n+z_n,\qquad n\in\mathbb Z,
\end{equation*}
we obtain
\begin{equation*}
y_{n+1}=F_n(y_n),\qquad n\in\mathbb Z.
\end{equation*}
Thus, $y=(y_n)_{n\in\mathbb Z}$ is an exact bi-infinite trajectory of $(F_n)_{n\in\mathbb Z}$. Since each $F_n$, $n\in\mathbb Z$, is a homeomorphism,
\begin{equation*}
y_n=F(n,0)(y_0),\qquad n\in\mathbb Z.
\end{equation*}
Finally, \eqref{zest} implies that
\begin{equation*}
\sup_{n\in\mathbb Z}\|x_n-F(n,0)(y_0)\|
=\sup_{n\in\mathbb Z}\|x_n-y_n\|=\sup_{n\in\mathbb Z}\|z_n\|
=\|z\|_\infty
\le \|z\|_\infty'
\le \epsilon.
\end{equation*}
Therefore, $(F_n)_{n\in\mathbb Z}$ has the shadowing property.

By Theorem~\ref{thm1}, the topological stability of $(T_n)_{n\in\mathbb Z}$ implies that $(T_n)_{n\in\mathbb Z}$ is generalized pseudo-hyperbolic. Let $G_{m,n}\in H(X)$, $m,n\in\mathbb Z$, be the corresponding Green family given by~\eqref{Gmndef}. Thus,
\begin{equation}\label{green_ident}
G_{m+1,n}-T_mG_{m,n}=
\begin{cases}
I,\quad & m=n,\\
0,\quad & m\neq n,
\end{cases}
\end{equation}
and there exist constants $D,\lambda>0$ such that
\begin{equation}\label{green_est}
\|G_{m,n}\|\le De^{-\lambda|m-n-1|},
\qquad m,n\in\mathbb Z.
\end{equation}
Set
\begin{equation}\label{mudef}
K_0:=D\sum_{j\in\mathbb Z}e^{-\lambda|j|}
=D\frac{1+e^{-\lambda}}{1-e^{-\lambda}}>0.
\end{equation}
Choose
\begin{equation}\label{eta_choice}
0<\eta<\frac1{K_0}.
\end{equation}
Suppose that $(f_n)_{n\in\mathbb Z}$ is a sequence of Lipschitz self-maps of $X$ such that
\begin{equation*}
\operatorname{Lip}(f_n)<\eta\qquad\text{and}\qquad F_n:=T_n+f_n\in Homeo(X)
\quad\text{for all $n\in\mathbb Z$}.
\end{equation*}
We prove that $(F_n)_{n\in\mathbb Z}$ is topologically stable.
Let $\epsilon>0$ and choose
\begin{equation}\label{delta_choice}
\delta:=\frac{1-\eta K_0}{K_0}\,\epsilon>0.
\end{equation}
Let $(g_n)_{n\in\mathbb Z}$ be an arbitrary sequence in $Homeo(X)$ satisfying
\begin{equation}\label{g_close}
\sup_{n\in\mathbb Z}\|F_n-g_n\|_\infty\le\delta.
\end{equation}
Set
\begin{equation}\label{pndef}
p_n:=F_n-g_n,
\qquad n\in\mathbb Z.
\end{equation}
Then
\begin{equation}\label{pn_est}
\sup_{n\in\mathbb Z}\|p_n\|_\infty\le\delta.
\end{equation}
Let $C_b(X)$ denote the space of bounded continuous self-maps of~$X$ equipped with the norm $\|\cdot\|_\infty$. Since $C_b(X)$ is a Banach space, the space $\ell_\infty(C_b(X))=\ell_\infty(\mathbb Z,C_b(X))$ is also a Banach space. Recall that the norm on $\ell_\infty(C_b(X))$ is given by
\begin{equation*}
\|q\|_\infty:=\sup_{n\in\mathbb Z}\|q_n\|_\infty<\infty\qquad\text{for $q=(q_n)_{n\in\mathbb Z}\in\ell_\infty(C_b(X))$}.
\end{equation*}
Given $q=(q_n)_{n\in\mathbb Z}\in\ell_\infty(C_b(X))$, define
\begin{equation}\label{Rg_def}
(\mathcal R_gq)_n(x)
:=\sum_{m\in\mathbb Z}G_{n,m}\bigl(q_m(g(m,n)(x))\bigr)
\end{equation}
for $n\in\mathbb Z$ and $x\in X$, where $g(m,n)$ denotes the evolution family associated with $(g_n)_{n\in\mathbb Z}$.
By~\eqref{green_est},
\begin{equation*}
\begin{aligned}
\|G_{n,m}(q_m(g(m,n)(x)))\|
&\le De^{-\lambda|n-m-1|}\,
\|q_m(g(m,n)(x))\|\\
&\le De^{-\lambda|n-m-1|}\|q\|_\infty.
\end{aligned}
\end{equation*}
Since
\begin{equation*}
\sum_{m\in\mathbb Z}e^{-\lambda|n-m-1|}
=\sum_{j\in\mathbb Z}e^{-\lambda|j|}<\infty,
\qquad n\in\mathbb Z,
\end{equation*}
the series in~\eqref{Rg_def} converges uniformly in $x\in X$, for every $n\in\mathbb Z$. Hence each $(\mathcal R_gq)_n$, $n\in\mathbb Z$, is continuous and
\begin{equation*}
\|\mathcal R_gq\|_\infty
:=\sup_{n\in\mathbb Z}\|(\mathcal R_gq)_n\|_\infty
\leq D\|q\|_\infty\sum_{m\in\mathbb Z}e^{-\lambda|n-m-1|}
= D\|q\|_\infty\sum_{j\in\mathbb Z}e^{-\lambda|j|}.
\end{equation*}
Hence, by~\eqref{mudef},
\begin{equation}\label{Rg_est}
	\|\mathcal R_gq\|_\infty\leq K_0\|q\|_\infty. 
\end{equation}
Thus, $\mathcal R_g$ maps $\ell_\infty(C_b(X))$ into itself.
Moreover, by the evolution property,
\begin{equation*}
g(m,n+1)\circ g_n=g(m,n),
\qquad m,n\in\mathbb Z.
\end{equation*}
Therefore, using~\eqref{Rg_def}, we obtain for $n\in\mathbb Z$ and $x\in X$,
\begin{equation*}
\begin{aligned}
&(\mathcal R_gq)_{n+1}(g_n(x))-T_n(\mathcal R_gq)_n(x)\\
&=\sum_{m\in\mathbb Z}G_{n+1,m}
\bigl(q_m(g(m,n+1)(g_n(x)))\bigr)
-T_n\sum_{m\in\mathbb Z}G_{n,m}
\bigl(q_m(g(m,n)(x))\bigr)\\
&=\sum_{m\in\mathbb Z}
\bigl(G_{n+1,m}-T_nG_{n,m}\bigr)
\bigl(q_m(g(m,n)(x))\bigr).
\end{aligned}
\end{equation*}
By~\eqref{green_ident}, the only nonzero term in the last sum corresponds to $m=n$. Since $g(n,n)=I$, it follows that for every $x\in X$,
\begin{equation}\label{Rg_rightinverse}
(\mathcal R_gq)_{n+1}(g_n(x))-T_n(\mathcal R_gq)_n(x)=q_n(x),
\qquad n\in\mathbb Z.
\end{equation}
For $h=(h_n)_{n\in\mathbb Z}\in\ell_\infty(C_b(X))$, define
\begin{equation}\label{N_def}
(\mathcal N(h))_n(x):=f_n(x+h_n(x))-f_n(x),
\qquad n\in\mathbb Z,\quad x\in X.
\end{equation}
Since $\operatorname{Lip}(f_n)<\eta$ for all $n\in\mathbb Z$, it follows that
\begin{equation}\label{N_bound}
\|\mathcal N(h)\|_\infty\le\eta\|h\|_\infty,\qquad h\in\ell_\infty(C_b(X)).
\end{equation}
Similarly,
\begin{equation}\label{N_lip}
\|\mathcal N(h)-\mathcal N(k)\|_\infty
\le\eta\|h-k\|_\infty,\qquad h,\,k\in\ell_\infty(C_b(X)).
\end{equation}
Thus, $\mathcal N$ is a Lipschitz self-map of $\ell_\infty(C_b(X))$.



For $h=(h_n)_{n\in\mathbb Z}\in\ell_\infty(C_b(X))$, define 
\begin{equation*}
(\mathcal L_gh)_n(x):=h_{n+1}(g_n(x))-T_nh_n(x),
\qquad n\in\mathbb Z,\quad x\in X.
\end{equation*}
We construct inductively a sequence
$h^{(k)}=(h_n^{(k)})_{n\in\mathbb Z}$, $k\in\mathbb N_0$,
in $\ell_\infty(C_b(X))$ such that
$\mathcal L_g h^{(k)}\in\ell_\infty(C_b(X))$ for every $k\in\mathbb N_0$.
Set
$h^{(0)}=0$.
Suppose that $h^{(k)}\in\ell_\infty(C_b(X))$ has been defined such that $\mathcal L_gh^{(k)}\in\ell_\infty(C_b(X))$ for some $k\in\mathbb N_0$. Set
\begin{equation}\label{res_def_general}
r^{(k)}:=p+\mathcal N(h^{(k)})-\mathcal L_gh^{(k)}\in\ell_\infty(C_b(X)),
\end{equation}
where $p=(p_n)_{n\in\mathbb Z}\in\ell_\infty(C_b(X))$ is defined by~\eqref{pndef}.
Define
\begin{equation}\label{ak_def}
a^{(k)}:=\mathcal R_gr^{(k)}\in\ell_\infty(C_b(X))
\end{equation}
and
\begin{equation}\label{hk_step}
h^{(k+1)}:=h^{(k)}+a^{(k)}\in\ell_\infty(C_b(X)).
\end{equation}
By~\eqref{Rg_rightinverse}, $\mathcal L_g\mathcal R_g q=q$ for all $q\in\ell_\infty(C_b(X))$. Hence
\begin{equation*}
\mathcal L_ga^{(k)}=\mathcal L_g\mathcal R_g r^{(k)}=r^{(k)}\in\ell_\infty(C_b(X)).
\end{equation*}
Since $\mathcal L_g$ is linear, this and~\eqref{hk_step} imply that
\begin{equation}\label{Lga_r}
\mathcal L_gh^{(k+1)}=\mathcal L_gh^{(k)}+r^{(k)}\in\ell_\infty(C_b(X)).
\end{equation} 
From \eqref{res_def_general} and~\eqref{Lga_r}, we obtain
\begin{equation*}
\begin{aligned}
r^{(k+1)}
&=p+\mathcal N(h^{(k+1)})-\mathcal L_gh^{(k+1)}\\
&=p+\mathcal N(h^{(k+1)})-\mathcal L_gh^{(k)}-r^{(k)}\\
&=\mathcal N(h^{(k+1)})-\mathcal N(h^{(k)}).
\end{aligned}
\end{equation*}
Consequently,~\eqref{N_lip}, \eqref{hk_step}, ~\eqref{ak_def} and \eqref{Rg_est}, yield
\begin{equation*}
\|r^{(k+1)}\|_\infty
\le\eta\|h^{(k+1)}-h^{(k)}\|_\infty
=\eta\|a^{(k)}\|_\infty
=\eta\|\mathcal R_g r^{(k)}\|_\infty
\le\eta K_0\|r^{(k)}\|_\infty
\end{equation*}
for $k\in\mathbb N_0$.
Since $r^{(0)}=p$, from the last inequality and~\eqref{pn_est}, we obtain
\begin{equation}\label{rk_est}
\|r^{(k)}\|_\infty\le(\eta K_0)^k\|r^{(0)}\|_\infty=(\eta K_0)^k\|p\|_\infty
\le(\eta K_0)^k\delta,
\qquad k\in\mathbb N_0.
\end{equation}

\begin{equation}\label{hk_dif}
\|h^{(k+1)}-h^{(k)}\|_\infty
=\|a^{(k)}\|_\infty=\|\mathcal R_g r^{(k)}\|_\infty
\le K_0\|r^{(k)}\|_\infty
\le K_0(\eta K_0)^k\delta.
\end{equation}

$h^{(k)}\to h$. Moreover, by~\eqref{hk_dif},
\begin{equation*}
\|h\|_\infty
\le\sum_{k=0}^\infty
\|h^{(k+1)}-h^{(k)}\|_\infty
\le K_0\delta\sum_{k=0}^\infty(\eta K_0)^k
=\frac{K_0\delta}{1-\eta K_0}
=\epsilon,
\end{equation*}
where the last equality follows from~\eqref{delta_choice}. Hence
\begin{equation}\label{h_est}
\sup_{n\in\mathbb Z}\|h_n\|_\infty\le\epsilon.
\end{equation}

\begin{equation*}
h_n^{(k)}(x)\rightarrow h_n(x)\qquad\text{and}\qquad h_{n+1}^{(k)}(g_n(x))\rightarrow h_{n+1}(g_n(x))\qquad\text{as $k\to\infty$}
\end{equation*}
for all $n\in\mathbb Z$ and $x\in X$. Moreover, by the continuity of~$T_n$,
\begin{equation*}
T_n h^{(k)}_n(x)\to T_n h_n(x)\qquad\text{as $k\to\infty$}
\end{equation*}
for all $n\in\mathbb Z$ and $x\in X$.
Therefore, passing to the limit in~\eqref{res_def_general}, pointwise for the fixed $n$ and $x$, yields
\begin{equation*}
0=p_n(x)+(\mathcal N(h))_n(x)
-h_{n+1}(g_n(x))+T_nh_n(x).
\end{equation*}
This, combined with~\eqref{N_def}, implies for $n\in\mathbb Z$ and $x\in X$,
\begin{equation}\label{cohom_eq_2}
h_{n+1}(g_n(x))-T_nh_n(x)
=p_n(x)+f_n(x+h_n(x))-f_n(x).
\end{equation}

\begin{equation}\label{Hn_def}
H_n:=I+h_n,
\qquad n\in\mathbb Z.
\end{equation}
Each $H_n$, $n\in\mathbb Z$, is continuous. Using~\eqref{cohom_eq_2}, \eqref{pndef}, and $F_n=T_n+f_n$, we obtain for $n\in\mathbb Z$ and $x\in X$,
\begin{equation*}
\begin{aligned}
H_{n+1}(g_n(x))
&=g_n(x)+h_{n+1}(g_n(x))\\
&=g_n(x)+T_nh_n(x)+p_n(x)
+f_n(x+h_n(x))-f_n(x)\\
&=g_n(x)+T_nh_n(x)+F_n(x)-g_n(x)
+f_n(x+h_n(x))-f_n(x)\\
&=T_nx+T_nh_n(x)+f_n(x+h_n(x))\\
&=F_n(x+h_n(x))\\
&=F_n(H_n(x)).
\end{aligned}
\end{equation*}
Hence
\begin{equation}\label{semiconj_final}
H_{n+1}\circ g_n=F_n\circ H_n,
\qquad n\in\mathbb Z.
\end{equation}
Moreover, by~\eqref{h_est},
\begin{equation*}
\sup_{n\in\mathbb Z}\|H_n-I\|_\infty
=\sup_{n\in\mathbb Z}\|h_n\|_\infty
\le\epsilon.
\end{equation*}
Since $(g_n)_{n\in\mathbb Z}$ was an arbitrary sequence in $Homeo(X)$ satisfying~\eqref{g_close}, it follows that $(F_n)_{n\in\mathbb Z}$ is topologically stable.
