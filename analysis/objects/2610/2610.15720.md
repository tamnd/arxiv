\label{ex:CFKM}
An explicit example of an intermittent map to which our theory applies is the map $T:M\to M$, $M=[0,1]$, given by
\begin{equation}
    \label{eq:CFKM}
    Tx = \begin{cases} 
        x(1+3^{1/\alpha} x^{1/\alpha}), & x \in[0, \frac{1}{3})  , \\
        3x-1, & x\in[\frac{1}{3}, \frac{2}{3})  , \\
        1 - (1-x) (1 + 3^{1/\alpha} (1-x)^{1/\alpha}), & x\in[ \frac{2}{3},1],
  \end{cases}
\end{equation}
where $\alpha\in(1,2]$. There is a unique absolute continuous $T$-invariant probability measure $\mu_M$.
This example is taken from~\cite{CFKM20}; the symmetry around $x=\frac12$ is irrelevant but we require that the exponents $1+1/\alpha$ at the neutral fixed points coincide.

Let $a_n=n^{1/\alpha}$ for $\alpha\in(1,2)$ and
$a_n=(n\log n)^{1/2}$ for $\alpha=2$.
Given $v:M\to{\mathbb R}^d$, we define $v_n=\sum_{j=0}^{n-1}v\circ T^j$.

\begin{thm} \label{thm:CFKM}
Let $T:M\to M$ be the map~\eqref{eq:CFKM} with $\alpha\in(1,2]$.
Let $v:M\to{\mathbb R}$ be a H\"older observable with $\int_M v\,d\mu_M=0$ and satisfying $v(0),\,v(1)\neq0$ and $v(0)\neq v(1)$.  It is known (see for example~\cite{CFKM20}) that $a_n^{-1}v_n$ converges in distribution to a two-sided $\alpha$-stable law $X_\alpha$; we denote the density of $X_\alpha$ by $g$.

 Under an aperiodicity assumption that holds typically,
\[
\lim_{n\to\infty}a_n\int_M \zeta_1\,\zeta_2\circ T^n\,\xi\circ(v_n-qa_n)\,d\mu_M
=g(q)\int_M \zeta_1\,d\mu_M \int_M \zeta_2\,d\mu_M \int_{\mathbb R} \xi\,d\operatorname{Leb},
\]
for all $\zeta_1,\,\zeta_2\in C_c\big((0,1)\big)$, $\xi\in C_c({\mathbb R})$, $q\in{\mathbb R}$.
\end{thm}

The $\xi$ component encompasses the stable local limit theorem, whereas the test functions $\zeta_1$ and $\zeta_2$ represent mixing properties of $T$. Hence Theorem~\ref{thm:CFKM} is called a \emph{mixing local limit theorem (MLLT)}.

\label{ex:Thaler}
We can extend to $C^\eta$ vector-valued observables $v:M\to{\mathbb R}^d$ of Thaler maps (see Section~\ref{sec:LSV}) with $k$ neutral fixed points $x_1,\dots,x_k$ all with exponent $1+1/\alpha$. We again assume that $v(x_i)\neq0$, $i=1,\dots,k$, and we require further that the vectors $\{(v(x_i),1),\, i=1,\dots,k\}$ span ${\mathbb R}^{d+1}$. (In particular, $k\ge d+1$.)
We also assume an aperiodicity assumption that holds typically.
Provided $d<\alpha^2$, we obtain a stable MLLT of the form
\[
\lim_{n\to\infty}a_n^d\int_M \zeta_1\,\zeta_2\circ T^n\,\xi\circ(v_n-qa_n)\,d\mu_M
=g(q)\int_M \zeta_1\,d\mu_M \int_M \zeta_2\,d\mu_M \int_{{\mathbb R}^d} \xi\,d\operatorname{Leb},
\]
for all $\zeta_1,\,\zeta_2\in C_c\big(M\setminus\{x_1,\dots,x_k\}\big)$, $\xi\in C_c({\mathbb R}^d)$, $q\in{\mathbb R}^d$.
Here $g$ is the density of the $d$-dimensional $\alpha$-stable law limit for $n^{-1/\alpha}v_n$.

The assumption $d<\alpha^2$ can be 
removed provided $\eta\in(0,1]$ is sufficiently close to $1$ and a further technical condition is satisfied. For this it suffices that $v(x_i)$ has positive first coordinate for all $i=1,\dots,k$.

where
\[
Q_m(t,u)=\mu\big(y\in A_0\cap F^{-m}B_0:\,
\Psi_m(y)  \in (qa_t,
t-m{\bar{\tau}})+K_u\big).
\]

Fixing $L>0$, we can write
\[
{\bar{\tau}}  \omega_t= S_1(t,L)+S_2(t,L)+o(a_t^{-d})
\]
where
\[
  S_1(t,L)=\sum_{\stackrel{m>t^\gamma:}{|m-t/{\bar{\tau}}|\le La_t}}\int_I Q_m(t,u)\,du, \qquad
  S_2(t,L)=\sum_{\stackrel{m>t^\gamma:}{|m-t/{\bar{\tau}}|> La_t}}\int_I Q_m(t,u)\,du.
\]

Thus, it suffices to show:
\begin{align} \label{eq:S1}
& \lim_{L\to\infty}\lim_{t\to\infty} a_t^d S_1(t,L)=
 {\bar{\tau}} g(q)\mu^\tau(A)\mu^\tau(B)|K|, \\
& \lim_{L\to\infty}\limsup_{t\to\infty} a_t^d S_2(t,L)=0.
\label{eq:S2}
\end{align}

In~\eqref{eq:S1}, we are in the range $|m-t/{\bar{\tau}}|\le La_t$.
It follows that
$a_t\sim {\bar{\tau}}^{1/\alpha}a_m$ and $z_{m,t}=\Big(\dfrac{qa_t}{a_m},\dfrac{t-m{\bar{\tau}}}{a_m}\Big)$ is bounded.
By \ref{itm:N1},
\begin{align*}
a_m^{d+1}Q_m(t,u) & =
g_{d+1}(z_{m,t})\mu(A_0)\mu(B_0)|K_u|+h_m(t,u)
\\ &  =
{\bar{\tau}} g_{d+1}(z_{m,t})\mu(A_0)\mu^\tau(B)|K|+h_m(t,u)
\end{align*}
where $\lim_{t\to\infty}h_m(t,u)=0$ uniformly in $|m-t/{\bar{\tau}}|\le La_t$ and $u\in I$.
Hence
\[
a_t^d\int_I Q_m(t,u)\,du  
                                                                                        
={\bar{\tau}}^{2+(d+1)/\alpha} a_t^{-1} g_{d+1}\big(q{\bar{\tau}}^{1/\alpha},(t/{\bar{\tau}}-m){\bar{\tau}}^{1+1/\alpha} a_t^{-1}\big)\kappa+\tilde h_m(t)
\]
where $\kappa=\mu^\tau(A)\mu^\tau(B)|K|$ and
$\tilde h_m(t)=o(a_t^{-1})$ as $t\to\infty$ uniformly in $|m-t/{\bar{\tau}}|\le La_t$.
It follows that
\begin{align*}
  a_t^d  S_1(t,L) 
 ={\bar{\tau}}^{2+(d+1)/\alpha} a_t^{-1}\sum_{|j|\le La_t} 
g_{d+1}\big(q{\bar{\tau}}^{1/\alpha},j{\bar{\tau}}^{1+1/\alpha} a_t^{-1}\big)\kappa+o(1)
\end{align*}
as $t\to\infty$ for each fixed $L$.
Approximating Riemann integrals by Riemann sums with increments of width ${\bar{\tau}}^{1+1/\alpha} a_t^{-1}$ yields
\[
{\bar{\tau}}^{1+1/\alpha} a_t^{-1}
\sum_{|j|\le La_t}
g_{d+1}\big(q{\bar{\tau}}^{1/\alpha},j{\bar{\tau}}^{1+1/\alpha} a_t^{-1}\big)
 \to  
\int_{-L{\bar{\tau}}^{1+1/\alpha}}^{L{\bar{\tau}}^{1+1/\alpha}} g_{d+1}(q{\bar{\tau}}^{1/\alpha},y)\,dy
\]
as $t\to\infty$.
Hence
\[
\lim_{L\to\infty} \lim_{t\to\infty}a_t^dS_1(t,L)
  = {\bar{\tau}}^{1+d/\alpha} g_d(q{\bar{\tau}}^{1/\alpha})\, \kappa
  = {\bar{\tau}} g(q)\, \kappa
\]
by definition of $g_d$ and $g$ in~\eqref{eq:gd}.
This completes the verification of~\eqref{eq:S1}.

\vspace{1ex}
Next, $Q_m(t,u)\le 
\sup_{u\in I} \mu(\Psi_m\in \big(qa_t,t-m{\bar{\tau}})+K_u\big)$.
In~\eqref{eq:S2}, we are
in the range $a_t\ll |m-t/{\bar{\tau}}|$. Hence $|(qa_t,t-m{\bar{\tau}})|\approx |t-m{\bar{\tau}}|$ and it follows from~\ref{itm:N2} that
\[
Q_m(t,u)\ll {\widetilde Q}_m(t) \quad\text{where}\quad
 {\widetilde Q}_m(t)=\frac{m}{a_m^{d+1}}\, \frac{\tilde\ell(|t-m{\bar{\tau}}|)}{|t-m{\bar{\tau}}|^\alpha}\Big(1+\Big(\frac{|t-m{\bar{\tau}}|}{a_m}\Big)^r\Big)
\]
for any $r>0$ sufficiently small.
Hence~\eqref{eq:S2} is an immediate consequence of the next three results.


\begin{prop} \label{prop:1}
\(
\lim_{L\to\infty}\limsup_{t\to\infty}a_t^d\sum_{La_t<| m-t/{\bar{\tau}}|<ct}{\widetilde Q}_m(t)=0
\)
for all $c\in(0,1/{\bar{\tau}})$.
\end{prop}

\begin{proof}
By the choice of $c$, in this range, $m\approx t$.
Choose $r\in(0,\alpha-1)$. Then
\begin{align*}
\sum_{La_t<  |m-t/{\bar{\tau}}|<ct} {\widetilde Q}_m(t) 
  & \ll \frac{t}{a_t^{d+1}}
\sum_{ m-t/{\bar{\tau}}> L a_t}
\frac{\tilde\ell(m{\bar{\tau}}-t)}{(m{\bar{\tau}}-t)^\alpha} \Big(1+\Big(\frac{m{\bar{\tau}}-t}{a_t}\Big)^r\Big) \\
  & \ll \frac{t}{a_t^{d+1+r}}
\sum_{ m-t/{\bar{\tau}}> L a_t}
\frac{\tilde\ell(m{\bar{\tau}}-t)}{(m{\bar{\tau}}-t)^{\alpha-r}} \\
&  \approx 
\frac{t}{a_t^{d+1+r}} \frac{\tilde\ell(a_t)}{(La_t)^{\alpha-1-r}}
\approx \frac{1}{a_t^d} \frac{1}{L^{\alpha-1-r}}\frac{t\tilde\ell(a_t)}{a_t^{\alpha}}
\end{align*}
by Karamata. 
The result follows since $a_t^\alpha\sim t\tilde\ell(a_t)$.
\end{proof}


\begin{prop} \label{prop:2}
\(
\lim_{t\to\infty}a_t^d\sum_{m-t/{\bar{\tau}} \ge ct}{\widetilde Q}_m(t)=0
\)
for all $c>0$.
\end{prop}

\begin{proof}
In this range, $m\gg t$ so choosing $r\in(0,\alpha-1)$, 
\[
\sum_{ m-t/{\bar{\tau}}\ge ct} {\widetilde Q}_m(t) 
 \ll \frac{t}{a_t^{d+1}}
\sum_{ m-t/{\bar{\tau}}\ge ct}
\frac{\tilde\ell(m{\bar{\tau}}-t)}{(m{\bar{\tau}}-t)^{\alpha-r}}
 \approx \frac{t}{a_t^{d+1}} \frac{\tilde\ell(t)}{t^{\alpha-1-r}}
 =  \frac{1}{a_t^d}\;\frac{\tilde\ell(t)}{t^{\alpha-2-r}a_t}.
\]
By Potter's bounds,
\[
  \frac{\tilde\ell(t)}{t^{\alpha-2-r}a_t}\ll \frac{t^s}{t^{\alpha+1/\alpha-2-r-s}}
=t^{-(\alpha+1/\alpha-2-r-2s)},
\]
for all $s>0$.
Since $\alpha+1/\alpha>2$, we can shrink $r,\,s$ so that $\alpha+1/\alpha>2+r+2s$. Then
  $\dfrac{\tilde\ell(t)}{t^{\alpha-2-r}a_t}=o(1)$
as required.
\end{proof}

\begin{prop} \label{prop:3}
\(
\lim_{t\to\infty}a_t^d\sum_{m\ge t^\gamma,\,m-t/{\bar{\tau}}\le -ct}{\widetilde Q}_m(t)=0
\)
for all $c>0$,
\end{prop}

\begin{proof}
Write the range as $t^\gamma\le m\le c't$ where
$c'=1/{\bar{\tau}}-c$. (If $c'\le0$, then there is nothing to prove.)
Let $r,\,s>0$.
Since $t-m{\bar{\tau}}\gg t$,
\begin{align*}
\sum_{t^\gamma\le m\le c't} {\widetilde Q}_m(t)
\ll \sum_{t^\gamma\le m\le c't}\frac{m}{a_m^{d+1}}\,\frac{\tilde\ell(t)}{t^{\alpha-r}}
& \ll t^{-\alpha+r+s} \sum_{t^\gamma\le m\le c't}\frac{1}{m^{(d+1)/\alpha-s-1}}
\\ & \approx t^{-\alpha+r+s} 
\big\{(t^\gamma)^{2-(d+1)/\alpha+s}
+t^{2-(d+1)/\alpha+s}\big\}.
\end{align*}
Since $\alpha+1/\alpha>2$, we can shrink $r,\,s>0$ so that
\[
a_t^d\sum_{t^\gamma\le m\le c't} {\widetilde Q}_m(t)\ll t^{d/\alpha-\alpha+r+2s}(t^\gamma)^{2-(d+1)/\alpha+s}+o(1).
\]

If $d<\alpha^2$, then $\gamma=0$ and we see immediately that $a_t^d\sum_{1\le m\le c't} {\widetilde Q}_m(t)=o(1)$ for $r,\,s>0$ sufficiently small.

If $d\ge \alpha^2$, then the choice of $\gamma>\gamma_0$ ensures that
$d/\alpha-\alpha+\gamma(2-(d+1)/\alpha)<0$ and the result again follows.
\end{proof}

This completes the proof of the lemma.

It suffices to show that $S_t=o(a_t^{-d})$.

Write $W=\omega\tau+Z=\omega\tau-\int_Y\omega\tau\,d\mu+{\widetilde Z}$ where
${\widetilde Z}=Z-\int_Y Z\,d\mu$.
Then ${\widetilde Z}_m=W_m-(\omega\tau)_m+m\int_Y\omega\tau\,d\mu$.
If $|(\omega\tau)_m|\ge c\tau_m$, then  
the conditions $|W_m|\ll a_t$, $\tau_m\approx t$ and $m\ll t^\gamma$ imply that $|{\widetilde Z}_m|\approx t$ since $\gamma<1$. Hence $I_m\subset I_m'\cup I_m''$ where
\begin{align*}
I_m' & =
\Big\{(y,u)\in Y^\tau:|(\omega\tau)_m|< c\tau_m
,\,\tau_m(y)\le u+t<\tau_{m+1}(y),\,|u|\ll 1\Big\},
\\
I_m'' & =
\Big\{(y,u)\in Y^\tau: |{\widetilde Z}_m(y)|\approx t
,\,\tau_m(y)\le u+t<\tau_{m+1}(y),\,|u|\ll 1\Big\}.
\end{align*}

The sets $\{\tau_m(y)\le u+t<\tau_{m+1}(y)\}$ are disjoint, so
\begin{align*}
\sum_{1\le m\le t^\gamma}\mu^\tau(I_m')
& 	\le \mu^\tau\Big(\min_{1\le m\le t^\gamma}|(\omega\tau)_m|/\tau_m < c,\,|u|\ll1\Big)
\\ & \ll
	\mu\Big(\min_{1\le m\le t^\gamma}|(\omega\tau)_m|/\tau_m < c\Big)
=o(a_t^{-d}).
\end{align*}

Similarly,
\[
\sum_{1\le m\le t^\gamma}\mu^\tau(I_m'')
\ll \mu\Big(\max_{1\le m\le t^\gamma}|{\widetilde Z}_m|\gg t\Big).
\]
It follows from~\eqref{eq:Psireg} that ${\widetilde Z}$ is piecewise H\"older.
Since  ${\widetilde Z}\in L^2$,
standard moment estimates~\cite{M09b} for Gibbs-Markov maps yield
$\big|\max_{1\le m\le t}|{\widetilde Z}_m|\,\big|_p\ll t^{1/2}$.
By the definitions
\[
p(1-\tfrac12 \gamma_0)=\tfrac12 p\, \frac{d+\alpha^2-4\alpha+2}{d+1-2\alpha}>\frac{d}{\alpha}.
\]
Hence we can shrink $\gamma>\gamma_0$ so that by Markov's inequality,
\[
\sum_{1\le m\le t^\gamma}\mu^\tau(I_m'')
\ll (t^\gamma)^{p/2}/t^p = t^{-p(1-\frac12\gamma)}=o(a_t^{-d}),
\]
completing the proof.

\[

=\Big|\sum_{j=0}^{R(y)-1}(v(T^j y)-v(0))\Big|
\le |v|_\eta \sum_{j=0}^{R(y)-1}(T^j y)^\eta.

\label{lem:vague}
Suppose that
\begin{equation} \label{eq:vague}
\lim_{t\to\infty}a_\gamma(t) {\mathbb P} (\omega\in A\cap F_t^{-1}B:\,
\zeta_{t,\gamma}(\omega)\in E) = {\mathbb P}(A){\mathbb P}(B)\operatorname{Leb}(E)
\end{equation}
uniformly in $\gamma\in\Gamma$,
for all $A,B\in{\mathcal A}$, $E\in{\mathcal E}$.
Suppose that $H_1,H_2:\Omega\to{\mathbb R}^m$ satisfy property~\ref{itm:H}.
Then
\[
\lim_{t\to\infty}a_\gamma(t) {\mathbb P} \left(\omega\in A\cap F_t^{-1}B:\, \zeta_{t,\gamma}(\omega)+H_1(\omega)+H_2(F_t\omega)\in E\right) = {\mathbb P}(A){\mathbb P}(B)\operatorname{Leb}(E)
\]
uniformly in $\gamma\in\Gamma$,
for all $A,B\in{\mathcal A}$, $E\in{\mathcal E}$.

Fix $A,\,B\in{\mathcal A}$ and 
introduce the family of measures
${\mathbb Q}_t(D)=a_\gamma(t)\, {\mathbb P}(A\cap F_t^{-1}B\cap D)$ on ${\mathbb R}^m$.
Also, let ${\mathbb Q}(D)={\mathbb P}(A){\mathbb P}(B)\operatorname{Leb}(D)$.
Define
\[
S^{t,\gamma}:\Omega\to{\mathbb R}^m, \qquad
S^{t,\gamma}=\zeta_{t,\gamma}+H_1+H_2\circ F_t.
\]
We are required to prove that
$\lim_{t\to\infty}S^{t,\gamma}_*\,{\mathbb Q}_t(E)={\mathbb Q}(E)$ for all
$E\in{\mathcal E}$.
Note that $S^{t,\gamma}_*\,{\mathbb Q}_t({\mathbb R}^m)=a_\gamma(t)$, so the measures $S^{t,\gamma}_*\,{\mathbb Q}_t$ are trivially Radon measures (finite on compact sets).
Also ${\mathbb Q}$ is a Radon measure.
By Lemma~\ref{lem:Port} below,
 it suffices to prove that
\begin{equation} \label{eq:port}
\lim_{t\to\infty}\int_{{\mathbb R}^m}g\,d S^{t,\gamma}_*\,{\mathbb Q}_t=\int_{{\mathbb R}^m}g\,d{\mathbb Q}
\end{equation}
uniformly in $\gamma$, for all $g\in C_c({\mathbb R}^m)$.

Let $g\in C_c({\mathbb R}^m)$, and define
\[
G:({\mathbb R}^m)^3\to {\mathbb R},
\qquad G(x,y,z)=g(x+y+z).
\]
We claim that
\begin{align} \label{eq:vagueclaim} \nonumber
\lim_{t\to +\infty}a_\gamma(t)&\int_{A\cap F_t^{-1}B}G(H_1(\omega),H_2(F_t \omega),\zeta_{t,\gamma}(\omega))\, d{\mathbb P}(\omega)\\
&=\int_{A\times B\times{\mathbb R}^m}G(H_1(\omega),H_2(\omega'),z)d({\mathbb P}\times{\mathbb P}\times\operatorname{Leb})(\omega,\omega',z)  
\end{align}
uniformly in $\gamma$.
Suppose that the claim is true. Then by definition of $G$,
\begin{align*}
\lim_{t\to +\infty}a_\gamma(t)&\int_{A\cap F_t^{-1}B}g(\zeta_{t,\gamma}(\omega)+H_1(\omega)+H_2(F_t \omega))\, d{\mathbb P}(\omega)\\
&=\int_{A\times B\times{\mathbb R}^m}g(H_1(\omega)+H_2(\omega')+z)d({\mathbb P}\times{\mathbb P}\times\operatorname{Leb})(\omega,\omega',z) 
\\ &=\int_{A\times B\times{\mathbb R}^m}g(z)d({\mathbb P}\times{\mathbb P}\times\operatorname{Leb})(\omega,\omega',z) 
\end{align*}
uniformly in $\gamma$,
yielding~\eqref{eq:port}.

It remains to prove the claim.
	Since $g$ is compactly supported and $H_1(A)$, $H_2(B)$ are bounded, we can find a closed disk $D\subset{\mathbb R}^m$ 
	such that $\operatorname{supp} G\subset {\mathbb R}^m\times{\mathbb R}^m\times D$.
For any ${\epsilon}>0$,
there exists $\delta\in(0,{\epsilon}/\{\operatorname{Leb}(D)(\| G\|_\infty+1)\})$ such that 
\begin{equation} \label{eq:G}
|G(x,y,z)-G(x',y',z')|<{\epsilon} \quad\text{for all}\quad
|x-x'|<\delta,\ |y-y'|<\delta,\ |z-z'|<\delta.
\end{equation} 
We partition $D$ into a finite number of disjoint compact
subsets $C_1,C_2,\dots$ with $\operatorname{Leb}(\partial C_k)=0$ and $\operatorname{diam} C_k<\delta$.

Since $H_1$, $H_2$ satisfy \ref{itm:H}, we can choose
disjoint subsets $A_i\subset A$ and disjoint subsets 
$B_j\subset B$ with $A_1,\dots,A_m;\,B_1,\dots,B_n\in{\mathcal A}$ such that
\begin{alignat}{2} \label{eq:N4a}
& {\mathbb P}(A\setminus A') < \delta, & \qquad 
& {\mathbb P}(B\setminus B') < \delta, \\
\label{eq:N4b}
& \max_{1\le i\le m} \operatorname{diam} H_1(A_i)  < \delta , & \qquad
& \max_{1\le j\le n} \operatorname{diam} H_2(B_j)  < \delta ,
\end{alignat}
where $A'=\bigcup_{i=1}^mA_i$ and $B'=\bigcup_{j=1}^nB_j$.

Now,
\begin{align*}
a_\gamma(t) & \int_{A\cap F_t^{-1}B}  G(H_1(\omega),H_2(F_t \omega),\zeta_{t,\gamma}(\omega))\, d{\mathbb P}(\omega)
\\ & =
a_\gamma(t)\int_{A\cap F_t^{-1}B\cap\zeta_{t,\gamma}^{-1}D}  G(H_1(\omega),H_2(F_t \omega),\zeta_{t,\gamma}(\omega))\, d{\mathbb P}(\omega)
\\ & =
a_\gamma(t)\int_{A'\cap F_t^{-1}B'\cap\zeta_{t,\gamma}^{-1}D}G(H_1(\omega),H_2(F_t \omega),\zeta_{t,\gamma}(\omega))\, d{\mathbb P}(\omega)+I_1(t)+I_2(t),
\end{align*}
where
\begin{align*}
|I_1(t,\gamma)| & \le \|G\|_\infty\, a_\gamma(t)\,
{\mathbb P}\big((A\setminus A')\cap F_t^{-1}B\cap \zeta_{t,\gamma}^{-1}D\big), \\
|I_2(t,\gamma)| & \le \|G\|_\infty\, a_\gamma(t)\,
{\mathbb P}\big(A'\cap F_t^{-1}(B\setminus B')\cap \zeta_{t,\gamma}^{-1}D\big).
\end{align*}
By~\eqref{eq:vague} and~\eqref{eq:N4a},
\[
\lim_{t\to\infty}a_\gamma(t)\,
{\mathbb P}\big((A\setminus A')\cap F_t^{-1}B\cap \zeta_{t,\gamma}^{-1}D\big)
={\mathbb P}(A\setminus A'){\mathbb P}(B)\operatorname{Leb}(D)<\delta\operatorname{Leb}(D)
\]
uniformly in $\gamma$.
Hence $\limsup_{t\to\infty}\sup_{\gamma\in\Gamma}|I_1(t,\gamma)|<\delta\|G\|_\infty\operatorname{Leb}(D)<{\epsilon}$.
Similarly, $\limsup_{t\to\infty}\sup_{\gamma\in\Gamma}|I_2(t,\gamma)|<{\epsilon}$.

Next, by~\eqref{eq:G} and~\eqref{eq:N4b},
\begin{align*}
a_\gamma(t) & \int_{A'\cap F_t^{-1}B'\cap\zeta^{-1}D} G(H_1(\omega),H_2(F_t \omega),\zeta_{t,\gamma}(\omega))\, d{\mathbb P}(\omega) \\
&= \sum_{i,j,k} a_\gamma(t) \int_{A_i\cap F_t^{-1}B_j\cap \zeta_{t,\gamma}^{-1}C_k}G(H_1(\omega),H_2(F_t \omega),\zeta_{t,\gamma}(\omega))\, d{\mathbb P}(\omega) \\
&= \sum_{i,j,k}  \Big\{\min_{H_1(A_i)\times H_2(B_j)\times C_k}G\Big\} a_\gamma(t){\mathbb P}(A_i\cap F_t^{-1}B_j\cap \zeta_{t,\gamma}^{-1}C_k)+I_3(t,\gamma),
\end{align*}
where
\[
|I_3(t,\gamma)|\le \delta\sum_{i,j,k} a_\gamma(t){\mathbb P}(A_i\cap F_t^{-1}B_j\cap \zeta_{t,\gamma}^{-1}C_k).
\]
Hence $\limsup_{t\to\infty}|I_3(t,\gamma)|\le \delta{\mathbb P}(A){\mathbb P}(B)\operatorname{Leb}(D)<{\epsilon}$.

By~\eqref{eq:vague},
\begin{align*}
 \lim_{t\to\infty} \sum_{i,j,k} &   \Big\{\min_{H_1(A_i)\times H_2(B_j)\times C_k}G\Big\} a_\gamma(t){\mathbb P}(A_i\cap F_t^{-1}B_j\cap \zeta_{t,\gamma}^{-1}C_k)
\\ & \qquad =
 \sum_{i,j,k}    \Big\{\min_{H_1(A_i)\times H_2(B_j)\times C_k}G\Big\} {\mathbb P}(A_i){\mathbb P}(B_j)\operatorname{Leb}(C_k).
\end{align*}
Reversing the first two steps of the calculations in a simpler situation,
\begin{align*}
 \sum_{i,j,k}    & \Big\{\min_{H_1(A_i)\times H_2(B_j)\times C_k}G\Big\} {\mathbb P}(A_i){\mathbb P}(B_j)\operatorname{Leb}(C_k).
\\ & =
\int_{A\times B\times {\mathbb R}^m}G(H_1(\omega),H_2(\omega'),z)\,d({\mathbb P}\times{\mathbb P}\times\operatorname{Leb})(\omega,\omega',z)+I_4,
\end{align*}
where $|I_4|\le \delta\operatorname{Leb}(D)+2\delta\|G\|_\infty\operatorname{Leb}(D)<3{\epsilon}$.

Since ${\epsilon}>0$ is arbitrary, the claim~\eqref{eq:vagueclaim} follows.

<!-- lemma-7: body_md -->
\label{lem:Port}
Let $\lambda_t^\gamma$, $t>0$, $\gamma\in\Gamma$, and $\lambda$ be Radon measures on ${\mathbb R}^m$. 
Let ${\mathcal E}_\lambda$ denote the collection of bounded measurable subsets of ${\mathbb R}^m$ with $\lambda(\partial E)=0$.
Then the following are equivalent:
\begin{itemize}
\item[(a)] $\int_{{\mathbb R}^m} g\,d\lambda_t^\gamma\to \int_{{\mathbb R}^m}g\,d\lambda$
uniformly in $\gamma$ for each $g\in C_c({\mathbb R}^m)$.
\item[(b)] $\lim_{t\to\infty}\lambda_t^\gamma(E)= \lambda(E)$ uniformly in $\gamma$
for each $E\in{\mathcal E}_\lambda$.
\end{itemize}
<!-- end lemma-7 -->
<!-- proof-19: body_md -->
First, suppose that (a) holds and let $K\subset {\mathbb R}^m$ be compact.
Choose a sequence of compact sets $K_j\subset{\mathbb R}^m$ and a sequence
of functions
$f_j\in C_c({\mathbb R}^m)$ such that
$K_{j+1}\subset K_j$, $\bigcap_j K_j=K$, and
$1_K\le f_j\le 1_{K_j}$ for all $j$.
Then by (a), for each fixed $j$,
\[
\lambda_t^\gamma(K)\le \int_{{\mathbb R}^m}f_j\,d\lambda_t^\gamma\to \int_{{\mathbb R}^m}f_j\,d\lambda
\]
uniformly in $\gamma$.
Hence $\limsup_{t\to\infty} \sup_\gamma \lambda_t^\gamma(K)\le \int_{{\mathbb R}^m}f_j\,d\lambda$.
Also, $|f_j|\le 1_{K_1}$ which is $\lambda$-integrable and $f_j\to 1_K$ pointwise as $j\to\infty$, so by the dominated convergence theorem $\limsup_{t\to\infty} \sup_\gamma \lambda_t^\gamma(K)\le \lambda(K)$.


Second, continue to suppose that (a) holds and let $G\subset{\mathbb R}^m$ be open and bounded. 
Choose a sequence of open sets $G_j\subset{\mathbb R}^m$ and a sequence of functions
$f'_j\in C_c({\mathbb R}^m)$ such that
$G_j\subset G_{j+1}$, $\bigcup_j G_j=G$, and
$1_G\ge f'_j\ge 1_{G_j}$. Then the same argument as before (but with dominating function $1_G$) shows that
$\liminf_{t\to\infty} \inf_\gamma \lambda_t^\gamma(G)\ge \lambda(G)$.

Now let $E\in{\mathcal E}_\lambda$. Taking $K=\bar E$ and noting that $E\subset K$ and $\lambda(K)=\lambda(E)$, it follows from the first calculation that
$\limsup_{t\to\infty} \sup_\gamma \lambda_t^\gamma(E)\le \lambda(E)$.
Similarly, taking $G=\operatorname{Int} E$ it follows from the second calculation that
$\liminf_{t\to\infty} \inf_\gamma \lambda_t^\gamma(E)\ge \lambda(E)$.
This completes the proof that (a) implies (b).

Next, we show that (b) implies (a). Let $g\in C_c({\mathbb R}^m)$, and set $M=\sup|g|$. 
There exists $R>0$ such that $\operatorname{supp} g\subset B_R(0)$.
Note that $\lambda(\partial B_r(0))>0$ for at most countably many $r\in [R, R+1]$. Hence there exists $r>0$ such that $D=\overline{B_r(0)}$ contains $\operatorname{supp} g$ and $D\in{\mathcal E}_\lambda$.

Given ${\epsilon}>0$, choose $c_0<c_1<\dots<c_k$ such that $[-M,M]\subset(c_0,c_k)$ and $c_i-c_{i-1}<{\epsilon}$ for $i=1,\dots,k$.
Since $\lambda(g=c)>0$ for at most countably many values of $c$, we can arrange that $\lambda(g=c_i)=0$ for $i=1,\dots,k$.

Let $E_i=g^{-1}(c_{i-1},c_i)$. Then $\partial E_i\subset \{g=c_{i-1}\}\cup\{g=c_i\}$ so $\lambda(\partial E_i)=0$.
Hence $E_i\cap D\in{\mathcal E}_\lambda$.
(Note that one of the $E_i$ contains ${\mathbb R}^m\setminus\operatorname{supp} g$ which is unbounded, hence the need to intersect with the bounded set $D$.)
By (b), there exists $T>0$ such that
\begin{equation} \label{eq:D1}
\Big|\sum_{i=1}^k c_i\lambda_t^\gamma(E_i\cap D)-
\sum_{i=1}^k c_i\lambda(E_i\cap D)\Big|\le {\epsilon} ,
\end{equation}
for all $t>T$, $\gamma$.

By definition of $E_i$, 
\[
c_{i-1}\lambda_t^\gamma(E_i\cap D)\le 
\int_{E_i\cap D}g\,d\lambda_t^\gamma
\le c_i\lambda_t^\gamma(E_i\cap D),
\]
for all $t$, $\gamma$, $i$. Since $\int_{{\mathbb R}^m}g\,d\lambda_t^\gamma=\int_Dg\,d\lambda_t^\gamma$, it follows that
\begin{equation} \label{eq:D2}
\Big|\int_{{\mathbb R}^m} g\,d\lambda_t^\gamma-\sum_{i=1}^k c_i\lambda_t^\gamma(E_i\cap D)\Big|<{\epsilon} \lambda_t^\gamma( D),
\end{equation}
for all $t$, $\gamma$.
Similarly,
\begin{equation} \label{eq:D3}
\Big|\int_{{\mathbb R}^m} g\,d\lambda-\sum_{i=1}^k c_i\lambda(E_i\cap D)\Big|<{\epsilon} \lambda(D).
\end{equation}
Since $D\in{\mathcal E}_\lambda$, by (b) we can increase $T$ if need be so that $\lambda_t^{\gamma}(D)\le \lambda(D)+1$
for all $t>T$, $\gamma$.
This together with~\eqref{eq:D1},~\eqref{eq:D2},~\eqref{eq:D3}, yields that
\[
\sup_\gamma\Big|\int_{{\mathbb R}^m} g\,d\lambda_t^\gamma-\int_{{\mathbb R}^m} g\,d\lambda\Big|
<2{\epsilon}(1+\lambda(D)),
\]
for all $t>T$.
Since ${\epsilon}>0$ is arbitrary, this completes the proof that (b) implies (a).

$c(\mu(A)-{\epsilon})\le \liminf_{n\to\infty,\,w\to z} q_{n,w}(A)
\le \limsup_{n\to\infty,\,w\to z} q_{n,w}(A)
\le c(\mu(A)+{\epsilon})$.
Since ${\epsilon}>0$ is arbitrary,
$\lim_{n\to\infty,\,w\to z}q_{n,w}(A)=c\mu(A)$ and the result follows.

<!-- thm-8: body_md -->
\label{thm:LLTtwo}
Let $(F,Y,\mu)$ be a mixing two-sided Gibbs-Markov map with one-sided quotient
${\bar{\pi}}:Y\to{\bar Y}$, and
let ${\mathcal A}$ denote the collection of measurable subsets $A\subset Y$ with $\mu(\partial A)=0$.

Assume that $\Psi=(u,v):Y\to{\mathbb R}^{m_1}\times{\mathbb Z}^{m_2}$ is aperiodic, nondegenerate, regularly varying with index $-\alpha$,  $\alpha\in(1,2]$.
Suppose further that $v$ is locally constant and that $u$ admits a decomposition of the form
\begin{equation} \label{eq:chi0}
u={\bar u}\circ\bar\pi+\chi\circ F-\chi,
\end{equation}
where ${\bar u}:{\bar Y}\to{\mathbb R}^{m_1}$ satisfies~\eqref{eq:Psireg0} and
$\chi|_A:A\to{\mathbb R}^{m_1}$ is uniformly continuous
for each $A\in{\mathcal A}$.

Then
\begin{align*}
\lim_{n\to\infty} \sup_{\genfrac{}{}{0pt}{}{z\in{\mathbb R}^m}{|z|\le L}}
\Big|a_n^m\mu\big(y\in A\cap  F^{-n}B:\, u_n(y) \in z_1a_n +K, &
\, {\hat v}^{(n)}(y)=[z_2a_n]   \big)
\\ & - g_{X_\alpha}(z)\mu(A)\mu(B)|K|\Big|=0,
\end{align*}
for all $A,B\in{\mathcal A}$,
$K\subset{\mathbb R}^{m_1}$ bounded with $|\partial K|=0$, and $L>0$.

<!-- proof-28: body_md -->
Fix $\gamma\in(0,1)$.
Recall that a test function $\zeta:Y\to{\mathbb R}$ is dynamically H\"older if $\zeta$ is bounded and there exist constants $C>0$, $\gamma\in(0,1)$ such that $|\zeta(F^ny)-\zeta(F^ny')|\le C(\gamma^{s(y,y')}+\gamma^{s(y,y')-n})$ for all
$y,y'\in Y$ and $n\ge0$.

First, we prove the MLLT with $1_A$ and $1_B$ replaced by dynamically H\"older test functions
$\zeta_1,\zeta_2:Y\to{\mathbb R}$.
Define
\[
\zeta_{r,k}:Y\to{\mathbb R}, \qquad \zeta_{r,k}(y)=\inf\{\zeta_r(F^ky'):s(y,y')\ge 2k\},
\]
for $r=1,2$ and $k\ge1$.
Then $\zeta_{r,k}={\bar{\zeta}}_{r,k}\circ {\bar{\pi}}$ where
${\bar{\zeta}}_{r,k}\in L^\infty({\bar Y})$ with
$|{\bar{\zeta}}_{r,k}|_\infty\le |\zeta_r|_\infty$. It is standard that
$|\zeta_r\circ F^k-\zeta_{r,k}|_\infty\ll \gamma^k$. Moreover, we can choose $\gamma$ so that
$\sup_k\|R^k{\bar{\zeta}}_{r,k}\|<\infty$.

Let ${\tilde u}_n={\bar u}_n\circ{\bar{\pi}}$. Since $v$ is locally constant, there exists ${\bar v}^{(n)}$ such that
${\hat v}^{(n)}={\bar v}^{(n)}\circ{\bar{\pi}}$. Then 
\begin{align*}
\int_Y & \zeta_1\;\zeta_2\circ F^n \;1_{\{{\tilde u}_n\in z_1a_n+K,\,{\hat v}^{(n)}=[z_2a_n]\}}\,d\mu
\\ & =
\int_Y \zeta_1\circ F^k\;\zeta_2\circ F^{n+k} \,1_{\{{\tilde u}_n\in z_1a_n+K,\,{\hat v}^{(n)}=[z_2a_n]\}}\circ F^k\,d\mu
 = J_1+J_2+J_3,
\end{align*}
where
\begin{align*}
J_1 & = 
\int_Y (\zeta_1\circ F^k-\zeta_{1,k})\;\zeta_2\circ F^{n+k} 
\,1_{\{{\tilde u}_n\in z_1a_n+K,\,{\hat v}^{(n)}=[z_2a_n]\}}\circ F^k\,d\mu
\ll \gamma^k, \\
J_2 & = 
\int_Y \zeta_{1,k}\;(\zeta_2\circ F^k-\zeta_{2,k})\circ F^n 
\,1_{\{{\tilde u}_n\in z_1a_n+K,\,{\hat v}^{(n)}=[z_2a_n]\}}\circ F^k\,d\mu
\ll \gamma^k, \\
J_3 & = 
\int_Y \zeta_{1,k}\;\zeta_{2,k}\circ F^n 
\,1_{\{{\tilde u}_n\in z_1a_n+K,\,{\hat v}^{(n)}=[z_2a_n]\}}\circ F^k\,d\mu
 \\ & =
\int_{\bar Y} {\bar{\zeta}}_{1,k}\;{\bar{\zeta}}_{2,k}\circ {\bar F}^n 
\,1_{\{{\bar u}_n\in z_1a_n+K,\,{\bar v}^{(n)}=[z_2a_n]\}}\circ {\bar F}^k\,d{\bar{\mu}}
 \\ & =
\int_{\bar Y} \big(R^k{\bar{\zeta}}_{1,k}\big)\;{\bar{\zeta}}_{2,k}\circ {\bar F}^{n-k} 
\,1_{\{{\bar u}_n\in z_1a_n+K,\,{\bar v}^{(n)}=[z_2a_n]\}}\,d{\bar{\mu}}.
\end{align*}
We take $k=q\log n$ where $q$ is sufficiently large that $\lim_{n\to\infty}a_n^m\gamma^{q\log n}=0$.
Then
\[
a_n^m\int_Y \zeta_1\;\zeta_2\circ F^n \;1_{\{{\tilde u}_n\in z_1a_n+K,\,{\hat v}^{(n)}=[z_2a_n]\}}\,d\mu
= \nu_{n,z}(K)+o(1)
\]
as $n\to\infty$ uniformly in $z$, where
\[
\nu_{n,z}(K)=a_n^m\int_{\bar Y} \big(R^k{\bar{\zeta}}_{1,k}\big)\;{\bar{\zeta}}_{2,k}\circ {\bar F}^{n-k} 
\,1_{\{{\bar u}_n\in z_1a_n+K,\,{\bar v}^{(n)}=[z_2a_n]\}}\,d{\bar{\mu}}.
\]

Let $D={\mathbb R}^{m_1}\times[-\pi,\pi]^{m_2}\times{\bar Y}$ and $q=n-k$.
Proceeding as in the proof of Theorem~\ref{thm:LLT},
\begin{align*}
(2\pi)^{m_2} & \int_{{\mathbb R}^{m_1}}h \,d\nu_{n,w}  \\ & =
                                  
                                   
                                                                        
                                    
          
a_n^m\int_De^{-is_1\cdot w_1a_n}
e^{-is_2\cdot([w_2a_n]+c_n)}\hat h(s_1)
\big(R^k{\bar{\zeta}}_{1,k}\big)\;{\bar{\zeta}}_{2,k}\circ {\bar F}^q e^{is\cdot {\overline{\Psi}}_n}
\,d{\bar{\mu}}\,ds
 \\ &  =
a_n^m\int_{D}e^{-is_1\cdot w_1a_n}e^{-is_2\cdot([w_2a_n]+c_n)}\hat h(s_1)
\big(R^k{\bar{\zeta}}_{1,k}\big)\;\big({\bar{\zeta}}_{2,k}e^{is\cdot{\overline{\Psi}}_k}\big)\circ {\bar F}^q e^{is\cdot {\overline{\Psi}}_q}\,
d{\bar{\mu}}\,ds,
\end{align*}
where ${\bar{\Psi}}=\big({\bar u},v-\int_Y v\,d\mu\big)$
and $c_n=\big[n\int_Y v\,d\mu\big]-n\int_Y v\,d\mu$.
Hence, by Lemma~\ref{lem:LLT},
\begin{align*}
\int_{{\mathbb R}^m}h\,d\nu_{n,w} & =
 I_1 + O(\|R^k{\bar{\zeta}}_{1,k} \|\,{\textstyle\sup_s}|{\bar{\zeta}}_{2,k}e^{is\cdot{\overline{\Psi}}_k}  |_1 a_n^m a_q^{-(m+1)})=I_1+O(a_n^{-1})
\end{align*}
where
\begin{align*}
I_1  
 & = a_n^m {\bar{\mu}}\big(R^k{\bar{\zeta}}_{1,k}\big)\,
\int_{|s|< {\epsilon} }
e^{-is_1\cdot w_1a_n}e^{-is_2\cdot([w_2a_n]+c_n)}\hat h(s) 
{\bar{\mu}}\big({\bar{\zeta}}_{2,k}e^{is\cdot{\overline{\Psi}}_k}\big)\,
\lambda(s)^q ds
 \\ & = {\bar{\mu}}\big(R^k{\bar{\zeta}}_{1,k}\big)\,
\int_{|s|< a_n {\epsilon} }
e^{-is_1\cdot w_1}e^{-ia_n^{-1}s_2\cdot([w_2a_n]+c_n)}\hat h(a_n^{-1}s) 
{\bar{\mu}}\big({\bar{\zeta}}_{2,k}e^{ia_n^{-1}s\cdot{\overline{\Psi}}_k}\big)\,
\lambda(a_n^{-1}s)^q ds.
\end{align*}
But
\begin{align*}
{\bar{\mu}}\big(R^k{\bar{\zeta}}_{1,k}\big) & =
\mu\big( \zeta_{1,k}\big)=\mu\big( \zeta_1\big)+O(\gamma^k), \\
{\bar{\mu}}\big( {\bar{\zeta}}_{2,k}e^{is\cdot{\overline{\Psi}}_k}\big) & =
\mu\big( \zeta_{2,k}e^{is\cdot{\widetilde \Psi}_k}\big)=
\mu\big( \zeta_2\circ F^k e^{is\cdot{\widetilde \Psi}_k}\big)+O(\gamma^k).
\end{align*}
Hence
\begin{align*}
\lim_{n\to\infty,\,w\to z}
\int_{{\mathbb R}^m}h\,d\nu_{n,w}  & =
\mu(\zeta_1)\,\mu(\zeta_2)\,
\int_{{\mathbb R}^m}\hat h(0)e^{-s\cdot x}{\mathbb E}(e^{is\cdot X_\alpha})\,ds
\\ & =
\mu(\zeta_1)\,\mu(\zeta_2)\,
\int_{{\mathbb R}^m}h(x)\,dx\,g_{X_\alpha}(z).
\end{align*}
As in the proof of Theorem~\ref{thm:LLT}, it follows from~\cite[Theorem~10.7]{Breiman} that
 \[
\lim_{n\to\infty} \sup_{\genfrac{}{}{0pt}{}{z\in{\mathbb R}^m}{|z|\le L}}
\Big|a_n^m\int_Y \zeta_1\;\zeta_2\circ F^n \; 
1_{\{{\tilde u}_n\in z_1a_n+K,\,{\hat v}^{(n)}=[z_2a_n]\}}\,d\mu
- g_{X_\alpha}(z)\mu(\zeta_1)\mu(\zeta_2) |K|\Big|=0.
\]

Next, given a measurable subset $A\subset{\mathcal A}$ with
$\mu(\partial A)= 0$, we approximate $A$ and $Y\setminus A$ from within as in the proof of Proposition~\ref{prop:A}. In place of cylinders $\beta_n$, we use sets of the form $\beta_n(y)=\{F^ny':s(y,y')>2n\}$. Similarly for $B$. In this way, we obtain that
\begin{align*}
\lim_{n\to\infty} \sup_{\genfrac{}{}{0pt}{}{z\in{\mathbb R}^m}{|z|\le L}}
\Big|a_n^m\mu\big(y\in A\cap  F^{-n}B:\, {\tilde u}_n(y) & \in z_1a_n +K,
\, {\hat v}^{(n)}(y)=[z_2a_n]   \big)
\\ & - g_{X_\alpha}(z)\mu(A)\mu(B)|K|\Big|=0
\end{align*}
for $A,B,K$ as in the statement of the Theorem.


Finally, we use the same argument as in Subsection~\ref{sec:pf2} to pass from
${\tilde u}_n$ to
$u_n={\tilde u}_n+\chi\circ F^n-\chi$.
By assumption, $\chi$ restricts to a uniformly continuous function on $A$ and $B$. Hence
a simpler version of the argument in Proposition~\ref{prop:N4} shows that
the functions $H_1=-\chi$ and $H_2=\chi$
satisfy condition~\ref{itm:H}.
We can now apply Lemma~\ref{lem:vague} to deduce the result.
