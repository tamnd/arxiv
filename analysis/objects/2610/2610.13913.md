<!-- theorem-1: body_md -->
\label{thm_no_rigidity}
    For every $n\geqslant 2$, there exists $S\subseteq{\mathbb{H}}^n$ satisfying the following properties.     
    \begin{enumerate}
        \item $S$ is a smooth, connected, embedded, two-sided, complete hypersurface without boundary;
        \item $S$ is non-characteristic;
        \item $S$ is the boundary of a global perimeter minimizer. In particular, $S$ is minimal, stable and satisfies the area growth conditions \eqref{updesubintro} for every $p\in S$.
        \item $S$ is an entire, smooth intrinsic graph along $X_1$;
        \item $S$ is not a vertical hyperplane;
        \item $S$ does not verify any of \eqref{eq:P1}, \eqref{eq:P2}, \eqref{eq:P3}.
    \end{enumerate}

<!-- theorem-2: body_md -->
\label{thm_main_1}
For every $n\geqslant 2$, there exists $u\in C^\infty({\mathbb{R}}^n)$ such that
\begin{equation}\label{eq:target}
 \operatorname{div}\left(\frac{Du}{\sqrt{|Du|^2+4|x|^2}}\right)=0,
 \qquad \frac{\partial u}{\partial x_1}\geqslant 1\qquad\text{on }{\mathbb{R}}^n.
\end{equation}
Moreover, $u$ is odd in $x_1$ and invariant under orthogonal transformations of $x_2,\ldots,x_n$.

<!-- proposition-1: body_md -->
\label{propcriteriointrinsic}
Let $G\in C^\infty({\mathbb{H}}^n)$, and set 
$S=\{G=0\}$. If $X_1G\geqslant 1$ on ${\mathbb{H}}^n$, then $S$ is an entire, smooth intrinsic graph along $X_1$.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
The map $(w,s)\mapsto w\cdot\exp(sX_1)$ is a smooth diffeomorphism
from $\mathbb{W}\times\mathbb{R}$ onto ${\mathbb{H}}^n$. By assumption, recalling \cite[Lemma 2.12.1]{MR746308} and for every
$w\in\mathbb{W}$,
\[
    \frac{d}{ds}G\bigl(w\cdot\exp(sX_1)\bigr)
    =X_1G\bigl(w\cdot\exp(sX_1)\bigr)\geqslant 1.
\]
Therefore, $s\mapsto G(w\cdot\exp(sX_1))$ is strictly increasing
and tends to $\pm\infty$ as $s\to\pm\infty$. Hence it vanishes at
a unique $\varphi(w)\in\mathbb{R}$. To prove smoothness, fix $w_0\in\mathbb{W}$. The map
$(w,s)\mapsto G(w\cdot\exp(sX_1))$ is smooth, vanishes at
$(w_0,\varphi(w_0))$, and has nonzero derivative with respect to
$s$ at that point. Therefore, the implicit function theorem
provides an open neighborhood $U_0\subseteq\mathbb{W}$ of $w_0$
and a function $\varphi_0\in C^\infty(U_0)$ such that
$\varphi_0(w_0)=\varphi(w_0)$ and
\[
    G\bigl(w\cdot\exp(\varphi_0(w)X_1)\bigr)=0
    \qquad\text{for every }w\in U_0.
\]
For every $w\in U_0$, both $\varphi_0(w)$ and $\varphi(w)$
are zeros of $s\mapsto G(w\cdot\exp(sX_1))$. By the uniqueness
proved above, $\varphi_0=\varphi$ on $U_0$. Therefore
$\varphi\in C^\infty(\mathbb{W})$, and the thesis follows.

<!-- proposition-2: body_md -->
\label{prop_comparison_principle}
   Let $\Omega\subseteq{\mathbb{R}}^n$ be a bounded domain. Let $u,v\in C^2(\Omega)\cap C(\overline\Omega)$. Assume $\mathcal Q^{\varepsilon,\sigma} u\geqslant0$, $\mathcal Q^{\varepsilon,\sigma} v\leqslant 0$ on $\Omega$. If $u\leqslant v$ on $\partial\Omega$, then $u\leqslant v$ on $\Omega$. In particular, if $\mathcal Q^{\varepsilon,\sigma} u,\mathcal Q^{\varepsilon,\sigma} v=0$ on $\Omega$ and $u=v$ on $\partial\Omega$, then $u= v$ on $\Omega$. The same conclusions hold when $\mathcal Q^{\varepsilon,\sigma}$ is replaced by $\mathcal N^{\varepsilon,\sigma}$.
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
Since $u\in C^1(\Omega)$, $\mathcal Q^{\varepsilon,\sigma}$ and $\mathcal N^{\varepsilon,\sigma}$ are locally uniformly elliptic with respect to $u$, in the sense of \cite[Chapter 10]{MR1814364}. The thesis then follows by \cite[Theorem 10.1]{MR1814364}.
<!-- end proof-2 -->
<!-- proposition-3: body_md -->
\label{propositiondiriboundcondex}
Let $R,K>0$ and $\varepsilon\in(0,1]$. The problem
\begin{equation}\label{eq_diri_ball_eps}
 \mathcal Q^\varepsilon u=0\quad\text{on }B_R,
 \qquad u=Kx_1\quad\text{on }\partial B_R
\end{equation}
has a unique solution $u\in C^\infty(\overline B_R)$. Moreover, $u$ is odd in $x_1$, and invariant under orthogonal transformations of $x_2,\ldots,x_n$.
<!-- end proposition-3 -->
<!-- proof-3: body_md -->
First, we show existence. We rely on the well-known Leray-Schauder fixed point approach. To this aim, 
    Fix any $\alpha\in(0,1)$. By \cite[Theorem 13.8]{MR1814364}, \eqref{eq_diri_ball_eps} is solvable in $C^{2,\alpha}(\overline B_R)$ provided that solutions to 
    \begin{equation}\label{eq_diri_ball_eps_sigma}
 \mathcal N^{\varepsilon,\sigma} u=0\quad\text{on }B_R,
 \qquad u=\sigma Kx_1\quad\text{on }\partial B_R
\end{equation}
are bounded in $C^1(\overline B_R)$ uniformly in $u$ and $\sigma\in[0,1]$. Fix any such solution. When $\sigma=0$, then $u\equiv 0$ solves \eqref{eq_diri_ball_eps_sigma}, and it is the unique solution by \Cref{prop_comparison_principle}. Therefore, we assume that $\sigma> 0$. 

\smallskip
\noindent \textbf{Step 1.} We first show that $u\in C^\infty(\overline B_R)$. Indeed, setting $\mathcal A^\varepsilon(x)=A^\varepsilon(x,Du(x))$, then $u$ satisfies 
\begin{equation}\label{linear_elliptic_forreg}
\sum_{i,j=1}^n\mathcal A^\varepsilon(x)_{ij}D^2 u_{ij}-4\sigma \langle Du,x\rangle=0\qquad \text{on $B_R$.}
\end{equation}
The latter, is a second-order \emph{linear} equation without zero-order terms. Since $u\in C^{2,\alpha}(\overline B_R)$, then \eqref{eigenvaluesofaeps} implies that \eqref{linear_elliptic_forreg} is uniformly elliptic, and that $\mathcal A^\varepsilon_{ij}\in C^{1,\alpha}(\overline B_R)$. Therefore, \cite[Theorem 6.19]{MR1814364} grants $u\in C^{3,\alpha}(\overline B_R)$. Bootstrapping the above argument, $u\in C^\infty(\overline B_R).$

\smallskip
\noindent \textbf{Step 2.} Next, we derive uniform estimates in $C^1(\overline B_R)$. First, notice that the constant functions $-KR$ and $KR$ are solutions to the equation. Moreover, $-KR\leqslant u\leqslant KR$ on $\partial B_R.$ Then, \Cref{prop_comparison_principle} yields $-KR\leqslant u\leqslant KR$ on $\overline B_R$, i.e. $|u|\leqslant KR$ on $\overline B_R$. 

\smallskip
\noindent \textbf{Step 3.} In order to achieve boundary gradient estimates, we exhibit an upper and a lower barrier realizing the boundary datum. Accordingly, set $b_\pm\coloneqq \sigma Kx_1\pm(R^2-|x|^2)$. Clearly, $b_+=b_-=u$ on $\partial B_R$. We show that $\mathcal N^{\varepsilon,\sigma} b_+\leqslant 0$. Indeed, since
\begin{equation*}
    Db_+=(\sigma K-2x_1,-2x_2,\ldots,-2x_n),\quad Db_-=(\sigma K+2x_1,2x_2,\ldots,2x_n),\quad D^2 b_+=-2I_n,\quad D^2 b_-= 2I_n,
\end{equation*}
then 
\begin{equation*}
    \begin{split}
        \mathcal N^{\varepsilon,\sigma} b_+\overset{\eqref{definizione_matrici_eqregeps}}&{=}-2n\big(|Db_+|^2+4|x|^2+\varepsilon^2\big)+2|Db_+|^2-4\sigma\langle D b_+,x\rangle \\
        &=-2(n-1)|Db_+|^2-4\sigma\langle D b_+,x\rangle-8n|x|^2-2n\varepsilon^2\\
        &=-2(n-1)\left|Db_++\frac{\sigma x}{n-1}\right|^2+\left(\frac{2\sigma^2}{n-1}-8n\right)|x|^2-2n\varepsilon^2\\
        &\leqslant 0.
    \end{split}
\end{equation*}
Similarly,
\begin{equation*}
    \begin{split}
         \mathcal N^{\varepsilon,\sigma} b_-\overset{\eqref{definizione_matrici_eqregeps}}&{=}2n\big(|Db_-|^2+4|x|^2+\varepsilon^2\big)-2|Db_-|^2-4\sigma\langle D b_-,x\rangle \\
        &=2(n-1)|Db_-|^2-4\sigma\langle D b_-,x\rangle+8n|x|^2+2n\varepsilon^2\\
        &=2(n-1)\left|Db_--\frac{\sigma x}{n-1}\right|^2+\left(8n-\frac{2\sigma^2}{n-1}\right)|x|^2+2n\varepsilon^2\\
        &\geqslant 0.
    \end{split}
\end{equation*}
Therefore, \Cref{prop_comparison_principle} yields $b_- \leqslant u\leqslant b_+$ on $\overline B_R$. Since both barriers coincide with $u$ on $\partial B_R$ and $|D b_\pm|\leqslant K+2R$ on $\overline B_R$, we conclude that $|D u|\leqslant K+2R$ on $\partial B_R$. 

\smallskip
\noindent \textbf{Step 4.} 
To conclude, we show that 
\begin{equation}\label{quasi_max_principle}
    \max_{\overline B_R} |Du|^2\leqslant \max\left\{ \max_{\partial B_R} |Du|^2,4R^2\right\}.
\end{equation}
By contradiction, assume \eqref{quasi_max_principle} does not hold. Let $\tilde x\in B_R$ be a point where \eqref{quasi_max_principle} fails. Then, we may assume that $\tilde x$ is an interior maximum point of $w$. For notational simplicity, set $w=|Du|^2$. Therefore
\begin{equation}\label{maximum_point_of_w_in_max_principle}
    Dw=2 D^2 u\,Du=2\left(\sum_{i=1}^n u_{1i} u_i,\ldots,\sum_{i=1}^n u_{ni} u_i\right)=0,\qquad D^2 w\leqslant 0\qquad\text{at $\tilde x$.}
\end{equation}
In particular, 
\begin{equation}\label{equazione_in_pt_massimo_w_max_principle_34578}
(w+4|x|^2+\varepsilon^2)\Delta u\overset{\eqref{eq_diri_ball_eps_sigma}}{=}4\sigma \langle Du,x\rangle\qquad\text{at $\tilde x$.}
\end{equation}
By direct computation, at $\tilde x$,
\begin{equation}\label{eq_ausiliarie_28486948375}
    \partial _i\langle Du,x\rangle =u_i+\sum_{j=1}^nu_{ij} x_j,\qquad \partial _i A^\varepsilon_{jk}\coloneqq \partial_i\left(A^\varepsilon_{jk}(x,Du)\right)\overset{\eqref{maximum_point_of_w_in_max_principle}}{=}8x_i\delta_{jk}-u_{ij}u_k-u_{ik}u_j\qquad i=1,\ldots, n.
\end{equation}

Therefore, at $\tilde x$,
\begin{equation*}
    \begin{split}
        0\overset{\eqref{eq_diri_ball_eps_sigma}}&{=}\sum_{i=1}^n\partial_i\left(\mathcal N^{\varepsilon,\sigma} u\right)\,u_i\\
        \overset{\eqref{maximum_point_of_w_in_max_principle},\eqref{eq_ausiliarie_28486948375}}&{=}\sum_{i,j,k=1}^n\partial_iA^\varepsilon_{jk}\,u_{jk}\,u_i+\sum_{i,j,k=1}^nA^\varepsilon_{jk}\,u_{ijk}\,u_i-4\sigma w\\
        \overset{\eqref{maximum_point_of_w_in_max_principle},\eqref{eq_ausiliarie_28486948375}}&{=}8\Delta u\langle Du,x\rangle+\sum_{i,j,k=1}^nA^\varepsilon_{jk}\,u_{ijk}\,u_i-4\sigma w,
    \end{split}
\end{equation*}
whence
\begin{equation}\label{eq_ausiliariaperderivataaaaaa384954}
    \sum_{i,j,k=1}^nA^\varepsilon_{jk}\,u_{ijk}\,u_i=4\sigma w-8\Delta u\langle Du,x\rangle\qquad\text{at $\tilde x$.}
\end{equation}
Noticing that $\operatorname{trace}\left(A^\varepsilon\cdot D^2 w\right)\leqslant 0$ at $\tilde x$ by \eqref{maximum_point_of_w_in_max_principle}, then, at $\tilde x$,  
\begin{equation*}
\begin{split}
      0&\geqslant \frac{1}{2}\sum_{i,j=1}^nA^\varepsilon_{ij}w_{ij}\\
      &=\sum_{i,j,k=1}^nA^\varepsilon_{ij}\partial_i\big(u_{jk} u_k\big)\\
      &=\sum_{i,j,k=1}^nA^\varepsilon_{ij}u_{ijk} u_k+\sum_{i,j,k=1}^nA^\varepsilon_{ij}u_{jk} u_{ik}\\
      \overset{\eqref{eq_ausiliariaperderivataaaaaa384954}}&{=}4\sigma w-8\Delta u\langle Du,x\rangle+\sum_{i,j,k=1}^nA^\varepsilon_{ij}u_{jk} u_{ik}\\
      \overset{\eqref{maximum_point_of_w_in_max_principle}}&{=}4\sigma w-8\Delta u\langle Du,x\rangle+(w+4|x|^2+\varepsilon)^2|D^2 u|^2.
\end{split}   
\end{equation*}
In particular, $4\sigma w-8\Delta u\langle Du,x\rangle\leqslant 0$ at $\tilde x$. On the other hand, at $\tilde x$,
\begin{equation*}
    \begin{split}
        4\sigma w-8\Delta u\langle Du,x\rangle\overset{\eqref{equazione_in_pt_massimo_w_max_principle_34578}}&{=}4\sigma \left(w-\frac{8\langle Du,x\rangle^2}{w+4|x|^2+\varepsilon^2}\right)\\
        &\geqslant 4\sigma w\left(1-\frac{8|x|^2}{w+4|x|^2+\varepsilon^2}\right)\\
        &=4\sigma w\left(\frac{w-4|x|^2+\varepsilon^2}{w+4|x|^2+\varepsilon^2}\right),
    \end{split}

Set $v=x_1u_1-u$. By \eqref{prole1}, \eqref{prole2}, \eqref{prole3} and \eqref{prole4}, 
\begin{equation*}\label{eq:mono-V-full}
\begin{split}
    \mathcal L^\varepsilon v&=x_1\mathcal L^\varepsilon u_1+u_1\mathcal L^\varepsilon x_1+2\sum_{i=1}^nA^\varepsilon_{1i}u_{1 i}-8\langle x, Du\rangle +2(4|x|^2+\varepsilon^2)\,\Delta u\\
    &=x_1\big(4 u_1-8x_1\Delta u\big)+u_1\left(2 u_1\Delta u-2\sum_{i=1}^nu_{1 i}u_i-4x_1\right)+2\sum_{i=1}^nA^\varepsilon_{1i}u_{1 i}-8\langle x, Du\rangle +2(4|x|^2+\varepsilon^2)\,\Delta u\\
    &=-8x_1^2\Delta u+2 u_1^2\Delta u-4u_1\sum_{i=1}^nu_{1 i}u_i+2W_\varepsilon^2u_{11}-8\langle x, Du\rangle +2(4|x|^2+\varepsilon^2)\,\Delta u\\
    &=2W_\varepsilon^2u_{11}-4u_1\sum_{i=1}^nu_{1 i}u_i-8\langle x, Du\rangle +2\big(u_1^2+4r^2+\varepsilon^2\big)\,\Delta u.
\end{split}
\end{equation*}
Exploiting \eqref{eq_diri_ball_eps},
\begin{equation}\label{serveperplapropdoudn3859}
    \begin{split}
        \mathcal L^\varepsilon v&=2W_\varepsilon^2u_{11}-4u_1\sum_{i=1}^nu_{1 i}u_i-2W_\varepsilon^2\,\Delta u+2\sum_{i,j=1}^n u_{ij}u_i u_j +2\big(u_1^2+4r^2+\varepsilon^2\big)\,\Delta u.        .
    \end{split}
\end{equation}
Therefore, when $r> 0$ and recalling \eqref{gradienteinroti394969},
\begin{equation}\label{eq:mono-V}
    \begin{split}
        \mathcal L^\varepsilon v
        &=2\big(u_r^2+4|x|^2+\varepsilon^2\big)u_{11}-8x_1^2\Delta u-2u_r^2\,\Delta u+2\sum_{i,j=2}^n u_{ij}u_i u_j\\
        &=2\big(u_r^2+4|x|^2+\varepsilon^2\big)u_{11}-8x_1^2\Delta u-2u_r^2\,\Delta u+2u_r^2 u_{rr}\\
        \overset{\eqref{laplacianincoordinatesdfrfr}}&{=}2\big(4|x|^2+\varepsilon^2\big)u_{11}-8x_1^2\Delta u-2(n-2)\frac{u_r^3}{r}
    \end{split}
\end{equation}
Notice that $v=0$ when $x_1=0$. Moreover, on $\partial B_R\cap\{x_1> 0\}$, \eqref{eq:mono-boundary} gives $v=dx_1^2R^{-1}\geqslant 0$. Assume by contradiction that \eqref{eq:mono-dilation-conclusion} fails. Then, there exists $\tilde x\in B_R\cap\{x_1> 0\}$ which is a minimum point for $v$ and such that $v(\tilde x)< 0$. Assume first $r(\tilde x)>0$. Then $v_1=v_r=0$ at $\tilde x$, whence $u_{11}=0$ and $x_1 u_{1r}=u_r$ at $\tilde x$.
The combination of two facts with \eqref{eq:mono-meridional} and \eqref{laplacianincoordinatesdfrfr} grants 
\begin{equation}\label{nonricordopiu38495885}
 (u_1^2+4|x|^2+\varepsilon^2)\Delta u
 =4\langle x,Du\rangle+\frac{2u_1u_r^2}{x_1}-(n-2)\frac{u_r^3}{r} \qquad\text{at }\tilde x,
\end{equation}
and moreover, by \eqref{eq:mono-V},
\begin{equation}\label{auxiliare39459694939393993}
    \mathcal L^\varepsilon v=-8x_1^2\Delta u-2(n-2)\frac{u_r^3}{r}\qquad \text{at }\tilde x.
\end{equation}
Therefore, since $\tilde x$ is an interior minimum point of $v$ and $\mathcal L^\varepsilon$ is elliptic, \eqref{nonricordopiu38495885} and \eqref{auxiliare39459694939393993} yield
\begin{equation}
 \begin{split}
 \mathcal L^\varepsilon v=-
 \frac{8x_1^2}{u_1^2+4|x|^2+\varepsilon^2}
 \left(4\langle x,Du\rangle+\frac{2u_1u_r^2}{x_1}\right)-2(n-2)\frac{u_r^3}{r}
 \left(\frac{u_1^2+4r^2+\varepsilon^2}{u_1^2+4|x|^2+\varepsilon^2}\right)<0,
 \end{split}
\end{equation}
where the last inequality follows combining \eqref{eq:mono-transverse-positive} and \eqref{eq:mono-first-positive}. This is a contradiction.
Finally, assume that $r(\tilde x)=0$.
 In this case, recall that $u_i(\tilde x)=0$ for $i=2,\ldots,n$ by invariance along $x_2,\ldots,x_n$. Moreover, again by minimality of $\tilde x$, $u_{11}(\tilde x)=0$. Therefore \eqref{eq_diri_ball_eps} yields 
\begin{equation}\label{segnodfdfdfdf4578}
 (u_1^2+4x_1^2+\varepsilon^2)\Delta u=4x_1u_1> 0\qquad \text{at }\tilde x,
\end{equation}
the last inequality following by \eqref{eq:mono-first-positive}. 
Therefore, \eqref{serveperplapropdoudn3859} and \eqref{segnodfdfdfdf4578} grant  
$$\mathcal L^\varepsilon v=-8x_1^2\Delta u<0,$$
again a contradiction. The thesis follows.
