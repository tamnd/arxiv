<!-- head-1: body_md -->
Throughout the paper, we work over a field $\Bbbk$ of characteristic $0$, unless explicitly stated otherwise.
We do not repeat standard definitions 
$\big($Lie algebras, Leibniz algebras, solvable and nilpotent algebras, etc.$\big)$ 
and refer to the literature, see e.g.\ \cite{k23}.
For a set of vectors $S$ we denote by $\big\langle S\big\rangle$ the $\Bbbk$-linear span of $S$.
For the commutator, associator, anti-commutator, and anti-associator, we use
\[
[x,y]=xy-yx,
\qquad
(x,y,z)=(xy)z-x(yz),
\qquad
x\circ y=xy+yx,
\qquad
\{x,y,z\}=(xy)z+x(yz).
\]
When some identities are verified by symbolic computations, we use the computer algebra package {\rm Albert}.
We always assume that algebras under consideration are nontrivial 
$\big($the multiplication is not identically zero$\big)$.

<!-- definition-1: body_md -->
For a fixed multilinear polynomial 
\begin{center}
    $
\mathfrak f :=\sum\limits_{\sigma \in \mathbb{S}_3} \alpha_\sigma^j \big(x_{\sigma(1)}x_{\sigma(2)}\big)x_{\sigma(3)}+
    \sum\limits_{\sigma \in \mathbb{S}_3} \beta_\sigma^j x_{\sigma(1)}\big(x_{\sigma(2)}x_{\sigma(3)}\big),$
\end{center}   
    we define  that the $\mathfrak M$arseille anti-polynomial 
  $\mathfrak{Ma(f)}$ as follows 
\begin{center}  $
\mathfrak{Ma(f)} :=\sum\limits_{\sigma \in \mathbb{S}_3} \alpha_\sigma^j \big(x_{\sigma(1)}x_{\sigma(2)}\big)x_{\sigma(3)} - 
    \sum\limits_{\sigma \in \mathbb{S}_3} \beta_\sigma^j x_{\sigma(1)}\big(x_{\sigma(2)}x_{\sigma(3)}\big).$ 
\end{center}

<!-- proposition-2: body_md -->
\label{basisAAs}
Let $\mathfrak{Ma}\textrm{-}\mathrm{As}\<X\>$ be the free $\mathfrak{Ma}$-associative algebra, generated by the set $X=\big\{x_i \big\}_{i \in I}.$ 
Then the following set of elements 
\begin{center}
    ${\mathfrak B}=\big\{  \ x_i,\  x_{i}x_{j},\  x_{i} (x_{j}x_{k})  \ \big\}_{i, j, k \in I}$
\end{center}
gives a basis of $\mathfrak{Ma}$-$\mathrm{As}\<X\>.$
<!-- end proposition-2 -->
<!-- proof-1: body_md -->
Let us remember that $\mathfrak{Ma}\textrm{-}\mathrm{As}\<X\>$ is a $3$-step nilpotent algebra, 
hence, $x_{k_1}\ldots x_{k_t}=0$ for $t \ge 4.$
All basis elements are constructed from words of length $1$, $2$, or $3.$
Due to anti-associativity, all words given by 
$\big\{\ x_{i} (x_{j}x_{k}) \  \big\}_{i,   j, k \in I} $ are independent.
Hence, ${\mathfrak B}$ is a basis of  
$\mathfrak{Ma}\textrm{-}\mathrm{As}\<X\>.$

<!-- proposition-4: body_md -->
\label{basisAAs}
Let $\mathfrak{Ma}\textrm{-}S\mathrm{As}\<X\>$ be the free $\mathfrak{Ma}$-shift associative algebra, generated by the set $X=\big\{x_i \big\}_{i \in I}.$ 
Then the following set of elements 
\begin{center}
    ${\mathfrak B}=\big\{  \ x_i,\  x_{i}x_{j},\  x_{i} (x_{j}x_{k}), \  (x_{i} x_{j})(x_{m} x_{k})   \ \big\}_{i, j, k, m \in I; \  j<m}$
\end{center}
gives a basis of $\mathfrak{Ma}\textrm{-}S\mathrm{As}\<X\>.$
<!-- end proposition-4 -->
<!-- proof-2: body_md -->
Let us remember that $\mathfrak{Ma}\textrm{-}S\mathrm{As}\<X\>$ is a $4$-step nilpotent algebra, 
hence, $x_{k_1}\ldots x_{k_t}=0$ for $t \ge 5.$
All basis elements are constructed from words of length $1$, $2$,  $3,$ or $4.$
Due to anti-shift associativity, all words given by 
$\big\{\ x_{i} (x_{j}x_{k}) \  \big\}_{i,   j, k \in I} $ are independent. It is easy to see that 
\begin{longtable}{lclclcl}
$(xy)(zt)$&$=$&$-\  y\big((zt)x\big) $&$=$&$ y \big(t(xz)\big);$ \\ 
$(xy)(zt)$&$=$&$-\ \big(t(xy)\big)z$&$= $&$\big((yt)x\big)z$&$=$&$- \ (xz)(yt).$
\end{longtable}
Hence, ${\mathfrak B}$ is a basis of   $\mathfrak{Ma}\textrm{-}S\mathrm{As}\<X\>.$

<!-- theorem-1: body_md -->
For the varieties of $\mathfrak{Ma}$-alternative and $\mathfrak{Ma}$-associative algebras, one has
\[
\mathfrak{Ma}\textrm{-}\mathrm{Alt} \neq \mathfrak{Ma}\textrm{-}\mathrm{As}_2 .
\]
<!-- end theorem-1 -->
<!-- proof-3: body_md -->
It is immediate that
\[
\mathfrak{Ma}\textrm{-}\mathrm{Alt} \supseteq \mathfrak{Ma}\textrm{-}\mathrm{As}_2 .
\]
To show that the inclusion is strict, we compare the free two-generated algebras
$\mathfrak{Ma}\textrm{-}\mathrm{Alt}\langle a,b\rangle$ and $\mathfrak{Ma}\textrm{-}\mathrm{As}\langle a,b\rangle$.
Using the computer algebra system Albert, we compute bases of these algebras.

For $\mathfrak{Ma}\textrm{-}\mathrm{Alt}\langle a,b\rangle$ we obtain the following basis elements:
\[
a,\ b,\ bb,\ ba,\ ab,\ aa,
\]
\[
(bb)b,\ (bb)a,\ (ba)b,\ (ab)b,\ (ba)a,\ (ab)a,\ (aa)b,\ (aa)a,
\]
\[
\big((ba)b\big)b,\ \big((ab)a\big)b,\ \big((aa)b\big)b,\ \big((aa)a\big)b.
\]
Moreover, starting from degree $5$ the algebra $\mathfrak{Ma}\textrm{-}\mathrm{Alt}\langle a,b\rangle$ is nilpotent.
On the other hand, by Proposition~\ref{basisAAs}, the algebra
$\mathfrak{Ma}\textrm{-}\mathrm{As}\langle a,b\rangle$ is nilpotent of index $4$.

Therefore,
\begin{center}
$\mathfrak{Ma}\textrm{-}\mathrm{Alt}\<a,b\>\ncong \mathfrak{Ma}\textrm{-}\mathrm{As}_2\<a,b\>$
\ and\ 
$\mathfrak{Ma}\textrm{-}\mathrm{Alt} \varsupsetneq \mathfrak{Ma}\textrm{-}\mathrm{As}_2.$
\end{center}

<!-- theorem-2: body_md -->
\label{BasisALeib}
The set $\mathcal{AL}$ is a basis of the algebra $\mathfrak{Ma}$-${\rm Leib\<X\>}$.
<!-- end theorem-2 -->
<!-- proof-5: body_md -->
Firstly, we show that the set $\mathcal{AL}$ spans $\mathfrak{Ma}$-${\rm Leib}\<X\>$. 
Let $m$ be an arbitrary monomial of degree $k$ of the form
\[
m\ =\ \big(\cdots((x_i w_1)w_2)\cdots\big)w_n.
\]
Using the identities \eqref{ALeib2} and \eqref{ALeib3}, we can reduce $m$ whenever $\deg(w_j)\ge 3$. Using the identity \eqref{ALeib1}, we order the generators inside $w_j$. 
Next, we can interchange $w_j$ and $w_{j+1}$ by the defining identity of $\mathfrak{Ma}$-${\rm Leib}\<X\>$, namely,
\[
(aw_j)w_{j+1}=a(w_{j+1}w_{j})+(aw_{j+1})w_{j}.
\]
If both $w_j$ and $w_{j+1}$ are generators and $w_j>w_{j+1}$, then the monomials
$a(w_{j+1}w_{j})$ and $(aw_{j+1})w_{j}$ belong to $\mathcal{AL}$. 
If $\deg(w_{j})=\deg(w_{j+1})=2$, then the identity above reduces to
\[
(aw_j)w_{j+1}=(aw_{j+1})w_{j}.
\]
Thus, the set of remaining monomials coincides with the set $\mathcal{AL}$.

Let us consider the algebra $\mathcal{A}\<X\>$ with a basis $\mathcal{AL}$. 
We define a multiplication on $\mathcal{A}\<X\>$ as follows. 
Let $A_i\in\mathcal{AL}$ be an arbitrary basis monomial of degree $i$.

\begin{itemize}
\item If $A_t\in\mathcal{AL}$ has degree $t\ge 3$, then
\[
A_k\cdot A_t=0
\qquad (k\ge 1).
\]

\item If $t=2$, and
\[
A_k\ =\ \big(\cdots((x_i w_1)w_2)\cdots\big)w_n,
\]
then for any degree-$2$ basis monomial $w_p$ we set
\[
A_k\cdot w_p
=\big(\cdots((((\cdots((x_i w_1)w_2)\cdots)w_{r-1})w_p)w_r)\cdots\big)w_n,
\]
where $w_{r-1}\le w_p\le w_r$.

\item If $t=1$, and
\[
A_k=\big(\cdots(((\cdots((x_i x_{t_1})x_{t_2})\cdots)x_{t_l})w_{l+1})\cdots\big)w_n,
\]
then for any generator $x_p$ we define
\begin{multline*}
A_k\cdot x_p
=\big(\cdots(((\cdots((((\cdots((x_i x_{t_1})x_{t_2})\cdots)x_{t_{r-1}})x_p)x_{t_r})\cdots)x_{t_l})w_{l+1})\cdots\big)w_n \\
+\sum_{s=r}^{l}\big(\cdots((((\cdots(((\cdots((x_i x_{t_1})x_{t_2})\cdots)x_{t_{r-1}})x_{t_r})\cdots)x_{t_l})(x_p x_{t_s}))w_{l+1})\cdots\big)w_n,
\end{multline*}
where $x_{t_{r-1}}\le x_p<x_{t_r}$, and in the final sum there is no generator $x_{t_s}$ among $x_{t_r},\dots,x_{t_l}$.
\end{itemize}
A direct computation shows that the algebra $\mathcal{A}\<X\>$ belongs to $\mathfrak{Ma}\text{-}$${\rm Leib}$, that is, it satisfies the defining identity of $\mathfrak{Ma}\text{-}$${\rm Leib}$. 
It remains to note that
\[
\mathcal{A}\<X\>\cong \mathfrak{Ma}\text{-}{\rm Leib}\<X\>.
\]

<!-- theorem-3: body_md -->
\label{ThmANovBasis}
The basis of $\mathfrak{Ma}\text{-}$${\rm Nov}\<X\>$ coincides with the basis of the free bicommutative algebra ${\rm BiCom}\<X\>$.
<!-- end theorem-3 -->
<!-- proof-10: body_md -->
First, we show that every monomial in $\mathfrak{Ma}\text{-}$${\rm Nov}\<X\>$ can be rewritten as a linear combination of monomials from $\mathcal{AN}$.
We prove this by induction on the degree.

The base cases are $n=3$ and $n=4$. For $n=3$, we have
\[
x_{2}(x_{1}x_{3})=x_{1}(x_{2}x_{3}),\qquad
x_{3}(x_{2}x_{1})=x_{2}(x_{3}x_{1}),\qquad
x_{3}(x_{1}x_{2})=x_{1}(x_{3}x_{2}),
\]
and
\[
(x_{1}x_{2})x_{3}=(x_{1}x_{3})x_{2}+x_{1}(x_{3}x_{2})-x_{1}(x_{2}x_{3}),
\]
\[
(x_{2}x_{1})x_{3}=(x_{2}x_{3})x_{1}+x_{2}(x_{3}x_{1})-x_{1}(x_{2}x_{3}),
\]
\[
(x_{3}x_{1})x_{2}=(x_{3}x_{2})x_{1}+x_{2}(x_{3}x_{1})-x_{1}(x_{3}x_{2}).
\]

We endow $\mathfrak{Ma}\text{-}$${\rm Nov}\<X\>$ with the degree--lexicographical order.
We also write $m\equiv m'$ if $m$ and $m'$ have the same bracketing type and
$m-m'$ is a linear combination of basis monomials strictly smaller than $m$ in this order.
In degree $3$, this yields
\[
(x_{1}x_{2})x_{3}\equiv (x_{1}x_{3})x_{2},\qquad
(x_{2}x_{1})x_{3}\equiv (x_{2}x_{3})x_{1},\qquad
(x_{3}x_{1})x_{2}\equiv (x_{3}x_{2})x_{1}.
\]
As example, the identity \eqref{ANovId3} shows that
\[
(db)\big((ce)a\big)\equiv 0.
\]

For $n=4$, by the left-commutative identity, every monomial of the form
$x_{i_1}\bigl(x_{i_2}(x_{i_3}x_{i_4})\bigr)$ can be rewritten as one of the following monomials:
\[
x_{1}\bigl(x_{2}(x_{3}x_{4})\bigr),\quad
x_{1}\bigl(x_{2}(x_{4}x_{3})\bigr),\quad
x_{1}\bigl(x_{3}(x_{4}x_{2})\bigr),\quad
x_{2}\bigl(x_{3}(x_{4}x_{1})\bigr).
\]

For a monomial of the form $x_{i_1}\bigl((x_{i_2}x_{i_3})x_{i_4}\bigr)$, using the inductive hypothesis and \eqref{ANovId1}, we obtain
\[
x_{i_1}\bigl((x_{i_2}x_{i_3})x_{i_4}\bigr)\equiv
x_{i_1}\bigl((x_{i_2}x_{i_4})x_{i_3}\bigr)
=
x_{i_2}\bigl((x_{i_1}x_{i_4})x_{i_3}\bigr).
\]
In particular, modulo $\equiv$, we can permute the generators within the pairs $(x_{i_1},x_{i_2})$ and $(x_{i_3},x_{i_4})$ in order to achieve the required ordering.

Next, consider a monomial of the form $(x_{i_1}x_{i_2})(x_{i_3}x_{i_4})$.
By the left-commutative identity and the inductive hypothesis, we have
\[
(x_{i_1}x_{i_2})(x_{i_3}x_{i_4})\equiv 0,
\]
that is, there is no basis monomial with the same bracketing type, and this monomial can be expressed as a linear combination of smaller basis monomials.

Similarly, by \eqref{ANovId2} and the previous cases, we have
\[
\bigl(x_{i_1}(x_{i_2}x_{i_3})\bigr)x_{i_4}\equiv 0.
\]

Finally, using the defining identities of $\mathfrak{Ma}\text{-}$${\rm Nov}$ together with the previous reductions, we obtain
\[
\bigl((x_{i_1}x_{i_2})x_{i_3}\bigr)x_{i_4}
\equiv
\bigl((x_{i_1}x_{i_2})x_{i_4}\bigr)x_{i_3}
\equiv
\bigl((x_{i_1}x_{i_4})x_{i_2}\bigr)x_{i_3}.

Let $\mathrm{Var}=\mathcal F(E)/(R)$ be a quadratic presentation, where $\mathcal F(E)$ is the free symmetric operad
generated by a binary $\mathbb S$-module $E$ and $R\subset \mathcal F(E)(3)$ is the $\mathbb S_3$-submodule of
quadratic relations. We use the definition of the quadratic dual operad via orthogonal complement:
\[
\mathrm{Var}^{(!)}=\mathcal F(E^{\vee})/(R^{\perp}),
\qquad
R^{\perp}:=\{\,g\in \mathcal F(E^{\vee})(3)\mid \langle r,g\rangle=0\ \text{for all } r\in R\,\},
\]
where $\langle\cdot,\cdot\rangle$ is the standard nondegenerate $\mathbb S_3$-invariant pairing
between $\mathcal F(E)(3)$ and $\mathcal F(E^{\vee})(3)$.

In arity $3$, we have the direct sum decomposition
\[
\mathcal F(E)(3)=\mathcal F_L(E)(3)\oplus \mathcal F_R(E)(3),
\]
where $\mathcal F_L(E)(3)$ is spanned by the monomials
$\bigl(x_{\sigma(1)}x_{\sigma(2)}\bigr)x_{\sigma(3)}$ and
$\mathcal F_R(E)(3)$ is spanned by the monomials
$x_{\sigma(1)}\bigl(x_{\sigma(2)}x_{\sigma(3)}\bigr)$, for $\sigma\in\mathbb S_3$.
Hence every $f\in \mathcal F(E)(3)$ is uniquely written as $f=f_L+f_R$ with
$f_L\in \mathcal F_L(E)(3)$ and $f_R\in \mathcal F_R(E)(3)$.

By definition,
\[
\mathfrak{Ma}(f):=f_L-f_R.
\]
In particular, $\mathfrak{Ma}$ is $\Bbbk$-linear, $\mathbb S_3$-equivariant, and
\begin{equation}\label{eq:MaProperties}
\mathfrak{Ma}^2=\mathrm{id},\qquad
\mathfrak{Ma}\big|_{\mathcal F_L(E)(3)}=\mathrm{id},\qquad
\mathfrak{Ma}\big|_{\mathcal F_R(E)(3)}=-\,\mathrm{id}.
\end{equation}
Moreover, since $\operatorname{char}\Bbbk\neq 2$, we have the projections
\begin{equation*}
f_L=\frac12\bigl(f+\mathfrak{Ma}(f)\bigr),
\qquad
f_R=\frac12\bigl(f-\mathfrak{Ma}(f)\bigr).
\end{equation*}

We define $\mathfrak{Ma}$ on $\mathcal F(E^{\vee})(3)$ by the same rule:
it acts by $+1$ on the left-nested component and by $-1$ on the right-nested component.
Then for all $f\in\mathcal F(E)(3)$ and $g\in\mathcal F(E^{\vee})(3)$, we have
\begin{equation}\label{eq:MaAdjoint}
\langle \mathfrak{Ma}(f),\,g\rangle=\langle f,\,\mathfrak{Ma}(g)\rangle.
\end{equation}
Indeed, writing $f=f_L+f_R$ and $g=g_L+g_R$, we obtain
\[
\langle \mathfrak{Ma}(f),g\rangle
=\langle f_L-f_R,\,g_L+g_R\rangle
=\langle f_L,g_L\rangle-\langle f_R,g_R\rangle,
\]
because the mixed pairings $\langle f_L,g_R\rangle$ and $\langle f_R,g_L\rangle$ vanish.
On the other hand,
\[
\langle f,\mathfrak{Ma}(g)\rangle
=\langle f_L+f_R,\,g_L-g_R\rangle
=\langle f_L,g_L\rangle-\langle f_R,g_R\rangle,
\]
which proves \eqref{eq:MaAdjoint}.

We claim that
\begin{equation}\label{eq:OrthCommutesMa}
\bigl(\mathfrak{Ma}(R)\bigr)^{\perp}=\mathfrak{Ma}\bigl(R^{\perp}\bigr).
\end{equation}
Let $g\in \big(\mathfrak{Ma}(R)\big)^{\perp}$. Then for every $r\in R$ we have
\[
\langle \mathfrak{Ma}(r),\,g\rangle=0.
\]
Using \eqref{eq:MaAdjoint}, this is equivalent to $\langle r,\,\mathfrak{Ma}(g)\rangle=0$ for all $r\in R$,
hence $\mathfrak{Ma}(g)\in R^{\perp}$. Applying \eqref{eq:MaProperties}, we get
$g=\mathfrak{Ma}\big(\mathfrak{Ma}(g)\big)\in \mathfrak{Ma}(R^{\perp})$.
Thus $\big(\mathfrak{Ma}(R)\big)^{\perp}\subseteq \mathfrak{Ma}(R^{\perp})$.

Conversely, let $g\in \mathfrak{Ma}(R^{\perp})$. Then $g=\mathfrak{Ma}(h)$ for some $h\in R^{\perp}$.
For any $r\in R$ we obtain, using \eqref{eq:MaAdjoint} and \eqref{eq:MaProperties},
\[
\langle \mathfrak{Ma}(r),\,g\rangle
=\langle \mathfrak{Ma}(r),\,\mathfrak{Ma}(h)\rangle
=\langle r,\,h\rangle
=0.
\]
Hence $g\in \big(\mathfrak{Ma}(R)\big)^{\perp}$, and therefore
$\mathfrak{Ma}(R^{\perp})\subseteq \big(\mathfrak{Ma}(R)\big)^{\perp}$.
This proves \eqref{eq:OrthCommutesMa}.

Let ${\rm A}$ be an algebra, then 
$\mathfrak{aff}^+({\rm A})$ is Jacobi--Jordan if and only if
${\rm A}$ is $\mathcal{M}a$-Novikov.
<!-- end proposition-15 -->

Let $\mathfrak{aff}^+({\rm A})$ be Jacobi--Jordan and  $a_1, a_2, a_3 \in {\rm A},$ then 
    \begin{longtable}{lll}
$(a_1t^{n_1} \circ a_2 t^{n_2}) \circ a_3 t^{n_3}$&$=$&$
n_1(n_1+n_2-1)\ (a_1a_2)a_3+n_1n_3 \ a_3(a_1a_2)+$\\
\multicolumn{3}{r}{$n_2(n_1+n_2-1)\ (a_2a_1)a_3
+n_2n_3 \ a_3(a_2a_1);$}\\

$(a_2t^{n_2} \circ a_3 t^{n_3}) \circ a_1 t^{n_1}$&$=$&$
n_2(n_2+n_3-1)\ (a_2a_3)a_1+n_2n_1 \ a_1(a_2a_3)+$\\
\multicolumn{3}{r}{$n_3(n_2+n_3-1)\ (a_3a_2)a_1
+n_3n_1 \ a_1(a_3a_2);$}\\


$(a_3t^{n_3} \circ a_1 t^{n_1}) \circ a_2 t^{n_2}$&$=$&$
n_3(n_3+n_1-1)\ (a_3a_1)a_2+n_3n_2 \ a_2(a_3a_1)+$\\
\multicolumn{3}{r}{$n_1(n_3+n_1-1)\ (a_1a_3)a_2
+n_1n_2 \ a_2(a_1a_3).$}\\
    \end{longtable}
Summarizing the above relations and using the Jacobi--Jordan identity
\begin{enumerate}
    \item for $n_2=n_3=0,$ we have $(a_1a_2)a_3 = - (a_1a_2)a_3.$

    \item for $n_1=0,$ we have 
  
$n_2(n_2-1) \ (a_2a_1)a_3 + 
n_2n_3 \ a_3(a_2a_1) +
n_2(n_2+n_3-1) \ (a_2a_3)a_1 +
 $
\begin{flushright}
$n_3(n_2+n_3-1) \ (a_3a_2)a_1 + 
n_3(n_3-1) \ (a_3a_1)a_2 +
    n_3n_2 \ a_2(a_3a_1)=0.$
 \end{flushright}
   
The last is equivalent to 
$\big\{a_3,a_2,a_1 \big\} = - \big\{a_2,a_3,a_1\big\}.$\end{enumerate}
Hence, ${\rm A}$ is $\mathcal{M}a$-Novikov.
On the other side, it is easy to see that for each
$\mathcal{M}a$-Novikov, its symmetric affinization gives a Jacobi--Jordan algebra.
