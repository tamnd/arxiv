<!-- theorem-1: body_md -->
\label{thm:main-reliability}
Under equation~\ref{eq:stability_assumption}, assume $0\le\pi_m\le1$, $\sum_m\pi_m=1$, and $\varepsilon>0$. For masses $s_m=\int_Q\pi_m\,d\mu_Q$ and $s_m^\Gamma=\int_\Gamma\pi_m\,d\mu_\Gamma$, define continuum indicators by the corresponding $\varepsilon$-regularized weighted averages of $|r|^2$ and $|b|^2$, where $r=Lu_\Theta-f$ and $b=Bu_\Theta-g$. Then
\begin{equation}
\|u_\Theta-u\|_X^2\le C_{\mathrm{stab}}^2
\left[\sum_m(s_m+\varepsilon)\eta_m^{\mathrm{res}}
+\kappa_\Gamma\sum_m(s_m^\Gamma+\varepsilon)\eta_m^{\mathrm{bdry}}\right].
\label{eq:recap-reliability}
\end{equation}
The independent-sample certificate, including its finite-sample correction terms and assumptions, is stated and proved in Theorem~\ref{thm:recap-reliability} in Appendix~\ref{sec:further_theory}.

<!-- theorem-2: body_md -->
\label{thm:main-quiet-child}
Let $L=\partial_t+\beta\cdot\nabla$ and use equation~\ref{eq:mixture_assumption}. Retain the parent unchanged and translate the child so that $\ell_c=\ell_p$. Assume the gate is constant in $z$, $0\le\gamma_c\le\epsilon_g$, and $0\le\phi_c\le1$, $D\ge d_0>0$ on $Q\cup\Gamma$, with $\|L\phi_c\|_{L^\infty(Q)}\le K_\phi$ and $\|LD\|_{L^\infty(Q)}\le K_D$. For $J(v)=\|Lv-f\|_{L^2(Q)}^2+\kappa_\Gamma\|v-g\|_{L^2(\Gamma)}^2$, put $h=\ell_p-u^-$ and $K_\alpha=K_\phi/d_0+K_D/d_0^2$. Define the finite constant
\begin{equation}
C_{\mathrm{split}}=\left[
\left(K_\alpha\|h\|_{L^2(Q)}+d_0^{-1}\|Lh\|_{L^2(Q)}\right)^2
+\kappa_\Gamma d_0^{-2}\|h\|_{L^2(\Gamma)}^2
\right]^{1/2}.
\label{eq:split-constant}
\end{equation}
The post-split predictor satisfies
\begin{equation}
\begin{aligned}
\sqrt{J(u^+)}&\le\sqrt{J(u^-)}+\epsilon_gC_{\mathrm{split}},\\
J(u^+)-J(u^-)&\le2\epsilon_gC_{\mathrm{split}}\sqrt{J(u^-)}
+\epsilon_g^2C_{\mathrm{split}}^2.
\end{aligned}
\label{eq:sqrt_loss_shock}
\end{equation}
If admissible child gates can approach zero, the refined class contains the old class in its closure, so its optimal achievable loss cannot increase.

<!-- theorem-3: body_md -->
\label{thm:recap-reliability}
Under equation~\ref{app:eq:stability_assumption} and the partition-of-unity condition,
\begin{equation}
\|u_\Theta-u\|_X^2
\le C_{\rm stab}^2\left[
\sum_{m=1}^M(s_m+\varepsilon)\eta_m^{\rm res}
+\kappa_\Gamma\sum_{m=1}^M(s_m^\Gamma+\varepsilon)\eta_m^{\rm bdry}
\right].
\label{app:eq:recap-reliability}
\end{equation}

Moreover, freeze $u_\Theta$ and draw independent certification samples
$Z_i\sim\mu_Q$, $i=1,\ldots,N_c$, and
$Y_j\sim\mu_\Gamma$, $j=1,\ldots,N_\Gamma$. For $N_c,N_\Gamma\ge1$, let the empirical masses and indicators be defined by replacing
the integrals above with sample sums, and set
\begin{equation}
\widehat{\mathcal E}_{\mathrm{ReCAP}}^2
=\frac1{N_c}\sum_m(\widehat s_m+\varepsilon)\widehat\eta_m^{\rm res}
+\frac{\kappa_\Gamma}{N_\Gamma}
\sum_m(\widehat s_m^\Gamma+\varepsilon)\widehat\eta_m^{\rm bdry}.
\end{equation}
If $|r_\Theta|\le R_\infty$ on $Q$ and $|b_\Theta|\le G_\infty$ on $\Gamma$, then, for $0<\delta<1$, with probability at least
$1-\delta$,
\begin{equation}
\|u_\Theta-u\|_X
\le C_{\rm stab}\left[
\widehat{\mathcal E}_{\mathrm{ReCAP}}^2
+R_\infty^2\sqrt{\frac{\log(4/\delta)}{2N_c}}
+\kappa_\Gamma G_\infty^2\sqrt{\frac{\log(4/\delta)}{2N_\Gamma}}
\right]^{1/2}.
\label{app:eq:recap-sample-certificate}
\end{equation}
<!-- end theorem-3 -->
<!-- proof-1: body_md -->
By definition and the partition-of-unity identity,
\begin{align}
\sum_m(s_m+\varepsilon)\eta_m^{\rm res}
&=\sum_m\int_Q\pi_m|r_\Theta|^2\,d\mu_Q
=\int_Q\left(\sum_m\pi_m\right)|r_\Theta|^2\,d\mu_Q
=\|r_\Theta\|_{L^2(\mu_Q)}^2.
\end{align}
The same calculation gives
$\sum_m(s_m^\Gamma+\varepsilon)\eta_m^{\rm bdry}
=\|b_\Theta\|_{L^2(\mu_\Gamma)}^2$.
Substitution into equation~\ref{app:eq:stability_assumption} proves equation~\ref{app:eq:recap-reliability}.

For every realization of the certification samples, the same algebra gives
\begin{equation}
\frac1{N_c}\sum_m(\widehat s_m+\varepsilon)\widehat\eta_m^{\rm res}
=\frac1{N_c}\sum_{i=1}^{N_c}|r_\Theta(Z_i)|^2,
\end{equation}
and likewise on $\Gamma$. Since the predictor is frozen before the certification samples are drawn,
Hoeffding's inequality yields, with probability at least $1-\delta/2$,
\begin{equation}
\|r_\Theta\|_{L^2(\mu_Q)}^2
\le \frac1{N_c}\sum_i|r_\Theta(Z_i)|^2
+R_\infty^2\sqrt{\frac{\log(4/\delta)}{2N_c}},
\end{equation}
and, with probability at least $1-\delta/2$,
\begin{equation}
\|b_\Theta\|_{L^2(\mu_\Gamma)}^2
\le \frac1{N_\Gamma}\sum_j|b_\Theta(Y_j)|^2
+G_\infty^2\sqrt{\frac{\log(4/\delta)}{2N_\Gamma}}.
\end{equation}
A union bound and equation~\ref{app:eq:stability_assumption} prove equation~\ref{app:eq:recap-sample-certificate}.

<!-- proposition-1: body_md -->
\label{prop:cov_direction}
For a selected component, define
$\omega_i=\pi_m(z_i)|r_\Theta(z_i)|^2$, $W=\sum_i\omega_i>0$,
$\bar z=W^{-1}\sum_i\omega_i z_i$, and
$C=W^{-1}\sum_i\omega_i(z_i-\bar z)(z_i-\bar z)^\top$.
If $v_1$ is a unit eigenvector associated with $\lambda_{\max}(C)$, then
\begin{equation}
v_1\in\operatorname*{arg\,max}_{\|v\|_2=1}
\frac1W\sum_i\omega_i\bigl(v^\top(z_i-\bar z)\bigr)^2.
\end{equation}
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
The objective equals $v^\top Cv$. Writing $C=V\Lambda V^\top$ and $a=V^\top v$ gives
$v^\top Cv=\sum_k\lambda_k a_k^2\le\lambda_{\max}(C)\sum_k a_k^2
=\lambda_{\max}(C)$, with equality for a leading eigenvector.

<!-- lemma-1: body_md -->
\label{lem:poly_translation}
Let $\xi_p=(z-\mu_p)\oslash\sigma_p$ and $\xi_c=(z-\mu_c)\oslash\sigma_c$.
Set $d=(\mu_c-\mu_p)\oslash\sigma_p$ and $b=\sigma_c\oslash\sigma_p$. If
$\ell_p=\sum_{|\alpha|\le p}c_\alpha^p\xi_p^\alpha$ and the complete monomial basis through degree $p$ is
used, then the child coefficients
\begin{equation}
c_\beta^c=
\sum_{\substack{|\alpha|\le p\\\alpha\succeq\beta}}
c_\alpha^p\binom{\alpha}{\beta}d^{\alpha-\beta}b^\beta
\label{app:eq:translated_coef}
\end{equation}
satisfy $\ell_c(z)=\ell_p(z)$ for every $z$.
<!-- end lemma-1 -->
<!-- proof-3: body_md -->
The coordinates satisfy $\xi_p=d+b\odot\xi_c$. Substituting this identity into the parent polynomial and
applying the multi-index binomial theorem gives
\begin{equation}
\ell_p
=\sum_{|\alpha|\le p}c_\alpha^p
\sum_{\beta\preceq\alpha}\binom{\alpha}{\beta}
d^{\alpha-\beta}b^\beta\xi_c^\beta.
\end{equation}
Collecting the coefficient of each $\xi_c^\beta$ gives equation~\ref{app:eq:translated_coef}. Completeness of the basis
ensures that every resulting monomial is present.

<!-- theorem-4: body_md -->
\label{thm:quiet_child}
Translate a new child from parent $p$ according to Lemma~\ref{lem:poly_translation}, leave the parent
unchanged, and let $q_c=\gamma_c\phi_c$ with $0\le\gamma_c\le\epsilon_g$ and $\gamma_c$ constant in $z$. Assume $\mathcal L=\partial_t+\beta\cdot\nabla$, $0\le\phi_c\le1$ on $Q\cup\Gamma$, and that the following lower bound on $D$ holds on $Q\cup\Gamma$:
\begin{equation}
D\ge d_0>0,
\qquad
\|\mathcal L\phi_c\|_{L^\infty(Q)}\le K_\phi,
\qquad
\|\mathcal LD\|_{L^\infty(Q)}\le K_D.
\end{equation}
Set $h=\ell_p-u^-$,
$K_\alpha=K_\phi/d_0+K_D/d_0^2$, and
\begin{equation}
C_{\rm split}=\left[
\left(K_\alpha\|h\|_{L^2(Q)}+d_0^{-1}\|\mathcal Lh\|_{L^2(Q)}\right)^2
+\kappa_\Gamma d_0^{-2}\|h\|_{L^2(\Gamma)}^2
\right]^{1/2}.
\end{equation}
For
$J(v)=\|\mathcal Lv-f\|_{L^2(Q)}^2+
\kappa_\Gamma\|v-g\|_{L^2(\Gamma)}^2$, the post-split predictor satisfies
\begin{align}
\sqrt{J(u^+)}&\le\sqrt{J(u^-)}+\epsilon_gC_{\rm split},
\label{app:eq:sqrt_loss_shock}\\
J(u^+)-J(u^-)&\le
2\epsilon_gC_{\rm split}\sqrt{J(u^-)}+
\epsilon_g^2C_{\rm split}^2.
\label{app:eq:loss_shock}
\end{align}
If the child gate can approach zero, then the closure of the refined hypothesis class contains the pre-split
class and its optimal achievable loss cannot increase.
<!-- end theorem-4 -->
<!-- proof-4: body_md -->
By Lemma~\ref{lem:poly_translation}, the new child represents $\ell_p$ in global coordinates. Directly from
equation~\ref{app:eq:mixture_assumption},
\begin{equation}
u^+-u^-=\alpha h,
\qquad
\alpha=\frac{q_c}{D+q_c}.
\label{app:eq:split_identity}
\end{equation}
Since $q_c\le\epsilon_g$ and $D\ge d_0$, $0\le\alpha\le\epsilon_g/d_0$. Moreover,
\begin{equation}
\mathcal L\alpha=
\frac{D\,\mathcal Lq_c-q_c\,\mathcal LD}{(D+q_c)^2}.
\end{equation}
The gate is constant with respect to $z$ at insertion, so
$\|\mathcal Lq_c\|_\infty\le\epsilon_gK_\phi$. Therefore
\begin{equation}
\|\mathcal L\alpha\|_\infty
\le\epsilon_g\left(\frac{K_\phi}{d_0}+\frac{K_D}{d_0^2}\right)
=\epsilon_gK_\alpha.
\end{equation}
Applying the product rule to equation~\ref{app:eq:split_identity} gives
\begin{align}
\|\mathcal L(u^+-u^-)\|_{L^2(Q)}
&\le\epsilon_g\left(
K_\alpha\|h\|_{L^2(Q)}+d_0^{-1}\|\mathcal Lh\|_{L^2(Q)}
\right),\\
\|u^+-u^-\|_{L^2(\Gamma)}
&\le\epsilon_gd_0^{-1}\|h\|_{L^2(\Gamma)}.
\end{align}
Thus the product norm of the residual and boundary perturbations is at most
$\epsilon_gC_{\rm split}$. Minkowski's inequality proves equation~\ref{app:eq:sqrt_loss_shock}; squaring it proves
equation~\ref{app:eq:loss_shock}. Finally, retaining every old parameter and sending the child gate logit to $-\infty$
makes $q_c\to0$, so every pre-split predictor lies in the closure of the refined class.
<!-- end proof-4 -->
