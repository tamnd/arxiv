\emph{\cite[Definition 1.2]{Hamada-2001}}
Let $f:\mathbb{B}\to X$ be a normalized locally biholomorphic mapping and let $\alpha\in [0,1)$. We say that $f$ is starlike of order $\alpha$ if 
\begin{align*}
\displaystyle
\Re\left\lbrace ||z||/l_z([Df(z)]^{-1}f(z))\right\rbrace>\alpha,\quad z\in\mathbb{B}\backslash\{0\},\quad l_z\in T(z).
\end{align*}

\cite[Definition 1.1]{Liu-Liu-2010}
A normalized locally biholomorphic mapping $f\in\mathcal{H}(\mathbb{B})$ is said to be quasi-convex of type $B$ and order $\alpha\;(0\leq \alpha<1)$ if
\begin{align*}
\displaystyle
\Re\left\lbrace l_z\left([Df(z)]^{-1}\left(D^2f(z)(z,z)+Df(z)z\right)\right)\right\rbrace>\alpha ||z||,\quad z\in\mathbb{B}\backslash\{0\}.
\end{align*}

\label{Th-21}Let $f\in\mathcal{H}(\mathbb{B},\mathbb{C})$ with $f(0)=1$ and $F(z)=zf(z)\in\mathcal{S}^*_{\alpha}(\mathbb{B})$. Then 
 \begin{align}\label{Th-2.1}
 	\left|\frac{l_{z}(D^{k}F(0)(z^{k}))}{k!||z||^{k}}\right|\leq \frac{1}{(k-1)!}\prod\limits_{m=2}^{k}(m-2\alpha),\quad z\in\mathbb{B}\backslash\{0\}

\label{Lem1} {\em{\cite{Cho-Kumar-Ravichandran-2019}}}
	Let $p(z)=1+\sum_{n=1}^{\infty}p_nz^n\in\mathcal{P}$. Then $|p_n|\leq 2$ and $|p_n-p_mp_{n-m}|\leq 2$, $n\geq 2$, $1\leq m\leq n-1$.

<!-- proof-1: body_md -->
We have $f\in\mathcal{H}(\mathbb{B},\mathbb{C})$ with $f(0)=1$ and $F(z)=zf(z)\in\mathcal{S}^*_{\alpha}(\mathbb{B})$. Let $z\in \mathbb{B}\backslash\{0\}$ be fixed. Suppose $z_0=z/\|z\|$. Define $g:\mathbb{U}\to \mathbb{C}$ as follows
\begin{align}\label{Th-21.0}	
\displaystyle
g(\zeta)=\begin{cases}
\frac{\zeta}{l_{z_0}\left([D F(\zeta z_0)]^{-1}F(\zeta z_0)\right)},& \zeta\neq 0,\\[8pt]
1& \zeta=0.
\end{cases}
\end{align}

Clearly $g\in\mathcal{H}(\mathbb{U})$ and $g(0)=1$.
Since $l_{\zeta z}(.)=\frac{|\zeta|}{\zeta}l_z$ for each $\zeta\in\mathbb{C}\backslash\{0\}$, we have
\begin{align*}
g(\zeta)=\frac{\zeta}{l_{z_0}\left([D F(\zeta z_0)]^{-1}F(\zeta z_0)\right)}=\frac{\|\zeta z_0\|}{l_{\zeta z_0}\left([D F(\zeta z_0)]^{-1}F(\zeta z_0)\right)}
\end{align*}	
for $\zeta\in\mathbb{U}\backslash\{0\}$  and so  $\Re\{g(\zeta)\}>\alpha$
for all $\zeta\in\mathbb{U}$. Suppose
\begin{align}\label{Th-21.a}
g(\zeta)=1+g_1\zeta+g_2\zeta^2+g_3\zeta^3\ldots
\end{align}

Let $p(\zeta)=(g(\zeta)-\alpha)/(1-\alpha)$
for all $\zeta\in\mathbb{U}$. Clearly $p\in\mathcal{H}(\mathbb{U})$ with $\Re\{p(\zeta)\}>0$ for all $\zeta\in\mathbb{U}$ and $p(0)=1$. 
Suppose $h(\zeta)=1+p_1\zeta+p_2\zeta^2+p_3\zeta^3+\ldots$.
According to Lemma \ref{Lem1}, we have $|p_k|\leq 2$ for $k=1,2,\ldots$.
Clearly 
\begin{align*}
g(\zeta)=1+\sum\limits_{i=1}^{\infty}g_i\zeta^i=1+(1-\alpha)p_1\zeta+(1-\alpha)p_2\zeta^2+(1-\alpha)p_3\zeta^3+\ldots
\end{align*}

Consequently, for $k=1,2,\ldots$, we have
\begin{align}\label{Th-21.b}
|g_k|\leq (1-\alpha)|p_k|\leq 2(1-\alpha).
\end{align}

Using a similar method as in Theorem 7.1.14 \cite{Graham-Kohr}, we have
                             
                                                                                                           
            
           
\begin{align}\label{Th-21.2}
	[D F(z)]^{-1}\eta=\frac{1}{f(z)}\left\lbrack \eta-\frac{\frac{(Df(z)\eta)z}{f(z)}}{1+\frac{Df(z)z}{f(z)}}\right\rbrack.
\end{align}

Therefore 
\begin{align}\label{Th-21.2}
	[D F(z)]^{-1}F(z)=\frac{zf(z)}{f(z)+D f(z)z},
\end{align}
which implies that $\frac{\|z\|}{l_z\left([D F(z)]^{-1}F(z)\right)}=1+\frac{Df(z) z}{f(z)}$
and so
\begin{align}\label{Th-21.3}
	g(\zeta)=\frac{\|\zeta z_0\|}{l_{\zeta z_0}\left([D F(\zeta z_0)]^{-1}F(\zeta z_0)\right)}=1+\frac{Df(\zeta z_0) \zeta z_0}{f(\zeta z_0)}.
\end{align}

Let 
\begin{align}\label{Th-21.4}
f(z)=1+\sum\limits_{k=1}^{\infty} \frac{1}{k!}D^kf(0)(z^k).
\end{align}	

Clearly
\begin{align}\label{Th-21.5}
Df(z)(z)=\sum\limits_{k=1}^{\infty} \frac{k}{k!}D^kf(0)(z^k).
\end{align}	

If we take $ h(\zeta)=f(\zeta z_0)$, then from \eqref{Th-21.4}, we have
\begin{align}\label{Th-21.6}
\displaystyle
 h(\zeta)=f(\zeta z_0)=1+a_1\zeta+a_2\zeta^2+\ldots.
\end{align}
where
\begin{align}\label{Th-21.7}
a_k=\frac{1}{k!}D^k f(0)(z_0^k)
\end{align}
for $k=1,2,\ldots$
We see that \eqref{Th-21.5} yields
\begin{align}\label{Th-21.8}
\displaystyle
D f(\zeta z_0)(\zeta z_0)=\sum\limits_{k=1}^{\infty}ka_k\zeta^k
\end{align}

Obviously \eqref{Th-21.6} and \eqref{Th-21.8}, yields $ h'(\zeta)\zeta=D f(\zeta z_0)(\zeta z_0)$. So from \eqref{Th-21.3}, we get
\begin{align}\label{Th-21.9}
1+\frac{\zeta h'(\zeta)}{h(\zeta)}=g(\zeta)=\sum\limits_{k=0}^{\infty}g_k\zeta^k,
\end{align}
where $g_0=1$.
Note that 
\begin{align*}
\frac{1+(1-2\alpha)\zeta}{1-\zeta}=1+2(1-\alpha)\zeta(1-\zeta)^{-1}=1+2(1-\alpha)\sum\limits_{k=1}^{\infty}\zeta^k=\sum\limits_{k=0}^{\infty}b_k\zeta^k,
\end{align*}
where $b_0=1$, $b_k=2(1-\alpha)$, $k\in\mathbb{N}$. Now \eqref{Th-21.b} yields $|g_k|\leq b_k$, $k\in\mathbb{N}\cup\{0\}$. Hence
\begin{align*}
g(\zeta)\ll\frac{1+(1-2\alpha)\zeta}{1-\zeta}
\end{align*}
and so \eqref{Th-21.9} yields
\begin{align*}
1+\frac{\zeta h'(\zeta)}{h(\zeta)}\ll \frac{1+(1-2\alpha)\zeta}{1-\zeta}.
\end{align*}

Consequently 
               
                                                          
             
       
\begin{align}\label{Th-21.10}
h(\zeta)\ll \phi(\zeta):=\frac{1}{(1-\zeta)^{2(1-\alpha)}}=1+\sum\limits_{k=1}^{\infty}c_k\zeta^k,
\end{align}
where
\begin{align}\label{Th-21.11}
c_k=\frac{(2-2\alpha)(3-2\alpha)\ldots((k+1)-2\alpha)}{k!}.
\end{align}
Using \eqref{Th-21.6} to \eqref{Th-21.10}, we get $|a_k|\leq c_k$, $k\in\mathbb{N}$. Consequently \eqref{Th-21.7} and \eqref{Th-21.11} give
\begin{align}\label{Th-21.12}
\left|\frac{1}{k!}D^k f(0)(z_0^k)\right|\leq \frac{(2-2\alpha)(3-2\alpha)\ldots((k+1)-2\alpha)}{k!}=\frac{1}{k!}\prod\limits_{m=2}^{k+1}(m-2\alpha)
\end{align}
for $k=1,2,3,\ldots$.
Since $F(z)=zf(z)$, we have (see \cite[Theorem 2]{Xiong-Yu-Wang-2025})
\begin{align}\label{Th-21.13}
D^mF(0)(z^m)=mD^{m-1}f(0)(z^{m-1})z
\end{align}
where $m\in\mathbb{N}$. As $z_0=z/||z||$, we deduce from \eqref{Th-21.13} that
\begin{align}\label{Th-21.14}
\displaystyle
\frac{l_{z_0}(D^mF(0)(z_0^m))}{m!}=\frac{D^{m-1}f(0)(z_0^{m-1})}{(m-1)!}.
\end{align}

Finally from \eqref{Th-21.12} and \eqref{Th-21.14}, we have
\begin{align*}
\left|\frac{l_{z_0}(D^{k+1}F(0)(z_0^{k+1}))}{(k+1)!}\right|\leq \frac{1}{k!}\prod\limits_{m=2}^{k+1}(m-2\alpha),
\end{align*}
i.e., for $k=2,3,\ldots$, we get the desired inequalities \eqref{Th-2.1}.
               
                                                                                                                                                    
             


To establish the sharpness of the bound, let us consider the function $F$ defined by
\begin{align*}
F(z)=zf(z)=\frac{z}{\left(1-l_u(z)\right)^{2(1-\alpha)}},\quad z\in\mathbb{B},\quad \|u\|=1.
\end{align*}

According to Lemma 2.2 \cite{Xu-He-Xu}, obviously $F\in\mathcal{S}^*(\mathbb{B})$. Note that 
\begin{align*}
f(z)=1+\sum_{k=1}^{\infty}\frac{(2-2\alpha)(3-2\alpha)\ldots((k+1)-2\alpha)}{k!}(l_u(z))^{k},\quad z\in\mathbb{B}.
\end{align*}


Clearly \eqref{Th-21.13} yields
\begin{align*}
\frac{l_z(D^{k+1}F(0)(z^{k+1}))}{(k+1)!}=(l_u(z))^k\|z\|\;\frac{1}{k!}\prod\limits_{m=2}^{k+1}(m-2\alpha).
\end{align*}

Taking $z=ru\;(0<r<1)$, we get for $k=2,3,\ldots$
               
                                                                                                    
             
                                   
\begin{align*}
\left|\frac{l_z(D^{k}F(0)(z^{k}))}{k!||z||^{k}}\right|=\frac{1}{(k-1)!}\prod\limits_{m=2}^{k}(m-2\alpha),\quad z\in\mathbb{B}\backslash\{0\}.
\end{align*}

<!-- proof-2: body_md -->
By the given condition, we have $F\in\mathcal{K}^{\alpha}_{B}(\mathbb{B})$ and so 
\begin{align}\label{Lem1.1}
\displaystyle
\Re\left\lbrace \frac{l_{z_0}\left([DF(z)]^{-1}\phi(z)\right)}{||z||}\right\rbrace>\alpha,
\end{align}
$z\in\mathbb{B}\backslash\{0\}$ and $l_z\in T(z)$, where $\phi(z)=D^2 F(z)(z^2)+DF(z) z$.	
Let $z\in \mathbb{B}\backslash\{0\}$ be fixed and let $z_0=z/\|z\|$.
We define the function $H:\mathbb{U}\to \mathbb{C}$ as follows
\begin{align}\label{Th2.2}	
\displaystyle
H(\zeta)=\begin{cases}
\frac{l_{z_0}\left([DF(\zeta z_0)]^{-1}\phi(\zeta z_0)\right)}{\zeta},& \zeta\neq 0,\\[8pt]
1& \zeta=0.
\end{cases}
\end{align}

Clearly $H\in\mathcal{H}(\mathbb{U})$ and $H(0)=1$.
Since $l_{\zeta z}(.)=\frac{|\zeta|}{\zeta}l_z$ for each $\zeta\in\mathbb{U}\backslash\{0\}$, we have
\begin{align}\label{Th2.3}
\displaystyle
H(\zeta)=\frac{l_{\zeta z_0}\left([DF(\zeta z_0)]^{-1}\phi(\zeta z_0)\right)}{||\zeta z_0||}
\end{align}	
and so $\Re\{H(\zeta)\}>\alpha$ for all $\zeta\in\mathbb{U}$.
Suppose
\begin{align}\label{Th2.3a}
H(\zeta)=1+H_1\zeta+H_2\zeta^2+H_3\zeta^3+\ldots
\end{align}

Clearly 
\begin{align}\label{Th2.4}
|H_k|\leq 2(1-\alpha)
\end{align}
for $k=1,2,\ldots$.
Using a similar method as in Theorem 7.1.14 \cite{Graham-Kohr}, we have
\begin{align}\label{Th2.5}
\displaystyle
[D F(z)]^{-1}\phi(z)=\frac{\phi(z)}{f(z)+D f(z)z}.
\end{align}


As $F(z)$ is normalized, it has the power series expansion
\begin{align}\label{Th2.9a}
\displaystyle
F(z)=z+\sum\limits_{k=2}^{\infty}\frac{1}{k!}D^k F(0)(z^k).
\end{align}

Clearly
\begin{align}\label{Th2.9}
\displaystyle	
D F(z)=I+\sum\limits_{k=2}^{\infty}\frac{k}{k!}D^k F(0)(z^{k-1})
\end{align}
and 
\begin{align}\label{Th2.10}
\displaystyle
D^2 F(z)=\sum\limits_{k=2}^{\infty}\frac{k(k-1)}{k!}D^k F(0)(z^{k-2}).
\end{align}

Using \eqref{Th2.9} and \eqref{Th2.10}, we get
\begin{align}\label{Th2.11}
\displaystyle
D^2 F(z)(z^2)+DF(z)
z=&z+\sum\limits_{k=2}^{\infty}\frac{k^2}{k!}D^k F(0)(z^k).
\end{align}


Now substituting \eqref{Th-21.13} into \eqref{Th2.11}, we get
\begin{align}\label{Th2.13}
\displaystyle
D^2 F(z)(z^2)+DF(z)
z=&z+\sum\limits_{k=2}^{\infty}\frac{k^3}{k!}D^{k-1} f(0)(z^{k-1}) z.
\end{align}

Clearly 
\begin{align}\label{Th2.13a}
\displaystyle
\phi(\zeta z_0)=\zeta z_0\left(1+\sum\limits_{m=1}^{\infty}\frac{(m+1)^2}{m!}D^{m} f(0)(z^{m})\zeta^m\right).
\end{align}

Let
\begin{align}\label{Th2.14}
\displaystyle
f(\zeta z_0)=1+\sum\limits_{k=1}^{\infty}\frac{1}{k!}D^k f(0)(z_0^k)\zeta^k.
\end{align}

Clearly
\begin{align}\label{Th2.15}
\displaystyle
D f(\zeta z_0)(\zeta z_0)=\sum\limits_{k=1}^{\infty}\frac{k}{k!}D^k f(0)(z_0^k)\zeta^k.
\end{align}

Suppose $g(\zeta)=f(\zeta z_0)\zeta$. Then we have
\begin{align}\label{Th2.7}
g'(\zeta)=f(\zeta z_0)+Df(\zeta z_0)\zeta z_0.
\end{align}

In view of \eqref{Th2.7} and adding \eqref{Th2.9} and \eqref{Th2.10}, we obtain
\begin{align}\label{Th2.16}
\displaystyle
g'(\zeta)=f(\zeta z_0)+D f(\zeta z_0)\zeta z_0=1+\sum\limits_{k=1}^{\infty}\frac{k+1}{k!}D^k f(0)(z_0^k)\zeta^k.
\end{align}	

Clearly
\begin{align}\label{Th2.17}
\displaystyle
\zeta g''(\zeta)+g'(\zeta)=1+\sum\limits_{k=1}^{\infty}\frac{(k+1)^2}{k!}D^k f(0)(z_0^k)\zeta^k.
\end{align}

Now from \eqref{Th2.5}, \eqref{Th2.13a} and \eqref{Th2.16}, we get
\begin{align}\label{Th2.18}
\displaystyle
\frac{l_{\zeta z_0}\left([D F(\zeta z_0)]^{-1}\phi(\zeta z_0)\right)}{||\zeta z_0||}=\frac{1+\sum\limits_{m=1}^{\infty}\frac{(m+1)^2}{m!}D^{m} f(0)(z^{m})\zeta^m}{1+\sum\limits_{k=1}^{\infty}\frac{k+1}{k!}D^k f(0)(z_0^k)\zeta^k}.
\end{align}

Combining \eqref{Th2.16} and \eqref{Th2.17} to \eqref{Th2.18}, we obtain
\begin{align}\label{Th2.19}
\displaystyle
\frac{l_{\zeta z_0}\left([D F(\zeta z_0)]^{-1}\phi(\zeta z_0)\right)}{||\zeta z_0||}
=&\frac{\zeta g''(\zeta)+g'(\zeta)}{g'(\zeta)}=1+\zeta\frac{
g''(\zeta)}{g'(\zeta)}.
\end{align}

In view of \eqref{Th2.3} and \eqref{Th2.3a}, we get from \eqref{Th2.19} that
\begin{align}\label{Th2.20}
1+\zeta\frac{g''(\zeta)}{g'(\zeta)}=H(\zeta)=\sum\limits_{k=0}^{\infty}H_k\zeta^k,
\end{align}
where $H_0=1$.
Note that 
\begin{align*}
	\frac{1+(1-2\alpha)\zeta}{1-\zeta}=\sum\limits_{k=0}^{\infty}b_k\zeta^k,
\end{align*}
where $b_0=1$, $b_k=2(1-\alpha)$, $k\in\mathbb{N}$. So \eqref{Th2.4} yields $|H_k|\leq b_k$, $k\in\mathbb{N}\cup\{0\}$. Hence
\begin{align*}
H(\zeta)\ll\frac{1+(1-2\alpha)\zeta}{1-\zeta}
\end{align*}
and so \eqref{Th2.20} yields
\begin{align*}
	1+\frac{\zeta g''(\zeta)}{g'(\zeta)}\ll \frac{1+(1-2\alpha)\zeta}{1-\zeta}.
\end{align*}

Consequently 
               
                                                            
             
       
\begin{align}\label{Th2.21}
g'(\zeta)\ll \phi(\zeta):=\frac{1}{(1-\zeta)^{2(1-\alpha)}}=1+\sum\limits_{k=1}^{\infty}c_k\zeta^k,
\end{align}
where
\begin{align}\label{Th2.22}
	c_k=\frac{(2-2\alpha)(3-2\alpha)\ldots((k+1)-2\alpha)}{k!}
\end{align}
for $k=1,2,3,\ldots$.
Now using \eqref{Th2.16} to \eqref{Th2.21}, we get 
\begin{align*}
\left|\frac{k+1}{k!}D^k f(0)(z_0^k)\zeta^k\right|\leq c_k
\end{align*}
for $k=1,2,3,\ldots$. Consequently from \eqref{Th2.22}, we have
\begin{align}\label{Th2.23}
\left|\frac{k+1}{k!}D^k f(0)(z_0^k)\right|\leq \frac{(2-2\alpha)(3-2\alpha)\ldots((k+1)-2\alpha)}{k!}=\frac{1}{k!}\prod\limits_{m=2}^{k+1}(m-2\alpha)
\end{align}
for $k=1,2,3,\ldots$.
Finally from \eqref{Th2.23} and \eqref{Th-21.14}, we have
\begin{align*}
	\left|\frac{l_{z_0}(D^{k+1}F(0)(z_0^{k+1}))}{(k+1)!}\right|\leq \frac{1}{(k+1)!}\prod\limits_{m=2}^{k+1}(m-2\alpha),
\end{align*}
i.e., for $k=2,3,\ldots$,  we get the desired inequalities \eqref{Th-2.2}.
               
                                                                                                                                                 
             


To establish the sharpness of the bound, let us consider the function $F$ defined by
\begin{align*}
F(z)=zf(z)=\frac{f_1(l_u(z))}{l_u(z)}z,\quad z\in\mathbb{B},\quad \|u\|=1,
\end{align*}
where
\begin{align*}
f_1(l_u(z))=
\begin{cases}
\frac{1-(1-l_u(z))^{2\alpha-1}}{2\alpha-1}, & \alpha\neq \frac{1}{2}, \\[10pt]
-\log (1-l_u(z)), & \alpha=\frac{1}{2}.
\end{cases}
\end{align*}
 We know that $f_1\in\mathcal{K}_{\alpha}(\mathbb{U})$ (see \cite[Problem 2.3.7]{Graham-Kohr}).
According to Lemma 2.1 \cite{Liu-Liu-2010}, we have $F\in\mathcal{K}^{\alpha}_B(\mathbb{B})$.
Suppose $\alpha\neq 1/2$. In this case, we have
\begin{align*}
f(l_u(z))=\frac{1}{2\alpha-1}\sum\limits_{k=0}^{\infty}(-1)^{k+2}\binom{2\alpha-1}{k+1}(l_u(z))^{k}.
\end{align*}

Clearly \eqref{Th-21.13} yields
\begin{align*}
\frac{l_z(D^{k+1}F(0)(z^{k+1}))}{(k+1)!}=(-1)^k\frac{(2\alpha-2)(2\alpha-3)\ldots(2\alpha-k-1)}{(k+1)!}\;(l_u(z))^k\|z\|.
\end{align*}

Taking $z=ru\;(0<r<1)$, we get for $k=2,3,\ldots$
               
                                                                                                                    
             
                                   
\begin{align*}
	\left|\frac{l_z(D^{k}F(0)(z^{k}))}{k!||z||^{k}}\right|=\frac{1}{k!}\prod\limits_{m=2}^{k}(m-2\alpha),\quad z\in\mathbb{B}\backslash\{0\}.
\end{align*}

Suppose $\alpha=1/2$. In this case, we have
\begin{align*}
	f(z)=\sum_{k=0}^{\infty}\frac{1}{k+1}(l_u(z))^{k},\quad z\in\mathbb{B}.
\end{align*}
                                
               
                                                                       
             

Let us take $z=ru\;(0<r<1)$. Now using \eqref{Th-21.13}, we find for $k=2,3,\ldots$ that
               
                                                                   
             
                                   
\begin{align*}
	\left|\frac{l_z(D^{k}F(0)(z^{k}))}{k!||z||^{k}}\right|=\frac{1}{k},\quad z\in\mathbb{B}\backslash\{0\}.
\end{align*}
