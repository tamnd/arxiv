<!-- definition-1: body_md -->
The subspace of $U(N)$ invariant polynomials is denoted by
\begin{equation}
    \mathcal{H}_N = \qty{ \mathcal{F} (Z,W) \in \mathbb{C}[Z,W] \, \vert \, 
    \mathcal{F} (UZU^\dagger , UWU^\dagger) = \mathcal{F} (Z,W) \quad \forall U \in U(N)}.
\end{equation}
The space of invariants can be organised into subspaces of homogeneous bidegree, where $Z$ has degree $(1,0)$ and $W$ has degree $(0,1)$.
The subspaces of homogeneous degree $(m,n)$ are defined as
\begin{equation}
    \mathcal{H}_N(m,n) = \{ \mathcal{F}(Z,W) \in \mathcal{H}_N \, \vert \, \mathcal{F}(x Z, y W) = x^m y^n \mathcal{F}(Z,W) \, \forall x,y \in \mathbb{C}\},
\end{equation}
and give a decomposition
\begin{equation}
    \mathcal{H}_N = \bigoplus_{m,n \geq 0} \mathcal{H}_N(m,n).
\end{equation}

<!-- example-1: body_md -->
\label{ex: H_N(2,1)}
All elements in $\mathcal{H}_N(2,1)$ are linear combinations of 
\begin{equation}
\mathrm{tr}(ZZW), \quad \mathrm{tr}(ZZ)\mathrm{tr}(W), \quad \mathrm{tr}(ZW)\mathrm{tr}(Z), \quad \mathrm{tr}(Z)^2 \mathrm{tr}(W).
\end{equation}
In the stable regime $N \geq 3$, these elements form a basis. For $N < 3$ they are linearly dependent.

<!-- definition-2: body_md -->
Let $S_L$ be the symmetric group on $L$ letters.
The symmetric group algebra $\mathbb{C}[S_L]$ is the complex vector space whose basis is $S_L$
\begin{equation}
\mathbb{C}[S_L]
= \Big\{ \sum_{g\in S_L} c_g \, g \, \Big| \, c_g \in\mathbb{C} \Big\}.
\end{equation}
It becomes a finite-dimensional associative algebra when the group multiplication is extended bilinearly
\begin{equation}
\Big( \sum_{g \in S_L} c_{g}  g \Big)
\Big( \sum_{g' \in S_L} c_{g'} {g'} \Big)
= \sum_{g, g' \in S_L} c_{g}  \, c_{g'} \, g g' .
\end{equation}
<!-- end definition-2 -->
<!-- example-2: body_md -->
The product of elements $\sigma, \sigma' \in \mathbb{C}[S_3]$ given by
    \begin{equation}
        \sigma = (123) + (132), \quad \sigma' = (12) + (13)
    \end{equation}
    is given by
    \begin{equation}
        \sigma \sigma' = (123)(12) + (123)(13) + (132)(12) + (132)(13) = (12) + (13) + 2 \, (23).
    \end{equation}
We use the convention $(g_1g_2)(i)=g_2(g_1(i))$ following \cite{Padellaro:2024rld}.

<!-- definition-3: body_md -->
The permutation centraliser algebra (PCA) at degree $(m,n)$ is the subspace of the group algebra made out of elements that commute with $S_m \times S_n$
\begin{equation}
{\mathcal A}(m,n)
=
\{\sigma \in \mathbb{C}[S_{m+n}] \, \vert  \, \gamma \sigma \gamma^{-1}=\sigma, \quad \forall \gamma \in S_m \times S_n \}.
\label{def:PCA}
\end{equation}

The group algebra element $\sigma = (13)(24) - (23)(14)$ is not in ${\mathcal A}(2,2)$, but $\sigma = (13)(24) + (14)(23)$ is an element of ${\mathcal A}(2,2)$.

<!-- definition-4: body_md -->
\label{def:trace map}
For $g\in S_{m+n}$\,, we define the multi-trace $U(N)$ invariant
\begin{equation}
\mathscr{O}^{(N)}_{m,n}(g) \equiv \sum_{i_1,\dots,i_{m+n}=1}^N
\prod_{p=1}^m Z_{i_p i_{g(p)}}
\prod_{q=m+1}^{m+n} W_{i_q i_{g(q)}} \,.
\label{eq:Og-definition}
\end{equation}
For a group-algebra element $\sigma=\sum_{g\in S_{m+n}}c_g(\sigma)g$, we linearly extend the above map,
\begin{equation}
\mathscr{O}^{(N)}_{m,n}(\sigma) \equiv
\sum_{g\in S_{m+n}}c_g(\sigma)\,\mathscr{O}^{(N)}_{m,n}(g) .
\label{eq:Osigma-definition}
\end{equation}
Since $\mathscr{O}^{(N)}_{m,n}(\sigma) = \mathscr{O}^{(N)}_{m,n}(h \sigma h^{-1})$ for any $h \in S_m \times S_n$\,, we obtain a map from permutation centraliser algebras to invariants
\begin{equation}
\mathscr{O}^{(N)}_{m,n}: {\mathcal A}(m,n) \rightarrow \mathcal{H}_N(m,n).
\end{equation}

Let $\{ g_1,\dots, g_l\}$ be a complete set of representatives of $S_m \times S_n$ orbits of $S_{m+n}$.
For each representative $g_k$, define the following element in ${\mathcal A}(m,n)$
\begin{equation}
    P_{m,n}(g_k)=\sum_{h \in S_m \times S_n } hg_kh^{-1} \,.

\label{eq:orbit-basis}
\end{equation}

<!-- definition-6: body_md -->
\label{def:orbit-basis inner prod}
Define the linear function $\delta: \mathbb{C} [S_{m+n}] \rightarrow \mathbb{C}$ by
\begin{equation}
    \delta(g) = \begin{cases}
        1, &\text{ if $g=\mathrm{id}$}\\
        0, &\text{ if $g\neq \mathrm{id}$}.
    \end{cases}
\end{equation}
A natural inner product on ${\mathcal A}(m,n)$ is given by
\begin{equation}
\langle \sigma, \sigma' \rangle = \delta(\sigma {\mathcal S}(\sigma'))
\label{eq: inner prod}
\end{equation}
where ${\mathcal S}$ is the involution
\begin{equation}
    \mathcal{S}\qty(\sum_{k=1}^l c_k P_{m,n}(g_k)) = \sum_{k=1}^l c_k^* P_{m,n}(g_k^{-1}).
\end{equation}
The orbit basis is orthogonal with respect to this inner product, which we call the orbit-basis inner product.

The centre of $\mathbb{C}[S_L]$ is the subspace of elements that commute with all of $\mathbb{C}[S_L]$
\begin{equation}
    \mathcal{Z} ( \mathbb{C}[S_L] ) = \{ \omega \in \mathbb{C}[S_L] \, \vert \, g \omega = \omega g \quad \forall g \in S_L \}.

\label{def:left center}
Let $\omega \in \mathcal{Z}(\mathbb{C}[S_L])$ have an expansion
\begin{equation}
    \omega = \sum_{g \in S_L} c_g(\omega) g.
\end{equation}
The left action of $\omega$ on $\sigma \in {\mathcal A}(m,n)$ is given by
\begin{equation}
m^{\mathfrak{L}}: \mathcal{Z}(\mathbb{C}[S_{m+n}]) \otimes {\mathcal A}(m,n) \rightarrow {\mathcal A}(m,n) , \qquad
m^{\mathfrak{L}}[\omega ](\sigma) = \sum_{g \in S_L} c_g(\omega ) g \sigma.
\end{equation}

<!-- definition-10: body_md -->
\label{def:finite-N basis}
A basis for ${\mathcal A}(m,n)$ is called SEP
if it simultaneously diagonalises the left action of the centre.
Such bases can always be labelled by an integer partition $R \vdash L$ and $a \in \{1,\dots, g(R)\}$, where $g(R)$ is defined in equation \eqref{eq: g(R)}.
For example, write
\begin{equation}
    \mathcal{Q}^{R}_a = \sum_{k=1}^l \mathcal{Q}^{R}_a(g_k) P_{m,n}(g_k)
\label{cQRa expand}
\end{equation}
then the coefficients $\mathcal{Q}^{R}_a(g_k)$ are constrained by demanding
\begin{equation}
    m^{\mathfrak{L}}[ \omega ](\mathcal{Q}^R_a) = \frac{\chi^R(\omega )}{d_R} \mathcal{Q}^R_a.
\label{left-center Q^R_a}
\end{equation}
Here $\chi^R (\sigma)$ is the $S_L$ character for $\sigma \in \mathbb{C}[S_L]$, $d_R$ is the dimension of the irreducible representation of $S_L$ labelled by $R$, and $\frac{\chi^R(\omega )}{d_R}$ is called a central character.

<!-- proof-1: body_md -->
For the first claim, decompose an arbitrary element $\sigma\in{\mathcal A}(m,n)$ into its $R$-sector components in the SEP basis:
\begin{equation}
\sigma=\sum_R\sigma_R, \qquad \sigma_R=\sum_a\sigma_{R,a} \, {\mathcal Q}^R_a.
\end{equation}
Distinct $R$-sectors are orthogonal in the orbit-basis inner product, and $\Omega_N$ acts on each sector by the scalar $f^R$.
The $U(N)$ inner product in Definition \ref{prop:UN inner} therefore expresses the free two-point function as
\begin{equation}
    \left\langle \mathscr{O}^{(N)}_{m,n}(\sigma) \, \mathscr{O}^{(N)}_{m,n}(\sigma)^\dagger \right\rangle_{{\mathcal N}=4}
    = \langle \sigma,\sigma\rangle_{U(N)}
    = m!n! \sum_R f^R \langle \sigma_R,\sigma_R\rangle .
\label{eq:SEP-norm}
\end{equation}
The orbit-basis inner product is positive definite, so $\langle \sigma_R,\sigma_R\rangle>0$ whenever $\sigma_R\neq0$.
Since $f^R>0$ for $\ell(R)\leq N$ and $f^R=0$ otherwise, the right-hand side vanishes if and only if $\sigma_R=0$ for every $R$ with $\ell(R)\leq N$.
The free two-point function is positive definite on ${\mathcal H}_N(m,n)$, so the left-hand side vanishes if and only if $\sigma\in\ker(\mathscr{O}^{(N)}_{m,n})$.
Comparing these conditions proves \eqref{eq:radical-kernel}.
The same identity shows that the restriction of the $U(N)$ inner product to ${\mathcal A}_N(m,n)$ is positive definite.
For the second claim, the two summands in \eqref{eq:PCA-orth-decomp} are spanned by the $R$-sectors with $\ell(R)\leq N$ and $\ell(R)>N$, respectively.
They exhaust ${\mathcal A}(m,n)$ and are orthogonal in the orbit-basis inner product, proving the orthogonal decomposition.

<!-- proof-2: body_md -->
The permutations-to-invariants map $\mathscr{O}^{(N)}_{m,n}: {\mathcal A}(m,n) \rightarrow {\mathcal H}_N(m,n)$ is a linear map between finite-dimensional vector spaces.
We use the result from linear algebra that says that the restricted map is an isomorphism to the image.
In the present case, this says that we have an isomorphism
\begin{equation}
    \hat \mathscr{O}^{(N)}_{m,n}: {\mathcal A}_N(m,n) \rightarrow \operatorname{im} \mathscr{O}^{(N)}_{m,n}
\end{equation}
Since $\mathscr{O}^{(N)}_{m,n}$ is surjective, $\operatorname{im} \mathscr{O}^{(N)}_{m,n}  = {\mathcal H}_N(m,n)$ and we get ${\mathcal A}_N(m,n) \cong {\mathcal H}_N(m,n)$.
In other words, the restricted permutations-to-invariants map is an isomorphism to the space of invariants.

<!-- definition-14: body_md -->
\label{def: 1LPDO}
For $h \in S_L$, define the intermediate map
\begin{equation}
    \bar H^{(1)}_{\xi}(h) \equiv \sum_{\substack{i=1\\ i\ne h(i)}}^L
\left[ h - (i,h(i))\, h\,(i,h(i)) \right]
 +{\xi} \sum_{i=1}^L \sum_{\substack{j=1\\ h(j)\ne i\\ j\ne i}}^L
\left[ h\, (i,h(j)) - (i,j)\, h\,(i,h(j))\,(i,j) \right].
\label{eq:D1-orbit-basis}
\end{equation}
This map is covariant under conjugation as shown in \eqref{app:D1 SL covariance},
\begin{equation}
\bar H^{(1)}_{\xi}(\gamma h\gamma^{-1}) = \gamma\bar H^{(1)}_{\xi}(h)\gamma^{-1}, \qquad \gamma\in S_L.
\label{eq:H1-conjugation}
\end{equation}
Consequently, $P_{m,n}\circ\bar H^{(1)}_{\xi}$ is independent of the representative chosen within each $S_m\times S_n$ orbit.
In the orbit basis, with representatives $\{g_1,\dots,g_l\}$, define the one-loop permutation dilatation operator by
\begin{equation}
    \begin{aligned}
        & H^{(1)}_{\xi}: {\mathcal A}(m,n) \rightarrow {\mathcal A}(m,n) \\
        & H^{(1)}_{\xi}(P_{m,n}(g_k)) \equiv P_{m,n} \circ \bar H^{(1)}_{\xi}(g_k).
    \end{aligned}
\end{equation}
The first term of \eqref{eq:D1-orbit-basis} is the planar contribution, while the second term is the non-planar contribution and carries an explicit factor of ${\xi}$.
Writing the full composition explicitly gives
\begin{equation}
    \begin{aligned}
    & H^{(1)}_{\xi}(P_{m,n} (g_k))
    =
\sum_{\substack{i=1\\ i\ne g_k(i)}}^L
\left\{
P_{m,n} (g_k) - P_{m,n}\!\left((i,g_k(i))g_k(i,g_k(i))\right)
\right\}
\\
    &\quad\quad\quad\quad\quad\quad\quad\quad +{\xi}
\sum_{i=1}^L
\sum_{\substack{j=1\\ g_k(j)\ne i\\ j\ne i}}^L
\left\{
P_{m,n} \!\left(g_k(i,g_k(j))\right)
- P_{m,n} \!\left((i,j)g_k(i,g_k(j))(i,j)\right)
\right\}.
    \end{aligned}
\end{equation}

<!-- lemma-1: body_md -->
\label{cor:finite-N factorisation}
Let $( H_{\xi}^{(1)})_{ba}^{SR}$ be the matrix elements of the one-loop permutation dilatation operator in a SEP basis ${\mathcal Q}^R_a$.
It has block triangular structure at $\xi = 1/N$
\begin{equation}
    ( H_{\xi}^{(1)})_{ba}^{SR}\Big \vert_{\xi=\frac{1}{N}} = \begin{pmatrix}  \mathfrak{D}_N^{(1)} & 0 \\ H_{21}^{(1)} & H_{22}^{(1)} \end{pmatrix}, 
\end{equation}
where the first and second blocks correspond to ${\mathcal A}_N(m,n)$ and $\ker(\mathscr{O}^{(N)}_{m,n})$, respectively.
The zero block corresponds to $\ell(R)>N$ and $\ell(S)\leq N$.
The matrix in the top-left block is the operator $ \mathfrak{D}_N^{(1)}$ evaluated in the basis $O^R_a = \hat \mathscr{O}^{(N)}_{m,n}({\mathcal Q}^R_a)$.
<!-- end lemma-1 -->
<!-- proof-4: body_md -->
We consider two cases.
    The first case is $R$ with $\ell(R) \leq N$ such that ${\mathcal Q}^R_a \in {\mathcal A}_N(m,n)$.

    \begin{equation}
        ( H_{\xi}^{(1)})_{ba}^{SR}\Big \vert_{\xi=\frac{1}{N}} = ( \mathfrak{D}_N^{(1)})_{ba}^{SR}, \qquad \forall R,S \vdash (m+n) \text{ s.t. } \ell(R) \leq N, \ell(S) \leq N.
    \end{equation}
    This is the upper-left block in the lemma statement.

    \end{equation}

Expanding the generating functions in terms of irreducible characters of $U(2)$ gives

\begin{equation}

\begin{equation}

\end{equation}
and Pieri's formula which gives

    \chi^{[\lambda_1+1,\lambda_2+1]}(x,y)\,
    \end{aligned}
\end{equation}
where a character vanishes, by definition, if $\lambda_2 > \lambda_1$.

\Big) .
\end{equation}

<!-- proof-13: body_md -->
We give a simple argument based on counting using the inclusion-exclusion principle.
Note that $\dim {\mathcal D}(m,n)$ is equal to the number of orbit basis elements of ${\mathcal A}(m,n)$ with no fixed points.
The number of orbit basis elements of ${\mathcal A}(m,n)$ with at least one fixed point in $\{1,\dots,m\}$ is given by $\dim {\mathcal A}(m-1,n)$.
Similarly the number of orbits with at least one fixed point in $\{m+1,\dots,m+n\}$ is given by $\dim {\mathcal A}(m,n-1)$.
However, this overcounts by the overlap -- the orbits of permutations with at least one fixed point in each set -- which have cardinality $\dim {\mathcal A}(m-1,n-1)$.
The inclusion-exclusion principle gives
\begin{equation}
    \dim {\mathcal D}(m,n) = \dim {\mathcal A}(m,n) - \dim {\mathcal A}(m,n-1) - \dim {\mathcal A}(m-1,n) + \dim {\mathcal A}(m-1,n-1) \,.
\end{equation}
For $N \geq m+n$ we have $\dim {\mathcal A}(m,n) = \dim \mathcal{H}_N(m,n)$ and together with the factorisation in Proposition~\ref{prop:hilb factorisation} we get
\begin{equation}
    \begin{aligned}
        \dim {\mathcal D}(m,n) &=
        \dim {\mathcal H}_N(m,n) - \dim {\mathcal H}_N(m,n-1) - \dim {\mathcal H}_N(m-1,n) + \dim {\mathcal H}_N(m-1,n-1) \\
        &= \dim {\mathcal H}_N'(m,n) \,.
    \end{aligned}
\end{equation}
Since the map is surjective and the dimensions agree, the kernel is trivial.

Since $\mathcal{F}(Z,W) \in {\mathcal H}_N$ has a finite expansion into elements of $\mathcal{H}'_N \otimes \mathbb{C}[z,w]$,
\begin{equation}
    \label{eq: exp Fe}
    \mathcal{F}(Z,W) = \sum_{a,b = 0}^\infty \mathcal{F}_{a,b}(Z',W')z^a w^b,
\end{equation}
it suffices to prove this for polynomials of the form $\mathcal{F}'(Z',W')z^a w^b$. 
The substitution maps are omitted below to simplify the notation.

First, for $(X,x) \in \{(Z,z),(W,w)\}$, the traceless derivatives satisfy \eqref{eq: traceless D on trace}:
\begin{equation}
   \label{eq:traceless deriv orthogonality}
\mathcal L^{X'}_{ji} ( x ) =  0 \, .
\end{equation}
It follows that for $\mathcal{F}'(Z',W') \in {\mathcal H}_N'$
\begin{equation}
   \mathcal L^{X'}_{ji}  \qty( z^{a} w^{b}\mathcal{F}'(Z',W') ) =
   z^{a}w^{b}
   \mathcal L^{X'}_{ji} \Big( \mathcal{F}'(Z',W') \Big),  \quad \forall a,b \in \mathbb{Z}_{\geq 0} \,
\end{equation}
and using Lemma \ref{lemma:oneloop factorisation}
\begin{equation}
    \comm{ \mathfrak{D}^{(1)}_{N}}{x} =  0 \,.
\end{equation}
Secondly, using 
\begin{equation}
\label{eq: trace deriv orthogonality}
\pdv{}{x} X'_{ij} = \sum_{k=1}^N \pdv{}{X_{kk}}   \qty(X_{ij} - \delta_{ij}x) = \sum_{k=1}^N \qty(\delta_{ki} \delta_{kj} - \frac{\delta_{ij}}{N}) = 0
\end{equation}

\begin{aligned}
 \mathfrak{D}^{(1)}_N \pdv[c]{z} \pdv[d]{w} \mathcal{F}'(Z',W')z^a w^b

\quad \forall c,d \in \mathbb{Z}_{\geq 0}.
\end{aligned}
\end{equation}

The easy direction assumes $\mathcal{F}^{(e)}_{a,b}(Z',W') \in \mathcal{H}'_N(e)$ are eigenvectors of the dilatation operator with eigenvalue $e$
\begin{equation}
    \mathfrak{D}^{\prime(1)}_N \mathcal{F}_{a,b}^{(e)}(Z',W') = e \mathcal{F}_{a,b}^{(e)}(Z',W'), \quad \forall a,b \geq 0.
\end{equation}
Lemma \ref{lemma: weyl dil commute} gives 
\begin{equation}
    \mathfrak{D}^{(1)}_N \qty(z^{a}w^{b}\mathcal{F}_{a,b}^{(e)}(Z',W')) 
    = z^{a}w^{b} \,  \mathfrak{D}^{\prime(1)}_N \mathcal{F}_{a,b}^{(e)}(Z',W')
    = e z^{a}w^{b}\mathcal{F}^{(e)}_{a,b}(Z',W').
\end{equation}
By linearity, we get that $\mathcal{F}^{(e)}(Z,W)$ is an eigenvector with eigenvalue $e$.

For the other direction, we assume $\mathcal{F}^{(e)}(Z,W) \in \mathcal{H}_N(e)$ and deduce that $\mathcal{F}^{(e)}_{a,b}(Z',W') \in \mathcal{H}'_N(e)$.
For this, note that the single-letter-trace counting operators 
have non-negative integer eigenvalues as operators on $\mathcal{H}_N$
\begin{equation}
    \mathrm{Spectrum}(\hat{\mathcal{N}}^{(s)}_x) = \{0,1,2,\dots \}.
\end{equation}
They can be used to construct maps extracting the coefficients $\mathcal{F}^{(e)}_{a,b}(Z',W')$.
By assumption, $\mathcal{F}^{(e)}(Z,W)$ has bounded degree and so $\mathcal{F}^{(e)}_{a,b}(Z',W')$ all vanish for some finite upper limits $a > A, b > B$.
Using this, we define
\begin{equation}
\mathrm{Coeff}_z(a) = \frac{1}{a!}\pdv[a]{z}\prod_{\substack{c=0\\c \neq a}}^A \frac{\hat{\mathcal{N}}^{(s)}_z-c}{a-c}, \qquad
\mathrm{Coeff}_w(b) =\frac{1}{b!}\pdv[b]{w} \prod_{\substack{d=0\\d \neq b}}^B \frac{\hat{\mathcal{N}}^{(s)}_w-d}{b-d} \,.
\end{equation}
They satisfy 
\begin{equation}
\mathrm{Coeff}_z(a) \, \mathrm{Coeff}_w(b)\qty(\mathcal{F}^{(e)}(Z,W)) = \mathcal{F}^{(e)}_{a,b}(Z',W') \,,
\end{equation}
and commute with the dilatation operator since they are polynomials in $\hat{\mathcal{N}}^{(s)}_x, \pdv{x}$.
A short computation gives 
\begin{equation}
    \begin{aligned}
     \mathfrak{D}^{\prime(1)}_N \mathcal{F}^{(e)}_{a,b} (Z',W')
    &= \mathfrak{D}^{(1)}_N \qty[\mathrm{Coeff}_z(a) \, \mathrm{Coeff}_w(b) \, \mathcal{F}^{(e)}(Z,W) ] \\
    &= \mathrm{Coeff}_z(a) \, \mathrm{Coeff}_w(b) \, \mathfrak{D}^{(1)}_N \mathcal{F}^{(e)}(Z,W) \\
    &= \mathrm{Coeff}_z(a) \, \mathrm{Coeff}_w(b) \, e\mathcal{F}^{(e)}(Z,W) \\
    &= e \mathcal{F}^{(e)}_{a,b}(Z',W').
    \end{aligned}
\end{equation}

<!-- remark-3: body_md -->
\label{app:single-letter-trace SL}

The single-letter-trace counting operators \eqref{def:slcnt} have a permutation description. Their action on the $U(N)$-invariant operators $\mathscr{O}^{(N)}_{m,n}(g)$ for $g \in S_L$ can be computed in a similar way to Appendix \ref{app:induced action D1} as
\begin{equation}

\hat{\mathcal{N}}^{(s)}_w \cdot \mathscr{O}^{(N)}_{m,n}(g) = \frac{1}{N} \sum_{q=m+1}^{L} \mathscr{O}^{(N)}_{m,n} \bigl(g\,\llbracket q,g(q) \rrbracket\bigr) .
\end{equation}

The total single-letter-trace counting operator satisfies the following intertwining relation similar to \eqref{eq:intertwine},
\begin{equation}

\mathcal{N}^{(s)}_{\xi} (g) = 
{\sf Fix} (g) \, g + \xi \sum_{\substack{i=1\\ i\ne g(i)}}^L g (i,g(i)) 
\label{app:Ns induced SL}
\end{equation}
where ${\sf Fix} (g)$ counts the number of fixed points in $g$.

On a polynomial $\mathcal{F}'(Z',W')z^a w^b$, the operator $\hat{\mathcal{N}}^{(s)}$ has eigenvalue $(a+b)$. 
The intertwining relation \eqref{app:Ns induced SL} implies that the permutation operator $\mathcal{N}^{(s)}_{\xi}$ preserves $\ker\mathscr{O}^{(N)}_{m,n}$ at $\xi=1/N$.
The block-triangular argument used in Lemma \ref{cor:finite-N factorisation} gives
\begin{equation}

\end{equation}
where $Q^{(s,N)}_{m,n}(x)$ is the characteristic polynomial in the null sector.
