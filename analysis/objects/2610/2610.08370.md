\label{prop:differential-criterion}
  Fix \(v\in\bfa\), and suppose that \(\ell(\bfk_v)\le r\), \(\ell(\bfl_v)\le r\), and \(\ell(\bfk_v)+\ell(\bfl_v)\le\mu\).
  Let \(P:\M_{n+r}(\bbC)\rightarrow V_{\sigma_{n,v}}\otimes V_{\sigma'_{r,v}}\) be a polynomial.
  For \(X_1,Y_1\in\M_{n,\mu}(\bbC)\), \(X_2,Y_2\in\M_{r,\mu}(\bbC)\),
  put
  \[
    \widetilde P(X_1,X_2,Y_1,Y_2)
    =
    P\!\left(
    \begin{pmatrix}
        X_1{}^tY_1 & X_1{}^tY_2 \\
        X_2{}^tY_1 & X_2{}^tY_2
      \end{pmatrix}
    \right).
  \]
  Then the covariance relation above holds if and only if the following two conditions hold:
  \begin{enumerate}
    \item
          \(\widetilde P\) is pluriharmonic in each of the pairs
          \((X_1,Y_1)\) and \((X_2,Y_2)\).

    \item
          For all
          \(A_1,B_1\in\GL_n(\bbC)\) and
          \(A_2,B_2\in\GL_r(\bbC)\),
          \[
            P\!\left(
            \begin{pmatrix}
                A_1 & 0   \\
                0   & A_2
              \end{pmatrix}
            T
            \begin{pmatrix}
                {}^tB_1 & 0       \\
                0       & {}^tB_2
              \end{pmatrix}
            \right)
            =
            \bigl(
            \sigma_{n,v}(A_1,B_1)
            \otimes
            \sigma'_{r,v}(A_2,B_2)
            \bigr)P(T).
          \]
  \end{enumerate}

\label{rem:siegel-fourier-coefficients}
  The Fourier coefficients of \(E_{m,\kappa,\nu,\frakN}\) factor into local Fourier integrals;
  see \cite[\S\S18--19]{Shimura1997EulerProducts} and \cite[\S2.2]{Eischen2015EisensteinMeasure}.
  At almost all finite places these are given by the local Hermitian Siegel series,
  with explicit formulas in \cite{Hironaka1998LocalZeta,Hironaka1999Spherical}.
  At the remaining places they depend on the chosen sections.
  We do not need a uniform formula for these factors.

\label{lem:level-big-cell-criterion}
  Let \(m\ge1\),
  and let \(x=\begin{pmatrix}
    A & B \\
    C & D
  \end{pmatrix} \in G_m(F_v)\).
  Then
  \[
    x\in P_{m,0}(F_v)K_{m,v}(\frakN_v)
  \]
  if and only if \(D\) is invertible and \(D^{-1}C\in\M_m(\frakN_v\calO_{E,v})\).
<!-- end lem-1 -->
<!-- proof-1: body_md -->
If \(x=pk\) with \(p\in P_{m,0}(F_v)\) and \(k=
  \begin{pmatrix}
    a & b \\
    c & d
  \end{pmatrix}\in K_{m,v}(\frakN_v)\),
  then \(d\equiv I_m\pmod{\frakN_v}\),
  so \(d\) is invertible.
  Since the lower blocks of \(x\) are obtained from \((c,d)\) by the same invertible Levi factor,
  \[
    D^{-1}C=d^{-1}c\in\M_m(\frakN_v\calO_{E,v}).
  \]

  Conversely,
  suppose that \(D\) is invertible and
  \[
    U=D^{-1}C\in\M_m(\frakN_v\calO_{E,v}).
  \]
  From \(DC^*=CD^*\),
  we have \(U^*=U\),
  and hence
  \[
    \overline n(-U) = \begin{pmatrix} I_m & 0 \\ -U & I_m \end{pmatrix} \in K_{m,v}(\frakN_v).
  \]
  Since the lower-left block of \(x\overline n(-U)\) is zero,
  \[
    x\overline n(-U)\in P_{m,0}(F_v),
  \]
  proving the converse.

Put
  \[
    Z_v(g_1,s)
    =
    \int_{G_r(F_v)} \varepsilon_{n+r,v}^{\frakN_v}
    \bigl( \xi_r \iota_{n,r}(g_1,g_2) \xi_r^{-1}, s;\chi_v \bigr)
    \overline{\varphi_v^\dagger(g_2)} \,dg_2.
  \]

  We first determine the support and value of the section at \(\xi_r\iota_{n,r}(s_{n,r}(g),I_{2r})\xi_r^{-1}\).
  Write
  \[
    g=\begin{pmatrix}a&b\\c&d\end{pmatrix}\in G_r(F_v).
  \]
  With respect to the decomposition \(n=(n-r)+r\),
  the complementary blocks of size \(n-r\) are identity or zero blocks,
  and the nontrivial \(4r\times4r\) block of \(\xi_r \iota_{n,r}(s_{n,r}(g),I_{2r}) \xi_r^{-1}\)
  is
  \[
    \begin{pmatrix} a & -b & b & 0 \\ 0 & I_r & 0 & 0 \\ c & I_r-d & d & 0 \\ a-I_r & -b & b & I_r \end{pmatrix}.
  \]
  Its lower blocks are
  \[
    C_0= \begin{pmatrix} c & I_r-d \\ a-I_r & -b \end{pmatrix}, \qquad D_0= \begin{pmatrix} d & 0 \\ b & I_r \end{pmatrix}.
  \]
  By Lemma~\ref{lem:level-big-cell-criterion},
  the section is supported precisely when \(d\) is invertible and
  \[
    D_0^{-1}C_0 = \begin{pmatrix} d^{-1}c & d^{-1}-I_r \\ a-I_r-bd^{-1}c & -bd^{-1} \end{pmatrix} \in\M_{2r}(\frakN_v\calO_{E,v}).
  \]
  This is equivalent to \(a\equiv d\equiv I_r, b\equiv c\equiv0 \pmod{\frakN_v}\),
  hence to \(g\in K_{r,v}(\frakN_v)\).
  If \(g\in K_{r,v}(\frakN_v)\),
  then \(\xi_r\iota_{n,r}(s_{n,r}(g),I_{2r})\xi_r^{-1}\in K_{n+r,v}(\frakN_v)\).
  By the right \(K_{n+r,v}(\frakN_v)\)-invariance of the section and its normalization at the identity,
  we obtain
  \[
    \varepsilon_{n+r,v}^{\frakN_v} \bigl(\xi_r\iota_{n,r}(s_{n,r}(g),I_{2r})\xi_r^{-1},s;\chi_v\bigr) =\varepsilon_{n+r,v}^{\frakN_v}(1,s;\chi_v) =1.
  \]

  Thus
  \[
    \varepsilon_{n+r,v}^{\frakN_v}
    \bigl(\xi_r\iota_{n,r}(s_{n,r}(g),I_{2r})\xi_r^{-1},s;\chi_v\bigr)
    =\mathbbm1_{K_{r,v}(\frakN_v)}(g).
  \]

  By the calculation of the stabilizer of the open orbit in \cite[Fact~19(2)]{Takeda2026PullbackFormula} and the description of the support above,
  \(Z_v(g_1,s)\) is supported on \(P_{n,r}(F_v)K_{n,v}(\frakN_v)\).
  It has the same \(P_{n,r}(F_v)\)-equivariance and right \(K_{n,v}(\frakN_v)\)-invariance as \(\Psi_{n,r,v}^{\frakN_v}(\varphi_v;g_1,s;\chi_v)\).
  It therefore suffices to take \(g_1=s_{n,r}(h)\).
  The change of variables and the transformation law of the section yield
  \begin{align*}
    Z_v(s_{n,r}(h),s)
     & =\int_{G_r(F_v)}
    \varepsilon_{n+r,v}^{\frakN_v}
    \bigl(\xi_r\iota_{n,r}(s_{n,r}(h),(hg^{-1})^\natural)\xi_r^{-1},s;\chi_v\bigr)
    \overline{\varphi_v^\dagger((hg^{-1})^\natural)}\,dg   \\
     & =\int_{G_r(F_v)}
    \mathbbm1_{K_{r,v}(\frakN_v)}(g)\varphi_v(hg^{-1})\,dg \\
     & =\int_{K_{r,v}(\frakN_v)}\varphi_v(hg^{-1})\,dg     \\
     & =\vol(K_{r,v}(\frakN_v))
    \Psi_{n,r,v}^{\frakN_v}(\varphi_v;s_{n,r}(h),s;\chi_v).
  \end{align*}

<!-- thm-1: body_md -->
\label{thm:global-pullback}
  Assume \(\mu\ge n+r\).
  If \(\mu=n+r+1\) and \(F=\bbQ\), assume \(\chi_0\ne\chi_{E/F}^{n+r+1}\).
  If \(\mu=n+r\) and \(n=r+1\), assume that \(L^{(\frakN)}(1,f,\St)\) is holomorphic and nonzero.
  Then the Klingen--Eisenstein series \([f]_{\kappa,\nu,\frakN}^n(g,s;\chi)\) is regular at \(s=\mu/2\),
  and
  \[
    ( \calE_{\frakN}(g,*), f^\dagger )_{G_r}
    =
    C_{n+r,\kappa,\nu}^{(\frakN)}(f;\chi)
    [f]_{\kappa,\nu,\frakN}^n(g;\chi).
  \]
<!-- end thm-1 -->
<!-- proof-3: body_md -->
For \(\Re(s)>n+r\),
  unfold the pullback integral as in \cite[\S\S~5.2--5.3]{Takeda2026PullbackFormula}.
  Cuspidality eliminates all but the open orbit,
  and the local calculations there identify the remaining integral with a scalar multiple of \([f]_{\kappa,\nu,\frakN}^n(g,s;\chi)\).
  This identity extends meromorphically in \(s\).

  Put \(m=n+r\).
  Since \(\mu\ge m\),
  \cite[Proposition~17.7]{Shimura2000Arithmeticity}
  gives regularity of the degree-\(m\) Siegel Eisenstein series at \(s=\mu/2\),
  except at \(\mu=m+1\) with the exceptional character.
  This is exactly the case excluded in the statement.
  Its Petersson pairing with \(f^\dagger\) is therefore regular at \(\mu/2\),
  since \(f^\dagger\) is rapidly decreasing.

  At \(s=\mu/2\),
  Propositions~\ref{prop:unramified-local-pullback},
  \ref{prop:arche-local-pullback},
  and \ref{prop:principal-local-pullback} identify the scalar obtained from the local integrals with the nonzero value
  \(C_{n+r,\kappa,\nu}^{(\frakN)}(f;\chi)\) in \eqref{eq:C-explicit-formula}.
  Dividing the continued identity by this scalar proves regularity of \([f]_{\kappa,\nu,\frakN}^n(g,s;\chi)\) at \(\mu/2\).
  Evaluation at \(\mu/2\) gives the asserted formula.

\label{lem:fourier-coefficient-transformation}
  Suppose that
  \[
    t_{n,0}(h)\gamma
    =
    t_{n,0}(a^{-1})n_n(Z)\eta k,
  \]
  with \(a\in\GL_n(E)\),
  \(Z\in\Her_n(E)\),
  \(\eta\in\calC^{(n)}(\frakN)\),
  and \(k\in K_n(\frakN)\).
  If \(S=(a^*)^{-1}Ta^{-1}\),
  then
  \[
    A_{F_0}(t_{n,0}(h)\gamma,T)
    =
    \rho_S[a,Z]A_{F_0}(\eta,S).
  \]
  In particular,
  \begin{equation}\label{eq:fourier-coefficient-transformation}
    A_{F_0}(t_{n,0}(h_M)\gamma,T)
    =
    \rho_{T_M}[a_M,Z_M]A_{F_0}(\gamma[M],T_M).
  \end{equation}
  For fixed \(F_0\),
  the common value in \eqref{eq:fourier-coefficient-transformation} depends only on \(\gamma\),
  \(T\),
  and \(M\).

<!-- proof-4: body_md -->
The first identity follows from the Fourier coefficient transformation formula in \cite[Section~2.2]{Takeda2026Congruences},
  with \(\alpha=a^{-1}\) and \(X_0=a^{-1}Z(a^*)^{-1}\).
  Taking \((h,a,Z,\eta)=(h_M,a_M,Z_M,\gamma[M])\) gives \eqref{eq:fourier-coefficient-transformation}.

  For fixed \(h_M\),
  \eqref{eq:fourier-coefficient-transformation} shows that the value is independent of the parabolic decomposition.
  For another choice \(h_M'\),
  the element \(u=(h_M')^{-1}h_M\) preserves \(L_{\gamma,\frakN}^{(n)}\otimes\widehat{\calO}_E\) and satisfies \(u_v=I_n\) for \(v\mid\frakN\).

    =A_{F_0}(t_{n,0}(h_M')\gamma k,T)
    =A_{F_0}(t_{n,0}(h_M')\gamma,T).
  \]

\label{lem:primitive-reconstruction}
  For \(M\in\calP_\gamma(T)\), the map
  \[
    \begin{aligned}
      \{N\in\calP_\gamma(T):M\subseteq N\}
        & \xrightarrow{\sim}
      \calP_{\gamma[M]}(T_M), \\
      N & \longmapsto a_MN
    \end{aligned}
  \]
  is an isomorphism of posets.
  Moreover,
  \begin{equation}\label{eq:primitive-reconstruction}
    A_{F_0}(\gamma,T)
    =
    \sum_{M\in\calP_\gamma(T)}
    \rho_{T_M}[a_M,Z_M]
    A_{F_0}^{\prim}(\gamma[M],T_M).
  \end{equation}
<!-- end lem-3 -->
<!-- proof-5: body_md -->
Put \(L=L_{\gamma,\frakN}^{(n)}\) and \(P=\calP_\gamma(T)\).
  Since \(a_MM=L_{\gamma[M],\frakN}^{(n)}\) and every \(x\in E^n\) satisfies \((a_Mx)^*T_M(a_Mx)=x^*Tx\),
  the map \(N\mapsto a_MN\) gives the asserted poset isomorphism.
  It therefore identifies \(\mu_{\calP_{\gamma[M]}(T_M)}(a_MM,a_MN)\) with \(\mu_P(M,N)\).
  Write \(P_M=\calP_{\gamma[M]}(T_M)\),
  and choose \(h_{M,N'}\) for each lattice \(N'\in P_M\) at the cusp \(\gamma[M]\).
  Substituting the definition and then setting \(N'=a_MN\),
  we obtain
  \begin{align*}
     & \sum_{M\in P}
    \rho_{T_M}[a_M,Z_M]A_{F_0}^{\prim}(\gamma[M],T_M)               \\
     & \quad=\sum_{M\in P}\rho_{T_M}[a_M,Z_M]
    \sum_{N'\in P_M}\mu_{P_M}(a_MM,N')
    A_{F_0}(t_{n,0}(h_{M,N'})\gamma[M],T_M)                         \\
     & \quad=\sum_{M\in P}\sum_{\substack{N\in P                    \\M\subseteq N}}
    \mu_P(M,N)\rho_{T_M}[a_M,Z_M]
    A_{F_0}(t_{n,0}(h_{M,a_MN})\gamma[M],T_M)                       \\
     & \quad=\sum_{M\in P}\sum_{\substack{N\in P                    \\M\subseteq N}}
    \mu_P(M,N)A_{F_0}(t_{n,0}(h_N)\gamma,T)                         \\
     & \quad=\sum_{N\in P}
    \left(\sum_{L\subseteq M\subseteq N}\mu_P(M,N)\right)
    A_{F_0}(t_{n,0}(h_N)\gamma,T)                                   \\
     & \quad=\sum_{N\in P}\delta_{L,N}A_{F_0}(t_{n,0}(h_N)\gamma,T) \\
     & \quad=A_{F_0}(\gamma,T).
  \end{align*}
  The third equality follows by applying \eqref{eq:fourier-coefficient-transformation} first to \(M\in\calP_\gamma(T)\) and then to \(a_MN\in\calP_{\gamma[M]}(T_M)\).
  The last two equalities use the M\"obius identity and the choice \(h_L=I_n\).

Put \(L=L_{\gamma,\frakN}^{(n)}\) and
  \(\calP=\calP_\gamma(T)\).
  For \(\Re(s)>n+r\), set
  \[
    \widetilde E_{n+r,\kappa,\nu,\frakN}^{[n],\theta}(g,s;\chi)
    =
    \bbL^{(\frakN)}(n,\mu;\chi)
    E_{n+r,\kappa,\nu,\frakN}
    (g\theta_{n,r}^{-1},s;\chi)
  \]
  and
  \[
    \calE_{\frakN}(g,h;s)
    =
    (2\pi\sqrt{-1})^{-d_{\bfk,\bfl}}
    \bbD_{\bfk,\bfl}
    \widetilde E_{n+r,\kappa,\nu,\frakN}^{[n],\theta}
    (\iota_{n,r}(g,h),s;\chi).
  \]
  Thus
  \(\calE_{\frakN}(g,h;\mu/2)=\calE_{\frakN}(g,h)\).

  By Theorem~\ref{thm:global-pullback} and
  Definition~\ref{def:primitive-coefficient},
  \begin{align}
     & C_{n+r,\kappa,\nu}^{(\frakN)}(f;\chi)
    A_{\Kl}^{\prim}(\gamma,T)
    \notag                                   \\
     & =
    e^{2\pi\operatorname{tr}_\infty(T)}
    \sum_{M\in\calP}
    \mu_\calP(L,M)
    \int_{[\Her_n]}
    \bigl(
    \calE_{\frakN}
    (n_n(U)t_{n,0}(h_M)\gamma,*;\mu/2),
    f^\dagger
    \bigr)_{G_r}
    \overline{\psi_F(\operatorname{tr}(TU))}
    \,dU.
    \label{eq:primitive-pullback-sum}
  \end{align}

  We use the global pullback identity after continuation to \(s=\mu/2\).
  At this point the differentiated pullback kernel is holomorphic and of moderate growth.
  Since \(f^\dagger\) is cuspidal, the Petersson integral converges absolutely.
  The \(U\)-integration is over a compact quotient,
  and the Fourier expansions below may be inserted first on truncated Siegel sets and then integrated term by term;
  the limit follows from the rapid decay of \(f^\dagger\).

  For \(\delta\in\calG^{(r)}(\frakN)\), put
  \[
    \fraku_\delta
    =
    \fraku_\delta(\frakN)
    =
    \{V\in\Her_r(E):
    n_r(V)\in G_r(F)\cap
    \delta K_r(\frakN)\delta^{-1}\}.
  \]
  Let
  \[
    \calA_\delta
    =
    \{a\in\GL_r(E):
    t_{r,0}(a)\in
    G_r(F)\cap\delta K_r(\frakN)\delta^{-1}\},
  \]
  let \(Z_\delta\) be the kernel of the action
  \(Y\mapsto aYa^*\) of \(\calA_\delta\) on
  \((\Her_r(\bbC)_{>0})^\bfa\), put
  \(e_\delta=\#Z_\delta\), and choose a fundamental domain
  \(\calF_\delta\) for \(\calA_\delta/Z_\delta\).

  Fix \(M\in\calP\), and put
  \[
    \Xi_{M,\delta}
    =
    \xi_r\iota_{n,r}
    \bigl(t_{n,0}(h_M)\gamma,\delta\bigr)\xi_r^{-1}.
  \]

  By \cite[Fact~19 and Proposition~20]{Takeda2026PullbackFormula},
  after unfolding and pairing with \(f^\dagger\),
  only the open orbit represented by \(\xi_r\) contributes.
  We first describe the additive quotient occurring in this orbit.
  Put
  \(J=\widetilde I_r
  =
  \begin{pmatrix}
    0 \\
    I_r
  \end{pmatrix}
  \in\M_{n,r}(E)\).
  For \(U_0\in\Her_n\) and \(V_0\in\Her_r\), set
  \[
    S=
    \begin{pmatrix}
      0   & J \\
      J^* & 0
    \end{pmatrix},
    \qquad
    W=
    \begin{pmatrix}
      U_0 & 0   \\
      0   & V_0
    \end{pmatrix}.
  \]
  Direct multiplication gives
  \[
    \xi_r
    \iota_{n,r}\bigl(n_n(U_0),n_r(V_0)\bigr)
    \xi_r^{-1}
    =
    \begin{pmatrix}
      I-WS & W    \\
      -SWS & I+SW
    \end{pmatrix},
  \]
  where \(SWS
  =
  \begin{pmatrix}
    JV_0J^* & 0       \\
    0       & J^*U_0J
  \end{pmatrix}\).
  Hence
  \[
    \xi_r
    \iota_{n,r}\bigl(n_n(U_0),n_r(V_0)\bigr)
    \xi_r^{-1}
    \in P_{n+r,0}
  \]
  if and only if \(V_0=0\), \(J^*U_0J=0\).
  Writing
  \[
    U_0=
    \begin{pmatrix}
      U_{11}   & U_{12} \\
      U_{12}^* & U_{22}
    \end{pmatrix},
  \]
  the condition \(J^*U_0J=0\) is equivalent to \(U_{22}=0\).
  Therefore the map \(n_n(U_0)\longmapsto U_{22}\)
  identifies \((N_n\cap P_{n,r})\backslash N_n\simeq\Her_r\).
  We write \(C\) for this quotient coordinate.
  Accordingly, the corresponding adelic Fourier integration is over \(C\in{[\Her_r]}\).

  Write \(Z=V+\sqrt{-1}Y\in\prod_{v\in\bfa}\calH_r\) with \(V\in(\Her_r(\mathbf R))^\bfa\) and \(Y\in(\Her_r(\mathbf R)_{>0})^\bfa\),
  and
  \[
    R=
    \begin{pmatrix}
      A   & B \\
      B^* & Q
    \end{pmatrix}
    \in\Her_{n+r}(E).
  \]
  On the component represented by \(\delta\), we have
  \[
    f_\delta^\dagger(V+\sqrt{-1}Y)
    =
    \sum_{Q'>0}
    A_{f^\dagger}(\delta,Q')\,
    e^{2\pi\sqrt{-1}\operatorname{tr}_\infty(Q'V)}
    e^{-2\pi\operatorname{tr}_\infty(Q'Y)}.
  \]
  The \(R\)-th Fourier term of the differentiated
  degree-\((n+r)\) Siegel Eisenstein series is
  \[
    A_{\widetilde E_{n+r,\kappa,\nu,\frakN}^{[n],\theta}}
    (\Xi_{M,\delta},R)\,
    P_{\mu;\bfk,\bfl}^{(n,r)}(R)
    e^{2\pi\sqrt{-1}\operatorname{tr}_\infty(AU+QV)}
    e^{-2\pi\operatorname{tr}_\infty(A+QY)},
  \]
  together with its character on the remaining open-orbit additive
  quotient.

  For \(R=
  \begin{pmatrix}
    A   & B \\
    B^* & Q
  \end{pmatrix}\),
  put \(X=B^*T^{-1}\).
  For \(C\in\Her_r(\bbA_E)\),
  define
  \[
    \calC_{T,B}(C)
    =
    \begin{pmatrix}
      X^*CX & -X^*C \\
      -CX   & C
    \end{pmatrix}.
  \]
  Then
  \begin{align}
    \operatorname{tr}\bigl(R\calC_{T,B}(C)\bigr)
     & =
    \operatorname{tr}\!\left(
    \bigl(
    Q-XTX^*+X(A-T)X^*
    \bigr)C
    \right)                                      \notag \\
     & =
    \operatorname{tr}\!\left(
    \bigl(
    Q-B^*T^{-1}B
    +B^*T^{-1}(A-T)T^{-1}B
    \bigr)C
    \right).
    \label{eq:open-orbit-C-character}
  \end{align}
  Substituting these Fourier expansions into
  \eqref{eq:primitive-pullback-sum}, and decomposing the Petersson
  integral into its \(\delta\)-components, gives
  \begin{align}
     & C_{n+r,\kappa,\nu}^{(\frakN)}(f;\chi)
    A_{\Kl}^{\prim}(\gamma,T)
    \notag                                   \\
     & =
    \sum_{M\in\calP}
    \mu_\calP(L,M)
    \sum_{\delta\in\calG^{(r)}(\frakN)}
    \frac{\vol(K_r(\frakN))}{e_\delta}
    \sum_R\sum_{Q'>0}
    e^{2\pi\operatorname{tr}_\infty(T)}
    e^{-2\pi\operatorname{tr}_\infty(A)}
    \notag                                   \\
     & \quad\times
    A_{\widetilde E_{n+r,\kappa,\nu,\frakN}^{[n],\theta}}
    (\Xi_{M,\delta},R)
    P_{\mu;\bfk,\bfl}^{(n,r)}(R)
    \notag                                   \\
     & \quad\times
    \int_{{[\Her_n]}}
    \int_{{[\Her_r]}}
    \psi_F\!\left(
    \operatorname{tr}\bigl((A-T)(U+X^*CX)\bigr)
    +
    \operatorname{tr}\bigl((Q-XTX^*)C\bigr)
    \right)
    \,dC\,dU
    \notag                                   \\
     & \quad\times
    \int_{\calF_\delta}
    \int_{\fraku_\delta\backslash(\Her_r(\bbC))^\bfa}
    e^{2\pi\sqrt{-1}\operatorname{tr}_\infty((Q-Q')V)}
    e^{-2\pi\operatorname{tr}_\infty((Q+Q')Y)}
    \notag                                   \\
     & \qquad\qquad\times
    \left(
    1\otimes
    \left\langle
    \rho_r'
    \bigl((Y_v,{}^tY_v)_{v\in\bfa}\bigr)\,\cdot,
    A_{f^\dagger}(\delta,Q')
    \right\rangle_{\sigma_r'}
    \right)
    \,dV\,d\mu_r(Y),
    \label{eq:primitive-full-Fourier-substitution}
  \end{align}
  where \(d\mu_r(Y) = \prod_{v\in\bfa} \det(Y_v)^{-2r}\,dY_v\).

  For fixed \(C\), the translation \(U\mapsto U+X^*CX\) preserves the Haar measure on \({[\Her_n]}\).
  Hence Fourier orthogonality in \(U\) gives
  \[
    A=T.
  \]
  The remaining \(C\)-character is therefore \(C\mapsto \psi_F\!\left(\operatorname{tr}\bigl((Q-XTX^*)C\bigr)\right)\),
  and Fourier orthogonality in \(C\) gives
  \[
    Q=T[X^*].
  \]

  Finally, Fourier orthogonality in \(V\) forces
  \[
    Q'=Q=T[X^*].
  \]
  Since \(Q'>0\) and \(T>0\), this is equivalent to \(\rank_E(X)=r\).

  Hence, if we put
  \[
    c_\delta
    =
    \frac{\vol(K_r(\frakN))}{e_\delta}
    \covol_{dV}(\fraku_\delta),
  \]
  then \eqref{eq:primitive-full-Fourier-substitution} becomes
  \begin{align}
     & C_{n+r,\kappa,\nu}^{(\frakN)}(f;\chi)
    A_{\Kl}^{\prim}(\gamma,T)
    \notag                                   \\
     & =
    \sum_{M\in\calP}\mu_\calP(L,M)
    \sum_{\delta\in\calG^{(r)}(\frakN)}
    c_\delta
    \sum_{\substack{
    X\in\M_{r,n}(E)                          \\
        \rank_E(X)=r
      }}
    A_{\widetilde E_{n+r,\kappa,\nu,\frakN}^{[n],\theta}}
    (\Xi_{M,\delta},R_{T,X})
    \notag                                   \\
     & \quad\times
    \int_{\calF_\delta}
    e^{-4\pi\operatorname{tr}_\infty(T[X^*]Y)}
    \left(
    1\otimes
    \left\langle
    \rho_r'
    \bigl((Y_v,{}^tY_v)_{v\in\bfa}\bigr)\,\cdot,
    A_{f^\dagger}\bigl(\delta,T[X^*]\bigr)
    \right\rangle_{\sigma_r'}
    \right)
    P_{T,X}\,
    d\mu_r(Y).
    \label{eq:primitive-after-Fourier-orthogonality}
  \end{align}

  Moreover,
  \[
    (2\pi\sqrt{-1})^{-d_{\bfk,\bfl}}
    \bbD_{\bfk,\bfl}
    e^{2\pi\sqrt{-1}\operatorname{tr}_\infty(R_{T,X}Z)}
    =
    P_{T,X}
    e^{2\pi\sqrt{-1}\operatorname{tr}_\infty(R_{T,X}Z)}.
  \]

  Fix \(M\in\calP\), \(\delta\in\calG^{(r)}(\frakN)\),
  and a Fourier index \(R_{T,X}\) occurring in \eqref{eq:primitive-after-Fourier-orthogonality}.
  Put \(L_\delta=L_{\delta,\frakN}^{(r)}\).
  Since \(R_{T,X}
    =
    {}^t\!\begin{pmatrix}
    I_n & X
  \end{pmatrix}
  T
  \begin{pmatrix}
    I_n & X^*
  \end{pmatrix}\),
  we have
  \[
    \begin{pmatrix}x^*&y^*\end{pmatrix}
    R_{T,X}
    \begin{pmatrix}x'\\y'\end{pmatrix}
    =
    (x+X^*y)^*T(x'+X^*y')
  \]
  for \(x,x'\in M\) and \(y,y'\in L_\delta\).
  Hence the map
  \begin{eqnarray*}
    M\oplus L_\delta&\longrightarrow& N,\\(x,y)&\longmapsto& x+X^*y
  \end{eqnarray*}
  induces
  \begin{equation}\label{eq:primitive-quotient-lattice}
    \frac{M\oplus L_\delta}
    {(M\oplus L_\delta)\cap\ker R_{T,X}}
    \cong
    M+X^*L_\delta,
  \end{equation}
  under which the nondegenerate Hermitian form induced by
  \(R_{T,X}\) is identified with \(T\).

  We therefore put
  \[
    N=M+X^*L_\delta.
  \]
  If the corresponding finite-adelic Fourier coefficient is nonzero,
  then \(R_{T,X}\) is integral on \(M\oplus L_\delta\),
  and hence \(\operatorname{Tr}_{E/F}(z^*Tz')\in\frakd_F^{-1}\) for \(z,z'\in N\).
  Since \(L\subseteq M\subseteq N\),
  this implies \(N\subseteq L_T^\sharp\).
  For \(v\mid\frakN\), we have \(M_v=L_v\) by the definition of \(\calP_\gamma(T)\).
  Moreover,
  the local Fourier coefficient attached to \(R_{T,X,v}\) vanishes unless \(X^*L_{\delta,v}\subseteq L_v\).
  Hence \(N_v=M_v+X^*L_{\delta,v}=L_v\) for \(v\mid\frakN\).
  Consequently,
  \[
    N\in\calP_\gamma(T).
  \]

  By the singular Fourier coefficient formula \cite[\S\S18--19]{Shimura1997EulerProducts},
  applied to \eqref{eq:primitive-quotient-lattice},
  the nondegenerate quotient is the lattice \(N\) equipped with the Hermitian form \(T\).
  With the normalization of \(\widetilde E_{n+r,\kappa,\nu,\frakN}^{[n],\theta}\) fixed above,
  the formula gives
  \begin{equation}\label{eq:primitive-singular-coefficient}
    A_{\widetilde E_{n+r,\kappa,\nu,\frakN}^{[n],\theta}}
    \left(
    \Xi_{M,\delta},
    R_{T,X}
    \right)
    =
    A_{\Si}
    \bigl(t_{n,0}(h_N)\gamma,T\bigr).
  \end{equation}

  We next sum over \(M\).
  Put \(K=X^*L_\delta\).
  For \(M\in\calP\) with \(M+K\in\calP\),
  Lemma~\ref{lem:primitive-reconstruction},
  applied to the upper interval above \(M+K\),
  gives
  \begin{equation}\label{eq:primitive-upper-reconstruction}
    A_{\Si}
    \bigl(t_{n,0}(h_{M+K})\gamma,T\bigr)
    =
    \sum_{\substack{N'\in\calP\\M+K\subseteq N'}}
    (\tau_{\kappa,\nu})_{T_{N'}}[a_{N'},Z_{N'}]
    A_{\Si}^{\prim}(\gamma[N'],T_{N'}).
  \end{equation}
  Hence
  \begin{align}
     & \sum_{\substack{M\in\calP \\M+K\in\calP}}
    \mu_\calP(L,M)
    A_{\Si}
    \bigl(t_{n,0}(h_{M+K})\gamma,T\bigr)
    \notag                       \\
     & =
    \sum_{N'\in\calP}
    (\tau_{\kappa,\nu})_{T_{N'}}[a_{N'},Z_{N'}]
    A_{\Si}^{\prim}(\gamma[N'],T_{N'})
    \mathbf1_{K\subseteq N'}
    \sum_{L\subseteq M\subseteq N'}
    \mu_\calP(L,M)
    \notag                       \\
     & =
    \mathbf1_{K\subseteq L}
    (\tau_{\kappa,\nu})_{T_L}[a_L,Z_L]
    A_{\Si}^{\prim}(\gamma[L],T_L)
    \notag                       \\
     & =
    \mathbf1_{K\subseteq L}
    A_{\Si}^{\prim}(\gamma,T),
    \label{eq:primitive-mobius-cancellation}
  \end{align}
  where \(\mathbf1_{K_1\subseteq K_2}\) denotes the indicator of the condition \(K_1\subseteq K_2\).
  Here the second equality uses
  \[
    \sum_{L\subseteq M\subseteq N'}
    \mu_\calP(L,M)
    =
    \delta_{L,N'}.
  \]
  Thus the M\"obius sum leaves the condition \(X^*L_\delta\subseteq L\).
  Together with the condition \(R_{T,X}\in\Lambda_{\Xi_{\gamma,\delta}}(\frakN)\) already imposed on the Fourier index,
  this gives \(X\in\calX_{\gamma,\delta}(T)\).
  Therefore \eqref{eq:primitive-after-Fourier-orthogonality} becomes
  \begin{align}
     & C_{n+r,\kappa,\nu}^{(\frakN)}(f;\chi)
    A_{\Kl}^{\prim}(\gamma,T)
    \notag                                   \\
     & =
    A_{\Si}^{\prim}(\gamma,T)
    \sum_{\delta\in\calG^{(r)}(\frakN)}
    c_\delta
    \sum_{\substack{
    X\in\calX_{\gamma,\delta}(T)             \\
        \rank_E(X)=r
      }}
    \int_{\calF_\delta}
    e^{-4\pi\operatorname{tr}_\infty(T[X^*]Y)}
    \notag                                   \\
     & \qquad\qquad\times
    \left(
    1\otimes
    \left\langle
    \rho_r'
    \bigl((Y_v,{}^tY_v)_{v\in\bfa}\bigr)\,\cdot,
    A_{f^\dagger}\bigl(\delta,T[X^*]\bigr)
    \right\rangle_{\sigma_r'}
    \right)
    P_{T,X}\,
    d\mu_r(Y).
    \label{eq:primitive-after-mobius}
  \end{align}

  We now show that
  \begin{align}
    (\Phi_{\gamma,T},f^\dagger)_{G_r}= & \sum_{\delta\in\calG^{(r)}(\frakN)}
    c_\delta
    \sum_{\substack{
    X\in\calX_{\gamma,\delta}(T)                                             \\
        \rank_E(X)=r
      }}
    \int_{\calF_\delta}
    e^{-4\pi\operatorname{tr}_\infty(T[X^*]Y)}
    \notag                                                                   \\
                                       & \qquad\times
    \left(
    1\otimes
    \left\langle
    \rho_r'
    \bigl((Y_v,{}^tY_v)_{v\in\bfa}\bigr)\,\cdot,
    A_{f^\dagger}\bigl(\delta,T[X^*]\bigr)
    \right\rangle_{\sigma_r'}
    \right)
    P_{T,X}\,
    d\mu_r(Y).
    \label{eq:primitive-open-sum-after-mobius}
  \end{align}

  Put \(u_0=(\mu-n)/2\).
  For \(\Re(u)>r\), the Eisenstein series \(E_{\delta,\mathbf i}^{[n]}(\,\cdot\,,u)\) is absolutely convergent.
  Decomposing the Petersson pairing into its \(\delta\)-components and unfolding \(E_{\delta,\mathbf i}^{[n]}(\,\cdot\,,u)\),
  we obtain
  \begin{align}
     & \bigl(
    \Phi_{\gamma,T}(\,\cdot\,,u),
    f^\dagger
    \bigr)_{G_r}
    \notag    \\
     & =
    \sum_{\delta\in\calG^{(r)}(\frakN)}
    \frac{\vol(K_r(\frakN))}{e_\delta}
    \sum_{\mathbf i}
    \int_{P_{r,0}(F)\backslash G_r(\bbA_F)}
    \left\langle
    \Theta_{\gamma,T,\delta}^{(r)}
    (h;\phi_{\delta,\mathbf i})
    \varepsilon_{\delta,\mathbf i}^{[n]}(h,u),
    f^\dagger(h)
    \right\rangle_r
    \,dh.
    \label{eq:primitive-Phi-unfolding}
  \end{align}

  The identity obtained for \(\Re(u)>r\) continues to \(u=u_0\).
  Indeed, after pairing with the cusp form \(f^\dagger\),
  rapid decay gives absolute convergence at \(u_0\),
  and the unfolded integral represents the same continuation.
  At this point, using the Iwasawa decomposition on the
  \(\delta\)-component and the definition
  \eqref{eq:general-r-theta-definition}, the right-hand side becomes
  \begin{align}
     & \bigl(
    \Phi_{\gamma,T},
    f^\dagger
    \bigr)_{G_r}
    \notag          \\
     & =
    \sum_{\delta\in\calG^{(r)}(\frakN)}
    \frac{\vol(K_r(\frakN))}{e_\delta}
    \int_{\calF_\delta}
    \int_{\fraku_\delta\backslash(\Her_r(\bbC))^\bfa}
    \sum_{X\in\M_{r,n}(E)}
    \Biggl(
    \sum_{\mathbf i}
    \varepsilon_{\delta,\mathbf i}^{\bfh}
      (k_\bfh,u_0)
    \bigl(
      \omega_{T,\bfh}(k_\bfh)
      \phi_{\delta,\mathbf i}
      \bigr)(X)
    \Biggr)
    \notag          \\
     & \qquad\times
    e^{2\pi\sqrt{-1}\operatorname{tr}_\infty(T[X^*]V)}
    e^{-2\pi\operatorname{tr}_\infty(T[X^*]Y)}
    \notag          \\
     & \qquad\times
    \left\langle
    \rho_r'
    \bigl((Y_v,{}^tY_v)_{v\in\bfa}\bigr)
    P_{T,X},
    f_\delta^\dagger(V+\sqrt{-1}Y)
    \right\rangle_{\sigma_r'}
    \,dV\,d\mu_r(Y).
    \label{eq:primitive-Phi-Iwasawa}
  \end{align}
  By \eqref{eq:dual-basis-identity},
  \[
    \sum_{\mathbf i}
    \varepsilon_{\delta,\mathbf i}^{\bfh}
    (k_\bfh,u_0)
    \bigl(
    \omega_{T,\bfh}(k_\bfh)
    \phi_{\delta,\mathbf i}
    \bigr)(X)
    =
    \mathbbm1_{\calX_{\gamma,\delta}(T)}(X).
  \]
  Hence
  \begin{align}
     & \bigl(
    \Phi_{\gamma,T},
    f^\dagger
    \bigr)_{G_r}
    \notag          \\
     & =
    \sum_{\delta\in\calG^{(r)}(\frakN)}
    \frac{\vol(K_r(\frakN))}{e_\delta}
    \sum_{X\in\calX_{\gamma,\delta}(T)}
    \int_{\calF_\delta}
    \int_{\fraku_\delta\backslash(\Her_r(\bbC))^\bfa}
    e^{2\pi\sqrt{-1}\operatorname{tr}_\infty(T[X^*]V)}
    e^{-2\pi\operatorname{tr}_\infty(T[X^*]Y)}
    \notag          \\
     & \qquad\times
    \left\langle
    \rho_r'
    \bigl((Y_v,{}^tY_v)_{v\in\bfa}\bigr)
    P_{T,X},
    f_\delta^\dagger(V+\sqrt{-1}Y)
    \right\rangle_{\sigma_r'}
    \,dV\,d\mu_r(Y) \\
     & =
    \sum_{\delta\in\calG^{(r)}(\frakN)}
    \frac{\vol(K_r(\frakN))}{e_\delta}
    \sum_{X\in\calX_{\gamma,\delta}(T)}
    \sum_{Q'>0}
    \int_{\calF_\delta}
    \int_{\fraku_\delta\backslash(\Her_r(\bbC))^\bfa}
    e^{2\pi\sqrt{-1}
        \operatorname{tr}_\infty((T[X^*]-Q')V)}
    \notag          \\
     & \qquad\times
    e^{-2\pi\operatorname{tr}_\infty((T[X^*]+Q')Y)}
    \left(
    1\otimes
    \left\langle
    \rho_r'
    \bigl((Y_v,{}^tY_v)_{v\in\bfa}\bigr)\,\cdot,
    A_{f^\dagger}(\delta,Q')
    \right\rangle_{\sigma_r'}
    \right)
    P_{T,X}
    \,dV\,d\mu_r(Y).
    \label{eq:primitive-Phi-Fourier-substitution}
  \end{align}
  Fourier orthogonality in \(V\) gives
  \[
    Q'=T[X^*].
  \]
  Since \(Q'>0\) and \(T>0\), this is possible exactly when \(\rank_E(X)=r\).
  Moreover,
  \[
    \int_{\fraku_\delta\backslash(\Her_r(\bbC))^\bfa}
    1\,dV
    =
    \covol_{dV}(\fraku_\delta).
  \]
  Therefore,
  \begin{align}
    \bigl(
    \Phi_{\gamma,T},
    f^\dagger
    \bigr)_{G_r}
     & =
    \sum_{\delta\in\calG^{(r)}(\frakN)}
    c_\delta
    \sum_{\substack{
    X\in\calX_{\gamma,\delta}(T) \\
        \rank_E(X)=r
      }}
    \int_{\calF_\delta}
    e^{-4\pi\operatorname{tr}_\infty(T[X^*]Y)}
    \notag                       \\
     & \qquad\times
    \left(
    1\otimes
    \left\langle
    \rho_r'
    \bigl((Y_v,{}^tY_v)_{v\in\bfa}\bigr)\,\cdot,
    A_{f^\dagger}
    \bigl(\delta,T[X^*]\bigr)
    \right\rangle_{\sigma_r'}
    \right)
    P_{T,X}\,
    d\mu_r(Y).
