<!-- thm-1: body_md -->
\label{thm_main}
 Let ${\mathcal B}$ be a BLT-set of ${\mathcal Q}(4,q)$ whose full automorphism group $\operatorname{Aut}({\mathcal B})$ is non-solvable and acts transitively on ${\mathcal B}$. If $\operatorname{Aut}({\mathcal B})$ is reducible, then ${\mathcal B}$ is either a classical BLT-set or a Kantor semifield BLT-set.
<!-- end thm-1 -->
<!-- thm-2: body_md -->
\label{cor:full-classification}
Let ${\mathcal B}$ be a BLT-set of ${\mathcal Q}(4,q)$ whose full automorphism group is
non-solvable and transitive on ${\mathcal B}$. Then, up to equivalence, ${\mathcal B}$ is one
of the following:
\begin{enumerate}
\item a classical BLT-set;
\item a Kantor semifield BLT-set;
\item a Fisher--Thas--Walker BLT-set;
\item the Law--Penttila sporadic BLT-set over ${\mathbb F}_{29}$, with
$\operatorname{Aut}({\mathcal B})\cong S_6$;
\item the Law--Penttila sporadic BLT-set over ${\mathbb F}_{59}$, with
$\operatorname{Aut}({\mathcal B})\cong S_5$.
\end{enumerate}
<!-- end thm-2 -->
<!-- lemma-1: body_md -->
\label{lem_disc}
Let ${\mathcal B}$ be a set of at least three points of ${\mathcal Q}(4,q)$, where
$q$ is an odd prime power. Then ${\mathcal B}$ is a partial BLT-set if and only if, for every triple of distinct points $\langle\mathbf{x}\rangle,\langle\mathbf{y}\rangle,\langle\mathbf{z}\rangle$ of ${\mathcal B}$,
\[
-\frac{2B(\mathbf{x},\mathbf{y})B(\mathbf{x},\mathbf{z})B(\mathbf{y},\mathbf{z})}
{\operatorname{disc}(Q)}\in\blacksquare_q.

<!-- lemma-6: body_md -->
\label{lem_xisquare}
Let $\xi=\omega^{q-1}$, where $\omega$ is a primitive element of ${\mathbb F}_{q^2}$. Then for $1\leq k\leq q$,
	\[
	 \xi^{k}+\xi^{-k}-2=(\xi^{k/2}-\xi^{-k/2})^2\in\begin{cases}
	  \square_q,&\text{if $k$ is odd},\\
	  \blacksquare_q,&\text{if $k$ is even}.
	 \end{cases}
	\]
Here $\xi^{k/2}=\omega^{(q-1)k/2}$.
<!-- end lemma-6 -->
<!-- proof-4: body_md -->
Since $\xi=\omega^{q-1}$, we have $\xi^{(q+1)/2}=-1$ and
$\xi^{kq/2}=(-1)^{k}\xi^{-k/2}$. Thus
\[
(\xi^{k/2}-\xi^{-k/2})^q
=(-1)^{k+1}(\xi^{k/2}-\xi^{-k/2}).
\]
Put $u=\xi^{k/2}-\xi^{-k/2}$. Then $u\ne0$ because $1\leq k\leq q$.
If $k$ is odd, then $u^q=u$, so $u\in{\mathbb F}_q$ and $u^2\in\square_q$. If
$k$ is even, then $u^q=-u$, so
$(u^2)^{(q-1)/2}=u^{q-1}=-1$ and $u^2\in\blacksquare_q$.

<!-- thm-6: body_md -->
\label{thm_FFquadsum}

<!-- lemma-7: body_md -->
\label{lem_caseI_torus_sum}
{Let $\xi=\omega^{q-1}$ be as in Lemma \ref{lem_xisquare}, let $k$ be an
even divisor of $q+1$, and put  $Y:=\big\langle\xi^k\big\rangle$. Suppose that
$\alpha\in{\mathbb F}_{q^2}^{\times}$, $\beta\in{\mathbb F}_q$, and
$\beta^2-4\alpha\alpha^q\ne0$. Then
\[
\left|
\sum_{y\in Y}
\chi\bigl(\beta+\alpha y+\alpha^qy^{-1}\bigr)
\right|
\leq
{\frac{(2k-1)\sqrt q+1}{k}}
<
2\sqrt q.
\]}
<!-- end lemma-7 -->
<!-- proof-5: body_md -->
{The group $\langle\xi\rangle$ has order $q+1$.
For each $y\in\langle\xi\rangle$, we have $y^q=y^{-1}$, so
$\beta+\alpha y+\alpha^qy^{-1}\in{\mathbb F}_q$.} The map $\big\langle\xi\big\rangle\to Y$,
$x\mapsto x^k$, is $k$-to-one. Hence
\[
\sum_{y\in Y}\chi\bigl(\beta+\alpha y+\alpha^qy^{-1}\bigr)
=
\frac{1}{k}
\sum_{x\in\big\langle\xi \big\rangle}
\chi\bigl(\beta+\alpha x^k+\alpha^qx^{-k}\bigr).
\]

Choose $\delta\in{\mathbb F}_{q^2}^{\times}$ with
$\delta^q=-\delta$ and $\delta^2\in\blacksquare_q$. The map
\[
x(t)=\frac{t+\delta}{t-\delta},
\qquad t\in{\mathbb F}_q\cup\{\infty\},
\]
is a bijection onto $\big\langle\xi\big\rangle$, with $t=\infty$ corresponding to $x=1$.
Moreover,
$x(t)^q=x(t)^{-1}$ for all $t\in{\mathbb F}_q$. For each $t\in{\mathbb F}_q$, we have
\[
\beta+\alpha x(t)^k+\alpha^qx(t)^{-k}
=\frac{f(t)}{(t^2-\delta^2)^k},
\]
where
$f(t):=\alpha(t+\delta)^{2k}+\beta(t^2-\delta^2)^k
+\alpha^q(t-\delta)^{2k}$. Expanding the first and third terms gives
$\alpha(t+\delta)^{2k}+\alpha^q(t-\delta)^{2k}
=\sum_{i=0}^{2k}\binom{2k}{i}
\bigl(\alpha\delta^{2k-i}+\alpha^q(-\delta)^{2k-i}\bigr)t^i$,
and every coefficient lies in ${\mathbb F}_q$. Hence $f(t)\in{\mathbb F}_q[t]$. Since
$\delta^2\in \blacksquare_q$, $t^2-\delta^2\ne0$ for
$t\in{\mathbb F}_q$. Since $k$ is even, $(t^2-\delta^2)^k$ is a square. Therefore
$\chi(\beta+\alpha x(t)^k+\alpha^qx(t)^{-k})=\chi(f(t))$.


To use Weil's bound, it remains to determine the number of distinct roots of $f(t)$. Away from
$t=\pm\delta$, put $U=x(t)^k$. Then $f(t)=0$ is equivalent to
$\alpha U^2+\beta U+\alpha^q=0$. By hypothesis, its discriminant
$\beta^2-4\alpha^{q+1}$ is nonzero, so the equation has two distinct
nonzero roots. Since $k\mid (q+1)$, we have $p\nmid k$, and each root
therefore gives $k$ distinct solutions of $x(t)^k=U$.  Neither
$t=\delta$ nor $t=-\delta$ is a root of $f$. The point
$t=\infty$ corresponds to one of these $2k$ solutions precisely
when $\alpha+\beta+\alpha^q=0$. Thus the number $d$ of distinct finite roots of
$f$ is $2k$ if $\alpha+\beta+\alpha^q\ne0$, and $2k-1$ otherwise.
   
                
                                 
                               
              
All these roots are simple, so $f$ is not a square in ${\mathbb F}_q[t]$.
After multiplying $f$ by a nonzero element of ${\mathbb F}_q$ to make it monic, Theorem \ref{thm_FFquadsum} gives
\[
\left|\sum_{t\in{\mathbb F}_q}\chi(f(t))\right|
\leq(d-1)\sqrt q\leq(2k-1)\sqrt q.
\]
The point $t=\infty$ contributes
$\chi(\alpha+\beta+\alpha^q)$, whose absolute value is at most $1$.
Consequently,
\[
\left|
\sum_{x\in\big\langle\xi \big\rangle}\chi\bigl(\beta+\alpha x^k+\alpha^qx^{-k}\bigr)
\right|
\leq(2k-1)\sqrt q+1.
\]
Dividing by $k$ proves the first bound. Finally,
$2\sqrt q-((2k-1)\sqrt q+1)/k=(\sqrt q-1)/k>0$, which gives the desired
inequality.

\begin{enumerate}

 \item[(II)]$L=\{A\circ\sigma(A):A\in\textup{SL}_2(q_0)\}

We first exclude Case \textup{(I)}. Take $\langle(x,M)\rangle\in{\mathcal B}$. Put
\[
 u_1=\begin{pmatrix}1&1\\0&1\end{pmatrix},\qquad
 u_2=\begin{pmatrix}1&0\\1&1\end{pmatrix}.

$x=0$, then $M$

\end{equation}

 \lambda^{p^i+1}a&
 \lambda a z^{p^i}+\lambda^{1-p^i}b\\

\label{prop:A5-finite-verification}
Assume the hypotheses of Section \ref{sec:4} hold. Then the external-line case and Case \textup{(III)} of the hyperbolic case do not occur.

In each case, the two embedding classes are conjugate in $\textup{PGO}_5(q)$.
We therefore fix one representation $L$ in each model, with
$L\cong A_5$ in the external-line case and $L\cong2.A_5$
in the hyperbolic case. For $q=239$, an exhaustive Magma computation
tests the complete candidates, including compatibility between
distinct $\overline J$-orbits when $m=2$, and finds no BLT-set of size $240$.
For the remaining values of $q$, we use the following procedures,
implemented in Magma~\cite{Magma}.
\begin{description}
\setlength{\itemsep}{2pt}
\setlength{\parsep}{0pt}


\item[(i)] \emph{The external-line case with $q\ne239$.}
Consider the $20$ pairs $(q,k)$ in Proposition
\ref{prop_A5_scalar_bound} for which $(q+1)/k>2$. 
 Choose a representative $(u,\mathbf w)$ with $u^{q+1}=-2b_5\nu$ and $Q_{A_5}(\mathbf w)=2b_5\nu$. Fix a Klein four-group $\{1,\tau_1,\tau_2,\tau_3\}\leq H$. For each $i$, choose a nonzero vector $\mathbf w_i$ spanning the $(+1)$-eigenspace of $\tau_i$. Then these three eigenspaces form an orthogonal decomposition of $W$. Consequently, every admissible $\mathbf{w}$ determines
\[
a_i=
-\frac{Q_{A_5}(\mathbf{w}+\tau_i\mathbf{w})}{2b_5}
\qquad (1\leq i\leq3),
\]
where
\[
(a_1,a_2,a_3)
 \in(\mathcal A_{q,k}\cap\square_q)^3,
\qquad
a_1+a_2+a_3=-4\nu.
\]

\[
 -\frac{Q_{A_5}(\mathbf{w}+h\mathbf{w})}{2b_5}
 \in\mathcal A_{q,k}
\]
for every element $h\in H$ of order five.\\

\item[(ii)] \emph{The hyperbolic case \textup{(III)} with $q\ne239$.}
Choose
\[
 Q=\langle g_1,g_2\rangle\cong Q_8\leq L\cong 2.A_5,
 \qquad g_3=g_1g_2.
\]

\[
 (Mg_iM^{-1})_{11}=\delta t_i.
\]

\[

\]

\end{description}

external-line case nor Case \textup{(III)} occurs.
