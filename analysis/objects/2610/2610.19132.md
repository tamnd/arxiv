\label{thm:main}
Let $1<p<\infty$.  There exist real numbers
$a_{p,0},a_{p,1},\ldots$ such that, for every integer $M\geq0$,
\begin{equation}\label{eq:complete-zero-Poincare}
 \tau_n=n+\tfrac1p-\tfrac12+\sum_{k=0}^{M}\frac{a_{p,k}}{(n+\frac1p-\frac12)^{2k+1}}
 + O_M\left( \frac{1}{(n+\frac1p-\frac12)^{2M+3}}\right).
\end{equation}

<!-- lemma-2: body_md -->
\label{lem:Gamma-properties}
For \({x>\max\{0,\gamma\}}\),
\begin{equation}\label{eq:Gamma-phase}
 x^\alpha E_p(x)
 =\frac{\Gamma(1+\gamma)^2}{\pi} m_p(x)\sin\pi(x-\gamma).
\end{equation}
Moreover, for $x\ge 1$ and for every integer \(j\ge0\),
\begin{equation}\label{eq:m-symbol}
 \left|\frac{\,\mathrm d^j}{\,\mathrm d x^j}\{m_p(x)-1\}\right|
 \le C_{j,p}x^{-2-j}.
\end{equation}
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
The first assertion \eqref{eq:Gamma-phase} follows from the reflection formula for the gamma function. To arrive at \eqref{eq:m-symbol}, we use Fields's asymptotic expansion from \cite{Fields} (see also \cite{Frenzen} and 
\cite[Sec.~5.11(i)--(iii)]{DLMF}). The case $j=0$ is then immediate. Since Fields's expansion relies on a certain loop integral (see \cite[p. 43]{Fields}), we may differentiate it and thus obtain \eqref{eq:m-symbol} for all   $j>0$.

<!-- lemma-3: body_md -->
\label{lem:weighted-correction-estimates}
For every integer \(j\ge0\),
\begin{equation}\label{eq:weighted-Bernstein}
 \left\lVertf^{(j)}\right\rVert_\alpha\le C_{j,p}\left\lVertf\right\rVert_\alpha,
 \qquad f\in\mathcal X_\alpha.
\end{equation}
In addition,
\begin{equation}\label{eq:pointwiseH}
 |f(x)|
 \le C_p(1+x^2)^{-\alpha/2}\left\lVertf\right\rVert_\alpha.
\end{equation}
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
By a variant of a classical inequality of Plancherel and Pólya (see e.g. \cite[Section 2.3]{Young}),
\begin{equation} \label{eq:PP}
    \int_{-1}^1 \int_{-\infty}^\infty (1+|x+\,\mathrm i y|^2)^{\alpha}|f(x+\,\mathrm i y)|^2 \,\mathrm d x \,\mathrm d y \le C \| f\|_\alpha^2.\end{equation}
The first assertion follows by estimating $|f^{(j)}(x)|$ in terms of Cauchy estimates and then applying \eqref{eq:PP}. We get the second assertion by using subharmonicity to bound ${|f(x)|^2}$ by an area integral of $|f|^2$ in a disc of radius $1$ centered at $x$ and then invoking again \eqref{eq:PP}.

<!-- lemma-4: body_md -->
\label{lem:tail-dominance}
For every \(T>1\), there is a constant \(C_{p,T}>0\) such that 
\begin{equation}\label{eq:tail-dominance}
 \left\lVertf\right\rVert_\alpha^2
 \le C_{p,T}\int_{|x|>T}|x|^{2\alpha}|f(x)|^2\,\mathrm d x,
 \qquad f\in\mathcal X_\alpha.
\end{equation}
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
{If the assertion failed, there would be $f_j\in\mathcal X_\alpha$
with $\|f_j\|_\alpha=1$ and weighted tail integrals tending to zero.
Pass to a weakly convergent subsequence. By \eqref{eq:PP} and local
compactness it converges locally uniformly to an entire function $f$.
The vanishing tails force $f=0$ on $(T,T+1)$, hence $f\equiv0$.
The norm on $[-T,T]$ therefore tends to zero, as does the remaining
part since $(1+x^2)^\alpha\asymp_{p,T}|x|^{2\alpha}$ for $|x|>T$.
This contradicts $\|f_j\|_\alpha=1$.}

<!-- lemma-7: body_md -->
\label{prop:Gamma-defect}
The distribution
\[
 \Delta_p:=\widehat{x^2N_p(E_p)}
\]
is represented on \((-\pi,\pi)\) by a locally integrable function, still denoted \(\Delta_p\), satisfying
\begin{equation}\label{eq:defect-bound}
 |\Delta_p(\xi)|
 \le C_p\left\{1+(\pi-\xi)^{1-\alpha}
 +(\pi+\xi)^{1-\alpha}\right\}.
\end{equation}
Moreover,
\begin{equation}\label{eq:ell0}
 \ell_0(f)=\frac1{2\pi}
 \int_{-\pi}^{\pi}\Delta_p(\xi)\widehat f(\xi)\,\mathrm d\xi
\end{equation}
defines a bounded functional on \(\mathcal X_\alpha\).
<!-- end lemma-7 -->
<!-- proof-6: body_md -->
We may assume that \(p\ne2\) since trivially $\Delta_2|_{(-\pi,\pi)} =0$. By Lemma~\ref{lem:Gamma-properties}, for large positive \(x\),
\begin{equation}\label{eq:defect-tail-exact}
 x^2N_p(E_p(x))
 =D_p x^\alpha m_p(x)^{p-1}
 F_p\left(\pi x-\frac\pi p\right)
\end{equation}
with $D_p$ a constant. {We insert the cutoff $\chi(x)$ in the tail integrals below; the omitted compact part has a bounded Fourier transform.}
We insert \eqref{eq:cusp-series} into \eqref{eq:defect-tail-exact}, and our task is then to compute the Fourier transform of terms of the form $x^\alpha m_p(x)^{p-1}\cos(n\pi(x-\frac1p))$. Beginning with odd $n\ge 3$, we set
\[
 I_{n,\pm,\varepsilon}(\xi)
 \coloneqq \int_0^\infty
 {\chi(x)}x^\alpha m_p(x)^{p-1}e^{-\varepsilon x}
 e^{i(\pm n\pi-\xi)x}\,\mathrm d x.
\]
Let $q$ be an integer larger than $\alpha+1$. Since $|\pm n\pi - \xi|\ge(n-1)\pi$ when $|\xi|\le \pi$, $q$
integrations by parts on $[1,\infty]$ along with \eqref{eq:m-symbol} give
\[
 \sup_{0<\varepsilon\le1}\sup_{|\xi|\le\pi}
 |I_{n,\pm,\varepsilon}(\xi)|
 \le C_{q,p}.
\]
Taking into account the uniform bounds on $\beta_n(p)$ from \eqref{eq:cusp-series}, we conclude that the terms with odd $n\ge 3$ contribute at most a bounded term to $\Delta_p(\xi)$ when ${|\xi|\le\pi}$.

It remains to consider the contribution from the term with $n=1$.  Write \(m_p^{p-1}=1+s_p\).  By \eqref{eq:m-symbol},
\[
 |s_p^{(j)}(x)|\le C_{j,p}x^{-2-j}.
\]
Thus Lemma~\ref{lem:oddn} applies with $s=\alpha -2$ and so the term with $s_p$ yields a term bounded by a constant times the right-hand side of \eqref{eq:defect-bound}. Finally, to deal with the leading term when $n=1$, we see from {Lemma~\ref{lem:Abel-estimates}} with $s=\alpha$ that we may potentially have two singular terms behaving as $(\pi\mp\xi)^{-\alpha-1}$  near the respective endpoints $\xi=\pm \pi$. However, the coeffients of these terms will, according to \eqref{eq:Abel-monomial}, be 
 proportional to
\[
 \cos\left(\frac{\pi(\alpha+1)}2-\frac\pi p\right)
 =\cos\frac\pi2=0,
\]
because $\alpha=2/p$. In total, we have therefore established \eqref{eq:defect-bound}.

The final assertion, namely that $\ell_0$ defines a bounded functional on $\mathcal X_{\alpha}$, now follows from \eqref{eq:defect-bound}. For $\alpha<1$, we see this by noting that $\Delta_p$ is in $L^\infty$, and so its inverse Fourier transform is in $PW^2$. Hence the result follows from Plancherel's identity. For
$\alpha>1$, we use that $\Delta_p$ is in $L^1$ which implies that its inverse Fourier transform is bounded. Then applying Plancherel's identity again, we obtain the desired result because $\| f \|_{L^1(\mathbb R)} \le C \| f\|_{\alpha}$ when $\alpha>1$.

<!-- lemma-8: body_md -->
\label{lem:Bregman} We have
\begin{align}
 B_p(a,h)&\ge c_p\frac{|h|^2}{(|a|+|h|)^{2-p}}, \quad  B_p(a,h)\le C_p h^2|a|^{p-2}, &&1<p<2,
 \label{eq:Bregman-sub-lower}\\
 B_p(a,h)&\ge c_p|a|^{p-2}|h|^2, \qquad B_p(a,h)\le C_p\{|a|^{p-2}|h|^2+|h|^p\} &&p>2,
 \label{eq:Bregman-super-lower}
\end{align}
<!-- end lemma-8 -->
<!-- proof-7: body_md -->
These estimates follow from the formula
\begin{equation}\label{eq:Bpint}
 B_p(a,h)=(p-1)h^2\int_0^1(1-t)|a+th|^{p-2}\,\mathrm d t.
\end{equation}
To arrive at the first bound in \eqref{eq:Bregman-sub-lower} and the second bound in \eqref{eq:Bregman-super-lower}, we use that $|a+th|\le |a|+|h|$. To deal with the remaining two cases,  we write
\[B_p(a,h)=(p-1)h^2 |a|^{p-2} \int_0^1(1-t)|1+th/a|^{p-2}\,\mathrm d t \]
and note that the function $x\mapsto \int_0^1(1-t)|1+tx|^{p-2}\,\mathrm d t$ has a finite maximum when $1<p<2$ and a positive minimum when $p>2$.

<!-- lemma-10: body_md -->
\label{prop:Euler-solution}
For every $1<p<\infty$, $p\ne2$, the functional $\mathcal J$ has a unique minimizer $H_*$.  Moreover, for every $f$ in $\mathcal X_\alpha$,
\begin{equation}\label{eq:weak-correction}
 \ell_0(f)+\int_\mathbb R x^2
 \{N_p(E_p(x)+H_*(x))-N_p(E_p(x))\}f(x) \,\mathrm d x=0.
\end{equation}
<!-- end lemma-10 -->
<!-- proof-9: body_md -->
Write
\[
 \mathcal I(H)\coloneqq\int_\mathbb R x^2B_p(E_p(x),H(x))\,\mathrm d x,
 \qquad \mathcal J(H)=\ell_0(H)+\mathcal I(H).
\]
We first show that $\mathcal I(H)$ is finite for every
$H$ in $\mathcal X_\alpha$.  
Suppose first that $1<p<2$. Then
\[ \int_{|x|\ge 1} B_p(b,{v}) \,\mathrm d x \le C_p \int_{|x|\ge 1} |b|^{p-2} {v^2} \,\mathrm d x\le C_p \| H\|_{\alpha}^2\]
by the second bound in \eqref{eq:Bregman-sub-lower}. Since the integral of $|b|^{p-2}$ over all unit intervals is uniformly bounded, this yields the result since, by \eqref{eq:PP}, we obtain an equivalent norm on $\mathcal X_\alpha$ by summing the maxima of ${v^2}$ over all intervals $[k,k+1]$, $k\in \mathbb Z$. If $p>2$, then 
the second bound in \eqref{eq:Bregman-super-lower} gives
\[
 B_p(b,v)\le C_p\{|b|^{p-2}|v|^2+|v|^p\}.
\]
{Since $b$ is bounded, $\|v\|_\infty\le C_p\|H\|_\alpha$,
and $v$ is square-integrable, it follows that
$\mathcal I(H)\le C_p(\|H\|_\alpha^2+\|H\|_\alpha^p)$.} 
Taking into account also Lemma~\ref{prop:Gamma-defect}, we conclude that
$\mathcal J$ is finite on
$\mathcal X_\alpha$.

For fixed $a$, the map $h\mapsto B_p(a,h)$ is strictly convex.  If $H_0\ne H_1$ in $\mathcal X_\alpha$, then $H_0-H_1$ is a
nonzero entire function, so its real zeros are discrete.  The strict
convexity inequality therefore holds for almost every $x$.  Integration
against $x^2\,\mathrm d x$ preserves the strict inequality.  Since $\ell_0$ is
linear, we conclude that $\mathcal J$ is strictly convex.

We next verify that $\mathcal J$ is weakly lower semicontinuous. To this end,  suppose that
$H_j$ tend weakly to $H$ in $\mathcal X_\alpha$.  By
\eqref{eq:pointwiseH}, this implies that $H_j(x)\to H(x)$ for
every $x$.  Since $B_p$ is continuous and nonnegative, Fatou's lemma
gives
\[
 \mathcal I(H)\le\liminf_{j\to\infty}\mathcal I(H_j).
\]
Moreover, $\ell_0(H_j)\to\ell_0(H)$ because $\ell_0$ is bounded and
linear.  Thus
\[
 \mathcal J(H)\le\liminf_{j\to\infty}\mathcal J(H_j).
\]
This proves sequential weak lower semicontinuity.  In particular, every
sublevel set of $\mathcal J$ is norm closed.  These sets are convex, so
they are weakly closed by the Hahn--Banach separation theorem.  Thus $\mathcal J$ is weakly lower
semicontinuous.

We finally prove coercivity. For $1<p<2$, we get
\[ {\mathcal I(H)}\ge c_p \frac{\left\lVertH\right\rVert_\alpha^2}{(1+\left\lVertH\right\rVert_{\alpha})^{2-p}} \]
by using \eqref{eq:Bregman-scaling} and then \eqref{eq:Bregman-sub-lower} along with Lemma~\ref{lem:tail-dominance}. We conclude that coercivity holds because $\ell_0(H)\le C_p \| H\|_\alpha$. When $p>2$, appealing again to \eqref{eq:Bregman-scaling} and applying \eqref{eq:Bregman-super-lower} along with {Lemma~\ref{lem:Gamma-dominance}}, we get
\[ {\mathcal I(H)}\ge c_p \| H\|_{\alpha}^2.\]
We conclude that coercivity holds in this case as well.

Since we have shown that $\mathcal J$ is finite, coercive, strictly convex and weakly semicontinuous, we may apply the direct method from \cite[Chapter~II, Proposition~1.2, p.~35]{EkelandTemam} and conclude that there is a unique minimizer $H_\ast$ in $\mathcal X_\alpha$.

We finally compute the first variation.  Fix $H$ and $K$ in $\mathcal X_\alpha$
and, for each $x$, set
\[
 F_x(s)=x^2B_p(E_p(x),H(x)+sK(x)).
\]
The function $F_x$ is nonnegative and convex.  For $0<|t|\le1$, the
inequalities between the secant slopes of a convex function give
\[
 \left|\frac{F_x(t)-F_x(0)}{t}\right|
 \le F_x(-1)+2F_x(0)+F_x(1).
\]
The right-hand side is integrable, because $\mathcal I$ is finite at
$H-K$, $H$, and $H+K$.  Since $\Phi_p$ is continuously differentiable
for every $p>1$, the difference quotient tends pointwise to
\[
 x^2\{N_p(E_p+H)-N_p(E_p)\}K.
\]
Dominated convergence therefore gives
\[
 D\mathcal J(H)[K]=\ell_0(K)+\int_\mathbb R x^2
 \{N_p(E_p+H)-N_p(E_p)\}K\,\mathrm d x.
\]
At the minimizer $H_*$ this derivative is zero for every
$K$ in $\mathcal X_\alpha$, which proves \eqref{eq:weak-correction}.

Let $\mathbf t$ be an admissible sequence. Then
\begin{equation} \label{eq:Jacobian}
\partial_{t_j} z_k 
 =\begin{cases} 2p\,\frac{t_jz_k-t_kz_j}{t_j^2-t_k^2}, & j\neq k ;\\
 \kappa(\bm{t})\big((1+2p)R_k(\bm{t})+t_k \partial_{t_k} R_k(\bm{t})\big), & j=k.\end{cases} 
\end{equation}
In particular,
\begin{equation}\label{eq:normalized-scalar-hessian}
 D\mathscr Z(\boldsymbol\tau)=\lambda_p I,
 \qquad
 \lambda_p\coloneqq  \frac{\kappa(\bm{\tau})}{\mathscr C_p}.
\end{equation}

<!-- proof-12: body_md -->
We begin by differentiating $R_k=R_k(\bm{t})$. To this end, we 
note to begin with that
\begin{equation}\label{eq:help}
 \partial_{t_k}|\phi_{\mathbf t}(x)|^p
 =\frac{2p}{t_k}|\phi_{\mathbf t}(x)|^p
 \frac{x^2}{t_k^2-x^2},
\end{equation}
When $j\ne k$, we  use that
\[
 \frac{x^4}{(t_j^2-x^2)(t_k^2-x^2)}
 =\frac{t_k^2}{t_j^2-t_k^2}\frac{x^2}{t_k^2-x^2}
 -\frac{t_j^2}{t_j^2-t_k^2}\frac{x^2}{t_j^2-x^2}
\]
and find that 
\begin{equation}\label{eq:offdiag-Jacobian}
 \partial_{t_j} R_k
 =\frac{2p}{t_j}\frac{t_k^2R_k-t_j^2R_j}{t_j^2-t_k^2}.
\end{equation}
Noting that 
\[ \partial_{t_j} \kappa(\bm{t})=\frac{2p}{t_j} {\kappa(\bm t)},\]
we therefore get
\[ \partial_{t_j} z_k = \kappa(\bm{t})t_k\big(\frac{2p}{t_j}R_k+\partial_{t_j} R_k\big)=2p\,\frac{t_jz_k-t_kz_j}{t_j^2-t_k^2},\]
as desired. The case $j=k$ in \eqref{eq:Jacobian} follows immediately by Leibniz's rule. 

Since $R_k(\bm\tau)=0$, $D\mathscr Z(\bm{\tau})$ is a diagonal matrix. By \eqref{eq:Jacobian}, the $k$th diagonal entry is
\[  \kappa(\bm{\tau})t_k \partial_{t_k} R_k(\bm{\tau}).  \]
Hence it remains to show that
\begin{equation} \label{eq:ident1} \tau_k \partial_{\tau_k} R_k(\bm{\tau})=\| \varphi_p\|_p^p.\end{equation}
To prove this, we start by noting that 
\[ \partial_{\tau_k} R_k(\bm{\tau}) = \int_{\mathbb R} \frac{| \varphi_p(x)|^px^2}{(\tau_k^2-x^2)^2}\Big(\frac{2p}{\tau_k}x^2-2\tau_k\Big) \,\mathrm d x\]
by Leibniz's rule along with \eqref{eq:help}. It follows that
\begin{equation} \label{eq:almost} \tau_k \partial_{t_k} R_k(\bm{\tau})=
2\tau_k^2(p-1) \int_{\mathbb R} \frac{| \varphi_p(x)|^px^2}{(\tau_k^2-x^2)^2} \,\mathrm d x
= (p-1) \int_{\mathbb R} \frac{| \varphi_p(x)|^px^2}{(\tau_k-x)^2} \,\mathrm d x,
\end{equation}
where in the last step we used \eqref{eq:Lportho}. 
To compute the latter integral, we write
\begin{equation} \label{eq:tran} \int_{\mathbb R} \frac{| \varphi_p(x)|^px^2}{(\tau_k-x)^2} \,\mathrm d x=\tau_k^2\int_{\mathbb R} \frac{| \varphi_p(x)|^p}{(\tau_k-x)^2} \,\mathrm d x-\| \varphi_p\|_p^p\end{equation}
by \cite[Theorem 3.6]{BCOS}. Since $p>1$, we may use integration by parts to get
\[ \int_{\mathbb R}\frac{| \varphi_p(x)|^p}{(\tau_k-x)^2} \,\mathrm d x= -p \int_{\mathbb R}\frac{| \varphi_{p}(x)|^{p-2}\varphi_p(x) \varphi_p'(x)}{\tau_k-x} \,\mathrm d x.\]
Since the function $\frac{\varphi_p(x)}{(x-\tau_k)^2}-\frac{\varphi'(x)}{x-\tau_k}$ belongs to $PW^p$, another application of \cite[Theorem 3.6]{BCOS} yields
\[ \tau_k^2\int_{\mathbb R}\frac{| \varphi_p(x)|^p}{(\tau_k-x)^2} \,\mathrm d x=\frac{p}{p-1} \|\varphi_p\|_p^p. \]
Plugging this into \eqref{eq:tran} and returning to \eqref{eq:almost}, we arrive at the desired conclusion that $D\mathscr{Z}=\lambda_p I$.

 |\phi_{\bm{t}}'(t_k)|\asymp_p t_k^{-\alpha}

{|\partial_{t_k} z_k|} \le C(\bm{t})

uniformly in $k$. Here $C(\bm{t})$ and the implicit constants in \eqref{eq:local-product-envelope} and \eqref{eq:local-product-derivative} can be chosen to depend only on the $\ell^2$ norm of $\bm c - \bm t$ and the separation of $\bm t${, including its distance from zero}.

 \left(1+\frac{a_n}{c_n+z}\right).

<!-- lemma-15: body_md -->
\label{lem:squared-zero-shift}
Let $\bm t$ be a sequence in $\mathcal{B}(\bm \tau,\rho_0)$. Then  
\begin{equation}\label{eq:diagonal-residual-formula}
 \partial_{t_k} z_k =\lambda_p+2\frac{z_k}{t_k}+2p\left\{
 \sum_{j\geq1}\frac{z_j}{t_j}-t_k^2\sum_{j\ne k}
 \frac{\frac{z_k}{t_k}-\frac{z_j}{t_j}}{t_j^2-t_k^2}\right\}.
\end{equation}
<!-- end lemma-15 -->
<!-- proof-15: body_md -->
Let $A$ be the matrix with entries $A_{k,j}\coloneqq \partial_{t_j} z_k$, and let $A^\circ$ be the off-diagonal part of $A$. Set $\bm v\coloneqq (1/t_j)$. We will identify $A \bm v$ so that
\[ \partial_{t_k} z_k = A_{kk} = t_k (A \bm v)_k-t_k(A^\circ \bm v)_k.\]
In view of \eqref{eq:Jacobian},
\[ (A^\circ \bm v)_k =2p \sum_{j\neq k} \frac{t_jz_k-t_kz_j}{t_j^2-t_k^2} \frac{1}{t_j} ,\]
and so our task is to show that
\begin{equation}\label{eq:task} t_k (A \bm v)_k =\lambda_p+2\frac{z_k}{t_k}+2p
 \sum_{j\geq1}\frac{z_j}{t_j} .\end{equation}

We use the sequence  $v_j=t_j^{-1}$ because we rely on a computation involving  the deformation
\[
 t_j(\varepsilon)^2=t_j^2+\varepsilon,
 \qquad t_j'(0)=\frac1{2t_j}.
\]
Indeed, we may write 
\begin{equation} \label{eq:Jacobianv} (A\bm v)_k
 =2 \left.\frac{\,\mathrm d}{\,\mathrm d\varepsilon}
       \bigl(t_k(\varepsilon)r_k(\varepsilon)\bigr)
       \right|_{\varepsilon=0},
\end{equation}
where 
\[
 r_k(\varepsilon)\coloneqq \int_{\mathbb R}|F_{\bm t(\varepsilon)}(x)|^p
                         \frac{x^2}{t_k^2+\varepsilon-x^2}\,\mathrm d x, \qquad F_{\bm t}(x)\coloneqq \kappa(\bm t)^{\frac{1}{p}} \phi_{\bm t}(x).
\]
To compute the right-hand side of \eqref{eq:Jacobianv}, we put $G(x)\coloneqq |F_{\bm t}(x)|^p$ and note that $F_{\bm t(\varepsilon)}(x)
 =F_{\bm t}\bigl((x^2-\varepsilon)^{1/2}\bigr)$
Differentiation and
integration by parts give
\begin{align*}
 r_k'(0)
 &=-\frac12\int_{\mathbb R}\frac{xG'(x)}{t_k^2-x^2}\,\mathrm d x
   -\int_{\mathbb R}\frac{x^2G(x)}{(t_k^2-x^2)^2}\,\mathrm d x\\
 &=\frac12\int_{\mathbb R}\frac{G(x)}{t_k^2-x^2}\,\mathrm d x
  =\frac{\|F_{\bm t}\|_p^p+z_k/t_k}{2t_k^2}.
\end{align*}
Taking into account \eqref{eq:Jacobianv}, we therefore have 
\begin{equation}\label{eq:stop}
 t_k (A\bm v)_k
 = \|F_{\bm t}\|_p^p+2 \frac{z_k}{t_k}.
\end{equation}

We now consider the function
\[ R(\bm t)\coloneqq \| F_{\bm t} \|_p^p - 2p \sum_{j\ge 1} \frac{z_j}{t_j}\]
for which we have $R(\bm \tau)=\lambda_p$. Therefore it suffices to show that $R(\bm t)$ is constant for $\bm t$ in $\mathcal{B}(\bm \tau,\rho_0)$. {By continuity it suffices to check constancy on finite-coordinate
deformations, so it is enough that all partial derivatives vanish.
The differentiated sums below are justified by the local operator bounds
and $(1/t_j)\in\ell^2$; the matrix $A$ is symmetric by \eqref{eq:Jacobian}.} We find that
\begin{equation}\label{eq:incred}
 \partial_{t_k} \sum_{j\ge 1} \frac{z_j}{t_j}
 =\sum_{j\geq1}\frac{A_{j k}}{t_j}
      -\frac{z_k}{t_k^2}\\
 =(A\bm v)_k-\frac{z_k}{t_k^2}=\frac{\| F_{\bm t}\|_p^p}{t_k}+\frac{z_k}{t_k^2}, \end{equation}
where, in the last step, we used \eqref{eq:stop}. On the other hand, differentiation of the normalized product gives
\[
 \partial_{t_k}|F_{\bm t}(x)|^p
 =\frac{2p\,t_k}{t_k^2-x^2}|F_{\bm t}(x)|^p, 
\]
and so
\[
 \partial_{t_k} \|F_{\bm t} \|_p^p
 = 2p \Big( \frac{\| F_{\bm t}\|_p^p}{t_k} + \frac{z_k}{t_k^2}\Big).
\]
Combining this with \eqref{eq:incred}, we find that $R(\bm t)\equiv \lambda_p$ for $\bm t$ in $\mathcal{B}(\bm \tau,\rho_0)$, and so we conclude that \eqref{eq:task} holds.

\label{prop:local-residual-inverse}
There exist positive numbers $\rho_0, \sigma_0$ such that  
\begin{equation}\label{eq:local-residual-diffeomorphism}
 \mathscr Z: \mathcal{B}(\bm \tau,\rho_0)\longrightarrow V:= \mathscr Z(\mathcal{B}(\bm \tau, \rho_0))
\end{equation}
 is a diffeomorphism  and  $B(\bm 0, \sigma_0) \subset V$. Moreover,                                                                                                           
if $\mathscr Z \bm t$ is in $\ell_1^\infty$, then $D\mathscr Z(\mathbf t)^{-1}$ maps $\ell^\infty_1$ into $\ell^\infty_1$, with $\|D\mathscr Z(\mathbf t)^{-1}\|_{\ell^\infty_1\to\ell^\infty_1}$ uniformly bounded on {$\bm t \in \mathcal{B}(\bm \tau, \rho_0)$ satisfying $\|\mathscr Z\bm t\|_{\ell^\infty_1}\le B$, for each fixed $B$}.
<!-- end lemma-16 -->
<!-- proof-16: body_md -->
Lemma~\ref{lem:squared-zero-shift}
shows that the diagonal of $A=A(\bm t) \coloneqq D \mathscr Z \bm t$ is continuous in
$\ell^\infty$.  At $\boldsymbol\tau$, it equals
$\lambda_pI$ by Lemma~\ref{lem:normalized-zero-algebra}. We also have $A(\bm \tau)^\circ \bm \delta =0$, so we may write
\[ A(\bm t)^\circ\bm \delta=(A(\bm t)^\circ-A(\bm \tau)^\circ)\bm \delta .\]  A direct computation shows that $\|A(\bm t)^\circ-A(\bm \tau)^\circ\|_{2\to 2}\to 0$ when $ \bm t - \bm \tau  \to 0$ in $\ell^2$.  In total, we therefore find that
\begin{equation}\label{eq:differential-close-scalar}
 \|A (\mathbf t)-\lambda_pI\|_{2\to2}\longrightarrow0
 \qquad(\mathbf t -\bm \tau\to \bm 0 \text{ in }\ell^2).
\end{equation}
This implies that we may find $\rho$ such that
 \[ \|\lambda_p^{-1} A(\mathbf t)-I\|_{2\to2} \le \frac12, \qquad \bm t \in \mathcal B(\bm \tau, \rho).\]
We may therefore express the inverse of $A$ on $\ell^2$ as a convergent Neumann series,
\[ A^{-1} \bm \nu =\frac{1}{\lambda_p} \sum_{n=0}^\infty (I-\lambda_p^{-1} A)^n \bm \nu  \]
for every $\bm t$ in $\mathcal B(\bm \tau, \rho)$.
{Finite-coordinate differentiation and truncation, using the
locally continuous operator bounds above, show that $\mathscr Z$ is $C^1$.}
Since the Fr\'{e}chet derivative of $\bm t \to \mathscr Z \bm t$ is a bounded linear isomorphism in $\mathcal B(\bm \tau, \rho)$, the inverse function theorem \cite[p. 149]{Deimling} guarantees that there exists $0<\rho_0 \le  \rho$ such that  $\bm t \to \mathscr Z \bm t$ is invertible from $\mathcal B(\bm \tau, \rho_0)$ onto $V= \mathscr Z( B(\bm \tau, \rho_0))$. 

It remains to prove the assertion about how $D\mathscr Z(\mathbf t)^{-1}$ acts on $\ell^\infty_1$ when $\mathscr Z\bm t$ belongs to that space. 
                                                                                                                                                                                                                                                          
We make a block decomposition of $A$:
\[
A=\begin{pmatrix} Q & R \\ C & T \end{pmatrix},
\]
where $Q$ is the $N\times N$ matrix $(A_{kj})_{k,j=1}^N$
and $N$ is a positive integer to be suitably chosen.
The estimate $\|A-\lambda_pI\|_{2\to2}\le\lambda_p/2$
also holds for $Q$, so $Q$ is invertible and
$\|Q^{-1}\|_{2\to2}\le2/\lambda_p$.
Since both $A$ and $Q$ are invertible, their Schur complement $T-CQ^{-1}R$  is
invertible.
Setting $W\coloneqq (T-C Q^{-1} R)^{-1}$ and following
\cite[pp.~18--19]{HornJohnson}, we may write the blockwise inversion of $A$ as
\[
 A^{-1}=\begin{pmatrix}
 Q^{-1}+Q^{-1}RWCQ^{-1} & -Q^{-1}RW \\
 -WCQ^{-1} & W
 \end{pmatrix}.
\]
Recall that  
$\|v\|_X=\sup_k k|v_k| <\infty$. Fix $B$ with $\|z\|_X\le B$, and write
$X_N=\{(v_k)_{k>N}:\sup_{k>N}k|v_k|<\infty\}$.
For $j\ne k$, \eqref{eq:Jacobian} and the uniform separation give
\[
 \frac{k}{j}|A_{kj}|
 \le C_pB\frac{1+(k/j)^2}{|j-k|(j+k)}.
\]
On summing over $j>N$, the ranges $j<k/2$,
$k/2\le j\le2k$ (with $j\ne k$), and $j>2k$ contribute at most
$C_pB/N$, $C_pB\log(2+k)/k$, and $C_pB/k$, respectively.
Consequently
\begin{equation}\label{eq:weighted-tail-row-estimate}
 \|T^\circ\|_{X_N\to X_N}
 \le\sup_{k>N}\sum_{\substack{j>N\\j\ne k}}
                 \frac{k}{j}|A_{kj}|
 \le C_pB\frac{1+\log N}{N}.
\end{equation}
We must also estimate  $CQ^{-1}R$.
For $j\le N<k$, the same entrywise bound gives
\[
 k|A_{kj}|\le C_pB\frac{j+k^2/j}{(k-j)(k+j)}
 \le C_pB\left(\frac1j+\frac1{k-j}\right).
\]
The sum of the squares of the last expression over $j\le N$ is
bounded by $C_pB^2$, uniformly in $k>N$ and $N$.
Cauchy--Schwarz therefore gives
$\|C\|_{\ell^2_N\to X_N}\le C_pB$.
On the other hand, $R$ is a compression of $A-\lambda_pI$, and
\[
 \|v\|_2^2\le\|v\|_{X_N}^2\sum_{k>N}k^{-2}
 \le N^{-1}\|v\|_{X_N}^2.
\]
Consequently
\[
 \|R\|_{X_N\to\ell^2_N}\le\frac{\lambda_p}{2\sqrt N},
 \qquad
 \|CQ^{-1}R\|_{X_N\to X_N}\le\frac{C_pB}{\sqrt N}.
\]
Let $\Delta$ be the diagonal of $T$ and put
$E=T^\circ-CQ^{-1}R$. Since $|A_{kk}|\ge\lambda_p/2$,
\eqref{eq:weighted-tail-row-estimate} implies
\[
 \|\Delta^{-1}E\|_{X_N\to X_N}
 \le\frac{C_pB}{\lambda_p}
       \left(\frac{1+\log N}{N}+\frac1{\sqrt N}\right).
\]
Choose $N=N(p,B)$ so that this is at most $1/2$.
The Neumann series
\[
 (T-CQ^{-1}R)^{-1}
 =\sum_{m\ge0}(-\Delta^{-1}E)^m\Delta^{-1}
 \quad\text{on }X_N
\]
then converges, with inverse norm at most $4/\lambda_p$.
The displayed blockwise formula now shows that $A^{-1}$ maps
$X$ into itself: $C$ maps the finite head into $X_N$, $R$ maps
$X_N$ into the finite head, and $W$ is bounded on $X_N$.
The norm comparisons on the head cost only constants depending on
$N$; since $N$ depends only on $p,B$, the resulting bound is
$\|A^{-1}g\|_X\le C_{p,B}\|g\|_X$, as required.

\label{lem:finite-comparison-residual}
For every fixed $K$ and every $L\geq1$, there are real numbers
$\beta_{K,0},\ldots,\beta_{K,L-1}$ such that
\begin{equation}\label{eq:finite-comparison-residual}
 f_k^{(K)}=\sum_{q=0}^{L-1}\frac{\beta_{K,q}}{c_k^{2q+1}}

\end{equation}

<!-- proof-17: body_md -->
For the sequence in \eqref{eq:finite-comparison-sequence},
\[
 F_K(z)\coloneqq E_p(z)Q_K(z),\qquad
 Q_K(z)\coloneqq \prod_{j\leq K}\frac{\tau_j^2-z^2}{c_j^2-z^2}.
\]
Set $P(u)\coloneqq |\sin\pi u|^p$ and $W_K(x)\coloneqq x^2|F_K(x)|^p$.  By Fields's expansion of the gamma quotient in
\eqref{eq:A-m} (see \cite{Fields, Frenzen}) and the fact that $Q_K$ is a rational function of $x^2$, we get, for every $L$,
\begin{equation}\label{eq:finite-comparison-amplitude}
 W_K(x)=P(x-\gamma)\left\{
 \sum_{r=0}^{L}d_{K,r}x^{-2r}
 +O_{p,K,L}(x^{-2L-2})\right\}.
\end{equation}
We write $P(u)$ in terms of an absolutely convergent Fourier series,
\[
 P(u)=b_0+\sum_{m\geq1}b_m\cos(2\pi mu),
 \qquad b_m=O_p(m^{-p-1}).
\]
Since $P(0)=0$,
$P(u)=\sum_{m\geq1}b_m\{\cos(2\pi mu)-1\}$.  We set
$c=c_k=k+\gamma$ and use partial fractions to get
\begin{align*} \int_0^\infty\frac{\cos({2\pi  m}(x-c))-1}{c^2-x^2}\,\mathrm d x
 = &\frac{\cos({2\pi  m} c)}{2c}\int_{\mathbb R}\frac{\cos({2\pi  m} x)}{c-x}\,\mathrm d x
+\frac{\sin({2\pi  m} c)}{2c} \int_{\mathbb R}\frac{\sin({2\pi  m} x)}{c-x}\,\mathrm d x \\ & +
\frac{\sin({2\pi  m} c)}{c} \int_0^\infty \frac{\sin({2\pi  m} x)}{x+c}\,\mathrm d x,\end{align*}
where the first two integrals on the right-hand side are to be understood in the principal value sense. By contour integration, we find that they cancel each other. Hence, using $(x+c)^{-1}=\int_0^\infty {\,\mathrm e}^{-s(x+c)}\,\mathrm d s$, we find that
\begin{equation}\label{eq:periodic-integral-identity}
 \int_0^\infty\frac{\cos({2\pi  m}(x-c))-1}{c^2-x^2}\,\mathrm d x
 =\frac{\sin({2\pi  m} \gamma)}{c}
 \int_0^\infty\frac{e^{-{2\pi  m}cs}}{1+s^2}\,\mathrm d s.
\end{equation}
The finite identity
\[
 \frac1{1+s^2}=\sum_{q=0}^{L-1}(-1)^qs^{2q}
 +(-1)^L\frac{s^{2L}}{1+s^2}
\]
shows that the last integral equals
\[
 \sum_{q=0}^{L-1}\frac{(-1)^q(2q)!}{({2\pi  m}c)^{2q+1}}
 +O_L(({2\pi  m}c)^{-2L-1}).
\]
The estimate $\sum_m|b_m|\log(2+m)<\infty$ justifies termwise integration.
It follows that
\begin{equation}\label{eq:periodic-integral-expansion}
 \int_0^\infty\frac{P(x-\gamma)}{c_k^2-x^2}\,\mathrm d x
 =\sum_{q=0}^{L-1}\frac{\ell_q}{c_k^{2q+2}}
 +O_{p,L}(c_k^{-2L-2})
\end{equation}
for suitable constants $\ell_q$. 

It remains to estimate the contribution from the higher order terms in
\eqref{eq:finite-comparison-amplitude}.  Set
\[
 U_q(x)\coloneqq x^{2q}W_K(x)-P(x-\gamma)
 \sum_{r=0}^{q}d_{K,r}x^{2(q-r)},
 \qquad
 H_q(c)\coloneqq \int_0^\infty\frac{U_q(x)}{c^2-x^2}\,\mathrm d x.
\]
Then $U_q$ is integrable and
$U_{q+1}=x^2U_q-d_{K,q+1}P(x-\gamma)$. Writing  $U_q=\frac{c^2-x^2}{c^2}U_q+\frac{x^2}{c^2}U_q$, we therefore get
\begin{equation}\label{eq:moment-recursion}
 H_q(c)=\frac1{c^2}\left\{
 \int_0^\infty U_q(x)\,\mathrm d x+d_{K,q+1}
 \int_0^\infty\frac{P(x-\gamma)}{c^2-x^2}\,\mathrm d x+H_{q+1}(c)
 \right\}.
\end{equation}
                                                                           
$L$ iterations of \eqref{eq:moment-recursion}, together with
\eqref{eq:periodic-integral-expansion}, yield a polynomial of degree $L$ in $c^{-2}$ with remainder 
$O(c^{-2L-2})$, given that $H_L(c)=O(c^{-2})$. To see that the latter bound holds, we split the integral that defines $H_L$ as $\int_0^{c/2}+\int_{c/2}^{3c/2}+\int_{3c/2}^\infty$.
The first and third integrals are both $O(c^{-2})$ because  $U_L$ is integrable. To deal with the middle integral, we use
$U_L(c+u)=O(c^{-2}P(u))$ and get that it is $O(c^{-3} \log(2+c))$. 

Finally, since $W_K(x)$ is an even function,
\[
 f_k^{(K)}=2c_k\int_0^\infty\frac{W_K(x)}{c_k^2-x^2}\,\mathrm d x,
\]
and \eqref{eq:finite-comparison-residual} follows.

 \mathcal H_M(\mathbf t)=\mathcal H_M(\mathbf c)
\end{equation}

is bounded.}

The first assertion is trivial. If 
\eqref{eq:change-variable-assumption} holds, then the quotient
$u_j\coloneqq (t_j-c_j)/c_j$ is a finite linear combination of
$c_j^{-2},\ldots,c_j^{-2M}$ modulo $\ell_{2M}^2$.  Expanding

\]
to the order ${c_j^{-2M-1}}$, we get an upper-triangular change of basis with diagonal
entries one. Clearly, the sum of all higher order terms belongs to
$\ell_{2M+1}^2(\mathbf c)$. {The stated bounds control the coefficients, the weighted
Taylor remainders, and the inverse triangular change of basis, giving
the asserted uniform norm estimates.}

<!-- lemma-19: body_md -->
\label{lem:moment-subtraction}
Let $\bm t$ be an admissible sequence satisfying  $t_k=c_k+O(c_k^{-1})$. If $\bm q$ is in $\ell^2_{2L}$ with $L\ge 1$, then 
    
         
   
                                                  
                         
                             
{the moments}
\[
 \mu_\nu\coloneqq \sum_{j\ge 1}t_j^{2\nu}q_j,
 \qquad 0\leq\nu<L,
\]
converge absolutely and
\begin{equation}\label{eq:moment-subtraction}
 t_k\sum_{j\ne k}
 \frac{q_j}{t_j^2-t_k^2}
 =-\sum_{\nu=0}^{L-1}\frac{\mu_\nu}{t_k^{2\nu+1}}
 +\frac{Lq_k}{t_k}
 +\frac{(K_{\mathbf t}u)_k}{t_k^{2L-1}},
 \qquad u_j=t_j^{2L}q_j.
\end{equation}
<!-- end lemma-19 -->
<!-- proof-19: body_md -->
Absolute convergence of the moments follows from the Cauchy--Schwarz inequality.  Writing
\[
 \frac1{a-b}=-\sum_{\nu=0}^{L-1}\frac{a^\nu}{b^{\nu+1}}
 +\frac{a^L}{b^L(a-b)}
\]
with $a=t_j^2$ and $b=t_k^2$  summing over $j\neq k$, we get \eqref{eq:moment-subtraction}.

<!-- lemma-20: body_md -->
\label{lem:differential-expansions}
Let $\mathbf t$ be {in $\mathcal B(\boldsymbol\tau;\rho_0)$} satisfying
$t_k=c_k+O(c_k^{-1})$, and suppose that $\mathscr Z(\bm t)$ is in $\mathcal H_M(\mathbf t)$. 

If $M=0$ and $\bm v$ is in $\ell^\infty_1$, then
\begin{equation}\label{eq:offdiag-preserves-base}
 A^\circ v\in\mathcal H_0(\mathbf t).
\end{equation}
If $M\geq1$ and $\bm v$ is in $\mathcal H_{M-1}(\mathbf t)$, then
\begin{equation}\label{eq:offdiag-raises-order}
 A^\circ v\in\mathcal H_M(\mathbf t).
\end{equation}
In addition,
\begin{equation}\label{eq:diagonal-even-expansion}
 ((A_{kk}-\lambda_p)/t_k)
 \in \mathcal{H}_M(\bm t),
\end{equation}
with the coefficient $a_0$ of $(1/t_j)$ being $0$. {The operator bounds are uniform when
$\|\mathscr Z(\mathbf t)\|_{\mathcal H_M(\mathbf t)}$
and $\sup_k k|t_k-c_k|$ are bounded.}
<!-- end lemma-20 -->
<!-- proof-20: body_md -->
By the assumption on $z=\mathscr Z \bm t$, we may write
\begin{equation}\label{eq:start}
 z_j=\sum_{a=0}^{M}b_at_j^{-(2a+1)}+\eta_j,
 \qquad \bm\eta\in\ell_{2M+1}^2(\mathbf t).
\end{equation}
Hence, to prove \eqref{eq:offdiag-preserves-base} and \eqref{eq:offdiag-raises-order}, it suffices to show that $A_0^\circ \bm v, \ldots, A_M^\circ\bm v$, and $A_{\bm \eta}^\circ \bm v$ are all in $\mathcal H_M(\bm t)$, where  
\[ (A_a^\circ \bm v)_k=2p \sum_{j\neq k}  \frac{t_jt_k^{-2a-1}-t_k t_j^{-2a-1}}{t_j^2-t_k^2} v_j, \qquad 0\le a \le M,\]
and
\begin{equation}\label{eq:off}  (A_{\bm \eta}^\circ \bm v)_k=2p \sum_{j\neq k}  \frac{t_j\eta_k-t_k \eta_j}{t_j^2-t_k^2} v_j. \end{equation}
Polynomial division in
\eqref{eq:Jacobian} gives
\begin{align}\label{eq:pure-power-action}
 (A_a^\circ \bm v)_k
 = 2p\sum_{\ell=0}^{a}t_k^{-(2(a-\ell)+1)}
 \sum_{j\ge 1} t_j^{-(2\ell+1)}v_j
 {-2p(a+1)t_k^{-(2a+2)}v_k}.
\end{align}
{For $M=0$ the last term belongs to $\ell^2_1$ since
$v_j=O(c_j^{-1})$. For $M\ge1$, expand $v\in\mathcal H_{M-1}(\bm t)$:
multiplication by $t_k^{-2a-2}$ gives the required odd powers and a
remainder in $\ell^2_{2M+1}$. Thus $A_a^\circ\bm v\in\mathcal H_M(\bm t)$.}

To compute the contribution from the remainder $\eta_j$, we rewrite
\eqref{eq:off} as 
\begin{equation}\label{eq:remainder-offdiag-action}
 (A^\circ_\eta \bm v)_k
 =2p\eta_k\sum_{j\ne k}
 \frac{t_jv_j}{t_j^2-t_k^2}
 -2pt_k\sum_{j\ne k}
 \frac{\eta_jv_j}{t_j^2-t_k^2}.
\end{equation}
In both cases under consideration, $(t_jv_j)$ is bounded.  The first sum is
therefore uniformly bounded, and hence the first term
belongs to $\ell_{2M+1}^2(\mathbf t)$.  To estimate the second term, we apply
Lemma~\ref{lem:moment-subtraction} with
$q_j=\eta_jv_j$ and $L=M+1$. Indeed, since the sequence $(t_j v_j) $ is bounded, we have $\bm q$ in $\ell^2_{2L}$.
Formula \eqref{eq:moment-subtraction} gives the required odd powers and a
remainder in $\ell_{2M+1}^2$ since $K_{\bm t}$ is bounded on $\ell^2$ by Hilbert's inequality.  This proves
\eqref{eq:offdiag-preserves-base} and \eqref{eq:offdiag-raises-order}.

To deal with the diagonal, we use \eqref{eq:diagonal-residual-formula}. We consider the contribution from each of the terms in \eqref{eq:start} separately. If we set $t_j^{-2a-1}$ in place of $z_j$ in the expression
\begin{equation} \label{eq:express}
 \sum_j\frac{z_j}{t_j}-t_k^2\sum_{j\ne k}
 \frac{z_k/t_k-z_j/t_j}{t_j^2-t_k^2}
\end{equation}
in \eqref{eq:diagonal-residual-formula}, then it becomes, by polynomial division,  
{$(a+1)t_k^{-2a-2}$ plus a linear combination of
$t_k^{-2},\ldots,t_k^{-2a}$; after division by $t_k$ the additional
power lies in the required expansion or its remainder.} To deal with the contribution from the remainder $\eta_j$, we set $\rho_j\coloneqq \eta_j/t_j$ which we insert in place of $z_j/t_j$ in \eqref{eq:express}:
\begin{equation}\label{eq:rho}
 \sum_j\rho_j-t_k^2\sum_{j\ne k}
 \frac{\rho_k}{t_j^2-t_k^2}+t_k^2\sum_{j\ne k}
 \frac{\rho_j}{t_j^2-t_k^2}.
\end{equation}
Since $\bm \rho$ is in $\ell_{2M+2}^2$, we may apply
Lemma~\ref{lem:moment-subtraction} with $L=M+1$ to the third sum in \eqref{eq:rho}. Its zeroth moment cancels the first sum
$\sum_j\rho_j$.  The remaining moments
yield a linear combination of $t_k^{-2},\ldots,t_k^{-2M}$. The middle sum in \eqref{eq:rho} is in $\ell_{2M}^2$, since the sum 
 $\sum_{j\neq k} 
 \frac1{|t_j^2-t_k^2|}$ is uniformly bounded.
We conclude that also \eqref{eq:diagonal-even-expansion} holds.

\label{prop:inverse-preserves-expansions}
We make the same assumptions as in Lemma~\ref{lem:differential-expansions} and let $R>0$. Then
\begin{equation}\label{eq:inverse-preserves-expansions}
 D\mathscr Z(\mathbf t)^{-1}:\mathcal H_M(\mathbf t)
 \longrightarrow\mathcal H_M(\mathbf t)
\end{equation}
is uniformly bounded {on subsets of $\mathcal U(R,\rho_0)$
where $\|\mathscr Z(\mathbf t)\|_{\mathcal H_M(\mathbf t)}$
and $\sup_k k|t_k-c_k|$ are bounded.}

<!-- proof-21: body_md -->
We consider $\bm g=A\bm v$ in $\mathcal H_M(\mathbf t)$.  Lemma~\ref{prop:local-residual-inverse}
first shows that $\bm v$ satisfies $v_k=O(t_k^{-1})$.  When $M=0$,
Lemma~\ref{lem:differential-expansions} shows that 
$A^\circ \bm v$ is in $\mathcal H_0$ and
$(A_{kk}-\lambda_p)v_k$ is in $\ell_1^2$.  Therefore
\begin{equation}\label{eq:inverse-induction-identity}
 v=\lambda_p^{-1}
 \{g-A^\circ v-(A_{kk}-\lambda_p)v_k\}
 \in\mathcal H_0.
\end{equation}
Suppose now that the assertion is known at level $M-1$.  Since
$g,{z}\in\mathcal H_M(\bm t)\subset\mathcal H_{M-1}(\bm t)$, the induction hypothesis implies that
$v$ is in $\mathcal H_{M-1}(\bm t)$.  Lemma~\ref{lem:differential-expansions} puts

\eqref{eq:diagonal-even-expansion}, multiplied by
$v\in\mathcal H_{M-1}(\bm t)$, also belongs to $\mathcal H_M(\bm t)$.  For its weighted

\[
 t_k^{2M+1}|\delta_kv_k|
 \leq |t_k^{2M}\delta_k|\,|t_kv_k|.
\]
Formula \eqref{eq:inverse-induction-identity} completes the induction.  The
same estimates give the asserted uniform bounds.

Put $\alpha=2/p$ and $r_p=\varphi_p-K(p)E_p$, with $K(p)$ as in the

 P_m(x)=\prod_{n\ne m}

Splitting the sum at $m/2$ and $2m$ gives, uniformly on $I_m$,
\[
 \sum_{n\ne m}\left|\frac{\Delta_n}{c_n^2-x^2}\right|
 \le C_p\frac{\log(2+m)}m,
 \qquad
 P_m(x)=1+O_p\!\left(\frac{\log(2+m)}m\right).
\]
The exact product identity is
\[
 r_p(x)=K(p)\left[
 E_p(x)(P_m(x)-1)
 +\Delta_m\frac{E_p(x)}{c_m^2-x^2}P_m(x)\right].
\]
Lemma~\ref{lem:Gamma-properties} yields $|E_p(x)|\le C_pm^{-\alpha}$

\begin{equation}\label{eq:physical-rough}
 |r_p(x)|\le C_p
 \frac{\log(2+|x|)}{(1+|x|)^{\alpha+1}}.
\end{equation}
