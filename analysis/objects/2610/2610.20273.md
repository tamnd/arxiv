<!-- defn-1: body_md -->
\label{net poly def}
For an arbitrary field $K$, consider the affine coordinate ring 
\[
R_{n}=K[x_i, y_i]_{1\leq i \leq n}[(x_{i}-x_{j})^{-1}]/\langle f(x_{i}, y_{i})\rangle_{1\leq i \leq n},
\]
where $f$ is the defining polynomial (\ref{short EC}) for the elliptic curve $E/K$. Let $\bm{P}=(P_{1}, ..., P_{n}) \in E(K)^{n}$ and $\bm{v}=(v_{1}, \dotsc, v_{n}) \in \ensuremath{\mathbb{Z}}^{n}$. If $P_{i}\neq O$ for all $i$ and $P_i \neq \pm P_j$ for any $i \neq j$, then there exist rational functions $\Psi_{\bm{v}}(\bm{P})$, $\Phi_{\bm{v}}(\bm{P})$, $\bar\Omega_{\bm{v}}(\bm{P}) \in R_{n}$ such that 
\begin{equation}\label{notation}
\bm{v} \cdot \bm{P} = [v_{1}]P_{1} +...+[v_{n}]P_{n}=\left(\frac{\Phi_{\bm{v}}(\bm{P})}{\Psi^{2}_{\bm{v}}(\bm{P})}, \frac{\bar\Omega_{\bm{v}}(\bm{P})}{\Psi^{3}_{\bm{v}}(\bm{P})}\right).
\end{equation}
The polynomial $\Psi_{\bm{v}}$ is defined to be the \emph{$\bm{v}$-th net polynomial}, which is an elliptic net (for proof, see \cite[Theorem 4.1]{Stange}).

\cite[Theorem 1.5]{Akbary}\label{Akbary TFAE}
Let $E/K$ be an elliptic curve defined by the equation in (\ref{short EC}). Let $\bm{P}=(P_1, \ldots, P_n) \in E(K)^n$ be such that $P_i \not\equiv O \bmod \ensuremath{\mathfrak{p}}$ and $P_i\pm P_j \not\equiv O \bmod \ensuremath{\mathfrak{p}}$ for $1\leq i < j \leq n$. \begin{enumerate}
    \item[(a)] \text{There exists $1 \le i \le n$, such that} $\nu(\Psi_{2\bm{e_i}}(\bm{P})) > 0 \text{ \textit{and} } \nu(\Psi_{3\bm{e_i}}(\bm{P})) > 0$.
    \item[(b)] \text{There exists $1 \le i \le n$ such that for all $n \ge 2$ we have} $\nu(\Psi_{n\bm{e_i}}(\bm{P})) > 0$.
    \item[(c)] \text{There exists $\bm{v} \in \mathbb{Z}^n$ and $1 \le i \le r$ such that} $\nu(\Psi_{\bm{v}}(\bm{P})) > 0 \text{ and } \nu(\Psi_{\bm{v}+\bm{e_i}}(\mathbf{P})) > 0$.
 \item[(d)] \text{There exists $\bm{v} \in \mathbb{Z}^n$ such that} $\nu(\Psi_{\bm{v}}(\bm{P})) > 0 \text{ \text{and} } \nu(\Phi_{\bm{v}}(\bm{P})) > 0$.
    \item[(e)] \text{There exists $1 \le i \le r$ such that $P_i \pmod{\mathfrak{p}}$ is singular.}
\end{enumerate}

<!-- defn-2: body_md -->
\label{elliptic net}
Let $A$ be a free finitely-generated abelian group and $R$ be an integral domain. An \emph{elliptic net} is any map $W \colon A \rightarrow R$ with $W(\bm{0})=0$ and, for any $\bm{p}, \bm{q}, \bm{r}, \bm{s} \in A$,
\begin{multline*}
W(\bm{p}+\bm{q}+\bm{s})W(\bm{p}-\bm{q})W(\bm{r}+\bm{s})W(\bm{r}) \\
+ W(\bm{q}+\bm{r}+\bm{s})W(\bm{q}-\bm{r})W(\bm{p}+\bm{s})W(\bm{p}) \\ 
+ W(\bm{r}+\bm{p}+\bm{s})W(\bm{r}-\bm{p})W(\bm{q}+\bm{s})W(\bm{q})=0.
\end{multline*} \label{elliptic net def}
We call the rank of \(W\) the rank of the elliptic net.

\cite[Proposition 2.1]{Stange} \label{odd net}
Let $A$ be a free abelian group of finite rank, and $R$ be an integral domain. Let $W\colon A \rightarrow R$ be an elliptic net. Then for any $\bm{v} \in A, W(-\bm{v})=-W(\bm{v})$. In other words, an elliptic net is an odd function.

<!-- prop-1: body_md -->
\label{net poly}
The net polynomials satisfy the following properties:
\begin{enumerate}[(a)]
\item $\Psi_{\bm{v}}$ is an elliptic net, so the recurrence in \Cref{elliptic net} holds;
\item For all $\bm{v} \in \mathbb{Z}^{n}$, the values of the $\bm{v}$-th net polynomial are completely defined by the following initial conditions:
\begin{itemize}
\item $\Psi_{\bm{0}} = 0$;
\item $\Psi_{\bm{e}_i} = 1$ for $1\leq i \leq r$;
\item $\Psi_{2\bm{e}_i} = 2y_i + a_{1}x_i + a_{3} = \psi_{2}(P_i)$;
\item $\Psi_{\bm{e}_i + \bm{e}_j} = 1$, $i \neq j$;
\item $\Psi_{2\bm{e}_i + \bm{e}_j} = 2x_i + x_j - \left(\frac{y_j - y_i}{x_j - x_i}\right)^{2}
      - a_{1}\left(\frac{y_j - y_i}{x_j - x_i}\right) + a_{2}$, $i \neq j$.
\end{itemize}
\label{a values}
\item For $1\leq i \leq r$, we have $\Psi_{n\bm{e}_{i}}(\bm{P})=\psi_{n}(P_i)$.
\item For any $\bm{v}, \bm{u} \in \mathbb{Z}^{r}$,
\begin{equation}
\Psi_{\bm{v}}^{2}\Psi_{\bm{u}}^{2}\left(x\left(\bm{v} \cdot \bm{P}\right)
- x\left(\bm{u} \cdot \bm{P}\right)\right)
= -\Psi_{\bm{v} + \bm{u}}\Psi_{\bm{v} - \bm{u}},
\end{equation}
\label{notation}
where $x(\bm{v} \cdot \bm{P})$ represents the $x$-coordinate of the point $\bm{v} \cdot \bm{P}$.

\item Using the notations in \ref{notation}, for $1 \leq i \leq r$, substitute $\bm{u} = \bm{e}_i$ in \ref{notation}; then we have
\begin{equation}
\Phi_{\bm{v}}(\bm{P})
= \Psi_{\bm{v}}^{2}(\bm{P})\,x(P_i)
- \Psi_{\bm{v} + \bm{e}_i}(\bm{P})\Psi_{\bm{v} - \bm{e}_i}(\bm{P}).
\end{equation}
\label{net poly b}
\end{enumerate}

One can refer to \cite[Theorem 2.5]{Stange} for examples.

<!-- defn-3: body_md -->
\label{appropriate}
Let $E/K$ be a cubic curve defined by (\ref{short EC}), not necessarily smooth. A set of points $\{P_1, \ldots, P_n\}$ on the non--singular part of $E$ is called \emph{appropriate} if the following hold:
\begin{enumerate}
\item $P_i\neq O$ for all $i=1, \ldots, n$;
\item $[2]P_i \neq O$ for all $i=1, \ldots, n$;
\item $P_i \neq \pm P_j$ for any $i \neq j$; and 
\item $[3]P_1\neq O$ whenever $n=1$.
\end{enumerate}

If we have an appropriate $n$--tuple of points $\bm{P}$ on the non--singular part of $C/K$, then we may define a map 
\[
\Psi_{\bm{v}}(\bm{P})\colon \ensuremath{\mathbb{Z}}^n \rightarrow K
\]
known as the elliptic net associated to $C$ and $\bm{P}$, which is an elliptic net.

\cite[Corollary 5.2]{Stange}\label{net polynomial vanish}
For an elliptic net $\Psi_{\bm{v}}(\bm{P}) \colon \ensuremath{\mathbb{Z}}^n \rightarrow K$ associated to a cubic Weierstrass curve $E$ and appropriate points $\bm{P}$ on the non--singular part of $E/K$, we have $\Psi_{\bm{v}}(\bm{P})=0$ if and only if $\bm{v}\cdot \bm{P}=O$ on $E_{\text{ns}}$, where $E_{\text{ns}}$ denotes the set/group of non--singular points of $C$ defined over $K$ and $\bm{v}\cdot \bm{P}$ is as defined in \Cref{net poly def}.

<!-- defn-4: body_md -->
The Sylvester matrix of $f$ and $g$ with respect to $x$, denoted $\operatorname{Syl}_{x}(f, g)$ is the following $(m+n)\times (m+n)$ matrix:
\[
\operatorname{Syl}_{x}(f, g)=
\begin{pmatrix}
a_0 & 0 & \cdots & 0 & b_0 & 0 & \cdots & 0\\
a_1 & a_0 & \cdots & 0 & b_1 & b_0 & \cdots & 0\\
a_2 & a_1 & \ddots & 0 & b_2 & b_1 & \ddots & 0\\
\vdots & \vdots & \ddots & a_0 & \vdots & \vdots & \ddots & b_0\\
a_n & a_{n-1} & \cdots & a_0 & b_m & b_{m-1} & \cdots & b_0\\
0 & a_n & \cdots & a_1 & 0 & b_m & \cdots & b_1\\
\vdots & \vdots & \ddots & \vdots & \vdots & \vdots & \ddots & \vdots\\
0 & 0 & \cdots & a_n & 0 & 0 & \cdots & b_m
\end{pmatrix}.
\]
The resultant of $f$ and $g$, with respect to $x$, denoted $\operatorname{Res}_{x}(f, g)$, is the determinant of the Sylvester matrix. Thus, 
\[
\operatorname{Res}_{x}(f, g)=\operatorname{det}\left(\operatorname{Syl}_{x}(f, g)\right).
\]
<!-- end defn-4 -->
<!-- prop-3: body_md -->
Let $f(x)$ and $g(x)$ be polynomials in $K[x]$ of positive degree. 
\begin{enumerate}
\item The resultant $\operatorname{Res}_{x}(f, g)$ is an integer polynomial in the coefficients of $f$ and $g$. Furthermore, $f$ and $g$ have a common factor in $K[x]$ if and only if $\operatorname{Res}_{x}(f, g)=0$. 
\item There are polynomials $A, B \in K[x]$ such that 
\[
Af+Bg=\operatorname{Res}_{x}(f, g).
\]
Furthermore, the coefficients of $A$ and $B$ are integer polynomials in the coefficients of $f$ and $g$. 
\end{enumerate}

See \cite[Chapter 3, \S 6, Proposition 3, 5]{CoxAndLittle1}.

<!-- prop-4: body_md -->
\label{resultant properties}
Let $f(x)$ and $g(x)$ be polynomials in $K[x]$ of positive degree. 
\begin{enumerate}
\item  If $f(x)$ and $g(x)$ split into linear factors
\[
f(x)=a_0\prod_{i=1}^{n} (x-\lambda_i),\quad g(x)=b_0\prod_{j=1}^m(x-\mu_j),
\]
then
\[
\operatorname{Res}_{x}(f, g)=a_0^mb_0^m\prod_{i=1}^n\prod_{j=1}^m (\lambda_i-\mu_j)=a_0^m\prod_{i=1}^ng(\lambda_i)=(-1)^{mn}b_0^{n}\prod_{j=1}^{m}f(\mu_j).
\]
\item If $a_i$ is assigned with weight $n-i$ and $b_j$ is assigned with weight $m-j$, then $\operatorname{Res}_{x}(f, g)$ is homogenous of weight $mn$. 
\end{enumerate}

<!-- prop-6: body_md -->
\label{singular vanish}
Let $E/K \colon y^2=x^3+Ax+B$ be a singular cubic Weierstrass curve defined over $K$ and let $\bm{P}=(P_1, P_2) \in E\times E$ be a pair of distinct points. Let $\bm{v}=(v_1, v_2)\in \ensuremath{\mathbb{Z}}^2$. Then $\Phi_{\bm{v}}(\bm{P})$ and $\Psi_{\bm{v}}(\bm{P})$ vanish simultaneously if one of the following holds:
\begin{enumerate}
\item $|v_1|<2$, $|v_2|\geq 2$, and $P_2$ is the singular point;
\item $|v_2|<2$, $|v_1|\geq 2$, and $P_1$ is the singular point;
\item $|v_1|,|v_2|\geq 2$, and one of $P_1,P_2$ is the singular point.
\end{enumerate}
<!-- end prop-6 -->
<!-- remark-3: body_md -->
In the definition of net polynomials (see \Cref{net poly def}), it was stated that $\Phi_{\bm{v}}(\bm{P})$ and $\Psi_{\bm{v}}(\bm{P})$ are elements in the affine coordinate ring 
\[
R_{2}=K[x_i, y_i]_{1\leq i \leq 2}[(x_{i}-x_{j})^{-1}]/\langle f(x_{i}, y_{i})\rangle_{1\leq i \leq 2},
\]
where $f$ is the defining polynomial (\ref{short EC}) for the elliptic curve $E/K$. Therefore, we do not allow both points to be evaluated at the point singularity in this proposition.
<!-- end remark-3 -->
<!-- remark-4: body_md -->
The proposition implies that $\Phi_{\bm{v}}(\bm{P})$ and $\Psi_{\bm{v}}(\bm{P})$ cannot vanish simultaneously for $\bm{v}\in\{(0, \pm1), (\pm1, 0), (\pm1, \pm1), (\pm 1, \mp 1)\}$, the non-vanishing of $\Psi_{\bm{v}}(\bm{P})$ follows directly from the initial values. For the first three vectors, these values are $\pm1$, while $\Psi_{\pm(1,-1)}(\bm{P})=\pm(x_2-x_1)$ (see \cite[Theorem 2.5]{Stange} for the relevant recurrence). Since one of $P_1,P_2$ is the singular point $(x_*,0)$, the equality $x_1=x_2$ would force both points to equal $(x_*,0)$, which we have excluded from this proposition. Therefore, $\Phi_{\bm{v}}(\bm{P})$ and $\Psi_{\bm{v}}(\bm{P})$ cannot vanish simultaneously at these vectors.
<!-- end remark-4 -->

We have assumed that $E$ is singular, so let $P_*=(x_*,y_*)$ denote its singular point. Then the following statements hold:
\begin{enumerate}
    \begin{multicols}{2}
        \item $\Delta=4A^3+27B^2=0$;
        \item $\frac{\partial f}{\partial y}(P_*)=2y_*=0$, so $y_*=0$;
        \columnbreak
        \item $\frac{\partial f}{\partial x}(P_*)=3x_*^2+A=0$;
        \item $x_*^3-3x_*^3+B=0$, so $B=2x_*^3$.
    \end{multicols}
\end{enumerate}
This also means that the singular point is $P_*=(x_*, 0)$.

We now follow the proof of \cite[Lemma 2.4]{Stange} and induct on $\lVert\bm v\rVert_\infty$. The assertion has already been shown for all relevant vectors $\bm{v} \in \ensuremath{\mathbb{Z}}^2$ with $\lVert\bm{v}\rVert_\infty\leq4$. Let $N_0>4$, and suppose that the assertion holds for every $\bm u\in\mathbb Z^2$ satisfying $\lVert\bm u\rVert_\infty<N_0$. 

Let $\bm{v}\in \ensuremath{\mathbb{Z}}^2$ satisfy $\lVert\bm v\rVert_\infty=N_0$. For $i=1, 2$, define $w_i=\lceil \frac{v_i}{2}\rceil$. Stange showed that we can construct a recurrence, depending on the parity of $v_i$, demonstrating that $\Psi_{\bm{v}}$ can be expressed in terms of $\Psi_{\bm{u}}$ with $\lVert\bm{u}\rvert_{\infty} \leq N_0$. 

\underline{Case I:} without loss of generality, assume $v_1$ is odd and $v_2$ is even.
\[
\Psi_{(v_1, v_2)}=\Psi_{(w_1, w_2+1)}\Psi_{(w_1, w_2-1)}\Psi_{(w_1-1, w_2)}^2-\Psi_{(w_1-1, w_2+1)}\Psi_{(w_1-1, w_2-1)}\Psi_{(w_1, w_2)}^2.
\]
\underline{Case II:} both $v_1$ and $v_2$ are odd.
\[
\Psi_{(v_1, v_2)}=\Psi_{(w_1, w_2+1)}\Psi_{(w_1, w_2-1)}\Psi_{(w_1-1, w_2-1)}^2-\Psi_{(w_1-1, w_2)}\Psi_{(w_1-1, w_2-2)}\Psi_{(w_1, w_2)}^2.
\]
\underline{Case III:} both $v_1$ and $v_2$ are even.
\[
\Psi_{(v_1, v_2)}=\Psi_{(w_1+1, w_2+1)}\Psi_{(w_1, w_2-1)}\Psi_{(w_1-1, w_2)}\Psi_{(w_1, w_2)}-\Psi_{(w_1, w_2+1)}\Psi_{(w_1-1, w_2-1)}\Psi_{(w_1+1, w_2)}\Psi_{(w_1, w_2)}.
\]
If $v_i$ is even, every $i$-th coordinate appearing on the right-hand side belongs to $\{w_i-1,w_i,w_i+1\}$. Hence
\[
    |u_i|\leq |w_i|+1=\frac{|v_i|+2}{2}.
\]
If $v_i$ is odd, every such coordinate belongs to $\{w_i-2,w_i-1,w_i,w_i+1\}$. Since $w_i=(v_i+1)/2$, we may deduce that
\[
    |u_i|
    \leq \max\left(|w_i-2|,|w_i+1|\right)
    =\frac{|v_i|+3}{2}.
\]
Consequently, every vector $\bm{u}$ appearing on the right-hand side of the three recurrences satisfies
\[
    \lVert\bm{u}\rVert_\infty
    \leq \frac{\lVert\bm{v}\rVert_\infty+3}{2}
    =\frac{N_0+3}{2}<N_0,
\]
where the final inequality follows from $N_0>4$.
Suppose that $P_i=P_*$ and $|v_i|\geq 2$. We claim that each of the two monomials contains a factor $\Psi_{\bm{u}}$ with $|u_i|\geq 2$.

If $v_i$ is even, then $|w_i|\geq 1$ and each monomial contains factors whose $i$-th coordinates are $w_i-1$ and $w_i+1$. At least one of these coordinates has absolute value at least $2$. If $v_i$ is odd, then $|v_i|\geq 3$. In Cases I and II, the relevant coordinates are $w_i$ and $w_i-1$, at least one of which has absolute value at least $2$, except when $i=2$ in Case II. In that case, the first monomial contains coordinates $w_2+1$ and $w_2-1$, while the second contains coordinates $w_2$ and $w_2-2$. Since $|v_2|\geq 3$, each pair contains a coordinate of absolute value at least $2$.

Overall, each monomial contains a factor $\Psi_{\bm{u}}$ such that $\lVert\bm{u} \rVert_\infty<N_0$ and $|u_i|\geq 2$. By the inductive hypothesis, this factor vanishes at
$\bm{P}$. Both monomials therefore vanish, giving $\Psi_{\bm{v}}(\bm{P})=0$ and completing the induction.

Finally, we turn our attention to the numerator function $\Phi_{\bm{v}}$. From \Cref{net poly}, we can express $\Phi_{\bm{v}}$ in terms of $\Psi_{\bm{v}}$:
\begin{equation}\label{phi}
\Phi_{\bm{v}}=\Psi_{\bm{v}}^2x(P_i)-\Psi_{\bm{v}+\bm{e}_i}\Psi_{\bm{v}-\bm{e}_i},
\end{equation}
which allows us to conclude the proof:
\begin{enumerate}
\item If $\bm{v}=(v_1, \pm 1)$ with $|v_1|>1$, then choose $i=2$ in (\ref{phi}), which gives us 
\begin{align*}
\Phi_{\bm{v}}(P_*, P_2)
&=x_2\Psi_{\bm{v}}(P_*, P_2)^2-\Psi_{(v_1, v_2+1)}(P_*, P_2)\Psi_{(v_1, v_2-1)}(P_*, P_2) \\
&=0 \\
&=\Psi_{\bm{v}}(P_*, P_2).
\end{align*}
\item If $\bm{v}=(\pm1, v_2)$ with $|v_2|>1$, then choose $i=1$ in (\ref{phi}), similar to the case above, it also gives us 
\[
\Phi_{\bm{v}}(P_1, P_*)=\Psi_{\bm{v}}(P_1, P_*)=0.
\]
\item If $\bm{v}$ satisfies $\min_{i=1, 2}|v_i|\geq 2$, by the previous argument for $\Psi_{\bm{v}}$, we obtain
\[
\Phi_{\bm{v}}(P_1, P_*)=\Phi_{\bm{v}}(P_*, P_2)=\Psi_{\bm{v}}(P_*, P_2)=\Psi_{\bm{v}}(P_1, P_*)=0.
\]
\end{enumerate}
In all cases, $\Phi_{\bm{v}}$ and $\Psi_{\bm{v}}$ vanish simultaneously when one of $P_1$ and $P_2$ is evaluated at the singular point $P_*$.

<!-- lemma-3: body_md -->
\label{algebraic geometry}
For any odd prime $\theta$, $\Psi_{\bm{v}}^{*2}$ and $\Phi_{\bm{v}}^{*}$ do not share any common factor as elements in $\ensuremath{\mathbb{Z}}[x_1, x_2, y_1, y_2, A, B]/\langle \theta, f_1, f_2\rangle$, except $(x_2-x_1)$ possibly, where $f_i=y_i^2-x_i^3-Ax_i-B$ for $i=1, 2$.
<!-- end lemma-3 -->
<!-- proof-5: body_md -->
Suppose $F(x_1, x_2, y_1, y_2, A, B)\neq x_2-x_1$ is an irreducible common factor of $\Psi_{\bm{v}}^{*2}$ and $\Phi_{\bm{v}}^{*}$ in the quotient ring $\overline{\ensuremath{\mathbb{F}}_{\theta}}[x_1, x_2, y_1, y_2, A, B]/\langle f_1, f_2 \rangle$. If $P_1=(x_1, y_1), P_2=(x_2, y_2)$, then $f_1(P_1)=f_2(P_2)=0$.  Since we assumed $F$ is an irreducible common factor not associated to $x_2-x_1$, we may choose a point $(x_1, x_2, y_1, y_2, A, B) \in \ensuremath{\mathbb{V}}(f_1, f_2, F)$ with $x_2\neq x_1$. This means we have $P_1 \neq \pm P_2$ and if $F(P_1, P_2)=0$, by \Cref{Akbary TFAE} exactly one of $P_1$ and $P_2$ is singular $\bmod \theta$. Interchanging $P_1$ and $P_2$ if necessary, we may assume $P_1$ is singular $\bmod \theta$. This gives us $\partial f_1/\partial x_1 = \partial f_1/\partial y_1 = 0$, therefore showing 
\[
\ensuremath{\mathbb{V}}(f_1, f_2, F) \subset \ensuremath{\mathbb{V}}\left(\frac{\partial f_1}{\partial x_1}, \frac{\partial f_1}{\partial y_1}, f_1, f_2\right)
\]
over $\overline{\ensuremath{\mathbb{F}}_{\theta}}$. 

Now consider the $6$--dimensional affine space with coordinates $\{x_1, x_2, y_1, y_2, A, B\}$ over $\bar{\ensuremath{\mathbb{F}}}_{\theta}$. We know that $\ensuremath{\mathbb{V}}(f_1, f_2)$ is $4$--dimensional, so $\ensuremath{\mathbb{V}}(f_1, f_2, F)$ has dimension $4$ or $3$. However, also note that 
\begin{align*}
&\quad \ensuremath{\mathbb{V}}\left(\frac{\partial f_1}{\partial x_1}, \frac{\partial f_1}{\partial y_1}, f_1, f_2\right) \\
&=
\{(x_1, x_2, y_1, y_2, A, B) \in \mathbb{A}^6 \colon f_1(P_1, P_2)=f_2(P_1, P_2) =\frac{\partial f_1}{\partial x_1}(P_1, P_2) = \frac{\partial f_1}{\partial y_1}(P_1, P_2)= 0\} \\
&\subseteq 
\{(x_1, x_2, y_1, y_2, A, B) \in \mathbb{A}^6 \colon  -16(4A^3+27B^2)=0, y_2^2=x_2^3+Ax_2+B, -3x_1^2-A= 0, 2y_1= 0 \}.
\end{align*}
SInce $\overline{\ensuremath{\mathbb{F}}_{\theta}}$ is not a characteristic $2$ field, this shows that $\ensuremath{\mathbb{V}}(\frac{\partial f_1}{\partial x_1}, \frac{\partial f_1}{\partial y_1}, f_1, f_2)$ has dimension less than or equal to $2$ necessarily, so it is impossible to contain $\ensuremath{\mathbb{V}}(f_1, f_2, F)$ as a subset. By contradiction, such $F(x_1, x_2, y_1, y_2, A, B)$ cannot exist.

<!-- prop-8: body_md -->
\label{Shape}
Let $E$ be \textcolor{red}{a cubic} curve defined by the short Weierstrass equation (\ref{short EC}), so it has discriminant $\Delta=-16(4A^3+27B^2)$. Let $R_1, R_2$ be as in \Cref{R1 R2}. Then $\operatorname{Res}_{x_2}(R_1, R_2)(X)$ can be expressed as
\begin{equation*}
\operatorname{Res}_{x_2}(R_1, R_2)(X) = \pm X^{L_2(\bm{v})}\Delta^{L_1(\bm{v})}g_{t}(X),
\end{equation*}
where $L_i\colon \ensuremath{\mathbb{Z}}^2 \rightarrow \ensuremath{\mathbb{Z}}$ is a function in $v_1, v_2$ for $i=1, 2$, and $g_t(X) \in \ensuremath{\mathbb{Z}}[A, B, t][X]$ is a polynomial that \textcolor{red}{vanishes exactly on the closure of the set of $(A, B, X)$ such that $E$ is smooth and there exists} an appropriate pair of points $\bm{P}=(P_1, P_2) \in E(\overline{\ensuremath{\mathbb{Q}}})^2$ satisfying $x([v_1]P_1 \pm [v_2]P_2)=t$ and $[v_1]P_1 \mp [v_2]P_2=O$.
<!-- end prop-8 -->
<!-- proof-6: body_md -->
From previous sections, the resultant $\operatorname{Res}_{x_2}(R_1, R_2)(X)$ vanishes if and only if at least one of the following conditions holds:
\begin{enumerate}
\item \label{cond 1}There exists a pair of points $\bm{P}=(P_1, P_2) \in E(\overline{\ensuremath{\mathbb{Q}}})^2$ such that $X=x_2-x_1=0$, corresponding to $P_1=\pm P_2$. This follows from the definitions of $\Psi_{\bm{v}}^{*}$ and $\Phi_{\bm{v}}^{*}$ in (\ref{new poly}).
\item \label{cond 2}The curve $E\colon y^2=x^3+Ax+B$ is singular, or equivalently $\Delta=-16(4A^3+27B^2)=0$, as shown in \Cref{singular vanish}.
\item \label{cond 3}There exists a pair of appropriate points $\bm{P}=(P_1, P_2) \in E(\overline{\ensuremath{\mathbb{Q}}})^2$, we have $x([v_1]P_1 \pm [v_2]P_2)=t$ and $[v_1]P_1 \mp [v_2]P_2=O$. This follows from \Cref{zero condition}.
\end{enumerate}

\textcolor{red}{Now consider the ambient polynomial ring $\overline{\ensuremath{\mathbb{Q}}}[X, A, B]$ and the ambient space $\mathbb{A}_{\overline{\ensuremath{\mathbb{Q}}}}^3=\{(X, A, B)\colon X, A, B \in \overline{\ensuremath{\mathbb{Q}}}\}$. Conditions (\ref{cond 1}) and (\ref{cond 2}) define the Zariski closed sets $Z_1=\ensuremath{\mathbb{V}}(X)$ and $Z_2=\ensuremath{\mathbb{V}}(\Delta)$ respectively. Let $S$ denote the locus described by condition (\ref{cond 3}) on $X\Delta\neq0$. After extracting all powers of $X$ and $\Delta$ from $\operatorname{Res}_{x_2}(R_1, R_2)(X)$, denote the remaining factor by $g_t(X)$. By construction, $\ensuremath{\mathbb{V}}(g_t)$ is precisely the Zariski closure of $S$, since $S=\ensuremath{\mathbb{V}}(g_t)\setminus(Z_1\cup Z_2)$ and no irreducible component of $\ensuremath{\mathbb{V}}(g_t)$ can be contained in $Z_1\cup Z_2$.} Finally, \Cref{algebraic geometry} guarantees that there are no more other possible irreducible factors. Therefore, we have shown that 
\[
\operatorname{Res}_{x_2}(R_1, R_2)(X) = CX^{L_2(\bm{v})}\Delta^{L_1(\bm{v})}g_{t}(X), C\in \ensuremath{\mathbb{Z}}.
\]
Using the argument from \cite[Lemma 1.7.11(b)]{Connell} (or \cite{hindry_cassels} ), we claim that $C=\pm 1$. Suppose $C$ has an odd prime factor $p$, then $\operatorname{Res}_{x_2}(R_1, R_2)(X)$ vanishes identically $0$ in characteristic $p$. By \Cref{Akbary TFAE}, this forces $4A^3+27B^2$ to be a zero polynomial in $\ensuremath{\mathbb{F}}_p[A, B]$, which is clearly false. 

If $p=2$, the short Weierstrass equation (\ref{short EC}) is not an appropriate model, since every cubic curves of such form is singular in characteristic $2$. Instead, we apply the same argument to an integral general Weierstrass equation 
\[
E\colon  y^2+a_1xy+a_3y=x^3+a_2x^2+a_4x+a_6, a_i \in \ensuremath{\mathcal{O}}_K.
\]
The discriminant of the general Weierstrass equation cannot be a zero polynomial in characteristic $2$, otherwise all elliptic curves are singular in in characteristic $2$ (for example, $y^2+xy=x^3+1$ defines a nonsingular cubic over $\ensuremath{\mathbb{F}}_2$). Hence, the same argument rules out $2\mid C$ and we conclude that $C=\pm1$.

Hence, there is no compatibility issue in treating the case $p=2$ using an integral general Weierstrass equation.

<!-- lemma-4: body_md -->
\label{monic}
The polynomial $g_{t}(X) \in \ensuremath{\mathbb{Z}}[A, B, t][X]$ is a monic polynomial in the variable $X$.
<!-- end lemma-4 -->
<!-- proof-7: body_md -->
It suffices to prove that if $t, A, B$ are all algebraic integers and $X$ is a root of $g_t(X)$ in $\ensuremath{\mathbb{C}}$, then $X$ is an algebraic integer. We will prove by contradiction, so suppose $X$ is not an algebraic integer, then there exists a prime $\ensuremath{\mathfrak{q}} \in \ensuremath{\mathcal{O}}_{K(\bm{P})}$ such that $X \equiv \infty \bmod \ensuremath{\mathfrak{q}}$, where $K(\bm{P})$ is a finite extension of $K$ formed by adjoining the coordinates of $P_1, P_2 \in E(\overline{\ensuremath{\mathbb{Q}}})$. Without loss of generality, this implies that $P_1 \equiv \infty \bmod \ensuremath{\mathfrak{q}}$. From (\ref{zero condition}), if $X$ is a root of $g_t(X)$, then we know there are $P_1, P_2 \in E(\overline{\ensuremath{\mathbb{Q}}})$ that satisfy 
\[
x\left(\pm[v_1]P_1\pm[v_2]P_2\right)=t, \text{ and } \pm[v_1]P_1\mp[v_2]P_2=O.
\]
Together with $P_1 \equiv \infty \bmod \ensuremath{\mathfrak{q}}$, this implies that $t\equiv x\left([v_1]P_1+[v_2]P_2\right)\equiv \infty \bmod \ensuremath{\mathfrak{q}}$, so $t$ cannot be an algebraic integer.

<!-- prop-9: body_md -->
\label{local shape}
Let $\ensuremath{\mathfrak{q}}$ be a prime lying above $\ensuremath{\mathfrak{p}}$. For all $\overline{A}, \overline{B}, \in \ensuremath{\mathbb{F}}_{\ensuremath{\mathfrak{p}}}$ and $\overline{X}$ , $\overline{t} \in \overline{\ensuremath{\mathbb{F}}_{\ensuremath{\mathfrak{p}}}}$, we have $g_{\overline{t}}(\overline{X})=0$ if and only if there exists a pair of points $P_1, P_2 \in E(\overline{Q})$ such that $(\overline{A}, \overline{B}, \overline{P_1}, \overline{P_2})$ is in the Zariski closure of the union of the sets below.
\begin{equation}\label{local congruence}
\begin{aligned}
C_1&=\{\Delta(E)\not\equiv 0 \bmod \ensuremath{\mathfrak{q}}\}, \\
C_2&=\{[v_1]P_1\pm[v_2]P_2\equiv O \bmod \ensuremath{\mathfrak{q}}\}, \text{ and }\\
C_3&=\{x\left([v_1]P_1\mp[v_2]P_2\right)\equiv \overline{t} \bmod \ensuremath{\mathfrak{q}}\}.
\end{aligned}
\end{equation}
<!-- end prop-9 -->
<!-- remark-6: body_md -->
\Cref{Shape} guarantees $P_1, P_2 \in E(\overline{\ensuremath{\mathbb{Q}}})$ are a pair of appropriate, hence non-singular point. However, their reductions modulo $\ensuremath{\mathfrak{q}}$ need not be. Therefore, the expressions in (\ref{local congruence}) are interpreted by first performing the group operations in characteristic zero and then reducing the resulting points or coordinates modulo $\ensuremath{\mathfrak{q}}$.

<!-- proof-8: body_md -->
Let $\overline{A}, \overline{B}, \overline{X} \in \ensuremath{\mathbb{F}}_{\ensuremath{\mathfrak{p}}}$ and $\overline{t} \in \overline{\ensuremath{\mathbb{F}}_{\ensuremath{\mathfrak{p}}}}$ be such that $g_{\overline{t}}(\overline{X}) = 0$. We wish to show that there exist $A, B \in \ensuremath{\mathcal{O}}_K$ and $X, t \in \overline{\ensuremath{\mathbb{Z}}}$ such that $A \equiv \overline{A}$, $B \equiv \overline{B}$, and $X \equiv \overline{X} \bmod \ensuremath{\mathfrak{q}}$ for some prime ideal $\ensuremath{\mathfrak{q}}$ lying above $\ensuremath{\mathfrak{p}}$.

To construct $X$ and $\ensuremath{\mathfrak{q}}$, choose $Z \in \ensuremath{\mathcal{O}}_K$ such that $Z \equiv \overline{X} \bmod \ensuremath{\mathfrak{p}}$ and $g_{t}(Z) \equiv 0 \bmod \ensuremath{\mathfrak{p}}$. Consider the polynomial $g_t(Y + Z) \in \ensuremath{\mathbb{Z}}[A, B, t][Y]$. By \Cref{monic}, $g_t(Y + Z)$ is monic, so it factors over $\overline{\ensuremath{\mathbb{Q}}}$ as
\[
g_{t}(Y+Z) = \prod_{i=1}^{n} \left(Y - (Y_i - Z)\right),
\]
where $Y_1, \ldots, Y_n \in \overline{\ensuremath{\mathbb{Z}}}$ are the roots of $g_{t}(Y)$. Let $L = K(Y_1, \ldots, Y_n)$ be the splitting field over $K$, and let $\ensuremath{\mathfrak{q}} \subset \ensuremath{\mathcal{O}}_L$ be a prime ideal lying above $\ensuremath{\mathfrak{p}}$. Evaluating $g_t(Y + Z)$ at $Y = 0$ yields
\[
g_t(Z) = (-1)^n \prod_{i=1}^{n} (Y_i - Z).
\]
Since $g_t(Z) \equiv 0 \bmod \ensuremath{\mathfrak{p}}$, we have $g_t(Z) \in \ensuremath{\mathfrak{q}}$, which implies $\prod_{i=1}^{n} (Y_i - Z) \in \ensuremath{\mathfrak{q}}$ and there exists some $Y_0 \in \{Y_1, \ldots, Y_n\}$ such that $Y_0 - Z \equiv 0 \bmod \ensuremath{\mathfrak{q}}$. Setting $X = Y_0+Z \in \overline{\ensuremath{\mathbb{Z}}}$, we obtain 
\[
g_{t}(X)=g_t(Y_0+Z)=g(Y_0)=0 \text{ with } X\equiv Y_0+Z \equiv Z \bmod \ensuremath{\mathfrak{q}}.
\]
Having established suitable values $X, t \in \overline{\ensuremath{\mathbb{Z}}}$ and choose $A, B \in \ensuremath{\mathcal{O}}_K$ such that $E/K$ is defined by a minimal integral equation, we apply \Cref{Shape} to obtain a pair of points $(P_1, P_2) \in E(\overline{\ensuremath{\mathbb{Q}}})^2$ satisfying 
\[
x([v_1]P_1 \pm [v_2]P_2) = t \text{ and } [v_1]P_1 \mp [v_2]P_2 = O.
\]
Reducing these relations modulo $\ensuremath{\mathfrak{q}}$, we have found a set of $(A, B, X)$ that is contained in the Zariski closure of (\ref{local congruence}).

<!-- prop-10: body_md -->
\label{separate}
Let $k\in \ensuremath{\mathbb{N}}$ be a natural number, $\bm{v}=(v_1, v_2)\in \ensuremath{\mathbb{Z}}^2$ and $\bm{P}=(P_1, P_2) \in E(\overline{\ensuremath{\mathbb{Q}}})^2$. Let $X=x(P_2)-x(P_1)=x_2-x_1$. Suppose the reduction of $P_1$ and $P_2$ satisfies $\overline{P_1}\neq \pm \overline{P_2}$ and $\Psi_{\bm{v}}^{*}(\bm{P})\equiv \Phi_{\bm{v}}^{*}(\bm{P}) \equiv 0 \bmod \ensuremath{\mathfrak{p}}^k$ for a prime ideal $\ensuremath{\mathfrak{p}} \subset \ensuremath{\mathcal{O}}_K$, then there exists an algebraic integer $t \in \overline{\ensuremath{\mathbb{Z}}}$ such that no prime over $\ensuremath{\mathfrak{p}}$ divides $X^{L_2(\bm{v})}g_t(X)$.
<!-- end prop-10 -->
<!-- proof-9: body_md -->
Let $\nu$ be the valuation associated to the prime $\ensuremath{\mathfrak{p}}$ and assume $\Psi_{\bm{v}}^{*}(\bm{P})\equiv \Psi_{\bm{v}}^{*}(\bm{P}) \equiv 0 \bmod \ensuremath{\mathfrak{p}}^k$. Replacing $y_i$--variables using the Weierstrass equation $y_i^2=x_i^3+Ax_i+B$ will not reduce the $\ensuremath{\mathfrak{p}}$--valuation. Hence, by the definition of $R_1(X, x_2)$ and $R_2(X, x_2)$ in \Cref{R1 R2}, for every $t \in \ensuremath{\mathbb{Z}}$ we have $w(R_1), w(R_2)\geq k$ for some valuation $w$ extending $\nu$. This also implies that $w\left(X^{L_2(\bm{v})}\Delta^{L_1(\bm{v})}g_t(X)\right) \geq k$, which follows from \Cref{Shape}. Since we have also assumed the reduction of $P_1$ and $P_2$ satisfies $\overline{P_1}\neq \pm \overline{P_2}$, we have $w(X)=0$. Hence, it remains to show the existence of $t\in \overline{\ensuremath{\mathbb{Z}}}$ such that $w(g_t(X))=0$. 

We first assume that both $P_1$ and $P_2$ have good reduction modulo $\ensuremath{\mathfrak{p}}$, therefore $\overline{P_1}, \overline{P_2} \in E(\ensuremath{\mathbb{F}}_{\ensuremath{\mathfrak{p}}})$. 

Suppose there exists $t\in  \overline{\ensuremath{\mathbb{Z}}}$ such that $w(g_t(X))>0$, then by \Cref{local shape}, there exist $\overline{P_1}, \overline{P_2} \in E(\overline{\ensuremath{\mathbb{F}}_{\ensuremath{\mathfrak{p}}}})$ such that 
\begin{equation}\label{local congruence 2}
\begin{aligned}
\overline{x_2}-\overline{x_1}&=\overline{X}, \\
[v_1]\overline{P_1}\pm[v_2]\overline{P_2}&=\overline{O}, \\
x\left(\pm[v_1]\overline{P_1}\mp[v_2]\overline{P_2}\right)&=\overline{t},
\end{aligned}
\end{equation}
where $\overline{O}$ denotes the identity element on the reduced curve. 

Let $\ensuremath{\mathfrak{q}}$ be a prime lying above $\ensuremath{\mathfrak{p}}$; we split the proof into two cases.

\textbf{Case I:} $|v_1|\neq |v_2|$ \\
In this case, (\ref{local congruence 2}) tells us that 
\[
x_2\equiv X+x_1 \bmod \ensuremath{\mathfrak{q}} \text{ and } x\left([2v_1]P_1\right)\equiv x\left([2v_2]P_2\right)\equiv t \bmod \ensuremath{\mathfrak{q}}. 
\]
Using \Cref{net poly def} (with $n=1$, or equivalently the definition of division polynomials), the above congruence conditions can also be rewritten as 
\[
x\left([2v_1]P_1\right) \equiv \frac{\phi_{2v_1}(x_1)}{\psi_{2v_1}^2(x_1)}\equiv x\left([2v_2]P_2\right) \equiv \frac{\phi_{2v_2}(X+x_1)}{\psi_{2v_2}^2(X+x_1)} \equiv t \bmod \ensuremath{\mathfrak{q}}.
\]
From \cite[Theorem A]{AyadS}, we know that $\psi_{2v_1}^2(x_1)\equiv \psi_{2v_2}^2(X+x_1)\equiv 0 \bmod \ensuremath{\mathfrak{q}}$ if and only if one of the following holds:
	\begin{enumerate}
	\item $[2v_1]P_1 \equiv [2v_2]P_2 \equiv O \bmod \ensuremath{\mathfrak{q}}$; or 
	\item $P_1, P_2$ are both singular modulo $\ensuremath{\mathfrak{q}}$; or
	\item Without loss of generality,  $[2v_1]P_1\equiv O \bmod \ensuremath{\mathfrak{q}}$ and $P_2$ is singular modulo $\ensuremath{\mathfrak{q}}$. 
	\end{enumerate}
The first possibility cannot be true, as it implies $t \equiv \infty \bmod \ensuremath{\mathfrak{q}}$, violating the condition that $t \in \overline{\ensuremath{\mathbb{Z}}}$. If the second situation were to occur, then $X\equiv 0 \bmod \ensuremath{\mathfrak{q}}$ since there is only one singular point modulo $\ensuremath{\mathfrak{q}}$, contrary to our assumption. Finally, the last possibility implies that $t \in \overline{\ensuremath{\mathbb{Z}}}$ has to satisfy $t \equiv \infty \bmod \ensuremath{\mathfrak{q}}$ and $\overline{t}$ is the $x$--coordinate of the singular point modulo $\ensuremath{\mathfrak{q}}$ simultaneously, which cannot happen. Therefore, we may rearrange the above as
	\begin{equation}\label{local div poly}
	\phi_{2v_1}(x_1)\psi_{2v_2}^2(X+x_1)-\phi_{2v_2}(X+x_1)\psi_{2v_1}^2(x_1) \equiv 0 \bmod \ensuremath{\mathfrak{q}}.
	\end{equation}
Remain to show that this polynomial is nonzero. To do so, we recall from \cite[Chapter III, Exercise 3.7(b)]{Silverman1} that as polynomials in one variable, 
	\begin{align*}
	\phi_n(x)&=x^{n^2}+(\text{lower order terms}),\\
	\psi_n^2(x)&=n^2x^{n^2-1}+(\text{lower order terms}).
	\end{align*}

If $|v_1|\neq |v_2|$ and $X\equiv x(P_2)-x(P_1) \not\equiv 0 \bmod \ensuremath{\mathfrak{q}}$, then the polynomials in (\ref{local div poly}) cannot cancel out each other due to the difference in degree. Therefore, (\ref{local div poly}) is a polynomial in the variable $x_1$ over $\overline{\ensuremath{\mathbb{F}}_p}$ of degree at most $4v_1^2+4v_2^2-1$. Since $X$ is a fixed value, every solution of (\ref{local congruence 2}) immediately determines a value of $x_2 \bmod \ensuremath{\mathfrak{q}}$, hence $t \bmod \ensuremath{\mathfrak{q}}$ as well and therefore there are at most $4v_1^2+4v_2^2-1$ distinct values of $x_1$ as roots of the polynomial (\ref{local div poly}). This gives us a finite set of possible values for $\overline{t}\in \overline{\ensuremath{\mathbb{F}}_{\ensuremath{\mathfrak{p}}}}$ such that (\ref{local congruence 2}) is soluble for some $\overline{P_1}, \overline{P_2} \in E(\overline{\ensuremath{\mathbb{F}}_{\ensuremath{\mathfrak{p}}}})$, so we can simply choose $t\in \overline{\ensuremath{\mathbb{Z}}}$ that its reduction $\overline{t}$ is outside this finite set. 

\textbf{Case II:} $|v_1|=|v_2|=v$ \\
In this case, (\ref{local congruence 2}) tells us that there exist $\overline{P_1}, \overline{P_2} \in E(\overline{\ensuremath{\mathbb{F}}_{\ensuremath{\mathfrak{p}}}})$ such that 
\begin{align}
\overline{X}=\overline{x_2}-\overline{x_1} &\neq 0, \label{eq:X-nonzero}\\
[v](\overline{P_1}\pm \overline{P_2}) &= \overline{O}, \label{eq:plus-minus}\\
x\left([v](\overline{P_1}\mp \overline{P_2})\right) &= \overline{t}. \label{eq:t-condition}
\end{align}
Without loss of generality, we assume $[v](\overline{P_1}+ \overline{P_2}) \equiv O$ and $[v](\overline{P_1}- \overline{P_2})=\overline{t}$ with $\overline{X}$ being fixed here. Set $\overline{Q}=\overline{P_1}+\overline{P_2}$, then $\overline{Q} \in E[v]$. From (\ref{eq:X-nonzero}), $\overline{Q}\neq \overline{O}$, so there are at most $v^2-1$ choices for $\overline{Q}$. 

Next, for each choice of $\overline{Q} \in E[v]$, we consider the rational function
\[
f_{\overline{Q}}(\overline{P_1})=x(\overline{Q}-\overline{P_1})-x(\overline{P_1})-\overline{X} \in \overline{\ensuremath{\mathbb{F}}_{\ensuremath{\mathfrak{p}}}}(E)^*.
\]
This is indeed a rational function: we have shown that $\overline{Q}\neq \overline{O}$ and assumed $\overline{X}\neq0$, additionally the $x$--coordinate function and any nonzero constant function are rational functions.

As a rational function, $f_{\overline{Q}}(\overline{P_1})$ has $2$ poles at $\overline{P_1}=\overline{Q}$ and $\overline{P_1}=\overline{O}$ each respectively. Since we necessarily have $\deg(\operatorname{div}(f_{\overline{Q}}))=0$, this implies that $f_{\overline{Q}}(\overline{P_1})$ has $4$ zeros. Hence, for each choice of $Q$, there are at most $4$ distinct choices for $\overline{P_1}$ satisfying (\ref{eq:X-nonzero}) and overall a maximum of $4(v^2-1)$ choices for $\overline{P_1}$ satisfying all the congruence conditions simultaneously. Since $\overline{X}$ is fixed, every choice of $\overline{P_1}$ immediately determines a value of $\overline{P_2}$ using (\ref{eq:X-nonzero}). As in Case I, this gives us a finite set of possible values for $\overline{t}\in \overline{\ensuremath{\mathbb{F}}_{\ensuremath{\mathfrak{p}}}}$ such that (\ref{eq:t-condition}) is soluble for some $\overline{P_1}, \overline{P_2} \in E(\overline{\ensuremath{\mathbb{F}}_{\ensuremath{\mathfrak{p}}}})$, so we can simply choose $t\in \overline{\ensuremath{\mathbb{Z}}}$ that its reduction $\overline{t}$ is outside this finite set. 

Next, we handle the case where exactly one of $P_1$ and $P_2$ is singular mod $\ensuremath{\mathfrak{q}}$, say $P_2$. Denote the reduction of $P_1$ and $P_2$ modulo $\ensuremath{\mathfrak{q}}$ by $\overline{P_1}$ and $\overline{P_2}$ respectively. 

We have assumed that $\overline{X}\neq 0$, so $P_1$ cannot have singular reduction and therefore the group law of $E(\overline{\ensuremath{\mathbb{F}}_{\ensuremath{\mathfrak{p}}}})$ is applicable to $\overline{P_1}$. By \Cref{local shape}, $\overline{P_1}$ and $\overline{P_2}$ satisfy 

\begin{equation*}
\begin{aligned}
\overline{x_2}-\overline{x_1}&=\overline{X}, \\
\overline{[v_1]P_1\pm[v_2]P_2}&=\overline{O}, \\
\overline{x\left(\pm[v_1]P_1\mp[v_2]P_2\right)}&=\overline{t}.
\end{aligned}
\end{equation*}

Since $[v_1]\overline{P_1}$ is a non-singular point on $E(\overline{\ensuremath{\mathbb{F}}_{\ensuremath{\mathfrak{p}}}})$, so as $\overline{[v_2]P_2}$ by the second condition, therefore the above relations imply that we have $x([2v_1]P_1)\equiv t \bmod \ensuremath{\mathfrak{q}}$. Denote the singular point on $E(\overline{\ensuremath{\mathbb{F}}_{\ensuremath{\mathfrak{p}}}})$ by $P^*=(x^*, 0)$, then the assumption implies that $x(P_2)\equiv x^* \bmod \ensuremath{\mathfrak{q}}$ and therefore requiring $x(P_1)\equiv x^* -X \bmod \ensuremath{\mathfrak{q}}$. There are at most $2$ points on $E(\overline{\ensuremath{\mathbb{F}}_{\ensuremath{\mathfrak{p}}}})$ with $x(P_1)\equiv x^* -X \bmod \ensuremath{\mathfrak{q}}$, say $\overline{P_1}=\pm Q$, hence giving at most $2$ possible values for $\overline{t}\in \overline{\ensuremath{\mathbb{F}}_{\ensuremath{\mathfrak{p}}}}$ such that the above congruences is soluble. We may choose $t\in \overline{\ensuremath{\mathbb{Z}}}$ that its reduction $\overline{t}$ is outside this finite set, then we cannot have $w(g_t(X))>0$. 

Overall, we showed that for whenever $\overline{X}\neq0$ in $\overline{F_{\ensuremath{\mathfrak{p}}}}$, there exists $t\in \overline{\ensuremath{\mathbb{Z}}}$ such that $w(X^{L_2(\bm{v})}g_t(X))=0$. For this choice of $t \in \overline{\ensuremath{\mathbb{Z}}}$, we have the implication 
\[
w\left(X^{L_2(\bm{v})}\Delta^{L_1(\bm{v})}g_t(X)\right) \geq k \implies w\left(\Delta^{L_1(\bm{v})}\right) \geq k,
\]
therefore completing the proof.

\label{X=0 impossible}
If $\overline{X}=0$ in $\overline{F_{\ensuremath{\mathfrak{p}}}}$, it is possible that no such $t \in \overline{Z}$ exists. Specifically, if $|v_1|=|v_2|$, then the set of congruences in (\ref{local congruence 2}) collapses to a system of $2$ congruences only: 
\begin{align*}
\overline{x_2}-\overline{x_1}=\overline{X}&=0, \\
x\left([2v]\overline{P_1}\right)&=\overline{t}.  
\end{align*}
Furthermore, the multiplication map $[v]\colon E \rightarrow E$ is surjective, so any value $\overline{t} \in \overline{\ensuremath{\mathbb{F}}_{\ensuremath{\mathfrak{p}}}}$ is attained and our method fails. This is one of the reasons why we have the condition $P_1\not\equiv \pm P_2 \bmod \ensuremath{\mathfrak{p}}$ in our main theorem.

Let $\ensuremath{\mathfrak{p}}$ be a prime and $P_1, P_2 \in E(\overline{\ensuremath{\mathbb{Q}}})$ such that $P_1, P_2 \not\equiv O \bmod \ensuremath{\mathfrak{p}}$ and $P_1\not\equiv \pm P_2 \bmod \ensuremath{\mathfrak{p}}$. Under these assumptions, \cite[Proposition 1.6]{Akbary} asserts that $\nu\left(\Phi_{(v_1, v_2)}(\bm{P})\right), \nu\left(\Phi_{(v_1, v_2)}(\bm{P})\right)\geq 0$. If $\ensuremath{\mathfrak{p}}$ divides both $\Psi_{\bm{v}}^{*}$ and $\Phi_{\bm{v}}^{*}$, then \Cref{Akbary TFAE} tells us that $\ensuremath{\mathfrak{p}}$ divides the discriminant $\Delta$ necessarily.

Under the assumption of $P_1\pm P_2\not\equiv O \bmod \ensuremath{\mathfrak{p}}$, \Cref{separate} shows that if $P_1 \neq \pm P_2 \bmod \ensuremath{\mathfrak{p}}$, then there exists $t\in \overline{\ensuremath{\mathbb{Z}}}$ such that $g_{t}(X)$ and $\Delta$ cannot share positive $\ensuremath{\mathfrak{p}}$--valuation simultaneously. Hence, the valuation of $\Delta$ in $\operatorname{Res}_{x_2}(R_1, R_2)(X)$ bounds the common valuation of $\Psi_{\bm{v}}^{2}$ and $\Phi_{\bm{v}}$ associated to odd primes of bad reduction from above.

Note that $R_1(X, x_2)$ and $R_2(X, x_2)$ are homogenous in weight, where $X, x_2, A$ and $B$ have weights $2, 2, 4$ and $6$ respectively. Recall from \Cref{Section 4} that $\Psi_{\bm{v}}$ and $\Phi_{\bm{v}}$ are homogenous of weight $g(\bm{v})$ and $2(v_1^2+v_2^2-v_1v_2)$ respectively. Thus, we may deduce that
\begin{align*}
\operatorname{deg}_{x_2}(R_1)
&\leq \frac{1}{2}(8)\left((v_1^2+v_2^2-v_1v_2-1)+2(v_1^2+v_2^2-v_1v_2)\right) \\
&=4\left(3(v_1^2+v_2^2-v_1v_2)-1\right); \text{ and} \\
\operatorname{weight}(R_2)
&=4\cdot2(v_1^2+v_2^2-v_1v_2)=8(v_1^2+v_2^2-v_1v_2).
\end{align*}

Since $R_1$ and $R_2$ are both homogenous in weight, so is their resultant by \Cref{resultant properties}. Hence, we can bound the multiplicity of $\Delta$ in $\operatorname{Res}_{x_2}(R_1, R_2)(X)$ by assuming that $\Delta$ is the only term contributing to the weight. This bounds the multiplicity of $\Delta$ in $\operatorname{Res}_{x_2}(R_1, R_2)(X)$ from above by
\[
\frac{32}{12}\left(3(v_1^2+v_2^2-v_1v_2)-1\right)(v_1^2+v_2^2-v_1v_2).
\]
Using the ineuqality $2|v_1v_2|\leq v_1^2+v_2^2$, we obtain
\begin{align*}
\text{Multiplicity of $\Delta$ in $\operatorname{Res}_{x_2}(R_1, R_2)(X)$}
&\leq \frac{8}{3}\left(\frac{9}{2}v_1^2+\frac{9}{2}v_2^2-1\right)\left(\frac{3}{2}v_1^2+\frac{3}{2}v_2^2\right)\\
&\leq 18(v_1^2+v_2^2)^2-4(v_1^2+v_2^2).
\end{align*}
This proves the theorem.
