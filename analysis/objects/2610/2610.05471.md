\label{stabs11}
Let $\mathbb{F}$ be either $\mathbb{C}$ or $\mathbb{R}$, and let $\mathop{\rm G}\nolimits \subset \mathop{\rm GL}\nolimits_{n}(\mathbb{F})$ with $n=\sum_{j=1}^m n_j$.
For each $j\in \{1,\ldots,m\}$, let $H_j,\mathop{\rm G}\nolimits_j\subset \mathop{\rm GL}\nolimits_{n_j}(\mathbb{F})$, and set $H:=\bigoplus_{j=1}^m H_j$. Next, let 
 
\begin{equation*}
A=\bigoplus_{j=1}^{m}A_j^{}, \qquad
R=\bigoplus_{j=1}^{m}R_j^{},
\end{equation*}
 
where matrices $A_1,\ldots,A_m$ have pairwise disjoint spectra (i.e. $A_j$, $A_k$ have no common eigenvalues for $j\neq k$), and likewise, matrices $R_1,\ldots,R_m$ have pairwise disjoint spectra.
Moreover, suppose exactly one of the following conditions hold:

<!-- remark-1: body_md -->
\label{indef}
All nonsingular skew-symmetric matrices over $\mathbb{F}$ are congruent, and all nonsingular complex symmetric matrices are congruent; see
\cite[Corollaries 4.4.4, 4.4.19]{HornJohn},\cite[Theorem 2.5]{Thomp}.
Thus, by Proposition~\ref{conj}, centralizers of elements of an $H$-orthogonal/$H$-symplectic group can be transferred via congruence to the corresponding standard forms of $H$.
                                                                                                                                                           
Similarly, by Sylvester's inertia theorem, centralizers of elements of $\mathop{\rm U}\nolimits(n,H)$ are isomorphic to those of the corresponding elements of $\mathop{\rm U}\nolimits(p,q)$, where $H$ is congruent to $I_{p,q}$.

\label{eqATB2}

\end{align}

the equation

Next, obtain $A_j^{rr}$ by solving (\ref{eqATB2}). 
                                                                                                       
Its solution is $X=(A^{\star})^{-1}(Z+\frac{1}{2}B)$ with

$(A^{\star})^{-1}=
A_0^{rr}(B_{r})^{-1}$. Thus $Z$ parametrizes precisely the homogeneous solutions of the equation for $A_j^{rr}$, $j\ge1$. Moreover, $D_j^{r(r+p)}$ with $p\geq 1$ depends only on $A_n^{rs}$, with $n,r,s$ satisfying (\ref{induA1}).

(m_r^2-m_r)+(m_r^2+m_r)\big).
   
\end{align*}
\normalsize
 
Simplifying this expression yields the stated formula in (\ref{cdim}).

E \oplus E = & (X_1^{*} \oplus X_2^{*}) ( E \oplus E) (X_2 \oplus X_1)\nonumber\\

\end{align}


By conjugating both hand sides of the last equation of (\ref{QTQH2}) by $\Omega$ from Lemma \ref{lemaP}, recalling $\mathcal{F}=\Omega^{T}E\Omega$, and slightly manipulating it, we deduce
 
\begin{align}\label{ortoD3U}
  \Omega^{T}E \Omega = & (\Omega^{T} X_1\Omega)^{*}(\Omega^{T}E\Omega)(\Omega^{T} X_2\Omega)\\
\mathcal{F}= &  \mathcal{X}^{*}_1\mathcal{F} \mathcal{X}_2 \qquad \qquad (\mathcal{X}_1:=\Omega^{T} X_1\Omega, \quad \mathcal{X}_2:=\Omega^{T} X_2\Omega),\nonumber
\end{align}
  
The form of the solution $X=X_1\oplus X_2$ of (\ref{wJQQwJ}) yields $\mathcal{X}_1, \mathcal{X}_2\in \mathbb{T}^{\alpha,\mu}$ (cf. (\ref{0T0}).

$Q$
                                
is of the form
 
\begin{align}\label{QPsiU}
&Q=\widetilde{\Omega}(X_1\oplus X_2)\widetilde{\Omega}^{T}
=\Psi^{-1} \big(\mathcal{X}_1\oplus (\mathcal{X}^{*}_1)^{-1}\big)\Psi, \qquad \mathcal{X}_1\in \mathbb{T}^{\alpha,\mu},\\
&\qquad \Psi:=(I\oplus \mathcal{F})\big((\Omega^{T}\oplus \Omega^{T})\widetilde{\Omega}^{T}\big).\nonumber
\end{align}
 

\item           
Let the $H$-Hermitian normal form $\mathcal{A}$ with one eigenvalue $\lambda\in \mathbb{R}$ be:
 
\begin{align}
             
\mathcal{A}
=\bigoplus_{r=1}^{N}\Big( \bigoplus_{j=1}^{m_r}  \big( J_{\alpha_r}(\lambda)\big)\Big),\qquad

\end{align}

\begin{align}\label{ortoD2U}
                                                                                                                                                                                                       
&\mathcal{D} =  \mathcal{F}\mathcal{Q}^{*}\mathcal{F}\mathcal{D}\mathcal{Q},\\

\big)

\end{align}

\quad

\end{enumerate}
