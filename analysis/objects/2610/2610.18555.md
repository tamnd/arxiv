<!-- lemma-1: body_md -->
\label{lem:coco}
If $f$ is convex and $L$-smooth, then, for all $x,y\in\mathbb R^d$,
\begin{align}
f(x)&\ge f(y)+\langle \nabla f(y),x-y\rangle
+\frac{\lVert \nabla f(x)-\nabla f(y)\rVert^2}{2L},\label{eq:strongsmooth}\\
\langle \nabla f(x)-\nabla f(y),x-y\rangle
&\ge\frac{\lVert \nabla f(x)-\nabla f(y)\rVert^2}{L}.\label{eq:coco}
\end{align}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Fix $y$ and let $\psi(z)=f(z)-\langle \nabla f(y),z\rangle$. This function
is convex and $L$-smooth, with $\nabla\psi(y)=0$. Therefore
\[
\psi(z)\ge\psi(y)+\langle \nabla\psi(y),z-y\rangle=\psi(y),\qquad z\in\mathbb R^d.
\]
Applying this inequality and then smoothness at $x$ gives
\begin{align*}
\psi(y)
&\le\psi\left(x-\frac1L\nabla\psi(x)\right)\\
&\le\psi(x)+\left\langle\nabla\psi(x),-\frac1L\nabla\psi(x)\right\rangle
+\frac L2\left\|\frac1L\nabla\psi(x)\right\|^2\\
&=\psi(x)-\frac1{2L}\lVert \nabla\psi(x)\rVert^2.
\end{align*}
Substituting the definition of $\psi$ yields
\begin{align*}
f(y)-\langle \nabla f(y),y\rangle
&\le f(x)-\langle \nabla f(y),x\rangle
-\frac1{2L}\lVert \nabla f(x)-\nabla f(y)\rVert^2,\\
f(x)-f(y)
&\ge\langle \nabla f(y),x-y\rangle
+\frac1{2L}\lVert \nabla f(x)-\nabla f(y)\rVert^2.
\end{align*}
This proves \eqref{eq:strongsmooth}. Interchanging $x$ and $y$ and
adding the resulting inequalities gives
\begin{align*}
0
&\ge \langle \nabla f(y),x-y\rangle+\langle \nabla f(x),y-x\rangle
+\frac1L\lVert \nabla f(x)-\nabla f(y)\rVert^2\\
&=-\langle \nabla f(x)-\nabla f(y),x-y\rangle
+\frac1L\lVert \nabla f(x)-\nabla f(y)\rVert^2,
\end{align*}
which is \eqref{eq:coco}.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:proxstep}
Let $F=f+r$ satisfy Assumption~\ref{ass:convex}. For $x\in\operatorname{dom} r$,
$h\in\partial f(x)$, an arbitrary error $e$, and $\alpha>0$, define
$x^+=\operatorname{prox}_{\alpha r}(x-\alpha(h+e))$.
If $f$ is $L$-smooth and $\alpha\le1/(2L)$, then, for every $u\in\operatorname{dom} r$,
\begin{equation}\label{eq:prox-smooth}
\alpha(F(x^+)-F(u))
\le D(u,x)-D(u,x^+)+\alpha\langle e,u-x\rangle+\alpha^2\lVert e\rVert^2.
\end{equation}
If $f$ is $G$-Lipschitz, then for every $\alpha>0$,
\begin{equation}\label{eq:prox-nonsmooth}
\alpha(F(x^+)-F(u))
\le D(u,x)-D(u,x^+)+\alpha\langle e,u-x\rangle
+\alpha^2(\lVert e\rVert^2+4G^2).
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
The optimality condition for the proximal update gives
\[
\frac{x-x^+}{\alpha}-h-e\in\partial r(x^+).
\]
By the subgradient inequality,
\begin{align*}
r(u)&\ge r(x^+)+\left\langle\frac{x-x^+}{\alpha}-h-e,u-x^+\right\rangle,\\
\alpha\bigl(r(x^+)-r(u)+\langle h+e,x^+-u\rangle\bigr)
&\le\langle x-x^+,x^+-u\rangle.
\end{align*}
Expanding the square gives
\[
\lVert x-u\rVert^2=\lVert x-x^+\rVert^2+\lVert x^+-u\rVert^2
+2\langle x-x^+,x^+-u\rangle.
\]
Consequently,
\begin{equation}\label{eq:prox-threepoint}
\alpha\bigl(r(x^+)-r(u)+\langle h+e,x^+-u\rangle\bigr)
\le D(u,x)-D(u,x^+)-\frac12\lVert x^+-x\rVert^2.
\end{equation}

For smooth $f$, we have $h=\nabla f(x)$. Smoothness and convexity,
respectively, give
\begin{align*}
f(x^+)-f(x)&\le\langle h,x^+-x\rangle+\frac L2\lVert x^+-x\rVert^2,\\
f(x)-f(u)&\le\langle h,x-u\rangle.
\end{align*}
Adding these two inequalities, we obtain
\begin{align*}
f(x^+)-f(u)
&=f(x^+)-f(x)+f(x)-f(u)\\
&\le\langle h,x^+-x\rangle+\langle h,x-u\rangle+\frac L2\lVert x^+-x\rVert^2\\
&=\langle h,x^+-u\rangle+\frac L2\lVert x^+-x\rVert^2.
\end{align*}
Combining this with \eqref{eq:prox-threepoint} yields
\begin{align*}
\alpha(F(x^+)-F(u))
&\le D(u,x)-D(u,x^+)-\frac{1-L\alpha}{2}\lVert x^+-x\rVert^2
+\alpha\langle e,u-x^+\rangle\\
&=D(u,x)-D(u,x^+)+\alpha\langle e,u-x\rangle
-\frac{1-L\alpha}{2}\lVert x^+-x\rVert^2-\alpha\langle e,x^+-x\rangle\\
&\le D(u,x)-D(u,x^+)+\alpha\langle e,u-x\rangle
-\frac14\lVert x^+-x\rVert^2+\alpha\lVert e\rVert\lVert x^+-x\rVert\\
&=D(u,x)-D(u,x^+)+\alpha\langle e,u-x\rangle+\alpha^2\lVert e\rVert^2
-\left(\frac{\lVert x^+-x\rVert}2-\alpha\lVert e\rVert\right)^2\\
&\le D(u,x)-D(u,x^+)+\alpha\langle e,u-x\rangle+\alpha^2\lVert e\rVert^2.
\end{align*}
The second inequality uses $L\alpha\le1/2$ and Cauchy--Schwarz.

For $G$-Lipschitz $f$, first note that $\lVert h\rVert\le G$. If
$h\ne0$, then for any $a>0$,
\[
a\lVert h\rVert
\le f\left(x+a\frac h{\lVert h\rVert}\right)-f(x)
\le Ga.
\]
Hence, we can ensure that
\begin{align*}
f(x^+)-f(u)
&=f(x^+)-f(x)+f(x)-f(u)\\
&\le G\lVert x^+-x\rVert+\langle h,x-u\rangle\\
&=\langle h,x^+-u\rangle+\langle h,x-x^+\rangle+G\lVert x^+-x\rVert\\
&\le\langle h,x^+-u\rangle+2G\lVert x^+-x\rVert.
\end{align*}
Using \eqref{eq:prox-threepoint} once more gives
\begin{align*}
\alpha(F(x^+)-F(u))
&\le D(u,x)-D(u,x^+)+\alpha\langle e,u-x^+\rangle
+2\alpha G\lVert x^+-x\rVert-\frac12\lVert x^+-x\rVert^2\\
&\le D(u,x)-D(u,x^+)+\alpha\langle e,u-x\rangle
+\alpha(\lVert e\rVert+2G)\lVert x^+-x\rVert-\frac12\lVert x^+-x\rVert^2\\
&=D(u,x)-D(u,x^+)+\alpha\langle e,u-x\rangle
+\frac{\alpha^2}2(\lVert e\rVert+2G)^2\\
&\quad-\frac12\bigl(\lVert x^+-x\rVert-\alpha(\lVert e\rVert+2G)\bigr)^2\\
&\le D(u,x)-D(u,x^+)+\alpha\langle e,u-x\rangle
+\alpha^2(\lVert e\rVert^2+4G^2),
\end{align*}
where $(\lVert e\rVert+2G)^2\le2\lVert e\rVert^2+8G^2$ proves the last step.
<!-- end proof-2 -->
<!-- lemma-3: body_md -->
\label{lem:radius}
Under Assumptions~\ref{ass:convex} and \ref{ass:smooth}, let
$x^+=\operatorname{prox}_{\alpha r}(x-\alpha(\nabla f(x)+e))$
and set $0<\alpha\le1/L$. Then
\begin{equation}\label{eq:radiusrecursion}
\lVert x^+-x^\star\rVert^2
\le\lVert x-x^\star\rVert^2-2\alpha\langle e,x-x^\star\rangle+2\alpha^2\lVert e\rVert^2.
\end{equation}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
For $u=\operatorname{prox}_{\alpha r}(a)$ and $v=\operatorname{prox}_{\alpha r}(b)$, the proximal
optimality conditions and the subgradient inequalities imply
\begin{align*}
\frac{a-u}{\alpha}&\in\partial r(u),
&\frac{b-v}{\alpha}&\in\partial r(v),\\
r(v)-r(u)&\ge\frac1\alpha\langle a-u,v-u\rangle,
&r(u)-r(v)&\ge\frac1\alpha\langle b-v,u-v\rangle.
\end{align*}
Adding the last two inequalities and rearranging gives
\begin{align*}
0&\le\langle a-b-(u-v),u-v\rangle,\\
\lVert u-v\rVert^2&\le\langle a-b,u-v\rangle\le\lVert a-b\rVert\lVert u-v\rVert,\\
\lVert \operatorname{prox}_{\alpha r}(a)-\operatorname{prox}_{\alpha r}(b)\rVert&\le\lVert a-b\rVert.
\end{align*}
The last line also holds if $u=v$. Since $f$ is differentiable on
$\mathbb R^d$, optimality of $x^\star$ gives
\[
-\nabla f(x^\star)\in\partial r(x^\star),\qquad
x^\star=\operatorname{prox}_{\alpha r}(x^\star-\alpha\nabla f(x^\star)).
\]
Applying the preceding nonexpansiveness inequality, we obtain
\begin{align*}
\lVert x^+-x^\star\rVert^2
&\le\lVert x-x^\star-\alpha(\nabla f(x)-\nabla f(x^\star)+e)\rVert^2\\
&=\lVert x-x^\star\rVert^2
-2\alpha\langle \nabla f(x)-\nabla f(x^\star),x-x^\star\rangle
-2\alpha\langle e,x-x^\star\rangle\\
&\quad+\alpha^2\lVert \nabla f(x)-\nabla f(x^\star)+e\rVert^2\\
&\le\lVert x-x^\star\rVert^2-2\alpha\langle e,x-x^\star\rangle
-\frac{2\alpha}{L}\lVert \nabla f(x)-\nabla f(x^\star)\rVert^2\\
&\quad+2\alpha^2\lVert \nabla f(x)-\nabla f(x^\star)\rVert^2+2\alpha^2\lVert e\rVert^2\\
&=\lVert x-x^\star\rVert^2-2\alpha\langle e,x-x^\star\rangle+2\alpha^2\lVert e\rVert^2
+2\alpha\left(\alpha-\frac1L\right)\lVert \nabla f(x)-\nabla f(x^\star)\rVert^2\\
&\le\lVert x-x^\star\rVert^2-2\alpha\langle e,x-x^\star\rangle+2\alpha^2\lVert e\rVert^2.
\end{align*}
Here the second inequality uses \eqref{eq:coco} and
$\lVert a+b\rVert^2\le2\lVert a\rVert^2+2\lVert b\rVert^2$, and the last uses $\alpha\le1/L$.

<!-- lemma-4: body_md -->
\label{lem:prefix}
Let $F$ satisfy Assumption~\ref{ass:convex}, let $x_t\in\operatorname{dom} r$ for
$1\le t\le T+1$, and write $\Delta_t=F(x_t)-F^\star$.
Suppose that, for every $u\in\operatorname{dom} r$
and $1\le t\le T$,
\begin{equation}\label{eq:abstractstep}
\eta_t(F(x_{t+1})-F(u))
\le D(u,x_t)-D(u,x_{t+1})+\eta_t\langle e_t,u-x_t\rangle+\rho_t,
\end{equation}
where $\rho_t\ge0$. Then, for every $1\le s\le T$,
\begin{align}
P_s+\eta_sv_s\Delta_{s+1}
+S_{s+1}\sum_{j=2}^s(v_j-v_{j-1})\Delta_j
\le P_0+\sum_{t=1}^s v_t\rho_t
+\sum_{t=1}^s\langle e_t,A_t\rangle,\label{eq:prefix}
\end{align}
where $A_t=\eta_tv_{t-1}(z_{t-1}-x_t)$.
Every term on the left-hand side is nonnegative.
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
By equation \eqref{eq:potential}, we know that
\begin{align*}
z_t-x_t&=\frac{v_{t-1}}{v_t}(z_{t-1}-x_t),\\
v_tD(z_t,x_t)
&=\frac{v_{t-1}^2}{v_t}D(z_{t-1},x_t)
\le v_{t-1}D(z_{t-1},x_t)=P_{t-1},\\
\eta_tv_t\langle e_t,z_t-x_t\rangle
&=\eta_tv_{t-1}\langle e_t,z_{t-1}-x_t\rangle=\langle e_t,A_t\rangle.
\end{align*}
Putting $u=z_t$ in \eqref{eq:abstractstep} and multiplying by $v_t$ gives
\begin{equation}\label{eq:prefix-onestep}
P_t+\eta_tv_t(F(x_{t+1})-F(z_t))
\le P_{t-1}+v_t\rho_t+\langle e_t,A_t\rangle.
\end{equation}
To expand $F(z_t)$, first telescope the recursion for $z_t$:
\begin{align*}
v_tz_t-v_{t-1}z_{t-1}&=(v_t-v_{t-1})x_t,\\
v_tz_t-v_0x^\star&=\sum_{j=1}^t(v_j-v_{j-1})x_j,\\
v_0+\sum_{j=1}^t(v_j-v_{j-1})&=v_t.
\end{align*}
Convexity therefore gives
\begin{align*}
v_tF(z_t)
&\le v_0F^\star+\sum_{j=1}^t(v_j-v_{j-1})F(x_j),\\
v_t(F(z_t)-F^\star)
&\le\sum_{j=1}^t(v_j-v_{j-1})\Delta_j,\\
v_t(F(x_{t+1})-F(z_t))
&\ge v_t\Delta_{t+1}-\sum_{j=1}^t(v_j-v_{j-1})\Delta_j.
\end{align*}
Substitution in \eqref{eq:prefix-onestep} and summation over $t=1,\ldots,s$
yield
\begin{align}
P_s+\sum_{t=1}^s\eta_tv_t\Delta_{t+1}
-\sum_{t=1}^s\eta_t\sum_{j=1}^t(v_j-v_{j-1})\Delta_j
\le P_0+\sum_{t=1}^s v_t\rho_t+\sum_{t=1}^s\langle e_t,A_t\rangle.
\label{eq:prefix-summed}
\end{align}
The two sums containing $\Delta_j$ have the following exact expansions:
\begin{align*}
\sum_{t=1}^s\eta_tv_t\Delta_{t+1}
&=\eta_sv_s\Delta_{s+1}+\sum_{j=2}^s\eta_{j-1}v_{j-1}\Delta_j,\\
\sum_{t=1}^s\eta_t\sum_{j=1}^t(v_j-v_{j-1})\Delta_j
&=\sum_{j=1}^s(v_j-v_{j-1})\Delta_j\sum_{t=j}^s\eta_t\\
&=\sum_{j=2}^s(v_j-v_{j-1})(S_j-S_{s+1})\Delta_j.
\end{align*}
The last equality uses $v_1=v_0$. For $2\le j\le s$,
\begin{align*}
S_{j-1}-S_j&=\eta_{j-1},\\
v_j-v_{j-1}
&=\eta_T\left(\frac1{S_j}-\frac1{S_{j-1}}\right)
=\frac{\eta_T\eta_{j-1}}{S_jS_{j-1}},\\
\eta_{j-1}v_{j-1}-(v_j-v_{j-1})(S_j-S_{s+1})
&=S_j(v_j-v_{j-1})-(S_j-S_{s+1})(v_j-v_{j-1})\\
&=S_{s+1}(v_j-v_{j-1}).
\end{align*}
Consequently,
\begin{align*}
\sum_{t=1}^s\eta_tv_t\Delta_{t+1}
-\sum_{t=1}^s\eta_t\sum_{j=1}^t(v_j-v_{j-1})\Delta_j
&=\eta_sv_s\Delta_{s+1}
+S_{s+1}\sum_{j=2}^s(v_j-v_{j-1})\Delta_j.
\end{align*}
This proves \eqref{eq:prefix}. Its left-hand side is nonnegative because
$P_s\ge0$, $\Delta_j\ge0$, $S_{s+1}\ge0$, and $v_j\ge v_{j-1}$.

<!-- lemma-5: body_md -->
\label{lem:weights}
For $\eta_t=\eta(T-t+1)/T$, with $\eta>0$, the quantities in
\eqref{eq:potential} satisfy
\begin{equation}\label{eq:weights}
v_t=\frac2{k(k+1)},\quad
P_0\le\frac{R^2}{T(T+1)}\le\frac{R^2}{T^2},\quad
\eta_t^2v_t\le\frac{2\eta^2}{T^2},\quad
\eta_t\sqrt{v_{t-1}}\le\frac{\sqrt2\eta}{T},
\end{equation}
where $k=T-t+1$. Moreover, $v_T=1$ and $\eta_T=\eta/T$.
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
For $k=T-t+1$, the sum defining $S_t$ becomes
\begin{align*}
S_t&=\frac\eta T\sum_{j=t}^T(T-j+1)
=\frac\eta T\sum_{i=1}^k i=\frac{\eta k(k+1)}{2T},\\
v_t&=\frac{\eta/T}{\eta k(k+1)/(2T)}=\frac2{k(k+1)},\\
P_0&=\frac2{T(T+1)}\frac{\lVert x^\star-x_1\rVert^2}{2}
\le\frac{R^2}{T(T+1)}\le\frac{R^2}{T^2},\\
\eta_t^2v_t&=\frac{\eta^2k^2}{T^2}\frac2{k(k+1)}
=\frac{2\eta^2}{T^2}\frac{k}{k+1}\le\frac{2\eta^2}{T^2}.
\end{align*}
Since $v_{t-1}\le v_t$, the last bound also gives
\[
\eta_t\sqrt{v_{t-1}}\le\eta_t\sqrt{v_t}
=\sqrt{\eta_t^2v_t}\le\frac{\sqrt2\eta}{T}.
\]
Putting $t=T$ in the definitions gives $v_T=1$ and $\eta_T=\eta/T$.
All these calculations include $T=1$.

<!-- lemma-6: body_md -->
\label{lem:clip}
Let $\lambda>0$, $1<p\le2$, and let $h$ be measurable with respect to the conditioning sigma-field,
with $\lVert h\rVert\le\lambda/2$. Suppose that $g=h+\varepsilon$,
$\mathbb E_t\varepsilon=0$, and $\mathbb E_t\lVert \varepsilon\rVert^p\le\sigma^p$.
Define $e=\operatorname{clip}(g,\lambda)-h$, $b=\mathbb E_t e$, and $\xi=e-b$. Then
\begin{align}
\lVert e\rVert&\le\frac32\lambda,
&\mathbb E_t\lVert e\rVert^2&\le V:=2\sigma^p\lambda^{2-p},\label{eq:clipsecond}\\
\lVert b\rVert&\le B:=2\sigma^p\lambda^{1-p},
&\lVert \xi\rVert&\le2\lambda,
&\mathbb E_t\lVert \xi\rVert^2&\le V.\label{eq:clipbias}
\end{align}
<!-- end lemma-6 -->
<!-- proof-6: body_md -->
Clipping is Euclidean projection onto the ball of radius $\lambda$.
Its nonexpansiveness follows from the proximal argument in
Lemma~\ref{lem:radius}, with $r$ the indicator of that ball. Since
$\operatorname{clip}(h,\lambda)=h$,
\begin{align*}
\lVert e\rVert
&=\lVert \operatorname{clip}(g,\lambda)-\operatorname{clip}(h,\lambda)\rVert\le\lVert g-h\rVert=\lVert \varepsilon\rVert,\\
\lVert e\rVert
&\le\lVert \operatorname{clip}(g,\lambda)\rVert+\lVert h\rVert\le\lambda+\lambda/2=3\lambda/2,\\
\mathbb E_t\lVert e\rVert^2
&\le(3\lambda/2)^{2-p}\mathbb E_t\lVert e\rVert^p
\le(3\lambda/2)^{2-p}\mathbb E_t\lVert \varepsilon\rVert^p
\le2\sigma^p\lambda^{2-p}.
\end{align*}
The last line uses $0\le2-p<1$; when $p=2$, its first inequality is
simply an equality.

For the bias, $\mathbb E_t(g-h)=0$ implies
\[
b=\mathbb E_t[\operatorname{clip}(g,\lambda)-g],\qquad
\lVert \operatorname{clip}(g,\lambda)-g\rVert=(\lVert g\rVert-\lambda)\mathbf 1_{\{\lVert g\rVert>\lambda\}}.
\]
On $\{\lVert g\rVert>\lambda\}$, the triangle inequality gives
\[
\lVert \varepsilon\rVert\ge\lVert g\rVert-\lVert h\rVert>\lambda/2,
\qquad \lVert g\rVert-\lambda\le\lVert \varepsilon\rVert+\lVert h\rVert-\lambda\le\lVert \varepsilon\rVert.
\]
Thus
\begin{align*}
\lVert b\rVert
&\le\mathbb E_t\lVert \operatorname{clip}(g,\lambda)-g\rVert\\
&\le\mathbb E_t\bigl[\lVert \varepsilon\rVert\mathbf 1_{\{\lVert \varepsilon\rVert>\lambda/2\}}\bigr]\\
&=\mathbb E_t\bigl[\lVert \varepsilon\rVert^p\lVert \varepsilon\rVert^{1-p}
\mathbf 1_{\{\lVert \varepsilon\rVert>\lambda/2\}}\bigr]\\
&\le(\lambda/2)^{1-p}\mathbb E_t\lVert \varepsilon\rVert^p
\le2\sigma^p\lambda^{1-p}.
\end{align*}
The product in the third line is evaluated only on
$\{\lVert \varepsilon\rVert>\lambda/2\}$. Finally,
\begin{align*}
\xi&=\operatorname{clip}(g,\lambda)-\mathbb E_t\operatorname{clip}(g,\lambda),
&\mathbb E_t\xi&=0,\\
\lVert \xi\rVert&\le\lVert \operatorname{clip}(g,\lambda)\rVert+\mathbb E_t\lVert \operatorname{clip}(g,\lambda)\rVert\le2\lambda,\\
\mathbb E_t\lVert \xi\rVert^2
&=\mathbb E_t\bigl[\lVert e\rVert^2-2\langle e,b\rangle+\lVert b\rVert^2\bigr]\\
&=\mathbb E_t\lVert e\rVert^2-2\langle \mathbb E_t e,b\rangle+\lVert b\rVert^2
=\mathbb E_t\lVert e\rVert^2-\lVert b\rVert^2\le V.
\end{align*}
Here $b$ is measurable before the current sample is observed.

<!-- lemma-7: body_md -->
\label{lem:nonnegative}
Suppose $X_t$ is $\mathcal F_t$-measurable, $0\le X_t\le H$ almost surely,
and $\mathbb E_t X_t\le m_t$, where $m_t\ge0$ is $\mathcal F_{t-1}$-measurable and
$\sum_{t=1}^T m_t\le A$ almost surely for a deterministic $A$.
For $u>0$,
\[
\mathbb P\left\{\sum_{t=1}^T X_t\le2A+Hu\right\}\ge1-e^{-u}.
\]
<!-- end lemma-7 -->
<!-- proof-7: body_md -->
If $H=0$, then $X_t=0$ almost surely. For $H>0$, convexity of the
exponential gives, for $0\le x\le H$,
\[
\exp(x/H)\le(1-x/H)\exp(0)+(x/H)\exp(1)
=1+(\mathrm e-1)x/H.
\]
Consequently,
\[
\mathbb E_t\exp(X_t/H)\le1+(\mathrm e-1)\mathbb E_t X_t/H
\le1+(\mathrm e-1)m_t/H\le\exp((\mathrm e-1)m_t/H).
\]
Set
\[
Z_s=\exp\left(\frac1H\sum_{t=1}^s X_t
-\frac{\mathrm e-1}{H}\sum_{t=1}^s m_t\right),\qquad Z_0=1.
\]
Since $Z_{s-1}$ and $m_s$ are $\mathcal F_{s-1}$-measurable,
\begin{align*}
\mathbb E[Z_s\mid\mathcal F_{s-1}]
&=Z_{s-1}\exp\left(-\frac{\mathrm e-1}{H}m_s\right)
\mathbb E[\exp(X_s/H)\mid\mathcal F_{s-1}]\le Z_{s-1},\\
\mathbb E Z_T&\le\mathbb E Z_{T-1}\le\cdots\le\mathbb E Z_0=1.
\end{align*}
Using $\sum m_t\le A$, $\mathrm e-1<2$, and Markov's inequality,
\begin{align*}
\mathbb P\left\{\sum_{t=1}^T X_t>2A+Hu\right\}
&\le\mathbb P\left\{\sum_{t=1}^T X_t>(\mathrm e-1)A+Hu\right\}\\
&\le\mathbb P\{Z_T>e^u\}\le e^{-u}\mathbb E Z_T\le e^{-u}.
\end{align*}

<!-- lemma-8: body_md -->
\label{lem:freedman}
Let $M_s=\sum_{t=1}^s Y_t$, where $Y_t$ is $\mathcal F_t$-measurable,
$\mathbb E_t Y_t=0$, $|Y_t|\le a$, and $\sum_{t=1}^T\mathbb E_t Y_t^2\le w$ almost
surely for deterministic $a,w\ge0$. For every $u>0$,
\begin{equation}\label{eq:freedman}
\mathbb P\left\{\max_{1\le s\le T}M_s\le\sqrt{2wu}+2au\right\}\ge1-e^{-u}.
\end{equation}
<!-- end lemma-8 -->
<!-- proof-8: body_md -->
If $a=0$, all increments vanish. If $w=0$, then
$\sum_t\mathbb E Y_t^2=\mathbb E\sum_t\mathbb E_t Y_t^2=0$, so the same conclusion holds.
Assume $a,w>0$ and fix $0<\theta<3/a$. For $k\ge2$,
$k!\ge2\cdot3^{k-2}$ and $|Y_t|^k\le a^{k-2}Y_t^2$. Hence
\begin{align*}
\mathbb E_t e^{\theta Y_t}
&=1+\theta\mathbb E_t Y_t+\sum_{k=2}^\infty\frac{\theta^k}{k!}\mathbb E_t Y_t^k\\
&\le1+\sum_{k=2}^\infty\frac{\theta^ka^{k-2}}{k!}\mathbb E_t Y_t^2\\
&\le1+\frac{\theta^2}2\mathbb E_t Y_t^2\sum_{k=2}^\infty(\theta a/3)^{k-2}\\
&=1+\frac{\theta^2\mathbb E_t Y_t^2}{2(1-\theta a/3)}
\le\exp\left(\frac{\theta^2\mathbb E_t Y_t^2}{2(1-\theta a/3)}\right).
\end{align*}
Boundedness of $Y_t$ justifies taking conditional expectations of
the absolutely convergent series. Define
\[
W_s=\sum_{t=1}^s\mathbb E_t Y_t^2,\qquad
Z_s=\exp\left(\theta M_s-\frac{\theta^2W_s}{2(1-\theta a/3)}\right),
\qquad Z_0=1.
\]
The preceding conditional bound gives
\[
\mathbb E[Z_s\mid\mathcal F_{s-1}]
=Z_{s-1}\exp\left(-\frac{\theta^2\mathbb E[Y_s^2\mid\mathcal F_{s-1}]}{2(1-\theta a/3)}\right)
\mathbb E[e^{\theta Y_s}\mid\mathcal F_{s-1}]
\le Z_{s-1}.
\]
For $c>1$, let $\tau$ be the first $s\in\{1,\ldots,T\}$ with $Z_s\ge c$,
and let $\tau=T+1$ if there is no such index. Since
$\{\tau\ge s\}\in\mathcal F_{s-1}$,
\begin{align*}
Z_{s\wedge\tau}-Z_{(s-1)\wedge\tau}
&=\mathbf 1_{\{\tau\ge s\}}(Z_s-Z_{s-1}),\\
\mathbb E[Z_{s\wedge\tau}-Z_{(s-1)\wedge\tau}\mid\mathcal F_{s-1}]
&\le0,\\
c\mathbb P\{\tau\le T\}&\le\mathbb E Z_{T\wedge\tau}\le1.
\end{align*}
Thus $\mathbb P\{\max_{s\le T}Z_s\ge c\}\le c^{-1}$.

For a threshold $r>0$, take $\theta=r/(w+ar/3)$. Then $\theta a<3$ and
\[
1-\theta a/3=\frac{w}{w+ar/3},\qquad
\theta r-\frac{\theta^2w}{2(1-\theta a/3)}
=\frac{r^2}{2(w+ar/3)}.
\]
Since $W_s\le w$, we obtain
\begin{align*}
\mathbb P\{\max_{s\le T}M_s\ge r\}
&\le\mathbb P\left\{\max_{s\le T}Z_s
\ge\exp\left(\frac{r^2}{2(w+ar/3)}\right)\right\}\\
&\le\exp\left(-\frac{r^2}{2(w+ar/3)}\right).
\end{align*}
For $r=\sqrt{2wu}+2au$, direct expansion gives
\begin{align*}
r^2&=2wu+4au\sqrt{2wu}+4a^2u^2,\\
2u(w+ar/3)&=2wu+\frac23au\sqrt{2wu}+\frac43a^2u^2,\\
r^2-2u(w+ar/3)&=\frac{10}3au\sqrt{2wu}+\frac83a^2u^2\ge0.
\end{align*}
The last probability is therefore at most $e^{-u}$. This proves the
stated maximal form of Freedman's inequality \citep{freedman1975};
the maximum over $s$ requires no additional union bound.

<!-- lemma-9: body_md -->
\label{lem:composite-anchor}
Let $m=\lceil32\log(2/\delta)\rceil$, and let
$c\in\arg\min_{u\in\mathbb R^d}\sum_{j=1}^m\lVert g^{(0)}_j-u\rVert$ be a measurable
choice of geometric median. Then
\[
\mathbb P\{\lVert c-\nabla f(x_1)\rVert\le24\sigma\}\ge1-\delta/2.
\]
<!-- end lemma-9 -->
<!-- proof-9: body_md -->
Write $u_0=\nabla f(x_1)$. If $\sigma=0$, the conditional moment
bound implies $g^{(0)}_j=u_0$ almost surely for every $j$. The geometric
median is then $u_0$. Suppose that $\sigma>0$, and define
\[
J_j=\mathbf 1_{\{\lVert g^{(0)}_j-u_0\rVert>8\sigma\}}.
\]
Conditional Markov's inequality gives
\[
\mathbb E[J_j\mid\mathcal H_{j-1}]\le
\frac{\mathbb E[\lVert g^{(0)}_j-u_0\rVert^p\mid\mathcal H_{j-1}]}{(8\sigma)^p}
\le8^{-p}\le\frac18.
\]
Since $J_j\in\{0,1\}$, we have $2^{J_j}=1+J_j$. Thus
\begin{align*}
\mathbb E[2^{J_j}\mid\mathcal H_{j-1}]
&=1+\mathbb E[J_j\mid\mathcal H_{j-1}]\le\frac98,\\
\mathbb E\left[2^{\sum_{j=1}^mJ_j}\right]
&=\mathbb E\left[2^{\sum_{j=1}^{m-1}J_j}
\mathbb E[2^{J_m}\mid\mathcal H_{m-1}]\right]\\
&\le\frac98\mathbb E\left[2^{\sum_{j=1}^{m-1}J_j}\right]
\le\left(\frac98\right)^m.
\end{align*}
Markov's inequality now yields
\begin{align*}
\mathbb P\left\{\sum_{j=1}^mJ_j>\frac m4\right\}
&\le2^{-m/4}\left(\frac98\right)^m\\
&=\exp\left[-m\left(\frac{\log2}{4}-\log\frac98\right)\right]
\le e^{-m/32}\le\frac\delta2,
\end{align*}
where $(\log2)/4-\log(9/8)>1/32$. The pilot samples may
depend on the preceding pilot history.

On the complementary event, let
$\mathcal G=\{j:\lVert g^{(0)}_j-u_0\rVert\le8\sigma\}$. Then
$|\mathcal G|\ge3m/4$. The triangle inequality gives
\[
\lVert g^{(0)}_j-c\rVert-\lVert g^{(0)}_j-u_0\rVert
\ge
\begin{cases}
\lVert c-u_0\rVert-16\sigma,&j\in\mathcal G,\\
-\lVert c-u_0\rVert,&j\notin\mathcal G.
\end{cases}
\]
By the minimizing property of $c$,
\begin{align*}
0
&\ge\sum_{j=1}^m\bigl(\lVert g^{(0)}_j-c\rVert-\lVert g^{(0)}_j-u_0\rVert\bigr)\\
&\ge(2|\mathcal G|-m)\lVert c-u_0\rVert-16\sigma|\mathcal G|,\\
\lVert c-u_0\rVert
&\le\frac{16\sigma|\mathcal G|}{2|\mathcal G|-m}
\le24\sigma.
\end{align*}
The denominator is positive, and the last inequality is equivalent to
$|\mathcal G|\ge3m/4$. This proves the claim.
