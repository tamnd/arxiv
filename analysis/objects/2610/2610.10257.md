Assume that \ref{H1}--\ref{H4} of \Cref{subsec: assumptions} hold. In particular, \(M_1=M_2=\mathbb{T}^d\), \(c\) is non-degenerate near \(\Sigma\), and \(\Sigma\) is a \(d\)-dimensional \(C^{1,1}\) submanifold. Then the following holds.
\begin{enumerate}[(i)]
\item (\(\Gamma\)-expansion) The functionals \(\mathcal F_\varepsilon\) admit the first-order \(\Gamma\)-expansion
\begin{equation}
\label{eq: gamma-expansion}
\boxed{
 \mathcal F_\varepsilon - \frac{d}{2}\varepsilon\log(1/\varepsilon)
 \stackrel{\Gamma}{=}
 \mathcal F^{(0)}+\varepsilon\mathcal F^{(1)}+o(\varepsilon),
}
\end{equation}
which means that \((\mathcal F_\varepsilon-\min\mathcal F^{(0)})/\varepsilon-\frac{d}{2}\log(1/\varepsilon)\) \(\Gamma\)-converges to \(\mathcal F^{(1)}\) for the weak convergence of probability measures.
\end{enumerate}
As a consequence:
\begin{enumerate}[(i),resume]
\item (Asymptotics of the cost) The minimum of \(\mathcal F^{(1)}\) is finite, and
\begin{equation}
    \label{eq: energy-expansion}
    \boxed{
    \min\mathcal{F}_\varepsilon=\min \mathcal F^{(0)}+\dfrac{d}{2}\varepsilon\log(1/\varepsilon)+\varepsilon \min\mathcal F^{(1)}+o(\varepsilon).
    }
\end{equation}
\item (Entropic selection) The unique minimizer \(\gamma_\varepsilon\) of \(\mathcal F_\varepsilon\) converges weakly, as \(\varepsilon\to0\), to the unique minimizer \(\gamma_0\) of \(\operatorname{Ent}(\cdot\,|\,\mathbb P_0)\) among all optimal transport plans, which is also the unique minimizer of \(\mathcal F^{(1)}\).
\end{enumerate}

\label{thm: convergence-Gibbs-measure}
Under \ref{H1}--\ref{H4}, as \(\varepsilon\to0\), the Gibbs measure \(\operatorname{\mathbb{P}_\varepsilon}\) converges weakly to a probability measure \(\mathbb{P}_0\), given by
\[
\mathop{}\mathopen{}\mathrm{d}\mathbb{P}_0(z)
=
\frac{1}{Z_0}
\left(\det H_{z}E\right)^{-1/2}
\rho(z)
\mathop{}\mathopen{}\mathrm{d}\mathcal H^d_{|\Sigma}(z),
\qquad
Z_0
\stackrel{\text{def.}}{=}
\int_\Sigma
\left(\det H_{z}E\right)^{-1/2}
\rho(z)
\mathop{}\mathopen{}\mathrm{d}\mathcal H^d_{|\Sigma}(z).
\]
Moreover, \(Z_\varepsilon=(2\pi\varepsilon)^{k/2}(Z_0+o(1))\).

By \eqref{eq: normal-recovery-density}, and since \(\operatorname{\mathbb{P}_\varepsilon}\) has density \(Z_\varepsilon^{-1}e^{-E/\varepsilon}\rho\) with respect to \(\operatorname{vol}_\Omega\), where \(\rho\) is the density of \(\mu_1\otimes\mu_2\), we have, for \(\mathcal H^d\)-a.e.\ \(z\in\Sigma\) and every \(s\in\mathbb{B}_R\),
\begin{align*}
 \log\left(
 \frac{\mathop{}\mathopen{}\mathrm{d}\widetilde\gamma_{\varepsilon,R}}
      {\mathop{}\mathopen{}\mathrm{d}\operatorname{\mathbb{P}_\varepsilon}}(\Phi(z,\sqrt\varepsilon s))\right)
 &=\log\overline\rho(z)
   +\log\left(\frac{Z_\varepsilon}
   {\varepsilon^{k/2}Z_R(z)}\right)
   -\log\rho(\Phi(z,\sqrt\varepsilon s))
\\
 &\qquad+\frac{E(\Phi(z,\sqrt\varepsilon s))-\frac12H_{z}E(\sqrt\varepsilon s,\sqrt\varepsilon s)}
 {\varepsilon}
 -\log\mathcal J_\Phi(z,\sqrt\varepsilon s).
\end{align*}
As in the proof of \Cref{lemma: quadratic-detachment}, a Taylor expansion of \(E\) gives
\[
E(\Phi(z,\eta))=\frac12H_{z}E(\eta,\eta)+\mathcal R(z,\eta),
\qquad
|\mathcal R(z,\eta)|\leq C|\eta|^3.
\]
Since \(\operatorname{Ent}(\widetilde\gamma_{\varepsilon,R}\,|\,\operatorname{\mathbb{P}_\varepsilon})=\int_\Omega\log\frac{\mathop{}\mathopen{}\mathrm{d}\widetilde\gamma_{\varepsilon,R}}{\mathop{}\mathopen{}\mathrm{d}\operatorname{\mathbb{P}_\varepsilon}}\mathop{}\mathopen{}\mathrm{d}\widetilde\gamma_{\varepsilon,R}\), we may compute this integral in tubular coordinates by \eqref{eq: recovery sequence with wrong marginals 1}, that is, integrate the above identity against \(K_R(z,s)\mathop{}\mathopen{}\mathrm{d} s\,\mathop{}\mathopen{}\mathrm{d}\overline\gamma(z)\). We obtain
\begin{align*}
\operatorname{Ent}\left(\widetilde\gamma_{\varepsilon,R}\,|\,\operatorname{\mathbb{P}_\varepsilon} \right)
 &=\int_\Sigma\log\overline\rho(z)\overline\gamma(\mathop{}\mathopen{}\mathrm{d} z)
 +\int_\Sigma\log\left(\frac{Z_\varepsilon}
 {\varepsilon^{k/2}Z_R(z)}\right)
 \overline\gamma(\mathop{}\mathopen{}\mathrm{d} z)
 \\
 &\qquad-\int_\Sigma\int_{\mathbb{B}_R}
 \Bigl(\log\rho(\Phi(z,\sqrt\varepsilon s))
       -\frac{\mathcal R(z,\sqrt\varepsilon s)}{\varepsilon}
 \\
 &\qquad\qquad\qquad\qquad
       +\log\mathcal J_\Phi(z,\sqrt\varepsilon s)\Bigr)
 K_R(z,s)\mathop{}\mathopen{}\mathrm{d} s\,\overline\gamma(\mathop{}\mathopen{}\mathrm{d} z).
\end{align*}
These integrals are finite: since \(\operatorname{Ent}(\overline\gamma\,|\,\mathbb P_0)<\infty\) and the density of \(\mathbb P_0\) with respect to \(\mathcal H^d_{|\Sigma}\) is bounded from above and below by positive constants, we have
\(\int_\Sigma|\log\overline\rho|\,\mathop{}\mathopen{}\mathrm{d}\overline\gamma<\infty\).
Since \(|\mathcal R(z,\sqrt\varepsilon s)|\leq C\varepsilon^{3/2}|s|^3\), since \(\log\rho\) is Lipschitz (\(\rho\) being Lipschitz and bounded below by a positive constant), and since \(|\log\mathcal J_\Phi(z,\sqrt\varepsilon s)|\leq C\sqrt\varepsilon|s|\) by \eqref{eq: tube-jacobian-rescaled}, the double integral in the last line differs from \(\int_\Sigma\log\rho\mathop{}\mathopen{}\mathrm{d}\overline\gamma\) by at most
\[
C\sqrt\varepsilon\int_\Sigma\int_{\mathbb{B}_R}\bigl(|s|+|s|^3\bigr)K_R(z,s)\mathop{}\mathopen{}\mathrm{d} s\,\overline\gamma(\mathop{}\mathopen{}\mathrm{d} z)\leq C\sqrt\varepsilon\longrightarrow0,
\]
by \eqref{eq: gaussian-moments}.
Moreover, by \Cref{thm: convergence-Gibbs-measure},
\(Z_\varepsilon=(2\pi\varepsilon)^{k/2}(Z_0+o(1))\), so that, uniformly in \(z\),
\[
 \log\left(\frac{Z_\varepsilon}
 {\varepsilon^{k/2}Z_R(z)}\right)
 =\log\frac{(2\pi)^{k/2}Z_0}{Z_R(z)}+o(1)
 =\log\left(Z_0\sqrt{\det H_{z}E}\right)
  +\log\frac{Z_\infty(z)}{Z_R(z)}+o(1).
\]
Combining these identities yields
\begin{align*}
 \operatorname{Ent}\left(\widetilde\gamma_{\varepsilon,R}\,|\,\operatorname{\mathbb{P}_\varepsilon} \right)
 &\longrightarrow
 \int_\Sigma\log\frac{\overline\rho(z)}{\rho(z)}\overline\gamma(\mathop{}\mathopen{}\mathrm{d} z)
 +\int_\Sigma\log\left(Z_0\sqrt{\det H_{z}E}\right)
 \overline\gamma(\mathop{}\mathopen{}\mathrm{d} z)
 \\
 &\qquad+\int_\Sigma\log\frac{Z_\infty(z)}{Z_R(z)}
 \overline\gamma(\mathop{}\mathopen{}\mathrm{d} z)
 \\
 &=\operatorname{Ent}\left(\overline\gamma\,|\,\mathbb P_0\right)+b_R(\overline\gamma).
\end{align*}
Finally, the Gaussian tail estimate
\[
 0\leq Z_\infty(z)-Z_R(z)
 \leq\int_{|s|\geq R}\exp\left(-\frac{\lambda}{2}|s|^2\right)\,\mathop{}\mathopen{}\mathrm{d} s
 \longrightarrow0\qquad(R\to\infty)
\]
is uniform in \(z\). Together with \(Z_R(z)\geq c>0\), this shows
that \(b_R\geq0\) tends to zero uniformly over probability measures
on \(\Sigma\), and proves the claim.

\label{def: shadow} 
Let \(\mu_i, \widetilde{\mu}_i \in \mathcal{P}(M_i)\) with \(\widetilde\mu_i\ll\operatorname{vol}_{M_i}\), and let \(T_i\) denote the optimal transport map from \(\widetilde \mu_i\) to \(\mu_i\) for the quadratic cost, \(i=1,2\). Then, for a given \(\widetilde{\gamma} \in \Pi(\widetilde{\mu}_1,\widetilde\mu_2)\), its \emph{shadow} \(\gamma \in \Pi(\mu_1,\mu_2)\) is defined by
\[
\gamma \stackrel{\text{def.}}{=} \big(T_1,T_2\big)_\# \widetilde{\gamma}.
\]

In this proof, we write \(\gamma=\gamma_{\varepsilon,R}\), \(\widetilde\gamma=\widetilde\gamma_{\varepsilon,R}\) and \(\widetilde\mu_i=\widetilde\mu_{i,\varepsilon,R}\). Recall that \(\mathbb P_\varepsilon=f_\varepsilon\,\mu^\otimes\), where \(f_\varepsilon\stackrel{\text{def.}}{=} Z_\varepsilon^{-1}e^{-E/\varepsilon}\) (\Cref{prop: EOT-Gibbs}).
By \Cref{lemma: convergence entropy}, \(\operatorname{Ent}(\widetilde\gamma\,|\,\operatorname{\mathbb{P}_\varepsilon})<\infty\), hence \(\operatorname{Ent}(\widetilde\gamma\,|\,\mu^\otimes)\le\operatorname{Ent}(\widetilde\gamma\,|\,\operatorname{\mathbb{P}_\varepsilon})-\log Z_\varepsilon<\infty\) since \(E\geq0\).
Hence the marginal entropies \(\operatorname{Ent}(\widetilde\mu_i\,|\,\mu_i)\) are finite by projection, and so are all the
quantities below. By a change of reference measure, \(\operatorname{Ent}(\cdot\,|\,\operatorname{\mathbb{P}_\varepsilon})=\operatorname{Ent}(\cdot\,|\,\mu^\otimes)-\int_\Omega\log f_\varepsilon\,\mathop{}\mathopen{}\mathrm{d}(\cdot)\) on probability measures. Together with \Cref{prop: bound entropy shadow}, this gives
\begin{align*}
\operatorname{Ent}\left(\gamma\,|\,\operatorname{\mathbb{P}_\varepsilon}\right)
 &=\operatorname{Ent}\left(\gamma\,|\,\mu^\otimes\right)
 -\int_\Omega\log f_\varepsilon\,\mathop{}\mathopen{}\mathrm{d}\gamma
 \\
 &\leq\operatorname{Ent}\left(\widetilde\gamma\,|\,\widetilde\mu_1\otimes\widetilde\mu_2\right)
 -\int_\Omega\log f_\varepsilon\,\mathop{}\mathopen{}\mathrm{d}\gamma\\
 &=\operatorname{Ent}\left(\widetilde\gamma\,|\,\operatorname{\mathbb{P}_\varepsilon}\right)
 +\int_\Omega\log f_\varepsilon\,
 \mathop{}\mathopen{}\mathrm{d}(\widetilde\gamma-\gamma)
 \\
 &\qquad+\operatorname{Ent}\left(\widetilde\gamma\,|\,\widetilde\mu_1\otimes\widetilde\mu_2\right) -\operatorname{Ent}\left(\widetilde\gamma\,|\,\mu^\otimes\right).
\end{align*}
Since \(\widetilde\mu_i\ll\mu_i\) (\Cref{subsec: normal-recovery}), there exist
Borel functions \(a_i:M_i\to[0,\infty)\) such that
\(\widetilde\mu_i=a_i\,\mu_i\), so that
\(\widetilde\mu_1\otimes\widetilde\mu_2=a_1(x_1)a_2(x_2)\,\mu^\otimes\).
Consequently,
\begin{align*}
 \operatorname{Ent}\left(\widetilde\gamma\,|\,\widetilde\mu_1\otimes\widetilde\mu_2\right)-\operatorname{Ent}\left(\widetilde\gamma\,|\,\mu^\otimes\right)
 &=-\int_\Omega\log\bigl(a_1(x_1)a_2(x_2)\bigr)\mathop{}\mathopen{}\mathrm{d}\widetilde\gamma(x_1,x_2)
 \\
 &=-\sum_{i=1}^2\int_{M_i}\log a_i\,\mathop{}\mathopen{}\mathrm{d}\widetilde\mu_i
 =-\sum_{i=1}^2\operatorname{Ent}\left(\widetilde\mu_i\,|\,\mu_i\right)\leq0,
\end{align*}
where the last inequality follows from the nonnegativity of the relative
entropy between probability measures. Finally, since \(\log f_\varepsilon=-E/\varepsilon-\log Z_\varepsilon\) and both plans have total mass one,
\[
 \operatorname{Ent}\left(\gamma\,|\,\operatorname{\mathbb{P}_\varepsilon} \right)
\leq\operatorname{Ent}\left(\widetilde\gamma\,|\,\operatorname{\mathbb{P}_\varepsilon}\right)
 -\int_\Omega\log f_\varepsilon\,
 \mathop{}\mathopen{}\mathrm{d}(\gamma-\widetilde\gamma)
=\operatorname{Ent}\left(\widetilde\gamma\,|\,\operatorname{\mathbb{P}_\varepsilon}\right)
 +\frac1\varepsilon\int_\Omega E\,
 \mathop{}\mathopen{}\mathrm{d}(\gamma-\widetilde\gamma).
\]

\label{thm: Wasserstein between the marginals}
For each fixed \(R\geq1\), the marginals of the normal perturbation
\eqref{eq: recovery sequence with wrong marginals 1} satisfy
\begin{equation}
\label{eq: marginal-cancellation}
W_2(\widetilde\mu_{i,\varepsilon,R},\mu_i)=o_R(\sqrt\varepsilon)
 \qquad (i=1,2),\quad\text{as }\varepsilon\to0,
\end{equation}
where the subscript indicates that the rate of convergence may depend on \(R\).

<!-- proof-8: body_md -->
Let \(T_{\varepsilon,R}\stackrel{\text{def.}}{=}(T_1,T_2)\), where \(T_i\) is the optimal transport map from \(\widetilde\mu_{i,\varepsilon,R}\) to \(\mu_i\). By definition of the shadow,
\(\gamma_{\varepsilon,R}
 =(T_{\varepsilon,R})_\#\widetilde\gamma_{\varepsilon,R}\).
Set
\[
 \Delta_{\varepsilon,R}^2
 \stackrel{\text{def.}}{=}\int_\Omega \operatorname{dist}(\operatorname{\textbf{x}},T_{\varepsilon,R}(\operatorname{\textbf{x}}))^2
 \,\mathop{}\mathopen{}\mathrm{d}\widetilde\gamma_{\varepsilon,R}(\operatorname{\textbf{x}})
 =\sum_{i=1}^2W_2^2(\widetilde\mu_{i,\varepsilon,R},\mu_i).
\]
The pushforward identity gives
\[
 \frac1\varepsilon\int_\Omega E\,
 \mathop{}\mathopen{}\mathrm{d}(\gamma_{\varepsilon,R}-\widetilde\gamma_{\varepsilon,R})
 =\frac1\varepsilon\int_\Omega
 \bigl(E(T_{\varepsilon,R}(\operatorname{\textbf{x}}))-E(\operatorname{\textbf{x}})\bigr)
 \,\mathop{}\mathopen{}\mathrm{d}\widetilde\gamma_{\varepsilon,R}(\operatorname{\textbf{x}}).
\]

We identify \(\Omega\) with \(\mathbb T^{2d}=\mathbb{R}^{2d}/\mathbb Z^{2d}\), denote by \([y]\in\Omega\) the class of \(y\in\mathbb{R}^{2d}\), and by \(\widetilde E\stackrel{\text{def.}}{=} E([\,\cdot\,])\) the \(\mathbb Z^{2d}\)-periodic lift of \(E\), which belongs to \(C^{2,1}(\mathbb{R}^{2d})\) by \ref{H2}. We set
\[
 C_E\stackrel{\text{def.}}{=}\sup_{y\in\mathbb{R}^{2d}}\|D^2\widetilde E(y)\|<\infty,
\]
where \(\|\cdot\|\) denotes the operator norm.

We first estimate the gradient by the distance to \(\Sigma\). Let \(y\in\mathbb{R}^{2d}\). Since \(\Sigma\) is compact, there exists \(\bar y\in\mathbb{R}^{2d}\) such that \([\bar y]\in\Sigma\) and \(|y-\bar y|=\operatorname{dist}([y],\Sigma)\). Since \(\nabla\widetilde E(\bar y)=0\), the fundamental theorem of calculus gives
\begin{equation}
\label{eq: gradient-distance}
 |\nabla\widetilde E(y)|
 =\left|\int_0^1 D^2\widetilde E\bigl(\bar y+t(y-\bar y)\bigr)(y-\bar y)\mathop{}\mathopen{}\mathrm{d} t\right|
 \leq C_E\operatorname{dist}([y],\Sigma).
\end{equation}
Now, for each \(\operatorname{\textbf{x}}\in\Omega\), choose \(\widetilde\operatorname{\textbf{x}}\in\mathbb{R}^{2d}\) with \([\widetilde\operatorname{\textbf{x}}]=\operatorname{\textbf{x}}\), and a shortest displacement \(v_{\operatorname{\textbf{x}}}\in\mathbb R^{2d}\) such that
\[
 [\widetilde\operatorname{\textbf{x}}+v_{\operatorname{\textbf{x}}}]=T_{\varepsilon,R}(\operatorname{\textbf{x}}),
 \qquad |v_{\operatorname{\textbf{x}}}|=\operatorname{dist}(\operatorname{\textbf{x}},T_{\varepsilon,R}(\operatorname{\textbf{x}})).
\]
By the fundamental theorem of calculus,
\[
 E(T_{\varepsilon,R}(\operatorname{\textbf{x}}))-E(\operatorname{\textbf{x}})
 =\widetilde E(\widetilde\operatorname{\textbf{x}}+v_{\operatorname{\textbf{x}}})-\widetilde E(\widetilde\operatorname{\textbf{x}})
 =\int_0^1\bigl\langle\nabla\widetilde E(\widetilde\operatorname{\textbf{x}}+\tau v_{\operatorname{\textbf{x}}}),
                         v_{\operatorname{\textbf{x}}}\bigr\rangle\,\mathop{}\mathopen{}\mathrm{d}\tau.
\]
For every \(\tau\in[0,1]\), the triangle inequality gives \(\operatorname{dist}([\widetilde\operatorname{\textbf{x}}+\tau v_{\operatorname{\textbf{x}}}],\Sigma)\leq\operatorname{dist}(\operatorname{\textbf{x}},\Sigma)+\tau|v_{\operatorname{\textbf{x}}}|\). Together with \eqref{eq: gradient-distance}, we obtain
\begin{equation}
\label{eq: energy-difference-gradient}
 |E(T_{\varepsilon,R}(\operatorname{\textbf{x}}))-E(\operatorname{\textbf{x}})|
 \leq C_E \operatorname{dist}(\operatorname{\textbf{x}},T_{\varepsilon,R}(\operatorname{\textbf{x}}))
              \operatorname{dist}(\operatorname{\textbf{x}},\Sigma)
 +C_E \operatorname{dist}(\operatorname{\textbf{x}},T_{\varepsilon,R}(\operatorname{\textbf{x}}))^2.
\end{equation}
Integrating \eqref{eq: energy-difference-gradient} with respect to \(\widetilde\gamma_{\varepsilon,R}\) and applying the Cauchy--Schwarz inequality gives
\begin{align*}
 \frac1\varepsilon\left|\int_\Omega E\,
 \mathop{}\mathopen{}\mathrm{d}(\gamma_{\varepsilon,R}-\widetilde\gamma_{\varepsilon,R})\right|
 &\leq\frac{C_E\Delta_{\varepsilon,R}}{\varepsilon}
 \left(\int_\Omega \operatorname{dist}(\operatorname{\textbf{x}},\Sigma)^2
 \,\mathop{}\mathopen{}\mathrm{d}\widetilde\gamma_{\varepsilon,R}(\operatorname{\textbf{x}})\right)^{1/2}
 \\
 &\qquad+\frac{C_E\Delta_{\varepsilon,R}^2}{\varepsilon}.

\textbf{Step 1. The \(\Gamma\)-\(\liminf\) inequality \eqref{eq: gamma-liminf}.} Let \(\gamma_\varepsilon\rightharpoonup\overline\gamma\). \textbf{Step 2. \[
 \limsup_{\varepsilon\to0}\mathcal G_\varepsilon(\gamma_{\varepsilon,R})
 =\limsup_{\varepsilon\to0}\operatorname{Ent}\left(\gamma_{\varepsilon,R}\,|\,\operatorname{\mathbb{P}_\varepsilon}\right)
 \leq\mathcal G_0(\overline\gamma)+b_R(\overline\gamma),

while \Cref{prop: bound entropy shadow} and \eqref{eq: normal-perturbation-W2} give

\textbf{Step 1. Computing the flux.}
For \(\phi\in C_c^\infty(\mathbb R)\) and \(t\in[0,1]\),
\begin{align*}
\frac{\mathop{}\mathopen{}\mathrm{d}}{\mathop{}\mathopen{}\mathrm{d} t}\int_\mathbb{R}\phi\mathop{}\mathopen{}\mathrm{d}\mu_{t,\varepsilon}
&=\frac{\mathop{}\mathopen{}\mathrm{d}}{\mathop{}\mathopen{}\mathrm{d} t}\frac{1}{c_R}\int_{-R}^R\int_\mathbb{R}
\phi(y+t\sqrt\varepsilon z)\mu(y)e^{-z^2/2}\mathop{}\mathopen{}\mathrm{d} y\mathop{}\mathopen{}\mathrm{d} z\\
&=\frac{1}{c_R}\int_{-R}^R\int_\mathbb{R}
\phi'(y+t\sqrt\varepsilon z)\sqrt\varepsilon z\,
\mu(y)e^{-z^2/2}\mathop{}\mathopen{}\mathrm{d} y \mathop{}\mathopen{}\mathrm{d} z
=\int_\mathbb{R}\phi'(x)\operatorname{\mathfrak m_{t,\varepsilon}}(x)\mathop{}\mathopen{}\mathrm{d} x,
\end{align*}
where, by the change of variables \(y=x-t\sqrt\varepsilon z\),
\[
\operatorname{\mathfrak m_{t,\varepsilon}}(x)\stackrel{\text{def.}}{=}\frac{1}{c_R}\int_{-R}^R
\sqrt\varepsilon z\,\mu(x-t\sqrt\varepsilon z)
e^{-z^2/2}\mathop{}\mathopen{}\mathrm{d} z.
\]
Differentiation under the integral sign is justified since \(\phi'\) is bounded, \(\mu\in L^1\) and \(\int_{-R}^R |z|e^{-z^2/2}\mathop{}\mathopen{}\mathrm{d} z<\infty\).
Thus \(\partial_t\mu_{t,\varepsilon}+\partial_x\operatorname{\mathfrak m_{t,\varepsilon}}=0\) in the sense of
distributions. Note that \(\mathfrak m_{0,\varepsilon}=0\) by
symmetry of the Gaussian kernel.

Let us estimate the total variation \(\|\operatorname{\mathfrak m_{t,\varepsilon}}\|_{\mathrm{TV}}=\int_\mathbb{R}|\operatorname{\mathfrak m_{t,\varepsilon}}|\mathop{}\mathopen{}\mathrm{d} x\). The flux depends linearly on \(\mu\): for \(f\in L^1(\mathbb{R})\), let
\[
\operatorname{\mathfrak m_{t,\varepsilon}}[f](x)\stackrel{\text{def.}}{=}\frac{1}{c_R}\int_{-R}^R
\sqrt\varepsilon z\,f(x-t\sqrt\varepsilon z)
e^{-z^2/2}\mathop{}\mathopen{}\mathrm{d} z,
\qquad\text{so that }\operatorname{\mathfrak m_{t,\varepsilon}}=\operatorname{\mathfrak m_{t,\varepsilon}}[\mu].
\]
We will use the following bounds on the moments of the truncated Gaussian, which are uniform in \(R\) and follow from an integration by parts and the Cauchy--Schwarz inequality:
\begin{align}
\label{eq: 1D-second-moment}
\frac{1}{c_R}\int_{-R}^R z^2e^{-z^2/2}\mathop{}\mathopen{}\mathrm{d} z
&=1-\frac{2Re^{-R^2/2}}{c_R}\le1,
\\
\label{eq: 1D-first-moment}
\frac{1}{c_R}\int_{-R}^R |z|e^{-z^2/2}\mathop{}\mathopen{}\mathrm{d} z
&\le\left(\frac{1}{c_R}\int_{-R}^R z^2e^{-z^2/2}\mathop{}\mathopen{}\mathrm{d} z\right)^{1/2}
\le1.
\end{align}
By Fubini's theorem, the translation invariance of the Lebesgue measure and \eqref{eq: 1D-first-moment}, we get
\[
\|\operatorname{\mathfrak m_{t,\varepsilon}}[f]\|_{L^1}\le\sqrt\varepsilon\|f\|_{L^1}
\qquad\text{for every }f\in L^1(\mathbb{R}).
\]
If moreover \(f\in C_c^1(\mathbb{R})\), we write, with \(x_s\stackrel{\text{def.}}{=} x-s\sqrt\varepsilon z\),
\[
f(x-t\sqrt\varepsilon z)
=f(x)-\int_0^t\sqrt\varepsilon z f'(x_s)\mathop{}\mathopen{}\mathrm{d} s.
\]
Since the kernel is centered, that is,
\(\int_{-R}^R z e^{-z^2/2}\mathop{}\mathopen{}\mathrm{d} z=0\),
the term containing \(f(x)\) cancels, so that, by \eqref{eq: 1D-second-moment},
\[
\|\operatorname{\mathfrak m_{t,\varepsilon}}[f]\|_{L^1}
=\int_\mathbb{R}\left|\frac{\varepsilon}{c_R}\int_{-R}^R
z^2e^{-z^2/2}\int_0^tf'(x_s)\mathop{}\mathopen{}\mathrm{d} s\mathop{}\mathopen{}\mathrm{d} z\right|\mathop{}\mathopen{}\mathrm{d} x
\le t\varepsilon\|f'\|_{L^1}
\le\varepsilon\|f'\|_{L^1}.
\]
Now let \((\mu^k)_{k\in\mathbb{N}}\) be a sequence in \(C_c^1(\mathbb{R})\)
such that \(\mu^k\to\mu\) in \(L^1\). Writing \(\operatorname{\mathfrak m_{t,\varepsilon}}=\operatorname{\mathfrak m_{t,\varepsilon}}[\mu-\mu^k]+\operatorname{\mathfrak m_{t,\varepsilon}}[\mu^k]\), we obtain, for every \(k\in\mathbb{N}\) and \(t\in[0,1]\),
\[
\frac{\|\operatorname{\mathfrak m_{t,\varepsilon}}\|_{\mathrm{TV}}}{\sqrt\varepsilon}
\le\|\mu-\mu^k\|_{L^1}
+\sqrt\varepsilon\|(\mu^k)'\|_{L^1}.
\]
Define, for \(r>0\),
\[
\omega(r)\stackrel{\text{def.}}{=}\inf_{k\in\mathbb{N}}\left\{
\|\mu-\mu^k\|_{L^1}+r\|(\mu^k)'\|_{L^1}\right\}.
\]
For each fixed \(k\),
\(0\le\limsup_{r\downarrow0}\omega(r)\le\|\mu-\mu^k\|_{L^1}\), and
letting \(k\to\infty\) shows that \(\omega(r)\to0\) as \(r\to0\). Therefore
\begin{equation}
\label{eq: 1D-flux-bound}
\sup_{t\in[0,1]}\frac{\|\operatorname{\mathfrak m_{t,\varepsilon}}\|_{\mathrm{TV}}}{\sqrt\varepsilon}
\le\omega(\sqrt\varepsilon)\longrightarrow0
\qquad(\varepsilon\to0).
\end{equation}

\textbf{Step 2. Estimate of \(W_2(\mu,\widetilde\mu_\varepsilon)\).}
By Step~1, the curve \((\mu_{t,\varepsilon})_{t\in[0,1]}\) of probability measures with finite second moments solves, together with \((\operatorname{\mathfrak m_{t,\varepsilon}})_{t\in[0,1]}\), the continuity equation, and it is weakly continuous, with \(\mu_{0,\varepsilon}=\mu\) and \(\mu_{1,\varepsilon}=\widetilde\mu_\varepsilon\). Since \(|z|\le R\) in the definition of \(\operatorname{\mathfrak m_{t,\varepsilon}}\), we have
\[
|\operatorname{\mathfrak m_{t,\varepsilon}}(x)|
\le\frac{\sqrt\varepsilon R}{c_R}
\int_{-R}^R\mu(x-t\sqrt\varepsilon z)
e^{-z^2/2}\mathop{}\mathopen{}\mathrm{d} z
=\sqrt\varepsilon R\,\mu_{t,\varepsilon}(x)
\qquad\text{for a.e. }x\in\mathbb{R}.
\]
Thus \(\operatorname{\mathfrak m_{t,\varepsilon}}\ll\mu_{t,\varepsilon}\), and the velocity field
\[
v_{t,\varepsilon}
\stackrel{\text{def.}}{=}\frac{\mathop{}\mathopen{}\mathrm{d} \operatorname{\mathfrak m_{t,\varepsilon}}}{\mathop{}\mathopen{}\mathrm{d}\mu_{t,\varepsilon}}
\quad\text{satisfies}\quad |v_{t,\varepsilon}|\le\sqrt\varepsilon R
\quad\mu_{t,\varepsilon}\text{-a.e.}
\]
By the Benamou--Brenier formula (see, e.g., \cite{santambrogioOptimalTransportApplied2015}), with \((\mu_{t,\varepsilon},v_{t,\varepsilon})_{t\in[0,1]}\) as a competitor, and by \eqref{eq: 1D-flux-bound}, we get

&\le\int_0^1\int_\mathbb{R} |v_{t,\varepsilon}|^2

\\
&\le\sqrt\varepsilon R\sup_{t\in[0,1]}
\|\operatorname{\mathfrak m_{t,\varepsilon}}\|_{\mathrm{TV}}
\le \varepsilon R\,\omega(\sqrt\varepsilon).
\end{align*}
Hence \(W_2(\mu,\widetilde\mu_\varepsilon)\le\sqrt R\,\sqrt{\omega(\sqrt\varepsilon)}\,\sqrt\varepsilon\), which concludes the proof, since \(\omega\) depends only on \(\mu\).

&=\sum_{j\in J}\int_{V_j} \int_{\mathbb{B}_R} \big\langle \nabla \phi(\zeta_{\varepsilon,s,t}^j(x)), \partial_t\zeta_{\varepsilon,s,t}^j (x)\big\rangle \rho_j(x)K_j(x,s) \mathop{}\mathopen{}\mathrm{d} s \mathop{}\mathopen{}\mathrm{d} \operatorname{vol}_M(x)
\\
&=\sum_{j\in J}\int_{V_j} \int_{\mathbb{B}_R} \sqrt{\varepsilon}\big\langle \nabla \phi(\zeta_{\varepsilon,s,t}^j(x)), A_j(x)s  \big\rangle \rho_j(x)K_j(x,s) \mathop{}\mathopen{}\mathrm{d} s \mathop{}\mathopen{}\mathrm{d} \operatorname{vol}_M(x)
\\
&=
\int_M
\left\langle \nabla \phi,\mathop{}\mathopen{}\mathrm{d}\operatorname{\mathfrak m_{t,\varepsilon}}\right\rangle,
\end{align*}
where the differentiation under the integral sign is justified since \(\nabla\phi\) is bounded and \(|\partial_t\zeta^j_{\varepsilon,s,t}(x)|=\sqrt\varepsilon|A_j(x)s|\leq\sqrt\varepsilon R\), and where the flux \(\operatorname{\mathfrak m_{t,\varepsilon}}\in \mathcal M(M;TM)\) is the vector measure defined by
\[\operatorname{\mathfrak m_{t,\varepsilon}} \stackrel{\text{def.}}{=} \sqrt{\varepsilon}\sum_{j\in J} \int_{\mathbb{B}_R} (\zeta_{\varepsilon,s,t}^j)_{\#}\left(\rho_jK_j(\cdot,s)A_js\, \operatorname{vol}_M\right) \mathop{}\mathopen{}\mathrm{d} s. \]
Integrating in time against a test function in
\(C_c^1((0,1))\) yields
\[
\partial_t\mu_{t,\varepsilon}
+
\operatorname{div}\operatorname{\mathfrak m_{t,\varepsilon}}
=0
\qquad\text{in }\mathcal D'((0,1)\times M).
\]

\textbf{Step 3. Gaussian symmetry.} As in \Cref{subsubsec: 1D}, the flux depends linearly on the densities \(\rho_j\). For \(j\in J\), \(s\in\mathbb{B}_R\) and \(f\in L^1(V_j)\), let
\[
F_{j,s}[f](x) \stackrel{\text{def.}}{=}
\begin{cases}
f(x)K_j(x,s)A_j(x)s,
& x\in V_j,
\\[1mm]
0,
& x\notin V_j,
\end{cases}
\]
so that, by definition of the flux,
\[
\frac{\operatorname{\mathfrak m_{t,\varepsilon}}}{\sqrt{\varepsilon}}
= \sum_{j\in J}
\int_{\mathbb{B}_R}
(\zeta_{\varepsilon,s,t}^j)_{\#}
\big(
F_{j,s}[\rho_j]\operatorname{vol}_M
\big)
\mathop{}\mathopen{}\mathrm{d} s.
\]
By \eqref{eq: symmetry of the Gaussian}, \(\int_{\mathbb{B}_R}F_{j,s}[\rho_j]\mathop{}\mathopen{}\mathrm{d} s=0\). Subtracting this in each term, we obtain
\begin{equation}
\label{eq: flow-decomposition}
\frac{\operatorname{\mathfrak m_{t,\varepsilon}}}{\sqrt{\varepsilon}}
=\sum_{j\in J}
\int_{\mathbb{B}_R}
\left((\zeta_{\varepsilon,s,t}^j)_{\#}-\operatorname{Id}\right)\left(
F_{j,s}[\rho_j]\operatorname{vol}_M
\right)
\mathop{}\mathopen{}\mathrm{d} s
=\sum_{j\in J}\mathbf{I}_{t,\varepsilon}^j[\rho_j],
\end{equation}
where, for \(f\in L^1(V_j)\),
\[
\mathbf{I}_{t,\varepsilon}^j[f]
\stackrel{\text{def.}}{=}
\int_{\mathbb{B}_R}
\left((\zeta_{\varepsilon,s,t}^j)_{\#}-\operatorname{Id}\right)\left(
F_{j,s}[f]\operatorname{vol}_M
\right)
\mathop{}\mathopen{}\mathrm{d} s.
\]
 
\textbf{Step 4. Estimate of the displacement term.} Fix \(j\in J\). We estimate the total variation of \(\mathbf{I}_{t,\varepsilon}^j[f]\) in two ways.
 
\emph{(a) For \(f\in L^1(V_j)\).} Since the push-forward operation does not increase the total variation and \(|A(z)s|\leq|n(z)s|=|s|\), \eqref{eq: gaussian-moments} gives
\begin{align*}
\|\mathbf{I}_{t,\varepsilon}^j[f]\|_{\mathrm{TV}}
&\leq 2\int_{\mathbb{B}_R}\|F_{j,s}[f]\|_{L^1(V_j)}\mathop{}\mathopen{}\mathrm{d} s
\\
&\leq 2\int_{V_j}|f(x)|\int_{\mathbb{B}_R}|s|K_j(x,s)\mathop{}\mathopen{}\mathrm{d} s\mathop{}\mathopen{}\mathrm{d}\operatorname{vol}_M(x)
\leq 2M_1\|f\|_{L^1(V_j)}.
\end{align*}
 
\emph{(b) For \(f\in C_c^1(V_j)\) supported in a fixed compact set \(S\subseteq V_j\).} Write \(F_s\stackrel{\text{def.}}{=} F_{j,s}[f]\). Fix \(t\in[0,1]\) and \(s\in\mathbb{B}_R\), and write \(\zeta\stackrel{\text{def.}}{=}\zeta_{\varepsilon,s,t}^j\), \(\kappa\stackrel{\text{def.}}{=}\zeta^{-1}\) and \(W\stackrel{\text{def.}}{=}\zeta(V_j)\). Since \(|\zeta(x)-x|\leq\sqrt\varepsilon R\),
we may choose \(\varepsilon>0\) sufficiently small, depending only on \(S\) (and not on \(f\), \(s\) and \(t\)), so that
\[
S\Subset W
\qquad\text{and}\qquad
\zeta(S)\Subset V_j.
\]
By the area formula, the density of \(\zeta_{\#}
(F_s\operatorname{vol}_M)\) is
\(F_s\circ\kappa\, \mathcal J_{\kappa}\mathbf{1}_{W}\), and, since \(\operatorname{spt}(F_s)\subseteq S\), the above inclusions show that \(F_s\circ\kappa\,\mathbf 1_{W}\) and \(F_s\) are both supported in \(W\cap V_j\). Hence
\begin{align*}
\big\|
\zeta_{\#}
(F_s\operatorname{vol}_M)
-
F_s\operatorname{vol}_M
\big\|_{\mathrm{TV}}
&\leq\|\mathcal J_{\kappa}-1\|_{L^\infty(W)}
\int_{W}
|F_s\circ\kappa|
\mathop{}\mathopen{}\mathrm{d} \operatorname{vol}_M
\\
&\quad+\int_{W\cap V_j }
|F_s\circ\kappa-F_s|\mathop{}\mathopen{}\mathrm{d}\operatorname{vol}_M.
\end{align*}
By \Cref{lemma: C^1 diffeo}, applied with \(\tau\stackrel{\text{def.}}{=} t\sqrt\varepsilon|s|L_j\) and \(f\stackrel{\text{def.}}{=} A_js/(|s|L_j)\), where \(L_j\) is a Lipschitz constant of \(A_j\) on the convex set \(V_j\) (recall that \(n\) is Lipschitz and \(\alpha_j\) is \(C^{1,1}\)), we have, for \(\varepsilon\) small enough, \(\|\mathcal J_{\kappa}-1\|_{L^\infty(W)}+\|\mathcal J_{\zeta}-1\|_{L^\infty(V_j)}\leq C\sqrt{\varepsilon}|s|\), and in particular \(1/2\leq\mathcal J_\kappa,\mathcal J_\zeta\leq2\) a.e. Since, by the area formula (change of variables \(y=\kappa(x)\)),
\[
\int_W|F_s\circ\kappa|\mathop{}\mathopen{}\mathrm{d}\operatorname{vol}_M
\leq2\int_W|F_s\circ\kappa|\,\mathcal J_\kappa\mathop{}\mathopen{}\mathrm{d}\operatorname{vol}_M
=2\|F_s\|_{L^1(V_j)},
\]
the first term is at most \(C\sqrt\varepsilon|s|\,\|F_s\|_{L^1(V_j)}\). For the second term, recall that \(F_s\) is defined on \(M\) and vanishes outside \(V_j\). By the area formula (change of variables \(x=\zeta(y)\)),
\[
\int_{W\cap V_j}|F_s\circ\kappa-F_s|\mathop{}\mathopen{}\mathrm{d}\operatorname{vol}_M
\leq2\int_{V_j}|F_s\circ\zeta-F_s|\mathop{}\mathopen{}\mathrm{d}\operatorname{vol}_M
\leq4\sqrt\varepsilon|s|\,\|\nabla F_s\|_{L^1(V_j)}.
\]
To prove the last inequality, assume first that \(F_s\) is \(C^1\). Since \(u\mapsto\zeta^j_{\varepsilon,s,ut}(y)=y+ut\sqrt\varepsilon A_j(y)s\) parametrizes the segment from \(y\) to \(\zeta(y)\) and \(|A_j(y)s|\leq|s|\), we have
\[
|F_s(\zeta(y))-F_s(y)|\leq\sqrt\varepsilon|s|\int_0^1\bigl|\nabla F_s\bigl(\zeta^j_{\varepsilon,s,ut}(y)\bigr)\bigr|\mathop{}\mathopen{}\mathrm{d} u,
\]
and we conclude by the area formula (change of variables \(x=\zeta^j_{\varepsilon,s,ut}(y)\)), since the inverse of \(\zeta^j_{\varepsilon,s,ut}\) has Jacobian at most \(2\) by \Cref{lemma: C^1 diffeo} (applied with \(ut\) in place of \(t\)). In general, \(F_s\) is only Lipschitz on \(M\), since \(A_j\) and \(K_j(\cdot,s)\) are Lipschitz on \(V_j\) (see below) and \(f\) has compact support in \(V_j\). We then apply the above to mollifications of \(F_s\), which converge to \(F_s\) in \(W^{1,1}(M)\), and pass to the limit, using that \(\int_{V_j}|g\circ\zeta|\mathop{}\mathopen{}\mathrm{d}\operatorname{vol}_M\leq2\|g\|_{L^1(M)}\) for \(g\in L^1(M)\), since \(\mathcal J_\kappa\leq2\). Altogether,
\[
\big\|
\zeta_{\#}
(F_s\operatorname{vol}_M)
-
F_s\operatorname{vol}_M
\big\|_{\mathrm{TV}}
\leq
C\sqrt{\varepsilon}|s|
\left(
\|F_s\|_{L^1(V_j)}
+
\|\nabla F_s\|_{L^1(V_j)}
\right).
\]
It remains to bound the \(L^1\) norms of \(F_s\) and \(\nabla F_s\). By \eqref{eq: normal-ellipticity} and since \(Z_R\geq c\), we have \(K_j(x,s)\leq c^{-1}e^{-\lambda|s|^2/2}\). Moreover, \(x\mapstoH_{\alpha_j(x)}E\) is Lipschitz (since \(E\in C^{2,1}\), \(n\) is Lipschitz and \(\alpha_j\) is \(C^{1,1}\)), hence so are \(x\mapsto Z_R(\alpha_j(x))\) and \(K_j(\cdot,s)\), and differentiating a.e.\
\(K_j(x,s)=\mathbf1_{\mathbb{B}_R}(s)\,e^{-H_{\alpha_j(x)}E(s,s)/2}/Z_R(\alpha_j(x))\) with respect to \(x\) gives \(|\nabla_xK_j(x,s)|\leq C(1+|s|^2)e^{-\lambda|s|^2/2}\) for a.e.\ \(x\in V_j\). Since \(F_s=f\,K_j(\cdot,s)\,A_js\), with \(|A_js|\leq|s|\) and \(A_j\) Lipschitz, the product rule yields
\[
\|F_s\|_{L^1(V_j)}
\leq
C e^{-\lambda|s|^2/2}
|s|\|f\|_{L^1(V_j)},
\]
\[
\|\nabla F_s\|_{L^1(V_j)}
\leq
C e^{-\lambda|s|^2/2}
\left[
|s|\|\nabla f\|_{L^1(V_j)}
+
(|s|+|s|^3)\|f\|_{L^1(V_j)}
\right].
\]
Integrating with respect to \(s\in\mathbb{B}_R\), we obtain, uniformly in \(t\in[0,1]\),
\[
\|\mathbf{I}_{t,\varepsilon}^j[f]\|_{\mathrm{TV}}
\leq
C\sqrt{\varepsilon} (M_2+M_4)
\left(
\|f\|_{L^1(V_j)}
+
\|\nabla f\|_{L^1(V_j)}
\right),
\]
where \(M_2\) and \(M_4\) are defined in \eqref{eq: gaussian-moments}.
 
\emph{Conclusion.} Since \(\chi_j\) has compact support in \(U_j\), \(\rho_j\) vanishes outside the compact set \(\operatorname{\mathrm{proj}}_M(\operatorname{spt}\chi_j)\subseteq V_j\). By mollification, there exist a compact set \(S_j\subseteq V_j\) and a sequence \((\rho_j^{(k)})_{k\in\mathbb{N}}\) in \(C_c^1(V_j)\), all supported in \(S_j\), such that
\(\|\rho_j-\rho_j^{(k)}\|_{L^1(V_j)}\to0\) as \(k\to\infty\). Applying (b) with \(S=S_j\), there exists \(\varepsilon_0>0\), independent of \(k\), such that, for \(0<\varepsilon\leq\varepsilon_0\) and every \(k\in\mathbb{N}\), writing \(\mathbf{I}_{t,\varepsilon}^j[\rho_j]=\mathbf{I}_{t,\varepsilon}^j[\rho_j-\rho_j^{(k)}]+\mathbf{I}_{t,\varepsilon}^j[\rho_j^{(k)}]\), (a) and (b) give
\[
\sup_{t\in[0,1]}\|\mathbf{I}_{t,\varepsilon}^j[\rho_j]\|_{\mathrm{TV}}
\leq 2M_1\|\rho_j-\rho_j^{(k)}\|_{L^1(V_j)}
+
C\sqrt{\varepsilon}
\left(
\|\rho_j^{(k)}\|_{L^1(V_j)}
+
\|\nabla \rho_j^{(k)}\|_{L^1(V_j)}
\right).
\]
For \(r>0\), set
\[
\omega_j(r)
\stackrel{\text{def.}}{=}
\inf_{k\geq1}
\left\{
2M_1\|\rho_j-\rho_j^{(k)}\|_{L^1(V_j)}
+
rC
\left(
\|\rho_j^{(k)}\|_{L^1(V_j)}
+
\|\nabla \rho_j^{(k)}\|_{L^1(V_j)}
\right)
\right\},
\]
and \(\omega(r)\stackrel{\text{def.}}{=}\sum_{j\in J}\omega_j(r)\).
Since \(\rho_j^{(k)}\to\rho_j\) in \(L^1(V_j)\), we have
\(\omega(r)\to0\) as \(r\to 0\): one first fixes \(k\) sufficiently large so that the first
term is arbitrarily small, and then lets \(r\to0\).
Combining this with \eqref{eq: flow-decomposition} yields, for \(0<\varepsilon\leq\varepsilon_0\),
\[
\sup_{t\in[0,1]}
\frac{\|\operatorname{\mathfrak m_{t,\varepsilon}}\|_{\mathrm{TV}}}
{\sqrt{\varepsilon}}
\leq \sum_{j\in J} \sup_{t\in[0,1]} \|\mathbf{I}_{t,\varepsilon}^j[\rho_j]\|_{\mathrm{TV}} \leq
\omega(\sqrt{\varepsilon})
=o(1).
\]
 
\textbf{Step 5. Benamou--Brenier estimate.} Recall from Step~2 and \eqref{eq: interpolation-charts} that
\begin{align*}
\operatorname{\mathfrak m_{t,\varepsilon}} &= \sqrt{\varepsilon}\sum_{j\in J} \int_{\mathbb{B}_R} (\zeta_{\varepsilon,s,t}^j)_{\#}\left(\rho_jK_j(\cdot,s)A_js\, \operatorname{vol}_M\right) \mathop{}\mathopen{}\mathrm{d} s,
\\
\mu_{t,\varepsilon}&=\sum_{j\in J}\int_{\mathbb{B}_R}(\zeta_{\varepsilon,s,t}^j)_{\#}(\rho_jK_j(\cdot,s)\operatorname{vol}_M)\mathop{}\mathopen{}\mathrm{d} s.
\end{align*}
Since \(|A_j(x)s|\leq|n(\alpha_j(x))s|=|s|\leq R\), we get \(|\operatorname{\mathfrak m_{t,\varepsilon}}|\leq\sqrt\varepsilon R\,\mu_{t,\varepsilon}\). Thus \(\operatorname{\mathfrak m_{t,\varepsilon}}\ll\mu_{t,\varepsilon}\), and the velocity field
\[
v_{t,\varepsilon}
\stackrel{\text{def.}}{=}\frac{\mathop{}\mathopen{}\mathrm{d} \operatorname{\mathfrak m_{t,\varepsilon}}}{\mathop{}\mathopen{}\mathrm{d}\mu_{t,\varepsilon}}\in L^\infty(\mu_{t,\varepsilon};TM)
\quad\text{satisfies}\quad |v_{t,\varepsilon}|\le\sqrt\varepsilon R
\quad\mu_{t,\varepsilon}\text{-a.e.}
\]
Since \((\mu_{t,\varepsilon},v_{t,\varepsilon})\) satisfies the
continuity equation, with \(\mu_{0,\varepsilon}=\mu\) and \(\mu_{1,\varepsilon}=\widetilde\mu_\varepsilon\), the Benamou--Brenier formula (see, e.g., \cite{santambrogioOptimalTransportApplied2015}) yields
\begin{align*}
W_2^2(\mu,\widetilde\mu_\varepsilon)
&\leq\int_0^1\int_M |v_{t,\varepsilon}|^2
\,\mathop{}\mathopen{}\mathrm{d}\mu_{t,\varepsilon}\mathop{}\mathopen{}\mathrm{d} t=\int_0^1\int_M |v_{t,\varepsilon}|\,\mathop{}\mathopen{}\mathrm{d}|\operatorname{\mathfrak m_{t,\varepsilon}}|\mathop{}\mathopen{}\mathrm{d} t
\\
&\le\sqrt\varepsilon R\sup_{t\in[0,1]}
\|\operatorname{\mathfrak m_{t,\varepsilon}}\|_{\mathrm{TV}}.
\end{align*}
 
By Step~4, \(\sup_{t\in[0,1]}\|\operatorname{\mathfrak m_{t,\varepsilon}}\|_{\mathrm{TV}}\leq\sqrt\varepsilon\,\omega(\sqrt\varepsilon)\) for \(0<\varepsilon\leq\varepsilon_0\), hence
\[
W_2^2(\mu,\widetilde\mu_\varepsilon)
\leq
\varepsilon R\,\omega(\sqrt{\varepsilon}).
\]
Since \(\omega(r)\to0\) as \(r\to0\), this gives \(W_2(\mu,\widetilde\mu_\varepsilon)=o_R(\sqrt{\varepsilon})\).

\begin{align*}
\int_\Omega \phi\mathop{}\mathopen{}\mathrm{d}\operatorname{\widetilde{\mathbb{P}}_\varepsilon}

\leq(2\pi)^{-k/2}\beta^2\|\phi\|_\infty C_J\,\mathcal H^d(\Sigma\cap\omega)\int_{\mathbb{R}^k\setminus\mathbb{B}_R}e^{-\underline C|s|^2}\mathop{}\mathopen{}\mathrm{d} s
\xrightarrow[R\to+\infty]{}0,
\]
as the tail of an integrable Gaussian. This proves that the contribution of \(\mathcal T_r\setminus\mathcal T_{R\sqrt\varepsilon}\) is negligible.

\textbf{Step 3. Estimate on \(\Omega\setminus \Sigma_{r}\).}
If \(\mathcal K\setminus \Sigma_{r}= \emptyset\) there is nothing to do. Otherwise, \(\mathcal K\setminus\Sigma_r\) is a compact set which does not meet \(\Sigma\), so that \(E>0\) on it, and we denote

\min_{\mathcal K\setminus\Sigma_r}E>0.
\]
Since \(\operatorname{\widetilde{\mathbb{P}}_\varepsilon}\) is supported on \(\mathcal K\) and \(\mu^\otimes(\mathcal K)=1\),

\leq (2\pi\varepsilon)^{-k/2} \|\phi\|_{\infty}
e^{-\delta/\varepsilon}\xrightarrow[\varepsilon \to 0]{} 0.
\]

\[

\leq\left|\lim_{\varepsilon\to0}I_{1,\varepsilon,R}(\phi)-L(\phi)\right|+\sup_{\varepsilon}\big|I_{2,\varepsilon,R}(\phi)\big|,
\]
and the right-hand side tends to \(0\) as \(R\to+\infty\). In other words,
\[
\operatorname{\widetilde{\mathbb{P}}_\varepsilon}
\xrightharpoonup[]{\ast}
\bigl(\det \operatorname{Hess}^\perp E\bigr)^{-1/2}
\rho\mathbf 1_{\mathcal K}
\mathcal H^d_{|\Sigma}.
\]
Taking \(\phi\equiv1\), we get \(\operatorname{\widetilde{\mathbb{P}}_\varepsilon}(\Omega)=(2\pi\varepsilon)^{-k/2}Z_\varepsilon\to Z_0\), which is the asymptotics of \(Z_\varepsilon\). \[\operatorname{\mathbb{P}_\varepsilon} = \frac{\operatorname{\widetilde{\mathbb{P}}_\varepsilon}}{\operatorname{\widetilde{\mathbb{P}}_\varepsilon}(\Omega)}
\xrightharpoonup[]{\ast}
\frac1{Z_0}
\bigl(\det \operatorname{Hess}^\perp E\bigr)^{-1/2}
\rho\mathbf1_{\mathcal K}
\mathcal H^d_{|\Sigma},
\]
which proves the claim.

We follow the proof of \hyperref[thm: A]{Theorem~A} given in \Cref{sec: selection}, and only indicate the changes. By compactness of \(\Sigma\cap\overline\omega\), the tubular coordinates, the normal ellipticity and the quadratic detachment hold on \(\Sigma\cap\omega\) with uniform constants (see \Cref{subsec: Convergence of the Gibbs measure and the geometric Laplace's method}), and \Cref{thm: euclidean-Gibbs} replaces \Cref{thm: convergence-Gibbs-measure}. \textbf{Step 1. The marginals.} The marginal estimate of \Cref{thm: Wasserstein between the marginals} still holds: its proof only uses inverse charts of \(\operatorname{\mathrm{proj}}^\Sigma_{M_i}\), which exist on \(\Sigma\cap\omega\) by \Cref{prop: twist-geometry} and finitely many of which cover the compact set \(\Sigma\cap\mathcal K\), and the integrability of the chart densities of \(\overline\gamma\). The energy estimate of \Cref{prop: shadow-energy} still holds as well: since \(\Omega\) is convex, the segments used in its proof stay in \(\operatorname{int}\Omega\), and \eqref{eq: gradient-distance} holds with \(\Sigma\) replaced by \(\Sigma\cap\overline\omega\). \qquad\text{and}\qquad
\frac1\varepsilon\int E\,\mathop{}\mathopen{}\mathrm{d}(\gamma_{\varepsilon,R}-\widetilde\gamma_{\varepsilon,R})=o_R(1).

The entropy.} The entropy upper bound is the point to adapt since, as we wrote it, \(\operatorname{Ent}(\widetilde\gamma_{\varepsilon,R}\,|\,\mathbb{P}_\varepsilon)=+\infty\) in this case: \(\widetilde\gamma_{\varepsilon,R}\) charges \(\Omega\setminus\mathcal K\), where \(\operatorname{\mathbb{P}_\varepsilon}\) vanishes. We therefore consider the entropy with respect to the modified marginals:
\[

 \middle|\,\widetilde\mu_{1,\varepsilon,R}\otimes\widetilde\mu_{2,\varepsilon,R}\right).
\]
Since \(\mathop{}\mathopen{}\mathrm{d}\operatorname{\mathbb{P}_\varepsilon}=Z_\varepsilon^{-1}e^{-E/\varepsilon}\mathop{}\mathopen{}\mathrm{d}(\mu_1\otimes\mu_2)\), the shadow inequality (\Cref{prop: bound entropy shadow}) and Step~1 give, for fixed \(R\),
\[\begin{split}
 \operatorname{Ent}(\gamma_{\varepsilon,R}\,|\,\mathbb P_\varepsilon)&=\operatorname{Ent}(\gamma_{\varepsilon,R}\,|\,\mu_1\otimes\mu_2)+\log Z_\varepsilon+\frac1\varepsilon\int E\,\mathop{}\mathopen{}\mathrm{d}\gamma_{\varepsilon,R}\\
 &\leq I_{\varepsilon,R}+\log Z_\varepsilon+\frac1\varepsilon\int E\,\mathop{}\mathopen{}\mathrm{d}(\gamma_{\varepsilon,R}-\widetilde\gamma_{\varepsilon,R})+\frac1\varepsilon\int E\,\mathop{}\mathopen{}\mathrm{d}\widetilde\gamma_{\varepsilon,R}\\
 &\leq I_{\varepsilon,R}
 +\frac1\varepsilon\int E\,\mathop{}\mathopen{}\mathrm{d}\widetilde\gamma_{\varepsilon,R}
 +\log Z_\varepsilon+o_R(1).
\end{split}
\]
It remains to show that
\begin{equation}
\label{eq: euclidean-entropy-limsup}
 \limsup_{\varepsilon\to0}\left[
 I_{\varepsilon,R}
 +\frac1\varepsilon\int E\,\mathop{}\mathopen{}\mathrm{d}\widetilde\gamma_{\varepsilon,R}
 +\log Z_\varepsilon\right]
 \leq\operatorname{Ent}(\overline\gamma\,|\,\mathbb P_0)+b_R(\overline\gamma).
\end{equation}

Since \(\widetilde\mu_{1,\varepsilon,R}\otimes\widetilde\mu_{2,\varepsilon,R}\) has the product of the marginal densities as density, we have
\[
I_{\varepsilon,R}=\operatorname{Ent}(\widetilde\gamma_{\varepsilon,R})-\operatorname{Ent}(\widetilde\mu_{1,\varepsilon,R})-\operatorname{Ent}(\widetilde\mu_{2,\varepsilon,R}),

all these entropies being finite, since \(\widetilde\gamma_{\varepsilon,R}\) has finite entropy and all the measures are concentrated on fixed bounded sets. Hence
\begin{equation}
\label{eq: euclidean-entropy-decomposition}
I_{\varepsilon,R}
+\frac1\varepsilon\int E\,\mathop{}\mathopen{}\mathrm{d}\widetilde\gamma_{\varepsilon,R}
+\log Z_\varepsilon
=\Bigl(\operatorname{Ent}(\widetilde\gamma_{\varepsilon,R})+\frac1\varepsilon\int E\,\mathop{}\mathopen{}\mathrm{d}\widetilde\gamma_{\varepsilon,R}+\log Z_\varepsilon\Bigr)-\sum_{i=1}^2\operatorname{Ent}\left(\widetilde\mu_{i,\varepsilon,R}\right),
\end{equation}
and we pass to the limit in the two terms of the right-hand side in the next two steps.

\textbf{Step 3. The first term of \eqref{eq: euclidean-entropy-decomposition}.} Let \(\widetilde p_{\varepsilon,R}\) be the density \eqref{eq: normal-recovery-density} of \(\widetilde\gamma_{\varepsilon,R}\), and recall that \(K_R(z,s)=Z_R(z)^{-1}e^{-H_{z}E(s,s)/2}\) on \(\mathbb{B}_R\). \[

=\log\overline\rho(z)+\log\frac{Z_\varepsilon}{\varepsilon^{k/2}Z_R(z)}+\frac{\mathcal R(z,\sqrt\varepsilon s)}{\varepsilon}-\log\mathcal J_\Phi(z,\sqrt\varepsilon s).
\]

The integral of the last two terms is bounded by \(C\sqrt\varepsilon(M_1+M_3)\to0\), by \eqref{eq: tube-jacobian-rescaled} and \eqref{eq: gaussian-moments}. \[
\log\frac{Z_\varepsilon}{\varepsilon^{k/2}Z_R(z)}=\log\Bigl(Z_0\sqrt{\detH_{z}E}\Bigr)+\log\frac{Z_\infty(z)}{Z_R(z)}+o(1).
\]
Since \(\int_\Sigma|\log\overline\rho|\mathop{}\mathopen{}\mathrm{d}\overline\gamma<\infty\) (because \(\operatorname{Ent}(\overline\gamma\,|\,\mathbb P_0)<\infty\)) and \(\mathbb P_0\) has density \(Z_0^{-1}(\detH_{z}E)^{-1/2}\rho(z)\) with respect to \(\mathcal H^d_{|\Sigma}\) on \(\Sigma\cap\mathcal K\), we obtain
\begin{align*}
\lim_{\varepsilon\to0}\Bigl(\operatorname{Ent}(\widetilde\gamma_{\varepsilon,R})+\frac1\varepsilon\int E\,\mathop{}\mathopen{}\mathrm{d}\widetilde\gamma_{\varepsilon,R}+\log Z_\varepsilon\Bigr)
&=\int_\Sigma\log\frac{Z_0\,\overline\rho\sqrt{\detH_{z}E}}{\rho}\mathop{}\mathopen{}\mathrm{d}\overline\gamma+\int_\Sigma\log\rho\mathop{}\mathopen{}\mathrm{d}\overline\gamma+b_R(\overline\gamma)\\
&=\operatorname{Ent}(\overline\gamma\,|\,\mathbb P_0)+\operatorname{Ent}(\mu_1)+\operatorname{Ent}(\mu_2)+b_R(\overline\gamma),
\end{align*}
where we used that \(\overline\gamma\) has marginals \(\mu_1,\mu_2\) and \(\rho=\rho_1\otimes\rho_2\), so that \(\int\log\rho\mathop{}\mathopen{}\mathrm{d}\overline\gamma=\operatorname{Ent}(\mu_1)+\operatorname{Ent}(\mu_2)\).

\textbf{Step 4. The marginal entropies in \eqref{eq: euclidean-entropy-decomposition}.} By \eqref{eq: normal-perturbation-W2}, \(\widetilde\mu_{i,\varepsilon,R}\rightharpoonup\mu_i\) as \(\varepsilon\to0\), and all these measures are concentrated on the compact set \(K_i^1\). Since \(\operatorname{Ent}(\cdot)=\operatorname{Ent}(\cdot\,|\,\mathcal L_{|K_i^1})\) on such measures, and the relative entropy with respect to a finite measure is weakly lower semicontinuous, we get
\[
\liminf_{\varepsilon\to0}\operatorname{Ent}\left(\widetilde\mu_{i,\varepsilon,R}\right)\geq\operatorname{Ent}(\mu_i),\qquad i=1,2.
\]

\textbf{Conclusion.} Plugging Steps~3 and~4 into \eqref{eq: euclidean-entropy-decomposition} gives \eqref{eq: euclidean-entropy-limsup}, hence
\[
\limsup_{\varepsilon\to0}\operatorname{Ent}(\gamma_{\varepsilon,R}\,|\,\operatorname{\mathbb{P}_\varepsilon})\leq\operatorname{Ent}(\overline\gamma\,|\,\mathbb P_0)+b_R(\overline\gamma).
\] The diagonal argument of Step~2 of \Cref{subsec: selection-value-asymptotics} then provides a recovery sequence, and the remaining conclusions follow as in the proof of \hyperref[thm: A]{Theorem~A}.

On overlaps, \(n_b(z)=n_a(z)O_{ab}(z)\) with \(O_{ab}(z)\in O(k)\). The coordinate change \((z,s)\mapsto(z,O_{ab}(z)s)\) has absolute Jacobian one almost everywhere, even though \(O_{ab}\) depends on \(z\). It also preserves the normal Gaussian measure and \(\mathbb{B}_R\), that is 
\[
H_b=O_{ab}^{\top}H_aO_{ab},
\qquad
K_b(z,s)=K_a(z,O_{ab}s),
\] where \(H_a(z)\) is the matrix of the normal Hessian in the local orthonormal frame \(n_a(z)\). Thus the local normal perturbations agree on overlaps and define the recovery measure in \Cref{subsec: normal-recovery}. In the Euclidean case, we use a finite cover of the compact portion \(\Sigma\cap\overline\omega\) appearing in the Laplace proof.

\textbf{(iii) Marginal estimate.} The proof of \Cref{thm: Wasserstein between the marginals} uses local inverses of \(\operatorname{\mathrm{proj}}_{M_i}^\Sigma\), which, for \(N\geq3\), may have critical points (vertical branches). On the other hand, since \(|p(\Phi(z,\sqrt\varepsilon s))-p(z)|\leq\sqrt\varepsilon|s|\), the map \((z,s)\mapsto(p(z),p(\Phi(z,\sqrt\varepsilon s)))\) pushes \(K_R(z,s)\mathop{}\mathopen{}\mathrm{d} s\,\mathop{}\mathopen{}\mathrm{d}\overline\gamma^b(z)\) forward to a coupling of \(\mu^b\) and \(\widetilde\mu^b_\varepsilon\) with quadratic cost at most
\[

We follow the proofs of \Cref{thm: euclidean-Gibbs,thm: euclidean-selection}, with \(\Sigma_b\) in place of the \(d\)-dimensional contact set: these proofs do not use the dimension of the contact set, and they are simpler here, since the normal coordinates are global and \(E\) is exactly quadratic in them.

\textbf{Step 1. Gibbs measures and \(\Gamma\)-\(\liminf\).} The proof of \Cref{thm: euclidean-Gibbs} applies with \(\mathcal H^{(N-1)d}\) in place of \(\mathcal H^d\), \(\mathcal R\equiv0\), \(\mathcal J_\Phi\equiv1\) and \(\mathcal T_r=\Sigma_r\). Its assumption \(\mathcal H^{(N-1)d}(\Sigma_b\cap\partial\mathcal K)=0\), where \(\mathcal K=\prod_iK_i\), follows from \(\mathcal L^d(\partial K_i)=0\): for \(j\neq i\), parametrizing \(\Sigma_b\) by the coordinates \((x_l)_{l\neq j}\), the set \(\Sigma_b\cap\{x_i\in\partial K_i\}\) is the image of a negligible set by a Lipschitz map. Moreover, \(Z_0>0\), since \(\rho>0\) \(\gamma_M\)-a.e.\ and \(\gamma_M\ll\mathcal H^{(N-1)d}_{|\Sigma_b}\) (see Step~2). Since \(\detH_{z}E=(2N)^d\) is constant, this gives \(\operatorname{\mathbb{P}_\varepsilon}\rightharpoonup\mathbb P_0\) and \(Z_\varepsilon=(2\pi\varepsilon)^{d/2}(Z_0+o(1))\), and the \(\Gamma\)-\(\liminf\) inequality follows as in Step~1 of \Cref{subsec: selection-value-asymptotics}.

\textbf{Step 2. A finite-entropy optimal plan.} Parametrizing \(\Sigma_b\) by \((x_i)_{i\neq i_0}\), whose surface Jacobian is \(N^{d/2}\), we get \(\gamma_M=N^{-d/2}\prod_{i\neq i_0}\rho_i(x_i)\,\mathcal H^{(N-1)d}_{|\Sigma_b}\). Hence
\begin{equation}
\label{eq: harmonic-Rd-density}
\frac{\mathop{}\mathopen{}\mathrm{d}\gamma_M}{\mathop{}\mathopen{}\mathrm{d}\mathbb P_0}(\operatorname{\textbf{x}})=\frac{C_M}{\rho_{i_0}(x_{i_0})},
\qquad
C_M\stackrel{\text{def.}}{=} N^{-d/2}\int_{\Sigma_b}\rho\mathop{}\mathopen{}\mathrm{d}\mathcal H^{(N-1)d},
\end{equation}
and \(\operatorname{Ent}(\gamma_M\,|\,\mathbb P_0)=\log C_M-\operatorname{Ent}(\mu_{i_0})<\infty\), since \(\rho_{i_0}\) is bounded with compact support. This replaces Step~3 of \Cref{subsec: selection-value-asymptotics}, where the bound \(\overline\gamma\leq\beta\mathcal H^d_{|\Sigma}\) used \(\dim\Sigma=d\).

\textbf{Step 3. The marginals.} Fix \(\overline\gamma\in\Pi(\mu_1,\ldots,\mu_N)\) with \(\operatorname{Ent}(\overline\gamma\,|\,\mathbb P_0)<\infty\), and \(R\geq1\). Since \(H_{z}E=2N\,I_d\) does not depend on \(z\), neither does the kernel \(K_R(z,s)=K_R(s)\propto e^{-N|s|^2}\mathbf 1_{\mathbb{B}_R}(s)\). Hence, by \eqref{eq: recovery sequence with wrong marginals 1}, \(\widetilde\gamma_{\varepsilon,R}\) is the law of \(Z+\sqrt{\varepsilon/N}\,(S,\ldots,S)\), where \(Z\sim\overline\gamma\) and \(S\sim K_R\) are independent, and its \(i\)-th marginal is
\[
\widetilde\mu_{i,\varepsilon,R}=\mu_i*\eta_\varepsilon,
\qquad
\eta_\varepsilon\stackrel{\text{def.}}{=}\text{law of }\sqrt{\varepsilon/N}\,S.
\]
Writing \(\sqrt{\varepsilon/N}\,S=\sqrt{\varepsilon/(2N^2)}\,S'\), where \(S'=\sqrt{2N}\,S\) has density proportional to \(e^{-|s'|^2/2}\) on \(\mathbb{B}_{\sqrt{2N}R}\), we are in the setting of \Cref{prop: 1D}, with \(\varepsilon/(2N^2)\) and \(\sqrt{2N}R\) in place of \(\varepsilon\) and \(R\). Its proof applies verbatim in \(\mathbb{R}^d\), and gives \(W_2(\widetilde\mu_{i,\varepsilon,R},\mu_i)=o_R(\sqrt\varepsilon)\).

\textbf{Step 4. Energy and entropy.} The rest of the proof of \Cref{thm: euclidean-selection} is unchanged. The energy estimate of \Cref{prop: shadow-energy} holds, since \(|\nabla E(\mathbf y)|=2N\operatorname{dist}(\mathbf y,\Sigma_b)\) for every \(\mathbf y\in(\mathbb{R}^d)^N\). In Step~3 of the proof of \Cref{thm: euclidean-selection}, the remainder and Jacobian terms vanish, and \(\int|\log\overline\rho|\mathop{}\mathopen{}\mathrm{d}\overline\gamma<\infty\) follows from \(\operatorname{Ent}(\overline\gamma\,|\,\mathbb P_0)<\infty\) and \(\int|\log\rho|\mathop{}\mathopen{}\mathrm{d}\overline\gamma\leq\sum_i\int\rho_i|\log\rho_i|\mathop{}\mathopen{}\mathrm{d} x<\infty\).

\textbf{Step 5. Selection and value.} By \eqref{eq: harmonic-Rd-density}, \(\mathop{}\mathopen{}\mathrm{d}\gamma_M/\mathop{}\mathopen{}\mathrm{d}\mathbb P_0\) only depends on \(x_{i_0}\). Hence, for every \(\gamma\in\Pi(\mu_1,\ldots,\mu_N)\) with \(\operatorname{Ent}(\gamma\,|\,\mathbb P_0)<\infty\), as in the proof of \Cref{prop: MPW-selection},
\[
\operatorname{Ent}(\gamma\,|\,\mathbb P_0)=\operatorname{Ent}(\gamma\,|\,\gamma_M)+\int\log\frac{C_M}{\rho_{i_0}}\mathop{}\mathopen{}\mathrm{d}\mu_{i_0}=\operatorname{Ent}(\gamma\,|\,\gamma_M)+\operatorname{Ent}(\gamma_M\,|\,\mathbb P_0),

so that \(\gamma_M\) is the unique minimizer of \(\operatorname{Ent}(\cdot\,|\,\mathbb P_0)\) on \(\Pi(\mu_1,\ldots,\mu_N)\), that is, the selected plan. Finally, since \(C_M=2^{d/2}Z_0\),
\[
\min\mathcal F^{(1)}=\operatorname{Ent}(\gamma_M\,|\,\mathbb P_0)-\log Z_0-\frac d2\log(2\pi)
=-\operatorname{Ent}(\mu_{i_0})-\frac d2\log\pi,
\]
and the expansion of \(\min\mathcal F_\varepsilon\) follows from \hyperref[thm: A]{Theorem~A}.
