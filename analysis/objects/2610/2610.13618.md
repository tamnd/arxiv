    		\Delta\mathcal N f=f-\overline{f},\qquad \partial_n\mathcal N f=0,
    		\qquad\int_\Omega\mathcal N f \mbox{d}x=0.
    	\end{equation}
    	Thus this inverse is always applied after removal of the constant mode.
    	We first justify existence and uniqueness and explain the role of the subtracted average. For $1<q<\infty$, set
    	\[
    	V=\left\{v\in H^1(\Omega):\int_\Omega v \mbox{d}x=0\right\},
    	\qquad \lVert v\rVert_{V}:=\lVert \nabla v\rVert_{L^2}.
    	\]
    	
    	Poincar\'e's inequality shows that this norm is equivalent to the $H^1$ norm. The weak formulation for $w=\mathcal N f$ is
    	\[
    	\int_\Omega\nabla w\cdot\nabla v \mbox{d}x
    	=-\int_\Omega(f-\overline{f})v \mbox{d}x,\qquad v\in V.
    	\]
    	The right-hand side is bounded on $V$: in two dimensions
    	$H^1(\Omega)\hookrightarrow L^{q'}(\Omega)$ for every finite
    	$q'=q/(q-1)$, and
    	$\lVert f-\overline{f}\rVert_{L^q}\le2\lVert f\rVert_{L^q}$.
    	The bilinear form on the left is coercive on $V$. The Lax--Milgram theorem therefore gives a unique $w\in V$. Since $\int_\Omega(f-\overline{f})=0$, subtracting the mean of an arbitrary test function extends this identity to every $v\in H^1(\Omega)$. This is precisely the homogeneous Neumann boundary condition in the weak sense. Standard Neumann regularity on this domain gives $w\in W^{2,q}(\Omega)$. Alternatively, uniqueness follows by testing the difference of two solutions with itself: its gradient vanishes, and its zero average eliminates the remaining constant.
    	Thus \eqref{neumann-inverse} is well defined for every $f\in L^q$;
    	it is not an inverse of $\Delta$ on constants, since $\mathcal N c=0$.
    	
    	We shall use the following quantitative estimates for the Neumann Laplacian. They follow from the Calder\'on--Zygmund theory for singular	integral operators and the standard estimates for the two-dimensional
    	Neumann Green function; see \cite{CZ1952,TKB2013,Stein1970}. Here the explicit dependence on $a$ in \eqref{hessian} and	\eqref{projection} follows from the weak type $(1,1)$ estimate and the $L^2$ boundedness of the corresponding Calder\'on--Zygmund operators, together with the Marcinkiewicz interpolation theorem and duality.
    	
    	\begin{lem}\label{elliptic}
    		
    		There exists a constant $C_\Omega$ depending only on $\Omega$ such that:
    		
    		\begin{enumerate}
    			\item If $1<a\le2$, then
    			\begin{equation}\label{hessian}
    				\lVert D^2\mathcal N f\rVert_{L^a}\le {C_\Omega\over a-1}\lVert f\rVert_{L^a}.
    			\end{equation}
    			\item If $z$ is the zero-mean weak solution of
    			\[
    			\int_\Omega\nabla z\cdot\nabla\varphi
    			=\int_\Omega v\cdot\nabla\varphi\qquad(\varphi\in H^1(\Omega)),
    			\]
    			then for $2\le a<\infty$,
    			\begin{equation}\label{projection}
    				\lVert \nabla z\rVert_{L^a}\le C_\Omega a\lVert v\rVert_{L^a}.
    			\end{equation}
    			Equivalently, this corresponds to
    			\[
    			\Delta z=\operatorname{div} v,\qquad \partial_n z=v\cdot n
    			\]
    			in the weak sense.
    			\item The zero-mean Neumann Green function, with sign convention
    			$\Delta_yN(x,y)=\delta_x(y)-|\Omega|^{-1}$, satisfies
    			\begin{equation}\label{green-gradient}
    				|\nabla_yN(x,y)|\le {C_\Omega\over |x-y|}\quad(x\ne y).
    			\end{equation}
    		\end{enumerate}
    		
    	\end{lem}
    	
    	
    	
    	Recalling the definitions $F=(2\mu + \lambda(\rho))\mbox{div}u -P$ and $\omega=\partial_2 u_1- \partial_1 u_2$, we have
    	\begin{equation}\label{2-1}
    		\partial_t ( \mbox{div}(\rho u) ) + ( \partial_{11}(\rho u_1^2) + 2 \partial_1 \partial_2 (\rho u_1 u_2) + \partial_{22}( \rho u_2^2 ) ) = \Delta F + \rho \Delta \Phi + \nabla \rho \cdot \nabla \Phi.
    	\end{equation}
    	
    	Let $\xi$ solve
    	\begin{equation}\label{xi}
    		\Delta\xi=\mbox{div}(\rho u),\qquad (\partial_n\xi)|_{\partial \Omega}=0,
    		\qquad \int_{\Omega} \xi \mbox{d}x=0.
    	\end{equation}
    	Compatibility follows from $\rho (u\cdot n)=0$ on the boundary $\partial \Omega$. Define $\eta$ by
    	\begin{equation}\label{eta}
    		\Delta\eta=\mbox{div}\mbox{div} G:= \mbox{div}\mbox{div}(\rho u \otimes u),\qquad
    		\partial_n\eta|_{\partial \Omega}=((\mbox{div} G)\cdot n)|_{\partial \Omega},
    		\qquad\int_{\Omega}\eta \mbox{d}x=0.
    	\end{equation}
    	This Neumann problem is compatible by the divergence theorem. Since $u$
    	is tangent to the boundary,
    	\begin{align}
    		(\mbox{div} G)\cdot n
    		&=\rho((u\cdot\nabla)u)\cdot n
    		=-\rho\,u\cdot((u\cdot\nabla)n)
    		\quad\text{on }\partial\Omega.\label{curvature}
    	\end{align}
    	In particular, this datum is not in general zero.
    	
    	For completeness, the $L^a$ construction of \eqref{eta} can be carried out without estimating any boundary trace of $G$. For smooth $G$ with $Gn|_{\partial \Omega} = ((\rho u) u \cdot n)|_{\partial \Omega}=0$ and any smooth $\varphi$ with $\partial_n\varphi=0$, two integrations by parts give
    	\begin{align}
    		\int_{\Omega}\eta\Delta\varphi \mbox{d}x
    		&= \int_{\partial \Omega} \nabla \varphi \cdot n \eta \mbox{d}s - \int_{\Omega} \mbox{div} G \cdot \nabla \varphi \mbox{d}x =\int_{\Omega}\varphi\mbox{div}\mbox{div} G \mbox{d}x
    		-\int_{\partial\Omega}\varphi(\mbox{div} G)\cdot n \mbox{d}s\notag\\
    		& = -\int_{\Omega}\nabla\varphi\cdot\mbox{div} G \mbox{d}x + \int_{\partial\Omega} \varphi \mbox{div} G \cdot n \mbox{d}s-\int_{\partial\Omega}\varphi(\mbox{div} G)\cdot n \mbox{d}s\notag\\ 
    		&=\int_{\Omega}D^2\varphi : G \mbox{d}x - \int_{\partial\Omega} (Gn) \cdot \nabla \varphi \mbox{d}s
    		=\int_{\Omega} G:D^2\varphi \mbox{d}x.\label{eta-duality}
    	\end{align}
    	
    	If $a\ge2$, $a'=a/(a-1)$, and $f\in L^{a'}$, take
    	$\varphi=\mathcal{N} f$. The subtracted average must first be retained:
    	\begin{align*}
    		\int_{\Omega}\eta(f- \bar{f}) \mbox{d}x
    		&=\int_{\Omega}\eta\Delta\mathcal{N} f \mbox{d}x
    		=\int_{\Omega} G:D^2\mathcal{N} f \mbox{d}x,\\
    		\int_{\Omega}\eta(f-\bar{f}) \mbox{d}x
    		&=\int_{\Omega}\eta f \mbox{d}x-\bar{f}\int_{\Omega}\eta \mbox{d}x
    		=\int_{\Omega}\eta f \mbox{d}x.
    	\end{align*}
    	Thus $\bar{f}$ disappears because $\eta$ has zero average; the test
    	function $f$ itself need not have zero average. H\"older's inequality and
    	\eqref{hessian}, with $1<a'\le2$, give
    	\begin{align*}
    		\left|\int_{\Omega}\eta f \mbox{d}x\right|
    		&\le\| G\|_{L^a}\|D^2\mathcal{N} f\|_{L^{a'}}\le\frac{C}{a'-1}\| G\|_{L^a} \| f\|_{L^{a'}}
    		=C(a-1)\| G\|_{L^a}\|f\|_{L^{a'}}.
    	\end{align*}
    	Taking the supremum over $\|f\|_{L^{a'}}\le1$ gives
    	$\|\eta\|_{L^a}\le C(a-1)\| G\|_{L^a}$.
    	To make this duality argument explicit, if $\eta\ne0$, the admissible test function
    	\[
    	f_\eta=\frac{|\eta|^{a-2}\eta}{\|\eta\|_{L^a}^{a-1}}
    	\]
    	satisfies $\|f_\eta\|_{L^{a'}}=1$ and $\int_{\Omega}\eta f_\eta \mbox{d}x=\|\eta\|_{L^a}$; the reverse inequality for all admissible test functions follows from H\"older's inequality.
    	
    	There is no need to assume the desired $L^a$ regularity in this argument.
    	For $G\in L^a$, define
    	$\Lambda_G(f)=\int_{\Omega} G:D^2\mathcal{N} f \mbox{d}x$ on $L^{a'}$.
    	The preceding bound and $L^p$ duality produce a unique representing
    	$\eta\in L^a$. Since $\mathcal{N}1=0$, testing with $1$ gives
    	$\int_{\Omega}\eta\mbox{d}x=0$. For regular $G$ with $Gn=0$, the integration by
    	parts calculation identifies this representative with \eqref{eta}.
    	Finally, with the Frobenius matrix norm,
    	$|\rho u\otimes u|=\rho|u|^2$. We have proved
    	\begin{equation}\label{etaLp}
    		\|\eta\|_{L^a}\le Ca\|\rho|u|^2\|_{L^a},\qquad a\ge2.
    	\end{equation}
    	
    	
    	Therefore, following the argument in \cite{VK1995}, we obtain the following estimates for the solutions of \eqref{xi} and \eqref{eta}:
    	\begin{align}\label{2-4}
    		&\|\nabla \xi \|_{L^{2m}} \leq m C\|u\|_{L^{2mk}} \|\rho \|_{L^{\frac{2mk}{k-1}}},\nonumber\\[2mm]
    		& \|\nabla \xi \|_{L^{2-r}} \leq C\|\sqrt{\rho} u\|_{L^2}\|\rho \|^\frac{1}{2}_{L^{\frac{2-r}{r}}}, \nonumber\\[2mm]
    		& \| \eta\|_{L^{2m}}\le  m C\|u\|_{L^{4mk}}^2 \|\rho \|_{L^{\frac{2mk}{k-1}}}.
    	\end{align}
    	where $C$ is a positive constant independent of $k>1$, $m \geq 1$, and $r$ with $1 \geq 2r \geq 0$.
    	
    	
    	
    	Moreover, we obtain (\cite{VK1995}, Lemma 1-3)
    	\begin{align}\label{2-5}
    		&\|u\|_{L^{2m}} \leq m^{\frac{1}{2}} C \|\nabla u\|_{L^2}, \quad m>2,\nonumber\\[2mm]
    		& \|\xi \|_{L^{2m}} \leq m^{\frac{1}{2}} C \|\nabla \xi \|_{L^{\frac{2m}{m+1}}}, \quad m>2,
    	\end{align}
    	where $C$ is independent of $m$.
    	
    	Since $\|\nabla u\|^2_{L^2} \leq \int_{\Omega}( (2 \mu + \lambda)|\mbox{div}u|^2+ \mu |\omega|^2 )\mbox{d}x :=\psi(t),$ we infer that
    	\begin{align}\label{2-6}
    		&\| \xi \|_{L^{2m}} \leq m^{\frac{1}{2}} C \|\rho \|^{\frac{1}{2}}_{L^{m}},\nonumber\\[2mm]
    		& \|\nabla \xi \|_{L^{2m}} \leq m^{\frac{3}{2}} k^{\frac{1}{2}} C \|\rho \|_{L^{\frac{2mk}{k-1}}}(\psi(t))^{\frac{1}{2}},\nonumber\\[2mm]
    		& \|\eta \|_{L^{2m}} \leq  m^{2} k C \psi(t) \|\rho \|_{L^{\frac{2mk}{k-1}}},
    	\end{align}
    	for $m>2$ and $k>1$.
    	
    	Using the definitions of $\xi (x,t)$ and $\eta(x,t)$, we obtain
    	\begin{equation}\label{2-7}
    		\Delta( \xi_t + \eta - F -K + \frac{1}{|\Omega|}\int_{\Omega} F \mbox{d}x)=0,
    	\end{equation}
    	where we define
    	\begin{equation}\label{2-8}
    		\Delta K = \mbox{div}( \rho \nabla \Phi), \quad \int_{\Omega} K \mbox{d}x=0, \quad (\nabla K \cdot n)|_{\partial \Omega}=0.
    	\end{equation}
    
  
    	The boundary values of $\xi (x,t)$ and $\eta(x,t)$, together with the boundary conditions for $K$, we have
    	$$ \partial_n(\xi_t + \eta - F -K + \frac{1}{|\Omega|}\int_{\Omega} F \mbox{d}x)|_{\partial \Omega}= \mu \partial_{\tau}(A (u \cdot \tau))|_{\partial \Omega},$$
    	therefore by the uniqueness of the zero-mean solution of the Neumann
    	problem, imply
    	\begin{equation}\label{2-9}
    		\xi_t + \eta -F -K + \frac{1}{|\Omega|} \int_{\Omega} F \mbox{d}x=h.
    	\end{equation}
    	
    	For each fixed time for which $u\in H^1(\Omega)$, define $h$ as the
    	zero-mean weak solution of
    	\begin{equation}\label{h}
    		\Delta h=0,\qquad
    		(\partial_nh)|_{\partial \Omega}=(\mu\partial_\tau\big(A(u\cdot\tau)\big))|_{\partial \Omega},
    		\qquad\int_\Omega h \mbox{d}x=0.
    	\end{equation}
    	This definition requires both compatibility and a solvability argument.
    	Write $g_\partial=A(u\cdot\tau)|_{\partial\Omega}$. The trace theorem
    	gives $g_\partial\in H^{1/2}(\partial\Omega)$, so
    	$k=\mu\partial_\tau g_\partial\in H^{-1/2}(\partial\Omega)$.
    	Tangential differentiation on the closed boundary gives
    	\[
    	\langle k,1\rangle_{\partial\Omega}
    	=-\mu\langle g_\partial,\partial_\tau1\rangle_{\partial\Omega}=0.
    	\]
    	For smooth data this is simply the integral of a derivative around a
    	closed curve. It is exactly the Neumann compatibility condition
    	$\int_{\partial\Omega}\partial_nh\mbox{d}s=\int_\Omega\Delta h \mbox{d}x=0$.
    	
    	We now give a direct construction that also yields the required estimate.
    	Choose once and for all a vector field
    	$\mathcal{B} \in W^{1,\infty}(\Omega;\mathbb R^2)$ with boundary trace $A\tau$,
    	and put $g= \mathcal{B} \cdot u\in H^1(\Omega)$. Such a fixed extension exists by
    	the smoothness of the boundary and $A$. Only its boundary values are
    	prescribed; it need not be a unit tangent field inside $\Omega$.
    	For smooth $g$ the orientation convention gives
    	\[
    	\operatorname{div}\nabla^\perp g=0,\qquad
    	n\cdot\nabla^\perp g
    	=n_1\partial_2g-n_2\partial_1g=-\partial_\tau g.
    	\]
    	Integration by parts therefore gives the weak problem
    	\begin{equation}\label{hweak}
    		\int_\Omega\nabla h\cdot\nabla\varphi \mbox{d}x
    		=-\mu\int_\Omega\nabla^\perp g\cdot\nabla\varphi \mbox{d}x,
    		\qquad \varphi\in H^1(\Omega).
    	\end{equation}
    	The same identity holds for $g\in H^1$ by approximation and the
    	trace duality just described. The functional on its right is bounded:
    	\[
    	\left|\mu\int_\Omega\nabla^\perp g\cdot\nabla\varphi \mbox{d}x\right|
    	\le\mu\lVert \nabla g\rVert_{L^2}\lVert \nabla\varphi\rVert_{L^2}.
    	\]
    	On the zero-mean Hilbert space $V$, the left-hand bilinear form is coercive. The Lax--Milgram theorem yields a unique $h\in V$ satisfying \eqref{hweak} for test functions in $V$; subtracting the mean of a test function extends the identity to all of $H^1$. For compactly supported test functions, the right-hand side vanishes since $\operatorname{div}\nabla^\perp g=0$, proving $\Delta h=0$ in distributions. The identity for general tests then identifies the weak normal derivative with $\mu\partial_\tau g_\partial$. Thus it solves \eqref{h}. If two zero-mean solutions existed, their difference $d$ would satisfy $\int_\Omega|\nabla d|^2 \mbox{d}x=0$, and hence $d=0$.
    	
    	The construction is independent of the interior extension. Two choices
    	of $g$ with the same trace differ by $v\in H^1_0(\Omega)$, and
    	$\int_\Omega\nabla^\perp v\cdot\nabla\varphi \mbox{d}x=0$ for every
    	$\varphi\in H^1(\Omega)$, by approximation of $v$ with compactly
    	supported smooth functions. Their weak problems are identical.
    	
    	For the estimate, take $\varphi=h$ in \eqref{hweak}:
    	\begin{align*}
    		\lVert \nabla h\rVert_{L^2}^2
    		&=-\mu\int_\Omega\nabla^\perp g\cdot\nabla h \mbox{d}x \le\mu\lVert \nabla g\rVert_{L^2}\lVert \nabla h\rVert_{L^2}.
    	\end{align*}
    	Dividing by $\lVert \nabla h\rVert_{L^2}>0$ (the zero case is immediate) and applying Poincar\'e's inequality for functions with zero mean gives
    	\[
    	\lVert h\rVert_{H^1}\le C_\Omega\lVert \nabla h\rVert_{L^2}
    	\le C_\Omega\mu\lVert \nabla g\rVert_{L^2}.
    	\]
    	The product rule, with $\mathcal{B}$ fixed, gives
    	\begin{align*}
    		\partial_i g&=\sum_j\big((\partial_i \mathcal{B}_j)u_j+\mathcal{B}_j\partial_i u_j\big),\\
    		\lVert \nabla g\rVert_{L^2}
    		&\le C\big(\lVert D\mathcal{B}\rVert_{L^\infty}\lVert u\rVert_{L^2}
    		+\lVert \mathcal\rVert_{B}{L^\infty}\lVert \nabla u\rVert_{L^2}\big)
    		\le C_{\Omega,A}\lVert u\rVert_{H^1}.
    	\end{align*}
    	Combining this with Lemma \ref{lem-poincare} gives
    	\[
    	\lVert h\rVert_{L^a}\le C\sqrt a\,
    	\lVert h\rVert_{L^2}^{2/a}\lVert h\rVert_{H^1}^{1-2/a}
    	\le C\sqrt a\,\lVert h\rVert_{H^1}.
    	\]
    	Consequently,
    	\begin{equation}\label{hLp}
    		\lVert h\rVert_{L^a}\le C\sqrt a\| u\|_{H^1} ,\qquad a\ge2.
    	\end{equation}  	
    	
    	
    	
    	Applying $\eqref{origin}^1$, we obtain
    	\begin{equation}\label{2-10}
    		\xi_t + (2\mu + \lambda(\rho))\frac{1}{\rho}(\rho_t + \nabla \rho \cdot u) +P + \eta - K + \frac{1}{|\Omega|}\int_{\Omega} F \mbox{d}x - h=0.
    	\end{equation}
    	
    	Define the function $\theta(\rho)$ on $(0,+\infty)$ by
    	\begin{equation}\label{theta}
    		\theta(\rho)=\int_{1}^{\rho}\frac{1}{s}( 2\mu + \lambda(\rho))\mbox{d}s=2\mu \ln \rho + \frac{1}{\beta}(\rho^{\beta}-1),
    	\end{equation}
    	where we used that $\lambda(\rho)=\rho^{\beta}$.
    	
    	We then obtain
    	\begin{equation}\label{2-11}
    		\frac{\partial }{\partial t} ( \xi + \theta(\rho) ) + u \cdot \nabla ( \xi + \theta(\rho)) + P + \eta -K - u \cdot \nabla \xi + \frac{1}{|\Omega|} \int_{\Omega} F \mbox{d}x -h =0.
    	\end{equation}
    	
    	Define $f(x,t)$ by
    	\begin{equation}\label{2-12}
    		f=\max\{ 0 , \xi(x,t) + \theta(\rho(x,t)) \}, \quad (x,t)\in \bar{\Omega_T}
    	\end{equation}
    	
    We multiply equation \eqref{2-11} by $\rho f^{2m-1}$, where $m \geq 4$ is a natural number, and integrate over $\Omega$. Using the continuity equation and the boundary conditions for $u$, we obtain
    	\begin{align}\label{2-13}
    		& \frac{1}{2m} \frac{d}{dt} \int_{\Omega} \rho f^{2m} \mbox{d}x + \int_{\Omega} P \rho f^{2m-1} \mbox{d}x + \int_{\Omega} \rho \eta f^{2m-1} \mbox{d}x - \int_{\Omega}\rho K f^{2m-1} \mbox{d}x\nonumber\\[2mm]
    		& - \int_{\Omega} \rho(u \cdot \nabla) \xi  f^{2m-1} + \frac{1}{|\Omega|}\int_{\Omega} F \mbox{d}x \int_{\Omega} \rho f^{2m-1}\mbox{d}x - \int_{\Omega} \rho h f^{2m-1} \mbox{d}x= 0.
    	\end{align}
    	
    	Define
    	\begin{equation}\label{2-14}
    		Z(t)= ( \int_{\Omega}\rho f^{2m} \mbox{d}x )^{\frac{1}{2m}}.
    	\end{equation}
    	
    	Using the preceding estimates for $\xi$ and $\eta$, the positivity $\rho >0$, and H$\ddot{o}$lder's inequality, we estimate the relevant terms in \eqref{2-13} as follows:
    	
    	\begin{align}\label{2-15}
    		&|\int_{\Omega}\rho \eta f^{2m-1} \mbox{d}x| \leq C \|\rho \|^{\frac{1}{2m}}_{L^{2m\beta+1}} Z^{2m-1}(t) \|\eta\|_{L^{2m+\frac{1}{\beta}}} \leq C m^2 \|\rho \|^{1+\frac{1}{2m}}_{L^{2m\beta+1}} \psi(t) Z^{2m-1}(t),\nonumber\\[2mm]
    		& | \int_{\Omega} \rho (u \cdot \nabla)\xi f^{2m-1} \mbox{d}x| \leq C m^2 \|\rho\|^{1+\frac{1}{2m}}_{L^{2m\beta+1}} \psi(t) Z^{2m-1}(t),\nonumber\\[2mm]
    		& |\frac{1}{|\Omega|}\int_{\Omega} F \mbox{d}x \int_{\Omega} \rho f^{2m-1}\mbox{d}x| \leq C Z^{2m-1}(t)( 1 + (\psi(t))^{\frac{1}{2}} + \|\rho\|^{\frac{\beta}{2}}_{L^{2m\beta+1}} (\psi(t))^{\frac{1}{2}} ).
    	\end{align}
    	
    	
    	Recalling that $\Delta K=\mbox{div}( \rho \nabla \Phi )$, we apply the Poincar\'{e} inequalities (Lemma \ref{lem-poincare} and Lemma \ref{lem-poincare2}), Young's inequality, and H$\ddot{o}$lder's inequality to obtain
    	$$\|K\|_{L^{2m+1/\beta}} \leq C m^{\frac{1}{2}} \|\nabla K\|_{L^2} \leq C m^{\frac{1}{2}} \|\rho \nabla \Phi\|_{L^2} \leq Cm^2\|\rho \|_{L^{2m\beta +1}},$$
    	where we have again used Lemma \ref{lem-poincare}, together with Lemma \ref{lem-div-curl}, the identity $\Delta \Phi =\rho -b(x)$, and the boundedness of $b(x)$.
    	
    	Therefore, we obtain
    	\begin{align}\label{2-16}
    		&|\int_{\Omega}\rho K f^{2m-1}| \leq C m^2 \|\rho \|^{1+\frac{1}{2m}}_{L^{2m\beta+1}} Z^{2m-1}(t).
    	\end{align}
    	
    	For the last term, we have
    	\begin{align}
    		& |\int_{\Omega}\rho h f^{2m-1}| \leq C \|\rho \|^{\frac{1}{2m}}_{L^{2m\beta+1}} Z^{2m-1}(t) \|h\|_{L^{2m+\frac{1}{\beta}}} \nonumber\\[2mm]
    		& \leq C m^{\frac{1}{2}}\|\rho \|^{\frac{1}{2m}}_{L^{2m\beta+1}} Z^{2m-1}(t) \|u\|_{H^1} \leq C m^{\frac{1}{2}} \|\rho \|^{\frac{1}{2m}}_{L^{2m\beta+1}} Z^{2m-1}(t) (\psi(t))^{\frac{1}{2}}.
    	\end{align}
    	
    	Combining the preceding inequalities with the nonnegativity of the second term in \eqref{2-13}, Young's inequality, and the lower bound for the mass, we derive
    	\begin{equation}\label{2-17}
    		Z(t) \leq C( 1+ \int_{0}^{t} m^2 (\psi(t)+1) \|\rho\|^{1+\frac{1}{2m}}_{L^{2m\beta+1}} \mbox{d}\tau+ \int_{0}^{t}(\psi(t))^{\frac{1}{2}}\|\rho\|^{\frac{\beta}{2}}_{L^{2m\beta+1}}\mbox{d}\tau ),
    	\end{equation}
    	for $m \geq 4$.
    	    	
    	Following the argument in \cite{VK1995}(equations (31)--(34)), we obtain
    	\begin{equation}\label{2-18}
    		\|\rho\|^{\beta}_{L^{2m\beta+1}} \leq C (Z(t) +m^{\frac{\beta}{2 \beta-1}}).
    	\end{equation}
    	
    	Observe that the positive constant $C$ in the preceding inequalities is independent of $m \geq 4$. Thus, \eqref{2-17} and \eqref{2-18} imply
    	\begin{align}\label{2-19}
    		\|\rho\|^{\beta}_{L^{2m\beta+1}} \leq C(m^{\frac{\beta}{2 \beta-1}} + m^2 \int_{0}^{t} (\psi(t)+1) \|\rho\|^{1+\frac{1}{2m}}_{L^{2m\beta+1}} \mbox{d}\tau+\int_{0}^{t}(\psi(t))^{\frac{1}{2}}\|\rho\|^{\frac{\beta}{2}}_{L^{2m\beta+1}}\mbox{d}\tau ),
    	\end{align}
    	where $C$ remains independent of $m \geq 4$.
    	
    	Using the fact established in Lemma \ref{lem1} that $\psi (t) \in L^1(0,T)$ and applying Gronwall's inequality, we obtain
    	\begin{equation}\label{2-20}
    		\|\rho\|^{\beta}_{L^{2m\beta+1}} \leq C(m^{\frac{\beta}{2 \beta-1}} + m^2 \int_{0}^{t} (\psi(t)+1) \|\rho\|^{1+\frac{1}{2m}}_{L^{2m\beta+1}} \mbox{d}\tau).
    	\end{equation}
    	
    	Setting $y(t)=m^{-\frac{2}{\beta-1}}\|\rho\|_{L^{2m\beta+1}}(t)$ for $t \in [0,T]$, we obtain
    	\begin{equation}\label{2-21}
    		y^{\beta}(t) m^{\frac{2\beta}{\beta-1}} \leq C(m^{\frac{2\beta}{2 \beta-1}} + m^2m^{(1+\frac{1}{m})(\frac{2}{\beta-1})} \int_{0}^{t} (\psi(t)+1) y^{1+\frac{1}{m}}(\tau) \mbox{d}\tau).
    	\end{equation}

The standard local existence result, Lemma \ref{lem-localexistence}, ensures that the problem \eqref{origin}-\eqref{boundary2} with initial data $(\rho_0,u_0,\Phi_0)$ has a unique local solution $(\rho,u,\nabla \Phi)$ defined up to a positive time $T_0$, which may depend on $\inf_{x\in \Omega }\rho_0(x)$, and satisfying \eqref{localexistence} and \eqref{localinf}. We set
    	
    	\begin{equation}\label{p1-1}
    		T^*= \sup\{ T| \sup_{0 \leq t \leq T} (\|(\rho,u)\|_{H^2}+\|\nabla \Phi\|_{H^1}) < \infty \}.
    	\end{equation}
    	
    	Clearly, $T^* \geq T_0$. If $T^*< \infty$, we claim that there exists a positive constant $\hat{C}$ which may depend on $b(x)$, $T^*$ and $\inf_{x\in \Omega } \rho_0(x)$ such that, for all $0<T<T^*$,
    	\begin{equation}\label{p1-2}
    		\sup_{0 \leq t \leq T}\|\rho\|_{H^2} \leq \hat{C}.
    	\end{equation}
    	
    	This together with \eqref{lem15-1} contradicts \eqref{p1-1}. Thus, we have
    	\begin{equation}\label{p1-3}
    		T^*=\infty.
    	\end{equation}
    	
    	It remains to prove \eqref{p1-2}. First, standard calculations and \eqref{lem15-1} imply that, for any $T \in (0,T^*)$,
    	\begin{equation}\label{p1-4}
    		\inf_{(x,t)\in \Omega \times (0,T)} \rho(x,t) \geq \inf_{x\in \Omega } \rho_0(x) \exp^{-\int_{0}^{T}\|\mbox{div}u\|_{L^{\infty}}\mbox{d}t} \geq \hat{C}^{-1},
    	\end{equation}
    	Here and below, $\hat{C}$ denotes a generic positive constant depending on $\inf_{x\in \Omega } \rho_0(x)$ and $T^*$ but independent of $T$. In view of \eqref{initialvalue}, we define
    	\begin{equation}\label{p1-5}
    		\sqrt{\rho}\tilde{u}(x,t=0) = \rho_0^{-\frac{1}{2}}( \mu \Delta u_0+ \nabla ((\mu + \lambda(\rho_0))\mbox{div}u_0)- \nabla P(\rho_0)).
    	\end{equation}
    	
    	Integrating \eqref{13-16} with respect to $t$ over $(0,T)$ and using \eqref{initialvalue}, Lemma \ref{lem11}, and \eqref{p1-5}, we obtain
    	\begin{equation}\label{p1-6}
    		\sup_{0 \leq t \leq T} \int_{\Omega} \rho |\tilde{u}|^2 \mbox{d}x + \int_{0}^{T} \|\nabla \dot{u}\|^2_{L^2}\mbox{d}t \leq \hat{C}.
    	\end{equation}
    	
    	
    	Combining this with \eqref{15-11} for $\|\nabla^2 u\|_{L^2}$ and \eqref{lem15-1}, and recalling that $\Delta F= \mbox{div}( \rho \tilde{u})$ and $\mu \Delta \omega= \nabla^{\bot} \cdot (\rho \tilde{u})$, we obtain
    	\begin{align}\label{p1-7}
    		& \sup_{0 \leq t \leq T}( \|\nabla^2 u\|_{L^2} + \|\nabla F\|_{L^2}+ \|\nabla \omega\|_{L^2}) + \int_{0}^{T}( \|\nabla^2 F\|_{L^2}^2 + \|\nabla^2 \omega\|_{L^2}^2 )\mbox{d}t \nonumber\\[2mm]
    		& \leq \hat{C} \sup_{0 \leq t \leq T} \|\rho \tilde{u}\|_{L^2} + \hat{C}\int_{0}^{T} \|\nabla(\rho \tilde{u})\|_{L^2}^2 \mbox{d}t \nonumber\\[2mm]
    		& \leq \hat{C} + \hat{C}\int_{0}^{T}( \|\nabla \rho\|_{L^q}^2\|\tilde{u}\|_{L^{\frac{2q}{q-2}}}^2 + \|\nabla \tilde{u}\|_{L^2}^2 )\mbox{d}t \leq \hat{C} + \hat{C}\int_{0}^{T} \|\tilde{u}\|_{H^1}^2\mbox{d}t \leq \hat{C},
    	\end{align}
    	where we have used \eqref{5-3} with $k=1$ and $p=2$. The additional terms involving $u$ and $\tilde{u}$ are controlled by \eqref{p1-6}, Lemma \ref{lem15}, and \eqref{15-11}.

    		\partial_t \Psi_i + ( u \cdot \nabla )\Psi_i + ( 2\mu +\lambda(\rho))\nabla \rho \cdot \partial_i u+ \rho \partial_i F+ \rho \partial_i P +\Psi_i \mbox{div}u=0,
    	\end{equation}
    	multiplying the resulting identity by $\nabla \Psi_i$, integrating by parts, and using \eqref{p1-4} and \eqref{p1-7}, we obtain

    		\frac{d}{dt}\|\nabla \Psi\|_{L^2} &\leq \hat{C}(1+\|\nabla u\|_{L^{\infty}})( 1+\|\nabla \Psi\|_{L^2}+\|\nabla \rho\|_{L^4}^2+ \|\nabla^2 \rho\|_{L^2} )\nonumber\\[2mm]
    		&+ \hat{C}\| |\nabla \rho||\nabla^2u|\|_{L^2} + \hat{C}\| |\nabla \rho||\nabla F|\|_{L^2} + \hat{C}\|\nabla^2 F\|_{L^2}.
    	\end{align}

    		\|\nabla^2 \rho\|_{L^2}+\|\nabla \rho\|_{L^4}^2 & \leq \hat{C}\|\nabla \Psi\|_{L^2}+ \hat{C}\|\nabla \rho\|_{L^4}^2 \leq \hat{C}\|\nabla \Psi\|_{L^2}+ \hat{C} \|\nabla \rho\|_{L^{l}}^{\frac{l}{2}}\|\nabla^2 \rho\|_{L^2}^{\frac{4-l}{2}} \nonumber\\[2mm]
    		& \leq \hat{C}\|\nabla\Psi\|_{L^2} + \frac{1}{2}\|\nabla^2 \rho\|_{L^2}+\hat{C},
    	\end{align}
    	where we have chosen $l=\min\{4,q\}$ and used Poincar\'{e}'s inequality (Lemma \ref{lem-poincare}).

    	\begin{align}\label{p1-10}
    		\| |\nabla \rho||\nabla^2u|\|_{L^2} + \| |\nabla \rho||\nabla F|\|_{L^2} &\leq \hat{C}\|\nabla \rho\|_{L^q}\|\nabla^2 u\|_{L^2}^{1-\frac{2}{q}}\|\nabla^3 u\|_{L^2}^{\frac{2}{q}} + \hat{C}\|\nabla \rho\|_{L^q}\|\nabla F\|_{L^2}^{1-\frac{2}{q}}\|\nabla^2 F\|_{L^2}^{\frac{2}{q}}\nonumber\\[2mm]

    	\end{align}

    	\begin{align}\label{p1-11}
    		\|\nabla^3 u\|_{L^2} &\leq \hat{C}\|\nabla^2 \mbox{div} u\|_{L^2}+ \hat{C}\|\nabla^2 \omega\|_{L^2}\nonumber\\[2mm]
    		& \leq \hat{C}\|\nabla^2 F\|_{L^2} + \hat{C}\|\nabla^2 \rho\|_{L^2}+ \hat{C}\|\nabla \rho\|_{L^4}^2+ \hat{C}\||\nabla \rho||\nabla^2 u|\|_{L^2} \nonumber\\[2mm]
    		&+ \hat{C}( \|\nabla^2 \rho\|_{L^2} + \|\nabla \rho\|_{L^4}^2 )\|\nabla u\|_{L^{\infty}} + \hat{C}\|\nabla^2 \omega\|_{L^2}. 
    	\end{align}

    	\end{align}
    	which together with \eqref{p1-8} gives
    	\begin{equation}\label{p1-13}
    		\frac{d}{dt}\|\nabla \Psi\|_{L^2} \leq \hat{C}\|\nabla^2 F\|_{L^2} + \hat{C}\|\nabla^2 \omega\|_{L^2} + \hat{C}(1+\|\nabla \Psi\|_{L^2})(1+\|\nabla u\|_{L^{\infty}}).
    	\end{equation}
    	
    	This combined with \eqref{p1-7} and Gronwall's inequality yields
    	\begin{equation}\label{p1-14}
    		\sup_{0 \leq t \leq T} \|\nabla \Psi\|_{L^2} \leq \hat{C},
    	\end{equation}

    	\begin{equation}\label{p1-15}
    		\Delta \Phi = \rho - b(x),
    	\end{equation}

    	\begin{equation}\label{p1-16}

    	\end{equation}
