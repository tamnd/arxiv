<!-- theorem-1: body_md -->
\label{thrm:main}
Let $d$ be a positive integer and $q$ a prime power. Then
$$
\mathrm{H}(\mathrm{AGL}_d(q))=
\begin{cases}
d+1 & \textrm{when }(d,q)\ne(1,2),\\
1   & \textrm{when }(d,q)=(1,2).
\end{cases}
$$
Moreover,
$$
\mathrm{RC}(\mathrm{AGL}_d(q))=
\begin{cases}
d+2 & \textrm{when }q>2\textrm{ and }(d,q)\ne(1,3),\\
d+1 & \textrm{when }q=2\textrm{ and }d\textrm{ is odd},
      \textrm{ or when }(d,q)=(1,3),\\
d   & \textrm{when }q=2\textrm{ and }d\textrm{ is even},\\
                                 
\end{cases}
$$
Moreover, every subgroup of $\mathrm{AGL}_d(q)$ has relational complexity at most $d+2$ when $q>2$, and at most $d+1$ when $q=2$.

<!-- theorem-2: body_md -->
\label{thrm:main2}
Let
$G=V\rtimes G_0\le \mathrm{AGL}_d(q)$,
where $G_0$ is the stabilizer in $G$ of the zero vector and acts irreducibly on $V$. Suppose that
$\mathrm{RC}(G)=d+2$.
Then $q>2$ and one of the following holds:
\begin{enumerate}
\item\label{part1} $d=1$ and $|G_0|>2$,
\item\label{part2} $d\ge 2$ and there exists a divisor $f_1$ of $f$ such that
\begin{align*}
&\mathrm{SL}_d(p^{f_1})\unlhd G_0
\le \mathrm{GL}_d(p^{f_1})\mathbb{F}_q^\ast,\\
&\mathrm{SL}_d(p^{f_1})<G_0\cap \mathrm{GL}_d(p^{f_1}),
\end{align*}
where $\mathbb{F}_q^\ast$ is identified with the group of non-zero scalar matrices in $\mathrm{GL}_d(q)$.

\item\label{part3} $d= 2$, $p=3$, $f$ is even and
$G_0=\mathrm{SL}_2(5)Z$, where $\mathrm{SL}_2(5)$ is a maximal subgroup of $\mathrm{SL}_2(9)$ and $Z\le\mathbb{F}_q^\ast$ is a group of scalar matrices. Moreover, $G_0$ contains a scalar matrix of order $4$.
\end{enumerate}


The converse also holds, that is, if $G_0$ is as in parts~\eqref{part1},~\eqref{part2} or~\eqref{part3}, then $G=V\rtimes G_0$ has relational complexity $d+2$.

<!-- theorem-3: body_md -->
\label{thrm:main3}
Let
$G=V\rtimes G_0\le \mathrm{AGL}_d(2)$,
where $G_0$ is the stabilizer in $G$ of the zero vector and acts irreducibly on $V$. Suppose that
$\mathrm{RC}(G)=d+1$.
Then one of the following holds:
\begin{enumerate}
\item\label{part12} $d$ is odd and $G_0=\mathrm{SL}_d(2)$,
\item\label{part22} $d\geq 2$ is even and $G_0$ is the natural image of either $\mathrm{Alt}(d+1)$ or $\mathrm{Alt}(d+2)$ acting on the fully deleted permutation module,
\item\label{part32} $d\ge 6$, $d$ is even and $G_0$ is either $\Omega_d^+(2)$ or $\Omega_d^-(2)$,
\item\label{part42} $d=4$ and $G_0=\mathrm{Alt}(7)$.
\end{enumerate}

The converse also holds, that is, if $G_0$ is as in parts~\eqref{part12},~\eqref{part22},~\eqref{part32} or~\eqref{part42}, then $G=V\rtimes G_0$ has relational complexity $d+1$.

<!-- lemma-1: body_md -->
\label{l:0}
Let $\emptyset\ne \Lambda\subseteq V$. Then $\Lambda$ is an independent set for the action of $\mathrm{AGL}_d(q)$ if and only if $\Lambda$ is a frame for some affine subspace of $V$, with the single exception that, when $(d,q)=(1,2)$, the frame $\Lambda=V$ is not independent.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Suppose first that $\Lambda$ is a frame for an affine subspace $\pi$ of $V$ and that $\Lambda\ne V$ when $(d,q)=(1,2)$. Up to replacing $\Lambda$ with a suitable $\mathrm{AGL}_d(q)$-conjugate, we have
$$
\Lambda=\{0,e_1,\ldots,e_\ell\}
$$
for some $0\le\ell\le d$. A direct calculation shows that the pointwise stabilizer of $\Lambda$ is strictly contained in the pointwise stabilizer of each subset obtained by deleting one point from $\Lambda$. Thus $\Lambda$ is an independent set for the action of $\mathrm{AGL}_d(q)$ on $V$.

Conversely, suppose that $\Lambda$ is an independent set for the action of $\mathrm{AGL}_d(q)$ on $V$. Up to replacing $\Lambda$ with a suitable $\mathrm{AGL}_d(q)$-conjugate, we may suppose that $0\in\Lambda$. Thus $\Lambda=\{0,v_1,\ldots,v_\ell\}$,
for some $v_1,\ldots,v_\ell\in V\setminus\{0\}$. It follows from the independence of $\Lambda$ that $\{v_1,\ldots,v_\ell\}$ is an independent set for the action of $\mathrm{GL}_d(q)$ on $V$. Hence the vectors $v_1,\ldots,v_\ell$ are linearly independent. Therefore, $\Lambda$ is a frame for the affine subspace
$$
0+\langle v_1,\ldots,v_\ell\rangle.
$$
Finally, when $(d,q)=(1,2)$, the pointwise stabilizers of $V$ and of either singleton subset of $V$ are all trivial, so $V$ is not independent.

<!-- lemma-2: body_md -->
\label{l: almost independent}
Suppose that $m$ is a positive integer, that $I,J\in \Omega^{m+1}$, that $I\sim_{m} J$ but $I\not\sim_{m+1} J$. Then the underlying set of $I$ (and, therefore, of $J$) is an almost independent set.
<!-- end lemma-2 -->
<!-- lemma-3: body_md -->
\label{l:aiodd}
Suppose that $q>2$, and let $\Lambda$ be an almost independent subset of $V$ of cardinality at least $d+2$. Then $|\Lambda|=d+2$, and $\Lambda$ is $\mathrm{AGL}_d(q)$-conjugate to a set of the form
$$
\left\{0,e_1,\ldots,e_d,\sum_{i=1}^d\alpha_i e_i\right\},
$$
where $\alpha_i\ne 0$ for every $i$ and $\sum_{i=1}^d\alpha_i\ne 1$.
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
For every $\lambda\in\Lambda$, the set $\Lambda\setminus\{\lambda\}$ is independent. By Lemma~\ref{l:0}, it has cardinality at most $d+1$. Therefore,
$|\Lambda|\le d+2$,
and hence $|\Lambda|=d+2$.

Fix $\lambda\in\Lambda$. The set $\Lambda\setminus\{\lambda\}$ is a frame for $V$. After replacing $\Lambda$ by a suitable $\mathrm{AGL}_d(q)$-conjugate, we may assume that
$$
\Lambda\setminus\{\lambda\}=\{0,e_1,\ldots,e_d\}.
$$
Write 
$\lambda=\sum_{i=1}^d\alpha_i e_i$.
The result now follows from the observation preceding the lemma.

<!-- lemma-4: body_md -->
\label{l:ai}
Suppose that $q=2$ and $d\ge 2$, and let $\Lambda$ be an almost independent subset of $V$ of cardinality at least $d+1$. Then $\Lambda$ is $\mathrm{AGL}_d(2)$-conjugate to one of the following:
\begin{enumerate}
\item\label{l:ai1}$
\{0,e_1,\ldots,e_d\}$,
in which case $\Lambda$ is independent;

\item\label{l:ai2}
$\left\{0,e_1,\ldots,e_{d-1},
\sum_{i=1}^{d-1}e_i\right\}$,
where $d$ is odd;

\item\label{l:ai3}
$
\left\{0,e_1,\ldots,e_d,
\sum_{i=1}^d e_i\right\}$,
where $d$ is even.
\end{enumerate}
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
As in the proof of Lemma~\ref{l:aiodd}, we have
$|\Lambda|\le d+2$.

Suppose first that $|\Lambda|=d+1$. If $\Lambda$ is independent, then by Lemma~\ref{l:0} it is a frame for $V$ and is therefore conjugate to the set in~\eqref{l:ai1}.

Suppose that $\Lambda$ is not independent. Choose $\lambda\in\Lambda$. Since $\Lambda\setminus\{\lambda\}$ is an independent set of cardinality $d$, it is a frame for an affine subspace of dimension $d-1$. After replacing $\Lambda$ by a suitable conjugate, we may assume that
$$
\Lambda\setminus\{\lambda\}
=
\{0,e_1,\ldots,e_{d-1}\}.
$$
Since $\Lambda$ is not independent, $\lambda$ belongs to the affine span of these points. Thus
$$
\lambda=\sum_{i=1}^{d-1}\alpha_i e_i.
$$
The observation preceding Lemma~\ref{l:aiodd} shows that every $\alpha_i$ is non-zero and that
$$
\sum_{i=1}^{d-1}\alpha_i\ne 1.
$$
Since $q=2$, we have $\alpha_i=1$ for every $i$. The second condition is therefore equivalent to $d-1$ being even, or equivalently to $d$ being odd. This gives the set in~\eqref{l:ai2}.

Suppose now that $|\Lambda|=d+2$. As in the proof of Lemma~\ref{l:aiodd}, we may assume that
$$
\Lambda=
\left\{0,e_1,\ldots,e_d,
\sum_{i=1}^d\alpha_i e_i\right\}.
$$
Once again, every $\alpha_i$ is non-zero, and hence $\alpha_i=1$ for every $i$. Moreover,
$$
\sum_{i=1}^d\alpha_i\ne 1
$$
if and only if $d$ is even. This gives the set in~\eqref{l:ai3}.

<!-- lemma-5: body_md -->
\label{l:binarysubgroups}
Let $K\le \mathrm{AGL}_d(2)$. Then
$\mathrm{RC}(K)\le d+1$.
<!-- end lemma-5 -->
<!-- proof-4: body_md -->
The case $d=1$ is immediate, so assume $d\ge 2$.
Suppose that there exist $I$ and $J$ with $I\sim_{d+1}J$ and $I\not\sim_{d+2}J$. By Lemma~\ref{l: almost independent}, we may assume that the underlying sets $I'$ and $J'$ of $I$ and $J$ are almost independent sets of cardinality $d+2$. These sets are also almost independent sets for $\mathrm{AGL}_d(2)$ and hence, by Lemma~\ref{l:ai}, we have that $d$ is even and $I=(P_0,\ldots,P_d,P_{d+1})$ and $J=(Q_0,\ldots,Q_d,Q_{d+1})$, where $\{P_0,\ldots,P_d\}$ and $\{Q_0,\ldots,Q_d\}$ are frames and $P_{d+1}=\sum_iP_i$ and $Q_{d+1}=\sum_iQ_i$. Since $I\sim_{d+1}J$, we may assume $P_i=Q_i,$ for every $i\in \{0,\ldots,d\}$. Over the field with two elements, this implies $P_{d+1}=Q_{d+1}$ and $I=J$, contrary to the fact that $I\not\sim_{d+2}J$.
<!-- end proof-4 -->
<!-- proof-5: body_md -->
The formula for the height follows from Lemma~\ref{l:0}. Indeed, a frame of an affine subspace of dimension $\ell$ has cardinality $\ell+1$, and the largest affine subspaces of $V$ have dimension $d$. The exceptional case $(d,q)=(1,2)$ also follows from Lemma~\ref{l:0}.

It follows from~\eqref{eq:2.1} that
$\mathrm{RC}(\mathrm{AGL}_d(q))\le d+2$
when $(d,q)\ne(1,2)$. When $(d,q)=(1,2)$, we have
$\mathrm{AGL}_1(2)=\mathrm{Sym}(2)$,
and hence
$\mathrm{RC}(\mathrm{AGL}_1(2))=2$.
Similarly, when $(d,q)=(1,3)$, we have
$\mathrm{AGL}_1(3)=\mathrm{Sym}(3)$,
and hence 
$\mathrm{RC}(\mathrm{AGL}_1(3))=2$.

Assume now that $q>2$ and $(d,q)\ne(1,3)$. For
$a=(a_1,\ldots,a_d)\in(\mathbb{F}_q^\ast)^d$,
define
$$
T(a)=
\left(0,e_1,\ldots,e_d,\sum_{i=1}^d a_i e_i\right).
$$
Suppose that $a,b\in(\mathbb{F}_q^\ast)^d$ satisfy
\begin{equation}\label{eq:q+2}
\sum_{i=1}^d a_i\ne 1,
\qquad
\sum_{i=1}^d b_i\ne 1,
\qquad
a\ne b.
\end{equation}
We claim that
\[
T(a)\sim_{d+1}T(b) \textrm{  and  }
T(a)\not\sim_{d+2}T(b). \]

To prove the claim, observe first that, after deleting the last entry, the two tuples are identical. After deleting $e_i$, the remaining points form a frame for $V$ in each tuple because $a_i$ and $b_i$ are non-zero. After deleting $0$, the remaining points form a frame for $V$ because
$\sum_{i=1}^d a_i\ne 1$ and $\sum_{i=1}^d b_i\ne 1$.
Thus, since $\mathrm{AGL}_d(q)$ is transitive on ordered frames we deduce that $T(a)\sim_{d+1}T(b)$.
On the other hand, an element mapping $T(a)$ to $T(b)$ would fix
$0,e_1,\ldots,e_d$
and would therefore be the identity. This is impossible because $a\ne b$. Hence
$T(a)\not\sim_{d+2}T(b)$, as claimed. 

The claim implies that, so long as there exist $a,b$ satisfying~\eqref{eq:q+2}, we have $\mathrm{RC}(\mathrm{AGL}_d(q))=d+2$. It remains to show that such $a$ and $b$ exist.

Suppose first that $q>3$. Choose $\alpha\in\mathbb{F}_q^\ast$ such that
$d\alpha\ne 1$.
Then choose
$\lambda\notin
\{0,\alpha,1-(d-1)\alpha\}$.
Such choices are possible because $q>3$. We may now take
$a=(\alpha,\ldots,\alpha)$
and $b=(\lambda,\alpha,\ldots,\alpha)$.

It remains to consider $q=3$ and $d>1$. If $d\equiv2\pmod 3$, take
$a=(1,\ldots,1)$
and $b=(-1,1,\ldots,1)$.
If $d\equiv0\pmod 3$, take 
$a=(-1,\ldots,-1)$
and
$b=(1,-1,\ldots,-1)$.
Finally, if $d\equiv1\pmod 3$ and $d\ge4$, take
$a=(-1,\ldots,-1)$
and
$b=(1,1,-1,\ldots,-1)$.
In each case, all coordinates are non-zero, the coordinate sums are not equal to $1$, and $a\ne b$. Therefore,
$\mathrm{RC}(\mathrm{AGL}_d(q))=d+2$
whenever $q>2$ and $(d,q)\ne(1,3)$.

Assume now that $q=2$ and $d>1$.                     
                                 
                             
                
                                                                                                                                                                     
                                                                                                                                                                     
   
                                                  
   
                                                                                                                                                                                                                       
                
                                            
Suppose first that $d$ is odd and $d>1$. Consider the two $(d+1)$-tuples
\begin{align*}
I&=(0,e_1,\ldots,e_d),\\
J&=\left(0,e_1,\ldots,e_{d-1},
\sum_{i=1}^{d-1}e_i\right).
\end{align*}
                                                                                  
$I\not\sim_{d+1}J$.
We claim that 
$I\sim_d J$.
After deleting the last entry, the two tuples are identical. After deleting any $e_i$ with $1\le i\le d-1$, the remaining points form frames for affine subspaces of dimension $d-1$. Finally, after deleting $0$, the remaining points again form frames for affine subspaces of dimension $d-1$, because $d-1$ is even. Hence
$I\sim_d J$.
It follows that
$\mathrm{RC}(\mathrm{AGL}_d(2))=d+1$
when $d>1$ is odd by Lemma~\ref{l:binarysubgroups}.

Finally, suppose that $d$ is even. We first prove that
$\mathrm{RC}(\mathrm{AGL}_d(2))\ge d$.
When $d=2$, this is vacuous, so suppose that $d\ge4$. Consider the two $d$-tuples
\begin{align*}
I&=(0,e_1,\ldots,e_{d-1}),\\
J&=\left(0,e_1,\ldots,e_{d-2},
\sum_{i=1}^{d-2}e_i\right).
\end{align*}
The entries of $I$ form a frame for an affine subspace of dimension $d-1$, whereas the entries of $J$ do not. Thus
$I\not\sim_d J$.
The same deletion argument used in the odd-dimensional case shows that
$I\sim_{d-1}J$,
because $d-2$ is even. Therefore,
$\mathrm{RC}(\mathrm{AGL}_d(2))\ge d$.

It remains to prove the reverse inequality. Let $I,J\in V^{d+1}$ satisfy
$I\sim_d J$.
By Lemmas~\ref{l: almost independent} and~\ref{l:binarysubgroups}, it is enough to consider the case in which the sets of entries of $I$ and $J$ are almost independent and have cardinality $d+1$.
Since $d$ is even, Lemma~\ref{l:ai} shows that both sets of entries are frames for $V$. The regular action of $\mathrm{AGL}_d(2)$ on ordered frames therefore gives
 $I\sim_{d+1}J$.
Hence $
\mathrm{RC}(\mathrm{AGL}_d(2))\le d$.

We finish with the only assumption on $(d,q)$ being that it is not in $\{(1,2),(1,3)\}$.  Let $K$ be a subgroup of $\mathrm{AGL}_d(q)$. Then, from~\eqref{eq:2.1} and from the monotonicity of the height function, $\mathrm{RC}(K)\le \mathrm{H}(K)+1\le \mathrm{H}(\mathrm{AGL}_d(q))+1=d+2$. By Lemma~\ref{l:binarysubgroups}, the stronger inequality $\mathrm{RC}(K)\le d+1$ holds when $q=2$. This concludes the proof of Theorem~\ref{thrm:main}.

<!-- lemma-6: body_md -->
\label{l:irredN}
    Let $\mathbb{F}_{q_0}\leq \mathbb{F}_q$ be an inclusion of fields. Let $E\leq H$ be the group of all matrices of the form $I_d+w^\top\delta$ where $w$ ranges over $W:=\langle X\rangle_{\mathbb{F}_{q_0}}$ for some $X\subseteq V$ and $\delta$ is a fixed non-zero row vector. Suppose that the common fixed-point space $C_V(E)=\mathbb{F}_q\delta$, and that $H\geq E$ is irreducible on $V$. Then $N:=\langle E^g\mid g\in H\rangle$ is irreducible on $V$.
<!-- end lemma-6 -->
<!-- proof-6: body_md -->
Let $0\ne U\le V$ be an $N$-invariant subspace. In particular, $U$ is $E$-invariant. Suppose first that there exist $x\in U$ and $u\in W$ such that $xu^\top\ne 0$. Then
$$
x(I_d+u^\top\delta)-x=(xu^\top)\delta\in U,
$$
and hence
$$
\mathbb{F}_q\delta\le U.
$$
On the other hand, if $xu^\top=0$ for every $x\in U$ and every $u\in W$, then
$$
U\le {\bf C}_V(E)=\mathbb{F}_q\delta.
$$
Since $U\ne 0$, we again obtain $
\mathbb{F}_q\delta\le U$.
Since $U$ is $N$-invariant, it is invariant under $E^g$ for every $g\in H$. Applying the preceding argument to each $E^g$, we obtain
$\mathbb{F}_q\delta^g\le U,$
for every $g\in H$. Since $H$ acts irreducibly on $V$ and $\delta\ne 0$, we have
$$
V=\sum_{g\in H}\mathbb{F}_q\delta^g.
$$
Therefore, $U=V$, and $N$ is irreducible.

<!-- lemma-7: body_md -->
\label{l:d=1}
    Suppose that $d=1$. Then $\mathrm{RC}(G)=d+2=3$ if and only if $|G_0|>2$.
<!-- end lemma-7 -->
<!-- proof-7: body_md -->
Suppose first that $|G_0|>2$. Choose $
\gamma\in G_0\setminus\{1,-1\}$,
and set
$$
\alpha=\frac{1}{1+\gamma},
\qquad
\beta=\frac{\gamma}{1+\gamma}.
$$
Consider the triples
$I=(0,1,\alpha)$ and 
$J=(0,1,\beta)$.
The pairs obtained by deleting the third entries are identical. The map
$$
x\longmapsto \gamma x
$$
maps $(0,\alpha)$ to $(0,\beta)$, while the map
$$
x\longmapsto \gamma^{-1}x+1-\gamma^{-1}
$$
maps $(1,\alpha)$ to $(1,\beta)$. Since $\gamma,\gamma^{-1}\in G_0$, it follows that
$I\sim_2J$.
On the other hand, $\alpha\ne\beta$, because $\gamma\ne1$, and the only affine transformation fixing both $0$ and $1$ is the identity. Hence
$I\not\sim_3J$.
Therefore, $\mathrm{RC}(G)\ge3$. Since $\mathrm{H}(G)\le \mathrm{H}(\mathrm{AGL}_1(q))=2$ by Theorem~\ref{thrm:main},
equation~\eqref{eq:2.1} gives $\mathrm{RC}(G)\le3$. Thus $\mathrm{RC}(G)=3$.

Conversely, suppose that $|G_0|\le2$. If $|G_0|=1$ then $G$ is regular, hence $\mathrm{RC}(G)=2$. Suppose that $|G_0|=2$. Then $2\mid (q-1)$, so $q$ is odd and we have
$G_0=\{1,-1\}$.
Let $$I=(x_1,\ldots,x_n)\sim_2(y_1,\ldots,y_n)=J$$
be $2$-equivalent tuples. If all entries of $I$ are equal, then all entries of $J$ are equal, and a translation maps $I$ to $J$. Otherwise, choose $r,s$ such that $x_r\ne x_s$. After applying an element of $G$ to $J$, we may assume that
$y_r=x_r$ and $y_s=x_s$.
For every $k\leq n$, since $I\sim_2 J$ we deduce that there is some $(u,\ell)\in\mathbb{F}_q\rtimes\{1,-1\}$ such that $$(u+\ell x_r,u+\ell x_k)=(x_r,x_k)^{(u,\ell)}=(y_r,y_k)=(x_r,y_k).$$ In particular, $y_k-x_r=u+\ell x_k-u-\ell x_r=\ell(x_k-x_r)$, that is, for every $k\leq n$, $y_k-x_r=\pm(x_k-x_r)$ and similarly $y_k-x_s=\pm(x_k-x_s)$.
If $y_k\ne x_k$, the first equality gives $y_k=2x_r-x_k$,
while the second gives $y_k=2x_s-x_k$.
Since the characteristic is odd, this implies $x_r=x_s$, a contradiction. Therefore, $y_k=x_k$ for every $k$, and hence $I\sim_nJ$. Thus $\mathrm{RC}(G)=2$.

<!-- lemma-8: body_md -->
\label{l:Bidef}
    Let $i\in \{1,\dots,d\}$, and let $\delta$, $c_i$ and $B_i$ be defined as above. Then $$B_i=I_d+c_i\delta$$
<!-- end lemma-8 -->
<!-- proof-8: body_md -->
Since $(0,e_1,\dots,e_{i-1},e_{i+1},\dots,e_d,v)$ and $(0,e_1,\dots,e_{i-1},e_{i+1},\dots,e_d,w)$ are ordered frames, it suffices to show that $I_d+c_i\delta$ maps $v$ to $w$ and fixes $e_j$ for every $j\ne i$.

    Let $j\in\{1,\dots,d\}\setminus\{i\}$. Then $$e_j(I_d+c_i\delta)=e_j+\frac{1}{\alpha_i}e_je_i^\top \delta=e_j+0,$$ as desired. Moreover, $$v(I_d+c_i\delta)=v+\frac{1}{\alpha_i}\left(\sum_{j=1}^d\alpha_je_j\right)e_i^\top \delta=v+\delta=w,$$ as was to be shown.

<!-- lemma-10: body_md -->
The group $K$ contains an elementary abelian group $E=\{I_d+u\delta\mid u\in W\}$ of transvections of dimension $d-1$ over $\mathbb{F}_{q_0}$. Moreover, the common fixed-point space of $E$ on $V$ is $\mathbf{C}_V(E)=\mathbb{F}_q\delta$. Set $N:=\langle E^g\mid g\in G_0\rangle$. <!-- proof-10: body_md -->
Throughout this proof we will repeatedly use that $\lambda_j-1=\delta c_j$ for $j\in\{0,\dots, d\}$. This implies first that
\begin{equation}\label{eq:key}
\delta u_j=(\lambda_j-1)\delta c_r -(\lambda_r-1)\delta c_j=(\delta c_j)\delta c_r -(\delta c_r)\delta c_j=0.
\end{equation}
Moreover, for any $j\in\{1,\dots,d\}$, 
\[B_j^{-1}=(I_d+c_j\delta)^{-1}=I_d-\dfrac{c_j\delta}{\lambda_j}.\]
Therefore, for $j\in\{1,\ldots,d\}\setminus\{r\}$,
\[
(B_jB_r)^{-1}=\left(I_d-\dfrac{c_r\delta}{\lambda_r}\right)\left(I_d-\dfrac{c_j\delta}{\lambda_j}\right)=I_d-\frac{c_r\delta}{\lambda_r}-\frac{c_j\delta}{\lambda_j}+\frac{c_r\delta c_j\delta}{\lambda_r\lambda_j}= I_d-\frac{c_j\delta}{\lambda_j}-\frac{c_r\delta}{\lambda_r\lambda_j},\]
where the final equality is using (again) that $\delta c_j=\lambda_j-1$. Similarly, \[B_rB_j=(I_d+c_r\delta)(I_d+c_j\delta)=I_d+c_r\delta+c_j\delta+(\lambda_j-1)c_r\delta=I_d+\lambda_jc_r\delta+c_j\delta, 
\]
whence
\begin{align*}(B_jB_r)^{-1}B_rB_j&=\left(I_d-\frac{c_j\delta}{\lambda_j}-\frac{c_r\delta}{\lambda_r\lambda_j}\right)(I_d+\lambda_jc_r\delta+c_j\delta)\\&=I_d-\frac{c_j\delta}{\lambda_j}-\frac{c_r\delta}{\lambda_r\lambda_j}+\lambda_jc_r\delta-c_j\delta c_r\delta-\frac{c_r\delta c_r\delta}{\lambda_r}+c_j\delta-\frac{c_j\delta c_j\delta}{\lambda_j}-\frac{c_r\delta c_j\delta}{\lambda_r\lambda_j}\\&=
I_d-c_r\delta+\lambda_jc_r\delta-(\lambda_r-1)c_j\delta\\&=I_d+u_j\delta.\end{align*}

By~\eqref{eq:key},
$
(u_j\delta)^2=0,
$
thus, for $j\ne r$ the element $I_d+u_j\delta$ is a transvection in $K$.

The vectors $u_j$, with $j\in\{1,\ldots,d\}\setminus\{r\}$, are linearly independent over $\mathbb{F}_{q_0}$. Indeed, the vectors $c_1,\ldots,c_d$ form an $\mathbb{F}_q$-basis of the column-vector space, and, in any linear combination of the $u_j$, the coefficient of $c_j$ is $-(\lambda_r-1)$ times the coefficient of $u_j$.

If $u$ is a column vector satisfying $\delta u=0$, then 
$$
B_i^{-1}(I_d+u\delta)B_i=(I_d-\lambda_i^{-1}c_i\delta)(I_d+u\delta)(I_d+c_i\delta)=(I_d+u\delta-\lambda_i^{-1}c_i\delta)(I_d+c_i\delta)=I_d+\lambda_i u\delta.
$$
Also, if $\delta u=\delta u'=0$, then
$$
(I_d+u\delta)(I_d+u'\delta)=I_d+(u+u')\delta.
$$
It follows that $K$ contains $E$. Moreover, $\dim_{\mathbb{F}_{q_0}}(W)=d-1$, $|E|=q_0^{d-1}$,
and every nonidentity element of $E$ is a transvection in $K$, and hence in $G_0$: products add the parameters $u$, while conjugation by $B_i$
multiplies them by $\lambda_i$; thus the admissible scalars contain $\mathbb{F}_{q_0}$. We have now proved the first claim.

Observe next that
$$
\mathbb{F}_qW
=
\ker(\delta),
$$
where by abuse of notation we are identifying with $\delta$ the linear mapping $V^\top \to \mathbb{F}_q$ sending $u$ to $\delta u$. Given vectors $u,v\in V$, $v$ is fixed by $I+u^\top\delta$ if and only if $vu^\top=0$, whence the common fixed-point space of $E$ is (isomorphic to) the annihilator of $\mathbb{F}_q W=\ker(\delta)$, and hence has dimension 1 over $\mathbb{F}_q$. Therefore, 
$$
{\bf C}_V(E)=\mathbb{F}_q\delta.
$$

Clearly $N=\langle E^g\mid g\in G_0\rangle\unlhd G_0$ and $N\le \mathrm{SL}_d(q)$---that $N$ is irreducible follows from Lemma~\ref{l:irredN}.

<!-- lemma-20: body_md -->
\label{l:non-singular-basis}
Let $(V,Q)$ be a nondegenerate orthogonal space of even dimension $d$
over $\mathbb F_2$. Suppose either that $V$ has minus type, or that
$V$ has plus type and $d\geq6$. Then $V$ has a basis
$f_1,\ldots,f_d$ such that
$Q(f_i)=1$, for every $i\in\{1,\ldots,d\}$ and
$Q(f_1+\cdots+f_d)=1$.
<!-- end lemma-20 -->
<!-- proof-20: body_md -->
Let $p$ denote the associated polar form. We first describe a four-dimensional plus-type space which will be
used to increase the dimension by $4$. Let
$P=\langle \alpha,\beta\rangle\perp\langle \gamma,\delta\rangle$,
where $(\alpha,\beta)$ and $(\gamma,\delta)$ are hyperbolic pairs. Thus
\[
Q(\alpha)=Q(\beta)=Q(\gamma)=Q(\delta)=0,
\qquad
p(\alpha,\beta)=p(\gamma,\delta)=1,
\]
and all the other pairings among these vectors are zero. Set
\[
z_1=\gamma+\delta,\qquad
z_2=\beta+\gamma+\delta,\qquad
z_3=\alpha+\gamma+\delta,\qquad
z_4=\alpha+\beta+\delta.
\]
Then $z_1,z_2,z_3,z_4$ form a basis of $P$ and
\[
Q(z_i)=1
\]
for every $i$. Moreover,
\[
z_1+z_2+z_3+z_4=\gamma,
\]
and hence
\[
Q(z_1+z_2+z_3+z_4)=0.
\]

Consequently, suppose that a nondegenerate orthogonal space $W$ has a
basis $x_1,\ldots,x_n$ satisfying
\[
Q(x_i)=1
\quad\text{and}\quad
Q(x_1+\cdots+x_n)=1.
\]
Then
\[
x_1,\ldots,x_n,z_1,z_2,z_3,z_4
\]
is a basis of $W\perp P$ consisting of non-singular vectors, and
\[
Q\left(
x_1+\cdots+x_n+z_1+\cdots+z_4
\right)
=
Q(x_1+\cdots+x_n)+Q(\gamma)=1.
\]
Since $P$ has plus type, $W\perp P$ has the same type as $W$.

It therefore suffices to give the required bases in dimensions $2$
and $4$ for the minus type, and in dimensions $6$ and $8$ for the
plus type.

For the two-dimensional minus-type space, let $\alpha,\beta$ be an anisotropic
basis, so that
\[
Q(\alpha)=Q(\beta)=Q(\alpha+\beta)=1.
\]
Then $\alpha,\beta$ is the required basis.

For the four-dimensional minus-type space, write
\[
V=\langle \alpha,\beta\rangle\perp\langle \gamma,\delta\rangle,
\]
where $\langle \alpha,\beta\rangle$ is anisotropic and $(\gamma,\delta)$ is a hyperbolic
pair. Then
\[
\gamma+\delta,\qquad \beta,\qquad \beta+\delta,\qquad \alpha
\]
is a basis consisting of non-singular vectors, and its sum is $\alpha+\gamma$,
which is non-singular.

For the six-dimensional plus-type space, write
\[
V=
\langle \alpha,\beta\rangle
\perp
\langle \gamma,\delta\rangle
\perp
\langle \varepsilon,\zeta\rangle,
\]
where the three displayed pairs are hyperbolic. The vectors
\[
\begin{split}
&\varepsilon+\zeta,\qquad \delta+\varepsilon+\zeta,\qquad \gamma+\varepsilon+\zeta,\\
&\gamma+\delta+\zeta,\qquad \beta+\varepsilon+\zeta,\qquad \alpha+\varepsilon+\zeta
\end{split}
\]
form a basis of $V$ and are all non-singular. Their sum is
\[
\alpha+\beta+\varepsilon,
\]
which is non-singular.

Finally, for the eight-dimensional plus-type space, write
\[
V=
\langle \alpha,\beta\rangle
\perp
\langle \gamma,\delta\rangle
\perp
\langle \varepsilon,\zeta\rangle
\perp
\langle \eta,\mu\rangle,
\]
where all four displayed pairs are hyperbolic. The vectors
\[
\begin{split}
&\eta+\mu,\qquad \zeta+\eta+\mu,\qquad \varepsilon+\eta+\mu,\qquad \varepsilon+\zeta+\mu,\\
&\delta+\eta+\mu,\qquad \gamma+\eta+\mu,\qquad \beta+\eta+\mu,\qquad \alpha+\varepsilon+\zeta
\end{split}
\]
form a basis and are all non-singular. Their sum is
\[
\alpha+\beta+\gamma+\delta+\varepsilon+\zeta+\mu,
\]
which is non-singular.

The result now follows inductively by taking an orthogonal direct sum
with the four-dimensional space $P$. Indeed, the base dimensions
$2$ and $4$ cover all even dimensions of minus type, while the base
dimensions $6$ and $8$ cover all even dimensions at least $6$ of
plus type.

\begin{equation}\label{eq:T-decomposition-complete}
V=V_1\oplus\cdots\oplus V_m
\end{equation}
for some $m>1$, where the $V_k$ are irreducible $T$-modules of the same

Every $B_i$ normalizes $T$ and therefore permutes the summands in
\eqref{eq:T-decomposition-complete}. If $n\geq3$ and $B_i$ maps

$(B_i-I_d)|_{V_k}$ is injective, so
\[
\mathrm{rank}(B_i-I_d)\geq n>2,
\]
a contradiction. Hence every $B_i$ stabilizes every $V_k$ when
$n\geq3$.

Suppose that $n=2$. A non-trivial cycle of length $\ell$ on the blocks
contributes at least $2(\ell-1)$ to the rank of $B_i-I_d$. Thus the permutation
induced by a matrix $B_i$ on the summands is either trivial or a single transposition.                                   
                                                     
                                    
               
                                
                                                                
                             
Assume that some $B_r$ interchanges two summands; without loss of generality assume $B_r$ interchanges $V_1$ and $V_2$.

image. Since $\mathrm{rank}(B_r-I_d)=2$, this is the whole moved space.
Every non-zero vector in this image has non-zero components in both
$V_1$ and $V_2$. Thus Lemma~\ref{l:ainot0} implies that
$\delta$ has non-zero components in precisely $V_1$ and $V_2$.

$V_2$. On the other hand, if $B_i$ stabilizes every summand, then the restrictions of $B_i-I_d$ to both $V_1$ and
$V_2$ are non-zero since $\delta\in\mathrm{Im}(B_i-I_d)$ and the $V_1$- and $V_2$-components
of $\delta$ are non-zero. Since $\mathrm{rank}(B_i-I_d)=2$, it follows that $B_i$ fixes every other
summand pointwise, and similarly for a matrix $B_i$ interchanging $V_1$ and $V_2$. Hence every summand different from $V_1$ and $V_2$ is contained in the intersection in~\eqref{eq:intersection-Bi-zero-complete}. It follows that $m=2$ and $d=4$, a contradiction to the assumption that $d\geq 9$.

\eqref{eq:T-decomposition-complete} and that $n\geq 2$. The fixed spaces in
\eqref{eq:rank-two-fixed-i-complete} and
\eqref{eq:rank-two-fixed-zero-complete} are then decomposable with

$j\in\{1,\ldots,d-1\}$,
\[
\langle e_j\rangle
=
\bigcap_{\substack{1\leq i\leq d-1\\i\neq j}}
{\bf C}_V(B_i),
\]
so every line $\langle e_j\rangle$ is decomposable and is contained in
one summand.

$U_k$ is non-zero. Since the coordinate lines span $U$ and each of
them belongs to one summand, every $U_k$ contains some $e_j$.

\[
\varphi:U\longrightarrow\mathbb F_2,
\qquad
\varphi\left(\sum_{j=1}^{d-1}x_je_j\right)
=
\sum_{j=1}^{d-1}x_j.
\]
Then ${\bf C}_V(B_0)=\ker(\varphi)$. This is a decomposable hyperplane
of the decomposable space
$U=U_1\oplus\cdots\oplus U_m$. Therefore it contains $U_k$ for all
but at most one value of $k$. On the other hand, every $U_k$ contains an
$e_j$, and $\varphi(e_j)=1$. Thus no $U_k$ is contained in
$\ker(\varphi)$, a contradiction.

Therefore $T$ is irreducible. Finally, applying McLaughlin's classification, and using the exclusions already proved at the end of Section~\ref{sec:subsub4}, gives $T=\mathrm{SL}_d(2)=\mathrm{GL}_d(2)$.
Thus $G_0=\mathrm{GL}_d(2)$, contrary to
\eqref{eq:C-not-G0-complete}, hence the result.

\label{l:N-absolute-complete}
The group $N$ is absolutely irreducible on $V$.
<!-- end lemma-31 -->
<!-- proof-33: body_md -->
First we prove irreducibility. Since $N\unlhd G_0$, the $N$-socle of
$V$ is $G_0$-invariant. Hence $V$ is completely reducible as an
$N$-module. Suppose that
\[
V=V_1\oplus\cdots\oplus V_m
\]
is a decomposition into irreducible $N$-modules with $m>1$.
The fixed-point space ${\bf C}_V(N)$ is $G_0$-invariant. Since
$N\neq1$ and $G_0$ is irreducible, ${\bf C}_V(N)=0$. Therefore no
$V_k$ is a trivial $N$-module; in particular,
$\dim V_k\geq2$, because $\mathrm{GL}_1(2)=1$.

Every $B_i$ belongs to $N$ and hence stabilizes every $V_k$. Therefore
all the fixed spaces in
\eqref{eq:rank-two-fixed-i-complete} and
\eqref{eq:rank-two-fixed-zero-complete} are decomposable. The
fixed-space argument used above now applies without any permutation of
summands: $U={\bf C}_V(B_1)+{\bf C}_V(B_2)$ is decomposable, every
coordinate line $\langle e_j\rangle$ is decomposable, every non-zero
space $U\cap V_k$ contains some $e_j$, and the decomposable hyperplane
${\bf C}_V(B_0)=\ker(\varphi)$ cannot contain all but one of the spaces
$U\cap V_k$. This contradiction proves that $N$ is irreducible.

Let $E=\mathrm{End}_N(V)$.
By Schur's lemma, $E$ is a finite division ring, and hence
$E\cong\mathbb F_{2^s}$ for some $s\geq1$. Every
${\bf C}_V(B_i)$ is an $E$-subspace, because every element of $E$
commutes with $B_i$. Since $U={\bf C}_V(B_1)+{\bf C}_V(B_2)$,
the space $U$ is also an $E$-subspace. It follows that
$s$ divides $\dim_{\mathbb F_2}V=d$ and $s$ divides $\dim_{\mathbb F_2}U=d-1$.
Thus $s=1$, which is equivalent to absolute
irreducibility.

<!-- lemma-32: body_md -->
\label{l:N-primitive-complete}
The group $N$ is primitive on $\overline V$.
<!-- end lemma-32 -->
<!-- proof-34: body_md -->
Suppose that $\overline V=W_1\oplus\cdots\oplus W_m$
is a non-trivial system of imprimitivity for $N$, and put
$r=\dim_k W_j$.

Assume first that $r\geq3$. If some $B_i$ maps $W_j$ to a distinct
block, then $(B_i-I_d)|_{W_j}$ is injective, and hence
$\mathrm{rank}(B_i-I_d)\geq r>2$. Thus every $B_i$ stabilizes every
block. Their fixed spaces are therefore decomposable, and the
fixed-space argument in the proof of
Lemma~\ref{l:N-absolute-complete} -- now over $k$ -- gives a contradiction.

Suppose that $r=2$. A non-trivial cycle of length $\ell$ on the blocks
contributes at least $2(\ell-1)$ to the rank of $B_i-I_d$. Thus the permutation
induced by a matrix $B_i$ on the blocks is either trivial or a single transposition.
If every $B_i$ stabilizes every block, the same fixed-space argument
again gives a contradiction. Hence some $B_r$ interchanges two blocks; without loss of generality assume that $B_r$ interchanges $W_1$ and $W_2$. The restriction
$(B_r-I_d)|_{W_1}$ is injective and has dimension $2$, so its image is the
whole moved space of $B_r$. Every non-zero vector in that image has
non-zero components in both $W_1$ and $W_2$. Since
$\delta\in\mathrm{Im}(B_r-I_d)$, the vector $\delta$ has non-zero
components in precisely these two blocks.

The same reasoning shows that every $B_i$ which interchanges blocks
must interchange $W_1$ and $W_2$. If $B_i$ stabilizes
every block, then
$$
\mathrm{Im}(B_i-I_d)
=
\bigoplus_W
\mathrm{Im}\bigl((B_i-I_d)|_W\bigr).
$$
Since $\delta\in\mathrm{Im}(B_i-I_d)$ has non-zero components in
both $W_1$ and $W_2$, the restrictions of $B_i-I_d$ to $W_1$ and to $W_2$ are
both non-zero. Each of them therefore contributes at least $1$ to $\mathrm{rank}(B_i-I_d)$. Since $
\mathrm{rank}(B_i-I_d)=2$, we deduce that $(B_i-I_d)|_W=0$,
for every block $W_j$ with $j>2$. Thus $B_i$ fixes every such block
pointwise. Similarly, a matrix $B_i$ interchanging $W_1$ and $W_2$ also fixes every other block pointwise. Therefore
every other block lies in
$\bigcap_i{\bf C}_{\overline V}(B_i)=0$. Hence $m=2$ and $d=4$, which contradicts our assumption $d>8$.

It remains to consider $r=1$. Let $\Omega$ be the set of the
one-dimensional blocks, let
\[
\pi:N\longrightarrow P\leq\mathrm{Sym}(\Omega)
\]
be the induced permutation action, and set $D=\ker(\pi)$. The group
$P$ is primitive: a non-trivial block system for $P$ would give a
system of imprimitivity for $N$ with blocks of dimension greater than
$1$, which has already been excluded.

Let $x$ be one of the generating conjugates of a matrix $B_i$. If
$|x|=2$, then each transposition in the permutation induced by $x$
contributes 1 to $\mathrm{rank}(x-I_d)$, while a fixed
one-dimensional block contributes nothing. Hence $\pi(x)$ is a
product of two disjoint transpositions. If $|x|=3$, then either
$\pi(x)$ is a $3$-cycle, or $\pi(x)=1$ and $x$ is diagonal. It follows
that the non-trivial group $P$ (if $P=1$, then every one-dimensional block $W_\omega$ is $N$-invariant, contradicting the irreducibility of $N$ on 
$\overline{V}$) contains either a $3$-cycle or a double
transposition. By the classical Jordan Theorems for primitive
permutation groups, and since $d>8$,
\[
\mathrm{Alt}(\Omega)\leq P;
\]
see, for example, \cite[Section~3.3]{DixonMortimer}. Thus
$P=\mathrm{Alt}(\Omega)$ or $P=\mathrm{Sym}(\Omega)$.

Thus
$$\overline V=\bigoplus_{\omega\in\Omega}W_\omega,
\qquad
\dim_k(W_\omega)=1,$$
where $W_\omega$ denotes the one-dimensional block indexed by
$\omega$. Since $D$ is the kernel of the action on $\Omega$, every
element of $D$ stabilizes each line $W_\omega$. Consequently, for
every $\omega\in\Omega$ and every $x\in D$, there is a unique scalar
$\chi_\omega(x)\in k^\ast$ such that
\[
wx=\chi_\omega(x)w
\]
for every $w\in W_\omega$. The map
\[
\chi_\omega:D\longrightarrow k^\ast
\]
is a linear character, describing the action of $D$ on the
one-dimensional space $W_\omega$.

Suppose first that $D\neq1$. Equality of two such characters is a $P$-invariant
equivalence relation on $\Omega$. Since $P$ is primitive, either all
the characters are equal or they are pairwise distinct. In the first
case $D$ consists of scalar matrices. But every element of $D$ belongs
to $\mathrm{GL}_d(2)$, so the only scalar it can contain is the
identity, contrary to $D\neq1$. Hence the characters are pairwise
distinct, and the spaces $W_\omega$ are precisely the common
eigenspaces of $D$.

The Frobenius map
\[
\sigma:\overline V\longrightarrow\overline V,\qquad
(x_1,\ldots,x_d)\longmapsto(x_1^2,\ldots,x_d^2),
\]
commutes with $N$, and therefore permutes the common eigenspaces
$W_\omega$. The induced permutation centralizes $P$. Since
$P\geq\mathrm{Alt}(\Omega)$ and $|\Omega|=d>3$, its centralizer in
$\mathrm{Sym}(\Omega)$ is trivial. Thus every $W_\omega$ is
Frobenius-stable, and hence is defined over $\mathbb F_2$. We obtain
a decomposition of $V$ into one-dimensional $\mathbb F_2$-spaces
permuted by $N$. Since $\mathrm{GL}_1(2)=1$, the group acts by
permutation matrices and fixes the sum of a basis vector from each
line. This contradicts the irreducibility of $N$ on $V$.

Finally, suppose that $D=1$. Then $N\cong P$. The stabilizer of a line
$W_\omega$ acts on it through a linear character. If
$P=\mathrm{Alt}(d)$, this stabilizer is $\mathrm{Alt}(d-1)$ and is
perfect. If $P=\mathrm{Sym}(d)$, its abelianization has order $2$, but
$k^\ast$ has no element of order $2$. In either case the character
is trivial. Hence $\overline V$ is the ordinary permutation module of
$P$ on $\Omega$, which has a non-zero fixed vector. This again
contradicts irreducibility.

\label{l:N-tensor-complete}
The group $N$ is tensor-indecomposable on $\overline V$.
<!-- end lemma-33 -->
<!-- proof-35: body_md -->
We use Lemma~\ref{l:symmetric-pairing-complete}. 
Suppose that $N$ normalizes a non-trivial tensor decomposition
\begin{equation}\label{eq:tensor-decomposition-complete}
\overline V=V_1\otimes\cdots\otimes V_t,
\qquad
\dim V_j\geq2.
\end{equation}

We first show that no generating bireflection permutes the tensor
factors non-trivially. Let $x$ be such a generator and suppose first that it permutes the tensor factors non-trivially. If $|x|=2$ then it swaps two $r$-dimensional factors; without loss of generality assume that these are $V_1$ and $V_2$. Observe that the product of the dimensions
of the remaining factors is $d/r^2$. After changing bases in $V_1$ and $V_2$ and absorbing a scalar into the remaining tensor factor, the
action has the form $\tau\otimes y$, where $\tau$ is the swap on
$k^r\otimes k^r$ and $y$ is an involution on a $(d/r^2)$-dimensional
space. 

Abusing notation we identify $V_1$ and $V_2$ and choose a basis $\{e_1,\dots, e_r\}$ such that the swap $\tau$ satisfies $\tau(e_i\otimes e_j) = e_j\otimes e_i$ for all $i,j$. When $i\neq j$, this yields a Jordan block, $J_2$, on the space $\langle e_i\otimes e_j, e_j\otimes e_i\rangle$ corresponding to the matrix $\begin{pmatrix} & 1 \\ 1 & \end{pmatrix}$. Thus $\tau$ has
$r(r-1)/2$ Jordan blocks of size $2$. Since the rank of $J_2-I_2$ is equal to $1$, we  see that
\[
\mathrm{rank}(x-I_d)\geq \frac{d}{r^2}\cdot\frac{r(r-1)}2=\frac{d(1-\frac{1}{r})}{2}\geq d/4>2,\]
a contradiction.

If $|x|=3$ then $x$ induces a $3$-cycle
of $r$-dimensional factors; the cyclic permutation on
$(k^r)^{\otimes3}$ has eigenspace dimensions
\[
\frac{r^3+2r}{3},\qquad
\frac{r^3-r}{3},\qquad
\frac{r^3-r}{3}.
\]
Tensoring with any semisimple element of order dividing $3$ on the
remaining $\left(\frac{d}{r^3}\right)$-dimensional tensor factor cannot give a $1$-eigenspace
larger than $(d/r^3)(r^3+2r)/3$. Hence
\[
\mathrm{rank}(x-I_d)
\geq
\frac{d}{r^3}\cdot\frac{2(r^3-r)}3>2.
\]
Thus every generating bireflection stabilizes every tensor factor, and so we may absorb some factors and assume without loss of generality that $N$ normalizes a decomposition $V=V_1\otimes V_2$.

We next use the following elementary two-factor calculation. Let
$V_1$ and $V_2$ have dimensions $r,s\geq2$, and let
$x=a\otimes b$ have order $2$ or $3$. If both $a$ and $b$ are
projectively non-scalar, then
\begin{equation}\label{eq:tensor-rank-bound-complete}
\mathrm{rank}(x-I)\geq r+s-2.
\end{equation}
Indeed, for order $2$, after rescaling the two factors we may write
$a=I+P$ and $b=I+Q$, where $P^2=Q^2=0$. If
$p=\mathrm{rank}(P)$ and $q=\mathrm{rank}(Q)$, direct calculation gives
\[
\mathrm{rank}(a\otimes b-I)
=
ps+rq-2pq
\geq r+s-2.
\]
For order $3$, the two factors are semisimple. If their three
eigenvalue multiplicities are
$(r_0,r_1,r_2)$ and $(s_0,s_1,s_2)$, respectively, each triple has at
least two non-zero entries. The multiplicity of any prescribed
eigenvalue of $a\otimes b$ is at most
\[
(r-1)(s-1)+1.
\]
This proves~\eqref{eq:tensor-rank-bound-complete}.
Since $rs=d\geq 9$ it follows that $r+s-2\geq 4$, hence each generating bireflection is
projectively non-scalar on exactly one factor, without loss of generality suppose that $x$ is projectively non-scalar on $V_1$. After absorbing scalars,
$x=c\otimes I_s$, and hence
\[
2=\mathrm{rank}(x-I_d)
=
s\,\mathrm{rank}(c-I_r).
\]
Since the tensor decomposition is non-trivial, $s\geq2$, and therefore
$s=2$ and $r=d/2$. Since $d/2>2$, any generating bireflection which is projectively non-scalar on exactly one factor must act by scalar on $V_2$. Therefore, $N$ stabilizes every subspace
$V_1\otimes L$, where $L$ is a one-dimensional subspace of the
two-dimensional factor. This contradicts irreducibility.
