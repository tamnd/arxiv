<!-- head-1: body_md -->
We write $[n]=\{1,\ldots,n\}$ and use $\1\{A\}$ for the indicator of an event $A$.
The Euclidean inner product and norm are denoted by $\langle x,y\rangle$ and $\left\lVertx\right\rVert_2$,
and $B_2^d=\{x\in\mathbb{R}^d:\left\lVertx\right\rVert_2\le1\}$ is the Euclidean unit ball.
The notation $\widetilde O$ and $\widetilde\Omega$ suppresses polylogarithmic factors.
<!-- end head-1 -->
<!-- definition-1: body_md -->
\label{def:exp-concavity}
For $\alpha>0$, a function $f:\mathcal{K}\to\mathbb{R}$ is \emph{$\alpha$-exp-concave} if
$x\mapsto e^{-\alpha f(x)}$ is concave on $\mathcal{K}$.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:strong-convexity}
For $\lambda>0$, a differentiable function $f:\mathcal{K}\to\mathbb{R}$
is \emph{$\lambda$-strongly convex} if, for every $x,y\in \mathcal{K}$,
\[
f(y)\ge
f(x)+\langle\nabla f(x),y-x\rangle
+\frac{\lambda}{2}\left\lVerty-x\right\rVert_2^2.
\]
<!-- end definition-2 -->
<!-- definition-3: body_md -->
\label{def:comparator-paths}
A \emph{comparator path} is a sequence $u_{1:T}=(u_1,\ldots,u_T)\in \mathcal{K}^T$.
Its number of switches and number of distinct points are, respectively,
\[
k(u_{1:T})
:=
\sum_{t=2}^T \1\{u_t\ne u_{t-1}\},
\qquad
m(u_{1:T})
:=
|\{u_1,\ldots,u_T\}|.
\]
For integers $1\le m\le T$ and $0\le k\le T-1$, define the comparator class
\[
\mathcal U_{m,k}(\mathcal{K})
:=
\left\{
u_{1:T}\in \mathcal{K}^T:
m(u_{1:T})\le m,\quad
k(u_{1:T})\le k
\right\}.
\]
Thus $\mathcal U_{m,k}(\mathcal{K})$ consists of paths that switch
at most $k$ times among at most $m$ distinct points.
<!-- end definition-3 -->
<!-- definition-4: body_md -->
\label{def:regret}
For an algorithm $\mathcal A$ that outputs decisions
$x_{1:T}\in\mathcal{K}^T$ and faces losses $f_{1:T}$,
the tracking regret with at most $k$ switches and $m$ distinct points is
\[
\mathsf{Reg}_T(\mathcal A;m,k)
:=
\sum_{t=1}^T f_t(x_t)
-
\min_{u_{1:T}\in\mathcal U_{m,k}(\mathcal{K})}
\sum_{t=1}^T f_t(u_t).
\]
<!-- end definition-4 -->
<!-- definition-5: body_md -->
\label{def:adaptive-regret}
For an algorithm $\mathcal{A}$ that outputs decisions $x_{1:T}\in\mathcal{K}^T$ and faces losses $f_{1:T}$, the adaptive regret is
\[
    \mathsf{AReg}_T(\mathcal A)
    :=
    \max_{1\le r\le s\le T}
    \left\{
        \sum_{t=r}^s f_t(x_t)
        -\min_{u\in\mathcal{K}}\sum_{t=r}^s f_t(u)
    \right\}.
\]
<!-- end definition-5 -->
<!-- theorem-1: body_md -->
\label{thm:static-to-adaptive}
Suppose the losses are $\alpha$-exp-concave on $\mathcal{K}$.
Given a deterministic base learner $\mathcal B$ with nonnegative,
nondecreasing static-regret bound $R(T)$, pruned FLH produces an
algorithm $\mathcal A$ that maintains $O(\log T)$ active copies
and satisfies
\[
    \mathsf{AReg}_T(\mathcal A)
    \le O\!\left(
        R(T)\log T+\alpha^{-1}\log^2 T
    \right).
\]
The construction preserves oracle efficiency.
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{cor:static-to-switching}
Under the assumptions of Theorem~\ref{thm:static-to-adaptive},
the algorithm $\mathcal A$ satisfies, for every $1\le m\le T$
and $0\le k\le T-1$,
\[
    \mathsf{Reg}_T(\mathcal A;m,k)
    \le (k+1)\mathsf{AReg}_T(\mathcal A)
    \le O\!\left((k+1)\cdot \left[
        R(T)\log T+\alpha^{-1}\log^2 T
    \right]\right).
\]
The bound is independent of $m$, and the algorithm requires
neither $m$ nor $k$ nor the switch times.
<!-- end corollary-1 -->
<!-- theorem-2: body_md -->
\label{prop:strong}
Suppose every $f_t$ is twice differentiable, $\lambda$-strongly convex,
and $G$-Lipschitz on $\mathcal{K}$.
There is a deterministic oracle-efficient algorithm $\mathcal A$ such that,
for every $1\le m\le T$ and $1\le k\le T-1$,
\[
    \mathsf{Reg}_T(\mathcal A;m,k)
    \le O\!\left(\frac{G^2}{\lambda}k\log^2 T\right).
\]
<!-- end theorem-2 -->
<!-- proof-1: body_md -->
\citet{hazan2007logarithmic} show that projected online gradient descent, restarted with step sizes
$\eta_t=1/(\lambda t)$ in its local time, has static regret
\[
    R_{\mathrm{OGD}}(n)
    \le O\!\left(\frac{G^2}{\lambda}\log n\right)\,.
\]
Moreover, strong convexity and the gradient bound give
\[
    \nabla^2 f_t(x)\succeq\lambda I
    \succeq\frac{\lambda}{G^2}
        \nabla f_t(x)\nabla f_t(x)^\top,
\]
where for the RHS we used the Cauchy-Schwarz inequality, so each loss is $(\lambda/G^2)$-exp-concave. Applying
Corollary~\ref{cor:static-to-switching} with OGD as the base learner
and $\alpha=\lambda/G^2$ gives the desired algorithm $\mathcal A$
and proves the bound for every $m$ and $k$.
<!-- end proof-1 -->
<!-- theorem-3: body_md -->
\label{thm:ons-switching-upper}
Suppose $\mathcal{K}$ has Euclidean diameter at most $D$, and every loss $f_t$
is $\alpha$-exp-concave and $G$-Lipschitz on $\mathcal{K}$.
There is a deterministic oracle-efficient algorithm $\mathcal A$ such that,
for every $1\le m\le T$ and $1\le k\le T-1$,
\[
    \mathsf{Reg}_T(\mathcal A;m,k)
    \le O\!\left(d\left(\frac{1}{\alpha}+GD\right)
        k\log^2 T\right).
\]

<!-- proof-3: body_md -->
Only rounds with $(i,j,\ell)\in C_1\times C_2\times C_3$
contribute to $\Delta$. On every other round, the comparator
predicts zero on the queried coordinate, so
$f_t(u_t)=f_t(0)=1/8$ and the gain is zero.
On a contributing round, the comparator predicts $\gamma$.
Splitting according to whether the fourth vertex lies in the
clique, we obtain
\[
\begin{aligned}
    f_{ij\ell v}(0)-f_{ij\ell v}(u^{(1)})
    &=\frac18\left(Y_{ij\ell v}^2
                  -(\gamma-Y_{ij\ell v})^2\right)=\begin{cases}
        \displaystyle\frac{\gamma}{4}-\frac{\gamma^2}{8},
            &v\in C_4,\\[4pt]
        \displaystyle\frac{\gamma Y_{ij\ell v}}4-\frac{\gamma^2}{8},
            &v\in V_4\setminus C_4.
      \end{cases}
\end{aligned}
\]
Conditional on $C$, the first case is deterministic, while the
labels in the second case are independent unbiased signs.
The quadratic penalty is paid in both cases. Collecting all
deterministic terms therefore gives $\Delta = \mu_C + Z_C$ where
\[
\begin{aligned}
    \mu_C:=\mathbb E[\Delta\mid C]
    &=q_1q_2q_3\left(\frac{\gamma q_4}{4}
                    -\frac{\gamma^2|V_4|}{8}\right), \qquad Z_C =\frac{\gamma}{4}
        \sum_{\substack{i\in C_1,\;j\in C_2,\;\ell\in C_3\\
                        v\in V_4\setminus C_4}}
        Y_{ij\ell v}.
\end{aligned}
\]

With probability $1-o_N(1)$, the clique intersections concentrate
around their means $q|V_s|/N$: in particular,
$q_1,q_2,q_4=\Theta(q)$, $q_3=\Theta(q^3/N)$, and
$q_4\ge q|V_4|/(2N)=\gamma|V_4|$.
Condition on such a clique. Since $\gamma=q/(2N)$ and
$|V_4|=\Theta(N)$, the deterministic contribution satisfies
\[
    \mu_C\ge q_1q_2q_3\frac{\gamma^2|V_4|}{8}
    =\Omega\!\left(q^2\cdot\frac{q^3}{N}\cdot\frac{q^2}{N}\right)
    =\Omega\!\left(\frac{q^7}{N^2}\right).
\]
Thus the positive contribution from forced hyperedges outweighs
the quadratic penalties on all contributing rounds. The remaining term $Z_C$ is a sum of
\[
    q_1q_2q_3(|V_4|-q_4)
    =O\!\left(q^2\cdot\frac{q^3}{N}\cdot N\right)
    =O(q^5)
\]
independent, mean-zero random variables, each of magnitude
$\gamma/4$. Its conditional standard deviation is therefore
$O(\gamma\sqrt{q^5})=O(q^{7/2}/N)=o(\mu_C)$.
A standard Hoeffding bound therefore gives
$\Delta\ge\mu_C/2=\Omega(q^7/N^2)$ with high probability.

Finally, there are $q_1q_2=\Theta(q^2)$ active blocks, throughout
which the comparator stays at $u^{(1)}$. Each active block
contributes at most two switches: one on entry and one on exit.
Consecutive active blocks require no switch between them, so
the total number of switches is $O(q^2)$. Combining these bounds
with the concentration of the clique intersections proves the
claim. See Appendix~\ref{app:4hpc-signal-proof} for the detailed proof.

Recall that $q=\lfloor N^\beta\rfloor$ with
$\beta\in(1/3,1/2)$, $d=q^2$, and
\[
    |V_3|=d,
    \qquad |V_1|,|V_2|,|V_4|=\Theta(N).
\]
The planted clique $C$ is a uniformly random $q$-subset of $[N]$.
Write $C_s=C\cap V_s$ and $q_s=|C_s|$ for $s\in[4]$.
The comparator uses the two points
\[
    u^{(0)}=0,
    \qquad u^{(1)}=\gamma\mathbf 1_{C_3},
    \qquad \gamma=\frac{q}{2N}.
\]
It plays $u^{(1)}$ throughout block $(i,j)$ if
$(i,j)\in C_1\times C_2$, and plays $u^{(0)}$ throughout every
other block. We call the former blocks active.

\paragraph{Step 1: Concentration of the clique intersections.}
For each $s\in[4]$, the variable $q_s$ is hypergeometric, with mean
\[
    \mu_s:=\mathbb E[q_s]=\frac{q|V_s|}{N}.
\]
Indeed, each vertex of $V_s$ belongs to the uniformly chosen clique
with probability $q/N$. The multiplicative Chernoff bound for
sampling without replacement gives
\[
    \Pr(q_s<\mu_s/2)\le e^{-\mu_s/8},
    \qquad
    \Pr(q_s>2\mu_s)\le e^{-\mu_s/3}.
\]
Define the event
\[
    \mathcal E
    :=\left\{
        \frac{q|V_s|}{2N}\le q_s\le\frac{2q|V_s|}{N}
        \text{ for every }s\in[4]
      \right\}.
\]
A union bound, which does not require the four intersection sizes
to be independent, yields
\[
    \Pr(\mathcal E^c)
    \le 2\sum_{s=1}^4 e^{-\mu_s/8}
    \le 8e^{-\min_s\mu_s/8}
    =o_N(1).
\]
To justify the last equality, the means for $s=1,2,4$ are
$\Theta(q)$, while
\[
    \mu_3=\frac{qd}{N}=\frac{q^3}{N}
    =\Theta(N^{3\beta-1})\longrightarrow\infty.
\]
Thus every mean tends to infinity. This explains the assumption
$\beta>1/3$: it ensures concentration even for the smaller
coordinate part $V_3$.

For the next steps, fix any clique $C$ satisfying $\mathcal E$.
All expectations and probabilities conditional on $C$ are over the
remaining randomness of the hypergraph. On this event,
\[
    q_1,q_2,q_4=\Theta(q),
    \qquad q_3=\Theta\!\left(\frac{q^3}{N}\right).
\]
The constants in these estimates can be chosen uniformly over
all such cliques.

\paragraph{Step 2: Feasibility and number of switches.}
The origin is feasible, and the other point satisfies
\[
    \|u^{(1)}\|_2
    =\gamma\sqrt{q_3}
    \le\gamma\sqrt d
    =\frac{q^2}{2N}
    \longrightarrow0,
\]
where the limit uses $\beta<1/2$. Hence both points lie in
$B_2^d$ for sufficiently large $N$.

There are exactly $q_1q_2=\Theta(q^2)$ active blocks. The
comparator is constant within each block, so a switch can occur
only at a block boundary. Each active block can account for at
most two switches, one on entering it and one on leaving it.
Consecutive active blocks require no switch between them.
Consequently, the total number of switches is at most
$2q_1q_2=O(q^2)$. In fact, this upper bound holds for every
clique, since $q_1,q_2\le q$.

\paragraph{Step 3: Gain for a fixed active block and coordinate.}
Outside active blocks, the comparator equals the origin and its
gain is zero. On an active block, a round with $\ell\notin C_3$
also has zero gain, because $u^{(1)}_\ell=0$ and the loss depends
only on coordinate $\ell$. Thus only triples
$(i,j,\ell)\in C_1\times C_2\times C_3$ contribute to
\[
    \Delta=\sum_{t=1}^T\bigl(f_t(0)-f_t(u_t)\bigr).
\]
Fix one such triple. For every $v\in V_4$, expanding the square
gives
\[
\begin{aligned}
    f_{ij\ell v}(0)-f_{ij\ell v}(u^{(1)})
    &=\frac18\left(Y_{ij\ell v}^2
                  -(\gamma-Y_{ij\ell v})^2\right)\\
    &=\frac{\gamma Y_{ij\ell v}}4-\frac{\gamma^2}{8}.
\end{aligned}
\]
If $v\in C_4$, all four vertices lie in $C$, so the hyperedge is
forced to be present and $Y_{ij\ell v}=1$. If $v\notin C_4$,
the hyperedge is not forced; conditional on $C$, its label is
an unbiased sign. Therefore
\[
    \mathbb E[Y_{ij\ell v}\mid C]
    =\begin{cases}
        1,&v\in C_4,\\
        0,&v\notin C_4.
      \end{cases}
\]
Summing over the $|V_4|$ rounds for this triple, we obtain
\[
\begin{aligned}
    \mathbb E\!\left[
        \sum_{v\in V_4}
        \bigl(f_{ij\ell v}(0)-f_{ij\ell v}(u^{(1)})\bigr)
        \,\middle|\,C\right]
    &=\frac{\gamma q_4}{4}-\frac{\gamma^2|V_4|}{8}\\
    &\ge\frac{\gamma^2|V_4|}{8}\\
    &=\frac{q^2|V_4|}{32N^2}
      =\Omega\!\left(\frac{q^2}{N}\right).
\end{aligned}
\]
The inequality follows from
$q_4\ge q|V_4|/(2N)=\gamma|V_4|$, which holds on
$\mathcal E$. The positive bias from the forced hyperedges
therefore outweighs the quadratic penalty paid on all rounds.
This is a statement about the sum in expectation; an individual
round with label $-1$ has negative gain.

\paragraph{Step 4: Total expected gain.}
There are
\[
    M_C:=q_1q_2q_3
    =\Theta(q^2)\,\Theta\!\left(\frac{q^3}{N}\right)
    =\Theta\!\left(\frac{q^5}{N}\right)
\]
contributing triples. By linearity of expectation,
\[
\begin{aligned}
    \mu_C:=\mathbb E[\Delta\mid C]
    &=M_C\left(\frac{\gamma q_4}{4}
              -\frac{\gamma^2|V_4|}{8}\right)\\
    &\ge M_C\frac{q^2|V_4|}{32N^2}
     \ge c_0\frac{q^7}{N^2}
\end{aligned}
\]
for a constant $c_0>0$ independent of $N$ and of the clique
satisfying $\mathcal E$. In particular, the power $q^7/N^2$
comes from multiplying the number of active blocks, the number
of clique coordinates, and the expected gain for each such pair:
\[
    \underbrace{\Theta(q^2)}_{\text{active blocks}}
    \;\underbrace{\Theta(q^3/N)}_{\text{clique coordinates}}
    \;\underbrace{\Omega(q^2/N)}_{\text{gain per pair}}
    =\Omega\!\left(\frac{q^7}{N^2}\right).
\]

\paragraph{Step 5: Concentration of the total gain.}
It remains to show that the realized gain is unlikely to fall
far below its conditional mean. Define
\[
    \mathcal I_C
    :=C_1\times C_2\times C_3\times(V_4\setminus C_4),
    \qquad n_C:=|\mathcal I_C|.
\]
Conditional on $C$, both the gain from forced edges and all
quadratic penalty terms are deterministic. Hence
\[
    \Delta-\mu_C
    =\sum_{(i,j,\ell,v)\in\mathcal I_C}
      \frac{\gamma Y_{ij\ell v}}4.
\]
Every summand has mean zero and takes values in
$\{-\gamma/4,\gamma/4\}$. These summands are independent
conditional on $C$: they correspond to unforced hyperedges,
and distinct tuples give distinct hyperedges because the
four partition parts are disjoint. Their number satisfies
\[
    n_C=M_C(|V_4|-q_4)
    \le M_C|V_4|=O(q^5).
\]

When $n_C>0$, Hoeffding's inequality gives, for every $a>0$,
\[
    \Pr(\Delta-\mu_C\le-a\mid C)
    \le\exp\!\left(-\frac{2a^2}{n_C(\gamma/2)^2}\right)
    =\exp\!\left(-\frac{8a^2}{n_C\gamma^2}\right).
\]
The denominator uses the squared range length of each summand,
which is $(\gamma/2)^2$. If $n_C=0$, then
$\Delta=\mu_C$ deterministically and no concentration bound
is needed.

Choose $c=c_0/2$ and set $S_N=cq^7/N^2$. Since
$\mu_C\ge c_0q^7/N^2$, we have $S_N\le\mu_C/2$.
Using $a=\mu_C/2$ in the preceding bound yields
\[
\begin{aligned}
    \Pr(\Delta<S_N\mid C)
    &\le\Pr(\Delta-\mu_C\le-\mu_C/2\mid C)\\
    &\le\exp\!\left(-\frac{2\mu_C^2}{n_C\gamma^2}\right)\\
    &\le\exp\!\left(-\Omega\!\left(
        \frac{(q^7/N^2)^2}{q^5(q^2/N^2)}\right)\right)\\
    &=\exp\!\left(-\Omega\!\left(\frac{q^7}{N^2}\right)\right)
      =o_N(1).
\end{aligned}
\]
Here $\gamma^2=q^2/(4N^2)$, and
$q^7/N^2=\Theta(N^{7\beta-2})\to\infty$ because
$\beta>1/3$. The bound is uniform over cliques satisfying
$\mathcal E$. Averaging over those cliques and adding the
probability of $\mathcal E^c$ therefore gives
\[
    \Pr(\Delta<S_N)
    \le\Pr(\mathcal E^c)
      +\exp\!\left(-\Omega\!\left(\frac{q^7}{N^2}\right)\right)
    =o_N(1).
\]
Together with feasibility and the switch bound, this proves
Lemma~\ref{lem:4hpc-signal}.

Consider the losses
\[
    f_t(x)=\frac18(x_{\ell_t}-Y_t)^2,
\]
and let $\mathcal H_t$ be the information available just before
$Y_t$ is revealed, including the learner's history, its current
decision $x_t$, and the coordinate $\ell_t$. Thus
$b_t:=x_{t,\ell_t}$ is $\mathcal H_t$-measurable. For a
randomized learner, conditioning on $x_t$ includes the outcome
of its random choice for this round; we will average over that
choice when taking unconditional expectations.

Under the hypothesis of the lemma,
\[
    \Pr_0(Y_t=1\mid\mathcal H_t)
    =\Pr_0(Y_t=-1\mid\mathcal H_t)=\frac12.
\]
In our reduction, this property follows from the independence
of hyperedges under the null model and the fact that each round
queries a previously unqueried hyperedge. In particular,
observing past losses does not reveal the next label.

\paragraph{Step 1: Express the gain on one round.}
Since $Y_t^2=1$, the origin has loss $f_t(0)=1/8$. Expanding
the square gives
\[
\begin{aligned}
    g_t:=f_t(0)-f_t(x_t)
    &=\frac18\left(Y_t^2-(b_t-Y_t)^2\right)\\
    &=\frac{b_tY_t}{4}-\frac{b_t^2}{8}.
\end{aligned}
\]
Consequently,
\[
    \mathbb E_0[g_t\mid\mathcal H_t]
    =-\frac{b_t^2}{8}\le0.
\]
The linear term is a mean-zero random reward, while the
quadratic term is a deterministic penalty once the prediction
has been chosen. Nonpositive expected gain alone does not give
the desired exponential upper-tail bound, so we next control
the exponential moment of $g_t$.

\paragraph{Step 2: Bound the conditional exponential moment.}
Averaging over the two equally likely values of $Y_t$ yields
\[
\begin{aligned}
    \mathbb E_0[e^{g_t}\mid\mathcal H_t]
    &=\frac12e^{b_t/4-b_t^2/8}
      +\frac12e^{-b_t/4-b_t^2/8}\\
    &=e^{-b_t^2/8}\cosh(b_t/4),
\end{aligned}
\]
where $\cosh(z)=(e^z+e^{-z})/2$. For every real $z$,
\[
    \cosh(z)
    =\sum_{j=0}^{\infty}\frac{z^{2j}}{(2j)!}
    \le\sum_{j=0}^{\infty}\frac{(z^2/2)^j}{j!}
    =e^{z^2/2}.
\]
The termwise inequality uses $(2j)!\ge2^j j!$ and
$z^{2j}\ge0$. Substituting $z=b_t/4$ gives
\[
    \mathbb E_0[e^{g_t}\mid\mathcal H_t]
    \le e^{-b_t^2/8+b_t^2/32}
    =e^{-3b_t^2/32}
    \le1.
\]
This bound holds for every realized history and every current
decision, so it allows arbitrary adaptive and randomized
choices by the learner.

\paragraph{Step 3: Iterate the conditional bound.}
Define
\[
    G_0=0,
    \qquad G_t=\sum_{s=1}^t g_s,
    \qquad G=G_T.
\]
The accumulated gain $G_{t-1}$ is known given $\mathcal H_t$.
Since $G_t=G_{t-1}+g_t$, the tower property gives
\[
\begin{aligned}
    \mathbb E_0[e^{G_t}]
    &=\mathbb E_0\!\left[
        \mathbb E_0[e^{G_{t-1}}e^{g_t}\mid\mathcal H_t]
      \right]\\
    &=\mathbb E_0\!\left[
        e^{G_{t-1}}\mathbb E_0[e^{g_t}\mid\mathcal H_t]
      \right]\\
    &\le\mathbb E_0[e^{G_{t-1}}].
\end{aligned}
\]
Iterating this inequality from $t=T$ down to $t=1$ yields
\[
    \mathbb E_0[e^G]
    \le\mathbb E_0[e^{G_{T-1}}]
    \le\cdots\le\mathbb E_0[e^{G_0}]=1.
\]
The gains $g_t$ need not be independent, since the learner
can adapt to previous outcomes. Conditioning at each round
is what makes the iteration valid despite this dependence.
All expectations are finite because the losses lie in
$[0,1/2]$ and the horizon is finite.

\paragraph{Step 4: Derive the upper-tail bound.}
The exponential is increasing and $e^G$ is nonnegative.
Markov's inequality therefore gives, for every real $z$,
\[
    \Pr_0(G\ge z)
    =\Pr_0(e^G\ge e^z)
    \le\frac{\mathbb E_0[e^G]}{e^z}
    \le e^{-z}.
\]
In particular, for any $\delta\in(0,1)$, the probability that
the learner gains at least $\log(1/\delta)$ over the origin
is at most $\delta$, independently of the horizon $T$.
This proves both conclusions of Lemma~\ref{lem:4hpc-null}.

Since $C$ is a uniformly random $q$-subset of $[N]$,
$\mathbb E[|C\cap V|]=q|V|/N$ for each partition part $V$.
Thus, the expected sizes of $C_{B_a}$ and $C_{W_c}$ are $\Theta(q)$,
and those of $C_{D_j}$ are $\Theta(q^3/N)\to\infty$ because $\beta>1/3$.
A Chernoff bound therefore gives,
with probability $1-o_N(1)$,
\[
\frac{q|V|}{2N}\le |C\cap V|\le\frac{2q|V|}{N}
\qquad\text{for every partition part }V.
\]
Condition on a clique $C$ satisfying this event. In particular,
$|C_{B_a}|,|C_{W_c}|=\Theta(q)$ and
$|\mathcal C_D|=\Theta(q^{3s}/N^s)$. Moreover,
\[
\|u^{(1)}\|_2^2
=\gamma^2|\mathcal C_D|
\le 2^{s-2h}\frac{q^{2h+3s}}{N^{2h+s}}
\le N^{s/2-h}=o_N(1),
\]
where $\beta<1/2$ gives $q\le N^{1/2}$, and $h>s/2$.
Thus both comparator points lie in $B_2^d$ for sufficiently
large $N$. There are $\prod_{a=1}^p|C_{B_a}|=\Theta(q^p)$
active blocks, so the comparator makes $O(q^p)$ switches.

Outside active blocks the comparator is the origin, and outside
$\mathcal C_D$ its coordinates are zero. Thus only active
block--planted-coordinate pairs contribute to $\Delta$.
Write $\mathcal W=\prod_{c=1}^h W_c$,
$M_W=\prod_{c=1}^h|C_{W_c}|$, and $L_W=|\mathcal W|$.
For each such pair $(\mathbf i,\boldsymbol\ell)$,
\[
\begin{aligned}
\mathbb E\!\left[\sum_{\mathbf v\in\mathcal W}
\bigl(f_{\mathbf i,\boldsymbol\ell,\mathbf v}(0)
-f_{\mathbf i,\boldsymbol\ell,\mathbf v}(u^{(1)})\bigr)
\,\middle|\,C\right]
&=\mathbb E\!\left[\sum_{\mathbf v\in\mathcal W}
\left(\frac{\gamma Y_{\mathbf i,\boldsymbol\ell,\mathbf v}}4
-\frac{\gamma^2}8\right)\,\middle|\,C\right] \\
&=\frac{\gamma M_W}{4}-\frac{\gamma^2L_W}{8}
\ge\frac{\gamma^2L_W}{8}
=\Omega\left(\frac{q^{2h}}{N^h}\right),
\end{aligned}
\]
where the concentration event gives
$M_W\ge(q/2N)^hL_W=\gamma L_W$.
Since there are
$(\prod_{a=1}^p|C_{B_a}|)|\mathcal C_D|
=\Theta(q^{p+3s}/N^s)$ such pairs, some constant $c_0>0$ satisfies
\[
\mu_C:=\mathbb E[\Delta\mid C]
\ge c_0\frac{q^{p+3s+2h}}{N^{s+h}}.
\]

Conditional on $C$, the random part of $\Delta$ is a sum of at most
$O(q^{p+3s}N^{h-s})$ independent terms, each taking values in
$\{-\gamma/4,\gamma/4\}$.
Choose $c=c_0/2$. Hoeffding's inequality then gives
\[
\begin{aligned}
\Pr\!\left(\Delta<c\frac{q^{p+3s+2h}}{N^{s+h}}
\,\middle|\,C\right)
&\le\exp\left(-\Omega\left(
\frac{(q^{p+3s+2h}/N^{s+h})^2}
{q^{p+3s}N^{h-s}\gamma^2}\right)\right) \\
&=\exp\left(-\Omega\left(
\frac{q^{p+3s+2h}}{N^{s+h}}\right)\right)=o_N(1).
\end{aligned}
\]
The final equality follows from
$\beta>(s+h)/(p+3s+2h)$, which ensures
$q^{p+3s+2h}/N^{s+h}\to\infty$.
Combining this with the concentration event proves the claim.

<!-- proof-9: body_md -->
Suppose a randomized polynomial-time learner guarantees
$\widetilde O(k^a d^b)$ expected regret uniformly over instances
with $k=\Theta(d^\theta)$, for a fixed rational $\theta>0$
and fixed $a,b\ge0$ satisfying $\theta a+b<\theta+1/2$.
Choose positive integers $p,s$ with $p/(2s)=\theta$.
Then the hypothesis gives
\[
pa+2sb=2s(\theta a+b)<2s(\theta+1/2)=p+s.
\]
Set $h=\lceil s/2\rceil+1$ and $r=p+s+h\ge4$, and choose
\[
\beta\in\left(
\max\left\{\frac13,
\frac{s+h}{p+3s+2h-pa-2sb}\right\},\frac12\right).
\]
This interval is nonempty because
$p+3s+2h-pa-2sb>2(s+h)$.
It also satisfies the condition on $\beta$ used in the construction.

By Lemma~\ref{lem:r-hpc-signal}, on its high-probability event
the comparator uses two points and makes at most $k$ switches,
where $k$ is a sufficiently large constant multiple of $q^p$.
Since $d=q^{2s}$, we have $k=\Theta(d^{p/(2s)})=\Theta(d^\theta)$.
Moreover, $q=\Theta(N^\beta)$ gives
$T=\Theta(N^{p+h}d)=\Theta(d^{1+(p+h)/(2s\beta)})$,
which is polynomial in $d$.
Thus the assumed guarantee applies and is
$\widetilde O(q^{pa+2sb})$, whereas
\[
S_N=c\frac{q^{p+3s+2h}}{N^{s+h}},\qquad
\beta(p+3s+2h-pa-2sb)>s+h.
\]
Consequently, the assumed regret guarantee is $o(S_N)$.
For all sufficiently large $N$, it is at most $S_N/4$.
Lemma~\ref{lem:4hpc-detection} then yields a randomized
polynomial-time test for $r$-HPC, contradicting
Conjecture~\ref{conj:hpc}.

Fix any comparator path $u_{1:T}\in\mathcal U_{m,k}(\mathcal{K})$.
Let $I_1,\ldots,I_s$ be its maximal constant segments, where
$s\le k+1$, and let $v_j$ be its constant value on $I_j$.
Since $v_j$ is one possible fixed decision on $I_j$, the
definition of adaptive regret gives
\[
\begin{aligned}
    \sum_{t\in I_j}\bigl(f_t(x_t)-f_t(v_j)\bigr)
    &\le \sum_{t\in I_j}f_t(x_t)
        -\min_{v\in\mathcal{K}}\sum_{t\in I_j}f_t(v) \le \mathsf{AReg}_T(\mathcal A).
\end{aligned}
\]
Summing over the segments and considering the best comparator path in the set $\mathcal U_{m,k}(\mathcal{K})$ gives the claimed bound.
