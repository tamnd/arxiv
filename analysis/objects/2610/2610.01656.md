<!-- lem-1: body_md -->
For fixed $\eta$ satisfying  $0<\eta\le 1$,  the $C^\infty$ operator value function  $\xi\rightarrow T_{\xi,\eta}$  from $\mathbb{R}^d$ to the space of bounded linear operators $\mathcal{L}[L^2(\Omega;\mathbb{C}^d)]$
on $L^2(\Omega;\mathbb{C}^d)$  has an analytic continuation to a region  $\{\xi\in\mathbb{C}^d:  |\Im\xi|^2<c_d\eta\}$ for some constant $c_d>0$ depending only on $d$.  Furthermore,  for $\xi$ in this region there exists a constant 
$C_d$ depending only on $d$ such that
\begin{equation} \label{E3}
\|T_{\xi,\eta}\|_{L^2(\Omega;\mathbb{C}^d)} \ \le \ 1+\frac{C_d|\Im\xi|^2}{\eta} \ , \ \quad \|T_{\xi,\eta}-T_{\Re\xi,\eta}\|_{L^2(\Omega;\mathbb{C}^d)} \ \le \ \frac{C_d|\Im\xi|}{\sqrt{\eta}} \ .

<!-- proof-1: body_md -->
Since the operator $A=[A_1,\dots, A_d]$ in  (\ref{A3})  has the property that the $A_j, \ j=1,\dots,d$ form  a commuting set of self-adjoint operators, it is sufficient to prove the result for $A\equiv 0$. 
In that case  $T_{\xi,\eta}$ is  for $\xi\in\mathbb{R}^d$ a Hermitian matrix operator  acting on $\mathbb{C}^d$. 
The matrix $T_{\xi,\eta}$ has entries $ T_{\xi,\eta}(j,k), \ 1\le j,k\le d,$ where
\begin{multline} \label{F3}
T_{\xi,\eta}(j,k) \ = \ \frac{e_j(\xi)e_k(\xi)^*}{\eta+e(\xi)^*e(\xi)} \ = \ \frac{ \{e^{-i\xi_j}-1\}\{e^{i\xi_k}-1\}}{\eta+ \sum_{r=1}^d \{e^{i\xi_r}-1\}\{e^{-i\xi_r}-1\} }  \\
= \ \frac{ \{e^{-i\Re\xi_j+\Im\xi_j}-1\}\{e^{i\Re\xi_k-\Im\xi_k}-1\}}{\eta+ \sum_{r=1}^d \{e^{i\Re\xi_r-\Im\xi_r}-1\}\{e^{-i\Re\xi_r+\Im\xi_r}-1\} }  \ .
\end{multline} 
The denominator in the final expression of (\ref{F3}) may be written as
\begin{multline} \label{G3}
{\rm denom} \ = \ \eta+ e(\Re\xi)^*e(\Re\xi)+ \sum_{r=1}^d \{1-e^{\Im\xi_r}\}\{1-e^{-\Im\xi_r}\} \\  
+ \sum_{r=1}^d \{1-e^{\Im\xi_r}\}e_r(\Re\xi)  +\sum_{r=1}^d \{1-e^{-\Im\xi_r}\}e_r(\Re\xi)^* \ .
\end{multline}
We write the numerator in (\ref{F3}) as 
\begin{multline} \label{H3}
{\rm num}_{j,k} \ = \ e_j(\Re\xi)e_k(\Re\xi)^*+e^{-i\Re\xi_j+i\Re\xi_k}\{e^{\Im\xi_j}-1\}\{e^{-\Im\xi_k}-1\} \\
+e^{-i\Re\xi_j}\{e^{\Im\xi_j}-1\}e_k(\Re\xi)^*+e_j(\Re\xi)e^{i\Re\xi_k}\{e^{-\Im\xi_k}-1\}  \ .
\end{multline}
Then we have that
\begin{equation} \label{I3}
| {\rm denom}| \ \ge   \ \frac{\eta}{2}+\left[1-\frac{C_1|\Im\xi|^2}{\eta}\right]  |e(\Re\xi)|^2 \ , \quad {\rm for \ some \ constant \ } C_1.
\end{equation}
From (\ref{H3}) we also have for the $d\times d$  matrix ${\rm num}_{j,k}, \ 1\le j,k\le d,$ acting on $\mathbb{C}^d$   that for some constants $C_2,C_3$ one has the inequality
\begin{equation} \label{J3}
|{\rm num}  (v)| \le \ \left[   |e(\Re\xi)|^2  +C_2|\Im\xi|^2+ C_3|e(\Re\xi)||\Im\xi|                    \right]|v| \ , \quad v\in\mathbb{C}^d \ .
\end{equation}
On using the inequality 
\begin{equation} \label{K3}
C_3|e(\Re\xi)||\Im\xi|  \ \le \ \frac{C_3^2|\Im\xi|^2}{\eta}|e(\Re\xi)|^2+ \frac{\eta}{4} \ ,
\end{equation}
in (\ref{J3}) we see that (\ref{I3}), (\ref{J3}) implies the first inequality of (\ref{E3}). For the second inequality we use instead of (\ref{K3}) the inequality 
\begin{equation} \label{L3}
|e(\Re\xi)||\Im\xi|  \ \le \  \left(\frac{|\Im\xi|}{\sqrt{\eta}}\right) \sqrt{\eta} |e(\Re\xi)| \ ,
\end{equation} 
whence the inequality follows from (\ref{I3}).

{\rm (Hunt interpolation theorem)}
Let $(\mathcal{M},\mu)$ be a measure space and $ \mathcal{H}_1, \mathcal{H}_2$   Hilbert spaces.  Suppose that  $p_1,p_2,q_1,q_2\in\mathbb{R}$ satisfy $1\le p_1<p_2\le\infty, \ 1\le q_1< q_2\le \infty$, and  $T$ is a  bounded linear operator  from
$L^{p}(\mathcal{M};\mathcal{H}_1)$  to  $L^{q}(\mathcal{M};\mathcal{H}_2)$ when $p=p_1, q=q_1$ and $p=p_2, q=q_2$.   Hence there are constants $C_1,C_2$ such that 
\begin{equation} \label{AB3}
\|Tf\|_{q_1}\le C_1\|f\|_{p_1} \ , \quad \|Tf\|_{q_2}\le C_2\|f\|_{p_2}  \ , \quad {\rm all \ } f:\mathcal{M}\rightarrow\mathcal{H}_1 \ .
\end{equation}
Then for any $p,q$ satisfying the identities
\begin{equation} \label{AC3}
\frac{1}{p} \ = \ \frac{\theta}{p_1}+\frac{1-\theta}{p_2} \ ,  \quad \frac{1}{q} \ = \ \frac{\theta}{q_1}+\frac{1-\theta}{q_2} \ ,  \quad {\rm with \ } 0<\theta<1 \ ,
\end{equation}
one has the inequality
\begin{equation} \label{AD3}
\|Tf\|_{q,w}\le M C_1^\theta C_2^{1-\theta}\|f\|_{p,w} \ \ , \quad {\rm where \ } M \ {\rm depends \ only \ on \ } p_1,p_2,q_1,q_2,\theta \ .
\end{equation}

{\rm (Marcinkiewicz interpolation theorem)}
Let $(\mathcal{M},\mu)$ be a measure space and $ \mathcal{H}_1, \mathcal{H}_2$   Hilbert spaces.  Suppose that  $p_1,p_2,q_1,q_2\in\mathbb{R}$ satisfy $1\le p_1<p_2\le\infty, \ 1\le q_1< q_2\le \infty$, and  $T$ is a  bounded linear operator  from
$L^{p}(\mathcal{M};\mathcal{H}_1)$  to  $L^{q,w}(\mathcal{M};\mathcal{H}_2)$ when $p=p_1, q=q_1$ and $p=p_2, q=q_2$.   Hence there are constants $C_1,C_2$ such that 
\begin{equation} \label{AE3}
\|Tf\|_{q_1,w}\le C_1\|f\|_{p_1} \ , \quad \|Tf\|_{q_2,w}\le C_2\|f\|_{p_2}  \ , \quad {\rm all \ } f:\mathcal{M}\rightarrow\mathcal{H}_1 \ .
\end{equation}
Then for any $p,q$ satisfying the identities
\begin{equation} \label{AF3}
\frac{1}{p} \ = \ \frac{\theta}{p_1}+\frac{1-\theta}{p_2} \ ,  \quad \frac{1}{q} \ = \ \frac{\theta}{q_1}+\frac{1-\theta}{q_2} \ ,  \quad {\rm with \ } 0<\theta<1 \ ,
\end{equation}
one has the inequality
\begin{equation} \label{AG3}
\|Tf\|_{q}\le M C_1^\theta C_2^{1-\theta}\|f\|_{p} \ \ , \quad {\rm where \ } M \ {\rm depends \ only \ on \ } p_1,p_2,q_1,q_2,\theta \ .
\end{equation}

\left(\frac{\partial}{\partial\xi_1}\right)^{d-1}F(\xi+ai,\eta)  \ = \ P\left(  \xi+ai,\eta,\frac{\partial q(\xi+ai,\eta)}{\partial\xi_1}, \dots,      \frac{\partial^{d-1} q(\xi+ai,\eta)}{\partial\xi_1^{d-1}}                           \right) \ ,
\end{equation}
where $P$ is a polynomial in $\partial q(\xi,\eta)/\partial\xi_1,\dots,\partial ^{d-1}q(\xi,\eta)/\partial\xi_1^{d-1}$, with coefficients depending on $\xi+ai,\eta, q(\xi+ai,\eta)$.  A typical term in the polynomial has the form
\begin{equation} \label{O4}
A(\xi+ai,\eta)\prod_{j=1}^{d-1} \frac{\partial ^{r_j}q_{m_j,n_j}(\xi+ai,\eta)}{\partial\xi_1^{r_j}} \ , \quad {\rm with \ } r_1+\cdots+r_{d-1}=r \ \le  \ d-1 \ ,
\end{equation}
where the $r_j$ are non-negative integers.  One also sees that
\begin{equation} \label{P4}
|A(\xi+ai,\eta)| \ \le \ \frac{C}{|\xi|^{d-1-r}} \ , \quad \xi\in\mathbb{T}^d \ .
\end{equation}

To bound the integral on  $\mathbb{T}^d$ 
of the term (\ref{O4}) satisfying (\ref{P4}), we use the fact that if  $f_j\in L^{p_j}_w(\mathcal{M};\mathbb{R}), \ j=1,\dots,N,$ with 
 $1/p_1+\cdots+1/p_N=1/p\le1$, then $\prod_{j=1}^N f_j$ is in $L^p_w(\mathcal{M};\mathbb{R})$ and   $\left\|\prod_{j=1}^N f_j\right\|_{p,w}\le N^{1/p}\prod_{j=1}^N\|f_j\|_{p_j,w}$.  Then we apply a classical
 inequality for functions in weak spaces: Let  $g\in L^p_w(\mathcal{M};\mathbb{R})$ with $p>1$ and $E\subset \mathcal{M}$ a measurable set. 
 Then
 \begin{equation} \label{Q4}
 \int_E |g| \ \le \  \frac{p}{p-1}\|g\|_{p,w}m(E)^{1-1/p} \ .
 \end{equation}
 From Corollary 3.1 we have that the function $\xi\rightarrow \partial^{r_j} q(\xi+ai)/\partial\xi_1^{r_j}$  is in $L^{d/r_j}_w(\mathbb{T}^d;\mathcal{L}(\mathbb{C}^d)),     \ j=1,\dots,d-1$, whence the product of the derivatives in (\ref{O4}) is in
 $L^{d/r}_w(\mathbb{T}^d;\mathbb{C})$. From the H\"{o}lder inequality we have that for any $p\ge 1$ and  $1/p+1/p'=1$, 
 \begin{multline} \label{R4}
 \int_{\mathbb{T}^d} \left|     A(\xi+ai,\eta)\prod_{j=1}^{d-1} \frac{\partial ^{r_j}q_{m_j,n_j}(\xi+ai,\eta)}{\partial\xi_1^{r_j}}                   \right| \ d\xi \\
  \le \  \left\{     \int_{\mathbb{T}^d} \left|     A(\xi+ai,\eta)\right|^{p'}    \ d\xi                 \right\}^{1/p'}   \left\{     \int_{\mathbb{T}^d} \left|   \prod_{j=1}^{d-1} \frac{\partial ^{r_j}q_{m_j,n_j}(\xi+ai,\eta)}{\partial\xi_1^{r_j}}  \right|^{p}    \ d\xi                 \right\}^{1/p} \ .
 \end{multline}
The first integral on the  RHS of (\ref{R4}) is bounded provided $1/p'>(d-1-r)/d$, while the second integral is bounded  provided  $1/p >r/d$. Since $r\le d-1$ we are able to choose $p>1$ which satisfies both these inequalities.  It follows that
$|I_{d-1}(x)|\le C/(1+|x|)^{d-1}, \ x\in\mathbb{Z}^d$.  We have proven (\ref{E4}) for $\nu=1$. 

 To show that (\ref{E4}) continues to hold for $0<\nu\le 1$ 
 we do a fractional integration by parts in (\ref{M4}) and use the inequality for  (\ref{CB3}) given by Corollary 3.1. We choose $\Delta\xi\simeq 1/(1+|x|)$ such that $\Delta\xi\cdot x=\pi$. We have then that
 \begin{multline} \label{S4}
 2 I_{d-1}(x) \ = \ \frac{1}{(ix_1)^{d-1}}\int_{\mathbb{T}^d} e^{-i\xi\cdot x}  \frac{\partial^{d-1} F(\xi+ai,\eta)}{\partial\xi_1^{d-1}} \  d\xi  \\
 - \frac{1}{(ix_1)^{d-1}}\int_{\mathbb{T}^d} e^{-i\xi\cdot x}  \frac{\partial^{d-1} F(\xi+\Delta\xi+ai,\eta)}{\partial\xi_1^{d-1}} \  d\xi  \ .
 \end{multline}
 Since $\Delta\xi\simeq 1/(1+|x|)$  we have from (\ref{S4}) that
 \begin{equation}\label{T4}
 \left|I_{d-1}(x)\right| \ \le \  \frac{C}{|x|^{d-\nu}} \int_{\mathbb{T}^d}  \left|  \frac{1}{|\Delta\xi|^{1-\nu} } \left\{\frac{\partial^{d-1} F(\xi+ai,\eta)}{\partial\xi_1^{d-1}}-\frac{\partial^{d-1} F(\xi+\Delta\xi+ai,\eta)}{\partial\xi_1^{d-1}} \right\}                                \right| \ d\xi  \ .
 \end{equation}
 The integrand in (\ref{S4}) consists of a sum of terms of the form
 \begin{multline} \label{U4}
A(\xi+ai,\eta)\prod_{j=1}^{d-1} \frac{\partial ^{r_j}q_{m_j,n_j}(\xi+ai,\eta)}{\partial\xi_1^{r_j}} -A(\xi+\Delta\xi+ai,\eta)\prod_{j=1}^{d-1} \frac{\partial ^{r_j}q_{m_j,n_j}(\xi+\Delta\xi+ai,\eta)}{\partial\xi_1^{r_j}} \\
= \ 
\sum_{k=0}^{d-1} J_k(\xi+ai,\eta,\Delta\xi) \  ,
\end{multline}
 where
 \begin{multline} \label{V4}
 J_0(\xi+ai,\eta,\Delta\xi)  \ = \  \{A(\xi+ai,\eta)-A(\xi+\Delta\xi+ai,\eta)\}\prod_{j=1}^{d-1} \frac{\partial ^{r_j}q_{m_j,n_j}(\xi+ai,\eta)}{\partial\xi_1^{r_j}}  \ , \\
  {\rm For \ } k\ge 1, \quad J_k(\xi+ai,\eta,\Delta\xi)  \ = \  A(\xi+\Delta\xi+ai,\eta)\prod_{j=1}^{k-1} \frac{\partial ^{r_j}q_{m_j,n_j}(\xi+\Delta\xi+ai,\eta)}{\partial\xi_1^{r_j}}  \   \\
\times  \    \left\{         \frac{\partial ^{r_k}q_{m_j,n_j}(\xi+ai,\eta)}{\partial\xi_1^{r_k}}         -     \frac{\partial ^{r_k}q_{m_j,n_j}(\xi+\Delta\xi+ai,\eta)}{\partial\xi_1^{r_k} }                         \right\}
\prod_{j=k+1}^{d-1} \frac{\partial ^{r_j}q_{m_j,n_j}(\xi+ai,\eta)}{\partial\xi_1^{r_j}}  \ .
 \end{multline}
 To bound the integral of $J_0$ in (\ref{V4}) we write 
 \begin{multline}\label{W4}
     A(\xi+ai,\eta)-A(\xi+\Delta\xi+ai,\eta)     \\
      = \ A_0(\xi+ai,\eta,\Delta\xi) + {\rm Tr} \left[A_1(\xi+ai,\eta,\Delta\xi)\left\{       q(\xi+ai,\eta)-q(\xi+\Delta\xi+ai,\eta)               \right\} \right] \  .
\end{multline}
 The function $A_0(\xi+ai,\eta,\Delta\xi)$ and the $d\times d$ matrix $A_1(\xi+ai,\eta,\Delta\xi)$ satisfy inequalities similar to (\ref{P4}).   Thus we have that
 \begin{equation} \label{X4}
 \begin{array}{ccc}
 |A_0(\xi+ai,\eta,\Delta\xi)| \ &\le& \ C\left[  \frac{1}{|\xi|^{d-1-r}} +  \frac{1}{|\xi+\Delta\xi|^{d-1-r}} \right]\ , \quad \xi\in\mathbb{T}^d \ , \\
|A_0(\xi+ai,\eta,\Delta\xi)| \ &\le& \ C|\Delta\xi|\left[  \frac{1}{|\xi|^{d-r}} +  \frac{1}{|\xi+\Delta\xi|^{d-r}} \right]\ , \quad \xi\in\mathbb{T}^d \ , \\
|A_1(\xi+ai,\eta,\Delta\xi)| \ &\le& \ C\left[  \frac{1}{|\xi|^{d-1-r}} +  \frac{1}{|\xi+\Delta\xi|^{d-1-r}} \right]\ , \quad \xi\in\mathbb{T}^d \ .
\end{array}
\end{equation}
Using the first two inequalities in (\ref{X4}), we see that 
\begin{equation} \label{Y4}
  \int_{\mathbb{T}^d} \left|     \frac{A_0(\xi+ai,\eta,\Delta\xi)}{|\Delta\xi|^{1-\nu}}\right|^{p'}    \ d\xi    \ < \ \infty   \quad {\rm provided \ } \frac{1}{p'}>\frac{d-\nu-r}{d} \ .
\end{equation}
It follows from (\ref{Y4}) similarly to  before that the contribution to (\ref{T4}) corresponding to the term in $J_0$ containing  $A_0$ is integrable on $\mathbb{T}^d$  for all $\nu, \  0<\nu\le 1$.  To deal with the term in $J_0$ corresponding to the second term
on the RHS of (\ref{W4}) we observe from Corollary 3.1 that the function $\xi\rightarrow \left\{       q(\xi+ai,\eta)-q(\xi+\Delta\xi+ai,\eta)               \right\}/|\Delta\xi|^{1-\nu}$ is in the space $L_w^{d/(1-\nu)}(\mathbb{T}^d;\mathcal{L}(\mathbb{C}^d))$. 
Thus we have that the product of this function together with the other derivatives in the formula for $J_0$ is in $L_w^{d/(r+1-\nu)}(\mathbb{T}^d;\mathcal{L}(\mathbb{C}^d))$.Using the third inequality of (\ref{X4}) and arguing as before 
we conclude that that the term in $J_0$ corresponding to $A_1$ is integrable on $\mathbb{T}^d$  for all $\nu, \  0<\nu\le 1$.  We can similarly see that the contributions to (\ref{T4}) from  $J_k, \ k\ge 1,$ are also  integrable on $\mathbb{T}^d$  for all $\nu, \  0<\nu\le 1$.

Using analytic continuation again we have the representation
\begin{equation} \label{AF4}
\nabla\nabla^*G_{\mathbf{a},\eta}(x)  \ = \  \frac{e^{a\cdot x}}{(2\pi)^d}\int_{\mathbb{T}^d} \frac{e(\xi+ai)e(\xi-ai)^*e^{-ix\cdot\xi} }{\eta+e(\xi-ai)^*q(\xi+ai,\eta)e(\xi+ai)} \ d\xi \  .
\end{equation}
By choosing $a$ in (\ref{AF4})  appropriately we obtain the exponential falloff in (\ref{AE4}). 
Let $\phi:\mathbb{R}^d\rightarrow\mathbb{R}$ be a $C^\infty_c$ function   such that $\phi(\xi)=1$ for $|\xi|<20\pi$. To obtain the polynomial decay in (\ref{AE4}) we write the integral in (\ref{AF4}) as a sum,
$I_1(x)+I_2(x)$, where
\begin{equation} \label{AG4}
I_1(x) \ = \ \int_{\mathbb{T}^d} \phi(\{1+|x|\}\xi)\frac{e(\xi+ai)e(\xi-ai)^*e^{-ix\cdot\xi} }{\eta+e(\xi-ai)^*q(\xi+ai,\eta)e(\xi+ai)} \ d\xi \  .
\end{equation}
It easily follows from (\ref{I4}) that $|I_1(x)|\le C/(1+|x|)^d$.  

Assuming  that $|x_1|\ge |x|/d$ and using integration by parts, we write $I_2(x)$ as
\begin{multline} \label{AH4}
I_2(x) \ = \ \frac{1}{(ix_1)^{d}}\int_{\mathbb{T}^d}e^{-ix\cdot\xi} \frac{\partial^{d}}{\partial\xi_1^{d}}\Bigg[ \left\{1-\phi(\{1+|x|\}\xi)\right\}  \\
\times\frac{e(\xi+ai)e(\xi-ai)^* }{\eta+e(\xi-ai)^*q(\xi+ai,\eta)e(\xi+ai)} \Bigg] \ d\xi  \ = \ \sum_{j=0}^d I_{2,j}(x) \  .
\end{multline}
The sum over $j, \ 0\le j\le d,$ in (\ref{AH4}) comes from the application of Leibniz rule, with $I_{2,j}(x)$ being the term with $j$ derivatives acting on  the function $\xi\rightarrow 1-\phi(\{1+|x|\}\xi)$.
Hence we have for $j\ge 1$  that 
\begin{equation} \label{AI4}
|I_{2,j}(x)| \ \le \  \frac{C}{(1+|x|)^{d-j}} \int_{|\xi|<R/(1+|x|)}  \left|        \frac{\partial^{d-j}}{\partial\xi_1^{d-j}} \left\{\frac{e(\xi+ai)e(\xi-ai)^* }{\eta+e(\xi-ai)^*q(\xi+ai,\eta)e(\xi+ai)}            \right\}                           \right| \ d\xi \ ,
\end{equation}
for any ball in $\mathbb{R}^d$ of radius $R$  containing ${\rm supp} \  \phi(\cdot)$. 
Arguing as in the proof of Lemma 4.1, in particular using the inequality (\ref{Q4}), we see that the integral in (\ref{AI4}) is bounded by $C/(1+|x|)^j$, whence we conclude that $|I_{2,j}(x)|\le C/(1+|x|)^d$ if $j\ge 1$

To  bound $I_{2,0}(x)$ we  write similarly to (\ref{N4}), 
\begin{equation} \label{AJ4}
  \frac{\partial^{d}}{\partial\xi_1^{d}} \left\{\frac{e(\xi+ai)e(\xi-ai)^* }{\eta+e(\xi-ai)^*q(\xi+ai,\eta)e(\xi+ai)} \right\}     \ = \  P\left(  \xi+ ai,\eta,\frac{\partial q(\xi+ai,\eta)}{\partial\xi_1}, \dots,      \frac{\partial^{d} q(\xi+ai,\eta)}{\partial\xi_1^{d}}                           \right)  \ .
\end{equation}
A typical term in the polynomial  of (\ref{AJ4}) has the form
\begin{equation} \label{AK4}
A(\xi+ ai,\eta)\prod_{j=1}^{d} \frac{\partial ^{r_j}q_{m_j,n_j}(\xi+ai,\eta)}{\partial\xi_1^{r_j}} \ , \quad {\rm with \ } r_1+\cdots+r_{d}=r \ \le  \ d \ ,
\end{equation}
where the $r_j$ are non-negative integers.  One also sees that
\begin{equation} \label{AL4}
|A(\xi+ai,\eta)| \ \le \ \frac{C}{|\xi|^{d-r}} \ , \quad \xi\in\mathbb{T}^d \ .
\end{equation}

We may bound the integral of  a term (\ref{AK4}) as before provided $r<d$. To do this we do a fractional integration by parts in the integral
\begin{equation} \label{AM4}
\frac{1}{(ix_1)^{d}}\int_{\mathbb{T}^d}e^{-ix\cdot\xi} \left\{1-\phi(\{1+|x|\}\xi)\right\}  A(\xi+ ai,\eta)\prod_{j=1}^{d} \frac{\partial ^{r_j}q_{m_j,n_j}(\xi+ai,\eta)}{\partial\xi_1^{r_j}}  \ d\xi  \ ,
\end{equation}
and proceed in a similar way to (\ref{S4})-(\ref{W4}).  Thus as in (\ref{S4}) we do a translation  by $\Delta\xi$  in the integrand of (\ref{AM4}) where $\Delta\xi\cdot x=\pi$.  The expression (\ref{AM4})  is half the value of 
\begin{multline} \label{AN4}
\frac{1}{(ix_1)^{d}}\int_{\mathbb{T}^d}e^{-ix\cdot\xi} \left\{\phi(\{1+|x|\}[\xi+\Delta\xi])-\phi(\{1+|x|\}\xi)\right\}  A(\xi+ ai,\eta)\prod_{j=1}^{d} \frac{\partial ^{r_j}q_{m_j,n_j}(\xi+ai,\eta)}{\partial\xi_1^{r_j}}  \ d\xi  \\

The integration in the  first term of (\ref{AN4}) is confined to  the region $\{|\xi|<M/(1+|x|)\}$ for some constant $M>20\pi$.   Since we may also choose $\Delta\xi$ to satisfy $|\Delta\xi|<\pi/(1+|x|)$,  we see that
$\phi(\{1+|x|\}[\xi+\Delta\xi])-\phi(\{1+|x|\}\xi)=0$ if $|\xi|<10\pi/(1+|x|)$.  Hence we have from (\ref{AL4}) that the first term in (\ref{AN4}) is bounded by

\end{equation}
Applying the inequality (\ref{Q4}) in (\ref{AO4}) and using Corollary 3.1 we conclude that the first integral in (\ref{AN4}) is bounded by $C/(1+|x|)^d$. 

 To bound the second term in (\ref{AN4}) we use the formula  (\ref{U4}), replacing $d-1$ by $d$.   We consider the contribution   from  $J_k, $ as in (\ref{V4}), with $k\ge1$.   Thus we need to estimate an integral,
\begin{equation} \label{AP4}
\frac{1}{(1+|x|)^d} \int_{|\xi|>1/(1+|x|)}  |J_k(\xi+ai,\eta,\Delta\xi)| \ \frac{d\xi}{|\xi|^{d-r}} \ . 
\end{equation}
We have from Corollary 3.1 that the function $\xi\rightarrow J_k(\xi+ai,\eta,\Delta\xi)/|\Delta\xi|^{1-\nu}$ is in $L^{d/(r+1-\nu)}_w(\mathbb{T}^d; \mathbb{C})$. It follows then from (\ref{Q4}) that  the expression in (\ref{AP4}) is bounded by
\begin{multline} \label{AQ4}
\frac{C}{(1+|x|)^{r+1-\nu}} \sum_{n\ge 1} 2^{-(d-r)n}\int_{2^{n-1}/(1+|x|)<|\xi|<2^n/(1+|x|)}  \frac{|J_k(\xi+ai,\eta,\Delta\xi)|}{|\Delta\xi|^{1-\nu}} \ d\xi \\
\le \ \frac{C}{(1+|x|)^d}\sum_{n\ge 0} 2^{-(1-\nu)n} \ \le \ \ \frac{C}{(1+|x|)^d} \ .
\end{multline}
The $J_0$ term may be similarly  bounded  following the representation   (\ref{W4}) and inequalities (\ref{X4})  with  $d-1$ replaced by $d$. 

Finally we need to deal with the case when $r=d$ in (\ref{AK4}). Thus we need to bound the integral
\begin{multline} \label{AR4}
I_3(x) \ = \ \frac{1}{(ix_1)^{d}}\int_{\mathbb{T}^d}e^{-ix\cdot\xi}  \left\{1-\phi(\{1+|x|\}\xi)\right\}   A(\xi+ ai,\eta)B(\xi+ai,\eta) \  d\xi \ , \\

\end{multline}
 We first estimate (\ref{AR4}) in the case $\phi(\cdot)\equiv0$.  Note that the function $\xi\rightarrow A(\xi,\eta)$ is of the form (\ref{D4}), whence  by Lemma 4.1 its Fourier inverse $x\rightarrow\hat{A}(x,\eta)$ satisfies an inequality (\ref{E4}).    We have then by Parseval's theorem that
 \begin{equation} \label{AS4}
 \frac{e^{a\cdot x}}{(2\pi)^d} I_3(x) \ = \ \frac{1}{(ix_1)^{d}}\sum_{y\in\mathbb{Z}^d} \hat{A}(y,\eta)\hat{B}(x-y,\eta)  \ .
 \end{equation}
 We also have from Lemma 4.3 that
 \begin{equation} \label{AT4}
| \hat{B}(y,\eta) | \ \le \   \frac{Ce^{-c_d\sqrt{\lambda\eta}|y|}}{(1+|y|)^{\beta}} \ , \quad y\in\mathbb{Z}^d \ ,  \ 0<\eta\le 1.
\end{equation}
The inequality (\ref{AT4})  follows immediately from Lemma 4.3 when $r_1=d$ in (\ref{AR4}),  and otherwise by estimating convolutions.  For the bound on the function $x\rightarrow\hat{A}(x,\eta)$ we choose $\nu$ in (\ref{E4}) such that $\nu<\beta$, to conclude that
\begin{equation} \label{AU4}
\left|                  \sum_{y\in\mathbb{Z}^d} \hat{A}(y,\eta)\hat{B}(x-y,\eta)           \right| \ \le \ Ce^{-c_d\sqrt{\lambda\eta}|x|} \ , \quad x\in\mathbb{Z}^d \ ,
\end{equation}
for some constant $C$. 

We also use the Parseval identity to estimate the term involving $\phi(\cdot)$ in (\ref{AR4}).  We have that 
\begin{multline} \label{AV4}
\frac{1}{(2\pi)^d}I_4(x) \ = \ \frac{1}{(2\pi ix_1)^{d}}\int_{\mathbb{T}^d}e^{-ix\cdot\xi} \phi(\{1+|x|\}\xi)   A(\xi+ ai,\eta)B(\xi+ai,\eta)  \ d\xi \\
= \ \frac{1}{(ix_1)^{d}}\sum_{y\in\mathbb{Z}^d}  \frac{1}{(1+|x|)^d} \hat{\phi}\left(\frac{x-y}{1+|x|}\right) \hat{C}(y,\eta,a) \ , \quad {\rm where \ } C(\xi,\eta,a)  \ = \  A(\xi+ ai,\eta)B(\xi+ai,\eta)  \ .
\end{multline}
We may estimate the function $y\rightarrow \hat{C}(y,\eta,a)$ similarly to (\ref{AU4}).  Choosing $|a|=c_d\sqrt{\lambda\eta}$ with $c_d>0$ sufficiently small, we see  from Lemma 4.1 and Lemma 4.3 that $|\hat{C}(y,\eta,a)|\le K/(1+|y|)^{\beta-\nu}, \ y\in\mathbb{Z}^d$, for some constant $K$.  Since $\hat{\phi}(\cdot)$ is a Schwarz function it is easy to see that the summation over $y\in\mathbb{Z}^d$  in (\ref{AV4}) is bounded, independent of $x\in\mathbb{Z}^d$. We conclude that $|I_4(x)|\le C/(1+|x|)^d, \ x\in\mathbb{Z}^d$, for some constant $C$.

<!-- theorem-3: body_md -->
{\rm (weighted Calderon-Zygmund)}
For $1<p<\infty, \ 0<\eta\le 1,$ the operator $S_\eta$ is bounded on $\ell_{p,k}(\mathbb{Z}^d;\mathbb{C}^d)$ provided $kp'<d$.  The operator norm satisfies an inequality
\begin{equation} \label{M5}
\left\| S_\eta \ \right\|_{\ell_{p,k}(\mathbb{Z}^d;\mathbb{C}^d)} \ \le \  \kappa_{p,k} \ ,
\end{equation}
where $\kappa_{p,k}\ge1$ is independent of $\eta>0$.

<!-- corollary-3: body_md -->
For $R\ge 1$ let $\chi_R:\mathbb{Z}^d\rightarrow\{0,1\}$ be the characteristic function of the set $\{x\in\mathbb{Z}^d: \ |x|<R\}$. The convolution  operator $S_\eta^R$  on functions $f:\mathbb{Z}^d\rightarrow\mathbb{C}^d$  with kernel $\nabla\nabla^*G_\eta(\cdot)\chi_R(\cdot)$ is bounded on $\ell_{p,0}(\mathbb{Z}^d;\mathbb{C}^d)$  and 
\begin{equation} \label{O5}
\left\| S_\eta^R \ \right\|_{\ell_{p,0}(\mathbb{Z}^d;\mathbb{C}^d)} \ \le \  \kappa_{p,0} +\kappa \ ,
\end{equation}
where the constant $\kappa$ is independent of $p$ and also $\eta>0,R\ge 1$.

We also see  upon using  Theorem 5.1 that $T_{K,t,T}$ is a  bounded operator on $\ell_{p,k}(\mathbb{Z}^d;\mathbb{R}^d)$ with norm satisfying $\left\|         T_{K,t,T}            \right\|_{\ell_2(\mathbb{Z}^d;\mathbb{R}^d)}\le\exp[\kappa_{p,k}(1-\lambda)(T-t)/2]$. 
Since $\kappa_{2,0}=1$ we conclude from continuity of norms that there is a $k>0$ such that $\kappa_{2,k}(1-\lambda)<1$. In that case the RHS of (\ref{M6}) remains bounded as $T\rightarrow\infty$.  Since $\omega(\cdot,T)$ converges in distribution to $\omega(\cdot)$ as $T\rightarrow\infty$ we obtain the formula (\ref{L6}), with the kernel $K$ satisfying the inequality in the statement of the Lemma.
