\label{prop:wrapped_orth}
Let polynomials ${\bm Q}_{{\bm j}, {\bm k}, m}^n$ of $d_1+d_2$ variables be defined by
\begin{equation}\label{eq:wrapOP}
   {\bm Q}_{{\bm j}, {\bm k}, m}^n({\bm x}, {\bm y}) = {\bm R}_{{\bm j},n-m}^{(m)}({\bm y}) [\rho({\bm y})]^m {\bm P}_{{\bm k},m}\left(\frac{{\bm x}}{\rho({\bm y})}\right),  
\end{equation}
where $0 \le m \le n$, $|{\bm k}| = m$ with ${\bm k} \in {\mathbb N}_0^{d_1}$, and $|{\bm j}| = n-m$ with ${\bm j} \in {\mathbb N}_0^{d_2}$. 
Then the collection $\left\{  {\bm Q}_{{\bm j}, {\bm k}, m}^n: |{\bm k}| = m,\,  |{\bm j}| = n-m, 0 \le m \le n\right\}$ is an orthogonal basis 
of the space ${\mathcal V}_n\big(\Omega_1^{d_1}  \rtimes \Omega_2^{d_2}, {\bm W}^{\Omega_1\rtimes \Omega_2}\big)$.

For $\mu > - \frac12$, ${\boldsymbol{\kappa}} \in {\mathbb R}^{d_2}$ with ${\kappa}_j > -1$, $1 \le j \le d_2$, define 
\begin{equation} \label{eq:Qbasis}
 \bm Q_{\bm j, \bm k, m}^n(\bm x,\bm y) = \bm T_{\bm j, n-m}^{({\boldsymbol{\kappa}}, \alpha+2m)}(\bm y) (1-|\bm y|)^m 
      \bm B_{\bm k, m}^\mu \left(\frac{\bm x}{1-|\bm y| }\right),

<!-- thm-1: body_md -->
\label{thm:QOPd1=1}
The linear transformation $(x, \bm y) \in [-1,1] \rtimes \triangle^{d} \mapsto \bm u \in \triangle^{d+1}$ defined by
$$
  \bm u = \phi(x, \bm y) = \left(y_1, \ldots, y_{d-1}, \, \tfrac12(1-|\bm y|-x),  \tfrac12(1-|\bm y|+x) \right)
$$
is a bijection between $[-1,1]\rtimes \triangle^{d_2}$ and  $\triangle^{d+1}$. Moreover, under this linear transformation, 
the weight function $\bm W_{\mu,{\boldsymbol{\kappa}}}^{{\mathbb B}\rtimes \triangle}(x, \bm y)$ with ${\boldsymbol{\kappa}} = ({\kappa}_1,\ldots, {\kappa}_d)$ becomes
$$
    \bm W_{\bm {\beta}}^\triangle ({\bm u}) = 2^{2\mu-1} u_1^{{\kappa}_1} \cdots u_{d-1}^{{\kappa}_{d-1}} 
      (1-|\bm u|)^{{\kappa}_d} u_d^{\mu -\frac12} u_{d+1}^{\mu -\frac12}.
$$
In particular, let $\{\bm T_{\bm j, n}^{\bm {\beta}}: |\bm j|  = n, \, \bm j \in {\mathbb N}_0^{d+1}\}$ be an orthogonal basis
of ${\mathcal V}_n(\triangle^{d+1}, \bm W^\triangle_{\bm {\beta}})$ on the simplex $\triangle^{d+1}$; then
$$
   \bm Q_{\bm j, n}^{\mu, {\boldsymbol{\kappa}}} (x,\bm y) = \bm T_{\bm j, n}^{\bm {\beta}}\left(y_1, \ldots,  y_{d-1}, 
       \tfrac12(1-|\bm y|-x),  \tfrac12(1-|\bm y|+x) \right)
$$
for $|\bm j| = n$ consists of an orthogonal basis for 
${\mathcal V}_n\big([-1,1]\rtimes \triangle^d, \bm W_{\mu, {\boldsymbol{\kappa}}}^{{\mathbb B}\rtimes \triangle}\big)$.

<!-- proof-1: body_md -->
The inverse of the linear transform ${\bm u} \mapsto (x, \bm y)$ is easily seen as given by 
$$
  x= u_{d+1}-u_d, \, y_1 = u_1, \ldots, y_{d-1} = u_{d-1}, \, y_d = 1-|\bm u|.
$$
It follows that $u_{d+1}-u_d = x$ and $u_d+u_{d+1} = 1 - |\bm y|$. In particular, $|x| \le 1-|\bm y|$ for 
$(x, \bm y) \in {\mathbb B}^{1} \! \rtimes \triangle^{d_2}$ 
becomes $|u_d- u_{d+1}| \le u_d+u_{d+1}$, which is equivalent to $u_d \ge 0$ and $u_{d+1} \ge 0$. Moreover, $1-|\bm u| = y_d$, 
so that $(x,\bm y)$ is mapped to $\bm u \in \triangle^{d+1}$. Moreover, the Jacobian $\partial (x, \bm y)/\partial \bm u = 2$, which
shows that the linear transformation is a bijection. Furthermore, 
$$
  \bm W_{\mu,\bm {\kappa}}^{[-1,1]\rtimes \triangle}(x, \bm y) = \prod_{i=1}^d y_i^{{\kappa}_i} (1-|\bm y|-x)^{\mu-\frac12} (1-|\bm y|+x)^{\mu-\frac12} 
     = \bm W_{\bm {\beta}}^\triangle ({\bm u})  
$$
with $\bm u \in \triangle^{d+1}$. Since the affine transformation preserves orthogonality, an orthogonal basis for 
${\mathcal V}_n\big({\mathbb B}^1 \rtimes \triangle^{d_2}, \bm W_{\mu,{\boldsymbol{\kappa}}}^{{\mathbb B}\rtimes \triangle}\big)$ can be derived from 
that of ${\mathcal V}_n(\triangle^{d+1}, \bm W^\triangle_{\bm {\beta}})$.

For $\mu \ge 0$ and ${\boldsymbol{\kappa}} =({\kappa}_1,\ldots, {\kappa}_d)$ with ${\kappa}_i \ge -\frac12$, 
\begin{align} \label{eq:P=IntP}
  & \bm P_n^{d_1,d_2} \big(\bm W_{\mu, {\boldsymbol{\kappa}}}^{{\mathbb B}\rtimes \triangle}; (\bm x,\bm y), (\bm u, \bm v)\big)  \\
   = &\, c_{\mu-\frac12} \int_{-1}^1 
     \bm P_n^{1,d_2}\left(\bm W_{\mu+\frac{d_1 -1}{2}, {\boldsymbol{\kappa}}}^{[-1,1]\rtimes \triangle}; \big(\zeta (\bm x,\bm y,  \bm u,\bm v; s),\bm y\big), \big(1-|\bm v|, \bm v\big) \right) 
  (1-s^2)^{\mu-1} \mathrm{d} s, \notag 
\end{align} 
where  
$$
\zeta (\bm x,\bm y,  \bm u,\bm v; s) = \frac{{\langle} \bm x, \bm u {\rangle} + s \sqrt{(1-|\bm y|)^2 -\|\bm x\|^2}  \sqrt{(1-|\bm v|)^2 -\|\bm u\|^2}}{1-|\bm v|},
$$
and $|\zeta (\bm x,\bm y,  \bm u,\bm v; s)|\le 1$ for $(\bm x, \bm y), (\bm u, \bm v) \in {\mathbb B}^{d_1} \!\rtimes \triangle^{d_2}$ 
and $s \in [-1,1]$. Moreover,  the identity holds in the limit \eqref{eq:limitInt} if $\mu = 0$.

In terms of the orthogonal polynomials \eqref{eq:Qbasis} and their norms \eqref{eq:Q_OPNorm}, and the definition of 
the reproducing kernel on the unit ball, 
\begin{align*}                      
\bm P_n^{d_1,d_2} \left(\bm W_{\mu, {\boldsymbol{\kappa}}}^{{\mathbb B}\rtimes \triangle}; (\bm x,\bm y), (\bm u, \bm v)\right) & = \sum_{m=0}^n \frac{(|{\boldsymbol{\kappa}}|+ 2\mu+ d_1 + d_2)_{2m}}{(2\mu+d_1)_{2m}} \frac1{\bm h_{\bm k,m}^{\mu,{\mathbb B}}  
       \bm h_{\bm j,m}^{({\boldsymbol{\kappa}}, {\alpha}+2m),\triangle}} \\
 & \times  \sum_{\bm j = n-m} \bm T_{\bm j, n-m}^{({\boldsymbol{\kappa}}, 2m +2\mu+d_1-1)}(\bm y) 
      \bm T_{\bm j, n-m}^{({\boldsymbol{\kappa}}, 2m+2\mu+d_1-1)}(\bm v) \\
 & \times  (1-|\bm y|)^m (1-|\bm v|)^m \bm P_m\left(W_\mu^{\mathbb B}; \frac{\bm x}{1-|\bm y|}, \frac{\bm u}{1-|\bm v|} \right). \notag 
\end{align*}
Hence, by the addition formula \eqref{eq:add-ball}, we obtain
\begin{align}\label{eq:P=IntP2}
& \bm P_n^{d_1,d_2}  \left(\bm W_{\mu, {\boldsymbol{\kappa}}}^{{\mathbb B}\rtimes \triangle}; (\bm x,\bm y), (\bm u, \bm v)\right)
  = \sum_{m=0}^n \frac{(|{\boldsymbol{\kappa}}|+ 2\mu+ d_1 + d_2)_{2m}}{(2\mu+d_1)_{2m}} \frac{1}{ \bm h_{\bm j,m}^{({\boldsymbol{\kappa}}, {\alpha}+2m),\triangle}} \\
 &\qquad \times  \sum_{\bm j = n-m} \bm T_{\bm j, n-m}^{({\boldsymbol{\kappa}}, 2m +2\mu+d_1-1)}(\bm y) 
      \bm T_{\bm j, n-m}^{({\boldsymbol{\kappa}}, 2m+2\mu+d_1-1)}(\bm v) \notag \\
 & \qquad \times  (1-|\bm y|)^m (1-|\bm v|)^m c_\mu \int_{-1}^1 Z_m^{\mu+\frac{d_1-1}{2}}

  (1-s^2)^{\mu-1} \mathrm{d} s. \notag 
\end{align}

where we have used the relation 
$$

Setting $\mu \to \mu + \frac{d_1-1}{2}$ and $x = \zeta (\bm x,\bm y, \bm u,\bm v; s)$ in \eqref{eq:reprodPd=1},

\label{thm:addition_BT}
Let $\mu \ge 0$ and ${\kappa}_i \ge -\frac12$, $1\le i \le d_2$. For $(\bm x, \bm y)$ and $(\bm u, \bm v) \in {\mathbb B}^{d_1} \! \rtimes \triangle^{d_2}$,

  \bm P_n^{d_1,d_2}  \big(\bm W_{\mu, {\boldsymbol{\kappa}}}^{{\mathbb B}\rtimes \triangle},  (\bm x, \bm y), (\bm u, \bm v) \big) 
    = & \, c_{\mu,{\boldsymbol{\kappa}}} \int_{[-1,1]^{d_2+2}}   Z_{2n}^{|{\boldsymbol{\kappa}}| + 2\mu + d_1+d_2-1}
    \big( \xi(\bm x, \bm y, \bm u, \bm v; \bm s, \bm t)\big) \notag \\
   &\times \prod_{i=1}^{d_2} (1-t_i^2)^{{\kappa}_i-\frac12}  \prod_{i=1}^2 (1-s_i^2)^{\mu - 1}  \mathrm{d} \bm t\, \mathrm{d} \bm s,
\end{align}
where $c_{\mu, {\boldsymbol{\kappa}}} =  c_{{\kappa}_1}\cdots c_{{\kappa}_{d_2}} (c_{\mu-\frac12})^{2} $ and
\begin{align*}
& \xi(\bm x, \bm y, \bm u, \bm v; \bm s, \bm t) =  \sqrt{y_1 v_1} \, t_1 + \cdots + \sqrt{y_{d_2} v_{d_2}} \, t_{d_2}  \\ 
   & \quad + \sqrt{\frac12 \left( {\langle} \bm x, \bm u {\rangle}+(1-|\bm y|)(1-|\bm v|) +  s_1 \sqrt{(1-|\bm y|)^2-\|\bm x\|^2}\sqrt{(1-|\bm v|)^2-\|\bm u\|^2} \right)} \, s_2.
\end{align*}
Moreover, \eqref{eq:addition} holds under the limit \eqref{eq:limitInt} if one or more ${\kappa}_i= -\frac12$ and/or $\mu = 0$.

For $\mu \ge 0$ and ${\kappa}_i \ge -\frac12$, let $ {\alpha} = |{\boldsymbol{\kappa}}| + 2\mu + d_1+d_2-1$. For

\begin{equation} \label{eq:convolu}
 f\ast g (\bm x,\bm y) = \int_{{\mathbb B}^{d_1} \!\rtimes \triangle^{d_2}} f(\bm u,\bm v) T^{\mu,{\boldsymbol{\kappa}}} g\big ( (\bm x,\bm y), (\bm u,\bm v) \big)

\end{equation}
where the operator $g \mapsto T^{\mu,{\boldsymbol{\kappa}}} g$ is defined by 
\begin{align*}
T^{\mu,{\boldsymbol{\kappa}}} g \big ( (\bm x,\bm y), (\bm u,\bm v) \big)
 =  & \, \bm b_{\mu,{\boldsymbol{\kappa}}} \int_{[-1,1]^{d+1}} \int_{-1}^1 g \big( 2 \xi(\bm x, \bm y, \bm u, \bm v; \bm s, \bm t)^2-1 \big) \notag \\

\end{align*}
in which $\xi(\bm x, \bm y, \bm u, \bm v; \bm s, \bm t)$ is defined as in the addition formula \eqref{eq:addition}.

<!-- lem-1: body_md -->
\label{lem:translateT}
Let $g \in L^1([-1,1],w_{{\alpha}, - \frac12})$. Then, for each

By its definition, ${\mathsf d}_{{\mathbb S}\rtimes \triangle} (\cdot,\cdot)$ is symmetric in its variables.

by the Cauchy-Schwarz inequality, so that ${\mathsf d}_{{\mathbb S}\rtimes \triangle} \big((\bm x, \bm y), (\bm u,\bm v) \big) \ge 0$ for all

of ${\mathsf d}_{{\mathbb S}\rtimes \triangle}$ follows from that of ${\mathsf d}_{\mathbb S}$. This completes the proof.
