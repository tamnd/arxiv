\item ${\mathcal  F}$ is injective and its image is the open disk of radius  $\frac\pi4$: ${\mathcal  F}(\mathbb{R}^2)=\big\{|\xi|<\frac\pi4\big\}.$

\end{itemize}

We first compute the Jacobian matrix at $\xi=0$. 
The mixed terms vanish:
$$
\partial_{\xi_1}{\mathcal  F}_2(0)=\partial_{\xi_2}{\mathcal  F}_1(0)=\int_{\mathbb{R}^2} z_1 z_2 |z|^{-1} (1+|z|^2)^{-3} dz = 0,
$$
since the integrand is odd with respect to $z_1$ or $z_2$.
Moreover, by rotational symmetry,
$$
\partial_{\xi_1}{\mathcal  F}_1(0)=\partial_{\xi_2}{\mathcal  F}_2(0)=\frac{1}{2} \int_{\mathbb{R}^2} |z| \frac{1}{(1+|z|^2)^3}\,dz$$
and we immediately get $${\rm det}\, D{\mathcal  F}(0)>0.$$
For every rotation $\theta\in\mathbb{R}$, performing the change of variables $z \mapsto e^{{\rm i}\theta}z$ gives
$$
\mathcal{F}(e^{{\rm i}\theta}\xi) = e^{{\rm i}\theta} \mathcal{F}(\xi).
$$
Hence $\mathcal{F}$ is rotation-equivariant, and therefore there exists a scalar function $\phi:(0,\infty)\to\mathbb{R}$ such that
\begin{equation}\label{fib}
\mathcal{F}(\xi) = \phi(|\xi|)\,\xi.
\end{equation}
By choosing $\xi=(r,0)$ in \eqref{fib}, we write
$$
\phi(r)
=
\frac{1}{r}
\int_{\mathbb{R}^2}
|z+(r,0)|\,\frac{z_1}{(1+|z|^2)^3}\,dz.
$$
Split $\mathbb{R}^2$ into the regions $\{z_1>0\}$ and $\{z_1<0\}$, and perform the change of variables $z\mapsto -z$ in the second  integral. This yields
$$
\phi(r)
=
\frac{1}{r}
\int_{z_1>0}
\frac{z_1}{(1+|z|^2)^3}
\left(
|z+(r,0)| - |z-(r,0)|
\right) dz\quad \forall r>0.
$$
If $z_1>0$, then $|z+(r,0)| > |z-(r,0)|$. 
Hence
$$
\phi(r) > 0 \quad \text{for all } r>0.
$$
Differentiating under the integral sign (justified by dominated convergence), we obtain
$$
\phi(r)+r\phi'(r)=\frac{d}{dr}\big(r\phi(r)\big)
=
\int_{\mathbb{R}^2}
|z+(r,0)|^{-1}
\frac{(z_1+r)z_1}{(1+|z|^2)^3}
\,dz.
$$
Repeating the same symmetry argument as before, this can be rewritten as
$$\phi(r)+r\phi'(r)=
\int_{z_1>0}
\frac{z_1}{(1+|z|^2)^3}
\Big[
|z+(r,0)|^{-1}(z_1+r)
+
|z-(r,0)|^{-1}(z_1-r)
\Big] dz.
$$
Since   $|z+(r,0)|^{-1}(z_1+r)
>
|z-(r,0)|^{-1}(r-z_1)$ if $z_1>0$,  the term in brackets is  positive. Hence  \begin{equation}\label{moment}(r\phi(r))'=\phi(r)+r\phi'(r) > 0.\end{equation} Then by \eqref{fib}  $$
\det D\mathcal{F}(\xi)
=
\phi(|\xi|)\,(\phi(|\xi|)+|\xi|\phi'(|\xi|))
>0
\quad \text{for all } \xi \neq 0.
$$
 
Consequently $|{\mathcal  F}(\xi)|=\phi(|\xi|)|\xi|$ is  strictly increasing in $|\xi|$ by \eqref{moment}. Hence, the rotational equivariance of ${\mathcal  F}$ implies that ${\mathcal  F}$ is injective. 
Moreover, using integration by parts,  $$\begin{aligned}
r\phi(r)&=\int_{\mathbb{R}^2} |z+(r,0)|\frac{z_1}{(1+|z|^2)^3} dz=-\frac14\int_{\mathbb{R}^2} |z+(r,0)|\partial_{z_1}\bigg(\frac{1}{(1+|z|^2)^2}\bigg) dz\\ &=\frac14\int_{\mathbb{R}^2} \frac{z_1+r}{|z+(r,0)|}\frac{1}{(1+|z|^2)^2} dz\longrightarrow  \frac14\int_{\mathbb{R}^2}\frac{1}{(1+|z|^2)^2}dz=\frac\pi4\end{aligned}$$ as $r\to +\infty$, and  therefore
$$\{r\phi(r)\,|\, r\geq 0\}=\Big[0,\frac\pi4\Big).$$  
Finally we  derive $$|F(\xi)|=|\xi|\phi(|\xi|)\longrightarrow  \frac\pi4\hbox{ as }|\xi|\to +\infty$$
and 
the characterization of the image of ${\mathcal  F}$ follows immediately   $${\mathcal  F}(\mathbb{R}^2)=\{ r\phi(r) e^{{\rm i}\theta}\,|\,r\geq 0 , \,\theta\in \mathbb{R}\}=\Big\{\xi\,\Big|\,|\xi|< \frac\pi4\Big\}.$$

Let $\omega = e^{\frac{2\pi i}{N+1}}$. By the invariance of $\zeta$ under the rotation $y \mapsto \omega^k y$, we have
$$
\int_{\mathbb{R}^2} \zeta(y)\, y^\ell \, dy
= \int_{\mathbb{R}^2} \zeta(\omega^k y)\, (\omega^k y)^\ell \, dy
= \omega^{k\ell} \int_{\mathbb{R}^2} \zeta(y)\, y^\ell \, dy.
$$
Averaging over $k=0,\dots,N$ gives
$$
\int_{\mathbb{R}^2} \zeta(y)\, y^\ell \, dy
= \frac{1}{N+1} \sum_{k=0}^N \omega^{k\ell}
\int_{\mathbb{R}^2} \zeta(y)\, y^\ell \, dy.
$$

$$
\sum_{k=0}^N \omega^{k\ell} = 0 \quad \text{if } \frac{\ell}{N+1} \notin \mathbb{N},
$$
it follows that
$$
\int_{\mathbb{R}^2} \zeta(y)\, y^\ell \, dy = 0.
$$
Taking real and imaginary parts yields the claim.
