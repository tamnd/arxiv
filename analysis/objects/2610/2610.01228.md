\label{def:finite_rate}    
Let $m,n\in \mathbb{N}$ and $L:\mathbb{R}^{+}\times\mathbb{N}\to\mathbb{R}^{+}$
be a rate function. The computation rate of an $(n,m,\mathcal F_m,\lambda_1,\lambda_2)$ BFC code is $R$ with the rate function $L$ if $m =L(R,n)$.

<!-- proposition-1: body_md -->
\label{prop:finite_error}
For $0\leq S\leq 2^m$ and $n_t\geq 2$, consider the RS tagging construction in Subsection~\ref{sec:tagging_code} based on a $(T,K)$ RS code over $\mathbb{F}_q$, satisfying $1\leq K\leq T\leq q$ and $n_t=\lceil\log T\rceil+r$. Then, the construction is an
\begin{equation}
    \left(
        n_t,m,\mathcal F_m(\leq S),
        0,
        \min\left\{1,\frac{S(K-1)}{T}\right\}
    \right)
\end{equation}
BFC code over the noiseless binary channel.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Suppose $f(i)=1$, then the tag $c_u(i)\in D_f(u)$ for all $u\in[T]$ by definition of \eqref{equ:symbol_decoding_region}. Therefore, the decoder always declares $\hat{f}(a)=1$ given $a=(u,c_u(i))$, and FN probability is zero.

    Suppose $f(i)=0$, the FP occurs if there exists a message $j\neq i$ and $j\in f^{-1}(1)$ such that
    \begin{equation}
        \label{equ:collision}
        P_i(\alpha_u) = P_j(\alpha_u).
    \end{equation}
    Thus, the FP probability for a pair $(i,f)$ is given by
    \begin{equation}
        \label{equ:fp}
        \begin{split}
            P_{\rm FP}(i,f) & = \Pr\{\hat f(a(i))=1|i,f:f(i)=0\}  \\
            & = \frac{1}{T}\sum_{u=1}^T\1\{\exists\,j\in f^{-1}(1): P_i(\alpha_u) = P_j(\alpha_u)\} \\
            & = \frac{1}{T}\sum_{u=1}^T\1\left\{\bigcup_{j\in f^{-1}(1)}\{P_i(\alpha_u) = P_j(\alpha_u)\}\right\} \\
            & \leq  \frac{1}{T}\sum_{u=1}^T\sum_{j\in f^{-1}(1)}\1\{P_i(\alpha_u) = P_j(\alpha_u)\}
        \end{split}
    \end{equation}
    The property of $(T,K)$ RS code gives: for all $j\neq i$, the difference $P_i(x)-P_j(x)$ is a non-zero polynomial of degree at most $K-1$. Therefore, $\sum_{u=1}^T\1\{P_i(\alpha_u) = P_j(\alpha_u)\}\leq K-1$, yielding
    \begin{equation}
        \begin{split}
            P_{\rm FP}(i,f) & \leq \sum_{j\in f^{-1}(1)}\frac{K-1}{T} \\
            & \leq |f^{-1}(1)|\frac{K-1}{T}\\
            &= \frac{S_f(K-1)}{T} \\
            & \leq \frac{S(K-1)}{T}.
        \end{split}
    \end{equation}
    Combining with $P_{\rm FP}\leq 1$ and taking the maximum over all $(i,f)$ yields
    \begin{equation}
        \label{equ:fp_bound}
        P_{\rm FP} = \max_{i,f:f(i)=0}P_{\rm FP}(i,f)\leq\min\left\{1,\frac{S(K-1)}{T}\right\}.
    \end{equation}

<!-- theorem-1: body_md -->
\label{thm:finite_E2}
    Let $S:\mathbb{N}\to\mathbb{N}$ be non-decreasing and define
    \begin{equation}
        \label{equ:rate_function}
        L_S(R,n):=\max\left\{m\in\mathbb{N}:mS(m)\leq\left\lceil\frac{n}{2}\right\rceil2^{\left\lfloor \frac{n}{2}\right\rfloor}2^{n\left(R-\frac{1}{2}\right)}\right\}.
    \end{equation}
    For all $E_2\in(0,1/2)$ and $n_t\in\mathbb{N}$ with $n_t\geq 2$ such that the set in \eqref{equ:rate_function} is non-empty, set $r=\lceil n_t/2\rceil$, $T=2^{\lfloor n_t/2\rfloor}$, and $m=L_S(1/2-E_2,n_t)$. This yields an
    \begin{equation}
        (n_t, m, \mathcal F_m(\leq S(m)), 0, 2^{-n_tE_2})
    \end{equation}
    BFC code and the finite-blocklength computation rate of this code is $1/2-E_2$ with the rate function $L_S$.
<!-- end theorem-1 -->
<!-- proof-2: body_md -->
By the definition of \eqref{equ:rate_function} and $E_2\in(0,1/2)$, the selected $m$ satisfying
    \begin{equation}
        mS(m)\leq rT 2^{-n_tE_2} < rT.
    \end{equation}
    Since $S(m)\geq 1$, this implies that $m<rT$ and thus $K=\lceil m/r\rceil\leq T$
    With $q=2^r$, the selected $r=\lceil n_t/2\rceil$ and $T=2^{\lfloor n_t/2\rfloor}$ give $T\leq q$. Therefore, these selected RS parameters satisfying $1\leq K\leq T\leq q$ and resulting an RS tagging code of length
    \begin{equation}
        \lceil \log T\rceil + r = \lfloor n_t/2\rfloor + \lceil n_t/2\rceil = n_t.
    \end{equation}
    
    Since $K-1\leq m/r$, Proposition~\ref{prop:finite_error} gives
    \begin{equation}
        \begin{split}
            P_{\rm FP} & \leq \frac{S(m)(K-1)}{T} \\
            & \leq \frac{mS(m)}{rT} \\
            & \leq \frac{rT}{rT}2^{-n_tE_2} \\
            & = 2^{-n_tE_2}.
        \end{split}
    \end{equation}

    The finite-blocklength computation rate of $1/2-E_2$ follows directly from Definition~\ref{def:finite_rate}.

<!-- corollary-1: body_md -->
\label{cor:achievable}
    For all $\lambda_1,\lambda_2>0$ and $\eta\in(0,1/2)$, and for all large enough $n_t$, the RS tagging code construction in Subsection~\ref{sec:tagging_code} gives an $(n_t, m, \mathcal F_m(\leq S(m)), \lambda_1,\lambda_2)$ BFC code with $m=L_S(1/2-\eta,n_t)$. Thus, for all small error probabilities, the code supports an asymptotic computation rate arbitrarily close to $1/2$ with the rate function $L_S$ in \eqref{equ:rate_function}.

<!-- corollary-2: body_md -->
\label{cor:weight_regimes}
    Let $S^{\leftarrow}(x):=\max\{m\in\mathbb{N}:S(m)\leq x\}$. Under the code construction in Subsection~\ref{sec:tagging_code}, as $m\to\infty$
    \begin{itemize}
        \par\hangindent 40\p@ if $\log S(m)/\log m\to a\geq 0$, then $1/[2(1+a)]$ is an achievable computation rate with the rate function $L_{\rm small}(R,n)=2^{Rn}$.
        \par\hangindent 40\p@ if $\log S(m)/\log m\to \infty$, then $1/2$ is an achievable computation rate with the rate function $L_{\rm medium}(R,n)=S^{\leftarrow}(2^{Rn})$.
        \par\hangindent 40\p@ if $\log S(m)/m\to 1$, then $1/2$ is an achievable computation rate with the rate function $L_{\rm large}(R,n)=Rn$.
    \end{itemize}

<!-- proof-4: body_md -->
By Corollary~\ref{cor:achievable}, for all $\delta\in(0,1/2)$ and sufficiently large $n$, there exists a BFC code based on the RS tagging construction that supports a message length of at least $L_S(1/2-\delta,n)$ with arbitrarily small error probabilities. 
    By Corollary~\ref{cor:achievable}, to prove a rate $R_0$ is achievable with a rate function $L_\star$, i.e., the code supports computation rates arbitrarily close to $R_0$ with arbitrarily small error probabilities, it is sufficient to show that for all $\eta\in(0,R_0)$ and sufficiently large $n$
    \begin{equation}
        \label{equ:proof_achievability}
        L_\star(R_0-\eta,n)\leq L_S\left(\frac{1}{2}-\delta,n\right).
    \end{equation}

    {\bf Small regime}: When $\log S(m) / \log m \to a$ for $a\geq 0$, consider
    \begin{equation}
        R_0 = \frac{1}{2(1+a)}\quad \text{and}\quad L_{\rm small}(R,n)=2^{Rn}.
    \end{equation}
    For some $\eta\in(0,R_0)$, let
    \begin{equation}
        \tilde m = \left\lceil 2^{n(R_0-\eta)}\right\rceil \geq L_{\rm small}(R_0-\eta,n).
    \end{equation} 
    Since
    \begin{equation}
        \frac{\log\tilde m}{n} \to R_0-\eta\quad \text{and} \quad \frac{\log S(m)}{\log \tilde m} \to a,
    \end{equation}
    we have
    \begin{equation}
        \begin{split}
            \lim_{n\to\infty}\frac{1}{n}\log(\tilde mS(\tilde m)) & = \lim_{n\to\infty}\frac{\log \tilde m}{n}\left(1 + \frac{\log S(\tilde m)}{\log \tilde m}\right) \\
            & = (R_0-\eta)(1+a) \\
            &  = \frac{1}{2}-(1+a)\eta.
        \end{split}
    \end{equation}
    Choose any $0<\delta<(1+a)\eta$, for sufficiently large $n$ we have
    \begin{equation}
        \tilde mS(\tilde m)\leq 2^{n(1/2-\delta)}\leq \frac{n}{2}2^{n(1/2-\delta)}
    \end{equation}
    verifying \eqref{equ:proof_achievability}. Thus, rate $\frac{1}{2(1+a)}$ is achievable with the rate function $L_{\rm small}(R,n)=2^{Rn}$.


    {\bf Medium regime}: When $\log S(m) / \log m\to \infty$, consider
    \begin{equation}
        R_0 = \frac{1}{2}\quad \text{and}\quad L_{\rm medium}(R,n)=S^{\leftarrow}(2^{Rn}).
    \end{equation}
    For some $\eta\in(0,R_0)$, let
    \begin{equation}
        \tilde m = S^{\leftarrow}(2^{n(R_0-\eta)}) \geq L_{\rm medium}(R_0-\eta,n).
    \end{equation} 
    Since
    \begin{equation}
        \frac{\log\tilde m}{\log S(\tilde m)} \to 0\quad\text{and}\quad \frac{\log S(\tilde m)}{n} \leq R_0-\eta,
    \end{equation}
    we have,
    \begin{equation}
        \begin{split}
            \lim_{n\to\infty}\frac{1}{n}\log(\tilde mS(\tilde m)) & = \lim_{n\to\infty}\frac{\log S(\tilde m)}{n}\left(1 + \frac{\log\tilde m}{\log S(\tilde m)}\right) \\
            & \leq (R_0-\eta)(1+0) \\
            & = \frac{1}{2}-\eta.
        \end{split}
    \end{equation}
    Choose any $0<\delta<\eta$, for sufficiently large $n$ we have
    \begin{equation}
        \tilde mS(\tilde m)\leq 2^{n(1/2-\delta)}\leq \frac{n}{2}2^{n(1/2-\delta)}
    \end{equation}
    verifying \eqref{equ:proof_achievability}. Thus, rate $\frac{1}{2}$ is achievable with the rate function $L_{\rm medium}(R,n)=S^{\leftarrow}(2^{Rn})$.

    {\bf Large regime}: When $\log S(m) / m \to 1$, consider
    \begin{equation}
        R_0 = \frac{1}{2}\quad \text{and}\quad L_{\rm larage}(R,n)=Rn.
    \end{equation}
    For some $\eta\in(0,R_0)$, let
    \begin{equation}
        \tilde m = \lceil n(R_0-\eta)\rceil \geq L_{\rm large}(R_0-\eta,n).
    \end{equation} 
    Since
    \begin{equation}
        \frac{\log\tilde m}{n} \to 0\quad, \frac{\log S(m)}{m} \to 1,\quad  \text{and}\quad \frac{\tilde m}{n}\to R_0-\eta,
    \end{equation}
    we have,
    \begin{equation}
        \begin{split}
            \lim_{n\to\infty}\frac{1}{n}\log(\tilde mS(\tilde m)) & = \lim_{n\to\infty}\left(\frac{\log \tilde m}{n} + \frac{\log S(\tilde m)}{\tilde m}\frac{\tilde m}{n}\right) \\
            & = 0 + 1\cdot (R-\eta) \\
            & = \frac{1}{2}-\eta.
        \end{split}
    \end{equation}
    Choose any $0<\delta<\eta$, for sufficiently large $n$ we have
    \begin{equation}
        \tilde mS(\tilde m)\leq 2^{n(1/2-\delta)}\leq \frac{n}{2}2^{n(1/2-\delta)}
    \end{equation}
    verifying \eqref{equ:proof_achievability}. Thus, rate $\frac{1}{2}$ is achievable with the rate function $L_{\rm large}(R,n)=Rn$.

<!-- proposition-2: body_md -->
\label{prop:noisy_finite_error}
    Consider the packed concatenated RS tagging construction that concatenates $G$ noiseless BFC codewords in Proposition~\ref{prop:finite_error} using an $(N_b,N_i,\xi)$ channel code. For each function evaluation, the construction is an 
    \begin{equation}
        \left(n_{\rm eff},m,\mathcal F_m(\leq S),\xi,\min\left\{1,\frac{S(K-1)}{T}+\xi\right\}\right),
    \end{equation}
    effective BFC code, where $n_{\rm eff}$ is defined in \eqref{equ:eff_channel_use}
<!-- end proposition-2 -->
<!-- proof-5: body_md -->
Let $\mathcal{E}:=\{\hat V\neq V\}$ denote the channel decoding error event. The $(N_b,N_i,\xi)$ channel code definition implies $\Pr\{\mathcal{E}\}\leq\ \xi$. 
    We fix an arbitrary slot $g\in[G]$. Suppose $f_g(i_g)=1$, a correct channel decoding gives $\hat A_g=A_g$, and $\hat f_g=1$ according to Proposition~\ref{prop:finite_error}. 
    Therefore, 
    \begin{equation}
        \{\hat f_g=0\}\subseteq \{\hat A_g\neq A_g\}\subseteq \mathcal{E},
    \end{equation}
    and consequently the FN probability for slot $g$ is
    \begin{equation}
        P_{\rm FN}^{(g)} = \Pr\{\hat f_g=0|f_g(i_g)=1\} \leq \xi.
    \end{equation}

    Suppose $f_g(i_g)=0$ and let $\mathcal{C}_g$ denote the noiseless RS tag collision event in slot $g$. Specifically, 
    \begin{equation}
        \mathcal{C}_g:=\{\exists\,j\in f_g^{-1}(1):P_{i_g}(\alpha_{U_g})=P_j(\alpha_{U_g})|f_g(i_g)=0\}.
    \end{equation}
    Since $\{\hat f_g=1\}\cap \mathcal{E}^{\rm c}\subseteq \mathcal{C}_g$ and $\{\hat f_g=1\}\cap \mathcal{E}\subseteq \mathcal{E}$, the FP probability for slot $g$ is
    \begin{equation}
        \begin{split}
            P_{\rm FP}^{(g)} 
            & = \Pr\{\hat f_g=1\} \\
            & = \Pr\{\{\hat f_g=1\}\cap \mathcal{E}^{\rm c}\} + \Pr\{\{\hat f_g=1\}\cap \mathcal{E}\} \\
            & \leq \Pr\{\mathcal{C}_g\} + \Pr\{\mathcal{E}\}  \\
            & \leq \frac{S(K-1)}{T}+\xi.
        \end{split}
    \end{equation}
    
    Both the FN and FP probability bounds are uniform for all slots, messages and Boolean functions, taking the maximum and imposing an upper bound of one prove the proposition.

\label{cor:noisy_finite_E2}
    Define $L_{S,R_c}(R,n) := L_S(R/R_c,nR_c)$.
    Suppose $r$ and $T$ are chosen as in Theorem~\ref{thm:finite_E2}, and the channel code satisfies $\xi\leq 2^{-N_b E_{\rm ch}}\leq 1$, for some $E_{\rm ch}>0$.
    For all $E_2\in (0,R_c/2)$, set $m=L_{S,R_c}(R_c/2-E_2,n_{\rm eff})$, then the packed concatenated RS tagging construction gives an $(n_{\rm eff}, m, \mathcal F_m(\leq S(m)), \lambda_1,\lambda_2)$ effective BFC code where
    \begin{equation}
        \lambda_1= 2^{-N_bE_{\rm ch}},\quad \lambda_2=\min\{1, 2^{-n_{\rm eff}E_2}+2^{-N_bE_{\rm ch}}\},
    \end{equation}
    and the finite-blocklength computation rate of this code is $R_c/2-E_2$ with the rate function $L_{S,R_c}$.

Let $\tilde E_2=E_2/R_c\in(0,1/2)$. Since $n_t=n_{\rm eff}R_c$, 
    the rate function defined in \eqref{equ:rate_function} is
    \begin{equation}
        L_S\left(\frac{1}{2}-\tilde E_2,n_t\right) = L_{S,R_c}\left(\frac{R_c}{2}-E_2, n_{\rm eff}\right).
