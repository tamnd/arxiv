<!-- assumption-1: body_md -->
\label{ass:rank-law}
The function $ g:[0,\ensuremath{\left|\Omega\right|}]\longrightarrow\ensuremath{\mathbb R}\xspace$ is Lipschitz
continuous and nondecreasing. There exists a unique
$\alpha\in(0,\ensuremath{\left|\Omega\right|})$ such that $g(\alpha)=0.$ Consequently,
\begin{equation}
  g(r)<0
  \quad\text{for }r<\alpha,
  \qquad
  g(r)>0
  \quad\text{for }r>\alpha.
  \label{eq:g-sign}
\end{equation}

<!-- definition-1: body_md -->
\label{def:weak-solution}
A function $u\in \ensuremath{\ensuremath{\operatorname H}\xspace}^{1}_0(\Omega)\cap \ensuremath{\ensuremath{\operatorname C}\xspace^{0}}(\overline\Omega)$ is a weak
solution of \eqref{eq:grad-mercier} if
\begin{equation}
  \int_\Omega
  \nabla u\cdot\nabla\phi\,\mathrm dx
  =
  \int_\Omega
  g\ensuremath{\!\left({
    \ensuremath{\left|u\geq u(x)\right|}
  }\right)}
  \phi(x)\,\mathrm dx
  \quad\:\forall\: \phi\in \ensuremath{\ensuremath{\operatorname H}\xspace}^{1}_0(\Omega).
  \label{eq:weak-problem}
\end{equation}

<!-- theorem-1: body_md -->
\label{thm:continuum-well-posedness}
Let Assumption~\ref{ass:rank-law} hold. Then \eqref{eq:grad-mercier}
admits a unique weak solution $u$. Moreover, $u\in
\ensuremath{\ensuremath{\operatorname C}\xspace^{1,1}}(\overline\Omega), \Delta u\in \ensuremath{\ensuremath{\operatorname C}\xspace^{0}}(\overline\Omega).$ The
maximum set
\begin{equation}
  D_u
  :=
  \ensuremath{\left\{{
    x\in\Omega:
    u(x)=\max_{\overline\Omega}u
  }\right\}}
  \label{eq:continuum-dead-core}
\end{equation}
has the prescribed measure
\begin{equation}
  \ensuremath{\left|D_u\right|}=\alpha.
  \label{eq:continuum-dead-core-measure}
\end{equation}
In addition,
\begin{equation}
  \ensuremath{\left|u\geq u(x)\right|}
  =
  \alpha
  \qquad
  \text{for }x\in D_u,
  \label{eq:superlevel-measure-on-dead-core}
\end{equation}
whereas
\begin{equation}
  \ensuremath{\left|u\geq u(x)\right|}
  >
  \alpha
  \qquad
  \text{for }x\in\Omega\setminus D_u.
  \label{eq:superlevel-measure-off-dead-core}
\end{equation}
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
Existence, uniqueness, regularity, and
\eqref{eq:continuum-dead-core-measure} follow from
\cite{CaffarelliFarahRestrepo2025}. For $x\in D_u$, the superlevel set
at $u(x)$ is $D_u$, giving
\eqref{eq:superlevel-measure-on-dead-core}. If $x\notin D_u$, then
$u(x)<\max_{\overline\Omega}u$; by continuity,
$\{u>u(x)\}$ contains a neighbourhood of $D_u$ and hence has measure
strictly greater than $\alpha$. This proves
\eqref{eq:superlevel-measure-off-dead-core}.

\label{thm:discrete-energy-certificate}
Let $\ensuremath{\left\{{ u_h^j,\widehat u_h^{j+1} }\right\}}_{j=0}^n$ be generated by the
rearrangement iteration
\eqref{eq:iterative-rank-source}--\eqref{eq:rearrangement-iteration}.
For $0\leq j\leq n$, define the stepwise object
\begin{equation}
  \ell_h^j
  :=
  a_h\ensuremath{\!\left({
    \widehat u_h^{j+1},u_h^j
  }\right)}
  -
  \mathcal R_h(u_h^j)
  -
  \frac12
  a_h\ensuremath{\!\left({
    \widehat u_h^{j+1},
    \widehat u_h^{j+1}
  }\right)}.
  \label{eq:iteration-lower-bound}
\end{equation}
The best floor available after $n$ iterations, the current iterate
energy, and their gap are
\begin{equation}
  L_h^n
  :=
  \max_{0\leq j\leq n}\ell_h^j,
  \qquad
  \mathcal J_h^n
  :=
  \mathcal J_h(u_h^n),
  \qquad
  \eta_h^n
  :=
  \mathcal J_h^n-L_h^n.
  \label{eq:iteration-energy-bounds}
\end{equation}
Writing $\mathcal J_h^\star := \mathcal J_h(u_h)$ for the minimum
energy, where $u_h$ is the unique discrete minimiser, we have
\begin{equation}
  L_h^n
  \leq
  \mathcal J_h^\star
  \leq
  \mathcal J_h^n.
  \label{eq:certified-energy-bracket}
\end{equation}
Consequently,
\begin{equation}
  0
  \leq
  \mathcal J_h^n-\mathcal J_h^\star
  \leq
  \eta_h^n,
  \label{eq:certified-energy-error}
\end{equation}
and
\begin{equation}
  \ensuremath{\left\|
    \nabla(u_h^n-u_h)
  \right\|}_{\ensuremath{\ensuremath{\operatorname L}\xspace^{2}}(\Omega_h)}
  \leq
  \sqrt{2\eta_h^n}.
  \label{eq:certified-energy-norm-error}
\end{equation}

<!-- theorem-3: body_md -->
\label{thm:main-nonlinear}
Let the domain and mesh hypotheses above hold, and let
Assumptions~\ref{ass:rank-law} and~\ref{ass:DMP} hold.

For every sufficiently small $h$, the discrete problem
\eqref{eq:discrete-problem} has a unique solution $u_h\in K_h.$

For every $u_h^0\in K_h$ and every step sequence satisfying
\eqref{eq:step-size-conditions}, the iteration
\eqref{eq:rearrangement-poisson-step}--\eqref{eq:rearrangement-iteration}
converges to $u_h$:
\begin{equation}
  \ensuremath{\left\|
    u_h^n-u_h
  \right\|}_{\ensuremath{\ensuremath{\operatorname H}\xspace}^{1}_0(\Omega_h)}
  \longrightarrow0
  \qquad
  \text{as }n\longrightarrow\infty.
  \label{eq:iteration-convergence}
\end{equation}
For fixed $h$, convergence also holds in $L^\infty(\Omega_h)$.

Let $u$ be the continuum solution from
Theorem~\ref{thm:continuum-well-posedness}. Then, as $h\to0$,
\begin{align}
  \mathcal E_hu_h
  &\longrightarrow u
  &&\text{strongly in }\ensuremath{\ensuremath{\operatorname H}\xspace}^{1}_0(\Omega),
  \label{eq:main-H1-convergence}
  \\
  \mathcal E_hu_h
  &\longrightarrow u
  &&\text{strongly in }\ensuremath{\ensuremath{\operatorname L}\xspace^{\infty}}(\Omega).
  \label{eq:main-Linf-convergence}
\end{align}

For each $h$, choose any admissible equilibrium source
$q_h^\ast$ satisfying \eqref{eq:discrete-source-equation}, and define
\begin{equation}
  s_h
  :=
  s_h[q_h^\ast].
  \label{eq:equilibrium-source-reconstruction}
\end{equation}
Then, independently of the choice of $q_h^\ast$, its zero extension
satisfies
\begin{equation}
  \widetilde{s_h}
  \stackrel{*}{\rightharpoonup}
  g\ensuremath{\!\left({
    \ensuremath{\left|u\geq u(\cdot)\right|}
  }\right)}
  \qquad
  \text{in }\ensuremath{\ensuremath{\operatorname L}\xspace^{\infty}}(\Omega)
  \quad\text{as }h\to0.
  \label{eq:main-source-convergence}
\end{equation}
Equivalently,
\begin{equation}
\int_\Omega
\widetilde{s_h}\varphi\,\mathrm dx
\longrightarrow
  \int_\Omega
  g\ensuremath{\!\left({
    \ensuremath{\left|u\geq u(\cdot)\right|}
    }\right)}
  \varphi\,\mathrm dx
  \qquad
  \text{for every }\varphi\in \ensuremath{\ensuremath{\operatorname L}\xspace^{1}}(\Omega).
  \label{eq:main-source-convergence-duality}
\end{equation}
In particular, the limiting solution has a maximum plateau of measure
$\alpha$ and no positive-measure level sets below its maximum.

<!-- lemma-1: body_md -->
\label{lem:nodal-measure-convergence}
For every $\psi\in \ensuremath{\ensuremath{\operatorname C}\xspace^{0}}(\overline\Omega)$,
\begin{equation}
  \int_{\overline\Omega}
  \psi
  \ensuremath{\,\mathrm{d}}\mu_h
  =
  \sum_{i\in\mathcal N_h}
  \omega_i\psi(x_i)
  =
  \int_{\Omega_h}
  I_h\psi\,\mathrm dx.
  \label{eq:nodal-quadrature}
\end{equation}
Consequently,
\begin{equation}
  \mu_h
  \rightharpoonup
  \mathcal L^2|_\Omega
  \qquad
  \text{weakly as finite measures on }\overline\Omega.
  \label{eq:mu-weak}
\end{equation}
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
The first identity follows from the definition of the weights and the
partition of unity:
\[
  \sum_{i\in\mathcal N_h}
  \omega_i\psi(x_i)
  =
  \int_{\Omega_h}
  \sum_{i\in\mathcal N_h}
  \psi(x_i)\varphi_i\,\mathrm dx
  =
  \int_{\Omega_h}
  I_h\psi\,\mathrm dx.
\]
Since $I_h\psi\to\psi$ uniformly on $\Omega_h$ and
$\ensuremath{\left|\Omega\setminus\Omega_h\right|}\to0$,
\[
  \ensuremath{\left|
    \int_{\overline\Omega}
    \psi\ensuremath{\,\mathrm{d}}\mu_h
    -
    \int_\Omega\psi\,\mathrm dx
  \right|}
  \leq
  \ensuremath{\left|\Omega_h\right|}
  \ensuremath{\left\|I_h\psi-\psi\right\|}_{L^\infty(\Omega_h)}
  +
  \int_{\Omega\setminus\Omega_h}
  \ensuremath{\left|\psi\right|}\,\mathrm dx
  \longrightarrow0.
\]

<!-- lemma-3: body_md -->
\label{lem:mass-lumping-error}
For $p_h,r_h\in\widetilde V_h$,
\begin{equation}
  \ensuremath{\left|
    \sum_{i\in\mathcal N_h}
    \omega_i
    p_h(x_i)r_h(x_i)
    -
    \int_{\Omega_h}
    p_hr_h\,\mathrm dx
  \right|}
  \leq
  Ch
  \ensuremath{\left\|p_h\right\|}_{L^\infty(\Omega_h)}
  \ensuremath{\left\|\nabla r_h\right\|}_{L^2(\Omega_h)}.
  \label{eq:mass-lumping-error}
\end{equation}
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
On $K\in\ensuremath{\varTheta}\xspace$, the local quadrature error is $\int_K \bigl(
I_K(p_hr_h)-p_hr_h \bigr) \,\mathrm dx,$ where
\[
  I_K(p_hr_h)(x)-p_h(x)r_h(x) = \sum_{j=1}^3 \lambda_j(x)p_h(x_j)
  \bigl( r_h(x_j)-r_h(x) \bigr).
\]
Since $r_h$ is affine, $\ensuremath{\left|r_h(x_j)-r_h(x)\right|}\leq
Ch_K\ensuremath{\left|\nabla r_h|_K\right|}.$
Thus
\[
  \ensuremath{\left|
    \int_K
    \bigl(
      I_K(p_hr_h)-p_hr_h
    \bigr)
    \,\mathrm dx
  \right|}
  \leq
  Ch_K
  \ensuremath{\left\|p_h\right\|}_{L^\infty(K)}
  \ensuremath{\left|K\right|}
  \ensuremath{\left|\nabla r_h|_K\right|}.
\]
Summation and Cauchy--Schwarz give
\eqref{eq:mass-lumping-error}.

<!-- lemma-5: body_md -->
\label{lem:discrete-rearrangement-representation}
For every $v_h\inV_h$,
\begin{equation}
  \mathcal R_h(v_h)
  =
  \min_{(E_i)}
  \sum_{i\in\mathcal N_h}
  v_h(x_i)
  \int_{E_i}
  G(\rho) \ensuremath{\,\mathrm{d}}\rho.
  \label{eq:discrete-minimum-representation}
\end{equation}
The minimum assigns the lowest ranks to the largest nodal values. Thus
the nodes on a level $c\in\mathcal L_h(v_h)$ collectively receive the
rank interval
\begin{equation}
  \mathcal I_h(c;v_h).
  \label{eq:minimising-level-rank-interval}
\end{equation}
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
Suppose $v_h(x_i)>v_h(x_j)$, but equal-measure sets $A\subset E_i$ and
$B\subset E_j$ satisfy $\int_A G(\rho) \ensuremath{\,\mathrm{d}}\rho > \int_B G(\rho)
\ensuremath{\,\mathrm{d}}\rho.$ Exchanging $A$ and $B$ changes the pairing by
\[
  \ensuremath{\!\left({
    v_h(x_i)-v_h(x_j)
  }\right)}
  \ensuremath{\!\left({
    \int_B G(\rho) \ensuremath{\,\mathrm{d}}\rho
    -
    \int_A G(\rho) \ensuremath{\,\mathrm{d}}\rho
  }\right)}
  <
  0.
\]
Hence every minimising allocation is oppositely ordered to the nodal
values. Grouping equal values gives
\[
  \sum_{c\in\mathcal L_h(v_h)}
  c
  \int_{\mathcal I_h(c;v_h)}
  G(\rho) \ensuremath{\,\mathrm{d}}\rho
  =
  \mathcal R_h(v_h).
  \]
<!-- end proof-6 -->
<!-- lemma-6: body_md -->
\label{lem:discrete-rearrangement-concavity}
The functional $\mathcal R_h:V_h\to\ensuremath{\mathbb R}\xspace$ is concave and
\begin{equation}
  \ensuremath{\left|
    \mathcal R_h(v_h)-\mathcal R_h(w_h)
  \right|}
  \leq
  M_\alpha
  \sum_{i\in\mathcal N_h}
  \omega_i
  \ensuremath{\left|
    v_h(x_i)-w_h(x_i)
  \right|}
  \label{eq:discrete-rearrangement-Lipschitz}
\end{equation}
for all $v_h,w_h\inV_h$.
<!-- end lemma-6 -->
<!-- proof-7: body_md -->
The representation
\eqref{eq:discrete-minimum-representation} expresses $\mathcal R_h$ as
the pointwise minimum of linear functionals, so it is concave. If
$(E_i)$ minimises the representation at $w_h$, then
\[
  \mathcal R_h(v_h)-\mathcal R_h(w_h)
  \leq
  \sum_{i\in\mathcal N_h}
  \omega_iq_i
  \bigl(
    v_h(x_i)-w_h(x_i)
  \bigr),
\]
where $0\leq q_i\leq M_\alpha$. Interchanging $v_h$ and $w_h$ proves
\eqref{eq:discrete-rearrangement-Lipschitz}.

<!-- lemma-7: body_md -->
\label{lem:canonical-source-supergradient}
Let $c\in\mathcal L_h(v_h)$ and set
\[
  I_c
  :=
  \mathcal I_h(c;v_h),
  \qquad
  \omega_c
  :=
  \ensuremath{\left|I_c\right|}.
  \]
  There is a partition of $I_c$ among the nodes satisfying
$v_h(x_i)=c$ such that
\begin{equation}
  \frac{1}{\omega_i}
  \int_{E_i}G(\rho) \ensuremath{\,\mathrm{d}}\rho
  =
  \frac{1}{\omega_c}
  \int_{I_c}G(\rho) \ensuremath{\,\mathrm{d}}\rho.
  \label{eq:equal-average-allocation}
\end{equation}
Consequently, the canonical source
$\overline q_h[v_h]$ satisfies
\begin{equation}
  \mathcal R_h(w_h)-\mathcal R_h(v_h)
  \leq
  \sum_{i\in\mathcal N_h}
  \omega_i
  \overline q_{h,i}[v_h]
  \bigl(
    w_h(x_i)-v_h(x_i)
  \bigr)
  \label{eq:canonical-source-supergradient}
\end{equation}
for every $w_h\inV_h$.
<!-- end lemma-7 -->
<!-- proof-8: body_md -->
Let
\[
  \mathcal N_c
  :=
  \ensuremath{\left\{{
    i\in\mathcal N_h:
    v_h(x_i)=c
  }\right\}}
  =
  \ensuremath{\left\{{
    i_1,\ldots,i_m
  }\right\}}.
\]
On $I_c$, define the vector measure
\[
  \nu(E)
  :=
  \ensuremath{\!\left({
    \ensuremath{\left|E\right|},
    \int_E G(\rho)\ensuremath{\,\mathrm{d}}\rho
  }\right)},
  \qquad
  E\subset I_c
  \text{ measurable}.
\]
Since Lebesgue measure is nonatomic and $G\in L^\infty(I_c)$, the
vector measure $\nu$ is nonatomic. Hence, by the Lyapunov convexity
theorem, the range of the restriction of $\nu$ to any measurable
subset $A\subset I_c$ is convex. In particular, for every
$\lambda\in[0,1]$ there exists a measurable set $E\subset A$ such that
$\nu(E) = \lambda\nu(A).$

We now construct the required partition recursively. Set $A_1:=I_c$
and $W_k:=\sum_{\ell=k}^m\omega_{i_\ell}$. Since $W_1=\omega_c$, we
have $\nu(A_1)=\nu(I_c)=\frac{W_1}{\omega_c}\nu(I_c)$. Suppose that a
measurable remainder $A_k$ has been constructed with
$\nu(A_k)=\frac{W_k}{\omega_c}\nu(I_c)$.
By the preceding convexity property, there exists
$E_{i_k}\subset A_k$ such that
$\nu(E_{i_k})=\frac{\omega_{i_k}}{W_k}\nu(A_k)=
\frac{\omega_{i_k}}{\omega_c}\nu(I_c)$.
Set $A_{k+1} := A_k\setminus E_{i_k}.$ Then $\nu(A_{k+1}) =
\frac{W_{k+1}}{\omega_c}\nu(I_c),$ so the construction may be
continued. Taking the final remainder as $E_{i_m}$ gives a measurable
partition of $I_c$ satisfying
\[
  \ensuremath{\left|E_i\right|}
  =
  \omega_i,
  \qquad
  \int_{E_i}G(\rho)\ensuremath{\,\mathrm{d}}\rho
  =
  \frac{\omega_i}{\omega_c}
  \int_{I_c}G(\rho)\ensuremath{\,\mathrm{d}}\rho.
\]

The functional $-\mathcal R_h$ is convex and continuous, while
$v_h\longmapsto\frac12a_h(v_h,v_h)$ is strictly convex on $V_h$.
Hence $\mathcal J_h$ is strictly convex.
By Lemma~\ref{lem:discrete-energy-coercivity}, it is also coercive.
Since $V_h$ is finite dimensional, $\mathcal J_h$ therefore has a
unique minimiser.

We prove the equivalence in the order
$(1)\Longrightarrow(3)\Longrightarrow(2)\Longrightarrow(1)$.

Suppose first that $u_h$ minimises $\mathcal J_h$. The
finite-dimensional convex optimality condition gives a supergradient
$\ell_h\in\partial^+\mathcal R_h(u_h)$ satisfying
\begin{equation}
  a_h(u_h,v_h)
  =
  \ell_h(v_h)
  \qquad
  \text{for every }v_h\inV_h.
  \label{eq:technical-optimality-balance}
\end{equation}
By Lemma~\ref{lem:discrete-supergradient-representation}, there is an
admissible source $q_h^\ast$ at $u_h$ representing $\ell_h$. Hence
\[
  a_h(u_h,v_h)
  =
  \sum_{i\in\mathcal N_h}
  \omega_iq_{h,i}^\ast v_h(x_i)
  \qquad
  \text{for every }v_h\inV_h,
\]
which proves $(1)\Rightarrow(3)$.

Now suppose that $(3)$ holds. Since $q_h^\ast$ is admissible at $u_h$,
\[
  \mathcal R_h(v_h)-\mathcal R_h(u_h)
  \leq
  \sum_{i\in\mathcal N_h}
  \omega_iq_{h,i}^\ast
  \ensuremath{\!\left({
    v_h(x_i)-u_h(x_i)
  }\right)}.
\]
Using \eqref{eq:technical-discrete-source-equation} with
$v_h-u_h$ therefore gives $\mathcal R_h(v_h)-\mathcal R_h(u_h)
\leq a_h(u_h,v_h-u_h)$,
which is \eqref{eq:technical-discrete-VI}. Thus
$(3)\Rightarrow(2)$.

Finally, suppose that $(2)$ holds. Expanding the quadratic term gives
\begin{align*}
  \mathcal J_h(v_h)-\mathcal J_h(u_h)
  &=
  a_h(u_h,v_h-u_h)
  +
  \frac12
  a_h(v_h-u_h,v_h-u_h)
  \\
  &\qquad
  -
  \ensuremath{\!\left({
    \mathcal R_h(v_h)-\mathcal R_h(u_h)
  }\right)}.
\end{align*}
By \eqref{eq:technical-discrete-VI},
\[
  \mathcal J_h(v_h)-\mathcal J_h(u_h)
  \geq
  \frac12
  a_h(v_h-u_h,v_h-u_h)
  \geq
  0.
\]
Hence $u_h$ minimises $\mathcal J_h$, proving $(2)\Rightarrow(1)$.
The same inequality also recovers uniqueness from strict convexity.

  \int_{\Omega_h}
  s_h\phi\,\mathrm dx
  \qquad
  \text{for every }\phi\in \ensuremath{\ensuremath{\operatorname H}\xspace}^{1}_0(\Omega_h).
  \label{eq:technical-continuous-Poisson}
\end{equation}
Testing with $w(h)$ and using Poincaré's inequality gives
\begin{equation}
  \sup_h
  \ensuremath{\left\|\nabla w(h)\right\|}_{L^2(\Omega_h)}
  <\infty.
  \label{eq:technical-continuous-energy-bound}
\end{equation}

Let $z(h)\in \ensuremath{\ensuremath{\operatorname H}\xspace}^{1}_0(\Omega)$ solve the corresponding problem on the
fixed domain:
\begin{equation}
  \int_\Omega
  \nabla z(h)\cdot\nabla\phi\,\mathrm dx
  =
  \int_\Omega
  \widetilde{s_h}\phi\,\mathrm dx
  \qquad
  \text{for every }\phi\in \ensuremath{\ensuremath{\operatorname H}\xspace}^{1}_0(\Omega).
  \label{eq:technical-fixed-domain-problem}
\end{equation}
By \eqref{eq:technical-uniform-source-bound}, the zero extensions satisfy
\[
  \sup_h
  \ensuremath{\left\|
    \widetilde{s_h}
  \right\|}_{L^p(\Omega)}
  <\infty
  \qquad
  \text{for every }1\leq p<\infty.
\]
Since the fixed domain $\Omega$ has $\ensuremath{\ensuremath{\operatorname C}\xspace^{1,1}}$ boundary, elliptic
regularity therefore gives, for every fixed $p>2$,
\begin{equation}
  \ensuremath{\left\|z(h)\right\|}_{W^{2,p}(\Omega)}
  \leq
  C_p
  \ensuremath{\left\|
    \widetilde{s_h}
  \right\|}_{L^p(\Omega)}
  \leq
  C_p.
  \label{eq:technical-fixed-domain-W2p}
\end{equation}
In two dimensions,
\[
  W^{2,p}(\Omega)
  \hookrightarrow
  C^{1,\gamma}(\overline\Omega),
  \qquad
  \gamma
  =
  1-\frac{2}{p}
  >0,
\]
and the embedding
$C^{1,\gamma}(\overline\Omega)\hookrightarrow C^1(\overline\Omega)$
is compact. Hence $\{z(h)\}_h$ is precompact in
$\ensuremath{\ensuremath{\operatorname C}\xspace^{1}}(\overline\Omega)$.

If $z(h_k)\to z$ strongly in $\ensuremath{\ensuremath{\operatorname C}\xspace^{1}}(\overline\Omega)$ along a
subsequence, then passing to the limit in
\eqref{eq:technical-fixed-domain-problem} using
\eqref{eq:reconstructed-source-weak-star} shows that $z$ satisfies
\eqref{eq:limit-domain-poisson}. By uniqueness, $z=w$. Since every
convergent subsequence has the same limit, the full sequence satisfies
\begin{equation}
  z(h)
  \longrightarrow
  w
  \qquad
  \text{strongly in }\ensuremath{\ensuremath{\operatorname C}\xspace^{1}}(\overline\Omega).
  \label{eq:technical-fixed-domain-C1}
\end{equation}

We next compare the fixed-domain and varying-domain solutions. The
inclusion
\[
  \Omega_{-\delta_h}
  \subset
  \Omega_h
\]
implies
\begin{equation}
  \sup_{x\in\partial\Omega_h}
  \ensuremath{\operatorname{dist}}(x,\partial\Omega)
  \leq
  \delta_h.
  \label{eq:technical-boundary-distance}
\end{equation}
Indeed, if $x\in\partial\Omega_h$ satisfied
$\ensuremath{\operatorname{dist}}(x,\partial\Omega)>\delta_h$, then
$x\in\Omega_{-\delta_h}\subset\Omega_h$, contradicting
$x\in\partial\Omega_h$.

Since $\widetilde{s_h}=s_h$ on $\Omega_h$, the function
\[
  w(h)-z(h)|_{\Omega_h}
\]
is harmonic in $\Omega_h$ and has boundary value $-z(h)$ on
$\partial\Omega_h$. The maximum principle, the zero trace of $z(h)$ on
$\partial\Omega$, and \eqref{eq:technical-fixed-domain-W2p} give
\begin{align}
  \ensuremath{\left\|
    w(h)-z(h)
  \right\|}_{L^\infty(\Omega_h)}
  &\leq
  \ensuremath{\left\|z(h)\right\|}_{L^\infty(\partial\Omega_h)}
  \notag
  \\
  &\leq
  \ensuremath{\left\|\nabla z(h)\right\|}_{\ensuremath{\ensuremath{\operatorname L}\xspace^{\infty}}(\Omega)}
  \sup_{x\in\partial\Omega_h}
  \ensuremath{\operatorname{dist}}(x,\partial\Omega)
  \notag
  \\
  &\leq
  C\delta_h.
  \label{eq:technical-harmonic-comparison}
\end{align}
It follows from \eqref{eq:technical-fixed-domain-C1} that
\begin{equation}
  \ensuremath{\left\|
    w(h)-w
  \right\|}_{L^\infty(\Omega_h)}
  \longrightarrow0.
  \label{eq:technical-interior-Linf}
\end{equation}

It remains to control the part of $\Omega$ omitted by $\Omega_h$.
The domain inclusions give
\[
  \Omega\setminus\Omega_h
  \subset
  \ensuremath{\left\{{
    x\in\Omega:
    \ensuremath{\operatorname{dist}}(x,\partial\Omega)\leq\delta_h
  }\right\}}.
\]
Since $w\in \ensuremath{\ensuremath{\operatorname C}\xspace^{1}}(\overline\Omega)$ and $w=0$ on $\partial\Omega$,
\begin{equation}
  \ensuremath{\left\|w\right\|}_{L^\infty(\Omega\setminus\Omega_h)}
  \leq
  \delta_h
  \ensuremath{\left\|\nabla w\right\|}_{\ensuremath{\ensuremath{\operatorname L}\xspace^{\infty}}(\Omega)}

  \label{eq:technical-boundary-strip}
\end{equation}
Because $\mathcal E_hw(h)=0$ on $\Omega\setminus\Omega_h$,
\eqref{eq:technical-interior-Linf} and
\eqref{eq:technical-boundary-strip} yield
\begin{equation}
  \mathcal E_hw(h)
  \longrightarrow
  w
  \qquad
  \text{strongly in }\ensuremath{\ensuremath{\operatorname L}\xspace^{\infty}}(\Omega).
  \label{eq:technical-continuous-Linf}
\end{equation}

\[
  \ensuremath{\left\|\nabla w(h)\right\|}_{L^2(\Omega_h)}^2

  \int_\Omega
  \widetilde{s_h}

\[
  \int_\Omega
  \widetilde{s_h}

  \longrightarrow

Together with weak compactness in $\ensuremath{\ensuremath{\operatorname H}\xspace}^{1}_0(\Omega)$, this proves
\begin{equation}
  \mathcal E_hw(h)

\label{lem:continuum-rearrangement-properties}
The functional $\mathcal R:L^1(\Omega)\to\ensuremath{\mathbb R}\xspace$ is concave and
satisfies

    \mathcal R(v)-\mathcal R(w)

  \text{for every }v,w\in L^1(\Omega).

\end{equation}

Since $G$ is nondecreasing, the Hardy--Littlewood rearrangement
inequality and the characterisation of equimeasurable rearrangements
\cite{HardyLittlewoodPolya1952,Ryff1965},
\begin{equation}
  \mathcal R(v)
  =
  \inf_{q\sim G}
  \int_\Omega vq\,\mathrm dx.
  \label{eq:continuum-minimum-representation}
\end{equation}
Indeed, among all spatial rearrangements of $G$, the integral is
minimised by pairing the decreasing rearrangement of $v$ with the
increasing rearrangement of $G$, which is $G$ itself.

For every fixed $q\sim G$, the map
\[
  v
  \longmapsto
  \int_\Omega
  vq\,\mathrm dx
\]
is linear. Hence \eqref{eq:continuum-minimum-representation} expresses
$\mathcal R$ as an infimum of linear functionals, and therefore
$\mathcal R$ is concave.

Equimeasurability preserves the essential range. Since
\[
  0
  \leq
  G
  \leq
  M_\alpha
  \qquad
  \text{almost everywhere on }(0,\ensuremath{\left|\Omega\right|}),
\]
every $q\sim G$ satisfies
\begin{equation}
  0
  \leq
  q
  \leq
  M_\alpha
  \qquad
  \text{almost everywhere in }\Omega.
  \label{eq:continuum-rearranged-source-bound}
\end{equation}

Let $\varepsilon>0$ and choose $q_\varepsilon\sim G$ such that
\[
  \int_\Omega
  wq_\varepsilon\,\mathrm dx
  \leq
  \mathcal R(w)+\varepsilon.
\]
Using \eqref{eq:continuum-minimum-representation} for $v$ gives
\begin{align*}
  \mathcal R(v)-\mathcal R(w)
  &\leq
  \int_\Omega
  vq_\varepsilon\,\mathrm dx
  -
  \int_\Omega
  wq_\varepsilon\,\mathrm dx
  +
  \varepsilon
  \\
  &\leq
  M_\alpha
  \ensuremath{\left\|v-w\right\|}_{L^1(\Omega)}
  +
  \varepsilon.
\end{align*}
Letting $\varepsilon\to0$ and interchanging $v$ and $w$ proves
\eqref{eq:continuum-rearrangement-Lipschitz}.

\[
  v
  \longmapsto
  \frac12
  \int_\Omega
  \ensuremath{\left|\nabla v\right|}^2\,\mathrm dx
\]
is strictly convex on $\ensuremath{\ensuremath{\operatorname H}\xspace}^{1}_0(\Omega)$. Their sum $\mathcal J$ is
therefore strictly convex.

  >

\label{lem:continuum-variational-characterisation}
Let $u$ be the continuum Grad--Mercier solution and define
\begin{equation}
  q_u(x)
  :=
  G\bigl(
    \ensuremath{\left|u\geq u(x)\right|}
  \bigr).
  \label{eq:continuum-canonical-source}
\end{equation}
Then
\begin{equation}
  \mathcal R(v)-\mathcal R(u)
  \leq
  \int_\Omega
  q_u(v-u)\,\mathrm dx
  \qquad

\end{equation}
and $u$ is the unique minimiser of $\mathcal J$.

Let
\[
  D_u
  :=
  \ensuremath{\left\{{
    x\in\Omega:
    u(x)=\max_{\overline\Omega}u
  }\right\}}.
\]
By Theorem~\ref{thm:continuum-well-posedness},
\[
  \ensuremath{\left|D_u\right|}
  =
  \alpha.
\]
Moreover, Lemma~\ref{lem:continuum-no-secondary-plateaux} shows that
every level set below the maximum has zero measure.

Define the non-strict rank map
\[
  r_u(x)
  :=
  \ensuremath{\left|
    \ensuremath{\left\{{
      y\in\Omega:
      u(y)\geq u(x)
    }\right\}}
  \right|}.
\]
On the maximum plateau,
\begin{equation}
  r_u(x)
  =
  \alpha
  \qquad
  \text{for every }x\in D_u.
  \label{eq:continuum-plateau-rank}
\end{equation}
Away from $D_u$, the absence of positive-measure level sets implies
that the rank variable is uniformly distributed over
$(\alpha,\ensuremath{\left|\Omega\right|})$. More precisely, writing $\mathcal L^d$ for
$d$-dimensional Lebesgue measure, $\mu|_A$ for the restriction of a
measure $\mu$ to $A$, and $T_\#\mu$ for its pushforward under a map
$T$, we have
\begin{equation}
  (r_u)_\#
  \ensuremath{\!\left({
    \mathcal L^2|_{\Omega\setminus D_u}
  }\right)}
  =
  \mathcal L^1|_{(\alpha,\ensuremath{\left|\Omega\right|})}.
  \label{eq:continuum-rank-pushforward}
\end{equation}
Indeed, let $s\in(\alpha,\ensuremath{\left|\Omega\right|})$. Since the level sets of $u$
below its maximum have zero measure, the distribution function
\[
  t
  \longmapsto
  \ensuremath{\left|u\geq t\right|}
\]
is continuous below the maximum. Hence there exists $t_s$ such that
\[
  \ensuremath{\left|u\geq t_s\right|}
  =
  s.
\]
Up to a null set,
\[
  \ensuremath{\left\{{
    x\in\Omega\setminus D_u:
    r_u(x)\leq s
  }\right\}}
  =
  \ensuremath{\left\{{
    x\in\Omega:
    u(x)\geq t_s
  }\right\}}
  \setminus D_u,
\]
and therefore
\[
  \ensuremath{\left|
    \ensuremath{\left\{{
      x\in\Omega\setminus D_u:
      r_u(x)\leq s
    }\right\}}
  \right|}
  =
  s-\alpha.
\]
This is precisely the distribution function of Lebesgue measure on
$(\alpha,\ensuremath{\left|\Omega\right|})$, and proves
\eqref{eq:continuum-rank-pushforward}.

Set
\begin{equation}
  q_u(x)
  :=
  G\ensuremath{\!\left({r_u(x)}\right)}.
\end{equation}
Since $G=0$ on $[0,\alpha]$, the plateau $D_u$ contributes mass
$\alpha$ at the value zero. Together with
\eqref{eq:continuum-rank-pushforward}, this shows that $q_u$ and $G$
have the same distribution of values. Thus
\begin{equation}
  q_u
  \sim
  G.
  \label{eq:continuum-source-equimeasurable}
\end{equation}

The functions $u$ and $q_u$ are oppositely ordered. Indeed, if
$u(x)>u(y)$, then
\[
  r_u(x)
  \leq
  r_u(y),
\]
and, since $G$ is nondecreasing,
\[
  q_u(x)
  \leq
  q_u(y).
\]
The opposite-order case of the Hardy--Littlewood rearrangement
inequality therefore gives
\begin{equation}
  \mathcal R(u)
  =
  \int_\Omega
  uq_u\,\mathrm dx.
  \label{eq:continuum-source-attains-minimum}
\end{equation}
On the other hand, since $q_u\sim G$, the minimum representation
\eqref{eq:continuum-minimum-representation} gives
\begin{equation}
  \mathcal R(v)
  \leq
  \int_\Omega
  vq_u\,\mathrm dx
  \qquad
  \text{for every }v\in L^1(\Omega).
  \label{eq:continuum-fixed-source-upper-bound}
\end{equation}
Subtracting \eqref{eq:continuum-source-attains-minimum} from
\eqref{eq:continuum-fixed-source-upper-bound} yields
\begin{equation}
  \mathcal R(v)-\mathcal R(u)
  \leq
  \int_\Omega
  q_u(v-u)\,\mathrm dx,
\end{equation}
so $q_u$ is a supergradient of $\mathcal R$ at $u$.

Finally, $r_u(x)\geq\alpha$ throughout $\Omega$, and hence
\[
  q_u(x)
  =
  G\bigl(r_u(x)\bigr)
  =
  g\bigl(r_u(x)\bigr).
\]
The Grad--Mercier equation therefore gives
\[
  \int_\Omega
  \nabla u\cdot\nabla(v-u)\,\mathrm dx
  =
  \int_\Omega
  q_u(v-u)\,\mathrm dx
  \qquad
  \text{for every }v\in \ensuremath{\ensuremath{\operatorname H}\xspace}^{1}_0(\Omega).
\]
Using \eqref{eq:continuum-supergradient-inequality} and expanding the
quadratic term, we obtain
\begin{align*}
  \mathcal J(v)-\mathcal J(u)
  &=
  \int_\Omega
  \nabla u\cdot\nabla(v-u)\,\mathrm dx
  +
  \frac12
  \ensuremath{\left\|
    \nabla(v-u)
  \right\|}_{L^2(\Omega)}^2
  -
  \bigl(
    \mathcal R(v)-\mathcal R(u)
  \bigr)
  \\
  &\geq
  \frac12
  \ensuremath{\left\|
    \nabla(v-u)
  \right\|}_{L^2(\Omega)}^2.
\end{align*}
Thus $u$ minimises $\mathcal J$. The inequality is strict whenever
$v\neq u$, so $u$ is the unique minimiser.

  \int_0^m
  \ensuremath{\left|
    \widehat v_h^\ast(r)-v^\ast(r)
  \right|}
  \ensuremath{\,\mathrm{d}} r
  \longrightarrow0,
\]
and in particular
\begin{equation}
  \int_0^{m_h}
  \ensuremath{\left|
    v_h^\ast(r)-v^\ast(r)
  \right|}
  \ensuremath{\,\mathrm{d}} r
  \longrightarrow0.
  \label{eq:augmented-quantile-convergence}
\end{equation}

By the definitions of the discrete and continuum rearrangement
functionals,
\[
  \mathcal R_h(v_h)
  =
  \int_0^{m_h}
  G(r)v_h^\ast(r)\ensuremath{\,\mathrm{d}} r,
  \qquad
  \mathcal R(v)
  =
  \int_0^m
  G(r)v^\ast(r)\ensuremath{\,\mathrm{d}} r.
\]
Since $0\leq G\leq M_\alpha$ and $\ensuremath{\left|v^\ast\right|}\leq B$,
\begin{align*}
  \ensuremath{\left|
    \mathcal R_h(v_h)-\mathcal R(v)
  \right|}
  &\leq
  M_\alpha
  \int_0^{m_h}
  \ensuremath{\left|
    v_h^\ast(r)-v^\ast(r)
  \right|}
  \ensuremath{\,\mathrm{d}} r
  \\
  &\quad
  +
  M_\alpha B
  (m-m_h).
\end{align*}
The first term tends to zero by
\eqref{eq:augmented-quantile-convergence}, while the second tends to
zero because $\ensuremath{\left|\Omega_h\right|}\to\ensuremath{\left|\Omega\right|}$. This proves
\eqref{eq:rearrangement-functional-convergence}.

Existence, uniqueness and the inclusion
\[
  u_h\in K_h
\]
follow from Lemmas~\ref{lem:discrete-well-posedness}
and~\ref{lem:discrete-barrier-bound}. For each fixed $h$, convergence of
the rearrangement iteration to $u_h$ follows from
Lemma~\ref{lem:rearrangement-iteration-convergence}.

We now consider the limit $h\to0$. Let $h_k\to0$ be arbitrary. For each
$k$, choose any admissible equilibrium source
\[
  q_{h_k}^\ast
  =
  \ensuremath{\!\left({
    q_{h_k,i}^\ast
  }\right)}_{i\in\mathcal N_{h_k}}
\]
satisfying \eqref{eq:discrete-source-equation}, and define
\[
  s_{h_k}^\ast
  :=
  \sum_{i\in\mathcal N_{h_k}}
  q_{h_k,i}^\ast\varphi_i.
\]
The admissible source bounds imply
\[
  0
  \leq
  s_{h_k}^\ast
  \leq
  M_\alpha
  \qquad
  \text{in }\Omega_{h_k}.
\]
Hence there exist a subsequence, not relabelled, and
$f\in \ensuremath{\ensuremath{\operatorname L}\xspace^{\infty}}(\Omega)$ such that
\begin{equation}
  \widetilde{s_{h_k}^\ast}
  \stackrel{*}{\rightharpoonup}
  f
  \qquad
  \text{in }\ensuremath{\ensuremath{\operatorname L}\xspace^{\infty}}(\Omega)
  \quad\text{as }k\to\infty.
  \label{eq:technical-source-subsequence}
\end{equation}

The discrete equilibrium equation is a mass-lumped Poisson problem.
Lemma~\ref{lem:linear-compactness-consistency} therefore gives a
function $w\in \ensuremath{\ensuremath{\operatorname H}\xspace}^{1}_0(\Omega)$ such that
\begin{align}
  \mathcal E_{h_k}u_{h_k}
  &\longrightarrow
  w
  &&\text{strongly in }\ensuremath{\ensuremath{\operatorname H}\xspace}^{1}_0(\Omega),
  \label{eq:technical-subsequence-H1}
  \\
  \mathcal E_{h_k}u_{h_k}
  &\longrightarrow
  w
  &&\text{strongly in }\ensuremath{\ensuremath{\operatorname L}\xspace^{\infty}}(\Omega),
  \label{eq:technical-subsequence-Linf}
\end{align}
and
\begin{equation}
  \int_\Omega
  \nabla w\cdot\nabla\phi\,\mathrm dx
  =
  \int_\Omega
  f\phi\,\mathrm dx
  \qquad
  \text{for every }\phi\in \ensuremath{\ensuremath{\operatorname H}\xspace}^{1}_0(\Omega).
  \label{eq:technical-limit-Poisson}
\end{equation}

Let $\phi\in \ensuremath{\ensuremath{\operatorname C}\xspace^{\infty}}_c(\Omega)$ and set
\[
  \phi_{h_k}
  :=
  I_{h_k}\phi.
\]
For sufficiently large $k$, the support of $\phi$ lies inside
$\Omega_{h_k}$. Since $q_{h_k}^\ast$ is an admissible supergradient at
$u_{h_k}$,
\begin{align}
  \mathcal R_{h_k}(\phi_{h_k})
  -
  \mathcal R_{h_k}(u_{h_k})
  &\leq
  \sum_{i\in\mathcal N_{h_k}}
  \omega_iq_{h_k,i}^\ast
  \bigl(
    \phi_{h_k}(x_i)-u_{h_k}(x_i)
  \bigr)
  \nonumber\\
  &=
  \sum_{i\in\mathcal N_{h_k}}
  \omega_iq_{h_k,i}^\ast
  \phi_{h_k}(x_i)
  -
  a_{h_k}(u_{h_k},u_{h_k}).
  \label{eq:technical-pass-supergradient}
\end{align}

Lemma~\ref{lem:mass-lumping-error} gives
\[
  \sum_{i\in\mathcal N_{h_k}}
  \omega_iq_{h_k,i}^\ast
  \phi_{h_k}(x_i)
  -
  \int_{\Omega_{h_k}}
  s_{h_k}^\ast\phi_{h_k}\,\mathrm dx
  \longrightarrow0.
\]
Moreover,
\[
  \mathcal E_{h_k}\phi_{h_k}
  \longrightarrow
  \phi
  \qquad
  \text{uniformly on }\overline\Omega.
\]
Together with
\eqref{eq:technical-source-subsequence}, this gives
\begin{equation}
  \int_{\Omega_{h_k}}
  s_{h_k}^\ast\phi_{h_k}\,\mathrm dx
  \longrightarrow
  \int_\Omega
  f\phi\,\mathrm dx.
  \label{eq:technical-source-test-limit}
\end{equation}

The strong $\ensuremath{\ensuremath{\operatorname H}\xspace}^{1}_0(\Omega)$ convergence in
\eqref{eq:technical-subsequence-H1} gives
\[
  a_{h_k}(u_{h_k},u_{h_k})
  =
  \ensuremath{\left\|
    \nabla\mathcal E_{h_k}u_{h_k}
  \right\|}_{L^2(\Omega)}^2
  \longrightarrow
  \ensuremath{\left\|\nabla w\right\|}_{L^2(\Omega)}^2.
\]
Testing \eqref{eq:technical-limit-Poisson} with $\phi=w$ then yields
\begin{equation}
  a_{h_k}(u_{h_k},u_{h_k})
  \longrightarrow
  \int_\Omega
  fw\,\mathrm dx.
  \label{eq:technical-discrete-energy-limit}
\end{equation}

Lemma~\ref{lem:discrete-rearrangement-consistency}, applied to
\eqref{eq:technical-subsequence-Linf} and to the interpolants
$\phi_{h_k}$, gives
\begin{align}
  \mathcal R_{h_k}(u_{h_k})
  &\longrightarrow
  \mathcal R(w),
  \label{eq:technical-solution-rearrangement-limit}
  \\
  \mathcal R_{h_k}(\phi_{h_k})
  &\longrightarrow
  \mathcal R(\phi).
  \label{eq:technical-test-rearrangement-limit}
\end{align}
Passing to the limit in
\eqref{eq:technical-pass-supergradient} therefore yields
\begin{equation}
  \mathcal R(\phi)-\mathcal R(w)
  \leq
  \int_\Omega
  f(\phi-w)\,\mathrm dx
  \qquad
  \text{for every }\phi\in \ensuremath{\ensuremath{\operatorname C}\xspace^{\infty}}_c(\Omega).
  \label{eq:technical-limit-supergradient}
\end{equation}

Since $\ensuremath{\ensuremath{\operatorname C}\xspace^{\infty}}_c(\Omega)$ is dense in $L^1(\Omega)$,
$f\in \ensuremath{\ensuremath{\operatorname L}\xspace^{\infty}}(\Omega)$, and $\mathcal R$ is Lipschitz continuous on
$L^1(\Omega)$, the inequality extends to
\begin{equation}
  \mathcal R(\phi)-\mathcal R(w)
  \leq
  \int_\Omega
  f(\phi-w)\,\mathrm dx
  \qquad
  \text{for every }\phi\in L^1(\Omega).
  \label{eq:technical-limit-supergradient-L1}
\end{equation}

For $\phi\in \ensuremath{\ensuremath{\operatorname H}\xspace}^{1}_0(\Omega)$,
\eqref{eq:technical-limit-Poisson} gives
\[
  \int_\Omega
  \nabla w\cdot\nabla(\phi-w)\,\mathrm dx
  =
  \int_\Omega
  f(\phi-w)\,\mathrm dx.
\]
Combining this identity with
\eqref{eq:technical-limit-supergradient-L1} yields
\[
  \int_\Omega
  \nabla w\cdot\nabla(\phi-w)\,\mathrm dx
  \geq
  \mathcal R(\phi)-\mathcal R(w)
  \qquad
  \text{for every }\phi\in \ensuremath{\ensuremath{\operatorname H}\xspace}^{1}_0(\Omega).
\]
Thus $w$ minimises the continuum energy $\mathcal J$. By
Lemma~\ref{lem:continuum-variational-characterisation}, the unique
minimiser is the continuum Grad--Mercier solution, and therefore
\begin{equation}
  w
  =
  u.
  \label{eq:technical-subsequence-identification}
\end{equation}

It remains to identify the weak-star limit of the reconstructed sources.
Substituting $w=u$ into \eqref{eq:technical-limit-Poisson} gives
\begin{equation}
  \int_\Omega
  \nabla u\cdot\nabla\phi\,\mathrm dx
  =
  \int_\Omega
  f\phi\,\mathrm dx
  \qquad
  \text{for every }\phi\in \ensuremath{\ensuremath{\operatorname H}\xspace}^{1}_0(\Omega).
  \label{eq:technical-limit-source-equation}
\end{equation}
On the other hand, the weak formulation of the continuum
Grad--Mercier equation gives
\begin{equation}
  \int_\Omega
  \nabla u\cdot\nabla\phi\,\mathrm dx
  =
  \int_\Omega
  g\bigl(
    \ensuremath{\left|u\geq u(\cdot)\right|}
  \bigr)
  \phi\,\mathrm dx
  \qquad
  \text{for every }\phi\in \ensuremath{\ensuremath{\operatorname H}\xspace}^{1}_0(\Omega).
  \label{eq:technical-continuum-source-equation}
\end{equation}
Subtracting
\eqref{eq:technical-continuum-source-equation} from
\eqref{eq:technical-limit-source-equation} gives
\[
  \int_\Omega
  \ensuremath{\!\left[{
    f
    -
    g\ensuremath{\!\left({
      \ensuremath{\left|u\geq u(\cdot)\right|}
    }\right)}
  }\right]}
  \phi\,\mathrm dx
  =
  0
  \qquad
  \text{for every }\phi\in \ensuremath{\ensuremath{\operatorname C}\xspace^{\infty}}_c(\Omega).
\]
Therefore
\begin{equation}
  f
  =
  g\ensuremath{\!\left({
    \ensuremath{\left|u\geq u(\cdot)\right|}
  }\right)}
  \qquad
  \text{almost everywhere in }\Omega.
  \label{eq:technical-source-identification}
\end{equation}

The sequence $h_k\to0$ was arbitrary, and every convergent subsequence
has the same limit $u$. The subsequence principle therefore gives
\begin{align*}
  \mathcal E_hu_h
  &\longrightarrow
  u
  &&\text{strongly in }\ensuremath{\ensuremath{\operatorname H}\xspace}^{1}_0(\Omega),
  \\
  \mathcal E_hu_h
  &\longrightarrow
  u
  &&\text{strongly in }\ensuremath{\ensuremath{\operatorname L}\xspace^{\infty}}(\Omega)
\end{align*}
as $h\to0$.

The admissible equilibrium sources were also chosen arbitrarily. Every
weak-star accumulation point of their reconstructions is identified by
\eqref{eq:technical-source-identification}. Consequently, for every
choice of admissible equilibrium sources $q_h^\ast$,
\[
  \widetilde{s_h^\ast}
  \stackrel{*}{\rightharpoonup}
  g\ensuremath{\!\left({
    \ensuremath{\left|u\geq u(\cdot)\right|}
  }\right)}
  \qquad
  \text{in }\ensuremath{\ensuremath{\operatorname L}\xspace^{\infty}}(\Omega)
  \quad\text{as }h\to0.
\]

Finally, the maximum plateau has measure $\alpha$ by
Theorem~\ref{thm:continuum-well-posedness}, and the absence of
positive-measure level sets below the maximum follows from
Lemma~\ref{lem:continuum-no-secondary-plateaux}.
