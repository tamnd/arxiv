\label{thm:clt1}
    Fix $\beta>0$ and let $\Lambda_n=\{\lambda_1,\dots,\lambda_n\}\sim \textup{C$\beta$E}_n$. For any fixed $k$, the random vector
    \begin{equation}\label{eq:clt}
    \left(\sum_{i=1}^n\lambda_i, \sum_{i=1}^n\lambda_i^2, \cdots \sum_{i=1}^n\lambda_i^k\right)\Rightarrow \left(\xi_1,\xi_2,\dots,\xi_k\right)
    \end{equation}
    weakly as $n\to\infty$, where $\xi_j$'s are independent complex normal random variables with $\xi_j\sim{\mathbb C} N(0,\frac{2}{\beta}j)$ for each $j$.
<!-- end thmx-1 -->
<!-- theorem-1: body_md -->
\label{thm:main}
    Fix $\beta>0$. There exists a coupling of a sequence of random unitary matrices ${\mathcal C}_n,n\ge 1$ such that $ \operatorname{spec}\{{\mathcal C}_n\}\stackrel{d}{=} \operatorname{spec}\{\mathsf{Circ}_{n,\beta}\} \sim \text{C$\beta$E}_n$, 
    and for any fixed $k\ge 1$ we have 
    \begin{equation}\label{eq:conv_vec}  \left(\operatorname{tr}({\mathcal C}_n),\operatorname{tr}({\mathcal C}_n^2),\cdots,\operatorname{tr}({\mathcal C}_n^k)\right)\to \left(T_1,T_2,\dots,T_k\right) \quad \text{a.s. and in $L^2$ as $n\to\infty$,}
    \end{equation}
    where $T_j, j\ge 1$ is a sequence of independent complex normal random variables with $T_j\sim{\mathbb C} N(0,\frac{2}{\beta}j)$.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:1st}
    Under the coupling in Theorem \ref{thm:main}, the random variable $T_j$ can be represented as the trace of the $j$-th power of an infinite CMV matrix ${\mathcal C}_\infty$. In particular, 
    \begin{align*}

<!-- definition-1: body_md -->
\label{def:CMV}
 Let $\{\alpha_0,\alpha_1,\cdots\}$ be a sequence of complex coefficients with  $|\alpha_k|\le 1$. Define
\[
\Xi_k = \begin{pmatrix}
\bar\alpha_k & \rho_k\\ \rho_k& -\alpha_k
\end{pmatrix},\qquad \rho_k=\sqrt{1-|\alpha_k|^2},
\]
and set $\Xi_{-1}=(1)$. Then we define the {CMV matrix} corresponding to $\{\alpha_0,\alpha_1,\cdots\}$ as
\begin{align}\label{eq:CMV}
	{\mathcal C}(\alpha_0,\alpha_1,\cdots) :={\mathcal L}{\mathcal M},
\end{align}
where ${\mathcal L}, {\mathcal M}$ are block-diagonal matrices 
\begin{align*}
{\mathcal L}=\operatorname{diag}\left(\Xi_0,\Xi_2\cdots\right),\qquad {\mathcal M}=\operatorname{diag}\left(\Xi_{-1},\Xi_1\cdots\right).
\end{align*}
In the case when $|\{\alpha_0,\alpha_1,\cdots\}|=n$, we define $\Xi_{n-1}=(\bar\alpha_{n-1})$ to be $1\times 1$ matrix and set ${\mathcal L},{\mathcal M}$ to be $n\times n$ matrices
\begin{align*}
{\mathcal L}=\operatorname{diag}\left(\Xi_0,\Xi_2\cdots,\Xi_{2\lfloor\tfrac{n-1}{2}\rfloor}\right),\qquad {\mathcal M}=\operatorname{diag}\left(\Xi_{-1},\Xi_1\cdots,\Xi_{2\lfloor\tfrac{n}{2}\rfloor-1}\right).
\end{align*}

Recall Assumption \ref{coupling}. We denote the traces of ${\mathcal C}_\infty$ by $T_k:=\operatorname{tr}({\mathcal C}_\infty^k)$ if the limit exists. In particular, we have  
    \[
    T_1=-\lim_{n\to\infty}\sum_{j=-1}^n\alpha_j\bar\alpha_{j+1}=-\sum_{j= -1}^\infty  \alpha_j\bar \alpha_{j+1}
    \]
    with $\alpha_{-1}=-1$. Note that the limit exists since ${\rm   E}[|\alpha_n\bar \alpha_{n+1}|^2]=O(n^{-2})$. By Proposition \ref{prop:Gauss-Stein}, it suffices to prove that for any $f\in C^1({\mathbb C})$ with bounded first derivative, we have 
    \begin{align}\label{eq:1st_goal}
        {\rm   E}[\overline{T} \!\,_1 f(T_1)]=\tfrac{2}{\beta} {\rm   E}[\partial_{z}f(T_1)],\qquad  {\rm   E}[ T_1 f( T_1)]=\tfrac{2}{\beta} {\rm   E}[\partial_{\bar z}f( T_1)].
    \end{align}
    Note that for $j\ge 0$ we have 
    \begin{align*}
    \partial_{\alpha_j} f(T_1)& = -\bar \alpha_{j+1} \partial_zf(T_1)-\bar\alpha_{j-1}\partial_{\bar z}f(T_1),\\
    \partial_{\bar \alpha_j}f(T_1)& = -\alpha_{j-1}\partial_zf(T_1)-\alpha_{j+1}\partial_{\bar z}f(T_1).
\end{align*}
Recall $\beta_j=\frac{\beta}{2}(j+1)$.
By the Stein's identities \eqref{eq:beta-Stein_simple}, we have 
\begin{align}
    &\tfrac{\beta}{2}{\rm   E}[\bar\alpha_j \alpha_{j+1}f(T_1)]\notag\\
    &\quad = {\rm   E}[\beta_{j+1}\bar\alpha_j \alpha_{j+1}f(T_1)]-{\rm   E}[\beta_j\bar\alpha_j\alpha_{j+1}f(T_1)]\notag\\
    &\quad={\rm   E}[\rho_{j+1}^2\bar \alpha_j \partial_{\bar \alpha_{j+1}}f(T_1)]-{\rm   E}[\rho_j^2\alpha_{j+1}\partial_{\alpha_j}f(T_1)]\notag\\
                                                                                        
&\quad={\rm   E}\left[\partial_zf(T_1)\big(|\alpha_{j+1}|^2\rho_j^2-|\alpha_j|^2\rho_{j+1}^2)\right]+{\rm   E}\left[\partial_{\bar z}f(T_1)\big(\rho_j^2\bar\alpha_{j-1}\alpha_{j+1}-\rho_{j+1}^2\bar \alpha_j\alpha_{j+2}\big)\right]\label{eq:telescoping}.
\end{align}
Note that the terms on the right hand side of \eqref{eq:telescoping} are telescoping series.
Together with
\[
\tfrac{\beta}{2}{\rm   E}[\alpha_0f(T_1)] ={\rm   E}[\rho_0^2\partial_{\bar \alpha_0}f(T_1)]={\rm   E}[\rho_0^2\partial_{z}f(T_1)-\rho_0^2\alpha_1\partial_{\bar z}f(T_1)],

\[|\alpha_{j+1}|^2\rho_j^2-|\alpha_j|^2\rho_{j+1}^2 = |\alpha_{j+1}|^2 - |\alpha_j|^2,
\]
we get that 
\begin{align*}

\end{align*}
as claimed. This proves the first equality of \eqref{eq:1st_goal}, the second one can be proved similarly by applying Stein's method to the conjugated terms $\alpha_j\bar \alpha_{j+1}$ for $j\ge -1$. By Proposition \ref{prop:Gauss-Stein}, we conclude that $T_1\sim {\mathbb C} N(0,\frac{2}{\beta})$.

<!-- proposition-8: body_md -->
\label{prop:ab}
We have $\alpha_n\to 0$ a.s.~and in $L^2$ as $n\to\infty$.
For any fixed $k$, we have $b_k^{(n)}\to 0$ a.s.~as $n\to\infty$. Moreover, we have

 {\rm   E}[(b_k^{(n)})^2]\le C_{\beta}\frac{k^2}{n-k}, \qquad k\le n-1,
\]
where $C_\beta$ is independent of $k$ and $n$. 
Consequently, $T_k^{(n)}-\mathtt{T}^{(n)}_k\to 0$ a.s.~and in $L^2$ as $n\to\infty$.

<!-- proof-3: body_md -->
By definition, we have
    \begin{align*}
        \mbox{\rm P}(|\alpha_n|^2\ge x) = (1-x)^{\frac{\beta}{2}(n+1)},
    \end{align*}
    which is summable in $n$ for $x>0$. This shows that $\alpha_n\to 0$ a.s.~by the Borel-Cantelli Lemma. The $L^2$ convergence follows from the exact computation
    \begin{align}\label{eq:alpha_moments}
        {\rm   E}[|\alpha_n|^2]= \frac{1}{1+\frac{\beta}{2}(j+1)}.
    \end{align}
    (Indeed, we have $\alpha_n\to 0$ in $L^p$ for every $0<p<\infty$.)

    We now turn to the estimate of $b_k^{(n)}$.  By the Szeg\H o recursion \eqref{eq:Szego1}, we have 
    \begin{align*}
        B_{n+1}(z)= zB_n(z)-\bar\alpha_n +\alpha_n zB_n(z)B_{n+1}(z).
    \end{align*}
    By comparing the Taylor coefficients, we have 
    \begin{align*}
        b_k^{(n+1)} = b_{k-1}^{(n)}+\alpha_n [z^{k-1}](B_nB_{n+1}).
    \end{align*}
    Using the fact that $|(B_nB_{n+1})(z)|<1$ for $z\in{\mathbb D}$, and Cauchy's formula, we have $|[z^{k-1}](B_nB_{n+1})|\le 1$. Iterating this argument leads to the (rough) bound 
    \begin{align*}
        |b_k^{(n+1)}|\le |b_{k-1}^{(n)}|+|\alpha_n|\le |b_{k-2}^{(n-1)}|+|\alpha_n|+|\alpha_{n-1}|\le  \cdots\le \sum_{j=n-k}^{n}|\alpha_j|.
    \end{align*}
    This shows that $b_k^{(n)}\to 0$ a.s.~as $n\to\infty$ for every fixed $k$.
    By Minkowski's inequality and \eqref{eq:alpha_moments}, we have
    \begin{align*}
        \|b_k^{(n)}\|_2\le \sum_{j=n-k-1}^{n-1}\|\alpha_j\|_2\le \frac{k+1}{\sqrt{1+\frac{\beta}{2}(n-k)}}.
    \end{align*}
    This proves the $L^2$ convergence of $b_k^{(n)}$.
    Recall the definitions $T_k^{(n)}=-k\bar t_{k}^{(n)}$ and $\mathtt{T}_k^{(n)}=-k\overline{\mathtt{t}} \!\,_k^{(n)}$, the last statement follows from \eqref{eq:t_diff} and the convergence of $b_k^{(n)}$.

<!-- proof-4: body_md -->
Recall the recursion \eqref{eq:logPhi*}. Since $B_n$ is $\mathcal{F}_n$ measurable, $\alpha_n$ is rotationally invariant and independent of $\mathcal{F}_n$, we conclude that $\log \Phi_n^*(z),n\ge 0$ is a martingale for any fixed $z\in{\mathbb D}$. By  the same argument, one can show that for any fixed $k$, $\mathtt{t}_k^{(n)},n\ge 1$ is a $\mathcal{F}$-martingale. 
    (Note that by a more careful estimate of exponential moments of $\log\Phi_n^*$, one can show that $\log\Phi_n^*(z)\to\log\Phi_\infty^*(z)$  a.s.~as $n\to\infty$, uniformly on compacts sets of ${\mathbb D}$, see Proposition 3.1 in \cite{CN_GMC} for more details.) 

    For our purposes, we will derive the convergence using Proposition \ref{prop:ab}. Fix $k\ge 1$, we define the martingale difference
    \begin{align*}
        m_k^{(n)}:=\mathtt{t}_{k}^{(n+1)}-\mathtt{t}_k^{(n)}=-\sum_{j=1}^k \frac{\alpha_n^j}{j}[z^{k-j}]B_n(z)^j.
    \end{align*}
    Using the independence and rotational invariance of $\alpha_n$, we have 
    \begin{align*}
        {\rm   E}[|m_k^{(n)}|^2] = \sum_{j=1}^k\frac{{\rm   E}[|\alpha_n|^{2j}]}{j^2}{\rm   E}\left[\big|[z^{k-j}]B_n(z)^j\big|^2\right]\le C_\beta\frac{k^2}{n^2}.
    \end{align*}
    Indeed, when $k=1$, we have ${\rm   E}[|\alpha_n|^2]=O(n^{-1})$ and ${\rm   E}[|b_{0}^{(n)}|^2]={\rm   E}[|\alpha_{n-1}|^2]=O(n^{-1})$. For $k\ge 2$, we simply use the bounds ${\rm   E}[|\alpha_n|^{2j}]=O(n^{-j})$ and ${\rm   E}\left[\big|[z^{k-j}]B_n(z)^j\big|^2\right]\le 1$. Therefore, $\sum_{n}{\rm   E}[|m_k^{(n)}|^2] <\infty$ and we have $\mathtt{t}_{k}^{(n)}$ is bounded in $L^2$. This implies that there exists $t_k$ such that $\mathtt{t}_k^{(n)}\to t_k$ and $\mathtt{T}_k^{(n)}=-k\bar {\mathtt{t}}_k^{(n)}\to -k\bar t_k$ a.s~and in $L^2$ as $n\to\infty$. 
    
    It remains to prove that $-k\bar t_k = T_k=\operatorname{tr}({\mathcal C}_\infty^k)$. Define $S_k^{(n)}:=\sum_{j=1}^n [{\mathcal C}_\infty^k]_{jj}$. Since ${\mathcal C}_\infty$ is five-diagonal, $S_k^{(n)}$ is well defined, and we have $T_k=\lim_{n\to\infty} S_k^{(n)}$ whenever the limit exists. Therefore, it suffices to $S_k^{(n)}-\mathtt{T}_k^{(n)}\to 0$ a.s.~and in $L^2$ as $n\to\infty$ for any fixed $k$. The CMV matrices ${\mathcal C}_n$ and ${\mathcal C}_\infty^{(n)}$ are five-diagonal, and only differ by the last Verblunsky coefficient. Therefore, we have $[({\mathcal C}_\infty^{(n)})^k]_{jj}=[{\mathcal C}_n^k]_{jj}$ for $j\le n-k$. Every entry of ${\mathcal C}_\infty,{\mathcal C}_n$ has modulus of at most one, and there are at most $k5^k$ paths that contains the random variable $\eta$ in ${\mathcal C}_n$. Thus, we get the following crude bound
    \[
    |S_k^{(n)}-\mathtt{T}_k^{(n)}| \le k5^k\max_{|j-n|\le  k}|\alpha_j|\le  k5^k\sum_{j=n-k}^{n+k}|\alpha_j|.
    \]
   Applying Proposition \ref{prop:ab} shows that the error term $ |S_k^{(n)}-\mathtt{T}_k^{(n)}| \to 0$ a.s.~and in $L^2$ as $n\to\infty$. This finishes the proof.

Recall that $|\alpha_j|^2\sim \textup{Beta}(1,\beta_j)$ with $\beta_j=\frac{\beta}{2}(j+1)$, and recall that the Poisson bracket is defined as
\begin{align*}
    \{F,G\}_{n} = i\sum_{j=-1}^{n-1}\rho_j^2(\partial_{\bar \alpha_j}F\partial_{ \alpha_j}G-\partial_{ \alpha_j}F\partial_{\bar \alpha_j}G).
\end{align*}
Taking expectation for each term and applying Proposition \ref{prop:beta-Stein}, we have 
\begin{align*}
    {\rm   E}[\rho_j^2\partial_{\bar \alpha_j}F\partial_{ \alpha_j}G] &=  \beta_j {\rm   E}[\alpha_j F\partial_{\alpha_j}G]-{\rm   E}[\rho_j^2 F\partial_{\bar \alpha_j}\partial_{\alpha_j}G],\\

\end{align*}
wheneve the Wirtinger derivatives exist and have finite expectation.

\begin{align*}
    {\rm   E}[\rho_j^2(\partial_{\bar \alpha_j}F\partial_{ \alpha_j}G-\partial_{ \alpha_j}F\partial_{\bar \alpha_j}G)] = \beta_j {\rm   E}[F(\alpha_j\partial_{\alpha_j}-\bar\alpha_j\partial_{\bar \alpha_j})G].
\end{align*}

\begin{align}\label{eq:Poisson-D}
    {\rm   E}[\{F,G\}_n] = i \frac{\beta}{2}{\rm   E}[F\mathcal{D}_n G],
\end{align}

\begin{align*}
    \mathcal{D}_n = \sum_{j= -1}^{n-1}(j+1)(\alpha_j\partial_{\alpha_j}-\bar \alpha_j\partial_{\bar \alpha_j}).
\end{align*}
We claim that the traces $\mathtt{T}_k^{(n)}$ can be viewed as the 'eigenvectors' of ${\mathcal D}_n$. The proof of the claim is postponed after the proof of the theorem.
\begin{claim}\label{claim:D}
For any fixed $k$, we have $\mathcal{D}_n\overline{\mathtt{T}} \!\,_k^{(n)} = k\overline{\mathtt{T}} \!\,_k^{(n)}$.
                    
                                                                                                                          
                  
\end{claim}
Let ${\mathbf T}^{(n)}=(\mathtt{T}_1^{(n)},\mathtt{T}_2^{(n)},\cdots,\mathtt{T}_k^{(n)})$, $F\in C^1({\mathbb C}^k)$ with bounded Wirtinger derivatives, and choose 
$G(\mathbf{T}^{(n)})=\overline{\mathtt{T}} \!\,_j^{(n)}$. By \eqref{eq:Poisson-D} and Claim \ref{claim:D}, we have  
    \begin{align}\label{eq:remove_Poisson}
        {\rm   E}\left[\{F(\mathbf{T}^{(n)}),\overline{\mathtt{T}}_j^{(n)}\}_n\right] &= i\frac{\beta}{2}{\rm   E}\left[F(\mathbf{T}^{(n)})\mathcal{D}_n\overline{\mathtt{T}}_j^{(n)}\, \right]=i\frac{\beta j}{2}{\rm   E}\left[F(\mathbf{T}^{(n)})\overline{\mathtt{T}}_j^{(n)}\,\right].
    \end{align}
    On the other hand, since $\mathtt{T}_j^{(n)} = -j\overline{\mathtt{t}} \!\,_j^{(n)}$, it follows from \eqref{id:logPoisson3} that 
    \begin{equation}\label{eq:remove_Poisson2}
    \{F(\mathbf{T}^{(n)}),\overline{\mathtt{T}} \!\,_j^{(n)}\}_n = i\sum_{m=1}^k mj\left(\delta_{mj}-\sum_{l=1}^{\min\{m,j\}}\bar b_{m-l}^{(n)}b_{j-l}^{(n)}\right)\partial_{ z_m}F(\mathbf{T}^{(n)}).
    \end{equation}
    By Propositions \ref{prop:ab} and \ref{prop:mg}, we have 
    \[
    \bar b_{m-l}^{(n)} b_{j-l}^{(n)}\to 0,\quad l\le \min\{j,m\}, \qquad \mathtt{T}_j^{(n)}\to T_j,\quad 1\le j\le k
    \]
    a.s.~and in $L^2$ as $n\to\infty$.

    We may now pass to the limit. Let $\mathbf{T}=(T_1,\cdots,T_k)$, by \eqref{eq:remove_Poisson} and \eqref{eq:remove_Poisson2} we have
    \begin{align*}
        {\rm   E}[\overline{T} \!\,_jF(\mathbf{T})]  &= \lim_{n\to\infty}{\rm   E}[\overline{\mathtt{T}} \!\,_j^{(n)}F(\mathbf{T}^{(n)})]\\
        &=\lim_{n\to\infty}-i\frac{2}{\beta j}{\rm   E}\left[\{F(\mathbf{T}^{(n)}),\overline{\mathtt{T}}_j^{(n)}\}_n\right] \\
        &=\lim_{n\to\infty}\frac{2}{\beta j}\sum_{m=1}^k {\rm   E}\left[mj\Big(\delta_{mj}-\sum_{l=1}^{\min\{m,j\}}\bar b_{m-l}^{(n)}b_{j-l}^{(n)}\Big)\partial_{ z_m}F(\mathbf{T}^{(n)})\right]\\
        &=\frac{2j}{\beta}{\rm   E}\left[\partial_{z_j}F(\mathbf{T})\right]
    \end{align*}
    as desired. Applying the same argument to the complex conjugates gives that 
    \begin{align*}
        {\rm   E}[T_j F(\mathbf{T}) ]  = \frac{2j}{\beta}{\rm   E}[\partial_{\bar z_j}F(\mathbf{T})]. 
    \end{align*}
    Thus, by Stein's lemma in Proposition \ref{prop:Gauss-Stein}, we conclude that $T_j,1\le j\le k$ are independent random variables such that $T_j\sim{\mathbb C} N(0,\frac{2j}{\beta})$. This finishes the proof.

\label{claim:D}
For any fixed $k$, we have $\mathcal{D}_n\overline{\mathtt{T}} \!\,_k^{(n)} = k\overline{\mathtt{T}} \!\,_k^{(n)}$.

<!-- proof-8: body_md -->
Recall that $\mathtt{T}_k^{(n)}=-k\overline{\mathtt{t}} \!\,_k^{(n)}$, where $\mathtt{t}_k^{(n)}$ is the $k$-th Taylor coefficient of $\log \Phi_n^*$. We claim that 
\begin{align}\label{eq:D-Phi}
{\mathcal D}_n\Phi_n^*(z)  = z\partial_z\Phi_n^*(z),\qquad {\mathcal D} \Phi_n(z) = (z\partial_z -n) \Phi_n.
\end{align}
From this, it follows that 
\[
{\mathcal D}_n \log \Phi_n^*(z) = \frac{{\mathcal D}_n\Phi_n^*(z)}{\Phi_n^*(z)} = \frac{z\partial_z \Phi_n^*(z)}{\Phi_n^*(z)} = z\partial_z \log\Phi_n^*(z).
\]
By comparing the Taylor coefficients again, we obtain that
\[
{\mathcal D}_n \mathtt{t}_k^{(n)}=[z^{k-1}](\partial_z\log \Phi_n^*(z)) = k\mathtt{t}_k^{(n)},
\]
which implies ${\mathcal D}_n \overline{\mathtt{T}} \!\,_k^{(n)}=-k\overline{\mathtt{T}} \!\,_k^{(n)}$. 

It remains to prove \eqref{eq:D-Phi}. We proceed by induction on $n$. The case when $n=0$ is trivial. Assuming the identities \eqref{eq:D-Phi} hold for $k\le n$. Then, by Szeg\H o recursion \eqref{eq:Szego1} we have
\begin{align*}
    {\mathcal D}_{n+1}\Phi_{n+1}^*(z) &= {\mathcal D}_{n+1}(\Phi_n^*(z)-\alpha_n z\Phi_n(z))\\
    &= {\mathcal D}_n\Phi_n^*(z)- z\Phi_n(z){\mathcal D}_{n+1}(\alpha_n) -\alpha_n z{\mathcal D}_n\Phi_n(z),
\end{align*}
where we used the fact that $\Phi_n,\Phi_n^*$ only depend on $\alpha_0,\alpha_1,\cdots,\alpha_{n-1}$ in the second step. Since ${\mathcal D}_{n+1}\alpha_n=(n+1)\alpha_n$, by the induction hypothesis \eqref{eq:D-Phi} we get 
\begin{align*}
     {\mathcal D}_{n+1}\Phi_{n+1}^*(z) 
                                                                                           
     &=z\partial_z\Phi_n^*(z) - \alpha_nz\Phi_n(z) - \alpha_n z^2\partial_z\Phi_n(z)\\
     &=z\partial_z(\Phi_n^*(z)-\alpha_nz\Phi_n(z))=z\partial_z\Phi_{n+1}^*(z)
\end{align*}
as claimed. Repeating the same argument gives the second identity in \eqref{eq:D-Phi}. This completes the induction, and hence the proof.

<!-- remark-2: body_md -->
\label{rmk:infinite}
Using the infinitesimal generator argument, one may extend the definition of ${\mathcal D}_n$ to the infinite dimensional setting
\[
{\mathcal D}_\infty = \sum_{j=-1}^\infty (j+1)(\alpha_j\partial_{\alpha_j}-\bar\alpha_j\partial_{\bar\alpha_j}),
\]
and prove that  $\mathcal{D}_\infty\overline{T} \!\,_k = k\overline{T} \!\,_k$. Then the variables $t_k$ can be viewed as canonical coordinates or 'orthonormal eigenvector' for the infinite Ablowitz-Ladik Poisson bracket, in the sense that 
\[
\{t_j,t_k\}_\infty = 0,\qquad \{\bar t_j,t_k\}_\infty=i\delta_{jk}.
\]
Consequently, we have $\{F({\bf T}),\overline{T} \!\,_k\}_\infty = ik^2\partial_{z_k}F({\bf T})$ assuming the infinite AL Poisson bracket converges. The exact integration-by-parts formulas for the Verblunsky coefficients in Proposition \ref{prop:beta-Stein} then allow us to connect ${\rm   E}[\{F({\bf T}),\overline{T} \!\,_k\}_\infty]$ to ${\rm   E}[F({\bf T})\overline{T} \!\,_k]$. The Gaussian limits then follow from Stein's method.

<!-- remark-3: body_md -->
Using our method, we expect the convergence in Theorem \ref{thm:main} to remain valid when $k=k(n)$ tends to infinity. Indeed, by the estimate of ${\rm   E}[|t_k-\mathtt{t}_k^{(n)}|^2]\asymp \sum_{j\ge n}j^{-2}=O(n^{-1})$, we have ${\rm   E}[\sum_{j=1}^k|\mathtt{T}_j^{(n)}-T_j|^2]\asymp n^{-1}\sum_{j=1}^k j^2\asymp \frac{k^3}{n}$. This suggests that the convergence \eqref{eq:conv_vec} should still hold when $k=o(n^{1/3})$. However, this rate and range does not improve the result of Johansson-Lambert \cite{JohanssonLambert}, where they prove that the Wasserstein distance $W_2$ between the vector of traces and the corresponding Gaussian vectors is $O(k^2/n)$, which tends to $0$ when $k=o(n^{1/2})$.

The case when $n=0$ is trivial. We proceed by induction and assume the statements hold for $k\le n$. By the definition of the Poisson bracket we have 
    \[
    \{f,g\}_{n+1} = \{f,g\}_n + i\rho_n^2(\partial_{\bar \alpha_n}f\partial_{\alpha_n}g - \partial_{\alpha_n}f\partial_{\bar\alpha_n}g).
    \]
    Hence, by the Szeg\H o recursion \eqref{eq:Szego1}  we get 
    \begin{align*}
        \{\Phi_{n+1}(z),\Phi_{n+1}(w)\}_{n+1} &= \{z\Phi_n(z)-\bar\alpha_n\Phi_n^*(z),w\Phi_n(w)-\bar\alpha_n\Phi_n^*(w)\}_{n+1}\\
        &=-\bar\alpha_n \left(z\{\Phi_n(z),\Phi_n^*(w)\}_{n+1}-w\{\Phi_n(w),\Phi_n^*(z)\}_{n+1}\right)=0,
    \end{align*}
    where the last step follows from the induction hypothesis \eqref{id:Poisson2} and $\{\Phi_n,\Phi_n^*\}_{n+1}=\{\Phi_n,\Phi_n^*\}_{n}$. Similarly, we have $\{\Phi_{n+1}^*(z),\Phi_{n+1}^*(w)\}_{n+1}=0$.

    It remains to prove \eqref{id:Poisson2} at $n+1$. By Szeg\H o recursion \ref{eq:Szego1} again, we have
    \begin{align}
        \{\Phi_{n+1}(z),\Phi_{n+1}^*(w)\}_{n+1} &= \{z\Phi_n(z)-\bar \alpha_n\Phi_n^*(z), \Phi_{n}^*(w)-\alpha_nw\Phi_n(w)\}_{n+1}\notag\\
        &=z\{\Phi_n(z),\Phi_n^*(w)\}_n-|\alpha_n|^2 w\{\Phi_n(w),\Phi_n^*(z)\}_n+ w\Phi_n^*(z)\Phi_n(w)\{\bar\alpha_n,\alpha_n\}_{n+1}\notag\\
        &=\rho_n^2 \frac{izw}{z-w}(\Phi_n(z)\Phi_n^*(w)-\Phi_n(w)\Phi_n^*(z))+i\rho_n^2w\Phi_n^*(z)\Phi_n(w)\notag\\
                                                                                              
        &=\frac{iw}{z-w}\rho_n^2(z\Phi_n(z)\Phi_n^*(w)-w\Phi_n(w)\Phi_n^*(z))\label{eq:indunction}. 
    \end{align}
    On the other hand, the Szeg\H o recursion gives that
    \[ 
    \Phi_{n+1}(z)\Phi_{n+1}^*(w)-\Phi_{n+1}(w)\Phi_{n+1}^*(z)=\rho_n^2(z\Phi_n(z)\Phi_n^*(w)-w\Phi_n(w)\Phi_n^*(z)).
    \]
    Plugging into the right hand side of \eqref{eq:indunction} completes the induction and the proof.
