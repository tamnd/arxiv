<!-- remark-1: body_md -->
\label{rem:finit-back}
    After Lemma \ref{lem:guarded-backtracking}, we show that whenever the current iterate satisfies $\Delta(w^t)\leq \Gamma_t$, every $\alpha\leq h_t$ is guaranteed to satisfy the next diameter bound $\Delta(w^{t+1})\leq \Gamma_{t+1}$. Thus, the backtracking line search terminates after finitely many steps at each iteration, and the floor $\ell_t=\tau\min\{\bar\alpha_t,h_t\}$ provides a controlled lower scale for the accepted step. Detailed proof is provided in Appendix~\ref{app:proof_backtrack}.

<!-- theorem-1: body_md -->
\label{thm:unconditional}
Consider the GW Problem~(\ref{main_problem}) with symmetric $D_X\in\mathbb{R}^{n\times n}$,
$D_Y\in\mathbb{R}^{m\times m}$, and marginals $\mu\in\mathbb{R}^{n}_{++}$,
$\nu\in\mathbb{R}^{m}_{++}$ satisfying $\mathbf{1}_n^{\top}\mu=\mathbf{1}_m^{\top}\nu=1$,
and let $M_G:=\sup_{X\in\mathcal{C}_1\cup\mathcal{C}_2}\|G(X)\|_\infty<\infty$.
Let the initialization be entrywise positive, let the burn-in length $k_0\in\mathbb{N}$ be finite, and let the prefix penalty $\rho_{0}>0$, the proposal scale $\alpha_0>0$, the backtracking factor $\tau\in(0,1)$, the growth bound $B_\Delta\ge 8M_G$, and the guard slope and delayed-power exponent $ a\in (0,{2}/{3})$, and $\beta\in ({2}/{3}+{a}/{2},\,1)$. Then, the tail iterates $\{(\pi^{t},w^{t})\}_{t\ge 0}$ generated by Algorithm~\ref{alg:adaptive-delayed-power}, indexed from the switch at $k=k_0$, have the following properties:
\begin{enumerate}
  \item $\displaystyle\liminf_{t\to\infty}\bigl\|R_t\bigr\|_{M_t}=0$,
        with $\rho_t:=1/\alpha_t$ and $\|U\|_{M_t}^{2}:=\sum_{i,j}m^{t}_{ij}U_{ij}^{2}$, where
        $m^{t}_{ij}:=\min\{w^{t}_{ij},w^{t+1}_{ij}\}$;
  \item $\displaystyle\lim_{t\to\infty}\bigl\|x^{t+1}-\pi^{t+1}\bigr\|_F+\bigl\|\pi^{t+1}-w^{t+1}\bigr\|_F = 0$, where $x^{t}:=\mathcal{P}_{\mathcal{C}}(\pi^{t})$ denotes the
        Euclidean projection of $\pi^{t}$ onto $\mathcal{C}=\mathcal{C}_1\cap\mathcal{C}_2$;
        in particular, $\{x^{t}\}$, $\{\pi^{t}\}$ and $\{w^{t}\}$ have the same set of
        accumulation points.
\end{enumerate}
<!-- end theorem-1 -->
<!-- lemma-1: body_md -->
\label{lem:aae}
Let ${(\pi^t,w^t)}_{t\geq0}$ be the iterates generated by Algorithm~\ref{alg:adaptive-delayed-power}. Under the setting of Theorem \ref{thm:unconditional}, 
$\sum_{t=0}^{\infty}\bigl|D_h(\pi^{t+1},w^t)-D_h(w^t,\pi^{t+1})\bigr|
+
\bigl|D_h(w^{t+1},\pi^{t+1})
      -D_h(\pi^{t+1},w^{t+1})\bigr|
=O(\eta_t^3)<\infty$, i.e., AAE holds.
<!-- end lemma-1 -->
<!-- assumption-1: body_md -->
\label{assum:support}
For every $\{t_\ell\}$ such that $\|R^{t_\ell}\|_{M_{t_\ell}}\to0$ and $x^{t_\ell+1}=\mathcal P_{\mathcal C}(\pi^{t_\ell+1})\to\bar x$, define the active support of $\bar x$ as $S(\bar x):=\{(i,j):\bar x_{ij}>0\}$. Then for every $(i,j)\notin S(\bar x)$,
 $\liminf_{\ell\to\infty}R^{t_\ell}_{ij}\ge 0$.
<!-- end assumption-1 -->
<!-- proposition-1: body_md -->
\label{cor:mono-suff} 
Let $\{t_\ell\}$ be subsequence with $x^{t_\ell+1}\to\bar x$.  Under the setting of Theorem \ref{thm:unconditional}, Assumption~\ref{assum:support} holds along this subsequence if for every $(i,j)\notin S(\bar x)$,
\[\liminf_{\ell\to\infty}
\rho_{t_\ell}
\log \frac{w^{t_\ell}_{ij}}{w^{t_\ell+1}_{ij}}
\geq 0.\]
In particular, it suffices that
$w^{t_\ell+1}_{ij}\leq w^{t\ell}_{ij}(1+\varepsilon_\ell^{ij})$
with $\varepsilon_\ell^{ij}=o(\alpha_{t_\ell})$. Thus, eventual nonincrease of every inactive entry along the subsequence is sufficient.
<!-- end proposition-1 -->
<!-- theorem-2: body_md -->
\label{thm:stationarity}
Consider the GW Problem \ref{main_problem} and the A-KL-BAPG iterates ${(\pi^t,w^t)}_{t\geq0}$ generated by Algorithm~\ref{alg:adaptive-delayed-power} under the conditions of Theorem~\ref{thm:unconditional}. Let Assumption~\ref{assum:support} hold. Then every residual-vanishing subsequence has a limit $\bar x\in\mathcal C$ satisfying $0\in\nabla F(\bar x)+N_{\mathcal C}(\bar x)$. In particular, the iterates admit at least one stationary accumulation point.
<!-- end theorem-2 -->
<!-- proposition-2: body_md -->
\label{prop:no-universal-power-lower-bound}
Under the conditions of Theorem \ref{thm:unconditional}, there exists a valid $2\times2$ instance, constructed in Appendix \ref{app:proof_lower_bound}, such that
any guarded tail of A-KL-BAPG on this instance satisfies $\sum_{s=0}^{T-1}\alpha_s=O(\log T)$.
Consequently, no theorem using only the projective-diameter guard can guarantee an eventual lower bound $\alpha_t\geq c(t+1)^{-\beta}$ for any $c>0$ and any $\beta<1$.
<!-- end proposition-2 -->
<!-- proposition-3: body_md -->
\label{prop:automatic-marginal-feasibility}
Let $\{(\pi^t,w^t)\}_{t\geq0}$ be the tail iterates generated by Algorithm~\ref{alg:adaptive-delayed-power}. Under the conditions of Theorem \ref{thm:unconditional},  $\|w^t\mathbf 1_m-\mu\|_1 = O(\eta_t)$ and $\|(\pi^{t+1})^\top\mathbf 1_n-\nu\|_1=O(\eta_t)$.
Moreover, for all sufficiently large~$t$,
$D_h(\pi^{t+1},w^t) = O(\eta_t^2)$
and $D_h(\pi^{t+1},w^{t+1}) = O(\eta_t^2)$.
<!-- end proposition-3 -->
<!-- proposition-4: body_md -->
\label{prop:cubic-main}
Let $\{(\pi^t,w^t)\}_{t\geq0}$ be the tail iterates generated by Algorithm~\ref{alg:adaptive-delayed-power}.  Under the conditions of Theorem \ref{thm:unconditional}, for all sufficiently large $t$,
$|A_t^{\rm row}|+|A_t^{\rm col}|=O(\eta_t^3)$
and $|A_t|=O(\eta_t^3)$.
<!-- end proposition-4 -->
<!-- proposition-5: body_md -->
\label{prop:lyap-main}
For $\{(\pi^t,w^t)\}_{t\geq0}$ generated by Algorithm~\ref{alg:adaptive-delayed-power}, under the conditions of Theorem \ref{thm:unconditional},
\[\sum_{t=1}^{\infty}\rho_t\left[D_h(\pi^t,\pi^{t+1})+D_h(w^t,w^{t+1})\right]<\infty.\]
<!-- end proposition-5 -->
<!-- proposition-6: body_md -->
\label{prop:residual}
Let $\{R_t\}_{t\geq0}$ be the corrected residuals associated with the tail iterates generated by Algorithm~\ref{alg:adaptive-delayed-power} and suppose the conditions of Theorem \ref{thm:unconditional} hold. Then there exists an integer $K_R$ such that
$\sum_{t = K_R}^\infty \alpha_t \|R_t\|_{M_t}^2 < \infty$.
Consequently, for every $N \geq K_R + 1$,
\[\min_{0\leq i \leq N-1}\|R_i\|_{M_i}^2 = O\left(\frac{1}{{\log N}}\right).\]
<!-- end proposition-6 -->
<!-- head-1: body_md -->
Proposition~\ref{prop:automatic-marginal-feasibility} establishes vanishing marginal infeasibility through projective contraction and converts this estimate into quadratic KL cross-divergence, while Proposition~\ref{prop:cubic-main} shows that the resulting KL asymmetry is cubic. Proposition~\ref{prop:lyap-main} then incorporates these bounds into the Lyapunov analysis, which in turn yields the residual summability in Proposition~\ref{prop:residual}. Together with the guarded step-size bounds, this summability yields the vanishing residual, while Proposition~\ref{prop:automatic-marginal-feasibility}, Pinsker's inequality, and linear regularity show that the two iterates converge to their feasible projections, so all three sequences share the same accumulation points, as established in Theorem~\ref{thm:unconditional}. Moreover,  Assumption~\ref{assum:support} guarantees that every residual-vanishing subsequence limit satisfies the KKT conditions and is therefore stationary, as stated in Theorem~\ref{thm:stationarity}. Complete proofs are provided in Appendix~\ref{app:proofs}.

<!-- remark-2: body_md -->
Note that $r_{\rm stat}(\widehat W_t) = 0$ is equivalent to $\widehat W_t = \mathcal P_{\mathcal C} \big( \widehat W_t-\eta \nabla F(\widehat W_t) \big)$, i.e., $0\in\nabla F(\widehat W_t)+\mathcal N_{\mathcal C}(\widehat W_t)$, which is the KKT first-order stationarity condition for the GW problem over $\mathcal C$. A similar quantity $\big\| \pi-\operatorname{Proj}_{\mathcal C} \big(\pi+D_X\pi D_Y\big) \big\|_F$ is used by \citet[Proposition~3.1]{li2023convergent} to establish a Luo--Tseng error bound for GW.  Their quantity is the projected-gradient displacement with $\eta=1/2$, whereas ours uses a Lipschitz-based step size. 
When the projected-gradient residual is sufficiently small, the Luo--Tseng error bound \citep{li2023convergent} bounds the distance from $\widehat W_t$ to the GW stationary set by a constant multiple of $r_{\mathrm{stat}}(\widehat W_t)$.
Thus, a smaller residual indicates less first-order improvement remained and a closer proximity to the stationary set.

<!-- definition-1: body_md -->

For $K\in \mathbb{R}_{++}^{r\times s}$ and for all $u,v\in \mathbb{R}_{++}^s$,
\[
d_H(Ku,Kv) \leq \tanh\left(\frac{\Delta(K)}{4}\right)d_H(u,v).
\]

<!-- lemma-2: body_md -->
\label{lem:inv-hilb-proj}
For positive vectors $u,v\in \mathbb{R}_{++}^r$, $d_H(u^{-1},v^{-1})=d_H(u,v)$, where inversion is entrywise. 
Moreover, if $D_1$ and $D_2$ are positive diagonal matrices and $X$ is strictly positive, then $\Delta(D_1XD_2)=\Delta(X)$.
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
We know that 
    \[\max_{i} \frac{u_i^{-1}}{v_i^{-1}} = \max_i \frac{v_i}{u_i} = \frac{1}{\min_i \frac{u_i}{v_i}},\]
    and similarly \[\min_i \frac{u_i^{-1}}{v_i^{-1}} = \frac{1}{\max_i \frac{u_i}{v_i}}.\] Thus, the quotient inside the logarithm defining $d_H$ remains unchanged. 
    Denote $(D_1XD_2)_{ij} = a_iX_{ij}b_j$ with $a_i, b_j > 0$. For every $(i, i^\prime, j, j^\prime)$,
    \[
    \frac{(D_1XD_2)_{ij}(D_1XD_2)_{i^\prime j^\prime}}{(D_1XD_2)_{ij^\prime}(D_1XD_2)_{i^\prime j}} 
    = \frac{a_iX_{ij}b_j a_{i^\prime}X_{i^\prime j^\prime}b_{j^\prime}}{a_iX_{ij^\prime}b_{j^\prime }a_{i^\prime}X_{i^\prime j}b_j} 
    = \frac{X_{ij}X_{i^\prime j^\prime}}{X_{ij^\prime}X_{i^\prime j}}.
    \]
    Thus, taking the maxima of logarithms gives $\Delta (D_1 X D_2) = \Delta(X)$.

<!-- lemma-3: body_md -->
\label{lem:exact-row-ratio-dynamics}
Let $X\in \mathcal C_2$ be strictly positive. Then
$r(S(X)) = A_X\left(B_X(r(X)^{-1})\right)^{-1}$,
where inversion is entrywise.

<!-- proof-2: body_md -->
Recall that \[R(X)_{ij} = \frac{\mu_i}{\sum_{l}X_{il}}X_{ij} = X_{ij}/r_i.\] Denote $t_j = \sum_i X_{ij}/r_i$. \[\frac{t_j}{\nu_j} = \sum_{i} \frac{X_{ij}}{\nu_j} r_i^{-1} = \left(B_X(r^{-1})\right)_j.\] Then $S(X)_{ij} = \frac{\nu_j}{t_j} \frac{X_{ij}}{r_{i}}$. Since $X_{ij} = \mu_i r_i(A_X)_{ij}$,  
    \[
    \frac{(S(X)\mathbf 1_m)_i}{\mu_i} = \frac{1}{\mu_i r_i} \sum_j X_{ij}\frac{\nu_j}{t_j} = \sum_j (A_X)_{ij}\frac{\nu_j}{t_j} = \sum_j (A_X)_{ij}\left(B_X(r^{-1})\right)^{-1}_{j}.
    \]

<!-- lemma-4: body_md -->
\label{lem:sinkhorn-projective-contraction}
Let $X\in \mathcal C_2$ be strictly positive. Then
\[
d_H(r(S(X)),\mathbf 1_n)
\leq
\tanh^2\!\left(\frac{\Delta(X)}{4}\right)

<!-- proof-3: body_md -->
Since $A_X \mathbf 1_m = \mathbf 1_n$, $d_H(r(S(X)), \mathbf 1_n) = d_H\left(A_X\left(B_X(r^{-1})^{-1}\right), A_X(\mathbf 1_m)\right)$. By Lemma \ref{lem:inv-hilb-proj}, $d_H(B_X(r^{-1})^{-1}, \mathbf{1}_m) = d_H(B_X(r^{-1}), \mathbf{1}_m)$.

\]

<!-- lemma-5: body_md -->
\label{lem:projective-growth-cycle}
For any accepted A-KL-BAPG step with
step size $\alpha>0$,
    $\Delta(w^{t+1})\leq \Delta(w)+B_\Delta\alpha$,
where $B_\Delta := 8M_G$ is a conservative projective-growth constant.
<!-- end lemma-5 -->
<!-- proof-4: body_md -->
For one exponential perturbation $Z=X\odot\exp(\alpha H)$ with $\|H\|_\infty\leq M_G$, every log cross-ratio changes by $\alpha\bigl(H_{ij}+H_{i'j'}-H_{ij'}-H_{i'j}\bigr)$, whose absolute value is at most $4M_G\alpha$. Thus $\Delta(Z)\leq \Delta(X)+4M_G\alpha$. The row and column KL projections are positive diagonal scalings and therefore do not change cross-ratios.  A full A-KL-BAPG cycle contains two exponential perturbations, so
    \[\Delta(w^{t+1})\leq \Delta(w)+8M_G\alpha \leq \Delta(w)+B_\Delta\alpha.\]

<!-- lemma-7: body_md -->
\label{lem:guarded-backtracking}
Assume that the current tail iterate satisfies $\Delta(w^t)\leq\Gamma_t$.  Then the backtracking loop in Algorithm~\ref{alg:adaptive-delayed-power} terminates finitely for each iteration.  Moreover, the accepted step satisfies $\ell_t\leq \alpha_t \leq \bar\alpha_t$ and $\alpha_t \leq \alpha_{t-1}$.
Since $\beta<1$, there exist constants $c_-,c_+>0$ and $T_0$ such that, for all $t\geq T_0$,
    \[\frac{c_-}{t+2}\leq \alpha_t
    \leq c_+(t+1)^{-\beta}.\]
<!-- end lemma-7 -->
<!-- proof-6: body_md -->
Define $s_t:= \max\{\ell_t, \min\{\bar{\alpha}_t, \alpha_{t-1}\}\}$.
    By the construction of $h_t$ and $\bar \alpha_t$, the sequences $h_t$ and $\bar\alpha_t$ are nonincreasing. Then the sequence of $\ell_t$ is also nonincreasing and so is the sequence of $s_t$. Since $\alpha_{-1}=+\infty$, $\alpha_{-1}\geq\ell_0$ for $t = 0$. 
    
    For arbitrary $t \geq 1$, assume $\alpha_{t-1}\geq \ell_{t-1}\geq \ell_t$. Then $\ell_{t} \leq \alpha_{t-1}$ and $\min\{\bar \alpha_{t}, \alpha_{t-1}\} \leq \alpha_{t-1}$, i.e., $s_t \leq \alpha_{t-1}$. We also know that $\ell_t \leq s_t \leq \bar \alpha_t$.
    If $s_t\leq h_t$, we have shown that at this time the safeguard is satisfied, and we can directly accept $a_t = s_t$. Otherwise, when $s_t > h_t$, we backtrack until we find the first candidate $\alpha_t$ such that $\alpha_t \leq h_t$. Then by construction, $\alpha_t > \tau h_t$. Since $\ell_t \leq \tau h_t$, it holds that $\alpha_t \geq \ell_t$. 
    Moreover, given that the sequence of $s_t$ decreases, we obtain
    \[
    \ell_t\leq\alpha_t\leq s_t
    \leq \bar\alpha_t,
    \qquad
    \alpha_t\leq s_t\leq \alpha_{t-1}.
    \]
    
    For the final bound, we know that $\bar\alpha_t\asymp (t+1)^{-\beta}$ and
    \[
        h_t=\frac{a}{B_\Delta}\log\left(1+\frac{1}{t+1}\right).
    \]
    Since $\beta<1$, eventually $\bar\alpha_t\geq h_t$, so $\ell_t=\tau h_t$ for all sufficiently large $t$.  Since \[\log\left(1+\frac{1}{t+1}\right)\geq \frac{1}{t+2}\] and $\alpha_t\leq\bar\alpha_t$, there exists $c_{-} = \tau a/B_{\Delta} > 0$,  $c_+> 0$ and $T_0$ such that for all $t \geq T_0$, 
    \[
    \frac{c_{-}}{t+2} \leq \alpha_t \leq c_+(t+1)^{-\beta}.
    \]

<!-- lemma-8: body_md -->
\label{lem:full-cycle-comparison}
For every $t$ and every $i,j$,
\[
\exp(-2M_G\alpha_t) \leq \frac{\pi^{t+1}_{ij}}{\widehat \pi^{t+1}_{ij}} \leq \exp(2M_G\alpha_t), 
\qquad \exp(-6M_G\alpha_t) \leq
\frac{w^{t+1}_{ij}}{\widehat w^{t+1}_{ij}} \leq \exp(6M_G\alpha_t).
\]
Consequently,
$d_H\!\left(r(w^{t+1}),r(\widehat w^{t+1})\right) \leq 12M_G\alpha_t$.
<!-- end lemma-8 -->
<!-- proof-7: body_md -->
We know that 
    \[
    \frac{\pi^{t+1}_{ij}}{\widehat \pi^{t+1}_{ij}}  = \exp(\alpha_t G(w^t)_{ij})\frac{\sum_{l}w_{il}^t}{\sum_{\ell} w_{il}^t \exp(\alpha_t G(w^t)_{il})},
    \]
    with \[\exp(-M_G \alpha_t) \sum_{\ell} w_{il}^t \leq \sum_{\ell} w_{il}^t \exp(\alpha_t G(w^t)_{il}) \leq \exp(M_G \alpha_t)\sum_{\ell} w_{il}^t.\] 
    Thus \[\exp(-2M_G\alpha_t) \leq \frac{\pi^{t+1}_{ij}}{\widehat \pi^{t+1}_{ij}} \leq \exp(2M_G\alpha_t).\] 
    Recall that $z_2^t = \pi^{t+1}\odot \exp(\alpha_t G(\pi^{t+1}))$ and $\widehat w^{t+1} = \mathcal P_{\mathcal C_2}^{\mathrm{KL}}(\widehat \pi^{t+1})$, $w^{t+1} = \mathcal P_{\mathcal C_2}^{\mathrm{KL}}(z_2^t)$. Hence
    \[
    \frac{w^{t+1}_{ij}}{\widehat w^{t+1}_{ij}} = \frac{z_{2,ij}^t / \sum_\ell z_{2,\ell j}^t}{\widehat \pi^{t+1}_{ij} / \sum_\ell \widehat \pi_{\ell j}^{t+1}} = \frac{z_{2,ij}^t}{\widehat \pi^{t+1}_{ij}} \cdot \frac{\sum_\ell \widehat \pi_{\ell j}^{t+1}}{\sum_\ell z_{2,\ell j}^t},
    \quad \text{and} \quad
    \frac{z_{2,ij}^t}{\widehat \pi^{t+1}_{ij}} = \frac{\pi^{t+1}_{ij} \exp(\alpha_t G(\pi^{t+1})_{ij})}{\widehat \pi^{t+1}_{ij}} \in [e^{-3M_G\alpha_t}, e^{3M_G\alpha_t}].
    \]
    Summing over each column, we obtain
\[
\frac{\sum_\ell \pi^{t+1}_{\ell j} \exp(\alpha_t G(\pi^{t+1})_{\ell j})}{\sum_\ell \widehat \pi^{t+1}_{\ell j}} \in [e^{-3M_G\alpha_t}, e^{3M_G\alpha_t}] \quad \text{and} \quad \frac{\sum_\ell \widehat \pi^{t+1}_{\ell j}}{\sum_\ell \pi^{t+1}_{\ell j} \exp(\alpha_t G(\pi^{t+1})_{\ell j})} \in [e^{-3M_G\alpha_t}, e^{3M_G\alpha_t}].
\]
Putting these together, we obtain
\[\frac{w^{t+1}_{ij}}{\widehat w^{t+1}_{ij}} \in [e^{-3M_G\alpha_t} \cdot e^{-3M_G\alpha_t}, e^{3M_G\alpha_t} \cdot e^{3M_G\alpha_t}] = [e^{-6M_G\alpha_t}, e^{6M_G\alpha_t}].\]

<!-- lemma-9: body_md -->
\label{lem:polynomial-row-error}
Assume $\beta>p$. Then there exist constants $\widetilde C_\delta>0$ and $K_\delta\geq0$ such that, for every $t\geq K_\delta$,
    $d_H(r(w^t),\mathbf 1_n)\leq \widetilde C_\delta \eta_t$.
<!-- end lemma-9 -->
<!-- proof-8: body_md -->
By Lemma \ref{lem:full-cycle-comparison},
    $d_H(r(w^{t+1}),r(\widehat w^{t+1}))
    \leq 12M_G\alpha_t
    \leq C_\alpha(t+1)^{-\beta}$
for some constant $C_\alpha>0$ and all sufficiently large $t$.
By the accepted diameter guard,
$\Delta(w^t)\leq\Delta_0+a\log(t+1)$ and hence
\[
    1-\tanh^2\!\left(\frac{\Delta(w^t)}{4}\right)
    =\operatorname{sech}^2\!\left(\frac{\Delta(w^t)}{4}\right) \geq \exp(-\frac{\Delta(w^t)}{2})
    \geq c_\Delta(t+1)^{-p}
\]
for some $c_\Delta>0$.
Then \[\tanh^2\left(\frac{\Delta(w^t)}{4}\right) \leq 1 - c_\Delta(t+1)^{-p}.\] By Lemma \ref{lem:sinkhorn-projective-contraction}, 
\[d_H(r(\widehat w^{t+1}), \mathbf 1_n) =  d_H(r(S(w^{t})), \mathbf 1_n)\leq \tanh \left(\frac{\Delta(w^t)}{4}\right)^2 d_H(r(w^t), \mathbf 1_n) \leq \left(1 - c_\Delta(t+1)^{-p}\right) d_H(r(w^t), \mathbf 1_n) .\] 
By triangle inequality, $d_H(r(w^{t+1}), \mathbf 1_n) \leq d_H(r(w^{t+1}), r(\widehat w^{t+1})) + d_H(r(\widehat w^{t+1}), \mathbf 1_n)$. Then 
\[d_H(r(w^{t+1}), \mathbf 1_n) \leq \left(1 - c_\Delta(t+1)^{-p}\right)d_H(r(w^t), \mathbf 1_n) + C_\alpha (t+1)^{-\beta}.\] 

Let $s:=\beta-p>0$. If $d_H(r(w^t), \mathbf 1_n)\leq K(t+1)^{-s}$, then
    $\left(1-c_\Delta(t+1)^{-p}\right)d_H(r(w^t), \mathbf 1_n)
    \leq
    K(t+1)^{-s}-c_\Delta K(t+1)^{-(p+s)}$.
Since $p+s=\beta$, the recurrence gives
\[
    d_H(r(w^{t+1}), \mathbf 1_n)
    \leq K(t+1)^{-s}-(c_\Delta K-C_\alpha)(t+1)^{-\beta}.
\]
Choose $K$ sufficiently large such that $c_\Delta K>C_\alpha$. Since $p = a/2 < 1$, $\beta=p+s<s+1$ and the correction term
$(t+1)^{-\beta}$ decays more slowly than $(t+2)^{-s-1}$, which is the size of $(t+1)^{-s} - (t+2)^{-s}$.
Thus the implication
    \[d_H(r(w^{t}), \mathbf 1_n)\leq K(t+1)^{-s}
    \Rightarrow
    d_H(r(w^{t+1}), \mathbf 1_n)\leq K(t+2)^{-s},\]
holds for all sufficiently large $t$. Enlarging $K$ to absorb the finitely many earlier indices gives the claim with $\widetilde C_\delta=K$.

<!-- lemma-10: body_md -->
\label{lem:exact-projection-identities}
For every $t$, for row projection, the following identities hold: $\|\pi^{t+1}-z_1^t\|_1 = \|z_1^t\mathbf 1_m-\mu\|_1$,
\[
    D_h(\pi^{t+1},z_1^t) =

        \mu_i\log\frac{\mu_i}{r_i^t}-
        \mu_i+r_i^t \right],
\quad \text{and} \quad
    D_h(z_1^t,\pi^{t+1})=
    \sum_{i=1}^n\left[
        r_i^t\log\frac{r_i^t}{\mu_i}- r_i^t+\mu_i
    \right].
\]
Similarly, for column projection,
$\|w^{t+1}-z_2^t\|_1 = \|(z_2^t)^\top\mathbf 1_n-\nu\|_1$,
\[
    D_h(w^{t+1},z_2^t)  = \sum_{j=1}^m

        \nu_j\log\frac{\nu_j}{c_j^t} - \nu_j + c_j^t

    \sum_{j=1}^m \left[

    \right].
\]

We know that $\sum_{j=1}^m |\pi^{t+1}_{ij} - z_{1, ij}^{t}| = \left|\frac{\mu_i}{r_i^t}-1\right|\sum_{j = 1}^m z_{1, ij}^{t} = |\mu_i - r_i^t|$. Summing over $i$, we obtain $\|\pi^{t+1}-z_1^t\|_1 = \|z_1^t\mathbf 1_m-\mu\|_1$.

    \end{multline*}
The derivation of formula for $D_h(z_1^t , \pi^{t+1})$ is identical. The proof for column projection identities is analogous to the above proof for row projection identities.

\begin{proof}
 For each $t$, define the row ratio vector \[\theta_i^t := \frac{(w^t \mathbf 1_m)_i}{\mu_i}.\]

    Define
   $K_{\mathrm{row}}:=
    \min\left\{ t\geq K_\delta:\widetilde C_\delta\eta_t\leq 1
    \right\}$.

    $\|w^t \mathbf 1_m - \mu\|_1 \leq e \widetilde C_\delta \eta_t = C_{\operatorname{row}}\eta_t$.

        \leq \|\pi^{t+1} - z_1^t\|_1 + \|z_1^t - w^t\|_1 
        \leq \|z_1^t\mathbf 1_m - \mu\|_1 + \|z_1^t - w^t\|_1
