<!-- assumption-1: body_md -->
\label{ass:smooth}
The differentiable objective $F:\mathbb R^d\to\mathbb R$ has finite infimum
$F^*=\inf_xF(x)$ and satisfies
\begin{equation}
 \left\|\nabla F(x)-\nabla F(y)\right\|\le L\left\|x-y\right\|
 \quad\text{for all }x,y\in\mathbb R^d.
 \label{eq:population-smooth}
\end{equation}
<!-- end assumption-1 -->
<!-- assumption-2: body_md -->
\label{ass:oracle}
The stochastic oracle $g$ is unbiased and has
bounded variance:
\begin{equation}
 \mathbb E_\xi g(x,\xi)=\nabla F(x),\qquad
 \mathbb E_\xi\left\|g(x,\xi)-\nabla F(x)\right\|^2\le\sigma^2
 \quad\text{for all }x\in\mathbb R^d.
 \label{eq:bv}
\end{equation}
<!-- end assumption-2 -->
<!-- assumption-3: body_md -->
\label{ass:average}
Stochastic gradient differences satisfy
\begin{equation}
 \mathbb E_\xi\left\|g(x,\xi)-g(y,\xi)\right\|^2
       \le\mathcal L^2\left\|x-y\right\|^2\quad\text{for all }x,y\in\mathbb R^d.
 \label{eq:mss}
\end{equation}
Note that this assumption is also called mean-square smoothness. 
By Jensen's inequality, Assumptions~\ref{ass:oracle}
and~\ref{ass:average} imply that $F$ is also $\mathcal L$-smooth, so we assume $L\le\mathcal L$ without loss of generality.
<!-- end assumption-3 -->
<!-- assumption-4: body_md -->
\label{ass:pl}
For some $0<\mu\le L$, the objective satisfies
\begin{equation}
 \tfrac12\left\|\nabla F(x)\right\|^2\ge\mu(F(x)-F^*)
 \quad\text{for all }x\in\mathbb R^d.
 \label{eq:pl}
\end{equation}
<!-- end assumption-4 -->
<!-- assumption-5: body_md -->
\label{ass:convex}
The objective $F$ has a minimizer $x^*$ and satisfies
\begin{equation}
 F(y)\ge F(x)+\left\langle\nabla F(x),y-x\right\rangle
 \quad\text{for all }x,y\in\mathbb R^d.
 \label{eq:convexity}
\end{equation}
<!-- end assumption-5 -->
<!-- assumption-6: body_md -->
\label{ass:strong}
The objective $F$ has a minimizer $x^*$ and  satisfies
\begin{equation}
 F(y)\ge F(x)+\left\langle\nabla F(x),y-x\right\rangle
                         +\tfrac\lambda2\left\|y-x\right\|^2
 \quad\text{for all }x,y\in\mathbb R^d.
 \label{eq:strong-convexity}
\end{equation}
Strong convexity also implies the PL condition with $\mu=\lambda$, and the converse does not hold.
<!-- end assumption-6 -->
<!-- theorem-1: body_md -->
\label{thm:nonconvex}
Under Assumptions~\ref{ass:smooth}, \ref{ass:oracle}, and \ref{ass:average}, 
setting
$\eta_t=\eta=\frac1{2\mathcal L(T+2)^{1/3}}$, $
 a_t=(t+2)^{-2/3}$, we have the following guarantee
\begin{align}
 \frac1T\sum_{t=0}^{T-1}\mathbb E\left\|\nabla F(x_t)\right\|^2
 \le{}&\frac{4\mathcal L\Delta_0(T+2)^{1/3}}T
          +\frac{2^{2/3}\sigma^2}T+\frac{6\sigma^2}{T}
                 \bigl((T+2)^{1/3}-2^{1/3}\bigr).
\label{eq:ncbound}\end{align}
For $I$ uniform on $\{0,\ldots,T-1\}$ and independent of the trajectory,
\[
 \mathbb E\left\|\nabla F(x_I)\right\|
 =O(\sqrt{\mathcal L\Delta_0+\sigma^2}\,T^{-1/3}).
\]
When
$\sigma=0$, the alternative choice $\eta=1/(4\mathcal L)$ and $a_t=1/4$
gives 
\[
 \mathbb E\left\|\nabla F(x_I)\right\|
 =O(\sqrt{\mathcal L\Delta_0}/\sqrt T).
\]
<!-- end theorem-1 -->
<!-- head-1: body_md -->
Theorem~\ref{thm:nonconvex} gives an $O(T^{-1/3})$ rate and
the corresponding gradient complexity is $O(\varepsilon^{-3})$, matching the lower bound for nonconvex objectives with average smoothness 
\citep{arjevani2023lower}.
The theorem removes the logarithmic factor in the original STORM

<!-- lemma-1: body_md -->
\label{lem:geometry-main}
Under Assumptions~\ref{ass:smooth} and~\ref{ass:convex},
for every $0\le\eta\le1/L$ and all $x,y\in\mathbb R^d$,
\begin{equation}
 \left\|x-\eta\nabla F(x)-y+\eta\nabla F(y)\right\|
 \le\left\|x-y\right\|.
 \label{eq:nonexpansive-main}
\end{equation}
If strong convexity~(Assumption~\ref{ass:strong}) also holds, then
\begin{equation}
 \left\|x-\eta\nabla F(x)-y+\eta\nabla F(y)\right\|
 \le(1-\lambda\eta)\left\|x-y\right\|.
 \label{eq:contraction-main0}
\end{equation}
Under Assumption~\ref{ass:smooth} alone, even without convexity, we have
\begin{align}
 \left\|\nabla F(y)\right\|^2
 &\le2L(F(y)-F^*),
 \label{eq:self-bound-main}\\
 F(x)-F^*
 &\le2(F(y)-F^*)+L\left\|x-y\right\|^2.
 \label{eq:functioncompare}
\end{align}
<!-- end lemma-1 -->
<!-- head-4: body_md -->
Inequality~\eqref{eq:nonexpansive-main} shows that a gradient
step would not increase the distance between two points.
Under strong convexity, \eqref{eq:contraction-main0} can further give a contraction.
Both statements concern the exact gradient $\nabla F$ and therefore require no regularity of individual stochastic gradients, avoiding the requirement of average smoothness.
Once the distance between two trajectories is controlled, the inequality
\eqref{eq:functioncompare} transfers a function-gap guarantee
from one trajectory to the other.

<!-- lemma-3: body_md -->
\label{lem:tracking}
Under Assumptions~\ref{ass:smooth} and~\ref{ass:oracle},
for every deterministic $\eta_t\ge0$, the primal update satisfies
\begin{align}
 \Delta_{t+1}-\Delta_t
 \le{}&-\frac{\eta_t}{2}\mathbb E\left\|\nabla F(x_t)\right\|^2
       -\frac{\eta_t(1-L\eta_t)}{2}\mathbb E\left\|d_t\right\|^2
       +\frac{\eta_t}{2}q_t.
 \label{eq:descent}
\end{align}
If Assumption~\ref{ass:average} also holds and
$0\le a_{t+1}\le1$, then
\begin{equation}
 q_{t+1}\le(1-a_{t+1})^2q_t+2a_{t+1}^2\sigma^2
                       +2\mathcal L^2\eta_t^2\mathbb E\left\|d_t\right\|^2.
 \label{eq:tracking}
\end{equation}
<!-- end lemma-3 -->
<!-- proof-1: body_md -->
Smoothness and $x_{t+1}=x_t-\eta_td_t$ give
\[
 F(x_{t+1})\le F(x_t)
 -\eta_t\left\langle\nabla F(x_t),d_t\right\rangle
 +\frac{L\eta_t^2}{2}\left\|d_t\right\|^2.
\]
By the definition of $e_t$,
\[
 \left\|e_t\right\|^2
 =\left\|d_t\right\|^2+\left\|\nabla F(x_t)\right\|^2
                      -2\left\langle\nabla F(x_t),d_t\right\rangle.
\]
Solving this identity for the inner product and substituting yields
\begin{align*}
 F(x_{t+1})-F(x_t)
 &\le-\frac{\eta_t}{2}\left\|\nabla F(x_t)\right\|^2
      -\frac{\eta_t}{2}\left\|d_t\right\|^2
      +\frac{\eta_t}{2}\left\|e_t\right\|^2
      +\frac{L\eta_t^2}{2}\left\|d_t\right\|^2\\
 &=-\frac{\eta_t}{2}\left\|\nabla F(x_t)\right\|^2
      -\frac{\eta_t(1-L\eta_t)}{2}\left\|d_t\right\|^2
      +\frac{\eta_t}{2}\left\|e_t\right\|^2.
\end{align*}
Taking expectations proves \eqref{eq:descent}.

To prove the error recursion, abbreviate $a=a_{t+1}$ and write
$\zeta(x,\xi)=g(x,\xi)-\nabla F(x)$.
Subtracting $\nabla F(x_{t+1})$ from the estimator update gives the
exact identity
\begin{align*}
 e_{t+1}
 &=(1-a)e_t+\zeta(x_{t+1},\xi_{t+1})
                         -(1-a)\zeta(x_t,\xi_{t+1})\\
 &=(1-a)e_t+a\zeta(x_{t+1},\xi_{t+1})+(1-a)
       \bigl(\zeta(x_{t+1},\xi_{t+1})-\zeta(x_t,\xi_{t+1})\bigr).
\end{align*}
Conditional on $\mathcal F_{t+1}$, the fresh-sample terms together have mean
zero, so their cross term with the fixed $e_t$ vanishes. We then use
$\left\|u+v\right\|^2\le2\left\|u\right\|^2+2\left\|v\right\|^2$ to obtain
\begin{align*}
 \mathbb E[\left\|e_{t+1}\right\|^2\mid\mathcal F_{t+1}]
 &\le(1-a)^2\left\|e_t\right\|^2
       +2a^2\mathbb E[\left\|\zeta(x_{t+1},\xi_{t+1})\right\|^2\mid\mathcal F_{t+1}]\\
 &\quad+2(1-a)^2\mathbb E\bigl[
       \left\|\zeta(x_{t+1},\xi_{t+1})-\zeta(x_t,\xi_{t+1})\right\|^2
                           \mid\mathcal F_{t+1}\bigr].
\end{align*}
The first conditional noise moment is at most $\sigma^2$ by
\eqref{eq:bv} and fresh-sample independence. For the second one,
centering a random vector subtracts the squared norm of its mean:
\begin{align*}
 &\mathbb E\bigl[
    \left\|\zeta(x_{t+1},\xi_{t+1})-\zeta(x_t,\xi_{t+1})\right\|^2
                       \mid\mathcal F_{t+1}\bigr]\\
 &\quad=\mathbb E\bigl[
    \left\|g(x_{t+1},\xi_{t+1})-g(x_t,\xi_{t+1})\right\|^2
                       \mid\mathcal F_{t+1}\bigr]
          -\left\|\nabla F(x_{t+1})-\nabla F(x_t)\right\|^2\\
 &\quad\le\mathcal L^2\left\|x_{t+1}-x_t\right\|^2
       =\mathcal L^2\eta_t^2\left\|d_t\right\|^2.
\end{align*}
The inequality uses \eqref{eq:mss}. Substitution gives
\[
 \mathbb E[\left\|e_{t+1}\right\|^2\mid\mathcal F_{t+1}]
 \le(1-a)^2\left\|e_t\right\|^2+2a^2\sigma^2
                  +2(1-a)^2\mathcal L^2\eta_t^2\left\|d_t\right\|^2.
\]
Finally, $(1-a)^2\le1$ gives
\eqref{eq:tracking}.

\label{lem:certificate}
Under Assumptions~\ref{ass:smooth}, \ref{ass:oracle}, and
\ref{ass:average}, let
$\eta_t=\eta>0$ and $L\eta\le1/2$.
Choose an auxiliary weight $a_0>0$ and deterministic coefficients satisfying
\[
 0<a_{t+1}\le\frac12,\qquad
 a_{t+1}\ge4\mathcal L^2\eta^2,\qquad
 a_{t+1}^{-1}-a_t^{-1}\le\frac12.
\]
With
\[
 \Psi_t=\Delta_t+\frac{\eta}{2a_t}q_t,
\]
one has
\begin{equation}
 \Psi_{t+1}\le\Psi_t
       -\frac\eta2\mathbb E\left\|\nabla F(x_t)\right\|^2
       +\eta a_{t+1}\sigma^2.
 \label{eq:certificate}
\end{equation}

<!-- proof-2: body_md -->
Multiply \eqref{eq:tracking} by $\eta/(2a_{t+1})$ and add it to
\eqref{eq:descent}. This gives
\begin{align*}
 \Psi_{t+1}
 &\le\Delta_t-\frac\eta2\mathbb E\left\|\nabla F(x_t)\right\|^2+\left(\frac\eta2+
             \frac{\eta(1-a_{t+1})^2}{2a_{t+1}}\right)q_t\\
 &\quad+\left(-\frac{\eta(1-L\eta)}2
                   +\frac{\mathcal L^2\eta^3}{a_{t+1}}\right)
                                      \mathbb E\left\|d_t\right\|^2
        +\eta a_{t+1}\sigma^2.
\end{align*}
Subtract $\Psi_t=\Delta_t+\eta q_t/(2a_t)$.
The remaining coefficient of $q_t$ is
\begin{align*}
 \frac\eta2+\frac{\eta(1-a_{t+1})^2}{2a_{t+1}}
                     -\frac{\eta}{2a_t}
 &=\frac\eta2\left(
      1+a_{t+1}^{-1}-2+a_{t+1}-a_t^{-1}\right)\\
 &=\frac\eta2\left(
      a_{t+1}^{-1}-a_t^{-1}-1+a_{t+1}\right)\\
 &\le\frac\eta2\left(\frac12-1+\frac12\right)=0.

 -\frac{\eta(1-L\eta)}2+\frac{\mathcal L^2\eta^3}{a_{t+1}}

Fix an arbitrary integer $T\ge1$ and use:
\[
 \eta=\frac1{2\mathcal L(T+2)^{1/3}},\qquad
 a_t=(t+2)^{-2/3}.
\]
We first verify the conditions of Lemma~\ref{lem:certificate} for
$t=0,\ldots,T-1$. Since $L\le\mathcal L$,
\[
 L\eta\le\frac1{2(T+2)^{1/3}}\le\frac12.
\]
The mixing sequence decreases, and hence
\[
 a_{t+1}\le a_1=3^{-2/3}<\frac12,
 \qquad
 a_{t+1}\ge a_T=(T+2)^{-2/3}=4\mathcal L^2\eta^2.
\]
Because $u\mapsto u^{2/3}$ is concave, its unit increments
decrease as $u$ increases. Therefore
\[
 a_{t+1}^{-1}-a_t^{-1}
 =(t+3)^{2/3}-(t+2)^{2/3}
 \le3^{2/3}-2^{2/3}<\frac12.

Thus, all conditions of Lemma~\ref{lem:certificate} hold.
Rearrange \eqref{eq:certificate} and sum over the $T$ steps:
\begin{align*}
 \frac\eta2\sum_{t=0}^{T-1}\mathbb E\left\|\nabla F(x_t)\right\|^2
 &\le\sum_{t=0}^{T-1}(\Psi_t-\Psi_{t+1})
                        +\eta\sigma^2\sum_{t=0}^{T-1}a_{t+1}\\
 &=\Psi_0-\Psi_T+\eta\sigma^2\sum_{t=0}^{T-1}a_{t+1}\\
 &\le\Delta_0+\frac{\eta\sigma^2}{2a_0}
                        +\eta\sigma^2\sum_{t=0}^{T-1}a_{t+1}.
\end{align*}
Here $\Psi_T\ge0$ because $F(x_T)\ge F^*$ and $q_T\ge0$,
and the initial variance bound gives $q_0\le\sigma^2$.

\[
 \frac1T\sum_{t=0}^{T-1}\mathbb E\left\|\nabla F(x_t)\right\|^2
 \le\frac{2\Delta_0}{\eta T}
        +\frac{\sigma^2}{a_0T}
        +\frac{2\sigma^2}{T}\sum_{t=0}^{T-1}(t+3)^{-2/3}.
\]

Since $u^{-2/3}$ is decreasing, integrating over each interval

\begin{align*}
 \sum_{t=0}^{T-1}(t+3)^{-2/3}
 &\le\sum_{t=0}^{T-1}\int_{t+2}^{t+3}u^{-2/3}\,du\\
 &=\int_2^{T+2}u^{-2/3}\,du\\
 &=3\bigl((T+2)^{1/3}-2^{1/3}\bigr).
\end{align*}

Cauchy--Schwarz gives
$\mathbb E\left\|\nabla F(x_I)\right\|\le

Substituting $\eta_t=2/[\mu(t+s)]$ into \eqref{eq:plscalar-app} gives
\begin{equation}
 V_{t+1}\le\frac{t+s-2}{t+s}V_t
                   +\frac{8\sigma^2}{\mu(t+s)^2}.
 \label{eq:pl-reciprocal-recursion}
\end{equation}
The first two substitutions are
\begin{align*}
 V_1&\le\frac{s-2}{s}V_0+\frac{8\sigma^2}{\mu s^2},\\
 V_2&\le\frac{s-1}{s+1}\frac{s-2}{s}V_0
       +\frac{8\sigma^2}{\mu}
           \left(\frac{s-1}{(s+1)s^2}+\frac1{(s+1)^2}\right).
\end{align*}
Induction therefore gives
\begin{equation}\label{eq:pl-unrolled}
 V_T\le V_0\prod_{k=0}^{T-1}\frac{k+s-2}{k+s}
       +\frac{8\sigma^2}{\mu}\sum_{j=0}^{T-1}\frac1{(j+s)^2}
                 \prod_{k=j+1}^{T-1}\frac{k+s-2}{k+s}.
\end{equation}
For every $0\le u\le T$,
\begin{equation}\label{eq:pl-product}
 \prod_{k=u}^{T-1}\frac{k+s-2}{k+s}
 =\frac{(u+s-2)(u+s-1)}{(T+s-2)(T+s-1)}.
\end{equation}
Taking $u=0$ gives the initialization product.
Taking $u=j+1$ in each noise term gives
\begin{align*}
 \frac1{(j+s)^2}
                 \prod_{k=j+1}^{T-1}\frac{k+s-2}{k+s}
 &=\frac1{(j+s)^2}
      \frac{(j+s-1)(j+s)}{(T+s-2)(T+s-1)}\\
 &=\frac{(j+s-1)/(j+s)}{(T+s-2)(T+s-1)}.
\end{align*}
Consequently, \eqref{eq:pl-unrolled} becomes the explicit bound
\begin{equation}
 V_T\le
 \frac{(s-2)(s-1)V_0+
       \dfrac{8\sigma^2}{\mu}
                    \sum_{j=0}^{T-1}\dfrac{j+s-1}{j+s}}
      {(T+s-2)(T+s-1)}.
 \label{eq:pl-exact-scalar-bound}
\end{equation}

We now bound the initialization and noise terms separately.

For $r=1,2$, positivity of the denominators and $rT\ge0$ imply
\[
 \frac{s-r}{T+s-r}\le\frac{s}{T+s}.
\]
Indeed, after cross multiplication this is precisely
$(s-r)(T+s)\le s(T+s-r)$.
Multiplying the two inequalities gives
\[
 \frac{(s-2)(s-1)}{(T+s-2)(T+s-1)}
 \le\left(\frac{s}{T+s}\right)^2.
\]
Every summand $(j+s-1)/(j+s)$ in the noise numerator is at most
one. Furthermore,
\begin{align*}
 (T+s-2)(T+s-1)-T(T+s)=(s-3)(T+s)+2\ge0,
\end{align*}
because $s\ge16$. Hence
\[
 \frac{\sum_{j=0}^{T-1}(j+s-1)/(j+s)}{(T+s-2)(T+s-1)}
 \le\frac{T}{(T+s-2)(T+s-1)}\le\frac1{T+s}.
\]
Combining the two bounds yields
\begin{equation}
 V_T\le\left(\frac{s}{T+s}\right)^2V_0
                         +\frac{8\sigma^2}{\mu(T+s)}.
 \label{eq:pl-potential-bound}
\end{equation}
Using $\Delta_T\le V_T$ and
$V_0\le\Delta_0+\sigma^2/\mu$ proves \eqref{eq:plbound}.

\emph{The two-phase schedule.}

Now fix any integer $T\ge1$, put $m=\lfloor T/2\rfloor$, and
use \eqref{eq:pl-two-phase}. During the first $m$ updates,
\eqref{eq:plscalar-app} gives
\[
 V_{t+1}\le\left(1-\frac2s\right)V_t
                                    +\frac{8\sigma^2}{\mu s^2}.
\]
Repeated substitution gives
\begin{align*}
 V_m
 &\le\left(1-\frac2s\right)^mV_0
       +\frac{8\sigma^2}{\mu s^2}
                      \sum_{j=0}^{m-1}\left(1-\frac2s\right)^j\\
 &=\left(1-\frac2s\right)^mV_0
       +\frac{4\sigma^2}{\mu s}
                         \left[1-\left(1-\frac2s\right)^m\right]\\
 &\le e^{-2m/s}V_0+\frac{4\sigma^2}{\mu s}.
\end{align*}
The final line uses
$1-u\le e^{-u}$ and drops a factor between zero and one
from the noise term.

There remain $n=T-m$ updates. Their parameters are
$\eta_{m+j}=2/[\mu(s+j)]$ for $j=0,\ldots,n-1$.
Applying the bound \eqref{eq:pl-potential-bound}
with initial value $V_m$ gives
\[
 V_T\le\left(\frac{s}{s+n}\right)^2V_m
                                      +\frac{8\sigma^2}{\mu(s+n)}.
\]
The potential $V_m$ already includes the estimation error $q_m$.
Hence this argument applies directly to the remaining updates,
without restarting $x_m$ or $d_m$.

 &\le\left(\frac{s}{s+n}\right)^2 e^{-2m/s}V_0
       +\frac{4\sigma^2s}{\mu(s+n)^2}
       +\frac{8\sigma^2}{\mu(s+n)}\\

 &\le e^{-2m/s}V_0+\frac{24\sigma^2}{\mu(T+2s)}.
\end{align*}
The second line uses $s/(s+n)\le1$ and the third uses
$n=\lceil T/2\rceil\ge T/2$.
Finally, $\Delta_T\le V_T$ and the initial bound on $V_0$
give \eqref{eq:plphase}.

\emph{The estimation error has bounded second moment.}
Let $\mathcal F_t=\sigma(\xi_0,\ldots,\xi_{t-1})$ denote the information
available before sample $\xi_t$ is drawn.
Write $\mathbb E_t[\cdot]=\mathbb E[\cdot\mid\mathcal F_t]$ and define
\[
 e_t=d_t-\nabla F(x_t),\qquad e_{-1}=0.
\]
For $t\ge1$, subtract $\nabla F(x_t)$ from the estimator update:
\begin{align*}
 e_t
 &=\beta e_{t-1}+\nu_t,\\
 \nu_t
 &=g(x_t,\xi_t)-\nabla F(x_t)
   -\beta\bigl(g(x_{t-1},\xi_t)-\nabla F(x_{t-1})\bigr).
\end{align*}
Set $\nu_0=e_0$. The error recursion therefore holds also at $t=0$:
\begin{equation}
 e_t=\beta e_{t-1}+\nu_t,\qquad t\ge0.
 \label{eq:population-nc-error}
\end{equation}
Both query points in $\nu_t$ are $\mathcal F_t$-measurable. Since $\xi_t$
is independent of $\mathcal F_t$, oracle unbiasedness gives
\begin{equation}
 \mathbb E_t\nu_t=0.
 \label{eq:population-nc-centered}
\end{equation}
The two oracle errors in $\nu_t$ use the same sample and need not
be independent. We bound their sum by the conditional $L^2$ triangle
inequality and then use \eqref{eq:bv}:
\begin{align*}
 \bigl(\mathbb E_t\left\|\nu_t\right\|^2\bigr)^{1/2}
 &\le
 \bigl(\mathbb E_t\left\|g(x_t,\xi_t)-\nabla F(x_t)\right\|^2\bigr)^{1/2}+\beta
 \bigl(\mathbb E_t\left\|g(x_{t-1},\xi_t)-\nabla F(x_{t-1})\right\|^2\bigr)^{1/2}\\
 &\le(1+\beta)\sigma.
\end{align*}
For $t=0$, initialization gives the stronger bound
$\mathbb E\left\|\nu_0\right\|^2\le\sigma^2$. Thus, for every $t\ge0$,
\begin{equation}
 \mathbb E_t\left\|\nu_t\right\|^2\le(1+\beta)^2\sigma^2.
 \label{eq:population-nc-noise}
\end{equation}
Squaring \eqref{eq:population-nc-error} and conditioning on $\mathcal F_t$
eliminates its cross term, because $e_{t-1}$ is $\mathcal F_t$-measurable
and \eqref{eq:population-nc-centered} holds. Consequently,
\[
 \mathbb E\left\|e_t\right\|^2
 \le\beta^2\mathbb E\left\|e_{t-1}\right\|^2+(1+\beta)^2\sigma^2.
\]
Starting from $e_{-1}=0$, repeated substitution gives
\begin{equation}
 \mathbb E\left\|e_t\right\|^2
 \le(1+\beta)^2\sigma^2\sum_{j=0}^t\beta^{2j}
 \le\frac{(1+\beta)^2}{1-\beta^2}\sigma^2
 =\frac{1+\beta}{1-\beta}\sigma^2.
 \label{eq:population-nc-error-bound}
\end{equation}

\emph{An auxiliary iterate removes the inherited error.}
Define, only for the analysis,
\begin{equation}
 z_t=x_t-\frac{\eta\beta}{1-\beta}e_{t-1}.
 \label{eq:population-nc-virtual}
\end{equation}
Then $z_0=x_0$. The factor $\beta/(1-\beta)$ is chosen so that
the error cancels from the following update:
\begin{align*}
 z_{t+1}
 &=x_{t+1}-\frac{\eta\beta}{1-\beta}e_t\\
 &=x_t-\eta\nabla F(x_t)-\eta e_t
                         -\frac{\eta\beta}{1-\beta}e_t\\
 &=x_t-\eta\nabla F(x_t)-\frac{\eta}{1-\beta}e_t\\
 &=x_t-\frac{\eta\beta}{1-\beta}e_{t-1}
          -\eta\nabla F(x_t)-\frac{\eta}{1-\beta}\nu_t.
\end{align*}
The last equality uses \eqref{eq:population-nc-error}. Hence
\begin{equation}
 z_{t+1}=z_t-\eta\nabla F(x_t)-\frac{\eta}{1-\beta}\nu_t.
 \label{eq:population-nc-virtual-update}
\end{equation}
The update has centered noise, but evaluates the gradient at $x_t$
rather than $z_t$. Standard smoothness will control this difference.
Indeed, \eqref{eq:population-nc-virtual} gives the exact identity
\[
 \left\|x_t-z_t\right\|^2
 =\frac{\eta^2\beta^2}{(1-\beta)^2}\left\|e_{t-1}\right\|^2.
\]
No auxiliary sequence is computed by STORM.

\emph{Descent along the auxiliary sequence.}
Apply standard smoothness to
\eqref{eq:population-nc-virtual-update}:
\begin{align*}
 F(z_{t+1})
 \le{}F(z_t)-\eta\left\langle\nabla F(z_t),\nabla F(x_t)+\nu_t/(1-\beta)\right\rangle
 +\frac{L\eta^2}{2}
                 \left\|\nabla F(x_t)+\nu_t/(1-\beta)\right\|^2.
\end{align*}
Conditional on $\mathcal F_t$,
$\mathbb E_t\nu_t=0$. Expanding the squared norm before taking conditional
expectations therefore yields
\begin{align*}
 \mathbb E_t F(z_{t+1})
 \le{}F(z_t)-\eta\left\langle\nabla F(z_t),\nabla F(x_t)\right\rangle
                +\frac{L\eta^2}{2}\left\|\nabla F(x_t)\right\|^2+\frac{L\eta^2}{2(1-\beta)^2}\mathbb E_t\left\|\nu_t\right\|^2.
\end{align*}
To express the decrease in terms of the gradient at the actual iterate,
use $\left\langleu,v\right\rangle\le\left\|u\right\|^2+\left\|v\right\|^2/4$:
\begin{align*}
 -\left\langle\nabla F(z_t),\nabla F(x_t)\right\rangle
 &=-\left\|\nabla F(x_t)\right\|^2
   +\left\langle\nabla F(x_t)-\nabla F(z_t),\nabla F(x_t)\right\rangle\\
 &\le-\frac34\left\|\nabla F(x_t)\right\|^2
       +\left\|\nabla F(x_t)-\nabla F(z_t)\right\|^2\\
 &\le-\frac34\left\|\nabla F(x_t)\right\|^2
       +\frac{L^2\eta^2\beta^2}{(1-\beta)^2}\left\|e_{t-1}\right\|^2.
\end{align*}
Since
$L\eta\le1/2$, the coefficient of the squared gradient is at most
$-\eta(3/4-L\eta/2)\le-\eta/2$. We obtain
\begin{align*}
 \mathbb E_t F(z_{t+1})
 \le{}F(z_t)-\frac\eta2\left\|\nabla F(x_t)\right\|^2
       +\frac{L^2\eta^3\beta^2}{(1-\beta)^2}\left\|e_{t-1}\right\|^2+\frac{L\eta^2(1+\beta)^2}{2(1-\beta)^2}\sigma^2.
\end{align*}
Taking full expectations and using
\eqref{eq:population-nc-error-bound}, which also bounds
$\mathbb E\left\|e_{-1}\right\|^2=0$, gives
\begin{equation}\label{eq:population-nc-descent}
 \mathbb E F(z_{t+1})
 \le{}\mathbb E F(z_t)-\frac\eta2\mathbb E\left\|\nabla F(x_t)\right\|^2+\frac{L^2\eta^3\beta^2(1+\beta)}{(1-\beta)^3}\sigma^2
       +\frac{L\eta^2(1+\beta)^2}{2(1-\beta)^2}\sigma^2.
\end{equation}

\emph{Summation and the choice of parameters.}

\begin{align*}
 \frac\eta2\sum_{t=0}^{T-1}\mathbb E\left\|\nabla F(x_t)\right\|^2
 \le{}&F(z_0)-\mathbb E F(z_T)
       +\frac{L^2\eta^3T\beta^2(1+\beta)}{(1-\beta)^3}\sigma^2+\frac{L\eta^2T(1+\beta)^2}{2(1-\beta)^2}\sigma^2.
\end{align*}

 \frac1T\sum_{t=0}^{T-1}\mathbb E\left\|\nabla F(x_t)\right\|^2
 \le{}\frac{2\Delta_0}{\eta T}

\end{equation}
The condition $2L\eta\le1$ bounds the last term by
$\beta^2(1+\beta)L\eta\sigma^2/(1-\beta)^3$. Combining it with

\[
 \frac{(1+\beta)^2}{(1-\beta)^2}
 +\frac{\beta^2(1+\beta)}{(1-\beta)^3}
 =\frac{(1+\beta)((1+\beta)(1-\beta)+\beta^2)}{(1-\beta)^3}
 =\frac{1+\beta}{(1-\beta)^3}.
\]
This proves \eqref{eq:population-nc-general}.

 \ge\frac{\left\|\nabla F(x)-\nabla F(y)\right\|^2}{2L}.

\]
Taking square roots proves \eqref{eq:nonexpansive-main}.

\emph{Contraction under strong convexity.}
Assume first that $L>\lambda$ and set
$H(z)=F(z)-\lambda\left\|z\right\|^2/2$. Subtracting the quadratic term
from the strong-convexity lower model and the smoothness upper model gives
\[
 0\le H(z)-H(w)-\left\langle\nabla H(w),z-w\right\rangle
       \le\frac{L-\lambda}{2}\left\|z-w\right\|^2.
\]
Thus $H$ is convex and has an upper quadratic model with coefficient
$L-\lambda$. The preceding proof of nonexpansiveness uses only these
two properties, so it applies to $H$. Since $\eta\le1/L$ and
$L>\lambda$, we have $1-\lambda\eta>0$ and
\[
 \frac{\eta}{1-\lambda\eta}\le\frac1{L-\lambda};
\]
after multiplying by the positive denominators, this is just
$L\eta\le1$. Using $\nabla F(z)=\nabla H(z)+\lambda z$,
\begin{align*}
 &\left\|x-\eta\nabla F(x)-y+\eta\nabla F(y)\right\|\\
 &\quad=(1-\lambda\eta)
 \left\|x-\frac{\eta}{1-\lambda\eta}\nabla H(x)
          -y+\frac{\eta}{1-\lambda\eta}\nabla H(y)\right\|\\
 &\quad\le(1-\lambda\eta)\left\|x-y\right\|.
\end{align*}
If $L=\lambda$, the lower and upper models for $H$ coincide, so $H$
is affine. Hence $\nabla F(x)-\nabla F(y)=\lambda(x-y)$ and

\emph{Comparison of function gaps.}

Use $D_t=\mathbb E\left\|y_t-x^*\right\|^2$ and
$h_t=\nabla F(y_t)+\nu_t$. Note that \eqref{eq:convex-moment}--\eqref{eq:convex-distance}
apply. Put $M=\lceil (T+1)/4\rceil$ and $K=\lceil (T+1)/2\rceil$.
The first schedule block is $0,\ldots,K-1$, with
$\eta_t=\eta$.

\emph{A reference value in the first half.}
Discard the nonpositive term $-\eta\delta_t$ in
\eqref{eq:convex-distance} for $0\le t<M-1$ and sum. This yields
\[
 D_{M-1}\le D_0+(M-1)\eta^2S^2
                   =R^2+(M-1)\eta^2S^2.
\]
Now keep that decrease and sum from $t=M-1$ to $K-1$:
\begin{align*}
 \eta\sum_{t=M-1}^{K-1}\delta_t
 &\le D_{M-1}-D_K+(K-M+1)\eta^2S^2\\
 &\le R^2+\bigl((M-1)+(K-M+1)\bigr)\eta^2S^2\\
 &=R^2+K\eta^2S^2.
\end{align*}
Let $A_0$ be the minimum of $\delta_t$ over these $K-M+1$
indices. A minimum is at most the mean, so
\begin{equation}
 A_0\le\frac{R^2}{(K-M+1)\eta}
                   +\frac{K}{K-M+1}\eta S^2
       \le\frac{4R^2}{\eta (T+1)}+2\eta S^2.
 \label{eq:convex-first-reference}
\end{equation}
The last line uses $K-M+1\ge (T+1)/4$ and
$K/(K-M+1)\le2$ from Lemma~\ref{lem:rounded-geometry}.
We will also need the mean over the entire first block.
Summing directly from $0$ to $K-1$ gives
\begin{equation}
 \frac1K\sum_{t=0}^{K-1}\delta_t
 \le\frac{R^2}{K\eta}+\eta S^2
 \le\frac{2R^2}{\eta (T+1)}+\eta S^2,
 \label{eq:convex-first-mean}
\end{equation}
where $K\ge (T+1)/2$.

\emph{Comparison with an earlier point.}
Fix deterministic indices $p\le q$. For every $p\le t\le q$,
expand the squared distance from $y_p$:
\[
 \left\|y_{t+1}-y_p\right\|^2
 =\left\|y_t-y_p\right\|^2-2\eta_t\left\langleh_t,y_t-y_p\right\rangle
                                      +\eta_t^2\left\|h_t\right\|^2.
\]
Although $y_p$ is random, it is $\mathcal F_t$-measurable when
$t\ge p$. Conditional centering therefore gives
\begin{align*}
 \mathbb E\left\langleh_t,y_t-y_p\right\rangle
 &=\mathbb E\left\langle\nabla F(y_t),y_t-y_p\right\rangle\\
 &\ge\mathbb E[F(y_t)-F(y_p)]=\delta_t-\delta_p.
\end{align*}
The inequality follows by rearranging
$F(y_p)\ge F(y_t)+\left\langle\nabla F(y_t),y_p-y_t\right\rangle$.
After taking expectations and summing the distance identity,
the initial distance is zero and the terminal distance is nonnegative:
\begin{align*}
 2\sum_{t=p}^q\eta_t(\delta_t-\delta_p)
 &\le-\mathbb E\left\|y_{q+1}-y_p\right\|^2
                               +\sum_{t=p}^q\eta_t^2\mathbb E\left\|h_t\right\|^2\\
 &\le2L\sum_{t=p}^q\eta_t^2\delta_t+S^2\sum_{t=p}^q\eta_t^2.
\end{align*}
The right side still depends on the function errors. Write
$\delta_t=(\delta_t-\delta_p)+\delta_p$ there, and move the
first part to the left. This is an exact rearrangement, giving
\begin{equation}
 2\sum_{t=p}^q\eta_t(1-L\eta_t)(\delta_t-\delta_p)
 \le(2L\delta_p+S^2)\sum_{t=p}^q\eta_t^2.
 \label{eq:convex-reference}
\end{equation}
The differences $\delta_t-\delta_p$ need not be nonnegative.

\emph{Comparison of consecutive blocks.}
Use the comparison blocks \eqref{eq:comparison-blocks}.
For each $i$, choose a deterministic minimizer $\tau_i$ of
$\delta_t$ on $B_i$ and set $A_i=\delta_{\tau_i}$.
These are indices minimizing expected errors, not data-dependent
choices made by the algorithm. Write
$\bar\delta_i=|B_i|^{-1}\sum_{t\in B_i}\delta_t$ for a block mean.
In particular, $A_i\le\bar\delta_i$ and
$A_m=\bar\delta_m=\delta_T$ because $B_m=\{T\}$.

In \eqref{eq:convex-reference}, take $p=\tau_i$ and
$q=\max B_{i+1}$ for $0\le i<m$. On the remaining part of
$B_i$, each difference $\delta_t-A_i$ is nonnegative.
Its coefficient $2\eta_t(1-L\eta_t)$ is positive, so discarding
this source part can only decrease the left side.
Throughout the target block $B_{i+1}$ the stepsize equals
$\eta2^{-(i+1)}$. The remaining inequality is therefore
\begin{align*}
 &2\eta2^{-(i+1)}(1-L\eta2^{-(i+1)})|B_{i+1}|
                              (\bar\delta_{i+1}-A_i)\le(2LA_i+S^2)
                      \sum_{t=\tau_i}^{\max B_{i+1}}\eta_t^2.
\end{align*}
Every comparison span has at most $4|B_{i+1}|$ indices by
\eqref{eq:rounded-block-ratios}. Its largest stepsize is
at most $\eta2^{-i}$. Moreover,
$1-L\eta2^{-(i+1)}\ge1/2$. Thus the ratio of the sum of squared
stepsizes to the coefficient of $\bar\delta_{i+1}-A_i$ is at most
\[
 \frac{4|B_{i+1}|\eta^2 2^{-2i}}
      {|B_{i+1}|\eta2^{-(i+1)}}
 =8\eta2^{-i}=:c_i.
\]
Since $2LA_i+S^2\ge0$, it follows that
\begin{equation}
 A_{i+1}\le\bar\delta_{i+1}
            \le(1+2Lc_i)A_i+c_iS^2,
 \qquad c_i=8\eta2^{-i}.
 \label{eq:convex-block-recursion}
\end{equation}
The mean difference $\bar\delta_{i+1}-A_i$ itself may be negative;
division by its positive coefficient remains valid in that case.
The length bounds used here include $T=1$ and $T=2$, whose blocks
are respectively $B_0=\{0\}, B_1=\{1\}$ and
$B_0=\{0,1\}, B_1=\{2\}$.

\emph{Expansion of the scalar recursion.}
The first two substitutions in \eqref{eq:convex-block-recursion}
(the second when $m\ge2$) are
\begin{align*}
 A_1&\le(1+2Lc_0)A_0+c_0S^2,\\
 A_2&\le(1+2Lc_1)(1+2Lc_0)A_0
                  +\bigl(c_0(1+2Lc_1)+c_1\bigr)S^2.
\end{align*}
More generally, induction gives, for $0\le i<m$,
\begin{equation}
 \bar\delta_{i+1}
 \le A_0\prod_{j=0}^{i}(1+2Lc_j)
       +S^2\sum_{j=0}^{i}c_j
                    \prod_{\ell=j+1}^{i}(1+2Lc_\ell).
 \label{eq:convex-block-expanded}
\end{equation}
To see the induction step, use the same formula with index $i-1$
to bound $A_i\le\bar\delta_i$, multiply it by $1+2Lc_i$, and
add $c_iS^2$. This extends every existing product by its last
factor and adds the $j=i$ summand, whose empty product equals one.

The coefficients have a geometric sum:
\[
 \sum_{j=0}^{m-1}c_j=8\eta\sum_{j=0}^{m-1}2^{-j}\le16\eta.
\]
For any product occurring in \eqref{eq:convex-block-expanded}, use
$1+u\le\exp(u)$ for $u\ge0$ to obtain
\[
 \prod(1+2Lc_j)
 \le\exp\left(2L\sum_{j=0}^{m-1}c_j\right)
 \le\exp(32L\eta)\le\mathrm e.
\]
Hence every tail-block mean satisfies
\begin{align}
 \bar\delta_{i+1}
 &\le\mathrm e A_0+16\mathrm e\eta S^2\nonumber\\
 &\le\frac{4\mathrm e R^2}{\eta (T+1)}
                              +18\mathrm e\eta S^2.
 \label{eq:convex-each-block}
\end{align}
The second line inserts \eqref{eq:convex-first-reference}.
The coefficients $c_j$ form a geometric sequence, so neither term
incurs a factor proportional to the number of blocks.

\emph{The last iterate and the uniform mean.}
The last block is a singleton, so \eqref{eq:convex-each-block}
with $i+1=m$ is already the desired last-point inequality.
For the full mean, partition $0,\ldots,T$ into the entire
first schedule block and $B_1,\ldots,B_m$:
\[
 \frac1{T+1}\sum_{t=0}^T\delta_t
 =\frac K{T+1}\left(\frac1K\sum_{t=0}^{K-1}\delta_t\right)
       +\sum_{i=1}^m\frac{|B_i|}{T+1}\bar\delta_i.
\]
The coefficients on the right sum to one. The first mean is
bounded by \eqref{eq:convex-first-mean}, which is no larger than
the right side of \eqref{eq:convex-each-block}; every remaining
mean is bounded by \eqref{eq:convex-each-block} itself.
Their convex combination obeys the same bound. This completes
the proof of \eqref{eq:convex-companion-last}.

\end{align*}
Substitute $W_T=(T+1)(T+2s)/2$, $S^2\le9\sigma^2/4$, and
$\beta=1/(T+s+1)$ to obtain the sharper three-term bound
\begin{equation}
 \mathbb E[F(\bar x_T)-F^*]
 \le\frac{2\lambda s(s-1)R^2}{(T+1)(T+2s)}
       +\frac{18\sigma^2}{\lambda(T+2s)}
       +\frac{3\kappa\sigma^2}{\lambda(T+s+1)^2}.
 \label{eq:sc-mean-exact-app}
\end{equation}
For the simpler bound, use $s\ge4\kappa\ge4$:
\begin{align*}
 \frac{18\sigma^2}{\lambda(T+2s)}
 +\frac{3\kappa\sigma^2}{\lambda(T+s+1)^2}
 &\le\left(18+\frac{3\kappa}{T+s+1}\right)
                    \frac{\sigma^2}{\lambda(T+s+1)}\\
 &\le\left(18+\frac34\right)\frac{\sigma^2}{\lambda(T+s+1)}
 \le\frac{19\sigma^2}{\lambda(T+s+1)}.
\end{align*}
If $J$ is independent of the run with
$\Pr(J=t)=(t+s)/W_T$, the weighted average of expected STORM function gaps equals
$\mathbb E[F(x_J)-F^*]$. The same calculation therefore proves the
random-output assertion.

\label{lem:blocks}
Let $F$ be differentiable and convex, and consider
\[
 y_{t+1}=y_t-\eta_t h_t,\qquad
 \mathbb E[h_t\mid\mathcal F_t]=\nabla F(y_t),
\]
where $(\mathcal F_t)$ is a filtration and $y_t$ is
$\mathcal F_t$-measurable. Let $T\ge1$ be an integer and set
\[
 M=\lceil (T+1)/4\rceil,\qquad K=\lceil (T+1)/2\rceil,\qquad
 \gamma_t=\frac{c}{t+s+1},\qquad c>0,\ s\ge0.
\]
Use the boundaries in \eqref{eq:rounded-boundaries} and the stepsizes
$\eta_t=2^{-i}\gamma_t$ for $U_i\le t<U_{i+1}$.
Assume the displayed function-value expectations are finite.
If $\mathbb E\left\|h_t\right\|^2\le G^2$ for $M-1\le t\le T$, then
\begin{equation}
 \mathbb E F(y_T)
 \le\min_{M-1\le t\le K-1}\mathbb E F(y_t)+64G^2\gamma_T.
 \label{eq:block-result}
\end{equation}
It is enough to introduce $y_{T+1}$ virtually; the output is $y_T$.

Since $T\ge1$, we have $m\ge1$. We use the geometry proved in
Lemma~\ref{lem:rounded-geometry}:
$M\ge (T+1)/4$, $K\le2M$, and $n=K-M+1\ge (T+1)/4$.

\emph{Comparison with an earlier point.}
Fix deterministic indices $M-1\le p\le q\le T$.
For every $p\le t\le q$, expand the update around $y_p$:
\[
 \left\|y_{t+1}-y_p\right\|^2
 =\left\|y_t-y_p\right\|^2-2\eta_t\left\langleh_t,y_t-y_p\right\rangle
                       +\eta_t^2\left\|h_t\right\|^2.
\]
Although $y_p$ is random, it is $\mathcal F_t$-measurable because
$p\le t$. The same is true of $y_t$. Consequently,
\begin{align*}
 \mathbb E[\left\langleh_t,y_t-y_p\right\rangle\mid\mathcal F_t]
 &=\left\langle\mathbb E[h_t\mid\mathcal F_t],y_t-y_p\right\rangle\\
 &=\left\langle\nabla F(y_t),y_t-y_p\right\rangle\\
 &\ge F(y_t)-F(y_p).
\end{align*}
The last inequality is convexity applied with expansion point $y_t$:
\[
 F(y_p)\ge F(y_t)+\left\langle\nabla F(y_t),y_p-y_t\right\rangle.
\]
Taking expectations in the distance identity therefore gives
\[
 2\eta_t\bigl(\mathbb E F(y_t)-\mathbb E F(y_p)\bigr)
 \le \mathbb E\left\|y_t-y_p\right\|^2-\mathbb E\left\|y_{t+1}-y_p\right\|^2
                +\eta_t^2G^2.
\]
Summing from $t=p$ to $q$ cancels intermediate distance.
The first distance is $\left\|y_p-y_p\right\|^2=0$, so
\begin{align*}
 2\sum_{t=p}^q\eta_t\bigl(\mathbb E F(y_t)-\mathbb E F(y_p)\bigr)
 \le-\mathbb E\left\|y_{q+1}-y_p\right\|^2
                          +G^2\sum_{t=p}^q\eta_t^2\le G^2\sum_{t=p}^q\eta_t^2.
\end{align*}
We will use the resulting inequality in the form
\begin{equation}
 2\sum_{t=p}^q\eta_t\bigl(\mathbb E F(y_t)-\mathbb E F(y_p)\bigr)
 \le G^2\sum_{t=p}^q\eta_t^2.
 \label{eq:reference-comparison}
\end{equation}
All distances in this calculation have finite expectations: indeed,
$y_t-y_p=-\sum_{j=p}^{t-1}\eta_jh_j$ is a finite sum of random vectors
with finite second moments.

\emph{Comparison of adjacent blocks.}
Use the comparison blocks \eqref{eq:comparison-blocks}. They are
consecutive from $M-1$ through $T$, and $B_m=\{T\}$.
Choose index $\tau_i\in B_i$ minimizing
$\mathbb E F(y_t)$ on $B_i$, and write
\[
 H_i=\mathbb E F(y_{\tau_{i+1}})-\mathbb E F(y_{\tau_i})
 \qquad(0\le i<m).
\]
The index $\tau_i$ is deterministic because the quantities being
minimized are expectations, not observed sample-path values.

Apply \eqref{eq:reference-comparison} with
$p=\tau_i$ and $q=\max B_{i+1}$. The left sum consists of the
remaining part of the source block $B_i$ and all of the target block
$B_{i+1}$. On the source part, the choice of $\tau_i$ gives
\[
 \mathbb E F(y_t)-\mathbb E F(y_{\tau_i})\ge0.
\]
On the target block, the choice of $\tau_{i+1}$ gives
\[
 \mathbb E F(y_t)-\mathbb E F(y_{\tau_i})
 \ge\mathbb E F(y_{\tau_{i+1}})-\mathbb E F(y_{\tau_i})=H_i.
\]
All stepsizes are positive. Thus the whole left side of
\eqref{eq:reference-comparison} is at least
$2H_i\sum_{t\in B_{i+1}}\eta_t$. Dividing by this positive
weight sum gives
\begin{equation}
 H_i\le
 \frac{G^2\sum_{t=\tau_i}^{\max B_{i+1}}\eta_t^2}
      {2\sum_{t\in B_{i+1}}\eta_t}.
 \label{eq:adjacent-blocks}
\end{equation}
This reasoning is valid whether $H_i$ is positive, zero, or negative.

\emph{The effect of rounded block lengths.}
First consider $i=0$. By \eqref{eq:rounded-block-ratios}, the numerator
in \eqref{eq:adjacent-blocks} contains at most $3|B_1|$ terms.
Furthermore, $T+1\le4M$ and $s\ge0$ give
\[
 \eta_t\le\gamma_{M-1}=\frac{c}{M+s}
                    \le4\gamma_T
 \quad\text{throughout the first comparison span}.
\]
On $B_1$, the multiplier is $1/2$, and $t\le T$ implies
$\gamma_t\ge\gamma_T$. Each target stepsize is therefore
at least $\gamma_T/2$. Substituting these upper and lower
bounds into \eqref{eq:adjacent-blocks} yields
\[
 H_0\le
 \frac{3|B_1|G^2(4\gamma_T)^2}
      {2|B_1|(\gamma_T/2)}
 =48G^2\gamma_T.
\]

Next take $1\le i<m$. By \eqref{eq:rounded-block-ratios}, the
comparison span contains at most $4|B_{i+1}|$ indices.
All its indices satisfy $t\ge U_1=K\ge (T+1)/2$, whence
\[
 \gamma_t=\frac{c}{t+s+1}\le2\gamma_T.
\]
The largest multiplier in this span is $2^{-i}$, so its largest
stepsize is at most $2^{1-i}\gamma_T$.
On the target block the multiplier is $2^{-i-1}$, and every
stepsize is at least $2^{-i-1}\gamma_T$. Consequently,
\[
 H_i\le
 \frac{4|B_{i+1}|G^2(2^{1-i}\gamma_T)^2}
      {2|B_{i+1}|2^{-i-1}\gamma_T}
 =16\,2^{-i}G^2\gamma_T.
\]
When $m=1$, there are no comparisons of this second type.

\emph{Summation over the blocks.}
Since $\tau_m=T$, consecutive differences telescope:
\begin{align*}
 \mathbb E F(y_T)-\mathbb E F(y_{\tau_0})
 &=\sum_{i=0}^{m-1}H_i\\
 &\le\left(48+16\sum_{i=1}^{m-1}2^{-i}\right)
                                      G^2\gamma_T\\
 &\le64G^2\gamma_T.
\end{align*}
The defining property of $\tau_0$ now proves
\eqref{eq:block-result}. The last comparison uses the update at
$t=T$ only to produce the nonnegative terminal distance
$\mathbb E\left\|y_{T+1}-y_{\tau_{m-1}}\right\|^2$, which is discarded.
The point $y_{T+1}$ need not be returned or computed by the algorithm.

Apply Lemma~\ref{lem:companion}. In particular, the companion
starts at $y_0=x_0$ and has centered innovations satisfying
\[
 \mathbb E[\nu_t\mid\mathcal F_t]=0,\qquad
 \mathbb E[\left\|\nu_t\right\|^2\mid\mathcal F_t]\le S^2,
 \qquad S^2=(1+\beta)^2\sigma^2\le\frac94\sigma^2.
\]
Write
\[
 D_t=\mathbb E\left\|y_t-x^*\right\|^2,\qquad
 \delta_t=\mathbb E[F(y_t)-F^*],\qquad
 \gamma_t=\frac2{\lambda(t+s+1)},
\]
and let $M=\lceil (T+1)/4\rceil$, $K=\lceil (T+1)/2\rceil$.
The schedule \eqref{eq:sctaper} satisfies $\eta_t=\gamma_t$ for $0\le t<K$,
and $\eta_t\le\gamma_t$ everywhere. Since
$s\ge4\kappa^2\ge4\kappa$, every actual and virtual stepsize
is at most $1/(2L)$.

\emph{Distance decrease in the first half.}
Expand one companion update around $x^*$:
\[
 y_{t+1}-x^*
 =y_t-x^*-\eta_t\nabla F(y_t)-\eta_t\nu_t.
\]
The first three terms are $\mathcal F_t$-measurable, so the
cross term with $\nu_t$ vanishes after taking expectation.
Gradient-map contraction and $\nabla F(x^*)=0$ then imply
\begin{align}
 D_{t+1}
 &\le(1-\lambda\eta_t)^2D_t+\eta_t^2S^2\nonumber\\
 &\le(1-\lambda\eta_t)D_t+\eta_t^2S^2.
 \label{eq:last-distance}
\end{align}
The second line follows from $0\le\lambda\eta_t\le1$.
During the first block, where $\eta_t=\gamma_t$, we claim that
\begin{equation}
 D_t\le\frac{(s+1)^2R^2}{(t+s+1)^2}
                  +\frac{4S^2}{\lambda^2(t+s+1)},\qquad 0\le t\le K.
 \label{eq:burn-in}
\end{equation}
At $t=0$, the first term equals $D_0=R^2$ and the second is
nonnegative. For the induction step, put $u=t+s+1$.
Here $u\ge5$, so $1-2/u\ge0$ and substitution of the induction
upper bound into \eqref{eq:last-distance} preserves the inequality.
Since $\eta_t=2/(\lambda u)$, this gives
\begin{align*}
 D_{t+1}
 &\le\left(1-\frac2u\right)
       \left(\frac{(s+1)^2R^2}{u^2}
                    +\frac{4S^2}{\lambda^2u}\right)
                    +\frac{4S^2}{\lambda^2u^2}\\
 &=\frac{(s+1)^2R^2(u-2)}{u^3}
                    +\frac{4S^2(u-1)}{\lambda^2u^2}\\
 &\le\frac{(s+1)^2R^2}{(u+1)^2}
                    +\frac{4S^2}{\lambda^2(u+1)}.
\end{align*}
The last line follows term by term from
\[
 (u-2)(u+1)^2=u^3-3u-2\le u^3,
 \qquad (u-1)(u+1)=u^2-1\le u^2.
\]
This proves \eqref{eq:burn-in}.

\emph{Second moments of the search directions.}
For any $t\ge M-1$, discard contraction in
\eqref{eq:last-distance} and sum from $M-1$ to $t-1$:
\[
 D_t\le D_{M-1}+S^2\sum_{j=M-1}^{t-1}\eta_j^2
       \le D_{M-1}+S^2\sum_{j=M-1}^{\infty}\gamma_j^2.
\]
The infinite sum is only an upper bound on a finite sum; it does
not introduce additional algorithmic iterations. By monotonicity
of the summand and $M+s\ge1$,
\begin{align*}
 \sum_{j=M-1}^{\infty}\gamma_j^2
 &=\frac4{\lambda^2}\sum_{k=M}^{\infty}\frac1{(k+s)^2}\\
 &\le\frac4{\lambda^2}
       \left(\frac1{(M+s)^2}
                         +\int_M^\infty\frac{du}{(u+s)^2}\right)\\
 &=\frac4{\lambda^2}\left(\frac1{(M+s)^2}+\frac1{M+s}\right)
 \le\frac8{\lambda^2(M+s)}.
\end{align*}
Use \eqref{eq:burn-in} at $t=M-1$ to conclude that
\begin{equation}
 D_t\le\frac{(s+1)^2R^2}{(M+s)^2}
                    +\frac{12S^2}{\lambda^2(M+s)},
                    \qquad M-1\le t\le T.
 \label{eq:tail-distance}
\end{equation}

Now write $h_t=\nabla F(y_t)+\nu_t$ for the companion direction.
Since $y_t$ is $\mathcal F_t$-measurable, conditional centering gives
$\mathbb E\left\langle\nabla F(y_t),\nu_t\right\rangle=0$.
Standard smoothness and $\nabla F(x^*)=0$ imply
$\left\|\nabla F(y_t)\right\|\le L\left\|y_t-x^*\right\|$.
Consequently,
\begin{align*}
 \mathbb E\left\|h_t\right\|^2
 &=\mathbb E\left\|\nabla F(y_t)\right\|^2+\mathbb E\left\|\nu_t\right\|^2\\
 &\le L^2D_t+S^2\\
 &\le\frac{L^2(s+1)^2R^2}{(M+s)^2}
        +\left(1+\frac{12\kappa^2}{M+s}\right)S^2\\
 &\le\frac{L^2(s+1)^2R^2}{(M+s)^2}+4S^2=:G^2.
\end{align*}
For the last line, $M+s\ge s\ge4\kappa^2$ gives
$12\kappa^2/(M+s)\le3$. Thus the moment assumption of
Lemma~\ref{lem:blocks} holds on the required trajectory segment.
This argument uses smoothness of $F$ only; it imposes neither
sample smoothness nor a globally bounded stochastic gradient.

\emph{A reference function value.}
For $M-1\le t\le K-1$, the stepsize is $\gamma_t$.
Lemma~\ref{lem:convex-distance} applies because $\gamma_t\le1/(2L)$,
and gives
\[
 D_{t+1}\le D_t-\gamma_t\delta_t+\gamma_t^2S^2.
\]
Rearranging and summing makes the interior distances cancel:
\[
 \sum_{t=M-1}^{K-1}\gamma_t\delta_t
 \le D_{M-1}-D_K+S^2\sum_{t=M-1}^{K-1}\gamma_t^2
 \le D_{M-1}+S^2\sum_{t=M-1}^{K-1}\gamma_t^2.
\]
The left side is at least the smallest $\delta_t$ in the range
times the positive sum of weights. Hence
\[
 \min_{M-1\le t\le K-1}\delta_t
 \le\frac{D_{M-1}+S^2\sum_{t=M-1}^{K-1}\gamma_t^2}
              {\sum_{t=M-1}^{K-1}\gamma_t}.
\]
There are $n=K-M+1\ge (T+1)/4$ indices. Their largest denominator
in $\gamma_t$ is $K+s$ and their smallest is $M+s$, so
\[
 \sum_{t=M-1}^{K-1}\gamma_t\ge\frac{2n}{\lambda(K+s)},
 \qquad
 \sum_{t=M-1}^{K-1}\gamma_t^2\le\frac{4n}{\lambda^2(M+s)^2}.
\]
Also $K\le2M$, and therefore $K+s\le2(M+s)$. It follows that
\begin{align*}
 \min_{M-1\le t\le K-1}\delta_t
 &\le\frac{\lambda(K+s)}{2n}D_{M-1}
          +\frac{2S^2(K+s)}{\lambda(M+s)^2}\\
 &\le\frac{4\lambda(M+s)}{T+1} D_{M-1}
          +\frac{4S^2}{\lambda(M+s)}\\
 &\le\frac{4\lambda(s+1)^2R^2}{(T+1)(M+s)}
          +\frac{16S^2}{\lambda (T+1)}+\frac{4S^2}{\lambda(M+s)}\\
 &\le\frac{4\lambda(s+1)^2R^2}{(T+1)(M+s)}+\frac{32S^2}{\lambda (T+1)}.
\end{align*}
In the third line, we use the sharper bound
\eqref{eq:burn-in} at $M-1$, whose noise coefficient is $4$,
rather than the tail bound with coefficient $12$.
For the last line, use $M+s\ge (T+1)/4$.

\emph{The last companion iterate.}
Lemma~\ref{lem:blocks}, with $c=2/\lambda$, adds at most
\begin{align*}
 64G^2\gamma_T
 &=\frac{128L^2(s+1)^2R^2}{\lambda(T+s+1)(M+s)^2}
                        +\frac{512S^2}{\lambda(T+s+1)}\\
 &\le\frac{32\lambda(s+1)^2R^2}{(T+1)(M+s)}
                        +\frac{512S^2}{\lambda (T+1)}.
\end{align*}
For the initialization term, divide its coefficient by the
coefficient of $\lambda(s+1)^2R^2/[(T+1)(M+s)]$:
\[
 \frac{128\kappa^2(T+1)}{(T+s+1)(M+s)}
 =128\frac{\kappa^2}{M+s}\frac{T+1}{T+s+1}\le32.
\]
Combining this with the preceding bound on the minimum over
$M-1\le t\le K-1$, and using $4+32=36$ and
$32+512=544$, gives
\begin{equation}
 \mathbb E[F(y_T)-F^*]
 \le\frac{36\lambda(s+1)^2R^2}{(T+1)(M+s)}
                        +\frac{544S^2}{\lambda (T+1)}.
 \label{eq:last-companion-bound}
\end{equation}


\emph{Transfer to STORM.}
The comparison bound \eqref{eq:expected-transfer-app} gives
\[
 \mathbb E[F(x_T)-F^*]
 \le2\mathbb E[F(y_T)-F^*]
                  +\frac{3L\beta^2\sigma^2}{\lambda^2}.
\]
Substitute \eqref{eq:last-companion-bound}, then use
$S^2\le9\sigma^2/4$ and $\beta=1/(T+s+1)$:
\begin{align*}
 \mathbb E[F(x_T)-F^*]
 &\le\frac{72\lambda(s+1)^2R^2}{(T+1)(M+s)}
        +\frac{1088S^2}{\lambda (T+1)}
        +\frac{3\kappa\sigma^2}{\lambda(T+s+1)^2}\\
 &\le\frac{72\lambda(s+1)^2R^2}{(T+1)(M+s)}
        +\frac{2448\sigma^2}{\lambda (T+1)}
        +\frac{3\kappa\sigma^2}{\lambda(T+s+1)^2}.
\end{align*}
Finally, $s\ge4\kappa^2\ge\kappa$ and
$(T+s+1)^2\ge4(T+1)s\ge\kappa(T+1)$ imply
\[
 \frac{3\kappa\sigma^2}{\lambda(T+s+1)^2}
 \le\frac{3\sigma^2}{\lambda (T+1)}.
\]
Finally, \(M+s\ge(T+1)/4\), so \(72/[(T+1)(M+s)]\le288/(T+1)^2\), which proves the bound in Theorem~\ref{thm:strong}.
