<!-- lemma-1: body_md -->
\label{lem:reflection-positivity}
For every $i,b$ as above, $\widetilde{\mathbb P}_{L,\beta,h}$ is reflection
positive with respect to $\widetilde\vartheta_{i,b}$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Fix $i,b$, and abbreviate
$\widetilde\vartheta=\widetilde\vartheta_{i,b}$ and
$\widetilde{\mathbb P}=\widetilde{\mathbb P}_{L,\beta,h}$.  Let $\mathcal{F}_0$ be the $\sigma$-algebra generated by $X_z$'s with $z$ in the reflection plane: 
\[
 \mathcal F_0
 :=\sigma\bigl(X_z:z_i=2b\text{ or }z_i=2b+L\bigr).
\]
Apart from deterministic variables, this records precisely the spins on the original reflection plane and the bonds lying entirely within it.

Since the reflecting planes pass through lattice sites, \eqref{eq:ES-measure} implies that the two half-tori are conditionally
independent given $\mathcal F_0$.  Since $\vartheta_{i,b}$ preserves vertex parity, 
\[\mathbb P_{L,\beta,h}\bigl[\sigma,\omega\bigr]=\mathbb P_{L,\beta,h}\bigl[\vartheta_{i,b}\sigma,\vartheta_{i,b}\omega\bigr].\]
The reflection fixes all $X_z$'s generating
$\mathcal F_0$, so it exchanges the two conditional laws.  Hence
\[
 \left\langle F\,\widetilde\vartheta G\mid\mathcal F_0\right\rangle_{\widetilde{\mathbb P}}
 =
 \left\langle F\mid\mathcal F_0\right\rangle_{\widetilde{\mathbb P}}
 \left\langle G\mid\mathcal F_0\right\rangle_{\widetilde{\mathbb P}}.
\]
Averaging over the joint law of the variables generating
$\mathcal F_0$ gives
\begin{equation}\label{eq:rp-conditional-product}
 \langle F\,\widetilde\vartheta G\rangle_{\widetilde{\mathbb P}}
 =
 \left\langle
 \left\langle F\mid\mathcal F_0\right\rangle_{\widetilde{\mathbb P}}
 \left\langle G\mid\mathcal F_0\right\rangle_{\widetilde{\mathbb P}}
 \right\rangle_{\widetilde{\mathbb P}}.
\end{equation}
This expression is symmetric in $F,G$, and taking $G=F$ yields
\[
 \langle F\,\widetilde\vartheta F\rangle_{\widetilde{\mathbb P}}
 =
 \left\langle
 \left(\left\langle F\mid\mathcal F_0\right\rangle_{\widetilde{\mathbb P}}\right)^2
 \right\rangle_{\widetilde{\mathbb P}}
 \ge0.
\]
Thus both conditions for reflection positivity hold.

<!-- lemma-3: body_md -->
\label{lem:good-geometry}
Let $d\ge2$ and $\mu$ be a probability measure on $\{0,1\}^{\mathbb Z^d}$.  Call a vertex \emph{open} when $\eta_v=1$ and \emph{closed} when $\eta_v=0$.  Suppose that, for some $r>0$,
\begin{equation}\label{eq:abstract-bad-bound}
 \mu\bigl[\eta_v=0\text{ for every }v\in A\bigr]\le r^{\lvertA\rvert}, \qquad \forall A\Subset\mathbb Z^d.
\end{equation}
Set $\kappa_d=(3^d-1)^2$.
If $r<\kappa_d^{-1}$, then the open vertices have,
$\mu$-almost surely, a unique infinite nearest-neighbor component
$\mathcal C_\infty$.  Moreover, for every $v\in\mathbb Z^d$,
\begin{equation}\label{eq:good-density}
 \mu\bigl[v\notin\mathcal C_\infty\bigr]
 \le r+\frac{\kappa_dr}{(1-\kappa_dr)^2}
 =:\delta_d(r).
\end{equation}
In particular, $\delta_d(r)\to0$ as $r\downarrow0$.
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
For a finite connected set $A$, let $\partial_{\mathrm{ex}}A$ be the
vertices adjacent to $A$ in the infinite component of $\mathbb Z^d\setminus A$.
This boundary set is $*$-connected by \cite[Lemma~2]{Tim13}.

Put $\rho=\kappa_dr<1$.  Every $*$-connected set containing $v$ with at
least $n$ vertices contains a tree with $n$ vertices containing $v$.
A traversal of this tree is a $*$-nearest-neighbor walk of length
$2(n-1)$, and there are at most
$(3^d-1)^{2(n-1)}\le\kappa_d^n$ such walks.  Hence
\begin{equation}\label{eq:large-bad-animal}
 \mu[\exists A \text{ s.t. } v\in A, \ A\text{ is $*$-connected and entirely closed}, \lvertA\rvert\ge n]\le\rho^n.
\end{equation}
Letting $n\to\infty$ in \eqref{eq:large-bad-animal} and then taking a countable union over $v$ imply that almost surely there is no infinite closed $*$-component.
Call a finite set a \emph{separator of $v$} if it separates $v$ from infinity.
A $*$-connected separator of size $n$ meets
$\{v+je_1:1\le j\le n\}$.  The same count therefore gives
\begin{equation}\label{eq:bad-separator-count}
 \mu[\exists\text{ a closed $*$-connected separator }A\text{ of }v \text{ s.t. }
             \lvertA\rvert=n]\le n\rho^n.
\end{equation}

For existence of $\mathcal{C}_{\infty}$, suppose all open components are finite.  Making each vertex in a finite
box $B_N(v):=v+[-N,N]^d \cap \mathbb Z^d$ open cannot create an infinite component, since
only finitely many original components can be joined.  The open component
containing $B_N(v)$ (assuming each vertex in $B_N(v)$ is open) thus has a finite, closed $*$-connected exterior boundary,
which lies outside $B_N(v)$ and is closed also in the original configuration.
These boundaries give separators of $v$ of arbitrarily large size,
an event of probability zero since \eqref{eq:bad-separator-count} implies that $\sum_{n\ge N}n\rho^n\to0$ as $N\rightarrow \infty$.

For uniqueness, suppose $\mathcal{C},\mathcal{C}'$ are distinct infinite open components,
and let $U$ be the component of $\mathbb Z^d\setminus \mathcal{C}$ containing $\mathcal{C}'$.
Again by \cite[Lemma~2]{Tim13}, the vertices in $U$ adjacent to $\mathcal{C}$
form a closed $*$-connected set.  This set separates $\mathcal{C}$ from $\mathcal{C}'$
and is therefore infinite; this is because deleting finitely many vertices from
$\mathbb Z^d$, $d\ge2$, leaves only one infinite component.  This contradicts the statement right after
\eqref{eq:large-bad-animal}.

Finally, if $v\notin\mathcal C_\infty$, then $v$ is either closed or its finite
open component has a closed exterior boundary separating $v$ from infinity.
Summing \eqref{eq:bad-separator-count} over all sizes gives
\begin{equation}\label{eq:delta-explicit}
 \mu[v\notin\mathcal C_\infty]\le r+\sum_{m\ge1}m\rho^m
 =r+\frac{\rho}{(1-\rho)^2}=\delta_d(r).
\end{equation}
This completes the proof of the lemma.

<!-- lemma-4: body_md -->
\label{lem:tail-sign}
For each $s\in \{-1,+1\}$,  $A_s$ is a tail event. Furthermore,
\begin{equation}\label{eq:sign-half}
 \mathbb P_{\beta,h}[A_+]=\mathbb P_{\beta,h}[A_-]=1/2.
\end{equation}
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
A finite change of spins and bonds changes the $s$-good status of only finitely many blocks. Removing finitely many vertices from an infinite component of a locally finite graph leaves an infinite component. Adding finitely many vertices cannot join infinitely many previously finite components. Hence $A_s$ is invariant under finite coordinate changes and is a tail event.

Since $q$ is odd, the transformation
\begin{equation}\label{eq:combined-symmetry}
 (\mathcal T\sigma)_u=-\sigma_{u+qe_1},
 \qquad
 (\mathcal T\omega)_{uv}=\omega_{u+qe_1,v+qe_1},
\end{equation}
preserves the torus measure defined in \eqref{eq:ES-measure}, and hence its limit $\mathbb P_{\beta,h}$.  The infinite component of good blocks is unique and the corresponding FK mesh cluster has a common sign, so $\mathbb P_{\beta,h}[A_+]+\mathbb P_{\beta,h}[A_-]=1$. Since $\mathcal{T}$ shifts the coarse lattice by $e_1$ and reverses the sign of the FK mesh cluster, $\mathbb P_{\beta,h}[A_+]=\mathbb P_{\beta,h}[A_-]$. This proves \eqref{eq:sign-half}.

<!-- proposition-4: body_md -->

 \qquad uv \in F.
\]

<!-- proof-6: body_md -->
For a full configuration $(\sigma, \omega)$, let $\pi_{\Lambda}$ be the kernel that samples $(\sigma_{\Lambda}, \omega_{\bar{E}(\Lambda)})$ according to~\eqref{eq:ES-measure-general}, with exterior spins and bonds fixed. 
Then an easy calculation shows
\[\mathbb P_{L,\beta,h}[B]=\sum_{\eta \in \{-1,+1\}^{\mathbb{T}_L\setminus \Lambda}} \sum_{\xi \in \{0,1\}^{E(\mathbb{T}_L \setminus \Lambda)}} \pi_{\Lambda} [B \mid \eta, \xi] \mathbb P_{L,\beta,h}[\eta,\xi],\quad \forall \Lambda \subset \mathbb{T}_L, \text{ local event }B.\]	

Passing to the chosen subsequential limit,
\[\mathbb P_{\beta,h}[B]= \int \pi_{\Lambda} [B \mid \eta, \xi] \mathbb P_{\beta,h}[d\eta, d\xi], \qquad \forall \Lambda \Subset \mathbb Z^d, \text{ local event }B.\]
Therefore,
\begin{equation*}
	\mathbb P_{\beta,h}[A \cap B]= \int_A \pi_{\Lambda} [B \mid \eta, \xi] \mathbb P_{\beta,h}[d\eta, d\xi], \qquad \forall \text{ local event }B, \ A \in \mathcal{F}_{\Lambda^c}, \ \Lambda \Subset \mathbb Z^d.
\end{equation*}
These are the DLR equations for the joint measure. Since $A_s$ is a tail event by Lemma~\ref{lem:tail-sign},
\begin{equation}\label{eq:DLRconditional}
	\widehat\mathbb P^{\,s}_{\beta,h}[B]=\frac{\mathbb P_{\beta,h}[B \cap A_s]}{\mathbb P_{\beta,h}[A_s]}=\frac{\int_{A_s}  \pi_{\Lambda} [B \mid \eta, \xi] \mathbb P_{\beta,h}[d\eta, d\xi]}{\mathbb P_{\beta,h}[A_s]}=\int \pi_{\Lambda} [B \mid \eta, \xi]  \widehat\mathbb P^{\,s}_{\beta,h}[d\eta, d\xi]
\end{equation}
for each local event $B$ and $\Lambda \Subset \mathbb Z^d$. These are the DLR equations for $\widehat\mathbb P^{\,s}_{\beta,h}$, which completes the proof of the first statement.

If $B$ in \eqref{eq:DLRconditional} is a local event for the Ising model, we get
\[\widehat\mu^s_{\beta,h}[B]=\int \mu^{\mathrm{FM},\eta}_{\Lambda,\beta,h}[B] \widehat\mu^s_{\beta,h}[d\eta], \qquad \forall \text{ local Ising event }B, \ \Lambda \Subset \mathbb Z^d;\]
here $\mu^{\mathrm{FM},\eta}_{\Lambda,\beta,h}$ is understood as its corresponding kernel if $B$ is not supported in $\Lambda$. These are the DLR equations for the Ising model.
	

Finally, \eqref{eq:DLRconditional} implies that, given $\mathcal{F}_{\Lambda^c}$ and the $\sigma$-algebra generated by spins in $\Lambda$, the bonds in $\bar{E}(\Lambda)$ are independent with opening probabilities $p \mathbf 1_{\{\sigma_u=\sigma_v\}}$. For any finite bond set $F$, choosing $\Lambda$ containing its endpoints and applying the tower property complete the proof of the last statement.

<!-- lemma-5: body_md -->
\label{lem:crossing-observable}
The function $f_q$ is increasing in $\sigma_{Q_0}$. For every sufficiently large odd $q$,
\begin{equation}\label{eq:crossing-bounds}
 \langle f_q\rangle_{\mu_h^+}\ge1-2\delta_q,
 \quad \forall |h|\le h_q,
 \qquad
 \langle f_q\rangle_{\mu_0^-}\le\epsilon_q,
\end{equation}
where $\epsilon_q\to0$ as $q\to\infty$.
<!-- end lemma-5 -->
<!-- proof-7: body_md -->
Given the spins, sample a bond configuration in $\{0,1\}^{E(Q_0)}$ as
\[
 \omega_{uv}=b_{uv}\mathbf 1_{\{\sigma_u=\sigma_v\}},
 \qquad b_{uv}\text{'s} \sim \text{ independent Bernoulli}(p).
\]
For fixed $(b_{uv})$, increasing the spins preserves every positive
open path. Averaging over these variables proves the monotonicity.

On $\{0\in\mathcal G_\infty,A_+\}$, the crossing cluster of $Q_0$ belongs
to the positive FK mesh cluster, so $C_q^+$ occurs. Therefore, \eqref{eq:delta-choice} and \eqref{eq:sign-half} give
\begin{equation}\label{eq:crossinglowerbound}
	\langle f_q\rangle_{\widehat\mu^+_{\beta,h}}
	=\widehat\mathbb P^{\,+}_{\beta,h}[C_q^+] \ge 1-\frac{\mathbb P_{\beta,h}[0\notin\mathcal G_\infty, A_+]}
	{\mathbb P_{\beta,h}[A_+]}
	\ge1-2\delta_q.
\end{equation}
 
Stochastic ordering of the ferromagnetic Ising measures gives $\widehat\mu^+_{\beta,h}\preceq\mu_h^+$. So
monotonicity of $f_q$ proves the first bound.

For the second bound, let $\mathbb P_{\beta,0}^-$ be the zero-field infinite-volume Edwards--Sokal coupling measure with spin marginal $\mu_0^-$ and bond marginal the infinite-volume wired zero-field FK measure $\phi_\beta^{\mathrm w}$. Conditional on $\sigma_{Q_0}$, the
bonds in $E(Q_0)$ are again independent, with open probabilities
$p\mathbf 1_{\{\sigma_u=\sigma_v\}}$.
Hence
\[
 f_q(\sigma_{Q_0})
 =\mathbb P_{\beta,0}^-\bigl[C_q^+\mid\sigma_{Q_0}\bigr],
 \qquad
 \langle f_q\rangle_{\mu_0^-}=\mathbb P_{\beta,0}^-[C_q^+].
\]
Since the infinite FK cluster under $\mathbb P_{\beta,0}^-$  has sign $-$, every FK
crossing with positive sign lies in a finite cluster.
Write $\mathcal C(x)$ for the FK cluster of $x$ and
$\operatorname{rad}_x\mathcal C(x):=\sup_{y\in\mathcal C(x)}\|y-x\|_\infty$.
There exist $c=c(d,\beta), C=C(d,\beta) \in (0,\infty)$ such that
\[
 \phi_\beta^{\mathrm w}
 \bigl[n\le\operatorname{rad}_x\mathcal C(x)<\infty\bigr]
 \le Ce^{-cn},\qquad n\ge1.
\]
For $d\ge3$, this follows from \cite[Theorem~5.104]{Gri06} and \cite{Bod05}; for $d=2$, it follows from planar duality and subcritical exponential decay, see, e.g., \cite[Chapter~6]{Gri06}. Every such crossing starts at some $x$ on the face
$F_q^-:=\{x\in Q_0:x_1=-q\}$, and its cluster has radius at least $2q$ from $x$. A union bound therefore gives
\[
 \langle f_q\rangle_{\mu_0^-}
 \le\sum_{x\in F_q^-}\phi_\beta^{\mathrm w}
 \bigl[2q\le\operatorname{rad}_x\mathcal C(x)<\infty\bigr]
 \le C(2q+1)^{d-1}e^{-2cq}
 =:\epsilon_q\longrightarrow0.
\]
This finishes the proof of the lemma.

<!-- proposition-5: body_md -->
\label{prop:continuity}
	Let $d\ge2$ and $\beta>\beta_c(d)$. Then
	\begin{equation}\label{eq:local-continuity}
		\mu^{\mathrm{FM},\pm}_{\beta,h}\Rightarrow\mu^{\mathrm{FM},\pm}_{\beta,0}, \ \mu^{\mathrm{AF},\mathrm N\pm}_{\beta,h}\Rightarrow\mu^{\mathrm{AF},\mathrm N\pm}_{\beta,0}
		\qquad\text{as }h\to0,
	\end{equation}
where $\Rightarrow$ denotes weak convergence.
<!-- end proposition-5 -->
<!-- proof-8: body_md -->
By compactness, it suffices to identify any subsequential limit
$\mu_{h_n}^+\Rightarrow\nu$ with $h_n\to0$. By the DLR equations for $\mu_{h_n}^+$, for each $\Lambda \Subset \mathbb Z^d$ and local event $B$,
\begin{equation}\label{eq:DLRmu_h}
	\begin{aligned}
		\mu_{h_n}^+[B]&= \int \mu_{\Lambda,\beta,h_n}^{\mathrm{FM},\eta}\bigl[B\bigr]\mu_{h_n}^+[d\eta]\\
		&=\int \left(\mu_{\Lambda,\beta,h_n}^{\mathrm{FM},\eta}\bigl[B\bigr]-\mu_{\Lambda,\beta,0}^{\mathrm{FM},\eta}\bigl[B\bigr]\right) \mu_{h_n}^+(d\eta) + \int \mu_{\Lambda,\beta,0}^{\mathrm{FM},\eta}\bigl[B\bigr] \mu_{h_n}^+(d\eta);
	\end{aligned}
\end{equation}
here again $\mu_{\Lambda,\beta,h}^{\mathrm{FM},\eta}$ is understood as its kernel. Since 
\[\sup_{\eta}\left |\mu_{\Lambda,\beta,h_n}^{\mathrm{FM},\eta}\bigl[B\bigr]-\mu_{\Lambda,\beta,0}^{\mathrm{FM},\eta}\bigl[B\bigr]\right | \rightarrow 0 \text{ as } n\rightarrow \infty,\]
we send $n\rightarrow\infty$ in \eqref{eq:DLRmu_h} to obtain
\[\nu[B]=\int \mu_{\Lambda,\beta,0}^{\mathrm{FM},\eta}\bigl[B\bigr] \nu[d\eta].\]
These are the DLR equations for $\nu$. So $\nu$ is a zero-field Gibbs measure.

Write $T_z$ for translation by $z$. It acts on spin configurations by $(T_z\sigma)_u=\sigma_{u+z}$ and on measures by pushforward. Each $\mu_h^+$ is invariant under translations by $z$ with $|z|$ even, and $T_{e_1}\mu_h^+=\mu_{-h}^+$.
Then we have $\mu_{-h_n}^+ \Rightarrow T_{e_1} \nu$, and
\[
 \overline\nu:=\tfrac12(\nu+T_{e_1}\nu)
\]
is a translation-invariant zero-field Gibbs measure. The classification
of such measures gives
\begin{equation}\label{eq:mixture}
 \overline\nu=\alpha\mu_0^++(1-\alpha)\mu_0^-,
 \qquad \alpha\in[0,1],
\end{equation}
by the Aizenman--Higuchi theorem \cite{Aiz80,Hig81} for $d=2$ and Bodineau's theorem \cite{Bod06} for $d\ge3$.

Fix a sufficiently large odd $q$. We have $|h_n|\le h_q$ for all sufficiently large $n$.
Applying Lemma~\ref{lem:crossing-observable} to both $h_n$ and $-h_n$,
and using locality of $f_q$ to pass to the limit $h_n \rightarrow 0$, give
\[
 1-2\delta_q
 \le\langle f_q\rangle_{\overline\nu}
 \le\alpha+(1-\alpha)\epsilon_q.
\]
Letting $q\to\infty$ now gives $\alpha=1$. Since $\mu_0^+$ is extremal
among all Gibbs measures \cite[Lemma 6.65]{FV18},
$\overline\nu=\mu_0^+$ implies $\nu=\mu_0^+$.

Therefore $\mu_h^+\Rightarrow\mu_0^+$ as $h\rightarrow0$. Since a global spin flip maps
$\mu_{-h}^+$ to $\mu_h^-$, we have $\mu_h^-\Rightarrow\mu_0^-$ as $h\rightarrow0$.
The bipartite spin transformation then gives $\mu^{\mathrm{AF},\mathrm N\pm}_{\beta,h}\Rightarrow\mu^{\mathrm{AF},\mathrm N\pm}_{\beta,0}$.

Proposition \ref{prop:continuity} implies that
\[\lim_{h\rightarrow 0} \frac{\langle \sigma_0 \rangle^{\mathrm{FM},+}_{\beta,h}+\langle \sigma_{e_1} \rangle^{\mathrm{FM},+}_{\beta,h}}{2}=\langle \sigma_0 \rangle^{\mathrm{FM},+}_{\beta,0}>0.\]
So there is $h_0=h_0(d,\beta)>0$ such that
\[\frac{\langle \sigma_0 \rangle^{\mathrm{FM},+}_{\beta,h}+\langle \sigma_{e_1} \rangle^{\mathrm{FM},+}_{\beta,h}}{2}\geq \frac{\langle \sigma_0 \rangle^{\mathrm{FM},+}_{\beta,0}}{2}=:c>0, \qquad \forall |h|\leq h_0. \]
Combining this with \eqref{eq:m-symmetry}, we obtain
\[\frac{\langle \sigma_0 \rangle^{\mathrm{FM},+}_{\beta,h}-\langle \sigma_0 \rangle^{\mathrm{FM},-}_{\beta,h}}{2}= \mu_{\beta,h}^{\mathrm{FM},+}\bigl[\sigma_0=+1\bigr] - \mu_{\beta,h}^{\mathrm{FM},-}\bigl[\sigma_0=+1\bigr] \geq c, \qquad \forall |h|\leq h_0. \]
Monotonicity in the boundary condition implies that for each $n\geq 1$,
\[\mu^{\mathrm{FM},+}_{B_n,\beta,h}\bigl[\sigma_0=+1\bigr]
-\mu^{\mathrm{FM},-}_{B_n,\beta,h}\bigl[\sigma_0=+1\bigr] \geq \mu_{\beta,h}^{\mathrm{FM},+}\bigl[\sigma_0=+1\bigr] - \mu_{\beta,h}^{\mathrm{FM},-}\bigl[\sigma_0=+1\bigr] \geq c, \quad |h|\leq h_0.\]
This completes the proof of Proposition~\ref{prop:main-FM}
and thus Theorem~\ref{thm:main-AF}.
