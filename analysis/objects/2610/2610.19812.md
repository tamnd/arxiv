\label{A}
     Let $f:(z,w)\in \mathbb{C}^2 \mapsto (p(z), q(z,w))\in \mathbb{C}^2$ be a polynomial skew product of algebraic degree $d\geqslant 2$. Let $\mu$ be the Green measure associated to $f$ and $\lambda_1 > \lambda_2$ its Lyapunov exponents such that $\lambda_2$ is the Lyapunov exponent of $(p, \mu_p)$. Then, for $\mu$-almost every $x\in\mathbb{C}^2$, we have 
     \begin{equation*}
        \lim_{r\to0}\frac{\log\mu\left(\mathbb{D}^2(x, r)\right)}{\log r} =  \frac{\log d}{\lambda_1}+\frac{\log d}{\lambda_2}.
    \end{equation*}

<!-- theo-2: body_md -->
\label{Oseledets}
    Let $\lambda_1 > \lambda_2$ be the Lyapunov exponents of $(f,\mu)$ and $\lambda_p$ the Lyapunov exponent of $(p,\mu_p)$. Then $\{\lambda_1,\lambda_2\} = \{\lambda_p, \lambda_q\}.$ Moreover, if $\lambda_2 = \lambda_p$, then the Oseledets space associated to $\lambda_q$ is $\{0\}\times\mathbb{C}$.
<!-- end theo-2 -->
<!-- proof-1: body_md -->
We know that $\log \lvert p'\lvert $ is integrable against $\mu_p$ since the latter admits a continuous Green potential. The Birkhoff theorem gives 
    \begin{equation*}
        \frac1n\sum_{k=0}^{n-1}\log|p'(p^k(z))| \overset{n\to\infty}{\longrightarrow}\lambda_p. 
    \end{equation*}
    for $\mu_p$-almost every $z\in\mathbb{C}$. Similarly, the sum $\Lambda_f \coloneqq \lambda_1+\lambda_2$ is defined for $\mu$-almost every $x\in\mathbb{C}^2$ by:
    \begin{equation*}
        \frac1n\sum_{k=0}^{n-1}\log|\det Df_{f^k(x)}| \overset{n\to\infty}{\longrightarrow}\Lambda_f \coloneqq \int\log|\det Df_x|\text{d}\mu(x).
    \end{equation*}
    Since $f$ is a skew product, for every $(z,w)\in\mathbb{C}^2$, its Jacobian matrix is of the form 
    \begin{equation}
        Df_{(z,w)}=\begin{pmatrix}
            p'(z)&0\\\partial_zq(z,w)&\partial_wq(z,w)
        \end{pmatrix}.\label{jacobien}
    \end{equation}
    Thus, 
    \begin{equation*}
        \lambda_1+\lambda_2 = \Lambda_f = \int\log|\det Df_x|\text{d}\mu(x) = \int\log|p'(z)|\text{d}\mu_p(z) + \int\log|q'_z(w)|\text{d}\mu(z,w) = \lambda_p+\lambda_q.
    \end{equation*}
    However, since the Lyapunov exponent of $\mu_p$ is one for $\mu$ (see \cite{Jonsson1999, DupontTaflin2021}), we obtain $\lambda_p\in\{\lambda_1,\lambda_2\}$. Thus $\{\lambda_1,\lambda_2\} = \{\lambda_p, \lambda_q\}.$ We set, for almost every $\hat x\in X$, 
    \begin{equation*}
        H_1(\hat x) = \text{vect}\begin{pmatrix}
        a(\hat x)\\b(\hat x)
        \end{pmatrix} \quad \text{ and }\quad  H_2 = \text{vect}\begin{pmatrix}
         0\\1
         \end{pmatrix}.
    \end{equation*}

    \begin{itemize}
        \item an integer $n_{\hat{x}} \geqslant 1$ and real numbers $h_{\hat{x}} \geqslant 1$ and $0 < r_{\hat{x}}, \rho_{\hat{x}} \leqslant 1$,
        \item a sequence $(\varphi_{\hat{x}_n,k})_{k\geqslant0}$ of injective holomorphic maps:
            \begin{equation*}\begin{array}{ccccc}
                \varphi_{\hat{x}_n,k} &: &B\left(x_{n-k},\, r_{\hat{x}_n} \text{e}^{-k(\delta+2\varepsilon)}\right) &\longrightarrow &\mathbb{D}^2\left(\rho_{\hat{x}_n} \text{e}^{k\varepsilon}\right)   \end{array}

            \begin{equation*}
                \text{e}^{-k\lambda_i+k(\delta-\varepsilon)} \leqslant |a^i_{\hat{x},k}|\leqslant\text{e}^{-k\lambda_i+k(\delta+\varepsilon)},
            \end{equation*}
    \end{itemize}
    for which the following diagram commutes for every $n\geqslant\hat{n}_{\hat{x}}$:

        \begin{CD}
            B(x_n, \,r_{\hat{x}_n}) @>{f_{\hat{x}_n}^{-n}}>>B\left(x_{0},\,r_{\hat{x}_n} \text{e}^{-n(\delta+2\varepsilon)}\right)\\

        \end{CD}
    \end{equation*}

\label{Cepsi}

We follow the construction of $\mathcal{C}_\varepsilon$ in the proof of \cite[Theorem~S.2.10]{KatokHasselblatt1995}. By Theorem \ref{Oseledets}, for every $\hat x \in X$ there exist $a(\hat x)\neq 0$ and $b(\hat x)$ such that the Oseledets spaces at $\hat x$ are given by the span of
    \begin{equation*}
        H_1(\hat x) = \begin{pmatrix} a(\hat x)\\ b(\hat x) \end{pmatrix} \quad\text{and}\quad H_2(\hat x) = \begin{pmatrix} 0\\ 1 \end{pmatrix},
    \end{equation*}
    where $H_1$ is associated to $\lambda_2$ and $H_2$ to $\lambda_1$. We normalize $\lVert H_1(\hat x)\rVert = 1$ (the equality $\lVert H_2(\hat x)\rVert = 1$ is obvious). Since these lines are invariant, there exists $\xi_i(\hat x)\in\mathbb{C}$ with $|\xi_i(\hat x)|=1$ such that 
    \begin{equation}\label{d0g}
        \frac{d_0 g_{\hat{x}}(H_i(\hat{x}))}{\left\| d_0 g_{\hat{x}}(H_i(\hat{x})) \right\|} = \xi_i(\hat{x})\, H_i(\tau(\hat{x})).
    \end{equation}
    Writing $d_0 g^m_{\hat x}\coloneqq d_0 g_{\tau^{m-1}(\hat x)}\circ\cdots\circ d_0g_{\hat x}$, (and the inverse of such compositions when $m$ is negative), we set
    \begin{equation}\label{deftheta}
        \theta_1(\hat{x}) = \sum_{m \in \mathbb{Z}} \left\| d_0 g^{m}_{\hat{x}}(H_1(\hat{x})) \right\|^2 \text{e}^{2m\lambda_2}\, \text{e}^{-2\varepsilon |m|}\quad \text{ and } \qquad \theta_2(\hat{x}) = \sum_{m \in \mathbb{Z}} \left\| d_0 g^{m}_{\hat{x}}(H_2(\hat{x})) \right\|^2 \text{e}^{2m\lambda_1}\, \text{e}^{-2\varepsilon |m|}.
    \end{equation}
    These series converge (see \cite{KatokHasselblatt1995}) and satisfy $\theta_i \geqslant 1$ (sum of positive terms with the term $m=0$ that is equal to $1$). Re-indexing the sum defining $\theta_1(\tau(\hat x))$ and using $\text{e}^{-2\varepsilon}\leqslant \text{e}^{-2\varepsilon(|k-1|-|k|)}\leqslant\text{e}^{2\varepsilon}$, we get
    \begin{equation}\label{theta}
        \theta_1(\tau(\hat{x}))^{1/2}= \theta_1(\hat{x})^{1/2} \left\| d_0 g_{\hat{x}}(H_1(\hat{x})) \right\|^{-1}\theta\text{e}^{-\lambda_2},\qquad \text{e}^{-\varepsilon} \leqslant \theta \leqslant \text{e}^{\varepsilon}.
        \end{equation}
    and similarly for $\theta_2$. Define
    \begin{equation}\label{defC}
        \begin{array}{ccccl}
            \mathcal{C}_{\varepsilon,\hat{x}}^{-1} & :&\mathbb{C}^2&\longrightarrow &\mathbb{C}^2 \\
             &&(z, w) &\longmapsto &\frac{1}{\sqrt{2}}\left(z\, \theta_1(\hat{x})^{-1/2} H_1(\hat{x}) + w\, \theta_2(\hat{x})^{-1/2} H_2(\hat{x})\right).
        \end{array}
    \end{equation}
    Expanding, we obtain the following fibered map
    \begin{equation}\label{defCextented}
        \mathcal{C}_{\varepsilon,\hat{x}}^{-1}(z, w) = \frac{1}{\sqrt{2}}\left( z\, \theta_1(\hat{x})^{-1/2} a(\hat{x}),\; z\, \theta_1(\hat{x})^{-1/2} b(\hat{x}) + w\, \theta_2(\hat{x})^{-1/2} \right).
    \end{equation}
    Since $a(\hat x)\neq0$, the map $\mathcal{C}_{\varepsilon,\hat{x}}$ is well defined as the inverse of $\mathcal{C}_{\varepsilon,\hat{x}}^{-1}$ and is also fibered. Finally, combining \eqref{d0g} and \eqref{theta}, we get
    \begin{align*}
        d_0 g_{\hat{x}}\big(\mathcal{C}_{\varepsilon,\hat{x}}^{-1}(1,0)\big) &= \frac{1}{\sqrt 2}\theta_1^{-1/2}(\hat x)d_0g_{\hat x}(H_1(\hat x)) \\&= \frac{1}{\sqrt 2}\xi_1(\hat x)\|d_0g_{\hat x}(H_1(\hat x))\|\theta^{-1/2}_1(\hat x)H_1(\tau(\hat x))\\&= \xi_1(\hat x)\theta\text{e}^{-\lambda_2}\mathcal{C}_{\varepsilon,\tau(\hat{x})}^{-1}(1,0),
    \end{align*}
    and similarly for $(0,1)$. Thus
    \begin{equation*}
        \mathcal{C}_{\varepsilon,\tau(\hat{x})} \circ d_0 g_{\hat{x}} \circ \mathcal{C}_{\varepsilon,\hat{x}}^{-1} = \operatorname{diag}\big(a_2(\hat{x}), a_1(\hat{x})\big),
    \end{equation*}
    where $\text{e}^{-\lambda_i - \varepsilon} \leqslant |a_i(\hat{x})| \leqslant \text{e}^{-\lambda_i + \varepsilon}$. Moreover, we verify easily that $\lvert\mathcal{C}^{-1}_{\varepsilon,\hat x}(v)\lvert \leqslant \lvert v\vert$ form \eqref{defC}. Resolving a system from \eqref{defCextented} and using that $\lvert a(\hat x)\lvert ,\, \lvert b(\hat x)\lvert \leqslant 1$, we get 
    \begin{equation*}
        \lvert \mathcal{C}_{\varepsilon, \hat x}(v)\lvert \leqslant \lvert v\lvert\, \frac{1}{\lvert a(\hat x)\lvert }\left(\theta_1^{1/2}+2\theta_2^{1/2}\right).
    \end{equation*}
    The functions $\theta_1$, $\theta_2$ are tempered by their definition \eqref{deftheta} and using the Birkhoff theorem. The function $|a|^{-1}$ is also tempered since the angle between the Oseledets spaces is always tempered (see \cite{KatokHasselblatt1995,BarreiraPesinSchmeling1999}). Hence, $h:\hat x\mapsto \frac{1}{\lvert a(\hat x)\lvert }\left(\theta_1^{1/2}+2\theta_2^{1/2}\right)$ is tempered. Thus, there exist an $\varepsilon$-fast function $h_\varepsilon$ such that $h \leqslant h_\varepsilon$ for $\hat\mu$-almost every point $\hat x\in X$ (see \cite{BertelootDupont2019}) and this gives the upper bound of $\mathcal{C}_{\varepsilon, \hat x}$.

    \paragraph{Fact 2.} In Proposition \ref{3.2}, if $(W_n^\delta)_{n\in\mathbb{Z}}$ is assumed to be fibered, then the sequence $(T^1_n)_{n\in\mathbb{Z}}$ satisfying the stated properties can be chosen to be fibered.
    \begin{proof}
        We follow the proof of Proposition 3.2 in \cite{BertelootDupont2019}. Let $(G_n)_{n\in\mathbb{Z}}$ be a chain such that $d_0G_n = A_n^\delta$. Denote by $G_n^{(k)}$ the homogeneous part of degree $k$ of $G_n$. Suppose that there exists a sequence of homogeneous polynomial maps $\left(H_n^{(k)}\right)_{n\in\mathbb{Z}}$ of degree $k$ satisfying the homological equation
        \begin{equation}\label{eqhomol}
            G_n^{(k)} + H_{n+1}^{(k)}\circ A_n^\delta - A_n^\delta\circ H_n^{(k)} = 0.
        \end{equation}
        Then, if we set
        \begin{equation}\label{defsnk}
            S_n^{(k)} \coloneqq \mathrm{Id} + H_n^{(k)},
        \end{equation}
        the sequence $S_{n+1}^{(k)} \circ G_n \circ \left(S_n^{(k)}\right)^{-1}$ has no homogeneous part of order $k$, see \cite{BertelootDupont2019}. Proposition \ref{3.2} is then obtained by successively applying this procedure a finite number of times, starting with $G_n = W_n^{\delta}$ and $k=2$, then with $G_n = S_{n+1}^{(2)} \circ W_n^{\delta} \circ \left(S_n^{(2)}\right)^{-1}$ and $k=3$, and so on, up to $k = [\lambda_1^\delta/\lambda_2^\delta]+1$.

        By linearity, it is enough to solve \eqref{eqhomol} when the sequence $\left(G_n^{(k)}\right)_{n\in\mathbb{Z}}$ has only one nonzero coordinate, which is moreover a monomial. Since $(G_n)_{n\in\mathbb{Z}}$ is fibered, and according to the resonance conditions \eqref{notinb}, there are three cases to consider for each $k\geqslant 2$:
        \begin{enumerate}
            \item $G_n^{(k)}(z,w) = (a_n z^k,\, 0)$;
            \item $G_n^{(k)}(z,w) = (0,\, a_n z^{\alpha_2} w^{\alpha_1})$ where $\alpha_1 \lambda_1^{\delta} + \alpha_2 \lambda_2^{\delta} - \lambda_1^{\delta} > b$;

        \end{enumerate}

        \begin{equation*}
            \Gamma : \left(H_n^{(k)}\right)_{n \in \mathbb{Z}} \in \left(\mathcal{H}_k\right)^{\mathbb{Z}} \mapsto \left(H_{n+1}^{(k)} \circ A_n^\delta - A_n^\delta \circ H_n^{(k)}\right)_{n \in \mathbb{Z}}.
