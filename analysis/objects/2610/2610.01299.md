<!-- theorem-1: body_md -->
\label{thm:main}
For every $s>0$ there exists a constant $C_s>0$ such that
every measurable $E\subset\mathbb R$ with $|E|=s$, every
$f\in PW$ with $\|f\|_2=1$, and every optimal center $a_E$
satisfy
\begin{equation}
\mathcal D(E)+d_{a_E}(f)^2\le C_s\Delta(f,E).
\label{eq:main}
\end{equation}
Moreover, at the same center,
\begin{equation}
\left(\frac{|E\triangle(I_s+a_E)|}{s}\right)^2 + d_{a_E}(f)^2
\le C_s\Delta(f,E),
\label{eq:main-measure}
\end{equation}
after increasing $C_s$ if necessary.

<!-- prop-1: body_md -->
\label{prop:local}
There exists $K_s>0$ such that every normalized $f\in PW$,
every measurable $E$ with $|E|=s$, and every optimal center
$a_E$ satisfy
\begin{equation}
\Delta(f,E)\ge
\mathcal D(E)+\frac{\lambda_0(s)-\lambda_1(s)}{2}d_{a_E}(f)^2
-K_s\mathcal D(E)d_{a_E}(f).
\label{eq:local}
\end{equation}

<!-- proof-1: body_md -->
Translate both $E$ and $f$ so that $a_E=0$, and write
$I=I_s$, $\psi=\psi_s$.
Leaving $\psi$ unchanged, multiply $f$ by a unimodular constant so that
\[
\langle f,\psi\rangle=|\langle f,\psi\rangle|\ge0.
\]
Then $d_0(f)=\|f-\psi\|_2$. 

\medskip\noindent
\textbf{Step 0: decompose the deficit.}
Since $\mathcal D(E) = \lambda_0(s) - \int_E|\psi|^2$ and $\int_I|\psi|^2=\lambda_0(s)$,
\begin{align}
\Delta(f,E)
&= \lambda_0(s)-\int_E|f(x)|^2\,dx \nonumber\\
                                                      
             
&=\mathcal D(E)-\int_I\bigl(|f|^2-|\psi|^2\bigr)\,dx +
\int_{\mathbb R}(\mathbbm 1_I-\mathbbm 1_E)
\bigl(|f|^2-|\psi|^2\bigr)\,dx
\nonumber\\
&=\mathcal D(E)
+\left(\lambda_0(s)-\int_I|f|^2\,dx\right)
+
\int_{\mathbb R}(\mathbbm 1_I-\mathbbm 1_E)
\bigl(|f|^2-|\psi|^2\bigr)\,dx. \label{eq:decomposition}
\end{align}
               
             
                   
                                        
             
                                               
                                 
                          
             
We bound the last two terms separately.

\medskip\noindent
\textbf{Step 1: use the interval spectral gap.}
Write
\[
f=\langle f,\psi\rangle\psi+h,
\qquad h\perp\psi.
\]
The mixed terms vanish because $T_I\psi=\lambda_0(s)\psi$.
The norm of $T_I$ restricted to $\psi^\perp$ is $\lambda_1(s)$, so
\begin{align*}
\int_I|f|^2
&=|\langle f,\psi\rangle|^2\lambda_0(s)
+\langle T_Ih,h\rangle\\
&\le |\langle f,\psi\rangle|^2\lambda_0(s)
+\lambda_1(s)\bigl(1-|\langle f,\psi\rangle|^2\bigr).
\end{align*}
Consequently,
\[
\lambda_0(s)-\int_I|f|^2
\ge(\lambda_0(s)-\lambda_1(s))\bigl(1-|\langle f,\psi\rangle|^2\bigr).
\]
By the choice of phase,
\[
d_0^2(f)=2\bigl(1-|\langle f,\psi\rangle|\bigr).
\]
Using $1-t^2\ge1-t$ for $0\le t\le1$, we obtain
\begin{equation}
\lambda_0(s)-\int_I|f|^2\ge\frac{\lambda_0(s)-\lambda_1(s)}{2}d_0^2(f).
\label{eq:gap}
\end{equation}

\medskip\noindent
\textbf{Step 2: the two cancellations.}
Set
\[
u=\psi^2,\quad v=|f|^2-u.
\]
We use the following properties of $u$:
\begin{equation}
\begin{gathered}
u(s/2)>0,\qquad u'(-s/2)=-u'(s/2)>0,\\
u(x)>u(s/2)\quad (|x|<s/2),\qquad
u(x)<u(s/2)\quad (|x|>s/2).
\end{gathered}
\label{eq:prolate-level}
\end{equation}
These follow from the classical prolate differential equation and
its ground-state solution; see \cite[Section V]{SP}.

In particular, the only zeros of $u-u(s/2)$ are the two simple zeros
$\pm s/2$. Near these points, $|u-u(s/2)|$ is bounded below by a positive
multiple of the distance to the endpoints. Outside fixed neighborhoods
of the endpoints it has a positive lower bound, since it has no other
zeros and $u(x)\to0$ while $u(s/2)>0$. Thus, for some $c_s>0$,
\begin{equation}
|u(x)-u(s/2)|\ge c_s\min\{1,\operatorname{dist}(x,\partial I)\},
\qquad x\in\mathbb R.
\label{eq:weight}
\end{equation}

Equal measures give $\int(\mathbbm 1_I-\mathbbm 1_E)=0$.
Since zero is an optimal center, the function
\[
\Phi(a)=\int_Eu(x-a)\,dx
\]
has a maximum at zero. Boundedness of $u'$ and finiteness
of $|E|$ justify differentiation under the integral, so
$0=\Phi'(0)=-\int_Eu'$.
Also $\int_Iu'=u(s/2)-u(-s/2)=0$, since $u$ is even. Hence
\begin{equation}
\int_{\mathbb R}(\mathbbm 1_I-\mathbbm 1_E)\,dx=0,
\qquad
\int_{\mathbb R}(\mathbbm 1_I-\mathbbm 1_E) u'\,dx=0.
\label{eq:cancellations}
\end{equation}
The signs in \eqref{eq:prolate-level}, together with equal measures,
give, for every prescribed center $a$,
\begin{equation}
D_a(E)=\int_{E\triangle(I_s+a)}|u(x-a)-u(s/2)|\,dx.
\label{eq:weighted}
\end{equation}
Indeed, subtracting $u(s/2)$ from the two integrals
in $D_a(E)$ does not change their difference. This identity also
shows that $D_a(E)=0$ exactly when $E=I_s+a$ up to a null set.
At our optimal center $a_E=0$, it becomes
\begin{equation}
\mathcal D(E)=\int_{\mathbb R}(\mathbbm 1_I-\mathbbm 1_E)(u-u(s/2))\,dx
=\int_{\mathbb R}|\mathbbm 1_I-\mathbbm 1_E|\,|u-u(s/2)|\,dx.
\label{eq:centered-weighted}
\end{equation}
Choose
\[
A=\frac{v(-s/2)+v(s/2)}2,
\qquad
B=\frac{v(-s/2)-v(s/2)}{2u'(-s/2)},
\qquad R=v-A-Bu'.
\]
Since $u'(-s/2)=-u'(s/2)>0$, we have $R(-s/2)=R(s/2)=0$.
The two cancellations give
\[
\int_{\mathbb R}(\mathbbm 1_I-\mathbbm 1_E) v\,dx
=\int_{\mathbb R}(\mathbbm 1_I-\mathbbm 1_E) R\,dx.
\]

\medskip\noindent
\textbf{Step 2.1: bound the remainder.}
As is well known, Fourier inversion and Cauchy--Schwarz give, for $g\in PW$,
\begin{equation}
\|g^{(j)}\|_\infty
\le\frac{\pi^j}{\sqrt{2j+1}}\|g\|_2,
\qquad j=0,1,2,\ldots.
\label{eq:bernstein}
\end{equation}
The identity
\[
v=(f-\psi)\overline f+\psi\,\overline{(f-\psi)}
\]
and \eqref{eq:bernstein} give
\[
\|v\|_\infty
\le\|f-\psi\|_\infty(\|f\|_\infty+\|\psi\|_\infty)
\le2d_0(f).
\]
Differentiating the same identity yields
\begin{align*}
\|v'\|_\infty
&\le\|f'-\psi'\|_\infty(\|f\|_\infty+\|\psi\|_\infty)\\
&\quad+\|f-\psi\|_\infty(\|f'\|_\infty+\|\psi'\|_\infty)
\le\frac{4\pi}{\sqrt3}d_0(f).
\end{align*}
                                                            
                                                          
Similarly, we can write $u'=2\psi\psi'$ and $u''=2(\psi')^2+2\psi\psi''$, so
\eqref{eq:bernstein} gives
\[
\|u'\|_\infty\le2\|\psi\|_\infty\|\psi'\|_\infty\le\frac{2\pi}{\sqrt3},
\qquad
\|u''\|_\infty\le2\|\psi'\|_\infty^2+2\|\psi\|_\infty\|\psi''\|_\infty
\le2\pi^2\Bigl(\frac13+\frac1{\sqrt5}\Bigr)\le2\pi^2 .
\]
Since $u'(-s/2)>0$ depends only on $s$,
\[
|A|+|B|\le C_s d_0(f),
\qquad
\|R\|_\infty+\|R'\|_\infty\le C_s d_0(f).
\]
Since $R(-s/2)=R(s/2)=0$, the fundamental theorem of calculus,
together with the sup-norm bound above, gives
\[
|R(x)|\le C_s d_0(f) \min\{1,\operatorname{dist}(x,\partial I)\}
\le C_s d_0(f) \,|u(x)-u(s/2)|,
\]
where the last inequality is \eqref{eq:weight}.
This also bounds $R/(u-u(s/2))$, with continuous extension
at the endpoints.

Finally,
\begin{align*}
\left|\int_{\mathbb R}(\mathbbm 1_I-\mathbbm 1_E) v\,dx\right|
&=\left|\int_{\mathbb R}(\mathbbm 1_I-\mathbbm 1_E) R\,dx\right|\\
&\le C_s d_0(f)\int_{\mathbb R}|(\mathbbm 1_I-\mathbbm 1_E)|\,|u-u(s/2)|\,dx\\
&=C_s\mathcal D(E)d_0(f).
\end{align*}
Fix a constant $K_s>0$ for this estimate.
Combining it with \eqref{eq:decomposition} and \eqref{eq:gap}
proves \eqref{eq:local}.

<!-- lemma-1: body_md -->
\label{lem:compactness}
If $|E_n|=s$ and $\eta(E_n)\to0$, then
$\mathcal D(E_n)\to0$.
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
Suppose otherwise, and pass to a subsequence on which
$\mathcal D(E_n)\ge\varepsilon_0>0$.
Choose normalized top eigenfunctions $f_n$:
\[
T_{E_n}f_n=\lambda_{E_n}f_n,
\qquad \lambda_{E_n}\longrightarrow\lambda_0(s).
\]
For large $n$, $\int_{E_n}|f_n|^2\ge\lambda_0(s)/2$.
Since $|E_n|=s$, we can translate $E_n$ and $f_n$, and choose
a phase, so that
\[
f_n(0)\ge\sqrt{\lambda_0(s)/(4s)}.
\]
These operations do not change $\eta(E_n)$ or $\mathcal D(E_n)$.

After passing to a further subsequence,
$f_n\rightharpoonup f$ in $PW$.
Point evaluation is continuous, so
$f\ne0$. The same estimates give local uniform convergence:
weak convergence gives convergence at every point, and the
derivative bound gives equicontinuity.

We claim that
\begin{equation}
\int_{E_n}(f_n-f)\overline f\,dx\longrightarrow0.
\label{eq:cross-limit}
\end{equation}
On a fixed bounded interval this follows from local uniform
convergence. On its complement, Cauchy--Schwarz gives the bound
\[
\|f_n-f\|_2\,
\|f\mathbbm 1_{\{|x|>R\}}\|_2,
\]
which is uniformly small for large $R$.

Testing the eigenvalue equation against the fixed function $f$
now gives
\[
\int_{E_n}|f|^2\,dx
=\lambda_{E_n}\langle f_n,f\rangle+o(1)
\longrightarrow\lambda_0(s)\|f\|_2^2.
\]
In particular,
\begin{equation}
\lambda_0(s)\|f\|_2^2-\int_{E_n}|f|^2\,dx
\longrightarrow0.
\label{eq:profile}
\end{equation}

Let $A$ be a superlevel set of $|f|^2$ with $|A|=s$. Let $t>0$ be such that $A=\{x:|f(x)|^2>t\}$.
For any $B$ of measure $s$, equal measures give
\[
\int_A|f|^2-\int_B|f|^2
=\int_{A\setminus B}(|f|^2-t)
+\int_{B\setminus A}(t-|f|^2)\ge0.
\]
Therefore \eqref{eq:profile} and \eqref{eq:interval} imply
\[
\lambda_0(s)\|f\|_2^2
=\lim_n\int_{E_n}|f|^2
\le\int_A|f|^2
\le\lambda_0(s)\|f\|_2^2.
\]
The equality characterization \eqref{eq:equality} gives
$f=c\psi_s(\cdot-b)$ for some $c\ne0$ and $b\in\mathbb R$.
Testing the definition of $\mathcal D(E_n)$ at this center,
\[
0\le\mathcal D(E_n)
\le
\frac{\lambda_0(s)\|f\|_2^2-\int_{E_n}|f|^2}{\|f\|_2^2}
\longrightarrow0,
\]
contradicting $\mathcal D(E_n)\ge\varepsilon_0$.

<!-- proof-3: body_md -->
We first control the set discrepancy by the spectral deficit,
and then return to an arbitrary normalized function.

\medskip\noindent
\textbf{Step 1: compare the two deficits.}
Fix $E$, an optimal center $a_E$, and a normalized top
eigenfunction $f_E$.
Testing the Rayleigh quotient with $\psi_s(\cdot-a_E)$ gives
$\eta(E)\le \mathcal D(E)$. Proposition~\ref{prop:local}, applied to $f_E$,
and completion of the square yield
\begin{equation}
\begin{aligned}
0\le \mathcal D(E)-\eta(E)
&\le K_s\mathcal D(E)d_{a_E}(f_E)-\frac{\lambda_0(s)-\lambda_1(s)}{2}d_{a_E}^2(f_E)\\
&\le A_s\mathcal D(E)^2,
\end{aligned}
\label{eq:spectral-quadratic}
\end{equation}
where 
\begin{equation*}
    A_s=\frac{K_s^2}{2(\lambda_0(s)-\lambda_1(s))}>0.
\end{equation*}

\medskip\noindent
\textbf{Step 2: make the spectral comparison global.}
We claim that there is $S_s>0$ such that
\begin{equation}
\mathcal D(E)\le S_s\eta(E)
\qquad\text{for every }|E|=s.
\label{eq:global-spectral}
\end{equation}
Suppose not. There would be sets $E_n$ with
\[
\mathcal D(E_n)>n\eta(E_n).
\]
Since $\mathcal D(E_n)\le\lambda_0(s)$, we have
$\eta(E_n)\to0$. Lemma~\ref{lem:compactness} gives
$\mathcal D(E_n)\to0$. For large $n$,
$A_s\mathcal D(E_n)\le1/2$, and
\eqref{eq:spectral-quadratic} then implies
\[
\mathcal D(E_n)
\le\eta(E_n)+A_s\mathcal D(E_n)^2
\le\eta(E_n)+\tfrac12\mathcal D(E_n).
\]
Hence $\mathcal D(E_n)\le2\eta(E_n)$, a contradiction.
This proves \eqref{eq:global-spectral}.

\medskip\noindent
\textbf{Step 3: return to arbitrary functions.}
Let $\|f\|_2=1$.
Since $\eta(E)\le\Delta(f,E)$, \eqref{eq:global-spectral} gives
\[
\mathcal D(E)\le S_s\Delta(f,E).
\]
Moreover, Proposition~\ref{prop:local} implies
\[
\frac{\lambda_0(s)-\lambda_1(s)}{2}d_{a_E}^2(f)
\le\Delta(f,E)-\mathcal D(E)+K_s\mathcal D(E)d_{a_E}(f).
\]
Phase minimization gives
\[
d_{a_E}^2(f)=2\bigl(1-|\langle f,\psi_s(\cdot-a_E)\rangle|\bigr)\le2.
\]
Consequently,
\[
\frac{\lambda_0(s)-\lambda_1(s)}{2}d_{a_E}^2(f)
\le\Delta(f,E)+\sqrt2K_s\mathcal D(E)
\le(1+\sqrt2K_sS_s)\Delta(f,E).
\]
Together with the set estimate, this proves \eqref{eq:main}.

\medskip\noindent
\textbf{Step 4: compare with symmetric difference.}
For any prescribed center $a$, we show that
\begin{equation}
|E\triangle(I_s+a)|^2\le C_sD_a(E).
\label{eq:asymmetry}
\end{equation}
Translate to $a=0$ and put $\alpha=|E\triangle I_s|\le2s$.
The case $\alpha=0$ is immediate. Otherwise choose
\[
t=\frac{\alpha}{8(1+s)}<1.
\]
At most $4t\le\alpha/2$ of the symmetric difference lies
within distance $t$ of the two endpoints. Thus at least
$\alpha/2$ lies at distance at least $t$.
By \eqref{eq:weight} and \eqref{eq:weighted},
\[
D_0(E)\ge c_s t\,\frac\alpha2
=\frac{c_s}{16(1+s)}\alpha^2.
\]
This proves \eqref{eq:asymmetry}. Applying it at $a_E$ and
using \eqref{eq:main} proves \eqref{eq:main-measure}.
