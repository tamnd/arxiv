\label{thm1.1}
Suppose that $(\omega_{in},\theta_{in})$ is the initial data for \eqref{1.3} with $D=\mathbb{R}$ or $[0,\infty)$.  For all $m>\frac12$, there exists a constant $\varepsilon_0>0$ independent of $\nu$ and $\mu$ such that if
\begin{equation}\label{1.4}
	\|u_{0,in}^1\|+\left\|y{\partial}_y\omega_{0,in}\right\|+\sum_{j=0}^1\left\|(\nu^{\frac14}{\partial}_y)^j\langle{\partial}_x\rangle^{m-\frac{j}{4}}\omega_{in}\right\|+\nu^{-\frac14}\left\||{\partial}_x|^{m+\frac{1}{4}}(y\omega_{in},\nabla \phi_{in})\right\|\leq \varepsilon_0\min\{\nu^{\frac34},\mu^{\frac34},\nu^{-\frac14}\mu\},
\end{equation}
\begin{equation}\label{1.5}
	\left\|y{\partial}_y\theta_{0,in}\right\|+\sum_{j=0}^1\left\|(\mu^{\frac14}{\partial}_y)^j\langle{\partial}_x\rangle^{m-\frac{j}{4}}\theta_{in}\right\|+\mu^{-\frac14}\left\||{\partial}_x|^{m+\frac{1}{4}}y\theta_{in}\right\|\leq \varepsilon_0\min\{\nu^{\frac94},\mu^{\frac94}\},
\end{equation}
then for $0<\delta_0\ll 1$ (depending only on universal constants), $t\geq 0$ and $0<\nu,\mu\leq 1$,  the corresponding solution $(\omega,\theta)$ to \eqref{1.3} satisfies
\begin{equation}\label{1.6}
\sum_{j=0}^1\left\|(\nu^{\frac14}{\partial}_y)^j\langle{\partial}_x\rangle^{m-\frac{j}{4}}\omega_{\neq}\right\|+\nu^{-\frac14}\left\||{\partial}_x|^{m+\frac{1}{4}}(y\omega_{\neq},\nabla \phi_{\neq})\right\|\lesssim \varepsilon_0{\rm e}^{-\delta_0\min\{\nu,\mu\}^{\frac12} t},	
\end{equation}
\begin{equation}\label{1.7}
	\sum_{j=0}^1\left\|(\mu^{\frac14}{\partial}_y)^j\langle{\partial}_x\rangle^{m-\frac{j}{4}}\theta_{\neq}\right\|+\mu^{-\frac14}\left\||{\partial}_x|^{m+\frac{1}{4}}y\theta_{\neq}\right\|\lesssim \varepsilon_0{\rm e}^{-\delta_0\min\{\nu,\mu\}^{\frac12} t}.
\end{equation}

<!-- rem-1: body_md -->
According to the results about Navier-Stokes equations \cite{Coti2020,Del2023}, the authors used energy theory and semigroup estimation to get the transition threshold. 
  Because of the coupling effect of this system, it seems difficult to use semigroup estimation like Navier-Stokes equations.
  In this paper, we use a new energy estimate to study the enhanced dissipation of NSB equations, which will provide some reference for studying the stability of Poiseuille flow for other coupled models.
<!-- end rem-1 -->
<!-- rem-2: body_md -->
Our main results, the Theorems \ref{thm1.1} and \ref{thm1.2}, give the enhanced dissipation with decay rate ${\rm e}^{-\delta_0\min\{\nu,\mu\}^{\frac12} t}$,  which is consistent with the Navier-Stokes equations \cite{Del2023,Ding2025}. However, the transition threshold may not be sharp.  It is very interesting and challenging to study the optimal transition threshold of
 the Poiseuille flow in the future.

&=\mu\langle y\theta_k, y \Delta_k \theta_k\rangle -\mathfrak{R}\langle y\theta_k, y(u\cdot \nabla \theta)_k)\rangle \notag\\
&=\mu\|\theta_k\|^2-\mu\|y\nabla_k\theta_k\|^2-\mathfrak{R}\langle y\theta_k, y(u\cdot \nabla \theta)_k)\rangle,\notag
 \end{align}
hence we have \eqref{4.5}.
The proof of Lemma \ref{l4.1} is over.

Recalling  the definition of  $E_{\theta}[\theta_k]$ in \eqref{2.7}, and combining with Lemma \ref{l4.1}, we can deduce
\begin{align}\label{4.6}
&{\frac{{\rm d}}{{\rm d} t}} E_{\theta}[\theta_k] +\mu\|\nabla_k\theta_k\|^2+2^{-7}\mu^{\frac32}|k|^{-\frac12}\|{\partial}_y\nabla_k\theta_k\|^2+2^{-10}|k|\| y\theta_k\|^2+2^{-9}\mu^{\frac12}|k|^{\frac12}\|y\nabla_k\theta_k\|^2\notag\\
=&2^{-9}\mu^{\frac12}|k|^{\frac12}\|\theta_k\|^2-2^{-6}\mu^{\frac12}|k|^{-\frac12}\mathfrak{R}\langle iky\theta_k,{\partial}_y\theta_k\rangle-2^{-10}\mu|k|^{-1}\mathfrak{R}\langle \Delta_k \theta_k, iky{\partial}_{y}\theta_k\rangle\notag\\ 
&-\mathfrak{R}\langle\theta_k,(u\cdot \nabla \theta)_k\rangle-2^{-7}\mu^{\frac12}|k|^{-\frac12}\mathfrak{R}\langle{\partial}_y\theta_k,{\partial}_y(u\cdot \nabla \theta)_k\rangle +2^{-10}|k|^{-1}\mathfrak{R}\langle iky{\partial}_{y}\theta_k,(u\cdot \nabla \theta)_k\rangle\notag\\
&+2^{-11}|k|^{-1}\mathfrak{R}\langle ik\theta_k, (u\cdot \nabla \theta)_k\rangle-2^{-9}\mu^{-\frac12}|k|^{\frac12}\mathfrak{R}\langle y^2\theta_k,(u\cdot \nabla \theta)_k\rangle.
\end{align}
We first analyze the linear term on the right side of \eqref{4.6}.
In order to obtain the $L^2$ norm of $\theta$ on the left hand side of \eqref{4.6}, an integration by parts yields
\begin{equation}\notag
 \langle y \theta_k, {\partial}_y\theta_k\rangle= -\frac12  \|\theta_k\|^2.
\end{equation}
It means that for every $\sigma>0$,
\begin{equation}\notag
\left\lVert\theta_k\right\rVert^2\leq 2 \left\lVert{\partial}_y\theta_k\right\rVert\left\lVerty \theta\right\rVert\leq \sigma \left\lVert{\partial}_y\theta_k\right\rVert^2 +\frac{1}{\sigma}\left\lVerty \theta_k\right\rVert^2.
\end{equation}
By taking $\sigma=2^{-\frac{13}{2}}\mu^{\frac12}|k|^{-\frac12}$, we conclude that
\begin{equation}\label{4.7}
 2^{-\frac{13}{2}}\mu^{\frac12}|k|^{\frac12}\left\lVert\theta_k\right\rVert^2\leq \frac{\mu}{4}\left\lVert{\partial}_y\theta_k\right\rVert^2+2^{-11}|k|\left\lVerty\theta_k\right\rVert^2.
\end{equation}
Then, through direct calculation, the second and third terms on the right side of \eqref{4.6} can be estimated as
\begin{align}\label{4.8}
-2^{-6}\mu^{\frac12}|k|^{-\frac12}\mathfrak{R}\langle iky\theta_k,{\partial}_y\theta_k\rangle\leq \frac{\mu}{4} \|\nabla_k \theta_k\|^2+ 2^{-12} |k|\|y\theta_k\|^2,
\end{align}
\begin{align}\label{4.9}
-2^{-10}\mu|k|^{-1}\mathfrak{R}\langle \Delta_k \theta_k, iky{\partial}_{y}\theta_k\rangle=&2\bar\beta\mu\mathfrak{R}\langle y|k|^2 \theta_k, ik{\partial}_{y}\theta_k\rangle-2\bar\beta\mu\mathfrak{R}\langle {\partial}_{yy} \theta_k, iky{\partial}_{y}\theta_k\rangle\notag\\ 
\leq& -2^{-9}\mu\|{\partial}_{y} \nabla_k\theta_k\|\|y\nabla_k\theta_k\|  \notag\\
\leq&2^{-9}\mu^{\frac32}|k|^{-\frac12}\|{\partial}_{y} \nabla_k\theta_k\|^2  +2^{-11}\mu^{\frac12}|k|^{\frac12}\|y\nabla_k\theta_k\|^2.
\end{align}
Substituting \eqref{4.7}--\eqref{4.9} into \eqref{4.6}, and multiplying both sides of the resulting equation by ${\rm e}^{2\delta_0 \kappa^{\frac12}t}$, after performing some integration by parts, we obtain
\begin{align}\label{4.10}

\leq&-{\rm e}^{2\delta_0 \kappa^{\frac12}t}\sum_{k \neq 0}|k|^{2m}\mathfrak{R}\langle\theta_k,(u\cdot \nabla \theta)_k\rangle-2^{-7}\mu^{\frac12}{\rm e}^{2\delta_0 \kappa^{\frac12}t}\sum_{k \neq 0}|k|^{2m-\frac12}\mathfrak{R}\langle{\partial}_y\theta_k,{\partial}_y(u\cdot \nabla \theta)_k\rangle \notag\\ 
&+2^{-10}{\rm e}^{2\delta_0 \kappa^{\frac12}t}\sum_{k \neq 0}|k|^{2m-1}\mathfrak{R}\langle iky{\partial}_{y}\theta_k,(u\cdot \nabla \theta)_k\rangle+2^{-11}{\rm e}^{2\delta_0 \kappa^{\frac12}t}\sum_{k \neq 0}|k|^{2m-1}\mathfrak{R}\langle ik\theta_k, (u\cdot \nabla \theta)_k\rangle\notag\\
&-2^{-9}\mu^{-\frac12}{\rm e}^{2\delta_0 \kappa^{\frac12}t}\sum_{k \neq 0}|k|^{2m+\frac12}\mathfrak{R}\langle y^2\theta_k,(u\cdot \nabla \theta)_k\rangle:=\sum_{i=1}^5I_i.
\end{align}

{\it \underline{Estimate of $I_1$:}}
We can rewrite it as follows
\begin{align}\label{I1}
I_1=&{\rm e}^{2\delta_0 \kappa^{\frac12}t}\sum_{k \neq 0}|k|^{2m}\mathfrak{R}\langle \theta_k, \sum_{l \in \mathbb{Z}} \partial_y \phi_{k-l} \cdot il  \theta_{l}- \sum_{l \in \mathbb{Z}} i(k-l)\phi_{k-l} \cdot \partial_{y} \theta_{l}  \rangle\notag\\

 :=& I_1^1+I_1^2+I_1^3.

For the term $ I_1^1$, one can directly obtain
		\begin{align}\label{i11}
			 I_1^1&={\rm e}^{2\delta_0 \kappa^{\frac12}t}\sum_{k \neq 0}|k|^{2m} \mathfrak{R} \int_{\mathbb{R}}  \theta_k\cdot \overline{ ik\theta_k \partial_{y} \phi_{0}} \mathrm{d} y\notag\\

		\end{align}
		For $I_1^2$, applying H\"older inequality and definitions \eqref{2.2}--\eqref{2.13} yields

			I_1^2&\lesssim{\rm e}^{2\delta_0 \kappa^{\frac12}t}\sum_{k \neq 0}(|k|^{m} \|\theta_k\|) (|k|^{m+1}\|\phi_k\|_{L^\infty}) \| \partial_{y} \theta_0\|\notag\\

			& \lesssim \mu^{-\frac{1}{2}} \mathcal{E}^{\frac{1}{2}}_{\theta,\neq} \mathcal{D}^{\frac{1}{2}}_{\omega, \neq}\mathcal{D}^{\frac{1}{2}}_{\theta, 0} ,
		\end{align}

		{\rm e}^{2\delta_0 \kappa^{\frac12}t}\sum_{k\neq 0} |k|^{2m+2} \|\phi_{k} \|^2_{L^\infty} \lesssim& {\rm e}^{2\delta_0 \kappa^{\frac12}t}\sum_{k\neq 0}   (|k|^{m+\frac32} \|\phi_k\|) (|k|^{m+\frac12} \|\partial_{y}\phi_k\|) \notag\\ 
		\lesssim&{\rm e}^{2\delta_0 \kappa^{\frac12}t}\left(\sum_{k\neq 0}|k|^{2m+3} \|\phi_k\|^2\right)^{\frac12} \left(\sum_{k\neq 0}(|k|^{2m+1} \|\partial_{y}\phi_k\|^2\right)^{\frac12}\notag\\ 
		\lesssim& \mathcal{D}_{\omega, \neq}.

For $ I_1^3$, we need to decompose the frequency $k$. First, denote 
\begin{align}\label{A1}
	&A_1(k,l):=\left\{  (k,l) \in \mathbb{Z}\setminus\{0\}\ \text{and} \ k \neq l~\big| |k-l|\leq \frac{|l|}{2}\   \right\},\notag\\

We define the following two characteristic functions:
 		\begin{align*}
 			\chi_1(k,l)= \begin{cases}1, & \text { if } (k,l) \in A_1 \\ 0, & \text { if } (k,l) \in A_2\end{cases} \quad \text{and} \quad

 Notice that for $(k,l) \in A_1$, we have $|k|\leq |k-l|+|l|\leq \frac32 |l|$ and $|l|\leq |k-l|+|k|\leq \frac{|l|}{2}+|k|$, that is $|k|\approx|l|$; and for $(k,l) \in A_2$, we have $|l|< 2|k-l|$ and $|k|\leq |k-l|+|l|< 3|k-l|$.

	:=&I_1^{3,1}+I_1^{3,2}.
\end{align}
We use the two characteristic functions $\chi_1$ and $\chi_2$ defined in \eqref{A1} to
 decompose $I_1^{3,1}$ into two parts,  and then apply \eqref{psi} and $m>\frac12$ to get
\begin{align}\label{i131}
	I_1^{3,1} &= {\rm e}^{2\delta_0 \kappa^{\frac12}t}\sum_{\substack{ k,l \neq 0 \\ k \neq l }} |k|^{2m} \mathfrak{R}\langle \partial_y\theta_k,   i(k-l)\phi_{k-l} \cdot  \theta_l \rangle  \notag\\

			& \quad + {\rm e}^{2\delta_0 \kappa^{\frac12}t}\Big( \sum_{k\neq 0} |k|^{2m} \|\partial_{y} \theta_k \|^2  \Big)^{\frac{1}{2}}\Big( \sum_{k\neq 0} |k|^{2m+2} \|\phi_{k} \|^2_{L^\infty}  \Big)^{\frac{1}{2}} \Big( \sum_{k\neq 0} \|\theta_{k} \|  \Big)\notag\\
&\lesssim {\rm e}^{2\delta_0 \kappa^{\frac12}t}\Big( \sum_{k\neq 0} |k|^{2m} \|\partial_{y} \theta_k \|^2  \Big)^{\frac{1}{2}}\Big( \sum_{k\neq 0} |k|^{2m} \|\theta_{k} \|^2 \Big)^{\frac{1}{2}} \Big( \sum_{k\neq 0} |k|^{2m+2} \|\phi_{k} \|^2_{L^\infty}  \Big)^{\frac{1}{2}}  \Big( \sum_{k\neq 0} |k|^{-2m}   \Big)^{\frac{1}{2}}\notag\\
			& \quad + {\rm e}^{2\delta_0 \kappa^{\frac12}t}\Big( \sum_{k\neq 0} |k|^{2m} \|\partial_{y} \theta_k \|^2  \Big)^{\frac{1}{2}}\Big( \sum_{k\neq 0} |k|^{2m+2} \|\phi_{k} \|^2_{L^\infty}  \Big)^{\frac{1}{2}} \Big( \sum_{k\neq 0} |k|^{2m}\|\theta_{k} \|^2 \Big)^{\frac{1}{2}}\Big( \sum_{k\neq 0} |k|^{-2m}   \Big)^{\frac{1}{2}}\notag\\
			& \lesssim  \mu^{-\frac{1}{2}}\mathcal{E}_{\theta,\neq}^{\frac{1}{2}} \mathcal{D}_{\theta, \neq}^{\frac{1}{2}} \mathcal{D}_{\omega, \neq}^{\frac{1}{2}}.
 		\end{align}
Similar to $I_1^{3,1}$, we also decompose $I_1^{3,2}$ into two parts, namely
 \begin{align}\label{i132}
	I_1^{3,2} &\lesssim {\rm e}^{2\delta_0 \kappa^{\frac12}t}\sum_{\substack{ k,l \neq 0 \\ k \neq l }}|k|^{2m+1} \| \theta_{k}\|_{L^2} \| \partial_{y} \phi_{k-l}\|_{L^\infty} \|\theta_{l} \|_{L^2}\notag\\
			&\lesssim {\rm e}^{2\delta_0 \kappa^{\frac12}t}\sum_{\substack{ k,l \neq 0 \\ k \neq l }}|k|^{2m+1} \Big( \chi_1(k,l) + \chi_2(k,l) \Big)|k|^{2m+1} \| \theta_{k}\|_{L^2} \| \partial_{y} \phi_{k-l}\|_{L^\infty} \|\theta_{l} \|_{L^2}\notag\\
			&:= I_1^{3,2,\chi_1} + I_1^{3,2,\chi_2}.
		\end{align}
		If $ (k,l)\in A_1$, then $ |k|\approx |l|$. By direct calculation, we get
		\begin{align}\label{i1321}
		I_1^{3,2,\chi_1} &\lesssim {\rm e}^{2\delta_0 \kappa^{\frac12}t}\sum_{\substack{ k,l \neq 0 \\ k \neq l }}(|k|^{m+\frac{1}{4}} \|\theta_{k} \|) (|l|^{m+\frac{3}{4}} \|\theta_{l} \|) \| \partial_{y} \phi_{k-l}\|_{L^\infty} \notag\\
			&\lesssim \mu^{-\frac{1}{4}}{\rm e}^{\delta_0 \kappa^{\frac12}t}\mathcal{D}_{\theta,\neq}^{\frac{1}{2}}\Big( \sum_{k\neq 0} |k|^{2m+\frac{3}{2}} \| \theta_{k}\|^2\Big)^{\frac{1}{2}} \sum_{k\neq 0}\| \partial_{y} \phi_{k}\|_{L^\infty}\notag\\
	&\lesssim \mu^{-\frac{1}{4}}{\rm e}^{\delta_0 \kappa^{\frac12}t}\mathcal{D}_{\theta,\neq}^{\frac{1}{2}}\Big( \sum_{k\neq 0} (|k|^{2m+\frac{1}{2}} \| \theta_{k}\|^2)^{\frac13}(|k|^{2m+2} \| \theta_{k}\|^2)^{\frac23}\Big)^{\frac{1}{2}} \Big( \sum_{k\neq 0} |k|^{2m}\| \partial_{y} \phi_{k}\|_{L^\infty}^2\Big)^{\frac{1}{2}}\Big( \sum_{k\neq 0} |k|^{-2m}\Big)^{\frac{1}{2}}\notag\\

			If $ (k,l)\in A_2$, then $ |k|\lesssim |k-l|+ |l|\lesssim |k-l|$. We can estimate

Combining \eqref{I1}--\eqref{i12} and \eqref{i13}--\eqref{i1322}, we arrive at

	I_1\lesssim\mu^{-\frac{1}{2}} \mathcal{E}^{\frac{1}{2}}_{\theta,\neq} \mathcal{D}^{\frac{1}{2}}_{\omega, \neq}\mathcal{D}^{\frac{1}{2}}_{\theta, 0}+(\mu^{-\frac{1}{2}}+\nu^{-\frac{1}{4}} \mu^{-\frac{1}{4}})\mathcal{E}_{\theta,\neq}^{\frac{1}{2}}  \mathcal{D}_{\omega, \neq}^{\frac{1}{2}}\mathcal{D}_{\theta, \neq}^{\frac{1}{2}}+\mu^{-\frac{2}{3}}\mathcal{E}^{\frac{1}{2}}_{\omega,\neq} \mathcal{D}_{\theta,\neq}.

{\it \underline{Estimate of $I_2$:}} Notice that
\begin{align}\label{I2}
	I_2&= 2^{-7}\mu^{\frac12}{\rm e}^{2\delta_0 \kappa^{\frac12}t}\sum_{k \neq 0}|k|^{2m-\frac12}\mathfrak{R}\langle \partial_{y} \theta_k, \partial_{y}\Big(\sum_{l \in \mathbb{Z}} \partial_{y} \phi_{k-l} \cdot il \theta_l\Big) \rangle \notag\\

			&\quad -2^{-7}\mu^{\frac12}{\rm e}^{2\delta_0 \kappa^{\frac12}t}\sum_{\substack{ k,l \neq 0 \\ k \neq l }}|k|^{2m-\frac12}\mathfrak{R}\langle \partial_{y} \theta_k,  \partial_{y} \big(\nabla^\bot_{k-l} \phi_{k-l} \cdot \nabla_l \theta_l\big) \rangle   \notag\\
			& := I_2^1+I_2^2+I_2^3.

For $k=l$, using H\"older inequality and definitions \eqref{2.2}--\eqref{2.13} gives
\begin{align}\label{i21}
I_2^1=&2^{-7}\mu^{\frac12}{\rm e}^{2\delta_0 \kappa^{\frac12}t}\sum_{k \neq 0}|k|^{2m-\frac12}\mathfrak{R}\langle \partial_{y} \theta_k, \partial_{yy} \phi_{0} \cdot ik \theta_k+\partial_{y} \phi_{0} \cdot ik \partial_{y}\theta_k\rangle\notag\\ 
=&2^{-7}\mu^{\frac12}{\rm e}^{2\delta_0 \kappa^{\frac12}t}\sum_{k \neq 0}|k|^{2m-\frac12}\mathfrak{R}\langle \partial_{y} \theta_k, \partial_{yy} \phi_{0} \cdot ik \theta_k\rangle\notag\\ 
\lesssim&\mu^{\frac12}{\rm e}^{2\delta_0 \kappa^{\frac12}t}\sum_{k \neq 0}(|k|^{m}\|\partial_{y} \theta_k\|)(|k|^{m+\frac12}\|\theta_k\|)\|\partial_{yy} \phi_{0}\|_{L^\infty}\notag\\
\lesssim&{\rm e}^{\delta_0 \kappa^{\frac12}t}\mathcal{D}_{\theta,\neq}^{\frac12}\left(\sum_{k \neq 0}(|k|^{2m+\frac12}\|\theta_k\|^2)^{\frac23}(|k|^{2m+2}\|\theta_k\|^2)^{\frac13}\right)^{\frac12}\|\omega_{0}\|^{\frac12}\|{\partial}_y\omega_{0}\|^{\frac12}\notag\\
\lesssim&\nu^{-\frac18}\mu^{-\frac13}\mathcal{E}_{\omega,0}^{\frac12}\mathcal{D}_{\theta,\neq}.

For $l=0$, by \eqref{psi}, one has
\begin{align}\label{i22}
I_2^2=&2^{-7}\mu^{\frac12}{\rm e}^{2\delta_0 \kappa^{\frac12}t}\sum_{k \neq 0}|k|^{2m+\frac12}\mathfrak{R} \langle \partial_{yy} \theta_k, i\phi_{k} \cdot  \partial_{y} \theta_0\rangle   \notag\\
\lesssim&\mu^{\frac12}{\rm e}^{2\delta_0 \kappa^{\frac12}t}\sum_{k \neq 0}(|k|^{m-\frac{1}{4}} \| \partial_{yy} \theta_k\|)(|k|^{m+\frac{3}{4}} \| \phi_{k}\|_{L^\infty}) \| \partial_{y} \theta_0\| \notag\\
			\lesssim&\mu^{-\frac12}{\rm e}^{\delta_0 \kappa^{\frac12}t}\mathcal{E}_{\omega,0}^{\frac12}\mathcal{D}_{\theta,\neq}^{\frac12}\left(\sum_{k \neq 0}|k|^{2m+\frac{3}{2}} \| \phi_{k}\|_{L^\infty}^2\right)^{\frac{1}{2}}\lesssim\mu^{-\frac12}\mathcal{E}_{\theta,0}^{\frac12}\mathcal{D}_{\theta,\neq}^{\frac12}\mathcal{D}_{\omega,\neq}^{\frac12}.

Regarding $I_2^3$, it is noted that
\begin{align}\label{I23}
