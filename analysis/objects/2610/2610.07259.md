<!-- theorem-1: body_md -->
\label{main}
Assume that $N\geqslant1$ and
\begin{equation}\label{q}
1+\frac1N<q
\quad\text{if }N\in\{1,2\},
\qquad
1+\frac1N<q<\frac{N}{N-2}
\quad\text{if }N\geqslant3.
\end{equation}
Let
\begin{equation}\label{u0}
u_0(x)
=
\frac{C(x)}{|x|^{\frac1{q-1}}},
\qquad
C\in L^\infty(\mathbb{R}^N),
\qquad
C(\lambda x)=C(x)
\end{equation}
for every $\lambda>0$ and a.e. $x\in\mathbb{R}^N$.
Then the Cauchy problem
\begin{equation}\label{eq:main}
\begin{aligned}
u_t-\Delta u+\partial_{x_1}\bigl(|u|^{q-1}u\bigr)&=0,
&&x\in\mathbb{R}^N,\quad t>0,\\
u(x,0)&=u_0(x),
&&x\in\mathbb{R}^N,
\end{aligned}
\end{equation}
has a self-similar solution
\begin{equation}\label{u:self:main}
u(x,t)
=
t^{-\frac1{2(q-1)}}
U\left(\frac{x}{\sqrt t}\right),
\end{equation}
where the profile $U=U(x)$ satisfies, in a weak sense,  the elliptic equation
\begin{equation}\label{ell_U_q}
-\Delta U-\frac12x\cdot\nabla U
-\frac{1}{2(q-1)}U
+\partial_{x_1}\bigl(|U|^{q-1}U\bigr)=0
\qquad\text{in }\mathbb{R}^N.
\end{equation}
Moreover, writing
\(
U_0=e^\Delta u_0,
\) 
the following properties hold:
\begin{itemize}
\item
if
\(
q>1+\frac2N,
\)
then
\(
U-U_0\in W^{1,2}(\mathbb{R}^N);
\)

\item
if
\(
1+\frac1N<q\leqslant1+\frac2N,
\)
then
\(
U-U_0\in W^{1,p}(\mathbb{R}^N)
\)
for every
\[
\max\left\{
1,\frac{N(q-1)}q
\right\}
<p<N(q-1).
\]
\end{itemize}
In addition,
\(U\in L^p(\mathbb{R}^N)\) for every
\(p\in\bigl(N(q-1),\infty\bigr]\),
and the initial condition is attained in the sense that
\begin{equation}\label{initial:trace:main}
u(\cdot,t)\longrightarrow u_0
\qquad\text{in }L^r_{\rm loc}(\mathbb{R}^N)
\quad\text{as }t\to0
\end{equation}
for every
\(
1\leqslant r<N(q-1).
\)

<!-- remark-1: body_md -->
\label{rem:Lp:self}
Here, we record the natural integrability threshold for the self-similar profiles $U = U(x)$. The profiles constructed in Theorem~\ref{main} exhibit a structural barrier to fast spatial decay. Indeed, let $U$ be a bounded, sufficiently smooth solution to the profile equation~\eqref{ell_U_q} such that $U \in L^p(\mathbb{R}^N)$ for some $p > 1$. Testing equation \eqref{ell_U_q} against $|U|^{p-2}U$ and integrating over $\mathbb{R}^N$  (in the same way as in Lemma~\ref{lem:large:p:U}, below) yields
\begin{equation}\label{est:Lp:0}
\frac{4(p-1)}{p^2} \left\| \nabla |U|^{\frac{p}{2}} \right\|_2^2 + \left( \frac{N}{2p} - \frac{1}{2(q-1)} \right) \|U\|_p^p = 0,
\end{equation}
where the convection term vanishes identically upon integration by parts. Consequently, if
$                   
p \le N(q-1),
$                 
both terms in \eqref{est:Lp:0} are non-negative, forcing $U \equiv 0$. Thus, any nontrivial profile $U$ satisfying \eqref{est:Lp:0} must obey
\begin{equation*}
U \notin L^p(\mathbb{R}^N) \qquad \text{for all } 1 < p \le N(q-1).
\end{equation*}
In particular, when $q \ge 1 + \frac{2}{N}$, it follows that $U \notin L^2(\mathbb{R}^N)$. 

This cutoff matches the linear heat-profile estimate in Lemma~\ref{heat_L}, which yields $e^{\Delta} u_0 \in L^p(\mathbb{R}^N)$ for every $p>N(q-1)$.
This threshold aligns with the expected far-field asymptotics of $U(x)$. Because the initial datum $u_0$ is homogeneous of degree $-1/(q-1)$, one naturally expects $U$ to share the leading-order decay rate at spatial infinity:
$
U(x) = \mathcal{O}\!\left(|x|^{-{1}/(q-1)}\right)
$
as $ |x| \to \infty$.

<!-- remark-2: body_md -->
Let us explain the restriction
\(
q<{N}/({N-2})
\)
for 
$N\geqslant3$ in Theorem \ref{main}.
In the energy argument, \(V=U-U_0\) belongs to
\(W^{1,2}(\mathbb{R}^N)\). To use
\(\varphi\in W^{1,2}(\mathbb{R}^N)\) as a test function, we need
\(
\psi(U)\in L^2(\mathbb{R}^N).
\)
Since \(|\psi(s)|\leqslant |s|^q\), it is sufficient to have
\(U=V+U_0\in L^{2q}(\mathbb{R}^N)\).
The Sobolev embedding
\(
W^{1,2}(\mathbb{R}^N)
\hookrightarrow L^{\frac{2N}{N-2}}(\mathbb{R}^N)
\)
gives \(V\in L^{2q}(\mathbb{R}^N)\) when
\(q\leqslant N/(N-2)\). On the other hand, Lemma \ref{heat_L}
implies that \(U_0\in L^{2q}(\mathbb{R}^N)\) provided
\(
2q>N(q-1),
\)
or equivalently,
\(
q<{N}/({N-2}).
\)
                                                      
                                                               
Thus, the strict upper bound on \(q\) ensures that the nonlinear
term
\(
\int_{\mathbb{R}^N}
\psi(U)\,\partial_{x_1}\varphi\,\,\mathrm d x
\)
is well defined for  \(\varphi \in W^{1,2}(\mathbb{R}^N)\).

<!-- lemma-1: body_md -->
\label{C_omega:1} Let $N\geq 1$.
Let $q\geqslant 1+\frac{2}{N}$, let $\Omega\subset\mathbb{R}^N$ be a bounded
domain with $C^2$ boundary,
let $u_0$ be defined by \eqref{u0} and 
let $U_0=e^\Delta u_0$, and let
\(\psi\in W^{1,\infty}(\mathbb{R})\) be arbitrary.
Then the boundary value problem
\begin{align}\label{ell:bounded_4}
-\Delta V-\frac{x}{2}\cdot\nabla V-\frac{1}{2(q-1)}V
+\partial_{x_1}\psi(V+U_0)&=0,
&&x\in\Omega,\\
V&=0,
&&x\in\partial\Omega,
\end{align}
has a weak solution $V\in W^{1,2}_0(\Omega)$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Set
\[
c=\frac{1}{2(q-1)}.
\]
For \(W\in W^{1,2}_0(\Omega)\), define \(T(W)=Z\), where \(Z\in
W^{1,2}_0(\Omega)\) is the unique weak solution of
\[
-\Delta Z
=
\frac{x}{2}\cdot\nabla W+cW
-\partial_{x_1}\psi(W+U_0),
\qquad
Z|_{\partial\Omega}=0.
\]
Since \(\psi\in L^\infty(\mathbb{R})\), the right-hand side belongs to
\(W^{-1,2}(\Omega)\), and hence \(T\) is well defined by the
Lax--Milgram theorem.

The map
\(
T:W^{1,2}_0(\Omega)\longrightarrow W^{1,2}_0(\Omega)
\)
is continuous and compact. Indeed, continuity follows from the Lipschitz
continuity of \(\psi\) and the continuity of the inverse Dirichlet
Laplacian. To prove compactness, write \(T=T_1+T_2\), where
\[
T_1(W)
=
(-\Delta_D)^{-1}
\left(\frac{x}{2}\cdot\nabla W+cW\right),
\qquad
T_2(W)
=
(-\Delta_D)^{-1}
\left(-\partial_{x_1}\psi(W+U_0)\right).
\]
The map \(T_1\) is compact from \(W^{1,2}_0(\Omega)\) into itself by
elliptic \(W^{2,2}\)-regularity and the compact embedding
\(W^{2,2}(\Omega)\hookrightarrow W^{1,2}(\Omega)\). Moreover, if
\(\{W_n\}\) is bounded in \(W^{1,2}_0(\Omega)\), then, after passing to a
subsequence,
\(
W_n\to W
\)
in $L^2(\Omega)$.
Since \(\psi\) is globally Lipschitz,
\[
\psi(W_n+U_0)\to\psi(W+U_0)
\qquad\text{in }L^2(\Omega),
\]
and therefore \(T_2(W_n)\to T_2(W)\) in \(W^{1,2}_0(\Omega)\). Thus \(T\)
is compact.

It remains to obtain the Leray--Schauder {\it a priori} bound. Let
\(
V=\lambda T(V)
\)
for $\lambda \in [0,1]$.
Then
\[
-\Delta V
=
\lambda\left(
\frac{x}{2}\cdot\nabla V+cV
-\partial_{x_1}\psi(V+U_0)
\right).
\]
This equation holds in $\Omega$, together with the homogeneous Dirichlet
boundary condition.
Testing this equation with \(V\) and using the relation
\[
\frac12\int_\Omega (x\cdot\nabla V)V\,\,\mathrm d x
=
-\frac N4\int_\Omega V^2\,\,\mathrm d x,
\]
we obtain
\[
\|\nabla V\|_2^2
+
\lambda\left(
\frac N4-\frac{1}{2(q-1)}
\right)\|V\|_2^2
=
\lambda\int_\Omega
\psi(V+U_0)\partial_{x_1}V\,\,\mathrm d x.
\]
Since \(q\geq 1+2/N\), the second term on the left-hand side is
nonnegative. Hence
\[
\|\nabla V\|_2^2
\leqslant
|\Omega|^{1/2}\|\psi\|_\infty\|\nabla V\|_2,
\]
and consequently
\(
\|\nabla V\|_2
\leqslant
|\Omega|^{1/2}\|\psi\|_\infty.
\)
By the Poincar\'e inequality, this gives a bound for \(V\) in
\(W^{1,2}_0(\Omega)\), uniform with respect to \(\lambda\in[0,1]\).
The Leray--Schauder theorem therefore yields a fixed point
\(V\in W^{1,2}_0(\Omega)\) of \(T\), which is a weak solution of
\eqref{ell:bounded_4}.

<!-- lemma-2: body_md -->
\label{lemma_1}
Let $1<p<2$ and let $\Omega\subset\mathbb{R}^N$ be an arbitrary domain.
Suppose that
\[
|V|^{\frac p2-1}V\in W^{1,2}_0(\Omega).
\]
Then $V\in W^{1,p}_0(\Omega)$ and
\begin{equation}\label{lemma_1:estimate}
\int_\Omega |\nabla V|^p\,\mathrm d x
\leqslant
C(p)
\left(
\int_\Omega
\left|\nabla\big(|V|^{\frac p2-1}V\big)\right|^2\,\mathrm d x
+
\int_\Omega |V|^p\,\mathrm d x
\right).
\end{equation}
If, in addition, $\Omega$ is bounded, then
\begin{equation}\label{lemma_1:estimate:bounded}
\int_\Omega |\nabla V|^p\,\mathrm d x
\leqslant
C(p,\Omega)
\int_\Omega
\left|\nabla\big(|V|^{\frac p2-1}V\big)\right|^2\,\mathrm d x.
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Set
\[
w=|V|^{\frac p2-1}V,
\quad 
\text{then}
\quad 
V=|w|^{\frac2p-1}w.
\]
For smooth $w$ we have
\[
|\nabla V|
=
\frac2p |w|^{\frac2p-1}|\nabla w|,
\]
and consequently, by H\"older's inequality,
\begin{align*}
\int_\Omega|\nabla V|^p\,\mathrm d x
&=
\left(\frac2p\right)^p
\int_\Omega
|w|^{2-p}|\nabla w|^p\,\mathrm d x
\\
&\leqslant
\left(\frac2p\right)^p
\left(\int_\Omega|\nabla w|^2\,\mathrm d x\right)^{p/2}
\left(\int_\Omega|w|^2\,\mathrm d x\right)^{(2-p)/2}
\\
&\leqslant
C(p)
\left(
\int_\Omega|\nabla w|^2\,\mathrm d x
+
\int_\Omega|w|^2\,\mathrm d x
\right).
\end{align*}
The general case follows by approximation in $W^{1,2}_0(\Omega)$, which we describe in detail in Lemma \ref{C_psi_1} below.
Since $|w|^2=|V|^p$, this proves inequality \eqref{lemma_1:estimate}.
If $\Omega$ is bounded, inequality \eqref{lemma_1:estimate:bounded} follows
from the Poincar\'e inequality applied to $w$.

<!-- lemma-3: body_md -->
\label{lemma_2}
Let
\(
q\in\left(1+\frac1N,1+\frac2N\right]
\)
and let $\Omega\subset\mathbb{R}^N$ be a bounded domain with $C^2$ boundary.
Let $\psi\in W^{1,\infty}(\mathbb{R})$ be given by formula \eqref{psi1}  
with some $A>0$. 
Let $u_0$ be defined by \eqref{u0} and set $U_0=e^\Delta u_0$.
For every
\(
1<p<N(q-1),
\)
the boundary value problem
\begin{equation}\label{ell:bounded}
\begin{aligned}
-\Delta V-\frac{x}{2}\cdot\nabla V
-\frac{1}{2(q-1)}V
&=
-\partial_{x_1}\psi (V+e^\Delta u_0),
&&x\in\Omega,\\
V&=0,
&&x\in\partial\Omega,
\end{aligned}
\end{equation}
has a weak solution $V\in W^{1,p}_0(\Omega)$.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Set
\[
c=\frac{1}{2(q-1)}\quad \text{and} \qquad X=W^{1,p}_0(\Omega).
\]
Since \(q\leq 1+2/N\) and \(p<N(q-1)\), we have \(1<p<2\).
Moreover, by Lemma \ref{heat_L}, we have 
\(
U_0,\nabla U_0\in L^\infty(\mathbb{R}^N).
\)

For \(W\in X\), let \(T(W)=Z\), where \(Z\in X\) solves
\[
-\Delta Z
=
\frac{x}{2}\cdot\nabla W+cW
-\partial_{x_1}\psi(W+U_0),
\qquad
Z|_{\partial\Omega}=0.
\]
Since \(\psi\) is bounded and globally Lipschitz, standard elliptic
regularity estimates show that \(T:X\to X\) is well defined and continuous.
Moreover, the proof of Lemma~\ref{C_omega:1} shows that \(T\) is compact.
                                   
                                                                    
                                                                           
                                                                       
                                                           
                                 

We apply the Leray--Schauder theorem. Let
\(
V=\lambda T(V)
\)
for $\lambda \in [0,1]$.
Then
\begin{equation}\label{ell:bounded:homotopy:p}
-\Delta V
=
\lambda\left(
\frac{x}{2}\cdot\nabla V+cV
-\partial_{x_1}\psi(V+U_0)
\right).
\end{equation}
Standard elliptic bootstrap, using the chain rule in the nonlinear term and the boundedness of \(\psi'\), yields
enough regularity to justify the calculations below.
               
                                                                  
                 
Using a standard regularization of \(|V|^{p-2}V\) as a test function in
equation
\eqref{ell:bounded:homotopy:p} 
and then passing to the limit
(see the proof of Lemma \ref{C_psi_1} below for more detailed reasoning), we obtain
\begin{equation}\label{p_est_correct}
\begin{split}
&\frac{4(p-1)}{p^2}
\int_\Omega
\left|
\nabla\bigl(|V|^{\frac p2-1}V\bigr)
\right|^2\,\mathrm d x
+
\lambda\left(
\frac{N}{2p}-\frac{1}{2(q-1)}
\right)
\int_\Omega |V|^p\,\mathrm d x
\\
&\qquad\leqslant
\lambda
\left|
\int_\Omega
\psi'(V+U_0)
\bigl(\partial_{x_1}V+\partial_{x_1}U_0\bigr)
|V|^{p-2}V\,\,\mathrm d x
\right|.
\end{split}
\end{equation}
Here we used
\[
\frac12\int_\Omega
(x\cdot\nabla V)|V|^{p-2}V\,\,\mathrm d x
=
-\frac{N}{2p}\int_\Omega |V|^p\,\,\mathrm d x.
\]
Since \(p<N(q-1)\), the second term on the left-hand side of
\eqref{p_est_correct} is nonnegative.

By the definition \eqref{psi1} of the truncated nonlinearity,
\[
\psi'(V+U_0)=0
\qquad\text{a.e. on }\{|V+U_0|>A\}.
\]
Consequently, on the support of \(\psi'(V+U_0)\), we have
\(
|V|\leq A+\|U_0\|_\infty,
\)
and therefore
\[
\left|
\int_\Omega
\psi'(V+U_0)
\bigl(\partial_{x_1}V+\partial_{x_1}U_0\bigr)
|V|^{p-2}V\,\,\mathrm d x
\right|
\leq
C\left(
\|\nabla V\|_{L^1(\Omega)}+1
\right),
\]
where   \(C=C(U_0,|\Omega|,A)>0\) is independent of \(\lambda\).

Set
\[
E=
\int_\Omega
\left|
\nabla\bigl(|V|^{\frac p2-1}V\bigr)
\right|^2\,\mathrm d x.
\]
By Lemma \ref{lemma_1},
\[
\|\nabla V\|_p^p\leq C E,
\]
and hence
\[
\|\nabla V\|_1
\leq
C\|\nabla V\|_p
\leq C E^{1/p}.
\]
Dropping the nonnegative \(L^p\)-term in \eqref{p_est_correct}, we obtain
\[
E\leq C E^{1/p}+C.
\]
Since \(p>1\), the $\varepsilon$-Young inequality gives
\(
E\leq C,
\)
uniformly with respect to \(\lambda\in[0,1]\). Lemma \ref{lemma_1} and
the Poincar\'e inequality then yield
\(
\|V\|_{W^{1,p}_0(\Omega)}
\leq C,
\)
again independently of \(\lambda\).
Thus all the assumptions of the Leray--Schauder theorem are satisfied.
Hence \(T\) has a fixed point \(V\in W^{1,p}_0(\Omega)\), which is a weak
solution of \eqref{ell:bounded}.

<!-- lemma-4: body_md -->
\label{C_psi}
Assume that
\[
q>1+\frac{2}{N}
\quad\text{if }N\in\{1,2\},
\qquad
1+\frac{2}{N}<q<\frac{N}{N-2}
\quad\text{if }N\geqslant3.
\]
Let $\psi=\psi_A$ be the truncated nonlinearity defined in
\eqref{psi1}, with an arbitrary truncation level $A>0$. Let
$u_0$ be given by \eqref{u0} and set
\(
U_0=e^\Delta u_0.
\)
Then, for every $k\in\mathbb N$, the boundary value problem
\eqref{Vk_equ} has a weak solution
\(
V_k\in W^{1,2}_0(B_k).
\)
Moreover, there exists a constant
\(
C=C(A,N,q,U_0)>0,
\)
independent of $k$ and of the chosen solution $V_k$, such that
\begin{equation}\label{Vk_psi}
\|\nabla V_k\|_{L^2(B_k)}^2
+
\|V_k\|_{L^2(B_k)}^2
\leqslant C.
\end{equation}

If, in addition,
$                                            
1+\frac{2}{N}<q<2,
$                
then the constant in inequality  \eqref{Vk_psi} can be chosen in the form
\(
C=C(N,q,U_0),
\)
and is therefore independent of both $k$ and the truncation level
$A$.
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
The existence of $V_k$ follows from Lemma \ref{C_omega:1}.
It remains to prove the stated estimates.
Set
\[
\alpha
=
\frac{N}{4}-\frac{1}{2(q-1)}.
\]
The assumption $q>1+2/N$ implies that
$                                     
\alpha>0.
$                

Testing equation \eqref{Vk_equ} with $V_k$ and using
$V_k=0$ on $\partial B_k$, we obtain
\begin{align}
\|\nabla V_k\|_{L^2(B_k)}^2
+
\alpha\|V_k\|_{L^2(B_k)}^2
&=
\int_{B_k}
\psi_A(V_k+U_0)\,
\partial_{x_1}V_k\,dx.
\label{C_psi:energy}
\end{align}
         
    
                    
                           
   
                                   
    

Let
\(
\Psi_A(s)=\int_0^s\psi_A(z)\,dz.
\)
Since $\psi_A$ is globally Lipschitz, the Sobolev chain rule gives
\[
\partial_{x_1}\Psi_A(V_k)
=
\psi_A(V_k)\partial_{x_1}V_k
\qquad\text{a.e. in }B_k.
\]
Because $V_k=0$ on $\partial B_k$ and $\Psi_A(0)=0$, integration
by parts yields
\[
\int_{B_k}
\psi_A(V_k)\partial_{x_1}V_k\,dx
=
\int_{B_k}\partial_{x_1}\Psi_A(V_k)\,dx
=
0.
\]
Consequently,
\begin{align}
\left|
\int_{B_k}
\psi_A(V_k+U_0)\partial_{x_1}V_k\,dx
\right|
&=
\left|
\int_{B_k}
\bigl(
\psi_A(V_k+U_0)-\psi_A(V_k)
\bigr)
\partial_{x_1}V_k\,dx
\right|.
\label{C_psi:difference}
\end{align}

The truncations $\psi_A$ satisfy
\begin{equation}\label{C_psi:uniform-difference}
|\psi_A(a+b)-\psi_A(a)|
\leqslant
C(q)|b|
\bigl(
|a|^{q-1}+|b|^{q-1}

For simplicity, write
\[

\qquad\text{and}\qquad 
c=\frac{1}{2(q-1)}.
\]
All constants below are independent of $k$.

In the following reasoning, we present a regularization procedure which allows us to obtain $W^{1,p}$-estimates for $p\in (1,2)$ in this proof as well as in other parts of this paper.
For $\delta>0$, define
\[
\Phi_\delta(s)
=
s(s^2+\delta)^{\frac{p-2}{2}},
\qquad
Q_\delta(s)
=
\int_0^s\Phi_\delta(\tau)\,\,\mathrm d\tau.
\]
Then $\Phi_\delta$ is increasing and
\begin{equation}\label{Cpsi1:Phi-properties}
0\leqslant\Phi_\delta'(s)
\leqslant C(p)|s|^{p-2},
\qquad s\neq0.
\end{equation}
Moreover,
\[
\Phi_\delta(s)\longrightarrow |s|^{p-2}s,
\qquad
Q_\delta(s)\longrightarrow \frac1p|s|^p
\]
as $\delta\downarrow0$.
Testing the boundary value problem \eqref{ell:bounded:uni:p:Cpsi1} with
$\Phi_\delta(V)$ gives
\begin{align}
&\int_{B_k}\Phi_\delta'(V)|\nabla V|^2\,dx
+\frac N2\int_{B_k}Q_\delta(V)\,dx
-c\int_{B_k}V\Phi_\delta(V)\,dx
\nonumber\\
&\qquad
=
\int_{B_k}
\psi_A(V+U_0)\Phi_\delta'(V)\,
\partial_{x_1}V\,dx.
\label{Cpsi1:regularized-energy}
\end{align}
                                                                
                                  
                                  

Introduce
\[
P_\delta(v,u)
=
\int_0^v
\psi_A(s+u)\Phi_\delta'(s)\,ds.
\]
Then
\[
\partial_vP_\delta(v,u)
=
\psi_A(v+u)\Phi_\delta'(v).
\]
Consequently,
\[
\partial_{x_1}P_\delta(V,U_0)
=
\psi_A(V+U_0)\Phi_\delta'(V)\partial_{x_1}V
+
\partial_uP_\delta(V,U_0)\partial_{x_1}U_0.

Because \(V=0\) on \(\partial B_k\) and

=
-\int_{B_k}
\partial_uP_\delta(V,U_0)\partial_{x_1}U_0\,dx.
\end{equation}
Combining \eqref{Cpsi1:regularized-energy} and

-c\int_{B_k}V\Phi_\delta(V)\,dx
\nonumber\\
&\qquad
=
-\int_{B_k}
\partial_uP_\delta(V,U_0)\partial_{x_1}U_0\,dx.
\label{Cpsi1:regularized-energy:P}
\end{align}

 We next estimate the last integral using

$

If $p>N(q-1)$, set
\[
\beta_p
=
\frac{1}{2(q-1)}-\frac{N}{2p}>0.
\]
Then equation \eqref{large:p:energy} becomes
\begin{equation}\label{large:p:gradient:exact}
\left\|\nabla |U|^{p/2}\right\|_2^2
=
D_p\|U\|_p^p,
\qquad
D_p
=
\frac{p^2}{4(p-1)}\beta_p.
\end{equation}
                                       
Moreover, for $p\geqslant2$,
\[
D_p
\leqslant
\frac{p^2}{8(q-1)(p-1)}
\leqslant
\frac{p}{4(q-1)},
\]
which proves                       
\begin{equation}\label{large:p:grad}
\left\|\nabla |U|^{p/2}\right\|_{L^2(\mathbb{R}^N)}^2
\leqslant
C(q)p\|U\|_{L^p(\mathbb{R}^N)}^p.
\end{equation}

We now bootstrap the integrability of $U$.

\medskip
\noindent
\textit{Case 1: $N\geqslant3$.}
Set
\[
\kappa=\frac{N}{N-2}.
\]
Applying the Sobolev inequality to
$w=|U|^{p/2}$ and using
\eqref{large:p:gradient:exact}, we obtain
\[
\|U\|_{\kappa p}^p
=
\left\||U|^{p/2}\right\|_{\frac{2N}{N-2}}^2
\leqslant
C(N)\left\|\nabla |U|^{p/2}\right\|_2^2
=
C(N)D_p\|U\|_p^p.
\]
Hence
\begin{equation}\label{large:p:iteration:step}
\|U\|_{\kappa p}
\leqslant
\bigl(C(N)D_p\bigr)^{1/p}\|U\|_p.
\end{equation}

Let
\(
p_j=\kappa^jp_0.
\)
Since $p_j\geqslant p_0>1$,
\[
D_{p_j}\leqslant C(q,p_0)p_j.
\]
Iterating \eqref{large:p:iteration:step} yields
\[
\|U\|_{p_m}
\leqslant
\prod_{j=0}^{m-1}
\bigl(C(N)D_{p_j}\bigr)^{1/p_j}
\|U\|_{p_0}.
\]
The product is bounded uniformly in $m$ because
\[
\sum_{j=0}^\infty
\frac{\log\bigl(\max\{1,C(N)D_{p_j}\}\bigr)}{p_j}
\leqslant
C\sum_{j=0}^\infty
\frac{1+\log p_j}{p_j}
<\infty.
\]
Since $p_m\to\infty$, it follows that
\(
\|U\|_\infty
\leqslant
C(N,q,p_0)\|U\|_{p_0}.
\)

\medskip
\noindent
\textit{Case 2: $N=2$.}
The Gagliardo--Nirenberg inequality gives
\[
\|w\|_4^2
\leqslant
C\|w\|_2\|\nabla w\|_2.
\]
Taking $w=|U|^{p/2}$ and using
\eqref{large:p:gradient:exact}, we obtain
\[
\|U\|_{2p}^p
\leqslant
C\|U\|_p^{p/2}
\left\|\nabla |U|^{p/2}\right\|_2
=
CD_p^{1/2}\|U\|_p^p.
\]
Therefore,
\[
\|U\|_{2p}
\leqslant
\bigl(CD_p^{1/2}\bigr)^{1/p}\|U\|_p.
\]
Taking $p_j=2^jp_0$ and iterating as above gives
\[
\|U\|_\infty
\leqslant
C(q,p_0)\|U\|_{p_0}.
\]

\medskip
\noindent
\textit{Case 3: $N=1$.}
Using the inequality
\(
\|w\|_\infty^2
\leqslant
C\|w\|_2\|w'\|_2
\)
with $w=|U|^{p_0/2}$, we obtain
\begin{align*}
\|U\|_\infty^{p_0}
\leqslant
C\|U\|_{p_0}^{p_0/2}
\left\|\partial_x|U|^{p_0/2}\right\|_2
=
CD_{p_0}^{1/2}\|U\|_{p_0}^{p_0}.
\end{align*}
Consequently,
\[
\|U\|_\infty
\leqslant
\bigl(CD_{p_0}^{1/2}\bigr)^{1/p_0}
\|U\|_{p_0}
\leqslant
C(q,p_0)\|U\|_{p_0}.
\]

Write $V=V_k$ and set
\[
D=\|\partial_{x_1}U_0\|_{L^\infty(\mathbb{R}^2)},
\qquad
F=|U_0\partial_{x_1}U_0|.
\]
By Lemma~\ref{heat_L},
\[
|U_0(x)|+|\nabla U_0(x)|
\leqslant \frac{C(U_0)}{1+|x|}.
\]
Hence $D<\infty$ and
\[
F(x)\leqslant \frac{C(U_0)}{(1+|x|)^2},
\qquad
F\in L^s(\mathbb{R}^2)
\quad\text{for every }s>1.
\]

For each fixed $A$ and $k$, the boundedness of $\psi_A'$
and elliptic bootstrap give
$V\in W^{2,t}(B_k)$ for every finite $t>1$.
Thus the regularized power tests below are admissible.
The constants in this preliminary regularity argument
need not be uniform in $A$ or $k$.

For $1<s<2$, set
\[
w_s=|V|^{s/2-1}V,
\qquad
\alpha_s=\frac{4(s-1)}{s^2},
\qquad
\beta_s=\frac1s-\frac12.
\]
Unless otherwise indicated, norms of $V$ and $w_s$ are
taken over $B_k$.

Testing \eqref{endpoint:N2:q2:bounded} with
\[
\Phi_{s,\delta}(V)
=
V(V^2+\delta)^{(s-2)/2},
\]
integrating the convection term by parts, and letting
$\delta\downarrow0$, we obtain
\begin{align}
\alpha_s\|\nabla w_s\|_2^2
+\beta_s\|V\|_s^s
&\leqslant
\int_{B_k}|\partial_{x_1}U_0|
\left|
(s-1)\int_0^V
\psi_A'(z+U_0)|z|^{s-2}\,\,\mathrm d z
\right|\,\mathrm d x.
\label{endpoint:power:energy}
\end{align}
To justify the nonlinear integration by parts, use the
primitive
\[
P_{s,\delta}(v,u)
=
\int_0^v
\psi_A(z+u)\Phi_{s,\delta}'(z)\,\,\mathrm d z.
\]
Since $P_{s,\delta}(0,u)=0$ and $V=0$ on $\partial B_k$,
\[
\int_{B_k}
\psi_A(V+U_0)\Phi_{s,\delta}'(V)
\partial_{x_1}V\,\,\mathrm d x
=
-\int_{B_k}
\partial_uP_{s,\delta}(V,U_0)
\partial_{x_1}U_0\,\,\mathrm d x.
\]
Dominated convergence applies to the latter integral,
while Fatou's lemma gives the diffusion term in
\eqref{endpoint:power:energy}.

The bound $|\psi_A'(z)|\leqslant 2|z|$ gives
\begin{align*}
\left|
(s-1)\int_0^v
\psi_A'(z+u)|z|^{s-2}\,\,\mathrm d z
\right|
&\leqslant
2(s-1)\int_0^{|v|}
(z+|u|)z^{s-2}\,\,\mathrm d z\\
&=
\frac{2(s-1)}s|v|^s
+2|u||v|^{s-1}.
\end{align*}
Consequently, by H\"older's inequality,
\begin{equation}
\label{endpoint:basic:estimate}
\alpha_s\|\nabla w_s\|_2^2
+\beta_s\|V\|_s^s
\leqslant
\frac{2(s-1)}s D\|V\|_s^s
+
2\|F\|_{L^s(\mathbb{R}^2)}\|V\|_s^{s-1}.
\end{equation}
The factor $s-1$ in the first term on the right-hand side
allows us to absorb this term by choosing $s$ close to $1$.

Choose an auxiliary exponent
\[
1<r<
\min\left\{
p,\,
1+\frac{1}{1+4D}
\right\}.
\]
Then
\[
\gamma_r
:=
\beta_r-\frac{2(r-1)}rD
=
\frac{2-r-4D(r-1)}{2r}
>0.
\]
Applying \eqref{endpoint:basic:estimate} with $s=r$
and using Young's inequality, we obtain
\begin{align*}
\alpha_r\|\nabla w_r\|_2^2
+\gamma_r\|V\|_r^r
&\leqslant
2\|F\|_{L^r(\mathbb{R}^2)}\|V\|_r^{r-1}\\
&\leqslant
\frac{\gamma_r}{2}\|V\|_r^r+C(p,U_0).
\end{align*}
Therefore,
\[
\|\nabla w_r\|_2^2+\|V\|_r^r
\leqslant C(p,U_0).
\]
Lemma~\ref{lemma_1} now yields
\begin{equation}
\label{endpoint:auxiliary:W1r}
\|V\|_{W^{1,r}(B_k)}
\leqslant C(p,U_0),
\end{equation}
uniformly in $A$ and $k$.

Extend $V$ by zero outside $B_k$.
Since $V\in W^{1,r}_0(B_k)$, the Sobolev inequality
on $\mathbb{R}^2$ and \eqref{endpoint:auxiliary:W1r} imply
\[
\|V\|_{L^{r^*}(\mathbb{R}^2)}
\leqslant C(p,U_0),
\qquad
r^*=\frac{2r}{2-r}>2.
\]
Because $r<p<2<r^*$, interpolation gives
\begin{equation}
\label{endpoint:uniform:Lp}
\|V\|_{L^p(B_k)}
\leqslant C(p,U_0).
\end{equation}

Returning to \eqref{endpoint:basic:estimate} with $s=p$
and using \eqref{endpoint:uniform:Lp}, we obtain
\[
\|\nabla w_p\|_2^2
\leqslant C(p,U_0).
\]
A second application of Lemma~\ref{lemma_1} gives
\[
\|V\|_{W^{1,p}(B_k)}^p
\leqslant
C(p)\left(
\|\nabla w_p\|_2^2+\|V\|_p^p
\right)
\leqslant C(p,U_0),
\]
which proves \eqref{endpoint:N2:q2:W1p}.

Finally, apply the estimate with $p=3/2$.
After extending $V_k$ by zero and passing to the
whole-space limit, weak lower semicontinuity gives
\[
\|V\|_{W^{1,3/2}(\mathbb{R}^2)}
\leqslant C(U_0).
\]
The Sobolev embedding
$W^{1,3/2}(\mathbb{R}^2)\hookrightarrow L^6(\mathbb{R}^2)$
therefore yields
\[
\|V\|_{L^6(\mathbb{R}^2)}
\leqslant C(U_0).
\]
Since $U_0\in L^6(\mathbb{R}^2)\cap L^\infty(\mathbb{R}^2)$,
the whole-space profile $U=V+U_0$ satisfies
\[
\|U\|_{L^6(\mathbb{R}^2)}
\leqslant C(U_0).
\]
Applying Lemma~\ref{lem:large:p:U} to $U$ with $p_0=6$,
we obtain
\[
\|U\|_{L^\infty(\mathbb{R}^2)}
\leqslant
C\|U\|_{L^6(\mathbb{R}^2)}
\leqslant C(U_0),
\]
where the constant is independent of $\psi_A$ and hence
of $A$. Thus
\[
\|V\|_{L^\infty(\mathbb{R}^2)}
\leqslant
\|U\|_{L^\infty(\mathbb{R}^2)}
+\|U_0\|_{L^\infty(\mathbb{R}^2)}
\leqslant C(U_0).
\]
