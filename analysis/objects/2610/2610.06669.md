\label{thm:hyzz}
  Let $B$ be a smooth, purely even $\Bbbk$-analytic space, let $(\mathcal{H},\nabla)/B$ be maximal at $b\in B(\Bbbk)$, and let

\label{lem:lattice-independence}
  Let $(\mathcal{M},\nabla)$ be a regular-singular meromorphic connection over
  $\Bbbk((u))$.
  \begin{enumerate}
   \item The multiset in $\Bbbk/\mathbb{Z}$ of residue eigenvalues modulo $\mathbb{Z}$, counted

<!-- definition-1: body_md -->
\label{def:exponents}
  For $\lambda\in\Bbbk$, define \[   \mathcal{E}^{\lambda/u}:=\Bbbk((u))e_\lambda,   \qquad   \nabla_{\partial_u}e_\lambda   =-\lambda u^{-2}e_\lambda. \] Thus \[   \nabla_{\partial_u}(f e_\lambda)   =(f'-\lambda u^{-2}f)e_\lambda. \] After adjoining an invertible differential symbol $s_\lambda$ satisfying $s_\lambda'=\lambda u^{-2}s_\lambda$, the section $s_\lambda e_\lambda$ is horizontal. We write $s_\lambda=e^{-\lambda/u}$. 
  
  A meromorphic connection $\mathcal{M}$ is said to be of \emph{unramified exponential type} if it decomposes as
  \begin{equation}\label{eq:elementary}
   \mathcal{M}\cong\bigoplus_{\lambda\in\Lambda}
   \mathcal{E}^{\lambda/u}\otimes\mathcal{R}_\lambda
  \end{equation}
  for a finite indexing set $\Lambda\subset\Bbbk$ and a collection of regular-singular connections $\mathcal{R}_\lambda$. 
  
  For a regular-singular connection $\mathcal{R}$, we define its multiset of exponents:
  \[
  \operatorname{Exp}(\mathcal{R}):=\bigl\{\bigl\{ -\vartheta+\mathbb{Z} \bigm| \vartheta\text{ is a residue eigenvalue of } \mathcal{R} \bigr\}\bigr\},
  \]
  where the elements are counted with their algebraic multiplicities. We extend this definition to the elementary factors setting $\operatorname{Exp}(\mathcal{E}^{\lambda/u}\otimes\mathcal{R}_\lambda):=\operatorname{Exp}(\mathcal{R}_\lambda)$.
  
  A regular-singular connection has \emph{rational residue eigenvalues} if the residue eigenvalues of one, equivalently every, logarithmic lattice lie in $\mathbb{Q}$. We abbreviate this condition to \emph{rational residues}. We say that a connection of unramified exponential type, as in \eqref{eq:elementary}, has rational residues if this condition holds for every regular-singular component $\mathcal{R}_\lambda$. Because changing the underlying logarithmic lattice only shifts the residue eigenvalues by integers, Lemma~\ref{lem:lattice-independence} ensures that both the rationality condition and the multiset of exponents $\operatorname{Exp}$ are well-defined and lattice-independent.

<!-- lemma-3: body_md -->
\label{lem:splitting}
  Let $F$ be an algebraically closed field of characteristic zero, and let $\mathcal{L}$ be an $F[[u]]$-lattice in a meromorphic connection $\mathcal{M}$ over $F((u))$ such that $u^2\nabla_{\partial_u}\mathcal{L}\subseteq\mathcal{L}$. Define the leading endomorphism
  \[
  A_0:=(u^2\nabla_{\partial_u})\bmod u\mathcal{L}\in\operatorname{End}_F(\mathcal{L}/u\mathcal{L}),
  \]
  and consider the generalized eigenspace decomposition $\mathcal{L}/u\mathcal{L}=\bigoplus_{\lambda}V_\lambda$, where $V_\lambda$ corresponds to the eigenvalue $\lambda \in \operatorname{spec}(A_0)$. Then:
  \begin{enumerate}
   \item There exists a unique decomposition $\mathcal{L}=\bigoplus_\lambda\mathcal{L}_\lambda$ into $u^2\nabla_{\partial_u}$-stable $F[[u]]$-submodules such that the reduction modulo $u$ yields $\mathcal{L}_\lambda/u\mathcal{L}_\lambda=V_\lambda$.
   \item This lattice decomposition induces a direct sum decomposition of the meromorphic connection $\mathcal{M}=\bigoplus_\lambda\mathcal{M}_\lambda$, where each $\mathcal{M}_\lambda:=\mathcal{L}_\lambda[u^{-1}]$ is a differential submodule of rank $\operatorname{rk}\mathcal{M}_\lambda=\dim V_\lambda$.
   \item For each eigenvalue $\lambda$, the connection $\mathcal{E}^{\lambda/u}\otimes\mathcal{M}_\lambda$ has a nilpotent leading endomorphism given by $(A_0-\lambda)|_{V_\lambda}$. Consequently, the full connection factors as
   \[
   \mathcal{M}\cong\bigoplus_\lambda\mathcal{E}^{-\lambda/u}\otimes\bigl(\mathcal{E}^{\lambda/u}\otimes\mathcal{M}_\lambda\bigr).
   \]
  \end{enumerate}

<!-- corollary-1: body_md -->
\label{cor:leading-exponential} In the setting of Lemma~\ref{lem:splitting}, suppose that $\mathcal{M}$ is of unramified exponential type: \[   \mathcal{M}=\bigoplus_{\lambda\in\Lambda}\mathcal{N}_\lambda,   \qquad   \mathcal{N}_\lambda\cong\mathcal{E}^{\lambda/u}\otimes\mathcal{R}_\lambda, \] where every $\mathcal{R}_\lambda$ is regular singular. Then \[   \Lambda=-\operatorname{spec}(A_0),   \qquad   \mathcal{M}_\lambda=\mathcal{N}_{-\lambda},   \qquad   \operatorname{rk}\mathcal{R}_{-\lambda}=\dim V_\lambda. \]

<!-- proof-1: body_md -->
Choose a logarithmic lattice in each $\mathcal{R}_\lambda$. The induced lattice in $\mathcal{N}_\lambda$ has leading endomorphism $-\lambda I$. We first verify that a differential morphism $\phi: \mathcal{M}_{\lambda_1} \to \mathcal{N}_{\lambda_2}$ between two connections with quadratic lattices and disjoint leading spectra is zero, even if the morphism does not preserve the lattices. In chosen lattice bases, write the morphism as a Laurent matrix \[   \phi(u)=u^j\phi_j+u^{j+1}\phi_{j+1}+\cdots,   \qquad \phi_j\neq0. \] If the two connection matrices are $A(u)/u^2$ and $B(u)/u^2$, horizontality gives \[   u^2\phi'(u)+B \phi(u)-\phi(u) A=0. \] The coefficient of $u^j$ is \[   B_0\phi_j-\phi_jA_0=0. \] Disjoint leading spectra make this Sylvester operator invertible, a contradiction. Apply this observation to the components of the identity map between $\mathcal{M}=\bigoplus_{\lambda_1} \mathcal{M}_{\lambda_1}$ and $\mathcal{M}=\bigoplus_{\lambda_2}\mathcal{N}_{\lambda_2}$. Every component vanishes unless $\lambda_1=-\lambda_2$. Applying the same argument in the reverse direction identifies $\mathcal{M}_{\lambda_1}$ with the embedded summand $\mathcal{N}_{-\lambda_1}$. The rank formula follows from Lemma~\ref{lem:splitting}.

\label{prop:low-dimension}
  Let $W$ be a connected smooth complex projective variety of
  dimension at most two. Suppose that one of its generic Hodge
  factors is cyclic and of unramified exponential type with
  rational residues at a rational point. Its rank $m$ and
  regularized exponent multiset $F$ have one of the following types:
  \begin{enumerate}
   \item \emph{point type}:
   $m=1$ and $F=\{\!\{\overline{0}\}\!\}$;
   \item \emph{curve type}:
   $m=2$ and $F=\{\!\{\overline{1/2},\overline{1/2}\}\!\}$;
   \item \emph{surface type}:
   $m=3$ and $F=\{\!\{\overline{0},\overline{0},\overline{0}\}\!\}$.
  \end{enumerate}
  
  A generic Hodge factor of any projective space that satisfies the same hypotheses is of point type.

All transfers in this proof use Lemma~\ref{lem:separation};

  For a point, $H_{\operatorname{Hdg}}=\mathbb{Q}\mathbf1$, $G=0$, and $K^{\mathrm F}=-y_0\operatorname{id}$.

  projective space.

  Hodge-fixed base, the only additive coordinate is the unit
  coordinate, and the quantum product is the cup product.
  Writing $\eta\in H^2(C,\mathbb{Q})$ for the class with
  $\int_C\eta=1$, we have
  \[
  H_{\operatorname{Hdg}}(C)=\mathbb{Q}\mathbf1\oplus\mathbb{Q}\eta,
  \qquad

<!-- theorem-4: body_md -->
\label{thm:criterion}
  Let $X$ be a connected smooth complex projective variety of
  dimension $2\leq n\leq4$. Suppose that a generic Hodge factor
  has a cyclic representative of unramified exponential type
  with rational residues. Let $m$ be its rank and $F$ its
  regularized exponent multiset. If $X$ is rational, then:
  \begin{enumerate}
   \item if $n=2$, then
   \[
   (m,F)=(1,\{\!\{\overline{0}\}\!\});
   \]
   \item if $n=3$, then
   \[
   (m, F)\in
   \left\{
   (1,\{\!\{\overline{0}\}\!\}),
   (2,\{\!\{\overline{1/2},\overline{1/2}\}\!\})
   \right\};
   \]
   \item if $n=4$, then
   \[
   (m, F)\in
   \left\{
   (1,\{\!\{\overline{0}\}\!\}),
   (2,\{\!\{\overline{1/2},\overline{1/2}\}\!\}),
   (3,\{\!\{\overline{0},\overline{0},\overline{0}\}\!\})
   \right\}.
   \]
  \end{enumerate}
<!-- end theorem-4 -->
<!-- proof-9: body_md -->
Assume that $X$ is rational. By weak factorization
  \cite[Thm.~0.1.1]{AKMW}, a
  birational map $X\dashrightarrow\mathbb{P}^n$ factors into a finite
  sequence of blowups and blowdowns between smooth projective
  $n$-folds, with smooth centers of codimension at least two.
  
  Track the selected generic Hodge factor along this sequence.
  If the next variety is the blowup of the current one, use
  the converse direction of Lemma~\ref{lem:separation}, with
  the current variety as the specified summand. If the current
  variety is the blowup of the next one, use the forward
  direction of the lemma. The factor then transfers either
  to the next variety or to one of the copies of the center.
  At every step, its rank, cyclicity, unramified exponential
  type, rational residue eigenvalues, and exponent multiset
  are preserved.
  
  The process terminates either with a factor on a center
  of dimension at most $n-2$, or with a factor on $\mathbb{P}^n$.
  In the latter case Proposition~\ref{prop:low-dimension}
  gives point type. In the former case, the same proposition
  gives point type when $n=2$, point or curve type when
  $n=3$, and point, curve, or surface type when $n=4$.

<!-- lemma-8: body_md -->
\label{lem:application-matrix-recursion}
 A log-free solution of \eqref{eq:phi-system} satisfies the recursive relations
 \begin{equation}\label{eq:application-matrix-recursion}
  (\lambda\operatorname{id}-K^{\mathrm F})v_0=0,\qquad
  (\lambda\operatorname{id}-K^{\mathrm F})v_k
  =\bigl(G+(s+k-1)\operatorname{id}\bigr)v_{k-1}\quad(k\geq1).
 \end{equation}
 For every vector $w$ satisfying $w^\top(\lambda\operatorname{id}-K^{\mathrm F})=0$,
 these relations imply the solvability conditions
 \begin{equation}\label{eq:application-left-obstruction}
  w^\top\bigl(G+(s+k-1)\operatorname{id}\bigr)v_{k-1}=0.
 \end{equation}
 Furthermore, every exponent class of $\mathcal{R}_\lambda$ is represented modulo
 $\mathbb{Z}$ by the leading power $s$ of at least one such solution.
<!-- end lemma-8 -->
<!-- proof-10: body_md -->
Substituting the formal series into \eqref{eq:phi-system} and equating the coefficients of $u^s$ yields the primary relation $(\lambda\operatorname{id}-K^{\mathrm F})v_0=0$. Evaluating the coefficients at order $u^{s+k}$ gives the equality
 $(\lambda\operatorname{id}-K^{\mathrm F})v_k=(G+(s+k-1)\operatorname{id})v_{k-1}$. Applying the left kernel vector $w^\top$ annihilates the left-hand side, yielding the stated solvability condition.

 Regarding the final assertion, the formal spectral decomposition provided by
 Theorem~\ref{thm:fano}, after twisting by $\mathcal{E}^{\lambda/u}$, reads
 \[
 \mathcal{R}_\lambda\ \oplus\!
 \bigoplus_{\nu\ne\lambda}
 \mathcal{E}^{(\lambda-\nu)/u}\otimes\mathcal{R}_\nu.
 \]
 Consequently, $\mathcal{R}_\lambda$ operates as a direct summand of the twisted
 matrix connection: the formal isomorphism maps a vector in
 this initial summand to the original cohomology basis via a
 Laurent-series change of basis. For each residue eigenvalue of $\mathcal{R}_\lambda$,
 the constant-residue formal normal form of
 \cite[Prop.~3.12]{PutSinger} (valid over the algebraically
 closed characteristic-zero field $\Bbbk$ by
 \cite[Rem.~3.20(1)]{PutSinger}) yields an
 eigenvector solution carrying the opposite power of $u$ entirely free of
 logarithms. Expressing this solution in the original basis through the
 formal meromorphic isomorphism and the inclusion of the direct
 summand produces a solution to \eqref{eq:phi-system}. Since a Laurent-series change of basis modifies the leading power by an integer,
 its class modulo $\mathbb{Z}$ is the same.

  \begin{equation}\label{eq:app-polynomial-commutators}
   [\partial_x,p(x)]=p'(x),\qquad
   [p(x),u]=up'(x)u.
  \end{equation}
  The second identity follows from the first by commuting
  $p(x)$ with $u\partial_x=\operatorname{id}$.
  
  We next check that $\widehat\nabla_p$ preserves
  $\widehat\mathcal{H}$. Both $u\nabla_{X_p}$ and $p(x)$ preserve
  this lattice, and their reductions are equal:
  \[
  (u\nabla_{X_p})\bmod u=\mu(X_p)
  =p(K^{\mathrm F})=p(x)\bmod u.
  \]
  Their difference, therefore, has image in $u\widehat\mathcal{H}$.
  Moving $\partial_x=u^{-1}$ to the left in
  \eqref{eq:app-transport-operators}, using
  \eqref{eq:app-polynomial-commutators}, gives
  \begin{equation}\label{eq:app-lattice-transport}
   \widehat\nabla_p
   =u^{-1}\bigl(u\nabla_{X_p}-p(x)\bigr)+p'(x).
  \end{equation}
  Each term on the right preserves $\widehat\mathcal{H}$.
  
  Since $X_p$ is a vector field on $B$, it is independent of
  $u$. Flatness and the Leibniz rule give
  $[\nabla_{X_p},x]=[\nabla_{X_p},\partial_x]=0$.
  Using \eqref{eq:app-polynomial-commutators}, we obtain
  \[
  \begin{aligned}
  [\widehat\nabla_p,x]
  &=-[p(x)\partial_x,x]=-p(x),\\
  [\widehat\nabla_p,\partial_x]
  &=-[p(x),\partial_x]\partial_x=p'(x)\partial_x.
  \end{aligned}
  \]
  Commuting $\widehat\nabla_p$ with $u\partial_x=\operatorname{id}$
  then gives $[\widehat\nabla_p,u]=-up'(x)$, proving
  \eqref{eq:app-transport-commutators}.
  Finally, these identities can be written as
  \[
  \widehat\nabla_p u=u\bigl(\widehat\nabla_p-p'(x)\bigr),
  \qquad
  \widehat\nabla_p u^{-1}
  =u^{-1}\widehat\nabla_p+p'(x)u^{-1}.
  \]
  Starting with $\widehat\mathcal{H}$ and using that $p'(x)$
  preserves every shifted lattice, induction in both directions
  proves the assertion for $u^a\widehat\mathcal{H}$.

<!-- theorem-5: body_md -->
\label{cl:formal-persistence}
  Let $(\mathcal{H},\nabla)/B$ be a maximal F-bundle over a smooth,
  purely even base, and let $b\in B(\Bbbk)$. Use the notation
  \eqref{eq:app-polynomial-fields}, and put
  \begin{equation}\label{eq:app-euler-distribution}
   \mathcal D_w:=\operatorname{span}_{\Bbbk}
   \{X_p(w):p\in\Bbbk[z]\}
   =\operatorname{span}_{\Bbbk}
   \{(\mathsf{Eu}_w^{\mathrm F})^{\circ j}:j\geq0\}.
  \end{equation}
  Then the following statements hold.
  \begin{enumerate}
   \item There is a unique germ of a closed smooth analytic
   submanifold $W\subset B$ through $b$ such that
   $T_wW=\mathcal D_w$ for all sufficiently nearby rational
   points $w\in W(\Bbbk)$.
   \item The number of distinct eigenvalues of
   $\mathsf{Eu}^{\mathrm F}\circ(-)|_{T_W}$ and their algebraic
   multiplicities are locally constant on $W$.
   \item Suppose that $\widehat\mathcal{H}_b^{\mathrm{mer}}$ is of
   unramified exponential type. After shrinking $W$, the
   distinct eigenvalues of $K^{\mathrm F}|_W$ admit analytic
   labels $\lambda_\alpha(w)$, and there are regular-singular
   connections $\mathcal{R}_\alpha$ over $\Bbbk((u))$ such that
   \begin{equation}\label{eq:app-constant-factors}
    \widehat\mathcal{H}_w^{\mathrm{mer}}
    \cong\bigoplus_\alpha
    \mathcal{E}^{-\lambda_\alpha(w)/u}\otimes\mathcal{R}_\alpha,
    \qquad w\in W(\Bbbk).
   \end{equation}
   Here, the $\mathcal{R}_\alpha$ are the regular-singular factors at
   $b$ and do not depend on $w$, up to isomorphism.
  \end{enumerate}

<!-- proof-14: body_md -->
We first construct $W$ and prove the spectral assertion.
  We then construct parallel transport for
  $\widehat\nabla_p$ along a flow of $X_p$ and show that it
  preserves the regular-singular factors.
  
  \medskip
  \noindent\emph{Construction of the integral germ.}
  Let $n=\dim B$. By the Cayley--Hamilton theorem and
  \eqref{eq:F-Euler-powers}, the fields
  $X_1,X_z,\ldots,X_{z^{n-1}}$ generate the distribution
  \eqref{eq:app-euler-distribution} over $\mathcal{O}_B$.
  Equation~\eqref{eq:involution} shows that it is involutive.
  Its rank need not be constant in a neighborhood in $B$, so
  we construct its integral germ using flows.
  
  The local analytic flows of the fields $X_p$ exist by
  \cite[Lem.~3.6]{HYZZ}. For each such flow,
  \eqref{eq:involution}, followed by the Cayley--Hamilton theorem,
  gives a finite linear differential system for the pullbacks
  of the generators. Thus, the flow preserves $\mathcal D$ and
  its rank. Choose $r=\dim\mathcal D_b$ of the generators
  whose values at $b$ are independent. Their successive flows
  define an analytic map from an $r$-dimensional polydisk to
  $B$, with injective differential at the origin. After
  shrinking, its image is a closed embedded smooth germ $W$.
  Flow invariance gives $\dim\mathcal D_w=r$ on this image,
  and the differential of the parametrization has image
  $\mathcal D_w$. Hence $T_wW=\mathcal D_w$.
  Uniqueness follows because any other integral germ through $b$ has dimension $r$ and is preserved by the same flows, thus agreeing as a germ. This proves (i).
  
  \medskip
  \noindent\emph{Spectral constancy on $T_W$.}
  Write $L:=\mathsf{Eu}^{\mathrm F}\circ(-)|_{T_W}$; it preserves
  $T_W$ because $LX_q=X_{zq}$. Equation~\eqref{eq:involution}
  gives
  \[
  (\mathrm{Lie}_{X_p}L)(X_q)
  =[X_p,X_{zq}]-L[X_p,X_q]=-X_{pq}.
  \]
  Since the fields $X_q$ span $T_W$, this is the identity
  \begin{equation}\label{eq:Lie-derivative-equality}
   \mathrm{Lie}_{X_p}L=-p(L).
  \end{equation}
  Let $\Phi_s$ be the local flow of $X_p$, and let
  $\varphi_s$ be the scalar analytic flow determined by
  \begin{equation}\label{eq:app-scalar-flow}
   \partial_s\varphi_s(z)=-p(\varphi_s(z)),\qquad
   \varphi_0(z)=z.
  \end{equation}
  Thus \eqref{eq:Lie-derivative-equality} gives
  $\Phi_s^*L=\varphi_s(L)$. For small $s$, the function
  $\varphi_s$ is injective on the finite set $\operatorname{spec}(L_b)$
  and $\varphi_s'(\lambda)\ne0$ at each of its elements.
  On a generalized $\lambda$-eigenspace, write
  $L_b=\lambda\operatorname{id}+N$, where $N$ is nilpotent. Taylor expansion
  gives
  \begin{equation}\label{eq:app-jordan-transport}
   \varphi_s(\lambda\operatorname{id}+N)-\varphi_s(\lambda)\operatorname{id}
   =N g_s(N),\qquad g_s(0)=\varphi_s'(\lambda)\ne0.
  \end{equation}
  The operator $g_s(N)$ is invertible and commutes with $N$.
  Consequently, \eqref{eq:app-jordan-transport} has the same
  kernels of powers as $N$. The eigenvalue multiplicities, and
  indeed the Jordan block sizes, are therefore unchanged along
  each generating flow. Composing these flows proves (ii).
  
  \medskip
  \noindent\emph{Parallel transport on the completed lattices.}
  Fix a polynomial $p$ and a flow curve $b_s=\Phi_s(b)$.
  We use the operators $x$, $\partial_x$, and
  $\widehat\nabla_p$ of \eqref{eq:app-transport-operators}.
  In an analytic trivialization along this curve, write
  \begin{equation}\label{eq:app-analytic-operators}
   x=u^2\partial_u+C(s,u),\qquad
   \nabla_{X_p}=\partial_s+u^{-1}B(s,u),\qquad
   B(s,0)=p(C(s,0)),
  \end{equation}
  where $B$ and $C$ are analytic matrices. Put
  $\delta:=u\partial_u$, so that $x=u\delta+C(s,u)$.
  When $p(u\delta+C(s,u))$ is written with all powers of
  $\delta$ on the right, its constant term modulo $u$ is
  $p(C(s,0))$, and all its positive-degree coefficients are
  divisible by $u$. Equation~\eqref{eq:app-lattice-transport}
  therefore takes the form
  \begin{equation}\label{eq:app-bounded-transport}
   \widehat\nabla_p=\partial_s+A(s,u,\delta),
  \end{equation}
  with $A$ polynomial in $\delta$ and analytic in $(s,u)$.
  
  On a sufficiently small closed $u$-disk, $\delta$ is bounded
  in the Gauss norm, with $\|\delta\|\leq1$. Thus $A$ defines
  an analytic family of bounded operators. Write
  $A(s)=\sum_{j\geq0}A_js^j$, and choose $M\geq1$ such that
  $\|A_j\|\leq M^{j+1}$. Parallel transport is obtained by
  solving
  \begin{equation}\label{eq:app-transport-recursion}
  \begin{gathered}
  (\partial_s+A(s))T_s=0,\qquad T_0=\operatorname{id},\\
  T_s=\sum_{n\geq0}T_ns^n,\qquad
  (n+1)T_{n+1}=-\sum_{j=0}^n A_jT_{n-j}.
  \end{gathered}
  \end{equation}
  The absolute value on $\Bbbk$ is trivial on $\mathbb{Q}$, so
  $|n+1|=1$. The recursion gives $\|T_n\|\leq M^n$;
  hence $T_s$ converges in operator norm for $|s|<M^{-1}$.
  The equation $\partial_sS_s=S_sA(s)$, $S_0=\operatorname{id}$, gives
  the same bound. Differentiating $S_sT_s$ and $T_sS_s$
  shows that $S_s=T_s^{-1}$.
  
  To act on arbitrary formal series, perform the same
  construction on $\widehat\mathcal{H}_b/u^N\widehat\mathcal{H}_b$.
  The bounds are independent of $N$: the operators preserve
  the $u$-adic filtration, and the quotient Gauss norms satisfy
  the same estimates. Taking inverse limits therefore defines
  mutually inverse maps on the entire completed lattices,
  including formal series that converge on no $u$-disk.
  For a shifted lattice $u^a\widehat\mathcal{H}_b$, conjugation by
  $u^a$ replaces $\delta$ in
  \eqref{eq:app-bounded-transport} by $\delta+a$.
  Since $|a|\leq1$, the bounds remain uniform in $a\in\mathbb{Z}$.
  Uniqueness in \eqref{eq:app-transport-recursion} makes these
  constructions compatible. We obtain $\Bbbk$-linear
  isomorphisms
  \begin{equation}\label{eq:app-transport-maps}
   T_s\colon\widehat\mathcal{H}_b^{\mathrm{mer}}
   \longrightarrow\widehat\mathcal{H}_{b_s}^{\mathrm{mer}},\qquad
   T_s(u^a\widehat\mathcal{H}_b)=u^a\widehat\mathcal{H}_{b_s}
   \quad(a\in\mathbb{Z}).
  \end{equation}
  These maps need not be $\Bbbk((u))$-linear.
  
  \medskip
  \noindent\emph{Evolution of the operators and spectral summands.}
  For an operator $O$ along the flow, differentiating its
  conjugate gives
  \[
  \partial_s(T_s^{-1}O_{b_s}T_s)
  =T_s^{-1}[\widehat\nabla_p,O]_{b_s}T_s.
  \]
  Apply this identity to $x$, $\partial_x$, and $u$.
  The commutators \eqref{eq:app-transport-commutators} and
  the initial conditions at $s=0$ yield
  \begin{equation}\label{eq:app-integrated-transport}
  \begin{aligned}
  x_s&:=T_s^{-1}x_{b_s}T_s=\varphi_s(x_b),\\
  \partial_{x,s}&:=T_s^{-1}\partial_{x,b_s}T_s
  =\varphi_s'(x_b)^{-1}\partial_{x,b},\\
  u_s&:=T_s^{-1}uT_s=u\varphi_s'(x_b).
  \end{aligned}
  \end{equation}
  Here $\varphi_s$ is the scalar flow in
  \eqref{eq:app-scalar-flow}. Its functions of $x_b$ are
  defined by convergent power series in $s$ on each truncated
  lattice, followed by passage to the inverse limit.
  Notice that $u_s$ denotes the transported multiplication
  operator, rather than a new scalar coordinate on the original
  fiber. 
  
  Reduction of the first identity modulo $u$ conjugates
  $K_{b_s}^{\mathrm F}$ to $\varphi_s(K_b^{\mathrm F})$.
  Thus the distinct eigenvalues of $K^{\mathrm F}$ evolve as
  $\lambda_s=\varphi_s(\lambda)$, with their multiplicities
  unchanged. Along the successive flows parametrizing $W$,
  this gives analytic labels $\lambda_\alpha(w)$.
  
  By Lemma~\ref{lem:splitting}, write
  $\widehat\mathcal{H}_b=\bigoplus_\lambda\mathcal{L}_\lambda$ for the
  $x_b$-stable spectral decomposition. Each $\mathcal{L}_\lambda$
  is preserved by $\varphi_s(x_b)$ and
  $\varphi_s'(x_b)^{\pm1}$ for small $s$. Consequently,
  \eqref{eq:app-integrated-transport} gives
  \begin{equation}\label{eq:app-transported-lattice}
   u_s\mathcal{L}_\lambda=u\mathcal{L}_\lambda,\qquad
   u\,T_s(\mathcal{L}_\lambda)=T_s(u\mathcal{L}_\lambda).
  \end{equation}
  The image $T_s(\mathcal{L}_\lambda)$ is therefore a complete
  $\Bbbk[[u]]$-submodule, stable under $x_{b_s}$.
  Lifting a basis modulo $u$ gives a finite free basis of this
  image; its reduction is the generalized
  $\varphi_s(\lambda)$-eigenspace of $K_{b_s}^{\mathrm F}$.
  These images form a direct sum. The uniqueness in
  Lemma~\ref{lem:splitting} identifies them with the spectral
  lattices at $b_s$. Localizing
  \eqref{eq:app-transported-lattice} also identifies the
  corresponding meromorphic summands under $T_s$.
  
  \medskip
  \noindent\emph{Preservation of the regular-singular factors.}
  Assume now that $\widehat\mathcal{H}_b^{\mathrm{mer}}$ is of
  unramified exponential type. Fix a spectral summand with
  eigenvalue $\lambda$. By Corollary~\ref{cor:leading-exponential},
  it is isomorphic to $\mathcal{E}^{-\lambda/u}\otimes\mathcal{R}$, with
  $\mathcal{R}$ regular singular. Choose a constant-residue basis for
  $\mathcal{R}$ as in \cite[Prop.~3.12 and Rem.~3.20(1)]{PutSinger}.
  Write $v$ for the tuple of induced basis elements of the
  exponential twist and $R$ for the constant residue matrix, so
  \begin{equation}\label{eq:app-initial-normal-form}
   \nabla_{\partial_u}v
   =\left(\frac{\lambda}{u^2}+\frac Ru\right)v,
   \qquad x_bv=(\lambda+uR)v.
  \end{equation}
  On this summand abbreviate $x_b$ and $\partial_{x,b}$ to
  $x$ and $\partial_x$, and put $z:=x-\lambda$.
  Then $z=u(u\partial_u+R)$ in this basis. In particular,
  $z$ strictly raises the $u$-adic order on the
  constant-residue lattice $\mathcal{L}:=\Bbbk[[u]]v$. Since
  $[\partial_x,z]=\operatorname{id}$, \eqref{eq:app-initial-normal-form}
  implies
  \begin{equation}\label{eq:app-centered-residue}
   z\partial_xv=(\partial_xz-\operatorname{id})v=(R-\operatorname{id})v.
  \end{equation}
  
  Set $\lambda_s:=\varphi_s(\lambda)$ and factor the
  centered scalar flow as
  \begin{equation}\label{eq:app-centered-flow}
   \psi_s(z):=\varphi_s(\lambda+z)-\varphi_s(\lambda)
   =a_s z h_s(z),\qquad
   a_s:=\varphi_s'(\lambda)\ne0,\qquad h_s(0)=1.
  \end{equation}
  In the notation of \eqref{eq:app-integrated-transport},
  \begin{equation}\label{eq:app-centered-operators}
   z_s:=x_s-\lambda_s=\psi_s(z),\qquad
   \partial_{x,s}=\psi_s'(z)^{-1}\partial_x,\qquad
   u_s=u\psi_s'(z).
  \end{equation}
  We will find a basis for the scalar action defined by $u_s$
  in which $x_s$ has the same residue matrix $R$.
  
  Because $R$ commutes with $z$ and $z$ strictly raises
  the filtration on $\mathcal{L}$, the following formal operator and
  basis tuple are well-defined:
  \begin{equation}\label{eq:app-corrected-basis}
   H_s(z):=h_s(z)^{R-\operatorname{id}}
   =\exp\bigl((R-\operatorname{id})\log h_s(z)\bigr),\qquad
   w:=H_s(z)v.
  \end{equation}
  The commutator $[\partial_x,z]=\operatorname{id}$ gives
  $[\partial_x,H_s(z)]=H_s'(z)$, with
  \[
  H_s'(z)=(R-\operatorname{id})H_s(z)\frac{h_s'(z)}{h_s(z)}.
  \]
  Combining this with \eqref{eq:app-centered-residue} and
  \eqref{eq:app-centered-flow}, we obtain
  \begin{equation}\label{eq:app-residue-calculation}
  \begin{aligned}
  z_s\partial_{x,s}w
  &=\left(\operatorname{id}+z\frac{h_s'(z)}{h_s(z)}\right)^{-1}
    z\partial_x\bigl(H_s(z)v\bigr)\\
  &=\left(\operatorname{id}+z\frac{h_s'(z)}{h_s(z)}\right)^{-1}
    \bigl(zH_s'(z)v+H_s(z)(R-\operatorname{id})v\bigr)\\
  &=(R-\operatorname{id})w.
  \end{aligned}
  \end{equation}
  Also $[\partial_{x,s},z_s]=\operatorname{id}$ and
  $u_s\partial_{x,s}=\operatorname{id}$, by
  \eqref{eq:app-centered-operators}. Hence
  \eqref{eq:app-residue-calculation} gives
  \begin{equation}\label{eq:app-transported-normal-form}
   z_sw=u_s(\partial_{x,s}z_s)w=u_sRw,
   \qquad x_sw=(\lambda_s+u_sR)w.
  \end{equation}
  
  It remains to check that this operator calculation produces
  an actual $\Bbbk((u))$-basis at $b_s$. By
  \eqref{eq:app-corrected-basis}, $w\equiv v\pmod{u\mathcal{L}}$,
  while \eqref{eq:app-centered-operators} gives, for $k\geq0$,
  \[
  u_s(u^k\mathcal{L})=u^{k+1}\mathcal{L},\qquad
  u_s=a_su\pmod{\text{operators raising the $u$-adic order by two}}.
  \]
  Successive reduction modulo powers of $u$ therefore expresses
  each element of $\mathcal{L}$ uniquely as a formal series in $u_s$
  with coefficients in the $\Bbbk$-span of the entries of $w$.
  The lattice $\mathcal{L}$ is commensurable with the original spectral
  lattice, and \eqref{eq:app-transport-maps} ensures that $T_s$
  is continuous for their $u$-adic topologies. Since
  $T_su_s=uT_s$, the tuple $T_sw$ is a
  $\Bbbk[[u]]$-basis of $T_s(\mathcal{L})$ and a
  $\Bbbk((u))$-basis of the transported summand.
  Applying $T_s$ to \eqref{eq:app-transported-normal-form}
  and dividing by $u^2$ gives
  \begin{equation}\label{eq:app-final-normal-form}
   \nabla_{\partial_u}(T_sw)
   =\left(\frac{\lambda_s}{u^2}+\frac Ru\right)(T_sw).
  \end{equation}
  Thus \eqref{eq:app-final-normal-form} identifies the summand
  at $b_s$ with $\mathcal{E}^{-\lambda_s/u}\otimes\mathcal{R}$: its
  regular-singular factor has the same constant residue matrix
  as in \eqref{eq:app-initial-normal-form}.
  
  Apply this argument to every initial spectral summand and
  then to the successive generating flows parametrizing $W$.
  After shrinking their parameter polydisk, these identifications
  hold throughout $W$ and give \eqref{eq:app-constant-factors}.
  This proves (iii).

\label{rem:compatible-persistence}
  The formal-persistence conclusion also holds for an additional
  F-bundle $(\widetilde\mathcal{H},\widetilde\nabla)$ on the same leaf
  $W$, not necessarily maximal, provided that its action and
  residual endomorphism satisfies
  \begin{equation}\label{eq:app-compatible-action}
   \widetilde\mu(X_p)=p(\widetilde K^{\mathrm F}),
   \qquad p\in\Bbbk[z],\qquad
   X_p=p_\circ(\mathsf{Eu}^{\mathrm F}).
  \end{equation}
  Here $\mathsf{Eu}^{\mathrm F}$ is the Euler field of the original
  maximal F-bundle, and $\widetilde\mu$ and
  $\widetilde K^{\mathrm F}$ are defined by
  \eqref{eq:F-residual-action} for the additional bundle.
  If $\widehat{\widetilde\mathcal{H}}_b^{\mathrm{mer}}$ is of
  unramified exponential type, its regular-singular factors are
  locally constant up to isomorphism along $W$, with eigenvalues
  matched by the flows above.
  
  Indeed, maximality was used to construct the tangent algebra
  and the leaf. Condition~\eqref{eq:app-compatible-action}
  allows us to apply Lemma~\ref{lem:app-transport-operators}
  to the additional bundle. The parallel transport and
  constant-residue calculation then apply without change.
  If the additional bundle is a super vector bundle, all these
  operators are even, and the argument applies to each parity
  separately.
