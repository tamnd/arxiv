<!-- theorem-2: body_md -->
\label{T5}
    Let $M^2$ be a proper biharmonic Lagrangian surface in a 2-dimensional complex space form \(\widetilde{M}^{2}(4 \epsilon)\), where $\epsilon\in\{-1,0,1\}$. If $M^2$ has constant mean curvature, then $M^2$ is flat and $\epsilon = 1$. Precisely, $M^2$ is given by an open part of the Lagrangian tori \eqref{FL}.
<!-- end theorem-2 -->
<!-- proof-1: body_md -->
Since $|H|$ is a non-zero constant, $H$ vanishes nowhere, so
$\mathcal{N} = M^2$ and $a = 2|H|$ is a positive constant on $M^2$.
Then \eqref{F35} gives $\omega_1^2(e_1) = 0$, and hence
\eqref{F3811}--\eqref{E311} reduce to

        \begin{align}   \label{1FU1}
    \begin{cases}
        &e_2(\omega_{1}^{2}(e_{2}))-ab=0, \\
        &(a-c)^2+2b^2+c^2+(\omega_{1}^{2}(e_{2}))^2-5\epsilon=0,\\
        &c\omega_{1}^{2}(e_{2})=0, \\
        &b\omega_{1}^{2}(e_{2})=0.
        \end{cases}
        \end{align}
We claim that $\omega_1^2(e_2) = 0$ on $M^2$. Suppose, to the contrary,
that $\omega_1^2(e_2) \neq 0$ at any point of an open, connected
subset $\mathcal{U}$ of $M^2$. Then the last two equations of
\eqref{1FU1} yield $b = c = 0$ on $\mathcal{U}$. Combined with \eqref{F34}, this
gives $\omega_1^2(e_2) = 0$ on $\mathcal{U}$, a contradiction. Therefore
$\omega_1^2(e_2) = 0$ on $M^2$, and since $\omega_1^2(e_1) = 0$,
relation \eqref{F36} gives $K = 0$. Moreover, the first equation of
\eqref{1FU1} yields $b = 0$, and the second one reduces to

\begin{align}
(a-c)^2+c^2-5\epsilon=0,\label{C1}
\end{align}
which, in particular, forces $\epsilon$ to be positive, that is, $\epsilon = 1$.

Relation \eqref{F36}  implies now
    \begin{align}
ac - 2c^2 + 1 = 0. \label{C2}
        \end{align}
From \eqref{C1} and \eqref{C2} we get
$$
c^2=\frac{7\pm\sqrt{41}}{4} \quad \text{and} \quad a^2=\frac{13\pm\sqrt{41}}{2}.
$$
Consequently, either
$$
a=\sqrt{\frac{13 + \sqrt{41}}{2}}, \quad b=0 \quad \text{and} \quad c= \sqrt{\frac{7 + \sqrt{41}}{4}},
$$
or
$$
a=\sqrt{\frac{13 - \sqrt{41}}{2}}, \quad b=0 \quad \text{and} \quad c= - \sqrt{\frac{7 - \sqrt{41}}{4}}.
$$

We now recall a construction due to Reckziegel \cite{Reckziegel}, which
yields an explicit representation of the proper biharmonic Lagrangian
surface. Any Lagrangian immersion $\phi\colon M^2 \to \mathbb{C}P^2(4)$
can be lifted locally (and globally, if $M^2$ is simply connected) via
the Hopf fibration $\pi\colon \mathbb{S}^5(1) \to \mathbb{C}P^2(4)$ to a Legendrian
immersion $\bar\phi\colon M^2 \to \mathbb{S}^5(1)$ with $\pi \circ \bar\phi = \phi$.
Moreover, by standard arguments (see \cite{Chen2002, Reckziegel}), the
second fundamental form $\bar h$ of $\bar\phi$ in $\mathbb{S}^5(1)$ is the
horizontal lift of the second fundamental form $h$ of $\phi$, that is,
$d \pi \left(\bar h(X,Y)\right ) = h(X,Y)$. Consequently, $\hat\nabla_X Y$ is given by

    \begin{align}
    \hat{\nabla}_XY=  \nabla_XY+\bar{h}(X,Y)-\langle X, Y\rangle \bar{\phi},
 \end{align}
 where $\hat{\nabla}$ is the Levi-Civita connection of $\mathbb R^6\equiv \mathbb C^3$, and $X$ and $Y$ are vector fields tangent to $M^2$.

In our case, since $\omega_1^2 = 0$, the vector fields $e_1$ and
$e_2$ are parallel and commute. Hence there exist local coordinates
$(u, v)$ on $M^2$ such that $\partial_u = e_1$ and $\partial_v = e_2$;
in particular, the metric is $g = \mathrm{d}u^2 + \mathrm{d}v^2$. Using \eqref{F31} with
$b = 0$, we find that the Legendrian lift $\bar\phi$ satisfies the
system of PDEs

        \begin{align}   \label{FC2}
    \begin{cases}
        &\bar{\phi}_{uu}=\mathrm{i}(a-c)\bar{\phi}_{u}-\bar{\phi}, \\
        &\bar{\phi}_{uv}=\mathrm{i}c\bar{\phi}_{v},\\
        &\bar{\phi}_{vv}=\mathrm{i}c\bar{\phi}_{u}-\bar{\phi}.
    \end{cases}
 \end{align}
Integrating \eqref{FC2}, we find that the Legendrian lift $\bar\phi$ is given by
(see also \cite{Sasahara2007})

    \begin{align}
{\bar\phi}(u,v) = \left(
 \sqrt{\frac{c^2}{c^2+1}} \exp({-\frac{\mathrm{i}}{c}u}),\;
 \sqrt{\frac{1}{c^2+1}} \exp({\mathrm{i}cu}) \cos\sqrt{c^2+1}\,v,\right.\nonumber\\
 \left.
 \sqrt{\frac{1}{c^2+1}} \exp({\mathrm{i}cu}) \sin\sqrt{c^2+1}\,v
 \right).\nonumber
        \end{align}
Finally, acting on ${\bar\phi}$ with the linear holomorphic isometry of $\mathbb{R}^6$
$$
(z_1,z_2,z_3)\to\left(z_1,\frac{z_2+\mathrm{i}z_3}{\sqrt2},\frac{z_2-\mathrm{i}z_3}{\sqrt2}\right),
$$
and making a simple change of parameters of $\mathbb{R}^2$, we obtain the expression~\eqref{IntegralCParallelSurface}. The new parameters are still denoted by $(u,v)$. Moreover, the Hopf cylinder over $\phi_{\pm}$ in $\mathbb{S}^5(1)$, that is, $\pi^{-1}\left(\pi\left(\phi_{\pm}\right)\right)$ is an open part of
\begin{align}
    \pi\left(\mathbb{S}^1\left(\sqrt{\frac{9\pm\sqrt{41}}{20}}\right)\times \mathbb{S}^1\left(\sqrt{\frac{11\mp\sqrt{41}}{40}}\right)\times \mathbb{S}^1\left(\sqrt{\frac{11\mp\sqrt{41}}{40}}\right)\right), \nonumber
\end{align}
where $\pi: \mathbb{S}^5(1) \to \mathbb{C}P^2(4)$ is the Hopf fibration.

<!-- lemma-2: body_md -->
\label{le33}
The functions $b$ and $c$ satisfy $b^2c^2>0$ on an open, dense subset of $\mathcal{N}$.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
We argue by contradiction and assume that
    $b^2c^2=0$ on an open, connected subset $\mathcal{U}$ of $\mathcal{N}$. The proof then splits into the following cases.

    \textbf{Case A.} $b=c=0$ on $\mathcal{U}$. In this case, it follows from \eqref{E310} and \eqref{E311} that \begin{align*}
        \frac{3}{2}ae_1(a)=0,\quad
        \frac1 2ae_2(a)=0.
    \end{align*} Thus, $e_1(a)=e_2(a)=0$ on $\mathcal{U}$, so $\operatorname{grad}a=0$ there, contradicting $\operatorname{grad}a\ne0$ throughout $\mathcal{N}$.

    \textbf{Case B.} $b=0$ on $\mathcal{U}$ and $c\ne0$ at any point of an open, connected subset of $\mathcal{U}$,  redenoted by $\mathcal{U}$. Then, we obtain from  \eqref{F33}, \eqref{F34}  and \eqref{F35} that
    \begin{align}
        &e_2(c)=3c\omega_1^2(e_1)=\frac{3c}{a}e_2(a),\label{1}\\
        &e_1(c)=(a-3c)\omega_1^2(e_2).\label{3}
    \end{align}
    By \eqref{eq:3.1} we have $ac-2c^2=K-\epsilon$ and differentiating this further yields $$ce_i(a)+(a-4c)e_i(c)=0\quad \text{for}\quad i=1,2,$$ which together with \eqref{1} and \eqref{3} provides

Fix the sign and suppress the subscripts. The three phases in
    \eqref{eq:hopf-saturation} are given by the linear transformation
    \begin{equation}\label{eq:phase-matrix}
        A=\begin{pmatrix}
            -\alpha&-\alpha&-1\\
            1&0&-1\\
            0&1&-1
        \end{pmatrix},
        \qquad \det A=-(1+2\alpha)=-\frac1{r^2}\ne0.
    \end{equation}


    For $j=1,2$, write $X_j=(u_j,v_j,t_j)\in\mathbb R^3$ and set
    \[
    \delta u=u_2-u_1,\qquad
    \delta v=v_2-v_1,\qquad
    \delta t=t_2-t_1.
    \]
    Since $r,s>0$, we have
    \begin{equation}\label{eq:phase-coincidences}
        \Phi(X_1)=\Phi(X_2)
        \quad\Longleftrightarrow\quad
        A(X_2-X_1)\in2\pi\mathbb Z^3.
    \end{equation}
    Consequently, the full period lattice is $2\pi A^{-1}\mathbb Z^3$.
    Writing $A(X_2-X_1)=2\pi(m,n,p)^{\mathsf T}$, we obtain
    \[
    -\alpha(\delta u+\delta v)-\delta t=2\pi m,\qquad
    \delta u-\delta t=2\pi n,\qquad
    \delta v-\delta t=2\pi p
    \]
    for $m,n,p\in\mathbb Z$. Solving these equations gives
    \begin{equation}\label{eq:period-solution}
        \begin{aligned}
            \delta u&=2\pi\left(-r^2m+(1-s^2)n-s^2p\right),\\
            \delta v&=2\pi\left(-r^2m-s^2n+(1-s^2)p\right),\\
            \delta t&=-2\pi\left(r^2m+s^2(n+p)\right).
        \end{aligned}
    \end{equation}
    Thus
    \[
    (\delta u,\delta v,\delta t)
    =m\lambda_1+n\lambda_2+p\lambda_3,
    \]
    which proves \eqref{eq:full-period-lattice}. In particular, $\Lambda$
    is a rank-three lattice, and \eqref{eq:phase-coincidences} shows that
    $\overline\Phi$ is injective.


    A direct calculation gives
    \begin{align*}
        \Phi^*\langle\cdot,\cdot\rangle_{\mathbb{S}^5}
        &=r^2\left(\alpha(\mathrm{d}u+\mathrm{d}v)
        +\mathrm{d}t\right)^2
        +s^2(\mathrm{d}u-\mathrm{d}t)^2
        +s^2(\mathrm{d}v-\mathrm{d}t)^2\\
        &=s^2(\mathrm{d}u^2+\mathrm{d}v^2)
        +\frac{s^4}{r^2}(\mathrm{d}u+\mathrm{d}v)^2
        +\mathrm{d}t^2=g_{M\times \mathbb{R}},
    \end{align*}
    where the mixed terms involving $\mathrm{d}t$ vanish because

    \qquad
    B=\begin{pmatrix}1+\alpha&\alpha\\\alpha&1+\alpha\end{pmatrix}.
    \]

    \[
    \det B=1+2\alpha=\frac1{r^2},\qquad

    =\langle\phi_v,\mathrm{i}\phi\rangle=-r^2\alpha+s^2=0,

    f^*\langle\cdot,\cdot\rangle_{\mathbb{C}P^2(4)}=\phi^*\langle\cdot,\cdot\rangle_{\mathbb{S}^5}=g_M.
    \]
    It follows that $\overline f$ is an isometric Lagrangian immersion.
    Compactness and injectivity imply that it is an embedding. Finally,
    $\pi\circ\Phi(u,v,t)=f(u,v)$.

\label{rem:legendrian-periods}
    The lift $\phi_\pm$ is singly periodic, whereas its Hopf projection
    $\pi\circ\phi_\pm$ is doubly periodic. Since
    $\alpha=(7\mp\sqrt{41})/4$ is irrational, the periods of $\phi_\pm$
    are precisely $(2\pi m,-2\pi m)$, $m\in\mathbb Z$. Thus $\phi_\pm$
    descends to an injective isometric Legendrian immersion of the
    complete flat cylinder
    \[
    \left(\mathbb R^2/L,g_M\right),
    \qquad L=\mathbb Z(2\pi,-2\pi).
    \]
    This cylinder is horizontal with respect to the Hopf
    fibration, since
    \[
    \langle(\phi_\pm)_u,\mathrm{i}\phi_\pm\rangle
    =\langle(\phi_\pm)_v,\mathrm{i}\phi_\pm\rangle=0.
    \]
    In contrast, $\partial_t\Phi_\pm=-\mathrm{i}\Phi_\pm$ is tangent to the Hopf
    fibres. The cylinder immersion is not an embedding. Its image is
    dense in $\mathbb{S}^1(r)\times \mathbb{S}^1(s)\times \mathbb{S}^1(s)$, as follows from the
    irrationality of $\alpha$.
