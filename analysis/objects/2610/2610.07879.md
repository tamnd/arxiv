<!-- theorem-1: body_md -->
Let $(M^4, g)$ admit a smooth hypersurface-orthogonal timelike geodesic
congruence with spacelike hypersurfaces $\Sigma_\tau$. Suppose that the
initial spatial geometry satisfies the scale-invariant lower volume bound
\begin{equation}
\operatorname{Vol}_0(B_0(p,r)) \geq \kappa_0 r^3, \qquad 0 < r \leq r_0,
\label{eq:38}
\end{equation}
for some $\kappa_0 > 0$.

Assume that along the region swept out by the congruence,
\begin{equation}
R_{\mu\nu}u^\mu u^\nu \leq R_0, \qquad \sigma^2 \leq \Sigma_0^2,
\label{eq:39}
\end{equation}
and
\begin{equation}
|\theta| \leq \Theta_0, \qquad |\sigma|_h \leq \Sigma_0.
\label{eq:40}
\end{equation}

Let $\Theta$ solve
\begin{equation}
\dot\Theta = -\frac{1}{3}\Theta^2 - (R_0 + \Sigma_0^2), \qquad \Theta(0) = \theta_0,
\label{eq:41}
\end{equation}
and let $\tau_c$ denote the first finite blow-up time of the comparison
solution, if such a time exists.

Define
\begin{equation}
K = \frac{\Theta_0}{3} + \Sigma_0.
\label{eq:42}
\end{equation}

Then, for every
\begin{equation}
0 \leq \tau < \tau_c
\label{eq:43}
\end{equation}
and every radius satisfying
\begin{equation}
0 < r \leq r_0 e^{K\tau},
\label{eq:44}
\end{equation}
one has
\begin{equation}
\operatorname{Vol}_\tau(B_\tau(p_\tau, r)) \geq \kappa_{\mathrm{L}}(\tau) r^3,
\label{eq:45}
\end{equation}
where
\begin{equation}
\kappa_{\mathrm{L}}(\tau) = \kappa_0 e^{-3K\tau} \exp\!\left[\int_0^\tau \Theta(s)\,ds\right].
\label{eq:46}
\end{equation}

In particular,
\begin{equation}
\kappa_{\mathrm{L}}(\tau) > 0
\label{eq:47}
\end{equation}
for every finite $\tau < \tau_c$.

<!-- proof-1: body_md -->
Fix $p \in \Sigma_0$, $\tau < \tau_c$, and a radius $r$ satisfying
eq.~\eqref{eq:44}. Define
\begin{equation}
\rho = e^{-K\tau} r.
\label{eq:48}
\end{equation}

The radius restriction gives
\begin{equation}
\rho \leq r_0.
\label{eq:49}
\end{equation}

Therefore, the initial non-collapse assumption implies
\begin{equation}
\operatorname{Vol}_0(B_0(p,\rho)) \geq \kappa_0 \rho^3 = \kappa_0 e^{-3K\tau} r^3.
\label{eq:50}
\end{equation}

By the distance distortion estimate, eq.~\eqref{eq:37},
\begin{equation}
F_\tau(B_0(p,\rho)) \subset B_\tau(p_\tau, r).
\label{eq:51}
\end{equation}

Consequently,
\begin{align}
\operatorname{Vol}_\tau(B_\tau(p_\tau, r)) &\geq \operatorname{Vol}_\tau(F_\tau(B_0(p,\rho))) \notag \\
&\geq \exp\!\left(\int_0^\tau \Theta(s)\,ds\right) \operatorname{Vol}_0(B_0(p,\rho)).
\label{eq:52}
\end{align}

Using eq.~\eqref{eq:50}, we obtain
\begin{equation}
\operatorname{Vol}_\tau(B_\tau(p_\tau, r)) \geq \kappa_0 e^{-3K\tau} \exp\!\left(\int_0^\tau \Theta(s)\,ds\right) r^3.
\label{eq:53}
\end{equation}

This is precisely eq.~\eqref{eq:45} with
\begin{equation*}
\kappa_{\mathrm{L}}(\tau) = \kappa_0 e^{-3K\tau} \exp\!\left(\int_0^\tau \Theta(s)\,ds\right).
\end{equation*}

Since $\Theta$ is finite for every $\tau < \tau_c$, the exponential factor
is strictly positive. Hence $\kappa_{\mathrm{L}}(\tau) > 0$.
