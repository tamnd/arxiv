\label{thm:main}

<!-- definition-1: body_md -->
\label{def:echoing}
Let $\mathcal{A}$ be a finite alphabet.
An infinite word $w = w_0w_1\cdots \in \mathcal{A}^{\mathbb{N}}$ is called \emph{echoing} if, for each $\varepsilon>0$, there exist $N \in \mathbb{N}$, unbounded sequences of natural numbers $(r_{k})_{k \in \mathbb{N}}$, $(s_{k})_{k \in \mathbb{N}}$, and sets $\mathcal{I}_{k} \subset \{0,1,\dots,s_{k}{-}1\}$, with the following properties for all $k \in \mathbb{N}$:
\begin{enumerate}[\upshape (i)]
\itemsep1ex
\item \label{i:241}
$\{0 \le i < s_{k} \,:\, w_i \ne w_{i+r_{k}}\} \subset \mathcal{I}_{k}$,                                                                                     
\item \label{i:242}
 $r_{k} + | \mathcal{I}_{k} | < \varepsilon\, s_{k}$,
\item \label{i:243}
$\mathcal{I}_{k}$ is a union of at most $N$ intervals of natural numbers. 
\end{enumerate}
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:strongechoing}
An echoing word $w = w_0w_1\cdots \in \mathcal{A}^{\mathbb{N}}$ with $\mathcal{A} \subset \overline{\mathbb{Q}}$ is called \emph{$\beta$-echoing} for $\beta \in \overline{\mathbb{Q}} \setminus \{0\}$ if, for each $\varepsilon > 0$, the sequences $(r_{k})_{k \in \mathbb{N}}$ and $(s_{k})_{k \in \mathbb{N}}$ in Definition~\ref{def:echoing} can be chosen so that there exist sequences of natural numbers $(l_k)_{k\in\mathbb{N}}$, $(p_{k})_{k\in\mathbb{N}}$ and $(q_{k})_{k\in\mathbb{N}}$ with the following properties:
\begin{enumerate}[\upshape (i)]
\itemsep1ex
\setcounter{enumi}{3}
\item \label{eq:upper bound of p,q}
$0 \le l_k \le p_{k} \le q_{k} < s_{k}/2$  for all $k \in \mathbb{N}$,  $\lim_{k\to\infty} l_k/\log s_k = \infty$, 
\item \label{eq:non vanishing}
$\sum_{i=p_{k}}^{q_{k}} w_i \beta^{-i} \ne \sum_{i=p_{k}}^{q_{k}} w_{i+r_{k}} \beta^{-i}$ for all $k \in \mathbb{N}$, 
\item \label{eq:expanding gap}
$w_i = w_{i+r_{k}}$ for all integers $i \in [p_{k} {-} l_k, p_{k}) \cup (q_{k}, q_{k} {+} l_k]$, $k \in \mathbb{N}$.
\end{enumerate}

If $w$ is $\beta$-echoing for all $\beta \in \overline{\mathbb{Q}}$ with $|\beta| > 1$, then $w$ is called \emph{strong echoing}.

If moreover $X_{\boldsymbol{\sigma}}$ is dendric, then each $w \in X_{\boldsymbol{\sigma}}$ is strong echoing.

<!-- proposition-1: body_md -->
\label{prop:p-adic Subspace Theorem}
    Let $K$ be a number field and $S$ a finite set of places with $\mathcal{M}_\infty(K) \subset S \subset \mathcal{M}(K)$. Let 
    $n \geq 2$, $\varepsilon>0$. Let $L$ be a non-zero linear form in $n$ variables with coefficients in $\overline{\mathbb{Q}}$.
    Let $\nu_{0} \in S$ be a distinguished place. 
    Let $1 \leq j_{0} \leq n$ be distinguished index such that $x_{j_{0}}$ has non-zero coefficient in~$L$.

    \begin{equation}\label{eq:cor:p-adic Subspace Theorem}
|L(\mathbf{x})|_{\nu_{0}}\prod_{\substack{(\nu,j) \in S\times\{1,\dots,n\} \\ (\nu,j) \neq (\nu_{0},j_{0})}}|x_{j}|_{\nu}  \leq H(\mathbf{x})^{ - \varepsilon}
    \end{equation}
    lies in finitely many proper subspaces of~$K^{n}$.

\label{lem:Common root}

<!-- lemma-2: body_md -->
\label{lem:Expanding Gap}
    Let $w \in \mathcal{A}^{\mathbb{N}}$ be an echoing word with $\mathcal{A} \subset \overline{\mathbb{Q}}$.
    For each $\varepsilon > 0$, the constant $N \in \mathbb{N}$ and the sequences $(r_k), (s_k), (\mathcal{I}_k)$ of Definition~\ref{def:echoing} can be chosen such that $\lim_{k\to\infty} r_k = \infty$ and $\mathcal{I}_{k} = \bigcup_{i=1}^{N} I_{k,j}$ with non-empty ordered intervals of natural numbers $I_{k,1} < I_{k,2} < \cdots < I_{k,N}$ satisfying $0 \in I_{k,1}$ and
    \begin{equation} \label{e:dIk}
        \lim_{k\to\infty} \frac{d(I_{k,j},I_{k,j+1})}{\log s_{k}} = \infty \mbox{ for all } 1 \leq j < N.
    \end{equation}
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
Let $\varepsilon > 0$.
Since $w$ is echoing, there exist $N'$, $r'_k$, $s'_k$ and $\mathcal{I}'_{k}$ with the following properties for all $k \in \mathbb{N}$: $\{0 \le i < s'_{k} \,:\, w_i \ne w_{i+r'_{k}}\} \subset \mathcal{I}'_{k}$,  $r'_{k} + |\mathcal{I}'_{k}| < (\varepsilon/2)\, s'_{k}$, and $\mathcal{I}'_{k} = \bigcup_{i=1}^{N'_{k}} I'_{k,i}$ with non-empty ordered intervals of natural numbers $I'_{k,1} < I'_{k,2} < \cdots < I'_{k,N'_{k}}$ such that $N'_k \le N'$.
W.l.o.g., assume that $0 \in I'_{k,1}$.

Since $s'_{k}$ is unbounded, we can assume w.l.o.g.\ that $s'_{k} > (2N'/\varepsilon)^2$ for all $k \in \mathbb{N}$.
For $k \in \mathbb{N}$, $1 \le j < N'_k$, if $d(I'_{k,j},I'_{k,j+1}) < (s'_{k})^{1/2}$, then we merge $I'_{k,j}$ and $I'_{k,j+1}$ into a new interval $[\min I'_{k,j}, \max I'_{k,j+1}] \cap \mathbb{N}$.
In this way, we obtain intervals $I''_{k,j}$, $1 \le j < N''_k \le N'_k$, such that $\mathcal{I}''_k := \bigcup_{j=1}^{N''_k} \mathcal{I}''_{k,j} \supset \mathcal{I}'_k$, with
\[
|\mathcal{I}''_k| < |\mathcal{I}'_k| + (N'_k-1) (s'_{k})^{1/2} < (\epsilon/2) s'_k.
\]
By taking a subsequence of $k \in \mathbb{N}$ where $N''_k$ is constant, we have thus found sequences, $r_k$, $s_k$, $\mathcal{I}_k = \bigcup_{j=1}^N I_{k,j}$ satisfying Definition~\ref{def:echoing}, with $0 \in I_{k,1} < I_{k,2} < \cdots < I_{k,N}$ and $d(I_{k,j},I_{k,j+1}) \ge s_k^{1/2}$.
Since $s_{k}^{1/2}/\log(s_{k}) \to \infty$ when $k \to \infty$, we have 
\[
    \lim_{k \to \infty} \frac{d(I_{k,j},I_{k,j+1})}{\log(s_{k})} = \infty
\]
for all $1 \leq j < N$.

    Let $\beta \in \overline{\mathbb{Q}}$ with $|\beta|>1$.

<!-- proof-2: body_md -->
\label{proof:Echoing imply non-algebraic}
    Suppose that $(w)_{\beta} \neq 0$ is algebraic.
    We will show that $(w)_{\beta} \in \mathbb{Q}(\beta,\mathcal{A})$.
    Let $S$ be the finite set containing all the Archimedean places of $\mathbb{Q}(\beta,\mathcal{A})$ and all finite places $\nu$ of $\mathbb{Q}(\beta,\mathcal{A})$ corresponding to prime ideals $\mathfrak{p}$ of the ring of integers $\mathcal{O}_{\mathbb{Q}(\beta,\mathcal{A})}$ such that $|\beta|_{\nu} \neq 1$ or $|a|_{\nu} \neq 1$ for some $a \in \mathcal{A}$.
    Let $\nu_{0} \in S$ be the Archimedean place corresponding to the identity embedding $\mathbb{Q}(\beta,\mathcal{A}) \hookrightarrow \mathbb{C}: \beta \mapsto \beta$, $a \mapsto a$ for all $a \in \mathcal{A}$.
    We have $|\beta|_{\nu_0} = |\beta|^{1/\deg(\beta)}$ if $\nu_0$ is real infinite and $|\beta|_{\nu_0} = |\beta|^{2/\deg(\beta)}$ if $\nu_0$ is complex infinite, thus $|\beta|_{\nu_0} > 1$.
    Let $\kappa = \max_{\nu \in S, a,b \in \mathcal{A}}(|\beta|_{\nu},|a|_{\nu}, |a-b|_{\nu})$.
    Let $\varepsilon = \frac{\log(|\beta|_{\nu_0})}{4|S| \log(\kappa)}$.
    Let $r_k$, $s_k$, $\mathcal{I}_{k} = \bigcup_{j=1}^N I_{k,j}$ be as in Lemma~\ref{lem:Expanding Gap}, and set $t_k := \max I_{k,1}$.
    By the unboundedness of $r_k$ and $s_k$ and taking a subsequence of $k \in \mathbb{N}$, w.l.o.g., we can assume that $r_k$ and $s_k$ are strictly increasing.
    For each $k \in \mathbb{N}$, we define $\mathbf{z}_{k} = (z_{k,1},\dots,z_{k,N+2}) \in (\mathcal{O}_{S})^{N+2}$ by
    \begin{equation}
    \begin{split}
        z_{k,1} &:= \beta^{r_{k}}, \quad 
        z_{k,2} := 1, \\
        z_{k,3} &:= \sum_{i=0}^{r_{k}-1} w_{i} \beta^{r_{k}-i} +  \sum_{i \in I_{k,1}} (w_{r_{k}+i} - w_{i})\beta^{-i}, \\
        z_{k,j+2} &:= \sum_{i \in I_{k,j}} (w_{r_{k}+i} - w_{i})\beta^{-i} \quad (j = 2,\dots,N).
        \end{split}
    \end{equation}
    By taking a subsequence of $k \in \mathbb{N}$, we can assume w.l.o.g.\ that there exists $J \subset \{2,\dots,N\}$ such that $z_{k,j+2} \neq 0$ if and only if $j \in J$. 
    Now, define the linear form $L$ over $\mathbf{x} = (x_{1}, \dots, x_{|J|+3})$ by
    \begin{equation}\label{eq:Linear Combination}
        L(\mathbf{x}) := (w)_{\beta} \, x_{1} - (w)_{\beta} \, x_{2} -x_{3}-\sum_{j \in J}  x_{j+2}.
    \end{equation}
    Then
    \begin{equation*}
    \begin{split} 
        L(\mathbf{z}_{k})
        & = \beta^{r_{k}} (w)_{\beta} - (w)_{\beta} - \sum_{i=0}^{r_{k}-1} w_{i} \beta^{r_{k}-i} -  \sum_{i \in \mathcal{I}_{k}} (w_{r_{k}+i} - w_{i})\beta^{-i} \\
        &= \sum_{i=s_{k}}^{\infty} (w_{r_{k}+i} - w_{i})\beta^{-i}
    \end{split} 
    \end{equation*}
    and thus
    \begin{equation} \label{eq:Archimedean upper bound}
    |L(\mathbf{z}_{k})|_{\nu_0} \ll |\beta|_{\nu_0}^{-s_{k}}.
    \end{equation}    
    For the other places $\nu \in S \setminus \{ \nu_{0} \}$, using the product formula $\prod_{\nu \in S} |\beta|_{\nu} = 1$, we obtain that
    \begin{equation}\label{eq:a1 absolute value v0}
        \prod_{\nu \in S \setminus \{ \nu_{0} \}} | z_{k,1} |_{\nu} = |\beta|_{\nu_{0}}^{-r_{k}}.
    \end{equation}
    Recall that $\varepsilon s_{k} > r_{k}+t_{k}$, with $\varepsilon = \frac{\log(|\beta|_{\nu_0})}{4|S| \log(\kappa)}$.
    By the product formula, we also have
    \begin{equation}\label{eq:a3 other absolute value}
        \prod_{\nu \in S} |z_{k,3}|_{\nu} = \prod_{\nu \in S} |\beta^{t_k} z_{k,3}|_{\nu}
        \leq \prod_{\nu \in S} \sum_{i=0}^{r_{k}+t_{k}} \kappa^{i+1}
        \ll \prod_{\nu \in S} \kappa^{r_{k}+t_{k}}
        = \kappa^{(r_{k}+t_{k})|S|}
        = |\beta|_{\nu_0}^{s_{k}/4}.
    \end{equation}
    Similarly, using that $|\mathcal{I}_{k}| < \varepsilon s_{k}$ for all $k \in \mathbb{N}$, we obtain that
    \begin{equation}
        \label{eq:other absolute value}
        \prod_{j \in J} \prod_{\nu \in S} |z_{k,j+2}|_{\nu}
        \leq \prod_{j \in J} \prod_{\nu \in S} \sum_{i =0}^{|I_{k,j}|-1} \kappa^{i+1} \ll |\beta|_{\nu_0}^{s_{k}/4}.
    \end{equation}
    
    In order to apply Proposition~\ref{prop:p-adic Subspace Theorem}, we use the above inequalities \eqref{eq:Archimedean upper bound}--\eqref{eq:other absolute value} and the trivial equation $\prod_{\nu \in S} |z_{k,2}|_{\nu} = 1$ to obtain
    \begin{equation}\label{eq:asymtotic subspace theorem}
        |L(\mathbf{z}_{k})|_{\nu_{0}} \prod_{\substack{(\nu,j) \in S\times \{1,2,3\} \\ (\nu,j) \neq (\nu_{0} , 1)}}|z_{k,j}|_{\nu} \prod_{(\nu,j) \in S\times J}|z_{k,j+2}|_{\nu}  
        \ll |\beta|_{\nu_{0}}^{-s_{k}} |\beta|_{\nu_{0}}^{-r_{k}} |\beta|_{\nu_{0}}^{s_{k}/2}.
    \end{equation} 
    Since $r_{k} \to \infty$ as $k \to \infty$, we have

\label{proof:Echoing}
Let $\boldsymbol{\sigma} = (\sigma_{n})_{n\in\mathbb{N}}$ be as in Lemma~\ref{lem:density0}.
Let $h \ge 1$ be such that $|\sigma_{[0,h)}(a)| \ge s{+}r$ for all $a \in \mathcal{A}$, and by the recurrence, let $(n_k)_{k\in\mathbb{N}}$ be an increasing sequence of natural numbers such that $\sigma_{[n_k,n_k+h)} = \sigma_{[0,h)}$ for all $k \in \mathbb{N}$. 

Let $w \in X_{\boldsymbol{\sigma}}$.
Then there exist $w_0^{(k)}w_1^{(k)}\cdots \in X_{T^{n_k}\boldsymbol{\sigma}}$, $0 \le j_k < |\sigma_{[0,n_k)}(w_0^{(k)})|$, such that 
\[
w = T^{j_k} \sigma_{[0,n_k)}(w_0^{(k)}w_1^{(k)}\cdots).
\]
Let 
\begin{equation} \label{e:rk}
r_{k} := \lVertM_{\sigma_{[0,n_k)}} \lfloorr\mathbf{f}\rceil\rVert_1
\end{equation}
(where we recall that $\lfloorr\mathbf{f}\rceil$ is the nearest integer vector to~$r\mathbf{f}$), 
\begin{equation} \label{e:sk}
s_k := |\sigma_{[0,n_k)}(w^{(k)}_{[0,s)})| - j_k, 
\end{equation}
and 
\[
\mathcal{I}_k := \bigcup_{i\in\mathcal{J}_{r}(w^{(k)})\cap[0,s)} \big[|\sigma_{[0,n_k)}(w^{(k)}_{[0,i)})|{-}j_k, |\sigma_{[0,n_k)}(w^{(k)}_{[0,i+1)})|{-}j_k\big) \cap \mathbb{N},
\]
with $\mathcal{J}_r$ defined in \eqref{e:Jr}.
If $i \notin \mathcal{J}_{r}(w^{(k)}) \cap [0,s)$, then $w^{(k)}_i = w^{(k)}_{i+r}$ and
\[
r_k = \lVertM_{\sigma_{[0,n_k)}} \mathbf{l}(w^{(k)}_{[i,i+r)})\rVert_1 = \lVert\mathbf{l}(\sigma_{[0,n_k)}(w^{(k)}_{[i,i+r)}))\rVert_1 = |\sigma_{[0,n_k)}(w^{(k)}_{[i,i+r)})|.
\]
This implies that
\[ 
    \{0 \le i<s_k : w_i \ne w_{i+r_{k}}\} \subset \mathcal{I}_k,
\]
and $\mathcal{I}_k$ is a union of at most $s$ intervals of integers.

Since $|\sigma_{[0,h)}(a)| \ge s{+}r$ for all $a \in \mathcal{A}$, $w_{[0,s+r)}^{(k)}$ is a factor of $\sigma_{[0,h)}(ab)$ for some $a,b \in \mathcal{A}$.
Consider first the case that $w_{[0,s+r)}^{(k)}$ is a factor of $\sigma_{[0,h)}(a)$.
Then $w_{[0,s+r)}^{(k)}$ is the prefix of some $\tilde{w} = \tilde{w}_0\tilde{w}_1\cdots \in X_{\boldsymbol{\sigma}}$.
Then we have 
\[
\begin{gathered}
r_{k} \le r \max_{a\in\mathcal{A}} |\sigma_{[0,n_k)}(a)| \le \varepsilon\, s \min_{a\in\mathcal{A}} f_a \max_{a\in\mathcal{A}} |\sigma_{[0,n_k)}(a)| \le \varepsilon\,s \sum_{a\in\mathcal{A}} f_a |\sigma_{[0,n_k)}(a)|, \\
|\mathcal{I}_k| \le \sum_{a \in A} |\{ i \in \mathcal{J}_{r}\cap[0,s) : \tilde{w}_i = a \}|\ |\sigma_{[0,n_k)}(a)| \le \varepsilon\,s \sum_{a\in\mathcal{A}} f_a\, |\sigma_{[0,n_k)}(a)|, \\
s \sum_{a\in\mathcal{A}} f_a |\sigma_{[0,n_k)}(a)| \le 3 \sum_{a\in\mathcal{A}} |\tilde{w}_{[1,s/2)}|_a \  |\sigma_{[0,n_k)}(a)| < 3 s_k.
\end{gathered}
\]

Consider now the case that $w_{[0,s')}^{(k)}$ is a suffix of $\sigma_{[0,h)}(a)$ for some $s' < s+r$ and $w_{[s',s+r)}^{(k)}$ is a prefix of $\sigma_{[0,h)}(b)$.
Then $w_{[0,s')}^{(k)}$ is the prefix of some $w' \in X_{\boldsymbol{\sigma}}$ and $w_{[s',s+r)}^{(k)}$ is the prefix of some $\tilde{w} \in X_{\boldsymbol{\sigma}}$.
We have
\begin{align*}
    |\mathcal{I}_k| &\le \sum_{a \in A} \Big( \big| \{ i \in \mathcal{J}_{r}\cap[0,s) : \tilde{w}_i = a \} \big|+ r + \big| \{ i \in \mathcal{J}_{r}\cap[0,s) : w'_i = a \} \big|\Big) |\sigma_{[0,n_k)}(a)| \\ 
    &\le 3 \varepsilon\,s \sum_{a\in\mathcal{A}} f_a\, |\sigma_{[0,n_k)}(a)|.
\end{align*} 
If $s/2 \le s'$, then $w'_{[1,s/2)}$ is a factor of $w_{[0,s')}^{(k)}$, therefore we have
\[
    s \sum_{a\in\mathcal{A}} f_a |\sigma_{[0,n_k)}(a)| \le 3 \sum_{a\in\mathcal{A}} | w'_{[1,s/2)}|_a \, |\sigma_{[0,n_k)}(a)| < 3 s_k.
\]
If $s/2 > s'$, then $\tilde{w}_{[1,s/2)}$ is a factor of $w_{[s',s+r)}^{(k)}$ and we can apply the above argument to get the same bound.
Therefore, 
\[ 
r_{k} + |\mathcal{I}_k| < 9 \varepsilon s_k.
\]
This proves that $w$ is echoing.

<!-- lemma-7: body_md -->
\label{lem:returnwords}
Let $\mathcal{A} \subset \mathbb{C}$, $u \in F_{\mathcal{A}}$, $L \subset F_{\mathcal{A}}$, $\beta \in \mathbb{C}$ not a root of unity.
Assume that $\mathrm{ev}_\beta(P(uv)) = \mathrm{ev}_\beta(P(vu))$ for all $v \in L$.
Then $\mathrm{ev}_\beta(P(w)) = \mathrm{ev}_\beta(P(w'))$ for all $w,w' \in \langle uL\rangle$ such that $\ell(P(w)) = \ell(P(w'))$, in particular $|\langle uL\rangle \cap \mathcal{A}| \le 1$.
<!-- end lemma-7 -->
<!-- proof-7: body_md -->
First note that $\mathrm{ev}_\beta(f*g*h) = \mathrm{ev}_\beta(f*g'*h)$ if $\mathrm{ev}_\beta(g) = \mathrm{ev}_\beta(g')$ and $\ell(g) = \ell(g')$.
Therefore, we have $\mathrm{ev}_\beta(P(w)) = \mathrm{ev}_\beta(P(w'))$ for all pairs $w,w'$ that are obtained by replacing $uv$ by~$vu$, $v \in L$.
In particular, we have $\mathrm{ev}_\beta(P(uw)) = \mathrm{ev}_\beta(P(wu))$ for all $w \in \langle uL\rangle$, the subgroup of $F_{\mathcal{A}}$ generated by $\{uv : v \in L\}$, i.e., 
\[
\mathrm{ev}_\beta(P(u))\, \beta^{\ell(P(w))} + \mathrm{ev}_\beta(P(w)) = \mathrm{ev}_\beta(P(w))\,  \beta^{\ell(P(u))} + \mathrm{ev}_\beta(P(u)),
\]
thus
\[
\mathrm{ev}_\beta(P(w)) = \frac{\beta^{\ell(P(w))}-1}{\beta^{\ell(P(u))}-1}\, \mathrm{ev}_\beta(P(u)).
\]
This shows that $\mathrm{ev}_\beta(P(w)) = \mathrm{ev}_\beta(P(w'))$ for all $w, w' \in \langle uL\rangle$ with $\ell(P(w)) = \ell(P(w'))$.
For $a \in \mathcal{A}$, we have $\mathrm{ev}_\beta(P(a)) = a$ and $\ell(P(a)) = 1$, thus $a, b \in \langle uL\rangle \cap \mathcal{A}$ implies $a = b$.

<!-- lemma-8: body_md -->
\label{lem:expanding gap on return words}
Let $\boldsymbol{\sigma}$ be a primitive sequence of unimodular substitutions such that $X_{\boldsymbol{\sigma}}$ is dendric.
Let $n,r \ge 1$, $v = v_{[0,r)} \in \mathcal{A}^r$, and $w \in X_{\boldsymbol{\sigma}}$ be such that $\sigma_{[0,n)}(v\, v_0)$ is a factor of~$w$.
Then there exists a factor $u$ of $\sigma_{[0,n)}(v\,v_0)$ with $|u| = |\sigma_{[0,n)}(v)|$ such that, for every $j \in \mathbb{N}$, $w_{[j,j+|u|)} = u$ implies 
\[
w_{i} = w_{i+|u|} \quad \text{for all integers } i \in \bigg[j, j+ \frac{|\sigma_{[0,n)}(v_0)|}{|\mathcal{A}|}\bigg)\cdot
\]
If moreover $\tilde{w} = \tilde{w}_0 \tilde{w}_1 \cdots \in \mathcal{A}^\mathbb{N}$ is such that $\sigma_{[0,n)}(\tilde{w}_{[0,s/8)})$ is a factor of~$w$ and $\tilde{w}_{[0,s/8)}$ contains a set of return words of~$v\,v_0$ with abelianizations spanning~$\mathbb{R}^{\mathcal{A}}$, then all elements of $R_{X_{\boldsymbol{\sigma}}}(u)$ are factors of $\sigma_{[0,n)}(\tilde{w}_{[0,s/8)})$.
<!-- end lemma-8 -->
<!-- proof-8: body_md -->
Since $X_{\boldsymbol{\sigma}}$ is dendric, the number of factors of length~$k$ is $(|\mathcal{A}|{-}1)k{+}1$ for all $k\in\mathbb{N}$.
Therefore, for all $t \in \mathbb{N}$, at most $(|\mathcal{A}|{-}1)t$ factors of length $k$ have more than one right extension of length~$t$.
Suppose that all factors of $\sigma_{[0,n)}(v\,v_0)$ of length $\tilde{r} = |\sigma_{[0,n)}(v)|$ have more than one right extension of length $t = \lfloor|\sigma_{[0,n)}(v_0)|/|\mathcal{A}|\rfloor$.
Then one of these factors appears twice in $\sigma_{[0,n)}(v\,v_0)$, at positions bounded by $(|\mathcal{A}|{-}1)t$.
Since $|\mathcal{A}|t \le |\sigma_{[0,n)}(v_0)|$, there exists a length $\tilde{r}$ factor $u$ of $\sigma_{[0,n)}(v)$ which is $p$-periodic word for some $1 \le p \le (|\mathcal{A}|{-}1)t$.
Since $\sigma_{[0,n)}(v\,v_0)$ is $\tilde{r}$-periodic, we obtain that the length~$p$ right extension~$u'$ of~$u$ is also $p$-periodic and $\tilde{r}$-periodic.
By Fine and Wilf's theorem on $u'$, w.l.o.g., $p$ divides~$\tilde{r}$.
In particular, we have $\mathbf{l}(\sigma_{[0,n)}(v)) \in (\tilde{r}/p)\, \mathbb{Z}^{\mathcal{A}}$.
By the unimodularity of $\sigma_{[0,n)}$, we obtain that $\mathbf{l}(v) \in (\tilde{r}/p)\, \mathbb{Z}^{\mathcal{A}}$, thus $p\, |\{0 \le i < r : v_i = v_0\}| \in \tilde{r}\, \mathbb{Z}$, giving the contradiction 
\[
\tilde{r} \le p\,|\{0 \le i < r : v_i = v_0\}| < |\sigma_{[0,n)}(v_0)|\, |\{0 \le i < r : v_i = v_0\}| \le |\sigma_{[0,n)}(v)| = \tilde{r}.
\]
This implies that $\sigma_{[0,n)}(v\,v_0)$ has a factor $u \in \mathcal{A}^{\tilde{r}}$ occurring before position $(|\mathcal{A}|{-}1)t$ with a unique extension of length~$t$.
Since $|\mathcal{A}|t \le |\sigma_{[0,n)}(v_0)|$, this extension is a prefix of~$u$.
Hence, $w_{[i,i+\tilde{r})} = u$ implies $w_{[i,i+t)} = w_{[i+\tilde{r},i+\tilde{r}+t)}$ for all $w \in X_{\boldsymbol{\sigma}}$, $i \in \mathbb{N}$.

If $\tilde{w}_{[0,s/8)}$ contains return words of $v$ with abelianizations spanning~$\mathbb{R}^{\mathcal{A}}$, then the $\sigma_{[0,n)}$-images of these return words also span~$\mathbb{R}^{\mathcal{A}}$.
Therefore, the abelianizations of  return words of~$u$ that are factors of $\sigma_{[0,n)}(\tilde{w}_{[0,s/8)})$ also span~$\mathbb{R}^{\mathcal{A}}$.
By dendricity, this implies that each element of $R_{X_{\boldsymbol{\sigma}}}(u)$ is a factor of $\sigma_{[0,n)}(\tilde{w}_{[0,s/8)})$.

<!-- lemma-9: body_md -->
\label{lem:strong echoing and return word}
Let $w = w_0 w_1 \cdots \in \mathcal{A}^{\mathbb{N}}$ be an echoing word with sequences $(r_{k})_{k\in\mathbb{N}}$, $(s_k)_{k\in\mathbb{N}}$.
Assume that, for every $k \in \mathbb{N}$, $w$~has a factor $u = u_{[0,r_k)} \in \mathcal{A}^{r_{k}}$ such that
\begin{enumerate}[\upshape (i)]
\item 
every occurrence of~$u$ in~$w_{[0,s_{k}/2)}$ is followed by a prefix~$u_{[0,l_k)}$ of~$u$ with $\lim_{k \to \infty} l_{k}/\log(s_{k})=\infty$, 
\item
and $w_{[0,s_{k}/2)}$ contains a set of return words of~$u$ such that the generated subgroup of~$F_\mathcal{A}$ contains at least two different elements of~$\mathcal{A}$.
\end{enumerate}
Then $w$ is strong echoing.
<!-- end lemma-9 -->
<!-- proof-9: body_md -->
By Lemma~\ref{lem:returnwords}, for every $\beta \in \overline{\mathbb{Q}}$ that is not a root of unity, there exists a return word~$uv$ of~$u$ in $w_{[0,s_{k}/2)}$ (with $v \in F_{\mathcal{A}}$) such that $\mathrm{ev}_\beta(P(uv)) \neq \mathrm{ev}_\beta(P(vu))$.
Write
\[
\begin{aligned}
w & = \hspace{1.4em} w_{[0,n)}\ u\ v_1\, u\ v_2\, u\ v_3 \cdots \\
T^{r_{k}}(w) & = w_{[r_{k},n)}\ u\ v_1\, u\ v_2\, u\ v_3\, u \cdots \\
\end{aligned}
\]
Since $u$ is always followed by a prefix of $u$ of length $l_{k}$ in $w_{[0,s_{k}/2)}$, each pair $(u v_i, v_i u)$ starts with a matching interval at least length $l_{k}$. 

Take a length $>l_{k}$ factor $uv_{i'}\cdots u v \cdots u v_{i''}$ of $w_{[0,s_{k}/2)}$ and let 
\[
    w_{[p_{k},q_{k}]} = u_{[l_{k},|u|)}v_{i'} uv_{i'+1}\cdots u v \cdots u v_{i''}.
\]
Then, we have $w_{[p_{k}-l_{k},p_{k})} = w_{[p_{k}+r_{k}-l_{k},p_{k}+r_{k})} = w_{(q_{k},q_{k}+l_{k}]} = w_{(q_{k}+r_{k},q_{k}+r_{k}+l_{k}]}$.
Furthermore, by Lemma~\ref{lem:Common root}, we have $\mathrm{ev}_\beta \circ P(w_{[p_{k},q_{k}]}) \neq \mathrm{ev}_\beta \circ P(w_{[p_{k}+r_{k},q_{k}+r_{k}]})$.

<!-- proposition-4: body_md -->
\label{prop:strong echoing of dendric S-adic words}
    Let $\boldsymbol{\sigma}$ be a primitive and recurrent sequence of unimodular substitutions over~$\mathcal{A}$, and assume that $\mathcal{C}_{\boldsymbol{\sigma}}$ forms a tiling of $\mathbf{1}^\perp$ with compact sets.
    If $X_{\boldsymbol{\sigma}}$ is dendric, then each $w \in X_{\boldsymbol{\sigma}}$ is strong echoing.
<!-- end proposition-4 -->
<!-- proof-10: body_md -->
\label{proof:strong echoing of dendric S-adic words}
    Let $\varepsilon \leq 1/9$.
    Let $r$, $s$ be as in Lemma~\ref{lem:density0}, and let $n_k$, $w^{(k)}, w',\tilde{w}$, $s'$ be as in the proof of Lemma~\ref{lem:Echoing}.
    Let $a \in \mathcal{A}$ be a letter such that $|\sigma_{[0,n_k)}(a)| \ge |\sigma_{[0,n_k)}(b)|$ for all $b \in \mathcal{A}$.
    By Lemma~\ref{lem:density0}~(\ref{i:d02})--(\ref{i:d03}), we have 
    \[
        |\{0 \le i <  s/8 : a = w_i \ne w_{i+r} \}|
        \leq \varepsilon f_{a} \, s \leq f_{a} s/9 < |\tilde{w}_{[1,s/8)}|_a.
    \]
    Thus there exists a factor $v = v_{[0,r+1)} \in \mathcal{A}^{r+1}$ of $\tilde{w}_{[1, s/8)}$ such that $v_0 = a = v_{r}$.
    By Lemma~\ref{lem:density0} (\ref{i:d05}), we can assume that $vv'$~is a factor of $\tilde{w}_{[1, s/8)}$ for all $v' \in R_{X_{\boldsymbol{\sigma}}}(v)$.
    By Lemma~\ref{lem:expanding gap on return words}, there exists a factor $u$ of $\sigma_{[0,n_k)}(v)$ with $|u| = |\sigma_{[0,n_k)}(v_{[0,r)})|(=r_k)$ such that, for all $i$ an index of an occurrence of $u$ in $w$,
    \[
    w_{i} = w_{i+|u|} \quad \text{for all integers } i \in \bigg[j, j+ \frac{|\sigma_{[0,n)}(v_0)|}{|\mathcal{A}|}\bigg),
    \]
    and all the return words of $u$ are factors of $\sigma_{[0,n_k)}(\tilde{w}_{[1,s/8)})$.
    When $k\to\infty$, we have $|\sigma_{[0,n_k)}(v_0)| / (|\mathcal{A}| \log s_k) \to \infty$  since $s_k \le |\sigma_{[0,n_k)}(v_0)|\, s$ and $\boldsymbol{\sigma}$ is primitive.
    We have $s_{k}/2 \geq (1/2) \, \sum_{a \in \mathcal{A}} |\sigma_{[0,n_k)}(a)| \times |\{ 1 \leq i <  s : w^{(k)}_{i} = a \}|$ and
    \[
        |w^{(k)}_{[1,s/2)}|_a \geq |\tilde{w}_{[1,s/2)}|_a \geq \frac{f_{a}\, s}{3}
        \geq \frac{2f_{a}\, s}{7} \geq 2\, |\tilde{w}_{[1,s/8)}|_a.
    \]
    Thus, we have $s_{k}/2 > |\sigma_{[0,n_k)}(\tilde{w}_{[1,s/8)})|$.
    Therefore, all the return words of $u$ are factors of $w_{[0,s_k/2)}$.
    Since $X_{\boldsymbol{\sigma}}$ is dendric, they generate the free group~$F_{\mathcal{A}}$.
    Then, by Lemma~\ref{lem:strong echoing and return word}, $w$ is strong echoing.
