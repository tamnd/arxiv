<!-- theorem-1: body_md -->
\label{thm}
Let $m\geq 2$ be an integer.
Let $K$ be a number field. Let $d_K$ be the degree of $K$ over $\mathbb{Q}$, and let $\operatorname{Cl}(K)$ be the class group of $K$. 
For every ideal class $c\in\operatorname{Cl}(K)$ and for all $x\in\mathbb{R}_{>0}$, let $N_{m,c}(x)$ be the number of $m$-full ideals of $\mathcal{O}_K$ of ideal class $c$ and of norm up to $x$. Then
$$
N_{m,c}(x) = x^{\frac 1m}\frac{\rho_K\#\operatorname{Cl}(K)[m]}{h_K}\sum_{\substack{I\in \mathcal R_m\\ [I]\in c+m\operatorname{Cl}(K)}}N(I)^{-\frac 1m} + O(x^{\nu}(\log(x+1))^\mu),
$$
where $\rho_K$ is the residue of the Dedekind zeta function of $K$ at $s=1$, $\#\operatorname{Cl}(K)[m]$ is the cardinality of the $m$-torsion subgroup of $\operatorname{Cl}(K)$, $h_K$ is the class number of $K$, $\mathcal R_m$ is the set of ideals $I$ such that $I\subseteq \mathfrak p^{m+1}$ and $I\nsubseteq \mathfrak p^{2m}$ for all prime ideals $\mathfrak p$ that contain $I$,
$$
\nu = \begin{cases}\frac 1{m+1} & \text{ if } d_K\leq m+1, \\ \frac{d_K-1}{md_K} & \text{ otherwise,} \end{cases}
\qquad \text{ and } \qquad
\mu=\begin{cases} 1 & \text{ if } d_K=m+1,\\ 0 & \text{ otherwise.} \end{cases}
$$

<!-- cor-1: body_md -->
\label{cor}
Let $K$ be a number field, and let $m$ be a positive integer.
For all $x\in\mathbb{R}_{>0}$, let $N_{m}(x)$ be the number of $m$-full ideals of $\mathcal{O}_K$ of norm up to $x$. Then
$$
N_{m}(x) = x^{\frac 1m} \rho_K \prod_{\mathfrak p}\left(1+\sum_{i=m+1}^{2m-1}N(\mathfrak p)^{-\frac im}\right) + O(x^{\nu}\log(x+1)^\mu),
$$
with $\rho_K$, $h_K$, $\nu$ and $\mu$ as in Theorem \ref{thm}.

\label{thm:algebraic}
Let $K$ be a number field, and let $m$ be a positive integer. Let $\mu_K$ be the group of roots of unity in $K$, and $R_K$ the regulator of $K$. Let  $r_1$ and $r_2$ be the numbers of real and complex places of $K$, respectively. For all $x\in\mathbb{R}_{>0}$, let $N'_m(x)$ be the number of $m$-full elements of $\mathcal{O}_K$ of height at most $x$. Then 
$$
N'_m(x)=
\frac{\#\mu_K\rho_K C_K C_{K,m}}{h_K}  x^{\frac 1m}(\log x)^{r_1+r_2-1} + E
$$
where 
$$
C_K=\begin{cases}
1 & \text{ if } r_1+r_2=1,\\
((r_1+r_2-1)!R_K)^{-1} & \text{ if }r_1+r_2\geq 2,
\end{cases}
$$
$$
C_{K,m} = \begin{cases}
1 & \text{ if } m=1,\\
\#\operatorname{Cl}(K)[m]\sum_{\substack{I\in \mathcal R_m\\ [I]\in m\operatorname{Cl}(K)}}N(I)^{-\frac 1m} & \text{ if } m\geq 2,
\end{cases}
$$ 
$$
E= \begin{cases} 
O(x^{\nu}(\log(x+1))^\mu) & \text{ if } r_1+r_2=1, \\
O(x^{\frac 1m}(\log x)^{r_1+r_2-2}\log\log x) & \text{ if } r_1+r_2\geq 2,
\end{cases}
$$
and $\rho_K$, $h_K$, $\operatorname{Cl}(K)$, $\mathcal R_m$, $\nu$ and $\mu$ are as in Theorem \ref{thm}.

<!-- proposition-1: body_md -->
\label{prop:bateman-grosswald}
Let $a,b,c:\mathcal I\to\mathbb{C}$ be functions that define convergent Dirichlet series and such that 
\begin{equation}
\label{eq:product}
\left(\sum_{I\in\mathcal I}\frac{a(I)}{N(I)^{s}}\right)\left(\sum_{I\in\mathcal I}\frac{b(I)}{N(I)^{s}}\right)=\sum_{I\in\mathcal I}\frac{c(I)}{N(I)^{s}}
\end{equation}
for all $s\in\mathbb{C}$ with $\Re(s)\gg 1$.

Assume that 
$$
\sum_{N(I)\leq x}a(I)=\alpha_0x^{\lambda_0}+\dots + \alpha_r x^{\lambda_r} + O(x^{\lambda}\log^{\mu}(x+1)) \quad \text{ and } \quad \sum_{N(I)\leq x}|b(I)| = O(x^\nu)
$$
with $\lambda, \mu, \nu \in\mathbb{R}_{\geq 0}$, $\alpha_0,\dots,\alpha_r, \lambda_0,\dots,\lambda_r\in\mathbb{C}$.
Then 
$$
\sum_{N(I)\leq x}c(I) = \gamma_0x^{\lambda_0}+\dots + \gamma_r x^{\lambda_r} +O(x^{\max(\lambda,\nu)}(\log(x+1))^{\mu'}),
$$
where for $i\in\{0,\dots,r\}$,
$$
\gamma_i = \begin{cases}
\alpha_i \sum_{I\in\mathcal I} b(I)N(I)^{-\lambda_i} & \text{ if } \Re(\lambda_i)>\nu,\\
0 & \text{ otherwise,}
\end{cases}
$$
and
$$
\qquad \mu'=\begin{cases}
\mu & \text{ if }\lambda>\nu,\\
\mu+1 & \text{ if }\lambda = \nu,\\
1 & \text{ if } \lambda <\nu \text{ and } \Re(\lambda_i) = \nu \text{ for some } i\in\{0,\dots,r\},\\
0 & \text{ otherwise.} 
\end{cases}
$$
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
For all positive integers $n$, let 
$$a_n=\sum_{N(I)=n}a(I), \quad b_n=\sum_{N(I)=n}b(I), \quad c_n=\sum_{N(I)=n}c(I).$$ 
By \eqref{eq:product} we have $\left( \sum_{n\geq 1} a_n n^{-s}\right)\left(\sum_{n\geq 1}b_n n^{-s} \right) = \sum_{n\geq 1} c_n n^{-s}$ for $s\in\mathbb{C}$ with $\Re(s)\gg 1$. Moreover,
$$\sum_{n\leq x}a_n = \sum_{N(I)\leq x}a(I), \quad \sum_{n\leq x}|b_n|\leq \sum_{N(I)\leq x}|b(I)|, \quad \sum_{n\leq x}c_n = \sum_{N(I)\leq x}c(I).
$$
Then we conclude by \cite[Lemma 1]{BG58}.

<!-- proposition-2: body_md -->
\label{prop:bounds}
Let $m$ be a positive integer.
Let $c\in\operatorname{Cl}(K)$. For $x\in\mathbb{R}_{\geq 0}$,
\begin{gather*}
\sum_{N(I)\leq x}\mathds 1_{\mathcal I_m, c}(I)=\frac{\rho_K\#\operatorname{Cl}(K)[m]}{h_K} x^{\frac 1m}  +O\left(x^{\frac{d_K-1}{md_K}}\right),\\
\sum_{N(I)\leq x}\mathds 1_{\mathcal R_m,c}(I) = O(x^{\frac 1{m+1}}).
\end{gather*}
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
We prove first the asymptotic formula. By the bijection \eqref{eq:bij_power}, writing $I=J^m$ with $J\in\mathcal I$ gives
\begin{gather*}
\sum_{N(I)\leq x}\mathds 1_{\mathcal I_m,c}(I) 
= \sum_{\substack{c'\in\operatorname{Cl}(K)\\ mc'=c}}\sum_{\substack{N(J)\leq x^{\frac 1m}\\ [J]=c'}}1.
\end{gather*}
By \cite[Theorem VI.3.3]{Lan94}
the inner sum is 
\begin{equation}\label{eq:ideals of bounded norm}
\frac{\rho_K}{h_K} x^{\frac 1m} +O\left(x^{\frac {d_K-1}{md_K}}\right).
\end{equation} 
Since the leading term is independent of the choice of the ideal class $c'$ we get the asymptotic formula in the statement.

For the upper bound, write $\prod_{i=1}^{m-1}\zeta_K((m+i)s)=\sum_{I\in\mathcal I}a(I)N(I)^{-s}$. 
For all prime ideals $\mathfrak p\in\mathcal P$ and all $e\in\mathbb{Z}_{\geq 0}$ we have 
$$a(\mathfrak p^e)
=\#\{(e_1,\dots,e_{m-1})\in\mathbb{Z}_{\geq 0}^{m-1}:e_1(m+1)+\dots+e_{m-1}(2m-1)=e\}. 
$$
Thus $a(\mathfrak p^e)\geq \mathds 1_{\mathcal R_m}(\mathfrak p^{e})$, and 
$$
a(I) = \prod_{\mathfrak p}a(\mathfrak p^{v_{\mathfrak p(I)}})\geq \prod_{\mathfrak p}\mathds 1_{\mathcal R_m}(\mathfrak p^{v_{\mathfrak p(I)}})=\mathds 1_{\mathcal R_m}(I)
$$ 
for all $I\in\mathcal I$.
Then
$$
\sum_{N(I)\leq x}\mathds 1_{\mathcal R_m,c}(I) \leq\sum_{N(I)\leq x}\mathds 1_{\mathcal R_m}(I) \leq\sum_{N(I)\leq x}a(I).
$$
Since $\prod_{i=1}^{m-1}\zeta_K((m+i)s)$ is a Dirichlet series with nonnegative coefficients, holomorphic for $\Re(s)>\frac 1{m+1}$ with a simple pole at $s=\frac 1{m+1}$ and no other poles on the line $\Re(s)=\frac 1{m+1}$, a Tauberian theorem \cite[Theorem A]{PTBZ25}
gives the desired bound.
<!-- end proof-2 -->
<!-- proof-3: body_md -->
By the bijection \eqref{eq:bij_full}, every $I\in\mathcal F_m$ can be written uniquely as $I=\widetilde I J$ with $\widetilde I\in \mathcal R_m$ and $J\in\mathcal I_m$. Let $\widetilde c=[\widetilde I]$. Then $[J]=[I]-\widetilde c\in m\operatorname{Cl}(K)$. Thus
$$
\sum_{I\in\mathcal I}\frac{\mathds 1_{\mathcal F_m,c}(I)}{N(I)^{s}}=\sum_{\widetilde c\in c+m\operatorname{Cl}(K)}\left(\sum_{\widetilde I\in\mathcal I}\frac{\mathds 1_{\mathcal R_m,\widetilde c}(\widetilde I)}{N(\widetilde I)^{s}} \right)\left(\sum_{J\in\mathcal I}\frac{\mathds 1_{\mathcal I_m,c-\widetilde c}(J)}{N(J)^{s}}\right).
$$
An application of Proposition \ref{prop:bateman-grosswald} with $a(I)=\mathds 1_{\mathcal I_m,c-\widetilde c}(I)$ and $b(I)=\mathds 1_{\mathcal R_m,\widetilde c}(I)$ and with the estimates in Proposition \ref{prop:bounds} gives
$$
N_{m,c}(x) = x^{\frac 1m}\frac{\rho_K\#\operatorname{Cl}(K)[m]}{h_K}\sum_{\substack{I\in \mathcal R_m\\ [I]\in c+m\operatorname{Cl}(K)}}N(I)^{-\frac 1m} + O(x^{\nu}(\log(x+1))^\mu),
$$
where 
$$
\nu = \max\left\{\frac 1{m+1},\frac{d_K-1}{md_K}\right\}
\qquad \text{ and } \qquad
\mu=\begin{cases} 1 & \text{ if } \frac 1{m+1}=\frac{d_K-1}{md_K},\\ 0 & \text{ otherwise.} \end{cases}
$$
We conclude as $\frac 1{m+1}\geq \frac{d_K-1}{md_K}$ if and only if $d_K\leq m+1$.
<!-- end proof-3 -->
<!-- proof-4: body_md -->
By Theorem \ref{thm} the leading constant is $$
\frac{\rho_K\#\operatorname{Cl}(K)[m]}{h_K}\sum_{I\in \mathcal R_m}\sum_{c\in [I]+m\operatorname{Cl}(K)}N(I)^{-\frac 1m} = \rho_K\sum_{I\in \mathcal R_m}N(I)^{-\frac 1m}.$$
We conclude by computing the Euler product, as $\mathds 1_{\mathcal R_m}$ is a multiplicative function.

<!-- lemma-1: body_md -->
\label{lemma:volume polytope}
Let $n$ be a positive integer. For real numbers $0\leq c\leq b$,  let 
$$P_{n,b,c}=\{\underline a\in\mathbb{R}^n: c-b\leq \sum_{i\in S}a_i\leq b \ \forall S\subseteq \{1,\dots,n\},  S\neq\emptyset\}.$$
Then there is a homogeneous polynomial $Q_n\in\mathbb{Q}[u,v]$ of degree $n$ such that $$\operatorname{meas}_n(P_{n,b,c})=Q_n(b,c),$$ where $\operatorname{meas}_n$ is the Lebesgue measure on $\mathbb{R}^n$. In particular,
$$
Q_n(u,v)=\sum_{k=0}^n {n \choose k} \frac{u^k(u-v)^{n-k}}{k!(n-k)!}, \qquad Q_n(1,1)=\frac 1{n!}, \qquad Q_n(b,c)=O(b^n),
$$
and the implicit constant in the upper bound is independent of $c$.

For every coordinate subspace $V\subseteq \mathbb{R}^n$, let $p_V:\mathbb{R}^n\to V$ be the orthogonal projection onto $V$.
Then $p_V(P_{n,b,c}) = P_{\dim V,b,c}$.
<!-- end lemma-1 -->
<!-- proof-5: body_md -->
For every subset $T\subseteq\{1,\dots,n\}$, let $T^c=\{1,\dots,n\}\smallsetminus T$, and let
$$
R_T=\{\underline a\in\mathbb{R}^n: a_i\geq 0 \ \forall i\in T, a_j\leq 0 \ \forall j\in T^c\}
$$
be the quadrant associated to $T$. Then $\operatorname{meas}_n(P_{n,b,c})=\sum_{T\subseteq\{1,\dots,n\}}\operatorname{meas}_n(P_{b,c}\cap R_T)$.
Fix $T\subseteq\{1,\dots,n\}$. 
We use the convention that a sum over the empty set is zero.
Then
$$
P_{n,b,c}\cap R_T = \left\{\underline a\in R_T: c-b\leq \sum_{i\in T^c}a_i, \ \sum_{i\in T}a_i \leq b\right\},
$$
as 
$$
 \sum_{i\in T^c}a_i \leq \sum_{i\in S\cap T^c}a_i \leq \sum_{i\in S} a_i \leq \sum_{i\in S\cap T}a_i \leq \sum_{i\in T}a_i
$$
for all nonempty $S\subseteq \{1,\dots,n\}$ and all $\underline a\in R_T$.
Thus 
$
\operatorname{meas}_n(P_{n,b,c}\cap R_T)=\operatorname{meas}_{\#T}(b\Delta^{\#T})\operatorname{meas}_{n-\#T}((b-c)\Delta^{n-\#T}),
$
where for every nonnegative integer $k$, 
$$\Delta^k=\left\{\underline a\in\mathbb{R}^k: \sum_{i=1}^k a_i\leq 1, \ a_i\geq 0 \ \forall i\in\{1,\dots,k\}\right\}$$ is the $k$-dimensional simplex, and $\operatorname{meas}_k(\Delta^k)=\frac 1{k!}$. Thus
$$
\operatorname{meas}_n(P_{n,b,c})=\sum_{k=0}^n {n \choose k} \frac{b^k(b-c)^{n-k}}{k!(n-k)!}.
$$

Let $V\subseteq \mathbb{R}^n$ be a coordinate subspace, and let $U\subseteq\{1,\dots,n\}$ such that $V=\{\underline a\in\mathbb{R}^n: a_i=0\ \forall i\in\{1,\dots,n\}\smallsetminus U\}$. We observe that for every $T\subseteq\{1,\dots,n\}$ we have
$$
p_V(P_{n,b,c}\cap R_T)=\left\{\underline a\in V\cap R_T: c-b\leq \sum_{i\in U\cap T^c}a_i, \ \sum_{i\in U\cap T} a_i\leq b\right\}.
$$
Thus
$
p_V(P_{n,b,c}) = \bigcup_{T\subseteq \{1,\dots,n\}}p_V(P_{n,b,c}\cap R_T) = P_{\dim V,b,c}.
$

<!-- prop-1: body_md -->
\label{prop:integers of given norm}
Let $I$ be a nonzero principal ideal of $\mathcal{O}_K$. For all $x\in\mathbb{R}_{>0}$, let $N''_{N(I)}(x)$ be the set of elements of $\mathcal{O}_K$ of norm $N(I)$ and of height at most $x$.
Then for $x\geq N(I)$, we have $N''_{N(I)}(x)=\#\mu_K$ if $r_1+r_2=1$, and
$$
N''_{N(I)}(x) =  \frac{\#\mu_K Q_{r_1+r_2-1}(\log x,\log N(I))}{R_K} + O((\log x)^{r_1+r_2-2})
$$
if $r_1+r_2\geq 2$,
where 
$Q_{r_1+r_2-1}$ is the polynomial from Lemma \ref{lemma:volume polytope}, and the implicit constant in the error term is independent of $N(I)$.
<!-- end prop-1 -->
<!-- proof-6: body_md -->
The number $N''_{N(I)}(x)$ is the cardinality of the set 
$$
A=\{\alpha\in\mathcal{O}_K:N(\alpha)=N(I), H(\alpha)\leq x\}.
$$
Since 
$$
\ell: K^\times \to\mathbb{R}^{\Omega_{\infty}}, \qquad \alpha\mapsto (\log|\alpha|_v)_{v\in\Omega_{\infty}},
$$
is a group homomorphism with kernel the group $\mu_K$ of roots of unity in $K$, 
$$
N''_{N(I)}(x)=\#\mu_K\#\ell(A).
$$
We recall that $\Lambda = \ell(\mathcal{O}_K^\times)$ is a lattice in the hyperplane given by $\sum_{v\in\Omega_{\infty}}a_v=0$. Let $r=r_1+r_2=\#\Omega_{\infty}$, and let $\epsilon_1,\dots,\epsilon_{r-1}$ be independent fundamental units of $\mathcal{O}_K$. Then $\Lambda$ is defined by the matrix $(\ell(\epsilon_i))_{1\leq i\leq r-1}$.
Let $\underline d=(d_v)_{v\in\Omega_{\infty}}$, with $d_v=1/d_K$ if $v$ is real and $d_v=2/d_K$ if $v$ is complex.
Let $\Lambda_{N(I)}=\Lambda + \log N(I)\underline d$.
Then
$$
\Lambda_{N(I)} =  \ell\left(\{\alpha\in\mathcal{O}_K: N(\alpha)=N(I)\}\right),
$$
as units are the algebraic integers of norm $1$.
If $r=1$, then $\#\ell(A)=\#\Lambda_{N(I)}=1$. 
We assume that $r\geq 2$.
Since $H(\alpha)=\prod_{v\in\Omega_{\infty}}\max\{1,|\alpha|_v\}$, the condition $H(\alpha)\leq x$ is equivalent to $\prod_{v\in S}|\alpha|_v\leq x$ for all nonempty subsets $S\subseteq\Omega_{\infty}$. Thus
$$
\ell(A)=\left\{\underline a\in \Lambda_{N(I)}: \sum_{v\in S}a_v\leq \log x \ \forall S\subseteq \Omega_{\infty}, S\neq\emptyset\right\}.
$$
 Let $\tilde v\in\Omega_{\infty}$. 
Writing $a_{\tilde v}=\log N(I) - \sum_{v\in\Omega_{\infty}\smallsetminus\{\tilde v\}}a_v$, we get
$$
\ell(A)
=\left\{\underline a\in \Lambda_{N(I)}: \log N(I)-\log x\leq \sum_{v\in S}a_v\leq \log x \ \forall S\subseteq \Omega_{\infty}\smallsetminus\{\tilde v\}, S\neq\emptyset\right\}.
$$
The coordinate projection $\pi:\mathbb{R}^{\Omega_{\infty}}\to\mathbb{R}^{\Omega_{\infty}\smallsetminus\{\tilde v\}}$ induces an isomorphism from $\{\sum_{v\in\Omega_{\infty}}a_v=\log N(I)\}$ onto $\mathbb{R}^{\Omega_{\infty}\smallsetminus\{\tilde v\}}$, so that 
$$
\#\ell(A)=\#\left(\pi(\Lambda_{N(I)})\cap P_{r-1,\log x,\log N(I)}\right),
$$
where $P_{r-1,\log x,\log N(I)}\subseteq\mathbb{R}^{r-1}$ is the polytope in Lemma \ref{lemma:volume polytope}. 
We observe that $\pi(\Lambda_{N(I)})=\pi(\Lambda)+\log N(I) (d_v)_{v\in\Omega_{\infty}\smallsetminus\{\tilde v\}}$, and that $\pi(\Lambda)\subseteq \mathbb{R}^{\Omega_{\infty}\smallsetminus\{\tilde v\}}$ is the full rank lattice  given by the matrix $B=(\log|\epsilon_i|_v)_{1\leq i\leq r-1, v\in\Omega_{\infty}\smallsetminus\{\tilde v\}}$. 
Translating by $-\log N(I) (d_v)_{v\in\Omega_{\infty}\smallsetminus\{\tilde v\}}$ yields a lattice point counting problem. Performing a change of variables by $B$ and using the fact that volume is invariant under translation, we can apply \cite{Dav51} to get 
\begin{gather*}
\#\ell(A) 
= \#(\pi(\Lambda)\cap (P_{r-1,\log x,\log N(I)}-\log N(I) (d_v)_{v\in\Omega_{\infty}\smallsetminus\{\tilde v\}}))
\\ = \frac {\operatorname{meas}_{r-1}(P_{r-1,\log x,\log N(I)})}{\det B} + O\left(\sum_V\operatorname{meas}_{\dim V}(p_V(P_{r-1,\log x,\log N(I)}))\right),
\end{gather*}
where the sum runs over all coordinate subspaces of $\mathbb{R}^{\Omega_{\infty}\smallsetminus\{\tilde v\}}$. Since $R_K=\det B$ by definition, an application of Lemma \ref{lemma:volume polytope} gives
\begin{equation*}
\#\ell(A) = \frac{Q_{r-1}(\log x,\log N(I))}{R_K} + O((\log x)^{r-2}). 
\qedhere
\end{equation*}

<!-- proof-7: body_md -->
Let $\mathcal F_{m,0}= \{I\in\mathcal F_m: [I]=[\mathcal{O}_K]\}$.
Since principal ideals of $\mathcal{O}_K$ are equal if and only if they have the same norm,
$$N'_{m}(x)=1+ \sum_{I\in\mathcal F_{m,0}, N(I)\leq x} N''_{N(I)}(x).$$
If $r_1+r_2=1$ the result follows by Proposition \ref{prop:integers of given norm}, Theorem \ref{thm} for $m\geq 2$ and \eqref{eq:ideals of bounded norm} for $m=1$.
We assume that $r_1+r_2\geq 2$.
We apply Proposition \ref{prop:integers of given norm} and the bound $N_{m,[\mathcal{O}_K]}(x)=O(x^{\frac 1m})$ from Theorem \ref{thm} for $m\geq 2$ and from \eqref{eq:ideals of bounded norm} for $m=1$, to get
$$
N'_m(x)=\frac{\#\mu_K}{R_K}\sum_{I\in\mathcal F_{m,0}, N(I)\leq x}Q(\log x,\log N(I)) +O(x^{\frac 1m}(\log x)^{r_1+r_2-2}),
$$
where $Q$ is the polynomial $Q_{r_1+r_2-1}$ from Lemma \ref{lemma:volume polytope}. 
Since $Q$ is a homogeneous polynomial, $$Q(\log x,\log N(I))=(\log x)^{r_1+r_2-1}Q\left(1,\frac{\log N(I)}{\log x}\right).$$ The Taylor series expansion of $Q(1,t)$ at $t=1$ gives 
\begin{equation*}\label{eq:Q simplified}
Q(\log x,\log N(I))=(\log x)^{r_1+r_2-1} Q(1,1) + O((\log x)^{r_1+r_2-2}(\log x-\log N(I)).
\end{equation*}
Hence,
\begin{gather*}
\sum_{I\in\mathcal F_{m,0}, N(I)\leq x}Q(\log x,\log N(I))
\\ =Q(1,1)(\log x)^{r_1+r_2-1} N_{m,[\mathcal{O}_K]}(x) + O((\log x)^{r_1+r_2-2}(E_1 + E_2)),
\end{gather*}
where
$$
E_1 = \sum_{I\in\mathcal F_{m,0}, N(I)\leq \frac{x}{(\log x)^m}} (\log x-\log N(I)) \leq  N_{m,[\mathcal{O}_K]}\left(\frac x{(\log x)^m}\right) \log x \ll x^{\frac 1m},
$$
and 
\begin{gather*}
E_2 = \sum_{I\in\mathcal F_{m,0}, \frac{x}{(\log x)^m}\leq N(I)\leq x} (\log x-\log N(I)) 
\\ \ll  N_{ m,[\mathcal{O}_K]}(x) \log\log x \ll x^{\frac 1m}\log\log x,
\end{gather*}
as $\log x-\log N(I)\leq \log x-\log (\frac x{(\log x)^m})=m\log\log x$.
We conclude by applying Theorem \ref{thm} as $\nu = \max\left\{\frac 1{m+1},\frac{d_K-1}{md_K}\right\}<\frac 1m$.
