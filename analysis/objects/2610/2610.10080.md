<!-- definition-1: body_md -->
Let $\mathcal{X}\subset\mathbb{R}^p$ be an input domain. A stochastic process
$\{F(x):x\in\mathcal{X}\}$ is a GP, denoted by $F\sim \mathrm{GP}(m,K)$, if, for every finite collection
$x_1,\ldots,x_n\in\mathcal{X}$, the random vector
\[
\bigl(F(x_1),\ldots,F(x_n)\bigr)^\top\sim N(\mu,\Sigma),
\]
where $\mu=[m(x_1),\cdots,m(x_n)]^\top$ and $\Sigma_{ij}=K(x_i,x_j)$. $m$ is called the mean function while $K$ is called the covariance function or kernel function, which together uniquely determine the GP.
<!-- end definition-1 -->
<!-- assumption-1: body_md -->
\label{ass:parameter}
The parameter space is
\[
\Theta=[s_-,s_+]\times[l_-,l_+]\times[t_-,t_+]
\Subset(0,\infty)^3,
\]
i.e., $\Theta$ is compactly contained in $(0,\infty)^3$, and assume that $\theta_0\in\operatorname{int}(\Theta)$.
<!-- end assumption-1 -->
<!-- assumption-2: body_md -->
\label{ass:coverage}
There exist a fixed closed cube $Q\subset\mathcal{X}$ with positive side length and constants $C_h<\infty$ and $\kappa>0$ such that, for all sufficiently large $n$,
\begin{equation}\label{eq:coverage}
h_n(Q)\le C_h n^{-\kappa}.
\end{equation}

<!-- theorem-1: body_md -->
\label{thm:rbf}
Under \Cref{ass:parameter,ass:coverage}, the following assertions hold.
\begin{enumerate}[label=(\roman*)]
\item \textbf{Fisher information.} The normalized Fisher information is uniformly nondegenerate:
\begin{equation}\label{eq:information-bounds}
 cI_3\preceq J_n\preceq CI_3
\end{equation}
for all sufficiently large $n$. Its spatial block $J_{n,\mathrm{sp}}$ satisfies
$cI_2\preceq J_{n,\mathrm{sp}}\preceq CI_2$, and
\begin{equation}\label{eq:information-block}
 J_n=
 \begin{pmatrix}
 J_{n,\mathrm{sp}}&o(1)\\
 o(1)&(2t_0^2)^{-1}+o(1)
 \end{pmatrix}.
\end{equation}
\item \textbf{Consistency and convergence rates.} The MLE $\widehat{\theta}_n$ admits the expansion
\begin{equation}\label{eq:mle-expansion}
 D_n(\widehat\theta_n-\theta_0)
 =J_n^{-1}D_n^{-1}\nabla l_n(\theta_0)+o_{\mathbb P}(1).
\end{equation}
In particular, $\widehat\theta_n$ is consistent with rates
\begin{equation}\label{eq:rates}
 \widehat s_n-s_0=O_{\mathbb P}(b_n^{-p/2}),\quad
 \widehat l_n-l_0=O_{\mathbb P}(b_n^{-(p+2)/2}),\quad
 \widehat t_n-t_0=O_{\mathbb P}(n^{-1/2}).
\end{equation}
\item \textbf{Asymptotic normality.} Let $M_n$ be any $3\times3$ matrices with
 $M_n^\top M_n=\mathcal I_n(\theta_0)$. Then
\begin{equation}\label{eq:main-an}
 M_n(\widehat\theta_n-\theta_0)\ \Longrightarrow\ N_3(0,I_3).
\end{equation}
Moreover,
\begin{equation}\label{eq:nugget-an}
 \sqrt n(\widehat t_n-t_0)\Longrightarrow N(0,2t_0^2).
\end{equation}
\item \textbf{Exponential tail bounds.} Write $\widehat h_n=D_n(\widehat\theta_n-\theta_0)
 =(\widehat h_{n,\mathrm{sp}},\widehat h_{n,t})$, where
 $\widehat h_{n,\mathrm{sp}}\in\mathbb R^2$ collects the two spatial coordinates $s$ and $l$.
 There are constants $c,C>0$ and
 $n_0$ such that, for all $n\ge n_0$ and all $H\ge0$,
\begin{equation}\label{eq:tail-bound}
 \mathbb P_{\theta_0,n}\bigl\{\left\lVert\widehat h_{n,\mathrm{sp}}\right\rVert\ge H\bigr\}
 \le C\exp\{-c\min(H^2,b_n^{p/2}H)\},
 \qquad
 \mathbb P_{\theta_0,n}\bigl\{|\widehat h_{n,t}|\ge H\bigr\}
 \le Ce^{-cH^2}.
\end{equation}
\item \textbf{Moment convergence.} The normalized MLE has uniformly bounded exponential moments: for every
 $\lambda>0$ there is $n_\lambda$ such that
\begin{equation}\label{eq:exp-moments}
 \sup_{n\ge n_\lambda}\mathbb E_{\theta_0,n}
 \exp\bigl\{\lambda\left\lVertD_n(\widehat\theta_n-\theta_0)\right\rVert\bigr\}<\infty.
\end{equation}
 Consequently, let $Z\sim N_3(0,I_3)$, and let $f:\mathbb R^3\to\mathbb R$ be
 continuous with $|f(x)|\le Ce^{\lambda\left\lVertx\right\rVert}$ for some constants
 $C,\lambda>0$. Then, for every choice of $M_n$ in (iii),
\begin{equation}\label{eq:f-convergence}
 \mathbb E_{\theta_0,n}f\bigl(D_n(\widehat\theta_n-\theta_0)\bigr)
 -\mathbb E f\bigl(J_n^{-1/2}Z\bigr)\to0,
 \qquad
 \mathbb E_{\theta_0,n}f\bigl(M_n(\widehat\theta_n-\theta_0)\bigr)\to\mathbb E f(Z).
\end{equation}
 In particular, all moments converge, and
\begin{equation}\label{eq:moment-convergence}
 D_n\mathbb E_{\theta_0,n}(\widehat\theta_n-\theta_0)\to0,\qquad
 D_n\mathbb E_{\theta_0,n}\bigl[(\widehat\theta_n-\theta_0)(\widehat\theta_n-\theta_0)^\top\bigr]D_n
 -J_n^{-1}\to0.
\end{equation}
\end{enumerate}

<!-- remark-1: body_md -->
\label{rem:uniform}
Inspection of the proofs shows that parts (i) and (iv) also hold
uniformly in $\theta_0\in\Theta$. Parts~(ii), (iii), and the limits
in part~(v) hold uniformly over $\theta_0$ in compact subsets
$K$ of $\operatorname{int}(\Theta)$. For example, writing
$\mathcal L_{\theta_0}$ for the law under $\theta_0$ and
$d_{\mathrm{BL}}$ for the bounded Lipschitz distance,
\[
 \sup_{\theta_0\in K}d_{\mathrm{BL}}\bigl\{\mathcal L_{\theta_0}
 \bigl(M_n(\widehat\theta_n-\theta_0)\bigr),N_3(0,I_3)\bigr\}\to0.
\]
A sketch of the proofs of these uniform claims is given in Appendix~\ref{app:uniform}.

Uniformity is what makes these asymptotic statements usable for
inference when $\theta_0$ is unknown. It ensures that Wald confidence
intervals based on $\mathcal I_n(\theta_0)^{-1}$ (or $\mathcal I_n(\hat{\theta}_n)^{-1}$, as will be shown later) attain their nominal coverage
asymptotically, uniformly over $\theta_0\in K$, rather than only at each
fixed parameter value.

<!-- theorem-2: body_md -->
\label{thm:minimax}
Assume \Cref{ass:parameter}. Let $d_{n,j}=D_{n,jj}$ for $j\in\{s,l,t\}$. For each $j\in\{s,l,t\}$, there exist constants $a>0$ and $c_0>0$ such
that, for every fixed $\theta_0\in\operatorname{int}(\Theta)$,
\[
 \theta_{1,n}^{(j)}
 =\theta_0+a d_{n,j}^{-1}e_j\in\Theta
\]
for all sufficiently large $n$, and
\begin{equation}\label{eq:probability-lower}
 \inf_{\widetilde\theta_{n,j}}
 \max_{\theta\in\{\theta_0,\theta_{1,n}^{(j)}\}}
 \mathbb P_{\theta,n}\left\{
 |\widetilde\theta_{n,j}-\theta_j|
 \ge\frac{a}{2d_{n,j}}\right\}
 \ge c_0.
\end{equation}
The other two parameters are identical under the two alternatives, so
this lower bound continues to hold even when they are known. Consequently,
\begin{equation}\label{eq:risk-lower}
 \inf_{\widetilde\theta_{n,j}}
 \sup_{\theta\in\Theta}
 \mathbb E_{\theta,n}(\widetilde\theta_{n,j}-\theta_j)^2
 \ge c\,d_{n,j}^{-2}.
\end{equation}

If \Cref{ass:coverage} also holds, then the MLE satisfies
\begin{equation}\label{eq:minimax-upper}
 \sup_{\theta\in\Theta}
 \mathbb E_{\theta,n}(\widehat\theta_{n,j}-\theta_j)^2
 \le C\,d_{n,j}^{-2}.
\end{equation}
Therefore, for each parameter $j\in\{s,l,t\}$,
\begin{equation}\label{eq:minimax-rate}
 \inf_{\widetilde\theta_{n,j}}
 \sup_{\theta\in\Theta}
 \mathbb E_{\theta,n}(\widetilde\theta_{n,j}-\theta_j)^2
 \asymp d_{n,j}^{-2},
\end{equation}
and the MLE attains the minimax rate. Equivalently, the minimax root MSE rates for $(s,l,t)$ are
\[
 b_n^{-p/2},\qquad
 b_n^{-(p+2)/2},\qquad
 n^{-1/2},
\]
respectively.

<!-- corollary-1: body_md -->
\label{cor:log-scale}
Under \Cref{ass:parameter,ass:coverage}, all assertions of
\Cref{thm:rbf} and \Cref{thm:minimax} hold for $\widehat\lambda_n$, with
$\widehat\theta_n-\theta_0$ replaced by $\widehat\lambda_n-\lambda_0$,
$\mathcal I_n(\theta_0)$ by $\mathcal I_n^{\log}(\theta_0)$, $J_n$ by $J_n^{\log}$,
and $\nabla l_n(\theta_0)$ by $\Lambda_0\nabla l_n(\theta_0)$, after
adjusting the constants $c$ and $C$. In particular,
$M_n(\widehat\lambda_n-\lambda_0)\Rightarrow N_3(0,I_3)$ whenever
$M_n^\top M_n=\mathcal I_n^{\log}(\theta_0)$,
\[
 \sqrt n\,(\log\widehat t_n-\log t_0)\Longrightarrow N(0,2),
\]
and
\[
 \mathbb E(\log\widehat s_n-\log s_0)^2\asymp b_n^{-p},\qquad
 \mathbb E(\log\widehat l_n-\log l_0)^2\asymp b_n^{-(p+2)},\qquad
 n\mathbb E(\log\widehat t_n-\log t_0)^2\longrightarrow2.
\]
The uniformity statements of \Cref{rem:uniform} also carry over.

<!-- corollary-2: body_md -->
\label{cor:plugin}
Under \Cref{ass:parameter,ass:coverage}, let $\widehat\mathcal I_n$ denote
either the plug-in Fisher information $\mathcal I_n(\widehat\theta_n)$ or the observed
information $-\nabla^2l_n(\widehat\theta_n)$. Then the following assertions hold.
\begin{enumerate}[label=(\roman*)]
\item \textbf{Consistency of the estimated Fisher information.}
\begin{equation}\label{eq:plugin-information}
 D_n^{-1}\widehat\mathcal I_nD_n^{-1}-J_n=O_\mathbb P(\epsilon_n).
\end{equation}
In particular, $\widehat\mathcal I_n$ is positive definite with probability
tending to one, and for each $j\in\{s,l,t\}$,
\[
 \frac{[\widehat\mathcal I_n^{-1}]_{jj}}{[\mathcal I_n(\theta_0)^{-1}]_{jj}}
 =1+O_\mathbb P(\epsilon_n).
\]
\item \textbf{Coordinatewise intervals.} For each $j\in\{s,l,t\}$,
\[
 \frac{\widehat\theta_{n,j}-\theta_{0,j}}
 {[\widehat\mathcal I_n^{-1}]_{jj}^{1/2}}\Longrightarrow N(0,1).
\]
Consequently, for every $\alpha\in(0,1)$, letting $z_{1-\alpha/2}$ denote the $(1-\alpha/2)$ quantile of the standard normal distribution, the Wald interval
$\widehat\theta_{n,j}\pm z_{1-\alpha/2}[\widehat\mathcal I_n^{-1}]_{jj}^{1/2}$
contains $\theta_{0,j}$ with probability tending to $1-\alpha$.
\item \textbf{Joint inference.} Let $\widehat L_n$ be the
lower-triangular Cholesky factor of $\widehat\mathcal I_n$. Then
\[
 \widehat L_n^\top(\widehat\theta_n-\theta_0)\Longrightarrow N_3(0,I_3),
 \qquad
 (\widehat\theta_n-\theta_0)^\top\widehat\mathcal I_n(\widehat\theta_n-\theta_0)
 \Longrightarrow\chi^2_3.
\]
Hence the ellipsoid
$\{\theta:(\widehat\theta_n-\theta)^\top\widehat\mathcal I_n(\widehat\theta_n-\theta)
\le\chi^2_{3,1-\alpha}\}$ is an asymptotic $1-\alpha$ confidence region
for $\theta_0$.
\item \textbf{Log scale.} Let $\widehat\Lambda_n=\operatorname{diag}(\widehat\theta_n)$ and
$\widehat\mathcal I_n^{\log}=\widehat\Lambda_n\widehat\mathcal I_n\widehat\Lambda_n$.
Then (i)--(iii) hold for $\widehat\lambda_n-\lambda_0$, with
$\widehat\mathcal I_n$ replaced by $\widehat\mathcal I_n^{\log}$ and $J_n$ by
$J_n^{\log}$. In particular, the interval
\[
 \Bigl[\widehat\theta_{n,j}
 \exp\bigl\{-z_{1-\alpha/2}[(\widehat\mathcal I_n^{\log})^{-1}]_{jj}^{1/2}\bigr\},\;
 \widehat\theta_{n,j}
 \exp\bigl\{z_{1-\alpha/2}[(\widehat\mathcal I_n^{\log})^{-1}]_{jj}^{1/2}\bigr\}\Bigr]
\]
contains $\theta_{0,j}$ with probability tending to $1-\alpha$. When
$\widehat\mathcal I_n$ is the observed information, $\widehat\mathcal I_n^{\log}$
coincides with the observed information $-\nabla_\lambda^2l_n$ at
$\widehat\lambda_n$ whenever $\widehat\theta_n\in\operatorname{int}(\Theta)$, an event of
probability tending to one.
\end{enumerate}
On the event that $\widehat\mathcal I_n$ is not positive definite, the
intervals and regions in (ii)--(iv) may be defined arbitrarily.

<!-- corollary-3: body_md -->
\label{cor:random}
Let $X_1,X_2,\ldots$ be independent and identically distributed locations
in a fixed bounded set $\mathcal X$, independent of the process and the errors.
Suppose their distribution has a Lebesgue density $f$ satisfying
$f(x)\ge f_->0$ for almost every $x$ in a fixed cube $Q\subset\mathcal X$. Write
$\mathbf X=(X_1,X_2,\ldots)$ for the design sequence. The quantities
$\mathcal I_n(\theta_0)$, $J_n$, and $M_n$ in \Cref{thm:rbf} are evaluated at the
realized locations, and are therefore random.
\begin{enumerate}[label=(\alph*)]
\item \textbf{Conditionally on the design.} For almost every realization
of $\mathbf X$, \Cref{ass:coverage} holds, and \Cref{thm:rbf,thm:minimax}
and \Cref{cor:log-scale,cor:plugin} hold conditionally on $\mathbf X$. The
constants $c$ and $C$ in these results do not depend on $\mathbf X$; only
the thresholds $n_0$ and $n_\lambda$ may.
\item \textbf{Unconditionally.} Under the joint law of the design, the
process, and the errors, the following assertions hold.
\begin{enumerate}[label=(\roman*)]
\item \Cref{thm:rbf}(i) holds almost surely for all sufficiently large
$n$.
\item \Cref{thm:rbf}(ii) and (iii) hold.
\item \Cref{thm:rbf}(iv) holds with an additional term $Ce^{-cn^{3/4}}$
on the right-hand side of each tail bound.
\item In \Cref{thm:rbf}(v), the exponential moment bound holds, and
$\mathbb E g(M_n(\widehat\theta_n-\theta_0))\to\mathbb E g(Z)$ for every test function $g$ as
in \Cref{thm:rbf}(v). In particular, $D_n\mathbb E(\widehat\theta_n-\theta_0)\to0$ and
$\mathbb E[M_n(\widehat\theta_n-\theta_0)(\widehat\theta_n-\theta_0)^\top M_n^\top]\toI_3$.
\item \Cref{thm:minimax} holds, so the MLE attains the minimax rate.
\item \Cref{cor:plugin} holds, and (i)--(v) above hold on the log scale.
\end{enumerate}
\end{enumerate}

<!-- lemma-1: body_md -->
\label{lem:affinity-monotone}
Let $P,Q$ be probability laws on a measurable space
$(\mathcal Y,\mathcal F)$, and let $T:\mathcal Y\to\mathcal Y'$ be a
measurable map into another measurable space. Then
\[
 \operatorname{Aff}(P\circ T^{-1},Q\circ T^{-1})\ge\operatorname{Aff}(P,Q).
\]
In particular, if $Y\sim N_d(0,A)$ under one law and $Y\sim N_d(0,B)$ under
the other, and $H$ is a fixed $r\times d$ matrix of rank $r$, then
$\operatorname{Aff}(HAH^\top,HBH^\top)\ge\operatorname{Aff}(A,B)$.

\label{lem:fraction}
Let $X$ be a random variable with $|X|\le M$ almost surely and
$\mathbb E X^2\ge\sigma^2>0$. Then, for every $\lambda\in[0,1)$,
\[
 \mathbb P(|X|\ge\lambda\sigma)\ge(1-\lambda^2)\frac{\sigma^2}{M^2}.

on a finite set $\mathcal A$, then at least
$(1-\lambda^2)\sigma^2|\mathcal A|/M^2$ elements $\iota\in\mathcal A$ satisfy
$|f(\iota)|\ge\lambda\sigma$.

\label{lem:kolmogorov}
Let $k\in\mathbb N$, $m\ge1$, and $\alpha>k$. If a real process
$(\xi(x))_{x\in[0,1]^k}$ satisfies

\label{lem:convex-sup}

\label{cor:envelope}
Let $k=k_1+k_2$ with $k_1,k_2\in\mathbb N$, let $m>k$, and let
$U\subseteq\mathbb R^{k_1}\times\mathbb R^{k_2}$ be a product of compact intervals.
For $\gamma\in[1,\infty]$ and $r\ge0$ put $\psi_\gamma(r)=\min(r^2,\gamma r)$,
so that $\psi_\infty(r)=r^2$. Let $(\xi(u))_{u\in U}$ have continuous
paths, and suppose that, for constants $A\ge0$, $a>0$, $B\ge0$, and
$\gamma_1,\gamma_2\in[1,\infty]$,
\begin{align}
 \left\lVert\xi(u)\right\rVert_{L^m}^m
 &\le A\exp\bigl[-a\{\psi_{\gamma_1}(|u_1|)+\psi_{\gamma_2}(|u_2|)\}\bigr],
 &&u\in U,\label{eq:envelope-moment}\\
 \left\lVert\xi(u)-\xi(v)\right\rVert_{L^m}&\le B|u-v|,
 &&u,v\in U.\label{eq:envelope-lipschitz}
\end{align}
Put $a''=a(m-k)/(32m)$. Then, for all $H_1,H_2\ge0$,
\begin{equation}\label{eq:envelope-bound}
 \mathbb E\sup_{\substack{u\in U\\|u_1|\ge H_1,\,|u_2|\ge H_2}}|\xi(u)|^m
 \le C\bigl(A+B^kA^{1-k/m}\bigr)
 \exp\bigl[-a''\{\psi_{\gamma_1}(H_1)+\psi_{\gamma_2}(H_2)\}\bigr],
\end{equation}
where $C$ depends only on $k_1$, $k_2$, $m$, and $a$, and not on
$\gamma_1$, $\gamma_2$, or $U$.

<!-- lemma-6: body_md -->
\label{lem:projection}
Assume \Cref{ass:parameter} and bounded observation locations. For any
fixed $A>0$, there is a parameter-independent orthogonal projection $P_n$
of rank $r_n=O(b_n^p)$ such that, with $S_n(\theta)=sR_n(l)$,
\begin{equation}\label{eq:projection-approx}
 \sup_{\theta\in\Theta}
 \left\lVert\partial_\theta^\gamma
 (S_n-P_nS_nP_n)\right\rVert_{\mathrm{op}}\le n^{-A},
 \qquad |\gamma|\le3.
\end{equation}
For $\widetilde C_n=tI_n+P_nS_nP_n$ and its log likelihood,
\begin{equation}\label{eq:likelihood-approx}
 \sup_{\theta\in\Theta}
 |\partial_\theta^\gamma(l_n-\widetilde l_n)|
 =O_{\mathbb P}(n^{4-A}),\qquad |\gamma|\le3.
\end{equation}

For a spatial derivative with $a$ derivatives in $s$ and $c$ in $l$,
$1\le a+c\le3$, a nonzero derivative has $a\le1$ and satisfies
\begin{align}
 \sup_\theta\left\lVertC_n^{-1/2}
   \partial_s^a\partial_l^c C_n C_n^{-1/2}\right\rVert_{\mathrm{op}}
 &\le C b_n^c,\label{eq:raw-operator}\\
 \sup_\theta\left\lVertC_n^{-1/2}
   \partial_s^a\partial_l^c C_n C_n^{-1/2}\right\rVert_{\mathrm F}
 &\le C b_n^{p/2+c}.\label{eq:raw-frobenius}
\end{align}
The same bounds hold after compression. Precisely, let $U$ be any
$n\times r$ matrix with orthonormal columns, $1\le r\le n$, that does not
depend on $\theta$, and let $A_\theta=U^\top C_n(\theta)U$ be the compressed
covariance. Then \eqref{eq:raw-operator} and \eqref{eq:raw-frobenius}
hold with $C_n$ replaced by $A_\theta$, that is, for the compressed
derivatives $\partial_s^a\partial_l^cA_\theta
=U^\top(\partial_s^a\partial_l^cC_n)U$ whitened by $A_\theta^{-1/2}$, with
constants that do not depend on $U$. In the coordinates
$\theta=\theta_0+D_n^{-1}h$, normalized spatial derivatives of total
order $k\le3$, whitened either by $C_n$ or by a compression $A_\theta$,
have respective bounds $C\epsilon_n^k$ in operator norm and
$C\epsilon_n^{k-1}$ in Frobenius norm. The first normalized nugget derivative has operator
norm $O(n^{-1/2})$ and Frobenius norm $O(1)$, and all covariance
derivatives of order at least two involving $t$ vanish.

\label{lem:quadratic}

$\Delta_n=D_n^{-1}\nabla l_n(\theta_0)$ and

Consider a path varying only coordinate $j$. Differentiating its Gaussian
density $p_u$ under the integral and using Cauchy--Schwarz gives
\[
 \left\lVert\partial_up_u\right\rVert_{L^1}
 =\mathbb E_u|\partial_u\log p_u|
 \le\sqrt{\mathcal I_{jj,n}(u)}.
\]
Consequently,
\begin{equation}\label{eq:tv-information}
 \operatorname{TV}(P_{u_1,n},P_{u_0,n})
 \le\frac12\int_{u_0}^{u_1}
       \sqrt{\mathcal I_{jj,n}(u)}\,du.
\end{equation}
The uniform upper bounds in \Cref{lem:information} are
$\mathcal I_{jj,n}\le C d_{n,j}^2$. Choose $a>0$ small enough that the
right side of \eqref{eq:tv-information} is at most $1/4$ when
$u_1-u_0=a/d_{n,j}$. Both alternatives lie in $\Theta$ eventually.

We now reduce estimation to testing between the two alternatives
(Le Cam's two-point method). Write $u_0=\theta_{0,j}$,
$u_1=u_0+a/d_{n,j}$, $P_0=P_{u_0,n}$, and $P_1=P_{u_1,n}$, and let
$\delta=(u_1-u_0)/2=a/(2d_{n,j})$ be half the separation. Let
$\widetilde\theta_{n,j}$ be any estimator of the $j$th coordinate, that is,
any measurable function of the data vector $Y_n$. It defines a test of
$P_0$ against $P_1$: decide for whichever of the two candidate values
$u_0,u_1$ is closer to the estimate,
\[
 \psi=\mathbf 1\bigl\{|\widetilde\theta_{n,j}-u_1|<|\widetilde\theta_{n,j}-u_0|\bigr\},
\]
so that $\psi=1$ means ``decide $u_1$'' and $\psi=0$ means ``decide
$u_0$''. Equivalently, $\psi=1$ exactly when $\widetilde\theta_{n,j}$
exceeds the midpoint $(u_0+u_1)/2$.

A wrong decision forces a large estimation error. If $\psi=1$, then
$|\widetilde\theta_{n,j}-u_0|>|\widetilde\theta_{n,j}-u_1|$, and the
triangle inequality
$|\widetilde\theta_{n,j}-u_0|+|\widetilde\theta_{n,j}-u_1|\ge u_1-u_0$
gives $|\widetilde\theta_{n,j}-u_0|>\delta$. Likewise, if $\psi=0$, then
$|\widetilde\theta_{n,j}-u_1|\ge\delta$. Hence
\[
 P_0\{|\widetilde\theta_{n,j}-u_0|\ge\delta\}\ge P_0(\psi=1),
 \qquad
 P_1\{|\widetilde\theta_{n,j}-u_1|\ge\delta\}\ge P_1(\psi=0).
\]

The total variation distance bounds how well any test can separate the
two laws. For the event $E=\{\psi=1\}$,
\[
 P_0(\psi=1)+P_1(\psi=0)=1-\{P_1(E)-P_0(E)\}
 \ge1-\operatorname{TV}(P_0,P_1)\ge\frac34.
\]
Combining the last two displays, the two estimation error
probabilities sum to at least $3/4$, so the larger of them is at least
$3/8$. This proves \eqref{eq:probability-lower} with $c_0=3/8$, since the
estimator was arbitrary.

Finally, by Markov's inequality,
$\mathbb E_{\theta,n}(\widetilde\theta_{n,j}-\theta_j)^2
\ge\delta^2\,\mathbb P_{\theta,n}\{|\widetilde\theta_{n,j}-\theta_j|\ge\delta\}$.
Taking the larger of the two alternatives and using
\eqref{eq:probability-lower} gives
$\sup_{\theta\in\Theta}\mathbb E_{\theta,n}(\widetilde\theta_{n,j}-\theta_j)^2
\ge\frac38\cdot\frac{a^2}{4d_{n,j}^2}$, which is
\eqref{eq:risk-lower}.

For the upper bound, recall from \Cref{lem:tail} that the constants in
the tail bound \eqref{eq:tail-bound} do not depend on $\theta_0\in\Theta$,
and that interiority of $\theta_0$ is not needed. Integrating the tail
bound as in the proof of \Cref{thm:rbf}(v) therefore gives, for
$n\ge n_0$,
\[
 \sup_{\theta\in\Theta}
 \mathbb E_{\theta,n}\bigl\{d_{n,j}^2(\widehat\theta_{n,j}-\theta_j)^2\bigr\}
 \le C,
\]
with $C$ independent of $n$ and $\theta$. Since $\Theta$ is bounded, the
finitely many $n<n_0$ are covered by enlarging $C$. This proves
\eqref{eq:minimax-upper}.
Combining this with \eqref{eq:risk-lower} gives
\eqref{eq:minimax-rate}. Since
\[

\textbf{A coverage event with stretched-exponential failure probability.}
Let $Q$ have side length $L$. For $n\ge1$, let
$k_n=\lfloor n^{1/(4p)}\rfloor$, and partition $Q$ into $k_n^p$ closed
subcubes of side length $L/k_n$. Let $H_n$ be the event that every
subcube contains at least one of $X_1,\ldots,X_n$. As in the proof of
\Cref{lem:random}, each subcube has sampling probability at least
$f_-L^p/k_n^p$, so a union bound gives
\begin{equation}\label{eq:coverage-failure}
 \mathbb P(H_n^c)\le k_n^p\exp\Bigl(-\frac{nf_-L^p}{k_n^p}\Bigr)
 \le n^{1/4}\exp\bigl(-f_-L^pn^{3/4}\bigr),
\end{equation}
because $k_n^p\le n^{1/4}$. On $H_n$, every point of $Q$ lies within the
diameter $\sqrt p\,L/k_n$ of an observation. Since
$\lfloor y\rfloor\ge y/2$ for $y\ge1$, this gives
$h_n(Q)\le2\sqrt p\,Ln^{-1/(4p)}$. Hence, on $H_n$, \eqref{eq:coverage}
holds at $n$ with $\kappa=1/(4p)$ and $C_h=2\sqrt p\,L$. Fix these
constants from now on. The constants in
\Cref{lem:projection,lem:recovery,lem:affinity,lem:tail} depend on the
design only through $\mathcal X$, $Q$, $C_h$, and $\kappa$, and the bounds of
\Cref{lem:tail} hold at every $n\ge n_0$ at which \eqref{eq:coverage} is
satisfied. Therefore every bound of \Cref{thm:rbf}(iv) and (v) holds on
$H_n$ for $n\ge n_0$, with constants that do not depend on the design.
By \eqref{eq:coverage-failure}, $\sum_n\mathbb P(H_n^c)<\infty$, so by the first
Borel--Cantelli lemma, almost surely $H_n$ occurs for all sufficiently
large $n$.

$\left\lVert\widehat h_n\right\rVert\le C\sqrt n$.

\textbf{A common projection.}
Since $\mathcal X$ is bounded and $l\ge l_->0$, the quantity
$u_l(x,y)=\left\lVertx-y\right\rVert^2/(2l^2)$ is uniformly bounded. Set
\[
 T_{M,l}(x,y)=\sum_{j=0}^{M}\frac{(-1)^j}{j!}u_l(x,y)^j.
\]
For each fixed $k\le3$, differentiating $l^{-2j}$ introduces a
polynomial factor of degree $k$ in $j$, with bounded additional
powers of $l^{-1}$. Therefore, for a fixed $B<\infty$,
\begin{equation}\label{eq:taylor-derivative-tail}
 \sup_{x,y\in\mathcal X,\,l\in[l_-,l_+]}
 |\partial_l^k(R_l(x,y)-T_{M,l}(x,y))|
 \le C_k\sum_{j>M}\frac{(j+1)^kB^j}{j!}
 \le e^{-M\log M+C_kM}.
\end{equation}
The last inequality is the case $\gamma=1$ of the following factorial
tail bound. Since $e^j\ge j^j/j!$, Stirling's lower bound
\begin{equation}\label{eq:stirling}
 \log j!\ge j\log j-j,\qquad j\ge1,
\end{equation}
holds, together with the trivial upper bound $\log j!\le j\log j$. Fix
$k\ge0$, $B>0$, and $\gamma\in(0,1]$, and put $a_j=(j+1)^kB^j/(j!)^\gamma$.
Then
\begin{equation}\label{eq:factorial-tail}
 \sum_{j>M}a_j\le e^{-\gamma M\log M+CM},\qquad M\ge1,
\end{equation}
where $C$ depends only on $k$, $B$, and $\gamma$. Indeed,
$a_{j+1}/a_j=\{(j+2)/(j+1)\}^kB(j+1)^{-\gamma}\le2^kB(j+1)^{-\gamma}$, which
is at most $1/2$ once $j+1\ge(2^{k+1}B)^{1/\gamma}$. For $M$ beyond this
threshold, the tail is at most $2a_{M+1}$. By \eqref{eq:stirling},
$\log(M+1)!\ge(M+1)\log(M+1)-(M+1)\ge M\log M-M-1$, so
               
\begin{align*}
     \log(2a_{M+1}) &\le\log2+k\log(M+2)+(M+1)\log^+B-\gamma(M\log M-M-1) \\ 
     &\le -\gamma M\log M+CM.
\end{align*}
  
For the finitely many smaller $M$, the tail is at most
$\sum_ja_j<\infty$, which is absorbed by enlarging $C$.

Let $P_n$ project onto evaluations of polynomials of total degree at most
$2M$. Each column and row of the matrix $T_{M,l}$ lies in that space,
as do its first three lengthscale derivatives. Thus
$P_n(\partial_l^kT_{M,l})P_n=\partial_l^kT_{M,l}$ and
\[
 \left\lVert\partial_l^kR_n-P_n(\partial_l^kR_n)P_n\right\rVert_{\mathrm{op}}
 \le 2n e^{-M\log M+C_kM}.
\]
Choose $M=\lceil c_Ab_n\rceil$ with a fixed $c_A>A+1$. Then
$\log M=(1+o(1))\log\log n$, so $M\log M=(c_A+o(1))\log n$, while
$C_kM=o(\log n)$. Hence the last bound is $n^{1-c_A+o(1)}\le n^{-A}$ for
large $n$. The rank is at most
$\binom{2M+p}{p}=O(b_n^p)$. Derivatives in $s$ cause only constant
factors; derivatives of $S_n$ in $t$ vanish. This proves
\eqref{eq:projection-approx}.

Both exact and reduced covariance matrices are bounded below by
$t_-I_n$. Put $\Omega_n=C_n-\widetilde C_n=S_n-P_nS_nP_n$, so that
$\left\lVert\Omega_n\right\rVert_{\mathrm{op}}\le n^{-A}$ uniformly over $\Theta$ by
\eqref{eq:projection-approx}. The resolvent identity
$C_n^{-1}-\widetilde C_n^{-1}=-C_n^{-1}\Omega_n\widetilde C_n^{-1}$ gives
\[
 \left\lVertC_n^{-1}-\widetilde C_n^{-1}\right\rVert_{\mathrm{op}}\le t_-^{-2}n^{-A}.
\]
Every raw covariance derivative through order three has operator norm
$O(n)$ uniformly over $\Theta$. Formulas for likelihood derivatives
through order three contain finitely many traces and quadratic forms
with at most three covariance derivative factors. Replacing factors one
at a time, using \eqref{eq:projection-approx}, bounds each difference by
$Cn^{4-A}(1+\left\lVertY_n\right\rVert^2/n)$. For the zeroth-order term,
\[
 l_n(\theta)-\widetilde l_n(\theta)
 =-\frac12\bigl\{\log\det C_n-\log\det\widetilde C_n\bigr\}
 -\frac12Y_n^\top\bigl(C_n^{-1}-\widetilde C_n^{-1}\bigr)Y_n.
\]
The quadratic part is at most $t_-^{-2}n^{-A}\left\lVertY_n\right\rVert^2$. For the log
determinant, let $C_n(r)=\widetilde C_n+r\Omega_n=(1-r)\widetilde C_n+rC_n$
for $r\in[0,1]$. As a convex combination of two matrices bounded below
by $t_-I_n$, each $C_n(r)$ satisfies $C_n(r)\succeq t_-I_n$. Jacobi's
formula gives $\frac d{dr}\log\det C_n(r)=\operatorname{tr}\{C_n(r)^{-1}\Omega_n\}$, and
$|\operatorname{tr}(AB)|\le n\left\lVertA\right\rVert_{\mathrm{op}}\left\lVertB\right\rVert_{\mathrm{op}}$ for $n\times n$ matrices.
Hence
\[
 \bigl|\log\det C_n-\log\det\widetilde C_n\bigr|
 \le\int_0^1\bigl|\operatorname{tr}\{C_n(r)^{-1}\Omega_n\}\bigr|\,dr
 \le t_-^{-1}n^{1-A}.
\]
Thus all differences in \eqref{eq:likelihood-approx} are at most
$Cn^{4-A}(1+\left\lVertY_n\right\rVert^2/n)$ uniformly over $\Theta$. Since
$\mathbb E_{\theta_0}\left\lVertY_n\right\rVert^2=\operatorname{tr} C_n(\theta_0)=n(s_0+t_0)$,

\textbf{Feature differentiation.}
Define
\[
 \phi_{\alpha,l}(x)
 =e^{-\left\lVertx\right\rVert^2/(2l^2)}
   \frac{x^\alpha}{l^{|\alpha|}\sqrt{\alpha!}},
 \qquad R_l(x,y)=\sum_{\alpha\in\mathbb N_0^p}
                 \phi_{\alpha,l}(x)\phi_{\alpha,l}(y).
\]
This feature expansion of the kernel is derived in
\eqref{eq:kernel-expansion} in the proof of \Cref{lem:recovery}. The
series and its fixed-order lengthscale derivatives converge absolutely
and uniformly on bounded sets. Direct differentiation gives
\begin{equation}\label{eq:feature-raising}
 \partial_l\phi_{\alpha,l}
 =l^{-1}\left[
 -|\alpha|\phi_{\alpha,l}
 +\sum_{j=1}^p\sqrt{(\alpha_j+1)(\alpha_j+2)}
                \phi_{\alpha+2e_j,l}\right].
\end{equation}
For $M\ge0$, let $I_M=\{\alpha\in\mathbb N_0^p:|\alpha|\le M\}$, which has
$q_M=\binom{M+p}{p}$ elements, and let $B_M(l)\in\mathbb R^{n\times q_M}$ be the
observation matrix of the features of degree at most $M$:
\[
 [B_M(l)]_{i\alpha}=\phi_{\alpha,l}(x_{n,i}),
 \qquad 1\le i\le n,\ \alpha\in I_M.
\]
Thus the column of $B_M(l)$ indexed by $\alpha$ is the feature
$\phi_{\alpha,l}$ evaluated at the $n$ sampling points, and
\[
 [B_M(l)B_M(l)^\top]_{ij}
 =\sum_{\alpha\in I_M}\phi_{\alpha,l}(x_{n,i})\phi_{\alpha,l}(x_{n,j})
\]
is the degree-$M$ truncation of $R_n(l)$. For $M\le M'$, let
$E_{M,M'}\in\mathbb R^{q_{M'}\times q_M}$ be the coordinate embedding,
$E_{M,M'}e_\alpha=e_\alpha$ for $\alpha\in I_M$. It has orthonormal
columns, so $\left\lVertE_{M,M'}\right\rVert_{\mathrm{op}}=1$, and $B_M=B_{M'}E_{M,M'}$ because
the columns of $B_M$ are among those of $B_{M'}$.

Applying \eqref{eq:feature-raising} column by column gives
$\partial_lB_M(l)=B_{M+2}(l)V_M(l)$, where
$V_M(l)\in\mathbb R^{q_{M+2}\times q_M}$ has columns
\[
 V_M(l)e_\alpha
 =l^{-1}\Bigl[-|\alpha|e_\alpha
 +\sum_{j=1}^p\sqrt{(\alpha_j+1)(\alpha_j+2)}\,e_{\alpha+2e_j}\Bigr],
 \qquad\alpha\in I_M.
\]
In particular, $V_M(l)=l^{-1}V_M^\circ$ with $V_M^\circ$ independent of
$l$. Since $\sqrt{(\alpha_j+1)(\alpha_j+2)}\le|\alpha|+2\le M+2$, every
column has norm at most $(p+1)(M+2)/l$. A bound on column norms does not
by itself bound an operator norm, but it does once the matrix is split
into pieces with orthogonal columns. Write
\begin{align*}
 & lV_M(l)=-E_{M,M+2}N_M+\sum_{j=1}^pW_{M,j},\quad N_Me_\alpha=|\alpha|e_\alpha,\quad\text{and}\quad W_{M,j}e_\alpha=\sqrt{(\alpha_j+1)(\alpha_j+2)}\,e_{\alpha+2e_j}.
\end{align*}
In each of these $p+1$ matrices, distinct columns are multiples of
distinct coordinate vectors, because the maps $\alpha\mapsto\alpha$ and
$\alpha\mapsto\alpha+2e_j$ are injective. The columns are therefore
orthogonal. For a matrix $W$ with orthogonal columns,
$\left\lVertWx\right\rVert^2=\sum_\alpha x_\alpha^2\left\lVertWe_\alpha\right\rVert^2$, so its operator
norm equals its largest column norm, here at most $M+2$. Hence
\begin{equation}\label{eq:raising-norm}
 \left\lVertV_M(l)\right\rVert_{\mathrm{op}}\le l_-^{-1}(p+1)(M+2)\le C(M+1).
\end{equation}
Equivalently, each row of $V_M(l)$ also has at most $p+1$ nonzero entries, so the Schur test applies. We now show that, for $k\le3$,
\begin{equation}\label{eq:truncated-derivative}
 \partial_l^k\{B_M(l)B_M(l)^\top\}
 =B_{M+2k}(l)H_{M,k}(l)B_{M+2k}(l)^\top,
 \quad \left\lVertH_{M,k}(l)\right\rVert_{\mathrm{op}}\le C_k(M+1)^k,
\end{equation}
where $H_{M,k}(l)=l^{-k}H_{M,k}^\circ$ is a symmetric
$q_{M+2k}\times q_{M+2k}$ matrix and $H_{M,k}^\circ$ does not depend on
$l$. For $k=0$, take $H_{M,0}=I_{q_M}$. Suppose that
\eqref{eq:truncated-derivative} holds for some $k$, and abbreviate
$M_k=M+2k$ and $E=E_{M_k,M_k+2}$. Since $B_{M_k}=B_{M_k+2}E$ and
$\partial_lB_{M_k}=B_{M_k+2}V_{M_k}$, the product rule gives
\[
 \partial_l^{k+1}\{B_MB_M^\top\}
 =B_{M_k+2}\bigl\{V_{M_k}H_{M,k}E^\top+E(\partial_lH_{M,k})E^\top
 +EH_{M,k}V_{M_k}^\top\bigr\}B_{M_k+2}^\top.
\]
The matrix in braces is $H_{M,k+1}$. It is symmetric, and it has the form
$l^{-(k+1)}H_{M,k+1}^\circ$ because $V_{M_k}=l^{-1}V_{M_k}^\circ$ and
$\partial_lH_{M,k}=-kl^{-1}H_{M,k}$. Since $\left\lVertE\right\rVert_{\mathrm{op}}=1$, by
\eqref{eq:raising-norm},
\[
 \left\lVertH_{M,k+1}\right\rVert_{\mathrm{op}}
 \le\bigl\{2C(M+2k+1)+kl_-^{-1}\bigr\}\left\lVertH_{M,k}\right\rVert_{\mathrm{op}}
 \le C_{k+1}(M+1)^{k+1}.
\]
This is how all smaller feature spaces are embedded in the common space
through degree $M+2k$: the embeddings $E$ carry every intermediate term
into $\mathbb R^{q_{M+2k}}$, so that the single feature matrix $B_{M+2k}$
appears on both sides.

Positive feature truncation gives
$sB_{M+2k}B_{M+2k}^\top\preceq C_n$ and therefore
\begin{equation}\label{eq:feature-contraction}
 \left\lVert\sqrt{s}\,C_n^{-1/2}B_{M+2k}\right\rVert_{\mathrm{op}}\le1.
\end{equation}
The whitened truncated derivative consequently has operator norm
$O(M^k)$ and rank $O(M^p)$, so its Frobenius norm is $O(M^{p/2+k})$.

To control the derivative tail, collect terms with $|\alpha|=j$.
Absolute summation of the monomials bounds these terms by $B^j/j!$
on the fixed bounded set, and a $k$th lengthscale derivative adds at
most $C_k(j+1)^k$. Hence \eqref{eq:factorial-tail} with $\gamma=1$ gives
the tail bound in \eqref{eq:taylor-derivative-tail}. For $M$ a sufficiently large multiple
of $b_n$, its operator norm is an arbitrarily small power of $n^{-1}$,
and its Frobenius norm is at most $\sqrt n$ times that bound.
Whitening multiplies these errors by at most $t_-^{-1}$.
This proves \eqref{eq:raw-operator} and \eqref{eq:raw-frobenius} for
lengthscale derivatives. One $s$-derivative removes the factor $s$;
two $s$-derivatives give zero.

For a compression $A_\theta=U^\top C_nU$, the derivatives are
$U^\top(\partial_l^kC_n)U$, so by \eqref{eq:truncated-derivative} the

$s(U^\top B_{M+2k})H_{M,k}(U^\top B_{M+2k})^\top$. Since
$sB_{M+2k}B_{M+2k}^\top\preceq C_n$, also
$sU^\top B_{M+2k}B_{M+2k}^\top U\preceq U^\top C_nU=A_\theta$, which gives the
analogue $\left\lVert\sqrt s\,A_\theta^{-1/2}U^\top B_{M+2k}\right\rVert_{\mathrm{op}}\le1$ of
\eqref{eq:feature-contraction}. The rank is still at most $q_{M+2k}$.
The compressed tail $U^\top\{\partial_l^k(R_n-B_MB_M^\top)\}U$ has operator
norm at most that of the uncompressed tail and Frobenius norm at most
$\sqrt r\le\sqrt n$ times it, and $A_\theta\succeq t_-I_r$. Hence every
estimate above holds verbatim for $A_\theta$, with the same constants.

Finally,

 =b_n^{-(a+c)p/2-c}\partial_s^a\partial_l^c.

Combining this identity with the preceding estimates gives

\textbf{Kernel expansion.}
For $x,y\in\mathbb R^p$, $\left\lVertx-y\right\rVert^2=\left\lVertx\right\rVert^2+\left\lVerty\right\rVert^2-2\langle x,y\rangle$.
By the multinomial theorem,
$\langle x,y\rangle^j=\sum_{|\alpha|=j}\frac{j!}{\alpha!}x^\alpha y^\alpha$,
and therefore
\[
 \exp\Bigl(\frac{\langle x,y\rangle}{l^2}\Bigr)
 =\sum_{j=0}^\infty\frac{\langle x,y\rangle^j}{l^{2j}j!}
 =\sum_{j=0}^\infty\sum_{|\alpha|=j}\frac{x^\alpha y^\alpha}{l^{2j}\alpha!}
 =\sum_{\alpha\in\mathbb N_0^p}\frac{x^\alpha y^\alpha}{l^{2|\alpha|}\alpha!}.
\]
The rearrangement is justified because the same series with every term
replaced by its absolute value sums to
$\exp(\sum_i|x_iy_i|/l^2)<\infty$. Hence
\begin{equation}\label{eq:kernel-expansion}
 k_\theta(x,y)
 =s\,e^{-\left\lVertx\right\rVert^2/(2l^2)}e^{-\left\lVerty\right\rVert^2/(2l^2)}e^{\langle x,y\rangle/l^2}
 =s\sum_{\alpha\in\mathbb N_0^p}\phi_{\alpha,l}(x)\phi_{\alpha,l}(y),
\end{equation}
where
\[
 \phi_{\alpha,l}(x)=e^{-\left\lVertx\right\rVert^2/(2l^2)}
 \frac{x^\alpha}{l^{|\alpha|}\sqrt{\alpha!}}.
\]
In particular,
$\sum_\alpha\phi_{\alpha,l}(x)^2=k_\theta(x,x)/s=1$.

\textbf{An analytic representation.}
Let $\xi_\alpha$, $\alpha\in\mathbb N_0^p$, be independent standard normal
variables, and define
\begin{equation}\label{eq:analytic-field}
 F(x)=\sqrt{s}\sum_{\alpha\in\mathbb N_0^p}\xi_\alpha\phi_{\alpha,l}(x)
 =\sqrt{s}\,e^{-\left\lVertx\right\rVert^2/(2l^2)}
 \sum_{\alpha\in\mathbb N_0^p}
 \xi_\alpha\frac{x^\alpha}{l^{|\alpha|}\sqrt{\alpha!}}.
\end{equation}
For $\rho>0$ and $x\in[-\rho,\rho]^p$, $|\phi_{\alpha,l}(x)|\le
(\rho/l_-)^{|\alpha|}/\sqrt{\alpha!}$, so
\[
 \mathbb E\sum_\alpha|\xi_\alpha|\sup_{x\in[-\rho,\rho]^p}|\phi_{\alpha,l}(x)|
 \le\sum_{\alpha}\frac{(\rho/l_-)^{|\alpha|}}{\sqrt{\alpha!}}
 =\Bigl(\sum_{a=0}^\infty\frac{(\rho/l_-)^a}{\sqrt{a!}}\Bigr)^p<\infty.
\]
Hence, almost surely, the series in \eqref{eq:analytic-field} converges
absolutely and uniformly on every box, and the power series in it
defines an entire function. For fixed $x,y$, the partial sums also
converge in $L^2$ because $\sum_\alpha\phi_{\alpha,l}(x)^2=1$, and
independence of the $\xi_\alpha$ together with \eqref{eq:kernel-expansion}
gives
\[
 \operatorname{Cov}\{F(x),F(y)\}=s\sum_\alpha\phi_{\alpha,l}(x)\phi_{\alpha,l}(y)
 =k_\theta(x,y).
\]
Thus \eqref{eq:analytic-field} is an analytic version of the centered
Gaussian process with covariance \eqref{eq:rbf-intro}.

\textbf{Taylor coefficients and their tails.}
Let
\[
 H(x)=e^{\left\lVertx\right\rVert^2/(2l^2)}F(x)
 =\sqrt s\sum_{\gamma\in\mathbb N_0^p}\xi_\gamma
 \frac{x^\gamma}{l^{|\gamma|}\sqrt{\gamma!}},
 \qquad a_l=-\frac1{2l^2},
\]
so that $F(x)=e^{a_l\left\lVertx\right\rVert^2}H(x)$. Since
$e^{a_l\left\lVertx\right\rVert^2}=\prod_je^{a_lx_j^2}=\sum_{r\in\mathbb N_0^p}a_l^{|r|}x^{2r}/r!$,
the Cauchy product of these absolutely convergent series gives
\[
 F(x)=\sqrt s\sum_{r\in\mathbb N_0^p}\sum_{\gamma\in\mathbb N_0^p}
 \frac{a_l^{|r|}}{r!\,l^{|\gamma|}\sqrt{\gamma!}}\,\xi_\gamma\,x^{2r+\gamma}.
\]
Collecting the terms with $2r+\gamma=\alpha$, the Taylor coefficients
$f_\alpha=[x^\alpha]F(x)$ are
\begin{equation}\label{eq:coefficient-explicit}
 f_\alpha=\sqrt{s}
 \sum_{2r\le\alpha}
 \frac{a_l^{|r|}}
 {r!\,l^{|\alpha|-2|r|}\sqrt{(\alpha-2r)!}}
 \xi_{\alpha-2r}.
\end{equation}
The summands involve distinct independent normal variables. For each
coordinate, the terms in
\[
 \sum_{r=0}^{\lfloor q/2\rfloor}
 \frac{q!}{r!r!(q-2r)!}
 |a_l|^{2r}(l^{-2})^{q-2r}
\]
form a subset of the nonnegative terms in the multinomial expansion
of $(2|a_l|+l^{-2})^q=(2l^{-2})^q$. Therefore
\[
 \alpha!\operatorname{Var}_\theta(f_\alpha)
 \le s(2l^{-2})^{|\alpha|},
 \qquad
 \left\lVertf_\alpha\right\rVert_{L^2(P_\theta)}
 \le \frac{C^{|\alpha|+1}}{\sqrt{\alpha!}}.
\]
Let $T_KF(x)=\sum_{|\alpha|\le K}f_\alpha x^\alpha$, and write
$|x|^\alpha=\prod_i|x_i|^{\alpha_i}$. The number of $\alpha\in\mathbb N_0^p$ with
$|\alpha|=j$ is $\binom{j+p-1}{p-1}$. Hence the Cauchy--Schwarz
inequality over these indices and the multinomial theorem give
\begin{align}
 \Bigl(\sum_{|\alpha|=j}\frac{|x|^\alpha}{\sqrt{\alpha!}}\Bigr)^2
 &\le\binom{j+p-1}{p-1}\sum_{|\alpha|=j}\frac{x^{2\alpha}}{\alpha!}
 =\binom{j+p-1}{p-1}\frac1{j!}\sum_{|\alpha|=j}\frac{j!}{\alpha!}
 \prod_{i=1}^p(x_i^2)^{\alpha_i}\notag\\
 &=\binom{j+p-1}{p-1}\frac{\left\lVertx\right\rVert^{2j}}{j!}.
 \label{eq:degree-sum-bound}
\end{align}
Since $|F(x)-T_KF(x)|\le\sum_{j>K}\sum_{|\alpha|=j}|f_\alpha||x|^\alpha$,
Minkowski's inequality, the bound on $\left\lVertf_\alpha\right\rVert_{L^2(P_\theta)}$,
\eqref{eq:degree-sum-bound}, and $\binom{j+p-1}{p-1}\le(j+1)^{p-1}$ give,
for $x$ in a fixed bounded set,
\[
 \left\lVertF(x)-T_KF(x)\right\rVert_{L^2(P_\theta)}
 \le C\sum_{j>K}\frac{(j+1)^{(p-1)/2}(C\left\lVertx\right\rVert)^j}{(j!)^{1/2}}.
\]
By \eqref{eq:factorial-tail} with $\gamma=1/2$, which rests on
Stirling's bound $\log j!\ge j\log j-j$, the right side is at most
$e^{-(K\log K)/2+CK}$. Squaring yields
\begin{equation}\label{eq:field-taylor-tail}
 \sup_{\theta\in\Theta}\sup_{x\in\mathcal X}
 \mathbb E_\theta|F(x)-T_KF(x)|^2
 \le e^{-K\log K+CK}.
\end{equation}

\textbf{Weights and polynomial Gram matrix.}
Partition $Q$ into nearest-observation cells $V_{n,i}$, with a fixed
tie-breaking rule. Define $w_{n,i}=|V_{n,i}|/|Q|$. Then
\begin{equation}\label{eq:voronoi-weights}
 \sum_iw_{n,i}=1,\qquad
 0\le w_{n,i}\le Ch_n(Q)^p\le Cn^{-\beta}.
\end{equation}
Indeed, $V_{n,i}\subseteq B(x_{n,i},h_n(Q))$. Zero weights and repeated
locations are allowed.

Use the product Legendre basis on $[0,L]^p$, restricted to total degree
at most $K$, and normalized for uniform probability measure on $Q$.
Write its elements as $\psi_\nu$, $|\nu|\le K$, and set
\[
 \Phi_{i,\nu}=\psi_\nu(x_{n,i}),\quad
 W=\operatorname{diag}(w_{n,1},\ldots,w_{n,n}),\quad M_K=\Phi^\top W\Phi.
\]
The shifted Legendre polynomials have the explicit form
\begin{equation}\label{eq:legendre-coefficients}
 P_j(2x-1)
 =\sum_{a=0}^j(-1)^{j-a}
       \binom ja\binom{j+a}{a}x^a.
\end{equation}
This follows from Rodrigues' formula
$P_j(t)=(2^jj!)^{-1}\frac{d^j}{dt^j}(t^2-1)^j$
\citep[\S18.5]{dlmf}. With $t=2x-1$, we have $t^2-1=4(x^2-x)$ and
$\frac{d}{dt}=\frac12\frac{d}{dx}$, so
\begin{align*}
 P_j(2x-1) &=\frac1{j!}\frac{d^j}{dx^j}\bigl\{x^j(x-1)^j\bigr\}
 =\frac1{j!}\sum_{a=0}^j\binom ja(-1)^{j-a}\frac{d^j}{dx^j}x^{j+a} =\sum_{a=0}^j(-1)^{j-a}\binom ja\frac{(j+a)!}{j!\,a!}x^a,
\end{align*}
which is \eqref{eq:legendre-coefficients}. This identity implies bounds $e^{CK}K^C$ for these basis polynomials and their first derivatives on any fixed enlarged box. The same bounds hold for products of two basis polynomials, after changing $C$. Comparing the integral over each $V_{n,i}$ with its value at $x_{n,i}$ gives
\[
 |[M_K-I]_{\mu\nu}|
 \le h_n(Q)
 \sup|\nabla(\psi_\mu\psi_\nu)|.
\]
The supremum is taken on a fixed box containing all relevant line
segments. This remains valid when the assigned node is outside $Q$.
The number of basis elements is polynomial in $K$, and hence
\begin{equation}\label{eq:gram-approx}
 \left\lVertM_K-I\right\rVert_{\mathrm{op}}\le h_n(Q)e^{CK}K^C.
\end{equation}
For $K=K_n$, this is $n^{-\kappa+o(1)}$, so $M_{K_n}$ and its inverse
are uniformly bounded for all sufficiently large $n$.

\textbf{Weighted regression and coefficient extraction.}
Fit the polynomial with Legendre coefficient vector
\[
 \widehat c=M_{K_n}^{-1}\Phi^\top WY_n.
\]
The basis has $q_K=\binom{K+p}{p}$ elements. For $|\alpha|\le K$, let
$L_{\alpha,K}\in\mathbb R^{1\times q_K}$ be the row vector that maps the Legendre
coefficient vector of a polynomial to its normalized monomial
coefficient of index $\alpha$:
\[
 L_{\alpha,K}c=\sqrt{\alpha!}\,[x^\alpha]
 \sum_{|\nu|\le K}c_\nu\psi_\nu(x),
 \qquad c\in\mathbb R^{q_K}.
\]
By
\eqref{eq:legendre-coefficients}, individual unnormalized Legendre
coefficients are at most $8^j$ on $[0,1]$. Rescaling to $[0,L]$,
taking products, and allowing polynomial dimension factors gives
\begin{equation}\label{eq:extraction-bound}
 \left\lVertL_{\alpha,K}\right\rVert^2
 \le e^{m\log m+CK},\qquad |\alpha|\le m\le K.
\end{equation}
Here $\alpha!\le|\alpha|!\le m!\le e^{m\log m}$ was used, by the trivial
upper counterpart of \eqref{eq:stirling}.
Define $\widehat\zeta_\alpha=L_{\alpha,K_n}\widehat c$ for
$|\alpha|\le m_n$. This is a fixed linear function of $Y_n$.

Write $Y_n=F_n+\bar\varepsilon$, where
$F_n=(F(x_{n,1}),\ldots,F(x_{n,n}))^\top$ and
$\bar\varepsilon=(\varepsilon_{n,1},\ldots,\varepsilon_{n,n})^\top$.
Weighted regression reproduces $T_KF$ exactly. If
$r_i=F(x_{n,i})-T_KF(x_{n,i})$, the residual coefficient error obeys
\[
 |L_{\alpha,K}M_K^{-1}\Phi^\top Wr|^2
 \le (L_{\alpha,K}M_K^{-1}L_{\alpha,K}^\top)\,r^\top Wr.
\]
For the noise error, $\operatorname{Cov}_\theta(\bar\varepsilon)=tI_n$ and
$W^2\preceq(\max_iw_{n,i})W$ imply
\[
 \operatorname{Var}_\theta(L_{\alpha,K}M_K^{-1}\Phi^\top W\bar\varepsilon)
 \le t_+\max_iw_{n,i}\,
        L_{\alpha,K}M_K^{-1}L_{\alpha,K}^\top.
\]
The field residual and noise are independent. Using
\eqref{eq:field-taylor-tail}, \eqref{eq:voronoi-weights},
\eqref{eq:gram-approx}, and \eqref{eq:extraction-bound}, we obtain
\begin{equation}\label{eq:recovery-error-calculation}
 \sup_\theta\mathbb E_\theta(\widehat\zeta_\alpha-\zeta_\alpha)^2
 \le e^{m_n\log m_n+CK_n}
      \{n^{-\beta}+e^{-K_n\log K_n+CK_n}\}.
\end{equation}
Since $\log m_n$ and $\log K_n$ are both $(1+o(1))\log\log n$, we have
$m_n\log m_n=(\beta/4+o(1))\log n$ and
$K_n\log K_n=(\beta/2+o(1))\log n$, while $CK_n=o(\log n)$. Hence the two
contributions in
\eqref{eq:recovery-error-calculation} are respectively
$n^{-3\beta/4+o(1)}$ and $n^{-\beta/4+o(1)}$.
This proves \eqref{eq:recovery-mse}.

\textbf{Conditioning of the exact coefficient covariance.}
Let $e_\alpha$ denote the coordinate vector indexed by $\alpha$ in the
space $|\alpha|\le m$. Define
\[
 D_{m,l}e_\alpha=l^{-|\alpha|}e_\alpha
\]
and a truncated raising matrix
\[
 Q_me_\alpha=
 \sum_{\substack{1\le j\le p\\|\alpha|+2\le m}}
 \sqrt{(\alpha_j+1)(\alpha_j+2)}\,e_{\alpha+2e_j}.
\]
Since $Q_m$ raises the total degree by two, $Q_m^r=0$ for $2r>m$. As
for $V_M$ in the proof of \Cref{lem:projection}, $Q_m$ splits into $p$
matrices with orthogonal columns of norm at most $m$, so
$\left\lVertQ_m\right\rVert_{\mathrm{op}}\le pm$.

We now derive \eqref{eq:triangular-transform} below in four steps. For a
function $f$ given by a power series $f(x)=\sum_\alpha([x^\alpha]f)\,x^\alpha$
that converges absolutely on $\mathbb R^p$, its normalized coefficients are
\[
 [\mathcal Nf]_\alpha=\sqrt{\alpha!}\,[x^\alpha]f,
 \qquad\text{so that}\qquad
 f(x)=\sum_{\alpha\in\mathbb N_0^p}[\mathcal Nf]_\alpha\,
 \frac{x^\alpha}{\sqrt{\alpha!}}.
\]
Write $\mathcal N_mf=([\mathcal Nf]_\alpha:|\alpha|\le m)\in\mathbb R^{q_m}$. By
\eqref{eq:coefficient-definition}, $\zeta^{(m)}=\mathcal N_mF$.

\textbf{Step 1: multiplication by $\left\lVertx\right\rVert^2$.} Since
$\left\lVertx\right\rVert^2f(x)=\sum_{j=1}^p\sum_\alpha([x^\alpha]f)\,x^{\alpha+2e_j}$,
for every $\beta$,
\[
 [x^\beta](\left\lVertx\right\rVert^2f)=\sum_{j:\,\beta_j\ge2}[x^{\beta-2e_j}]f.
\]
Multiply by $\sqrt{\beta!}$ and write $\alpha=\beta-2e_j$. Then
$\beta!/\alpha!=(\alpha_j+1)(\alpha_j+2)$, so
\[
 [\mathcal N(\left\lVertx\right\rVert^2f)]_\beta
 =\sum_{j:\,\beta_j\ge2}\sqrt{(\alpha_j+1)(\alpha_j+2)}\,
 [\mathcal Nf]_{\beta-2e_j}.
\]
For $|\beta|\le m$, the right side involves only indices of degree at
most $m-2$, and it is exactly the $\beta$th entry of $Q_m\mathcal N_mf$.
Hence $\mathcal N_m(\left\lVertx\right\rVert^2f)=Q_m\mathcal N_mf$.

\textbf{Step 2: multiplication by $e^{a\left\lVertx\right\rVert^2}$.} By induction,
$\mathcal N_m(\left\lVertx\right\rVert^{2r}f)=Q_m^r\mathcal N_mf$ for every $r\ge0$. For
$a\in\mathbb R$, $e^{a\left\lVertx\right\rVert^2}f=\sum_{r\ge0}a^r\left\lVertx\right\rVert^{2r}f/r!$, and all these
power series converge absolutely, so coefficients may be extracted term
by term. Since $Q_m^r=0$ for $2r>m$,
\[
 \mathcal N_m\bigl(e^{a\left\lVertx\right\rVert^2}f\bigr)
 =\sum_{r\le m/2}\frac{a^r}{r!}Q_m^r\mathcal N_mf
 =e^{aQ_m}\mathcal N_mf.
\]

\textbf{Step 3: the coefficients of $H$.} By the definition of $H$,
$[x^\gamma]H=\sqrt s\,\xi_\gamma/(l^{|\gamma|}\sqrt{\gamma!})$, so
$[\mathcal NH]_\gamma=\sqrt s\,l^{-|\gamma|}\xi_\gamma$, that is,
$\mathcal N_mH=\sqrt s\,D_{m,l}\xi^{(m)}$.

\textbf{Step 4: the coefficients of $F$.} Since $F=e^{a_l\left\lVertx\right\rVert^2}H$
with $a_l=-1/(2l^2)$, Steps 2 and 3 give
\begin{equation}\label{eq:triangular-transform}
 \zeta^{(m)}=\mathcal N_mF
 =e^{a_lQ_m}\mathcal N_mH
 =\sqrt{s}\,e^{-Q_m/(2l^2)}D_{m,l}\xi^{(m)}.
\end{equation}
Entrywise, this is \eqref{eq:coefficient-explicit} multiplied by
$\sqrt{\alpha!}$.
The transform and its inverse have operator norms at most $e^{Cm}$:
$\left\lVerte^{\pm Q_m/(2l^2)}\right\rVert_{\mathrm{op}}\le e^{pm/(2l_-^2)}\le e^{Cm}$ and
$\left\lVertD_{m,l}^{\pm1}\right\rVert_{\mathrm{op}}\le e^{Cm}$. Constants can be absorbed
into the exponent for $m\ge1$. This proves
\eqref{eq:coefficient-eigenvalues}.

\textbf{Relative covariance error and affinity.}
Let $\delta_n=\widehat\zeta^{(m_n)}-\zeta^{(m_n)}$, a centered random
vector. Since $q_{m_n}\le(m_n+1)^p=n^{o(1)}$ and $e^{Cm_n}=n^{o(1)}$,
\eqref{eq:recovery-mse} and \eqref{eq:coefficient-eigenvalues} give
\begin{align*}
 \sup_\theta\mathbb E_\theta\left\lVert\delta_n\right\rVert^2
 &=\sup_\theta\sum_{|\alpha|\le m_n}
 \mathbb E_\theta(\widehat\zeta_\alpha-\zeta_\alpha)^2
 \le n^{-\beta/4+o(1)},\\
 \sup_\theta\mathbb E_\theta\left\lVert\zeta^{(m_n)}\right\rVert^2
 &=\sup_\theta\operatorname{tr}\Sigma_{m_n,\theta}\le q_{m_n}e^{Cm_n}=n^{o(1)}.
\end{align*}
Since $\widehat\zeta^{(m_n)}=\zeta^{(m_n)}+\delta_n$,
\[
 \widehat\Sigma_{n,\theta}-\Sigma_{m_n,\theta}
 =\mathbb E_\theta\bigl[\zeta^{(m_n)}\delta_n^\top\bigr]
 +\mathbb E_\theta\bigl[\delta_n\zeta^{(m_n)\top}\bigr]
 +\mathbb E_\theta\bigl[\delta_n\delta_n^\top\bigr].
\]
For unit vectors $u,v$, the Cauchy--Schwarz inequality gives
\[
 \bigl|u^\top\mathbb E_\theta[\zeta^{(m_n)}\delta_n^\top]v\bigr|
 \le\bigl\{\mathbb E_\theta(u^\top\zeta^{(m_n)})^2\,
 \mathbb E_\theta(v^\top\delta_n)^2\bigr\}^{1/2}
 \le\bigl\{\mathbb E_\theta\left\lVert\zeta^{(m_n)}\right\rVert^2\,
 \mathbb E_\theta\left\lVert\delta_n\right\rVert^2\bigr\}^{1/2},
\]
and similarly for the transposed term, while
$\left\lVert\mathbb E_\theta[\delta_n\delta_n^\top]\right\rVert_{\mathrm{op}}
=\sup_{\left\lVertu\right\rVert=1}\mathbb E_\theta(u^\top\delta_n)^2\le\mathbb E_\theta\left\lVert\delta_n\right\rVert^2$.
Therefore
\begin{align*}
 \left\lVert\widehat\Sigma_{n,\theta}-\Sigma_{m_n,\theta}\right\rVert_{\mathrm{op}}
 &\le \mathbb E_\theta\left\lVert\delta_n\right\rVert^2+
 2\sqrt{\mathbb E_\theta\left\lVert\delta_n\right\rVert^2\,
             \mathbb E_\theta\left\lVert\zeta^{(m_n)}\right\rVert^2}\\
 &\le n^{-\beta/4+o(1)}+n^{-\beta/8+o(1)}
 =n^{-\beta/8+o(1)}.
\end{align*}
Finally,
$\left\lVert\Sigma^{-1/2}A\Sigma^{-1/2}\right\rVert_{\mathrm{op}}\le\left\lVert\Sigma^{-1}\right\rVert_{\mathrm{op}}\left\lVertA\right\rVert_{\mathrm{op}}$
and $\left\lVert\Sigma_{m_n,\theta}^{-1}\right\rVert_{\mathrm{op}}\le e^{Cm_n}=n^{o(1)}$ by
\eqref{eq:coefficient-eigenvalues}, which proves
\eqref{eq:relative-recovery}.

Let $\eta_n=n^{-\beta/8+o(1)}$ be a common bound in that equation.
For large $n$,
\[
 (1-\eta_n)\Sigma_{m_n,\theta}
 \preceq\widehat\Sigma_{n,\theta}
 \preceq(1+\eta_n)\Sigma_{m_n,\theta}.
\]
The same inequalities hold for sums of two such covariance matrices.
Using \eqref{eq:gaussian-affinity} and the corresponding log determinant
bounds, the absolute difference of log affinities is at most
$Cq_{m_n}\eta_n=o(1)$. This proves \eqref{eq:recovery-affinity}.
The comparison is multiplicative for affinity, which will be essential
when the affinity itself is exponentially small.

\textbf{Triangular structure.}
Order the indices $\alpha\in I_m=\{|\alpha|\le m\}$ by total degree, with
an arbitrary order within each degree. Write $\alpha(1),\ldots,\alpha(q)$
for the ordered indices, where $q=q_m$, write $\zeta_k$ and $\xi_k$ for
the coordinates of $\zeta^{(m)}$ and $\xi^{(m)}$ with index $\alpha(k)$,
and put $\zeta_{<k}=(\zeta_1,\ldots,\zeta_{k-1})$ and
$\xi_{<k}=(\xi_1,\ldots,\xi_{k-1})$. By \eqref{eq:triangular-transform},
$\zeta^{(m)}=\sqrt s\,\mathcal T_\theta\xi^{(m)}$ with
$\mathcal T_\theta=e^{-Q_m/(2l^2)}D_{m,l}$. Since $Q_m$ raises total degree
by two, every nonzero entry of $Q_m^r$, $r\ge1$, lies in a row of
strictly higher degree than its column. Hence
$e^{-Q_m/(2l^2)}=\sum_{r\le m/2}(-2l^2)^{-r}Q_m^r/r!$ is unit lower
triangular in this order, and $\mathcal T_\theta$ is lower triangular with
diagonal entries $l^{-|\alpha(k)|}$:
\[
 \zeta_k=\sqrt s\,l^{-|\alpha(k)|}\xi_k
 +\sqrt s\sum_{i<k}(\mathcal T_\theta)_{ki}\,\xi_i,
 \qquad k=1,\ldots,q.
\]
The leading $(k-1)\times(k-1)$ block of $\mathcal T_\theta$ is invertible,
so $\zeta_{<k}$ and $\xi_{<k}$ are invertible linear functions of each
other, and $\xi_k$ is independent of $\xi_{<k}$. Therefore, under
$P_\theta$, the conditional law of $\zeta_k$ given $\zeta_{<k}$ is
$N(\mu_{\theta,k}(\zeta_{<k}),\omega_{\theta,k})$, where $\mu_{\theta,k}$ is a
linear function of $\zeta_{<k}$ and
\begin{equation}\label{eq:conditional-variance}
 \omega_{\theta,k}=\operatorname{Var}_\theta(\zeta_k\mid\zeta_{<k})
 =s\,l^{-2|\alpha(k)|}.
\end{equation}
The conditional variance does not depend on $\zeta_{<k}$.

\textbf{Univariate affinity.}
For normal densities with means $\mu,\mu'$ and variances $\omega,\omega'$, the
exponent of the product of the two square roots satisfies
\[
 -\frac{(x-\mu)^2}{4\omega}-\frac{(x-\mu')^2}{4\omega'}
 =-\frac{(x-\bar\mu)^2}{2w}-\frac{(\mu-\mu')^2}{4(\omega+\omega')},
 \qquad w=\frac{2\omega\omega'}{\omega+\omega'},
\]
for a suitable $\bar\mu$. Integrating with respect to $x$ gives
\begin{align*}
 \operatorname{Aff}\{N(\mu,\omega),N(\mu',\omega')\}
 &=\frac{\sqrt{2\pi w}}{\sqrt{2\pi}\,(\omega\omega')^{1/4}}
 \exp\Bigl\{-\frac{(\mu-\mu')^2}{4(\omega+\omega')}\Bigr\} =\left(\frac{2\sqrt{\omega\omega'}}{\omega+\omega'}\right)^{1/2}
   \exp\left\{-\frac{(\mu-\mu')^2}{4(\omega+\omega')}\right\}.
\end{align*}
Since $(\omega+\omega')/(2\sqrt{\omega\omega'})
=\frac12(\sqrt{\omega/\omega'}+\sqrt{\omega'/\omega})
=\cosh\{\frac12\log(\omega/\omega')\}$, it follows that
\begin{equation}\label{eq:univariate-affinity}
 \operatorname{Aff}\{N(\mu,\omega),N(\mu',\omega')\}
 \le\cosh\Bigl\{\frac12\log\frac\omega{\omega'}\Bigr\}^{-1/2}
 \qquad\text{for all }\mu,\mu'.
\end{equation}

\textbf{Successive integration.}
Let $\pi_\theta$ denote the density of $\zeta^{(m)}$ under $P_\theta$, and
$\pi_\theta(\zeta_k\mid\zeta_{<k})$ its conditional densities, so that
$\pi_\theta(\zeta)=\prod_{k=1}^q\pi_\theta(\zeta_k\mid\zeta_{<k})$. For
$0\le k\le q$ put
\[
 \mathcal A_k=\int_{\mathbb R^k}\prod_{i=1}^k
 \sqrt{\pi_\theta(\zeta_i\mid\zeta_{<i})\,
 \pi_{\theta_0}(\zeta_i\mid\zeta_{<i})}\;d\zeta_1\cdots d\zeta_k,
\]
so that $\mathcal A_0=1$ and
$\mathcal A_q=\operatorname{Aff}(\Sigma_{m,\theta},\Sigma_{m,\theta_0})$. The integrand is
nonnegative, so by Tonelli's theorem we may integrate over $\zeta_k$
first, with $\zeta_{<k}$ fixed. By \eqref{eq:univariate-affinity},
\[
 \int_\mathbb R\sqrt{\pi_\theta(\zeta_k\mid\zeta_{<k})\,
 \pi_{\theta_0}(\zeta_k\mid\zeta_{<k})}\;d\zeta_k
 \le\rho_k:=\cosh\Bigl\{\frac12\log\frac{\omega_{\theta,k}}{\omega_{\theta_0,k}}\Bigr\}^{-1/2}.
\]
This bound is the same for every value of $\zeta_{<k}$, because the
conditional variances \eqref{eq:conditional-variance} do not depend on
$\zeta_{<k}$. The conditional means, which do depend on $\zeta_{<k}$ and
on the parameter, enter only through the exponential factor of the
univariate affinity, which is at most one. Hence
$\mathcal A_k\le\rho_k\mathcal A_{k-1}$, and by induction
$\operatorname{Aff}(\Sigma_{m,\theta},\Sigma_{m,\theta_0})\le\prod_{k=1}^q\rho_k$.
Finally, with $u=\log(s/s_0)$ and $v=\log(l/l_0)$,
\eqref{eq:conditional-variance} gives
$\log(\omega_{\theta,k}/\omega_{\theta_0,k})=u-2|\alpha(k)|v$. Therefore
\begin{equation}\label{eq:coefficient-affinity-product}
 \operatorname{Aff}(\Sigma_{m,\theta},\Sigma_{m,\theta_0})
 \le\prod_{k=1}^q\cosh\bigl(u/2-|\alpha(k)|v\bigr)^{-1/2}
 =\exp\left\{-\frac12\sum_{|\alpha|\le m}
             \log\cosh(u/2-|\alpha|v)\right\}.
\end{equation}

\textbf{A fixed fraction of separated degrees.}
To bound the sum, select a multiindex $\alpha$ uniformly from $I_m$, let
$J=|\alpha|$, and put $Z=J/m$ for $m\ge1$. The number of indices of degree
$j$ is $\binom{j+p-1}{p-1}$. Binomial summation gives
\[
 \mathbb E Z=\frac{p}{p+1},\qquad
 \mathbb E Z^2=\frac{p}{p+2}
       +\frac{p}{(p+1)(p+2)m}.
\]
The matrix
$\mathbb M=\begin{pmatrix}1&-\mathbb E Z\\-\mathbb E Z&\mathbb E Z^2\end{pmatrix}$ has
$\det\mathbb M=\operatorname{Var} Z\ge\frac p{p+2}-\frac{p^2}{(p+1)^2}
=\frac p{(p+2)(p+1)^2}$ and $\operatorname{tr}\mathbb M\le2$, so
\[
 \lambda_{\min}(\mathbb M)\ge\frac{\det\mathbb M}{\operatorname{tr}\mathbb M}
 \ge c_p:=\frac p{2(p+2)(p+1)^2}.
\]
Put $a=u/2$, $b=mv$, and $L_0=\sqrt{a^2+b^2}$, so that
$a-bZ=u/2-|\alpha|v$. Then
\[

 =(a,b)\,\mathbb M\,(a,b)^\top\ge c_pL_0^2,
\]
and, since $0\le Z\le1$, $|a-bZ|\le|a|+|b|\le\sqrt2L_0$. If $L_0=0$, then
$u=v=0$ and \eqref{eq:degree-separation} below is trivial. Otherwise,
\Cref{lem:fraction} with $X=a-bZ$, $\sigma^2=c_pL_0^2$, $M=\sqrt2L_0$, and
$\lambda=1/\sqrt2$ shows that the set
\[

 \qquad c_0=(c_p/2)^{1/2},

contains at least $(c_p/4)q_m\ge(c_p/4)m^p/p!$ indices.

\textbf{Lower bound for the sum.} For all real $z$,
\begin{equation*}
 \log\cosh(1)\min(z^2,|z|)\le\log\cosh z\le\min(z^2/2,|z|).
\end{equation*}
The upper bound is elementary. For the lower bound, the function
$g(z)=2\log\cosh z-z\tanh z$ satisfies $g(0)=0$ and
$g'(z)=\operatorname{sech}^2z\,(\tfrac12\sinh2z-z)\ge0$ for $z\ge0$, so
$(\log\cosh z)/z^2$ is nonincreasing on $(0,\infty)$; this gives the
bound for $|z|\le1$. For $|z|\ge1$, it follows from convexity of
$\log\cosh$ and $\log\cosh0=0$. For $\alpha\in\mathcal G$, since
$\log\cosh$ is increasing in $|z|$ and $c_0\le1$,
\[
 \log\cosh(u/2-|\alpha|v)\ge\log\cosh(c_0L_0)
 \ge\log\cosh(1)\,c_0^2\min(L_0^2,L_0).
\]
Since $L_0^2\ge(u^2+m^2v^2)/4$, we have
$\min(L_0^2,L_0)\ge\frac14\min\{u^2+m^2v^2,(u^2+m^2v^2)^{1/2}\}$.
Discarding the nonnegative terms with $\alpha\notin\mathcal G$ and using
$|\mathcal G|\ge(c_p/4)m^p/p!$ gives
\begin{equation}\label{eq:degree-separation}
 \sum_{|\alpha|\le m}\log\cosh(u/2-|\alpha|v)
 \ge cm^p\min\bigl\{u^2+m^2v^2,(u^2+m^2v^2)^{1/2}\bigr\}.
\end{equation}

By \Cref{lem:affinity-monotone}, the affinity of the observation laws
is at most the affinity of the laws of the parameter-independent linear
statistic $H_nY_n$. By \Cref{lem:recovery}, the latter log affinity differs from that of the exact
coefficient vector by $o(1)$, uniformly over the parameters. Combining
this fact with \eqref{eq:coefficient-affinity-product},
\eqref{eq:degree-separation}, and $m_n\asymp b_n$ proves
\eqref{eq:affinity-separation}. The coefficient law has no nugget
parameter, and its recovery bound is uniform in $t$; this proves the
claimed uniformity in the nugget as well.

 =\frac12h^\top\{\nabla_h^2L_n(0)+J_n\}h
 +\frac16\sum_{a,b,c}\partial^3_{h_ah_bh_c}L_n(\bar h)\,h_ah_bh_c
\]
for some $\bar h$ on the segment from $0$ to $h$. By
\eqref{eq:hessian-local}, \eqref{eq:third-local}, and
$\sum_a|h_a|\le\sqrt3\left\lVerth\right\rVert$, the right side is
$O_{\mathbb P}(\epsilon_nH_n^2+\epsilon_nH_n^3)=O_{\mathbb P}(\epsilon_nH_n^3)$ uniformly
on the ball, because $H_n\ge1$. This proves \eqref{eq:quadratic-remainder},
and \eqref{eq:uniform-quadratic} follows.

\textbf{Sharpness (\Cref{rem:radius}).}
Along the variance direction $h=ue_s$, the covariance is
$C(\sigma)=\sigma R_n(l_0)+t_0I_n$ with $\sigma=s_0+b_n^{-p/2}u$, so
$\partial_\sigma C=R_n(l_0)$ and $\partial_\sigma^2C=0$. Differentiating
$l=-\frac12\log\det C-\frac12Y_n^\top C^{-1}Y_n$ gives $\partial_\sigma^3l = -\operatorname{tr}\{(C^{-1}R_n)^3\}+3Y_n^\top C^{-1}(R_nC^{-1})^3Y_n$ and $\partial_\sigma^4l = 3\operatorname{tr}\{(C^{-1}R_n)^4\}-12Y_n^\top C^{-1}(R_nC^{-1})^4Y_n$,
               
                                                                                
                        
                                                                               
             
with $R_n=R_n(l_0)$. At $u=0$, in terms of $X_n=C_{n,0}^{-1/2}Y_n$ and
the normalized derivative $B_s$,
$\partial_{h_s}^3L_n(0)=-\operatorname{tr}(B_s^3)+3X_n^\top B_s^3X_n$. Its mean is
$2\operatorname{tr}(B_s^3)$, and its standard deviation is
$3\sqrt2\left\lVertB_s^3\right\rVert_\mathrm F\le C\left\lVertB_s\right\rVert_{\mathrm{op}}^2\left\lVertB_s\right\rVert_\mathrm F
\le C\epsilon_n^2$ by \Cref{lem:projection}. The matrix $B_s$ is positive
semidefinite. By the Cauchy--Schwarz inequality over its eigenvalues,
$(\operatorname{tr} B_s^2)^2\le\operatorname{tr}(B_s)\operatorname{tr}(B_s^3)$. Here $\operatorname{tr} B_s^2=2J_{n,ss}\ge2c$ by
\Cref{lem:information}. Moreover, $B_s=b_n^{-p/2}R_n(s_0R_n+t_0I_n)^{-1}$
has eigenvalues $b_n^{-p/2}\mu/(s_0\mu+t_0)$, where $\mu$ ranges over the
eigenvalues of $R_n$. By \eqref{eq:projection-approx} and Weyl's
inequality, at most $r_n$ of these $\mu$ exceed $n^{-20}/s_0$, so
$\operatorname{tr} B_s\le b_n^{-p/2}(r_n/s_0+n^{-19}/(s_0t_0))\le C\epsilon_n^{-1}$.
Hence $\operatorname{tr}(B_s^3)\ge c\epsilon_n$, and
$\partial_{h_s}^3L_n(0)\ge c\epsilon_n$ with probability tending to one,
after decreasing $c$.

For the fourth derivative, all matrices involved commute with $R_n$. In
an eigenbasis of $R_n$, with $Y_n$ having independent coordinates of
variances $s_0\mu_i+t_0$, each eigenvalue of $C^{-1}R_n$ is at most
$C\min(1,\mu_i)$ for $|\sigma-s_0|\le s_0/2$. Therefore
$\sup_{|\sigma-s_0|\le s_0/2}|\partial_\sigma^4l|$ is bounded by a
constant times $\sum_i\min(1,\mu_i^4)(1+\chi_i^2)$ with $\chi_i$
independent standard normal, which is $O_{\mathbb P}(r_n)$ by the eigenvalue
count above. After normalization, the fourth derivative in $h_s$ is
$O_{\mathbb P}(b_n^{-2p}r_n)=O_{\mathbb P}(\epsilon_n^2)$ uniformly on
$|u|\le\kappa\epsilon_n^{-1/3}$.

Now let $u=\kappa\epsilon_n^{-1/3}$, so that $\sigma-s_0=\kappa\epsilon_n^{2/3}\to0$.
Taylor's formula in $u$, together with \eqref{eq:hessian-local}, gives
\[
 L_n(ue_s)-L_n(0)-u\Delta_{n,s}+\frac12J_{n,ss}u^2
 =O_{\mathbb P}(\epsilon_nu^2)+\frac16\partial_{h_s}^3L_n(0)\,u^3
 +O_{\mathbb P}(\epsilon_n^2u^4).
\]
The first and last terms are $O_{\mathbb P}(\kappa^2\epsilon_n^{1/3})$ and
$O_{\mathbb P}(\kappa^4\epsilon_n^{2/3})$, and the middle term is at least
$c\kappa^3/6$ with probability tending to one. This proves
\Cref{rem:radius}.

 \frac{\sum_j\lambda_{j,n}^4}

Let $Q$ have side length $L$, and let $A\ge1$ be a fixed constant. For
$n\ge3$, put
\[
 k_n=\left\lfloor\left(\frac{n}{A\log n}\right)^{1/p}\right\rfloor,
\]
and suppose $k_n\ge1$. Partition $Q$ into $k_n^p$ closed subcubes
$Q_1,\ldots,Q_{k_n^p}$ of side length $L/k_n$, which overlap only on
their boundaries. Each has volume $L^p/k_n^p$, so its sampling
probability satisfies
\[
 \pi_j=\mathbb P(X_1\in Q_j)=\int_{Q_j}f(x)\,dx\ge\frac{f_-L^p}{k_n^p}.
\]
Since $X_1,\ldots,X_n$ are independent and $1-x\le e^{-x}$,
\[
 \mathbb P(\text{no }X_i\text{ lies in }Q_j)=(1-\pi_j)^n
 \le\Bigl(1-\frac{f_-L^p}{k_n^p}\Bigr)^n
 \le\exp\Bigl(-\frac{nf_-L^p}{k_n^p}\Bigr).
\]
A union bound over the $k_n^p$ subcubes gives
\[
 \mathbb P(\text{some }Q_j\text{ is empty})
 \le k_n^p\exp\Bigl(-\frac{nf_-L^p}{k_n^p}\Bigr).
\]
By the definition of $k_n$, $k_n^p\le n/(A\log n)$. Hence
$k_n^p\le n$, because $A\log n\ge1$, and $n/k_n^p\ge A\log n$, so that
$\exp(-nf_-L^p/k_n^p)\le\exp(-f_-L^pA\log n)=n^{-f_-L^pA}$. Therefore
\[
 \mathbb P(\text{some }Q_j\text{ is empty})
 \le n\exp(-f_-L^pA\log n)=n^{1-f_-L^pA}.
\]
Given $q>0$, choose $A\ge\max\{1,(q+1)/(f_-L^p)\}$, so that this
probability is at most $n^{-q}$. When every subcube contains an
observation, every $x\in Q$ lies in some $Q_j$ together with an
observation, so it is within the diameter $\sqrt p\,L/k_n$ of that
observation. Since $\lfloor y\rfloor\ge y/2$ for $y\ge1$,
\[
 h_n(Q)\le\frac{\sqrt p\,L}{k_n}\le2\sqrt p\,L\Bigl(\frac{A\log n}n\Bigr)^{1/p}
\]
on this event, for all $n\ge3$ with $k_n\ge1$. This proves the
probability bound with $A_q=2\sqrt p\,LA^{1/p}$, after enlarging $C_q$ to
cover the finitely many $n$ with $n<3$ or $k_n=0$. With $q=2$, the first Borel--Cantelli lemma proves
\eqref{eq:random-fill}. Since
$(\log n/n)^{1/p}=o(n^{-\kappa})$ for $\kappa<1/p$,
the coverage conclusion follows.
