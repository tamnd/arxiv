<!-- definition-1: body_md -->
\label{def:measure-mtp2}
	Let $\mu$ be a probability measure on $(\mathbb{R}^d,\mathcal B_d)$. We say that $\mu$ is measure-theoretically multivariate totally positive of order two, and write $\mathrm{mMTP}_2(\mu)$, if
	\begin{equation}
		\mu(C)\mu(D)
		\leq
		\mu(C\wedge D)\mu(C\vee D)
		\label{eq:measure-mtp2}
	\end{equation}
	for all $C,D\in\mathcal B_d$ such that
	$C\wedge D,C\vee D\in\mathcal B_d$. A random vector $\boldsymbol{X}$ is said to be $\mathrm{mMTP}_2$, and we write $\mathrm{mMTP}_2(\boldsymbol{X})$, if its distribution $\mathbb P_{\boldsymbol{X}}$ satisfies $\mathrm{mMTP}_2(\mathbb P_{\boldsymbol{X}})$.
<!-- end definition-1 -->
<!-- remark-1: body_md -->
\label{rem:measure-mtp2}
	\begin{enumerate}
		
		\noindent\item[(i)]
		For $\boldsymbol{x}\in\mathbb{R}^d$, define
		$L_{\boldsymbol{x}}:=(-\infty,\boldsymbol{x}]=\prod_{i=1}^d(-\infty,x_i]$.
		Then, for every $\boldsymbol{x},\boldsymbol{y}\in\mathbb{R}^d$,
		$L_{\boldsymbol{x}}\wedge L_{\boldsymbol{y}}=L_{\boldsymbol{x}\wedge\boldsymbol{y}}$ and
		$L_{\boldsymbol{x}}\vee L_{\boldsymbol{y}}=L_{\boldsymbol{x}\vee\boldsymbol{y}}$.
		Consequently, $\mathrm{mMTP}_2(\mu)$ and
		\eqref{eq:measure-mtp2} imply
		$F_{\mu}(\boldsymbol{x})F_{\mu}(\boldsymbol{y})
		\leq
		F_{\mu}(\boldsymbol{x}\wedge\boldsymbol{y})F_{\mu}(\boldsymbol{x}\vee\boldsymbol{y})$,
		where
		$F_{\mu}(\boldsymbol{x}):=\mu(L_{\boldsymbol{x}})$.
		Thus, viewed as the mapping
		$\boldsymbol{x}\mapsto F_{\mu}(\boldsymbol{x})$,
		the distribution function of $\mu$ is $\mathrm{MTP}_2$. The converse does not follow directly from the definition, since
		$\mathrm{mMTP}_2(\mu)$ requires \eqref{eq:measure-mtp2} for all
		admissible pairs $C,D\in\mathcal B_d$, whereas the distribution function tests only the lower orthants $L_{\boldsymbol{x}}$ and $L_{\boldsymbol{y}}$.
		
		\item[(ii)]
		Let $\mu_i:=\mu\circ\pi_i^{-1}$, $i\in[d]$, and set $\nu:=\bigotimes_{i=1}^d\mu_i$. Definition~\ref{def:measure-mtp2} requires neither $\mu\boldsymbol{l}\nu$ nor the existence of a Radon-Nikodym derivative
		$f:=d\mu/d\nu$. If $\mu\boldsymbol{l}\nu$, the usual density-based condition is
		$f(\boldsymbol{x})f(\boldsymbol{y})\leq
		f(\boldsymbol{x}\wedge\boldsymbol{y})f(\boldsymbol{x}\vee\boldsymbol{y})$
		for $\boldsymbol{x},\boldsymbol{y}\in\mathbb{R}^d$, after choosing a suitable version of $f$. The relation between this pointwise property and
		$\mathrm{mMTP}_2(\mu)$ is not assumed here and will be established under appropriate conditions in Section~\ref{subsec:density-equivalence}.
		In particular, the present construction is meaningful when $\mu\perp\nu$, which is excluded by the density-based formulation in \citet{ColangeloEtAl2006}.
		
		\item[(iii)]
		Define
		$T_{\wedge}(\boldsymbol{x},\boldsymbol{y}):=\boldsymbol{x}\wedge\boldsymbol{y}$ and
		$T_{\vee}(\boldsymbol{x},\boldsymbol{y}):=\boldsymbol{x}\vee\boldsymbol{y}$ for
		$(\boldsymbol{x},\boldsymbol{y})\in\mathbb{R}^d\times\mathbb{R}^d$.
		Although $T_{\wedge}$ and $T_{\vee}$ are continuous, for
		$C,D\in\mathcal B_d$ the identities
		$C\wedge D=T_{\wedge}(C\times D)$ and
		$C\vee D=T_{\vee}(C\times D)$
		do not guarantee that $C\wedge D$ and $C\vee D$ are Borel. Accordingly, \eqref{eq:measure-mtp2} is imposed only when $C\wedge D,C\vee D\in\mathcal B_d$.The same measurability convention will be used for the conditional
		measure-theoretic formulation introduced in
		Section~\ref{subsec:definition}.
		
	\end{enumerate}

<!-- lemma-1: body_md -->
\label{lem:mmtp2-marginal}
	If $\mathrm{mMTP}_2(\mu)$ holds, then
	$\mathrm{mMTP}_2(\mu_A)$ holds for every non-empty
	$A\subseteq[d]$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Fix $\varnothing\neq A\subseteq[d]$ and let
	$C,D\in\mathcal B_A$ be such that
	$C\wedge D,C\vee D\in\mathcal B_A$. Set
	$\widetilde C:=\pi_A^{-1}(C)$ and
	$\widetilde D:=\pi_A^{-1}(D)$. Then
	$\widetilde C,\widetilde D\in\mathcal B_d$ and, since the canonical
	projection preserves coordinatewise lattice operations,
	$\widetilde C\wedge\widetilde D
	=\pi_A^{-1}(C\wedge D)$ and
	$\widetilde C\vee\widetilde D
	=\pi_A^{-1}(C\vee D)$.
	Hence, by Definition~\ref{def:measure-mtp2},
	$\mu_A(C)\mu_A(D)
	\leq
	\mu_A(C\wedge D)\mu_A(C\vee D)$.
	Therefore $\mathrm{mMTP}_2(\mu_A)$ holds.
<!-- end proof-1 -->
<!-- definition-2: body_md -->
\label{def:conditional-measure-mtp2}
	Let $A$ and $B$ be non-empty sets forming a partition of $[d]$. We say that $\boldsymbol{X}=(\boldsymbol{X}_A,\boldsymbol{X}_B)$ is conditionally measure-theoretically multivariate totally positive of order two, and
	write $\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$, if
	\begin{equation}
		\begin{split}
			&\mathbb P(\boldsymbol{X}_A\in C,\boldsymbol{X}_B\in L_{\boldsymbol{x}_B})
			\mathbb P(\boldsymbol{X}_A\in D,\boldsymbol{X}_B\in L_{\boldsymbol{y}_B})\leq
			\mathbb P(\boldsymbol{X}_A\in C\wedge D,
			\boldsymbol{X}_B\in L_{\boldsymbol{x}_B\wedge\boldsymbol{y}_B})
			\mathbb P(\boldsymbol{X}_A\in C\vee D,
			\boldsymbol{X}_B\in L_{\boldsymbol{x}_B\vee\boldsymbol{y}_B})
		\end{split}
		\label{eq:cmmtp2}
	\end{equation}
	for all $\boldsymbol{x}_B,\boldsymbol{y}_B\in\mathbb{R}^B$ and all $C,D\in\mathcal B_A$ such that $C\wedge D,C\vee D\in\mathcal B_A$.
<!-- end definition-2 -->
<!-- remark-2: body_md -->
\label{rem:conditional-measure-mtp2}
	\begin{enumerate}
		
		\noindent \item[(i)]
		Although the notation
		$\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$ emphasizes the partition $(A,B)$, the property is formulated entirely in terms of the joint
		law $\mathbb P_{\boldsymbol{X}}$. Indeed, if
		$\mu=\mathbb P_{\boldsymbol{X}}$, then \eqref{eq:cmmtp2} is equivalent to
		\begin{equation}
			\begin{split}
				&\mu(C\times L_{\boldsymbol{x}_B})\mu(D\times L_{\boldsymbol{y}_B})\leq
				\mu((C\wedge D)\times L_{\boldsymbol{x}_B\wedge\boldsymbol{y}_B})
				\mu((C\vee D)\times L_{\boldsymbol{x}_B\vee\boldsymbol{y}_B}),
			\end{split}
			\label{eq:cmmtp2-joint-law}
		\end{equation}
		for every admissible $C,D\in\mathcal B_A$ and
		$\boldsymbol{x}_B,\boldsymbol{y}_B\in\mathbb{R}^B$. More precisely,
		$\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$ is invariant under the choice of a version of the regular conditional distribution $K_{B\mid A}$.
		
		\item[(ii)]
		For $\boldsymbol{x}_A,\boldsymbol{y}_A\in\mathbb{R}^A$, set
		$C=L_{\boldsymbol{x}_A}$ and $D=L_{\boldsymbol{y}_A}$. Since,
		$L_{\boldsymbol{x}_A}\wedge L_{\boldsymbol{y}_A}=L_{\boldsymbol{x}_A\wedge\boldsymbol{y}_A}$, and
		$L_{\boldsymbol{x}_A}\vee L_{\boldsymbol{y}_A}=L_{\boldsymbol{x}_A\vee\boldsymbol{y}_A}$,
		\eqref{eq:cmmtp2} yields
		$F_{\boldsymbol{X}}(\boldsymbol{x}_A,\boldsymbol{x}_B)F_{\boldsymbol{X}}(\boldsymbol{y}_A,\boldsymbol{y}_B)
		\leq
		F_{\boldsymbol{X}}(\boldsymbol{x}_A\wedge\boldsymbol{y}_A,\boldsymbol{x}_B\wedge\boldsymbol{y}_B)
		F_{\boldsymbol{X}}(\boldsymbol{x}_A\vee\boldsymbol{y}_A,\boldsymbol{x}_B\vee\boldsymbol{y}_B)$. Hence,
		$\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$ implies
		$\mathrm{MTP}_2(\boldsymbol{X})$ in the distribution-function sense.
		
		\item[(iii)]
		The conditioning coordinates are not passive in
		\eqref{eq:cmmtp2}. Let $\mathbf t_B=(t_i)_{i\in B}$ with $t_i\uparrow\infty$ for every $i\in B$. Then $L_{\mathbf t_B}\uparrow\mathbb{R}^B$, and continuity from below implies
		$\mathbb P(\boldsymbol{X}_A\in C,\boldsymbol{X}_B\in L_{\mathbf t_B})
		\to\mu_A(C)$. Invoking \eqref{eq:cmmtp2} with
		$\boldsymbol{x}_B=\boldsymbol{y}_B=\mathbf t_B$ and passing to the limit yields $\mu_A(C)\mu_A(D)\leq
		\mu_A(C\wedge D)\mu_A(C\vee D)$.
		Therefore, $\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$ implies $\mathrm{mMTP}_2(\boldsymbol{X}_A)$.
		
	\end{enumerate}

<!-- proposition-1: body_md -->
\label{prop:cmmtp2-finite-measures}
	Let $A$ and $B$ be non-empty sets forming a partition of $[d]$. Then $\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$ holds if and only if
	\begin{equation}
		\begin{split}
			\mu_{\boldsymbol{x}_B}^{A}(C)\mu_{\boldsymbol{y}_B}^{A}(D)
			\leq
			\mu_{\boldsymbol{x}_B\wedge\boldsymbol{y}_B}^{A}(C\wedge D)
			\mu_{\boldsymbol{x}_B\vee\boldsymbol{y}_B}^{A}(C\vee D)
		\end{split}
		\label{eq:cmmtp2-finite-measures}
	\end{equation}
	for all $\boldsymbol{x}_B,\boldsymbol{y}_B\in\mathbb{R}^B$ and all
	$C,D\in\mathcal B_A$ such that
	$C\wedge D,C\vee D\in\mathcal B_A$.
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
By the definition of $\mu_{\boldsymbol{x}_B}^{A}$,
	$\mu_{\boldsymbol{x}_B}^{A}(C)=
	\mathbb P(\boldsymbol{X}_A\in C,\boldsymbol{X}_B\in L_{\boldsymbol{x}_B})$.
	Applying the same identity to
	$(\boldsymbol{y}_B,D)$,
	$(\boldsymbol{x}_B\wedge\boldsymbol{y}_B,C\wedge D)$, and
	$(\boldsymbol{x}_B\vee\boldsymbol{y}_B,C\vee D)$ shows that
	\eqref{eq:cmmtp2-finite-measures} is precisely
	\eqref{eq:cmmtp2}. The two conditions are therefore equivalent.

<!-- lemma-2: body_md -->
\label{lem:one-dimensional-lattice}
	Let $\lambda$ be a probability measure on
	$(\mathbb{R},\mathcal B(\mathbb{R}))$. Then, for all
	$E,F\in\mathcal B(\mathbb{R})$ such that
	$E\wedge F,E\vee F\in\mathcal B(\mathbb{R})$,
	\begin{equation}
		\lambda(E)\lambda(F)
		\leq
		\lambda(E\wedge F)\lambda(E\vee F).
		\label{eq:one-dimensional-lattice}
	\end{equation}
	In particular, every probability measure on $\mathbb{R}$ is
	measure-theoretically $\mathrm{MTP}_2$.
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
Equip $\mathbb{R}$ with its usual total order. The constant function
	$h\equiv1$ is $\mathrm{MTP}_2$ with respect to $\lambda$, since
	$h(s)h(t)=h(s\wedge t)h(s\vee t)=1$ for all $s,t\in\mathbb{R}$.
	The integral form of the basic composition inequality for
	$\mathrm{MTP}_2$ functions therefore gives
	$\lambda(E)\lambda(F)
	\leq
	\lambda(E\wedge F)\lambda(E\vee F)$
	for every admissible pair $E,F$. Hence the asserted
	measure-theoretic property follows.

<!-- proposition-2: body_md -->
\label{prop:common-factor}
	Let $Z$ be a real-valued random variable and let
	$\boldsymbol{X}=(g_1(Z),\ldots,g_d(Z))$, where
	$g_i:\mathbb{R}\to\mathbb{R}$, $i\in[d]$, are non-decreasing Borel measurable functions \citep{dunbar2026modeling}. Then
	$\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$ holds for every pair of non-empty sets $A,B$ forming a partition of $[d]$.
<!-- end proposition-2 -->
<!-- proof-4: body_md -->
Let $\lambda=\mathbb P_Z$ and fix a partition $(A,B)$. For $\boldsymbol{x}_B,\boldsymbol{y}_B\in\mathbb{R}^B$ and admissible $C,D\in\mathcal B_A$, define
	$E:=\{z:g_A(z)\in C,\ g_B(z)\leq\boldsymbol{x}_B\}$ and
	$F:=\{z:g_A(z)\in D,\ g_B(z)\leq\boldsymbol{y}_B\}$ \citep{powell2025genomic,pramanik2024analysis}.
	Then $\mu_{\boldsymbol{x}_B}^{A}(C)=\lambda(E)$, and
	$\mu_{\boldsymbol{y}_B}^{A}(D)=\lambda(F)$ \citep{powell2026role,pramanik2026bayesian}. Since every $g_i$ is non-decreasing,
	$g_i(s\wedge t)=g_i(s)\wedge g_i(t)$ and
	$g_i(s\vee t)=g_i(s)\vee g_i(t)$ for all $s,t\in\mathbb{R}$. Consequently, $E\wedge F$ is contained in $\{z:g_A(z)\in C\wedge D,\
	g_B(z)\leq\boldsymbol{x}_B\wedge\boldsymbol{y}_B\}$, where
	$E\vee F$ is contained in
	$\{z:g_A(z)\in C\vee D,\
	g_B(z)\leq\boldsymbol{x}_B\vee\boldsymbol{y}_B\}$. Lemma~\ref{lem:one-dimensional-lattice} therefore yields
	\begin{equation}
		\begin{split}
			\mu_{\boldsymbol{x}_B}^{A}(C)\mu_{\boldsymbol{y}_B}^{A}(D)
			&=\lambda(E)\lambda(F)\leq\lambda(E\wedge F)\lambda(E\vee F)\leq
			\mu_{\boldsymbol{x}_B\wedge\boldsymbol{y}_B}^{A}(C\wedge D)
			\mu_{\boldsymbol{x}_B\vee\boldsymbol{y}_B}^{A}(C\vee D).
		\end{split}
		\label{eq:common-factor-cmmtp2}
	\end{equation}
	The assertion follows from Proposition~\ref{prop:cmmtp2-finite-measures}.

<!-- theorem-1: body_md -->
\label{thm:conditioning-reduction}
	Let $A$ and $B$ be non-empty sets forming a partition of $[d]$, with
	$|A|\geq2$. If $\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$ holds, then, for every $k\in A$,
	$\mathrm{cmMTP}_2
	(\boldsymbol{X}_{B\cup\{k\}}\mid\boldsymbol{X}_{A\setminus\{k\}})$ holds.
<!-- end theorem-1 -->
<!-- proof-5: body_md -->
Fix $k\in A$ and set $A':=A\setminus\{k\}$ and
	$B':=B\cup\{k\}$. Let $C,D\in\mathcal B_{A'}$ be such that
	$C\wedge D,C\vee D\in\mathcal B_{A'}$, and let
	$\boldsymbol{x}_{B'},\boldsymbol{y}_{B'}\in\mathbb{R}^{B'}$. Write
	$\boldsymbol{x}_{B'}=(\boldsymbol{x}_B,x_k)$ and
	$\boldsymbol{y}_{B'}=(\boldsymbol{y}_B,y_k)$, and define
	$C_x:=C\times(-\infty,x_k]$ and
	$D_y:=D\times(-\infty,y_k]$.
	Then $C_x,D_y\in\mathcal B_A$. We first identify the lattice operations on these sets. By the
	componentwise definition of $\wedge$ and $\vee$,
	\begin{equation}
		\begin{split}
			C_x\wedge D_y
			&=(C\wedge D)\times(-\infty,x_k\wedge y_k],\\
			C_x\vee D_y
			&=(C\vee D)\times(-\infty,x_k\vee y_k].
		\end{split}
		\label{eq:conditioning-reduction-sets}
	\end{equation}
	In particular, both sets in
	\eqref{eq:conditioning-reduction-sets} belong to $\mathcal B_A$ \citep{vikramdeo2024mitochondrial}. Hence, $C_x$ and $D_y$ form an admissible pair in
	Definition~\ref{def:conditional-measure-mtp2} \citep{pramanik2024estimation,pramanik2025optimal}. Next observe that
	$\{\boldsymbol{X}_{A'}\in C,\boldsymbol{X}_{B'}\in L_{\boldsymbol{x}_{B'}}\}
	=
	\{\boldsymbol{X}_A\in C_x,\boldsymbol{X}_B\in L_{\boldsymbol{x}_B}\}$,
	and analogously
	$\{\boldsymbol{X}_{A'}\in D,\boldsymbol{X}_{B'}\in L_{\boldsymbol{y}_{B'}}\}
	=
	\{\boldsymbol{X}_A\in D_y,\boldsymbol{X}_B\in L_{\boldsymbol{y}_B}\}$ \citep{valdez2025association,valdez2025exploring}. Since, $\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$ holds,
	\eqref{eq:cmmtp2} applied to $C_x,D_y,\boldsymbol{x}_B,\boldsymbol{y}_B$ gives
	\begin{equation}
		\begin{split}
			&\mathbb P(\boldsymbol{X}_A\in C_x,\boldsymbol{X}_B\in L_{\boldsymbol{x}_B})
			\mathbb P(\boldsymbol{X}_A\in D_y,\boldsymbol{X}_B\in L_{\boldsymbol{y}_B})\leq
			\mathbb P(\boldsymbol{X}_A\in C_x\wedge D_y,
			\boldsymbol{X}_B\in L_{\boldsymbol{x}_B\wedge\boldsymbol{y}_B})\times
			\mathbb P(\boldsymbol{X}_A\in C_x\vee D_y,
			\boldsymbol{X}_B\in L_{\boldsymbol{x}_B\vee\boldsymbol{y}_B}).
		\end{split}
		\label{eq:conditioning-reduction-inequality}
	\end{equation}
Invoking \eqref{eq:conditioning-reduction-sets}, the first probability on the right-hand side of
	\eqref{eq:conditioning-reduction-inequality} equals
	$\mathbb P(\boldsymbol{X}_{A'}\in C\wedge D,
	\boldsymbol{X}_{B'}\in L_{\boldsymbol{x}_{B'}\wedge\boldsymbol{y}_{B'}})$,
	whereas the second equals
	$\mathbb P(\boldsymbol{X}_{A'}\in C\vee D,
	\boldsymbol{X}_{B'}\in L_{\boldsymbol{x}_{B'}\vee\boldsymbol{y}_{B'}})$. The two probabilities on the left-hand side are similarly
	$\mathbb P(\boldsymbol{X}_{A'}\in C,\boldsymbol{X}_{B'}\in L_{\boldsymbol{x}_{B'}})$ \citep{pramanik2024stochastic}
	and
	$\mathbb P(\boldsymbol{X}_{A'}\in D,\boldsymbol{X}_{B'}\in L_{\boldsymbol{y}_{B'}})$.
	Consequently, \eqref{eq:conditioning-reduction-inequality} is exactly the defining lattice inequality for
	$\mathrm{cmMTP}_2(\boldsymbol{X}_{B'}\mid\boldsymbol{X}_{A'})$. Since, $C,D$ and $\boldsymbol{x}_{B'},\boldsymbol{y}_{B'}$ were arbitrary, the assertion follows \citep{pramanik2025factors,pramanik2025strategies}.

<!-- theorem-2: body_md -->
\label{thm:cmmtp2-marginalization}
	Let $A$ and $B$ be non-empty sets forming a partition of $[d]$. For a
	$d$-dimensional random vector $\boldsymbol{X}$, we have:
	\begin{enumerate}
		\item[(i)]
		If $\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$ holds, then
		$\mathrm{cmMTP}_2(\boldsymbol{X}_{B'}\mid\boldsymbol{X}_{A'})$ holds for all non-empty
		$A'\subseteq A$ and $B'\subseteq B$.
		
		\item[(ii)]
		If $\mathrm{cmMTP}_2(\boldsymbol{X})$ holds, then
		$\mathrm{cmMTP}_2(\boldsymbol{X}_I)$ holds for every
		$I\subseteq[d]$ with $|I|\geq2$.
	\end{enumerate}
<!-- end theorem-2 -->
<!-- proof-7: body_md -->
We first prove (i). Suppose that
$\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$ holds. We begin with
marginalization in the conditioned block. The argument follows the usual marginalization principle for $\mathrm{MTP}_2$ measures, but is applied here to the mixed set-orthant formulation in
Definition~\ref{def:conditional-measure-mtp2}
\citep{ColangeloEtAl2006,FallatEtAl2017}.
Let $\varnothing\neq B'\subseteq B$ and set $J:=B\setminus B'$. Fix $\boldsymbol{x}_{B'},\boldsymbol{y}_{B'}\in\mathbb{R}^{B'}$ and admissible
$C,D\in\mathcal B_A$. For $n\geq1$, define
$\boldsymbol{x}_B^{(n)},\boldsymbol{y}_B^{(n)}\in\mathbb{R}^B$ by
$\boldsymbol{x}_B^{(n)}=(\boldsymbol{x}_{B'},n\boldsymbol{1}_J)$ and
$\boldsymbol{y}_B^{(n)}=(\boldsymbol{y}_{B'},n\boldsymbol{1}_J)$,
where $\boldsymbol{1}_J$ denotes the vector of ones indexed by $J$.
Definition~\ref{def:conditional-measure-mtp2} gives
	\begin{equation}
		\begin{split}
			&\mathbb P(\boldsymbol{X}_A\in C,\boldsymbol{X}_B\in L_{\boldsymbol{x}_B^{(n)}})
			\mathbb P(\boldsymbol{X}_A\in D,\boldsymbol{X}_B\in L_{\boldsymbol{y}_B^{(n)}})\\
			&\hspace{1cm}\leq
			\mathbb P(\boldsymbol{X}_A\in C\wedge D,
			\boldsymbol{X}_B\in L_{\boldsymbol{x}_B^{(n)}\wedge\boldsymbol{y}_B^{(n)}})
			\mathbb P(\boldsymbol{X}_A\in C\vee D,
			\boldsymbol{X}_B\in L_{\boldsymbol{x}_B^{(n)}\vee\boldsymbol{y}_B^{(n)}}).
		\end{split}
		\label{eq:conditioned-marginalization}
	\end{equation}
	Since $n\to\infty$, the four events in
	\eqref{eq:conditioned-marginalization} increase, respectively, to the
	corresponding events involving only $\boldsymbol{X}_{B'}$. Continuity from below
	therefore yields
	\begin{equation}
		\begin{split}
			&\mathbb P(\boldsymbol{X}_A\in C,\boldsymbol{X}_{B'}\in L_{\boldsymbol{x}_{B'}})
			\mathbb P(\boldsymbol{X}_A\in D,\boldsymbol{X}_{B'}\in L_{\boldsymbol{y}_{B'}})\\
			&\hspace{1cm}\leq
			\mathbb P(\boldsymbol{X}_A\in C\wedge D,
			\boldsymbol{X}_{B'}\in L_{\boldsymbol{x}_{B'}\wedge\boldsymbol{y}_{B'}})
			\mathbb P(\boldsymbol{X}_A\in C\vee D,
			\boldsymbol{X}_{B'}\in L_{\boldsymbol{x}_{B'}\vee\boldsymbol{y}_{B'}}).
		\end{split}
		\label{eq:conditioned-marginal-limit}
	\end{equation}
	Hence, $\mathrm{cmMTP}_2(\boldsymbol{X}_{B'}\mid\boldsymbol{X}_A)$ holds. Now let $\varnothing\neq A'\subseteq A$ and set
	$K:=A\setminus A'$. By Corollary~\ref{cor:iterated-conditioning-reduction},
	$\mathrm{cmMTP}_2(\boldsymbol{X}_{B\cup K}\mid\boldsymbol{X}_{A'})$ holds \citep{issah2026stochastic,pramanik2026viscosity,pramanik2026stochastic}. Applying the conditioned-block marginalization just proved, first retaining $B$ and then $B'\subseteq
	B$, yields $\mathrm{cmMTP}_2(\boldsymbol{X}_{B'}\mid\boldsymbol{X}_{A'})$. This proves (i). We now prove (ii). Suppose $\mathrm{cmMTP}_2(\boldsymbol{X})$ holds and fix
	$I\subseteq[d]$ with $|I|\geq2$. Let $A'$ and $B'$ be arbitrary non-empty sets forming a partition of $I$. Define $A:=A'\cup I^c$ and $B:=B'$. Then $A$ and $B$ form a partition of
	$[d]$, and Definition~\ref{def:cmmtp2-vector} yields
	$\mathrm{cmMTP}_2(\boldsymbol{X}_{B'}\mid\boldsymbol{X}_A)$. Applying part (i) with $A'\subseteq A$ gives
	$\mathrm{cmMTP}_2(\boldsymbol{X}_{B'}\mid\boldsymbol{X}_{A'})$. Since the partition	$(A',B')$ of $I$ was arbitrary,
	$\mathrm{cmMTP}_2(\boldsymbol{X}_I)$ follows from
	Definition~\ref{def:cmmtp2-vector}. This proves (ii).

<!-- theorem-3: body_md -->
\label{thm:basic-implications}
	Let $A$ and $B$ be non-empty sets forming a partition of $[d]$. For a
	$d$-dimensional random vector $\boldsymbol{X}$, we have:
	\begin{enumerate}
		\item[(i)]
		$\mathrm{mMTP}_2(\boldsymbol{X})$ implies
		$\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$.
		
		\item[(ii)]
		$\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$ implies
		$\mathrm{MTP}_2(\boldsymbol{X})$.
	\end{enumerate}
<!-- end theorem-3 -->
<!-- proof-8: body_md -->
We first prove (i). Let $\mu=\mathbb P_{\boldsymbol{X}}$ and suppose that
	$\mathrm{mMTP}_2(\mu)$ holds. Fix
	$\boldsymbol{x}_B,\boldsymbol{y}_B\in\mathbb{R}^B$ and
	$C,D\in\mathcal B_A$ such that
	$C\wedge D,C\vee D\in\mathcal B_A$. Define
	$E:=C\times L_{\boldsymbol{x}_B}$ and
	$F:=D\times L_{\boldsymbol{y}_B}$. Then $E,F\in\mathcal B_d$, and the
	componentwise lattice operations give
	\begin{equation}
		\begin{split}
			E\wedge F
			&=(C\wedge D)\times L_{\boldsymbol{x}_B\wedge\boldsymbol{y}_B},\
			\ E\vee F
			=(C\vee D)\times L_{\boldsymbol{x}_B\vee\boldsymbol{y}_B}.
		\end{split}
		\label{eq:product-set-lattice}
	\end{equation}
	Hence $E\wedge F,E\vee F\in\mathcal B_d$ \citep{pramanik2022lock,pramanik2022stochastic}. Applying
	Definition~\ref{def:measure-mtp2} to $E$ and $F$ yields
	\begin{equation}
		\begin{split}
			&\mathbb P(\boldsymbol{X}_A\in C,\boldsymbol{X}_B\in L_{\boldsymbol{x}_B})
			\mathbb P(\boldsymbol{X}_A\in D,\boldsymbol{X}_B\in L_{\boldsymbol{y}_B})\leq
			\mathbb P(\boldsymbol{X}_A\in C\wedge D,
			\boldsymbol{X}_B\in L_{\boldsymbol{x}_B\wedge\boldsymbol{y}_B})
			\mathbb P(\boldsymbol{X}_A\in C\vee D,
			\boldsymbol{X}_B\in L_{\boldsymbol{x}_B\vee\boldsymbol{y}_B}).
		\end{split}
		\label{eq:mmtp2-implies-cmmtp2}
	\end{equation}
	Since $C,D,\boldsymbol{x}_B,\boldsymbol{y}_B$ were arbitrary \citep{pramanik2024dependence,pramanik2024measuring},
	\eqref{eq:mmtp2-implies-cmmtp2} is precisely
	\eqref{eq:cmmtp2} \citep{pramanik2021consensus,pramanik2021optimization}. Therefore, $\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$ holds. We now prove (ii). Suppose
	$\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$ holds \citep{pramanik2016tail}. For
	$\boldsymbol{x}=(\boldsymbol{x}_A,\boldsymbol{x}_B)$, and
	$\boldsymbol{y}=(\boldsymbol{y}_A,\boldsymbol{y}_B)$ in $\mathbb{R}^d$, take
	$C=L_{\boldsymbol{x}_A}$ and $D=L_{\boldsymbol{y}_A}$ in
	Definition~\ref{def:conditional-measure-mtp2}. Since,
	$L_{\boldsymbol{x}_A}\wedge L_{\boldsymbol{y}_A}=L_{\boldsymbol{x}_A\wedge\boldsymbol{y}_A}$ and
	$L_{\boldsymbol{x}_A}\vee L_{\boldsymbol{y}_A}=L_{\boldsymbol{x}_A\vee\boldsymbol{y}_A}$, we obtain $F_{\boldsymbol{X}}(\boldsymbol{x})F_{\boldsymbol{X}}(\boldsymbol{y})\leq
	F_{\boldsymbol{X}}(\boldsymbol{x}\wedge\boldsymbol{y})F_{\boldsymbol{X}}(\boldsymbol{x}\vee\boldsymbol{y})$.
	Hence, the mapping $\boldsymbol{x}\mapsto F_{\boldsymbol{X}}(\boldsymbol{x})$ is $\mathrm{MTP}_2$, which proves $\mathrm{MTP}_2(\boldsymbol{X})$.

<!-- corollary-2: body_md -->

	For a $d$-dimensional random vector $\boldsymbol{X}$,

<!-- lemma-3: body_md -->
\label{lem:cmtp2-to-cmmtp2}
	Suppose that $\mu_A\boldsymbol{l}\nu_A$ and that versions of $f_A$ and
	$K_{B\mid A}$ can be chosen such that
	$(\boldsymbol{x}_A,\boldsymbol{x}_B)\mapsto g_{\boldsymbol{x}_B}(\boldsymbol{x}_A)$ is
	$\mathrm{MTP}_2$. Then
	$\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$ holds.
<!-- end lemma-3 -->
<!-- proof-10: body_md -->
Let $\boldsymbol{x}_B,\boldsymbol{y}_B\in\mathbb{R}^B$ and let
	$C,D\in\mathcal B_A$ be such that
	$C\wedge D,C\vee D\in\mathcal B_A$. By disintegration, $\mu_{\boldsymbol{x}_B}^{A}$ and $\mu_{\boldsymbol{y}_B}^{A}$ are absolutely continuous
	with respect to $\nu_A$, with densities
	$g_{\boldsymbol{x}_B}$ and $g_{\boldsymbol{y}_B}$, respectively \citep{pramanik2023path}. Hence,
	$\mu_{\boldsymbol{x}_B}^{A}(C)=\int_C g_{\boldsymbol{x}_B}\,d\nu_A$ and
	$\mu_{\boldsymbol{y}_B}^{A}(D)=\int_D g_{\boldsymbol{y}_B}\,d\nu_A$ \citep{pramanik2021optimization,pramanik2020motivation,pramanik2024motivation}. Since $(\boldsymbol{x}_A,\boldsymbol{x}_B)\mapsto g_{\boldsymbol{x}_B}(\boldsymbol{x}_A)$ is
	$\mathrm{MTP}_2$, for $\boldsymbol{u}_A,\boldsymbol{v}_A\in\mathbb{R}^A$ we have
	\begin{equation}
		\begin{split}
			g_{\boldsymbol{x}_B}(\boldsymbol{u}_A)g_{\boldsymbol{y}_B}(\boldsymbol{v}_A)
			\leq
			g_{\boldsymbol{x}_B\wedge\boldsymbol{y}_B}(\boldsymbol{u}_A\wedge\boldsymbol{v}_A)
			g_{\boldsymbol{x}_B\vee\boldsymbol{y}_B}(\boldsymbol{u}_A\vee\boldsymbol{v}_A).
		\end{split}
		\label{eq:density-set-inequality}
	\end{equation}
As $\nu_A=\bigotimes_{i\in A}\mathbb P_{X_i}$ is a product measure, the set-integral form of the $\mathrm{MTP}_2$ lattice inequality applied to \eqref{eq:density-set-inequality} implies
	\begin{equation}
		\begin{split}
			&\left(\int_C g_{\boldsymbol{x}_B}\,d\nu_A\right)
			\left(\int_D g_{\boldsymbol{y}_B}\,d\nu_A\right)\leq
			\left(\int_{C\wedge D}
			g_{\boldsymbol{x}_B\wedge\boldsymbol{y}_B}\,d\nu_A\right)
			\left(\int_{C\vee D}
			g_{\boldsymbol{x}_B\vee\boldsymbol{y}_B}\,d\nu_A\right).
		\end{split}
		\label{eq:density-to-measure-integrals}
	\end{equation}
	This is the set-based integration argument underlying the density-free formulation of $\mathrm{MTP}_2$ (see \citet[Theorem~3.10.14]{MullerStoyan2002} and
	\citet{ColangeloEtAl2006}). Invoking
	$\mu_{\boldsymbol{z}_B}^{A}(E)=\int_Eg_{\boldsymbol{z}_B}\,d\nu_A$
	for $E\in\mathcal B_A$ and $\boldsymbol{z}_B\in\mathbb{R}^B$,
	\eqref{eq:density-to-measure-integrals} becomes
	$\mu_{\boldsymbol{x}_B}^{A}(C)\mu_{\boldsymbol{y}_B}^{A}(D)
	\leq
	\mu_{\boldsymbol{x}_B\wedge\boldsymbol{y}_B}^{A}(C\wedge D)
	\mu_{\boldsymbol{x}_B\vee\boldsymbol{y}_B}^{A}(C\vee D)$.
	Proposition~\ref{prop:cmmtp2-finite-measures} now yields $\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$. This completes the proof.

<!-- theorem-4: body_md -->
\label{thm:density-equivalence}
	Let $A$ and $B$ be non-empty sets forming a partition of $[d]$. Suppose, for every $i\in A$, $\mathbb P_{X_i}$ admits a strictly
	positive continuous Lebesgue density $p_i$, and define $\nu_A:=\bigotimes_{i\in A}\mathbb P_{X_i}$ \citep{pramanik2022lock,pramanik2023path1}. Assume  $\mu_A\boldsymbol{l}\nu_A$ and that versions of $f_A$ and $K_{B\mid A}$ can be
	chosen such that $g(\boldsymbol{x}_A,\boldsymbol{x}_B):=K_{B\mid A}(\boldsymbol{x}_A,L_{\boldsymbol{x}_B})f_A(\boldsymbol{x}_A)$ is continuous on $\mathbb{R}^A\times\mathbb{R}^B$. Then $\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$ holds if and only if $\mathrm{cMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$ holds.
<!-- end theorem-4 -->
<!-- proof-11: body_md -->
By Lemma~\ref{lem:cmtp2-to-cmmtp2}, it remains to prove that $\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$ implies
	$\mathrm{cMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$. Suppose $\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$ holds and fix
	$\boldsymbol{x}_A,\boldsymbol{y}_A\in\mathbb{R}^A$ and $\boldsymbol{x}_B,\boldsymbol{y}_B\in\mathbb{R}^B$. For $r>0$, define
	the rectangles
	$C_r:=\prod_{i\in A}[x_i-r,x_i+r]$ and
	$D_r:=\prod_{i\in A}[y_i-r,y_i+r]$ \citep{hertweck2023clinicopathological,pramanik2022stochastic,pramanik2023optimal,pramanik2024optimization}.
	Then $C_r,D_r\in\mathcal B_A$ and
	\begin{equation}
		\begin{split}
			C_r\wedge D_r
			&=\prod_{i\in A}
			[x_i\wedge y_i-r,x_i\wedge y_i+r],\ \
			C_r\vee D_r
			=\prod_{i\in A}
			[x_i\vee y_i-r,x_i\vee y_i+r].
		\end{split}
		\label{eq:local-rectangles}
	\end{equation}
	Thus, $C_r\wedge D_r$ and $C_r\vee D_r$ are admissible in Definition~\ref{def:conditional-measure-mtp2}. Let $p_A(\boldsymbol{z}_A):=\prod_{i\in A}p_i(z_i)$ \citep{pramanik2024estimation,pramanik2025construction}. Since
	$\nu_A(d\boldsymbol{z}_A)=p_A(\boldsymbol{z}_A)d\boldsymbol{z}_A$, continuity of $g$ and $p_A$ implies, for every $\boldsymbol{z}_A\in\mathbb{R}^A$ and $\boldsymbol{z}_B\in\mathbb{R}^B$,
	\begin{equation}
		\begin{split}
			\lim_{r\downarrow0}
			\frac{\mu_{\boldsymbol{z}_B}^{A}
				\left(\prod_{i\in A}[z_i-r,z_i+r]\right)}
			{(2r)^{|A|}}
			&=
			\lim_{r\downarrow0}
			\frac{1}{(2r)^{|A|}}
			\int_{\prod_{i\in A}[z_i-r,z_i+r]}
			g_{\boldsymbol{z}_B}(\boldsymbol{u}_A)p_A(\boldsymbol{u}_A)\,d\boldsymbol{u}_A\\
			&=
			g_{\boldsymbol{z}_B}(\boldsymbol{z}_A)p_A(\boldsymbol{z}_A).
		\end{split}
		\label{eq:local-density-limit}
	\end{equation}
	Applying Proposition~\ref{prop:cmmtp2-finite-measures} to
	$C_r,D_r,\boldsymbol{x}_B,\boldsymbol{y}_B$ and invoking
	\eqref{eq:local-rectangles}, we obtain
	\begin{equation}
		\begin{split}
			\mu_{\boldsymbol{x}_B}^{A}(C_r)\mu_{\boldsymbol{y}_B}^{A}(D_r)
			\leq
			\mu_{\boldsymbol{x}_B\wedge\boldsymbol{y}_B}^{A}(C_r\wedge D_r)
			\mu_{\boldsymbol{x}_B\vee\boldsymbol{y}_B}^{A}(C_r\vee D_r).
		\end{split}
		\label{eq:localized-cmmtp2}
	\end{equation}
	Divide \eqref{eq:localized-cmmtp2} by $(2r)^{2|A|}$ and let $r\downarrow0$. By \eqref{eq:local-density-limit},
	\begin{equation}
		\begin{split}
			&g_{\boldsymbol{x}_B}(\boldsymbol{x}_A)g_{\boldsymbol{y}_B}(\boldsymbol{y}_A)
			p_A(\boldsymbol{x}_A)p_A(\boldsymbol{y}_A)\leq
			g_{\boldsymbol{x}_B\wedge\boldsymbol{y}_B}(\boldsymbol{x}_A\wedge\boldsymbol{y}_A)
			g_{\boldsymbol{x}_B\vee\boldsymbol{y}_B}(\boldsymbol{x}_A\vee\boldsymbol{y}_A)
			p_A(\boldsymbol{x}_A\wedge\boldsymbol{y}_A)p_A(\boldsymbol{x}_A\vee\boldsymbol{y}_A).
		\end{split}
		\label{eq:localized-density-inequality}
	\end{equation}
	Since $p_A$ is a product density,
	$p_A(\boldsymbol{x}_A)p_A(\boldsymbol{y}_A)= p_A(\boldsymbol{x}_A\wedge\boldsymbol{y}_A)p_A(\boldsymbol{x}_A\vee\boldsymbol{y}_A)$. Moreover, $p_A>0$ by assumption. Cancelling this common factor in \eqref{eq:localized-density-inequality} yields
	$g_{\boldsymbol{x}_B}(\boldsymbol{x}_A)g_{\boldsymbol{y}_B}(\boldsymbol{y}_A)
	\leq
	g_{\boldsymbol{x}_B\wedge\boldsymbol{y}_B}(\boldsymbol{x}_A\wedge\boldsymbol{y}_A)
	g_{\boldsymbol{x}_B\vee\boldsymbol{y}_B}(\boldsymbol{x}_A\vee\boldsymbol{y}_A)$. Since $\boldsymbol{x}_A,\boldsymbol{y}_A,\boldsymbol{x}_B,\boldsymbol{y}_B$ were arbitrary, $g$ is
	$\mathrm{MTP}_2$. Hence,
	$\mathrm{cMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$ holds.

<!-- theorem-5: body_md -->
\label{thm:cmmtp2-weak-closure}
	Let $A$ and $B$ be non-empty sets forming a partition of $[d]$, and
	let $\mu_n,\mu$ be probability measures on $\mathbb{R}^d$ such that $\mu_n\Rightarrow\mu$. Suppose
	$\mathrm{cmMTP}_2((\boldsymbol{X}_n)_B\mid(\boldsymbol{X}_n)_A)$ holds for every $n$, where $\boldsymbol{X}_n\sim\mu_n$. Then
	$\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$ holds for $\boldsymbol{X}\sim\mu$.
<!-- end theorem-5 -->
<!-- proof-14: body_md -->
By Lemma~\ref{lem:cmmtp2-determining-class}, it is enough to verify \eqref{eq:cmmtp2} for strongly disjoint noncomparable closed rectangles $C,D\subset\mathbb{R}^A$. Fix such $C,D$ and
	$\boldsymbol{x}_B,\boldsymbol{y}_B\in\mathbb{R}^B$. Choose decreasing sequences of strongly disjoint noncomparable open
	rectangles $C_m\downarrow C$ and $D_m\downarrow D$, and $\boldsymbol{x}_B^{(m)}\downarrow\boldsymbol{x}_B$,
	$\boldsymbol{y}_B^{(m)}\downarrow\boldsymbol{y}_B$, such that
	\begin{equation}
		\begin{aligned}
			E_{1m}&:=C_m\times L_{\boldsymbol{x}_B^{(m)}},&
			E_{2m}&:=D_m\times L_{\boldsymbol{y}_B^{(m)}},\\
			E_{3m}&:=(C_m\wedge D_m)\times
			L_{\boldsymbol{x}_B^{(m)}\wedge\boldsymbol{y}_B^{(m)}},&
			E_{4m}&:=(C_m\vee D_m)\times
			L_{\boldsymbol{x}_B^{(m)}\vee\boldsymbol{y}_B^{(m)}}
		\end{aligned}
		\label{eq:weak-continuity-approximation}
	\end{equation}
	are $\mu$-continuity sets. Such sequences exist since the boundaries of rectangles are contained in finite unions of coordinate hyperplanes, and each marginal distribution has at most countably
	many atoms. For every $n,m\geq1$,
	$\mathrm{cmMTP}_2((\boldsymbol{X}_n)_B\mid(\boldsymbol{X}_n)_A)$ gives
	\begin{equation}
		\mu_n(E_{1m})\mu_n(E_{2m})
		\leq
		\mu_n(E_{3m})\mu_n(E_{4m}).
		\label{eq:weak-closure-nm}
	\end{equation}
	Fixing $m$ and letting $n\to\infty$, the Portmanteau theorem yields
	$\mu_n(E_{jm})\to\mu(E_{jm})$, $j\in[4]$, and hence,
	$\mu(E_{1m})\mu(E_{2m})
	\leq
	\mu(E_{3m})\mu(E_{4m})$. Finally, $C_m\downarrow C$, $D_m\downarrow D$ imply
	$C_m\wedge D_m\downarrow C\wedge D$ and
	$C_m\vee D_m\downarrow C\vee D$. Letting $m\to\infty$ and using
	continuity from above gives
	\begin{equation}
		\begin{split}
			&\mu(C\times L_{\boldsymbol{x}_B})\mu(D\times L_{\boldsymbol{y}_B})\leq
			\mu((C\wedge D)\times L_{\boldsymbol{x}_B\wedge\boldsymbol{y}_B})
			\mu((C\vee D)\times L_{\boldsymbol{x}_B\vee\boldsymbol{y}_B}).
		\end{split}
		\label{eq:weak-closure-limit}
	\end{equation}
	Lemma~\ref{lem:cmmtp2-determining-class} now yields
	$\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$.

<!-- remark-3: body_md -->
\label{rem:weak-closure}
	Theorem~\ref{thm:cmmtp2-weak-closure} requires neither $\mu_n\boldsymbol{l}\lambda_d$ nor $\mu\boldsymbol{l}\lambda_d$. Thus it may happen that
	$\mu_n\boldsymbol{l}\lambda_d$ for every $n$, while
	$\mu_n\Rightarrow\mu$ and $\mu\perp\lambda_d$. More precisely, $\mu_n\Rightarrow\mu$ and
	$\mathrm{cmMTP}_2((\boldsymbol{X}_n)_B\mid(\boldsymbol{X}_n)_A)$ for every $n$ imply $\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$ even when $\mu$ is singular. Moreover, under the hypotheses of Lemma~\ref{lem:cmtp2-to-cmmtp2},
	$\mathrm{cMTP}_2((\boldsymbol{X}_n)_B\mid(\boldsymbol{X}_n)_A)$ implies
	$\mathrm{cmMTP}_2((\boldsymbol{X}_n)_B\mid(\boldsymbol{X}_n)_A)$. Hence, the weak limit retains the measure theoretic conditional property even when a density-based approach is unavailable at $\mu$.

\label{ex:weak-gaussian-limit}
	Let $Z,E_1,E_2$ be independent standard normal random variables and, for $\varepsilon>0$, define
	$X_{\varepsilon}:=Z+\varepsilon E_1$ and
	$Y_{\varepsilon}:=Z+\varepsilon E_2$. Then
	$(X_{\varepsilon},Y_{\varepsilon})\sim
	\mathcal N(\boldsymbol{0},\Sigma_{\varepsilon})$, where
	\begin{equation}
		\Sigma_{\varepsilon}
		=
		\begin{pmatrix}
			1+\varepsilon^2 & 1\\
			1 & 1+\varepsilon^2
		\end{pmatrix},
		\qquad
		\Sigma_{\varepsilon}^{-1}
		=
		\frac{1}{\varepsilon^2(2+\varepsilon^2)}
		\begin{pmatrix}
			1+\varepsilon^2 & -1\\
			-1 & 1+\varepsilon^2
		\end{pmatrix}.
		\label{eq:gaussian-weak-example}
	\end{equation}
	Hence, $\Sigma_{\varepsilon}^{-1}$ is an $M$-matrix and $(X_{\varepsilon},Y_{\varepsilon})$ is $\mathrm{dMTP}_2$ for every
	$\varepsilon>0$ \citep{FallatEtAl2017}. In particular, Theorem~\ref{thm:basic-implications}(i) yields $\mathrm{cmMTP}_2(Y_{\varepsilon}\mid X_{\varepsilon})$. Consider $\varepsilon\in\{1,1/2,1/5,1/10\}$. The
	correlations are
	\begin{center}
		\begin{tabular}{c|cccc}
			$\varepsilon$
			& $1$ & $1/2$ & $1/5$ & $1/10$\\ \hline
			$\operatorname{Corr}(X_{\varepsilon},Y_{\varepsilon})$
			& $0.5000$ & $0.8000$ & $0.9615$ & $0.9901$
		\end{tabular}
	\end{center}
	and therefore converge to $1$ as $\varepsilon\downarrow0$. The corresponding lower-orthant probabilities provide a direct
	probability-level illustration of the convergence. Since
	$\rho_{\varepsilon}:=
	\operatorname{Corr}(X_{\varepsilon},Y_{\varepsilon})
	=(1+\varepsilon^2)^{-1}$, the bivariate Gaussian orthant formula gives
	$\mathbb P(X_{\varepsilon}\leq0,Y_{\varepsilon}\leq0)
	=1/4+(2\pi)^{-1}\arcsin(\rho_{\varepsilon})$. Hence,
	
	\begin{center}
		\begin{tabular}{c|cccc}
			$\varepsilon$
			& $1$ & $1/2$ & $1/5$ & $1/10$\\ \hline
			$\mathbb P(X_{\varepsilon}\leq0,Y_{\varepsilon}\leq0)$
			& $0.3333$ & $0.3976$ & $0.4557$ & $0.4776$
		\end{tabular}
	\end{center}
	
	\noindent As $\varepsilon\downarrow0$, $\rho_{\varepsilon}\uparrow1$ and
	therefore,
	$\mathbb P(X_{\varepsilon}\leq0,Y_{\varepsilon}\leq0)\to1/2$.
	For the limiting vector $(Z,Z)$,
	$\mathbb P(Z\leq0,Z\leq0)=\mathbb P(Z\leq0)=1/2$.	Moreover,
	$(X_{\varepsilon},Y_{\varepsilon})\Rightarrow(Z,Z)$.
	The limiting distribution is concentrated on
	$\{(z,z):z\in\mathbb{R}\}$, and hence is singular with respect to $\mathbb P_Z\otimes\mathbb P_Z$. Taking $A=\{1\}$ and $B=\{2\}$, Theorem~\ref{thm:cmmtp2-weak-closure} implies
	$\mathrm{cmMTP}_2(Z\mid Z)$ for the limiting random vector $(Z,Z)$. Therefore, every member of the approximating family admits a
	strictly positive Gaussian density, whereas the limit has no density with respect to the product of its marginals. Nevertheless, the conditional measure theoretic total positivity property is preserved.

\label{lem:monotone-lattice-map}

<!-- proposition-3: body_md -->
\label{prop:monotone-transformations}
	Let $A$ and $B$ be non-empty sets forming a partition of $[d]$, and

			C'\vee D'=\phi_A^{-1}(C\vee D).

		\label{eq:inverse-lattice-transform}

			&\mathbb P(\boldsymbol{Y}_A\in C,\boldsymbol{X}_B\in L_{\boldsymbol{x}_B})
			\mathbb P(\boldsymbol{Y}_A\in D,\boldsymbol{X}_B\in L_{\boldsymbol{y}_B})\leq
			\mathbb P(\boldsymbol{Y}_A\in C\wedge D,
			\boldsymbol{X}_B\in L_{\boldsymbol{x}_B\wedge\boldsymbol{y}_B})
			\mathbb P(\boldsymbol{Y}_A\in C\vee D,
			\boldsymbol{X}_B\in L_{\boldsymbol{x}_B\vee\boldsymbol{y}_B}).
		\end{split}
		\label{eq:transform-conditioning-block}
	\end{equation}
	Therefore, $\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{Y}_A)$ holds.
	We now prove (ii). For $t\in\mathbb{R}$ and $i\in B$, set
	$H_i(t):=\{s\in\mathbb{R}:\phi_i(s)\leq t\}$. Since $\phi_i$ is strictly

	$\boldsymbol{x}_B\in\mathbb{R}^B$,

	\end{equation}

	Lemma~\ref{lem:lower-ray-extension} to
	$H_B(\boldsymbol{x}_B)$ and $H_B(\boldsymbol{y}_B)$. Since
	$H_B(\boldsymbol{x}_B\wedge\boldsymbol{y}_B)

	$H_B(\boldsymbol{x}_B\vee\boldsymbol{y}_B)

	defining inequality for
	$\mathrm{cmMTP}_2(\boldsymbol{Y}_B\mid\boldsymbol{Y}_A)$.

\label{thm:markov-product}
	Let $A,B,C$ be pairwise disjoint non-empty index sets and suppose
	\begin{equation}
		\mathbb P(d\boldsymbol{x}_A,d\boldsymbol{x}_B,d\boldsymbol{x}_C)
		=
		\lambda(d\boldsymbol{x}_A)K_1(\boldsymbol{x}_A,d\boldsymbol{x}_B)K_2(\boldsymbol{x}_B,d\boldsymbol{x}_C),
		\label{eq:markov-product-law}
	\end{equation}
	where $\lambda$ is $\mathrm{mMTP}_2$,
	$K_1\in\mathcal K_{\mathrm{mLTP}_2}(A,B)$, and
	$K_2\in\mathcal K_{\mathrm{LTP}_2}(B,C)$. Then
	\begin{enumerate}
		\item[(i)]
		$K_{C\mid A}=K_1K_2\in
		\mathcal K_{\mathrm{LTP}_2}(A,C)$.
		\item[(ii)]
		$\mathrm{cmMTP}_2(\boldsymbol{X}_C\mid\boldsymbol{X}_A)$ holds.
	\end{enumerate}
<!-- end theorem-7 -->
<!-- proof-22: body_md -->
We first prove (i). By \eqref{eq:markov-product-law},
	$\boldsymbol{X}_A\perp\boldsymbol{X}_C\mid\boldsymbol{X}_B$, and disintegration gives, for $E\in\mathcal B_C$, $K_{C\mid A}(\boldsymbol{x}_A,E)
	=\int_{\mathbb{R}^B}K_2(\boldsymbol{x}_B,E)\\ K_1(\boldsymbol{x}_A,d\boldsymbol{x}_B)
	=(K_1K_2)(\boldsymbol{x}_A,E)$. Since,
	$K_1\in\mathcal K_{\mathrm{mLTP}_2}(A,B)$ and
	$K_2\in\mathcal K_{\mathrm{LTP}_2}(B,C)$,
	Proposition~\ref{prop:kernel-composition} yields
	$K_{C\mid A}\in\mathcal K_{\mathrm{LTP}_2}(A,C)$.
	We now prove (ii). For $\boldsymbol{x}_C\in\mathbb{R}^C$ and
	$E\in\mathcal B_A$, define
	$\eta_{\boldsymbol{x}_C}^{A}(E):=
	\mathbb P(\boldsymbol{X}_A\in E,\boldsymbol{X}_C\in L_{\boldsymbol{x}_C})$. By (i),
	\begin{equation}
		\eta_{\boldsymbol{x}_C}^{A}(E)
		=
		\int_EK_{C\mid A}(\boldsymbol{x}_A,L_{\boldsymbol{x}_C})\lambda(d\boldsymbol{x}_A).
		\label{eq:markov-product-finite-measure}
	\end{equation}

	$E,F\in\mathcal B_A$. Since
	$K_{C\mid A}\in\mathcal K_{\mathrm{LTP}_2}(A,C)$,
	\begin{equation}
		\begin{split}
			&K_{C\mid A}(\boldsymbol{x}_A,L_{\boldsymbol{x}_C})
			K_{C\mid A}(\boldsymbol{y}_A,L_{\boldsymbol{y}_C})\leq
			K_{C\mid A}(\boldsymbol{x}_A\wedge\boldsymbol{y}_A,L_{\boldsymbol{x}_C\wedge\boldsymbol{y}_C})
			K_{C\mid A}(\boldsymbol{x}_A\vee\boldsymbol{y}_A,L_{\boldsymbol{x}_C\vee\boldsymbol{y}_C}).
		\end{split}
		\label{eq:markov-product-pointwise}
	\end{equation}
	As $\lambda$ is $\mathrm{mMTP}_2$, the integral
	four-functions inequality applied to
	\eqref{eq:markov-product-pointwise} yields

		\begin{split}
			\eta_{\boldsymbol{x}_C}^{A}(E)\eta_{\boldsymbol{y}_C}^{A}(F)
			&\leq
			\eta_{\boldsymbol{x}_C\wedge\boldsymbol{y}_C}^{A}(E\wedge F)
			\eta_{\boldsymbol{x}_C\vee\boldsymbol{y}_C}^{A}(E\vee F).
		\end{split}
		\label{eq:markov-product-cmmtp2}
	\end{equation}
	Proposition~\ref{prop:cmmtp2-finite-measures} therefore yields $\mathrm{cmMTP}_2(\boldsymbol{X}_C\mid\boldsymbol{X}_A)$.

<!-- theorem-8: body_md -->
\label{thm:archimedean-cmmtp2}
	Let $\boldsymbol{U}=(U_1,\ldots,U_d)$ have a strict Archimedean copula $C_\psi$, and let $A$ and $B$ form a partition of $[d]$ with
	$|A|=k$. Suppose $\psi$ is $k$-times continuously differentiable on $(0,\infty)$, $h_k:=(-1)^k\psi^{(k)}>0$, and the conditional
	mapping required in Theorem~\ref{thm:density-equivalence} is
	continuous. Then the following statements are equivalent
	\begin{enumerate}
		\item[(i)] $\mathrm{cmMTP}_2(\boldsymbol{U}_B\mid\boldsymbol{U}_A)$;
		\item[(ii)] $\mathrm{cMTP}_2(\boldsymbol{U}_B\mid\boldsymbol{U}_A)$;
		\item[(iii)] $h_k$ is log-convex on $(0,\infty)$.
	\end{enumerate}
<!-- end theorem-8 -->
<!-- proof-24: body_md -->
Since $U_i\sim\operatorname{Unif}(0,1)$, $i\in A$, we have
	$\nu_A=\bigotimes_{i\in A}\mathbb P_{U_i}
	=\lambda^k|_{(0,1)^k}$, whose density is strictly positive and
	continuous. Hence, under the stated regularity assumptions,
	Theorem~\ref{thm:density-equivalence} gives
	$
	\mathrm{cmMTP}_2(\boldsymbol{U}_B\mid\boldsymbol{U}_A)
	\Longleftrightarrow
	\mathrm{cMTP}_2(\boldsymbol{U}_B\mid\boldsymbol{U}_A).
	$
	It remains to prove $(ii)\Longleftrightarrow(iii)$.
	For $\boldsymbol{u}_A\in(0,1)^k$ and $\boldsymbol{u}_B\in(0,1]^{d-k}$,
	differentiating \eqref{eq:archimedean-copula} with respect to $\boldsymbol{u}_A$ yields
	\begin{equation}
		\mathbb P(\boldsymbol{U}_B\leq\boldsymbol{u}_B\mid\boldsymbol{U}_A=\boldsymbol{u}_A)c_A(\boldsymbol{u}_A)
		=
		h_k\!\left(s_A(\boldsymbol{u}_A)+s_B(\boldsymbol{u}_B)\right)
		\prod_{i\in A}\{-\varphi'(u_i)\},
		\label{eq:archimedean-conditional-map}
	\end{equation}
	where
	$c_A(\boldsymbol{u}_A)
	=h_k(s_A(\boldsymbol{u}_A))
	\prod_{i\in A}\{-\varphi'(u_i)\}$.
	Therefore, the mapping defining $\mathrm{cMTP}_2$ is
	$$
	g(\boldsymbol{u}_A,\boldsymbol{u}_B)
	=
	h_k\!\left(s_A(\boldsymbol{u}_A)+s_B(\boldsymbol{u}_B)\right)
	\prod_{i\in A}\{-\varphi'(u_i)\}.
	$$
	The product factor is lattice-neutral, since, for
	$\boldsymbol{u}_A,\boldsymbol{v}_A\in(0,1)^k$,
	\[
	\prod_{i\in A}\{-\varphi'(u_i)\}
	\prod_{i\in A}\{-\varphi'(v_i)\}
	=
	\prod_{i\in A}\{-\varphi'(u_i\wedge v_i)\}
	\prod_{i\in A}\{-\varphi'(u_i\vee v_i)\}.
	\]
Consequently, the $\mathrm{MTP}_2$ inequality for $g$ reduces to
\begin{equation}
	\begin{split}
		h_k\!\left(s_A(\boldsymbol{u}_A)+s_B(\boldsymbol{u}_B)\right)
		h_k\!\left(s_A(\boldsymbol{v}_A)+s_B(\boldsymbol{v}_B)\right)&\leq
		h_k\!\left(s_A(\boldsymbol{u}_A\wedge\boldsymbol{v}_A)
		+s_B(\boldsymbol{u}_B\wedge\boldsymbol{v}_B)\right)\\
		&\hspace{1cm}\times
		h_k\!\left(s_A(\boldsymbol{u}_A\vee\boldsymbol{v}_A)
		+s_B(\boldsymbol{u}_B\vee\boldsymbol{v}_B)\right).
	\end{split}
	\label{eq:archimedean-hk-lattice}
\end{equation}
Since $\varphi$ is decreasing,
$s_A(\boldsymbol{u}_A\vee\boldsymbol{v}_A)+s_B(\boldsymbol{u}_B\vee\boldsymbol{v}_B)
\leq
\min\{r,s\}\leq
\max\{r,s\}\leq
s_A(\boldsymbol{u}_A\wedge\boldsymbol{v}_A)+s_B(\boldsymbol{u}_B\wedge\boldsymbol{v}_B)$,
where
$r:=s_A(\boldsymbol{u}_A)+s_B(\boldsymbol{u}_B)$ and
$s:=s_A(\boldsymbol{v}_A)+s_B(\boldsymbol{v}_B)$, while the sums of the two extreme arguments and of $r,s$ coincide. Hence,
\eqref{eq:archimedean-hk-lattice} is equivalent to convexity of
$\log h_k$ \citep{MullerScarsini2005,FuchsWang2027}. Thus
$(ii)\Longleftrightarrow(iii)$, and the assertion follows. Combining this with
	$(i)\Longleftrightarrow(ii)$ proves the assertion.

<!-- corollary-5: body_md -->
\label{cor:clayton-cmmtp2}
	Let $\boldsymbol{U}$ have the $d$-dimensional Clayton copula with parameter
	$\theta>0$. Then, for every non-trivial partition $(A,B)$ of
	$[d]$,
	$\mathrm{cmMTP}_2(\boldsymbol{U}_B\mid\boldsymbol{U}_A)$ holds.

<!-- proof-25: body_md -->
For $\theta>0$, the Clayton generator is
	$\psi_\theta(t)=(1+t)^{-1/\theta}$. If $|A|=k$, then
	\begin{equation}
		h_k(t)
		=
		(-1)^k\psi_\theta^{(k)}(t)
		=
		\left\{\prod_{j=0}^{k-1}
		\left(\frac{1}{\theta}+j\right)\right\}

\label{thm:gaussian-cmmtp2}
	Let $\boldsymbol{X}\sim N_d(\boldsymbol\mu,\Sigma)$, $\Sigma\succ0$,
	and $A\dot\cup B=[d]$, $A,B\neq\varnothing$.

	$M:=\Sigma_{BA}\Sigma_{AA}^{-1}$,
	$V:=\Sigma_{BB}-\Sigma_{BA}

			\quad i\neq j,\ i,j\in A,\\
			(-M^\top H(\boldsymbol{z}))_{ij}&\geq0,
			\quad i\in A,\ j\in B,\\
			H_{ij}(\boldsymbol{z})&\geq0,
			\quad i\neq j,\ i,j\in B.
		\end{split}
		\label{eq:gaussian-conditional-criterion}
	\end{equation}

		Theorem~\ref{thm:archimedean-cmmtp2}.

	$\max_{r\in[m]}[-Z_r]_+$, $Z\sim N_m(\mathbf0,R)$, is continuous

	\begin{enumerate}

		\item[(ii)] if $\boldsymbol\Delta=\mathbf0$,
		$\mathbb P_\mu(\varphi_n=1)\to\alpha$;
		\item[(iii)] if $\min_{r\in[m]}\Delta_r<0$,
		$\mathbb P_\mu(\varphi_n=1)\to1$.
	\end{enumerate}

\label{thm:local-power}
	Let $(\mu_n)$ be a sequence of laws with lattice contrasts
	$\boldsymbol\Delta_n$ and suppose
	$\sqrt n(\widehat{\boldsymbol\Delta}_n-\boldsymbol\Delta_n)
	\Rightarrow N_m(\mathbf0,\Omega)$,
	$\sqrt n\,\boldsymbol\Delta_n\to\boldsymbol\delta\in\mathbb{R}^m$, and
	$\widehat\Omega_n\stackrel{p}{\to}\Omega$, where
	$\sigma_r^2:=\Omega_{rr}>0$, $r\in[m]$. Let
	$D:=\operatorname{diag}(\sigma_1,\ldots,\sigma_m)$ and
	$R:=D^{-1}\Omega D^{-1}$. Then
	\begin{equation}
		T_n\Rightarrow
		T(\boldsymbol\delta):=
		\max_{r\in[m]}
		\left[-Z_r-\frac{\delta_r}{\sigma_r}\right]_+,
		\quad
		\boldsymbol{Z}\sim N_m(\mathbf0,R).
		\label{eq:local-power}
	\end{equation}
	If the distribution of $T(\boldsymbol\delta)$ is continuous at
	$c_{1-\alpha}(R)$, then
	$\mathbb P_{\mu_n}(\varphi_n=1)\to
	\mathbb P\{T(\boldsymbol\delta)>c_{1-\alpha}(R)\}$.

Let $\widehat D_n:=
	\operatorname{diag}(\widehat\sigma_{n,1},\ldots,
	\widehat\sigma_{n,m})$. Since
	$\widehat\Omega_n\stackrel{p}{\to}\Omega$ and
	$\min_{r\in[m]}\sigma_r^2>0$,
	$\widehat D_n\stackrel{p}{\to}D$. Furthermore,
	\[
	\widehat D_n^{-1}\sqrt n\,\widehat{\boldsymbol\Delta}_n
	=
	\widehat D_n^{-1}
	\sqrt n(\widehat{\boldsymbol\Delta}_n-\boldsymbol\Delta_n)
	+
	\widehat D_n^{-1}\sqrt n\,\boldsymbol\Delta_n
	\Rightarrow
	\boldsymbol{Z}+D^{-1}\boldsymbol\delta,
	\]
	by Slutsky's theorem. Since
	$\boldsymbol{z}\mapsto\max_{r\in[m]}[-z_r]_+$ is continuous,
	the continuous mapping theorem yields
	\eqref{eq:local-power}. Moreover,
	$\widehat R_n\stackrel{p}{\to}R$ implies
	$c_{1-\alpha}(\widehat R_n)\stackrel{p}{\to}
	c_{1-\alpha}(R)$. Continuity of the distribution of
	$T(\boldsymbol\delta)$ at $c_{1-\alpha}(R)$ therefore yields
	$\mathbb P_{\mu_n}(\varphi_n=1)\to
	\mathbb P\{T(\boldsymbol\delta)>c_{1-\alpha}(R)\}$.

\label{ex:simultaneous-lattice-test}
	Let $n=500$, $m=2$, and suppose the empirical probabilities associated
	with two comparisons in $\mathcal Q$ yield
	$\widehat{\boldsymbol\Delta}_n=(-0.020,0.008)^\top$ and
	$\widehat\Omega_n=\operatorname{diag}(0.0400,0.0256)$.
	Thus
	$\widehat\sigma_{n,1}=0.20$,
	$\widehat\sigma_{n,2}=0.16$, and $\widehat R_n=I_2$. From
	\eqref{eq:max-lattice-test},
	\[
	T_n
	=
	\max\left\{
	\frac{\sqrt{500}(0.020)}{0.20},
	0
	\right\}
	=2.236.
	\]
	For $\alpha=0.05$ and $\boldsymbol{Z}\sim N_2(\mathbf0,I_2)$,
	$c_{0.95}(I_2)=\Phi^{-1}(\sqrt{0.95})\approx1.955$.
	Hence $T_n>c_{0.95}(I_2)$ and $H_0$ is rejected. The first comparison
	therefore provides evidence of a lattice violation within
	$\mathcal Q$. This conclusion concerns the selected finite system and
	does not, by itself, constitute an omnibus test of
	$\mathrm{cmMTP}_2(\boldsymbol{X}_B\mid\boldsymbol{X}_A)$.
