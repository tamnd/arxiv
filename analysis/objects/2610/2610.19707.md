<!-- theorem-1: body_md -->
\label{thm:main}
There exists a universal constant $c>0$ such that, for every fixed
integer $n\geq0$,
\[
   \liminf_{g\to\infty}~
   \frac{\operatorname{diam}_{\mathrm{WP}}(\mathcal M_{g,n})}{\sqrt g\log g}\geq c.
\]
<!-- end theorem-1 -->
<!-- remark-1: body_md -->
With normalization \eqref{eq:wp-normalization}, one may take the constant $c$ above to be
\[
   c=\frac1{454\sqrt6}.
\] 
This Theorem \ref{thm:main} together with the work \cite{CP} of Cavendish--Parlier implies that for every fixed $n\geq0$,
\begin{equation}
\operatorname{diam}_{\mathrm{WP}}(\mathcal \mathcal M_{g,n})
   \asymp
   \sqrt g\log g. \nonumber
\end{equation}

<!-- theorem-2: body_md -->
\label{thm:integrated-intro}
For every $\eta>0$ there is a constant $A_\eta>0$, independent of $g$,
with the following property. Suppose that
$\gamma:[0,L]\to\mathcal T_g$ is a unit-speed Weil--Petersson geodesic,
its projection to $\mathcal M_g$ is globally length minimizing, and its
endpoints lie over
$\mathcal M_g^{\geq\eta}$. Then
\begin{equation}\label{eq:intro-ricci}
 \int_0^L-\operatorname{Ric}_{\mathrm{WP}}(\dot\gamma(t),\dot\gamma(t))\,dt
 \leq A_\eta L.
\end{equation}
Here $\operatorname{Ric}_{\mathrm{WP}}$ is the real Riemannian Ricci tensor.
One may take $A_\eta=C(1+\eta^{-2})$ with $C>1$ universal.
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm:ball-intro}
Let $A_\eta\geq1$ be as in Theorem~\ref{thm:integrated-intro}. Then there is a universal $C>0$ such
that, for all
$X\in\mathcal M_g^{\geq\eta}$ and $R>0$,
\begin{equation}\label{eq:intro-ball}
 \operatorname{Vol}_{\mathrm{WP}}\bigl(B_{\mathcal M_g}(X,R)\cap\mathcal M_g^{\geq\eta}\bigr)
 \leq \exp\bigl(Cg+\sqrt{6A_\eta g}\,R\bigr).
\end{equation}

<!-- lemma-1: body_md -->
\label{lem:packing}
For every $z\in\mathbb H$,
\begin{equation}\label{eq:packing}
 \sum_{\alpha\in\sigma}\ \sum_{\substack{L\in\mathscr L_\alpha\\z\notin\operatorname{Int} C_L}}
 \frac{\operatorname{sech}^2 d(z,L)}{\ell_\alpha}\leq76.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Fix $z\in\mathbb H$. For each axis in the sum, set
$u_L=d(z,L)-W_\alpha\geq0$. Along the perpendicular path from $z$ to $L$,
choose $p_L$ at distance $1/2$ inside the nearer boundary of $C_L$.
Then $d(z,p_L)=u_L+1/2$, and the balls $B(p_L,1/4)$ lie in the
disjoint strip interiors.

Let $\mathscr A_j$ consist of the axes with $j\leq u_L<j+1$.
Their balls lie in $B(z,j+2)$, so comparison of hyperbolic areas gives
\[
 \#\mathscr A_j\leq
 \frac{\cosh(j+2)-1}{\cosh(1/4)-1}.
\]
Since $e^{-W_\alpha}=\tan\ell_\alpha\leq(101/100)\ell_\alpha$,
\[
 \ell_\alpha^{-1}\operatorname{sech}^2d(z,L)
 \leq4\ell_\alpha^{-1}e^{-2W_\alpha-2u_L}
 \leq4(101/100)^2\ell_0e^{-2u_L}.
\]
Summing over the shells and using $\cosh(j+2)-1\leq e^{j+2}/2$
bounds the LHS of Equation \eqref{eq:packing} by
\[
 \frac{4(101/100)^2\ell_0e^2}
 {2(\cosh(1/4)-1)}
 \sum_{j=0}^\infty e^{-j}
 =
 \frac{4(101/100)^2\ell_0e^2}
 {2(\cosh(1/4)-1)(1-e^{-1})}<76
\]
as desired.

<!-- proposition-1: body_md -->
\label{prop:fourier}
For every harmonic Beltrami differential $\mu\in T_X\mathcal T_g$,
\begin{equation}\label{eq:fourier-sum}
 \sum_{\alpha\in\sigma}\ell_\alpha|c_{\alpha,0}|^2
 +\sum_{\alpha\in\sigma}\frac{Q_\alpha^\perp(\mu)}{\ell_\alpha}
 \leq128\left\|\mu\right\|_{\mathrm{WP}}^2.
\end{equation}
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
We first estimate the principal coefficient. In the cylinder model,
\[
 |\mu|^2\,dA=|f_\alpha(x+iy)|^2\sin^2y\,dx\,dy.
\]
Parseval's identity on each horizontal circle gives
\[
 \int_0^{\ell_\alpha}|f_\alpha(x+iy)|^2\,dx
 =\ell_\alpha\sum_{k\in\mathbb Z}|c_{\alpha,k}|^2
       e^{-4\pi ky/\ell_\alpha}
 \geq\ell_\alpha|c_{\alpha,0}|^2.
\]
The interval $\pi/4\leq y\leq3\pi/4$ lies in every standard collar
under consideration, and
$\int_{\pi/4}^{3\pi/4}\sin^2y\,dy\geq\pi/4$. Integrating the last
inequality on this interval yields
\begin{equation}\label{eq:principal-collar}
     \ell_\alpha|c_{\alpha,0}|^2\leq\frac4\pi N_\alpha^2.
\end{equation}

\noindent Since the standard collars are disjoint, we have 
\begin{equation}\label{eq:principal-sum}
 \sum_{\alpha\in\sigma}\ell_\alpha|c_{\alpha,0}|^2
 \leq \frac4\pi\sum_{\alpha\in\sigma}N_\alpha^2
 \leq \frac4\pi\left\|\mu\right\|_{\mathrm{WP}}^2.
\end{equation}


Next split the cylinder integral defining $Q_\alpha^\perp$ into the
inner collar $C_\alpha$ and its complement. On $C_\alpha$,
\eqref{eq:BWlocal} says that the integrand
$|f_\alpha^\perp|^2\sin^4y$ is bounded pointwise by
$16N_\alpha^2$. The Euclidean area of the parameter rectangle for
$C_\alpha$ is
$\ell_\alpha(\pi-4\ell_\alpha)<\pi\ell_\alpha$, and hence
\[
 Q_{\alpha,\mathrm{in}}^\perp\leq16\pi\ell_\alpha N_\alpha^2,
\]
which implies that
\begin{equation}\label{eq:nonprincipal-inner}
 \sum_{\alpha\in\sigma}
 \frac{Q_{\alpha,\mathrm{in}}^\perp}{\ell_\alpha}
 \leq16\pi\left\|\mu\right\|_{\mathrm{WP}}^2.
\end{equation}

The exterior part requires a different argument, because a pointwise
estimate on one collar would not remain uniform after summing over
$\sigma$. For every $y$, orthogonality of the Fourier modes gives
\[
 \int_0^{\ell_\alpha}|f_\alpha^\perp(x+iy)|^2\,dx
 \leq\int_0^{\ell_\alpha}|f_\alpha(x+iy)|^2\,dx.
\]
Thus we may replace the nonprincipal part by the full differential
before unfolding. To see the unfolded integrand explicitly, note that
on the cyclic cover
\[
 |\mu|^2\operatorname{sech}^2d(\,\cdot\,,L)\,\,dA
 =|f_\alpha|^2\sin^4y\,dx\,dy
\]
for the distinguished lift of the core, by \eqref{eq:sech}. Summing
over the deck translates of a fundamental domain and then over
$\alpha$, with Tonelli's theorem justifying the nonnegative sums, gives
\begin{align}\label{eq:nonprincipal-outer}
 \sum_{\alpha\in\sigma}\frac{Q_{\alpha,\mathrm{out}}^\perp}{\ell_\alpha}
 &\leq\int_X|\mu(z)|^2
 \sum_{\alpha\in\sigma}\ \sum_{\substack{L\in\mathscr L_\alpha\\
 \widetilde z\notin\operatorname{Int} C_L}}
 \frac{\operatorname{sech}^2d(\widetilde z,L)}{\ell_\alpha}\,dA(z)
 \leq76\left\|\mu\right\|_{\mathrm{WP}}^2,
\end{align}
where $\widetilde z$ is any lift of $z$; the last inequality is
Lemma~\ref{lem:packing}.

Adding \eqref{eq:principal-sum}, \eqref{eq:nonprincipal-inner}, and
\eqref{eq:nonprincipal-outer} proves the claim.

<!-- proposition-2: body_md -->
\label{prop:Hformula}
Let $\ell_\alpha\leq\ell_0$ and let $v\in T_X\mathcal T_g$ have the
Fourier expansion \eqref{eq:fourier}. Then
\begin{equation}\label{eq:firstvariation}
 a_\alpha(v)^2+b_\alpha(v)^2
 =\frac{\ell_\alpha}{4}|c_{\alpha,0}|^2,
\end{equation}
and
\begin{equation}\label{eq:Hbound}
 0\leq\frac{H_\alpha(v,v)}{r_\alpha}
 \leq\frac2\pi\frac{Q_\alpha^\perp(v)}{\ell_\alpha}.
\end{equation}

<!-- lemma-2: body_md -->
\label{lem:multiplier}
If $f^\perp=\sum_{k\ne0}c_ke^{2\pi ikw/\ell}$ and
$\int_{R_\ell}|f^\perp|^2\sin^4y<\infty$, then
\begin{equation}\label{eq:multiplier}
 \int_{R_\ell}|Af^\perp|^2\sin^2y
 \leq\frac43\int_{R_\ell}|f^\perp|^2\sin^4y.
\end{equation}
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
For $\xi\ne0$, put $I_j(\xi)=\int_0^\pi e^{-2\xi y}\sin^j y\,dy$.
Expanding $\sin^2y$ and $\sin^4y$ into cosines gives
\[
 I_2(\xi)=\frac{1-e^{-2\pi\xi}}{4\xi(\xi^2+1)},
 \qquad
 I_4(\xi)=\frac{3(1-e^{-2\pi\xi})}
 {4\xi(\xi^2+1)(\xi^2+4)}.
\]
Thus, for the mode $f_k(w)=c_ke^{2\pi ikw/\ell}$ with
$\xi=2\pi k/\ell$,
\begin{equation}\label{eq:single-mode}
 \int_{R_\ell}|Af_k|^2\sin^2y
 =\frac{\xi^2+4}{3(\xi^2+1)}
   \int_{R_\ell}|f_k|^2\sin^4y
 \leq\frac43\int_{R_\ell}|f_k|^2\sin^4y.
\end{equation}
Orthogonality in $x$ proves \eqref{eq:multiplier} for finite sums. It then follows from
Parseval's identity that the Fourier
partial sums converge in $L^2(\sin^4y\,dx\,dy)$. Applying
\eqref{eq:single-mode} to their differences gives convergence of their
images in $L^2(\sin^2y\,dx\,dy)$, with limit $Af^\perp$.
Passing to the limit proves the claim.

<!-- lemma-3: body_md -->
\label{lem:Hvariation}
Under the hypotheses of Proposition~\ref{prop:Hformula},
\begin{equation}\label{eq:principal-variation}
 a_\alpha(v)=\frac{r_\alpha}{2}\operatorname{Re} c_{\alpha,0},
 \qquad
 b_\alpha(v)=-\frac{r_\alpha}{2}\operatorname{Im} c_{\alpha,0},
\end{equation}
and
\begin{equation}\label{eq:Hexact}
 H_\alpha(v,v)=\frac1{2\pi r_\alpha}
 \left(2\int_{R_{\ell_\alpha}}|Af_\alpha^\perp|^2\sin^2y
       -\operatorname{Re}\int_{R_{\ell_\alpha}}(Af_\alpha^\perp)^2\sin^2y\right).
\end{equation}
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
Suppress the index $\alpha$. Our holomorphic coefficient is $-4$
times Wolpert's coefficient. Consequently,
\cite[Theorem~3.3]{W08} reads
\[
 d\ell(v)=\ell\operatorname{Re} c_0,\qquad
 \operatorname{Hess}\ell(v,v)=\frac1\pi
 \left(2\int_{R_\ell}|Af|^2\sin^2y
       -\operatorname{Re}\int_{R_\ell}(Af)^2\sin^2y\right).
\]
The second variation is the Riemannian Hessian because harmonic
Beltrami coordinates are WP normal \cite[Section~2.3]{W08}.
Since $J\mu=i\mu$, the holomorphic coefficient representing $Jv$
is $-if$. The identities $dr=d\ell/(2r)$ and $dr(Jv)=-b(v)$
therefore give \eqref{eq:principal-variation}.

Write $f=c_0+f^\perp$. Since $A(c_0)=-c_0$, orthogonality in $x$
eliminates the mixed terms. The constant-mode contribution to
$\operatorname{Hess}\ell(v,v)$ is
\[
 \frac{\ell}{2}(\operatorname{Re} c_0)^2
 +\frac{3\ell}{2}(\operatorname{Im} c_0)^2
\]
and cancels in
\eqref{eq:H-Wolpert}, leaving \eqref{eq:Hexact}.
All integrals are finite by Proposition~\ref{prop:fourier} and
Lemma~\ref{lem:multiplier}. The same weighted $L^2$ convergence
justifies the bilinear integral, since
$\|u^2-v^2\|_{L^1}\leq\|u-v\|_{L^2}\|u+v\|_{L^2}$.
<!-- end proof-4 -->
<!-- proof-5: body_md -->
Equation~\eqref{eq:firstvariation} follows from
\eqref{eq:principal-variation}. Set
\[
 U=\int_{R_{\ell_\alpha}}|Af_\alpha^\perp|^2\sin^2y.
\]
The absolute value of the second integral in \eqref{eq:Hexact}
is at most $U$. Hence
\[
 0\leq H_\alpha(v,v)
 \leq\frac{3U}{2\pi r_\alpha}
 \leq\frac{2}{\pi r_\alpha}Q_\alpha^\perp(v),
\]
where the last inequality is Lemma~\ref{lem:multiplier}.
Dividing by $r_\alpha$ proves \eqref{eq:Hbound}.

<!-- proposition-3: body_md -->
\label{prop:summed-H}
For every $v\in T_X\mathcal T_g$,
\begin{align}
 \sum_{\alpha\in\sigma}
 \bigl(a_\alpha(v)^2+b_\alpha(v)^2\bigr)
 &\leq \left\|v\right\|_{\mathrm{WP}}^2,\label{eq:Bessel}\\
 \sum_{\alpha\in\sigma}\frac{H_\alpha(v,v)}{r_\alpha}
 &\leq81\left\|v\right\|_{\mathrm{WP}}^2.\label{eq:Hsum}
\end{align}
For any real $(e_\alpha)_{\alpha\in\sigma}$,
\begin{equation}\label{eq:dual-Bessel}
 \left\|\sum_{\alpha\in\sigma}e_\alpha J\lambda_\alpha\right\|_{\mathrm{WP}}^2
 \leq\sum_{\alpha\in\sigma}e_\alpha^2.
\end{equation}
<!-- end proposition-3 -->
<!-- proof-6: body_md -->
Equations~\eqref{eq:principal-sum} and \eqref{eq:firstvariation} give
\[
 \sum_{\alpha\in\sigma}(a_\alpha^2+b_\alpha^2)
 \leq\frac1\pi\left\|v\right\|_{\mathrm{WP}}^2\leq\left\|v\right\|_{\mathrm{WP}}^2.
\]
Similarly, \eqref{eq:nonprincipal-inner},
\eqref{eq:nonprincipal-outer}, and \eqref{eq:Hbound} give
\[
 \sum_{\alpha\in\sigma}\frac{H_\alpha(v,v)}{r_\alpha}
 \leq\left(32+\frac{152}{\pi}\right)\left\|v\right\|_{\mathrm{WP}}^2
 \leq81\left\|v\right\|_{\mathrm{WP}}^2.
\]
For \eqref{eq:dual-Bessel}, set
$w=\sum_{\alpha\in\sigma}e_\alpha J\lambda_\alpha$.
Cauchy--Schwarz and \eqref{eq:Bessel} give, for every $v$,
\[
 |\left\langle w,v\right\rangle|
 =\left|\sum_{\alpha\in\sigma}e_\alpha b_\alpha(v)\right|
 \leq\left(\sum_{\alpha\in\sigma}e_\alpha^2\right)^{1/2}
      \left\|v\right\|_{\mathrm{WP}}.
\]
Taking the supremum over unit vectors $v$ proves the claim.

<!-- lemma-4: body_md -->
\label{lem:Bergman}
For $0<\varepsilon\leq\ell_0$ and all $z\in X$,
\begin{equation}\label{eq:Bergman-bound}
 \mathcal B_X(z)\leq6\varepsilon^{-2}+
 3\sum_{\alpha\in\sigma_\varepsilon(X)}\ell_\alpha^{-1}K_{\alpha,4}(z).
\end{equation}
<!-- end lemma-4 -->
<!-- proof-7: body_md -->
Suppose that $z\in C_\alpha$ for some
$\alpha\in\sigma_\varepsilon(X)$. Write the lift of a unit harmonic
differential as the sum of its principal and nonprincipal terms:
\[
 \mu=\overline{c_{\alpha,0}}\sin^2y
     +\overline{f_\alpha^\perp(w)}\sin^2y.
\]
Since $N_\alpha\leq\left\|\mu\right\|_{\mathrm{WP}}=1$,
\eqref{eq:principal-collar} gives
$|c_{\alpha,0}|\leq2/(\sqrt{\pi\ell_\alpha})$, while
\eqref{eq:BWlocal} bounds the second term by $4$. Hence
\[
 |\mu(z)|\leq4+\frac2{\sqrt{\pi\ell_\alpha}}\sin^2y,
 \qquad
 |\mu(z)|^2\leq32+\frac3{\ell_\alpha}\sin^4y.
\]
For the second inequality, we used $(A+B)^2\leq2A^2+2B^2$ and
$8/\pi<3$. The chosen lift of the core contributes
$\sin^4y$ to $K_{\alpha,4}(z)$ by \eqref{eq:sech}. Since
$32\leq6\varepsilon^{-2}$, taking the supremum in
\eqref{eq:Bergman-eval} proves \eqref{eq:Bergman-bound} at every point
of the chosen inner collars.

If $z$ lies outside these collars, we claim that
\begin{equation}\label{eq:inj-outside-inner}
 \operatorname{inj}_X(z)\geq\varepsilon/2.
\end{equation}
Suppose otherwise. Since $\varepsilon/2\leq1/20<\varepsilon_2$, the thin-part
decomposition places $z$ in the standard collar of a primitive closed
geodesic $\alpha$ of length
$\ell_\alpha \leq 2\operatorname{inj}_X(z)<\varepsilon$. In this collar the injectivity radius
agrees with the cyclic injectivity radius. If $z$ were outside
$C_\alpha$, formula \eqref{eq:inj-cylinder}, together with
$\sin(2\ell_\alpha)\leq2\ell_\alpha$ and
$\sinh(\ell_\alpha/2)\geq\ell_\alpha/2$, would give
\[
 \sinh\operatorname{inj}_X(z)\geq\frac{\sinh(\ell_\alpha/2)}
 {\sin(2\ell_\alpha)}\geq\frac14,
\]
whereas $\operatorname{inj}_X(z)<1/20$ implies
$\sinh\operatorname{inj}_X(z)<\sinh(1/20)<1/4$. This contradiction proves
\eqref{eq:inj-outside-inner}. Apply Teo's local estimate with radius $\varepsilon/4$ to obtain
\[
 |\mu(z)|^2
 \leq C(\varepsilon/4)^2
       \int_{B(z,\varepsilon/4)}|\mu|^2\,dA.
\]
For completeness, the numerical bound follows from
$R=\tanh(\varepsilon/8)$ and \eqref{eq:Teo-constant}:
\[
 C(\varepsilon/4)^2
 =\frac{3}{4\pi[1-(1-R^2)^3]}
 \leq\frac1{4\pi R^2(1-R^2)^2}
 <6\varepsilon^{-2}.
\]
Indeed, $\tanh x\geq x/(1+x)$ and $\tanh x\leq x$ give
$R\geq10\varepsilon/81$ and $1-R^2>99/100$ for $\varepsilon\leq1/10$.
Taking the supremum over unit $\mu$ completes the proof.

<!-- lemma-5: body_md -->
\label{lem:resolvent}
For every curve $\alpha$,
\begin{equation}\label{eq:resolvent}
 D(K_{\alpha,4})=\frac12K_{\alpha,2}.
\end{equation}
Moreover, if $\mu$ has expansion \eqref{eq:fourier}, then
\begin{equation}\label{eq:unfold-Q}
 \int_X|\mu|^2K_{\alpha,2}\,dA
 =Q_\alpha^\perp(\mu)+\frac{3\pi}{8}\ell_\alpha|c_{\alpha,0}|^2.
\end{equation}
<!-- end lemma-5 -->
<!-- proof-8: body_md -->
In Fermi coordinates about a geodesic in $\mathbb H$, the Laplacian on a
function of signed distance $s$ is $h''(s)+\tanh(s)h'(s)$. Direct
differentiation gives
\[
 (2-\Delta)\operatorname{sech}^2s=4\operatorname{sech}^4s.
\]
Apply this identity to the distance from each axis
$L\in\mathscr L_\alpha$. The local $C^2$ convergence established after
\eqref{eq:kernel} permits termwise differentiation and gives
\[
 (2-\Delta)K_{\alpha,2}=4K_{\alpha,4}.
\]
The operator $2-\Delta$ is positive and invertible on the closed
surface $X$. Applying $2(2-\Delta)^{-1}$ to the last identity proves
\eqref{eq:resolvent}.

For the second assertion, choose a fundamental domain in $\mathbb H$ for the
surface group and unfold the nonnegative integral to the cyclic cover with respect to $\alpha$. Equations
\eqref{eq:sech} and \eqref{eq:fourier} give
\[
 \int_X|\mu|^2K_{\alpha,2}\,dA
 =\int_0^{\ell_\alpha}\int_0^\pi|f_\alpha|^2\sin^4y\,dy\,dx.
\]
For each fixed $y$, Parseval's identity separates the zeroth mode from
all other modes. The latter contribute precisely
$Q_\alpha^\perp(\mu)$, while the former contributes
\[
 \ell_\alpha|c_{\alpha,0}|^2
 \int_0^\pi\sin^4y\,dy
 =\frac{3\pi}{8}\ell_\alpha|c_{\alpha,0}|^2.
\]
This is \eqref{eq:unfold-Q}.

<!-- proposition-4: body_md -->
\label{prop:Ricci}
For every $0<\varepsilon\leq\ell_0$ and every $v\in T_X\mathcal T_g$,
\begin{equation}\label{eq:directional-Ricci}
 -\operatorname{Ric}_{\mathrm{WP}}(v,v)\leq 32\varepsilon^{-2}\left\|v\right\|_{\mathrm{WP}}^2+
30\sum_{\alpha\in\sigma_\varepsilon(X)}
 \frac{a_\alpha(v)^2+b_\alpha(v)^2}{r_\alpha^2}.
\end{equation}
<!-- end proposition-4 -->
<!-- proof-9: body_md -->
For $\left\|\mu\right\|_{\mathrm{WP}}=1$, choose the Hermitian orthonormal basis with
$\nu_1=\mu$. Sum the sectional-curvature estimate \cite[(2.1)]{BW} over
$\{\nu_j,J\nu_j\}_{j=1}^{3g-3}\setminus\{\mu\}$, an orthonormal
basis of the real orthogonal complement of $\mu$.
The Ricci tensor is the trace of the curvature operator in this real
basis. Moreover, multiplication by $J$ preserves pointwise norm, so the
sum of the pointwise squared norms of these basis elements is at most
$2\mathcal B_X$. The curvature estimate, the positivity of $D$, and
the self-adjointness of $D$ therefore give
\begin{equation}\label{eq:Ricci-Bergman}
 -\operatorname{Ric}_{\mathrm{WP}}(\mu,\mu)
 \leq4\int_X|\mu|^2D(\mathcal B_X)\,dA.
\end{equation}
Homogeneity extends the estimate to arbitrary $\mu$.

Apply $D$ to \eqref{eq:Bergman-bound}. Positivity of $D$ preserves the
inequality, while $D(1)=1$ and Lemma~\ref{lem:resolvent} yield
\[
 D(\mathcal B_X)\leq6\varepsilon^{-2}
 +\frac32\sum_{\alpha\in\sigma_\varepsilon(X)}
 \ell_\alpha^{-1}K_{\alpha,2}.
\]
Substitution in \eqref{eq:Ricci-Bergman} gives
\[
\begin{aligned}
 &   -\operatorname{Ric}_{\mathrm{WP}}(\mu,\mu)
 \leq24\varepsilon^{-2}\left\|\mu\right\|_{\mathrm{WP}}^2+
 6\sum_{\alpha\in\sigma_\varepsilon(X)}
 \frac1{\ell_\alpha}\int_X|\mu|^2K_{\alpha,2}\,dA \\
 &\leq 24\varepsilon^{-2}\left\|\mu\right\|_{\mathrm{WP}}^2+
 \sum_{\alpha\in\sigma_\varepsilon(X)}
 \left(\frac{9\pi}{4}|c_{\alpha,0}|^2+6\frac{Q_\alpha^\perp(\mu)}{\ell_\alpha} \right).
\end{aligned}
\]
Here we apply \eqref{eq:unfold-Q} in the last inequality. By \eqref{eq:nonprincipal-inner} and
\eqref{eq:nonprincipal-outer}, the nonprincipal part satisfies
\[
 \sum_{\alpha\in\sigma_\varepsilon(X)}
 \frac{Q_\alpha^\perp(\mu)}{\ell_\alpha}
 \leq(16\pi+76)\left\|\mu\right\|_{\mathrm{WP}}^2
 <127\left\|\mu\right\|_{\mathrm{WP}}^2.
\]
By \eqref{eq:firstvariation},
$|c_{\alpha,0}|^2
=4(a_\alpha^2+b_\alpha^2)/r_\alpha^2$. We consequently obtain
\[
 -\operatorname{Ric}_{\mathrm{WP}}(v,v)
 \leq(24\varepsilon^{-2}+762)\left\|v\right\|_{\mathrm{WP}}^2+
 9\pi\sum_{\alpha\in\sigma_\varepsilon(X)}
 \frac{a_\alpha(v)^2+b_\alpha(v)^2}{r_\alpha^2}.
\]
Since $9\pi<30$ and $\varepsilon^{-2}\geq100$, this is bounded by the
right-hand side of \eqref{eq:directional-Ricci}.

<!-- lemma-6: body_md -->
\label{lem:infinitesimal}
Suppose that $r_\alpha\leq u\leq\sqrt{\ell_0}$ for all
$\alpha\in\sigma$. If $|\theta_\alpha|\leq1/2$ and
$W_\theta=\sum\limits_{\alpha\in\sigma}\theta_\alpha V_\alpha$, then
\begin{equation}\label{eq:infinitesimal}

 \leq 81u^4\left\|v\right\|_{\mathrm{WP}}^2.
\end{equation}

\label{lem:Jacobi}
Set $m=d-1$ and $q(s)=-\operatorname{Ric}(\dot\gamma(s),\dot\gamma(s))$.
For every $\kappa>0$,
\begin{equation}\label{eq:Jacobi}

\end{equation}

<!-- proof-18: body_md -->
Let $\mathbb{A}(s)$ be the radial Jacobi tensor in a parallel
orthonormal frame, with $\mathbb{A}(0)=0$ and $\mathbb{A}'(0)=I$.
For $s>0$, set
\[
 \mathcal J(s)=\det \mathbb{A}(s),\qquad
 S(s)= \mathbb{A}'(s)\ \mathbb{A}(s)^{-1},\qquad
 h=\operatorname{tr}S=\mathcal J'/\mathcal J.
\]
The Riccati equation and $\operatorname{tr}(S^2)\geq h^2/m$ imply
that $y=\mathcal J^{1/m}$ satisfies
\begin{equation}\label{eq:y-ineq}
 y''\leq\frac qm y,\qquad y(0)=0,\quad y'(0)=1;
\end{equation}
see \cite[Example~4.3]{Ball16}.
The initial data follow from the Jacobi equation:
$\mathbb{A}(s)=sI+O(s^3)$, so $y(s)=s+O(s^3)$, with the
corresponding differentiated expansion.

Nonpositive sectional curvature makes the radial shape operator
positive semidefinite. To see this directly, let $U$ be a radial Jacobi
field with $U(0)=0$. Integration by parts in the Jacobi equation gives
\[
 \langle U'(s),U(s)\rangle
 =\int_0^s\left(
 |U'|^2-\langle R(U,\dot\gamma)\dot\gamma,U\rangle\right)d\tau.
\]
The integrand is nonnegative when sectional curvature is nonpositive,
so $\langle S (U),U\rangle\geq0$. Hence $S$ is positive semidefinite,
$h\geq0$, and therefore $y'\geq0$.

Now fix $\kappa>0$ and set $F=(y')^2+\kappa^2y^2$. Since
$y,y'\geq0$, equation \eqref{eq:y-ineq} gives
\[
 \begin{split}
 F'\leq2\left(\frac qm+\kappa^2\right)yy'
 &=\left(\kappa+\frac{q}{m\kappa}\right)(2\kappa yy')\\
 &\leq\left(\kappa+\frac{q}{m\kappa}\right)F,
 \end{split}
\]
where the last step is $2\kappa yy'\leq(y')^2+\kappa^2y^2$.
Because $F(0)=1$, Gr\"onwall's inequality yields
\[
 F(t)\leq
 \exp\left(\kappa t+\frac1{m\kappa}\int_0^tq(s)\,ds\right).
\]
Finally $\kappa^2y(t)^2\leq F(t)$. Raising the resulting estimate for
$y(t)$ to the $m$-th power, and using $\mathcal J=y^m$, proves
\eqref{eq:Jacobi}.

\label{prop:thickvolume}
For all sufficiently large $g$,
\begin{equation}\label{eq:thickvolume}

<!-- proof-20: body_md -->
Let $\widehat V_{h,k}(\mathbf L)$ denote the volumes in
\cite{Mir13}, retaining its boundary-labeling and orbifold
conventions, including the one-handle factors in the integration
formula. Its symplectic form satisfies
$\widehat\omega_{\mathrm{WP}}=2\omega_{\mathrm{WP}}.$
Hence, for closed surfaces of sufficiently large genus,
\[
d\widehat V_g
=2^{3g-3}\,d\operatorname{Vol}_{\mathrm{WP}},
\qquad
\operatorname{Vol}_{\mathrm{WP}}(\mathcal M_g)
=2^{-(3g-3)}\widehat V_g.
\]
The normalized probability measures therefore agree.
All cutting formulas and cross-dimensional volume ratios below
are used in the hatted normalization.

Write $\mathbb P_g$
for normalized WP measure and $\mathbb E_g$ for its expectation.
Let $N(X)$ count simple closed geodesics of length less than $1$.
Since $\{\ell_{\mathrm{sys}}<1\}=\{N\geq1\}$, Markov's inequality gives
\[
 \mathbb P_g(\ell_{\mathrm{sys}}<1)\leq\mathbb E_g[N].
\]

The integration formula \cite[Theorem~2.2]{Mir13}, with symmetry
factors bounded by one, gives
\[
 \mathbb P_g(\ell_{\mathrm{sys}}<1)\leq\mathbb E_g[N]
 \leq\frac1{\widehat V_g}\int_0^1 t
 \left(\widehat V_{g-1,2}(t,t)
 +\sum_{i=1}^{\lfloor g/2\rfloor}
 \widehat V_{i,1}(t)\widehat V_{g-i,1}(t)\right)\,dt.
\]
The first term counts nonseparating curves and the sum counts
separating curves. 

The coefficient bound and volume polynomial in
\cite[Lemma~3.2(2) and (3.1)]{Mir13} imply
\[
 \widehat V_{h,k}(\mathbf L)
 \leq\widehat V_{h,k}\prod_{j=1}^k\cosh(L_j/2).
\]
Moreover, \cite[Theorem~1.2 and (3.19)]{Mir13} gives
\[
 \frac{\widehat V_{g-1,2}}{\widehat V_g}=1+O(g^{-1}),
 \qquad
 \frac{\sum_{i=1}^{\lfloor g/2\rfloor}
 \widehat V_{i,1}\widehat V_{g-i,1}}{\widehat V_g}=O(g^{-1}).
\]
For $0\leq t\leq1$, the nonseparating contribution divided by
$\widehat V_g$ is consequently at most
$(1+O(g^{-1}))\cosh^2(t/2)$, uniformly in $t$, and the total
separating contribution is $O(g^{-1})\cosh^2(t/2)$. Hence
\[
 \mathbb P_g(\ell_{\mathrm{sys}}<1)
 \leq\int_0^1t\cosh^2(t/2)\,dt+O(g^{-1})
 =\frac34-\frac1{2e}+O(g^{-1})<\frac23
\]
for large $g$. This proves the first inequality in \eqref{eq:thickvolume}.

\cite[Corollary~3.6]{Mir13} and Stirling's formula give
$\log\widehat V_g=2g\log g+O(g)$. Passing to our volume normalization
subtracts $(3g-3)\log2$, and the factor $1/3$ subtracts only $\log3$.
Both changes are absorbed by the $O(g)$ term, proving the second
inequality.

<!-- lemma-12: body_md -->
\label{lem:puncture-transfer}
For every fixed $n\geq1$,
\[
 \liminf_{g\to\infty}\frac{D_{g,n}}{\sqrt g\log g}
 \geq
 \liminf_{g\to\infty}\frac{D_{g}}{\sqrt g\log g}.
\]
<!-- end lemma-12 -->
<!-- proof-21: body_md -->
This is essentially \cite[Lemma~6.1]{CP}. Apply the lower-bound assertion of \cite[Lemma~6.1]{CP} with $h(g)=\sqrt g\log g$. The scale satisfies that as $g\to \infty$,
\[
 \frac{h(g+1)}{h(g)}\longrightarrow1 \quad \textit{$\mathrm{and}$} \quad 
 \frac{\sqrt{\log g}}{h(g)}\longrightarrow0.
\]
Thus the bounded genus shifts and the pinching error in that
lemma do not change the normalized lower limit. For every constant
$c$ below the closed surface lower limit, the cited result gives
$$\liminf\limits_{g\to\infty}\frac{D_{g,n}}{h(g)}\geq c.$$
Letting $c$ increase to that lower limit proves the claim.

<!-- proof-22: body_md -->
By Lemma~\ref{lem:puncture-transfer}, it suffices to treat closed surfaces.
Choose $X\in\mathcal M_g^{\geq1}$, which is
nonempty for large $g$ by Proposition~\ref{prop:thickvolume}.
For $R>D_g$, the ball $B_{\mathcal M_g}(X,R)$ contains $\mathcal M_g^{\geq1}$.
Theorem~\ref{thm:ball-intro} and Proposition~\ref{prop:thickvolume}
give
\[
 2g\log g-Cg
 \leq\log\operatorname{Vol}_{\mathrm{WP}}(\mathcal M_g^{\geq1})
 \leq Cg+\sqrt{6A_1g}\,R.
\]
Absorbing the linear error terms gives
\[
 R\geq\frac{2g\log g-Cg}{\sqrt{6A_1g}}.
\]
Let $R\downarrow D_g$ and use $A_1=823\,580$ from
\eqref{eq:numerical-integrated}. We obtain
\[
 D_g\geq\frac{2\sqrt g\log g}{\sqrt{6\cdot823\,580}}-C\sqrt g.
\]
Since $823\,580<4\cdot454^2$, this yields
 \[
 \liminf_{g\to\infty}\frac{\operatorname{diam}_{\mathrm{WP}}(\mathcal M_g)}{\sqrt g\log g}
 \geq\frac1{454\sqrt6}
\]
as desired.
