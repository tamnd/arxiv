<!-- thm-1: body_md -->
\label{stmt:main}
  There exists $c > 0$ such that the following holds for every $k \ge 3$ and
  $\delta > 0$.
  Let $C = C(\delta, k) > 0$ and $n \in \mathbb{N}$ be sufficiently large.
  If $p \ge C n^{-c / \log \log k}$, then
  \[
    \mathbb{P}\big(\exists \, S \subset [n], \, |S| \ge \delta n : \mathcal{D}_k(S) \cap [n]_p = \emptyset \big) \to 0
  \]
  as $n \to \infty.$
<!-- end thm-1 -->
<!-- lemma-1: body_md -->
\label{stmt:fingerprints}
  There exists $c > 0$ such that the following holds for every $k \ge 3$ and
  $\delta > 0$.
  Let $C = C(k) > 0$ and $\gamma = \gamma(\delta, k) > 0$, and let $n$ be
  sufficiently large.
  There exists a collection $\mathcal{F} \subset 2^{[n]}$ with the following
  properties:
  \begin{enumerate}[(a)]
    \item $|\mathcal{F}| \le C n \exp(n^{1-c/\log \log k})$.\label{item:sizeOfFamily}
    \item For each $S \subset [n]$ with $|S| \ge \delta n$, there exists
          $F \in \mathcal{F}$ such that
          $F \subset S$.\label{item:fingerprintForEverySet}
    \item Every $F \in \mathcal{F}$ satisfies $|\mathcal{D}_k(F)| \ge \gamma n$.
          \label{item:diffsInF}
  \end{enumerate}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Let $\mathcal{F}$ be given by \Cref{stmt:fingerprints} for $k$, $\delta$ and $n$.
  By \Cref{stmt:fingerprints}\ref{item:fingerprintForEverySet}, observing that
  $\mathcal{D}_k(F) \subset \mathcal{D}_k(S)$ holds trivially when $F \subset S$, we
  have
  \begin{equation}
    \label{eq:unionBound}
    \mathbb{P}(\exists \, S \subset [n], \, |S| \ge \delta n : \mathcal{D}_k(S) \cap [n]_p = \emptyset) \le \sum_{F \in \mathcal{F}} \mathbb{P}(\mathcal{D}_k(F) \cap [n]_p = \emptyset)
  \end{equation}
  by taking a union bound over all $F \in \mathcal{F}$.
  Moreover, it follows from \Cref{stmt:fingerprints}\ref{item:diffsInF} that
  \begin{equation}
    \label{eq:probabilityThatMisses}
    \mathbb{P}\big(\mathcal{D}_k(F) \cap [n]_p = \emptyset\big) \le (1 - p)^{|\mathcal{D}_k(F)|}
                                                     \le \exp(- \gamma p n),
  \end{equation}
  for every $F \in \mathcal{F}$.
  Substituting \eqref{eq:probabilityThatMisses} and
  \Cref{stmt:fingerprints}\ref{item:sizeOfFamily} in \eqref{eq:unionBound}, we
  conclude that
  \begin{equation*}
    \sum_{F \in \mathcal{F}} \mathbb{P}\big(\mathcal{D}_k(F) \cap [n]_p = \emptyset\big)
    \le \exp\!\left( n^{1-c / \log \log k} + 2 \log n - \gamma p n \right)
    \le \exp\!\left( - \gamma n^{1-c / \log \log k} \right)
  \end{equation*}
  by our choice of $p \ge C n^{-c / \log \log k}$, for any
  ${C > 3 \gamma^{-1}}$.
<!-- end proof-1 -->
<!-- thm-2: body_md -->
\label{stmt:varnavides}
  Let $k \in \mathbb{N}$, $\delta > 0$ and let $n \in \mathbb{N}$ be sufficiently large.
  If $S \subset [n]$ satisfies $|S| \ge \delta n$, then it contains at least
  $\varepsilon n^2$ $k$-APs, for $\varepsilon = \varepsilon(\delta, k) > 0$.
<!-- end thm-2 -->
<!-- thm-3: body_md -->
\label{stmt:greenRuzsaArithmeticKakeya}
  There exists $c > 0$ such that the following holds for every $k \ge 3$ for
  some $C = C(k) > 0$.
  For every sufficiently large $n \in \mathbb{N}$, there exists $Y \subset [C n]$ with
  $|Y| \le n^{1-c / \log \log k}$ which contains, for every $d \in [n]$, a
  $k$-AP with common difference $d$.
<!-- end thm-3 -->
<!-- proof-2: body_md -->
Let $Y$ be the set given by \Cref{stmt:greenRuzsaArithmeticKakeya}, and, for
  each $d \in [n]$, let $P_d \subset Y$ be the $k$-AP with common difference
  $d$, and fix a set $S \subset [n]$ with $|S| \ge \delta n$.
  Observing that each ${P \subset S}$, where $P$ is a $k$-AP, can be written as
  $P = t + P_d$ for some pair ${(t, d) \in \{-C n, \ldots, C n\} \times [n]}$,
  it follows from \Cref{stmt:varnavides} that there are $\varepsilon n^2$ pairs
  $(t, d)$ where $t + P_d \subset S$.
  This implies that there is one $t_*$ such that, letting
  \begin{equation}
    \label{eq:formOfTheFingerprint}
    F = (t_* + Y) \cap S,
  \end{equation}
  we have that the number of $t_* + P_d \subset F$ is at least
  $$
    \frac{\varepsilon n^2}{2 C n + 1} \ge \frac{\varepsilon n}{2C + 1}.
  $$
  Consequently, as every $P_d \subset Y$ has a distinct common difference $d$,
  we conclude that
  \begin{equation}
    \label{eq:propertyOfFingerprint}
    |\mathcal{D}_k(F)| \ge \frac{\varepsilon n}{2C + 1} = \gamma n
  \end{equation}
  for $\gamma = \varepsilon / (2 C + 1)$.
  To complete the proof, we let $\mathcal{F}$ be the family of the $F$ satisfying
  \eqref{eq:formOfTheFingerprint} and \eqref{eq:propertyOfFingerprint} for some
  $S \subset [n]$ with $|S| \ge \delta n$, and note that they are fully
  specified by choosing $t_* \in \{-Cn, \ldots, Cn\}$ and one of the
  $2^{|Y|} \le \exp(n^{1-c / \log \log k})$ sets $Y \cap (S - t_*) \subset Y$.
<!-- end proof-2 -->
