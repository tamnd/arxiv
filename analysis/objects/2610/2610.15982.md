<!-- head-1: body_md -->
Throughout, all regularities lie in $(0,S)$, the constants $R\geq1$, $U>1$, $\nu>1$ and $\gamma\in(0,1)$ are fixed, and privacy is bounded, $0<\alpha\leq\bar\alpha$, which covers the levels used in practice\footnote{See \url{https://desfontain.es/blog/real-world-differential-privacy.html} for privacy levels used in real-world deployments of differential privacy.} \citep[e.g.][]{DJW18}. The level $\alpha$ may depend on $n$, as long as $n\alpha^2\to\infty$. Constants may depend on the regularities involved, on $R$, $U$, $S$, $\nu$, $\bar\alpha$, $\gamma$ and on the basis, but not on $n$ or $\alpha$.

<!-- thm-1: body_md -->
\label{thm:main-adaptation}
Let $\mathcal R\subset(0,S)$ be a set of regularities with smallest element $r$. Under the standing assumptions, the following statements hold.
\begin{enumerate}
\item If $\mathcal R\subseteq[r,4r/3]$, there exist confidence sets based on non-interactive $\alpha$-LDP reports that are honest over $\Sigma(r,R,U)$ at level $1-\gamma$, for every $n$, and whose diameter is of order $\ell_n\delta_{n,\mathrm{est}}(t,\alpha)$ over $\Sigma(t,R,U)$, for every $t\in\mathcal R$.
\item If $\mathcal R$ contains some $s>4r/3$ and $\gamma<1/2$, no confidence sets based on non-interactive $\alpha$-LDP reports can be honest over $\Sigma(r,R,U)$ at level $1-\gamma$ and have diameter of order $\ell_n\delta_{n,\mathrm{est}}(s,\alpha)$ over $\Sigma(s,R,U)$.
\end{enumerate}
<!-- end thm-1 -->
<!-- prop-1: body_md -->
\label{thm:testing-lower}
Under the standing assumptions, let $\rho_n=o\{\rho_{n,\mathrm{test}}(r,\alpha)\}$. Then, for every sequence of non-interactive $\alpha$-LDP channels and every sequence of tests $\Psi_n=\Psi_n(Z_1,\ldots,Z_n)$,
$$
\begin{aligned}
\liminf_{n\to\infty}\Big[&\sup_{f\in\Sigma(s,R,U)}P_f^{Q^n}(\Psi_n=1)\\
&+\sup_{f\in H_1(\rho_n)}P_f^{Q^n}(\Psi_n=0)\Big]\geq1.
\end{aligned}
$$

<!-- thm-2: body_md -->
\label{thm:projected-loss}
Under the standing assumptions, $C_n(\gamma)$ is honest over $\Sigma(r,R,U)$ at level $1-\gamma$ for every $n$. For every $t\in[r,S)$, its diameter is of order $\ell_n\{\delta_{n,\mathrm{est}}(t,\alpha)\vee\rho_{n,\mathrm{test}}(r,\alpha)\}$ over $\Sigma(t,R,U)$.

<!-- thm-3: body_md -->
\label{thm:density-separated}
Assume the standing assumptions, let $r<s$, and let
$$\rho_n\geq M\ell_n^2\{\delta_{n,\mathrm{est}}(s,\alpha)\vee\rho_{n,\mathrm{test}}(r,\alpha)\}$$
for a sufficiently large constant $M$. The above confidence set $C_n^{\mathrm{den}}(\gamma)$ is honest over $\Sigma(s,R,U)\cup H_1(\rho_n)$ at level $1-\gamma$ for every $n$. Its diameter is of order $\ell_n\delta_{n,\mathrm{est}}(s,\alpha)$ over $\Sigma(s,R,U)$, and \[|C_n^{\mathrm{den}}(\gamma)|_2=O_{P_f^{Q^n}}\{\ell_n\delta_{n,\mathrm{est}}(r,\alpha)\}\] uniformly over $H_1(\rho_n)$.

<!-- lem-1: body_md -->
\label{lem:wavelet-basic-bounds}
For every $j\geq j_0$, one has $\|\psi_{jk}\|_\infty\leq C_\psi2^{j/2}$ for all $k\in N_j$, and, for every coefficient vector $b_j=(b_{jk})_{k\in N_j}$,
$$\Bigl\|\sum_{k\in N_j}b_{jk}\psi_{jk}\Bigr\|_\infty\leq C_\psi2^{j/2}\max_{k\in N_j}|b_{jk}|\leq C_\psi2^{j/2}\|b_j\|_{\ell^2}.$$
Similarly, $\|\varphi_{j_0k}\|_\infty\leq C_\psi2^{j_0/2}$ for all $k\in N_{-1}$.
<!-- end lem-1 -->
<!-- proof-1: body_md -->
For $x\in[0,1]$, (W3) gives
$$
\begin{aligned}
\Bigl|\sum_{k\in N_j}b_{jk}\psi_{jk}(x)\Bigr|
&\leq\max_{k\in N_j}|b_{jk}|\sum_{k\in N_j}|\psi_{jk}(x)|\\
&\leq C_\psi2^{j/2}\max_{k\in N_j}|b_{jk}|,
\end{aligned}
$$
and $\max_{k\in N_j}|b_{jk}|\leq\|b_j\|_{\ell^2}$. The bounds on $\|\psi_{jk}\|_\infty$ and $\|\varphi_{j_0k}\|_\infty$ follow from (W3), since each term of a sum of non-negative functions is bounded by the sum.
<!-- end proof-1 -->
<!-- lem-2: body_md -->
\label{lem:besov-tail}
Let $0<t<S$ and $\|f\|_{B^t_{2,\infty}}\leq R$. Then $E_j(f)\leq R^22^{-2jt}$ for every $j\geq j_0$ and, for every $J>j_0$,
\[\|(I-K_J)f\|_2^2\leq b_t(J)\coloneqq\frac{R^22^{-2Jt}}{1-2^{-2t}}.\]
<!-- end lem-2 -->
<!-- proof-2: body_md -->
The first bound restates $2^{jt}\|\beta_j(f)\|_{\ell^2}\leq R$. Since $(I-K_J)f$ consists of the levels $j\geq J$, (W1) and Parseval's identity give
$$
\begin{aligned}
\|(I-K_J)f\|_2^2&=\sum_{j=J}^\infty E_j(f)\leq R^2\sum_{j=J}^\infty2^{-2jt}\\
&=\frac{R^22^{-2Jt}}{1-2^{-2t}}.
\end{aligned}
$$
<!-- end proof-2 -->
<!-- lem-3: body_md -->
\label{lem:release-privacy}
For every deterministic $J>j_0$, the release $Z_{i,J}$ is non-interactive $\alpha$-LDP. Moreover,
$$\sup_{J>j_0}\Delta_J<\infty.$$
<!-- end lem-3 -->
<!-- proof-3: body_md -->
Fix $J>j_0$. Conditional on $X_i=x$, independence of the Laplace variables implies that $Z_{i,J}$ has density
$$q_J(z\mid x)=\prod_{a\in\mathcal I_J}\frac{1}{2\sigma_{J,a}}\exp\left\{-\frac{|z_a-\psi_a(x)|}{\sigma_{J,a}}\right\}.$$
Hence, for $x,x'\in[0,1]$,
$$\log\frac{q_J(z\mid x)}{q_J(z\mid x')}=\sum_{a\in\mathcal I_J}\frac{|z_a-\psi_a(x')|-|z_a-\psi_a(x)|}{\sigma_{J,a}}.$$
By the reverse triangle inequality,
$$|z_a-\psi_a(x')|-|z_a-\psi_a(x)|\leq|\psi_a(x)-\psi_a(x')|.$$
Using the definition of $\sigma_{J,a}$ therefore gives
$$\log\frac{q_J(z\mid x)}{q_J(z\mid x')}\leq\frac{\alpha}{\Delta_J}\sum_{a\in\mathcal I_J}\frac{|\psi_a(x)-\psi_a(x')|}{d_{j(a)}}\leq\alpha.$$
Thus $q_J(z\mid x)\leq e^\alpha q_J(z\mid x')$ for every $z,x,x'$. Integrating this inequality over any measurable set of outputs proves the $\alpha$-LDP condition. Since the release of individual $i$ depends only on $X_i$ and independent randomness, the mechanism is non-interactive.

It remains to show that $\Delta_J$ is bounded uniformly in $J$. By property (W3) of Appendix~\ref{app: bes_wav}, for every $x\in[0,1]$,
$$\sum_{k\in N_{-1}}\frac{|\varphi_{j_0k}(x)|}{2^{j_0/2}}\leq C_\psi,$$
and, for every resolution level $j\geq j_0$,
$$\sum_{k\in N_j}\frac{|\psi_{jk}(x)|}{(j\vee1)^\nu2^{j/2}}\leq C_\psi(j\vee1)^{-\nu}.$$
Applying these bounds at both $x$ and $x'$ yields
$$\Delta_J\leq2C_\psi\left\{1+\sum_{j=j_0}^{J-1}(j\vee1)^{-\nu}\right\}.$$
Since $\nu>1$, the series $\sum_{j\geq j_0}(j\vee1)^{-\nu}$ converges. Its partial sums are therefore uniformly bounded in $J$, which proves $\sup_{J>j_0}\Delta_J<\infty$.
<!-- end proof-3 -->
<!-- lem-4: body_md -->
\label{lem:release-moments}
Let $f$ be a density satisfying $\|f\|_\infty\leq U$ for $U>0$. Then, for every $J>j_0$ and $a\in\mathcal I_J$,
\[\mathbb{E}_f^Q Z_{i,J,a}=\langle f,\psi_a\rangle,\qquad \operatorname{Var}_f^Q(Z_{i,J,a})\leq U+2\sigma_{J,a}^2.\]
Moreover, uniformly over $J>j_0$ and $a\in\mathcal I_J$,
$$\sigma_{J,a}^2\leq C\alpha^{-2}d_{j(a)}^2.$$
<!-- end lem-4 -->
<!-- proof-4: body_md -->
Since $W_{i,a}$ is independent of $X_i$, centred and of variance two, $\mathbb{E}_f^QZ_{i,J,a}=\mathbb{E}_f\{\psi_a(X_i)\}=\langle f,\psi_a\rangle$ and
$$
\begin{aligned}
\operatorname{Var}_f^Q(Z_{i,J,a})&=\operatorname{Var}_f\{\psi_a(X_i)\}+2\sigma_{J,a}^2\\
&\leq\int_0^1\psi_a(x)^2f(x)\,dx+2\sigma_{J,a}^2\\
&\leq U+2\sigma_{J,a}^2,
\end{aligned}
$$
where the last step uses $f\leq U$ and $\|\psi_a\|_2=1$, by (W1). Finally, $\sigma_{J,a}^2=\alpha^{-2}\Delta_J^2d_{j(a)}^2\leq C\alpha^{-2}d_{j(a)}^2$ uniformly over $J$ and $a$, since $\sup_{J>j_0}\Delta_J<\infty$ by Lemma~\ref{lem:release-privacy}.

<!-- prop-2: body_md -->
\label{prop:adaptive-private-estimator}
For every $m\geq1$ and $0<t<S$,
$$\sup_{f\in\Sigma(t,R,U)}\mathbb{E}_f^{Q^m}\|\widetilde f_m-f\|_2^2\leq a_m(t).$$
Under bounded privacy $\alpha\leq \bar \alpha$,
$$a_m(t)\lesssim\ell_m^2\delta_{m,\mathrm{est}}(t,\alpha)^2.$$
<!-- end prop-2 -->
<!-- proof-5: body_md -->
We prove the two assertions in turn, starting with the risk bound in three steps. We first show that, on an event $\mathcal E_m$ of probability at least $1-\varepsilon_m$, every released coefficient lies within $\tau_a/2$ of its mean. We then bound the error of each thresholded coefficient on $\mathcal E_m$. Finally, we combine these bounds through Parseval's identity, together with the contribution of the levels that are not released and a crude bound off $\mathcal E_m$.

Fix $a\in\mathcal I_{K_m}$. The coefficient released by individual $i$ decomposes as $Z_{i,K_m,a}-\theta_a=Y_{i,a}+\sigma_aW_{i,a}$, where $Y_{i,a}=\psi_a(X_i)-\theta_a$ is the sampling error and the Laplace variable $W_{i,a}$ is independent of $Y_{i,a}$. The sampling error is centred, satisfies $|Y_{i,a}|\leq2B_a$ because $|\psi_a(X_i)|\leq B_a$ and $|\theta_a|\leq\mathbb{E}_f|\psi_a(X_i)|\leq B_a$, and has variance at most $U$, as in the proof of Lemma~\ref{lem:release-moments}. Bernstein's bound on its moment generating function therefore gives, for $|\lambda|<3/(2B_a)$,
$$\log\mathbb{E}_fe^{\lambda Y_{i,a}}\leq\frac{U\lambda^2}{2(1-2B_a|\lambda|/3)}.$$
For the privacy noise, $\mathbb{E} e^{uW_{i,a}}=(1-u^2)^{-1}$ for $|u|<1$, and since
$$-\log(1-u^2)=\sum_{k=1}^\infty\frac{u^{2k}}{k}\leq\frac{u^2}{1-u^2}\leq\frac{u^2}{1-|u|},$$
we get for $|\lambda|<\sigma_a^{-1}$,
$$\log\mathbb{E} e^{\lambda\sigma_aW_{i,a}}\leq\frac{\sigma_a^2\lambda^2}{1-\sigma_a|\lambda|}=\frac{2\sigma_a^2\lambda^2}{2(1-\sigma_a|\lambda|)}.$$
By independence the two bounds add, and since $2B_a/3\leq c_a$ and $\sigma_a\leq c_a$, for $|\lambda|<c_a^{-1}$,
$$\log\mathbb{E}_f^Qe^{\lambda(Z_{i,K_m,a}-\theta_a)}\leq\frac{(U+2\sigma_a^2)\lambda^2}{2(1-c_a|\lambda|)}.$$
Since $\widehat\theta_a-\theta_a$ is the average of $m$ independent copies of $Z_{i,K_m,a}-\theta_a$, it follows that, with $\kappa_a=(U+2\sigma_a^2)/m$ and for $|\lambda|<m/c_a$,
$$\log\mathbb{E}_f^{Q^m}e^{\lambda(\widehat\theta_a-\theta_a)}\leq\frac{\kappa_a\lambda^2}{2(1-c_a|\lambda|/m)}.$$
Fix $x>0$ and put $\zeta_a(x)=\sqrt{2\kappa_ax}+c_ax/m$. For $\lambda=\sqrt{2\kappa_ax}/\{\kappa_a+(c_a/m)\sqrt{2\kappa_ax}\}$, which lies in $(0,m/c_a)$, a direct computation gives
$$\lambda\zeta_a(x)-\frac{\kappa_a\lambda^2}{2(1-c_a\lambda/m)}=x,$$
so that Markov's inequality applied to $e^{\lambda(\widehat\theta_a-\theta_a)}$ yields $P_f^{Q^m}\{\widehat\theta_a-\theta_a>\zeta_a(x)\}\leq e^{-x}$. The bound on the logarithmic moment generating function depends on $\lambda$ only through $|\lambda|$, so the lower tail is handled in the same way, and
$$P_f^{Q^m}\{|\widehat\theta_a-\theta_a|>\zeta_a(x)\}\leq2e^{-x}.$$
By definition of $\tau_a$, $\zeta_a(x_m)=\tau_a/2$. Let
$$\mathcal E_m=\bigcap_{a\in\mathcal I_{K_m}}\{|\widehat\theta_a-\theta_a|\leq\tau_a/2\}.$$
A union bound over the $|\mathcal I_{K_m}|=2^{K_m}$ released coefficients gives
$$P_f^{Q^m}(\mathcal E_m^c)\leq2^{K_m+1}e^{-x_m}=\varepsilon_m.$$

We now bound the error of each thresholded coefficient on $\mathcal E_m$, where $|\widehat\theta_a-\theta_a|\leq\tau_a/2$ for every released $a$. Let $a$ be a detail coefficient. If it is discarded, then $\widetilde\theta_a=0$ and $|\widehat\theta_a|\leq\tau_a$. The error $(\widetilde\theta_a-\theta_a)^2$ equals $\theta_a^2$, and it is at most $9\tau_a^2/4$ because $|\theta_a|\leq|\widehat\theta_a|+\tau_a/2\leq3\tau_a/2$. If it is retained, then $\widetilde\theta_a=\widehat\theta_a$ and $|\widehat\theta_a|>\tau_a$. The error is at most $\tau_a^2/4$, and it is at most $\theta_a^2$ because $|\theta_a|\geq|\widehat\theta_a|-\tau_a/2>\tau_a/2$. In both cases the error is bounded by $\theta_a^2$ and by $9\tau_a^2/4$, so that, on $\mathcal E_m$,
$$(\widetilde\theta_a-\theta_a)^2\leq\frac94\min\{\theta_a^2,\tau_a^2\}.$$

Summing over a fixed level $j$ gives
\begin{align*}
\sum_{a:j(a)=j}(\widetilde\theta_a-\theta_a)^2&\leq\frac94\sum_{a:j(a)=j}\min\{\theta_a^2,\tau_a^2\}\\
&\leq\frac94\min\left\{\sum_{a:j(a)=j}\theta_a^2,\sum_{a:j(a)=j}\tau_a^2\right\}.
\end{align*}

We finally combine these bounds. By (W1), the error of $\widehat f_m$ splits into the contributions of the different coefficients:
$$
\begin{aligned}
\|\widehat f_m-f\|_2^2
&=\sum_{\substack{a\in\mathcal I_{K_m}\\j(a)=-1}}(\widehat\theta_a-\theta_a)^2\\
&\quad+\sum_{j=j_0}^{K_m-1}\sum_{a:j(a)=j}(\widetilde\theta_a-\theta_a)^2
+\sum_{j=K_m}^\infty E_j(f).
\end{aligned}
$$
The last sum is at most $b_t(K_m)$ by Lemma~\ref{lem:besov-tail}, and the middle one is controlled on $\mathcal E_m$ by the previous display. The scaling coefficients are not thresholded, and by independence across individuals and Lemma~\ref{lem:release-moments}, $\mathbb{E}_f^{Q^m}(\widehat\theta_a-\theta_a)^2=\operatorname{Var}_f^Q(Z_{i,K_m,a})/m\leq(U+2\sigma_a^2)/m$ for each of them. It remains to account for the projection and for $\mathcal E_m^c$. Since $\|f\|_2^2\leq\|f\|_\infty\int_0^1f\leq U$, the density $f$ lies in the closed convex ball onto which $\widehat f_m$ is projected, and projection onto a closed convex set does not increase distances to its points, so that $\|\widetilde f_m-f\|_2\leq\|\widehat f_m-f\|_2$. As $\widetilde f_m$ and $f$ both lie in that ball, also $\|\widetilde f_m-f\|_2^2\leq4U$. Therefore
$$\|\widetilde f_m-f\|_2^2\leq\|\widehat f_m-f\|_2^2\1_{\mathcal E_m}+4U\1_{\mathcal E_m^c},$$
and taking expectations, with the three bounds above and $P_f^{Q^m}(\mathcal E_m^c)\leq\varepsilon_m$, gives $\mathbb{E}_f^{Q^m}\|\widetilde f_m-f\|_2^2\leq A_m(t)$. Together with $\|\widetilde f_m-f\|_2^2\leq4U$, this yields $\mathbb{E}_f^{Q^m}\|\widetilde f_m-f\|_2^2\leq a_m(t)$ for every $f\in\Sigma(t,R,U)$, which proves the risk bound.

It remains to establish the rate under bounded privacy. In this part, constants may also depend on $t$. Put $H_m=\log(e+N_m)$, so that $\ell_m^2=H_m^{2\nu+1}$. Since $a_m(t)\leq4U$ and $\ell_m^2\delta_{m,\mathrm{est}}(t,\alpha)^2\geq N_m^{-t/(t+1)}$, the claim is immediate when $N_m$ is bounded, and we may then assume that $N_m$ exceeds a constant. Then $K_m=\lceil\frac12\log_2(e+N_m)\rceil$, so that
$$K_m\lesssim H_m,\qquad 2^{K_m}\asymp N_m^{1/2},\qquad x_m\lesssim H_m.$$
We bound the four terms of $A_m(t)$ in turn. There are $2^{j_0}$ coefficients at the initial resolution level, and $U+2\sigma_a^2\lesssim\alpha^{-2}$ for each of them by Lemma~\ref{lem:release-moments} and $\alpha\leq\bar\alpha$, so the first term is $O(N_m^{-1})$. Next, $b_t(K_m)\lesssim2^{-2K_mt}\leq N_m^{-t}$ and $4U\varepsilon_m\lesssim N_m^{-2}$. For the other coefficients, let $a$ be an index at level $j<K_m$. Lemmas~\ref{lem:wavelet-basic-bounds} and~\ref{lem:release-moments}, with $\alpha\leq\bar\alpha$, give $B_a^2\lesssim2^j\lesssim\alpha^{-2}2^j$ and $U+2\sigma_a^2\lesssim\alpha^{-2}(j\vee1)^{2\nu}2^j$, hence also $c_a^2\lesssim\alpha^{-2}(j\vee1)^{2\nu}2^j$. Since $(x+y)^2\leq2x^2+2y^2$, $x_m\lesssim H_m$ and $H_m\lesssim m$,
$$\tau_a^2\lesssim\frac{(j\vee1)^{2\nu}2^j}{m\alpha^2}\Bigl(x_m+\frac{x_m^2}{m}\Bigr)\lesssim\frac{(j\vee1)^{2\nu}2^jH_m}{N_m}.$$
As level $j$ contains $2^j$ coefficients and $j\vee1\leq K_m\lesssim H_m$,
$$\sum_{a:j(a)=j}\tau_a^2\lesssim\frac{2^{2j}}{R_m},\qquad R_m=\frac{N_m}{H_m^{2\nu+1}}.$$
Let $k_m$ be an integer with $2^{k_m}\asymp R_m^{1/(2t+2)}$. Since $R_m\to\infty$ and $R_m^{1/(2t+2)}\leq N_m^{1/(2t+2)}$ is of smaller order than $2^{K_m}\asymp N_m^{1/2}$, one has $j_0\leq k_m<K_m$ for $N_m$ large enough. Using the bound in $\tau_a$ at the levels $j\leq k_m$ and the bound $R^22^{-2jt}$ at the levels $j>k_m$, we obtain
$$
\begin{aligned}
&\sum_{j=j_0}^{K_m-1}\min\Bigl\{R^22^{-2jt},\sum_{a:j(a)=j}\tau_a^2\Bigr\}\\
&\quad\lesssim\frac1{R_m}\sum_{j=j_0}^{k_m}2^{2j}+\sum_{j>k_m}2^{-2jt}\\
&\quad\lesssim\frac{2^{2k_m}}{R_m}+2^{-2k_mt}\asymp R_m^{-t/(t+1)},
\end{aligned}
$$
the choice of $k_m$ balancing the two terms. Collecting the four bounds,
$$
\begin{aligned}
a_m(t)&\lesssim R_m^{-t/(t+1)}+N_m^{-1}+N_m^{-t}+N_m^{-2}\\
&\lesssim H_m^{(2\nu+1)t/(t+1)}N_m^{-t/(t+1)},
\end{aligned}
$$
because $t/(t+1)<\min\{1,t\}$ and $H_m\geq1$. Finally, $H_m^{(2\nu+1)t/(t+1)}\leq H_m^{2\nu+1}=\ell_m^2$ and $N_m^{-t/(t+1)}\leq\delta_{m,\mathrm{est}}(t,\alpha)^2$, so that $a_m(t)\lesssim\ell_m^2\delta_{m,\mathrm{est}}(t,\alpha)^2$, which concludes the proof.

<!-- lem-5: body_md -->
\label{lem:release-covariance}
For every $J>j_0$ and every density $f$ satisfying $\|f\|_\infty\leq U$,
$$\operatorname{Cov}_f^Q(Z_{i,J})\preceq\mathsf V_J.$$
<!-- end lem-5 -->
<!-- proof-6: body_md -->
Let $c=(c_a)_{a\in\mathcal I_J}$ and $h=\sum_{a\in\mathcal I_J}c_a\psi_a$, so that $c^\top Z_{i,J}=h(X_i)+\sum_{a\in\mathcal I_J}c_a\sigma_{J,a}W_{i,a}$. The Laplace variables are independent of $X_i$ and of one another, with variance two, so
$$c^\top\operatorname{Cov}_f^Q(Z_{i,J})\,c=\operatorname{Var}_f\{h(X_i)\}+2c^\top\operatorname{diag}_{a\in\mathcal I_J}(\sigma_{J,a}^2)\,c.$$
A variance is at most the second moment about any constant. Taking the constant $\int_0^1h=\langle h,1\rangle=c^\top e_J$, and using $f\leq U$ and (W1), one obtains
$$
\begin{aligned}
\operatorname{Var}_f\{h(X_i)\}
&\leq U\int_0^1\{h(x)-c^\top e_J\}^2\,dx\\
&=U\{\|h\|_2^2-(c^\top e_J)^2\}\\
&=Uc^\top(I-e_Je_J^\top)c.
\end{aligned}
$$
Hence $c^\top\operatorname{Cov}_f^Q(Z_{i,J})\,c\leq c^\top\mathsf V_Jc$ for every $c$, which is the claim.

Write $\mathsf C_J=\operatorname{Cov}_f^Q(Z_{i,J})$ and $\chi_i=Z_{i,J}-\theta_J(f)$ for $i=1,\ldots,m$. The vectors $\chi_1,\ldots,\chi_m$ are independent of the first split and of one another, centred, with covariance $\mathsf C_J$, and $Z_{i,J}-\theta_J(g)=\mu+\chi_i$. Substituting this identity in the definition of $U_m(g,J)$, one obtains
$$U_m(g,J)=q_J(g)^2+\Lambda+\Omega,$$
where
$$\Lambda=\frac2m\sum_{i=1}^m\mu^\top\chi_i,\qquad\Omega=\frac1{m(m-1)}\sum_{i\neq i'}\chi_i^\top\chi_{i'}$$
are the linear and degenerate parts of the statistic. Both are conditionally centred, the second because $\chi_i$ and $\chi_{i'}$ are independent and centred for $i\neq i'$. Hence $\mathbb{E}_2U_m(g,J)=q_J(g)^2$.

We next compute the conditional variance. The two parts are conditionally uncorrelated: $\mathbb{E}_2(\Lambda\Omega)$ is a combination of terms $\mathbb{E}_2\{(\mu^\top\chi_i)(\chi_{i_1}^\top\chi_{i_2})\}$ with $i_1\neq i_2$, and in each of them one of $\chi_{i_1}$ and $\chi_{i_2}$ appears only once, so that independence and centring make the term vanish. By independence,
$$
\begin{aligned}
\operatorname{Var}_2(\Lambda)&=\frac4{m^2}\sum_{i=1}^m\operatorname{Var}_2(\mu^\top\chi_i)\\
&=\frac4m\mu^\top\mathsf C_J\mu.
\end{aligned}
$$
In the expansion
$$
\begin{aligned}
\mathbb{E}_2\Omega^2&=\frac1{m^2(m-1)^2}\\
&\quad\times\sum_{i_1\neq i_2}\sum_{i_3\neq i_4}\mathbb{E}_2\{(\chi_{i_1}^\top\chi_{i_2})(\chi_{i_3}^\top\chi_{i_4})\},
\end{aligned}
$$
a term vanishes as soon as one of the indices appears only once, that is, unless $\{i_3,i_4\}=\{i_1,i_2\}$. In that case, integrating first over $\chi_{i_2}$,
$$\mathbb{E}_2(\chi_{i_1}^\top\chi_{i_2})^2=\mathbb{E}_2(\chi_{i_1}^\top\mathsf C_J\chi_{i_1})=\operatorname{tr}(\mathsf C_J^2).$$
Each of the $m(m-1)$ ordered pairs $(i_1,i_2)$ has two such partners, so that $\mathbb{E}_2\Omega^2=2\operatorname{tr}(\mathsf C_J^2)/\{m(m-1)\}$, and therefore
$$\operatorname{Var}_2\{U_m(g,J)\}=\frac4m\mu^\top\mathsf C_J\mu+\frac{2\operatorname{tr}(\mathsf C_J^2)}{m(m-1)}.$$

By Lemma~\ref{lem:release-covariance}, $0\preceq\mathsf C_J\preceq\mathsf V_J$. Hence $\mu^\top\mathsf C_J\mu\leq\|\mathsf V_J\|_{\mathrm{op}}\|\mu\|_{\ell^2}^2$, and
$$\operatorname{tr}(\mathsf V_J^2)-\operatorname{tr}(\mathsf C_J^2)=\operatorname{tr}\{(\mathsf V_J-\mathsf C_J)(\mathsf V_J+\mathsf C_J)\}\geq0,$$
since the trace of a product of two positive semidefinite matrices is non-negative. By the definitions of $v_J$, $w_J$ and $q_J(g)$, this gives $\operatorname{Var}_2\{U_m(g,J)\}\leq4v_Jq_J(g)^2+w_J^2$.

Finally, since $q_J(g)^2\geq0$, one has $U_m(g,J)^+\leq q_J(g)^2+(\Lambda+\Omega)^+$. As $\Lambda+\Omega$ is conditionally centred, its positive and negative parts have the same conditional expectation, so that, by the Cauchy-Schwarz inequality,
$$\mathbb{E}_2(\Lambda+\Omega)^+=\frac12\mathbb{E}_2|\Lambda+\Omega|\leq\frac12\bigl[\operatorname{Var}_2\{U_m(g,J)\}\bigr]^{1/2}.$$
The variance bound completes the proof.

\label{prop:projected-loss-upper}
For every $\gamma\in(0,1)$, conditionally on the data used to construct $g$ and uniformly over $f\in\Sigma(r,R,U)$,
$$P_2\left[q_J(g)^2\leq\mathcal H_{\gamma,J}\{U_m(g,J)^+\}\right]\geq1-\gamma\quad\text{almost surely}.$$
Moreover, for every $\eta\in(0,1)$ and $u\geq0$,
$$\mathcal H_{\eta,J}(u)\leq2u+F_\eta(J).$$

Fix $\gamma\in(0,1)$ and write
$$q^2=q_J(g)^2,\qquad U=U_m(g,J).$$
By Lemma \ref{lem:projected-loss-moments},
$$\mathbb{E}_2U=q^2,\qquad \operatorname{Var}_2(U)\leq4v_Jq^2+w_J^2.$$
Cantelli's inequality states that, for any centred random variable $Y$ with finite variance and any $a>0$,
$$P(Y\leq-a)\leq\frac{\operatorname{Var}(Y)}{\operatorname{Var}(Y)+a^2}.$$
Apply this with $Y=U-q^2$ and
$$a=\sqrt{c_\gamma(4v_Jq^2+w_J^2)},\qquad c_\gamma=\frac{1-\gamma}{\gamma}.$$
Since $\operatorname{Var}_2(U)\leq4v_Jq^2+w_J^2$,

\begin{aligned}

q^2-U>\sqrt{c_\gamma(4v_Jq^2+w_J^2)}
\right\}\\
&\quad\leq
\frac{4v_Jq^2+w_J^2}
{4v_Jq^2+w_J^2+c_\gamma(4v_Jq^2+w_J^2)}\\
&\quad=
\frac{1}{1+c_\gamma}
=\gamma.
\end{aligned}

Therefore, with conditional probability at least $1-\gamma$,
$$q^2\leq U+\sqrt{c_\gamma(4v_Jq^2+w_J^2)}.$$

$$q^2\leq u+\sqrt{c_\gamma(4v_Jq^2+w_J^2)},\qquad u=U^+.$$

If $q^2\leq u$, then $q^2\leq\mathcal H_{\gamma,J}(u)$ immediately because $\mathcal H_{\gamma,J}(u)\geq u$. Suppose instead that $q^2>u$. Both sides of
$$q^2-u\leq\sqrt{c_\gamma(4v_Jq^2+w_J^2)}$$
are then non-negative, so we may square to obtain
$$(q^2-u)^2\leq4c_\gamma v_Jq^2+c_\gamma w_J^2.$$
Writing $y=q^2$, this is equivalent to
$$y^2-(2u+4c_\gamma v_J)y+u^2-c_\gamma w_J^2\leq0.$$
The larger root of the corresponding quadratic equation is
$$\begin{aligned}y_+&=\frac{2u+4c_\gamma v_J+\sqrt{(2u+4c_\gamma v_J)^2-4(u^2-c_\gamma w_J^2)}}{2}\\&=u+2c_\gamma v_J+\sqrt{4c_\gamma v_Ju+4c_\gamma^2v_J^2+c_\gamma w_J^2}\\&=\mathcal H_{\gamma,J}(u).\end{aligned}$$
Since the quadratic is non-positive only between its two roots, we have $q^2=y\leq y_+$. Hence
$$P_2\left[q_J(g)^2\leq\mathcal H_{\gamma,J}\{U_m(g,J)^+\}\right]\geq1-\gamma.$$

$$\frac{\partial}{\partial u}\mathcal H_{\eta,J}(u)=1+\frac{2c_\eta v_J}{\sqrt{4c_\eta v_Ju+4c_\eta^2v_J^2+c_\eta w_J^2}}.$$
The denominator is at least $2c_\eta v_J$, so
$$\frac{\partial}{\partial u}\mathcal H_{\eta,J}(u)\leq2.$$
Therefore
$$\mathcal H_{\eta,J}(u)-\mathcal H_{\eta,J}(0)=\int_0^u\frac{\partial}{\partial s}\mathcal H_{\eta,J}(s)\,ds\leq2u.$$
Since $F_\eta(J)=\mathcal H_{\eta,J}(0)$, this gives
$$\mathcal H_{\eta,J}(u)\leq2u+F_\eta(J).$$

and $e^\alpha-1\leq\alpha e^{\alpha}\leq\alpha e^{\bar\alpha}$. Finally, $\sum_j\widetilde\lambda_j\leq\sum_j\left(\lambda_j/(e^{2\bar\alpha}\alpha^2)+L^{-1}\right)\leq2$, so that, by the Cauchy-Schwarz inequality,
$$L^2\leq\Bigl(\sum_{j=1}^L\widetilde\lambda_j\Bigr)\Bigl(\sum_{j=1}^L\widetilde\lambda_j^{-1}\Bigr)\leq2\sum_{j=1}^L\widetilde\lambda_j^{-1}.$$

<!-- lem-10: body_md -->
\label{lem:alternative-integral-regularity}
For every $\omega\in\{-1,1\}^L$, the function $f_\omega$ integrates to one, and its wavelet coefficients coincide with those of the uniform density except at level $J$, where
$$\frac{L^{-r}}{\sqrt2}\leq\|\beta_J(f_\omega)\|_{\ell^2}=\|f_\omega-1\|_2\leq L^{-r}.$$
Consequently, $\|f_\omega\|_{B^r_{2,\infty}}\leq R$ and
$$\operatorname{dist}_2(f_\omega,\Sigma(s,R,U))\geq\frac{L^{-r}}{\sqrt2}-RL^{-s}.$$
<!-- end lem-10 -->
<!-- proof-15: body_md -->
The function $f_\omega-1$ lies in $\mathcal V_J$, so it has mean zero and its only non-zero wavelet coefficients are at level $J$. By orthonormality of the $u_j$, $\|f_\omega-1\|_2^2=L^{-2r-2}\sum_j\widetilde\lambda_j^{-1}$, which gives the displayed bounds by Lemma~\ref{lem:spectral-consequences}. The coefficients of $f_\omega$ at level $-1$ are those of the uniform density, whose Euclidean norm is one by (W1) and (W4). Since $2^{Jr}=L^r$, it follows that $\|f_\omega\|_{B^r_{2,\infty}}=\max\{1,L^r\|f_\omega-1\|_2\}\leq1\leq R$. Finally, for $h\in\Sigma(s,R,U)$, one has $\|\beta_J(h)\|_{\ell^2}\leq R2^{-Js}=RL^{-s}$, so that, by (W1) and the reverse triangle inequality,
$$\|f_\omega-h\|_2\geq\|\beta_J(f_\omega)-\beta_J(h)\|_{\ell^2}\geq\frac{L^{-r}}{\sqrt2}-RL^{-s},$$
and the bound on the distance follows by taking the infimum over $h$.

<!-- lem-11: body_md -->
\label{lem:probabilistic-positivity}
For every $p\in(0,1)$, one has $\pi(\mathcal A_{L,p})\geq1-p$, and every $\omega\in\mathcal A_{L,p}$ satisfies
$$\|f_\omega-1\|_\infty\leq C_\psi L^{-r}\{2\log(2L/p)\}^{1/2}.$$
In particular, $f_\omega\in\Sigma(r,R,U)$ for every $\omega\in\mathcal A_{L,p}$ as soon as the right-hand side is at most $1\wedge(U-1)$.
<!-- end lem-11 -->
<!-- proof-16: body_md -->
Since $f_\omega-1$ lies in $\mathcal V_J$, one has $f_\omega-1=\sum_{k\in N_J}\beta_{Jk}(f_\omega)\psi_{Jk}$, where, for every $k\in N_J$,
$$\beta_{Jk}(f_\omega)=L^{-r-1}\sum_{j=1}^L\omega_j\widetilde\lambda_j^{-1/2}\langle u_j,\psi_{Jk}\rangle.$$
Under $\pi$, this is a sum of the independent Rademacher variables $\omega_1,\ldots,\omega_L$ with deterministic weights, whose squares sum to
$$
\begin{aligned}
L^{-2r-2}\sum_{j=1}^L\widetilde\lambda_j^{-1}\langle u_j,\psi_{Jk}\rangle^2
&\leq L^{-2r-1}\sum_{j=1}^L\langle u_j,\psi_{Jk}\rangle^2\\
&=L^{-2r-1},
\end{aligned}
$$
because $\widetilde\lambda_j^{-1}\leq L$ and, by Parseval's identity in $\mathcal V_J$, $\sum_j\langle u_j,\psi_{Jk}\rangle^2=\|\psi_{Jk}\|_2^2=1$. For the level $L^{-r-1/2}\{2\log(2L/p)\}^{1/2}$ that defines $\mathcal A_{L,p}$, whose square is $2L^{-2r-1}\log(2L/p)$, Hoeffding's inequality gives
$$
\begin{aligned}
&\pi\Bigl[|\beta_{Jk}(f_\omega)|>L^{-r-1/2}\{2\log(2L/p)\}^{1/2}\Bigr]\\
&\quad\leq2\exp\Bigl\{-\frac{2L^{-2r-1}\log(2L/p)}{2L^{-2r-1}}\Bigr\}=\frac pL.
\end{aligned}
$$
The complement $\mathcal A_{L,p}^c$ is the union of these events over the $L$ indices $k\in N_J$, so that a union bound gives $\pi(\mathcal A_{L,p}^c)\leq p$.

On $\mathcal A_{L,p}$, Lemma~\ref{lem:wavelet-basic-bounds} and $2^{J/2}=L^{1/2}$ give
$$
\begin{aligned}
\|f_\omega-1\|_\infty
&\leq C_\psi2^{J/2}\max_{k\in N_J}|\beta_{Jk}(f_\omega)|\\
&\leq C_\psi L^{1/2}\cdot L^{-r-1/2}\{2\log(2L/p)\}^{1/2}\\
&=C_\psi L^{-r}\{2\log(2L/p)\}^{1/2}.
\end{aligned}
$$
If this bound is at most $1\wedge(U-1)$, then $0\leq f_\omega\leq U$, and Lemma~\ref{lem:alternative-integral-regularity} completes the proof.

<!-- lem-12: body_md -->
\label{lem:conditioned-prior-second-moment}
Let $\mathcal A\subseteq\{-1,1\}^L$ satisfy $\pi(\mathcal A)\geq1-p$, and suppose that $f_\omega$ is a density for every $\omega\in\mathcal A$. Let $\Pi$ be the law of $f_\omega$ when $\omega$ is drawn from $\pi(\cdot\mid\mathcal A)$, and let $\mathcal L_n=dP_\Pi^{Q^n}/dP_1^{Q^n}$. Then
$$\mathbb{E}_1^{Q^n}\mathcal L_n^2\leq(1-p)^{-2}\exp\Bigl(\frac{n^2e^{4\bar\alpha}\alpha^4}{2L^{4r+3}}\Bigr).$$
<!-- end lem-12 -->
<!-- proof-17: body_md -->
For $\omega\in\{-1,1\}^L$ and each $i$, put
$$\mathcal L_{\omega,i}=1+L^{-r-1}\sum_{j=1}^L\omega_j\widetilde\lambda_j^{-1/2}\mathcal T_iu_j,$$
which, for $\omega\in\mathcal A$, is the likelihood ratio of the $i$th private release under $f_\omega$ with respect to its law $\mu_i$ under the uniform density. Under $\mu_i$, each $\mathcal T_iu_j$ is centred: by Fubini's theorem, which applies because $|q_i(z\mid x)u_j(x)|\leq e^\alpha|u_j(x)|$,
$$
\begin{aligned}
\int\mathcal T_iu_j\,d\mu_i
&=\int_0^1u_j(x)\Bigl\{\int q_i(z\mid x)\,d\mu_i(z)\Bigr\}dx\\
&=\int_0^1u_j(x)\,dx=0,
\end{aligned}
$$
since $u_j$ has mean zero by (W4). Consequently,
$$
\begin{aligned}
\int\mathcal L_{\omega,i}\mathcal L_{\omega',i}\,d\mu_i
&=1+L^{-2r-2}\sum_{j,l=1}^L\omega_j\omega'_l\,\widetilde\lambda_j^{-1/2}\widetilde\lambda_l^{-1/2}\\
&\qquad\times\int\mathcal T_iu_j\,\mathcal T_iu_l\,d\mu_i,
\end{aligned}
$$
and, by definition of the $u_j$, $n^{-1}\sum_i\int\mathcal T_iu_j\,\mathcal T_iu_l\,d\mu_i=\lambda_j\1\{j=l\}$. Hence, by the inequality $1+y\leq e^y$, for all $\omega,\omega'\in\{-1,1\}^L$,
$$\prod_{i=1}^n\int\mathcal L_{\omega,i}\mathcal L_{\omega',i}\,d\mu_i\leq\exp\Bigl(\frac{n}{L^{2r+2}}\sum_{j=1}^L\frac{\lambda_j}{\widetilde\lambda_j}\omega_j\omega'_j\Bigr).$$
The private releases are independent, and
$$\mathcal L_n=\frac{1}{\pi(\mathcal A)}\mathbb{E}_\pi\Bigl\{\1_{\mathcal A}(\omega)\prod_{i=1}^n\mathcal L_{\omega,i}(Z_i)\Bigr\},$$
where $\mathbb{E}_\pi$ integrates over $\omega$ only. Since $\mathcal L_{\omega,i}\geq0$ for $\omega\in\mathcal A$, Fubini's theorem and the last display give
$$\mathbb{E}_1^{Q^n}\mathcal L_n^2\leq\frac{1}{\pi(\mathcal A)^2}\mathbb{E}_{\pi\otimes\pi}\exp\Bigl(\frac{n}{L^{2r+2}}\sum_{j=1}^L\frac{\lambda_j}{\widetilde\lambda_j}\omega_j\omega'_j\Bigr),$$
where the indicators of $\mathcal A$ have been dropped from a positive integrand. Put $y_j=nL^{-2r-2}\lambda_j/\widetilde\lambda_j$, so that $0\leq y_j\leq nL^{-2r-2}e^{2\bar\alpha}\alpha^2$ by Lemma~\ref{lem:spectral-consequences}. Under $\pi\otimes\pi$, the products $\omega_1\omega'_1,\ldots,\omega_L\omega'_L$ are independent Rademacher random variables. Therefore
$$
\begin{aligned}
\mathbb{E}_{\pi\otimes\pi}\exp\Bigl(\sum_{j=1}^Ly_j\omega_j\omega'_j\Bigr)
&=\prod_{j=1}^L\mathbb{E}_{\pi\otimes\pi}\exp(y_j\omega_j\omega'_j)\\
&=\prod_{j=1}^L\frac{e^{y_j}+e^{-y_j}}{2}\\
&=\prod_{j=1}^L\cosh(y_j).
\end{aligned}
$$
Since $(2k)!\geq2^kk!$ for every $k\geq0$, one has $\cosh y=\sum_{k\geq0}y^{2k}/(2k)!\leq\sum_{k\geq0}(y^2/2)^k/k!=e^{y^2/2}$, and therefore
$$
\begin{aligned}
\prod_{j=1}^L\cosh(y_j)
&\leq\exp\Bigl(\frac12\sum_{j=1}^Ly_j^2\Bigr)\\
&\leq\exp\Bigl(\frac L2\cdot\frac{n^2e^{4\bar\alpha}\alpha^4}{L^{4r+4}}\Bigr)=\exp\Bigl(\frac{n^2e^{4\bar\alpha}\alpha^4}{2L^{4r+3}}\Bigr).
\end{aligned}
$$
Combined with the previous display and $\pi(\mathcal A)\geq1-p$, this proves the lemma.

<!-- prop-5: body_md -->
\label{prop:private-composite-lower}
Let $\rho_n=o(\rho_n^\ast)$. For every sequence of non-interactive $\alpha$-LDP channels and every sequence of tests $\Psi_n$,
$$
\begin{aligned}
\liminf_{n\to\infty}\Bigl[&\sup_{f\in\Sigma(s,R,U)}P_f^{Q^n}(\Psi_n=1)\\
&+\sup_{f\in H_1(\rho_n)}P_f^{Q^n}(\Psi_n=0)\Bigr]\geq1.
\end{aligned}
$$
<!-- end prop-5 -->
<!-- proof-18: body_md -->
Replacing $\rho_n$ by $\rho_n\vee(\rho_n^\ast)^2$ shrinks $H_1(\rho_n)$, and therefore strengthens the claim, while preserving the assumption, because $\rho_n^\ast\to0$. We may thus assume that $\rho_n>0$. Let $L_n=2^{J_n}$ be the smallest power of two that is at least $(n\alpha^2)^{2/(4r+3)}(\rho_n^\ast/\rho_n)^{1/(2r)}$. Then $L_n\to\infty$, and
$$L_n^{-r}>2^{-r}(n\alpha^2)^{-2r/(4r+3)}\Bigl(\frac{\rho_n}{\rho_n^\ast}\Bigr)^{1/2}=2^{-r}\rho_n^\ast\Bigl(\frac{\rho_n}{\rho_n^\ast}\Bigr)^{1/2}.$$
We deduce that
$$
\begin{aligned}
\frac{\rho_n}{L_n^{-r}}&\leq2^r\Bigl(\frac{\rho_n}{\rho_n^\ast}\Bigr)^{1/2}\to0,\\
\frac{n^2\alpha^4}{L_n^{4r+3}}&\leq\Bigl(\frac{\rho_n}{\rho_n^\ast}\Bigr)^{(4r+3)/(2r)}\to0.
\end{aligned}
$$
Apply the construction of Appendix~\ref{app:prior} at level $J_n$, with the given channels, and let $\mathcal A_n=\mathcal A_{L_n,p_n}$ with $p_n=L_n^{-2}$. By Lemma~\ref{lem:probabilistic-positivity}, $\pi(\mathcal A_n)\geq1-p_n$. Moreover, the bound on $\|f_\omega-1\|_\infty$ in that lemma is of order $L_n^{-r}(\log L_n)^{1/2}$, which tends to zero, so that $f_\omega\in\Sigma(r,R,U)$ for every $\omega\in\mathcal A_n$ when $n$ is large enough. By Lemma~\ref{lem:alternative-integral-regularity}, the distance from each such $f_\omega$ to $\Sigma(s,R,U)$ is at least $L_n^{-r}/\sqrt2-RL_n^{-s}$, which exceeds $\rho_n$ for $n$ large enough, because $s>r$ and $\rho_n=o(L_n^{-r})$. The conditioned prior $\Pi_n$ of Lemma~\ref{lem:conditioned-prior-second-moment} is therefore supported on $H_1(\rho_n)$, and that lemma gives
$$\mathbb{E}_1^{Q^n}\mathcal L_n^2\leq(1-L_n^{-2})^{-2}\exp\Bigl(\frac{n^2\alpha^4}{2L_n^{4r+3}}\Bigr)\to1.$$
Since $\mathbb{E}_1^{Q^n}\mathcal L_n=1$, it follows that
$$
\begin{aligned}
\|P_{\Pi_n}^{Q^n}-P_1^{Q^n}\|_{\operatorname{TV}}&=\frac12\mathbb{E}_1^{Q^n}|\mathcal L_n-1|\\
&\leq\frac12(\mathbb{E}_1^{Q^n}\mathcal L_n^2-1)^{1/2}\to0.
\end{aligned}
$$
Finally, let $\Psi_n$ be any test. Applying the definition of the total variation distance to the event $\{\Psi_n=1\}$ gives
$$
\begin{aligned}
P_{\Pi_n}^{Q^n}(\Psi_n=0)&=1-P_{\Pi_n}^{Q^n}(\Psi_n=1)\\
&\geq1-P_1^{Q^n}(\Psi_n=1)\\
&\quad-\|P_{\Pi_n}^{Q^n}-P_1^{Q^n}\|_{\operatorname{TV}},
\end{aligned}
$$
that is, $P_1^{Q^n}(\Psi_n=1)+P_{\Pi_n}^{Q^n}(\Psi_n=0)\geq1-\|P_{\Pi_n}^{Q^n}-P_1^{Q^n}\|_{\operatorname{TV}}$. We now bound the two terms on the left by the two suprema in the claim. Since the uniform density belongs to $\Sigma(s,R,U)$,
$$P_1^{Q^n}(\Psi_n=1)\leq\sup_{f\in\Sigma(s,R,U)}P_f^{Q^n}(\Psi_n=1).$$
Since $P_{\Pi_n}^{Q^n}=\int P_f^{Q^n}\,d\Pi_n(f)$ and, for $n$ large enough, $\Pi_n$ is supported on $H_1(\rho_n)$,
$$
\begin{aligned}
P_{\Pi_n}^{Q^n}(\Psi_n=0)&=\int P_f^{Q^n}(\Psi_n=0)\,d\Pi_n(f)\\
&\leq\sup_{f\in H_1(\rho_n)}P_f^{Q^n}(\Psi_n=0).
\end{aligned}
$$
Hence, for $n$ large enough, the bracket in the claim is at least $1-\|P_{\Pi_n}^{Q^n}-P_1^{Q^n}\|_{\operatorname{TV}}$. Since the total variation distance tends to zero, the limit inferior of the bracket is at least one.

Under bounded privacy, the private term of $\rho_{n,\mathrm{test}}(r,\alpha)$ dominates (Section~\ref{sec:model}), so that $\rho_{n,\mathrm{test}}(r,\alpha)\asymp(n\alpha^2)^{-2r/(4r+3)}=\rho_n^\ast$. For every $f\in\Sigma(r,R,U)$ and every $J>j_0$,
$$
\begin{aligned}
&\inf_{g\in\Sigma(s,R,U)}\|K_J(f-g)\|_2^2\\

Let $g\in\Sigma(s,R,U)$. For $j\geq J$, the Besov constraints give $\|\beta_j(f)-\beta_j(g)\|_{\ell^2}\leq R(2^{-jr}+2^{-js})$, so that, by (W1),

<!-- lem-14: body_md -->
\label{lem:smooth-private-estimator}
The smooth class is a non-empty, closed and convex subset of $L^2([0,1])$. Consequently, the $L^2$ projection $\overline f_s$ of $\widetilde f_{m_1}$ onto $\Sigma(s,R,U)$ is well defined, it is a measurable function of the first-split release, and
$$\sup_{f\in\Sigma(s,R,U)}\mathbb{E}_f^{Q^{m_1}}\|\overline f_s-f\|_2^2\leq a_{m_1}(s).$$
<!-- end lem-14 -->
<!-- proof-22: body_md -->
The class contains the uniform density, and it is convex because the constraints $f\geq0$, $\int_0^1f=1$, $\|f\|_\infty\leq U$ and $\|f\|_{B^s_{2,\infty}}\leq R$ are preserved by convex combinations.

Let $f$ be the $L^2$ limit of a sequence in $\Sigma(s,R,U)$, so that the sequence also converges in $L^1([0,1])$ and $\int_0^1f=1$. Moreover, every sequence converging in $L^2$ has a subsequence converging almost everywhere. Since every element $h$ of the sequence satisfies $0\leq h\leq U$ almost everywhere and the interval $[0,U]$ is closed, the almost-everywhere limit satisfies $0\leq f\leq U$. Next, each wavelet coefficient is a continuous linear functional on $L^2([0,1])$: by the Cauchy-Schwarz inequality and (W1), $|\langle h-f,\psi_{jk}\rangle|\leq\|h-f\|_2$, so every coefficient converges along the sequence. Since each level contains finitely many coefficients, the norms $\|\beta_j(h)\|_{\ell^2}$ converge to $\|\beta_j(f)\|_{\ell^2}$, and the bound $2^{js}\|\beta_j(h)\|_{\ell^2}\leq R$, valid for every element $h$ and every $j\geq j_0$, therefore also holds for $f$. We conclude that $\|f\|_{B^s_{2,\infty}}\leq R$. The class is therefore closed.


The projection onto a non-empty closed convex subset of a Hilbert space exists, is unique and is non-expansive, hence continuous, so that $\overline f_s$ is a measurable function of $\widetilde f_{m_1}$. For $f\in\Sigma(s,R,U)$, non-expansiveness gives $\|\overline f_s-f\|_2\leq\|\widetilde f_{m_1}-f\|_2$, and Proposition~\ref{prop:adaptive-private-estimator} completes the proof.

<!-- prop-6: body_md -->
\label{prop:separated-test}
Let $J>j_0$. Then
$$\sup_{f\in\Sigma(s,R,U)}P_f^{Q^{m_1+m_2}}(\widehat\Psi_J=1)\leq\gamma_n,$$
and, if $\rho^2\geq b_{r,s}(J)+\mathcal H_{\gamma_n,J}(\tau_J)$,
$$\sup_{f\in H_1(\rho)}P_f^{Q^{m_1+m_2}}(\widehat\Psi_J=0)\leq\gamma_n.$$
<!-- end prop-6 -->
<!-- proof-23: body_md -->
Given the first split, $\overline f_s$ is fixed, and Lemma~\ref{lem:projected-loss-moments}, applied with $g=\overline f_s$ and $m=m_2$, gives $\mathbb{E}_2\widehat T_J=q_J(\overline f_s)^2$,
$$\operatorname{Var}_2(\widehat T_J)\leq4v_Jq_J(\overline f_s)^2+w_J^2,$$
and $\mathbb{E}_2\widehat T_J^+\leq q_J(\overline f_s)^2+\frac12\{4v_Jq_J(\overline f_s)^2+w_J^2\}^{1/2}$, where $q_J(\overline f_s)^2=\|K_J(f-\overline f_s)\|_2^2$.

Let $f\in\Sigma(s,R,U)$. The squared projected loss $q_J(\overline f_s)^2$ depends only on the first split, and, since $K_J$ is a contraction, Lemma~\ref{lem:smooth-private-estimator} gives
$$\mathbb{E}_f^{Q^{m_1}}q_J(\overline f_s)^2\leq\mathbb{E}_f^{Q^{m_1}}\|f-\overline f_s\|_2^2\leq a_{m_1}(s).$$
Taking expectations over the first split in the bound on $\mathbb{E}_2\widehat T_J^+$ gives
$$
\begin{aligned}
\mathbb{E}_f^{Q^{m_1+m_2}}\widehat T_J^+
&\leq\mathbb{E}_f^{Q^{m_1}}q_J(\overline f_s)^2\\
&\quad+\frac12\mathbb{E}_f^{Q^{m_1}}\{4v_Jq_J(\overline f_s)^2+w_J^2\}^{1/2}.
\end{aligned}
$$
The map $y\mapsto(4v_Jy+w_J^2)^{1/2}$ is concave and non-decreasing on $[0,\infty)$, so Jensen's inequality and the previous bound give
$$
\begin{aligned}
&\mathbb{E}_f^{Q^{m_1}}\{4v_Jq_J(\overline f_s)^2+w_J^2\}^{1/2}\\
&\quad\leq\{4v_J\mathbb{E}_f^{Q^{m_1}}q_J(\overline f_s)^2+w_J^2\}^{1/2}\\
&\quad\leq\{4v_Ja_{m_1}(s)+w_J^2\}^{1/2}.
\end{aligned}
$$
Hence, by the definition of $\tau_J$,
$$
\begin{aligned}
\mathbb{E}_f^{Q^{m_1+m_2}}\widehat T_J^+
&\leq a_{m_1}(s)+\frac12\{4v_Ja_{m_1}(s)+w_J^2\}^{1/2}\\
&=\gamma_n\tau_J.
\end{aligned}
$$
As $\tau_J>0$, the events $\{\widehat T_J>\tau_J\}$ and $\{\widehat T_J^+>\tau_J\}$ coincide, and Markov's inequality gives
$$
\begin{aligned}
P_f^{Q^{m_1+m_2}}(\widehat\Psi_J=1)
&=P_f^{Q^{m_1+m_2}}(\widehat T_J^+>\tau_J)\\
&\leq\frac{\mathbb{E}_f^{Q^{m_1+m_2}}\widehat T_J^+}{\tau_J}\leq\gamma_n.
\end{aligned}
$$

Now let $f\in H_1(\rho)$. Since $\overline f_s\in\Sigma(s,R,U)$, Lemma~\ref{lem:projected-density-separation} gives
$$q_J(\overline f_s)^2\geq\rho^2-b_{r,s}(J)\geq\mathcal H_{\gamma_n,J}(\tau_J)>\tau_J.$$
The map $y\mapsto(y-\tau_J)^2-c_{\gamma_n}(4v_Jy+w_J^2)$ is a convex quadratic whose larger root is $\mathcal H_{\gamma_n,J}(\tau_J)$, as computed in the proof of Proposition~\ref{prop:projected-loss-upper}. It is therefore non-negative at $y=q_J(\overline f_s)^2$, which gives
$$\{q_J(\overline f_s)^2-\tau_J\}^2\geq c_{\gamma_n}\{4v_Jq_J(\overline f_s)^2+w_J^2\}.$$
Cantelli's applied, given the first split, to $\widehat T_J$ whose conditional mean is $q_J(\overline f_s)^2$ and whose conditional variance is at most $4v_Jq_J(\overline f_s)^2+w_J^2$, and the distance $q_J(\overline f_s)^2-\tau_J>0$ gives
$$
\begin{aligned}
P_2(\widehat T_J\leq\tau_J)
&\leq\frac{4v_Jq_J(\overline f_s)^2+w_J^2}{4v_Jq_J(\overline f_s)^2+w_J^2+\{q_J(\overline f_s)^2-\tau_J\}^2}\\
&\leq\frac{1}{1+c_{\gamma_n}}=\gamma_n,
\end{aligned}
$$
where we used that $\{\widehat T_J\leq\tau_J\}$ is exactly the event that $\widehat T_J$ lies at a distance below $q_J(\overline f_s)^2-\tau_J$ from its expectation, and the second inequality uses the lower bound on $\{q_J(\overline f_s)^2-\tau_J\}^2$ obtained above and $1+c_{\gamma_n}=1/\gamma_n$. Averaging over the first split completes the proof.

We prove honesty at level $1-\gamma$, and the two diameter bounds in turn.

For honesty, note first that both classes in the claim are contained in $\Sigma(r,R,U)$, so there is nothing to prove when $C_n^{\mathrm{den}}(\gamma)=\Sigma(r,R,U)$. We may therefore assume that $\rho_n\geq\rho_{\mathrm{FS}}$. For $t\in\{r,s\}$ and $f\in\Sigma(t,R,U)$, one has $f\in C_t$ unless $\|\widetilde f_{m_3}-f\|_2^2>2a_{m_3}(t)/\gamma$, an event of probability at most $\gamma/2$ by Markov's inequality and Proposition~\ref{prop:adaptive-private-estimator}. For $f\in\Sigma(s,R,U)$, non-coverage requires $f\notin C_s$ or $\widehat\Psi_{J_{\mathrm{sep}}}=1$, while for $f\in H_1(\rho_n)$, it requires $f\notin C_r$ or $\widehat\Psi_{J_{\mathrm{sep}}}=0$. Since $\rho_n^2\geq\rho_{\mathrm{FS}}^2$, Proposition~\ref{prop:separated-test} bounds both testing errors by $\gamma_n\leq\gamma/2$, and a union bound gives $P_f^{Q^n}\{f\notin C_n^{\mathrm{den}}(\gamma)\}\leq\gamma$ in both cases, for every $n$.

We next show that $\rho_n\geq\rho_{\mathrm{FS}}$ for all large $n$. By Proposition~\ref{prop:projected-loss-upper}, $\mathcal H_{\gamma_n,J}(\tau_J)\leq2\tau_J+F_{\gamma_n}(J)$. Since $\{4v_Ja_{m_1}(s)+w_J^2\}^{1/2}\leq a_{m_1}(s)+v_J+w_J$, one has $\tau_J\leq\frac32\gamma_n^{-1}\{a_{m_1}(s)+v_J+w_J\}$, and, as in the proof of Lemma~\ref{lem:finite-resolution-rates}, $F_{\gamma_n}(J)\leq4c_{\gamma_n}v_J+\sqrt{c_{\gamma_n}}w_J\leq4\gamma_n^{-1}(v_J+w_J)$, because $c_{\gamma_n}\leq\gamma_n^{-1}$ and $\gamma_n\leq1$. Moreover, since $s>r$, one has $b_s(J)\leq b_r(J)$ and the last term of $b_{r,s}(J)$ is at most $2b_r(J)$, so that $b_{r,s}(J)\leq4b_r(J)$. Evaluating the criterion defining $J_{\mathrm{sep}}$ at every $J$ and using Lemma~\ref{lem:finite-resolution-rates}, we obtain
$$
\begin{aligned}
\rho_{\mathrm{FS}}^2
&\leq\frac{7}{\gamma_n}\Bigl[a_{m_1}(s)+\inf_{J>j_0}\{b_r(J)+v_J+w_J\}\Bigr]\\
&\lesssim\frac{1}{\gamma_n}\Bigl[a_{m_1}(s)+\inf_{J>j_0}\{b_r(J)+\Gamma_{m_2,\alpha}(J)\}\Bigr].
\end{aligned}
$$
By Proposition~\ref{prop:adaptive-private-estimator} and Lemma~\ref{lem:finite-resolution-rates}, the bracket is at most a constant multiple of $\ell_{m_1}^2\delta_{m_1,\mathrm{est}}(s,\alpha)^2+\ell_{m_2}^2\rho_{m_2,\mathrm{test}}(r,\alpha)^2$. Since $n/5\leq m_1=m_2\leq m_3\leq n$ and $\ell_m$ increases with $m$, this is at most a constant multiple of $\ell_n^2\{\delta_{n,\mathrm{est}}(s,\alpha)\vee\rho_{n,\mathrm{test}}(r,\alpha)\}^2$. As $\gamma_n^{-1}\leq2\ell_n^2/\gamma$, it follows that
$$\rho_{\mathrm{FS}}\lesssim_\gamma\ell_n^2\{\delta_{n,\mathrm{est}}(s,\alpha)\vee\rho_{n,\mathrm{test}}(r,\alpha)\},$$
so that $\rho_n\geq\rho_{\mathrm{FS}}$ for all large $n$ when the constant $M$ of Theorem~\ref{thm:density-separated} exceeds the implicit constant.

It remains to bound the diameters, for $n$ large enough that $\rho_n\geq\rho_{\mathrm{FS}}$. Let $f\in\Sigma(s,R,U)$. By Proposition~\ref{prop:separated-test}, $C_n^{\mathrm{den}}(\gamma)=C_s$ with probability at least $1-\gamma_n$, which tends to one because $\ell_n\to\infty$, and, by Proposition~\ref{prop:adaptive-private-estimator},
$$|C_s|_2\leq2\Bigl\{\frac{2a_{m_3}(s)}{\gamma}\Bigr\}^{1/2}\lesssim_\gamma\ell_n\delta_{n,\mathrm{est}}(s,\alpha),$$
which proves the bound over $\Sigma(s,R,U)$. For the bound over $H_1(\rho_n)$, note that $a_m(s)\leq a_m(r)$, because the only terms of $A_m(t)$ that depend on $t$, namely $b_t(K_m)$ and the minima in which $R^22^{-2jt}$ appears, are non-increasing in $t$. Hence both $C_r$ and $C_s$ have diameter at most $2\{2a_{m_3}(r)/\gamma\}^{1/2}\lesssim_\gamma\ell_n\delta_{n,\mathrm{est}}(r,\alpha)$, whatever the outcome of the test, which completes the proof.
