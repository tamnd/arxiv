<!-- lemma-1: body_md -->
\label{lem: feasibility}
There exists a feasible control sequence $u_T \in \mathbb{R}^{mT}$ for Problem \eqref{eq: meci} if and only if
\begin{equation}
    \gamma \geq \underline{\gamma}_{T} \triangleq 
    x_f^\top \left(\Sigma_{e_T}^{-1}
    - \Sigma_{e_T}^{-1}
C_T\left(C_T^\top\Sigma_{e_T}^{-1}C_T\right)^\dagger
    C_T^\top\Sigma_{e_T}^{-1}\right)x_f.
    \label{eq: min_fes}
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Let $g(u)=(C_Tu-x_f)^\top\Sigma_{e_T}^{-1}(C_Tu-x_f)$. The constraint in \eqref{eq: meci} is feasible if and only if
$\gamma\ge \underset{u}{\min}\: g(u)\triangleq \underline{\gamma}_{T}^{\text{(nor)}}$. Note that $\underline{\gamma}_{T}^{\text{(nor)}}$ is the projection error of $x_f$ on Range$(C_T)$ with respect to the $\Sigma_{e_T}^{-1}$ weighted norm. The corresponding projection matrix is given by $ P = C_T(C_T^\top \Sigma_{e_T}^{-1}C_T)^\dagger C_T^\top \Sigma_{e_T}^{-1}.$
Then, we have
\begin{align*}
\underline{\gamma}_{T}^{\text{(nor)}}
&=
(x_f-P x_f)^\top
\Sigma_{e_T}^{-1}
(x_f-Px_f) \\
&=
x_f^\top
(I-P)^\top
\Sigma_{e_T}^{-1}
(I-P)x_f 
=
x_f^\top
\Sigma_{e_T}^{-1}
(I-P)x_f, 
\end{align*}
where the last equality follows from  the symmetryidempotence ($P^2=P$) and self-adjointness ($P^\top\Sigma_{e_T}^{-1}= \Sigma_{e_T}^{-1}P$) of $P$.
Substituting $P$, we get the result. 
\qed

<!-- theorem-1: body_md -->
\label{thm: optimal_cost}
Assume $\gamma \geq \underline{\gamma}_{T}$. Define
\begin{align}
    S_T \triangleq \frac{\Sigma_{e_T}}{\nu_T} + C_TQ_T^{-1}C_T^\top,
\end{align}
and $\nu^*_T > 0$ be the unique solution to the following equation
\begin{align}
f(\nu_T) &\triangleq \frac{1}{\nu_T^{2}}x_f^\top S_T^{-1}\Sigma_{e_T} S_T^{-1} x_f = \gamma.
\label{eq: nu*}
\end{align}
Further, define
\begin{align}
C_{T,s} &\triangleq \big[ Q_T^{-1/2}C_T^\top S_T^{-1}\ \big]^\dagger Q_T^{1/2}, \quad 
W_{T,s} \triangleq 
C_{T,s}Q_T^{-1}C_{T,s}^\top.
\end{align}
Then, the solution to QCQP \eqref{eq: meci} is
\begin{equation}
u_T^* =
\left\{
\begin{aligned}
&0, \quad \text{if } \gamma \geq x_f^\top \Sigma_{e_T}^{-1} x_f,\\[3mm]
&Q_T^{-1}C_{T,s}^{\top} W_{T,s}^\dagger x_f,
\quad \text{if } \underline{\gamma}_{T} \le \gamma < x_f^\top \Sigma_{e_T}^{-1} x_f.\\[3mm]
\end{aligned}
\right.
\label{eq: solution}
\end{equation}
Further, the optimal cost is given by
\begin{align}
   J^*_T &= x_f^\top W_{T,s}^\dagger x_f
    = x_f^\top S_T^{-1}(C_TQ^{-1}_TC_T^\top)S_T^{-1}x_f. 
    \label{eq: optimal_cost}
\end{align}
<!-- end theorem-1 -->
<!-- proof-2: body_md -->
If $x_f^\top \Sigma_{e_T}^{-1} x_f \geq \gamma$, then \(u_T=0\) is feasible for \eqref{eq: meci} and yields cost \(u_T^\top Q_T u_T = 0\). Since the cost is nonnegative, it follows that \(u_T^* = 0\).
Next, we consider the case $\underline{\gamma}_{T} \leq \gamma < x_f^\top \Sigma_{e_T}^{-1}x_f$ and use KKT conditions to solve \eqref{eq: meci}. Consider the Lagrangian
\begin{align}
  \label{eq:lagrangian}
    \mathcal{L}(u_T, \nu_T) &= u_T^\top Q_Tu_T \\ &+ \nu_T [(C_Tu_T-x_f)^\top \Sigma_{e_T}^{-1}(C_Tu_T-x_f)-\gamma], \nonumber
\end{align}
where $\nu_T \geq 0$ is the Lagrange multiplier. Further, let $u_T^*$ be the optimal solution for QCQP \eqref{eq: meci} and $\nu_T^*$ be the corresponding lagrange multiplier in \eqref{eq:lagrangian}.

Clearly, $u_T^* = 0$ is infeasible for $\underline{\gamma}_{T} \leq \gamma < x_f^\top \Sigma_{e_T}^{-1}x_f$ which implies that $\nu^* \neq 0$, and hence $\nu_T^* > 0$. Hence, from complementary slackness, the constraint must be active, i.e.
\begin{equation}
     (C_Tu_T^*-x_f)^\top\Sigma_{e_T}^{-1}(C_Tu_T^*-x_f) = \gamma.
     \label{eq: short}
\end{equation}
Also, using stationarity KKT condition i.e. $\nabla _{u_T} \mathcal{L}(u_T,\nu_T) = 0$, we get
\begin{align}
\label{eq: optimal_u}
u_T^* &= (Q_T + \nu_T^*C_T^\top \Sigma_{e_T}^{-1}C_T)^{-1} \nu_T^* C_T ^\top \Sigma_{e_T}^{-1}x_f. \\
& \overset{(a)}{=} Q_T^{-1}C_T^\top \big(\frac{\Sigma_{e_T}}{\nu_T^*} + C_T Q_T^{-1}C_T^\top \big)^{-1}x_f
\!\overset{(b)}{=} Q_T^{-1}C_{T,s}^\top W_{T,s}^\dagger x_f, \nonumber
\end{align}
where $(a)$ follows from Woodbury identity and $(b)$ follows from the identity i.e. $B^\top (BB^\top)^\dagger = B^\dagger$. This proves \eqref{eq: solution}. Note that $u_T^*$ in \eqref{eq: optimal_u} is zero in two cases: either $x_f = 0$ or $x_f \in ker(W_{T,s}^\dagger)= ker(C_{T,s}^\top)$, i.e. if $x_f$ is orthogonal to $Range(C_{T,s})$. 
Next, to solve for  $\nu_T^* > 0$,
we substitute the expression for $u^*$ in \eqref{eq: short} and obtain
\begin{align}
C_T u_T^* -x_f
&= (C_TQ_T^{-1}C_T^\top) \big(\frac{\Sigma_{e_T}}{\nu_T^*} + C_TQ_T^{-1}C_T^\top \big)^{-1}x_f - x_f \nonumber \\
&= -\frac{\Sigma_{e_T}}{\nu_T^*}\underbrace{\big(\frac{\Sigma_{e_T}}{\nu_T^*} + C_TQ_T^{-1}C_T^\top \big)^{-1}}_{S_T(\nu_T^*)^{-1}}x_f 
\label{eq: woodbury}
\end{align}
Substituting \eqref{eq: woodbury} in \eqref{eq: short}, we obtain \eqref{eq: nu*}.
Further, to show that $\nu_T^*$ is unique, note that since $S_T\succ0$ and $\Sigma_{e_T}\succ0$, we have $x_f^\top S_T^{-1}\Sigma_{e_T}S_T^{-1}x_f>0$
for $x_f\neq0$. Hence, $f(\nu_T)$ is well-defined for $\nu_T>0$, and since $\frac{d f}{d\nu_T}
=
-\frac{2}{\nu_T^3}
x_f^\top S_T^{-1}\Sigma_{e_T}S_T^{-1}x_f
<0,$
$f(\nu_T)$ is strictly decreasing for $\nu_T > 0$. Hence the equation $f(\nu_T) = \gamma$ admits a unique solution $\nu_T^*$.
Finally, using \eqref{eq: optimal_u}, the closed form expression for optimal cost is
\begin{align}
u_T^{*\top}Q_Tu_T^*
&= (Q_T^{-1}C_{T,s}^\top W_{T,s}^\dagger x_f)^\top
Q_T
(Q_T^{-1}C_{T,s}^\top W_{T,s}^\dagger x_f) \nonumber\\
&=x_f^\top W_{T,s}^\dagger
(C_{T,s}Q_T^{-1}C_{T,s}^\top)
W_{T,s}^\dagger x_f \nonumber\\
&=x_f^\top W_{T,s}^\dagger W_{T,s}W_{T,s}^\dagger x_f
=x_f^\top W_{T,s}^\dagger x_f \nonumber\\
&\overset{(a)}{=}x_f^\top S_T^{-1}C_TQ_T^{-1}C_T^\top S_T^{-1}x_f,
\label{eq: simple_cost}
\end{align}
where $(a)$ follows from $
\left[D^\dagger(D^\dagger)^\top\right]^\dagger
= D^\top D,$
with $D=Q_T^{-1/2}C_T^\top S_T^{-1}.$
This proves \eqref{eq: optimal_cost}. \qed
