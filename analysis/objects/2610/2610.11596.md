<!-- definition-1: body_md -->
\label{oiih2} Let $(Y,\rho)$ be a metric space. The center
of distances of $(Y,\rho)$ is the subset $C(Y)\subseteq\mathbb{R}^+$,
such that  $t\in C(Y)$ if and only if the equation
\(
\rho(y,p)=t
\)
has a solution for every $p\in Y$.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{aarvg}
Let $(Y,\rho)$ be a metric space and let $C(Y)$ be the center of distances of $(Y,\rho)$. The central graph $CG_Y$ of $(Y,\rho)$ is a simple graph such that
\(
V(CG_Y)=Y
\)
and 
\[
\{u,v\}\in E(CG_Y)
\Leftrightarrow
\rho(u,v)\in C(Y)
\]
for all $u,v\in Y$. We will write that the central graph $CG_Y$ is $k$-central if $k=|C(Y)|$.
<!-- end definition-2 -->
<!-- problem-1: body_md -->
\label{qw3fhu}
Describe, up to graph isomorphism, the central graphs $CG_Y$ which have the smallest size
\(
|E(CG_Y)|
\)
and a given order $|V(CG_Y)|$.
<!-- end problem-1 -->
<!-- definition-3: body_md -->
\label{ganb}
Let $X$ be a non-empty set. A {\it metric} on  $X$ is a function $d : X \times X \to \mathbb{R}^+$,  such that for all $x,y,z \in X$:
\begin{enumerate}
\item[{\it(i)}] $d(x,y)=d(y,x)$,
\item[{\it(ii)}]  $d(x,y)=0 \iff x=y$,
\item[{\it(iii)}]  $d(x,y)\le d(x,z)+d(z,y)$.
\end{enumerate}
<!-- end definition-3 -->
<!-- definition-4: body_md -->
\label{ojifrs} Let $(Y,\rho)$ be a metric space. Then the set
\(
D(Y):=\{\rho(x,y):x,y\in Y\}
\)
is called the distance set of the space $(Y,\rho)$.
<!-- end definition-4 -->
<!-- definition-5: body_md -->
\label{eddeej} Let $G_1$ and $G_2$ be graphs. A bijective
 mapping
\(
F: V(G_1) \to V(G_2)
\)
is called an isomorphism of $G_1$ and $G_2$ if the equivalence
\[
\{u,v\} \in E(G_1)
\quad\iff\quad
\{F(u),F(v)\} \in E(G_2)
\]
is valid for all $u,v \in V(G_1)$. The graphs $G_1$ and $G_2$ are
isomorphic if there exists an isomorphism of $G_1$ and $G_2$.
<!-- end definition-5 -->
<!-- definition-6: body_md -->
\label{rre}
The {\it complement} of a graph $G$ is the graph $\overline{G}$ with
$V(\overline{G})=V(G)$, and such that
\[
\{u,v\}\in E(\overline{G})
\iff
\{u,v\}\notin E(G)
\]
for all distinct $u,v\in V(G).$

<!-- proposition-3: body_md -->
\label{mmhhws}
Let $(Y,\rho)$ be a metric space. Then the following statements are equivalent.
\begin{enumerate}
    \item[(i)] The central graph $CG_Y$ is empty,
    \begin{equation}
        \label{ev1}
        E(CG_Y)=\varnothing.
        \end{equation}

    \item[(ii)] There exists a vertex $v_0\in V(CG_Y)$ such that
    \begin{equation}
        \label{ev2}
        \delta_{CG_Y}(v_0)=0. 
        \end{equation}

    \item[(iii)] The center of distances $C(Y)$ of $(Y,\rho)$ contains the unique point $0$,
    \begin{equation}
\label{ev3}
        C(Y)=\{0\}. 
        \end{equation}
\end{enumerate}
<!-- end proposition-3 -->
<!-- proof-1: body_md -->
\(
(i)\Rightarrow(ii).
\)
The implication $(i)\Rightarrow(ii)$ is evidently valid, since equality \eqref{ev1} holds if and only if we have equality \eqref{ev2} for each $v_0\in V(CG_Y)$.

\(
(ii)\Rightarrow(iii).
\)
Let statement $(ii)$ hold. To prove statement $(iii)$, we note that the
inclusion $C(Y)\supseteq \{0\}$ holds by Lemma \ref{khan}.
Thus, equality \eqref{ev3} holds if and only if
\(
t\notin C(Y)
\)
for each $t> 0$.
Suppose, on the contrary that we can find 
\(
t_0\in C(Y)\) such that $t_0>0$.
Then, by Definition \ref{oiih2}, there is $y_0\in Y$ satisfying the equality
\(
\rho(v_0,y_0)=t_0.
\)
Consequently,
\(
\{v_0,y_0\}\in E(CG_Y)
\)
is valid by Definition \ref{aarvg}.
Now using formula \eqref{mmks} with $v=v_0$ and $G=CG_Y$
we obtain the inequality
\(
\delta_{CG_Y}(v_0)\geq 1.
\)
The last inequality contradicts equality \eqref{ev2}. Equality \eqref{ev3}
follows.

\(
(iii)\Rightarrow(i).
\)
If equality \eqref{ev3} holds, then we 
have \eqref{ev1},
because
the central graph $CG_Y$ is simple 
by Definition~\ref{aarvg}, 
and,
consequently, any two adjacent in $CG_Y$ vertices are different.

\label{pujas}
Let $(Y,\rho)$ be a metric space with the distance set $D(Y)$ and the center of distances $C(Y)$. Then the following statements are equivalent.
\begin{enumerate}
    \item[(i)] The central graph $CG_Y$ is complete.
    \item[(ii)] The equality
    \begin{equation}
        \label{wr1}

        \end{equation}

<!-- end proposition-4 -->
<!-- proof-2: body_md -->
$(i)\Rightarrow(ii)$. Let $CG_Y$ be a complete graph. To prove equality \eqref{wr1}, we note that the inclusion
\begin{equation}
    \label{wrwr1}
    D(Y)\supseteq C(Y)
\end{equation}
directly follows from Definition \ref{oiih2}.

Let us prove the converse inclusion
\begin{equation}
    \label{wr2}
    C(Y) \supseteq D(Y).
\end{equation}


Suppose, on the contrary, that there is
\begin{equation}
    \label{wr3}
    t_0\in D(Y)
\end{equation}
such that
\begin{equation}
\label{wr4}
    t_0\notin C(Y).
\end{equation}
Since the inclusion $C(Y)\supseteq \{0\}$ holds by Lemma \ref{khan}, relations \eqref{wr3} and \eqref{wr4} imply
\(
    t_0\in D(Y)\setminus\{0\}.

Consequently, we can find two distinct points $u_0,v_0\in Y$ for which
\begin{equation}

    \rho(u_0,v_0)=t_0>0.
\end{equation}
Definition  \ref{aarvg} and formula \eqref{wr5} imply that $u_0$ and $v_0$ are distinct non-adjacent vertices of the graph $CG_Y$, contrary to statement (i). Thus \eqref{wr2} holds. Equality \eqref{wr1} follows from inclusions \eqref{wrwr1} and \eqref{wr2}.

$(ii)\Rightarrow(i)$. The validity of the implication $(ii)\Rightarrow(i)$  follows from Definitions \ref{aarvg} and \ref{ganb}.

<!-- theorem-3: body_md -->
\label{wsswppj}
Let $H$ be a finite graph with the vertex set $X$ such that $|X|\geq 2$. Then the following statements are equivalent.
\begin{enumerate}
    \item[(i)] 
There exists a metric $d:X\times X\to\mathbb{R}^+$ satisfying the equalities
\begin{equation}
    \label{y1}
H=G_{X}=CG_{X},
\end{equation}
where $G_{X}$ and $CG_{X}$ are the diametrical graph of the
space $(X,d)$ and, respectively, the central graph of this
space.
    \item[(ii)] There exists a metric space $(Y,\rho)$ such that the central graph $CG_Y$ is bicentral and isomorphic to $H$.
    \item[(iii)] The inequality
    \begin{equation}
        \label{m1}
        \delta_H(u) \geq 1
        \end{equation}
    holds for every $u \in X$.
\end{enumerate}
<!-- end theorem-3 -->
<!-- proof-4: body_md -->
$(i) \Rightarrow (ii)$. 
Let $d\colon X\times X\to\mathbb{R}^+$ be a metric satisfying
equalities  \eqref{y1}.
Then, using Definition \ref{aarvg}, Definition \ref{okiu} and formula \eqref{y1}, we obtain the
equality
\begin{equation}
    \label{mmn}
C_X=\{0,\operatorname{diam} X\}. 
\end{equation}

The inequality $|X|\geq 2$ gives us the strict inequality
\(
\operatorname{diam}X>0.
\)
The last inequality and equality \eqref{mmn} imply that $CG_X$ is bicentral
by Definition \ref{aarvg}. Hence statement $(ii)$ is valid with $Y=X$ and $\rho=d$.


$(ii) \Rightarrow (iii)$. Let 
statement $(ii)$ hold. Then there is a metric space $(Y,\rho)$  with bicentral graph $CG_Y$ such that
$CG_Y$ and $H$ are isomorphic. We must prove that
inequality \eqref{m1} holds for each $u \in X$.
Suppose on the contrary  that there exists $u_0 \in X$ satisfying the equality
\begin{equation}
    \label{m2}
\delta_H(u_0)=0. 
\end{equation}
Since \(H\) and \(CG_Y\) are isomorphic, there is an isomorphism
\(
F:V(H)\to V(CG_Y).
\)
Definition \ref{eddeej} and equality \eqref{m2} give us the equality
\begin{equation}
    \label{m3}
\delta_{CG_Y}(F(u_0))=0. 
\end{equation}
The point \(F(u_0)\) belongs to the metric space \((Y,\rho)\) by Definition \ref{aarvg}. Hence \eqref{m3} implies the equality
\begin{equation}
    \label{m4}
C(Y)=\{0\}
\end{equation}
by Proposition \ref{mmhhws}. By Definition \ref{aarvg} we have the equality
\begin{equation}
    \label{m5}
|C(Y)|=2,
\end{equation}
since \(CG_Y\) is bicentral by statement $(ii)$. Equalities \eqref{m4} and \eqref{m5} cannot hold simultaneously.

Thus  equality \eqref{m2} is false. Statement $(iii)$ follows.

$(iii) \Rightarrow (i)$.
Let inequality \eqref{m1} hold for each $u\in X$. Based on the conditions of the theorem being proved, we have the inequality
\(
|X|\geq 2.
\)


Let $\overline{H}$ be the complement of the graph $H$. Since
$H$ is finite, the graph $\overline{H}$  also is finite and,
consequently, there is an injective mapping
\(
f:E(\overline{H})\to[1,2).
\)

Let us define a metric $d \colon X\times X \to \mathbb R^+$ by the rule
\begin{equation}
    \label{p3}
d(x,y):=
\begin{cases}
2, & \text{if }\{x,y\}\in E(H),\\
f(\{x,y\}), & \text{if }\{x,y\}\in E(\overline H),\\
0, & \text{otherwise}
\end{cases}
\end{equation}
for all $x,y\in X$. 
It follows directly from equality \eqref{p3} that $d$ is a symmetric function on $X\times X$, 
\begin{equation}
    \label{p4}
d(x,y)=d(y,x)
\end{equation}
for all $x,y\in X$, and the equality
\begin{equation}
    \label{p5}
    d(x,y)=0
\end{equation}
holds
if and only if $x=y$.
Moreover,
\eqref{p3} implies the double inequality
\begin{equation}
    \label{p6}
    1\leq d(x,y)\leq 2
\end{equation}
for all distinct $x,y\in X$.
Using formulas \eqref{p4}--\eqref{p6} one can easily prove that the triangle inequality
\[
d(x,y) \leq d(x,z) + d(z,y)
\]
holds for all $x,y,z \in X$.
Thus $(X,d)$ is a metric space. 

To prove the validity of statement $(i)$  it suffices to show that \eqref{y1} holds.


Let us prove the equality
\begin{equation}
    \label{sa1}
H=G_X.
\end{equation}
By conditions of the theorem being proved, we have the equality
\(
V(H)=X.
\)
Moreover, Definition \ref{okiu} implies that
\(
V(G_X)=X.
\)
Hence the equality 
\(
V(H)=V(G_X)
\)
holds.
Consequently equality \eqref{sa1}
holds if and only if
\(
E(H)=E(G_X).
\)
The inclusion
\(
f(E(\overline H))\subseteq [1,2),
\)
statement $(iii)$ and formula \eqref{p3} imply the equality
\begin{equation}
    \label{sa3}
\operatorname{diam}X=2.
\end{equation}
Now, using Definition \ref{okiu}, formula \eqref{p3} and equality \eqref{sa3} we obtain the equivalence
\[
\{x,y\}\in E(H)
\Longleftrightarrow
\{x,y\}\in E(G_X)
\]
for all $x,y\in X$. Hence we have  $E(H)=E(G_X)$.
Equality \eqref{sa1} follows.

To complete the proof of \eqref{y1}, it is sufficient to show that
\(
G_X=CG_X.
\)
The last equality holds if and only if the set
\(
\{0,2\}
\)
is the center of distances of the metric space $(X,d)$,
\begin{equation}
    \label{uy1}
C(X)=\{0,2\}.
\end{equation}

Let us prove equality \eqref{uy1}. This equality evidently holds if 
\(
|X|=2.
\)

Let us consider the case 
\begin{equation}
    \label{uy2}
    |X|\geq 3.
\end{equation}

Statement $(iii)$, formula \eqref{p3} 
and Definition \ref{aarvg} imply
the inclusion
\(
C(X)\supseteq\{0,\operatorname{diam}X\}.
\)
Consequently, \eqref{uy1} does not hold if and only if there is $t_0\in C(X)$
such that
\(
0\neq t_0\neq\operatorname{diam}X
\).

Let $0\neq t_0\neq \operatorname{diam}X$ and $t_0\in C(X)$ hold.
Then it follows from \(
0\neq t_0\neq\operatorname{diam}X
\) and formula 
\eqref{p3} that there is 
\(
\{x_0,y_0\}\in E(\overline{H})
\)
satisfying the equality
\(
f(\{x_0,y_0\})=t_0.

Inequality \eqref{uy2} implies the existence $z_0\in X$ such that
\begin{equation}
    \label{wew1}
x_0\neq z_0\neq y_0.
\end{equation}

By Definition \ref{oiih2}, the relations $t_0\in C(X)$ and $t_0\neq 0$ imply that there is $x^*\in X$ such that

The last equality and formula \eqref{p3} give us the equality

\label{savy}
Let $(X,d)$ be a finite metric space containing $n\geq 2$ points.
If the central graph $CG_X$ is non-empty, then the
inequality
\begin{equation}\label{rer1}
|E(CG_X)|\geq \left\lceil\frac{n}{2}\right\rceil
\end{equation}
holds, where
\(
x\mapsto \lceil x\rceil
\)
is the ceiling function.
Moreover, for every integer $n\geq 2$ there exists a
metric space $(Y,\rho)$ such that
\(
|Y|=n
\) 
and
\begin{equation}\label{rer2}
|E(CG_Y)|=\left\lceil\frac{n}{2}\right\rceil.
\end{equation}

<!-- proof-5: body_md -->
Let \(X^*\) be a finite set containing \(n\geq 2\) points and let \(d^*:X^*\times X^*\to\mathbb{R}^+\) be a metric on $X^*$ such that the central graph \(CG_{X^*}\) is non-empty and
\(
\left|E(CG_{X^*})\right|\leq \left|E(CG_Y)\right|
\)
holds for every metric space \((Y,\rho)\) with \(|Y|=|X^*|\) and  non-empty \(CG_Y\). 
It follows directly from the definition of the space $(X^*,d^*)$ that, to prove inequality \eqref{rer1}, it suffices to show that 
\begin{equation}
    \label{rer3}
\left|E(CG_{X^*})\right|
=
\left\lceil\frac{n}{2}\right\rceil.
\end{equation}

Let us prove equality \eqref{rer3}.

If $X^*$ contains exactly $2$ points, then the central
graph $CG_{X^*}$ is isomorphic to the path $P_2$, because $P_2$ is
the unique non-empty graph of order $2$. Since the equalities
\[
|E(P_2)|=1=\left\lceil\frac{1}{2}\right\rceil
\]
hold, we have \eqref{rer3} for $n=2$.


If \(n=3\), then, by Theorem \ref{wsswppj},
the central graph \(CG_{X^*}\) is isomorphic either to the path \(P_3\)
or to the cycle \(C_3\) (see Example \ref{piigdr}). Since
\[
\left|E(P_3)\right|=2<3=\left|E(C_3)\right|
\]
holds,
\(CG_{X^*}\) is isomorphic to \(P_3\) by definition of $(X^*,d^*)$.
Consequently, we have the equalities
\begin{equation}
    \label{rer11}
\left|E(CG_{X^*})\right|
=
\left|E(P_3)\right|
=
2.
\end{equation}
By the definition of ceiling function we also have the equality
\(
\left\lceil\frac{3}{2}\right\rceil=2.
\)
The last equality and \eqref{rer11} imply \eqref{rer3} for \(n=3\).

Let us consider the case
\(
n>3.
\)



First, we will show that the graph \(CG_{X^*}\) is disconnected. Suppose, on the contrary, 
that \(CG_{X^*}\) is a connected graph. Then, by Theorem \ref{kiii}, there is a tree
\begin{equation}
    \label{ur1}
T \subseteq CG_{X^*}
\end{equation}
such that
\(
V(T)=X^*.
\)
By Definition \ref{avct} every tree is a connected graph and, consequently, the
inequality $|X^*|>3$ implies 
\(
\delta_T(v)\geq 1
\)
 for every \(v\in X^*\). Hence, by Theorem \ref{wsswppj}, the tree \(T\) is isomorphic
to the bicentral graph \(CG_Y\) of a metric space \((Y,\rho)\) with \(|Y|=n\).
Since every bicentral graph is non-empty, inclusion
\eqref{ur1}
and the definition of  \((X^*,d^*)\) give us the inequalities
\[
|E(T)|\leq |E(CG_{X^*})|\leq |E(T)|.
\]
Thus the equality
\(
|E(T)|=|E(CG_{X^*})|
\)
holds. The last equality, 
the equalities
\(
n=|V(CG_{X^*})|=|V(T)|,
\)
and Proposition \ref{hfdsxbj} imply
\begin{equation}
    \label{bdhjh}
|E(CG_{X^*})|=n-1.
\end{equation}

The inequality $n> 3$ implies $n-2\geq 2$ and, consequently, we can consider the paths $P_2$ and $P_{n-2}$ with
\(
V(P_2)\cap V(P_{n-2})=\varnothing.
\)
Let $H$ be the disjoint union of $P_2$ and $P_{n-2}$,
\begin{equation*}
H:=P_2\cup P_{n-2}.
\end{equation*}
Then, using Definition \ref{ewaf} and formula \eqref{gasl}, we obtain the equalities
\begin{equation}
    \label{tcr}
|V(H)|=|V(P_2)|+|V(P_{n-2})|
=2+(n-2)=n,
\end{equation}
\begin{equation}
    \label{tcd}
|E(H)|=|E(P_2)|+|E(P_{n-2})|
=(2-1)+(n-2-1)=n-2,
\end{equation}
and the inequality
\(
\delta_H(u)\geq 1
\)
for each $u\in V(H)$. The last inequality and Theorem \ref{wsswppj}
imply the existence of a metric space $(Y^\circ,\rho^\circ)$ such
that the central graph $CG_{Y^\circ}$ is bicentral and isomorphic to $H$. Consequently, equalities \eqref{tcr}, \eqref{tcd} imply $|Y^\circ|=n$ and
\[
|E(CG_{Y^\circ})|=n-2.
\]
The last equality and equality \eqref{bdhjh} give us
the strict inequality
\begin{equation}
    \label{fav}
|E(CG_{Y^\circ})|<|E(CG_{X^*})|. 
\end{equation}
Since each bicentral graph is non-empty, inequality \eqref{fav} contradicts
the definition of the metric space $(X^*,d^*)$.
Thus the central graph $CG_{X^*}$ is disconnected, as stated above. 

Let $G_1^*, G_2^*,\ldots, G_m^*$ be the components of the central graph $CG_{X^*}$.
Then, using Definition \ref{ewaf}, we obtain the equalities
\begin{equation}
    \label{bb1}
|V(CG_{X^*})|=\sum_{i=1}^{m}|V(G_{i}^*)|, 
\end{equation}
\begin{equation}
    \label{bb2}
|E(CG_{X^*})|=\sum_{i=1}^{m}|E(G_{i}^*)|,
\end{equation}
and the equality
\begin{equation}
    \label{bb3}
\delta_{G_i^*}(x_i)=\delta_{CG_{X^*}}(x_i)
\end{equation}
for each \(i\in\{1,\ldots,m\}\) and every \(x_i\in V(G_i^*)\).

Since the central graph \(CG_{X^*}\) is non-empty, Proposition \ref{mmhhws} and equality \eqref{bb3} give us 
\(
\delta_{G_i^*}(x_i)\geq 1
\)
 for each \(i\in\{1,\ldots,m\}\) and every \(x_i\in V(G_i^*)\). Hence the inequality
\begin{equation}
    \label{bb5}
|V(G_i^*)|\geq 2 
\end{equation}
holds for each \(i\in\{1,\ldots,m\}\).


As in the proof of the disconnectedness of graph $CG_{X^*}$ we can show that, for every $i\in\{1,\ldots,m\}$, the inequality
\(
|V(G_i^*)| > 3
\)
imply that $G_i^*$ is a disconnected graph. Since all components of any graph are connected and $G_1^*,\ldots,G_m^*$ are the components of $CG_{X^*}$, the inequality 
\begin{equation}
|V(G_i^*)|  \leq 3
\label{eq:case-two-0}
\end{equation}
holds for every $i\in\{1,\ldots,m\}$.
Now  inequalities \eqref{bb5} and \eqref{eq:case-two-0} imply that we have either
\(
|V(G_i^*)| = 2
\)
or
\begin{equation}
|V(G_i^*)| = 3
\label{eq:case-three}
\end{equation}
whenever $i\in\{1,\ldots,m\}$.

We assert that there is at most one index $i\in\{1,\ldots,m\}$ for which equality \eqref{eq:case-three} holds. Suppose, on the contrary, that $i_1$ and $i_2$ are two distinct indices such that
\begin{equation}
|V(G_{i_1}^*)| = 3 = |V(G_{i_2}^*)|.
\label{eq:two-three}
\end{equation}
Without losing generality, we can assume $i_1=1$ and $i_2=2$.
The connectedness
of $G_1^*$ and $G_2^*$, the equality
$V(G_1^*) \cap V(G_2^*)=\varnothing$, and equalities
\eqref{eq:two-three} with $i_1=1$ and $i_2=2$
imply
\begin{equation}
|E(G_{1}^*\cup G_{2}^*)| \geq 4.
\label{eq:union-edges}
\end{equation}




Let $3P_2$ be the disjoint union
of $3$ copies of the path $P_2$
such that
\begin{equation}
V(3P_2)=V(G_1^*\cup G_2^*).
\label{eq:1}
\end{equation}
Definition \ref{ewaf} and formula \eqref{gasl}
imply the equalities
\begin{equation}
|E(3P_2)|=3
\label{eq:2}
\end{equation}
and
\begin{equation}
\delta_{3P_2}(u)=1
\label{eq:3}
\end{equation}
for each $u\in V(3P_2)$. Let us
consider the disjoint union $G^{**}$
of $3P_2$ and  
$G_3^*,\ldots,G_m^*$
\begin{equation}
\label{u0}
G^{**}:=3P_2\cup\left(\bigcup_{i=3}^{m}G_i^*\right).
\end{equation}
Then using \eqref{bb1}, \eqref{bb2},  \eqref{eq:1},  and \eqref{eq:2} we obtain
the equalities
\begin{equation}
|V(G^{**})|
=\sum_{i=1}^{m}|V(G_i^*)|=|X^*|
=n
\label{eq:4}
\end{equation}
and
\begin{equation}
\label{eq:5}
|E(G^{**})|
=|E(3P_2)|
+\sum_{i=3}^{m}|E(G_i^*)|
=3+|E(CG_{X^*})|
-|E(G_1^*\cup G_2^*)|.
\end{equation}
Inequality \eqref{eq:union-edges}, equality \eqref{eq:2}, and equality \eqref{eq:5} imply 
\(
|E(G^{**})|
\leq |E(CG_{X^*})|-1.
\)
Hence we have the strict inequality
\begin{equation}
\label{ie1}
\left|E(G^{**})\right| < \left|E(CG_{X^*})\right|.
\end{equation}
Since the central graph $CG_{X^*}$ is non-empty, the inequality
\(
\delta_{CG_{X^*}}(u) \geq 1
\)
holds for each $u \in V(CG_{X^*})$ by Proposition~\ref{mmhhws}. Moreover, it was noted above that equality~\eqref{eq:3} holds for every vertex $u \in V(3P_2)$. Hence \eqref{u0} and 
Definition~\ref{ewaf} give us the inequality
\[
\delta_{G^{**}}(u) \geq 1
\]
for every vertex $u \in V(G^{**})$. Consequently, by Theorem \ref{wsswppj}, there is a metric space $(Y^{\circ \circ},d^{\circ \circ})$ such that the central graph $CG_{Y^{\circ\circ}}$ is bicentral and isomorphic to $G^{**}$. Since $CG_{Y^{\circ\circ}}$ and $G^{**}$ are isomorphic, inequality \eqref{ie1} implies the inequality
\[
\left|E(CG_{Y^{\circ\circ}})\right|
<
\left|E(CG_{X^*})\right|.
\]
The last inequality and equalities \eqref{eq:4} contradict the definition of the metric space $(X^*,d^*)$.

Our assertion is proved. 



Let us consider an arbitrary component \(G_i^*\) of \(CG_{X^*}\).
If \(|V(G_i^*)|=2\) holds, then \(G_i^*\) is isomorphic to the
path \(P_2\), because \(P_2\) is the unique, up to isomorphism,
connected graph having order two.
Similarly, equality
\begin{equation}
    \label{rr1}
|V(G_i^*)|=3
\end{equation}
implies that \(G_i^*\) is isomorphic
either to the cycle \(C_3\) or to the path \(P_3\).
Now, reasoning in the same way as in the case \(|X^*|=n=3\) and
using equality \eqref{bb2}, we can show that \eqref{rr1} holds if and only if \(G_i^*\) and \(P_3\) are
isomorphic.


Thus, either all components $G_1^*,\ldots,G_m^*$ of the graph $CG_{X^*}$ are isomorphic to $P_2$, or exactly one of them is isomorphic to $P_3$ and the other are isomorphic to $P_2$.


In the first case, $CG_{X^*}$ is isomorphic to the graph $mP_2$ and, consequently, the equalities
\[
|X^*|=|V(CG_{X^*})|=2m
\]
and
\[
|E(CG_{X^*})|=m
\]
hold. These equalities and the equality $n=|X^*|$ give us  equality \eqref{rer3}.

In the second case, the central graph $CG_{X^*}$ is isomorphic to the disjoint union $P_3\cup(m-1)P_2$ and, consequently, we have the equalities
\begin{equation}
    \label{p11}
|X^*|=|V(CG_{X^*})|
=3+2(m-1)=2m+1
\end{equation}
and
\begin{equation}
    \label{p22}
|E(CG_{X^*})|
=2+m-1=m+1. 
\end{equation}
The equality $n=|X^*|$, equalities \eqref{p11}, \eqref{p22} and  definition of ceiling function imply
\[
\left\lceil\frac{n}{2}\right\rceil
=
\left\lceil\frac{2m+1}{2}\right\rceil
=m+1
=|E(CG_{X^*})|.
\]
Thus, equality \eqref{rer3} holds in the second case also.


Now, to complete the proof, it suffices to note that
equality \eqref{rer3} becomes equality \eqref{rer2}  if \((Y,\rho)=(X^*,d^*)\).

The proof is completed.

The theorem being proven is obviously true if $|X|=1$. 


Let us consider the case $|X| \geq 2$. 
Let $C(X)$ be the center of distances of $(X,d)$. Then the relation 
\begin{equation}
    \label{okok}
    \operatorname{diam}X \in C(X)
\end{equation}
holds by Theorem \ref{kmjkusa}. 
 Definition  \ref{aarvg}, Definition \ref{okiu} and relation \eqref{okok} imply the inclusion 
\begin{equation}
    G_X \subseteq CG_X,
    \label{eq:3.3}
\end{equation}
where $G_X$ is the diametrical graph of $(X,d)$. Using Theorem \ref{kmjkusa} and Definition \ref{okiu} again, it is easy to see that $G_X$ is connected. Consequently, by Theorem \ref{kiii}, there is a tree $T$ such that
\begin{equation}
    T \subseteq G_X
    \qquad\text{and}\qquad
    V(T)=V(G_X).
    \label{eq:3.4}
\end{equation}
Now it follows from \eqref{eq:3.3} and \eqref{eq:3.4} that $T$ is a subgraph of $CG_X$ and
\(
    V(T)=V(CG_X).
\)
Hence, we have the inequality
\begin{equation}
    |E(T)| \leq |E(CG_X)|.
    \label{eq:3.5}
\end{equation}
Proposition \ref{hfdsxbj} implies the equality
\begin{equation}
    |E(T)|=|V(T)|-1.
    \label{eq:3.6}
\end{equation}
Inequality \eqref{eq:3.1}  follows from \eqref{eq:3.5}, \eqref{eq:3.6}, and 
\(
    V(T)=V(CG_X)=X
\)
.

Let equality \eqref{eq:3.2} hold. We need to prove that  $CG_X$ is a star graph. To do this,
we first note that $CG_X$ is a connected graph, that follows from inclusion \eqref{eq:3.3}, since the diametrical graph $G_X$ is connected and
\(
V(G_X)=V(CG_X).
\)

Since $CG_X$ is a finite connected graph, equality \eqref{eq:3.2} implies that $CG_X$ is a tree by Proposition \ref{hfdsxbj}. 
Using the equivalence $(i)\Longleftrightarrow(ii)$ from Proposition \ref{hfdsxbj}, it is easy to prove that, for each tree $T$  and every connected  $G\subseteq T$ the equality
\(
V(G)=V(T)
\)
holds
 if and only if 
\(
G=T.
\)
Hence, inclusion \eqref{eq:3.3} implies 
 the equality
\begin{equation}
    \label{sf1}
G_X=CG_X. 
\end{equation}
and, consequently, the diametrical graph $G_X$ also is a tree.



It is easy to see that, that each finite tree  $T$ with $|V(T)|\geq 2$ contains a vertex $v \in V(T)$ such that  
\(
\delta_T(v)=1.
\)
Hence, there is $x_1\in X$ such that
\(
\delta_{G_X}(x_1)=1. 
\)


By Theorem \ref{kmjkusa} the diametrical graph $G_X$ is complete $k$-partite with $k\geq 2$.
Let \(X_1,\ldots,X_k\) be the parts of \(G_X\), and let \(x_1\in X_1\) satisfy the equality 
\(\delta_{G_X}(x_1)=1.\)  Then, using Definition \ref{edrtaa} and formula \eqref{mmks}
we obtain the equalities
\(
k=2
\) and \(
|X_2|=1.
\)
Let \(c\) be the unique point of \(X_2\). Then Definition \ref{edrtaa}
and Definition \ref{okiu} imply the relation
\(
\{c,x\}\in E(G_X)
\)
for each \(x\in X_1\), and the relation
\(
\{u,v\}\notin E(G_X),
\)
whenever \(u,v\in X_1\).
Thus \(G_X\) is a star graph by Definition \ref{hihu}. Hence
\(CG_X\)  also is a star graph by equality \eqref{sf1}.




Now to complete the proof we only note that equality \eqref{eq:3.2} follows from Proposition \ref{hfdsxbj}, if $CG_X$ is a star graph, because  each star graph is a tree.

The proof is completed.
