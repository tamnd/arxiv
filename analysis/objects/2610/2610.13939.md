<!-- mainthm-1: body_md -->
\label{threefoldtatemainl}
	Let $X$ be a Calabi--Yau threefold over a finite field $k$ which satisfies one of the following conditions:
	\begin{enumerate}
		\item $X$ is  ordinary;
		\item $X$ is $1$-ordinary and strict, i.e.\   $\mathrm{H}^1(X,\mathcal{O}_X) = \mathrm{H}^2(X,\mathcal{O}_X) = 0$. 
	\end{enumerate}
	Then $X$ satisfies  the integral geometric Tate conjecture.
<!-- end mainthm-1 -->
<!-- mainthm-2: body_md -->
\label{divisortate}
	Let $X$ be a $2$-ordinary Calabi--Yau variety with   $\mathrm{H}^2(X, \mathcal{O}_X) = 0$.
	Then $X$ satisfies the integral geometric Tate conjecture for divisor classes.
<!-- end mainthm-2 -->
<!-- remark-1: body_md -->
We can replace the ordinarity assumption in Theorem A(1) and the $2$-ordinarity assumption in Theorem B by the condition that $X$ is Bloch--Kato $2$-ordinary with $\mathrm{H}^{\dim(X)}_{\acute et}(X_{\overline{k}}, \mathbf{Z}_p)$ torsion-free.
<!-- end remark-1 -->
<!-- theorem-1: body_md -->
Let $Y$ be a smooth complex projective variety (or a compact K\"{a}hler manifold).
	The cycle class map $$ \mathrm{Pic}(Y) \longrightarrow \mathrm{H}^{2}(Y,\mathbf{Z}) \cap \mathrm{H}^{1,1}(Y,\mathbf{C})$$
	is surjective.
<!-- end theorem-1 -->
<!-- lemma-1: body_md -->
\label{liftgivesdivisors}
Let $X$ be a smooth projective variety 
 over a  field $k \subseteq \overline{\mathbf{F}}_{p}$.
Assume that 
$X$ admits a smooth projective lift $\mathcal{X}$  to the Witt vectors $W(k)$
satisfying $\mathrm{H}^2(\mathcal{X}_K, \mathcal{O}_{\mathcal{X}_K})  = 0 $ for $K = \mathrm{Frac}(W(k))$.	Then $X$ satisfies the integral geometric Tate conjecture on codimension-1 cycles.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Maulik--Poonen's refinement \cite[Proposition 3.6]{zbMATH06073742} of Grothendieck's specialisation results  
	\cite[Expos\'{e} X, Appendice 7]{ferrand2006theorie} gives a commutative diagram
	$$ \begin{tikzcd}
		\operatorname{NS}(\mathcal{X}_{\overline{K}}) \otimes \mathbf{Z}_{\ell} \arrow[r,hook,"c_1"] \arrow[d,"\mathrm{sp}"] & \mathrm{H}^2_{\acute et} (\mathcal{X}_{\overline{K}}, \mathbf{Z}_\ell(1))\arrow[d,"\cong"] 
		\\ 
		\operatorname{NS}(X_{\overline{k}}) \otimes \mathbf{Z}_{\ell} \arrow[r,hook, "c_1"] & \mathrm{H}^2_{\acute et} (X_{\overline{k}}, \mathbf{Z}_\ell(1))
	\end{tikzcd}$$
	To verify the claim, it suffices to check that the top horizontal map is an isomorphism, and by \cite[Lemma 6.2]{totaro2021integral}, it is enough to do this
	 rationally (note that $\mathcal{X}_{\overline{K}}$ descends to some finitely generated subfield over $\mathbf{Q}$).
This is in turn a consequence of the Lefschetz $(1,1)$ theorem (cf.\ e.g.\  \cite[Proposition 2.3]{gregory2019p}).
<!-- end proof-1 -->
<!-- proof-2: body_md -->
By \cite[Theorem C]{CYDef}, $X$ admits a canonical lift to $W(k)$, and the claim follows from Lemma \ref{liftgivesdivisors} 
	as upper semicontinuity (cf.\ \cite[0BDN]{stacks-project}) implies $\mathrm{H}^2(\mathcal{X}_K, \mathcal{O}_{\mathcal{X}_K}) = 0$.
<!-- end proof-2 -->
<!-- lemma-2: body_md -->
\label{threefoldtategenerallemma}
	Let $X$ be a strict Calabi--Yau threefold  over a field $k \subseteq \overline{\mathbf{F}}_{p}$
 and assume $X$
 admits a smooth projective lift $\mathcal{X}$ to $W(k)$.
	Then $X$ satisfies the integral geometric Tate conjecture.
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
Let us write $K = \mathrm{Frac}(W(k))$ and fix an 
	  embedding  $\overline{K}  \hookrightarrow \mathbf{C}$ throughout.
	We begin by noting that since $X$ is strict, line bundles lift uniquely (cf.\ e.g.\ \cite[Lemma 26.15]{gortz2023algebraic}), which implies that $\mathcal{X}$ has trivial canonical bundle.
	
The case $i=1$ corresponding to divisor classes follows immediately from Lemma \ref{liftgivesdivisors}, as we have $\mathrm{H}^2(\mathcal{X}_K, \mathcal{O}_{\mathcal{X}_K}) = 0$ by upper  semicontinuity.
	We now verify the case $i=2$ of curve classes; our argument is a variant of the proof of  \cite[Lemma  6.3]{totaro2021integral}.
	Let us start with an element $u$  in $\mathrm{H}^{4}_{\acute et} (X_{\overline{k}}, \mathbf{Z}_{\ell} (2))$.	
	Pick an ample line bundle $\mathcal{L}$ on the projective variety $X$ and write $H= c_1(\mathcal{L}) $ for the corresponding class in $ \mathrm{H}_{\acute et}^2(\mathcal{X}_{\overline{k}}, \mathbf{Z}_{\ell}(1))$. By the hard Lefschetz theorem (cf.\ \cite[Théorème 4.1.1]{deligne1980conjecture}), we can pick  a class  $v \in \mathrm{H}_{\acute et}^2(\mathcal{X}_{\overline{k}}, \mathbf{Z}_{\ell}(1))$ satisfying 
	$$  N u  =v  \cup H \ \  \mbox{in} \ \   \mathrm{H}^4_{\acute et}(\mathcal{X}_{\overline{k}}, \mathbf{Z}_{\ell}(2))$$
	for some positive integer $N$.  
	
By smooth proper base change, we can  lift $u$ to  a class $\overline{u}$ in $\mathrm{H}^{4}_{\acute et} (\mathcal{X}_{\overline{K}}, \mathbf{Z}_{\ell} (2))$ and $v$ and $H$ to classes $\overline{v}$ and $\overline{H}$ in $ \mathrm{H}^2_{\acute et}(\mathcal{X}_{\overline{K}}, \mathbf{Z}_{\ell}(1))$, respectively.
 Lemma \ref{liftgivesdivisors} and its proof
show that  both $\overline{v}$ and $\overline{H}$ are  algebraic, and so  the class $$N\overline{u} = \overline{v} \cup \overline{H}$$  is also a  $\mathbf{Z}_\ell$-linear combination of classes
	of subvarieties of $\mathcal{X}_{\overline{K}}$. 
	The image of $N\overline{u}$ under the isomorphism $$\mathrm{H}^{4}_{\acute et} (\mathcal{X}_{\overline{K}}, \mathbf{Z}_{\ell} (2)) \xrightarrow{\ \cong\ } \mathrm{H}^{4}(\mathcal{X}_{\mathbf{C}}, \mathbf{Z}) \otimes  \mathbf{Z}_{\ell}$$ is a $\mathbf{Z}_{\ell}$-linear combination of integral Hodge classes, and since $$\mathrm{H}^{4}(\mathcal{X}_{\mathbf{C}}, \mathbf{Z})/ (\mathrm{H}^{4}(\mathcal{X}_{\mathbf{C}}, \mathbf{Z}) \cap \mathrm{H}^{2,2}(\mathcal{X}_{\mathbf{C}}, \mathbf{C}))$$ is torsion-free, the class $\overline{u}$ itself must be a $\mathbf{Z}_\ell$-linear combination of Hodge classes. 
	
	By Totaro's refinement \cite[Theorem 4.1]{totaro2021integral} of Voisin's  \cite[Theorem 2]{voisin2006integral}, this implies  	(as in the proof of \cite[Lemma 6.3]{totaro2021integral})
	that $\overline{u}$ is a 
	$\mathbf{Z}_{\ell}$-linear  combination of curve classes on $\mathcal{X}_{\overline{K}}$.
	The claim then follows by specialising  to $X_{\overline{k}}$.
	
	The cases $i=0$ and $i=3$ are immediate.
<!-- end proof-3 -->
<!-- proof-4: body_md -->
Let us first assume that  $X$ is Bloch--Kato $2$-ordinary  with 
	$\mathrm{H}^3_{\acute et}(X_{\overline{k}}, \mathbf{Z}_p)$ torsion-free. By 
	\cite[Corollary 1.7, Theorem C]{CYDef}, $X$ admits a canonical lift  $\mathcal{X}$ over $W(k)$.
As the Newton polygon of its special fibre lies above the Hodge polygon of its generic fibre (cf.\ e.g.\ \cite[8.3.6]{arthur1994f}),  $\mathcal{X}_K$ is  also a \mbox{Calabi--Yau variety.}
	 
	If $b_1(X)=0$, then the same argument as in Lemma \ref{threefoldtategenerallemma} implies the result.

So let us assume that  $b_1(X)>0$ and  start with the case of divisor classes. By the proof of  \cite[Theorem 12.8]{patakfalvi2026weak}, there is  a \mbox{generically finite dominant morphism} $$ Z \longrightarrow X_{\overline{k} } $$
with $Z$ either a product of an elliptic curve with a $K3$ surface or an abelian threefold. Here we have used that 
$X_{\overline{k}}$ is $F$-split by  \cite[Proposition 2.2.1]{achinger2018serre}.



By \cite[Lemma 6.2]{totaro2021integral} and 
\cite[Theorem 5.2]{tate1994conjectures} (see also \cite[Lemma 2.2.1]{ambrosi2019note}), it suffices to verify the rational Tate conjecture for divisors on abelian threefolds as well as products of elliptic curves with $K3$ surfaces.
But these statements follow from Tate's result on divisor classes on abelian varieties and  
the more recent works on K3 surfaces cited in the introduction.
	
	
	

		
		The case of curve classes is more subtle.  
		Let us pick an ample line bundle $\mathcal{L}$ on $X$.
	Given  a   class $u$ in $ \mathrm{H}^{4}_{\acute et} (X_{\overline{k}}, \mathbf{Z}_{\ell} (2))$ which is fixed by $\mathrm{Gal}(\overline{k}/l)$ for some finite field extension $l/k$, we  can use hard Lefschetz to write $$ N u =  v \cup H,$$ 
		where $H= c_1(\mathcal{L}) \in\mathrm{H}_{\acute et}^2(X_{\overline{k}}, \mathbf{Z}_{\ell}(1))$, $v \in \mathrm{H}_{\acute et}^2(X_{\overline{k}}, \mathbf{Z}_{\ell}(1))$, and $N$ is a positive integer.
		
		
		
		
	As cupping with $H$ is Galois-equivariant, 
	we may,
 after possibly replacing $l$ by a finite extension  as in the proof of \cite[Lemma 6.3]{totaro2021integral},
  assume  that 
	$v$ is fixed by $\mathrm{Gal}(\overline{k}/l)$ and hence a Tate class.
Hence the claim for divisors shows that $v$ is algebraic, and  we can   write 
$$v= \sum_{i=1}^n \lambda_i c_1(\mathcal{L}_i) $$		
		for suitably chosen line bundles $\mathcal{L}_i$ on $X_{\overline{k}}$ and scalars $\lambda_i\in \mathbf{Z}_{\ell}$.
We then use  \cite[Theorem  8.28]{CYDef}
to lift the line bundles   above 
to corresponding line bundles $\overline{\mathcal{L}}, \overline{\mathcal{L}}_1,\ldots, \overline{\mathcal{L}}_n$ on the generic fibre $\mathcal{X}_{\overline{K}}$.
 The algebraic classes  $$\overline{H} = c_1(\overline{\mathcal{L}})  \ \ \ \ \ \ \ \ \ \ \ \ \ \overline{v} =  \sum_{i=1}^n \lambda_i c_1(\overline{\mathcal{L}}_i)$$		
specialise to $H$ and $v$, respectively, and we deduce  that $u$ is algebraic by the same argument as in the proof of Lemma \ref{threefoldtategenerallemma}.


 
 If $X$ is strict and $1$-ordinary, \cite[Theorem A]{Ward} shows that $X_{\overline{k}}$ admits a smooth projective lift to   $W(\overline{k})$. Since $b_1(X)=0$, the claim follows from Lemma \ref{threefoldtategenerallemma}.
 
 	
 The cases $i=0$ and $i=3$ are again immediate.
