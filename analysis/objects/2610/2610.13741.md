<!-- theorem-1: body_md -->
\label{thm:fischer} Let $\bar {\bf u} \in {\mathcal S}(0)$, where ${\mathcal S}$ is the solution mapping from \eqref{solm}. Assume that ${\mathcal S}(0)$ is locally closed around $\bar {\bf u}$ and that the following properties hold for some positive constant $c$:
\begin{enumerate}[noitemsep,topsep=2pt]
\item {\rm(}calmness of solution mapping{\rm)} the solution mapping ${\mathcal S}$ is calm at $0\in {\bf H}'$ for $\bar {\bf u}$;

\item {\rm(}solvability of subproblems{\rm)} there exists a positive constant $\varepsilon_2$ such that for any $\tilde {\bf u} \in \mathbb{B}_{\varepsilon_2}(\bar {\bf u})$   the localized solution set
\begin{equation*}
\big\{ {\bf u}\in {\bf H}\, \big|\, 0\in {\cal A}({\bf u}, \tilde {\bf u})+ G({\bf u})\; \textrm{ and }\; \|{\bf u}-\tilde {\bf u}\|\leq c\, {\rm dist}(\tilde {\bf u}, {\mathcal S}(0))\big\}
\end{equation*}
is nonempty;

\item {\rm(}precision of approximation{\rm)} there exist a positive constant $\varepsilon_3$ such that 
\begin{equation*}
 \sup\big\{\|\Psi({\bf u}) - {\cal A}({\bf u}, \tilde {\bf u})\|~\big|~{\bf u}\in {\bf H}\; \textrm{ with }\;\| {\bf u} - \tilde {\bf u}\|\leq  c\, {\rm dist}(\tilde {\bf u}, {\mathcal S}(0))\big\}\leq o\big({\rm dist}(\tilde {\bf u}, {\mathcal S}(0))\big),
\end{equation*}
for all $\tilde {\bf u}\in \mathbb{B}_{\varepsilon_3}(\bar {\bf u})$.
\end{enumerate}
Then there exists $\varepsilon_0>0$ such that for any starting point ${\bf u}^0$ chosen from $\mathbb{B}_{\varepsilon_0}(\bar {\bf u})$, 
the iterative scheme in \eqref{ukk} 
generates a sequence $\{{\bf u}^k\}_{k\in {\rm I\!N}}$ that is convergent to some $\hat {\bf u} \in {\mathcal S}(0)$. Moreover, the rate of convergence is Q-superlinear.

\label{prop:subr3}
   Assume that  $({\bar{x}}, \bar{y})$ is a solution to the KKT system \eqref{vs}
   and the SOSC in \eqref{sosc} holds. Then, there are positive constants $\tau_1\leq \tau_2$  such that 
\begin{equation}\label{subr3}
\tau_1\big(\|x-{\bar{x}}\|+{\rm dist}(y,\Lambda({\bar{x}}))\big)\leq r(x, y)\leq \tau_2 \big(\|x-{\bar{x}}\|+{\rm dist}(y,\Lambda({\bar{x}}))\big)
\end{equation}
for all $(x, y)$ in some neighborhood of $({\bar{x}}, \bar{y})$.

<!-- proposition-2: body_md -->
\label{ex_sosc}
Let $({\bar{x}}, \bar{y})$ be a solution to the KKT system \eqref{vs} such that the SOSC \eqref{sosc} holds at $({\bar{x}}, \bar{y})$ and $\sigma$ satisfy the upper bound in \eqref{eb}. Set $\bar{z} :=\Phi({\bar{x}})$. Then there exist positive constants $\varepsilon$ and $\gamma$ such that for all $(x, y) \in \mathbb{B}_\varepsilon({\bar{x}}, \bar{y}))$ we have
\begin{equation}\label{s7}
\langle \nabla^2_{xx}L(x, y)u, u\rangle+{\rm d}^2g(\bar{z}, \bar{y})\big(\nabla\Phi(x)u-\sigma(x, y)w\big)+\sigma(x, y)\|w\|^2 \geq \gamma(\|u\|^2+\sigma(x, y)\|w\|^2)
\end{equation}
for all $(u, w) \in {\overline{  K}}(x, y):= \big\{(u, w)~\big|~ \nabla\Phi(x)u-\sigma(x, y)w\in K_g(\bar{z}, \bar{y})\big\}$.
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
Suppose by contradiction that there exist sequences $({x^k}, y^k) \to ({\bar{x}}, \bar{y})$ and $(u^k, w^k)\in {\overline{  K}}({x^k}, y^k)$ satisfying
\begin{equation}\label{s1}
\langle {H_k}u^k, u^k\rangle+{\rm d}^2g(\bar{z}, \bar{y})\big(\nabla\Phi({x^k})u^k-{\sigma_k}w^k\big)+{\sigma_k}\|w^k\|^2 < \frac{1}{k}(\|u^k\|^2+{\sigma_k}\|w^k\|^2),
\end{equation}
with ${H_k}:= \nabla^2_{xx}L({x^k}, y^k)$ and ${\sigma_k}:= \sigma({x^k}, y^k)$. Set $\beta_k:=\sqrt{\|u^k\|^2+{\sigma_k}\|w^k\|^2}$. The latter estimate obviously yields $\beta_k>0$. Passing to a subsequence, if necessary, we can assume that
\begin{equation*}
\Big(\frac{u^k}{\beta_k}, \frac{\sqrt{{\sigma_k}}w^k}{\beta_k}\Big)\to (u, w)\;\;\textrm{as }\, k\to \infty\;\;\textrm{for some }\, (u, w)\neq 0.
\end{equation*}
Recall that $\nabla\Phi({x^k})u^k-{\sigma_k}w^k \in K_g(\bar{z}, \bar{y})$ for all $k\in {\rm I\!N}$. Dividing the left-hand side of the latter by $\beta_k>0$ and passing through the limit as $k\to\infty$, we get  $\nabla\Phi({\bar{x}})u\in K_g(\bar{z}, \bar{y})$, due to ${\sigma_k}\to 0$ as $k\to\infty$. 
It is worth mentioning that ${\rm d}^2g(\bar{z}, \bar{y})$ is homogeneous of order 2 and convex (cf. \cite[Propositions~13.5 \& 13.20(a)]{rw}. It is therefore continuous relative to its domain ${\rm dom}\, {\rm d}^2g(\bar{z}, \bar{y}) = K_g(\bar{z}, \bar{y})$. Dividing both sides of \eqref{s1} by $\beta_k^2$ gives us
\begin{align*}
\Big\langle {H_k}\frac{u^k}{\beta_k}, \frac{u^k}{\beta_k}\Big\rangle +{\rm d}^2g(\bar{z}, \bar{y})\Big(\nabla\Phi({x^k})\frac{u^k}{\beta_k}-\frac{{\sigma_k}w^k}{\beta_k}\Big)+\frac{{\sigma_k}\|w^k\|^2}{\beta_k^2}
< \frac{1}{k},
\end{align*}
Passing through the limit as $k\to\infty$ then yields
\begin{equation*}
\langle\nabla_{xx}^2L(\bar x,\bar{y})u,u\rangle+{\rm d}^2g (\bar{z},\bar{y}) (\nabla \Phi({\bar{x}})u)+\|w\|^2\leq 0.
\end{equation*}
The SOSC \eqref{sosc} then implies that $u = 0$, and therefore $w = 0$, a contradiction. The proof is then complete.

<!-- corollary-1: body_md -->
\label{cor:SOSC}
Let $({\bar{x}}, \bar{y})$ be a solution to the KKT system \eqref{vs} such that the SOSC \eqref{sosc} holds at $({\bar{x}}, \bar{y})$ and $\sigma$ satisfy the upper bound in \eqref{eb}. Then there exists $\varepsilon>0$ such that for all $(x, y) \in \mathbb{B}_\varepsilon({\bar{x}}, \bar{y})$ satisfying $\sigma(x, y)>0$ the matrix
\begin{equation}\label{s2}
\begin{bmatrix}
\nabla^2_{xx} L(x, y)&\nabla\Phi(x)^*\\
-\nabla\Phi(x)&\sigma(x, y)I
\end{bmatrix}
\end{equation}
is nonsingular.
<!-- end corollary-1 -->
<!-- proof-3: body_md -->
Let $\varepsilon>0$ and $\gamma>0$ be such that \eqref{s7} holds for all $(x, y)\in \mathbb{B}_\varepsilon({\bar{x}}, \bar{y})$ and all $(u, w)\in {\overline{  K}}(x, y)$. 
Fix $(x, y) \in \mathbb{B}_\varepsilon({\bar{x}},\bar{y})$ with $\sigma(x, y)>0$ and let $(u, w)$ be in the kernel of the matrix defined in \eqref{s2}. That means
\begin{equation}\label{s4}
\begin{cases}
\nabla^2_{xx} L(x, y)u+\nabla\Phi(x)^*w = 0,\\
\nabla\Phi(x)u - \sigma(x, y)w = 0.
\end{cases}
\end{equation}
Multiplying both sides of the first equality by $u$ and taking the second one into account, we obtain
\begin{equation}\label{s3}
0 = \langle \nabla^2_{xx} L(x, y)u, u\rangle+\langle\nabla\Phi(x)u, w\rangle =  \langle \nabla^2_{xx} L(x, y)u, u\rangle+\sigma(x, y)\|w\|^2
                                                                    
\end{equation}
Moreover, the second equality of \eqref{s4} particularly implies ${\rm d}^2g(\Phi({\bar{x}}), \bar{y})(\nabla\Phi(x)u-\sigma(x, y)w)=0$ and $(u, w)\in {\overline{  K}}(x, y)$. Thus, combining \eqref{s7} and \eqref{s3}, we get
\begin{equation*}
\|u\|^2+\sigma(x, y)\|w\|^2=0,
\end{equation*}
which yields $(u, w)=(0,0)$, since $\sigma(x, y)>0$, and therefore the matrix in \eqref{s2} is nonsingular. The proof is then complete.

<!-- proposition-3: body_md -->
\label{prop:existence}
Let $({\bar{x}}, \bar{y})$ be a solution to the KKT system \eqref{vs} such that the SOSC \eqref{sosc} holds at $({\bar{x}}, \bar{y})$ and $\sigma$ satisfy the upper bound in \eqref{eb}. Let $\bar{z} := \Phi({\bar{x}})$.
Then there exist $\varepsilon>0$ such that for all $(x, y) \in \mathbb{B}_\varepsilon({\bar{x}},\bar{y})$ with $\sigma(x, y)>0$, the following auxiliary optimization problem
\begin{align}\label{sSQP-1}
    \operatorname*{minimize\;}_{(s, q) \in {{\mathbb R}^n}\times {{\mathbb R}^m}} &\;\langle\nabla \varphi(x),s-x\rangle + \tfrac{1}{2}\langle\nabla^2_{xx}L(x, y)(s - x), s - x\rangle \nonumber\\
    &\; +\langle \bar{y}, \Phi(x) + \nabla\Phi(x)(s - x) - \sigma(x, y)(q - y) -\bar{z}\rangle\\
    &\; + \tfrac{1}{2}{\rm d}^2g(\bar{z}, \bar{y})\big(\Phi(x) + \nabla\Phi(x)(s - x) - \sigma(x, y)(q - y) - \bar{z}\big) + \tfrac{1}{2}\sigma(x, y)\|q\|^2\nonumber
\end{align}
has a nonempty compact solution set.
<!-- end proposition-3 -->
<!-- proof-4: body_md -->
Let $\varepsilon>0$ and $\gamma>0$ be such that \eqref{s7} holds for all $(x, y)\in \mathbb{B}_\varepsilon({\bar{x}}, \bar{y})$ and all $(u, w)\in {\overline{  K}}(x, y)$. 
Fix $(x, y)\in\mathbb{B}_\varepsilon({\bar{x}}, \bar{y})$ with $\sigma(x, y) > 0$ and let $(\hat{u}, \hat{w})$ be the unique solution to the linear system
\begin{equation*}
\begin{bmatrix}
\nabla^2_{xx} L(x, y)&\nabla\Phi(x)^*\\
-\nabla\Phi(x)&\sigma(x, y) I
\end{bmatrix}
\begin{bmatrix} u\\ w\end{bmatrix}
=\begin{bmatrix}
-\nabla_x L(x, y)\\ \Phi(x)-\bar{z}
\end{bmatrix},
\end{equation*}
which exists due to nonsingularity of the matrix on the left-hand side, established in Corollary \ref{cor:SOSC}. Define $(u, w):= (s-x-\hat{u}, q-y-\hat{w})$ and observe that
\begin{equation*}
\Phi(x)+\nabla\Phi(x)(s-x)-\sigma(x, y)(q-y) - \bar{z}
= \nabla\Phi(x)u-\sigma(x, y) w
\end{equation*}
and
\begin{align*}
   &\langle \nabla\varphi(x), s-x\rangle +\tfrac{1}{2}\langle \nabla^2_{xx} L(x, y)(s-x), s-x\rangle + \tfrac{1}{2}\sigma(x, y)\|q\|^2\\
      &=  \tfrac{1}{2}\langle \nabla^2_{xx} L(x, y)u, u\rangle + \tfrac{1}{2}\sigma(x, y)\|w\|^2 -  \langle y + \hat{w}, \nabla\Phi(x)u - \sigma(x, y) w\rangle + c,
\end{align*}
with $c = c (x, y, \hat{u}, \hat{w})$ being independent of $(s, q)$.
Problem \eqref{sSQP-1} therefore can be reformulated, using variable $(u, w)$, as follows
 \begin{align}\label{s6}
\operatorname*{minimize\;}_{(u,w) \in {{\mathbb R}^n}\times {{\mathbb R}^m}}&\; \tfrac{1}{2}\Big[\langle \nabla^2_{xx} L(x, y)u, u\rangle+{\rm d}^2g(\bar{z}, \bar{y})(\nabla\Phi(x)u-\sigma(x, y) w)+\sigma(x, y)\|w\|^2\Big]\nonumber\\
& - \langle y + \hat{w} - \bar{y}, \nabla\Phi(x)u -\sigma(x, y) w\rangle.
\end{align}
Let $\ell(u, w)$ denote the objective function in \eqref{s6}.
Employing the estimate \eqref{s7}, we get
\begin{equation}\label{s14}
\ell(u, w)\geq \frac{\gamma}{2}(\|u\|^2+\sigma(x, y)\|w\|^2) - \langle y + \hat{w} - \bar{y}, \nabla\Phi(x)u -\sigma(x, y) w\rangle
\end{equation}
for all $(u, w)\in {\overline{  K}}(x, y)$. This inequality is evidently true for $(u, w)\notin {\overline{  K}}(x, y)$. It follows from  the Cauchy-Schwarz inequality that
\begin{align*}
- \langle y + \hat{w} - \bar{y}, \nabla\Phi(x)u &-\sigma(x, y) w\rangle  = - \langle \nabla\Phi(x)^*(y + \hat{w} - \bar{y}), u\rangle +\sigma(x, y)\langle y + \hat{w} - \bar{y},  w\rangle\\
 \geq &- \|y + \hat{w} - \bar{y}\|\sqrt{\| \nabla\Phi(x)\|^2+\sigma(x, y)}\cdot\sqrt{\|u\|^2+\sigma(x, y)\|w\|^2},
\end{align*}
or, equivalently,
\begin{equation*}
-\frac{\langle y + \hat{w} - \bar{y}, \nabla\Phi(x)u -\sigma(x, y) w\rangle}{\|u\|^2+\sigma(x, y)\|w\|^2}\geq -\frac{ \|y + \hat{w} - \bar{y}\|\sqrt{\| \nabla\Phi(x)\|^2+\sigma(x, y)}}{\sqrt{\|u\|^2+\sigma(x, y)\|w\|^2}}.
\end{equation*}
Observe that the right-hand side goes to $0$ as $\|(u, w)\|\to \infty$. Thus, for $(u, w)$ with $\|(u, w)\|$ sufficiently large, we have
\begin{equation*}
 - \langle y + \hat{w} - \bar{y}, \nabla\Phi(x)u -\sigma(x, y) w\rangle \geq -\frac{\gamma}{4}(\|u\|^2+\sigma(x, y)\|w\|^2).
\end{equation*}
This, together with \eqref{s14}, proves coercivity of the objective function $\ell(u, w)$. 
We can then find a large positive constant $M$ such that solving \eqref{s6} is equivalent to minimizing $\ell(u, w)$ over the ball $M\mathbb{B}\subset{{\mathbb R}^n}\times{{\mathbb R}^m}$. The just described optimization problem obviously has nonempty and compact solution set, due to the classical Weierstrass theorem, and so does the problem \eqref{sSQP-1}. The proof is then complete.

<!-- proposition-4: body_md -->
\label{prop:solv}
Let $({\bar{x}}, \bar{y})$ be a solution to the KKT system \eqref{vs} such that the SOSC \eqref{sosc} holds at $({\bar{x}}, \bar{y})$ and let $\sigma$ satisfy the upper bound in \eqref{eb}. Then, there exists a positive constant $\varepsilon$   such that for all $(x, y)\in \mathbb{B}_\varepsilon({\bar{x}}, \bar{y})$ with $\sigma(x, y)>0$,  the solution set to the variational system
\begin{equation}\label{au-KKT}
\begin{cases}
\quad \;\;0 = \nabla\varphi(x)+\nabla^2_{xx}L(x, y)(s-x) + \nabla\Phi(x)^*q,\\
q - \bar{y} \in \partial\big(\tfrac{1}{2}{\rm d}^2g(\bar{z}, \bar{y})\big)\big(\Phi(x)+\nabla\Phi(x)(s-x)-\sigma(x, y)(q-y)-\bar{z}\big).
\end{cases}
\end{equation}
is always nonempty.
<!-- end proposition-4 -->
<!-- proof-5: body_md -->
Take the positive constant $\varepsilon$ from  Proposition~\ref{prop:existence} and let $(x, y)\in \mathbb{B}_\varepsilon({\bar{x}}, \bar{y})$ with $\sigma(x, y)>0$. 
According to Proposition~\ref{prop:existence}, the auxiliary optimization problem \eqref{sSQP-1}  has a solution, says $(s, q)$. Thus, the first-order optimality condition   for \eqref{sSQP-1} can be expressed via the chain rule in \eqref{sub_cplq} as 
\begin{equation*}
\begin{cases}
   0 = \nabla\varphi(x)+\nabla^2_{xx}L(x, y)(s-x) + \nabla\Phi(x)^*(\bar{y} +\nu),\\
   0 = \sigma(x, y)(q - \bar{y}) - \sigma(x, y) \nu,\\ 
    \nu \in \partial\big(\tfrac{1}{2}{\rm d}^2g(\bar{z}, \bar{y})\big)\big(\Phi(x)+\nabla\Phi(x)(s-x)-\sigma(x, y)(q-y)-\bar{z}\big),
\end{cases}
\end{equation*}
The second equality and $\sigma(x, y)>0$  yield $\nu = q - \bar{y}$, which in turn tells us that $(s, q)$ indeed solves the  variational system \eqref{au-KKT} and thus completes the proof.

<!-- proposition-5: body_md -->
\label{prop:eb}
Let $({\bar{x}}, \bar{y})$ be a solution to the KKT system \eqref{vs} such that the SOSC \eqref{sosc} holds at $({\bar{x}}, \bar{y})$ and $\sigma$ satisfy both error bound estimates in \eqref{eb}. Then there exist positive constants $\varepsilon$ and $\kappa$ such that for all $(x, y)\in \mathbb{B}_\varepsilon({\bar{x}}, \bar{y})$ with $\sigma(x, y)>0$, the error bound estimate
\begin{equation}\label{eb3}
\|s-x\|+\|q-y\|\leq \kappa\sigma(x, y)
\end{equation}
holds for any solution $(s, q)$ to the system \eqref{au-KKT}.
<!-- end proposition-5 -->
<!-- proof-6: body_md -->
Assume by contradiction that the estimate \eqref{eb3} fails. Then there exist sequence $({x^k}, y^k)\to ({\bar{x}}, \bar{y})$, with $\sigma_k:= \sigma({x^k}, y^k)>0$, and $(s^k, q^k)$ such that
\begin{equation}\label{s8}
\|s^k-{x^k}\|+\|q^k-y^k\|>k \sigma_k,
\end{equation}
where $(s^k, q^k)$ satisfies
\begin{equation}\label{s9}
    0=\nabla\varphi({x^k})+ {H_k}(s^k-{x^k})+\nabla\Phi({x^k})^*q^k,\quad
q^k - \bar{y} \in \partial\big(\tfrac{1}{2}{\rm d}^2g(\bar{z}, \bar{y})\big)(z^k - \bar{z}),
\end{equation}
where ${H_k} := \nabla^2_{xx}L({x^k}, y^k)$, $z^k :
= \Phi({x^k})+\nabla\Phi({x^k})(s^k-{x^k})-\sigma_k(q^k-y^k)$, and $\bar{z} := \Phi({\bar{x}})$.
Denoting by $\beta_k := \|s^k-{x^k}\|+\|q^k-y^k\|$, we infer via \eqref{s8} that $\sigma_k = o(\beta_k)$ as $k\to\infty$. By the lower estimate in \eqref{eb} and the fact that $({x^k}, y^k)\to ({\bar{x}}, \bar{y})$ as $k\to\infty$, we then obtain  
\begin{equation}\label{s10}
\|{x^k} -{\bar{x}}\| = o(\beta_k)\quad\textrm{and}\quad \|y^k -\hat y^k\|=o(\beta_k)\quad\textrm{as } k\to\infty,
\end{equation}
where $\hat y^k := P_{{\Lambda}({\bar{x}})}(y^k)$. Passing to a subsequence, if necessary, we can assume that
\begin{equation}\label{s11}
\Big(\frac{s^k-{x^k}}{\beta_k}, \frac{q^k-y^k}{\beta_k}\Big)\to (u, w)\;\;\textrm{as }\, k\to \infty\;\;\textrm{for some }\, (u, w)\neq 0.
\end{equation}
Recalling that $\hat y^k\in {\Lambda}({\bar{x}})$, we can then equivalently rewrite the first equality in \eqref{s9} as
\begin{equation*}
0=\nabla_x L({x^k}, y^k) - \nabla_x L({\bar{x}}, \hat y^k) +{H_k}(s^k-{x^k})+\nabla\Phi({x^k})^*(q^k-y^k).
\end{equation*}
Dividing both sides by $\beta_k>0$,   passing through the limit as $k\to\infty$, and taking  into account \eqref{s10} and \eqref{s11}, we arrive at
\begin{equation}\label{s12}
0=\nabla^2_{xx}L({\bar{x}}, \bar{y})u+\nabla\Phi({\bar{x}})^*w,
\end{equation}
which yields
\begin{equation}\label{s13}
0=\langle\nabla^2_{xx}L({\bar{x}}, \bar{y})u, u\rangle+\langle\nabla\Phi({\bar{x}})u, w\rangle.
\end{equation}
On the one hand, we observe from the second relation in \eqref{s9} that $z^k - \bar{z} \in {\rm dom}\, {\rm d}^2g(\bar{z}, \bar{y}) = K_g(\bar{z}, \bar{y})$.
Thus, we get via \eqref{s10}, \eqref{s11}, and ${\sigma_k} = o(\beta_k)$ that
\begin{equation}\label{s16}
\frac{z^k-\bar{z}}{\beta_k} = \frac{\Phi({x^k})-\Phi({\bar{x}})}{\beta_k}+\nabla\Phi({x^k})\frac{s^k-{x^k}}{\beta_k}-\frac{\sigma_k}{\beta_k}(q^k-y^k)\to \nabla\Phi({\bar{x}})u\; \; \textrm{as }\; k\to\infty,
\end{equation}
and arrive at $\nabla\Phi({\bar{x}})u\in K_g(\bar{z}, \bar{y})$. 
On the other hand, since that ${\rm d}^2g(\bar{z}, \bar{y})$ is proper, convex, and positively homogeneous of degree $2$, the second inclusion in \eqref{s9} leads us to
\begin{equation*}
{\rm d}^2g(\bar{z}, \bar{y})\Big(\frac{z^k-\bar{z}}{\beta_k}\Big)=\Big\langle \frac{z^k-\bar{z}}{\beta_k},\frac{q^k - \bar{y}}{\beta_k}\Big\rangle,
\end{equation*}
see, e.g., the proof of \cite[Proposition~3.9]{s20}.
This equality in turn implies that
\begin{equation}\label{s15}
{\rm d}^2g(\bar{z}, \bar{y})\Big(\frac{z^k-\bar{z}}{\beta_k}\Big)=\Big\langle \frac{z^k-\bar{z}}{\beta_k},\frac{q^k-\hat y^k}{\beta_k}\Big\rangle+ \frac{1}{\beta_k^2}\langle z^k-\bar{z},\, \hat y^k - \bar{y}\rangle
\leq \Big\langle \frac{z^k-\bar{z}}{\beta_k},\frac{q^k-\hat y^k}{\beta_k}\Big\rangle,
\end{equation}
due to $z^k - \bar{z} \in K_g(\bar{z}, \bar{y}) = N_{\partial g(\bar{z})}(\bar{y})$ and $\hat y^k\in \partial g(\bar{z})$.
Passing to the limit as $k\to\infty$ in \eqref{s15}  and using to \eqref{s11} and \eqref{s16} bring us to 
\begin{equation*}
{\rm d}^2g(\bar{z}, \bar{y})\big(\nabla\Phi({\bar{x}})u\big)\leq \langle \nabla\Phi({\bar{x}})u, w\rangle.
\end{equation*}
Combining this with \eqref{s13}, we arrive at
\begin{equation*}
\langle\nabla^2_{xx}L({\bar{x}}, \bar{y})u, u\rangle+{\rm d}^2g(\bar{z}, \bar{y})\big(\nabla\Phi({\bar{x}})u\big)\leq 0,
\end{equation*}
and which implies via SOSC \eqref{sosc} and $\nabla\Phi({\bar{x}})u\in K_g(\bar{z}, \bar{y})$ that $u=0$. 
Consequently, we get from \eqref{s12} that $\nabla\Phi({\bar{x}})^*w=0$. 

Denote $E:= {\rm ker}\,\nabla\Phi({\bar{x}})^*$. 
We finish the proof by showing that $w=P_E(w)=0$. 
Recall that $z^k - \bar{z}\in K_g(\bar{z}, \bar{y})$ with $K_g(\bar{z}, \bar{y})$ being described in \eqref{domcr}.
Passing to a subsequence,  if necessary, we can assume without loss of generality that there is some index $i_0\in I(\bar{z})$ such that $z^k-\bar{z} \in K_{C_{i_0}}(\bar{z}, \bar{y}_{i_0})$ for all $k$, where $\bar{y}_{i_0}:= \bar{y}-A_{i_0}\bar{z}-a_{i_0}$. Since $K_{C_{i_0}}(\bar{z}, \bar{y}_{i_0})$ is a polyhedral cone, it has the representation
\begin{equation}\label{s21}
K_{C_{i_0}}(\bar{z}, \bar{y}_{i_0}) = \big\{ v\in {\mathbb R}^m\, \big|\, \langle c^j, v\rangle\leq 0, \; j\in J_0\big\}
\end{equation}
for some  $ c^j \in {\mathbb R}^m, j \in J_0$, a finite index set. Denote by $J_k$ the set of the indexes of active inequalities that define $K_{C_{i_0}}(\bar{z}, \bar{y}_{i_0})$, in \eqref{s21}, at $z^k-\bar{z}$ and assume without loss of generality that $J_k = J\subset J_0$ for all $k$. Define
\begin{equation*}
\mathcal{B}:=\big\{v\in K_{C_{i_0}}(\bar{z}, \bar{y}_{i_0}) \, \big|\, \langle c^j, v\rangle =0, \; j\in J\big\}
\end{equation*}
and observe that
\begin{equation}\label{s20a}
  \mathcal{B}^*={\rm span}\,\big\{c^j \,\big|\, j\in J\big\}+{\rm cone}\,\big\{c^j \,\big|\, j\in J_0\setminus J\big\}.
\end{equation}
Clearly, $\mathcal{B}$ is a cone with $z^k-\bar{z}\in \mathcal{B}$ for all $k$, and therefore
\begin{equation}\label{s17}
P_E\Big(\frac{z^k-\bar{z}}{\sigma_k\beta_k}\Big)\in P_E(\mathcal{B}).
\end{equation}
It is worth noting that $P_E$ is a linear mapping and $P_E(\nabla\Phi({\bar{x}}){\mathbb R}^n)=\{0\}$. Since $\Phi$ is $\mathcal{C}^2$-smooth, we have
\begin{equation*}
\frac{\Phi({x^k})-\Phi({\bar{x}})}{\sigma_k\beta_k}=\frac{1}{\sigma_k\beta_k}\nabla\Phi({\bar{x}})({x^k} - {\bar{x}})+O\Big(\frac{\|{x^k} - {\bar{x}}\|^2}{\sigma_k\beta_k}\Big).
\end{equation*}
which in turn, by the fact that $\|{x^k} -{\bar{x}}\|= O({\sigma_k})=o(\beta_k)$, yields
\begin{equation}\label{s18}
P_E\Big(\frac{\Phi({x^k})-\Phi({\bar{x}})}{\sigma_k\beta_k}\Big)= P_E\Big(O\Big(\frac{\|{x^k} - {\bar{x}}\|^2}{\sigma_k\beta_k}\Big)\Big) \to 0 \quad\textrm{as } k\to\infty.
\end{equation}
Moreover, by the same fact and $s^k - {x^k} = o(\beta_k)$, we also obtain  
\begin{equation}\label{s19}
P_E\Big(\frac{1}{\sigma_k\beta_k}\nabla\Phi({x^k})(s^k-{x^k})\Big) = P_E\Big(\frac{\nabla\Phi({x^k})-\nabla\Phi({\bar{x}})}{\sigma_k}\frac{s^k-{x^k}}{\beta_k}\Big) \to 0\quad\textrm{as } k\to\infty.
\end{equation}
Combining \eqref{s11}, \eqref{s16}, and \eqref{s17}--\eqref{s19}, we arrive at $-P_E(w)\in P_E(\mathcal{B})$. Since $w\in E$, the latter inclusion confirms that 
\begin{equation*}
-w =-P_E(w) =P_E(b)\quad\textrm{for some}\quad b\in \mathcal{B}. 
\end{equation*}
Moreover, observe from the second relation in \eqref{s9} and \eqref{pwfor1} that
\begin{equation}\label{s22a}
q^k-\bar{y} \in D(\partial g)(\bar{z}, \bar{y})(z^k-\bar{z})\subset \big\{v\in{\mathbb R}^m\, \big|\, v-A_{i_0}(z^k-\bar{z})\in N_{K_{C_{i_0}}(\bar{z},\, \bar{y}_{i_0})}(z^k-\bar{z})\big\},
\end{equation}
where we utilize the  fact that $z^k-\bar{z}\in K_{C_{i_0}}(\bar{z}, \bar{y}_{i_0})$ for all $k$. Taking into account \eqref{s21} and the definition of the index set $J$, we get from \eqref{s22a} that
\begin{equation}\label{s20}
q^k - \bar{y} - A_{i_0}(z^k-\bar{z})\in N_{K_{C_{i_0}}(\bar{z},\, \bar{y}_{i_0})}(z^k-\bar{z}) = {\rm cone}\,\big\{c^j \,\big|\, j\in J\big\}.
\end{equation}
On the other hand, since $\hat y^k, \bar{y}\in\partial g(\bar{z})$ and $\hat y^k \to \bar{y}$ as $k\to\infty$, we conclude from the reduction relation in \eqref{rl2} and the formulas in \eqref{domcr} and \eqref{pwfor2} that
\begin{equation*}
\hat y^k-\bar{y} \in \big(K_g(\bar{z}, \bar{y})\big)^*\subset \big(K_{C_{i_0}}(\bar{z}, \bar{y}_{i_0})\big)^* = {\rm cone}\,\big\{c^j \,\big|\, j\in J_0\big\},
\end{equation*}
where we use \eqref{s21} to derive the last equality.
Combining this and \eqref{s20} and using \eqref{s20a} result in
\begin{align*}
\hat y^k-q^k+A_{i_0}(z^k-\bar{z}) &\in {\rm cone}\,\big\{c^j \,\big|\, j \in J_0\big\}-{\rm cone}\,\big\{c^j \,\big|\, j\in J\big\}\\
&={\rm span}\,\big\{c^j \,\big|\, j\in J\big\}+{\rm cone}\,\big\{c^j \,\big|\, j\in J_0\setminus J\big\} = \mathcal{B}^*.
\end{align*}
Recall that  $\beta_k>0$. Thus,  we have
\begin{equation*}
-\frac{q^k-\hat y^k}{\beta_k}+A_{i_0}\frac{z^k-\bar{z}}{\beta_k}\in \mathcal{B}^*.
\end{equation*}
Passing to the limit as $k\to\infty$ and taking into account \eqref{s10}, \eqref{s11}, and \eqref{s16} with $u=0$, we deduce that $-w \in \mathcal{B}^*$.
In summary, we have shown that
\begin{equation*}
\begin{cases}
-w =-P_E(w) =P_E(b), \; \; b\in \mathcal{B},\\
-w\in \mathcal{B}^*.
\end{cases}
\end{equation*}
This clearly tell us that  
\begin{equation*}
0\leq \langle w, b\rangle = \langle P_E(w), b\rangle =\langle w, P_E(b)\rangle=\langle w, -w\rangle = -\|w\|^2,
\end{equation*}
and therefore $w=0$. Since $u=0$, the latter clearly contradicts \eqref{s11} and thus completes the proof.

\label{prop:solv2}
Let $({\bar{x}}, \bar{y})$ be a solution to the KKT system \eqref{vs} with $g$ being a polyhedral function. Assume that $\bar{y}\in {\rm ri}\, \partial g(\Phi({\bar{x}}))$ is noncritical and $\sigma$ satisfies both error bound estimates in \eqref{eb}.
Then there exists a positive constant $\varepsilon$ such that for all $(x, y)\in \mathbb{B}_\varepsilon({\bar{x}}, \bar{y})$ with $\sigma(x, y)>0$, the solution set to the variational system in \eqref{au-KKT} is always nonempty.

Set $\bar{z}:=\Phi({\bar{x}})$ and $K:=K_g(\bar{z},\bar{y})$.
Since $g$ is polyhedral, we get from \eqref{domcr} that ${\rm d}^2g(\bar{z},\bar{y})=\delta_K$, and hence,
\begin{equation*}
    D(\partial g)(\bar{z},\bar{y})=\partial\left(\tfrac{1}{2} {\rm d}^2g(\bar{z},\bar{y})\right)=N_K.
\end{equation*}
Moreover,  the critical cone $K=N_{\partial g(\bar{z})}(\bar{y})$ is a linear subspace, due to $\bar{y}\in{\rm ri}\,\,\partial g(\bar{z})$, and we have
\begin{equation*}
    N_K(v)=K^\perp \qquad \mbox{for all } \;v\in K.
\end{equation*}
Thus, the variational  system \eqref{au-KKT} takes the form
\begin{equation}\label{eq:linear}
\begin{cases}
\nabla\varphi(x)+\nabla^2_{xx}L(x,y)(s-x)+\nabla\Phi(x)^*q=0,\\
\Phi(x)+\nabla\Phi(x)(s-x)-\sigma(x,y)(q-y)-\bar z\in K,\\
q-\bar y\in K^\perp.
\end{cases}
\end{equation}
Set $u:=s-x$ and $w:=q-\bar{y}$.
We can then equivalently express \eqref{eq:linear} as the problem of finding
$(u, w)\in{{\mathbb R}^n}\times K^\perp$ such that
\begin{equation}\label{eq:projected}
\begin{cases}
\nabla^2_{xx}L(x,y)u+\nabla\Phi(x)^*w=-\nabla_xL(x,\bar{y}),\\
P_{K^\perp}\big(\nabla\Phi(x)u-\sigma(x,y)w\big)=-P_{K^\bot}\big(\Phi(x)-\bar{z} +\sigma(x, y)(y-\bar{y})\big).
\end{cases}
\end{equation}
Given the pair $(x,y)\in {\mathbb R}^n\times {\mathbb R}^m$, we define the linear operator $ M_{x,y}:{\mathbb R}^n\times K^\perp\to {\mathbb R}^n\times K^\perp$
by
\begin{equation*}
    M_{x,y}(u,w):=
  \Big(\nabla^2_{xx}L(x,y)u+\nabla\Phi(x)^*w,
  P_{K^\perp}\big(\nabla\Phi(x)u-\sigma(x,y)w\big)\Big).
\end{equation*}
We claim that for any pair $(x,y)$ sufficiently close to $({\bar{x}},\bar{y})$ with $\sigma(x,y)>0$,  $M_{x,y}$ is nonsingular. This clearly implies that   \eqref{eq:projected} has a solution.
To justify this claim, suppose on the contrary that it fails.
That means  we can find sequences $({x^k},y^k)\to({\bar{x}},\bar{y})$, with ${\sigma_k}:=\sigma({x^k},y^k)>0$ for all $k$, and $(u^k,w^k)\neq (0, 0)$ in ${\mathbb R}^n\times K^\perp$ satisfying
\begin{equation}\label{eq:non}
  H_ku^k+\nabla\Phi({x^k})^*w^k=0,
  \quad
  \nabla\Phi({x^k})u^k- {\sigma_k} w^k \in K,
\end{equation}
where $H_k:=\nabla^2_{xx}L({x^k}, y^k)$. 
Observe from the inclusion in \eqref{eq:non} that $u^k\neq0$. 
Since, otherwise, we get ${\sigma_k} w^k \in K$, and therefore $w^k=0$ due to ${\sigma_k}\neq0$ and $w^k\in K^\bot$.
We can then assume, without loss of generality, that the sequence $u^k/\|u^k\|$ converges to some nonzero vector $\xi$.
The relations in \eqref{eq:non} then give us
\begin{equation}\label{eq:non1}
H_k\frac{u^k}{\|u^k\|} + \big(\nabla\Phi({x^k})-\nabla\Phi({\bar{x}})\big)^*\frac{w^k}{\|u^k\|}\in \nabla\Phi({\bar{x}})^*K^\bot,
\quad
\nabla\Phi({x^k})\frac{u^k}{\|u^k\|} - {\sigma_k} \frac{w^k}{\|u^k\|} \in K.
\end{equation}
We now claim that $w^k = O(\|u^k\|)$.
Assume otherwise that $w^k\neq 0$ for all $k$ and $u^k = o(\|w^k\|)$ as $k\to\infty$. Dividing the equality and inclusion in \eqref{eq:non} respectively by $\|w^k\|$ and ${\sigma_k}\|w^k\|$, we get that
\begin{equation}\label{eq:non2}
H_k\frac{u^k}{\|w^k\|} + \nabla\Phi({x^k})^*\frac{w^k}{\|w^k\|} = 0,
\quad
\frac{\nabla\Phi({x^k})}{{\sigma_k}}\frac{u^k}{\|w^k\|} -\frac{w^k}{\|w^k\|} \in K.
\end{equation}
Assume, without loss of generality, that $w^k/\|w^k\| \to \eta \neq 0$. Passing to the limit in the first relationship in \eqref{eq:non2}    yields $\nabla\Phi({\bar{x}})^*\eta = 0$.
Denote $E:= {\rm ker}\,\nabla\Phi({\bar{x}})^*$.
We know that the projection $P_E$ is a linear mapping satisfying $P_E(\nabla\Phi({\bar{x}}){{\mathbb R}^n}) = \{0\}$.
Since  ${x^k} - {\bar{x}} = O({\sigma_k})$ due to  \eqref{eb} and $u^k = o(\|w^k\|)$, we have
\begin{equation*}
    P_E\Big(\frac{\nabla\Phi({x^k})}{{\sigma_k}}\frac{u^k}{\|w^k\|}\Big) = P_E\Big(\frac{\nabla\Phi({x^k})-\nabla\Phi({\bar{x}})}{{\sigma_k}}\frac{u^k}{\|w^k\|}\Big) \to 0\quad\textrm{as }\; k\to\infty,
\end{equation*}
This, combined with the inclusion in \eqref{eq:non2}, gives us $-P_E(\eta) \in P_E(K)$.
We can then select $\hat\eta\in K$ with $P_E(\hat\eta) = P_E(\eta) = \eta$, since $\eta \in E$. On the other hand, it follows from $w^k\in K^\perp$ that $\eta\in K^\perp$. Thus, we obtain 
\begin{equation*}
    0 = \langle \eta, \hat\eta\rangle =  \langle \eta, P_E(\hat\eta)+P_{E^\perp}(\hat\eta)\rangle = \langle \eta, P_E(\hat\eta)\rangle = \|\eta\|^2.
\end{equation*}
We then have $\eta = 0$, a contradiction. This proves our cliam that $w^k = O(\|u^k\|)$. 
Recalling the error bounds \eqref{eb}, we get ${\sigma_k}\to0$ as $k\to\infty$.
Passing to the limit in both relationships in \eqref{eq:non1} and taking into account that $\nabla\Phi({\bar{x}})^*K^\bot$ is a convex polyhedral set, we arrive at
\begin{equation*}
  \nabla\Phi({\bar{x}})\xi\in K,\qquad \nabla^2_{xx}L({\bar{x}}, \bar{y})\xi \in \nabla\Phi({\bar{x}})^*K^\bot=\nabla\Phi({\bar{x}})^*D(\partial g)(\bar{z},\bar{y})(\nabla\Phi({\bar{x}})\xi).
\end{equation*}
This along with $\xi\neq 0$ tells us  that $\bar{y}$ is a critical multiplier, a contradiction, and therefore completes the proof.

which contradicts $r>0$.  Thus the auxiliary system has no solution with $(x, y) =(x_t,y_t)$ for any $t\in (0,1/2)$. 

To justify the same conclusion for \eqref{ssqp-kkt}, we observe first that 
$$
\partial g(z)=
    \begin{cases}
    \{0\} & \mbox{if}\;\; z\le 0,\\      
    \{2z\} & \mbox{if}\;\; z>0.
    \end{cases}
$$
Thus, the inclusion $q\in\partial g(r)$ in \eqref{ssqp-kkt}, together with $q=s$  and $r=(1-t)s+t^2$, leads us to the same cases as those of the auxiliary system,  discussed above.  Therefore, the variational system \eqref{ssqp-kkt} has no solution with $(x,y)=(x_t,y_t)$ whenever $t\in(0,1/2)$.

<!-- proof-9: body_md -->
Assume by contradiction that the estimate \eqref{eb3} fails. 
Recall the equivalent form \eqref{eq:linear} of \eqref{au-KKT} when $g$ is a polyhedral function and $\bar{y}\in {\rm ri}\, \partial g(\Phi({\bar{x}}))$.
Then there exist sequences $({x^k}, y^k)\to ({\bar{x}}, \bar{y})$, with $\sigma_k:= \sigma({x^k}, y^k)>0$, and $(s^k, q^k)$ such that
\begin{equation}\label{s8.1}
\|s^k-{x^k}\|+\|q^k-y^k\|>k \sigma_k,
\end{equation}
where $(s^k, q^k)$ satisfies
\begin{equation}\label{s9.1}
    0=\nabla\varphi({x^k})+ {H_k}(s^k-{x^k})+\nabla\Phi({x^k})^*q^k, \quad z^k - \bar{z} \in K, \quad q^k - \bar{y} \in K^\bot,
\end{equation}
where ${H_k} := \nabla^2_{xx}L({x^k}, y^k)$, $z^k :
= \Phi({x^k})+\nabla\Phi({x^k})(s^k-{x^k})-\sigma_k(q^k-y^k)$, $\bar{z} := \Phi({\bar{x}})$, and $K := K_g(\bar{z}, \bar{y})$.
Set $\beta_k := \|s^k-{x^k}\|+\|q^k-y^k\|$ and $\hat y^k := P_{{\Lambda}({\bar{x}})}(y^k)$.
Repeating the arguments leading to \eqref{s12} and \eqref{s16} in the proof of Proposition~\ref{prop:eb}, which do not employ SOSC \eqref{sosc}, we get
\begin{equation}\label{s12.1}
0=\nabla^2_{xx}L({\bar{x}}, \bar{y})u+\nabla\Phi({\bar{x}})^*w,
\quad 
\nabla\Phi({\bar{x}})u\in K
\end{equation}
where $(u, w)\neq 0$ is defined similarly as in \eqref{s11}.
We now claim that $w\in K^\bot$.
Recall first that $\hat y^k, \bar{y}\in\partial g(\bar{z})$ and $\hat y^k \to \bar{y}$ as $k \to\infty$. The reduction relation in \eqref{rl2}, together with \eqref{pwfor2}, then yields $\hat y^k - \bar{y} \in K_g(\bar{z}, \bar{y})^* = K^\bot$. Combining this with the last inclusion in \eqref{s9.1} leads us to
\begin{equation*}
    \frac{q^k - y^k}{\beta_k} - \frac{\hat y^k - y^k}{\beta_k}= \frac{q^k - \bar{y}}{\beta_k} - \frac{\hat y^k - \bar{y}}{\beta_k}   \in  K^\bot.
\end{equation*}
Passing through the limit as $k\to\infty$ and recalling that $\|y^k - \hat y^k\| = o(\beta_k)$, as in \eqref{s10}, we arrive at $w \in K^\bot$, which proves our claim.
This, combined with \eqref{s12.1} and noncriticality of $\bar{y}$, implies that $u=0$. 
The first equality in \eqref{s12.1} then implies that $\nabla\Phi({\bar{x}})^*w=0$. 

Arguing similarly to the second half of the proof of Proposition~\ref{prop:eb}, we end up with $w =0$, which is not possible. This  completes the proof.

\label{prop:calm}
Let $({\bar{x}}, \bar{y})$ be a solution to the KKT system \eqref{vs} such that there exists a positive constant $\varepsilon$ for which the error bound estimates in \eqref{subr3} hold for all $(x,y)\in\mathbb{B}_\varepsilon({\bar{x}},\bar{y})$.

<!-- proof-11: body_md -->
Observing that $r(x, y) = 0$ for all $(x, y)\in  {\mathcal S}(0, 0)$, we   deduce from the first estimate in \eqref{subr3} that
\begin{equation}\label{calm1}
    {\mathcal S}(0, 0) \cap B_\varepsilon({\bar{x}}, \bar{y}) = \big(\{{\bar{x}}\}\times\Lambda({\bar{x}})\big) \cap B_\varepsilon({\bar{x}}, \bar{y}).
\end{equation}
Moreover, as discussed above, we can infer from \cite[Proposition~3.8]{s20} and the first estimate in \eqref{subr3} that there exists $ 0<\varepsilon'\leq \varepsilon$ such that 
\begin{equation}\label{calm2}
    {\mathcal S}(u, w) \cap \mathbb{B}_{\varepsilon'}({\bar{x}}, \bar{y}) \subset \big(\{{\bar{x}}\}\times\Lambda({\bar{x}})\big) + 2\tau_1^{-1}\|(u, w)\|\mathbb{B}
\end{equation}
for all $(u, w) \in \varepsilon'\mathbb{B}$, where the choice of the modulus on the right-hand side is due to the proof of the latter result.
Shrinking $\varepsilon'$ if necessary, we can assume that $\max\{\varepsilon', 2\varepsilon'\tau_1^{-1}\}\leq \varepsilon/2$.
It   follows from \eqref{calm1} and \eqref{calm2} that
\begin{align*}
    {\mathcal S}(u, w)\cap B_{\varepsilon'}({\bar{x}}, \bar{y}) &\subset \big(\big(\{{\bar{x}}\}\times\Lambda({\bar{x}})\big) + 2\tau_1^{-1}\|(u, w)\|\mathbb{B}\big)\cap\mathbb{B}_{\varepsilon/2}({\bar{x}}, \bar{y})\\
    &\subset \big(\{{\bar{x}}\}\times\Lambda({\bar{x}})\big)\cap \mathbb{B}_\varepsilon({\bar{x}}, \bar{y}) + 2\tau_1^{-1}\|(u, w)\|\mathbb{B}\\
    &= {\mathcal S}(0, 0)\cap \mathbb{B}_\varepsilon({\bar{x}}, \bar{y}) + 2\tau_1^{-1}\|(u, w)\|\mathbb{B}\\
    &\subset {\mathcal S}(0, 0) + 2\tau_1^{-1}\|(u, w)\|\mathbb{B},
\end{align*}
for all $(u, w) \in \varepsilon'\mathbb{B}$, which verifies the calmness of ${\mathcal S}$ at $(0, 0)$ for $({\bar{x}}, \bar{y})$ with some constant $\ell \in (0, 2/\tau_1]$.
The proof is then complete.

<!-- proposition-9: body_md -->
\label{prop:precision}
Let $({\bar{x}}, \bar{y})$ be a solution to the KKT system \eqref{vs} such that there exists a positive constant $\varepsilon$ for which the error bound estimates in \eqref{subr3} hold for all $(x,y)\in\mathbb{B}_\varepsilon({\bar{x}},\bar{y})$. 
Assume also that $\sigma$ satisfies the upper bound in \eqref{eb} for all $(x, y) \in \mathbb{B}_\varepsilon({\bar{x}},\bar{y})$.
Then, for the mappings $\Psi$ and ${\cal A}$ taken from \eqref{ALMge} and \eqref{A}, we have
\begin{equation}\label{pre}
    \|\Psi(x, y) - {\cal A}(x, y, \tilde{x}, \tilde{y})\| \leq o\big({\rm dist}((\tilde{x}, \tilde{y}), {\mathcal S}(0, 0))\big)
\end{equation}
for all $(\tilde{x}, \tilde{y}) \in \mathbb{B}_\varepsilon({\bar{x}}, \bar{y})$ and all $(x, y)$ satisfying 
\begin{equation}\label{pre2}
    \|x-\tilde{x}\| + \|y - \tilde{y}\|\leq \kappa \sigma(\tilde{x}, \tilde{y})
\end{equation}
for some $\kappa>0$.
<!-- end proposition-9 -->
<!-- proof-12: body_md -->
As shown in the proof of Proposition~\ref{prop:calm}, the error bound estimates in \eqref{subr3} imply the local representation of ${\mathcal S}(0, 0)$ in \eqref{calm1}.
This tells us that
\begin{equation*}
    {\rm dist}((x, y), {\mathcal S}(0, 0)) = \|x - {\bar{x}}\| + {\rm dist}(y, {\Lambda}({\bar{x}}))
\end{equation*}
for all $(x, y)$ close to $({\bar{x}}, \bar{y})$.
By the arguments at the end of Subsection~\ref{sect:fischer} and by the upper bound   in \eqref{eb}, we know that the estimate in \eqref{pre} holds for all $(\tilde{x}, \tilde{y})$ sufficiently close to $({\bar{x}}, \bar{y})$ and all $(x, y)$ satisfying \eqref{pre2}.
The proof is then complete.

Let $({\bar{x}}, \bar{y})$ be a solution to the KKT system \eqref{vs} such that the SOSC \eqref{sosc} holds at $({\bar{x}}, \bar{y})$ and  $\sigma$ satisfy both error bound estimates in \eqref{eb}.
