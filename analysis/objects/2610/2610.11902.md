<!-- definition-1: body_md -->
\label{defweightsequ}
$\mathbf{M}=(M_p)_p\in\mathbb{R}_{>0}^{\mathbb{N}}$ is called a \emph{weight sequence} if $M_0=1$ and
$$\lim_{p\rightarrow+\infty}(M_p)^{1/p}=+\infty.$$

<!-- lemma-1: body_md -->
\label{lemma1}
Let $\mathbf{M}=(M_p)_{p\in\mathbb{N}}$ be a log-convex weight sequence. Then
\begin{equation}\label{lemma1equ}
\omega_{\mathbf{M}}(t)=0,\;\text{for}\;t\in[0,\mu_1],\;\;\;\;\omega_{\mathbf{M}}(t)=\log\left(\frac{M_0t^p}{M_p}\right)\;\text{for}\;t\in[\mu_p,\mu_{p+1}],\;p\ge 1.
\end{equation}

<!-- remark-1: body_md -->
\label{normrem}
For any weight function $v:[0,+\infty)\rightarrow(0,+\infty)$ in the weighted entire setting there exists a normalized $v^n$ such that $v^n(t)=v(t)$ for all $t$ sufficiently large. Indeed, fix $t_0>1$ such that $v(t_0)<\min\{1,v(0)\}(<t_0)$ and such $t_0$ exists since $v(t)$ tends to $0$ as $t\rightarrow+\infty$. Then put $v^n(t):=1$ for $t\in[0,1]$, $v^n(t):=v(t)$ for all $t\ge t_0$ and, finally $v^n(t):=\frac{v(t_0)-1}{t_0-1}t+\frac{t_0-v(t_0)}{t_0-1}$ for all $t\in(1,t_0)$; i.e. affine linear interpolation. $v^n$ satisfies all required conditions since the slope $\frac{v(t_0)-1}{t_0-1}$ is negative.

Similarly, for $v_{\mathbb{D}}$ fix $t_0\in(0,1)$ such that $v_{\mathbb{D}}(t_0)<\min\{1,v_{\mathbb{D}}(0)\}$, put $v^n_{\mathbb{D}}(0):=1$, $v^n_{\mathbb{D}}(t):=v_{\mathbb{D}}(t)$ for all $t\ge t_0$ and $v_{\mathbb{D}}^n(t):=\frac{v_{\mathbb{D}}(t_0)-1}{t_0}t+1$ for all $t\in(0,t_0)$.
<!-- end remark-1 -->
<!-- lemma-2: body_md -->
\label{MG1equivlemma}
Let $\mathbf{M}$ be a log-convex weight sequence. Then the following are equivalent:
\begin{itemize}
\item[$(i)$] $\mathbf{M}\hyperlink{approx}{\approx}\mathbf{G}^1$ holds.

\item[$(ii)$] The sequence of quotients $\mu=(\mu_p)_p$ satisfies

0<\inf_{p\in\mathbb{N}_{>0}}\frac{\mu_p}{p}\le\sup_{p\in\mathbb{N}_{>0}}\frac{\mu_p}{p}<+\infty.

\item[$(iii)$] It holds that

\item[$(iv)$] It holds that

\item[$(v)$] It holds that
$$\exists\;C,C_1\ge 1\;\forall\;t\ge 0:\;\;\;\frac{1}{C_1}\exp(-Ct)\le v_{\mathbf{M}}(t)\le C_1\exp(-t/C).$$
\end{itemize}

<!-- theorem-2: body_md -->
\label{Diffopthm}
Let $\mathbf{M}$ be a log-convex weight sequence. Then the following are equivalent:
\begin{itemize}
	\item[$(i)$] The differentiation operator $D: H^{\infty}_{v_{\mathbf{M}}}(\mathbb{C})\rightarrow H^{\infty}_{v_{\mathbf{M}}}(\mathbb{C})$ is continuous.
	
	\item[$(ii)$] $-\log(v_{\mathbf{M}}(t))=O(t)$ as $t\rightarrow+\infty$.
	
	\item[$(iii$)] $\omega_{\mathbf{M}}(t)=O(t)$  as $t\rightarrow+\infty$.
	
	\item[$(iv)$] $\liminf_{p\rightarrow+\infty}\left(\frac{M_p}{p!}\right)^{1/p}>0$ (i.e. $\mathbf{G}^{1}\hyperlink{preceq}{\preceq}\mathbf{M}$).

    \item[$(v)$] $\liminf_{p\rightarrow+\infty}\frac{\mu_p}{p}>0$.
\end{itemize}

Moreover, the following are equivalent:

\begin{itemize}
	\item[$(i)'$] The differentiation operator $D: H^{\infty}_{v_{\mathbf{M}}}(\mathbb{C})\rightarrow H^{\infty}_{v_{\mathbf{M}}}(\mathbb{C})$ is compact.
	
	\item[$(ii)'$] $-\log(v_{\mathbf{M}}(t))=o(t)$ as $t\rightarrow+\infty$.
	
	\item[$(iii)'$] $\omega_{\mathbf{M}}(t)=o(t)$ as $t\rightarrow+\infty$.
	
	\item[$(iv)'$] $\lim_{p\rightarrow+\infty}\left(\frac{M_p}{p!}\right)^{1/p}=+\infty$ (i.e. $\mathbf{G}^{1}\hyperlink{mtriangle}{\vartriangleleft}\mathbf{M}$).

\item[$(v)'$]  $\lim_{p\rightarrow+\infty}\frac{\mu_p}{p}=+\infty$.
\end{itemize}

However, in view of Lemma \ref{normconvexrem} in any situation we can switch to a normalized weight and which does not effect the underlying spaces; the corresponding associated weight sequences are then related by \eqref{assocsequnormrelation}.

\label{Intopthm}
Let $\mathbf{M}$ be a log-convex weight sequence. Consider the following assertions:
	\begin{itemize}
		\item[$(i)$] The integration operator $J: H^{\infty}_{v_{\mathbf{M}}}(\mathbb{C})\rightarrow H^{\infty}_{v_{\mathbf{M}}}(\mathbb{C})$ is continuous.
		
		\item[$(ii)$] $\limsup_{t\rightarrow+\infty}v_{\mathbf{M}}(t)\int_0^t\frac{1}{v_{\mathbf{M}}(r)}dr<+\infty$.
		
		\item[$(iii$)] $\liminf_{t\rightarrow+\infty}v_{\mathbf{M}}(t)\left(\frac{1}{v_{\mathbf{M}}}\right)'(t)>0$.
		
		\item[$(iv)$] $\sup_{p\in\mathbb{N}_{>0}}\frac{\mu_p}{p}<+\infty$.

\item[$(v)$] $\sup_{p\in\mathbb{N}_{>0}}\left(\frac{M_p}{p!}\right)^{1/p}<+\infty$ (i.e. $\mathbf{M}\hyperlink{preceq}{\preceq}\mathbf{G}^{1}$).
	\end{itemize}
	
Moreover, consider the following assertions:

	\begin{itemize}
		\item[$(i)'$] The integration operator $J: H^{\infty}_{v_{\mathbf{M}}}(\mathbb{C})\rightarrow H^{\infty}_{v_{\mathbf{M}}}(\mathbb{C})$ is compact.
		
		\item[$(ii)'$] $\lim_{t\rightarrow+\infty}v_{\mathbf{M}}(t)\int_0^t\frac{1}{v_{\mathbf{M}}(r)}dr=0$.
		
		\item[$(iii)'$] $\lim_{t\rightarrow+\infty}v_{\mathbf{M}}(t)\left(\frac{1}{v_{\mathbf{M}}}\right)'(t)=+\infty$.
		
		\item[$(iv)'$] $\lim_{p\rightarrow+\infty}\frac{\mu_p}{p}=0$.

\item[$(v)'$] $\lim_{p\rightarrow+\infty}\left(\frac{M_p}{p!}\right)^{1/p}=0$ (i.e. $\mathbf{M}\hyperlink{mtriangle}{\vartriangleleft}\mathbf{G}^1$).
	\end{itemize}
Then $(i)\Leftrightarrow(ii)\Leftrightarrow(iii)\Leftrightarrow(iv)\Rightarrow(v)$ and $(i)'\Leftrightarrow(ii)'\Leftrightarrow(iii)'\Leftrightarrow(iv)'\Rightarrow(v)'$ is valid.

Moreover:
\begin{itemize}
\item[$(*)$] When $\mathbf{M}$ satisfies in addition \hyperlink{mg}{$(\operatorname{mg})$}, then the implications $(v)\Rightarrow(iv)$, $(v)'\Rightarrow(iv)'$ are valid and hence equivalence of all assertions.

\item[$(*)$] When in addition even $\left(\frac{M_p}{p!}\right)_{p\in\mathbb{N}}$ is log-convex, i.e. $\mathbf{M}$ is \emph{strongly log-convex,} then one has $\lim_{p\rightarrow+\infty}\left(\frac{M_p}{p!}\right)^{1/p}=\lim_{p\rightarrow+\infty}\frac{\mu_p}{p}\in(0,+\infty]$ (cf. \eqref{stronglc}). Consequently, on the one hand, this gives directly $(iv)\Leftrightarrow(v)$ and hence again equivalence of the first five assertions, but, on the other hand, both $(iv)'$ and $(v)'$ are violated.
\end{itemize}

\label{orliczpaperrem}
Motivated by the proof of $(iii)\Rightarrow(iv)$ in Theorem \ref{Intopthm} we see that the technical assumption $\mu_p<\mu_{p+1}$ for all $p\in\mathbb{N}_{>0}$ in \cite[Lemma 4.9]{orliczpaper} is superfluous: Indeed, the given first part of $(i)\Rightarrow(ii)$ on \cite[p. 938]{orliczpaper} implies $\mu_j<\lambda_{\lceil\frac{j}{c_1}\rceil}$ for all $j\ge j_{\epsilon_1}$ such that $\mu_j<\mu_{j+1}$ and w.l.o.g. $\mu_{j_{\epsilon_1}}<\mu_{j_{\epsilon_1}+1}$. If now $j\ge j_{\epsilon_1}$ is such that $\mu_j<\mu_{j+1}=\dots=\mu_{j+1+\ell}<\mu_{j+2+\ell}$ for some $\ell\in\mathbb{N}_{>0}$, then indeed the arguments yield $\Sigma_{\mathbf{L}}(t)<\lceil\frac{j}{c_1}\rceil$ and hence $t<\lambda_{\lceil\frac{j}{c_1}\rceil}$ for all $t\in[\mu_j,\mu_{j+1})$. Thus $\mu_{j+1}\le\lambda_{\lceil\frac{j}{c_1}\rceil}$ holds which implies by log-convexity of $\mathbf{L}$ that $\mu_{j+1+i}\le\lambda_{\lceil\frac{j+i}{c_1}\rceil}$ for all $i=0,\dots,\ell$. Summarizing, $\mu_j\le\lambda_{\lceil\frac{j}{c_1}\rceil}$ is valid for all $j\ge j_{\epsilon_1}$.

For the second part in this implication, when $\mu_j=\dots=\mu_{j+\ell}<\mu_{j+\ell+1}$, then one can choose $t=\mu_{j+\ell}$ and conclude again for all indices $j+i$, $i=0,\dots,\ell$, by the log-convexity of $\mathbf{L}$.

The rest of the proof of \cite[Lemma 4.9]{orliczpaper} does not require any changes.

\label{example1}
There exists a strongly log-convex weight sequence $\mathbf{M}$ such that \eqref{hypercyclthmequ} holds.

Let $\mathbf{b}=(b_j)_{j\in\mathbb{N}_{>0}}$ be a (fixed) strictly increasing sequence such that $b_j\in(0,1)$ for all $j\in\mathbb{N}_{>0}$ and let $\mathbf{a}:=(a_j)_{j\in\mathbb{N}_{>0}}$ be a strictly increasing sequence of integers with $a_1:=1$. Then define $\mathbf{M}$ by
\begin{equation}\label{Mdef}
\forall\;p\in\mathbb{N}_{>0}:\;\;\;M_p:=M_0\prod_{i=1}^p\mu_i,\hspace{15pt}M_0:=1,
\end{equation}
with
\begin{equation}\label{example1mudef}
\mu_p:=b_jp,\;\;\;a_j\le p\le a_{j+1}-1,\;j\in\mathbb{N}_{>0}.
\end{equation}
Obviously, $p\mapsto\frac{\mu_p}{p}$ is non-decreasing and $(0<)b_1\le\frac{\mu_p}{p}\le\sup_{j\in\mathbb{N}_{>0}}b_j\le 1$ which gives strong log-convexity and the equivalence $\mathbf{M}\hyperlink{approx}{\approx}\mathbf{G}^1$ as a necessary condition for \eqref{hypercyclthmequ} via Corollary \ref{hypercyclthmequlemmacor}. Moreover, by this equivalence it is immediate that $\mathbf{M}$ is also a log-convex weight sequence according to Definition \ref{defweightsequ}. Finally, for all $j\in\mathbb{N}_{>0}$ we obtain
$$\frac{M_{a_{j+1}-1}}{(a_{j+1}-1)!}=\prod_{i=1}^{a_{j+1}-1}\frac{\mu_i}{i}=\prod_{i=1}^jb_i^{a_{i+1}-a_i}\le\prod_{i=1}^jb_j^{a_{i+1}-a_i}=b_j^{a_{j+1}-a_1}=b_j^{a_{j+1}-1},$$
and when $\mathbf{a}$ is increasing sufficiently fast then, since $b_j\in(0,1)$ for all $j$, we can achieve that $b_j^{a_{j+1}-1}\le\frac{1}{j}$ for all $j\in\mathbb{N}_{>0}$. This gives \eqref{hypercyclthmequ} and note that, when modifying $\mathbf{a}$ accordingly, $\frac{1}{j}$ can be replaced by $\frac{1}{c_j}$, with $\mathbf{c}=(c_j)_{j\in\mathbb{N}_{>0}}$ denoting an arbitrary strictly increasing sequence tending to infinity; i.e. the rate of approaching $0$ (in \eqref{hypercyclthmequ}) is not effecting the validity of this statement.

\label{example2}

<!-- example-3: body_md -->
There exists a log-convex weight sequence $\mathbf{M}=(M_p)_{p\in\mathbb{N}}$ with \eqref{hypercyclthmequ} but such that
\begin{equation}\label{hypercyclthmequviol}
\sup_{p\in\mathbb{N}_{>0}}\frac{\mu_p}{p}=+\infty.
\end{equation}
Let $\mathbf{a}=(a_j)_{j\in\mathbb{N}_{>0}}$ be a strictly increasing sequence of integers with $a_1=1$ and let $\mathbf{b}=(b_j)_{j\in\mathbb{N}_{>0}}$ be an arbitrary strictly increasing sequence in $\mathbb{R}_{>0}$ such that $\lim_{j\rightarrow+\infty}b_j=+\infty$. Then introduce $\mathbf{M}$ again like in \eqref{Mdef} but with
$$\mu_p:=a_jb_j,\;\;\;\;\;\;a_j\le p\le a_{j+1}-1,\;j\in\mathbb{N}_{>0}.$$
It is immediate that $p\mapsto\mu_p$ is non-decreasing and $\lim_{p\rightarrow+\infty}\mu_p=+\infty$, hence $\mathbf{M}$ is a weight sequence according to Definition \ref{defweightsequ} and log-convex. Moreover,
$$\sup_{p\in\mathbb{N}_{>0}}\frac{\mu_p}{p}\ge \sup_{j\in\mathbb{N}_{>0}}\frac{\mu_{a_j}}{a_j}=\sup_{j\in\mathbb{N}_{>0}}b_j=+\infty$$
and so \eqref{hypercyclthmequviol} follows. However, if $\mathbf{a}$ is assumed to be growing fast enough, then also \eqref{hypercyclthmequ} holds: For this fix $\epsilon\in(0,1)$ and the auxiliary sequence $\mathbf{b}$, then when $a_j$ is given (start with $a_1:=1$) we can find $a'_j>a_j$ with $a'_j\le a_{j+1}-1$ such that $\frac{\mu_p}{p}=\frac{a_jb_j}{p}\le\epsilon$ for all $p>a'_j$. Thus, when choosing $a_{j+1}>a_j$ large enough one can ensure
\begin{equation}\label{hypercycliccounterex2equ}
\forall\;j\in\mathbb{N}_{>0}:\;\frac{\mu_{a_j}\cdots\mu_{a_{j+1}-1}}{a_j\cdots(a_{j+1}-1)}=\frac{\mu_{a_j}\cdots\mu_{a'_j}}{a_j\cdots a'_j}\frac{\mu_{a'_j+1}\cdots\mu_{a_{j+1}-1}}{(a'_j+1)\cdots(a_{j+1}-1)}\le\frac{(a_jb_j)^{a'_j-a_j+1}}{a_j\cdots a'_j}\epsilon^{a_{j+1}-a'_j-1}\le\frac{1}{2},
\end{equation}
which implies, by setting $C_j:=\frac{\mu_{a_j}\cdots\mu_{a_{j+1}-1}}{a_j\cdots(a_{j+1}-1)}=\frac{(a_jb_j)^{a_{j+1}-a_j}}{a_j\cdots(a_{j+1}-1)}$, that $\frac{M_{a_{k+1}-1}}{(a_{k+1}-1)!}=\prod_{j=1}^kC_j\le\frac{1}{2^k}$ for all $k\in\mathbb{N}_{>0}$. This proves \eqref{hypercyclthmequ} and concludes the construction. Note: Similarly as before, $\frac{1}{2}$ in \eqref{hypercycliccounterex2equ} can be replaced by any arbitrary strictly decreasing sequence $\mathbf{c}=(c_j)_{j\in\mathbb{N}_{>0}}$ such that $\lim_{j\rightarrow+\infty}c_j=0$ when modifying $\mathbf{a}$ accordingly. \label{hypercyclthmequrem1}
First, also in $(i)$ the necessity of $(f)$ can be shown by involving Theorem \ref{hypercyclthm34}. For this note that the sequence $\mathbf{N}$ defined via \eqref{Nauxdef} violates \eqref{hypercyclthmequ} since for all $p\in\mathbb{N}_{>0}$
$$\frac{N_p}{p!}=\frac{(p+1/2)^{1/2}}{e^{1/2}}\frac{(p+1/2)^p}{e^pp!}\ge\frac{p^{1/2}}{e^{1/2}}\frac{p^p}{e^pp!},$$
and again by \emph{Stirling's formula} it holds that $p\mapsto\frac{p^{1/2}}{e^{1/2}}\frac{p^p}{e^pp!}$ is approaching $\frac{1}{\sqrt{2\pi e}}$. Now, when $\mathbf{M}$ is given and satisfying \eqref{hypercyclcor36equ3}, then also \eqref{hypercyclthmequ} fails for $\mathbf{M}$.\vspace{6pt}

Second, concerning $(ii)$ in the above theorem in view of \eqref{stirling} assertion \eqref{hypercyclcor36equ7} precisely means that $\mathbf{M}\hyperlink{preceq}{\preceq}\mathbf{G}^{1}$ has to be satisfied and hence \eqref{hypercyclcor36equ7} is obviously preserved under equivalence of weight sequences. On the other hand, for the implication $(d)\Rightarrow(e)$ the parameter $\alpha$ plays the crucial role here but relation \hyperlink{approx}{$\approx$} \emph{does not recognize this subtle behavior.} More precisely, fix $\alpha_0>0$ in \eqref{hypercyclcor36equ7}, then each sequence $\mathbf{M}^{\alpha}$ (given by $M^{\alpha}_p:=p^p(e\alpha)^{-p}$) with $\alpha\ge\alpha_0$ satisfies \eqref{hypercyclcor36equ7} whereas this condition fails for all $\mathbf{M}^{\alpha}$ with $\alpha\in(0,\alpha_0)$. On the other hand, recall that obviously $\mathbf{M}^{\alpha}\hyperlink{approx}{\approx}\mathbf{M}^{\beta}(\hyperlink{approx}{\approx}\mathbf{G}^1)$ holds for all $\alpha,\beta>0$.

<!-- remark-6: body_md -->
\label{omegaMvanishingrem}
By definition \eqref{weights} one has $-\log(v_{\mathbf{M}}(t))=\omega_{\mathbf{M}}(t)$ for all $t\ge 0$, but $\omega_{\mathbf{M}}(t)=0$ for all $t\in[0,\mu_1]$ which follows by Lemma \ref{lemma1} resp. by \eqref{assointrepr}. Therefore, whenever considering $\omega_{\mathbf{M}}(t)$ in the denominator of a fraction it is natural to investigate (all) $t>\mu_1$ which is not restricting the generality since for $t\mapsto\omega_{\mathbf{M}}(t)$ one is interested in having $t\rightarrow+\infty$ and thus the relevant information is ``localized at infinity.'' For example, see the proof of \cite[Prop. A.1 $(ii)$]{conjugateweightfunction} where we restricted to all integers $p\ge 2$ sufficiently large resp. to all $t>\mu_1$ and in the first part of \cite[Prop. A.1 $(i)$]{conjugateweightfunction} one should also consider $t>\mu_1$ but this issue does not effect the validity of this statement.

<!-- theorem-10: body_md -->
\label{Thm30&31}
Let $\mathbf{M}$ be a log-convex weight sequence and put $w_{\mathbf{M},\mathbb{D}}(t):=(1-t)v_{\mathbf{M},\mathbb{D}}(t)$, $t\in[0,1)$. Then the following hold:
\begin{itemize}
\item[$(i)$] The integration operator $J: H^{\infty}_{w_{\mathbf{M},\mathbb{D}}}(\mathbb{D})\rightarrow H^{\infty}_{v_{\mathbf{M},\mathbb{D}}}(\mathbb{D})$ is continuous.

\item[$(ii)$] The differentiation operator $D: H^{\infty}_{v_{\mathbf{M},\mathbb{D}}}(\mathbb{D})\rightarrow H^{\infty}_{w_{\mathbf{M},\mathbb{D}}}(\mathbb{D})$ is not continuous and surjective (i.e. $D$ is either not continuous or not surjective).

\item[$(iii)$] The differentiation operator $D: H^{\infty}_{v_{\mathbf{M},\mathbb{D}}}(\mathbb{D})\rightarrow H^{\infty}_{w_{\mathbf{M},\mathbb{D}}}(\mathbb{D})$ is not continuous.
\end{itemize}

<!-- lemma-10: body_md -->
\label{Thm30&31lemma}
Let $\mathbf{M}$ be a log-convex weight sequence, then
\begin{equation}\label{Thm30&31lemmaequ}
\forall\;k\in\mathbb{N}_{>0}:\;\;\;\lim_{n\rightarrow+\infty}\frac{v_{\mathbf{M},\mathbb{D}}(1-2^{-n-k})}{v_{\mathbf{M},\mathbb{D}}(1-2^{-n})}=0,
\end{equation}
and
\begin{equation}\label{Thm30&31lemmaequ2}
\forall\;\gamma>1:\;\;\;\lim_{t\rightarrow 1^{-}}\frac{v_{\mathbf{M},\mathbb{D}}(t^{\gamma})}{v_{\mathbf{M},\mathbb{D}}(t)}=+\infty.
\end{equation}
