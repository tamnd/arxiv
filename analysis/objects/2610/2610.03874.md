\label{conj_GK}
Let $M$ be as above, and $s\in\mathbb Z$. There exist universal functions
\begin{equation} \begin{aligned}\label{Cz Cz12}
Q_s, R_s, T_s, V_s, W_s, X_s  &\in \mathbb C[[z]]~, \\
S_s, S_{j,s}, Y_s, Y_{j,s}, Z_s,  Z_{jk,s} &\in \mathbb C[[z^{\frac{1}{2}}]]~,
\end{aligned} \end{equation}
for all $1\leq j\leq k\leq \rho-1$, such that for any $\alpha\in K(S)$ with $\text{rk}(\alpha)=s$ we have 
{\normalfont
\begin{equation} \begin{aligned}\label{GK conjecture}
\mathscr S_\alpha[p',x']=\mathscr U_\alpha[p',x'] \Big|_{z^{\frac12\text{vd}\, M}}~,
\end{aligned} \end{equation}
}
where 
{\normalfont
\begin{equation} \begin{aligned}\label{universal series}
\mathscr U_\alpha[p',x']& \coloneqq \rho^{2-\chi_{\rm h}+K^2}V_s^{c_2(\alpha)}W_s^{c_1(\alpha)^2}X_s^{\chi_{\rm h}} Y_s^{c_1(\alpha)K}Z_s^{K^2}e^{x'^2 Q_s + c_1(\alpha)x' R_s + x'K S_{s}  +  p'\, T_s} \\
&\quad \times \sum_{(a_1,\dots, a_{\rho-1})}\prod_{j=1}^{\rho-1}\varepsilon_\rho^{j a_j c_1}\mathcal{SW}(a_j) Y_{j,s}^{c_1(\alpha)a_j}e^{a_j x' S_{j,s}} \!\!\!\!\!\prod_{1\leq j\leq k\leq \rho-1}Z_{jk,s}^{a_ja_k}~,
\end{aligned} \end{equation}
}
and 
\begin{equation} \begin{aligned}\label{changeofvar}
z = t (1+(1-\tfrac{s}{\rho}) t)^{1-\frac{s}{\rho}}~.
\end{aligned} \end{equation}
Here, 
\begin{equation} \begin{aligned}\label{Vs Ws Xs Qs Rs Ts} 
V_s(t) &= (1+(1-\tfrac{s}{\rho})t)^{1-s} (1+(2-\tfrac{s}{\rho})t)^s (1+(1-\tfrac{s}{\rho})t)^{\rho-1} , \\ 
W_s(t) &= (1+(1-\tfrac{s}{\rho})t)^{\frac{1}{2}s-1} (1+(2-\tfrac{s}{\rho})t)^{\frac{1}{2}(1-s)} (1+(1-\tfrac{s}{\rho})t)^{\frac{1}{2} - \frac{1}{2} \rho}, \\
X_s(t) &=  (1+(1-\tfrac{s}{\rho})t)^{\frac{1}{2} s^2-s} (1+(2-\tfrac{s}{\rho})t)^{-\frac{1}{2}s^2+\frac{1}{2}} (1+(1-\tfrac{s}{\rho})(2-\tfrac{s}{\rho})t)^{-\frac{1}{2}} (1+(1-\tfrac{s}{\rho})t)^{-\frac{(\rho-1)^2}{2\rho} s}, \\
Q_s(t) &= \tfrac{1}{2}t(1+(1-\tfrac{s}{\rho}) t), \quad R_s(t) = t, \quad T_s(t) = \rho t (1+ \tfrac{1}{2}(1-\tfrac{s}{\rho})(2-\tfrac{s}{\rho})t),
\end{aligned} \end{equation}
and moreover $Y_s$, $Y_{j,s}$,  $Z_s$, $Z_{jk,s}$, $S_s$, $S_{j,s}$ are all algebraic functions.

\label{conj:DZSW0=Ua}
Let $X$ be a smooth, oriented, compact and simply-connected four-manifold of Seiberg–Witten simple type, with $b_2^+ > 1$. Let $Z_{\mathrm{SW},0}$ be the contribution of the $\mathcal{N}=1$ singularity $\vec{u}=\vec{\mathtt u}^{(0)}_{N_f}$ to the topological partition function of $\mathcal{N} = 2$ $\mathrm{SU}(N)$ SQCD with $1\leq N_f <2N$ hypermultiplets with large equal mass $m$ on $X$, and $\mathscr D$ the decoupling factor~\eqref{eq:decouping_factor}. \begin{equation}\label{eq:ZSW0=universal}
\mathscr D Z_{\mathrm{SW},0} [p, x] = \frac{1}{N}\mathscr{U}_\alpha [p', x'] .
\end{equation}

\begin{equation}\label{z in terms of m}

\begin{eqsp}\label{change of variables}

\label{thm:discriminant}
Define the binomial coefficients
\begin{equation} \begin{aligned}
    r=\binom{2g+1}{g+1}~,\qquad n=\binom{2g}{g+1}~.
\end{aligned} \end{equation}
There exists a Siegel modular form $\varphi_g$ of weight $4r$ on $\Gamma_g(2)$, constructed from the hyperelliptic even theta constants, such that
\begin{equation} \begin{aligned}\label{theorem identity}
    \pi^{-4gr}(\det A)^{4r}\Delta^n=\varphi_g(\Omega)~.
\end{aligned} \end{equation}

<!-- theorem-2: body_md -->
The following is true:
\begin{enumerate}
\item $\kappa\left(\mathfrak{g}_\alpha, \mathfrak{g}_\beta\right)=0$ if $\alpha, \beta \in \mathfrak{h}^*$ with $\alpha+\beta \neq 0$. In particular $\kappa\left(\mathfrak{h}, \mathfrak{g}_\alpha\right)=0$ for all $\alpha \in \Phi$. \item  $\Phi$ spans $\mathfrak{h}^*$.
\item If $\alpha \in \Phi$ then $-\alpha \in \Phi$.
\item If $\alpha \in \Phi$, then for $x \in \mathfrak{g}_\alpha, y \in \mathfrak{g}_{-\alpha}$, we have $[x, y]=\kappa(x, y) t_\alpha$. In particular $\left[\mathfrak{g}_\alpha, \mathfrak{g}_{-\alpha}\right]$ is one-dimensional with basis $t_\alpha$.
\item $\alpha\left(t_\alpha\right)=\kappa\left(t_\alpha, t_\alpha\right) \neq 0$ for $\alpha \in \Phi$.
\item If $\alpha \in \Phi$ and $x_\alpha \in \mathfrak{g}_\alpha \backslash\{0\}$, then there exists $y_\alpha \in \mathfrak{g}_{-\alpha}$ such that the subalgebra spanned by $x_\alpha, y_\alpha$ and $z_\alpha\coloneqq \left[x_\alpha, y_\alpha\right]$ is isomorphic to $\mathfrak{su}(2)$ under the map
\begin{equation}
x_\alpha \mapsto\left(\begin{array}{ll}
0 & 1 \\
0 & 0
\end{array}\right), \quad y_\alpha \mapsto\left(\begin{array}{ll}
0 & 0 \\
1 & 0
\end{array}\right), \quad z_\alpha \mapsto\left(\begin{array}{cc}
1 & 0 \\
0 & -1
\end{array}\right) \text {. }
\end{equation}
Moreover
\begin{equation}
z_\alpha=\frac{2 t_\alpha}{\kappa\left(t_\alpha, t_\alpha\right)}, \quad z_\alpha=-z_{-\alpha} .
\end{equation}
\item $\operatorname{dim} \mathfrak{g}_\alpha=1$ for $\alpha \in \Phi$.
\end{enumerate}
