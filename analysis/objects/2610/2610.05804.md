<!-- proposition-1: body_md -->
\label{prop:leadingOrder}
 Consider a super-critical neutral multi-type branching diffusion with exponentially scaled generator Eq.~(\ref{genOfZ}) where $\alpha > 0$ and $\gamma$ 
 is parametrised by Eq.~(\ref{gammaParam}).  Then as $\theta \to 0$ the solution to the forward equation with initial condition $\mathbf{Z}(0) = \mathbf{x}(0)$ is 
\begin{equation}	\label{approxFZ}
f(t, \mathbf{z}; \mathbf{x}(0)) = \begin{cases}
\prod_{i = 1}^d f_{\text{1-allelle}}\left( t, z_i; x_i(0) \right) + \mathcal{O}(\theta), & t < t_{\rm c}, \\ \\
\det \left(e^{-\gamma(t - t_{\rm c})} \right) \prod_{i = 1}^d f_{\text{1-allelle}}\left( \infty, (\mathbf{z}^{\rm T} e^{-\gamma(t - t_{\rm c})})_i; x_i(0) \right) 
											+ \mathcal{O}(\theta), & t > t_{\rm c}, \\
\end{cases}
\end{equation}
where 
\begin{equation}	\label{tcDef}
t_{\rm c} = - \alpha^{-1} \log\theta.  
\end{equation}
<!-- end proposition-1 -->
<!-- head-1: body_md -->
A parent-independent rate matrix is one whose off-diagonal elements are of the form $\gamma_{ij} = \gamma_j$, independent of $i$.  In Eq.~(\ref{gammaParam}) set 
 $\theta = 2\sum_{j = 1}^d \gamma_j$ and $p_i = \gamma_i/\sum_{j = 1}^d \gamma_j$ for $i = 1, \ldots, d$. Then 
\[
Q_{ij} = p_j  - \delta_{ij}, \quad i, j = 1, \ldots, d,   
\]
where $p_j \ge 0$ and $\sum_{j = 1}^d p_j = 1$, and hence $Q^2 = -Q$, leading to   
\[
e^{\gamma t} = e^{\frac{1}{2}\theta t Q} = I + (1 - e^{-\frac{1}{2}\theta t}) Q.  
\]
In the second phase of Eq.~(\ref{approxFZ}), 
\[
\det \left(e^{-\gamma(t - t_{\rm c})} \right) = \exp{\rm tr}\left(-\gamma(t - t_{\rm c})\right) = e^{\frac{1}{2}(d - 1)\theta (t - t_{\rm c})},
\]
and 
\[
\left(\mathbf{z}^{\rm T} e^{-\gamma(t - t_{\rm c})}\right)_i = |\mathbf z|p_i + \left(z_i - |\mathbf z|p_i\right)  e^{\frac{1}{2}(d - 1)\theta (t - t_{\rm c})},
\]
where $|\mathbf z| = \sum_{l = 1}^d z_l$.

<!-- lemma-1: body_md -->
\label{lemma:zetaPerturb}
The perturbative solution to Eq.~(\ref{zetapde}) with the above initial condition and $\gamma_{ij}$ as in Eq.~(\ref{gammaParam}) is 
\begin{eqnarray}	\label{zetaOfTPhi}
\zeta(t, \bm{\phi}) = \zeta^{(0)}(t, \bm{\phi}) \left(1 - \theta \sum_{j = 1}^d x_j(0) \xi_j^{(1)}(t, \bm{\phi}) + \mathcal{O}(\theta^2) \right), \quad \text{as }\theta \to 0, 
\end{eqnarray}
where 
\[
\zeta^{(0)}(t, \bm{\phi}) = \prod_{j = 1}^d  \zeta_{\text{1-allele}}(t, \phi_j; x_j(0)),  
\]
with $\zeta_{\text{1-allele}}$ defined in Eq.~(\ref{zeta1Allele}), 
\begin{eqnarray}	\label{xi1Of_tphi}
\xi_i^{(1)}(t, \bm{\phi}) 
	& = & \frac{\left(\frac{\phi_i}{2\alpha}\right)^2}{\left(1 + \frac{\phi_i}{\mu(t)}\right)^2} \sum_{j = 1}^d Q_{ij} \frac{1}{\frac{2\alpha}{\phi_j} - e^{-\alpha t}} \times \nonumber \\ 
	& & \qquad \left\{ \left(\frac{2\alpha}{\phi_i} - e^{-\alpha t} \right)^2 \alpha t - \left(  \frac{2\alpha}{\phi_i} 
							- \frac{2\alpha}{\phi_j} \right)^2 \log \left(1 + \frac{\phi_j}{\mu(t)}\right) \right\},  
\end{eqnarray}
and $\mu(t)$ defined by Eq.~(\ref{muBetaDef}).

<!-- theorem-1: body_md -->
\label{thm:InverseZeta}
The inverse LT of $\zeta^{\text{1-ancestor}}(t, \bm{\phi})$ is 
\begin{equation}	\label{Z1ancestorDensity}
f^{\text{1-ancestor}}(t, \mathbf{z}) = \mu(t)^d F(\mu(t) \mathbf{z}), 
\end{equation}
where the function $F(\mathbf{u})$ for $\mathbf{u} \in \mathbb{R}_{\ge 0}^d$ takes the form of a sum over line and surface densities 
\begin{eqnarray}	\label{fU}
\lefteqn{F(\mathbf{u}) = F_0(u_i) \prod_{l \ne i} \delta(u_l) + \theta\left(F^{\text{line-par}}(u_i) \prod_{l \ne i} \delta(u_l) 
							+ \sum_{j \ne i} F_j^\text{line-perp}(u_j) \prod_{l \ne j} \delta(u_l) \right.}  \nonumber \\
	& & \qquad\qquad\qquad\qquad\qquad\qquad \left. + \sum_{j \ne i} F_j^\text{surface}(u_i, u_j) \prod_{l \ne i,j} \delta(u_l)\right) + \mathcal{O}(\theta^2),
\end{eqnarray}
as $\theta \to 0$, with
\begin{align*}
F_0(u_i) &= e^{-u_i}  - \beta(t) u_i e^{-u_i} \sum_{j \ne i} \frac{Q_{ij}}{\lambda_{ij}(u_i)}, \nonumber \\ 
F^{\text{line-par}}(u_i) & = \frac{1}{2\alpha} Q_{ii} \left\{ \alpha t (1 - u_i e^{\alpha t}) + (1 - e^{\alpha t})\gamma_{\rm EM} u_i \right\} e^{-u_i}, \nonumber \\
F_j^{\text{line-perp}}(u_j) &= \frac{1}{2\alpha} Q_{ij} \left\{E_1(u_j) - E_1(u_j e^{\alpha t}) e^{ (e^{\alpha t} - 1)u_j}\right\}, \nonumber \\ 
F_j^\text{surface}(u_i, u_j) &= \frac{1}{2\alpha} Q_{ij} \left\{(e^{\alpha t} - 1) u_j^{\lambda_{ij}(u_i)\theta -1} e^{-u_j} u_i \phantom{u_j^{\lambda_{ij}}} \right. \nonumber \\
					& \qquad\quad \left.\phantom{u_j^{\lambda_{ij}}} -e^{\alpha t}(u_i e^{\alpha t} - 2)E_1(u_j e^{\alpha t}) e^{(e^{\alpha t} - 1)u_j} 
										+ (u_i - 2)E_1(u_j) \right\}  e^{-u_i}, \nonumber 
\end{align*}  
$\beta(t)$ is defined by Eq.~(\ref{muBetaDef}), $\gamma_{\rm EM}$ is the Euler-Mascheroni constant, $E_1$ is the exponential integral 
\citep[Eq.~(5.1.1)]{Abramowitz:1965sf}, and  $\lambda_{ij}(u_i) > 0$ for $j \ne i$ are undetermined finite, analytic functions on $\mathbb{R}_{>0}$.

<!-- proof-1: body_md -->
Writing LT$_i$ and LT$_j$ for LTs with respect to $u_i$ and $u_j$ respectively and using LTs from Table~\ref{LaplaceTransforms}, 
\begin{eqnarray*}
\lefteqn{\text{LT}(\text{rhs of Eq.~(\ref{LT_T3Part1})})} \nonumber \\ 
	& = & \text{LT}_i\left(u_ie^{-u_i}\,  \text{LT}_j \left(\left(\frac{1}{\lambda_{ij}(u_i) \theta} - \gamma_{\rm EM} \right)\delta(u_j) 
									- u_j^{\lambda_{ij}(u_i)\theta -1} e^{-u_j}\right) \right) \nonumber \\
	& = & \text{LT}_i\left(u_ie^{-u_i} \left( \log\left(1 + s_j \right) + \mathcal{O}(\theta)\right)\right) \nonumber \\
	& = & \text{LT}_i\left(u_ie^{-u_i}\right) \log\left(1 + s_j \right) + \mathcal{O}(\theta) \nonumber \\
	& = & \frac{1}{(1 + s_i)^2} \log\left(1 + s_j\right) + \mathcal{O}(\theta), \qquad \text{as }\theta \to 0.  
\end{eqnarray*}
