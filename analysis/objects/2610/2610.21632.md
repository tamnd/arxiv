<!-- introdefn-1: body_md -->
\label{defn:escaping quasi-local coupling}
Let $(X,d)$ be a discrete metric space of bounded geometry. An \emph{escaping quasi-local coupling} on $X$ is a sequence of symmetric finitely supported probability measures $\{\nu_n\}_n$ on $X\times X$ satisfying:
\begin{description}[
  font=\normalfont\bfseries,
  leftmargin=3em,
  labelsep=0.5em,
  style=multiline
]
\item[\textbf{(E)}] For any $S>0$, $\nu_n\bigl(\{(x,y)\in X^2:d(x,y)\le S\}\bigr)\to 0$;
\item[\textbf{(Q)}] For any $\varepsilon>0$, there is $R>0$ such that $\nu_n(A\times B)<\varepsilon$ for any $n$ and $A,B \subseteq X$ with $d(A,B)>R$.
\end{description}
<!-- end introdefn-1 -->
<!-- introthm-1: body_md -->
\label{thm:esc and ql coupling}
A discrete metric space $(X,d)$ of bounded geometry can be coarsely embedded into Hilbert space \emph{if and only if} $X$ admits no escaping quasi-local coupling.
<!-- end introthm-1 -->
<!-- introthm-2: body_md -->
\label{introcor:ghost proj are compact}
Let $(X,d)$ be a sparse discrete metric space of bounded geometry which is coarsely embeddable into Hilbert space. Then for every ghost projection \(P\in\mathfrak{B}(\ell^2(X))\), 
\[
\operatorname{dist}\big(P,C_{uq}^*(X)) = \operatorname{dist}\big(P,C_u^*(X)\big)
=
\begin{cases}
0,& \text{if }P\text{ is compact};\\
\frac12,&\text{if }P\text{ is non-compact}.
\end{cases}
\]
In particular, any ghost projection in $C^*_{uq}(X)$ is compact.
<!-- end introthm-2 -->
<!-- introthm-3: body_md -->
\label{thm:char by multiplier}
Let $(X,d)$ be a discrete metric space of bounded geometry. Then the following are equivalent:
\begin{enumerate}
\item The space $X$ can be coarsely embedded into some Hilbert space;
\item There exists a sequence $\{\varphi_n\}_{n\in\mathbb N}\subseteq \mathscr{P}_0(X)$ such that $\|M_{\varphi_n}(T)-T\| \to 0$ as $n\to \infty$, for any $T \in C^*_{uq}(X)$;
\item Any positive $T\in C^*_u(X)$ with $M_\varphi(T)\in \mathfrak{K}(\ell^2(X))$ for any $\varphi\in\mathscr{P}_0(X)$, is compact;
\item Any positive $T\in C^*_{uq}(X)$ with $M_\varphi(T)\in \mathfrak{K}(\ell^2(X))$ for any $\varphi\in\mathscr{P}_0(X)$, is compact;
\item $J_0(X)=\mathfrak{K}(\ell^2(X))$.
\end{enumerate}
<!-- end introthm-3 -->
<!-- introdefn-2: body_md -->
\label{defn:ONSL}
A discrete metric space $(X,d)$ of bounded geometry is said to have the \emph{operator norm state localisation property (ONSL)} if for any $R>0$ and $\varepsilon \in (0,1)$, there exists a non-increasing function $\omega:[0,\infty)\to [0,1]$ with $\lim_{s\to\infty}\omega(s)=0$ such that for any positive operator $T\in \mathfrak{B}(\ell^2(X))$ with $\mathrm{prop}(T)\le R$, there exists a positive trace-class operator $D$ on $\ell^2(X)$ with $\mathrm{Tr}(D)=1$ satisfying $\mathrm{Tr}(TD)\ge(1-\varepsilon)\|T\|$ and $|D_{xy}|\le \omega(d(x,y))\sqrt{D_{xx}D_{yy}}$ for any $x,y\in X$.
<!-- end introdefn-2 -->
<!-- introthm-4: body_md -->
\label{thm:CE and ONL}
Let $(X,d)$ be a discrete metric space of bounded geometry. Then $X$ can be coarsely embedded into Hilbert space \emph{if and only if} $X$ satisfies ONSL.
<!-- end introthm-4 -->
<!-- defn-1: body_md -->
\label{defn:ppg and quasi-local}
Let $(X,d)$ be a discrete metric space, $\mathcal{H}_0$ be a Hilbert space and $T \in \mathfrak{B}(\ell^2(X;\mathcal{H}_0))$.
\begin{enumerate}
  \item The \emph{propagation} of $T$ is defined to be $\mathrm{prop}(T):=\sup\{d(x,y): T_{xy} \neq 0\}$.
  We say that $T$ has \emph{finite propagation} if $\mathrm{prop}(T) < \infty$.
  \item We say that $T$ is \emph{quasi-local} if for any $\varepsilon>0$, there exists $R>0$ such that for any $A,B \subseteq X$ with $d(A,B) > R$, we have $\|\chi_A T \chi_B\| < \varepsilon$. 
                                                                                                                                                                               
  \item We say that $T$ is \emph{locally compact} if \(\chi_FT\) and \(T\chi_F\) are compact for every finite subset \(F\subseteq X\).
                                                  
\end{enumerate}

\label{defn:Roe and quasi-local algebra}
Let $(X,d)$ be a discrete metric space of bounded geometry and $\mathcal{H}_0$ a Hilbert space.
\begin{enumerate}
  \item The \emph{Roe algebra} of $X$, denoted by $C^*(X;\mathcal{H}_0)$, is the norm closure of all finite propagation and locally compact operators in $\mathfrak{B}(\ell^2(X;\mathcal{H}_0))$.
                                                                                                                                                                                                                                                                                                                    
  \item The \emph{quasi-local algebra} of $X$, denoted by $C^*_q(X;\mathcal{H}_0)$, is the set of all quasi-local and locally compact operators in $\mathfrak{B}(\ell^2(X;\mathcal{H}_0))$.
\end{enumerate}

\label{defn:ghost}
Let $(X,d)$ be a discrete metric space of bounded geometry and $\mathcal{H}_0$ a Hilbert space. Say that $T \in \mathfrak{B}(\ell^2(X;\mathcal{H}_0))$ is a \emph{ghost} operator if for any $\varepsilon>0$, there exists a finite subset $F \subseteq X$ such that $\|T_{xy}\| < \varepsilon$ for any $(x,y) \in (X \times X) \setminus (F \times F)$.

<!-- defn-4: body_md -->
\label{defn:generalised expanders}
Given $k,r > 0$, a finite metric space $(X,d)$ is called a \emph{generalised $(k,r)$-expander} if there is a symmetric probability measure $\mu$ on $X\times X$ such that $\mathrm{supp}(\mu) \subseteq \{(x,y) \in X^2: d(x,y)\geq r\}$ and 
\begin{equation}\label{EQ:generalised expander}
 \sum_{x,y \in X}\|F(x)-F(y)\|^2 \mu(x,y) \leq k^2
\end{equation}
for any Hilbert-space-valued map $F:X \to \mathcal{H}$ satisfying
\[
\|F(x) - F(y)\|_{\mathcal{H}} \leq d(x,y) \quad \text{whenever} \quad d(x,y) \geq 1. 
\]
                                                                    

A sequence of finite metric spaces $\{(X_n,d_n)\}_n$ is called a sequence of \emph{generalised expanders} if there exist $k>0$ and a sequence $r_n \to \infty$ such that each $X_n$ is a generalised $(k,r_n)$-expander.

\label{prop:tessera-general}
A metric space $X$ does not coarsely embed into Hilbert space \emph{if and only if} there exists a sequence of finite subsets $\{Y_n\}_n$ of $X$ which forms a sequence of generalised expanders.

\label{prop:sparse-reduction}
Let $(X,d)$ be a locally finite metric space. Then $X$ does not coarsely embed into Hilbert space \emph{if and only if} there exists a sequence of disjoint finite subsets $\{Y_n\}_n$ of $X$ with $d(Y_n, Y_m) \to \infty$ as $n+m \to \infty$ and $n \neq m$ such that $\{Y_n\}_n$ is a sequence of generalised expanders.
<!-- end prop-2 -->
<!-- proof-1: body_md -->
We only need to prove necessity. Assume that $X$ does not coarsely embed into Hilbert space. 
By Proposition~\ref{prop:tessera-general}, there exist $k>0$, $r_j\to\infty$, finite subsets $Z_j\subseteq X$ and symmetric probability measures $\mu_j$ on $Z_j\times Z_j$ with support contained in $\{(x,y)\in Z_j^2:d(x,y)\geq r_j\}$ such that 
\begin{equation}\label{eq:sparse-source-energy}
 \sum_{x,y\in Z_j}\|G(x)-G(y)\|^2\mu_j(x,y)\le k^2,
\end{equation}
for any Hilbert-space-valued map $G:Z_j\to \mathcal{H}$ satisfying $\|G(x) - G(y)\| \leq d(x,y)$ whenever $d(x,y) \geq 1$.
Extend each $\mu_j$ by zero to $X\times X$, still denoted by $\mu_j$. 

First, claim that $\mu_j(A\times X)\to 0$ for every finite subset $A\subseteq X$.
Indeed, given $\varepsilon>0$, choose $R>0$ so that $k^2/R^2<\varepsilon$. Consider the $1$-Lipschitz function
\[
 f(z):=\min\{d(z,A),R\} \quad \text{for} \quad z\in X.
\]
Since $f\equiv 0$ on $A$ and $f \equiv R$ on $X\setminus \mathcal{N}_R(A)$, \eqref{eq:sparse-source-energy} gives
\[
 \mu_j\bigl(A\times(X\setminus \mathcal{N}_R(A))\bigr)\le \frac{k^2}{R^2}<\varepsilon.
\]
On the other hand, for any $(x,y) \in A\times \mathcal{N}_R(A)$ we have $d(x,y)\leq \mathrm{diam}(A)+R$. Hence, for sufficiently large $j$, the support condition of $\mu_j$ gives $\mu_j(A\times \mathcal{N}_R(A))=0$. Combining these estimates, we prove the claim.

We now construct $Y_n$ recursively. Choose $j_1$ with $r_{j_1} \geq 1$, and set $Y_1 = Z_{j_1}$ and $\nu_1:=\mu_{j_1}$. Suppose that $Y_1,\ldots,Y_{n-1}$ have been chosen for some $n \geq 2$, and we set $B_n:=\mathcal{N}_n(Y_1\cup\cdots\cup Y_{n-1})$, which is finite. Since $X$ is infinite, choose $o_n\in X\setminus B_n$ and define
\[
 L_n:=1+ \max_{b\in B_n} d(o_n,b)>0 \quad \text{and} \quad A_n:=\mathcal{N}_{L_n}(B_n).
\]
By the claim above, we choose $j_n \geq n$ sufficiently large such that $\mu_{j_n}(A_n\times X)<1/4$. Hence, the symmetry of $\mu_{j_n}$ gives
\[
 q_n:=\mu_{j_n}\bigl((X\setminus A_n)^2\bigr)
 \ge 1-2\mu_{j_n}(A_n\times X)>\frac12.
\]
Now set
\begin{equation}\label{eq:sparse-buffer-choice}
 Y_n:=(Z_{j_n}\setminus B_n)\cup\{o_n\}
 \quad \text{and} \quad 
 \nu_n:=q_n^{-1}\mu_{j_n}|_{(X\setminus A_n)^2}.
\end{equation}
Then $Y_n$ is finite, and we regard $\nu_n$ as a symmetric probability measure on $Y_n\times Y_n$ with support contained in $\{(x,y)\in Y_n^2:d(x,y)\ge r_{j_n}\}$. 

Now we verify the Poincar\'e inequality. Let $F:Y_n\to \mathcal{H}$ be a Hilbert-space-valued map satisfying $\|F(x) - F(y)\| \leq d(x,y)$ whenever $d(x,y) \geq 1$. After translation, we assume $F(o_n)=0$. Set $t_x:=d(x,B_n)$ and $\theta_x:=\min\left\{1,\frac{t_x}{L_n}\right\}$ for $x\in Y_n$, and define a map $G:Z_{j_n}\to \mathcal{H}$ by
\[
 G(x):=
 \begin{cases}
  0,& \text{if } x\in Z_{j_n}\cap B_n, \text{ or }x\in Z_{j_n}\setminus B_n \text{ with } d(x,o_n) < 1;\\
  \theta_x \cdot F(x),& \text{if }x\in Z_{j_n}\setminus B_n \text{ with } d(x,o_n) \geq 1.
 \end{cases}
\]
We claim that $\|G(x) - G(y)\| \leq 3d(x,y)$ whenever $d(x,y) \geq 1$. To see this, fix $x,y\in Z_{j_n}$ with $d(x,y) \geq 1$. Note that when $x\in Z_{j_n}\setminus B_n$ and \(d(x,o_n)\ge1\), we have
\begin{equation}\label{eq:sparse-anchor-bound}
\|F(x)\| =\|F(x)-F(o_n)\| \le d(x,o_n) \le t_x+L_n.
\end{equation}
\textbf{Case I.} $x,y\in Z_{j_n}\setminus B_n$. If both \(G(x)\) and \(G(y)\) are zero, the estimate is trivial. Assume that \(G(x)\neq0\) and \(G(y)\neq0\). Without loss of generality, assume $t_x\le t_y$, so $\theta_x\le\theta_y$. Then we have
\[
\|G(x)-G(y)\| \leq \theta_y\|F(x)-F(y)\| +(\theta_y-\theta_x)\|F(x)\|.
\]
If $F(x)=0$, then $\|G(x)-G(y)\| \leq \theta_y\|F(x)-F(y)\|\leq \|F(x)-F(y)\|$ and the claim holds. Now we only need to consider the case when $F(x)\neq 0$.
If $\theta_x=\theta_y$, the right hand side is at most $d(x,y)$. If $\theta_x<\theta_y$, then $t_x<L_n$ and hence, \eqref{eq:sparse-anchor-bound} shows $\|F(x)\|\le2L_n$ and
$|\theta_x-\theta_y|\le d(x,y)/L_n$. Consequently,
\[
 \|G(x)-G(y)\|
 \le d(x,y)+\frac{d(x,y)}{L_n}\cdot 2L_n =3d(x,y).
\]
If exactly one of \(G(x)\) and \(G(y)\) is zero, by symmetry assume that \(G(x)=0\) and \(G(y)\neq0\). Since \(x\notin B_n\) and $F(x)\neq 0$, the definition of \(G\) implies $d(x,o_n)<1$. Since \(G(y)\neq0\), we have \(d(y,o_n)\ge1\) and hence
\[
\|G(y)\|
=\theta_y\|F(y)\|
\le \|F(y)-F(o_n)\|
\le d(y,o_n).
\]
Therefore, we have
\[
\|G(x)-G(y)\|
=\|G(y)\|
\le d(y,x)+d(x,o_n)
< d(x,y)+1
\le2d(x,y).
\]
                                                                                                                                                                                                                                               
   
                                                                       
   
                                                                                                                                                                              
                                                   
   
               
                                                   
   
                                                                                                                                                                                                                                    
   
         
                  
                    
              
   
                    
   
              
          
                    
           
            
   
\textbf{Case II. } $x\in Z_{j_n}\setminus B_n$ and $y\in Z_{j_n}\cap B_n$. We can assume $G(x) \neq 0$, otherwise the result holds trivially. Now \eqref{eq:sparse-anchor-bound} gives
\[
 \|G(x)-G(y)\|=\|G(x)\|
 \le\min\{1,t_x/L_n\} \cdot (t_x+L_n)
 \le2t_x\le2d(x,y).
\]
\textbf{Case III.} $x,y\in B_n$. Then $G(x) = G(y) = 0$. Hence, we prove the claim.

Now applying \eqref{eq:sparse-source-energy} to $G/3$ yields
\[
 \sum_{x,y\in Z_{j_n}}\|G(x)-G(y)\|^2\mu_{j_n}(x,y)\le9k^2.
\]
For \(x\in Z_{j_n}\setminus A_n\), we have \(t_x>L_n\) and hence, \(\theta_x=1\). Moreover, \(d(x,o_n)\ge1\) by the choice of $L_n$. Therefore, $G(x)=F(x)$ for any $x\in Z_{j_n}\setminus A_n$. 
Since $\nu_n$ is supported on $(Z_{j_n}\setminus A_n)^2$, it follows that
\[
 \sum_{x,y\in Y_n}\|F(x)-F(y)\|^2\nu_n(x,y) =q_n^{-1}\! \sum_{x,y\in Z_{j_n}\setminus A_n} \|G(x)-G(y)\|^2\mu_{j_n}(x,y)\leq 9q_n^{-1}k^2<18k^2.
\]
Thus, $Y_n$ is a generalised $(3\sqrt2\,k,r_{j_n})$-expander. 
Since $d(Y_n, Y_1 \cup \cdots \cup Y_{n-1}) \geq n$, we conclude the proof.

\label{prop:Tessera_generalised_expander}
Let $\{(X_n,d_n)\}_{n\in \mathbb{N}}$ be a sequence of finite metric spaces and $(X,d)$ be their coarse disjoint union. Then $X$ can be coarsely embedded into Hilbert space \emph{if and only if} no subsequence $\{X_{n_k}\}_k$ forms generalised expanders.
<!-- end prop-3 -->
<!-- proof-2: body_md -->
If a subsequence \(\{X_{n_k}\}_k\) forms a sequence of generalised expanders, then \(X\) cannot coarsely embed into Hilbert space by Proposition~\ref{prop:tessera-general}. 

Conversely, assume that \(X\) cannot be coarsely embedded into Hilbert space. 
For each $n$, choose $a_n\geq n+2\mathrm{diam}(X_n)$. Replace the metric $d$ such that $d(x,y) = a_n + a_m$ for $x\in X_n$ and $y\in X_m$ with $n\neq m$, which retains the same coarse equivalence class. By Proposition~\ref{prop:tessera-general}, there are finite subsets \(Z_j\subseteq X\), \(r_j\to\infty\), \(K_0>0\) and symmetric probability measures \(\widetilde\mu_j\) on \(Z_j^2\) such that $\mathrm{supp}(\widetilde\mu_j)
 \subseteq\{(x,y):d(x,y)\ge r_j\}$ and
\begin{equation}\label{eq:tessera-global}
 \sum_{x,y} \|F(x)-F(y)\|^2\widetilde\mu_j(x,y) \leq K_0^2,
\end{equation}
for any Hilbert-space-valued map $F: Z_j \to \mathcal{H}$ satisfying $\|F(x) - F(y)\| \leq d(x,y)$ whenever $d(x,y) \geq 1$. After discarding finitely many \(j\), we may assume that $r_j>2$.

Let $\{e_n\}_{n}$ be the standard orthonormal basis of $\ell^2(\mathbb{N})$ and define $G: X \longrightarrow \ell^2(\mathbb{N})$ by $G(x):=a_n e_n$ for $x\in X_n$, which is $1$-Lipschitz by the choice of $d$. If $\widetilde\mu_j(x,y) \neq 0$ for some $x\in X_n$, $y\in X_m$ with $n\ne m$, then 
\[
 \|G(x)-G(y)\|^2=a_n^2+a_m^2
 \geq\frac{(a_n+a_m)^2}{2} = \frac{d(x,y)^2}{2} \ge\frac{r_j^2}{2}.
\]
Applying \eqref{eq:tessera-global} to the map $G$ gives
\[
 \widetilde\mu_j\left(\bigcup_{n\ne m}X_n\times X_m\right)
 \le\frac{2K_0^2}{r_j^2}\longrightarrow 0 \quad \text{as} \quad j \to +\infty.
\]
After discarding finitely many $j$, we may therefore assume that
\begin{equation}\label{eq:withinmass}
 \sum_n\widetilde\mu_j(X_n\times X_n)\ge\frac12.
\end{equation}

Fix such a $j$, and write $p_n:=\widetilde\mu_j(X_n\times X_n)$. Since $\widetilde\mu_j$ has finite support, there are only finitely many $n$ such that $p_n>0$. For such $n$, set $\nu_n:=p_n^{-1} \cdot \widetilde\mu_j|_{X_n\times X_n}$ and we have $D_n:=\mathrm{diam}(X_n)\ge r_j>2$.
Define
\[
 P_n:=\sup_F\sum_{x,y\in X_n}
 \|F(x)-F(y)\|^2 \nu_n(x,y),
\]
where the supremum is taken over all Hilbert-space-valued maps $F:X_n\to H$ satisfying $\|F(x) - F(y)\| \leq d(x,y)$ whenever $d(x,y) \geq 1$. Since \(\nu_n\) is supported on pairs \((x,y)\) with \(d(x,y)\ge r_j>1\), we have $P_n\le \mathrm{diam}(X_n)^2<\infty$.

We claim: 
\begin{equation}\label{EQ:pnKn}
\sum_n p_nP_n\le K_0^2. 
\end{equation}
To see this, for each \(n\) with \(p_n>0\), fix a Hilbert-space-valued map $F_n:X_n\to \mathcal{H}_n$ satisfying $\|F_n(x) - F_n(y)\| \leq d(x,y)$ whenever $d(x,y) \geq 1$. Choose \(u_n,v_n\in X_n\) with $d(u_n,v_n)=D_n$. After translating, assume \(F_n(u_n)=0\). Given $x\in X_n$, if \(d(x,u_n)\ge1\), then
$\|F_n(x)\|\le d(x,u_n)\le D_n$. If \(d(x,u_n)<1\), then $d(x,v_n)>D_n-1>1$ and hence,
\[
\|F_n(x)\| \le \|F_n(x)-F_n(v_n)\|+\|F_n(v_n)\|\le d(x,v_n)+d(u_n,v_n)\le2D_n \leq a_n.
\]
Consider $F:X \to \bigoplus_n \mathcal{H}_n$ such that $F|_{X_n}:=F_n$ for those finitely many $n$ with $p_n>0$, and $0$ otherwise. By the choice of $d$, for \(x\in X_n\) and \(y\in X_m\) with \(n\neq m\), we have
\[
\|F(x)-F(y)\|
\le \sqrt{a_n^2+a_m^2}
\le a_n+a_m
=d(x,y).
\]
Therefore, $\|F(x) - F(y)\| \leq d(x,y)$ whenever $d(x,y) \geq 1$. 
Applying \eqref{eq:tessera-global} to $F$, we deduce the following
\[
 K_0^2\ge
 \sum_n\sum_{x,y\in X_n} 
 \|F_n(x)-F_n(y)\|^2 \widetilde\mu_j(x,y) = \sum_n p_n \sum_{x,y\in X_n}
 \|F_n(x)-F_n(y)\|^2 \nu_n(x,y).
\]
Since only finitely many \(p_n\) are non-zero, taking the supremum independently over each \(F_n\) proves the claim.


Combining \eqref{eq:withinmass} and \eqref{EQ:pnKn}, there exists $n=n(j) \in \mathbb{N}$ with $p_n>0$ such that $P_n\leq 2K_0^2$. Set $\mu_j:=\nu_{n(j)}$, whose support is contained in $\big\{(x,y)\in X_{n(j)}^2:d(x,y)\ge r_j\big\}$.
Moreover, for any Hilbert space-valued map $F:X_{n(j)}\to \mathcal{H}$ with $\|F(x) - F(y)\| \leq d(x,y)$ whenever $d(x,y) \geq 1$, we have:
\[
 \sum_{x,y\in X_{n(j)}}
 \|F(x)-F(y)\|^2 \mu_j(x,y) \leq 2K_0^2
\]
Since $r_j \to \infty$ and $\mu_j(X_{n(j)}^2) = 1$, we obtain $n(j) \to +\infty$. Passing to a subsequence, we may assume that \(n(j)\) is strictly increasing. This finishes the proof.

\label{lem:comparison}
Let $Y$ be a finite metric space with $|Y| \geq 2$, and $\mu$ be a symmetric probability measure on $Y\times Y$ with $\mu(x,x)=0$ for every $x\in Y$. Suppose that $C>0$ satisfies
\begin{equation}\label{eq:comparison-hyp}
 \sum_{x,y\in Y}\|F(x)-F(y)\|^2 \mu(x,y) \leq C
\end{equation}
for any $1$-Lipschitz Hilbert-space-valued map $F: Y \to \mathcal{H}$. Then there is a symmetric non-negative function $\eta$ on $Y \times Y$ with $\eta(x,x)=0$ for every $x\in Y$, such that
\begin{equation}\label{eq:second-moment}
 \sum_{x,y\in Y}d(x,y)^2\eta(x,y)\le C
\end{equation}
and for every $f:Y\to\mathbb{C}$, we have
\begin{equation}\label{eq:dirichlet-domination}
 \sum_{x,y\in Y}|f(x)-f(y)|^2 \mu(x,y) 
 \le
 \sum_{x,y\in Y}|f(x)-f(y)|^2 \eta(x,y).
\end{equation}
<!-- end lem-1 -->
<!-- proof-3: body_md -->
Fix $o\in Y$. Let $\mathcal C$ be the set of kernels $\psi(x,y)=\|F(x)-F(y)\|^2$ arising from $1$-Lipschitz Hilbert-space-valued maps $F$ on $Y$. Write $Y=\{o,x_1,\ldots,x_m\}$ and for each kernel $\psi$, define
\[
 G_\psi(i,j):=\frac12\bigl(\psi(x_i,o)+\psi(x_j,o)-\psi(x_i,x_j)\bigr) \quad \text{for} \quad i,j=1,\cdots,m.
\]
Claim: Given a symmetric kernel $\psi$ on $Y$ with $\psi(x,x)=0$ and $0\le\psi(x,y)\le d(x,y)^2$ for any $x,y\in Y$, we have $\psi \in \mathcal C$ if and only if $G_\psi \geq 0$. 

Indeed, first assume that $\psi \in \mathcal C$. Replacing $F$ by $F-F(o)$, we assume $F(o)=0$. After realifying the target Hilbert space if necessary, we compute directly that $G_\psi(i,j) = \langle F(x_i),F(x_j)\rangle$ and hence, $G_\psi\ge0$. Conversely, suppose that $G_\psi\ge0$. Then there exist vectors $v_1,\cdots,v_m$ in some real Hilbert space such that $\langle v_i,v_j\rangle=G_\psi(i,j)$. Define $F(o)=0$ and $F(x_i):=v_i$ for $i=1,\cdots, m$. A direct calculation shows that
\[
\|F(x_i)-F(x_j)\|^2 = G_\psi(i,i)+G_\psi(j,j)-2G_\psi(i,j)=\psi(x_i,x_j)
\]
and $\|F(x_i)-F(o)\|^2=\psi(x_i,o)$. Hence, we obtain $\psi \in \mathcal C$. Using this characterisation for $\mathcal C$, it is clear that $\mathcal C$ is compact and convex.


Let $\mathcal P$ be the set of symmetric probability measures on $Y^2$ which vanish on the diagonal. Then $\mathcal P$ is a  compact and convex subset of $\mathrm{RR}^{Y \times Y}$. Moreover, we have
\[
 \max_{p\in\mathcal P}\sum_{x\ne y}p(x,y)\frac{\psi(x,y)}{d(x,y)^2}
 =\max_{x\ne y}\frac{\psi(x,y)}{d(x,y)^2}.
\]
To see this, first note that the left-hand side is at most the right-hand side. Conversely, if the
right-hand side is attained at $(x_0,y_0)$, then the measure $p$ defined by $p(x_0,y_0)=p(y_0,x_0)=\frac12$ and $p(x,y)=0$ otherwise, attains the same value.

Now scaling \eqref{eq:comparison-hyp} together with the above gives
\begin{equation}\label{eq:homogenized}
 \sum_{x,y}\psi(x,y) \mu(x,y)
 \le C\max_{x\ne y}\frac{\psi(x,y)}{d(x,y)^2} = C\max_{p\in\mathcal P}\sum_{x\ne y}p(x,y)\frac{\psi(x,y)}{d(x,y)^2}.
\end{equation}
For $\psi\in\mathcal C$ and $p\in\mathcal P$, define
\[
 \Lambda(\psi,p) :=C\sum_{x\ne y}p(x,y)\frac{\psi(x,y)}{d(x,y)^2} -\sum_{x,y}\mu(x,y)\psi(x,y).
\]
Then \eqref{eq:homogenized} implies that $\max_{p\in\mathcal P}\Lambda(\psi,p)\ge0$. The function $\Lambda$ is continuous and affine in each variable. Since $\mathcal C$ and $\mathcal P$ are compact and convex, the finite-dimensional
minimax theorem gives
\[
 \max_{p\in\mathcal P}\min_{\psi\in\mathcal C}\Lambda(\psi,p) =\min_{\psi\in\mathcal C}\max_{p\in\mathcal P}\Lambda(\psi,p)\ge0.
\]
Since the function $p\mapsto\min_{\psi\in\mathcal C}\Lambda(\psi,p)$ is continuous on $\mathcal P$, there exists $p_0\in\mathcal P$ such that $\min_{\psi\in\mathcal C}\Lambda(\psi,p_0)\geq 0$. Equivalently, $\Lambda(\psi,p_0)\ge0$ for any $\psi\in\mathcal C$.

Finally, we define
\begin{equation}\label{eq:eta-from-p}
 \eta(x,y)=
 \begin{cases}
 \displaystyle\frac{C\,p_0(x,y)}{d(x,y)^2},&x\ne y,\\[2mm]
 0,&x=y.
 \end{cases}
\end{equation}
Then $\eta$ is symmetric and non-negative, and \eqref{eq:second-moment} follows from:
\[
 \sum_{x,y}d(x,y)^2\eta(x,y)
 =C\sum_{x\ne y}p_0(x,y)=C.
\]
It remains to prove \eqref{eq:dirichlet-domination}. For a $1$-Lipschitz map $f:Y\to\mathbb{C}$, consider the kernel $\psi_f(x,y):=|f(x)-f(y)|^2$, which belongs to $\mathcal C$. Then $\Lambda(\psi_f, p_0) \geq 0$ implies
\[
 \sum_{x,y}\mu(x,y)|f(x)-f(y)|^2 \le C\sum_{x\ne y}p_0(x,y) \frac{|f(x)-f(y)|^2}{d(x,y)^2}=\sum_{x,y}\eta(x,y)|f(x)-f(y)|^2.
\]
For arbitrary $f:Y\to\mathbb{C}$, both sides of \eqref{eq:dirichlet-domination} are zero if $f$ is constant. Otherwise, 
\[
 L_f:=\max_{x\ne y}\frac{|f(x)-f(y)|}{d(x,y)}
\]
is finite and positive, and $f/L_f$ is $1$-Lipschitz. Applying the preceding
inequality to $f/L_f$, we conclude \eqref{eq:dirichlet-domination} for $f$.

\label{prop:matrix-obstruction}
Let $(X,d)$ be a discrete metric space of bounded geometry which does not coarsely embed into Hilbert space. Then there exist a sparse subspace $Y=\bigsqcup_{n \in \mathbb{N}} Y_n\subseteq X$, a constant $C\ge1$, a sequence $r_n\to\infty$, and self-adjoint matrices $S_n,M_n\in \mathfrak{B}(\ell^2(Y_n))$ satisfying the following conditions:
\begin{enumerate}
\item $\|S_n\|\le2$ with the maximal eigenvalue $\lambda_{\max}(S_n)\ge C^{-1}$;
\item $S_n\le M_n$ and $\|M_n\|\le1$;
\item $(M_n)_{xy}\ge0$ for all $x,y\in Y_n$, and $(M_n)_{xy}=0$ whenever $d(x,y)<r_n$;
\item Setting $S_n^{(R)}$ the restriction of $S_n$ on $E_R$ for $R\geq 1$, we have $\|S_n-S_n^{(R)}\|\le R^{-2}$.
\end{enumerate}
In particular, $S=\mathrm{(SOT)}$-$\sum_{n} S_n$ belongs to $C_u^*(X)$.

\label{lem:weighted Schur test}
For a finite set $Y$, a symmetric matrix $K$ on $Y$ with non-negative entries and weights $w_x>0$ such that $\sum_y K_{xy}w_y\leq a w_x$, we have $\|K\|\le a$.
<!-- end lem-2 -->
<!-- proof-4: body_md -->
For any $\xi, \zeta \in \ell^2(Y)$, we have
\[
 |\langle\xi,K\zeta\rangle| \leq \sum_{x,y}K_{xy}|\xi_x|\cdot |\zeta_y| \leq\left(\sum_{x,y}K_{xy}\frac{w_y}{w_x}|\xi_x|^2\right)^{1/2}\left(\sum_{x,y}K_{xy}\frac{w_x}{w_y}|\zeta_y|^2\right)^{1/2}
 \leq a\|\xi\|\cdot\|\zeta\|.
\]
This shows that $\|K\|\le a$.

<!-- proof-5: body_md -->
Replacing \(d\) if necessary, by the coarsely equivalent metric
\[
\widetilde d(x,y)=
\begin{cases}
0,& \text{if }x=y,\\
\max\{d(x,y),1\},& \text{if }x\ne y,
\end{cases}
\]
we may assume throughout this proof that \(d(x,y)\ge1\) for \(x\ne y\). This replacement does not change the coarse structure and the uniform Roe algebra. 


By Proposition~\ref{prop:sparse-reduction}, choose a sparse subspace $Z=\bigsqcup_n Z_n\subseteq X$ whose components form a sequence of generalised $(k,r_n)$-expanders for some $k\geq 1$ and $r_n \to +\infty$, witnessed by measures $\mu_n$ on $Z_n \times Z_n$. Since $d(x,y) \geq 1$ for $x\neq y\in X$, we have
\[ 
\sum_{x,y\in Z_n}\|F(x)-F(y)\|^2 \mu_n(x,y) \leq k^2
\]
for any $1$-Lipschitz Hilbert-space-valued map $F: Z_n \to \mathcal{H}$. After discarding finitely many terms we may also assume \(r_n>1\) in the sequel, so the conclusions obtained below are unchanged for the original metric.


Applying Lemma~\ref{lem:comparison} to each $Z_n$ with constant $C:=k^2$, we obtain symmetric non-negative functions $\eta_n$ on $Z_n \times Z_n$ satisfying \eqref{eq:second-moment} and \eqref{eq:dirichlet-domination}. For $x\in Z_n$, set
\[
 m_n(x):=\sum_{y\in Z_n}\mu_n(x,y), \quad c_n(x):=\sum_{y \in Z_n}\eta_n(x,y) \quad \text{and} \quad h_n(x):=\sum_{y\in Z_n} d(x,y)^2\eta_n(x,y).
\]
Taking $f=\chi_{\{x\}}$ in \eqref{eq:dirichlet-domination} gives $m_n(x)\leq c_n(x)$. By the choice of the metric, we obtain
\begin{equation}\label{eq:weights}
 0\le m_n\le c_n\le h_n \quad \text{and} \quad 1\leq Q_n:=\sum_xh_n(x)\le C.
\end{equation}

For each $n$, set $Y_n:=\{x \in Z_n: h_n(x)>0\}$. It follows from \eqref{eq:weights} that on $Z_n \setminus Y_n$, $m_n(x) = c_n(x) = 0$, so $\mu_n$ and $\eta_n$ are supported on $Y_n \times Y_n$. Given $f: Y_n \to \mathbb{C}$, denote $D_f$ the diagonal operator on $\ell^2(Y_n)$. Define the following $Y_n$-by-$Y_n$ matrices:
\[
 A_{\mu_n}:=\left(\mu_n(x,y)\right)_{x,y\in Y_n}, \quad A_{\eta_n}=\left(\eta_n(x,y)\right)_{x,y\in Y_n}
\]
and 
\[
L_{\mu_n}:=D_{m_n} - A_{\mu_n}, \quad L_{\eta_n}:=D_{c_n}-A_{\eta_n}.
\]
Also define:
\begin{equation*}\label{eq:matrix-form}
 M_n=D_{h_n}^{-1/2}A_{\mu_n}D_{h_n}^{-1/2} \quad \text{and} \quad
 S_n=D_{h_n}^{-1/2}(D_{m_n}-L_{\eta_n})D_{h_n}^{-1/2}.
\end{equation*}

For the weight $w_x=\sqrt{h_n(x)}$ on $Y_n$, the matrix $M_n$ and the off-diagonal part $S'_n$ of $S_n$, we have
\[
 \sum_y(M_n)_{xy}w_y
 =\frac{m_n(x)}{h_n(x)}w_x\le w_x
 \quad \text{and} \quad
 \sum_y(S'_n)_{xy}w_y
 =\frac{c_n(x)}{h_n(x)}w_x\le w_x.
\]
Applying Lemma~\ref{lem:weighted Schur test}, we obtain $\|M_n\|\le1$ and $\|S'_n\|\le1$. Since $M_n$ is self-adjoint, we have $M_n \leq \mathrm{Id}_{\ell^2(Y_n)}$. The diagonal part of $S_n$ has norm at most $1$ by \eqref{eq:weights} and hence, $\|S_n\|\le2$. On the other hand, for each $R>0$, set $S_n^{(R)}$ to be the restriction of $S_n$ on the $R$-diagonal of $Y_n\times Y_n$. Since
\[
\sum_y(S_n - S_n^{(R)})_{xy}w_y = \sum_{y:d(x,y)>R}(S'_n)_{xy}w_y
 =\frac1{\sqrt{h_n(x)}}\sum_{y:d(x,y)>R}\eta_n(x,y)
 \le R^{-2}w_x,
\]
the fact above implies that $\|S_n - S_n^{(R)}\| \leq R^{-2}$. Hence, $S:=\mathrm{(SOT)}$-$\sum_n S_n \in C^*_u(X)$. 

According to \eqref{eq:dirichlet-domination} and $L_{\mu_n}, L_{\eta_n}$ being symmetric, we have $L_{\mu_n} \leq L_{\eta_n}$ and hence, $D_{m_n}-L_{\eta_n}\le A_{\mu_n}$. This implies $S_n \leq M_n$. Define a unit vector $u_n \in \ell^2(Y_n)$ by $u_n(x):=\sqrt{h_n(x)/Q_n}$. Then we have
\[
\langle u_n,S_nu_n\rangle = \frac1{Q_n}\left(\sum_x(m_n(x)-c_n(x))
                     +\sum_{x\ne y}\eta_n(x,y)\right)=\frac1{Q_n}\ge C^{-1}.
\]
This finishes the proof.

<!-- lem-3: body_md -->
\label{lem:cuts}
For a finite subset $V$ of a Hilbert space $\mathcal{H}$ and $t>0$, there exist finitely many maps $c_1, \cdots, c_N: V \to \{-1,1\}$ and weights $\lambda_j \geq 0$ with $\sum_j\lambda_j=1$ such that 
\[
\sum_{j=1}^N\lambda_jc_j(u)c_j(v) =e^{-t\|u-v\|} \quad \text{for any} \quad u,v \in V.
\]
<!-- end lem-3 -->
<!-- proof-6: body_md -->
Let us realify $\mathcal{H}$, translate one point in $V$ to the origin, and identify the linear span of $V$ with $\mathrm{RR}^d$. The case $d=0$ holds trivially, and we assume $d\geq 1$. 
Let $\sigma$ be the probability Lebesgue measure on $S^{d-1}$, $e_1=(1,0,\cdots,0) \in S^{d-1}$ and
\[
 c_d=\left(\int_{S^{d-1}} |\langle e_1, \vartheta\rangle| \mathrm{d}\sigma(\vartheta)\right)^{-1}.
\]
By rotation invariance and homogeneity, for any $w\in\mathrm{RR}^d$ we have
\begin{equation}\label{EQ:norm of w}
 c_d\int_{S^{d-1}}|\langle w,\vartheta\rangle|\mathrm{d}\sigma(\vartheta)=\|w\|.
\end{equation}
By the layer-cake identity, for any $u,v \in \mathcal{H}$ and $\vartheta \in S^{d-1}$ we have
\[
|\langle u-v,\vartheta\rangle| = \int_{\mathrm{RR}} \left| \chi_{(-\infty,\langle u,\vartheta\rangle)}(\alpha) - \chi_{(-\infty,\langle v,\vartheta\rangle)}(\alpha)\right|\mathrm{d}\alpha.
\]
Combining with \eqref{EQ:norm of w}, we obtain:
\[
 \|u-v\|= c_d \int_{S^{d-1}} \int_{\mathrm{RR}} \left| \chi_{(-\infty,\langle u,\vartheta\rangle)}(\alpha) -
 \chi_{(-\infty,\langle v,\vartheta\rangle)}(\alpha) \right| \mathrm{d}\alpha\mathrm{d}\sigma(\vartheta).
\]

For each $\vartheta\in S^{d-1}$ and $\alpha\in\mathrm{RR}$, set $E_{\vartheta,\alpha}:=\{z\in V:\langle z,\vartheta\rangle>\alpha\}$.
Then we have $\chi_{(-\infty,\langle z,\vartheta\rangle)}(\alpha) = \chi_{E_{\vartheta,\alpha}}(z)$. Thus each pair $(\vartheta,\alpha)$ determines a cut $V=E_{\vartheta,\alpha}\sqcup (V \setminus E_{\vartheta,\alpha})$.
Since $V$ is finite, only finitely many distinct cuts occur. Grouping together all $(\vartheta,\alpha)$ which determine the same cut, we obtain subsets $E_1,\cdots,E_m\subseteq V$ and numbers $a_1,\cdots,a_m\ge0$ such that
\begin{equation}\label{eq:l1cuts}
 \|u-v\| = \sum_{\ell=1}^m a_\ell \left| \chi_{E_\ell}(u)-\chi_{E_\ell}(v) \right| = \sum_{\ell=1}^m a_\ell \chi_{D_{E_\ell}}(u,v).
\end{equation}

For $\ell=1,\cdots, m$, set $p_\ell:=(1-e^{-t a_\ell})/2$. For $\eta=(\eta_1,\cdots,\eta_m)\in\{0,1\}^m$, define
\[
 c_\eta(u) := (-1)^{\sum_{\ell=1}^m \eta_\ell\chi_{E_\ell}(u)} = \prod_{\ell=1}^m (-1)^{\eta_\ell\chi_{E_\ell}(u)} \quad \text{and} \quad 
 \lambda_\eta:=\prod_{\ell=1}^m p_\ell^{\eta_\ell}(1-p_\ell)^{1-\eta_\ell}.
\]
Clearly each $\lambda_\eta \geq 0$, and
\[
 \sum_{\eta\in\{0,1\}^m}\lambda_\eta =
 \prod_{\ell=1}^m \left( \sum_{\eta_\ell=0}^1 p_\ell^{\eta_\ell}(1-p_\ell)^{1-\eta_\ell} \right) =
 \prod_{\ell=1}^m\bigl(p_\ell+(1-p_\ell)\bigr) =1.
\]
For each $\ell =1,\cdots,m$ and $u,v\in V$, we have
\begin{equation}\label{EQ:p ell}
 (1-p_\ell) + p_\ell (-1)^{\left|\chi_{E_\ell}(u)-\chi_{E_\ell}(v)\right|} =
 \exp\left(-ta_\ell \left|\chi_{E_\ell}(u)-\chi_{E_\ell}(v)\right|\right).
\end{equation}
Indeed, if $u$ and $v$ lie on the same side of the cut $E_\ell$, both sides equal $1$. If they lie on opposite sides, the left-hand side is $1-2p_\ell=e^{-ta_\ell}$. Therefore, we have
\begin{align*}\label{EQ:cut estimate compute}
\sum_{\eta\in\{0,1\}^m}
\lambda_\eta c_\eta(u)c_\eta(v) 
&=
\prod_{\ell=1}^m \left(\sum_{\eta_\ell=0}^1p_\ell^{\eta_\ell}(1-p_\ell)^{1-\eta_\ell} (-1)^{\eta_\ell(\chi_{E_\ell}(u)+\chi_{E_\ell}(v))} \right) \notag\\
&=
 \prod_{\ell=1}^m\left( (1-p_\ell) + p_\ell(-1)^{\left|\chi_{E_\ell}(u)-\chi_{E_\ell}(v)\right|}\right) \notag\\
&=
 \exp\left(-t \sum_{\ell=1}^m a_\ell |\chi_{E_\ell}(u)-\chi_{E_\ell}(v)|\right) \notag\\
&=
 e^{-t\|u-v\|},
\end{align*}
where the penultimate equality comes from \eqref{EQ:p ell} and the last from \eqref{eq:l1cuts}.

\label{lem:ql-approx}
Let $X$ be a discrete metric space of bounded geometry, and $F:X\to \mathcal{H}$ be a coarse map with control function $\rho_+$. For $t>0$, put
\begin{equation}\label{EQ:phi t}
 \varphi_t(x,y):=e^{-t\|F(x)-F(y)\|} \quad \text{for} \quad x,y\in X.
\end{equation}
Then for any $T\in\mathfrak{B}(\ell^2(X))$, $\xi,\eta\in\ell^2(X)$ and $R>0$, we have
\begin{equation}\label{eq:matrix-coefficient-schur}
\big|\langle \eta,(T-M_{\varphi_t}(T))\xi\rangle\big| \le 2\sup_{d(A,B)>R}
\big|\langle \eta,(\chi_AT\chi_B+\chi_BT\chi_A)\xi\rangle\big| +4\|T\|\cdot\|\xi\|\cdot\|\eta\|\sqrt{p_t(R)},
\end{equation}
where $p_t(R):=N_X(R) \cdot q_t(\rho_+(R))$.
<!-- end prop-5 -->
<!-- proof-8: body_md -->
We first prove \eqref{eq:matrix-coefficient-schur} when $\xi$ and $\eta$ have finite support. Choose a finite subset $Y\subseteq X$ containing both supports and set $T_Y:=\chi_YT\chi_Y$, viewed as an operator on $\ell^2(Y)$. 
Applying Lemma~\ref{lem:cuts} to $F(Y)$ and pulling the resulting maps back to $Y$, we obtain maps $c_j:Y\to\{-1,1\}$ and weights $\lambda_j\ge0$ with $\sum_j\lambda_j=1$ such that
\begin{equation}\label{eq:color-representation-schur}
 \varphi_t(x,y)=\sum_j\lambda_jc_j(x)c_j(y)
 \quad \text{for any} \quad x,y\in Y.
\end{equation}
For each $j$, define the cut $C_j:=\{v\in Y: c_j(v)=1\}$ and denote $D_j:=D_{C_j}$. Then \eqref{eq:color-representation-schur} can be reformulated to be
\begin{equation}\label{eq:cut-representation-schur}
 q_t(\|F(x)-F(y)\|) = \sum_j\lambda_j\chi_{D_j}(x,y) \quad \text{for any} \quad x,y\in Y.
\end{equation}

To decompose $M_{\varphi_t}(T)$ accordingly, for each $j$ let $U_j\in\mathfrak{B}(\ell^2(Y))$ be the diagonal unitary defined by $U_j\delta_x=c_j(x)\delta_x$ for $x\in Y$, and put $S_j:=T_Y-U_jT_YU_j$. Then
\[
 \chi_Y(T-M_{\varphi_t}(T))\chi_Y=\sum_j\lambda_jS_j, \qquad \|S_j\| \leq 2\|T_Y\| \leq 2\|T\|,
\]
and
\begin{equation}\label{EQ:Sj}
 (S_j)_{xy} =\left(1-c_j(x)c_j(y)\right)T_{xy} =2\chi_{D_j}(x,y) T_{xy}.
\end{equation}

Fix $R>0$. For each $j$, consider the $R$-boundary
\[
 E_j:=\{x\in Y:\text{there exists }z\in B_Y(x,R)\text{ with }(x,z)\in D_j\}.
\]
Then we have $d(C_j^+,C_j^-)>R$ for $C_j^+:=C_j\setminus E_j$ and $C_j^-:=C_j^c\setminus E_j$. 
Moreover, for $x\in Y$, \eqref{eq:cut-representation-schur} shows that
\[
 \sum_j\lambda_j\chi_{E_j}(x)
\leq \sum_j\lambda_j \sum_{z\in B_Y(x,R)}\chi_{D_j}(x,z) = \sum_{z\in B_Y(x,R)}q_t(\|F(x)-F(z)\|)\leq p_t(R).
\]
For each $j$, let $P_j:=\chi_{E_j}$ and $Q_j:=\mathrm{Id}_{\ell^2(Y)}-P_j$. Then the inequality above shows
\begin{equation}\label{EQ:estimate of P}
\sum_j \lambda_j P_j\leq p_t(R) \mathrm{Id}_{\ell^2(Y)}.
\end{equation}
Moreover, note that $\mathrm{Ran}(Q_j) = \ell^2(C_j^+) \oplus \ell^2(C_j^-)$.  Hence, according to this decomposition, \eqref{EQ:Sj} implies:
\[
 Q_jS_jQ_j
 =2\begin{pmatrix}
 0&\chi_{C_j^+}T\chi_{C_j^-}\\
 \chi_{C_j^-}T\chi_{C_j^+}&0
 \end{pmatrix}.
\]


On the other hand, we have the decomposition:
\[
\sum_j \lambda_j S_j = \sum_j \lambda_j Q_jS_jQ_j + \sum_j \lambda_j P_jS_j + \sum_j \lambda_j Q_jS_jP_j.
\]
For any $\xi, \eta \in \ell^2(Y)$, we have
\begin{align*}
\left|
\left\langle
\eta,
\sum_j\lambda_jQ_jS_jQ_j\xi
\right\rangle
\right|
\le
2\sup_{d(A,B)>R}
\left|
\left\langle
\eta,
(\chi_AT\chi_B+\chi_BT\chi_A)\xi
\right\rangle
\right|.
\end{align*}
Next, by the Cauchy--Schwarz inequality, \eqref{EQ:estimate of P} and $\|S_j\|\le2\|T\|$, we obtain:
\begin{align*}
 \left|\left\langle\eta,
 \sum_j\lambda_jP_jS_j\xi\right\rangle\right|
 &=\left|\sum_j\lambda_j\langle P_j\eta,S_j\xi\rangle\right|\leq
 \left(\sum_j\lambda_j\|P_j\eta\|^2\right)^{1/2}
 \left(\sum_j\lambda_j\|S_j\xi\|^2\right)^{1/2}\\
 &\leq
 \left\langle\eta, \left(\sum_j\lambda_jP_j\right)\eta\right\rangle^{1/2}
 \cdot 2\|T\|\cdot \|\xi\| \leq 2\|T\|\cdot \|\eta\|\cdot \|\xi\| \sqrt{p_t(R)}.
\end{align*}
Similarly,
\[
\left|
\left\langle
\eta,
\sum_j\lambda_jQ_jS_jP_j\xi
\right\rangle
\right|=
\left|
\sum_j\lambda_j
\langle S_j^*Q_j\eta,P_j\xi\rangle
\right|\le
2\|T\|\cdot \|\eta\|\cdot \|\xi\| \sqrt{p_t(R)}.
\]
Combining these estimates proves \eqref{eq:matrix-coefficient-schur} for finitely supported $\xi,\eta$. The general case follows by approximating $\xi$ and $\eta$ in terms of finitely supported ones.

\label{lem:generalised expander implies coupling}
Let $\{Y_n\}_n$ be finite subsets of $(X,d)$ which form a sequence of generalised expanders. Extending their witnessing measures by zero to $X\times X$ gives an escaping quasi-local coupling on $X$.

Then $f$ is $1$-Lipschitz, equals zero on $A$, and equals $R$ on $B$. Hence,

 \leq \sum_{x,y \in Y_n}|f(x)-f(y)|^2 \mu_n(x,y) \leq k^2,

which implies $\mu_n(A\times B) \leq k^2/R^2 <\varepsilon$ and finishes the proof.

<!-- proof-10: body_md -->
First, assume that \(X\) does not coarsely embed into any Hilbert space. Proposition~\ref{prop:tessera-general} provides finite subsets $Y_n\subseteq X$ which form a sequence of generalised expanders. Then Lemma~\ref{lem:generalised expander implies coupling} provides an escaping quasi-local coupling on $X$, which shows the sufficiency. 

Conversely, let $F:X\to\mathcal{H}$ be a coarse embedding with control functions $\rho_-,\rho_+$, and suppose that $\{\nu_n\}_n$ is an escaping quasi-local coupling. For each $n$, let
\[
 m_n(x):=\sum_{y\in X}\nu_n(x,y),
 \quad \text{and} \quad
 Y_n:=\{x\in X:m_n(x)>0\}.
\]
Since $\nu_n$ is finitely supported and symmetric, $Y_n$ is finite and $\nu_n$ is supported on $Y_n\times Y_n$. Define $v_n\in\ell^2(Y_n)$ and a matrix $K_n\in\mathfrak{B}(\ell^2(Y_n))$ by
\[
 v_n(x):=\sqrt{m_n(x)}
 \quad \text{and} \quad
 (K_n)_{xy}:=\frac{\nu_n(x,y)}{\sqrt{m_n(x)m_n(y)}}
 \quad \text{for} \quad x,y\in Y_n.
\]
Then $\|v_n\|=1$ and $\sum_{y\in Y_n}(K_n)_{xy}v_n(y)=v_n(x)$. Hence, Lemma~\ref{lem:weighted Schur test} shows $\|K_n\|= 1$. Regard $K_n$ and $v_n$ as supported on $Y_n\subseteq X$. Direct calculations show 
\begin{equation}\label{eq:coupling-as-coefficient}
 \langle v_n,\chi_AK_n\chi_Bv_n\rangle=\nu_n(A\times B) \quad \text{for any} \quad A,B\subseteq X.
\end{equation}

For $R>0$, set $\alpha(R):=\sup\{\nu_n(A\times B): n\in \mathbb{N} \text{ and } A,B \subseteq X \text{ with } d(A,B) > R\}$. 
Condition \textbf{(Q)} implies that $\alpha(R)\to 0$ as $R\to\infty$. Applying Proposition~\ref{lem:ql-approx} to $T=K_n$ and $\xi=\eta=v_n$ and using \eqref{eq:coupling-as-coefficient}, we obtain
\begin{equation}\label{eq:coupling-from-schur}
\big|\langle v_n,(K_n-M_{\varphi_t}(K_n))v_n\rangle\big|\le4\alpha(R)+4\sqrt{p_t(R)}
\end{equation}
for every $R,t>0$, where $\varphi_t(x,y)=e^{-t\|F(x)-F(y)\|}$ and $p_t(R)=N_X(R) \cdot q_t(\rho_+(R))$. Direct calculations show 
\begin{equation}\label{EQ:calculation coupling}
\big|\langle v_n,(K_n-M_{\varphi_t}(K_n))v_n\rangle\big| = \left|1-\sum_{x,y\in Y_n}\varphi_t(x,y)\nu_n(x,y)\right|.
\end{equation}

On the other hand, for every fixed $t>0$ and $S>0$, we have
\begin{align*}
0\le\sum_{x,y\in Y_n}\varphi_t(x,y)\nu_n(x,y)
&\le \nu_n\bigl(\{(x,y):d(x,y)\le S\}\bigr)+e^{-t\rho_-(S)}.
\end{align*}
Condition \textbf{(E)} therefore gives
\[
\limsup_{n\to\infty}\sum_{x,y\in Y_n}\varphi_t(x,y)\nu_n(x,y)
 \le e^{-t\rho_-(S)}.
\]
Letting $S\to\infty$, we obtain
\begin{equation}\label{eq:coupling-kernel-vanishes}
 \lim_{n\to \infty}\sum_{x,y\in Y_n}\varphi_t(x,y)\nu_n(x,y)= 0 \quad \text{for each fixed} \quad t>0.
\end{equation}


Taking $n\to\infty$ in \eqref{eq:coupling-from-schur} and using \eqref{EQ:calculation coupling}, \eqref{eq:coupling-kernel-vanishes} yields 
\[
 1\le4\alpha(R)+4\sqrt{p_t(R)}
 \quad \text{for any} \quad R,t>0.
\]
For fixed $R$, we have $p_t(R)\to0$ as $t\to0+$ and hence $1\le4\alpha(R)$, which contradicts condition \textbf{(Q)}. Thus, no escaping quasi-local coupling can exist on $X$.

\label{prop:ghost proj are compact}
Let $\{(X_n,d_n)\}_{n}$ be a sequence of finite metric spaces of uniformly bounded geometry, $X$ be their coarse disjoint union, and $\mathcal{H}_0$ be a separable Hilbert space. If $X$ is coarsely embeddable into Hilbert space, then for every locally compact ghost projection \(P\in\mathfrak{B}(\ell^2(X;\mathcal{H}_0))\), we have
\[
\operatorname{dist}\big(P,C_q^*(X;\mathcal{H}_0)) = \operatorname{dist}\big(P,C^*(X;\mathcal{H}_0)\big)
=
\begin{cases}
0,& \text{if }P\text{ is compact};\\
\frac12,&\text{if }P\text{ is non-compact}.
\end{cases}
\]
In particular, any ghost projection in $C^*_q(X;\mathcal{H}_0)$ is compact.
<!-- end thm-1 -->

The Roe-algebra analogue of the compactness conclusion of Theorem~\ref{prop:ghost proj are compact} is already known. Indeed, if \(X\) is coarsely embeddable, then the inclusion $\mathfrak{K}(\ell^2(X;\mathcal{H}_0)) \hookrightarrow G(X;\mathcal{H}_0)$ induces an isomorphism on \(K\)-theory by \cite[Corollary 36]{Fin14} (see also \cite[Theorem C]{WFZ25}). <!-- lem-5: body_md -->
\label{lem:ghost projeciton elementary}
Let \(X\) be a discrete metric space of bounded geometry, and \(\mathcal H_0\) be a separable Hilbert space.
\begin{enumerate}
 \item If \(P\in\mathfrak{B}(\ell^2(X;\mathcal{H}_0))\) is a ghost projection and \(Q\) is a quasi-local projection with \(\|P-Q\|<1\), then \(Q\) is a ghost.
 \item For every locally compact projection \(P\in\mathfrak{B}(\ell^2(X;\mathcal{H}_0))\), we have
\[
   \operatorname{dist}\big(P,C^*(X;\mathcal{H}_0)\big)\le\frac12.
\]
\end{enumerate}
<!-- end lem-5 -->
<!-- proof-12: body_md -->
(1). Since \(P\) is a ghost projection, for each fixed \(R>0\) we have
\[
\|P\chi_{B(x,R)}\|^2 \le \sum_{y\in B(x,R)}\|P\chi_{\{y\}}\|^2 =\sum_{y\in B(x,R)}\|P_{yy}\|\le N_X(R) \cdot \sup_{y\in B(x,R)}\|P_{yy}\| \to 0
\]
as $x\to \infty$. Given \(\varepsilon>0\), quasi-locality of \(Q\) provides \(R>0\) such that
\[
\|\chi_{X\setminus B(x,R)}Q\chi_{\{x\}}\|<\varepsilon \quad \text{for any} \quad x\in X.
\]
Hence, $\|PQ\chi_{\{x\}}\| \le \|P\chi_{B(x,R)}\|+\varepsilon$ and hence, \(\|PQ\chi_{\{x\}}\|\to0\) as \(x\to\infty\). Note that
\[
\|Q\chi_{\{x\}}\| \le \|(Q-P)Q\chi_{\{x\}}\|+\|PQ\chi_{\{x\}}\|\le \|P-Q\| \cdot \|Q\chi_{\{x\}}\|+\|PQ\chi_{\{x\}}\|.
\]
Since $\|P-Q\|<1$, the above implies \(\|Q\chi_{\{x\}}\|\to 0\) as \(x\to\infty\). Since $Q$ is a projection, we have $\|Q_{xy}\| \le \|Q\chi_{\{x\}}\|\cdot\|Q\chi_{\{y\}}\|$. This concludes that \(Q\) is a ghost.

(2). Fix $\varepsilon>0$. Since $P$ is locally compact, we can choose a diagonal operator $e:=\mathrm{(SOT)}$-$\sum_{x\in X}e_{x}$, where $e_x$ is a finite rank projection on $\ell^2(\mathbb{C} \delta_x; \mathcal{H}_0) \cong \mathcal{H}_0$, such that $\|(1-e)P\|<\varepsilon$. The projection \(e\) is locally compact and has propagation zero, so \(e\in C^*(X;\mathcal{H}_0)\). Moreover, we have
\[
\|P-ePe\| \le \|(1-e)P\|+\|eP(1-e)\| <2\varepsilon.
\]
Since \(0\le ePe\le e\), we have $\left\|ePe-\frac12e\right\|\le\frac12$. Therefore,
\[
\operatorname{dist}\bigl(P,C^*(X;\mathcal H_0)\bigr) \le \left\|P-\frac12e\right\| <\frac12+2\varepsilon.
\]
Letting $\varepsilon \to 0+$, we conclude (2).

<!-- prop-6: body_md -->
\label{prop:general ghost compact distance}
Let \(X\) be a discrete metric space of bounded geometry, and let \(\mathcal H_0\) be a separable Hilbert space. Then the following are equivalent:
\begin{enumerate}
 \item Every ghost projection in \(C_q^*(X;\mathcal H_0)\) is compact;
 \item Every ghost projection \(P\in\mathfrak{B}(\ell^2(X;\mathcal{H}_0))\) with $\operatorname{dist}(P,C_q^*(X;\mathcal{H}_0))<\frac12$ is compact;
 \item Every locally compact ghost projection \(P\in\mathfrak{B}(\ell^2(X;\mathcal{H}_0))\) satisfies:
 \[
\operatorname{dist}\big(P,C_q^*(X;\mathcal{H}_0)) = \operatorname{dist}\big(P,C^*(X;\mathcal{H}_0)\big)
=
\begin{cases}
0,& \text{if }P\text{ is compact};\\
\frac12,&\text{if }P\text{ is non-compact}.
\end{cases}
\]
\end{enumerate}
<!-- end prop-6 -->
<!-- proof-13: body_md -->
``(1) $\Rightarrow$ (2)'': Let \(P\in\mathfrak{B}(\ell^2(X;\mathcal{H}_0))\) be a ghost projection which satisfies $\operatorname{dist}(P,C_q^*(X;\mathcal{H}_0))<\frac12$. Choose self-adjoint \(T\in C_q^*(X;\mathcal{H}_0)\) such that $\delta:=\|P-T\|<\frac12$. Since \(P\) is a projection, we have $\sigma(T)\subseteq[-\delta,\delta]\cup[1-\delta,1+\delta]$. Hence, functional calculus provides a projection $Q:=\chi_{(1/2,\infty)}(T)\in C_q^*(X;\mathcal{H}_0)$ satisfying
\[
\|P-Q\|
\le\|P-T\|+\|T-Q\|
\le2\delta<1.
\]
By Lemma~\ref{lem:ghost projeciton elementary}, \(Q\) is a ghost projection and hence compact by (1). Hence, $Q$ has finite rank. Since \(\|P-Q\|<1\), the projection \(P\) must also have finite rank.

``(2) $\Rightarrow$ (3)'': Let \(P\in\mathfrak{B}(\ell^2(X;\mathcal{H}_0))\) be a locally compact ghost projection. If \(P\) is compact, then \(P\in C^*(X;\mathcal{H}_0)\), so both distances are zero. Now suppose that \(P\) is non-compact. By (2), we have $\operatorname{dist}(P,C_q^*(X;\mathcal{H}_0))\geq \frac12$. Combining with Lemma~\ref{lem:ghost projeciton elementary}(2), we conclude (3). 
Finally, ``(3) $\Rightarrow$ (1)'' is trivial.

<!-- prop-7: body_md -->
\label{prop:traceratio}
Let $\{(X_n,d_n)\}_{n\in \mathbb{N}}$ be a sequence of finite metric spaces of uniformly bounded geometry, $X$ be their coarse disjoint union, and $\mathcal{H}_0$ be a separable Hilbert space. Assume $X$ coarsely embeds into Hilbert space. Let $T:=\mathrm{(SOT)}$-$\sum_{n}T_n \in C^*_q(X;\mathcal{H}_0)$ be a positive ghost operator such that each $T_n$ is in the trace class. If 
\begin{equation}\label{eq:trbound}
 \sup_{n:T_n\neq 0} \frac{\mathrm{Tr}(T_n)}{\mathrm{Tr}(T_n^2)}<\infty,
\end{equation}
then only finitely many $T_n$ are non-zero. In particular $T$ is compact.

\label{lem:trace coupling}
Let $(Y,d)$ be a finite metric space and $\mathcal{H}_0$ be a separable Hilbert space. For a non-zero positive trace-class operator $T \in \mathfrak{B}(\ell^2(Y;\mathcal{H}_0))$, set
\[
c_T=\frac{\mathrm{Tr}(T)}{\mathrm{Tr}(T^2)} \quad \text{and} \quad \nu_T(x,y)=\frac{\|T_{xy}\|_{2}^2}{\mathrm{Tr}(T^2)} \quad \text{for} \quad x,y\in Y.
\]
Then $\nu_T$ is a symmetric probability measure. Moreover, we have:
\begin{enumerate}
 \item For any $A,B \subseteq Y$, $\nu_T(A\times B) \leq c_T \cdot \|\chi_AT\chi_B\|$;
 \item For any $R \geq 0$, 
 \[
 \nu_T(\{(x,y) \in Y^2: d(x,y) \leq R\}) \leq c_T \cdot \sup_{x\in Y}|B(x,R)| \cdot \sup_{x\in Y}\|T_{xx}\|.
 \]
\end{enumerate}
<!-- end lem-6 -->
<!-- proof-14: body_md -->
Since $T$ is a nonzero positive trace-class operator, $0<\mathrm{Tr}(T^2)<\infty$. Also
\[
 \sum_{x,y\in Y}\|T_{xy}\|_2^2=\|T\|_2^2=\mathrm{Tr}(T^2),
\]
and $T_{yx}=T_{xy}^*$ gives the symmetry of $\nu_T$. For any $A,B \subseteq Y$, both $\chi_A T^{1/2}$ and $\chi_B T^{1/2}$ are Hilbert-Schmidt. Then we have
\[
\|\chi_A T \chi_B\|_1 \leq \|\chi_A T^{1/2}\|_2\cdot \|T^{1/2}\chi_B\|_2 =\sqrt{\mathrm{Tr}(\chi_A T \chi_A)\mathrm{Tr}(\chi_B T \chi_B)}\leq\mathrm{Tr}(T).
\]
Moreover, recall that $\|S\|_2^2 \leq \|S\| \cdot \|S\|_1$ for any trace class operator $S$. Hence, 
\[
\nu_T(A\times B)=\frac{\|\chi_AT\chi_B\|_2^2}{\mathrm{Tr}(T^2)} \leq\frac{\mathrm{Tr}(T)}{\mathrm{Tr}(T^2)} \|\chi_AT\chi_B\|,
\]
which concludes (1).

As for (2), let $J_x:\mathcal{H}_0\to \ell^2(Y;\mathcal{H}_0)$ be the inclusion at $x \in Y$, and put $B_x=T^{1/2}J_x$. Then $T_{xy}=B_x^*B_y$ and $\|B_x\|^2=\|T_{xx}\|$. Hence
\[
 T_{xy}^*T_{xy}
 =B_y^*B_xB_x^*B_y
 \le\|T_{xx}\|\,B_y^*B_y
 =\|T_{xx}\|\,T_{yy}.
\]
Therefore, we obtain
\begin{align*}
 \sum_{d(x,y)\le R}\|T_{xy}\|_2^2 &=\sum_{d(x,y)\le R}\mathrm{Tr}(T_{xy}^*T_{xy}) \leq\sup_{x\in Y}\|T_{xx}\| \cdot \sum_{y\in Y}|B_Y(y,R)|\cdot \mathrm{Tr}(T_{yy}) \\
 &\leq\sup_{x\in Y}\|T_{xx}\| \cdot \sup_{x\in Y}|B(x,R)| \cdot \mathrm{Tr}(T).
\end{align*}
Dividing by $\mathrm{Tr}(T^2)$, we conclude (2).

Assume that there are infinitely many $n\in \mathbb{N}$ such that $T_n \neq 0$, and we restrict to that subsequence. Denote $C:=\sup_{n} \mathrm{Tr}(T_n)/\mathrm{Tr}(T_n^2)$, and let $\nu_n=\nu_{T_n}$ be the measures from Lemma~\ref{lem:trace coupling}. 

Given $\varepsilon>0$, there exists $R>0$ such that $\|\chi_A T \chi_B\|<\varepsilon/C$ whenever $d(A,B)>R$. For $A,B\subseteq X$ with $d(A,B)>R$, Lemma~\ref{lem:trace coupling}(1) gives

 \nu_n(A\times B) = \nu_n\left((A \cap X_n)\times (B \cap X_n)\right) \le C\|\chi_{A \cap X_n}T_n\chi_{B \cap X_n}\|
 =C\|\chi_{A \cap X_n}T\chi_{B \cap X_n}\|<\varepsilon.

\]
which goes to $0$ since $T$ is a ghost. Hence, $\nu_n$ provides an escaping quasi-local coupling. This contradicts the coarse embeddability of $X$ by Theorem~\ref{thm:esc and ql coupling}.

\label{lem:cutoff of quasi-local operators}
Given $T \in C^*_q(X;\mathcal{H}_0)$ and $\delta>0$, choose $R>0$ such that for any $A,B \subseteq X$ with $d(A,B) > R$, then $\|\chi_A T \chi_B\| < \delta$. Let $F$ be a finite subset of $X$ with $d(x,x') > R$ for any distinct points $x,x' \in F$. Then
\[
  \big\|\chi_F T \chi_F - \sum_{x\in F} \chi_{\{x\}}T\chi_{\{x\}} \big\|\leq 2\delta.
\]
<!-- end lem-7 -->
<!-- proof-17: body_md -->
For a map $\varepsilon: F \to \{-1,1\}$, set $U_\varepsilon:=\sum_{x\in F}\varepsilon(x) \chi_{\{x\}} \in \mathfrak{B}(\ell^2(F;\mathcal{H}_0))$. Averaging over all choices of \(\varepsilon\) gives
\begin{equation*}\label{EQ:averaging}
\frac1{2^{|F|}}
\sum_{\varepsilon\in\{-1,1\}^{F}}
U_\varepsilon(\chi_FT\chi_F)U_\varepsilon
=
\sum_{x\in F}\chi_{\{x\}}T\chi_{\{x\}}
\quad \text{since} \quad
\frac1{2^{|F|}}
\sum_{\varepsilon}
\varepsilon(x)\varepsilon(y) = 
\begin{cases}
0,& \text{if } x\neq y;\\
 1, & \text{if } x=y.
\end{cases}
\end{equation*}

Fixing a map \(\varepsilon\), set $F_+:=\{x\in F:\varepsilon(x)=1\}$ and $F_-:=\{x\in F:\varepsilon(x)=-1\}$. Then
\[
\|\chi_FT\chi_F-
U_\varepsilon(\chi_FT\chi_F)U_\varepsilon\|
= 2 \|\chi_{F_+}T\chi_{F_-} + \chi_{F_-}T\chi_{F_+}\| \leq 2\delta.
\]
Combining them together, we obtain
\[
\big\| \chi_FT\chi_F - \sum_{x\in F}\chi_{\{x\}}T\chi_{\{x\}} \big\| =
\big\|\frac1{2^{|F|}}\sum_{\varepsilon}\big(\chi_FT\chi_F-U_\varepsilon(\chi_FT\chi_F)U_\varepsilon\big)\big\|
\leq 2\delta.
\]
This finishes the proof.

<!-- lem-8: body_md -->
\label{lem:covering maps ghosts to ghosts}
Let $\{X_n\}_n, \{Y_n\}_n$ be two sequences of finite metric spaces of uniformly bounded geometry, and $(X,d_X), (Y,d_Y)$ be their coarse disjoint unions, respectively. For each $n$, let $\pi_n: X_n \to Y_n$ be a surjective map, and set $\pi:=\bigsqcup_n \pi_n: X \to Y$. Assume that $\pi$ is coarse. Let $\mathcal{H}_0$ be an infinite-dimensional separable Hilbert space, and $V: \ell^2(X;\mathcal{H}_0) \to \ell^2(Y;\mathcal{H}_0)$ be an isometry covering $\pi$. Then:
\begin{enumerate}
 \item For any $T \in C^*_q(X;\mathcal{H}_0)$, we have $\mathrm{Ad}_V(T) \in C^*_q(Y;\mathcal{H}_0)$;
 \item For each $n\in \mathbb{N}$, denote $s_n:=\inf\{d_X(x,x'):x\neq x' \text{ in } X_n \text{ with } \pi_n(x)=\pi_n(x')\}$, where $\inf\emptyset=+\infty$. If $s_n \to \infty$, then $\mathrm{Ad}_V(T) \in G_q(Y;\mathcal{H}_0)$ for any $T \in G_q(X;\mathcal{H}_0)$.
\end{enumerate}
<!-- end lem-8 -->
<!-- proof-18: body_md -->
(1) comes directly from \cite[Proposition 2.9]{BCZ23}, so we focus on (2). Let $T$ be a ghost in $C^*_q(X;\mathcal{H}_0)$, and we consider $S:=\mathrm{Ad}_V(T) = VTV^*$. 

Take $C\geq 0$ such that $\mathrm{supp}(V) \subseteq \{(y,x)\in Y\times X:d_Y(y,\pi(x))\leq C\}$. 
For \(y\in Y\), put $E_y:=\pi^{-1}(B_Y(y,C))$. Then for each $y\in Y$, we have 
\begin{equation}\label{EQ:chi y V}
\chi_{\{y\}}V=\chi_{\{y\}}V\chi_{E_y}. 
\end{equation}
Given $\varepsilon>0$, choose $R>0$ such that for any $A,B \subseteq X$ with $d(A,B) > R$, we have $\|\chi_A TT^* \chi_B\| < \frac{\varepsilon^2}{4N_Y(C)^2}$. Since $G_q(X;\mathcal{H}_0)$ is an ideal, $TT^*$ is ghost. Hence there is a finite subset $K \subseteq X$ such that 
\[
\|(TT^*)_{xx'}\| \leq \frac{\varepsilon^2}{2N_Y(C)^2} \quad  \text{for any} \quad (x,x') \notin K \times K.
\]
Choose $N_0 \in \mathbb{N}$ such that for any $n > N_0$, we have $X_n \cap K = \emptyset$, $s_n > R$ and for any $m\neq n$, $d_X(X_m, X_n) > R$ and $d_Y(Y_m, Y_n) > C$. 

Claim: For any $n > N_0$, any $y\in Y_n$ and $z\in Y$, we have $\|S_{yz}\| \leq \varepsilon$. To see this, fix $y\in Y_n$ and $z\in Y$. From \eqref{EQ:chi y V}, we have
\[
\|\chi_{\{y\}}VT\| = \|\chi_{\{y\}}V\chi_{E_y}T\| \leq \sum_{w\in B_Y(y,C)} \|\chi_{\{y\}}V\chi_{\pi^{-1}(w)}T\| \leq N_Y(C) \cdot \sup_{w\in B_Y(y,C)}\|\chi_{\pi^{-1}(w)}T\|.
\]
Since $d_Y(Y_m, Y_n) > C$ for any $m \neq n$, we obtain $E_y \subseteq X_n$. Hence for each $w\in B_Y(y,C)$, different points in $\pi^{-1}(w)$ have distance at least $s_n$. By Lemma~\ref{lem:cutoff of quasi-local operators}, we have
\begin{align*}
  \|\chi_{\pi^{-1}(w)} TT^* \chi_{\pi^{-1}(w)}\| &\leq \left\|\chi_{\pi^{-1}(w)} TT^* \chi_{\pi^{-1}(w)}- \sum_{x\in \pi^{-1}(w)} \chi_{\{x\}}TT^*\chi_{\{x\}} \right\| + \sup_{x\in \pi^{-1}(w)} \|(TT^*)_{xx}\| \\
  &\leq 2 \cdot \frac{\varepsilon^2}{4N_Y(C)^2} + \frac{\varepsilon^2}{2N_Y(C)^2} = \frac{\varepsilon^2}{N_Y(C)^2}.
\end{align*}
Therefore, we obtain
\[
\|S_{yz}\| = \|\chi_{\{y\}} VTV^* \chi_{\{z\}}\| \leq \|\chi_{\{y\}}VT\| \leq N_Y(C) \cdot \frac{\varepsilon}{N_Y(C)} = \varepsilon.
\]
This proves the claim. Applying the same argument for $T^*$, we conclude that $S$ is ghost. This finishes the proof.

<!-- lem-9: body_md -->
\label{lem:SC implies ghost}
Let $(X,d)$ be a discrete metric space of bounded geometry. Then $J_0(X)$ is a closed two-sided ideal of $C^*_u(X)$ and $\mathfrak{K}(\ell^2(X)) \subseteq J_0(X) \subseteq G_u(X)$. Moreover, for a positive operator $T\in C^*_u(X)$, $T \in J_0(X)$ \emph{if and only if} $M_\varphi(T)\in \mathfrak{K}(\ell^2(X))$ for any $\varphi\in\mathscr{P}_0(X)$.
<!-- end lem-9 -->
<!-- proof-20: body_md -->
Since the Schur multiplier $M_\varphi$ is linear, positive and contractive, it is routine to check that $J_0(X)$ is a norm-closed linear subspace in $C^*_u(X)$. For example, for any $T_1,T_2\in J_0(X)$ and $\varphi\in\mathscr{P}_0(X)$, positivity of $M_\varphi$ gives
\[

Since $\mathfrak{K}(\ell^2(X))$ is hereditary, we obtain $T_1+T_2 \in J_0(X)$. 

Now we check that $J_0(X)$ is an ideal in $C^*_u(X)$. For $T \in J_0(X)$ and $S \in C^*_u(X)$, we have $0\le M_\varphi ((ST)^*(ST))\le\|S\|^2 M_\varphi(T^*T)$, which proves that $ST \in J_0(X)$. To see that $TS \in J_0(X)$, first assume that $S$ is a diagonal operator. Then 
\[
M_\varphi((TS)^*(TS))=M_\varphi(S^*T^*TS)=S^*M_\varphi(T^*T)S\in\mathfrak{K}(\ell^2(X)).
\]
Next, we assume that $S=\chi_K$ for some partial translation $K$ and let $t: D \to R$ be the bijection, where $D,R \subseteq X$, such that $S$ is the graph of $t$. Define a kernel $\varphi'$ on $X \times X$ by 
\[

 \begin{cases}
 \varphi(t^{-1}(x), t^{-1}(y)), & \text{if } x,y\in R,\\
 \delta_{x=y}, & \text{otherwise}.
 \end{cases}
\]
Here $\delta_{x=y}$ equals $1$ if $x=y$, and $0$ if $x\neq y$. It is routine to check $\varphi' \in \mathscr{P}_0(X)$ and
\[
M_{\varphi}((TS)^*(TS)) =S^*M_{\varphi'}(T^*T)S \in \mathfrak{K}(\ell^2(X)).
\]
Any finite propagation operator has the form $S=\sum_{j=1}^n f_j V_{t_j}$ for $f_j \in \ell^\infty(X)$ and bijections $t_j$ as above. Taking norm closure, we obtain $TS \in J_0(X)$ for any $S \in C^*_u(X)$.

Since $M_\varphi$ preserves $\mathfrak{K}(\ell^2(X))$ for each $\varphi \in \mathscr{P}_0(X)$, we have $\mathfrak{K}(\ell^2(X)) \subseteq J_0(X)$. Take $\Delta:=\{(x,x): x\in X\}$ and $\varphi = \chi_\Delta$, which is in $\mathscr{P}_0(X)$. Given $T \in J_0(X)$, we have $M_\varphi(T^*T)_{xx} = \|T\delta_x\|^2$ and $M_\varphi(TT^*)_{xx} = \|T^*\delta_x\|^2$. Compactness shows that $\|T\delta_x\| \to 0$ and $\|T^*\delta_x\| \to 0$ as $x\to \infty$, where the second follows from $T^* \in J_0(X)$. Since $|A_{xy}|\leq \min\left\{\|A\delta_y\|,\|A^*\delta_x\|\right\}$ for any $x,y\in X$, we obtain that $T$ is a ghost.

Since $M_\varphi$ is a completely positive contraction, $0\leq M_\varphi(T)^2\leq M_\varphi(T^2)$, which shows that $M_\varphi(T)$ is compact.

 &=\omega(r_n)\langle v_n,M_nv_n\rangle \le\omega(r_n).
\end{align}
Furthermore, $\left|\mathrm{Tr}\left((S_n^{(R)}-S_n)D_n^{Y}\right)\right| \le\|S_n^{(R)}-S_n\|\cdot\|D_n^{Y}\|_1 \leq R^{-2}$.
Combining this with \eqref{eq:onsl-lower} and \eqref{EQ:upper control on trace}, we obtain
\[
 \omega(r_n) \geq \mathrm{Tr}(S_nD_n) \geq \mathrm{Tr}(S_n^{(R)}D_n) - R^{-2} \geq  C^{-1}-2R^{-2}-6\varepsilon
 >\frac1{2C}.
\]
This contradicts $\lim_{n\to \infty} \omega(r_n) = 0$, which finishes the proof.

<!-- lem-12: body_md -->
\label{lem:ONL and rank one}
A discrete metric space $(X,d)$ of bounded geometry has ONL \emph{if and only if} $X$ has ONSL such that each $D$ can be chosen of rank one.
<!-- end lem-12 -->
<!-- proof-27: body_md -->
Assume $X$ has ONL. Fixing $R>0$ and $\varepsilon \in (0,1)$, ONL provides $S>0$ such that for any $T \geq 0$ with $\mathrm{prop}(T) \leq R$, there is a unit vector $v \in \ell^2(X)$ with $\mathrm{diam}(\mathrm{supp} v) \leq S$ and $\langle Tv,v\rangle \geq (1-\varepsilon)\|T\|$. Let $D=P_v$ be the rank-one projection in $\mathfrak{B}(\ell^2(X))$ onto $\mathbb{C} v$. Then $\mathrm{Tr}(TD) = \langle Tv,v\rangle \geq (1-\varepsilon) \|T\|$. On the other hand, define
\[
 \omega(t)=
 \begin{cases}
 1,&0\le t\le S,\\
 0,&t>S.
 \end{cases}
\]
Note that $D_{xy}=v_x\overline{v_y}$, which shows that $|D_{xy}|=\sqrt{D_{xx}D_{yy}}$. Then $D_{xx}D_{yy}\ne0$ implies $x,y\in\mathrm{supp} v$. In this case, we have $d(x,y)\le S$ and hence, $|D_{xy}|\le \omega(d(x,y))\sqrt{D_{xx}D_{yy}}$.

Conversely, fixing $R>0$ and $\varepsilon\in (0,1)$, let $\omega$ be supplied by ONSL. Choose $S>0$ with $\omega(S)<1$. For a positive $T$ with $\mathrm{prop}(T)\le R$, denote the rank-one witness density operator by $D$. Since $\mathrm{Tr}(D)=1$ and $D$ is positive of rank one, $D=P_v$ the rank-one projection for some unit vector $v\in \ell^2(X)$. If $x,y\in\mathrm{supp} v$, then
\[
 \omega(d(x,y)) \geq  \frac{|D_{xy}|}{\sqrt{D_{xx}D_{yy}}}=1, 
\]
which implies that $d(x,y) \leq S$. Hence, $\mathrm{diam}(\mathrm{supp} v) \leq S$ and proves ONL.

<!-- prop-11: body_md -->
\label{prop:ONL and bounded rank}
A discrete metric space $(X,d)$ of bounded geometry has ONL \emph{if and only if} $X$ has ONSL with uniformly finite rank.
<!-- end prop-11 -->
<!-- proof-28: body_md -->
Necessity follows directly from Lemma~\ref{lem:ONL and rank one}, so we only focus on sufficiency. 
Fix $R>0$ and $\varepsilon \in (0,1)$. Applying ONSL with uniformly finite rank for $R$ and $\varepsilon$, we obtain $k \in \mathbb{N}$ and a function $\omega$. Choose $S>0$ such that $\omega(S)<\frac1k$.

Let $T\in\mathfrak{B}(\ell^2(X))$ be a positive operator with $\mathrm{prop}(T)\le R$. Choose a density operator $D$ supplied by ONSL with rank at most $k$. Then there are vectors $v_x\in\mathbb{C}^k$ such that $D_{xy}=\langle v_x,v_y\rangle_{\mathbb{C}^k}$ for any $x,y\in X$. Set $Z:=\{x\in X:D_{xx}>0\}$, and define $u_x:=v_x/\|v_x\|$ for $x\in Z$. Then $\|u_x\|=1$ and we have
\[
 |\langle u_x,u_y\rangle|
 =\frac{|D_{xy}|}{\sqrt{D_{xx}D_{yy}}}
 \le\omega(d(x,y)).
\]
Claim: $Z$ cannot contain $k+1$ points whose pairwise distances are
greater than $S$.  Indeed, if $x_1,\ldots,x_{k+1}\in Z$ had this property,
then
\[
 |\langle u_{x_i},u_{x_j}\rangle| \leq  \omega(d(x_i,x_j)) \leq \omega(S) <\frac1k \quad \text{for} \quad i\neq j.
\]
Then the Gram matrix $G=\left(\langle u_{x_i},u_{x_j}\rangle\right)_{i,j=1}^{k+1}$ has diagonal entries equal to $1$, while in each row the sum of the absolute values of the off-diagonal entries is strictly less than $1$.  Hence $G$ is strictly diagonally dominant and therefore, invertible. This would make $u_{x_1},\ldots,u_{x_{k+1}}$ linearly independent in $\mathbb C^k$, which is a contradiction.

Choose a maximal family $z_1,\ldots,z_m\in Z$ with pairwise distances greater than $S$. The claim above gives $m\le k$, while maximality shows $Z\subseteq\bigcup_{j=1}^m B(z_j,S)$. Due to bounded geometry, we have
\begin{equation}\label{EQ:bdd geom}
 |Z|\le kN_X(S) <\infty.
\end{equation}
Since $D \geq 0$, we have $|D_{xy}|^2\le D_{xx}D_{yy}$. Hence $D_{xx}=0$ forces $D_{xy}=0$ for every $y$, which implies that $D= \chi_ZD\chi_Z$. 

Decompose $Z=Z_1\sqcup\cdots\sqcup Z_q$ into its $R$-connected components, \emph{i.e.}, the connected components of the graph on \(Z\) where \(x\) and \(y\) are connected whenever \(d(x,y)\le R\). Then $d(Z_i,Z_j)>R$ for $i\neq j$.  Moreover, \eqref{EQ:bdd geom} shows that each $Z_i$ has at most $kN_X(S)$ points and hence, 
\begin{equation}\label{EQ:diam of Zi}
 \mathrm{diam}(Z_i)\le (kN_X(S) -1)R.
\end{equation}
Let $T_i:=\chi_{Z_i}T\chi_{Z_i}$ and $D_i:=\chi_{Z_i}D\chi_{Z_i}$. 
Since $\mathrm{prop}(T)\le R$, we have $\chi_Z T \chi_Z = \sum_{i=1}^q T_i$. Then we obtain
\[
 \mathrm{Tr}(TD)=\mathrm{Tr}(\chi_ZT\chi_ZD\chi_Z) = \sum_{i=1}^q\mathrm{Tr}(T_iD_i).
\]
Since $T_i,D_i\ge0$, we have $\mathrm{Tr}(T_iD_i)\le\|T_i\|\mathrm{Tr}(D_i)$ and hence, 
\[
 (1-\varepsilon)\|T\| \le\mathrm{Tr}(TD) \le\Big(\max_i\|T_i\|\Big)\sum_i\mathrm{Tr}(D_i) =\max_i\|T_i\|.
\]
Thus, there exists $i$ such that $\|T_i\|\ge(1-\varepsilon)\|T\|$. 
Since $T_i$ is positive on the finite-dimensional space $\ell^2(Z_i)$, there
is a unit vector $v\in\ell^2(Z_i)$ such that
\[
 \langle Tv,v \rangle
 =\langle T_iv,v\rangle
 =\|T_i\|
 \ge(1-\varepsilon)\|T\|.
\]
By \eqref{EQ:diam of Zi}, we have $\mathrm{diam}(\mathrm{supp} v) \leq (kN_X(S) -1)R$, which concludes ONL.
