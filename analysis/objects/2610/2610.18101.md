<!-- lemma-5: body_md -->
\label{L:2.2}
Let $\phi  \in$ Hom$(G, {\text{Aut}}_k(K))$ be a Galois action. Then the orthogonal idempotents $\{\epsilon_g\}_{g\in G}\subset K \otimes_k K$ satisfy

\end{equation}

\epsilon_g(K\otimes_k K) = \epsilon_{g}(K \otimes_k 1) = \epsilon_{g}(1 \otimes_k K),\ \ \forall g \in G.
\end{equation}

\epsilon_g'(x \otimes_k 1) = \epsilon_g'(1 \otimes_k \phi(g^{})(x)),\ \forall x \in K.
\end{equation}

<!-- definition-5: body_md -->
Let $G$ be a group with identity $e$ and let $K$ be a commutative ring containing a subring $k.$
A $k$-algebra is \textit{strongly $G$-graded} if it admits an additive decomposition
\begin{equation}\label{eq:D:1}
A_G = \bigoplus_{g \in G} A_g
\end{equation}
such that for every $g,h \in G$
\begin{equation}\label{AgAh}A_g\cdot A_h=A_{gh}.\end{equation}

<!-- definition-6: body_md -->
\label{cliffextdef}                       
(\textit{i}) A $k$-algebra homomorphism
$\psi_G : A_G \rightarrow A'_G$ between two strongly $G$-graded $k$-algebras
\begin{equation}\label{AA'}
A_G = \bigoplus_{g \in G}A_g \text{ and }A'_G =  \bigoplus_{g \in G}A_g'
\end{equation}
                                                      
is a \textit{graded homomorphism} if $\psi_G(A_g) = A'_g$ for every $g \in G$.
The graded algebras \eqref{AA'} are \textit{graded isomorphic}, and written $A_G\stackrel{G}\cong A'_G$, if there exists a graded isomorphism between them.

extension} of $R$ is a pair $(A_G,\iota)$ where $A_G$ is a strongly
$G$-graded $k$-algebra \eqref{eq:D:1} and $\iota : R \to A_G$ is a
$k$-algebra embedding with $\iota(R) = A_e$.

system extensions $(A_G,\iota)$ and $(A'_G,\iota')$ of $R$ are equivalent if
there exists a graded isomorphism $\psi_G : A_G \xrightarrow{G} A'_G $ such
that $\psi_G\circ \iota = \iota'$. Denote the equivalence class of $(A_G,\iota)$  by $[(A_G,\iota)]$.

Let $M$ be a left $K_{\phi}G$-module such that ${\text{res}}([M])=[K]$.
Then there exists an isomorphism $\varphi:K\xrightarrow{\cong} M$ of $K$-modules. For every $g\in G$ let
$$\varphi_g:=\varphi^{-1}(u_g\star\varphi(1))\in K.$$

\textit{Step 1.} For every $g\in G$, the element $\varphi_g$ is invertible. Moreover,\begin{eqnarray}\label{cocy}\pi_{{\varphi}}=\pi_{M,\varphi}:\begin{array}{ccc}
G&\to& K^*\\
g&\mapsto &\varphi_g
\end{array}\end{eqnarray}
is a 1-cocycle, where the $G$-module structure of $K^*$ is via $\phi$.

\textit{Proof of Step 1.}
For every $g,h\in G$
\begin{eqnarray}\label{1cc}\begin{array}{rl}
\varphi_{gh}&=\varphi^{-1}(u_{gh}\star\varphi(1))=\varphi^{-1}(u_gu_h\star\varphi(1))=\varphi^{-1}(u_g\star(u_h\star\varphi(1)))\\
&=\varphi^{-1}(u_g\star\varphi(\varphi_h))=\varphi^{-1}(u_g\star(\varphi_h\cdot\varphi(1)))=\varphi^{-1}(u_g\cdot \varphi_h\star(\varphi(1)))\\
&=\varphi^{-1}(g(\varphi_h)\cdot u_g\star(\varphi(1)))=g(\varphi_h)\cdot \varphi^{-1}(u_g\star(\varphi(1)))=g(\varphi_h)\cdot \varphi_g.
\end{array}\end{eqnarray}
Since $\varphi_e=1$, putting $h=g^{-1}$ in \eqref{1cc} we deduce that $\varphi_g\in K^*$ for every $g\in G$.
In fact, \eqref{1cc} says that $g\mapsto \varphi_g$
                                                                                                                                        
is a 1-cocycle.

\textit{Step 2.} With the notation \eqref{cocy}, the rule $[M]\mapsto [\pi_{M,\varphi}]$ determines a well-defined map $\ker({\text{res}})\to H^1(G,K^*_{\phi})$.

\textit{Proof of Step 2.} We show that the cohomology class
$[\pi_{M,{\varphi}}]\in H^1(G,K^*_{\phi})$ neither depends on the representative $M\in [M]$, nor on the isomorphism $\varphi$.
Indeed, let $M$ and $\tilde{M}$ be left $K_{\phi}G$-modules via the actions $``\star"$ and $``\tilde{\star}"$ respectively, such that $[M]=[\tilde{M}] \in \ker({\text{res}})$,
and let $\lambda:M \stackrel{\cong}\rightarrow \tilde{M}$ be a $K_{\phi}G$-module isomorphism.
In particular, for every $g\in G$ and $m\in M$
\begin{equation}\label{lamb}
\lambda(u_g\star m)=u_g\tilde{\star}\lambda(m).
\end{equation}
                                                                                                  
Let
\begin{equation}\label{varphitilde}
{\varphi}:K\xrightarrow{\cong} M,\ \ \ \tilde{\varphi}:K\xrightarrow{\cong} \tilde{M}\end{equation}
be $K$-module isomorphisms determining the corresponding 1-cocycles
$\pi_{M,{\varphi}}$ and $\pi_{\tilde{M},\tilde{\varphi}}$ as in \eqref{cocy}.
We claim that these 1-cocycles defer by the 1-coboundary
$$g\mapsto g(x)\cdot x^{-1},\text{  where  } x:=\varphi^{-1}\circ\lambda^{-1}\circ\tilde{\varphi}(1)\in K^*.$$
First, since $\tilde{\varphi}^{-1}\circ\lambda\circ \varphi$ is a $K$-morphism we have
$$x\cdot(\tilde{\varphi}^{-1}\circ\lambda\circ \varphi( 1))=\tilde{\varphi}^{-1}\circ\lambda\circ \varphi( x\cdot 1)=\tilde{\varphi}^{-1}\circ\lambda\circ \varphi( x)
=\tilde{\varphi}^{-1}\circ\lambda\circ \varphi(\varphi^{-1}\circ\lambda^{-1}\circ\tilde{\varphi}(1))=1,$$
hence $\tilde{\varphi}^{-1}\circ\lambda\circ \varphi( 1)=x^{-1}.$
Next, for every $g\in G$
$$\begin{array}{rl}
\tilde{\varphi}_g&=\tilde{\varphi}^{-1}(u_g\tilde{\star}\tilde{\varphi}(1))=\tilde{\varphi}^{-1}(u_g\tilde{\star}\lambda\circ\varphi\circ\varphi^{-1}\circ\lambda^{-1}\circ\tilde{\varphi}(1))\\
&=\tilde{\varphi}^{-1}(u_g\tilde{\star}\lambda\circ\varphi((\varphi^{-1}\circ\lambda^{-1}\circ\tilde{\varphi}(1))\cdot 1))
=\tilde{\varphi}^{-1}(u_g\tilde{\star}\lambda(\varphi(x\cdot 1)))=\cdots
\end{array}$$
The element $x\in K$ can be pulled out of the $K$-morphism $\varphi$, thus
$$\begin{array}{rl}
\cdots&=\tilde{\varphi}^{-1}(u_g\tilde{\star}\lambda(x\cdot\varphi( 1)))=\tilde{\varphi}^{-1}(u_g\cdot x\tilde{\star}\lambda(\varphi( 1)))
                                              
=\tilde{\varphi}^{-1}(g(x)\cdot u_g\tilde{\star}\lambda(\varphi( 1)))=\cdots
\end{array}$$
By \eqref{lamb}
$$\begin{array}{rl}
\cdots&=\tilde{\varphi}^{-1}(g(x)\cdot \lambda(u_g\star\varphi( 1)))=g(x)\cdot\tilde{\varphi}^{-1}( \lambda(u_g\star\varphi( 1)))=g(x)\cdot\tilde{\varphi}^{-1}( \lambda(\varphi( \varphi_g)))\\
&=g(x)\cdot\tilde{\varphi}^{-1}( \lambda(\varphi( \varphi_g\cdot 1)))=\cdots
\end{array}$$
Again, pull $\varphi_g\in K$ out of the $K$-morphism $\tilde{\varphi}^{-1}\circ\lambda\circ \varphi$ and apply the above description of $x^{-1}$ to obtain
$$\cdots=g(x)\cdot\varphi_g\cdot(\tilde{\varphi}^{-1}\circ \lambda\circ \varphi( 1))=g(x)\cdot\varphi_g\cdot x^{-1}.$$
We conclude that $\tilde{\varphi}_g\cdot\varphi_g^{-1}=g(x)\cdot x^{-1}$
and hence we are done.

\textit{Step 3.}
The well-defined map
\begin{equation}\label{Pi-1}\begin{array}{ccc}\ker({\text{res}})&\to& H^1(G,K_{\phi}^*)\\~[M]&\mapsto &[\pi_{M,\varphi}]\end{array}
\end{equation} is a group homomorphism.

\textit{Proof of Step 3.}
Let $M$ and $\tilde{M}$ be (not necessarily isomorphic) left $K_{\phi}G$-modules such that $[M],[\tilde{M}] \in \ker({\text{res}})$.
Let \eqref{varphitilde} be $K$-module isomorphisms determining the corresponding 1-cocycles
$\pi_{M,{\varphi}}$ and $\pi_{\tilde{M},\tilde{\varphi}}$ as in \eqref{cocy}.
Let us compute the 1-cocycle
$$\pi_{M\otimes_K\tilde{M},\hat{\varphi}}:G\to K^*$$
with respect to the $K$-module isomorphism
\begin{eqnarray}\label{cocytens}\hat{\varphi}:\begin{array}{ccc}
K&\to& M\otimes_K\tilde{M}\\
x&\mapsto &\varphi(1)\otimes_K\tilde{\varphi}(x)
\end{array}.\end{eqnarray}
By the rule in step 1 (see \eqref{cocy}) applied on \eqref{cocytens}, and the $K_{\phi}G$-action \eqref{tensrepr} on the tensor product, we have for every $g\in G$
$$\begin{array}{rl}
\pi_{M\otimes_K\tilde{M},\hat{\varphi}}(g)&=\hat{\varphi}_g=\hat{\varphi}^{-1}(u_g(\hat{\varphi}(1)))=\hat{\varphi}^{-1}(u_g(\varphi(1)\otimes_K\tilde{\varphi}(1)))\\
&=\hat{\varphi}^{-1}(u_g*\varphi(1)\otimes_Ku_g\tilde{*}\tilde{\varphi}(1))=\hat{\varphi}^{-1}(\varphi(\varphi_g)\otimes_K \tilde{\varphi}(\tilde{\varphi}_g))\\
&=\hat{\varphi}^{-1}(\varphi(\varphi_g\cdot 1)\otimes_K \tilde{\varphi}(\tilde{\varphi}_g))=\varphi_g\cdot\hat{\varphi}^{-1}(\varphi( 1)\otimes_K \tilde{\varphi}(\tilde{\varphi}_g))\\
&=\varphi_g\cdot\tilde{\varphi}_g=\pi_{M,{\varphi}}(g)\cdot\pi_{\tilde{M},\tilde{\varphi}}(g).
\end{array}$$
In particular, the $H^1(G,K^*_{\phi})$-classes $[\pi_{M\otimes_K\tilde{M},\hat{\varphi}}]$ and $[\pi_{M,{\varphi}}]\cdot[\pi_{\tilde{M},\tilde{\varphi}}]$ are equal, proving the homomorphism property.

\textit{Step 4. The map \eqref{Pi-1} is an isomorphism.}

\textit{Proof of Step 4.}
Here is a construction of the inverse of the group homomorphism \eqref{Pi-1}.
We endow the trivial element in Mod$(K)$, that is free $K$-module of rank 1, with a left $K_{\phi}G$-action for
a given 1-cocycle $\pi:G\to K^*_{\phi}$ as follows.
                                                                                                                                              
\begin{equation}\label{act*}(\sum_{g\in G}x_gu_g)\star x:=\sum_{g\in G}x_g\cdot\pi(g)\cdot g(x),\ \ \ \ \sum_{g\in G}x_gu_g\in K_{\phi}G,\ x\in K_{\pi}.\end{equation}
                                                                 
Then it is not hard to verify, using the 1-cocycle condition, that \eqref{act*} indeed describes a left $K_{\phi}G$-module on $K$.
Denote this module by $K_{\pi}$.

Furthermore, if $[\pi]=[\hat{\pi}]\in H^1(G,K^*_{\phi})$ then the corresponding $K_{\phi}G$-modules $K_{\pi}$ and $K_{\hat{\pi}}$ are isomorphic as follows.
The fact that the 1-cocycles are cohomologically equivalent says that there exists $y\in K^*$ such that
$$\pi(g)=y^{-1}\cdot g(y)\cdot \hat{\pi}(g),\ \ \ \forall g\in G.$$
Then it is easily checked that $\begin{array}{ccc}K_{\pi}&\to &K_{\hat{\pi}}\\ x&\mapsto& y\cdot x\end{array}$ is a $K_{\phi}G$-module isomorphism.
In other words, there is a well-defined map
\begin{equation}\label{piphi}
H^1(G,K_{\phi}^*)\to \ker({\text{res}}).\end{equation}
To conclude the proof, the fact that \eqref{piphi} is indeed the inverse of the group homomorphism \eqref{Pi-1} is left to the reader.

<!-- lemma-39: body_md -->
Let $\phi \in$ Hom$(G, {\text{Aut}}_k(K))$ be a Galois action. Then with the above notation,
\begin{eqnarray}\label{prepsi}\begin{array}{ccc}
{\text{Mod}} (K_{\phi}G) &\rightarrow& {\text{Mod}}(k)\\
~[M] &\mapsto& [M^{\phi}]\end{array}.\end{eqnarray}
                                                   
is a well-defined isomorphism of monoids.
<!-- end lemma-39 -->
<!-- proof-24: body_md -->
The fact that \eqref{prepsi} is well-defined is clear.
Next, the Galois setting yields an isomorphism of $K$-modules (see \cite[IV(2.1)]{FDEI})
\begin{equation}\label{theta2}K \otimes_k M^{\phi} \cong M. \end{equation}
By \eqref{theta2} we deduce that for every $[M_1],[M_2] \in{\text{Mod}}(K_{\phi}G)$
                                                                          
\begin{eqnarray}\label{eqarr}\begin{array}{rl}
K \otimes_k M_1^{\phi} \otimes_k M_2^{\phi} \cong& K \otimes_k M_1^{\phi} \otimes_K K \otimes_k M_2^{\phi}\\
\cong& M_1 \otimes_K M_2 \\
\cong& K \otimes_k (M_1\otimes_K M_2)^{\phi}.
\end{array}\end{eqnarray}
By the Galois assumption, $K$ is a $k$-progenerator. This property allows us to cancel out $K\otimes_k-$ on both hands of \eqref{eqarr} \cite[I, Prop. 3.3]{FDEI} so as to obtain a $k$-module isomorphism
$$(M_1\otimes_K M_2)^{\phi} \cong M_1^{\phi} \otimes_k M_2^{\phi}, $$
thereby proving that \eqref{prepsi} is a group homomorphism. To complete the proof, here is an inverse of \eqref{prepsi}
\begin{eqnarray}\label{prepsiinv}                  
\begin{array}{ccc}
{\text{Mod}} (k) &\rightarrow& {\text{Mod}}(K_{\phi}G)\\
~[P] &\mapsto& [K\otimes_kP]\end{array},\end{eqnarray}
where the the $K_{\phi}G$-module $K \otimes_k P$, whose $G$-module action is given by
$$u_g*(x \otimes_k p) :=\ \phi(g)(x)\otimes_k p, \ \ \ x \in K, p \in P,g \in G.$$
Indeed, \eqref{theta2} ascertains that \eqref{prepsiinv} is a left inverse of \eqref{prepsi}.
Conversely, by putting $M:=K \otimes_k P,\ \ [P]\in {\text{Mod}} (k)$ in \eqref{theta2}, we have
$$K \otimes_k (K \otimes_k P)^{\phi} \cong K\otimes_k P, \ \  [P]\in {\text{Mod}} (k).$$
By the $k$-progenerator property of $K$, this further yields $(K \otimes_k P)^{\phi} \cong P$, proving that \eqref{prepsiinv} is also a right inverse of \eqref{prepsi}, and hence the claim.

\begin{equation}\label{Miyash}

\begin{equation}\label{pass}

\end{equation}
In particular, for every $h\in G$, $\epsilon_h$ is full, that is
\begin{equation}\label{prefull}

We prove the left equalities in \eqref{Miyash} and in \eqref{pass}. The right ones are proven in a similar way.
                                     
Let us first show that $\sum\limits_i(a^g_i \otimes_k 1)\epsilon_h(b^g_i \otimes_k 1)$
is an idempotent which satisfies an appropriate equation of the form \eqref{theorth}. We compute
$$\sum\limits_i(a^g_i \otimes_k 1)\epsilon_h(b^g_i \otimes_k 1)\sum\limits_j(a^g_j \otimes_k 1)\epsilon_h(b^g_j \otimes_k 1)=

$$\cdots=\sum\limits_{i,j}(a^g_i \otimes_k 1)(b^g_i\otimes_k 1)(a^g_j \otimes_k 1)\epsilon_h^2(b^g_j \otimes_k 1)=\sum\limits_j(a^g_j \otimes_k 1)\epsilon_h(b^g_j \otimes_k 1).$$
                                                                                                                                                                    
Therefore, $\sum\limits_i(a^g_i \otimes_k 1)\epsilon_h(b^g_i \otimes_k 1)\in K \otimes_k K$ is an idempotent for every $g,h\in G$.
We next check condition \eqref{theorth}. For $x \in K$ we have
$$\sum\limits_i(a^g_i \otimes_k 1)\epsilon_h(b^g_i \otimes_k 1)(x\otimes_k 1) = \sum\limits_i(a^g_i \otimes_k 1)\epsilon_h(b^g_ix\otimes_k 1)=\cdots$$

\end{array}$$

Let us compute the kernel of $\rho'_{\phi}$.
By \cite[Prop. 5.3]{AMGO}, $[A_G]\in\ker(\rho'_{\phi})$ if and only if $A_G \stackrel{G}{\cong}$ End$_k(P)$  for some faithful progenerator $k$-module $P$.
