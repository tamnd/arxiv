<!-- asm-1: body_md -->
\label{asm:func}Regarding $f, g$ and $F$, we assume:\\
    $\bullet$ $f(\mathbf{w}, \mathbf{y})$ is $C_f$-Lipschitz in $\mathbf{y}$ and jointly $L_f$-smooth in terms of $(\mathbf{w}, \mathbf{y})$\\
    $\bullet$ $g(\mathbf{w}, \mathbf{y})$ is jointly $L_g$-smooth and $\mu_g$-strongly convex in terms of $\mathbf{y}$ for any $\mathbf{w}$\\
    $\bullet$ $f$ is twice continuously differentiable, and $\nabla^2 g(\mathbf{w}, \mathbf{y})$ is jointly $Q_g$-Lipschitz continuous.\\
    $\bullet$ $F(\mathbf{w})$ is lower bounded, i.e., $\inf_{\mathbf{w}\in\mathbb{R}^{d_w}} F(\mathbf{w})> -\infty$
<!-- end asm-1 -->
<!-- asm-2: body_md -->
\label{asm:oracle}There exist stochastic gradients of $f$ and $g$ that are unbiased and have bounded variance $\sigma_f^2, \sigma_g^2$, respectively, i.e.
    \begin{align}
        \mathbf{E}_{\xi_f} [\nabla f(\mathbf{w}, \mathbf{y}, \xi_f)] = \nabla f(\mathbf{w}, \mathbf{y})&, \quad\mathbf{E}_{\xi_g} [\nabla g(\mathbf{w}, \mathbf{y}, \xi_g)] = \nabla g(\mathbf{w}, \mathbf{y})\nonumber\\
        \mathbf{E}_{\xi_f}\left\|\nabla f(\mathbf{w}, \mathbf{y}, \xi_f) - \nabla f(\mathbf{w}, \mathbf{y})\right\|^2 \leq \sigma_f^2&, \quad \mathbf{E}_{\xi_g}\left\|\nabla g(\mathbf{w}, \mathbf{y}, \xi_g) - \nabla g(\mathbf{w}, \mathbf{y})\right\|^2 \leq \sigma_g^2.
    \end{align}
<!-- end asm-2 -->
<!-- asm-3: body_md -->
\label{asm:mss}  $g$ is mean-square smooth in $\mathbf{y}$, i.e.
\begin{align}
    \mathbf{E}_{\xi_g}\left\|\nabla g(\mathbf{w}, \mathbf{y}_1, \xi_g) - \nabla g(\mathbf{w}, \mathbf{y}_2, \xi_g)\right\|^2 \leq \tilde L_g^2 \left\|\mathbf{y}_1 - \mathbf{y}_2\right\|^2.
\end{align}

<!-- lem-1: body_md -->
\label{lem:penal_property}
Under Assumption~\ref{asm:func}, for $\lambda\ge 2L_f/\mu_g$, it holds for $\forall \mathbf{w}\in\mathbb{R}^{d_w}$that
\begin{align}
                                                                                                   
&|\mathcal{L}^*_\lambda(\mathbf{w}) - F(\mathbf{w})| = O\left(\frac{\bar{\ell}\kappa^2}{\lambda}\right),
\quad \left\|\nabla\mathcal{L}^*_\lambda(\mathbf{w})-\nabla F(\mathbf{w})\right\| = O\left(\frac{\bar{\ell}\kappa^3}{\lambda}\right),
\label{eq:obj_grad_surrogate_approx}\\
&\left\|\nabla\mathcal{L}^*_\lambda(\mathbf{w}_1) - \nabla\mathcal{L}^*_\lambda(\mathbf{w}_2)\right\| \leq L_*\left\|\mathbf{w}_1-\mathbf{w}_2\right\|, \text{ where } L_*=O(\bar{\ell}\kappa^3).
\label{eq:sm_surrogate}
\end{align}

<!-- thm-1: body_md -->
\label{thm:std}
Under Assumptions~\ref{asm:func} and~\ref{asm:oracle}, let the batch size be $B=O(1)$. With a warm-up stage for $\mathbf{y}$ such that the initialization satisfies the condition that $\mathbf{E}[\|\mathbf{y}_0 - \mathbf{y}^*_\lambda(\mathbf{w}_0)\|^2]\leq O(1/\kappa)$, by setting
\[
    \lambda=\Theta\left(\frac{\kappa^3}{\epsilon}\right),\qquad
    \beta=\Theta\left(\frac{\epsilon^4}{\kappa^6}\right),\qquad
    \gamma=\Theta\left(\frac{\epsilon^4}{\kappa^7}\right),\qquad
    \eta=\Theta\left(\frac{\epsilon^4}{\kappa^{11}}\right),
\]
Algorithm~\ref{alg:s2f2bo} returns $\mathbf{w}_\tau$ satisfying
\[
    \mathbf{E}\left\|\nabla F(\mathbf{w}_\tau)\right\|^2\leq\epsilon^2
\]
within $T=O(\kappa^{11}\epsilon^{-6})$ stochastic first-order oracle calls, where $\tau$ is uniformly sampled from $\{0, 1, \cdots, T-1\}$.
<!-- end thm-1 -->
<!-- head-1: body_md -->
Note that the involved optimization problem in~(\ref{def:penal_ll_sol}) that define $\mathbf{y}_\lambda^*(\mathbf{w}_0)$ is strongly convex, hence the warm-up stage is cheap. For example, running a simple SGD for $T_0=O(\kappa^3)$-iterations for solving $\min_{\mathbf{y}}\mathcal{L}_\lambda(\mathbf{w}_0,\mathbf{y})$ will find a solution $\mathbf{y}_0$ such that $\mathbf{E}[\|\mathbf{y}_0 - \mathbf{y}^*_\lambda(\mathbf{w}_0)\|^2]\leq O(1/(\mu^2_g T_0))=O(1/\kappa)$.
<!-- end head-1 -->
<!-- thm-2: body_md -->
\label{thm:mss}
Under Assumptions~\ref{asm:func},~\ref{asm:oracle} and~\ref{asm:mss}, let the batch size be $B=O(1)$. With the same warm-up initialization and by setting
\[
    \lambda=\Theta\left(\frac{\kappa^3}{\epsilon}\right),\qquad
    \beta=\Theta\left(\frac{\epsilon^2}{\kappa^2}\right),\qquad
    \gamma=\Theta\left(\frac{\epsilon^2}{\kappa^5}\right),\qquad
    \eta=\Theta\left(\frac{\epsilon^2}{\kappa^9}\right),
\]
Algorithm~\ref{alg:s2f2bo} returns $\mathbf{w}_\tau$ satisfying
\[
    \mathbf{E}\left\|\nabla F(\mathbf{w}_\tau)\right\|^2\leq\epsilon^2
\]
within $T=O(\kappa^9\epsilon^{-4})$ stochastic first-order oracle calls, where $\tau$ is uniformly sampled from $\{0, 1, \cdots, T-1\}$.
<!-- end thm-2 -->
<!-- head-2: body_md -->
To the best of our knowledge, this rate also has the best dependence on $\kappa$ compared with that of other fully first-order methods.

\label{lem:ll_sol_prox}
    Under assumption~\ref{asm:func}, for $\lambda\geq\frac{2L_f}{\mu_g}$, it holds that
    \begin{align}\label{eq:ll_sol_prox}
        \left\|\mathbf{y}^*(\mathbf{w}) - \mathbf{y}_\lambda^*(\mathbf{w})\right\|\leq \frac{C_f}{\mu_g\lambda}
    \end{align}
<!-- end lem-2 -->
<!-- proof-1: body_md -->
It holds from definition of $\mathbf{y}_\lambda^*(\mathbf{w})$ that $\nabla_2 f(\mathbf{w}, \mathbf{y}_\lambda^*(\mathbf{w})) + \lambda \nabla_2 g(\mathbf{w}, \mathbf{y}_\lambda^*(\mathbf{w})) = \textbf{0}$. Then we have
    \begin{align*}
        \left\|\mathbf{y}^*(\mathbf{w}) - \mathbf{y}_\lambda^*(\mathbf{w})\right\|
        \leq \frac{1}{\mu_g}\left\|\nabla_2 g(\mathbf{w}, \mathbf{y}_\lambda^*(\mathbf{w}))\right\|
        = \frac{1}{\mu_g\lambda}\left\|\nabla_2 f(\mathbf{w}, \mathbf{y}_\lambda^*(\mathbf{w}))\right\|
        \leq \frac{C_f}{\mu_g\lambda}  
    \end{align*}
    where the first inequality holds from strongly convexity of $g(\mathbf{w}, \cdot)$.

<!-- lem-3: body_md -->
\label{lem:y_lip}
    Under assumption~\ref{asm:func}, for $\lambda\geq \frac{2L_f}{\mu_g}$, $\mathbf{y}_\lambda^*(\cdot)$ is $C_{y*} = \frac{2(L_f+\lambda L_g)}{\lambda \mu_g} = O(\kappa)$-lipschitz.
<!-- end lem-3 -->
<!-- proof-2: body_md -->
It holds from the definition of $\mathbf{y}_\lambda^*(\mathbf{w})$ that $\nabla_2 \mathcal{L}_\lambda(\mathbf{w}, \mathbf{y}_\lambda^*(\mathbf{w})) = \textbf{0}$. Differentiate LHS w.r.t. $\mathbf{w}$, we have 
\begin{align*}
    \nabla_{21}^2 \mathcal{L}_\lambda(\mathbf{w}, y^*(\mathbf{w})) + \nabla y^*(\mathbf{w})\nabla_{22}^2 \mathcal{L}_\lambda(\mathbf{w}, y^*(\mathbf{w}))= \textbf{0}.
\end{align*}
Rearranging the term and multiplying both side by $\nabla_{22}^2 \mathcal{L}_\lambda^*(\mathbf{w}, y^*(\mathbf{w}))^{-1}$ yields
\begin{align}\label{eq:grad_y_lamstar}
    \nabla y^*_\lambda(\mathbf{w}) = -\left(\frac{1}{\lambda}\nabla^2_{21}f(\mathbf{w}, y^*_\lambda(\mathbf{w})) + \nabla^2_{21}g(\mathbf{w}, y^*_\lambda(\mathbf{w}))\right)\left(\frac{1}{\lambda}\nabla^2_{22}f(\mathbf{w}, y^*_\lambda(\mathbf{w})) + \nabla^2_{22}g(\mathbf{w}, y^*_\lambda(\mathbf{w}))\right)^{-1}
\end{align}
It finishes the proof by noting that $\nabla^2_{22}f(\mathbf{w}, y^*_\lambda(\mathbf{w}))\preceq L_f\mathbf{I}, \nabla^2_{21}f(\mathbf{w}, y^*_\lambda(\mathbf{w}))\preceq L_f\mathbf{I}$ and $\nabla^2_{22}g(\mathbf{w}, y^*_\lambda(\mathbf{w}))\succeq \mu_g\mathbf{I}, \nabla^2_{21}g(\mathbf{w}, y^*_\lambda(\mathbf{w}))\preceq L_g\mathbf{I}$

\label{lem:s_lip}
Under assumption~\ref{asm:func}, if $\lambda\geq \frac{2L_f}{\mu_g}$, we have that for any $t$:

\label{lem:y_recur}

\end{align}

    &\leq (1-\frac{\gamma\mu_y}{2})\left\|\mathbf{y}_t - \mathbf{y}_{\lambda, t-1}^*\right\|^2 + \frac{2C_{y*}^2\eta^2}{\gamma\mu_y}\left\|\mathbf{v}_{t}\right\|^2 + \frac{\gamma^2\sigma_y^2}{B}  
\end{align}
where $\mu_y = \mu_g - \frac{L_f}{\lambda}$ and $L_y = \frac{L_f}{\lambda} + L_g$ are the strongly convexity coefficient and the smoothness coefficient of $\mathcal{L}_y^\lambda(\mathbf{w}_t, \cdot) = \frac{1}{\lambda}f(\mathbf{w}_t, \cdot) + g(\mathbf{w}_t, \cdot)$. \label{lem:cross_gradnormsq_lb}
Under assumption~\ref{asm:func}, we have that the following two inequality hold:
\begin{align}
    \langle G_t^s, \mathbf{s}_t - \mathbf{s}_t^*\rangle &\geq \frac{\mu_g}{2}\left\|\mathbf{s}_t - \mathbf{s}_t^*\right\|^2 - \frac{1}{\mu_g} \left(\frac{1}{\lambda^2} L_f^2 + Q_g^2r_\lambda^2\right)\left\|\mathbf{y}_t - \mathbf{y}_\lambda^*(\mathbf{w}_t)\right\|^2\label{eq:s_cross_lb},\\
    \left\|G_t^s\right\|^2 &\leq 2L_g^2\left\|\mathbf{s}_t - \mathbf{s}_t^*\right\|^2 + 2\left(\frac{2}{\lambda^2} L_f^2 + 2Q_g^2r_\lambda^2\right)\left\|\mathbf{y}_t - \mathbf{y}_\lambda^*(\mathbf{w}_t)\right\|^2.\label{eq:s_gradnorm_lb}   
\end{align}

<!-- proof-6: body_md -->
We first show that the mapping $H(\mathbf{w},\mathbf{y}, \mathbf{s}) \coloneqq \frac{1}{\lambda}\nabla_2 f(\mathbf{w},\mathbf{y}) + \nabla_2 g(\mathbf{w}, \mathbf{y}) - \nabla_2 g(\mathbf{w}, \mathbf{y}-\mathbf{s})$ is $\mu_g$-strongly monotone and $L_g$ lipschitz in $\mathbf{s}$, i.e. 
\begin{align}
    \langle H(\mathbf{w}, \mathbf{y}, \mathbf{s}_1) - H(\mathbf{w}, \mathbf{y}, \mathbf{s}_2), \mathbf{s}_1 - \mathbf{s}_2\rangle &\geq \mu_g \left\|\mathbf{s}_1 - \mathbf{s}_2\right\|^2 ,\label{eq:H_smnt_cross}\\
    \mu_g^2\left\|\mathbf{s}_1 - \mathbf{s}_2\right\|^2 \leq \left\|H(\mathbf{w}, \mathbf{y}, \mathbf{s}_1) - H(\mathbf{w}, \mathbf{y}, \mathbf{s}_2)\right\|^2 &\leq L_g^2\left\|\mathbf{s}_1 - \mathbf{s}_2\right\|^2. \label{eq:H_smnt_sq}
\end{align}
The proof is straightforward, we directly plug the definition of $H$ in the LHS of~\ref{eq:H_smnt_cross}:
\begin{align}
    \text{LHS} &= \langle \nabla_2 g(\mathbf{w}, \mathbf{y}-\mathbf{s}_2) - \nabla_2 g(\mathbf{w}, \mathbf{y}-\mathbf{s}_1), \mathbf{s}_1 - \mathbf{s}_2\rangle \nonumber\\
    &= \langle \nabla_2 g(\mathbf{w}, \mathbf{y}-\mathbf{s}_2) - \nabla_2 g(\mathbf{w}, \mathbf{y}-\mathbf{s}_1), (\mathbf{y} - \mathbf{s}_2) - (\mathbf{y} - \mathbf{s}_1)\rangle \nonumber\\
    &\geq \mu_g\left\|\mathbf{s}_1 - \mathbf{s}_2\right\|^2
\end{align}
where the inequality holds from the strongly convexity of $g$. Then the first inequality of~\ref{eq:H_smnt_sq} holds obviously by combining the fact that 
\[\left\|H(\mathbf{w}, \mathbf{y}, \mathbf{s}_1) - H(\mathbf{w}, \mathbf{y}, \mathbf{s}_2)\right\|\left\|\mathbf{s}_1 - \mathbf{s}_2\right\|\geq \langle H(\mathbf{w}, \mathbf{y}, \mathbf{s}_1) - H(\mathbf{w}, \mathbf{y}, \mathbf{s}_2), \mathbf{s}_1 - \mathbf{s}_2\rangle\]
with~\ref{eq:H_smnt_cross} while the second inequality holds from smoothness of $g$. \\
We note that $G_t^s = H(\mathbf{w}_t, \mathbf{y}_t, \mathbf{s}_t)$ and that 
\begin{align*}
    H(\mathbf{w}_t, \mathbf{y}_\lambda^*(\mathbf{w}_t), \mathbf{s}^*_t) &= H(\mathbf{w}_t, \mathbf{y}_\lambda^*(\mathbf{w}_t), \mathbf{y}_\lambda^*(\mathbf{w}_t) - \mathbf{y}^*(\mathbf{w}_t)) \\
    &= \frac{1}{\lambda}\nabla_2 f(\mathbf{w}_t,\mathbf{y}_\lambda^*(\mathbf{w}_t)) + \nabla_2 g(\mathbf{w}_t, \mathbf{y}_\lambda^*(\mathbf{w}_t)) - \nabla_2 g(\mathbf{w}_t, \mathbf{y}^*(\mathbf{w}_t))\\
    &= \textbf{0}-\textbf{0} = \textbf{0},
\end{align*}
then the cross term becomes:
\begin{align}
    \langle G_t^s, \mathbf{s}_t - \mathbf{s}_t^*\rangle &= \langle H(\mathbf{w}_t, \mathbf{y}_t, \mathbf{s}_t) - H(\mathbf{w}_t, \mathbf{y}_t, \mathbf{s}^*_t), \mathbf{s}_t - \mathbf{s}_t^*\rangle\nonumber\\&\qquad + \langle H(\mathbf{w}_t, \mathbf{y}_t, \mathbf{s}^*_t) - H(\mathbf{w}_t, \mathbf{y}_\lambda^*(\mathbf{w}_t), \mathbf{s}^*_t), \mathbf{s}_t - \mathbf{s}_t^*\rangle\nonumber\\
    &\geq \mu_g\left\|\mathbf{s}_t - \mathbf{s}_t^*\right\|^2+ \langle H(\mathbf{w}_t, \mathbf{y}_t, \mathbf{s}^*_t) - H(\mathbf{w}_t, \mathbf{y}_\lambda^*(\mathbf{w}_t), \mathbf{s}^*_t), \mathbf{s}_t - \mathbf{s}_t^*\rangle\nonumber\\
    &\geq \mu_g\left\|\mathbf{s}_t - \mathbf{s}_t^*\right\|^2 - \frac{1}{2\mu_g} \left\|H(\mathbf{w}_t, \mathbf{y}_t, \mathbf{s}^*_t) - H(\mathbf{w}_t, \mathbf{y}_\lambda^*(\mathbf{w}_t), \mathbf{s}^*_t)\right\|^2 - \frac{\mu_g}{2} \left\|\mathbf{s}_t - \mathbf{s}_t^*\right\|^2\nonumber\\
    &\geq \frac{\mu_g}{2}\left\|\mathbf{s}_t - \mathbf{s}_t^*\right\|^2 - \frac{1}{2\mu_g} \left\|H(\mathbf{w}_t, \mathbf{y}_t, \mathbf{s}^*_t) - H(\mathbf{w}_t, \mathbf{y}_\lambda^*(\mathbf{w}_t), \mathbf{s}^*_t)\right\|^2 .\label{eq:cross_split}
\end{align}
For the second term in RHS, we note that $H(\mathbf{w}, \mathbf{y}, \mathbf{s}) =  \frac{1}{\lambda}\nabla_2 f(\mathbf{w},\mathbf{y}) + \int_0^1\nabla^2_{22} g(\mathbf{w}, \mathbf{y}-\tau\mathbf{s})\mathbf{s} d\tau$, then
\begin{align}
    \left\|H(\mathbf{w}_t, \mathbf{y}_t, \mathbf{s}^*_t) - H(\mathbf{w}_t, \mathbf{y}_\lambda^*(\mathbf{w}_t), \mathbf{s}^*_t)\right\|^2 &\leq \frac{2}{\lambda^2} L_f^2\left\|\mathbf{y}_t - \mathbf{y}_\lambda^*(\mathbf{w}_t)\right\|^2 \nonumber\\ + 2\left\|\mathbf{s}^*\right\|^2&\left\|\int_0^1\left(\nabla^2_{22} g(\mathbf{w}, \mathbf{y}_t-\tau\mathbf{s}) - \nabla^2_{22} g(\mathbf{w}, \mathbf{y}_\lambda^*(\mathbf{w}_t)-\tau\mathbf{s})\right) d\tau\right\|^2 \nonumber\\
    &\leq \frac{2}{\lambda^2} L_f^2\left\|\mathbf{y}_t - \mathbf{y}_\lambda^*(\mathbf{w}_t)\right\|^2 + 2Q_g^2\left\|\mathbf{s}\right\|^2\left\|\mathbf{y}_t - \mathbf{y}_\lambda^*(\mathbf{w}_t)\right\|^2 \nonumber\\
    &\leq \left(\frac{2}{\lambda^2} L_f^2 + 2Q_g^2r_\lambda^2\right)\left\|\mathbf{y}_t - \mathbf{y}_\lambda^*(\mathbf{w}_t)\right\|^2.\label{eq:H_y_lips}
\end{align}
Plugging~\ref{eq:H_y_lips} into~\ref{eq:cross_split} leads to~\ref{eq:s_cross_lb}. The~\ref{eq:s_gradnorm_lb} can be proved in a similar way: 
\begin{align*}
    \left\|G_t^s\right\|^2 &\leq 2\left\|H(\mathbf{w}_t, \mathbf{y}_t, \mathbf{s}_t) - H(\mathbf{w}_t, \mathbf{y}_t, \mathbf{s}_t^*)\right\|^2 + 2\left\|H(\mathbf{w}_t, \mathbf{y}_t, \mathbf{s}_t^*)\right\|^2\nonumber\\
    &\leq 2L_g^2\left\|\mathbf{s}_t - \mathbf{s}_t^*\right\|^2 + 2\left(\frac{2}{\lambda^2} L_f^2 + 2Q_g^2r_\lambda^2\right)\left\|\mathbf{y}_t - \mathbf{y}_\lambda^*(\mathbf{w}_t)\right\|^2.
\end{align*}
The second inequality holds from~\ref{eq:H_smnt_sq} and~\ref{eq:H_y_lips}.

By plugging~\ref{eq:projed_grad_est_unpack} into ~\ref{eq:mmt_err_mss} and plugging~\ref{eq:s_lip} into~\ref{eq:s_desc_pre_mss}, we have
