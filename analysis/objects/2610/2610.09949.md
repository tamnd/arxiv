<!-- definition-1: body_md -->
\label{def:entropyapprox}
Let $(X,G)$ be an amenable group action and $\mu\in\mathcal M(X,G)$.
\begin{enumerate}
\item We say that $\mu$ is \emph{almost entropy-approximable} if for every
neighborhood $U$ of $\mu$, every $h\in(0,h_\mu(X,G))$, and every
$\varepsilon,\beta>0$, there exist a nonempty compact $G$-invariant set
$\Lambda\subset X$ and $\gamma\in(0,\varepsilon)$ such that
\[
\mathcal M(\Lambda,G)\subset U,
\qquad h(\Lambda,G)>h,
\qquad h(\Lambda,G,\gamma)<h+\beta.
\]
\item We say that $\mu$ is \emph{entropy-approximable} if for every
neighborhood $U$ of $\mu$, every $h\in(0,h_\mu(X,G))$ and every $\beta>0$
there exists a nonempty compact $G$-invariant set $\Lambda$ such that
\[
\mathcal M(\Lambda,G)\subset U,
\qquad h<h(\Lambda,G)<h+\beta.
\]
\item The system is \emph{entropy-generic} if the following holds for
each $0\leq\alpha<h(X,G)$:
\[
\mathcal M_e(X,G,\alpha):=\{\nu\in\mathcal M_e(X,G):h_\nu(X,G)=\alpha\}
\]
is residual in
\[
\mathcal M^\alpha(X,G):=\{\nu\in\mathcal M(X,G):h_\nu(X,G)\ge\alpha\}.
\]
\end{enumerate}
If $h_\mu(X,G)=0$, the first two notions mean that for every neighborhood $U$ of $\mu$ there exists a nonempty compact $G$-invariant set $\Lambda$ with $\mathcal M(\Lambda,G)\subset U$.

<!-- definition-2: body_md -->
\label{df:tiling}
	A quasitiling $\mathcal T$ is determined by two objects:
\begin{enumerate}
	\item a finite collection $\mathcal{S}(\mathcal{T})$ of finite subsets of $G$ containing the unity $e_G$, called the \textit{shapes}.
	\item a finite collection $\mathcal{C}(\mathcal{T}) = \{C(S): S\in\mathcal{S}(\mathcal{T})\}$ of disjoint subsets of $G$, called \textit{center sets} (for the shapes).
\end{enumerate}
The quasitiling is then the family $\mathcal{T} = \{(S,c): S\in\mathcal{S}(\mathcal{T}), c\in C(S)\}$.
We require that the map $(S,c)\mapsto Sc$ is injective.
Hence, by the \textit{tiles} of $\mathcal{T}$ (denoted by the letter $T$) we will mean either the sets $Sc$ or the pairs $(S,c)$ (i.e., the tiles with defined centers), depending on the context.

A quasitiling $\mathcal{T}$ is called an (exact) \textit{tiling} if it is a partition of $G$, i.e.

\[
G=\bigsqcup_{S\in\mathcal S}\bigsqcup_{c\in C(S)}Sc,
\]

admissible family by adding right translates of one sufficiently invariant
finite set, the  approximate product property implies the finite version.

Conversely, take $\delta,\varepsilon\in(0,1)$, a sequence
$\{x_k\}_{k\ge1}\in\mathscr C(X)$ and 
$\{F_k\}_{k\ge1}\in\mathcal{DF}(G)$ such that each $F_k$ is
$\bigl(K(\delta,\varepsilon),\eta(\delta,\varepsilon)\bigr)$-invariant.


For each $k\geq 1$, let
\[
C_k=\left\{z:\left|\{s\in F_k:\rho(sz,sx_k)>\varepsilon\}\right|
\leq\delta|F_k|\right\}
\]
and $Y_k = \bigcap\limits _{j=1}^{k}C_k$.
Writing $r_k=\lfloor\delta|F_k|\rfloor$, we have
\[
C_k=\bigcup_{\substack{A\subset F_k\\|A|\geq|F_k|-r_k}}
\bigcap_{s\in A}\{z:\rho(sz,sx_k)\leq\varepsilon\}.
\]
Thus $C_k$ is closed. By  the finite approximate product property, the set $Y_k$ is a closed non-empty subset of $X$. Thus we have $\bigcap_{k=1}^{\infty} Y_k\ne\varnothing$ and $z \in \bigcap\limits _{k=1}^{\infty} Y_k$ can $(\{F_k\}_{k\ge1},\delta,\varepsilon)$-trace $\{x_k\}_{k\ge1}$.

<!-- example-1: body_md -->
\label{ex:app-not-specification}
Let $G$ be any infinite countable amenable group. Define
\[
Y=\{b\in\{0,1\}^G:|\{g\in G:b(g)=1\}|\leq1\},
\qquad X=\{0,1\}^G\times Y.
\]
Equip $X$ with the shift action
\[
(s(a,b))(g)=(a(gs),b(gs)),\qquad s,g\in G.
\]
Then $(X,G)$ has $\mathrm{APP}_G$ in the prescribed-position sense of
Definition~\ref{APP}, but it does not have specification. Moreover, it is
expansive and has topological entropy $\log 2$.
<!-- end example-1 -->
<!-- proof-3: body_md -->
The set $Y$ is closed and invariant: its complement is the union of the
cylinders requiring a $1$ at each of two distinct coordinates.
Thus $X$ is a compact invariant subspace of the full shift over the alphabet
$\{0,1\}^2$.

Enumerate $G=\{g_1,g_2,\dots\}$ with $g_1=e_G$, and use the compatible metric
\[
d((a,b),(a',b'))=
\sum_{j=1}^{\infty}2^{-j}
\mathbf 1_{\{(a(g_j),b(g_j))\ne(a'(g_j),b'(g_j))\}}.
\]
We verify $\mathrm{APP}_G$ with the boundary convention
$|\partial_KF|<\eta|F|$ used in the manuscript.

Fix $\delta,\varepsilon\in(0,1)$. Choose $N$ such that
$2^{-N}<\varepsilon$, and put $W=\{g_1,\dots,g_N\}$.
Choose a finite set $K\supset W$ satisfying
\[
|K|>\frac{2|W|}{\delta},\qquad \eta=\frac\delta2.
\]
This choice of $(K,\eta)$ depends only on $\delta$ and $\varepsilon$.
Every nonempty $(K,\eta)$-invariant set $F$ satisfies
\begin{equation}\label{eq:example-block-size}
|F|\geq|K|>\frac{2|W|}{\delta}.
\end{equation}
Indeed, if $|F|<|K|$, then $\operatorname{Int}_KF=\varnothing$.
Since $e\in K$, one has $F\subset\operatorname{Cl}_KF=\partial_KF$,
contradicting $\eta<1$.

Let $(F_i)_{i\geq1}$ be any pairwise disjoint family of nonempty
$(K,\eta)$-invariant sets, and let $x_i=(a_i,b_i)\in X$.
Define $z=(a,0^G)\in X$ by
\[
a(g)=
\begin{cases}
a_i(g),&g\in F_i,\\
0,&g\notin\bigcup_iF_i.
\end{cases}
\]
This is well-defined because the $F_i$ are disjoint.
For each $i$, let
\[
\begin{aligned}
B_i^{\rm bd}&=F_i\setminus\operatorname{Int}_KF_i,\\
B_i^{\rm sp}&=\{s\in F_i:b_i(ws)=1\text{ for some }w\in W\}.
\end{aligned}
\]
Then $|B_i^{\rm bd}|\leq|\partial_KF_i|<\eta|F_i|$.
Since $b_i$ has at most one nonzero coordinate, $|B_i^{\rm sp}|\leq|W|$:
if that coordinate is $p_i$, every such $s$ lies in $W^{-1}p_i$.
For $s\in F_i\setminus(B_i^{\rm bd}\cup B_i^{\rm sp})$, the points
$sz$ and $sx_i$ agree at all coordinates in $W$, because
$Ws\subset Ks\subset F_i$ and $b_i$ vanishes on $Ws$.
Therefore $d(sz,sx_i)\leq2^{-N}<\varepsilon$. Using
\eqref{eq:example-block-size}, we obtain
\[
\begin{split}
|\{s\in F_i:d(sz,sx_i)>\varepsilon\}|
&\leq |B_i^{\rm bd}|+|B_i^{\rm sp}|\\
&<\frac\delta2|F_i|+|W|<\delta|F_i|.
\end{split}
\]
This proves the infinite-family version of $\mathrm{APP}_G$ directly.

To disprove specification, fix $\varepsilon_0=1/4$.
For any proposed finite separation set $L\subset G$, choose
$t\notin L\cup L^{-1}\cup\{e\}$. The sets
$E_1=\{e\}$ and $E_2=\{t\}$ satisfy
$LE_i\cap E_j=\varnothing$ for $i\ne j$.
For $p\in G$, write $b^{(p)}$ for the configuration whose unique $1$
is at $p$. Take
\[
x_1=(0^G,b^{(e)}),\qquad x_2=(0^G,b^{(t)}).
\]
If $z=(a,b)$ traces $x_1$ on $E_1$ and $x_2$ on $E_2$ with tolerance
$\varepsilon_0$, then the first inequality forces $b(e)=1$, and the
second forces $b(t)=1$. Indeed, a disagreement at coordinate $e=g_1$
has distance at least $1/2$. This contradicts $b\in Y$.
Thus specification fails.

The same observation shows expansiveness: distinct configurations differ
at some coordinate $s$, so their $s$-translates have distance at least
$1/2$. Finally, for every nonempty finite $F\subset G$,
\[
|\{x|_F:x\in X\}|=2^{|F|}(|F|+1).
\]
The symbolic pattern-count formula for amenable-group entropy gives
\[
h(X,G)=\lim_{n\to\infty}\frac1{|F_n|}
\log\bigl(2^{|F_n|}(|F_n|+1)\bigr)=\log 2
\]
along any F\o lner sequence $(F_n)$.

The sparse coordinate in this example is zero almost surely for every

<!-- remark-3: body_md -->
\label{rem:classical-app}
The positions of the sets $F_i$ in $\mathrm{APP}_G$ are prescribed. This differs
from the classical approximate product property, which allows small
transition intervals or shifts of blocks. For example, an irrational
circle rotation has the classical property: for every accuracy, a bounded
number of iterates suffices to move any phase close to any prescribed
phase, after which the rotation traces the entire next orbit segment.
The bounded transition times are a vanishing proportion of sufficiently
long segments.

In contrast, no isometric action on a space with two distinct points has
$\mathrm{APP}_G$. Choose $x_1,x_2$ with $\rho(x_1,x_2)>2\varepsilon$ and
$\delta<1$, and take two disjoint admissible sets. Since
$\rho(sz,sx_i)=\rho(z,x_i)$, the tracing requirement forces
$\rho(z,x_i)\leq\varepsilon$ for both $i$, which is impossible.
The irrational rotation therefore also shows why $\mathrm{APP}_G$ cannot be
identified with the classical hypothesis in \cite{sun2025ergodic}.

<!-- lemma-2: body_md -->
\label{lem:exhaustive-tempered}
Every infinite countable amenable group $G$ admits an increasing left
tempered F\o lner sequence $(F_n)$ such that
\[
e\in F_1,\qquad \bigcup_{n\geq1}F_n=G,\qquad |F_n|\geq n^2.
\]
In particular, $|F_n|/\log(n+1)\to\infty$.
<!-- end lemma-2 -->
<!-- proof-5: body_md -->
We first observe that, for finite sets $L,D\subset G$ with $e\in L$,
any $\eta>0$ and any $M>0$, there exists a finite set $F\supset D$
such that $|F|\geq M$ and $|LF\setminus F|<\eta|F|$.
Indeed, amenability gives a nonempty finite set $A$ satisfying
$|LA\setminus A|<(\eta/2)|A|$.
Since $G$ is infinite, we may take arbitrarily many pairwise disjoint
right translates of $A$. Their union $H$ satisfies
$|LH\setminus H|<(\eta/2)|H|$ and can be chosen so large that
$|H|\geq M$ and $|LD|<(\eta/2)|H|$.
Then $F=H\cup D$ has the required properties, since
\[
|LF\setminus F|
\leq |LH\setminus H|+|LD|
<\eta|H|\leq\eta|F|.
\]

Enumerate $G=\{g_1,g_2,\ldots\}$ with $g_1=e$, and put $F_0=\varnothing$.
Having chosen $F_1,\ldots,F_{n-1}$, set
\[
D_n=F_{n-1}\cup\{g_1,\ldots,g_n\},\qquad
L_n=\{g_1,\ldots,g_n\}\cup\bigcup_{j<n}F_j^{-1}.
\]
Apply the preceding observation to choose $F_n\supset D_n$ with
\[
|F_n|\geq n^2,\qquad |L_nF_n\setminus F_n|<\frac1n|F_n|.
\]
The sequence is increasing and exhaustive, and $e\in F_1$.
For every fixed $g_k$ and every $n\geq k$,
\[
\frac{|g_kF_n\triangle F_n|}{|F_n|}
=\frac{2|g_kF_n\setminus F_n|}{|F_n|}<\frac2n,
\]
so it is a left F\o lner sequence. Moreover,
\[
\left|\bigcup_{j<n}F_j^{-1}F_n\right|
\leq |L_nF_n|
<\left(1+\frac1n\right)|F_n|\leq2|F_n|.
\]
Thus the sequence is tempered with constant $2$.
The required growth condition follows from $|F_n|\geq n^2$.

\end{equation}
where the middle bound is omitted if $h_{\mu_0}(X,G)=\infty$.
Set $h_1=h_0+5\beta$ and $h^*=h_0+10\beta$.
Proposition~\ref{mainseppro1}, along the auxiliary sequence $(F_n)$,
gives constants $\delta^*,\varepsilon^*>0$ and, for all sufficiently
large $n$, sets
\begin{equation}\label{eq:generic-labels-new}
\Gamma_n^*\subset X_{F_n,B(\mu_0,\eta)},\qquad
|\Gamma_n^*|\geq e^{h^*|F_n|},
\end{equation}
which are $(\delta^*,F_n,\varepsilon^*)$-separated.
Choose
\begin{equation}\label{eq:Delta-choice-new}
0<\varepsilon<\min\{\varepsilon^*/20,\eta,\gamma_0/5\}.
\end{equation}
Let $Q\subset X$ be a finite $\varepsilon/4$-spanning set with the minimum cardinality. 
Next choose $0<\delta<\min\{1/2,\delta^*/10,\eta\}$ with
\begin{equation}\label{eq:delta-choice-new}
H_{\mathrm b}(\delta)+\delta\log \#Q<\beta,
\end{equation}
and obtain $m(\delta,\varepsilon)=(K_{\rm APP},\eta_{\rm APP})$.
Finally choose $0<\kappa<1/4$ and $\theta>0$ such that
\begin{align}
&3\kappa<\min\{\delta^*/2,\eta\},\qquad
\frac{3|K_{\rm APP}|\kappa}{1-3\kappa}<\frac{\eta_{\rm APP}}2,
\label{eq:main-trim-choice}\\
&\theta+4\kappa<\eta,\qquad
(\theta+4\kappa)\log \#Q<\beta,
\label{eq:bad-parameters-new}\\
&h_1(1-\theta)(1-4\kappa)>h_0.
\label{eq:low-parameters-new}
\end{align}
These choices are possible because $\delta$ and the APP data are already
fixed before $\kappa$ is chosen.

                                                                     
Apply Lemma~\ref{lem:admissible-subtiles} with
$(K,\eta_K)=(K_{\rm APP},\eta_{\rm APP})$. Choose its block tail
large enough that \eqref{eq:generic-labels-new} holds and
\[
e^{\beta(1-3\kappa)|F_n|}>2
\]
throughout that tail. By Lemma~\ref{eslemma1}, each $\Gamma_{n_i}^*$
remains $(\delta^*/2,J_{S,i,c},\varepsilon^*)$-separated, and its
empirical measures on $J_{S,i,c}$ lie in $\mathcal B(\mu_0,2\eta)$.
Choose $\Gamma_{S,i,c}\subset\Gamma_{n_i}^*$ of cardinality
$\lceil e^{h_1|J_{S,i,c}|}\rceil$. The tail condition and
$h_1+\beta<h^*$ ensure both its existence and
\begin{equation}\label{eq:label-cardinality-new}
e^{h_1|J_{S,i,c}|}\leq|\Gamma_{S,i,c}|
<e^{(h_1+\beta)|J_{S,i,c}|}.
\end{equation}

Index the tiles by $Sd\in\mathcal T$ and set
\[
\Omega=\prod_{Sd\in\mathcal T}\prod_{i=1}^t
\prod_{c\in C_{S,i}}\Gamma_{S,i,c}.
\]
For $\omega\in\Omega$, apply $\mathrm{APP}_G$ on the pairwise disjoint blocks
$J_{S,i,c}cd$ to the points $d^{-1}c^{-1}\omega_{S,d,i,c}$.
Right invariance \eqref{eq:right-boundary} makes these blocks admissible.
Using the approximate product property, the set
\[
Z^\#(\omega)=\left\{z:
\begin{array}{l}
|\{u\in J_{S,i,c}:\rho(ucdz,u\omega_{S,d,i,c})>\varepsilon\}|\\
\hfill{}\leq\delta|J_{S,i,c}|\quad\text{for every }S,d,i,c
\end{array}\right\}
\]
is nonempty. The relation $R=\{(z,\omega):z\in Z^\#(\omega)\}$
is closed in the compact space $X\times\Omega$, since each block
condition is a finite union of intersections of closed conditions.
Consequently $Z^\#=\pi_X(R)$ is nonempty and compact. Put
\[
Y= \bigcup_{g\in G} g Z^\# = GZ^\#,\text{ and } \Lambda=\overline Y.
\]
Then $\Lambda$ is a nonempty, compact and $G$-invariant subset of $X$.

                                                                      
For $z\in Z^\#$ choose an admissible labeling $\omega$ so that $z\in Z^{\#}(w)$.  On a good block
$P=Jcd$, where $J=J_{S,i,c}$ has label $x$, Lemma~\ref{measures3} gives
\[
\begin{aligned}
D(\mathcal E(z,P),\mu_0)
&\leq D(\mathcal E(cdz,J),\mathcal E(x,J))
 +D(\mathcal E(x,J),\mu_0)\\
&<\varepsilon+\delta+2\eta<4\eta.
\end{aligned}
\]
By Lemma~\ref{lem:complete-local}, the complete tiles in $K_mg$
cover more than $(1-\theta)|K_m|$ points, uniformly in $g$, for all
large $m$. Their good subtiles cover more than
$(1-\theta)(1-4\kappa)|K_m|$ points. Convexity and
Lemma~\ref{measures1} therefore yield
\begin{equation}\label{eq:uniform-empirical-localization}
D(\mathcal E(z,K_mg),\mu_0)<4\eta+\theta+4\kappa<5\eta
\end{equation}
uniformly in $z,g$ and sufficiently large $m$.
By Lemma~\ref{lem:complete-local}, we have
$D(\nu,\mu_0)\leq5\eta<\eta_0$ for every
$\nu\in\mathcal M(\Lambda,G)$. Thus we  prove  item (1).

                                                       
Let $\Lambda_{K_m}=\{Sd\in\mathcal T:Sd\subset K_m\}$. We call each such $Sd$ good tile of $K_m$. On the good
subtiles of these complete tiles, choose all labels independently, keeping
a fixed reference label on every other block. For each such labeling
$\xi$, choose a tracing point $z_\xi\in Z^\#$.
If $\xi\ne\xi'$, they differ on some good block $Jcd$. Their distinct
labels $x,x'$ are separated on at least $(\delta^*/2)|J|$ coordinates.
The two tracing points together have at most $2\delta|J|$ mistakes there.
Since $2\delta<\delta^*/2$, a separating non-mistake coordinate $u\in J$
remains, and
\[
\rho(ucdz_\xi,ucdz_{\xi'})>
\varepsilon^*-2\varepsilon>\varepsilon.
\]
As $ucd\in K_m$, the points $z_\xi$ form an
$(K_m,\varepsilon)$-separated set. Thus
\[
\begin{split}
\log s(\Lambda,K_m,\varepsilon)
&\geq\sum_{Sd\in\mathcal I_m}\sum_{i,c}\log|\Gamma_{S,i,c}|\\
&\geq h_1\sum_{Sd\in\mathcal I_m}\sum_{i,c}|J_{S,i,c}|\\
&>h_1(1-\theta)(1-4\kappa)|K_m|
\end{split}
\]
for every sufficiently large $m$. Consequently,
\[
h(\Lambda,G)\geq h(\Lambda,G,\varepsilon)
\geq h_1(1-\theta)(1-4\kappa)>h_0.
\]
Thus we prove item (2).

                                                                    
Set $\gamma=5\varepsilon<\gamma_0$. For each $y\in Y$, fix a
representation $y=g_yz_y$ and an admissible labeling for $z_y$.
We will encode its orbit on $K_m$ in four parts.


                                   
The ideas in this part are from \cite{liu2026ergodic}.
Recall that $\mathcal S$ is the finite family of shapes of the fixed
tiling $\mathcal T$. Put
\[
S_*=\bigcup_{S\in\mathcal S}S,
\qquad L_m=K_m\cup S_*^{-1}K_m.
\]
The set $S_*$ is finite and independent of $m$. Since $(K_m)$ is a
 F\o lner sequence, we have
\[
0\leq\frac{|L_m\setminus K_m|}{|K_m|}
\leq\sum_{s\in S_*}\frac{|s^{-1}K_m\setminus K_m|}{|K_m|}
\longrightarrow0.
\]
Consequently $|L_m|/|K_m|\to1$. Moreover, for every fixed $a\in G$,
\[
\begin{aligned}
	|aL_m\triangle L_m|
	&\leq |aL_m\triangle aK_m|
	+|aK_m\triangle K_m|+|K_m\triangle L_m|\\
	&=2|L_m\setminus K_m|+|aK_m\triangle K_m|.
\end{aligned}
\]
Dividing by $|L_m|\geq|K_m|$ shows that $(L_m)$ is also a 
F\o lner sequence.

Let $\mathcal A=\mathcal S\cup\{0\}$ be the alphabet of the center
configuration $x_{\mathcal T}$. For $g\in G$, define
\[
p_m(g)\in\mathcal A^{L_m},\qquad
p_m(g)(a)=x_{\mathcal T}(ag)\quad(a\in L_m).
\]
Thus all patterns are indexed by the same finite set $L_m$, even though
they describe the different windows $L_mg$ in the original tiling.
The value $p_m(g)(a)=S\in\mathcal S$ means that $Sag$ is a tile of
$\mathcal T$; the value $0$ means that $ag$ is not a tile center.

We enlarge $K_m$ to $L_m$ because the center of a tile meeting $K_mg$
need not belong to $K_mg$. Indeed, if $Sd\cap K_mg\ne\varnothing$,
then $sd=kg$ for some $s\in S$ and $k\in K_m$, and hence
\[
dg^{-1}=s^{-1}k\in S^{-1}K_m\subset L_m.
\]
Therefore the center and shape of every such tile are recorded by
$p_m(g)$. More precisely, after right multiplication by $g^{-1}$, the
tiles meeting the window are exactly
\[
\{Sa:a\in L_m,\ p_m(g)(a)=S\in\mathcal S,
\ Sa\cap K_m\ne\varnothing\}.
\]
In particular, the same pattern also identifies the complete tiles,
which are precisely those satisfying $Sa\subset K_m$.

For each shape $S$, the sets $J_{S,i,c}$ and the centers
$c\in C_{S,i}$ were fixed before the labeling was chosen. Thus, if
$p_m(g)(a)=S$, the good subtiles in the tile $Sag$ are
$J_{S,i,c}cag$, and their coordinates relative to $K_mg$ are
$J_{S,i,c}ca$. It follows that $p_m(g)$ determines all these relative
subtile locations, including which lie in complete tiles. In particular,
two translations giving the same pattern give exactly the same
geometric domains for the subsequent parts of the code. The pattern
records these domains, while the orbit labels on them will be recorded
separately.

It remains to bound the number of possible patterns. By the shift
convention $(gx)(a)=x(ag)$,
\[
p_m(g)=(gx_{\mathcal T})|_{L_m}.
\]
Let $X_{\mathcal T}=\overline{Gx_{\mathcal T}}$ and set
$N_m=\#\{p_m(g):g\in G\}$. Since the alphabet $\mathcal{A}$ is finite, every
pattern on $L_m$ appearing in $X_{\mathcal T}$ already appears in
some translate of $x_{\mathcal T}$: the cylinder specifying that
finite pattern is open and meets the dense orbit $Gx_{\mathcal T}$.
Hence
\[
N_m=\#\{x|_{L_m}:x\in X_{\mathcal T}\}.
\]
By the symbolic pattern-count formula along the F\o lner sequence $(L_m)$,
tand since$h(X_{\mathcal T},G)=h(\mathcal T)=0$, we
\[
\lim_{m\to\infty}\frac{\log N_m}{|L_m|}=0.
\]
Since $|L_m|/|K_m|\to1$, it follows that
\[
\frac{\log N_m}{|K_m|}
=\frac{|L_m|}{|K_m|}\frac{\log N_m}{|L_m|}
\longrightarrow0.
\]
Therefore, for the previously fixed $\beta>0$, there exists $m_0$
such that
\begin{equation}\label{eq:phase-number-new}
	\#\{p_m(g):g\in G\}\leq e^{\beta|K_m|}
	\qquad(m\geq m_0).
\end{equation}
This estimates all translations simultaneously; in particular, the
threshold $m_0$ is independent of the chosen point $y$ and its
representative $g_y$.
For $y = g_y z_y$, define
\[
\pi^1_m(y) := p_m(g_y).
\]
and $Im(\pi^1_m) := \{\pi^1_m(y) : y\in Y\}$.


For fixed $p_m(g_y)$, record the labels on good subtiles inside complete
tiles of $K_mg_y$ by $(S,d,i,c)$. For $p\in Im(\pi^1_m)$, let $Y(p)= \{y\in Y : \pi^1_m(y) = p\}$. Define
\[
\pi^2_m(y) :=  \prod_{S,d,i,c} \omega_{S,d,i,c} \in  \prod_{S,d,i,c} \Gamma_{{S,i,c}}
\]
Since these subtiles are disjoint, we have
\begin{equation}\label{eq:label-code-upper}
\#\{\pi^2_m(y) : y\in Y(p)\} \leq  \prod_{S,d,i,c}  \#\Gamma_{S,i,c}\leq e^{(h_1+\beta)|K_m|}.
\end{equation}


                                
Fix $y = g_yz_y \in Y(p)$. On each good subtile $(S,d,i,c)$, record
\[
M_{S,d,i,c}(y)=\{u\in J_{S,i,c}:
\rho(ucdz_y,u\omega_{S,d,i,c})>\varepsilon\}.
\]
Its size is at most $\delta|J_{S,i,c}|$. Define
\[
\pi^3_m(y) := \prod_{S,d,i,c} M_{S,d,i,c} cd
\]
By the finite-$N$ estimate
\eqref{eq:finite-mistake-count}, 
\begin{equation}\label{eq:mistake-code-upper}
\# \{\pi^3_m(y) : y\in Y(p)\} \leq \prod_{S,d,i,c}e^{H_{\mathrm b}(\delta)|J_{S,i,c}|}
\leq e^{H_{\mathrm b}(\delta)|K_m|}.
\end{equation}
In particular, no asymptotic error is accumulated over the individual blocks.

                                                    
Fix $p \in Im(\pi^1_m)$ and $y = g_yz_y \in Y(p)$. Let $\Lambda_{K_m g_y} = \{Sd \in \mathcal{T} : Sd \subset K_mg_y\}$ and $\widetilde{K_mg_y} = \bigcup_{Sd\in \Lambda_{K_m g_y}} Sd$. Set $B_{\rm geo}(p) =K_m \setminus \widetilde{K_mg_y}$, which consists of the coordinates whose translates
in $K_mg_y$ lie outside the good subtiles of complete tiles. Uniformly in
the tiling pattern, we have
\[
|B_{\rm geo}(p)|\leq(\theta+4\kappa)|K_m|.
\]
Let $B_{Mist}(y) = \bigcup\limits_{S,d,i,c} M_{S,d,i,c}(y) c d$. Then $|B_{Mist}(p)| \leq \delta |K_m| $.
Let  $B_m(y)= B_{\rm geo}(p) \cup B_{Mist}(y).$ We have
\[
|B_m(y)|\leq(\theta+4\kappa+\delta)|K_m|.
\]
Recall that $Q$ is a finite $\varepsilon/4$-spanning set with the minimum cardinality and write $Q = \{q_1,\cdots,q_{\# Q}\}$. For each $s\in B_m(y)$, let $l_s := \min \{1\leq l \leq \# : \rho(sy, q_l) < \frac{\varepsilon}{4}\}$ and set 
\[
\pi^4_m(y)(s) := q_{l_s}.
\]
The domain is already determined by the pattern and mistake codes.
Thus the value code for uncontrolled  positions has at most
$|Q|^{(\theta+4\kappa+\delta)|K_m|}$ possibilities. Define
\[
\pi^4_m(y) := \prod_{s\in B_m(y)} \pi^4_m(y)(s) \in Q^{B_m(y)}.
\]
Then we have
\[
\# \{\pi^4_m(y): y\in Y(p)\}\leq \#Q^{(\theta+4\kappa+\delta)|K_m|}.
\]

For $y = g_y z_y$, let \[
\Pi_m(y)=\bigl(\pi^1_m(y)),\pi^2_m(y),\pi^3_m(y),\pi^4_m(y)\bigr)
\]
be the complete code, where the four components record, respectively,
the tiling pattern, the labels on complete good subtiles, the mistake
sets on those subtiles, and the $Q$-values at the uncontrolled sites.
We index the label and mistake components by the relative data
$(S,a,i,c)$, where $a=dg_y^{-1}$ for a complete tile $Sd\subset K_mg_y$.
Thus the corresponding subtile has coordinates $J_{S,i,c}ca\subset K_m$.
This convention allows codes belonging to different translations $g_y$
to be compared on the same observation window $K_m$.

For each fixed pattern, the number of label codes is at most
$e^{(h_1+\beta)|K_m|}$. For each fixed pattern and label code, the
number of mistake codes is at most $e^{H_{\mathrm b}(\delta)|K_m|}$.
Once these components have been fixed, the set of uncontrolled
coordinates $B_m(y)$ is determined, and its cardinality is at most
$(\theta+4\kappa+\delta)|K_m|$. Thus the number of value codes over
each such choice is at most $|Q|^{(\theta+4\kappa+\delta)|K_m|}$.
Multiplying these uniform upper bounds, together with the pattern bound
\eqref{eq:phase-number-new}, gives
\[
\begin{aligned}
	\#\{\Pi_m(y):y\in Y\}
	&\leq e^{\beta|K_m|}e^{(h_1+\beta)|K_m|}
	e^{H_{\mathrm b}(\delta)|K_m|}
	\#Q^{(\theta+4\kappa+\delta)|K_m|}\\
	&=\exp\!\Bigl(\bigl[h_1+2\beta
	+H_{\mathrm b}(\delta)+\delta\log\#Q\\
	&\hspace{42mm}+(\theta+4\kappa)\log \#Q\bigr]|K_m|\Bigr).
\end{aligned}
\]
The two parameter choices \eqref{eq:delta-choice-new} and
\eqref{eq:bad-parameters-new} imply
\[
H_{\mathrm b}(\delta)+\delta\log\#Q<\beta,
\qquad (\theta+4\kappa)\log\#Q<\beta.
\]
Consequently, for all sufficiently large $m$,
\begin{equation}\label{eq:total-code-number-new}
	\#\{\Pi_m(y):y\in Y\}\leq e^{(h_1+4\beta)|K_m|}.
\end{equation}
This multiplication only uses uniform bounds on the number of extensions
of each partial code; independence of the four components is not needed.


Suppose $y_1,y_2\in Y$ satisfy $\Pi_m(y_1)=\Pi_m(y_2)$, and write
$y_j=g_jz_j$ using their fixed representations. Equality of the pattern
components identifies the complete tiles and good subtiles in relative
coordinates. Equality of the mistake components then implies
\[
B_m(y_1)=B_m(y_2)=:B.
\]
We compare the orbit coordinates at each $r\in K_m$.

If $r\in B$, equality of the value components gives a common point
$q_r\in Q$ such that
\[
\rho(ry_1,q_r)<\varepsilon/4,
\qquad \rho(ry_2,q_r)<\varepsilon/4.
\]
Hence the triangle inequality yields
\[
\rho(ry_1,ry_2)<\varepsilon/2.
\]

If $r\in K_m\setminus B$, the common pattern places $r$ in a good
subtile of a complete tile, and the common mistake code says that this
coordinate is not a mistake for either point. More explicitly, there
are common relative indices $(S,a,i,c)$ and a common
$u\in J_{S,i,c}$ such that
\[
r=uca.
\]
The corresponding tiles in the original tiling have centers
$d_j=ag_j$, and their good subtiles are $J_{S,i,c}cd_j$.
Equality of the label components supplies the same label
$w\in\Gamma_{S,i,c}$ on these two corresponding subtiles.
Since $u$ belongs to neither mistake set, the tracing inequalities give
\[
\rho(ucd_jz_j,uw)\leq\varepsilon
\qquad(j=1,2).
\]
Using the left-action convention and $d_j=ag_j$, we obtain
\[
ry_j=(uca)(g_jz_j)=ucd_jz_j.
\]
Thus both orbit coordinates are within $\varepsilon$ of the same
point $uw$, and
\[
\rho(ry_1,ry_2)
\leq\rho(ry_1,uw)+\rho(uw,ry_2)\leq2\varepsilon.
\]
The two cases exhaust $K_m$. Therefore
\[
\Pi_m(y_1)=\Pi_m(y_2)
\quad\Longrightarrow\quad
\rho_{K_m}(y_1,y_2)
=\max_{r\in K_m}\rho(ry_1,ry_2)\leq2\varepsilon.
\]


Let $E\subset Y$ be $(K_m,3\varepsilon)$-separated. Two distinct
points of $E$ cannot have the same code, since their Bowen distance
would then be at most $2\varepsilon$. Hence $\Pi_m|_E$ is injective,
and \eqref{eq:total-code-number-new} gives
\[
|E|\leq\#\{\Pi_m(y):y\in Y\}
\leq e^{(h_1+4\beta)|K_m|}.
\]
Taking the supremum over such sets $E$ yields
\[
s(Y,K_m,3\varepsilon)\leq e^{(h_1+4\beta)|K_m|}.
\]

Finally, let $E\subset\Lambda=\overline Y$ be a finite
$(K_m,5\varepsilon)$-separated set. Since $K_m$ is finite and the
action is continuous, for each $x\in E$ the set
\[
\{y\in\Lambda:\rho_{K_m}(x,y)<\varepsilon/2\}
\]
is a relative open neighborhood of $x$. By density of $Y$ in $\Lambda$,
choose $y_x\in Y$ in this neighborhood. If $x\ne x'$ belong to $E$,
the triangle inequality in the Bowen metric implies
\[
\begin{aligned}
	\rho_{K_m}(y_x,y_{x'})
	&\geq\rho_{K_m}(x,x')-\rho_{K_m}(x,y_x)
	-\rho_{K_m}(x',y_{x'})\\
	&>5\varepsilon-\varepsilon=4\varepsilon>3\varepsilon.
\end{aligned}
\]
Thus the points $y_x$ are distinct and form a
$(K_m,3\varepsilon)$-separated subset of $Y$ with cardinality $|E|$.
It follows that
\[
s(\Lambda,K_m,5\varepsilon)
\leq s(Y,K_m,3\varepsilon)
\leq e^{(h_1+4\beta)|K_m|}.
\]
Taking logarithms, dividing by $|K_m|$, and passing to the upper limit,
we conclude that
\[
h(\Lambda,G,5\varepsilon)\leq h_1+4\beta
=h_0+9\beta<h_0+\beta_0.
\]
Since $\gamma=5\varepsilon<\gamma_0$, this proves item (3).


Let $\Pi_m(y)$ be the resulting complete code. Combining these estimates
with \eqref{eq:delta-choice-new} and \eqref{eq:bad-parameters-new}, we get
\begin{equation}\label{eq:total-code-number-new}
\#\{\Pi_m(y):y\in Y\}\leq e^{(h_1+4\beta)|K_m|}.
\end{equation}
If two points have the same code, their orbit coordinates at every
uncontrolled site are within $\varepsilon/2$, and at every remaining
site they are within $2\varepsilon$, because both trace the same local
coordinate of the same label. Hence
\[
\Pi_m(x)=\Pi_m(y)\quad\Longrightarrow\quad
\rho_{K_m}(x,y)\leq2\varepsilon.
\]
It follows that
$s(Y,K_m,3\varepsilon)\leq e^{(h_1+4\beta)|K_m|}$.

Finally, any $(K_m,5\varepsilon)$-separated set in $\Lambda=\overline Y$
can be moved into $Y$ by less than $\varepsilon/2$ in the Bowen metric,
giving a $(K_m,3\varepsilon)$-separated set of the same cardinality.
Therefore
\[
h(\Lambda,G,5\varepsilon)\leq h_1+4\beta
=h_0+9\beta<h_0+\beta_0.
\]
This proves item (3).

\label{prop:entropy-generic-abstract}
Suppose that $(X,G)$ is asymptotically entropy expansive and every measure in
$\mathcal M(X,G)$ is entropy-approximable. Then $(X,G)$ is entropy-generic.

Fix $0\le\alpha<h(X,G)$. By \cite{ren2016local}, asymptotic entropy expansiveness
implies upper semicontinuity of the entropy map. Hence
\[
\mathcal M^\alpha(X,G)=\{\mu:h_\mu(X,G)\ge\alpha\}
\]
is compact and is therefore a Baire space. For $\alpha'>\alpha$ let
\[
\mathcal M_e(\alpha,\alpha')
:=\{\nu\in\mathcal M_e(X,G):\alpha\le h_\nu(X,G)<\alpha'\}.
\]
The ergodic measures form a $G_\delta$ subset of $\mathcal M(X,G)$, and upper
semicontinuity makes $\{\nu:h_\nu(X,G)<\alpha'\}$ open. Thus
$\mathcal M_e(\alpha,\alpha')$ is a $G_\delta$ subset of
$\mathcal M^\alpha(X,G)$.

We prove density. Let $\mu\in\mathcal M^\alpha(X,G)$ and let $V$ be a neighborhood of
$\mu$. Take $\eta>0$ small such that $\mathcal{B}(\mu,2\eta) \subset V$. Since $\alpha<h(X,G)$, the variational principle gives
an invariant measure $\mu_X$ with $h_{\mu_X}(X,G)>\alpha$. For sufficiently small
$t>0$,
\[
\mu'=(1-t)\mu+t\mu_X
\]
belongs to $\mathcal{B}(\mu,2\eta)$, and affinity of entropy gives $\alpha < h_{\mu'}(X,G) < \alpha'$. By Theorem \ref{continuitythm}, there is an ergodic measure $\nu$ satisfying
\[
D(\nu, \mu') < \eta \text{ and } h_{\nu}(X,G) \in (\alpha, \alpha').
\]
Hence we have proved density.

$\mathcal M^\alpha(X,G)$. Taking $\alpha'=\alpha+1/n$ and intersecting over $n\ge1$
gives
\[
\mathcal M_e(X,G,\alpha)
=\bigcap_{n=1}^{\infty}\mathcal M_e(\alpha,\alpha+1/n),
\]
which is residual in $\mathcal M^\alpha(X,G)$.

Fix $\mu\in\mathcal M(X,G)$, $\eta_0>0$ and a finite $h_0<h_\mu(X,G)$.
Using the ergodic decomposition and the fact that the entropy is an affine function on $\mathcal M(X,G)$,
 it follows that with no loss of generality we may assume $\mu$ is a finite sum of ergodic measures $\mu_1,\cdots,\mu_L$, i.e.
 \[
 \mu = \frac{1}{L}\sum_{l=1}^{L}\mu_l,
 \]
 where the $\mu_l$'s are not necessarily distinct. We will finish the proof under this assumption. 
Choose finite $a_l\geq0$, strictly below $h_{\mu_l}(X,G)$ whenever
$h_{\mu_l}(X,G)>0$, and equal to zero otherwise, so that
\[
A:=\frac1L\sum_l a_l>h_0.
\]
Put $L_+=\{l:a_l>0\}$. For $l\in L_+$ apply
Proposition~\ref{mainseppro1} with target $a_l$, obtaining
$\delta_l^*,\varepsilon_l^*>0$. For $l\notin L_+$ choose one generic
point $x_l$ for $\mu_l$ along the F{\o}lner sequence $(F_n)$.

Choose $b>0$ with $2b<\eta_0/8$, then choose $0<\Delta,\delta<1$
so that $\Delta+\delta<\eta_0/8$ and
\[
\Delta<\frac14\min_{l\in L_+}\varepsilon_l^*,
\qquad 2\delta<\frac12\min_{l\in L_+}\delta_l^*.
\]
Obtain $m(\delta,\Delta)=(K_{\rm APP},\eta_{\rm APP})$ and only then
choose $\kappa,\theta>0$ satisfying
\begin{gather}
3\kappa<b,\quad 5L\kappa<1,\quad
\frac{3|K_{\rm APP}|\kappa}{1-3\kappa}<\frac{\eta_{\rm APP}}2,
\label{eq:appendix-small}\\
(5L+4)\kappa+\theta<\eta_0/4,
\qquad (1-\theta)(1-5L\kappa)A>h_0.
\label{eq:appendix-entropy-budget}
\end{gather}
When $L_+\ne\varnothing$, we also require
$3\kappa<\frac12\min_{l\in L_+}\delta_l^*$.
If $A=0$, then $h_0<0$, and the last inequality is automatic.


For color $l$, define
\[
\mathcal T_{l}(S) := \{ Jc ; Jc \text{ is of color } l\}.
\]
and $w_l(S) := \frac{1}{S} \sum\limits_{Jc \text{ of color } l}|J|$.
By Lemma \ref{lem:balanced-coloring}, we have
\[
\big|w_l(S)- \frac{1}{L}\big| < 5\kappa 
\]
and the uncolored remainder has proportion less than $4\kappa$.
Also we take the block tail
so far out that all generic-label conclusions for the finitely many
measures hold in $\mathcal B(\mu_l,b)$. By Lemma~\ref{eslemma1}, for a good subtile $J$ of color
$l\in L_+$, we can choose a label set $\Gamma_J^{(l)}$ with
\[
|\Gamma_J^{(l)}|\geq e^{a_l|J|},
\]
which is $(\delta_l^*/2,J,\varepsilon_l^*)$-separated and whose empirical
measures lie in $\mathcal B(\mu_l,2b)$.
For $l\notin L_+$, let $\Gamma_J^{(l)} =\{x_l\}$ and by Lemma~\ref{measures2}, we also have 
$
\mathcal{E}_{J}(x_l) \in \mathcal{B}(\mu_l, 2b).
$

Fix the coloring and the
label sets once for each shape, and translate this data to every tile.


Index the tiles by $Sd\in\mathcal T$ and set
\[
\Omega=\prod_{Sd\in\mathcal T}\prod_{i=1}^t
\prod_{c\in C_{S,i}}\Gamma_{S,i,c}.
\]
For $\omega = (\omega_{S,d,i,c})\in\Omega$, $\omega_{S,d,i,c} \in \Gamma^{l}_{J_{S,i,c}}$ if $J_{S,i,c} c\in \mathcal T_j(S)$. Apply $\mathrm{APP}_G$ on the pairwise disjoint blocks
$J_{S,i,c}cd$ to the points $d^{-1}c^{-1}\omega_{S,d,i,c}$.
Right invariance \eqref{eq:right-boundary} makes these blocks admissible.
Using the approximate product property, the set
\[
Z^\#(\omega)=\left\{z:
\begin{array}{l}
	|\{u\in J_{S,i,c}:\rho(ucdz,u\omega_{S,d,i,c})>\varepsilon\}|\\
	\hfill{}\leq\delta|J_{S,i,c}|\quad\text{for every }S,d,i,c
\end{array}\right\}
\]
is nonempty. The relation $R=\{(z,\omega):z\in Z^\#(\omega)\}$
is closed in the compact space $X\times\Omega$, since each block
condition is a finite union of intersections of closed conditions.
Consequently $Z^\#=\pi_X(R)$ is nonempty and compact. Put
\[
Y= \bigcup_{g\in G} g Z^\# = GZ^\#,\text{ and } \Lambda=\overline Y.
\]
Then $\Lambda$ is a nonempty, compact and $G$-invariant subset of $X$.
On each good subtile of color $l$, the empirical measure of a tracing
point is within
$e_0:=2b+\Delta+\delta<\eta_0/4$ of $\mu_l$.
We spell out the error over a complete tile, since the coloring is by
area and its error depends on $L$.

Fix a tile $Sd$. Its color weights satisfy \eqref{eq:color-weights},
and let $r_S=1-\sum_{i=1}^{L}w_i(S)<4\kappa$.

Recall that for $\lambda,\lambda' \in \mathcal M(X)$, 
\[
D(\lambda,\lambda')
=
\sum_{j=1}^{\infty}2^{-j}
\left|
\int\varphi_j\,d\lambda-\int\varphi_j\,d\lambda'
\right|,
\qquad 0\leq\varphi_j\leq1.
\]
For each test function \(\varphi_j\), we compare its average over \(Sd\) with its integral against \(\mu\). There are three contributions to the error. The first is the tracing error on the prescribed orbit pieces. The second comes from replacing the actual color weights \(w_l(S)\) by the target weights \(1/L\). Since the averages of \(\varphi_j\) associated with each color belong to \([0,1]\), this contribution is at most
\[
\sum_{i=1}^{L}\left|w_i(S)-\frac1L\right|.
\]
The third contribution comes from the remainder of the tile and is at most its relative size \(r_S\), again because \(0\leq\varphi_j\leq1\).

Writing \(\tau_j\) for the tracing-error contribution for \(\varphi_j\), we therefore obtain
\[
\left|
\int\varphi_j\,d\mathcal E(z,Sd)
-
\int\varphi_j\,d\mu
\right|
\leq
\tau_j+
\sum_{i=1}^{L}\left|w_i(S)-\frac1L\right|+r_S.
\]
The preceding tracing estimate gives
\(\sum_{j=1}^{\infty}2^{-j}\tau_j\leq e_0\).
By the choice of metric $D$, we conclude that
\[
\begin{aligned}
	D\bigl(\mathcal E(z,Sd),\mu\bigr)
	&\leq
	e_0+\sum_{i=1}^{L}\left|w_i(S)-\frac1L\right|+r_S\\
	&<e_0+(5L+4)\kappa.
\end{aligned}
\]
This bound holds uniformly over all complete tiles.

We next pass from individual tiles to \(K_mg\), where $g\in G$. Let $\Lambda_{K_m g} = \{Sd\in \mathcal T : Sd \subset Kmg\}$.  Write $\Lambda_{K_m g}  = \{T_1,\ldots,T_N\}$
 and put
\[
U_{m,g}=\bigcup_{\ell=1}^{N}T_\ell,
\qquad
R_{m,g}=K_mg\setminus U_{m,g}.
\]
For all sufficiently large \(m\), uniformly in \(g\), the complete-tile coverage estimate implies
\[
\frac{|R_{m,g}|}{|K_m|}<\theta.
\]
Since the tiles are disjoint, the empirical measure on \(K_mg\) decomposes as
\[
\mathcal E(z,K_mg)
=
\sum_{\ell=1}^{N}
\frac{|T_\ell|}{|K_m|}
\,\mathcal E(z,T_\ell)
+
\frac{|R_{m,g}|}{|K_m|}\,\mathcal E(z,R_{m,g}).
\]


Moreover, \(D(\lambda,\lambda')\leq1\) for any two probability measures, because every \(\varphi_j\) takes values in \([0,1]\). Applying these observations to the preceding decomposition yields
\[
\begin{aligned}
	D\bigl(\mathcal E(z,K_mg),\mu\bigr)
	&\leq
	\sum_{\ell=1}^{N}\frac{|T_\ell|}{|K_m|}
	D\bigl(\mathcal E(z,T_\ell),\mu\bigr)
	+\frac{|R_{m,g}|}{|K_m|}\\
	&\leq
	(1-\frac{|R_{m,g}|}{|K_m|})\bigl(e_0+(5L+4)\kappa\bigr)
	+\frac{|R_{m,g}|}{|K_m|}\\
	&< e_0+(5L+4)\kappa+\theta
	<\frac{\eta_0}{2}.
\end{aligned}
\]
Here the last inequality follows from the choice of parameters. In particular, the estimate holds for all sufficiently large \(m\), with the threshold independent of \(g\).

By Lemma~\ref{lem:complete-local}, applied to $Z^\#$, this uniform
estimate implies
\begin{equation}\label{eq:appendix-measure-localization}
D(\nu,\mu)\leq\eta_0/2<\eta_0
\quad\text{for every }\nu\in\mathcal M(\Lambda,G).
\end{equation}

If $h_0<0$, any ergodic invariant measure on the nonempty compact system
$\Lambda$ already has entropy greater than $h_0$, so localization completes
the proof in this case. Suppose now that $h_0\geq0$. Then $A>0$ and
$L_+\ne\varnothing$. Vary the labels independently on good subtiles in
complete tiles of $K_m$. Distinct choices differ on a color in $L_+$,
because all other label sets are singletons. The same separation argument
as in Step 4 of Proposition~\ref{mainprop1} gives a common separation
scale, for instance $\frac12\min_{i\in L_+}\varepsilon_i^*$.
The logarithm of the number of labels on a single complete tile is at least
\[
\begin{split}
\sum_i a_i w_i(S)|S|
&\geq\left(\frac1L-5\kappa\right)\sum_i a_i|S|\\
&=(1-5L\kappa)A|S|.
\end{split}
\]
Consequently,
\[
h(\Lambda,G)\geq(1-\theta)(1-5L\kappa)A>h_0.
\]
The variational principle and ergodic decomposition yield
$\nu\in\mathcal M_e(\Lambda,G)$ with $h_\nu(X,G)>h_0$.
Together with \eqref{eq:appendix-measure-localization}, this proves
entropy-denseness.
