<!-- definition-1: body_md -->
The {\it Dirichlet convolution\/} of arithmetic functions $f_1, f_2, \dots, f_j$
is defined as
\begin{align}
 (f_1 * f_2 * \cdots * f_j)(n) &= \sum_{n_1 n_2 \cdots n_j = n} f_1(n_1) f_2(n_2) \cdots f_j(n_j),
\label{econvo}
\end{align}
summing over all ordered factorisations of $n \in {\mathbb N}$ into $j$ factors.
The convolution is evidently commutative and linear in each function $f_1, \dots, f_j$.
The $j$th convolution power is notated as $f^{*j} := f * f * \cdots * f$,
where $f$ is repeated $j$ times on the right-hand side.
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{thm:Db}
For $j\in\mathbb N_0$,
$$ b_j = (|\mu| - e)^{*j}. $$
The corresponding Dirichlet series has the form
\[
\sum_{n=1}^\infty \frac{b_j}{n^s}=\left ( \frac{\zeta(s)}{\zeta(2s)}-1\right)^j
\qquad (\mathop{\rm Re} s > 1).
\]
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
From equation (\ref{lembj})
\begin{align*}
 b_j(n) &= \lambda(n) \sum_{n_1 n_2 \cdots n_j = n} \prod_{k=1}^j (\mu(n_k) - e(n_k))
 = \sum_{n_1 n_2 \cdots n_j = n} \prod_{k=1}^j \lambda(n_k) (\mu(n_k) - e(n_k))
 & \\
 &= \sum_{n_1 n_2 \cdots n_j = n} \prod_{k=1}^j (|\mu(n_k)| - e(n_k))
 = (|\mu| - e)^{*j}(n)
\qquad (n\in\mathbb N),
 \end{align*}
where we used the complete multiplicativity of
$\lambda$
and the facts that
$\lambda \mu = |\mu|$
and
$\lambda e = e.$

For the Dirichlet series, we first note that for a general arithmetic function $f(n)$ with Dirichlet series $F(s)$, the arithmetic function $f(n)-e(n)$ has Dirichlet series
\[
\sum_{n=1}^\infty \frac{f(n)-e(n)}{n^s}=\sum_{n=1}^\infty \frac{f(n)}{n^s}-1=F(s)-1.
\]
Also, the Dirichlet series of a convolution of arithmetic functions is equal
to the product of their individual Dirichlet series.
Hence the $j$-th convolution power $(f(n)-e(n))^{*j}$ has Dirichlet series $(F(s)-1)^j$.

The Dirichlet series for Liouville's function $\lambda$ is given by $\zeta(2s)/\zeta(s)$ \cite{Lehman}.
Moreover, the Dirichlet convolution inverse of $\lambda$ is
the absolute value of the Möbius function \cite[Ex.\ I.8.(8)]{rSchSp},
$\lambda^{-*}(n) = \mu(n)\,\lambda(n) = |\mu(n)| = \mu^2(n)$, which is the characteristic function of the squarefree integers.

It follows that the function $(|\mu| - e)$ has the Dirichlet series $\zeta(s)/\zeta(2s)-1$ (see also \cite[Corollary 8.20]{NZM}), and consequently
\[
\sum_{n=1}^\infty \frac{b_j(n)}{n^s}=\left ( \frac{\zeta(s)}{\zeta(2s)}-1\right)^j.
\qedhere
\]

\label{thm11}
For each $n\in\mathbb N$,
the divisor function polynomials $P(n,\cdot)$ and $Q(n,\cdot)$ satisfy the identity
\begin{equation}\label{PQlem}
P\left(n,x-1\right)=Q\left(n,x\right) \quad (x \in \mathbb{C}).
\end{equation}

<!-- proof-2: body_md -->
Applying Newton's binomial theorem
\begin{equation}
 \frac{1}{(1+x)^s} = \sum_{l=0}^\infty \binom{l + s - 1}{l} (-1)^l  x^l
\end{equation}
in the commutative Dirichlet convolution algebra of arithmetic functions, we obtain the identity
\begin{align*}
 \mu^{*L} &= 1^{*-L}
 = (e + (1 - e))^{*-L}
 = \sum_{l=0}^\infty \binom{l + L - 1}{l} (-1)^l  (1 - e)^{*l}
\end{align*}
and consequently
\begin{align*}
 c_L^{(-L)} &= (e - \mu)^{*L} * 1^{*L} * \mu^{*L}\\
 &= (1 - e)^{*L} * \sum_{l=0}^\infty \binom{l + L - 1}{l} (-1)^l  (1 - e)^{*l}\\
 &= \sum_{l=0}^\infty \binom{l + L - 1}{l} (-1)^l  (1 - e)^{*l+L}
 = \sum_{l=0}^\infty \binom{l + L - 1}{l} (-1)^l  c_{l + L}.
\end{align*}
Therefore we find
\begin{align*}
 \sum_{L=1}^\infty (x - 1)^{L-1}  c_L &= \sum_{j=0}^\infty (x - 1)^j  c_{j+1}
 = \sum_{j=0}^\infty \sum_{k=0}^j \binom{j}{k} x^k (-1)^{j-k}  c_{j+1}\\
 &= \sum_{k=0}^\infty x^k \sum_{l=0}^\infty \binom{l + k}{k} (-1)^l c_{l+k+1}
 \\
 &= \sum_{L=1}^\infty x^{L-1} \sum_{l=0}^\infty \binom{l + L - 1}{l} (-1)^l  c_{l+L}
 = \sum_{L=1}^\infty x^{L-1}  c_L^{(-L)}.
\end{align*}
(Note that all sums of associated divisor functions are pointwise finite, so
there is no issue of convergence in the above calculations.)

<!-- theorem-3: body_md -->
\label{thm12}
The non-trivial divisor function polynomial $P$ has, for $\mathop{\rm Re} s > 1$                                  
and $x\in\mathbb C$ such that $|x| < |\zeta(\mathop{\rm Re} s) - 1|^{-1}$, the Dirichlet series
\begin{equation} \label{eq:DP}
D_{P(\cdot, x)}(s) = \sum_{n=1}^\infty \frac{P(n, x)}{n^s} = \frac{\zeta(s) - 1}{1 + x - x\,\zeta(s)};
\end{equation}
the rational function on the right-hand side gives its analytic continuation
to all $(s, x) \in\mathbb C^2$ such that $s\neq 1$ and $x\neq (\zeta(s) - 1)^{-1}$.

The analytically continued Dirichlet series for the balanced divisor function
polynomial $Q$ and the shifted polynomials $P + e$ and $Q + e$ take,
for $s\in\mathbb C\setminus\{1\}$,
 the form
\begin{align}
&D_{Q(\cdot, x)}(s) = \frac{\zeta(s) - 1}{x + (1 - x)\,\zeta(s)} \qquad (x\in\mathbb C\setminus
\{{\textstyle \frac{\zeta(s)}{\zeta(s) - 1}}\}),
\label{eq:DQ}
\\
&D_{P(\cdot, x) + e}(s) = D_{P(\cdot, x)} + 1 = \frac{x + (1 - x)\,\zeta(s)}{1 + x - x\,\zeta(s)} \qquad (x\in\mathbb C\setminus
\{{\textstyle \frac 1{\zeta(s) - 1}}\}),
\label{eq:DP1}
\\
&D_{Q(\cdot, x) + e}(s) = D_{Q(\cdot, x)}(s) + 1 = \frac{x - 1 + (2 - x)\,\zeta(s)}{x + (1 - x)\,\zeta(s)} \qquad (x\in\mathbb C\setminus
\{{\textstyle \frac{\zeta(s)}{\zeta(s) - 1}}\}).
\label{eq:DQ1}
\end{align}
Moreover, the following identities hold,
\begin{align}
 &D_{P(\cdot, x)} = D_{P(\cdot, x)+e}\,D_{Q(\cdot, x)};
 \label{eq:DPQ}
\\
 &D_{P(\cdot, x) + e}\,D_{Q(\cdot, x) + e} = 2\,D_{P(\cdot, x)} + 1.
 \nonumber
\end{align}
<!-- end theorem-3 -->
<!-- proof-3: body_md -->
For $N\in\mathbb N$, let $\hat\Omega(N) := \max\limits_{1\le n\le N} \Omega(n)
= \lfloor \log_2 N \rfloor$.
For $x > 0$, we have
\begin{equation*}
 \sum_{n=1}^N \frac{P(n, x)}{n^s} = \sum_{n=1}^N \sum_{k=1}^{\hat\Omega(N)} x^{k-1} \frac{c_k(n)}{n^s}
 = \sum_{k=1}^{\hat\Omega(N)} x^{k-1} \sum_{n=1}^N \frac{c_k(n)}{n^s}
 = \sum_{k=1}^{\infty} x^{k-1} \sum_{n=1}^N \frac{c_k(n)}{n^s}.
\end{equation*}
Here
\begin{align*}
\left| x^{k-1} \sum_{n=1}^N \frac{c_k(n)}{n^s} \right|
&= |x|^{k-1}\left|\sum_{n=1}^N \frac{c_k(n)}{n^s} \right|
\le|x|^{k-1}\sum_{n=1}^N \frac{c_k(n)}{|n^s|}
\\
&= |x|^{k-1} \sum_{n=1}^N \frac{c_k(n)}{n^{\mathop{\rm Re} s}}
\le  |x|^{k-1}\,|\zeta(\mathop{\rm Re} s) - 1|^k,
\end{align*}
which is independent of $N$, and
$$
 \sum_{k=1}^\infty |x|^{k-1}\,|\zeta(\mathop{\rm Re} s) - 1|^k
= |\zeta(\mathop{\rm Re} s) - 1|\sum_{k=1}^\infty (|x|\,|\zeta(\mathop{\rm Re} s) - 1|)^{k-1} < \infty
$$
for $|x| < |\zeta(\mathop{\rm Re} s) - 1|^{-1}$, and also
\begin{equation*}
\lim_{N \rightarrow\infty} x^{k-1} \sum_{n=1}^N \frac{c_k(n)}{n^s}
= x^{k-1} (\zeta(s) - 1)^k
\end{equation*}
for each $k\in\mathbb N$,
so by the dominated convergence theorem
\begin{equation*}
D_{P(\cdot, x)}(s) = \lim_{N\rightarrow\infty} \sum_{n=1}^N \frac{P(n, x)}{n^s}
= \sum_{k=1}^\infty x^{k-1}\,(\zeta(s) - 1)^k
= \frac {\zeta(s) - 1}{1 - x\,(\zeta(s) - 1)},
\end{equation*}
giving (\ref{eq:DP}).
Hence (\ref{eq:DP1}) follows by a simple calculation. Equations (\ref{eq:DQ})
and (\ref{eq:DQ1}) arise by replacing $x$ in the formulae for $D_P$ by $x - 1$
(see Theorem \ref{thm11}).
The final two identities are a direct consequence of the formulae
(\ref{eq:DP}) -- (\ref{eq:DQ1}).

<!-- theorem-4: body_md -->
\label{thm112}
The arithmetic functions $P(\cdot,x)$ and $Q(\cdot,x)$, where $x \in \mathbb{C}$, obey the sum-over-divisor relations
\begin{equation}\label{115}
P(n,x)=x\sum_{d\mid n,\ d<n}\left (P(d,x)+\frac{e(d)}{x}\right),
\end{equation}
\begin{equation}\label{116}
Q(n,x)=(x-1)\sum_{d\mid n,\ d<n}\left (Q(d,x)+\frac{e(d)}{x-1}\right),
\end{equation}
and
\begin{equation}\label{117}
P(n,x)=\sum_{d\mid n,\ d<n}\left (P(d,x)+e(d)\right)Q\left(\frac{n}{d},x\right).
\end{equation}
<!-- end theorem-4 -->
<!-- proof-4: body_md -->
By the definition of Dirichlet convolution, we see that for every arithmetic
function $f$,
\begin{equation*}
\sum_{d | n,\ d < n} f(d) = (f * (1 - e))(n) \qquad (n\in\mathbb N).
\end{equation*}
Hence the right-hand side of equation (\ref{115}), abstracting from the variable
$n$, is equal to
\begin{align*}
 (x\, P(\cdot, x) + e) * (1 - e) &= x\,P(\cdot, x) * (1 - e) + e * (1 - e)
\\
 &= x \sum_{j=1}^\infty x^{j-1}\,c_j * (1 - e) + 1 - e
\\
 &= \sum_{j=1}^\infty x^j\,(1 - e)^{*(j+1)} + 1 - e
 = \sum_{k=2}^\infty x^{k-1}\,(1 - e)^{*k} + 1 - e
\\
 &= \sum_{k=1}^\infty x^{k-1}\,(1 - e)^{*k}
 = P(\cdot, x).
\end{align*}
Equation (\ref{116}) now follows by Theorem \ref{thm11} as
\begin{align*}
 Q(\cdot, x) &= P(\cdot, x - 1)
 = ((x - 1) P(\cdot, x - 1) + e) * (1 - e)
\\
 &= ((x - 1) Q(\cdot, x) + e) * (1 - e).
\end{align*}
Finally, as the Dirichlet series of a convolution of arithmetic functions
is the product of their separate Dirichlet series, we
deduce from equation (\ref{eq:DPQ}) that
\begin{align*}
P(n, x) &= \left((P(\cdot, x) + e) * Q(\cdot, x)\right)(n)
= \sum_{d | n} (P(d, x) + e(d))\,Q\left(\frac n d, x\right)
\end{align*}
for all $n\in\mathbb N$, giving equation (\ref{117}) since
$Q(1, x) = 0$.

<!-- definition-3: body_md -->
The {\it Bernoulli numbers\/} $B_n$ form a special sequence of rational numbers which are related to the Riemann zeta function at even integer arguments. The sequence begins $(B_0, B_1, B_2, B_3, B_4, B_5, B_6, \dots) = (1,-\frac{1}{2},\frac{1}{6},0,-\frac{1}{30},0,\frac{1}{42},\dots)$, defined by the generating function \cite[p.\ 48]{comtet}
\[
\frac{t}{e^t-1}=\sum_{n=0}^\infty \frac{t^n B_n}{n!};
\]
there is also the historic variant \cite[p.\ 107]{conway}
\[
\frac{t e^t}{e^t-1}=\sum_{n=0}^\infty \frac{t^n B_n^+}{n!},
\]
where $B_n^+=B_n$ for $n\neq 1$, $B_1^+=\frac{1}{2}$.

The {\it ordered Bell numbers\/} (or {\it Fubini numbers\/}) $F_n$,  and {\it ordered Bell polynomials\/} $F_n(x)$ are defined by
\[
F_n=\sum_{k=1}^nk!S(nk),\quad \text{and}\quad F_n(x)=\sum_{k=1}^nk!S(n,k)x^k,
\]
respectively,
where $S(n,k)$ are the Stirling numbers of the second kind. The ordered Bell numbers count the number of weak orderings or equivalently the number of distinct \emph{ordered} partitions of a set of $n$ elements \cite{tanny}.

Lastly, we define the {\it signed ordered Bell numbers\/} $F_n^-$ and {\it signed ordered Bell polynomials\/} $F_n^-(x)$ as
\[
F_n^-=\sum_{k=1}^n(-1)^{n-k}k!S(n,k)\quad \text{and}\quad F_n^-(x)=\sum_{k=1}^n(-1)^{n-k}k!S(n,k)x^k.
\]

<!-- theorem-5: body_md -->
\label{thm115}
Let $n$ be a squarefree number with $\omega(n)=\Omega(n)$ distinct prime factors. Then
\[
x\,P(n,x)=F_{\omega(n)}(x),\qquad x\,Q(n,x)=F^-_{\omega(n)}(x).
\]
<!-- end theorem-5 -->
<!-- proof-5: body_md -->
Every non-trivial divisor of the squarefree number $n$ is also squarefree, so the
number of non-trivial factorisations of $n$ is equal to the number of
non-trivial squarefree factorisations, i.e.\ $b_j(n) = c_j(n)$.
Moreover, factorising $n$ into $j$ non-trivial factors is equivalent to
assigning its $\Omega(n) = \omega(n)$ prime factors to $j$ non-empty sets,
with each non-trivial factor being the (commutative) product of the primes in
one of the sets. Hence
\begin{equation}\label{eq:sff}
c_j(n) = |c_j^{(-j)}(n)| = b_j(n) = S(\omega(n), j)\,j!.
\end{equation}
It follows that
\begin{equation*}
x\,P(n, x) = \sum_{j=1}^{\omega(n)} x^j\,c_j(n)
 = \sum_{j=1}^{\omega(n)} x^j\,j!\,S(\omega(n), j)
 = F_{\omega(n)}(x).
\end{equation*}
Similarly,
\begin{align*}
x\,Q(n, x) &= \sum_{j=1}^{\omega(n)} x^j\,c_j^{(-j)}(n)
 = \sum_{j=1}^{\omega(n)} x^j\,(-1)^{j+\omega(n)}\,b_j(n)
\\
 &= \sum_{j=1}^{\omega(n)} x^j\,(-1)^{j+\omega(n)}\,S(\omega(n), j)\,j!
 = F_{\omega(n)}^-(x).
\qedhere
\end{align*}

By \cite[Proposition 2.1]{kellner}
the Bernoulli numbers satisfy the identity
\begin{equation*}
B_j = \sum_{k=1}^j\frac{(-1)^k k!S(j,k)}{k+1} \qquad (j\geq 1).
\end{equation*}
Combining this formula with equation (\ref{eq:sff}),
we find
\begin{align*}
B_{\omega(n)} &= \sum_{k=1}^{\omega(n)} \frac{(-1)^k k!\,S(\omega(n), k)}{k+1}
= \sum_{k=1}^{\omega(n)} \frac{(-1)^k c_k(n)}{k+1}
\\
&= \sum_{k=1}^{\omega(n)} c_k(n) \int_{-1}^0 x^k\,d x
= \int_{-1}^0 \sum_{k=1}^{\omega(n)} x^k c_k(n)\,d x
= \int_{-1}^0 x\,P(n, x)\,d x.
\end{align*}

$c_k^{(-k)}(n) = (-1)^{k+\omega(n)}\,|c_k^{(-k)}(n)|$, we also have
\begin{equation}\label{eq:Bwn}
B_{\omega(n)} = \sum_{k=1}^{\omega(n)} \frac{(-1)^k\,(-1)^{k+\omega(n)} c_k^{(-k)}(n)}{k+1}
= (-1)^{\omega(n)} \sum_{k=1}^{\omega(n)} \frac{c_k^{(-k)}(n)}{k+1}.
\end{equation}
Since $B_1 = - B_1^+$ and $B_j = 0$ for all odd $j \ge 3$ \cite[p.48 {[14d]}]{comtet}, it follows that
\begin{align*}
B_{\omega(n)}^+ &= \sum_{k=1}^{\omega(n)} \frac{c_k^{(-k)}(n)}{k+1}
\\
&= \sum_{k=1}^{\omega(n)} c_k^{(-k)}(n) \int_0^1 x^k\,d x
= \int_0^1 \sum_{k=1}^{\omega(n)} x^k\,c_k^{(-k)}(n)\,d x
= \int_0^1 x\,Q(n, x)\,d x.
\end{align*}
The equality with integrals over the (standard and signed) ordered Bell

<!-- proof-7: body_md -->
Starting from the right-hand side and the identity
\begin{equation*}
m!\,S(L, m) = \sum_{k=0}^m (-1)^{m-k} {m\choose k} k^L
\end{equation*}
(see \cite[Theorem 5.1.A]{comtet}),
we find
\begin{align*}
\sum_{L=1}^\infty & m!\,S(L-1, m-1)\,c_L =
\sum_{L=1}^\infty  m \sum_{k=0}^{m-1} (-1)^{m-1-k} {m-1 \choose k} k^{L-1} c_L
\\
&= m \sum_{j=1}^m (-1)^{m-j} {m-1 \choose j-1} \sum_{L=1}^\infty (j-1)^{L-1} c_L
\\
&= m \sum_{j=1}^m (-1)^{m-j} {m-1 \choose j-1}\,P(\cdot, j-1)\\
&= m \sum_{j=1}^m (-1)^{m-j} {m-1 \choose j-1}\,Q(\cdot, j)
\\
&= m \sum_{j=1}^m (-1)^{m-j} {m-1 \choose j-1} \sum_{L=1}^\infty j^{L-1} c_L^{(-L)}
\\
&= \sum_{L=1}^\infty \sum_{j=1}^m (-1)^{m-j}\,\frac{(m-1)!}{(m-j)!\,(j-1)!}\,j^{L-1}\,c_L^{(-L)}
\\
&= \sum_{L=1}^\infty \sum_{j=0}^m (-1)^{m-j} {m\choose j} j^L c_L^{(-L)}
= \sum_{L=1}^\infty m!\,S(L, m)\,c_L^{(-L)}
\end{align*}
using Theorem \ref{thm11}.

<!-- theorem-8: body_md -->
\label{thm:DN}
For natural numbers $m,n$, $m\ge 2$, the Dirichlet series for the arithmetic function $\mathcal{N}_m$, which counts the number of $m$-part sum systems, is given by
\begin{equation}\label{eq:DN}
\sum_{n=1}^\infty \frac{\mathcal{N}_m(n)}{n^s}
= \sum_{L=1}^\infty m!\,S(L-1, m-1)\,(\zeta(s) - 1)^L
\end{equation}
for $s\in\mathbb C$, $\mathop{\rm Re} s > 1$, such that                       
$\zeta(\mathop{\rm Re} s) < \frac m{m-1}=1+\frac{1}{m-1}$.
The latter condition is satisfied whenever $\mathop{\rm Re} s > r_m$, with
the unique real number $r_m > 1$ such that $\zeta(r_m) = \frac m{m-1}$.

For $m=2$, we have $r_2 \approx 1.728649\dots$; for $m\ge 3$,
\begin{equation*}
\frac{\log (m-1)}{\log 2} < r_m < 1 + \frac{\log (m-1)}{\log\frac 3 2},
\end{equation*}
so $r_m$  is of order $\log m$ for large values of $m$.

<!-- proof-8: body_md -->
By equation (\ref{eq:Hylock}), we find
\begin{align*}
\sum_{n=1}^\infty \frac{\mathcal N_m(n)}{n^s} &= \sum_{n=1}^\infty \frac 1{n^s} \sum_{L=1}^{\Omega(n)} m!\,S(L-1, m-1)\,c_L(n)
\\
&= \lim_{N\rightarrow\infty} \sum_{n=1}^N \frac 1{n^s} \sum_{L=1}^{\hat\Omega(N)} m!\,S(L-1, m-1)\,c_L(n)
\\
&= \lim_{N\rightarrow\infty} \sum_{L=1}^{\hat\Omega(N)} m!\,S(L-1, m-1) \sum_{n=1}^N \frac{c_L(n)}{n^s}
\\
&= \lim_{N\rightarrow\infty} \sum_{L=1}^{\infty} m!\,S(L-1, m-1) \sum_{n=1}^N \frac{c_L(n)}{n^s}.
\end{align*}
The Stirling numbers of the second kind satisfy the estimate
$m!~\,S(L, m)~\le~m^L$
(cf.\ \cite[Theorem 5.2.C (3f)]{comtet}).
Therefore
\begin{align*}
\left| m!\,S(L-1, m-1)\sum_{n=1}^N \frac{c_L(n)}{n^s} \right|
&\le m!\,S(L-1, m-1) \sum_{n=1}^N \frac{c_L(n)}{|n^s|}
\\
&\le m\,(m-1)^{L-1} \sum_{n=1}^\infty \frac{c_L(n)}{n^{\mathop{\rm Re} s}}
\\
&= m\,(m-1)^{L-1}\,(\zeta(\mathop{\rm Re} s) - 1)^L,
\end{align*}
which are the ($N$-independent) terms of a convergent geometric series (with summation parameter $L$) if
$(m-1)\,(\zeta(\mathop{\rm Re} s) - 1) < 1$. This latter condition is easily seen to
be equivalent to
$\zeta(\mathop{\rm Re} s) < \frac m{m-1}$.
Furthermore, we have pointwise convergence (for fixed $L\in\mathbb N$)
\begin{equation*}
\lim_{N\rightarrow\infty} m!\,S(L-1, m-1) \sum_{n=1}^N \frac{c_L(n)}{n^s}
= m!\,S(L-1, m-1)\,(\zeta(s) - 1)^L,
\end{equation*}
so equation (\ref{eq:DN}) follows by dominated convergence.

It remains to see for which $s\in\mathbb C$ the convergence condition
$\zeta(\mathop{\rm Re} s) < \frac m{m-1}$ holds.
On the real interval $(1, \infty)$, the $\zeta$ function is strictly monotone
decreasing and continuous, with limits
$\zeta(r)\rightarrow\infty$ $(r \rightarrow 1)$ and
$\lim\limits_{r\rightarrow\infty} \zeta(r) = 1$.
Therefore there is a unique $r_m > 1$ such that $\zeta(r_m) = \frac m{m-1}$,
and we have convergence for all $s\in\mathbb C$ with $\mathop{\rm Re} s > r_m$.

If $m\ge 3$ and $r \ge 1 + \log(m-1)/\log\frac 3 2$, then
$r > 2$, so
\begin{equation*}
\log (m-1) \le (r - 1) \log \frac 3 2 < (r - 1) \log \frac 3 2 + \log (r - 1)
\end{equation*}
and therefore
\begin{equation*}
1 + \frac 1{r-1}\left(\frac 32\right)^{1-r} < 1 + \frac 1{m-1} = \frac m{m-1}.
\end{equation*}
The upper bound on $r_m$ in the theorem now follows by observing (using the
convexity of the function $x^{-r}$ $(x > 0)$) that
\begin{align*}
\zeta(r) &= 1 + \sum_{n=2}^\infty \frac 1{n^r}
 < 1 + \sum_{n=2}^\infty \int_{n-\frac 12}^{n+\frac 12} \frac 1{x^r}\,d x
= 1 + \int_{\frac 32}^\infty \frac{d x}{x^r}
= 1 + \frac 1{r-1} \left(\frac 32\right)^{1-r}.
\end{align*}
For the lower bound on $r_m$, we note that if $r \le \log(m-1)/\log 2$, then
$2^r \le m - 1$ and hence
\begin{equation*}
 \zeta(r) \ge 1 + \frac 1{2^r} \ge 1 + \frac 1{m - 1} = \frac m{m-1}
\qedhere
\end{equation*}

Following the calculation in the proof of Theorem \ref{thm:DN} but starting
from equation (\ref{enum22}) instead of equation (\ref{eq:Hylock}) and using
the Dirichlet series for the balanced divisor function $c_j^{(-j)}$
(equation (\ref{eq:Dbal})),
we obtain an alternative form for the Dirichlet series of $\mathcal N_m$,
\begin{equation*}
\sum_{n=1}^\infty \frac{\mathcal N_m(n)}{n^s} = \sum_{L=1}^\infty m!\,S(L,m) \left(1 - \frac 1{\zeta(s)}\right)^L.
\end{equation*}
This series is absolutely convergent for $\mathop{\rm Re} s > r_m$, but we note that the
majorant
\begin{equation*}
\left| m!\,S(L, m) \sum_{n=1}^N \frac{c_L^{(-L)}(n)}{n^s} \right|
\le m!\,S(L, m) \sum_{n=1}^N \frac{b_L(n)}{n^{\mathop{\rm Re} s}}
\end{equation*}
\begin{equation*}
\le m!\,S\left(L,m)\,(\frac{\zeta(\mathop{\rm Re} s)}{\zeta(2 \mathop{\rm Re} s)} - 1\right)^L
\end{equation*}
(using $|c_L^{(-L)}| \le b_L$ and the Dirichlet series in Theorem \ref{thm:Db}) will be summable if and only
if $\frac{\zeta(s)}{\zeta(2s)} < \frac{m+1}m=1+\frac{1}{m}$, which is a slightly different
condition.
