<!-- theorem-1: body_md -->
(Williamson's normal form \@nx\protect\@empty\@nx\csname cite \endcsname{jw01})  \label{wnffd}
 		Let $A$ be a strictly positive matrix of order $2n$. Then there exists a symplectic matrix $M$ of order $2n$ and a strictly positive diagonal matrix D of order $n$ such that
 		$$
 		A = 
 		M^t\begin{bmatrix}
 			D & 0 \\
 			0 & D
 		\end{bmatrix}M.
 		$$
 		Furthermore, D is unique up to the order of its entries.

<!-- theorem-2: body_md -->
(Williamson's normal form \@nx\protect\@empty\@nx\csname cite \endcsname{BJ19}) \label{wnfifd}
 		Let $\mathcal{H}$ be a real separable Hilbert space and $A$ be a strictly positive invertible operator on $ \mathcal{H} \oplus \mathcal{H}$. Then there exists a positive invertible operator $D$ on $\mathcal{H}$ and a symplectic operator $M$ on $\mathcal{H} \oplus \mathcal{H}$ such that 
 		$$
 		A = M^t\begin{bmatrix}
 			D & 0 \\
 			0 & D
 		\end{bmatrix}M.$$ 
 		Furthermore, $D$ is unique up to conjugation with an orthogonal transformation.

<!-- definition-1: body_md -->
\label{def:mhsb}
Let $(X, \mathscr{B}, \mu)$ be a $\sigma$-finite Borel
measure space. A \emph{measurable Hilbert space bundle} over $X$ is an assignment $\lambda \mapsto \relax{h}_\lambda$ of separable real Hilbert spaces together with a countable measurable partition $X = \bigsqcup_{n\geq 1}X_n$ (with $X_n \in \mathscr{B}$) such that $\dim\relax{h}_\lambda = n$, for all $\lambda \in X_n$. The \emph{direct integral} $\mathcal{K} = \int^\oplus_X \relax{h}_\lambda\,d\mu(\lambda)$ is the Hilbert space of measurable sections $v(\lambda)\in\relax{h}_\lambda$ with
$\int_X\|v(\lambda)\|^2\,d\mu(\lambda)<\infty$ and the inner product is given by 
\begin{equation}\langle u,\, v \rangle_{\mathcal{K}} \coloneqq \int_X \langle u(\lambda),\, v(\lambda) \rangle_{\relax{h}_\lambda}\,d\mu(\lambda).\end{equation}

A bounded operator $T$ on $\mathcal{K}$ is \emph{decomposable} if $(Tv)(\lambda) = T(\lambda)v(\lambda)$ $\mu$-a.e.\ for bounded operators $T(\lambda)$ on $\relax{h}_\lambda$; denoted as \begin{equation}T = \int^\oplus_{X}
T(\lambda)\,d\mu(\lambda).\end{equation}
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:fiberwise-symp}
Given a bundle $\lambda\mapsto\relax{h}_\lambda$ over $(X,\mu)$, set $\mathcal{K}\cong \int^\oplus_X (\relax{h}_\lambda\oplus\relax{h}_\lambda) \,d\mu(\lambda)$. The \emph{fibrewise symplectic form} is $\langle \cdot, \mathbf{J}_\mathcal{K} \cdot \rangle$, where
\[
  \mathbf{J}_\mathcal{K} \coloneqq \int^\oplus_X J_\lambda\,d\mu(\lambda),
  \qquad
  J_\lambda \coloneqq
  \begin{pmatrix}0 & I_{\relax{h}_\lambda}\\
    -I_{\relax{h}_\lambda} & 0\end{pmatrix},
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
satisfying $\mathbf{J}_\mathcal{K}^t = -\mathbf{J}_\mathcal{K}$ and
$\mathbf{J}_\mathcal{K}^2 = -\mathrm{I}_{\mathcal{K}}$.

<!-- theorem-3: body_md -->
\label{thm:di-wnf}
Let $(\mathbf{H}, \langle \cdot, \cdot \rangle, J)$ be a real separable symplectic Hilbert space, and let $A \in \mathcal{B}(\mathbf{H})$ be a bounded positive operator such that $\ker(A)$ is $J$-invariant. Then there exist:
\begin{enumerate}
    \@par\hangindent 2em A $\sigma$-finite Borel measure $\mu$ on $[0, \infty)$ and a measurable Hilbert space bundle $\lambda \mapsto \relax{h}_\lambda$ over $[0, \infty)$ with direct integral space 
    \[
    \mathcal{K} = \int_{[0,\infty)}^\oplus \relax{h}_\lambda \, d\mu(\lambda);
    \relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
    \@par\hangindent 2em A real Hilbert space $\relax{h}_0 \subset \mathbf{H}_0$ such that $\mathbf{H}_0 = \relax{h}_0 \oplus J_0 \relax{h}_0$;
    \@par\hangindent 2em A closed, densely defined linear operator 
    \[
    \Phi = \Phi_0 \oplus \Phi_R' : \operatorname{dom}(\Phi) \subset \mathbf{H} \longrightarrow \mathcal{K}_{\mathrm{tot}},
    \relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
    where $\mathcal{K}_{\mathrm{tot}} = (\relax{h}_0 \oplus \relax{h}_0) \oplus (\mathcal{K} \oplus \mathcal{K})$, $\Phi_0 : \mathbf{H}_0 \to \relax{h}_0 \oplus \relax{h}_0$ is a bounded orthogonal isomorphism, and $\Phi_R' = M^{-1/2}\Phi_R$ is a densely defined, closable operator on $\mathcal{H}_R$ with domain $\operatorname{dom}(\Phi_R') = \{ x \in \mathcal{H}_R : \Phi_R x \in \mathrm{ran}(M^{1/2}) \}$;
\end{enumerate}
such that on $\operatorname{dom}(\Phi) = \mathbf{H}_0 \oplus \operatorname{dom}(\Phi_R')$, the following identities hold:
\[
A = \Phi^t M_A \Phi \quad \text{and} \quad \Phi J \Phi^t = J_{\mathcal{K}_{\mathrm{tot}}},
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
where $M_A = 0_{\relax{h}_0 \oplus \relax{h}_0} \oplus M$, $M = \int_{[0,\infty)}^\oplus \lambda \cdot I_{\relax{h}_\lambda \oplus \relax{h}_\lambda} \, d\mu(\lambda)$, and $J_{\mathcal{K}_{\mathrm{tot}}} = J_{\relax{h}_0} \oplus J_{\mathcal{K}}$ with $J_{\relax{h}_0} = \begin{bmatrix} 0 & I_{\relax{h}_0} \\ -I_{\relax{h}_0} & 0 \end{bmatrix}$. 
Furthermore, $\Phi$ is injective with dense range, and the measure class $[\mu]$ and multiplicity function $\lambda \mapsto \dim \relax{h}_\lambda$ are uniquely determined by $A$.

We prove the theorem by elaborating the ideas given in Section {{\rm\expandafter\@setref\csname r@secproofoutline\endcsname\@firstoftwo{secproofoutline}}}. We construct the transformation $\Phi = \Phi_0 \oplus \Phi_R'$ in two distinct steps:

\smallskip
\noindent\textbf{Step 1: The Kernel Component $\Phi_0$.} \\
Since $J_0^2 = -I_{\mathbf{H}_0}$ and $J_0^t = -J_0$, Theorem {{\rm\expandafter\@setref\csname r@appen:thm_polar\endcsname\@firstoftwo{appen:thm_polar}}} yields a maximal isotropic closed subspace $\relax{h}_0 \subset \mathbf{H}_0$ such that 
\[
\mathbf{H}_0 = \relax{h}_0 \oplus J_0 \relax{h}_0.
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
Define the map $\Phi_0 : \mathbf{H}_0 \to \relax{h}_0 \oplus \relax{h}_0$ by $\Phi_0(u + J_0 v) = (u, v)$ for $u, v \in \relax{h}_0$. Then $\Phi_0$ is a bounded orthogonal isomorphism satisfying $\Phi_0^t = \Phi_0^{-1}$. Under $\Phi_0$, we have $\Phi_0 J_0 \Phi_0^t = J_{\relax{h}_0}$, where $J_{\relax{h}_0} = \begin{bmatrix} 0 & I_{\relax{h}_0} \\ -I_{\relax{h}_0} & 0 \end{bmatrix}$. Since $A|_{\mathbf{H}_0} = 0$, we trivially have
\[
A|_{\mathbf{H}_0} = \Phi_0^t \cdot 0 \cdot \Phi_0 = 0.
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi

\smallskip
\noindent\textbf{Step 2: The Range Component $\Phi_R'$.} \\
Note that the operator $A_R$ is bounded, positive, and injective operator on $\mathbf{H}_R$. We work on $\mathbf{H}_R$ henceforth and adjoin the kernel part at the end.

\begin{itemize}
    \@par\hangindent 2em \underline{\textbf{Step 1:} The $A_R$-inner product space $\mathcal{H}_A$ and the Riesz factorisation.} 
    \\ 
    Due to the positivity and injectivity of $A_R$, the bilinear form $\langle x,\, y \rangle_A := \langle A_R x,\, y \rangle_{\mathbf{H}_R}$ is an inner product on $\mathbf{H}_R$, which need not be complete. Let $\mathcal{H}_A$ denote its completion and $\iota: \mathbf{H}_R\hookrightarrow\mathcal{H}_A$ be the inclusion. Since for all $x \in \mathbf{H}_R, \|\iota(x)\|_A^2 = \langle A_R x,\, x \rangle \leq \|A_R\|\|x\|^2$, the map $\iota$ is bounded. Moreover, for $u\in\mathcal{H}_A, y\in\mathbf{H}_R, |\langle u,\, \iota(y) \rangle_A|\leq\|u\|_A\|y\|_A\leq
\|u\|_A\|A_R\|^{1/2}\|y\|$. Hence, the adjoint $\iota^t:\mathcal{H}_A\to\mathbf{H}_R$, defined by $\langle \iota^t u,\, y \rangle_{\mathbf{H}_R} = \langle u,\, \iota y \rangle_A$, for $u\in\mathcal{H}_A$,
$y\in\mathbf{H}_R$, satisfies $\|\iota^t\| \leq \|A_R\|^{1/2}$. Also since $\langle \iota^t\iota x,\, y \rangle = \langle x,\, y \rangle_A = \langle A_R x,\, y \rangle$ for all $x,y\in\mathbf{H}_R$,
\begin{equation}
\label{eq:factor}
  A_R = \iota^t\iota.
\end{equation}
 In addition $\iota$ has a dense range in $\mathcal{H}_A$: if $\langle u,\, x \rangle_A = 0$ for all $x\in\mathbf{H}_R$, taking $u\in\mathbf{H}_R$ gives $\|u\|_A^2 = 0$, so $u = 0$.


\@par\hangindent 2em \underline{\textbf{Step 2: }Extending $T_R = J_R A_R$ to a bounded skew-symmetric injective operator}  \\ \underline{$\widehat{T}$ on $\mathcal{H}_A$.} \\
$T_R$ is injective due to the injectivity of $J_R$ and $A_R$ on $\mathbf{H}_R$. Also since $J_R$ is skew-symmetric on $\mathbf{H}_R$, we have
\[
  \langle T_R x,\, y \rangle_A
  = \langle A_R J_R A_R x,\, y \rangle
  = \langle J_R A_R x,\, A_R y \rangle
  = -\langle A_R x,\, J_R A_R y \rangle
  = -\langle x,\, T_R y \rangle_A,
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi that is, the operator $T_R = J_RA_R$ is skew-symmetric with respect to $\langle \cdot,\, \cdot \rangle_A$. Moreover, since $J_R$ is orthogonal and
$\|A_R x\|^2 \leq \|A_R\|\langle A_R x,\, x \rangle = \|A_R\|\|x\|_A^2$, we have
\[
  \|T_R x\|_A^2
  = \langle A_R J_R A_R x,\, J_R A_R x \rangle
  \leq \|A_R\|\|A_R x\|^2
  \leq \|A_R\|^2\|x\|_A^2,
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi that is $T_R$ is a bounded operator on $\mathbf{H}_R$. Hence, by the density of $\mathbf{H}_R$ in $\mathcal{H}_A$ and skew-symmetry, $T_R$ extends to $\widehat{T}:\mathcal{H}_A\to\mathcal{H}_A$ which is bounded, skew-symmetric and injective on $\mathcal{H}_A$.

\@par\hangindent 2em \underline{\textbf{Step 3:} Spectral decomposition of $\widehat{S} := -\widehat{T}^2$.}
\\
Since $\widehat{T}$ is bounded, skew-symmetric and injective, the operator $\widehat{S} = (\widehat{T})^t\widehat{T} = -\widehat{T}^2$ is bounded, positive and injective on $\mathcal{H}_A$. Thus $\sigma(\widehat{S}) \subseteq [0,\infty)$. Since $\widehat{T}$ is skew-adjoint, it commutes with
$\widehat{S}$:
\begin{equation}
\label{eq:commute}
  \widehat{T}\,\widehat{S}
  = \widehat{T}(-\widehat{T}^2)
  = -\widehat{T}^3
  = (-\widehat{T}^2)\widehat{T}
  = \widehat{S}\,\widehat{T}.
\end{equation}
Now apply the spectral theorem \@nx\protect\@empty\@nx\csname cite \endcsname[Theorem~VIII.4]{RS81} and
multiplicity theory \@nx\protect\@empty\@nx\csname cite \endcsname[Section~VIII.2]{RS81} to $\widehat{S}$.
There exist a $\sigma$-finite Borel measure $\nu_0$ on $[0,\infty)$
and a measurable bundle $s\mapsto\widetilde{\relax{h}}_s$ with
\begin{equation}
\label{eq:S-decomp}
  \mathcal{H}_A \;\cong\;
  \int^\oplus_{[0,\infty)}\widetilde{\relax{h}}_s\,d\nu_0(s),
  \qquad
  \widehat{S} \;\longleftrightarrow\;
  \int^\oplus_{[0,\infty)} s\cdot I_{\widetilde{\relax{h}}_s}\,d\nu_0(s).
\end{equation}
By \eqref{eq:commute} and the commutant theorem for direct integrals
\@nx\protect\@empty\@nx\csname cite \endcsname[Theorem~XIII.83]{RS78}, $\widehat{T}$ is decomposable:
\begin{equation}
\label{eq:T-decomp}
  \widehat{T} \;\longleftrightarrow\;
  \int^\oplus_{[0,\infty)} \tau_s\,d\nu_0(s),
\end{equation}
where $\tau_s^t = -\tau_s$ and $-\tau_s^2 = s\cdot
I_{\widetilde{\relax{h}}_s}$ on each fibre.

\@par\hangindent 2em \underline{\textbf{Step 4:} Fibre-level structure and the isomorphism $\Gamma_0$.} \\
Since $s=0$ is an element of the continuous spectrum, the set \(\{0\}\) is null with respect to measure \(\nu_0\). Thus, we are justified in removing it from the domain of our direct integral decomposition henceforth. 
\\ Fix $s>0$. The operator $j_s := s^{-1/2}\tau_s$ satisfies $j_s^t = -j_s$ and $j_s^2= - I_{\widetilde{\relax{h}}_s}$, making it a
\emph{complex structure} on $\widetilde{\relax{h}}_s$. Therefore, we choose any closed subspace $\relax{h}_s\subset\widetilde{\relax{h}}_s$ maximal subject
to $\relax{h}_s\perp j_s\relax{h}_s$ when $s>0$ (see Theorem {{\rm\expandafter\@setref\csname r@appen:thm_polar\endcsname\@firstoftwo{appen:thm_polar}}}). By maximality,
$\widetilde{\relax{h}}_s = \relax{h}_s\oplus j_s\relax{h}_s$, and
\begin{equation}
\label{eq:gamma}
  \gamma_s:\relax{h}_s\oplus\relax{h}_s\xrightarrow{\;\sim\;}\widetilde{\relax{h}}_s,
  \qquad
  \gamma_s(u,v) := u + j_s v,
\end{equation}
is a real orthogonal isomorphism (since $\langle u,j_s v\rangle = 0$ by maximality and $j_s$ is orthogonal). Under $\gamma_s$:
\begin{equation}
\label{eq:fiber-tau}
  \gamma_s^{-1}\,\tau_s\,\gamma_s = s^{1/2}\,J_{\relax{h}_s},
  \qquad
  J_{\relax{h}_s} :=
  \begin{pmatrix}0&I_{\relax{h}_s}\\-I_{\relax{h}_s}&0\end{pmatrix},
\end{equation}
since $\tau_s(u+j_s v) = s^{1/2}j_s(u+j_s v) = s^{1/2}(j_s u-v)$,
giving $\gamma_s^{-1}\tau_s\gamma_s(u,v) = s^{1/2}(-v,u) =
s^{1/2}J_{\relax{h}_s}(u,v)$.

\noindent Now we move to the measure $\mu$. Fix $s\in (0,\infty)$, let $\lambda = s^{1/2}$, push $\nu_0$ forward under $s\mapsto s^{1/2}$ to a measure $\mu$ on $(0,\infty)$, set
$\relax{h}_\lambda := \relax{h}_{\lambda^2}$ and $\gamma_\lambda :=
\gamma_{\lambda^2}$. Assemble the fibrewise isomorphisms into a global real orthogonal isomorphism
\begin{equation}
\label{eq:Gamma0}
  \Gamma_0 : \mathcal{H}_A \xrightarrow{\;\sim\;}
  \mathcal{K}\oplus\mathcal{K} :=\int^\oplus_{(0,\infty)}
  (\relax{h}_\lambda\oplus\relax{h}_\lambda)\,d\mu(\lambda).
\end{equation}
Under $\Gamma_0$, equations \eqref{eq:S-decomp}--\eqref{eq:T-decomp}
become:
\begin{align}
  \Gamma_0\,\widehat{S}\,\Gamma_0^{-1}
  &= M^2 := \int^\oplus_{(0,\infty)} \lambda^2\cdot
    I_{\relax{h}_\lambda\oplus\relax{h}_\lambda}\,d\mu(\lambda),
  \label{eq:S-fiberwise}\\
  \Gamma_0\,\widehat{T}\,\Gamma_0^{-1}
  &= M_J :=\int^\oplus_{(0,\infty)} \lambda\cdot J_{\relax{h}_\lambda}\,d\mu(\lambda)
    = M\,\mathbf{J}_\mathcal{K} = \mathbf{J}_\mathcal{K}\,M,
  \label{eq:T-fiberwise}
\end{align}
where $M =\int^\oplus_{(0,\infty)} \lambda\cdot I_{\relax{h}_\lambda\oplus\relax{h}_\lambda}
\,d\mu(\lambda)$, and the commutativity $M\mathbf{J}_\mathcal{K} = \mathbf{J}_\mathcal{K} M$
holds fibrewise since both are diagonal in the direct integral
structure.

\@par\hangindent 2em \underline{\textbf{Step 5:} Construction of $\Phi_R$.}
\\
Define the isometry
\begin{equation}
\label{eq:PhiR-def}
  \Phi_R := \Gamma_0\circ\iota : \mathbf{H}_R \longrightarrow \mathcal{K}\oplus\mathcal{K}
\end{equation}
Since $\iota:\mathbf{H}_R\to\mathcal{H}_A$ and $\Gamma_0:\mathcal{H}_A\to\mathcal{K}\oplus\mathcal{K}$ are both bounded, $\Phi_R$ is bounded. Since $\iota$ has dense range in
$\mathcal{H}_A$ and $\Gamma_0$ is an isomorphism, $\Phi_R$ has dense range. Since $\iota$ is injective (as $\|\iota x\|_A = \|x\|_A > 0$ for $x\neq 0$) and $\Gamma_0$ is injective, $\Phi_R$ is injective. Next we show that $\Phi_R^t$ is densely defined. For $z = \Phi_R y
\in \operatorname{ran}(\Phi_R)$ and any $x\in\mathbf{H}_R$:
\[
  |{\langle z,\, \Phi_R x \rangle_{\mathcal{K}}}|
  = |\langle \Phi_R y,\, \Phi_R x \rangle_{\mathcal{K}}|
  = |\langle y,\, x \rangle_A|
  \leq \|y\|_A\|x\|_A
  \leq \|y\|_A\|A_R\|^{1/2}\|x\|,
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
so $x\mapsto\langle z,\, \Phi_R x \rangle_{\mathcal{K}}$ is bounded for each
$z\in\operatorname{ran}(\Phi_R)$. Hence $\operatorname{ran}(\Phi_R)\subseteq\operatorname{dom}(\Phi_R^t)$.
Since $\operatorname{ran}(\Phi_R)$ is dense in $\mathcal{K}$, $\operatorname{dom}(\Phi_R^t)$
is dense.
 Next we prove that $A_R = \Phi_R^t\Phi_R$. For all $x,y\in\mathbf{H}_R$,
\begin{align*}
  \langle \Phi_R^t\Phi_R x,\, y \rangle_{\mathbf{H}_R}
  &= \langle \Phi_R x,\, \Phi_R y \rangle_{\mathcal{K}}
   = \langle \Gamma_0\iota x,\, \Gamma_0\iota y \rangle_{\mathcal{K}} \\
   &= \langle \iota x,\, \iota y \rangle_A
   = \langle x,\, y \rangle_A \\
   &= \langle A_R x,\, y \rangle_{\mathbf{H}_R}
\end{align*}
$\Rightarrow \Phi_R^t\Phi_R = A_R$.
\\
Now for $y = \Phi_R x\in\operatorname{ran}(\Phi_R)$, note that $\Phi_R^t y =
\Phi_R^t\Phi_R x = A_R x$. Then, using \eqref{eq:T-fiberwise}
in the form $\Gamma_0\widehat{T} = M_J\Gamma_0$:
\begin{align*}
  \Phi_R J_R\Phi_R^t y
  &= \Phi_R J_R A_R x
   = \Gamma_0\iota(J_R A_R x)
   = \Gamma_0\widehat{T}(\iota x) \\
   &= M_J\Gamma_0\iota x
   = M_J\Phi_R x \\
   &= M\mathbf{J}_\mathcal{K} y.
\end{align*}
Since $\operatorname{ran}(\Phi_R)$ is dense in $\mathcal{K}$ and both sides
are continuous (the right side is bounded since $M$ and $\mathbf{J}_\mathcal{K}$ are bounded), the above identity extends to all of $\mathcal{K}$, that is, $\Phi_R J_R\Phi_R^t = M\mathbf{J}_\mathcal{K}$.

\@par\hangindent 2em \underline{\textbf{Step 6:} The equivalent Williamson's normal form}
\\
Define $\Phi_R^\prime := M^{-1/2}\Phi_R$. Then the domain
\[
  \operatorname{dom}(\Phi_R^\prime) := \{x\in\mathbf{H}_R : \Phi_R x\in\operatorname{ran}(M^{1/2})\}.
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
is dense. This can be seen as follows. Define the subspace $
\mathcal{D} := J_R \Phi_R^t \bigl(\operatorname{ran}(M^{1/2})\bigr).$ We first show that $\mathcal{D} \subseteq \operatorname{dom}(\Phi_R^\prime)$. Let $x \in \mathcal{D}$. Then $x = J_R \Phi_R^t M^{1/2} z$ for some $z \in \mathcal{K} \oplus \mathcal{K}$. Using the symplectic identity $\Phi_R J_R \Phi_R^t = M J_{\mathcal{K}}$ and the commutativity of $M$ and $J_{\mathcal{K}}$, we obtain:
$$
\Phi_R x = \Phi_R J_R \Phi_R^t M^{1/2} z = M J_{\mathcal{K}} M^{1/2} z = M^{1/2} (M J_{\mathcal{K}} z).
$$
Since $M J_{\mathcal{K}} z \in \mathcal{K} \oplus \mathcal{K}$, it follows that $\Phi_R x \in \operatorname{ran}(M^{1/2})$, and hence $x \in \operatorname{dom}(\Phi_R^\prime)$. Thus,
\begin{equation} \label{eq:D_subset_dom}
    \mathcal{D} \subseteq \operatorname{dom}(\Phi_R^\prime).
\end{equation}
Now it remains to show that $\mathcal{D}$ is dense in $H_R$. Since $M$ is positive and injective, $M^{1/2}$ is also injective, so
$
\overline{\operatorname{ran}(M^{1/2})} = (\ker M^{1/2})^\perp = \mathcal{K} \oplus \mathcal{K},
$ which implies that $\operatorname{ran}(M^{1/2})$ is dense in $\mathcal{K} \oplus \mathcal{K}$. Because $\Phi_R^t$ is a bounded linear operator, the image of a dense set under $\Phi_R^t$ is dense in $\operatorname{ran}(\Phi_R^t)$, so
\[
\overline{\Phi_R^t \bigl(\operatorname{ran}(M^{1/2})\bigr)} = \overline{\operatorname{ran}(\Phi_R^t)}.
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
Furthermore, for any bounded operator between Hilbert spaces, $\overline{\operatorname{ran}(\Phi_R^t)} = (\ker \Phi_R)^\perp$. By the injectivity of $\Phi_R$, we have $\ker \Phi_R = \{0\}$, which yields
\[
\overline{\operatorname{ran}(\Phi_R^t)} = H_R.
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
Finally, since $J_R$ is orthogonal, it preserves closures of dense subspaces, that is,
\[
\overline{\mathcal{D}} = \overline{J_R \Phi_R^t \bigl(\operatorname{ran}(M^{1/2})\bigr)} = J_R \overline{\Phi_R^t \bigl(\operatorname{ran}(M^{1/2})\bigr)} = J_R H_R = H_R.
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
Combining this with \eqref{eq:D_subset_dom}, we conclude that $\overline{\operatorname{dom}(\Phi_R^\prime)} = H_R$, establishing that $\Phi_R^\prime$ is densely defined.
\\
Next we prove that the operator $\Phi_R^\prime$ is closable. Since $(M^{-1/2}\Phi_R)^t\supseteq\Phi_R^t M^{-1/2}$, we have $\operatorname{dom}((\Phi_R^\prime)^t)
\supseteq \operatorname{dom}(M^{-1/2}) = \operatorname{ran}(M^{1/2})$, which is dense. Hence $(\Phi_R^\prime)^t$ is densely defined and $\Phi_R^\prime$ is closable. 
\\ Finally, we observe the following. For $x,y\in\operatorname{dom}(\Phi_R^\prime)$, using
self-adjointness of $M^{-1/2}$ and $M^{-1/2}MM^{-1/2} = I$ on $\operatorname{ran}(M^{1/2})$:
\begin{align*}
  \langle (\Phi_R^\prime)^t M\Phi_R^\prime x,\, y \rangle
  &= \langle M\Phi_R^\prime x,\, \Phi_R^\prime y \rangle_{\mathcal{K}}
   = \langle M\cdot M^{-1/2}\Phi_R x,\, M^{-1/2}\Phi_R y \rangle_{\mathcal{K}}\\
  &= \langle \Phi_R x,\, M^{-1/2}MM^{-1/2}\Phi_R y \rangle_{\mathcal{K}} \\
  &= \langle \Phi_R x,\, \Phi_R y \rangle_{\mathcal{K}} = \langle \Phi_R^t \Phi_Rx,\, y \rangle_{\mathcal{K}} \\
  &= \langle A_Rx,\, y \rangle_{\mathcal{K}}
\end{align*}
$\Rightarrow (\Phi_R^\prime)^t M\Phi_R^\prime = A_R$. For symplecticity, take $x\in\operatorname{dom}(M^{-1/2}) = \operatorname{ran}(M^{1/2})$. Then,  
\begin{align}
   M^{-1/2}\Phi_R J_R\Phi_R^t M^{-1/2}x
   = M^{-1/2}(M\mathbf{J}_\mathcal{K})M^{-1/2}x.\notag
\end{align}
Since $M^{-1/2}M = M^{1/2}$ and $\mathbf{J}_\mathcal{K}$ commutes with $M$:
\begin{align}
  M^{-1/2}(M\mathbf{J}_\mathcal{K})M^{-1/2}
  &= M^{-1/2}M^{1/2}\mathbf{J}_\mathcal{K} M^{1/2}M^{-1/2}
   = \mathbf{J}_\mathcal{K}
   \qquad\text{on }\operatorname{ran}(M^{1/2}).
  \label{eq:form1-symp-clean}
\end{align}
Hence $\Phi_R^\prime J_R(\Phi_R^\prime)^t = \mathbf{J}_\mathcal{K}$ on the dense domain
$\operatorname{ran}(M^{1/2})\subseteq\operatorname{dom}(\Phi_R^t)$.


\@par\hangindent 2em \underline{\textbf{Step 8:} Uniqueness.}
\\
Suppose that
\[
\widetilde{\Phi}_R:H_R\longrightarrow\widetilde{\mathcal K}
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
is another bounded operator with dense range satisfying
\[
\widetilde{\Phi}_R^t\widetilde{\Phi}_R=A_R,
\qquad
\widetilde{\Phi}_R J_R\widetilde{\Phi}_R^t
=
\widetilde M J_{\widetilde{\mathcal K}},
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
where
\[
\widetilde M
=
\int_{(0,\infty)}^\oplus
\lambda I_{\widetilde{\relax h}_\lambda}
\,d\widetilde\mu(\lambda).
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi

Define
\[
U_0:\operatorname{ran}(\Phi_R)
\longrightarrow
\operatorname{ran}(\widetilde{\Phi}_R),
\qquad
U_0(\Phi_Rx):=\widetilde{\Phi}_Rx.
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
Since $\Phi_R$ is injective, $U_0$ is well-defined. Moreover,
\[
\begin{aligned}
\|U_0\Phi_Rx\|_{\widetilde{\mathcal K}}^2
&=
\|\widetilde{\Phi}_Rx\|_{\widetilde{\mathcal K}}^2\\
&=
\langle
\widetilde{\Phi}_R^t\widetilde{\Phi}_Rx,x
\rangle_{H_R}\\
&=
\langle A_Rx,x\rangle_{H_R}\\
&=
\langle
\Phi_R^t\Phi_Rx,x
\rangle_{H_R}\\
&=
\|\Phi_Rx\|_{\mathcal K}^2.
\end{aligned}
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
Hence $U_0$ is an isometry. Since both $\operatorname{ran}(\Phi_R)$ and
$\operatorname{ran}(\widetilde{\Phi}_R)$ are dense, $U_0$ extends uniquely to a unitary operator
\(
U:\mathcal K\longrightarrow\widetilde{\mathcal K}.
\)
By construction,
\(
\widetilde{\Phi}_R=U\Phi_R.
\)
Consequently,
$$
\widetilde M J_{\widetilde{\mathcal K}} =
\widetilde{\Phi}_R J_R\widetilde{\Phi}_R^t =
U\Phi_RJ_R\Phi_R^tU^t = U(MJ_{\mathcal K})U^t.
$$
Since $M$ commutes with $J_{\mathcal K}$,
$J_{\mathcal K}^t=-J_{\mathcal K}$, and
$J_{\mathcal K}^2=-I$, we have
$$
(MJ_{\mathcal K})^t(MJ_{\mathcal K}) =
(-J_{\mathcal K}M)(MJ_{\mathcal K}) =
-J_{\mathcal K}M^2J_{\mathcal K} = M^2.
$$ Therefore \(
|MJ_{\mathcal K}|=M. \) Similarly,
\(
|\widetilde M J_{\widetilde{\mathcal K}}|
=
\widetilde M.
\) Using the unitary invariance of the functional calculus we have,
\[
\widetilde M
=
|\widetilde M J_{\widetilde{\mathcal K}}|
=
U|MJ_{\mathcal K}|U^t
=
UMU^t.
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
Thus $M$ and $\widetilde M$ are unitarily equivalent. By the uniqueness of spectral measure class and multiplicity function for self-adjoint operators
\@nx\protect\@empty\@nx\csname cite \endcsname[Section~VIII.2]{RS81}, $[\mu] = [\widetilde\mu]$ and $\dim\relax{h}_\lambda = \dim\widetilde\relax{h}_\lambda$ $\mu$-a.e.
\end{itemize}

\smallskip
\noindent\textbf{Step 3: Direct Sum Construction on $\mathcal{H} = \mathbf{H}_0 \oplus \mathcal{H}_R$.} \\
Define $\mathcal{K}_{\mathrm{tot}} = (\relax{h}_0 \oplus \relax{h}_0) \oplus (\mathcal{K} \oplus \mathcal{K})$ and the operator 
\[
\Phi = \Phi_0 \oplus \Phi_R' : \mathbf{H}_0 \oplus \operatorname{dom}(\Phi_R') \subset \mathcal{H} \longrightarrow \mathcal{K}_{\mathrm{tot}}.
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
Since $\Phi_0$ is bounded and $\operatorname{dom}(\Phi_R')$ is dense in $\mathcal{H}_R$, $\operatorname{dom}(\Phi) = \mathbf{H}_0 \oplus \operatorname{dom}(\Phi_R')$ is dense in $\mathcal{H}$. Taking the transpose (adjoint) in block-diagonal form, $\Phi^t = \Phi_0^t \oplus (\Phi_R')^t$.

Combining the kernel and range results:
\[
\Phi^t M_A \Phi = \begin{bmatrix} \Phi_0^t & 0 \\ 0 & (\Phi_R')^t \end{bmatrix} \begin{bmatrix} 0 & 0 \\ 0 & M \end{bmatrix} \begin{bmatrix} \Phi_0 & 0 \\ 0 & \Phi_R' \end{bmatrix} = \begin{bmatrix} 0 & 0 \\ 0 & (\Phi_R')^t M \Phi_R' \end{bmatrix} = 0 \oplus A_R = A.
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
Similarly, for the symplectic structure:
\[
\Phi J \Phi^t = \begin{bmatrix} \Phi_0 J_0 \Phi_0^t & 0 \\ 0 & \Phi_R' J_R (\Phi_R')^t \end{bmatrix} = \begin{bmatrix} J_{\relax{h}_0} & 0 \\ 0 & J_{\mathcal{K}} \end{bmatrix} = J_{\mathcal{K}_{\mathrm{tot}}}.
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
Injectivity and density of $\mathrm{ran}(\Phi)$ follow directly from the injectivity and dense ranges of $\Phi_0$ and $\Phi_R^\prime$.

The proof has five stages.

\underline{\textbf{Stage I:} Extending $T_A, T_B, T_C$ to $\mathcal{H}_C$.}
\\
For $x \in \mathbf{H}_R$,
\[
\|T_C x\|_C^2 = \langle J_R C_R x,\, J_R C_R x \rangle_C
= \langle C_R J_R C_R x,\, J_R C_R x \rangle_{\mathbf{H}_R} \leq \|C_R\|^2 \|x\|_C^2,
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
so $T_C$ is bounded. Also, for $x,y \in \mathbf{H}_R$,
\[
\langle T_C x,\, y \rangle_C = \langle C_R J_R C_R x,\, y \rangle_{\mathbf{H}_R}
= -\langle x,\, T_C y \rangle_C,
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
so $T_C$ is skew-symmetric. Hence, by the density of $\mathbf{H}_R$ in $\mathcal{H}_C$, $T_C$ extends to a bounded, skew-symmetric, and injective operator $\widehat{T}_C$ on $\mathcal{H}_C$. Similarly, using $A_R \leq C_R$ and $B_R \leq C_R$, $T_A$ and $T_B$ extend to bounded operators $\widehat{T}_A$ and $\widehat{T}_B$ on $\mathcal{H}_C$. By linearity, density, and the commutativity hypothesis, we have
\[
\widehat{T}_A + \widehat{T}_B = \widehat{T}_C,\qquad
\widehat{T}_A \widehat{T}_B = \widehat{T}_B \widehat{T}_A,
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
and $\widehat{T}_A,\widehat{T}_B$ commute with $\widehat{S}_C := -\widehat{T}_C^2$.

\underline{\textbf{Stage II:} Spectral decomposition of $\widehat{S}_C$.}
\\
Define $\widehat{S}_C := -\widehat{T}_C^2 = \widehat{T}_C^t\widehat{T}_C$. This is a bounded, positive, self-adjoint, injective operator on $\mathcal{H}_C$ with spectrum in $[0,\infty)$. Applying the spectral theorem \@nx\protect\@empty\@nx\csname cite \endcsname[Theorem~VIII.4]{RS81} and multiplicity theory \@nx\protect\@empty\@nx\csname cite \endcsname[Section~VIII.2]{RS81} to $\widehat{S}_C$, there exist a $\sigma$-finite Borel measure $\mu_1$ on $[0,\infty)$ and an orthogonal isomorphism
\[
\Gamma_0 : \mathcal{H}_C \xrightarrow{\sim} 
\mathcal{K} := \int^\oplus_{[0,\infty)} \widetilde{\relax{h}}_\lambda \, d\mu_1(\lambda)
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
such that $\Gamma_0 \widehat{S}_C \Gamma_0^{-1} = \Lambda_C^2$ (multiplication by $\lambda^2$). Since $\widehat{T}_A$ and $\widehat{T}_B$ commute with $\widehat{S}_C$, they are decomposable by the commutant theorem for direct integrals \@nx\protect\@empty\@nx\csname cite \endcsname[Theorem~XIII.83]{RS78}:
\[
\Gamma_0 \widehat{T}_A \Gamma_0^{-1} = \int^\oplus_{[0,\infty)} \tau^\lambda_A \, d\mu_1(\lambda),\qquad
\Gamma_0 \widehat{T}_B \Gamma_0^{-1} = \int^\oplus_{[0,\infty)} \tau^\lambda_B \, d\mu_1(\lambda).
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
On each fibre, $\tau^\lambda_C=\tau^\lambda_A+\tau^\lambda_B$, $[\tau^\lambda_A,\tau^\lambda_B]=0$, and $(\tau^\lambda_C)^t\tau^\lambda_C=\lambda^2 I$.

\underline{\textbf{Stage III:} Fibre-level analysis and splitting.}
\\
As before, we omit the $\{0\}$ from our domain since it has measure zero. Fix $\lambda>0$ and set $j_\lambda := \lambda^{-1}\tau^\lambda_C$. From $(\tau^\lambda_C)^t\tau^\lambda_C = \lambda^2 I$ and $(\tau^\lambda_C)^t = -\tau^\lambda_C$, we have $j_\lambda^t=-j_\lambda$ and $j_\lambda^2=-I_{\widetilde{\relax{h}}_\lambda}$, so $j_\lambda$ is a complex structure. By the standard complex structure polarisation (Theorem~{{\rm\expandafter\@setref\csname r@appen:thm_polar\endcsname\@firstoftwo{appen:thm_polar}}}), there exists a splitting $\widetilde{\relax{h}}_\lambda = \relax{h}_\lambda \oplus j_\lambda \relax{h}_\lambda$ and an isomorphism $\gamma_\lambda : \relax{h}_\lambda \oplus \relax{h}_\lambda \xrightarrow{\sim} \widetilde{\relax{h}}_\lambda$ such that $\gamma_\lambda^{-1} j_\lambda \gamma_\lambda = J_{\relax{h}_\lambda}$. Define $a_\lambda := -j_\lambda \tau^\lambda_A$ and $b_\lambda := -j_\lambda \tau^\lambda_B$. Then
\[
\gamma_\lambda^{-1} \tau_A^\lambda \gamma_\lambda = J_{\relax{h}_\lambda}a_\lambda,\qquad
\gamma_\lambda^{-1} \tau_B^\lambda \gamma_\lambda = J_{\relax{h}_\lambda}b_\lambda,
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
and $a_\lambda,b_\lambda$ are bounded, positive, self-adjoint operators satisfying
\[
a_\lambda + b_\lambda = \lambda I,\qquad
[a_\lambda, b_\lambda] = 0,\qquad
[a_\lambda, j_\lambda] = [b_\lambda, j_\lambda] = 0.
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi

\underline{\textbf{Stage IV:} Second-stage diagonalisation (fibrewise).}
\\
We now diagonalise $a_\lambda$ within each fibre. In the $\gamma_\lambda$-basis, $a_\lambda$ is a positive operator commuting with $J_{\relax{h}_\lambda}$. Applying the real spectral theorem, since the spectral projections commute with $J_{\relax{h}_\lambda}$, $J_{\relax{h}_\lambda}$ decomposes fibrewise as the canonical symplectic operator. Thus, for each $\lambda>0$, there exist a measure $\nu_\lambda$ on $[0,\lambda]$ and an isomorphism
\[
V_\lambda : \widetilde{\relax{h}}_\lambda \xrightarrow{\sim}
\int^\oplus_{[0,\lambda]} (\relax{h}^{(\lambda)}_\alpha \oplus \relax{h}^{(\lambda)}_\alpha) \, d\nu_\lambda(\alpha)
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
such that
\[
a_\lambda \leftrightarrow \int^\oplus_{[0,\lambda]} \alpha\cdot I\,d\nu_\lambda(\alpha),\qquad
b_\lambda \leftrightarrow \int^\oplus_{[0,\lambda]} (\lambda-\alpha)\cdot I\,d\nu_\lambda(\alpha),\qquad
j_\lambda \leftrightarrow \int^\oplus_{[0,\lambda]} J_{\relax{h}^{(\lambda)}_\alpha}\,d\nu_\lambda(\alpha).
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi

\underline{\textbf{Stage V:} Defining the transformation $\Phi$.}
\\
The joint index set and measure are
\[
\Lambda_+ := \bigsqcup_{\lambda>0} \{(\lambda,\alpha) : \alpha \in \sigma(a_\lambda)\},\qquad
d\mu_+(\lambda,\alpha) := d\nu_\lambda(\alpha)\,d\mu_1(\lambda).
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
The total direct integral space is
\[
\mathcal{K}_{\mathrm{tot}}:=
(\mathcal{K}_0\oplus\mathcal{K}_0)\oplus
\int^\oplus_{\Lambda_+} (\relax{h}_{(\lambda,\alpha)}\oplus\relax{h}_{(\lambda,\alpha)}) \, d\mu_+(\lambda,\alpha),
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
where $\mathcal{K}_0 := \relax{h}_*$ encodes the kernel.

For the kernel piece, choose $\relax{h}_* \subset \mathbf{H}_0$ maximal with $\relax{h}_* \perp J_0\relax{h}_*$ and define
\[
\Phi_0 := \gamma_*^{-1} : \mathbf{H}_0 \xrightarrow{\sim} \relax{h}_*\oplus\relax{h}_* = \mathcal{K}_0\oplus\mathcal{K}_0,
\qquad \gamma_*(u,v) := u + J_0 v.
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
Then $\Phi_0 J_0 \Phi_0^{-1} = J_{\relax{h}_*}$ and $A|_{\mathbf{H}_0}=B|_{\mathbf{H}_0}=0$.

For the range piece, define bounded operators:
\begin{align}
  \Phi_R^{(0)} &:= \Gamma_0 \circ \iota : \mathbf{H}_R \to \mathcal{K}, \\
  \Phi_R^{(1)} &:= \left(\int^\oplus_{(0,\infty)} V_\lambda \, d\mu_1(\lambda)\right) \circ \Phi_R^{(0)} : \mathbf{H}_R \to \mathcal{K}_{\mathrm{tot}}\oplus\mathcal{K}_{\mathrm{tot}}.
\end{align}
Then $\Phi_R^{(1)}$ satisfies
\[
(\Phi_R^{(1)})^t \Phi_R^{(1)} = C_R,\qquad
\Phi_R^{(1)} J_R (\Phi_R^{(1)})^t = M_C \mathbf{J}_{\mathcal{K}_{\mathrm{tot}}},
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
\[
(\Phi_R^{(1)})^t M_A M_C^{-1} \Phi_R^{(1)} = A_R,\qquad
(\Phi_R^{(1)})^t M_B M_C^{-1} \Phi_R^{(1)} = B_R,
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
where $M_A=\int_{\Lambda_+}^{\oplus}\alpha\,d\mu_+$, $M_B=\int_{\Lambda_+}^{\oplus}(\lambda-\alpha)\,d\mu_+$, $M_C=M_A+M_B$, and $\mathbf{J}_{\mathcal{K}_{\mathrm{tot}}}=\int_{\Lambda_+}^{\oplus}\mathbf{J}_{\relax{h}_{\alpha}^{(\lambda)}}\,d\mu_+$. To verify the third identity, take $U = \left(\int^{\oplus}_{(0,\infty)}V_\lambda\,d\mu_1(\lambda)\right)\circ\Gamma_0$. Then $U\widehat{T}_A U^{-1}=\mathbf{J}_{\mathcal{K}_{\mathrm{tot}}} M_A$ and $U\widehat{T}_C U^{-1}=\mathbf{J}_{\mathcal{K}_{\mathrm{tot}}} M_C$, so for $x,y\in\mathbf{H}_R$,
\begin{align*}
\langle (\Phi_R^{(1)})^t M_A M_C^{-1}\Phi_R^{(1)}x,y\rangle_{\mathbf{H}_R}
&=\langle \iota x,\widehat{T}_{A}\widehat{T}_{C}^{-1}\iota y\rangle_{C}\\
&=\langle x,A_R y\rangle_{\mathbf{H}_R}.
\end{align*}

Now define the (generally unbounded) transformation
\[
\Phi_R^\prime := M_C^{-1/2} \Phi_R^{(1)},\qquad
\operatorname{dom}(\Phi_R^\prime) := \{ x \in \mathbf{H}_R : \Phi_R^{(1)} x \in \operatorname{ran}(M_C^{1/2}) \}.
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
The domain is dense in $\mathbf{H}_R$ by an argument analogous to Step 6 in the proof of Theorem~{{\rm\expandafter\@setref\csname r@thm:di-wnf\endcsname\@firstoftwo{thm:di-wnf}}}. Finally, define
\[
\Phi := \Phi_0 \oplus \Phi_R^\prime,\qquad
\operatorname{dom}(\Phi) := \mathbf{H}_0 \oplus \operatorname{dom}(\Phi_R^\prime).
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi

For $A$, on $\mathbf{H}_0$, $\Phi_0^t \cdot 0 \cdot \Phi_0 = 0 = A|_{\mathbf{H}_0}$; and on $\operatorname{dom}(\Phi_R^\prime)$:
\[
(\Phi_R^\prime)^t M_A \Phi_R^\prime = (\Phi_R^{(1)})^t M_A M_C^{-1} \Phi_R^{(1)} = A_R.
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
Similarly, $(\Phi_R^\prime)^t M_B \Phi_R^\prime = B_R$.

For $J$, on $\mathbf{H}_0$, $\Phi_0 J_0 \Phi_0^t = \mathbf{J}_{\mathcal{K}_0}$; and on $\operatorname{ran}(\Phi_R^\prime)$:
\[
\Phi_R^\prime J_R (\Phi_R^\prime)^t = M_C^{-1/2} (M_C \mathbf{J}_{\mathcal{K}_{\mathrm{tot}}}) M_C^{-1/2} = \mathbf{J}_{\mathcal{K}_{\mathrm{tot}}}.
\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
Since $\operatorname{ran}(\Phi_R^\prime)$ is dense, the identity extends to all of $\mathcal{K}_{\mathrm{tot}}$.

The uniqueness of the joint measure class $[\mu_{\mathrm{tot}}]$ and the multiplicity function follows from standard spectral theory for commuting self-adjoint operators \@nx\protect\@empty\@nx\csname cite \endcsname[Section~VIII.2]{RS81}. This completes the proof.
