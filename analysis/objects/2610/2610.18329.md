<!-- definition-1: body_md -->
\label{def:ordinary-dich}
	               
    We say that system~\eqref{eq:Lin} (or the sequence $\{A_n\}_{n\in \mathbb{N}}$) admits a nonuniform  $\mu$-ordinary dichotomy with exponent $c\ge 0$ and scale $M>0$ 
                                                                                               
 if there are a sequence of projections $\{P_k\}_{k \in \mathbb{N}}$ on $X$ and $K \geq 1$ such that:
	\begin{enumerate}[(od1)]
		\item\label{def:od1} $A_k P_k=P_{k+1} A_k$, for all $k \in \mathbb{N}$;
		                                                                 
		\item\label{def:od3} for every $k \in \mathbb{N}, {A_k}_{\mid}: \operatorname{Ker} P_k \rightarrow \operatorname{Ker} P_{k+1}$ is an isomorphism, where ${A_k}_{\mid}$ denotes the restriction of $A_k$ to $\mathrm{Ker} P_k$;
		\item\label{def:od4} $\left\|\Phi_A(m, k) P_k \right\| \leq K\mu_k^c $, for all $m \ge k$ with $\mu_m/\mu_k \le M$;
		\item\label{def:od5} $\left\|\Phi_A(m, k) (\mathrm{Id}-P_k)\right\| \leq K\mu_k^c$, for all $m\le k$ with $\mu_k/\mu_m\le M$.
	\end{enumerate}

<!-- definition-2: body_md -->
\label{def:mu-dich-norms}
Let $\mu=(\mu_n)_{n\in \mathbb{N}_0}$ be a discrete growth rate.
             
We say that system~\eqref{eq:Lin} (or the sequence $\{A_n\}_{n\in \mathbb{N}}$) admits a nonuniform $\mu$-dichotomy
                                                                                            
with exponents $(\nu,\varepsilon)$ where $\nu>0$ and $\varepsilon \ge 0$ if there is a constant $N \geq 1$ and a sequence of projections $\{P_k\}_{k \in \mathbb{N}}$ on $X$ such that (od\ref{def:od1})--(od\ref{def:od3}) in Definition~\ref{def:ordinary-dich} are satisfied and additionally:
\begin{enumerate}[($\mu$1)]
	\item\label{def:mu1} 
	$\left\|\Phi_A(m, k) P_k \right\| \leq N \left(\frac{\mu_m}{\mu_k}\right)^{-\nu} \mu_k^\varepsilon $, for every  $m \ge k$;
	\item\label{def:mu2} $\left\|\Phi_A(m, k) (\mathrm{Id}-P_k)\right\|\leq N \left(\frac{\mu_k}{\mu_m}\right)^{-\nu} \mu_k^\varepsilon$, for every  $m \le k$.
\end{enumerate}

<!-- remark-1: body_md -->
When, in Definition~\ref{def:mu-dich-norms}, we set $\mu_n=e^n$, for $n \in \mathbb{N}_0$, we obtain the notion of nonuniform exponential dichotomy.
                                                                                      
Similarly, when we let $\mu_n=n+1$ for $n \in \mathbb{N}_0$, we have the notion of nonuniform polynomial dichotomy. 
                                                                                      
When $\varepsilon=0$ we recover the notion of (uniform) $\mu$-dichotomy.

\label{lem:scale}
	Let $\theta \ge 1$ and $\mu,\eta$ be growth rates satisfying 
	\begin{equation}\label{eq:bound-rates-1-0}
		\frac{\mu_{n+1}}{\mu_n}\le \theta \quad \text{ and } \quad \frac{\eta_{n+1}}{\eta_n}\le \theta, \quad \text{for all} \quad n \in \mathbb{N}_0.
	\end{equation} 
	Then, 
	\begin{enumerate}[(i)]
		\item\label{lem:scale1} for every $r\in\mathbb N_0$, $\eta_r\le \mu_{\lfloor\widetilde\mu^{-1}(\eta_r)\rfloor+1}\le \theta \eta_r$.
		\item\label{lem:scale2} for every $r\in\mathbb N_0$, $\dfrac{\mu_{\lfloor\widetilde\mu^{-1}(\eta_{r+1})\rfloor+1}}{\mu_{\lfloor\widetilde\mu^{-1}(\eta_r)\rfloor+1}} \le \theta^2$.
		\item\label{lem:scale3} if $\lfloor\widetilde\mu^{-1}(\eta_{n+1})\rfloor+1\ge m\ge \lfloor\widetilde\mu^{-1}(\eta_n)\rfloor+1$, then
		$$\dfrac{\mu_m}{\mu_{\lfloor\widetilde\mu^{-1}(\eta_n)\rfloor+1}} \le \theta^2 \quad \text{and} \quad \dfrac{\mu_{\lfloor\widetilde\mu^{-1}(\eta_{n+1})\rfloor+1}}{\mu_m} \le \theta^2$$
	\end{enumerate}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Let $r\in\mathbb N_0$, $t=\widetilde{\mu}^{-1}(\eta_r)$. By the definition of $\widetilde{\mu}$ and since it is increasing, we have
	\[
	\eta_r=\tilde\mu(\tilde\mu^{-1}(\eta_r))\le \tilde\mu(\lfloor\tilde\mu^{-1}(\eta_r)\rfloor+1)=\mu_{\lfloor\tilde\mu^{-1}(\eta_r)\rfloor+1}
	\]
	and also, by~\eqref{eq:bound-rates-1-0},
	\[
	\mu_{\lfloor\tilde\mu^{-1}(\eta_r)\rfloor+1}= \frac{\mu_{\lfloor\tilde\mu^{-1}(\eta_r)\rfloor+1}}{\mu_{\lfloor\tilde\mu^{-1}(\eta_r)\rfloor}}\mu_{\lfloor\tilde\mu^{-1}(\eta_r)\rfloor}\le \theta \mu_{\lfloor\tilde\mu^{-1}(\eta_r)\rfloor}=\theta \widetilde{\mu}(\lfloor\widetilde{\mu}^{-1}(\eta_r)\rfloor)\le\theta \widetilde{\mu}(\widetilde{\mu}^{-1}(\eta_r))=\theta\eta_r.
	\]
	This proves \eqref{lem:scale1}. 
	
	Since, by~\eqref{lem:scale1}, we have $\mu_{\lfloor\widetilde\mu^{-1}(\eta_r)\rfloor +1}\ge \eta_r$ and $\mu_{\lfloor\widetilde\mu^{-1}(\eta_{r+1})\rfloor +1}\le \theta \eta_{r+1}$, using~\eqref{eq:bound-rates-1-0} we have
	\[
	\frac{\mu_{\lfloor\widetilde\mu^{-1}(\eta_{r+1})\rfloor+1}}{\mu_{\lfloor\widetilde\mu^{-1}(\eta_r)\rfloor+1}}
	\le \frac{\theta \eta_{r+1}}{\eta_r}
	\le \theta^2. 
	\]
	Thus, we obtain \eqref{lem:scale2}. 
	
	If $\lfloor\widetilde\mu^{-1}(\eta_{n+1})\rfloor+1\ge m\ge \lfloor\widetilde\mu^{-1}(\eta_n)\rfloor+1$, then $\mu_{\lfloor\widetilde\mu^{-1}(\eta_{n+1})\rfloor+1}\ge \mu_m\ge \mu_{\lfloor\widetilde\mu^{-1}(\eta_n)\rfloor+1}$ and thus, from~\eqref{lem:scale2},
	\[
	\frac{\mu_m}{\mu_{\lfloor\widetilde\mu^{-1}(\eta_n)\rfloor+1}}
	\le \frac{\mu_{\lfloor\widetilde\mu^{-1}(\eta_{n+1})\rfloor+1}}{\mu_{\lfloor\widetilde\mu^{-1}(\eta_n)\rfloor+1}}
	\le\theta^2
	\quad \text{ and } \quad
	\frac{\mu_{\lfloor\widetilde\mu^{-1}(\eta_{n+1})\rfloor+1}}{\mu_m}
	\le \frac{\mu_{\lfloor\widetilde\mu^{-1}(\eta_{n+1})\rfloor+1}}{\mu_{\lfloor\widetilde\mu^{-1}(\eta_n)\rfloor+1}}
	\le\theta^2.
	\]
	Hence, \eqref{lem:scale3} holds.

\label{teo:main1}

	\begin{equation}\label{eq:bound-rates-1-a}

\label{teo:main}
                                                                                
Let $\theta \ge 1$ and $\mu, \eta$ be growth rates satisfying
\begin{equation}\label{eq:bound-rates-1-b}
	\frac{\mu_{n+1}}{\mu_n}\le \theta \quad \text{and} \quad \frac{\eta_{n+1}}{\eta_n} \le \theta, \quad \text{for all} \quad n \in \mathbb{N}_0.
\end{equation} 
If system~\eqref{eq:Lin} admits a nonuniform $\mu$-ordinary dichotomy                                                                                        
relative to a sequence of projections $\{P_k\}_{k \in \mathbb{N}}$, with exponent $c$ and scale $\theta^2$, and system~\eqref{eq:Lin-Q} admits a nonuniform $\eta$-dichotomy 
                                                                                              
relative to the sequence of projections $\left\{P^\eta_k\right\}_{k \in \mathbb{N}}$, with exponents $(\nu,\varepsilon)$, then, if $\nu>c$, system~\eqref{eq:Lin} admits a nonuniform $\mu$-dichotomy 
                                                                                      
relative to a sequence of projections $\{P_k\}_{k \in \mathbb{N}}$, with exponents $(\nu-c,\varepsilon+2c)$.

Assume that~\eqref{eq:Lin} admits a nonuniform $\mu$-ordinary dichotomy 
                                                                                       
relative to a sequence of projections $\{P_k\}_{k \in \mathbb{N}}$ with exponent $c$ at scale $M:=\theta^2$: for some $K \geq 1$  we have, for all  $m \ge k$ with $\mu_m/\mu_k \le M$,
\begin{equation}\label{eq:ordinary-dich1-iii->i}
\left\|\Phi_A(m, k) P_k \right\|\leq K\mu_k^c
\end{equation}
and, for all $m \le k$ with $\mu_k/\mu_m\le M$,
\begin{equation}\label{eq:ordinary-dich2-iii->i}
\left\|\Phi_A(m, k) (\mathrm{Id}-P_k)\right\| \leq K\mu_k^c. 
\end{equation}

Assume also that system~\eqref{eq:Lin-Q} admits a nonuniform $\eta$-dichotomy 
                                                                                            
relative to the projections $\left\{P^\eta_k\right\}_{k \in \mathbb{N}}$: there are $N \geq 1, \nu>0$ and  $\varepsilon \ge 0$ such that, for all $m \ge k$, we have
\begin{equation}\label{eq:eta-dich1-iii->i}
\left\|\Phi_{Q^{\mu,\eta}}(m,k) P_k^\eta \right\| \leq N \left(\frac{\eta_m}{\eta_k}\right)^{-\nu}\eta_k^\varepsilon
\end{equation}
and for $m\le k$,
\begin{equation}\label{eq:eta-dich2-iii->i}
\left\|\Phi_{Q^{\mu,\eta}}(m,k) (\mathrm{Id}-P_k^\eta)\right\| \leq N \left(\frac{\eta_k}{\eta_m}\right)^{-\nu}\eta_k^\varepsilon.
\end{equation}
We claim that there exists $N_1\ge 1$ such that  for $m \ge k \ge 1$, we have
\begin{equation}\label{eq:claim1}
	\left\|\Phi_A(m, k) P_k \right\|
	\le N_1 \left(\frac{\mu_{m}}{\mu_{k}}\right)^{-\nu+c} \mu_k^{\varepsilon+2c}.
\end{equation}
We split the proof of~\eqref{eq:claim1} into two cases:\\
\underline{Case I} - There are $n,j \in \mathbb{N}_0$ with $n>j$ such that 
$$\lfloor\widetilde\mu^{-1}(\eta_{n+1})\rfloor +1> m \ge \lfloor\widetilde\mu^{-1}(\eta_n)\rfloor +1 \ge \lfloor\widetilde\mu^{-1}(\eta_{j+1})\rfloor +1 >  k
\ge \lfloor\widetilde\mu^{-1}(\eta_j)\rfloor+1.$$
 Recalling that $P_{\lfloor {\widetilde \mu}^{-1} (\eta_{j+1}) \rfloor +1}=P_{j+2}^\eta$ 
                                                                                             
, by~\eqref{eq:bound-rates-1-b},~\eqref{eq:ordinary-dich1-iii->i},~\eqref{eq:eta-dich1-iii->i} and~\eqref{lem:scale3} in Lemma~\ref{lem:scale}, we get 
\[
\begin{split}
& \left\|\Phi_A(m, k) P_k \right\| \\
	& =\left\|
	\Phi_A\left(m, \lfloor\widetilde\mu^{-1}(\eta_n)\rfloor +1\right) P_{\lfloor \widetilde{\mu}^{-1}(\eta_n)\rfloor+1} \Phi_A\left(\lfloor\widetilde\mu^{-1}(\eta_n)\rfloor +1, k\right) P_k \right\| \\
	& \le K \mu_{\lfloor\widetilde\mu^{-1}(\eta_n)\rfloor +1}^c \left\| \Phi_A\left(\lfloor\widetilde\mu^{-1}(\eta_n)\rfloor +1, \lfloor\widetilde\mu^{-1}(\eta_{j+1})\rfloor +1\right)
	\Phi_A\left(\lfloor\widetilde\mu^{-1}(\eta_{j+1})\rfloor +1,k\right) P_k \right\|\\
&\le K\mu_m^c\left\|\Phi_{Q^{\mu,\eta}}\left(n+1,j+2\right)P_{j+2}^\eta \Phi_A\left(\lfloor\widetilde\mu^{-1}(\eta_{j+1})\rfloor+1,k\right) P_k \right\| \\
	& \leq NK  \mu_m^c\left(\frac{\eta_{n+1}}{\eta_{j+2}}\right)^{-\nu}
	\eta_{j+2}^\varepsilon \, \left\|\Phi_A\left(\lfloor\widetilde\mu^{-1}(\eta_{j+1})\rfloor+1,k\right) P_k\right\| \\
	& \leq NK^2   \left(\frac{\mu_m}{\mu_k}\right)^c\left(\frac{\eta_{n+1}}{\eta_{j+2}}\right)^{-\nu}
	\eta_{j+2}^\varepsilon\mu_k^{2c}\\
	& \leq NK^2 \theta^{3\nu+2\varepsilon}  \left(\frac{\mu_m}{\mu_k}\right)^c\left(\frac{\mu_m}{\mu_k}\right)^{-\nu}
	\mu_k^\varepsilon\mu_k^{2c}\\
	& \leq NK^2 \theta^{3\nu+2\varepsilon} \left(\frac{\mu_m}{\mu_k}\right)^{-\nu+c} \mu_k^{\varepsilon+2c},
\end{split}
\]
as 
\[
\begin{split}
\frac{\eta_{n+1}}{\eta_{j+2}}=\frac{\eta_{n+1}}{\mu_m}\cdot \frac{\mu_m}{\mu_k}\cdot \frac{\mu_k}{\eta_{j+2}} &\ge \frac{\eta_{n+1}}{\mu_{\lfloor \widetilde{\mu}^{-1}(\eta_{n+1})\rfloor+1}}\cdot \frac{\mu_m}{\mu_k}\cdot \frac{\mu_{\lfloor \widetilde{\mu}^{-1}(\eta_j)\rfloor+1}}{\eta_{j+2}}\\
&\ge \frac{\eta_{n+1}}{\mu_{\lfloor \widetilde{\mu}^{-1}(\eta_{n+1})\rfloor+1}}\cdot \frac{\mu_m}{\mu_k}\cdot \frac{\eta_j}{\eta_{j+2}}\\
&\ge \theta^{-3}\frac{\mu_m}{\mu_k}.
\end{split}
\]

\underline{Case II} - There is $n \in \mathbb{N}_0$ such that $$\lfloor\widetilde\mu^{-1}(\eta_{n+1})\rfloor +1> m \ge k
\ge \lfloor\widetilde\mu^{-1}(\eta_n)\rfloor+1.$$
By~\eqref{eq:ordinary-dich1-iii->i} and~\eqref{lem:scale2} in Lemma~\ref{lem:scale}, we get 
\[
\begin{split}
	\left\|\Phi_A(m, k) P_k \right\|	\le K\mu_k^c 
	&=K\mu_k^c\left (\frac{\mu_m}{\mu_k}\right )^{-\nu} \left (\frac{\mu_m}{\mu_k}\right )^{\nu}  \\
&\le K\mu_k^c\left (\frac{\mu_m}{\mu_k}\right )^{-\nu} \left (\frac{\mu_{\lfloor\tilde\mu^{-1}(\eta_{n+1})\rfloor+1}}{\mu_{\lfloor\tilde\mu^{-1}(\eta_n)\rfloor+1}}\right )^{\nu}\\
& \le K \theta^{2\nu}
                   
\left(\frac{\mu_{m}}{\mu_{k}}\right)^{-\nu}\mu_k^c.
\end{split}
\]
We conclude that~\eqref{eq:claim1} holds and the claim follows. 

Our next claim is that there exists $N_2\ge 1$ such that for $1\le m\le k$, we have
                                                                                                                                                                                       
\begin{equation}\label{eq:claim2}
	\left\|\Phi_A(m, k) (\mathrm{Id}-P_k) \right\|
	\leq N_2 \left(\frac{\mu_{k}}{\mu_{m}}\right)^{-\nu}   \mu_k^{\varepsilon+2c}.
\end{equation}
We again split the proof of~\eqref{eq:claim2} into two cases.

\underline{Case I} - There are $n, j\in \mathbb{N}_0$ with $n<j$ such that 
\[
\lfloor \tilde \mu^{-1}(\eta_n)\rfloor +1\le m < \lfloor \tilde \mu^{-1}(\eta_{n+1})\rfloor +1 \le  \lfloor \tilde \mu^{-1}(\eta_{j})\rfloor +1\le k< \lfloor \tilde{\mu}^{-1}(\eta_{j+1})\rfloor +1.

By~\eqref{eq:bound-rates-1-b},~\eqref{eq:ordinary-dich2-iii->i},~\eqref{eq:eta-dich2-iii->i} and~\eqref{lem:scale3} in Lemma~\ref{lem:scale} we have that 
\[

	& =\left\|\Phi_A\left(m, \lfloor\widetilde\mu^{-1}(\eta_{n+1})\rfloor+1\right) (\mathrm{Id}- P_{\lfloor\widetilde\mu^{-1}(\eta_{n+1})\rfloor+1})\Phi_A\left(\lfloor\widetilde\mu^{-1}(\eta_{n+1})\rfloor +1, k\right) (\mathrm{Id}-P_k)\right\|\\

&\le K^2\theta^{2c}\mu_k^c\mu_m^c \|\Phi_{Q^{\mu,\eta}}\left(n+2,j+1\right)(\mathrm{Id}-P_{j+1}^\eta)\|\\
& \le NK^2\theta^{2c}\mu_k^{2c}\left(\frac{\mu_m}{\mu_k}\right)^c \left(\frac{\eta_{j+1}}{\eta_{n+2}}\right)^{-\nu}\theta^\varepsilon\mu_k^\varepsilon\\
&\le NK^2 \theta^{2c+3\nu+\varepsilon}\left (\frac{\mu_k}{\mu_m}\right )^{-(\nu+c)}\mu_k^{\varepsilon+2c},
                                                                                                               
\end{split} 
\]
as \[\frac{\eta_{j+1}}{\mu_k}\le \frac{\eta_{j+1}}{\mu_{\lfloor \widetilde{\mu}^{-1}(\eta_j)\rfloor+1}}\le \frac{\eta_{j+1}}{\eta_j}\le \theta, \]
and
\[
\frac{\eta_{j+1}}{\eta_{n+2}}=\frac{\eta_{j+1}}{\mu_k}\cdot  \frac{\mu_m}{\eta_{n+2}}\cdot \frac{\mu_k}{\mu_m}\ge \frac{\eta_{j+1}}{\mu_{\lfloor \widetilde{\mu}^{-1}(\eta_{j+1})\rfloor+1}}\cdot \frac{\mu_{\lfloor \widetilde{\mu}^{-1}(\eta_n)\rfloor+1}}{\eta_{n+2}}\cdot \frac{\mu_k}{\mu_m}\ge \theta^{-1}\frac{\eta_n}{\eta_{n+2}}\cdot \frac{\mu_k}{\mu_m}\ge \theta^{-3}\frac{\mu_k}{\mu_m}.
\]

\underline{Case II} - There is $n\in \mathbb{N}_0$ such that
\[
\lfloor \tilde{\mu}^{-1}(\eta_n)\rfloor +1 \le m\le k<\lfloor \tilde{\mu}^{-1}(\eta_{n+1})\rfloor+1.
\]
By~~\eqref{lem:scale2} in Lemma~\ref{lem:scale}
and~\eqref{eq:ordinary-dich2-iii->i} we have that 
\[
\begin{split}
\left\|\Phi_A(m, k) (\mathrm{Id}-P_k)\right\| \le K\mu_k^c &=K\left (\frac{\mu_k}{\mu_m}\right )^{-\nu} \left (\frac{\mu_k}{\mu_m}\right )^{\nu} \mu_k^c \\
&\le K\left (\frac{\mu_k}{\mu_m}\right )^{-\nu} \left (\frac{\mu_{\lfloor\tilde\mu^{-1}(\eta_{n+1})\rfloor+1}}{\mu_{\lfloor\tilde\mu^{-1}(\eta_n)\rfloor+1}}\right )^{\nu} \mu_k^c \\
&\le K\theta^{2\nu} \left (\frac{\mu_k}{\mu_m}\right )^{-\nu}\mu_k^c.
\end{split}
\]
We conclude that~\eqref{eq:claim2} holds and the claim follows.

We conclude that system~\eqref{eq:Lin} admits a $\mu$-nonuniform dichotomy 
                                                                                       
relative to the projections $\{P_k\}_{k \in \mathbb{N}}$ with exponents $(\nu-c,\varepsilon+2c)$. The proof of the theorem is completed.

\label{ex:discrete}
We equip $\mathbb{R}^2$ with the maximum norm. Define a sequence $(b_n)_{n\in\mathbb N_0}$ by
\begin{equation}\label{eq:discrete-b}
 b_n=
 \begin{cases}
  n-3\cdot 2^{j-1}, & 2^j\leq n\leq 3\cdot 2^{j-1}\text{ for some }j\in\mathbb{N},\\[1mm]
  0, & \text{otherwise}.
 \end{cases}
\end{equation}
Note that the intervals in~\eqref{eq:discrete-b} are pairwise disjoint. Consider the system
\begin{equation}\label{eq:discrete-system}
 x_{n+1}=A_nx_n,
 \qquad
 A_n=
 \begin{pmatrix}
  \operatorname{e}^{-1+b_{n+1}-b_n} & 0\\
  0 & \operatorname{e}^{1-b_{n+1}+b_n}
 \end{pmatrix}.
\end{equation}

The evolution operators associated with~\eqref{eq:discrete-system} are
\begin{equation}\label{eq:discrete-evolution}
 \Phi_A(m,k)=
 \begin{pmatrix}
  \operatorname{e}^{-(m-k)+b_m-b_k} & 0\\
  0 & \operatorname{e}^{m-k-b_m+b_k}
 \end{pmatrix},
 \qquad m,k\in\mathbb N_0.
\end{equation}
For $m<k$, the expression in~\eqref{eq:discrete-evolution} denotes the inverse of $\Phi_A(k,m)$.  Let
\[
 P=
 \begin{pmatrix}
  1&0\\
  0&0
 \end{pmatrix}.
\]
Clearly, $A_nP=PA_n$ for every $n$, and the restriction of $A_n$ to $\ker P$ is an isomorphism.

We highlight two elementary properties of $(b_n)$.  Namely,
\begin{equation}\label{eq:discrete-bounds}
 -\frac n2\leq b_n\leq0 \qquad\text{and}\qquad b_{n+1}-b_n\leq1 \quad\text{for every } n\in\mathbb N_0.
\end{equation}
Indeed, if $n=2^j+r$ with $0\leq r\leq 2^{j-1}$, then
$$b_n=-\frac{2^j}{2}+r\geq-\frac{2^j}{2}\ge -\frac n2.$$
When $b_n=0$, we immediately have $b_n\ge -n/2$.  To obtain the second inequality in~\eqref{eq:discrete-bounds}, we note the following: when $m$ and $m+1$ belong to the same interval, $[2\cdot 2^{j-1},3\cdot 2^{j-1}]$, we have $b_{m+1}-b_m=1$; in all the remaining cases $b_{m+1}-b_m\le 0$.

It follows from~\eqref{eq:discrete-bounds} that
\begin{equation}\label{eq:discrete-one-sided-lipschitz}
 b_m-b_k\leq m-k \qquad\text{whenever }m\geq k.
\end{equation}
Consequently, for $m\geq k$,
$$\|\Phi_A(m,k)P\|=\operatorname{e}^{-(m-k)+b_m-b_k}\leq1.$$
For $m\leq k$, applying~\eqref{eq:discrete-one-sided-lipschitz} to the pair $(k,m)$ gives
$$\|\Phi_A(m,k)(\mathrm{Id}-P)\| =\operatorname{e}^{-(k-m)+b_k-b_m}\leq1.$$
Thus system~\eqref{eq:discrete-system} admits a nonuniform exponential ordinary dichotomy with $c=0$ and $K=1$. In fact, these estimates hold without any restriction on
$|m-k|$, so the scale $M>1$ can be chosen arbitrarily.

If $m\geq k$, we have $b_m\leq0$ and $b_k\geq-k/2$, and hence 
\begin{equation}\label{eq:discrete-stable}
 \|\Phi_A(m,k)P\|
 \leq \operatorname{e}^{-(m-k)}\operatorname{e}^{k/2}.
\end{equation}
If $m\leq k$, then $b_k\leq0$ and $b_m\geq-m/2$, and therefore
\begin{equation}\label{eq:discrete-unstable}
 \begin{split}
 \|\Phi_A(m,k)(\mathrm{Id}-P)\|
 &=\operatorname{e}^{-(k-m)-b_m+b_k}\\
 &\leq \operatorname{e}^{-(k-m)}\operatorname{e}^{m/2}
 \leq \operatorname{e}^{-(k-m)}\operatorname{e}^{k/2}.
 \end{split}
\end{equation}
Equations~\eqref{eq:discrete-stable} and~\eqref{eq:discrete-unstable} prove that system~\eqref{eq:discrete-system} admits a nonuniform exponential dichotomy with
$N=1$, $\nu=1$, and $\varepsilon=1/2$.  Notice in particular that $\varepsilon<\nu$.

We claim that system~\eqref{eq:discrete-system} does not admit a uniform exponential dichotomy, even if a different sequence of projections is allowed. In fact, for
$$2^j\leq n<3\cdot 2^{j-1}$$
we have $b_{n+1}-b_n=1$, and thus $A_n=\mathrm{Id}$.  Hence
\begin{equation}\label{eq:discrete-identity-block}
 \Phi_A(3\cdot 2^{j-1},2^j)=\mathrm{Id}, \quad \text{with} \quad
 3\cdot2^{j-1}-2^j=2^{j-1}\longrightarrow+\infty.
\end{equation}

Suppose, to the contrary, that the system admitted a uniform exponential dichotomy with invariant projections $(R_n)$ and constants $D\geq1$ and $\alpha>0$.  Invariance on the identity block gives $R_{3\cdot 2^{j-1}}=R_{2^j}$. The stable and unstable estimates, applied to the two endpoints of this
block, would imply
$$\|R_{2^j}\|\leq D\operatorname{e}^{-\alpha 2^{j-1}} \qquad\text{and}\qquad \|\mathrm{Id}-R_{2^j}\|=\|\mathrm{Id}-R_{3\cdot 2^{j-1}}\|\leq D\operatorname{e}^{-\alpha 2^{j-1}}.$$
Therefore,
$$1=\|\mathrm{Id}\| \leq \|R_{2^j}\|+\|\mathrm{Id}-R_{2^j}\| \leq2D\operatorname{e}^{-\alpha 2^{j-1}},$$
which is impossible as $j\to\infty$.  This proves the claim.

Finally, for $\mu_n=\eta_n=e^n$, one has $\theta=e$ and $Q_n^{\mu,\eta}=A_n$. The ordinary dichotomy obtained above holds at every scale and hence, in particular, at the scale $\theta^2=e^2$. Moreover, $\nu=1>0=c$. Thus, all the hypotheses of Theorem~\ref{teo:main} are satisfied.

\label{thmc}
Let $\{T(t, s)\}_{t\ge s\ge 0}$ be an evolution family and $\mu, \eta \colon [0, \infty)\to [1, \infty)$ growth rates. The following statements are equivalent:
\begin{enumerate}
\item $\{T(t, s)\}_{t\ge s\ge 0}$ admits a nonuniform $\mu$-dichotomy with exponents $(\nu, \varepsilon)$;
\item $\{T^{\mu, \eta}(t, s)\}_{t\ge s \ge 0}$ admits a nonuniform $\eta$-dichotomy with exponents $(\nu, \varepsilon)$.
\end{enumerate}

<!-- proof-5: body_md -->
$i.\implies ii.$ Suppose that  $\{T(t, s)\}_{t\ge s\ge 0}$ admits a nonuniform $\mu$-dichotomy with exponents $(\nu, \varepsilon)$ relative to a family of projections $\{P(t)\}_{t\ge 0}$. Set
\[
\tilde P(t):=P(\mu^{-1}(\eta(t))), \quad t\ge 0.
\]
Then,
\[
\begin{split}
\tilde P(t)T^{\mu, \eta}(t, s) &=P(\mu^{-1}(\eta(t)))T(\mu^{-1}(\eta(t)), \mu^{-1}(\eta(s))) \\
&=T(\mu^{-1}(\eta(t)), \mu^{-1}(\eta(s)))P(\mu^{-1}(\eta(s)))\\
&=T^{\mu, \eta}(t, s)\tilde P(s),
\end{split}
\]
for $t\ge s\ge 0$. Moreover, for such $t$ and $s$, the restriction of $T^{\mu, \eta}(t, s)$ to $\mathrm{Ker} \tilde P(s)$ coincides with the restriction of $T(\mu^{-1}(\eta(t)), \mu^{-1}(\eta(s)))$ to $\mathrm{Ker} P(\mu^{-1}(\eta(s)))$ which is invertible.

Furthermore, for $t\ge s\ge 0$ we have 
\[
\begin{split}
\|T^{\mu, \eta}(t, s)\tilde P(s)\| &=\|T(\mu^{-1}(\eta(t)), \mu^{-1}(\eta(s)))P(\mu^{-1}(\eta(s)))\| \\
&\le N \left (\frac{\mu(\mu^{-1}(\eta(t)))}{\mu(\mu^{-1}(\eta(s)))}\right )^{-\nu}\mu(\mu^{-1}(\eta(s)))^\varepsilon \\
&=N\left (\frac{\eta(t)}{\eta(s)}\right )^{-\nu}\eta(s)^\varepsilon.
\end{split}
\]
Similarly, for $0\le t\le s$ we have 
\[
\begin{split}
\|T^{\mu, \eta}(t, s)(\mathrm{Id}-\tilde P(s))\| &=\|T(\mu^{-1}(\eta(t)), \mu^{-1}(\eta(s)))(\mathrm{Id}-P(\mu^{-1}(\eta(s))))\| \\
&\le N \left (\frac{\mu(\mu^{-1}(\eta(s)))}{\mu(\mu^{-1}(\eta(t)))}\right )^{-\nu}\mu(\mu^{-1}(\eta(s)))^\varepsilon \\
&=N\left (\frac{\eta(s)}{\eta(t)}\right )^{-\nu}\eta(s)^\varepsilon.
\end{split}
\]
We conclude that $ii.$ holds.
 
$ii. \implies i.$ Suppose that  $\{T^{\mu, \eta}(t, s)\}_{t\ge s\ge 0}$ admits a nonuniform $\eta$-dichotomy with exponents $(\nu, \varepsilon)$ relative to a family of projections $\{\tilde P(t)\}_{t\ge 0}$. Set
\[
P(t):=\tilde P(\eta^{-1}(\mu(t))), \quad t\ge 0.
\]
Then (see~\eqref{inverse}) we have that for $t\ge s\ge 0$,
\[
\begin{split}
P(t)T(t, s) &=\tilde P(\eta^{-1}(\mu(t)))T^{\mu, \eta}(\eta^{-1}(\mu(t)), \eta^{-1}(\mu(s))) \\
&=T^{\mu, \eta}(\eta^{-1}(\mu(t)), \eta^{-1}(\mu(s)))\tilde P(\eta^{-1}(\mu(s))) \\
&=T(t, s)P(s).
\end{split}
\]
Moreover, the restriction of $T(t, s)$ to $\mathrm{Ker} P(s)$ which coincides with the restriction of $T^ {\mu, \eta}(\eta^{-1}(\mu(t)), \eta^{-1}(\mu(s)))$ to $\mathrm{Ker} \tilde P(\eta^{-1}(\mu(s)))$ 
 is invertible. In addition, for $t\ge s\ge 0$,
 \[
 \begin{split}
\|T(t, s)P(s)\| &=\|T^{\mu, \eta}(\eta^{-1}(\mu(t)), \eta^{-1}(\mu(s)))\tilde P(\eta^{-1}(\mu(s)))\| \\
&\le N\left (\frac{\eta (\eta^{-1}(\mu(t)))}{\eta (\eta^{-1}(\mu(s)))}\right )^{-\nu}\eta(\eta^{-1}(\mu(s)))^\varepsilon \\
&=N\left (\frac{\mu(t)}{\mu(s)}\right )^{-\nu}\mu(s)^\varepsilon.
 \end{split}
 \]
 Similarly, for $0\le t\le s$ we have 
 \[
 \|T(t, s)(\mathrm{Id}-P(s))\|\le N\left (\frac{\mu(s)}{\mu(t)}\right )^{-\nu}\mu(s)^\varepsilon.
 \]
 We conclude that $i.$ holds and the proof of the theorem is completed.

<!-- remark-3: body_md -->
\label{remc1}
So far, we have considered evolution families over $\mathbb R^+$. However, one can easily adapt the arguments to deal with evolution families  over $\mathbb R$. In this case, a (continuous-time) growth rate is an arbitrary continuous and strictly increasing map $\mu \colon \mathbb R \to (0,+\infty)$ such that $\mu(0)=1$, $\lim_{t\to \infty}\mu(t)=\infty$ and $\lim_{t\to -\infty}\mu(t)=0$. Given two such growth rates $\mu, \eta\colon \mathbb R\to (0,+\infty)$ and an evolution family $\{T(t, s)\}_{t\ge s }$, one can consider an evolution family $\{T^{\mu, \eta}(t, s)\}_{t\ge s}$ as in~\eqref{newef} (but for $t, s\in \mathbb R$, $t\ge s$).  One can now establish the version of Theorem~\ref{thmc} for evolution families on $\mathbb R$ whose proof would be identical to the proof of Theorem~\ref{thmc}.

<!-- example-2: body_md -->
\label{ex:nonuniform-polynomial-continuous}
Let $X=\mathbb R^2$, endowed with the maximum norm, and consider the growth rates
\[
\mu(t)=1+t
\qquad\text{and}\qquad
\eta(\tau)=e^\tau,
\]
and define $b\colon[0,+\infty)\to\mathbb R$ by
\begin{equation}\label{eq:continuous-polynomial-b}
b(\tau)=
\begin{cases}
-4^{j-1}\tau+4^{2j-1},	& 4^j\leq\tau\leq 4^j+1 \text{ for some }j\in\mathbb N\\[1mm]
-5\cdot 4^{j-1}+\tau-1, & 4^j+1\leq\tau\leq 5\cdot 4^{j-1}+1 \text{ for some }j\in\mathbb N\\[1mm]
0, &\text{otherwise}
\end{cases}.
\end{equation}
The blocks $[4^j,5\cdot4^{j-1}+1]$, $j\in\mathbb N$, are pairwise disjoint, and the expressions $-4^{j-1}\tau+4^{2j-1}$ and $-5\cdot 4^{j-1}+\tau-1$ in \eqref{eq:continuous-polynomial-b} agree at $\tau=4^j+1$. Consider the differential equation
\begin{equation}\label{eq:continuous-polynomial-system}
x'=A(t)x,
\qquad
A(t)=\frac{1}{1+t}
\begin{pmatrix}
-1+b'(\ln(1+t)) & 0\\
0 & 1-b'(\ln(1+t))
\end{pmatrix},
\end{equation}
for $t\geq0$, where the equality defining $A(t)$ is understood almost everywhere. The evolution family associated with equation~\eqref{eq:continuous-polynomial-system} is
\begin{equation}\label{eq:continuous-polynomial-evolution}
T(t,s)=
\begin{pmatrix}
\displaystyle
\frac{1+s}{1+t} \
e^{b(\ln(1+t))-b(\ln(1+s))}&0\\[4mm]
0&\displaystyle
\frac{1+t}{1+s} \
e^{-b(\ln(1+t))+b(\ln(1+s))}
\end{pmatrix},
 \ \ t\geq s\geq0.
\end{equation}
Let
\[
P=
\begin{pmatrix}
1&0\\
0&0
\end{pmatrix}.
\]
Clearly, $P$ is invariant and the evolution on $\ker P$ is invertible. Note that
\begin{equation}\label{eq:continuous-polynomial-bounds}
-\frac{\tau}{4}\leq b(\tau)\leq0
\quad\text{for every }\tau\geq0, \qquad b'(\tau)\leq1
\quad\text{for almost every }\tau\geq0.
\end{equation}
Indeed, on the interval $[4^j,4^j+1]$ the function decreases linearly from $0$ to $-4^{j-1}$, while on the interval $[4^j+1,5\cdot4^{j-1}+1]$ it increases linearly from $-4^{j-1}$ to $0$. The lower bound in \eqref{eq:continuous-polynomial-bounds} follows immediately on both subintervals.

If $t\geq s$, then $b(\ln(1+t))\leq0$ and
$b(\ln(s+1))\geq-\frac14\ln(s+1)$. Consequently,
$$\|T(t,s)P\| =\frac{1+s}{1+t}
e^{b(\ln(1+t))-b(\ln(s+1))} \leq
\left(\frac{1+t}{1+s}\right)^{-1}(1+s)^{1/4} =\left(\frac{\mu(t)}{\mu(s)}\right)^{-1}\mu(s)^{1/4}.
$$
For $0\leq t\leq s$, using the inverse of the evolution on $\ker P$, we similarly obtain
\[
\begin{split}
\|T(t,s)(\mathrm{Id}-P)\|
&= \frac{1+t}{1+s} \
e^{-b(\ln(1+t))+b(\ln(s+1))} \leq
\left(\frac{1+s}{1+t}\right)^{-1}(1+t)^{1/4}\\
&\leq \left(\frac{\mu(s)}{\mu(t)}\right)^{-1}\mu(s)^{1/4}.
\end{split}
\]
Hence, equation~\eqref{eq:continuous-polynomial-system} admits a nonuniform $\mu$-dichotomy with $N=1$, $\nu=1$, and $\varepsilon=1/4$.
	
We next describe the time rescaling explicitly. Since $\mu^{-1}(e^\tau)=e^\tau-1$, it follows from \eqref{eq:continuous-polynomial-evolution} that
\begin{equation}\label{eq:continuous-polynomial-rescaled}
\begin{split}
T^{\mu,\eta}(\tau,\sigma)
&=T(e^\tau-1,e^\sigma-1)\\
&=
\begin{pmatrix}
e^{-(\tau-\sigma)+b(\tau)-b(\sigma)} & 0\\
0 & e^{\tau-\sigma-b(\tau)+b(\sigma)}
\end{pmatrix},
\qquad \tau\geq\sigma\geq0.
\end{split}
\end{equation}
The estimates already established show that this family admits a nonuniform exponential dichotomy with exponents
$$\nu=1 \qquad\text{and}\qquad \varepsilon=\frac14.$$
	
We will now show that the nonuniformity cannot be removed. Set
$$ s_j=e^{4^j+1}-1 \qquad\text{and}\qquad t_j=e^{5\cdot 4^{j-1}+1}-1. $$
For $t\in[s_j,t_j]$, we have
\[
\ln(1+t)\in[4^j+1,5\cdot 4^{j-1}+1]
\]
and thus $b'(\ln(1+t))=1$ almost everywhere. Hence, $A(t)=0$ almost everywhere on $[s_j,t_j]$ and
\begin{equation}\label{eq:continuous-polynomial-identity}
T(t_j,s_j)=\mathrm{Id}.
\end{equation}
Moreover,
\begin{equation}\label{eq:continuous-polynomial-ratio}
\frac{\mu(t_j)}{\mu(s_j)}=\frac{1+t_j}{1+s_j}=e^{4^{j-1}}\longrightarrow+\infty.
\end{equation}
	
Assume, for contradiction, that a uniform dichotomy with respect to $\mu$ exists. Let $(R(t))_{t\geq0}$ be the associated invariant family of projections, and let $D\geq1$ and $\alpha>0$ be the dichotomy constants. Invariance and \eqref{eq:continuous-polynomial-identity} imply that $R(t_j)=R(s_j)$. Applying the stable and unstable estimates to the endpoints of $[s_j,t_j]$ and using \eqref{eq:continuous-polynomial-ratio}, we obtain
\[
\|R(s_j)\|\leq De^{-\alpha 4^{j-1}}
\qquad\text{and}\qquad
\|\mathrm{Id}-R(t_j)\|\leq De^{-\alpha 4^{j-1}}.
\]
Therefore,
\[
1=\|\mathrm{Id}\|
\leq\|R(s_j)\|+\|\mathrm{Id}-R(s_j)\|
\leq2De^{-\alpha 4^{j-1}},
\]
which is impossible as $j\to\infty$. Thus, no uniform $\mu$-dichotomy exists.
	
This example provides a concrete
illustration of Theorem~\ref{thmc}: the nonuniform polynomial dichotomy of
the original system becomes a nonuniform exponential dichotomy after the
time rescaling $\tau=\ln(1+t)$. The example also shows that the nonuniformity
cannot, in general, be removed.

\label{robustness}
Let $\mu \colon [0, \infty)\to [1, \infty)$ be a continuously differentiable growth rate such that $\mu'(t)>0$ for each $t\in [0, \infty)$.
Suppose the following:
\begin{enumerate}
\item $\{T(t, s)\}_{t\ge s\ge 0}$ admits a nonuniform $\mu$-dichotomy with exponents $(\nu, \varepsilon)$;
                                          
                            
                                                                                           
               
\item there is $c>0$ such that 
\begin{equation}\label{smallpert}
\|B(t)\| \le \frac{c\mu'(t)}{\mu(t)^{1+\varepsilon}}, \quad t\ge 0.
\end{equation}
\end{enumerate}
Then, provided that $c$ is sufficiently small, $\{T_B(t, s)\}_{t\ge s\ge 0}$ admits a nonuniform $\mu$-dichotomy.

Let $\eta(t)=e^t$, $t\ge 0$.

\begin{split}
&T_B^{\mu, \eta}(t, s)\\
&=T_B(\mu^{-1}(e^t), \mu^{-1}(e^s))\\
&=T(\mu^{-1}(e^t), \mu^{-1}(e^s))
+\int_{\mu^{-1}(e^s)}^{\mu^{-1}(e^t)}T(\mu^{-1}(e^t), r)B(r)T_B(r, \mu^{-1}(e^s))\, dr \\
&=T^{\mu, \eta}(t, s)+\int_s^t T(\mu^{-1}(e^t), \mu^{-1}(e^z))B(\mu^{-1}(e^z))T_B(\mu^{-1}(e^z), \mu^{-1}(e^s)) \frac{e^z}{\mu'(\mu^{-1}(e^z))}\, dz \\
&=T^{\mu, \eta}(t, s)+\int_s^t T^{\mu, \eta}(t, z) B(\mu^{-1}(e^z))T_B^{\mu, \eta}(z, s)\frac{e^z}{\mu'(\mu^{-1}(e^z))} \, dz \\
&=T^{\mu, \eta}(t, s)+\int_s^t T^{\mu, \eta}(t, z) \tilde B(z) T_B^{\mu, \eta}(z, s)\,  dz, \\
\end{split}
\end{equation}
for $t\ge s\ge 0$,
where

\]
Note that~\eqref{smallpert} gives that
\begin{equation}\label{smallB2}
\|\tilde B(z)\|  \le ce^{-\varepsilon z}, \quad z\ge 0.

Hence, it follows from Theorem~\ref{thmc} that $\{T_B(t, s)\}_{t\ge s\ge 0}$ admits  a nonuniform $\mu$-dichotomy, which completes the proof of the theorem.

<!-- remark-4: body_md -->
One can easily establish the version of Theorem~\ref{robustness} for evolution families over $\mathbb R$ (see the discussion in Remark~\ref{remc1}). A result of this type has been proved in~\cite[Theorem 1]{BV}. In order to compare the two results, we write $\mu$ in the form  $\mu(t)=e^{\rho(t)}$.  Now~\eqref{smallpert} reduces to 
\begin{equation}\label{824pert}
\|B(t)\| \le \frac{c\rho'(t)}{e^{\varepsilon \rho(t)}}, \quad t\ge 0.
\end{equation}
In comparison, \cite[Theorem 1]{BV} (adjusted to the case of evolution families on $\mathbb R^+$) requires that 
\begin{equation}\label{838pert}
\|B(t)\| \le \frac{c\rho'(t)}{e^{3\varepsilon \rho(t)}}, \quad t\ge 0,
\end{equation}
and in addition that $\{T(t, s)\}$ admits a nonuniform $\mu$-dichotomy with exponents $(\nu, \varepsilon)$ that satisfy $2\varepsilon<\nu$. 

Clearly, \eqref{824pert} is weaker than~\eqref{838pert}. In addition, in Theorem~\ref{robustness} we make no assumptions regarding the constants associated with the dichotomy of $\{T(t, s)\}$. Finally, our proof of Theorem~\ref{robustness} is completely different from  the proof of~\cite[Theorem 1]{BV}, which makes no use of the relationship between exponential dichotomies and dichotomies with growth rates.

for $t\ge s$. We have proved that for each $y\in Y_\infty$, there exists $x\in Y_\infty$ such that 
\begin{equation}\label{1020adm}
x(t)=T^{\mu, \exp}(t, s)x(s)+\int_s^t T^{\mu, \exp}(t, z)y(z)\, dz, \quad \text{for $t\ge s$.}
\end{equation}
Moreover, such $x$ is unique. Indeed, suppose that $x_i$, $i=1, 2$ are such that 
\[
x_i(t)=T^{\mu, \exp}(t, s)x_i(s)+\int_s^t T^{\mu, \exp}(t, z)y(z)\, dz \quad \text{for $t\ge s$ and $i=1, 2$.}
\]
Then, $\tilde x:=x_1-x_2\in Y_\infty$ and 
\[
\tilde x(t)=T^{\mu, exp}(t, s)\tilde x(s), \quad t\ge s.
\]
Let 
\[
\underline{x}(t):=\tilde x(\ln \mu(t)), \quad t\in \mathbb R.
\]
Then, 
\[
\underline{x}(t)=T(t, s)\underline{x}(s), \quad t\ge s, 
\]
which means that 
\[
\underline{x}(t)=T(t, s)\underline{x}(s)+\int_s^t \frac{\mu'(r)}{\mu(r)}T(t, r)w(r)\, dr \quad \text{for $t\ge s$,}
\]
where $w\in Y_\infty$ is given by $w(t)=0$ for $t\in \mathbb R$. Due to the uniqueness in $(ii)$, we conclude that $\underline{x}=0$, which means that $x_1=x_2$. 

It now follows from~\cite[Theorem 4.28]{CL} that $\{T^{\mu, exp}(t, s)\}_{t\ge s}$ admits a uniform exponential dichotomy, which implies $(i)$.

$(i)\implies (ii)$ It follows by an analogous reasoning. Since $\{T(t, s)\}_{t\ge s}$ admits a uniform $\mu$-dichotomy, $\{T^{\mu, exp}(t, s)\}_{t\ge s}$ admits a uniform exponential dichotomy. By~\cite[Theorem 4.28]{CL}, for each $y\in Y_\infty$, there exists a unique $x\in Y_\infty$ such that~\eqref{1020adm}. By reversing the arguments in the proof of the implication $(ii)\implies (i)$, one can show that $(ii)$ holds.

\label{Lyapunov}
Suppose that there exists $M>0$ such that 
\begin{equation}\label{1010b}
\|A(t)\| \le M \frac{\mu'(t)}{\mu(t)}, \quad t\in \mathbb R.
\end{equation}
The following properties are equivalent:
\begin{enumerate}
\item[(i)] \eqref{209lde} admits a uniform $\mu$-dichotomy;
\item[(ii)] there exists a bounded and differentiable map $W\colon \mathbb R\to \mathcal B(X)$ such that $W(t)$ is invertible and self-adjoint for each $t$, $\sup_{t\in \mathbb R}\|W(t)^{-1}\|<\infty$, and for some $\delta>0$ we have 
\begin{equation}\label{lyap}
\left \langle W'(t)x+A(t)^*W(t)x+W(t)A(t)x, x \right \rangle \le -\delta\frac{\mu'(t)}{\mu(t)}\langle x, x\rangle,
\end{equation}
for every $t\in \mathbb R$ and $x\in X$.
\end{enumerate}

Let $\tilde A\colon \mathbb R\to \mathcal B(X)$ be given by~\eqref{AA}. Note that~\eqref{1010b} implies that $\sup_{t\in \mathbb R}\|\tilde A(t)\|<\infty$.

$(i)\implies (ii)$ Assume that~\eqref{209lde} admits a uniform $\mu$-dichotomy. Then the  linear equation
\begin{equation}\label{1:14}
x'=\tilde A(t)x, \quad t\in \mathbb R
\end{equation}
admits a uniform exponential dichotomy.  By~\cite[Corollary 4.48]{CL} there is a bounded and differentiable map $\tilde W\colon \mathbb R\to \mathcal B(X)$ such that $\tilde W(t)$ is self-adjoint and invertible for each $t$, $\sup_{t\in \mathbb R}\|\tilde W(t)^{-1}\|<\infty$, and for some $\delta>0$ we have 
\begin{equation}\label{1104}
    \left \langle \tilde W'(t)x+\tilde A(t)^*\tilde W(t)x+\tilde W(t)\tilde A(t)x, x\right \rangle \le -\delta \langle x, x \rangle,
\end{equation}
for every $t\in \mathbb R$ and $x\in X$. Set $W(t):=\tilde W(\ln \mu(t))$, $t\in \mathbb R$. Then $W$ is bounded and  differentiable, $W(t)$ is self-adjoint and invertible for each $t$ and $\sup_{t\in \mathbb R}\|W(t)^{-1}\|<\infty$. Moreover, \eqref{1104} implies that 
\[
\left \langle \tilde W'(\ln \mu(t))x+\frac{\mu(t)}{\mu'(t)}A(t)^*W(t)x+\frac{\mu(t)}{\mu'(t)}W(t)A(t)x, x\right \rangle \le -\delta \langle x, x\rangle,
\]
for each $t\in \mathbb R$ and $x\in X$. By taking into account that $W'(t)=\frac{\mu'(t)}{\mu(t)}\tilde W'(\ln \mu(t))$, we conclude that~\eqref{lyap} holds.

$(ii)\implies (i)$ It follows by reversing the arguments in the proof that $(i)$ implies $(ii)$. More precisely, suppose that there exists $W\colon \mathbb R\to \mathcal B(X)$ as in $(ii)$. Set \[\tilde W(t):=W(\mu^{-1}(e^t)), \quad  t\in \mathbb R.\]
Then $\tilde W\colon \mathbb R\to \mathcal B(X)$ is bounded  and differentiable, $\tilde W(t)$ is self-adjoint and invertible for each $t\in \mathbb R$, $\sup_{t\in \mathbb R}\|\tilde W(t)^{-1}\|<\infty$ and~\eqref{1104} holds. Then~\eqref{1:14} admits a uniform exponential dichotomy, which implies $(i)$.

\item[(ii)] for each $t\ge 0$, $\mathbf S_t$ is homeomorphic to $X_0(t)$, where

<!-- theorem-10: body_md -->
\label{thm:stable-manifold-mu}
Let $\mu \colon [0, \infty)\to [1, \infty)$ be a continuously differentiable growth rate with $\mu'(t)>0$ for each $t$. 
Suppose that $f$ satisfies the following:
\begin{enumerate}
\item[(a)] $f(t, 0)=0$ for $t\ge 0$;
\item[(b)] there is $c>0$ such that
\begin{equation}\label{lip2}
\|f(t, x)-f(t, y)\| \le c\frac{\mu'(t)}{\mu(t)}\|x-y\|, \quad \text{for $t\ge 0$ and $x, y\in X$.}
\end{equation}
\end{enumerate}
Moreover, suppose that $\{T(t, s)\}_{t\ge s\ge 0}$ admits a uniform $\mu$-dichotomy. Provided that $c$ is sufficiently small, there is an invariant stable manifold $\mathbf S$ for~\eqref{integeq}. Moreover, there are $\nu, C_\nu>0$ such that for each $t_0>0$ and solutions $u_1, u_2$ of~\eqref{integeq} on $[t_0, \infty)$ such that $u_i(t_0)=v_i+g_{t_0}(v_i)\in \mathbf S_{t_0}$ with $v_i\in X_0(t_0)$, $i=1, 2$, we have 
\[
\|u_1(t)-u_2(t)\| \le C_\nu \left (\frac{\mu(t)}{\mu(t_0)}\right )^{-\nu}\|v_1-v_2\|, \quad t\ge t_0.
\]

<!-- proof-10: body_md -->
We define $h\colon \mathbb R^+\times X\to X$ by 
\[
h(t, x):=\frac{e^t}{\mu'(\mu^{-1}(e^t))}f(\mu^{-1}(e^t), x), \quad (t, x)\in \mathbb R^+\times X.
\]
Then $h(t, 0)=0$ for $t\ge 0$. Moreover, it follows from~\eqref{lip2} that 
\[
\|h(t, x)-h(t, y)\| \le c\|x-y\|, \quad \text{for $t\ge 0$ and $x, y\in X$.}
\]
In other words, $h$ satisfies the assumptions of Theorem~\ref{HuyTM}. Furthermore, since $\{T(t, s)\}_{t\ge s\ge 0}$ admits a uniform $\mu$-dichotomy, $\{T^{\mu, exp}(t, s)\}_{t\ge s \ge 0}$ admits an exponential dichotomy. We consider the associated integral equation
\begin{equation}\label{inteq2}
u(t)=T^{\mu, exp}(t, s)u(s)+\int_s^t T^{\mu, exp}(t, r)h(r, u(r))\, dr, \quad t\ge s.
\end{equation}
By Theorem~\ref{HuyTM}, \eqref{inteq2} has an invariant stable manifold $\mathbf{\bar S}$ such that for $t\ge 0$,
\[
\mathbf {\bar S}_t=\{w\in X: \ (t, w)\in \mathbf {\bar S}\}=\{v+g_t(v): \ v\in  X_0(t)\},
\]
where $g_t\colon  X_0(t)\to   X_1(t)$ and $X=X_0(t)\oplus  X_1(t)$, $t\ge 0$ is the splitting associated with $\mathbf {\bar S}$ (or dichotomy of $\{T^{\mu, exp}(t, s)\}_{t\ge s}$).


It remains to observe that if $t\mapsto u(t)$ is a (bounded)  solution of~\eqref{inteq2} on $[t_0, \infty)$, then  $t\mapsto u(\ln \mu(t))$ is a (bounded) solution of~\eqref{integeq} on $[\mu^{-1}(e^{t_0}), \infty)$, and if $t\mapsto u(t)$ is a  (bounded) solution of~\eqref{integeq} on $[t_0, \infty)$, then $t\mapsto u(\mu^{-1}(e^t))$ is a (bounded) solution of~\eqref{inteq2} on $[\ln \mu(t_0), \infty)$. The invariant stable manifold $\mathbf S$ for~\eqref{integeq} is thus given by 
\[
\mathbf S_t=\{v+g_{\ln \mu(t)}(v): \ v\in  X_0(\ln \mu(t))\}, \quad t\ge 0.
\]

<!-- remark-8: body_md -->
Theorem~\ref{thm:stable-manifold-mu} is obtained from Huy's exponential	stable manifold theorem, Theorem 4.6 in~\cite{Huy}, through the	time rescaling $\tau=\ln\mu(t)$.
	In fact, a uniform $\mu$-dichotomy for the original evolution family becomes a uniform exponential dichotomy after this time change of variables. Moreover, the factor $\mu'(t)/\mu(t)$ in~\eqref{lip2} exactly compensates for the Jacobian of the change of time,	so that the transformed nonlinearity satisfies the uniform Lipschitz condition required in Theorem 4.6 in~\cite{Huy}. Finally, exponential contraction in the new time variable, $e^{-\nu(\tau-\tau_0)}$, becomes $\left(\mu(t)/\mu(t_0)\right)^{-\nu}$ in the original time variable. Thus, the invariant stable manifold, together with the  quantitative decay estimate for trajectories on it, is transported directly from Huy's
	exponential setting.
	
	Stable manifold theorems for nonautonomous equations under nonuniform exponential, nonuniform polynomial, and more general nonuniform dichotomies have also been established in~\cite{BV2,BS3,BS4}. Those results allow more general nonuniform linear dynamics and impose different conditions on the nonlinear perturbation. Therefore, they are not directly comparable with Theorem~\ref{thm:stable-manifold-mu}. 
	
	The specific feature of our result is that, under hypotheses exactly adapted to the time change	of variables $\tau=\ln\mu(t)$, a global Lipschitz stable manifold for a uniform $\mu$-dichotomy follows directly from the corresponding exponential result, without requiring a new graph-transform construction.
