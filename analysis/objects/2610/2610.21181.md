<!-- mainthm-1: body_md -->
\label{thmA}
Let $S\subset\mathbb R^n_+$ be compact and convex with $0\in S$, and assume that $\mathbb R_+S$ is a rational polyhedral cone, that is, the set
of nonnegative linear combinations of finitely many vectors with rational
coordinates. If $K\subset\mathbb C^n$ is compact and $S$-convex, then every function holomorphic in a neighbourhood of $K$ can
be approximated uniformly on $K$ by elements of $\mathcal P^S$. Moreover, a
compact set $K$ is $S$-convex if and only if $K\subset U_S$ and the image
of $K$ under a monomial map defined by generators of $\mathcal P^S$ is
polynomially convex.
<!-- end mainthm-1 -->
<!-- mainthm-2: body_md -->
\label{thmB}
Let $S\subset\mathbb R^n_+$ be compact and convex with $0\in S$, and let
$K\subset\mathbb C^n$ be a nonempty compact $S$-convex set. The following
conditions are equivalent.
\begin{enumerate}[label=\textup{(\roman*)}]
\item Every function holomorphic in a neighbourhood of $K$ is a uniform
limit on $K$ of elements of $\mathcal P^S$.
\item There is a finitely generated monomial subring $\mathcal B\subset\mathcal P^S$
such that $\widehat K^{\mathcal B}$ is compact.
\item There are $\beta^1,\ldots,\beta^r\in G_S$ and
$q_1,\ldots,q_r\in\mathcal P^S$ such that
\[
 \Bigl\|\,1-\sum_{k=1}^r z^{\beta^k}q_k\Bigr\|_K<1 .
\]
\item $K$ is contained in a bounded sparse polyhedron.
\end{enumerate}
<!-- end mainthm-2 -->
<!-- mainthm-3: body_md -->
\label{thmC}
Let $S\subset\mathbb R^n_+$ be compact and convex with $0\in S$, and let $K$ be a
compact $S$-convex set. The conditions of Theorem~\ref{thmB} hold, and
hence the Oka--Weil theorem holds on $K$, in each of the following cases:
\begin{enumerate}[label=\textup{(\alph*)}]
\item $n=2$;
\item $K\subset(\mathbb C^*)^n$;
\item there is a nonempty proper subset $I\subset\{1,\ldots,n\}$ such that
 \begin{itemize}
 \item $e_i\in\mathbb R_+S$ for every $i\in I$;
 \item for all $i\in I$ and $k\notin I$ there is an integer $m\ge1$ with
 $me_i+e_k\in\mathbb R_+S$;
 \item every nonzero $\alpha\in\mathbb R_+S\cap\mathbb N^n$ satisfies
 $\alpha-\delta e_i\in\mathbb R_+S$ for some $i\in I$ and some $\delta>0$.
 \end{itemize}
\end{enumerate}
Moreover, for arbitrary $n$ and $S$ the Oka--Weil theorem holds on every
compact $S$-convex set contained in a coordinate subspace of dimension at
most two.

<!-- mainthm-4: body_md -->
\label{thmD}
Let $K\subset\mathbb C^n$ be compact and assume that $\widehat K^{\mathcal B}$ is
compact for some finitely generated monomial subring $\mathcal B\subset\mathcal P^S$; by
Theorems~\ref{thmA} and~\ref{thmC} this holds, for instance, for every
compact $S$-convex set when $\mathbb R_+S$ is rational polyhedral or $n=2$. A
point $z\in\mathbb C^n$ lies outside $\widehat K^S$ if and only if there is a
continuous family $(p_t)_{t\ge0}$ in some $\mathcal P^S_m$ such that
\begin{itemize}
\item $p_0(z)=0$ and no $p_t$ vanishes on $K$;
\item the hypersurfaces $\{p_t=0\}$ leave every compact subset of $\mathbb C^n$
as $t\to\infty$.
\end{itemize}

<!-- lemma-1: body_md -->
\label{lem:strata}
Let $\Gamma\subset\mathbb R^n_+$ be a convex cone, let $z\in\mathbb C^n$, and let
$J=\{j:\ z_j\ne0\}$. The following conditions are equivalent.
\begin{enumerate}[label=\textup{(\arabic*)}]
\item There is $\beta\in G_\Gamma\cap\mathbb R^J$; equivalently, $z\in U_\Gamma$.
\item $M_J$ spans $\mathbb R^J$, and for every $i\notin J$ there is
$\alpha\in M_\Gamma\cap\mathbb R^{J\cup\{i\}}$ with $\alpha_i>0$.
\item The fibre
\[
 F(z)=\bigl\{w\in\mathbb C^n:\ w^\alpha=z^\alpha\ \text{for all}\
 \alpha\in M_\Gamma\bigr\}
\]
is bounded.
\end{enumerate}
If these conditions hold, then $F(z)=\{z\}$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
A good exponent $\beta$ satisfies $z^\beta\ne0$ if and only if
$\beta\in\mathbb R^J$, which gives the equivalence inside (1). If (1) holds and
$w\in F(z)$, then $w^\beta=z^\beta\ne0$ and
\[
 w_j=\frac{w^{\beta+e_j}}{w^\beta}=\frac{z^{\beta+e_j}}{z^\beta}=z_j
 \qquad(j=1,\ldots,n),
\]
so $F(z)=\{z\}$ and (3) holds.

Assume that (2) fails. If $M_J$ does not span $\mathbb R^J$, choose a nonzero
$v\in\mathbb R^J$ orthogonal to $M_J$ and put, for $t\in\mathbb R$,
\[
 w(t)_j=z_je^{tv_j}\quad(j\in J),\qquad w(t)_i=0\quad(i\notin J).
\]
If $\alpha\in M_\Gamma$ is supported in $J$, then
$w(t)^\alpha=z^\alpha e^{t\langle\alpha,v\rangle}=z^\alpha$; otherwise
both $w(t)^\alpha$ and $z^\alpha$ vanish. Thus the unbounded curve $w(t)$
lies in $F(z)$. If for some $i\notin J$ no
$\alpha\in M_\Gamma\cap\mathbb R^{J\cup\{i\}}$ has $\alpha_i>0$, then every
$\alpha\in M_\Gamma$ either has $\alpha_i=0$ or has a positive coordinate
outside $J\cup\{i\}$, and in both cases $w\mapsto w^\alpha$ is constant on
the line $z+\mathbb C e_i$; hence this line lies in $F(z)$. Therefore (3)
implies (2).

Assume (2). Choose linearly independent
$\gamma^1,\ldots,\gamma^{|J|}\in M_J$, let $C\subset\Gamma$ be the cone
they generate, and let $\gamma=\gamma^1+\cdots+\gamma^{|J|}$ (with
$\gamma=0$ if $J=\varnothing$). Then $\gamma$ lies in the interior of $C$ relative to $\mathbb R^J$: in the
basis $\gamma^1,\ldots,\gamma^{|J|}$ of $\mathbb R^J$ the cone $C$ is the
positive orthant, and $\gamma$ has all coordinates equal to $1$. Since
$\gamma+e_j/k\to\gamma$ as $k\to\infty$, for $j\in J$ we have
\[
 k\gamma+e_j=k\Bigl(\gamma+\frac{e_j}k\Bigr)\in C
 \qquad\text{for all large }k .
\]
For $i\notin J$ let $\alpha$ be as in (2) and write
$\alpha/\alpha_i=e_i+v_i$ with $v_i\in\mathbb R^J$. For large $k$ we have $k\gamma-v_i=k(\gamma-v_i/k)\in C$ for the same
reason, and hence, since $\Gamma$ is closed under addition and
$\alpha/\alpha_i\in\Gamma$,
\[
 k\gamma+e_i=(k\gamma-v_i)+\frac{\alpha}{\alpha_i}\in\Gamma .
\]
Taking one large integer $k$ for all indices, we obtain
$\beta=k\gamma\in G_\Gamma\cap\mathbb R^J$.

<!-- corollary-1: body_md -->
\label{cor:goodcover}
Let $\Gamma\subset\mathbb R^n_+$ be a convex cone and let $K\subset\mathbb C^n$ be a
nonempty compact set with $\widehat K^\Gamma=K$ or $\mathcal R_\Gamma(K)=K$.
Then $K\subset U_\Gamma$, so that there are finitely many
$\beta^1,\ldots,\beta^r\in G_\Gamma$ with
\[
 K\subset\bigcup_{k=1}^r\{z^{\beta^k}\ne0\}.
\]
Moreover, $\Gamma$ has nonempty interior, and $0\notin K$ unless
$\Gamma=\mathbb R^n_+$.
<!-- end corollary-1 -->
<!-- proof-2: body_md -->
Every element of $\mathcal P^\Gamma$ is constant on $F(z)$, so $F(z)$ is
contained in both hulls of $K$ when $z\in K$. Hence $F(z)$ is bounded and
Lemma~\ref{lem:strata} gives $K\subset U_\Gamma$; the finite cover follows
from compactness. If $\beta\in G_\Gamma$, then $e_j=(\beta+e_j)-\beta$
lies in the linear span of $M_\Gamma$ for every $j$, so $\Gamma$ has
nonempty interior. If $0\in K$, then condition (2) of
Lemma~\ref{lem:strata} with $J=\varnothing$ gives $e_i\in\Gamma$ for all
$i$, so $\Gamma=\mathbb R^n_+$.

<!-- example-1: body_md -->
\label{ex:triangle}
Let $S=\operatorname{conv}\{0,e_1,e_1+e_2\}$. Then
\[
 \Gamma_S=\{x\in\mathbb R^2_+:\ x_2\le x_1\},\qquad
 \mathcal P^S=\mathbb C[z_1,\,z_1z_2],
\]
since $M_{\Gamma_S}$ is generated by $(1,0)$ and $(1,1)$. A pair
$\beta=(a,b)$ is good exactly when $b<a$, so $U_S=\{z_1\ne0\}$. For $0<r<2$
and $s>0$ the map $(z_1,z_2)\mapsto(z_1,z_1z_2)$ identifies the set
\[
 K_{r,s}=\bigl\{z\in\mathbb C^2:\ |z_1-2|\le r,\ |z_1z_2|\le s\bigr\}
\]
with a closed bidisc, so $K_{r,s}$ is compact and $S$-convex, and it meets
$\{z_2=0\}$. For the unit torus $\mathbb T^2$, however,
\[
 \widehat{\mathbb T^2}^{\,S}=\bigl\{z\in\mathbb C^2:\ |z_1|\le1,\ |z_1z_2|\le1\bigr\}
\]
contains the line $\{z_1=0\}$; the hull of a compact set need not be
bounded. This example is taken from \cite[Section~1]{Sig}, where it is
also observed that the hull of a single point $(0,z_2)$ is the whole line
$\{z_1=0\}$.

<!-- lemma-2: body_md -->
\label{lem:chartlemma}
Let $\mathcal B$ be a monomial algebra and let $\varphi:\mathcal B\to\mathbb C$ be a unital
homomorphism with $\varphi(z^\beta)\ne0$ for some $\beta\in G(\mathcal B)$. Put
\[
 z_j=\frac{\varphi(z^{\beta+e_j})}{\varphi(z^\beta)},\qquad j=1,\ldots,n.
\]
Then $z^\beta\ne0$, and $\varphi(b)=b(z)$ for every $b\in\mathcal B$.
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
Let $z^\alpha\in\mathcal B$. Since
$\alpha+|\alpha|\beta=\sum_j\alpha_j(\beta+e_j)$, we have the identity
\[
 z^\alpha\,(z^\beta)^{|\alpha|}=\prod_{j=1}^n(z^{\beta+e_j})^{\alpha_j}
\]
between products of elements of $\mathcal B$; here
$|\alpha|=\alpha_1+\cdots+\alpha_n$, and all factors belong to $\mathcal B$
because $z^\alpha\in\mathcal B$ and $\beta\in G(\mathcal B)$. Since $\varphi$ is
multiplicative and $\varphi(z^{\beta+e_j})=z_j\,\varphi(z^\beta)$ by the
definition of $z_j$, applying $\varphi$ gives
\[
 \varphi(z^\alpha)\,\varphi(z^\beta)^{|\alpha|}
 =\prod_{j=1}^n\bigl(z_j\,\varphi(z^\beta)\bigr)^{\alpha_j}
 =z^\alpha\,\varphi(z^\beta)^{|\alpha|},
\]
hence $\varphi(z^\alpha)=z^\alpha$. For $\alpha=\beta$ this gives
$z^\beta=\varphi(z^\beta)\ne0$, and the assertion follows by linearity.

<!-- lemma-3: body_md -->
\label{lem:nopieces}
Let $L\subset\mathbb C^N$ be compact and let $E$ be a compact set disjoint from
$L$. If $\widehat L=L\cup E$ or $\mathcal R(L)=L\cup E$, then $E=\varnothing$.
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
Let $X$ be $\widehat L$ or $\mathcal R(L)$; it is compact and, respectively,
polynomially or rationally convex. Since $L$ and $E$ are disjoint compact
sets, they have disjoint open neighbourhoods $V_L$ and $V_E$, and the
function equal to $0$ on $V_L$ and to $1$ on $V_E$ is holomorphic on the
neighbourhood $V_L\cup V_E$ of $X$. By the Oka--Weil theorem or
its rational version \cite[Ch.~1]{Stout} there is a polynomial,
respectively a rational function $G=P/Q$ with $Q\ne0$ on $X$, such that
\[
 |G|<\tfrac14\ \text{on}\ L,\qquad |G|>\tfrac34\ \text{on}\ E .
\]
In the polynomial case every $y\in E$ satisfies
$|G(y)|\le\|G\|_L<1/4$ because $y\in\widehat L$, a contradiction. In the
rational case let $y\in E$ and $c=G(y)$. The polynomial $P-cQ$ vanishes at
$y\in\mathcal R(L)$, hence at some $x\in L$, and $G(x)=c$ because $Q(x)\ne0$;
this contradicts $|c|>3/4>|G(x)|$.

<!-- theorem-1: body_md -->
\label{thm:chart}
Let $\mathcal B$ be a finitely generated monomial algebra, and let
$K\subset\mathbb C^n$ be a compact set with $K\subset U_\mathcal B$.
\begin{enumerate}[label=\textup{(\alph*)}]
\item If $\widehat K^\mathcal B=K$, then $\pi(K)$ is polynomially convex, every
character of $A_\mathcal B(K)$ is the evaluation at a point of $K$, and there
are $\beta^1,\ldots,\beta^r\in G(\mathcal B)$ and $q_1,\ldots,q_r\in\mathcal B$ with
\begin{equation}\label{eq:certificate}
 \Bigl\|\,1-\sum_{k=1}^r z^{\beta^k}q_k\Bigr\|_K<1 .
\end{equation}
Every function holomorphic in a neighbourhood of $K$ belongs to
$A_\mathcal B(K)$.
\item If $\mathcal R_\mathcal B(K)=K$, then $\pi(K)$ is rationally convex, and every
function holomorphic in a neighbourhood of $K$ belongs to
$A^{\mathrm{rat}}_\mathcal B(K)$.
\end{enumerate}
<!-- end theorem-1 -->
<!-- proof-5: body_md -->
Choose $\beta^1,\ldots,\beta^r\in G(\mathcal B)$ such that the sets
$\{z^{\beta^k}\ne0\}$ cover $K$. Enlarging the set of generators we may
assume that $z^{\beta^k}=g_{i_k}$, so that
\[
 z^{\beta^k}=w_{i_k}\circ\pi,\qquad k=1,\ldots,r,
\]
where $w_1,\ldots,w_N$ are the coordinates of $\mathbb C^N$. Put $L=\pi(K)$.

(a) Let $W=\widehat L$. For $w\in W$ and $b=Q\circ\pi\in\mathcal B$ put
$\varphi_w(b)=Q(w)$. This is well defined, since $Q\circ\pi=Q'\circ\pi$
implies
\[
 |(Q-Q')(w)|\le\|Q-Q'\|_L=0 .
\]
Thus $\varphi_w$ is a unital homomorphism with $|\varphi_w(b)|\le\|b\|_K$.
Let
\[
 E=\bigl\{w\in W:\ w_{i_1}=\cdots=w_{i_r}=0\bigr\}.
\]
If $w\in W\setminus E$, then $\varphi_w(z^{\beta^k})\ne0$ for some $k$,
and Lemma~\ref{lem:chartlemma} gives a point $z$ with $\varphi_w(b)=b(z)$
for all $b\in\mathcal B$. Then $|b(z)|\le\|b\|_K$ for all $b$, so
$z\in\widehat K^\mathcal B=K$, and
\[
 w=\bigl(\varphi_w(g_1),\ldots,\varphi_w(g_N)\bigr)=\pi(z)\in L .
\]
Hence $W=L\cup E$. The set $E$ is closed in the compact set $W$, being a
common zero set of coordinate functions, and hence compact. It is disjoint
from $L$, because every point of $K$ has some $z^{\beta^k}\ne0$. Therefore
Lemma~\ref{lem:nopieces} gives $W=L$.

Let $\psi$ be a character of $A=A_\mathcal B(K)$ and put
$w=(\psi(g_1),\ldots,\psi(g_N))$. For every polynomial $Q$ on $\mathbb C^N$ we
have $Q(w)=\psi(Q\circ\pi)$, hence
\[
 |Q(w)|\le\|Q\circ\pi\|_K=\|Q\|_L ,
\]
and $w\in W=L$. Thus $w=\pi(z)$ with $z\in K$, and $\psi$ agrees with the
evaluation at $z$ on the generators, hence on $\mathcal B$ and, by continuity, on
$A$.

Since $K$ is covered by the sets $\{z^{\beta^k}\ne0\}$ and every
character is the evaluation at a point of $K$, no character of $A$
vanishes on all the functions $z^{\beta^k}$. By (G3) there are
$h_k\in A$ with
\[
 \sum_{k=1}^r z^{\beta^k}h_k=1\qquad\text{on }K .
\]
Choose $\eta>0$ with $\eta\sum_k\|z^{\beta^k}\|_K<1$ and $q_k\in\mathcal B$
with $\|h_k-q_k\|_K<\eta$. Then
\[
 \Bigl\|\,1-\sum_{k=1}^r z^{\beta^k}q_k\Bigr\|_K
 =\Bigl\|\sum_{k=1}^r z^{\beta^k}(h_k-q_k)\Bigr\|_K
 \le\eta\sum_{k=1}^r\|z^{\beta^k}\|_K<1,
\]
which is \eqref{eq:certificate}. Moreover, on $K$ we have
\[
 z_j=\sum_{k=1}^r z^{\beta^k+e_j}h_k,\qquad z^{\beta^k+e_j}\in\mathcal B,
\]
so every coordinate function belongs to the algebra $A$, and hence every
polynomial does.
Since $\widehat K\subset\widehat K^\mathcal B=K$, the set $K$ is polynomially
convex, and the classical Oka--Weil theorem shows that every function
holomorphic near $K$ lies in the closure of the polynomials on $K$, which
is contained in $A$.

(b) Let $R=\mathcal R(L)$. For $w\in R$ and $b=Q\circ\pi$ put again
$\varphi_w(b)=Q(w)$; if $Q\circ\pi=Q'\circ\pi$, then $Q-Q'$ vanishes on
$L$, so that $(Q-Q')(w)\in(Q-Q')(L)=\{0\}$. If $\varphi_w(z^{\beta^k})\ne0$
for some $k$, Lemma~\ref{lem:chartlemma} gives $z$ with
$\varphi_w(b)=b(z)$ for all $b$, and
\[
 b(z)=Q(w)\in Q(L)=b(K)\qquad\text{for all }b=Q\circ\pi\in\mathcal B ,
\]
so $z\in\mathcal R_\mathcal B(K)=K$ and $w=\pi(z)\in L$. As before, $R=L\cup E$ with $E$
compact and disjoint from $L$, and Lemma~\ref{lem:nopieces} gives $R=L$.

Let $\psi$ be a character of $A^{\mathrm{rat}}=A^{\mathrm{rat}}_\mathcal B(K)$
and $w=(\psi(g_1),\ldots,\psi(g_N))$. If $Q$ is a polynomial with
$Q(w)=0$, then $Q$ has a zero on $L$. Otherwise $b=Q\circ\pi$ has no zero
on $K$, so that $1/b\in A^{\mathrm{rat}}$ and
\[
 1=\psi(b)\,\psi(1/b)=Q(w)\,\psi(1/b)=0 .
\]
Applying this to $Q-Q(w)$ we get $w\in R=L$, so $w=\pi(z)$ with $z\in K$.
Then $\psi(b)=b(z)$ on $\mathcal B$, and $\psi(b/c)=b(z)/c(z)$ for every
admissible quotient because $\psi(b)=\psi(c)\psi(b/c)$; by continuity
$\psi$ is the evaluation at $z$. The argument of~(a) now shows that the
coordinate functions belong to $A^{\mathrm{rat}}$. A polynomial $q$ without zeros on $K$ therefore belongs to
$A^{\mathrm{rat}}$, and no character vanishes on it because characters
are point evaluations at points of $K$; by (G3), $1/q\in A^{\mathrm{rat}}$.
Thus $A^{\mathrm{rat}}$ contains every rational function without poles on
$K$. Finally
$\mathcal R(K)\subset\mathcal R_\mathcal B(K)=K$, and the rational Oka--Weil theorem
\cite[Ch.~1]{Stout} completes the proof.

<!-- theorem-2: body_md -->
\label{thm:A}
Let $\Gamma\subset\mathbb R^n_+$ be a rational polyhedral cone, let
$g_1,\ldots,g_N$ be monomial generators of $\mathcal P^\Gamma$, and let
$\pi=(g_1,\ldots,g_N)$. For a compact set $K\subset\mathbb C^n$ the following
hold.
\begin{enumerate}[label=\textup{(\alph*)}]
\item $K$ is $\Gamma$-convex if and only if $K\subset U_\Gamma$ and
$\pi(K)$ is polynomially convex. In this case the Oka--Weil property holds
on $K$.
\item If $\mathcal R_\Gamma(K)=K$, then every function holomorphic in a
neighbourhood of $K$ is a uniform limit on $K$ of quotients $p/q$ with
$p,q\in\mathcal P^\Gamma$ and $q$ without zeros on $K$.
\end{enumerate}
In particular, Theorem~\ref{thmA} holds.
<!-- end theorem-2 -->
<!-- proof-6: body_md -->
We first show that $\mathcal P^\Gamma$ is a finitely generated monomial algebra.
After clearing denominators, $\Gamma$ is the set of nonnegative linear
combinations of finitely many integer vectors $v_1,\ldots,v_m$. Let
$\alpha\in M_\Gamma$ and write $\alpha=\sum_lt_lv_l$ with $t_l\ge0$.
Splitting each $t_l$ into its integer part $\lfloor t_l\rfloor$ and its
fractional part $t_l-\lfloor t_l\rfloor\in[0,1)$, we get
\[
 \alpha=\sum_{l=1}^m\lfloor t_l\rfloor\,v_l+\gamma,\qquad
 \gamma=\sum_{l=1}^m\bigl(t_l-\lfloor t_l\rfloor\bigr)v_l .
\]
The vector $\gamma=\alpha-\sum_l\lfloor t_l\rfloor v_l$ has integer
coordinates and lies in $\Gamma\subset\mathbb R^n_+$, so $\gamma\in M_\Gamma$;
moreover $\gamma$ lies in the bounded set
$\Pi=\{\sum_ls_lv_l:\ 0\le s_l<1\}$, which contains only finitely many
lattice points. Hence $M_\Gamma$ is generated, as a semigroup, by
$v_1,\ldots,v_m$ and the finitely many points of $M_\Gamma\cap\Pi$, and
$\mathcal P^\Gamma$ is generated by the corresponding monomials. (This is the
usual proof of Gordan's lemma \cite[Proposition~1.2.17]{CLS}.) If $K$ is $\Gamma$-convex, Corollary~\ref{cor:goodcover} gives
$K\subset U_\Gamma$, and Theorem~\ref{thm:chart}(a) shows that $\pi(K)$ is
polynomially convex and that the Oka--Weil property holds. Conversely,
assume that $K\subset U_\Gamma$ and that $\pi(K)$ is polynomially convex.
By \eqref{eq:pullback},
\[
 \widehat K^\Gamma=\pi^{-1}\bigl(\pi(K)\bigr).
\]
If $\pi(z)=\pi(z')$ with $z'\in K$, then every element of $\mathcal P^\Gamma$
takes the same value at $z$ and at $z'$. Choosing $\beta\in G_\Gamma$ with
$z'^\beta\ne0$ we get $z^\beta=z'^\beta\ne0$ and
\[
 z_j=\frac{z^{\beta+e_j}}{z^\beta}=\frac{z'^{\beta+e_j}}{z'^\beta}=z'_j
 \qquad(j=1,\ldots,n).
\]
Hence $\widehat K^\Gamma=K$. Part (b) follows in the same way from
Corollary~\ref{cor:goodcover} and Theorem~\ref{thm:chart}(b).

<!-- example-2: body_md -->
\label{ex:cusp}
This classical example shows what goes wrong without good charts. Let $\mathcal B=\mathbb C[z^2,z^3]\subset\mathbb C[z]$ and let $K$ be the closed unit disc.

<!-- theorem-3: body_md -->
\label{thm:B}
Let $\Gamma\subset\mathbb R^n_+$ be a convex cone and let $K\subset\mathbb C^n$ be a
nonempty compact $\Gamma$-convex set. Let $A_\Gamma(K)$ be the uniform
closure of $\mathcal P^\Gamma|_K$. The following conditions are equivalent.
\begin{enumerate}[label=\textup{(\roman*)}]
\item The Oka--Weil property holds on $K$.
\item Every character of $A_\Gamma(K)$ is the evaluation at a point of
$K$.
\item There are $\beta^1,\ldots,\beta^r\in G_\Gamma$ and
$q_1,\ldots,q_r\in\mathcal P^\Gamma$ with
\[
 \Bigl\|\,1-\sum_{k=1}^r z^{\beta^k}q_k\Bigr\|_K<1 .
\]
\item $K$ is contained in a bounded sparse polyhedron.
\item There is a finitely generated monomial algebra
$\mathcal B_0\subset\mathcal P^\Gamma$ such that $\widehat K^{\mathcal B_0}$ is compact.
\end{enumerate}
If these conditions hold, then every neighbourhood of $K$ contains a
bounded sparse polyhedron around $K$.

<!-- proof-7: body_md -->
(i)$\Rightarrow$(ii). By (i), $A_\Gamma(K)$ contains all polynomials, so
it is the uniform closure $P(K)$ of the polynomials on $K$. If $\psi$ is a
character and $w=(\psi(z_1),\ldots,\psi(z_n))$, then
\[
 |q(w)|=|\psi(q)|\le\|q\|_K\qquad\text{for every polynomial }q,
\]
so $w\in\widehat K\subset\widehat K^\Gamma=K$. Thus $\psi$ is the
evaluation at $w$ on the polynomials, and hence, by the continuity (G1)
of $\psi$, on $P(K)$.

(ii)$\Rightarrow$(iii). By Corollary~\ref{cor:goodcover}, $K$ is covered
by finitely many sets $\{z^{\beta^k}\ne0\}$ with $\beta^k\in G_\Gamma$. By
(ii), no character of $A_\Gamma(K)$ vanishes on all $z^{\beta^k}$, and (G3) gives $h_k\in A_\Gamma(K)$ with $\sum_kz^{\beta^k}h_k=1$ on
$K$. Choosing $q_k\in\mathcal P^\Gamma$ with $\|h_k-q_k\|_K<\eta$, where
$\eta\sum_k\|z^{\beta^k}\|_K<1$, we obtain (iii) exactly as in the proof
of Theorem~\ref{thm:chart}.

(iii)$\Rightarrow$(iv). Put
\[
 F=\sum_{k=1}^r z^{\beta^k}q_k,\qquad \vartheta=\|1-F\|_K<1,
\]
and consider the set $D$ of all $z\in\mathbb C^n$ such that
\[
 |1-F(z)|<\frac{1+\vartheta}2,\qquad
 |q_k(z)|<\|q_k\|_K+1,\qquad
 |z^{\beta^k+e_j}|<\|z^{\beta^k+e_j}\|_K+1
\]
for all $k$ and $j$. It is a sparse polyhedron around $K$: the defining
functions $1-F$, $q_k$ and $z^{\beta^k+e_j}$ belong to $\mathcal P^\Gamma$, and
each bound exceeds the norm on $K$ of the corresponding function, since
$(1+\vartheta)/2>\vartheta$. On $D$ the first inequality gives
$|F(z)|\ge1-|1-F(z)|>(1-\vartheta)/2$, and the second gives
$|F(z)|\le\sum_k|z^{\beta^k}|\,|q_k(z)|\le C\max_k|z^{\beta^k}|$. Thus
\[
 \delta:=\frac{1-\vartheta}2<|F(z)|\le C\max_k|z^{\beta^k}|,
 \qquad C=\sum_{k=1}^r\bigl(\|q_k\|_K+1\bigr),
\]
so some $k$ satisfies $|z^{\beta^k}|\ge\delta/C$, and then
\begin{equation}\label{eq:coordbound}
 |z_j|=\frac{|z^{\beta^k+e_j}|}{|z^{\beta^k}|}
 \le\frac C\delta\,\max_l\bigl(\|z^{\beta^l+e_j}\|_K+1\bigr),
 \qquad j=1,\ldots,n.
\end{equation}
Hence $D$ is bounded.

(iv)$\Rightarrow$(v). Let $\mathcal B_0$ be generated by the monomials occurring
in the polynomials $p_j$ that define the polyhedron $D$. Every
$z\in\widehat K^{\mathcal B_0}$ satisfies
\[
 |p_j(z)|\le\|p_j\|_K<c_j\qquad(j=1,\ldots,N),
\]
so the closed set $\widehat K^{\mathcal B_0}$ lies in the bounded set $D$.

(v)$\Rightarrow$(i). Let $f$ be holomorphic in an open set
$\Omega\supset K$. By Corollary~\ref{cor:goodcover} there are
$\beta^1,\ldots,\beta^r\in G_\Gamma$ with
\[
 K\subset V:=\bigcup_{k=1}^r\{z^{\beta^k}\ne0\},
\]
and we may assume $\Omega\subset V$. Enumerate
$M_\Gamma=\{\alpha^1,\alpha^2,\ldots\}$ and let $\mathcal B_j$ be the algebra
generated by $\mathcal B_0$, the monomials $z^{\beta^k}$ and $z^{\beta^k+e_i}$
for all $k$ and $i$, and $z^{\alpha^1},\ldots,z^{\alpha^j}$. The sets
\[
 K_j=\widehat K^{\mathcal B_j},\qquad j=1,2,\ldots,
\]
are closed subsets of the compact set $\widehat K^{\mathcal B_0}$, they
decrease, and
\[
 \bigcap_jK_j=\widehat K^\Gamma=K,
\]
because $\bigcup_j\mathcal B_j=\mathcal P^\Gamma$. Hence $K_j\subset\Omega$ for some
$j$: otherwise the sets $K_j\setminus\Omega$ would form a decreasing
sequence of nonempty compact sets, and a point in their intersection
would be a point of $K\setminus\Omega$. For $b\in\mathcal B_j$ we have
$\|b\|_{K_j}\le\|b\|_K$ by the definition of $K_j$, and
$\|b\|_K\le\|b\|_{K_j}$ because $K\subset K_j$; hence
$\widehat{K_j}^{\mathcal B_j}=\widehat K^{\mathcal B_j}=K_j$, that is, $K_j$ is
$\mathcal B_j$-convex, and $K_j\subset V\subset U_{\mathcal B_j}$ because every
$\beta^k$ belongs to $G(\mathcal B_j)$. Theorem~\ref{thm:chart}(a) shows that $f$
is a uniform limit on $K_j$, and hence on $K$, of elements of
$\mathcal B_j\subset\mathcal P^\Gamma$.

For the last assertion, let $D$ be the bounded polyhedron constructed in
the proof of (iii)$\Rightarrow$(iv) and let $\Omega\supset K$ be open.
For each $w$ in the compact set $\overline D\setminus\Omega$ there are
$p_w\in\mathcal P^\Gamma$ and $t_w>0$ with
\[
 |p_w(w)|>t_w>\|p_w\|_K ,
\]
because $w\notin K=\widehat K^\Gamma$. Finitely many of the open sets
$\{|p_w|>t_w\}$ cover $\overline D\setminus\Omega$, and intersecting $D$
with the corresponding sets $\{|p_w|<t_w\}$ gives a bounded sparse
polyhedron $D'$ around $K$. The closure of $D'$ is contained in
$\overline D\cap\bigcap\{|p_w|\le t_w\}$, which does not meet
$\overline D\setminus\Omega$; hence $\overline{D'}\subset\Omega$. This is
the sparse analogue of the classical fact that a polynomially convex set
has a basis of neighbourhoods consisting of polynomial polyhedra
\cite[Section~2.7]{Hor}.

\label{rem:open}
Let $K$ be a compact $\Gamma$-convex set. By Lemma~\ref{lem:chartlemma}, a
character of $A_\Gamma(K)$ that does not vanish on all good monomials is
the evaluation at a point $z$ with $|b(z)|\le\|b\|_K$ for all
$b\in\mathcal P^\Gamma$, that is, at a point of $K$. Therefore the Oka--Weil
property on $K$ fails exactly when $A_\Gamma(K)$ has a character that
annihilates every $z^\beta$, $\beta\in G_\Gamma$, that is, when the
spectrum of $A_\Gamma(K)$ is strictly larger than $K$. By
Theorem~\ref{thm:A} this does not happen for rational polyhedral cones,
and by Corollary~\ref{cor:dimtwo} below it does not happen when $n=2$. We
do not know whether it can happen for $n\ge3$.

\label{ex:certificate}
Let
\[
 \Gamma_0=\Bigl\{x\in\mathbb R^3_+:\ x_3\le x_1+x_2+\sqrt{x_1x_2}\Bigr\}.
\]
The function $x_1+x_2+\sqrt{x_1x_2}$ is concave and positively
homogeneous, so $\Gamma_0$ is a closed convex cone with nonempty
interior. It is not polyhedral, and therefore $\mathcal P^{\Gamma_0}$ is not
finitely generated: otherwise the closed cone generated by $M_{\Gamma_0}$,
which equals $\Gamma_0$ because the rational points are dense in
$\Gamma_0$, would be polyhedral. The exponents $e_1$, $e_2$, $e_1+e_3$ and
$e_2+e_3$ lie in $\Gamma_0$, and $e_1,e_2\in G_{\Gamma_0}$; for instance
$e_1$, $2e_1$, $e_1+e_2$ and $e_1+e_3$ all satisfy the defining inequality.

Let $r>0$ and $s>0$, let
\[
 \mathbb B_r=\bigl\{(z_1,z_2)\in\mathbb C^2:\ |z_1-1|^2+|z_2-1|^2\le r^2\bigr\}
\]
be the closed Euclidean ball in $\mathbb C^2$ with centre $(1,1)$ and radius $r$,
and put
\begin{equation}\label{eq:Krs}
 K_{r,s}=\bigl\{z\in\mathbb C^3:\ (z_1,z_2)\in\mathbb B_r,\ |z_1z_3|\le s,\
 |z_2z_3|\le s\bigr\}.
\end{equation}
We assume
\[
 1<r<\sqrt2 ,
\]
and the two bounds play different roles. The distance from $(1,1)$ to
$(0,0)$ in $\mathbb C^2$ is $\sqrt2$, so $r<\sqrt2$ means exactly that
$(0,0)\notin\mathbb B_r$. Since $\mathbb B_r$ is compact, there is then
$c>0$ with $\max(|z_1|,|z_2|)\ge c$ on $\mathbb B_r$, and the last two
inequalities in \eqref{eq:Krs} give $|z_3|\le s/c$ on $K_{r,s}$; hence
$K_{r,s}$ is closed and bounded, that is, compact. On the other hand,
$r>1$ means that $\mathbb B_r$ meets the line $\{z_1=0\}$: the point
$(0,z_2)$ lies in $\mathbb B_r$ when $|z_2-1|^2\le r^2-1$. Thus $K_{r,s}$
contains the points $(0,z_2,0)$ with $|z_2-1|^2\le r^2-1$ and,
symmetrically, points $(z_1,0,0)$; in particular it meets the hyperplanes
$\{z_1=0\}$, $\{z_2=0\}$ and $\{z_3=0\}$, but not the axis
$\{z_1=z_2=0\}$.

We show that $K_{r,s}$ is $\mathcal B$-convex for
\[
 \mathcal B=\mathbb C[z_1,\,z_2,\,z_1z_3,\,z_2z_3]\subset\mathcal P^{\Gamma_0}.
\]
First, the projection of $K_{r,s}$ to the first two coordinates is all of
$\mathbb B_r$, because $(z_1,z_2,0)\in K_{r,s}$ for every
$(z_1,z_2)\in\mathbb B_r$. Second, $\mathbb B_r$ is polynomially convex:
if $w\notin\mathbb B_r$, the Hahn--Banach theorem gives a complex linear
function $\ell$ on $\mathbb C^2$ with
$\operatorname{Re}\ell(w)>\max_{\mathbb B_r}\operatorname{Re}\ell$, and
then $|e^{\ell(w)}|>\max_{\mathbb B_r}|e^\ell|$, where $e^\ell$ is a
uniform limit of polynomials on compact sets. Now let
$z\in\widehat{K_{r,s}}^{\,\mathcal B}$. Every polynomial $q(z_1,z_2)$ belongs to
$\mathcal B$, so
\[
 |q(z_1,z_2)|\le\|q\|_{K_{r,s}}=\|q\|_{\mathbb B_r},
\]
and therefore $(z_1,z_2)\in\widehat{\mathbb B_r}=\mathbb B_r$. Moreover
\[
 |z_1z_3|\le\|z_1z_3\|_{K_{r,s}}\le s,\qquad
 |z_2z_3|\le\|z_2z_3\|_{K_{r,s}}\le s .
\]
Hence $z\in K_{r,s}$, that is, $\widehat{K_{r,s}}^{\,\mathcal B}=K_{r,s}$. Since
$\mathcal B\subset\mathcal P^{\Gamma_0}$, the $\mathcal P^{\Gamma_0}$-hull is contained in the
$\mathcal B$-hull, and $K_{r,s}$ is also $\Gamma_0$-convex. In particular,
condition (v) of Theorem~\ref{thm:B} holds with $\mathcal B_0=\mathcal B$.

Condition (iii) can also be checked by hand, and this is where $r<\sqrt2$
enters a second time. With $q_1=q_2=1/2$ and $\beta^1=e_1$,
$\beta^2=e_2$, the Cauchy--Schwarz inequality gives
\[
 \Bigl\|\,1-\frac{z_1+z_2}2\Bigr\|_{K_{r,s}}
 =\frac12\,\bigl\|(1-z_1)+(1-z_2)\bigr\|_{K_{r,s}}
 \le\frac1{\sqrt2}\,\sup_{\mathbb B_r}\bigl(|1-z_1|^2+|1-z_2|^2\bigr)^{1/2}
 =\frac r{\sqrt2}<1 .
\]
Either way, the Oka--Weil property holds on $K_{r,s}$. Neither $|z_1|$ nor
$|z_2|$ is bounded below on $K_{r,s}$, but their maximum is.

\label{prop:reinhardt}
Let $\Gamma\subset\mathbb R^n_+$ be a convex cone and let $K\subset(\mathbb C^*)^n$ be
a compact Reinhardt set with $A=\operatorname{Log} K$. For $z\in(\mathbb C^*)^n$ the following

\item $z\in\widehat K^\Gamma$;
\item $|z^\alpha|\le\|z^\alpha\|_K$ for all $\alpha\in M_\Gamma$;
\item $\operatorname{Log} z\in\operatorname{conv}(A)-\Gamma_M^\vee$.
\end{enumerate}

Clearly (1) implies (2). Assume (2) and let
$p=\sum c_\alpha z^\alpha\in\mathcal P^\Gamma$. For $w\in K$ the torus
$\{(w_1e^{i\theta_1},\ldots,w_ne^{i\theta_n})\}$ lies in $K$, and the Cauchy formula
\[
 c_\alpha w^\alpha=\frac1{(2\pi)^n}\int_{[0,2\pi]^n}
 p\bigl(w_1e^{i\theta_1},\ldots,w_ne^{i\theta_n}\bigr)\,
 e^{-i\langle\alpha,\theta\rangle}\,d\theta
\]
gives $|c_\alpha|\,|w^\alpha|\le\|p\|_K$. Hence
\[
 |c_\alpha|\,\|z^\alpha\|_K\le\|p\|_K
 \qquad\text{and}\qquad
 |p(z)|\le\sum_\alpha|c_\alpha|\,|z^\alpha|\le N(p)\,\|p\|_K,
\]
where $N(p)$ is the number of monomials of $p$. Applying this to $p^k\in\mathcal P^\Gamma$, whose number of monomials is at most
$\binom{k+N-1}{N-1}\le(k+1)^{N-1}$ when $p$ has $N$ monomials, we get
\[
 |p(z)|^k\le(k+1)^{N-1}\,\|p\|_K^k ,
\]
and letting $k\to\infty$ after taking $k$-th roots gives
$|p(z)|\le\|p\|_K$. This use of high powers is the standard device for
passing from estimates of coefficients to estimates of values.

For (2)$\Leftrightarrow$(3) note that $\|z^\alpha\|_K=e^{h_A(\alpha)}$,
so that (2) says
\[
 \langle\alpha,x\rangle\le h_A(\alpha)\qquad
 \text{for } x=\operatorname{Log} z \text{ and all } \alpha\in M_\Gamma .
\]
This inequality extends to $\Gamma_M$. Indeed, if
$s=\sum_l(a_l/b)\alpha_l$ with $\alpha_l\in M_\Gamma$ and nonnegative
integers $a_l$, $b\ge1$, then $bs\in M_\Gamma$ because $M_\Gamma$ is a
semigroup, and the positive homogeneity of $h_A$ gives
$\langle s,x\rangle=b^{-1}\langle bs,x\rangle\le b^{-1}h_A(bs)=h_A(s)$;
such $s$ are dense in $\Gamma_M$, and both sides are continuous in $s$. The closed
convex set $\operatorname{conv}(A)-\Gamma_M^\vee$ has support function $h_A$ on
$\Gamma_M$ and $+\infty$ elsewhere, because
$\Gamma_M^{\vee\vee}=\Gamma_M$; hence
\[
 \operatorname{conv}(A)-\Gamma_M^\vee=\bigl\{x\in\mathbb R^n:\ \langle s,x\rangle\le h_A(s)\
 \text{for all}\ s\in\Gamma_M\bigr\}. \qedhere
\]

<!-- theorem-4: body_md -->
\label{thm:anchor}
Let $\Gamma\subset\mathbb R^n_+$ be a convex cone, let
$\varnothing\ne I\subsetneq\{1,\ldots,n\}$, assume \textup{(A1)}, and let
$K\subset\mathbb C^n$ be compact.
\begin{enumerate}[label=\textup{(\alph*)}]
\item If $0\notin\widehat{\operatorname{pr}_I(K)}$, the ordinary polynomial hull in
$\mathbb C^I$, then $\widehat K^\mathcal B$ is compact for
\[
 \mathcal B=\mathbb C\bigl[z_i,\ z_i^{m_{ik}}z_k:\ i\in I,\ k\notin I\bigr].
\]
\item If $\Gamma$ satisfies \textup{(A2)} and $K$ is $\Gamma$-convex, then
$\widehat K^\mathcal B$ is compact for some finitely generated monomial algebra
$\mathcal B\subset\mathcal P^\Gamma$.
\end{enumerate}
In both cases, if $K$ is $\Gamma$-convex, the Oka--Weil property holds on
$K$.
<!-- end theorem-4 -->
<!-- proof-10: body_md -->
In both parts we show that, for a suitable finitely generated algebra
$\mathcal B\supset\mathbb C[z_i,z_i^{m_{ik}}z_k]$ and some $\varepsilon>0$,
\begin{equation}\label{eq:anchorbound}
 \rho(z):=\max_{i\in I}|z_i|\ge\varepsilon\qquad
 \text{for all }z\in\widehat K^\mathcal B .
\end{equation}
This suffices. On $\widehat K^\mathcal B$ we have $|z_i|\le\|z_i\|_K$ for
$i\in I$, and for $k\notin I$ we choose $i\in I$ with
$|z_i|\ge\varepsilon$ and obtain
\[
 |z_k|\le\varepsilon^{-m_{ik}}\,\|z_i^{m_{ik}}z_k\|_K .
\]
Thus the closed set $\widehat K^\mathcal B$ is bounded, and Theorem~\ref{thm:B}
gives the Oka--Weil property when $K$ is $\Gamma$-convex.

(a) The algebra $\mathcal B$ contains all polynomials in $(z_i)_{i\in I}$, so
\[
 \operatorname{pr}_I\bigl(\widehat K^\mathcal B\bigr)\subset\widehat{\operatorname{pr}_I(K)} .
\]
The latter is a compact set not containing $0$, which gives
\eqref{eq:anchorbound}.

(b) By (A2), every nonzero $\alpha\in M_\Gamma$ has a positive coordinate
indexed by $I$. Hence every element of $\mathcal P^\Gamma$ is constant on the
unbounded set
\[
 V_I=\{z\in\mathbb C^n:\ z_i=0\ \text{for}\ i\in I\},
\]
and the compact $\Gamma$-convex set $K$ does not meet $V_I$; in
particular $0\notin K$. Choose
\[
 p=c+\sum_{\alpha\in F}c_\alpha z^\alpha\in\mathcal P^\Gamma,\qquad
 F\subset M_\Gamma\setminus\{0\}\ \text{finite},\qquad
 |c|=|p(0)|>\|p\|_K .
\]
For $\alpha\in F$ choose $i(\alpha)\in I$ and $\delta_\alpha>0$ as in
(A2), and integers $N_\alpha,j_\alpha\ge1$ with
$j_\alpha/N_\alpha<\delta_\alpha$. The point
$\alpha-(j_\alpha/N_\alpha)e_{i(\alpha)}$ lies on the segment between
$\alpha$ and $\alpha-\delta_\alpha e_{i(\alpha)}$, hence in $\Gamma$, and
therefore
\[
 \gamma_\alpha:=N_\alpha\alpha-j_\alpha e_{i(\alpha)}
 =N_\alpha\Bigl(\alpha-\frac{j_\alpha}{N_\alpha}\,e_{i(\alpha)}\Bigr)
 \in M_\Gamma .
\]
Let $\mathcal B$ be generated by the $z_i$, the $z_i^{m_{ik}}z_k$, and the
monomials $z^\alpha$ and $z^{\gamma_\alpha}$ for $\alpha\in F$. The
identity $N_\alpha\alpha=j_\alpha e_{i(\alpha)}+\gamma_\alpha$ gives, for
$z\in\widehat K^\mathcal B$,
\[
 |z^\alpha|
 \le|z_{i(\alpha)}|^{j_\alpha/N_\alpha}\,
 \|z^{\gamma_\alpha}\|_K^{1/N_\alpha}
 \le\rho(z)^{j_\alpha/N_\alpha}\,\|z^{\gamma_\alpha}\|_K^{1/N_\alpha}.
\]
Consequently, for $z\in\widehat K^\mathcal B$,
\[
 |p(z)-c|\le\sum_{\alpha\in F}|c_\alpha|\,
 \rho(z)^{j_\alpha/N_\alpha}\,\|z^{\gamma_\alpha}\|_K^{1/N_\alpha}.
\]
The right-hand side tends to $0$ as $\rho(z)\to0$, so there is
$\varepsilon>0$ such that it is smaller than $|c|-\|p\|_K$ whenever
$\rho(z)<\varepsilon$. For such $z$ the reverse triangle inequality gives
$|p(z)|>\|p\|_K$, which is impossible for $z\in\widehat K^\mathcal B$ because
$p\in\mathcal B$. Hence $\rho\ge\varepsilon$ on $\widehat K^\mathcal B$, which is
\eqref{eq:anchorbound}.

<!-- proposition-2: body_md -->
\label{prop:block}
Let $S\subset\mathbb R^n_+$ be compact and convex with $0\in S$, let
$\Gamma=\Gamma_S$, and let $\varnothing\ne J\subsetneq\{1,\ldots,n\}$.
Assume that
\begin{enumerate}[label=\textup{(B\arabic*)}]
\item $M_J=M_\Gamma\cap\mathbb R^J$ spans $\mathbb R^J$;
\item for every $k\notin J$ there is $\alpha^{(k)}\in M_\Gamma$ supported
in $J\cup\{k\}$ with $\alpha^{(k)}_k>0$.
\end{enumerate}
Let $K\subset\mathbb C^n$ be compact and assume that the hull $H_J$ of
$\operatorname{pr}_J(K)$ with respect to $\mathcal P^{S_J}$ in $\mathbb C^J$ is a compact subset of
$(\mathbb C^*)^J$. Then $\widehat K^\mathcal B$ is compact for some finitely generated
monomial algebra $\mathcal B\subset\mathcal P^S$. In particular, if $K$ is $S$-convex,
the Oka--Weil property holds on $K$.
<!-- end proposition-2 -->
<!-- proof-11: body_md -->
The argument combines Corollary~\ref{cor:torus} with lower bounds for
coordinates obtained from monomial inequalities, as in
\cite[Lemma~2.3]{Sig}. By (B1) the cone $\mathbb R_+S_J$ has nonempty interior in
$\mathbb R^J$. The set $H_J$
is a compact $S_J$-convex subset of $(\mathbb C^*)^J$, and
Corollary~\ref{cor:torus}, applied in $\mathbb C^J$, gives $\beta\in G_{S_J}$
with all coordinates positive and $q\in\mathcal P^{S_J}$ with
\[
 \vartheta:=\|1-z^\beta q\|_{H_J}<1 .
\]
Let $\mathcal B_J$ be generated by the monomials of $q$ and by $z^\beta$ and
$z^{\beta+e_j}$, $j\in J$. Since $\|b\|_{H_J}=\|b\|_{\operatorname{pr}_J(K)}$ for
$b\in\mathcal P^{S_J}$, every $w$ in the $\mathcal B_J$-hull $H'$ of $\operatorname{pr}_J(K)$ satisfies
\[
 |1-w^\beta q(w)|\le\vartheta,\qquad |q(w)|\le\|q\|_{H_J},\qquad
 |w^{\beta+e_j}|\le\|z^{\beta+e_j}\|_{H_J},
\]
and hence
\[
 |w^\beta|\ge\frac{1-\vartheta}{\|q\|_{H_J}}>0,\qquad
 |w_j|=\frac{|w^{\beta+e_j}|}{|w^\beta|}\le
 \frac{\|q\|_{H_J}\,\|z^{\beta+e_j}\|_{H_J}}{1-\vartheta}
 \quad(j\in J).
\]
Since all coordinates of $\beta$ are positive, these bounds give, for
each $j\in J$,
\[
 |w_j|^{\beta_j}=\frac{|w^\beta|}{\prod_{l\in J,\,l\ne j}|w_l|^{\beta_l}}
 \ge\frac{1-\vartheta}{\|q\|_{H_J}}
 \prod_{l\in J,\,l\ne j}\Bigl(\frac{1-\vartheta}
 {\|q\|_{H_J}\,\|z^{\beta+e_l}\|_{H_J}}\Bigr)^{\beta_l}>0 .
\] Thus $H'$ is a compact subset of $(\mathbb C^*)^J$.

Let $\mathcal B\subset\mathcal P^S$ be generated by the monomials of $\mathcal B_J$, regarded
as polynomials on $\mathbb C^n$, and by $z^{\alpha^{(k)}}$, $k\notin J$. If
$z\in\widehat K^\mathcal B$, then $\operatorname{pr}_J(z)\in H'$, so that
$0<c\le|z_j|\le C$ for $j\in J$ with constants independent of $z$. For
$k\notin J$ the inequality $|z^{\alpha^{(k)}}|\le\|z^{\alpha^{(k)}}\|_K$
gives
\[
 |z_k|^{\alpha^{(k)}_k}\le
 \frac{\|z^{\alpha^{(k)}}\|_K}{\prod_{j\in J}|z_j|^{\alpha^{(k)}_j}}
 \le\frac{\|z^{\alpha^{(k)}}\|_K}{\prod_{j\in J}c^{\,\alpha^{(k)}_j}} .
\]
Hence $\widehat K^\mathcal B$ is bounded, and it is closed. The last assertion
follows from Theorem~\ref{thm:B}.

\label{cor:dimtwo}
Let $S\subset\mathbb R^2_+$ be compact and convex with $0\in S$. Then the
conditions of Theorem~\ref{thm:B} hold, and the Oka--Weil property holds,
for every compact $S$-convex set $K\subset\mathbb C^2$.

<!-- proof-12: body_md -->
We may assume $K\ne\varnothing$, so that $\Gamma=\Gamma_S$ has nonempty
interior by Corollary~\ref{cor:goodcover}. If $e_1,e_2\in\Gamma$, then
$\Gamma=\mathbb R^2_+$ and the classical Oka--Weil theorem applies. If
$e_1,e_2\notin\Gamma$, then every nonzero $\alpha\in M_\Gamma$ has two
positive coordinates, every good exponent is nonzero, $U_\Gamma\subset
(\mathbb C^*)^2$, and Corollaries~\ref{cor:goodcover} and~\ref{cor:torus} apply.

By symmetry it remains to consider the case $e_1\in\Gamma$,
$e_2\notin\Gamma$. Then $x_1>0$ for every $x\in\Gamma\setminus\{0\}$, and
we put
\[
 \sigma=\sup\Bigl\{\frac{x_2}{x_1}:\ x\in\Gamma\setminus\{0\}\Bigr\}
 \in(0,\infty].
\]
For $0\le\tau<\sigma$ there is $x\in\Gamma$ with $\sigma'=x_2/x_1>\tau$,
and
\[
 (1,\tau)=\Bigl(1-\frac\tau{\sigma'}\Bigr)e_1+\frac\tau{\sigma'}\cdot
 \frac x{x_1}
\]
is a nonnegative combination of $e_1$ and $x$, so $(1,\tau)\in\Gamma$. If $\Gamma$ is generated by $e_1$ and a lattice
point, then $\Gamma$ is rational polyhedral, and Theorems~\ref{thm:A}
and~\ref{thm:B} apply. Otherwise we verify (A1) and (A2) for $I=\{1\}$.
Condition (A1) holds because
\[
 me_1+e_2=m\Bigl(1,\frac1m\Bigr)\in\Gamma\qquad\text{when }\frac1m<\sigma .
\]
Let $\alpha=(a,b)\in M_\Gamma\setminus\{0\}$, so that $a\ge1$ and
$b/a\le\sigma$. If $b/a<\sigma$, choose $0<\delta<a$ with
$b/(a-\delta)<\sigma$; then
\[
 \alpha-\delta e_1=(a-\delta)\Bigl(1,\frac b{a-\delta}\Bigr)\in\Gamma .
\]
If $b/a=\sigma$, then $\Gamma$ lies in $\{x_2\le\sigma x_1\}\cap\mathbb R^2_+$,
which is the cone generated by $e_1$ and the lattice point $\alpha$; since
both lie in $\Gamma$, this is the excluded rational polyhedral case.
Theorem~\ref{thm:anchor}(b) completes the proof.

<!-- theorem-5: body_md -->
\label{thm:OS}
Let $\Gamma\subset\mathbb R^n_+$ be a convex cone and let $K\subset\mathbb C^n$ be a
compact set such that $\widehat K^{\mathcal B_0}$ is compact for some finitely
generated monomial algebra $\mathcal B_0\subset\mathcal P^\Gamma$. For $z\in\mathbb C^n$ the
following are equivalent.
\begin{enumerate}[label=\textup{(\roman*)}]
\item $z\notin\widehat K^\Gamma$.
\item There is a continuous family $(p_t)_{t\ge0}$ in $\mathcal P^\Gamma$ such
that $p_0(z)=0$, no $p_t$ has a zero on $K$, and the hypersurfaces
$\{p_t=0\}$ escape.
\end{enumerate}
In \textup{(ii)} one may take $p_t=p-(1+t)$ with $p\in\mathcal P^\Gamma$; in
particular, all $p_t$ lie in one space $\mathcal P^S_m$ when $\Gamma=\Gamma_S$.
The implication \textup{(i)$\Rightarrow$(ii)} holds for every compact
$K$.
<!-- end theorem-5 -->
<!-- proof-14: body_md -->
(i)$\Rightarrow$(ii). Choose $p\in\mathcal P^\Gamma$ with $p(z)=1>\|p\|_K$; this
is possible for every compact $K$. The polynomials
\[
 p_t=p-(1+t),\qquad t\ge0,
\]
satisfy $p_0(z)=0$ and $|p_t|\ge1+t-\|p\|_K>0$ on $K$, and $\{p_t=0\}$
escapes because $p$ is bounded on compact sets. Since $\mathcal P^S_m$ contains
the constants, $p_t\in\mathcal P^S_m$ whenever $p\in\mathcal P^S_m$.

(ii)$\Rightarrow$(i). By Remark~\ref{rem:nonconvex},
$H=\widehat K^\Gamma$ is compact and the Oka--Weil property holds on $H$.
Suppose that $z\in H$ and let
\[
 T=\bigl\{t\ge0:\ \{p_t=0\}\cap H\ne\varnothing\bigr\}.
\]
Then $0\in T$ because $z\in H$, and $T$ is bounded because the
hypersurfaces escape and $H$ is compact. Moreover $T$ is closed: if
$t_l\in T$, $t_l\to t$, and $y_l\in H$ satisfy $p_{t_l}(y_l)=0$, then a
subsequence of $(y_l)$ converges to some $y\in H$, and $p_t(y)=0$ by the
joint continuity of $(t,w)\mapsto p_t(w)$. Let $t_*=\max T$ and choose $y\in H$ with
$p_{t_*}(y)=0$. For $t>t_*$ the function $1/p_t$ is holomorphic in a
neighbourhood of $H$, so it is a uniform limit on $H$ of elements
$a_j\in\mathcal P^\Gamma$. Since $y\in H$ and $|a_j(y)|\le\|a_j\|_K$, we get
\[
 \frac1{|p_t(y)|}\le\sup_K\frac1{|p_t|}\qquad\text{for all }t>t_* .
\]
The right-hand side is bounded for $t\in(t_*,t_*+1]$, because $p_t(w)\ne0$
on the compact set $[t_*,t_*+1]\times K$, whereas the left-hand side tends to $+\infty$ as $t\downarrow t_*$,
because $p_t(y)\to p_{t_*}(y)=0$ and $p_t(y)\ne0$ for $t>t_*$ (indeed
$y\in H$ and $t\notin T$). This contradiction proves (i).

\label{rem:OSlocal}
(1) Without a compact finite stage the hull may be unbounded, and the
criterion must be formulated in ambient spaces. If
$\mathcal B_1\subset\mathcal B_2\subset\cdots$ are finitely generated monomial algebras

\]

<!-- theorem-6: body_md -->
\label{thm:rational}
Let $\Gamma\subset\mathbb R^n_+$ be a convex cone and let $K\subset\mathbb C^n$ be a
nonempty compact set with $\mathcal R_\Gamma(K)=K$. Assume that one of the
following conditions holds:
\begin{enumerate}[label=\textup{(\alph*)}]
\item $\Gamma$ is rational polyhedral;
\item $\mathcal R_{\mathcal B_0}(K)$ is compact for some finitely generated monomial
algebra $\mathcal B_0\subset\mathcal P^\Gamma$; this holds in particular when
$\widehat K^{\mathcal B_0}$ is compact;
\item $K\subset(\mathbb C^*)^n$.
\end{enumerate}
Then every function holomorphic in a neighbourhood of $K$ is a uniform
limit on $K$ of quotients $p/q$ with $p,q\in\mathcal P^\Gamma$ and $q$ without
zeros on $K$. Moreover, if $\Gamma$ has nonempty interior, then
\[
 \mathcal R_\Gamma(L)=\mathcal R(L)\qquad\text{for every compact }L\subset(\mathbb C^*)^n .
\]
<!-- end theorem-6 -->
<!-- proof-15: body_md -->
Case (a) is Theorem~\ref{thm:A}(b). In case (b) we repeat the proof of
(v)$\Rightarrow$(i) in Theorem~\ref{thm:B} with
\[
 K_j=\mathcal R_{\mathcal B_j}(K),\qquad j=1,2,\ldots
\]
These sets are closed subsets of the compact set $\mathcal R_{\mathcal B_0}(K)$, they
decrease, and $\bigcap_jK_j=\mathcal R_\Gamma(K)=K$. Moreover
$\mathcal R_{\mathcal B_j}(K_j)=K_j$, because $b(K_j)\subset b(K)$ for $b\in\mathcal B_j$. Let $f$ be holomorphic on an open set $\Omega\supset K$. By
Corollary~\ref{cor:goodcover}, $K$ lies in a finite union $V$ of good
charts $\{z^{\beta^k}\ne0\}$, and we may assume $\Omega\subset V$ and that
every $\mathcal B_j$ contains the monomials $z^{\beta^k}$ and $z^{\beta^k+e_i}$.
As in the proof of Theorem~\ref{thm:B}, for large $j$ we have
$K_j\subset\Omega\subset V\subset U_{\mathcal B_j}$, and
Theorem~\ref{thm:chart}(b), applied to $K_j$ and $\mathcal B_j$, shows that $f$
is a uniform limit on $K_j$, hence on $K$, of quotients $b/c$ with
$b,c\in\mathcal B_j\subset\mathcal P^\Gamma$ and $c\ne0$ on $K_j\supset K$.

For (c) and the last assertion, $\Gamma$ has nonempty interior by
Corollary~\ref{cor:goodcover} in case (c) and by assumption otherwise.
Choose a lattice point $\beta$ in the interior of $\Gamma$; its
coordinates are positive. For a polynomial $p$ and a large integer $k$,
all exponents of $z^{k\beta}p$ lie in $\Gamma$, since
\[
 \alpha+k\beta=k\Bigl(\beta+\frac\alpha k\Bigr).
\]
Let $L\subset(\mathbb C^*)^n$ be compact. A point with a vanishing coordinate is
not in $\mathcal R_\Gamma(L)$, since $z^\beta$ vanishes there and not on $L$. If
$w\in(\mathbb C^*)^n\setminus\mathcal R(L)$, a polynomial $p$ vanishes at $w$ and not on
$L$, and so does $z^{k\beta}p\in\mathcal P^\Gamma$. Hence
$\mathcal R_\Gamma(L)\subset\mathcal R(L)$, and the opposite inclusion is clear. In case
(c), $K=\mathcal R(K)$ is rationally convex, so a function holomorphic near $K$
is a uniform limit of rational functions $p/q$ with $q\ne0$ on $K$, and
\[
 \frac pq=\frac{z^{k\beta}p}{z^{k\beta}q},\qquad
 z^{k\beta}p,\ z^{k\beta}q\in\mathcal P^\Gamma\ \text{for large }k. \qedhere
\]
