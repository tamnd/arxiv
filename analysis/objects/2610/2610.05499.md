<!-- theorem-1: body_md -->
\label{thm:main}
Let $\mathscr F_s$ be given by \eqref{sGs} and let $\lambda$ be defined by \eqref{lambda},
let $(u_n)$ be a sequence in $L^2(\mathbb R^N)$
 weakly converging to a function $u$, and let $(s_n)$ be an infinitesimal sequence in $(0,1)$.
 Assume that the functions $u_n$ vanish outside a common compact set. 
 Then
 $$
  \liminf_{n\to+\infty} \mathscr F_{s_n}(u_n)\ge
  (1-\lambda)N \omega_N \liminf_{n\to+\infty}\|u_n\|_{L^2(\mathbb R^N)}^2.
 $$
<!-- end theorem-1 -->
<!-- remark-1: body_md -->
\label{rk:direct}\normalfont
When the functions $u_n$ vanish outside a common ball $B_{R_0}$, the inequality in
Theorem~\ref{thm:main} can also be obtained by integrating the kernel directly. If $x\in B_{R_0}$
and $2R_0<|z|<r_s-R_0$, then $x+z\in B_{r_s}\setminus B_{R_0}$, where $u_n$ vanishes, so that
\begin{eqnarray*}
\mathscr F_{s}(u_n)&\ge& 2s\int_{B_{R_0}}|u_n(x)|^2\int_{\{2R_0<|z|<r_s-R_0\}}\frac{dz}{|z|^{N+2s}}\,dx
\\
&&= N\omega_N\Big[(2R_0)^{-2s}-(r_s-R_0)^{-2s}\Big]\|u_n\|_{L^2}^2,
\end{eqnarray*}
and the bracket tends to $1-\lambda$. The argument below is different in nature: it is
qualitative, and it handles the lack of weak continuity of the energies by a compactness
property of monotone radial profiles (the rearrangements converge strongly), rather than by an
explicit computation with the kernel.
<!-- end remark-1 -->
<!-- theorem-2: body_md -->
\label{thm:pws}
 Let 
 $u$ be a Lipschitz function on $\mathbb R^N$  with compact support. 
 Then
 $$
 \limsup_{s\to0} \mathscr F_s(u)\le  (1-\lambda)N \omega_N \|u\|^2_{L^2(\mathbb R^N)}.
 $$
<!-- end theorem-2 -->
<!-- remark-2: body_md -->
\label{rk:dec_F}\normalfont
It is useful to point out the special role of the case $r_s\equiv+\infty$. Let $u$ be a function with support in a ball $B_{R_0}$, and let $r_s>R_0$. 
The domain of integration $\mathbb R^N\times\mathbb R^N$ in the definition of $\mathscr F_s^0$ can be split taking into account the disappearance of $u$ outside $B_{r_s}$. Thus, we can decompose $\mathscr F_s(u)$ as follows:
\begin{equation}\label{eq:dec_F}
\begin{split}
\mathscr F_s(u)=&\ \mathscr F_s^0(u)-\delta_s(u^2),\\
&\text{with}\ \ 
\delta_s(f)=2s\iint_{B_{R_0}\times (\mathbb R^N\setminus B_{r_s})}\frac{f(x)}{|x-y|^{N+2s}}\, dx\, dy.
\end{split}
\end{equation}
This decomposition will be applied appropriately in the sequel to recover the general case from the asymptotics of $\mathscr F_s^0$. For example, Proposition \ref{prop:corr} in Section~\ref{sec:convergence} implies that if $u\in C^1_c(\mathbb R^N)$ then $\delta_s(u^2)$ tends to $\lambdaN\omega_N \|u\|^2_{L^2}$ as $s\to0$. As a consequence, it is enough to prove Theorem \ref{thm:pws} for $\mathscr F_s^0$.
 
                                                                                                                                                                                                                                 
Note that Theorem~\ref{thm:main} cannot be derived from the case $r_s\equiv+\infty$ through this
decomposition, since the convergence of $\delta_s(u^2)$ fails if $u$ is replaced by a weakly
convergent sequence; this is one reason why a compactness mechanism, here provided by
rearrangements, is needed.
<!-- end remark-2 -->
<!-- corollary-1: body_md -->
\label{cor:Gamma}
Let $(s_n)$ be an infinitesimal sequence in $(0,1)$. Then the functionals $F_{s_n}$ $\Gamma$-converge to $F_0$ with respect to the weak $L^2(\Omega)$-convergence  as $n\to+\infty$.
<!-- end corollary-1 -->
<!-- head-1: body_md -->
($\Gamma$-$\liminf$) Let $(u_n)$ be a sequence in $L^2(\Omega)$
 weakly converging to a function $u$. Let $\tilde u_n$ and $\tilde u$ denote the extension of $u_n$ and $u$, respectively, to the whole $\mathbb R^N$ with zero value outside $\Omega$.
Since $\Omega$ is bounded, and $(\tilde u_n)$ weakly converges to $\tilde u$ in $L^2(\mathbb R^N)$, we can apply Theorem \ref{thm:main}, and deduce that
\begin{align*}
\liminf_{n\to+\infty} F_{s_n}(u_n)&=\liminf_{n\to+\infty} \mathscr F_{s_n}(\tilde u_n)\\
    &\ge (1-\lambda)N\omega_N\|\tilde u\|_{L^2(\mathbb R^N)}^2
    =(1-\lambda)N\omega_N\|u\|_{L^2(\Omega)}^2\,.
\end{align*}
\par\noindent
($\Gamma$-$\limsup$) Theorem \ref{thm:pws} guarantees the $\Gamma$-$\limsup$ inequality for $C^\infty_c(\Omega)$  functions. The general case follows by a standard approximation and diagonalization argument.
 \qed

<!-- proposition-1: body_md -->
\label{prop:pws_radial}
 Let $\phi\colon[0,+\infty)\to[0,M]$ be a $C^1$ decreasing function, with $\phi(r)=0$ if $r\ge R_0$ for some $R_0>0$. 
 Let $w(x)=\phi(|x|)$. Then
 \begin{equation}\label{eq:ptw_conv}
   \lim_{s\to 0} \mathscr F_s(w)=(1-\lambda)N\omega_N\|w\|_{L^2(\mathbb R^N)}^2.
 \end{equation}

<!-- proposition-2: body_md -->
\label{prop:corr}
For any $x\in \mathbb R^N$, define
$$
\eta_s(x)=2s\int_{\mathbb R^N\setminus B_{r_s}}\frac1{|x-y|^{N+2s}}\,dy.
$$
Then $\eta_s\to \lambdaN\omega_N$ uniformly on compact sets.
<!-- end proposition-2 -->
<!-- head-2: body_md -->
We set 
$$
I_s=2s\int_{\mathbb R^N\setminus B_{r_s}}\frac1{|y|^{N+2s}}\,dy\,.
$$
If $|x|\le R<r_s$ then we have
$$
|\eta_s(x)- I_s|\le
    2s\int_{\mathbb R^N\setminus B_{r_s}}\frac1{|y|^{N+2s}}\left|\left(\frac{|y|}{|x-y|}\right)^{N+2s}-1\right| dy\,.
$$
For every $y\in\mathbb R^N\setminus B_{r_s}$, from the inequality $\bigl||x-y|-|y|\bigr|\le |x|\le R$ we deduce that
$|x-y|\ge r_s-R$, and
$$
\left|\frac{|y|}{|x-y|}-1\right|\le\frac{|x|}{|x-y|}\le\frac{R}{r_s-R}.
$$
Therefore, to estimate the integral above, for $r_s$ sufficiently large we consider the function 
$t\mapsto t^{N+2s}$ in a neighbourhood of $t=1$, where the function is Lipschitz.
It follows that there exists a constant $L>0$ such that for every $s$
$$
\bigg|\Big(\frac{|y|}{|x-y|}\Big)^{N+2s}-1\bigg|\le L \left|\frac{|y|}{|x-y|}-1\right|
    \le L \frac{R}{r_s-R}.
$$
Hence,
$$
|\eta_s(x)- I_s|\le L \frac{R}{r_s-R} I_s\,.
$$
A direct computation shows that $I_s$ tends to $\lambda N\omega_N$ as $s\to 0$, thus concluding the proof. \qed

<!-- head-3: body_md -->
It is enough to consider the case $r_s\equiv +\infty$; i.e., $\mathscr F_s^0$. Indeed, by the decomposition in \eqref{eq:dec_F}, we have $\mathscr F_s(w)=\mathscr F_s^0(w)-\delta_s(w^2)$, and $\delta_s(w^2)\to\lambdaN\omega_N\|w\|_{L^2}^2$ as $s\to0$
by Proposition \ref{prop:corr}.



The proof will proceed by approximation of $w$ by radially symmetric step functions.
\par
Fix $h>0$ and let $n=\lfloor M/h \rfloor$. For every $i=0,\ldots,n$, let 
$$
B_i=\{x\in\mathbb R^N:\ w(x)>i h\}=\{x\in \mathbb R^N:\ \phi(|x|)>ih\}.
$$
Note that the radial symmetry of $w$ implies that $B_i$ is a ball centered at the origin. We denote by $r_i$ the radius. We can assume that $r_0=R_0$ and $B_n\neq\emptyset$.
Thus, since $\phi(r_i)=ih$ and $\phi$ is a $C^1$-function, we have
$0<r_{i+1}< r_i\le R_0$. 
\par
Define
\begin{equation}\label{eq:wh}
w^h=\sum_{i=0}^n h 1_{B_i}.
\end{equation}

In Step 1 we prove equality \eqref{eq:ptw_conv} for the characteristic function of a ball; then (in Step 2 and Step 3)
we consider the general case of simple functions. In Step~4 we show that $\mathscr F_s^0(w)$ can be approximated
by $\mathscr F_s^0(w^h)$ as $h\to0$, \emph{uniformly with respect to $s$}. This allows us to conclude. Indeed, 
from Step 3 and 4 we deduce that
\begin{align*}
\lim_{s\to0} \mathscr F_s^0(w)&=\lim_{s\to0}\lim_{h\to0}\mathscr F_s^0(w^h)=\lim_{h\to0}\lim_{s\to0} \mathscr F_s^0(w^h)\\
    &=N\omega_N \lim_{h\to0} \|w^h\|_{L^2(\mathbb R^N)}^2
    =N\omega_N\|w\|_{L^2(\mathbb R^N)}^2.
\end{align*}

<!-- lemma-1: body_md -->
\label{lem}
 Let $(w_n)_n$ be a weakly convergent sequence in $L^2(\mathbb R^N)$. Let $w$ be the limit function.
 Then, for every $\varepsilon>0$ there exists $M>0$ such that 
 $$
 \liminf_{n\to+\infty}\|w_n^M\|_{L^2(\mathbb R^N)}\ge \|w\|_{L^2(\mathbb R^N)}-\varepsilon,
 $$
where $w_n^M=(w_n\wedge M)\vee(-M)$.
<!-- end lemma-1 -->
<!-- head-4: body_md -->
We may assume that $w\not\equiv 0$. For every $M>0$ we have
$$
 \int_{\mathbb R^N} w_n^M w\, dx
    =\int_{\mathbb R^N} w_n w\,dx - \int_{\mathbb R^N} (w_n- w_n^M)w\,dx\,.
$$
We show that the second integral on the right-hand side can be made arbitrarily small for a suitable $M$ independent of $n$.
Indeed,
\begin{align*}
\left| \int_ {\mathbb R^N} (w_n-w_n^M) w\, dx\right|&\le \int_{\{|w_n|>M\}}(|w_n| -M) |w|\, dx\le \int_{\{|w_n|>M\}}|w_n w|\, dx\\
    &\le \| w_n\|_{L^2(\mathbb R^N)} \left( \int_{\{|w_n|>M\}}w^2\, dx\right)^{1/2}.
\end{align*}
The weak convergence of $(w_n)$ implies that $\|w_n\|_{L^2(\mathbb R^N)}\le C$ for a suitable $C$ independent of $n$. 
By Chebyshev's inequality $|\{|w_n|>M\}|\le (C/M)^2$. Thus, the integrability of $w^2$ implies that
for every $\sigma>0$ we can find a value $M$ such that
$\bigl(\int_{\{|w_n|>M\}}w^2\, dx\bigr)^{1/2}\le \sigma$ for every $n$. 
Then
$$
\liminf_{n\to+\infty} \int_{\mathbb R^N} w_n^M w\, dx\ge \liminf_{n\to+\infty} \int_{\mathbb R^N} w_n w\,dx - \sigma C
    =\|w\|_{L^2(\mathbb R^N)}^2- \sigma C.
$$
The left-hand side is majorized by $\|w\|_{L^2}\liminf_n\|w_n^M\|_{L^2}$. Thus
$$
\liminf_n\|w_n^M\|_{L^2}\ge \|w\|_{L^2} -\sigma C/\|w\|_{L^2}.
$$
We conclude by taking $\sigma=\varepsilon \|w\|_{L^2}/C$. \qed
