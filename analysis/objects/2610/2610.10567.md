<!-- theorem-1: body_md -->
\cite{KITTANEH} (\textbf{Kittaneh Numerical Radius Inequalities}) \label{KT}
	Let $T \in M_n(\mathbb{C})$. Then 
\begin{align}\label{KI}
	\frac{1}{2}\sqrt{\|T^*T+TT^*\|}\leq w_\mathbb{C}(T)\leq 	\frac{1}{\sqrt{2}}\sqrt{\|T^*T+TT^*\|}.	
\end{align}
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
Let $T \in M_n(\mathbb{O})$. Then 	
\begin{align*}
		\frac{\|T\|}{8}\leq w_\mathbb{O}(T)\leq \|T\|.
\end{align*}
<!-- end theorem-2 -->
<!-- proof-1: body_md -->
Clearly second inequality comes from Cauchy-Schwarz inequality. Let $x, y \in \mathbb{O}^n.$ By the generalized polarization identity, 
	\begin{align*}
		4 \langle y, Tx \rangle =& \langle (x+y), T(x+y) \rangle- \langle (x-y), T(x-y) \rangle\\
		&+\sum_{j=1}^{7}(\langle (x+ye_j), T(x+ye_j) \rangle- \langle (x-ye_j), T(x-ye_j))e_j.
	\end{align*}
Hence
\begin{align*}
	4| \langle y, Tx \rangle|&\leq w_\mathbb{O}(T)\left(\|x+y\|^2+\|x-y\|^2+\sum_{j=1}^{7}(\|x+ye_j\|^2+\|x-ye_j\|^2)\right)\\
	&=w_\mathbb{O}(T)\left(2\|x\|^2+2\|y\|^2+\sum_{j=1}^{7}(2\|x\|^2+2\|y\|^2)\right)\\
	&=16w_\mathbb{O}(T)(\|x\|^2+\|y\|^2).
\end{align*}
 Therefore 
 \begin{align*}
 	\|T\|=\sup_{x, y \in \mathbb{O}^n, \|x\|=\|y\|=1}|\langle y, Tx\rangle|\leq 8 w_\mathbb{O}(T).
 \end{align*}
<!-- end proof-1 -->
<!-- theorem-3: body_md -->
\label{QKT}
(\textbf{Octonionic Kittaneh Numerical Radius Inequalities})
Let $T \in M_n(\mathbb{O})$. Assume that 
\begin{align*}
	T=A_0+\sum_{j=1}^{7}e_jA_j \quad \text{ for some } A_0, \dots, A_7 \in M_n(\mathbb{O})
\end{align*}
satisfying the following conditions. 
	\begin{enumerate}[\upshape(i)]
		\item $A^*_j=A_j$ for all $0\leq j \leq 7$.
	\item $e_jA_j=A_je_j$ for all $1\leq j \leq 7$.
	\item $\sum_{j,k=1, j \neq k}^{7}\{e_jA_j, e_kA_k\}=0$.
		\end{enumerate}
Then 
\begin{align}\label{MI}
	\frac{\|T\|}{8}\leq	\frac{1}{4\sqrt{2}}\sqrt{\|T^*T+TT^*\|}\leq w_\mathbb{O}(T)\leq\frac{1}{\sqrt{2}}\sqrt{\|T^*T+TT^*\|}\leq \|T\|.	
\end{align}
<!-- end theorem-3 -->
<!-- proof-2: body_md -->
We have 
	\begin{align*}
		T=A_0+\sum_{j=1}^{7}e_jA_j, \quad T^*=A_0-\sum_{j=1}^{7}e_jA_j.
	\end{align*}
Therefore 
\begin{align*}
	T^*T+TT^*&=\left(A_0-\sum_{j=1}^{7}e_jA_j\right)\left(A_0+\sum_{j=1}^{7}e_jA_j\right)+\left(A_0+\sum_{j=1}^{7}e_jA_j\right)\left(A_0-\sum_{j=1}^{7}e_jA_j\right)\\
	&=2\left(\sum_{j=0}^{7}A^2_j\right)-2\sum_{j,k=1, j \neq k}^{7}\{e_jA_j, e_kA_k\}\\
	&=2\left(\sum_{j=0}^{7}A^2_j\right).
\end{align*}
For $h \in \mathbb{O}^n$ with $\|h\|=1$, we have
\begin{align*}
|\langle h, Th \rangle |^2&=\left|\left\langle h, \left(A_0+\sum_{j=1}^{7}e_jA_j\right)h \right\rangle \right|^2\\	
&=\left|\left\langle h, A_0h \right \rangle+ \sum_{j=1}^{7}e_j\left\langle h, A_jh \right \rangle\right|^2\\
&=\sum_{j=0}^{7}|\langle h, A_jh  \rangle|^2\\
&\leq \sum_{j=0}^{7}\|A_jh\|^2\\
&= \sum_{j=0}^{7}\langle h, A_j^2h \rangle \\
&=\left\langle h, \left(\sum_{j=0}^{7}A_j^2\right)h \right \rangle
\end{align*}
Since all $A_j$'s  are self-adjoint, we have
\begin{align*}
	w_\mathbb{O}(T)^2&= \sup_{h \in \mathbb{O}^n, \|h\|=1} |\langle h, Th \rangle |^2\\
	&\leq \sup_{h \in \mathbb{O}^n, \|h\|=1} \left\langle h, \left(\sum_{j=0}^{7}A_j^2\right)h \right\rangle\\
	&=\left\|\sum_{j=0}^{7}A_j^2\right\|\\
	&=\frac{1}{2}\|	T^*T+TT^*\|.
\end{align*}
On the other hand, for $h \in \mathbb{O}^n$ with $\|h\|=1$, using the Cauchy-Schwarz inequality, we also have 
\begin{align*}
|\langle h, Th \rangle |^2&=\left|\left\langle h, \left(A_0+\sum_{j=1}^{7}e_jA_j\right)h \right\rangle \right|^2\\	
&=\left|\left\langle h, A_0h \right \rangle+ \sum_{j=1}^{7}e_j\left\langle h, A_jh \right \rangle\right|^2\\
&=\sum_{j=0}^{7}|\langle h, A_jh  \rangle|^2\\
&\geq 	\frac{1}{8}\left(\sum_{j=0}^{7}|\langle h, A_jh  \rangle|\right)^2\\
&\geq 	\frac{1}{8}\left|\sum_{j=0}^{7}\langle h, (\pm A_j)h  \rangle\right|^2\\
&=\frac{1}{8}\left|\left\langle h, \sum_{j=0}^{7}(\pm A_jh) \right\rangle\right |^2.
\end{align*}
Since all $A_j$'s are self-adjoint, we have
\begin{align*}
	w_\mathbb{O}(T)^2&= \sup_{h \in \mathbb{O}^n, \|h\|=1} |\langle h, Th \rangle |^2\\
	&\geq \frac{1}{8}\sup_{h \in \mathbb{O}^n, \|h\|=1}\left|\left\langle h, \sum_{j=0}^{7}(\pm A_jh) \right\rangle\right  |^2\\

\end{align*}

\end{align*}

\end{align*}

<!-- problem-1: body_md -->
(\textbf{Octonionic Crouzeix Problem})
	Whether there is a universal constant $R_\mathbb{O}$ satisfying the following: For every $d \in \mathbb{N}$ and for every $A\in \mathbb{M}_{d}(\mathbb{O})$, we have 
	\begin{align*}
		\|p(A)\|\leq R_\mathbb{O} \sup\left\{|p(z)|: z \in W_\mathbb{O}(A)\right\},  \quad \forall p(z)\coloneqq a_0+a_1z+a_2z^2+\cdots +a_nz^n \in \mathbb{O}[z], n \in \mathbb{N}
	\end{align*}
	where $\mathbb{O}[z]$ is the set of all polynomials over $\mathbb{O}$. 
	We conjecture that $R_\mathbb{O}$ exists and $R_\mathbb{O}=8$.
