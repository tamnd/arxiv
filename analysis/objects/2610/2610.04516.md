<!-- head-1: body_md -->
For a given group $G$, decide for which $\mathbb{Z}G$-modules $M$ there exists a topological space $X$ such that
\begin{enumerate}
    \item $H_{\ast }(X;\mathbb{Z})\cong M$;
    \item the group of self-homotopy equivalences $\mathcal{E}(X)$ is isomorphic to $G$;
    \item the action of $G$ on $M$ is equivalent to the canonical action of $\mathcal{E}(X)$ on $H_{\ast }(X;\mathbb{Z})$.
\end{enumerate}
<!-- end head-1 -->
<!-- theorem-1: body_md -->
\cite[Theorem 3.1]{CC}\label{qzy}
    Let $G$ be an arbitrary group, let $V$ be a set and let $\rho: G\rightarrow \mathrm{Sym}(V)$ be a (not necessarily faithful) permutation representation. Then, there is a simple, undirected graph $\mathcal{G}$ such that

    \begin{enumerate}
        \item $V\subset V(\mathcal{G})$ and $V$ is invariant under the $\mathrm{aut}(\mathcal{G})$-action;
        \item $\mathrm{aut}(\mathcal{G}) \cong G$;
        \item the restriction map $G\cong \mathrm{aut}(\mathcal{G}) \rightarrow \mathrm{Sym}(V)$ is precisely $\rho$;
        \item there is a faithful representation $\bar{\rho}: G\cong \mathrm{aut}(\mathcal{G}) \rightarrow  \mathrm{Sym}( V(\mathcal{G}) \backslash V)$ such that the restriction map $G\cong \mathrm{aut}(\mathcal{G}) \rightarrow \mathrm{Sym}(V(\mathcal{G}))$ is $\rho \oplus \bar{\rho}$.
    \end{enumerate}
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{fi}
    Let $k\geq 2$ be an integer and $G$ an arbitrary group. Let $\left\{M_i\right\}_{i=1}^{k}$ be a finite collection of permutation $RG$-modules. Suppose that there exists an integer $n$ satisfying
    $$n\geq 3k+5, \ \  \rho(R) >20n^3-33n^2+8n+4. $$
    Then, there exists an $(8n^2-6n+1)$-connected finite-dimensional CW-complex $X$ such that
    \begin{enumerate}
        \item $\mathcal{E}(X_{R})\cong G$;
        \item $H_{k(8n^2-2n-2)+i+1}(X_{R};\mathbb{Z})\cong M_i$ for $i=1,\dots, k$;
        \item the action of $\mathcal{E}(X_{R})$ on $H_{k(8n^2-2n-2)+i+1}(X_{R};\mathbb{Z})$ is equivalent to the action of $G$ on $M_i$ for $i=1,\dots, k$.
    \end{enumerate}
    Here $X_{R}$ denotes the $R$-localization of $X$. In particular, $X$ can be chosen to be a finite CW-complex when $G$ is a finite group and $\left\{M_i\right\}_{i=1}^{k}$ is a collection of finitely generated $RG$-modules.
<!-- end theorem-2 -->
<!-- remark-1: body_md -->
Theorem \ref{fi} deals with the case $k\geq 2$. The case $k=1$ can be settled by applying Theorem \ref{fi} to the arbitrary group $G$ and the collection of permutation $RG$-modules $\left\{M_i\right\}_{i=1}^{2}=\{M_1,R\}$ with the trivial $RG$-module $R$.
<!-- end remark-1 -->
<!-- corollary-1: body_md -->
\label{co1}
    For every group $G$ and positive integer $N$, there exists a positive integer $P_{G,N}$ such that, for every subring $R\subset {\mathbb Q}$ with $\rho(R)>P_{G,N}$, there exists an $N$-connected finite-dimensional CW-complex $X$ satisfying $\mathcal{E}(X_{R})\cong G$, where $X_{R}$ is the $R$-localization of $X$. In particular, $X$ can be chosen to be a finite CW-complex when $G$ is a finite group.
<!-- end corollary-1 -->
<!-- definition-1: body_md -->
A free dgl over $R$ is called {\it $r$-mild} if it is generated as a Lie algebra by the range of dimensions $r$ through $r\rho(R)-1$.
<!-- end definition-1 -->
<!-- theorem-3: body_md -->
\label{HEP}
    Let $(\mathbb{L}(W),d)\rightarrow (\mathbb{L}(W\oplus U),d)$ be an inclusion of $r$-mild dgl's. Let $f:(\mathbb{L}(W\oplus U),d)\rightarrow (L,d^\prime)$ be a morphism between $r$-mild dgl's, and let $F:I\mathbb{L}(W)\rightarrow L$ be a homotopy between $f|_{\mathbb{L}(W)}$ and some other morphism $g$. Then there is an extension $\widetilde{F}:I\mathbb{L}(W\oplus U)\rightarrow L$ of $F$ to a homotopy between $f$ and an extension $\widetilde{g}$ of $g$.

<!-- lemma-1: body_md -->
\label{ho}
    Let $(\mathbb{L}(W_{\leq k}),d)$ and $(\mathbb{L}(W^\prime),d^\prime)$ be $r$-mild dgl's. Let
    $$\varphi_1,\varphi_2: (\mathbb{L}(W_{\leq k}),d) \rightarrow (\mathbb{L}(W^\prime),d^\prime)$$
    be two morphisms satisfying
    \begin{align*}
    \varphi_2(x)&=\varphi_1(x), \ \ x\in W_{<k},  \\
    \varphi_2(z)&=\varphi_1(z)+ d^\prime m_z, \ \ z\in W_{k},
    \end{align*}
    for some $m_z\in \mathbb{L}(W^\prime)$. Then $\varphi_1$ and $\varphi_2$ are homotopic.

<!-- lemma-2: body_md -->
\label{lm1}
    The graded $R$-module $\mathcal{L}^{\mathcal{G},0}$ is trivial in degree $|b|+3n$.
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
For $n\geq 3k+5\geq 11$, we have
    $$3n<|x_1|=|x_2|<|y|<|z|<|t|<|a_{v}|<|b|=|b_{(v,w)}|.$$
    It suffices to show the following equation
    \begin{equation}\label{eq0}
        k_1|x_1|+k_2|y|+k_3|z|+k_4|t|=|b|+3n
    \end{equation}
    with variables $k_1,k_2,k_3,k_4$ admits no non-negative integer solution.

    Suppose the equation has a non-negative integer solution $(k_1,k_2,k_3,k_4)$. By direct computation, we have
    $$k_1\leq 5n-4, \ \ k_2\leq 5n-6, \ \ k_3\leq 5n-6, \ \ k_4\leq \frac{5n}{2}-2.$$
    Note that if $(k_1,k_2,k_3,k_4)$ is a solution, then $(k_1+2,k_2-1,k_3+1,k_4-1)$ is also a solution. By induction, we can reach a solution with the second or fourth entry being zero. Hence, we may suppose $k_2$ or $k_4$ equals zero without loss of generality.

    {\bf Case 1}: $k_2=0$.

    After substituting $k_2$ and the degrees of the generators, Equation {(\ref{eq0})} becomes
    $$(2n-1)(4n-1)k_1+(2n-1)(4n+1)k_3+(2(2n-1)(4n-1)+1)k_4=40n^3-58n^2+10n+6.$$
    Modulo both sides by $2n-1$, we obtain the congruence
    $$k_4\equiv n+1 \ \ (\mathrm{mod} \ 2n-1).$$
    Since $k_4\leq \frac{5n}{2}-2$, it follows that $k_4=n+1$. Then we have
    \begin{equation}\label{eq1}
        (4n-1)k_1+(4n+1)k_3=12n^2-25n-3.
    \end{equation}
    Modulo both sides by $4n-1$, we get
    $$2k_3\equiv 2n-9 \ \ (\mathrm{mod} \ 4n-1).$$
    Since $k_3\leq 5n-6$ and $n\geq 11$, it follows that $k_3=3n-5$. By Equation (\ref{eq1}), we have $k_1=-2$, a contradiction.

    {\bf Case 2}: $k_4=0$.

    After substituting $k_4$ and the degrees of the generators, Equation (\ref{eq0}) becomes
    $$(2n-1)(4n-1)k_1+((2n-1)(4n+1)-1)k_2+(2n-1)(4n+1)k_3=40n^3-58n^2+10n+6.$$
    Modulo both sides by $2n-1$, we obtain the congruence
    $$-k_2\equiv n+1 \ \ (\mathrm{mod} \ 2n-1).$$
    Since $k_2\leq 5n-6$ and $n\geq 11$, it follows that $k_2=n-2$ or $3n-3$.

    {\bf Subcase 2.1}: $k_2=n-2$.

    After substitution, Equation (\ref{eq0}) becomes
    \begin{equation}\label{eq2}
        (4n-1)k_1+(4n+1)k_3=16n^2-12n-2.
    \end{equation}
    Modulo both sides by $4n-1$, we get
    $$2k_3\equiv -4 \ \ (\mathrm{mod} \  4n-1).$$
    Since $k_3\leq 5n-6$ and $n\geq 11$, it follows that $k_3=4n-3$. By Equation (\ref{eq2}), we have $k_1=-1$, a contradiction.

    {\bf Subcase 2.2}: $k_2=3n-3$.

    After substitution, Equation (\ref{eq0}) becomes
    \begin{equation}\label{eq3}
        (4n-1)k_1+(4n+1)k_3=8n^2-10n.
    \end{equation}
    Modulo both sides by $4n-1$, we get
    $$2k_3\equiv -2 \ \ (\mathrm{mod} \  4n-1).$$
    Since $k_3\leq 5n-6$ and $n\geq 11$, it follows that $k_3=4n-2$. By Equation (\ref{eq3}), we have $k_1=-2(n+1)$, a contradiction.

    This completes the proof of Lemma \ref{lm1}.

$$(\mathrm{ad} a_{v^\prime})^3(\mathrm{ad} x_1)^{2n-1} (\mathrm{ad} y)^{2n-3(k-h_{v^\prime})}(\mathrm{ad} z)^{n-5-3h_{v^\prime}}(x_2) $$
and
$$(\mathrm{ad} a_{v^\prime})(\mathrm{ad} a_{w^\prime})(\mathrm{ad} x_1)^{2n-3}(\mathrm{ad} y)^{2n+1-(2k-h_{v^\prime}-h_{w^\prime})}(\mathrm{ad} z)^{n-6-(h_{v^\prime}+h_{w^\prime})}([x_2,t]) $$
for $(v^\prime,w^\prime)\in E(\mathcal{G})$ in (\ref{ab1}) and (\ref{ab3}), we conclude that
\begin{equation}\label{xi1}
    \eta_{(v,w)(p,q)}=\delta_{vp}^3=\delta_{vp}\delta_{wq}\neq 0,\ \ \eta_{(v,w)(p^\prime,q^\prime)}=0 \ \ \mathrm{for} \ (p^\prime,q^\prime)\neq (p,q).
\end{equation}
In particular, $\eta_{(v,w)(p,q)}\neq 0$ implies $(p,q)\in E(\mathcal{G})$. Since $\mathcal{G}$ is a graph (which implies that if $(v,w)\in E(\mathcal{G})$ then $(w,v)\in E(\mathcal{G})$ too), by similar arguments for the edge $(w,v)$ we deduce that
\begin{equation}\label{xi2}
    \delta_{wq}^3=\delta_{wq}\delta_{vp}.
\end{equation}
Combining (\ref{xi1}) and (\ref{xi2}) gives 
$$\eta_{(v,w)(p,q)}=\delta_{vp}=\delta_{wq}=1,$$
which proves (\ref{xi}).

By virtue of the uniqueness of $p$, we can define a morphism of graphs
\begin{equation*}
    \begin{split}
        \sigma : \mathcal{G} &\longrightarrow \mathcal{G}  \\
        v \  &\longmapsto \  p \\
        (v,w) &\longmapsto (p,q).
    \end{split}
\end{equation*}
    Our final step is to show that the above $\sigma$ is an automorphism of graphs. As $f$ is a homotopy equivalence, it induces an isomorphism on homology in degree $|a_v|$, which implies that $\sigma$ is a permutation of $V(\mathcal{G})$. As a result, $f$ is an isomorphism on $\mathcal{L}^{\mathcal{G}}_{<|b|}$. For any $(v,w)\in E(\mathcal{G})$, we claim that $(\sigma^{-1}(v),\sigma^{-1}(w))\in E(\mathcal{G})$. Indeed, since $db_{(v,w)}$ is a boundary of degree $|b|-1$, it follows that
    \begin{equation*}
    \begin{split}
    f&^{-1}(db_{(v,w)})=(\mathrm{ad} a_{\sigma^{-1}(v)})^3(\mathrm{ad} x_1)^{2n-1} (\mathrm{ad} y)^{2n-3(k-h_v)}(\mathrm{ad} z)^{n-5-3h_v}(x_2) \\
    &+(\mathrm{ad} a_{\sigma^{-1}(v)})(\mathrm{ad} a_{\sigma^{-1}(w)})(\mathrm{ad} x_1)^{2n-3} \\
    &\ \ \ \ \ \ \ \  \circ(\mathrm{ad} y)^{2n+1-(2k-h_v-h_w)}(\mathrm{ad} z)^{n-6-(h_v+h_w)}([x_2,t])
    \end{split}
    \end{equation*}
is also a boundary. However, the last term above only appears in the differential of $b_{(\sigma^{-1}(v),\sigma^{-1}(w))}$, and the linear independence by Lemma \ref{lm3} forces the presence of the generator $b_{(\sigma^{-1}(v),\sigma^{-1}(w))}$. Therefore, we can deduce that $(\sigma^{-1}(v),\sigma^{-1}(w))\in E(\mathcal{G})$. It follows from the definition that $\sigma\in \mathrm{aut}(\mathcal{G})$. This completes the proof of Lemma \ref{lm4}.

\label{hx2}
    With notation as in Lemma \ref{hx1}. For every morphism $f$ with $[f]\in \mathcal{E}(\mathcal{L}^{\mathcal{G}})$, there exists $\sigma\in \mathrm{aut}(\mathcal{G})$ such that $f\simeq f_{\sigma}$.
<!-- end lemma-7 -->
<!-- proof-6: body_md -->
In Lemma \ref{lm4}, we have concluded that
\begin{align*}
    f|_\mathcal{L}&=f_{\sigma}|_\mathcal{L}, \\
    f(b)&=f_{\sigma}(b)+B, \\
    f(b_{(v,w)})&=f_{\sigma}(b_{(v,w)})+B_{(v,w)}, \ \  (v,w)\in E(\mathcal{G}),
\end{align*}
where $B$ and $B_{(v,w)}$'s are boundaries in $ \mathcal{L}^{\mathcal{G}}$. By Lemma \ref{ho}, we have
$$f|_{\mathcal{L}^{\mathcal{G},0}}\simeq g_{\sigma}= f_{\sigma}|_{\mathcal{L}^{\mathcal{G},0}},$$
By Theorem \ref{HEP}, we deduce that $f$ is homotopic to a morphism, denoted by $f_0$, whose restriction on $\mathcal{L}^{\mathcal{G},0}$ is $g_{\sigma}$.

We next compare the restriction of $f_0$ on $\mathcal{L}^{\mathcal{G},1}$ and $g_{\sigma,1}=f_{\sigma}|_{\mathcal{L}^{\mathcal{G},1}}$. Recall that $\{\phi_{1,\alpha}\}_{\alpha\in I_1}$ is a free basis for $\Phi_{|b|+1}$. For $\alpha\in I_1$, we have
$$d\left(f_0(\phi_{1,\alpha})-g_{\sigma,1}(\phi_{1,\alpha})\right)=f_0(d\phi_{1,\alpha})-g_{\sigma,1}(d\phi_{1,\alpha})=0,$$
since $d\phi_{1,\alpha}\in \mathcal{L}^{\mathcal{G},0} $, and $f_0,g_{\sigma,1}$ coincide on $\mathcal{L}^{\mathcal{G},0}$. By Lemma \ref{lm2}\,(4), it follows that $f_0(\phi_{1,\alpha})-g_{\sigma,1}(\phi_{1,\alpha})$ is a boundary. Applying Lemma \ref{ho} and Theorem \ref{HEP} again yields that
$$f_0|_{\mathcal{L}^{\mathcal{G},1}}\simeq g_{\sigma,1} =f_{\sigma} |_{\mathcal{L}^{\mathcal{G},1}},$$
and $f_0$ is homotopic to a morphism, denoted by $f_1$, whose restriction on $\mathcal{L}^{\mathcal{G},1}$ is $g_{\sigma,1}$.

Iterating the foregoing arguments, we obtain morphisms $f_1,f_2,\dots,f_{3n}$ such that
$$f_s|_{\mathcal{L}^{\mathcal{G},s}}=g_{\sigma,s}$$
for $s=1,2,\dots,3n$, and
$$f\simeq f_0\simeq f_1\simeq \dots \simeq f_{3n}. $$
Note that $f_{3n}$ and $f_{\sigma}$ coincide on $\mathcal{L}^{\mathcal{G},3n}=\mathcal{L}^{\mathcal{G}}$. Thus, we finally conclude that $f_{3n}= f_{\sigma}$, and hence $f\simeq f_{\sigma}$.

It is easy to verify that $\theta_{\mathcal{G}}$ is a group morphism, and Lemma \ref{hx2} implies $\theta_{\mathcal{G}}$ is surjective.

    To show that $\theta_{\mathcal{G}}$ is injective, suppose that $[f_{\sigma}]=[f_{\sigma^\prime}] $. Thus, we obtain  $H(f_{\sigma})=H(f_{\sigma^\prime})$. For $i=0,1,\dots, k$, we have
    $$H_{(k-i)|y|+i|z|}(\mathcal{L}^{\mathcal{G}})=\langle a_{v}  |\   v\in V_i \  \rangle \oplus \mathbb{L}^{k-i,i}(y,z).$$
    It follows that
    $$a_{\sigma(v)}=H(f_{\sigma})(a_v)=H(f_{\sigma^\prime})(a_v) =a_{\sigma^\prime(v)} $$
    for $v\in V_i$ and $i=0,1,\dots,k$. Since $V(\mathcal{G})= \cup_{i=0}^kV_i $, we conclude that $\sigma(v)=\sigma^\prime(v)$ for $v\in V(\mathcal{G})$, which implies $\sigma=\sigma^\prime$. This completes the proof of Lemma \ref{lm5}.
