<!-- theorem-1: body_md -->
Let $C$ be an Arthur orbit. For every $M$-orbit $D\subset V$,
$$
 C\leq D,\qquad p(C)\leq p(D)
 \quad\Longrightarrow\quad C=D.
$$
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
For every Arthur orbit $C$,
$$
 U_C\cap\mathcal F^{-1}U_{p(C)}=\mathbb C T_C.
$$
The distribution $T_C$ is the unique element of this intersection
satisfying
$$
 T_C\equiv S_C\pmod{U_{>C}}.
$$
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
Every orbit in $\mathscr C_{\mathrm{rel}}$ is Arthur, and the
saturation map
$$
 C\longmapstoG^\vee C
$$
is injective on $\mathscr C_{\mathrm{rel}}$. It is also surjective
onto the ambient orbits in the relative special piece in
classical types, and for exceptional types when $\mathcal O^\vee$
is even.
<!-- end theorem-3 -->
<!-- definition-1: body_md -->
\label{def:arthur-pair}
An ordered pair $(C_1,C_2)$ of $H$-orbits in $V$ is an
\emph{Arthur pair at $\lambda$} if
$(C_1,C_2)=(C_\psi,C_{\psi^t})$ for an Arthur parameter
$\psi$ with $\lambda_\psi=\lambda$.
An orbit occurring as the first member is an \emph{Arthur orbit}.
<!-- end definition-1 -->

<!-- lemma-1: body_md -->
\label{lem:strict-saturation}
If $C<D$ are $H$-orbits in $V=\mathfrak j(2)$, then
$\mathcal O_C^\vee<\mathcal O_D^\vee$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Since $J^\circ$ is reductive, complete reducibility gives a
$J$-stable complement $\mathfrak g^\vee=\mathfrak j\oplus U$
\cite[Corollary~22.43]{MilneAG}. For $x^\vee\in V$, we have
$[U,x^\vee]\subseteq U$ and
$[\mathfrak j(r),x^\vee]\subseteq\mathfrak j(r+2)$. Hence
\begin{equation}
 [\mathfrak g^\vee,x^\vee]\cap V
 =[\mathfrak j,x^\vee]\cap\mathfrak j(2)
 =[\mathfrak j(0),x^\vee]
 =[\operatorname{Lie}(H),x^\vee].
\end{equation}
By \cite[\S2.3, Lemma]{Vinberg1976}, each $H^\circ$-orbit is closed
in its ambient orbit intersected with $V$. The same holds for an
$H$-orbit, a finite union of $H^\circ$-orbits. Thus, if
$\mathcal O_C^\vee=\mathcal O_D^\vee=\mathcal O$, then
\begin{equation}
 C\subseteq\overline D\cap\mathcal O\cap V=D,
\end{equation}
contrary to $C\ne D$. Since saturation preserves closure inclusion,
$\mathcal O_C^\vee<\mathcal O_D^\vee$.

<!-- theorem-4: body_md -->
\label{thm:conormal-norm}
For every $(x^\vee,y^\vee)\in\Lambda$,
\begin{equation}\label{eq:conormal-norm-bound}
 \nu(x^\vee)+\nu(y^\vee)\leq\|h^\vee\|^2.
\end{equation}
Consequently, for every $H$-orbit $C\subset V$,
\begin{equation}\label{eq:orbit-conormal-bound}
 \nu(C)+\nu(p(C))\leq\|h^\vee\|^2.
\end{equation}
<!-- end theorem-4 -->
<!-- proof-3: body_md -->
Choose a homogeneous triple $(x^\vee,h_x^\vee,f_x^\vee)$ in
$\mathfrak j$, using \cite[Theorem~1]{Vinberg1979}; take the zero
triple when $x^\vee=0$. Set $z^\vee=h^\vee-h_x^\vee$.
Homogeneity shows that $z^\vee$ centralizes the triple, so
\begin{equation}
 B(h_x^\vee,z^\vee)
 =B([x^\vee,f_x^\vee],z^\vee)
 =B(x^\vee,[f_x^\vee,z^\vee])=0.
\end{equation}
The commuting semisimple elements $h^\vee$ and $h_x^\vee$ lie in
$[\mathfrak g^\vee,\mathfrak g^\vee]$ and have integral adjoint eigenvalues. After simultaneous
conjugation into a Cartan subalgebra, they and $z^\vee$ therefore
lie in its real coroot span.

Decompose $y^\vee=\sum_m y_m^\vee$ into $h_x^\vee$-weight vectors.
Since $[x^\vee,y^\vee]=0$, each nonzero $y_m^\vee$ is a
highest-weight vector for the triple, and hence $m\geq0$.
As $[h^\vee,y^\vee]=-2y^\vee$, we obtain
\begin{equation}
 [z^\vee,y_m^\vee]=(-2-m)y_m^\vee,
 \qquad
 y^\vee\in\bigoplus_{r\geq1}\mathfrak g^\vee(-z^\vee/2;r).
\end{equation}
Applying \eqref{eq:GS-bound} yields
\begin{equation}
 \nu(y^\vee)\leq\|z^\vee\|^2
 =\|h^\vee\|^2-\|h_x^\vee\|^2
 =\|h^\vee\|^2-\nu(x^\vee).
\end{equation}
The same conclusion is immediate if $y^\vee=0$.
Finally, choose a conormal pair in $C\times C^*$ and use
\eqref{eq:partner-norm} to obtain \eqref{eq:orbit-conormal-bound}.

<!-- theorem-5: body_md -->
\label{thm:arthur-rigidity}
If $C\subset V$ is an Arthur orbit, then
\begin{equation}\label{eq:arthur-norm-equality}
 \nu(C)+\nu(p(C))=\|h^\vee\|^2.
\end{equation}
For every $H$-orbit $D\subset V$,
\begin{equation}\label{eq:arthur-rigidity}
 C\leq D,\qquad p(C)\leq p(D)
 \quad\Longrightarrow\quad C=D.
\end{equation}
<!-- end theorem-5 -->
<!-- proof-4: body_md -->
Choose commuting triples defining $C=He_1^\vee$, with
$h_1^\vee+h_2^\vee=h^\vee$. Lemma~\ref{lem:cunningham} gives
$p(C)=He_2^\vee$. Since the triples commute,
\begin{equation}
 B(h_1^\vee,h_2^\vee)
 =B(e_1^\vee,[f_1^\vee,h_2^\vee])=0,
 \qquad
 \nu(C)+\nu(p(C))
 =\|h_1^\vee\|^2+\|h_2^\vee\|^2=\|h^\vee\|^2.
\end{equation}
If $C<D$ and $p(C)\leq p(D)$, strict monotonicity
\eqref{eq:graded-strict-norm} and Theorem~\ref{thm:conormal-norm} give
\begin{equation}
 \|h^\vee\|^2
 =\nu(C)+\nu(p(C))
 <\nu(D)+\nu(p(D))
 \leq\|h^\vee\|^2,
\end{equation}
a contradiction.

\label{prop:reconstruction-ambiguity}

to the second identity in \eqref{eq:ambiguity-support}. The converse is proved in the same way.

<!-- proposition-4: body_md -->
\label{prop:unipotent-llc-hypotheses}
\begin{enumerate}
\item Lusztig's unipotent correspondence satisfies the inner-form,

<!-- proof-15: body_md -->
For (1), the inner-form parametrization is part of Lusztig's
correspondence \cite{Lus95,Lus02}. The multiplicity formula follows
from his Hecke-algebra realization and
\cite[\S\S10.4--10.8]{LusGHAII}, while \cite{LusGHAIII} supplies the
induction and temperedness results. Their formulations for reductive
groups, including the Levi subgroups needed here, are given in
\cite[Theorem~1]{SolLLC} and \cite[Theorem~5.4(b)]{Sol}.

For (2), we use Waldspurger's tempered normalization of Lusztig's
correspondence \cite[Introduction]{WaldI}. This is consistent with
Antor--Okada's normalization in \cite[\S4.9]{AO}.
The stability and twisted-transfer
identities recalled in \cite[\S2.1]{WaldI}, together with the
endoscopic identities proved in \cite[\S1.4]{WaldII}, characterize

\label{thm:classical-test}
The orbit $H e_1^\vee$ is Arthur if and only if, for every $d$,
the eigenvalues (with multiplicity) of  $z_d$ form a union of the strings
\begin{equation}\label{eq:dynkin-strings}
 b-1,b-3,\ldots,1-b\qquad(b\geq1),
\end{equation}
with multiplicities $n_{d,b}\in\mathbb Z_{\geq0}$ satisfying:
\begin{enumerate}
\item for a general linear factor, there is no additional restriction;
\item for an orthogonal factor, $n_{d,b}$ is even whenever $b$ is even;
\item for a symplectic factor, $n_{d,b}$ is even whenever $b$ is odd.
\end{enumerate}
For $\mathfrak g^\vee=\mathfrak{sl}(W)$, the general linear string test applies
to each $U_d$ separately. In particular, it requires
$\operatorname{tr}(z_d)=0$ for every $d$, beyond the weighted
trace condition in \eqref{eq:classical-centralizers}.

Equivalently, let $\mu_d(j)$ denote the multiplicity of $j$ in
$z_d$. Its eigenvalues are integers, since $h^\vee$ and $h_1^\vee$
commute and have integral weights on $W$. The test is
$\mu_d(j)=\mu_d(-j)$, together with nonnegativity and the above
parity conditions for
\begin{equation}\label{eq:string-multiplicities}
 n_{d,b}=\mu_d(b-1)-\mu_d(b+1)\qquad(b\geq1).
\end{equation}
These multiplicities are uniquely determined.
<!-- end theorem-11 -->
<!-- proof-20: body_md -->
By Proposition~\ref{thm:centralizer-criterion}, it suffices to test whether
$z^\vee$ is a Dynkin element in $\mathfrak c_1$.
The classical classification of $\mathfrak{sl}_2$-modules with
invariant forms gives exactly the stated weight strings and parity
conditions on each $U_d$; see \cite[Ch.~5]{CM}.
Conversely, these conditions construct a triple whose neutral element
has the spectrum of $z_d$. Semisimple elements with this spectrum
are conjugate under the general linear or full isometry group, so
we may take the neutral element to be $z_d$ itself.
The resulting triples assemble in $\mathfrak c_1$; in the special
linear case, each has trace zero on every $U_d$.

Finally, for $j\geq0$, the weight strings give
$$
 \mu_d(j)=\sum_{\ell\geq0}n_{d,j+1+2\ell}.
$$
Taking successive differences gives
\eqref{eq:string-multiplicities}; conversely, telescoping recovers
$\mu_d(j)$, and symmetry recovers the negative weights.
This also proves uniqueness of the multiplicities.

<!-- proposition-7: body_md -->
\label{prop:orthogonal-refinement}
Let $(n_{d,b})$ be an admissible array for an orthogonal $W$.
\begin{enumerate}
\item Its $\mathrm O(W)$-class splits into two $\mathrm{SO}(W)$-classes
if and only if
\begin{equation}\label{eq:orthogonal-splitting}
 n_{d,b}=0\quad\text{for all odd }d,b.
\end{equation}
Otherwise it gives one $\mathrm{SO}(W)$-class.
\item If the diagonal partition is very even, exactly one of the
array's two classes lies over the prescribed diagonal class.
Otherwise all its classes lie over that diagonal class.
\item Factor exchange sends the classes for $n$ to those for $n^t$.
If $n=n^t$ and \eqref{eq:orthogonal-splitting} holds, it preserves
the two classes individually when
$$
 (-1)^{\sum_{d\equiv2\!\!\pmod4}n_{d,d}}=1,
$$
and interchanges them when this sign is $-1$. For a very-even
diagonal, a self-transpose array gives a fixed point of the partner operation.
\end{enumerate}
<!-- end proposition-7 -->
<!-- proof-22: body_md -->
An orthogonal class splits precisely when its centralizer contains
no element of determinant $-1$. By
\eqref{eq:two-factor-classification}, a centralizer element has
$$
 \det(g|_W)=\prod_{d,b}\det(g_{d,b})^{db}.
$$
This can be $-1$ exactly when an odd--odd block occurs: its symmetric
multiplicity space admits a reflection. This proves (1).

For (2), the diagonal orthogonal class splits precisely when its
partition is very even; see \cite[Ch.~5]{CM}.
In that case, \eqref{eq:diagonal-multiplicities} forces $d+b$ odd
on every nonzero block, so the array also splits. A determinant-$-1$
conjugation exchanges both pairs of classes, leaving exactly one
array class over each diagonal class. Otherwise the diagonal has
one special orthogonal class, over which all array classes lie.

For (3), factor exchange always maps the classes for $n$ bijectively
to those for $n^t$. Suppose $n=n^t$ and the array splits. Choose
isometries between paired multiplicity spaces and implement exchange
by tensor flips. Swapping the blocks with indices $d<b$ contributes
$(-1)^{db n_{d,b}}$; on $L_d\otimes L_d$, the flip is $-1$ on
$\bigwedge^2 L_d$. Hence the resulting isometry $q$ has
$$
 \det q
 =(-1)^{\sum_{d<b}db n_{d,b}+\sum_d n_{d,d}d(d-1)/2}
 =(-1)^{\sum_{d\equiv2\!\!\pmod4}n_{d,d}},
$$
since every occupied block has $db$ even. This sign is independent
of the choices, as centralizer elements have determinant one;
it determines whether the two classes are fixed or exchanged.
For a very-even diagonal there are no diagonal blocks, so the
selected class is fixed.

<!-- theorem-16: body_md -->
\label{thm:weak-packet-decomposition}
Under the hypotheses of this section, for every neutral grading of
classical or exceptional type,
\begin{equation}\label{eq:weak-packet-decomposition}
 \Pi_{\mathcal O^\vee}^{\mathrm{weak}}
 =\bigcup_{C\in\mathscr C_{\mathrm{rel}}}\Pi_{p(C)}^\mathrm{mic},
\end{equation}
and every packet in this union is a microlocal Arthur packet.
The equality holds on each pure inner twist. With packets taken
over all pure inner twists, a microlocal Arthur packet satisfies
$$
 \Pi_B^\mathrm{mic}\subseteq\Pi_{\mathcal O^\vee}^{\mathrm{weak}}
 \quad\Longleftrightarrow\quad p(B)\in\mathscr C_{\mathrm{rel}}.
$$

<!-- proof-30: body_md -->
If $C\in\mathscr C_{\mathrm{rel}}$ and $\pi_a\in\Pi_{p(C)}^\mathrm{mic}$,
Fourier symmetry gives $m_C(P_{f(a)})>0$, hence
$C\leq C_{f(a)}$. Since
$G^\vee C\leqG^\vee C_{f(a)}\leq\mathcal O^\vee$, order reversal under
$d$ gives $d(G^\vee C_{f(a)})=d(\mathcal O^\vee)$. Thus
\eqref{eq:weak-fourier-set} puts $\pi_a$ in the weak packet.
Conversely, for a member $\pi_a$ of the weak packet, put
$C=C_{f(a)}\in\mathscr C_{\mathrm{rel}}$. Then
$$
 m_{p(C)}(P_a)=m_C(P_{f(a)})=\dim\rho_{f(a)}>0.
$$
This proves the union formula, including its restriction to each
pure inner twist.

For the containment criterion, the reverse implication follows
from this formula. For the forward implication, put $D=p(B)$ and
choose $a$ with $P_{f(a)}=\operatorname{IC}(\overline D,\mathcal L_1)$.
Then $m_B(P_a)=1$, so containment and
\eqref{eq:weak-fourier-set} force $D\in\mathscr C_{\mathrm{rel}}$.
Finally, Theorem~\ref{thm:weak-injectivity} and factor exchange show
that all the packet bases $p(C)$ in the union are Arthur.
