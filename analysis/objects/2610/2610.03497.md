<!-- definition-2: body_md -->
Let $X$ and $Y$ be graphs.
    \begin{itemize}
        \item The {\texttt{disjoint union of graphs}} $X\sqcup Y$ has both vertices and edges the disjoint union of the vertices and edges of $X$ and $Y$ respectively.
        \item The {\texttt{box product}} $X \mathrel{\square} Y$ is the graph with vertices $V(X\mathrel{\square} Y)=V(X)\times V(Y)$ and edges
              \begin{align*}
                  \big\{ \left\{ (x, y_1), (x, y_2) \right\} \,\big|\, x \in V(X), \left\{ y_1, y_2 \right\} \in E(Y) \big\}\cup\big\{ \left\{ (x_1, y), (x_2, y) \right\} \,\big|\, \left\{ x_1, x_2 \right\} \in E(X), y \in V(Y) \big\}
              \end{align*}
        \item The {\texttt{categorical product}} $X\times Y$ is the graph containing all vertices and edges like $X \mathrel{\square} Y$ with the additional edges
              \begin{align*}
                  \big\{ \left\{ (x_1,y_1),(x_2,y_2) \right\} \,\big|\,\left\{ x_1, x_2 \right\} \in E(X),\left\{ y_1, y_2 \right\} \in E(Y) \big\}
              \end{align*}
    \end{itemize}

<!-- definition-5: body_md -->
~
    Let $X,Y$ be graphs.
    \begin{itemize}
        \item Let $f,g\colon X\to Y$ be graph maps. A graph map $H:X\mathrel{\square} I_m \to Y$ is called a {\texttt{homotopy from $f$ to $g$}} if for all $x\in X$ it holds that
              \[
                  H(x,0)=f(x)\text{ and } H(x,m)=g(x).
              \]
        \item For graphs $X$ and $Y$ and subgraphs $A\subset X$ and $B\subset Y$, a map of pairs $$f:(X,A)\to (Y,B)$$ is a graph map $f:X\to Y$ such that $f(A)\subset B$.
        \item Let $*_X\in V(X)$. A graph map $f:I_\infty^n\to X$ is called {\texttt{almost everywhere constant (a.~e.~c.) with basepoint $*_X$}} if all but finitely many vertices in $I_\infty^n$ get mapped to $*_X$.
        \item Given two almost everywhere constant maps $f,g:I_\infty^n\to X$ with basepoint $*_X$, an {\texttt{almost everywhere constant homotopy}} is a homotopy $H:I_\infty^n\mathrel{\square} I_m \to X$ from $f$ to $g$ such that all but finitely many vertices get mapped to $*_X$.
    \end{itemize}

<!-- definition-6: body_md -->
\label{dfn:htpygroups}
    For every $n\geq 1$, the {\texttt{$n$-th discrete homotopy group}} is the set of equivalence classes of almost everywhere constant graph maps
    \begin{align*}
        \mathcal{A}_n(X,*_X)=\left\{ f\colon I_\infty^n\to X \;\middle|\; f\text{ a. e. c. with basepoint } *_X \right\}/\sim
    \end{align*}
    where the equivalence relation $\sim$ is almost everywhere constant homotopy. For $n\geq 1$, the group multiplication is given by concatenation via
    \[
        [f]\cdot[g]=[f\cdot g]
    \]
    and the neutral element is the constant map with basepoint $*_X$.
    We also define $\mathcal{A}_0(X,*_X)$ as the set of connected components of $X$, pointed by the component containing $*_X$. A sequence of pointed sets $(A,*_A) \xrightarrow{f} (B,*_B) \xrightarrow{g} (C,*_C)$ is called {\texttt{exact}} at $B$ if $\mathrm{im}(f) = g^{-1}(*_C)$.

<!-- proposition-1: body_md -->
Let $(X,*)$ be a based graph and $n\in\mathbb{N}$.
    The following are naturally isomorphic as sets:
    \begin{enumerate}
        \item $$\mathcal{A}_n(X,*)$$
        \item $$\left\{f:(I_m^n,\partial I_m^n)\to(X,*) \, \middle\vert\, m\in\mathbb{N}\right\}/\sim$$
        \item $$\left\{f:(\partial I_m^{n+1},\mathbf{0})\to(X,*) \, \middle\vert\, m\in\mathbb{N}\right\}/\sim$$
    \end{enumerate}
    where in the cases (2) and (3) the equivalence relation $\sim$ is generated by both homotopies that are also maps of pairs of graphs, and precomposition with the morphisms of ${\boldsymbol{\mathsf{LR}}}_n$ and ${\boldsymbol{\mathsf{LR}}}_{n+1}$ respectively. Note that in case (3) we identify $\lambda_m^{n+1}$ and $\rho_m^{n+1}$ with their restrictions $\partial I_{m+1}^{n+1}\to \partial I_{m}^{n+1}$.

<!-- definition-7: body_md -->
Let $X,Y$ be graphs and $\varphi\colon X\to Y$ a graph map.
    \begin{itemize}
        \item $\varphi$ is called an $n${\texttt{-equivalence}} if for all $*_X\in V(X)$ the induced map $\varphi_*\colon \mathcal{A}_k(X,*_X) \to \mathcal{A}_k(Y,\varphi(*_X))$ is an isomorphism for all $k< n$, and an epimorphism for $k=n$.
        \item $\varphi$ is called a {\texttt{weak equivalence}} if $\varphi$ is an $n$-equivalence for all $n \geq 1$.
        \item $\varphi$ is called a {\texttt{homotopy equivalence}} if there is a map $\psi\colon Y\to X$ such that $\psi\circ \varphi$ and $\varphi \circ \psi$ are homotopic to the identity map on $X$ and $Y$ respectively.
    \end{itemize}

<!-- lemma-1: body_md -->
\label{lem:colimithomotopygroups}
    Let $X_1,X_2,\ldots$ be a sequence of connected graphs with a family of inclusions of induced subgraphs $\iota_k: X_k \to X_{k+1}$, such that $\iota_k$ induces an isomorphism $\iota_{k*}:\mathcal{A}_i(X_k) \to \mathcal{A}_i(X_{k+1})$ for all $i\leq k$. Let $X_\infty$ be the colimit of this sequence. Then the universal cocone map $\hat{\iota}_k\colon X_k\to X_\infty$ induces isomorphisms $\hat{\iota}_{k*}:\mathcal{A}_i(X_k) \to \mathcal{A}_i(X_\infty)$ for all $k\geq 1$ and $i\leq k$.
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
Let $i\geq 1$. We have to show that $\hat{\iota}_k\colon X_k\to X_\infty$ induces isomorphisms of homotopy groups in degrees $i\leq k$. We show that $\hat{\iota}_{k*}:\mathcal{A}_i(X_k) \to \mathcal{A}_i(X_\infty)$ is injective and surjective separately. Firstly, consider a a. e. c. map $f:I_\infty^i\to X_k$ such that $\hat{\iota}_{k*}([f])=0$. Then there is a a. e. c. homotopy $H:I_\infty^i\mathrel{\square} I_m\to X_\infty$ from $\hat{\iota}_k\circ f$ to the constant map. Since the image of $H$ is finite, there is an $N\geq k$ such that the image of $H$ is contained in $X_N$. Writing $\iota_{N,k} \coloneqq \iota_{N-1} \circ \dots \circ \iota_k : X_k \to X_N$, since $(\iota_{N,k})_*:\mathcal{A}_i(X_k) \to \mathcal{A}_i(X_N)$ is an isomorphism, we have that $[f]=0$ in $\mathcal{A}_i(X_k)$ and thus $\hat{\iota}_{k*}$ is injective. Secondly, consider a a. e. c. map $f:I_\infty^i\to X_\infty$. Since the image of $f$ is finite, there is an $N\geq k$ such that the image of $f$ is contained in $X_N$. Since $(\iota_{N,k})_*:\mathcal{A}_i(X_k) \to \mathcal{A}_i(X_N)$ is an isomorphism, there is a a. e. c. map $g:I_\infty^i\to X_k$ such that $(\iota_{N,k})_*([g])=[f]$ in $\mathcal{A}_i(X_N)$. Thus, we have that $\hat{\iota}_{k*}([g])=[f]$ in $\mathcal{A}_i(X_\infty)$ and thus $\hat{\iota}_{k*}$ is surjective.

<!-- definition-9: body_md -->
\label{dfn:cov}

\label{lem:covering_Umg}

    \[

    \begin{enumerate}

        \item For every $x\in X$ and every $g\in G$, the intersection $\mathcal{U}_2(x)\cap \mathcal{U}_2(g\triangleright x)$ is nonempty if and only if $g$ is the identity in $G$.

<!-- lemma-4: body_md -->
\label{lem:FundGrpCovering}
    Let $(X,*_X)$ be a connected, based graph with a freely sparse left action of a group $G$, then the quotient map $q:(X,*_X)\to (G\backslash X,*_ {G\backslash X})$ is a regular covering and there is an isomorphism
    \[
        \Phi:\faktor{\mathcal{A}_1(G\backslash X,*_{G\backslash X})}{q_*(\mathcal{A}_1(X,*_X))}\cong G
    \]
    such that $\Phi$ is compatible with the action of $G$ on the fiber $q^{-1}(*_{G\backslash X})$ and the monodromy action in the sense that for every $x\in q^{-1}(*_{G\backslash X})$ and every $[\gamma]\in \mathcal{A}_1(G\backslash X,*_{G\backslash X})$, it holds that
    \[
        (h \triangleright *_X) \triangleleft [\gamma] = (h\cdot \Phi([\gamma]q_*(\mathcal{A}_1(X,*_X))))\triangleright ( *_X).
    \]
<!-- end lemma-4 -->
<!-- proof-10: body_md -->
Since $G$ acts freely and sparsely on $X$, for every $x\in X$, the preimage of the neighborhood can be characterized as
    \[
        q^{-1}(\mathcal{U}_2(q(x)))= \bigsqcup_{g\in G} \mathcal{U}_2(g\triangleright x).
    \]
    Therefore, by \ref{lem:covering_Umg}, it holds that $q$ is a covering. Furthermore, there is a canonical identification between $q^{-1}(*_{G\backslash X})$ and $G$ where we identify $x\in q^{-1}(*_{G\backslash X})$ with the element $g\in G$ such that $g\triangleright *_X=x$. Note that this element $g$ is unique for a given $x$ since the action of $G$ on $X$ is free. Using this, we consider the map
    \begin{align*}
        \mathcal{A}_1(G\backslash X,*_{G\backslash X}) & \longrightarrow q^{-1}(*_{G\backslash X}) \\
        [\gamma]                               & \longmapsto *_X\triangleleft [\gamma]
    \end{align*}
    where $*_X\triangleleft [\gamma]$ denotes the monodromy action of $[\gamma]$ on $*_X$, i.e., the endpoint of the unique lift of $\gamma$ starting at $*_X$. Using the identification of $q^{-1}(*_{G\backslash X})$ and $G$ above, we obtain a map $\Psi:\mathcal{A}_1(G\backslash X,*_{G\backslash X})\to G$ defined by the equality $\Psi([\gamma])\triangleright *_X = *_X\triangleleft [\gamma]$. We will show that $\Psi$ is an epimorphism of groups with kernel $q_*(\mathcal{A}_1(X,*_X))$. Firstly, let $[\gamma_1],[\gamma_2]\in \mathcal{A}_1(G\backslash X,*_{G\backslash X})$ and let $g_i=\Psi([\gamma_i])$. The unique lift of $\gamma_1\cdot\gamma_2$ starting at $*_X$ is the concatenation of the lift of $\gamma_1$ starting at $*_X$ and the lift of $\gamma_2$ starting at $*_X\triangleleft [\gamma_1] = g_1\triangleright *_X$. Let $\tilde{\gamma}_2:I_n\to X$ be the lift of $\gamma_2$ starting at $*_X$ and consider the path \begin{align*}
        \tilde{\gamma}'_2:I_n & \longrightarrow X                                      \\
        t                     & \longmapsto g_1\triangleright \tilde{\gamma}_2(t).
    \end{align*}

    Then, by the uniqueness of lifts of paths for coverings (cf.\ \cite[4.5]{Udit:Fundamentalgroupoid}), this second lift $\tilde{\gamma}'_2$ is the lift of $\gamma_2$ starting at $g_1\triangleright *_X$. Its endpoint is then
    \[
        g_1\triangleright (g_2\triangleright *_X) = (g_1g_2)\triangleright *_X~.
    \]
    This shows $\Psi([\gamma_1\cdot\gamma_2]) = g_1g_2 = \Psi([\gamma_1])\Psi([\gamma_2])$, i.e. $\Psi$ is a group homomorphism.

    To see that $\Psi$ is surjective, let $g\in G$. Since $X$ is connected, there exists a path $\tilde{\gamma}$ in $X$ from $*_X$ to $g\triangleright *_X$. Then $\gamma=q\circ \tilde{\gamma}$ is a loop in $G\backslash X$ based at $*_{G\backslash X}$, and its unique lift starting at $*_X$ is exactly $\tilde{\gamma}$. Therefore, $*_X\triangleleft [\gamma] = g\triangleright *_X$, which implies $\Psi([\gamma])=g$.

    Finally, we determine the kernel of $\Psi$. A loop $[\gamma]\in \mathcal{A}_1(G\backslash X,*_{G\backslash X})$ is in $\ker(\Psi)$ if and only if $\Psi([\gamma])=e$, the identity in $G$, i.e. $*_X\triangleleft [\gamma] = e\triangleright *_X = *_X$. This holds if and only if the lift of $[\gamma]$ starting at $*_X$ is a loop in $X$, i.e. an element of $\mathcal{A}_1(X,*_X)$.

    Since the group $G$ acts transitively on the fiber $q^{-1}(*_{G\backslash X})$ by deck transformations, $q$ is a regular covering. The existence of the isomorphism $\Phi$ follows from the first isomorphism theorem for groups.

Let $G$ be a group. We recursively construct a sequence of graphs $E_n(G)$ equipped with a left action of $G$ as follows:
    \begin{itemize}
        \item Let $E_0(G) = G$ be the discrete graph with vertex set $G$ and no edges. The group $G$ acts on $E_0(G)$ by left multiplication.
        \item For $n \geq 1$ define $E_n(G)$ as
              \[
                  E_n(G) = G \mathrel{\star}_{n+2} E_{n-1}(G) = \left( \bigsqcup_{g\in G} \{g\}\times I_{n+2} \times E_{n-1}(G) \right) \Big/ \sim
              \]
              where $\sim$ is the equivalence relation generated by:
              \begin{align*}
                  (g,n+2,x) & \sim (g,n+2,y), \\
                  (g,0, x)  & \sim (h,0,x).
              \end{align*}
              The action of $G$ on $E_n(G)$ is given by $h\triangleright (g,k,x) := (h\cdot g,k,h\triangleright x)$.
    \end{itemize}
    We then define $EG$ as the colimit
    \[
        EG := \underset{}{\operatorname{colim}~}\left( E_0(G) \hookrightarrow E_1(G) \hookrightarrow \dots \right).
    \]
    We denote the quotients $B_n(G) \coloneqq G\backslash E_n(G)$ and $BG \coloneqq G\backslash EG$.

We show this inductively on $n$. For $n=0$, $E_0(G)=G$ is a discrete graph with no edges, so the action of $G$ is freely sparse.
    Now assume that for some $n \geq 0$, the action of $G$ on $E_n(G)$ is freely sparse. Consider $E_{n+1}(G) = G \mathrel{\star}_{n+3} E_n(G)$. Let $(g,k,x)\in E_{n+1}(G)$ and $h\in G\setminus \{e\}$. And assume there is a path of length at most $4$ from $(g,k,x)$ to $h\triangleright (g,k,x) = (h\cdot g, k, h\triangleright x)$. Firstly, consider the case $k\geq 2$. Then, since $n+3\geq 3$, the $2$-neighborhood of $(g,k,x)$ is contained in the induced subgraph with vertices
    \[
        \{g\}\mathrel{\star}_{n+3} E_n(G)
    \]
    while the $2$-neighborhood of $(h\cdot g,k,x)$ is contained in the induced subgraph with vertices
    \[
        \{h\cdot g\}\mathrel{\star}_{n+3} E_n(G)
    \]
    Therefore, $k=2$ and a path from $(g,k,x)$ to $(h\cdot g,k,h\triangleright x)$ must be of the form
    \[
        (g,2,x) - (g,1,x_1) - (g,0,x_2) - (h\cdot g,1,x_2) - (h\cdot g,2,h\triangleright x),
    \]
    This implies that $x_2 \in \mathcal{U}_2(x)$ and $x_2\in \mathcal{U}_2(h\triangleright x)$. Therefore by induction, such a path cannot exist.


    Secondly, consider the case $k\leq 1$. Then, since $n+3 \geq 3$ every path of length at most two starting at $(g,k,x)$ and containing a cone point must terminate at this cone point. Furthermore, the only cone point that could be contained in the $2$-neighborhood of $(g,k,x)$ (of $(h\cdot g,k,h\triangleright x)$) is $(g,n+3,x)$ (resp. $(h\cdot g,n+3,h\triangleright x)$).  Thus the $2$-neighborhood of $(g,k,x)$ is contained in the subgraphs of $E_{n+1}G$ induced by
    \[
        G\mathrel{\star}_{n+3} \mathcal{U}_2(x) \subset G\mathrel{\star}_{n+3} E_n(G).
    \]
    and the $2$-neighborhood of $h\triangleright (g,k,x)$ is contained in
    \[
        G\mathrel{\star}_{n+3} \mathcal{U}_2(h\triangleright x) \subset G\mathrel{\star}_{n+3} E_n(G).
    \]
    Again by induction, it follows that these two induced subgraphs only intersect in the cone points, however, $(g,n+3,x)$ and $(h\cdot g,n+3,h\triangleright x)$ are distinct if $g\neq h\cdot g$, that is, if $h\neq e$. Hence, this shows that the action of $G$ on $E_{n+1}(G)$ is freely sparse.

    For the action of $G$ on $EG$, assume we are given a vertex $y\in EG$. Since $EG$ is the colimit of the sequence $E_0(G)\hookrightarrow E_1(G)\hookrightarrow \dots$, it follows that $y$ is contained in some $E_n(G)$. Assume that there is an $h\in G$ with $h\triangleright \mathcal{U}_2(y)\cap\mathcal{U}_2(y)\neq\emptyset$. Then there exists $k\in\mathbb{N}$ and $z\in E_k(G)$ with $h\triangleright z\in\mathcal{U}_2(y)$ and $z\in\mathcal{U}_2(y)$. Let $m=\max(n,k)$; since the action is freely sparse on $E_m(G)$ it follows that $h=e$.

<!-- proposition-8: body_md -->
\label{prop:En_connected}
    $E_n(G)$ is $(n-1)$-connected for all $n> 0$.
<!-- end proposition-8 -->
<!-- proof-12: body_md -->
First, we will show by induction that for every $n\geq 2$, $E_n(G)$ is simply connected. For this consider a loop $\gamma:I_m\to E_n(G)$. Since $m$ is finite, it follows that $\gamma$ is contained in $X\coloneqq \{g_1,\ldots, g_k\}\mathrel{\star}_{n+2}E_{n-1}(G)$ for finitely many $g_1,\ldots, g_k \in G$. We will show that $X$ is simply connected: Let $X_i=\{g_1,\ldots,g_i\}\mathrel{\star}_{n+2}E_{n-1}(G)$ for $i=1,\ldots, k$. $X_1$ is a cone over $E_{n-1}(G)$, hence contractible and in particular simply connected. Now assume $X_i$ is simply connected and set $A=X_{i+1}\setminus\{(g_{i+1},n+2)\}$ and $B=X_{i+1}\setminus \{(g_l,n+2)\mid 1\leq l \leq i\}$. Then $A$ is homotopy equivalent to $X_i$ and $B \simeq \{g_{i+1}\} \mathrel{\star}_{n+2} E_{n-1}(G)$ is contractible. Their intersection $A \cap B \simeq E_{n-1}(G)$ is connected since $n \geq 2$. It follows from the Seifert-van Kampen theorem for graphs~(\ref{thm:seifertvankampen}) that $\mathcal{A}_1(X_{i+1})=\mathcal{A}_1(X_i)=0$. Note that this argument does not work in the case $n=1$, since the intersection $A\cap B$ would not be connected.

    We proceed by showing inductively that $E_n(G)$ is $(n-1)$-connected. Clearly the graph $E_1(G) = G\mathrel{\star}_{3}G$ is connected, i.e. $0$-connected, and $E_2(G)$ is simply connected by the first part of the proof, i.e. $1$-connected. We will now show, for $n\geq 2$, that $E_{n+1}(G)$ is $n$-connected under the assumption that $E_n(G)$ is $(n-1)$-connected. Recall that $E_{n+1}(G)=G\mathrel{\star}_{n+3}E_n(G)  =(G\times I_{n+3}\times E_n(G))/\sim$. Consider the two induced subgraphs
    \[
        A\coloneqq E_{n+1}(G)\setminus\left\{\,(g,n+3)\,\middle|\, g\in G  \,\right\}
    \]
    and
    \[
        B\coloneqq E_{n+1}(G)\setminus\left\{\,(g,0,x)\,\middle|\, x\in E_n(G), g\in G  \,\right\}.
    \]
    The distance between $A\setminus B$ and $B\setminus A$ is $n+3$. We thus obtain a partial Mayer-Vietoris sequence up to degree $n+1$. While $A$ is homotopy equivalent to $E_n(G)$, $B$ is homotopy equivalent to a disjoint union of $|G|$ vertices. The intersection $A\cap B$ has the homotopy type of $|G|$ copies of $E_n(G)$. In degrees $1\leq k\leq n-1$ it holds that $H_k(A\cap B)$, $H_k(A)$ and $H_k(B)$ vanish, therefore we have $H_k(E_{n+1}(G))=0$ for $1\leq k\leq n-1$. For degree $k=n$ consider the relevant part of the Mayer-Vietoris sequence:
    \[
        \begin{tikzcd}
            H_n(A\cap B) \ar[r,"\iota"] & H_n(A) \oplus H_n(B)  \ar[r,"j"] & H_n(E_{n+1}(G)) \ar[r] & H_{n-1}(A\cap B)=0
        \end{tikzcd}
    \]
    Since $H_n(B)$ also vanishes, the map $j$ is hence the map induced by the inclusion $\iota_A:A\to E_{n+1}(G)$. However, it can be seen that $\iota_A$ is homotopic to a constant map as follows: Firstly we can deform $A$ to $E_n(G)$ via a deformation retraction
    \begin{align*}
        h_1: A \mathrel{\square} I_{n+3} & \longrightarrow  A                  \\
        ((g,k,x),t)           & \longmapsto (g,\min(k,n+3-t),x)
    \end{align*}
    and hence deform $\iota_A$ to the inclusion $\iota_{E_n(G)}: E_n(G) \hookrightarrow E_{n+1}(G)$. This inclusion is homotopic to a constant map via:
    \begin{align*}
        h_2:E_{n}(G) \mathrel{\square} I_{n+3} & \longrightarrow E_{n+1}(G) \\
        (x,t)                       & \longmapsto (e,t,x).
    \end{align*}
    It follows that in the Mayer-Vietoris sequence the map $j:H_n(A)\oplus H_n(B)\to H_n(E_{n+1}(G))$ is the $0$-map, by the inductive assumption $H_{n-1}(A\cap B)=0$ and therefore $H_n(E_{n+1}(G))=0$.

    We have now shown that $E_{n+1}(G)$ is simply connected and the first $n$ homology groups vanish, hence by the Hurewicz theorem for graphs it follows that $E_{n+1}(G)$ is $n$-connected, completing the induction.

<!-- theorem-10: body_md -->
\label{thm:blakersmassey}
    Let $X$ be a graph and $A,B\subseteq X$ be induced subgraphs with non-empty intersection such that $X=A\cup B$. Furthermore, let $m,n\geq 1$ such that the distance between $A\setminus B$ and $B\setminus A$ is at least $m+n$. If it holds that
    \begin{align*}
        \mathcal{A}_i(A,A\cap B,*)=0 & ~\textnormal{for } 0<  i< m, \text{ and} \\
        \mathcal{A}_i(B,A\cap B,*)=0 & ~\textnormal{for } 0<  i< n
    \end{align*}
    for all basepoints $*\in A\cap B$, then it holds that the map induced by the inclusion $$\iota_*: \mathcal{A}_k(A,A\cap B,*)\to \mathcal{A}_k(X,B,*)$$ is bijective for $1\leq k< m+n-2$ and surjective for $k=m+n-2$.
<!-- end theorem-10 -->
<!-- proof-14: body_md -->
Firstly, observe that in the $(n+m-1)$-skeleton $|\mathbf{sk}_{n+m-1}N_1(X)|$, it holds that $$|\mathbf{sk}_{n+m-1}N_1(X)|=|\mathbf{sk}_{n+m-1}N_1(A)|\cup |\mathbf{sk}_{n+m-1}N_1(B)|.$$ This is because the distance between $A\setminus B$ and $B\setminus A$ is at least $m+n$, so any singular $k$-cube $I_1^k\to X$ factors through $A$ or $B$ for $k\leq m+n-1$. Since for CW-complexes, the $k$-th homotopy groups only depend on the $(k+1)$-skeleton for all $k$, we obtain a commutative diagram
    \[
        \begin{tikzcd}
            {\boldsymbol{\mathsf{Graph}}} \ar[d,"\mathcal{A}_k"] \ar[r,"N_1"] &{\boldsymbol{\mathsf{cSet}}} \ar[r,"\mathbf{sk}_{n+m-1}"] &{\boldsymbol{\mathsf{cSet}}}\ar[d,"|\cdot|"]\\
            {\boldsymbol{\mathsf{Grp}}} & & {\boldsymbol{\mathsf{Top}}} \ar[ll,"\pi_k"]
        \end{tikzcd}
    \]
    for all $k\leq m+n-2$. The claim then follows, using \ref{lem:relativemagic}, from the classical Blakers-Massey theorem (cf.\ \cite[6.4.1]{Dieck:AT}) applied to the spaces $|\mathbf{sk}_{n+m-1}N_1(A)|$, $|\mathbf{sk}_{n+m-1}N_1(B)|$ and $|\mathbf{sk}_{n+m-1}N_1(X)|$. Note that the assumption in \cite[6.4.1]{Dieck:AT} is that the subspaces are open; however, for a CW-pair $(Y,A)$, there is an open neighborhood $U$ of $A$ in $Y$ such that $A$ is a deformation retract of $U$ (cf.\ \cite[5.4.4]{Dieck:AT}).

\label{thm:postnikov}
    Let $(X,*_X)$ be a connected based graph and $n\geq 1$. Then there is a sequence of based graphs $(X_j,*_{X_j})$, $0\leq j\leq n$ and based graph maps $p_j:(X_j,*_{X_j})\to (X_{j-1},*_{X_{j-1}})$ and $\varphi_j:X\to X_j$ fitting in a commutative diagram

        \end{tikzcd}

    Finally, we set $\varphi_n\coloneqq s_n\circ \hat{\varphi}_n\colon X\to X_n$, and for $j<n$ the maps $\varphi_j:X\to X_j$ are defined as the composition of $\varphi_n$ with the maps $p_{j+1}\circ \cdots \circ p_{n-1}\circ p_{n}$, finishing the conditions of the theorem.

<!-- construction-2: body_md -->
\label{const:KPIN}
    Let $M$ be an Abelian group and $n\geq 2$. We construct a connected graph $K(M,n)$ such that $\mathcal{A}_n(K(M,n))\cong M$ and $\mathcal{A}_i(K(M,n))=0$ for all $i\neq n$.
    \begin{enumerate}
        \item Let $X = BM$ be a connected graph such that $\mathcal{A}_1(X,*)\cong M$; this exists by \ref{thm:BG}.
        \item Then, consider the $(n-1)$-fold suspension $\Sigma_{n+2}^{n-1} X$,
        \item and build the partial Postnikov system of length $n$.
    \end{enumerate}
    The $n$-th piece of the tower $(\Sigma_{n+2}^{n-1} X)_n$ is a $K(M,n)$-graph.
<!-- end construction-2 -->
<!-- proof-18: body_md -->
Given $X$ as in the construction, firstly observe that, since $M$ is Abelian, $H_1(X)=M$, by the Hurewicz theorem for graphs. Iteratively applying the Mayer-Vietoris sequence we obtain that
    \[
        H_k(\Sigma_{n+2}^{n-1} X) = 0 ~\textnormal{for } 0< k < n \text{ and } H_n(\Sigma_{n+2}^{n-1} X)=M.
    \]
    It now follows from the Hurewicz theorem for graphs that $\mathcal{A}_k(\Sigma_{n+2}^{n-1} X)=0$ for $0< k < n$ and $\mathcal{A}_n(\Sigma_{n+2}^{n-1} X)\cong M$. Since the Postnikov tower kills the higher homotopy groups, the $n$-th term $(\Sigma_{n+2}^{n-1} X)_n$ is a $K(M,n)$.

\label{cor:sequence_homotopy_groups}
    For every sequence of groups $M_1,M_2,\ldots$ such that $M_n$ is Abelian for all $n\geq 2$, there is a connected graph $X$ such that $\mathcal{A}_n(X)\cong M_n$ for all $n\geq 1$.
<!-- end corollary-2 -->
<!-- proof-19: body_md -->
Consider the sequence of based graphs $(X_1,*_1),(X_2,*_2),\ldots$ where $X_1 = BM_1$ from \ref{thm:BG} and $X_n$ is a $K(M_n,n)$ from \ref{const:KPIN} for $n\geq 2$, and consider the colimit
    \[
        X=\underset{n}{\operatorname{colim}~}\left(X_1\xrightarrow{\operatorname{id}_{X_1}\times c_{*_2}} X_1\times X_2\xrightarrow{\operatorname{id}_{X_1}\times\operatorname{id}_{X_2} \times c_{*_3}} \cdots \right).
    \]
    For every pair of based graphs $(X,*_X)$ and $(Y,*_Y)$ it holds that $\mathcal{A}_n(X \times Y, (*_X, *_Y)) \cong \mathcal{A}_n(X, *_X) \times \mathcal{A}_n(Y, *_Y)$(cf. \cite{Chris:Revisited}). Since every $X_n$ is $(n-1)$-connected, the maps $\operatorname{id}_{X_1}\times \cdots \times \operatorname{id}_{X_{n-1}}\times c_{*_n}$ are $(n-1)$-equivalences. By \ref{lem:colimithomotopygroups}, we can then conclude that
    \[
        \mathcal{A}_n(X)=\mathcal{A}_n(X_1\times \cdots \times X_n)\cong\mathcal{A}_n(X_1)\times \cdots \times \mathcal{A}_n(X_n)\cong M_n,
    \]
    as desired.

<!-- theorem-12: body_md -->
\label{thm:whitehead}
    Let $(X,*_X)$ be a connected based graph. Then there exists a sequence of based graphs $(X^{(i)},*_i)$, $i\geq 0$, and graph maps $\psi_i$
    \[
        \cdots \xrightarrow{\psi_3} X^{(2)} \xrightarrow{\psi_2} X^{(1)} \xrightarrow{\psi_1} X^{(0)}=X
    \]
    such that
    \begin{enumerate}
        \item $\mathcal{A}_i(X^{(j)})=0$ for $0<i\leq j$,
        \item $\psi_j$ induces isomorphisms $\mathcal{A}_i(X^{(j)})\to \mathcal{A}_i(X^{(j-1)})$ for all $i>j$,
        \item $\psi_j$ is a fibration with fiber $K(\mathcal{A}_j(X),j-1)$.
    \end{enumerate}
<!-- end theorem-12 -->
<!-- proof-20: body_md -->
We construct the sequence inductively. Let $X^{(0)}=X$ and let $X^{(1)}$ be the universal cover with covering map $\psi_1: X^{(1)} \to X$. Then $\psi_1$ induces an isomorphism on higher homotopy groups by \ref{prop:UditChrisCovering} and the fiber is discrete and can be endowed with a group structure isomorphic to $\mathcal{A}_1(X)$ by \ref{lem:FundGrpCovering}.
    Now, assume we have constructed $(X^{(j)},\psi_j)$ as desired for all $j\leq n-1$ and consider the partial Postnikov tower for $X^{(n-1)}$:
    \[
        \begin{tikzcd}[row sep=large, column sep=large]
            X^{(n-1)} \arrow[d,swap, "\varphi_n"']
            \arrow[dr, "\varphi_{n-1}"]
            \arrow[drrr, bend left=9, "\varphi_1"]
            \arrow[drrrr, bend left=12, "\varphi_0"] \\
            \left(X^{(n-1)}\right)_{n} \arrow[r, "p_n"]
            & \left(X^{(n-1)}\right)_{n-1} \arrow[r, "p_{n-1}"]
            & \cdots \arrow[r, "p_2"]
            & \left(X^{(n-1)}\right)_1 \arrow[r, "p_1"]
            & \left(X^{(n-1)}\right)_0=\{*\}
        \end{tikzcd}
    \]
    Note that for $i<n$, $\left(X^{(n-1)}\right)_i$ is weakly contractible and $\left(X^{(n-1)}\right)_n$ is a $K(\mathcal{A}_n(X),n)$.
    Let
    \begin{align*}
        X^{(n)} & \coloneqq\mathrm{hofib}(\varphi_n)                                                                                                                                                                           \\
                & =\left\{\,(\omega,x)\in P\left(\left(X^{(n-1)}\right)_{n}\right)\times \left(X^{(n-1)}\right) \,\middle|\,\underset{m\to \infty}{\operatorname{lim}}~\omega(m) = \varphi_n(x),\, \omega(0)=*_{\left(X^{(n-1)}\right)_n} \,\right\},
    \end{align*}
    the homotopy fiber of $\varphi_n$. By \ref{thm:postnikov}, $\varphi_n$ induces an isomorphism on homotopy groups $\mathcal{A}_i(X^{(n-1)},*)\cong \mathcal{A}_i(\left(X^{(n-1)}\right)_n,*)$ for $i\leq n$, it thus follows from the long exact sequence of a fibration, that
    \[
        \mathcal{A}_i(X^{(n)})=\begin{cases} 0 & i\leq n \\ \mathcal{A}_i\left(X\right) & i>n . \end{cases}
    \]
    Thus by setting
    \begin{align*}
        \psi_n \colon X^{(n)} & \longrightarrow X^{(n-1)} \\
        (\omega,x)           & \longmapsto x
    \end{align*}
    we have constructed the $n$-th stage of the tower as desired.
