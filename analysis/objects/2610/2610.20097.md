<!-- lemma-1: body_md -->
\label{lem:continuity}
Suppose that the smooth metrics $g_\varepsilon$, on the closure of a fixed
bounded smooth domain $\Omega$, $C^2$-converge to the smooth metric $g_0$ as $\varepsilon\rightarrow0$. Then, for every $k\in\mathbb{N}$,
\[
 \lambda_k(\Omega,g_\varepsilon)\longrightarrow\lambda_k(\Omega,g_0),\qquad
 \Lambda_k(\Omega,g_\varepsilon)\longrightarrow\Lambda_k(\Omega,g_0),
 \qquad\text{as}~\varepsilon\rightarrow0.
 \]
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
The Sobolev spaces for metrics $g_\varepsilon$ and $g_0$ are topologically equivalent. We use the same notation for these spaces  $L^2(\Omega)$, $H^1(\Omega)$ and $H^2(\Omega)$ and the respective norms $\|u\|_{L^2(\Omega)}^2 := \int_\Omega|u|^2{\,\mathrm{d}} V_{g_0} $, $\|u\|_{H^1(\Omega)} ^2:= \||\nabla u|\|^2_{L^2(\Omega)}+ \|u\|^2_{L^2(\Omega)}$, and $\|u\|_{H^2(\Omega)}^2 := \||D^2u|\|^2_{L^2(\Omega)}+ \|u\|^2_{H^1(\Omega)}$ will be for convenience evaluated in the limiting metric $g_0$.
	We introduce the symmetric bilinear forms
	\[
	\begin{aligned}
		\mathfrak{d}_g[u,v]&=\int_\Omega\langle\nabla u,\nabla v\rangle_g{\,\mathrm{d}} V_g,&\qquad\mathrm{dom}\,\mathfrak{d}_g = H^1_0(\Omega),\\
		\mathfrak{a}_g[u,v]&=\int_\Omega(\Delta_g u)(\Delta_g v){\,\mathrm{d}} V_g,&\qquad
		\mathrm{dom}\,\mathfrak{a}_g = H^2_0(\Omega),\\
		\mathfrak{b}_g[u,v]&=\int_\Omega uv{\,\mathrm{d}} V_g,&\qquad
		\mathrm{dom}\,\mathfrak{b}_g = L^2(\Omega).
	\end{aligned}
	\]
	The Laplace-Beltrami operator in local coordinates acts as
	\[
	\Delta_g u=|g|^{-1/2}\partial_i
	\bigl(|g|^{1/2}g^{ij}\partial_j u\bigr).
	\]
	Uniform convergence of the coefficients in the expression for the Laplace-Beltrami operator
	implies that there exists $\delta_\varepsilon>0$ satisfying $\delta_\varepsilon\rightarrow0$ as $\varepsilon\rightarrow 0$ such that
	\[
	\begin{aligned}
		|\mathfrak{a}_{g_\varepsilon}[u]-\mathfrak{a}_{g_0}[u]|
		&\leq\delta_\varepsilon\|u\|_{H^2(\Omega)}^2,\qquad &\text{for all}\, u\in H^2_0(\Omega),\\
		|\mathfrak{d}_{g_\varepsilon}[u]-\mathfrak{d}_{g_0}[u]|
		&\leq\delta_\varepsilon\|u\|_{H^1(\Omega)}^2,\qquad &\text{for all}\, u\in H^1_0(\Omega),\\
		|\mathfrak{b}_{g_\varepsilon}[u]-\mathfrak{b}_{g_0}[u]|
		&\leq\delta_\varepsilon\|u\|_{L^2(\Omega)}^2,&\qquad\text{for all}\, u\in L^2(\Omega).
	\end{aligned} 
	\]
	The Dirichlet elliptic estimate and Poincar\'e inequality yield
	that there exists a constant $C \ge 1$ such that
	$\|u\|_{H^2(\Omega)}^2\leq C \mathfrak{a}_{g_0}[u]$ for all $u\in H^2_0(\Omega)$; $\|u\|_{H^1(\Omega)}^2\le C\mathfrak{d}_{g_0}[u]$ for all $u\in H^1_0(\Omega)$. Thus,  setting
	$\hat\delta_\varepsilon = C\delta_\varepsilon$, we get
	\[
	\begin{split}
		(1-\hat\delta_\varepsilon)\mathfrak{a}_{g_0}
		&\leq \mathfrak{a}_{g_\varepsilon}\leq(1+\hat\delta_\varepsilon)\mathfrak{a}_{g_0},\\
		(1-\hat\delta_\varepsilon)\mathfrak{d}_{g_0}
		&\leq \mathfrak{d}_{g_\varepsilon}\leq(1+\hat\delta_\varepsilon)\mathfrak{d}_{g_0},\\
		(1-\hat\delta_\varepsilon)\mathfrak{b}_{g_0}
		&\leq \mathfrak{b}_{g_\varepsilon}\leq(1+\hat\delta_\varepsilon)\mathfrak{b}_{g_0},
	\end{split}
	\]
	as quadratic forms. The two
	min-max principles~\eqref{eq:dirichlet-minmax} and~\eqref{eq:buckling-minmax} combined with the above estimates bound each perturbed eigenvalue corresponding to the metric $g_\varepsilon$ between factors
	tending to one times its unperturbed value corresponding to the metric $g_0$, from which the desired continuity of the eigenvalues follows.

<!-- theorem-1: body_md -->
\label{thm:flat-cylinder}
If $\rho\in(0,\frac{1}{2\sqrt{2}\pi})$, then
\[
 \Lambda_2(C_\rho)=4z_1^2<9\pi^2=\lambda_3(C_\rho),
\]
where $z_1\in(\pi,3\pi/2)$ is the first positive nonzero root of the transcendental equation
$\tan z=z$.
<!-- end theorem-1 -->
<!-- proof-2: body_md -->
By separation of variables, the Dirichlet eigenvalues of the Laplacian on $C_\rho$ can be listed as
\begin{equation}
 \lambda_{n,m}=\pi^2n^2+\frac{m^2}{\rho^2},
 \qquad n\in\mathbb{N},\quad m\in\mathbb Z.
 \label{eq:cylinder-dirichlet}
\end{equation}
Clearly, the total sequence of Dirichlet eigenvalues of the Laplacian on $C_\rho$ is the total enumeration of all the above values, where repetitions account for multiplicities.
The assumption on $\rho$ places the first non-constant angular level ($|m| = 1$)
strictly above $9\pi^2$. Thus the first three eigenvalues of the Dirichlet Laplacian
on $C_\rho$ are simple and given by $\pi^2$, $4\pi^2$, and $9\pi^2$ (all of them correspond to $m = 0$). In particular, $\lambda_3(C_\rho) = 9\pi^2$.


Again applying the separation of variables, we observe that the buckling eigenvalues on $C_\rho$ can be listed as
\begin{equation}
 \Lambda_{n,m}=\min_{\substack{E\subset H^2_0(0,1)\\ \dim E=n}}
 \max_{0\neq f\in E}\frac{\displaystyle\int_0^1\left(-f'' + \tfrac{m^2}{\rho^2}f\right)^2{\,\mathrm{d}} x}{\displaystyle\int_0^1 \left((f')^2+\tfrac{m^2}{\rho^2}f^2\right){\,\mathrm{d}} x},
 \qquad n\in\mathbb{N},\quad m\in\mathbb Z.
 \label{eq:cylinder-dirichlet}
\end{equation}
The total family of buckling eigenvalues on $C_\rho$ is the enumeration of all the above values, where repetitions account for multiplicities.
The values $\Lambda_{n,0}$ for all $n\in\mathbb{N}$
can be found as values $q^2$ for $q > 0$ for which the following system
\[
\begin{cases}
    f^{(4)}+ q^2 f'' = 0,\qquad\qquad\qquad\qquad \text{in}\,\,(0,1),\\
    f(0)=f(1) =f'(0)=f'(1)=0,
\end{cases}
\]
has a non-trivial solution in $C^\infty([0,1])$. The existence of several linearly independent solutions for a given value $q^2$ accounts for the multiplicity of the respective eigenvalue.

The general solution of $f^{(4)}+q^2f''=0$ is
\[
 f(x)=a+bx+c\cos(qx)+d\sin(qx),
\]
with constants $a,b,c,d\in\mathbb{R}$.
The conditions at $x=0$ give $a=-c$ and $b=-qd$.
The remaining equations are
\[
 \begin{pmatrix}\cos q-1&\sin q-q\\-\sin q&\cos q-1\end{pmatrix}
 \begin{pmatrix}c\\d\end{pmatrix}=0.
\]
The corresponding determinant equals
\begin{equation}
 2(1-\cos q)-q\sin q
 =4\sin(q/2)\bigl[\sin(q/2)-(q/2)\cos(q/2)\bigr].
 \label{eq:cylinder-determinant}
\end{equation}
From the construction we notice that all the eigenvalues are simple.
The apparent root $q=0$ gives no nonzero clamped eigenfunction and
is excluded. The positive roots are $q=2k\pi$, $k\in\mathbb{N}$ and $q=2z_k$,
where $z_k\in(\pi k,\pi k+\pi/2)$, $k\in\mathbb{N}$ solves the transcendental equation $\tan z=z$.
On the intervals $(\pi k,\pi k+\pi/2)$ the function $\tan z-z$ increases from a negative number to
$+\infty$, with derivative $\tan^2z$; there are no other positive
roots. Therefore we get that
$\Lambda_{1,0} = 4\pi^2$ and $\Lambda_{2,0} = 4z_1^2$. Moreover, we notice that $4z_1^2<9\pi^2$.

Let $m\in\mathbb{Z}\setminus\{0\}$. Let us expand $f\in H^2_0(0,1)\setminus\{0\}$ into the series
$f(x) = \sum_{k=1}^\infty c_k\sqrt{2}\sin(\pi k x)$ with some real coefficients $\{c_k\}_{k\in\mathbb{N}}$.
This is a series in terms of normalized eigenfunctions of the Dirichlet Laplacian on the interval $(0,1)$. Since $f$ is a non-zero function, we can exclude the case when all coefficients $c_k$ are equal to zero. Then we obtain that
\[
 \frac{\displaystyle \int_0^1\left(-f''+\tfrac{m^2}{\rho^2}f\right)^2{\,\mathrm{d}} x}{\displaystyle\int_0^1 \left((f')^2+\tfrac{m^2}{\rho^2}f^2\right){\,\mathrm{d}} x}=
 \frac{\displaystyle\sum_{k=1}^\infty\left(\pi^2k^2+\tfrac{m^2}{\rho^2}\right)^2|c_k|^2}
        {\displaystyle\sum_{k=1}^\infty\left(\pi^2k^2+\tfrac{m^2}{\rho^2}\right)|c_k|^2}
 \geq\pi^2+\frac{1}{\rho^2}>9\pi^2.
\]
Hence, we conclude from~\eqref{eq:cylinder-dirichlet} that $\Lambda_{1,m} > 9\pi^2$ for all $m\in\mathbb{Z}\setminus\{0\}$. Thus, $\Lambda_2(C_\rho) = 4z_1^2$ and the claim follows
from the observation that $4z_1^2 < 9\pi^2 = \lambda_3(C_\rho)$.

<!-- theorem-2: body_md -->
\label{thm:hyperbolic-cylinder}
For every $\kappa<0$, there is a bounded smooth domain $\Omega$  in a two-dimensional manifold of constant Gaussian curvature $\kappa$  for which
$\Lambda_2(\Omega)<\lambda_3(\Omega)$.
<!-- end theorem-2 -->
<!-- proof-3: body_md -->
On the manifold $\mathbb R\times\mathbb S^1$ we define the metric
\[
 g_\varepsilon={\,\mathrm{d}} x^2+
       \rho^2\cosh^2(\sqrt{\varepsilon}\,x){\,\mathrm{d}}\theta^2,\qquad \varepsilon > 0,
 \]
with $\rho  = \frac{1}{10} < \frac{1}{2\sqrt{2}\pi}$.
Consider now the domain $\Omega=(-1/2,1/2)\times\mathbb S^1$.

For a metric ${\,\mathrm{d}} x^2+f(x)^2{\,\mathrm{d}}\theta^2$, the Gaussian curvature
is $K(x,\theta) = -f''(x)/f(x)$ ({\it cf.}\,~\cite[\S 4-6]{doCarmo1976}). Here $f''=\varepsilon f$, so $K=-\varepsilon$.
These metrics converge to the flat cylindrical surface metric in $C^\infty$
on $\overline\Omega$ as $\varepsilon\rightarrow0$. The strict gap in
Theorem~\ref{thm:flat-cylinder} and Lemma~\ref{lem:continuity}
imply $\Lambda_2(\Omega, g_\varepsilon)<\lambda_3(\Omega,g_\varepsilon)$
for every sufficiently small $\varepsilon>0$.

Fix one such $\varepsilon$, write $\kappa=-a$ with $a > 0$, and rescale the metric by 
\[
 \widehat g=\frac{\varepsilon}{a}g_\varepsilon.
\]
Equation~\eqref{eq:scaling} gives $K_{\widehat g}=-a=\kappa$,
while multiplying both eigenvalue sequences by $a/\varepsilon$
preserves their strict ordering.

<!-- theorem-3: body_md -->
\label{thm:hemisphere}
For every $d\geq2$,
\begin{equation}
 \Lambda_{d+1}(\mathbb S^d_+)
 \leq Q_d:=\frac{19d^2+56d+48}{7d+8}
 <3(d+2)=\lambda_{d+2}(\mathbb S^d_+).
 \label{eq:hemisphere-bound}
\end{equation}
The strictly positive variational margin is
\begin{equation}
 3(d+2)-Q_d=\frac{2d(d+5)}{7d+8}.
 \label{eq:hemisphere-margin}
\end{equation}
<!-- end theorem-3 -->
<!-- proof-4: body_md -->
We split the argument into two steps. In the first step, we compute $\lambda_{d+2}({\mathbb S}^d_+)$ and in the second step we estimate $\Lambda_{d+1}({\mathbb S}^d_+)$ from above.

\noindent {\it Step 1. Dirichlet Laplacian eigenvalues on ${\mathbb S}^d_+$.}
Odd reflection across the totally geodesic equator extends 
an eigenfunction of the Dirichlet Laplacian on ${\mathbb S}^d_+$  to an eigenfunction of the Laplacian on ${\mathbb S}^d$.
Conversely, the restrictions to ${\mathbb S}^d_+$ of the eigenfunctions of the Laplacian on ${\mathbb S}^d$ which are odd with respect to reflection $x_{d+1}\mapsto -x_{d+1}$ constitute the complete family of  the eigenfunctions of the Dirichlet Laplacian on ${\mathbb S}^d_+$.
Below we rely on the analysis of spherical harmonics in~\cite[\S 1.2.3]{LevitinMangoubiPolterovich2023}.
 Degree $\ell\in{\mathbb N}_0$ spherical harmonics correspond to the eigenvalue
$\ell(\ell+d-1)$ of the Laplacian on ${\mathbb S}^d$. Using the explicit characterization of spherical harmonics
we can single out those which are odd with respect to reflection $x_{d+1}\mapsto-x_{d+1}$.
We observe that for $\ell = 0$ there are no such harmonics, for $\ell =1$
the basis of such spherical harmonics is $\{x_{d+1}\}$, while for $\ell = 2$ the basis of such spherical harmonics is $\{x_{d+1}x_1,\ldots,x_{d+1}x_d\}$.
There are non-zero odd spherical harmonics corresponding to $\ell=3$; for instance
$x_{d+1}x_1x_2$. 
Hence
\begin{equation}
 \lambda_1({\mathbb S}^d_+)=d,\qquad
 \lambda_2({\mathbb S}^d_+)=\cdots=\lambda_{d+1}({\mathbb S}^d_+)=2(d+1),\qquad
 \lambda_{d+2}({\mathbb S}^d_+)=3(d+2).
 \label{eq:hemisphere-dirichlet}
\end{equation}

\medskip

\noindent\emph{Step 2. Trial space for the clamped buckling problem.}
Take the  trial space
\[
 E=\operatorname{span}\big\{x^2_{d+1},x^2_{d+1}x_1,\ldots,x^2_{d+1}x_d\big\}.
\]
These $d+1$ functions are clearly linearly independent.
Their values and gradients vanish on $\partial{\mathbb S}^d_+$, so
$E\subset H^2_0(\mathbb S^d_+)$. Let us denote in the following by $\Delta$ and $\nabla$, respectively, the Laplacian and the gradient on ${\mathbb S}^d$ and by $\langle\cdot,\cdot\rangle$ the inner product in the tangent space to ${\mathbb S}^d$ with respect to the round metric.
We have standard identities for $i, j\in\{1,2,\dots,d+1\}$
\[
 \Delta x_j=-dx_j,\qquad
 \langle\nabla x_i,\nabla x_j\rangle=\delta_{ij}-x_ix_j.
\]
The product rule gives
\begin{equation}
	\label{eq:sphere-t2}
\begin{aligned}
 \Delta(x_{d+1}^2)&=2-2(d+1)x^2_{d+1},\\
 \Delta(x_{d+1}^2x_i)&=  x_i[2-2(d+1)x_{d+1}^2]-dx_{d+1}^2x_i+4x_{d+1}\langle\nabla x_{d+1},\nabla x_i\rangle\\
 &=2x_i-3(d+2)x_{d+1}^2x_i,\qquad 1\leq i\leq d.
                        
\end{aligned}
\end{equation}
Reflections $x_i\mapsto-x_i$ show that both bilinear forms
$(\Delta u,\Delta v)_{L^2({\mathbb S}^d_+)}$ and $(-\Delta u,v)_{L^2({\mathbb S}^d_+)}$ are diagonal in the basis
$\{x_{d+1}^2,x_{d+1}^2x_1,\dots,x_{d+1}^2x_d\}$, that is for all $i,j\in\{1,2,\dots,d\}$ with $i\ne j$
\begin{equation}\label{eq:orth}
	\begin{aligned}
	\big(\Delta(x_{d+1}^2), \Delta(x_{d+1}^2x_i) \big)_{L^2({\mathbb S}^d_+)} &= 0,\\
	 \big(\Delta(x_{d+1}^2x_i), \Delta(x_{d+1}^2x_j) \big)_{L^2({\mathbb S}^d_+)} &= 0,\\
	 \big(\Delta(x_{d+1}^2x_i), x_{d+1}^2x_j \big)_{L^2({\mathbb S}^d_+)}&=0,\\
	 \big(\Delta(x_{d+1}^2), x_{d+1}^2x_j\big)_{L^2({\mathbb S}^d_+)} &=0.
\end{aligned}	
\end{equation}
Let us introduce the notation for the clamped buckling Rayleigh quotient
\[
	{\mathsf R}[u] := \frac{\displaystyle \int_{{\mathbb S}^d_+} (\Delta u)^2{\,\mathrm{d}} V}{\displaystyle\int_{{\mathbb S}^d_+} (-\Delta u)u{\,\mathrm{d}} V },\qquad u\in H^2_0({\mathbb S}^d_+)\setminus\{0\}.
\]
This Rayleigh quotient is deduced from~\eqref{eq:buckling-minmax} upon integration by parts in the denominator.
Permutations among the first $d$ coordinates imply that the Rayleigh quotients evaluated on the last
$d$ functions $\{x_{d+1}^2x_j\}_{j=1}^d$ are
equal; {\it i.e.}\,
\[
	{\mathsf R}[x_{d+1}^2x_i]= {\mathsf R}[x_{d+1}^2x_j],\qquad i,j\in\{1,2,\dots,d\}.
\]
We use the mapping ${\mathsf E}\colon L^1({\mathbb S}^d)\rightarrow {\mathbb R}$ defined by
\[
	{\mathsf E}[u] := \frac{\displaystyle \int_{{\mathbb S}^d}u{\,\mathrm{d}} V}{|{\mathbb S}^d|}.
\]
We stress that every integrand below is even in $x_{d+1}$, and its
hemisphere mean equals its full-sphere mean.
For nonnegative integers $\{a_j\}_{j=1}^{d+1}$, we find
\begin{equation}
\begin{aligned}
 {\mathsf E}\left[\prod_{j=1}^{d+1}x_j^{2a_j}\right] &= \frac{\displaystyle \int_{{\mathbb R}^{d+1}} e^{-|x|^2} \left(\prod_{j=1}^{d+1}x_j^{2a_j}\right){\,\mathrm{d}} x}{\displaystyle  \int_{{\mathbb R}^{d+1}} e^{-|x|^2}|x|^{\sum_{j=1}^{d+1}2a_j} {\,\mathrm{d}} x}\\
 &=\frac{\Gamma((d+1)/2)}{\Gamma((d+1)/2+\sum_{j=1}^{d+1} a_j)}
   \prod_{j=1}^{d+1}\frac{\Gamma(a_j+1/2)}{\Gamma(1/2)}.
\end{aligned}
 \label{eq:sphere-moments}
\end{equation}
In particular, for any $i\in\{1,2,\dots,d\}$
\begin{equation}\label{eq:E}
\begin{gathered}
 {\mathsf E}[x_{d+1}^2]=\frac1{d+1},\qquad
 {\mathsf E}[x_{d+1}^4]=\frac3{(d+1)(d+3)},\qquad
 {\mathsf E}[x_i^2]=\frac1{d+1},\\
 {\mathsf E}[x_{d+1}^2x_i^2]=\frac1{(d+1)(d+3)},\qquad
 {\mathsf E}[x_{d+1}^4x_i^2]=\frac3{(d+1)(d+3)(d+5)}.
\end{gathered}
\end{equation}
By \eqref{eq:sphere-t2} combined with~\eqref{eq:E} we obtain
\[
\begin{aligned}
 {\mathsf E}[x_{d+1}^2(-\Delta(x_{d+1}^2))]&=2(d+1){\mathsf E}[x_{d+1}^4]-2{\mathsf E}[x_{d+1}^2]
       =\frac{4d}{(d+1)(d+3)},\\
 {\mathsf E}[(\Delta(x_{d+1}^2))^2]&=4(d+1)^2{\mathsf E}[x_{d+1}^4]-8(d+1){\mathsf E}[x_{d+1}^2]+4=\frac{8d}{d+3}.
\end{aligned}
\]
Thus, we get
\begin{equation}\label{eq:R1}
	{\mathsf R}[x_{d+1}^2] = \frac{{\mathsf E}[(\Delta(x_{d+1}^2))^2]}{{\mathsf E}[x_{d+1}^2(-\Delta(x_{d+1}^2))]} = 2(d+1).
\end{equation}
 
                                          
Analogously, using \eqref{eq:sphere-t2} combined with~\eqref{eq:E} we get for $i\in\{1,2,\dots,d\}$
\[
\begin{aligned}
 {\mathsf E}[x_{d+1}^2x_i(-\Delta(x_{d+1}^2x_i))]&=3(d+2){\mathsf E}[x_{d+1}^4x_i^2]-2{\mathsf E}[x_{d+1}^2x_i^2]\\
   &=\frac{7d+8}{(d+1)(d+3)(d+5)},\\
 {\mathsf E}[(\Delta(x_{d+1}^2x_i))^2]&=9(d+2)^2{\mathsf E}[x_{d+1}^4x_i^2]-12(d+2){\mathsf E}[x_{d+1}^2x_i^2]+4{\mathsf E}[x_i^2]\\
 &=\frac{19d^2+56d+48}{(d+1)(d+3)(d+5)}.
\end{aligned}
\]
Thus, we obtain that  for all $i\in\{1,2,\dots,d\}$
\begin{equation}\label{eq:R2}
	{\mathsf R}[x_{d+1}^2x_i] = \frac{{\mathsf E}[(\Delta(x_{d+1}^2x_i))^2]}{{\mathsf E}[x_{d+1}^2x_i(-\Delta(x_{d+1}^2x_i))]}
	= \frac{19d^2+56d+48}{7d+8}= Q_d.
\end{equation}
Moreover, we observe that
\[
 Q_d-2(d+1)=\frac{5d^2+26d+32}{7d+8}>0.
\]
We conclude from~\eqref{eq:R1},~\eqref{eq:R2} combined with~\eqref{eq:orth} that
\[
	\max_{u\in E\setminus\{0\}}{\mathsf R}[u]=Q_d.
\]
The min-max principle gives the first inequality in \eqref{eq:hemisphere-bound};
subtracting $Q_d$ from $3(d+2)$ gives \eqref{eq:hemisphere-margin}.
