<!-- proposition-1: body_md -->
\label{prop:Emui_basis_single_site}
An orthogonal\footnote{We use the Hilbert–Schmidt inner product for linear operators: $\langle A,B\rangle_{\text{HS}}=\text{Tr}(A^\dagger B)$ and denote by $\text{Tr}_i$ the trace restricted to edge $i$.} basis of errors acting on each edge $i$ is
\begin{equation}\label{eq:Emui}
E_i^{h,\mu}:=L_i^hE_i^\mu\,,\qquad E_i^\mu:=\sqrt{d_\Gamma}\,Z^\Gamma_{ab,i}\,.
\end{equation}
Here $h\in G$ is a group element with $L^h$ defined in \eqref{eq:LR_def} while $\mu=(\Gamma,a,b)$, with $\Gamma\in\text{Irr}(G)$ and $1\le a,b\le d_\Gamma$, labels the irrep operators defined in \eqref{eq:ZGamma_def}. They satisfy
\begin{equation}\label{eq:Emunu_delta}
\text{Tr}_i\!\left[(E_i^{h,\mu})^\dagger E_i^{k,\nu}\right]=|G|\,\delta_{h,k}\delta_{\mu,\nu}\,.
\end{equation}
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Since $L_i^g$ has diagonal entries which are all zero for $g\neq\text{id}$, Schur orthogonality of the irreps \cite{Serre:1977} then gives
\begin{equation}\label{eq:setup-10}\begin{aligned}
\text{Tr}_i\!\left[(L_i^hZ^\Gamma_{ab,i})^\dagger L_i^kZ^\Lambda_{cd,i}\right]&=\delta_{h,k}\sum_{g}\Gamma(g)_{ab}^*\Lambda(g)_{cd}\cr
&=\delta_{h,k}\frac{|G|}{d_\Gamma}\delta_{\Gamma\Lambda}\delta_{a,c}\delta_{b,d}\,,
\end{aligned}\end{equation}
which, using the normalization of \eqref{eq:Emui}, is \eqref{eq:Emunu_delta}. Therefore, these $|G|\sum_\Gamma d_\Gamma^2=|G|^2$ orthogonal operators form a basis of the operators on each edge of the lattice.
<!-- end proof-1 -->
<!-- lemma-1: body_md -->
\label{lemma:PBEPB}
Let $p\in p(i)$, and suppose that only the edge $i\in\partial p$ carries a non-trivial multiplication error $L_i^h$. Then, for every $h\neq\text{id}$,
\begin{equation}
B_pL_i^hB_p=0\,,\qquad B_pE_i^{h,\mu}B_p=0\,.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
For a bulk plaquette with edges labeled as in \eqref{eq:AvBp}, $B_p$ imposes the trivial flux $\text{id}=g_1g_2g_3^{-1}g_4^{-1}$. The error $L_i^h$ changes the flux depending on the site index $i$ as follows:
\begin{equation}\begin{aligned}\label{eq:L_flux_change}
L_1^h:\ \text{id}&\mapsto(hg_1)g_2g_3^{-1}g_4^{-1}=h\,,\cr
L_2^h:\ \text{id}&\mapsto g_1(hg_2)g_3^{-1}g_4^{-1}=g_1hg_1^{-1}\,,\cr
L_3^h:\ \text{id}&\mapsto g_1g_2(hg_3)^{-1}g_4^{-1}=g_4h^{-1}g_4^{-1}\,,\cr
L_4^h:\ \text{id}&\mapsto g_1g_2g_3^{-1}(hg_4)^{-1}=h^{-1}\,.
\end{aligned}\end{equation}
The flux of the plaquette $p\in p(i)$ thus lies in the conjugacy classes $[h]$ or $[h^{-1}]$, which are both distinct from $[\text{id}]=\{\text{id}\}$ for $h\neq\text{id}$, so $B_pL_i^hB_p=0$. Finally $Z^\Gamma_{ab,i}$ is diagonal in the group basis and preserves each flux sector, so $B_pE_i^{h,\mu}B_p=0$.
<!-- end proof-2 -->
<!-- proposition-2: body_md -->
\label{eq:prop_Avmu}
The generalized vertex operators
\begin{equation}\label{eq:Avmu_def}
A_v^{(\Gamma,a,b)}:=\frac{d_\Gamma}{|G|}\sum_{g\in G}\Gamma(g)_{ab}A_v^g\,,
\end{equation}
satisfy the matrix-unit algebra
\begin{equation}\label{eq:Avmu_comm}
A_v^{(\Gamma,a,b)}A_v^{(\Lambda,c,d)}=\delta_{\Gamma,\Lambda}\delta_{b,c}A_v^{(\Gamma,a,d)}\,,
\end{equation}
and
\begin{equation}\label{eq:Avmu-adjoint}
\bigl(A_v^{(\Gamma,a,b)}\bigr)^\dagger=A_v^{(\Gamma,b,a)}\,.
\end{equation}
<!-- end proposition-2 -->
<!-- lemma-2: body_md -->
\label{lemma:vertex-violations}
Let $v\in v(i)$, and suppose that $i$ is the only edge of $\star v$ carrying an error. Then, for every $\mu=(\Gamma,a,b)$ with $\Gamma\neq1$,
\begin{equation}\label{eq:AvEAv}
A_vE_i^\mu A_v=0\,,
\end{equation}
so every nontrivial vertex constraint in $v(i)$ is violated.
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
If $i$ is oriented away from $v$, then $A_v^g$ acts by $L_i^g$, and \eqref{eq:ZL_comm} gives
\begin{equation}
E_i^{(\Gamma,a,b)}A_v=\frac{1}{d_\Gamma}\sum_{c=1}^{d_\Gamma}A_v^{(\Gamma,a,c)}E_i^{(\Gamma,c,b)}\,.
\end{equation}
while if $i$ is oriented towards $v$, $A_v^g$ acts on $i$ by $R_i^g$, and \eqref{eq:ZR_comm} gives
\begin{equation}
E_i^{(\Gamma,a,b)}A_v=\frac{1}{d_\Gamma}\sum_{c=1}^{d_\Gamma}A_v^{(\Gamma^{*},b,c)}E_i^{(\Gamma,a,c)}\,,
\end{equation}
In either case, left multiplication by $A_v=A_v^{(1,1,1)}$ annihilates the right-hand side for $\Gamma\neq1$ by the matrix-unit algebra \eqref{eq:Avmu_comm}.
<!-- end proof-3 -->

<!-- theorem-2: body_md -->
\label{thm:Knill-Laflamme}
Let $C$ be a quantum code with code-space projector $P$, and let $\mathcal E$ be a quantum operation with Kraus operators $\{\widetilde{E}_\mu\}$. A necessary and sufficient condition for the existence of a recovery operation $\mathcal R$ correcting $\mathcal E$ on $C$ is
\begin{equation}\label{eq:Knill-Laflamme}
P\widetilde{E}_\mu^\dagger\widetilde{E}_\nu P=\alpha_{\mu\nu}P\,.
\end{equation}
Here $\alpha=(\alpha_{\mu\nu})$ is positive semidefinite and Hermitian.
<!-- end theorem-2 -->
<!-- proof-6: body_md -->
Since $\alpha$ is Hermitian, it is diagonalizable by a unitary $u$, such that
\begin{equation}
u^\dagger\alpha u=d\,,
\end{equation}
with $d$ diagonal. The conditions \eqref{eq:Knill-Laflamme} for the rotated Kraus operators
\begin{equation}
E_\kappa=\sum_\mu u_{\mu\kappa}\widetilde{E}_\mu\,,
\end{equation}
then become
\begin{equation}\label{eq:E_orth}
P E_\kappa^\dagger E_\lambda P=d_{\kappa\kappa}\delta_{\kappa\lambda}P\,.
\end{equation}
Restricting $\kappa$ to the indices such that $d_{\kappa\kappa}>0$, and for each such $\kappa$ the error syndrome projector are defined as
\begin{equation} \label{eq:KL-diag}
P_\kappa=\frac{E_\kappa P E_\kappa^\dagger}{d_{\kappa\kappa}}\,.
\end{equation}
By \eqref{eq:E_orth} the $P_\kappa$ are mutually orthogonal projectors onto the error subspaces and $E_\kappa/\sqrt{d_{\kappa\kappa}}$ is an isometry on the code space, which extends to a unitary $U_\kappa$ on the full Hilbert space\footnote{$U_\kappa$ exists by the matrix polar decomposition of $E_\kappa P$ and \eqref{eq:E_orth}.} such that:
\begin{equation}\label{eq:EP_UP}
E_\kappa P=\sqrt{d_{\kappa\kappa}}\,U_\kappa P\,.
\end{equation}
The recovery channel measures the syndromes $\{P_\kappa\}$ and applies the corresponding $U_\kappa^\dagger$,
\begin{equation}\begin{aligned}\label{eq:recovery-channel}
\mathcal R(\sigma)&=\sum_{\kappa} U_\kappa^\dagger P_\kappa\sigma P_\kappa U_\kappa+P_\perp\sigma P_\perp\,,\cr
P_\perp&:=\mathbb{I}-\sum_\kappa P_\kappa\,.
\end{aligned}\end{equation}
The complementary projector $P_\perp$ vanishes on the corrupted code space and completes $\mathcal R$ to a trace-preserving channel.
{In the following when we refer to projective measurement, we imply that such a complementary term can always be added.}
                                
                                                                                       
     
Then for every code state $\rho=P\rho P$,\footnote{Using \eqref{eq:E_orth}-\eqref{eq:EP_UP}, one has:
\begin{equation}
U_\kappa^\dagger P_\kappa E_\lambda P=\sqrt{d_{\kappa\kappa}}\,\delta_{\kappa\lambda}P\,.
\end{equation}
}
the recovery channel is
\begin{equation}\begin{aligned}
\mathcal R(\mathcal E(\rho))&=\sum_{\kappa,\lambda} U_\kappa^\dagger P_\kappa E_\lambda\rho E_\lambda^\dagger P_\kappa U_\kappa=\left(\sum_\kappa d_{\kappa\kappa}\right)\rho\,.
\end{aligned}\end{equation}
The factor $\sum_\kappa d_{\kappa\kappa}=\text{Tr}[\mathcal E(\rho)]$ is independent of $\rho$ and any trace-preserving error channel $\mathcal E$ is corrected exactly. Necessity is proven in e.g. \cite[Theorem~10.1]{Nielsen:2012yss}.

The operator $\sum_{c=1}^{d_\Gamma}Z^\Gamma_{ac,i}Z^\Gamma_{cb,\mathrm{va}}$ is diagonal in the group basis, with eigenvalue
\begin{equation}\begin{aligned} 
\label{O-eigenvalue}
&\Bigl(\sum_{c=1}^{d_\Gamma}Z^\Gamma_{ac,i}Z^\Gamma_{cb,\mathrm{va}}\Bigr)\ket x_i\ket y_{\mathrm{va}}\cr 
&=\sum_{c=1}^{d_\Gamma}\Gamma(x)_{ac}\Gamma(y)_{cb}\ket x_i\ket y_{\mathrm{va}}\cr 
&=\Gamma(xy)_{ab}\ket x_i\ket y_{\mathrm{va}}\,.
\end{aligned}\end{equation}
Conjugation by $CL_{i,\mathrm{va}}$ therefore removes the dependence on $x$,
\begin{equation}
\label{CL-conjugation}
\begin{aligned} 
&CL_{i,\mathrm{va}}\Bigl(\sum_{c=1}^{d_\Gamma}Z^\Gamma_{ac,i}Z^\Gamma_{cb,\mathrm{va}}\Bigr)CL_{i,\mathrm{va}}^\dagger\ket x_i\ket y_{\mathrm{va}}\cr 
&=CL_{i,\mathrm{va}}\Bigl(\sum_{c=1}^{d_\Gamma}Z^\Gamma_{ac,i}Z^\Gamma_{cb,\mathrm{va}}\Bigr)\ket x_i\ket{x^{-1}y}_{\mathrm{va}}\cr 
&=\Gamma(y)_{ab}\ket x_i\ket y_{\mathrm{va}}\,.
\end{aligned}
\end{equation}

For the second identity, the operator $\sum_{c=1}^{d_\Gamma}Z^\Gamma_{cb,i}Z^{\Gamma^*}_{ca,\mathrm{va}}$ is diagonal in the group basis, which acts on the state $\ket x_i\ket y_{\mathrm{va}}$ with  eigenvalue 
\begin{equation} 
\sum_{c=1}^{d_\Gamma}\Gamma(y)^*_{ca}\Gamma(x)_{cb}=\sum_{c=1}^{d_\Gamma}\Gamma(y^{-1})_{ac}\Gamma(x)_{cb}=\Gamma(y^{-1}x)_{ab} \,,
\end{equation} 
where we used that $\Gamma(y)$ is unitary. 
Since $CL_{i,\mathrm{va}}\ket x_i\ket y_{\mathrm{va}}=\ket x_i\ket{xy}_{\mathrm{va}}$, conjugation by $CL_{i,\mathrm{va}}^\dagger$ gives
\begin{equation}
\label{CL-conjugation-dagger}
\begin{aligned}
&CL^\dagger_{i,\mathrm{va}}\Bigl(\sum_{c=1}^{d_\Gamma}Z^\Gamma_{cb,i}Z^{\Gamma^*}_{ca,\mathrm{va}}\Bigr)CL_{i,\mathrm{va}}\ket x_i\ket y_{\mathrm{va}}\cr
&=\Gamma\bigl((xy)^{-1}x\bigr)_{ab}\ket x_i\ket y_{\mathrm{va}}=\Gamma(y^{-1})_{ab}\ket x_i\ket y_{\mathrm{va}}\,.
\end{aligned}
\end{equation}
Thus, $\Gamma(y^{-1})_{ab}=\Gamma(y)^*_{ba}=\Gamma^*(y)_{ba}$ is the eigenvalue of $Z^{\Gamma^*}_{ba,\mathrm{va}}$, which proves the second identity.
