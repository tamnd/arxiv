If $\psi$ is a Dirichlet character, we say that a pair $(\theta, A)\in [0, 1/4]\times [0, \infty)$ is $\psi$-\emph{admissible} if for all $t \in \mathbb{R}$ and all primitive quadratic characters $\chi$ we have 
\begin{equation}\label{eq:defipsiadmissible}
    |L(1/2 + it, \chi \psi)| \ll_{\psi, \varepsilon} (1 + |t|)^{A + \varepsilon} \mathfrak{q}(\chi)^{\theta + \varepsilon},
\end{equation}
where $\mathfrak{q}(\chi)$ is the conductor of $\chi$. We say that $\theta \in [0, 1/4]$ is admissible if for any $\psi$ there is some $A \geq 0$, possibly depending on $\psi$, such that the pair $(\theta, A)$ is $\psi$-admissible.

<!-- thm-1: body_md -->
\label{thm:maintheorem}
For any $q \geq 1$, for any class function $\varphi$ on $\widetilde{\Gamma}$, and for any admissible exponent $\theta$ we have 
\begin{equation*}
    \Psi_{\Gamma}(x; \varphi) =  \langle \varphi, \mathbbm{1}\rangle x + O_{\varphi, \varepsilon}(x^{\eta(\theta) + \varepsilon}),
\end{equation*} 
where 
\begin{equation*}
    \langle \varphi, \mathbbm{1}\rangle := \frac{1}{|\widetilde{\Gamma}|}\sum_{\gamma \in \widetilde{\Gamma}}\varphi(\gamma)
\end{equation*}
and
\begin{equation}\label{eq:definitionexponent}
    \eta(y) := 1/2 + \frac{\sqrt{81y^2 + 106y + 33}-5 - 9y}{4}.
\end{equation}

<!-- cor-1: body_md -->
\label{cor:pgtforcongruencesubgroups}
Let $\Gamma\subset \operatorname{SL}_2(\mathbb{R})$ be a congruence subgroup of an indefinite quaternion algebra over $\mathbb{Q}$. Then, for any admissible exponent $\theta$ we have
\begin{equation*}
    \Psi_\Gamma(x) = x + O_\varepsilon(x^{\eta(\theta) + \varepsilon}).
\end{equation*}

<!-- prop-1: body_md -->
\label{prop:myzerodensity}
For any Dirichlet character $\psi$ and any $\psi$-admissible pair $(\theta, A)$ we have
\begin{equation*}
    \sum_{\chi \in S(Q)} N(\sigma, T, \chi\psi) \ll_{\psi, \varepsilon} (QT)^\varepsilon Q^{2(1 + 2\theta)(1-\sigma)}T^{1 + 4A(1-\sigma)}.
\end{equation*}
<!-- end prop-1 -->
<!-- proof-1: body_md -->
In this proof the character $\chi\psi$ is given by $n \mapsto \chi(n)\psi(n)$ and is not necessarily primitive. This is not an issue since $L(s, \chi\psi)$ and $L(s, (\chi\psi)^*)$ have the same zeros in the region $\text{Re}(s) > 0$, where $(\chi\psi)^*$ is the primitive character corresponding to $\chi\psi$. To prove Proposition \ref{prop:myzerodensity} we will follow \cite[Section 11]{heathbrownquadlargesieve} very closely. First of all, note that we can assume that $\sigma \geq 1/2 + \theta/(1 + 2\theta) $, since otherwise Proposition \ref{prop:myzerodensity} follows from the Riemann--von Mangoldt formula \cite[Theorem 5.8]{iwanieckowalski}. Also, since $\psi$ is fixed, we can assume that $\chi\psi$ is nontrivial by ignoring the contribution of at most a single $\chi$.  Define
\begin{equation*}
\begin{aligned}
    \mathcal{R}(\sigma, \tau) & := [\sigma, \sigma + (\log QT)^{-1}] \times [\tau, \tau + (\log QT)^{-1}],\\
    N'(\sigma, \tau) & := \#\{\chi \in S(Q): L(s, \chi \psi) \text{ has a zero in } \mathcal{R}(\sigma, \tau)\}.
\end{aligned}
\end{equation*}
By subdividing the region $[\sigma, 1]\times [-T, T]$ into small rectangles and summing the different contributions trivially, we see that in order to prove Proposition \ref{prop:myzerodensity} it is enough to show that  
\begin{equation}\label{eq:tobeproved}
    N'(\sigma, \tau) \ll_{\psi, \varepsilon} (QT)^\varepsilon Q^{2(1 + 2\theta)(1-\sigma)}T^{4A(1-\sigma)}
\end{equation}
holds for all $(\sigma, \tau) \in [1/2 + \theta/(1 + 2\theta), 1]\times [-T, T]$. Let $100 \leq X \leq Y \leq (QT)^{100}$ be large parameters and consider 
\begin{equation}\label{eq:inversedirpoly}
    M_X(s,\chi\psi) := \sum_{n \leq X}\frac{\mu(n)\chi(n)\psi(n)}{n^s}.
\end{equation}
If $\rho = \beta + i\gamma \in \mathcal{R}(\sigma, \tau) $ is a zero of $L(s, \chi\psi)$, then a routine argument \cite[Equation (12.24)]{montgomerytopics} using Mellin inversion shows that 
\begin{equation}\label{eq:auxeq}
    e^{-1/Y} + \sum_{n > X}\frac{a(n)\chi(n)\psi(n)}{n^\rho}e^{-n/Y} = \frac{1}{2\pi i}\int_{(1/2 - \text{Re}(\rho))} L(s + \rho, \chi\psi)M_X(s + \rho, \chi\psi)\Gamma(s)Y^s \, ds,
\end{equation}
where 
\begin{equation*}
    a(n) := \sum_{\substack{d \mid n\\ d \leq X}} \mu(d), \quad \text{ so that }\quad |a(n)| \leq \tau(n).
\end{equation*}
Let $\mathcal{U} := \{2^i X: 0 \leq i \leq \log_2(Y^2/X)\}$. Note that $|\mathcal{U}| \asymp \log Y$. Equation \eqref{eq:auxeq} implies that either we have
\begin{equation*}
\left|\int_{-(QT)^\varepsilon}^{(QT)^\varepsilon} L(1/2 + i\gamma + iu, \chi \psi)M_X(1/2 + i\gamma + iu, \chi \psi)Y^{1/2 - \beta + iu}\Gamma(1/2 - \beta + iu)\, du\right| \gg 1,
\end{equation*} 
or else there is $U \in \mathcal{U}$ such that 
\begin{equation*}
    \left|\sum_{U< n \leq 2U}\frac{a(n)\chi(n)\psi(n)}{n^\rho}e^{-n/Y}\right| \gg |\mathcal{U}|^{-1}
\end{equation*}
Therefore, using Equation \eqref{eq:auxeq} for each $\chi \in S(Q)$ such that $L(s, \chi \psi)$ has a zero in $\mathcal{R}(\sigma, \tau)$ we arrive at two different cases:
\begin{enumerate}
    \item [(1)] There are $\gg N'(\sigma, \tau)$ characters $\chi \in S(Q)$ for which
\begin{equation}\label{eq:firstcase}
    \left|\int_{-(QT)^\varepsilon}^{(QT)^\varepsilon} L(1/2 + i\gamma_\chi + iu, \chi \psi)M_X(1/2 + i\gamma_\chi + iu, \chi \psi)Y^{1/2 - \beta_\chi + iu}\Gamma(1/2 - \beta_\chi + iu)\, du\right| \gg 1
\end{equation}
holds for some $\rho_\chi = \beta_\chi + i\gamma_\chi \in \mathcal{R}(\sigma, \tau)$.
    \item [(2)] Otherwise, there is an integer $U \in \mathcal{U}$ and $\gg N'(\sigma, \tau)|\mathcal{U}|^{-1}$ characters $\chi \in S(Q)$ for which
\begin{equation*}
    \left|\sum_{U< n \leq 2U}\frac{a(n)\chi(n)\psi(n)}{n^{\rho_\chi}}e^{-n/Y}\right| \gg |\mathcal{U}|^{-1}
\end{equation*}
holds for some $\rho_\chi = \beta_\chi + i\gamma_\chi \in \mathcal{R}(\sigma, \tau)$.
\end{enumerate}
We can deal with the second case exactly as in Heath-Brown, obtaining  
\begin{equation*}
    N'(\sigma, \tau) \ll_{\psi, \varepsilon} (QT)^\varepsilon (QX^{1-2\sigma} + Y^{2-2\sigma})
\end{equation*}
in that case. Our proof differs from \cite[Section 11]{heathbrownquadlargesieve} in the treatment of the first case. Squaring \eqref{eq:firstcase}, applying Cauchy-Schwarz and the subconvexity bound from \eqref{eq:defipsiadmissible} we deduce that 
\begin{equation}\label{eq:auxcase2intermediate}
    N'(\sigma, \tau) \ll_{\psi, \varepsilon} (QT)^{\varepsilon} Q^{2\theta}T^{2A} Y^{1-2\sigma}\int_{\tau -(QT)^\varepsilon}^{\tau + (QT)^\varepsilon} \sum_{\chi \in S(Q)}|M_X(1/2 + iu, \chi\psi)|^2 \, du.
\end{equation}
We can write 
\begin{equation*}
    M_X(s, \chi\psi) = \sum_{i = 0}^{\lfloor \log_2 X\rfloor} M_{X, i}(s, \chi\psi)
\end{equation*}
where 
\begin{equation*}
    M_{X, i}(s, \chi\psi) := \begin{dcases}
        \sum_{2^i \leq n < 2^{i + 1}}\frac{\mu(n)\chi(n)\psi(n)}{n^s}, & 0 \leq i < \lfloor \log_2 X\rfloor,\\
        \sum_{2^i \leq n \leq X}\frac{\mu(n)\chi(n)\psi(n)}{n^s}, & i = \lfloor \log_2 X\rfloor.
    \end{dcases}
\end{equation*}
It follows from \eqref{eq:auxcase2intermediate} that 
\begin{equation*}
    N'(\sigma, \tau) \ll_{\psi, \varepsilon} (QT)^{\varepsilon} Q^{2\theta}T^{2A} Y^{1-2\sigma}\sum_{i = 0}^{\lfloor \log_2 X\rfloor}\int_{\tau -(QT)^\varepsilon}^{\tau + (QT)^\varepsilon} \sum_{\chi \in S(Q)}|M_{X, i}(1/2 + iu, \chi\psi)|^2 \, du,
\end{equation*}
and applying the large sieve inequality \eqref{eq:quadlargesieve} we deduce that
\begin{equation*}
    N'(\sigma, \tau) \ll_{\psi, \varepsilon} (QT)^{\varepsilon} Q^{2\theta}T^{2A} (Q + X)Y^{1 - 2\sigma}.
\end{equation*}
It follows that, in any case, we have 
\begin{equation*}
    N'(\sigma, \tau)  \ll_{\psi, \varepsilon} (QT)^\varepsilon \left(Q^{2\theta}T^{2A}(Q + X)Y^{1-2\sigma} + QX^{1-2\sigma} + Y^{2-2\sigma}\right). 
\end{equation*}
If we let $X = Q$ and $Y = Q^{1 + 2\theta}T^{2A}$ we arrive at \eqref{eq:tobeproved}, which concludes the proof of Proposition \ref{prop:myzerodensity}.

<!-- lem-1: body_md -->
\label{lem:numberofrepeatedfundamentaldiscriminants}
For each fundamental discriminant $D > 0$ we have  
\begin{equation*}
    \#\{3 \leq n \leq X : n^2 - 4 = Df^2 \text{ for some }f\} \ll \log X,
\end{equation*}
where the implied constant is independent in $D$.
<!-- end lem-1 -->
<!-- proof-2: body_md -->
Let $K$ be the field $Q(\sqrt{D})$, with ring of integers $\mathcal{O}_K$. We consider $K$ embedded in $\mathbb{R}$ so that $\sqrt{D} > 0$. By Dirichlet's unit theorem we know that $\mathcal{O}_K^1 = \{\pm z_K^m: m \in \mathbb{Z}\}$ for a unique $z_K$ satisfying $z_K > 1$. It is easy to see that we have $z_K \geq (3 + \sqrt{5})/2$. If $n \geq 3$ is such that $n^2 - 4 = Df^2$ for some integer $f$, then the element $z_n := (n + \sqrt{n^2 - 4})/2$ belongs to $\mathcal{O}_K^1$ and satisfies $z_n > 1$, so we can write $z_n = z_K^m$ for some $m \geq 1$. It follows that 
\begin{equation*}
    \#\{3 \leq n \leq X : n^2 - 4 = Df^2 \text{ for some }f\} \ll \frac{\log X}{\log z_K} \ll \log X,
\end{equation*}
as desired.

<!-- lem-2: body_md -->
\label{lem:lindelof}
If $L(s, \psi_D\psi)$ has no zero in the rectangle $[\sigma, 1]\times [-T, T]$, then 
\begin{equation*}
    L(s, \psi_D\psi) \ll_{q, \varepsilon} (DT)^\varepsilon
\end{equation*}
for $s$ such that $\text{Re}(s) > \sigma + \varepsilon$, $|\text{Im}(s)| < T - 1$ and $|s-1| \geq \varepsilon$.
<!-- end lem-2 -->
<!-- proof-3: body_md -->
This is well-known, see for example \cite[Lemma 1]{bykovskiiprimegeodesicshortintervals}.

<!-- lem-3: body_md -->
\label{lem:mostzetasatisfylindelof}
Let $1/2 \leq \sigma \leq 1$ and $x \geq 1$. For any $\varepsilon > 0$ there is a set $\mathcal{B}_\sigma \subset (x^{1/2}, 2x^{1/2}]$ of cardinality $|\mathcal{B}_\sigma| \ll_\varepsilon x^{2(1 + 2\theta)(1-\sigma) + \varepsilon}$ such that the bound 
\begin{equation}\label{eq:lindelofsigma}
    Z(s; n) \ll_\varepsilon x^\varepsilon
\end{equation}
holds uniformly for $n \in (x^{1/2}, 2x^{1/2}] \setminus \mathcal{B}_\sigma$ and all $s$ such that $\text{Re}(s) \geq \sigma$, $|\text{Im}(s)|\leq x^\varepsilon$ and $|s-1| \geq \varepsilon$.
<!-- end lem-3 -->
<!-- proof-4: body_md -->
The lemma follows by combining Proposition \ref{prop:myzerodensity}, identities \eqref{eq:zetaintermsofsignedzetas} and \eqref{eq:expressionforgeneralzagierzeta}, the bounds in \eqref{eq:boundsbadprimes}, and Lemmas \ref{lem:numberofrepeatedfundamentaldiscriminants} and \ref{lem:lindelof}.

<!-- prop-2: body_md -->
\label{prop:smoothingsum}
We have 
\begin{equation}\label{eq:smoothingtobeproved}
    \Psi_\Gamma(x; \chi) = \delta_{\chi = 1}x - \int_0^Y (\Psi_\Gamma(x + u; \chi) - \Psi_\Gamma(x; \chi) - \delta_{\chi = 1}u)\omega_Y(u) \, du + O_\varepsilon(x^{7/8 + \varepsilon}Y^{-1/4} + \delta_{k = 0}x^{1/2 + 7/64}). 
\end{equation}

<!-- prop-3: body_md -->
\label{prop:mylocalinterval}
Let $\theta$ be an admissible exponent. For $1 \leq V \leq x$, $1/2 \leq \sigma \leq 1$ and $x^{1/2 + \varepsilon} \leq u \leq x^{1-\varepsilon}$ we have 
\begin{equation*}
    \Psi_\Gamma(x + u; \chi) - \Psi_\Gamma(x; \chi) = \delta_{\chi = 1}u + O_\varepsilon\left(x^{1/2 + \varepsilon}\left(x^{2(1 + 2\theta)(1-\sigma) + \theta}V^{-1/2} + V^{1/2} + ux^{-1/2}V^{\sigma - 1}\right)\right).
\end{equation*}
<!-- end prop-3 -->
<!-- proof-6: body_md -->
Let $\omega \in C_c^\infty(0, \infty)$ satisfy $\int_\mathbb{R} \omega(x) \, dx = 1$. Let $\widehat{\omega}$ denote the Mellin transform of $\omega$. Note that $|\widehat{\omega}(s)| \ll_B (1 + |s|)^{-B}$ for arbitrary $B > 0$ on any fixed vertical strip. Consider 
\begin{equation}\label{eq:approximationtogeometricsideshortintervals}
    \mathcal{S}_V := \frac{\operatorname{Vol}(\Gamma\backslash \mathbb{H})\pi}{2V}\sum_{c\geq 1}\sum_{X < n \leq X'} (n^2 - 4)^{1/2} a(c; n)\omega(c \ell/V)
\end{equation}
with $X = \sqrt{x} + 1/\sqrt{x}$ and $X' = \sqrt{x + u} + 1/\sqrt{x + u}$.
We know from \cite[Equation (146)]{reche2026chebotarevgeodesictheoremsplit} that 
\begin{equation}\label{eq:auxevaluationapprox}
    \mathcal{S}_V = \delta_{\chi = 1}u + O_{q, \varepsilon}(V^{1/2 + \varepsilon}x^{1/2} + uV^{-1}).
\end{equation}
If we execute first the summation over $c$ in \eqref{eq:approximationtogeometricsideshortintervals}, apply Mellin inversion formula, move the contour to the left, pick up the residues at $s = 1$ and use \eqref{eq:pgtsumofresidues} we obtain
\begin{equation}\label{eq:auxeeqaftermellin}
    \mathcal{S}_V = \Psi_\Gamma(x + u; \chi) - \Psi_\Gamma(x; \chi) + \frac{\operatorname{Vol}(\Gamma\backslash \mathbb{H})\pi}{2} \sum_{X < n \leq X'}(n^2 - 4)^{1/2}I_V(n^2 - 4),
\end{equation}
where 
\begin{equation*}
    I_V(n^2 - 4) := \frac{1}{2\pi i}\int_{(\eta)} Z(s; n)\widehat{\omega}(s)V^{s-1}\, ds
\end{equation*}
with $1/2 \leq \eta < 1$. We bound the sum of integrals by applying Lemma \ref{lem:mostzetasatisfylindelof} and distinguishing two cases:
\begin{enumerate}
    \item [i)] If $n \in \mathcal{B}_\sigma$, we let $\eta = 1/2$ and use \eqref{eq:subconvexitybykovskii} to obtain 
    \begin{equation*}
        I_V(n^2 - 4) \ll_\varepsilon x^{\theta + \varepsilon} V^{-1/2}.
    \end{equation*}
    \item [ii)] If $n \in (X, X']\setminus \mathcal{B}_\sigma$ we let $\eta = \sigma + \varepsilon$ and use the bound in \eqref{eq:lindelofsigma} and the rapid decay of $\widehat{\omega}(s)$ to deduce 
    \begin{equation*}
        I_V(n^2 - 4) \ll_\varepsilon x^{\varepsilon} V^{\sigma - 1}.
    \end{equation*}
\end{enumerate}
We know that $|\mathcal{B}_\sigma| \ll x^{2(1+2\theta)(1-\sigma) + \varepsilon}$ and $X' - X \asymp ux^{-1/2}$. Summing over $n$ we get 
\begin{equation}\label{eq:auxeqboundsumintegrals}
    \sum_{X < n \leq X'}(n^2 - 4)^{1/2}I_V(n^2 - 4) \ll_\varepsilon x^{1/2 + \varepsilon}\left( x^{2(1 + 2\theta)(1-\sigma) + \theta} V^{-1/2} + ux^{-1/2}V^{\sigma - 1}\right). 
\end{equation}
The proof of Proposition \ref{prop:mylocalinterval} is concluded by combining \eqref{eq:auxevaluationapprox}, \eqref{eq:auxeeqaftermellin} and \eqref{eq:auxeqboundsumintegrals}.
