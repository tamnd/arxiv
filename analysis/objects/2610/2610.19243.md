<!-- mainthm-1: body_md -->
\label{thm:classification}
Let $n\geq1$, and let $f$ be meromorphic on $\mathbb C$. If $f$ and $f^{(n)}$
are algebraically dependent over $\mathbb C$, then either $f\in\mathcal W$, or
\begin{equation}\label{eq:affine}
 f^{(n)}=af+b\qquad\text{for some }a,b\in\mathbb C.
\end{equation}
In particular, every meromorphic solution of an irreducible equation
\eqref{eq:BB} with $\deg P\geq2$ belongs to $\mathcal W$.
<!-- end mainthm-1 -->
<!-- mainthm-2: body_md -->
\label{thm:composition}
Let $n\geq1$ and let $R,S\in\mathbb C[T]$ have the same degree $d\geq2$.
Suppose that $h$ is nonconstant and entire and
\begin{equation}\label{eq:RS}
 \mathrm D^nR(h)=S(h).
\end{equation}
Then
\begin{equation}\label{eq:h-classified}
 h(z)=Ae^{bz}+Be^{-bz}+C
\end{equation}
for some $A,B,C,b\in\mathbb C$ with $b\neq0$ and $(A,B)\neq(0,0)$.
In particular, $R(h)$ is a Laurent polynomial in an exponential.
<!-- end mainthm-2 -->
<!-- mainthm-3: body_md -->
\label{thm:germ}
Let $f$ be meromorphic on $\mathbb C$, and let $\alpha,\beta\in\overline{\mathbb Q}$ be
distinct. Suppose that
\[
 f(\alpha+t)\in\overline{\mathbb Q}((t)),\qquad f(\beta)\in\mathbb P^1(\overline{\mathbb Q}).
\]
If $f$ and $f^{(n)}$ are algebraically dependent over $\mathbb C$ for some
$n\geq1$, then $f\in\overline{\mathbb Q}(z)$.

<!-- lemma-1: body_md -->
\label{lem:poly-diagram}
If $u$ is entire of exponential type and $R\in\mathbb C[T]$ has degree $d\geq1$, then
\[
 K(R(u))=dK(u),\qquad \tau(R(u))=d\tau(u).
\]
If $R$ is constant, then $K(R(u))=\{0\}$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
The leading term of $R$ gives
\[
 \log^+|R(w)|=d\log^+|w|+O_R(1)
 \qquad(w\in\mathbb C),
\]
where the bound is uniform: it follows from leading-term comparison for large $|w|$ and boundedness on a fixed disc for the remaining values.
Substitute $w=u(re^{i\theta})$, divide by $r$, and pass to the limsup.
The resulting identity is
\[
 \max\{h_{R(u)}(\theta),0\}
 =d\max\{h_u(\theta),0\}.
\]
These are the support functions of $K(R(u))$ and $dK(u)$, respectively.
Equality of the diagrams and \eqref{eq:type-diagram} give the assertions.
The constant case is immediate.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:inverse}
Let $L\subset\mathbb C$ be nonempty, compact, and convex.
Suppose that $G\in\operatorname{Exp}(L)$ and that $p\in\mathbb C[\zeta]$ has no zero on $L$.
There exists $v\in\operatorname{Exp}(L)$ such that $p(\mathrm D)v=G$.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Since the zero set of $p$ is disjoint from the compact set $L$, a sufficiently small neighbourhood of $L$ contains no zero of $p$.
Choose a positively oriented contour $\Gamma$ around $L$ in that neighbourhood, with its interior also avoiding the zeros of $p$, and put
\[
 v(z)=\frac1{2\pi i}\int_\Gamma
       \frac{\mathcal B G(\zeta)}{p(\zeta)}e^{\zeta z}\,d\zeta.
\]
The integrand is holomorphic off $L$ in this neighbourhood. Any two
such contours have the same winding number about $L$ and can be
compared by Cauchy's theorem, so the integral is independent of the
contour. Differentiation under the integral, followed by \eqref{eq:polya}, gives $p(\mathrm D)v=G$.

For every sufficiently small $\varepsilon>0$, choose $\Gamma$ in the $\varepsilon$-neighbourhood of $L$. Each $\zeta\in\Gamma$ then satisfies
\[
 \operatorname{Re}(\zeta z)\leq H_L(z)+\varepsilon|z|.
\]
The contour integral yields \eqref{eq:uniform-support}, with a constant depending on the contour and hence on $\varepsilon$.
Contour independence ensures that all these estimates concern the same function $v$.
Thus $v\in\operatorname{Exp}(L)$.
<!-- end proof-2 -->
<!-- lemma-3: body_md -->
\label{lem:zero-independence}
Let $\mu_1,\ldots,\mu_s$ be distinct complex numbers, and let $u_1,\ldots,u_s$ be entire functions of exponential type zero. If
\[
 \sum_{j=1}^s u_j(z)e^{\mu_jz}=0,
\]
then $u_1=\cdots=u_s=0$.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
The transform $\mathcal B u_j$ is holomorphic on $\mathbb C\setminus\{0\}$ and vanishes at infinity.
The product rule, applied in a neighbourhood of infinity where the series converge absolutely, gives
\begin{equation}\label{eq:borel-shift}
 \mathcal B\bigl(u_je^{\mu_j\,\cdot}\bigr)(\zeta)
 =\mathcal B u_j(\zeta-\mu_j).
\end{equation}
Indeed, for each $\ell\geq0$,
\[
 \sum_{m\geq\ell}\binom m\ell
 \frac{\mu^{m-\ell}}{\zeta^{m+1}}
 =\frac1{(\zeta-\mu)^{\ell+1}}.
\]
Taking Borel transforms of the given identity and using analytic continuation yields
\[
 \sum_{j=1}^s\mathcal B u_j(\zeta-\mu_j)=0
 \qquad\bigl(\zeta\notin\{\mu_1,\ldots,\mu_s\}\bigr).
\]
Near $\mu_j$, every summand other than the $j$th is holomorphic.
Hence the possible isolated singularity of $\mathcal B u_j(\zeta-\mu_j)$ at $\mu_j$ is removable.
Thus $\mathcal B u_j$ is entire and vanishes at infinity.
Liouville's theorem makes it zero, and \eqref{eq:polya} gives $u_j=0$.
<!-- end proof-3 -->
<!-- lemma-4: body_md -->
\label{lem:derivative-type}
If $u$ is entire of exponential type and $m\geq1$, then
\[
 \tau\bigl(u^{(m)}\bigr)=\tau(u).
\]
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Cauchy's estimate on unit circles gives
\[
 M(r,u^{(m)})\leq m!M(r+1,u),
\]
so $\tau(u^{(m)})\leq\tau(u)$. Conversely, Taylor's formula with integral remainder on the segment from $0$ to $z$ gives
\[
 u(z)=\sum_{j=0}^{m-1}\frac{u^{(j)}(0)}{j!}z^j
 +\frac{z^m}{(m-1)!}\int_0^1(1-s)^{m-1}u^{(m)}(sz)\,ds.
\]
Consequently,
\[
 M(r,u)\leq C(1+r^{m-1})+\frac{r^m}{m!}M(r,u^{(m)}).
\]
This proves the reverse inequality. If $u^{(m)}=0$, then $u$ is a polynomial and both types are zero.
<!-- end proof-4 -->
<!-- lemma-5: body_md -->
\label{lem:automatic-growth}
Let $n\geq1$, let $R,S\in\mathbb C[T]$ have the same degree $d\geq1$,
and let $h$ be a nonconstant entire function satisfying
$\mathrm D^nR(h)=S(h)$. Then $h$ and $R(h)$ have finite positive
exponential type.
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
Let $r_d,s_d$ be the leading coefficients of $R,S$, and put
$F=R(h)$, $a=s_d/r_d\neq0$, and $Q=S-aR$. Then $F$ is nonconstant and $\deg Q<d$.
Polynomial comparison gives constants $C>0$ and $0\leq\gamma<1$
such that, on $\mathbb C$,
\[
 |F^{(n)}-aF|=|Q(h)|\leq C(1+|F|)^\gamma.
\]
For nonconstant $Q$ one can take $\gamma=\deg Q/d$; for constant
or zero $Q$ take $\gamma=0$.
In particular, $|F^{(n)}|\leq C_1(1+|F|)$. Along each ray,
\[
 U_\theta(t)=1+\sum_{j=0}^{n-1}|F^{(j)}(te^{i\theta})|
\]
is locally absolutely continuous and satisfies
$U_\theta'\leq C_2U_\theta$ almost everywhere, with $C_2$ and
$U_\theta(0)$ independent of $\theta$. Gronwall's inequality gives
$M(r,F)\leq C_3e^{C_2r}$.

Choose $z_r$ with $|z_r|=r$ and $|F(z_r)|=M(r,F)$.
Since $F$ is nonconstant, $M(r,F)\to\infty$, so the first estimate
implies $|F^{(n)}(z_r)|\geq |a|M(r,F)/2$ for all sufficiently large $r$.
Fix $L>0$ with $\eta:=|a|L^n/(2n!)>1$. Cauchy's estimate on
$D(z_r,L)$ then yields
\[
 M(r+L,F)\geq\frac{L^n}{n!}|F^{(n)}(z_r)|
             \geq\eta M(r,F).
\]
Iteration gives $\tau(F)\geq L^{-1}\log\eta>0$.
Finally, polynomial comparison gives
$|h|^d\leq C_4(1+|F|)$, so $h$ has finite exponential type.
Lemma~\ref{lem:poly-diagram} now yields $\tau(F)=d\tau(h)$,
and hence $0<\tau(h)<\infty$.
<!-- end proof-5 -->
<!-- lemma-6: body_md -->
\label{lem:polygon}
There are a nonempty set $\Lambda\subseteq\{\lambda:\lambda^n=a\}$ and nonzero constants $c_\lambda$ such that
\begin{equation}\label{eq:spectral}
 F(z)=\sum_{\lambda\in\Lambda}c_\lambda e^{\lambda z}+v(z),
 \qquad v\in\operatorname{Exp}(\vartheta K).
\end{equation}
Moreover,
\begin{equation}\label{eq:polygon}
 K=\operatorname{conv}\bigl(\{0\}\cup\Lambda\bigr),\qquad \tau(F)=\rho_*.
\end{equation}
<!-- end lemma-6 -->
<!-- proof-6: body_md -->
Let $p(\zeta)=\zeta^n-a$. Its roots are simple and lie on $|\zeta|=\rho_*$. Variation of constants gives
\begin{equation}\label{eq:variation}
 F(z)=\sum_{\lambda^n=a}b_\lambda e^{\lambda z}
 +\sum_{\lambda^n=a}\frac1{p'(\lambda)}
   \int_0^z e^{\lambda(z-t)}G(t)\,dt,
\end{equation}
where the constants account for the initial data. Integrate on $t=sz$, $0\leq s\leq1$. For every $\varepsilon>0$, the exponential-type bound on $G$ gives
\[
 \left|\int_0^z e^{\lambda(z-t)}G(t)\,dt\right|
 \leq C_\varepsilon |z|
 e^{\max\{\rho_*,\tau(G)+\varepsilon\}|z|}.
\]
It follows that
\begin{equation}\label{eq:initial-type-bound}
 \tau(F)\leq\max\{\rho_*,\tau(G)\}.
\end{equation}
Together with \eqref{eq:contraction} and $\tau(F)>0$, this implies $\tau(F)\leq \rho_*$: otherwise both $\rho_*$ and $\vartheta\tau(F)$ would be smaller than $\tau(F)$.

Thus $K$ lies in the closed disc of radius $\rho_*$, and $\vartheta K$ lies in the smaller disc of radius $\vartheta\rho_*$.
In particular, $p$ has no zero on $\vartheta K$.
Lemma~\ref{lem:inverse} provides a particular solution $v\in\operatorname{Exp}(\vartheta K)$ of $p(\mathrm D)v=G$.
The difference $F-v$ belongs to $\ker(\mathrm D^n-a)$, which gives \eqref{eq:spectral} after discarding the zero coefficients.
If none remained, then $K\subseteq \vartheta K$, forcing $K=\{0\}$ by boundedness and $\vartheta<1$.
This contradicts $\tau(F)>0$, so $\Lambda$ is nonempty.

Fix $\lambda\in\Lambda$ and take the unit direction
\[
 \nu=\frac{\overline\lambda}{\rho_*}.
\]
This conjugation is determined by the pairing in \eqref{eq:support}. We have
\[
 \operatorname{Re}(\lambda\nu)=\rho_*,
 \qquad \operatorname{Re}(\mu\nu)<\rho_*
 \quad(\mu\in\Lambda\setminus\{\lambda\}).
\]
Since $v$ has type at most $\vartheta\rho_*<\rho_*$, the $\lambda$-mode strictly dominates every other term on $z=t\nu$ as $t\to+\infty$.
Hence $h_F(\arg\nu)=\rho_*$.
The diagram $K$ is contained in $|\zeta|\leq \rho_*$, and the unique point of this disc satisfying $\operatorname{Re}(\zeta\nu)=\rho_*$ is $\lambda$.
Thus $\lambda\in K$.

Put $K_0=\operatorname{conv}(\{0\}\cup\Lambda)$. We have proved $K_0\subseteq K$. Conversely, the indicator bound for a finite sum in \eqref{eq:spectral} yields
\[
 H_K(z)\leq\max\{H_{K_0}(z),\vartheta H_K(z)\}
 \qquad(z\in\mathbb C).
\]
Both support functions are nonnegative.
If $H_K(z)>H_{K_0}(z)$ at some $z$, the last inequality would give $H_K(z)\leq \vartheta H_K(z)$ with $H_K(z)>0$, a contradiction.
Therefore $K\subseteq K_0$, proving the diagram identity.
Since $\Lambda$ contains a point of modulus $\rho_*$, the type identity follows as well.
<!-- end proof-6 -->
<!-- lemma-7: body_md -->
\label{lem:edge}
The set $\Lambda$ in Lemma~\ref{lem:polygon} consists of one point or two antipodal points.
<!-- end lemma-7 -->
<!-- proof-7: body_md -->
Suppose otherwise.
The circle $|\lambda|=\rho_*$ has two consecutive points of $\Lambda$ bounding an empty arc of angular length less than $\pi$.
For at least three points, this follows because the circular gaps sum to $2\pi$; for two non-antipodal points, choose the shorter arc.
We may write the endpoints as
\[
 \lambda_1=\rho_*e^{i(\phi-\alpha)},\qquad
 \lambda_2=\rho_*e^{i(\phi+\alpha)},\qquad
 0<\alpha<\frac\pi2.
\]
With $\nu=e^{-i\phi}$ and $\sigma_*=\rho_*\cos\alpha$, we obtain
\begin{equation}\label{eq:positive-edge}
 \begin{split}
 \operatorname{Re}(\lambda_1\nu)
 &=\operatorname{Re}(\lambda_2\nu)=\sigma_*>0,\\
 \operatorname{Re}(\lambda\nu)&<\sigma_*
 \qquad(\lambda\in\Lambda\setminus\{\lambda_1,\lambda_2\}).
 \end{split}
\end{equation}
Thus the chord is an exposed edge of $K=\operatorname{conv}(\{0\}\cup\Lambda)$; adjoining the origin does not change the edge because $\sigma_*>0$. Figure~\ref{fig:edge} illustrates the two planes used in the argument.

Set $c_j=c_{\lambda_j}$ and
\[
 E(z)=c_1e^{\lambda_1z}+c_2e^{\lambda_2z}.
\]
Its zeros are
\begin{equation}\label{eq:edge-zeros}
 z_k=\frac{\ell+2\pi i k}{\lambda_1-\lambda_2},\qquad
 k\in\mathbb Z,\qquad e^\ell=-\frac{c_2}{c_1}.
\end{equation}
They are simple, since
\[
 E'(z_k)=c_1e^{\lambda_1z_k}(\lambda_1-\lambda_2)\neq0.
\]
Indeed,
\[
\frac{2\pi i}{\lambda_1-\lambda_2}
=-\frac{\pi}{\rho_*\sin\alpha}\,\nu.
\]
Choosing the indices tending to $-\infty$ in
\eqref{eq:edge-zeros} and relabelling, we obtain
\[
 z_k=t_k\nu+z_*,\qquad t_k\longrightarrow+\infty,
\]
with $z_*$ fixed.

Choose $\delta>0$ smaller than half the distance between consecutive zeros in \eqref{eq:edge-zeros}, and let $D_k=D(z_k,\delta)$. For $|w|=\delta$,
\[
 E(z_k+w)=c_1e^{\lambda_1z_k}
 \bigl(e^{\lambda_1w}-e^{\lambda_2w}\bigr).
\]
The factor in parentheses has no zero on $|w|=\delta$, so its minimum modulus there is positive. Since
$|e^{\lambda_1z_k}|=e^{\sigma_*t_k+\operatorname{Re}(\lambda_1z_*)}$, there is $c>0$, independent of $k$, such that
\begin{equation}\label{eq:edge-large}
 \min_{\partial D_k}|E|\geq ce^{\sigma_*t_k}.
\end{equation}

By \eqref{eq:positive-edge}, the remaining exponential modes are $O(e^{(\sigma_*-\eta)t_k})$ uniformly on $\partial D_k$, for some fixed $\eta>0$ when such modes are present.
For the remainder, boundedness of $K$ and $z-z_k$ gives
\[
 H_K(z)\leq t_kH_K(\nu)+O(1)=\sigma_*t_k+O(1),
 \qquad |z|\leq t_k+O(1)
 \quad(z\in\partial D_k).
\]
Estimate~\eqref{eq:uniform-support} therefore yields
\begin{equation}\label{eq:edge-remainder}
 \max_{\partial D_k}|v|
 \leq C_\varepsilon\exp\bigl(\vartheta\sigma_*t_k+\varepsilon t_k+O(1)\bigr).
\end{equation}
Take $\varepsilon=(1-\vartheta)\sigma_*/2$. Equations~\eqref{eq:edge-large}
and \eqref{eq:edge-remainder}, together with the estimate for the other
modes, give constants $C_1,\gamma>0$, independent of $k$, such that
\[
 \max_{\partial D_k}\left|\frac{F-E}{E}\right|
 \leq C_1 e^{-\gamma t_k}.
\]
For all large $k$, Rouch\'e's theorem gives
\begin{equation}\label{eq:first-zero-count}
 N_{D_k}(F)=N_{D_k}(E)=1.
\end{equation}
The same estimates give the separate boundary conclusion
\begin{equation}\label{eq:boundary-properness}
 \min_{\partial D_k}|F|\geq\frac c2e^{\sigma_*t_k}
 \longrightarrow\infty.
\end{equation}

Now use $F=R(h)$. For every $M>0$, the quantity
\[
 C_M=\max_{|w|\leq M}|R(w)|
\]
is finite. Once $\min_{\partial D_k}|F|>C_M$, no value of $h$ on $\partial D_k$ can have modulus at most $M$. Thus
\[
 \min_{\partial D_k}|h|\longrightarrow\infty.
\]
Choose $M$ large enough that
\[
 |R(w)-r_dw^d|<|r_dw^d|\qquad(|w|>M).
\]
For all sufficiently large $k$, this inequality holds with $w=h(z)$ throughout $\partial D_k$. A second application of Rouch\'e's theorem gives
\begin{equation}\label{eq:second-zero-count}
 N_{D_k}(F)=N_{D_k}(r_dh^d)=dN_{D_k}(h).
\end{equation}
This compares total zero counts; it does not identify the zeros of $R(h)$ with those of $h$.
In particular, nonzero roots and the constant term of $R$ cause no exception.
Equations~\eqref{eq:first-zero-count} and \eqref{eq:second-zero-count} contradict $d\geq2$, proving the lemma.

\noindent\emph{Sectorial inversion.}
Fix $0<\delta<\pi/2$ and consider the closed sector
\[
 S_\delta=\{w\neq0:|\arg w|\leq\pi/2-\delta\}.

For $w\in S_\delta$, put $x=\operatorname{Re}w$, so $x\geq(\sin\delta)|w|$. By \eqref{eq:segment-diagrams} and \eqref{eq:uniform-support},
\begin{equation}\label{eq:V-sector}
 |V(w)|\leq C_\varepsilon\exp\bigl(m_Qx+\varepsilon|w|\bigr)
 \qquad(w\in S_\delta)
\end{equation}
for every $\varepsilon>0$. Since $m_Q<d$, take $\varepsilon=(d-m_Q)\sin\delta/2$ in

we obtain
\begin{equation}\label{eq:sector-leading}
 R(\widetilde h(w))=c_+e^{dw}
 \bigl(1+O_\delta(e^{-\eta_\delta|w|})\bigr)
\end{equation}
uniformly as $|w|\to\infty$ in $S_\delta$.

Set $T(w)=\widetilde h(w)-c_0$.
The modulus of \eqref{eq:sector-leading} tends uniformly to infinity.
Polynomial properness first gives $|T(w)|\to\infty$ uniformly, and leading-term comparison in \eqref{eq:centered-R} then gives
\begin{equation}\label{eq:T-size}
 |T(w)|\asymp e^x\qquad(w\in S_\delta,\ |w|\to\infty).
\end{equation}
Indeed, for large $|T|$, the quantities $|\widetilde R(T)|$ and $|T|^d$ are comparable, whereas \eqref{eq:sector-leading} makes $|\widetilde R(T(w))|$ comparable to $e^{dx}$.
Dividing \eqref{eq:centered-R} by $e^{dw}$, the lower-degree terms are $O(e^{-2x})$ by \eqref{eq:T-size}.

\[
 \bigl(T(w)e^{-w}\bigr)^d\longrightarrow\frac{c_+}{r_d}
\]
uniformly in the sector.

The roots of $X^d-c_+/r_d$ are distinct and nonzero.

The truncated sector $S_\delta\cap\{|w|>R_0\}$ is connected, so continuity confines the quotient to one of these discs.

\begin{equation}\label{eq:chosen-root}
 T(w)e^{-w}\longrightarrow A,\qquad A^d=c_+/r_d,
\end{equation}
uniformly in $S_\delta$, for one fixed root $A\neq0$.
The sectors are nested and all contain the positive real ray, so the root is independent of $\delta$.
This argument selects the root through connectedness; no global branch of an inverse function is required.

Factor
\[
 T^d-A^de^{dw}=(T-Ae^w)

\]
By \eqref{eq:chosen-root}, the sum on the right is asymptotic to $dA^{d-1}e^{(d-1)w}$ uniformly in $S_\delta$.
From \eqref{eq:normalized-spectral} and \eqref{eq:centered-R},
\[
 r_d(T^d-A^de^{dw})
 =c_-e^{-dw}+V(w)-\sum_{j=0}^{d-2}\widetilde r_jT^j.
\]
Combining this identity with \eqref{eq:T-size} and \eqref{eq:V-sector} yields
\begin{equation}\label{eq:sector-error}
 |T-Ae^w|\leq C_{\varepsilon,\delta}
 \left(\exp\bigl((m_Q-d+1)x+\varepsilon|w|\bigr)
       +e^{-x}+e^{-(2d-1)x}\right)
\end{equation}
for all sufficiently large $|w|$ in $S_\delta$. The three terms come, respectively, from $V$, the lower powers of $T$, and the opposite exponential mode.

Since $m_Q\leq d-1$ and $\varepsilon$ can be arbitrarily small, \eqref{eq:sector-error} implies that the indicator of $T-Ae^w$ is at most zero on every ray in the open right half-plane.

\smallskip
\noindent\emph{The type-zero conclusion.}
If $c_-=0$, take $B=0$.
On the open left half-plane, $K(\widetilde h)=[0,1]$ already gives indicator at most zero for $\widetilde h$.
The same is true of the constant $c_0$ and of $Ae^w$, so $u$ has indicator at most zero there.

If $c_-\neq0$, repeat the sectorial argument with $w$ replaced by $-w$.
It gives $B\neq0$, with $B^d=c_-/r_d$, such that $\widetilde h-c_0-Be^{-w}$ has indicator at most zero on the open left half-plane.
Subtracting $Ae^w$, which decreases exponentially there, preserves the bound.
On the open right half-plane, subtracting $Be^{-w}$ from \eqref{eq:sector-error} likewise preserves the bound.

In either case, $u$ is of exponential type and has indicator at most zero away from the imaginary axis.
If $u\neq0$, continuity of its indicator gives the same conclusion on the two imaginary directions.
Formula~\eqref{eq:type-diagram} now gives $\tau(u)=0$.
The case $u=0$ is immediate.
Notice that this last step does not require sectorial estimates uniform as $\delta\to0$.

\smallskip
\noindent\emph{Eliminating the remainder.}
In the $w$-variable, the differential equation reads
\begin{equation}\label{eq:scaled-ODE}
 (b^n\mathrm D_w^n-a)R(\widetilde h)=Q(\widetilde h),\qquad a=(db)^n.
\end{equation}
Substitute $\widetilde h=c_0+Ae^w+Be^{-w}+u$.
Each side becomes a finite sum of $e^{jw}$, $j\in\mathbb Z$, with entire coefficients of type zero.
Lemma~\ref{lem:zero-independence} therefore allows comparison of the coefficients at each frequency.

The coefficient of $e^{(d-1)w}$ in $R(\widetilde h)=\widetilde R(Ae^w+Be^{-w}+u)$ is
\begin{equation}\label{eq:critical-coefficient}
 dr_dA^{d-1}u.
\end{equation}
To check this, choose $k_+$ factors $Ae^w$, $k_-$ factors $Be^{-w}$, and $k_0$ factors $u$ from the top power. The conditions for frequency $d-1$ are
\[
 k_+-k_-=d-1,\qquad k_++k_-+k_0=d,
\]
so $2k_-+k_0=1$.
Hence $k_-=0$ and $k_0=1$, with multiplicity $d$.
No power of degree at most $d-2$ can reach frequency $d-1$, and the degree-$(d-1)$ term was removed in \eqref{eq:centered-R}.

Let $q_{d-1}$ be the coefficient of $T^{d-1}$ in $Q$, taken as zero if $m_Q<d-1$.
Translation by $c_0$ does not change that coefficient, and the coefficient of $e^{(d-1)w}$ in $Q(\widetilde h)$ is $q_{d-1}A^{d-1}$.
Using
\[
 \mathrm D_w^n\bigl(a_j(w)e^{jw}\bigr)
 =e^{jw}(\mathrm D_w+j)^n a_j(w)
\]
in \eqref{eq:scaled-ODE} and dividing by $dr_dA^{d-1}b^n\neq0$, we obtain
\begin{equation}\label{eq:residual-ODE}
 \left[(\mathrm D_w+d-1)^n-d^n\right]u
 =\frac{q_{d-1}}{dr_db^n}.
\end{equation}
The characteristic roots are
\[
 \rho=d\zeta-(d-1),\qquad \zeta^n=1.
\]
They are simple, and none is zero because $(d-1)/d$ is not a root of unity.
Also $(d-1)^n-d^n\neq0$, so \eqref{eq:residual-ODE} has a constant particular solution $u_0$.
Every solution therefore has the form
\[
 u(w)=u_0+\sum_{\rho}a_\rho e^{\rho w}.
\]
The function $u-u_0$ has type zero.
Apply Lemma~\ref{lem:zero-independence} to this identity with frequency $0$ included: every $a_\rho$ vanishes.
Thus $u$ is constant.
Setting $C=c_0+u_0$ and returning to $z$ proves \eqref{eq:h-classified}.

\label{prop:known-input}
Let $f$ be a transcendental meromorphic solution of \eqref{eq:BB} on $\mathbb C$.
\begin{enumerate}
\item If $f$ has a pole, then $f\in\mathcal W$.

Theorem~D of \cite{ELN} originates in Eremenko's work
\cite{Eremenko82}; its proof is reproduced in the appendix of
\cite{ELN}. Part~\textup{(i)} is \cite[Theorem~1]{ELN}.
For part~\textup{(ii)}, choose an irreducible factor of the defining polynomial. On its
smooth projective normalization $X$, use the notation of
\cite[Section~3]{ELN}: the coordinate functions $p,q$ satisfy
\[
 p\circ\phi=f^{(n)},\qquad q\circ\phi=f,
\]
where $\phi:\mathbb C\to X$ is the holomorphic solution map. Let $I$ be
the pole set of $q$ and, for $\xi\in I$, put
\[
 \kappa(\xi)=\frac{\operatorname{ord}_\xi p}{\operatorname{ord}_\xi q}.
\]
The numerator and denominator are orders of meromorphic functions,
with a pole having negative order. We use the following precise
parts of \cite[Theorem~D]{ELN}, stated here in this notation:
\begin{enumerate}[label=\textup{(\alph*)},start=2]
\item For every $\xi\in I$, either $\kappa(\xi)=1$ or

\item In the latter case, $\phi^{-1}(\xi)$ is infinite, and each of
its points is a pole of $f$ of order $m$.
\item If $\kappa(\xi)=1$, then $\phi^{-1}(\xi)=\varnothing$.
\end{enumerate}
If $f$ is entire, alternative~\textup{(c)} is impossible: it supplies
actual poles in the source plane, not merely a formal branch at
infinity. Since $q$ is nonconstant on the compact curve $X$, the set
$I$ is nonempty. Hence at least one $\xi\in I$ has
$\kappa(\xi)=1$; in fact all do. Now \cite[Lemma~1]{ELN} gives
order one and normal type.

maximum-modulus convention of Section~\ref{sec:borel}. Indeed,

 T(r,f)\leq\log^+M(r,f),\qquad
 \log^+M(r,f)\leq3T(2r,f)+O(1),
\]
and normal type at order one means
$0<\limsup_{r\to\infty}T(r,f)/r<\infty$.
This deduction uses no hypothesis on the multiplicities of the

\label{lem:parametrization}
Let $f$ be a transcendental entire solution of \eqref{eq:BB}.
Then either $f\in\mathcal W$, or there are a transcendental entire function $h$ of finite positive exponential type and polynomials $R,S\in\mathbb C[T]$ of equal positive degree such that
\[
 f=R(h),\qquad f^{(n)}=S(h).
\]
<!-- end lemma-8 -->
<!-- proof-10: body_md -->
Choose an irreducible factor of $P$ vanishing on $(f,f^{(n)})$.
Let $X$ be the smooth projective normalization of its plane curve, and let $x,y\in\mathbb C(X)$ denote the coordinate functions.
The pair induces a nonconstant holomorphic map
\begin{equation}\label{eq:normalized-map}
 \phi:\mathbb C\longrightarrow X,\qquad
 x\circ\phi=f,\quad y\circ\phi=f^{(n)}.
\end{equation}
The meromorphic pair defines a holomorphic map to the projective
plane curve. Since the source is a Riemann surface, this nonconstant
map lifts uniquely by the universal property of normalization.
Neither coordinate on $X$ is constant: a constant $f^{(n)}$ would
make $f$ a polynomial.

If $X$ has genus at least two, lift $\phi$ to the unit disc, its universal cover.
The lift is bounded and entire, hence constant by Liouville's theorem, a contradiction.
If $X$ has genus one, write its universal cover as $\pi:\mathbb C\to\mathbb C/\Gamma\simeq X$.
Since $\mathbb C$ is simply connected, $\phi=\pi\circ H$ for an entire function $H$, necessarily nonconstant.
Picard's theorem says that $H$ omits at most one complex number.
Every fibre of $\pi$ is infinite, so $H$ meets every fibre and $\phi$ is surjective.
But the nonconstant meromorphic function $x$ on the compact curve $X$ has a pole, and its pullback under a surjective nonconstant map has a pole.
This contradicts the fact that $f=x\circ\phi$ is entire.

Therefore $X\simeq\mathbb P^1$.
The union of the pole sets of $x$ and $y$ is nonempty and is omitted by $\phi$, because both functions in \eqref{eq:normalized-map} are entire.
Picard's theorem bounds this union by two points.

Suppose first that it consists of two points.
Choose a coordinate $t$ sending them to $0$ and $\infty$, and put $h=t\circ\phi$.
Then $h$ is entire and zero-free, and $f=R(h)$ for a nonconstant rational function $R$.
The rational-composition identity for the Nevanlinna characteristic \cite[Chapter~1]{Hayman} is
\[
 T(r,R(h))=(\deg R)T(r,h)+O(1),
\]
where $\deg R$ is the degree of the rational map. Proposition~\ref{prop:known-input} gives $T(r,f)=O(r)$, hence $T(r,h)=O(r)$. The standard estimate
\[
 \log M(r,h)\leq3T(2r,h)+O(1)
\]
for an entire function shows that $h$ has finite exponential type.
Hadamard factorization for a zero-free entire function of finite order at most one gives $h(z)=e^{az+b}$; nonconstancy forces $a\neq0$.
Thus $f\in\mathcal W$.
This is the two-omitted-point argument also used in \cite[Lemma~2]{ELN}.

Suppose instead that the union has a single point.
Choose a coordinate $t$ on $\mathbb P^1$ whose only pole is there.
Then $x=R(t)$ and $y=S(t)$ are polynomials of positive degree.
Since $\phi$ omits that point, $h=t\circ\phi$ is entire.
It is transcendental, since otherwise $f=R(h)$ would be a polynomial.
Leading-term comparison gives
\[
 \log^+|R(h(z))|=(\deg R)\log^+|h(z)|+O(1).
\]
Taking maxima and using Proposition~\ref{prop:known-input} proves
\[
 0<\tau(h)=\frac{\tau(f)}{\deg R}<\infty.
\]
Finally, Lemmas~\ref{lem:poly-diagram} and \ref{lem:derivative-type} give
\[
 (\deg R)\tau(h)=\tau(f)=\tau(f^{(n)})=(\deg S)\tau(h).
\]
Since $\tau(h)>0$, we obtain $\deg R=\deg S$.

\label{lem:descent}
Let $k\subseteq\mathbb C$ be a subfield, and let $F_1,\ldots,F_N\in k((t))$.
If these series are linearly dependent over $\mathbb C$, then they are linearly dependent over $k$.
<!-- end lemma-9 -->
<!-- proof-12: body_md -->
Choose a common lower bound and write
\[
 F_j=\sum_{m\geq m_0}a_{m,j}t^m,\qquad a_{m,j}\in k.
\]
The rows $(a_{m,1},\ldots,a_{m,N})$ lie in $k^N$.
Select finitely many rows forming a basis, and put them in a matrix $B$.
The rank of $B$ is unchanged by the scalar extension $k\subseteq\mathbb C$, since rank is detected by nonzero minors.
A nontrivial complex dependence of the $F_j$ is in the complex kernel of $B$, so this rank is less than $N$.
The kernel over $k$ is therefore also nonzero.
Any nonzero vector in that kernel annihilates every original coefficient row, giving a dependence over $k$.

The constant case follows from either algebraic value.
Suppose that $f$ is nonconstant.
Choose an irreducible factor of $P$ over $k$ vanishing at $(f,f')$.
Since $k$ is algebraically closed, the associated smooth projective normalization $X$ and its coordinate functions $x,y\in k(X)$ are defined over $k$.
As in Lemma~\ref{lem:parametrization}, projective compactification and normalization give a nonconstant holomorphic map
\[
 \phi:\mathbb C\longrightarrow X(\mathbb C),\qquad
 x\circ\phi=f,\quad y\circ\phi=f'.
\]

\smallskip
\noindent\emph{Algebraic points and the pulled-back differential.}
The nonconstant rational function $x$ defines a finite morphism $X\to\mathbb P^1_k$. Every point of a fibre over $\mathbb P^1(k)$ is defined over $k$: its
residue field is a finite extension of the algebraically closed field
$k$. Hence
\begin{equation}\label{eq:algebraic-curve-points}
 \phi(\alpha),\phi(\beta)\in X(k).
\end{equation}
This includes either of the values $f(\alpha)$ and $f(\beta)$ being infinite.

The function $y$ is not identically zero, since $f$ is nonconstant. Define
\[
 \omega=\frac{dx}{y}.
\]
This is a nonzero rational differential over $k$. On a nonempty open set,
\[
 \phi^*\omega=\frac{df}{f'}=dz,
\]
and the identity extends as an identity of meromorphic differentials everywhere.
To record its local consequence, let $e\geq1$ be the local degree of $\phi$ at $z_0$ and let $m=\operatorname{ord}_{\phi(z_0)}\omega$.
In local parameters, the pullback of a differential of the form $s^m v(s)\,ds$, with $v(0)\neq0$, under $s=a(z-z_0)^e+\cdots$ has order
\[
 em+(e-1)=e(m+1)-1.
\]
Since $dz$ has order zero, we obtain
\begin{equation}\label{eq:local-omega}
 e(m+1)=1.
\end{equation}
Since $e$ and $m+1$ are integers and $e\geq1$, we have $e=1$ and $m=0$.
In particular, $\phi$ is unramified and omits every zero and pole of $\omega$.
This calculation allows zeros of $f'$ and singularities of the plane model; the passage to the normalized curve has removed any need to assume a nonvanishing separant.

Uniformization excludes genus at least two, as in Lemma~\ref{lem:parametrization}. We treat the two remaining genera separately.

\smallskip
\noindent\emph{Genus zero.}
Choose an isomorphism $X\simeq\mathbb P^1_k$.
The support of $\operatorname{div}\omega$ is omitted by $\phi$, so Picard's theorem bounds it by two points.
It is nonempty because a rational differential on $\mathbb P^1$ has divisor of degree $-2$.
Its points are defined over $k$.

If the support is a single point, send it to infinity by a coordinate $t$ defined over $k$.
The differential then has the form $\omega=c\,dt$ with $c\in k^\times$: the quotient $\omega/dt$ is a rational function with no zeros or poles.
Since $\phi$ omits infinity, the function $h=t\circ\phi$ is entire.
Pullback gives $ch'=1$.
By \eqref{eq:algebraic-curve-points}, $h(\alpha)\in k$, and therefore
\[
 h(z)=h(\alpha)+\frac{z-\alpha}{c}\in k[z].
\]
As $x\in k(t)$, it follows that $f\in k(z)$.

If the support has two points, send them to $0$ and $\infty$ by a coordinate over $k$. A rational differential with no other zeros or poles has the form
\[
 \omega=ct^m\,dt,\qquad c\in k^\times,\quad m\in\mathbb Z.
\]
Now $h=t\circ\phi$ is entire and zero-free, and
\[
 ch^mh'=1.
\]
For $m\neq-1$, this gives
\[
 (h^{m+1})'=\frac{m+1}{c}\neq0.
\]
Even when $m+1<0$, the function $h^{m+1}$ is entire and zero-free.
It cannot be a nonconstant affine function, which has a zero.
Hence $m=-1$, and integration gives
\[
 \frac{h(\beta)}{h(\alpha)}
 =\exp\left(\frac{\beta-\alpha}{c}\right).
\]
The quotient on the left belongs to $k^\times$ by \eqref{eq:algebraic-curve-points}; the exponent is a nonzero algebraic number.
Hermite--Lindemann contradicts this equality; see \cite[Theorem~4]{Waldschmidt}.

\smallskip
\noindent\emph{Genus one.}
Take the algebraic point $O=\phi(\alpha)$ as origin, so that $X$ becomes an elliptic curve over $k$. Choose a Weierstrass model
\[
 E:\quad V^2=4U^3-g_2U-g_3,
 \qquad g_2,g_3\in k,\quad g_2^3-27g_3^2\neq0,
\]
and put $\eta=dU/V$. The complex uniformization $\pi:\mathbb C\to E(\mathbb C)$ can be normalized by
\[
 \pi(0)=O,\qquad \pi^*\eta=du,
 \qquad \pi(u)=(\wp(u),\wp'(u))
\]
away from the period lattice, where $\wp$ has invariants $g_2,g_3$.

Lift $\phi$ to an entire function $H$ with $\phi=\pi\circ H$ and $H(\alpha)=0$.
The lift is nonconstant.
Each fibre of $\pi$ is infinite, whereas Picard's theorem permits $H$ to omit at most one complex number.
Thus $\phi$ is surjective.
Since $\phi$ omits every zero and pole of $\omega$, there are none: $\omega$ is holomorphic and nowhere zero.

The space of holomorphic differentials on $E$ over $k$ is one-dimensional, so
\[
 \omega=c\eta\qquad(c\in k^\times).
\]
The algebraicity of $c$ also follows directly: $\omega/\eta$ is a constant rational function defined over $k$.
Pullback gives $cH'=1$, and the normalization at $\alpha$ yields
\begin{equation}\label{eq:elliptic-argument}
 H(z)=\frac{z-\alpha}{c},\qquad
 \phi(\beta)=\pi\left(\frac{\beta-\alpha}{c}\right).
\end{equation}
The argument $(\beta-\alpha)/c$ is nonzero and algebraic.
Schneider's theorem states that for algebraic invariants and a nonzero algebraic $u$, the number $u$ is not a pole of $\wp$ and $\wp(u)$ is transcendental; see \cite{Schneider} and \cite[Theorem~20\textup{(1)}]{Waldschmidt}.
Consequently the point on the right of \eqref{eq:elliptic-argument} is not in $E(k)$, contrary to \eqref{eq:algebraic-curve-points}.
The theorem also rules out a nonzero period as the argument, so the projective value $\infty$ introduces no exception.

Only the rational genus-zero case remains, which proves the result.

\label{lem:affine-arithmetic}
Let $\alpha,\beta\in k$ be distinct, and let $f$ be meromorphic on $\mathbb C$ with
\[
 f(\alpha+t)\in k((t)),\qquad f(\beta)\in\mathbb P^1(k).
\]
If $f^{(n)}=af+b$ for some $n\geq1$ and $a,b\in\mathbb C$, then $f\in k[z]$.

<!-- proof-15: body_md -->
If $f$ had a pole of order $m$, then $f^{(n)}$ would have a pole of order $m+n$, whereas $af+b$ would have order at most $m$.
Hence $f$ is entire.
The constant case follows from the germ, so suppose that $f$ is nonconstant.

Apply Corollary~\ref{cor:descent} to the monomials $f^{(n)},f,1$.
The coefficient of $f^{(n)}$ in the resulting nontrivial relation over $k$ cannot vanish, because $1$ and $f$ are linearly independent.
Thus the affine equation has coefficients in $k$.
These coefficients are unique for a nonconstant $f$, so the given $a,b$ themselves belong to $k$.

If $a=0$, repeated integration with the algebraic initial derivatives at $\alpha$ gives a polynomial in $k[z]$.
If $a\neq0$, put $g=f+b/a$.
Then $g^{(n)}=ag$.
Let $\lambda_1,\ldots,\lambda_n$ be the roots of $X^n-a$.
They are distinct, nonzero, and algebraic, and
\begin{equation}\label{eq:affine-expansion}
 g(z)=\sum_{j=1}^n c_je^{\lambda_j(z-\alpha)}.
\end{equation}
For $0\leq\ell<n$, evaluation of the derivatives at $\alpha$ gives
\[
 g^{(\ell)}(\alpha)=\sum_{j=1}^n c_j\lambda_j^\ell\in k.
\]
The coefficient matrix is a Vandermonde matrix with determinant
$\prod_{i<j}(\lambda_j-\lambda_i)\neq0$ in $k$. Hence $c_j\in k$ for every $j$.

Because $f$ is entire, the hypothesis on $f(\beta)$ gives a finite algebraic value, and $g(\beta)\in k$. The numbers
\[
 0,\ \lambda_1(\beta-\alpha),\ldots,\lambda_n(\beta-\alpha)
\]
are distinct and algebraic.
Lindemann--Weierstrass makes their exponentials linearly independent over $k$; see \cite[Theorem~6]{Waldschmidt}.
Evaluating \eqref{eq:affine-expansion} at $\beta$ therefore forces $c_1=\cdots=c_n=0$.
In this case $f=-b/a\in k$, completing the proof.

\label{ex:general-operator}
Put
\[
 h(z)=e^z+e^{iz},\qquad R(T)=T^2+T,\qquad S(T)=T^2+2T,
\]
and let $L=p(\mathrm D)$, where
\begin{equation}\label{eq:counter-polynomial}
 p(\zeta)=\frac35\zeta^4-(2+2i)\zeta^3
             +6i\zeta^2+(4-4i)\zeta-\frac35.
\end{equation}
One checks that
\[
 p(1)=p(i)=2,\qquad p(2)=p(1+i)=p(2i)=1.
\]
Since the frequencies of $h$ are $1,i$ and those of $h^2$ are
$2,1+i,2i$, it follows that
\[
 L R(h)=S(h).
\]
Yet $h$ is not of the form $Ae^{bz}+Be^{-bz}+C$, by linear
independence of exponentials with distinct frequencies.
More strongly, put $f=R(h)$ and $g=Lf=S(h)$. Then $g-f=h$ and
\begin{equation}\label{eq:general-operator-relation}
 (g-f)^2+g-2f=0.
\end{equation}
The polynomial $(Y-X)^2+Y-2X$ is irreducible: in the coordinates
$T=Y-X,X$ it is $T^2+T-X$. The entire function $f$ is not in
$\mathcal W$, because its nonzero frequencies do not lie on one real line,
whereas those of an entire nonpolynomial member of $\mathcal W$ lie in
$b\mathbb Z$ for some $b\neq0$. Nor does $g=af+b$ hold: comparison at
frequencies $1$ and $2$ would give $a=2$ and $a=1$.
