<!-- definition-1: body_md -->
A $U^p(a,b)$ atom is a step function
\[
a(t)=\sum_{k=1}^K \mathds{1}_{[t_{k-1},t_{k})}(t) \phi_k , \quad \sum_{k=1}^K\|\phi_k\|_{L^2(\mathbb{R}^n)}^p=1,
\]
$U^p(a,b)$ is the atomic space based on these atoms, that is $u \in U^p(a,b)$ if
\[
u = \sum_{j \in \mathbb{N}} \lambda_j a_j,
\]
where $(\lambda_j) \in l^1$ and $a_j$ are atoms as above; $U^p(a,b)$ is equipped with the standard norm
\[
\|u\|_{U^p(a,b)} = \inf \{ \sum_{j \in \mathbb{N}}  |\lambda_j |:  u = \sum_{j \in \mathbb{N}} \lambda_j a_j \ \mbox{as above} \}. 
\]
<!-- end definition-1 -->
<!-- definition-2: body_md -->
Given a function $v: (a,b) \rightarrow L^2(\mathbb{R}^n)$ we define
\[
|v|_{V^p} =\sup_{\tau=\{(t_j)_{j=1,N}\}} \left( \sum_j \| v(t_{j+1}) -v(t_j) \|_{L^2(\mathbb{R}^n)}^p \right)^\frac1p, 
\quad  \|v\|_{V^p} = |v|_{V^p} + \|v\|_{L^\infty_t L^2_x}. 
\]
We let $V^p=V^p(a,b)$ is the space of functions  $v: (a,b) \rightarrow L^2(\mathbb{R}^n)$ for which $ \|v\|_{V^p}$ is finite.

<!-- proposition-1: body_md -->
\label{Pref}
Let $M_1,M_2 \in (0,+\infty)$. For $k \in \mathbb{N}$, let $T_k : L^2(\mathbb{R}^n) \rightarrow L^2(\mathbb{R}^n)$ be a linear and bounded operator (acting spatially) such that
\[
M_1^{-1} \| \sum_{k \in \mathbb{N}}  T_k f \|_{L^2} \leq \left( \sum_{k \in \mathbb{N}}  \|T_k f \|^2_{L^2} \right)^\frac12 \leq M_2 \|f\|_{L^2}, \quad \forall f \in L^2.
\]
If $1 \leq p \leq 2$, then for all $u \in U^p$ we have the bound
\begin{equation} \label{U2dec}
\left( \sum_{k \in \mathbb{N}}  \|T_k u \|^2_{U^p} \right)^\frac12 \leq M_2 \|u\|_{U^p}.
\end{equation}
If $ p \geq 2$, then for all $v \in V^p$ we have the bound
\begin{equation} \label{V2dec}
 \| \sum_{k \in \mathbb{N}}  T_k v \|_{V^p}  \leq M_1 \left( \sum_{k \in \mathbb{N}}  \|T_k v \|^2_{V^p} \right)^\frac12. 
\end{equation}

<!-- lemma-1: body_md -->
\label{l2V2}
Assume that $v \in V^2$ is localized at frequency $2^k$ and we decompose
\[
v=\sum_{q \in \mathcal{C}_j} \chi_q(D_x) v
\]
for some $j \leq k$. If $q > 2$, then the following holds true
\begin{equation}
\sum_{q \in \mathcal{C}_k} \|\chi_q(D_x) v\|^2_{U^q} \lesssim \langle k-j \rangle^2 \|v\|_{V^2}^2. 
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
We set $N \in \mathbb{N}$ such that $2N(1-\frac2q) \approx 3(k-j)$. 
We use the decomposition $v=u_N+v-u_N$ from \eqref{V2U2q}. From \eqref{U2dec}
we have that
\begin{equation} \label{aux1a}
\sum_{q \in \mathcal{C}_k} \|\chi_q(D_x) u_{N}\|^2_{U^2} \lesssim \|u_N\|_{U^2}^2.  
\end{equation}
On the other hand, for each $q$ it follows that
\[
\|\chi_q(D_x) (v-u_N)\|_{U^q} \lesssim \|v-u_N\|_{U^q} \lesssim \|v-u_N\|_{V^2}. 
\]
From this we obtain
\begin{equation} \label{aux1b}
\sum_{q \in \mathcal{C}_j} \|\chi_q(D_x) (v-u_N)\|^2_{U^q} \lesssim 2^{3(k-j)} 
 \|v-u_N\|_{U^q}^2 \lesssim 2^{2N(1-\frac2q)} \|v-u_N\|_{U^q} \lesssim \|v\|_{V^2}^2,
\end{equation}
given that the cardinality of cubes $q \in \mathcal{C}_j$ for which $\chi_q(D_x) (v-u_N)\neq 0$ is $\approx 2^{3(k-j)}$; here we implicitly use that $v$ and $u_N$ are localized at frequency $2^k$.  

The result of the Lemma then follows from \eqref{aux1a}, \eqref{aux1b}, and the inclusion $U^2 \subset U^q$.

\]
A direct summation argument shows that $T$ is bounded on $\dot X^{-\frac12}$. Moreover, we also obtain the contraction type bound
\[
\| Tz_1 - Tz_2 \|_{\dot X^{-\frac12}} \lesssim (\| z_1 \|_{\dot X^{-\frac12}} + \| z_2 \|_{\dot X^{-\frac12}}) \| z_1-z_2 \|_{\dot X^{-\frac12}}. 
\]
Therefore, under the assumption that the initial data $\|z_0\|_{\dot H^{-\frac12}}$ is small enough, $T$ becomes a contraction in $\dot X^s$, which in turns leads to the existence of a fixed point $z$ which corresponds to a global solution of the equation \eqref{QNLS}. Scattering is a consequence of the fact that functions in $V^2_\Delta$ have well-defined limits as $t \rightarrow \pm \infty$.
