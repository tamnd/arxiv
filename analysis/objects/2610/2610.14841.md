\label{TheoremPY}
For $1<p<\infty$ and $T$ defined in \eqref{EqPY}, we have
\begin{equation*}
\|Tf\|_p\lesssim \|f\|_p,\ \ \ \forall f\in \mathcal{S}(\mathbb{R}^3).
\end{equation*}

<!-- lemma-1: body_md -->
\label{LemmaMaximalMajorArc}
Let $\lambda\geq 1$. Let $P_3\in \mathbb{R}[Y_1, Y_2]$ be a homogeneous polynomial of degree $3$. Then there exists an absolute constant $c>0$ such that
\begin{equation*}  
\big{\|}\sup_{R>0} \sup_{|\tau|\sim \lambda R^{-3}} |\int_{\mathbb{R}^2} f(x_{\leq 2}-y, x_3-|y|^2) e^{2\pi i \tau P_3(y)} \rho_R (y) dy| \big{\|}_2\lesssim \lambda^{-c}\|f\|_2, \ \ \ \forall f\in \mathcal{S}(\mathbb{R}^3).
\end{equation*}
Here $\rho_R(y):=\frac{1}{R^2}\rho(\frac{y}{R})$, and $\rho\in \mathcal{S}(\mathbb{R}^3)$ is supported in $\{|y|<5\}$ with $\|\rho\|_{C^1}\leq 1$.
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{LemmaContinuousMajorArcBi-Parameter}
Let $\lambda\gtrsim 1$. Let $\tau_{1, (\cdot)}, \tau_{2, (\cdot)}: \mathbb{R}^3\rightarrow \mathbb{R}$ be arbitrary measurable functions such that $\max\{|\tau_{1, x}|, |\tau_{2, x}| \}>\lambda, \forall x\in \mathbb{R}^3$. Let
\begin{equation*}
Af(x):=\int_{\mathbb{R}^2} f(x_{\leq 2}-y, x_3-|y|^2) e^{2\pi i (\tau_{1, x}y_1^3+\tau_{2, x}y_2^3)} \rho (y) dy,
\end{equation*}
where $\rho\in \mathcal{S}(\mathbb{R}^3)$ is supported in $\{|y|<5\}$ with $\|\rho\|_{C^1}\leq 1$. Then there exists an absolute constant $c>0$ such that
\begin{equation*}
\|Af\|_2\lesssim \lambda^{-c}\|f\|_2, \ \ \ \forall f\in L^2(\mathbb{R}^3).
\end{equation*}

<!-- theorem-2: body_md -->
\label{TheoremPYBi-Parameter}
Define the bi-parameter Pierce--Yung operator:
\begin{equation*}
\mathfrak{T}f(x):=\sup_{\tau_1, \tau_2\in \mathbb{R}} |\int_{\mathbb{R}^2} f(x_{\leq 2}-y, x_3-|y|^2) e^{2\pi i (\tau_1 y_1^3+\tau_2 y_2^3)} K(y) dy|.
\end{equation*}
Then for $1<p<\infty$ we have
\begin{equation*}
\|\mathfrak{T}f\|_p\lesssim \|f\|_p,\ \ \ \forall f\in \mathcal{S}(\mathbb{R}^3).
\end{equation*}

<!-- theorem-3: body_md -->
\label{TheoremFindingPatterns}
Let $q$ be a prime. Let $A$ be a subset of $\mathbb{F}_q^4$ with density $\alpha:=\frac{\#A}{q^4}$. Then there is an absolute constant $c_0>0$ such that
\begin{equation}  \label{EqFindingPatterns}
\begin{split}
\frac{1}{q^6}\# \big{\{}(x, y)\in \mathbb{F}_q^4\times \mathbb{F}_q^2: & \big{(}x_{\leq 3}, x_4\big{)}, \big{(}x_{\leq 3}-(y_1, y_2, y_1^2+y_2^2), x_4 \big{)}, \big{(} x_{\leq 3}, x_4-(y_1^3+y_2^3) \big{)}\in A \big{\}}   \\
&=\mathbb{E}_{x\in \mathbb{F}_q^4} 1_A(x)\big{(} \mathbb{E}_{x_{\leq 3}\in \mathbb{F}_q^3} 1_A(\cdot, x_4) \big{)} \big{(} \mathbb{E}_{x_4\in \mathbb{F}_q} 1_A(x_{\leq 3}, \cdot)  \big{)}+O(q^{-c_0}).
\end{split}
\end{equation}
Moreover, the right hand side is at least $\alpha^3+O(q^{-c_0})$.

<!-- lemma-3: body_md -->
\label{LemmaExponentialSumFiniteFields}
Let $q$ be a prime. Let $Q\in \mathbb{Z}[X]$ be a polynomial with degree $0<d<q$. Suppose $q$ does not divide the leading coefficient of $Q$. Then
\begin{equation*}
\big{|}\mathbb{E}_{t\in [q]} e\big{(} Q(t) \big{)} \big{|}\lesssim_d q^{-\frac{1}{d}}.
\end{equation*}

<!-- lemma-4: body_md -->
\label{LemmaFiniteFieldsMajorArc1}
Let $q$ be a prime. Let $\tau_{(\cdot)}: \mathbb{F}_q^3\rightarrow \mathbb{F}_q$ be any function. Suppose that $\tau_x\neq 0, \forall x\in \mathbb{F}_q^3$. Then for
\begin{equation}  \label{EqFiniteFieldsAveragingOperator}
Af(x):=\mathbb{E}_{y\in \mathbb{F}_q^2} f \big( x-\Gamma(y) \big) e\big( \tau_x\cdot P(y) \big)
\end{equation}
we have
\begin{equation*}
\|Af\|_{2}\lesssim q^{-c} \|f\|_2.
\end{equation*}
<!-- end lemma-4 -->
<!-- remark-1: body_md -->
Recall the notations defined in \eqref{EqDefinitionOfGammaAndP}.
<!-- end remark-1 -->
<!-- proof-1: body_md -->
WLOG assume $q>3$. We write $Af$ into a kernel representation on the Fourier side:
\begin{equation*}
Af(x)=\mathbb{E}_{y\in \mathbb{F}_q^2} \sum_{\xi\in \mathbb{F}_q^3} e \big(\xi\cdot (x-\Gamma(y)) \big)\hat{f}(\xi)e\big( \tau_x\cdot P(y) \big)=\sum_{\xi\in \mathbb{F}_q^3} K(x, \xi)\hat{f}(\xi),
\end{equation*}
where $K(x, \xi):=e(\xi\cdot x)\mathbb{E}_{y\in \mathbb{F}_q^2} e \big(-\xi\cdot\Gamma(y)+\tau_{x}P(y) \big)$. By Plancherel, it suffices to show the following operator has norm $O(q^{-c})$.
\begin{equation*}
f\mapsto \sum_{\xi\in \mathbb{F}_q^3} K(x, \xi)f(\xi)=:\mathfrak{A}f.
\end{equation*}
Then we compute the kernel of $\mathfrak{A}\mathfrak{A}^*$ as
\begin{equation}  \label{EqFiniteFieldsSummand1}
\begin{split}
\widetilde{K}(x, z):&=\sum_{\xi\in \mathbb{F}_q^3} K(x, \xi)\overline{K(z, \xi)} \\
& =\sum_{\xi\in \mathbb{F}_q^3} e\big( (x-z)\cdot\xi \big) \mathbb{E}_{y, y' \in \mathbb{F}_q^2} e \big(-\xi\cdot\Gamma(y)+\tau_{x}P(y) \big)e \big(-\xi\cdot\Gamma(y')+\tau_{x}P(y') \big).
\end{split}
\end{equation}
By Schur's test, it suffices to show
\begin{align}
& \mathbb{E}_{z\in \mathbb{F}_q^3} |\widetilde{K}(x, z)|\lesssim q^{-2c},\ \forall x\in \mathbb{F}_q^3, \label{EqFiniteFieldsSchurTest1}   \\
& \mathbb{E}_{x\in \mathbb{F}_q^3} |\widetilde{K}(x, z)|\lesssim q^{-2c},\ \forall z\in \mathbb{F}_q^3.  \label{EqFiniteFieldsSchurTest2}
\end{align}
Since $\widetilde{K}(x, z)=\overline{\widetilde{K}(z, x)}$, we only need to show \eqref{EqFiniteFieldsSchurTest1}. Fix $x$ from now on. Change the variable $z\mapsto x-z$. We exploit the oscillation by writing \eqref{EqFiniteFieldsSummand1} into:
\begin{equation}  \label{EqFiniteFieldsSummand2}
\begin{split}
\widetilde{K}(x, x-z) &=\mathbb{E}_{y, y'\in \mathbb{F}_q^2}\sum_{\xi\in \mathbb{F}_q^3}  e(z\cdot\xi) e\Big(-\xi\cdot \big( \Gamma(y)-\Gamma(y') \big) \Big) e \big(\tau_{x}P(y)-\tau_{x-z}P(y') \big) \\
&=\mathbb{E}_{y, y'\in \mathbb{F}_q^2} q^31_{\Gamma(y)-\Gamma(y')=z} e \big(\tau_{x}P(y)-\tau_{x-z}P(y') \big).
\end{split}
\end{equation}
Note that $\Gamma(y)-\Gamma(y')=z$ is equivalent to
\begin{equation}  \label{EqFiniteFieldsRangeOfy}
y-y'=z_{\leq 2},\ \ \ y\cdot z_{\leq 2}=2^{-1}(z_3+z_1^2+z_2^2).
\end{equation}
Here $2^{-1}$ is the inverse of $2$ modulo $q$. When $z_{\leq 2}=0$, we have
\begin{equation}  \label{EqFiniteFieldsRDegenerateCase}
\text{RHS of \eqref{EqFiniteFieldsSummand2}}=\begin{cases}
0,\ z_3\neq 0, \\
\frac{1}{q^4} \sum_{y=y'} q^3=q,\ z_3=0
\end{cases}
\end{equation}
When $z_{\leq 2}\neq 0$, \eqref{EqFiniteFieldsRangeOfy} is equivalent to $y-y'=z_{\leq 2}$ and $y$ belongs to the line $\{t\in \mathbb{F}_q: tz_{\leq 2}^{\perp}+b_z\}$. Here $z_{\leq 2}^{\perp}\neq 0, b_z$ are arbitrarily fixed elements in $\mathbb{F}_q^2$ such that $z_{\leq 2}^{\perp}\cdot z_{\leq 2}=0, b_z\cdot z_{\leq 2}=2^{-1}(z_3+z_1^2+z_2^2)$ respectively. Thus
\begin{equation}  \label{EqFiniteFieldsSummand3}
\text{RHS of \eqref{EqFiniteFieldsSummand2}}=\mathbb{E}_{t\in \mathbb{F}_q} e \big(\tau_{x}P(tz_{\leq 2}^{\perp}+b_z)-\tau_{x-z}P(tz_{\leq 2}^{\perp}+b_z-z_{\leq 2}) \big)=:\mathbb{E}_{t\in \mathbb{F}_q} e \big( Q_z(t) \big).
\end{equation}
In particular, \eqref{EqFiniteFieldsSummand3} implies $|\widetilde{K}(x, x-z)|\leq 1$ when $z_{\leq 2}\neq 0$. To prove \eqref{EqFiniteFieldsSchurTest1}, we shall select an exceptional set $\Upsilon\subset \mathbb{F}_q^3\backslash \{z: z_{\leq 2}=0\}$ such that $\#\Upsilon\lesssim q^{-2c}\cdot q^3$ and $|\widetilde{K}(x, x-z)|\lesssim q^{-2c}$ for $z\in \Upsilon_{\neq 0}^c:=\mathbb{F}_q^3\backslash (\Upsilon\cup \{z: z_{\leq 2}=0\})$. With this $\Upsilon$, we can use its properties and \eqref{EqFiniteFieldsRDegenerateCase} to estimate
\begin{equation*}
\begin{split}
\mathbb{E}_{z\in \mathbb{F}_q^3} |\widetilde{K}(x, x-z)| & =\frac{1}{q^3} \sum_{z: z_{\leq 2}=0} |\widetilde{K}(x, x-z)|+\frac{1}{q^3}\sum_{z\in \Upsilon}|\widetilde{K}(x, x-z)|+\frac{1}{q^3}\sum_{z\in \Upsilon_{\neq 0}^c} |\widetilde{K}(x, x-z)| \\
& \lesssim \frac{1}{q^3}\cdot q+\frac{1}{q^3}\#\Upsilon+\frac{1}{q^3}\cdot q^3\cdot q^{-2c}\lesssim q^{-2c},
\end{split}
\end{equation*}
which gives \eqref{EqFiniteFieldsSchurTest1}.

We select $\Upsilon$ to ensure that the polynomial $Q_z(t)$ in RHS of \eqref{EqFiniteFieldsSummand3} has degree strictly larger than zero for $z\in \Upsilon_{\neq 0}^c$. The coefficients of $t^3$ and $t^2$ are
\begin{align}
& (\tau_{x}-\tau_{x-z})\big( (z_{\leq 2}^{\perp})_1^3+(z_{\leq 2}^{\perp})_1^3 \big), \label{EqFiniteFieldsCoefficient1} \\
(\tau_{x}-\tau_{x-z})\big( (z_{\leq 2}^{\perp})_1^2(b_z)_1 & +(z_{\leq 2}^{\perp})_2^2(b_z)_2 \big)+\tau_{x-z}\big( (z_{\leq 2}^{\perp})_1^2z_1+(z_{\leq 2}^{\perp})_2^2z_2 \big), \label{EqFiniteFieldsCoefficient2}
\end{align}
respectively. Let $\Upsilon:=\{z: z_{\leq 2}\neq 0, (z_{\leq 2}^{\perp})_1^3+(z_{\leq 2}^{\perp})_1^3=0 \text{ or } (z_{\leq 2}^{\perp})_1^2z_1+(z_{\leq 2}^{\perp})_2^2z_2=0 \}$. Recall $\tau_x\neq 0, \forall x\in \mathbb{F}_q^3$. We see either $t^3$ or $t^2$ has a non-zero coefficient in $Q_z(t)$ when $z\in \Upsilon_{\neq 0}^c$. In this case, the desired estimate $|\widetilde{K}(x, x-z)|\lesssim q^{-2c}$ follows from Lemma \ref{LemmaExponentialSumFiniteFields}. It remains to check $\#\Upsilon\lesssim q^{-2c}\cdot q^3$. Let $z\in \Upsilon$. WLOG assume $z_1\neq 0$. Then we can write $z_2=rz_1$, and $z_{\leq 2}^{\perp}\neq 0$ is parallel with $(-r, 1)$. Plug these into the definition of $\Upsilon$. We see that $r$ is in the zero sets of $(-r)^3+1$ or $(-r)^2+r$, which means $\Upsilon$ is contained in the union of $O(1)$ many two-dimensional subspaces of $\mathbb{F}_q^3$. This implies $\#\Upsilon\lesssim q^{-1}\cdot q^3$ and completes the proof.

<!-- lemma-5: body_md -->
\label{LemmaPopularityPrinciple}
Let $0\leq Y\leq 1$ be a random variable in an arbitrary probability space. Let $\lambda\in (0,1)$. Suppose $\mathbb{E}Y>\lambda$. Then $P(Y>\frac{\lambda}{2})>\frac{\lambda}{2}$.
<!-- end lemma-5 -->
<!-- proof-2: body_md -->
Assume $P(Y>\frac{\lambda}{2})\leq \frac{\lambda}{2}$. Then
\begin{equation*}
\mathbb{E}Y=\mathbb{E}(Y: Y>\frac{\lambda}{2})+\mathbb{E}(Y: Y>\frac{\lambda}{2})\leq 1\cdot \frac{\lambda}{2}+ \frac{\lambda}{2}\cdot 1=\lambda,
\end{equation*}
which contradicts the condition.
<!-- end proof-2 -->
<!-- lemma-6: body_md -->
\label{LemmaFiniteFieldsMajorArc2}
Let $q$ be a prime and $X$ be a subset of $\mathbb{F}_q^3$. Let $\tau_{(\cdot)}: X\rightarrow \mathbb{F}_q$ be any function. Let $\delta\in (0, 1)$ such that $q>C_0\delta^{-C_0}$. Suppose the following holds for some $f$ ($Af$ defined by \eqref{EqFiniteFieldsAveragingOperator}):
\begin{equation*}
(\mathbb{E}_{x\in X} |Af|^2)^{\frac{1}{2}}>\delta\|f\|_2.
\end{equation*}
Then there is $X_0\subset X$ with $\#X_0\gtrsim \delta^{O(1)}q^3$ such that $\tau_{x}=0, \forall x\in X_0$.
<!-- end lemma-6 -->
<!-- proof-3: body_md -->
We use the same notations as in the proof of Lemma \ref{LemmaFiniteFieldsMajorArc1}. Note that $(\mathbb{E}_{x\in X} |Af|^2)^{\frac{1}{2}}=\|1_{x\in X}Af\|_2$. Use Plancherel as before. We see the operator $1_{x\in X}A$ with kernel $1_{x\in X}K(x, \xi)$ has operator norm larger than $\delta$, which means $(1_{x\in X}A)(1_{x\in X}A)^*$ has operator norm larger than $\delta^2$. Apply Schur's test in its contrapositive. We get there exists an $x\in \mathbb{F}_q^3$ such that
\begin{equation*}
\mathbb{E}_{z\in \mathbb{F}_q^3} |1_{x\in X}\widetilde{K}(x, z)1_{z\in X}|>\delta^2.
\end{equation*}
Let $z\mapsto x-z$ and split the sum as before. We have
\begin{equation*}
\mathbb{E}_{z\in \mathbb{F}_q^3} |1_{x\in X}\widetilde{K}(x, x-z)1_{x-z\in X}|\leq O(q^{-2c})+\frac{1}{q^3}\sum_{z\in \Upsilon_{\neq 0}^c} |\widetilde{K}(x, x-z)1_{x-z\in X}|.
\end{equation*}
Thus
\begin{equation*}
\frac{1}{q^3}\sum_{z\in \Upsilon_{\neq 0}^c} |\widetilde{K}(x, x-z)1_{x-z\in X}|>\frac{\delta^2}{2}.
\end{equation*}
Then Lemma \ref{LemmaPopularityPrinciple} gives us a set $Z$ with $\#Z>\frac{\delta^2}{4}q^3$ such that for any $z\in Z$ we have $z\in \Upsilon_{\neq 0}^c$, $x-z\in X$, and 
\begin{equation*}
|\widetilde{K}(x, x-z)|>\frac{\delta^2}{4}.
\end{equation*}
For each such $z$, the two coefficients \eqref{EqFiniteFieldsCoefficient1}\eqref{EqFiniteFieldsCoefficient2} must be $0$ simultaneously; otherwise Lemma \ref{LemmaExponentialSumFiniteFields} gives a contradiction. Our selection of $\Upsilon$ then implies $\tau_{x}-\tau_{x-z}=0, \tau_{x-z}=0$. Hence $X_0:=x-Z$ satisfies the requirements.
<!-- end proof-3 -->
<!-- lemma-7: body_md -->
\label{LemmaContinuousMajorArc}
Let $\lambda\gtrsim 1$. Let $\tau_{(\cdot)}: \mathbb{R}^3\rightarrow \mathbb{R}$ be any measurable function such that $|\tau_{x}|>\lambda, \forall x\in \mathbb{R}^3$. Let
\begin{equation*}
Af(x):=\int_{\mathbb{R}^2} f \big( x-\Gamma(y) \big) e^{2\pi i \tau_x P(y)} \rho (y) dy,
\end{equation*}
where $\rho\in \mathcal{S}(\mathbb{R}^3)$ is supported in $\{|y|<5\}$ with $\|\rho\|_{C^1}\leq 1$. Then we have
\begin{equation*}
\|Af\|_2\lesssim \lambda^{-c}\|f\|_2, \ \ \ \forall f\in L^2(\mathbb{R}^3).
\end{equation*}
<!-- end lemma-7 -->
<!-- proof-4: body_md -->
Note that the the operator $Af$ is local, so it suffices to show $\|Af\|_{L^2(B)}\lesssim \lambda^{-c}\|f\|_2$ for any fixed ball $B$ of radius $1$. By approximation, we can also assume $\hat{f}$ is supported in $\{|\xi|<R\}$ for some large $N$. Let $\{|\xi|\leq 1\}\prec \chi\prec \{|\xi|\leq 2\}$ be a bump function. We only need to show $\big{\|} 1_{B}(x)A \big( \chi(\frac{D}{R})f \big)(x) \big{\|}_2\lesssim \lambda^{-c}\|f\|_2$. Write $1_{B}(x)A \big( \chi(\frac{D}{R})f \big)(x)$ into a kernel representation on the Fourier side and apply Plancherel as before. Our goal becomes to show the following operator has operator norm $O(\lambda^{-c})$:
\begin{equation*}
\mathfrak{A}f:=\int_{\mathbb{R}^3} K(x, \xi) f(\xi) d\xi:=\int_{\mathbb{R}^3} e^{2\pi i x\cdot \xi} 1_B(x) \big( \int_{\mathbb{R}^2} e^{2\pi i\big( -\xi\cdot\Gamma(y)+\tau_{x}P(y)\big) }\rho (y) dy \big) \chi(\frac{\xi}{R}) f(\xi) d\xi.
\end{equation*}
Then we compute the kernel of $\mathfrak{A}\mathfrak{A}^*$ as
\begin{equation*}
\begin{split}
&\widetilde{K}(x, z):=\int_{\mathbb{R}^3} K(x, \xi)\overline{K(z, \xi)} d\xi \\
&=1_{B^2}(x, z)\int_{\mathbb{R}^3} e^{2\pi i( (x-z)\cdot\xi )} \chi^2(\frac{\xi}{R}) \big( \int_{(\mathbb{R}^2)^2} e^{2\pi i\big( -\xi\cdot\Gamma(y)+\tau_{x}P(y)\big) } e^{-2\pi i\big( -\xi\cdot\Gamma(y')+\tau_{z}P(y')\big) }\rho(y)\rho(y') \big) \\
&=1_{B^2}(x, z)\int_{(\mathbb{R}^2)^2} e^{2\pi i\big( \tau_{x}P(y)-\tau_{z}P(y') \big)} (\chi^2)_R^{\vee} \Big( x-z-\big( \Gamma(y)-\Gamma(y') \big) \Big) \rho(y)\rho(y').
\end{split}
\end{equation*}
Apply Schur's test, use the symmetry of $\widetilde{K}(x, z)$, and let $z\mapsto x-z$ as in the proof of Lemma \ref{LemmaFiniteFieldsMajorArc1}. It suffices to show the following for fixed $x$:
\begin{equation}  \label{EqContinuousSchurTest}
\int_{|z|<2} \Big| \int_{(\mathbb{R}^2)^2} e^{2\pi i\big( \tau_{x}P(y)-\tau_{x-z}P(y') \big)} (\chi^2)_R^{\vee} \Big( z-\big( \Gamma(y)-\Gamma(y') \big) \Big) \rho(y)\rho(y') \Big|\lesssim \lambda^{-2c}.
\end{equation}
Let $\widetilde{\mathcal{K}}(x, x-z)$ denote the inner integral $\int_{(\mathbb{R}^2)^2}\cdots$ in LHS of \eqref{EqContinuousSchurTest}. We now change the variable to fix the entry of $(\chi^2)_R^{\vee}(\cdot)$ so that we can exploit the oscillation. Let $w:=y-y'$ and $v:=\frac{w}{|w|}$. (Only need to consider the case $w\neq 0$.) Define $v^{\perp}$ to be the unique unit vector such that $v^{\perp}\cdot v=0$ and $\det (v, v^{\perp})>0$. Let $y=tv^{\perp}+sv=:U(t,s)$. Then
\begin{equation*}
\Gamma(y)-\Gamma(y')=(w, -|w|^2+2y\cdot w)=(w, -|w|^2+2s|w|).
\end{equation*}
Thus
\begin{equation} \label{EqContinuousSummand}
\begin{split}
\widetilde{\mathcal{K}}(x, x-z) & =\int_{w\in \mathbb{R}^2} \int_{s\in \mathbb{R}} (\chi^2)_R^{\vee}\big (z-(w, -|w|^2+2s|w|) \big) I(w, s) \\
&=\int_{0<|w|<10} \int_{|s|\lesssim 1} (\chi^2)_R^{\vee}\big (z-(w, -|w|^2+2s|w|) \big) I(w, s).
\end{split}
\end{equation}
Here
\begin{equation} \label{EqContinuousSummandOscillatoryPart}
I(w, s):=\int_{t\in \mathbb{R}} e^{2\pi i\big( \tau_{x}P(tv^{\perp}+sv)-\tau_{x-z}P(tv^{\perp}+sv-w) \big) } \rho\big( U(t, s) \big)\rho\big( U(t, s)-w \big).
\end{equation}
We shall select an exceptional set $\Upsilon\subset \{0<|w|<10\}$ such that $|\Upsilon|\lesssim \lambda^{-2c}$ and $|I(w, s)|\lesssim \lambda^{-2c}$ for $w\in \{0<|w|<10\}\backslash \Upsilon, |s|\lesssim 1$. With this $\Upsilon$, we can then use its properties to estimate
\begin{equation*}
|\widetilde{\mathcal{K}}(x, x-z)|\lesssim \int_{0<|w|<10} \int_{|s|\lesssim 1} \big| (\chi^2)_R^{\vee}\big (z-(w, -|w|^2+2s|w|) \big) \big| \big(\lambda^{-2c}+1_{\Upsilon}(w) \big),
\end{equation*}
so that $\text{LHS of \eqref{EqContinuousSchurTest}}=\int_{|z|<2} |\widetilde{\mathcal{K}}(x, x-z)|$ is bounded by
\begin{equation*}
\begin{split}
\int_{|z|<2} \int_{0<|w|<10} & \int_{|s|\lesssim 1}\big| (\chi^2)_R^{\vee}\big (z-(w, -|w|^2+2s|w|) \big) \big|\big(\lambda^{-2c}+1_{\Upsilon}(w) \big) \\
&\lesssim \lambda^{-2c}+\int_{|s|\lesssim 1} \int_{0<|w|<10} \int_{|z|<2}\big| (\chi^2)_R^{\vee}\big (z-(w, -|w|^2+2s|w|) \big) \big| 1_{\Upsilon}(w)  \\
&\lesssim \lambda^{-2c}+\int_{|s|\lesssim 1} \int_{0<|w|<10} 1_{\Upsilon}(w)\lesssim \lambda^{-2c}
\end{split}
\end{equation*}
as desired. 

We select $\Upsilon$ to ensure that the polynomial $Q_{w, s}(t):=\tau_{x}P(tv^{\perp}+sv)-\tau_{x-z}P(tv^{\perp}+sv-w)$ in RHS of \eqref{EqContinuousSummandOscillatoryPart} has a large cubic or quadratic coefficient for $w\in \{0<|w|<10\}\backslash\Upsilon$. The coefficients of $t^3$ and $t^2$ are
\begin{align}
& (\tau_{x}-\tau_{x-z})\big( (v_1^{\perp})^3+(v_2^{\perp})^3 \big), \label{EqContinuousCoefficient1} \\
(\tau_{x}-\tau_{x-z})\big( (v_1^{\perp})^2sv_1 & +(v_2^{\perp})^2sv_2 \big)+\tau_{x-z}\big( (v_1^{\perp})^2w_1+(v_2^{\perp})^2w_2 \big), \label{EqContinuousCoefficient2}
\end{align}
respectively. Recall $|\tau_{x}|>\lambda, \forall x\in \mathbb{R}^3$. Let $\Upsilon:=\{0<|w|<10: |(v_1^{\perp})^3+(v_2^{\perp})^3|\leq \lambda^{-c_1} \text{ or } |(v_1^{\perp})^2w_1+(v_2^{\perp})^2w_2|\leq \lambda^{-c_1} \}$. Let $w\in \{0<|w|<10\}\backslash\Upsilon$. Then if $|\tau_{x}-\tau_{x-z}|>\lambda^{2c_1}$, we have $|\eqref{EqContinuousCoefficient1}|>\lambda^{c_1}$; and if $|\tau_{x}-\tau_{x-z}|\leq \lambda^{2c_1}$, we have $|\eqref{EqContinuousCoefficient2}|>\lambda^{1-c_1}-O(\lambda^{2c_1})\gtrsim \lambda^{1-c_1}$. In each case, the estimate $|I(w, s)|\lesssim \lambda^{-2c}$ follows from the van der Corput lemma \cite[Proposition 2.1]{SWPolynomialCarleson}. It remains to check $|\Upsilon|\lesssim \lambda^{-2c}$. This is straightforward via the polar coordinate system $w=r(\cos \theta, \sin \theta)$. We have $v^{\perp}=(-\sin \theta, \cos \theta)$. Thus
\begin{align*}
|(v_1^{\perp})^3+(v_2^{\perp})^3|\leq \lambda^{-c_1} & \Longrightarrow |-\sin \theta+\cos \theta|\lesssim \lambda^{-c_1}, \\
|(v_1^{\perp})^2w_1+(v_2^{\perp})^2w_2|\leq \lambda^{-c_1}& \Longrightarrow |\sin \theta|\lesssim \lambda^{-\frac{c_1}{3}} \text{ or } |\cos \theta|\lesssim \lambda^{-\frac{c_1}{3}} \text{ or } |\sin \theta+\cos \theta|\lesssim \lambda^{-\frac{c_1}{3}},
\end{align*}
which implies $|\Upsilon|\lesssim \lambda^{-\frac{c_1}{3}}$ and completes the proof.

<!-- lemma-8: body_md -->
\label{LemmaFindingPatternsDual-DifferenceInterchange}
Let $Y$ be any index set and $F_y: \mathbb{F}_q\rightarrow \mathbb{C}$ be $1$-bounded functions for $y\in Y$. Set $F(x):=\mathbb{E}_{y\in Y} F_y(x)$. Then for any $\mathcal{H}\subset \mathbb{F}_q^n$ and $\phi: \mathbb{F}_q^n\rightarrow \mathbb{F}_q$ we have
\begin{equation*}
\Big( \mathbb{E}_{h\in \mathcal{H}} \big| \mathbb{E}_{x\in \mathbb{F}_q}\Delta_h F(x)e\big( \phi(h)x \big) \big|   \Big)^{2^n}\lesssim \mathbb{E}_{h^{(0)}, h^{(1)}\in \mathcal{H}} \big| \mathbb{E}_{x\in \mathbb{F}_q} \mathbb{E}_{y\in Y} (\Delta_{h^{(0)}-h^{(1)}} F_y)(x)e\big( \Phi(h^{(0)}, h^{(1)})x \big) \big|.
\end{equation*}
Here $\Phi(h^{(0)}, h^{(1)}):=\sum_{\omega\in \{0,1\}^n} (-1)^{|\omega|} \phi(h^{(\omega)})$ with $h^{(\omega)}:=(h_1^{(\omega_1)}, \cdots, h_n^{(\omega_n)})$.
<!-- end lemma-8 -->
<!-- lemma-9: body_md -->
\label{LemmaFindingPatternsLowRankControl}
Let $0\leq m\leq n$ and $\phi_1, \cdots, \phi_m: \mathbb{F}_q^{n-1}\rightarrow \mathbb{F}_q$ be arbitrary functions. Let $f: \mathbb{F}_q\rightarrow \mathbb{C}$ be $1$-bounded. Then
\begin{equation*}
\mathbb{E}_{h_1, \cdots, h_n\in \mathbb{F}_q} \Big| \mathbb{E}_{x\in \mathbb{F}_q} \Delta_h f(x)e \Big( \big( \sum_{j=1}^m \phi_j(h_1, \cdots, h_{j-1}, h_{j+1}, \cdots, h_n) \big)x \Big) \Big|\lesssim \|f\|_{U^{n+1}}^{2^{n-m}}.
\end{equation*}
<!-- end lemma-9 -->
<!-- proof-7: body_md -->
First assume $n=2$ which is the easiest case. Our condition reads as
\begin{equation*}
\mathbb{E}_{x_{\leq 3}\in \mathbb{F}_q^3} \|F_2(x_{\leq 3}, \cdot )\|_{U^2}^{4}>\delta.
\end{equation*}
By Lemma \ref{LemmaPopularityPrinciple}, we can find a set $\Upsilon\subset \mathbb{F}_q^3$ with $\#\Upsilon>\frac{\delta}{2}q^3$ such that
\begin{equation*}
\|F_2(x_{\leq 3}, \cdot )\|_{U^2}^{4}>\frac{\delta}{2},\ \forall x_{\leq 3}\in \Upsilon.
\end{equation*}
Recall that the $U^2$ inverse theorem $\|g\|_{U^2}\leq \|\hat{g}\|_{\infty}^{\frac{1}{2}}$ holds for any 1-bounded function $g$. Apply this to each $x_{\leq 3}\in \Upsilon$ and find $\phi_{x_{\leq 3}}\in \mathbb{F}_q$ which maximizes $\|\widehat{F_2(x_{\leq 3}, \cdot )}\|_{\infty}$. We get
\begin{equation} \label{EqFindingPatternsDegree-Lowering1}
|\mathbb{E}_{x_4\in \mathbb{F}_q} F_2(x_{\leq 3}, x_4) e(-\phi_{x_{\leq 3}}\cdot x_4)|>(\frac{\delta}{2})^{\frac{1}{2}},\ \forall x_{\leq 3}\in \Upsilon.
\end{equation}
Summing \eqref{EqFindingPatternsDegree-Lowering1} over $x_{\leq 3}\in \Upsilon$ and expanding the definition of $F_2$, we obtain
\begin{equation*}   
\mathbb{E}_{x_{\leq 3}\in \Upsilon} \big| \mathbb{E}_{x_4\in \mathbb{F}_q} \mathbb{E}_{y\in \mathbb{F}_q^2} f_0\big( x_{\leq 3}, x_4+P(y) \big)f_1 \big( x_{\leq 3}-\Gamma(y), x_4+P(y) \big) e(-\phi_{x_{\leq 3}}\cdot x_4)  \big|\gtrsim \delta^{O(1)}.
\end{equation*}
Let $x_4\mapsto x_4-P(y)$, take $\mathbb{E}_{x_4\in \mathbb{F}_q}$ outside the absolute value, and then pigeonhole on $x_4$. We get one $x_4\in \mathbb{F}_q$ such that
\begin{equation*}
\mathbb{E}_{x_{\leq 3}\in \Upsilon} \big| \mathbb{E}_{y\in \mathbb{F}_q^2} f_1 \big( x_{\leq 3}-\Gamma(y), x_4 \big) e \big( \phi_{x_{\leq 3}}\cdot P(y) \big)  \big|\gtrsim \delta^{O(1)}.
\end{equation*}
We apply Lemma \ref{LemmaFiniteFieldsMajorArc2} to find $\Upsilon_0\subset \Upsilon$ with $\Upsilon_0\gtrsim \delta^{O(1)}q^3$ such that $\phi_{x_{\leq 3}}=0, \forall x_{\leq 3}\in \Upsilon_0$. Now go back to \eqref{EqFindingPatternsDegree-Lowering1}. Summing \eqref{EqFindingPatternsDegree-Lowering1} over $x_{\leq 3}\in \Upsilon_0$ and extending the sum to all $x_{\leq 3}\in \mathbb{F}_q^3$ by positivity, we obtain
\begin{equation*}
\mathbb{E}_{x_{\leq 3}\in \mathbb{F}_q^3} |\mathbb{E}_{x_4\in \mathbb{F}_q} F_2(x_{\leq 3}, x_4) \big| \gtrsim \delta^{O(1)}.
\end{equation*}
This gives the conclusion by Cauchy--Schwartz.

Then we handle the $n\geq 3$ case. Write our condition as
\begin{equation*}
\mathbb{E}_{x_{\leq 3}\in \mathbb{F}_q^3} \mathbb{E}_{h\in \mathbb{F}_q^{n-2}}  \|(\Delta_h F_2)(x_{\leq 3}, \cdot )\|_{U^2}^{4}>\delta.
\end{equation*}
By Lemma \ref{LemmaPopularityPrinciple}, we can find a set $\Upsilon\subset \mathbb{F}_q^3\times \mathbb{F}_q^{n-2}$ with $\#\Upsilon>\frac{\delta}{2}q^{n+1}$ such that
\begin{equation*}
\|(\Delta_h F_2)(x_{\leq 3}, \cdot )\|_{U^2}^{4}>\frac{\delta}{2},\ \forall (x_{\leq 3}, h)\in \Upsilon.
\end{equation*}
We shall use the following fibers of $\Upsilon$:
\begin{equation*}
\mathcal{H}(x_{\leq 3}):=\{h\in \mathbb{F}_q^{n-2}: (x_{\leq 3}, h)\in \Upsilon \},\ \ \ X(h):=\{x_{\leq 3}\in \mathbb{F}_q^3: (x_{\leq 3}, h)\in \Upsilon \}.
\end{equation*}
For each $(x_{\leq 3}, h)\in \Upsilon$, apply the $U^2$ inverse theorem and find $\phi_{x_{\leq 3}}(h)\in \mathbb{F}_q$ which maximizes $\|\widehat{(\Delta_h F_2)(x_{\leq 3}, \cdot )}\|_{\infty}$. We get
\begin{equation} \label{EqFindingPatternsDegree-Lowering2}
\big| \mathbb{E}_{x_4\in \mathbb{F}_q} (\Delta_h F_2)(x_{\leq 3}, x_4) e \big( -\phi_{x_{\leq 3}}(h)\cdot x_4 \big) \big| >(\frac{\delta}{2})^{\frac{1}{2}},\ \forall (x_{\leq 3}, h)\in \Upsilon.
\end{equation}
Extend $\phi_{x_{\leq 3}}(h)$ to the whole $\mathbb{F}_q^3\times \mathbb{F}_q^{n-2}$ by arbitrarily defining $\phi_{x_{\leq 3}}(h)$ for $(x_{\leq 3}, h)\notin \Upsilon$. Sum \eqref{EqFindingPatternsDegree-Lowering2} over $(x_{\leq 3}, h)\in \Upsilon$:
\begin{equation*}
\mathbb{E}_{x_{\leq 3}\in \mathbb{F}_q^3}\mathbb{E}_{h\in \mathcal{H}(x_{\leq 3})} \Big( \big| \mathbb{E}_{x_4\in \mathbb{F}_q} (\Delta_h F_2)(x_{\leq 3}, x_4) e \big( -\phi_{x_{\leq 3}}(h)\cdot x_4 \big) \big| \Big)^{2^{n-2}} \gtrsim \delta^{O(1)}.
\end{equation*}
Then applying Lemma \ref{LemmaFindingPatternsDual-DifferenceInterchange} yields
\begin{equation*}
\begin{split}
\mathbb{E}_{x_{\leq 3}\in \mathbb{F}_q^3}\mathbb{E}_{h^{(0)}, h^{(1)}\in \mathcal{H}(x_{\leq 3})} \big| & \mathbb{E}_{x_4\in \mathbb{F}_q} \mathbb{E}_{y\in \mathbb{F}_q^2} (\Delta_{h^{(0)}-h^{(1)}} f_0) \big( x_{\leq 3}, x_4+P(y) \big) (\Delta_{h^{(0)}-h^{(1)}} f_1) \\
& \big( x_{\leq 3}-\Gamma(y), x_4+P(y) \big) e \big( -\Phi_{x_{\leq 3}}(h^{(0)}, h^{(1)})\cdot x_4 \big) \big|\gtrsim \delta^{O(1)}.
\end{split}
\end{equation*}
Control the left hand side by $\mathbb{E}_{h^{(1)}\in \mathbb{F}_q^{n-2}}\mathbb{E}_{x_{\leq 3}\in \mathbb{F}_q^3}\mathbb{E}_{h^{(0)}\in \mathcal{H}(x_{\leq 3})} |\cdots|$. By pigeonholing, we can fix one $h^{(1)}\in \mathbb{F}_q^{n-2}$ such that
\begin{equation*}
\begin{split}
\mathbb{E}_{x_{\leq 3}\in \mathbb{F}_q^3}\mathbb{E}_{h^{(0)}\in \mathcal{H}(x_{\leq 3})} \big| & \mathbb{E}_{x_4\in \mathbb{F}_q} \mathbb{E}_{y\in \mathbb{F}_q^2} (\Delta_{h^{(0)}-h^{(1)}} f_0) \big( x_{\leq 3}, x_4+P(y) \big) (\Delta_{h^{(0)}-h^{(1)}} f_1) \\
& \big( x_{\leq 3}-\Gamma(y), x_4+P(y) \big) e \big( -\Phi_{x_{\leq 3}}(h^{(0)}, h^{(1)})\cdot x_4 \big) \big| \gtrsim \delta^{O(1)}.
\end{split}
\end{equation*}
Let $x_4\mapsto x_4-P(y)$, take $\mathbb{E}_{x_4\in \mathbb{F}_q}$ outside the absolute value, and then pigeonhole on $x_4$. We get one $x_4\in \mathbb{F}_q$ such that
\begin{equation*}
\mathbb{E}_{(x_{\leq 3}, h^{(0)})\in \Upsilon} \big| \mathbb{E}_{y\in \mathbb{F}_q^2} (\Delta_{h^{(0)}-h^{(1)}} f_1)\big( x_{\leq 3}-\Gamma(y), x_4 \big) e \big( \Phi_{x_{\leq 3}}(h^{(0)}, h^{(1)})\cdot P(y) \big)  \big|\gtrsim \delta^{O(1)}.
\end{equation*}
Write $\mathbb{E}_{(x_{\leq 3}, h^{(0)})\in \Upsilon}=\mathbb{E}_{h^{(0)}\in \mathbb{F}_q^{n-2}}\mathbb{E}_{x_{\leq 3}\in X(h^{(0)})}$. Applying Lemma \ref{LemmaPopularityPrinciple} gives $\gtrsim \delta^{O((1)}q^{n-2}$ many $h^{(0)}$ such that
\begin{equation*}
\mathbb{E}_{x_{\leq 3}\in X(h^{(0)})} \big| \mathbb{E}_{y\in \mathbb{F}_q^2} (\Delta_{h^{(0)}-h^{(1)}} f_1)\big( x_{\leq 3}-\Gamma(y), x_4 \big) e \big( \Phi_{x_{\leq 3}}(h^{(0)}, h^{(1)})\cdot P(y) \big) \big|\gtrsim \delta^{O(1)}.
\end{equation*}
We apply Lemma \ref{LemmaFiniteFieldsMajorArc2} to find $X_0(h^{(0)})\subset X(h^{(0)})$ with $X_0(h^{(0)})\gtrsim \delta^{O(1)}q^3$ and $\Phi_{x_{\leq 3}}(h^{(0)}, h^{(1)})=0, \forall x_{\leq 3}\in X_0(h^{(0)})$. Altogether, we get $\gtrsim \delta^{O((1)}q^{n+1}$ many pairs $(x_{\leq 3}, h^{(0)})\in \Upsilon$ such that $\Phi_{x_{\leq 3}}(h^{(0)}, h^{(1)})=0$. Let $\Upsilon'$ denote the set of these pairs. For each $(x_{\leq 3}, h^{(0)})\in \Upsilon'$, expanding the definition of $\Phi_{x_{\leq 3}}(h^{(0)}, h^{(1)})$ yields
\begin{equation}  \label{EqFindingPatternsDegree-Lowering3}
\phi_{x_{\leq 3}}(h^{(0)})=-\sum_{\omega\in \{0,1\}^n\backslash \{(0, \cdots, 0)\}} (-1)^{|\omega|} \phi_{x_{\leq 3}}(h^{(\omega)}),\ \forall (x_{\leq 3}, h^{(0)})\in \Upsilon'. 
\end{equation}
Let $\psi_{x_{\leq 3}}(h^{(0)})$ denote RHS of \eqref{EqFindingPatternsDegree-Lowering3} for any $h^{(0)}\in \mathbb{F}_q^{n-2}$. Recall that $h^{(1)}$ is fixed, so $\psi_{x_{\leq 3}}(h^{(0)})$ has rank lower than $n-2$ in the sense that each summand in RHS of \eqref{EqFindingPatternsDegree-Lowering3} depends on at most $n-3$ coordinates of $h^{(0)}$. Now go back to \eqref{EqFindingPatternsDegree-Lowering2}. Since $\#\Upsilon'\gtrsim \delta^{O(1)}q^{n+1}$, Lemma \ref{LemmaPopularityPrinciple} implies that there are $\gtrsim \delta^{O(1)}q^{3}$ many $x_{\leq 3}$ such that $\mathcal{H}'(x_{\leq 3}):=\{h\in \mathbb{F}_q^{n-2}: (x_{\leq 3}, h)\in \Upsilon'\}$ has cardinality $\gtrsim \delta^{O(1)}q^{n-2}$. We obtain ($\phi_{x_{\leq 3}}(h)=\psi_{x_{\leq 3}}(h)$ for $h\in \mathcal{H}'(x_{\leq 3})$)
\begin{equation*}
\mathbb{E}_{h\in \mathbb{F}_q^{n-2}}\big| \mathbb{E}_{x_4\in \mathbb{F}_q} (\Delta_h F_2)(x_{\leq 3}, x_4) e \big( -\psi_{x_{\leq 3}}(h)\cdot x_4 \big) \big|
\end{equation*}
Applying Lemma \ref{LemmaFindingPatternsLowRankControl} with $m, n$ there both replaced by $n-2$, we get $\|F_2(x_{\leq 3}, \cdot)\|_{U^{n-1}}\gtrsim \delta^{O(1)}$. This holds for $\gtrsim \delta^{O(1)}q^{3}$ many $x_{\leq 3}$. Summing these $x_{\leq 3}$ gives the desired conclusion.

Define the maximal Radon transforms $M^{\Gamma}$ and $M^{\Gamma, K}$ by

\end{equation*}

<!-- proof-9: body_md -->
Note that we have
\begin{equation*}
|T_l f(x)| \lesssim M^{\Gamma}f(x),
\end{equation*}
so $\|T_l f\|_p\lesssim \|f\|_p$ for $1<p<\infty$. By interpolation, it suffices to show $\|T_l f\|_2\lesssim 2^{-cl}\|f\|_2$. We can assume $l\gtrsim 1$.

The argument is based on a two-step paraproduct decomposition. For each $|s|\leq N$, we decompose
\begin{equation*}
1_{\xi\neq 0}=\sum_{k, k'\in \mathbb{Z}^2} \psi_{-s+k}(\xi_{\leq 2})\psi_{-2s+k'} (\xi_3).
\end{equation*}
Here the first $\psi$ is a two-dimensional Littlewood--Paley projection, and the second is one-dimensional. Plug this into $T_l$. We write $T_l f$ as
\begin{equation}  \label{EqPYMainDecomposition}
\sum_{|s|\leq N}\sum_{(k, k')\in \mathbb{Z}^2} 1_{|\tau_x|\sim 2^{-3s+l}}\int_{\mathbb{R}^2} \big( \psi_{-s+k}(D_{\leq 2}) \otimes \psi_{-2s+k'} (D_3)f \big) \big( x-\Gamma(y) \big) e^{2\pi i \tau_x P(y)} K_s(y).
\end{equation}
Note that our assumption on $f$ implies there are only finitely many non-zero summands above, so we can freely reorganize the sum. For any $\Upsilon\subset \mathbb{Z}^2$, we define $T_l^{\Upsilon} f$ to be the operator in \eqref{EqPYMainDecomposition} with $\sum_{(k, k')\in \mathbb{Z}^2}$ replaced by $\sum_{(k, k')\in \Upsilon}$. Fix a large absolute constant $C$. We partition $\mathbb{Z}^2$ into the following four subsets
\begin{align*}
\Upsilon_{<}&:=\big{\{} (k, k')\in \mathbb{Z}^2: \max\{k, k'\}\leq l-C \big{\}}, \\
\Upsilon_{>, \neq}&:=\big{\{} (k, k')\in \mathbb{Z}^2: \max\{k, k'\}\geq l+C, |k-k'|\geq C \big{\}}, \\
\Upsilon_{>, =}&:=\big{\{} (k, k')\in \mathbb{Z}^2: \max\{k, k'\}\geq l+C, |k-k'|<C \big{\}}, \\
\Upsilon_{=}&:=\big{\{} (k, k')\in \mathbb{Z}^2: l-C<\max\{k, k'\}< l+C \big{\}}.
\end{align*}
Decompose $T_l f=T_l^{\Upsilon_{<}}f+T_l^{\Upsilon_{>, \neq}}f+T_l^{\Upsilon_{>, =}}f+T_l^{\Upsilon_{=}}f$ accordingly. We shall prove $\|T_l^{\Upsilon}f\|_2\lesssim 2^{-cl}\|f\|_2$ for $\Upsilon=\Upsilon_{<}, \Upsilon_{>, \neq}, \Upsilon_{>, =}, \Upsilon_{=}$ separately. The off-diagonal parts $T_l^{\Upsilon_{<}}f, T_l^{\Upsilon_{>, \neq}}f$ have rapid decay, and we will control them pointwisely by the strong maximal operator. The two types of diagonal parts $T_l^{\Upsilon_{>, =}}f, T_l^{\Upsilon_{=}}f$ have frequency orthogonality for slightly different reasons, and we will sum them together using the two types of orthogonality.

Now we go to the details. Recall the strong maximal operator defined by
\begin{equation*}
M^{\text{str}} f(x):=\sup_{R_i>0, i\in [3]} \frac{1}{R_1 R_2 R_3}\int_{|y_i|\leq R_i, i\in [3]} |f(x-y)|.
\end{equation*}
One can control $M^{\text{str}} f$ by $(M_1 \circ M_2 \circ M_3)f$, where $M_i$ is the one-dimensional Hardy--Littlewood maximal operator acting on the $i\text{-th}$ coordinate only. Hence $\|M^{\text{str}} f\|_p\lesssim \|f\|_p$ for $1<p<\infty$.

For $T_l^{\Upsilon_{<}}f$, it suffices to show
\begin{equation}  \label{EqPYOperatorPartOne}
|T_l^{\Upsilon_{<}}f(x)|\lesssim 2^{-cl} M^{\text{str}} f(x),\ \forall x\in \mathbb{R}^3.
\end{equation}
Fix $x$ and abbreviate $\tau_x$ as $\tau$. There is at most one $s$ such that $|\tau|\sim 2^{-3s+l}$. Assume such an $s$ exists. Fix it as well. It suffices to show
\begin{equation}  \label{EqPYOperatorPartOne1}
\big| \sum_{(k, k')\in \Upsilon_{<}} \int_{\mathbb{R}^2} \big( \psi_{-s+k}(D_{\leq 2}) \otimes \psi_{-2s+k'} (D_3)f \big) \big( x-\Gamma(y) \big) e^{2\pi i \tau P(y)} K_s(y) \big| \lesssim 2^{-cl} M^{\text{str}} f(x).
\end{equation}
Write
\begin{equation} \label{EqPYOperatorPartOne2}
\text{LHS of \eqref{EqPYOperatorPartOne1}}=\big| \int_{\mathbb{R}^2} \big( \phi_{-s+l-C}(D_{\leq 2}) \otimes \phi_{-2s+l-C} (D_3)f \big) \big( x-\Gamma(y) \big) e^{2\pi i \tau_x P(y)} K_s(y) \big|.
\end{equation}
Note that $\big( \phi_{-s+l-C}(D_{\leq 2})\otimes \phi_{-2s+l-C} (D_3)f \big) \big( x-\Gamma(y) \big)$ is equal to
\begin{equation*}
\begin{split}
\int_{\mathbb{R}^3} f\big( x-\Gamma(y)-t \big)& (\phi_{-s+l-C}\otimes \phi_{-2s+l-C})^{\vee}(t) dt \\
&=\int_{\mathbb{R}^3} f(x-t) (\phi_{-s+l-C}\otimes \phi_{-2s+l-C})^{\vee} \big( t-\Gamma(y) \big) dt.
\end{split}
\end{equation*}
Plug this into RHS of \eqref{EqPYOperatorPartOne2}. We see
\begin{equation} \label{EqPYOperatorPartOne3}
\begin{split}
\text{LHS of \eqref{EqPYOperatorPartOne1}} &=\Big| \int_{\mathbb{R}^3} f(x-t) \Big( \int_{\mathbb{R}^2} e^{2\pi i \tau P(y)}(\phi_{-s+l-C}\otimes \phi_{-2s+l-C})^{\vee} \big( t-\Gamma(y) \big)K_s(y) dy \Big) dt \Big| \\
&=:|\int_{\mathbb{R}^3} f(x-t) H_{s, l}^{\tau}(t) dt|.
\end{split}
\end{equation}
Moreover, changing the variables $y\mapsto 2^s y$ and using the homogeneous property of $K$, we have
\begin{equation*}
H_{s, l}^{\tau}(t)=\int_{\mathbb{R}^2} e^{2\pi i 2^{3s}\tau P(y)}2^{-4s}(\phi_{l-C}\otimes \phi_{l-C})^{\vee} \big( (\frac{t_{\leq 2}}{2^s}, \frac{t_3}{2^{2s}})-\Gamma(y) \big)K_0(y) dy=2^{-4s}H_{0, l}^{2^{3s}\tau}(\frac{t_{\leq 2}}{2^s}, \frac{t_3}{2^{2s}}).
\end{equation*}
Plug this into RHS of \eqref{EqPYOperatorPartOne3}. We see \eqref{EqPYOperatorPartOne1} is implied by the following estimate for $\tau\sim 2^l$:
\begin{equation}  \label{EqPYOperatorPartOne4}
|H_{0, l}^{\tau}(t)|\lesssim 2^{-cl}(1+|t|)^{-20}.
\end{equation}
To prove \eqref{EqPYOperatorPartOne4}, we plug the Fourier inversion formula of $(\phi_{l-C}\otimes \phi_{l-C})^{\vee}$ into $H_{0, l}^{\tau}$:
\begin{equation}  \label{EqPYOperatorPartOne5}
H_{0, l}^{\tau}(t)=\int_{\mathbb{R}^2} e^{2\pi i \tau P(y)} (\phi_{l-C}\otimes \phi_{l-C})^{\vee} \big( t-\Gamma(y) \big)K_0(y)=\int_{\mathbb{R}^3} e^{2\pi i \xi\cdot t} \phi(\frac{\xi_{\leq 2}}{2^{l-C}}) \phi(\frac{\xi_3}{2^{l-C}}) b(\xi) d\xi,
\end{equation}
where
\begin{equation}  \label{EqPYOperatorPartOne6}
b(\xi):=\int_{\mathbb{R}^2} e^{2\pi i \big( \tau P(y)-\xi\cdot \Gamma(y) \big) }K_0(y) dy.
\end{equation}
In the support of $\phi(\frac{\xi_{\leq 2}}{2^{l-C}}) \phi(\frac{\xi_3}{2^{l-C}})$ in RHS of \eqref{EqPYOperatorPartOne5}, we have
\begin{equation}  \label{EqPYOperatorPartOne7}
\big| \nabla_{y} \big( \tau P(y)-\xi\cdot \Gamma(y) \big) \big| \geq c_2|\tau|-O(|\xi|)\gtrsim 2^l,\ \forall |y|\sim 1.
\end{equation}
Thus integration by parts in \eqref{EqPYOperatorPartOne6} gives
\begin{equation*}
|\partial^{\alpha} b(\xi)|\lesssim_{N'} 2^{-lN'},\ \forall N'\in \mathbb{N}_+.
\end{equation*}
Then integration by parts in \eqref{EqPYOperatorPartOne5} gives \eqref{EqPYOperatorPartOne4}, which finishes the proof of \eqref{EqPYOperatorPartOne}.

For $T_l^{\Upsilon_{>, \neq}}f$, we prove the similar estimate
\begin{equation} \label{EqPYOperatorPartTwo}
|T_l^{\Upsilon_{>, \neq}}f(x)|\lesssim 2^{-cl} M^{\text{str}} f(x),\ \forall x\in \mathbb{R}^3.
\end{equation}
Fix $x$, abbreviate $\tau_x$ as $\tau$, and fix $s$ such that $|\tau|\sim 2^{-3s+l}$ as above. Note that
\begin{equation*}
\sum_{(k, k')\in \Upsilon_{>, \neq}} \psi_{-s+k} \otimes \psi_{-2s+k'}=\sum_{k\geq l+C} \psi_{-s+k} \otimes \phi_{-2s+k-C}+\sum_{k'\geq l+C} \phi_{-s+k'-C} \otimes \psi_{-2s+k'},
\end{equation*}
so it suffices to prove the following for $k\geq l+C$ and $k'\geq l+C$:
\begin{align}
\big| \int_{\mathbb{R}^2} \big( \psi_{-s+k}(D_{\leq 2}) \otimes \phi_{-2s+k-C} (D_3)f \big) \big( x-\Gamma(y) \big) e^{2\pi i \tau_x P(y)} K_s(y) \big| &\lesssim 2^{-ck} M^{\text{str}} f(x), \label{EqPYOperatorPartTwo1} \\
\big| \int_{\mathbb{R}^2} \big( \phi_{-s+k'-C}(D_{\leq 2}) \otimes \psi_{-2s+k'} (D_3)f \big) \big( x-\Gamma(y) \big) e^{2\pi i \tau_x P(y)} K_s(y) \big| &\lesssim 2^{-ck'} M^{\text{str}} f(x). \label{EqPYOperatorPartTwo2}
\end{align}
We prove \eqref{EqPYOperatorPartTwo1}\eqref{EqPYOperatorPartTwo2} using the same argument as in the $T_l^{\Upsilon_{<}}f$ part above. The only difference is that, instead of \eqref{EqPYOperatorPartOne7}, we use
\begin{equation*}
\big| \nabla_{y} \big( \tau P(y)-\xi\cdot \Gamma(y) \big) \big| \geq c_2|\xi_{\leq 2}|-O(|\tau|+|\xi_3|)\gtrsim 2^k,\ \forall |y|\sim 1.
\end{equation*}
for \eqref{EqPYOperatorPartTwo1} and
\begin{equation*}
\big| \nabla_{y} \big( \tau P(y)-\xi\cdot \Gamma(y) \big) \big| \geq c_2|\xi_3|-O(|\tau|+|\xi_{\leq 2}|)\gtrsim 2^{k'},\ \forall |y|\sim 1.
\end{equation*}
for \eqref{EqPYOperatorPartTwo2}. This finishes the proof of \eqref{EqPYOperatorPartTwo}.

For $T_l^{\Upsilon_{>, =}}f$, we write
\begin{equation*} 
\int_{\mathbb{R}^3} \big| \sum_{|s|\leq N}\sum_{(k, k')\in \Upsilon_{>, =}} 1_{|\tau_x|\sim 2^{-3s+l}} \int_{\mathbb{R}^2} \big( \psi_{-s+k}(D_{\leq 2}) \otimes \psi_{-2s+k'} (D_3)f \big) \big( x-\Gamma(y) \big) e^{2\pi i \tau_x P(y)} K_s(y) \big|^2
\end{equation*}
as
\begin{equation}\label{EqPYOperatorPartThree}  
\int_{\mathbb{R}^3} \sum_{|s|\leq N} \big| 1_{|\tau_x|\sim 2^{-3s+l}} \int_{\mathbb{R}^2} \big( \sum_{(k, k')\in \Upsilon_{>, =}}  \psi_{-s+k}(D_{\leq 2}) \otimes \psi_{-2s+k'} (D_3)f \big) \big( x-\Gamma(y) \big) e^{2\pi i \tau_x P(y)} K_s(y) \big|^2.
\end{equation}
For each $s$, applying a rescaled version of Lemma \ref{LemmaContinuousMajorArc} gives
\begin{equation*}
\begin{split}
\int_{\mathbb{R}^3} \big| 1_{|\tau_x|\sim 2^{-3s+l}} \int_{\mathbb{R}^2} & \big( \sum_{(k, k')\in \Upsilon_{>, =}}  \psi_{-s+k}(D_{\leq 2}) \otimes \psi_{-2s+k'} (D_3)f \big) \big( x-\Gamma(y) \big) e^{2\pi i \tau_x P(y)} K_s(y) \big|^2 \\
&\lesssim 2^{-2cl} \int_{\mathbb{R}^3} \big| \big( \sum_{(k, k')\in \Upsilon_{>, =}}  \psi_{-s+k}(D_{\leq 2}) \otimes \psi_{-2s+k'} (D_3)f \big)   \big|^2 \\
&\lesssim 2^{-2cl} \int_{\mathbb{R}^3} \sum_{(k, k')\in \Upsilon_{>, =}} \big| \big( \psi_{-s+k}(D_{\leq 2}) \otimes \psi_{-2s+k'} (D_3)f \big) \big|^2,
\end{split}
\end{equation*}
where we used Plancherel and frequency orthogonality in the last equation. Thus
\begin{equation}  \label{EqPYOperatorPartThree1}
\text{RHS of \eqref{EqPYOperatorPartThree}}\lesssim 2^{-2cl} \int_{\mathbb{R}^3} \sum_{|s|\leq N, (k, k')\in \Upsilon_{>, =}} \big| \big( \psi_{-s+k}(D_{\leq 2}) \otimes \psi_{-2s+k'} (D_3)f \big) \big|^2.
\end{equation}
When $(s, k_1, k_2)\in [-N, N]\times \Upsilon_{>, =}$ varies, the Fourier supports of $\psi_{-s+k}\otimes \psi_{-2s+k'}$ are boundedly overlapping because the following system has $O(1)$ solutions for any given $a, a'$:
\begin{equation*}
\begin{cases}
|(-s+k)-a|<100 \\
|(-2s+k')-a'|<100 \\
|k-k'|<C.
\end{cases}
\end{equation*}
Hence using Plancherel and frequency orthogonality yields
\begin{equation*}
\text{RHS of \eqref{EqPYOperatorPartThree1}}\lesssim 2^{-2cl} \int_{\mathbb{R}^3} |f|^2,
\end{equation*}
which means $\|T_l^{\Upsilon_{>, =}}f\|_2\lesssim 2^{-cl}\|f\|_2$ as desired.

For $T_l^{\Upsilon_{=}}f$, use the same argument as in the $T_l^{\Upsilon_{>, =}}f$ part above. We get that
\begin{equation*} 
\int_{\mathbb{R}^3} \big| \sum_{|s|\leq N}\sum_{(k, k')\in \Upsilon_{=}} 1_{|\tau_x|\sim 2^{-3s+l}} \int_{\mathbb{R}^2} \big( \psi_{-s+k}(D_{\leq 2}) \otimes \psi_{-2s+k'} (D_3)f \big) \big( x-\Gamma(y) \big) e^{2\pi i \tau_x P(y)} K_s(y) \big|^2
\end{equation*}
is controlled by
\begin{equation}\label{EqPYOperatorPartFour}  
2^{-2cl} \sum_{|s|\leq N} \int_{\mathbb{R}^3} \big| \big( \sum_{(k, k')\in \Upsilon_=}  \psi_{-s+k}(D_{\leq 2}) \otimes \psi_{-2s+k'} (D_3)f \big) \big|^2.
\end{equation}
The Fourier support of $\sum_{(k, k')\in \Upsilon_=}  \psi_{-s+k}\otimes \psi_{-2s+k'}$ is contained in
\begin{equation}  \label{EqPYOperatorPartFour1}
\{\xi: \frac{1}{100}\cdot 2^{-s+l-C} \leq |\xi_{\leq 2}|\leq 100\cdot 2^{-s+l+C}\}\cup \{\xi: \frac{1}{100}\cdot 2^{-2s+l-C} \leq |\xi_3|\leq 100\cdot 2^{-2s+l+C}\}.
\end{equation}
When $s$ varies, each strip in \eqref{EqPYOperatorPartFour1} varies in a boundedly overlapping manner. Hence the Fourier supports of $\sum_{(k, k')\in \Upsilon_=}  \psi_{-s+k}\otimes \psi_{-2s+k'}$ are boundedly overlapping. From Plancherel and frequency orthogonality, we deduce
\begin{equation*}
\text{RHS of \eqref{EqPYOperatorPartFour}}\lesssim 2^{-2cl} \int_{\mathbb{R}^3} |f|^2,
\end{equation*}
which means $\|T_l^{\Upsilon_=}f\|_2\lesssim 2^{-cl}\|f\|_2$ as desired. This completes the proof.
