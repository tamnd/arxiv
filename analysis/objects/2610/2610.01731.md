<!-- theorem-1: body_md -->
\label{thm:main-intro}
Let $r\ge2$, let $q$ be any prime power and set $X=\mathcal H(2r-1,q^2)$,
the Hermitian variety of even dimension $2r-2$ \textup{(}the
odd-dimensional Hermitian varieties $\mathcal H(2r,q^2)$ being covered by
Proposition~\textup{\ref{prop:belowmiddle}(i)}\textup{)}.
Then the standard conjecture of Hodge type holds for $X$\textup{:} the
symmetric bilinear form
\[
  (x,y)\;\longmapsto\;(-1)^{r-1}\langle x,y\rangle
\]
is positive definite on $\mathcal P^{2r-2}(X)_{\mathbb{Q}}$.

In fact more is true: the form is, up to one constant factor, a
euclidean inner product. Precisely, there are a linear subspace
$U_{r}\subseteq\mathbb{Q}^{\mathcal G}$ and an isomorphism of $\mathbb{Q}$-vector
spaces
\[
  \psi\;:\;U_{r}\;\xrightarrow{\ \sim\ }\;\mathcal P^{2r-2}(X)_{\mathbb{Q}}
\]
such that
\[
  (-1)^{r-1}\bigl\langle\psi(u),\psi(u')\bigr\rangle
  \;=\;q^{\,r-1}N_{r-1}\cdot\bigl\langle u,u'\bigr\rangle_{\mathrm{std}}
  \qquad\text{for all }u,u'\inU_{r} .
\]
Here $U_{r}$ is an eigenspace of an explicit integer
matrix, constructed in \S\textup{\ref{sec:decomp}}.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:tate-intro}
For every $r\ge2$ and every prime power $q$, the geometric Frobenius
$\mathrm{Fr}_{q^2}$ acts on $H^{2r-2}(X_{\overline\mathbb{F}_p},\mathbb{Q}_\ell)$ by the scalar
$q^{2r-2}$, and $H^{2k}(X_{\overline\mathbb{F}_p},\mathbb{Q}_\ell(k))$ is spanned by
algebraic cycle classes for every $k$. In particular the Tate
conjecture holds for $X$ unconditionally.

\label{thm:fermat-intro}
Let $Z$ be a nonsingular projective variety of even dimension over
$\overline{\mathbb{F}}_p$ admitting a finite surjective morphism from a
Hermitian variety --- for instance a nonsingular finite quotient of
one. Then the standard conjecture of Hodge type holds for $Z$.

<!-- corollary-1: body_md -->
\label{cor:fermat-intro}
The standard conjecture of Hodge type holds for the Fermat variety
$X_N^{2m}$ over $\overline{\mathbb{F}}_p$ whenever
\begin{equation}
\label{eq:sscongruence}
  p^{\nu}\equiv-1\pmod N\qquad\text{for some }\nu\ge1 ,
\end{equation}
which is precisely the condition for a suitable $q$ to exist. By
Theorem~II of \cite{SK1979}, for $N\ge4$ condition
\eqref{eq:sscongruence} characterises the \emph{supersingular} Fermat
varieties of even dimension, supersingular meaning here that all even
cohomology is spanned by algebraic cycle classes
\textup{(}Definition~\ref{def:ss}\textup{)}. In
odd dimension the conjecture holds for $X_N^{n}$ with no hypothesis on
$N$ beyond $p\nmid N$, by Proposition~\ref{prop:belowmiddle}\textup{(i)}.

<!-- proposition-1: body_md -->
\label{prop:belowmiddle}
Let $Y\subset\mathbb P^{n+1}$ be a nonsingular hypersurface of
dimension $n\ge1$ over an algebraically closed field, and let $L$ be
the hyperplane class. In \textup{(ii)--(v)} assume moreover that
$n=2m$ is even, and write $\langle a,b\rangle:=\deg(a\mathbin{\smallsmile} b)$ for the
Poincar\'e pairing on $H^{n}(Y,\mathbb{Q}_\ell(m))$. Then\textup{:}
\begin{enumerate}
\item[\textup{(i)}] For every $i$ with $2i<n$,
  \[
    \mathcal A^{i}(Y)=\mathbb{Q}\cdot L^{i},
    \qquad
    \mathcal A^{i}_{\mathrm{prim}}(Y)=
    \begin{cases}
      \mathbb{Q} & (i=0),\\[2pt]
      0  & (i\ge1),
    \end{cases}
  \]
  and Conjecture~\textup{\ref{conj:hdg}} holds for $Y$ in each such
  degree $i$. In particular, if $n$ is odd then
  Conjecture~\textup{\ref{conj:hdg}} holds for $Y$ outright.
\item[\textup{(ii)}] $H^{\mathrm{odd}}(Y,\mathbb{Q}_\ell)=0$, and
  $H^{2k}(Y,\mathbb{Q}_\ell(k))=\mathbb{Q}_\ell\cdot L^{k}$ is one-dimensional for
  $0\le k\le n$, $k\ne m$.
\item[\textup{(iii)}] $\langle L^{m},L^{m}\rangle=\deg Y>0$.
\item[\textup{(iv)}] $\mathcal P^{n}(Y)=\ker\bigl(L:H^{n}(Y,\mathbb{Q}_\ell(m))\to H^{n+2}(Y,\mathbb{Q}_\ell(m+1))\bigr)
  =\bigl(L^{m}\bigr)^{\perp}$.
\item[\textup{(v)}] The pairing restricts to a nondegenerate form on each
  summand of the orthogonal decomposition
  \[
    H^{n}\bigl(Y,\mathbb{Q}_\ell(m)\bigr)\;=\;\mathbb{Q}_\ell\cdot L^{m}\;\perp\;\mathcal P^{n}(Y).
  \]
\end{enumerate}
Only weak Lefschetz and Poincar\'e duality are used\textup{;} the hard
Lefschetz theorem \textup{\cite[\S4.1]{DeligneWeilII}} is not required.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
All statements follow from weak Lefschetz (Fact~\ref{fact:weakL}) and Poincar\'e
duality (Fact~\ref{fact:PD}). For $k\ne m$, $H^{2k}(Y,\mathbb{Q}_\ell(k))=\mathbb{Q}_\ell L^k$ with
$\deg L^n=\deg Y>0$. So for $2i<n$ no nonzero multiple of $L^i$ with $i\ge1$ is
primitive, and in degree $0$ the form is multiplication by $\deg Y$. For (iv),
$z\mapsto\deg(z\mathbin{\smallsmile} L^{m-1})$ is an isomorphism
$H^{n+2}(Y,\mathbb{Q}_\ell(m+1))\to\mathbb{Q}_\ell$, so $L\mathbin{\smallsmile} a=0$ if and only if $\langle a,L^m\rangle=0$.
Part (v) follows from (iii), (iv) and Fact~\ref{fact:PD}.
<!-- end proof-1 -->

<!-- fact-5: body_md -->
\label{fact:chern}
Let $Y\subset\mathbb P^{n+1}$ be a nonsingular hypersurface of degree
$d$ and dimension $n$ over an algebraically closed field of
characteristic $\ne\ell$, and write $\beta_m(Y):=\dim_{\mathbb{Q}_\ell}H^{m}(Y,\mathbb{Q}_\ell)$. Then
$\beta_m(Y)=\beta_m(\mathbb P^n)$ for $m\ne n$, and
\begin{equation}
\label{eq:classicalbetti}
  \beta_n(Y)\;=\;\frac{(d-1)^{\,n+2}+(-1)^{n}(d-1)}{d}+\frac{1+(-1)^n}{2}
\end{equation}
\cite[Ch.~5, (3.3)]{Dimca}, \cite[Exp.~XI]{SGA7}; for Hermitian varieties see
also \cite{Shimada2001}.

<!-- fact-6: body_md -->
\label{fact:excess}
Let $Y$ be smooth, $P'\subset Y$ regularly embedded,
meeting $P$ cleanly along $Z$, with $Z\subset P$ regularly embedded.
If $\dim P+\dim P'=\dim Y$, then
\begin{equation}
\label{eq:excessformula}
  P\cdot P'\;=\;\deg\bigl(\mathrm{c}_{e}(E)\mathbin{\smallfrown}[Z]\bigr),
  \qquad e\;=\;\mathrm{codim}(P',Y)-\mathrm{codim}(Z,P)\;=\;\mathrm{rank}\,E .
\end{equation}
The case $P=P'$ is the self-intersection formula
\cite[Cor.~6.3]{Fulton}, where $Z=P$, $N_{Z/P}=0$ and $E=N_{P/Y}$.

<!-- definition-1: body_md -->
\label{def:valency}
Let $\Gamma$ be distance-regular of diameter $r$. For $0\le j\le r$
the \emph{$j$-th valency} of $\Gamma$ is
\[
  k_j\;:=\;\#\bigl\{w\in\Gamma\;:\;\partial(u,w)=j\bigr\},
\]
the number of vertices at distance $j$ from a fixed vertex $u$; it does
not depend on $u$. Thus $k_0=1$, $k_1$ is the valency, and
$A_jJ=JA_j=k_jJ$.
<!-- end definition-1 -->
<!-- lemma-1: body_md -->
\label{lem:valency}
$k_1=b_0$, and for $1\le j\le r$
\begin{equation}
\label{eq:valency}
  k_j\;=\;\frac{b_0b_1\cdots b_{j-1}}{c_1c_2\cdots c_j} .
\end{equation}
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
Count in two ways the pairs $(w,w')$ with $\partial(u,w)=j-1$,
$\partial(u,w')=j$ and $w\sim w'$. Counting from $w$ gives
$k_{j-1}b_{j-1}$, counting from $w'$ gives $k_jc_j$; hence
$k_j=k_{j-1}b_{j-1}/c_j$, and \eqref{eq:valency} follows by induction
from $k_0=1$.
<!-- end proof-2 -->

<!-- definition-2: body_md -->
\label{def:herm}
The \emph{Hermitian variety} $\mathcal H(2r-1,q^2)$ is
the hypersurface
\[
 \sum_{i=0}^{2r-1}x_i^{\,q+1}=0
  \ \subset\ \mathbb P^{2r-1},
\]
defined over $\mathbb{F}_{q^2}$.
<!-- end definition-2 -->
<!-- definition-3: body_md -->
\label{def:generator}
A \emph{generator} of $\mathcal H(2r-1,q^2)$ is an $\mathbb{F}_{q^{2}}$-rational
linear subspace $P\subset\mathbb P^{2r-1}$ whose corresponding subspace
$\widetilde P$ of $\mathbb{F}_{q^2}^{2r}$
is totally isotropic for the Hermitian form, of the maximal possible
dimension, i.e., $\dim_{\mathbb{F}_{q^2}}\widetilde P=r$, so that
$\dim P=r-1$. Note $P\subset\mathcal H(2r-1,q^2)$
\textup{(}Lemma~\textup{\ref{lem:contained}}\textup{)}. Write $\mathcal G$ for the set of
generators.
<!-- end definition-3 -->
<!-- proposition-2: body_md -->
\label{prop:generators}
$\mathrm{PGU}_{2r}(q)$ acts transitively on $\mathcal G$, and
\[
  |\mathcal G|\;=\;N_r\;=\;\prod_{i=1}^{r}\bigl(q^{2i-1}+1\bigr).
\]
The graph on $\mathcal G$ with $P\sim P'$ iff $\dim(P\cap P')=r-2$ is
the dual polar graph of type ${}^{2}A_{2r-1}(q)$: it is
distance-regular of diameter $r$, its distance-$j$ relation being
\[
  R_j\;=\;\bigl\{(P,P')\;:\;\dim(P\cap P')=r-1-j\bigr\},
  \qquad 0\le j\le r .
\]
<!-- end proposition-2 -->
<!-- proof-3: body_md -->
Transitivity is Fact~\ref{fact:witt}; the count is the classical one
\cite[Thm.~2.19]{HT}; distance-regularity is Fact~\ref{fact:dualpolar}.
<!-- end proof-3 -->
<!-- lemma-2: body_md -->
\label{lem:array}
For $0\le i\le r$,
\[
  c_i\;=\;\frac{q^{2i}-1}{q^{2}-1},\qquad
  b_i\;=\;q^{\,2i+1}\,\frac{q^{2(r-i)}-1}{q^{2}-1},
\]
with $c_0=0$ and $b_r=0$. In particular
$b_0=q\,\dfrac{q^{2r}-1}{q^{2}-1}$ is the valency. Moreover
\begin{equation}
\label{eq:aisc}
  a_i\;=\;(q-1)\,c_i \qquad (0\le i\le r).
\end{equation}
<!-- end lemma-2 -->
<!-- proof-4: body_md -->
The values of $c_i,b_i$ are \cite[Thm.~9.4.3]{BCN}. For \eqref{eq:aisc},
substitute them into $a_i=b_0-b_i-c_i$.
<!-- end proof-4 -->
<!-- lemma-3: body_md -->
\label{lem:kj}
Fix $P\in\mathcal G$ and $0\le j\le r$. The $j$-th valency of the dual
polar graph --- that is \textup{(}Definition~\ref{def:valency}\textup{)},
the number of $P'\in\mathcal G$ with $\partial(P,P')=j$, equivalently
with $\dim(P\cap P')=r-1-j$ --- equals
\[
  k_j\;=\;q^{\,j^{2}}\binom{r}{j}_{\!q^{2}} .
\]
For instance $k_0=1$ and $k_1=b_0=q\,\frac{q^{2r}-1}{q^{2}-1}$.
<!-- end lemma-3 -->
<!-- proof-5: body_md -->
Substitute Lemma~\ref{lem:array} into \eqref{eq:valency}:
\[
  k_j=\frac{q^{\sum_{i=0}^{j-1}(2i+1)}
  \prod_{i=0}^{j-1}\frac{q^{2(r-i)}-1}{q^{2}-1}}
  {\prod_{i=1}^{j}\frac{q^{2i}-1}{q^{2}-1}}
  =q^{\,j^{2}}\prod_{i=1}^{j}\frac{q^{2(r-i+1)}-1}{q^{2i}-1}
  =q^{\,j^{2}}\binom{r}{j}_{\!q^{2}} . \qedhere
\]
<!-- end proof-5 -->
<!-- lemma-4: body_md -->
\label{lem:sumkj}
For every $r\ge1$,
\[
  \sum_{j=0}^{r}k_j=N_r,
  \qquad
  \sum_{j=0}^{r}k_j\,q^{-2j}=\frac{q+1}{q}\,N_{r-1},
  \qquad
  \sum_{j=0}^{r}k_j\,(-q)^{\,r-j}=0 .
\]
<!-- end lemma-4 -->
<!-- proof-6: body_md -->
All three are instances of the $q$-binomial theorem
(Fact~\ref{fact:qbinom}) with $Q=q^{2}$, so that
$Q^{\binom j2}=q^{\,j^{2}-j}$ and
$\binom rj_{q^2}Q^{\binom j2}x^{j}=k_j\,q^{-j}x^{j}$.

Taking $x=q$ gives $\sum_{j=0}^{r}k_j$ on the right and
$\prod_{i=0}^{r-1}(1+q^{2i+1})=\prod_{i=1}^{r}(1+q^{2i-1})=N_r$ on the
left.

Taking $x=q^{-1}$ gives $\sum_{j=0}^{r}k_j q^{-2j}$ on the right and
$\prod_{i=0}^{r-1}(1+q^{2i-1})=(1+q^{-1})\prod_{i=1}^{r-1}(1+q^{2i-1})
=\frac{q+1}{q}N_{r-1}$ on the left.

Taking $x=-1$ gives $\sum_{j=0}^{r}(-1)^{j}q^{\,j^{2}-j}\binom rj_{q^{2}}$ on
the right and $\prod_{i=0}^{r-1}(1-q^{2i})=0$ on the left, the factor
$i=0$ vanishing. Multiplying by $(-q)^{r}$ and using
$k_j=q^{\,j^{2}}\binom rj_{q^{2}}$ turns the right side into
$\sum_{j=0}^{r}k_j(-q)^{\,r-j}$.
<!-- end proof-6 -->

\label{lem:f}

\label{prop:gram}
With the notation of \S\ref{subsec:generators}\textup{:}
\begin{enumerate}
\item[\textup{(i)}] $M=\sum_{j=0}^{r}f(r-1-j)\,A_j$.
\item[\textup{(ii)}] $A_j=p_j(A)$ for a polynomial $p_j\in\mathbb{Q}[x]$ of
  degree exactly $j$; consequently
  \[
    M\;=\;F(A),\qquad F:=\sum_{j=0}^{r}f(r-1-j)\,p_j\in\mathbb{Q}[x],
    \quad\deg F\le r .
  \]
\item[\textup{(iii)}] $A$ is a symmetric integer matrix with exactly $r+1$
  distinct real eigenvalues $\theta_0>\theta_1>\cdots>\theta_r$, and
  $A\mathbf 1=b_0\mathbf 1$ for the
vector $\mathbf 1=\sum_P e_P$ all of whose coordinates equal $1$. Put
  $W_0:=\mathbb{Q}_\ell\cdot\mathbf 1$ and $m_0:=1$.
\item[\textup{(iv)}] If $\theta\in\mathbb{Q}$ is an eigenvalue of $A$, every matrix of the form $\sum_j \lambda_jA_j$ acts on
  $\ker(A-\theta)\subset\mathbb{Q}_\ell^{\mathcal G}$ by a scalar; in particular $M$ acts on $W_0$ by
  $\mu_0:=F(b_0)$.
\end{enumerate}

\textup{(i)} By Lemma~\ref{lem:f}, $P\cdot P'$ depends only on
$\dim(P\cap P')$; the entry of $M$ at $(P,P')\in R_j$ is the constant
$f(r-1-j)$.

\textup{(ii)} Induction on $j$: $A_0=\mathrm{Id}=p_0(A)$ and
$A_1=A=p_1(A)$; solving \eqref{eq:3termA} for $A_{j+1}$,
\[
  A_{j+1}=\frac{1}{c_{j+1}}\bigl(A\,A_j-a_jA_j-b_{j-1}A_{j-1}\bigr)
  =\frac{1}{c_{j+1}}\bigl(x\,p_j-a_jp_j-b_{j-1}p_{j-1}\bigr)(A),
\]
legitimate since $c_{j+1}\ne0$ for $j<r$ (Lemma~\ref{lem:array}); the
degree rises by exactly one at each step, the leading coefficient
being $1/(c_1\cdots c_j)\ne0$.

\textup{(iii)} $A$ is real symmetric, hence orthogonally
diagonalisable. The $A_j$ are nonzero with pairwise disjoint supports,
hence linearly independent; by (ii) they span the same space as
$\mathrm{Id},A,\dots,A^{r}$, so those $r+1$ powers are linearly
independent and the minimal polynomial of $A$ has degree exactly
$r+1$. A symmetric matrix has minimal polynomial

exactly $r+1$ of them. The graph is connected of diameter $r$ and
regular of valency $b_0$, so $A\mathbf 1=b_0\mathbf 1$.

\textup{(iv)} Any $\sum_j\lambda_jA_j=\bigl(\sum_j\lambda_jp_j\bigr)(A)$ is a

obtained by substituting $\theta$.

\label{thm:bottomrow}
Put
\[
  \theta_\star\;:=\;-c_r\;=\;-\,\frac{q^{2r}-1}{q^{2}-1}.
\]
Then, for                    
every prime power $q$,
\begin{equation}
\label{eq:bottomrow}
  A\,M'\;=\;\theta_\star\,M' .
\end{equation}
Consequently $\theta_\star$ is an eigenvalue of $A$; writing
$W_\star:=\ker(A-\theta_\star)$, we have
$\mathrm{im}\,M'\subseteq W_\star$, and $W_\star\ne W_0$.
<!-- end theorem-4 -->
<!-- proof-14: body_md -->
Write $g_j:=(-q)^{\,r-j}$, so $M'=\sum_{j=0}^{r}g_jA_j$. Applying
\eqref{eq:3termA} and collecting the coefficient of $A_m$,
\[
  A\,M'\;=\;\sum_{m=0}^{r}
  \bigl(g_{m+1}b_m+g_ma_m+g_{m-1}c_m\bigr)A_m ,
\]
and $g_{m+1}=-g_m/q$, $g_{m-1}=-q\,g_m$ turn the bracket into
$g_m\bigl[-\tfrac1q b_m+a_m-q\,c_m\bigr]$. So \eqref{eq:bottomrow}
amounts to the assertion that
\begin{equation}
\label{eq:bracket}
  -\tfrac1q b_m+a_m-q\,c_m\;=\;\theta_\star
  \qquad (0\le m\le r).
\end{equation}
The point is that the left-hand side does not depend on $m$. But
\eqref{eq:bracket} is immediate from the explicit formulas of
Lemma~\ref{lem:array}: by \eqref{eq:aisc} the left-hand side equals
$-\tfrac1q b_m-c_m$, in which the terms $q^{2m}$ cancel, leaving
$-c_r=\theta_\star$. The conventions $b_{-1}=c_{r+1}=0$,
$c_0=0$, $b_r=0$ make the computation valid at $m=0$ and $m=r$ as
well.

of $M'$ as a $\theta_\star$-eigenvector of $A$; every vector in
$\mathrm{im}\,M'$ is a combination of columns, whence
$\mathrm{im}\,M'\subseteq W_\star$. And $W_\star\ne W_0$ because
$\theta_\star<0<b_0$ and $A\mathbf 1=b_0\mathbf 1$.

\textup{(i)} Write $\Phi$ for the matrix of $\phi$ and $G$ for the

of dimension $\ge3$ in $\mathbb P^{n+1}$ over an algebraically closed
field has $\mathrm{Pic}\cong\mathbb{Z}$, generated by the hyperplane class
$L$. Hence every ample class on $Y$ is $eL$ for some integer $e\ge1$.
Corollary~\ref{cor:std} gives $\mathrm{Hdg}(Y,L)$ in the middle degree
and Proposition~\ref{prop:belowmiddle}(i) gives it in every degree below
the middle, so $\mathrm{Hdg}(Y,L)$ holds; Lemma~\ref{lem:scaling} then
gives $\mathrm{Hdg}(Y,eL)$.

If $m=1$ then $Y$ is a nonsingular projective surface, and for a
surface the standard conjecture of Hodge type with respect to any
ample class is the Hodge index theorem
\textup{\cite[Thm.~V.1.9]{Hartshorne}}, which holds unconditionally.

<!-- theorem-9: body_md -->
\label{cor:quotient}
Let $m\ge1$, let $q$ be a prime power, and let $Z$ be a nonsingular
projective variety of dimension $2m$ over $\overline{\mathbb{F}}_p$ admitting
a finite surjective morphism
\[
  \pi\;:\;\mathcal H(2m+1,q^{2})\;\longrightarrow\;Z .
\]
Then the standard conjecture of Hodge type holds for $Z$, with respect
to every ample class on $Z$. In particular this applies whenever
$Z=\mathcal H(2m+1,q^{2})/G$ is a nonsingular quotient by a finite group
$G$ of automorphisms.

Let $L_Z$ be ample on $Z$. Since $\pi$ is finite, $\pi^{*}L_Z$ is
ample on $\mathcal H(2m+1,q^{2})$, so Lemma~\ref{lem:allpol} supplies the
hypothesis of Lemma~\ref{lem:descent}, which gives

<!-- lemma-16: body_md -->
\label{lem:cover}
Let $N\ge2$ with $p\nmid N$, let $q=p^{\nu}$ be such that $N\mid q+1$,
and set $e=(q+1)/N$. Let $\mu_e^{2m+2}/\mu_e$ act on
$\mathbb P^{2m+1}$ by $[x_i]\mapsto[\zeta_ix_i]$ with
$\zeta_i^{\,e}=1$. Then this action preserves $\mathcal H(2m+1,q^{2})$, and
the quotient map is
\[
  \pi\;:\;\mathcal H(2m+1,q^{2})\;\longrightarrow\;X_N^{2m},
  \qquad
  [x_0:\cdots:x_{2m+1}]\;\longmapsto\;[x_0^{\,e}:\cdots:x_{2m+1}^{\,e}] ,
\]
a finite surjective morphism of nonsingular projective varieties of
dimension $2m$, with $\pi^{*}\mathcal O_{X_N}(1)=\mathcal
O_{\mathcal H}(e)$ and $\deg(\pi)=e^{\,2m+1}$. In particular $X_N^{2m}$ is
a nonsingular finite quotient of $\mathcal H(2m+1,q^{2})$.
<!-- end lemma-16 -->
<!-- proof-31: body_md -->
Since $e\mid q+1$, we have $\zeta_i^{\,q+1}=1$, so the action
preserves $\sum_ix_i^{\,q+1}$ and hence $\mathcal H(2m+1,q^{2})$; the
invariants of $\mu_e^{2m+2}/\mu_e$ on $\mathbb P^{2m+1}$ are generated
by the $x_i^{\,e}$, so the quotient map is $[x_i]\mapsto[x_i^{\,e}]$.

This assignment is in any case a morphism
$\mathbb P^{2m+1}\to\mathbb P^{2m+1}$, since the $x_i^{\,e}$ have no
common zero, and it is finite. If $\sum_ix_i^{\,q+1}=0$ then
$\sum_i(x_i^{\,e})^{N}=\sum_ix_i^{\,eN}=\sum_ix_i^{\,q+1}=0$, so it
maps $\mathcal H(2m+1,q^{2})$ into $X_N^{2m}$. Both are nonsingular of
dimension $2m$ (here $p\nmid N$ and $p\nmid q+1$), and $X_N^{2m}$ is
irreducible, so the image, being closed of dimension $2m$, is all of
$X_N^{2m}$. The map is given by forms of degree $e$, whence
$\pi^{*}\mathcal O(1)=\mathcal O(e)$, and the projection formula gives
$\deg(\pi)\cdot N=\deg(\pi)\,\deg(L_Z^{2m})=\deg\bigl((\pi^{*}L_Z)^{2m}\bigr)
=e^{2m}\deg(L_Y^{2m})=e^{2m}(q+1)$, so $\deg(\pi)=e^{2m}(q+1)/N=e^{2m+1}$.

<!-- proposition-7: body_md -->
\label{prop:pdescent}
\begin{enumerate}
\item[\textup{(i)}] $q^m\deg(\pi)\cdot\mathrm N^m(Z)\subset\Lambda_Z$, and $\Lambda_Z^\vee/\Lambda_Z$ is
  annihilated by $q^m\deg(\pi)$. In particular, if $p\nmid\deg\pi$, then the
  $p$-part of $\Lambda_Z^\vee/\Lambda_Z$ is annihilated by $q^m$, and
  $q^m\,\mathrm N^m(Z)\subset\Lambda_Z\otimes\mathbb{Z}_{(p)}$.
\item[\textup{(ii)}] For every prime $\ell\ne p$ with $\ell\nmid\deg\pi$, the lattice
  $\Lambda_Z\otimes\mathbb{Z}_\ell=\mathrm N^m(Z)\otimes\mathbb{Z}_\ell$ is unimodular.
\end{enumerate}
<!-- end proposition-7 -->
<!-- proof-33: body_md -->
By \cite[Thm.~1.1]{Shimada2001}, $\Lambda^\vee/\Lambda$ is a $p$-group. So
Corollary~\ref{cor:gentrace}(ii),(iii) give $q^m\Lambda^\vee\subset\Lambda$ and
$q^m\mathrm N^m(X)\subset\Lambda$, and $\Lambda\otimes\mathbb{Z}_\ell$ is unimodular for
$\ell\ne p$.

(i) Let $y\in\mathrm N^m(Z)$. Then $q^m\pi^*y\in\Lambda$, and applying $\pi_*$ gives
$q^m\deg(\pi)\,y\in\pi_*\Lambda=\Lambda_Z$. Let $x\in\Lambda_Z^\vee$. For
$\lambda\in\Lambda$ we have $\langle\pi^*x,\lambda\rangle_X=\langle x,\pi_*\lambda\rangle_Z\in\mathbb{Z}$, so
$\pi^*x\in\Lambda^\vee$. Hence $q^m\pi^*x\in\Lambda$ and $q^m\deg(\pi)\,x\in\Lambda_Z$.

(ii) Over $\mathbb{Z}_\ell$ the same argument, with $\Lambda^\vee\otimes\mathbb{Z}_\ell=\Lambda\otimes\mathbb{Z}_\ell$,
gives $\deg(\pi)(\Lambda_Z^\vee\otimes\mathbb{Z}_\ell)\subset\Lambda_Z\otimes\mathbb{Z}_\ell$. As
$\deg\pi$ is an $\ell$-adic unit, $\Lambda_Z\otimes\mathbb{Z}_\ell$ is unimodular. Since
$\Lambda_Z\subset\mathrm N^m(Z)\subset\mathrm N^m(Z)^\vee\subset\Lambda_Z^\vee$, all four
coincide $\ell$-adically.

<!-- corollary-6: body_md -->
\label{cor:pdescent}
Let $T\subset\mathrm{PGU}_{2m+2}(q)$ be a finite subgroup such that $Z=X/T$ is
nonsingular, and let $\pi\colon X\to Z$ be the quotient map, so $\deg\pi=|T|$.
\begin{enumerate}
\item[\textup{(i)}] $\langle\pi_*P,\pi_*P'\rangle_Z=\sum_{t\in T}M_{P,tP'}$ for $P,P'\in\mathcal G$.
\item[\textup{(ii)}] If $p\nmid|T|$, then $\pi^*$ induces isomorphisms
  $\Lambda_Z\otimes\mathbb{Z}_p\cong(\Lambda\otimes\mathbb{Z}_p)^T$ and
  $\mathrm N^m(Z)\otimes\mathbb{Z}_p\cong(\mathrm N^m(X)\otimes\mathbb{Z}_p)^T$, which multiply the
  intersection form by the unit $|T|$.
\item[\textup{(iii)}] Let $L_Z$ be an ample class on $Z$ with $\pi^*L_Z\in\mathbb{Q}_{>0}L$ (this is
  automatic if $m\ge2$, and holds for the Fermat quotients below). Under $\pi^*$, the primitive part of $\mathcal A^m(Z)$ with respect to $L_Z$ is identified with
  the $T$-invariants of $\mathcal P^{2m}(X)_\mathbb{Q}\congU_{r}$. On it,
  $(-1)^m\langle\,,\rangle_Z$ is $|T|^{-1}q^mN_m$ times $\langle\,,\rangle_{\mathrm{std}}$.
\end{enumerate}
For $Z=X_N^{2m}$ with $N\mid q+1$, take $T=\mu_e^{2m+2}/\mu_e\subset S$ with
$e=(q+1)/N$ (Lemma~\ref{lem:cover}); here $|T|=e^{2m+1}$ is prime to $p$.
<!-- end corollary-6 -->
<!-- proof-34: body_md -->
Now $\pi^*\pi_*=\sum_{t\in T}t_*$, and $T$ permutes $\mathcal G$ isometrically.

(i) $\langle\pi_*P,\pi_*P'\rangle_Z=|T|^{-1}\sum_{t,t'}\langle tP,t'P'\rangle_X
=\sum_t\langle P,tP'\rangle_X$.

(ii) With $e_T=|T|^{-1}\sum_tt\in\mathbb{Z}_p[T]$ we get
$\pi^*(\Lambda_Z\otimes\mathbb{Z}_p)=\pi^*\pi_*(\Lambda\otimes\mathbb{Z}_p)=e_T(\Lambda\otimes\mathbb{Z}_p)
=(\Lambda\otimes\mathbb{Z}_p)^T$. Moreover $\pi^*$ is injective because $\pi_*\pi^*=|T|$ is a
unit. The same argument applies to $\mathrm N^m$.

(iii) The map $\pi^*$ preserves primitivity (Lemma~\ref{lem:descent}), and the
isomorphism $\psi$ of Theorem~\ref{thm:main-intro} is equivariant for the
automorphisms of $X$ preserving $\mathcal G$. Now use the projection formula.

<!-- remark-3: body_md -->
\label{rem:autPGU}
The group of linear automorphisms of $X$, i.e.\ of the projective
transformations of $\mathbb P^{2m+1}$ preserving $X$, is $\mathrm{PGU}_{2m+2}(q)$

automorphism of $X$ is linear \cite{MM1964}, so $\mathrm{Aut}(X)=\mathrm{PGU}_{2m+2}(q)$.
