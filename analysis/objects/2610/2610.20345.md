<!-- assumption-1: body_md -->
\label{ass:smooth}
The function $f:\mathbb R^{d_x}\times\mathbb R^{d_y}\to\mathbb R$ is twice continuously differentiable. For all ${\bf w},{\bf v}\in\mathbb R^{d_x+d_y}$, we assume that
\begin{equation}\label{eq:smooth}
\lVert \nabla^2f({\bf w})\rVert\le L~~~\text{and}~~~
\lVert \nabla^2f({\bf w})-\nabla^2f({\bf v})\rVert\le\rho\lVert {\bf w}-{\bf v}\rVert,
\end{equation}
where $L,\rho>0$.
<!-- end assumption-1 -->
<!-- assumption-2: body_md -->
\label{ass:concave}
For every ${\bf x}$, the function $f({\bf x},\cdot)$ is $\mu$-strongly concave, where $0<\mu\le L$. We write $\kappa=L/\mu$.
<!-- end assumption-2 -->
<!-- assumption-3: body_md -->
\label{ass:gap}
The primal function $\Phi(\cdot)$ is bounded below, and the initial point ${\bf z}_0$ satisfies $\Phi({\bf z}_0)-\inf_{{\bf z}}\Phi({\bf z})\le\Delta$ for some $\Delta>0$.
<!-- end assumption-3 -->
<!-- definition-1: body_md -->
\label{def:sosp}
For $\varepsilon,\eta>0$, a point $\hat{\bf x}$ is an $\varepsilon$-first-order stationary point of $\Phi(\cdot)$ if $\lVert \nabla\Phi(\hat{\bf x})\rVert\le\varepsilon$. 
It is an $(\varepsilon,\eta)$-second-order stationary point if, in addition,
$
\nabla^2\Phi(\hat{\bf x})\succeq-\eta{\bf I}_{d_x}.
$

<!-- lemma-1: body_md -->
\label{lem:cubic}
Let $g:\mathbb R^d\to\mathbb R$ be twice continuously differentiable with a $\rho_g$-Lipschitz Hessian, where $\rho_g>0$, and suppose $g({\bf z}_0)-\inf_{{\bf z}}g({\bf z})\le\Delta$. If the derivative routine satisfies~\eqref{eq:model-contract} at every iterate, then Algorithm~\ref{alg:inexact-crn} returns ${\bf z}_{K+1}$ after $K+1$ iterations, where
\begin{equation}\label{eq:cubic-sum}
K\le4\cdot8^{\frac{3}{2}}\Delta\sqrt{\rho_g}\,\varepsilon^{-\frac{3}{2}}
\quad\text{and}\quad
\sum_{k=0}^{K-1}\lVert {\bf s}_k\rVert^3\le\frac{4\Delta}{\rho_g}.
\end{equation}
Its output satisfies
\begin{equation}\label{eq:cubic-stop}
\lVert \nabla g({\bf z}_{K+1})\rVert<\frac{\varepsilon}{2}
\quad\text{and}\quad
\nabla^2g({\bf z}_{K+1})\succeq-\frac{3}{4}\sqrt{\rho_g\varepsilon}\,{\bf I}_d.
\end{equation}

<!-- proposition-1: body_md -->
\label{prop:primal}
Under Assumptions~\ref{ass:smooth}--\ref{ass:concave}, the regularized objective in~\eqref{eq:regularized-objective} has the following properties.
\begin{enumerate}[label=(\roman*),leftmargin=2em,itemsep=3pt]
\item For every ${\bf z}\in\mathbb R^{d_x}$, $f_{{\bf z}}(\cdot)$ is $L$-strongly convex in ${\bf u}$ and $\mu$-strongly concave in ${\bf y}$, and has the unique saddle point~\eqref{eq:saddle-map}. Its Hessian is nonsingular, with
\begin{equation}\label{eq:regularized-inverse-bound}
\lVert \bigl(\nabla^2f_{{\bf z}}({\bf w})\bigr)^{-1}\rVert\le \mu^{-1}.
\end{equation}
\item For every ${\bf z}\in\mathbb R^{d_x}$ and every ${\bf w},{\bf v}\in\mathbb R^{d_x+d_y}$, we have
\begin{equation}\label{eq:regularized-hessian-bounds}
\begin{aligned}
\lVert \nabla^2f_{{\bf z}}({\bf w})\rVert\le 3L~~~\text{and}~~~\lVert \nabla^2f_{{\bf z}}({\bf w})-\nabla^2f_{{\bf z}}({\bf v})\rVert\le\rho\lVert {\bf w}-{\bf v}\rVert.
\end{aligned}
\end{equation}
\end{enumerate}

<!-- proposition-2: body_md -->
\label{prop:envelope-basic}
Under Assumptions~\ref{ass:smooth}--\ref{ass:concave}, the envelope $\Phi_{2L}(\cdot)$ is twice continuously differentiable, with
\begin{equation}\label{eq:envelope-basic}
\nabla\Phi_{2L}({\bf z})=2L\bigl({\bf z}-{\bf u}^*({\bf z})\bigr)=\nabla\Phi({\bf u}^*({\bf z}))
\end{equation}
and
\begin{equation}\label{eq:envelope-hessian}
\nabla^2\Phi_{2L}({\bf z})=2L{\bf I}_{d_x}-4L^2{\bf M}({\bf w}^*({\bf z}))^{-1},
\end{equation}
where ${\bf M}({\bf w})$ is defined by
\begin{equation}\label{eq:schur-complement}
{\bf M}({\bf w}):=\nabla^2_{uu}f({\bf w})+2L{\bf I}_{d_x}
-\nabla^2_{uy}f({\bf w})\bigl(\nabla^2_{yy}f({\bf w})\bigr)^{-1}\nabla^2_{yu}f({\bf w}).
\end{equation}
For every ${\bf w}\in\mathbb R^{d_x+d_y}$, we have ${\bf M}({\bf w})\succeq L{\bf I}_{d_x}$. The envelope has the following properties.
\begin{enumerate}[label=(\roman*),leftmargin=2em,itemsep=3pt]
\item Its infimum is $\inf_{{\bf z}}\Phi_{2L}({\bf z})=\inf_{{\bf z}}\Phi({\bf z})$. 
Under Assumption~\ref{ass:gap}, we have $\Phi_{2L}({\bf z}_0)-\inf_{{\bf z}}\Phi_{2L}({\bf z})\le\Delta$.
\item If ${\bf z}$ is an $(\varepsilon,\eta)$-second-order stationary point of $\Phi_{2L}(\cdot)$, then ${\bf u}^*({\bf z})$ is an $(\varepsilon,2L\eta/(2L+\eta))$-second-order stationary point of $\Phi(\cdot)$, and hence also an $(\varepsilon,\eta)$-second-order stationary point.
\end{enumerate}

<!-- lemma-2: body_md -->
\label{lem:saddle-map}
Under Assumptions~\ref{ass:smooth}--\ref{ass:concave}, the saddle-point map ${\bf w}^*(\cdot)$ is continuously differentiable and for every ${\bf z},{\bf z}'\in\mathbb R^{d_x}$, it holds that
\begin{equation}\label{eq:global-displacement}
\lVert {\bf w}^*({\bf z})-{\bf w}^*({\bf z}')\rVert
\le \sqrt{4+\kappa}\lVert {\bf z}-{\bf z}'\rVert.
\end{equation}
Moreover, its distance from any point ${\bf w}\in\mathbb R^{d_x+d_y}$ is bounded by the gradient norm:
\begin{equation}\label{eq:residual-to-distance}
\lVert {\bf w}-{\bf w}^*({\bf z})\rVert\le \mu^{-1}\lVert \nabla f_{{\bf z}}({\bf w})\rVert.
\end{equation}

<!-- lemma-3: body_md -->
\label{lem:envelope}
Under Assumptions~\ref{ass:smooth}--\ref{ass:concave}, the Hessian of $\Phi_{2L}(\cdot)$ is $\widetilde{\rho}$-Lipschitz continuous, where $\widetilde{\rho}:=\rho(4+\kappa)^{3/2}$. Specifically, for every ${\bf z},{\bf z}'\in\mathbb R^{d_x}$,
\begin{equation}\label{eq:H}
\lVert \nabla^2\Phi_{2L}({\bf z})-\nabla^2\Phi_{2L}({\bf z}')\rVert
\le\widetilde{\rho}\lVert {\bf z}-{\bf z}'\rVert.
\end{equation}
<!-- end lemma-3 -->
<!-- proof-1: body_md -->
We write ${\bf P}:=[2L{\bf I}_{d_x},{\bf 0}_{d_x\times d_y}]^\top$. 
Block elimination gives
\[
4L^2{\bf M}({\bf w})^{-1}={\bf P}^\top\bigl(\nabla^2f_{{\bf z}}({\bf w})\bigr)^{-1}{\bf P}.
\]
Thus, for any points ${\bf w},{\bf v}$, the inverse-difference identity gives
\begin{equation}\label{eq:envelope-hessian-difference}
\begin{aligned}
&4L^2\bigl({\bf M}({\bf v})^{-1}-{\bf M}({\bf w})^{-1}\bigr)\\
&\quad={\bf P}^\top\bigl(\nabla^2f_{{\bf z}}({\bf w})\bigr)^{-1}
\bigl(\nabla^2f_{{\bf z}}({\bf w})-\nabla^2f_{{\bf z}}({\bf v})\bigr)
\bigl(\nabla^2f_{{\bf z}}({\bf v})\bigr)^{-1}{\bf P}.
\end{aligned}
\end{equation}
Using~\eqref{eq:envelope-hessian-difference} and symmetry of the Hessians, we have
\begin{equation}\label{eq:schur-inverse-lipschitz}
\begin{aligned}
&4L^2\lVert {\bf M}({\bf v})^{-1}-{\bf M}({\bf w})^{-1}\rVert\\
&\quad\le
\lVert \bigl(\nabla^2f_{{\bf z}}({\bf w})\bigr)^{-1}{\bf P}\rVert
\lVert \nabla^2f_{{\bf z}}({\bf w})-\nabla^2f_{{\bf z}}({\bf v})\rVert
\lVert \bigl(\nabla^2f_{{\bf z}}({\bf v})\bigr)^{-1}{\bf P}\rVert\\
&\quad\stackrel{\eqref{eq:T-bound}}{\le}
(4+\kappa)\lVert \nabla^2f_{{\bf z}}({\bf w})-\nabla^2f_{{\bf z}}({\bf v})\rVert\\
&\quad\stackrel{\eqref{eq:regularized-hessian-bounds}}{\le}
\rho (4+\kappa)\lVert {\bf w}-{\bf v}\rVert.
\end{aligned}
\end{equation}
Apply \eqref{eq:schur-inverse-lipschitz} with ${\bf w}={\bf w}^*({\bf z})$ and ${\bf v}={\bf w}^*({\bf z}')$ yields
\[
\begin{aligned}
&\lVert \nabla^2\Phi_{2L}({\bf z})-\nabla^2\Phi_{2L}({\bf z}')\rVert\overset{\eqref{eq:envelope-hessian}}{\leq} 4L^2\lVert {\bf M}({\bf v})^{-1}-{\bf M}({\bf w})^{-1}\rVert
\\
&~~~\overset{\eqref{eq:schur-inverse-lipschitz}}{\le}\rho (4+\kappa)\lVert {\bf w}^*({\bf z})-{\bf w}^*({\bf z}')\rVert
\overset{\eqref{eq:global-displacement}}{\le}\rho (4+\kappa)^{\frac{3}{2}}\lVert {\bf z}-{\bf z}'\rVert.
\end{aligned}
\]
This proves~\eqref{eq:H}.

<!-- lemma-4: body_md -->
\label{lem:pce-dm}
Under Assumptions~\ref{ass:smooth}--\ref{ass:concave}, let $D_{\rm init}:=\lVert {\bf w}_{\rm init}-{\bf w}^*({\bf z})\rVert$. When the initial gradient norm exceeds $\bar\varepsilon$, Algorithm~\ref{alg:pce-dm} reaches~\eqref{eq:inner-target} within $T_{\rm PCE}(\bar\varepsilon)$ iterations and at most $2T_{\rm PCE}(\bar\varepsilon)+1$ gradient evaluations, where

T_{\rm PCE}(\bar\varepsilon):=\Bigl\lceil720\sqrt\kappa
\log\frac{2592L^2\kappa^{\frac{3}{2}}D_{\rm init}^2}{\bar\varepsilon^2}\Bigr\rceil.
\end{equation}

\label{lem:warm-start}

<!-- lemma-7: body_md -->
\label{lem:tracking}
Under Assumptions~\ref{ass:smooth}--\ref{ass:concave}, let $R,\bar\varepsilon$ satisfy~\eqref{eq:tracking-conditions}. If $\lVert \nabla f_{\bar{{\bf z}}}(\bar{{\bf w}})\rVert\le\bar\varepsilon$, Algorithm~\ref{alg:newton-track} returns an approximate solution $\hat{{\bf w}}={\bf w}_m$ satisfying $\lVert \nabla f_{{\bf z}}(\hat{{\bf w}})\rVert\le\bar\varepsilon$. With $m=\lceil\lVert {\bf z}-\bar{{\bf z}}\rVert/R\rceil$, it uses at most $m T_{\rm N}(\bar\varepsilon)$ Hessian evaluations and $m(T_{\rm N}(\bar\varepsilon)+1)$ gradient evaluations, where $T_{\rm N}(\bar\varepsilon)$ is given by~\eqref{eq:local-newton-horizon}.
<!-- end lemma-7 -->
<!-- proof-3: body_md -->
We prove by induction that the iterates in Algorithm~\ref{alg:newton-track} satisfy
\[
\lVert \nabla f_{{\boldsymbol\xi}_j}({\bf w}_j)\rVert\le\bar\varepsilon~~~\text{for}~~~ j=0,\ldots,m.
\]
The input condition gives the claim at $j=0$, since ${\bf w}_0=\bar{{\bf w}}$ and ${\boldsymbol\xi}_0=\bar{{\bf z}}$. Suppose the claim holds at $j-1$, where $1\le j\le m$. Since $\lVert {\boldsymbol\xi}_j-{\boldsymbol\xi}_{j-1}\rVert\le R$, Lemma~\ref{lem:warm-start} gives
\[
\lVert {\bf w}_{j-1}-{\bf w}^*({\boldsymbol\xi}_j)\rVert\le\frac{\mu}{12\rho}.
\]
Lemma~\ref{lem:local-newton} and the stopping test in Algorithm~\ref{alg:newton} therefore ensure that the returned iterate ${\bf w}_j$ satisfies
\[
\lVert \nabla f_{{\boldsymbol\xi}_j}({\bf w}_j)\rVert\le\bar\varepsilon.
\]
This completes the induction. Since $\hat{{\bf w}}={\bf w}_m$ and ${\boldsymbol\xi}_m={\bf z}$, we obtain
\[
\lVert \nabla f_{{\bf z}}(\hat{{\bf w}})\rVert
=\lVert \nabla f_{{\boldsymbol\xi}_m}({\bf w}_m)\rVert\le\bar\varepsilon.
\]
Each Newton call uses at most $T_{\rm N}(\bar\varepsilon)$ Hessian evaluations and $T_{\rm N}(\bar\varepsilon)+1$ gradient evaluations. Summing over the $m$ calls proves the stated oracle bounds.

<!-- lemma-8: body_md -->
\label{lem:oracle}
Under Assumptions~\ref{ass:smooth}--\ref{ass:concave}, let $\varepsilon>0$ and set
\begin{equation}\label{eq:parameters}
\widetilde\varepsilon:=\min\Bigl\{\frac{\mu\varepsilon}{256L},\
\frac{\mu\sqrt{\widetilde{\rho}\varepsilon}}{128\rho(4+\kappa)},\
\frac{\mu^2}{48\rho},\
\frac{\mu\sqrt{\widetilde{\rho}\varepsilon}}{4\rho(1+\kappa)^3}\Bigr\}.
\end{equation}
For any point ${\bf w}=({\bf u},{\bf y})$ satisfying $\lVert \nabla f_{{\bf z}}({\bf w})\rVert\le\widetilde\varepsilon$, we define
\begin{equation}\label{eq:approx-derivatives}
\widehat{{\bf g}}:=2L({\bf z}-{\bf u})
\quad\text{and}\quad
\widehat{{\bf H}}:=2L{\bf I}_{d_x}-4L^2{\bf M}({\bf w})^{-1}.
\end{equation}
Then, it holds that the matrix $\widehat{{\bf H}}$ is symmetric and the derivative errors satisfy
\begin{equation}\label{eq:model-errors}
\lVert \widehat{{\bf g}}-\nabla\Phi_{2L}({\bf z})\rVert\le\frac{\varepsilon}{128}
\quad\text{and}\quad
\lVert \widehat{{\bf H}}-\nabla^2\Phi_{2L}({\bf z})\rVert\le\frac{\sqrt{\widetilde{\rho}\varepsilon}}{128}.
\end{equation}
                                                                                                            
Moreover, if ${\bf z}$ is an $(\varepsilon/2,3\sqrt{\widetilde{\rho}\varepsilon}/4)$-second-order stationary point of $\Phi_{2L}(\cdot)$, then ${\bf u}$ is an $(\varepsilon,\sqrt{\widetilde{\rho}\varepsilon})$-second-order stationary point of $\Phi(\cdot)$.

<!-- theorem-1: body_md -->
\label{thm:upper}

<!-- proof-4: body_md -->
Let $K$ be the index at which ECRN satisfies its stopping condition, $r_k=\lVert {\bf s}_k\rVert$, and $m_k=\lceil r_k/R\rceil$. Lemma~\ref{lem:pce-dm}, applied to the initialization with $\bar\varepsilon=\widetilde\varepsilon$, and the stopping test in Algorithm~\ref{alg:pce-dm} give
\[
\lVert \nabla f_{{\bf z}_0}({\bf w}_0)\rVert\le\widetilde\varepsilon.
\]
The parameter choices satisfy
\[
\widetilde\varepsilon\stackrel{\eqref{eq:parameters}}{\le}\frac{\mu^2}{48\rho}
\quad\text{and}\quad
R=\frac{\mu}{48\rho\sqrt\kappa}
\le\frac{\mu}{16\rho\sqrt{4+\kappa}}.
\]
Applying Lemma~\ref{lem:tracking} with $\bar\varepsilon=\widetilde\varepsilon$ and $(\bar{{\bf w}},\bar{{\bf z}},{\bf z})=({\bf w}_k,{\bf z}_k,{\bf z}_{k+1})$ inductively yields
\[
\lVert \nabla f_{{\bf z}_k}({\bf w}_k)\rVert\le\widetilde\varepsilon~~~\text{for}~~~ k=0,\ldots,K+1.
\]
For each $k=0,\ldots,K$, Lemma~\ref{lem:oracle} gives
\[
\lVert \widehat{{\bf g}}_k-\nabla\Phi_{2L}({\bf z}_k)\rVert\stackrel{\eqref{eq:model-errors}}{\le}\frac{\varepsilon}{128}
\quad\text{and}\quad
\lVert \widehat{{\bf H}}_k-\nabla^2\Phi_{2L}({\bf z}_k)\rVert\stackrel{\eqref{eq:model-errors}}{\le}\frac{\sqrt{\widetilde{\rho}\varepsilon}}{128}.
\]
By Proposition~\ref{prop:envelope-basic}, we also have
\[
\Phi_{2L}({\bf z}_0)-\inf_{{\bf z}}\Phi_{2L}({\bf z})
=\Phi_{2L}({\bf z}_0)-\inf_{{\bf z}}\Phi({\bf z})
\stackrel{\eqref{eq:envelope}}{\le}\Phi({\bf z}_0)-\inf_{{\bf z}}\Phi({\bf z})\le\Delta.
\]
Applying Lemma~\ref{lem:cubic} with $g(\cdot)=\Phi_{2L}(\cdot)$ and $\rho_g=\widetilde{\rho}$ yields
\begin{equation}\label{eq:nonterminal-step-bounds}
K\stackrel{\eqref{eq:cubic-sum}}{\le}4\cdot8^{\frac{3}{2}}\Delta\sqrt{\widetilde{\rho}}\,\varepsilon^{-\frac{3}{2}}
\quad\text{and}\quad
\sum_{k=0}^{K-1}r_k^3\stackrel{\eqref{eq:cubic-sum}}{\le}\frac{4\Delta}{\widetilde{\rho}}.
\end{equation}
At iteration $K$, Lemma~\ref{lem:cubic} gives
\[
\lVert \nabla\Phi_{2L}({\bf z}_{K+1})\rVert\stackrel{\eqref{eq:cubic-stop}}{<}\frac{\varepsilon}{2}
\quad\text{and}\quad
\nabla^2\Phi_{2L}({\bf z}_{K+1})
\stackrel{\eqref{eq:cubic-stop}}{\succeq}-\frac{3}{4}\sqrt{\widetilde{\rho}\varepsilon}\,{\bf I}_{d_x}.
\]
The stationarity conclusion of Lemma~\ref{lem:oracle}, applied with ${\bf z}={\bf z}_{K+1}$ and ${\bf w}={\bf w}_{K+1}$, yields
\[
\lVert \nabla\Phi({\bf u}_{K+1})\rVert\le\varepsilon
\quad\text{and}\quad
\nabla^2\Phi({\bf u}_{K+1})\succeq-\sqrt{\widetilde{\rho}\varepsilon}\,{\bf I}_{d_x}.
\]
Then, we bound the total oracle complexity of ECRN.
The definitions of $R$ and $\widetilde{\rho}$, together with $\varepsilon\le L^2/\widetilde{\rho}$, give
\begin{align}
\label{eq:delta-R-bound-1}
&\frac{\sqrt{\varepsilon}}{R\sqrt{\widetilde{\rho}}}
=\frac{\sqrt{\widetilde{\rho}\varepsilon}}{\widetilde{\rho}R}
\le\frac{L}{\widetilde{\rho}R}
=48\left(\frac{\kappa}{4+\kappa}\right)^{\frac{3}{2}}\le48,
\end{align}
which gives
\begin{align}
\label{eq:delta-R-bound-2}
\frac{\Delta}{R\varepsilon}
=\frac{\sqrt{\varepsilon}}{R\sqrt{\widetilde{\rho}}}
\Delta\sqrt{\widetilde{\rho}}\,\varepsilon^{-\frac{3}{2}}
\stackrel{\eqref{eq:delta-R-bound-1}}{\le}48\Delta\sqrt{\widetilde{\rho}}\,\varepsilon^{-\frac{3}{2}}.
\end{align}
For $k=0,\ldots,K-1$, the stopping condition is not satisfied, so $r_k>\sqrt{\varepsilon/(8\widetilde{\rho})}$. Thus, it holds that
\begin{equation}\label{eq:r_k_bound}
r_k=\frac{r_k^3}{r_k^2}\le\frac{8\widetilde{\rho}}{\varepsilon}r_k^3.
\end{equation}
The number of segments in NewtonTrack method satisfies that
\begin{equation}\label{eq:m_k_bound}
m_k=\left\lceil \frac{r_k}{R}\right\rceil\le1+\frac{r_k}{R}.
\end{equation}
Summing up \eqref{eq:m_k_bound} gives
\begin{equation}\label{eq:nonterminal-segments}
\begin{aligned}
\sum_{k=0}^{K-1}m_k
&\stackrel{\eqref{eq:m_k_bound}}{\le}K+\frac{1}{R}\sum_{k=0}^{K-1}r_k\stackrel{\eqref{eq:r_k_bound}}{\le}
K+\frac{8\widetilde{\rho}}{R\varepsilon}\sum_{k=0}^{K-1}r_k^3\\
&\stackrel{\eqref{eq:nonterminal-step-bounds}}{\le}
K+\frac{32\Delta}{R\varepsilon}\stackrel{\eqref{eq:nonterminal-step-bounds},\eqref{eq:delta-R-bound-2}}{=}
\mathcal O(\Delta\sqrt{\widetilde{\rho}}\,\varepsilon^{-\frac{3}{2}}).
\end{aligned}
\end{equation}
At iteration $K$, the stopping test gives $r_K\le\sqrt{\varepsilon/(8\widetilde{\rho})}$. Thus, we have
\[
m_K\stackrel{\eqref{eq:m_k_bound}}{\le}1+\frac{r_K}{R}
\le1+\frac{\sqrt{\varepsilon}}{\sqrt8R\sqrt{\widetilde{\rho}}}
\stackrel{\eqref{eq:delta-R-bound-1}}{\le}1+\frac{48}{\sqrt8}.
\]
Consequently, the total number of inner solves performed by NewtonTrack is
\begin{equation}\label{eq:total-segments}
N_{\rm seg}:=\sum_{k=0}^{K}m_k
=\mathcal O\bigl(\Delta\sqrt{\widetilde{\rho}}\,\varepsilon^{-\frac{3}{2}}\bigr).
\end{equation}
The initialization cost from Lemma~\ref{lem:pce-dm} and the per-call evaluation counts from Lemma~\ref{lem:tracking} give the final oracle complexity
\[
\begin{aligned}
&N_{\rm grad}+N_{\rm Hess}\\
&\quad\stackrel{\eqref{eq:local-newton-horizon}}{\le}
\underbrace{\widetilde{\mathcal O}(\sqrt\kappa)}_{\text{PCE-DM}}+\underbrace{K+1}_{\text{outer level}}
+\underbrace{\left(2\left\lceil\log_2\log_2\frac{6L\mu}{\rho\widetilde\varepsilon}\right\rceil+1\right)N_{\rm seg}}_{\text{inner level}}\\
&\quad\stackrel{\eqref{eq:cubic-sum}}{\le}\widetilde{\mathcal O}(\sqrt\kappa)+ \mathcal{O}(\Delta \sqrt{\tilde{\rho}}\epsilon^{-\frac{3}{2}})
+\mathcal O\!\left(N_{\rm seg}\log\log\frac{L\mu}{\rho\widetilde\varepsilon}\right)\\
&\quad \stackrel{\eqref{eq:total-segments}}{\le} \tilde{O}(\Delta\sqrt\rho\,\kappa^{\frac{3}{4}}\varepsilon^{-\frac{3}{2}} ).
\end{aligned}
\]

<!-- definition-2: body_md -->
\label{dfn:so-alg-class}
Let ${\mathbb{O}}({\bf w}) = (\nabla f({\bf w}), \nabla^2 f({\bf w}))$ be the second-order oracle.
A deterministic second-order algorithm \texttt{A} consists of a sequence of measurable mappings $\{ \texttt{A}^{(t)} \}_{t \in {\mathbb{N}}}$ such that $\texttt{A}^{(t)}$ takes in the first $t-1$ oracle responses  to produce the $t$-th query,  \textit{i.e.}, \texttt{A} generates a sequence $\{{\bf w}^{(t)}\}_{t \in {\mathbb{N}}}$ with
\begin{align*}
    {\bf w}^{(t)} = \texttt{A}^{(t)} ( {\mathbb{O}}({\bf w}^{(0)}), {\mathbb{O}}({\bf w}^{(1)}), \cdots, {\mathbb{O}}({\bf w}^{(t-1)}) ).
\end{align*}

\lVert \nabla^2\psi({\bf y})-\nabla^2\psi({\bf y}')\rVert \le\frac{\beta^3\rho_h}{(\frac{L}{2}-c_\varepsilon)^3}\lVert {\bf y}-{\bf y}'\rVert\ \le\frac{(\frac{5}{8})^{\frac{3}{2}}}{10 \cdot (\frac{3}{8})^3} \lVert {\bf y}-{\bf y}'\rVert <\rho\lVert {\bf y}-{\bf y}'\rVert.
\]
And the gradient smoothness in Assumption \ref{ass:smooth} is also satisfied, because for $\kappa \ge 16$,  we have
\[
\lVert \nabla^2f\rVert
\le \frac{L}{2}+\beta+\frac{5}{3}\mu
\le L\left(\frac{1}{2}+\sqrt{\frac{5}{8\kappa}}+\frac{5}{3\kappa}\right)
\le L.
\]
Second, the strong concavity Assumption \ref{ass:concave} holds due to the lower curvature lower bound of ${\bf U}^\top {\bf y}$ in Lemma \ref{lem:pro-psi} and the curvature in ${\bf y}_\perp$ is $2\beta^2/L \in[\mu,5\mu/4]$. Third, we examine the final Assumption \ref{ass:gap}. Since $\Phi_{\bf U}({\bf x}) = \alpha \bar h_T({\bf U}^\top{\bf x}/\sigma)$, Lemma \ref{lem:carmon}(i) indicates that
\begin{equation}\label{eq:scaled-gap-and-grad}
\Phi_{\bf U}({\bf 0})-\inf_{{\bf x} \in \mathbb R^d} \Phi_{\bf U}({\bf x})\le12T\alpha.
        
                                                          
\end{equation}
Hence, $\Phi_{\bf U}$ has initial gap at most $\Delta$ if we take
\begin{equation}\label{eq:chain-length}
T=
\left \lfloor \frac{\Delta}{12 \alpha}  \right \rfloor = 
\left\lfloor\frac{\Delta\sqrt{\rho_h}}{12\cdot 2^{\frac{3}{2}}\sqrt{\ell_2}\,\varepsilon^{\frac{3}{2}}}\right\rfloor.
\end{equation}
Since $\alpha/\sigma = 2 \varepsilon$ under the parameter setting in \eqref{eq:scaling}, we can combine Lemma \ref{lem:carmon}(v) and Lemma \ref{lem:resisting} to conclude that
\[
\lVert \nabla\Phi_{\bf U}({\bf x}^{(t)})\rVert
= \frac{\alpha}{\sigma} \lVert {\bf U} \nabla \bar h_T({{\bf U}^\top{\bf x}^{(t)}}/{\sigma})\rVert
= \frac{\alpha}{\sigma} \lVert \nabla \bar h_T({{\bf U}^\top{\bf x}^{(t)}}/{\sigma})\rVert>2\varepsilon.
\]
for any $t <T$. Substitution of $T$ in \eqref{eq:chain-length} proves Theorem~\ref{thm:lower} after choosing the universal constants.
