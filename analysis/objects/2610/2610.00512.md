<!-- theorem-1: body_md -->
\label{nutemp}
    For every $t\geq0$, the convolution kernel $\nu_t$ defines a tempered distribution, i.e, $\nu_t\in \mathcal S'(\mathbb{R}^d)$.
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
By assumptions $(P1), (P2)$, and $(P3)$, the function $\mathbb{R}^d\ni\xi\mapsto e^{t\Psi(-i\xi)}$, $t\geq0$, is infinitely differentiable
with polynomially bounded derivatives. Therefore it belongs to $\mathcal O_M(\mathbb{R}^d)$, and by \cite{horvath}, Chap.4, Sec.11, Theorem 2, it defines a tempered distribution. Hence, by duality,
$\nu_t\in\mathcal S'(\mathbb{R}^d)$ satisfying
\begin{equation}\label{nuhat}
\widehat{\nu_t}(\xi)=e^{t\Psi(-i\xi)},\qquad \xi\in\mathbb{R}^d.
\end{equation}
<!-- end proof-1 -->
<!-- assumption-1: body_md -->
\label{assumptions} Throughout this work, we will assume the following: 

\begin{enumerate}
    \item\label{I} $f:\mathbb{R}\rightarrow \mathbb{R}$ is a function in the Schwartz space on $\mathbb{R}$ (i.e $f\in \mathcal S(\mathbb{R})$).
    \item\label{II} $V:\mathbb{R}^d\to\mathbb{R}$ is a coercive polynomial, i.e, $|V(\phi)|\to\infty$ as $|\phi|\to\infty$.
    \item\label{III} The symbol of the pseudodifferential operator $\Psi(\nabla)$ is of Lévy type, that is 
    \begin{equation}\label{levysymbol}
\Psi(-i\xi) = i\langle a, \xi\rangle - \langle \xi, D\xi\rangle 
+ z \int_{\mathbb{R}^d \setminus \{0\}} \big(e^{i\langle \varphi,\xi\rangle} - 1\big)\, dr(\varphi), \qquad \xi\in\mathbb{R}^d.
\end{equation}
where $a \in \mathbb{R}^d$, $D\in \mathcal M_d(\mathbb R)$ symmetric positive semi-definite matrix,  { $z\geq0$ is the jump intensity,}
 and $r$ is a probability measure on $\mathbb{R}^d \setminus \{0\}$ with entire analytic Fourier transform.
\end{enumerate}
<!-- end assumption-1 -->
<!-- theorem-2: body_md -->
\label{fschwartz}
    For $\beta \in \mathbb{R}^*$, we have $f(\beta V)\in\mathcal S(\mathbb{R}^d)$.
<!-- end theorem-2 -->
<!-- proof-2: body_md -->
The idea of the proof is based on the extension of \cite[Theorem 2.3]{galbis} through \cite[Remark 2.4]{galbis}. We first prove that the map $\phi\mapsto \beta V(\phi)$ is a Schwartz symbol on $\mathbb{R}^d$. For that, we fix $\beta\in\mathbb{R}^*$. By Assumption \ref{assumptions},
the polynomial $P=V^2$ is coercive. By
\cite[Section~2, Eq.~(2), and Corollary~3.4]{bajbar},
there exist $a,R,\ell>0$ such that
\begin{equation}
V(\phi)^2\geq a|\phi|^\ell,
\qquad |\phi|\geq R.
\end{equation}

Hence
\begin{equation}
|\beta V(\phi)|
\geq |\beta|\sqrt a\,|\phi|^{\ell/2},
\qquad |\phi|\geq R.
\end{equation}

Let now $c=|\beta|\sqrt a>0$ and $q=\ell/2>0$, and choose $K>1$ such that
\begin{equation}
K\geq\max\left\{R,\frac{2}{q},c^{-2/q}\right\}.
\end{equation}

For $|\phi|\geq K$, we have $c|\phi|^{q/2}\geq1$
and $1/K\leq q/2$. Hence
\begin{equation}
|\beta V(\phi)|
\geq c|\phi|^q
=\bigl(c|\phi|^{q/2}\bigr)|\phi|^{q/2}
\geq|\phi|^{q/2}
\geq|\phi|^{1/K},
\end{equation}
which proves the $d$-dimensional analogue of condition (ii) in \cite[Theorem 2.3]{galbis}.

For every multiindex $\alpha$, the derivative
$\partial^\alpha(\beta V)$ is a polynomial.
Therefore we can find constants $C_{\alpha,\beta},p_\alpha>0$ such that
\begin{equation}
|\partial^\alpha(\beta V)(\phi)|
\leq C_{\alpha,\beta}
     (1+|\beta V(\phi)|^2)^{p_\alpha},
\qquad \phi\in\mathbb{R}^d.
\end{equation}

The analogue growth conditions in the sufficient
part of \cite[Theorem~2.3 and Remark~2.4]{galbis} are satisfied. Since $f\in\mathcal S(\mathbb{R})$ by
Assumption \ref{assumptions}, we obtain
$f(\beta V)\in\mathcal S(\mathbb{R}^d)$.
<!-- end proof-2 -->
<!-- theorem-3: body_md -->
\label{existuniq}
Under assumptions \ref{assumptions}, the Cauchy problem \eqref{problem} has a unique solution \\$\Phi_t^\beta\in\mathcal S'(\mathbb{R}^d)$ given by
\begin{equation}\label{phisol}
\Phi_t^\beta(\phi)=(\nu_t*f_\beta)(\phi),\qquad \phi\in \mathbb{R}^d, \quad \beta\in\mathbb{R}^*
\end{equation}
<!-- end theorem-3 -->
<!-- proof-3: body_md -->
{
By Theorem \ref{fschwartz}, we have $f_\beta\in\mathcal S$. On the other hand by (P2)-(P3), the multiplier $e^{t\Psi(-i\xi)}$ and its time derivative have polynomially bounded derivatives. Thus $e^{t\Psi(-i\xi)}\widehat{f_\beta}$ defines a tempered distribution, with initial value $\widehat{f_\beta}$ and derivative $\Psi(-i\xi)e^{t\Psi(-i\xi)}\widehat{f_\beta}$. Applying the Fourier transform proves existence and equality \eqref{phisol} holds. For uniqueness, the Fourier transform of the difference of two solutions satisfies the homogeneous multiplier equation with zero initial value. Multiplying by $e^{-t\Psi(-i\xi)}$ gives zero, hence the difference is zero in $\mathcal S'$.
}

<!-- lemma-1: body_md -->
\label{leib} Let $f,g\in \mathcal C^\infty(\mathbb{R}^d)$ be real valued functions. Then, for $\alpha \in \mathbb{N}_0^d$, we have
\begin{equation}
\partial^\alpha (fg)
  = \sum_{\gamma \le \alpha} \binom{\alpha}{\gamma}\,
    \partial^\gamma f \, \partial^{\alpha-\gamma} g.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-4: body_md -->
The proof is achieved via induction on $|\alpha|$ by applying the product rule. The two terms at each step are combined by Pascal's identity for the multiindex binomial coefficients.

<!-- theorem-4: body_md -->
\label{faadibruno}
Let $h\in \mathcal C^\infty(\mathbb{R}^d)$, $g\in \mathcal C^\infty(\mathbb{R})$ be real valued functions. Then, for every  $\alpha\in{ \mathbb{N}_0^d\setminus\{0\}}$,
\begin{equation}\label{faadibrunoeq}
\partial^\alpha\bigl[g(h(\phi))\bigr]
=
\sum_{k=1}^{|\alpha|}
g^{(k)}(h(\phi))
\sum_{(j_\mu)\in\mathcal J_{\alpha,k}}
\frac{\alpha!}{\displaystyle\prod_\mu j_\mu!\,(\mu!)^{j_\mu}}
\prod_\mu \bigl(\partial^\mu h(\phi)\bigr)^{j_\mu},
\end{equation}
<!-- end theorem-4 -->
<!-- proof-5: body_md -->
This follows from \cite{hernandez} Theorem 1.1 .
<!-- end proof-5 -->
<!-- corollary-1: body_md -->
\label{faadibrunopol}
If $V:\mathbb{R}^d\to\mathbb{R}$ is a polynomial, $g\in \mathcal C^\infty(\mathbb{R})$, and $\alpha\in{ \mathbb{N}_0^d\setminus\{0\}}$, then $\partial^\alpha[g(V(\phi))]$ is a finite sum of terms of the form
\begin{equation}
P_{\alpha,k}(\phi)\,g^{(k)}(V(\phi)),\qquad 1\leq k \leq |\alpha|,
\end{equation}
where each $P_{\alpha,k}$ is a polynomial in $\phi$.
<!-- end corollary-1 -->
<!-- proof-6: body_md -->
By applying Theorem \ref{faadibruno}, and by the fact that for every multiindex $\mu\leq\alpha$, 
$\partial^\mu V$ is again a polynomial, the result follows.

<!-- theorem-5: body_md -->
\label{infdiff}For $m\in\mathbb{N}$, the { map $\beta\mapsto f(\beta V)$} is infinitely differentiable { on $\mathbb{R}^*$} and
    \begin{equation}\label{differen}\partial^{m}_\beta f(\beta V(\phi)):=\frac{\,\mathrm{d}^{m}}{\,\mathrm{d} \beta^{m}}f(\beta V(\phi))=V^{m}(\phi)f^{(m)}(\beta V(\phi))\qquad \text{ in } \mathcal{S}(\mathbb{R}^d),
\end{equation}
where $V^m(\phi)=\bigl(V(\phi)\bigr)^m$.
<!-- end theorem-5 -->
<!-- proof-7: body_md -->
Let $\beta\ne0$ and let $J\subset\mathbb{R}^*$ be a compact interval  containing $\beta$. By the coercivity of $V$, $|V(\phi)|\ge c|\phi|^q$ on $\{|\phi|>r\}$ for some $r,c,q>0$. The chain rule and rapid decay of every derivative of $f$ imply
\begin{equation}
\sup_{\sigma\in J}\big\|V^r f^{(m)}(\sigma V)\big\|_{\alpha,N}<\infty,
\end{equation}
for all $r,N,m\in\mathbb{N}_0$ and $\alpha\in\mathbb{N}_0^d$. Since spatial derivative is a finite sum of polynomial factors times $f^{(m+j)}(\sigma V)$, the lower bound absorbs any polynomial factor uniformly for $\sigma\in J$. By Taylor's integral remainder, we get
\begin{equation}\label{taylor1}
\frac{V^m f^{(m)}((\beta+h)V)-V^m f^{(m)}(\beta V)}{h}
-V^{m+1}f^{(m+1)}(\beta V)
=h\!\int_0^1(1-\theta)V^{m+2}f^{(m+2)}((\beta+\theta h)V)\,d\theta.
\end{equation}

The Schwartz seminorm of the L.H.S is at most $C_{\alpha,N,m,\beta}|h|$ for small $h$. Finally, proceeding by induction on $m$ proves \eqref{differen}.

<!-- definition-1: body_md -->
\label{def:nu-pairing}
For every $f\in\mathcal S(\mathbb R^d)$, we use the notation $\langle f\rangle_{\nu_t}$ to indicate the canonical pairing between $f\in \mathcal{S}$ and $\nu_t\in \mathcal{S}'(\mathbb{R}^d)$, that is
\begin{equation}\label{eq:nu-pairing}
\langle f\rangle_{\nu_t}
:=
\langle \nu_t,f\rangle.
\end{equation}
Under Assumption \ref{assumptions}(3), $\nu_t$ is the probability measure with characteristic function $e^{t\Psi(-i\xi)}$. The analyticity of the Fourier transform of $r$ implies finite polynomial moments of every order; the same holds for the Gaussian and compound-Poisson convolution in \eqref{levysymbol}. We use the same notation for any $\nu_t$-integrable function:
$$
\langle f\rangle_{\nu_t}:=\int_{\mathbb{R}^d}f(\phi)\,\nu_t(d\phi).
$$

\begin{equation}
\lim_{\beta\rightarrow0}\partial^m_\beta \Phi^\beta_t=f^{(m)}(0)\langle V^m_\phi\rangle_{\nu_t}\qquad\text{ in }\mathcal{S}'(\mathbb{R}^d),
\end{equation}

\label{physical}
Under Assumptions \ref{assumptions} and Theorem \ref{fschwartz} 
\begin{equation}\label{physical-limit}
\lim_{\beta\to0} V^m f^{(m)}(\beta V)=f^{(m)}(0)V^m
\qquad\text{ in }\mathcal{S}'(\mathbb{R}^d), 
\end{equation}
<!-- end lemma-2 -->

For each $\psi\in\mathcal S(\mathbb{R}^d)$, the integrand $V^m(\phi)f^{(m)}(\beta V(\phi))\psi(\phi)$ converges pointwise to \\$f^{(m)}(0)V^m(\phi)\psi(\phi)$ and is bounded by $\|f^{(m)}\|_\infty|V^m\psi|\in L^1$. <!-- remark-1: body_md -->
By Theorem \ref{expansion}, the solution $\Phi_t^\beta$ admits a formal expansion whose coefficients are distributional moments of the powers of $V_\phi$. These moments can be reorganized combinatorially and represented as sums indexed by a specific class of graphs. Each graph corresponds to one contribution in the expansion, and its value is determined by a fixed evaluation procedure, referred to as the \emph{Feynman rules}.

<!-- definition-2: body_md -->
Let $\sigma,\pi\in\mathcal P([m])$. 
\begin{enumerate}

\item We say that $\sigma$ is finer than $\pi$ (or that $\pi$ is coarser than $\sigma$) and we denote it $\sigma\le \pi$ whenever every block of $\sigma$
is contained in some block of $\pi$.
Equivalently, $\pi$ can be obtained by repeatedly merging blocks of $\sigma$.

\item We define the join of $\sigma$ and $\pi$, denoted $\sigma\vee\pi$, as the the smallest partition which is coarser than both $\sigma$ and $\pi$.

\item If $\sigma\le \pi$, we define the \textit{Möbius inversion function} $\mu(\sigma,\pi)$
\begin{equation}
\mu(\sigma,\pi)
=
(-1)^{|\sigma|-|\pi|}
\prod_{B\in\pi}
\bigl(k_B-1\bigr)!,
\end{equation}
where $k_B$ is the number of blocks of $\sigma$ contained in $B$.
If $\sigma\not\le\pi$, then $\mu(\sigma,\pi)=0$.
\end{enumerate}
<!-- end definition-2 -->
<!-- proposition-1: body_md -->
\label{prop:refinement-poset}
Let $m\in\mathbb{N}$. The relation "$\le$" on $\mathcal P([m])$ is a partial order.
<!-- end proposition-1 -->
<!-- proof-10: body_md -->
Reflexivity: every block of $\sigma$ is contained in itself, so $\sigma\le\sigma$.
Transitivity: if $\sigma\le\pi$ and $\pi\le\rho$, then every block of $\sigma$ lies in a block of $\pi$,
and that block lies in a block of $\rho$, hence $\sigma\le\rho$.
Antisymmetry: if $\sigma\le\pi$ and $\pi\le\sigma$, then each block of $\sigma$ is contained in a block of $\pi$
and conversely; because the blocks are disjoint and cover $[m]$, this forces $\sigma=\pi$.
<!-- end proof-10 -->
<!-- definition-3: body_md -->
\label{distruncdef}
For $\alpha=(\alpha_1,\cdots,\alpha_d)\in \mathbb{N}_0^d$ { with $|\alpha|\geq1$}, the $\alpha$-th truncated distributional moment of $\nu_t$ is defined by
\begin{equation}\label{distrunc}

\label{cluster}

$i_1,\dots,i_m\in\{1,\dots,d\}$. Then, for every $m\ge1$,

\sum_{\pi\in\mathcal P([m])}
\prod_{B\in\pi}
\Big\langle \prod_{r\in B}\phi_{i_r}\Big\rangle_{\nu_t}^T.
\end{equation}

\Big\langle \prod_{r=1}^m \phi_{i_r}\Big\rangle_{\nu_t}^T

\prod_{B\in\pi}
\Big\langle \prod_{r\in B}\phi_{i_r}\Big\rangle_{\nu_t},
\end{equation}
where $\hat{1}$ is the maximal element of $\mathcal P([m])$, i.e, the coarsest partition.

<!-- definition-5: body_md -->
\label{Feyndef}
A \emph{generalized amputated Feynman graph} is a graph with three types of vertices:
\emph{inner full}, \emph{inner empty}, and \emph{outer empty} vertices.
Inner full vertices are distinguishable and have distinguishable legs; empty vertices are non-distinguishable and have non-distinguishable legs.
Each outer empty vertex is met by exactly one edge. { Each inner empty vertex is met by at least one edge. Every edge joins an inner full vertex to an empty vertex. Parallel edges are distinguished by the labels of their legs at the full vertex.}


Fix $m\in\mathbb N_0$. Let $p_i$, $1\leq i \leq m$, be the numbers of legs of the $m$ inner full vertices, with ${ 0\leq}p_j\le \bar p$ and $\lambda^{(p_j)}\neq 0$.
The collection of all such graphs is denoted by ${\mathcal F}(m)$.
<!-- end definition-5 -->
<!-- remark-3: body_md -->
The above construction establishes the one-to-one correspondence between pairs $(K,I)$ and graphs in ${\mathcal F}(m)$, see \cite{main}, Lemma 2.9 for more details.

<!-- definition-6: body_md -->
\label{feynman}
Let $G\in{\mathcal F}(m)$, and let $p_1,\dots,p_m$ denote the numbers of legs attached to the inner full vertices of $G$.
For $\phi\in\mathbb{R}^d$ and $t{ \geq}0$, the value $V[G](t,\phi)\in\mathbb{R}$ is defined according to the following procedure:
\begin{enumerate}
\item To each leg of an inner full vertex, assign an index $i_a\in\{1,\dots,d\}$, where $a\in\Omega(p_1,\dots,p_m)$ labels that leg. 
\item For each outer empty vertex, multiply by the factor $({ +}\phi_{i_b})$, where $b\in\Omega(p_1,\dots,p_m)$ is the label of the leg of the inner full vertex that is directly connected to that outer empty vertex. 
\item For each inner empty vertex with $l$ legs, multiply by the factor
\begin{equation}
    t\,{ l!}C_J \;=\; t\,{ l!}C{ ^{(l)}}_{i_{a_1}\cdots i_{a_l}},
\end{equation}
where $J=\{a_1,\dots,a_l\}\subseteq\Omega(p_1,\dots,p_m)$ is the subset of labels corresponding to the $l$ legs of inner full vertices directly connected to that inner empty vertex. 
\item For the $j$th inner full vertex (with $p_j$ legs), multiply by the factor
\begin{equation}
\lambda^{(p_j)}_{i^j_1\cdots i^j_{p_j}}
\end{equation}
and sum over all assignments of the indices $i^j_1,\dots,i^j_{p_j}\in\{1,\dots,d\}$. 
\end{enumerate}
<!-- end definition-6 -->
<!-- remark-4: body_md -->
\hfill
\begin{enumerate}

\item  We use the convention that $\mathcal F(0)$ consists of the empty graph only, denoted $\varnothing$, and that
\begin{equation}
V[\varnothing](t,\phi)=1.
\end{equation}

\item By construction, the value of a graph $G$ is obtained by multiplying the local factors associated with { all its vertices} and then summing over all assignments of indices. It is therefore natural to expect this value to factorize when the graph splits into disjoint components.
\end{enumerate}

<!-- proposition-2: body_md -->
\label{multiplicativity}
Let $m_1,m_2\in\mathbb N_0$, and let $G_1\in \mathcal F(m_1)$ and $G_2\in \mathcal F(m_2)$ be two generalized amputated Feynman graphs. Let 
\begin{equation}
G=G_1\sqcup G_2,
\end{equation}
where $\sqcup$ denotes the disjoint union{ , with the full-vertex labels of $G_2$ shifted by $m_1$}. Then $G\in \mathcal F(m_1+m_2)$ and
\begin{equation}
V[G](t,\phi)=V[G_1](t,\phi)\,V[G_2](t,\phi).
\end{equation}
<!-- end proposition-2 -->
<!-- proof-13: body_md -->
Since $G_1\in\mathcal F(m_1)$ and $G_2\in\mathcal F(m_2)$, the graphs $G_1$ and $G_2$ have respectively $m_1$ and $m_2$ inner full vertices. Their disjoint union $G=G_1\sqcup G_2$ has $m_1+m_2$ inner full vertices and still satisfies the defining properties of a generalized amputated Feynman graph given in Definition \ref{Feyndef}, hence $G\in\mathcal F(m_1+m_2)$. By Definition \ref{feynman}, the value $V[G](t,\phi)$ is obtained by multiplying the factors assigned to the vertices of $G$ and summing over all index assignments. 

Since $G=G_1\sqcup G_2$, both the factors and the index assignments split into indices associated with $G_1$ and $G_2$ respectively. Therefore the evaluation defining $V[G](t,\phi)$ factorizes into the product of the evaluations defining $V[G_1](t,\phi)$ and $V[G_2](t,\phi)$. 

Thus
\begin{equation}
V[G](t,\phi)=V[G_1](t,\phi)\,V[G_2](t,\phi).
\end{equation}

\label{lem:multi-to-part}

\begin{equation}

\begin{enumerate}
\item A decomposition of the $n$ labeled copies of $V_\phi$ into $k$ ordered groups of respective sizes $r_1,\dots,r_k$.

\begin{equation}

\end{equation}

is a bijection labeling the blocks of $\pi$, with the property that for each
$a\in\{1,\dots,k\}$, the unique block $B\in\pi$ satisfying $l(B)=a$ has

<!-- proposition-3: body_md -->
\label{truncated-to-graph}

\begin{equation}\label{trunctograph}

\end{equation}

The product of the block cumulants equals the sum of $V[G]$ over graphs with connectedness partition $\sigma(G)=\tau$. On the other hand, the moment expansion in Theorem \ref{expansion2} together with Definition \ref{feynman}, gives the full moment as a sum over all $G\in\mathcal F(C)$. Finally, Theorem \ref{cluster} gives the same moment as a sum over $\tau\in\mathcal P(C)$ of products of block cumulants. We conclude by cancelling contributions of $\tau\ne\hat1_C$, which proves the Proposition.

\sum_{\substack{G\in\mathcal F(n)\\ \sigma(G)\vee\pi=\hat 1}} V[G](t,\phi).
\end{equation}

Therefore
\begin{equation}

=
\sum_{n\ge1}\beta^n
\sum_{\pi\in\mathcal P([n])}
w(\pi)\,
\sum_{\substack{G\in\mathcal F(n)\\ \sigma(G)\vee\pi=\hat 1}} V[G](t,\phi).
\end{equation}

This concludes the proof.

\label{Watson-Sokal}
Let $F$ be analytic in $\mathcal C_R$. Assume that there exist
constants $A,\sigma>0$ and coefficients $(a_m)_{m\geq0}$ such that
\begin{equation}\label{eq:Sokal-remainder}
\left|
F(\zeta)-\sum_{m=0}^{N-1}a_m\zeta^m
\right|
\leq
A\sigma^N N!|\zeta|^N,
\qquad
N\geq1,\quad \zeta\in\mathcal C_R.
\end{equation}

Then the Borel transform
\begin{equation}\label{eq:Borel-transform}
\mathcal BF(\tau)
:=
\sum_{m\geq0}\frac{a_m}{m!}\tau^m
\end{equation}
converges for $|\tau|<\sigma^{-1}$ and admits an analytic continuation
to the strip-like region
\begin{equation}\label{eq:Ssigma}
\mathcal S_\sigma
:=
\left\{
\tau\in\mathbb C:
\operatorname{dist}(\tau,\mathbb R_+)<\sigma^{-1}
\right\}.
\end{equation}

For every $\sigma'>\sigma$, there exists $K_{\sigma'}>0$ such that
\begin{equation}\label{eq:Borel-growth}
|\mathcal BF(\tau)|
\leq
K_{\sigma'}
\exp\left(\frac{|\tau|}{R}\right),
\qquad
\tau\in\mathcal S_{\sigma'}.
\end{equation}

Moreover, for every $\zeta\in\mathcal C_R$,
\begin{equation}\label{eq:Borel-Laplace}
F(\zeta)
=
\frac{1}{\zeta}
\int_0^\infty
e^{-\tau/\zeta}\mathcal BF(\tau)\,d\tau.
\end{equation}

Conversely, suppose that $B$ is analytic in some
$\mathcal S_\sigma$ and, for every $\sigma'>\sigma$, satisfies
\begin{equation}\label{eq:Borel-growth-converse}
|B(\tau)|
\leq
K_{\sigma'}
\exp\left(\frac{|\tau|}{R}\right),
\qquad
\tau\in\mathcal S_{\sigma'}.
\end{equation}

Then
\begin{equation}\label{eq:Laplace-converse}
F(\zeta)
:=
\frac{1}{\zeta}
\int_0^\infty e^{-\tau/\zeta}B(\tau)\,d\tau
\end{equation}
is analytic in $\mathcal C_R$ and satisfies
\eqref{eq:Sokal-remainder}, uniformly on every
$\mathcal C_{R'}$ with $0<R'<R$, with
$$
a_m=B^{(m)}(0).
$$

Under Assumption \ref{Borel}, the function $H$ is analytic in

<!-- proof-19: body_md -->
{
We differentiate the integral in \eqref{M} with respect to
$\zeta$. This gives
\begin{equation}
M^{(j)}(\zeta)
=
(-1)^j\langle V_\phi^j e^{-\zeta V_\phi}\rangle_{\nu_t},
\end{equation}
and hence \eqref{eq:H-from-M} and the analyticity of $H$.
Using \eqref{eq:M-expansion}, the formal coefficient of $\zeta^n$
in \eqref{eq:H-from-M} is
\begin{equation}
a_n\sum_{j=0}^{\min\{k,n\}}
p_j(-1)^j\frac{n!}{(n-j)!}
=
\langle V_\phi^n\rangle_{\nu_t}
\sum_{j=0}^{\min\{k,n\}}
\frac{p_j(-1)^{n-j}}{(n-j)!}
=
\frac{f^{(n)}(0)}{n!}
\langle V_\phi^n\rangle_{\nu_t}.
\end{equation}

This gives \eqref{eq:H-coefficients} at the level of formal
coefficients. Their Borel transform satisfies \eqref{eq:BH}. It remains to verify the analytic continuation and growth properties
of $\mathcal BH$. By Assumptions \ref{Borel}, compact support of the jump distribution gives
finite exponential moments of linear growth for $\nu_t$. Since $\sqrt{V_\phi}$ has at most
linear growth, we can choose $c>0$ such that
\begin{equation}
A:=\langle e^{c\sqrt{V_\phi}}\rangle_{\nu_t}<\infty.
\end{equation}

It follows for every $N\geq1$,
\begin{equation}
\langle V_\phi^N\rangle_{\nu_t}
\leq Ac^{-2N}(2N)!
\leq A\sigma^N(N!)^2,\quad \sigma=4c^{-2}.
\end{equation}

Taylor's remainder for $e^{-\zeta V_\phi}$ now gives
\begin{equation}
\left|M(\zeta)-\sum_{n=0}^{N-1}a_n\zeta^n\right|
\leq A\sigma^N N!|\zeta|^N,
\qquad \zeta\in\mathcal C_\infty.
\end{equation}

The constants $A$ and $\sigma$ do not depend on $R$. Thus, for
every $\widetilde R>0$ and $\widetilde\sigma>\sigma$,
Theorem \ref{Watson-Sokal} gives
\begin{equation}
|\mathcal BM(w)|
\leq K_{\widetilde\sigma,\widetilde R}
e^{|w|/\widetilde R},
\qquad w\in\mathcal S_{\widetilde\sigma}.
\end{equation}

Now, let $R>0$ be fixed. We choose $\widetilde R>R$, and let
$\sigma<\widetilde\sigma<\sigma'$. Cauchy's estimates bound the
derivatives of $\mathcal BM$ on $\mathcal S_{\sigma'}$ using the
preceding bound on $\mathcal S_{\widetilde\sigma}$. The sum in
\eqref{eq:BH} is finite, and its polynomial factors $\tau^j$ can
be absorbed into $e^{|\tau|/R}$. Therefore $\mathcal BH$ is
analytic in $\mathcal S_\sigma$ and satisfies
\begin{equation}\label{bound}
|\mathcal BH(\tau)|
\leq C_{\sigma',R}e^{|\tau|/R},
\qquad \tau\in\mathcal S_{\sigma'}.
\end{equation}

For $\beta\in(0,R)$, we use the Borel representation
\begin{equation}
M(\beta)=\int_0^\infty e^{-s}\mathcal BM(\beta s)\,ds.
\end{equation}

The bounds in \eqref{bound} allow differentiation under
the integral. Equation \eqref{eq:H-from-M} then gives
\begin{equation}
H(\beta)
=
\int_0^\infty e^{-s}\mathcal BH(\beta s)\,ds
=
\frac{1}{\beta}\int_0^\infty
e^{-\tau/\beta}\mathcal BH(\tau)\,d\tau.
\end{equation}

Both sides are analytic in $\mathcal C_R$, hence by Theorem 3.2.6 in \cite{ablowitz}, we can
extend the equality to $\mathcal C_R$. Since $R$ is arbitrary,
\eqref{eq:H-Borel-Laplace} holds throughout $\mathcal C_\infty$.

Finally, the converse part of Theorem \ref{Watson-Sokal} gives
a factorial remainder estimate for this Borel--Laplace integral
on every $\mathcal C_{R_0}$, where $R_0<R$. Its coefficients are
the coefficients calculated above. Hence \eqref{eq:H-expansion}
is Borel summable to $H$.
}
