<!-- theorem-1: body_md -->
\label{thm:aleksandrov-sharp}
  For every real-valued Borel measurable function \(f\), any classical subsolution \(u\in C(\overline{Q^-(P_0)})\cap C^{1,1,2}(Q^-(P_0))\) of~\eqref{eq:kin}, with \(f_+\in L^p(Q^-(P_0))\), satisfies
  \[
    \sup_{\overline{Q^-(P_0)}}u
    \leq \sup_{\partial_k Q^-(P_0)}u_+
    +C_{n,\lambda,\Lambda,p}\,\|f_+\|_{L^p(Q^-(P_0))}
    \refstepcounter{equation}\tag{\theequation}
    \label{eq:abp-kin}
  \]
  provided either \(A=A(t,v)\) and \(p>2n+1\), or \(n=1\), \(A=a(x,v)\), and \(p>4\). Here, \(C_{n,\lambda,\Lambda,p}>0\) is a constructive constant depending only on \(n\), \(\lambda\), \(\Lambda\), and \(p\).

<!-- conjecture-1: body_md -->
\label{conjecture1}
There is a finite $p_0=p_0(n,\lambda,\Lambda)>1$ such that, for every $p>p_0$, every classical sub-solution $u\in C(\overline Q)$ of~\eqref{eq:kin}, with $f_+\in L^p(Q)$, satisfies
\[
    \sup_{\overline Q}u
    \leq \sup_{\partial_kQ}u_++C\left\| f_+ \right\|_{L^p(Q)},
    \refstepcounter{equation}\tag{\theequation}
    \label{eq:abp-kin-general}
\]
where $\partial_kQ$ consists of the initial, diffusive and incoming transport boundary, excluding the relative interior of the final-time face, and $C$ depends only on $n,\lambda,\Lambda,p$ and $Q$.

<!-- theorem-2: body_md -->
\label{thm:mixing}
  There are $C,c>0$, depending only on $n,\frac\Lambda\lambda$, such that
  \[
    \sup_{v\in \mathbb S^{n-1}}\left\| \widehat K_{s,t}^\xi(v,\cdot) \right\|_{\mathrm{TV}}
    \leq C\exp\left(-c(t-s)
    \min\left\{\frac{\lvert\xi\rvert^2}{\lambda},\lambda^{\frac{1}{2}}\lvert\xi\rvert^{\frac{1}{2}}\right\}\right),
    \qquad s<t,\quad \xi \in{\mathbb Z}^n\setminus\{0\}.
    \refstepcounter{equation}\tag{\theequation}
    \label{e:spherical-DF}
  \]
  In particular, the rate is $\lambda^{\frac{1}{2}}\lvert\xi\rvert^{\frac{1}{2}}$ when $\lvert\xi\rvert \geq\lambda$, hence for every non-zero mode if $\lambda\leq1$. The frequency power $\frac12$ is optimal.

                                                                                                                                                                                                                                                                                                                                                                                  
      
                                                 
                                                                
                                                  
                     
                                                 
                             
      
                                                                                                                                            
      
                                                         
                                                                
                                              
      
                                                                                                   
In the autonomous case $A=A(v)$, there exist a probability measure $\mu$ on $\mathbb S^{n-1}$ and a strongly continuous, positivity-preserving solution semigroup for~\eqref{e:spherical-equation} on $L^2(\pi)$, where $\pi={\, \mathrm d} x\otimes\mu$ and ${\, \mathrm d} x$ is normalised Haar measure on ${\mathbb T}^n$. The measure $\mu$ is the unique invariant velocity probability for this semigroup. In particular,
\begin{equation}\label{e:spherical-invariant}
\int_{\mathbb S^{n-1}}A:\nabla_S^2\varphi{\, \mathrm d}\mu=0
\qquad\forall\varphi\in C^\infty(\mathbb S^{n-1}).
\end{equation}
For every initial datum $u(s)\in L^2(\pi)$, the solution $u\in C([s,\infty);L^2(\pi))$ supplied by this semigroup conserves the mean $m=\int u(s){\, \mathrm d}\pi$ and satisfies
\begin{equation}\label{e:spherical-L2-gap}
\left\| u(t)-m \right\|_{L^2(\pi)}
\leq C\exp\bigl(-c(t-s)\min\{\lambda,\lambda^{-1}\}\bigr)\left\| u(s)-m \right\|_{L^2(\pi)},
\quad t\geq s.
\end{equation}
On the invariant subspace $\big\{f\in L^2(\pi):\int f{\, \mathrm d}\pi=0\big\}$, the generator of this solution semigroup has spectrum contained in
\[
\bigl\{z\in{\mathbb C}:\operatorname{Re}z
\leq-c\min\{\lambda,\lambda^{-1}\}\bigr\}.
\]

  \end{equation}

<!-- proposition-3: body_md -->
\label{p:decay-loc}
There are $C,c>0$, depending only on $n,\lambda,\Lambda$, such that, if
$0\leq s<s+c_0R^2\leq1$, then
\[
    \sup_{v\in B_{\frac{3R}{4}}(v_0)}
    \left\| \widehat\omega_{s,v}^{D,\xi} \right\|_{\mathrm{TV}}
    \leq C\exp\left(-c(R^3\lvert\xi\rvert)^{\frac{1}{3}}\right),
    \qquad\xi\in{\mathbb R}^n.
    \refstepcounter{equation}\tag{\theequation}
    \label{e:local-decay}
\]

<!-- remark-1: body_md -->
The loss from $\frac{2}{3}$ to $\frac{1}{3}$ comes from the contribution of velocity-boundary data at arbitrarily short elapsed times.
<!-- end remark-1 -->
<!-- proof-1: body_md -->
We first compare the evolution killed at the velocity boundary with the whole-space evolution. Their difference is the mass that has reached the boundary. For $v\in B_{\frac{3R}{4}}(v_0)$ and $0<h\leq c_0R^2$, we will show that
\[
    \left\| \widehat K^{D,\xi}_{s,s+h}(v,\cdot) \right\|_{\mathrm{TV}}
    \leq C\left(e^{-\frac{cR^2}{h}}+e^{-ch\lvert\xi\rvert^{\frac{2}{3}}}\right).
    \refstepcounter{equation}\tag{\theequation}
    \label{e:local-Dirichlet}
\]
The first term bounds the mass that has reached the boundary by time $s+h$; the second is the Fourier decay of the whole-space kernel. By scaling, it is enough to take $R=1$. We fix $t=s+h$ and, for $0\leq r\leq h$, write this lost mass as
\[
    q(r,v')=1-P^D_{t-r,t}(v',D).
\]
Then $q(0,\cdot)=0$ in $D$, $q=1$ on $\partial D$ for $r>0$, and
$\partial_rq-\mbox{Tr}(A(t-r,v')D_{v'}^2q)=0$. In particular, the mass of
$K_{s,s+h}-K^D_{s,s+h}$ is $q(h,v)$.
We set $\delta=(4\sqrt n)^{-1}$.  For $\alpha>0$,
\[
    \Psi(r,v')=\sum_{j=1}^n\sum_{\varepsilon=\pm1}
    \exp\left(\alpha\varepsilon(v'_j-v_j)-\alpha\delta
    +\Lambda\alpha^2r\right)
\]
satisfies
$(\partial_r-\mbox{Tr}(A(t-r,v')D_{v'}^2))\Psi\geq0$.  If
$v'\in\partial B_1(v_0)$ and $v\in B_{\frac{3}{4}}(v_0)$, then
$\varepsilon(v'_j-v_j)\geq\delta$ for some $j$ and $\varepsilon$.
Comparison, with $\alpha=\frac{\delta}{2\Lambda h}$, gives
\[
  q(h,v)\leq Ce^{-\frac{c}{h}}.
    \refstepcounter{equation}\tag{\theequation}
    \label{e:short-boundary}
\]
At scale $R$ the right-hand side is $Ce^{-\frac{cR^2}{h}}$.  Since
$0\leq K^D\leq K$, the difference is positive with this mass;
Proposition~\ref{p:decay} and the triangle inequality give
\eqref{e:local-Dirichlet}.

We now include the data on the velocity boundary by splitting $\omega^D_{s,v}$ at time $s+h$, with $0<h\leq c_0R^2$. The part that reaches the velocity boundary before $s+h$ has mass at most $Ce^{-\frac{cR^2}{h}}$. The remaining part first evolves inside $D$ according to $K^D_{s,s+h}$. From time $s+h$ it continues to the terminal face or the velocity boundary according to a non-negative kernel of mass one. This last evolution cannot increase total variation, so~\eqref{e:local-Dirichlet} gives
\[
  \left\| \widehat\omega_{s,v}^{D, \xi} \right\|_{\mathrm{TV}}
  \leq C\left(e^{-\frac{cR^2}{h}}+e^{-ch\lvert\xi\rvert^{\frac{2}{3}}}\right).
\]
For $R^3\lvert\xi\rvert\geq1$ we choose $h=c_0R\lvert\xi\rvert^{-\frac{1}{3}}$; for the other
frequencies we use the mass bound.  This proves~\eqref{e:local-decay}.
<!-- end proof-1 -->
