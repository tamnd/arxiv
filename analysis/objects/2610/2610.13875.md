<!-- definition-1: body_md -->
For a fixed generating field $W=J_{\sigma+1}$ and deformation parameter $\gamma=\frac{\alpha_\sigma\tau^\sigma}{2\pi i}$, the \emph{$\sigma$-deformed KdV charges} are the formal series
\begin{equation}\label{eq: dkdv definition}
    \boldsymbol{\mathcal{I} }_s^{(\sigma)}:=\mathbf{I}_s+
    \sum_{n=1}^{\infty}\frac{\gamma^n}{(n-1)!\,2^{n-1}}
    \mathsf{Q}\left(M_{s,n}\right).
\end{equation}
Equivalently, they are the cylinder zero-modes of the deformed currents
\begin{equation}
    \mathsf J_s:=J_{s+1}+
    \sum_{n=1}^{\infty}\frac{\gamma^n}{(n-1)!\,2^{n-1}}M_{s,n}
    \pmod{\partial},
    \qquad
    \boldsymbol{\mathcal{I} }_s^{(\sigma)}=\mathsf{Q}\left(\mathsf J_s\right).
\end{equation}
We refer to $\boldsymbol{\mathcal{I} }_s^{(\sigma)}$ as the $\sigma$-deformed KdV charge of spin-$s$.
<!-- end definition-1 -->
<!-- proposition-1: body_md -->
\label{prop: relation M fields}
Let $M_{s,n}$ be defined by~\eqref{eq: Ms1} and~\eqref{eq: Msn recursion}, with generating field $W=J_{\sigma+1}$. Then, for every $n\geq1$,
\begin{equation}
    M_{1,n+1}=2nM_{\sigma,n}=[W^{n+1}]
    \pmod{\partial}.
\end{equation}
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
We prove the two identities by induction, using the skew-symmetry relation
\begin{equation}
    (AB)_1+(BA)_1=\partial(AB)_2
    \pmod{\partial^2}.
\end{equation}
First, we show that $M_{1,n}=[W^n]$ modulo total derivatives for all $n\geq1$. Since $J_2=T$, the initial condition gives
\begin{equation}
    M_{1,1}=\partial^{-1}(TW)_1
    =\partial^{-1}\partial W
    =W=[W]
    \pmod{\partial}.
\end{equation}
Suppose that $M_{1,k}=[W^k]$ modulo total derivatives for $1\leq k\leq n$. Substituting into~\eqref{eq: Msn recursion} yields
\begin{equation}
    M_{1,n+1}=\frac{1}{n+1}\partial^{-1}\Bigg\{
    \partial[W^{n+1}]
    +2n\sum_{k=1}^{n}\binom{n-1}{k-1}
    \bigl([W^k][W^{n+1-k}]\bigr)_1
    \Bigg\}
    \pmod{\partial}.
\end{equation}
The binomial coefficients are invariant under $k\mapsto n+1-k$. Symmetrising the sum and applying skew-symmetry therefore gives
\begin{equation}
\begin{aligned}
    &\sum_{k=1}^{n}\binom{n-1}{k-1}
    \bigl([W^k][W^{n+1-k}]\bigr)_1\\
    &\qquad=\frac12\partial\sum_{k=1}^{n}\binom{n-1}{k-1}
    \bigl([W^k][W^{n+1-k}]\bigr)_2\\
    &\qquad=\frac12\partial[W^{n+1}]
    \pmod{\partial^2},
\end{aligned}
\end{equation}
where the last equality follows from~\eqref{eq: W recursion}. Hence
\begin{equation}
    M_{1,n+1}
    =\frac{1}{n+1}\partial^{-1}
    \bigl\{\partial[W^{n+1}]+n\partial[W^{n+1}]\bigr\}
    =[W^{n+1}]
    \pmod{\partial}.
\end{equation}
Next, we show that $M_{\sigma,n}=[W^{n+1}]/(2n)$ modulo total derivatives\footnote{It looks as if when $\sigma =1$ this contradicts the previous statement. There is no contradiction. For $\sigma=1$, the generating field is $W=T$. Since $(TT)_2=2T$,
the recursion~\eqref{eq: W recursion} gives
\[
    [T^n]=2^{n-1}(n-1)!\,T \pmod{\partial}.
\]
Thus $[T^{n+1}]=2n[T^n]$ modulo total derivatives, and the
two identities in the proposition remain consistent when

It is useful to absorb the combinatorial factors into the fields. Define
\begin{equation}
    A_{s,0}:=J_{s+1},

    A_{s,n}:=\frac{M_{s,n}}{2^{n-1}(n-1)!}

\end{equation}

    (n+1)A_{s,n+1}
    =\partial^{-1}\sum_{i+j=n}(A_{s,i}K_j)_1
    \pmod{\partial},
    \qquad n\geq0,
\end{equation}

\begin{equation}\label{eq: W recursion now K recursion}
    (n+1)K_{n+1}
    =\frac12\sum_{i+j=n}(K_iK_j)_2
    \pmod{\partial}.
\end{equation}
Using skew-symmetry,
\begin{equation}
    (AB)_1+(BA)_1=\partial(AB)_2
    \pmod{\partial^2},
\end{equation}
and the symmetry of the sum under $i\leftrightarrow j$, we obtain
\begin{equation}
    \sum_{i+j=n}(K_iK_j)_1
    =\frac12\partial\sum_{i+j=n}(K_iK_j)_2
    \pmod{\partial^2}.
\end{equation}
Consequently,
\begin{equation}
    (n+1)K_{n+1}
    =\partial^{-1}\sum_{i+j=n}(K_iK_j)_1
    \pmod{\partial}.
\end{equation}
One could choose $A_{\sigma,n}=K_n$, consistently with Proposition~\ref{prop: relation M fields}. Define
\begin{equation}
    C_{s,r;N}:=\sum_{i+j=N}(A_{s,i}A_{r,j})_1.
\end{equation}

\begin{equation}
    C_{s,r;0}=(J_{s+1}J_{r+1})_1
    =0\pmod{\partial}
\end{equation}
by the assumed commutativity of the undeformed KdV currents. Now let $N\geq1$. Suppose that the fields have been constructed through order $N-1$ and that, for every pair $s,r$,
\begin{equation}\label{eq: sum of AA}
    C_{s,r;m}=\sum_{i+j=m}(A_{s,i}A_{r,j})_1
    =0\pmod{\partial},

\end{equation}

\begin{equation}
    \sum_{i+j=N-1}(A_{s,i}K_j)_1
    =0\pmod{\partial}.
\end{equation}
This proves existence by induction since the primitive of the left hand side is what defines the next $A_{s,N}$ in \eqref{eq: compacted recursion}, which can be done since we've shown that it must be a total derivative.

    \partial^{-1}\sum_{i+j=m}(A_{s,i}K_j)_1
    =(m+1)A_{s,m+1}\pmod{\partial},
    \qquad 0\leq m<N.
\end{equation}

\begin{equation}
    NC_{s,r;N}
    =\sum_{i+j=N}(i+j)(A_{s,i}A_{r,j})_1,
\end{equation}

\begin{equation}\label{eq: N CsrN}
    NC_{s,r;N}
    =\sum_{i+j=N-1}(j+1)
    \bigl\{(A_{s,i}A_{r,j+1})_1
    -(A_{r,i}A_{s,j+1})_1\bigr\}
    \pmod{\partial}.
\end{equation}
Differentiating~\eqref{eq: prim A K = m+1 A} yields
\begin{equation}
    (j+1)\partial A_{s,j+1}
    =\sum_{u+v=j}(A_{s,u}K_v)_1
    \pmod{\partial^2}.
\end{equation}
This and $(\partial A\,B)_1=0$ imply that differentiating \eqref{eq: N CsrN} yields
\begin{equation}
\begin{aligned}
    N\partial C_{s,r;N}
    &=\sum_{i+j=N-1}(j+1)
    \bigl\{(A_{s,i}\partial A_{r,j+1})_1
    -(A_{r,i}\partial A_{s,j+1})_1\bigr\}\\
    &=\sum_{i+u+v=N-1}
    \bigl\{(A_{s,i}(A_{r,u}K_v)_1)_1
    -(A_{r,u}(A_{s,i}K_v)_1)_1\bigr\}
    \pmod{\partial^2},
\end{aligned}
\end{equation}
where we have interchanged $i$ and $u$ in the second term of the last sum. Application of the Jacobi Identity brings us to
\begin{equation}
\begin{aligned}
    N\partial C_{s,r;N}
    &=\sum_{i+u+v=N-1}
    \bigl((A_{s,i}A_{r,u})_1K_v\bigr)_1\\
    &=\sum_{v=0}^{N-1}(C_{s,r;N-1-v}K_v)_1\\
    &=0\pmod{\partial^2}.
\end{aligned}
\end{equation}
The last equality follows from the induction hypothesis: each $C_{s,r;N-1-v}$ is a total derivative, whose first-order pole in the first OPE slot vanishes.

Thus $\partial C_{s,r;N}\in\partial^2V$. The field $C_{s,r;N}$ has positive conformal weight, so $\ker\partial=\mathbb{C}\mathbf{1}$ excludes an integration constant. It follows that $C_{s,r;N}\in\partial V$, completing the induction.

\label{lem: ao commutativity}
Let $A,B,K\in V$ be homogeneous fields of weights $h_A,h_B,h_K\geq2$. Suppose that their first-order poles are total derivatives,
\begin{equation}
    (AB)_1=\partial H,
    \qquad (AK)_1=\partial F,
    \qquad (BK)_1=\partial G,
\end{equation}
with homogeneous local primitives $H,F,G$. Then
\begin{equation}
    (FB)_1+(AG)_1\in\partial V.
\end{equation}
The same conclusion holds coefficient by coefficient for formal series, or for series truncated at any order, whose coefficient weights are respectively
\begin{equation}
    h_A+n(\sigma-1),
    \qquad h_B+n(\sigma-1),
    \qquad h_K+n(\sigma-1).
\end{equation}

\begin{aligned}

    &=(A(BK)_1)_1-(B(AK)_1)_1\\
    &=((AB)_1K)_1\\
    &=((\partial H)K)_1=0.
\end{aligned}
\end{equation}

\end{equation}

\end{equation}

\label{prop: ao hierarchy}
Recall that the undeformed KdV currents have first-order OPE poles in $\partial V$, and fix the full series $\mathsf K$ by the exact flow~\eqref{eq: ao K flow}. Then:
\begin{enumerate}
    \item All local primitives required by~\eqref{eq: Ms1} and~\eqref{eq: Msn recursion} exist.
    \item The currents $\mathsf J_s$ satisfying~\eqref{eq: djdgamma = }, with initial values $\mathsf J_s(0)=J_{s+1}$ and grading~\eqref{eq: ao coefficient grading}, are uniquely determined modulo total derivatives.
    \item Every pair of fields in $\{\mathsf K,\mathsf J_1,\mathsf J_3,\ldots\}$ commutes locally:
    \begin{equation}
        (AB)_1\in\partial V[\![\gamma]\!],
        \qquad A,B\in\{\mathsf K,\mathsf J_1,\mathsf J_3,\ldots\}.
    \end{equation}
\end{enumerate}
Here $\partial V[\![\gamma]\!]$ denotes the formal series whose coefficients are total derivatives.

<!-- proof-6: body_md -->
We prove existence and local commutativity simultaneously by induction on the power of $\gamma$, keeping $\mathsf K$ fixed throughout. We then prove uniqueness.

At order zero, $j_{s,0}=J_{s+1}$ and $k_0=W$ are undeformed KdV currents, so every pair commutes locally. Suppose that the currents $\mathsf J_s$ have been constructed through degree $N$, satisfy their flow equations through degree $N-1$, and commute locally with one another and with $\mathsf K$ through degree $N$. For $N=0$, the condition on the flow equations is empty. By the induction hypothesis, there are homogeneous local fields $U_{s,n}$ such that
\begin{equation}
    (\mathsf J_s\mathsf K)_1
    =\sum_{n=0}^{N}\partial U_{s,n}\gamma^n
    +O(\gamma^{N+1}).
\end{equation}
The coefficient of $\gamma^N$ depends only on the fields already constructed. We may therefore define
\begin{equation}\label{eq: ao next J}
    j_{s,N+1}=\frac{U_{s,N}}{N+1}\pmod{\partial}.
\end{equation}
This extends the flow equation through degree $N$. The primitive $U_{s,N}$ has weight $s+1+(N+1)\\(\sigma-1)>0$, so the prescribed grading is preserved. The corresponding coefficient of $\mathsf K$ is already fixed exactly
\begin{equation}
    k_{N+1}
    =\frac{1}{2(N+1)}\sum_{a=0}^{N}(k_a k_{N-a})_2.
\end{equation}
By~\eqref{eq: ao K primitive flow}, this coefficient also satisfies the primitive flow modulo total derivatives. Thus every member of the extended family satisfies its primitive flow through degree $N$.

To propagate commutativity, choose any two members $A,B$ of this family. Modulo $\gamma^{N+1}$, the induction hypothesis gives
\begin{equation}
    (A\mathsf K)_1=\partial F,
    \qquad (B\mathsf K)_1=\partial G,
    \qquad (AB)_1=\partial H,
\end{equation}
while the extended flow equations give
\begin{equation}
    \frac{\partial A}{\partial\gamma}= F,
    \qquad
    \frac{\partial B}{\partial\gamma}= G
    \pmod{\partial,\gamma^{N+1}}.
\end{equation}
Derivative corrections do not affect the calculation modulo $\partial$, since
\begin{equation}
    ((\partial X)B)_1=0,
    \qquad (A(\partial Y))_1=\partial(AY)_1.
\end{equation}
Applying Lemma~\ref{lem: ao commutativity} to these truncated series therefore yields
\begin{equation}
\begin{aligned}
    \frac{\partial}{\partial\gamma}(AB)_1
    &=\left(\frac{\partial A}{\partial\gamma}B\right)_1
    +\left(A\frac{\partial B}{\partial\gamma}\right)_1\\
    &=(FB)_1+(AG)_1\\
    &=0\pmod{\partial,\gamma^{N+1}}.
\end{aligned}
\end{equation}
Writing
\begin{equation}
    (AB)_1=\sum_{n\geq0}C_n\gamma^n
\end{equation}
and comparing the coefficient of $\gamma^N$ gives
\begin{equation}
    (N+1)C_{N+1}=0\pmod{\partial}.
\end{equation}
Hence $C_{N+1}\in\partial V$, and every pair commutes locally through degree $N+1$. This closes the induction. Proposition~\ref{prop: ao flows} then also establishes the existence of every local primitive in the original recursion.

For uniqueness, let $\mathsf J_s$ and $\widetilde{\mathsf J}_s$ be two solutions for the same fixed $\mathsf K$, with the same initial value and coefficient grading. Suppose inductively that
\begin{equation}
    j_{s,n}-\widetilde j_{s,n}\in\partial V,
    \qquad 0\leq n\leq N.
\end{equation}
This holds at $N=0$. A derivative in the first OPE slot has no first-order pole, so the $\gamma^N$ term in $((\mathsf J_s-\widetilde{\mathsf J}_s)\mathsf K)_1$ vanishes. The degree-$N$ inputs to the two primitive flows therefore agree exactly. Their homogeneous primitives also agree: their difference lies in $\ker\partial$, while their weight $s+1+(N+1)(\sigma-1)$ is positive. Comparing the flow equations at degree $N$ now gives
\begin{equation}
    (N+1)\bigl(j_{s,N+1}-\widetilde j_{s,N+1}\bigr)
    =0\pmod{\partial}.
\end{equation}
Induction proves uniqueness modulo total derivatives. This uniqueness statement holds for the prescribed full series $\mathsf K$; it does not permit arbitrary changes to its derivative terms.
