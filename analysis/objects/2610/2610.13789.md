<!-- assumption-1: body_md -->
\label{ass:objective}
$f_*>-\infty$ and $f$ is $L$-smooth. For every $t\ge0$, $\mathbb E_t g_t=\nabla f(x_t)$ and $\mathbb E_t\lVert g_t-\nabla f(x_t)\rVert^2\le\sigma^2$.
<!-- end assumption-1 -->
<!-- assumption-2: body_md -->
\label{ass:transport}
Let $\mu_t=\mathbb E_t C_t$. For constants $b,v\ge0$ and every $t\ge1$,
\begin{align}
\lVert \mu_t-C^\star_{t}\rVert&\le b\lVert s_t\rVert^2,\label{eq:bias-assumption}\\
\mathbb E_t\lVert C_t-\mu_t\rVert^2&\le v^2\lVert s_t\rVert^2.\label{eq:variance-assumption}
\end{align}
The gradient $g_t$ and the correction $C_t$ may share oracle randomness.
<!-- end assumption-2 -->
<!-- assumption-3: body_md -->
\label{ass:query}
The probability $q\in(0,1]$ is fixed. For every $t\ge1$, $J_t\mid\mathcal F_{t-1}\sim\operatorname{Bernoulli}(q)$, and $J_t$ is conditionally independent of $(C_t,B_t,g_t)$ given $\mathcal F_{t-1}$.
<!-- end assumption-3 -->
<!-- lemma-1: body_md -->
\label{lem:identities}
Under Assumptions~\ref{ass:transport} and~\ref{ass:query}, for every $t\ge1$,
\begin{align}
\mathbb E_t\widehat C_t&=\mu_t,\label{eq:mean-id}\\
\mathbb E_t\lVert \widehat C_t-\mu_t\rVert^2
&=\mathbb E_t\lVert C_t-\mu_t\rVert^2\notag\\
&\quad +(q^{-1}-1)\mathbb E_t\lVert C_t-B_t\rVert^2,\label{eq:variance-id}\\
\mathbb E_t\langle \zeta_t,\widehat C_t-\mu_t\rangle
&=\mathbb E_t\langle \zeta_t,C_t-\mu_t\rangle.\label{eq:covariance-id}
\end{align}
The same second-moment increment holds for centering at any $\mathcal F_{t-1}$-measurable point, in particular at $C^\star_{t}$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Write $\widehat C_t=C_t+(J_t/q-1)(C_t-B_t)$. Conditional on $(\mathcal F_{t-1},C_t,B_t,g_t)$, the added term has zero mean and second moment $(q^{-1}-1)\lVert C_t-B_t\rVert^2$. Expanding gives the three identities.
<!-- end proof-1 -->
<!-- theorem-1: body_md -->
\label{thm:lmo}
Under Assumptions~\ref{ass:objective}--\ref{ass:query}, Algorithm~1 with $\eta>0$ and schedule~\eqref{eq:alpha-schedule} satisfies
\begin{align}
\mathbb ES_{\mathcal K}(\nabla f(x_U))\le{}&\frac{\Delta}{\eta T}+\frac{L\eta R^2}{2}
+\frac{4R\sigma}{T}+\frac{4\sqrt2R\sigma}{\sqrt T}\notag\\
&+\frac{2\overline{\beta} b\eta^2R^3}{\underline{\alpha}}+2R\sqrt{W_T}.\label{eq:lmo-bound}
\end{align}
<!-- end theorem-1 -->
<!-- head-1: body_md -->
With $e_t=m_t-\nabla f(x_t)$ and $\beta_t=1-\alpha_t$,
\begin{equation}\label{eq:error-recursion}
e_t=\beta_t e_{t-1}+\beta_t r_t+w_t,
\quad r_t=\mu_t-C^\star_{t},
\end{equation}
where $w_t=\beta_t(\widehat C_t-\mu_t)+\alpha_t\zeta_t$ is a martingale difference. Unrolling gives a weighted initial error, a drift sum, and a martingale sum. The schedule yields
$\prod_{i=1}^t\beta_i\le2/[(t+1)(t+2)]$, so the cumulative initial-error weight is at most two. The remaining terms follow from weighted second-moment bounds and the alignment inequality
\begin{equation}\label{eq:lmo-alignment}
\langle \nabla f(x_t),u_t\rangle
\geS_{\mathcal K}(\nabla f(x_t))-2R\lVert e_t\rVert.
\end{equation}
Appendix~\ref{app:lmo} gives the proof and the corresponding bound for arbitrary deterministic $\alpha_t$.

\begin{corollary}[Complete stationarity bound]\label{cor:rate}
Let $c_\eta>0$ and $0<c_\alpha\le1$, and suppose $\mathcal L_T\le\bar\kappa^2\eta^2R^2$ for some $\bar\kappa\ge0$. Set $\eta=c_\eta T^{-2/3}$ and $\underline{\alpha}=c_\alpha T^{-2/3}$ in~\eqref{eq:alpha-schedule}. For $\kappa\ge0$ define
\begin{align}
Z_q(\kappa)&=c_\alpha\sigma^2+\sigma vc_\eta R\notag\\
&\quad+\frac{c_\eta^2R^2}{c_\alpha}\left[v^2+(q^{-1}-1)\kappa^2\right],\label{eq:Zq}\\
\mathsf A_q(\kappa)&=\Delta/c_\eta+2R\sqrt{Z_q(\kappa)},\notag\\
\mathsf U&=4\sqrt2R\sigma,\qquad \mathsf I=4R\sigma,\notag\\
\mathsf D&=Lc_\eta R^2/2+2bc_\eta^2R^3/c_\alpha.\label{eq:rate-constants}
\end{align}
Then, under the assumptions of Theorem~\ref{thm:lmo}, for every $T\ge1$,
\begin{align}
\mathbb ES_{\mathcal K}(\nabla f(x_U))\le{}&\mathsf A_q(\bar\kappa)T^{-1/3}
+\mathsf U T^{-1/2}\notag\\
&+\mathsf D T^{-2/3}+\mathsf I T^{-1}.\label{eq:rate}
\end{align}
\end{corollary}
The four terms collect the descent and tracking error, the noise of the decreasing phase, smoothness and Taylor drift, and the initial momentum error. For fixed $q>0$ and a uniform $\bar\kappa$, the bound is $O(T^{-1/3})$.

For $B_t=0$ and $\lVert \mu_t\rVert\le M\lVert s_t\rVert$, Assumption~\ref{ass:transport} gives $\mathbb E_t\lVert C_t\rVert^2\le(v^2+M^2)\lVert s_t\rVert^2$, so Corollary~\ref{cor:rate} applies with $\bar\kappa^2=v^2+M^2$. Both transport oracles satisfy this with $M=L$. In general, prediction accuracy enters only through $v^2+(q^{-1}-1)\bar\kappa^2$.

<!-- proposition-1: body_md -->
\label{prop:ratio}
Let $\omega=L^2/\delta^2$. The ratio of the leading coefficients in~\eqref{eq:leading-budget} at $q$ and at $1$ is
\begin{equation}\label{eq:ratio}
\mathcal Q(q;\omega)=\frac{1+q}{4}
\left(1+\sqrt{\frac{1+\omega}{q}-\omega}\right).
\end{equation}
There exists $q\in(0,1)$ with $\mathcal Q(q;\omega)<1$ if and only if $\omega<1$.

<!-- theorem-3: body_md -->
\label{thm:regret}
Under Assumptions~\ref{ass:transport} and~\ref{ass:query}, the LMO update~\eqref{eq:lmo-update}, and $\lVert \mu_t\rVert\le M\lVert s_t\rVert$, let $n=T-1\ge1$. For every fixed $h\in[-h_{\max},h_{\max}]^d$,
\begin{align}
\sum_{t=1}^n\mathbb E[\ell_t(h_t)-\ell_t(h)]
\le \frac{dh_{\max}^2}{2\eta_h}
+\frac{4\eta_h n(h_{\max}^2+V_C^2)}q.\label{eq:regret}
\end{align}
The choice $\eta_h=h_{\max}\sqrt{dq/[8n(h_{\max}^2+V_C^2)]}$ makes the right side equal to
\begin{equation}\label{eq:regret-value}
\mathrm{Reg}_n=h_{\max}\sqrt{8dn(h_{\max}^2+V_C^2)/q}.
\end{equation}
Consequently,
\begin{align}
\mathcal L_T\le\frac{\eta^2R^2}{T}
\left[\inf_{h\in[-h_{\max},h_{\max}]^d}\sum_{t=1}^n\mathbb E\ell_t(h)+\mathrm{Reg}_n\right].\label{eq:online-conversion}
\end{align}

\label{cor:learned-rate}

<!-- lemma-2: body_md -->
\label{lem:tracking}
Suppose
\[
e_t=\beta_t e_{t-1}+\beta_t r_t+w_t,
\qquad \mathbb E_t w_t=0,
\qquad \mathbb E\lVert e_0\rVert^2\le\sigma^2,
\]
where $r_t$ is $\mathcal F_{t-1}$-measurable and $w_t$ is $\mathcal F_t$-measurable. Let $\bar r_i=\mathbb E\lVert r_i\rVert$, and let $\bar w_i\ge\mathbb E\lVert w_i\rVert^2$ be deterministic. Then
\begin{equation}\label{eq:general-tracking}
\frac1T\sum_{t=0}^{T-1}\mathbb E\lVert e_t\rVert
\le\frac1T\sum_{t=0}^{T-1}\left[
\sigma p_{0,t}+\sum_{i=1}^t p_{i,t}\beta_i\bar r_i
+\sqrt{\sum_{i=1}^t p_{i,t}^2\bar w_i}\right].
\end{equation}
If $e_t=m_t-\nabla f(x_t)$, Assumption~\ref{ass:objective} and the LMO update~\eqref{eq:lmo-update} give a stationarity bound obtained by multiplying the right side by $2R$ and adding $\Delta/(\eta T)+L\eta R^2/2$.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Unrolling the recursion gives
\[
e_t=p_{0,t}e_0+\mathcal D_t+\mathcal M_t,
\qquad \mathcal D_t=\sum_{i=1}^t p_{i,t}\beta_i r_i,
\qquad \mathcal M_t=\sum_{i=1}^t p_{i,t}w_i.
\]
All weights are deterministic. For $i<j$, $w_i$ is $\mathcal F_{j-1}$-measurable, so the tower property gives $\mathbb E\langle w_i,w_j\rangle=0$ and
\[
\mathbb E\lVert \mathcal M_t\rVert^2=\sum_{i=1}^t p_{i,t}^2\mathbb E\lVert w_i\rVert^2
\le\sum_{i=1}^t p_{i,t}^2\bar w_i.
\]
The triangle inequality and Cauchy--Schwarz give~\eqref{eq:general-tracking}; no independence between $e_0$, $\mathcal D_t$, and $\mathcal M_t$ is needed.

For the exact LMO direction $u_t$ and a maximizer $u_t^*$ of $\langle \nabla f(x_t),u\rangle$ over $\mathcal K$,
\[
\langle \nabla f(x_t),u_t\rangle
=\langle m_t,u_t\rangle-\langle e_t,u_t\rangle
\ge\langle m_t,u_t^*\rangle-\langle e_t,u_t\rangle
\geS_{\mathcal K}(\nabla f(x_t))-2R\lVert e_t\rVert.
\]
Smoothness and $\lVert u_t\rVert\le R$ yield
\[
f(x_{t+1})\le f(x_t)-\etaS_{\mathcal K}(\nabla f(x_t))
+2\eta R\lVert e_t\rVert+L\eta^2R^2/2.
\]
Summing over $t=0,\ldots,T-1$, taking expectations, and using $f(x_T)\ge f_*$ proves the last assertion.

<!-- proposition-2: body_md -->
\label{prop:mars-scaling}
Under Assumptions~\ref{ass:objective}--\ref{ass:query}, update~\eqref{eq:mars-scaled} with schedule~\eqref{eq:alpha-schedule} and constant $\gamma\ge0$ satisfies
\begin{equation}\label{eq:mars-bound}
\mathbb ES_{\mathcal K}(\nabla f(x_U))
\le \Theta_T+\frac{2\overline{\beta}\gamma b\eta^2R^3}{\underline{\alpha}}
+\frac{2\overline{\beta}|1-\gamma|L\eta R^2}{\underline{\alpha}}
+2R\sqrt{W_T^\gamma}.
\end{equation}
<!-- end proposition-2 -->
<!-- proof-3: body_md -->
With $e_t=m_t-\nabla f(x_t)$,
\[
e_t=\beta_t e_{t-1}+\beta_t r_t^\gamma+w_t^\gamma,
\quad r_t^\gamma=\gamma\mu_t-C^\star_{t},
\quad w_t^\gamma=\beta_t\gamma(\widehat C_t-\mu_t)+\alpha_t\zeta_t.
\]
The predictable drift satisfies
\[
\lVert r_t^\gamma\rVert
\le\gamma\lVert \mu_t-C^\star_{t}\rVert+|1-\gamma|\lVert C^\star_{t}\rVert
\le\gamma b\eta^2R^2+|1-\gamma|L\eta R.
\]
Lemma~\ref{lem:identities} gives $\mathbb E_t w_t^\gamma=0$ and
\[
\mathbb E\lVert w_t^\gamma\rVert^2
\le\alpha_t^2\sigma^2+2\alpha_t\beta_t\gamma\sigma v\eta R
+\beta_t^2\gamma^2\left[v^2\eta^2R^2+(q^{-1}-1)\mathbb E\lVert C_t-B_t\rVert^2\right].
\]
For arbitrary deterministic $\alpha_t$, these estimates can be substituted into Lemma~\ref{lem:tracking}. For schedule~\eqref{eq:alpha-schedule},~\eqref{eq:tail-weights} bounds the average drift contribution by
$\overline{\beta}(\gamma b\eta^2R^2+|1-\gamma|L\eta R)/\underline{\alpha}$. Equations~\eqref{eq:initial-weights}--\eqref{eq:pointwise-correction-weights} applied to the noise bound repeat the calculation of Theorem~\ref{thm:lmo} with $v$ replaced by $\gamma v$ and the residual second moment multiplied by $\gamma^2$, so the initial-error and martingale contributions are at most $2\sigma/T$ and $\sqrt{W_T^\gamma}+2\sqrt2\sigma/\sqrt T$. The descent statement of Lemma~\ref{lem:tracking} gives~\eqref{eq:mars-bound}.

<!-- corollary-3: body_md -->
\label{cor:mars-rates}
Under Proposition~\ref{prop:mars-scaling}, suppose $\mathcal L_T\le\bar\kappa^2\eta^2R^2$ with $\bar\kappa\ge0$ independent of $T$. Fix $q\in(0,1]$, $c_\eta>0$, and $0<c_\alpha\le1$, and write
$\Omega_q(\bar\kappa)=v^2+(q^{-1}-1)\bar\kappa^2$. The following bounds hold for every integer $T\ge1$ under the stated conditions.
\begin{enumerate}
\item Suppose $0\le\gamma_T\le\bar\gamma$ and $|1-\gamma_T|\le c_\gamma T^{-1/3}$ for constants $\bar\gamma,c_\gamma\ge0$. With $\eta=c_\eta T^{-2/3}$ and $\underline{\alpha}=c_\alpha T^{-2/3}$,
\begin{align}
\mathbb ES_{\mathcal K}(\nabla f(x_U))\le{}&
\left[\frac\Deltac_\eta+2R\sqrt{c_\alpha\sigma^2+\bar\gamma\sigma vc_\eta R+
\frac{\bar\gamma^2c_\eta^2R^2}{c_\alpha}\Omega_q(\bar\kappa)}
+\frac{2c_\gamma Lc_\eta R^2}{c_\alpha}\right]T^{-1/3}\notag\\
&+4\sqrt2R\sigma T^{-1/2}
+\left(\frac{Lc_\eta R^2}{2}+\frac{2\bar\gamma bc_\eta^2R^3}{c_\alpha}\right)T^{-2/3}
+4R\sigma T^{-1}.\label{eq:standard-explicit-fast}
\end{align}
\item For any fixed $\gamma\ge0$, with $\eta=c_\eta T^{-3/4}$ and $\underline{\alpha}=c_\alpha T^{-1/2}$,
\begin{align}
\mathbb ES_{\mathcal K}(\nabla f(x_U))\le{}&
\left[\frac\Deltac_\eta+2R\sigma\sqrtc_\alpha+
\frac{2|1-\gamma|Lc_\eta R^2}{c_\alpha}\right]T^{-1/4}
+2R\sqrt{\gamma\sigma vc_\eta R}\,T^{-3/8}\notag\\
&+\left[4\sqrt2R\sigma+2\gammac_\eta R^2\sqrt{\frac{\Omega_q(\bar\kappa)}c_\alpha}\right]T^{-1/2}
+\frac{Lc_\eta R^2}{2}T^{-3/4}
+\left[4R\sigma+\frac{2\gamma bc_\eta^2R^3}{c_\alpha}\right]T^{-1}.\label{eq:standard-explicit-slow}
\end{align}
\end{enumerate}
<!-- end corollary-3 -->
<!-- proof-4: body_md -->
Using $\overline{\beta}\le1$ and $\underline{\alpha}(2-\underline{\alpha})\ge\underline{\alpha}$ in~\eqref{eq:scaling-noise} gives
\[
W_T^\gamma\le\sigma^2\underline{\alpha}+\gamma\sigma v\eta R
+\frac{\gamma^2\eta^2R^2}{\underline{\alpha}}\Omega_q(\bar\kappa).
\]
For the first tuning,
\[
2R\sqrt{W_T^{\gamma_T}}
\le2R\sqrt{c_\alpha\sigma^2+\bar\gamma\sigma vc_\eta R+
\frac{\bar\gamma^2c_\eta^2R^2}{c_\alpha}\Omega_q(\bar\kappa)}\,T^{-1/3},
\]
and the two drift terms in~\eqref{eq:mars-bound} are at most
$2\bar\gamma bc_\eta^2R^3T^{-2/3}/c_\alpha$ and $2c_\gamma Lc_\eta R^2T^{-1/3}/c_\alpha$.
For the second tuning, splitting the square root into its three nonnegative components gives
\[
2R\sqrt{W_T^\gamma}
\le2R\sigma\sqrtc_\alpha\,T^{-1/4}
+2R\sqrt{\gamma\sigma vc_\eta R}\,T^{-3/8}
+2\gammac_\eta R^2\sqrt{\frac{\Omega_q(\bar\kappa)}c_\alpha}\,T^{-1/2},
\]
and the drift terms are at most $2\gamma bc_\eta^2R^3T^{-1}/c_\alpha$ and
$2|1-\gamma|Lc_\eta R^2T^{-1/4}/c_\alpha$.
Adding the four terms of $\Theta_T$ proves both bounds.

<!-- proposition-3: body_md -->
\label{prop:residual-scaling}
Under Assumptions~\ref{ass:objective}--\ref{ass:query} with a predictable predictor, define
\[
\mathcal P_T=\frac1T\sum_{t=1}^{T-1}\mathbb E\lVert \mu_t-B_t\rVert.
\]
For~\eqref{eq:residual-scaled} with schedule~\eqref{eq:alpha-schedule} and constant $\gamma\ge0$,
\begin{equation}\label{eq:residual-scaled-bound}
\mathbb ES_{\mathcal K}(\nabla f(x_U))
\le \Theta_T+\frac{2\overline{\beta} b\eta^2R^3}{\underline{\alpha}}
+\frac{2\overline{\beta} R|1-\gamma|}{\underline{\alpha}}\mathcal P_T
+2R\sqrt{W_T^\gamma}.
\end{equation}
<!-- end proposition-3 -->
<!-- proof-5: body_md -->
The tracking recursion has drift~\eqref{eq:residual-scaled-drift} and the same centered noise $w_t^\gamma$ as in Proposition~\ref{prop:mars-scaling}, so the initial and martingale terms have the same bounds. For $\mathcal D_t=\sum_{i=1}^t p_{i,t}\beta_i(\widetilde\mu_i-C^\star_{i})$, exchanging the finite sums and using~\eqref{eq:tail-weights},
\begin{align*}
\frac1T\sum_{t=0}^{T-1}\mathbb E\lVert \mathcal D_t\rVert
\le\frac{\overline{\beta}}{\underline{\alpha} T}\sum_{i=1}^{T-1}\mathbb E\lVert \widetilde\mu_i-C^\star_{i}\rVert
\le\frac{\overline{\beta}}{\underline{\alpha}}\left[b\eta^2R^2+|1-\gamma|\mathcal P_T\right].
\end{align*}
The descent statement of Lemma~\ref{lem:tracking} concludes. For arbitrary deterministic $\alpha_t$, the same lemma applies before the tail-weight simplification.

<!-- corollary-4: body_md -->
\label{cor:residual-scaling-rate}
Under Proposition~\ref{prop:residual-scaling}, suppose $\mathcal L_T\le\bar\kappa^2\eta^2R^2$ and $\mathcal P_T\le b_B\eta^2R^2$ with $\bar\kappa,b_B\ge0$ independent of $T$. Fix $\gamma\ge0$, $q\in(0,1]$, $c_\eta>0$, and $0<c_\alpha\le1$. For every integer $T\ge1$, schedule~\eqref{eq:alpha-schedule} with $\eta=c_\eta T^{-2/3}$ and $\underline{\alpha}=c_\alpha T^{-2/3}$ gives
\begin{align}
\mathbb ES_{\mathcal K}(\nabla f(x_U))\le{}&
\left[\frac\Deltac_\eta+2R\sqrt{c_\alpha\sigma^2+\gamma\sigma vc_\eta R+
\frac{\gamma^2c_\eta^2R^2}{c_\alpha}\Omega_q(\bar\kappa)}\right]T^{-1/3}\notag\\
&+4\sqrt2R\sigma T^{-1/2}
+\left[\frac{Lc_\eta R^2}{2}+\frac{2(b+|1-\gamma|b_B)c_\eta^2R^3}{c_\alpha}\right]T^{-2/3}
+4R\sigma T^{-1}.\label{eq:residual-explicit-fast}
\end{align}
A sufficient mean-prediction condition is $\lVert \mu_t-B_t\rVert\le b_B\lVert s_t\rVert^2$ almost surely at every correction index.
<!-- end corollary-4 -->
<!-- proof-6: body_md -->
The total drift contribution in~\eqref{eq:residual-scaled-bound} satisfies
\[
\frac{2\overline{\beta} b\eta^2R^3}{\underline{\alpha}}
+\frac{2\overline{\beta} R|1-\gamma|}{\underline{\alpha}}\mathcal P_T
\le\frac{2(b+|1-\gamma|b_B)c_\eta^2R^3}{c_\alpha}T^{-2/3},
\]
and
\[
2R\sqrt{W_T^\gamma}
\le2R\sqrt{c_\alpha\sigma^2+\gamma\sigma vc_\eta R+
\frac{\gamma^2c_\eta^2R^2}{c_\alpha}\Omega_q(\bar\kappa)}\,T^{-1/3}.
\]
Adding the four terms of $\Theta_T$ proves~\eqref{eq:residual-explicit-fast}. The sufficient condition gives
$\mathcal P_T\le b_B\eta^2R^2(T-1)/T\le b_B\eta^2R^2$, since $\lVert s_t\rVert\le\eta R$.
