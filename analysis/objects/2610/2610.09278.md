<!-- assumption-1: body_md -->
\label{ass:filtering}
The functions
\[
 b:\mathbb R^d\times U\to\mathbb R^d,
 \qquad
 \sigma:\mathbb R^d\times U\to\mathbb R^{d\times r},
 \qquad
 h:\mathbb R^d\times U\to\mathbb R^m
\]
are Borel measurable.  There is a constant $K\ge1$ such that, for all $x,y\in\mathbb R^d$ and $v\in U$,
\begin{align}
 &\left\lvert b(x,v)-b(y,v)\right\rvert+\left\lVert \sigma(x,v)-\sigma(y,v)\right\rVert
   \le K\left\lvert x-y\right\rvert,
   \label{eq:filtering-lipschitz}\\
 &\left\lvert b(x,v)\right\rvert+\left\lVert \sigma(x,v)\right\rVert+\left\lvert h(x,v)\right\rvert
   \le K(1+\left\lvert x\right\rvert+\left\lvert v\right\rvert).
   \label{eq:filtering-growth}
\end{align}

<!-- definition-1: body_md -->
\label{def:bspde-solution}
Let $\theta\le\tau_k$ be an $\mathbb F^Y$-stopping time and
let $\gamma\in\mathscr G_\theta^{n_*+1}$.
We call a pair $(f,g)$ a weighted Sobolev solution
of the BSPDE \eqref{eq:intro-dual-bspde} if, for some $\ell>0$,
the field $f$ is $\mathbb F^Y$-adapted, $g$ is
$\mathbb F^Y$-predictable, $f\in L^\infty(\Omega\times[0,\theta]\times\mathbb R^d)$ and
\begin{align}
\mathbb E^\mathbb Q\left[
	\operatorname*{ess\,sup}_{0\le s\le\theta}\left\lVert f_s\right\rVert_{H_\ell^{n_*}}^2
	+\int_0^\theta
	\left\lVert g_s\right\rVert_{H_\ell^{n_*}(\mathbb R^d;\mathbb R^m)}^2\,\,\mathrm d s
	\right]<\infty.
	\label{eq:solution-norm}
\end{align}
Moreover, for every $\varphi\in C_c^\infty(\mathbb R^d)$
and $t\in[0,T]$, the identity
\begin{align}
	\left\langle f_{t\wedge\theta},\varphi\right\rangle
	={}&\left\langle \gamma,\varphi\right\rangle
	+\int_{t\wedge\theta}^{\theta}
	\left\langle f_s,L_s^*\varphi\right\rangle\,\,\mathrm d s
	\notag\\
	&+\int_{t\wedge\theta}^{\theta}
	\left\langle h(\cdot,u_s)^\top g_s,\varphi\right\rangle\,\,\mathrm d s
	-\int_{t\wedge\theta}^{\theta}
	\left\langle g_s,\varphi\right\rangle^\top\,\,\mathrm d Y_s
	\label{eq:weak-bspde}
\end{align}
holds $\mathbb Q$-a.s..

<!-- theorem-1: body_md -->
\label{thm:main-bspde}
	Suppose Assumptions \ref{ass:filtering} and
	\ref{ass:bspde-regularity} hold.
	Fix $k\in\mathbb N$, an $\mathbb F^Y$-stopping time
	$\theta\le\tau_k$, and $\gamma\in\mathscr G_\theta^{n_*+1}$.
	Then \eqref{eq:intro-dual-bspde} admits a unique
	weighted Sobolev solution $(f,g)$ with bounded first
	component, satisfying \eqref{eq:solution-norm}
	with $\ell=d/2+2$.
	Moreover, $f$ admits a jointly measurable version that is
	$\mathbb Q$-almost surely continuous in $(t,x)$ on
	$[0,\theta]\times\mathbb R^d$.
	For every $\lambda\ge d/2+2$ and every multi-index
	$\alpha$ with $|\alpha|\le2$, the solution satisfies
	\begin{align}
		\mathbb E^\mathbb Q\Bigg[
		&\operatorname*{ess\,sup}_{0\le s\le\theta}\sup_{x\in\mathbb R^d}
		\frac{|D^\alpha f_s(x)|^2}{\left\langle x\right\rangle^{2\lambda}}
		+\int_0^\theta\sup_{x\in\mathbb R^d}
		\frac{|D^\alpha g_s(x)|^2}{\left\langle x\right\rangle^{2\lambda}}\,\mathrm d s
		\Bigg]<\infty.
		\label{eq:weighted-sup-2}
	\end{align}

<!-- remark-1: body_md -->
The derivatives $D^\alpha f$ and $D^\alpha g$ in
	\eqref{eq:weighted-sup-2} are understood as weak spatial
	derivatives. Since $n_*>d/2+2$, local Sobolev embedding gives
	$C^2$  representatives of $f_s$ and $g_s$.
	Their weak derivatives up to order two coincide with the
	classical derivatives of these representatives. Therefore, 
	the spatial suprema in \eqref{eq:weighted-sup-2} are taken
	over the resulting continuous representatives of the derivatives.

<!-- lemma-2: body_md -->
\label{lem:flow-estimates}
	Under Assumptions \ref{ass:filtering} and \ref{ass:bspde-regularity}.
	Let $\widetilde{\mathbb Q}$ be any probability measure equivalent to $\mathbb Q$
	under which $W$ remains an $\mathbb F$-Brownian motion.
	For every $2\le q<\infty$ and $j=1,\ldots,n_*$,
	\begin{align}
		\mathbb E^{\widetilde{\mathbb Q}}
		\sup_{t\le s\le T}|\Phi_{t,s}(x)|^q
		&\le C_{k,q}\left\langle x\right\rangle^q,
		\label{eq:flow-state-moment}\\
		\sup_{0\le t\le T}\sup_{x\in\mathbb R^d}
		\mathbb E^{\widetilde{\mathbb Q}}
		\sup_{t\le s\le T}|D_x^j\Phi_{t,s}(x)|^q
		&\le C_{j,q}.
		\label{eq:flow-derivative-moment}
	\end{align}
	Moreover,
	\begin{align}
		\sup_{0\le t\le T}\sup_{x\in\mathbb R^d}
		\mathbb E^{\widetilde{\mathbb Q}}\sup_{t\le s\le T}
		\Bigg[
		&J_{t,s}(x)^q+J_{t,s}(x)^{-q}
		+\sum_{j=1}^{n_*}
		\big|(D^j\Psi_{t,s})(\Phi_{t,s}(x))\big|^q
		\Bigg]
		\le C_{k,q}.
		\label{eq:maximal-inverse-flow-moments}
	\end{align}

<!-- proof-1: body_md -->
First, it is easy to verify:
	\begin{align*}
		&\mathbb E^{\widetilde{\mathbb Q}}
		\sup_{t\le a\le s}|\Phi_{t,a}(x)|^q\\
		&\quad\le C_q\bigl(1+|x|^q+k^{q/2}\bigr)
		+C_q\int_t^s
		\mathbb E^{\widetilde{\mathbb Q}}
		\sup_{t\le r\le a}|\Phi_{t,r}(x)|^q\,\mathrm d a.
	\end{align*}
Then,	Gronwall's inequality yields
	\[
	\mathbb E^{\widetilde{\mathbb Q}}
	\sup_{t\le s\le T}|\Phi_{t,s}(x)|^q
	\le C_q\bigl(1+|x|^q+k^{q/2}\bigr)
	\le C_{k,q}\left\langle x\right\rangle^q.
	\]
This	proves \eqref{eq:flow-state-moment}.
	
Denote
	\[
	B_s:=\chi_s^tD_xb(\Phi_{t,s}(x),u_s),
	\quad
	C_s^\nu:=\chi_s^tD_x\sigma^\nu(\Phi_{t,s}(x),u_s),
	\]
where $\sigma^\nu$  denotes the $\nu$th column of the diffusion matrix  $\sigma$. 
Then, we have, 
	\[
	|B_s|+\sum_{\nu=1}^r|C_s^\nu|\le C.
	\]
	For $1\le j\le n_*$,  we have 
\begin{align*}
	\,\mathrm d D_x^j\Phi_{t,s}(x)
	={}&\bigl[B_sD_x^j\Phi_{t,s}(x)+R_s^{(j)}\bigr]\,\mathrm d s\\
	&+\sum_{\nu=1}^r
	\bigl[C_s^\nu D_x^j\Phi_{t,s}(x)
	+S_s^{(j),\nu}\bigr]\,\mathrm d W_s^\nu,
\end{align*}
with
\[
D_x\Phi_{t,t}(x)=\mathrm{Id},
\qquad
D_x^j\Phi_{t,t}(x)=0\quad(j\ge2).
\]
Here, the remainders $R_s^{(j)}$ and $S_s^{(j),\nu}$ collect
the chain-rule terms involving only spatial derivatives of
the flow of orders below $j$.
Note that $R^{(1)}_s=S^{(1),\nu}_s=0$, and for $j\ge2$,
	\begin{equation}
		|R_s^{(j)}|+|S_s^{(j)}|
		\le C_j\sum_{\ell=2}^j
		\ \sum_{\substack{i_1+\cdots+i_\ell=j\\i_a\ge1}}
		\prod_{a=1}^{\ell}|D_x^{i_a}\Phi_{t,s}(x)|,
		\label{eq:flow-derivative-forcing}
	\end{equation}
	
For $j= 1$, \eqref{eq:flow-derivative-moment} is clearly true, we proceed by induction on $j$. 	Suppose the estimate holds for every order below $j$ and every
	finite exponent $q\ge2$. For each product in
	\eqref{eq:flow-derivative-forcing}, all $i_a<j$,  H\"older's
	inequality gives
	\begin{align*}
		&\mathbb E^{\widetilde{\mathbb Q}}\sup_{t\le s\le T}
		\prod_{a=1}^{\ell}|D_x^{i_a}\Phi_{t,s}(x)|^q\\
		&\quad\le
		\prod_{a=1}^{\ell}
		\left(
		\mathbb E^{\widetilde{\mathbb Q}}\sup_{t\le s\le T}
		|D_x^{i_a}\Phi_{t,s}(x)|^{q\ell}
		\right)^{1/\ell}
		\le C_{j,q}.
	\end{align*}
	Consequently,
	\[
	\mathbb E^{\widetilde{\mathbb Q}}\int_t^T
	\bigl(|R_s^{(j)}|^q+|S_s^{(j)}|^q\bigr)\,\mathrm d s
	\le C_{j,q}.
	\]
Then, BDG and Gronwall inequalities yield \eqref{eq:flow-derivative-moment}, which
completes the induction.
	
Now, we set
	\[
	\mathcal J_s:=D_x\Phi_{t,s}(x),
	\qquad
	\mathcal K_s:=\mathcal J_s^{-1}.
	\]
Then the	Matrix It\^o calculus yields
	\[
	\,\mathrm d\mathcal K_s
	=\mathcal K_s\left(-B_s+\sum_{\nu=1}^rC_s^\nu C_s^\nu\right)\,\mathrm d s
	-\sum_{\nu=1}^r\mathcal K_sC_s^\nu\,\mathrm d W_s^\nu,
	\qquad \mathcal K_t=\mathrm{Id}.
	\]
Since	the coefficients are uniformly bounded, we have
	\begin{equation}\label{x2}
	\sup_{0\le t\le T}\sup_{x\in\mathbb R^d}
	\mathbb E^{\widetilde{\mathbb Q}}\sup_{t\le s\le T}|\mathcal K_s|^q
	\le C_q
	\qquad(q\ge2).
	\end{equation}
	For $j\ge2$, we have 
	\begin{align*}
		&\big|(D^j\Psi_{t,s})(\Phi_{t,s}(x))\big|\\
		&\quad\le C_j|\mathcal K_s|^j
		\sum_{\ell=1}^{j-1}
		\sum_{\substack{i_1+\cdots+i_\ell=j\\i_a\ge1}}
		\big|(D^\ell\Psi_{t,s})(\Phi_{t,s}(x))\big|
		\prod_{a=1}^{\ell}|D_x^{i_a}\Phi_{t,s}(x)|,
	\end{align*}
by induction which implies 
	\begin{equation}\label{x1}
	\sup_{0\le t\le T}\sup_{x\in\mathbb R^d}
	\mathbb E^{\widetilde{\mathbb Q}}\sup_{t\le s\le T}
	\big|(D^j\Psi_{t,s})(\Phi_{t,s}(x))\big|^q
	\le C_{j,q},
	\qquad 1\le j\le n_*.
	\end{equation}
	
Note that, $J_{t,s}(x)=\det\mathcal J_s>0$ and
	$J_{t,s}(x)^{-1}=\det\mathcal K_s$. Then the Hadamard's inequality gives
	\[
	J_{t,s}(x)^q+J_{t,s}(x)^{-q}
	\le C_{d,q}\left(
	|\mathcal J_s|^{dq}+|\mathcal K_s|^{dq}
	\right).
	\]
Finally, combine \eqref{x2}, \eqref{x1} and \eqref{eq:flow-derivative-moment} prove
	\eqref{eq:maximal-inverse-flow-moments}.

and $r\log^+r\le r\log r+e^{-1}$, where
$\log^+r:=\max\{\log r,0\}$, for every $R>1$, and as $R\to\infty$, we have 
\[
\sup_j\mathbb E^\mathbb Q\!\left[Z_{\rho_j^{t,x}}^{t,x}\mathbf 1_{\{Z_{\rho_j^{t,x}}^{t,x}>R\}}\right]
\le \frac{1}{\log R}\sup_j\mathbb E^\mathbb Q[Z_{\rho_j^{t,x}}^{t,x}\log^+Z_{\rho_j^{t,x}}^{t,x}]
\le \frac{C_k\left\langle x\right\rangle^2+e^{-1}}{\log R}
\longrightarrow0.
\]

Thus, $\{Z_{\rho_j^{t,x}}^{t,x}\}_{j\ge1}$ is uniformly integrable, and $Z_{\rho_j^{t,x}}^{t,x}\to Z_T^{t,x}$ a.s., which implies that  $Z^{t,x}$ is a uniformly
integrable $\mathbb Q$-martingale on $[0,T]$.
Therefore,
\eqref{eq:Qhat-tx} defines an equivalent probability measure.
Note that, $Z_T^{t,x}=Z_\theta^{t,x}$ because $H^{t,x}=0$ after $\theta$. Then, 
Girsanov's theorem applied to $(W,Y)$  shows that
$W$ and $\widehat Y^{t,x}$ are independent $\mathbb F$-Brownian
motions under $\widehat\mathbb Q^{t,x}$. This completes the proof.

<!-- proof-3: body_md -->
For fixed $N$, spatial differentiability through order $n_*$
in $\mathbb S^q\times\mathbb H^q\times\mathbb H^q$ follows
by applying standard stability estimates for Lipschitz BSDEs
successively to spatial difference quotients, using
\eqref{eq:hN-derivatives}, the regularity of $\gamma$,
and Lemma~\ref{lem:flow-estimates}.
We choose jointly measurable versions of the solutions
and their spatial derivatives.
	
Suppress  parameters  $(N,t,x)$.
	Differentiating \eqref{eq:truncated-BSDE} gives
	\begin{equation}
		\left\{
		\begin{aligned}
			\,\mathrm d D_x^\beta F_s
			&=-\left[H_s^\top D_x^\beta G_s+R_s^\beta\right]\,\mathrm d s
			+(D_x^\beta G_s)^\top\,\mathrm d Y_s
			+(D_x^\beta\widetilde G_s)^\top\,\mathrm d W_s,\\
			D_x^\beta F_T&=D_x^\beta\Gamma,
		\end{aligned}
		\right.
		\label{eq:derivative-BSDE}
	\end{equation}
	where
	\begin{equation}
		R_s^\beta
		:=\sum_{0<\eta\le\beta}\binom{\beta}{\eta}
		(D_x^\eta H_s)^\top D_x^{\beta-\eta}G_s,
		\qquad R_s^0:=0.
		\label{eq:R-beta}
	\end{equation}
	Here, $0<\eta\le\beta$ means that $\eta$ is a nonzero multi-index
	with $0\le\eta_k\le\beta_k$ for every $k=1,\ldots,d$, and
	$\displaystyle \binom{\beta}{\eta}
	:=\prod_{k=1}^d\binom{\beta_k}{\eta_k}
	=\prod_{k=1}^d\frac{\beta_k!}{\eta_k!(\beta_k-\eta_k)!}$
	is the multi-index binomial coefficient.
	
	Write
	\[
	\widehat\mathbb Q:=\widehat\mathbb Q^{N,t,x}, \qquad 
	\widehat\mathbb E_\rho[\cdot]
	:=\mathbb E^{\widehat\mathbb Q}[\cdot\mid\mathcal F_\rho], 	
	\qquad
	\mathfrak D_j(\rho)
	:=1+\sum_{i=1}^j|D_x^iX_\rho^{t,x}|.
	\]
	Under $\widehat\mathbb Q$,    \eqref{eq:derivative-BSDE} becomes:
	\begin{equation}
		\,\mathrm d D_x^\beta F_s
		=-R_s^{\beta}\,\mathrm d s
		+(D_x^\beta G_s)^\top\,\mathrm d\widehat Y_s^{N,t,x}
		+(D_x^\beta\widetilde G_s)^\top\,\mathrm d W_s.
		\label{eq:derivative-under-Qhat}
	\end{equation}
	
	
	We prove by induction on $j$, that for every
	$2\le q<\infty$,  $|\beta|\le j$ and $t\le\rho\le T$, the following holds:
	\begin{equation}
		\begin{aligned}
			|D_x^\beta F_\rho|^q
			&+\widehat\mathbb E_\rho\left[
			\left(\int_\rho^T
			\bigl(|D_x^\beta G_s|^2
			+|D_x^\beta\widetilde G_s|^2\bigr)\,\mathrm d s
			\right)^{q/2}\right]\\
			&\le P_{j,q}\bigl(\mathfrak D_j(\rho)\bigr).
		\end{aligned}
		\label{eq:closed-conditional-induction}
	\end{equation}

	
Indeed, 	for $j=0$, \eqref{eq:FN-representation} and
	Lemma~\ref{lem:conditional-linear-BSDE} give
	\eqref{eq:closed-conditional-induction} with a constant polynomial.
	Suppose the estimate holds through order $j-1$ at every finite
	exponent, and let $|\beta|=j$.
	Since $\Gamma=\gamma(X_T^{t,x})$, the chain rule and
	Lemma~\ref{lem:conditional-flow-polynomial} give
	\[
	\widehat\mathbb E_\rho\left[
	|D_x^\beta\Gamma|^q
	+\sum_{0<\eta\le\beta}
	\left(\int_\rho^T|D_x^\eta H_s|^2\,\mathrm d s\right)^q
	\right]
	\le P_{j,q}\bigl(\mathfrak D_j(\rho)\bigr).
	\]
	Moreover, conditional Cauchy--Schwarz yields
	\begin{equation}\label{xx1}
	\begin{aligned}
		&\widehat\mathbb E_\rho
		\left(\int_\rho^T|R_s^\beta|\,\mathrm d s\right)^q\\
		&\quad\le C_{\beta,q}
		\sum_{0<\eta\le\beta}
		\left\{
		\widehat\mathbb E_\rho
		\left(\int_\rho^T|D_x^\eta H_s|^2\,\mathrm d s\right)^q
		\right\}^{1/2}
		\left\{
		\widehat\mathbb E_\rho
		\left(\int_\rho^T|D_x^{\beta-\eta}G_s|^2\,\mathrm d s\right)^q
		\right\}^{1/2}\\
		&\quad\le P_{j,q}\bigl(\mathfrak D_j(\rho)\bigr),
	\end{aligned}
	\end{equation}
	where the last inequality uses the induction hypothesis at
	exponent $2q$. Applying Lemma~\ref{lem:conditional-linear-BSDE} to
	\eqref{eq:derivative-under-Qhat} then closes the induction.
	
Applying \eqref{eq:closed-conditional-induction} at times in a
countable dense subset of $[t,T]$, and using path continuity
together with the equivalence of $\widehat{\mathbb Q}$ and $\mathbb Q$, we obtain
	\[
	\sup_{t\le s\le T}|D_x^\beta F_s|^q
	\le
	\sup_{t\le s\le T}
	P_{|\beta|,q}\bigl(\mathfrak D_{|\beta|}(s)\bigr),

	\]
Note that,  for some finite
	integer $M$, we have 
	\[

	P_{|\beta|,q}\bigl(\mathfrak D_{|\beta|}(s)\bigr)
	\le
	C_{\beta,q}\left(
	1+\sum_{i=1}^{|\beta|}
	\sup_{t\le s\le T}|D_x^iX_s^{t,x}|^M

			&\mathbb E^\mathbb Q\left[
			\sup_{t\le s\le T}|D_x^\beta F_s|^q

			\left(\mathbb E^\mathbb Q\sup_{t\le s\le T}
			|D_x^\beta F_s|^{2q}\right)^{1/2}
			\left[
			\mathbb E^\mathbb Q\left(1+\int_t^T|H_s|^2\,\mathrm{d}s\right)^q
			\right]^{1/2}\\
			&\quad\le C_{k,\beta,q}\left\langle x\right\rangle^q.
		\end{aligned}
	\end{equation}
	
	We now prove \eqref{eq:uniform-G-parameter} by induction on
	$|\beta|$, simultaneously for all finite exponents $q\ge2$.
	For $\beta=0$, we have $R^0=0$, so the preceding energy
	estimate and \eqref{001} give
	\eqref{eq:uniform-G-parameter} at every such exponent.
	
	Suppose that \eqref{eq:uniform-G-parameter} holds for all
	derivatives of order at most $j-1$ and every finite exponent
	at least $2$, and let $|\beta|=j$.
	Using \eqref{eq:R-beta}, we obtain

			&\quad\le C_{\beta,q}
			\sum_{0<\eta\le\beta}

			\int_t^T|D_x^\eta H_s|^2\,\mathrm{d}s
			\right)^q
			\right\}^{1/2}
			\left\{
			\mathbb E^\mathbb Q\left(
			\int_t^T|D_x^{\beta-\eta}G_s|^2\,\mathrm{d}s
			\right)^q
			\right\}^{1/2}\\
			&\quad\le C_{k,\beta,q}\left\langle x\right\rangle^q.
		\end{aligned}
	\end{equation}
	Substituting \eqref{001} and \eqref{00} into the energy
	estimate proves \eqref{eq:uniform-G-parameter} for
	$|\beta|=j$ and closes the induction.
	All constants are independent of $N,t,x$.

	By \eqref{eq:Ztx} and Proposition~\ref{prop:param-likelihood},

	For $t\le s\le T$, define

	M_s:=\mathbb E^\mathbb Q[Z_T\Gamma\mid\mathcal F_s].

This is a continuous uniformly integrable martingale. Therefore, there exists predictable locally
square-integrable processes $K^Y$ and $K^W$ such that

	M_s=M_t+\int_t^s(K_a^Y)^\top\,\mathrm{d}Y_a
	+\int_t^s(K_a^W)^\top\,\mathrm{d}W_a.

	Set

	F_s:=\frac{M_s}{Z_s}
	=\mathbb E^{\widehat\mathbb Q}[\Gamma\mid\mathcal F_s],

	and define
	\begin{equation}
		G_s:=\frac{K_s^Y}{Z_s}-F_sH_s,
		\qquad
		\widetilde G_s:=\frac{K_s^W}{Z_s}.
		\label{eq:limit-integrands-definition}
	\end{equation}
Then,  processes in
\eqref{eq:limit-integrands-definition} are locally square-integrable, and

		|F_s|\le\|\gamma\|_\infty,
		\qquad t\le s\le T.
		\label{eq:limit-F-bound}
	\end{equation}

	\mathrm{d}F_s
	=-H_s^\top G_s\,\mathrm{d}s
	+G_s^\top\,\mathrm{d}Y_s
	+\widetilde G_s^\top\,\mathrm{d}W_s,
	\qquad F_T=\Gamma.

	We next verify square integrability under $\mathbb Q$.
	The growth assumption, $\theta\le\tau_k$, and

		\le C\mathbb E^\mathbb Q\int_t^T
		\chi_s^t\bigl(1+|X_s^{t,x}|^2+|u_s|^2\bigr)
		\,\mathrm{d}s
		\le C_k\left\langle x\right\rangle^2.

	\eta_j:=\inf\left\{s\in[t,T]:

	\bigl(|G_a|^2+|\widetilde G_a|^2+|H_a|^2\bigr)

		\mathbb E^\mathbb Q\int_t^T|H_s|^2\,\mathrm{d}s

	+\overline{\widetilde G}_s^\top\,\mathrm{d}W_s,
	\qquad \bar F_T=0.

	\bar F_s=\mathbb E^{\widehat\mathbb Q}[\bar F_T\mid\mathcal F_s]=0,
	\qquad t\le s\le T.

	\,\mathrm{d}s=0,
	\qquad \widehat\mathbb Q\text{-a.s.},

		\longrightarrow0.
		\label{eq:H-strong}

		&\quad\le
		\left(\mathbb E^\mathbb Q\int_t^T|H_a^N-H_a|^2\,\mathrm{d}a\right)^{1/2}
		\left(\mathbb E^\mathbb Q\int_t^T
		(|H_a^N|+|H_a|)^2\,\mathrm{d}a\right)^{1/2}
		\longrightarrow0.
	\end{align*}
	Therefore,
	\begin{align*}
		\mathbb E^\mathbb Q	&\sup_{0\le s\le T}
		|\log Z_s^N-\log Z_s|
		\longrightarrow0,
	\end{align*}
	which implies
	\begin{equation}
		\sup_{0\le s\le T}|Z_s^N-Z_s|
		\longrightarrow0
		\qquad\text{in $\mathbb Q$-probability}.
		\label{eq:Z-uniform-probability}
	\end{equation}

	The entropy estimates \eqref{eq:ZN-entropy} and
	\eqref{eq:param-entropy}, together with
	$r\log^+r\le r\log r+e^{-1}$, imply
	\[
	\sup_N\mathbb E^\mathbb Q[Z_T^N\log^+Z_T^N]
	+\mathbb E^\mathbb Q[Z_T\log^+Z_T]<\infty.
	\]
	Thus de la Vall\'ee--Poussin's criterion and
	\eqref{eq:Z-uniform-probability} yield
	\begin{equation}
		Z_T^N\longrightarrow Z_T
		\qquad\text{in }L^1(\mathbb Q).
		\label{eq:Z-L1}
	\end{equation}
	
	For $t\le s\le T$, define
	\[
	M_s^N:=\mathbb E^\mathbb Q[Z_T^N\Gamma\mid\mathcal F_s].
	\]
	By Bayes' formula,
	\[
	F_s^N=\frac{M_s^N}{Z_s^N},
	\qquad
	|F_s^N|\vee|F_s|\le\|\gamma\|_\infty.
	\]
	By \eqref{eq:Z-L1}, we have 
	\[
	\|Z_T^N\Gamma-Z_T\Gamma\|_{L^1(\mathbb Q)}
	\le\|\gamma\|_\infty\|Z_T^N-Z_T\|_{L^1(\mathbb Q)}
	\longrightarrow0.
	\]
	Hence,  for every
	$\varepsilon>0$,
	\[
	\mathbb Q\left(
	\sup_{t\le s\le T}|M_s^N-M_s|>\varepsilon
	\right)
	\le\frac1\varepsilon
	\mathbb E^\mathbb Q|(Z_T^N-Z_T)\Gamma|
	\longrightarrow0.
	\]
	Thus
	\begin{equation}
		\sup_{t\le s\le T}|M_s^N-M_s|
		\longrightarrow0
		\qquad\text{in $\mathbb Q$-probability}.
		\label{eq:M-uniform-probability}
	\end{equation}
	
	Define, 
	$\underline Z:=\inf_{t\le s\le T}Z_s>0$. It follows that 
	\begin{equation}
		\sup_{t\le s\le T}|F_s^N-F_s|
		\longrightarrow0
		\qquad\text{in $\mathbb Q$-probability}.
		\label{eq:F-uniform-probability}
	\end{equation}
	For every $\eta>0$,
	\begin{align*}
		\mathbb E^\mathbb Q\sup_{t\le s\le T}|F_s^N-F_s|^2
		&\le \eta^2
		+4\|\gamma\|_\infty^2
		\mathbb Q\left(
		\sup_{t\le s\le T}|F_s^N-F_s|>\eta
		\right).
	\end{align*}
	Letting first $N\to\infty$ and then $\eta\downarrow0$
	proves \eqref{eq:F-strong}.
	
	We next prove convergence of the martingale integrands.
	Write
	\[
	\mathrm{d}M_s^N
	=(K_s^{N,Y})^\top\,\mathrm{d}Y_s
	+(K_s^{N,W})^\top\,\mathrm{d}W_s.
	\]
	It\^o's formula applied to $M^N=Z^NF^N$ gives
	\begin{equation}
		K^{N,Y}=Z^N(G^N+F^NH^N),
		\qquad K^{N,W}=Z^N\widetilde G^N.
	\end{equation}
	Similarly,
	\begin{equation}
		K^Y=Z(G+FH),
		\qquad K^W=Z\widetilde G.
		\label{eq:K-identity}
	\end{equation}
	
	Apply Lemma~\ref{lem:localized-martingale-stability}
	with $a=t$, $B=(Y,W)$,

\label{lem:weighted-multiplier}

   \sup_{0\le s\le\theta}
   \left\langle \Phi_s(y)\right\rangle^{-m}J_s(y)|P_s(y)|

 \label{eq:weighted-multiplier}

<!-- proof-7: body_md -->
Let  $V_m(x)=\left\langle x\right\rangle^{-m}$, direct differentiation gives
\begin{align*}
 |DV_m(x)|\le C_mV_m(x)\left\langle x\right\rangle^{-1}, \qquad
 |D^2V_m(x)|\le C_mV_m(x)\left\langle x\right\rangle^{-2},
\end{align*}
and 
\begin{align*}
	\bigl|b(x,v)^\top DV_m(x)\bigr|
	+\left|\sum_{i,j=1}^d
	a^{ij}(x,v)D_{ij}V_m(x)\right|
	&\le C_m(1+|v|^2)V_m(x),\\
	\bigl|DV_m(x)\sigma(x,v)\bigr|^2
	&\le C_m(1+|v|^2)V_m(x)^2.
\end{align*}

Set $V_s=V_m(\Phi_s(y))$.  By It\^o's formula,
\[
\,\mathrm d V_s=V_s\mu_s\,\mathrm d s+V_s\nu_s^\top\,\mathrm d W_s,
\]
where
\begin{align*}
	\mu_s
	&:=
	\frac{\chi_s^0}{V_m(\Phi_s(y))}
	\left[
	\sum_{i=1}^d
	b^i(\Phi_s(y),u_s)D_iV_m(\Phi_s(y))
	+\frac12\sum_{i,j=1}^d
	a^{ij}(\Phi_s(y),u_s)D_{ij}V_m(\Phi_s(y))
	\right],\\
	\nu_s^\ell
	&:=
	\frac{\chi_s^0}{V_m(\Phi_s(y))}
	\sum_{i=1}^d
	D_iV_m(\Phi_s(y))
	\sigma^{i\ell}(\Phi_s(y),u_s),
	\qquad \ell=1,\ldots,r.
\end{align*}
It follows that
\[
\int_0^T\bigl(|\mu_s|+|\nu_s|^2\bigr)\,\mathrm d s
\le C_m\int_0^\theta(1+|u_s|^2)\,\mathrm d s
\le C_m(T+k).
\]


Therefore, for every finite $q\ge1$, we have 
\begin{equation}
 \mathbb E^\mathbb Q\sup_{s\le\theta}V_m(\Phi_s(y))^q
 \le C_{m,q,k}V_m(y)^q.
 \label{eq:negative-flow-moment}
\end{equation}

All finite moments of
$\sup_{s\le\theta}|J_s(y)P_s(y)|$ are bounded uniformly in $y$ by
\eqref{eq:flow-derivative-moment} and \eqref{eq:maximal-inverse-flow-moments}.  Hence, H\"older's inequality and
\eqref{eq:negative-flow-moment} give
\[
 \left\|\sup_{s\le\theta}

 \le C_{m,q,k,P}\left\langle y\right\rangle^{-m}.
\]

\begin{align*}
	&\mathbb E^\mathbb Q\int_0^\theta\int_{B_R}
	\left|
	h_N(x,u_s)^\top g_s^N(x)
	-h(x,u_s)^\top g_s(x)
	\right|\,\mathrm{d}x\,\mathrm{d}s\\
	&\quad\le
	\left(
	\mathbb E^\mathbb Q\int_0^\theta\int_{B_R}
	|h_N(x,u_s)-h(x,u_s)|^2\,\mathrm{d}x\,\mathrm{d}s
	\right)^{1/2}\\
	&\qquad\quad\times
	\left(
	\mathbb E^\mathbb Q\int_0^\theta\int_{B_R}
	|g_s^N(x)|^2\,\mathrm{d}x\,\mathrm{d}s
	\right)^{1/2}\\
	&\qquad+
	\left(
	\mathbb E^\mathbb Q\int_0^\theta\int_{B_R}
	|h(x,u_s)|^2\,\mathrm{d}x\,\mathrm{d}s
	\right)^{1/2}\\
	&\qquad\quad\times
	\left(
	\mathbb E^\mathbb Q\int_0^\theta\int_{B_R}
	|g_s^N(x)-g_s(x)|^2\,\mathrm{d}x\,\mathrm{d}s
	\right)^{1/2}
	\longrightarrow0.
\end{align*}

Finally, BDG gives

and using continuity in time and in the test function,
we obtain a common event of probability one on which
this identity holds simultaneously for all $t\in[0,T]$
and all $\varphi\in C_c^\infty(\mathbb R^d)$.

<!-- lemma-10: body_md -->
\label{lem:maximal-lsc}
Let $\mathcal H$ be a separable Hilbert space and $\theta\le T$ a stopping time.  On $L^2(\Omega\times(0,T);\mathcal H)$ define
\[
 \mathfrak J(v)
 :=\mathbb E^\mathbb Q\operatorname*{ess\,sup}_{0\le s\le\theta}\left\lVert v_s\right\rVert_{\mathcal H}^2,
\]
with value $+\infty$ when the right-hand side is infinite.  Then $\mathfrak J$ is convex and weakly lower semicontinuous.
<!-- end lemma-10 -->
<!-- proof-14: body_md -->
Convexity is immediate.  Suppose  $v^j\to v$
strongly in $L^2(\Omega\times(0,T);\mathcal H)$, if 
$
\liminf_{j\to\infty}\mathfrak J(v^j) =+\infty,
$ 
the desired inequality is immediate.
Otherwise choose a subsequence  $(v^{j_k})$ such that
\begin{equation}\label{cc}
\mathfrak J(v^{j_k})\longrightarrow \liminf_{j\to\infty}\mathfrak J(v^j) .
\end{equation}
Strong $L^2$ convergence allows us to extract a further
subsequence, still denoted by $(v^{j_k})$, for which
\[
\|v_s^{j_k}(\omega)-v_s(\omega)\|_{\mathcal H}
\longrightarrow0
\qquad
\text{for }\mathrm{d}\mathbb Q\otimes\mathrm{d}s
\text{-a.e. }(\omega,s).
\]
and  the convergence \eqref{cc} is preserved
under this further extraction.

By Fubini's theorem, for $\mathbb Q$-a.e.\ $\omega$, this convergence
holds for Lebesgue-a.e.\ $s\in(0,T)$.
For each such $\omega$,  we have 
\[
\operatorname*{ess\,sup}_{0<s\le\theta(\omega)}
\|v_s(\omega)\|_{\mathcal H}^2
\le
\liminf_{k\to\infty}
\operatorname*{ess\,sup}_{0<s\le\theta(\omega)}
\|v_s^{j_k}(\omega)\|_{\mathcal H}^2.
\]
Consequently, Fatou's lemma yields
\[
\begin{aligned}
	\mathfrak J(v)
	&\le
	\mathbb E^\mathbb Q\!\left[
	\liminf_{k\to\infty}
	\operatorname*{ess\,sup}_{0<s\le\theta}
	\|v_s^{j_k}\|_{\mathcal H}^2
	\right]\\
	&\le
	\liminf_{k\to\infty}\mathfrak J(v^{j_k})
	=\liminf_{j\to\infty}\mathfrak J(v^j).
\end{aligned}
\]
Thus $\mathfrak J$ is lower semicontinuous in the norm topology.
Since it is also convex, it is weakly lower semicontinuous
by \citet[Corollary~3.9, p.~61]{Brezis2011}.

\label{prop:weighted-Sobolev}
Let $(f,g)$ be the fields defined in
\eqref{eq:reconstruct-limit}. 
Set $\ell_0=d/2+2$.  Then
\begin{equation}
 \mathbb E^\mathbb Q\operatorname*{ess\,sup}_{s\le\theta}\left\lVert f_s\right\rVert_{H_{\ell_0}^{n_*}}^2
 +\mathbb E^\mathbb Q\int_0^\theta
    \left\lVert g_s\right\rVert_{H_{\ell_0}^{n_*}(\mathbb R^d;\mathbb R^m)}^2\,\mathrm d s
 <\infty.
 \label{eq:weighted-Sobolev-estimate}
\end{equation}
<!-- end proposition-8 -->
<!-- proof-15: body_md -->
Using the change of variables $x=\Phi_s(y)$, with
$\mathrm{d}x=J_s(y)\,\mathrm{d}y$, and applying
\eqref{eq:chain-f}, we obtain
\begin{align*}
 &\int_{\mathbb R^d}|D^\alpha v_s^N(x)|^2
       \left\langle x\right\rangle^{-2\ell_0}\,\mathrm d x
 \le C_\alpha\sum_{|\beta|\le|\alpha|}
 \int_{\mathbb R^d}|D_y^\beta V_s^{N,0,y}|^2
 |P_{\alpha,\beta,s}(y)|^2\\
 &\hspace{9em}{}
 \times\left\langle \Phi_s(y)\right\rangle^{-2\ell_0}J_s(y)\,\mathrm d y.
\end{align*}
Applying H\"older's inequality, Proposition \ref{prop:uniform-parameter}, and Lemma 
\ref{lem:weighted-multiplier},   we obtain
\begin{align}
 &\sup_N\mathbb E^\mathbb Q\operatorname*{ess\,sup}_{s\le\theta}\left\lVert f_s^N\right\rVert_{H_{\ell_0}^{n_*}}^2
 +\sup_N\mathbb E^\mathbb Q\int_0^\theta
    \left\lVert g_s^N\right\rVert_{H_{\ell_0}^{n_*}}^2\,\mathrm d s\\
 &\qquad\le C_k\int_{\mathbb R^d}
   \left(\left\langle y\right\rangle^{-2\ell_0}+\left\langle y\right\rangle^{2-2\ell_0}\right)\,\mathrm d y<\infty. \label{eq:explicit-weight-integral}
\end{align}

L^2\!\left(\Omega\times(0,T);
H_{\ell_0}^{n_*}(\mathbb R^d)\times
H_{\ell_0}^{n_*}(\mathbb R^d;\mathbb R^m)\right).

The local strong convergence in Proposition~\ref{prop:local-eulerian}
identifies its weak limit with $(f,g)$. Applying
Lemma~\ref{lem:maximal-lsc} to the first component and weak lower
semicontinuity of the $L^2$ norm to the second component gives
\eqref{eq:weighted-Sobolev-estimate}.

\label{prop:weighted-pointwise}
With $\ell_0=d/2+2$, for every $\lambda\ge\ell_0$ and every
$|\alpha|\le2$, \eqref{eq:weighted-sup-2} holds.
<!-- end proposition-9 -->
<!-- proof-16: body_md -->
Fix a multi-index $\alpha$ with $|\alpha|\le2$, and let
	$v\in H_{\ell_0}^{n_*}(\mathbb R^d)$.
	For $z\in\mathbb Z^d$, define $Q_z=z+(-1,1)^d$.
	Since $n_*-|\alpha|>d/2$, the Sobolev embedding
	\citep[Theorem~4.12, Part~I, Case~A, p.~85]{AdamsFournier2003}
	gives
	\begin{equation}
		\sup_{x\in Q_z}|D^\alpha v(x)|^2
		\le C\sum_{|\beta|\le n_*}
		\int_{Q_z}|D^\beta v(x)|^2\,\mathrm{d}x,
		\label{eq:local-Sobolev}
	\end{equation}
	where we use the continuous representative of $D^\alpha v$.
	By translation, the constant $C$ is independent of $z$.
	
	There exists a constant $c\ge1$, independent of $z$, such that
	\[
	c^{-1}\left\langle z\right\rangle\le\left\langle x\right\rangle\le c\left\langle z\right\rangle,
	\qquad x\in Q_z.
	\]
	Since the cubes $(Q_z)_{z\in\mathbb Z^d}$ cover $\mathbb R^d$,
	\eqref{eq:local-Sobolev} implies
	\begin{align*}
		\sup_{x\in\mathbb R^d}
		\frac{|D^\alpha v(x)|^2}{\left\langle x\right\rangle^{2\ell_0}}
		&\le C\sup_{z\in\mathbb Z^d}\left\langle z\right\rangle^{-2\ell_0}
		\sum_{|\beta|\le n_*}
		\int_{Q_z}|D^\beta v(x)|^2\,\mathrm{d}x\\
		&\le C\sup_{z\in\mathbb Z^d}
		\sum_{|\beta|\le n_*}
		\int_{Q_z}
		|D^\beta v(x)|^2\left\langle x\right\rangle^{-2\ell_0}\,\mathrm{d}x\\
		&\le C\left\lVert v\right\rVert_{H_{\ell_0}^{n_*}}^2.
	\end{align*}
	The same estimate holds for $\mathbb R^m$-valued functions by applying
	the scalar estimate to each component and summing.
	
	Applying this estimate to $v=f_s(\omega,\cdot)$ and
	$v=g_s(\omega,\cdot)$, taking the essential supremum in time
	for $f$ and integrating in time for $g$, and then taking
	expectations, Proposition~\ref{prop:weighted-Sobolev} yields
	\eqref{eq:weighted-sup-2} for $\lambda=\ell_0$.
	Finally, for $\lambda\ge\ell_0$,
	$\left\langle x\right\rangle^{-2\lambda}\le\left\langle x\right\rangle^{-2\ell_0}$, which proves the
	remaining cases.

<!-- proposition-10: body_md -->
\label{prop:local-time-continuity}
	Let $(f,g)$ be a weighted Sobolev solution of
	\eqref{eq:intro-dual-bspde}.
	Then $f$ admits a version such that, $\mathbb Q$-almost surely,
	for every $R<\infty$,
	\begin{equation}
		f\in C\bigl([0,\theta];H^{n_*-2}(B_R)\bigr)
		\hookrightarrow C\bigl([0,\theta];C(\overline B_R)\bigr),
		\label{eq:local-time-continuity}
	\end{equation}
<!-- end proposition-10 -->
<!-- proof-17: body_md -->
Fix $R<\infty$. By the coefficient assumptions and Leibniz's rule,
	\begin{align*}
		\|L_sf_s\|_{H^{n_*-2}(B_R)}
		&\le C_R(1+|u_s|^2)\|f_s\|_{H^{n_*}(B_{R})},\\
		\|h(\cdot,u_s)^\top g_s\|_{H^{n_*-2}(B_R)}
		&\le C_R(1+|u_s|)\|g_s\|_{H^{n_*-2}(B_{R})}.
	\end{align*}
	Together with \eqref{eq:solution-norm}, Cauchy--Schwarz, and
	$\int_0^\theta |u_s|^2\,\mathrm d s\le k$, these estimates imply
	\begin{equation}\label{xp}
	\int_0^\theta
	\left(
	\|L_sf_s\|_{H^{n_*-2}(B_R)}
	+\|h(\cdot,u_s)^\top g_s\|_{H^{n_*-2}(B_R)}
	\right)\,\mathrm d s<\infty
	\qquad\mathbb Q\text{-a.s.}
	\end{equation}
	Moreover, \eqref{eq:solution-norm} gives
	\begin{equation}\label{xp1}
	\mathbb E^\mathbb Q\int_0^\theta
	\|g_s\|_{H^{n_*-2}(B_R;\mathbb R^m)}^2\,\mathrm d s<\infty.
	\end{equation}
By \eqref{xp} and \eqref{xp1},	the drift integral has continuous paths in $H^{n_*-2}(B_R)$,
	so does the Hilbert-valued stochastic integral
	\citep[Chapter~4]{DaPratoZabczyk2014}.
	Hence, the weak identity defines
	an $H^{n_*-2}(B_R)$-continuous modification of $f$.


Choose the modifications consistently for $R\in\mathbb N$, agreement
almost everywhere and time continuity imply indistinguishability on
overlaps. Finally,
$n_*-2>d/2$ and the Sobolev embedding give
\eqref{eq:local-time-continuity}.

\label{prop:f-bound}

\begin{equation}
 |f_s(\omega,x)|\le\|\gamma\|_\infty
 \quad\text{for every }\omega\in\Omega_0,

 \label{eq:f-bound-common-version}
\end{equation}

By \eqref{eq:FN-representation},
$|F_r^{N,0,y}|\le \|\gamma\|_\infty$ for rational $(r,y)$.
	Proposition~\ref{prop:uniform-parameter} and Kolmogorov's continuity
	theorem yield versions jointly continuous in $(s,y)$.
	Taking a countable intersection over $N$ and rational $(r,y)$,
	and then using continuity, gives
	\[
	|F_s^{N,0,y}|\le \|\gamma\|_\infty
	\quad\text{for all }N\in\mathbb N,\ s\in[0,T],\ y\in\mathbb R^d
	\]
	on a common event of probability one. Consequently,
	\[
	|f_s^N(x)|=|F_s^{N,0,\Psi_s(x)}|\le \|\gamma\|_\infty
	\quad\text{for all }N\in\mathbb N,\ 0\le s\le\theta,\ x\in\mathbb R^d.
	\]

 Since
	$(|f_s(x)|-\|\gamma\|_\infty)_+\le |f_s(x)-f_s^N(x)|$,

	\[
	\mathbb E^\mathbb Q\sup_{s\le\theta}
	\bigl\|(|f_s|-\|\gamma\|_\infty)_+\bigr\|_{L^2(B_R)}^2
	\le
	\mathbb E^\mathbb Q\sup_{s\le\theta}

	\qquad\text{for every }s\le\theta.
	\]

	$\bigcup_{R=1}^{\infty}B_R=\mathbb R^d$.

\label{prop:bspde-uniqueness}
	Suppose Assumptions~\ref{ass:filtering} and
	\ref{ass:bspde-regularity} hold.
	Fix $k\in\mathbb N$, an $\mathbb F^Y$-stopping time $\theta\le\tau_k$,

Let $\bar f=f^1-f^2$ and $\bar g=g^1-g^2$, where $f^1$ and $f^2$
are taken to be the locally space--time continuous versions provided
by Proposition~\ref{prop:local-time-continuity}.

			\bar G_s^x:=\bar g_s(X_s^{t,x}),\\
			\overline{\widetilde G}_s^x
			&:=\sigma(X_s^{t,x},u_s)^\top
			\nabla\bar f_s(X_s^{t,x}).
		\end{aligned}
		\label{eq:uniqueness-characteristic-fields}
	\end{equation}
	By the forward formula \eqref{eq:forward-flow-IW} in
	Lemma~\ref{lem:flow-composition}, for Lebesgue-a.e.\ $x$,

	\begin{equation}
		\,\mathrm d\bar F_s^x
		=-(H_s^{t,x})^\top\bar G_s^x\,\mathrm d s
		+(\bar G_s^x)^\top\,\mathrm d Y_s
		+(\overline{\widetilde G}_s^x)^\top\,\mathrm d W_s,
		\qquad \bar F_\theta^x=0.
		\label{eq:zero-terminal-flow-BSDE}
	\end{equation}

Proposition~\ref{prop:density-stability}, we have
	\begin{equation}
		\begin{aligned}
			&\mathbf 1_{A_t}\bar F_s^x=0
			\quad(t\le s\le\theta),\qquad \mathbb Q\text{-a.s.},\\
			&\mathbf 1_{A_t}\bar G^x=0,\qquad
			\mathbf 1_{A_t}\overline{\widetilde G}^{x}=0,
			\quad \,\mathrm d\mathbb Q\otimes\,\mathrm d s\text{-a.e.},
		\end{aligned}
		\label{eq:characteristic-zero}
	\end{equation}

		\label{eq:integrated-f-zero}
