<!-- assumption-1: body_md -->
\label{ass:powerlaw}
There are constants $0<c_{\mathrm{sp}}\le C_{\mathrm{sp}}$ and $\beta>1$ such
that
\begin{equation}\label{eq:powerlaw}
 c_{\mathrm{sp}}\,i^{-\beta}\;\le\;\sigma_i^2\;\le\;C_{\mathrm{sp}}\,i^{-\beta},
 \qquad i=1,\ldots,d.
\end{equation}
<!-- end assumption-1 -->
<!-- definition-1: body_md -->
\label{def:truncated-class}
For $K\le d$, $\eta>0$ and a probability law $p$ on $\mathbb{R}^K$, the truncated
sampler is the law
\begin{equation}\label{eq:output-law}
 \widetilde p^\eta
 =\bigl(p*\mathcal{N}(0,\eta^2I_K)\bigr)\otimes\mathcal{N}(0,\eta^2I_{d-K}),
\end{equation}
written in the coordinates of the basis $V$. They are compared with the
smoothed data law, expressed in the same coordinates,
\begin{equation}\label{eq:smoothed-target}
 q^\eta:=q*\mathcal{N}(0,\eta^2I_d)=\mathrm{Law}(X+\eta Z'),
\end{equation}
where $Z'\sim\mathcal{N}(0,I_d)$ is independent of $X$.
The ideal truncated sampler is the choice $p=q_K$; we denote its law by
$\widetilde q_K^{\,\eta}$.
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{thm:truncation-cost}
Let $q=\mathcal{N}(0,C)$, $1\le K<d$ and $\eta>0$.
\begin{enumerate}
\item[(a)] With $G_K:=\min\bigl\{1,\|(g_i)_{i>K}\|_2\bigr\}$, the ideal
truncated sampler satisfies
\[
 \tfrac1{100}\,G_K
 \;\le\;
 \mathrm{TV}\bigl(\widetilde q_K^{\,\eta},q^\eta\bigr)
 \;\le\;
 \tfrac32\,G_K,
\]
and, with explicit constants,
$\mathrm{TV}(\widetilde q_K^{\,\eta},q^\eta)\le\bigl(\frac18\sum_{i>K}x_i^2/(1+x_i)\bigr)^{1/2}$.
\item[(b)] For every probability law $p$ on $\mathbb{R}^K$, the truncated sampler
$\widetilde p^\eta$ of \eqref{eq:output-law} satisfies
\[
 \mathrm{TV}\bigl(\widetilde p^\eta,q^\eta\bigr)
 \;\ge\;
 \mathrm{TV}\bigl(\widetilde q_K^{\,\eta},q^\eta\bigr).
\]
\end{enumerate}
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:num-directions}
Let $q=\mathcal{N}(0,C)$ with the exact power law \eqref{eq:powerlaw2}, and let
$\eta>0$.
\begin{enumerate}
\item[(a)] \emph{Sufficiency.} For every $1\le K<d$,
\begin{equation}\label{eq:sharp-tv}
 \mathrm{TV}\bigl(\widetilde q_K^{\,\eta},q^\eta\bigr)
 \le\frac{C_{\mathrm{sp}}}{\eta^2\sqrt{8(2\beta-1)}}\,K^{-(2\beta-1)/2}.
\end{equation}
In particular, the truncation error is at most $\varepsilon/2$ whenever
$K_\varepsilon(\eta)\le K<d$, where
\[
 K_\varepsilon(\eta)
 :=\biggl\lceil
 \Bigl(\frac{C_{\mathrm{sp}}}{\eta^2\,\varepsilon\sqrt{2(2\beta-1)}}\Bigr)^{2/(2\beta-1)}
 \biggr\rceil.
\]
\item[(b)] \emph{Necessity.} Let $\varepsilon<1/200$ and $2K\le d$. If some
truncated sampler that retains $K$ directions satisfies

<!-- proposition-2: body_md -->
\label{prop:learnable}
Let $X\in\mathbb{R}^{d}$ satisfy $\mathbb{E} X=0$ and $\mathbb{E}\|X\|^2<\infty$, and write
$\sigma_i^2:=\mathbb{E} X_i^2$ for its coordinate variances in a fixed orthonormal
basis. For a noise level $\eta>0$ let $Y=X+\eta Z$ with
$Z\sim\mathcal{N}(0,I_d)$ independent of $X$, and let $p_\eta:=\mathrm{Law}(Y)$ (the smoothed
law $q^\eta$ of \eqref{eq:smoothed-target} when $X\sim q$). Then, for every
coordinate $i$,
\[
 \partial_i\log p_\eta(y)
 =\underbrace{-\frac{y_i}{\eta^2}}_{\text{Gaussian reference}}
 +\underbrace{\frac{m_i(y)}{\eta^2}}_{\text{data-dependent correction}},
 \qquad
 m_i(y):=\mathbb{E}[X_i\mid Y=y],
\]
and, writing $g_i(\eta):=\sigma_i^2/(\sigma_i^2+\eta^2)$ for the Wiener gain,
\begin{equation}\label{eq:learnable-fraction}
 g_i(\eta)^2
 \;\le\;
 \frac{\mathbb{E}\,m_i(Y)^2}{\mathbb{E}\,Y_i^2}
 \;\le\;
 g_i(\eta).
\end{equation}
The lower bound is attained when $X$ is Gaussian with covariance diagonal in
the chosen basis.
<!-- end proposition-2 -->
<!-- proof-1: body_md -->
The decomposition is Tweedie's formula
$\nabla\log p_\eta(y)=(\mathbb{E}[X\mid Y=y]-y)/\eta^2$, and
$\mathbb{E} Y_i^2=\sigma_i^2+\eta^2$ because $X$ and $Z$ are independent and centered.
Upper bound: by Jensen's inequality for conditional expectations,
$\mathbb{E}\,m_i(Y)^2\le\mathbb{E} X_i^2=\sigma_i^2$, so the ratio is at most
$\sigma_i^2/(\sigma_i^2+\eta^2)=g_i$.
Lower bound: by orthogonality of the conditional mean,
$\mathbb{E}\,m_i(Y)^2=\sigma_i^2-\mathbb{E}\bigl(X_i-m_i(Y)\bigr)^2$, and since the
conditional mean minimises the mean-squared error over all measurable
functions of $Y$,
\[
 \mathbb{E}\bigl(X_i-m_i(Y)\bigr)^2
 \le\mathbb{E}\bigl(X_i-g_iY_i\bigr)^2
 =\sigma_i^2-2g_i\sigma_i^2+g_i^2(\sigma_i^2+\eta^2)
 =\sigma_i^2(1-g_i).
\]
Hence $\mathbb{E}\,m_i(Y)^2\ge g_i\sigma_i^2$, and the ratio is at least
$g_i\sigma_i^2/(\sigma_i^2+\eta^2)=g_i^2$. If $X$ is Gaussian with diagonal
covariance, then $m_i(y)=g_iy_i$ and $\mathbb{E}\,m_i(Y)^2=g_i^2(\sigma_i^2+\eta^2)$,
which is the lower bound.

<!-- lemma-1: body_md -->
\label{lem:sharp-residual}
\label{prop:residual-kl-sharpness}
For every $1\le K<d$ and $\eta>0$,
\begin{equation}\label{eq:sharp-kl}
 \mathrm{KL}\bigl(\widetilde q_K^{\,\eta}\,\|\,q^\eta\bigr)
 =\frac12\sum_{i>K}\Bigl[\log(1+x_i)-\frac{x_i}{1+x_i}\Bigr]
 =\frac12\sum_{i>K}\bigl[-g_i-\log(1-g_i)\bigr]
 \le\frac14\sum_{i>K}\frac{x_i^2}{1+x_i}.
\end{equation}
If $x_i\le1$ for all $i>K$, then also
$\mathrm{KL}(\widetilde q_K^{\,\eta}\|q^\eta)\ge\frac1{16}\sum_{i>K}x_i^2$.
Moreover, with $S:=\sum_{i>K}\min\{x_i^2,1\}$,
\begin{equation}\label{eq:tv-lower}
 \frac12\min\Bigl\{1,\ \frac{S}{35}\Bigr\}
 \;\le\;
 \mathrm{TV}\bigl(\widetilde q_K^{\,\eta},q^\eta\bigr)
 \;\le\;
 \Bigl(\frac18\sum_{i>K}\frac{x_i^2}{1+x_i}\Bigr)^{1/2}.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
\emph{Identity.} Since the two laws agree on the first $K$ coordinates and
are products, the divergence is the sum over $i>K$ of the divergences between
$\mathcal{N}(0,\eta^2)$ and $\mathcal{N}(0,\eta^2(1+x_i))$. The one-dimensional formula
$\mathrm{KL}(\mathcal{N}(0,a)\|\mathcal{N}(0,b))=\frac12(a/b-1-\log(a/b))$ with $a/b=1/(1+x_i)=1-g_i$
gives $\frac12[\log(1+x_i)-x_i/(1+x_i)]=\frac12[-g_i-\log(1-g_i)]$.

\emph{Upper bound.} For $g\in[0,1)$,
\[
 -g-\log(1-g)=\sum_{k\ge2}\frac{g^k}{k}
 \le\frac{g^2}{2}\sum_{j\ge0}g^j
 =\frac{g^2}{2(1-g)}
 =\frac{x^2}{2(1+x)} .
\]

\emph{Lower bound in divergence.} Let $\phi(x)=\log(1+x)-x/(1+x)$. On
$[0,1]$ the function $\phi(x)-x^2/8$ vanishes at $0$ and has derivative
$x\bigl((1+x)^{-2}-\frac14\bigr)\ge0$, so $\phi(x)\ge x^2/8$ there; the outer
factor $\frac12$ gives the claim.

\emph{Upper bound in total variation.} Pinsker's inequality
$\mathrm{TV}\le\sqrt{\mathrm{KL}/2}$ and \eqref{eq:sharp-kl}.

\emph{Lower bound in total variation.} Let
$\mathrm{BC}(\mu,\nu)=\int\sqrt{\mathrm d\mu\,\mathrm d\nu}$ be the
Bhattacharyya coefficient. It is multiplicative over independent
coordinates, and $1-\mathrm{BC}(\mu,\nu)\le\mathrm{TV}(\mu,\nu)$ (Le Cam's inequality
$H^2\le\mathrm{TV}$ with $H^2=1-\mathrm{BC}$). For centered one-dimensional Gaussians
with variances $a,b$, $\mathrm{BC}=\bigl(2\sqrt{ab}/(a+b)\bigr)^{1/2}$; with
$a=\eta^2$ and $b=\eta^2(1+x_i)$ this equals $1-\psi(x_i)$, where
\[
 \psi(x):=1-\Bigl(\frac{2\sqrt{1+x}}{2+x}\Bigr)^{1/2}.
\]
The function $\psi$ is increasing on $[0,\infty)$, because the ratio of the
geometric to the arithmetic mean of $1$ and $1+x$ decreases in $x$. The ratio
$\psi(x)/x^2$ decreases on $(0,1]$ from its limit $1/16$ at $0$ to
$\psi(1)\approx0.0290>1/35$. Hence $\psi(x)\ge\min\{x^2,1\}/35$ for all
$x\ge0$, and
\[
 \mathrm{TV}\bigl(\widetilde q_K^{\,\eta},q^\eta\bigr)
 \ge1-\prod_{i>K}\bigl(1-\psi(x_i)\bigr)
 \ge1-\exp\Bigl(-\sum_{i>K}\psi(x_i)\Bigr)
 \ge1-e^{-S/35}
 \ge\frac12\min\Bigl\{1,\frac{S}{35}\Bigr\},
\]
using $1-e^{-u}\ge\frac12\min\{u,1\}$ for $u\ge0$.

\label{lem:traces}
Under the upper side of Assumption~\ref{ass:powerlaw}, namely
$\sigma_i^2\le C_{\mathrm{sp}}i^{-\beta}$ with $\beta>1$, use the Wiener-gain

\[

\medskip\noindent\textbf{Bounding \(\operatorname{tr}(\Lambda)\).}

Since \(\lambda_i=\lambda_0g_i\) and \(0\le g_i\le1\),
\[
 \operatorname{tr}(\Lambda)
 =
 \lambda_0\sum_{i=1}^K g_i
 \le

\]

Since \(0\le g_i\le1\), we have \(g_i^2\le g_i\le1\). Hence
\[
 \operatorname{tr}(\Lambda^2)
 =
 \lambda_0^2\sum_{i=1}^K g_i^2
 \le
 \lambda_0^2 K.
\]

\[
 m_{2,\Lambda}^2
 =
 \sum_{i=1}^K \lambda_i\sigma_i^2
 =

\]
Since \(g_i\le1\),
\[

 \le

 m_{2,\Lambda}^2

 \le

\]

 \le
 \lambda_0 C_{\mathrm{sp}}\zeta(\beta),

independently of \(K\) and \(d\).

 \lambda_i\ge\lambda_0\,\frac{c_{\mathrm{sp}}}{c_{\mathrm{sp}}+C_{\mathrm{sp}}}
 =\lambda_\star,
 \qquad
 \kappa\le\frac{\lambda_0}{\lambda_\star}=\kappa_0 .
\]

<!-- proof-9: body_md -->
The sampler outputs
$\tilde{X} = V_K\hat{X} + Z$ with $\hat{X} \sim p$ and
$Z \sim \nu_\perp$ independent, supported on the orthogonal
subspaces $S_K$ and $S_K^\perp$ respectively. The law of
$\tilde{X}$ is $\tilde{p} = p \otimes \nu_\perp$ (product on
$S_K \times S_K^\perp \cong \mathbb{R}^d$).

\begin{align}
 \mathrm{TV}(\tilde{p},\,q)
 &= \mathrm{TV}(p\otimes\nu_\perp,\;q) \notag\\
 &\le \mathrm{TV}(p\otimes\nu_\perp,\;q_K\otimes\nu_\perp)
 + \mathrm{TV}(q_K\otimes\nu_\perp,\;q).
 \label{eq:app-triangle}
\end{align}

The distributions $p\otimes\nu_\perp$ and $q_K\otimes\nu_\perp$ share
the factor $\nu_\perp$. We claim:
\begin{equation}\label{eq:app-product-tv}
 \mathrm{TV}(\mu_1\otimes\nu,\;\mu_2\otimes\nu)
 = \mathrm{TV}(\mu_1,\,\mu_2)
\end{equation}
for any probability measures $\mu_1,\mu_2$ and any probability measure $\nu$.
Indeed, let $\Delta:=\mu_1-\mu_2$ be the signed measure difference. Since
$\nu$ is a positive probability measure, the variation measure factorizes:
\[
 |\Delta\otimes\nu|=|\Delta|\otimes\nu .
\]
Therefore
\[
\begin{aligned}
 \mathrm{TV}(\mu_1\otimes\nu,\mu_2\otimes\nu)
 &=\frac12 |(\mu_1-\mu_2)\otimes\nu|(\mathcal X\times\mathcal Y)\\
 &=\frac12 |\mu_1-\mu_2|(\mathcal X)\,\nu(\mathcal Y)\\
 &=\mathrm{TV}(\mu_1,\mu_2),
\end{aligned}
\]
which proves~\eqref{eq:app-product-tv}.

Applying~\eqref{eq:app-product-tv} with $\mu_1 = p$,
$\mu_2 = q_K$, $\nu = \nu_\perp$ to the first term
of~\eqref{eq:app-triangle}:
\[
 \mathrm{TV}(p\otimes\nu_\perp,\;q_K\otimes\nu_\perp)
 = \mathrm{TV}(p,\,q_K).
\]
Substituting into~\eqref{eq:app-triangle} and using the
definition of $\varepsilon_{\mathrm{PCA}}$:
\[
 \mathrm{TV}(\tilde{p},\,q)
 \le \mathrm{TV}(p,\,q_K) + \varepsilon_{\mathrm{PCA}}.
\]

\[

 \widehat\lambda_i

 \lambda_0

 \le
 c_0
 \min\left\{
  \sigma^2,\,
  \sigma_K^2,\,
  \delta_{\mathrm{gap}},\,
  \frac{\eta^2\delta_{\mathrm{gap}}}{\sigma_1^2\sqrt K}
 \right\},
\]

 \delta_{\mathrm{gap}}:=\sigma_K^2-\sigma_{K+1}^2.

non-diagonal:
\[
 \widehat C_K:=\widehat V_K^\top\mathcal C\,\widehat V_K.
\]

active-space OU process with initial covariance \(\widehat C_K\) and rate matrix
\(\widehat\Lambda\), namely
\[
 \widehat q_{K,t}
 =
 \mathcal N(0,\widehat D_t),
 \qquad
 \widehat D_t
 =
 e^{-\widehat\Lambda t}\widehat C_K e^{-\widehat\Lambda t}
 +
 \sigma^2(I_K-e^{-2\widehat\Lambda t}).
\]

\[
 \widehat s_t(u)=\nabla\log \widehat q_{K,t}(u)
 =
 -\widehat D_t^{-1}u.

 \underbrace{
 \widetilde O\!\left(\sqrt{\frac K N}\right)

For the forward OU \eqref{eq:generic-forward-OU} on $\mathbb{R}^r$,

<!-- proof-12: body_md -->
Solve \eqref{eq:generic-forward-OU} explicitly:
\[
U_\tau = e^{-\bm{\Lambda}\tau}U_0
 + \int_0^\tau e^{-\bm{\Lambda}(\tau-s)}\sqrt{2\bm{\Lambda}}\,\sigma\,dB_s.
\]
Then:
\[
\mathbb{E}\left[\left\|U_\tau\right\|^2\right]
= \mathbb{E}\left[\left\|e^{-\bm{\Lambda}\tau}U_0\right\|^2\right]

&\le \sum_{i=1}^r \sigma^2 = r\sigma^2.
\end{align*}

<!-- lemma-6: body_md -->
\label{lem:fisher}
Under Assumption~\ref{ass:lipschitz} (applied to the coefficient-space
marginals $q_{r,t}$ on $\mathbb{R}^r$), for every $t \in [\tau_0, T]$ and every
diagonal positive semidefinite matrix
$M = \mathrm{diag}(m_1, \dots, m_r)$, $m_i \ge 0$,
\[
  \mathbb{E}_{U \sim q_{r,t}}\bigl[\|\nabla \log q_{r,t}(U)\|_M^2\bigr]
  \le L \,\mathrm{tr}(M).
\]
In particular, for $M = I_r$,
$\mathbb{E}_{U \sim q_{r,t}}\bigl[\|\nabla \log q_{r,t}(U)\|^2\bigr] \le L r$.

<!-- proof-13: body_md -->
Write $q_{r,t}=e^{-V_t}$ for some $V_t\in C^2(\mathbb R^r)$, as
guaranteed by Assumption~\ref{ass:lipschitz}. Fix a coordinate
\(i\in\{1,\ldots,r\}\). Since \(q_{r,t}=e^{-V_t}\), we have
\[
 \partial_i q_{r,t}(x)
 =
 -q_{r,t}(x)\,\partial_i V_t(x).
\]
Therefore
\[
 \partial_i\!\left(q_{r,t}\,\partial_i V_t\right)
 =
 (\partial_i q_{r,t})\,\partial_i V_t
 +
 q_{r,t}\,\partial_{ii}^2 V_t
 =
 -q_{r,t}(\partial_i V_t)^2
 +
 q_{r,t}\,\partial_{ii}^2 V_t.
\]
Rearranging gives the pointwise identity
\[
 q_{r,t}(\partial_i V_t)^2
 =
 q_{r,t}\,\partial_{ii}^2 V_t
 -
 \partial_i\!\left(q_{r,t}\,\partial_i V_t\right).
\]

Integrating over the cube \(Q_R=[-R,R]^r\) and applying the divergence theorem,
\[
\begin{aligned}
 \int_{Q_R} q_{r,t}(x)(\partial_i V_t(x))^2\,dx
 &=
 \int_{Q_R} q_{r,t}(x)\,\partial_{ii}^2 V_t(x)\,dx \\
 &\quad
 -
 \int_{\partial Q_R}
 q_{r,t}(x)\,\partial_i V_t(x)\,n_i(x)\,dS(x).
\end{aligned}
\]
By the boundary condition in Assumption~\ref{ass:lipschitz}, the
boundary term converges to zero as \(R\to\infty\).

Since the score is \(L\)-Lipschitz and \(V_t\in C^2\), we have
\[
 \|\nabla^2 V_t(x)\|_{\mathrm{op}}\le L,
\]
and hence, for every \(i\),
\[
 \partial_{ii}^2 V_t(x)\le L.
\]
Moreover \(|\partial_{ii}^2 V_t(x)|\le L\), so dominated convergence applies to
the term
\[
 \int_{Q_R} q_{r,t}(x)\,\partial_{ii}^2V_t(x)\,dx,
\]
while the left-hand side converges by monotone convergence, its integrand
being nonnegative.
Letting \(R\to\infty\), we obtain
\[
 \mathbb E_{U\sim q_{r,t}}
 \left[
  (\partial_i V_t(U))^2
 \right]
 =
 \mathbb E_{U\sim q_{r,t}}
 \left[
  \partial_{ii}^2 V_t(U)
 \right]
 \le
 L.
\]

Since
\[
 \nabla\log q_{r,t} = -\nabla V_t,
\]
we have
\[
 (\partial_i\log q_{r,t})^2
 =
 (\partial_i V_t)^2.
\]
Therefore
\[
 \mathbb E_{U\sim q_{r,t}}
 \left[
  (\partial_i\log q_{r,t}(U))^2
 \right]
 \le
 L.
\]

Multiplying by \(m_i\ge0\) and summing over \(i=1,\ldots,r\) gives
\[
\begin{aligned}
 \mathbb E_{U\sim q_{r,t}}
 \left[
  \|\nabla\log q_{r,t}(U)\|_M^2
 \right]
 &=
 \sum_{i=1}^r
 m_i\,
 \mathbb E_{U\sim q_{r,t}}
 \left[
  (\partial_i\log q_{r,t}(U))^2
 \right] \\
 &\le
 \sum_{i=1}^r
 m_i L
 =
 L\,\operatorname{tr}(M).
\end{aligned}
\]
Taking \(M=I_r\) gives
\[
 \mathbb E_{U\sim q_{r,t}}
 \left[
  \|\nabla\log q_{r,t}(U)\|^2
 \right]
 \le
 Lr.
\]
This completes the proof.

\begin{align*}

$dU_t=-\bm{\Lambda} U_t\,dt+\sigma\sqrt{2\bm{\Lambda}}\,dB_t$,
the increment satisfies

U_{\tau'}-U_\tau
=\int_\tau^{\tau'}(-\bm{\Lambda} U_s)\,ds

$\bm{\Lambda}$-norm:

\label{eq:split-weighted}

\textbf{Drift term.}
By the Cauchy-Schwarz inequality:
\[
\left\|\int_\tau^{\tau'}\bm{\Lambda} U_s\,ds
\right\|^2_{\bm{\Lambda}}
\le\Delta\int_\tau^{\tau'}
 \|\bm{\Lambda} U_s\|^2_{\bm{\Lambda}}\,ds.
\]
Now:
\[
\|\bm{\Lambda} U_s\|^2_{\bm{\Lambda}}
=\sum_i\lambda_i(\lambda_i U_{s,i})^2
=\sum_i\lambda_i^3 U_{s,i}^2
\le\lambda_{\max}\sum_i\lambda_i^2 U_{s,i}^2
=\lambda_{\max}\|\bm{\Lambda}^{1/2}U_s\|^2_{\bm{\Lambda}}.
\]
Since $\|\bm{\Lambda}^{1/2}U_s\|^2_{\bm{\Lambda}}
=\sum_i\lambda_i\cdot\lambda_i U_{s,i}^2
=\|U_s\|^2_{\bm{\Lambda}^2}
\le\lambda_{\max}\|U_s\|^2_{\bm{\Lambda}}$,
we have:
\[
\|\bm{\Lambda} U_s\|^2_{\bm{\Lambda}}
\le\lambda_{\max}^2\|U_s\|^2_{\bm{\Lambda}}.
\]

\;\leq\;

\Bigl(m_{2,\Lambda}^2 + \sigma^2\,\mathrm{tr}(\Lambda)\Bigr),
\label{eq:drift-weighted}
\end{equation}

\begin{equation*}

 \;=\; \sum_i \lambda_i\,\mathbb{E}[U_{s,i}^2]

   \Bigl[e^{-2\lambda_i s}\mathbb{E}[U_{0,i}^2]
    + \sigma^2(1 - e^{-2\lambda_i s})\Bigr]
 \;\leq\; \sum_i \lambda_i\,\mathbb{E}[U_{0,i}^2]
   + \sigma^2\sum_i \lambda_i
\end{equation*}
\begin{equation}\label{eq:ou-second-moment-lambda}

\end{equation}

$\mathbb{E}[U_{s,i}^2] = e^{-2\lambda_i s}\mathbb{E}[U_{0,i}^2]

Since $\rho_i<1$ for all $i$, the covariance matrix
$\Sigma=\sigma^2(I-D^2)$ is positive definite.

\medskip
\noindent
Let $q_{Y'}$ be the density of $Y'=DY$. By the change-of-variables formula,
\[
q_{Y'}(u)=|\det D|^{-1}q(D^{-1}u),
\]
hence
\[
s_{Y'}(u):=\nabla\log q_{Y'}(u)=D^{-1}s(D^{-1}u).
\]
Since $\rho_i\in[1-\zeta,1)$ and $\zeta\le 1/2$, we have
$\rho_i\ge 1/2$, so
\[
\|D^{-1}\|_{\mathrm{op}}\le 2,
\qquad
\|D^{-1}-I\|_{\mathrm{op}}\le 2\zeta.
\]
Therefore
\begin{align}
\|s_{Y'}(u)-s(u)\|
&=\|D^{-1}s(D^{-1}u)-s(u)\| \notag\\
&\le \|D^{-1}\|_{\mathrm{op}}\,\|s(D^{-1}u)-s(u)\|
 +\|D^{-1}-I\|_{\mathrm{op}}\,\|s(u)\| \notag\\
&\le 2L\|D^{-1}u-u\|+2\zeta\|s(u)\| \notag\\
&\le 4L\zeta\|u\|+2\zeta\|s(u)\|.
\label{eq:syprime-minus-s}
\end{align}
In particular,
\begin{equation}
\|s_{Y'}(u)\|
\le \|s(u)\|+\|s_{Y'}(u)-s(u)\|
\le 4L\zeta\|u\|+(1+2\zeta)\|s(u)\|.
\label{eq:syprime-bound}
\end{equation}

\medskip
\noindent
Since $\Sigma\succ 0$, the density of $X=Y'+\varepsilon$ is
\[
\tilde q(u)=\int_{\mathbb{R}^d} q_{Y'}(y)\,\varphi_\Sigma(u-y)\,dy,
\]
where $\varphi_\Sigma$ is the Gaussian density with covariance $\Sigma$.

Differentiating under the integral sign and using
$\nabla_u\varphi_\Sigma(u-y)=-\Sigma^{-1}(u-y)\varphi_\Sigma(u-y)$,
we obtain
\[
\tilde s(u)
=
-\Sigma^{-1}
\frac{\int_{\mathbb{R}^d}(u-y)\,q_{Y'}(y)\varphi_\Sigma(u-y)\,dy}
{\tilde q(u)}
=
-\Sigma^{-1}\bigl(u-\mathbb{E}[Y'\mid X=u]\bigr).
\]
Equivalently, if we set
\[
\bar\varepsilon:=\mathbb{E}[u-Y'\mid X=u],
\]

\begin{equation}
\tilde s(u)=-\Sigma^{-1}\bar\varepsilon.
\label{eq:posterior-score}
\end{equation}

Also, integrating by parts in $y$,
\begin{align*}
\nabla \tilde q(u)
&=
\int q_{Y'}(y)\,\nabla_u\varphi_\Sigma(u-y)\,dy \\
&=
-\int q_{Y'}(y)\,\nabla_y\varphi_\Sigma(u-y)\,dy \\
&=
\int \nabla_y q_{Y'}(y)\,\varphi_\Sigma(u-y)\,dy \\
&=
\int q_{Y'}(y)\,s_{Y'}(y)\,\varphi_\Sigma(u-y)\,dy.
\end{align*}
Dividing by $\tilde q(u)$ yields
\begin{equation}
\tilde s(u)=\mathbb{E}[s_{Y'}(Y')\mid X=u].
\label{eq:denoising-identity}
\end{equation}

\[
\|s_{Y'}(u)-s_{Y'}(v)\|
=
\|D^{-1}(s(D^{-1}u)-s(D^{-1}v))\|
\le
\|D^{-1}\|_{\mathrm{op}}^2 L \|u-v\|.
\]

\begin{equation}
L'\le 4L.
\label{eq:Lprime}
\end{equation}

Then

\qquad\text{for all }z\in\mathbb{R}^d,

\[

\sigma^2\max_i(1-\rho_i^2)
\le 2\sigma^2\zeta.

\[
L'\|\Sigma\|_{\mathrm{op}}
\le 4L\cdot 2\sigma^2\zeta

\[
\nabla^2V_u(\varepsilon)\succeq \frac12\,\Sigma^{-1}
\qquad\text{for all }\varepsilon.
\]

\[

 \frac{1}{n_{\mathrm{patch}}}
 \sum_{j=1}^{n_{\mathrm{patch}}}
 x^{(j)}(x^{(j)})^\top .

constant $C>0$ such that, for every $t\geq 0$, with probability at least

 \|\widehat{\Sigma}-\Sigma\|_{\mathrm{op}}
 \leq

 \sqrt{\frac{D_p+t}{n_{\mathrm{patch}}}}

 \frac{D_p+t}{n_{\mathrm{patch}}}
 \right).

 \right).

 C'\sigma_1^2

On the event of Lemma~\ref{lem:bernstein},

 \le

 =

For any unit vector $a\in\mathbb R^K$, set $u=\widehat V_Ka$. Since
$\widehat P_Ku=u$,

 =\|(I-P_K)\widehat P_Ku\|
 \le \|(I-P_K)\widehat P_K\|_{\mathrm{op}}
 \le \|\widehat P_K-P_K\|_{\mathrm{op}}.

assumption $\Delta_n\le c_0\delta_{\mathrm{gap}}$, we have
\[
 \|\widehat P_K-P_K\|_{\mathrm{op}}
 \le

Choose the universal constant $c_0>0$ small enough that
\((2c_0)^2\le1/2\); for instance, $c_0\le1/(2\sqrt2)$ suffices under the
constant convention of Lemma~\ref{lem:dk}. Then
\[
 \|P_Ku\|^2
 =1-\|(I-P_K)u\|^2
 \ge \frac12.

\[
\begin{aligned}

 &=u^\top\mathcal C u \\
 &\ge \sigma_K^2\|P_Ku\|^2 \\
 &=\sigma_K^2\bigl(1-\|(I-P_K)u\|^2\bigr)
 \ge \frac12\sigma_K^2.
\end{aligned}

Because $K\le d_{\mathrm{eff}}^{\sigma}$, the exact power law gives

\[
 \widehat C_K\succeq \frac12\sigma^2 I_K.

 +\sigma^2(I_K-e^{-2\widehat\Lambda t}),

we consequently have, for every unit vector $a$,
\[
\begin{aligned}
 a^\top\widehat D_t a
 &\ge \frac{\sigma^2}{2}\|e^{-\widehat\Lambda t}a\|^2
  +\sigma^2\langle a,(I_K-e^{-2\widehat\Lambda t})a\rangle \\

  \left(1-\frac12 e^{-2\widehat\lambda_i t}\right)a_i^2
 \ge \frac{\sigma^2}{2}.
\end{aligned}

We first show that the empirical perturbation assumptions and the empirical

Under the exact power-law spectrum,

 =C_{\mathrm{sp}}\bigl(K^{-\beta}-(K+1)^{-\beta}\bigr).

\(K_{\max}\). We also need uniform control of the empirical mismatch factor.
By the mean-value theorem, there exist constants
\(0<c_\beta\le C_\beta<\infty\), depending only on \(\beta\) and

\[
 c_\beta K^{-\beta-1}
 \le
 \delta_{\mathrm{gap}}(K)
 \le
 C_\beta K^{-\beta-1}.
\]
Consequently, for every \(K\le K_{\max}\),
\[
 \frac{\sqrt K}{\delta_{\mathrm{gap}}(K)}
 \le
 C_\beta'
 \frac{\sqrt{K_{\max}}}{\delta_{\mathrm{gap}}(K_{\max})},
\]
where \(C_\beta'>0\) is another constant depending only on the spectral
power-law constants. Therefore the most restrictive smallness and
basis-mismatch requirements over \(K\le K_{\max}\) are controlled by their
values at \(K_{\max}\), up to fixed constants.

By Assumption~\ref{ass:sufficient_data} and

 \Delta_n=C_\Delta\sigma_1^2\sqrt{\frac{D_p}{n_{\mathrm{patch}}}},

the first term in the maximum defining the sample-size condition ensures

 \Delta_n\le c_0 M_{K_{\max}},

\label{def:broken}
For $A>0$, a breakpoint $b\ge1$ and exponents $\beta_1\le\beta_2$, the continuous broken power law is
\[
\sigma_i^2 \;=\;
\begin{cases}
A\, i^{-\beta_1}, & i\le b,\\[2pt]
A\, b^{\beta_2-\beta_1}\, i^{-\beta_2}, & i> b,
\end{cases}
\]
or equivalently
$\log\sigma_i^2=\log A-\beta_1\log i-(\beta_2-\beta_1)(\log i-\log b)_+$.
It decays like $i^{-\beta_1}$ before the breakpoint and like $i^{-\beta_2}$ after it, and is continuous at $i=b$. The
case $\beta_1=\beta_2$ is the exact power law~(2).

\label{prop:broken-envelope}
Let $\sigma_i^2$ be a continuous broken power law and $1\le\ell\le b\le h$. For every $\beta\in[\beta_1,\beta_2]$,
$c\, i^{-\beta}\le\sigma_i^2\le C\, i^{-\beta}$ for $\ell\le i\le h$, where
\[
\frac{C}{c}=\max\Bigl\{\Bigl(\frac{b}{\ell}\Bigr)^{\beta-\beta_1},\;\Bigl(\frac{h}{b}\Bigr)^{\beta_2-\beta}\Bigr\}
\]
and $C=A\,b^{\beta-\beta_1}$. In particular, with $\beta=\beta_2$ the ratio is $(b/\ell)^{\beta_2-\beta_1}$, which
does not depend on $h$: a broken power law with a fixed breakpoint satisfies Assumption \ref{ass:powerlaw} with exponent $\beta_2$ and
constants independent of the dimension.
