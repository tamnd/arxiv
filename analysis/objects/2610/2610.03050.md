<!-- proposition-1: body_md -->
\label{prop:structure}
For every nonzero Banach space $X$, the following statements hold.
\begin{enumerate}[label=\textup{(\roman*)}]
  \item $\mathcal S_{X,t}(0)=\mathcal J_t[0,X]=1$.
  \item For $\tau>0$,
  \[
     \mathcal S_{X,t}(\tau)=\tau\mathcal S_{X,t}(\tau^{-1}),
     \qquad
     \mathcal J_t[\tau,X]=\tau\mathcal J_t[\tau^{-1},X].
  \]
  \item If $1\leq r<s\leq\infty$, then
  \begin{align*}
     \mathcal S_{X,r}(\tau)&\leq\mathcal S_{X,s}(\tau)
       \leq2^{1/r-1/s}\mathcal S_{X,r}(\tau),\\
     \mathcal J_r[\tau,X]&\leq\mathcal J_s[\tau,X]
       \leq2^{1/r-1/s}\mathcal J_r[\tau,X].
  \end{align*}
  \item The functions $\tau\mapsto\mathcal S_{X,t}(\tau)$ and
  $\tau\mapsto\mathcal J_t[\tau,X]$ are $1$-Lipschitz on $[0,\infty)$.
\end{enumerate}
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Part (i) follows directly from $\left\lVert x\right\rVert=\left\lVert y\right\rVert=1$.  For (ii), multiply
the expression with parameter $\tau^{-1}$ by $\tau$ and then interchange
$x$ and $y$:
\begin{align*}
 \tau\mathcal M_t\bigl(\left\lVert x+\tau^{-1}y\right\rVert,\left\lVert \tau^{-1}x-y\right\rVert\bigr)
  &=\mathcal M_t\bigl(\left\lVert \tau x+y\right\rVert,\left\lVert x-\tau y\right\rVert\bigr)\\
  &=\mathcal M_t\bigl(\left\lVert y+\tau x\right\rVert,\left\lVert \tau y-x\right\rVert\bigr).
\end{align*}
Taking the infimum or supremum gives the assertion.  Part (iii) is
\eqref{eq:mean-comparison} followed by the same extremal operation.

For (iv), fix $x,y\inS_X$ and $\tau,\sigma\geq0$.  The reverse triangle
inequality gives
\[
 \left\lvert \left\lVert x+\tau y\right\rVert-\left\lVert x+\sigma y\right\rVert\right\rvert\leq\left\lvert \tau-\sigma\right\rvert,
 \qquad
 \left\lvert \left\lVert \tau x-y\right\rVert-\left\lVert \sigma x-y\right\rVert\right\rvert\leq\left\lvert \tau-\sigma\right\rvert.
\]
By \eqref{eq:mean-lipschitz}, the corresponding power-mean profiles differ
by at most $\left\lvert \tau-\sigma\right\rvert$, uniformly in $(x,y)$.  Infima and suprema of
uniformly Lipschitz families preserve the same Lipschitz constant.

<!-- theorem-1: body_md -->
\label{thm:hilbert}
Let $H$ be a real Hilbert space with $\dim H\geq2$.  For every $\tau\geq0$
and $1\leq t\leq\infty$,
\begin{align}
 \mathcal S_{H,t}(\tau)
   &=\min\left\{\sqrt{1+\tau^2},
           \mathcal M_t(1+\tau,\left\lvert 1-\tau\right\rvert)\right\},\label{eq:Hilbert-S}\\
 \mathcal J_t[\tau,H]
   &=\max\left\{\sqrt{1+\tau^2},
           \mathcal M_t(1+\tau,\left\lvert 1-\tau\right\rvert)\right\}.
           \label{eq:Hilbert-J}
\end{align}
Equivalently,
\[
\mathcal S_{H,t}(\tau)=
\begin{cases}
 \mathcal M_t(1+\tau,\left\lvert 1-\tau\right\rvert),&1\leq t\leq2,\\[2pt]
 \sqrt{1+\tau^2},&2\leq t\leq\infty,
\end{cases}
\]
and the two cases are reversed for $\mathcal J_t[\tau,H]$.
<!-- end theorem-1 -->
<!-- proof-2: body_md -->
For $x,y\in S_H$, set $s=\left\langle x,y\right\rangle\in[-1,1]$.  Then
\begin{align*}
 \left\lVert x+\tau y\right\rVert^2&=1+\tau^2+2\tau s,\\
 \left\lVert \tau x-y\right\rVert^2&=1+\tau^2-2\tau s.
\end{align*}
Every $s\in[-1,1]$ occurs because $\dim H\geq2$.  If $1\leq t<\infty$,
the $t$th power of the profile is therefore
\begin{equation}\label{eq:hilbert-profile}
 \frac{(c_{\tau}+2\tau s)^{t/2}+(c_{\tau}-2\tau s)^{t/2}}{2}.
\end{equation}
For $1\leq t<2$, the two summands in \eqref{eq:hilbert-profile} are
concave as functions of their nonnegative arguments; the even profile is
maximal at $s=0$ and minimal at $s=\pm1$.  For $t>2$, convexity reverses
the two extrema.  At $t=2$ the profile is identically $c_{\tau}$.  The
endpoint $t=\infty$ follows directly by minimizing or maximizing
\[
 \max\{(c_{\tau}+2\tau s)^{1/2},
        (c_{\tau}-2\tau s)^{1/2}\}.
\]
This proves the piecewise formulas.  The min--max forms
\eqref{eq:Hilbert-S}--\eqref{eq:Hilbert-J} also follow from the power-mean
comparison with the quadratic mean.

<!-- theorem-2: body_md -->
\label{thm:Lp}
Let $1\leq p\leq\infty$, and let $X=L_p(\Omega,\mu)$ have dimension at
least two.  Then
\begin{equation}\label{eq:Lp-main}
 \
 \mathcal S_{X,p}(\tau)=m_p(\tau)=\frac{1+\tau^2}{\nu_p(\tau)},
 \qquad
 \mathcal J_p[\tau,X]=\nu_p(\tau).
\end{equation}
In particular,
\begin{equation}\label{eq:product-identity}
  \mathcal S_{X,p}(\tau)\,\mathcal J_p[\tau,X]=1+\tau^2.
\end{equation}
Here the convention for $p=\infty$ is that the outer mean is the maximum.
<!-- end theorem-2 -->
<!-- proof-3: body_md -->
Assume first that $1\leq p<\infty$.  The scalar minimum-modulus inequality
gives, almost everywhere,
\[
 |x+\tau y|^p+|\tau x-y|^p
 \geq m_p(\tau)^p\bigl(|x|^p+|y|^p\bigr).
\]
After integration, every $x,y\in S_X$ satisfy
\[
 \frac{\left\lVert x+\tau y\right\rVert_p^p+\left\lVert \tau x-y\right\rVert_p^p}{2}
 \geq m_p(\tau)^p.
\]
The operator-norm inequality gives the analogous upper estimate with
$\nu_p(\tau)^p$.  Hence
\begin{equation}\label{eq:Lp-prebounds}
 \mathcal S_{X,p}(\tau)\geq m_p(\tau),
 \qquad
 \mathcal J_p[\tau,X]\leq\nu_p(\tau).
\end{equation}

Because $\dim L_p(\mu)\geq2$, there exist disjointly supported unit vectors
$f,g\in L_p(\mu)$.  Let $z=(a,b)\in\ell_p^2$ be a unit vector.  Define
\begin{equation}\label{eq:symmetrizing-vectors}
  x=af+bg,
  \qquad
  y=bf-ag.
\end{equation}
Then $\left\lVert x\right\rVert_p=\left\lVert y\right\rVert_p=1$, and disjointness gives
\begin{align*}
 \left\lVert x+\tau y\right\rVert_p^p
 &=|a+\tau b|^p+|b-\tau a|^p,\\
 \left\lVert \tau x-y\right\rVert_p^p
 &=|\tau a-b|^p+|\tau b+a|^p.
\end{align*}
The two right-hand sides are equal to $\left\lVert A_\tau z\right\rVert_p^p$.  Choosing $z$
to minimize and then to maximize the scalar ratio shows that both estimates
in \eqref{eq:Lp-prebounds} are attained.  Formula
\eqref{eq:minmod-involution} yields \eqref{eq:Lp-main} and
\eqref{eq:product-identity}.

For $p=\infty$, apply the pointwise inequality
\[
 \max\{|x+\tau y|,|\tau x-y|\}
 \geq m_\infty(\tau)\max\{|x|,|y|\}
\]
and its operator-norm counterpart.  Taking essential suprema proves the
analogues of \eqref{eq:Lp-prebounds}.  The same disjoint-support
construction \eqref{eq:symmetrizing-vectors}, now with
$\max\{|a|,|b|\}=1$, attains the two scalar extrema.

<!-- remark-1: body_md -->
\label{rem:balance}
The operator minimum modulus is taken over arbitrary scalar pairs $(a,b)$,
whereas \eqref{eq:def-S-intro} requires two vectors of equal norm.  The
second coordinate pair $(b,-a)$ in \eqref{eq:symmetrizing-vectors} is not
cosmetic: it interchanges, up to signs, the two coordinates of
$A_\tau(a,b)$.  It therefore balances the input norms without changing the
output ratio.  This is the step that upgrades an operator estimate to an
exact sphere formula.

<!-- corollary-1: body_md -->
\label{cor:distinguished}
Let $X=L_p(\mu)$ have dimension at least two.
\begin{enumerate}[label=\textup{(\roman*)}]
 \item If $p=1$ or $p=\infty$, then
 \[
   \mathcal S_{X,p}(\tau)=\frac{1+\tau^2}{1+\tau},
   \qquad
   \mathcal J_p[\tau,X]=1+\tau.
 \]
 \item If $p=2$, then
 \[
   \mathcal S_{X,2}(\tau)=\mathcal J_2[\tau,X]=\sqrt{1+\tau^2}.
 \]
 \item If $\tau=1$, then
 \begin{align*}
   \mathcal J_p[1,X]&=2^{\max\{1/p,\,1-1/p\}},\\
   \mathcal S_{X,p}(1)&=2^{\min\{1/p,\,1-1/p\}}.
 \end{align*}
 \item If $p'=p/(p-1)$, with the usual endpoint convention, then
 \[
   \mathcal S_{L_p,p}(\tau)=\mathcal S_{L_{p'},p'}(\tau),
   \qquad
   \mathcal J_p[\tau,L_p]=\mathcal J_{p'}[\tau,L_{p'}].
 \]
\end{enumerate}
<!-- end corollary-1 -->
<!-- proof-4: body_md -->
The $\ell_1^2$ and $\ell_\infty^2$ matrix norms are the maximum column and
row sums, respectively, both equal to $1+\tau$.  The Euclidean identity
$A_\tau^2=c_{\tau} I_2$ gives $\nu_2(\tau)=\sqrt{c_{\tau}}$.  When $\tau=1$,
interpolation between $p=1$ and $p=2$, together with the test vector
$(1,0)$, gives $\nu_p(1)=2^{1/p}$ for $1\leq p\leq2$.  Duality gives
$\nu_p(1)=2^{1-1/p}$ for $2\leq p\leq\infty$.  Finally,
$A_\tau$ is symmetric, so its $\ell_p^2$ norm equals its
$\ell_{p'}^2$ norm.  Apply \cref{thm:Lp}.

<!-- example-1: body_md -->
\label{ex:Banas-Fraczek}
Let $\lambda>1$, and consider the classical Bana\'s--Fr\k{a}czek space
\[
 X_\lambda=(\mathbb R^2,\left\lVert \cdot\right\rVert_\lambda),
 \qquad
 \left\lVert (u,v)\right\rVert_\lambda
 =\max\left\{\lambda |u|,\sqrt{u^2+v^2}\right\};
\]
see \cite{YangYangBF}.  Then, for the genuinely skew range
\[
 \sqrt{1-\lambda^{-2}}\leq \tau\leq1,
\]
one has the exact formula
\begin{equation}\label{eq:Banas-Fraczek-S}
 \
 \mathcal S_{X_\lambda,2}(\tau)
 =\left(\frac{(1+\tau^2)(1+\lambda^{-2})}{2}\right)^{1/2}.
\end{equation}

Indeed, write $x=(a,b)$ and $y=(c,d)$ with
$\left\lVert x\right\rVert_\lambda=\left\lVert y\right\rVert_\lambda=1$.  By the definition of the norm,
\[
 \max\{\lambda^2a^2,a^2+b^2\}=1,
 \qquad
 \max\{\lambda^2c^2,c^2+d^2\}=1.
\]
Moreover,
\begin{align*}
 &\left\lVert x+\tau y\right\rVert_\lambda^2+\left\lVert \tau x-y\right\rVert_\lambda^2\\
 &\quad\geq
 \max\Bigl\{
   \lambda^2\bigl((a+\tau c)^2+(\tau a-c)^2\bigr),
   \left\lVert x+\tau y\right\rVert_2^2+\left\lVert \tau x-y\right\rVert_2^2
 \Bigr\}\\
 &\quad=
 (1+\tau^2)
 \max\Bigl\{
   \lambda^2(a^2+c^2),
   \left\lVert x\right\rVert_2^2+\left\lVert y\right\rVert_2^2
 \Bigr\}.
\end{align*}
The unit-sphere conditions imply
\[
 \max\Bigl\{
   \lambda^2(a^2+c^2),
   \left\lVert x\right\rVert_2^2+\left\lVert y\right\rVert_2^2
 \Bigr\}
 \geq1+\lambda^{-2}.
\]
Hence
\[
 \mathcal S_{X_\lambda,2}(\tau)
 \geq
 \left(\frac{(1+\tau^2)(1+\lambda^{-2})}{2}\right)^{1/2}.
\]

For the reverse inequality, take
\[
 x=\left(\frac1\lambda,0\right),
 \qquad y=(0,1).
\]
Then $x,y\in S_{X_\lambda}$.  Since
$\tau^2\geq1-\lambda^{-2}$ and $0\leq\tau\leq1$,
\[
 \left\lVert x+\tau y\right\rVert_\lambda^2=\lambda^{-2}+\tau^2,
 \qquad
 \left\lVert \tau x-y\right\rVert_\lambda^2=1+\tau^2\lambda^{-2}.
\]
Therefore
\[
 \frac{\left\lVert x+\tau y\right\rVert_\lambda^2+\left\lVert \tau x-y\right\rVert_\lambda^2}{2}
 =\frac{(1+\tau^2)(1+\lambda^{-2})}{2},
\]
which proves \eqref{eq:Banas-Fraczek-S}.

The threshold $\tau=\sqrt{1-\lambda^{-2}}$ is intrinsic to this example:
for the extremizing pair above, it is precisely the point at which the
Euclidean component of $\left\lVert x+\tau y\right\rVert_\lambda$ becomes dominant.  Thus
\eqref{eq:Banas-Fraczek-S} exhibits a genuine interaction between the skew
parameter and the non-Euclidean geometry of the Bana\'s--Fr\k{a}czek unit ball.

<!-- theorem-3: body_md -->
\label{thm:sharp-chain}
For every $\tau\geq0$ and $1\leq t\leq\infty$,
\begin{equation}\label{eq:sharp-chain}
  r_{\tau}
  \leq \mathcal S_{X,t}(\tau)
  \leq h_{t}(\tau)
  \leq \mathcal J_t[\tau,X]
  \leq 1+\tau.
\end{equation}
Every bound in \eqref{eq:sharp-chain} is sharp over the class of nonzero
real Banach spaces.
<!-- end theorem-3 -->
<!-- proof-5: body_md -->
Fix $x,y\inS_X$ and write
\[
  u=x+\tau y,
  \qquad v=\tau x-y,
  \qquad a=\left\lVert u\right\rVert,\quad b=\left\lVert v\right\rVert.
\]
The involution identity \eqref{eq:A-intro} reads
\begin{equation}\label{eq:inverse-pair}
  u+\tau v=c_{\tau} x,
  \qquad
  \tau u-v=c_{\tau} y.
\end{equation}
Consequently,
\[
  c_{\tau}\leq a+\tau b,
  \qquad
  c_{\tau}\leq\tau a+b.
\]
Adding these inequalities and using the monotonicity of power means gives
\[
  \mathcal M_t(a,b)\geq\mathcal M_1(a,b)=\frac{a+b}{2}
  \geq\frac{c_{\tau}}{1+\tau}=r_{\tau}.
\]
Taking the infimum proves the first inequality.

Choosing $y=x$ gives the pair of norms $1+\tau,\left\lvert 1-\tau\right\rvert$, and hence
$\mathcal S_{X,t}(\tau)\leqh_{t}(\tau)\leq\mathcal J_t[\tau,X]$.  Finally, both entries in
the defining power mean are at most $1+\tau$, which proves the last bound.

It remains to verify sharpness.  In a one-dimensional real space every
pair of unit vectors is $(x,x)$ or $(x,-x)$, so
$\mathcal S_{X,t}(\tau)=\mathcal J_t[\tau,X]=h_{t}(\tau)$; this makes both middle bounds
sharp.  For the lower bound, first suppose $0\leq\tau\leq1$ and take
$X=\ell_1^2$.  With its canonical basis $(e_1,e_2)$, put
\[
 x=\frac{e_1+\tau e_2}{1+\tau},
 \qquad
 y=\frac{\tau e_1-e_2}{1+\tau}.
\]
Then $x,y\inS_X$ and
\[
 x+\tau y=r_{\tau} e_1,
 \qquad
 \tau x-y=r_{\tau} e_2.
\]
Thus $\mathcal S_{\ell_1^2,t}(\tau)=r_{\tau}$.  The case $\tau>1$ follows from
\cref{prop:structure}(ii).  Finally, in $\ell_1^2$ the choice
$x=e_1,y=e_2$ makes both output norms equal to $1+\tau$, proving sharpness
of the rightmost bound.

<!-- remark-2: body_md -->
\label{rem:involution-vs-classical}
For the classical pair $x+\tau y,x-\tau y$, independent scaling leads to
an even convex sum without further modification.  For the skew pair, that
property fails because the coefficient $\tau$ changes sides.  Identity
\eqref{eq:inverse-pair} is the replacement: it gives the sharp lower scale
$r_{\tau}$, but later arguments requiring homogeneous normalization must first
be symmetrized.  This distinction is responsible for the main proof in
\cref{sec:normal}.

<!-- theorem-4: body_md -->
\label{thm:gap-characterization}
Let $\tau>0$ and $1\leq t\leq\infty$, and exclude only the pair
$(t,\tau)=(1,1)$.  Then the following are equivalent:
\begin{enumerate}[label=\textup{(\roman*)}]
 \item $X$ is uniformly nonsquare;
 \item $\displaystyle
       \mathcal S_{X,t}(\tau)>\frac{1+\tau^2}{1+\tau}$.
\end{enumerate}
Consequently, for any fixed $t>1$, strictness at one positive value of
$\tau$ is equivalent to strictness at every positive value of $\tau$.
<!-- end theorem-4 -->
<!-- proof-7: body_md -->
By \cref{prop:structure}(ii), it suffices to consider $0<\tau\leq1$.  We
first prove the contrapositive of (i)$\Rightarrow$(ii).  Suppose that $X$
is not uniformly nonsquare.  There are $p_n,q_n\inS_X$ such that
\begin{equation}\label{eq:square-sequences}
 \left\lVert p_n+q_n\right\rVert\longrightarrow2,
 \qquad
 \left\lVert p_n-q_n\right\rVert\longrightarrow2.
\end{equation}
For $0<\tau\leq1$, the triangle inequality and \eqref{eq:square-sequences}
give
\begin{align*}
 \left\lVert p_n+\tau q_n\right\rVert
 &\geq\left\lVert p_n+q_n\right\rVert-(1-\tau)\longrightarrow1+\tau,\\
 \left\lVert \tau p_n-q_n\right\rVert
 &\geq\left\lVert p_n-q_n\right\rVert-(1-\tau)\longrightarrow1+\tau.
\end{align*}
The opposite inequalities are automatic, so both limits equal $1+\tau$.
Put
\[
  x_n=\frac{p_n+\tau q_n}{1+\tau},
  \qquad
  y_n=\frac{\tau p_n-q_n}{1+\tau}.
\]
Then $\left\lVert x_n\right\rVert,\left\lVert y_n\right\rVert\to1$, while the involution identity gives
\begin{equation}\label{eq:almost-lower-pair}
 x_n+\tau y_n=r_{\tau} p_n,
 \qquad
 \tau x_n-y_n=r_{\tau} q_n.
\end{equation}
Normalize $x_n$ and $y_n$ separately.  Since the normalization errors tend
to zero, \eqref{eq:almost-lower-pair} produces unit pairs whose two output
norms converge to $r_{\tau}$.  Therefore
$\mathcal S_{X,t}(\tau)\leqr_{\tau}$, and equality follows from
\cref{thm:sharp-chain}.

Conversely, suppose $\mathcal S_{X,t}(\tau)=r_{\tau}$.  Choose $x_n,y_n\inS_X$ and
set
\[
 u_n=x_n+\tau y_n,
 \quad v_n=\tau x_n-y_n,
 \quad a_n=\left\lVert u_n\right\rVert,
 \quad b_n=\left\lVert v_n\right\rVert,
\]
so that $\mathcal M_t(a_n,b_n)\tor_{\tau}$.  The proof of
\cref{thm:sharp-chain} shows
\begin{equation}\label{eq:sum-convergence}
 r_{\tau}\leq\frac{a_n+b_n}{2}
 \leq\mathcal M_t(a_n,b_n)\longrightarrowr_{\tau}.
\end{equation}
It also gives the nonnegative errors
\begin{align*}
 e_n&:=a_n+\tau b_n-c_{\tau}\geq0,\\
 d_n&:=\tau a_n+b_n-c_{\tau}\geq0,
\end{align*}
whose sum is
\[
 e_n+d_n=(1+\tau)(a_n+b_n)-2c_{\tau}\longrightarrow0.
\]
If $0<\tau<1$, subtracting the two relations shows
$(1-\tau)(a_n-b_n)=e_n-d_n\to0$.  Thus
$a_n,b_n\tor_{\tau}$.  If $\tau=1$ and $1<t<\infty$, the same conclusion
follows from \eqref{eq:sum-convergence} and the strict convexity of the
scalar $\ell_t^2$ norm: the only nonnegative pair with arithmetic mean and
$t$-mean both equal to $1$ is $(1,1)$.  For $t=\infty$, it follows directly
from $\max\{a_n,b_n\}\to1$ and $a_n+b_n\geq2$.  This is the point at which
the excluded case $(t,\tau)=(1,1)$ would fail.

For large $n$, define $p_n=u_n/a_n$ and $q_n=v_n/b_n$.  Choose
$f_n,g_n\in S_{X^*}$ norming the two identities in
\eqref{eq:inverse-pair}, so that
\begin{align*}
 c_{\tau}
 &=f_n(u_n+\tau v_n)
   =a_n f_n(p_n)+\tau b_n f_n(q_n),\\
 c_{\tau}
 &=g_n(\tau u_n-v_n)
   =\tau a_n g_n(p_n)-b_n g_n(q_n).
\end{align*}
Since $e_n,d_n\to0$ and $a_n,b_n\tor_{\tau}>0$, we obtain
\[
 f_n(p_n)\to1,
 \quad f_n(q_n)\to1,
 \qquad
 g_n(p_n)\to1,
 \quad g_n(q_n)\to-1.
\]
Hence
\[
 \left\lVert p_n+q_n\right\rVert\geq f_n(p_n+q_n)\longrightarrow2,
 \qquad
 \left\lVert p_n-q_n\right\rVert\geq g_n(p_n-q_n)\longrightarrow2.
\]
Thus $J(X)=2$, so $X$ is not uniformly nonsquare.  This proves the reverse
contrapositive and completes the equivalence.

\label{ex:gap-models}
For $1<p<\infty$, $L_p$ is uniformly nonsquare \cite{Clarkson}, and hence

<!-- lemma-1: body_md -->
\label{lem:lifting}
Let $z,w\in X\setminus\{0\}$ and
$m=\min\{\left\lVert z\right\rVert,\left\lVert w\right\rVert\}$.  Then
\begin{equation}\label{eq:lifting}
 \Gamma_{\tau,t}(z,w)
 \geq4m^t\mathcal S_{X,t}(\tau)^t.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-9: body_md -->
Write $z=\alpha p,w=\beta q$ with $p,q\inS_X$ and
$\alpha,\beta>0$.  For fixed $\beta$, each of
\begin{align*}
 s&\longmapsto\left\lVert sp+\tau\beta q\right\rVert^t
                  +\left\lVert sp-\tau\beta q\right\rVert^t,\\
 s&\longmapsto\left\lVert \beta q+\tau sp\right\rVert^t
                  +\left\lVert \beta q-\tau sp\right\rVert^t
\end{align*}
is even and convex.  It is therefore nondecreasing on $[0,\infty)$.  The
same statement holds when the roles of the two scaling variables are
interchanged.  Since $\alpha,\beta\geq m$,
\[
 \Gamma_{\tau,t}(\alpha p,\beta q)
 \geq\Gamma_{\tau,t}(mp,mq)
 =m^t\Gamma_{\tau,t}(p,q).
\]
For unit $p,q$, the four terms split into the two skew orientations:
\begin{align*}
 \Gamma_{\tau,t}(p,q)
  =&\ \bigl(\left\lVert p+\tau q\right\rVert^t+\left\lVert \tau p-q\right\rVert^t\bigr)\\
   &+\bigl(\left\lVert q+\tau p\right\rVert^t+\left\lVert \tau q-p\right\rVert^t\bigr).
\end{align*}
Each parenthesis is at least $2\mathcal S_{X,t}(\tau)^t$, proving
\eqref{eq:lifting}.
<!-- end proof-9 -->
<!-- theorem-5: body_md -->
\label{thm:James-estimate}
Let $0\leq\tau,\varepsilon\leq1$ and $1\leq t<\infty$.  Set
\begin{equation}\label{eq:AB}
 A_{\tau,\varepsilon}
 :=1+\tau+\varepsilon(1-\tau),
 \qquad
 B_{\tau,\varepsilon}
 :=1-\tau+\varepsilon(1+\tau).
\end{equation}
Then
\begin{equation}\label{eq:James-main}
 2\mathcal S_{X,t}(\tau)^tJ(\varepsilon,X)^t
 \leq A_{\tau,\varepsilon}^t+B_{\tau,\varepsilon}^t.
\end{equation}
Equivalently,
\begin{equation}\label{eq:James-mean}
 \mathcal S_{X,t}(\tau)J(\varepsilon,X)
 \leq\mathcal M_t(A_{\tau,\varepsilon},B_{\tau,\varepsilon}).
\end{equation}
The constant in \eqref{eq:James-main} is sharp: equality holds in
$X=\ell_1^2$ when $\tau=\varepsilon=1$.
<!-- end theorem-5 -->
<!-- proof-10: body_md -->
Fix $x,y\inS_X$ and apply \cref{lem:lifting} to
\[
  z=x+\varepsilon y,
  \qquad
  w=x-\varepsilon y.
\]
If $z=0$ or $w=0$, the pointwise estimate below is immediate because its
left-hand side contains $\min\{\left\lVert z\right\rVert,\left\lVert w\right\rVert\}^t=0$.  We may therefore
assume that both vectors are nonzero and invoke the lemma.
The four vectors in \eqref{eq:Gamma} are
\begin{align*}
 z+\tau w&=(1+\tau)x+\varepsilon(1-\tau)y,\\
 z-\tau w&=(1-\tau)x+\varepsilon(1+\tau)y,\\
 w+\tau z&=(1+\tau)x-\varepsilon(1-\tau)y,\\
 w-\tau z&=(1-\tau)x-\varepsilon(1+\tau)y.
\end{align*}
Their norms are bounded, in order, by
$A_{\tau,\varepsilon},B_{\tau,\varepsilon},
 A_{\tau,\varepsilon},B_{\tau,\varepsilon}$.  Consequently,
\begin{align*}
 4\mathcal S_{X,t}(\tau)^t
 \min\{\left\lVert x+\varepsilon y\right\rVert,\left\lVert x-\varepsilon y\right\rVert\}^t
 &\leq\Gamma_{\tau,t}(z,w)\\
 &\leq2A_{\tau,\varepsilon}^t
       +2B_{\tau,\varepsilon}^t.
\end{align*}
Take the supremum over $x,y\inS_X$ to obtain
\eqref{eq:James-main}.

For sharpness, $\mathcal S_{\ell_1^2,t}(1)=1$ by
\cref{thm:sharp-chain}, while $J(1,\ell_1^2)=2$.  At
$\tau=\varepsilon=1$, both quantities in \eqref{eq:AB} equal $2$, so the
two sides of \eqref{eq:James-main} are equal to $2^{t+1}$.

<!-- corollary-5: body_md -->
\label{cor:diagonal}
For $0\leq\tau\leq1$ and $1\leq t<\infty$,
\begin{equation}\label{eq:diagonal}
 2\mathcal S_{X,t}(\tau)^tJ(\tau,X)^t
 \leq(1+2\tau-\tau^2)^t+(1+\tau^2)^t.
\end{equation}
For $t=\infty$, the limiting forms of
\eqref{eq:James-mean} and \eqref{eq:diagonal} are, respectively,
\begin{align*}
 \mathcal S_{X,\infty}(\tau)J(\varepsilon,X)
 &\leq A_{\tau,\varepsilon},\\
 \mathcal S_{X,\infty}(\tau)J(\tau,X)
 &\leq1+2\tau-\tau^2.
\end{align*}
<!-- end corollary-5 -->
<!-- proof-11: body_md -->
Set $\varepsilon=\tau$ in \cref{thm:James-estimate}.  The limiting
statements follow from \cref{cor:t-infty-limit} and the convergence of
power means.  Notice that
$A_{\tau,\varepsilon}\geq B_{\tau,\varepsilon}$ on the stated parameter
range.

<!-- corollary-6: body_md -->
\label{cor:normal-criterion}
Let $0\leq\tau\leq1$, $0<\varepsilon\leq1$, and
$1\leq t\leq\infty$.  If
\begin{equation}\label{eq:normal-general}
 \mathcal S_{X,t}(\tau)
 >\frac{\mathcal M_t(A_{\tau,\varepsilon},B_{\tau,\varepsilon})}
         {g(\varepsilon)},
\end{equation}
then $X$ has uniform normal structure.  In particular, it is enough that
for some $0<\tau\leq1$,
\begin{equation}\label{eq:normal-diagonal}
 \mathcal S_{X,t}(\tau)
 >\frac{1}{g(\tau)}
  \left(\frac{(1+2\tau-\tau^2)^t+(1+\tau^2)^t}{2}\right)^{1/t}
\end{equation}
when $t<\infty$, with the maximum replacing the power mean for
$t=\infty$.
<!-- end corollary-6 -->
<!-- proof-12: body_md -->
Under \eqref{eq:normal-general}, \cref{thm:James-estimate} gives
$J(\varepsilon,X)<g(\varepsilon)$.  Apply Saejung's criterion
\eqref{eq:Saejung}.  The diagonal statement follows by taking
$\varepsilon=\tau$.

<!-- remark-4: body_md -->
\label{rem:Gao-threshold}
At $\tau=\varepsilon=1$ and $t=2$, condition
\eqref{eq:normal-general} becomes
\[
 \mathcal S_{X,2}(1)>\frac{2}{(1+\sqrt5)/2}.
\]
If Gao's quadratic parameter is written as
$f(X)=2\mathcal S_{X,2}(1)^2$, this is exactly
\[
 f(X)>12-4\sqrt5,
\]
the familiar Pythagorean threshold discussed in
\cite{Gao,Saejung,ZuoTang}.  The two-parameter form
\eqref{eq:normal-general} is stronger as a certificate: one may keep the
skew parameter $\tau$ independent of the James scale $\varepsilon$ and
optimize the resulting bound for the space under consideration.
<!-- end remark-4 -->
<!-- remark-5: body_md -->
The supremal invariant \eqref{eq:skew-james-intro} and the infimal invariant
\eqref{eq:def-S-intro} coincide in Hilbert space at the quadratic exponent,
but they encode opposite operator radii on $L_p$.  The product identity
\eqref{eq:product-identity}, the gap characterization
\cref{thm:gap-characterization}, and the symmetrized estimate
\cref{thm:James-estimate} show that the infimum theory is not a formal
restatement of the skew James theory.  Rather, the two theories are linked
by the scaled involution $A_\tau^2=(1+\tau^2)I$, which is the common
algebraic source of their exact model behavior.
