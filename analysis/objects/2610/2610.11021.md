		\[
		F_{1,L}(W)
		=
		\nu(L^*WL)
		=
		\nu(LL^*W),
		\]
		which is linear in $W$. We therefore assume throughout the rest of
		the proof that
		$
		1<s\le2.
		$
		
		We divide the proof into three steps.
		
		\smallskip
		\noindent
		\emph{Step 1: an Ando-type jointly convex functional.}
Equip $L^2(\mathcal N,\nu)$ with the inner product
\[
\langle Y,Z\rangle_2:=\nu(Y^*Z).
\]
Since $\nu(\mathbf 1)<\infty$ and $L\in\mathcal N$, we have
\[
\|L\|_2^2
=
\nu(L^*L)
\le
\|L\|_\infty^2\nu(\mathbf 1)
<\infty.
\]
Thus $L\in L^2(\mathcal N,\nu)$.

For $W,X\in\mathcal N^{++}$, define the left- and
right-multiplication operators on $L^2(\mathcal N,\nu)$ by
\[
\lambda_W(Y):=WY,
\qquad
\rho_X(Y):=YX,
\qquad Y\in L^2(\mathcal N,\nu).
\]
		These operators are bounded and positive. Indeed, for every
$Y\in L^2(\mathcal N,\nu)$,
\[
\|\lambda_WY\|_2
\le \|W\|_\infty\|Y\|_2,
\qquad
\|\rho_XY\|_2
\le \|X\|_\infty\|Y\|_2,
\]

\begin{align*}
	\langle Y,\lambda_WY\rangle_2
	&=
	\nu(Y^*WY)
	=
	\|W^{1/2}Y\|_2^2,\\
	\langle Y,\rho_XY\rangle_2
	&=
	\nu(Y^*YX)
	=
	\|YX^{1/2}\|_2^2.
\end{align*}
Since $W,X\in\mathcal N^{++}$, there exist
$\varepsilon_W,\varepsilon_X>0$ such that
\[
W\ge\varepsilon_W\mathbf 1,
\qquad
X\ge\varepsilon_X\mathbf 1.
\]
It follows that
\[
\lambda_W\ge\varepsilon_W I_{L^2},
\qquad
\rho_X\ge\varepsilon_X I_{L^2},
\]
so both operators are invertible. Here and below, real powers of
positive invertible elements and operators are defined by functional
calculus. Its compatibility with left and right multiplication gives
\[
\lambda_W^a=\lambda_{W^a},
\qquad
\rho_X^a=\rho_{X^a},
\qquad a\in\mathbb R.
\]
Every left-multiplication operator commutes with every
right-multiplication operator:
\[
\lambda_W\rho_X(Y)
=
WYX
=
\rho_X\lambda_W(Y).
\]
In particular,
\begin{equation}\label{eq:cross-commutation}
	[\lambda_{W_i},\rho_{X_j}]=0,
	\qquad i,j\in\{1,2\}.
\end{equation}

We next recall the operator-perspective argument in the precise
Hilbert-space form needed here. Let
\[
f_s(u):=u^s,\qquad u>0.
\]
This function is operator convex for $1\le s\le2$.
Suppose that $P_i,Q_i$, $i=1,2$, are bounded positive
invertible operators on a Hilbert space such that
\[
[P_i,Q_j]=0,
\qquad i,j\in\{1,2\}.
\]
For $0<\theta<1$, put
\[
P_\theta:=\theta P_1+(1-\theta)P_2,
\qquad
Q_\theta:=\theta Q_1+(1-\theta)Q_2,
\]
and define
\[
C_1
:=
\theta^{1/2}Q_1^{1/2}Q_\theta^{-1/2},
\qquad
C_2
:=
(1-\theta)^{1/2}Q_2^{1/2}Q_\theta^{-1/2}.
\]
Then
\[
C_1^*C_1+C_2^*C_2=\mathbf 1
\]
and
\[
Q_\theta^{-1/2}P_\theta Q_\theta^{-1/2}
=
\sum_{i=1}^2
C_i^*
\bigl(Q_i^{-1/2}P_iQ_i^{-1/2}\bigr)
C_i.
\]
Effros formulated the corresponding perspective theorem in the
matrix setting \cite[Theorem~2.2]{Effros}. Since its proof is based on
the Hansen--Pedersen--Jensen inequality, we record the same argument
on the possibly infinite-dimensional Hilbert space under
consideration.

The Hansen--Pedersen operator Jensen inequality
\cite[Theorem~2.1(ii)]{HansenPedersen} gives
\[
f_s\!\left(
Q_\theta^{-1/2}P_\theta Q_\theta^{-1/2}
\right)
\le
\sum_{i=1}^2
C_i^*
f_s\!\left(
Q_i^{-1/2}P_iQ_i^{-1/2}
\right)
C_i.
\]
Taking the congruence by $Q_\theta^{1/2}$ preserves the operator
order.  Write
\[
\alpha_1:=\theta,
\qquad
\alpha_2:=1-\theta.
\]
The cross-commutation assumptions imply
\[
[P_\theta,Q_\theta]=0,
\qquad
[P_i,Q_i]=0,\quad i=1,2.
\]
Since $f_s(u)=u^s$, functional calculus therefore gives
\[
Q_\theta^{1/2}
f_s\!\left(
Q_\theta^{-1/2}P_\theta Q_\theta^{-1/2}
\right)
Q_\theta^{1/2}
=
P_\theta^sQ_\theta^{\,1-s}.
\]
Moreover, from
\[
C_i=\alpha_i^{1/2}Q_i^{1/2}Q_\theta^{-1/2}
\]
we obtain
\[
Q_\theta^{1/2}C_i^*
=
\alpha_i^{1/2}Q_i^{1/2},
\qquad
C_iQ_\theta^{1/2}
=
\alpha_i^{1/2}Q_i^{1/2},
\]
and hence
\[
\begin{aligned}
	&Q_\theta^{1/2}C_i^*
	f_s\!\left(
	Q_i^{-1/2}P_iQ_i^{-1/2}
	\right)
	C_iQ_\theta^{1/2}\\
	&\qquad=
	\alpha_iQ_i^{1/2}
	\left(Q_i^{-1/2}P_iQ_i^{-1/2}\right)^s
	Q_i^{1/2}
	=
	\alpha_iP_i^sQ_i^{\,1-s}.
\end{aligned}
\]
Consequently,
\begin{equation}\label{eq:commuting-perspective}
	P_\theta^sQ_\theta^{\,1-s}
	\le
	\theta P_1^sQ_1^{\,1-s}
	+
	(1-\theta)P_2^sQ_2^{\,1-s}.
\end{equation}
Thus the Hansen--Pedersen argument yields the dimension-free
Hilbert-space form of Effros's commuting-perspective inequality.
No finite-dimensional approximation is involved.
Now let $W_i,X_i\in\mathcal N^{++}$, $i=1,2$. Applying
\eqref{eq:commuting-perspective} with
\[
P_i=\lambda_{W_i},
\qquad
Q_i=\rho_{X_i},
\]
is legitimate by \eqref{eq:cross-commutation}. Since the assignments
$W\mapsto\lambda_W$ and $X\mapsto\rho_X$ are linear, we obtain
\begin{align*}
	&
	\lambda_{\theta W_1+(1-\theta)W_2}^{\,s}
	\rho_{\theta X_1+(1-\theta)X_2}^{\,1-s}
	\\
	&\qquad\le
	\theta\lambda_{W_1}^{\,s}\rho_{X_1}^{\,1-s}
	+
	(1-\theta)
	\lambda_{W_2}^{\,s}\rho_{X_2}^{\,1-s}.
\end{align*}
Hence
\[
(W,X)\longmapsto
\lambda_W^s\rho_X^{\,1-s}
\]
is jointly operator convex on
$\mathcal N^{++}\times\mathcal N^{++}$.

Evaluating this operator inequality at the fixed vector
$L\in L^2(\mathcal N,\nu)$, we obtain
\begin{align*}
	\left\langle
	L,\lambda_W^s\rho_X^{\,1-s}L
	\right\rangle_2
	&=
	\left\langle
	L,W^sLX^{1-s}
	\right\rangle_2\\
	&=
	\nu(L^*W^sLX^{1-s}).
\end{align*}
Consequently,
\begin{equation}\label{eq:Ando-form}
	(W,X)
	\longmapsto
	\nu(L^*W^sLX^{1-s})
\end{equation}
is jointly convex on
$\mathcal N^{++}\times\mathcal N^{++}$.

It remains to allow $W$ to be noninvertible. Let
$W_i\in\mathcal N_+$, $X_i\in\mathcal N^{++}$, $i=1,2$, and apply the
preceding inequality to

\qquad \varepsilon>0.
\]
Since
\[

+(1-\theta)(W_2+\varepsilon\mathbf 1)
=

\]
letting $\varepsilon\downarrow0$ and using the norm continuity
of $W\mapsto W^s$ on $\mathcal N_+$ proves that
\eqref{eq:Ando-form} is jointly convex on
\[

	\emph{Step 2: a tracial variational formula.}
	We claim that, for every $C\in\mathcal N_+$,
	\begin{equation}\label{eq:root-variational}
		\nu(C^{1/s})
		=
		\frac1s
		\inf_{B\in\mathcal N^{++}}
		\left\{
		\nu(CB^{1-s})+(s-1)\nu(B)

	\end{equation}
	
	Fix $B\in\mathcal N^{++}$ and define
	\[
	F_B(A):=\nu(A^sB^{1-s}),
	\qquad A\in\mathcal N_+.
	\]
	Here $B$ is kept fixed, and all derivatives below are taken with
	respect to the variable $A$.  The linear functional
	\[
	\ell_B(Y):=\nu(YB^{1-s})
	\]
	is positive.  Indeed, for $Y\in\mathcal N_+$, traciality gives
	\[
	\ell_B(Y)
	=
	\nu\!\left(
	B^{(1-s)/2}YB^{(1-s)/2}
	\right)
	\geq0.
	\]
	Since $u\mapsto u^s$ is operator convex, it follows that
	$F_B=\ell_B\circ(u\mapsto u^s)$ is convex.
	
	Because $B$ is an interior point of $\mathcal N_+$, the
	supporting-hyperplane inequality at $A=B$ gives, for every
	$A\in\mathcal N_+$,
	\begin{equation}\label{eq:FB-supporting}
		F_B(A)
		\geq
		F_B(B)
		+
		DF_B(B)[A-B].
	\end{equation}
 
	
	We next compute the derivative in \eqref{eq:FB-supporting}.  Let
	$p_s(u):=u^s$.  Since $B\in\mathcal N^{++}$, holomorphic functional calculus
	gives, for $H\in\mathcal N_{\rm sa}$,
	\[
	D(p_s)_B[H]
	=
	\frac{1}{2\pi i}
	\int_\Gamma
	z^s(z\mathbf 1-B)^{-1}H(z\mathbf 1-B)^{-1}\,dz,
	\]
	where $\Gamma\subset\mathbb C\setminus(-\infty,0]$ is a positively oriented
	contour surrounding $\sigma(B)$.  Put
	\[
	R_z:=(z\mathbf 1-B)^{-1}.
	\]
	Since $B$ is fixed in the definition of $F_B$,
	\[
	DF_B(B)[H]
	=
	\nu\!\left(D(p_s)_B[H]B^{1-s}\right).
	\]
	The trace $\nu$ is a bounded linear functional because
	$\nu(\mathbf 1)<\infty$.  We may therefore pass it through the contour
	integral.  Using traciality and the fact that $R_z$ commutes with
	$B^{1-s}$, we obtain
	\begin{align*}
		DF_B(B)[H]
		&=
		\frac{1}{2\pi i}
		\int_\Gamma
		z^s\nu(R_zHR_zB^{1-s})\,dz\\
		&=
		\nu\!\left(
		HB^{1-s}
		\frac{1}{2\pi i}
		\int_\Gamma z^sR_z^2\,dz
		\right).
	\end{align*}
	Cauchy's differentiation formula gives
	\[
	\frac{1}{2\pi i}
	\int_\Gamma z^s(z\mathbf 1-B)^{-2}\,dz
	=
	sB^{s-1}.
	\]
	Consequently,
	\begin{align*}
		DF_B(B)[H]
		&=
		\nu\!\left(HB^{1-s}sB^{s-1}\right)\\
		&=
		s\nu(H).
	\end{align*}
	
 	Since $A-B\in\mathcal N_{\rm sa}$, the preceding derivative formula gives
	\[
	DF_B(B)[A-B]
	=
	s\nu(A-B).
	\]
	Substituting this identity and $F_B(B)=\nu(B)$ into
	\eqref{eq:FB-supporting}, we obtain
	\begin{align*}
		\nu(A^sB^{1-s})
		&\geq
		\nu(B)+s\nu(A-B)\\
		&=
		\nu(B)+s\nu(A)-s\nu(B)\\
		&=
		s\nu(A)+(1-s)\nu(B).
\end{align*}
	Equivalently,
	\begin{equation}\label{eq:tracial-Young}
		\nu(A^sB^{1-s})+(s-1)\nu(B)
		\geq
		s\nu(A).
	\end{equation}
	
	Set $A=C^{1/s}$, inequality
	\eqref{eq:tracial-Young} becomes
	\[
	\nu(CB^{1-s})+(s-1)\nu(B)
	\geq
	s\nu(C^{1/s})
	\]
	for every $B\in\mathcal N^{++}$.  Hence
	\[
	\frac1s
	\inf_{B\in\mathcal N^{++}}
	\left\{
	\nu(CB^{1-s})+(s-1)\nu(B)
	\right\}
	\geq
	\nu(C^{1/s}).
	\]
	
	Suppose first that $C$ is invertible.  Equality is attained by choosing
	$
	B=C^{1/s}.
	$
	
	It remains to consider a possibly noninvertible $C$.  Put
	\[
	A:=C^{1/s},
	\qquad
	B_\varepsilon:=A+\varepsilon\mathbf 1\in\mathcal N^{++},
	\qquad \varepsilon>0.
	\]
	Because $A$ and $B_\varepsilon$ commute, functional calculus gives
	\[
	A^sB_\varepsilon^{1-s}
	=
	A^s(A+\varepsilon\mathbf 1)^{1-s}.
	\]
	For $a\geq0$, the scalar inequality
	\[
	0
	\leq
	a-a^s(a+\varepsilon)^{1-s}
	\leq
	(s-1)\varepsilon
	\]
	implies, again by functional calculus,
	\[
	0
	\leq
	A-A^s(A+\varepsilon\mathbf 1)^{1-s}
	\leq
	(s-1)\varepsilon\mathbf 1.
	\]
	Applying $\nu$ and using $\nu(\mathbf 1)<\infty$, we obtain
	\[
	0
	\leq
	\nu(A)
	-
	\nu\!\left(A^s(A+\varepsilon\mathbf 1)^{1-s}\right)
	\leq
	(s-1)\varepsilon\nu(\mathbf 1).
	\]
	Therefore,
	\[
	\nu\!\left(A^s(A+\varepsilon\mathbf 1)^{1-s}\right)
	\longrightarrow
	\nu(A)
	\]
	as $\varepsilon\downarrow0$.  Moreover,
	\[
	\nu(B_\varepsilon)
	=
	\nu(A)+\varepsilon\nu(\mathbf 1)
	\longrightarrow
	\nu(A).
	\]
	Since $C=A^s$, it follows that
	\[
	\nu(CB_\varepsilon^{1-s})
	+
	(s-1)\nu(B_\varepsilon)
	\longrightarrow
	s\nu(A)
	=
	s\nu(C^{1/s}).
	\]
	Thus the infimum in \eqref{eq:root-variational} is at most
	$s\nu(C^{1/s})$.  Together with the opposite inequality proved above,
	this establishes \eqref{eq:root-variational} for every $C\in\mathcal N_+$.

		\noindent
		\emph{Step 3: convexity of $F_{s,L}$.}
		Applying \eqref{eq:root-variational} to
		\[
		C=L^*W^sL
		\]
		gives
		\begin{equation}\label{eq:CL-variational}
			F_{s,L}(W)
			=
			\frac1s
			\inf_{X\in\mathcal N^{++}}
			\Psi(W,X),
		\end{equation}
		where
		\[
		\Psi(W,X)
		:=
		\nu(L^*W^sLX^{1-s})+(s-1)\nu(X).
		\]
		The first term is jointly convex in $(W,X)$ by
		\eqref{eq:Ando-form}, while the second term is linear in $X$.
		Therefore $\Psi$ is jointly convex.
		
		We spell out why taking the infimum over $X$ preserves convexity.
		Let $W_1,W_2\in\mathcal N_+$, $0\le\theta\le1$, and choose arbitrary
		$X_1,X_2\in\mathcal N^{++}$. Since
		\[
		X_\theta:=\theta X_1+(1-\theta)X_2\in\mathcal N^{++},
		\]
		joint convexity gives
		\begin{align*}
			sF_{s,L}\bigl(\theta W_1+(1-\theta)W_2\bigr)
			&\le
			\Psi\bigl(
			\theta W_1+(1-\theta)W_2,X_\theta
			\bigr)\\
			&\le
			\theta\Psi(W_1,X_1)
			+
			(1-\theta)\Psi(W_2,X_2).
		\end{align*}
		Taking the infimum independently over $X_1$ and $X_2$, and using
		\eqref{eq:CL-variational}, yields
		\[
		F_{s,L}\bigl(\theta W_1+(1-\theta)W_2\bigr)
		\le
		\theta F_{s,L}(W_1)
		+
		(1-\theta)F_{s,L}(W_2).
		\]
		Thus $F_{s,L}$ is convex.
		
		Finally, for every $c\ge0$,
		\begin{align*}
			F_{s,L}(cW)
			&=
			\nu\!\left(
			\bigl(L^*(cW)^sL\bigr)^{1/s}
			\right)\\
			&=
			\nu\!\left(
			\bigl(c^sL^*W^sL\bigr)^{1/s}
			\right)\\
			&=
			cF_{s,L}(W).
		\end{align*}
		Hence $F_{s,L}$ is positively homogeneous.
		
		To prove \eqref{eq:CL-subadditivity}, put
		\[
		c:=\sum_{j=1}^m c_j.
		\]
		If $c=0$, the assertion is immediate. If $c>0$, convexity and
		positive homogeneity give
		\begin{align*}
			F_{s,L}\!\left(\sum_{j=1}^m c_jW_j\right)
			&=
			cF_{s,L}\!\left(
			\sum_{j=1}^m\frac{c_j}{c}W_j
			\right)\\
			&\le
			c\sum_{j=1}^m\frac{c_j}{c}F_{s,L}(W_j)\\
			&=
			\sum_{j=1}^m c_jF_{s,L}(W_j).
		\end{align*}
		This completes the proof.

	\[
	P_iP_jP_i=|\langle u_i,u_j\rangle|^2P_i,
	\]

	\[
	\mathcal K_r^{\operatorname{Tr}}(P_i,P_j)=|\langle u_i,u_j\rangle|^{2r}.
	\]

	\[
	\bigl[\mathcal K_r^{\operatorname{Tr}}(P_i,P_j)\bigr]_{i,j=0}^3
	=
	\begin{pmatrix}
		1&a_r&0&a_r\\
		a_r&1&a_r&0\\
		0&a_r&1&a_r\\
		a_r&0&a_r&1
	\end{pmatrix}.
	\]

	\[
	1-2a_r=1-2^{1-r},
	\]

	matrix is not positive semidefinite for any $1/2\le r<1$, even on pure
	qubit states, and the order-three conclusion of
	Theorem~\ref{thm:three-point} is sharp throughout this range. At $r=1$,
	by contrast, $\mathcal K_1^{\operatorname{Tr}}(P,Q)=\operatorname{Tr}(PQ)$ is a Hilbert--Schmidt Gram kernel

	mixed-sign vector in $\mathbb R^3$ has such a sign pattern, whereas four
	points already permit a two-versus-two pattern. Related higher-block
	compression obstructions appear in \cite{Audenaert,LinDriessche}.

		\qquad i=1,2,3.
		\]
		
		
			Applying Theorem~\ref{thm:three-point} in $e_n\mathcal M e_n$ with
			the restricted trace, $r=1/t$, and $P_i=A_{i,n}^{t/2}$,
			we obtain
		
		\[
		\bigl[Q_t(A_{i,n},A_{j,n})\bigr]_{i,j=1}^3\ge0.
		\]
		By the joint $L^1$-continuity from Lemma~\ref{lem:Qt-analytic},
		\[
		Q_t(A_{i,n},A_{j,n})
		\longrightarrow
		Q_t(A_i,A_j).
		\]
		Since the cone of positive semidefinite $3\times3$ matrices is closed,
		we obtain
		\[
		\bigl[Q_t(A_i,A_j)\bigr]_{i,j=1}^3\ge0.
		\]
		
		Now take $A_1=A$, $A_2=B$, and $A_3=C$. The preceding positivity
		gives vectors $\xi_1,\xi_2,\xi_3$ in a Hilbert space such that
		\[
		Q_t(A_i,A_j)=\langle \xi_i,\xi_j\rangle.
		\]
		Since $Q_t(A_i,A_i)=\tau(A_i)$, we have
		\[
		d_t(A_i,A_j)^2
		=
		\frac12\|\xi_i-\xi_j\|^2.
		\]
		Therefore
		\[
		d_t(A,C)
		=
		\frac1{\sqrt2}\|\xi_1-\xi_3\|
		\le
		\frac1{\sqrt2}\|\xi_1-\xi_2\|
		+
		\frac1{\sqrt2}\|\xi_2-\xi_3\|,
		\]
		that is,
		\[
		d_t(A,C)\le d_t(A,B)+d_t(B,C).
		\]
		This proves the triangle inequality.
		
		
			For the endpoint comparison, repeat the preceding approximation
			argument with
			\[
			S:=A+B,\qquad e_n:=\mathbf 1_{[1/n,n]}(S),
			\]
			and set
			\[
			A_n:=e_nAe_n,\qquad B_n:=e_nBe_n.
			\]
			As above,
			\[
			\tau(e_n)\le n\tau(S)<\infty,\qquad
			0\le A_n\le ne_n,\qquad 0\le B_n\le ne_n.
			\]
			Thus $A_n,B_n$ are bounded positive elements of the
			finite-tracial corner $e_n\mathcal M e_n$, equipped with the
			restricted trace. Moreover,
			\[
			\|A_n-A\|_1+\|B_n-B\|_1\longrightarrow0.
			\]
			Applying Corollary~\ref{cor:finite-realization} in this corner gives
			\[
			d_2(A_n,B_n)\le d_t(A_n,B_n)\le d_1(A_n,B_n).
			\]
			By the continuity of the trace on $L^1$,
			Lemma~\ref{lem:Qt-analytic}, and the definition of $d_s$,
			\[
			d_s(A_n,B_n)\longrightarrow d_s(A,B),
			\qquad s\in\{1,t,2\}.
			\]
			Passing to the limit yields
			\[
			d_2(A,B)\le d_t(A,B)\le d_1(A,B).
			\]
		
		These are the middle inequalities in \eqref{eq:endpoint-bounds}.
		The first inequality in \eqref{eq:endpoint-bounds} follows from
		$Q_2^\tau(A,B)\le\sqrt{ab}$.
		
		For $t=1$, traciality gives
		\[
		Q_1(A,B)=\tau(A^{1/2}B^{1/2}),
		\]
		and hence
		\[
		d_1(A,B)^2
		=
		\frac12\|A^{1/2}-B^{1/2}\|_2^2.
		\]
		The noncommutative $L^p$-extension of the Powers--St\o rmer
		inequality, applied with $p=1$ and $\theta=1/2$, gives
		\[
		\|A^{1/2}-B^{1/2}\|_2
		\leq
		\|A-B\|_1^{1/2};
		\]
		see \cite[Appendix, Theorem, p.~289]{HiaiNakamura}. Squaring this
		inequality and combining it with the preceding identity yields
		\eqref{eq:Powers-Stormer-bound}. The corresponding classical
		bounded-operator inequality appears in
		\cite[Lemma~4.1, p.~21]{PowersStormer}.
		
		It remains to prove \eqref{eq:FvdG-bound}. Let
		\[
		P:=\mathbf 1_{(0,\infty)}(A-B)
		\]
		and set
		\[
		a_1=\tau(PAP),\quad a_2=\tau((\mathbf 1-P)A(\mathbf 1-P)),\quad
		b_1=\tau(PBP),\quad b_2=\tau((\mathbf 1-P)B(\mathbf 1-P)).
		\]
		Let
		\[
		X:=A^{1/2}B^{1/2}\in L^1(\mathcal M,\tau),
		\]
		and let
		\[
		X=V|X|
		\]
		be its polar decomposition. By the polar decomposition theorem, $V$
		is a partial isometry with initial projection
		\[
		V^*V=s(|X|).
		\]
		Put $U:=V^*$. Then
		\[
		UU^*=V^*V=s(|X|)\leq\mathbf 1
		\]
		and
		\[
		UX=V^*V|X|=|X|.
		\]
		Hence, by adjoint invariance of the $L^1$-norm and traciality,
		\begin{align*}
			Q_2^\tau(A,B)
			&=\|B^{1/2}A^{1/2}\|_1\\
			&=\|A^{1/2}B^{1/2}\|_1\\
			&=\tau(UX)\\
			&=\tau(XU)\\
			&=\langle A^{1/2},B^{1/2}U\rangle_2\\
			&=\bigl|\langle A^{1/2},B^{1/2}U\rangle_2\bigr|.
		\end{align*}
		We now decompose the two $L^2$-elements according to $P$ and
		$\mathbf 1-P$:
		\[
		A^{1/2}
		=
		PA^{1/2}+(\mathbf 1-P)A^{1/2},
		\qquad
		B^{1/2}U
		=
		PB^{1/2}U+(\mathbf 1-P)B^{1/2}U.
		\]
		Since $P(\mathbf 1-P)=0$, the corresponding cross terms in the
		$L^2$-inner product vanish. Hence
		\begin{align*}
			\langle A^{1/2},B^{1/2}U\rangle_2
			&=
			\langle PA^{1/2},PB^{1/2}U\rangle_2\\
			&\quad+
			\langle(\mathbf 1-P)A^{1/2},
			(\mathbf 1-P)B^{1/2}U\rangle_2.
		\end{align*}
		Taking absolute values and applying the triangle inequality followed by
		the Hilbert-space Cauchy--Schwarz inequality, we obtain
		\begin{align*}
			Q_2^\tau(A,B)
			&\leq
			\|PA^{1/2}\|_2\,\|PB^{1/2}U\|_2\\
			&\quad+
			\|(\mathbf 1-P)A^{1/2}\|_2\,
			\|(\mathbf 1-P)B^{1/2}U\|_2.
		\end{align*}
		
		Moreover,
		\[
		\|PA^{1/2}\|_2^2=a_1,
		\qquad
		\|(\mathbf 1-P)A^{1/2}\|_2^2=a_2,
		\]
		while $UU^*\le\mathbf 1$ and traciality imply
		\[
		\begin{aligned}
			\|PB^{1/2}U\|_2^2
			=\tau\!\left(U^*B^{1/2}PB^{1/2}U\right)
			\le b_1,\quad
			\|(\mathbf 1-P)B^{1/2}U\|_2^2
			\le b_2.
		\end{aligned}
		\]
		Consequently,
		\[
		Q_2^\tau(A,B)
		\le
		\sqrt{a_1b_1}+\sqrt{a_2b_2}.
		\]
		Together with
		\[
		\|A-B\|_1=(a_1+b_2)-(b_1+a_2),
		\qquad
		a+b=(a_1+b_2)+(b_1+a_2),
		\]
		and
		\[
		\left(\sqrt{a_1b_1}+\sqrt{a_2b_2}\right)^2
		\le
		(a_1+b_2)(b_1+a_2),
		\]
		this yields
		\[
		\|A-B\|_1^2\le (a+b)^2-4Q_2^\tau(A,B)^2.
		\]
		The second inequality in \eqref{eq:FvdG-bound} follows from
		\begin{align*}
			(a+b)^2-4Q_2^\tau(A,B)^2
			&=(a+b-2Q_2^\tau(A,B))(a+b+2Q_2^\tau(A,B)),\\
			a+b-2Q_2^\tau(A,B)
			&=2d_2(A,B)^2,
		\end{align*}
		and $2Q_2^\tau(A,B)\le a+b$.
		
		The estimates also prove separation. Indeed, if $d_t(A,B)=0$,
		then $d_2(A,B)\leq d_t(A,B)$ implies $d_2(A,B)=0$. Hence
		\eqref{eq:FvdG-bound} gives
		\[
		\|A-B\|_1^2
		\leq
		4\bigl(\tau(A)+\tau(B)\bigr)d_2(A,B)^2
		=0,
		\]
		and therefore $A=B$.
		
		Let $(X_n)$ be $d_t$-Cauchy. By
		\eqref{eq:endpoint-bounds}, the sequence
		$\bigl(\sqrt{\tau(X_n)}\bigr)$ is Cauchy and hence bounded.
		Therefore, $\bigl(\tau(X_n)\bigr)$ is bounded. Since $d_2\le d_t$,
		\eqref{eq:FvdG-bound} implies that $(X_n)$ is $L^1$-Cauchy. Hence
		$X_n\to X\in L^1(\mathcal M,\tau)_+$. By
		\eqref{eq:Powers-Stormer-bound},
		\[
		d_1(X_n,X)\to0,
		\]
		and therefore $d_t(X_n,X)\to0$.
		
			Thus $(L^1(\mathcal M,\tau)_+,d_t)$ is complete.
		
		Conversely, $L^1$-convergence implies
		$d_t$-convergence by \eqref{eq:Powers-Stormer-bound} and $d_t\le d_1$.
		The same estimates also show that the $d_t$-Cauchy sequences and the
		$L^1$-Cauchy sequences coincide.
		
		
			Finally, suppose that $d_t(X_n,X)\to0$ for
			$X_n,X\in L^1(\mathcal M,\tau)_+$. By \eqref{eq:endpoint-bounds},
			\[
			\left|\sqrt{\tau(X_n)}-\sqrt{\tau(X)}\right|
			\le \sqrt2\,d_t(X_n,X)\longrightarrow0.
			\]
			Thus $\bigl(\tau(X_n)\bigr)$ is bounded. Using
			\eqref{eq:FvdG-bound} and $d_2\le d_t$, we obtain
			\[
			\|X_n-X\|_1^2
			\le
			4\bigl(\tau(X_n)+\tau(X)\bigr)d_t(X_n,X)^2
			\longrightarrow0.
			\]
			Together with the converse implication proved above, this shows
			that convergence in $d_t$ is equivalent to convergence in $L^1$.
			Since both topologies are metrizable, they coincide.

			d_t^{\operatorname{Tr}}(P,Q)^2=1-2^{-q}.

		\begin{equation}\label{eq:scalar-Clarkson}
			|a+b|^q+|a-b|^q
			\leq 2\bigl(|a|^s+|b|^s\bigr)^{q/s}.
		\end{equation}
		We verify strictness for $0<b<a$ using Friedrichs' argument
		as recalled in \cite[pp.~1364--1365]{MaligrandaSabourova}.

		\begin{aligned}
			F(\lambda)
			&=(1+\lambda^{1-q}z)(1+\lambda z)^{q-1}\\
			&\quad +(1-\lambda^{1-q}z)(1-\lambda z)^{q-1}.
		\end{aligned}
		\]
		Differentiation gives
		\[
		F'(\lambda)
		=(q-1)z(1-\lambda^{-q})
		\bigl((1+\lambda z)^{q-2}-(1-\lambda z)^{q-2}\bigr)<0
		\]
		for $z^{s-1}<\lambda<1$, since $q>2$.
		Using $(s-1)(q-1)=1$ and $q-1=q/s$, we obtain
		\[
		(1+z)^q+(1-z)^q
		=F(1)<F(z^{s-1})=2(1+z^s)^{q/s}.
		\]
		Multiplication by $a^q$ proves the required strictness.

		\eqref{eq:scalar-Clarkson} for $a,b\geq0$ holds exactly when
		$ab=0$ or $a=b$. In particular, $a=x=3/4$ and $b=y=1/4$

			1+2^{-q}<2(x^s+y^s)^{q/s}=2\mu.

		whose four positive parts belong to $L^2(\mathcal M,\tau)_+$.
		Their complex linear span is therefore all of $L^2(\mathcal M,\tau)$,
		which proves \eqref{eq:L2-core-condition}.

\label{cor:trace-GNS}

	Let $\tau$ be a faithful, densely defined, lower semicontinuous,
	semifinite trace on a $C^*$-algebra $\mathcal A$, where densely defined means
	that
	\[
	\overline{\mathcal A_{\tau,+}}^{\,\|\cdot\|}=\mathcal A_+,
	\]
	with $\|\cdot\|$ denoting the $C^*$-norm.
  Let
		\[
		\mathcal M_\tau:=\pi_\tau(\mathcal A)'',
		\]
		and let $\bar\tau$ be the canonical faithful normal semifinite extension.
		For every $1\le t\le2$, the metric completion of
		$(\mathcal A_{\tau,+},d_t^\tau)$ is canonically isometric to
		\begin{equation}\label{eq:trace-GNS-completion}
			\bigl(L^1(\mathcal M_\tau,\bar\tau)_+,d_t^{\bar\tau}\bigr).
		\end{equation}
		The completion of the trace-one part is
		\begin{equation}\label{eq:trace-GNS-states}
			\{\rho\in L^1(\mathcal M_\tau,\bar\tau)_+:\bar\tau(\rho)=1\}.
		\end{equation}

Let
		\[
		(\mathcal H_\tau,\pi_\tau,\Lambda_\tau)
		\]
		be the GNS construction associated with $\tau$. Since $\tau$
		is faithful, $\pi_\tau$ is faithful, and we identify $\mathcal A$
		with its image $\pi_\tau(\mathcal A)\subseteq\mathcal M_\tau$.
		
	The case $\mathcal A=\{0\}$ is immediate, so assume $\mathcal A\ne\{0\}$.
	The trace $\tau$ is a KMS weight with modular group
	$\sigma_s=\mathrm{id}_{\mathcal A}$, $s\in\mathbb R$;
	see \cite[Definition~1.10]{KustermansVaesWeights}.
	Its $W^*$-lift $\bar\tau$ to
	$\mathcal M_\tau=\pi_\tau(\mathcal A)''$ satisfies
	\[
	\bar\tau\circ\pi_\tau=\tau
	\quad\text{on }\mathcal A_+
	\]
	by \cite[Definition~2.10]{KustermansVaesWeights}.
	
	By \cite[Proposition~2.22 and the paragraph following its proof]
	{KustermansVaesWeights},
	$\bar\tau$ is faithful, normal, and semifinite, with modular
	group extending $\sigma$.
	This extension is trivial by normality and the ultraweak density
	of $\pi_\tau(\mathcal A)$ in $\mathcal M_\tau$.
	Hence $\bar\tau$ is a trace by
	\cite[Proposition~2.7]{KustermansVaesWeights}.
	In particular,
	\[
	\bar\tau\!\left(\pi_\tau(a)^*\pi_\tau(a)\right)
	=
	\tau(a^*a),
	\qquad a\in\mathfrak n_\tau(\mathcal A).
	\]
	
	By \cite[Proposition~2.16 and its proof]{KustermansVaesWeights},
	using the tracial realization of the GNS space of $\bar\tau$,
	the map
	\[
	\Lambda_\tau(a)\longmapsto\pi_\tau(a),
	\qquad a\in\mathfrak n_\tau(\mathcal A),
	\]
	extends to a unitary
	\[
	\mathcal H_\tau\longrightarrow L^2(\mathcal M_\tau,\bar\tau).
	\]
	The density of $\Lambda_\tau(\mathfrak n_\tau(\mathcal A))$
	in $\mathcal H_\tau$ therefore gives
	\begin{equation}\label{eq:GNS-L2-density}
		\overline{
			\pi_\tau(\mathfrak n_\tau(\mathcal A))
		}^{\,\|\cdot\|_{2,\bar\tau}}
		=
		L^2(\mathcal M_\tau,\bar\tau).
	\end{equation}
		
		The embedding $\pi_\tau$ also preserves the metric. Indeed, for
		$a,b\in\mathcal A_{\tau,+}$, continuous functional calculus gives
		\[
		\begin{aligned}
			&\left(
			\pi_\tau(a)^{t/4}
			\pi_\tau(b)^{t/2}
			\pi_\tau(a)^{t/4}
			\right)^{1/t}\\
			&\qquad=
			\pi_\tau\!\left(
			\left(a^{t/4}b^{t/2}a^{t/4}\right)^{1/t}
			\right).
		\end{aligned}
		\]
		Since $\bar\tau\circ\pi_\tau=\tau$ on $\mathcal A_+$, it follows that
		\[
		Q_t^{\bar\tau}(\pi_\tau(a),\pi_\tau(b))
		=
		Q_t^\tau(a,b)
		\]
		and hence
		\[
		d_t^{\bar\tau}(\pi_\tau(a),\pi_\tau(b))
		=
		d_t^\tau(a,b).
		\]
		
	Since $\bar\tau\circ\pi_\tau=\tau$, we have
	\[
	\mathfrak n_{\bar\tau}(\pi_\tau(\mathcal A))
	=\pi_\tau(\mathfrak n_\tau(\mathcal A)).
	\]
	Consequently, \eqref{eq:GNS-L2-density} is exactly condition
	\eqref{eq:L2-core-condition} for $\pi_\tau(\mathcal A)\subseteq\mathcal M_\tau$.
	Theorem~\ref{thm:L2-core} therefore gives
	\[
	\overline{\pi_\tau(\mathcal A_{\tau,+})}^{\,\|\cdot\|_{1,\bar\tau}}
	=L^1(\mathcal M_\tau,\bar\tau)_+.
	\]
	Together with the preceding isometry, the completion assertion
	of that theorem proves \eqref{eq:trace-GNS-completion}.
		
		It remains to consider the trace-one part. Let
		\[
		\mathsf S_\tau(\mathcal A)
		:=
		\{a\in\mathcal A_{\tau,+}:\tau(a)=1\},
		\]
		and let
		\[
		\rho\in L^1(\mathcal M_\tau,\bar\tau)_+,
		\qquad
		\bar\tau(\rho)=1.
		\]
		By the first part, there exist $b_n\in\mathcal A_{\tau,+}$ such that
		\[
		\pi_\tau(b_n)\longrightarrow\rho
		\qquad\text{in }L^1(\mathcal M_\tau,\bar\tau).
		\]
		Set
		\[
		m_n:=\tau(b_n)
		=\bar\tau(\pi_\tau(b_n)).
		\]
		The continuity of the mass functional on $L^1$ gives
		$m_n\to1$. Hence $m_n>0$ for all sufficiently large $n$.
		For such $n$, put
		\[
		c_n:=\frac{b_n}{m_n}\in\mathsf S_\tau(\mathcal A).
		\]
		Then
		\begin{align*}
			\|\pi_\tau(c_n)-\rho\|_1
			&\leq
			\left\|
			\frac{\pi_\tau(b_n)}{m_n}-\pi_\tau(b_n)
			\right\|_1
			+
			\|\pi_\tau(b_n)-\rho\|_1\\
			&=
			|1-m_n|
			+
			\|\pi_\tau(b_n)-\rho\|_1
			\longrightarrow0.
		\end{align*}
		By Theorem~\ref{thm:classification}\textup{(i)}, this convergence
		also holds with respect to $d_t^{\bar\tau}$. Thus
		$\mathsf S_\tau(\mathcal A)$ is dense in the trace-one shell
		\[
		\{\rho\in L^1(\mathcal M_\tau,\bar\tau)_+:\bar\tau(\rho)=1\}.
		\]
		
		Finally, the mass functional is $L^1$-continuous, so this shell
		is $L^1$-closed and hence $d_t^{\bar\tau}$-closed. Since
		$L^1(\mathcal M_\tau,\bar\tau)_+$ is complete for $d_t^{\bar\tau}$,
		the trace-one shell is complete. Its metric completion is therefore
		exactly \eqref{eq:trace-GNS-states}.

<!-- remark-4: body_md -->
\label{rem:faithful-densities}
	A faithful density in $L^1(\mathcal M,\tau)_+$ exists if and only if
	$\mathcal M$ is $\sigma$-finite. Indeed, such a density determines a
	faithful normal positive functional, whereas every
	$\sigma$-finite von Neumann algebra admits a faithful normal state.
	In this case, faithful densities are dense in
	$L^1(\mathcal M,\tau)_+$: if $h$ is one fixed faithful density, then
	\[
	A+\varepsilon h\longrightarrow A
	\qquad\text{in }L^1
	\]
	as $\varepsilon\downarrow0$, and hence also in $d_t$.
	The cone of faithful densities is not complete, since its
	$L^1$-closure contains nonfaithful elements. If $\mathcal M$ is not
	$\sigma$-finite, then no faithful normal state, and consequently
	no faithful state density, exists.
