\label{thm:dne-viscosity}
Let $m>0$, $p\geq 2$ and define $\lambda:= m - \frac{1}{p-1}$. Set $S:= \{(t,s) \in \mathbb{R}^2 \mid 0 < s < t \}$ and let $\Omega \subset \mathbb{R}^d$ be a bounded, convex domain with smooth boundary $\partial \Omega$. Suppose $\omega: \overline{\Omega} \times \overline{\Omega} \times S \rightarrow \mathbb{R}$ is a continuous function, which is twice differentiable in the first and second arguments and differentiable in the third and fourth arguments. Suppose that $\omega(x,y;t,s) \rightarrow \infty$ whenever $t \rightarrow s^+$ for $x\ne y$ and $\lim_{t\to s^+} \omega(x,x;t,s) \geq 0$ for all $x \in \overline{\Omega}$. In addition, suppose there exists a strictly increasing, differentiable function $\beta:[0,\infty) \to [0,\infty)$ with $\beta(0)=0$ and a function $A:[0,\infty) \to [0,\infty)$ such that $A^2\beta^{\lambda}$ is increasing and $\omega$ satisfies
\begin{align}
\partial_{t} \omega + \beta^{-\lambda(p-1)}(t)|\nabla_{x}\omega|^{p} &\geq 0\qquad \text{ in $\overline{\Omega} \times \overline{\Omega} \times S$},\label{eq:dne-euclidean-omega-ineq-1-visc} \\
\partial_{s} \omega - \beta^{-\lambda(p-1)}(s)|\nabla_{y}\omega|^{p} &\geq 0\qquad \text{ in $\overline{\Omega} \times \overline{\Omega} \times S$},\label{eq:dne-euclidean-omega-ineq-2-visc}
\end{align} 
\begin{equation}\label{eq:dne-euclidean-omega-ineq-3-visc}
\begin{split}
    A^{2}(t) \beta^{-\lambda(p-2)}(t)\Delta_{p,x}\omega
+ A^{2}(s) \beta^{-\lambda(p-2)}(s)\Delta_{p,y}\omega& 
\\+ 2A(t)A(s)\beta^{-\frac{\lambda(p-2)}{2}}(t)\beta^{-\frac{\lambda(p-2)}{2}}(s)\Delta_{p,xy} \omega& \\ \leq A^{2}(t)\beta^{\lambda-1}(t) \beta'(t)-& A^{2}(s) \beta^{\lambda-1}(s) \beta'(s)
\end{split}
\end{equation}
in $\overline{\Omega} \times \overline{\Omega} \times S$, where \[
\begin{split}
    \Delta_{p,xy} \omega:= |\nabla_{x}\omega|&^{\frac{p-4}{2}}|\nabla_{y}\omega|^{\frac{p-4}{2}}\times \\&\left(|\nabla_{x}\omega||\nabla_{y}\omega|\sum_{i=1}^d \frac{\partial^2 \omega}{\partial x_i\partial y_i} - (p-2)\sum_{i,j=1}^d \frac{\partial \omega}{\partial x_i} \frac{\partial^2 \omega}{\partial x_i\partial y_j} \frac{\partial \omega}{\partial y_j}\right).
\end{split}
\] Finally, assume that 
\begin{equation}\label{eq:V-convexity-compact-dne-visc}
    \begin{split}
    \nabla_x \omega \cdot \nu(x) \geq 0, \qquad \text{for all $x \in \partial \Omega$, $y \in \overline{\Omega}$, and $0<s<t$,}\\
    \nabla_y \omega \cdot \nu(y) \geq 0, \qquad \text{for all $x \in \overline{\Omega}$, $y \in \partial \Omega$, and $0<s<t$.}
\end{split}
\end{equation}
Then every positive viscosity solution $u$ of
\begin{equation}\label{eq:dne-euclidean-neumann-visc}
\begin{cases}
        \partial_t u = \Delta_{p} (u^m)  &\text{in $\Omega \times (0,\infty)$} \\
        |\nabla (u^m)|^{p-2}\nabla (u^m) \cdot \nu = 0 \quad &\text{on $\partial \Omega \times (0,\infty)$}
    \end{cases}
\end{equation}
satisfies the Harnack inequality
\begin{equation}\label{eq:harnack-dne-lam>0-visc}
    \psi(\beta(t)u(x,t)) \geq \psi(\beta(s)u(y,s)) - \omega(x,y;t,s)
\end{equation}
for all $x,y \in \overline{\Omega}$ and $0<s<t$, where 
\[
\psi(u):= \begin{cases}
   \dfrac{m}{\lambda}u^{\lambda} & \text{if $\lambda \ne 0$,}\\[12pt] 
    \dfrac{1}{p-1}\log u & \text{if $\lambda=0$.}
\end{cases}
\]

\label{thm:Euclidean-Schroedinger-full-visc}
Let $\Omega \subseteq \mathbb{R}^d$ be a bounded, convex domain with smooth boundary $\partial \Omega$ and let $V \in C^2(\Omega)$ be bounded from below.
Let $\omega: \overline{\Omega} \times \overline{\Omega} \times S \rightarrow\mathbb{R}$ be a continuous function, which is twice differentiable in the first and second arguments and differentiable in the third and fourth arguments. Suppose that $\omega(x,y;t,s) \rightarrow \infty$ whenever $t \rightarrow s^+$ for $x\ne y$ and $\lim_{t\to s^+} \omega(x,x;t,s) \geq 0$ for all $x \in \overline{\Omega}$. In addition, assume that $\omega$ satisfies
\begin{align}
    \partial_t \omega + |\nabla_x\omega|^2 &\geq V(x) \ \ \qquad \text{ in $\overline{\Omega} \times \overline{\Omega} \times S$}, \label{eq:schroe-omega-ineq-1-visc}\\
    \partial_s \omega - |\nabla_y\omega|^2 &\geq -V(y) \qquad \text{ in $\overline{\Omega} \times \overline{\Omega} \times S$,}\label{eq:schroe-omega-ineq-2-visc}
\end{align}
and there exist strictly increasing, differentiable functions $\beta:[0,\infty) \to [0,\infty)$ satisfying $\beta(0)=0$ and $A:[0,\infty) \to [0,\infty)$ such that
\begin{equation}
    A(t)^2\Delta_x\omega +A(s)^2\Delta_y \omega + 2A(t)A(s) \Delta _{xy} \omega \leq A(t)^2 \frac{\beta'(t)}{\beta(t)}-A(s)^2 \frac{\beta'(s)}{\beta(s)}\label{eq:schroe-omega-ineq-3-visc}
\end{equation}
holds in $\overline{\Omega} \times \overline{\Omega} \times S$.
Finally, assume that 
\begin{equation}\label{eq:V-convexity-compact-visc}

<!-- remark-1: body_md -->
We immediately mention that our definition of $\mathcal{P}^{2,\pm}u(x_0,t_0)$ does not coincide with the standard definition encountered in literature, which may be found, for example, in \cite{Users-guide}. 
    \begin{enumerate}
        \item We allow for the possibility that the function $u$ may depend on multiple time variables, as is the case for the functions $\omega$ and $Z$ utilised in our proofs of the Harnack inequality.
        \item In the typical definition in literature (for $k=1$), the inequalities are understood in the two-sided limit $t \to t_0$. However, as commented by Juutinen \cite{Juutinen-2001}, one often proceeds in the parabolic setting with the intuition that the behaviour of a solution to a parabolic equation at time $t_0$ is determined only by information about the solution at times before $t_0$ and not after. Therefore, in our definition, we only consider $t_i<t_{0i}$ in the one-sided limits $t_i \to t_{0i}^{-}$.
        \item In Definition \ref{def:par-semijets}, $|\cdot|_{1}$ denotes the $1$-norm and \[
        |t_0-t|_1 = \sum_{i=1}^{k}|t_{0i}-t_i| = \sum_{i=1}^{k}(t_{0i}-t_i),
        \]
        since we always take $t_i<t_{0i}$.
        \item As is the case for $J^{2,\pm}_{\Omega}u(x_0)$, the domain of the function $u:\Omega \times S \to \mathbb{R}$ is often not relevant when considering $\mathcal{P}^{2,\pm}_{\Omega \times S} u(x_0,t_0)$, since the definition is equivalent when one replaces $\Omega \times S$ with any other domain for which $(x_0,t_0)$ is an interior point. In addition, if $I_i = (a,b]$ for some $i$ and we consider the parabolic jets of $u$ at a point $(x_0,t_0)$ such that $t_{0i}=b$, then since we are only concerned with the behaviour of $u$ to the left of $t_{0i}=b$, we have that $\mathcal{P}^{2,\pm}_{\Omega \times S} u(x_0,t_0) = \mathcal{P}^{2,\pm}_{\Omega \times S'} u(x_0,t_0)$, where $S'$ is constructed by replacing the interval $I_i$ in the definition of $S$ by another interval $I_i'$ such that $I_i \subset I_i'$ and $b$ is an interior point of $I_i'$.
    \end{enumerate}

<!-- remark-2: body_md -->
We essentially consider pairs of times $(t,s) \in (0,\infty) \times (0, \infty)$ as being partially ordered with respect to the Euclidean norm, that is, we consider $(t,s)$ as occurring before $(t_0,s_0)$ if and only if $|(t,s)| < |(t_0,s_0)|$. Through this idea, one may also generalise Definition \ref{def:first-touching-point} for functions of any finite number of spatial or time variables. However, Definition \ref{def:first-touching-point} only compares the function values of $Z$ at $(t_0,s_0)$ with those at times before $(t_0,s_0)$. In particular, the definition makes no assumption about what occurs at other times $(t,s)$ for which $|(t,s)| = |(t_0,s_0)|$. Therefore, if $Z$ touches a value $c \in \mathbb{R}$ for the first time at a point $P_0 = (x_0,y_0;t_0,s_0)$, we do not exclude the possibility that there is another point $\hat{P}_0 = (\hat{x}_0,\hat{y}_0;\hat{t}_0,\hat{s}_0)$ with $|(\hat{t}_0,\hat{s}_0)| = |(t_0,s_0)|$, such that $Z$ also touches $c$ for the first time at $P_0$, but $(\hat{t}_0,\hat{s}_0) \ne (t_0,s_0)$.

\end{cases}

We identify the function 
\begin{equation}\label{eq:nonlinear-omega}
\omega(x,y;t,s)=\frac{1}{p'}\left( \frac{K}{d} \right) ^{p'-1} \frac{|x-y|^{p'}}{(t^{Kp/d} - s^{Kp/d})^{p'-1}}
\end{equation} 
to be an optimal candidate for the function $\omega$.
One sees that the function $\omega$ defined in \eqref{eq:nonlinear-omega} satisfies the inequalities \eqref{eq:dne-euclidean-omega-ineq-1-visc} and \eqref{eq:dne-euclidean-omega-ineq-2-visc} with equality. In addition, \eqref{eq:dne-euclidean-omega-ineq-3-visc} is also satisfied with equality once one chooses $A(t) = \frac{d}{Kp}t^{(\frac{p}{d} + \frac{\lambda(p-2)}{2})K}$ and $\beta(t)=t^K$ with $K=\frac{1}{m(p-1)-1+\frac{p}{d}}$. Our choice is further supported by the fact that the Barenblatt solution $\mathcal{B}$ for the doubly nonlinear heat equation satisfies the resulting Harnack inequality \eqref{eq:harnack-dne-lam>0-visc} with equality for all $x,y \in \mathbb{R}^d$ and $0<s<t$ in the support of $\mathcal{B}$ satisfying \[

<!-- remark-4: body_md -->
Theorem \ref{thm:dne-viscosity} requires that the function $\beta$ is strictly increasing, which if $\beta(t) = t^K$, only occurs if $K=\frac{1}{m(p-1)-1+\frac{p}{d}}>0$. This introduces the restriction \[
    p> \frac{d(m+1)}{1+dm},
    \]
    which must be satisfied in order to deduce a Harnack inequality using Theorem \ref{thm:dne-viscosity}. In the case $m=1$ corresponding to the $p$-heat equation, this restriction reads as \[
    p> p_{*}:=\frac{2d}{d+1},
    \]
    which matches the range for which both the Harnack inequality (see, for instance, \cite{DiBenedetto-Harnack-Book}) and its differentiated form due to Esteban and Vázquez \cite{Esteban-Vasquez-1990} are known to hold. Moreover, with $\omega$ chosen as in \eqref{eq:nonlinear-omega}, we recover the Harnack inequality
    \[
    \frac{1}{\lambda}(t^{K}u(x,t))^{\lambda} - \frac{1}{\lambda}(s^{K}u(y,s))^{\lambda} + \frac{1}{p'}\left( \frac{K}{d}\right)^{p'-1} \frac{|x-y|^{p'}}{(t^{Kp/d} - s^{Kp/d})^{p'-1}} \geq 0
    \]
    satisfied by every positive solution of the $p$-heat equation for every $x,y \in \mathbb{R}^d$, $0<s<t$, which coincides with the result in \cite{Auchmuty-Bao-1994}.
    
    Similarly, for $p=2$, one has the restriction \[
    m> 1 - \frac{2}{d},
    \]
    which already appeared in the work of Aronson and B\'enilan \cite{Aronson-Benilan-1979} on the porous medium equation. In addition, we also obtain the Harnack inequality 
    \[
    \frac{m}{m-1}(t^{K}u(x,t))^{m-1} - \frac{m}{m-1}(s^{K}u(y,s))^{m-1} +  \frac{K}{2d} \frac{|x-y|^{2}}{t^{2K/d} - s^{2K/d}} \geq 0
    \]
     satisfied by every positive solution of the porous medium equation for every $x,y \in \mathbb{R}^d$, $0<s<t$ \cite{Auchmuty-Bao-1994}.

<!-- definition-7: body_md -->
Let $\Omega \subseteq \mathbb{R}^d$, $T>0$, $m>0$, and  $p>1$. We call a function $u:\Omega \times [0,T] \to [0,\infty)$ a \emph{(nonnegative) weak solution of \eqref{eq:dne-euclidean-neumann-visc}} if \[
    u \in C([0,T]; L^{m+1}(\Omega)), \quad u^m \in L^{p}(0,T;W^{1,p}(\Omega)),
    \]
    and \[
    \int_{0}^{T} \int_{\Omega} -u \partial_{t}\varphi + |\nabla u^{m}|^{p-2}\nabla u^{m}\nabla \varphi \ \mathrm{d} x  \ \mathrm{d} t =0 
    \]
    for all nonnegative functions \[
    \varphi \in W^{1, 1+ \frac{1}{m}}(0,T;L^{1+ \frac{1}{m}}(\Omega)) \cap L^p(0,T;W^{1,p}_0(\Omega)).
    \]

<!-- definition-8: body_md -->
A function $u \in C([0,\infty];L^1(\Omega))$ is called a mild solution of the Cauchy problem \eqref{eq:dne-cauchy} if $u(0)=u_{0}$ in $L^1(\Omega)$, and for every $\varepsilon >0$, there is a partition $\pi_\varepsilon:0=t_0<t_1<\ldots < t_{N_{\varepsilon}}=T$ such that $\max_{1, \ldots, N_{\varepsilon}} (t_i - t_{i-1})<\varepsilon$, and there is a sequence $(u_{i, \varepsilon})_{i=0}^{N_\varepsilon} \subseteq D(A)$ defined by $u_{0,\varepsilon}=u_0$ and $u_{i,\varepsilon}$ is defined via
    \[
    u_{i,\varepsilon} + (t_{i,\varepsilon} - t_{i-1,\varepsilon})Au_{i,\varepsilon} \ni u_{i-1,\varepsilon}
    \]
    for $i=1, \ldots, N_\varepsilon - 1$, such that the sequence $(u_{\varepsilon})_{\varepsilon >0}$ of step functions \[
    u_{\varepsilon}(t):= \sum_{i=0}^{N_{\varepsilon}-1} u_{i,\varepsilon} \mathbbm{1}_{[t_{i,\varepsilon},t_{i,\varepsilon+1})}
    \]
    converges to $u$ in $C([0,\infty];L^1(\Omega))$ as $\varepsilon \to 0^{+}$.

<!-- theorem-5: body_md -->
\label{thm:approximation}
Let $u_0 \in L^1(\mathbb{R}^d)$ be positive and let $u$ be the unique weak solution to \eqref{eq:dne-manifold} on $\mathbb{R}^d$. Then there exists a sequence $(u_n)_{n \geq 1}$ of functions $u_n: \Omega_n \to (0,\infty)$ defined on an increasing sequence of bounded convex sets $\Omega_n \subseteq \mathbb{R}^d$ such that $u_n$ is a positive viscosity solution of the homogeneous Neumann problem \eqref{eq:dne-euclidean-neumann-visc} for all $n \geq 1$ and $u_n \to u$ pointwise on $\mathbb{R}^d \times (0,\infty)$.
