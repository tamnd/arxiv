<!-- theorem-1: body_md -->
\label{general}
Let $\{x_n\}_{n\geq0}$ be a sequence of rational functions in
$\mathbb{Q}(x_0,\ldots,x_{m-1})$ satisfying \eqref{eqn:form-of-rec}. Suppose that $\{x_n\}_{n\geq0}$ has the Laurent property and that there exists $1\leq t\leq m-1$ such that
$c_{t,m-t}\neq 0$.
    
                
                        
              
    
Consider the rational map
$$
f\colon\mathbb{A}^m\longrightarrow\mathbb{A}^m,
\qquad
f(x_0,\ldots,x_{m-1})
=
(x_1,\ldots,x_m).
$$
Then
$$
\deg(f^n)
=
1+\left(
\sum_{i=0}^{m-1}
\max\{k_j^{(i)}\mid n\leq j\leq n+m-1\}
\right)
$$
for $0\leq i\leq m-1$ the sequence $\{k_n^{(i)}\}_{n \geq 0}$ satisfies the initial conditions  
$$
k_0^{(i)}=\cdots=k_{m-1+i}^{(i)}=0,
\qquad
k_{m+i}^{(i)}=1,
$$
and of the form
\[
k^{(i)}_{n+m} = \max_{(s,t)\in C}\big(k^{(i)}_{n+s}+k^{(i)}_{n+t}\big) - k^{(i)}_{n}, \quad n\ge i+1,
\]
where $C = \{(s,t) : 1\le s,t\le m-1,\ c_{s,t}\neq 0\}$.

<!-- definition-1: body_md -->
Let $\mathbb{F}$ be a field. A discrete valuation is a map $v\colon \mathbb{F}\setminus \{0\}\rightarrow \mathbb{Z}$ with the following properties:
 	\begin{enumerate}
 		\item $v$ is surjective,
 		\item $v(xy)=v(x) + v(y)$ for all $x$, $y$ in $\mathbb{F}\setminus \{0\}$,
 		\item $v(x+y)\geq \min(v(x),v(y))$ for all $x$, $y$ in $\mathbb{F}\setminus \{0\}$ with $x+y\neq 0$.
 	\end{enumerate}
 The set $R_v=\{x\in\mathbb{F}\mid v(x)\geq0 \}\cup\{0\}$ is then a subring, called the valuation ring of $v$. An integral domain is called a discrete valuation ring if it is the valuation ring of a discrete valuation $v$. A valuation ring $R_v$ has a unique maximal ideal $M_v:=\{x\in\mathbb{F}\mid v(x)>0 \}\cup \{0\}$.

First, we will show by induction on $n$ that $\mu_i(x_n)>0$ for all $0\leq i\leq m-1$ and all $n\geq0$. It is clear that $\mu_i(x_j)=1$ for $0\leq i,j \leq m-1$. Now, we fix $i\in \{0,\ldots,m-1\}$ and $n\geq0$. Assume that we have $\mu_i(x_j)>0$ for all $j\leq n+m-1$. We will show that $\mu_i(x_{n+m})>0$. By equation \eqref{eqn:form-of-rec}, we have $$\mu_i(x_{n+m}x_n)=\mu_i\left( \sum_{s=1}^{m-1}\sum_{t=1}^{m-1}c_{s,t}x_{n+s}x_{n+t} \right).$$ 
  Now, we look at the $x_i$-adic valuation of the sum $$S=\sum_{s=1}^{m-1}\sum_{t=1}^{m-1}c_{s,t}x_{n+s}x_{n+t}.$$ Let $$\nu=\min\{v_{x_i}(x_{n+s}x_{n+t})\mid 1\leq s,t \leq m-1, c_{s,t}\neq0  \}$$ be the minimal valuation among the terms of the sum $S$. We split the sum $S$ into two summands $S=S_1+S_2$, where $S_1$ is the sum of the terms $c_{s,t}x_{n+s}x_{n+t}$ of valuation $\nu$, and $S_2$ is the sum of the terms of higher valuation. We claim that $v_{x_i}(S_1)=\nu$; indeed, passing to the residue field of the valuation $v_{x_i}$, and looking at the $x_i$-adic images of the terms in $S_1$, we know by induction that the leading terms $\mu_i(c_{s,t}x_{n+s}x_{n+t})$ are strictly positive. It follows that the leading term of the sum of these $x_i$-adic images is also positive, and in particular does not vanish. Hence indeed, $v_{x_i}(S_1)=\nu$ and $\mu_i(S_1)>0$. Since the terms in $S_2$ have higher valuation, we have $v_{x_i}(S)=v_{x_i}(S_1)$ and $\mu_i(S)=\mu_i(S_1)>0$. Hence, $$\mu_i(x_{n+m})\mu_i(x_n)=\mu_i(x_{n+m}x_n)>0.$$ Therefore, $\mu_i(x_{n+m})>0.$ Furthermore, we obtain that 
  \begin{align}\label{xiadicval}
  v_{x_i}(x_{n+m})=\min\{v_{x_i}(x_{n+s})+v_{x_i}(x_{n+t}) \mid c_{s,t}\neq 0  \}-v_{x_i}(x_n),
  \end{align}
  for $1\leq s,t\leq m-1$.
   
                                                                                                                                                                                                   
  
Observe that for the initial variables we have $x_i$-adic valuations for $0\leq i,j \leq m-1$
\[
    v_{x_i}(x_j)=\begin{cases}
        0 &\text{ if } j\neq i,\\
        1 &\text{ otherwise}.
    \end{cases}
\]
     
We claim that for $0\le i\le m-1$
$$
v_{x_i}(x_n)=
\begin{cases}
0 & \text{if } 0\le n\le m+i-1,\ n\neq i,\\
1 & \text{if } n=i,\\
-1 & \text{if } n=m+i.
\end{cases}
$$
Indeed, fix $0\leq i \leq m-1$, and let $n=n'+m$ with $0\leq n'\leq i$. Note that $P$ is not divisible by $x_{i-n'}$. Thus, there is a pair $(s,t)\in C$ with $i-n'\notin \{s,t\}$. Therefore, by induction on $n'$ we get that the minimum valuation in \eqref{xiadicval} is $0$. Hence, $v_{x_i}(x_{n'+m})=-v_{x_i}(x_{n'})$ for $0\leq n'\leq i$ which proves the claim.
 

Now set $\kappa_n^{(i)}:=-v_{x_i}(x_n)$ for $n\neq i$ and $\kappa_i^{(i)}:=0$. Then
$$\kappa_0^{(i)}=\cdots=\kappa_{m-1+i}^{(i)}=0,\qquad \kappa_{m+i}^{(i)}=1,$$
and, using \eqref{xiadicval} and $\kappa_n^{(i)}:=-v_{x_i}(x_n)$ we obtain
$$\kappa^{(i)}_{n+m} = \max_{(s,t)\in C}\big(\kappa^{(i)}_{n+s}+\kappa^{(i)}_{n+t}\big) - \kappa^{(i)}_{n}
$$
for $n\geq i+1$.
 

Now, we will observe that $\kappa_n^{(i)}\geq 0$ for all $n\geq 0$. By assumption, $(t,m-t)\in C$, hence
$$\kappa_{n+m}^{(i)}+\kappa_n^{(i)}\geq \kappa_{n+t}^{(i)}+\kappa_{n+m-t}^{(i)}, \qquad n\geq i+1.$$
We show by strong induction on $N$ that $\kappa_N^{(i)}\geq \kappa_{N-t}^{(i)}$ for $N\geq t$ and $\kappa_N^{(i)}\geq \kappa_{N-(m-t)}^{(i)}$ for $N\geq m-t$. If $N\leq m+i$, this is clear from the initial values, since the only nonzero value among $\kappa_0^{(i)},\ldots,\kappa_{m+i}^{(i)}$ is $\kappa_{m+i}^{(i)}=1$. If $N\geq m+i+1$, write $N=n+m$. 
  
Now assume that for $n'< N$ we have
\[
\kappa^{(i)}_{n'}\geq \kappa^{(i)}_{n'-t}, \text { for } n'\geq t \text{ and } \kappa^{(i)}_{n'}\geq \kappa^{(i)}_{n'-(m-t)} \text{ for } n'\geq m-t.
\]
Thus, we get
$$\kappa_{N}^{(i)}-\kappa_{N-t}^{(i)}\geq \kappa_{n+t}^{(i)}-\kappa_n^{(i)}\geq 0
\quad\text{and}\quad
\kappa_{N}^{(i)}-\kappa_{N-(m-t)}^{(i)}\geq \kappa_{n+m-t}^{(i)}-\kappa_n^{(i)}\geq 0,$$
since $n+t<N$ and $n+m-t<N$. Now, for $N\geq 0$ write $N=qt+r$ with $0\leq r<t$. Then
$$\kappa_N^{(i)}\geq\kappa_{N-t}^{(i)}\geq\cdots\geq\kappa_r^{(i)}=0.$$
Consequently, $v_{x_i}(x_n)\leq 0$ for all $n\neq i$ since $\kappa_n^{(i)}=-v_{x_i}(x_n)$ for $n \neq i$.
 
 

Note that $\{x_n\}_{n\geq0}$ has the Laurent property. Therefore, we can write $x_n$ uniquely in the form
\begin{equation}\label{Lproperty}
x_n=\dfrac{g_n}{x_0^{k_n^{(0)}}\cdots x_{m-1}^{k_n^{(m-1)}}}, \qquad n\geq 0,
\end{equation}
where $g_n$ is a polynomial in $x_0,\ldots,x_{m-1}$, the $k_{n}^{(i)}$ are non-negative integers, and for each $0\leq i\leq m-1$ either $k_n^{(i)}=0$ or $x_i\nmid g_n$. In particular,
$$v_{x_i}(x_n)=v_{x_i}(g_n)-k_n^{(i)}, \qquad v_{x_i}(g_n)\geq 0.$$
Now let $n\neq i$. If $k_n^{(i)}>0$, then $x_i\nmid g_n$, so $v_{x_i}(g_n)=0$ and $k_n^{(i)}=-v_{x_i}(x_n)$. If $k_n^{(i)}=0$, then $v_{x_i}(g_n)=v_{x_i}(x_n)\leq 0$. Thus $v_{x_i}(g_n)=0$, which means $x_i\nmid g_n$. Therefore, $k_n^{(i)}=0=-v_{x_i}(x_n)$. Thus, for $n\neq i$ we have $x_i\nmid g_n$ and
$$k_n^{(i)}=-v_{x_i}(x_n)=\kappa_n^{(i)}.$$
For $n=i$ we have $x_i=x_i/1$, so $k_i^{(i)}=0=\kappa_i^{(i)}$. Therefore $k_n^{(i)}=\kappa_n^{(i)}$ for all $n\geq 0$, and the sequences $\{k_n^{(i)}\}_{n\geq0}$ satisfy the initial conditions and the recurrence in the statement.


                                                                                                  
                                   
                                                                               
                                                                                                                       
                                                                                                           
 
 
                                                                                                                                                                                                                                                                                                                                                                                 

    
Finally, we can compute the degree of iterates of the map $f$. We write the $n^{\text{th}}$ iterate of the map f as $$f^n\colon (x_0,\ldots,x_{m-1})\mapsto (x_n,\ldots,x_{n+m-1}).$$ We denote $m_n^{(i)}:=\max\{k_j^{(i)}\mid n\leq j\leq n+m-1 \}$ for $0\leq i \leq m-1$. In projective coordinates, the map $f^n$ takes the form
	 
	\begin{align*}
	(x_0:\dots:x_{m-1}:y)\mapsto \bigg(&G_nx_0^{(m_n^{(0)}-k_{n}^{(0)})}\cdots x_{m-1}^{(m_{n}^{(m-1)}-k_{n}^{(m-1)})}:\\
	&\dots:\\
	&G_{n+m-1}x_0^{(m_n^{(0)}-k_{n+m-1}^{(0)})}\cdots x_{m-1}^{(m_{n}^{(m-1)}-k_{n+m-1}^{(m-1)})}:\\
	&x_0^{m_n^{(0)}}\cdots x_{m-1}^{m_n^{(m-1)}}y \bigg),
	\end{align*} 
	where $G_j=y^{\deg(g_j)}g_j\left(\frac{x_0}{y},\dots,\frac{x_{m-1}}{y}\right)$  are homogenizations of the polynomials $g_j$'s, and are of degree $$k_j^{(0)}+\dots+k_j^{(m-1)}+1.$$ We claim that the projective coordinates of $f^n$ have no common factor for all $n\geq 0$. Indeed, $y$ can not be a common factor since $G_j$'s are homogenizations of the polynomials $g_j$'s. 
                                                                                             
    Now, if $m_n^{(i)}>0$ for any $0\leq i\leq m-1$, then $m_n^{(i)}=k_j^{(i)}>0$ for some $n\leq j \leq n+m-1$. Thus, $x_i \nmid g_j$, so $x_i \nmid G_j$. Now we look at $j^\text{th}$ projective coordinate of the $f^n$ and we see that $$x_i \nmid G_jx_0^{m_n^{(0)}-k_j^{(0)}}\dots x_{i-1}^{m_n^{(i-1)}-k_j^{(i-1)}}x_{i+1}^{m_n^{(i+1)}-k_j^{(i+1)}}\dots x_{m-1}^{m_n^{(m-1)}-k_j^{(m-1)}},$$ since $m_n^{(i)}=k_j^{(i)}$. Additionally, if $m_n^{(i)}=0$, then $x_i \nmid x_0^{m_n^{(0)}}\cdots x_{m-1}^{m_n^{(m-1)}}y$. Hence, $x_i$ can not be a common factor for projective coordinates. 
	Therefore, $$\deg(f^n)=m_n^{(0)}+\cdots+m_n^{(m-1)}+1.$$\qedhere

We claim that by the assumptions of the Theorem~\ref{general} the sequences $\{k_n^{(i)}\}_{n\geq 0}$ are non-decreasing sequences for all $0\leq i\leq m-1.$
         
        We will show this by induction on $n$. By equation \eqref{eqn:form-of-rec}, we see that $$k_0^{(i)}=k_1^{(i)}=\cdots=k_{m-1+i}^{(i)}=0,$$ and $k_{m+i}^{(i)}=1$. Now suppose, $k_j^{(i)}\geq k_{j-1}^{(i)}$ for some $n>i$ and all $n\leq j\leq n+m-1$. Thus, $$k_{n+m}^{(i)}\geq k_{n+m-1}^{(i)},$$ which completes the inductive proof. Therefore, the result follows then immediately from Theorem \ref{general} since the $m_n^{(i)}=k_{n+m-1}^{(i)}$ for all $n$.

Call a sequence $\{k_n\}_{n\geq1}$ of real numbers max-linearly recurrent if there exist $d\geq 1$, $s\geq 1$ and real numbers $c_j^{(i)}$ for $1\leq i\leq s$ and $1\leq j \leq d$ such that $$k_n=\max\bigg(\sum_{j=1}^{d}c_j^{(1)}k_{n-j},\sum_{j=1}^{d}c_j^{(2)}k_{n-j},\ldots,\sum_{j=1}^{d}c_j^{(s)}k_{n-j}\bigg), \qquad n\geq d+1.$$

<!-- proof-4: body_md -->
For an integer $1\leq n \leq N-2,$ write $n=(i-1)(m-1)+j$ with $1\leq i \leq m-2$ and $1\leq j \leq m-1$. First, we claim that 
	 
	\begin{equation}\label{formula for k_n p=1 q=2}
	k_n=\binom{i}{2}+\max(j-m+i,0), \qquad 1\leq n \leq N-2.
	\end{equation} The formula is clear for $1\leq n \leq m$. If $n$ is such that $1\leq n \leq N-m-2$, then we put 
	\begin{align*}
	\delta_1=\begin{cases}
	1 & \text{if } i+j \geq m,\\
	0 & \text {otherwise} \end{cases}, \ 
	\delta_2=\begin{cases}1 & \text{if } i+j\geq m-1, \\ 0 & \text{otherwise}\end{cases}, \ 
	\delta_3=\begin{cases}
	1 & \text{if } j=1\\
	0 & \text{otherwise}.
	\end{cases}
    \end{align*} Then a direct computation implies that for all $n$ such that the formula \eqref{formula for k_n p=1 q=2} holds for all the terms of the sequence $\{k_n\}_{n\geq1}$ occurring in the corresponding equations, we have
    \begin{align*}
    &k_{n+1}=k_n+\delta_1,\\
    &k_{n+2}=k_n+\delta_1+\delta_2,\\
    &k_{n+m-1}=k_{n}+i+\delta_1,\\
    &k_{n+m-2}=k_{n}+i-\delta_3,\\
    &k_{n+m}=k_{n}+i+\delta_1+\delta_2.
    \end{align*} Suppose the formula \eqref{formula for k_n p=1 q=2} holds for $k_1,\ldots,k_{n+m-1}$. Then 
    \begin{align*}
    k_{n+m}=&\max(k_{n+m-1}+k_{n+1}-k_n,k_{n+m-2}+k_{n+2}-k_n)\\&=\max(k_{n}+i+2\delta_1,k_n+i+\delta_1+\delta_2-\delta_3)\\&=k_n+i+\delta_1+\delta_2+\max(\delta_1-\delta_2,-\delta_3).
    \end{align*} We have $\delta_1-\delta_2 \in \{0,-1\}$, $-\delta_3\in \{0,-1\}$ and $\max(\delta_1-\delta_2,-\delta_3)=0$, since if we had $$\delta_1-\delta_2=-\delta_3=-1,$$ then $i+j=m-1$, and $j=1$, which would mean that $n=N-m$, a contradiction. Thus $$k_{n+m}=k_n+i+\delta_1+\delta_2$$ and therefore the formula \eqref{formula for k_n p=1 q=2} is also satisfied for $k_{n+m}$, and hence by induction for all $1\leq n \leq N-2$. 
    
    Applying \eqref{formula for k_n p=1 q=2} for $ N-m-1\leq n\leq N-2$ we get that \begin{align}\label{formula delta4}
    k_n=n-\dfrac{N}{2}+\delta_4,
    \end{align} where 
    \begin{align*}
    \delta_4=\begin{cases}
    1 & \text{if } n\in \{N-m-1,N-m\},\\
    0 & \text{otherwise}.
    \end{cases}
    \end{align*}
   Using \eqref{2orderrec p=1 q=2} and \eqref{formula delta4} we get that 
   \begin{equation*}
   k_{N-1}=k_N=\dfrac{N}{2}-1.
   \end{equation*} 
   Thus, for $N-m-1\leq n \leq N$ we have 
   \begin{equation}\label{k_N}
   k_n=n-\dfrac{N}{2}+\delta_5,
   \end{equation}where 
   \begin{align*}
  \delta_5=\begin{cases}
   1 & \text{if } n\in \{N-m-1,N-m\},\\
   -1 & \text{if } n=N,\\
   0 & \text{ otherwise}.
   \end{cases}
   \end{align*}
   Using equations \eqref{2orderrec p=1 q=2} and \eqref{k_N}, it is easy to directly compute the values $k_{N+1},\ldots,k_{N+m}$, obtaining 
                                                                                           
                             
                     
                           
                                                                                                     
                  
                                                                
                                                                            
                                                                                                      
                                                                                               
                                                                                                                                                               
   \begin{equation}\label{k_N+n}
   k_{N+n}=\dfrac{N}{2}+n+\delta_6, \qquad 1\leq n \leq m,
   \end{equation} where 
   \begin{align*}
   \delta_6=\begin{cases}
   1 & \text{if } n=m,\\
   0 & \text{otherwise}.
   \end{cases}
   \end{align*}
   Now, we claim that 
	\begin{equation}\label{claim on k_N+n}
	k_{N+n}=\dfrac{N+2n}{2}+k_n
	\end{equation} for $1\leq n \leq N$. Since the form of the recurrence \eqref{2orderrec p=1 q=2} is invariant under changing the sequence by adding to it an arithmetic progression, we see that the sequence $\{l_n\}_{n\geq1}$ given by the formula $$l_n=k_{N+n}-\dfrac{N}{2}-n$$ satisfies the same recurrence \eqref{2orderrec p=1 q=2} and with initial values $l_1=\dots=l_{m-1}=0$ and $l_m=1$. Hence, $l_n=k_n$ for all $n\geq 1$.
	
	Finally, we define the sequence $\{c_n\}_{n\geq 1}$ as $c_n=2Nk_n-n^2$ for all $n\geq 1$. We use the equation \eqref{claim on k_N+n} to check that  
	\begin{align*}
	c_{N+n}&=2Nk_{n+N}-\left(n+N \right)^2=2Nk_n-n^2=c_n
	\end{align*} for all $n\geq 1$. Consequently, we see that $c_n$ is a periodic sequence with period $N$ and $k_n=\dfrac{n^2+c_n}{2N}$.

<!-- lemma-1: body_md -->
\label{Lemma on f(i,j,m)}
   	Let $\{k_n\}_{n\geq 1}$ be the growth rate sequence of three-term Gale--Robinson with $p=1$, $q=2$ and $r\geq 3$. For $1\leq n \leq N/2$ write $n$ in the form $n=(i-1)(m-1)+j$ for $1\leq i$ and $1\leq j \leq m-1$, and define $$f(i,j,m)\footnotemark=f(n):=\binom{i}{2}+\max\left(\ceil[\bigg]{\dfrac{j+2(i-1)-(m-1)}{2}},0\right).$$
    \footnotetext{Since the triple $(i,j,m)$ determines the n we note the function by $f(n)$ if $(i,j,m)$ clear by the context}
    Then the following equations hold. 
        \begin{enumerate}
        
   		\item[i)] $k_n=f(n)$ if $1\leq n<N/2$, 
   		\item[ii)] $k_{N/2}=f(N/2)$ if $m$ is even, 
   		\item[iii)] $k_{N/2}=f(N/2)-1$ if $m$ is odd.
   	\end{enumerate}

Obviously, we see that $k_n=f(n)$ for $1\leq n \leq m$. If $n$ is such that $1\leq n < N/2-m$, then we set the followings;
   	\begin{align*}
   	\delta_1=\begin{cases}
   	1 & \text{if }  j\geq m-2i\\
   	& \text{and } 2 \nmid m+j,\\
   	0 & \text {otherwise} \end{cases}, \ 
   	\delta_2=\begin{cases}
   	1 & \text{if } j\geq m-2i, \\ 
   	0 & \text{otherwise}\end{cases}, \ 
   	\delta_3= \begin{cases}
   	1 & \text{if } m-2i\leq j\leq m-3\\ 
   	  & \text{and }2\nmid m+j, \\
   	0 & \text{otherwise} 
   	\end{cases},
   	\end{align*}  
   	 
   	\begin{align*}
   	 \delta_4=\begin{cases}
   	 i-1 & \text{if }  j=1,\\
   	 i & \text {otherwise} \end{cases}, \ 
   	 \delta_5=\begin{cases}
   	 i-1 & \text{if } j\leq 2, \\ 
   	 i & \text{otherwise}\end{cases}, \
   	\delta_6= \begin{cases}
   	1 & \text{if } j\geq m-2i-1,\\ 
   	0 & \text{otherwise}. 
   	\end{cases}
   	\end{align*}
   	Then we see that the formula $f(n)$ gives the following equivalent formulas:
   	\begin{align*}
   	&f(n+1)=f(n)+\delta_1,\\
   	&f(n+2)=f(n)+\delta_2,\\
   	&f(n+r)=f(n)+\delta_5,\\
   	&f(n+m-1)=f(n)+i+\delta_2,\\
   	&f(n+m-2)=f(n)+\delta_4+\delta_1,\\
   	&f(n+m-r)=f(n)+\delta_3+\delta_6,\\
   	&f(n+m)=f(n)+i+\delta_1+\delta_6.
   	\end{align*} 
   	for the terms of the sequence $\{k_n\}_{n\geq1}$ for all $1\leq n < N/2-m$. Suppose the formula $f(n)$ holds for $k_1,\ldots,k_{n+m-1}$. Then 
   	\begin{align*}
   	k_{n+m}&=\max(k_{n+1}+k_{n+m-1}-k_n,k_{n+2}+k_{n+m-2}-k_n,k_{n+r}+k_{n+m-r}-k_n)\\
   	&=k_n+\max(\delta_1+i+\delta_2,\delta_1+\delta_2+\delta_4,\delta_3+\delta_5+\delta_6)\\
   	&=k_n+\begin{cases}
   	i+2 & \text{if } j\geq m-2i \text{ and } 2\nmid m+j,\\
   	i+1 & \text{if } j=m-2i-1,\\
   	i+1 & \text{if } j\geq m-2i-1 \text{ and } 2\mid m+j,\\
   	i   & otherwise.
   	\end{cases}\\
   	&=k_n+i+\delta_1+\delta_6.
   	\end{align*}
   	Thus, the formula $f(n)$ is also true for $k_{n+m}$ and therefore by induction for all $1\leq n < N/2$. This completes the proof of the first claim. 
   	
    Therefore, direct computations show the second and third claim using the recurrence of the sequence $\{k_n\}_{n\geq1}$ and the formula $f(n)$.

<!-- lemma-2: body_md -->
\label{Lemma 2 on p1 q2 r arbit}
     	Let $\{k_n\}_{n\geq 1}$ be the growth rate sequence of three-term Gale--Robinson with $p=1$, $q=2$ and $r\geq 3$. Then $k_{N/2+n}=k_{N/2-n}+n$ where $1\leq n < N/2.$ Moreover, $k_{N+n}=\frac{N+2n}{2}+k_n$ for $1\leq n\leq N$.
<!-- end lemma-2 -->
<!-- proof-8: body_md -->
First, we want to prove that 
    	\begin{equation}\label{eqn k_N/2}
    	k_{N/2+n}=k_{N/2-n}+n, \qquad 1\leq n< N/2.
    	\end{equation}
    	We prove this by induction on $n$. In order to use induction, we need to check that formula \eqref{eqn k_N/2} holds for $1\leq n \leq m$. Assume that $m=p+q+r$ is an even integer, which means $r$ is odd. Then we compute that $$N=p^2+q^2+r^2+pqr=(m-1)(m-3)+5,$$ which implies the following 
    	\begin{equation}\label{eqn N/2}
    	N/2=(m-1)\left(\dfrac{m-4}{2}\right) + \dfrac{m+4}{2}.
    	\end{equation}
    	Using equation \eqref{eqn N/2} we write the following equations;
    	\begin{align*}
    	&N/2-m+1=(m-1)\left(\dfrac{m-6}{2}\right)+\dfrac{m+4}{2},\\
    	&N/2-m+2=(m-1)\left(\dfrac{m-6}{2}\right)+\dfrac{m+6}{2},\\
    	&N/2-m+3=(m-1)\left(\dfrac{m-6}{2}\right)+\dfrac{m+8}{2},\\
    	&N/2-m+r+1=(m-1)\left(\dfrac{m-4}{2}\right)+\dfrac{m}{2},\\
    	&N/2-1=(m-1)\left(\dfrac{m-4}{2}\right)+\dfrac{m+2}{2},\\
    	&N/2+1-r=(m-1)\left(\dfrac{m-6}{2}\right)+\dfrac{m+10}{2}.\\
    	\end{align*}
    	By the recurrence of the $\{k_n\}_{n\geq 1}$ we have that 
    	\begin{align*}
    	k_{N/2+1}&=max\left(k_{N/2+2-m}+k_{N/2},k_{N/2-m+3}+k_{N/2-1},k_{N/2-m+r+1}+k_{N/2+1-r}\right) \\ &-k_{N/2-m+1}.
    	\end{align*}
        Now we use the equations above and the Lemma \ref{Lemma on f(i,j,m)}. To be rigorous, we assume that $m\geq 12$. Notice that if $m<12$, then we need to arrange some of the equations above to have $1\leq j \leq m-1$ for the formula $f(n)$. Moreover, one can observe that for $m\in \{6,8,10\}$ the formula in \eqref{eqn k_N/2} holds via direct calculations.
        Therefore,

   
\begin{align*}
k_{N/2+1}
=\max\Big[f\Big(\tfrac{m-4}{2},\tfrac{m+6}{2},m\Big)+f\Big(\tfrac{m-2}{2},\tfrac{m+4}{2},m\Big),\\
f\Big(\tfrac{m-4}{2},\tfrac{m+8}{2},m\Big)+f\Big(\tfrac{m-2}{2},\tfrac{m+2}{2},m\Big),\\
f\Big(\tfrac{m-2}{2},\tfrac{m}{2},m\Big)+f\Big(\tfrac{m-4}{2},\tfrac{m+10}{2},m\Big)\Big]\\
-f\Big(\tfrac{m-4}{2},\tfrac{m+4}{2},m\Big)\\
=\binom{(m-2)/2}{2}+\Big\lceil\tfrac{m-4}{4}\Big\rceil+1=k_{N/2-1}+1.
\end{align*}
  
Now, assume that $m$ is odd, so that $r=m-3$ is even. Then $\gcd(q,r)=2$. Thus,
$$N=r^2+q^2+2p^2+pqr=r^2+2r+6=(m-1)(m-3)+6.$$
\begin{equation}\label{eqn:N/2-when-m-is-odd}
    N/2=(m-1)\Big(\frac{m-3}{2}\Big)+3 
\end{equation}
Now we use Lemma~\ref{Lemma on f(i,j,m)}(iii) we have $k_{N/2}=f(N/2)-1$. We set
\[
a=\dfrac{m-3}{2} \text{ and } b=\dfrac{m-1}{2}.
\]
We have $N/2-m+r+1=N/2-2$ and $N/2+1-r=N/2-m+4$ since $r=m-3$. Moreover,
\begin{align*}
N/2-m+1&=(m-1)(a-1)+3,\\  
N/2-m+2&=(m-1)(a-1)+4,\\
N/2-m+3&=(m-1)(a-1)+5,\\
N/2-m+4&=(m-1)(a-1)+6,\\
N/2-2&=(m-1)(b-1)+1,\\
N/2-1&=(m-1)(b-1)+2.
\end{align*}
 
By Lemma~\ref{Lemma on f(i,j,m)} we obtain
\begin{align*}
    &k_{N/2-m+1}=k_{N/2-m+2}=\binom{a}{2},\\ &k_{N/2-m+3}=k_{N/2-m+4}=\binom{a}{2}+1,\\ &k_{N/2-2}=k_{N/2-1}=k_{N/2}=\binom{b}{2}.
\end{align*}
Hence
\begin{align*}
k_{N/2+1}&=\max\big(k_{N/2-m+2}+k_{N/2},\;k_{N/2-m+3}+k_{N/2-1},\;k_{N/2-2}+k_{N/2-m+4}\big)-k_{N/2-m+1}\\
&=\max\Big(\binom{b}{2},\,\binom{b}{2}+1,\,\binom{b}{2}+1\Big)=\binom{b}{2}+1=k_{N/2-1}+1.
\end{align*}
Note that here all indices satisfy $1\le j\le m-1$ for every odd $m\ge7$, so no small cases need separate treatment.

The remaining base cases $2\leq n\leq m$ follow from analogous computations using Lemma~\ref{Lemma on f(i,j,m)} and the decompositions above. For $m$ even, they are based on \eqref{eqn N/2} and $k_{N/2}=f(N/2)$. For $m$ odd, they are based on \eqref{eqn:N/2-when-m-is-odd} and $k_{N/2}=f(N/2)-1$.

Therefore, we have equation~\eqref{eqn k_N/2} holds for $1\leq n \leq m$. Suppose that equation \eqref{eqn k_N/2} holds for all $t\leq n+m-1$ with $1\leq n<N/2-m$. Now, we will show that equation \eqref{eqn k_N/2} holds for $t=n+m$. We know that
        \begin{align*}
        k_{N/2+n+m}=&\max(k_{N/2+n+1}+k_{N/2+n+m-1},k_{N/2+n+2}+k_{N/2+n+m-2},\\
        &k_{N/2+n+r}+k_{N/2+n+m-r})-k_{N/2+n}.
        \end{align*} We use the induction assumption in the previous equation, and we get 
        \begin{align*}
        k_{N/2+n+m}=&\max(k_{N/2-n-1}+n+1+k_{N/2-n-m+1}+n+m-1,\\
        &k_{N/2-n-2}+n+2+k_{N/2-n-m+2}+n+m-2,\\
        &k_{N/2-n-r}+n+r+k_{N/2-n-m+r}+n+m-r)-(k_{N/2-n}+n)\\
        &=k_{N/2-(n+m)}+n+m.
        \end{align*}
        Moreover, using equation \eqref{eqn k_N/2} and the recurrence of the $\{k_n\}_{n\geq 1}$ we directly compute that $$k_{N+n}=\dfrac{N+2n}{2}+k_n, \qquad 1\leq n \leq m.$$ 
        We see that the sequence $\{l_n\}_{n\geq1}$ is given by $$l_n=k_{N+n}-\dfrac{N+2n}{2}$$ satisfies the recurrence of $\{k_n\}_{n\geq1}$ and with initial values $l_1=\dots=l_{m-1}=0$ and $l_m=1$. Hence, $l_n=k_n$ for all $n\geq 1$.
