We denote scalar values as   $\lambda$, vectors with bold letters as $\bm{\lambda}$, matrices with bold capital letters as $\mathbf{\Lambda}$ and random variables and random vector with the tilde notation as $\tilde{\xi}$ and $\tilde{\bm{\xi}}$. We use $\mathbf{1}$ to denote a vector of all ones. $\mathbb{R}_n$ denotes the set of $n$ dimensional real vectors while $\mathbb{R}_n^+$ denotes the set of nonnegative vectors of size $n$. $\mathbf{A} \succeq 0$ denotes a positive semidefinite matrix $\mathbf{A}$ and $\mathbf{A} \geq 0$ denotes a nonnegative matrix $\mathbf{A}$. The trace of a matrix $\mathbf{A}$ is denoted as $trace(\mathbf{A})$. The inner product  between two matrices of the same dimension is denoted as $\mathbf{A}\bullet\mathbf{\Lambda}=trace(\mathbf{A}^T\mathbf{\Lambda})$. <!-- lemma-1: body_md -->
\label{lemma:nphard}
The problem \eqref{sup_problem} is NP-hard even for the case where $\Xi = \{0,1\}^n$.
<!-- end lemma-1 -->
<!-- head-2: body_md -->
We will prove the result for the case of $K=1$. Then let $\mathbf{A} = \mathbf{A}_1, \mathbf{b} = \mathbf{b}_1$ and $c = c_1$. Denote by $\theta(\bm{\xi}) = \mathbb{P}(\bm{\tilde{\xi}} = \bm{\xi})$. Then the problem \eqref{sup_problem} can be written as a linear program over $2^n$ decision variables as follows,
\begin{align}
\max_{\theta} &\, \sum_{\bm{\xi} \in \{0,1\}^n}\theta(\bm{\xi}) (\bm{\xi}^T\mathbf{A}\bm{\xi}+\bm{\xi}^T\bm{b} +c) \label{exp_size_lp_k1} \\
\text{s.t} &\, \sum_{\bm{\xi} \in \{0,1\}^n}\theta(\bm{\xi}) = 1, \nonumber\\
&\, \sum_{\bm{\xi} \in \{0,1\}^n}\theta(\bm{\xi}) \xi_i = \mu_i, \ \forall i \in [n],  \nonumber \\
                                                                                                            
&\, \sum_{\bm{\xi} \in \{0,1\}^n}\theta(\bm{\xi}) \xi_i \xi_j \geq \Sigma_{ij}, \ \forall (i,j) \in \binom{[n]}{2},\\
&\, \theta(\bm{\xi}) \geq 0, \ \forall \bm{\xi} \in \{0,1\}^n.
\nonumber
\end{align}
Here we omit the constraint $\text{E}_\mathbb{P}[\mbox{diag}(\bm{\tilde{\xi}}\bm{\tilde{\xi}}^T)] = \mbox{diag}(\mathbf{\Sigma})$ since for Bernoulli random variables, $\tilde{\xi}_i^2 = \tilde{\xi}_i \forall i \in [n]$ and we must have $\mbox{diag}(\mathbf{\Sigma}) = \bm{\mu}$. The dual of \eqref{exp_size_lp_k1} can be written as,
\begin{align*}
    \min &\, \rho + \bm{\alpha}^{\top} \bm{\mu} + \bm{\Gamma} \bullet \Sigma \\
    &\, \rho + \bm{\alpha}^{\top} \bm{\xi} + \bm{\xi}^{\top}  \bm{\Gamma} \bm{\xi} \geq c + \bm{b}^{\top} \bm{\xi} + \bm{\xi}^{\top}  \bm{A}  \bm{\xi}, \  \forall \bm{\xi} \in \{0,1\}^n\\
    & \,\mbox{diag}(\bm{\Gamma}) = \bm{0}\\
    &\,  \bm{\Gamma} \leq 0.
\end{align*}
                                                                                                                                                                             
The separation problem thus boils down to,
\begin{align}
\label{separation_problem}
    \min_{\bm{\xi} \in \{0,1\}^n} (\rho -c ) + (\bm{\alpha} - \bm{b})^{\top}\bm{\mu} + \bm{\xi}^{\top} (\bm{\Gamma-A})\bm{\xi} 
\end{align}
with $\rho, c, \bm{\alpha}, \bm{b}, \bm{\Gamma}, \bm{A}$ fixed. Since the matrix $\bm{A}$ does not have any particular structure, the separation problem is an instance of a QUBO (quadratic unconstrained binary optimization) problem which is known to be NP-hard \cite{gareyJohnson}. Hence hardness of $\eqref{sup_problem}$ is proved by equivalence of separation and optimization \cite{GLS93}.

<!-- proposition-1: body_md -->
\label{prop:decomposition}
    For the decomposition of \eqref{eqn:cpp_decomposition}, define $\mathcal{T}^0=\{t\in[T]:\alpha_t=0\}$ and $\mathcal{T}^+=\{t\in[T]:\alpha_t>0\}$ where $T=\sum_{k\in[K]\cup\{0\}}L_k$. Then,
    \begin{enumerate}
        \par\hangindent 40\p@ $\bm{\eta}_t/\alpha_t=\bm{1}-\bm{\beta}_t/\alpha_t$ and $\bm{\beta}_t/\alpha_t\leq\bm{1}$ for $\forall t\in \mathcal{T}^+$,
        \par\hangindent 40\p@ $\bm{\eta}_t=\bm{\beta}_t=\bm{0}$ for $\forall t\in\mathcal{T}^0$.
    \end{enumerate}

<!-- head-3: body_md -->
From decomposition \eqref{eqn:cpp_decomposition}, we have for all $k \in [K]$
\begin{align*}
    &\lambda_{0,k}=\sum_{t=1 + T_k - 1}^{T_k}\alpha_t^2, \quad \bm{\lambda}_{1,k}=\sum_{t=1+T_{k-1}}^{T_k}\alpha_t\bm{\beta}_t, \quad \bm{\lambda}_{2,k}=\sum_{t=1+T_{k-2}}^{T_k}\alpha_t\bm{\eta}_t,\\
    &\mathbf{\Lambda}_{1,k}=\sum_{t=1+T_{k-1}}^{T_k}\bm{\beta}_t\bm{\beta}_t^T,\quad \mathbf{\Lambda}_{2,k}=\sum_{t=1+T_{k-1}}^{T_k}\bm{\eta}_t\bm{\beta}_t^T, \quad \text{and} \quad \mathbf{\Lambda}_{3,k}=\sum_{t=1+T_{k-1}}^{T_k}\bm{\eta}_t\bm{\eta}_t^T.
\end{align*}
Consider the constraints \eqref{C5} and \eqref{C6}:
\begin{align*}
\begin{array}{cc}
    \sum_{k\in[K]}\sum_{t=1+T_{k-1}}^{T_k}\alpha_t(\beta_t(i)+\eta_t(i))=1,\\
    \sum_{k\in[K]}\sum_{t=1+T_{k-1}}^{T_k}(\beta_t^2(i)+2\beta_t(i)\eta_t(i)+\eta_t(i)^2)=1
\end{array}
    \quad \Rightarrow \quad \begin{array}{cc}
         \sum_{t\in[T]}\alpha_t(\beta_t(i)+\eta_t(i))=1,\\
    \sum_{t\in[T]}(\beta_t(i)+\eta_t(i))^2=1
    \end{array}
\end{align*}
for $\forall i \in [n]$ where $T=\sum_{k\in[K]\cup\{0\}}L_k$.
\\From the constraint \eqref{C4},
\begin{align*}
    \left(\sum_{t\in[T]}\alpha_t(\beta_t(i)+\eta_t(i))\right)^2=\sum_{t\in[T]}\alpha_t^2\sum_{t\in[T]}(\beta_t(i)+\eta_t(i))^2
\end{align*}
By the equality conditions of Cauchy-Schwartz inequality, $\exists\delta_i$ such that $\delta_i\alpha_t=\beta_t(i)+\eta_t(i)$ for $\forall i\in[n]$ and $\forall t\in[T]$. In fact, from the results in \cite{Burer09}$, \delta_i=1$ for $\forall i \in [n]$. Then,
\begin{align*}
    \beta_t(i)+\eta_t(i)=0 \quad \forall t\in {\mathcal{T}^0} \quad &\Rightarrow \quad \beta_t(i)=\eta_t(i)=0
\end{align*}
as $\beta_t(i),\eta_t(i)\geq 0$. Hence, (ii) holds. Furthermore,
\begin{align*}
    \frac{\beta_t(i)}{\alpha_t}+\frac{\eta_t(i)}{\alpha_t}=1 \quad \forall t\in {\mathcal{T}^+} \quad &\Rightarrow \quad \frac{\eta_t(i)}{\alpha_t}=1-\frac{\beta_t(i)}{\alpha_t}\geq 0\\
    &\Rightarrow 0\leq \frac{\beta_t(i)}{\alpha_t} \leq 1.
\end{align*}
Hence, (i) holds.

We have 
\begin{align*}

\end{align*}
By Proposition \ref{prop:decomposition}(ii), $\zeta_t(i)\in[0,1]$ for $\forall i\in [n]$ and $\forall t\in\mathcal{T}_k^+, \forall k\in[K]$. Then, $\zeta_t(i)^2\leq\zeta_t(i)$.

\end{align*}
Since $p_t>0$ and $\zeta_t(i)-\zeta_t(i)^2\geq 0$ for $\forall t\in\mathcal{T}^+$, we have $\zeta_t(i)-\zeta_t(i)^2=0$. Hence, $\zeta_t(i)\in\{0,1\}$.

<!-- theorem-1: body_md -->
\label{thm:bernoulli}
    Suppose $\mathcal{P}$ is a non-empty set, then $value\eqref{sup_problem}=value\eqref{CPP}$ for  $\Xi = \{0,1\}^n$.
<!-- end theorem-1 -->
<!-- head-5: body_md -->
Due to the interpretation for the decision variables and constraints provided in \Cref{subsec:01support}, we know that $value\eqref{CPP}\geq value\eqref{sup_problem}$. Now it suffices to show that $value\eqref{CPP}\leq value\eqref{sup_problem}$. Consider the optimal solution to \eqref{CPP}, denoted as 
$
    \left\{(\lambda_{0,k}^\ast, \bm{\lambda}_{1,k}^\ast, \bm{\lambda}_{2,k}^\ast, \mathbf{\Lambda}_{1,k}^\ast,\mathbf{\Lambda}_{2,k}^\ast, \mathbf{\Lambda}_{3,k}^\ast)\right\}_{k\in[K]}.
$
The optimal solution has the following completely positive decomposition by Propositions \ref{prop:decomposition} and \ref{prop:binary},
\begin{align*}
    \begin{bmatrix}
        \lambda_{0,k}^\ast & {\bm{\lambda}^\ast_{1,k}}^T & {\bm{\lambda}^\ast_{2,k}}^T \\
        {\bm{\lambda}^\ast_{1,k}} & {\mathbf{\Lambda}_{1,k}^\ast} & {\mathbf{\Lambda}_{2,k}^\ast}^T \\
        {\bm{\lambda}^\ast_{2,k}} & {\mathbf{\Lambda}_{2,k}^\ast} & {\mathbf{\Lambda}_{3,k}^\ast}
    \end{bmatrix}=&\sum_{t\in\mathcal{T}_k^{+}} p_t^\ast\begin{pmatrix}
        1 \\ \bm{\zeta}_t \\ \bm{1}-\bm{\zeta}_t
    \end{pmatrix}\begin{pmatrix}
        1 \\ \bm{\zeta}_t \\ \bm{1}-\bm{\zeta}_t
    \end{pmatrix}^T
\end{align*}
where $\bm{\zeta}_t\in\{0,1\}^n$, $\forall t \in\mathcal{T}_k^+$ and $\forall k\in[K]$. The decomposition allows for the  construction of a feasible distribution.
                                
               
                                                                                                                                                                                 
                                                                                                                                                                    
                                              
             
In particular a distribution $\mathbb{P}^*$ over $\bm{\tilde{\xi}}$ can be constructed as follows:
\begin{enumerate}
    \par\hangindent 40\p@ Generate a random variable $\tilde{z} = k$ with probability $\lambda_{0,k}^*$.
    \par\hangindent 40\p@ Given the realization of $\tilde{z} =k$, generate $\bm{\tilde{\xi}}$ using the following conditional probability distribution, $\mathbb{P}(\bm{\tilde{\xi}} = \bm{\zeta_t}| \tilde{z} = k) = p_t^*/\lambda_{0,k}^\ast$ $ \forall t \in \mathcal{T}_k^+$.
\end{enumerate}
$\mathbb{P}^*$ is a valid distribution since,
$
    \sum_{k \in K} \lambda_{0,k}^* \sum_{t \in \mathcal{T}_k^+} p^*_t / \lambda^*_{0,k} = \sum_{t \in \mathcal{T}^+} p^*_t = 1.
$
The constructed distribution $\mathbb{P}^* \in \mathcal{P}$ since,
\begin{align*}
    \mathbb{E}_{\mathbb{P}^*}[\bm{\tilde{\xi}}]= \sum_{k \in [K]}\lambda_{0,k}^* \sum_{t \in \mathcal{T}_k^+} p^*_t \bm{\zeta_t}/ \lambda_{0,k} = \sum_{k \in [K]} \sum_{t \in \mathcal{T}_k^+} p^*_t \bm{\zeta}_t = \sum_{k \in [K]}\bm{\lambda}^*_{1,k} 
\end{align*} 
Constraint \eqref{C1} in  \eqref{CPP} ensures that $\sum_{k \in [K]}\lambda^*_{1,k}(i) \ = \mu_i \forall i \in [n] $.
                                                                                                 
               
                                                                                                                                                                                                                                                                  
             
                
                                                                                                                                                
Also, 
\begin{align*}
    \mathbb{E}_{\mathbb{P}^*}[\bm{\tilde{\xi}}\bm{\tilde{\xi}}^{\top}]= \sum_{k \in [K]} \lambda_{0,k}^* \sum_{t \in \mathcal{T}_k^+} p_t \bm{\zeta}_t \bm{\zeta_t}^{\top}/\lambda_{0,k} = \sum_{k \in [K]} \Lambda_{1,k}^*
\end{align*}
From  constraint \eqref{C2} in \eqref{CPP}, we have $\sum_{k \in [K]} \Lambda_{1,k}^* (i,i) = \Sigma(i,i)$ for all $i \in [n]$ and from constraint \eqref{C3} in \eqref{CPP}, we have $\sum_{k \in [K]} \Lambda_{1,k}^* (i,j) \geq \Sigma(i,j)$ for all $(i,j) \in \binom{[n]}{2}$.
                                                                                               
                                                                                                                                                                                                                                                                                                                   
             
              
               
                                                                                                                                                                                                         
             
Thus, the constructed distribution is a feasible solution to \eqref{sup_problem}, and further,
\begin{align*}
    value\eqref{sup_problem}& \geq\text{E}_{\mathbb{P}^*}\left[\max_{k\in[K]}\left\{\bm{\tilde \xi}^T\bm{A}_k\bm{\tilde \xi}+\bm{\tilde \xi}^T\bm{b}_k+c_k\right\}\right]\\
    & \geq \sum_{k\in[K]} \mathbb{P}^*(\tilde{z} = k)\left(\mathbb{E}_{\mathbb{P}^*}\left[\bm{\tilde \xi}\bm{\tilde \xi}^T| \tilde{z} = k \right]\bullet\mathbf{A}_k+\mathbb{E}_{\mathbb{P}^*}\left[\bm{\tilde \xi}| \tilde{z} = k\right]^T\bm{b}_k
    +c_k\right)\\
=&\sum_{k\in[K]}\lambda_{0,k}\left[\left(\sum_{t\in\mathcal{T}_k^+}\frac{p_t^\ast\bm{\zeta}_t\bm{\zeta}_t^T}{\lambda_{0,k}^\ast}\right)\bullet\mathbf{A}_k+\left(\sum_{t\in\mathcal{T}_k^+}\frac{p_t^\ast\bm{\zeta}_t}{\lambda_{0,k}^\ast}\right)^T\bm{b}_k+c_k\right]\\
&=\sum_{k\in[K]}\left[\bm{\Lambda}_{1,k}^*\bullet\mathbf{A}_k+ (\bm{\lambda}_{1,k}^*)^T\bm{b}_k+ \lambda_{0,k}^\ast c_k\right]\\
    &= value\eqref{CPP}.
\end{align*}
The first inequality is true because $\mathbb{P}^*$ is a feasible distribution, the second inequality arises since we use a specific value of $k$ instead of the maximizer.
               
                                                                                                                                                                            
                                                                                                                                                                                                                                              
                                            
                                                                                                                                                                                                                                                         
                        
             
Therefore since $value\eqref{sup_problem}\leq value\eqref{CPP}$ by construction, together with this proof $value\eqref{sup_problem}=value\eqref{CPP}$.

<!-- theorem-2: body_md -->
\label{thm:R_n_+_and_R_n}
    Suppose $\mathcal{P}$ is a non-empty set, then $value\eqref{sup_problem}=value\eqref{CPP}$ for supports $\mathbb{R}_n^+$ and $\mathbb{R}_n$.

<!-- head-6: body_md -->
Similar to the proof of Theorem \ref{thm:bernoulli}, it is sufficient to show that $value\eqref{CPP}\leq value\eqref{sup_problem}$ for both support cases. The proof of Theorem \ref{thm:R_n_+_and_R_n} relies on constructing a sequence of random vectors and feasible solutions with respect to the decomposition \eqref{eqn:gen_sep_decomp}.
 Consider the optimal solution to \eqref{CPP} denoted as $\left\{\left(\lambda_{0,k}^\ast, \bm{\lambda}_{1,k}^\ast, \mathbf{\Lambda}^\ast_{1,k}\right)\right\}_{k\in[K]}$. The optimal solution can be written in the decomposition \eqref{eqn:gen_sep_decomp} form as follows:
\begin{align*}
    \begin{bmatrix}
        \lambda_{0,k}^\ast & {\bm{\lambda}_{1,k}^\ast}^T\\
        {\bm{\lambda}_{1,k}^\ast} & {\mathbf{\Lambda}_{1,k}^\ast}
    \end{bmatrix}=\sum_{t\in\mathcal{T}_k^+}{(\alpha_t^\ast)}^2\begin{pmatrix}
        1 \\
        \bm{\beta}_t^\ast/\alpha_t^\ast
    \end{pmatrix}\begin{pmatrix}
        1 \\
        \bm{\beta}_t^\ast/\alpha_t^\ast
    \end{pmatrix}^T + \sum_{t\in\mathcal{T}_k^0}\begin{pmatrix}
        0 \\
        \bm{\beta}_t^\ast
    \end{pmatrix}\begin{pmatrix}
        0 \\
        \bm{\beta}_t^\ast
    \end{pmatrix}^T
\end{align*}
{\it Case (1):  $\Xi = \mathbb{R}^+_n$}. We construct a sequence of random vectors so that the limiting distribution of the sequence is feasible to \eqref{sup_problem}. Let $\epsilon\in(0,1)$, then construct the distribution as follows.
\begin{enumerate}
\par\hangindent 40\p@ Generate a random variable $\tilde{z} = k$ with probability $\lambda_{0,k}^\ast$.
\par\hangindent 40\p@ Given the realization of $\tilde{z} = k$, generate $\bm{\tilde{\xi}}_\epsilon$ using the following conditional distribution.
\begin{align*} \mathbb{P}^\ast\left(\tilde{\bm{\xi}}_\epsilon=\bm{\beta}_t^\ast/\alpha^\ast_t \middle| \tilde{z}=k\right)&=(1-\epsilon^2)(\alpha_t^*)^2/\lambda_{0,k}^* \quad \forall t\in\mathcal{T}_k^+ \text{ and } \forall k\in[K],\\
\mathbb{P}^\ast\left(\tilde{\bm{\xi}}_\epsilon=\frac{\bm{\beta}_t\sqrt{|\mathcal{T}^0|}}{\epsilon} \middle|\tilde{z}=k\right)&=\frac{\epsilon^2}{|\mathcal{T}^0|\lambda^*_{0,k}} \quad \forall t\in\mathcal{T}_k^0 \text{ and } \forall k\in[K]
\end{align*}
where $\mathcal{T}^0=\bigcup_{k\in[K]}\mathcal{T}_k^0$. 
\end{enumerate}
This is a valid distribution as
\begin{align*}
    \sum_{k\in[K]}\lambda_{0,k}^*\left[\sum_{t\in\mathcal{T}_k^+}\mathbb{P}^\ast\Bigg(\tilde{\bm{\xi}}_\epsilon=\frac{\bm{\beta}_t^\ast}{\alpha^\ast_t} \middle| \tilde{z}=k \right)&+\sum_{t\in\mathcal{T}_k^0}\mathbb{P}^\ast\left(\tilde{\bm{\xi}}_\epsilon=\frac{\bm{\beta}_t\sqrt{|\mathcal{T}^0|}}{\epsilon} \middle| \tilde{z}=k\right)\Bigg]\\
    =&(1-\epsilon^2)\sum_{t\in\mathcal{T}^+}(\alpha_t^\ast)^2+\epsilon^2\sum_{t\in\mathcal{T}^0}\frac{1}{|\mathcal{T}^0|}\\
    =&(1-\epsilon^2)+\epsilon^2\frac{|\mathcal{T}^0|}{|\mathcal{T}^0|} \quad \text{(due to \eqref{C3})}\\
    =&1.
\end{align*}
The mean of the conditional distribution of $\bm{\tilde \xi}_\epsilon$ given $\tilde{z} = k$ is
\begin{align*}
    \lim_{\epsilon\downarrow0} \mathbb{E}_{\mathbb{P}^\ast}\left[\bm{\tilde \xi}_\epsilon \middle|\tilde{z} = k \right]=&\lim_{\epsilon\downarrow0}\sum_{t\in\mathcal{T}_k^+}\frac{\bm{\beta}_t^\ast}{\alpha_t^\ast}P^\ast\Bigg(\hat{\bm{\xi}}_\epsilon=\frac{\bm{\beta}_t^\ast}{\alpha^\ast_t} | \tilde{z} = k\Bigg)\\
    &+\lim_{\epsilon\downarrow0}\sum_{t\in\mathcal{T}_k^0}\frac{\bm{\beta}_t^\ast\sqrt{|\mathcal{T}^0|}}{\epsilon}P^\ast\left(\hat{\bm{\xi}}_\epsilon=\frac{\bm{\beta}_t\sqrt{|\mathcal{T}^0|}}{\epsilon}\middle| \tilde{z} = k\right)\\
    =&\lim_{\epsilon\downarrow0}\sum_{t\in\mathcal{T}_k^+}\frac{\bm{\beta}_t^\ast}{\alpha_t^\ast \lambda_{0,k}^*}(1-\epsilon^2)(\alpha_t^\ast)^2+\lim_{\epsilon\downarrow0}\sum_{t\in\mathcal{T}_k^0}\frac{\bm{\beta}_t^\ast\sqrt{|\mathcal{T}^0|}}{\epsilon}\frac{\epsilon^2}{|\mathcal{T}^0|\lambda_{0,k}^*}\\
    =&\sum_{t\in\mathcal{T}_k^+}\alpha_t^\ast\bm{\beta}_t^\ast/\lambda_{0,k}^*
    \end{align*}
    and satisfies
    \begin{align*}
        \lim_{\epsilon\downarrow0}\text{E}_{\mathbb{P}^\ast}[\bm{\tilde \xi}_\epsilon]=\sum_{k\in[K]}\lambda_{0,k}^*\sum_{t\in\mathcal{T}_k^+}\alpha_t^\ast\bm{\beta}_t^\ast/\lambda_{0,k}^* \quad \Rightarrow \quad \sum_{t\in\mathcal{T}^+}\alpha_t^\ast\bm{\beta}_t^\ast = \bm{\mu}.
                                                        
    \end{align*}
    Similarly, the conditional second moment matrix is
    \begin{align*}
        \lim_{\epsilon\downarrow0} \text{E}_{\mathbb{P}^\ast}\left[\bm{\tilde \xi}_\epsilon\bm{\tilde \xi}_\epsilon^T| \tilde{z} = k\right]=&\lim_{\epsilon\downarrow0}\sum_{t\in\mathcal{T}_k^+}\frac{\bm{\beta}_t^\ast}{\alpha_t^\ast}\frac{\bm{\beta}_t^\ast}{\alpha_t^\ast}^T \frac{(1-\epsilon^2)(\alpha^\ast_t)^2}{\lambda_{0,k}^*}\\
    &+\lim_{\epsilon\downarrow0}\sum_{t\in\mathcal{T}_k^0}\frac{\bm{\beta}_t^\ast\sqrt{|\mathcal{T}^0|}}{\epsilon}\frac{{\bm{\beta}_t^\ast}^T\sqrt{|\mathcal{T}^0|}}{\epsilon}\frac{\epsilon^2}{|\mathcal{T}^0|\lambda_{0,k}^*}\\
    =&\sum_{t\in\mathcal{T}_k}{\bm{\beta}_t^\ast}{\bm{\beta}_t^\ast}^T/\lambda_{0,k}^*
    \end{align*}
    and satisfies
    \begin{align*}
        \lim_{\epsilon\downarrow0} \text{E}_{\mathbb{P}^\ast}\left[{\tilde \xi}_\epsilon(i){\tilde \xi}_\epsilon(i)\right]=\sum_{k\in[K]}\lambda_{0,k}^*\sum_{t\in\mathcal{T}_k}{{\beta}_t^\ast(i)}{{\beta}_t^\ast}(i)/\lambda_{0,k}^* \quad \Rightarrow \quad \sum_{t\in\mathcal{T}}{{\beta}_t^\ast(i)}{{\beta}_t^\ast}(i)=\Sigma_{ii} 
    \end{align*}
    for $\forall i\in[n]$ and
    \begin{align*}
        \lim_{\epsilon\downarrow0} \text{E}_{\mathbb{P}^\ast}\left[{\tilde \xi}_\epsilon(i){\tilde \xi}_\epsilon(j)\right]=\sum_{k\in[K]}\lambda_{0,k}^*\sum_{t\in\mathcal{T}_k}{{\beta}_t^\ast(i)}{{\beta}_t^\ast}(j)/\lambda_{0,k}^* \quad \Rightarrow \quad \sum_{t\in\mathcal{T}}{{\beta}_t^\ast(i)}{{\beta}_t^\ast}(j)\geq\Sigma_{ij}
    \end{align*}
    for $\forall(i,j) \in \binom{[n]}{2}$. Hence, the constructed sequence of random vectors are a feasible solution to the original problem \eqref{sup_problem}. Furthermore,
    \begin{align*}
        value\eqref{sup_problem}\geq& \lim_{\epsilon\downarrow0} \text{E}_{\mathbb{P}^\ast}\left[\max_{k\in[K]}\left\{\bm{\tilde{\xi}}_\epsilon^T\mathbf{A}_k\bm{\tilde{\xi}}_\epsilon+\bm{\tilde{\xi}}_\epsilon^T\bm{b}_k+c_k\right\}\right]\\
        \geq &\lim_{\epsilon\downarrow0} \sum_{k\in[K]}\mathbb{P}^*(\tilde{z} = k)\text{E}_{\mathbb{P}^\ast}\left[\bm{\tilde{\xi}}_\epsilon\bm{\tilde{\xi}}_\epsilon^T| \tilde{z} =k\right]\bullet\mathbf{A}_k\\
        &+\lim_{\epsilon\downarrow0}\sum_{k\in[K]}\mathbb{P}^*(\tilde{z} = k)\text{E}_{\mathbb{P}^\ast}\left[\bm{\tilde{\xi}}_\epsilon|\tilde{z} =k\right]^T\bm{b}_k+\lim_{\epsilon\downarrow0}\sum_{k\in[K]}\mathbb{P}^*(\tilde{z}=k)c_k\\
        =&\sum_{k\in[K]}\lambda_{0,k}^*\Bigg[\left(\sum_{t\in\mathcal{T}_k}\frac{{\bm{\beta}^\ast_t}{\bm{\beta}^\ast_t}^T}{\lambda_{0,k}^*}\right)\bullet\mathbf{A}_k+\left(\sum_{t\in\mathcal{T}^+_k} \frac{\alpha^\ast_t\bm{\beta}_t^\ast}{\lambda_{0,k}^*}\right)^T\bm{b}_k+c_k\Bigg]\\
        =&value\eqref{CPP}.
    \end{align*}
    The first inequality is true because $\mathbb{P}^*$ is a feasible distribution in the limit, the second inequality arises since we use a specific value of $k$ instead of the maximizer.\\
{\it Case (2):  $\Xi = \mathbb{R}_n$}. Here, we construct the following distribution $\mathbb{P}^*$ over $\bm{\tilde{\xi}}$.
    \begin{enumerate}
    \par\hangindent 40\p@ Generate a random variable $\tilde{z} = k$ with probability $\lambda_{0,k}^*$.
    \par\hangindent 40\p@ Given the realization of $\tilde{z} =k$, generate $\bm{\tilde{\xi}}$ from a normal distribution with mean $\bm{\lambda}^\ast_{1,k}/\lambda_{0,k}^\ast$ and second moment  $\mathbf{\Lambda}_{1,k}^\ast/\lambda_{0,k}^\ast$ 
\end{enumerate}
The distribution is a mixture of normal distributions and hence valid. Since,
\begin{align*}
        \mathbb{E}_{\mathbb{P}^\ast}[\bm{\tilde \xi}]=&\sum_{k\in[K]} \mathbb{E}_{\mathbb{P}^\ast}\left[\bm{\tilde \xi}|\tilde{z}=k\right]\mathbb{P}^\ast(\tilde{z}=k)
        =\sum_{k\in[K]} \frac{\bm{\lambda}^\ast_{1,k}}{\lambda_{0,k}^\ast}\lambda_{0,k}^\ast \quad \Rightarrow \sum_{k\in[K]}\bm{\lambda}^\ast_{1,k}  = \bm{\mu}
    \end{align*}
     \begin{align*}
        \mathbb{E}_{\mathbb{P}^\ast}[\tilde{\xi}_i\tilde{\xi}_j]&=\sum_{k\in[K]} \mathbb{E}_{\mathbb{P}^\ast}\left[{\tilde \xi}_i{\tilde \xi}_j^T|\tilde{z}=k\right]\mathbb{P}^\ast(\tilde{z}=k)=\sum_{k\in[K]}\frac{{\Lambda}_{1,k}^\ast(i,j)}{\lambda_{0,k}^\ast}\lambda_{0,k}^\ast \\
        & \quad \Rightarrow \quad \sum_{k\in[K]}{\Lambda}_{1,k}^\ast(i,i)= \Sigma_{ii}      \quad \forall i\in[n], \text{ and}\\
        & \quad \quad \quad \sum_{k\in[K]}{\Lambda}_{1,k}^\ast(i,j)\geq \Sigma_{ij}      \quad \forall(i,j)\in\binom{[n]}{2}.
    \end{align*}
                   
                                                           
                 
                                                                                                                                  
                   
                                                                                                                                           
                 
                              
                   
                                                                                                                                                                
                 
                                                                                                               
                  
                   
                                                                                                                                                                                   
                                                                                                                                                                                                      
                 
        
                   
                                                                                                                                                                                                              
                                                                                                                                                                              
                 
                                 
    We have $\mathbb{P}^* \in \mathcal{P}$. We also have,
    \begin{align*}
        value\eqref{sup_problem}& \geq \mathbb{E}_{\mathbb{P}^\ast}\left[\max_{k\in[K]}\left\{\bm{\tilde \xi}^T\mathbf{A}_k\bm{\tilde \xi}+\bm{\tilde \xi}^T\bm{b}_k+c_k\right\}\right]\\
        &\geq \sum_{k\in[K]}\lambda_{0,k}^\ast \left[\mathbb{E}_{\mathbb{P}^\ast}\left[\bm{\tilde{\xi}}\bm{\tilde{\xi}}^T|\tilde{z}=k\right]\bullet\mathbf{A}_k+\sum_{k\in[K]}\mathbb{E}_{\mathbb{P}^\ast}\left[\bm{\tilde{\xi}}|\tilde{z}=k\right]^T\bm{b}_k+ c_k\right]\\
        &=\sum_{k\in [K]}\left[\mathbf{\Lambda}_{1,k}^\ast\bullet\mathbf{A_k}+{\bm{\lambda}_{1,k}^\ast}^T\bm{b}_k+\lambda_{0,k}^\ast\right]=value\eqref{CPP}.
    \end{align*}
    Since $value\eqref{sup_problem}\leq value\eqref{CPP}$ by construct, $value\eqref{sup_problem}=value\eqref{CPP}$ for both supports.
