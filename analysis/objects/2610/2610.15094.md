<!-- thm-1: body_md -->
The random variable $\mathcal{T}$ has exponential distribution with probability density function \begin{equation}\mathbb{P}\big(\mathcal{T}\in dt\big)/dt=\lambda p\,e^{-\lambda p t},\qquad t\in\mathbb{R}^+.\label{lpdens}\end{equation}
<!-- end thm-1 -->
<!-- proof-1: body_md -->
We recall that, in view of the standard theory of Poisson processes, the interarrival times $T_k-T_{k-1}$, $k\in\mathbb{N}$, are independent and identically distributed exponential random variables with common characteristic function $$\mathbb{E}\Big[e^{i\alpha T_1}\Big]=\frac{\lambda}{\lambda-i\alpha},\qquad \alpha\in\mathbb{R}.$$
Thus, for $\alpha\in\mathbb{R}$, the identity (\ref{identityT}) permits us to express the characteristic function of $\mathcal{T}$ in the form
\begin{align}
\mathbb{E}\Big[e^{i\alpha \mathcal{T}}\Big]=&\mathbb{E}\Big[e^{i\alpha \sum_{k=1}^{N(\mathcal{T})}(T_{k}-T_{k-1})}\Big]=\mathbb{E}\left[\mathbb{E}\Big[e^{i\alpha T_1}\Big]^{N(\mathcal{T})}\right]\nonumber\\
=&\mathbb{E}\left[\left(\frac{\lambda}{\lambda-i\alpha}\right)^{N(\mathcal{T})}\right]=\frac{\lambda p}{\lambda p-i\alpha} \label{cflp}
\end{align}
where we have used the fact that $N(\mathcal{T})$ has a geometric distribution in the last step. The proof is completed by observing that the characteristic function (\ref{cflp}) is the Fourier transform of the probability density function (\ref{lpdens}).

For $n\in\mathbb{N}$ and $x\in\mathbb{R}$, it holds that

f_{0,2n+1}(x)=&\frac{\lambda e^{-\frac{\lambda}{c}\lvert x\lvert}}{4^n\,c}{2n-1\choose n-1}\nonumber\\

By defining the random variable \begin{equation}E_{2n+1}:=c\,(T_{2n+1}-T_{2n})\label{E2n+1def}\end{equation} and taking into account that the Poisson interarrival times are exponentially distributed, the probability density function of $E_{2n+1}$ is \begin{equation}h(x)=\frac{\lambda}{c}\,e^{-\frac{\lambda}{c} x},\qquad x\in\mathbb{R}^+.\label{taradatta}\end{equation} By conditioning equation (\ref{oddsumXT}) on the initial direction being $d_0$, we have that
$$X(\mathcal{T})\,\Big\lvert \{N(\mathcal{T})=2n+1,\,D(0)=d_0\}\;=\,E_{2n+1}+c\sum_{k=1}^{n}L_k$$ where the random variables $L_k,\,k=1,...,n$, are defined in formula (\ref{lkdef}). In other words, the conditional random variable $X(\mathcal{T})\,\Big\lvert \{N(\mathcal{T})=2n+1,\,D(0)=d_0\}$ coincides with the sum of $n$ independent Laplace random variables with common density function (\ref{lxmor}) and an exponentally distributed random variable with density (\ref{taradatta}). Thus, by taking into account that the sum of the Laplace variables has probability density function (\ref{sumlaplacedensity}), the conditional density of $X(\mathcal{T})$ can be expressed by means of the convolution integral
\begin{align}f_{0,2n+1}(x)&=\int_0^{+\infty}f_{2n}(x-z)\,h(z)\,dz\nonumber\\
&=\left(\frac{\lambda}{2c}\right)^n\frac{\lambda}{c\,(n-1)!}\,\sum_{k=0}^{n-1}\frac{(n-1+k)!}{(n-1-k)!\;k!}\;\left(\frac{c}{2\lambda}\right)^k\int_0^{+\infty}e^{-\frac{\lambda}{c} \lvert x-z \lvert-\frac{\lambda}{c} z}\,\lvert x-z\lvert^{n-k-1}\mathop{dz}.\label{intsts}\end{align}
To reformulate the expression (\ref{intsts}), we consider the cases of positive and negative values of $x$ separately.\\
\noindent For $x>0$, we can write that 
\begin{align}
f_{0,2n+1}(x)&=\left(\frac{\lambda}{2c}\right)^n\frac{\lambda\,e^{-\frac{\lambda}{c}x}}{c\,(n-1)!}\,\sum_{k=0}^{n-1}\frac{(n-1+k)!}{(n-1-k)!\;k!}\;\left(\frac{c}{2\lambda}\right)^k\int_0^{x}(x-z)^{n-k-1}\mathop{dz}\nonumber\\
&\qquad\qquad\qquad+\left(\frac{\lambda}{2c}\right)^n\frac{\lambda\,e^{-\frac{\lambda}{c}x}}{c\,(n-1)!}\,\sum_{k=0}^{n-1}\frac{(n-1+k)!}{(n-1-k)!\;k!}\;\left(\frac{c}{2\lambda}\right)^k\int_0^{+\infty}e^{-\frac{2\lambda}{c}z}\,z^{n-k-1}\mathop{dz}\nonumber\\
&=\left(\frac{\lambda}{2c}\right)^n\frac{\lambda\,e^{-\frac{\lambda}{c}x}}{c\,(n-1)!}\,\sum_{k=0}^{n-1}\frac{(n-1+k)!}{(n-k)!\;k!}\;\left(\frac{c}{2\lambda}\right)^k x^{n-k}+\frac{1}{4^n}\frac{\lambda\,e^{-\frac{\lambda}{c}x}}{c\,(n-1)!}\,\sum_{k=0}^{n-1}\frac{(n-1+k)!}{k!}\nonumber\\
&=\left(\frac{\lambda}{2c}\right)^n\frac{\lambda\,e^{-\frac{\lambda}{c}x}}{c\,(n-1)!}\,\sum_{k=0}^{n-1}\frac{(n-1+k)!}{(n-k)!\;k!}\;\left(\frac{c}{2\lambda}\right)^k x^{n-k}+\frac{\lambda\,e^{-\frac{\lambda}{c}x}}{c\,4^n}\,{2n-1\choose n-1},\qquad x>0\label{smokdisp}
\end{align}
where we have used the hockey-stick identity
\begin{equation}\sum_{j=0}^r{m+j \choose j}={m+r+1 \choose r}.\label{hockstiid}\end{equation}
Similarly, in the case $x<0$, it holds that 
\begin{align}
f_{0,2n+1}(x)&=\left(\frac{\lambda}{2c}\right)^n\frac{\lambda\,e^{\frac{\lambda}{c}x}}{c\,(n-1)!}\,\sum_{k=0}^{n-1}\frac{(n-1+k)!}{(n-1-k)!\;k!}\;\left(\frac{c}{2\lambda}\right)^k\int_0^{+\infty}e^{-\frac{2\lambda}{c}z}\,(z-x)^{n-k-1}\mathop{dz}\nonumber\\
&=\left(\frac{\lambda}{2c}\right)^n\frac{\lambda\,e^{\frac{\lambda}{c}x}}{c\,(n-1)!}\,\sum_{k=0}^{n-1}\frac{(n-1+k)!}{(n-1-k)!\;k!}\;\left(\frac{c}{2\lambda}\right)^k\sum_{j=0}^{n-k-1}{n-k-1 \choose j}\;j!\;\left(\frac{c}{2\lambda}\right)^{j+1}(-x)^{n-k-j-1}\nonumber\\
&=\left(\frac{\lambda}{2c}\right)^n\frac{\lambda\,e^{\frac{\lambda}{c}x}}{c}\,\sum_{r=0}^{n-1}\frac{(-x)^{n-r-1}}{(n-r-1)!}\;\left(\frac{c}{2\lambda}\right)^{r+1}\sum_{j=0}^{r}{n-1+r-j \choose r-j}\nonumber\\
&=\left(\frac{\lambda}{2c}\right)^n\frac{\lambda\,e^{\frac{\lambda}{c}x}}{c\,n!}\,\sum_{r=0}^{n-1}\frac{(n+r)!}{(n-r-1)!\;r!}\;\left(\frac{c}{2\lambda}\right)^{r+1}(-x)^{n-r-1},\qquad x<0\label{smokdisplan}
\end{align} where we used the identity (\ref{hockstiid}) in the last step. The result (\ref{delfmigun}) now follows by combining formulas (\ref{smokdisp}) and (\ref{smokdisplan}) in a suitable manner. Finally, the identity (\ref{delfmigdu}) is a direct consequence of the fact that $V(0)$ has distribution (\ref{V0dis}).

<!-- corollary-1: body_md -->
\label{cor:olap}For $n\in\mathbb{N}_0$ and $x\in\mathbb{R}$, it holds that
$$f_{2n+1}(x)=\left(\frac{\lambda}{2c}\right)^{n+1}e^{-\frac{\lambda}{c}\lvert x \lvert}\sum_{k=0}^{n}{n+k\choose k}\,\frac{1}{(n-k)!}\,\left(\frac{c}{2\lambda}\right)^{k}\lvert x\lvert ^{n-k}.$$
<!-- end corollary-1 -->
<!-- proof-3: body_md -->
For $n=0$, the result is trivial. For $n\ge1$, it holds that
\begin{align}
f_{2n+1}(x)=&\frac{1}{2}\big[f_{0,2n+1}(x)+f_{1,2n+1}(x)\big]=\frac{1}{2}\big[f_{0,2n+1}(x)+f_{0,2n+1}(-x)\big]\nonumber\\
=&\frac{\lambda e^{-\frac{\lambda}{c}\lvert x\lvert}}{4^n\,c}{2n-1\choose n-1}+\left(\frac{\lambda}{2c}\right)^{n+1}\frac{e^{-\frac{\lambda}{c}\lvert x\lvert}}{(n-1)!}\sum_{k=0}^{n-1}\frac{(n-1+k)!}{(n-k)!\,k!}\left(\frac{c}{2\lambda}\right)^k \lvert x\lvert^{n-k}\nonumber\\
&\qquad\qquad+\left(\frac{\lambda}{2c}\right)^{n+1}\frac{e^{-\frac{\lambda}{c}\lvert x\lvert}}{n!}\sum_{r=0}^{n-2}\frac{(n+r)!}{(n-r-1)!\,r!}\left(\frac{c}{2\lambda}\right)^{r+1}\lvert x\lvert^{n-r-1}\nonumber\\
=&\frac{\lambda e^{-\frac{\lambda}{c}\lvert x\lvert}}{4^n\,c}{2n-1\choose n-1}+\left(\frac{\lambda}{2c}\right)^{n+1}\frac{e^{-\frac{\lambda}{c}\lvert x\lvert}}{n!} \;\lvert x\lvert^{n}\nonumber\\
&\qquad\qquad+\left(\frac{\lambda}{2c}\right)^{n+1}\frac{e^{-\frac{\lambda}{c}\lvert x\lvert}}{(n-1)!}\sum_{k=1}^{n-1}\frac{(n-1+k)!}{(n-k)!\,k!}\left(\frac{c}{2\lambda}\right)^k \lvert x\lvert^{n-k}\nonumber\\
&\qquad\qquad+\left(\frac{\lambda}{2c}\right)^{n+1}\frac{e^{-\frac{\lambda}{c}\lvert x\lvert}}{n!}\sum_{r=0}^{n-2}\frac{(n+r)!}{(n-r-1)!\,r!}\left(\frac{c}{2\lambda}\right)^{r+1}\lvert x\lvert^{n-r-1}\nonumber\\
=&\frac{\lambda e^{-\frac{\lambda}{c}\lvert x\lvert}}{4^n\,c}{2n-1\choose n-1}+\left(\frac{\lambda}{2c}\right)^{n+1}\frac{e^{-\frac{\lambda}{c}\lvert x\lvert}}{n!} \;\lvert x\lvert^{n}\nonumber\\
&\qquad\qquad+\left(\frac{\lambda}{2c}\right)^{n+1}\,e^{-\frac{\lambda}{c}\lvert x\lvert}\,\sum_{k=1}^{n-1}{n+k\choose n}\frac{1}{(n-k)!}\left(\frac{c}{2\lambda}\right)^k \lvert x\lvert^{n-k}\nonumber\\
=&\left(\frac{\lambda}{2c}\right)^{n+1}\,e^{-\frac{\lambda}{c}\lvert x\lvert}\,\sum_{k=0}^{n}{n+k\choose n}\frac{1}{(n-k)!}\left(\frac{c}{2\lambda}\right)^k \lvert x\lvert^{n-k}\nonumber
\end{align}
where we have used the identity $${2n-1\choose n-1}=\frac{1}{2}{2n\choose n}.$$ This completes the proof.

By recalling that the number of direction changes $N(\mathcal{T})$ before the killing has geometric distribution (\ref{geometricdistribution}), and employing formula \ref{sumlaplacedensity}, we can write that
\begin{align}
f_{\mathrm{even}}(x)&=\sum_{n=1}^{\infty}p(1-p)^{2n-1}f_{2n}(x)\nonumber\\
&=p\,e^{-\frac{\lambda}{c}\lvert x\lvert}\sum_{n=1}^{\infty}\frac{(1-p)^{2n-1}}{(n-1)!}\sum_{k=0}^{n-1}\frac{1}{4^k}\left(\frac{\lambda}{2c}\right)^{n-k}\frac{(n-1+k)!}{(n-1-k)!\,k!}\lvert x\lvert^{n-k-1}\nonumber\\
&=p\,e^{-\frac{\lambda}{c}\lvert x\lvert}\sum_{k=0}^{\infty}\sum_{n=k+1}^{\infty}\frac{(1-p)^{2n-1}}{4^k}\left(\frac{\lambda}{2c}\right)^{n-k}\frac{(n-1+k)!}{(n-1-k)!\,k!\,(n-1)!}\lvert x\lvert^{n-k-1}\nonumber\\
&=p\,e^{-\frac{\lambda}{c}\lvert x\lvert}\sum_{k=0}^{\infty}\sum_{n=0}^{\infty}\frac{(1-p)^{2(n+k)+1}}{4^k}\left(\frac{\lambda}{2c}\right)^{n+1}\frac{(n+2k)!}{n!\,k!\,(n+k)!}\lvert x\lvert^n\nonumber\\
&=\frac{\lambda p(1-p)}{2c}\,e^{-\frac{\lambda}{c}\lvert x\lvert}\sum_{n=0}^{\infty}\frac{1}{n!}\left(\frac{\lambda(1-p)^2}{2c}\lvert x\lvert\right)^n\sum_{k=0}^{\infty}\binom{n+2k}{k}\left(\frac{(1-p)^2}{4}\right)^k.\label{stepleb}\end{align}
By using the identity (see Lebedev \cite{lebedev}, p. 253)
$$\sum_{k=0}^{\infty}{n+2k\choose k}t^k=\frac{1}{\sqrt{1-4t}}\left(\frac{2}{1+\sqrt{1-4t}}\right)^n\qquad\lvert t\lvert<\frac{1}{4}$$

\begin{align}f_{\mathrm{even}}(x)&=\frac{\lambda p(1-p)}{2c}\,e^{-\frac{\lambda}{c}\lvert x\lvert}\sum_{n=0}^{\infty}\frac{1}{n!}\left(\frac{\lambda(1-p)^2}{2c}\lvert x\lvert\right)^n\frac{1}{\sqrt{1-(1-p)^2}}\left(\frac{2}{1+\sqrt{1-(1-p)^2}}\right)^n\nonumber\\
&=\frac{\lambda(1-p)}{2c}\sqrt{\frac{p}{2-p}}\,e^{-\frac{\lambda}{c}\lvert x\lvert}\sum_{n=0}^{\infty}\frac{1}{n!}\left(\frac{\lambda}{c}\left(1-\sqrt{p(2-p)}\right)\lvert x\lvert\right)^n\nonumber\\
&=\frac{\lambda(1-p)}{2c}\sqrt{\frac{p}{2-p}}\exp\left\{-\frac{\lambda}{c}\sqrt{p(2-p)}\,\lvert x\lvert\right\}.\nonumber
\end{align}
Similarly, when the number of direction changes is odd, it holds that
\begin{align}
f_{\mathrm{odd}}(x)&=\sum_{n=0}^{\infty}p(1-p)^{2n}f_{2n+1}(x)\nonumber\\
&=\sum_{n=0}^{\infty}p(1-p)^{2n}\left(\frac{\lambda}{2c}\right)^{n+1}e^{-\frac{\lambda}{c}\lvert x\lvert}\sum_{k=0}^{n}\binom{n+k}{k}\left(\frac{c}{2\lambda}\right)^k\frac{\lvert x\lvert^{n-k}}{(n-k)!}\\
&=\frac{\lambda p}{2c}\,e^{-\frac{\lambda}{c}\lvert x\lvert}\sum_{k=0}^{\infty}\sum_{n=k}^{\infty}\frac{(1-p)^{2n}}{4^k}\left(\frac{\lambda}{2c}\right)^{n-k}\binom{n+k}{k}\frac{\lvert x\lvert^{n-k}}{(n-k)!}\nonumber\\
&=\frac{\lambda p}{2c}\,e^{-\frac{\lambda}{c}\lvert x\lvert}\sum_{k=0}^{\infty}\sum_{n=0}^{\infty}\frac{(1-p)^{2(n+k)}}{4^k}\left(\frac{\lambda}{2c}\right)^n\binom{n+2k}{k}\frac{\lvert x\lvert^n}{n!}\nonumber\\
&=\frac{\lambda p}{2c}\,e^{-\frac{\lambda}{c}\lvert x\lvert}\sum_{n=0}^{\infty}\frac{1}{n!}\left(\frac{\lambda(1-p)^2}{2c}\lvert x\lvert\right)^n\sum_{k=0}^{\infty}\binom{n+2k}{k}\left(\frac{(1-p)^2}{4}\right)^k\nonumber\\
&=\frac{\lambda p}{2c}\,e^{-\frac{\lambda}{c}\lvert x\lvert}\sum_{n=0}^{\infty}\frac{1}{n!}\left(\frac{\lambda(1-p)^2}{2c}\lvert x\lvert\right)^n\frac{1}{\sqrt{1-(1-p)^2}}\left(\frac{2}{1+\sqrt{1-(1-p)^2}}\right)^n\nonumber\\
&=\frac{\lambda}{2c}\sqrt{\frac{p}{2-p}}\,e^{-\frac{\lambda}{c}\lvert x\lvert}\sum_{n=0}^{\infty}\frac{1}{n!}\left(\frac{\lambda}{c}\left(1-\sqrt{p(2-p)}\right)\lvert x\lvert\right)^n\nonumber\\
&=\frac{\lambda}{2c}\sqrt{\frac{p}{2-p}}\exp\left\{-\frac{\lambda}{c}\sqrt{p(2-p)}\,\lvert x\lvert\right\}\nonumber
\end{align} which completes the proof.

<!-- proof-6: body_md -->
By defining the auxiliary function
\begin{equation}\widetilde{g}(x,t)=e^{\lambda(1-p)t}g(x,t)\label{botsw}\end{equation}
and taking into account equation (\ref{gpde}), it can be verified that $\widetilde{g}(x,t)$ satisfies the partial differential equation
\begin{equation}\frac{\partial^2\widetilde{g}}{\partial t^2}-c^2\frac{\partial^2\widetilde{g}}{\partial x^2}=\lambda^2(1-p)^2\widetilde{g}.\label{gtildeeq}\end{equation}
The change of variables
\begin{equation}z=\sqrt{c^2t^2-x^2}\label{besseltrans}\end{equation}
now transforms equation (\ref{gtildeeq}) into the Bessel-type equation
$$\frac{d^2\widetilde{g}}{dz^2}+\frac{1}{z}\frac{d\widetilde{g}}{dz}-\frac{\lambda^2(1-p)^2}{c^2}\widetilde{g}=0,$$
whose general solution is
$$\widetilde{g}(z)=A\, I_0\left(\frac{\lambda(1-p)}{c}z\right)+B\, K_0\left(\frac{\lambda(1-p)}{c}z\right).$$
We disregard the term involving the modified Bessel function of the second kind $K_0(z)$, which diverges at $z=0$. Since equation (\ref{gpde}) is linear and has constant coefficients, the time derivative of any of its solutions is again a solution. Therefore, after inverting transformations (\ref{besseltrans}) and (\ref{botsw}), we consider a solution of the form
\begin{equation}g(x,t)=e^{-\lambda(1-p)t}\left[A I_0\left(\frac{\lambda(1-p)}{c}\sqrt{c^2t^2-x^2}\right)+B\frac{\partial}{\partial t}I_0\left(\frac{\lambda(1-p)}{c}\sqrt{c^2t^2-x^2}\right)\right].\label{Besseleqdenstel}\end{equation}
In order to determine the coefficients $A$ and $B$, we recall that the support of the process $X(t)$ coincides with the interval $[-ct,ct]$, and the probability mass at the endpoints of this interval is given in formula (\ref{probmassendpts}). Hence, in the interior of the support, it holds that
\begin{equation}\int_{-ct}^{ct}g(x,t)\,dx=1-e^{-\lambda(1-p)t}.\label{intcodn}\end{equation}
In order to impose the condition (\ref{intcodn}), we use the identities
$$\int_{-ct}^{ct}I_0\left(\frac{\lambda(1-p)}{c}\sqrt{c^2t^2-x^2}\right)dx=\frac{2c}{\lambda(1-p)}\operatorname{sinh}\big(\lambda(1-p)t\big)$$
and
$$\int_{-ct}^{ct}\frac{\partial}{\partial t}I_0\left(\frac{\lambda(1-p)}{c}\sqrt{c^2t^2-x^2}\right)dx=2c\left[\operatorname{cosh}\big(\lambda(1-p)t\big)-1\right].$$
Therefore,
\begin{align}
\int_{-ct}^{ct}g(x,t)\,dx=e^{-\lambda(1-p)t}\Bigg\{&\frac{Ac}{\lambda(1-p)}\left[e^{\lambda(1-p)t}-e^{-\lambda(1-p)t}\right]\nonumber\\
&+Bc\left[e^{\lambda(1-p)t}+e^{-\lambda(1-p)t}-2\right]\Bigg\}.\label{allinecaz}
\end{align}
By comparing now formulas (\ref{intcodn}) and (\ref{allinecaz}), we immediately obtain that
$$A=\frac{\lambda(1-p)}{2c},\qquad B=\frac{1}{2c},$$
which completes the proof.

<!-- thm-7: body_md -->
\label{infcondchf}
For every $n\geq 1$, the characteristic function of $\big(X(\mathcal{T}),\;Y(\mathcal{T}))$ conditional on $N(\mathcal{T})=n$ is
\begin{equation}\mathbb{E}\left(e^{i[\alpha X(\mathcal{T})+\beta Y(\mathcal{T})]}\Big \lvert N(\mathcal{T})=n\right)=\left(\frac{\lambda}{\sqrt{\lambda^2+c^2(\alpha^2+\beta^2)}}\right)^n,\qquad \alpha,\beta\in\mathbb{R}.\label{infcondf}\end{equation}
<!-- end thm-7 -->
<!-- proof-8: body_md -->
Since the interarrival times $T_k-T_{k-1},\;k\ge1,$ are exponentially distributed, it holds that
\begin{align}
\mathbb{E}\left(e^{i[\alpha X(\mathcal{T})+\beta Y(\mathcal{T})]}\Big\lvert N(\mathcal{T})=n\right)
&=\mathbb{E}\left[\exp\left(i\alpha c\sum_{k=1}^{n}(T_k-T_{k-1})\cos\Theta_k+i\beta c\sum_{k=1}^{n}(T_k-T_{k-1})\sin(\Theta_k)\right)\right] \nonumber \\
&=\int_{0}^{\infty}ds_1\cdots\int_{0}^{\infty}ds_n\,\int_{0}^{2\pi}d\theta_1\cdots\int_{0}^{2\pi}d\theta_n\;\left(\frac{\lambda}{2\pi}\right)^n e^{-\lambda\sum_{k=1}^{n}s_k}\nonumber \\
&\qquad\qquad\qquad \cdot\exp\left(i\alpha c\sum_{k=1}^{n}s_k\cos\theta_k+i\beta c\sum_{k=1}^{n}s_k\sin\theta_k\right) \nonumber \\
&=\int_{0}^{\infty}ds_1\cdots\int_{0}^{\infty}ds_n\,\lambda^ne^{-\lambda\sum_{k=1}^{n}s_k}\prod_{k=1}^{n}\frac{1}{2\pi}\int_{0}^{2\pi}e^{i\alpha c s_k\cos\theta_k+i\beta c s_k\sin\theta_k}\,d\theta_k \nonumber \\
&=\int_{0}^{\infty}ds_1\cdots\int_{0}^{\infty}ds_n\,\lambda^ne^{-\lambda\sum_{k=1}^{n}s_k}\prod_{k=1}^{n}J_0\left(c s_k\sqrt{\alpha^2+\beta^2}\right) \nonumber
\end{align}
where we have used the following integral representation for the Bessel function (see Watson \cite{watson})
\begin{equation*}
J_0\left(x\sqrt{a^2+b^2}\right)=\frac{1}{2\pi}\int_{0}^{2\pi}\exp\left\{ix\left(a\cos\theta+b\sin\theta\right)\right\}\,d\theta,\qquad x\in\mathbb{R}.
\end{equation*}
By recalling now the Laplace transform of the Bessel function $J_0(\cdot)$, that is
\begin{equation*}
\int_{0}^{+\infty}e^{-y s}J_0(as)\,ds=\frac{1}{\sqrt{y^2+a^2}},\qquad a\in\mathbb{R},\,y>0
\end{equation*}
we can write that
\begin{align}
\mathbb{E}\left(e^{i[\alpha X(\mathcal{T})+\beta Y(\mathcal{T})]}\Big\lvert N(\mathcal{T})=n\right)&=\lambda^n\prod_{k=1}^{n}\int_{0}^{\infty}e^{-\lambda s_k}J_0\left(c s_k\sqrt{\alpha^2+\beta^2}\right)\,ds_k \nonumber \\
&=\lambda^n\prod_{k=1}^{n}\frac{1}{\sqrt{\lambda^2+c^2(\alpha^2+\beta^2)}} \nonumber \\
&=\left(\frac{\lambda}{\sqrt{\lambda^2+c^2(\alpha^2+\beta^2)}}\right)^n\nonumber
\end{align}
which completes the proof.

\exp\left\{-t-\frac{\lambda^2(x^2+y^2)}{4c^2t}+\frac{\lambda^2(1-p)^2(x^2+y^2)}{4c^2t}\right\}\,dt \nonumber\\
&=\frac{\lambda^2p(1-p)}{4\pi c^2}\int_0^\infty \frac{1}{t}
\exp\left\{-t-\frac{\lambda^2p(2-p)(x^2+y^2)}{4c^2t}\right\}\,dt \nonumber\\
&=\frac{\lambda^2p(1-p)}{2\pi c^2}\;K_0\!\left(\frac{\lambda}{c}\sqrt{p(2-p)}\,\sqrt{x^2+y^2}\right).\nonumber
\end{align}
As for the odd-order terms, by definition of $f_{\mathrm{odd}}(x,y)$, we can write that
\begin{align}
f_{\mathrm{odd}}(x,y)
&= \sum_{n=0}^{\infty} p(1-p)^{2n}\frac{1}{2^{n+\frac{1}{2}}\pi}\left(\frac{\lambda}{c}\right)^{n+\frac32}
\frac{(x^2+y^2)^{\frac{1}{2}(n-\frac{1}{2})}}{\Gamma\left(n+\frac{1}{2}\right)}
K_{n-\frac{1}{2}}\!\left(\frac{\lambda}{c}\sqrt{x^2+y^2}\right). \nonumber
\end{align}

By separating the term \(n=0\) from the remaining terms, and recalling that
\begin{equation}
K_{-\frac{1}{2}}(z)=K_{\frac{1}{2}}(z)= \sqrt{\frac{\pi}{2z}}\,e^{-z}\label{Khalfpm}
\end{equation}
we can write that
\begin{align}
f_{\mathrm{odd}}(x,y)
&= \frac{\lambda p}{2\pi c}\frac{e^{-\frac{\lambda}{c}\sqrt{x^2+y^2}}}{\sqrt{x^2+y^2}}+\sum_{n=1}^{\infty} p(1-p)^{2n}\frac{1}{2^{n+\frac{1}{2}}\pi}\left(\frac{\lambda}{c}\right)^{n+\frac{3}{2}}
\frac{(x^2+y^2)^{\frac{1}{2}(n-\frac{1}{2})}}{\Gamma\left(n+\frac{1}{2}\right)}
K_{n-\frac{1}{2}}\!\left(\frac{\lambda}{c}\sqrt{x^2+y^2}\right).\nonumber
\end{align}
The proof is complete by using the gamma duplication formula
$$\Gamma\left(n+\frac{1}{2}\right)= 2^{1-2n}\sqrt{\pi}\,\frac{\Gamma(2n)}{\Gamma(n)}$$ and the identity (\ref{papalameri}).

Denoting by $f(x,y)$ the inverse Fourier transform of (\ref{infiunccf}), it holds that
\begin{align}
f(x,y) &= \frac{1}{(2\pi)^2}\int_{-\infty}^{+\infty}\int_{-\infty}^{+\infty} e^{-i\alpha x-i\beta y}\frac{\lambda p}{\sqrt{\lambda^2+c^2(\alpha^2+\beta^2)}-\lambda(1-p)}\,d\alpha\,d\beta \nonumber\\
&= \frac{\lambda p}{(2\pi)^2}\int_0^{+\infty}\int_0^{2\pi}\frac{e^{-ixr\cos\theta-iyr\sin\theta}}{\sqrt{\lambda^2+c^2r^2}-\lambda(1-p)}\,r\,d\theta\,dr= \frac{\lambda p}{2\pi}\int_0^{+\infty}\frac{J_0\left(r\sqrt{x^2+y^2}\right)\,r}{\sqrt{\lambda^2+c^2r^2}-\lambda(1-p)}\,dr \nonumber\\
&= \frac{\lambda p}{2\pi}\int_0^{+\infty}\frac{rJ_0\left(r\sqrt{x^2+y^2}\right)\left[\sqrt{\lambda^2+c^2r^2}+\lambda(1-p)\right]}{c^2r^2+\lambda^2p(2-p)}\,dr \nonumber\\
&= \frac{\lambda^2p(1-p)}{2\pi}\int_0^{+\infty}\frac{rJ_0\left(r\sqrt{x^2+y^2}\right)}{c^2r^2+\lambda^2p(2-p)}\,dr+ \frac{\lambda p}{2\pi}\int_0^{+\infty}\frac{r\sqrt{\lambda^2+c^2r^2}\,J_0\left(r\sqrt{x^2+y^2}\right)}{c^2r^2+\lambda^2p(2-p)}\,dr \nonumber\\
&= \frac{\lambda^2p(1-p)}{2\pi c^2}K_0\left(\frac{\lambda\sqrt{p(2-p)}\sqrt{x^2+y^2}}{c}\right) + \frac{\lambda p}{2\pi}\int_0^{+\infty}\frac{r\,\sqrt{\lambda^2+c^2r^2}\,J_0\left(r\sqrt{x^2+y^2}\right)}{c^2r^2+\lambda^2p(2-p)}\,dr \nonumber\\
&= \frac{\lambda^2p(1-p)}{2\pi c^2}K_0\left(\frac{\lambda\sqrt{p(2-p)}\sqrt{x^2+y^2}}{c}\right)+ \frac{\lambda p}{2\pi}\int_0^{+\infty}\frac{r\,J_0\left(r\sqrt{x^2+y^2}\right)}{\sqrt{c^2r^2+\lambda^2}}\sum_{k=0}^{\infty}\left(\frac{\lambda^2(1-p)^2}{\lambda^2+c^2r^2}\right)^k\,dr.\nonumber
\end{align}
By reversing the order of summation and integration and using formula (\ref{JKform}), we finally obtain that
\begin{align}
f(x,y) &= \frac{\lambda^2p(1-p)}{2\pi c^2}K_0\left(\frac{\lambda\sqrt{p(2-p)}\sqrt{x^2+y^2}}{c}\right)+ \frac{\lambda p}{2\pi}\sum_{n=0}^{\infty}\left(\lambda^2(1-p)^2\right)^n\int_0^{+\infty}\frac{r\,J_0\left(r\sqrt{x^2+y^2}\right)}{(c^2r^2+\lambda^2)^{n+\frac{1}{2}}}\,dr\nonumber\\
&= \frac{\lambda^2p(1-p)}{2\pi c^2}K_0\left(\frac{\lambda\sqrt{p(2-p)}\sqrt{x^2+y^2}}{c}\right)\nonumber\\
&\qquad\qquad+ \frac{\lambda p}{2\pi c}\sqrt{\frac{2\lambda}{c\sqrt{x^2+y^2}}}\sum_{n=0}^{\infty}\left(\frac{\lambda(1-p)^2\sqrt{x^2+y^2}}{2c}\right)^{\!\!n}\; \frac{K_{n-\frac{1}{2}}\left(\frac{\lambda}{c}\sqrt{x^2+y^2}\right)}{\Gamma\left(n+\frac{1}{2}\right)}.\label{traum}
\end{align}
Isolating the first term of the infinite series in (\ref{traum}) yields
\begin{align}
f(x,y) &= \frac{\lambda^2p(1-p)}{2\pi c^2}K_0\left(\frac{\lambda\sqrt{p(2-p)}\sqrt{x^2+y^2}}{c}\right)\nonumber\\
&\qquad\qquad+ \frac{\lambda p}{2\pi c}\sqrt{\frac{2\lambda}{c\sqrt{x^2+y^2}}}\Bigg[\frac{1}{\sqrt{\pi}}\,K_{-\frac{1}{2}}\left(\frac{\lambda}{c}\sqrt{x^2+y^2}\right)\nonumber\\
&\qquad\qquad\qquad\qquad\qquad\qquad+\sum_{n=0}^{\infty}\left(\frac{\lambda(1-p)^2\sqrt{x^2+y^2}}{2c}\right)^{\!\!n+1}\; \frac{K_{n+\frac{1}{2}}\left(\frac{\lambda}{c}\sqrt{x^2+y^2}\right)}{\Gamma\left(n+\frac{3}{2}\right)}\Bigg].\nonumber
\end{align}
The identities (\ref{papalameri}) and (\ref{Khalfpm}) now imply that
\begin{align}
f(x,y) &= \frac{\lambda^2p(1-p)}{2\pi c^2}K_0\left(\frac{\lambda\sqrt{p(2-p)}\sqrt{x^2+y^2}}{c}\right)\nonumber\\
&\qquad\qquad+ \frac{\lambda p}{2\pi c}\;\frac{e^{-\frac{\lambda}{c}\sqrt{x^2+y^2}}}{\sqrt{x^2+y^2}}\left[1+\sqrt{\pi}\sum_{n=0}^{\infty}\left(\frac{\lambda(1-p)^2\sqrt{x^2+y^2}}{2c}\right)^{\!\!n+1}\; \frac{y_n\left(\frac{c}{\lambda \sqrt{x^2+y^2}}\right)}{\Gamma\left(n+\frac{3}{2}\right)}\right].\nonumber
\end{align}
By applying the gamma duplication formula, we finally obtain
\begin{align}
f(x,y) &= \frac{\lambda^2p(1-p)}{2\pi c^2}K_0\left(\frac{\lambda\sqrt{p(2-p)}\sqrt{x^2+y^2}}{c}\right)\nonumber\\
&\qquad\qquad+ \frac{\lambda p}{2\pi c}\;\frac{e^{-\frac{\lambda}{c}\sqrt{x^2+y^2}}}{\sqrt{x^2+y^2}}\left[1+\frac{1}{2}\sum_{n=0}^{\infty}\left(\frac{2\lambda(1-p)^2\sqrt{x^2+y^2}}{c}\right)^{\!\!n+1}\,\frac{n!}{(2n+1)!}\;y_n\left(\frac{c}{\lambda \sqrt{x^2+y^2}}\right)\right]\nonumber
\end{align}
which coincides with the probability density function (\ref{mutubenult}). This completes the proof.
