<!-- definition-1: body_md -->
\label{def:vstl}
For a fixed number of triangles $T$, the \emph{valid-STL} (VSTL) distribution assigns equal probability to every valid STL file.
Floats and attribute words are sampled independently and uniformly over their respective admissible encodings, giving $26T$ unconstrained bytes and $12T$ exponent pairs that form mutually independent groups.
<!-- end definition-1 -->
<!-- lemma-1: body_md -->
\label{lem:singular_count}
For every $v \in \mathcal{V}$ and every block of $T$ records, the count $\#(v)$ is distributed as the convolution of $\Lambda_{U,v}$ over the $26T$ unconstrained positions with $\Lambda_{\mathcal{X},v}$ over the $12T$ exponent pairs.
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{lem:joint_count}
For all distinct $v_0, v_1 \in \mathcal{V}$ and every block of $T$ records, the pair of counts $\bigl( \#(v_0), \#(v_1) \bigr)$ is jointly distributed as the two-dimensional convolution of $\Lambda_{U,v_0,v_1}$ over the $26T$ unconstrained positions with $\Lambda_{\mathcal{X},v_0,v_1}$ over the $12T$ exponent pairs.
<!-- end lemma-2 -->
<!-- proposition-1: body_md -->
\label{prop:moments}
For every triangle count $T$, the block entropy has mean~\eqref{eq:muterms} and variance~\eqref{eq:varterms} given by the cardinality-weighted sums over $\Gamma$
\begin{equation}\label{eq:mean}
\mu^{\mathrm{VSTL}}(T) = \sum_{\gamma \in \Gamma} \lvert \gamma \rvert\, \theta_\gamma,
\end{equation}
\begin{equation}\label{eq:var}
\begin{aligned}
\left[ \sigma^{\mathrm{VSTL}}(T) \right]^2 &= \sum_{\gamma \in \Gamma} \lvert \gamma \rvert \theta^{(2)}_\gamma + \sum_{\substack{\gamma \in \Gamma \\ \lvert \gamma \rvert \ge 2}} \lvert \gamma \rvert \left( \lvert \gamma \rvert - 1 \right) \theta_{\gamma\gamma} \\
&\quad + \sum_{\substack{\gamma_0, \gamma_1 \in \Gamma \\ \gamma_0 \neq \gamma_1}} \lvert \gamma_0 \rvert \lvert \gamma_1 \rvert \theta_{\gamma_0 \gamma_1} - \left[ \mu^{\mathrm{VSTL}}(T) \right]^{2} .
\end{aligned}
\end{equation}
<!-- end proposition-1 -->
<!-- corollary-1: body_md -->
\label{cor:uniform}
Under the uniform byte model, with $L=\lvert\mathcal{V}\rvert=256$, Proposition~\ref{prop:moments} reduces to
\begin{equation}\label{eq:wu}
\begin{gathered}
\mu^{\mathrm{uniform}}(T) = L \theta_{\mathcal{V}}, \\
\left[ \sigma^{\mathrm{uniform}}(T) \right]^{2} = L \theta^{(2)}_{\mathcal{V}} + L \left(L - 1 \right) \theta_{\mathcal{V}\mathcal{V}} - \left[ \mu^{\mathrm{uniform}}(T) \right]^{2}.
\end{gathered}
\end{equation}
<!-- end corollary-1 -->
<!-- proof-1: body_md -->
Assuming uniformity drops the validity constraint on the exponent pairs.
All byte values are then equivalent, so the partition $\Gamma$ reduces to a set of one category, that category being the alphabet $\mathcal{V}$ itself.
Substituting this partition into~\eqref{eq:mean} and~\eqref{eq:var} yields~\eqref{eq:wu}.

Equation~\eqref{eq:wu} therefore reproduces the mean and variance of~\cite[Sec.~3.3]{local_entropy_2013}.
<!-- end proof-1 -->
<!-- lemma-3: body_md -->
\label{lem:crossentropy}
Every block $S$ of $T$ records satisfies the identity
\begin{equation}\label{eq:crossentropy}
H(S) = \sum_{v \in \mathcal{V}} Q_S(v)\lambda \left( \bar{P}(v) \right) - D(Q_S \| \bar{P}) ,
\end{equation}
where
\begin{equation}\label{eq:kl}
D(Q_S \| \bar{P}) = - \sum_{v \in \mathcal{V}} Q_S(v) \lambda \left( \frac{Q_S(v)}{\bar{P}(v)} \right)
\end{equation}
is the Kullback--Leibler divergence in bits~\cite[Sec.~2.3]{cover_thomas_2006}.
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
It is convenient to note a property of the surprisal function~\eqref{eq:plugin} first: the quotient rule of the underlying logarithm carries over, so differences of surprisals collapse to ratios of their arguments. For positive $x$ and $y$,
\begin{equation}\label{eq:lambda_quotient_rule}
\lambda(x) - \lambda(y) = -\left( \log_2(x) - \log_2(y) \right) = -\log_2\left(\frac{x}{y}\right) = \lambda \left( \frac{x}{y} \right).
\end{equation}

By~\eqref{eq:pbarvals}, $\forall v \in \mathcal{V}, \bar{P}(v) > 0$; hence $\lambda(\bar{P}(v))$ is finite and the ratio $Q_S(v) / \bar{P}(v)$ is well defined.
Adding and subtracting $\lambda(\bar{P}(v))$ inside each summand of~\eqref{eq:plugin} then gives
\begin{equation}\label{eq:cross_entropy_divergence_decomposition_proof}
\begin{aligned}
H(S)
&= \sum_{v \in \mathcal{V}} \varphi \left( Q_S(v) \right) \\
&= \sum_{v \in \mathcal{V}} Q_S(v) \left[ \lambda \left( Q_S(v) \right) + \lambda \left( \bar{P}(v) \right) - \lambda \left( \bar{P}(v) \right) \right] \\
&= \sum_{v \in \mathcal{V}} Q_S(v) \lambda \left( \bar{P}(v) \right) + \sum_{v \in \mathcal{V}} Q_S(v) \lambda \left( \frac{Q_S(v)}{\bar{P}(v)} \right) \\
&= \sum_{v \in \mathcal{V}} Q_S(v) \lambda \left( \bar{P}(v) \right) - D(Q_S \| \bar{P}).
\end{aligned}
\end{equation}
Terms with $Q_S(v) = 0$ contribute nothing to $H(S)$, since $\varphi(0) = 0$; in the chain above, they likewise vanish because $Q_S(v)$ multiplies every summand and carries its zero through.
The identity is therefore applied only where both of its arguments are positive.
<!-- end proof-2 -->
<!-- lemma-4: body_md -->
\label{lem:score}
The two moments of the record score are
\begin{equation}\label{eq:Cmom}
\begin{gathered}
E[C] = 26 E \left[ \lambda \left( \bar{P}(B_U) \right) \right] + 12 E \left[ \lambda \left( \bar{P}(B_0) \right) + \lambda \left( \bar{P}(B_1) \right) \right] = 50 H(\bar{P}), \\
\operatorname{Var}[C] = 26 \operatorname{Var} \left[ \lambda \left( \bar{P}(B_U) \right) \right] + 12 \operatorname{Var} \left[ \lambda \left( \bar{P}(B_0) \right) + \lambda \left( \bar{P}(B_1) \right) \right].
\end{gathered}
\end{equation}
Consequently, the moments of the block average score are
\begin{equation}\label{eq:CSmom}
\begin{gathered}
E \left[ \bar{C}_S \right] = \frac{E[C]}{50} = H(\bar{P}), \\
\operatorname{Var} \left[ \bar{C}_S \right] = \frac{\operatorname{Var}[C]}{50 N},
\end{gathered}
\end{equation}
exactly and at every $T$.
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
Write $B_{U,r}$ for the multiset of the unconstrained bytes and $B_{\mathcal{X},r}$ for the multiset of its exponent pairs of the $r$-th record.
Regrouping the fifty terms of~\eqref{eq:score} along this split leaves the sum unchanged,
\begin{equation}\label{eq:Cgroups}
C_r = \sum_{b_U \in B_{U,r}} \lambda \left( \bar{P}(b_U) \right) + \sum_{(b_0, b_1) \in B_{\mathcal{X},r}} \left( \lambda \left( \bar{P}(b_0) \right) + \lambda \left( \bar{P}(b_1) \right) \right).
\end{equation}
Under (A2) the $38$ summands are mutually independent.

\emph{Mean.}
Linearity of expectation in~\eqref{eq:Cgroups} gives
\begin{equation}\label{eq:ECr}
\begin{aligned}
E[C] &= 26 E \left[ \lambda \left( \bar{P}(B_U) \right) \right] + 12 E \left[ \lambda \left( \bar{P}(B_0) \right) \right] + 12 E \left[ \lambda \left( \bar{P}(B_1) \right) \right] \\
&= 26 \sum_{v \in \mathcal{V}} P_{B_U}(v) \lambda \left( \bar{P}(v) \right) + 12 \sum_{v \in \mathcal{V}} P_{B_0}(v) \lambda \left( \bar{P}(v) \right) + 12 \sum_{v \in \mathcal{V}} P_{B_1}(v) \lambda \left( \bar{P}(v) \right) \\
&= \sum_{v \in \mathcal{V}} \left[ 26 P_{B_U} + 12 P_{B_0} + 12 P_{B_1} \right](v) \lambda \left( \bar{P}(v) \right) \\
&= 50 \sum_{v \in \mathcal{V}} \varphi \left( \bar{P}(v) \right) = 50 H(\bar{P}).
\end{aligned}
\end{equation}

The same contribution can be computed while keeping the exponent pair $(B_0,B_1)$ intact and evaluating it under its joint law $P_{B_{0,1}}$.
This approach uses the fact that~\eqref{eq:pb0} and~\eqref{eq:pb1} are the marginals of~\eqref{eq:pb01}, which Table~\ref{tab:marginals} verifies.
By extending the sum to $\mathcal{V}^2$ --- with no change in value, since $P_{B_{0,1}}$ vanishes off $\mathcal{X}$ --- and rearranging the resulting terms, we obtain the same expression:
\begin{equation}\label{eq:pairsplitsum}
\begin{gathered}
\sum_{(v_0, v_1) \in \mathcal{X}} P_{B_{0,1}}(v_0, v_1) \left( \lambda \left( \bar{P}(v_0) \right) + \lambda \left( \bar{P}(v_1) \right) \right) \\
= \sum_{v_0, v_1 \in \mathcal{V}} P_{B_{0,1}}(v_0, v_1) \lambda \left( \bar{P}(v_0) \right) + \sum_{v_0, v_1 \in \mathcal{V}} P_{B_{0,1}}(v_0, v_1) \lambda \left( \bar{P}(v_1) \right) \\
= \sum_{v_0 \in \mathcal{V}} \lambda \left( \bar{P}(v_0) \right) \sum_{v_1 \in \mathcal{V}} P_{B_{0,1}}(v_0, v_1) + \sum_{v_1 \in \mathcal{V}} \lambda \left( \bar{P}(v_1) \right) \sum_{v_0 \in \mathcal{V}} P_{B_{0,1}}(v_0, v_1) \\
= \sum_{v \in \mathcal{V}} P_{B_0}(v) \lambda \left( \bar{P}(v) \right) + \sum_{v \in \mathcal{V}} P_{B_1}(v) \lambda \left( \bar{P}(v) \right).
\end{gathered}
\end{equation}
Thus, for the mean, evaluating the pair jointly gives the same result as evaluating its two components through their marginal laws.

\emph{Variance.}
The $38$ groups of~\eqref{eq:Cgroups} are the finest partition of a record into independent parts, and variances add over them and over no coarser split.
Because validity couples the two bytes of every exponent pair, they must be treated jointly, with their contribution given by the variance of $\lambda(\bar{P}(B_0)) + \lambda(\bar{P}(B_1))$, which does not equal the sum of the corresponding single-byte variances, as shown in~\eqref{eq:paircov} below.
By the per-position marginals of Section~\ref{subsec:layout}, the $26$ unconstrained groups and the $12$ exponent pairs are identically distributed under~\eqref{eq:pu} and~\eqref{eq:pb01}, respectively, giving
\begin{equation}\label{eq:varCform}
\operatorname{Var}[C] = 26 \operatorname{Var} \left[ \lambda \left( \bar{P}(B_U) \right) \right] + 12 \operatorname{Var} \left[ \lambda \left( \bar{P}(B_0) \right) + \lambda \left( \bar{P}(B_1) \right) \right] .
\end{equation}
Expanding each of the two terms leaves
\begin{equation}\label{eq:varmoments}
\begin{gathered}
\operatorname{Var} \left[ \lambda \left( \bar{P}(B_U) \right) \right] = E \left[ \lambda \left( \bar{P}(B_U) \right)^2 \right] - \left( E \left[ \lambda \left( \bar{P}(B_U) \right) \right] \right)^2 , \\
\begin{aligned}
\operatorname{Var} \left[ \lambda \left( \bar{P}(B_0) \right) + \lambda \left( \bar{P}(B_1) \right) \right] &= E \left[ \left( \lambda \left( \bar{P}(B_0) \right) + \lambda \left( \bar{P}(B_1) \right) \right)^{2} \right] \\
&\quad - \left( E \left[ \lambda \left( \bar{P}(B_0) \right) + \lambda \left( \bar{P}(B_1) \right) \right] \right)^2 .
\end{aligned}
\end{gathered}
\end{equation}

Unlike the mean, the variance cannot be expressed in terms of $\bar{P}$ alone.
The squared means in~\eqref{eq:varmoments} are not linear in the three class marginals, so~\eqref{eq:pbar} cannot fold them into $\bar{P}$ as it did in~\eqref{eq:ECr}.
Moreover, expanding the exponent pair's second moment introduces the cross term $E \left[ \lambda \left( \bar{P}(B_0) \right) \lambda \left( \bar{P}(B_1) \right) \right]$, which depends on the joint law~\eqref{eq:pb01}, not on its marginals.
The evaluation of the variance therefore requires computing the per-class moments and the moments of the exponent pair under its joint law.

\Delta = H_\infty - H_\infty^{\mathrm{VSTL}} \approx 0.00008520391 \text{ bits}.
\end{equation}

For each $v \in \mathcal{V}$, its counts in the $T$ records are i.i.d.\ by Definition~\ref{def:vstl}, bounded, and have common mean $50 \bar{P}(v)$ by~\eqref{eq:pbar}.
The strong law of large numbers~\cite[Sec.~7.5]{grimmett_stirzaker_2001} gives $Q_S(v) \to \bar{P}(v)$ almost surely for each $v$; since $\mathcal{V}$ is finite, these coordinatewise convergences hold simultaneously, and hence $Q_S \to \bar{P}$ almost surely.
The continuity of $H$ on the probability simplex then gives $H(S) = H(Q_S) \to H(\bar{P}) = H_\infty^{\mathrm{VSTL}}$ almost surely.

Taking expectations in~\eqref{eq:crossentropy} and using $E[\bar{C}_S] = H(\bar{P})$ from Lemma~\ref{lem:score} gives~\eqref{eq:ceilingid}.
Since the divergence is nonnegative and vanishes only at $Q_S = \bar{P}$, its expectation is strictly positive, proving the inequality.
Indeed, for every finite $T$, the empirical byte distribution is not fixed at $\bar{P}$ and therefore differs from it with positive probability.

Finally, $\bar{P}$ is nonuniform, so $H(\bar{P}) = H_\infty^{\mathrm{VSTL}} < H_\infty$, and~\eqref{eq:deficit} follows.

As $T \to \infty$, the VSTL reference mean admits the following second-order expansion:

\mu^{\mathrm{VSTL}}(T) = {}& H_\infty^{\mathrm{VSTL}} - \frac{1}{100 \ln(2) N} \sum_{v \in \mathcal{V}} \frac{E[\tilde{X}_v^{2}]}{\bar{P}(v)} \\

\end{aligned}

By Proposition~\ref{prop:ceiling} and~\eqref{eq:ceilingid}, it suffices to expand $E \left[ D(Q_S \| \bar{P}) \right]$.

Equations~\eqref{eq:XvsQ} and~\eqref{eq:Xtilde} show that each histogram deviation is a normalized sum of centred i.i.d.\ record counts:
\begin{equation}\label{eq:devsum}
\begin{aligned}
\delta_{S,v} &= \frac{1}{N} \sum_{r=1}^{T} X_{v,r} - \bar{P}(v) \\
&= \frac{1}{N} \sum_{r=1}^{T} \left( X_{v,r} - 50 \bar{P}(v) \right) \\
&= \frac{1}{N} \sum_{r=1}^{T} \tilde{X}_{v,r}.
\end{aligned}
\end{equation}
Consequently, the moments of the histogram deviations can be expressed through moments of the centred record counts.

Raising the final identity in~\eqref{eq:devsum} to the $k$-th power yields a sum over $k$-tuples $(r_1, \ldots, r_k)$ of not necessarily distinct record indices, where each summand is a product of the form $\tilde{X}_{v,r_1} \cdots \tilde{X}_{v,r_k}$.
Independence factorizes the expectation of each term, with each represented record contributing a centred moment of order equal to its multiplicity.
Since $E[\tilde{X}_v]=0$, a term vanishes unless each of its constituent records has multiplicity at least two.
For $k \ge 2$,
\begin{equation}\label{eq:devexplicit}
\begin{aligned}
E \left[ \delta_{S,v}^{k} \right] &= \frac{1}{N^{k}} \sum_{r_1 = 1}^{T} \cdots \sum_{r_k = 1}^{T} E \left[ \tilde{X}_{v,r_1} \cdots \tilde{X}_{v,r_k} \right] \\
&= \frac{1}{N^{k}}\displaystyle \sum_{\ell = 1}^{\lfloor k/2 \rfloor} \binom{T}{\ell} \sum_{\substack{a_1, \ldots, a_\ell \ge 2 \\ a_1 + \cdots + a_\ell = k}} \binom{k}{a_1, \ldots, a_\ell} \prod_{i=1}^{\ell} E \left[ \tilde{X}_v^{a_i} \right].
\end{aligned}
\end{equation}
The integer $\ell$ is the number of distinct records represented in a surviving $k$-tuple, and the binomial coefficient $\binom{T}{\ell}$ counts the ways to select these $\ell$ records from the block.
The integer $a_i$ is the multiplicity of the $i$-th selected record, and the multinomial coefficient $\binom{k}{a_1,\ldots,a_\ell}$ counts the corresponding arrangements.

Since every $a_i\ge2$ and $a_1+\cdots+a_\ell=k$, at most $\lfloor k/2\rfloor$ distinct records can be represented in an arrangement of the $k$ factors.
Consequently, the upper limit on $\ell$ in~\eqref{eq:devexplicit} also determines the asymptotic order
\begin{equation}\label{eq:mean_prob_diff_asymptotics}
E \left[ \delta_{S,v}^{k} \right] = O \left( N^{- \lceil k / 2 \rceil} \right).
\end{equation}

Therefore, the deviation moments contributing through $O(N^{-2})$ in the Taylor expansion are
\begin{equation}\label{eq:dmoments}
\begin{aligned}
E \left[ \delta_{S,v}^{2} \right] = {}& \frac{E \left[ \tilde{X}_v^{2} \right]}{50 N}, \\
E \left[ \delta_{S,v}^{3} \right] = {}& \frac{E \left[ \tilde{X}_v^{3} \right]}{50 N^2}, \\
E \left[ \delta_{S,v}^{4} \right] = {}& \frac{3 E \left[ \tilde{X}_v^{2} \right]^{2}}{2500 N^2} + \frac{E \left[ \tilde{X}_v^{4} \right] - 3 E \left[ \tilde{X}_v^{2} \right]^{2} }{50 N^3}. \\
                                                                                                                                                                                                                                                                           
                                                                                                                                                                                                                                
                                                                                                                                           
\end{aligned}
\end{equation}

Equations~\eqref{eq:rembound} and~\eqref{eq:mean_prob_diff_asymptotics} now provide everything needed to choose an appropriate value for $K$ so that the remainder bound has the required asymptotic order.
The first equation involves an absolute moment and the second, a signed one; for odd $K+1$, the two are conceptually different: positive and negative deviations can cancel in the signed moment, whereas both contribute positively to the absolute moment.
However, the two coincide when $K+1$ is even, so we choose the smallest such power whose moment is $O(N^{-3})$, namely $K+1=6$, and hence fix $K=5$.
For this choice, the resulting bound is
\begin{equation}\label{eq:remorder}
\begin{gathered}
\sum_{v \in \mathcal{V}} \Bigl\lvert E \left[ R_{5,v} \right] \Bigr\rvert \le \sum_{v \in \mathcal{V}} \frac{E \left[ \delta_{S,v}^{6} \right]}{5 \ln(2) \bar{P}(v)^{5}} \\
\sum_{v \in \mathcal{V}} \Bigl\lvert E \left[ R_{5,v} \right] \Bigr\rvert = O(N^{-3}).
\end{gathered}
\end{equation}

Substituting~\eqref{eq:dmoments} into~\eqref{eq:dtaylor} determines the leading terms, while $O(N^{-3})$ absorbs all remaining terms and, by~\eqref{eq:remorder}, the remainder.

\end{gathered}
\end{equation}

The index $r_0$ identifies the record contributing $\tilde{C}$.
The integer $\ell$ is the number of additional distinct records represented among the $k$ deviation factors, and the binomial coefficient $\binom{T-1}{\ell}$ counts the ways to select them from the remaining records.
The integer $b$ is the multiplicity of $r_0$ among those factors, while $a_i$ is the multiplicity of the $i$-th additional record; the multinomial coefficient $\binom{k}{a_1,\ldots,a_\ell,b}$ counts the corresponding arrangements.

Since $b\ge1$ and every $a_i\ge2$, at most $\lfloor(k-1)/2\rfloor$ additional records can be represented among the $k$ deviation factors.
Consequently, the upper limit on $\ell$ in~\eqref{eq:covexplicit} determines the asymptotic order
\begin{equation}\label{eq:cov_prob_diff_asymptotics}
\operatorname{Cov} \left[ \bar{C}_S, \delta_{S,v}^{k} \right] = O \left( N^{- \lfloor k / 2 \rfloor - 1} \right).
\end{equation}

Therefore, the covariance terms contributing through $O(N^{-2})$ in~\eqref{eq:covexpand} are
\begin{equation}\label{eq:covterms}
\begin{aligned}
\operatorname{Cov} \left[ \bar{C}_S, \delta_{S,v}^{2} \right] &= \frac{E \left[ \tilde{C} \tilde{X}_v^{2} \right]}{50 N^2}, \\
\operatorname{Cov} \left[ \bar{C}_S, \delta_{S,v}^{3} \right] &= \frac{3 E \left[ \tilde{C} \tilde{X}_v \right] E \left[ \tilde{X}_v^{2} \right]}{2500 N^2} + \frac{E \left[ \tilde{C} \tilde{X}_v^{3} \right] - 3 E \left[ \tilde{C} \tilde{X}_v \right] E \left[ \tilde{X}_v^{2} \right]}{50 N^3}. \\
                                                                                                                                                                                                                                                                      
                                                                                                                                                                                                                                    
                                                                                                                                                                                          
                                                                                                                                                                                                                                                                                                                                                     
                                                                                                                                                                                                                                                                                                                                                                                                     
\end{aligned}
\end{equation}

We next consider the covariance between powers of the deviations at two byte values $v_0$ and $v_1$.
The covariance is obtained by subtracting the product of the two marginal moments from their joint moment:
\begin{equation}\label{eq:pair_cov}
\operatorname{Cov} \left[ \delta_{S,v_0}^{k_0}, \delta_{S,v_1}^{k_1} \right] = E \left[ \delta_{S,v_0}^{k_0} \delta_{S,v_1}^{k_1} \right] - E \left[ \delta_{S,v_0}^{k_0} \right] E \left[ \delta_{S,v_1}^{k_1} \right].
\end{equation}
It remains only to evaluate the joint moment, since equations~\eqref{eq:devexplicit}--\eqref{eq:dmoments} already determine the product of the marginal moments in~\eqref{eq:pair_cov}.

Substituting~\eqref{eq:devsum} for both deviations in $E \left[ \delta_{S,v_0}^{k_0} \delta_{S,v_1}^{k_1} \right]$ yields a sum over $(k_0+k_1)$-tuples of not necessarily distinct record indices, where each summand contains $k_0$ factors associated with $v_0$ and $k_1$ factors associated with $v_1$.
Independence factorizes the expectation of each term; a record represented $a$ times among the factors associated with $v_0$ and $b$ times among those associated with $v_1$ contributes the joint centred moment $E[\tilde{X}_{v_0}^{a}\tilde{X}_{v_1}^{b}]$.
Since $E[\tilde{X}_{v_0}]=E[\tilde{X}_{v_1}]=0$, a term vanishes unless every represented record carries at least two factors in total.
For $k_0, k_1 \ge 2$, we have
\begin{equation}\label{eq:varexplicit}
\begin{gathered}
E \left[ \delta_{S,v_0}^{k_0} \delta_{S,v_1}^{k_1} \right] = \frac{1}{N^{k_0+k_1}} \sum_{r_1 = 1}^{T} \cdots \sum_{r_{k_0+k_1} = 1}^{T} E \left[ \prod_{i=1}^{k_0} \tilde{X}_{v_0,r_i} \prod_{i=1}^{k_1} \tilde{X}_{v_1,r_{k_0+i}} \right] \\
= \frac{1}{N^{k_0+k_1}} \sum_{\ell = 1}^{\lfloor (k_0+k_1)/2 \rfloor} \binom{T}{\ell} \sum_{\substack{a_1 + \cdots + a_\ell = k_0 \\ b_1 + \cdots + b_\ell = k_1 \\ a_i, b_i \ge 0 \\ a_i + b_i \ge 2}} \binom{k_0}{a_1, \ldots, a_\ell} \binom{k_1}{b_1, \ldots, b_\ell} \prod_{i=1}^{\ell} E \left[ \tilde{X}_{v_0}^{a_i} \tilde{X}_{v_1}^{b_i} \right].
\end{gathered}
\end{equation}

The integer $\ell$ is the number of distinct records represented in a surviving $(k_0+k_1)$-tuple, and the binomial coefficient $\binom{T}{\ell}$ counts the ways to select them from the block.
The integers $a_i$ and $b_i$ are the multiplicities of the $i$-th selected record among the factors associated with $v_0$ and $v_1$, respectively; the two multinomial coefficients count the corresponding arrangements within the two groups.

Since every $a_i+b_i\ge2$ and their sum over all represented records is $k_0+k_1$, at most $\lfloor(k_0+k_1)/2\rfloor$ distinct records can appear.
Consequently, the upper limit on $\ell$ in~\eqref{eq:varexplicit} determines the asymptotic order
\begin{equation}\label{eq:pair_prob_diff_asymptotics}
E \left[ \delta_{S,v_0}^{k_0} \delta_{S,v_1}^{k_1} \right] = O \left( N^{- \lceil (k_0 + k_1) / 2 \rceil} \right)
\end{equation}

It follows from~\eqref{eq:mean_prob_diff_asymptotics} and~\eqref{eq:pair_prob_diff_asymptotics} that the product of the marginal moments is of no greater asymptotic order than the joint moment. Their difference~\eqref{eq:pair_cov} therefore obeys the same bound:
\begin{equation}\label{eq:pair_cov_asymptotics}
\operatorname{Cov} \left[ \delta_{S,v_0}^{k_0}, \delta_{S,v_1}^{k_1} \right] = O \left( N^{- \lceil (k_0 + k_1) / 2 \rceil} \right)
\end{equation}

The only joint moment required through $O(N^{-2})$ is
\begin{equation}\label{eq:varchi}
\begin{aligned}
E \left[ \delta_{S,v_0}^{2} \delta_{S,v_1}^{2} \right] &= \frac{E \left[ \tilde{X}_{v_0}^{2} \right] E \left[ \tilde{X}_{v_1}^{2} \right] + 2 E \left[ \tilde{X}_{v_0} \tilde{X}_{v_1} \right]^{2}}{2500 N^{2}} \\
&\quad + \frac{E \left[ \tilde{X}_{v_0}^{2} \tilde{X}_{v_1}^{2} \right] - E \left[ \tilde{X}_{v_0}^{2} \right] E \left[ \tilde{X}_{v_1}^{2} \right] - 2 E \left[ \tilde{X}_{v_0} \tilde{X}_{v_1} \right]^{2}}{50 N^{3}}. \\

                                                                                                                                                                                                                                                                                                                                                                                                                                   
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                          
\end{aligned}
\end{equation}

All remaining terms involve the Taylor remainder, whose variance provides the common bound needed for their analysis.
Since $R_{5,v}$ is square-integrable, we may use the standard identity $\operatorname{Var}[Y]=E[Y^{2}]-E[Y]^{2}$, which implies $\operatorname{Var}[Y] \le E[Y^{2}]$.
Squaring~\eqref{eq:rembound} at $K=5$, taking expectations, and applying~\eqref{eq:mean_prob_diff_asymptotics} with $k=12$ therefore determines the asymptotic order of $\operatorname{Var}[R_{5,v}]$:
\begin{equation}\label{eq:remsq}
\begin{gathered}
\operatorname{Var} \left[ R_{5,v} \right] \le E \left[ R_{5,v}^{2} \right] \le \frac{E \left[ \delta_{S,v}^{12} \right]}{25 \ln(2)^{2} \bar{P}(v)^{10}}, \\
\operatorname{Var} \left[ R_{5,v} \right] = O(N^{-6}).
\end{gathered}
\end{equation}

The next term to be analyzed is the covariance between the score fluctuation and the remainder.
The Cauchy--Schwarz inequality bounds the absolute value of the covariance between two random variables by the square root of the product of their variances~\cite[App.~10B]{papoulis_pillai_2002}.
Lemma~\ref{lem:score} shows that $\operatorname{Var}[\bar{C}_S]=O(N^{-1})$; combining this with the bound for $R_{5,v}$ in~\eqref{eq:remsq} gives
\begin{equation}\label{eq:cov_cs_r5}
\begin{gathered}
\Bigl\lvert \operatorname{Cov} \left[ \bar{C}_S, R_{5,v} \right] \Bigr\rvert \le \sqrt{\operatorname{Var} \left[ \bar{C}_S \right] \operatorname{Var} \left[ R_{5,v} \right]} \le \sqrt{\operatorname{Var} \left[ \bar{C}_S \right] E \left[ R_{5,v}^{2} \right]} \\
\Bigl\lvert \operatorname{Cov} \left[ \bar{C}_S, R_{5,v} \right] \Bigr\rvert = O \left( N^{-7/2} \right)
\end{gathered}
\end{equation}

Likewise, the same Cauchy--Schwarz inequality is applied to $\operatorname{Cov}[\delta_{S,v_0}^{k},R_{5,v_1}]$, with $v_0$ indexing the deviation and $v_1$ the remainder~\cite[App.~10B]{papoulis_pillai_2002}.
Equation~\eqref{eq:mean_prob_diff_asymptotics}, together with the general variance bound established above, shows that $\operatorname{Var} \left[ \delta_{S,v_0}^{k} \right] \le E \left[ \delta_{S,v_0}^{2k} \right] = O(N^{-k})$; combining this with the bound for $R_{5,v_1}$ in~\eqref{eq:remsq} gives
\begin{equation}\label{eq:cov_prob_diff_r5}
\begin{gathered}
\Bigl\lvert \operatorname{Cov} \left[ \delta_{S,v_0}^{k}, R_{5,v_1} \right] \Bigr\rvert \le \sqrt{\operatorname{Var} \left[ \delta_{S,v_0}^{k} \right] \operatorname{Var} \left[ R_{5,v_1} \right]} \\
\Bigl\lvert \operatorname{Cov} \left[ \delta_{S,v_0}^{k}, R_{5,v_1} \right] \Bigr\rvert = O \left( N^{- k / 2 - 3} \right)
\end{gathered}
\end{equation}

The final remainder contribution in~\eqref{eq:varDexpand} is the variance of the sum of $R_{5,v}$ over all byte values.
The finite-dimensional Cauchy--Schwarz inequality~\cite[Thm.~1.41]{apostol_1967} then bounds the squared sum before the expectation is taken:
\begin{equation}\label{eq:var_sum_r5}
\begin{gathered}
\operatorname{Var} \left[ \sum_{v \in \mathcal{V}} R_{5,v} \right] \le E \left[ \left( \sum_{v \in \mathcal{V}} R_{5,v} \right)^2 \right] \le \lvert \mathcal{V} \rvert \sum_{v \in \mathcal{V}} E \left[ R_{5,v}^{2} \right] \\
\operatorname{Var} \left[ \sum_{v \in \mathcal{V}} R_{5,v} \right] = O \left( N^{-6} \right) .
\end{gathered}
\end{equation}

Substituting the results from~\eqref{eq:covterms}, \eqref{eq:pair_cov}, \eqref{eq:dmoments}, and~\eqref{eq:varchi} into the expansions~\eqref{eq:covexpand} and~\eqref{eq:varDexpand} determines all contributions through order $N^{-2}$.
Equations~\eqref{eq:cov_prob_diff_asymptotics}, \eqref{eq:pair_cov_asymptotics}, and~\eqref{eq:remsq}--\eqref{eq:var_sum_r5} show that every remaining contribution is $O(N^{-3})$ or smaller.
Equation~\eqref{eq:varsplit}, together with Lemma~\ref{lem:score}, then gives~\eqref{eq:asyvar}.

<!-- proof-7: body_md -->
Assume uniformity --- then the validity constraint no longer couples the exponent bytes $B_0$ and $B_1$ in each float, so they become independent, while every byte marginal, along with $\bar{P}$, coincides with $P_{B_U}=\dfrac{1}{L}$.
Thus, a block consists of $N$ independent uniform bytes, and its entropy converges to $H_\infty^{\mathrm{uniform}}=\log_2 L=8$.
For finite $N$, however, concavity makes the plug-in entropy biased downward~\cite[Thm.~2.7.3]{cover_thomas_2006}, so its reference mean requires a correction, while its reference variance is determined by fluctuations in the byte counts.

Under this uniform model, for any $v\in\mathcal{V}$, $X_v\sim\mathrm{Binomial}(50,P_{B_U})$, so $E[X_v]=50P_{B_U}$.
Moreover, every byte contributes the constant surprisal $\lambda(P_{B_U})$, giving $C=50H_\infty^{\mathrm{uniform}}=E[C]$.
It follows that $\operatorname{Var}[C]=0$ and $\tilde{C}=0$; therefore, all moments containing $\tilde{C}$ vanish.

Substituting the binomial moments of $X_v$ into~\eqref{eq:asymean} yields the mean in~\eqref{eq:uniformvalues}.
This expansion agrees, at the corresponding orders, with the results of Miller~\cite{miller_1955}, Basharin~\cite[Eq.~(3)]{basharin_1959}, and, after conversion to bits, Harris~\cite[Eq.~(14)]{harris_1977}.

For distinct byte values $v_0\ne v_1$, independence of the $50$ uniform bytes gives the record-count covariance $E[\tilde{X}_{v_0}\tilde{X}_{v_1}]=-50P_{B_U}^{2}$.
Substituting this covariance and the binomial moments of $X_v$ into~\eqref{eq:asyvar} yields the variance in~\eqref{eq:uniformvalues}.
After conversion to bits, this expansion agrees with Harris~\cite[Eq.~(23)]{harris_1977}; Basharin~\cite[Eq.~(4)]{basharin_1959} likewise notes that the $N^{-1}$ term vanishes for equiprobable symbols.

<!-- corollary-3: body_md -->
\label{cor:clt}
As $T \to \infty$,
\begin{equation}\label{eq:clt}
\frac{H(S) - \mu^{\mathrm{VSTL}}(T)}{\sigma^{\mathrm{VSTL}}(T)} \xrightarrow{D} \mathcal{N}(0, 1),
\end{equation}
so, with $\Phi$ the standard normal distribution function, the two-sided acceptance interval for the global measure at significance level $\alpha$ is
\begin{equation}\label{eq:interval}
\mu^{\mathrm{VSTL}}(T) \pm \Phi^{-1} \left( 1 - \frac{\alpha}{2} \right) \sigma^{\mathrm{VSTL}}(T),
\end{equation}
where $\mu^{\mathrm{VSTL}}(T)$ and $\sigma^{\mathrm{VSTL}}(T)$ are the reference values of Propositions~\ref{prop:asymean} and~\ref{prop:asyvar}.
<!-- end corollary-3 -->
<!-- proof-8: body_md -->
Collect the centred record counts into the vector $\tilde{X}_{\langle r \rangle} = \bigl\langle \tilde{X}_{v,r} \bigr\rangle_{v \in \mathcal{V}}$.
These vectors are i.i.d.\ across records by Definition~\ref{def:vstl}, and equation~\eqref{eq:devsum} holds simultaneously in all $L = \lvert \mathcal{V} \rvert$ coordinates.
The central limit theorem for i.i.d.\ summands~\cite{lindeberg_1922},~\cite[Thm.~(5.10.4)]{grimmett_stirzaker_2001}, in its multivariate form in $\mathbb{R}^{L}$~\cite[Ex.~2.18]{vandervaart_1998}, therefore gives
\begin{equation}\label{eq:histclt}
\begin{gathered}
\sqrt{N} \delta_S = \frac{1}{\sqrt{N}} \sum_{r=1}^{T} \tilde{X}_{\langle r \rangle} \xrightarrow{D} \mathcal{N}_{L} \left( 0, \Sigma \right), \\
\Sigma = \left\langle \frac{E \left[ \tilde{X}_{v_0} \tilde{X}_{v_1} \right]}{50} \right\rangle_{v_0, v_1 \in \mathcal{V}},
\end{gathered}
\end{equation}
where $\Sigma$ is the finite limiting covariance matrix formed from the per-record moments~\eqref{eq:mixedadd} already evaluated for Proposition~\ref{prop:asyvar}.

For each $v \in \mathcal{V}$, $\bar{P}(v) > 0$ and lies in the domain on which $\varphi$ is smooth; therefore, entropy is differentiable at the expansion point.
Since entropy is separable across byte values, the $L$ entries of its gradient are the $L$ evaluations of the one-variable derivative~\eqref{eq:phider}, exactly as in~\eqref{eq:htaylor}.
Thus,
\begin{equation}\label{eq:gradient}
\nabla H \left( \bar{P} \right) = \left\langle \varphi' \left( \bar{P}(v) \right) \right\rangle_{v \in \mathcal{V}} = \left\langle \lambda \left( \bar{P}(v) \right) - \frac{1}{\ln(2)} \right\rangle_{v \in \mathcal{V}} .
\end{equation}

Because $H$ is differentiable at $\bar{P}$, the delta method~\cite[Thm.~3.1]{vandervaart_1998} transforms the multivariate convergence in~\eqref{eq:histclt} into a univariate normal limit. Its variance is the quadratic form $\nabla H(\bar{P})^{T} \Sigma \nabla H(\bar{P})$, which projects the histogram covariance along the entropy gradient.
Since $H(S) = H(Q_S)$ and $H(\bar{P}) = H_\infty^{\mathrm{VSTL}}$ by Proposition~\ref{prop:ceiling}, this gives
\begin{equation}\label{eq:deltasandwich}
\sqrt{N} \left( H(S) - H_\infty^{\mathrm{VSTL}} \right) \xrightarrow{D} \mathcal{N} \left( 0, \nabla H \left( \bar{P} \right)^{T} \Sigma \nabla H \left( \bar{P} \right) \right).
\end{equation}

To evaluate the limiting variance in~\eqref{eq:deltasandwich}, consider the inner product of the gradient~\eqref{eq:gradient} with the centred byte counts $\tilde{X}_{v}$ for a generic record.
This calculation follows an approach similar to~\eqref{eq:linterm}:
\begin{equation}\label{eq:gradcontract}
\begin{aligned}
\sum_{v \in \mathcal{V}} \varphi' \left( \bar{P}(v) \right) \tilde{X}_{v}
&= \sum_{v \in \mathcal{V}} \left( \lambda \left( \bar{P}(v) \right) - \frac{1}{\ln(2)} \right) \tilde{X}_{v} \\
&= \sum_{v \in \mathcal{V}} \lambda \left( \bar{P}(v) \right) \tilde{X}_{v} - \frac{1}{\ln(2)} \sum_{v \in \mathcal{V}} \tilde{X}_{v} \\
&= \sum_{v \in \mathcal{V}} \lambda \left( \bar{P}(v) \right) X_{v} - E \left[ \sum_{v \in \mathcal{V}} \lambda \left( \bar{P}(v) \right) X_{v} \right] \\
&= C - E[C] = \tilde{C}.
\end{aligned}
\end{equation}
 The constant term in $\varphi'(x)$ is eliminated because $\displaystyle \sum_{v \in \mathcal{V}} X_v = \sum_{v \in \mathcal{V}} E[X_v] = 50$ and therefore $\displaystyle \sum_{v \in \mathcal{V}} \tilde{X}_v = 0$.

Substituting the entries of $\Sigma$ from~\eqref{eq:histclt} into the quadratic form and applying bilinearity allows it to be written in terms of the inner product in~\eqref{eq:gradcontract}:
\begin{equation}\label{eq:sandwich}
\begin{aligned}
\nabla H \left( \bar{P} \right)^{T} \Sigma \, \nabla H \left( \bar{P} \right)
&= \frac{1}{50} \sum_{v_0 \in \mathcal{V}} \sum_{v_1 \in \mathcal{V}} \varphi' \left( \bar{P}(v_0) \right) E \left[ \tilde{X}_{v_0} \tilde{X}_{v_1} \right] \varphi' \left( \bar{P}(v_1) \right) \\
&= \frac{1}{50} E \left[ \left( \sum_{v_0 \in \mathcal{V}} \varphi' \left( \bar{P}(v_0) \right) \tilde{X}_{v_0} \right) \left( \sum_{v_1 \in \mathcal{V}} \varphi' \left( \bar{P}(v_1) \right) \tilde{X}_{v_1} \right) \right] \\
&= \frac{E \left[ \tilde{C}^{2} \right]}{50} = \frac{\operatorname{Var}[C]}{50}.
\end{aligned}
\end{equation}
Equation~\eqref{eq:deltasandwich} therefore becomes
\begin{equation}\label{eq:cltraw}
\sqrt{N} \left( H(S) - H_\infty^{\mathrm{VSTL}} \right) \xrightarrow{D} \mathcal{N} \left( 0, \frac{\operatorname{Var}[C]}{50} \right).
\end{equation}

Rewriting the left-hand side of~\eqref{eq:clt} in terms of the $\sqrt{N}$-scaled convergence about $H_\infty^{\mathrm{VSTL}}$ in~\eqref{eq:cltraw} gives
\begin{equation}\label{eq:cltquotient}
\frac{H(S) - \mu^{\mathrm{VSTL}}(T)}{\sigma^{\mathrm{VSTL}}(T)}
= \frac{\sqrt{N} \left( H(S) - H_\infty^{\mathrm{VSTL}} \right) - \sqrt{N} \left( \mu^{\mathrm{VSTL}}(T) - H_\infty^{\mathrm{VSTL}} \right)}{\sqrt{N} \, \sigma^{\mathrm{VSTL}}(T)} .
\end{equation}
The first term in the numerator converges in distribution by~\eqref{eq:cltraw}; the mean expansion~\eqref{eq:asymean} with the coefficients~\eqref{eq:meanconstants} shows that the second term in the numerator is $O \left( N^{-1/2} \right)$ and therefore vanishes as $T \to \infty$; the leading term of~\eqref{eq:asyvar} shows that the denominator converges to a positive constant:
\begin{equation}\label{eq:cltthree}
\begin{gathered}
\sqrt{N} \left( H(S) - H_\infty^{\mathrm{VSTL}} \right) \xrightarrow{D} \mathcal{N} \left( 0, \frac{\operatorname{Var}[C]}{50} \right), \\[16pt]
\sqrt{N} \left( \mu^{\mathrm{VSTL}}(T) - H_\infty^{\mathrm{VSTL}} \right) = O \left( N^{-1/2} \right) \to 0, \\[16pt]
N \left[ \sigma^{\mathrm{VSTL}}(T) \right]^2 = \frac{\operatorname{Var}[C]}{50} + O \left( N^{-1} \right) \to \frac{\operatorname{Var}[C]}{50}, \\
\therefore \sqrt{N} \sigma^{\mathrm{VSTL}}(T) \to \sqrt{\frac{\operatorname{Var}[C]}{50}}.
\end{gathered}
\end{equation}
Slutsky's theorem~\cite{slutsky_1925},~\cite[Exercise~(7.2.5)]{grimmett_stirzaker_2001}, applied to the numerator and denominator of~\eqref{eq:cltquotient}, together with the scaling rule for normal distributions~\cite[Example~(4.4.4)]{grimmett_stirzaker_2001}, therefore yields
\begin{equation}\label{eq:cltunit}
\frac{H(S) - \mu^{\mathrm{VSTL}}(T)}{\sigma^{\mathrm{VSTL}}(T)} \xrightarrow{D} \frac{\displaystyle \mathcal{N} \left( 0, \frac{\operatorname{Var}[C]}{50} \right)}{\displaystyle \sqrt{\frac{\operatorname{Var}[C]}{50}}} \sim \mathcal{N} \left( 0, 1 \right).
\end{equation}
This proves~\eqref{eq:clt}.

The central $1-\alpha$ region of $\mathcal{N}(0,1)$ is bounded by $\pm \Phi^{-1}(1-\alpha/2)$; applying these bounds to~\eqref{eq:clt} gives~\eqref{eq:interval}.

<!-- corollary-5: body_md -->
\label{cor:localclt}
Let $S_1, \ldots, S_k$ be pairwise disjoint blocks of $T_B^{\ast}$ consecutive records from a VSTL ciphertext.
Their average entropy has reference mean $\mu_{\bar{H}}^{\mathrm{VSTL}}$ and standard deviation $\sigma_{\bar{H}}^{\mathrm{VSTL}}$ given in~\eqref{eq:localref}.
Then, as the block count $k$ grows,
\begin{equation}\label{eq:localclt}
\frac{\bar{H}_{k, N_B^{\ast}}(S) - \mu_{\bar{H}}^{\mathrm{VSTL}}}{\sigma_{\bar{H}}^{\mathrm{VSTL}}} \xrightarrow{D} \mathcal{N}(0, 1) ,
\end{equation}
so the two-sided acceptance interval at level $\alpha$ is
\begin{equation}\label{eq:localinterval}
\mu_{\bar{H}}^{\mathrm{VSTL}} \pm \Phi^{-1}\!\left( 1 - \frac{\alpha}{2} \right) \sigma_{\bar{H}}^{\mathrm{VSTL}}.
\end{equation}
<!-- end corollary-5 -->
<!-- proof-11: body_md -->
By Lemma~\ref{lem:aligned}, the block entropies are independent and share the same distribution.
Since the variance in~\eqref{eq:localref} is finite and positive, the central limit theorem~\cite{lindeberg_1922},~\cite[Thm.~(5.10.4)]{grimmett_stirzaker_2001} gives~\eqref{eq:localclt}.

The central $1-\alpha$ region of $\mathcal{N}(0,1)$ is bounded by $\pm \Phi^{-1}(1-\alpha/2)$; applying these bounds to~\eqref{eq:localclt} gives~\eqref{eq:localinterval}.
