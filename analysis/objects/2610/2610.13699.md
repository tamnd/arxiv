\textbf{\label{TR}}Suppose $a\leq 1$, $b>0$\ and $c\geq \max \left\{
-a,b\right\} $. If $c\geq \max \left\{ 1-2a,2b\right\} $, then 
\begin{equation}
\left[ F\left( -a,b;c;x\right) \right] ^{1/a}\geq M_{p}\left( 1-x;\frac{b}{c} 
\right)  \label{Hm>Pm-p}
\end{equation} 
\ for all $x\in \left( 0,1\right) $ if and only if $p\leq p_{0}$, where 
\begin{equation}
p_{0}=\frac{a+c}{1+c}.  \label{p0}
\end{equation} 
If $c\leq \min \left\{ 1-2a,2b\right\} $, then 
\begin{equation}
\left[ F\left( -a,b;c;x\right) \right] ^{1/a}\leq M_{p}\left( 1-x;\frac{b}{c} 
\right)  \label{Hm<Pm-p}
\end{equation} 
for all $x\in \left( 0,1\right) ,$if and only if $p\geq p_{0}$. If $ 
c=1-2a=2b $; or $c=\max \left\{ -a,b\right\} $; or $a=1$, then the equality
holds.

\textbf{Case 3.2}: $\delta \left( p_{0}^{\ast }\right) =0$, $\Delta \gamma
_{1}\left( p_{0}^{\ast }\right) <0$ and $\gamma _{1}\left( p_{0}^{\ast
}\right) =0$. By Lemma \ref{L-sgngn-p} (i) we see that $\gamma _{n}\leq 0$
for all $n\geq 1$. Using the same technique as Case 1, we can prove that $ 
\left\{ \xi _{n}\right\} _{n\geq 0}$ is a PN-type sequence, and the
assertion (\ref{A-gp0*-PN}) holds.

\textbf{Case 4}: $\delta \left( p_{0}^{\ast }\right) <0$, $\Delta \gamma
_{1}\left( p_{0}^{\ast }\right) >0$ and $\gamma _{1}\left( p_{0}^{\ast
}\right) \geq 0$. From Lemma \ref{L-sgngn-p} (ii) there is an $n_{1}\geq 1$
such that $\gamma _{n}>0$ for $1\leq n\leq n_{1}-1$ and $\gamma _{n}<0$ for $ 
n\geq n_{1}+1$. By using the same technique as Case 2, we can prove that $ 
\left\{ \xi _{n}\right\} _{n\geq 0}$ is a PN-type sequence, and the
assertion (\ref{A-gp0*-PN}) holds.

Finally, we prove that $p_{0}^{\ast }$ is the best possible. In fact,
solving the inequality 
\begin{equation*}
\lim_{x\rightarrow 1}\frac{\left[ F\left( -a,b;c;x\right) \right] ^{1/a}}{ 
M_{p}\left( 1-x,b/c\right) }\geq 1.
\end{equation*} 
for $p$ gives $p\leq p_{0}^{\ast }$; this together with the increasing
property of $p\mapsto M_{p}\left( 1-x,b/c\right) $ implies that $p_{0}^{\ast
}$ can not be replaced by any larger number than $p_{0}^{\ast }$. This
completes the proof.
