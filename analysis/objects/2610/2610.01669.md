\label{thm:adaptive-moment-locality}
			Let $(\mathcal{L},\mathcal{T}_f)$ be the dissipative Bose--Hubbard model \eqref{eq:lindblad} on a lattice $(V,E)$ of uniformly bounded maximum degree $\Delta_G$, weight satisfying $Z_\kappa^*<\infty$, and hopping coefficients $J_*=\sup_{\{i,j\}\in E}|J_{ij}|<\infty$ for any $\ell>2$ and loss rates $0<\gamma_{\min}\le\gamma_j\le\gamma_{\max}<\infty$.
			Then, there are constants $C,v_{\mathrm{LR}}>0$, depending only on $\Delta_G$, such that, for every $\lambda\ge2$, $p>2$, $t>0$, $X\subset R\subseteq V$, and $A\in\mathcal{B}(\mathcal{H}_X)$,
			\begin{equation}\label{eq:adaptive-physical-buffer-bound}
			\begin{aligned}
				\|\tau_t(A)-\tau_t^R(A)\|_\infty
				&\le C J_*|E_R|\,\|A\|_\infty
					\lambda^{-(p-2)/2}I_p(t)\\
				&\quad+C|\partial_E R|\,\|A\|_\infty e^{-d(X,\partial_E R)}
					\left(e^{v_{\mathrm{LR}}J_*(\lambda+1)I_p(t)}-1\right)
			\end{aligned}
			\end{equation}
			for the constants $\Delta_{\ell,p}=\frac{2}{\gamma_{\min}\ell}\Bigl(\gamma_{\max}\frac{(\ell+1)\ell^2}{2}+2^ppJ_*\Delta_G(1+e^\kappa)\Bigr)$ and
			\begin{equation}\label{eq:adaptive-moment-root}
				I_p(t)=(Z_\kappa^*)^{1/p}\left[
					\frac{\ell-1}{\ell-2}
					\left(\frac{2p}{(\ell-1)\gamma_{\min}\ell}\right)^{1/(\ell-1)}
					t^{(\ell-2)/(\ell-1)}+\Delta_{\ell,p}t\right].
			\end{equation}
			If $J_*>0$ and $\infty>d(X,\partial_E R)\ge8v_{\mathrm{LR}}J_*I_p(t)$, then
			\begin{equation}\label{eq:adaptive-distance-bound}
			\begin{aligned}
				\|\tau_t(A)-\tau_t^R(A)\|_\infty
				&\le C J_*|E_R|\,\|A\|_\infty I_p(t)\!
					\left(\frac{4v_{\mathrm{LR}}J_*I_p(t)}{d(X,\partial_E R)}\right)^{\!\!(p-2)/2}\!\!\!\!\!\!+C|\partial_E R|\,\|A\|_\infty e^{-d(X,\partial_E R)/2}.
			\end{aligned}
			\end{equation}
			The estimates are uniform in the initial state and the finite volume.

\label{thm:uniform-lrb}
		Under the hypotheses of Theorem~\ref{thm:adaptive-moment-locality}, let $2<p<2(\ell-1)$ and set
		\begin{equation}\label{eq:theta-p-ell}
				\Theta_{p,\ell}(t)
				=
				\frac{1}{1-\frac{p}{2(\ell-1)}}
				\left(\frac{2p}{(\ell-1)\gamma_{\min}\ell}\right)^{\frac{p}{2(\ell-1)}}
				t^{1-\frac{p}{2(\ell-1)}}+\Delta_{\ell,p}^{p/2}t\,.
			\end{equation}
			Here and below, $C_{p,\kappa}$ denotes a finite constant depending only on $p$ and the fixed Sobolev weights, coming from the scalar truncation estimate and the normalization factor $Z_\kappa^*$. Then, for every $A\in\mathcal{B}(\mathcal{H}_X)$, every $X\subset R\subseteq V$, every integer $M\ge2$, and every $t>0$,
			\begin{equation}\label{eq:main-uniform-lrb}
			\begin{aligned}
				\|\tau_t(A)-\tau_t^{R,M}(A)\|_\infty
				&\le 2C_{p,\kappa}J_* |E_R|\,\|A\|_\infty
				M^{-\frac{p-2}{2}}\Theta_{p,\ell}(t)\\
				&\quad +C |\partial_E R|\,\|A\|_\infty
				e^{-d(X,\partial_E R)}\bigl(e^{v_{\mathrm{LR}} J_*Mt}-1\bigr).
			\end{aligned}
			\end{equation}
			In particular, if $J_*t>0$, $d(X,\partial_E R)<\infty$, and this distance is larger than a constant multiple of $J_*t$, an integer choice of $M\ge2$ with $v_{\mathrm{LR}}J_*Mt\le d(X,\partial_E R)/2$ and $M\gtrsim d(X,\partial_E R)/(J_*t)$ gives
			\begin{equation}\label{eq:main-uniform-lrb-optimized}
				\|\tau_t(A)-\tau_t^{R,M}(A)\|_\infty
				\le
				\widetilde C_{p,\kappa} J_*\|A\|_\infty |E_R|\Theta_{p,\ell}(t)
				\left(\frac{J_*t}{d(X,\partial_E R)}\right)^{\frac{p-2}{2}}
				+C\|A\|_\infty |\partial_E R|e^{-d(X,\partial_E R)/2}.
			\end{equation}
			Equivalently, for a prescribed tolerance $\varepsilon\in(0,1)$, the volume factor in the particle-number tail can be absorbed into the cutoff. Namely, if
			\begin{equation}\label{eq:main-cutoff-accuracy-choice}
				M\ge
				\left(
				\frac{4C_{p,\kappa}J_*|E_R|\Theta_{p,\ell}(t)}{\varepsilon}
				\right)^{\frac{2}{p-2}}
			\end{equation}
			and
			\begin{equation}\label{eq:main-buffer-accuracy-condition}
				d(X,\partial_E R)\ge
				2v_{\mathrm{LR}}J_*Mt+\log\!\left(\frac{2C|\partial_E R|}{\varepsilon}\right),
			\end{equation}
			then the right side of Equation~\eqref{eq:main-uniform-lrb} is at most $\varepsilon\|A\|_\infty$. The estimate is in the uniform topology, i.e. in operator norm, and the constants are independent of the local Hilbert-space dimension and of the initial state.

\label{cor:adaptive-superpolynomial-locality}
		On nearest-neighbor subgraphs of $\mathbb Z^D$, fix a finite support $X$ and $T>0$. Under the hypotheses of Theorem~\ref{thm:adaptive-moment-locality}, let $R=X[r]$ be the graph ball of integer radius $r\ge0$ around $X$. There are constants $C,c>0$ such that, for every $q>0$ and $A\in\mathcal{B}(\mathcal{H}_X)$,

  \ifx(N_x+\1)^k e^{s\mathcal{L}_R^\gamma}(\rho)\empty \else \left[ (N_x+\1)^k e^{s\mathcal{L}_R^\gamma}(\rho) \right] \fi^{1/k}
				\le C_k^{\mathrm{mom}}\bigl((\gamma s)^{-1/3}+1\bigr)
				\le b_k(s)\coloneqq C_k^{\mathrm{mom}}(s^{-1/3}+1)
		\end{equation}
		for every state $\rho$, $s>0$, $x\in R$, and $C_k^{\mathrm{mom}}\ge1$ only depending on $k$ and the fixed model bounds. In the cited commutator estimate, occupation shifts of size at most four give coefficients bounded by $Ck5^{2k}$. The coefficient before scalar absorption is therefore at most $Ce^{Ck}$, so $(c_k/\mu)^{1/(k+3)}\le Ce^{Ck}$ and $C_k^{\mathrm{mom}}\le Ce^{Ck}$, with $C$ independent of $k$.

		We next verify the polynomial truncation estimate. For $M\ge8$, put
		\begin{equation*}
			S_e=\sum_{z\in e}(N_z+\1),\qquad
			Q_e^{(M)}=\1-P_e^{(M)},\qquad
			\widetilde Q_e^{(M)}=\1-P_e^{(M-4)}.
		\end{equation*}
		By \cite[Lem.~E.2]{Gondolf.2024}, $BS_e^{-2}$ is bounded for every monomial $B$ of degree at most four. Moreover,
		\begin{equation*}
			B-P_e^{(M)}BP_e^{(M)}
			=BQ_e^{(M)}+Q_e^{(M)}BP_e^{(M)},
			\qquad
			Q_e^{(M)}BP_e^{(M)}
			=Q_e^{(M)}BP_e^{(M)}\widetilde Q_e^{(M)}.
		\end{equation*}
		The second identity follows because $B$ changes each occupation by at most four. Since the projections commute with $S_e$ and $Q_e^{(M)}\le\widetilde Q_e^{(M)}$, for every $\psi\in\mathcal{H}_f$,
		\begin{equation*}
			\|(B-P_e^{(M)}BP_e^{(M)})\psi\|
			\le\|BQ_e^{(M)}\psi\|
				+\|Q_e^{(M)}BP_e^{(M)}\widetilde Q_e^{(M)}\psi\|
			\le C\|S_e^2\widetilde Q_e^{(M)}\psi\|.
		\end{equation*}
		Squaring gives the quadratic-form inequality
		\begin{equation*}
			(B-P_e^{(M)}BP_e^{(M)})^\dagger
			(B-P_e^{(M)}BP_e^{(M)})\le C\widetilde Q_e^{(M)}S_e^4\widetilde Q_e^{(M)}.
		\end{equation*}
		For $M\ge8$, the range of $\widetilde Q_e^{(M)}$ satisfies $S_e\ge M/2$. Hence, for $p>4$,
		\begin{equation*}
			\widetilde Q_e^{(M)}S_e^4\widetilde Q_e^{(M)}
			\le C_pM^{-(p-4)}\sum_{z\in e}(N_z+\1)^p.
		\end{equation*}
		Then, Cauchy--Schwarz shows, for every $\sigma\geq0$ with $\operatorname{tr} 
  \ifx\empty \else \left[  \right] \fi\sigma\le1$ and finite local $p$-moments,
		\begin{equation}\label{eq:cat-polynomial-tail}
			\|(H_e-H_e^{(M)})\sigma\|_1+\|\sigma(H_e-H_e^{(M)})\|_1\le C_pJ_{\mathrm{cat}}M^{-(p-4)/2}\sum_{z\in e}\operatorname{tr} 
  \ifx(N_z+\1)^p\sigma\empty \else \left[ (N_z+\1)^p\sigma \right] \fi^{1/2}.
		\end{equation}
		Enlarging $C_p$ covers $2\le M<8$. The scalar weight estimates above give $C_p\le Ce^{Cp}$. The same weighted bound gives $\|H_e^{(M)}\|_\infty\le CJ_{\mathrm{cat}}M^2$ and therefore $\|[-iH_e^{(M)},\cdot]\|_{\infty\to\infty}\le CJ_{\mathrm{cat}}M^2$.

Let $T_s^\gamma=e^{s\mathcal{L}^\gamma}$, and let $\mathcal{L}_{R,M}^\gamma$ be obtained from $\mathcal{L}_R^\gamma$ by replacing every $H_e$ with $H_e^{(M)}$. 		\textbf{First, for all edges in $E_R$,} decompose $H_e$ into the truncation tail $H_e-H_e^{(M(s))}$ and the bounded part $H_e^{(M(s))}$. For boundary edges, the bounded part is absent from the cutoff buffer generator and is estimated below. For an integer $\widetilde M\ge2$, choose

		\textbf{Second, for all edges in $\partial_E R$,} the bounded terms $H_e^{(M(s))}$ introduced above are absent from the cutoff buffer generator. 		\end{equation*}

			C|\partial_E R|\,\|A\|_\infty e^{-d(X,\partial_E R)}
			\left(e^{C_pJ_{\mathrm{cat}}\int_0^tM(s)^2\,ds}-1\right).

				\le(\lambda+1)^2\int_0^t b_p(s)^2\,ds
				\le6(C_p^{\mathrm{mom}})^2(\lambda+1)^2(t^{1/3}+t),

		\begin{equation}\label{eq:graph-bounded-interaction-lrb}
			\|K_Y\tau_{s,t}(A)\|_\infty
			\le
			\frac{1}{2}\|K_Y\|_{\infty\to\infty}\|A\|_\infty\min\{|X|,|Y|\}\,
			e^{-d(X,Y)}
			\left(e^{2\mathrm e\int_s^t\|\Phi(u)\|_1\,du}-1\right).
		\end{equation}
		For time-independent edge terms, the exponent is $2\mathrm e\|\Phi\|_1(t-s)$, with $\|\Phi\|_1\le\Delta_G\sup_e\|\Phi_e\|_{\infty\to\infty}$. The bound is independent of the on-site generators and the local Hilbert-space dimensions.

We first keep the on-site generators bounded and denote the resulting propagator by $\tau_{s,t}^{(L)}$. The adaptive-Dyson expansion in \cite[Thm.~3.4]{Rubiliani.2026} contains only paths of edge terms, since one-site terms cannot occur in a self-avoiding path. Retaining the instantaneous interaction norm in the path count of \cite[Lem.~4.3]{Rubiliani.2026} gives
		\begin{equation*}
			\|K_Y\tau_{s,t}^{(L)}(A)\|_\infty
			\le
			\frac{1}{2}\|K_Y\|_{\infty\to\infty}\|A\|_\infty\min\{|X|,|Y|\}\,
			\sum_{n\ge d(X,Y)}\frac{\left(2\int_s^t\|\Phi(u)\|_1\,du\right)^n}{n!}.
		\end{equation*}
		Indeed, paths with $n$ edges require $n\ge d(X,Y)$, and the ordered time integral of $\prod_{j=1}^n\|\Phi(u_j)\|_1$ equals $\bigl(\int_s^t\|\Phi(u)\|_1\,du\bigr)^n/n!$. Multiplying each summand by $e^{n-d(X,Y)}\ge1$ and summing proves Equation~\eqref{eq:graph-bounded-interaction-lrb} for $\tau_{s,t}^{(L)}$. If $d(X,Y)=+\infty$, no connecting path exists and the left side vanishes. Convex interpolation at the finitely many jumps preserves the contraction-generator property. Convergence in the time-integrated norm then extends the bound to the stated schedules by Duhamel's formula. We next remove the on-site cutoff. The bounded-perturbation expansion on each time interval and convergence of finite products give strong convergence of the perturbed predual propagators. Hence $\tau_{s,t}^{(L)}(A)\to\tau_{s,t}(A)$ weak*. Normality of $K_Y$ and weak* lower semicontinuity of the operator norm give Equation~\eqref{eq:graph-bounded-interaction-lrb}.

<!-- lem-5: body_md -->
\label{lem:cat-finite-adiabatic}
		Consider the cat model in Section~\ref{subsec:local-adiabatic-limit}. Then the dissipative evolution is approximated by the projected Hamiltonian evolution with error
		\begin{equation}\label{eq:cat-finite-adiabatic}
			\|e^{t\mathcal{L}_R^\gamma}(\sigma)
				-e^{t\mathcal K_R^{\mathrm{proj}}}(\sigma)\|_1
				\le \frac{C(t|R|^2+|R|)}{\gamma}
		\end{equation}
		for every finite region $R$, $t\ge0$, $\gamma>0$, and state $\sigma=P_R\sigma P_R$. The constant $C<\infty$ depends only on $\sup_x|\alpha_x|$, $J_{\mathrm{cat}}$, the uniform bound on the coefficients of $H_x$, and $\Delta_G$. It is independent of $R$, $t$, and $\gamma$. For every fixed $k\ge1$, $\|(N_x+\1)^kP_x\|_\infty$ is uniformly bounded in $x$.
<!-- end lem-5 -->
<!-- proof-18: body_md -->
We first bound moments on the code space. A one-site code state is stationary under $\mathcal{L}[L_x]$. Apply Equation~\eqref{eq:cat-local-moments} at order $2k$ to the one-site model with zero Hamiltonian and unit loss rate, and let time tend to infinity. Taking the supremum over code states bounds $\|(N_x+\1)^kP_x\|_\infty$ uniformly for bounded $|\alpha_x|$.

		Next, we verify a gap for $L'_R=\sum_{x\in R}L_x^\dagger L_x$ and bound the Hamiltonian relative to it. The gap estimates \cite[Lem.~E.5, E.11]{Moebus.2025} give $L'_R\ge24(\1-P_R)$. The one-site polynomial estimates \cite[Lem.~C.1 and Cor.~C.4]{Moebus.2025} imply, uniformly in $x$,
		\begin{equation*}
			\|(N_x+\1)^2\psi\|
				\le C\bigl(\|L_x^\dagger L_x\psi\|+\|\psi\|\bigr).
		\end{equation*}
		Every Hamiltonian monomial has degree at most four and support on at most two sites. Its norm is bounded by a constant times the sum of these number weights on its support. Since the positive $L_x^\dagger L_x$ commute, $\|L_x^\dagger L_x\psi\|\le\|L'_R\psi\|$ on $\operatorname{dom}(L'_R)$. Summing $\mathcal O(|R|)$ local terms gives
		\begin{equation*}
			\|H_R^{\mathrm{cat}}\psi\|
			\le C|R|\bigl(\|L'_R\psi\|+\|\psi\|\bigr).
		\end{equation*}
		We also need bounds for Hamiltonian terms acting on code states. The code moment bounds give $\|H_R^{\mathrm{cat}}P_R\|_\infty\le C|R|$. In $L'_RH_R^{\mathrm{cat}}P_R$, the loss factor at $x$ annihilates a Hamiltonian term supported away from $x$. The remaining $\mathcal O(|R|)$ terms have uniformly bounded norm, so
		\begin{equation*}
			\|L'_RH_R^{\mathrm{cat}}P_R\|_\infty\le C|R|.
		\end{equation*}
		Finally, we apply the adiabatic estimate. The all-order Sobolev estimates and \cite[Lem.~D.2]{Rico.2025} justify the two Duhamel identities in the proof of \cite[Thm.~A.1, case $\mathcal K=0$]{Moebus.2025} for this generated dynamics. Its gap-dependent estimate, with the three bounds above, gives Equation~\eqref{eq:cat-finite-adiabatic}.
