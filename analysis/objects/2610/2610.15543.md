<!-- definition-1: body_md -->
\label{def:refresh-regular}
For known positive rates $(\lambda_m)_{m=1}^M$, with
$\lambda_{\min}=\min_m\lambda_m$, a realized schedule is
$\lambda$-regular if, for every $m$ and all $t-s\ge2/\lambda_m$,
\[
 \tfrac12\lambda_m(t-s)\le N_m(t)-N_m(s)\le2\lambda_m(t-s).
\]

<!-- theorem-1: body_md -->
\label{thm:refresh-upper}
Under the assumptions of Section~\ref{sec:refresh-model}, condition all
expectations on a fixed $\lambda$-regular schedule and let $c=K\tau$ and
$\beta_0=4\sqrt{2/3}$. For Algorithm~\ref{alg:refresh}, every
$t\in\mathcal{I}_j$ satisfies
\begin{align}
\label{eq:refresh-pointwise}
\mathbb{E}[F^t(x_j)-\min_\mathcal{X} F^t]
&\le \frac{4LD^2}{K(K+1)}
 +\beta_0D\sqrt{\sum_m\frac{b_m}{h_{m,j}}}
 +2D\sum_m a_mh_{m,j}
 +D\bar\gamma(c+\Delta),
\end{align}
and for every $T\ge s_0+c$ the reported trajectory satisfies
\begin{align}
\label{eq:refresh-average}
\mathrm{AvgErr}_T
&\le \frac{4LD^2}{K(K+1)}
 +D\Bigl(2\sum_m a_mh_m+
         \beta_0\sqrt{\sum_m b_m/h_m}\Bigr)
 +D\bar\gamma(c+\Delta)
 +\mathcal{T}_T,\\
\label{eq:refresh-transient}
\mathcal{T}_T
&:=\beta_0D\,\Xi
   \Bigl(\frac{2}{\sqrt T}+\frac{\Delta}{T\sqrt{s_0}}\Bigr)
   +\frac{GD(s_0+c)}{T}.
\end{align}

<!-- corollary-1: body_md -->
\label{cor:refresh-opt}
Suppose $\mathcal{S}_{\gamma}>0$ and $\sigma_m>0$ for every client with $\gamma_m>0$. For
such clients set
\[

 \qquad h_m=\max\{h_m^\star,\,2K/\lambda_m\},
\]
and set $h_m=\infty$ when $\gamma_m=0$. Then
\begin{align}
\label{eq:refresh-optimized}
\mathrm{AvgErr}_T
\le {}&6(2/3)^{1/3}D\mathcal{H}_{\gamma}
 +\frac{4LD^2}{K(K+1)}
 +4KD\sum_m\frac{a_m}{\lambda_m}
 +D\bar\gamma(K\tau+\Delta)+\mathcal{T}_T .
\end{align}
Before the sample-count floors are imposed, $(h_m^\star)$ exactly
minimizes the statistical-drift expression in parentheses
in~(\ref{eq:refresh-average}).

\label{cor:refresh-dyadic}

<!-- proposition-1: body_md -->
\label{prop:periodic-main}
Let $\ell(x;Y)=x^2/2-Yx$ on $[-2,2]$, with two equally weighted clients
and $\tau=0$. For an integer $r\ge2$, client one has mean
$\cos(t+\Phi)$, rate $r$, and Rademacher noise of scale $1/r$; client
two has mean zero, rate one, and independent Rademacher noise of scale
$\sqrt2$. The phase $\Phi$ is arbitrary and fixed. Every finite shared
window $h\ge1$ has limiting inferior risk at least $1/16$, and the full
prefix attains $1/16$. The separate choice $h_1=1/r$, $h_2=\infty$ has
long-run risk
\begin{equation}
\label{eq:periodic-main}
 \frac18\Bigl[1-\frac{\sin(1/r)}{1/r}+\frac1{r^2}\Bigr]
 \le\frac7{48r^2}.
\end{equation}
Hence the ratio between the best shared-window risk and the risk of this
separate policy is at least $3r^2/7$, on a fixed compact domain.

<!-- head-1: body_md -->
Alternate arrivals at each client are assigned to a training and a
validation stream. For each of $E$ fixed memory vectors $h^1,\ldots,h^E$,
a separate reset SGD \emph{state} runs on the shared training array with
disjoint batches within each solve; these experts never see validation
samples or mixture weights. The server reports a convex mixture of the
expert outputs. Fresh validation samples from each reporting interval
score all experts, and two interleaved exponential-weight forecasters
(for even and for odd intervals) ensure that a full interval of feedback
has arrived before the forecaster that used it reports again. The
aggregation principle is standard \citep{freund1997hedge}; the
independent split and a source-time correction make it valid here.
Appendix~\ref{app:adaptive-memory} specifies the communication pipeline
and proves the complete form of the following statement
(Theorem~\ref{thm:adaptive-memory}).

       +D\bar\gamma\Delta .
\end{equation}
Neither the updates nor the learning rate uses a drift bound.

<!-- lemma-1: body_md -->
\label{lem:refresh-accel}
Let $F$ be convex and $L$-smooth, $L>0$, on a compact convex set $\mathcal{X}$
of diameter $D>0$. Start from $x^0=v^0=x_{\circ}\in\mathcal{X}$ and, for
$k=1,\ldots,K$, let
\begin{align*}
\theta_k&=\frac{2}{k+1},&
y^k&=(1-\theta_k)x^{k-1}+\theta_kv^{k-1},\\
v^k&=\Pi_{\mathcal{X}}(v^{k-1}-k\eta\,g^k),&
x^k&=(1-\theta_k)x^{k-1}+\theta_kv^k .
\end{align*}
Let $(\mathcal{F}_k)_{k\ge0}$ be a filtration such that $y^k,v^{k-1},x^{k-1}$
are $\mathcal{F}_{k-1}$-measurable, and suppose
\[
g^k=\nabla F(y^k)+\delta_k+\varepsilon_k,
\qquad \left\lVert \delta_k\right\rVert\le\delta,
\]
where $\delta_k$ is $\mathcal{F}_{k-1}$-measurable and
\[
\mathbb{E}[\varepsilon_k\mid\mathcal{F}_{k-1}]=0,
\qquad
\mathbb{E}[\left\lVert \varepsilon_k\right\rVert^2\mid\mathcal{F}_{k-1}]\le V_k
\]
for deterministic $V_k\ge0$. Then, for every $0<\eta\le1/(4L)$,
\begin{equation}
\label{eq:refresh-accel-general}
\mathbb{E}[F(x^K)-\min_\mathcal{X} F]
\le\frac{2}{K(K+1)}
\Bigl(\frac{D^2}{2\eta}+\eta\sum_{k=1}^Kk^2V_k\Bigr)+\delta D .
\end{equation}
If $S=\sum_{k=1}^Kk^2V_k>0$, the choice $\eta=\min\{1/(4L),D/\sqrt{2S}\}$
gives
\begin{equation}
\label{eq:refresh-accel-tuned}
\mathbb{E}[F(x^K)-\min_\mathcal{X} F]
\le\frac{4LD^2+2\sqrt2D\sqrt S}{K(K+1)}+\delta D ,
\end{equation}
and for $S=0$ the same bound holds with $\eta=1/(4L)$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Fix $x^{\star}\in\operatorname*{arg\,min}_\mathcal{X} F$ and write $r_k=\left\lVert v^k-x^{\star}\right\rVert^2$.
Since $x^k-y^k=\theta_k(v^k-v^{k-1})$ and
$y^k=(1-\theta_k)x^{k-1}+\theta_kv^{k-1}$, smoothness together with the
two supporting-hyperplane inequalities at $y^k$ gives
\begin{align}
F(x^k)-F(x^{\star})
&\le(1-\theta_k)[F(x^{k-1})-F(x^{\star})]
+\theta_k\langle \nabla F(y^k), v^k-x^{\star}\rangle
+\frac{L\theta_k^2}{2}\left\lVert v^k-v^{k-1}\right\rVert^2 .
\label{eq:refresh-accel-smooth}
\end{align}
The variational inequality of the projection that defines $v^k$ gives
\begin{equation}
\label{eq:refresh-accel-projection}
\langle g^k, v^k-x^{\star}\rangle
\le\frac{r_{k-1}-r_k-\left\lVert v^k-v^{k-1}\right\rVert^2}{2k\eta}.
\end{equation}
Substitute $\nabla F(y^k)=g^k-\delta_k-\varepsilon_k$ into
(\ref{eq:refresh-accel-smooth}). Since $v^k,x^{\star}\in\mathcal{X}$, the bias
contributes $-\theta_k\langle \delta_k, v^k-x^{\star}\rangle\le\theta_k\delta D$.
For the noise, write $v^k-x^{\star}=(v^{k-1}-x^{\star})+(v^k-v^{k-1})$. The
stepsize restriction gives $L\theta_k\cdot k\eta\le2L\eta\le1/2$, so
after (\ref{eq:refresh-accel-projection}) the coefficient of
$-\left\lVert v^k-v^{k-1}\right\rVert^2$ is at least $\theta_k/(4k\eta)$, and Young's
inequality yields
\[
-\theta_k\langle \varepsilon_k, v^k-v^{k-1}\rangle
\le\theta_k k\eta\left\lVert \varepsilon_k\right\rVert^2
+\frac{\theta_k}{4k\eta}\left\lVert v^k-v^{k-1}\right\rVert^2 .
\]
The squared-step terms cancel, and
$\mathbb{E}\langle \varepsilon_k, v^{k-1}-x^{\star}\rangle=0$ by conditional centering.
Hence, with $\mathcal E_k=\mathbb{E}[F(x^k)-F(x^{\star})]$,
\begin{equation}
\label{eq:refresh-accel-recursion}
\mathcal E_k\le(1-\theta_k)\mathcal E_{k-1}
+\frac{\theta_k}{2k\eta}\mathbb{E}[r_{k-1}-r_k]
+\theta_kk\eta V_k+\theta_k\delta D .
\end{equation}
Let $\Gamma_k=2/[k(k+1)]$. For $k\ge2$ we have
$(1-\theta_k)/\Gamma_k=1/\Gamma_{k-1}$, and for $k=1$ the coefficient of
$\mathcal E_0$ vanishes. Moreover,
\[
\frac{\theta_k}{2k\eta\,\Gamma_k}=\frac1{2\eta},
\qquad
\frac{\theta_kk\eta}{\Gamma_k}=k^2\eta,
\qquad
\frac{\theta_k}{\Gamma_k}=k .
\]
Dividing (\ref{eq:refresh-accel-recursion}) by $\Gamma_k$, summing over
$k$, and using $r_0\le D^2$, $r_K\ge0$, and
$\Gamma_K\sum_{k=1}^Kk=1$ proves (\ref{eq:refresh-accel-general}).

If the tuned stepsize equals $D/\sqrt{2S}$, the two terms in parentheses
in (\ref{eq:refresh-accel-general}) sum to $\sqrt2D\sqrt S$. If the cap
$1/(4L)$ is active, then $S\le8L^2D^2$ and
\[
\frac{D^2}{2\eta}+\eta S
=2LD^2+\frac S{4L}
\le2LD^2+\frac{D\sqrt S}{\sqrt2}.
\]
Both cases give (\ref{eq:refresh-accel-tuned}); the case $S=0$ follows
directly from (\ref{eq:refresh-accel-general}).

Condition on the complete exogenous arrival schedule. All clocks, buffer
choices, batch partitions, and variance-based stepsizes are then fixed
without revealing any sample value. Recall $s_j=s_0+j\Delta$, $c=K\tau$,
and $h_{m,j}=\min\{s_j,h_m\}$ with $\min\{s_j,\infty\}=s_j$.

\emph{Step 1: buffer ages and sample counts.}
If $s_j<h_m$, the frozen buffer of client $m$ is the prefix $(0,s_j]$, of
length $h_{m,j}=s_j$. If $h_m\le s_j$ and $h_m<\infty$, let
$p=\lfloor s_j/h_m\rfloor$; the frozen buffer is $((p-1)h_m,ph_m]$, of
length $h_{m,j}=h_m$, and since $ph_m\le s_j<(p+1)h_m$ every sample in
it has age less than $2h_m$ at the solve start. Infinite windows always
use the prefix. Hence every selected timestamp $t_{m,i}$ satisfies
\begin{equation}
\label{eq:refresh-age}
s_j-2h_{m,j}<t_{m,i}\le s_j .
\end{equation}
By $s_0=2K/\lambda_{\min}$ and $h_m\ge2K/\lambda_m$ we have
$h_{m,j}\ge2K/\lambda_m$, so the lower regularity inequality applies to
the frozen interval and gives
\begin{equation}
\label{eq:refresh-buffer-count}
n_{m,j}\ge\frac{\lambda_mh_{m,j}}2\ge K .
\end{equation}
Splitting the $n_{m,j}$ observations into $K$ batches whose sizes differ
by at most one yields batch sizes
\begin{equation}
\label{eq:refresh-split-count}
n_{m,j,k}\ge\Bigl\lfloor\frac{n_{m,j}}K\Bigr\rfloor
\ge\frac{n_{m,j}}{2K},
\end{equation}
where the last inequality uses $n_{m,j}/K\ge1$.

\emph{Step 2: the filtration of one solve.}
Fix a solve $j$. Its iterates reset to the public point $x_{\circ}$, and
its updates use only the batches of this solve. Let $\mathcal{F}_{j,k}$ be
generated by the fixed schedule and the values of batches $1,\ldots,k$ of
this solve at all clients. Then $y^k$ and the earlier iterates are
$\mathcal{F}_{j,k-1}$-measurable, and conditional independence of the
time-stamped samples makes batch $k$ independent of the earlier batches
of the solve.

The same samples may have appeared in earlier solves. We do not condition
on those uses or on earlier reported outputs: the current solve is a
measurable function of its own frozen buffers, its partition, and the
fixed initialization, so this marginal filtration bounds its expected
risk even when buffers overlap across solves. Earlier outputs change
neither the marginal laws of the buffer samples nor the fact that each
batch is used once within this solve. Warm starts or sample-dependent
windows would require a different argument.

\emph{Step 3: the conditional oracle at a fixed report time.}
Fix $t\in\mathcal{I}_j=[s_j+c,s_{j+1}+c)$. The conditional mean of the
aggregate gradient at step $k$ is
\[
\mathbb{E}[g^k\mid\mathcal{F}_{j,k-1}]
=\sum_m\frac{w_m}{n_{m,j,k}}
\sum_{i\in\mathcal{B}_{m,j,k}}\nabla f_m^{t_{m,i}}(y^k) .
\]
By the clientwise drift bounds and (\ref{eq:refresh-age}),
\begin{align*}
\bigl\|\mathbb{E}[g^k\mid\mathcal{F}_{j,k-1}]-\nabla F^t(y^k)\bigr\|
&\le\sum_m\frac{w_m\gamma_m}{n_{m,j,k}}
\sum_{i\in\mathcal{B}_{m,j,k}}(t-t_{m,i})\\
&\le 2\sum_m a_mh_{m,j}+\bar\gamma(t-s_j)
=:\delta_j(t).
\end{align*}
Conditionally on $\mathcal{F}_{j,k-1}$, the centered noises of different
samples and clients are independent, so their second moments add:
\[
\mathbb{E}\Bigl[\bigl\|g^k-\mathbb{E}[g^k\mid\mathcal{F}_{j,k-1}]\bigr\|^2
\Bigm|\mathcal{F}_{j,k-1}\Bigr]
\le V_{j,k}:=\sum_m\frac{w_m^2\sigma_m^2}{n_{m,j,k}} .
\]
Apply Lemma~\ref{lem:refresh-accel} with $F=F^t$ and $\delta=\delta_j(t)$.
With $S_j=\sum_{k=1}^Kk^2V_{j,k}$ and the solve output $x_j$, this gives
the exact-count bound
\begin{equation}
\label{eq:refresh-exact-proof}
\mathbb{E}[F^t(x_j)-\min_\mathcal{X} F^t]
\le\frac{4LD^2+2\sqrt2D\sqrt{S_j}}{K(K+1)}
+D\Bigl(2\sum_m a_mh_{m,j}+\bar\gamma(t-s_j)\Bigr).
\end{equation}
The argument applies separately to every deterministic report time; it
yields the pointwise expectation needed below without any uniform
continuous-time high-probability statement.

\emph{Step 4: the heterogeneous count bound.}
Put $\bar V_j=\sum_mw_m^2\sigma_m^2/n_{m,j}$. From
(\ref{eq:refresh-split-count}), $V_{j,k}\le2K\bar V_j$, whence
\[
S_j\le2K\bar V_j\sum_{k=1}^Kk^2
\le\frac23\bar V_jK^2(K+1)^2 .
\]
Thus the statistical term in (\ref{eq:refresh-exact-proof}) is at most
$4D\sqrt{\bar V_j}/\sqrt3$, and by (\ref{eq:refresh-buffer-count})
\[
\bar V_j\le2\sum_m\frac{b_m}{h_{m,j}} .
\]
Using $t-s_j<c+\Delta$ we arrive at
\begin{align}
\mathbb{E}[F^t(x_j)-\min_\mathcal{X} F^t]
&\le\frac{4LD^2}{K(K+1)}
+\beta_0D\sqrt{\sum_m\frac{b_m}{h_{m,j}}}
+2D\sum_m a_mh_{m,j}
+D\bar\gamma(c+\Delta),
\label{eq:refresh-r-proof}
\end{align}
with $\beta_0=4\sqrt{2/3}$; this is (\ref{eq:refresh-pointwise}).

\emph{Step 5: time averaging, including prefixes and stationary clients.}
For $h_m=\infty$ we use the conventions $1/h_m=0$ and $a_mh_m=0$, the
latter only when $\gamma_m=0$. Then
\[
\frac1{h_{m,j}}=\frac1{\min\{s_j,h_m\}}
\le\frac1{s_j}+\frac1{h_m},
\qquad
\sum_m a_mh_{m,j}\le\sum_m a_mh_m,
\]
and therefore, with $\Xi^2=\sum_mb_m$,
\[
\sqrt{\sum_m\frac{b_m}{h_{m,j}}}
\le\sqrt{\sum_m\frac{b_m}{h_m}}+\frac{\Xi}{\sqrt{s_j}} .
\]
In particular, a moving client's prefix count is never replaced by
$\lambda_mh_m/2$ before its first complete window exists.

When $T\ge s_0+c$, the report intervals that meet $[0,T]$ have
consecutive indices $0,1,\ldots,j_T$. Since $s\mapsto s^{-1/2}$ is
decreasing,
\begin{align*}
\sum_{j=0}^{j_T}\frac{|\mathcal{I}_j\cap[0,T]|}{\sqrt{s_j}}
&\le\frac{\Delta}{\sqrt{s_0}}
+\sum_{j=1}^{j_T}\frac{\Delta}{\sqrt{s_0+j\Delta}}
\le\frac{\Delta}{\sqrt{s_0}}
+\int_{s_0}^{s_{j_T}}s^{-1/2}\,ds
\le\frac{\Delta}{\sqrt{s_0}}+2\sqrt T ,
\end{align*}
with the usual convention that an empty sum is zero. Before the first
report, the uniform gradient bound gives
$F^t(x_{\circ})-\min_\mathcal{X} F^t\le GD$. Integrating (\ref{eq:refresh-r-proof})
over $[0,T]$, applying the preceding inequalities, and dividing by $T$
proves
\begin{align*}
\mathrm{AvgErr}_T
\le{}&\frac{4LD^2}{K(K+1)}
+\beta_0D\sqrt{\sum_m\frac{b_m}{h_m}}
+2D\sum_m a_mh_m+D\bar\gamma(c+\Delta)\\
&+\beta_0D\,\Xi\Bigl(\frac2{\sqrt T}
+\frac{\Delta}{T\sqrt{s_0}}\Bigr)
+\frac{GD(s_0+c)}T ,
\end{align*}
which is (\ref{eq:refresh-average})--(\ref{eq:refresh-transient}).
All expectations were conditional on the fixed schedule; the same
inequalities hold unconditionally if the regularity assumption holds
almost surely. The clocks and snapshots use arrived data only, and the
$K$ rounds cost $c=K\tau\le\Delta$, so the solves and the reporting
trajectory are causal.

<!-- lemma-3: body_md -->
\label{lem:refresh-sgd-bias}
Let $F$ be convex and $L$-smooth, $L>0$, on a compact convex set $\mathcal{X}$
of diameter $D>0$. Fix $z^0=x_{\circ}\in\mathcal{X}$, an integer $K\ge1$, and a
deterministic $0<\eta\le1/(2L)$. For $k=1,\ldots,K$ let
\[
 z^k=\Pi_{\mathcal{X}}(z^{k-1}-\eta g^k),\qquad
 g^k=\nabla F(z^{k-1})+\delta_k+\varepsilon_k,
\]
where $z^{k-1}$ and $\delta_k$ are $\mathcal{F}_{k-1}$-measurable,
$\|\delta_k\|\le\delta$, and
\[
 \mathbb{E}[\varepsilon_k\mid\mathcal{F}_{k-1}]=0,
 \qquad
 \mathbb{E}[\|\varepsilon_k\|^2\mid\mathcal{F}_{k-1}]\le V_k
\]
for finite deterministic $V_k\ge0$. Then, for
$\bar z=K^{-1}\sum_{k=1}^Kz^k$,
\begin{equation}
\label{eq:refresh-sgd-general}
 \mathbb{E}[F(\bar z)-\min_\mathcal{X} F]
 \le\frac{D^2}{2\eta K}+\frac\eta K\sum_{k=1}^K V_k+\delta D .
\end{equation}
Write $U=\sum_{k=1}^K V_k$. With $\eta=\min\{1/(2L),D/\sqrt{2U}\}$ if
$U>0$, and $\eta=1/(2L)$ if $U=0$,
\begin{equation}
\label{eq:refresh-sgd-tuned}
 \mathbb{E}[F(\bar z)-\min_\mathcal{X} F]
 \le\frac{LD^2}{K}+\frac{\sqrt2D\sqrt U}{K}+\delta D .
\end{equation}
<!-- end lemma-3 -->
<!-- proof-5: body_md -->
Fix $x^{\star}\in\operatorname*{arg\,min}_\mathcal{X} F$, write $r_k=\|z^k-x^{\star}\|^2$, and combine smoothness at $z^{k-1}$ with the
supporting-hyperplane inequality for $F(x^{\star})$:
\[
 F(z^k)-F(x^{\star})
 \le\langle \nabla F(z^{k-1}), z^k-x^{\star}\rangle
       +\frac L2\|z^k-z^{k-1}\|^2 .
\]
The projection variational inequality gives
\[
 \langle g^k, z^k-x^{\star}\rangle
 \le\frac{r_{k-1}-r_k-\|z^k-z^{k-1}\|^2}{2\eta}.
\]
After substituting $\nabla F(z^{k-1})=g^k-\delta_k-\varepsilon_k$, the
bias term is bounded by $-\langle \delta_k, z^k-x^{\star}\rangle\le\delta D$,
because both $z^k$ and $x^{\star}$ lie in $\mathcal{X}$. Split the noise term as
\[
 -\langle \varepsilon_k, z^k-x^{\star}\rangle
 =-\langle \varepsilon_k, z^{k-1}-x^{\star}\rangle
  -\langle \varepsilon_k, z^k-z^{k-1}\rangle.
\]
Since $\eta\le1/(2L)$, we have $1/(2\eta)-L/2\ge1/(4\eta)$, and Young's
inequality gives
\[
 -\langle \varepsilon_k, z^k-z^{k-1}\rangle
 \le\eta\|\varepsilon_k\|^2+\frac1{4\eta}\|z^k-z^{k-1}\|^2 ,
\]
so the squared-step terms have a nonpositive sum. Conditional centering
removes $\mathbb{E}\langle \varepsilon_k, z^{k-1}-x^{\star}\rangle$. Therefore
\begin{equation}
\label{eq:refresh-sgd-recursion}
 \mathbb{E}[F(z^k)-F(x^{\star})]
 \le\frac{\mathbb{E} r_{k-1}-\mathbb{E} r_k}{2\eta}
       +\eta V_k+\delta D .
\end{equation}
Summing over $k$, using $r_0\le D^2$ and $r_K\ge0$, and applying
convexity to the average of the post-update points proves
(\ref{eq:refresh-sgd-general}); the accumulated bias term remains
$\delta D$ after division by $K$.

If the chosen stepsize is $D/\sqrt{2U}$, then
$D^2/(2\eta)+\eta U=\sqrt2D\sqrt U$. If the cap $1/(2L)$ is active, then
$U\le2L^2D^2$ and
\[
 \frac{D^2}{2\eta}+\eta U
 =LD^2+\frac U{2L}
 \le LD^2+\frac{D\sqrt U}{\sqrt2}.
\]
Both cases imply (\ref{eq:refresh-sgd-tuned}). When $U=0$, the cap in
(\ref{eq:refresh-sgd-general}) gives $LD^2/K+\delta D$ directly.

then deterministic. In solve $j$, reveal only its batches $1,\ldots,k$ at

analysis of one reset solve; earlier uses of the same observations in
other solves are not conditioned upon.

Fix a deterministic $t\in\mathcal{I}_j=[s_j+c,s_{j+1}+c)$, where $c=K\tau$.
The pointwise oracle assumptions, independence within the current batch,
and the age bound (\ref{eq:refresh-age}) give
\begin{align*}
 \bigl\|\mathbb{E}[g_j^k\mid\mathcal{F}_{j,k-1}]-\nabla F^t(z_j^{k-1})\bigr\|

\label{prop:refresh-stationary}
Fix $D>0$, $T>0$, positive $(w_m,\lambda_m,\sigma_m)_m$, nonnegative
$(\gamma_m)_m$, and a $\lambda$-regular schedule on $[0,\infty)$. For
the fixed-vector class and the pooled causal rules of

\[
 \mathcal{R}_T^\star(\gamma)
 \ge\frac D4\min\Bigl\{G,\
       \frac1{\sqrt8}
       \Bigl(\sum_m\frac{w_m^2\sigma_m^2}{\lambda_mT+2}\Bigr)^{1/2}
       \Bigr\}.
\]
For every fixed rule, one of two stationary linear instances attains the
bound.

<!-- proof-7: body_md -->
Choose amplitudes $A_m\ge0$ and one uniform sign $\varsigma\in\{-1,+1\}$. At
every time, client $m$ supplies an independent sample from
$\mathcal N(\varsigma A_m,\sigma_m^2)$ with loss $\ell(x;\xi)=\xi x_1$. Let
$C=\sum_mw_mA_m$. The objectives have zero drift and population gradient
bound $C$, so both signs give admissible instances whenever $C\le G$.

For every $T>0$, regularity implies $N_m(T)\le2\lambda_mT+4$: when
$T\ge2/\lambda_m$ the upper inequality applies directly, and otherwise
$N_m(T)\le N_m(2/\lambda_m)\le4$. The two full transcripts through time
$T$ therefore have KL divergence at most
\[
 \sum_m\frac{2A_m^2N_m(T)}{\sigma_m^2}
 \le4\sum_m\frac{(\lambda_mT+2)A_m^2}{\sigma_m^2}.
\]
If the last sum, without the factor $4$, is at most $1/8$, the full
transcripts have total variation at most $1/2$, and so do the partial
transcripts at every earlier time. At every time, the two objective gaps
of any common candidate action add up to $DC$, so the two-point Bayes
argument gives expected gap at least $DC/4$ at every time and hence for
the average over $[0,T]$.

By the Cauchy--Schwarz inequality,
\[
 \sup\Bigl\{\sum_mw_mA_m:\ \sum_m\frac{(\lambda_mT+2)A_m^2}{\sigma_m^2}\le\frac18\Bigr\}
 =\frac1{\sqrt8}\Bigl(\sum_m\frac{w_m^2\sigma_m^2}{\lambda_mT+2}\Bigr)^{1/2},
\]
with equality for nonnegative $A_m$ proportional to
$w_m\sigma_m^2/(\lambda_mT+2)$. If
this value exceeds $G$, scale the amplitudes down until $C=G$; the
information constraint remains valid. Maximizing the Bayes bound over
these amplitudes and selecting one of the two signs proves the claim.
When $G=0$ the bound is zero and there is nothing to prove.

Under (\ref{eq:periodic-stream})--(\ref{eq:periodic-interior}), the

\[
 \mathcal K_n(\varphi)=\frac1n\sum_{l=0}^{n-1}e^{-\mathrm{i} l\varphi},\qquad
 \operatorname{sinc}(z)=\begin{cases}\sin(z)/z,&z\ne0,\\1,&z=0,\end{cases}

and, for a finite window of $n$ samples, the bias factor
\begin{equation}
\label{eq:periodic-factor}
 \Upsilon_n(\varphi)=1+|\mathcal K_n(\varphi)|^2
       -2\operatorname{Re}\Bigl\{
          \mathcal K_n(\varphi)\,e^{-\mathrm{i} n\varphi/2}
                       \operatorname{sinc}(n\varphi/2)\Bigr\}.
\end{equation}
If $\sin(\varphi/2)\ne0$, an equivalent real expression is
\begin{equation}
\label{eq:periodic-factor-real}
 \Upsilon_n(\varphi)=1+\Bigl(\frac{\sin(n\varphi/2)}{n\sin(\varphi/2)}\Bigr)^2
       -2\,\frac{\sin(n\varphi/2)}{n\sin(\varphi/2)}\,
          \operatorname{sinc}(n\varphi/2)\cos\bigl((2n-1)\varphi/2\bigr).
\end{equation}
A client with a finite window of $n_m$ samples contributes
\begin{equation}
\label{eq:periodic-client-risk}
 \frac{w_m^2}{2L}
      \Bigl[\frac{A_m^2}{2}\Upsilon_{n_m}(\varphi_m)
                                      +\frac{\sigma_m^2}{n_m}\Bigr]
\end{equation}
to $\mathcal{Q}_\infty$; a prefix client contributes the same expression
with the bracket replaced by
$\frac12A_m^2(1+\mathbf1\{\varphi_m\in2\pi\mathbb Z\})$. The long-run
risk is the sum of these contributions over clients. In particular, a
stationary prefix contributes zero, and a nonaliased moving prefix
contributes $w_m^2A_m^2/(4L)$.

<!-- proof-9: body_md -->
The population objective is $F^t(x)=Lx^2/2-\vartheta(t)x$ with
$\vartheta(t)=\sum_mw_m[\bar\vartheta_m+A_m\cos(\omega_mt+\Phi_m)]$, and
its gap at $x$ is $(Lx-\vartheta(t))^2/(2L)$. For $t\ge t_{\mathrm{s}}$, the error
of client $m$'s empirical mean is
\[
 A_m\operatorname{Re}\Bigl\{
       e^{\mathrm{i}\Phi_m}\bigl[\mathcal K_m(t)-e^{\mathrm{i}\omega_mt}\bigr]
                        \Bigr\}
 +\frac{\sigma_m}{n_m(t)}\sum_{i\in\mathcal M_m(t)}\varepsilon_{m,i}.
\]
It has zero mean over phase and noise, and independence across clients
removes all cross terms even when the replacement times differ. The
identity $\mathbb{E}_\Phi(\operatorname{Re}(e^{\mathrm{i}\Phi}z))^2=|z|^2/2$ and the
independence of the noise give (\ref{eq:periodic-finite}), including the
fixed-initialization interval.

For a completed window of length $h=n/\lambda$ and $t=ph+u$ with
$0\le u<h$, the noiseless complex coefficient is
$\mathcal K(t)=e^{\mathrm{i}\omega ph}\mathcal K_n(\varphi)$, so the phase-averaged
squared bias, divided by $A^2/2$, equals
$|\mathcal K_n(\varphi)-e^{\mathrm{i}\omega u}|^2$. Averaging over $u\in[0,h)$ and
using
\[
 \frac1h\int_0^h e^{-\mathrm{i}\omega u}\,du
       =e^{-\mathrm{i} n\varphi/2}\operatorname{sinc}(n\varphi/2)
\]
gives (\ref{eq:periodic-factor}); the geometric sum for $\mathcal K_n$ gives
(\ref{eq:periodic-factor-real}). The noise variance is exactly
$\sigma^2/n$. The initial finite prefix and the incomplete last holding
interval vanish after division by $T$.

For a prefix with $N(t)=\lfloor\lambda t\rfloor$ samples, the complex
sample average tends to zero if $\varphi\notin2\pi\mathbb Z$, by the
finite geometric-sum formula, and is identically one if
$\varphi\in2\pi\mathbb Z$; in the latter case $\omega>0$ and the time
average of $|1-e^{\mathrm{i}\omega t}|^2$ equals two. The prefix noise variance
$\sigma^2/N(t)$ has vanishing time average. Boundedness justifies
passing to these limits in (\ref{eq:periodic-finite}).

<!-- proposition-5: body_md -->
\label{prop:periodic-common-prefix}
Suppose one client has a nonaliased sinusoidal mean with amplitude $A$,
frequency $\omega$, and weight $w$, while all other clients are
stationary, and recall $\Xi^2=\sum_mw_m^2\sigma_m^2/\lambda_m$.
Restrict the estimator to uniformly weighted completed-window empirical
minimization with one fixed shared length $h$, or to the full prefix at
every client. For a real length $h$, the completed interval
$((p-1)h,ph]$ contains the actual observations of each client, and its
sample count may vary with $p$. For every real $h$ with $\lambda_mh\ge1$
at every client, and for every fixed phase,
\begin{equation}
\label{eq:periodic-common-all-real}
 \liminf_{T\to\infty}\mathrm{AvgErr}_T^{\rm shared}
 \ge\frac{w^2A^2}{4L}
             +\frac{\Xi^2-2w^2A^2/\omega}{2Lh}.
\end{equation}
Hence $\Xi^2\ge2w^2A^2/\omega$ makes every such shared window no better
than the prefix, and strict inequality makes every finite shared window
worse. For the phase-averaged risk at commensurate lengths
$\lambda_mh\in\mathbb N$, the sharper sufficient condition
$\Xi^2\ge w^2A^2/\omega$ holds.
<!-- end proposition-5 -->
<!-- proof-10: body_md -->
First let $h$ be any real length satisfying the count condition, so that
every completed interval contains at least one observation per client.
The empirical minimizer is constant on each holding interval, even if
the server reports more often. Whatever its deterministic conditional
mean on that interval, its integrated squared bias is at least the
within-interval variance of $wA\cos(\omega t+\Phi)$. The long-run
average of $\cos^2(\omega t+\Phi)$ is $1/2$, whereas the square of the
mean cosine over any interval of length $h$ is at most
$\operatorname{sinc}^2(\omega h/2)$. Therefore the long-run squared bias has lower
limit at least $w^2A^2[1/2-\operatorname{sinc}^2(\omega h/2)]$.

If $n_{m,p}$ is the count in client $m$'s $p$th completed interval, the
independent noise contributes variance $\sum_mw_m^2\sigma_m^2/n_{m,p}$.
The average count satisfies
$P^{-1}\sum_{p=1}^Pn_{m,p}=\lfloor\lambda_mPh\rfloor/P\to\lambda_mh$,
so Jensen's inequality gives
$\liminf_P P^{-1}\sum_{p=1}^P1/n_{m,p}\ge1/(\lambda_mh)$. Finally,
\[
 h\operatorname{sinc}^2(\omega h/2)
 \le\min\{h,\,4/(\omega^2h)\}\le2/\omega .
\]
Adding bias and variance and dividing by $2L$ proves
(\ref{eq:periodic-common-all-real}); bounded startup and incomplete last
intervals do not affect the lower limit. Nonaliasing ensures that the
full prefix has risk $w^2A^2/(4L)$ for every fixed phase.

For the sharper phase-averaged statement, use a commensurate length and
write $v=e^{-\mathrm{i}\omega h/2}\operatorname{sinc}(\omega h/2)$. Completing the square in
(\ref{eq:periodic-factor}) yields
\[
 1-\Upsilon_n(\varphi)
 =2\operatorname{Re}(\mathcal K_nv)-|\mathcal K_n|^2
 \le |v|^2=\operatorname{sinc}^2(\omega h/2).
\]
The shared-window risk minus the prefix risk is therefore at least
\[
 \frac1{2L}\Bigl[
          \frac{\Xi^2}h-\frac{w^2A^2}{2}\bigl(1-\Upsilon_n(\varphi)\bigr)\Bigr]
 \ge\frac{\Xi^2-w^2A^2/\omega}{2Lh},
\]
and the assertion follows.

<!-- corollary-4: body_md -->
\label{cor:periodic-separation}
Take two clients with $w_1=w_2=1/2$, $L=1$, $\mathcal{X}=[-2,2]$ (so $D=4$),
$\bar\vartheta_1=\bar\vartheta_2=0$, $A_1=\omega_1=1$, and $A_2=0$. For
an integer $r\ge2$, set
\[
 \lambda_1=r,\quad\sigma_1=1/r,\qquad
 \lambda_2=1,\quad\sigma_2=\sqrt2 .
\]
Among all fixed real shared lengths $h\ge1$ in
Proposition~\ref{prop:periodic-common-prefix}, including the full prefix,
the smallest lower limit of the average risk is $1/16$, for every fixed
phase. The client-specific choice $h_1=1/r$ with a prefix at client two
has risk
\begin{equation}
\label{eq:periodic-separation}
 \mathcal{Q}_\infty^{\rm sep}
 =\frac18\Bigl[1-\operatorname{sinc}(1/r)+\frac1{r^2}\Bigr]
 \le\frac7{48r^2}.
\end{equation}
Thus the ratio between the best shared-window risk and this
client-specific risk is at least $3r^2/7$. The client-specific limit
also holds for every fixed phase $\Phi_1$.

<!-- theorem-5: body_md -->
\label{thm:adaptive-memory}
Under Section~\ref{sec:refresh-model}, except that $\gamma$ need not be
known to the learner, and under
(\ref{eq:adaptive-oscillation})--(\ref{eq:adaptive-timing}), the causal
procedure above satisfies
\begin{equation}
\label{eq:adaptive-oracle}
 \mathcal{R}_T^{\rm mix}
 \le\min_{1\le e\le E}\mathcal{R}_T^e
       +2B\sqrt{\frac{\Delta\log E}{T}}
       +D\bar\gamma\Delta .
\end{equation}
In particular, with $\widetilde b_m=3b_m$,
$\widetilde\Xi^2=\sum_m\widetilde b_m$, and
\[
 \widetilde{\mathcal{T}}_T
 =\beta_{\rm sgd}D\widetilde\Xi
       \Bigl(\frac2{\sqrt T}+\frac{\Delta}{T\sqrt{s_0}}\Bigr)
       +\frac{GD(s_0+c)}T ,
\]
we have
\begin{align}
\label{eq:adaptive-explicit}
 \mathcal{R}_T^{\rm mix}
 &\le \frac{LD^2}{K}
  +D\min_{h\in\mathcal{G}}
     \Bigl\{2\sum_m a_mh_m
          +\beta_{\rm sgd}\sqrt{\sum_m\widetilde b_m/h_m}\Bigr\}
       \nonumber\\
 &\quad+D\bar\gamma(c+2\Delta)
       +\widetilde{\mathcal{T}}_T
       +2B\sqrt{\frac{\Delta\log E}{T}} .
\end{align}
The algorithm uses the noise and rate bounds but not the values
$a_m=w_m\gamma_m$ that appear in the oracle bound.

<!-- proof-13: body_md -->
We first prove the selection inequality, which needs no gradient
concentration event. Condition on the full arrival schedule and on the
entire training array. Every expert point is then fixed, the validation
observations remain mutually independent, and at report $j$ the weight
vector $\pi_j$ depends only on previously used validation intervals and
is independent of the observations in $(t_j,t_{j+1}]$.

Population losses are centered at $x_{\circ}$ before integrating over time;
this removes arbitrary loss offsets, which the assumptions do not bound.
Define the centered average population objective on a reporting interval
and the centered population objective at the validation source times by
\[
 \overline F_j(x)=\frac1\Delta\int_{t_j}^{t_{j+1}}
                         [F^t(x)-F^t(x_{\circ})]\,dt,
 \qquad
 F_j^{\rm val}(x)=\sum_m\frac{w_m}{n_{m,j}^{\rm val}}
                  \sum_{(u,\xi)\in\mathcal V_{m,j}}
                         [f_m^u(x)-f_m^u(x_{\circ})] ;
\]
the latter uses only the timestamps $u$. Conditionally on the stated
information and on past validation feedback,
\begin{equation}
\label{eq:adaptive-unbiased}
 \mathbb{E}[\widehat{\ell}_{j,e}]=F_j^{\rm val}(x_j^e),
 \qquad |\widehat{\ell}_{j,e}|\le B .
\end{equation}
This conditional expectation evaluates finitely many training-measurable
points on a separate sample array; it asserts no uniform concentration
over queries.

Consider either parity of reporting intervals. The exponential-weight
potential and Hoeffding's lemma for losses in $[-B,B]$ give, for any
expert $e$ and $n$ intervals of that parity,
\[
 \sum_j\Bigl(\sum_v\pi_{j,v}\widehat{\ell}_{j,v}
                         -\widehat{\ell}_{j,e}\Bigr)
 \le\frac{\log E}{\eta_{\mathrm{ew}}}+\frac{\eta_{\mathrm{ew}} B^2n}{2} :
\]
the one-step upper bound for the log potential ratio is
$-\eta_{\mathrm{ew}}\sum_v\pi_{j,v}\widehat{\ell}_{j,v}+\eta_{\mathrm{ew}}^2B^2/2$, and the terminal weight of
expert $e$ gives the matching lower bound
$-\eta_{\mathrm{ew}}\sum_j\widehat{\ell}_{j,e}-\log E$. All earlier losses of a parity are
available before its next report. Adding the inequalities of the two
parities (one of which may be empty) and substituting $\eta_{\mathrm{ew}}$ yields the
pathwise bound
\begin{equation}
\label{eq:adaptive-hedge}
 \sum_{j=0}^{R-1}\Bigl(\sum_v\pi_{j,v}\widehat{\ell}_{j,v}
                          -\widehat{\ell}_{j,e}\Bigr)
 \le 2B\sqrt{R\log E}.
\end{equation}

Source-time drift gives
$\|\nabla F_j^{\rm val}(x)-\nabla\overline F_j(x)\|\le\bar\gamma\Delta$
uniformly on $\mathcal{X}$, because every source time and every integration time
lie in an interval of length $\Delta$. Consequently, for all $x,y\in\mathcal{X}$,
\[
 \bigl|[F_j^{\rm val}(x)-F_j^{\rm val}(y)]
       -[\overline F_j(x)-\overline F_j(y)]\bigr|
 \le D\bar\gamma\Delta ,
\]
and convexity of each $F^t$, hence of $\overline F_j$, gives
\[
 \overline F_j(x_j)-\overline F_j(x_j^e)
 \le\sum_v\pi_{j,v}\bigl[F_j^{\rm val}(x_j^v)-F_j^{\rm val}(x_j^e)\bigr]
                      +D\bar\gamma\Delta .
\]
Take expectations, use (\ref{eq:adaptive-unbiased}) and
(\ref{eq:adaptive-hedge}), and multiply by $\Delta/T$. The current
minimum cancels in the risk difference, and both procedures report
$x_{\circ}$ before $t_0$. We obtain
\[
 \mathcal{R}_T^{\rm mix}-\mathcal{R}_T^e
 \le\frac{2B\Delta\sqrt{R\log E}}T
          +\frac{R\Delta}{T}D\bar\gamma\Delta
 \le2B\sqrt{\frac{\Delta\log E}{T}}+D\bar\gamma\Delta .
\]
This holds for every deterministic index $e$, so taking the minimum of
the expected expert risks proves (\ref{eq:adaptive-oracle}).

Finally, Lemma~\ref{lem:adaptive-counts} supplies the only arrival-count
input used in Proposition~\ref{prop:refresh-sgd}, with lower rate
$\lambda_m/3$. Frozen training intervals have length
$\min\{s_j,h_m^e\}\ge6K/\lambda_m$ and contain at least $K$
observations. Each expert retains the original reset and disjoint-batch
filtration, and shared observations between experts impose no
independence requirement on their marginal bounds. The proof of that
proposition therefore applies with $b_m$ replaced by $3b_m$ and with the
initialization time of (\ref{eq:adaptive-timing}). Substituting into
(\ref{eq:adaptive-oracle}) proves (\ref{eq:adaptive-explicit}).
