<!-- theorem-1: body_md -->
\label{thm:main}
Fix $\alpha>0$ and let $M/N\to\alpha$.  If $\mu$ has mean zero, variance
one, and a finite fourth moment, then
\[
\inf_{\varepsilon\,\mathrm{online}}\mathbb{E}_\mu D_{M,N}\longrightarrow R_\alpha.
\]
Moreover:
\begin{enumerate}[label=(\roman*),leftmargin=2.4em]
\item For every fixed $\kappa>R_\alpha$ and $K<\infty$, there is a
randomized online algorithm, depending only on $(\alpha,\kappa,K)$, such
that for every centered law $\mu$ with variance one and $\mathbb{E}_\mu|\xi|^4\le K$,
\[
\mathbb{P}_\mu(D_{M,N}>\kappa)+\mathbb{E}_\mu(D_{M,N}-\kappa)_+
\longrightarrow0.
\]
\item For every fixed $\kappa<R_\alpha$,
\[
\sup_{\varepsilon\,\mathrm{online}}\mathbb{P}_\mu(D_{M,N}\le\kappa)
\longrightarrow0.
\]
\end{enumerate}
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{cor:sbp}
For every $\kappa>0$ and every $\mu$ as in \cref{thm:main},
$\alpha_{\mathrm{on}}^{\mathrm{SBP},\mu}(\kappa)$ is independent of $\mu$ and is the unique $\alpha>0$
satisfying
\[
\kappa=R_\alpha=\sqrt{\alpha}\,V^\infty_{1/\alpha}.
\]
Every fixed density below this value is achievable by an online
algorithm.  At every fixed density above it, the supremal success
probability tends to zero.
<!-- end corollary-1 -->
<!-- theorem-2: body_md -->
\label{thm:quantitative}
Fix $\alpha>0$, $\kappa>R_\alpha$, $K<\infty$, and $A>0$.
There is a randomized online algorithm, depending only on these
parameters, such that every centered law $\mu$ with variance one and
$\mathbb{E}_\mu|\xi|^4\le K$ satisfies
\begin{equation}
\mathbb{P}_\mu(D_{M,N}>\kappa)+\mathbb{E}_\mu(D_{M,N}-\kappa)_+
\le C_A\left[N^{-A}+\mathfrak t_\mu(N^{1/3})\right].
\label{eq:main-tail-modulus}
\end{equation}
The bound holds uniformly for all sufficiently large $N$ and all $M/N$
in a sufficiently small fixed neighborhood of $\alpha$.
The constant, the neighborhood, and the lower bound on $N$ depend only on
$(\alpha,\kappa,K,A)$.

<!-- corollary-2: body_md -->
\label{cor:higher-moment-rate}
Fix $\alpha>0$, $q>4$, $K<\infty$, and $\kappa>R_\alpha$, and let
$M/N\to\alpha$.  One randomized online algorithm satisfies
\begin{equation}
\sup_{\substack{\mathbb{E}_\mu\xi=0,\ \mathbb{E}_\mu\xi^2=1\\
                \mathbb{E}_\mu|\xi|^q\le K}}
\left\{\mathbb{P}_\mu(D_{M,N}>\kappa)+\mathbb{E}_\mu(D_{M,N}-\kappa)_+\right\}
=O\!\left(N^{-(q-4)/2}\right).
\label{eq:higher-moment-rate}
\end{equation}

<!-- lemma-1: body_md -->
\label{lem:low-influence-response}
Let $\zeta_1,\ldots,\zeta_M$ be independent copies of a law $\nu$ with
mean $m$, variance $\sigma^2\in[1/2,2]$, and bounded third absolute moment.
Write $\zeta=(\zeta_1,\ldots,\zeta_M)$, suppose that
$\lVert v\rVert_2=1$ and $\lVert v\rVert_\infty\le1/2$, and put $d=m\sum_jv_j$.
For $0\le\rho\lec_{*}-\delta$,
\begin{align}
\mathbb{E}[\zeta_i\psi_\rho(\langle v,\zeta\rangle)]&=\rho v_i+e_i,\nonumber\\
|e_i|&\le C\rho\left[
 |v_i|(\lVert v\rVert_\infty+|\sigma^2-1|+|d|)+|m|\right].
\label{eq:response-bound}
\end{align}
Moreover,
\begin{equation}
t:=\mathbb{E}\psi_\rho(\langle v,\zeta\rangle),\qquad
|t|\le C\rho\left(\sum_j|v_j|^3+|d|\right)
\le C\rho(\lVert v\rVert_\infty+|d|).
\label{eq:sign-bias}
\end{equation}
Constants depend only on $\delta$ and the third-moment bound.
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
Write $\zeta_i^\circ=\zeta_i-m$ and
$A_i=\sum_{j\ne i}v_j\zeta_j^\circ$.  A first-order Taylor expansion,
using $\mathbb{E}\zeta_i^\circ=0$, gives
\begin{align}
\mathbb{E}[\zeta_i^\circ\psi_\rho(A_i+d+v_i\zeta_i^\circ)]
 &=v_i\sigma^2\mathbb{E}\psi_\rho'(A_i+d)+R_i,
\label{eq:Taylor-response}\\
|R_i|&\le C\rho v_i^2\mathbb{E}|\zeta_i^\circ|^3.
\label{eq:Taylor-remainder}
\end{align}
For $p\ge1$, we write $\mathbf d_p(\mu,\nu)$ for the $p$-Wasserstein
distance: the infimum of $\|X-Y\|_{L^p}$ over couplings with laws
$\mu$ and $\nu$.  The Wasserstein Berry--Esseen inequality for independent
weighted sums
\cite[Theorem~3.1]{ChenGoldsteinShao11} implies
\begin{equation}
\mathbf d_1\!\left(\operatorname{Law}(A_i),
       N(0,\sigma^2(1-v_i^2))\right)
 \le C\sum_{j\ne i}|v_j|^3\le C\lVert v\rVert_\infty.
\label{eq:Wasserstein-BE}
\end{equation}
The variance $\sigma^2(1-v_i^2)\ge3/8$ keeps the normalizing factor
in the Berry--Esseen bound uniformly bounded.  By Lipschitzness of
$\psi_\rho'$,
\begin{equation}
\left|\mathbb{E}\psi_\rho'(A_i+d)-\mathbb{E}\psi_\rho'(G)\right|
 \le C\rho(\lVert v\rVert_\infty+|\sigma^2-1|+|d|).
\label{eq:derivative-CLT}
\end{equation}
Gaussian integration by parts gives
\begin{equation}
\mathbb{E}\psi_\rho'(G)=\mathbb{E}[G\psi_\rho(G)]=\rho.
\label{eq:Gaussian-IBP}
\end{equation}
The remaining mean contribution is
$m\,\mathbb{E}\psi_\rho(\langle v,\zeta\rangle)$, bounded by $C\rho|m|$.
Combining these estimates proves \eqref{eq:response-bound}.
For \eqref{eq:sign-bias}, compare $\sum_jv_j\zeta_j^\circ$ with
$\sigma G$ in Wasserstein distance and shift by $d$.
Oddness gives $\mathbb{E}\psi_\rho(\sigma G)=0$ for every $\sigma$, yielding
\eqref{eq:sign-bias} directly.

<!-- lemma-2: body_md -->
\label{lem:exact-tilt}
Under \eqref{eq:sign-selection}, with
$t=\mathbb{E}\psi_\rho(\langle v,\zeta\rangle)$, for $\sigma\in\{-1,1\}$,
\begin{equation}
\mathbb{P}(\varepsilon=\sigma)=\frac{1+\sigma t}{2},\qquad
\frac{\,\mathrm{d}\operatorname{Law}(\zeta\mid\varepsilon=\sigma)}{\,\mathrm{d}\nu^{\otimes M}}(x)

\label{eq:product-tilt-density}
\end{equation}
In particular the conditional density lies between
\begin{equation}
\frac{1-a_\delta}{1+a_\delta}
\quad\text{and}\quad

\label{eq:tilt-bounds}
\end{equation}

<!-- lemma-3: body_md -->
\label{lem:product-innovation}
Under the hypotheses of \cref{lem:low-influence-response}, let $\varepsilon$
be chosen by \eqref{eq:sign-selection}.  Couple the actual pair
$(\varepsilon,\zeta)$ to a fair Rademacher sign $S$ and a vector
$Z\sim\nu^{\otimes M}$ such that $Z$ is independent of $(\varepsilon,S)$.
Writing $Y=\varepsilon\zeta$, one can arrange, for every $s\ge3$
with $\mathbb{E}|\zeta_1|^s<\infty$,
\begin{align}
\mathbb{E}|Y_i-SZ_i|^2&\le C(\rho|v_i|+|t|),
\label{eq:YZ-second}\\
\mathbb{E}|Y_i-SZ_i|^{s}&\le C_{s}\mathbb{E}|\zeta_1|^{s}.
\label{eq:YZ-q}
\end{align}
For
\begin{equation}
D=Y-SZ-\mathbb{E} Y,
\label{eq:D-definition}
\end{equation}
we have
\begin{equation}
\mathbb{E} D_i=0,\qquad
\mathbb{E} D_i^2\le C\rho(|v_i|+\lVert v\rVert_\infty+\sqrt M|m|),\qquad
\mathbb{E}|D_i|^{s}\le C_{s}\mathbb{E}|\zeta_1|^{s}.
\label{eq:D-moment-bounds}
\end{equation}
The constants in the second-moment bounds depend only on $\delta$ and
a third-moment bound; the $s$th moment enters linearly as displayed.
For fixed $\nu$, the conditional law of $(S,Z)$ given $(\varepsilon,\zeta)$
is measurable in the parameters $(\rho,v,t)$ and the observed pair.

<!-- proof-4: body_md -->
Maximally couple $\varepsilon$ to a fair sign $S$.  Given the observed pair,
sample $S$ from the conditional law
\[
\mathbb{P}(S=-\varepsilon\mid\varepsilon,\zeta)
 =\frac{(\varepsilon t)_+}{1+\varepsilon t}.
\]
This conditional probability depends only on $\varepsilon$, and
$\mathbb{P}(S\ne\varepsilon)=|t|/2$.  Conditioning further on $S$ therefore preserves
\eqref{eq:product-tilt-density}.

Fix $\varepsilon=\sigma$ and $S$.  Put $h_\sigma(x)=1+\sigma\psi_\rho(\langle v,x\rangle)$ and
\begin{equation}
H_i(x_1,\ldots,x_i)
 =\int h_\sigma(x_1,\ldots,x_i,y_{i+1},\ldots,y_M)
      \prod_{j>i}\nu(\,\mathrm{d} y_j).
\label{eq:Hi-definition}
\end{equation}
Here $H_0=1+\sigma t$ and $H_M=h_\sigma$.  Write
$x_{<i}=(x_1,\ldots,x_{i-1})$.  The conditional density of $\zeta_i$
with respect to $\nu$, given $\zeta_{<i}=x_{<i}$, is
\begin{equation}
g_i(x)=\frac{H_i(x_{<i},x)}{H_{i-1}(x_{<i})}.
\label{eq:gi-density}
\end{equation}
The normalization $1+\sigma t$ cancels; $H_{i-1}\ge1-a_\delta$.
Comparing $x$ with an independent $\zeta_i'$ gives
\begin{equation}
|g_i(x)-1|\le C_\delta\rho|v_i|(|x|+\mathbb{E}|\zeta_1|).
\label{eq:gi-minus-one}
\end{equation}
Apply maximal coupling to $g_i\,\,\mathrm{d}\nu$ and $\,\mathrm{d}\nu$ successively.
Each $Z_i$ is sampled conditionally on the observed prefix
$(\zeta_1,\ldots,\zeta_i)$, as described in \cref{app:coupling-kernels}.
Its conditional law given the preceding $\zeta$ and $Z$ coordinates and
$(\varepsilon,S)$ is $\nu$.  Induction therefore gives the asserted marginal law and independence.
The residual measures give
\begin{equation}
\mathbb{E}[|\zeta_i-Z_i|^2\mid\varepsilon,S,\zeta_{<i},Z_{<i}]
 \le C\int(1+x^2)|g_i(x)-1|\,\nu(\,\mathrm{d} x)
 \le C\rho|v_i|.
\label{eq:maximal-second-proof}
\end{equation}
Using
$Y_i-SZ_i=\varepsilon(\zeta_i-Z_i)+(\varepsilon-S)Z_i$
and independence of $Z$ from $(\varepsilon,S)$ proves
\[
\mathbb{E}|Y_i-SZ_i|^2
 \le C\rho|v_i|+8\mathbb{E} Z_i^2\,\mathbb{P}(S\ne\varepsilon)
 \le C(\rho|v_i|+|t|).
\]
The $s$th-moment bound follows from the marginals and
$|x-y|^{s}\le2^{s-1}(|x|^{s}+|y|^{s})$.
Since $\mathbb{E}[SZ_i]=0$, subtracting the mean proves
\eqref{eq:D-moment-bounds}, using \eqref{eq:sign-bias}.
The kernels are measurable by \cref{app:coupling-kernels}.

<!-- lemma-4: body_md -->
\label{lem:delocalization}
Fix $s>2$ and $\Lambda>0$.  Let $b_k$ be determined before the fresh
Rademacher perturbation $\eta_k$ is drawn.  Conditional on the information
available at that time,
\begin{equation}
\mathbb{P}\left(
\lVert \widetilde r_k\rVert_2^2<
\frac12(\lVert b_k\rVert_2^2+\gamma^2M)
\right)\le2e^{-cM}.
\label{eq:perturbation-norm-event}
\end{equation}
If in addition $M^{-1}\sum_i|b_{k,i}|^s\le\Lambda^s$, then on the
complement of this event,
\begin{equation}
\frac{\lVert \widetilde r_k\rVert_\infty}
     {\lVert \widetilde r_k\rVert_2}
\le C_{\Lambda,\gamma}M^{1/s-1/2}.
\label{eq:perturbation-delocalization}
\end{equation}
Along $M/N\to\alpha$, if the empirical moment bound holds at every
step, a union bound gives \eqref{eq:perturbation-delocalization}
simultaneously over $O(N)$ steps with probability $1-e^{-\Omega(N)}$.
Moreover,
\begin{equation}
\max_{i\le M,\,\ell\le N}\left|\frac\gamma N
\sum_{k=1}^{\ell}\eta_{k,i}\right|
=O_{\mathbb{P}}\left(\gamma\sqrt{\frac{\log M}{N}}\right)=o_{\mathbb{P}}(1),
\label{eq:perturbation-terminal}
\end{equation}
and
\begin{equation}
\frac{\lVert b_k+\gamma\eta_k\rVert_2}{\sqrt M}
\le\frac{\lVert b_k\rVert_2}{\sqrt M}+\gamma.
\label{eq:perturbation-energy}
\end{equation}
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
Expanding the square,
\begin{equation}
\lVert b_k+\gamma\eta_k\rVert_2^2
=\lVert b_k\rVert_2^2+\gamma^2M+2\gamma\langle b_k,\eta_k\rangle.
\label{eq:perturbation-square}
\end{equation}
A Rademacher tail bound proves \eqref{eq:perturbation-norm-event}; the
exponent is of order $M$ because
$(\lVert b\rVert_2^2+\gamma^2M)^2\ge4\gamma^2M\lVert b\rVert_2^2$.  On the complement of
the event in \eqref{eq:perturbation-norm-event}, the denominator in
\eqref{eq:perturbation-delocalization} is at least
$c\gamma\sqrt M$, while $\lVert b_k\rVert_\infty\le\Lambda M^{1/s}$.  Equation
\eqref{eq:perturbation-terminal} follows from Rademacher tail bounds and a union bound over the
coordinates and partial sums, and \eqref{eq:perturbation-energy} is the triangle inequality.

<!-- proposition-2: body_md -->
\label{prop:iterated-coupling}
Assume $M\asymp N$ and $\mu\in\mathcal{M}_4(K)$.  For all sufficiently large
$N$, on an enlarged probability space there are independent pairs
$(S_k,Z_k)$, with each $S_k$ a fair Rademacher sign, such that
\begin{equation}
\operatorname{Law}(S_k,Z_k\mid\mathcal{H}_{k-1})
 =\operatorname{Law}(S_k)\otimes\nu_N^{\otimes M}.
\label{eq:product-innovations}
\end{equation}
For every $0\le\ell\le N$,
\begin{equation}
\frac1{\sqrt N}\sum_{k\le\ell}\varepsilon_k\zeta_k
 =\frac1{\sqrt N}\sum_{k\le\ell}S_kZ_k
  +\frac1N\sum_{k\le\ell}r_k+E_\ell+Q_\ell,
\label{eq:iterated-decomposition}
\end{equation}
where $E_\ell$ is the accumulated predictable bias and $Q_\ell$ is
a martingale in $(\mathcal{H}_\ell)$.  They obey
\begin{align}
\max_{\ell\le N}\lVert E_\ell\rVert_\infty
 &\le C N^{-1/2+2/s},
\label{eq:iterated-bias}\\
\left\|\max_{\ell\le N}\lVert Q_\ell\rVert_\infty\right\|_{L^{s}}
 &\le C_s N^{-(s-4)/(6s)}.
\label{eq:iterated-martingale}
\end{align}
The constants and the lower bound on $N$ depend only on $K,s,\Lambda,\delta$
and fixed positive lower and upper bounds on $M/N$.
<!-- end proposition-2 -->
<!-- proof-6: body_md -->
Iterate the measurable kernel in \cref{lem:product-innovation} conditional
on the causal history.  The one-step identity is
\begin{equation}
\varepsilon_k\zeta_{k,i}
 =S_kZ_{k,i}+\frac{r_{k,i}}{\sqrt N}+e_{k,i}+D_{k,i}.
\label{eq:conditional-one-step}
\end{equation}
By \eqref{eq:truncation-moments} and the drift bounds,
\begin{align}
|e_{k,i}|&\le C\left[
 \frac{|r_{k,i}|}{\sqrt N}
 (M^{1/s-1/2}+N^{-2/3}+N^{-1/2})+N^{-1}\right],
\label{eq:conditional-e-bound}\\
\mathbb{E}[D_{k,i}\mid\mathcal{H}_{k-1}]&=0,
\label{eq:D-centered-conditional}\\
\mathbb{E}[D_{k,i}^2\mid\mathcal{H}_{k-1}]&\le C N^{-1/2+1/s},\qquad
\mathbb{E}[|D_{k,i}|^{s}\mid\mathcal{H}_{k-1}]
 \le C_{s} N^{(s-4)/3}.
\label{eq:D-conditional-moments}
\end{align}
The one-step residual $D_k$ is $\mathcal{H}_k$-measurable and centered
conditionally on $\mathcal{H}_{k-1}$.  The current perturbation has already
been revealed in $\mathcal{H}_{k-1}$, whereas future algorithmic coins remain
independent of that field.
The conditional second-moment bound includes the cost of coupling
$\varepsilon_k$ to $S_k$, as in \eqref{eq:D-moment-bounds}.
The fixed conditional law in \eqref{eq:product-innovations} makes the
pairs $(S_k,Z_k)$ independent, with the entire sequence $(S_k)$
independent of the array $(Z_{k,i})$.

Define $E_{\ell,i}=N^{-1/2}\sum_{k\le\ell}e_{k,i}$ and
$Q_{\ell,i}=N^{-1/2}\sum_{k\le\ell}D_{k,i}$.
Since $\lVert r_k\rVert_\infty\le\Lambda M^{1/s}$,
\begin{equation}
\max_{\ell,i}|E_{\ell,i}|
 \le C\left(N^{-1/2+2/s}+N^{-2/3+1/s}
                 +N^{-1/2+1/s}+N^{-1/2}\right)
 \le C N^{-1/2+2/s}.
\label{eq:bias-calculation}
\end{equation}
For each coordinate, Burkholder--Rosenthal \cite{Burkholder73} gives
\begin{align}
\mathbb{E}\max_{\ell\le N}|Q_{\ell,i}|^{s}
 &\le C_{s}\mathbb{E}\left(
 \frac1N\sum_{k=1}^N\mathbb{E}[D_{k,i}^2\mid\mathcal{H}_{k-1}]
 \right)^{s/2}
 +\frac{C_{s}}{N^{s/2}}\sum_{k=1}^N
 \mathbb{E}[|D_{k,i}|^{s}].
\label{eq:BR-coordinate}
\end{align}
Summing over $M=O(N)$ coordinates yields
\begin{equation}
\sum_i\mathbb{E}\max_{\ell\le N}|Q_{\ell,i}|^{s}
 \le C_{s}\left(N^{3/2-s/4}
       +N^{2-s/2+(s-4)/3}\right).
\label{eq:BR-summed}
\end{equation}
After taking the $s$th root, the two terms are
$N^{-1/4+3/(2s)}$ and $N^{-(s-4)/(6s)}$.  The first is smaller because
$s>12$.  This proves \eqref{eq:iterated-martingale}.  Only the jump term
uses the growing $s$th moment; the conditional variance bound uses a
uniform third-moment bound.

<!-- proposition-3: body_md -->
\label{prop:truncated-polynomial}
For every $\kappa_0>R_\alpha$, $K<\infty$, and $A>0$, parameters for
\cref{alg:online} can be chosen, independently of $\mu\in\mathcal{M}_4(K)$, so that
\begin{equation}
\mathbb{P}(D^{\mathrm{tr}}_{M,N}>\kappa_0)
 +\mathbb{E}[(D^{\mathrm{tr}}_{M,N}-\kappa_0)_+]
 \le C_A N^{-A}.
\label{eq:truncated-polynomial}

For the linearly interpolated martingales,
$\mathbb{E}|B_i(t)-B_i(s)|^4\le C_K|t-s|^2$ uniformly in $i$.

To obtain the sharp energy bound, take $L=\log M$ and truncate the drift:
$b_{k,i}=a_{k,i}\1_{\{|a_{k,i}|\le L\}}$.
If $\|b_k\|_2\ge M^{1/4}$, put $v=b_k/\|b_k\|_2$.  Then
$\|v\|_2=1$ and $\|v\|_\infty\le L/M^{1/4}$, so
\begin{align}
\frac{\|b_k\|_2}{\sqrt N}
 &=\mathbb{E}[\varepsilon_k\langle v,\xi_k\rangle\mid\mathcal{F}_{k-1}]
 \le\mathbb{E}[|\langle v,\xi_k\rangle|\mid\mathcal{F}_{k-1}]
 \lec_{*}+C_K L/M^{1/4}.
\label{eq:array-sharp-energy}
\end{align}
The last step is the Wasserstein Berry--Esseen inequality for independent
summands with different distributions.  If $\|b_k\|_2<M^{1/4}$, its
empirical squared norm is less than $M^{-1/2}$.  Consequently,
$M^{-1}\sum_i b_{k,i}^2\lec_{\alpha}^2+o(1)$ at every time, deterministically
and uniformly over online algorithms.
Let $a_i(t)=a_{k,i}$ and $b_i(t)=b_{k,i}$ for
$(k-1)/N<t\le k/N$, and set
$\widetilde A_i(t)=\int_0^t b_i(r)\,\,\mathrm{d} r$.  Then
\[
\frac1M\sum_i\lVert A_i-\widetilde A_i\rVert_\infty
 \le\frac1{ML}\sum_i\int_0^1 a_i(t)^2\,\,\mathrm{d} t
 \le\frac{N}{ML}\longrightarrow0.
\]
Thus truncation preserves every empirical limit.  Lower semicontinuity
of the weighted energies $\int_0^1 w(t)|\dot A_t|^2\,\,\mathrm{d} t$,
first for an arbitrary nonnegative continuous weight $w$ and then for
a countable dense family of such weights, shows that almost every limiting empirical
law has $\mathbb{E} a_t^2\lec_{\alpha}^2$ for almost every $t$.

To identify the Brownian motion in each limiting law, use bounded
continuous functions of finitely many past evaluations of $(A_i,B_i)$,
together with a bounded $C^3$ function $f:\mathbb{R}\to\mathbb{R}$ whose first three
derivatives are bounded, in the averaged discrete martingale problem.
By \eqref{eq:array-martingale-bounds}, the averaged linear martingale
term has variance $O(1/M)$.  The total error in its quadratic
compensator is at most
$C(MN)^{-1}\sum_k\lVert u_k\rVert_2^2\le C/M$, where $C$ depends on the test.
The centered quadratic term uses the identity
\[
d_{k,i}^2-\mathbb{E}[d_{k,i}^2\mid\mathcal{F}_{k-1}]
 =(\xi_{k,i}^2-1)-2u_{k,i}d_{k,i}.
\]
For predictable weights $|g_i|\le C$, independence of the current raw
coordinates and \eqref{eq:array-martingale-bounds} give conditional
variance at most $C_K M$ for the weighted sum of these terms.
After the factor $1/(MN)$ and summation over $N$ steps its variance is
$O(1/(MN))$.  The averaged cubic Taylor remainder has expectation
$O_K(N^{-1/2})$.
Round the observation times in the cylinder functions down to the mesh.
Tightness makes the rounding and Riemann-sum errors vanish.  The resulting bounded
continuous empirical test functionals converge, and the residual
vanishes in probability for a countable determining family of tests.
Every random empirical limit is supported
on laws under which $B$ is Brownian in the \emph{joint} $(A,B)$
filtration.  Combined with the sharp bound on drift energy, these are exactly the
admissible laws of \cref{prop:rectangular-compactness}.  Projection onto the Brownian
filtration and closedness of the terminal support condition then prove
infeasibility below $R_\alpha$, and hence the expected lower bound.

<!-- lemma-7: body_md -->
\label{lem:strict-monotonicity}
The function $\mathcal R$ is continuous and strictly decreasing on
$(0,\infty)$.  Consequently, $\alpha\mapsto R_\alpha$ is continuous and
strictly increasing from zero to infinity.
<!-- end lemma-7 -->
<!-- proof-17: body_md -->
A finite convex function on an open interval is continuous.  If a convex
nonincreasing function is constant on a nontrivial interval, its one-sided
slopes are zero there.  Convexity makes later slopes nonnegative, while
monotonicity makes them nonpositive, so the function remains constant
thereafter.  For $\mathcal R$ this would force the constant to be zero,
since $\mathcal R(c)\to0$ as $c\to\infty$.  But the entropy lower bound of
\cite[Proposition~6.5]{FiedlerJacksonLackerNilesWeed26}, after scaling,
gives $\mathcal R(c)>0$ for every finite $c>0$.
Thus $\mathcal R$ is strictly decreasing.  The identity
$R_\alpha=\mathcal R(c_{*}/\sqrt\alpha)$ proves the claim for $R_\alpha$.
<!-- end proof-17 -->
<!-- proof-18: body_md -->
By \cref{thm:main}, a fixed density $\alpha$ is achievable at margin
$\kappa$ whenever $R_\alpha<\kappa$, and is infeasible when
$R_\alpha>\kappa$.  By \cref{lem:strict-monotonicity}, the equation $R_\alpha=\kappa$ has
a unique solution.  The definition as a supremum identifies that
solution with $\alpha_{\mathrm{on}}^{\mathrm{SBP},\mu}(\kappa)$ for every admissible $\mu$.

<!-- lemma-8: body_md -->
\label{lem:Brownian-projection}
Let $B$ be a Brownian motion in a filtration $\mathcal{G}=(\mathcal{G}_t)_{0\le t\le1}$, and
let $a$ be progressively measurable with respect to $\mathcal{G}$, with
$\mathbb{E}\int_0^1a_t^2\,\,\mathrm{d} t<\infty$.  Let $C\subset\mathbb{R}$ be closed and convex and
suppose
\begin{equation}
B_1+\int_0^1a_t\,\,\mathrm{d} t\in C\qquad\text{a.s.}
\label{eq:projection-hypothesis}
\end{equation}
Let $u$ be the orthogonal projection of $a$ in
$L^2([0,1]\times\Omega,\,\mathrm{d} t\otimes\mathbb{P})$ onto the closed subspace of
processes predictable with respect to $\mathcal{F}^B$.  Since the Brownian filtration is
continuous, this projection satisfies
$u_t=\mathbb{E}[a_t\mid\mathcal{F}_t^B]$ for almost every $t$.  Then
\begin{equation}
B_1+\int_0^1u_t\,\,\mathrm{d} t
=\mathbb{E}\left[B_1+\int_0^1a_t\,\,\mathrm{d} t\,\middle|\,\mathcal{F}_1^B\right]
\in C\qquad\text{a.s.},
\label{eq:projection-terminal}
\end{equation}
and
\begin{equation}
\mathbb{E} u_t^2\le\mathbb{E} a_t^2
\qquad\text{for a.e.\ }t.
\label{eq:projection-energy}
\end{equation}
<!-- end lemma-8 -->
<!-- proof-19: body_md -->
Because $B$ is Brownian in $\mathcal{G}$, for every integrable $\mathcal{G}_t$-measurable
random variable $H$,
\begin{equation}
\mathbb{E}[H\mid\mathcal{F}_1^B]=\mathbb{E}[H\mid\mathcal{F}_t^B].
\label{eq:immersion-identity}
\end{equation}
Apply this to bounded truncations of $a_t$ and use conditional Fubini and
$L^1$ convergence.  This gives the equality in
\eqref{eq:projection-terminal}.  Conditional expectation preserves a closed
convex subset of the real line, which gives the last assertion of
\eqref{eq:projection-terminal}.  The energy bound is conditional Jensen.

<!-- lemma-9: body_md -->
\label{lem:empirical-moments}
Let $A_{j,i}$, $1\le j\le J_N$, $1\le i\le M$, be such that for each fixed
$j$ the variables $(A_{j,i})_{i\le M}$ are independent and identically
distributed, and suppose
\begin{equation}
J_N\le CN,
\qquad
M\asymp N,
\qquad
\sup_{N,j}\mathbb{E}|A_{j,1}|^p\le C_p
\label{eq:empirical-hypotheses}
\end{equation}
for some $p\ge2s$.  Then, for each fixed $\varepsilon>0$,
\begin{equation}
\mathbb{P}\left(
\max_{j\le J_N}\left|
\frac1M\sum_{i=1}^M|A_{j,i}|^s-
\mathbb{E}|A_{j,1}|^s
\right|>\varepsilon
\right)
\le C_{p,s,\varepsilon}J_N M^{-p/(2s)}
\le C'_{p,s,\varepsilon}N^{1-p/(2s)}.
\label{eq:empirical-moment-tail}
\end{equation}
The same estimate holds with $s$ replaced by any $1\le r\le s$.
<!-- end lemma-9 -->
<!-- proof-20: body_md -->
Apply Rosenthal's inequality with exponent $p/s\ge2$ to the centered
variables $|A_{j,i}|^s-\mathbb{E}|A_{j,i}|^s$.  It gives
\begin{equation}
\mathbb{E}\left|\frac1M\sum_{i=1}^M
(|A_{j,i}|^s-\mathbb{E}|A_{j,i}|^s)\right|^{p/s}
\le C M^{-p/(2s)}.
\label{eq:Rosenthal-mean}
\end{equation}
Markov's inequality and a union bound give the first estimate.
The assumptions $J_N=O(N)$ and $M\asymp N$ give the second.
