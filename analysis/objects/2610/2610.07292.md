<!-- remark-1: body_md -->
\label{rem:Phi-objective}
Although our methodology is formulated directly in terms of the estimating equation $\Psi(\ensuremath{\theta^{\star}})=0$, the operator $\Psi$ is also the gradient of a natural convex population objective. In particular, if we define
\begin{equation}
\label{eq:Phi-theta}
    \Phi(\theta)
    :=
    2\,\mathbb{E}\Bigl[
        \exp\Bigl\{-\mfrac{1}{2}Y\left\langle X,\theta \right\rangle\Bigr\}
    \Bigr],
    \qquad\text{for}\,~\theta\in\Theta_R,
\end{equation}
then $\nabla\Phi(\theta)=\Psi(\theta)$. The function $\Phi$ is the population exponential loss, a classical convex surrogate in binary classification; see, e.g., \citet{FriedmanHastieTibshirani2000logreg,Zhang2004classification,BartlettJordanMcAuliffe2006}. Thus, our stochastic approximation procedure can equivalently be viewed as projected stochastic gradient descent on $\Phi$, where we use the stochastic gradient $\widehat{\Psi}(\theta; Y, M, Z)$ as an unbiased estimator of $\nabla \Phi(\theta)$.

<!-- head-1: body_md -->
First, choose the universal constants $c_1, C_0, C_1$ in the sample-size condition Eq.~\eqref{eq:sample-size-cond-main-thm} to satisfy 
\begin{equation}
\label{eq:sample-size-cond-master}
    n \ge
    \frac{C}{\mu^2} \cdot \max\Biggl[
        \frac{e^{R/8}}{qR^2}\;\!,\;\!
        \exp\biggl(
            C \min\biggl\{
                \frac{R}{\sqrt{q}},\frac{1}{\mu}
            \biggr\}
            \log\biggl(\frac{e}{q}\biggr)
        \biggr)
    \Biggr].
\end{equation}
Since $e^{R/8} \geq 1$, this implies that the lower bound on the sample size $n$ in Proposition~\ref{prop:regime-dependent-lower} is in force.  

Applying Proposition~\ref{prop:baseline-assouad}, we obtain the lower bound
\begin{equation}
\label{eq:baseline-lower-final}
    \mathfrak R_n(R,\mu,q) \ge c\cdot\min\left\{
        R^2,\frac{e^{R/4}}{nq\mu^2} 
    \right\} 
    \ge
    c \frac{e^{R/8}}{nq\mu^2},
\end{equation}
where the final inequality follows as the sample-size condition
in Eq.~\eqref{eq:sample-size-cond-master} implies $e^{R/8}/(nq\mu^2) \leq R^2$.

Continuing, we define the quantity
\[
    \zeta(q,R,\mu)
    :=
    \begin{cases}
        \displaystyle \frac{R}{\sqrt q}\log(2-q),
        & \text{if } R\mu\le\sqrt q,\\ 
        \displaystyle \frac{1}{\mu}\log\left(1+\frac{1-q}{q}R^2\mu^2\right),
        & \text{if } \sqrt q<R\mu\le1,\\ 
        \displaystyle \frac{1}{\mu}\log(1/q),
        & \text{if } R\mu>1,
    \end{cases}
\]
and apply Proposition~\ref{prop:regime-dependent-lower} in conjunction with the numeric inequality $\log(2 - q) \leq 1$ to obtain the lower bound 
\begin{equation}
\label{eq:regime-lower-final}
    \mathfrak R_n(R,\mu,q)
    \ge
    \frac{c}{nq\mu^2}\exp\{c\,\zeta(q,R,\mu)\}.
\end{equation}
Finally, combining \crtCref{eq:baseline-lower-final,eq:regime-lower-final} and using $\max\{a,b\}\ge\sqrt{ab}$, we obtain
\[
    \mathfrak R_n(R,\mu,q)
    \ge
    \frac{c\,e^{R/16}}{nq\mu^2} \exp\left\{\frac{c}{2}\zeta(q,R,\mu)\right\}.
\]
This yields the desired result. \hfill \qed

<!-- remark-3: body_md -->
\label{rem:proof-intuition}
At this juncture, it is useful to pause to understand the intuition behind our construction through a heuristic calculation. 
For a complete observation $(Y,X)=(y,x(a,b))$, combining \crtCref{eq:generic-two-point-P1X} with the conditional laws of $Y$ under $P_0$ and $P_1$ yields
\[
\frac{\mathrm{d}P_1}{\mathrm{d}P_0}(y,x(a,b))
=
e^{\alpha yb}F_\alpha(a,b)
=
e^{\alpha yb}\bigl\{\cosh(\alpha)-yb\sinh(\alpha) \cdot J(T)\bigr\}.
\]
If, heuristically, we could take $J(T)=1$, then
\[
\frac{\mathrm{d}P_1}{\mathrm{d}P_0}(y,x(a,b))
=
e^{\alpha yb}\bigl\{\cosh(\alpha)-yb\sinh(\alpha)\bigr\}
=
e^{\alpha yb}e^{-\alpha yb}
=
1.
\]
Thus, the factor $F_\alpha$ is designed to cancel exactly the change in the conditional law of $Y$ induced by perturbing $\theta_0$ to $\theta_1$. Of course, an odd function $J$ cannot satisfy $J(T) \equiv c$ uniformly for any constant $c \neq 0$. Instead, we construct $J$ so that $\mathbb{E}_0[J(T)\mid T_S]=1$ whenever $|S|\le r$, since this suffices to control the observed-data likelihood ratio $\mathrm{d}Q_1/\mathrm{d}Q_0$; see \crtCref{lem:observed-likelihood-ratio-exact}.

\end{equation}

\]

\[

    \quad\Longrightarrow\quad
    \alpha^2 \le \frac{\widetilde{c}}{C}q \le q,
\]
 
since, by definition, $\widetilde{c} \le 3C/32 < C$. Similarly, since $\|J\|_\infty^2\le G_\lambda^m$ by \crtCref{lem:two-point-validity},
\[

\]
since, by definition, $\widetilde{c} \le 3C/32 < C/4$. Thus, $\alpha\|J\|_\infty\le\nicefrac{1}{2}$. Finally,
\[
    n \ge \frac{C}{\mathcal{B}_{m,\lambda,q}} \frac{p^2}{qR^2}
    \quad\Longrightarrow\quad
    p\alpha^2 \le \frac{\widetilde{c}}{C}R^2 \le \frac{3R^2}{32}.
\] 
Therefore, all conditions of \crtCref{lem:large-R-assouad-validity,lem:large-R-assouad-hamming} hold. 
Together with \crtCref{eq:generic-assouad-deterministic-conditions}, \crtCref{lem:large-R-assouad-validity} therefore ensures that every $P_\omega$ belongs to the required model class.
 
Hence, for every $k\in[p]$ and every neighboring pair $\bigl(\omega,\omega^{(k)}\bigr)$, \crtCref{lem:large-R-assouad-hamming} and \crtCref{eq:generic-assouad-alpha-choice} together yield
\[

        Q_{\omega^{(k)}}^{\otimes n},
        Q_\omega^{\otimes n}
    \right)
    \le
    \exp\left\{
        \frac{Cnq\alpha^2}{p}\,
        \mathcal{B}_{m,\lambda,q}
    \right\}-1
    \overset{\1}{=}
    \exp(\widetilde{c}\,C)-1
    \overset{\2}{\le}
    1,
\]

\end{equation}

    \frac{\widetilde{c}}{2}
    \frac{p^2}
    {nq\,\mathcal{B}_{m,\lambda,q}},
\]
which proves the lemma after renaming $c := \widetilde{c}/2$.

\begin{proof}[Proof of \crtCref{claim:lower-bound-regime-I}]

\medskip\noindent
\underline{\emph{Case 1: $u\le U_0$}}.
By \crtCref{prop:baseline-assouad} and the fact that $e^{R/4}\ge1$,
\[
    \mathfrak{R}_n(R,\mu,q)\ge c \cdot \min\left\{R^2,\frac{1}{nq\mu^2}\right\}.
\]
Hence, whenever $n\ge C/(q\mu^2R^2)$, we obtain $\mathfrak{R}_n(R,\mu,q)\ge c/(nq\mu^2)$. Since $u\le U_0$, we have
\begin{equation}
\label{eq:regime-I-small-u}
    \mathfrak{R}_n(R,\mu,q)\ge\frac{c}{nq\mu^2}\exp(cu)
    =\frac{c}{nq\mu^2}\exp\left\{c\frac{R}{\sqrt{q}}\right\}.
\end{equation}

\noindent
\underline{\emph{Case 2: $u > U_0$}}.
Set
\begin{equation}
\label{eq:regime-I-parameters}
    \lambda:=2\sqrt{q},
    \quad\,\text{and}\quad\,
    p:=\left\lfloor\frac{c''}{\mu}\right\rfloor.
\end{equation}

Also, let $m$ be the largest positive odd integer satisfying $m\le c'u$. Then, thanks to the choice of $U_0$, we have $m\ge5$, and since consecutive positive odd integers differ by $2$,
\begin{equation}
\label{eq:regime-I-m}
    \frac{2c'u}{3}\le m\le c'u.
\end{equation}
By \crtCref{eq:choice-a-b}, $m\lambda\le2c'R\le R/(4\sqrt{2})$. Moreover, since $u\le\mu^{-1}$ and $\lambda\le2$,
\[
    m\le\frac{c'}{\mu}\le\frac{\operatorname{sech}^2(2)}{2\mu}
    \le\frac{\operatorname{sech}^2(\lambda)}{2\mu}.
\]
Also, $u>U_0\ge2/c''$ and $u\le\mu^{-1}$ imply $c''/\mu>2$. Thus,
\begin{equation}
\label{eq:regime-I-p}
    \frac{c''}{2\mu}\le p\le\frac{c''}{\mu}.
\end{equation}
Since $c''\le1/2$, we also have $p\le (2\mu)^{-1}$. Therefore, the deterministic conditions \eqref{eq:generic-assouad-deterministic-conditions} in \crtCref{lem:generic-assouad-reduction} are satisfied.

We next control $\mathcal{B}_{m,\lambda,q}$ (\crtCref{eq:def-Bmlq}). For the present choice of $\lambda$, let $t := q/\sinh^2(2\sqrt{q})$.

\[
    \underline{\beta}:=\beta\left(\frac{1}{\sinh^2(2)}\right)>0,
    \qquad
    \overline{\beta}:=\beta\left(\frac{1}{4}\right)=\frac{1+\sqrt{17}}{8}<1.
\]

Since $\beta(\cdot)$ is a strictly increasing function on $[0,\infty)$, we have $\underline{\beta}\le\beta(t)\le\overline{\beta}$. Hence, using \crtCref{eq:regime-I-m},
\begin{equation}
\label{eq:regime-I-B-bounds}
    \exp(\underline{c}u)\le\mathcal{B}_{m,\lambda,q}^{-1}\le\exp(\overline{c}u)
\end{equation}
for universal constants $\underline{c},\overline{c}>0$.

\begin{equation}
\label{eq:regime-I-sample-size-control}

\begin{equation}
\label{eq:regime-I-large-u-sample-size}
    n\ge\frac{C}{\mu^2}
    \exp\left\{C'\frac{R}{\sqrt{q}}\log\left(\frac{e}{q}\right)\right\}.
\end{equation}

\begin{equation}
\label{eq:regime-I-large-u}

    \ge\frac{c}{nq\mu^2}
    \exp\left\{c\frac{R}{\sqrt{q}}\right\}.

\end{equation}

\begin{proof}[Proof of \crtCref{claim:lower-bound-regimes-II-III}]
Set $h:=\min\{R\mu,1\}$. Then, $\sqrt{q}\le h\le1$ and $h\le R\mu$. We first consider the range $0<\mu\le\mu_{\mathrm{crit}}$, where the Boolean construction with $m\ge5$ applies.

\medskip\noindent
\underline{\emph{Case 1: $0<\mu\le\mu_{\mathrm{crit}}$}}.
Let $m$ be the largest positive odd integer satisfying $m\le c'/\mu$, and set
\begin{equation}
\label{eq:regimes-II-III-parameters}
    \lambda := 2h = 2 \min\{R\mu,1\},
    \quad\,\text{and}\quad\,
    p:=\left\lfloor\frac{c''}{\mu}\right\rfloor.
\end{equation}
Since $\mu\le\mu_{\mathrm{crit}}\le\min\{c'/6,c''/2\}$, we have
\begin{equation}
\label{eq:regimes-II-III-mp}
    \frac{2c'}{3\mu}\le m\le\frac{c'}{\mu},
    \quad\,\text{and}\quad\,
    \frac{c''}{2\mu}\le p\le\frac{c''}{\mu}.
\end{equation}
Moreover, $m\lambda\le 2c'R\le R/(4\sqrt{2})$, while $\lambda\le2$ and \crtCref{eq:choice-a-b} give
\[
    m\le\frac{c'}{\mu}\le\frac{\operatorname{sech}^2(2)}{2\mu}
    \le\frac{\operatorname{sech}^2(\lambda)}{2\mu}.
\]
Since $c''\le1/2$, we have $p\le1/(2\mu)$. Hence, the deterministic conditions of \crtCref{lem:generic-assouad-reduction} are satisfied.

We next bound $\mathcal{B}_{m,\lambda,q}$. First, note that for any $t > 0$, $\beta(t)$ satisfies the following:
\begin{align}
\label{eq:beta-quadratic}
    \beta(t)^2
    &= 
    \frac{1}{4} (t^2 + 2t\sqrt{t^2 + 4t} + t^2 + 4t)
    =
    \frac{2t}{4} \Bigl[t + \sqrt{t^2 + 4t} + 2\Bigr]
    =
    \frac{t}{2} \Bigl[2 \, \beta(t) + 2\Bigr]
    =
    t\{1+\beta(t)\}.
\end{align}
 
Write $t:=q/\sinh^2(2h)$ and $z:=1/\beta(t)$. \crtCref{eq:beta-quadratic} yields that
\[
    z(z+1)=\frac{1}{t}=\frac{\sinh^2(2h)}{q}.
\]
Since $h\ge\sqrt{q}$ and $\sinh(2h)>2h$, we have $z(z+1)>4$, and hence $z>4/3$. Furthermore,
\[
    \frac{1-q}{q}h^2\le\frac{h^2}{q}\le\frac{z(z+1)}{4}.
\]
For $z>4/3$, $1+z(z+1)/4\le z^2$, and therefore
\[
    \log(z)\ge\frac{1}{2}\log\Bigl(1+\frac{1-q}{q}h^2\Bigr).
\]
Since $\mathcal{B}_{m,\lambda,q}^{-1}=z^m$, \crtCref{eq:regimes-II-III-mp} yields
\begin{equation}
\label{eq:regimes-II-III-B-lower}
    \mathcal{B}_{m,\lambda,q}^{-1}
    \ge
    \exp\biggl\{
        \frac{c'}{3\mu}
        \log\biggl(
            1+\frac{1-q}{q}\min\{R^2\mu^2,1\}
        \biggr)
    \biggr\}.
\end{equation}

It remains to verify the sample-size requirement. Since $2\sqrt{q}\le\lambda\le2$, we have $G_\lambda\le C/q$. Moreover, $z(z+1)=\sinh^2(2h)/q\le C/q$, so $z\le C/\sqrt{q}$, and hence
\[
    \mathcal{B}_{m,\lambda,q}^{-1}\le\exp(CmL_q),
    \quad\,\text{and}\quad\,
    \bigl(G_\lambda\bigr)^m\le\exp(CmL_q).
\]
Using $m\le c'/\mu$, $p\le c''/\mu$, and $R\mu>\sqrt{q}$, it follows that
\begin{equation}
\label{eq:regimes-II-III-sample-control}
    \frac{1}{\mathcal{B}_{m,\lambda,q}}
    \max\left\{
        \frac{p}{q^2}, \frac{p}{q}\bigl(G_\lambda\bigr)^m, \frac{p^2}{qR^2}
    \right\}
    \le
    \frac{C}{\mu^2}
    \exp\left\{\frac{C'}{\mu}L_q\right\}.
\end{equation}

\begin{equation*}

    \exp\left\{\frac{C_1}{\mu}\log\left(\frac{e}{q}\right)\right\}
\end{equation*}
allows us to invoke \crtCref{lem:generic-assouad-reduction}. Combining that lemma with \crtCref{eq:regimes-II-III-mp,eq:regimes-II-III-B-lower} yields
\begin{equation*}

    \ge
    \frac{c_0}{nq\mu^2}
    \exp\biggl\{
        \frac{c_1}{\mu}
        \log\biggl(
            1+\frac{1-q}{q}\min\{R^2\mu^2,1\}
        \biggr)
    \biggr\},
\end{equation*}
for universal constants $c_0,c_1>0$.

\noindent
\underline{\emph{Case 2: $\mu_{\mathrm{crit}}<\mu\le\nicefrac{1}{3}$}}.
 
In this range, the choice $m\asymp\mu^{-1}$ may no longer yield $m\ge5$, so we instead use a two-point construction with $m=1$. 
Since $\mu\ge\mu_{\mathrm{crit}}$, the quantity $1/\mu$ is bounded by a universal constant, and a fixed-dimensional construction is sufficient. 

Write $h:=\min\{R\mu,1\}$, and choose
\[
    \lambda:=\frac{h}{8},
    \quad~\text{and}\quad~
    \alpha^2:=c'''\frac{\sinh^2(\lambda)}{nq^2}
\]
for a sufficiently small universal constant $c'''$. In this case, we may assume a less stringent sample-size condition
\begin{equation}
\label{eq:moderate-mu-sample-size}
    n\ge C q^{-3}
\end{equation}
for a sufficiently large universal constant $C > 0$. We show the same lower bound on the minimax risk as in \crtCref{eq:lower-bound-regimes-II-III} holds under the sample-size condition in \crtCref{eq:moderate-mu-sample-size}. For brevity, we defer the formal construction and statement (see \crtCref{lem:moderate-mu-lower-bound}) to \crtCref{proof:lem:moderate-mu-lower-bound}.

Also, note that there exist sufficiently large universal constants $C_0,C_1>0$ such that the sample-size condition as in \crtCref{eq:regimes-II-III-sample-size} implies the sample-size condition as in \crtCref{eq:moderate-mu-sample-size}, since for $\mu_{\mathrm{crit}}<\mu\le\nicefrac{1}{3}$, choosing $C_1\ge1$ yields
\[
    \exp\left\{\frac{C_1}{\mu}\log\left(\frac{e}{q}\right)\right\}
    \ge q^{-C_1/\mu}
    \ge q^{-3}.
\]

Combining the two cases, we have
\begin{equation}
\label{eq:lower-bound-regimes-II-III-unified}
    \mathfrak{R}_n(R,\mu,q)
    \ge
    \frac{c_0}{nq\mu^2}
    \exp\biggl\{
        \frac{c_1}{\mu}
        \log\Bigl(
            1+\frac{1-q}{q}\min\{R^2\mu^2,1\}
        \Bigr)
    \biggr\}
\end{equation}
whenever $R\mu>\sqrt{q}$, and the sample-size condition in \crtCref{eq:regimes-II-III-sample-size} holds. This proves \crtCref{claim:lower-bound-regimes-II-III}.
\end{proof}

Let $r \in \mathbb{N} \cup \{0\}$ and suppose that $m = 2r + 1$.  The Fourier coefficients $\{g_U:U \subseteq [m]\}$ admit the representation
\[
g_U = 
\begin{cases}
~0\,, & \text{if~}\, |U|~ \text{is even};\\ 
~\displaystyle\frac{1}{2^{m-1}}
        \sum_{\substack{C\subseteq[m],\\ |C|\leq r}}
        (-1)^{|C\cap U|}\,, & \text{if~}\, |U|~ \text{is odd}.
\end{cases}
\]
Moreover, $\sum_{U \subseteq [m]} \, g_U^2 = 1$.
<!-- end lemma-11 -->
<!-- proof-5: body_md -->
We first consider the case in which $\lvert U \rvert$ is even.  Recall that
\[
g_U = \frac{1}{2^m} \sum_{t\in\{-1,1\}^m} f(t) \, \chi_U(t).
\]
Since the majority function is odd and the Fourier characters satisfy $\chi_U(-x)=(-1)^{|U|}\chi_U(x)$, it follows that $g_U = (-1)^{|U|+1}g_U$.  Since $\lvert U \rvert$ is even, we deduce that $g_U = 0$. 

Next, consider the case in which $\lvert U \rvert$ is  odd.  For $x\in\{-1,1\}^m$, define the set $C(x) \subseteq [m]$ to denote the indices of the negative components of $x$:
\[
    C(x):=\{j\in[m]:x_j=-1\},
\]
and use this to write the Fourier characters as $\chi_U(x)=(-1)^{|C(x)\cap U|}$.  Further, note that since $m = 2r + 1$, it holds that $f(x) = 1$ if and only if $\lvert C(x) \rvert \leq r$.  Combining these two facts yields the equivalent representation 
\[
g_U 
= \frac{1}{2^m} \sum_{t\in\{-1,1\}^m} f(t) \, \chi_U(t) 
= \frac{1}{2^m} \sum_{C \subseteq [m]} f_C \cdot (-1)^{|C\cap U|},
\]
where $f_C=1$ if $|C|\leq r$, and $f_C=-1$ if $|C|\geq r+1$.

Continuing, pair every set $C\subseteq[m]$ with its complement $C^c$. Since $m=2r+1$, exactly one member of each pair has cardinality at most $r$. Moreover, $f_{C^c}=-f_C$, and
\[
    (-1)^{|C^c\cap U|}
    =
    (-1)^{|U|-|C\cap U|}
    =
    -(-1)^{|C\cap U|},
\]
where the last equality uses the fact that $|U|$ is odd. Hence the two terms in each pair are equal. It thus follows that
\[
    g_U
    =
    \frac{1}{2^{m-1}}
    \sum_{\substack{C\subseteq[m]\\|C|\leq r}}
    (-1)^{|C\cap U|},
\]
as desired.

Finally, we note that the relation $\sum_{U \subseteq [m]} g_U^2 = 1$ follows from Parseval's identity.

<!-- lemma-12: body_md -->
\label{lem:boolean-cancellation}
Let $r \in \mathbb{N} \cup \{0\}$ and let $m = 2r + 1$.  The Fourier coefficients $\{g_U\}_{U \subseteq [m]}$ of the majority function satisfy the following.  For every $S\subseteq[m]$ satisfying $|S|\leq r$, and every $C\subseteq S$,
\[
    \sum_{\substack{
        U \subseteq [m],\\
        |U|\ \mathrm{odd},\\
        U\cap S=C
    }} g_U
    \,=\,
    \mathbbm{1}\{C=\varnothing\}.
\]
<!-- end lemma-12 -->
<!-- proof-6: body_md -->
Fix $S\subseteq[m]$ with $|S|\leq r$. For every $x_S\in\{-1,1\}^S$, consider the vector $(x_S,\boldsymbol{1}_{S^c})\in\{-1,1\}^m$. Regardless of the values of $x_S$, this vector has at least $m-|S|$ positive coordinates and at most $|S|$ negative coordinates. Since
\[
    m-|S| \geq 2r+1-r = r+1 > |S|,
\]
the positive coordinates form a strict majority. Therefore,
\[
    f(x_S,\boldsymbol{1}_{S^c})=1
    \qquad
    \text{for every}\,~x_S\in\{-1,1\}^S.
\]
 
Using the Fourier expansion of $f$, we thus have
\begin{align}
    1
    =
    f(x_S,\boldsymbol{1}_{S^c})
    =
    \sum_{U\subseteq[m]} g_U \chi_U(x_S,\boldsymbol{1}_{S^c}) \nonumber
    &=
    \sum_{U\subseteq[m]} g_U \prod_{j\in U\cap S} x_j \nonumber \\
    &=
    \sum_{C\subseteq S} \Bigg(\sum_{\substack{U\subseteq[m]\\U\cap S=C}} g_U\Bigg) \prod_{j\in C} x_j \nonumber\\
    &=
    \Bigg(\sum_{\substack{U\subseteq[m]\\U\cap S=\varnothing}} g_U\Bigg) +
    \sum_{\varnothing \neq C\subseteq S} \Bigg(\sum_{\substack{U\subseteq[m]\\U\cap S=C}} g_U\Bigg) \prod_{j\in C} x_j.
\label{eq:extended-maj-fourier-1}
\end{align}

On the other hand, note that the constant function $1$ has Fourier coefficient $1$ at $C=\varnothing$ and $0$ otherwise, since
\begin{align}
\label{eq:extended-maj-fourier-2}
    1
    =
    1 + \sum_{\varnothing \neq C\subseteq S} 0 \cdot \prod_{j\in C} x_j
\end{align}

Since the functions $\{\chi_C:C\subseteq S\}$ form an orthonormal basis for the space of real-valued functions on $\{-1,1\}^S$~\citep[Theorem~1.5]{odonnell2021analysisbooleanfunctions}, the Fourier expansion is unique. Therefore, comparing the Fourier coefficients of $f(x_S,\boldsymbol{1}_{S^c}) \equiv 1$ from \crtCref{eq:extended-maj-fourier-1,eq:extended-maj-fourier-2}, we obtain
\[
    \sum_{\substack{U\subseteq[m]\\U\cap S=C}} g_U
    =
    \mathbbm{1}\{C=\varnothing\},
    \qquad\text{for every}\,~ C\subseteq S.
\]

Finally, we use the result from \crtCref{lem:majority-coefficients}~\citep[or,][Theorem~5.19]{odonnell2021analysisbooleanfunctions} that $g_U=0$ for every even-cardinality $U$, so the inner sum may equivalently be restricted to odd-cardinality $U$. Thus,
\[
    \sum_{\substack{
        U \subseteq [m]\\
        |U|\ \mathrm{odd}\\
        U\cap S=C
    }}
    g_U
    \,=\,
    \mathbbm{1}\{C = \varnothing\},
\]
which completes the proof.

\label{lem:exact-hU}
For each $U \subseteq [m]$, the quantity $h_U$ as in \crtCref{eq:def-hU} depends only on $k:=|U|$, and has the following closed-form expression
\[
    h_U
    =
    \begin{cases}
        \, 1, & \text{if}\;~ k=0, \\ 
        \, 0, & \text{if}\;~ 1\le k\le r, \\ 
        \, \displaystyle \dfrac{1}{2^{k-1}} \, (-1)^r \, \binom{k-1}{r}, & \text{if}\;~ r+1\le k\le m.
    \end{cases}
\]
<!-- end lemma-13 -->
<!-- proof-7: body_md -->
Fix $U \subseteq [m]$, and write $k:=|U|$. Define the restriction
\[
    f_U(z) := f(z,\boldsymbol 1_{U^c}),
    \qquad
    \text{for}\;~z\in\{-1,1\}^U.
\]
Restricting the Fourier expansion of $f$ gives
\[
    f_U(z) = \sum_{W\subseteq[m]}g_W \cdot \chi_{W\cap U}(z).
\]
Therefore, the Fourier coefficient of the full monomial $\chi_U$ in $f_U$ is
\[
    \widehat{f}_U(U) = \sum_{W\supseteq U}g_W = h_U.
\]
Hence,
\begin{equation}
\label{eq:hU-restricted-Fourier}
    h_U
    =
    \frac{1}{2^{k}} \sum_{z\in\{-1,1\}^U} f_U(z) \cdot \chi_U(z).
\end{equation}

Suppose that $z$ has exactly $j$ coordinates equal to $-1$. Then, $\chi_U(z)=(-1)^j$, and the sum of all $m$ coordinates of $(z,\boldsymbol 1_{U^c})$ is
\[
    (m-k)+(k-2j) = m-2j = 2(r-j)+1.
\]
Since $m=2r+1$,
\[
    f_U(z)
    =
    \begin{cases}
        +1, & \text{if}\,~ j\le r,\\
        -1, & \text{if}\,~ j\ge r+1.
    \end{cases}
\]
Thus,
\begin{equation}
\label{eq:hU-binomial-sum}
    h_U
    = 2^{-k}
    \left\{
        \sum_{j=0}^{\min\{r,k\}}
        (-1)^j\binom{k}{j}
        -
        \sum_{j=r+1}^{k}
        (-1)^j\binom{k}{j}
    \right\},
\end{equation}
where the second sum is $0$ when $k\le r$.

\begin{itemize}
\item 
If $k=0$, then $h_\varnothing=f(\boldsymbol 1_m)=1$. Suppose henceforth that $k\ge1$. Since
\[
    \sum_{j=0}^k(-1)^j\binom{k}{j}=0,
\]
\crtCref{eq:hU-binomial-sum} becomes
\[
    h_U
    =
    2^{1-k} \sum_{j=0}^{\min\{r,k\}} (-1)^j\binom{k}{j}.
\]
\item 
If $1\le k\le r$, the sum on the right is the complete alternating binomial sum, and hence $h_U=0$.

\item
If $k\ge r+1$,  we have
\begin{align*}
    h_U = 2^{1-k} \sum_{j=0}^r(-1)^j\binom{k}{j}
    &=
    2^{1-k} \sum_{j=0}^r(-1)^j
    \left\{
        \binom{k-1}{j}
        +
        \binom{k-1}{j-1}
    \right\}
    \\
    &=
    2^{1-k} \left[\sum_{j=0}^r(-1)^j\binom{k-1}{j}
    -
    \sum_{j=0}^{r-1}(-1)^j\binom{k-1}{j}\right]
    \\
    &=
    \frac{1}{2^{k-1}} (-1)^r\binom{k-1}{r}.
\end{align*}
\end{itemize}
This proves the claimed formula.

$\mathcal{W}_{m,q}$, defined in \crtCref{eq:def-Wmq}, admits the representation:

Let us denote $\sigma_\rho:=\sqrt{1-\rho^2}$, and for all $U \subseteq [m] \setminus \varnothing$, define
\begin{equation}
\label{eq:psi-T}
    \psi_U(T)
    :=
    \prod_{j\in U} \underbrace{\frac{T_j-\rho}{\sigma_\rho}}_{=:\,\beta_{j}(\rho)},
    \qquad
    \psi_\varnothing\equiv1.
\end{equation}

Since $T_1,\ldots,T_m$ are independent, with $\mathbb{E}_0[T_j]=\rho$ and
$\operatorname{Var}_0(T_j)=1-\rho^2$, the collection $\{\psi_U:U \subseteq [m]\}$ is orthonormal under the null distribution of $T$. To see this, note the following.
\begin{itemize}
    \item For $A,B \subseteq [m]$ such that $A \neq B$, we have
    \begin{align*}
    \mathbb{E}\left[ \psi_A(T) \cdot \psi_B(T) \right]
    &=
    \mathbb{E}\left[ \prod_{j\in A \setminus B} \beta_{j}(\rho) \right] \cdot 
    \mathbb{E}\left[ \prod_{j\in B \setminus A} \beta_{j}(\rho) \right] \cdot 
    \mathbb{E}\left[ \prod_{j\in A \cap B} \beta_{j}(\rho)^2 \right] \\
    &= \prod_{j\in A \Delta B} \mathbb{E}\left[ \beta_{j}(\rho) \right] \cdot \prod_{j\in A \cap B} \mathbb{E}\left[ \beta_{j}(\rho)^2 \right] \\
    &= 0 \cdot \prod_{j\in A \cap B} \mathbb{E}\left[ \beta_{j}(\rho)^2 \right] \\
    &= 0.
    \end{align*}

    \item For $A = B$, we have
    \begin{align*}
    \mathbb{E}\left[ \psi_A(T)^2 \right]
    =
    \mathbb{E}\left[ \prod_{j\in A} \beta_{j}(\rho)^2 \right] 
    &= \prod_{j\in A} \mathbb{E}\left[ \beta_{j}(\rho)^2 \right] \\
    &= \prod_{j\in A} \frac{1}{1-\rho^2}\operatorname{var}\left[ T_{j} \right] \\
    &= \prod_{j\in A} \frac{1}{1-\rho^2} \left[ \frac{\exp(\lambda) + \exp(-\lambda)}{2 \cdot \cosh(\lambda)} - \rho^2 \right] 
    = \prod_{j\in A} \frac{1-\rho^2}{1-\rho^2} = 1.
    \end{align*}
\end{itemize}

Recall the definition of $\beta_j(\rho)$ in \crtCref{eq:psi-T}. For every $j\in[m]$, write
\[
    T_j
    =
    \rho + \sigma_\rho \frac{T_j-\rho}{\sigma_\rho}
    :=
    \rho + \sigma_\rho \, \beta_j(\rho).
\]
Thus, for every $W\subseteq[m]$, we have
\begin{align*}
    \chi_W(T) = \prod_{j \in W} (\rho + \sigma_\rho \beta_j(\rho)) 
    &= \prod_{j \in W} \left( \sum_{\mathcal{I}_j \in \{0, 1\}} \rho^{1 - \mathcal{I}_j} (\sigma_\rho \beta_j(\rho))^{\mathcal{I}_j} \right) \\ 
    &= \sum_{(\mathcal{I}_j)_{j \in W} \in \{0, 1\}^{|W|}} \prod_{j \in W} \rho^{1 - \mathcal{I}_j} (\sigma_\rho \beta_j(\rho))^{\mathcal{I}_j} \\ 
    &= \sum_{U \subseteq W} \left( \prod_{j \in W \setminus U} \rho \right) \left( \prod_{j \in U} \sigma_\rho \beta_j(\rho) \right) \quad \text{(letting } U = \{j \in W : \mathcal{I}_j = 1\}) \\ 
    &= \sum_{U \subseteq W} \rho^{|W| - |U|} \sigma_\rho^{|U|} \prod_{j \in U} \beta_j(\rho)\\ 
    &= \rho^{|W|} \sum_{U \subseteq W} \left(\frac{\sigma_\rho}{\rho}\right)^{|U|} \prod_{j \in U} \beta_j(\rho)
    ~=~ \rho^{|W|} \sum_{U \subseteq W} \left(\frac{\sigma_\rho}{\rho}\right)^{|U|} \psi_U(T)
\end{align*}

\begin{align*}

    =
    \sum_{W\subseteq[m]}
    g_W\rho^{-|W|}\chi_W(T)
    &= 
    \sum_{W\subseteq[m]}
    g_W\rho^{-|W|} 
    \rho^{|W|} \sum_{U \subseteq W} \left(\frac{\sigma_\rho}{\rho}\right)^{|U|} \psi_U(T)
    \\
    &=
    \sum_{U \subseteq [m]}
    \left(
        \frac{\sigma_\rho}{\rho}
    \right)^{|U|}
    \left(
        \sum_{W\supseteq U}g_W
    \right)
    \psi_U(T)\\ 
    &=
    \sum_{U \subseteq [m]}
    \left(
        \frac{\sigma_\rho}{\rho}
    \right)^{|U|}
    h_U \, \psi_U(T).
\end{align*}

    \right)^{|U|}

Write $h:=\min\{R\mu,1\}$. Set
\[
    \lambda:=\frac{h}{8},
    \quad~\text{and}\quad~
    \alpha^2:=c'' \frac{\sinh^2(\lambda)}{nq^2}.
\]
Since $R\mu>\sqrt{q}$, we have $\sqrt{q}\le h\le1$ and $0<\lambda\le1/8$. Also, the sample-size condition $n \ge Cq^{-3}$ implies that $1/(nq^2)\le q/C\le1/C$. In particular,
\[
    \alpha^2\le\frac{c'' \sinh^2(1/8)}{C}\,q\le q,
    \quad\;\text{and}\quad\;
    \frac{\alpha}{\tanh(\lambda)}
    =\cosh(\lambda)\sqrt{\frac{c''}{nq^2}}
    \le\cosh(1/8)\sqrt{\frac{c''}{C}}\le\frac{1}{2},
\]
after increasing $C$ if necessary.

We next verify the curvature condition. Since $\alpha^2\le q\le1$, we have $\alpha\le1$. Therefore, using $\sinh(2\alpha)\le2\alpha\cosh(2\alpha)\le2\alpha\cosh(2)$ and $\sinh(2\lambda)\ge2\lambda$,
\[
    \frac{\sinh(2\alpha)}{\sinh(2\lambda)}
    \le
    \cosh(2) \cdot \frac{\alpha}{\lambda}
    \le
    C\sqrt{\frac{c''}{nq^2}}
    \le
    C\sqrt{\frac{c''}{C}}
    \le\frac{1}{3} \le 1-2\mu,
\]
for $C$ sufficiently large. Finally, $h\le R\mu$ and $\mu\le1/3$ give $8\lambda^2=h^2/8\le R^2/72$. Moreover,
\[
    \alpha^2
    \le
    \frac{c''\cosh^2(1/8)}{64C}\,h^2,
\]
so, after increasing $C$ once more, $8(\lambda^2+\alpha^2)\le R^2$. Thus all the conditions of \crtCref{lem:two-dimensional-patch} are satisfied.
 
Since $\sinh(\lambda)\ge\lambda=h/8$, \crtCref{eq:two-dimensional-patch-lower} yields
\begin{equation}
\label{eq:moderate-mu-two-dimensional-bound}
    \mathfrak{R}_n(R,\mu,q)
    \ge
    c\frac{h^2}{nq^2}.
\end{equation}

We also use the baseline lower bound from \crtCref{prop:baseline-assouad}. Since $R\mu>\sqrt{q}$, we have $R^2>q/\mu^2$, and therefore
\[
    \frac{1}{q\mu^2R^2}<\frac{1}{q^2}.
\]
Thus, under the minimum sample size assumption $n\ge C q^{-3}$, we have $1/(nq\mu^2)\le R^2$. Since $\exp(R/4)\ge1$, \crtCref{prop:baseline-assouad} consequently yields
\begin{equation}
\label{eq:moderate-mu-baseline-bound}
    \mathfrak{R}_n(R,\mu,q)\ge\frac{c}{nq\mu^2}.
\end{equation}

Finally, choose a sufficiently small universal constant $c_1\in(0,\mu_{\mathrm{crit}})$, and define $\eta:=c_1/\mu\in(0,1)$ and $y:=(1-q)h^2/q$. Since $s\mapsto s^\eta$ is concave on $[0,\infty)$,
\[
    \frac{1}{nq\mu^2}(1+y)^\eta
    \le
    \frac{1}{nq\mu^2}(1+y)
    \le
    \frac{1}{nq\mu^2}
    +
    \frac{h^2}{nq^2\mu_{\mathrm{crit}}^2}.
\]
Combining \crtCref{eq:moderate-mu-two-dimensional-bound,eq:moderate-mu-baseline-bound}, and adjusting the universal constant gives
\begin{equation*}
    \mathfrak{R}_n(R,\mu,q)
    \ge
    \frac{c_0}{nq\mu^2}
    \exp\biggl\{
        \frac{c_1}{\mu}
        \log\biggl(
            1+\frac{1-q}{q}\min\{R^2\mu^2,1\}
        \biggr)
    \biggr\},
\end{equation*}
where we used $h^2=\min\{R^2\mu^2,1\}$. This completes the proof.
