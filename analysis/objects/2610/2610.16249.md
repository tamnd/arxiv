<!-- theorem-1: body_md -->
\label{thm:MLcond}
    Let $\ensuremath{\mathbf{G}} \in {\frak{G}_\xi^n}$ be a given FOM transfer function. Suppose that $\ensuremath{\mathbf{G}}_r \in {\frak{G}_\xi^r}$ is a local minimizer of $E$ in~\eqref{eq:H2errfunc}, then
\begin{equation}
    \label{eq:MLC}
    \ensuremath{\mathbf{G}}_r\big(\ensuremath{\widetilde{\lambda}}_j\big) = \ensuremath{\mathbf{G}}\big(\ensuremath{\widetilde{\lambda}}_j\big), \quad \ensuremath{\mathbf{G}}_r'\big(\ensuremath{\widetilde{\lambda}}_j\big) = \ensuremath{\mathbf{G}}'\big(\ensuremath{\widetilde{\lambda}}_j\big) \quad (j=1,\ldots,r),
\end{equation}
where $\ensuremath{\widetilde{\lambda}}_j$ denote the mirror image poles of $\ensuremath{\mathbf{G}}_r$.

<!-- theorem-2: body_md -->
\label{thm:thmnewconds}
    Let the FOM transfer function $\ensuremath{\mathbf{G}} \in {\frak{G}_\xi^n}$. Consider the \emph{equilibrium function} $F_\ensuremath{\mathbf{G}} : \ensuremath{\mathbb{D}}^r \to \ensuremath{\mathbb{C}}^r$, where for $\mathbf{z} = (z_1,\ldots,z_r) \in \ensuremath{\mathbb{D}}^r$,
    \begin{equation}\label{eq:FG}
\big[F_\ensuremath{\mathbf{G}}(\mathbf{z})\big]_j:= \sum_{k=1}^n \frac{\mathfrak{r}_k}{1 - \overline{\lambda_k} z_j} \overline{B^{\mathbf{z}}(\lambda_k)} \quad (j=1,\ldots,r),
    \end{equation}
where $B^{\mathbf{z}}$ is an $r$-term Blaschke product (see~\eqref{eq:Bprod}).
    If an order-$r$ ROM transfer function $\ensuremath{\mathbf{G}}_r \in {\frak{G}_\xi^r}$ is a local minimizer of $E$ from~\eqref{eq:H2errfunc}, then 
    \begin{equation}
        \label{eq:equibfun}
F_\ensuremath{\mathbf{G}}\big(\boldsymbol{\ensuremath{\widetilde{\lambda}}}\big) = 0,
    \end{equation}
    where $\boldsymbol{\ensuremath{\widetilde{\lambda}}} =                                                      
    \big(\ensuremath{\widetilde{\lambda}}_j\big)_{j=1}^r \in \ensuremath{\mathbb{D}}^r$       
    coincides with the mirror image poles of $\ensuremath{\mathbf{G}}_r$.
<!-- end theorem-2 -->
<!-- proof-1: body_md -->
Note that $\ensuremath{\mathbf{G}}_r = \ensuremath{\mathcal{S}}_r^{\boldsymbol{\ensuremath{\widetilde{\lambda}}}}[\ensuremath{\mathbf{G}}]$. By~\eqref{eq:discPointWise}, for any $\boldsymbol{\ensuremath{\widetilde{\lambda}}} \in \ensuremath{\mathbb{D}}^r$, since $\ensuremath{\mathbf{G}}_r = \ensuremath{\mathcal{S}}_r^{\boldsymbol{\ensuremath{\widetilde{\lambda}}}}[\ensuremath{\mathbf{G}}]$, we have 
    \begin{equation}
        \label{eq:interp}
        \ensuremath{\mathbf{G}}(\ensuremath{\widetilde{\lambda}}_j) = \ensuremath{\mathbf{G}}_r(\ensuremath{\widetilde{\lambda}}_j) \quad (j=1,\ldots,r).
    \end{equation}
    By Theorem~\ref{thm:MLcond}, a local minimzer $\ensuremath{\mathbf{G}}_r$ also satisfies
    \begin{equation}
        \label{eq:hinterp}
        \ensuremath{\mathbf{G}}'(\ensuremath{\widetilde{\lambda}}_j) = \ensuremath{\mathbf{G}}_r'(\ensuremath{\widetilde{\lambda}}_j) \quad (j=1,\ldots,r).
    \end{equation}
    We show that~\eqref{eq:hinterp} is equivalent to~\eqref{eq:equibfun} under our assumptions. Differentiating both sides of~\eqref{eq:discPointWise} gives
    \begin{multline}
        \label{eq:deriv}
       \ensuremath{\mathbf{G}}'(w) - \left( \ensuremath{\mathcal{S}}_r^{\mathbf{z}}[\ensuremath{\mathbf{G}}] \right)'(w) \\ =  \left(B^{\mathbf{z}}\right)'(w) \left( \sum_{k=1}^n \frac{\mathfrak{r}_k}{1 - \overline{\lambda}_k w} \overline{B^{\mathbf{z}}(\lambda_k)} \right) \\ + B^{\mathbf{z}}(w)  \left( \sum_{k=1}^n \frac{\mathfrak{r}_k}{1 - \overline{\lambda}_k w} \overline{B^{\mathbf{z}}(\lambda_k)}\right)',
    \end{multline}
    where $w \in \ensuremath{\mathbb{D}}$. Notice that by~\eqref{eq:Bprod}, for $w \in \big\{z_j\big\}_{j=1}^r$ we have $B^{\mathbf{z}}(w) = 0$. Fixing $\mathbf{z} = \boldsymbol{\ensuremath{\widetilde{\lambda}}}$ and choosing $w \in \big\{\ensuremath{\widetilde{\lambda}}_j\big\}_{j=1}^r$, i.e., the parameters that determine $\ensuremath{\mathbf{G}}_r$, it suffices to ensure
    $$
    \left( B^{\boldsymbol{\ensuremath{\widetilde{\lambda}}}}\right)'(\ensuremath{\widetilde{\lambda}}_j) \left( \sum_{k=1}^n \frac{\mathfrak{r}_k}{1 - \overline{\lambda}_k \ensuremath{\widetilde{\lambda}}_j} \overline{B^{\widetilde{\boldsymbol{\lambda}}}(\lambda_k)} \right) = 0 \quad (j=1,\ldots,r).
    $$
   We note that $\left(B^{\boldsymbol{\ensuremath{\widetilde{\lambda}}}}\right)'(\ensuremath{\widetilde{\lambda}}_j) = 0$ is only satisfied if $\ensuremath{\widetilde{\lambda}}_j$ is a mirror image pole with multiplicity at least $2$. Indeed, since
   $$
        B^{\boldsymbol{\ensuremath{\widetilde{\lambda}}}}(\ensuremath{\widetilde{\lambda}}_j) = \frac{P(\ensuremath{\widetilde{\lambda}}_j)}{Q(\ensuremath{\widetilde{\lambda}}_j)} = 0
   $$
   for polynomials $P$ and $Q$ of degree $r$ (see~\eqref{eq:Bprod}),
   we have $P(\ensuremath{\widetilde{\lambda}}_j) = 0$,  However,
   $$
        \left( B^{\boldsymbol{\ensuremath{\widetilde{\lambda}}}}\right)'(\ensuremath{\widetilde{\lambda}}_j) = \frac{P{'}(\ensuremath{\widetilde{\lambda}}_j) Q(\ensuremath{\widetilde{\lambda}}_j) - P(\ensuremath{\widetilde{\lambda}}_j)Q{'}(\ensuremath{\widetilde{\lambda}}_j)}{\left( Q(\ensuremath{\widetilde{\lambda}}_j) \right)^2} = \frac{P{'}(\ensuremath{\widetilde{\lambda}}_j)}{Q(\ensuremath{\widetilde{\lambda}}_j)},
   $$
   and thus $\left( B^{\boldsymbol{\ensuremath{\widetilde{\lambda}}}}\right)'(\ensuremath{\widetilde{\lambda}}_j) = 0$ is equivalent to $P'(\ensuremath{\widetilde{\lambda}}_j) = 0$. Since by assumption, the mirror image poles of $\ensuremath{\mathbf{G}}_r$ are simple, we therefore find $\left( B^{\boldsymbol{\ensuremath{\widetilde{\lambda}}}}\right)'(\ensuremath{\widetilde{\lambda}}_j) \neq 0$. Thus,  
    \begin{equation*}
        \ensuremath{\mathbf{G}}'(\ensuremath{\widetilde{\lambda}}_j) - \ensuremath{\mathbf{G}}_r'(\ensuremath{\widetilde{\lambda}}_j) = 0 
        \iff \big[F_{\ensuremath{\mathbf{G}}}(\boldsymbol{\ensuremath{\widetilde{\lambda}}})\big]_j = 0 \quad (j=1,\ldots,r).
    \end{equation*}
