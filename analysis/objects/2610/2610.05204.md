\label{thm:completion}
For every $t<0$, the metric space $(Z'_t,d_t^{Z'})$ is the metric
completion of $(R'_t,d_{g_t^{Z'}})$. In particular, $d_t^{Z'}$ is
finite-valued, and $(Z'_t,d_t^{Z'})$ is a proper length space.
If $X^{z'}$ is the metric soliton associated with $z'$, its canonical
embedding satisfies
\begin{equation}
 \iota_{z'}\bigl(X^{z'}_{(-\infty,0)}\bigr)
       =Z'_{(-\infty,0)}.
 \label{eq:surjectivity}
\end{equation}

\label{thm:convexity}
For every $t<0$, any minimizing geodesic in $(Z'_t,d_t^{Z'})$
that meets $R'_t$ has its entire interior in $R'_t$.
In particular, any two points of $R'_t$ are joined by a smooth
minimizing $g_t^{Z'}$-geodesic contained in $R'_t$.

\label{lem:cutoff-package}
There are smooth functions $\eta_\rho$ on the regular negative-time
spacetime, taking values in $[0,1]$, such that
\begin{equation}
 \begin{gathered}
 \eta_\rho>0\ \Longrightarrow\ r_{\operatorname{Rm}}\ge\rho,\qquad
 \eta_\rho=1\text{ on }\{r_{\operatorname{Rm}}\ge2\rho\},\\
 \rho|\nabla\eta_\rho|+
 \rho^2\bigl(|\partial_{\mathfrak t'}\eta_\rho|+|\nabla^2\eta_\rho|\bigr)
       \le C.
 \end{gathered}
 \label{eq:singular-cutoffs}
\end{equation}
Their supports on bounded regions of compact negative-time slabs
are relatively compact in the regular spacetime. For every
$\theta>0$, every compact interval $J\Subset(-\infty,0)$, and
every bounded spacetime region $Q\subset Z'_J$, we have, for all
sufficiently small $\rho>0$,
\begin{equation}
 \begin{split}
 \int_J\int_{Q_t\cap\{r_{\operatorname{Rm}}<2\rho\}}\,\mathrm d V_{g_t^{Z'}}\,\mathrm d t
       &\le C_{Q,J,\theta}\rho^{4-\theta},\\
 \sup_{t\in J}\operatorname{Vol}_{g_t^{Z'}}\bigl(Q_t\cap\{r_{\operatorname{Rm}}<2\rho\}\bigr)
       &\le C_{Q,J,\theta}\rho^{4-\theta}.
 \end{split}
 \label{eq:parabolic-bad-volume}
\end{equation}
Here $Q_t=Q\capR'_t$.
On the reference slice, for $A\ge1$,
\begin{equation}
 \operatorname{Vol}_g\bigl(B_g(p,A)\cap\{r_{\operatorname{Rm}}<2\rho\}\bigr)
       \le C_{A,\theta}\rho^{4-\theta}.
 \label{eq:slice-bad-volume}
\end{equation}
Finally, for every $J\Subset(-\infty,0)$ there is $C_J<\infty$
such that, for $A\ge2$,
\begin{equation}
 \sup_{t\in J}\operatorname{Vol}_{g_t^{Z'}}\{f\le A^2\}\le C_J A^n.
 \label{eq:sublevel-volume}
\end{equation}

<!-- proof-1: body_md -->
The cutoff properties and the first bound in
\eqref{eq:parabolic-bad-volume} follow from
\cite[Proposition~8.19 and equation~(8.19)]{FangLiStructure}.
For the second bound, sweep
$Q_t\cap\{r_{\operatorname{Rm}}<2\rho\}$ through a fixed small interval of soliton
times using $\boldsymbol{\psi}^s$. The swept set lies in a fixed bounded
negative-time region and has curvature radius at most $C_J\rho$.
Moreover, $\mathfrak t'\circ\boldsymbol{\psi}^s=e^{-s}\mathfrak t'$ and the slice volume
scales by $e^{-ns/2}$, so its spacetime volume is bounded below by
a fixed multiple of the original slice volume, uniformly for
$t\in J$. Applying the parabolic bound to this swept set gives
the exponent $4-\theta$ on every slice.
The reference-slice estimate \eqref{eq:slice-bad-volume} is also
\cite[Theorem~8.22(ii)]{FangLiStructure}.
Finally, \eqref{eq:sublevel-volume} follows from
\cite[Proposition~5.36]{FangLiStructure} and
\eqref{eq:slab-potential}, using soliton scaling to make the
constant uniform on $J$.

<!-- definition-1: body_md -->
\label{def:subcritical}
A function $u$ on $R'_J$, for $J\Subset(-\infty,0)$, has
\emph{Gaussian-subcritical growth} if, for every $\epsilon>0$, there is
$C_{J,\epsilon}<\infty$ such that
\begin{equation}
 |u(q)|\le C_{J,\epsilon}e^{\epsilon f(q)}\qquad(q\inR'_J).
 \label{eq:subcritical-growth}
\end{equation}
By \eqref{eq:slab-potential}, this is equivalently growth of order
$\exp(o(d_{g_t^{Z'}}(p_t,\cdot)^2))$, uniformly on $J$.
In particular, every polynomial-growth function satisfies this
condition.

<!-- lemma-2: body_md -->
\label{lem:kernel-decay}
For some $c,C>0$,
\begin{equation}
 K(y)\le Ce^{-cf(y)}\qquad(y\inR'_s,\ s\in I).
 \label{eq:kernel-potential-decay}
\end{equation}
Moreover, after decreasing $c$, for every $\epsilon>0$,
\begin{equation}
 \begin{split}
 |\nabla u|(y)&\le C_\epsilon(1+r_{\operatorname{Rm}}(y)^{-1})e^{\epsilon f(y)},\\
 |\nabla K|(y)&\le C(1+r_{\operatorname{Rm}}(y)^{-1})e^{-cf(y)}.
 \end{split}
 \label{eq:gradient-envelopes}
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Let $z_s$ be an $H_n$-center of $q$. The center inequality and
Wasserstein monotonicity give
\begin{align*}
 d_s(z_s,p_s)
 &\le d_{W_1}^{d_s}(\delta_{z_s},\nu_{q;s})
       +d_{W_1}^{d_s}(\nu_{q;s},\nu_{z';s})
       +d_{W_1}^{d_s}(\nu_{z';s},\delta_{p_s})\\
 &\le C_I+d_{W_1}^{d_\beta}(\nu_{q;\beta},\nu_{z';\beta}).
\end{align*}
The last term is finite. The reference centers $p_s$ have uniformly
bounded second moments for $\nu_{z';s}$ on $I$, by soliton scaling.
Thus the centers $z_s$ remain at bounded distance from $p_s$.
Equations~\eqref{eq:gaussian-kernel} and
\eqref{eq:slab-potential} imply
\eqref{eq:kernel-potential-decay}.

For $y\inR'_s$, $s\in I$, set
\[
 r=\frac{1}{2}\min\{1,r_{\operatorname{Rm}}(y),\operatorname{dist}(I,\{a,t\})^{1/2}\}.
\]
The cylinders $P(y,3r/2)$ lie in the regular spacetime over a fixed
compact interval $I^+\Subset(a,t)$. Apply the preceding kernel
bound on $I^+$. The curvature-radius bound and Shi's estimates on
these larger cylinders control the rescaled coefficients on
$P(y,r)$. By \eqref{eq:quadratic-potential}--\eqref{eq:slab-potential}
and standard local estimates,
\begin{equation}
 C^{-1}f(y)-C\le f(w)\le Cf(y)+C
       \qquad(w\in P(y,r)),
 \label{eq:local-potential-comparison}
\end{equation}
with uniform constants. Interior estimates for the heat equation give
\[
 |\nabla u|(y)\le Cr^{-1}\sup_{P(y,r)}|u|.
\]
Apply \eqref{eq:subcritical-growth} with a smaller exponent and use
\eqref{eq:local-potential-comparison} to obtain the first bound in
\eqref{eq:gradient-envelopes}. The same comparison gives
$\sup_{P(y,r)}K\le Ce^{-c'f(y)}$ for some $c'>0$.
Interior estimates for the conjugate heat equation, with the time
variable reversed, give the second bound.

<!-- corollary-1: body_md -->
\label{cor:moment}
For $s<t<0$ and $q\inR'_t$,
\[
 |t|\left(f(q)-\frac n2\right)
 =|s|\int_{R'_s}\left(f(y)-\frac n2\right)\,\mathrm d\nu_{q;s}(y).
\]
Equivalently,
\begin{equation}
 \int_{R'_s}f(y)\,\mathrm d\nu_{q;s}(y)
 =\frac{|t|}{|s|}f(q)
   +\left(1-\frac{|t|}{|s|}\right)\frac n2.
 \label{eq:moment-solved}
\end{equation}
<!-- end corollary-1 -->

\[

 =-\left(f-\frac n2\right)+\tau\frac{f-n/2}{\tau}=0.

This function has at most quadratic growth in the slice distance on

<!-- lemma-3: body_md -->
\label{lem:modified-time}
Let $J\Subset(-\infty,0)$ and $A<\infty$. Suppose $q_t$ is a
trajectory of $\partial_{\mathfrak t'}-\nabla f$ with $f(q_t)\le A$.
For $s,t\in J$ sufficiently close,
\begin{equation}
 d^*(q_s,q_t)\le C(n,Y,J,A)\sqrt{|t-s|}.
 \label{eq:modified-time-distance}
\end{equation}
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
Throughout the proof, $C$ depends only on $n,Y,J,A$ and may change
from line to line.
Assume $s<t$. Along the trajectory,
$f$ is constant and
$|\nabla f|^2+\mathrm{R}=f/\tau\le C$. The lower heat-kernel estimate
obtained from the reduced length of this regular curve gives
\begin{equation}
 K_{Z'}(q_t;q_s)\ge(4\pi(t-s))^{-n/2}e^{-C(t-s)}.
 \label{eq:curve-kernel-lower}
\end{equation}
This is the estimate used in
\cite[equation~(7.15)]{FangLiStructure}, obtained by applying
\cite[Corollary~9.4]{Perelman2002} on the closed approximating Ricci flows and
passing to the limit on the compact regular trajectory.
For an $H_n$-center $z_s$ of $q_t$, compare
\eqref{eq:curve-kernel-lower} with
\eqref{eq:gaussian-kernel}. This yields
$d_s(q_s,z_s)\le C\sqrt{t-s}$.
Equation~\eqref{eq:center} then implies
\[
 d_{W_1}^{d_s}(\delta_{q_s},\nu_{q_t;s})
 \le d_s(q_s,z_s)+\sqrt{H_n(t-s)}\le C\sqrt{t-s}.
\]
Choosing the comparison time $s$ in \eqref{eq:dstar} proves the
claim.

<!-- proof-5: body_md -->
Let $J\Subset(-\infty,0)$ be a compact interval. We prove that the
associated metric soliton restricted to $J$ is complete for $d^*$.
Let $(q_i)$ be a $d^*$-Cauchy sequence in that restriction.
By density we may replace it by a regular sequence at $d^*$-distance
tending to zero. Write $t_i=\mathfrak t'(q_i)\to t_0\in J$ and fix
$s<t_0$. For large $i$ all $t_i>s$.

The definition of $d^*$ and Wasserstein monotonicity imply
\[
 d_{W_1}^{d_s}(\nu_{q_i;s},\nu_{q_j;s})\longrightarrow0.
\]
If $c_i$ is an $H_n$-center of $q_i$ at time $s$, then
\begin{align*}
 d_s(c_i,c_j)
 &\le\sqrt{H_n(t_i-s)}
   +d_{W_1}^{d_s}(\nu_{q_i;s},\nu_{q_j;s})
   +\sqrt{H_n(t_j-s)}.
\end{align*}
Thus the centers remain in a bounded slice ball. The concentration
estimate gives the stronger conclusion
\[
 \int d_s(p_s,y)^2\,\mathrm d\nu_{q_i;s}(y)
 \le2d_s(p_s,c_i)^2+2H_n(t_i-s)\le C.
\]

By \eqref{eq:quadratic-potential},
$\int f\,\mathrm d\nu_{q_i;s}\le C$. Equation~\eqref{eq:moment-solved}
therefore yields
\begin{equation}
 \sup_i f(q_i)<\infty.
 \label{eq:cauchy-potential-bound}
\end{equation}
Transport $q_i$ by $\partial_{\mathfrak t'}-\nabla f$ to a point
$\widehat q_i$ at time $t_0$. The potential is unchanged, so the
lower bound in \eqref{eq:quadratic-potential} places
$\widehat q_i$ in a bounded ball of the associated metric soliton
slice. By the associated-soliton structure recalled in
Section~\ref{subsec:tangent-background}, this slice is a complete,
locally compact length space and hence proper. After passing to a subsequence,
\[
 \widehat q_i\longrightarrow q_\infty\quad\text{in }d_{t_0}.
\]
Lemma~\ref{lem:modified-time} and
\eqref{eq:cauchy-potential-bound} give
$d^*(q_i,\widehat q_i)\to0$, while
\eqref{eq:elementary-distances} gives
$d^*(\widehat q_i,q_\infty)\to0$. The original Cauchy sequence
thus converges in the associated metric soliton.

For a point of $Z'_{(-\infty,0)}$, approximate it by regular
spacetime points in a compact interval contained in $(-\infty,0)$.
The preceding completeness argument shows that its limit belongs
to the image of $\iota_{z'}$. This proves
\eqref{eq:surjectivity}. Since the associated metric soliton is
continuous, the distance-recovery identity used in the proof of
\cite[Lemma~6.5]{FangLiStructure} identifies its slice metrics with
the Fang--Li slice metrics in \eqref{eq:slice-distance-definition}.
The intrinsic
completion property of the associated soliton therefore identifies
the whole of
$(Z'_t,d_t^{Z'})$ with the completion of
$(R'_t,d_{g_t^{Z'}})$.

\label{thm:comparison}
There is $C=C(n)>1$ such that, for every $\tau>0$ and
$q\in Z'_{-\tau}$,
\begin{equation}
 C^{-1}\bigl(\sqrt\tau+d_{-\tau}^{Z'}(p_{-\tau},q)\bigr)
 \le d_{Z'}(z',q)
 \le C\bigl(\sqrt\tau+d_{-\tau}^{Z'}(p_{-\tau},q)\bigr).
 \label{eq:radial-comparison}
\end{equation}
The continuous extension of $f$ to $Z'_{-\tau}$ satisfies
\begin{equation}
 C^{-1}\frac{d_{Z'}(z',q)^2}{\tau}
 \le 1+f(q)\le C\frac{d_{Z'}(z',q)^2}{\tau}.
 \label{eq:potential-spacetime}
\end{equation}
In particular, for $r\ge1$,
\begin{equation}
 \begin{split}
 B_{d_{Z'}}(z',r)\cap Z'_{-1}
       &\subset B_{d_{-1}^{Z'}}(p,Cr),\\
 B_{d_{-1}^{Z'}}(p,r)
       &\subset B_{d_{Z'}}(z',Cr).
 \end{split}
 \label{eq:ball-comparison}
\end{equation}
<!-- end theorem-4 -->
<!-- proof-7: body_md -->
First take $q\inR'_{-\tau}$. The comparison time $-\tau$,
\eqref{eq:base-second-moment}, and the triangle inequality give
\begin{equation}
 D(q)\le d_{-\tau}(p_{-\tau},q)+C\sqrt\tau.
 \label{eq:radial-upper}
\end{equation}
Also $D(q)\ge\sqrt\tau$ by the time inequality.

Fix $L>D(q)$. There is a comparison time $-a$, with
$\tau\le a<L^2$, such that
\begin{equation}
 d_{W_1}^{d_{-a}}(\nu_{z';-a},\nu_{q;-a})<L.
 \label{eq:optimizing-time}
\end{equation}
For $a>\tau$, let $c$ be an $H_n$-center of $q$ at time $-a$;
for $a=\tau$, set $c=q$ and $\nu_{q;-a}=\delta_q$. We have
\begin{align}
 d_{-a}(p_{-a},c)
 &\le C\sqrt a+L+\sqrt{H_n(a-\tau)}\le CL,
 \label{eq:comparison-center}\\
 \int d_{-a}(p_{-a},y)^2\,\mathrm d\nu_{q;-a}(y)
 &\le2d_{-a}(p_{-a},c)^2+2H_n(a-\tau)\le CL^2.
 \label{eq:comparison-second-moment}
\end{align}
The upper potential estimate gives
\[
 \int f\,\mathrm d\nu_{q;-a}\le C\left(1+\frac{L^2}{a}\right).
\]
On the other hand, \eqref{eq:moment-solved} gives
\[
 \frac\tau a f(q)
   =\int f\,\mathrm d\nu_{q;-a}-\left(1-\frac\tau a\right)\frac n2
   \le C\left(1+\frac{L^2}{a}\right).
\]
Since $a<L^2$, we conclude that
$f(q)\le CL^2/\tau$. Using the lower potential estimate at time
$-\tau$, and then $L\searrow D(q)$, gives
\begin{equation}
 d_{-\tau}(p_{-\tau},q)\le C D(q).
 \label{eq:radial-lower}
\end{equation}
Together with \eqref{eq:radial-upper} and
$D\ge\sqrt\tau$, this proves
\eqref{eq:radial-comparison} on the regular set.

By \eqref{eq:f-identities}, $\sqrt f$ is
$1/(2\sqrt\tau)$-Lipschitz for the intrinsic slice distance and
extends continuously to the completion.
The function $D$ is also continuous there, since
$|D(q)-D(q')|\le d_{Z'}(q,q')\le d_{-\tau}(q,q')$.
Regular density thus proves \eqref{eq:radial-comparison} and the
quadratic potential bounds on the whole slice.
Combining these two comparisons yields
\eqref{eq:potential-spacetime}. Finally, setting $\tau=1$ proves
\eqref{eq:ball-comparison}, after increasing $C$.

<!-- corollary-3: body_md -->
\label{cor:proper-embedding}
For every $\tau>0$, the inclusion
\[
 (Z'_{-\tau},d_{-\tau}^{Z'})\longrightarrow(Z',d_{Z'})
\]
is proper and is a homeomorphism onto its image. A sequence on the
slice is intrinsically bounded if and only if its spacetime distance
from $z'$ is bounded. In particular, $Z'_{-\tau}$ is compact if and
only if it is bounded with respect to $d_{Z'}$.
<!-- end corollary-3 -->
<!-- proof-8: body_md -->
The inclusion is $1$-Lipschitz. By
\eqref{eq:radial-comparison}, the inverse image of a compact
spacetime set is bounded in the slice metric. It is also closed
there, hence compact by properness of the slice. To see the inverse
continuity directly, a spacetime-convergent slice sequence is
intrinsically bounded, so every subsequence has an intrinsically
convergent subsubsequence. The $1$-Lipschitz inclusion identifies
its limit with the given spacetime limit. This proves continuity
of the inverse. The final assertion follows again from properness
and the radial comparison.

<!-- lemma-4: body_md -->
\label{lem:initial-cap}
There are $h>0$, $\theta\in(0,h)$, and $V_0,C_0<\infty$, independent
of large $i$ and of the minimizing curve, such that these curves
remain in a fixed compact regular coordinate cap on $[1,1+h]$,
$|V_i(1)|\le V_0$, and
\begin{equation}
 D Z_i(v,1+\theta)\ge-C_0.
 \label{eq:initial-Z}
\end{equation}
<!-- end lemma-4 -->
<!-- proof-10: body_md -->
Choose a relatively compact regular spacetime neighborhood about
$x$. In its time-preserving coordinates the pulled-back metrics
are uniformly equivalent and the coordinate drift is uniformly
bounded. Before a first exit, displacement on $[1,1+h]$ is bounded
by
\[
 C\int_1^{1+h}|V_i|\,\mathrm d\tau+Ch\le C\sqrt{hE_0}+Ch.
\]
For sufficiently small $h$, this is less than the distance to the
lateral boundary. Thus an exit cannot occur in that interval.
The regular cap has uniform bounds for $\mathrm{Ric}_i$ and $\nabla \mathrm{R}_i$.
Equation~\eqref{eq:action-geodesic} gives
\[
 \frac{\mathrm d}{\mathrm d\tau}|V_i|^2
 =-2\mathrm{Ric}_i(V_i,V_i)+\langle\nabla \mathrm{R}_i,V_i\rangle
                         -\frac{|V_i|^2}\tau,
\]
so the absolute derivative is bounded by $C(1+|V_i|^2)$.
Some time in this interval has $|V_i|^2\le E_0/h$.
A simple ODE estimate then bounds the speed throughout the cap.

We give the corresponding Jacobi-field estimate. In regular local
frames along the curve, let $A_i(\tau)$ have columns given by the
components of $Y_{i,a}(\tau)=\mathrm dE_{i,\tau}(e_a)$, where
$\{e_a\}$ is $g_i(-1)$-orthonormal and the frames agree with it at
$\tau=1$. The $L$-Jacobi equation has the form
\[
 A_i''+B_iA_i'+C_iA_i=0,\qquad A_i(1)=0,\qquad A_i'(1)=\operatorname{Id};
\]
see \cite[Lemma~6.10, equation~(6.6), and Lemma~6.19]{MorganTian}.
Primes denote differentiation in $\tau$. Smooth convergence on the
fixed regular cap bounds the metric coefficients and their
required derivatives. Together with the speed bound, this gives
uniform bounds for $B_i$ and $C_i$, independent of $i$ and of the
minimizing curve. The ODE first bounds $A_i$ and $A_i'$ uniformly.
Integrating
\[
 A_i'(1+q)-\operatorname{Id}
       =-\int_1^{1+q}(B_iA_i'+C_iA_i)\,\mathrm d\tau
\]
and then integrating once more gives, uniformly as $q\searrow0$,
\[
 A_i(1+q)=q\operatorname{Id}+O(q^2),\qquad A_i'(1+q)=\operatorname{Id}+O(q).
\]
Let $G_i(\tau)$ be the metric matrix in these frames. Then
$G_i(1)=\operatorname{Id}$, and $G_i$, $G_i^{-1}$, and $G_i'$ are uniformly
bounded. For sufficiently small $q>0$, $\det A_i(1+q)>0$, and
\[
 J_i=\det A_i\sqrt{\det G_i},\qquad
 D\log J_i
   =\operatorname{tr}(A_i'A_i^{-1})
       +\frac12\operatorname{tr}(G_i^{-1}G_i')
   =\frac nq+O(1).
\]

\label{lem:relative-speed}
There is $C_V<\infty$, independent of $\xi$, such that every bad
minimizing curve satisfies
\begin{equation}
 |\dot\gamma_i|_g\le C_V\quad(\bar s\le s\le1),\qquad
 |V_i-U_i|_{g_i(-\bar\tau)}\le C_V'

<!-- proof-13: body_md -->
The terminal segment minimizes
$\widehat{A}_{i;\bar s,1}$ with its endpoints fixed and
therefore satisfies the Euler--Lagrange equation. For a tangent vector $w$,
\[
 (\partial_v\widehat L_i)[w]=2a_i(v,w)+2b_i(w).
\]
Define the positive energy
\[
 e_i=a_i(v,v)-c_i+2
       =(\partial_v\widehat L_i)[v]-\widehat L_i+2,
 \qquad v=\dot\gamma_i.
\]
The linear term cancels. Along an Euler--Lagrange curve,
\[
 \frac{\mathrm d}{\mathrm ds}
       \bigl(v^\alpha L_{v^\alpha}-L\bigr)
 =v^\alpha\left(\frac{\mathrm d}{\mathrm ds}L_{v^\alpha}
                              -L_{z^\alpha}\right)-L_s
 =-\partial_sL.
\]
Applying this identity to $\widehat L_i$ gives
\[
 e_i'=-(\partial_sa_i)(v,v)-2(\partial_sb_i)(v)-\partial_sc_i,
\]
where $\partial_s$ is taken at fixed $(z,v)$.

 1+\frac{1}{2}|v|^2\le e_i\le3+2|v|^2,
 \qquad |e_i'|\le C\epsilon_{i,\xi}e_i\le Ce_i.
\]
For any $s,t$ on the terminal segment, $e_i(s)\le e^Ce_i(t)$.

\eqref{eq:terminal-integral-energy},
\[
 e_i(s)\le\frac{e^C}{1-\bar s}\int_{\bar s}^1e_i(t)\,\mathrm d t
       \le\frac{8e^C}{\delta_b}(3+2E_1).
\]
This proves the first bound. At the contact point,
\[

       \le\sqrt{2\tau}\,\tau^{-3/2}|\dot\gamma_i|_g
       \le\sqrt2 C_V.
\]
All quantities extend continuously to the contact point in the
slightly larger regular coordinate domain.

<!-- lemma-7: body_md -->
\label{lem:flux}
On the part of this track contained in the normal region, the
absolute $n$-dimensional Jacobian of the restricted projection is
\begin{equation}
 \operatorname{Jac} P=\frac{J(v,4)}{J(v,\tau)}
                     |\langle V-U,\nu\rangle_{g(\tau)}|.
 \label{eq:flux-jacobian}
\end{equation}
Consequently, for a measurable subset $H$ of the track contained
in the normal region,
\begin{equation}
 \operatorname{Vol}_{g(4)}(P(H))
 \le\int_H\frac{J(v,4)}{J(v,\tau)}
              |\langle V-U,\nu\rangle|\,\mathrm d\sigma_\tau\,\mathrm d\tau.
 \label{eq:flux-inequality}
\end{equation}
<!-- end lemma-7 -->
<!-- proof-14: body_md -->
Work in oriented local charts and let $\omega_\tau$ be the spatial
volume form, extended to spacetime by annihilating $\partial_\tau$.
Put
\[
 \Omega=\omega_\tau\wedge\mathrm d\tau,\qquad
 X=\partial_\tau+V,\qquad
 \alpha=\iota_{X}\Omega.
\]
Since $P\circ\Xi(v,\tau)=E_4(v)$, we have
$\mathrm dP(X)=0$. Moreover,
$\mathrm d\Xi(\partial_\tau)=X$ and
$\alpha(X,\cdot)=0$. Choose a fixed orthonormal basis
$e_a$ in the initial-velocity space. Evaluating on the spatial frame
$Y_a=\mathrm dE_\tau(e_a)$ therefore gives the full pullback identity
\[
 \Xi^*\alpha=\pm J(v,\tau)\,\mathrm d v,
 \qquad
 \Xi^*(P^*\omega_4)=E_4^*\omega_4=J(v,4)\,\mathrm d v.
\]
Here local orientations can be chosen consistently; the sign of the
first identity will disappear on taking absolute densities.
Since $\Xi$ is a diffeomorphism, comparison of these identities yields
\begin{equation}
 P^*\omega_4
   =\pm\frac{J(v,4)}{J(v,\tau)}
       \iota_{\partial_\tau+V}
                   (\omega_\tau\wedge\mathrm d\tau)
 \label{eq:flux-form-identity}
\end{equation}
on the normal region, with $v$ understood through $\Xi^{-1}$.

At a point of the moving track, choose a spatial orthonormal
tangential frame $e_1,\ldots,e_{n-1}$ and let
$T=\partial_\tau+U$ be its time tangent. Since
$X=T+(V-U)$, alternation gives
\begin{align*}
 |\alpha(e_1,\ldots,e_{n-1},T)|
 &=|\Omega(V-U,e_1,\ldots,e_{n-1},T)|\\
 &=|\omega_\tau(V-U,e_1,\ldots,e_{n-1})|\\
 &=|\langle V-U,\nu\rangle_{g(\tau)}|.
\end{align*}
The chosen frame has unit density for
$\,\mathrm d\sigma_\tau\,\mathrm d\tau$. Restricting
\eqref{eq:flux-form-identity} to it proves
\eqref{eq:flux-jacobian}. Only the normal component of $U$ matters,
so the formula does not depend on the tangential parametrization
of the moving hypersurface.

Apply the area formula in relatively compact coordinate patches.
If $N(P,H,y)$ denotes the number of preimages of $y$ in $H$, it gives
\[
 \int_H\operatorname{Jac} P\,\,\mathrm d\sigma_\tau\,\mathrm d\tau
   =\int_M N(P,H,y)\mathrm dV_{g(4)}(y)
   \ge\operatorname{Vol}_{g(4)}(P(H)).
\]
Summing over a countable coordinate decomposition proves
\eqref{eq:flux-inequality}.

the center of $B$, with each auxiliary ball chosen smaller than its
distance to $X\setminus B$. This is the usual John-domain proof
for a metric ball in a geodesic space. The relative volume bound in
\eqref{eq:local-volume-comparison}, with exponent $n$, gives the
Sobolev--Poincar\'e estimate for $u-u_{B,\mathfrak m}$ on the
same chains. Adding the mean gives \eqref{eq:local-sobolev}, as in the local
Sobolev theory of Saloff-Coste~\cite{SaloffCoste1992}.
The preceding Sobolev identification extends the estimates to the stated function class.
Finally, \eqref{eq:local-potential-control} bounds the oscillation
of $f$ on $B_d(p,L)$. Weighted and unweighted averages are therefore
comparable with constants depending only on $n,L$, giving the
unweighted statements.

\label{lem:entropy-density}
For every $z\in X$ and every tangent cone $C$ at $z$,
\begin{equation}
 N_z(0)=\log\vartheta(C).
 \label{eq:entropy-density}
\end{equation}
In particular, the right-hand side is independent of the choice
of tangent cone. There is $\epsilon_n>0$ such that
\begin{equation}
 z\inS\quad\Longrightarrow\quad
 \vartheta(C)\le e^{-\epsilon_n}<1.
 \label{eq:singular-density-gap}
\end{equation}
A point of $X$ is regular if and only if it has a Euclidean tangent
cone.
<!-- end lemma-11 -->
<!-- proof-23: body_md -->
The static tangent soliton has potential $d_C(o,\cdot)^2/(4\tau)$
up to its normalization constant. Its vertex conjugate heat kernel is
\[
 K_C(o,w;\tau)=\vartheta(C)^{-1}(4\pi\tau)^{-n/2}
               \exp\left(-\frac{d_C(o,w)^2}{4\tau}\right).
\]
Indeed, cone polar integration gives
\[
 \int_C e^{-d_C(o,w)^2/(4\tau)}\,\mathrm dH^n(w)
       =\vartheta(C)(4\pi\tau)^{n/2},
 \qquad
 \int_C\frac{d_C(o,w)^2}{4\tau}K_C\,\mathrm dH^n(w)=\frac n2.
\]
Thus its pointed Nash entropy is $\log\vartheta(C)$.
Apply this calculation to a sequence of parabolic dilations whose
spatial limit is $C$. Entropy convergence for these dilations gives
\eqref{eq:entropy-density}; see the entropy convergence and tangent
flow results in \cite[Theorems~2.10--2.11]{BamlerStructure}.

At a regular point the small-scale entropy is zero. Conversely,
if $N_z(0)>-\epsilon_n$, then at a sufficiently small
positive scale the entropy satisfies the regularity threshold in
\cite[Proposition~7.7]{FangLiStructure}, after decreasing
$\epsilon_n$ if necessary. Hence $z$ is regular. Its contrapositive
and \eqref{eq:entropy-density} give
\eqref{eq:singular-density-gap}. A Euclidean tangent cone has
$\vartheta=1$, so the same argument proves the last assertion.

<!-- lemma-12: body_md -->
\label{lem:global-bochner}
Let $v$ be bounded and Lipschitz on $X$, and set $h=P_sv$ for
$s>0$. Then $|\nabla^2h|\in L^2_{\mathrm{loc}}(X,\mathfrak m)$
and $|\nabla h|^2\in W^{1,2}_{\mathrm{loc}}(X,d,\mathfrak m)$.
Moreover,
\begin{equation}
 (\partial_s-\Delta_f)|\nabla h|^2
       \le-2|\nabla^2h|^2-|\nabla h|^2
 \label{eq:weighted-Bochner}
\end{equation}
in the distributional sense on $X\times(0,\infty)$. The Hessian is computed on
$R$ and extended by zero across $S$.
<!-- end lemma-12 -->
<!-- proof-24: body_md -->
Write $q=|\nabla h|^2$. Equation~\eqref{eq:heat-gradient} gives
$|\nabla h|\le L:=\|\nabla v\|_\infty$. Since
$\mathfrak m(X)=1$, we have $v\in L^2(X,\mathfrak m)$.
Spectral calculus for the nonnegative self-adjoint operator
$-\Delta_f$ gives
\[
 \Delta_fh\in W^{1,2}(X,d,\mathfrak m),\qquad
 h\in C^1\bigl((0,\infty);W^{1,2}(X,d,\mathfrak m)\bigr),
 \qquad \partial_sh=\Delta_fh,
\]
with the relevant norms bounded on compact positive-time intervals.
On $R$, the smooth weighted Bochner identity is
\[
 \frac12\Delta_fq
       =|\nabla^2h|^2+\frac12q
                     +\langle\nabla h,\nabla\Delta_fh\rangle.
\]

Choose a compactly supported Lipschitz function $0\le\chi\le1$
on $X$ and put $\zeta_\rho=\chi\eta_\rho$, with the singular
cutoffs restricted to the reference slice. Its support is compact
in $R$, so integration by parts and
$|\nabla q|\le2L|\nabla^2h|$ give
\[
 \begin{split}
 \int_{R}\zeta_\rho^2|\nabla^2h|^2\,\mathrm d\mathfrak m
 &\le 2L\int_{R}\zeta_\rho|\nabla\zeta_\rho|
                    |\nabla^2h|\,\mathrm d\mathfrak m
       +\int_{R}\zeta_\rho^2
                    |\nabla h|\,|\nabla\Delta_fh|\,\mathrm d\mathfrak m\\
 &\le\frac12\int_{R}\zeta_\rho^2|\nabla^2h|^2
                    \,\mathrm d\mathfrak m
       +2L^2\int_{R}|\nabla\zeta_\rho|^2\,\mathrm d\mathfrak m
       +\|\nabla h\|_2\|\nabla\Delta_fh\|_2.
 \end{split}
\]
Here the norms are in $L^2(X,\mathfrak m)$. By
\eqref{eq:zero-capacity}, the cutoff energy on the right is bounded
independently of $\rho$. Absorption and Fatou's lemma prove
$|\nabla^2h|\in L^2_{\mathrm{loc}}(X,\mathfrak m)$ by taking
$\chi=1$ on any prescribed compact set. These bounds are uniform
on compact positive-time intervals. The bound for $\nabla q$
and the Sobolev identification in
Subsection~\ref{subsec:local-functional} then give
$q\in W^{1,2}_{\mathrm{loc}}(X,d,\mathfrak m)$.

Now test the smooth identity with $\phi\eta_\rho$, where
$\phi$ is compactly supported and Lipschitz on $X$. The only
additional gradient term vanishes, since
\[
 \left|\int_{R}
       \phi\langle\nabla q,\nabla\eta_\rho\rangle
                         \,\mathrm d\mathfrak m\right|
 \le\|\phi\|_\infty
       \|\nabla q\|_{L^2(\operatorname{supp}\phi)}
       \|\nabla\eta_\rho\|_{L^2(\operatorname{supp}\phi)}
 \longrightarrow0.
\]
All other terms pass to the limit by the established integrability.
Thus
\[
 -\frac12\int_X\langle\nabla q,\nabla\phi\rangle\,\mathrm d\mathfrak m
   =\int_X\phi\bigl(|\nabla^2h|^2+\frac12q
                  +\langle\nabla h,\nabla\Delta_fh\rangle\bigr)
                  \,\mathrm d\mathfrak m.
\]
Finally, the $W^{1,2}$ differentiability of $h$ gives
$\partial_sq=2\langle\nabla h,\nabla\Delta_fh\rangle$ in $L^1$.
Integrating against compactly supported time tests proves
\eqref{eq:weighted-Bochner}, in fact with equality.

$r_0=r_0(n,L)\in(0,\min\{1,\sqrt{s_0/2}\}]$ such that

\begin{equation}
 0\le\chi\le1,\qquad
 r|\nabla\chi|+r^2|\Delta_f\chi|\le C.
 \label{eq:heat-cutoff}
\end{equation}
The Laplacian bound holds for the global generator.

 |h_s^\pm-d^\pm|\le Cs+e,

 \label{eq:heat-pointwise}
\end{equation}
For $t\in[\delta,1-\delta]$,
\begin{equation}
 \fint_{B_d(\gamma(t),r)}e\,\mathrm d\mathfrak m\le Cr^2.
 \label{eq:mean-excess}
\end{equation}
\item There is a single $s_r\in[r^2/2,2r^2]$ for which
\begin{equation}
 \int_\delta^{1-\delta}
 \fint_{B_d(\gamma(t),5r)}
 \bigl(|\nabla^2h_{s_r}^-|^2+|\nabla^2h_{s_r}^+|^2\bigr)
             \,\mathrm d\mathfrak m\,\mathrm d t\le C.
 \label{eq:integrated-hessian}
\end{equation}

\end{equation}

\end{enumerate}

<!-- proposition-7: body_md -->
\label{prop:small-ball-comparison}
Fix $D,L,\ell_0>0$ and $0<\delta<1/2$. For every $\epsilon>0$
there are $a_0,r_0>0$, depending only on these parameters, $n$,
and $\epsilon$, such that the following holds. If
$\gamma:[0,\ell]\toR$ is a unit-speed minimizing geodesic
contained in $B_d(p,D)$, with $\ell_0\le\ell\le L$, then
\begin{equation}
 d_{\mathrm{GH}}\bigl(\overline B_d(\gamma(s),r),
             \overline B_d(\gamma(t),r)\bigr)\le\epsilon r
 \label{eq:small-ball-comparison}
\end{equation}
whenever $s,t\in[\delta\ell,(1-\delta)\ell]$,
$|s-t|\le a_0\ell$, and $0<r\le r_0\ell$.
In particular, the constants do not depend on the smallest
regularity scale along $\gamma$.
<!-- end proposition-7 -->
<!-- proof-27: body_md -->
Normalize $\ell=1$ and work on a fixed interior interval, with a
slightly larger interval available for the estimates. We follow
\cite[Lemmas~4.1,~4.2,~4.4 and Theorem~4.5]{HuangLiWang}.
By Theorem~\ref{thm:weak}, the inward radial flows toward $q_-$
and $q_+$ are defined along regular minimizing segments for almost
every initial point. Their vector fields are $-\nabla d^-$ and
$+\nabla d^+$, respectively. Weighted polar comparison gives,
for either flow $\Phi_v$ on an annulus away from its endpoint,
\begin{equation}
 \mathfrak m(\Phi_v A)\ge e^{-Cv}\mathfrak m(A),\qquad
 (\Phi_v)_\#(\mathfrak m|_A)\le e^{Cv}\mathfrak m.
 \label{eq:radial-compression}
\end{equation}
For $t\in(0,1)$, concatenate any ambient minimizing geodesic from
$q_-$ to $\gamma(t)$ with a short continuation of $\gamma$ beyond
$t$; the resulting curve is still minimizing.
At the regular junction it has no corner, so backward uniqueness
for the smooth geodesic equation along $\gamma$ identifies the
original curve with $\gamma|_{[0,t]}$.
Properness then implies that all minimizing geodesics from $q_-$
to nearby endpoints remain in a regular neighborhood of this compact
segment. Since $\gamma$ continues minimizing beyond $t$, there is
no conjugate point at $\gamma(t)$; the local inverse of the
exponential map therefore shows that $d^-$ is smooth near
$\gamma(t)$. The reversed argument applies to $q_+$, and compactness
gives a common smooth neighborhood along any closed interior
interval.

Now let $\Phi_v$ denote the inward flow toward $q_-$.
The Riccati identity, endpoint Laplacian comparison, and the Jacobi
equation give the usual core estimate
$d(\Phi_vz,\gamma(b-v))\le e^{C\sqrt v}d(z,\gamma(b))$.
The core radius may depend on $\gamma$, but the constant does not;
see the argument of \cite[Lemma~4.1]{HuangLiWang}.
The corresponding estimate toward $q_+$ follows by reversing
$\gamma$.

Stop the flows when they leave doubled moving balls. Under the
temporary half-volume bounds for the surviving points,
Lemma~\ref{lem:heat-approximation}, the segment inequality, and
\eqref{eq:radial-compression} control the averaged distance
distortion. Chebyshev's inequality gives a good set $T$ and partner
sets $T(x)$, whose relative complements have measure at most
$C\eta$, with pairwise distortion at most
$C\eta^{-2}r\sqrt v$ before exit. To remove the stopping assumption,
choose $0<\alpha\ll1$, $0<\xi\ll\alpha$, and
$0<\eta\ll\xi^n$. Connect good points through the finite chain of
radii $r,\xi r,\ldots,\xi^Jr$ ending inside the smooth core.
The distortion estimate prevents exit from the moving
$(1+\alpha)r$ balls for a time depending only on the uniform
parameters. Applying the opposite endpoint flow and
\eqref{eq:thin-annulus} strictly improves the half-volume bounds,
so this time is independent of $J$ and of the core radius.
This is the stopping argument of
\cite[Lemmas~4.2 and~4.4]{HuangLiWang}.

Relative volume comparison makes $T$ $C\eta^{1/n}r$-dense in the
initial ball. Choosing common partners extends the distortion
estimate to all pairs in $T$. The opposite endpoint flow and
\eqref{eq:radial-compression} also show that the image misses at
most a $C(\alpha+\eta+v)$ fraction of the target ball's measure.
Consequently, it is $C(\alpha+\eta+v)^{1/n}r$-dense there.
Moving image points outside the target ball into it by at most
$\alpha r$ gives
\[
 \begin{split}
 &r^{-1}d_{\mathrm{GH}}\bigl(\overline B_d(\gamma(b),r),
                    \overline B_d(\gamma(b-v),r)\bigr)\\
 &\qquad\le C\bigl(\eta^{-2}\sqrt v+\eta^{1/n}
                    +(\alpha+\eta+v)^{1/n}+\alpha\bigr).
 \end{split}
\]
Choose $\alpha$ small, then $\xi$ and $\eta$ as above, and finally
$v$ sufficiently small. This proves
\eqref{eq:small-ball-comparison} with the stated uniformity.

\label{lem:regular-limit-geodesic}
For every $x\inR$ and $y\in X$, there is a minimizing
geodesic $\sigma:[0,1]\to X$ from $x$ to $y$ with
$\sigma([0,1))\subsetR$.
<!-- end lemma-15 -->
<!-- proof-28: body_md -->
The statement is immediate for $y=x$. Otherwise, choose good
endpoints $y_j\to y$ from Theorem~\ref{thm:weak} and regular
minimizing geodesics $\gamma_j$ from $x$ to $y_j$, parametrized at constant
speed on $[0,1]$. The good endpoints are dense because their
complement has zero measure and $\mathfrak m$ has full support.
Properness gives a uniformly convergent subsequence with a minimizing
limit $\sigma$. The lengths converge to $d(x,y)>0$, and the curves
lie in a common bounded region.

Let $J=\{t\in(0,1):\sigma(t)\inR\}$. This set is open
and nonempty, since $x$ is regular. We show that it is relatively
closed in $(0,1)$. Suppose $t_i\in J$ and $t_i\to t\in(0,1)$.
For any fixed $\epsilon>0$, Proposition~\ref{prop:small-ball-comparison}
gives uniform $a_0,r_0>0$ on an interior interval containing these
times. For $i$ large, pass its estimate from $\gamma_j$ to $\sigma$:
\begin{equation}
 d_{\mathrm{GH}}\bigl(\overline B_d(\sigma(t_i),r),
             \overline B_d(\sigma(t),r)\bigr)\le\epsilon r
 \quad(0<r<r_0).
 \label{eq:limit-ball-comparison}
\end{equation}
This passage is at a fixed radius. In a proper length space,
closed balls of the same radius converge in Hausdorff distance
when their centers converge, by moving points along minimizing
segments if necessary.

For this fixed $i$, smoothness at $\sigma(t_i)$ implies
\[
 \lim_{r\searrow0}r^{-1}d_{\mathrm{GH}}
 \bigl(\overline B_d(\sigma(t_i),r),
                    \overline B_{\mathbb R^n}(0,r)\bigr)=0.
\]
The triangle inequality and \eqref{eq:limit-ball-comparison} give
an upper limit at most $\epsilon$ for the analogous expression at
$\sigma(t)$. Since $\epsilon$ is arbitrary, every tangent cone at
$\sigma(t)$ has a unit ball isometric, as a metric space, to the
Euclidean unit ball. Its $n$-dimensional volume is therefore
$\omega_n$. Lemma~\ref{lem:entropy-density} gives
$N_{\sigma(t)}(0)=0$, hence $\sigma(t)\inR$.
This proves closedness. Consequently $J=(0,1)$, as required.
