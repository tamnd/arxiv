   \rho_t\in C([0,T];H^1),\\
 &u,\theta-\bar\theta\in C([0,T];H^2\cap H_0^1)
    \cap L^2(0,T;H^3),\\
 &\sqrt\rho\,\dot u,\sqrt\rho\,\dot\theta
      \in L^\infty(0,T;L^2),

A global strong solution is a strong solution on $[0,T]$ for every finite
$T>0$.

\label{thm:main}
Let $R,\mu,\kappa>0$ and $K_0\geq1$.  Fix mass bounds
$0<m_*<m^*$, and assume
\begin{equation}\label{eq:exponents}
 \alpha>2,\qquad \beta>2\alpha+4.
\end{equation}
There exists
\[
 \Theta_0=\Theta_0(R,\mu,\kappa,\alpha,\beta,K_0,m_*,m^*)\geq1
\]
with the following property.  For every boundary temperature
$\bar\theta\geq\Theta_0$, prescribe the initial trace
\[
 (\rho,u,\theta)(x,0)=(\rho_0,u_0,\theta_0)(x),
 \qquad x\in(0,1).
\]
The complete assumptions on the initial triple
$(\rho_0,u_0,\theta_0)$ and its compatibility data are the following; no
additional initial-data hypothesis is imposed elsewhere.
\begin{enumerate}[label=\textup{(H\arabic*)},leftmargin=3.2em,itemsep=4pt]
\item \emph{Initial density and total mass.}  The initial density obeys
\begin{equation}\label{eq:data-density}
 \rho_0\in H^2,\qquad \rho_0\geq0,\qquad
 m_*\leq m:=\int_0^1\rho_0\mathop{}\!\mathrm d x\leq m^*,\qquad
 \lVert\rho_0\rVert_{H^2}\leq K_0.
\end{equation}
No positive lower bound is imposed on $\rho_0$; beyond the compatibility
condition \textup{(H3)}, no smallness or measure restriction is placed on its
zero set.

\item \emph{Velocity and temperature data.}  The remaining data obey
\begin{equation}\label{eq:data-dynamic}
 u_0\in H^2\cap H_0^1,\qquad
 \theta_0-\bar\theta\in H^2\cap H_0^1,\qquad
 \lVertu_0\rVert_{H^2}^2+\lVert\theta_0-\bar\theta\rVert_{H^2}^2\leq K_0,
 \qquad
 \frac34\bar\theta\leq\theta_0\leq\frac54\bar\theta.
\end{equation}
Thus, in particular,
\[
 u_0(0)=u_0(1)=0,\qquad
 \theta_0(0)=\theta_0(1)=\bar\theta.
\]

\item \emph{Compatibility conditions.}  There exist
$g_1,g_2\in L^2(0,1)$, taken to vanish almost everywhere on
$\{\rho_0=0\}$, such that the relations
\begin{equation}\label{eq:compatibility}
\left\{
\begin{aligned}
 &(\mu\theta_0^\alpha u_{0x})_x-(R\rho_0\theta_0)_x
       =\sqrt{\rho_0}\,g_1,\\
 &(\kappa\theta_0^\beta\theta_{0x})_x
       +\mu\theta_0^\alpha u_{0x}^2
       -R\rho_0\theta_0u_{0x}
       =\sqrt{\rho_0}\,g_2
\end{aligned}\right.
\end{equation}
hold almost everywhere.  In particular, the second relation includes the
pressure-work term $-R\rho_0\theta_0u_{0x}$.

\item \emph{Natural compatibility scale.}  The compatibility data satisfy
\begin{equation}\label{eq:natural-data}
 \bar\theta^{-2\alpha}\lVertg_1\rVert_{L^2}^2
 +\bar\theta^{-2\beta}\lVertg_2\rVert_{L^2}^2\leq K_0.
\end{equation}
The number $K_0$ is arbitrary and is not assumed small.  The powers in
\eqref{eq:natural-data} merely normalize the principal parts of
\eqref{eq:compatibility}: for fixed $u_0$ and $\theta_0-\bar\theta$, their natural
sizes are $g_1=O(\bar\theta^\alpha)$ and $g_2=O(\bar\theta^\beta)$.
\end{enumerate}
Then problem \eqref{eq:NSF}--\eqref{eq:initial} admits at least one global
strong solution in the sense of Definition~\ref{def:strong}.
\smallskip

\noindent\emph{Uniform pointwise bounds.}
There are constants $M$ and $C$ depending only on
$R,\mu,\kappa,\alpha,\beta$, $K_0$, $m_*$, and $m^*$.  They are independent
of time, $\bar\theta$, and any positive lower bound for $\rho_0$, and satisfy
\begin{equation}\label{eq:pointwise-main}
 0\leq\rho(x,t)\leq M,\qquad
 \frac12\bar\theta\leq\theta(x,t)\leq\frac32\bar\theta
 \quad (x\in[0,1],\ t\geq0),
\end{equation}
\smallskip

Put
\begin{equation}\label{eq:sigma-main}
 \tau_{\bar\theta}=\bar\theta^{1-\beta},\qquad
 \sigma_{\bar\theta}(t)=\min\{1,t+\tau_{\bar\theta}\}.
\end{equation}

\noindent\emph{Initial-layer and post-layer energy estimates.}
\nopagebreak[4]
\begin{align}\label{eq:uniform-main}
 &\operatorname*{ess\,sup}_{0\leq t\leq1}\Bigl[
  \sigma_{\bar\theta}\bar\theta^{\alpha-1}\lVertu_x(t)\rVert_{L^2}^2
 +\sigma_{\bar\theta}\bar\theta^{\beta-2}\lVert\theta_x(t)\rVert_{L^2}^2
 +\sigma_{\bar\theta}^2\bar\theta^{-1}
       \lVert\sqrt{\rho}\,\dot u(t)\rVert_{L^2}^2
 +\sigma_{\bar\theta}^2\bar\theta^{-2}
       \lVert\sqrt{\rho}\,\dot\theta(t)\rVert_{L^2}^2
 \Bigr]\nonumber\\
 &\quad+\int_0^1\Bigl[
  \bar\theta^{\alpha-1}\lVertu_x\rVert_{L^2}^2
 +\bar\theta^{\beta-2}\lVert\theta_x\rVert_{L^2}^2
 +\sigma_{\bar\theta}\bar\theta^{-1}
       \lVert\sqrt{\rho}\,\dot u\rVert_{L^2}^2
 +\sigma_{\bar\theta}\bar\theta^{-2}
       \lVert\sqrt{\rho}\,\dot\theta\rVert_{L^2}^2\nonumber\\[-1pt]

 +\sigma_{\bar\theta}^2\bar\theta^{\beta-2}\lVert\dot\theta_x\rVert_{L^2}^2
 \Bigr]\mathop{}\!\mathrm d t\leq C.
\end{align}
At the endpoint of the initial layer, $\sigma_{\bar\theta}(1)=1$; the post-layer

 \operatorname*{ess\,sup}_{t\geq1}\mathcal A(t)
 +\int_1^\infty\mathcal D(t)\mathop{}\!\mathrm d t\leq C,
\end{equation}
where $\mathcal A$ and $\mathcal D$ are defined in
\eqref{eq:A}--\eqref{eq:D} below.
\smallskip

\noindent\emph{Finite-time higher-order estimates.}
For each $T<\infty$ there is a constant $C_T$, depending only on
$T$, $\bar\theta$, $R,\mu,\kappa,\alpha,\beta$, $K_0$, $m_*$, and $m^*$.
It may grow with $T$, but is independent of any positive lower bound for
$\rho_0$, and satisfies
\begin{align}\label{eq:finite-high}
 &\sup_{0\leq t\leq T}
   \bigl(\lVert\rho(t)\rVert_{H^2}+\lVertu(t)\rVert_{H^2}
        +\lVert\theta(t)-\bar\theta\rVert_{H^2}\bigr)+\int_0^T\bigl(\lVertu\rVert_{H^3}^2+\lVert\theta-\bar\theta\rVert_{H^3}^2
       +\lVert\dot u\rVert_{H^1}^2+\lVert\dot\theta\rVert_{H^1}^2\bigr)\mathop{}\!\mathrm d t
 \leq C_T.
\end{align}
\smallskip

\noindent\emph{Persistence of vacuum.}
For every $a\in[0,1]$, let $X(a,t)$ be the particle path defined by
\[
 X_t(a,t)=u(X(a,t),t),\qquad X(a,0)=a.
\]
If $\rho_0(a)=0$, then $\rho(X(a,t),t)=0$ for every finite $t$.

\label{thm:relaxation}
Let $(\rho,u,\theta)$ be the global strong solution furnished by
Theorem~\ref{thm:main}, and let
$m=\int_0^1\rho_0\mathop{}\!\mathrm d x$.  There exist constants $c_{\rm d},C_{\rm d}>0$,
depending only on $R,\mu,\kappa,\alpha,\beta,K_0,m_*$, and $m^*$, but
independent of $\bar\theta$ and of any positive lower bound for the density, such
that, with
\begin{equation}\label{eq:decay-rate}
 \lambda_{\rm d}=c_{\rm d}\bar\theta^{1-\alpha},
\end{equation}
one has, for almost every $t\geq1$,
\begin{align}
 &\lVert\rho(t)-m\rVert_{L^2}^2
 +\bar\theta^{\alpha-2}\lVertu(t)\rVert_{H^1}^2
 +\bar\theta^{\beta-2}\lVert\theta(t)-\bar\theta\rVert_{H^1}^2
  +\bar\theta^{-2}\left(
   \lVert\sqrt\rho\,\dot u(t)\rVert_{L^2}^2
  +\lVert\sqrt\rho\,\dot\theta(t)\rVert_{L^2}^2\right)
 \leq C_{\rm d}e^{-\lambda_{\rm d}(t-1)}.
 \label{eq:low-order-decay}
\end{align}
The first three terms in \eqref{eq:low-order-decay} satisfy the same estimate
for every $t\geq1$.  Moreover, for every $t\geq1$,
\begin{align}
 &\int_t^\infty\Bigl[
  \bar\theta^{\alpha-1}\lVertu_x\rVert_{L^2}^2
 +\bar\theta^{\beta-2}\lVert\theta_x\rVert_{L^2}^2
 +\bar\theta^{-2}\Bigl(
   \lVert\sqrt{\rho}\,\dot u\rVert_{L^2}^2
  +\lVert\sqrt{\rho}\,\dot\theta\rVert_{L^2}^2\Bigr)
 \nonumber\\
 &\qquad +\bar\theta^{\alpha-2}\lVert\dot u_x\rVert_{L^2}^2
 +\bar\theta^{\beta-2}\lVert\dot\theta_x\rVert_{L^2}^2
 \Bigr]\mathop{}\!\mathrm d s
 \leq C_{\rm d}e^{-\lambda_{\rm d}(t-1)}.
 \label{eq:decay-tail}
\end{align}
Consequently, for every $1\leq p<\infty$ there are constants $C_p,c_p>0$
with
\begin{equation}\label{eq:density-Lp-decay}
 \lVert\rho(t)-m\rVert_{L^p}
 \leq C_p\exp\!\left(-c_p\bar\theta^{1-\alpha}(t-1)\right)
 \qquad(t\geq1),\qquad
 c_p=\frac{c_{\rm d}}{\max\{2,p\}}.
\end{equation}
If, in addition, $\rho_0(a_0)=0$ for some $a_0\in[0,1]$, set
$c_{\rm v}:=m^2/(2C_{\rm d}^{1/2})$.  Then, for every $t\geq1$,
\begin{equation}\label{eq:gradient-growth}
 \lVert\rho(t)-m\rVert_{L^\infty}\geq m,
 \qquad
 \lVert\rho_x(t)\rVert_{L^2}
 \geq\max\left\{\sqrt{3}\,m,
                 c_{\rm v}e^{\lambda_{\rm d}(t-1)/2}\right\},
\end{equation}
and, for every $T>1$,
\begin{equation}\label{eq:gradient-growth-integrated}
 \int_1^T\lVert\rho_x(t)\rVert_{L^2}^2\mathop{}\!\mathrm d t
 \geq\max\left\{3m^2(T-1),
       \frac{c_{\rm v}^2}{\lambda_{\rm d}}
       \left(e^{\lambda_{\rm d}(T-1)}-1\right)\right\}.
\end{equation}
In particular,
$\lVert\rho(t)-m\rVert_{H^1}\geq
c_{\rm v}e^{\lambda_{\rm d}(t-1)/2}$ for $t\geq1$.
Thus, in the genuine-vacuum case, density relaxation cannot hold in
$L^\infty$, $H^1$, or any stronger topology that controls point values.

<!-- remark-2: body_md -->
\label{rem:scope}
Theorem~\ref{thm:main} permits arbitrarily large compatible data: $K_0$ is
fixed at an arbitrary value before the threshold $\Theta_0(K_0)$ is chosen.
It does not assert a single temperature threshold uniform over data of
unbounded size.  The normalization \eqref{eq:natural-data} is dictated by the
principal differential operators and is not a perturbative smallness
condition.  Substantially enlarging the exponent range would require a
different commutator closure.
Theorem~\ref{thm:relaxation} gives exponential relaxation precisely below
the density-gradient level.  If $\rho_0$ vanishes somewhere, the density
cannot converge to its positive mean in $L^\infty$ or in any topology that
controls its point values or spatial derivative.  The approximation argument
proves existence; uniqueness in the vacuum class is not addressed here.

<!-- remark-3: body_md -->
\label{rem:proof-mechanisms}
Table~\ref{tab:comparison} focuses on the one-dimensional global theory.
The multidimensional local-vacuum results and the vacuum free-boundary
results discussed in the introduction are not repeated there, because their
geometry and solution classes are different.  The comparison is intended to
record structural distinctions, not to claim optimality of the exponent
range.  Within the precise regime of
Theorems~\ref{thm:main}--\ref{thm:relaxation}, the main analytical
difficulties and the mechanisms used here to resolve them are as follows.
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=2.2em,itemsep=5pt]
\item \emph{Coercivity compatible with vacuum.}
The density may vanish, so neither the momentum equation nor the temperature
equation can be divided by $\rho$.  The relative entropy
\eqref{eq:relative-energy}, centered at the conserved mean $m$, remains
finite at vacuum.  Its thermal part is density weighted and therefore does
not control the temperature on the vacuum set; the missing unweighted
control is recovered from the fixed boundary temperature and the Dirichlet
Poincar\'e inequality.  Hence the basic energy estimate requires no positive
lower density bound.

\item \emph{Pointwise bounds without division by the density.}
For the physical effective viscous flux
\[
 F=\mu\theta^\alpha u_x-R\rho\theta,
 \qquad F_x=\rho\dot u,
\]
the quantity $\Pi$ in \eqref{eq:Pi} fixes the spatial constant through a
$\theta^{-\alpha}$-weighted mean.  This yields the anchor
\eqref{eq:F-anchor} in terms of $\sqrt\rho\,\dot u$, without invoking
$\rho^{-1}$.  A Zlotnik argument for $\log\rho$ along positive-density
particle paths then gives the density upper bound uniformly in the
approximation parameter.  The heat-flux identity \eqref{eq:heat-flux}
supplies the derivative control needed in the thermal commutators, while the
temperature bootstrap is improved using the scaled $L^2$ estimate of
$\theta_x$ and the fixed boundary trace.

\item \emph{Initial-layer smoothing and post-layer closure.}
The time scale $\tau_{\bar\theta}=\bar\theta^{1-\beta}$ and the shifted weight
$\sigma_{\bar\theta}(t)=\min\{1,t+\tau_{\bar\theta}\}$ capture the rapid thermal initial
layer without requiring $u_0$ or $\theta_0-\bar\theta$ to shrink as
$\bar\theta\to\infty$.  Multiplying the material momentum identity by $\bar\theta$ before
adding the material temperature identity cancels the leading mixed term at
equilibrium.  The resulting weighted estimate places the solution at the
normalized post-layer scale at $t=1$.  From that time onward, the hierarchy
\eqref{eq:A}--\eqref{eq:D} and Lemma~\ref{lem:bookkeeping} close the
variable-coefficient commutators under
$\alpha>2$ and $\beta>2\alpha+4$.  This is a sufficient exponent range, not
a claim of sharpness.

\item \emph{Passage from positive density to genuine vacuum.}
The local problem is solved for $\rho_0^\delta=\rho_0+\delta$.  The modified
compatibility residuals satisfy the uniform naturally scaled bounds
\eqref{eq:gdelta}--\eqref{eq:natural-data-delta}.  The two-stage estimates are
independent of $\inf\rho_0^\delta$, while the density derivatives are
propagated only on finite time intervals by the transport estimates of
Section~\ref{sec:higher}.  Compactness then produces a strong vacuum-limit
solution, and the exact flow identity \eqref{eq:flow-formula} transports the
initial zero set.  No smallness or measure condition is imposed on that set,
although uniqueness of the vacuum-limit solution is not asserted.

\item \emph{Relaxation below the density-gradient level.}
The basic entropy contains no direct density dissipation.
Lemma~\ref{lem:decay-coercivity} recovers the missing $L^2$ control of
$\rho-m$ from the effective flux without using $\rho_x$.  Adding a scaled
relative entropy to the differentiated energy gives the strict Lyapunov
inequality in Proposition~\ref{prop:strict-lyapunov} and the rate
$\lambda_{\rm d}=c_{\rm d}\bar\theta^{1-\alpha}$.  This proves finite-$L^p$
density relaxation, $H^1$ relaxation of $u$ and $\theta-\bar\theta$, and weighted
$L^2$ decay of the material derivatives.  When a vacuum point is present,
the transported zero rules out density convergence in $L^\infty$ and $H^1$;
together with the $L^2$ decay it yields the quantitative vacuum-layer
sharpening in \eqref{eq:gradient-growth}--
\eqref{eq:gradient-growth-integrated}.
\end{enumerate}
Thus the contribution is the conjunction of global vacuum-compatible
large-data existence, an initial-layer estimate uniform in the density lower
bound, and low-order exponential relaxation after that layer.  The exponent
range is not claimed optimal, and uniqueness in the vacuum class remains
open.

\label{lem:entropy}
Every positive smooth solution of \eqref{eq:NSF}--\eqref{eq:boundary}
satisfies
\begin{align}\label{eq:entropy-identity}
 \frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\mathcal E(t)
 &+\int_0^1\left(
 \mu\bar\theta\theta^{\alpha-1}u_x^2
 +\kappa\bar\theta\theta^{\beta-2}\theta_x^2
 \right)\mathop{}\!\mathrm d x=0.
\end{align}
The same identity, with equality replaced by the corresponding integrated
inequality, holds for the vacuum limit constructed below.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
The continuity equation implies
\[
 \frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\int_0^1\frac12\rho u^2\mathop{}\!\mathrm d x
 =\int_0^1\rho\dot u\,u\mathop{}\!\mathrm d x.
\]
Testing the momentum equation by $u$, using $u=0$ at the endpoints, and
integrating the temperature equation give, respectively,
\begin{align*}
& \frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\int_0^1\frac12\rho u^2\mathop{}\!\mathrm d x
 +\int_0^1\mu\theta^\alpha u_x^2\mathop{}\!\mathrm d x
 =\int_0^1R\rho\theta u_x\mathop{}\!\mathrm d x,\\
 &\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\int_0^1\rho\theta\mathop{}\!\mathrm d x
 +\int_0^1R\rho\theta u_x\mathop{}\!\mathrm d x
 =\int_0^1\mu\theta^\alpha u_x^2\mathop{}\!\mathrm d x
   +\left[\kappa\theta^\beta\theta_x\right]_{0}^{1}.
\end{align*}
Their sum is the total-energy balance
\begin{equation}\label{eq:total-energy-balance}
 \frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\int_0^1
 \left(\frac12\rho u^2+\rho\theta\right)\mathop{}\!\mathrm d x
 =\left[\kappa\theta^\beta\theta_x\right]_{0}^{1}.
\end{equation}
On the other hand, division of the temperature equation by $\theta$ and
use of $D_t\rho=-\rho u_x$ yield the local entropy balance
\begin{equation}\label{eq:local-entropy-balance}
 (\rho\log\theta-R\rho\log\rho)_t
 +(\rho u\log\theta-R\rho u\log\rho)_x
 =\mu\theta^{\alpha-1}u_x^2
  +\kappa\theta^{\beta-2}\theta_x^2
  +(\kappa\theta^{\beta-1}\theta_x)_x.
\end{equation}
Integrate \eqref{eq:local-entropy-balance}, multiply it by $\bar\theta$, and
subtract it from \eqref{eq:total-energy-balance}.  The boundary fluxes
cancel exactly because $\theta=\bar\theta$ on $\partial(0,1)$:
\[
 \kappa\theta^\beta\theta_x
 -\bar\theta\kappa\theta^{\beta-1}\theta_x=0.
\]
The functional obtained by this subtraction is
\begin{equation*}
 \widetilde{\mathcal E}(t)=\int_0^1\left[
 \frac12\rho u^2+\rho\theta-\bar\theta\rho\log\theta
 +R\bar\theta\rho\log\rho\right]\mathop{}\!\mathrm d x.
\end{equation*}
Expanding the two relative functions in \eqref{eq:relative-energy} gives
\begin{align*}
 \mathcal E(t)-\widetilde{\mathcal E}(t)
 =\int_0^1\bigl[&-R\bar\theta\rho\log m-R\bar\theta\rho+R\bar\theta m
 -\bar\theta\rho+\bar\theta\rho\log\bar\theta\bigr]\mathop{}\!\mathrm d x.
\end{align*}
Since $\int_0^1\rho\mathop{}\!\mathrm d x=m$ and $|(0,1)|=1$, the right-hand side equals
\[
 -R\bar\theta m\log m-\bar\theta m+\bar\theta m\log\bar\theta,
\]
which is constant in time.  Therefore $\mathcal E'=\widetilde{\mathcal E}'$, and the
subtracted balance is exactly \eqref{eq:entropy-identity}.  This calculation
also shows explicitly why centering at the conserved mean $m$ changes only
the normalization and places the minimum at the actual equilibrium density.

For the vacuum limit, integrate the positive-density identity between
$s$ and $t$.  The strong convergence in $C([0,T];H^1)$ established in
Section~\ref{sec:construction} passes every term of $\mathcal E^\delta(s)$ and
$\mathcal E^\delta(t)$ to the limit.  The temperature bounds make the diffusion
coefficients converge uniformly, while weak lower semicontinuity gives
\begin{align*}
 \int_s^t\!\int_0^1\left(
 \mu\bar\theta\theta^{\alpha-1}u_x^2
 +\kappa\bar\theta\theta^{\beta-2}\theta_x^2\right)\mathop{}\!\mathrm d x\mathop{}\!\mathrm d r
\leq\liminf_{\delta\to0}
 \int_s^t\!\int_0^1\left(
 \mu\bar\theta(\theta^\delta)^{\alpha-1}|u_x^\delta|^2
 +\kappa\bar\theta(\theta^\delta)^{\beta-2}|\theta_x^\delta|^2\right)\mathop{}\!\mathrm d x\mathop{}\!\mathrm d r.
\end{align*}
Thus the integrated identity passes to the asserted integrated inequality
without using a positive lower bound for the density.

\label{lem:flux-anchor}
Suppose $0\leq\rho\leq M$ and
$\bar\theta/4\leq\theta\leq2\bar\theta$.  Then
\begin{align}
 \lVertF+\Pi\rVert_{L^\infty}
 \leq m^{1/2}\lVert\sqrt\rho\dot u\rVert_{L^2},
 \label{eq:F-anchor}\\
 \lVertu_x\rVert_{L^\infty}
 \leq C_{\rm F}\bar\theta^{1-\alpha}
   +C_{\rm F}\bar\theta^{-\alpha}\lVert\sqrt\rho\dot u\rVert_{L^2}.
 \label{eq:ux-anchor}
\end{align}
Here $C_{\rm F}=C_{\rm F}(M,R,\mu,\alpha)>0$ can be chosen
nondecreasing in $M$ and is independent of $t$, $\bar\theta$, and
$\inf\rho$.  In particular, \eqref{eq:F-anchor} itself has no dependence
on $M$ and neither estimate uses a lower bound for $\rho$.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Since $\int_0^1u_x\mathop{}\!\mathrm d x=0$, definitions
\eqref{eq:flux}--\eqref{eq:Pi} give
\[
 \int_0^1\theta^{-\alpha}(F+\Pi)\mathop{}\!\mathrm d x=0.
\]
Hence $F(\xi,t)+\Pi(t)=0$ at some $\xi\in[0,1]$.  Therefore
\[
 |F(x,t)+\Pi(t)|\leq\lVertF_x\rVert_{L^1}
 \leq m^{1/2}\lVert\sqrt\rho\dot u\rVert_{L^2}.
\]
The temperature comparison also yields
\[
 \int_0^1\rho\theta^{1-\alpha}\mathop{}\!\mathrm d x
 \leq C_\alpha m\bar\theta^{1-\alpha},\qquad
 \int_0^1\theta^{-\alpha}\mathop{}\!\mathrm d x
 \geq c_\alpha\bar\theta^{-\alpha}.
\]
Consequently,
\[
 |\Pi|\leq C_{\alpha,R}m\bar\theta.
\]
Solving \eqref{eq:flux} for $u_x$ and adding and subtracting $\Pi$ give
\[
 u_x=\frac{\theta^{-\alpha}}{\mu}
       \bigl((F+\Pi)-\Pi+R\rho\theta\bigr).
\]
Since $\theta^{-\alpha}\leq4^\alpha\bar\theta^{-\alpha}$,
$\rho\theta\leq2M\bar\theta$, and $m=\int_0^1\rho\mathop{}\!\mathrm d x\leq M$, we obtain
\begin{align*}
 \lVertu_x\rVert_{L^\infty}
 &\leq\frac{4^\alpha}{\mu}\bar\theta^{-\alpha}
 \left(\lVertF+\Pi\rVert_{L^\infty}+|\Pi|
       +R\lVert\rho\theta\rVert_{L^\infty}\right)\\
 &\leq C_{\alpha,\mu}\bar\theta^{-\alpha}
 \left(m^{1/2}\lVert\sqrt\rho\dot u\rVert_{L^2}
       +(C_{\alpha,R}m+2RM)\bar\theta\right)\\
 &\leq C_{\rm F}\bar\theta^{1-\alpha}
       +C_{\rm F}\bar\theta^{-\alpha}
          \lVert\sqrt\rho\dot u\rVert_{L^2},
\end{align*}
which is \eqref{eq:ux-anchor} with the stated dependence of $C_{\rm F}$.

\label{lem:thermal-anchor}
Under the hypotheses of Lemma~\ref{lem:flux-anchor},
\begin{align}
 \lVert\theta_x\rVert_{L^\infty}
 &\leq C_{\rm G}\bar\theta^{-\beta}\left(
  \lVert\sqrt\rho\dot\theta\rVert_{L^2}+\bar\theta\lVertu_x\rVert_{L^2}
  +\bar\theta^\alpha\lVertu_x\rVert_{L^2}^2\right),
 \label{eq:thetax-infty}\\
 \lVert\theta_{xx}\rVert_{L^2}
 &\leq C_{\rm G}\bar\theta^{-\beta}\left(
  \lVert\sqrt\rho\dot\theta\rVert_{L^2}+\bar\theta\lVertu_x\rVert_{L^2}
  +\bar\theta^\alpha\lVertu_x\rVert_{L^\infty}\lVertu_x\rVert_{L^2}\right)
  +C_{\rm G}\bar\theta^{-1}\lVert\theta_x\rVert_{L^\infty}
                         \lVert\theta_x\rVert_{L^2}.
 \label{eq:thetaxx}
\end{align}
Here
$C_{\rm G}=C_{\rm G}(M,R,\mu,\kappa,\alpha,\beta)>0$ can be chosen
nondecreasing in $M$ and is independent of $t$, $\bar\theta$, and $\inf\rho$.

<!-- proof-3: body_md -->
Because $\theta(0,t)=\theta(1,t)$,
$\int_0^1G\theta^{-\beta}\mathop{}\!\mathrm d x=\kappa\int_0^1\theta_x\mathop{}\!\mathrm d x=0$.
Thus $G$ has a zero and $\lVertG\rVert_{L^\infty}\leq\lVertG_x\rVert_{L^1}$.
To estimate the latter norm, equation \eqref{eq:heat-flux},
Cauchy--Schwarz, mass conservation, the density upper bound, and
$|(0,1)|=1$ give
\begin{align*}
 \lVertG_x\rVert_{L^1}
 &\leq\lVert\rho\dot\theta\rVert_{L^1}
      +R\lVert\rho\theta u_x\rVert_{L^1}
      +\mu\lVert\theta^\alpha u_x^2\rVert_{L^1}\\
 &\leq m^{1/2}\lVert\sqrt\rho\dot\theta\rVert_{L^2}
      +2RM\bar\theta\lVertu_x\rVert_{L^1}
      +2^\alpha\mu\bar\theta^\alpha\lVertu_x\rVert_{L^2}^2\\
 &\leq m^{1/2}\lVert\sqrt\rho\dot\theta\rVert_{L^2}
      +2RM\bar\theta\lVertu_x\rVert_{L^2}
      +2^\alpha\mu\bar\theta^\alpha\lVertu_x\rVert_{L^2}^2.
\end{align*}
Because $G=\kappa\theta^\beta\theta_x$ and
$\theta^{-\beta}\leq4^\beta\bar\theta^{-\beta}$,
\[
 \lVert\theta_x\rVert_{L^\infty}
 \leq\frac{4^\beta}{\kappa}\bar\theta^{-\beta}
       \lVertG\rVert_{L^\infty}
 \leq\frac{4^\beta}{\kappa}\bar\theta^{-\beta}
       \lVertG_x\rVert_{L^1}.
\]
Using $m\leq M$ proves \eqref{eq:thetax-infty} with the stated constant.

For the second derivative, the three terms in
\eqref{eq:heat-flux} satisfy
\begin{align*}
& \lVert\rho\dot\theta\rVert_{L^2}^2
 =\int_0^1\rho^2|\dot\theta|^2\mathop{}\!\mathrm d x
 \leq M\lVert\sqrt\rho\dot\theta\rVert_{L^2}^2,\\
 &R\lVert\rho\theta u_x\rVert_{L^2}
 \leq2RM\bar\theta\lVertu_x\rVert_{L^2},\\
& \mu\lVert\theta^\alpha u_x^2\rVert_{L^2}
 \leq2^\alpha\mu\bar\theta^\alpha
       \lVertu_x^2\rVert_{L^2}
 \leq2^\alpha\mu\bar\theta^\alpha
       \lVertu_x\rVert_{L^\infty}\lVertu_x\rVert_{L^2}.
\end{align*}
Consequently,
\[
 \lVertG_x\rVert_{L^2}
 \leq M^{1/2}\lVert\sqrt\rho\dot\theta\rVert_{L^2}
 +2RM\bar\theta\lVertu_x\rVert_{L^2}
 +2^\alpha\mu\bar\theta^\alpha
       \lVertu_x\rVert_{L^\infty}\lVertu_x\rVert_{L^2}.
\]
Finally, expanding $G_x$ gives the pointwise identity
\[
 \theta_{xx}=\frac{\theta^{-\beta}}{\kappa}G_x
              -\beta\theta^{-1}\theta_x^2.
\]
The bootstrap lower bound for $\theta$ therefore implies
\begin{align*}
 \lVert\theta_{xx}\rVert_{L^2}
 &\leq\frac{4^\beta}{\kappa}\bar\theta^{-\beta}
          \lVertG_x\rVert_{L^2}
      +4|\beta|\bar\theta^{-1}\lVert\theta_x^2\rVert_{L^2}\leq\frac{4^\beta}{\kappa}\bar\theta^{-\beta}
          \lVertG_x\rVert_{L^2}
      +4|\beta|\bar\theta^{-1}\lVert\theta_x\rVert_{L^\infty}
                              \lVert\theta_x\rVert_{L^2}.
\end{align*}
Substitution of the preceding $L^2$ bound for $G_x$ proves
\eqref{eq:thetaxx} and displays explicitly every dependence entering
$C_{\rm G}$.

<!-- lemma-4: body_md -->
\label{lem:zlotnik}
Let $y$ and $b$ be absolutely continuous and satisfy
$y'\leq g(y)+b'$ on $[0,T]$, where $g(s)\to-\infty$ as $s\to\infty$.
If
\[
 b(t_2)-b(t_1)\leq N_0+N_1(t_2-t_1)
 \quad(0\leq t_1<t_2\leq T)
\]
and $g(s)\leq-N_1$ for $s\geq\zeta$, then
$y(t)\leq\max\{y(0),\zeta\}+N_0$.
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Set $M_0=\max\{y(0),\zeta\}$.  Suppose, to the contrary, that there is a
time $t_2\in(0,T]$ such that $y(t_2)>M_0+N_0$.  By continuity of the
absolutely continuous function $y$, there is a last time
\[
 t_1:=\max\{s\in[0,t_2]:y(s)=M_0\}.
\]
This set is nonempty: if $M_0=y(0)$ it contains $0$, whereas if
$M_0=\zeta>y(0)$ the intermediate-value theorem supplies a crossing before
$t_2$.  The choice of $t_1$ implies
$y(s)>M_0\geq\zeta$ for $t_1<s\leq t_2$, and therefore
$g(y(s))\leq-N_1$ almost everywhere on this interval.  Integrating the
differential inequality gives
\begin{align*}
 y(t_2)-M_0
 =y(t_2)-y(t_1)
 \leq\int_{t_1}^{t_2}g(y(s))\mathop{}\!\mathrm d s+b(t_2)-b(t_1)
 &\leq-N_1(t_2-t_1)+N_0+N_1(t_2-t_1)=N_0,
\end{align*}
contradicting $y(t_2)>M_0+N_0$.  Hence
$y(t)\leq M_0+N_0$ on $[0,T]$.

\label{lem:material}
Assume $0\leq\rho\leq M$ and $\bar\theta/4\leq\theta\leq2\bar\theta$.  Then
\begin{align}
 \frac12\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\lVert\sqrt\rho\dot u\rVert_{L^2}^2+c\bar\theta^\alpha \lVert\dot u_x\rVert_{L^2}^2
 &\leq C\bar\theta^{-\alpha}\lVert\sqrt\rho\dot\theta\rVert_{L^2}^2+C\bar\theta^{2-\alpha}\lVertu_x\rVert_{L^2}^2
   +C\bar\theta^\alpha\lVertu_x\rVert_{L^\infty}^2\lVertu_x\rVert_{L^2}^2
   +C\bar\theta^{\alpha-2}\lVertu_x\rVert_{L^2}^2\lVert\dot\theta_x\rVert_{L^2}^2,
\label{eq:material-u}\\
 \frac12\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\lVert\sqrt\rho\dot\theta\rVert_{L^2}^2+c\bar\theta^\beta \lVert\dot\theta_x\rVert_{L^2}^2
 &\leq C\lVertu_x\rVert_{L^\infty} \lVert\sqrt\rho\dot\theta\rVert_{L^2}^2
 +C\bar\theta \lVert\sqrt\rho\dot\theta\rVert_{L^2}\bigl(\lVert\dot u_x\rVert_{L^2}+\lVertu_x\rVert_{L^\infty} \lVertu_x\rVert_{L^2}\bigr)
 \nonumber\\
 &\quad+C\bar\theta^\alpha \lVert\dot\theta_x\rVert_{L^2}\lVertu_x\rVert_{L^\infty} \lVertu_x\rVert_{L^2}^2
 +C\bar\theta^{\alpha-1}\lVert\dot\theta_x\rVert_{L^2}^2\lVertu_x\rVert_{L^2}^2
 \nonumber\\
 &\quad
 +C\bar\theta^\alpha \lVert\dot\theta_x\rVert_{L^2}\lVertu_x\rVert_{L^2}\lVert\dot u_x\rVert_{L^2}
 +C\bar\theta^{\beta-1}\lVert\theta_x\rVert_{L^2}\lVert\dot\theta_x\rVert_{L^2}^2
 \nonumber\\
 &\quad+C\bar\theta^\beta\lVertu_x\rVert_{L^\infty} \lVert\theta_x\rVert_{L^2} \lVert\dot\theta_x\rVert_{L^2} .
\label{eq:material-theta}
\end{align}

$c=c(\mu,\kappa,\alpha,\beta)>0$ denotes a coercivity constant, whereas
$C=C(M,R,\mu,\kappa,\alpha,\beta)>0$ denotes a generic upper-bound
constant that may change from line to line.  Both are independent of
$t$, $\bar\theta$, and $\inf\rho$.

<!-- proof-6: body_md -->
For a scalar function $f$, define in this proof the conservative material
operator
\[
 \mathcal C f=f_t+(uf)_x=D_t f+u_xf.
\]
The continuity equation and direct differentiation give
\begin{equation}\label{eq:material-commutators}
 \mathcal C(\rho f)=\rhoD_t f,\qquad
 \mathcal C(q_x)=(D_t q)_x,
 \qquad D_t u_x=\dot u_x-u_x^2,
 \qquad D_t\theta_x=\dot\theta_x-u_x\theta_x .
\end{equation}
The constitutive laws also give
\[
 D_t(R\rho\theta)=R\rho\dot\theta-R\rho\theta u_x,\qquad
 D_t(\mu\theta^\alpha)=\mu\alpha\theta^{\alpha-1}\dot\theta,\qquad
 D_t(\kappa\theta^\beta)=\kappa\beta\theta^{\beta-1}\dot\theta.
\]
Applying $\mathcal C$ to the momentum equation and using
\eqref{eq:material-commutators}, we obtain the exact equation
\begin{equation}\label{eq:exact-material-momentum}
 \rhoD_t\dot u+\bigl(R\rho\dot\theta-R\rho\theta u_x\bigr)_x
 =\bigl(\mu\alpha\theta^{\alpha-1}\dot\theta u_x
       +\mu\theta^\alpha\dot u_x-\mu\theta^\alpha u_x^2\bigr)_x .
\end{equation}
Since $\dot u=0$ on $\partial(0,1)$, testing by $\dot u$ yields
\begin{align}
 &\frac12\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\lVert\sqrt\rho\dot u\rVert_{L^2}^2
 +\mu\int_0^1\theta^\alpha|\dot u_x|^2\mathop{}\!\mathrm d x\nonumber\\
 &
=R\int_0^1\rho\dot\theta\dot u_x\mathop{}\!\mathrm d x
  -R\int_0^1\rho\theta u_x\dot u_x\mathop{}\!\mathrm d x  -\mu\alpha\int_0^1\theta^{\alpha-1}\dot\theta u_x\dot u_x\mathop{}\!\mathrm d x
  +\mu\int_0^1\theta^\alpha u_x^2\dot u_x\mathop{}\!\mathrm d x .
 \label{eq:exact-material-u-energy}
\end{align}
Here we used
$\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\int_0^1\rho f^2\mathop{}\!\mathrm d x
=2\int_0^1\rho fD_t f\mathop{}\!\mathrm d x$, which follows
from the continuity equation and $u|_{\partial(0,1)}=0$.  For every
$\varepsilon>0$, the four terms on the right satisfy
\begin{align*}
& R\left|\int_0^1\rho\dot\theta\dot u_x\mathop{}\!\mathrm d x\right|
 \leq\varepsilon\bar\theta^\alpha \lVert\dot u_x\rVert_{L^2}^2
      +C_\varepsilon\bar\theta^{-\alpha}\lVert\sqrt\rho\dot\theta\rVert_{L^2}^2,\\
 &R\left|\int_0^1\rho\theta u_x\dot u_x\mathop{}\!\mathrm d x\right|
 \leq\varepsilon\bar\theta^\alpha \lVert\dot u_x\rVert_{L^2}^2
      +C_\varepsilon\bar\theta^{2-\alpha}\lVertu_x\rVert_{L^2}^2,\\
& \mu\alpha\left|\int_0^1\theta^{\alpha-1}\dot\theta u_x\dot u_x\mathop{}\!\mathrm d x\right|
 \leq C\bar\theta^{\alpha-1}\lVert\dot\theta_x\rVert_{L^2}\lVertu_x\rVert_{L^2}\lVert\dot u_x\rVert_{L^2}
  \leq\varepsilon\bar\theta^\alpha \lVert\dot u_x\rVert_{L^2}^2
      +C_\varepsilon\bar\theta^{\alpha-2}\lVertu_x\rVert_{L^2}^2\lVert\dot\theta_x\rVert_{L^2}^2,\\
& \mu\left|\int_0^1\theta^\alpha u_x^2\dot u_x\mathop{}\!\mathrm d x\right|
 \leq C\bar\theta^\alpha\lVertu_x\rVert_{L^\infty} \lVertu_x\rVert_{L^2}\lVert\dot u_x\rVert_{L^2}
  \leq\varepsilon\bar\theta^\alpha \lVert\dot u_x\rVert_{L^2}^2
      +C_\varepsilon\bar\theta^\alpha\lVertu_x\rVert_{L^\infty}^2\lVertu_x\rVert_{L^2}^2.
\end{align*}
Choosing $\varepsilon$ sufficiently small proves
\eqref{eq:material-u}.

We next apply $\mathcal C$ to the temperature equation.  Keeping every
commutator term gives
\begin{align}
 \rhoD_t\dot\theta-(\kappa\theta^\beta\dot\theta_x)_x
 &=-R\rho\dot\theta u_x-R\rho\theta\dot u_x+R\rho\theta u_x^2
   -\mu\theta^\alpha u_x^3\nonumber\\
 &\quad+\mu\alpha\theta^{\alpha-1}\dot\theta u_x^2
   +2\mu\theta^\alpha u_x\dot u_x
   +\partial_x\bigl(\kappa\beta\theta^{\beta-1}\dot\theta\theta_x
                     -\kappa\theta^\beta u_x\theta_x\bigr).
 \label{eq:exact-material-temperature}
\end{align}
Testing by $\dot\theta$ and using
$\dot\theta|_{\partial(0,1)}=0$ gives
\begin{align}
 &\frac12\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\lVert\sqrt\rho\dot\theta\rVert_{L^2}^2
 +\kappa\int_0^1\theta^\beta|\dot\theta_x|^2\mathop{}\!\mathrm d x\nonumber\\
 &=-R\int_0^1\rho u_x|\dot\theta|^2\mathop{}\!\mathrm d x
 -R\int_0^1\rho\theta\dot u_x\dot\theta\mathop{}\!\mathrm d x +R\int_0^1\rho\theta u_x^2\dot\theta\mathop{}\!\mathrm d x
 -\mu\int_0^1\theta^\alpha u_x^3\dot\theta\mathop{}\!\mathrm d x+\mu\alpha\int_0^1\theta^{\alpha-1}|\dot\theta|^2u_x^2\mathop{}\!\mathrm d x\nonumber\\
 &\quad
 +2\mu\int_0^1\theta^\alpha u_x\dot u_x\dot\theta\mathop{}\!\mathrm d x -\kappa\int_0^1\bigl(\beta\theta^{\beta-1}\dot\theta\theta_x
                        -\theta^\beta u_x\theta_x\bigr)
       \dot\theta_x\mathop{}\!\mathrm d x\nonumber\\
 &\leq C\lVertu_x\rVert_{L^\infty}\lVert\sqrt\rho\dot\theta\rVert_{L^2}^2
 +C\bar\theta \lVert\sqrt\rho\dot\theta\rVert_{L^2}\lVert\dot u_x\rVert_{L^2}
 +C\bar\theta \lVert\sqrt\rho\dot\theta\rVert_{L^2}\lVertu_x\rVert_{L^\infty}\lVertu_x\rVert_{L^2}+C\bar\theta^{\alpha-1}\lVert\dot\theta_x\rVert_{L^2}^2\lVertu_x\rVert_{L^2}^2\nonumber\\
 &\quad 
 +C\bar\theta^\alpha\lVert\dot\theta_x\rVert_{L^2}\lVertu_x\rVert_{L^2}\lVert\dot u_x\rVert_{L^2}
 +C\bar\theta^{\beta-1}\lVert\theta_x\rVert_{L^2}\lVert\dot\theta_x\rVert_{L^2}^2
 +C\bar\theta^\beta\lVertu_x\rVert_{L^\infty}\lVert\theta_x\rVert_{L^2}\lVert\dot\theta_x\rVert_{L^2}.
 \label{eq:exact-material-temperature-energy}
\end{align}
The inequality in \eqref{eq:exact-material-temperature-energy} follows
from \eqref{eq:dot-poincare}: the seven groups on the right-hand side of
the equality are bounded, in their displayed order, by the seven
corresponding groups following the inequality sign.  These are precisely
the terms on the right of \eqref{eq:material-theta}.  In particular, only
the boundary Poincar\'e
estimate for $\dot u$ and $\dot\theta$ is used; at no point is an unweighted
material derivative estimated by a density-weighted norm.

<!-- proof-7: body_md -->
Let $C_S$ be the norm of the embedding $H^2(0,1)\hookrightarrow L^\infty(0,1)$
and let $C_Z$ be the fixed threshold constant used in the Zlotnik estimate
for the logarithmic density inequality derived below.  Before estimating the
energy, fix
\begin{equation}\label{eq:Min-choice}
 M_{\rm in}>2e\max\{C_SK_0+1,C_Zm^*,1\}.
\end{equation}
All constants below are now computed under the density bootstrap
$\rho\leq2M_{\rm in}$.  The energy constant $K_{\rm in}$ is fixed next,
and the temperature threshold is chosen only after both constants have been
fixed.

The relative-entropy identity and the pointwise bootstrap give
\begin{equation}\label{eq:layer-basic}
 \sup_{0\leq t\leq S}\mathcal E(t)\leq C\bar\theta,
 \qquad
 \int_0^S\left(\bar\theta^{\alpha-1}\lVertu_x\rVert_{L^2}^2
                   +\bar\theta^{\beta-2}\lVert\theta_x\rVert_{L^2}^2\right)\mathop{}\!\mathrm d t\leq C.
\end{equation}
Set
\begin{align*}
 \mathcal H_u&=\frac\mu2\int_0^1\theta^\alpha u_x^2\mathop{}\!\mathrm d x
   -R\int_0^1(\rho\theta-m\bar\theta)u_x\mathop{}\!\mathrm d x,\qquad
 \mathcal H_\theta=\frac\kappa2\int_0^1\theta^\beta\theta_x^2\mathop{}\!\mathrm d x.
\end{align*}
For parameters $0<\eta<1<L$ to be fixed below, define
\begin{equation}\label{eq:layer-functional}
 \mathcal W(t)=\frac{\sigma}{\bar\theta}\mathcal H_u
 +\frac{\sigma}{\bar\theta^2}\mathcal H_\theta
 +\frac{\eta\sigma^2}{2\bar\theta^2}(\bar\theta \lVert\sqrt\rho\dot u\rVert_{L^2}^2+\lVert\sqrt\rho\dot\theta\rVert_{L^2}^2)
 +L\bar\theta^{-\alpha}\mathcal E(t).
\end{equation}
We first prove the pressure-oscillation estimate needed here, without using
any estimate from the subsequent post-layer argument.  Since
\[
 \rho\theta-m\bar\theta=\theta(\rho-m)+m(\theta-\bar\theta),
\]
the pointwise temperature bootstrap, \eqref{eq:entropy-coercivity}, and the
Dirichlet Poincar\'e inequality imply
\begin{align*}
 R\lVert\rho\theta-m\bar\theta\rVert_{L^2}
 \leq R\lVert\theta\rVert_{L^\infty}\lVert\rho-m\rVert_{L^2}
       +Rm\lVert\theta-\bar\theta\rVert_{L^2}
 \leq C\bar\theta\bigl(\bar\theta^{-1}\mathcal E\bigr)^{1/2}
       +C\lVert\theta_x\rVert_{L^2}
 \leq C(\bar\theta\mathcal E)^{1/2}+C\lVert\theta_x\rVert_{L^2}.
\end{align*}
Here and below in this calculation,
$C=C(M_{\rm in},m_*,m^*,R)$ is independent of $\bar\theta$, $\sigma$, and any
positive lower bound for the density.  Consequently, Cauchy--Schwarz gives
\begin{align*}
 \frac{R\sigma}{\bar\theta}
 \left|\int_0^1(\rho\theta-m\bar\theta)u_x\mathop{}\!\mathrm d x\right|
 &\leq C\sigma\bar\theta^{-1}
 \left((\bar\theta\mathcal E)^{1/2}+\lVert\theta_x\rVert_{L^2}\right)
 \lVertu_x\rVert_{L^2}\\
 &=C\left(\sigma^{1/2}\bar\theta^{(\alpha-1)/2}\lVertu_x\rVert_{L^2}\right)
 \left(\sigma^{1/2}\bar\theta^{-\alpha/2}\mathcal E^{1/2}
       +\sigma^{1/2}\bar\theta^{-(\alpha+1)/2}\lVert\theta_x\rVert_{L^2}\right).
\end{align*}
Applying Young's inequality separately to the two products, for every
$\varepsilon>0$ we obtain
\begin{align}\label{eq:layer-cross}
 \frac{R\sigma}{\bar\theta}\left|\int_0^1(\rho\theta-m\bar\theta)u_x\mathop{}\!\mathrm d x\right|
 &\leq\varepsilon\sigma\bar\theta^{\alpha-1}\lVertu_x\rVert_{L^2}^2
 +C_\varepsilon\sigma\bar\theta^{-\alpha}\mathcal E
 +C_\varepsilon\sigma\bar\theta^{-\alpha-1}\lVert\theta_x\rVert_{L^2}^2.
\end{align}
Choosing first $\varepsilon$ and then $L$ gives
\begin{equation}\label{eq:layer-equivalence}
 c\mathcal B_\sigma+c\bar\theta^{-\alpha}\mathcal E
 \leq\mathcal W
 \leq C\mathcal B_\sigma+C\bar\theta^{-\alpha}\mathcal E.
\end{equation}

The initial value of this functional is bounded without any smallness of
$u_0$ or $\theta_0-\bar\theta$.  Indeed,
$\lVert\sqrt{\rho_0}\dot u(0)\rVert_{L^2}=\lVertg_1\rVert_2$ and
$\lVert\sqrt{\rho_0}\dot\theta(0)\rVert_{L^2}=\lVertg_2\rVert_2$, whereas
\eqref{eq:data-dynamic},
\eqref{eq:natural-data}, and $\tau=\bar\theta^{1-\beta}$ give
\begin{align*}
 &\tau\bar\theta^{\alpha-1}\lVertu_x(0)\rVert_{L^2}^2\leq K_0\bar\theta^{\alpha-\beta},
 \qquad
 \tau\bar\theta^{\beta-2}\lVert\theta_x(0)\rVert_{L^2}^2\leq K_0\bar\theta^{-1},\\
 &\tau^2\bar\theta^{-1}\lVert\sqrt{\rho_0}\dot u(0)\rVert_{L^2}^2
   \leq K_0\bar\theta^{2\alpha+1-2\beta},
 \qquad
 \tau^2\bar\theta^{-2}\lVert\sqrt{\rho_0}\dot\theta(0)\rVert_{L^2}^2\leq K_0.
\end{align*}
Since $\mathcal E(0)\leq C\bar\theta$, it follows that
\begin{equation}\label{eq:layer-initial-value}
 \mathcal W(0)\leq C(K_0).
\end{equation}

We next derive the differential inequality.  Put $K:=2K_{\rm in}$.  Under
\eqref{eq:layer-bootstrap}, \eqref{eq:layer-B} gives
\begin{equation}\label{eq:layer-sizes}
 \lVertu_x\rVert_{L^2}^2\leq K\sigma^{-1}\bar\theta^{1-\alpha},\quad
 \lVert\theta_x\rVert_{L^2}^2\leq K\sigma^{-1}\bar\theta^{2-\beta},\quad
 \lVert\sqrt\rho\dot u\rVert_{L^2}^2\leq K\sigma^{-2}\bar\theta,\quad
 \lVert\sqrt\rho\dot\theta\rVert_{L^2}^2\leq K\sigma^{-2}\bar\theta^2.
\end{equation}
In particular,
\begin{equation}\label{eq:layer-smallness}
 \sigma^{-1}\bar\theta^{-\beta}\leq\bar\theta^{-1},\qquad
 \frac{\lVert\theta_x\rVert_{L^2}}{\bar\theta}\leq C_K\bar\theta^{-1/2},\qquad
 \bar\theta^{(\alpha-\beta-1)/2}\lVertu_x\rVert_{L^2}\leq C_K\bar\theta^{-1/2}.
\end{equation}
The flux anchor also yields the time-weighted Lipschitz bound
\begin{equation}\label{eq:layer-ux}
 \sigma\lVertu_x\rVert_{L^\infty}
 \leq C\sigma\bar\theta^{1-\alpha}+C\sigma\bar\theta^{-\alpha}\lVert\sqrt\rho\dot u\rVert_{L^2}
 \leq C_K\bar\theta^{1-\alpha}.
\end{equation}

Multiply \eqref{eq:first-u} by $\sigma\bar\theta^{-1}$ and
\eqref{eq:first-theta} by $\sigma\bar\theta^{-2}$.  Their nonlinear groups are
bounded as follows:
\begin{align}
 \sigma\bar\theta^{-1}\lVert\sqrt\rho\dot\theta\rVert_{L^2}\lVertu_x\rVert_{L^2}
 &\leq\varepsilon\sigma\bar\theta^{-2}\lVert\sqrt\rho\dot\theta\rVert_{L^2}^2
       +C_\varepsilon\lVertu_x\rVert_{L^2}^2,                 \label{eq:layer-I1}\\
 \sigma \lVertu_x\rVert_{L^2}^2&\leq\lVertu_x\rVert_{L^2}^2,                     \label{eq:layer-I2}\\
 \sigma\bar\theta^{\alpha-1}\lVertu_x\rVert_\infty \lVertu_x\rVert_{L^2}^2
 &\leq C_K\lVertu_x\rVert_{L^2}^2,                           \label{eq:layer-I3}\\
 \sigma\bar\theta^{\alpha-2}\lVert\dot\theta_x\rVert_{L^2}\lVertu_x\rVert_{L^2}^2
 &\leq\varepsilon\sigma^2\bar\theta^{\beta-2}\lVert\dot\theta_x\rVert_{L^2}^2
       +C_{\varepsilon,K}\bar\theta^{\alpha-2}\lVertu_x\rVert_{L^2}^2,          \label{eq:layer-I4}\\
 \sigma\bar\theta^{\beta-2}\lVertu_x\rVert_\infty \lVert\theta_x\rVert_{L^2}^2
 &\leq C_K\bar\theta^{\beta-\alpha-1}\lVert\theta_x\rVert_{L^2}^2,             \label{eq:layer-J3}\\
 \sigma\bar\theta^{\beta-3}\lVert\dot\theta_x\rVert_{L^2}\lVert\theta_x\rVert_{L^2}^2
 &\leq\varepsilon\sigma^2\bar\theta^{\beta-2}\lVert\dot\theta_x\rVert_{L^2}^2
       +C_{\varepsilon,K}\bar\theta^{\beta-3}\lVert\theta_x\rVert_{L^2}^2.     \label{eq:layer-J4}
\end{align}
Here \eqref{eq:layer-I4}--\eqref{eq:layer-J4} use, respectively,
\[
 \bar\theta^{2\alpha-\beta-2}\lVertu_x\rVert_{L^2}^4
 \leq C_K\bar\theta^{\alpha-2}\lVertu_x\rVert_{L^2}^2,
 \qquad
 \bar\theta^{\beta-4}\lVert\theta_x\rVert_{L^2}^4
 \leq C_K\bar\theta^{\beta-3}\lVert\theta_x\rVert_{L^2}^2,
\]
which follow from \eqref{eq:layer-sizes} and the first inequality in
\eqref{eq:layer-smallness}.  For clarity, the two thermal terms having the
same algebraic form as \eqref{eq:layer-I1} and \eqref{eq:layer-I4} are,
explicitly,
\begin{align*}
 &\sigma\bar\theta^{-2}
 \left|R\int_0^1\rho\theta u_x\dot\theta\mathop{}\!\mathrm d x\right|
 \leq C\sigma\bar\theta^{-1}\lVertu_x\rVert_{L^2}
             \lVert\sqrt\rho\dot\theta\rVert_{L^2}\leq\varepsilon\sigma\bar\theta^{-2}
             \lVert\sqrt\rho\dot\theta\rVert_{L^2}^2
          +C_\varepsilon\lVertu_x\rVert_{L^2}^2,\\
 &\sigma\bar\theta^{-2}
 \left|\mu\int_0^1\theta^\alpha u_x^2\dot\theta\mathop{}\!\mathrm d x\right|
 \leq C\sigma\bar\theta^{\alpha-2}\lVert\dot\theta_x\rVert_{L^2}
                    \lVertu_x\rVert_{L^2}^2\leq\varepsilon\sigma^2\bar\theta^{\beta-2}
                    \lVert\dot\theta_x\rVert_{L^2}^2
          +C_{\varepsilon,K}\bar\theta^{\alpha-2}\lVertu_x\rVert_{L^2}^2.
\end{align*}
Thus the six displayed estimates together with these two instances cover,
term by term, all four terms in each of \eqref{eq:first-u} and
\eqref{eq:first-theta}.

The decisive step is to use the exact material identities before applying
Young's inequality.  Multiply \eqref{eq:exact-material-u-energy} by $\bar\theta$
and add \eqref{eq:exact-material-temperature-energy}.  The two leading
mixed terms satisfy the exact cancellation
\begin{align}\label{eq:layer-cancellation}
 \bar\theta R\int_0^1\rho\dot\theta\dot u_x\mathop{}\!\mathrm d x
 -R\int_0^1\rho\theta\dot u_x\dot\theta\mathop{}\!\mathrm d x
 =R\int_0^1\rho(\bar\theta-\theta)\dot\theta\dot u_x\mathop{}\!\mathrm d x.
\end{align}
Consequently,
\begin{align}\label{eq:layer-material-exact}
 &\frac12(\bar\theta \lVert\sqrt\rho\dot u\rVert_{L^2}^2+\lVert\sqrt\rho\dot\theta\rVert_{L^2}^2)'
 +\mu\bar\theta\int_0^1\theta^\alpha|\dot u_x|^2\mathop{}\!\mathrm d x
 +\kappa\int_0^1\theta^\beta|\dot\theta_x|^2\mathop{}\!\mathrm d x=R\int_0^1\rho(\bar\theta-\theta)\dot\theta\dot u_x\mathop{}\!\mathrm d x
 +\mathcal R_m,
\end{align}
where
\begin{align*}
 \mathcal R_m={}&
 -R\bar\theta\int_0^1\rho\theta u_x\dot u_x\mathop{}\!\mathrm d x
 -\mu\alpha\bar\theta\int_0^1\theta^{\alpha-1}\dot\theta u_x\dot u_x\mathop{}\!\mathrm d x
 +\mu\bar\theta\int_0^1\theta^\alpha u_x^2\dot u_x\mathop{}\!\mathrm d x
 -R\int_0^1\rho u_x|\dot\theta|^2\mathop{}\!\mathrm d x\\
 &+R\int_0^1\rho\theta u_x^2\dot\theta\mathop{}\!\mathrm d x
 -\mu\int_0^1\theta^\alpha u_x^3\dot\theta\mathop{}\!\mathrm d x
 +\mu\alpha\int_0^1\theta^{\alpha-1}|\dot\theta|^2u_x^2\mathop{}\!\mathrm d x
 +2\mu\int_0^1\theta^\alpha u_x\dot u_x\dot\theta\mathop{}\!\mathrm d x\\
 &-\kappa\beta\int_0^1\theta^{\beta-1}\dot\theta\theta_x\dot\theta_x\mathop{}\!\mathrm d x
 +\kappa\int_0^1\theta^\beta u_x\theta_x\dot\theta_x\mathop{}\!\mathrm d x .
\end{align*}
All undisplayed integrals in this formula are over $(0,1)$.

Multiply \eqref{eq:layer-material-exact} by
$\eta\sigma^2\bar\theta^{-2}$.  Since
$\lVert\theta-\bar\theta\rVert_\infty\leq \lVert\theta_x\rVert_{L^2}\leq C_K\bar\theta^{1/2}$,
the remaining mixed term satisfies
\begin{align}\label{eq:layer-main-mixed}
 &\eta\sigma^2\bar\theta^{-2}
 \left|\int_0^1\rho(\bar\theta-\theta)\dot\theta\dot u_x\mathop{}\!\mathrm d x\right|\leq\varepsilon\eta\sigma^2\bar\theta^{\alpha-1}\lVert\dot u_x\rVert_{L^2}^2
 +C_{\varepsilon,K}\bar\theta^{-\alpha}
       \sigma\bar\theta^{-2}\lVert\sqrt\rho\dot\theta\rVert_{L^2}^2.
\end{align}
The ten terms in $\mathcal R_m$, in their displayed order, obey
\begin{align}
 \eta\sigma^2\lVertu_x\rVert_{L^2}\lVert\dot u_x\rVert_{L^2}
 &\leq\varepsilon\eta\sigma^2\bar\theta^{\alpha-1}\lVert\dot u_x\rVert_{L^2}^2
 +C_\varepsilon\sigma^2\bar\theta^{1-\alpha}\lVertu_x\rVert_{L^2}^2,              \label{eq:layer-M1}\\
 \eta\sigma^2\bar\theta^{\alpha-2}\lVert\dot\theta_x\rVert_{L^2}\lVertu_x\rVert_{L^2}\lVert\dot u_x\rVert_{L^2}
 &\leq C_K\bar\theta^{-1/2}\eta
  (\sigma^2\bar\theta^{\alpha-1}\lVert\dot u_x\rVert_{L^2}^2
   +\sigma^2\bar\theta^{\beta-2}\lVert\dot\theta_x\rVert_{L^2}^2),                       \label{eq:layer-M2}\\
 \eta\sigma^2\bar\theta^{\alpha-1}\lVertu_x\rVert_\infty \lVertu_x\rVert_{L^2}\lVert\dot u_x\rVert_{L^2}
 &\leq\varepsilon\eta\sigma^2\bar\theta^{\alpha-1}\lVert\dot u_x\rVert_{L^2}^2
 +C_\varepsilon\eta(\sigma\lVertu_x\rVert_\infty)^2
       \bar\theta^{\alpha-1}\lVertu_x\rVert_{L^2}^2,              \label{eq:layer-M3}\\
 \eta\sigma^2\bar\theta^{-2}\lVertu_x\rVert_\infty \lVert\sqrt\rho\dot\theta\rVert_{L^2}^2
 &\leq C_K\bar\theta^{1-\alpha}\eta\sigma\bar\theta^{-2}\lVert\sqrt\rho\dot\theta\rVert_{L^2}^2,       \label{eq:layer-N1}\\
 \eta\sigma^2\bar\theta^{-1}\lVert\sqrt\rho\dot\theta\rVert_{L^2}\lVertu_x\rVert_\infty \lVertu_x\rVert_{L^2}
 &\leq\varepsilon\sigma\bar\theta^{-2}\lVert\sqrt\rho\dot\theta\rVert_{L^2}^2
 +C_{\varepsilon,K}(\sigma\lVertu_x\rVert_\infty)^2
       \lVertu_x\rVert_{L^2}^2,                              \label{eq:layer-N2}\\
 \eta\sigma^2\bar\theta^{\alpha-2}\lVert\dot\theta_x\rVert_{L^2}\lVertu_x\rVert_\infty \lVertu_x\rVert_{L^2}^2
 &\leq\varepsilon\eta\sigma^2\bar\theta^{\beta-2}\lVert\dot\theta_x\rVert_{L^2}^2
 +C_{\varepsilon,K}(\sigma\lVertu_x\rVert_\infty)^2
       \bar\theta^{\alpha-2}\lVertu_x\rVert_{L^2}^2,              \label{eq:layer-N3}\\
 \eta\sigma^2\bar\theta^{\alpha-3}\lVert\dot\theta_x\rVert_{L^2}^2\lVertu_x\rVert_{L^2}^2
 &\leq C_K\bar\theta^{-1}\eta\sigma^2\bar\theta^{\beta-2}\lVert\dot\theta_x\rVert_{L^2}^2,  \label{eq:layer-N4}\\
 \eta\sigma^2\bar\theta^{\alpha-2}\lVert\dot\theta_x\rVert_{L^2}\lVertu_x\rVert_{L^2}\lVert\dot u_x\rVert_{L^2}
 &\leq C_K\bar\theta^{-1/2}\eta
  (\sigma^2\bar\theta^{\alpha-1}\lVert\dot u_x\rVert_{L^2}^2
   +\sigma^2\bar\theta^{\beta-2}\lVert\dot\theta_x\rVert_{L^2}^2),                       \label{eq:layer-N5}\\
 \eta\sigma^2\bar\theta^{\beta-3}\lVert\theta_x\rVert_{L^2}\lVert\dot\theta_x\rVert_{L^2}^2
 &\leq C_K\bar\theta^{-1/2}\eta\sigma^2\bar\theta^{\beta-2}\lVert\dot\theta_x\rVert_{L^2}^2,\label{eq:layer-N6}\\
 \eta\sigma^2\bar\theta^{\beta-2}\lVertu_x\rVert_\infty \lVert\theta_x\rVert_{L^2}\lVert\dot\theta_x\rVert_{L^2}
 &\leq\varepsilon\eta\sigma^2\bar\theta^{\beta-2}\lVert\dot\theta_x\rVert_{L^2}^2
 +C_\varepsilon\eta(\sigma\lVertu_x\rVert_\infty)^2
       \bar\theta^{\beta-2}\lVert\theta_x\rVert_{L^2}^2.          \label{eq:layer-N7}
\end{align}
Only \eqref{eq:layer-sizes}--\eqref{eq:layer-ux}, the boundary
Poincar\'e estimates in \eqref{eq:dot-poincare}, and $\rho\leq2M_{\rm in}$
are used here.  Thus no density lower bound is hidden in the table.

It remains to account for the derivative of the time weight.  On the set
where $\sigma'=1$,
\begin{align}
 \frac{\sigma'}{\bar\theta}|\mathcal H_u|
 &\leq C\bigl(\bar\theta^{\alpha-1}\lVertu_x\rVert_{L^2}^2
                 +\bar\theta^{\beta-2}\lVert\theta_x\rVert_{L^2}^2\bigr)
        +C\bar\theta^{-\alpha}\mathcal E,
 \label{eq:layer-weight-u}\\
 \frac{\sigma'}{\bar\theta^2}|\mathcal H_\theta|
 &\leq C\bar\theta^{\beta-2}\lVert\theta_x\rVert_{L^2}^2.       \label{eq:layer-weight-theta}
\end{align}
The derivative of the material weight is
\[
 \eta\sigma\sigma'(\bar\theta^{-1}\lVert\sqrt\rho\dot u\rVert_{L^2}^2+\bar\theta^{-2}\lVert\sqrt\rho\dot\theta\rVert_{L^2}^2),
\]
which is absorbed by the two first-identity dissipations after $\eta$ is
chosen sufficiently small.  The sources
\eqref{eq:layer-weight-u}--\eqref{eq:layer-weight-theta} are retained on the
right and are integrable by \eqref{eq:layer-basic}; no pointwise absorption
of them is used.  Combining
\eqref{eq:layer-I1}--\eqref{eq:layer-weight-theta}, choose first $\eta$,
then every Young parameter $\varepsilon$ smaller than a fixed multiple of
$\eta$, next $L$, and finally $\bar\theta$.  This gives
{\small
\begin{equation}\label{eq:layer-differential}
 \mathcal W'\!+c\sigma\bigl[\bar\theta^{-1}\lVert\sqrt\rho\dot u\rVert_{L^2}^2+\bar\theta^{-2}\lVert\sqrt\rho\dot\theta\rVert_{L^2}^2
 +\eta\sigma\bigl(\bar\theta^{\alpha-1}\lVert\dot u_x\rVert_{L^2}^2+\bar\theta^{\beta-2}\lVert\dot\theta_x\rVert_{L^2}^2\bigr)\bigr]
 \leq C\bigl(\bar\theta^{\alpha-1}\lVertu_x\rVert_{L^2}^2+\bar\theta^{\beta-2}\lVert\theta_x\rVert_{L^2}^2+\bar\theta^{-\alpha}\mathcal E\bigr).
\end{equation}
}
Integration, \eqref{eq:layer-basic}, \eqref{eq:layer-equivalence}, and
\eqref{eq:layer-initial-value} prove the energy and dissipation parts of
\eqref{eq:layer-improvement}--\eqref{eq:layer-dissipation}, with a strict
improvement after $K_{\rm in}$ is fixed sufficiently large.

For the temperature, \eqref{eq:layer-B} and $\sigma\geq\tau$ give
\[
 \lVert\theta_x(t)\rVert_{L^2}^2\leq C\sigma^{-1}\bar\theta^{2-\beta}\leq C\bar\theta.
\]
Since $\theta-\bar\theta$ has zero boundary trace,
$\lVert\theta-\bar\theta\rVert_\infty\leq \lVert\theta_x\rVert_{L^2}\leq C\bar\theta^{1/2}$.  Increasing
$\Theta_{\rm in}$ improves the temperature bootstrap.

For the density, we first derive the logarithmic inequality needed for the
Zlotnik argument, rather than invoking its later post-layer formulation.
Along a positive-density particle path $X(t)$, put
$y(t)=\log\rho(X(t),t)$.  The continuity equation and
\eqref{eq:flux} give
\begin{equation*}
 y'=\frac{\dot\rho}{\rho}=-u_x=-\frac{F+R\rho\theta}{\mu\theta^\alpha}
 =-\frac{R}{\mu}\rho\theta^{1-\alpha}+\frac{1}{\mu}\theta^{-\alpha}\Pi
 -\frac{1}{\mu}\theta^{-\alpha}(F+\Pi).
\end{equation*}
We now estimate the three terms separately.  Since
$\bar\theta/4\leq\theta\leq2\bar\theta$ and $\alpha>2$,
\[
 -\frac{R}{\mu}\rho\theta^{1-\alpha}
 \leq-c_0\bar\theta^{1-\alpha}\rho
 =-c_0\bar\theta^{1-\alpha}e^y.
\]
Moreover, the definition \eqref{eq:Pi}, the same pointwise temperature
bounds, and $\int_0^1\rho\mathop{}\!\mathrm d x=m$ imply
\begin{equation*}
 0\leq\Pi(t)=R\frac{\displaystyle\int_0^1\rho\theta^{1-\alpha}\mathop{}\!\mathrm d x}
          {\displaystyle\int_0^1\theta^{-\alpha}\mathop{}\!\mathrm d x}
 \leq C\frac{m\bar\theta^{1-\alpha}}{\bar\theta^{-\alpha}}\leq C m\bar\theta.
\end{equation*}
and hence
\[
 \frac{1}{\mu}\theta^{-\alpha}\Pi(t)
 \leq C_0m\bar\theta^{1-\alpha}.
\]
Finally, the flux-anchor estimate \eqref{eq:F-anchor} gives
\[
 \frac{1}{\mu}\theta^{-\alpha}|F+\Pi|
 \leq C_0m^{1/2}\bar\theta^{-\alpha}
       \lVert\sqrt\rho\dot u(t)\rVert_{L^2}.
\]
Combining these estimates proves, at this first point of use,
\begin{equation*}
 y'\leq-c_0\bar\theta^{1-\alpha}e^y
 +C_0m\bar\theta^{1-\alpha}
 +C_0m^{1/2}\bar\theta^{-\alpha}
       \lVert\sqrt\rho\dot u(t)\rVert_{L^2}.
\end{equation*}
The constants $c_0,C_0>0$ depend only on the displayed fixed parameters and
the pointwise temperature ratios; in particular, they are independent of
$\bar\theta$ and of any positive lower bound for the density.

We next estimate the increment in the last term.  By
\eqref{eq:layer-B},
\begin{equation}\label{eq:layer-P-pointwise}
 \lVert\sqrt\rho\dot u(t)\rVert_{L^2}\leq C_{K_{\rm in}}\bar\theta^{1/2}\sigma(t)^{-1}.
\end{equation}
Hence, for $0\leq t_1<t_2\leq S$,
\begin{align}\label{eq:layer-P-increment}
 \bar\theta^{-\alpha}\int_{t_1}^{t_2}\lVert\sqrt\rho\dot u(t)\rVert_{L^2}\mathop{}\!\mathrm d t
 &\leq C_{K_{\rm in}}\bar\theta^{1/2-\alpha}
       \int_{t_1}^{t_2}\frac{\mathop{}\!\mathrm d t}{\sigma(t)}\leq C_{K_{\rm in}}\bar\theta^{1/2-\alpha}
       \bigl((\beta-1)\log\bar\theta+1\bigr)=:N_0(\bar\theta),
\end{align}
where we split the integral at $1-\tau$ if that point belongs to
$[t_1,t_2]$.  Thus $N_0(\bar\theta)\to0$ because $\alpha>2$.  Combining the
logarithmic density inequality just proved with this increment estimate, in
Lemma~\ref{lem:zlotnik} we take
\[
 g_{\bar\theta}(y)=-c_0\bar\theta^{1-\alpha}e^y+C_0m\bar\theta^{1-\alpha},
 \qquad N_1=0,
\]
and choose the fixed threshold $\zeta=\log(C_Zm)$.  After increasing
$\Theta_{\rm in}$ so that $N_0(\bar\theta)\leq1$, the lemma gives
\[
 \rho(x,t)\leq e\max\{\lVert\rho_0\rVert_\infty+1,C_Zm\},
\]
uniformly in the positive approximation.  By \eqref{eq:Min-choice}, this
strictly improves the density bootstrap and involves no circular choice.
Finally,
$\sigma(1)=1$ and \eqref{eq:layer-B} imply
\eqref{eq:layer-entry}.  The proof is complete.

From \eqref{eq:A}, the bootstrap gives
\begin{equation}\label{eq:scaled-size}
 \lVertu_x\rVert_{L^2}\leq K^{1/2}\bar\theta^{1-\alpha/2},\quad
 \lVert\theta_x\rVert_{L^2}\leq K^{1/2}\bar\theta^{1-\beta/2},\quad

Lemmas~\ref{lem:flux-anchor} and \ref{lem:thermal-anchor} then imply
\begin{equation}\label{eq:scaled-pointwise}
 \lVertu_x\rVert_{L^\infty}\leq C_{M,K}\bar\theta^{1-\alpha},\qquad
 \lVert\theta_x\rVert_{L^\infty}\leq C_{M,K}\bar\theta^{2-\beta},\qquad
 \lVert\theta_{xx}\rVert_{L^2}\leq C_{M,K}\bar\theta^{1-\beta}.
\end{equation}
Here $C_{M,K}$ depends, besides $M$ and $K$, only on
$R,\mu,\kappa,\alpha,\beta$, and is independent of $\inf\rho$.

\eqref{eq:first-u}--\eqref{eq:material-theta}.  Constants depending only on

 \bar\theta^{-2}\lVert\sqrt\rho\dot\theta\rVert_{L^2}\lVertu_x\rVert_{L^2}
 &\leq\varepsilon\bar\theta^{-2}\lVert\sqrt\rho\dot\theta\rVert_{L^2}^2
       +C_\varepsilon\bar\theta^{-2}\lVertu_x\rVert_{L^2}^2,
 \label{eq:book-I1}\\
 \bar\theta^{-1}\lVertu_x\rVert_{L^2}^2
 &\quad\text{is retained without further estimation},
 \label{eq:book-I2}\\
 \bar\theta^{\alpha-2}\lVertu_x\rVert_{L^\infty} \lVertu_x\rVert_{L^2}^2
 &\leq\varepsilon\bar\theta^{-2}\lVert\sqrt\rho\dot u\rVert_{L^2}^2
       +C_\varepsilon\bigl(\bar\theta^{-1}+K\bar\theta^{-\alpha}\bigr)
          \lVertu_x\rVert_{L^2}^2,\label{eq:book-I3}\\
 \bar\theta^{\alpha-3}\lVert\dot\theta_x\rVert_{L^2}\lVertu_x\rVert_{L^2}^2
 &\leq\varepsilon\bar\theta^{\beta-2}\lVert\dot\theta_x\rVert_{L^2}^2
       +C_\varepsilon K\bar\theta^{\alpha-\beta-2}\lVertu_x\rVert_{L^2}^2.
\label{eq:book-I4}
\end{align}
Here \eqref{eq:book-I3} follows by inserting the two terms of
\eqref{eq:ux-anchor} before applying Young's inequality.  The four terms in
\eqref{eq:first-theta} obey
\begin{align}
 \bar\theta^{\alpha-2}\lVert\dot\theta_x\rVert_{L^2}\lVertu_x\rVert_{L^2}^2
 &\leq\varepsilon\bar\theta^{\beta-2}\lVert\dot\theta_x\rVert_{L^2}^2
       +C_\varepsilon K\bar\theta^{\alpha-\beta}\lVertu_x\rVert_{L^2}^2,
\label{eq:book-J1}\\
 \bar\theta^{-1}\lVert\sqrt\rho\dot\theta\rVert_{L^2}\lVertu_x\rVert_{L^2}
 &\leq\varepsilon\bar\theta^{-2}\lVert\sqrt\rho\dot\theta\rVert_{L^2}^2
       +C_\varepsilon\lVertu_x\rVert_{L^2}^2,
\label{eq:book-J2}\\
 \bar\theta^{\beta-2}\lVertu_x\rVert_{L^\infty} \lVert\theta_x\rVert_{L^2}^2
 &\leq\varepsilon\bar\theta^{-2}\lVert\sqrt\rho\dot u\rVert_{L^2}^2
       +C_\varepsilon\bigl(\bar\theta^{\beta-\alpha-1}
          +K\bar\theta^{\beta-2\alpha}\bigr)\lVert\theta_x\rVert_{L^2}^2,
\label{eq:book-J3}\\
 \bar\theta^{\beta-3}\lVert\dot\theta_x\rVert_{L^2}\lVert\theta_x\rVert_{L^2}^2
 &\leq\varepsilon\bar\theta^{\beta-2}\lVert\dot\theta_x\rVert_{L^2}^2
       +C_\varepsilon K\bar\theta^{-2}\lVert\theta_x\rVert_{L^2}^2.
\label{eq:book-J4}
\end{align}
The two pressure factors in these displays are harmless constants depending
on $M$.

After multiplication by $\eta\bar\theta^{-2}$, the four terms on the right of
\eqref{eq:material-u} are bounded by
\begin{align}
 \bar\theta^{-\alpha-2}\lVert\sqrt\rho\dot\theta\rVert_{L^2}^2,\qquad
 \bar\theta^{-\alpha}\lVertu_x\rVert_{L^2}^2,\qquad
 \bar\theta^{-\alpha}\lVertu_x\rVert_{L^2}^2
   +K\bar\theta^{-2\alpha}\lVert\sqrt\rho\dot u\rVert_{L^2}^2,\qquad
 K\bar\theta^{-2}\lVert\dot\theta_x\rVert_{L^2}^2 .
\label{eq:book-M}
\end{align}
Finally, the eight scalar groups on the right of
\eqref{eq:material-theta}, in their displayed order, satisfy
\begin{align}
 \bar\theta^{-2}\lVertu_x\rVert_{L^\infty} \lVert\sqrt\rho\dot\theta\rVert_{L^2}^2
 &\leq C(1+K^{1/2})\bar\theta^{-\alpha-1}
       \lVert\sqrt\rho\dot\theta\rVert_{L^2}^2,\label{eq:book-N1}\\
 \bar\theta^{-1}\lVert\sqrt\rho\dot\theta\rVert_{L^2}\lVert\dot u_x\rVert_{L^2}
 &\leq\varepsilon\bar\theta^{\alpha-2}\lVert\dot u_x\rVert_{L^2}^2
       +C_\varepsilon\bar\theta^{-\alpha}\lVert\sqrt\rho\dot\theta\rVert_{L^2}^2,
\label{eq:book-N2a}\\
 \bar\theta^{-1}\lVert\sqrt\rho\dot\theta\rVert_{L^2}\lVertu_x\rVert_{L^\infty} \lVertu_x\rVert_{L^2}
 &\leq\varepsilon\bar\theta^{-2}\lVert\sqrt\rho\dot\theta\rVert_{L^2}^2
       +C_\varepsilon\bar\theta^{2-2\alpha}\lVertu_x\rVert_{L^2}^2
       +C_\varepsilon K\bar\theta^{2-3\alpha}
          \lVert\sqrt\rho\dot u\rVert_{L^2}^2,\label{eq:book-N2b}\\
 \bar\theta^{\alpha-2}\lVert\dot\theta_x\rVert_{L^2}\lVertu_x\rVert_{L^\infty} \lVertu_x\rVert_{L^2}^2
 &\leq\varepsilon\bar\theta^{\beta-2}\lVert\dot\theta_x\rVert_{L^2}^2
       +C_\varepsilon K\bar\theta^{2-\alpha-\beta}\lVertu_x\rVert_{L^2}^2
       +C_\varepsilon K^2\bar\theta^{2-2\alpha-\beta}
          \lVert\sqrt\rho\dot u\rVert_{L^2}^2,
\label{eq:book-N3}\\
 \bar\theta^{\alpha-3}\lVert\dot\theta_x\rVert_{L^2}^2\lVertu_x\rVert_{L^2}^2
 &\leq K\bar\theta^{-1}\lVert\dot\theta_x\rVert_{L^2}^2,\label{eq:book-N4}\\
 \bar\theta^{\alpha-2}\lVert\dot\theta_x\rVert_{L^2}\lVertu_x\rVert_{L^2}\lVert\dot u_x\rVert_{L^2}
 &\leq C K^{1/2}\bigl(
       \bar\theta^{\alpha-\beta/2-1}\lVert\dot u_x\rVert_{L^2}^2
       +\bar\theta^{\beta/2-1}\lVert\dot\theta_x\rVert_{L^2}^2\bigr),
\label{eq:book-N5}\\
 \bar\theta^{\beta-3}\lVert\theta_x\rVert_{L^2}\lVert\dot\theta_x\rVert_{L^2}^2
 &\leq C K^{1/2}\bar\theta^{\beta/2-2}\lVert\dot\theta_x\rVert_{L^2}^2,
 \label{eq:book-N6}\\
 \bar\theta^{\beta-2}\lVertu_x\rVert_{L^\infty} \lVert\theta_x\rVert_{L^2}\lVert\dot\theta_x\rVert_{L^2}
 &\leq\varepsilon\bar\theta^{\beta-2}\lVert\dot\theta_x\rVert_{L^2}^2
       +C_\varepsilon\bar\theta^{\beta-2\alpha}\lVert\theta_x\rVert_{L^2}^2
       +C_\varepsilon K\bar\theta^{-2\alpha}\lVert\sqrt\rho\dot u\rVert_{L^2}^2.
\label{eq:book-N7}
\end{align}
These are exhaustive: \eqref{eq:book-I1}--\eqref{eq:book-I4} correspond to
the four terms in \eqref{eq:first-u},
\eqref{eq:book-J1}--\eqref{eq:book-J4} to the four terms in
\eqref{eq:first-theta}, \eqref{eq:book-M} to
\eqref{eq:material-u}, and \eqref{eq:book-N1}--\eqref{eq:book-N7} to
\eqref{eq:material-theta}.  To illustrate the normalization, the pressure
 work in \eqref{eq:first-u} gives
\[
 R\bar\theta^{-2}\left|\int_0^1\rho\theta u_x^2\mathop{}\!\mathrm d x\right|
 \leq C\bar\theta^{-1}\lVertu_x\rVert_{L^2}^2,
\]
whereas the mixed material term gives
\begin{align*}
 \bar\theta^{\alpha-2}\lVert\dot\theta_x\rVert_{L^2}\lVertu_x\rVert_{L^2}\lVert\dot u_x\rVert_{L^2}
 &\leq C K^{1/2}\bar\theta^{1-\beta/2}
 \left(\bar\theta^{\alpha-2}\lVert\dot u_x\rVert_{L^2}^2+\bar\theta^{\beta-2}\lVert\dot\theta_x\rVert_{L^2}^2\right).
\end{align*}
Thus the constants in the table are independent of $\inf\rho$.

The choices are made in the following order.  First fix $\eta>0$ in
\eqref{eq:Y} sufficiently small.  Next choose every Young parameter
$\varepsilon$ smaller than a fixed multiple of $\eta$, so that only
$\mathcal D_h$ is absorbed.  Then choose $L$ in \eqref{eq:Y} to secure the
equivalence \eqref{eq:Y-equivalence}.  Every remaining power containing
$K$ is bounded by
$C_0(1+K)^N\bar\theta^{-\gamma}$, whereas the $K$-independent low-order terms have
coefficient at most $C_0$.  Adding
$L\bar\theta^{-\alpha-1}$ times \eqref{eq:entropy-identity} now gives
\eqref{eq:closure-differential}.  Only $\mathcal D_h$ is absorbed; the
$\mathcal D_0$ terms remain on the right and are integrated with the basic
entropy estimate.  The lower bound on $\bar\theta$ is chosen only afterward in
Proposition~\ref{prop:uniform}.

\label{prop:uniform}
There exist $M_1,K_1,\Theta_1$, depending only on
$R,\mu,\kappa,\alpha,\beta$, $K_0$, $m_*$, and $m^*$, with the following
property.  Let a smooth positive-density solution exist on $[0,T]$, with
$T\geq1$, and suppose that the entry estimates furnished by
Proposition~\ref{prop:initial-layer} hold at $t=1$.  Suppose also, as
bootstrap assumptions on $[1,T]$, that
\begin{equation}\label{eq:bootstrap}
 \rho\leq2M_1,\qquad \frac14\bar\theta\leq\theta\leq2\bar\theta,\qquad
 \mathcal A\leq2K_1 .
\end{equation}
If $\bar\theta\geq\Theta_1$, then on $[1,T]$
\begin{equation}\label{eq:uniform-prop}
 \rho\leq M_1,\qquad \frac12\bar\theta\leq\theta\leq\frac32\bar\theta,\qquad
 \sup_{1\leq t\leq T}\mathcal A(t)+\int_1^T\mathcal D(t)\mathop{}\!\mathrm d t\leq K_1.
\end{equation}
All constants are independent of $\inf\rho$ and of $T$.

Choose the constants in a triangular order.  First fix
\begin{equation}\label{eq:M1-choice}
 M_1=2e\max\{M_{\rm in}+2,C_Z(m^*+1),1\},
\end{equation}
where $C_Z$ is the fixed pressure constant in the threshold supplied by
Lemma~\ref{lem:zlotnik}.  It is important that $C_Z$ is independent of a
density bootstrap upper bound: the trajectory inequality used below follows
directly from \eqref{eq:flux}, \eqref{eq:F-anchor}, and the bound
$|\Pi|\leq C_{\alpha,R}m\bar\theta$ proved in
Lemma~\ref{lem:flux-anchor}, none of which requires $\rho\leq M$.
Thus \eqref{eq:M1-choice} does not involve the $M$-dependent constants
$C_{\rm F}$ or $C_{\rm G}$.  After $M_1$ has been fixed, apply
Lemma~\ref{lem:bookkeeping} with density bootstrap number $2M_1$.
This fixes $C_{\rm F}(2M_1)$, $C_{\rm G}(2M_1)$, and hence $C_0$ and $N$,
before $K_1$ is chosen.  The order is therefore triangular and contains no
logical circle.

The entry bound \eqref{eq:layer-entry}, the equivalence
\eqref{eq:Y-equivalence}, and $\mathcal E(1)\leq\mathcal E(0)\leq C\bar\theta$ imply
$\mathcal Y(1)\leq C_{\rm ent}$, where $C_{\rm ent}$ is fixed by the data class
and, in particular, dominates the equivalence constant times
$1+K_{\rm in}$.  By \eqref{eq:entropy-identity}, there is a fixed
$C_{\rm basic}$ such that, for every $T\geq1$,
\begin{equation}\label{eq:basic-integrated}
 \int_1^T\left(\bar\theta^{\alpha-1}\lVertu_x\rVert_{L^2}^2+
                    \bar\theta^{\beta-2}\lVert\theta_x\rVert_{L^2}^2\right)\mathop{}\!\mathrm d t\leq C_{\rm basic}.
\end{equation}
Let $C_{\rm eq}$ dominate the fixed constants in
\eqref{eq:layer-entry}, \eqref{eq:Y-equivalence}, and in the integration of
\eqref{eq:closure-differential}, and set
\[
 C_*=2C_{\rm eq}
       \bigl(1+C_{\rm ent}+C_0C_{\rm basic}+K_{\rm in}\bigr).
\]
This number is independent of $T$ and, crucially, does not involve the
bootstrap number $K_1$.  Set
\[
 K_1=2\max\{K_0,C_*\}.
\]
Finally choose $\Theta_1$ so large that
\begin{equation}\label{eq:theta-choice}
 C_0(1+2K_1)^N\Theta_1^{-\gamma}
 \leq\min\{c/2,1\}.
\end{equation}
Only the high-order part of the last term of
\eqref{eq:closure-differential} is absorbed.  Its low-order part is
combined with \eqref{eq:basic-integrated}.  Integration, followed by
\eqref{eq:Y-equivalence}, gives
\begin{equation}\label{eq:A-D-pre}
 \sup_{1\leq t\leq T}\mathcal A(t)+\int_1^T\mathcal D(t)\mathop{}\!\mathrm d t
 \leq C_*\leq\frac12K_1.
\end{equation}
This improves the energy bootstrap without a circular choice of constants.

It remains to improve the density and temperature bootstrap.  The
logarithmic density inequality was derived in detail above in the proof of
Proposition~\ref{prop:initial-layer}, before
\eqref{eq:layer-P-pointwise}.  The same derivation uses only the continuity
equation, \eqref{eq:flux}, \eqref{eq:Pi}, \eqref{eq:F-anchor}, and the
pointwise temperature bounds, and therefore applies on the present
post-layer interval.  We record it in the form used here:
\begin{equation}\label{eq:logrho}
 y'\leq-c_0\bar\theta^{1-\alpha}e^y
 +C_0m\bar\theta^{1-\alpha}
 +C_0m^{1/2}\bar\theta^{-\alpha}\lVert\sqrt\rho\dot u(t)\rVert_{L^2}.
\end{equation}
Moreover, \eqref{eq:A-D-pre} gives
$\int_1^T \lVert\sqrt\rho\dot u\rVert_{L^2}^2\mathop{}\!\mathrm d t\leq C_*\bar\theta^2$.  For
$1\leq t_1<t_2\leq T$ and $s=t_2-t_1$,
\begin{equation*}
 C\bar\theta^{-\alpha}\int_{t_1}^{t_2}\lVert\sqrt\rho\dot u\rVert_{L^2}\mathop{}\!\mathrm d t
 \leq C C_*^{1/2}\bar\theta^{1-\alpha}s^{1/2}\leq C C_*\bar\theta^{1-\alpha}
   +\frac{c_0}{4}\bar\theta^{1-\alpha}s .
\end{equation*}

$C C_*\Theta_1^{1-\alpha}\leq1$.  Then
Lemma~\ref{lem:zlotnik}, applied to \eqref{eq:logrho}, yields
\[
 \rho(x,t)\leq e\max\{M_{\rm in}+1,C_Zm\}\leq\frac12M_1,
\]
uniformly for the positive approximations.
This improves the first condition in \eqref{eq:bootstrap}.

Finally, \eqref{eq:A-D-pre} and the equal boundary values of $\theta$ give
\[
 \lVert\theta(t)-\bar\theta\rVert_{L^\infty}\leq\lVert\theta_x(t)\rVert_{L^2}\leq C_*^{1/2}\bar\theta^{1-\beta/2}.
\]
Increase $\Theta_1$ once more so that this is at most $\bar\theta/2$.  This improves the
temperature bootstrap and proves \eqref{eq:uniform-prop}.

<!-- lemma-8: body_md -->
\label{lem:density-high}
Let a smooth solution satisfy \eqref{eq:pointwise-main} and
\eqref{eq:finite-layer-base} on $[0,T]$.  Then
\begin{equation}\label{eq:density-H2}
 \sup_{0\leq t\leq T}\lVert\rho(t)\rVert_{H^2}\leq C_T,
\end{equation}
where $C_T$ does not depend on a positive lower bound for $\rho$.
<!-- end lemma-8 -->
<!-- proof-10: body_md -->
Differentiate $\dot\rho=-\rho u_x$ and eliminate $u_{xx}$ with the momentum
equation.  This gives the exact identity
\begin{align}\label{eq:rhox-transport}
 D_t\rho_x+
 \left(2u_x+\frac R\mu\rho\theta^{1-\alpha}\right)\rho_x
={}&-\frac1\mu\rho^2\theta^{-\alpha}\dot u
 -\frac R\mu\rho^2\theta^{-\alpha}\theta_x
 +\alpha\rho\theta^{-1}\theta_xu_x .
\end{align}
For any $f\in H^1$ and $u|_{\partial(0,1)}=0$,
\[
 \int_0^1(D_t f)f\mathop{}\!\mathrm d x
 =\frac12\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\lVertf\rVert_{L^2}^2
  -\frac12\int_0^1u_xf^2\mathop{}\!\mathrm d x.
\]
Thus, multiplication of \eqref{eq:rhox-transport} by $\rho_x$ gives the
positive, but degenerate, term
$\frac R\mu\int_0^1\rho\theta^{1-\alpha}\rho_x^2\mathop{}\!\mathrm d x$.
The three terms on the
right are estimated without dividing by $\rho$:
\begin{align*}
 \left|\int_0^1\rho^2\theta^{-\alpha}\dot u\rho_x\mathop{}\!\mathrm d x\right|
 &\leq C\lVert\dot u\rVert_{L^2}\lVert\rho_x\rVert_{L^2}
 \leq C \lVert\dot u_x\rVert_{L^2}^2+C\lVert\rho_x\rVert_{L^2}^2,\\
 \left|\int_0^1\rho^2\theta^{-\alpha}\theta_x\rho_x\mathop{}\!\mathrm d x\right|
 &\leq C \lVert\theta_x\rVert_{L^2}^2+C\lVert\rho_x\rVert_{L^2}^2,\\
 \left|\int_0^1\rho\theta^{-1}\theta_xu_x\rho_x\mathop{}\!\mathrm d x\right|
 &\leq C\lVert\theta_x\rVert_{L^\infty} \lVertu_x\rVert_{L^2}\lVert\rho_x\rVert_{L^2}
 \leq C\lVert\theta_x\rVert_{L^\infty}^2\lVertu_x\rVert_{L^2}^2+C\lVert\rho_x\rVert_{L^2}^2.
\end{align*}
Together with \eqref{eq:dot-poincare} and the uniform pointwise bounds,
these estimates imply
\begin{align}\label{eq:rhox-gronwall}
 \frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\lVert\rho_x\rVert_{L^2}^2
 +c\bar\theta^{1-\alpha}\int_0^1\rho\rho_x^2\mathop{}\!\mathrm d x
 \leq C(1+\lVertu_x\rVert_{L^\infty})\lVert\rho_x\rVert_{L^2}^2
 +C\bigl(\lVert\dot u_x\rVert_{L^2}^2+\lVert\theta_x\rVert_{L^2}^2+\lVert\theta_x\rVert_{L^\infty}^2\lVertu_x\rVert_{L^2}^2\bigr).
\end{align}
The last three terms are integrable on every finite interval by
\eqref{eq:finite-layer-base}.  Gronwall's inequality yields
\[
 \sup_{t\leq T}\lVert\rho_x(t)\rVert_{L^2}\leq C_T.
\]
Notice that the positive dissipation in \eqref{eq:rhox-gronwall} is the
degenerate quantity $\int_0^1\rho\rho_x^2\mathop{}\!\mathrm d x$, not
$\lVert\rho_x\rVert_{L^2}^2$.

Put
\[
 A=2u_x+\frac R\mu\rho\theta^{1-\alpha},\qquad
 B=-\frac1\mu\rho^2\theta^{-\alpha}\dot u
 -\frac R\mu\rho^2\theta^{-\alpha}\theta_x
 +\alpha\rho\theta^{-1}\theta_xu_x .
\]
Differentiating \eqref{eq:rhox-transport} once more gives
\begin{equation}\label{eq:rhoxx-transport}
 D_t\rho_{xx}+(A+u_x)\rho_{xx}=B_x-A_x\rho_x .
\end{equation}
No second derivative of $\rho$ occurs on the right.  Indeed,
\begin{align}
 A_x={}&2u_{xx}+\frac R\mu\theta^{1-\alpha}\rho_x
 +\frac R\mu(1-\alpha)\rho\theta^{-\alpha}\theta_x,
\label{eq:Ax}\\
 B_x={}&-\frac1\mu\left(
 2\rho\rho_x\theta^{-\alpha}\dot u
 -\alpha\rho^2\theta^{-\alpha-1}\theta_x\dot u
 +\rho^2\theta^{-\alpha}\dot u_x\right)\nonumber\\
&-\frac R\mu\left(
 2\rho\rho_x\theta^{-\alpha}\theta_x
 -\alpha\rho^2\theta^{-\alpha-1}\theta_x^2
 +\rho^2\theta^{-\alpha}\theta_{xx}\right)\nonumber\\
&+\alpha\left(
 \rho_x\theta^{-1}\theta_xu_x
 -\rho\theta^{-2}\theta_x^2u_x
 +\rho\theta^{-1}\theta_{xx}u_x
 +\rho\theta^{-1}\theta_xu_{xx}\right).
\label{eq:Bx}
\end{align}
The momentum equation further gives
\begin{equation}\label{eq:uxx-elliptic}
 \mu\theta^\alpha u_{xx}
 =\rho\dot u+R\theta\rho_x+R\rho\theta_x
  -\mu\alpha\theta^{\alpha-1}\theta_xu_x,
\end{equation}
so the already obtained $\rho_x$ estimate, \eqref{eq:dot-poincare}, and
\eqref{eq:finite-layer-base} give
\begin{equation}\label{eq:uxx-finite}
 \lVertu_{xx}\rVert_{L^2}\leq C_T(1+\lVert\dot u_x\rVert_{L^2}+\lVert\sqrt\rho\dot u\rVert_{L^2}+\lVert\theta_x\rVert_{L^2}).
\end{equation}
Put $Y(t)=1+\lVert\rho_{xx}(t)\rVert_{L^2}^2$.  Using the one-dimensional inequality
\[
 \lVert\rho_x\rVert_{L^\infty}^2
 \leq C\lVert\rho_x\rVert_{L^2}
       \bigl(\lVert\rho_x\rVert_{L^2}+\lVert\rho_{xx}\rVert_{L^2}\bigr),
\]
the first-derivative bound yields $\lVert\rho_x\rVert_{L^\infty}^2\leq C_TY$.
The products in \eqref{eq:Ax}--\eqref{eq:Bx} fall into the following
classes:
\begin{align*}
 \lVert\rho_x\dot u\rVert_{L^2}^2
 &\leq\lVert\rho_x\rVert_{L^2}^2\lVert\dot u\rVert_{L^\infty}^2
 \leq C_T\lVert\dot u_x\rVert_{L^2}^2,\\
 \lVertu_{xx}\rho_x\rVert_{L^2}^2
 &\leq\lVertu_{xx}\rVert_{L^2}^2\lVert\rho_x\rVert_{L^\infty}^2
 \leq C_T\lVertu_{xx}\rVert_{L^2}^2Y,\\
 \lVert\rho_x^2\rVert_{L^2}^2
 &\leq\lVert\rho_x\rVert_{L^\infty}^2\lVert\rho_x\rVert_{L^2}^2
 \leq C_TY,\\
 \lVert\rho_x\theta_x\rVert_{L^2}^2
 &\leq C_T\lVert\theta_x\rVert_{L^\infty}^2,\\
 \lVert\theta_xu_{xx}\rVert_{L^2}^2
 &\leq\lVert\theta_x\rVert_{L^\infty}^2\lVertu_{xx}\rVert_{L^2}^2,\\
 \lVert\theta_x^2\rVert_{L^2}^2
 &\leq\lVert\theta_x\rVert_{L^\infty}^2\lVert\theta_x\rVert_{L^2}^2,\\
 \lVert\dot u_x\rVert_{L^2}^2+\lVert\theta_{xx}\rVert_{L^2}^2
 &=\lVert\dot u_x\rVert_{L^2}^2+\lVert\theta_{xx}\rVert_{L^2}^2.
\end{align*}
Every remaining factor in \eqref{eq:Ax}--\eqref{eq:Bx} is a bounded
coefficient multiplying one of these seven classes.  Consequently,
\begin{equation}\label{eq:AB-estimate}
 \lVertB_x-A_x\rho_x\rVert_{L^2}^2
 \leq H_T(t)\bigl(1+\lVert\rho_{xx}\rVert_{L^2}^2\bigr),
 \qquad H_T\in L^1(0,T).
\end{equation}
One admissible choice is
\[
 H_T(t)=C_T\left[1+\lVert\dot u_x\rVert_{L^2}^2+\lVert\dot\theta_x\rVert_{L^2}^2+\lVertu_{xx}\rVert_{L^2}^2
 +\lVert\theta_{xx}\rVert_{L^2}^2+\lVertu_x\rVert_{L^\infty}^2
 +\lVert\theta_x\rVert_{L^\infty}^4\right].
\]
Its integrability follows from \eqref{eq:finite-layer-base},
\eqref{eq:thetaxx}, and \eqref{eq:uxx-finite}.  Multiplication of
\eqref{eq:rhoxx-transport} by $\rho_{xx}$ and use of
\eqref{eq:AB-estimate} yield
\[
 \frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\bigl(1+\lVert\rho_{xx}\rVert_{L^2}^2\bigr)
 \leq C\bigl(1+\lVertu_x\rVert_{L^\infty}+H_T(t)\bigr)
       \bigl(1+\lVert\rho_{xx}\rVert_{L^2}^2\bigr).
\]
Another application of Gronwall proves \eqref{eq:density-H2}.

<!-- lemma-9: body_md -->
\label{lem:elliptic-recovery}
Under the assumptions of Lemma~\ref{lem:density-high},
\begin{align}\label{eq:elliptic-finite}
 &\sup_{0\leq t\leq T}
 \left(\lVertu(t)\rVert_{H^2}+\lVert\theta(t)-\bar\theta\rVert_{H^2}\right)
 \leq C_T,\\
 &\int_0^T
 \left(\lVertu\rVert_{H^3}^2+\lVert\theta-\bar\theta\rVert_{H^3}^2\right)\mathop{}\!\mathrm d t
 \leq C_T.\nonumber
\end{align}
<!-- end lemma-9 -->
<!-- proof-11: body_md -->
We first recover the second derivatives.  Formula
\eqref{eq:uxx-elliptic}, the bounds for $\rho$ and $\theta$, and
$\lVert\rho\dot u\rVert_{L^2}\leq M^{1/2}
\lVert\sqrt\rho\dot u\rVert_{L^2}$ give
\begin{align*}
 \lVertu_{xx}\rVert_{L^2}
 \leq C_T\bigl(&\lVert\sqrt\rho\dot u\rVert_{L^2}
 +\lVert\rho_x\rVert_{L^2}+\lVert\theta_x\rVert_{L^2}
 +\lVert\theta_x\rVert_{L^\infty}\lVertu_x\rVert_{L^2}\bigr).
\end{align*}
Lemma~\ref{lem:thermal-anchor}, Lemma~\ref{lem:density-high}, and
\eqref{eq:finite-layer-base} make the right-hand side uniformly bounded on
$[0,T]$.  Since $u=0$ at the endpoints, Poincar\'e's inequality now yields
$\sup_{t\leq T}\lVertu(t)\rVert_{H^2}\leq C_T$.

Similarly, the temperature equation gives
\begin{equation}\label{eq:thetaxx-elliptic}
 \kappa\theta^\beta\theta_{xx}
 =\rho\dot\theta+R\rho\theta u_x-\mu\theta^\alpha u_x^2
  -\kappa\beta\theta^{\beta-1}\theta_x^2,
\end{equation}
and therefore
\begin{align*}
 \lVert\theta_{xx}\rVert_{L^2}
 \leq C_T\bigl(&\lVert\sqrt\rho\dot\theta\rVert_{L^2}
 +\lVertu_x\rVert_{L^2}
 +\lVertu_x\rVert_{L^\infty}\lVertu_x\rVert_{L^2}
 +\lVert\theta_x\rVert_{L^\infty}\lVert\theta_x\rVert_{L^2}\bigr).
\end{align*}
The flux anchor \eqref{eq:ux-anchor}, the thermal anchor
\eqref{eq:thetax-infty}, and \eqref{eq:finite-layer-base} bound this
right-hand side uniformly.  Applying the Dirichlet Poincar\'e inequality to
$\theta-\bar\theta$ proves
$\sup_{t\leq T}\lVert\theta(t)-\bar\theta\rVert_{H^2}\leq C_T$.

Differentiating the two elliptic identities gives
\begin{align}
 \mu\theta^\alpha u_{xxx}={}&
 \rho_x\dot u+\rho\dot u_x+R\theta\rho_{xx}
 +2R\rho_x\theta_x+R\rho\theta_{xx}
 -(\mu\theta^\alpha)_{xx}u_x
 -2(\mu\theta^\alpha)_xu_{xx},\label{eq:uxxx}\\
 \kappa\theta^\beta\theta_{xxx}={}&
 \rho_x\dot\theta+\rho\dot\theta_x+(R\rho\theta)_xu_x
 +R\rho\theta u_{xx}
 -(\mu\theta^\alpha)_xu_x^2-2\mu\theta^\alpha u_xu_{xx}
 -(\kappa\theta^\beta)_{xx}\theta_x
 -2(\kappa\theta^\beta)_x\theta_{xx}.
\label{eq:thetaxxx}
\end{align}
We estimate every class of terms in these identities.  First,
\begin{align*}
 \lVert\rho_x\dot u\rVert_{L^2}
 &\leq\lVert\rho_x\rVert_{L^2}\lVert\dot u\rVert_{L^\infty}
 \leq C_T\lVert\dot u_x\rVert_{L^2},
 &\lVert\rho\dot u_x\rVert_{L^2}&\leq M\lVert\dot u_x\rVert_{L^2},\\
 \lVert\rho_x\dot\theta\rVert_{L^2}
 &\leq C_T\lVert\dot\theta_x\rVert_{L^2},
 &\lVert\rho\dot\theta_x\rVert_{L^2}&\leq M\lVert\dot\theta_x\rVert_{L^2},
\end{align*}
where \eqref{eq:dot-poincare} is used in the first and third estimates.
The density terms satisfy
\begin{align*}
 \lVert\theta\rho_{xx}\rVert_{L^2}
 +\lVert\rho_x\theta_x\rVert_{L^2}
 +\lVert\rho\theta_{xx}\rVert_{L^2}
 \leq C_T,
\end{align*}
because $\rho\in L^\infty(0,T;H^2)$ and
$\theta-\bar\theta\in L^\infty(0,T;H^2)$.

For the variable coefficients, direct differentiation gives
\begin{align*}
 (\theta^\alpha)_x=\alpha\theta^{\alpha-1}\theta_x, \qquad (\theta^\alpha)_{xx}
 =\alpha\theta^{\alpha-1}\theta_{xx}
   +\alpha(\alpha-1)\theta^{\alpha-2}\theta_x^2,
\end{align*}
and the same formulas hold with $\alpha$ replaced by $\beta$.
Consequently, using $H^1(0,1)\hookrightarrow L^\infty(0,1)$,
\begin{align*}
 \lVert(\theta^\alpha)_{xx}u_x\rVert_{L^2}
 &\leq C_T\bigl(\lVert\theta_{xx}\rVert_{L^2}
    +\lVert\theta_x\rVert_{L^\infty}\lVert\theta_x\rVert_{L^2}\bigr)
       \lVertu_x\rVert_{L^\infty}\leq C_T,\\
 \lVert(\theta^\alpha)_xu_{xx}\rVert_{L^2}
 &\leq C_T\lVert\theta_x\rVert_{L^\infty}\lVertu_{xx}\rVert_{L^2}\leq C_T,\\
 \lVert(\theta^\beta)_{xx}\theta_x\rVert_{L^2}
 +\lVert(\theta^\beta)_x\theta_{xx}\rVert_{L^2}
 &\leq C_T\bigl(\lVert\theta_x\rVert_{L^\infty}
       \lVert\theta_{xx}\rVert_{L^2}
       +\lVert\theta_x\rVert_{L^\infty}^2
        \lVert\theta_x\rVert_{L^2}\bigr)\leq C_T.
\end{align*}
The remaining products in \eqref{eq:thetaxxx} obey
\begin{align*}
 \lVert(\rho\theta)_xu_x\rVert_{L^2}
 +\lVert\rho\theta u_{xx}\rVert_{L^2}
 &\leq C_T,\\
 \lVert(\theta^\alpha)_xu_x^2\rVert_{L^2}
 +\lVert\theta^\alpha u_xu_{xx}\rVert_{L^2}
 &\leq C_T,
\end{align*}
using the already established $H^2$ bounds for $u$ and $\theta$.
After division by the uniformly positive coefficients
$\mu\theta^\alpha$ and $\kappa\theta^\beta$, we therefore obtain
\begin{align*}
 \lVertu_{xxx}\rVert_{L^2}
 +\lVert\theta_{xxx}\rVert_{L^2}
 \leq C_T\bigl(1+\lVert\dot u_x\rVert_{L^2}
                    +\lVert\dot\theta_x\rVert_{L^2}\bigr).
\end{align*}
Squaring and integrating over $(0,T)$, then using
\eqref{eq:finite-layer-base}, proves the time-integrated $H^3$ estimate.
Every constant used above depends on the density only through its upper
bound and its $H^2$ norm; no density lower bound, $F_{xx}$ estimate, or
auxiliary-flux estimate is hidden in the argument.

\label{lem:linear-local}
Let $0<\tau\leq1$.  Suppose
\[
 v\in L^\infty(0,\tau;H^2\cap H_0^1)\cap L^2(0,\tau;H^3),
 \qquad v_t\in L^2(0,\tau;H^1).
\]
Then the solution of
\[
 r_t+vr_x+rv_x=0,\qquad r|_{t=0}=r_0\in H^2,
\]
satisfies
\begin{align}
 \lVertr(t)\rVert_{H^2}^2
 &\leq \lVertr_0\rVert_{H^2}^2
 \exp\left(C\int_0^t\bigl(1+\lVertv(s)\rVert_{H^3}\bigr)\mathop{}\!\mathrm d s\right),
 \label{eq:linear-transport-H2}\\
 r(X_v(a,t),t)
 &=r_0(a)\exp\left(-\int_0^t v_x(X_v(a,s),s)\mathop{}\!\mathrm d s\right),
 \label{eq:linear-transport-positive}
\end{align}
where $(X_v)_t=v(X_v,t)$.

Let, in addition, $q,A\in L^\infty(0,\tau;H^2)$,
$q_t,A_t\in L^2(0,\tau;H^1)$ and
\[
 0<q_*\leq q\leq q^*,\qquad 0<A_*\leq A\leq A^*.
\]
For the Dirichlet problem
\begin{equation}\label{eq:linear-parabolic}
 qw_t+qvw_x-(Aw_x)_x=f,\qquad
 w|_{\partial(0,1)}=0,\qquad w|_{t=0}=w_0,
\end{equation}
assume $w_0\in H^2\cap H_0^1$, $f\in L^2(0,\tau;H^1)$,
$f_t\in L^2(0,\tau;H^{-1})$, and that the equation at $t=0$
defines $w_t(0)\in L^2$.  Then \eqref{eq:linear-parabolic} has a unique
solution satisfying
\begin{align}
 &w\in C([0,\tau];H^2\cap H_0^1)\cap L^2(0,\tau;H^3),\nonumber\\
 &w_t\in C([0,\tau];L^2)\cap L^2(0,\tau;H^1),
 \label{eq:linear-parabolic-regularity}
\end{align}
and
\begin{align}
 &\sup_{t\leq\tau}\left(\lVertw(t)\rVert_{H^2}^2
                         +\lVertw_t(t)\rVert_{L^2}^2\right)
 +\int_0^\tau\left(\lVertw\rVert_{H^3}^2
                         +\lVertw_t\rVert_{H^1}^2\right)\mathop{}\!\mathrm d t
 \nonumber\\
 &\qquad\leq C_{\rm lin}\left[
  \lVertw_0\rVert_{H^2}^2+\lVertw_t(0)\rVert_{L^2}^2
  +\lVertf\rVert_{L^2(0,\tau;H^1)}^2+\lVertf_t\rVert_{L^2(0,\tau;H^{-1})}^2\right].
 \label{eq:linear-parabolic-estimate}
\end{align}
Here $C_{\rm lin}$ depends only on the displayed positive bounds and on the
coefficient norms, and is locally bounded as a function of those quantities.

We first prove the transport estimate.  Multiplying the equation by $r$
and integrating the transport term by parts gives
\[
 \frac12\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\lVertr\rVert_{L^2}^2
 +\frac12\int_0^1v_xr^2\mathop{}\!\mathrm d x=0.
\]
The first and second differentiated equations are
\begin{align*}
 r_{xt}+vr_{xx}+2v_xr_x+v_{xx}r&=0,\\
 r_{xxt}+vr_{xxx}+3v_xr_{xx}+3v_{xx}r_x+v_{xxx}r&=0.
\end{align*}
Testing them by $r_x$ and $r_{xx}$, respectively, and integrating the
terms containing $v r_{xx}$ and $v r_{xxx}$ by parts, we obtain
\begin{align*}
 \frac12\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\lVertr_x\rVert_{L^2}^2
 \leq C\lVertv\rVert_{H^2}\lVertr\rVert_{H^1}^2, \qquad
 \frac12\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\lVertr_{xx}\rVert_{L^2}^2
 \leq C\lVertv\rVert_{H^3}\lVertr\rVert_{H^2}^2.
\end{align*}
Here, for example,
\begin{align*}
 \left|\int_0^1v_{xx}r_xr_{xx}\mathop{}\!\mathrm d x\right|
 &\leq \lVertv_{xx}\rVert_{L^2}\lVertr_x\rVert_{L^\infty}
             \lVertr_{xx}\rVert_{L^2},\\
 \left|\int_0^1v_{xxx}rr_{xx}\mathop{}\!\mathrm d x\right|
 &\leq \lVertv_{xxx}\rVert_{L^2}\lVertr\rVert_{L^\infty}
             \lVertr_{xx}\rVert_{L^2},
\end{align*}
and $H^1((0,1))\hookrightarrow L^\infty((0,1))$.  Adding the three
inequalities yields
\[
 \frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\lVertr\rVert_{H^2}^2
 \leq C\bigl(1+\lVertv\rVert_{H^3}\bigr)\lVertr\rVert_{H^2}^2,
\]
so Gronwall's inequality proves \eqref{eq:linear-transport-H2}.  Along the
flow $X_v$,
\[
 \frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}r(X_v(a,t),t)
 =-v_x(X_v(a,t),t)r(X_v(a,t),t),
\]
and solving this scalar ODE gives \eqref{eq:linear-transport-positive}.

We next give the parabolic estimates.  They may first be performed on a
Galerkin approximation.  Testing \eqref{eq:linear-parabolic} by $w_t$ and
using $w_t=0$ at the boundary gives the exact identity
\begin{align*}
 \int_0^1q|w_t|^2\mathop{}\!\mathrm d x
 +\frac12\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\int_0^1A|w_x|^2\mathop{}\!\mathrm d x
 =\int_0^1(f-qvw_x)w_t\mathop{}\!\mathrm d x
  +\frac12\int_0^1A_t|w_x|^2\mathop{}\!\mathrm d x.
\end{align*}
For any $\varepsilon>0$, the right-hand side is bounded by
\begin{align*}
 \varepsilon\lVertw_t\rVert_{L^2}^2
 +C_\varepsilon\bigl(\lVertf\rVert_{L^2}^2
       +\lVertv\rVert_{L^\infty}^2\lVertw_x\rVert_{L^2}^2\bigr)
 +\frac12\lVertA_t\rVert_{L^\infty}\lVertw_x\rVert_{L^2}^2.
\end{align*}
Since $A_t\in L^2H^1\hookrightarrow L^1L^\infty$ on the finite time
interval, choosing $\varepsilon<q_*/2$ and applying Gronwall's inequality
gives
\begin{equation}\label{eq:linear-first-energy}
 \sup_{t\leq\tau}\lVertw_x(t)\rVert_{L^2}^2
 +\int_0^\tau\lVertw_t\rVert_{L^2}^2\mathop{}\!\mathrm d t
 \leq C_{\rm lin}\left(\lVertw_0\rVert_{H^1}^2
                    +\lVertf\rVert_{L^2(0,\tau;L^2)}^2\right).
\end{equation}

To expose the estimate for $w_t$, differentiate
\eqref{eq:linear-parabolic} in time.  The resulting equation is
\begin{align*}
 qw_{tt}+qv w_{tx}-(Aw_{tx})_x
 ={}& f_t-q_tw_t-(q_tv+qv_t)w_x+(A_tw_x)_x.
\end{align*}
This identity is justified at the Galerkin level, or equivalently by time
difference quotients.  Pair it with $w_t\in H_0^1$.  The principal terms
satisfy
\begin{align*}
 \int_0^1q w_{tt}w_t\mathop{}\!\mathrm d x
 &=\frac12\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\int_0^1q|w_t|^2\mathop{}\!\mathrm d x
   -\frac12\int_0^1q_t|w_t|^2\mathop{}\!\mathrm d x,\\
 -\int_0^1(Aw_{tx})_xw_t\mathop{}\!\mathrm d x
 &=\int_0^1A|w_{tx}|^2\mathop{}\!\mathrm d x,
\end{align*}
whereas
\begin{align*}
 \left|\langle f_t,w_t\rangle\right|
 &\leq\varepsilon\lVertw_{tx}\rVert_{L^2}^2
      +C_\varepsilon\lVertf_t\rVert_{H^{-1}}^2,\\
 \left|\int_0^1A_tw_xw_{tx}\mathop{}\!\mathrm d x\right|
 &\leq\varepsilon\lVertw_{tx}\rVert_{L^2}^2
      +C_\varepsilon\lVertA_t\rVert_{L^\infty}^2\lVertw_x\rVert_{L^2}^2.
\end{align*}
The convection and coefficient-derivative terms are estimated in exactly
the same way, using
$q_t,v_t,A_t\in H^1\hookrightarrow L^\infty$ and the coefficient bounds;
more explicitly, their sum is at most
\begin{align*}
 \varepsilon\lVertw_{tx}\rVert_{L^2}^2
 +C_\varepsilon\bigl(1+\lVertq_t\rVert_{H^1}^2
     +\lVertv_t\rVert_{H^1}^2+\lVertA_t\rVert_{H^1}^2\bigr)
       \bigl(\lVertw_t\rVert_{L^2}^2+\lVertw_x\rVert_{L^2}^2\bigr).
\end{align*}
After choosing $\varepsilon$ so that these terms are absorbed by
$A_*\lVertw_{tx}\rVert_{L^2}^2$, integration and Gronwall's inequality, together
with \eqref{eq:linear-first-energy}, yield
\begin{align*}
 \sup_{t\leq\tau}\lVertw_t(t)\rVert_{L^2}^2
 +\int_0^\tau\lVertw_{tx}\rVert_{L^2}^2\mathop{}\!\mathrm d t
 \leq C_{\rm lin}\left[
  \lVertw_t(0)\rVert_{L^2}^2+\lVertf\rVert_{L^2(0,\tau;H^1)}^2
  +\lVertf_t\rVert_{L^2(0,\tau;H^{-1})}^2+\lVertw_0\rVert_{H^2}^2\right].
\end{align*}

It remains to recover the spatial derivatives without differentiating the
Dirichlet condition.  Put $h=qw_t+qvw_x-f$.  From
\[
 Aw_{xx}=h-A_xw_x
\]
and the estimates above, together with the standard implication
$f\in L^2H^1$, $f_t\in L^2H^{-1}\Rightarrow f\in C L^2$, we obtain
$w\in L^\infty H^2$.  Differentiating the same elliptic identity once gives
\[
 Aw_{xxx}=h_x-A_{xx}w_x-2A_xw_{xx}.
\]
Every term on the right belongs to $L^2(0,\tau;L^2)$: in particular,
$h_x$ is controlled by $w_{tx}$, $w_x$, $w_{xx}$, $f_x$, and the displayed
coefficient norms.  Division by $A\geq A_*$ therefore gives
$w\in L^2H^3$ and completes \eqref{eq:linear-parabolic-estimate}.

The estimates are uniform in the Galerkin dimension, so weak compactness
and the strong convergence of lower-order terms produce a solution.
The time-difference-quotient bounds give
$w_t\in C([0,\tau];L^2)$; the elliptic identity then gives
$w\in C([0,\tau];H^2)$.  Finally, applying
\eqref{eq:linear-first-energy} to the difference of two solutions with zero
initial data gives uniqueness.  This proves
\eqref{eq:linear-parabolic-regularity}--\eqref{eq:linear-parabolic-estimate}.

We use a Picard construction based on Lemma~\ref{lem:linear-local}; its
structure is the positive-density iteration introduced in the classical
compressible-flow theory~\cite{Tani1977}, but the estimates below are in the
Sobolev spaces of Definition~\ref{def:strong}.  Set
$\vartheta=\theta-\bar\theta$.  Starting from iterates having the same initial and
Dirichlet data, define
$\rho^{n+1}$ by
\[
 \rho^{n+1}_t+u^n\rho^{n+1}_x+\rho^{n+1}u_x^n=0
\]
and solve the two linear parabolic Dirichlet problems
\begin{align*}
 \rho^{n+1}(u^{n+1}_t+u^n u^{n+1}_x)
 -\partial_x\!\left(\mu(\theta^n)^\alpha u^{n+1}_x\right)
 &=-\partial_x(R\rho^{n+1}\theta^n),\\
 \rho^{n+1}(\theta^{n+1}_t+u^n\theta^{n+1}_x)
 -\partial_x\!\left(\kappa(\theta^n)^\beta\theta^{n+1}_x\right)
 &=-R\rho^{n+1}\theta^n u_x^n
   +\mu(\theta^n)^\alpha(u_x^n)^2 .
\end{align*}
The powers of $\theta^n$ in these two equations are frozen coefficients.
The characteristic formula is
\begin{equation}\label{eq:local-density-formula}
 \rho^{n+1}(X^n(a,t),t)=\rho_0(a)
 \exp\left(-\int_0^t u_x^n(X^n(a,s),s)\mathop{}\!\mathrm d s\right).
\end{equation}
Thus every iterate satisfies $\rho^{n+1}\geq\delta/2$ after $\tau$ is
chosen so that
$\int_0^\tau\lVertu_x^n\rVert_{L^\infty}\mathop{}\!\mathrm d t\leq\log2$.
Use the norm
\begin{align*}
 \lVert(\rho,u,\vartheta)\rVert_{\mathcal X_\tau}^2={}&
 \sup_{t\leq\tau}\lVert(\rho,u,\theta-\bar\theta)(t)\rVert_{H^2}^2
 +\sup_{t\leq\tau}\left[
   \lVert\rho_t(t)\rVert_{H^1}^2+\lVertu_t(t)\rVert_{L^2}^2
   +\lVert\vartheta_t(t)\rVert_{L^2}^2\right]\\
 &+\int_0^\tau\left[
  \lVert(u,\vartheta)\rVert_{H^3}^2
 +\lVert(u_t,\vartheta_t)\rVert_{H^1}^2\right]\mathop{}\!\mathrm d t .
\end{align*}
Let $B_L\subset\mathcal X_\tau$ be the closed ball determined by this norm,
the two Dirichlet conditions, and
$\rho\geq\delta/2$, $\theta\geq c_0/2$.  On $B_L$ the density multiplier and
the two diffusion coefficients are uniformly positive.  The product laws
in $H^1$ and $H^2$ show that the two parabolic right-hand sides belong to
$L^2H^1$ and their time derivatives to $L^2H^{-1}$.  Applying
Lemma~\ref{lem:linear-local} to the transport equation and then to the two
Dirichlet equations gives
\begin{equation}\label{eq:local-iteration}
 \lVert(\rho^{n+1},u^{n+1},\theta^{n+1}-\bar\theta)\rVert_{\mathcal X_\tau}^2
 \leq C_{\delta,c_0}(\text{data})
 +C_{\delta,c_0}\tau^{1/2}(1+L)^3 .
\end{equation}
Here the parabolic estimate is applied with homogeneous boundary data for
$u$ and $\vartheta$; it does not rely on discarding boundary terms after two
spatial differentiations.
Furthermore,
\[
 \sup_{t\leq\tau}\lVert\theta^{n+1}(t)-\theta_0\rVert_{L^\infty}
 \leq \tau^{1/2}\lVert\theta_t^{n+1}\rVert_{L^2(0,\tau;H^1)}.
\]
After $L$ is fixed, reducing $\tau$ makes this quantity at most $c_0/2$;
hence the temperature lower bound is also invariant under the iteration.
The compatibility conditions give the $L^2$ traces
$\sqrt{\rho_0}\dot u(0)=g_1$ and
$\sqrt{\rho_0}\dot\theta(0)=g_2$; since $\rho_0\geq\delta$, these are the
initial time-derivative quantities in
\eqref{eq:linear-parabolic-estimate}.  Choosing first
$L=2C_{\delta,c_0}(\text{data})$ and then $\tau$ so that the second term in
\eqref{eq:local-iteration} is at most $L/2$ makes the iteration invariant.
For completeness, subtract the equations for two consecutive iterates and
put
\begin{align*}
 d_n:={}&\sup_{0\leq t\leq\tau}
 \bigl(\lVertZ_\rho^n(t)\rVert_{L^2}^2+\lVertZ_u^n(t)\rVert_{L^2}^2
       +\lVertZ_\theta^n(t)\rVert_{L^2}^2\bigr)
 +\int_0^\tau\bigl(\lVertZ_{u,x}^n\rVert_{L^2}^2
                         +\lVertZ_{\theta,x}^n\rVert_{L^2}^2\bigr)\mathop{}\!\mathrm d t,
\end{align*}
where $Z^n$ is the difference of the $n$th and $(n-1)$st triples.
The density difference satisfies the explicit equation
\begin{equation}\label{eq:local-density-difference}
 Z_{\rho,t}^{n+1}+u^nZ_{\rho,x}^{n+1}+u_x^nZ_\rho^{n+1}
 =-Z_u^n\rho_x^n-\rho^nZ_{u,x}^n,
 \qquad Z_\rho^{n+1}(0)=0.
\end{equation}
Testing by $Z_\rho^{n+1}$, integrating the transport term by parts, and
using $H^1\hookrightarrow L^\infty$ gives, for every $\varepsilon>0$,
\begin{align}
 \frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\lVertZ_\rho^{n+1}\rVert_{L^2}^2
 \leq{}& C_L\lVertZ_\rho^{n+1}\rVert_{L^2}^2
 +\varepsilon\lVertZ_{u,x}^n\rVert_{L^2}^2
 +C_{L,\varepsilon}\lVertZ_u^n\rVert_{L^\infty}^2.
 \label{eq:local-density-difference-energy}
\end{align}

The momentum difference has the same positive principal part as the
$(n+1)$st equation and can be written as
\begin{align*}
 &\rho^{n+1}\bigl(Z_{u,t}^{n+1}+u^nZ_{u,x}^{n+1}\bigr)
 -\partial_x\bigl(\mu(\theta^n)^\alpha Z_{u,x}^{n+1}\bigr)\\
 &=-Z_\rho^{n+1}(u_t^n+u^{n-1}u_x^n)
   -\rho^{n+1}Z_u^n u_x^n
   +\partial_x\!\left(\mu\bigl[(\theta^n)^\alpha
                       -(\theta^{n-1})^\alpha\bigr]u_x^n\right)-\partial_x\!\left(RZ_\rho^{n+1}\theta^n
                         +R\rho^nZ_\theta^n\right).
\end{align*}

\begin{equation}\label{eq:local-power-difference}
 \lVert(\theta^n)^\gamma-(\theta^{n-1})^\gamma\rVert_{L^p}
 \leq C_{c_0,L,\gamma}\lVertZ_\theta^n\rVert_{L^p},
 \qquad \gamma\in\{\alpha,\beta\},\quad 2\leq p\leq\infty.
\end{equation}

continuity equation for $\rho^{n+1}$ and hence introduce no derivative of
the density.  The two principal diffusion terms give
\begin{align*}

                   -(\theta^{n-1})^\alpha\bigr]u_x^n

   +C_{L,\varepsilon}\lVertu_x^n\rVert_{L^\infty}^2

The term involving $Z_\rho^{n+1}$ is absorbed by Gronwall's inequality.

 \mathcal Z'+c_3(\mathcal D_0+\mathcal D_h)\leq0,
\end{equation*}
which is \eqref{eq:Z-differential}.

Let $c_Y,C_Y>0$ be the constants in \eqref{eq:Y-equivalence}.  Since
$\bar\theta^{-\alpha-1}\leq\bar\theta^{-1}$, we have
\begin{align*}
 \mathcal Z
 &\geq c_Y\mathcal A+c_Y\bar\theta^{-\alpha-1}\mathcal E
       +k_*\bar\theta^{-1}\mathcal E
 \geq\min\{c_Y,k_*\}(\mathcal A+\bar\theta^{-1}\mathcal E),\\
 \mathcal Z
 &\leq C_Y\mathcal A+C_Y\bar\theta^{-\alpha-1}\mathcal E
       +k_*\bar\theta^{-1}\mathcal E
 \leq\max\{C_Y,C_Y+k_*\}(\mathcal A+\bar\theta^{-1}\mathcal E).
\end{align*}
This proves \eqref{eq:Z-equivalence}, and in particular $\mathcal Z\geq0$.

The upper equivalence and Lemma~\ref{lem:decay-coercivity} give a fixed
$C_L>0$ such that
\[
 \mathcal Z(t)\leq C_L\bar\theta^{\alpha-1}
   \bigl(\mathcal D_0(t)+\mathcal D_h(t)\bigr).
\]
Thus
\[
 \mathcal D_0+\mathcal D_h\geq C_L^{-1}\bar\theta^{1-\alpha}\mathcal Z,
\]
and \eqref{eq:Z-differential} implies
\[
 \mathcal Z'+\lambda_0\bar\theta^{1-\alpha}\mathcal Z\leq0,
 \qquad \lambda_0:=c_3C_L^{-1}>0.
\]
Multiplying by
$\exp(\lambda_0\bar\theta^{1-\alpha}(t-1))$ and integrating from $1$ to $t$
proves \eqref{eq:Z-decay}.

Finally, integration of \eqref{eq:Z-differential} over $[t,T]$ yields
\[
 c_3\int_t^T(\mathcal D_0+\mathcal D_h)(s)\mathop{}\!\mathrm d s
 \leq\mathcal Z(t)-\mathcal Z(T)\leq\mathcal Z(t).
\]
Letting $T\to\infty$ by monotone convergence and then using
\eqref{eq:Z-decay} proves \eqref{eq:Z-tail}.

 \stackrel{*}{\rightharpoonup}\sqrt\rho\,\dot\theta
 \quad\text{in }L^\infty(1,T;L^2).
\end{align*}

 &\leq\liminf_{\delta\to0}
       \int_t^T(\mathcal D_0^\delta+\mathcal D_h^\delta)(s)\mathop{}\!\mathrm d s
 \leq C\exp\!\left(-c\bar\theta^{1-\alpha}(t-1)\right).
\end{align*}
Letting $T\to\infty$ and using monotone convergence proves

 \lVert\rho-m\rVert_{L^2}^2
 &\leq C\bar\theta^{-1}\mathcal E,\\

 \bar\theta^{\beta-2}\lVert\theta-\bar\theta\rVert_{H^1}^2
 &\leq C\bar\theta^{\beta-2}\lVert\theta_x\rVert_{L^2}^2\leq C\mathcal A.

For $1\leq p\leq2$, the finite measure of $(0,1)$ gives the same conclusion
with exponent $\lambda_{\rm d}/2$.  This proves all the relaxation
assertions of Theorem~\ref{thm:relaxation}.

\label{prop:vacuum}
Let $(\rho,u,\theta)$ be a solution furnished by
Theorem~\ref{thm:main} and let
$X_t(a,t)=u(X(a,t),t)$, $X(a,0)=a$.  Then
\begin{equation}\label{eq:flow-formula}
 X_a(a,t)=\exp\left(\int_0^t u_x(X(a,s),s)\mathop{}\!\mathrm d s\right)>0,

 \rho(X(a,t),t)X_a(a,t)=\rho_0(a).
\end{equation}

Fix an arbitrary $T<\infty$.  We first justify the flow map and its
invertibility on $[0,T]$.  By Definition~\ref{def:strong} and the
one-dimensional embedding $H^1(0,1)\hookrightarrow L^\infty(0,1)$,
\begin{equation*}
 L_T:=\int_0^T\lVertu_x(s)\rVert_{L^\infty}\mathop{}\!\mathrm d s
 \leq C\int_0^T\lVertu(s)\rVert_{H^2}\mathop{}\!\mathrm d s<\infty.
\end{equation*}
Thus $u(\cdot,t)$ is spatially Lipschitz with an integrable Lipschitz
constant.  The nonautonomous ODE defining $X$ consequently has a unique
solution for every $a\in[0,1]$.  Moreover, because the boundary condition
gives $u(0,t)=u(1,t)=0$, uniqueness implies
\begin{equation*}
 X(0,t)=0,\qquad X(1,t)=1,
 \qquad 0\leq t\leq T,
\end{equation*}
so every trajectory remains in $[0,1]$.

The same spatial regularity permits differentiation of the flow with
respect to its label.  The derivative $J(a,t):=X_a(a,t)$ satisfies the
variational equation
\begin{equation*}
 J_t(a,t)=u_x(X(a,t),t)J(a,t),
 \qquad J(a,0)=1.
\end{equation*}
Solving this scalar equation gives
\begin{equation*}
 X_a(a,t)=\exp\!\left(\int_0^t
 u_x(X(a,s),s)\mathop{}\!\mathrm d s\right).
\end{equation*}
In particular,
\begin{equation}\label{eq:flow-Jacobian-bounds}
 e^{-L_T}\leq X_a(a,t)\leq e^{L_T}
 \qquad (a\in[0,1],\ 0\leq t\leq T).
\end{equation}
Hence $a\mapsto X(a,t)$ is strictly increasing.  Since it fixes the two
endpoints, it maps $[0,1]$ bijectively onto itself.  Integrating
\eqref{eq:flow-Jacobian-bounds} between two labels gives
\begin{equation*}
 e^{-L_T}|a-b|\leq|X(a,t)-X(b,t)|
 \leq e^{L_T}|a-b|,
\end{equation*}
and the inverse map has the same type of Lipschitz bound.  Thus the flow is
bi-Lipschitz on every finite time interval.

It remains to identify the density transported by this flow.  The strong
solution regularity
$\rho\in C([0,T];H^2)$ and
$\rho_t\in C([0,T];H^1)$ justifies the chain rule along each trajectory.
Using the continuity equation, we obtain
\begin{align*}
 \frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\rho(X(a,t),t)
 &=\rho_t(X(a,t),t)+X_t(a,t)\rho_x(X(a,t),t)\\
 &=\rho_t+u\rho_x
 =-\rho u_x
 \qquad\text{at }(X(a,t),t).
\end{align*}
Combining this identity with the variational equation for $X_a$ yields
\begin{align*}
 \frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\bigl(\rho(X(a,t),t)X_a(a,t)\bigr)
 &=(-\rho u_x)X_a+\rho(u_xX_a)=0.
\end{align*}
Since $X(a,0)=a$ and $X_a(a,0)=1$, integration in time gives
\begin{equation*}
 \rho(X(a,t),t)X_a(a,t)=\rho_0(a),
\end{equation*}
which proves the second identity in \eqref{eq:flow-formula}.  Equivalently,
\begin{equation*}
 \rho(X(a,t),t)=\rho_0(a)
 \exp\!\left(-\int_0^t u_x(X(a,s),s)\mathop{}\!\mathrm d s\right).
\end{equation*}

Finally, let
$\mathcal V_0=\{a\in[0,1]:\rho_0(a)=0\}$ and
$\mathcal V(t)=\{x\in[0,1]:\rho(x,t)=0\}$.  Because $X_a>0$ and
$X(\cdot,t)$ is onto, the preceding identity shows both inclusions
\begin{equation*}
 \mathcal V(t)=X(\mathcal V_0,t).
\end{equation*}
Thus an initial vacuum point remains a vacuum point along its particle
path, while a particle starting with positive density cannot become vacuum
at any finite time.  This proves the asserted transport of the zero set.

Set $x_{\rm v}(t)=X(a_0,t)$.  Proposition~\ref{prop:vacuum} gives
$\rho(x_{\rm v}(t),t)=0$, which proves the first inequality in
\eqref{eq:gradient-growth}.  Put $f=\rho-m$.  Since
$\int_0^1f\mathop{}\!\mathrm d x=0$, integration by parts gives the exact representation
\[
 f(x_{\rm v})=\int_0^{x_{\rm v}}s f_x(s)\mathop{}\!\mathrm d s-\int_{x_{\rm v}}^1(1-s)f_x(s)\mathop{}\!\mathrm d s.
\]
Hence, by the Cauchy--Schwarz inequality,
\[
 m=|f(x_{\rm v})|\leq\left(\frac{x_{\rm v}^3+(1-x_{\rm v})^3}{3}\right)^{1/2}
 \lVert\rho_x(t)\rVert_{L^2}\leq\frac1{\sqrt{3}}\lVert\rho_x(t)\rVert_{L^2}.
\]
This proves the time-independent lower bound in
\eqref{eq:gradient-growth}.

Since $f$ is continuous and has zero mean, there is $z_t\in[0,1]$ such
that $f(z_t,t)=0$.  Therefore
\begin{equation*}
 m^2=\left|f^2(x_{\rm v}(t),t)-f^2(z_t,t)\right|
 \leq2\lVert\rho(t)-m\rVert_{L^2}\lVert\rho_x(t)\rVert_{L^2}.
\end{equation*}
The density part of \eqref{eq:low-order-decay} now yields
\[
 \lVert\rho_x(t)\rVert_{L^2}\geq \frac{m^2}{2C_{\rm d}^{1/2}}e^{\lambda_{\rm d}(t-1)/2}.
\]
By the definition of $c_{\rm v}$, this is precisely the exponential lower
bound in \eqref{eq:gradient-growth}.  Combining the two pointwise lower
bounds, squaring, and integrating proves
\eqref{eq:gradient-growth-integrated} and completes the proof of
Theorem~\ref{thm:relaxation}.
