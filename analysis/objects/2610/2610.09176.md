<!-- problem-1: body_md -->
\label{prob:gasull}
Are there smooth vector fields in $\mathbb{R}^{3}$ under the hypotheses of the
Markus--Yamabe conjecture and having periodic orbits?
<!-- end problem-1 -->
<!-- theorem-1: body_md -->
\label{main-theorem}
There exists an small open region $U$ in the square $(0,1/500)\times (0,1/6)$  such that for each pair $(\mu,e)\in U$ the corresponding polynomial vector field $\mathbf{F}_{\mu,e}(x,y,z)$ given in \eqref{campo-principal} is Hurwitz on $\mathbb{R}^3$, it has a unique singular point at the origin, which is locally asymptotically stable, and exactly two hyperbolic $2\pi$-periodic orbits
\[
\gamma_1:=\left\{(x,y,z)\in \mathbb{R}^3 \mid x^2+y^2=1,\ z=-\frac{1}{10}\right\},
\quad
\gamma_2:=\left\{(x,y,z) \in \mathbb{R}^3\mid x^2+y^2=4,\ z=-\frac{2}{5}\right\}.
\]
Moreover,  $\gamma_1$ is 
 of saddle type and $\gamma_2$ is locally asymptotically stable.
<!-- end theorem-1 -->
<!-- lemma-1: body_md -->
\label{propiedades-dinamicas}
Each polynomial vector field $\mathbf{F}_{\mu,e}(x,y,z)$ given in \eqref{campo-principal}, with  $0<\mu\leq 588/6575$, has a unique singular point at the origin, which is  locally asymptotically stable, and exactly two hyperbolic $2\pi$-periodic orbits
\[
\gamma_1:=\left\{(x,y,z)\in \mathbb{R}^3 \mid x^2+y^2=1,\ z=-\frac{1}{10}\right\},
\quad
\gamma_2:=\left\{(x,y,z) \in \mathbb{R}^3\mid x^2+y^2=4,\ z=-\frac{2}{5}\right\}.
\]
Moreover,  $\gamma_1$ is of saddle type and $\gamma_2$ is locally asymptotically stable.
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{invarianza-de-matriz-jacobiana}
The characteristic polynomial of the Jacobian matrix $D\mathbf{F}_{\mu,e}(x,y,z)$ of the polynomial vector field $\mathbf{F}_{\mu,e}(x,y,z)$ given in \eqref{campo-principal} is invariant under  rotations around the $z$-axis.
<!-- end lemma-2 -->
<!-- lemma-3: body_md -->
\label{Hurwitz-en-plano-xz}
Let $\mu=\mu_0=1/1000$ and $e=e_0=1/10$.  The Jacobian matrix $D\mathbf{F}_{\mu_0,e_0}(x,y,z)$ is Hurwitz for every point $(x,0,z)\in\mathbb{R}^3$.
<!-- end lemma-3 -->
<!-- proof-1: body_md -->
The uniqueness of the origin as singular point and the existence of the two periodic orbits $\gamma_1$ and $\gamma_2$, as well as their dynamical properties are proved by Lemma~\ref{propiedades-dinamicas}. Hence, it remains to prove the Hurwitz property of the vector field.

Lemma~\ref{invarianza-de-matriz-jacobiana} implies that it is enough to prove the Hurwitz property  for points of the form $(x,0,z)$.
The characteristic polynomial of $D\mathbf{F}_{\mu,e}(x,0,z)$ has the form
$$
P_{\mu,e}(\lambda)=\lambda^3+c_2\lambda^2
+c_1\lambda+c_0,
$$
where
\begin{align*}
c_2&=1+ \mu \left( 6\, \left( x^2-{\frac {125}{63}} \right) ^{2}+{\frac {800\,{z}
^{2}}{21}}-{\frac {18650}{1323}} \right), \\
c_1&=\frac{1}{100}\big( 2ex^2+g(x,z)\big)g(x,z)+\frac{\mu }{21}f_1(x,z)+\frac{5\mu^2}{441}f_2(x,z)f_3(x,z),\\
c_0&=\frac{1}{100}\big(g(x,z)\big)^2+\frac{5\mu^2 }{441}f_3(x,z)f_4(x,z),
\end{align*}
with 
\begin{align*}
g(x,z)&=ex^2+10ez+10,\\
f_1(x,z)&=126x^4-160x^2z-500x^2+800z^2+200,\\
f_2(x,z)&=21x^4-75x^2+80z^2+20,\\
f_3(x,z)&=21x^4-125x^2+400z^2+100,\\
f_4(x,z)&=21x^4-32x^2z-75x^2+80z^2+20.
\end{align*}
We prove the Hurwitz property of $D\mathbf{F}_{\mu,e}(x,0,z)$  by proving that $P_{\mu,e}(\lambda)$ is stable by using the Routh--Hurwitz criterion for cubic polynomials, that is, we will prove that $c_2>0, c_1>0, c_0>0,$ and $c_2c_1-c_0>0$. 
In fact, we need only to prove the inequalities 
$$
c_2>0,\ c_0>0, \quad \textrm{and} \quad d:=c_2c_1-c_0>0,
$$ 
because the last one immediately yields $c_1=(d+c_0)/c_2>0$.


\smallskip
A straightforward computation gives
$$
d=c_2c_1-c_0=d_0+\mu\Big(d_1+d_2\mu+d_3\mu^2\Big),
$$
where
\begin{align*}
d_0&=\frac{e}{50}x^2\big(g(x,z)\big)\\
d_1&=\frac{1}{21}\Bigg(400+400ez+
(200e^2+1600)z^2+1600ez^3+800e^2z^4
\\ &\qquad \qquad 
+(80e-1000)x^2
-20e(25e-16)z^2x^2+
(80e^2-1000e-160)zx^2
\\ &\qquad \qquad \quad 
+(6e^2-200e+252)x^4 +320e^2z^3x^2-4e(50e-63)zx^4\\
& \qquad \qquad \qquad 
+150e^2z^2x^4+\frac{252}{5}e^2zx^6-\frac{3}{5}e(25e-84)x^6+\frac{189}{50}e^2x^8\Bigg)\\
\end{align*}
\begin{align*}
d_2&=\frac{1}{441}\Bigg(
40000+320000z^2+640000z^4-64000z^3x^2-800000z^2x^2\\ 
&\qquad \qquad \quad
-16000zx^2-200000x^2+201600z^2x^4
+60000zx^4
\\ 
&\qquad \qquad \qquad 
+300400x^4-16800zx^6-
126000x^6+15876x^8\Bigg)\\
d_3&=\frac{10}{9261}\bigg(\Big(63x^4-250x^2+
400z^2+100\Big)f_2(x,z)f_3(x,z)\bigg)
\end{align*}

\medskip
We divide the proof into three steps. In Step 1 we prove that $c_2>0$ for suitable positive values of $\mu$. In Step 2 we prove that $c_0>0$ for suitable positive values of $\mu$ and $e$. Finally, according to the constraints we obtain 
for $\mu$ and $e$ in the first two steps, together with Lemma~\ref{Hurwitz-en-plano-xz}, we prove that $d>0$ for values in an open region $U\subset (0,1/500)\times(0,1/6)$ for which
all the roots of $P_{\mu,e}(\lambda)$ have negative real part.  

\medskip

{\bf Step 1.} The coefficient of $\mu$ in $c_2$ has two critical minima at $(\pm \sqrt{125/63},0)$ of the $xz$-plane, whose global minimum value is $-18650/1323$. Therefore, for $0<\mu<\mu_2^*:=1323/18650$, we get $c_2>0$ for any $(x,z)$.

\medskip
{\bf Step 2.}  The functions $f_3(x,z)$ and $f_4(x,z)$ can be written as
$$
f_3(x,z)=21\, \left( {x}^{2}-{\frac {125}{42}} \right) ^{2}+400\,{z}^{2}-{
\frac {7225}{84}}
$$
and
$$
f_4(x,z)=21\, \left( {x}^{2}-{\frac {16\,z}{21}}-{\frac {25}{14}} \right) ^{2}+
{\frac {1424}{21} \left( z-{\frac {75}{178}} \right) ^{2}}-{\frac {
21005}{356}}
$$
Thus, the global minimum values of $f_3(x,z)$ and $f_4(x,z)$ are $-7225/84$ and $-21005/356$, respectively. From the above expressions we get that  the real algebraic curves $\{f_3(x,z)=0\}$ and  $\{f_4(x,z)=0\}$ both are formed by two disjoint closed connected components (symmetric respect to the $z$-axis). Therefore, we look for rectangles $[x_3^{-},x_3^{+}]\times[z_3^{-},z_3^{+}]$ and $[x_4^{-},x_4^{+}]\times[z_4^{-},z_4^{+}]$ that contain the algebraic curves $\{f_3(x,z)=0\}$ and  $\{f_4(x,z)=0\}$, respectively.

The smallest and largest values of $x$ and $z$ in the semiplane $\{x\geq 0\}$ for the algebraic curves $\{f_3(x,z)=0\}$  are computed by using the real zeros of the resultants
$$
Res(f_3,(f_3)_z,z)=640000(21x^2-20)(x^2-5)
$$
and
$$
Res(f_3,(f_3)_x,x)=441000000(4z^2+1)(1344z^2-289)^2,
$$
respectively. Since $\sqrt{20/21}<\sqrt{5}<5/2$ and $\sqrt{289/1344}<1/2$ we obtain that $\{f_3(x,z)=0\}$ is contained in the closed rectangle $[-5/2,5/2]\times [-1/2,1/2]$. Moreover,  straightforward computations give that $f_3(x,z)$ restricted to such rectangle reaches the maximum value $3825/16$. {\bf Step 3.}
We regard $d=c_2c_1-c_0$ as a polynomial in $(x,z,\mu,e)$ and  $(\mu,e)$ varying over the compact rectangle $\mathcal{R}_2:=[1/2000,3/2000]\times[1/20,3/20]$, which contains
$(\mu_0,e_0)=(1/1000, 1/10)$ in its interior.
 Assign weight $1$ to $x$ and weight $2$ to $z$,
so that each monomial $x^{i}z^{j}$ in $d$ 
has weighted degree $i+2j$, and set $s:=x^{4}+z^{2}$. Thus, the weighted degrees of $d_0$,
$d_1$, $d_2$, and $d_3$ are $4$, $8$, $8$, and $12$, respectively. Hence, the highest weighted degree part of $d$ is
the part of $\mu^{3}d_3$ of weighted degree $12$, namely
$$
D_{12}:=\frac{10\mu^{3}}{9261}\left(63x^{4}+400z^{2}\right)
\left(21x^{4}+80z^{2}\right)
\left(21x^{4}+400z^{2}\right).
$$
Then, we can write
$$
d=D_{12}+R,
$$
where $R$ consists of the remaining monomials in $d$, each of which has degree at most four in $z$ and weighted degree at most 10.


On the one hand, the last three factors of $D_{12}$ are bounded below by $63s$, $21s$,
and $21s$, respectively. Thus, we have
\begin{equation}
    \label{primera-desigualdad}
D_{12}\geq 30\,\mu^3\, s^3\geq 
\bigg(\frac{30}{(2000)^3}\bigg)s^3.
\end{equation}
On the other hand,
Since $s=x^{4}+z^{2}$, $|x|\le s^{1/4}$ and $|z|\le s^{1/2}$. Thus, every monomial of weighted degree $w$ is
bounded by $s^{w/4}$. Hence, from the definition of $R$ we get that 
\begin{equation}
    \label{segunda-desigualdad}
    |R|\leq \sum_{j=0}^4\sum_{i=0}^{10-2j}\big{|}C_{ij}(\mu,e)\big{|}|x|^i|z|^j \leq C s^{5/2}, 
    \quad \mbox{for $s\geq 1$.}
\end{equation}
where $C$ is the maximum over $\mathcal{R}_2$ of the sum of the absolute values of the
coefficients of $R$, which satisfies 
$$
C\geq \big{|}C_{00}(\mu,e)\big{|} =\frac{400\,\mu}{21}+\frac{
40000\,{\mu}^{2}}{441}+\frac{2000000\,{\mu}^{3}}{9261}\geq \frac{353641}{37044000}.
$$

Combining the inequalities \eqref{primera-desigualdad} and \eqref{segunda-desigualdad}, 
for each $(\mu,e)\in \mathcal{R}_2$, we have 
$$
d\ \geq \ s^{5/2}\bigg(\frac{30}{(2000)^{3}}\,s^{1/2}-C\bigg)>0
\qquad\text{whenever }
s>s_0:=\bigg(\frac{(2000)^3 C}{30}\bigg)^{2}
 \geq 1.
$$

On the compact set $K:=\{s\le s_0\}$, the proof of
Lemma~3 gives $\delta:=\min_{K}d(\,\cdot\,;\mu_0,e_0)>0$, and by uniform
continuity of $d$ on $K\times P$ there is an open neighbourhood
$U\subset\operatorname{int} \mathcal{R}_2$ of $(\mu_0,e_0)$ with $d>\delta/2$ on $K\times U$.
Therefore, we have proved that $d>0$ on $\mathbb{R}^{2}\times U$. 


In summary, from Lemma~\ref{invarianza-de-matriz-jacobiana}, Steps 1, 2 and 3, the Routh--Hurwitz criterion shows that $DF_{\mu,e}(x,y,z)$ is
Hurwitz on $\mathbb{R}^{3}$ for every $(\mu,e)\in U$.
