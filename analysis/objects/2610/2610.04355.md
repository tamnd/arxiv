<!-- theorem-1: body_md -->
\label{thm:glob_err}
Consider the initial value problem for a non-autonomous evolution equation 
\begin{equation*}
\begin{cases}
u'(t) = F(t) \, u(t) = A(t) \, u(t) + B(t) \, u(t) + C(t) \, u(t)\,, \quad t \in [t_0, T]\,, \\
u(t_0) \, \text{ given}\,, 
\end{cases}
\end{equation*}
and assume that the linear operators $A(t): D_A \to X$, $B(t): D_B \to X$, and $C(t): D_C \to X$, $t \in [t_0, T]$, are defined on certain subspaces $D_A, D_B, D_C \subseteq X$ of the underlying Banach space $(X, \|\cdot\|_X)$ with non-empty intersection.
Suppose that the associated evolution operators are well-defined and in particular satisfy stability bounds of the form 
\begin{equation*}
\begin{gathered}
                                                                                 
                                                                                 
                                                                                 
                                                                              
\|E_G(\tau;\cdot)\|_{X \leftarrow X} \leq {\rm e}^{\, \text{Const} \, \tau}\,, \quad 
G\in\{F,\,A,\,B,\,C\}\,, \quad 
\tau \in [0, T - t_0]\,, 
\end{gathered}
\end{equation*}
with generic constants, uniformly in the second and third arguments, respectively. 
Assume further that the prescribed initial value and hence the exact solution values are bounded
\begin{equation*}
\|u(t)\|_D \leq \text{Const}\,, \quad t \in [t_0, T]\,, 
\end{equation*}
with respect to the norm of a suitably restricted subspace $D \subseteq X$ that is defined by the requirement that the commutators arising in the expansion of the local error remain bounded
\begin{equation*}
\big\|[A(s),B(t)]\big\|_{X \leftarrow D}
+ \big\|[A(s),C(t)]\big\|_{X \leftarrow D} + \big\|[B(s),C(t)]\big\|_{X \leftarrow D}\leq \text{Const}\,, \quad s, t \in [t_0, T]\,,  
\end{equation*}
and that also their compositions with evolution operators associated with the original problem and the corresponding subproblems remain bounded. 
Under these general requirements, the proposed first-order time integration method based on the Lie--Trotter splitting retains its classical order and satisfies the global error estimate 
\begin{equation*}
\big\|u_n - u(t_n)\big\|_X \leq \text{Const} \, \max_{\nu \in \{0, \dots, n-1\}} \tau_{\nu}\,, \quad
n \in \{1, \dots, N\}\,,
\end{equation*}
where the arising constant is independent of the time increment and the number of time steps.

<!-- remark-1: body_md -->
\label{rem:differentiability}
For the first approach, a sufficient abstract hypothesis for the differentiability statements above, on Banach spaces $Y \hookrightarrow X$ ($Y=X$ for pure multiplication operators), is
\begin{equation}\label{ass:timereg_abstract}
A,\, B,\, C \in W^{1,\infty}\big(0,T;\mathcal{L}(Y,X)\big)\,.
\end{equation}
Indeed, \eqref{ass:timereg_abstract} directly yields, for $s,t\in[0,T]$,
\begin{equation*}
\|A(t)-A(s)\|_{\mathcal{L}(Y,X)} \le M_A \, |t-s|\,, \quad \|A'(t)\|_{\mathcal{L}(Y,X)} \le M_A\,,
\end{equation*}
and analogously for $B, C$ (with constants $M_B, M_C$), which is precisely the differentiability and boundedness required above. For the Bloch--Torrey equation (Section~\ref{sec:BT}), Hypothesis~\eqref{ass:timereg_abstract} is verified concretely in terms of the physical coefficients $\vec{\mathrm{b}}, \vec{\mathrm{v}}$ in Proposition~\ref{prop:timereg_BT}, where in fact $B' \equiv 0$ exactly, since $B$ does not depend on time there.
For the second approach, since $F(t) = A(t) + B(t) + C(t)$, Hypothesis~\eqref{ass:timereg_abstract} gives $F' = A' + B' + C'$ bounded; for the Bloch--Torrey equation, where $B' \equiv 0$, this bound is made explicit at the end of Proposition~\ref{prop:timereg_BT}.

\label{prop:timereg_BT}
For
\begin{equation}\label{ass:timereg_BT}
\vec{\mathrm{b}}\in W^{1,\infty}(0,T;L^\infty(\Omega;\mathbb{R}^3)),\qquad \vec{\mathrm{v}}\in W^{1,\infty}(0,T;L^\infty(\Omega;\mathbb{R}^3)),
\end{equation}
there is a constant $C_A$, depending only on $\Omega$, such that
\begin{align*}
\|A(t)-A(s)\|_{L^2(\Omega;\mathbb{R}^4)\leftarrow L^2(\Omega;\mathbb{R}^4)} &\le C_A\gamma\,|t-s|\,\|\partial_{t}\vec{\mathrm{b}}\|_{L^\infty(0,T;L^\infty(\Omega))}\,,\\
\|A'(t)\|_{L^2(\Omega;\mathbb{R}^4)\leftarrow L^2(\Omega;\mathbb{R}^4)} &\le C_A\gamma\,\|\partial_{t}\vec{\mathrm{b}}(t)\|_{L^\infty(\Omega)}\,,\\
\|C(t)-C(s)\|_{L^2(\Omega;\mathbb{R}^4)\leftarrow H^1(\Omega;\mathbb{R}^3)\times L^2(\Omega)} &\le |t-s|\,\|\partial_{t}\vec{\mathrm{v}}\|_{L^\infty(0,T;L^\infty(\Omega))}\,,\\
\|C'(t)\|_{L^2(\Omega;\mathbb{R}^4)\leftarrow H^1(\Omega;\mathbb{R}^3)\times L^2(\Omega)} &\le \|\partial_{t}\vec{\mathrm{v}}(t)\|_{L^\infty(\Omega)}\,.
\end{align*}
Here the stationary diffusion part $C_{\text{diff}}$ cancels exactly in $C(t)-C(s)$ and in $C'(t)$; boundedness of the full operator $C(t)$ from $H^1(\Omega;\mathbb{R}^3)\times L^2(\Omega)$ to $L^2(\Omega;\mathbb{R}^4)$ is \emph{not} claimed here.
Since $R_1,R_2$ depend only on $x$, $B$ is autonomous: $B(t)-B(s)=0$ and $B'\equiv0$ exactly, with no regularity assumption needed.

Consequently, for $t\in[t_n,t_n+\tau_n]$, the differences arising in Approach~1 satisfy
\[
\Big\|A\big(t_n+\tfrac13(t-t_n)\big)-A(t)\Big\|_{L^2(\Omega;\mathbb{R}^4)\leftarrow L^2(\Omega;\mathbb{R}^4)} \le \tfrac23\,C_A\gamma\,\tau_n\,\|\partial_{t}\vec{\mathrm{b}}\|_{L^\infty_tL^\infty_x},
\]
\[
\Big\|C\big(t_n+\tfrac23\tau_n+\tfrac13(t-t_n)\big)-C(t)\Big\|_{L^2(\Omega;\mathbb{R}^4)\leftarrow H^1(\Omega;\mathbb{R}^3)\times L^2(\Omega)} \le \tfrac23\,\tau_n\,\|\partial_{t}\vec{\mathrm{v}}\|_{L^\infty_tL^\infty_x},
\]
while the corresponding difference for $B$ vanishes identically. For Approach~2, since $B'=0$, $F'(t)=A'(t)+C'(t)$ satisfies
\[
\|F'(t)\|_{L^2(\Omega;\mathbb{R}^4)\leftarrow H^1(\Omega;\mathbb{R}^3)\times L^2(\Omega)} \le C_A\gamma\,\|\partial_{t}\vec{\mathrm{b}}(t)\|_{L^\infty(\Omega)} + \|\partial_{t}\vec{\mathrm{v}}(t)\|_{L^\infty(\Omega)}.
\]
