<!-- lemma-2: body_md -->
\label{lem:has-even-stretch}

	Every $v\in B(A)$ has an even stretch in $A$.
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
Let $M(v)\cap A = \{a_{i_1},\dots,a_{i_k}\}$ where $i_1< i_2 < \dots < i_k$ and note that $k \ge 3$.
	For $1\le j<k$, let $A_j=a_{i_j}\hbox{-} a_{i_j+1}\hbox{-} \dots \hbox{-} a_{i_{j+1}}$ and let $A_k=a_{i_k}\hbox{-} a_{i_k+1}\hbox{-} \dots \hbox{-} a_{i_1}$.
	Note that each $A_j$ is a stretch of $v$.
	Observe that $A=\cup_{i=1}^k A_i$ and that $\sum_{i=1}^k |A_i| = |A|+k$ because every vertex in $M(v)\cap A$ is counted twice.
	Since $|A|$ is odd, it follows that one of the stretches is even.
	This completes the proof.

<!-- lemma-3: body_md -->
\label{lem:every-s-is-small}
	
	Every vertex $s\in S$ is small for $A$.
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
Suppose that $s\in S$ is big.
	Then, by Lemma \ref{lem:has-even-stretch}, $s$ has an even stretch $P$ in $A$.
	Since $\alpha(S) = 2$, $s$ does not have two consecutive non-neighbors in $A$.
	Together with the fact that $s$ is big, it follows that $2 < |P| \le |A|-3$.
	Let $A' = P \cup \{s\}$.
	Then, $A'$ is an odd antihole in $S$ shorter than $A$, which is a contradiction.
	This completes the proof.

<!-- lemma-4: body_md -->
\label{lem:at-most-6-small}
	
	Let $L$ be an $S$-full component of $G\setminus S$.
	Then, there exists a subset $L'\subseteq L$ such that $G[L']$ is connected, $L'$ is full to $A$, and such that $L'$ contains at most 6 small vertices.
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
Let $L' \subseteq L$ be connected and full to $A$, such that $|L'|$ is as small as possible.
	Suppose first that $L'$ is a path.
	Let $l_1,l_2$ be the ends of $L'$.
	By the minimality of $|L'|$, for $i \in \{1,2\}$ there exists $a(i) \in A$ such that $l_i$ is the unique neighbor of $a(i)$ in $L'$.
	It follows that for every vertex $v$ in $(L')^*$ that is small for $A$, $M_A(v)\subseteq \{a(1),a(2)\}$.
	Since $A$ is odd, there exist two consecutive vertices $a_k,a_{k+1}$ that are complete to all the small vertices in $(L')^*$.
	Suppose that $(L')^*$ contains at least 5 small vertices.
	Then, 3 of them form a stable set which together with $\{a_k,a_{k+1}\}$ forms a $K_{2,3}$ in $G$, contrary to the fact that $G\in \mathcal G_2'$.
	Thus, $L'$ contains at most 6 small vertices as required.

	Now, we may assume that $L'$ is not a path, and so there exist three distinct vertices $l_1,l_2,l_3 \in L'$ such that for each $i$, $L' \setminus l_i$ is connected.
	By the minimality of $|L'|$, for each $i$, there is $a(i) \in A$ such that $l_i$ is the unique neighbor of $a(i)$ in $L'$.
	It follows that the vertices $a(1)$, $a(2)$ and $a(3)$ are all distinct and every vertex of $L' \setminus \{l_1,l_2,l_3\}$ is anticomplete to $\{a(1), a(2),a(3)\}$.
	Consequently, no vertex of $L' \setminus \{l_1,l_2,l_3\}$ is small, and the assertion of the lemma holds.

<!-- lemma-5: body_md -->
\label{lem:long-stretch}

	Suppose that $b\in B(A)$ has a long stretch $P=\{a_i,\dots,a_j\}$ in $A$ where $i<j$.
	Then, every other stretch of $b$ has exactly 3 vertices.
<!-- end lemma-5 -->
<!-- proof-4: body_md -->
First, suppose that $b$ is anticomplete to two consecutive vertices $a_k,a_{k+1}$.
	Note that in that case $X=\{b,a_k,a_{k+1}\}$ is a stable set of size 3.
	Since $b$ is big, $|X\cap P|\le 1$.
	Hence, there exist $a_p,a_{p+1}\in P\setminus X$ that are complete to $X$.
	This contradicts $G$ being $K_{2,3}$-free.

	Thus, $b$ is adjacent to at least one of $a_k,a_{k+1}$ for every $k \in \mathbb N$.
	By Lemma \ref{lem:has-even-stretch}, $b$ has an even stretch in $A$.
	Let $E=\{a_q,\dots,a_w\}$ be a shortest even stretch of $b$ in $A$ and let $A' = E\cup \{b\}$.
	Suppose that $b$ has a long stretch $P'$ in $A\setminus E^*$.
	Since $b$ is big for $A$, $A'$ is shorter than $A$ and $|E\cap P'|\le 1$.
	Hence, there exist $a_p,a_{p+1}\in P'\setminus E$ that are complete to $A'$.
	Then, there exists a minimal $a_p\hbox{-} a_{p+1}$-separator $S'$ in $G$ that contains $A'$, contrary to $S$ being a shortest odd minimal separator in $G$.
	It follows that $E=P$ and by minimality of $E$ every stretch of $b$ in $A\setminus P^*$ has size at most 3.
	Since $b$ has no two consecutive non-neighbors in $A$, every other stretch has size exactly three.
	This completes the proof.

<!-- lemma-6: body_md -->
\label{lem:cover-big-vertices}
	
	Let $B\subseteq B(A)$.
	Then, one of the following holds:
	\begin{enumerate}
		\item There exist two vertices $b_1,b_2\in B$ such that $A\subseteq N(\{b_1,b_2\})$.
		\item There exists a set $A'\subseteq A$ of size at most 22 such that $B\subseteq B(A')$.
	\end{enumerate}
<!-- end lemma-6 -->
<!-- proof-5: body_md -->
We may assume $B$ is non-empty.
	Let $b \in B$ be a vertex with the longest stretch in $A$, say $\{a_i,\ldots,a_j\}$ where $i<j$.
	Suppose that some vertex $b' \in B$ is complete to $\{a_{i-1},a_i,a_{i+1},a_{j-1},a_j,a_{j+1}\}$.
	By Lemma \ref{lem:long-stretch}, it follows that either $b'$ has a longer stretch than $b$, which is a contradiction, or outcome (1) holds.

	Therefore, we may assume that every vertex $b' \in B$ is non-adjacent to one of $a_{i-1},a_i,a_{i+1},\allowbreak a_{j-1},a_j,a_{j+1}$.
	Let $A'=\{a_{i-5},\ldots,a_{i+5},\allowbreak a_{j-5}, \ldots,a_{j+5}\}$.
	We show that every $b'\in B$ is big for $A'$.
	If $b'$ has a long stretch, then this follows from Lemma \ref{lem:long-stretch}.
	Otherwise, every stretch of $b'$ is of size at most $4$, and hence $b'$ is also big for $A'$.
	Thus, outcome (2) holds.
	This completes the proof.

<!-- lemma-7: body_md -->
\label{lem:subgraph-separator}
	
	Let $G$ be a graph, let $G'$ be an induced subgraph of $G$, and let $S'$ be a minimal separator in $G'$.
	Then there exists a minimal separator $S$ in $G$ such that $S' \subseteq S$.
<!-- end lemma-7 -->
<!-- proof-6: body_md -->
First, we describe the algorithm.
	
	Applying the algorithm from Theorem \ref{thm:k23-im}, we decide if $G$ is $K_{2,3}$-induced-minor-free.
	If not, we output ``$G\notin \mathcal G_2$''.
	Next, let $\mathcal A$ be the set of subsets of $V(G)$ of size at most 22.
	For each 4-tuple $(l_1,l_2,r_1,r_2)$ of vertices of $G$, add a shortest odd antihole $A$ of $N(\{l_1,l_2\})\cap N(\{r_1,r_2\})$, if one exists, to $\mathcal A$.
	This is possible by Theorem \ref{thm:find-shortest-odd}.
	For each $A\in \mathcal A$ and each set $L\subseteq V(G)\setminus A$ of size at most 6 do the following:
	\begin{enumerate}
		\item Mark each vertex of $V(G)\setminus A$ as big or small for $A$.
		\item For each component $C_1$ of $G[B(A)\cup L]$ and each component $C_2$ of $G\setminus N[C_1]$, test whether $N(C_1)\cap N(C_2)$ contains an odd antihole and if so, output ``$G\notin \mathcal G_2$''.
	\end{enumerate}
	If the algorithm has not yet output ``$G\notin \mathcal G_2$'', then output ``$G\in \mathcal G_2$''.
	
	\
	
	{\em Proof of correctness:}
	
	\
	
	It is clear that the algorithm runs in polynomial time.
	By Lemma \ref{lem:separator-alpha-2}, if $G\in \mathcal G_2$ then $G$ is $K_{2,3}$-induced-minor-free.
	Thus, if the algorithm detects a $K_{2,3}$ induced minor, the output ``$G\notin \mathcal G_2$'' is correct.
	Note also that every other time our algorithm outputs ``$G\notin \mathcal G_2$'', it finds an odd minimal separator in an induced subgraph of $G$.
	Lemma \ref{lem:subgraph-separator} implies that the output ``$G\notin \mathcal G_2$'' is always correct.
	Therefore, it suffices to show that if $G$ is $K_{2,3}$-induced-minor-free and $G\notin \mathcal G_2$, then our algorithm outputs ``$G\notin \mathcal G_2$''.
	
	Thus, suppose that $G$ is $K_{2,3}$-induced-minor-free but not in $\mathcal G_2$.
	Then, there exists a minimal separator $S$ in $G$ that is not bisimplicial.
	This means that $G[S]^C$ is not bipartite and therefore contains an odd cycle.
	Let $C$ be a shortest odd induced cycle in $G[S]^C$.
	As $\alpha(S)\le 2$, $G[S]^C$ is triangle-free and therefore $C$ is an odd hole.
	Thus, $G$ contains an odd minimal separator.
	Our goal now is to detect an odd minimal separator in $G$.
	
	Let $\tilde S$ be a shortest odd minimal separator in $G$, and let $\tilde L,\tilde R$ be two $\tilde S$-full components of $G\setminus \tilde S$.
	We claim that:
	
	\medskip\medskip\refstepcounter{tbox}\noindent{\parbox{\textwidth}{(\thetbox) {\em \label{sta:core-for-big}
	There exists a set $A\in \mathcal A$ and a shortest odd antihole $\tilde A\supseteq A$ in $\tilde S$ such that either $\tilde L \cap B(\tilde A)\subseteq B(A)$ or $\tilde R \cap B(\tilde A)\subseteq B(A)$.}}}\vspace*{0.3cm}
	
	Suppose that there exist $l_1,l_2\in \tilde L$ and $r_1,r_2\in \tilde R$ such that $X=N(\{l_1,l_2\})\cap N(\{r_1,r_2\})$ contains a shortest odd antihole of $\tilde S$.
	Then, our algorithm will detect such a 4-tuple $(l_1,l_2,r_1,r_2)$ and will add a shortest odd antihole $A$ of $X$ to $\mathcal A$.
	Note that $A$ is then also a shortest odd antihole of $\tilde S$.
	Taking $\tilde A=A$, the claim follows.
	Otherwise, by possibly interchanging $\tilde L$ and $\tilde R$, we may assume that there exists a shortest odd antihole $\tilde A$ of $\tilde S$ such that $\tilde A\not \subseteq N(\{l_1,l_2\})$ for every pair $l_1,l_2\in \tilde L$.
	Then, Lemma \ref{lem:cover-big-vertices} implies that there exists a set $A\subseteq \tilde A$ of size at most 22, such that $\tilde L \cap B(\tilde A)\subseteq B(A)$.
	This completes the proof of \eqref{sta:core-for-big}.
	
	Let $A\in \mathcal A$ be a set and $\tilde A\supseteq A$ be a shortest odd antihole in $\tilde S$ as in \eqref{sta:core-for-big}.
	By possibly interchanging $\tilde L$ and $\tilde R$, we may assume that $\tilde L \cap B(\tilde A)\subseteq B(A)$.
	By Lemma \ref{lem:every-s-is-small}, every $s\in \tilde S$ is small for $\tilde A$ and hence also for $A$.
	Therefore, $B(A)$ is disjoint from $\tilde S$.
	Lemma \ref{lem:at-most-6-small} implies that in some iteration, our algorithm will find a connected set $C_1\subseteq \tilde L$ that is full to $\tilde A$.
	Hence, it will also find $C_2\supseteq \tilde R$ such that $\tilde A\subseteq N(C_1)\cap N(C_2)$ and will output ``$G\notin \mathcal G_2$'' as required.
	This completes the proof of correctness.

\label{lem:dominating-subset}

	Then, there exists a non-edge $e\in \overline E(S)$, a vertex $x\in X$, and a component $X'$ of $G[M(e)\cup \{x\}]$ such that $X'\subseteq X$ and $N(X') \cap e' \neq \emptyset$ for every $e'\in \overline E(S)$.

For $e\in \overline E(S)$ and $x\in X$, let $C(e,x)$ denote the component of $G[M(e)\cup \{x\}]$ containing $x$.

	Suppose that there exists $e'\in \overline E(S)$ such that $N(C(e,x))\cap e' = \emptyset$.

<!-- lemma-9: body_md -->
\label{lem:two-cliques}
	
	Let $S\subseteq V(G)$ be a set and let $X_1,X_2\subseteq V(G)\setminus S$ be such that for each $i\in\{1,2\}$ and every $e\in \overline E(S)$, $N(X_i)\cap e \neq \emptyset$.
	Then, $S\setminus(N(X_1)\cap N(X_2))$ is bisimplicial.
<!-- end lemma-9 -->
<!-- proof-8: body_md -->
Define \begin{itemize}
		\item $S' = N(X_1)\cap N(X_2)$,
		\item $P = (S \cap N(X_1)) \setminus N(X_2)$, and
		\item $Q = S \setminus (S'\cup P)$.
	\end{itemize}
	We claim that $P$ and $Q$ are both cliques.
	Suppose that $e\in \overline E(S)$ is contained in $P$.
	Then, $N(X_2)\cap e = \emptyset$, which is a contradiction.
	Otherwise, suppose that $e$ is contained in $Q$.
	Observe that $S\cap N(X_1) \subseteq S'\cup P$.
	Since $N(X_1) \cap e \neq \emptyset$ and $Q$ is disjoint from $S'\cup P$, the claim follows.
	It is straightforward to observe that $P\cup Q = S\setminus S'$.
<!-- end proof-8 -->
<!-- proof-9: body_md -->
First, we describe the algorithm.
	
	For $e\in \overline E(G)$ and $x\in V(G)$, let $C(e,x)$ denote the component of $G[M(e)\cup \{x\}]$ containing $x$.
	Let $\mathcal C = \{\emptyset\}\cup \{C(e,x)\mid e\in \overline E(G), x\in V(G)\}$.
	For every pair $C_1,C_2\in \mathcal C$, define $S=N(C_1)\cap N(C_2)$ and do the following.
	Let \[D = \{d_{xy}\mid xy\in \overline E(G\setminus S) \text{ such that } \alpha(S\cup \{x,y\})\ge 3 \}\] be a set of new vertices.
	Define a new graph $G'$ with vertex set $(V(G)\setminus S)\cup D$ and edges as follows:
	\begin{itemize}
		\item $G'[V(G)\setminus S]=G\setminus S$, and
		\item $N(d_{xy}) = V(G')\setminus \{x,y,d_{xy}\}$ for every $d_{xy}\in D$.
	\end{itemize}
	We create an instance $\mathcal I=(U,V,P,Q)$ of the list 2-clique cutset problem for $G'$ in the following way.
	Let $X=\{x\in V(G) \mid \alpha(S\cup \{x\})\ge 3\}$ and set
	\begin{itemize}
		\item $U = V(G')\setminus (\{v\} \cup D)$,
		\item $V = V(G')\setminus (\{u\} \cup D)$,
		\item $P = Q = V(G')\setminus (X\cup \{u,v\})$.
	\end{itemize}
	Suppose that $\mathcal I$ admits a solution $(U',V',P',Q')$.
	Then, set $S'= (S\cup P'\cup Q')\cap V(G)$.
	Check if $S'$ is a triad-free $u\hbox{-} v$ separator in $G$, and if so then output it.
	If after all iterations no triad-free $u\hbox{-} v$ separator in $G$ has been found, then output ``no solution exists''.
	
	\
	
	{\it Proof of correctness:}
	
	\
	
	Computing the set $\mathcal C$ takes polynomial time since there are at most $O(|V(G)|^3)$ pairs $(e,x)$ for which we need to compute $C(e,x)$.
	By Theorem \ref{thm:list-2-clique-cutset}, constructing an instance $\mathcal I$ and solving it also takes polynomial time.
	Lastly, checking that $S'$ is a triad-free $u\hbox{-} v$ separator can also be done in polynomial time.
	
	It is clear that if our algorithm outputs a set $S'$, then it is a triad-free $u\hbox{-} v$ separator.
	Hence, it suffices to show that if there exists a triad-free $u\hbox{-} v$ separator, then our algorithm will find it.
	
	Thus, suppose that there exists a triad-free $u\hbox{-} v$ separator.
	Note that then there exists a minimal triad-free $u\hbox{-} v$ separator $\tilde S$.
	Suppose that $\tilde S$ is a clique.
	Then, for $C_1=C_2=\emptyset$ our algorithm will set $S=D=\emptyset$, meaning $G'=G$.
	Thus, the created instance $\mathcal I$ of the list 2-clique cutset problem will admit a solution $(\tilde U,\tilde V,\tilde S,\emptyset)$ for some sets $\tilde U,\tilde V$ with $u\in \tilde U,v\in \tilde V$.
	By Theorem \ref{thm:list-2-clique-cutset}, the list 2-clique cutset algorithm will find a solution $(U',V',P',Q')$ to $\mathcal I$ and thus a bisimplicial $u\hbox{-} v$ separator $P'\cup Q'$.
	
	Therefore, we may assume that $\alpha(\tilde S)=2$.
	Let $\tilde U,\tilde V$ be the $\tilde S$-full components of $G\setminus \tilde S$ containing $u$ and $v$, respectively.
	Let $\tilde U'$ be the union of all components of $G\setminus (\tilde S\cup \tilde V)$.
	By Lemma \ref{lem:dominating-subset}, $\mathcal C$ contains sets $C_1\subseteq \tilde U$ and $C_2\subseteq \tilde V$ such that for each $C_i$ and every non-edge $e\in \overline E(\tilde S)$, $N(C_i)\cap e\neq \emptyset$.
	Let $S=N(C_1)\cap N(C_2)$.
	By Lemma \ref{lem:two-cliques}, we get that $\tilde S\setminus S$ is bisimplicial.
	Observe that $D$ is a clique in $G'$ such that each $d\in D$ has at most one non-neighbor in $\tilde S\setminus S$.
	Thus, the created instance $\mathcal I$ of the list 2-clique cutset problem admits a solution $(\tilde U',\tilde V, \tilde P, \tilde Q)$ for some cliques $\tilde P,\tilde Q$ such that $\tilde P \cup \tilde Q = (\tilde S\setminus S)\cup D$.
	Let $(U',V',P',Q')$ be a solution to $\mathcal I$ that our algorithm found.
	By the way we defined the instance, we get that $u\in U',v\in V'$ and that $U'$ is anticomplete to $V'$.
	It follows that $S'=(S\cup P'\cup Q')\cap V(G)$ is a $u\hbox{-} v$ separator in $G$.
	Now, suppose that $S'$ contains a triad $\{a,b,c\}$.
	Since $P'$ and $Q'$ are both cliques and are disjoint from $X=\{x\in V(G) \mid \alpha(S\cup \{x\})\ge 3\}$, it follows that (after possibly relabeling $a,b,c$) $a\in S,b\in P',c\in Q'$.
	Note that $D\subseteq P'\cup Q'$ follows by our definition of $\mathcal I$.
	But then $\{d_{bc},b,c\}$ is a triad in $P'\cup Q'$, which is a contradiction.
	It follows that $S'$ is a triad-free $u\hbox{-} v$ separator in $G$.
	This completes the proof.

We will show a reduction to the problem of finding a $u\hbox{-} v$ triad-free separator.
	Let $G$ be a graph and let $\mathcal I=(U,V,S)$ be an instance of the list triad-free cutset problem.
	Let $X=V(G)\setminus S$ and let $A=\{a^1_x,a^2_x\mid x\in X\}$.
	Obtain $G'$ from $G$ by adding the vertices $\{u,v\}\cup A$ such that
	\begin{itemize}
		\item $N(u)=V(G')\setminus (V\cup \{u,v\})$,
		\item $N(v)=V(G')\setminus (U\cup \{u,v\})$, and
		\item $N(a^1_x)=N(a^2_x)=V(G')\setminus \{x,a^1_x,a^2_x\}$ for every $x\in X$.
	\end{itemize}
	Use the algorithm from Theorem \ref{thm:triad-free-cutset} to find a triad-free $u\hbox{-} v$ separator $S'$ in $G'$.
	If none exists, output ``no solution exists''.
	Otherwise, let $U'$ be the component of $G'\setminus S'$ containing $u$, let $V'$ be the union of all the other components of $G'\setminus S'$, and output $(U'\cap V(G),V'\cap V(G),S'\cap V(G))$.
	
	\
	
	{\it Proof of correctness:}
	
	\
	
	We show that $\mathcal I$ admits a solution if and only if there exists a triad-free $u\hbox{-} v$ separator in $G'$.
	
	Suppose that $\mathcal I$ admits a solution $(\tilde U,\tilde V,\tilde S)$.
	Then, $\{u\}\cup \tilde U$ is anticomplete to $\{v\}\cup \tilde V$ in $G'$.
	Let $A$ be as defined by our algorithm and note that since $\tilde S\subseteq S$, we have that $\tilde S\cup A$ is triad-free.
	Thus, $\tilde S\cup A$ is a triad-free $u\hbox{-} v$ separator in $G'$.
	
	For the other direction, suppose that there exists a triad-free $u\hbox{-} v$ separator $S'$ in $G'$.
	Define $U',V'$ as in the algorithm description.
	It is straightforward to check that $(U'\cap V(G),V'\cap V(G),S'\cap V(G))$ is a partition of $V(G)$.
	Since $U'$ and $V'$ consist of disjoint components of $G'\setminus S'$, we have that $U'\cap V(G)$ is anticomplete to $V'\cap V(G)$.
	Observe that $A\subseteq S'$ since both $u$ and $v$ are complete to it.
	It follows that $S'$ cannot contain a vertex $x$ from $X=V(G)\setminus S$ because then $\{a^1_x,a^2_x,x\}$ would form a triad, which is a contradiction.
	Hence, from the definitions of $N(u)$ and $N(v)$, it follows that $U'\cap V(G)\subseteq U,V'\cap V(G)\subseteq V$ and $S'\cap V(G)\subseteq S$.
	Lastly, $S'\cap V(G)$ is triad-free since $S'$ is triad-free.
	Therefore, $(U'\cap V(G),V'\cap V(G),S'\cap V(G))$ is a solution to $\mathcal I$.
	This completes the proof.

	First, create an instance $\mathcal I'=(U,V,S,\emptyset)$ of the list 2-clique cutset problem.
	If $\mathcal I'$ admits a solution $(U',V',S',\emptyset)$, then output $(U',V',S')$.
	Otherwise, for each of the induced paths $a\hbox{-} b\hbox{-} c$ in $S$ do the following.

	If $\mathcal I_{abc}$ admits a solution for some path $a\hbox{-} b \hbox{-} c$, then output it.
	Otherwise, output ``no solution exists''.
	
	\
	
	{\it Proof of correctness:}
	
	\
	
	Suppose that $\mathcal I$ admits a solution $(\tilde U, \tilde V,\tilde S)$.
	If $\tilde S$ is a clique, then $\mathcal I'$ admits a solution.
	It is also clear that every solution to $\mathcal I'$ is a solution to $\mathcal I$.

	Since $\tilde S$ is connected, there exists an induced path $a\hbox{-} b \hbox{-} c$ in $\tilde S$.

	This completes the proof.

	The extension minimal bipartite vertex cover problem is NP-hard.

<!-- proof-12: body_md -->
We show a reduction from the extension minimal bipartite vertex cover problem to the minimal list triad-free cutset problem.
	Let $G$ and $X\subseteq V(G)$ be an instance of the extension minimal bipartite vertex cover problem.
	Let $(A,B)$ be a bipartition of $G$.
	Obtain the graph $G'$ from $G$ by making $A$ and $B$ cliques and adding the vertices $a$ and $b$ such that $N(a)=A$ and $N(b)=B$.
	Note that $G'$ is the complement of a bipartite graph, so in particular $V(G)$ is bisimplicial.
	Let $A'=A\cup\{a\}$ and $B'=B\cup \{b\}$.
	Create an instance $\mathcal I=(A'\setminus X,B'\setminus X,A\cup B)$ of the minimal list triad-free cutset problem.
	
	Let $C$ be a minimal vertex cover of $G$ with $X\subseteq C$.
	We claim that $\mathcal S=(A'\setminus C,B'\setminus C,C)$ is a solution to $\mathcal I$.
	Let $\mathcal I'=\mathcal I$ be an instance of the list triad-free cutset problem.
	It is straightforward to check that $\mathcal S$ is a solution to $\mathcal I'$.
	To show the minimality of $\mathcal S$, let $c\in C$ be a vertex and suppose that $c\in A$.
	Then, $c$ is adjacent to $a\in A'\setminus C$ by construction.
	Since $C$ is a minimal vertex cover of $G$, it also follows that there exists a vertex $b'\in B\setminus C$ such that $c$ is adjacent to it.
	The case of $c\in B$ follows analogously.
	
	For the converse, let $(A'\setminus S,B'\setminus S,S)$ be a solution to $\mathcal I$.
	It is clear that $X\subseteq S$ and that $S$ is a vertex cover of $G$.
	Since $N(a)=A$ and $N(b)=B$, it follows that for every $s\in S$ there exists a vertex in $(A\cup B)\setminus S$ that is adjacent to $s$.
	Thus, $S$ is a minimal vertex cover of $G$.
	This completes the proof.
