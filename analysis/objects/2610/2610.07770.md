<!-- theorem-1: body_md -->
\label{thm:coefficients}
For every $K>1$, there are $\varepsilon_K>0$ and a map
$f_K^0=h_K^0+\overline{g_K^0}\in\mathcal S_H^0(K)$ of maximal
dilatation exactly $K$ with the following properties.
\begin{enumerate}
\item[(i)] The inequality \eqref{eq:css-difference} fails at every sufficiently
large index.  More precisely,
\begin{equation}\label{eq:coefficient-growth}
 |a_n|-|b_n|\asymp_K n^{1+\varepsilon_K}
 \qquad(n\to\infty).
\end{equation}
In particular,
\[
 \frac{|a_n|-|b_n|}{n}\longrightarrow\infty.
\]
\item[(ii)] Its precise Hardy range is
\[
 f_K^0\in h^p\quad\Longleftrightarrow\quad
 p<\frac1{2+\varepsilon_K}.
\]
\end{enumerate}

\label{thm:boundary}
Let $f=h+\overline g$ be a sense-preserving harmonic
quasiconformal homeomorphism of $\mathbb D$.  Suppose that
$\omega=g'/h'$ satisfies
\[
 \lim_{|z|\to1}(1-|z|^2)|\omega'(z)|=0.
\]
Then the following conclusions hold.
\begin{enumerate}
\item[(i)] The map $f$ belongs to $h^p$ for every $0<p<1/2$.
\item[(ii)] For every $\varepsilon>0$,
\[
 \max_{|z|=r}|f(z)|
 =O_{f,\varepsilon}\bigl((1-r)^{-2-\varepsilon}\bigr)
 \qquad(r\uparrow1).
\]
\end{enumerate}

<!-- theorem-3: body_md -->
\label{thm:qc-sobolev}
Let $K>1$. Then the following are true.
\begin{enumerate}
\item[(i)] If $2\le s\le\infty$ and $F\in\mathcal Q_{K,s}$
maps $\mathbb D$ onto $\Omega$, then any conformal map
$H:\mathbb D\to\Omega$ satisfies
\[
 \sup\{p>0:F\in\mathcal H^p\}
 =\sup\{p>0:H\in H^p\}.
\]
In particular, $p_{\mathrm{qc}}(K,s)=1/2$.
The universal endpoint fails even for a map of maximal
dilatation exactly $K$ whose coefficient is smooth and
compactly supported in $\mathbb D$.
\item[(ii)] If $1\le s<2$, then
\[
 p_{\mathrm{qc}}(K,s)=\frac1{2K}.
\]
For every $\eta>0$, there is a map of maximal dilatation
exactly $K$ with $\|\mu_F\|_{W^{1,s}(\mathbb D)}<\eta$ that does
not belong to $\mathcal H^{1/(2K)}$.
\end{enumerate}

<!-- lemma-1: body_md -->
\label{lem:angular-width}
The function $\Psi$ is strictly increasing, and the map
$F$ in \eqref{eq:halfplane-spiral} is injective on $\mathbb H$ if
and only if $W\le2\pi$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Let $U=\log|F(e^{x+it})|$ and choose the continuous argument
$V=bx+\arg Q(t)$.  Since $U=ax+\log|Q(t)|$, we have
\[
 a\Psi'(t)
 =\det\frac{\partial(U,V)}{\partial(x,t)}
 =\frac{e^{2x}J_F(e^{x+it})}{|F(e^{x+it})|^2}>0.
\]
At each fixed image modulus, the arguments therefore form an
open interval of width $W$.  If $W\le2\pi$, no two arguments
in that interval differ by an integer multiple of $2\pi$.
If two images coincide, their arguments must therefore come from the
same value of $t$.  Their equal moduli then imply the same value of $x$.
If $W>2\pi$, two distinct values of $t$ have image arguments
differing by $2\pi$ on the same image-modulus level.  Their
images coincide.  This proves both directions.

<!-- proposition-1: body_md -->
\label{prop:spiral}
Let $K>1$, $k=(K-1)/(K+1)$, $b_K=b(k)$, and
$\delta_K=\delta(k)$.  There is a sense-preserving harmonic
homeomorphism $f_K$ of $\mathbb D$ with maximal dilatation exactly
$K$ such that
\[
 |f_K(z)|\asymp_K |1-z|^{-2-\delta_K}
 \qquad(z\in\mathbb D).
\]
Its precise Hardy range is
\[
 f_K\in h^p\quad\Longleftrightarrow\quad
 p<\frac1{2+\delta_K}.
\]
Its analytic dilatation extends analytically across
$\partial\mathbb D\setminus\{1\}$ and has no limit at $1$.
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
In \eqref{eq:halfplane-spiral}, take
\[
 q=k,\quad a=2+\delta_K,\quad b=b_K,
 \quad R=ke^{-2\pi b},\quad \vartheta=\pi\delta_K.
\]
The definition of $\delta_K$ gives
$2\sin\vartheta=k-R$, with $0<\vartheta<\pi/6$.
At the two endpoints, the factors $1+B(t)$ are
\[
 C_+=1-ik e^{-i\vartheta},\qquad
 C_-=1-iR e^{i\vartheta}.
\]
The identity $2\sin\vartheta=k-R$ is equivalent to
\[
 C_+=e^{-2i\vartheta}C_-.
\]
Thus $|C_+|=|C_-|$.  Both arguments lie between $-\pi/2$ and
$0$, and their difference is $-2\vartheta$.  On the full
interval, $1+B(t)$ stays in the right half-plane, so this
argument difference is taken in the same continuous branch.
Since
\[
 \Psi(t)=\left(a+\frac{b^2}{a}\right)t
       +\arg(1+B(t))-\frac ba\log|1+B(t)|,
\]
we obtain
\begin{equation}\label{eq:width}
 W=2\pi+\pi\left(\frac{b^2}{2+\delta_K}-\delta_K\right).
\end{equation}

To estimate this width, put $s=\sqrt{1+4\pi k}>1$.
Then
\[
 2\pi b=s-s^{-1}=2\sinh(\log s)>2\log s,
\]
so $e^{-2\pi b}<s^{-2}$.  As $\arcsin u>u$ for $0<u<1$,
\[
 \delta_K>
 \frac{k(1-e^{-2\pi b})}{2\pi}
 >\frac{k(1-s^{-2})}{2\pi}
 =\frac{2k^2}{s^2}
 =\frac{b^2}{2}
 >\frac{b^2}{2+\delta_K}.
\]
It follows from \eqref{eq:width} that $W<2\pi$.
Therefore, Lemma~\ref{lem:angular-width} proves global injectivity on
$\mathbb H$.

Now let
\[
 \phi(z)=\frac2{1-z},\qquad f_K=F\circ\phi.
\]
The map $\phi$ takes $\mathbb D$ conformally onto
$\{\operatorname{Re}w>1\}$. Conformal precomposition
preserves harmonicity, and the injectivity makes
$f_K$ a homeomorphism onto its image.  Its maximal dilatation
is still $K$, since this smaller half-plane contains points
with arguments arbitrarily close to $\pi/2$.
The continuous function $Q$ is bounded above and away from
zero on $[-\pi/2,\pi/2]$.  Hence
\[
 |F(w)|\asymp |w|^a\quad(w\in\mathbb H),\qquad
 |f_K(z)|\asymp |1-z|^{-a}\quad(z\in\mathbb D).
\]
For $r$ near $1$ and $|t|$ small,
$|1-re^{it}|\asymp(1-r)+|t|$.  Therefore, for a fixed small
$t_0>0$,
\[
 \int_{-t_0}^{t_0}|f_K(re^{it})|^p\,\mathrm{d}t
 \asymp\int_0^{t_0}\frac{\mathrm{d}t}{((1-r)+t)^{ap}}.
\]
This is uniformly bounded exactly when $ap<1$.  At $ap=1$
it diverges logarithmically, and for $ap>1$ it diverges as a
power of $(1-r)^{-1}$.  The map is bounded on arcs away from
$1$, which proves the stated Hardy range.

The analytic dilatation of $F$ is
\begin{equation}\label{eq:spiral-omega}
 \frac{G'(w)}{H'(w)}
 =ik e^{-\pi b}\frac{a-ib}{a+ib}\,w^{-2ib}.
\end{equation}
The map $\phi$ extends analytically across every boundary
point except $1$, and its image stays in the right half-plane
near every such point.  The dilatation of $f_K$ therefore
extends across those points.  Along $z=r\uparrow1$, however,
$\phi(r)\to+\infty$ and $\phi(r)^{-2ib}$ turns indefinitely.
It has no limit.

<!-- proof-3: body_md -->
The map $f_K^0$ constructed above belongs
to $\mathcal S_H^0(K)$ and has maximal dilatation exactly $K$.
Use the parameters in \eqref{eq:normalized-parameters} and the
explicit representation in \eqref{eq:normalized-parts}.
Write
\[
 P_\tau(z)=\sum_{n=1}^\infty u_nz^n,
\]
and set $a_1=1$, $b_1=0$.  The binomial series gives
\[
 \frac{(1-z)^{-\tau}-1}{\tau}
 =\sum_{n=1}^\infty
   \frac{(\tau+1)\cdots(\tau+n-1)}{n!}\,z^n.
\]
The empty product for $n=1$ is $1$.  Since $P_{\bar\tau}$ has coefficients $\overline{u_n}$,
substitution into \eqref{eq:normalized-parts} gives
\begin{equation}\label{eq:binomial-coefficients}
 u_n=\frac1n\prod_{j=1}^{n-1}\left(1+\frac{\tau}{j}\right),
 \qquad
 a_n=\frac{u_n-\rho^2\overline{u_n}}{1-\rho^2},
 \qquad
 b_n=\frac{i\rho(\overline{u_n}-u_n)}{1-\rho^2}.
\end{equation}
The two moduli need not be estimated separately.  Their
difference of squares has a simpler expression.

Put $u_n=x_n+iy_n$.  Then
\[
 a_n=x_n+i\frac{1+\rho^2}{1-\rho^2}y_n,
 \qquad b_n=\frac{2\rho}{1-\rho^2}y_n.
\]
The identity
\[
 \left(\frac{1+\rho^2}{1-\rho^2}\right)^2
 -\left(\frac{2\rho}{1-\rho^2}\right)^2=1
\]
therefore implies
\begin{equation}\label{eq:coefficient-identity}
 |a_n|^2-|b_n|^2=|u_n|^2.
\end{equation}
None of the factors
in the formula for $u_n$ vanishes, since $\operatorname{Re}\tau=a>2$.
Hence $|a_n|>|b_n|$.  We also have
\[
 |u_n|\le |a_n|\le\frac{1+\rho^2}{1-\rho^2}|u_n|,
 \qquad
 |b_n|\le\frac{2\rho}{1-\rho^2}|u_n|.
\]
Thus
\[
 |u_n|\le |a_n|+|b_n|
       \le\frac{1+\rho}{1-\rho}|u_n|.
\]
Dividing \eqref{eq:coefficient-identity} by $|a_n|+|b_n|$
now gives the two-sided bound
\begin{equation}\label{eq:coefficient-comparison}
 \frac{1-\rho}{1+\rho}|u_n|
 \le |a_n|-|b_n|\le |u_n|.
\end{equation}
Thus the coefficient difference has the same order as $|u_n|$;
the subtraction of the two moduli does not remove its leading
growth.

To estimate its modulus, observe that, for large $j$,
\[
 \log\left|1+\frac\tau j\right|
 =\frac12\log\left(1+\frac{2a}{j}+\frac{|\tau|^2}{j^2}\right)
 =\frac a j+O_\tau(j^{-2}).
\]
The remainders are summable, and the difference between
$\sum_{j=1}^{n-1}j^{-1}$ and $\log n$ converges.  Thus there
is a real constant $C_\tau$ such that
\[
 \log|u_n|=(a-1)\log n+C_\tau+o(1).
\]
Exponentiating, we obtain $|u_n|\asymp n^{a-1}$.  Since
$a-1=1+\varepsilon_K$, equation~\eqref{eq:coefficient-comparison}
proves \eqref{eq:coefficient-growth}.  Dividing by $n$ gives
\[
 \frac{|a_n|-|b_n|}{n}\longrightarrow\infty,
\]
and proves part~(i).


For part~(ii), the invertible real-affine normalization used to
obtain \eqref{eq:normalized-growth} preserves membership in every
$h^p$, as shown above.  Apply Proposition~\ref{prop:spiral} with
parameter $q_K$ and use $\delta(q_K)=\varepsilon_K$ to obtain
the asserted exact Hardy range.

<!-- proof-4: body_md -->

 f_r(z)=\frac{f(rz)}r=h_r(z)+\overline{g_r(z)}
\]
has $N$-th coefficient difference $r^{N-1}(|a_N|-|b_N|)>N$.
It is normalized, has maximal dilatation at most $K_0$, and is
harmonic and injective on a neighborhood of $\overline\mathbb D$.

We first obtain a lower Lipschitz bound on the closed disk.
Since $Df_r$ is continuous and nonsingular,
$m=\min_{z\in\overline\mathbb D,\,|v|=1}|Df_r(z)v|>0$.
Uniform continuity of $Df_r$ and integration along segments give
\[
 |f_r(w)-f_r(z)-Df_r(z)(w-z)|\le\frac m2|w-z|
\]
for all sufficiently close pairs.  Their image separation is
therefore at least $(m/2)|w-z|$.  For pairs separated by a fixed
positive distance, injectivity and compactness give a positive
minimum of $|f_r(w)-f_r(z)|/|w-z|$.  Hence, for some $c_r>0$,
\begin{equation}\label{eq:polynomial-stability}
 |f_r(w)-f_r(z)|\ge c_r|w-z|
 \qquad(z,w\in\overline\mathbb D).
\end{equation}

Let $h_{r,M}$ and $g_{r,M}$ be the Taylor polynomials of degree
$M$ of $h_r$ and $g_r$, and put $P_M=h_{r,M}+\overline{g_{r,M}}$.
They converge to the respective parts in $C^1(\overline\mathbb D)$,
since the parts are analytic on a larger disk.  For large $M$,
$\|D(P_M-f_r)\|_\infty<c_r/2$; integration along segments and
\eqref{eq:polynomial-stability} then give
\[
 |P_M(w)-P_M(z)|\ge\frac{c_r}{2}|w-z|.
\]
Thus $P_M$ is injective on $\overline\mathbb D$.

Since $\min_{\overline\mathbb D}|h_r'|>0$, the ratios
$g_{r,M}'/h_{r,M}'$ converge uniformly to $g_r'/h_r'$, whose
supremum is at most $(K_0-1)/(K_0+1)<(K-1)/(K+1)$.
For large $M$, the map $P_M$ is therefore sense-preserving with
maximal dilatation less than $K$.  Its normalizations are exact,
and taking $M\ge N$ preserves the violating coefficients.
Finally, $h_{r,M}'$ is nonzero on the closed disk and hence in a
neighborhood of it, so $g_{r,M}'/h_{r,M}'$ extends analytically
there.

Let $k=\|\omega\|_\infty<1$.  At each point $a\in\mathbb D$, we
move $a$ to the origin and make the dilatation vanish there.
Set
\[

 \qquad c=\omega(a),\qquad

\]
The map $T_c\circ f\circ\varphi_a$ is injective and harmonic.
Subtract its value at $0$ and divide by its analytic derivative
there.  The resulting harmonic univalent map
$U_a=H_a+\overline{G_a}$ satisfies
\[
 H_a(0)=G_a(0)=G_a'(0)=0,\qquad H_a'(0)=1.
\]
Its dilatation is a unimodular multiple of
\[
 \Omega_a(\zeta)=
 \frac{\omega(\varphi_a(\zeta))-\omega(a)}
      {1-\overline{\omega(a)}\omega(\varphi_a(\zeta))}.
\]

The hypothesis makes $\Omega_a$ tend to zero on compact sets
as $|a|\to1$.  Indeed, the automorphism identity
\[
 \frac{|\varphi_a'(\zeta)|}{1-|\varphi_a(\zeta)|^2}
 =\frac1{1-|\zeta|^2}
\]
shows that, for each $R<1$ and $|a|>R$,
\[
 \sup_{|\zeta|\le R}|\Omega_a'(\zeta)|
 \le C_{k,R}\sup_{|z|\ge r_a}(1-|z|^2)|\omega'(z)|
 \longrightarrow0,
\]
where $r_a=(|a|-R)/(1-|a|R)\to1$.
Since $\Omega_a(0)=0$, integration along segments proves
the locally uniform convergence to zero.

The class $\mathcal S_H^0$ of univalent harmonic maps with this
normalization is compact for locally uniform convergence
\cite[Theorem~2.3]{CSS}.  Therefore every sequence
$a_n\to\partial\mathbb D$ has a subsequence on which $U_{a_n}$
converges to a normalized univalent harmonic map.  Local
uniform convergence also gives convergence of the analytic and
coanalytic derivatives.  Since the dilatations tend to zero on
compact sets, the coanalytic derivative of the limit vanishes,
so the limit is analytic.  Its second coefficient has modulus
at most $2$ by the classical analytic bound.

Let $A_2(a)$ be the second analytic coefficient of $U_a$.
If its modulus exceeded $2$ by a fixed positive amount along
some sequence approaching the boundary, the preceding
subsequence argument would give an analytic limit violating
that bound.  Hence
\[
 \limsup_{|a|\to1}|A_2(a)|\le2.
\]

To use the coefficient bound, put
$H_c=h-\overline c\,g$, where $c=\omega(a)$.  As
$g'=\omega h'$, we have
\[
 H_c'(a)=(1-|c|^2)h'(a),\quad
 H_c''(a)=(1-|c|^2)h''(a)-\overline c\,\omega'(a)h'(a).
\]
Differentiating the analytic part of $U_a$ at the origin gives
\[
 2A_2(a)=(1-|a|^2)
  \left(\frac{h''(a)}{h'(a)}
  -\frac{\overline c\,\omega'(a)}{1-|c|^2}\right)
  -2\overline a.
\]
The hypothesis makes the term containing $\omega'(a)$ tend
to zero after multiplication by $1-|a|^2$.  Hence, uniformly
in $t$ as $r\to1$,
\[
 (1-r^2)\operatorname{Re}\left(
   e^{it}\frac{h''(re^{it})}{h'(re^{it})}\right)
 \le2r+4+o(1).
\]
The real part is $\frac{\mathrm{d}}{\mathrm{d}r}\log|h'(re^{it})|$.
Since
\[
 \frac{2r+4}{1-r^2}=\frac3{1-r}+\frac1{1+r},
\]
we can absorb the uniform $o(1)$ term into an arbitrarily small
multiple of $(1-r)^{-1}$ near the boundary.  On a fixed smaller
disk the derivatives are bounded.  Integration along radii
therefore gives, for every $\varepsilon>0$,
\[
 |h'(re^{it})|+|g'(re^{it})|
 \le C_\varepsilon(1-r)^{-3-\varepsilon}.
\]
Integrating once more gives
\begin{equation}\label{eq:growth}
 \max_{|z|=r}|f(z)|\le C_\varepsilon
                  (1-r)^{-2-\varepsilon}.
\end{equation}
For any $p<1/2$, choose $\varepsilon>0$ so that
$p(2+\varepsilon)<1$.  The radial criterion applied to
\eqref{eq:growth} proves $f\in h^p$.

If $\omega$ extends continuously to $\overline\mathbb D$, Cauchy's
estimate on $B(z,(1-|z|)/2)$, applied after subtracting
$\omega(z)$, shows that
$(1-|z|^2)|\omega'(z)|\to0$ by uniform continuity.

order $2$ and does not belong to $h^{1/2}$.  For each $K>1$,
the map $\kappa+k\overline\kappa$, with
$k=(K-1)/(K+1)$, has constant dilatation $k$ and the
same growth and Hardy threshold.

\label{thm:energy-growth}
Let $K>1$, $0<\gamma<1/2$, and $c,\eta>0$.  There is a
sense-preserving harmonic homeomorphism $f=h+\overline g$ of
$\mathbb D$ with maximal dilatation exactly $K$ whose analytic

\[

\begin{equation}\label{eq:super-koebe}
 |f(z)|\asymp |1-z|^{-2}

\end{equation}

Fix $0<\gamma<1/2$ and $c>0$, and choose
\[
 \beta=\frac{1-\gamma}{4},\qquad
 \nu=\gamma+\beta,\qquad
 B=\frac{c\gamma}{\nu}.
\]
These choices give $\gamma+2\beta<1$ and $B\nu=c\gamma$.
On the outer half-sector
$\{w:\operatorname{Re}w>0,\ |w|>R_0\}$, with $R_0$ large,
use the principal branches with $L=\log w$ to define
\[
 H(w)=w^2\exp\bigl(cL^\gamma+iBL^\nu\bigr),\quad
 G(w)=iw^2L^{-\beta}
        \exp\bigl(cL^\gamma-iBL^\nu\bigr),
 \quad F=H+\overline G.
\]

We check injectivity before transferring $F$ to the disk.
Write $w=e^{x+it}$, where $|t|<\pi/2$, and set
\[
 q=x^{-\beta},\quad u=c\gamma x^{\gamma-1},\quad
 v=B\nu x^{\nu-1},\quad
 S_x=e^{2x+cx^\gamma+iBx^\nu}.
\]
These quantities satisfy $qv=u$ and $v^2=o(u)$.  The first
identity produces the endpoint cancellation.  The second ensures
that comparing points with equal image modulus, rather than equal
source radius, changes the angular width by a smaller-order term.
Uniformly for $|t|\le\pi/2$,
\[
 \frac{F(e^{x+it})}{S_x}=e^{2it}+O(q+v).
\]
Thus $F$ has no zeros for large $x$, and its argument has a
continuous branch close to $Bx^\nu+2t$.  Differentiation of the analytic parts gives $|G'/H'|=O(q)<1$,
so the Jacobian is positive.  We also need radial monotonicity
of the image modulus.  In the explicit expression for $F/S_x$,
the remainder after $e^{2it}$ has $x$-derivative
$O(q/x+v/x+u/x)$.  It follows that
\[
 \partial_x\log|F(e^{x+it})|=2+O(v)>0.
\]
Fix $X_0$ so large that this radial derivative is positive for
$x\ge X_0$.  The moduli $|F(e^{X_0+it})|$ are uniformly bounded
in $t$, while $|F(e^{x+it})|\to\infty$ uniformly as $x\to\infty$.
Thus, for each sufficiently large image modulus $\rho$ and every
$-\pi/2\le t\le\pi/2$, there is a unique solution
$x=x_\rho(t)$ of $|F(e^{x+it})|=\rho$.
Put $U(x,t)=\log|F(e^{x+it})|$ and
$V(x,t)=\arg F(e^{x+it})$, with the continuous branch chosen above.
Differentiating $U(x_\rho(t),t)=\log\rho$ gives
$x_\rho'(t)=-U_t/U_x$.  Hence
\[
 \frac{\mathrm{d}}{\mathrm{d}t}V(x_\rho(t),t)
 =\frac{U_xV_t-U_tV_x}{U_x}
 =\frac{e^{2x}J_F(e^{x+it})}{|F(e^{x+it})|^2U_x}>0,
\]
where $x=x_\rho(t)$ on the right.
Thus the image angle increases strictly along this level.

We find the width of this level from its endpoints.  Taylor
expansion at $t=\pm\pi/2$ first gives
\[
 -\frac{F(e^{x\pm i\pi/2})}{S_x}
 =(1-iq)\pm\frac\pi2(-v+iu)(1+iq)+o(u).
\]
Since $qv=u$, we have the exact identity
\[
 (-v+iu)\frac{1+iq}{1-iq}=-v-iu.
\]
Factoring out $1-iq$ therefore gives
\begin{equation}\label{eq:slow-endpoints}
 F(e^{x\pm i\pi/2})
 =-S_x(1-iq)\left[1\mp\frac\pi2(v+iu)+o(u)\right].

$\tau\in\{\gamma,\nu,-\beta\}$, shows that the remainder in
\eqref{eq:slow-endpoints} is
$O(v^2+u/x+v/x+q/x)=o(u)$.
Here $v^2/u\asymp x^{\gamma+2\beta-1}\to0$, and
$q/(xu)\asymp x^{-\beta-\gamma}\to0$.

We now put the endpoints on a common image-modulus level.
At equal $x$, the leading term $e^{2it}$ contributes $2\pi$
to the continuous argument, while the endpoint factors in
\eqref{eq:slow-endpoints} contribute $-\pi u+o(u)$.
The logarithmic endpoint moduli differ by $O(v)$.
Since the radial derivative
of $\log|F|$ tends to $2$, putting the endpoints on the same
image modulus changes $x$ by $O(v)$.  This changes the phase
$Bx^\nu$ by $O(v^2)=o(u)$, $q$ by $O(qv/x)=o(u)$,
and $u$ by $O(uv/x)=o(u)$.
Consequently the actual angular width is, with $u$ evaluated
at either endpoint of the level,
\[
 W_\rho=2\pi-\pi u+o(u)<2\pi.
\]
The image angle is strictly increasing on each such level and
varies by less than one full turn.  Thus no two points on the
level have the same image.  Choose $R$ so
large that every image modulus of $\{\operatorname{Re}w>R\}$
is in this range.  Hence $F$ is injective on that half-plane.

dilatation of $F$ is
\begin{equation}\label{eq:slow-omega}
 \Omega(w)=iL^{-\beta}e^{-2iBL^\nu}
 \frac{2+c\gamma L^{\gamma-1}-\beta/L-iB\nu L^{\nu-1}}

\end{equation}

\[
 |\Omega(w)|=O((\log|w|)^{-\beta}),\qquad
 |\Omega'(w)|
 =O\!\left(\frac1{|w|(\log|w|)^{1-\gamma}}\right).
\]

\begin{equation}\label{eq:slow-energy}

\[
 \omega_s=\frac{\Omega\circ\phi_R+s}

\]
The denominator stays bounded away from zero, so the formula is

Since $\Omega\circ\phi_R(z)\to0$ as $z\to1$, we have

 |\omega_s'|\le\frac{|(\Omega\circ\phi_R)'|}

\]
The denominator is bounded below by $(1-k^2)^2$, independently

\]
since $|1-re^{it}|\asymp(1-r)+|t|$ there and the exponential
factor is at least $1$ near $1$.  Higher exponents fail by
inclusion of Hardy spaces.  This completes the proof.

Precompose $H$ with a disk automorphism so that $H(0)=F(0)$.
This does not change its Hardy spaces.  Factor $F=H\circ\phi$,
where $\phi:\mathbb D\to\mathbb D$ is quasiconformal with coefficient
$\mu_F$ and $\phi(0)=0$.  The map $\phi$ extends homeomorphically to
$\overline\mathbb D$, and reflection across the unit circle gives a
global quasiconformal map $\widehat\phi$ fixing $\infty$.
Its quasisymmetry constants depend only on the
maximal dilatation of $F$.

Write
\[
 \Delta_{\xi,t}=\phi((1-t)\xi)-\phi(\xi).
\]
We claim that
\begin{equation}\label{eq:qc-half-ratio}
 \lim_{t\downarrow0}\sup_{\xi\in\partial\mathbb D}
 \left|\frac{\Delta_{\xi,t/2}}{\Delta_{\xi,t}}
                        -\frac12\right|=0.
\end{equation}
To see this, take arbitrary sequences $\xi_n\in\partial\mathbb D$
and $t_n\downarrow0$, and rescale at the boundary point by setting
\[
 G_n(w)=
 \frac{\widehat\phi(\xi_n(1-t_nw))-\phi(\xi_n)}
      {\Delta_{\xi_n,t_n}}.
\]
Each $G_n$ fixes $0$, $1$, and $\infty$.  The compactness theorem
for normalized global quasiconformal maps \cite[Theorem~3.9.4]{AIM}
allows us to pass to a subsequence that converges locally uniformly
to a quasiconformal homeomorphism $G$.
We may also assume that $\xi_n\to\xi$ and
$m_{\xi_n,t_n}\to\alpha$.

The disks $|1-t_nw|<1$ converge to

$a_n=\phi(\xi_n)$ and $\Delta_n=\Delta_{\xi_n,t_n}$.

\[
 \left\{w:

      +|\Delta_n|^2|w|^2<0\right\}.
\]

$|\Delta_n|\to0$.  A subsequence of these disks

\begin{equation}\label{eq:qc-boundary-distance}
 1-|\phi(x)|\le|\Delta_{\xi,t}|
       \le C_K(1-|\phi(x)|).
\end{equation}

$\phi(y)$ is a nearest boundary point to $\phi(x)$.  Then
$|x-\xi|=t\le |x-y|$, and quasisymmetry gives
\[
 |\phi(x)-\phi(\xi)|
 \le C_K|\phi(x)-\phi(y)|
 =C_K(1-|\phi(x)|).
\]

rescaled image disk is bounded below independently of $n$.

 \overline{\xi_n}^{\,2}

 G(w)=\frac{w+c\bar w}{1+c}.

 2^{-1-\varepsilon}|\Delta_{\xi,t}|

 M_H(1-C'_\varepsilon t^{1-\varepsilon})

An open map has no interior maximum of its modulus, so $M_G$
is nondecreasing.  By \eqref{eq:radial-criterion}, Hardy
membership is equivalent to $I_G(p)<\infty$.  In that case,
monotonicity gives
\begin{equation}\label{eq:radial-integral-bound}
 tM_G(1-t)^p
 \le\int_{1-t}^1M_G(r)^p\,\mathrm{d}r
 \le I_G(p).
\end{equation}

Suppose first that $H\in H^p$.  Applying
\eqref{eq:radial-integral-bound} to the upper comparison of
radial maxima, we get
\[
 M_F(1-t)\le C t^{-(1+\varepsilon)/p}.
\]
The radial criterion then gives $F\in\mathcal H^q$ whenever
$q(1+\varepsilon)<p$.  Every $q<p$ is allowed by choosing
$\varepsilon$ sufficiently small.
Conversely, if $F\in\mathcal H^p$, the lower comparison gives
\[
 M_H(1-C'_\varepsilon t^{1-\varepsilon})
 \le C t^{-1/p}.
\]
With $u=C'_\varepsilon t^{1-\varepsilon}$, this becomes
\[
 M_H(1-u)\le C'u^{-1/[p(1-\varepsilon)]}.
\]
Thus $H\in H^q$ whenever $q<p(1-\varepsilon)$.  Given any
$q<p$, we can choose $\varepsilon$ small enough for this
inequality to hold.
The two implications show that the suprema of the admissible
exponents agree, whether finite or infinite.

The usual quadratic growth estimate for the conformal
map $H$, together with the upper estimate for $M_F$, gives
\[
 M_F(r)\le C_\varepsilon
                 (1-r)^{-2-2\varepsilon}.
\]
Relabeling $\varepsilon$ gives the stated estimate.
Choosing $\varepsilon$ with $(2+\varepsilon)p<1$ in the
radial criterion gives $F\in\mathcal H^p$ for $p<1/2$.

Suppose $\mu_F\in W^{1,2}(\mathbb D)$.  The caps $U_{\xi,t}$

On the bounded disk, $W^{1,s}\subset W^{1,2}$ when $s\ge2$.

$\sigma\in C_c^\infty((1/4,1/2))$ with
$0\le\sigma\le1$ and $\max\sigma=1$.  Put
\[
 \lambda(r)=1+(K-1)\sigma(r),\qquad
 \rho(r)=\exp\left(-\int_r^1\frac{\lambda(t)}t\,\mathrm{d}t\right),
 \qquad
 \chi(re^{i\theta})=\rho(r)e^{i\theta}.
\]
Set $\chi(0)=0$.
The ratio of radial to tangential stretching of $\chi$
is $\lambda(r)$.  Therefore $\chi$ has maximal
dilatation exactly $K$, and
\[
 \mu_\chi(re^{i\theta})
    =e^{2i\theta}\frac{\lambda(r)-1}{\lambda(r)+1}
       \in C_c^\infty(\mathbb D).
\]
Since $\chi(z)=z$ for $|z|\ge1/2$, the map
$\kappa\circ\chi$, where
$\kappa(z)=z/(1-z)^2$, agrees with the Koebe function
near the boundary.  Its circular $1/2$-means diverge.

\cite[Theorem~3.2]{AK} gives

\[
\begin{aligned}
 C(z)&=\frac{1+z}{1-z},&
 S_R(w)&=w\left(1+\frac{|w|^2}{R^2}\right)^{(K-1)/2},\\
 \phi_R&=C^{-1}\circ S_R\circ C,&
 F_R&=\kappa\circ\phi_R\qquad(R\ge1).
\end{aligned}
\]
The radial map $S_R$ preserves the right half-plane.
Its radial to tangential stretch ratio increases from
$1$ to $K$ as $|w|\to\infty$.  Thus $F_R$ has maximal
dilatation exactly $K$.

\begin{equation}\label{eq:qc-mu-R}
\mu_{F_R}(z)
 =\frac{(K-1)|C(z)|^2}
         {2R^2+(K+1)|C(z)|^2}
       \frac{1-z^2}{1-\bar z^{\,2}}.
\end{equation}
Formula~\eqref{eq:qc-mu-R} has an apparent singularity at
$z=-1$.  Cancelling the corresponding factor gives
\[
 \mu_{F_R}(z)=
 \frac{(K-1)(1+z)^2(1-z)}
 {(1-\bar z)
  \bigl(2R^2|1-z|^2+(K+1)|1+z|^2\bigr)}.
\]
The denominator in this expression does not vanish near $-1$,
so the coefficient extends smoothly there.  To estimate it
uniformly in $R$, write $u=1-z$, $N=(1+z)^2u/\bar u$, and
$D_R=2R^2|u|^2+(K+1)|1+z|^2$.  Then
$D_R\asymp_K1+R^2|u|^2$,
$|\nabla N|\le C/|u|$, and
$|\nabla D_R|\le C_K(R^2|u|+1)$.
The quotient rule gives
\[
 |\mu_{F_R}(z)|
   \le\frac{C_K}{1+R^2|u|^2},
 \qquad
 |\nabla\mu_{F_R}(z)|
   \le\frac{C_K}{|u|(1+R^2|u|^2)}.
\]

for $1\le s<2$,
\[
 \int_\mathbb D|\nabla\mu_{F_R}|^s\,\mathrm{d}A
 \le C_{K,s}R^{s-2}
       \int_0^{2R}\frac{v^{1-s}}{(1+v^2)^s}\,\mathrm{d}v
 =O_{K,s}(R^{s-2})\longrightarrow0.
\]
The first bound and dominated convergence also show
$\|\mu_{F_R}\|_{L^s(\mathbb D)}\to0$.

\[
 |1-\phi_R(z)|\asymp_R|1-z|^K,
 \qquad
 |F_R(z)|\asymp_R|1-z|^{-2K}.
\]

\[
 \int_{-t_0}^{t_0}
     |F_R(re^{it})|^{1/(2K)}\,\mathrm{d}t
 \gtrsim_R
 \int_{-t_0}^{t_0}
       \frac{\mathrm{d}t}{\sqrt{(1-r)^2+t^2}}
 \longrightarrow\infty.
\]
Thus $F_R\notin\mathcal H^{1/(2K)}$, and choosing $R$
large makes $\|\mu_{F_R}\|_{W^{1,s}}<\eta$.
This proves part~(ii).

\label{prop:supercritical}
Let $F:\mathbb D\to\mathbb C$ be an orientation-preserving
quasiconformal homeomorphism with
$\mu_F\in W^{1,s}(\mathbb D)$ for some $2<s\le\infty$, and let
$H:\mathbb D\to F(\mathbb D)$ be conformal.  Then, for every $p>0$,
\[
 F\in\mathcal H^p\quad\Longleftrightarrow\quad H\in H^p.
\]
Moreover,
\[
 \max_{|z|=r}|F(z)|=O_F((1-r)^{-2})\qquad(r\uparrow1).
\]

<!-- proof-10: body_md -->
As in Lemma~\ref{lem:qc-cap}, normalize $H(0)=F(0)$ and write
$F=H\circ\phi$, where $\phi:\mathbb D\to\mathbb D$ is quasiconformal,
$\phi(0)=0$, and $\mu_\phi=\mu_F$.
Morrey's inequality gives a representative of $\mu_F$ in
$C^\alpha(\overline\mathbb D)$ for some $0<\alpha<1$.
For finite $s$ we may take any $0<\alpha<1-2/s$; for
$s=\infty$ we may take any $0<\alpha<1$.
Let $k=\|\mu_F\|_\infty<1$.

To obtain regularity up to the unit circle, extend the coefficient
to the plane.  Let $\Pi:\mathbb C\to\overline\mathbb D$
be nearest-point projection, and choose a smooth compactly
supported function $\chi$ with $0\le\chi\le1$ and $\chi=1$
on a neighborhood of $\overline\mathbb D$.  Since $\Pi$ is
Lipschitz, the function
\[
 \nu(z)=\chi(z)\mu_F(\Pi(z))
\]
is compactly supported and H\"older continuous, agrees with
$\mu_F$ on $\mathbb D$, and has supremum norm at most $k$.
The classical regularity theorem for the Beltrami equation
with a compactly supported $C^\alpha$ coefficient gives a
$C^{1,\alpha}$ quasiconformal diffeomorphism
$\Psi:\mathbb C\to\mathbb C$ solving this equation.  In
particular, $D\Psi$ is nonsingular.  This form of Schauder
regularity is discussed in \cite{AIM} and recalled in
\cite[Section~1]{MOV}.

The domain $\Psi(\mathbb D)$ is bounded by a $C^{1,\alpha}$ Jordan
curve.  Let $R:\Psi(\mathbb D)\to\mathbb D$ be conformal.  The
Kellogg--Warschawski theorem
\cite[Theorem~3.6]{Pommerenke} gives $C^{1,\alpha}$ extensions
of $R$ and $R^{-1}$ to their closed domains, with nonvanishing
conformal derivatives.  Thus $\psi=R\circ\Psi|_\mathbb D$ is a
$C^1$ diffeomorphism of the closed disk and has coefficient
$\mu_F$.  The Beltrami composition formula makes
$\phi\circ\psi^{-1}$ a conformal disk automorphism.
Consequently $\phi$ and $\phi^{-1}$ are $C^1$ on the closed
disk as well.  Their derivatives are bounded, and the disk
is convex, so both maps are Lipschitz.

Choose $L\ge1$ to be a Lipschitz constant for both maps.
Comparing distances to the unit circle gives
\[
 L^{-1}(1-|z|)\le1-|\phi(z)|\le L(1-|z|).
\]
Since $\phi(0)=0$, the same Jordan-domain inclusion argument
as in Lemma~\ref{lem:qc-cap} now yields
\begin{equation}\label{eq:linear-radial-comparison}
 M_H(1-Lt)\le M_F(1-t)\le M_H(1-L^{-1}t)
\end{equation}
for all sufficiently small $t>0$.
For every fixed $p>0$, linear changes of variable in these
integrals show that
\[
 \int_0^1M_F(r)^p\,\mathrm{d}r<\infty
 \quad\Longleftrightarrow\quad
 \int_0^1M_H(r)^p\,\mathrm{d}r<\infty.
\]
The radial criterion proves the equivalence of Hardy
memberships, without loss in $p$.  Finally, the conformal
quadratic growth bound for $H$ and the upper estimate in
\eqref{eq:linear-radial-comparison} give the asserted bound
for $F$.

\label{ex:critical-endpoint}
For every $K>1$, there is a quasiconformal homeomorphism
$F:\mathbb D\to\kappa(\mathbb D)$ of maximal dilatation at most $K$ such
that
\[
 \mu_F\in W^{1,2}(\mathbb D)\cap C(\overline\mathbb D),\qquad
 F\in\mathcal H^p\quad\Longleftrightarrow\quad0<p\le\tfrac12.

In contrast, $\kappa\notin H^{1/2}$, although $F$ and
$\kappa$ have the same image and the same critical Hardy
exponent.

Fix $m>1$, and choose $B_0>2m$ so large that
$(1-2m/B_0)^{-1}<K$.  For $r\ge0$, put
\[
 L(r)=B_0+\log(1+r^2),\qquad
 s(r)=rL(r)^{-m},\qquad
 S(re^{i\theta})=s(r)e^{i\theta}.
\]
The ratio of radial to tangential stretching is
\begin{equation}\label{eq:log-stretch-ratio}
 \lambda(r)=\frac{rs'(r)}{s(r)}
 =1-\frac{2mr^2}{(1+r^2)L(r)}.
\end{equation}
It lies between $1-2m/B_0$ and $1$.  Hence $s$ is strictly
increasing, starts at $0$, and tends to infinity.  The map
$S$ is an orientation-preserving quasiconformal homeomorphism
of the plane, with maximal dilatation less than $K$, and
preserves the right half-plane.
Let
\[
 C(z)=\frac{1+z}{1-z},\qquad
 \phi=C^{-1}\circ S\circ C,\qquad F=\kappa\circ\phi.
\]
Then $\phi$ maps $\mathbb D$ onto itself, and $F$ has the required
image and distortion bound.

The only boundary points requiring separate checks are $-1$
and $1$, which $C$ sends to $0$ and infinity.  Put
\[
 \eta(r)=\frac{mr^2}{(1+r^2)L(r)}.
\]
Using \eqref{eq:log-stretch-ratio} and the conformal
precomposition formula, we obtain
\begin{equation}\label{eq:log-stretch-coefficient}
 \mu_F(z)
 =-\frac{\eta(|C(z)|)}{1-\eta(|C(z)|)}
      \frac{1-z^2}{1-\bar z^{\,2}}.
\end{equation}
Near $r=0$, we can write $\eta(r)=r^2E(r^2)$ with $E$
smooth.  The coefficient of $S$ near $w=0$ is consequently
\[
 -\frac{E(|w|^2)}{1-|w|^2E(|w|^2)}\,w^2,
\]
which is smooth.  Since $C$ is conformal near $z=-1$,
$\mu_F$ extends smoothly across that point.  Away from $1$
the remaining factors are smooth as well.

For large $r$, direct differentiation gives
\[
 \eta(r)=O((\log r)^{-1}),\qquad
 \eta'(r)=O\bigl(r^{-1}(\log r)^{-2}\bigr).
\]
Near $z=1$, let $u=1-z$.  We have
$|C(z)|\asymp|u|^{-1}$, $|C'(z)|\asymp|u|^{-2}$, and the
gradient of $(1-z^2)/(1-\bar z^{\,2})$ is $O(|u|^{-1})$.
Differentiating \eqref{eq:log-stretch-coefficient} therefore
gives
\[
 |\nabla\mu_F(z)|
 \le\frac{C}{|u|\log(e/|u|)}.
\]
Also $\mu_F(z)\to0$ as $z\to1$.  Thus the coefficient is
continuous on $\overline\mathbb D$, and polar integration about $1$
shows that
\[
 \int_{|1-z|<t_0}|\nabla\mu_F(z)|^2\,\mathrm{d}A(z)
 \le C\int_0^{t_0}\frac{\mathrm{d}t}{t[\log(e/t)]^2}<\infty.
\]
Together with smoothness away from $1$, this proves
$\mu_F\in W^{1,2}(\mathbb D)$.

It remains to determine the Hardy behavior.  The identity
\[
 \kappa(C^{-1}(w))=\frac{w^2-1}{4}
\]
shows that, uniformly as $z\to1$ within $\mathbb D$,
\begin{equation}\label{eq:log-stretch-growth}
 |F(z)|\asymp |1-z|^{-2}
      [\log(e/|1-z|)]^{-2m}.
\end{equation}
The map is bounded on boundary arcs away from $1$.
The function $t^{-2}[\log(e/t)]^{-2m}$ decreases as $t$
increases, for small $t$.  Since $|1-z|\ge1-r$ on $|z|=r$,
the upper bound in \eqref{eq:log-stretch-growth}, together
with its lower bound at $z=r$, gives
\[
 M_F(1-t)\asymp t^{-2}[\log(e/t)]^{-2m}
 \qquad(t\downarrow0).
\]
The radial integral at $p=1/2$ is finite because
\[
 \int_0^{t_0}\frac{\mathrm{d}t}{t[\log(e/t)]^m}<\infty.
\]
For $p<1/2$ it is also finite.  For $p>1/2$, choose
$0<\varepsilon<2p-1$ and use
$[\log(e/t)]^{-2mp}\ge t^\varepsilon$ for sufficiently small
$t$.  The radial integral then dominates a divergent power
integral.  The radial criterion \eqref{eq:radial-criterion}
proves the precise Hardy range claimed in the example.
