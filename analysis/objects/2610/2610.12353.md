<!-- definition-1: body_md -->
\label{def:almost_surely}
Let $\mathbb{P}$ denote probability, and let $x$ be a random variable.
 
A property of $x$ is said to hold \textit{almost surely} if it holds with probability one under $\mathbb{P}$.
 
Equivalently, the property holds for every value $x$ can take except possibly for a set of values whose total probability is zero, and such exceptional values can therefore be disregarded in practice.
 
Equivalently, the set of values of $x$ for which the property fails has measure zero.
 
For example, if $x$ is uniformly distributed on $[0,1]$, then $x\neq0.5$ holds almost surely, since the single value $x=0.5$ has probability zero, even though $x=0.5$ remains a possible outcome.
<!-- end definition-1 -->
<!-- lemma-1: body_md -->
\label{lem:F_monotone}
The Fisher information matrix ${\mathcal F}_k$ satisfies ${\mathcal F}_{k+1}\succeq {\mathcal F}_k$ for all $k$, surely.
 
Consequently, $\lambda_{\min}({\mathcal F}_k)$ is a nondecreasing real-valued sequence, for every sample path.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
It follows from \eqref{eq:fisher_information} that
${\mathcal F}_{k+1}-{\mathcal F}_k=\phi_k\phi_k^{\rm T} \succeq 0$, since $\phi_k\phi_k^{\rm T}$ is positive semidefinite for every $\phi_k$.
 
Monotonicity of $\lambda_{\min}(\cdot)$ under the Loewner order then implies that $\lambda_{\min}({\mathcal F}_{k+1})\geq\lambda_{\min}({\mathcal F}_k)$.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:sinusoidal_PE}
Let $\omega_1,\ldots,\omega_n$ be distinct frequencies in $(0,\pi)$, and define
\begin{align}
    s_k
    \stackrel{\triangle}{=}
    \begin{bmatrix}
        \sin(\omega_1 k) \\
        \cos(\omega_1 k) \\
        \vdots \\
        \sin(\omega_n k) \\
        \cos(\omega_n k)
    \end{bmatrix}
    \in
    {\mathbb R}^{2n}.
    \label{eq:s_k_def}
\end{align}
Then the sequence $\{s_k\}$ is persistently exciting.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
For each $\omega_j$, the sequences
$\sin(\omega_j k)$ and $\cos(\omega_j k)$ are linearly independent.
Furthermore, sinusoidal sequences associated with distinct frequencies in $(0,\pi)$ are linearly independent.
Consequently, there exist an integer $N>0$ and a scalar $\alpha_s>0$ such that
\begin{align}
    \sum_{i=k}^{k+N-1}
        s_i s_i^{\rm T}
        &\geq
        \alpha_s I_{2n}
    \label{eq:sinusoidal_basis_PE}
\end{align}
for all $k$.
Thus, $\{s_k\}$ is persistently exciting.

<!-- proposition-3: body_md -->
\label{prop:constant_sigma}
Consider the system \eqref{eq:system} with input \eqref{eq:nominal_plus_white_noise} and $\sigma_k\equiv\sigma>0$.
 
Let $\phi_k^w$ be the steady-state regressor generated by $w_k$, and assume $R_w\stackrel{\triangle}{=}\mathbb E[\phi_k^w(\phi_k^w)^{\rm T}]\succ0$.
 
Define ${\mathcal F}_k^w\stackrel{\triangle}{=}\sum_{i=0}^{k-1}\sigma^2\phi_i^w(\phi_i^w)^{\rm T}$ and $\bar P_k\stackrel{\triangle}{=}\left(P_0^{-1}+\mathbb E[{\mathcal F}_k^w]\right)^{-1}$.
 
Then the following hold.
\begin{enumerate}
    \item[(a)] $\mathbb E[\phi_k\phi_k^{\rm T}]\succeq\sigma^2R_w$ for all $k$.
     
    Consequently, for every integer $N>0$,
    \begin{align}
        \mathbb E\left[\sum_{i=k}^{k+N-1}\phi_i\phi_i^{\rm T}\right]
            &\succeq
            N\sigma^2\lambda_{\min}(R_w)I_{2n},
        \nonumber
    \end{align}
    so $\{\phi_k\}$ is persistently exciting in expectation.
     
    \item[(b)] $\mathbb E[{\mathcal F}_k^w]=k\sigma^2R_w$ and
    \begin{align}
        \lambda_{\max}(\bar P_k)
            &\leq
            \frac{1}{\lambda_{\min}(P_0^{-1})+k\sigma^2\lambda_{\min}(R_w)}
            \nonumber \\
            &=
            {\mathcal O}\left((k\sigma^2)^{-1}\right).
        \label{eq:P_decay_rate}
    \end{align}
\end{enumerate}

\label{prop:information_bound}

\label{thm:self_regulating}
Under the hypotheses of Proposition~\ref{prop:information_bound}, almost surely,
\begin{align}
    \sigma_k
        &\rightarrow
        0,
    \\
    p_k
        &\rightarrow
        0,
    \\
    \lambda_{\min}({\mathcal F}_k)
        &\rightarrow
        \infty,
    \\
    \hat\theta_k
        &\rightarrow
        \theta,
    \\
    \frac{\sigma_k}{p_k}
        &\rightarrow
        \infty.
\end{align}

<!-- remark-1: body_md -->
\label{rem:beta_effect}
Simulations over the tested range of $\beta$ indicate that $p_k={\mathcal O}(k^{-\gamma})$, where the rate $\gamma$ decreases monotonically as $\beta$ increases.
 
A larger $\beta$ gives a smaller $\sigma_k$ for the same $p_k$, so less excitation is injected and $p_k$ decays more slowly.
 
The exponent $\beta$ therefore trades the rate of covariance decay against the amount of excitation retained.
 
A rigorous characterization of $\gamma$ as a function of $\beta$ is left to future work.

\label{ex:three_case}
\textup{Effect of self-regulating excitation on persistent excitation and parameter convergence.}
Consider the third-order system
\begin{align}
    y_k
        &=
        \frac{0.5 q^2 - 0.2 q + 0.1}{q^3 - 0.8 q^2 - 0.03 q + 0.09}
        u_k,
\end{align}
with poles at $0.5$, $0.6$, and $-0.3$.
 
Define $\phi_k\stackrel{\triangle}{=}[-y_{k-1}\ -y_{k-2}\ -y_{k-3}\ u_{k-1}\ u_{k-2}\ u_{k-3}]^{\rm T}$, so that $y_k=\phi_k^{\rm T}\theta$ with $\theta=[-0.8\ -0.03\ 0.09\ 0.5\ -0.2\ 0.1]^{\rm T}$.
 
To apply RLS, let $P_0=10I_6$ and $\hat\theta_0=0$.
 
Consider the plant input defined in \eqref{eq:nominal_plus_white_noise}, with the nominal input
\begin{align}
\bar u_k
&=
\sin(0.3k)+\sin(0.75k+0.4),
\end{align}
which is a two-tone multisine input.
 
Since $\phi_k\in{\mathbb R}^6$ requires three distinct frequencies for $H$ in \eqref{eq:regressor_sinusoidal_mapping} to have full rank, the two-tone input $\bar u_k$ alone is not persistently exciting.

Three choices of $\sigma_k$ are considered: $\sigma_k=0$ (Case 1); $\sigma_k=0.5$, constant (Case 2); and $\sigma_k$ self-regulating per \eqref{eq:self_regulating_sigma} with $\sigma_0=0.5$, $\beta=0.8$ (Case 3).
 
Figure~\ref{fig:three_case} shows, for all three cases, the parameter error $\|\theta_k-\theta\|$, the covariance measure $\lambda_{\max}(P_k)$, the Fisher information measure $\lambda_{\min}(\mathcal F_k)$, and the excitation amplitude $\sigma_k$.

 
In Case 1, the parameter error does not converge to zero, $\lambda_{\max}(P_k)$ plateaus, and $\lambda_{\min}(\mathcal F_k)$ plateaus rather than growing without bound, confirming the lack of persistent excitation identified above.

 
In Cases 2 and 3, the injected noise restores persistent excitation: the parameter error and $\lambda_{\max}(P_k)$ converge to zero, and $\lambda_{\min}(\mathcal F_k)$ grows without bound, consistent with Theorem~\ref{thm:self_regulating}.
 
However, $\sigma_k$ remains constant in Case 2, whereas $\sigma_k$ vanishes in Case 3, confirming the remaining claim of Theorem~\ref{thm:self_regulating}.

Since $P_k^{-1}=P_0^{-1}+{\mathcal F}_k$ and ${\mathcal F}_k$ is nondecreasing in the Loewner order, $p_k$ is nonincreasing and converges to some $p_\infty\geq0$ on every sample path.

Consider a sample path with $p_\infty>0$.
 
Then $\sigma_k\rightarrow\bar\sigma\stackrel{\triangle}{=}\sigma_0(p_\infty/p_0)^\beta>0$, so there exists $K\geq1$ such that $\sigma_{k-1}^2\geq\bar\sigma^2/2$ for all $k\geq K$.
 
Define $t_j\stackrel{\triangle}{=} K+jN$, ${\mathcal G}_j\stackrel{\triangle}{=}{\mathcal H}_{t_j-1}$, and $\Delta_j\stackrel{\triangle}{=}{\mathcal F}_{t_{j+1}}-{\mathcal F}_{t_j}$.
 
By Proposition~\ref{prop:information_bound} with $k=t_j$, ${\mathbb E}[\Delta_j\mid{\mathcal G}_j]\succeq(c\bar\sigma^2/2)I_{2n}$.
 
Since $\|\phi_i\|^2\leq B$, $\|\Delta_j\|\leq NB$.
 
Hence $D_j\stackrel{\triangle}{=}\Delta_j-{\mathbb E}[\Delta_j\mid{\mathcal G}_j]$ is a bounded martingale difference sequence, and the strong law of large numbers for martingales gives $S_n/n\rightarrow0$ almost surely, where $S_n\stackrel{\triangle}{=}\sum_{j=0}^{n-1}D_j$.
 
Then
\begin{align}
    \lambda_{\min}\left(\frac{{\mathcal F}_{t_n}}{n}\right)
        &\geq
        \frac{c\bar\sigma^2}{2}
        -
        \left\|\frac{S_n}{n}\right\|
        -
        \frac{\|{\mathcal F}_{t_0}\|}{n},
    \nonumber
\end{align}
so $\liminf_n\lambda_{\min}({\mathcal F}_{t_n})/n\geq c\bar\sigma^2/2>0$.
 
Since $\lambda_{\min}({\mathcal F}_k)$ is nondecreasing, $\lambda_{\min}({\mathcal F}_k)\rightarrow\infty$ almost surely on this event.
 
This gives $p_k\leq\left(\lambda_{\min}(P_0^{-1})+\lambda_{\min}({\mathcal F}_k)\right)^{-1}\rightarrow0$, which contradicts $p_\infty>0$.
 
Therefore $p_\infty=0$ almost surely.

It follows that $\sigma_k\rightarrow0$ and $p_k\rightarrow0$ almost surely.

        \infty

\end{align}
