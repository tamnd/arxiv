\label{defn:1.1}

\label{lem:1.2}
	For any real-valued smooth functions $\mathbf{f}$ and $\mathbf{g}$ with compact support in $\mathbb{R}^{2}$, the following hold:\\
	1. $\| \mathbf{f}^{2}\mathbf{g}^{2}\|_{L^{1}} \le \| \mathbf{f}\|_{L^{2}}\| \mathbf{g}\|_{L^{2}}\| \nabla\mathbf{f}\|_{L^{2}}\| \nabla\mathbf{g}\|_{L^{2}}$,\\
	2. $\|\mathbf{f}\|^{4}_{L^{4}} \le 2 \|\mathbf{f}\|^{2}_{L^{2}}\| \nabla\mathbf{f}\|^{2}_{L^{2}}$.

<!-- proof-2: body_md -->
For any $(x, y)$, by the fundamental theorem of calculus, the function can be written as
	\begin{equation}
		\label{eq:1.4}
		\mathbf{f}(x,y) = \int_{-\infty}^{x} \partial_{1}\mathbf{f}(s,y)gds = - \int_{x}^{-\infty} \partial_{1}\mathbf{f}(s,y)gds.
	\end{equation}
	\begin{equation}
		\label{eq:1.5}
		\mathbf{g}(x,y) = \int_{-\infty}^{y} \partial_{2}\mathbf{g}(x,t)gdt = - \int_{y}^{-\infty} \partial_{2}\mathbf{g}(x,t)gdt.
	\end{equation}
	By \eqref{eq:1.4} we have
	\begin{equation}
		\label{eq:1.6}
		|\mathbf{f}(x,y)| = \frac{1}{2} \left|\left[ \int_{-\infty}^{x}\partial_{1}\mathbf{f}(s,y)gds +\int_{x}^{-\infty} -\partial_{1}\mathbf{f}(s,y)gds\right]\right| \le \frac{1}{2} \int_{-\infty}^{\infty} |\partial_{1}\mathbf{f}(s,y)g|ds. 
	\end{equation}
	In the same way 
	\begin{equation}
		\label{eq:1.7}
		|\mathbf{g}(x,y)| \le \frac{1}{2} \int_{-\infty}^{\infty} |\partial_{2}\mathbf{g}(x,t)g|dt.
	\end{equation}
	Using \eqref{eq:1.6} and \eqref{eq:1.7}, and integrating with respect to $x$ and $y$,
	\begin{equation}
		\label{eq:1.8}
		\left|\int_{-\infty}^{\infty}\mathbf{f}(x,y)\mathbf{g}(x,y)dxdy \right|\le \frac{1}{4} \int_{-\infty}^{\infty} |\partial_{1}\mathbf{f}(s,y)g|ds\int_{-\infty}^{\infty} |\partial_{2}\mathbf{g}(x,t)g|dt.
	\end{equation}
	\eqref{eq:1.8} is equivalent to
	\begin{equation}
		\label{eq:1.9}
		\| \mathbf{f}\mathbf{g}\|_{L^{1}} \le \frac{1}{4} \|\partial_{1}\mathbf{f}\|_{L^{1}}\|\partial_{2}\mathbf{g}\|_{L^{1}}.
	\end{equation} 
	Replacing $\mathbf{f}$ and $\mathbf{g}$ by $\mathbf{f}^{2}$  and $\mathbf{g}^{2}$ , respectively, it yields 
	\begin{equation}
		\label{eq:1.10}
		\| \mathbf{f}^{2}\mathbf{g}^{2}\|_{L^{1}} \le  \|\mathbf{f}\partial_{1}\mathbf{f}\|_{L^{1}}\|\mathbf{g}\partial_{2}\mathbf{g}\|_{L^{1}}.
	\end{equation}
	Using the Cauchy-Schwarz inequality twice gives
	\begin{equation}
		\label{eq:1.11}
		\begin{aligned}
		\| \mathbf{f}^{2}\mathbf{g}^{2}\|_{L^{1}} &\le  \|\mathbf{f}\|_{L^{2}}\|\mathbf{g}\|_{L^{2}}\|\partial_{1}\mathbf{f}\|_{L^{2}}\|\partial_{2}\mathbf{g}\|_{L^{2}}
		\\&\le \|\mathbf{f}\|_{L^{2}}\|\mathbf{g}\|_{L^{2}}\|\nabla\mathbf{f}\|_{L^{2}}\|\nabla\mathbf{g}\|_{L^{2}}.
	\end{aligned}
	\end{equation}
	Therefore, the proof of the first inequality is completed. Setting $\mathbf{f} =\mathbf{g}$ in \eqref{eq:1.11}, and using the Cauchy-Schwarz inequality, it yields
\begin{equation}
	\label{eq:1.12}
	\|\mathbf{f}\|^{4}_{L^{4}} \le 2 \|\mathbf{f}\|^{2}_{L^{2}}\| \nabla\mathbf{f}\|^{2}_{L^{2}}.
\end{equation}

		\begin{equation}
			\label{eq:1.25}
			(G,J)_{\mathcal{P}_{T}} = \textbf{E}\int_{0}^{T} tr\{G(s)QJ^{*}(s)\}ds \quad \forall \,G,J \in \mathcal{P}_{T}.
		\end{equation}
		The map $J : \mathcal{P}_{T} \to L^{2}(\Omega_{T})$, defined by the real-valued integral,  $J_{t}(G) = \int_{0}^{t}(G(s)dW(s), e_{j}(s) )_{g}$ is linear and continuous. Hence, $(\sigma_{n}(s, u_{n}(s))P_{n}, R)_{\mathcal{P}_{T}} \to (SdW (s), R)_{\mathcal{P}_{T}}$  for any  $R \in \mathcal{P}_{T}$ as $n \to \infty$ by the weak convergence of $\sigma_{n}(s, u_{n}(s))P_{n} \to S$ in $L^{2}_{Q}(H_{0,g}; H_{g} )$. Thus we can conclude that
		\begin{equation}
			\label{eq:1.26}
		\int_{0}^{T} (\sigma_{n}(s, u_{n}(s))dW_{n}(s), e_{j}(s))_{g} \to \int_{0}^{T} \phi(s)(S(s)dW(s), e_{j})_{g}, \quad as\,\, n \to \infty \,\,for\,\, each\,\, j.
		\end{equation}
		and
		\begin{equation}
			\label{eq:1.27}
			\int_{0}^{T} (F_{n}( u_{n}(s)), e_{j}(s))_{g} \to \int_{0}^{T} (F_{0}, e_{j}(s))_{g}ds, \quad as \,\, n \to \infty \,\, for\,\, each\,\, j.
		\end{equation}
		Taking the limit of each term in \eqref{eq:1.24}, we obtain
		\begin{equation}
			\label{eq:1.28}
			\begin{aligned}
			(\eta, e_{j})_{g}\phi(T) &= (u_{0}, e_{j})_{g} + \int_{0}^{T} (F_{0}, e_{j})_{g}\phi(s)ds + \int_{0}^{T} \left( u(s), \dfrac{de_{j}(s)}{ds} \right)_{g}ds 
			\\& + \int_{0}^{T} (f(s), e_{j}(s))_{g}ds + \sqrt{\varepsilon}\int_{0}^{T} \phi(s)(S(s)dW(s), e_{j})_{g}.
			\end{aligned}
		\end{equation}
		Choose the approximating sequence so that its derivative converges to $\delta_{t}$ in the distributional sense. Thus, for any $\psi \in C([0,T]; H_{g})$, we have
		\begin{equation}
			\label{eq:1.29}
			\underset{k \to \infty}{\rm{lim}} \int_{0}^{T} \phi'_{k}(s) \psi(s)ds  = -\underset{k \to \infty}{\rm{lim}} \int_{0}^{T} \phi_{k}(s) \psi'(s)ds= -\psi(t).
		\end{equation}
		Choose functions $\{\phi_{k}\}$ such that $\phi_{k} \to  \textbf{1}_{[0,T]}$ and $\phi'_{k} \to \delta_{t}$ weakly as $k \to \infty$. Substituting $\phi_{k}$ for $\phi$ in \eqref{eq:1.28} and then letting $k \to \infty$, we get that
		\begin{equation}
			\label{eq:1.30}
			(u(t), e_{j})_{g} = (u_{0}, e_{j})_{g} + \int_{0}^{T}(F_{0}, e_{j})_{g}ds + \sqrt{\varepsilon}\int_{0}^{T}(S(s)dW(s), e_{j})_{g}.
		\end{equation}
		When $t < T$ with $(u_{T} , e_{ j} )_{g} = (\eta, e_{j})_{g}$ for all $j$, we obtain 
		\begin{equation}
			\label{eq:1.31}
			u(t) = u(0) + \int_{0}^{t}F_{0}(s)ds + \sqrt{\varepsilon}\int_{0}^{t}S(s)dW(s)
		\end{equation}
		with $u(T) = \eta$.
		\par
		Let $v \in L^{\infty}(\Omega_{T}; H_{m})$ for $m \le n$, and  define $r(t) = \frac{27}{\nu^{3}} \int_{0}^{t}\| v(s)\|^{4}_{L^{4}(\mathcal{O})}ds$. 	By using the monotonicity, we get
		\begin{equation}
			\label{eq:1.32}
			\begin{aligned}
			X_{n}(T) &:=  2{\textbf{E}}\int_{0}^{T} e^{-r(s)}(F(u_{n}(s))- F(v(s)), u_{n}(s)-v(s))_{g}ds 
				- {\textbf{E}}\int_{0}^{T} \dfrac{dr(s)}{dt}e^{-r(s)}\| u_{n}(s)-v(s)\|^{2}_{g} ds 
				\\& + {\textbf{E}}\int_{0}^{T}e^{-r(s)} \|\sigma_{n}(s,u_{n}(s)) -\sigma_{n}(s,v(s)) \|^{2}_{L_{Q}}ds
				\\& \le 0.
			\end{aligned}
		\end{equation}
		Write $X_{n}(T)$ = $Y_{n} + Z_{n}$, where
		\begin{equation}
			\label{eq:1.33}
			\begin{aligned}
				Y_{n} &= 2{\textbf{E}}\int_{0}^{T} (F(u_{n}(s)), u_{n}(s))_{g}e^{-r(s)}ds - {\textbf{E}}\int_{0}^{T} \dfrac{dr(s)}{dt}e^{-r(s)}\| u_{n}(s)|^{2}_{g} ds 
				\\& + {\textbf{E}}\int_{0}^{T}e^{-r(s)} \|\sigma_{n}(s,u_{n}(s))\|^{2}_{L_{Q}}ds,
			\end{aligned}
		\end{equation}
		and
		\begin{equation}
			\label{eq:1.34}
			\begin{aligned}
			Z_{n} &= 2{\textbf{E}}\int_{0}^{T} \left( (F(v(s)), v(s))_{g} -(F(v(s)), u_{n}(s))_{g} -(F(u_{n}(s), v(s))_{g}) \right)e^{-r(s)}ds 
			\\& - {\textbf{E}}\int_{0}^{T} \dfrac{dr(s)}{dt}e^{-r(s)} (\| v(s)\|^{2}_{H_{g}} - 2(u_{n}(s),v(s))_{g})ds 
			\\&  +{\textbf{E}}\int_{0}^{T}e^{-r(s)} \{ \|\sigma_{n}(s,v(s))\|^{2}_{L_{Q}} - 2(\sigma_{n}(u_{n}(s)), \sigma_{n}(v(s))_{L_{Q}}) \}ds.
			\end{aligned}
		\end{equation}
		By It\^o formula, we obtain 
		\begin{equation}
			\label{eq:1.35}
			\begin{aligned}
			e^{-r(T)}\|u_{n}(T)\|^{2}_{g} - \|u_{n}(0)\|^{2}_{g} &=  2{\textbf{E}} \int_{0}^{T}e^{-r(s)}(F(u_{n}(s)), u_{n}(s))_{g}ds - {\textbf{E}} \int_{0}^{T}\frac{dr(s)}{dt}e^{-r(s)} \|u_{n}(s)\|^{2}_{g}ds 
			\\&+ {\textbf{E}} \int_{0}^{T}e^{-r(s)}\|\sigma_{n}(s,u_{n}(s))\|^{2}_{L_{Q}}ds + 2{\textbf{E}} \int_{0}^{T}e^{-r(s)}(f(s), u_{n}(s))_{g}ds.	
			\end{aligned}
		\end{equation}
		By the lower-semicontinuity of $\|u_n(T)\|^{2}_{g}$, we obtain 
		\begin{equation}
			\label{eq:1.36}
			\begin{aligned}
			\underset{n\to \infty}{\rm{lim}} \rm{inf}Y_{n} &= {\textbf{E}}e^{-r(T)}\|u_{n}(T)\|^{2}_{g} - \|u_{n}(0)\|^{2}_{g} \ge {\textbf{E}}e^{-r(T)}\|u_{n}(T)\|^{2}_{g} - \|u(0)\|^{2}_{g} 
			\\&=  2{\textbf{E}} \int_{0}^{T}e^{-r(s)}(F(u_{0}(s)), u(s))_{g}ds - {\textbf{E}} \int_{0}^{T}\frac{dr(s)}{dt}e^{-r(s)} \|u(s)\|^{2}_{g}ds 
		\\& + {\textbf{E}} \int_{0}^{T}e^{-r(s)}\|S(s)\|^{2}_{L_{Q}}ds + {\textbf{E}} \int_{0}^{T}e^{-r(s)}(S(s)dW(s), u(s))_{g}.
			\end{aligned}
		\end{equation}
		In \eqref{eq:1.34}, by taking the limit for each term, we obtain
		\begin{equation}
			\label{eq:1.37}
			\begin{aligned}
		\underset{n\to \infty}{\rm{lim}} Z_{n} &= 2{\textbf{E}}\int_{0}^{T} \left[ (F(v(s)), v(s))_{g} -(F(v(s)), u(s))_{g} -(F_{0}(s), v(s))_{g} \right] e^{-r(s)}ds 
		\\& - {\textbf{E}}\int_{0}^{T} \dfrac{dr(s)}{ds}e^{-r(s)} (\| v(s)\|^{2}_{{g}} - 2(u(s),v(s))_{g})ds 
		\\&  + {\textbf{E}}\int_{0}^{T}e^{-r(s)}  \left(\|\sigma(s,v(s))\|^{2}_{L_{Q}} - 2(S(s), \sigma(v(s))_{L_{Q}}) \right)ds.	
			\end{aligned}
		\end{equation}
		The monotonicity condition and $X_{n} = Y_{n} + Z_{n} \le 0$ imply
		\begin{equation}
			\label{eq:1.38}
			\begin{aligned}
				&2{\textbf{E}}\int_{0}^{T} e^{-r(s)}(F_{0}(s) - F(v(s)), u(s) - v(s))_{g} ds  - {\textbf{E}}\int_{0}^{T} \dfrac{dr(s)}{ds}e^{-r(s)} \| u(s) - v(s)\|^{2}_{g}ds 
				\\&+ {\textbf{E}}\int_{0}^{T}e^{-r(s)}\|S(s) - \sigma(s,v(s))\|^{2}_{L_{Q}}ds
				\\& \le \underset{n\to \infty}{\rm{lim}}\rm{inf}\, X_{n} \le 0.
			\end{aligned}
		\end{equation}
		Setting $v = u$ gives $S(s) = \sigma(s, u(s))$. Next, set $v = u - \lambda w$ with $\lambda > 0$ to obtain 
		\begin{equation}
			\label{eq:1.39}
			\begin{aligned}
				&2\lambda{\textbf{E}}\int_{0}^{T} e^{-r(s)}(F_{0}(s) - F(u(s) - \lambda w(s)), w(s))_{g} ds + \lambda^{2}{\textbf{E}}\int_{0}^{T} \dfrac{dr(s)}{ds}e^{-r(s)} \| w(s)\|^{2}_{g}ds
				\\& + \varepsilon L \lambda^{2}\int_{0}^{T}e^{-r(s)}\|w(s)\|^{2}_{V_{g}}  \le 0.	
			\end{aligned}
		\end{equation}
		Dividing by $\lambda$ on both sides of the inequality above, and letting $\lambda \to 0$, we can conclude
		\begin{equation}
			\label{eq:1.40}
			{\textbf{E}}\int_{0}^{T}(F_{0}(s) - F(u(s)), w(s))_{g}ds \le 0.
		\end{equation}
		Since $w$ is arbitrary, $F_{0}$ = $F(u(s))$. Thus the existence of a strong solution has been proved.
		\par
		We next prove pathwise uniqueness. Let $u_{1}$ and $u_{2}$ be two solutions  of the $g$-Navier--Stokes equation  \eqref{eq:1.23}. For $i = 1, 2$, and $N > 0$, define $\tau^{ i}_{ N} := \inf\{t \le T: \|u_{i} (t)\|_{g} \ge N \}$. Set $\tau_{N} = \min\{\tau^{1}  _{N}, \tau^{2}_{N}\}$. By Chebyshev's inequality, we obtain 
		\begin{equation}
			\label{eq:1.41}
			P\{\tau_{N}  < T \} \le P\{\max \{\underset{t\in [0,T]}{\rm{sup}} \|u_{1}(t) \|_{g}, \underset{t\in [0,T]}{\rm{sup}} \|u_{2}(t) \|_{g} \}  \ge N\} \le \dfrac{C}{N^{2}}.
		\end{equation}
		 Thus $\underset{N \to \infty}{\rm{lim}} \tau_{N}= T$ a.s. 
		\par
		By \eqref{eq:1.23} and It\^o formula, we have for $s \in [0, t \wedge \tau_{N}]$,
		\begin{equation}
			\label{eq:1.42}
			\begin{aligned}
		&\| u_{1}( t \wedge \tau_{N} ) - u_{2}( t \wedge \tau_{N} ) \|^{2}_{g} + 2\nu \int_{0}^{t \wedge \tau_{N}} \|u_{1}(s)- u_{2}(s)\|^{2}_{V_{g}}ds 
		\\& + 2\int_{0}^{t \wedge \tau_{N}} (B_{g}(u_{1}(s), u_{1}(s))- B_{g}(u_{2}(s), u_{2}(s)), u_{1}(s)- u_{2}(s))_{g} ds 
		\\& + 2\nu \int_{0}^{t \wedge \tau_{N}} (R_{g}(u_{1}(s)) - R_{g}(u_{2}(s)), u_{1}(s)- u_{2}(s))_{g} ds 
		\\& =    2 \sqrt{\varepsilon} \int_{0}^{t \wedge \tau_{N}} (u_{1}(s)- u_{2}(s), (\sigma(s,u_{1}(s))- \sigma(s,u_{2}(s)))  dW(s))_{g} 
		\\& + \varepsilon \int_{0}^{t \wedge \tau_{N}} \|\sigma(s,u_{1}(s))- \sigma(s,u_{2}(s))\|^{2}_{L_{Q}} ds.
			\end{aligned}
		\end{equation}
		We now estimate the third term on the left-hand side of \eqref{eq:1.42}. By \eqref{eq:2.3} and the bilinearity of $B_{g}$, we have
		\begin{equation}
			\label{eq:1.44}
			\begin{aligned}
			&\left| b_{g}(u_{1}, u_{1}, u_{1} - u_{2}) - b_{g}(u_{2}, u_{2}, u_{1} - u_{2}) \right| 
			\\&\le | b_{g}(u_{1}, u_{1}, u_{1} - u_{2}) - b_{g}(u_{2}, u_{1}, u_{1} - u_{2}) |
			 + |b_{g}(u_{2}, u_{1}, u_{1} - u_{2}) - b_{g}(u_{2}, u_{2}, u_{1} - u_{2})|
			\\& = |b_{g}(u_{1} - u_{2}, u_{1}, u_{1} - u_{2})| + |b_{g}(u_{2}, u_{1} - u_{2}, u_{1} - u_{2})|.
		\end{aligned}
		\end{equation}
		By Ladyzhenskaya inequality and H\"older inequality, we obtain
		\begin{equation}
			\label{eq:1.45}
			\begin{aligned}
			|b_{g}(u_{1}(s) - u_{2}(s), u_{1} ,u_{1}(s) - u_{2}(s) )| & \le \int_{\mathcal{O}}^{}  |u_{1}(s) - u_{2}(s)| |\nabla u_{1}||u_{1}(s) - u_{2}(s)| gdx 
			\\& \le \|u_{1}(s) - u_{2}(s)\|^{2}_{L^{4}}\|\nabla u_{1}\|_{g}
			\\& \le \frac{\sqrt{2M_{0}}}{m_{0}}\|u_{1}(s) - u_{2}(s)\|_{g}\|u_{1}(s) - u_{2}(s)\|_{V_{g}} \|u_{1}\|_{V_{g}}
			\\& \le \frac{15\nu }{8}\|u_{1}(s) - u_{2}(s)\|^{2}_{V_{g}} + \frac{8 M_{0}}{15\nu m^{2}_{0}}\|u_{1}(s) - u_{2}(s)\|^{2}_{{g}} \|u_{1}\|^{2}_{V_{g}}.
			\end{aligned}
		\end{equation}
		For the fourth term on the left-hand side of \eqref{eq:1.42},  by \eqref{eq:2.10}, we get
		\begin{equation}
			\label{eq:1.46}
			\begin{aligned}
		2\nu \int_{0}^{t \wedge \tau_{N}} (R_{g}(u_{1}(s)) -R_{g}(u_{2}(s)), u_{1}(s) - u_{2}(s))_{g}ds  
		& \le 2\nu \int_{0}^{t \wedge \tau_{N}}\frac{\|\nabla g\|_{\infty}}{m_{0}\sqrt{\lambda_{1}}} \|u_{1}(s) - u_{2}(s)\|^{2}_{V_{g}}ds
		\\&\le \frac{\nu}{8} \int_{0}^{t \wedge \tau_{N}} \|u_{1}(s) - u_{2}(s)\|^{2}_{V_{g}}ds.
			\end{aligned}
		\end{equation}
		For the last term on the right-hand side of \eqref{eq:1.42}, by condition (\textbf{A2}), we have 
		\begin{equation}
			\label{eq:1.43}
			\begin{aligned}
			\varepsilon \int_{0}^{t \wedge \tau_{N}} \|\sigma(s,u_{1}(s))- \sigma(s,u_{2}(s))\|^{2}_{L_{Q}} ds  \le \varepsilon L\, \int_{0}^{t \wedge \tau_{N}}  \|u_{1}(s) - u_{2}(s)\|^{2}_{g}ds. 
			\end{aligned}
		\end{equation}
		Substituting \eqref{eq:1.45} - \eqref{eq:1.43} into \eqref{eq:1.42}, and taking the expectation, we obtain
		\begin{equation}
			\label{eq:1.47}
			\begin{aligned}
	 \textbf{E} \| u_{1}( t \wedge \tau_{N} ) - u_{2}( t \wedge \tau_{N} ) \|^{2}_{g} 	\le \textbf{E} \int_{0}^{t \wedge \tau_{N}} \|u_{1}(s) - u_{2}(s)\|^{2}_{g} \left( \frac{8 M_{0}}{15 \nu m^{2}_{0}}\|u_{1} (s) \|^{2}_{V_{g}} + \varepsilon L  \right)ds. 
			\end{aligned}
		\end{equation}
		Finally, using Gronwall inequality we know that
		\begin{equation}
			\label{eq:4.58}
			\begin{aligned}
			\textbf{E} \| u_{1}( t \wedge \tau_{N} ) - u_{2}( t \wedge \tau_{N} ) \|^{2}_{g} = 0.
			\end{aligned}
		\end{equation}
		Taking $N \to \infty$ and applying Fatou’s lemma, the pathwise uniqueness can be proved. This proof is completed.

\label{drfn:1.8}(Rate function). 
	A function $I: \mathcal{E} \to [0, \infty]$ is called a rate function on $\mathcal{E}$ if it is lower semicontinuous in $\mathcal{E}$. A rate function $I$ on $\mathcal{E}$ is said to be a good rate function on $\mathcal{E}$ if for every $0 \le C < \infty$, the level set $\{x \in \mathcal{E}: I(x) \le C\}$ is a compact subset of $\mathcal{E}$.

\label{drfn:1.9}(Large Deviation Principle). 
	The family $\{u^{\varepsilon}\}$ is said to satisfy the LDP in $\mathcal{E}$ with a rate function $I: \mathcal{E} \to [0, \infty]$, if for every Borel subset $A$ of $\mathcal{E}$,
	\begin{equation}
		\label{eq:1.53}
		\begin{aligned}
			- \underset{x\in A^{\circ}}{\inf}\, I(x)  \le \underset{\varepsilon \to 0}{\lim\, \inf} \varepsilon\,  \log\mathbb{P} (u^{\varepsilon} \in A) \le \underset{\varepsilon \to 0}{\lim\, \sup}\,  \varepsilon\,  \log\mathbb{P}(u^{\varepsilon} \in A) \le - \underset{x\in \bar{A}}{\inf}\, I(x),
		\end{aligned}
	\end{equation}
	where $A^{\circ}$ and $\bar{A}$ are the interior and the closure of $A$ in $\mathcal{E}$, respectively.

\label{drfn:1.10}(Laplace principle). The family $\{u^{\varepsilon}\}$ is said to satisfy the Laplace principle in $\mathcal{E}$ with a rate function $I: \mathcal{E} \to [0, \infty]$ if for all bounded and continuous $\phi:  \mathcal{E} \to \mathbb{R}$,

	\end{equation}

Under conditions (\textbf{A1})--(\textbf{A3}), by the argument of [\cite{ref Da Prato3}, Theorem 7.10], one can show that for every $T > 0$, $v \in L^{2}(0, T; H_{0,g})$ and $u_{0} \in H_{g}$, problem \eqref{eq:1.56}  has a unique solution $u_{v}\in C([0, T], H_{g}) \cap L^{2}( 0,T; V_{g})$. 
\par 
Next, we prove \eqref{eq:1.84}. By \eqref{eq:2.10}, \eqref{eq:1.56}, we get
\begin{equation}
	\label{eq:1.86}
	\begin{aligned}
	d\| u_{v}(t)\|^{2}_{g} + 2\nu \|u_{v}(t)\|^{2}_{V_{g}}dt \le \frac{\nu }{8} \|u_{v}(t)\|^{2}_{V_{g}}dt + 2(f(t), u_{v}(t))_{g} dt
	 +   2(\sigma(t,  u_{v}(t))v(t),  u_{v}(t))_{g}dt.
	\end{aligned}
\end{equation}
For the second term on the right-hand side of \eqref{eq:1.86}, by H\"older inequality, we obtain
\begin{equation}
	\label{eq:4.60}
	\begin{aligned}
	2\left|(f(t), u_{v}(t))_{g}  \right| \le \dfrac{\nu }{8}\|u_{v}(t)\|^{2}_{V_{g}} + \frac{8}{\nu} \|(f(t)\|^{2}_{V^{*}}. 
	\end{aligned}
\end{equation}
For the last term on the right-hand side of \eqref{eq:1.86}, by (\textbf{A2}) and  Young's inequality, we find
\begin{equation}
	\label{eq:1.87}
	\begin{aligned}
	2(\sigma(t,  u_{v}(t))v(t),  u_{v}(t))_{g} &= 2\int_{\mathcal{O}}^{}\sigma(t,  u_{v}(t))v(t) \cdot u_{v}(t)\,gdx 
	\\& \le \frac{3\nu }{4KR_{2}}\|\sigma(t,  u_{v}(t))\|^{2}_{L_{Q}} \|v(t)\|^{2}_{L^{2}(0,T; H_{0,g})} + \frac{4KR_{2}}{3\nu }\|u_{v}(t)\|^{2}_{g}
	\\& \le \frac{3\nu }{4}\|u_{v}(t)\|^{2}_{V_{g}} + \frac{3\nu }{4} + \frac{4KR_{2}}{3\nu }\|u_{v}(t)\|^{2}_{{g}}.

		+  \frac{2\lambda^{1/2}_{1}c_{4}}{k}\|\nabla u_{v}\|^{2} 
		\\& + \frac{c_{6}}{m_{0}k} \left(\|\nabla u_{v}\|^{2} + \|u_{v}\|^{6}\right)+ \frac{C^{2}\sqrt{\varepsilon L}R_{3}M_{0}}{\lambda_{1} k}\|u_{v}\|^{2},
		\end{aligned}
	\end{equation}
	where $c_{4}, c_{6} > 0$ depend on $C, \lambda_{1}$, and $\varepsilon$, but not on $k$. \par 
	Then, by Gronwall's inequality, we get for any $t \in [0, T]$, 
	\begin{equation}
		\label{eq:3.98}
		\begin{aligned}
	&	\|\varrho_{k} u_{v}(t)\|^{2} \le \|\varrho_{k} u_{0}\|^{2}e^{- \frac{\nu \alpha}{2}t} 
		\\& + e^{ \frac{\nu \alpha}{2}t} \times \int_{0}^{t} \left( \frac{4}{k \lambda^{1/2}_{1} }\| u_{v_{t}}(s)\|\|u_{v}(s)\|  + \frac{1}{8}\| u_{v_{t}}(s)\|\|u_{v}(s)\| + \frac{2\lambda^{1/2}_{1}c_{4}}{k}\|\nabla u_{v}\|^{2} \right)ds
		\\& + e^{ \frac{\nu \alpha}{2}t} \times	\left(  \frac{c_{6}}{m_{0} k}(\|\nabla u_{v}(s)\|^{2}   + \|u_{v}(s)\|^{6} )\right)ds
		\\& +  e^{ \frac{\nu \alpha}{2}t} \times \int_{0}^{t} \left(  \frac{C^{2}\sqrt{\varepsilon L}R_{3}M_{0}}{\lambda_{1} k^{2}}\|u_{v}(s)\|^{2} \right)ds.
		\end{aligned}
	\end{equation}
	For the ﬁrst term on the right-hand side of \eqref{eq:3.98}, by $u_{0} \in H_{g}$ and Lebesgue's Dominated Convergence Theorem, we can obtain that 
	\begin{equation}
		\label{eq:4.73}
		\begin{aligned}
			e^{\frac{- \nu \alpha }{2} t}\|\varrho_{k} u_{0}\|^{2}  \le \int_{\{(x_{1},x_{2}) \in \mathcal{O}:|x_{1}| \ge \frac{k}{2} \}}^{} |u(0,x)|^{2}dx \to 0,\,\, as\,\, k \to \infty.
		\end{aligned}
	\end{equation}
For the second and third term on the right-hand side of \eqref{eq:3.98}, by Lemma \ref{lem:1.13} and   Lemma 2.2 in Wang \cite{ref Wang. B4} we have 
	\begin{equation}
		\label{eq:4.79}
		\begin{aligned}
		e^{ \frac{\nu \alpha}{2}t} \times \int_{0}^{t} \left( \frac{4}{\lambda^{1/2}_{1} k}\| u_{v_{t}}(s)\|\|u_{v}(s)\| + \frac{1}{8}\| u_{v_{t}}(s)\|\|u_{v}(s)\|  + \frac{2\lambda^{1/2}_{1}c_{4}}{k}\|\nabla u_{v}\|^{2} \right) ds \to 0,\,\, as\,\, k \to \infty.
		\end{aligned}
	\end{equation}
	For the fourth  term on the right-hand side of \eqref{eq:3.98}, by Proposition \ref{prop:1.22} and   Lemma 2.2 in Wang \cite{ref Wang. B4} we have
	\begin{equation}
		\label{eq:3.99}
		\begin{aligned}
	 e^{ \frac{\nu \alpha}{2}t} \times  \left(\int_{0}^{t} \frac{c_{6}}{m_{0}k}\left(  \|\nabla u_{v}(s)\|^{2} + \|u_{v}(s)\|^{6} \right) ds\right) \to 0, \,as\,\, k \to \infty. 
		\end{aligned}
	\end{equation}
	For the last term on the right-hand side of \eqref{eq:3.98}, by $\| v \|_{L^{2}(0, T ; H_{0,g})} \le R_{3}$ and Lemma \ref{lem:1.13} we have
	\begin{equation}
		\label{eq:4.80}
		\begin{aligned}
		e^{ \frac{\nu \alpha}{2}t} \times \int_{0}^{t} \left(  \frac{C^{2}\sqrt{\varepsilon L}R_{3}M_{0}}{\lambda_{1} k^{2}}\|u_{v}(s)\|^{2} \right)ds \to 0, \,as\,\, k \to \infty. 
		\end{aligned}
	\end{equation}
	It follows from \eqref{eq:4.73}-\eqref{eq:4.80} that for every $\varepsilon' > 0 $, there exists $K^{*} = K^{*}(\alpha, \lambda_{1}, T, u_{0}, R_{3}, \varepsilon') > 0$ such that for all $k \ge K^{*}$ and $t \in [0, T]$,
	\begin{equation}
		\label{eq:3.100}
		\begin{aligned}
			\int_{|x| \ge k}^{} |u_{v}(t,x)|^{2}dx \le  \|\varrho_{k} u_{v}(t)\|^{2}  < \varepsilon'.
		\end{aligned}
	\end{equation}
	This proof is complete.

	For $m \in \mathbb{N}$, define $\textbf{Z}_{m} = \{u \in H_{g}: \|u\|_{g} \le  \textbf{C}_{4}, \,\, u(x)=0\,\, for\,\, almost\,\, all \,\, |x| \ge m \}$. Therefore, for every $m\in \mathbb{N}$ and $t \in [0,T]$,

	By \eqref{eq:4.13} and \eqref{eq:4.14}, we obtain that $\{\tilde{\textbf{{u}}}^{m}_{v_{n_{k}}}(t) + \hat{\textbf{u}}^{m}_{v_{n_{k}}}(t)\}^{\infty}_{k=1}$ has a finite open cover with radius $\varepsilon'$ in $V_{g}$, which implies that $\{u_{v_{n_{k}}}(t)\}^{\infty}_{k=1}$ is precompact in $V_{g}$.

	Therefore, by \eqref{eq:4.2} and the Arzela-Ascoli Theorem imply the asserted strong convergence in $C([0,T];H_g)$.

	\textbf{Step 3: prove the convergence}.
	\begin{equation}

		\end{aligned}

	To prove \eqref{eq:4.62}, by \eqref{eq:4.3} we only need to show $\tilde{\mathbf{u}} = u_{v}$. Equation \eqref{eq:3.35} gives, for every $t \in [0,T]$ and $\eta \in H_{g}$,
	\begin{equation}

		\\& = (u_{0}, \eta)_{g} + \int_{0}^{t}(f(s), \eta)_{g}ds + \int_{0}^{t}(\sigma(s,u_{v_{n_{k}}}(s))v_{n_{k}}(s), \eta)_{g}ds.

	We now prove $\tilde{\mathbf{u}} = u_{v}$ by taking the limit of \eqref{eq:4.63} as $k \to \infty$. We ﬁrst deal with the nonlinear term on the left-hand side of \eqref{eq:4.63}. By \eqref{eq:4.3} we infer that, up to a subsequence
	\[
			u_{v_{n_{k}}}  \to \tilde{\mathbf{u}} \,\,\,a.e.\,\, on \,\,  (0, T ) \times \mathbb{R}^{2}.
	\]	
	By the definition of $F$ in Proposition \ref{prop:1.4} and its local monotonicity property, we obtain 
	\begin{equation}
		\label{eq:4.89}

	\end{equation}
	For the last term on the right-hand side of \eqref{eq:4.63}, we have for any $t \in [0,T]$, and $\eta \in H_{g}$,
	\begin{equation}
		\label{eq:4.65}
		\begin{aligned}
		&\int_{0}^{t}(\sigma(s,u_{v_{n_{k}}}(s))v_{n_{k}}(s), \eta)_{g}ds 
		\\& = \int_{0}^{t}((\sigma(s,u_{v_{n_{k}}}(s)) - \sigma(s,\tilde{\mathbf{u}}(s) ))v_{n_{k}}(s), \eta)_{g}ds + \int_{0}^{t}(\sigma(s,\tilde{\mathbf{u}}(s))v_{n_{k}}(s), \eta)_{g}ds. 
		\end{aligned}
	\end{equation}
	By \eqref{eq:4.3}, condition (\textbf{A3}) and H\"older inequality, we obtain
	\begin{equation}

		\\& \le \|\eta\| \int_{0}^{t} \|\sigma(s,u_{v_{n_{k}}}(s)) - \sigma(s,\tilde{\mathbf{u}}(s)\|_{L_{Q}} \|v_{n_{k}}(s)\|_{L^{2}(0,T; H_{0,g})} ds
		\\& \le  \sqrt{M L}  \|\eta\| \left( \int_{0}^{t} \| u_{v_{n_{k}}}(s) - \tilde{\mathbf{u}}(s)\|^{2}_{g} ds      \right)^{1/2} \,\, \to\,\, 0\,\,as\,\,k\to\infty.

	\mathcal{T}(\omega) = \int_{0}^{t}\sigma(s,\tilde{\mathbf{u}}(s))\omega(s) g ds,\,\, \forall\, \omega \in L^{2}(0,T; H_{0,g}). 
	\]
	It follows from condition (\textbf{A2}) and \eqref{eq:4.2} that $\mathcal{T}$ is well-defined, and

    \| \mathcal{T}(\omega) \|^{2}_{g} 
    & \le \left(\int_{0}^{t}\| \sigma(s,\tilde{\mathbf{u}}(s))\|^{2}_{L^{2}(H_{0,g}; H_{g})} ds   \right)\left(\int_{0}^{t}\|\omega(s)\|^{2}_{L^{2}(0,T; H_{0,g})} \right)
    \\& \le \left(\int_{0}^{t}\left(K(1+ \|\tilde{\mathbf{u}}(s)\|^{2}_{V_{g}}) \right)  ds   \right) \left(\int_{0}^{t}\|\omega(s)\|^{2}_{L^{2}(0,T; H_{0,g})} \right),
	\end{align*}
	which shows $\mathcal{T}$ is a bounded linear operator and thus it is weakly continuous. Since $v_{n_{k}} \rightharpoonup v$ in $ L^{2}(0,T; H_{0,g})$, we have $\mathcal{T}(v_{n_{k}}) \to \mathcal{T}(v)$ weakly in $H_{g}$. Hence, for all $t \in [0,T]$, 
	\begin{equation}

	It follows from \eqref{eq:4.65}--\eqref{eq:4.68} that we have for any $t\in [0,T]$ and $\eta \in H_{g}$,
\begin{equation}

	\begin{aligned}
	\underset{k \to \infty}{\lim }\int_{0}^{t}(\sigma(s, u_{v_{n_{k}}}(s))v_{n_{k}}(s) ,\eta  )_{g}ds = \int_{0}^{t}(\sigma(s,\tilde{\mathbf{u}}(s))v(s), \eta)_{g}ds.
	\end{aligned}
\end{equation}
	Now, letting $k \to \infty$ in \eqref{eq:4.63}, by \eqref{eq:4.2}, \eqref{eq:4.89} and \eqref{eq:4.87}, we deduce that for any $t \in [0,T]$ and $\eta \in H_{g}$, 
	\begin{equation}

		\begin{aligned} 
			&(\tilde{\mathbf{u}}(t), \eta)_{g} + \int_{0}^{t} (\nu A_{g}\tilde{\mathbf{u}}(s), \eta)_{g} ds + \int_{0}^{t} (B_{g}(\tilde{\mathbf{u}}(s), \tilde{\mathbf{u}}(s)), \eta)_{g} ds

		\label{eq:4.18}
		\begin{aligned}
		&2 \left| \int_{0}^{t} (\sigma(s,u^{\varepsilon}_{v^{\varepsilon}}(s)v^{\varepsilon}(s) - \sigma(s,u_{v^{\varepsilon}}(s))v^{\varepsilon}(s), u^{\varepsilon}_{v^{\varepsilon}}(s) - u_{v^{\varepsilon}}(s))_{g}ds  \right|
		\\& \le 2\int_{0}^{t}\|\sigma(s,u^{\varepsilon}_{v^{\varepsilon}}(s))-\sigma(s,u_{v^{\varepsilon}}(s)) \|_{L_{Q}} \|v^{\varepsilon}(s)\|_{L^{2}(0,T; H_{0,g})} \|u^{\varepsilon}_{v^{\varepsilon}}(s) - u_{v^{\varepsilon}}(s)\|_{g}ds
		\\& \le 2\sqrt{L} \int_{0}^{t}\|v^{\varepsilon}(s)\|_{L^{2}(0,T; H_{0,g})} \|u^{\varepsilon}_{v^{\varepsilon}}(s) - u_{v^{\varepsilon}}(s)\|^{2}_{g}ds. 
		\end{aligned}
	\end{equation}
	It follows from \eqref{eq:4.16} - \eqref{eq:4.18} that for all $t \in [0,T]$,  
	\begin{equation}
		\label{eq:4.19}
		\begin{aligned}
		& \| u^{\varepsilon}_{v^{\varepsilon}}(t) - u_{v^{\varepsilon}}(t)\|^{2}_{g} + \nu  \int_{0}^{t}\| u^{\varepsilon}_{v^{\varepsilon}}(s) - u_{v^{\varepsilon}}(s) \|^{2}_{V_{g}} ds
		\\&  \le \frac{16M_{0}}{7\nu m^{2}_{0}}\int_{0}^{t} \|u^{\varepsilon}_{v^{\varepsilon}}(s) - u_{v^{\varepsilon}}(s)\|^{2}_{g}\| u_{v^{\varepsilon}}(s)\|^{2}_{V_{g}}ds + 2\sqrt{L} \int_{0}^{t}\|v^{\varepsilon}(s)\|_{L^{2}(0,T; H_{0,g})} \|u^{\varepsilon}_{v^{\varepsilon}}(s) - u_{v^{\varepsilon}}(s)\|^{2}_{g}ds 
		\\& +  \varepsilon \int_{0}^{t} \|\sigma(s,u^{\varepsilon}_{v^{\varepsilon}}(s))\|^{2}_{L_{Q}}ds + 2\sqrt{\varepsilon}\int_{0}^{t} (\sigma(s,u^{\varepsilon}(s)_{v^{\varepsilon}})dW(s),u^{\varepsilon}_{v^{\varepsilon}}(s) - u_{v^{\varepsilon}}(s))_{g}.
		\end{aligned}
	\end{equation}
		Given $ N >0$ and $\varepsilon \in (0,1)$, define a stopping time 
	\begin{equation}
		\label{eq:3.29}
		\begin{aligned}
			\tau_{N,\varepsilon} := inf\left\{t \ge 0:  \|u^{\varepsilon}_{v^{\varepsilon}}(t)\|^{2}_{g}  \ge  N\right\} \wedge T.
		\end{aligned}
	\end{equation}
	By \eqref{eq:4.19} we have for all $t \in [0,T]$,  
	\begin{equation}
		\label{eq:4.20}
		\begin{aligned}
		& \underset{0\le r\le t}{\sup} \left( \| u^{\varepsilon}_{v^{\varepsilon}}(r \wedge \tau_{N,\varepsilon}) - u_{v^{\varepsilon}}(r \wedge \tau_{N,\varepsilon})\|^{2}_{g} + \nu  \int_{0}^{r \wedge \tau_{N,\varepsilon}}\| u^{\varepsilon}_{v^{\varepsilon}}(s) - u_{v^{\varepsilon}}(s) \|^{2}_{V_{g}} ds \right)
		\\&  \le \int_{0}^{r \wedge \tau_{N,\varepsilon}} \left( \frac{16M_{0}}{7\nu m^{2}_{0}}\|u_{v^{\varepsilon}}(s)\|^{2}_{V_{g}} + 2\sqrt{L}\|v^{\varepsilon}(s)\|_{L^{2}(0,T; H_{0,g})}  \right) \underset{0 \le r \le s}{\sup}\|u^{\varepsilon}_{v^{\varepsilon}}(r \wedge \tau_{N,\varepsilon}) - u_{v^{\varepsilon}}(r \wedge \tau_{N,\varepsilon})\|^{2}_{g}ds 
		\\& +  \varepsilon \int_{0}^{r \wedge \tau_{N,\varepsilon}}\|\sigma(s,u^{\varepsilon}_{v^{\varepsilon}}(s))\|^{2}_{L_{Q}}ds + 2\sqrt{\varepsilon}\underset{0\le r\le t}{\sup}\int_{0}^{r \wedge \tau_{N,\varepsilon}} (\sigma(s,u^{\varepsilon}(s)_{v^{\varepsilon}})dW(s),u^{\varepsilon}_{v^{\varepsilon}}(s) - u_{v^{\varepsilon}}(s))_{g}.
		\end{aligned}
	\end{equation}
	By \eqref{eq:4.20} and Gronwall's inequality, we have that for any $t \in [0,T]$, 
	\begin{equation}
		\label{eq:4.21}
		\begin{aligned}
		& \underset{0\le r\le T}{\sup} \left( \| u^{\varepsilon}_{v^{\varepsilon}}(r \wedge \tau_{N,\varepsilon}) - u_{v^{\varepsilon}}(r \wedge \tau_{N,\varepsilon})\|^{2}_{g} + \nu  \int_{0}^{r \wedge \tau_{N,\varepsilon}}\| u^{\varepsilon}_{v^{\varepsilon}}(s) - u_{v^{\varepsilon}}(s) \|^{2}_{V_{g}} ds \right)
	    \\& \le  \varepsilon e^{c_{8}}  \int_{0}^{r \wedge \tau_{N,\varepsilon}} \|\sigma(s,u^{\varepsilon}_{v^{\varepsilon}}(s) )\|^{2}_{L_{Q}} ds
		+ 2\sqrt{\varepsilon}e^{c_{8}}  \underset{0\le r\le T}{\sup}\int_{0}^{T \wedge \tau_{N,\varepsilon}} (\sigma(s,u^{\varepsilon}_{v^{\varepsilon}}(s))dW(s),u^{\varepsilon}_{v^{\varepsilon}}(s) - u_{v^{\varepsilon}}(s))_{g}, 
		\end{aligned}
	\end{equation}
	where $c_{8} = (\frac{16M_{0}}{7m^{2}_{0}}C_{1} + 2\sqrt{LM})T > 0$.  For the first term on the right-hand side \eqref{eq:4.21}, by condition (\textbf{A2}) and \eqref{eq:4.38}, we have 
	\begin{equation}
		\label{eq:4.22}
		\begin{aligned}
		  \underset{\varepsilon \to 0}{\rm{lim}}\,\varepsilon e^{c_{8}}  \int_{0}^{T \wedge \tau_{N,\varepsilon}} \|\sigma(s,u^{\varepsilon}_{v^{\varepsilon}}(s) )\|^{2}_{L_{Q}} ds
		  \le \underset{\varepsilon \to 0}{\rm{lim}}\,\varepsilon e^{c_{8}} K\int_{0}^{T \wedge \tau_{N,\varepsilon}} (1+ \|u^{\varepsilon}_{v^{\varepsilon}}(s)\|^{2}_{V_{g}}) ds
		 = 0, \mathbb{P} - almost\,surely. 
		\end{aligned}
	\end{equation}
	In addition, by Lemma  \ref{lem:1.13}, we have that there exists $ C'_{1} = C'_{1}(M, T) > 0$ such that 
	\begin{equation}
		\label{eq:4.23}
		\begin{aligned}
			\underset{\varepsilon \in (0, 1)}{\sup} \|u_{v^{\varepsilon}} \|_{g} \le C'_{1}. 
		\end{aligned}
	\end{equation}
	For the last term on the right-hand side of \eqref{eq:4.21}, by \eqref{eq:4.22}, \eqref{eq:4.23} and Doob's maximal inequality, we obtain
	\begin{equation}
		\label{eq:4.24}
		\begin{aligned}
	& \underset{\varepsilon \to 0}{\rm{lim}}   \textbf{E} \left[ \underset{0\le r\le T}{\sup} \left|  \int_{0}^{r \wedge \tau_{N,\varepsilon}}2\sqrt{\varepsilon}\, e^{c_{8}}\, (\sigma(s,u^{\varepsilon}_{v^{\varepsilon}}(s))dW(s),u^{\varepsilon}_{v^{\varepsilon}}(s) - u_{v^{\varepsilon}}(s))_{g} \right|^{2} \right]
	\\& \le  \underset{\varepsilon \to 0}{\rm{lim}} 16 \varepsilon e^{2c_{8}}  \textbf{E} \left[\int_{0}^{T \wedge \tau_{N,\varepsilon}} \|u^{\varepsilon}_{v^{\varepsilon}}(s) - u_{v^{\varepsilon}}(s)\|^{2}_{g} \| \sigma(s,u^{\varepsilon}(s))\|^{2}_{L_{Q}}ds\right]
	\\& \le 16e^{2c_{8}} (N + C'_{1})^{2} \underset{\varepsilon \to 0}{\rm{lim}} \textbf{E} \left[\varepsilon \int_{0}^{ T \wedge \tau_{N,\varepsilon}} \| \sigma(s,u^{\varepsilon}(s))\|^{2}_{L_{Q}} ds\right]
	\\& =0.
		\end{aligned}
	\end{equation}
	It follows from \eqref{eq:4.21}, \eqref{eq:4.22} and \eqref{eq:4.24} that 
	\begin{equation}
		\label{eq:4.25}
		\begin{aligned}
		\underset{\varepsilon \to 0}{\rm{lim}}\,\,  \underset{0\le r\le T}{\sup} \|u^{\varepsilon}_{v^{\varepsilon}}(r \wedge \tau_{N,\varepsilon}) - u_{v^{\varepsilon}}(r \wedge \tau_{N,\varepsilon})\|^{2}_{g} = 0 \,\,in\, \, probability, 
		\end{aligned}
	\end{equation}
	and 
	\begin{equation}
		\label{eq:4.26}
		\begin{aligned}
			\underset{\varepsilon \to 0}{\rm{lim}}  \int_{0}^{T\wedge \tau_{N,\varepsilon}} \|u^{\varepsilon}_{v^{\varepsilon}}(s) - u_{v^{\varepsilon}}(s)\|^{2}_{V_{g}} ds = 0 \,\,in\, \, probability.
		\end{aligned}
	\end{equation}
	On the other hand by  \eqref{eq:2.32} and  \eqref{eq:3.29} we have for all $\varepsilon \in (0,1)$,  
	\begin{equation}
		\label{eq:4.27}
		\begin{aligned}
		\mathbb{P}(\tau_{N,\varepsilon} < T  ) \le \mathbb{P}\left(\underset{t \in [0,T]}{\sup} \|u^{\varepsilon}_{v^{\varepsilon}}(t)\|_{g} \ge N \right) \le \dfrac{1}{N^{2}}\textbf{E}\left( \underset{t \in [0,T]}{\sup} \|u^{\varepsilon}_{v^{\varepsilon}}(t)\|^{2}_{g}\right) \le \frac{M_{1}}{N^{2}},
		\end{aligned}
	\end{equation}
	where $M_{1} = M_{1}( T, M) >0$. It follows from \eqref{eq:4.27} that for every $\eta > 0$,
	\begin{align*}
		& \mathbb{P}\left(\underset{t \in [0,T]}{\sup}\|u^{\varepsilon}_{v^{\varepsilon}}(t) - u_{v^{\varepsilon}}(t) \|_{g}  > \eta \right) 
		\\& \le \mathbb{P}\left(\underset{t \in [0,T]}{\sup}\|u^{\varepsilon}_{v^{\varepsilon}}(t) - u_{v^{\varepsilon}}(t) \|_{g}  > \eta, \tau_{N,\varepsilon} =T \right) + \mathbb{P}\left(\underset{t \in [0,T]}{\sup}\|u^{\varepsilon}_{v^{\varepsilon}}(t) - u_{v^{\varepsilon}}(t) \|_{g}  > \eta, \tau_{N,\varepsilon} < T \right) 
		\\& \le \mathbb{P}\left(\underset{t \in [0,T]}{\sup}\|u^{\varepsilon}_{v^{\varepsilon}}(t \wedge \tau_{N,\varepsilon}) - u_{v^{\varepsilon}}(t \wedge \tau_{N,\varepsilon}) \|_{g}  > \eta \right) + \frac{M_{1}}{N^{2}}.
	\end{align*}
	First letting $\varepsilon \to 0$ and  then $N \to \infty$, we get from \eqref{eq:4.25} that 
	\begin{equation}
		\label{eq:4.29}
		\begin{aligned}
		\underset{\varepsilon \to 0}{\rm{lim}} \left( u^{\varepsilon}_{v^{\varepsilon}} - u_{v^{\varepsilon}} \right) = 0 \,\, in\,\, C([0,T]; H_{g}) \,\, in\, probability.
		\end{aligned}
	\end{equation}
	Similarly, by \eqref{eq:4.27} we have, for every $\eta > 0$, 
	\begin{align*}
		& \mathbb{P}\left(\int_{0}^{T} \|u^{\varepsilon}_{v^{\varepsilon}}(t) - u_{v^{\varepsilon}}(t) \|^{2}_{V_{g}}dt  > \eta^{2} \right) 
		\\& \le \mathbb{P}\left(\int_{0}^{T} \|u^{\varepsilon}_{v^{\varepsilon}}(t) - u_{v^{\varepsilon}}(t) \|^{2}_{V_{g}} dt > \eta^{2}, \tau_{N,\varepsilon} =T \right) + \mathbb{P}\left(\int_{0}^{T} \|u^{\varepsilon}_{v^{\varepsilon}}(t) - u_{v^{\varepsilon}}(t) \|^{2}_{V_{g}}dt  > \eta^{2}, \tau_{N,\varepsilon} < T \right) 
		\\& \le \mathbb{P}\left(\int_{0}^{T \wedge \tau_{N,\varepsilon}} \|u^{\varepsilon}_{v^{\varepsilon}}(t) - u_{v^{\varepsilon}}(t) \|^{2}_{V_{g}}dt  > \eta^{2} \right) + \frac{M_{1}}{N^{2}}.
		\end{align*}
		First letting $\varepsilon \to 0$ and then $N \to \infty$, we get from \eqref{eq:4.26} that 
		\begin{equation}
			\label{eq:4.31}
			\begin{aligned}
				\underset{\varepsilon \to 0}{\rm{lim}} \left( u^{\varepsilon}_{v^{\varepsilon}} - u_{v^{\varepsilon}} \right) = 0 \,\, in\,\, L^{2}(0,T; V_{g}) \,\, in\, probability.
			\end{aligned}
		\end{equation}
		It follows from \eqref{eq:4.29} and \eqref{eq:4.31} that 
		\begin{equation}
			\label{eq:4.32}
			\begin{aligned}
				\underset{\varepsilon \to 0}{\rm{lim}} \left( u^{\varepsilon}_{v^{\varepsilon}} - u_{v^{\varepsilon}} \right) = 0 \,\, in\,\, C([0,T]; H_{g}) \cap  L^{2}(0,T; V_{g}) \,\, in\, probability.
			\end{aligned}
		\end{equation}
		By assumption, we know $\{ v^{\varepsilon} \}$ converges in distribution to $v$ as $\mathcal{S}_{M}$-valued random element, and hence by Skorokhod's Theorem, there exist a probability space $(\tilde{\Omega}, \tilde{\mathcal{F}}, \tilde{\mathbb{P}}  )$ and $\mathcal{S}_{M}$-valued random variables $\tilde{v}^{\varepsilon}$ and $\tilde{v}$ on $(\tilde{\Omega}, \tilde{\mathcal{F}}, \tilde{\mathbb{P}}  )$ such that the law of $\tilde{v}^{\varepsilon}$ and $\tilde{v}$ are equal to that of $v^{\varepsilon}$ and $v$ respectively. Furthermore, $\{\tilde{v}^{\varepsilon}\}$ converges to $\tilde{v}$ almost surely in $\mathcal{S}_{M}$, it follows from Lemma \ref{lem:3.6} that
		\begin{equation}
			\label{eq:4.33}
			\begin{aligned}
			u_{\tilde{v}^{\varepsilon}} \to  u_{\tilde{v}}	\,\, in\,\, C([0,T]; H_{g}) \cap  L^{2}(0,T; V_{g}) \,\, almost\,\, surely,
			\end{aligned}
		\end{equation}
		and hence it is also convergent in distribution, which implies that 
\begin{equation}
	\label{eq:4.34}
	\begin{aligned}
		u_{v^{\varepsilon}} \to  u_{v}	\,\, in\,\, C([0,T]; H_{g}) \cap  L^{2}(0,T; V_{g}) \,\, in \,\, distribution.
	\end{aligned}
\end{equation}
	By \eqref{eq:4.33} and \eqref{eq:4.34} we infer that 
	\begin{equation}
		\label{eq:4.35}
		\begin{aligned}
			u^{\varepsilon}_{v^{\varepsilon}} \to  u_{v}	\,\, in\,\, C([0,T]; H_{g}) \cap  L^{2}(0,T; V_{g}) \,\, in \,\, distribution, 
		\end{aligned}
	\end{equation}
	as desired. The proof is thus completed.
