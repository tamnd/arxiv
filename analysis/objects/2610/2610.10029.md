<!-- head-1: body_md -->
\rm
We shall use the Einstein summation convention and the following notations:
<!-- end head-1 -->
<!-- lemma-1: body_md -->
\label{lemma2.1}
For any $u\in L^2({\mathbb R}^d)$ with $0\leq u \leq M_0$ a.e. we get that:
$$0\leq \eta=\mathcal{S}u \leq M_0 \quad \mbox{ a.e. in } {\mathbb R}^d.$$
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
We have that:
$$0\leq (\mathcal{S}u)(x)=\int_{{\mathbb R}^d}{\cal B} (x-y)u(y) \, dy \leq M_0 \int_{{\mathbb R}^d}{\cal B} (x-y) dy = M_0$$ 
a.e. $x\in {\mathbb R}^d$.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lemma2.2}
For any $\beta >0$ there exists a constant $C_{\beta} >0$ such that for any $u\in L^2_{\beta}({\mathbb R}^d)$ we have that
$$\|\mathcal{S}u\|_{2,\beta}\leq C_{\beta}\|u\|_{2,\beta} \quad \mbox{ and } \quad \|\nabla(\mathcal{S}u)\|_{2,\beta}\leq C_{\beta}\|u\|_{2,\beta}.$$
This implies that $\mathcal{S}:L^2_{\beta}({\mathbb R}^d)\rightarrow H^1_{\beta}({\mathbb R}^d)$ is a bounded operator.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
$$\langle x \rangle ^{\beta} |(\mathcal{S}u)(x)|\leq \int_{{\mathbb R}^d}\langle x \rangle ^{\beta} {\cal B} (x-y)|u(y)| \, dy $$
and by Proposition \ref{appendix1} (in Appendix)
$$\leq C_{\beta} \int_{{\mathbb R}^d} \langle x-y \rangle ^{\beta} {\cal B} (x-y) \langle y \rangle ^{\beta} |u(y)| \, dy = C_{\beta}\left( {\cal B} _{\beta}*(\langle y \rangle ^{\beta}|u(y)|)\right)(x)$$
(here ${\cal B} _{\beta} = \langle \cdot \rangle ^{\beta} {\cal B} \in L^1({\mathbb R}^d)$).
By Young's inequality, we get that
$$\|\mathcal{S}u\|_{2,\beta}\leq C_{\beta}\|{\cal B} _{\beta}\|_{L^1}\|u\|_{2,\beta}.$$
Analogously we obtain that $\|\nabla (\mathcal{S}u)\|_{2,\beta}\leq C_{\beta}\|\langle \cdot \rangle ^{\beta} \nabla {\cal B}\|_{L^1}\|u\|_{2,\beta}$.
<!-- end proof-2 -->
<!-- remark-1: body_md -->
For any $u\in L^{\infty}({\mathbb R}^d)$ we have that $\mathcal{S}u \in C^1_b({\mathbb R}^d)$.
<!-- end remark-1 -->
<!-- proof-3: body_md -->
For any $x_0,x\in {\mathbb R}^d$:
$$|(\mathcal{S}u)(x)-(\mathcal{S}u)(x_0)|=\left| \int_{{\mathbb R}^d}\left[ {\cal B} (x-y)-{\cal B} (x_0-y)\right]u(y) \, dy\right| $$
$$\leq \| u \|_{L^{\infty}}\int_{{\mathbb R}^d}|{\cal B} (x-y)-{\cal B} (x_0-y)|dy \rightarrow 0 \mbox{ as } x\rightarrow x_0$$ 
(we have used a change of variable $y-x_0=z$ to obtain the convergence).
$$\Rightarrow \mathcal{S}u\in C({\mathbb R}^d).$$

\noindent
On the other hand, we have:
$$|(\mathcal{S}u)(x)|=\left| \int_{{\mathbb R}^d}{\cal B} (x-y)u(y) dy\right| \leq \|u\|_{L^{\infty}}\int_{{\mathbb R}^d}{\cal B} (x-y) \, dy=\|u \|_{L^{\infty}}$$
$$\Rightarrow \mathcal{S}u\in C_b({\mathbb R}^d).$$

\noindent
We have $\nabla (\mathcal{S}u)(x)=\int_{{\mathbb R}^d}\nabla {\cal B} (x-y)u(y) \, dy$, where 

$\nabla {\cal B} (x)=C\displaystyle \int^{\infty}_0e^{-\theta}\theta ^{- \frac{d}{2}}e^{-\frac{|x|^2}{4\theta}}\frac{-x}{2\theta} d\theta$ with $C=\displaystyle \frac{1}{(4\pi)^{\frac{d}{2}}}$.

\noindent
So, 
$$\int_{{\mathbb R}^d}|\nabla{\cal B} (x)| dx\leq C\int^{\infty}_0 e^{-\theta}\theta ^{- \frac{d}{2}}\int_{{\mathbb R}^d} e^{-\frac{|x|^2}{4\theta}}\frac{|x|}{2\theta} \, dx \, d\theta .$$ 
We do a change of variable $\frac{x}{2\sqrt{\theta}} =y \Rightarrow dx = (2\sqrt{\theta})^d dy$ and get:
$$\int_{{\mathbb R}^d}|\nabla{\cal B} (x)| \, dx\leq C\int^{\infty}_0 e^{-\theta}\theta ^{- \frac{d}{2}}\int_{{\mathbb R}^d} e^{-|y|^2}|y| \frac{1}{\theta^{\frac{1}{2}}}(2\sqrt{\theta})^d \, dy \, d\theta$$
$$=\tilde{C}\int^{\infty}_0 e^{-\theta}\theta^{-\frac{1}{2}} \,  d\theta \, \int_{{\mathbb R}^d} e^{-|y|^2}|y| \, dy < +\infty$$
$$\Rightarrow \nabla {\cal B} \in L^1({\mathbb R}^d;{\mathbb R}^d).$$

It follows analogously that $\nabla (\mathcal{S}u) \in C_b({\mathbb R}^d) \Rightarrow \mathcal{S}u \in C^1_b({\mathbb R}^d).$
<!-- end proof-3 -->
<!-- lemma-3: body_md -->
\label{lemma2.3}
Let $\beta >0$. The following compact embedding holds
$$H^1({\mathbb R}^d)\cap L^2_{\beta }({\mathbb R}^d) \hookrightarrow L^2({\mathbb R}^d) \, . $$
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
Let $(f_n)$ be bounded in $H^1({\mathbb R}^d)\cap L^2_{\beta}({\mathbb R}^d)$. Let $R>0$ be arbitrary and fixed.
By Rellich-Kondrachov's theorem, we get that $f_n\rightarrow f$ in $L^2(B_R)$ on a subsequence (also denoted by $(f_n)$).

On the other hand, we have:

$$\int_{B_R^c}|f_n|^2 dx \leq R^{-2\beta}\int_{B_R^c}\langle x \rangle ^{2\beta}|f_n|^2 dx \leq R^{-2\beta}\sup_n \|f_n\|^2_{2,\beta}.$$

Thus we get $(f_{1_n})$ a subsequence convergent to $f$ on $L^2(B_1)$, $(f_{2_n})$ a subsequence of the previous sequence which converges to $f$ on $L^2(B_2)$, etc.
We choose $(f_{n_n})$ convergent to $f$ in $L^2(B_R), \forall R>0$.

We have that $\forall \varepsilon >0$, there exists a $R_\varepsilon \in {\mathbb N}^*$ such that $R_\varepsilon ^{-2\beta} \sup_k \|f_k\|^2_{2,\beta} < \frac{\varepsilon^2}{4}$.
So, $\|f_k\|^2_{L^2(B_{R_\varepsilon}^c)}< \frac{\varepsilon^2}{4}$.
Moreover, there exists $n_\varepsilon \in {\mathbb N}$ such that $\forall n \geq n_\varepsilon$ we have that $\|f_{n_n}-f\|^2_{L^2(B_{R_\varepsilon})}<\frac{\varepsilon^2}{4}$. Thus we obtain that $(f_{n_n})$ is a Cauchy sequence in $L^2({\mathbb R}^d)$ and therefore convergent to $f$ in $L^2({\mathbb R}^d)$.

<!-- lemma-4: body_md -->
\label{lemma2.4}
Let $d\in \{ 1,2,3\}$ and let $f_{\varepsilon }\in L^{\infty }({\mathbb R}^d)$ satisfy
$$supp \ f_{\varepsilon }\subset B_{\varepsilon }(x_0), \quad \| f_{\varepsilon }\|_{L^{\infty }({\mathbb R}^d)}\leq M .$$
Let $\eta _{\varepsilon }=\mathcal{S}f_{\varepsilon }$. Then there exists a constant $C$ independent of $x_0\in {\mathbb R}^d$, $M>0$ and $\varepsilon >0$, such 
that for all sufficiently small $\varepsilon $
$$\| \eta _{\varepsilon }\|^2_{H^1({\mathbb R}^d)}\leq C \, M^2m_{\varepsilon }^{\frac{3}{2}}.$$
Moreover, if $d=3$, the following sharper estimate holds
$$\| \eta _{\varepsilon }\|^2_{H^1({\mathbb R}^d)}\leq C \, M^2m_{\varepsilon }^{\frac{5}{3}}.$$
If $d=2$, it holds
$$\| \eta _{\varepsilon }\|^2_{H^1({\mathbb R}^d)}\leq C \, M^2m_{\varepsilon }^2(1+|\log \varepsilon |),$$
and if $d=1$:
$$\| \eta _{\varepsilon }\|^2_{H^1({\mathbb R}^d)}\leq C \, M^2m_{\varepsilon }^2.$$
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
Since $\eta _{\varepsilon }=\mathcal{S}f_{\varepsilon }$ the weak formulation of $(I-\Delta )\eta _{\varepsilon }=f_{\varepsilon }$ gives
$$\| \eta_{\varepsilon }\|^2_{H^1({\mathbb R}^d)}=\int_{{\mathbb R}^d}f_{\varepsilon }\eta _{\varepsilon } \, dx=\int_{B_{\varepsilon }(x_0)}
\int_{B_{\varepsilon }(x_0)}f_{\varepsilon }(x)\mathcal{B}(x-y)f_{\varepsilon }(y) \, dy \, dx ,$$
thus
$$\| \eta_{\varepsilon }\|^2_{H^1({\mathbb R}^d)}\leq M^2\int_{B_{\varepsilon }(x_0)}
\int_{B_{\varepsilon }(x_0)}\mathcal{B}(x-y) \, dy \, dx .$$
The Bessel kernel is translation invariant and satisfies the following local bounds near the origin
$$\mathcal{B}(z)\leq C \quad \mbox{\rm if } d=1, $$
$$\mathcal{B}(z)\leq C(1+|\log |z||) \quad \mbox{\rm if } d=2, $$
$$\mathcal{B}(z)\leq C(1+|z|^{-1}) \quad \mbox{\rm if } d=3 $$
(where $C$ is a positive constant) 
and the exponential decay at infinity is irrelevant, since both variables are integrated over the same small ball.

For $d=1$, the kernel is bounded near the origin, therefore
$$\int_{B_{\varepsilon }(x_0)}\int_{B_{\varepsilon }(x_0)}\mathcal{B}(x-y) \, dy \, dx \leq C \, m^2_{\varepsilon }.$$

For $d=2$, after translating $x_0$ to the origin, we have
$$\int_{B_{\varepsilon }}\int_{B_{\varepsilon }}\mathcal{B}(x-y) \, dy \, dx \leq C \int_{B_{\varepsilon }}\int_{B_{\varepsilon }}(1+|\log |x-y||) \, dy \, dx,$$
where, for each $x\in B_{\varepsilon }$, the inner integral is bounded by
$$C \int_{B_{2\varepsilon }}(1+|\log |z||) \, dz \leq C \varepsilon ^2(1+|\log \varepsilon |).$$
Then we get that 
$$\int_{B_{\varepsilon }}\int_{B_{\varepsilon }}\mathcal{B}(x-y) \, dy \, dx \leq C\varepsilon ^4(1+|\log \varepsilon |)=Cm^2_{\varepsilon }(1+|\log \varepsilon |) .$$

For $d=3$, the same argument gives 
$$\int_{B_{\varepsilon }}\int_{B_{\varepsilon }}\mathcal{B}(x-y) \, dy \, dx\leq C\int_{B_{\varepsilon }}\int_{B_{\varepsilon }}(1+|x-y|^{-1}) \, dy \, dx.$$
For each fixed $x\in B_{\varepsilon }$,
$$\int_{B_{\varepsilon }}|x-y|^{-1} \, dy \leq \int_{B_{2\varepsilon }}|z|^{-1}dz\leq C \, \varepsilon ^2,$$
hence
$$\int_{B_{\varepsilon }}\int_{B_{\varepsilon }}\mathcal{B}(x-y) \, dy \, dx\leq C \, \varepsilon ^3\varepsilon ^2=C \, m^{5/3}_{\varepsilon }.$$
Therefore, for sufficiently small $\varepsilon $, the bounds in dimensions one, two, and three are all bounded by $C \, m^{3/2}_{\varepsilon }$, proving the lemma.

<!-- proposition-1: body_md -->
\label{proposition3.4}
Let $u_n,u\in{\cal U}_\beta$ and assume that $u_n\rightharpoonup u$ weakly in $L^2_\beta({\mathbb R}^d)$. Then
$$\rho^{u_n}\longrightarrow \rho^u \quad \mbox{in } C([0,T];L^2({\mathbb R}^d)),$$
and
$$\rho^{u_n}\longrightarrow \rho^u \quad \mbox{in } L^2(0,T;H^1({\mathbb R}^d)).$$

<!-- end proposition-1 -->
<!-- proof-8: body_md -->
Let
\[
\eta_n:=\mathcal S u_n,\qquad \eta:=\mathcal S u.
\]
By Theorem \ref{theorem2.5}, $\eta_n\to \eta$ in $H^1({\mathbb R}^d)$ and in $C^1_{\rm loc}({\mathbb R}^d)$. Since $0\le u_n,u\le M_0$ and $\nabla\mathcal B\in L^1({\mathbb R}^d;{\mathbb R}^d)$, the sequence $\nabla \eta_n$ is uniformly bounded in $L^\infty({\mathbb R}^d;{\mathbb R}^d)$. Hence
\[
 b(\eta_n),\quad q(\eta_n),\quad \nabla_x q(\eta_n),\quad \frac{\partial q}{\partial s}(\eta_n),\quad \frac{\partial q}{\partial s}(\eta_n)\nabla\eta_n
\]
converge pointwise a.e. to the corresponding quantities evaluated at $\eta$, and they are uniformly bounded.

Set $z_n:=\rho^{u_n}-\rho^u$. We write
\[
B_n=b(\eta_n),\qquad B=b(\eta),\qquad Q_n=q(\eta_n),\qquad Q=q(\eta),
\]
and
\[
D_{i,n}:=\frac{\partial Q_{ij,n}}{\partial x_j}
=\frac{\partial q_{ij}}{\partial x_j}(\eta_n)+\frac{\partial q_{ij}}{\partial s}(\eta_n)\frac{\partial \eta_n}{\partial x_j},\qquad
D_i:=\frac{\partial Q_{ij}}{\partial x_j}.
\]
The weak formulation of the difference equation gives, for every $\varphi\in H^1({\mathbb R}^d)$,
\begin{align*}
\left\langle \frac{d z_n}{dt},\varphi\right\rangle_{H^{-1},H^1}
&=\int_{{\mathbb R}^d}B_nz_n\cdot\nabla\varphi \,dx
-\frac12\int_{{\mathbb R}^d}Q_{ij,n}\frac{\partial z_n}{\partial x_j}\frac{\partial \varphi}{\partial x_i}\,dx
-\frac12\int_{{\mathbb R}^d}D_{i,n}z_n\frac{\partial \varphi}{\partial x_i}\,dx
\\
&\quad+\int_{{\mathbb R}^d}(B_n-B)\rho^u\cdot\nabla\varphi\,dx
-\frac12\int_{{\mathbb R}^d}(Q_{ij,n}-Q_{ij})\frac{\partial \rho^u}{\partial x_j}\frac{\partial \varphi}{\partial x_i}\,dx
\\
&\quad-\frac12\int_{{\mathbb R}^d}(D_{i,n}-D_i)\rho^u\frac{\partial \varphi}{\partial x_i}\,dx .
\end{align*}
Taking $\varphi=z_n(t)$, using the uniform ellipticity of $q$, the boundedness of the lower-order coefficients, and Young's inequality, we obtain
\begin{align*}
\frac12\frac{d}{dt}\|z_n(t)\|_{L^2}^2+c_0\|\nabla z_n(t)\|_{L^2}^2
&\le C\Big( \|z_n(t)\|_{L^2}^2
+\|(B_n-B)\rho^u(t)\|_{L^2}^2\\
&\quad+\|(Q_n-Q)\nabla\rho^u(t)\|_{L^2}^2
+\|(D_n-D)\rho^u(t)\|_{L^2}^2\Big),
\end{align*}
where $c_0>0$ and $C>0$ are independent of $n$. Since $z_n(0)=0$, Gronwall's lemma yields
\begin{align*}
\sup_{t\in[0,T]}\|z_n(t)\|_{L^2}^2+\int_0^T\|\nabla z_n(t)\|_{L^2}^2\,dt
&\le C\Big[ \int_0^T\|(B_n-B)\rho^u(t)\|_{L^2}^2\,dt\\
&\quad+\int_0^T\|(Q_n-Q)\nabla\rho^u(t)\|_{L^2}^2\,dt\\
&\quad+\int_0^T\|(D_n-D)\rho^u(t)\|_{L^2}^2\,dt\Big].
\end{align*}
The three terms on the right-hand side tend to zero by dominated convergence. Indeed, the coefficient differences converge pointwise a.e. and are uniformly bounded, whereas $\rho^u\in L^2((0,T)\times{\mathbb R}^d)$ and $\nabla\rho^u\in L^2((0,T)\times{\mathbb R}^d;{\mathbb R}^d)$, hence
\[
\sup_{t\in[0,T]}\|z_n(t)\|_{L^2}\to0,\qquad
\int_0^T\|\nabla z_n(t)\|_{L^2}^2\,dt\to0\,,
\]
which proves the asserted convergence in $C([0,T];L^2({\mathbb R}^d))$ and in $L^2(0,T;H^1({\mathbb R}^d))$.

<!-- theorem-3: body_md -->
\label{existence_optimal_control}
Under hypotheses (H1)-(H4) and (H4)', problem $(P)$ admits at least one optimal control $u^*\in{\cal U}_\beta$.

Because $h(0)=0$ and $G,G_T\geq0$, the zero control is admissible and has finite cost. Hence
$$m:=\inf_{u\in{\cal U}_\beta}J(u)$$
is finite and non-negative. Let $(u_n)$ be a minimizing sequence such that $J(u_n)\to m$. From (H4)' we obtain
$$\frac{\alpha}{2}\int_{{\mathbb R}^d}\langle x\rangle^{2\beta}|u_n(x)|^2\,dx\leq J(u_n)\leq m+1$$
for all sufficiently large $n$. Thus $(u_n)$ is bounded in $L^2_\beta({\mathbb R}^d)$. It is also bounded in $L^\infty({\mathbb R}^d)$ and satisfies $0\leq u_n\leq M_0$ a.e. By Banach-Alaoglu's theorem and weak compactness in the Hilbert space $L^2_\beta({\mathbb R}^d)$, there exists $u^*\in L^2_\beta({\mathbb R}^d)\cap L^\infty({\mathbb R}^d)$ such that, up to a subsequence,
$$u_n\rightharpoonup u^*\quad \mbox{weakly in }L^2_\beta({\mathbb R}^d),\qquad u_n\stackrel{*}{\rightharpoonup}u^*\quad \mbox{weakly * in }L^\infty({\mathbb R}^d).$$

$$\rho^{u_n}\to\rho^{u^*}\quad \mbox{in }L^2((0,T)\times{\mathbb R}^d),\qquad \rho^{u_n}(T)\to\rho^{u^*}(T)\quad \mbox{in }L^2({\mathbb R}^d).$$

$$u\mapsto \int_{{\mathbb R}^d}\langle x\rangle^{2\beta}h(u(x))\,dx$$
is weakly lower semicontinuous on $L^2_\beta({\mathbb R}^d)$, because $h$ is convex and continuous on $[0,M_0]$. Hence
$$J(u^*)\leq \liminf_{n\to\infty}J(u_n)=m.$$
Since $u^*\in{\cal U}_\beta$, we also have $m\leq J(u^*)$, and therefore $J(u^*)=m$.

<!-- theorem-4: body_md -->
\label{theorem4.1} (Pontryagin's principle)
Let $u^*$ be an optimal control and let $p$ be the weak solution to \eqref{eqp}. Define
$$F_1(x):=\int_0^T \frac{\partial b}{\partial s}(x,(\mathcal{S}u^*)(x))\rho(t,x)\cdot\nabla p(t,x)\,dt,$$
and
$$F_2(x):=\int_0^T \frac{\partial q}{\partial s}(x,(\mathcal{S}u^*)(x))\rho(t,x)\Delta p(t,x)\,dt.$$
Then $F_1,F_2\in L^1({\mathbb R}^d)$. Set
$$\Phi_1:=\mathcal{S}F_1={\cal B}*F_1,
\qquad
\Phi_2:=\mathcal{S}F_2={\cal B}*F_2.$$
For a.e. $x\in{\mathbb R}^d$ one has
$$u^*(x)\in\operatorname*{arg\,min}_{w\in[0,M_0]}\left\{w\Phi_1(x)+\frac12 w\Phi_2(x)+\langle x\rangle^{2\beta}h(w)\right\}.$$
If the function $h$ is strictly convex, the minimizer is unique for a.e. $x$.
<!-- end theorem-4 -->
<!-- proof-10: body_md -->
The proof is carried out at an arbitrary point $x_0$ which is simultaneously a Lebesgue point of $u^*$, $h\circ u^*$, $\Phi_1$, and $\Phi_2$, which is enough, since these functions are locally integrable and the set of such points has full Lebesgue measure.

Fix $w\in[0,M_0]$ and define the spike perturbation
$$u_\varepsilon(x)=w\mathds{1}_{B_\varepsilon(x_0)}(x)+u^*(x)\mathds{1}_{B_\varepsilon(x_0)^c}(x),
\qquad \delta_\varepsilon:=u_\varepsilon-u^*=(w-u^*)\mathds{1}_{B_\varepsilon(x_0)}.$$
Let $\rho_\varepsilon:=\rho^{u_\varepsilon}$ and $z_\varepsilon:=\rho_\varepsilon-\rho$. Since $u^*$ is optimal
\begin{equation}\label{ineqzeps_blue}
0\leq J(u_\varepsilon)-J(u^*)
=\int_0^T\int_{{\mathbb R}^d}Gz_\varepsilon\,dx \, dt+
\int_{{\mathbb R}^d}G_Tz_\varepsilon(T)\,dx+
\int_{B_\varepsilon(x_0)}\langle x\rangle^{2\beta}(h(w)-h(u^*(x)))\,dx.
\end{equation}
The last term satisfies
\begin{equation}\label{control_term_blue}
\frac1{m_\varepsilon}\int_{B_\varepsilon(x_0)}\langle x\rangle^{2\beta}(h(w)-h(u^*(x)))\,dx
\longrightarrow
\langle x_0\rangle^{2\beta}(h(w)-h(u^*(x_0))).
\end{equation}

The duality identity obtained by multiplying the equation for $z_\varepsilon$ by the adjoint $p$ and integrating by parts is (after an easy calculation)
\begin{align}\label{duality_blue}
&\int_0^T\int_{{\mathbb R}^d}Gz_\varepsilon\,dx \, dt+
\int_{{\mathbb R}^d}G_Tz_\varepsilon(T)\,dx \\
&=\int_0^T\int_{{\mathbb R}^d}(b(\mathcal{S}u_\varepsilon)-b(\mathcal{S}u^*))\rho\cdot\nabla p\,dx \,dt
+\frac12\int_0^T\int_{{\mathbb R}^d}(q(\mathcal{S}u_\varepsilon)-q(\mathcal{S}u^*))\rho\Delta p\,dx \, dt \nonumber\\
&\quad +\int_0^T\int_{{\mathbb R}^d}(b(\mathcal{S}u_\varepsilon)-b(\mathcal{S}u^*))z_\varepsilon\cdot\nabla p\,dx \, dt
+\frac12\int_0^T\int_{{\mathbb R}^d}(q(\mathcal{S}u_\varepsilon)-q(\mathcal{S}u^*))z_\varepsilon\Delta p\,dx \, dt. \nonumber
\end{align}
The integrations by parts are justified by the regularity quoted before the theorem.

We first estimate the state increment. In particular, the difference $z_\varepsilon$ solves the equation obtained by subtracting the two Fokker-Planck equations, and the forcing terms are produced by the coefficient differences
\[
 b(\mathcal S u_\varepsilon)-b(\mathcal S u^*),\qquad
 q(\mathcal S u_\varepsilon)-q(\mathcal S u^*),\qquad
 \nabla \left(q(\mathcal S u_\varepsilon)\right)-\nabla (q(\mathcal S u^*))\,,
\]
hence, by the mean value theorem and the boundedness assumptions on the coefficients, each of these differences is bounded by a constant times either $|\mathcal S\delta_\varepsilon|$ or $|\nabla (\mathcal S\delta_\varepsilon)|$, and Lemma \ref{lemma2.4} gives
\[
\|\mathcal S\delta_\varepsilon\|_{H^1({\mathbb R}^d)}\le C m_\varepsilon^{3/4},
\]
for all sufficiently small $\varepsilon$. Since $d\le3$, the Sobolev embedding $H^1({\mathbb R}^d)\subset L^4({\mathbb R}^d)$ also gives
\[
\|\mathcal S\delta_\varepsilon\|_{L^4({\mathbb R}^d)}\le C m_\varepsilon^{3/4}.
\]
Testing the equation for $z_\varepsilon$ by $z_\varepsilon$, using the uniform ellipticity of $q$, and applying Young's inequality, we obtain
\begin{align*}
\|z_\varepsilon\|_{L^\infty(0,T;L^2)}^2+\|z_\varepsilon\|_{L^2(0,T;H^1)}^2
&\le C\|\mathcal S\delta_\varepsilon\|_{L^4}^2
\left(\|\rho\|_{L^2(0,T;L^4)}^2+\|\nabla\rho\|_{L^2(0,T;L^4)}^2\right)
\\
&\quad+C\|\nabla(\mathcal S\delta_\varepsilon )\|_{L^2}^2\|\rho\|_{L^2(0,T;L^\infty)}^2 .
\end{align*}
The right-hand side is bounded by $C m_\varepsilon^{3/2}$, because $\rho\in L^2(0,T;H^2({\mathbb R}^d))$ and $H^2({\mathbb R}^d)\subset W^{1,4}({\mathbb R}^d)\cap L^\infty({\mathbb R}^d)$ for $d\le3$, hence
\begin{equation*}
\|z_\varepsilon\|_{L^\infty(0,T;L^2({\mathbb R}^d))}+\|z_\varepsilon\|_{L^2(0,T;H^1({\mathbb R}^d))}\le C m_\varepsilon^{3/4}\,.
\end{equation*}
Consequently,
\begin{equation}\label{z_terms_vanish_blue}
\frac1{m_\varepsilon}\left|\int_0^T\int_{{\mathbb R}^d}(b(\mathcal{S}u_\varepsilon)-b(\mathcal{S}u^*))z_\varepsilon\cdot\nabla p\,dx\ dt\right|\to0\,,
\end{equation}
and
\begin{equation}\label{z_terms_vanish_q_blue}
\frac1{m_\varepsilon}\left|\int_0^T\int_{{\mathbb R}^d}(q(\mathcal{S}u_\varepsilon)-q(\mathcal{S}u^*))z_\varepsilon\Delta p\,dx\ dt\right|\to0.
\end{equation}
Indeed, for the drift term,
\begin{align*}
\frac1{m_\varepsilon}\left|\int_0^T\int_{{\mathbb R}^d}(b(\mathcal S u_\varepsilon)-b(\mathcal S u^*))z_\varepsilon\cdot\nabla p\,dx\ dt\right|
&\le \frac{C}{m_\varepsilon}\|\mathcal S\delta_\varepsilon\|_{L^4}\|z_\varepsilon\|_{L^2(0,T;L^4)}\|\nabla p\|_{L^2(0,T;L^2)}\to0,
\end{align*}
and the diffusion term is analogous, with $\Delta p$ in place of $\nabla p$.

We now identify the two first-order terms. Since $\rho\in L^2((0,T)\times{\mathbb R}^d)$, $\nabla p\in L^2((0,T)\times{\mathbb R}^d)$, and $\Delta p\in L^2((0,T)\times{\mathbb R}^d)$, while $\frac{\partial b}{\partial s}$ and $\frac{\partial q}{\partial s}$ are bounded, we have $F_1,F_2\in L^1({\mathbb R}^d)$. The Bessel kernel ${\cal B}$ belongs to $L^1({\mathbb R}^d)$; hence $\Phi_1={\cal B}*F_1$ and $\Phi_2={\cal B}*F_2$ belong to $L^1({\mathbb R}^d)$ and are defined a.e.

By the mean value theorem,
$$b(\mathcal{S}u_\varepsilon)-b(\mathcal{S}u^*)=\frac{\partial b}{\partial s}(\mathcal{S}u^*)\mathcal{S}\delta_\varepsilon+R^b_\varepsilon,$$
where
$$|R^b_\varepsilon|\leq C|\mathcal{S}\delta_\varepsilon|^2.$$
Therefore,
\begin{equation*}
\int_0^T\int_{{\mathbb R}^d}(b(\mathcal{S}u_\varepsilon)-b(\mathcal{S}u^*))\rho\cdot\nabla p\,dx \, dt
=\int_{{\mathbb R}^d}F_1(x)(\mathcal{S}\delta_\varepsilon)(x)\,dx+o(m_\varepsilon)\,,
\end{equation*}
\noindent
and the remainder is $o(m_\varepsilon)$: since $d\le3$, $H^2({\mathbb R}^d)\subset W^{1,4}({\mathbb R}^d)$ and
\[
\frac1{m_\varepsilon}\int_0^T\int_{{\mathbb R}^d}|\mathcal{S}\delta_\varepsilon|^2|\rho||\nabla p|\,dx\,dt
\le
\frac{\|\mathcal{S}\delta_\varepsilon\|_{L^4}^2}{m_\varepsilon}
\|\rho\|_{L^2(0,T;L^4)}\|\nabla p\|_{L^2(0,T;L^4)}
\le C m_\varepsilon^{1/2}\to0\,,
\]
\noindent
then, using the self-adjointness of $\mathcal{S}$, or equivalently Fubini's theorem and the symmetry of ${\cal B}$,
$$\int_{{\mathbb R}^d}F_1(x)(\mathcal{S}\delta_\varepsilon)(x)\,dx
=\int_{B_\varepsilon(x_0)}(w-u^*(y))\Phi_1(y)\,dy.$$
Since $x_0$ is a Lebesgue point of $u^*$ and $\Phi_1$,
\begin{equation}\label{drift_limit_blue}
\frac1{m_\varepsilon}\int_0^T\int_{{\mathbb R}^d}(b(\mathcal{S}u_\varepsilon)-b(\mathcal{S}u^*))\rho\cdot\nabla p\,dx \, dt
\to (w-u^*(x_0))\Phi_1(x_0).
\end{equation}

The diffusion term is identical. We write
$$q(\mathcal{S}u_\varepsilon)-q(\mathcal{S}u^*)=\frac{\partial q}{\partial s}(\mathcal{S}u^*)\mathcal{S}\delta_\varepsilon+R^q_\varepsilon,
\qquad |R^q_\varepsilon|\leq C|\mathcal{S}\delta_\varepsilon|^2.$$
Then
\begin{equation}\label{diffusion_limit_blue}
\frac1{m_\varepsilon}\int_0^T\int_{{\mathbb R}^d}(q(\mathcal{S}u_\varepsilon)-q(\mathcal{S}u^*))\rho\Delta p\,dx \, dt
\to (w-u^*(x_0))\Phi_2(x_0).
\end{equation}
The proof is the same as above, with $F_2$ and $\Phi_2$ in place of $F_1$ and $\Phi_1$.

Dividing \eqref{ineqzeps_blue} by $m_\varepsilon$ and using \eqref{control_term_blue}, \eqref{duality_blue}, \eqref{z_terms_vanish_blue}, \eqref{z_terms_vanish_q_blue}, \eqref{drift_limit_blue}, and \eqref{diffusion_limit_blue}, we obtain
$$0\leq \langle x_0\rangle^{2\beta}(h(w)-h(u^*(x_0)))+(w-u^*(x_0))\Phi_1(x_0)+\frac12(w-u^*(x_0))\Phi_2(x_0).$$
Equivalently,
$$\langle x_0\rangle^{2\beta}h(u^*(x_0))+u^*(x_0)\Phi_1(x_0)+\frac12u^*(x_0)\Phi_2(x_0)
\leq
\langle x_0\rangle^{2\beta}h(w)+w\Phi_1(x_0)+\frac12w\Phi_2(x_0)$$
for every $w\in[0,M_0]$. Since $x_0$ was an arbitrary common Lebesgue point, the conclusion follows for a.e. $x\in{\mathbb R}^d$.

<!-- proposition-3: body_md -->
\label{appendix_parabolic_regularity}
Let $T>0$ and let $\tilde a = (\tilde a_{ij})_{i,j\in \{1,2,...,d\}}, \tilde b =(\tilde b_i )_{i\in \{1,2,...,d\}}, \tilde c$ be time-independent coefficients on ${\mathbb R}^d$. Assume that
\[
\tilde a=\tilde a^T,
\qquad
\tilde a\in W^{1,\infty}({\mathbb R}^d;{\mathbb R}^{d\times d}),
\qquad
\tilde b\in L^\infty({\mathbb R}^d;{\mathbb R}^d),
\qquad
\tilde c\in L^\infty({\mathbb R}^d)\,,
\]
and that there exists a constant \(0<\lambda\) such that
\[
\lambda |\xi|^2\le \tilde a_{ij}(x)\xi_i\xi_j\ ,
\qquad\forall x\in{\mathbb R}^d\,,
\quad\forall \xi\in{\mathbb R}^d\,.
\]
If \(f\in L^2((0,T)\times{\mathbb R}^d)\) and \(v_0\in H^1({\mathbb R}^d)\), then the Cauchy problem
$$\frac{\partial v}{\partial t}-\tilde a_{ij}\frac{\partial^2 v}{\partial x_i \partial x_j}-\tilde b_i\frac{\partial v}{\partial x_i}-\tilde cv=f,
\qquad v(0)=v_0,$$
has a unique solution satisfying
\[
v\in L^2(0,T;H^2({\mathbb R}^d))\cap W^{1,2}(0,T;L^2({\mathbb R}^d)).
\]
Moreover,
\[
\|v\|_{L^2(0,T;H^2({\mathbb R}^d))}+
\|\frac{\partial v}{\partial t}\|_{L^2((0,T)\times{\mathbb R}^d)}
\le
C\left(
\|f\|_{L^2((0,T)\times{\mathbb R}^d)}+
\|v_0\|_{H^1({\mathbb R}^d)}
\right),
\]
where \(C\) depends only on
\[
T,d,\lambda,
\|\tilde a\|_{W^{1,\infty}},
\|\tilde b\|_{L^\infty},
\|\tilde c\|_{L^\infty}.
\]
The analogous backward terminal-value problem follows from the change of variables \(s=T-t\).
<!-- end proposition-3 -->
<!-- proof-12: body_md -->
This is the standard global Sobolev regularity theorem for uniformly parabolic equations on the whole space with Lipschitz leading coefficients and bounded lower-order coefficients. More precisely, the statement follows from the classical $L^2$-theory for non-divergence form parabolic operators with time-independent coefficients, applied to
\[
\widetilde L v:=\widetilde a_{ij}(x)\frac{\partial^2 v}{\partial x_i\partial x_j}+\widetilde b_i(x)\frac{\partial v}{\partial x_i}+\widetilde c(x)v.
\]
The assumptions $\widetilde a=\widetilde a^T$, $\widetilde a\in W^{1,\infty}({\mathbb R}^d)$, and uniform ellipticity imply the elliptic estimate
\[
\|v\|_{H^2({\mathbb R}^d)}\le C\left(\|\widetilde L v\|_{L^2({\mathbb R}^d)}+\|v\|_{L^2({\mathbb R}^d)}\right),\qquad v\in H^2({\mathbb R}^d),
\]
where the lower-order terms are controlled by $\|\widetilde b\|_{L^\infty}$ and $\|\widetilde c\|_{L^\infty}$. Combining this elliptic estimate with the standard energy estimate for $\displaystyle \frac{\partial v}{\partial t}-\widetilde L v=f$ gives the asserted maximal $L^2$-regularity estimate. The result is stated, in equivalent forms, in Ladyzhenskaya-Solonnikov-Ural'tseva \cite[Chapter IV]{Ladyzhenskaya1968}, Krylov \cite[Chapters 2 and 3]{Krylov2008}, and Lieberman \cite[Chapters VII and VIII]{Lieberman1996}. The terminal-value problem is reduced to the initial-value problem by the change of variables $s=T-t$.
