For each $i$, we assume $K_i$ is an ellipsoid congruent to
$$\widetilde{K}_i = \{ (x,y,z) \in {\mathbb R}^3 : x^2 + y^2 + (\beta z)^2 \leq r^2_i \} ,$$
where $r_i > 0$ is a small constant and $\beta > 1$ is a fixed parameter controlling the
eccentricity in the $z$-direction. Let $\Gamma = \{z=0\}$ in ${\mathbb R}^3$. We assume each $K_i$
is a translate of $\widetilde{K}_i$, namely
$$K_i = \widetilde{K}_i + O_i := \{ \tilde{x} + O_i : \tilde{x} \in \widetilde{K}_i \},$$
where $O_i \in \Gamma$ is the centre of $K_i$. We further assume there exist constants
$d_0 > 0$ and $0 < \tilde{r}_0 < r_0 < d_0/3$ such that
$$d_0 < |O_i O_j| \leq 1 \quad \text{for all } i \neq j, \qquad \tilde{r}_0 \leq r_i \leq r_0.$$
The condition (H) implies that there exists a constant $c_1 > 0$ such that for any three
consecutive reflection points $q_1, q_2, q_3$ of a billiard trajectory in $\Omega$ and
outward unit normal $\nu(q_2)$ at $q_2$, the angle $\varphi$ between $\nu(q_2)$ and
$(q_3-q_2)/\|q_3-q_2\|$ satisfies $\cos\varphi \geq c_1$. Since $r_i \leq r_0 < d_0/3$,
consecutive reflection points satisfy
\begin{equation}
d_i = \|q_i - q_{i+1}\| \geq d_0 - 2r_0 > d_0/3. \label{eq:di}
\end{equation}

Clearly if $O_1, \ldots, O_{k_0} \in \Gamma$ with no three collinear, then
sufficiently small ellipsoids $K_i$ centred at $O_i$ automatically satisfy (H) and all
the assumptions above.

<!-- thm-1: body_md -->
Under the above assumptions, the billiard ball map $B$ on $M_0$
has exactly two positive Lyapunov exponents $\lambda_1(\beta) \geq \lambda_2(\beta) > 0$
with respect to any Gibbs measure $\mu$. If in addition $\beta > 1/c_1^2$, then the
following hold:
\begin{itemize}
\@par\hangindent 2em[{\rm (i)}] The exponents are distinct: $\lambda_1(\beta) > \lambda_2(\beta) > 0$.
\@par\hangindent 2em[{\rm (ii)}] The smaller exponent $\lambda_2(\beta) = \lambda_2$ is independent of
$\beta$ and equals the Lyapunov exponent of the restriction of the billiard flow to $\Gamma$.
\@par\hangindent 2em[{\rm (iii)}] The gap satisfies $\lambda_1(\beta) \geq \lambda_2 + \log(\beta c_1^2)$.
\end{itemize}

is a linear symplectic map with matrix representation
\begin{equation}
P^{(m)}_\gamma =
 \left(
\begin{array}{cc}
s_m & 0\\
0  & s_m
\end{array}
\right) \cdot
\left(
\begin{array}{cc}
I & d_m I\\
\psi_m  & I + d_m \psi_m
\end{array}
\right)
\cdots
\left(
\begin{array}{cc}
I & d_1 I\\
\psi_1  & I + d_1 \psi_1
\end{array}
\right).
\end{equation}
}

\end{equation}

We have $1 < 1/r_0 \leq \kappa_i \leq 1/\tilde{r}_0$,
$2c_1 \leq \theta_i \leq 2$ and $d_0/3 \leq d_i \leq d_0 < 1$ for all $i$. Setting
$$s = s(\beta) = \min\left\{ \log \left( \frac{\beta d_0 \tilde{r}_0}{6c_1} \right)\;,\,
\log (\beta c_1^2) \right\} > 0,$$
one verifies (as in \@nx\protect\@empty\@nx\csname cite \endcsname{AmSt}) that
\begin{equation}
b_i \geq e^s\, \tilde{b}_i \quad \text{and} \quad a_i \geq e^s\, \tilde{a}_i \quad \text{for all } i \geq 1. \label{eq:key}
\end{equation}

\bigskip

\begin{lem}\label{lem:inductive}
{\it There exists a sequence $\{t_i\}_{i\geq 1}$ of non-negative numbers
such that, setting $u_1 = 1$, $v_1 = e^s$, and
$$u_{i+1} = e^{t_{i+1}}\,u_i \quad,\quad v_i = e^s\,u_i,$$
we have $u_i \geq 1$ for all $i \geq 1$,
\begin{equation}
\Delta^{(2)}_i \geq u_i\,\widetilde{\Delta}^{(2)}_i \quad \text{and} \quad \Delta^{(4)}_i \geq v_i\,\widetilde{\Delta}^{(4)}_i \label{eq:uv}
\end{equation}
for all $i \geq 1$, and $0 \leq t_{i+1} < s$ for all $i \geq 1$.}
\end{lem}

\medskip

\noindent
<!-- end head-3 -->
<!-- lem-4: body_md -->
\label{lem:inductive}
{\it There exists a sequence $\{t_i\}_{i\geq 1}$ of non-negative numbers
such that, setting $u_1 = 1$, $v_1 = e^s$, and
$$u_{i+1} = e^{t_{i+1}}\,u_i \quad,\quad v_i = e^s\,u_i,$$
we have $u_i \geq 1$ for all $i \geq 1$,
\begin{equation}
\Delta^{(2)}_i \geq u_i\,\widetilde{\Delta}^{(2)}_i \quad \text{and} \quad \Delta^{(4)}_i \geq v_i\,\widetilde{\Delta}^{(4)}_i \label{eq:uv}
\end{equation}
for all $i \geq 1$, and $0 \leq t_{i+1} < s$ for all $i \geq 1$.}
<!-- end lem-4 -->
<!-- head-4: body_md -->
We will do the construction by induction. Set $u_1 = 1$ and $v_1 = e^s$.

From \eqref{eq:recA} and \eqref{eq:recB} we have $\Delta^{(2)}_1 = d_1 = \widetilde{\Delta}^{(2)}_1$ and, using \eqref{eq:key},
$\Delta^{(4)}_1 = b_1 \geq e^s \tilde{b}_1 = v_1 \widetilde{\Delta}^{(4)}_1$.
So \eqref{eq:uv} holds for $i = 1$.

Assume that $u_i$, $t_i$, $v_i = e^s u_i$ have been constructed for some $i \geq 1$
so that \eqref{eq:uv} holds. We will now define $u_{i+1}$, $t_{i+1}$ and $v_{i+1} = e^s u_{i+1}$.

Using \eqref{eq:recA} and the inductive hypothesis \eqref{eq:uv} we get
$$\Delta^{(2)}_{i+1} = \Delta^{(2)}_i + d_{i+1}\,\Delta^{(4)}_i
\geq u_i\,\widetilde{\Delta}^{(2)}_i + d_{i+1}\, e^s u_i\,\widetilde{\Delta}^{(4)}_i.$$
From \eqref{eq:recB}, $d_{i+1}\,\widetilde{\Delta}^{(4)}_i = \widetilde{\Delta}^{(2)}_{i+1} - \widetilde{\Delta}^{(2)}_i$, so the above yields
\begin{equation}
\Delta^{(2)}_{i+1} \geq u_i\bigl(e^s\,\widetilde{\Delta}^{(2)}_{i+1} - (e^s - 1)\widetilde{\Delta}^{(2)}_i\bigr). \label{eq:D2est}
\end{equation}
From \eqref{eq:recB} and Lemma~{{\rm\expandafter\@setref\csname r@lem:delta\endcsname\@firstoftwo{lem:delta}}}, $\widetilde{\Delta}^{(2)}_{i+1} \geq (1+d_0)\,\widetilde{\Delta}^{(2)}_i$, hence
$\widetilde{\Delta}^{(2)}_i \leq \widetilde{\Delta}^{(2)}_{i+1}/(1+d_0)$. Substituting into \eqref{eq:D2est} gives
$$\Delta^{(2)}_{i+1} \geq u_i\left(\frac{d_0 e^s + 1}{1+d_0}\right)\widetilde{\Delta}^{(2)}_{i+1}.$$
Define $t_{i+1} > 0$ by
\begin{equation}
t_{i+1} = \log\frac{d_0 e^s + 1}{1+d_0} < s, \label{eq:ti}
\end{equation}
and set $u_{i+1} = e^{t_{i+1}} u_i \geq u_i \geq 1$.
Then $\Delta^{(2)}_{i+1} \geq u_{i+1}\,\widetilde{\Delta}^{(2)}_{i+1}$, so \eqref{eq:uv} holds for $\Delta^{(2)}_{i+1}$.

Now set $v_{i+1} = e^s u_{i+1}$. We check that \eqref{eq:uv} holds for $\Delta^{(4)}_{i+1}$.
It follows from \eqref{eq:recA}, \eqref{eq:key} and the inductive assumptions that
\begin{eqnarray*}
\Delta^{(4)}_{i+1}
&=& a_{i+1}\,\Delta^{(2)}_i + b_{i+1}\,\Delta^{(4)}_i
\geq e^s\tilde{a}_{i+1}\,\Delta^{(2)}_i + e^s\tilde{b}_{i+1}\,\Delta^{(4)}_i\\
&\geq& e^s\bigl(\tilde{a}_{i+1}\,u_i\,\widetilde{\Delta}^{(2)}_i + \tilde{b}_{i+1}\, e^s u_i\,\widetilde{\Delta}^{(4)}_i\bigr)\\
&=& e^s u_{i+1}\bigl(\tilde{a}_{i+1}\, e^{-t_{i+1}}\,\widetilde{\Delta}^{(2)}_i
   + \tilde{b}_{i+1}\, e^{s-t_{i+1}}\,\widetilde{\Delta}^{(4)}_i\bigr).
\end{eqnarray*}
We will check now that $v_{i+1}\,\widetilde{\Delta}^{(4)}_{i+1} = e^s u_{i+1}\,\widetilde{\Delta}^{(4)}_{i+1}$ satisfies
$$e^s u_{i+1}\bigl(\tilde{a}_{i+1}\, e^{-t_{i+1}}\,\widetilde{\Delta}^{(2)}_i
+ \tilde{b}_{i+1}\, e^{s-t_{i+1}}\,\widetilde{\Delta}^{(4)}_i\bigr) \geq v_{i+1}\,\widetilde{\Delta}^{(4)}_{i+1},$$
that is,
\begin{equation}
\frac{\tilde{b}_{i+1}\,\widetilde{\Delta}^{(4)}_i}{\tilde{a}_{i+1}\,\widetilde{\Delta}^{(2)}_i}
\geq \frac{e^{t_{i+1}}-1}{e^s - e^{t_{i+1}}}. \label{eq:ratio}
\end{equation}
The definition \eqref{eq:ti} and a simple calculation give
$$\frac{e^{t_{i+1}} -1}{e^s - e^{t_{i+1}}} = \frac{ \frac{d_0 e^s + 1}{1+d_0} -1}{e^s -  \frac{d_0 e^s + 1}{1+d_0}}
= \frac{d_0 e^s + 1 -1 -d_0}{e^s + d_0 e^s - d_0 e^s -1} = \frac{d_0 (e^s-1)}{e^s-1} = d_0 .$$
On the other hand, for the left-hand side of \eqref{eq:ratio}, it follows from Lemma~{{\rm\expandafter\@setref\csname r@lem:delta\endcsname\@firstoftwo{lem:delta}}} that
$$\frac{\tilde{b}_{i+1}\,\widetilde{\Delta}^{(4)}_i}{\tilde{a}_{i+1}\,\widetilde{\Delta}^{(2)}_i}
\geq \frac{\tilde{b}_{i+1}}{\tilde{a}_{i+1}}
= \frac{1 + \kappa_{i+1} d_{i+1} L_{i+1}\theta_{i+1}}{\kappa_{i+1} L_{i+1}\theta_{i+1}}
> d_{i+1} \geq d_0,$$
since $d_{i+1} \geq d_0$. This proves \eqref{eq:ratio}. The latter in turn implies
$\Delta^{(4)}_{i+1} \geq e^s u_{i+1}\,\widetilde{\Delta}^{(4)}_{i+1} = v_{i+1}\,\widetilde{\Delta}^{(4)}_{i+1}$,
and therefore \eqref{eq:uv} holds with $i$ replaced by $i+1$.

This completes the induction.
{\rule{6pt}{6pt}}

\bigskip

We will now show that both $\lambda(v)$ and $\lambda(\tilde{v})$ are strictly positive.


\begin{lem}\label{lem:positive}
{\it We have $\lambda(v) > 0$ and $\lambda(\tilde{v}) > 0$.}
\end{lem}

\medskip

\noindent
<!-- end head-4 -->
<!-- lem-5: body_md -->
\label{lem:positive}
{\it We have $\lambda(v) > 0$ and $\lambda(\tilde{v}) > 0$.}
<!-- end lem-5 -->
<!-- head-5: body_md -->
We first show $\lambda(\tilde{v}) > 0$. Since $\tilde{a}_{i+1}, \widetilde{\Delta}^{(2)}_i \geq 0$, the
recurrence \eqref{eq:recB} gives
$$\widetilde{\Delta}^{(4)}_{i+1} = \tilde{a}_{i+1}\,\widetilde{\Delta}^{(2)}_i + \tilde{b}_{i+1}\,\widetilde{\Delta}^{(4)}_i \geq \tilde{b}_{i+1}\,\widetilde{\Delta}^{(4)}_i.$$
Using $\kappa_i \geq 1/r_0$, $d_i \geq d_0/3$ (from \eqref{eq:di}), $L_i \geq 1$, and
$\theta_i \geq 2c_1$:
$$\tilde{b}_i = 1 + \kappa_i d_i L_i\theta_i \geq 1 + \frac{2c_1 d_0}{3r_0}.$$
Setting $c_0 = \log\!\left(1 + \frac{2c_1 d_0}{3r_0}\right) > 0$, it follows by induction that
$$\widetilde{\Delta}^{(4)}_m \geq e^{c_0 m}\,\widetilde{\Delta}^{(4)}_0 = e^{c_0 m} \quad \text{for all } m \geq 1,$$
since $\widetilde{\Delta}^{(4)}_0 = 1$. Therefore $\lambda(\tilde{v}) = \lim_{m\to\infty}\frac{1}{m}\log\widetilde{\Delta}^{(4)}_m
\geq c_0 > 0$.

The same argument applies to $\lambda(v)$, replacing $\tilde{b}_i$ by $b_i \geq \tilde{b}_i$ throughout
(since $\hat\kappa_i = \beta\kappa_i > \kappa_i L_i$ as $\beta > L_0 \geq L_i$). Hence
$\Delta^{(4)}_m \geq e^{c_0 m}$ for all $m \geq 1$, and $\lambda(v) \geq c_0 > 0$.
{\rule{6pt}{6pt}}

\bigskip

{\bf {To finish the proof of Theorem {{\rm\expandafter\@setref\csname r@thm1.1\endcsname\@firstoftwo{thm1.1}}}}}, let $v \in E_0$ and $\tilde{v} \in F_0$ with
$\|v\| = \|\tilde{v}\| = 1$. Then \eqref{eq:lyap} and \eqref{eq:uv} (with $v_m = e^s u_m \geq e^s$) imply
\begin{eqnarray*}
\lambda(v)
& = & \lim_{m\to\infty} \frac{1}{m} \log \Delta^{(4)}_m
  \geq \lim_{m\to\infty} \frac{1}{m} \log (e^{ms}\,\widetilde{\Delta}^{(4)}_m)\\
& = & \lim_{m\to\infty} \frac{1}{m} \left(ms + \log\widetilde{\Delta}^{(4)}_m\right)
  = s + \lambda(\tilde{v}).
\end{eqnarray*}

Since $s \geq \log(\beta c_1^2) > 0$, we have $\lambda(v) > \lambda(\tilde{v})$, so we can now
identify $\lambda_1(\beta) = \lambda(v)$ and $\lambda_2 (\beta) = \lambda(\tilde{v})$. Since $\lambda(\tilde{v})$ depends only on the dynamics restricted to $\Gamma$, it is
independent of $\beta$, so $\lambda_2(\beta) = \lambda_2$. By Lemma~{{\rm\expandafter\@setref\csname r@lem:positive\endcsname\@firstoftwo{lem:positive}}},
$\lambda_2 = \lambda(\tilde{v}) > 0$, and the gap satisfies
$$\lambda_1(\beta) - \lambda_2 \geq s \geq \log(\beta c_1^2) > 0,$$
establishing $\lambda_1 (\beta) \geq \lambda_2 + \log(\beta c_1^2) > \lambda_2 > 0$.  This completes the proof of Theorem~1.1.
