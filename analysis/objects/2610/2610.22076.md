<!-- theorem-1: body_md -->
\label{main-result}
    Let $G$ be a nilpotent group. For every $d, k\in \mathbb{N}$, any polynomials $p_{i,j}\in \mathbb{Z}[t]$, $i\in [d]$, $j\in [k]$, every $G$-measure-preserving system $(X,\mathcal B,\mu,G)$, with the transformations $T_1,\ldots,T_d\in G$, and every $f_0,\ldots,f_k\in L^\infty(\mu)$ bounded by $1$, we have the decomposition
    $$\int f_0\cdot \left( \prod_{i=1}^{d} T_i^{p_{i,1}(n)} \right) f_1 \cdot \ldots \cdot  \left( \prod_{i=1}^{d} T_i^{p_{i,k}(n)} \right) f_k d\mu = {\mathcal{N}}(n) + {\mathbf{e}}(n),  \quad n\in \mathbb{N}$$
where ${\mathcal{N}}$ is a generalized nilsequence bounded by $1$, and ${\mathbf{e}}$ is a null-sequence.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{Frantzikinakis-nilpotent}
    For every $s,d,k\in \mathbb{N}$, any polynomials $p_{i,j}\in \mathbb{Z}[t]$, $i\in [d]$, $j\in [k]$, there is $\ell\in \mathbb{N}$ such that for every measure-preserving system $(X,\mathcal{B},\mu,T_1,\ldots,T_{d})$ with the transformations $T_1,\ldots,T_{d}$ spanning a nilpotent group of class at most $s$, every $f_1,\ldots,f_k\in L^\infty(\mu)$ bounded by $1$, and $\varepsilon>0$ we have the decomposition 
\begin{equation*}
  \int \left( \prod_{i=1}^{d} T_i^{p_{i,1}(n)} \right) f_1 \cdot \ldots \cdot  \left( \prod_{i=1}^{d} T_i^{p_{i,k}(n)} \right) f_k d\mu = {\mathcal{N}}_{\varepsilon}(n) + {\mathbf{e}}_\varepsilon(n),  \quad n\in \mathbb{N}  
\end{equation*}
where ${\mathcal{N}}_{\varepsilon}$ is an $\ell$-step nilsequence bounded by $1$ and ${\mathbf{e}}_\varepsilon$ is an $\epsilon$-null-sequence.
<!-- end theorem-2 -->
<!-- proposition-1: body_md -->
\label{Weight-multicor-average}
Let $G$ be a nilpotent group and $(X,\mu,T)$ a $G$-system. Let $g_1,\ldots,g_k,h_1,\ldots,h_l$ be polynomial sequences in $G$. For $u_0,\ldots,u_k\in L^\infty(\mu)$, set
    $$\mathcal{D}(n)=\int u_0 \cdot \prod_{j=1}^k h_j(n)u_j d\mu. $$
    Then, for functions $f_1,\ldots,f_k\in L^\infty(\mu)$, the uniform limit
    \begin{equation}\label{eq-limit-correlation-and-functions}
        \lim_{N-M \to \infty} \frac{1}{N-M} \sum_{n=M+1}^N \mathcal{D}(n) \cdot  \prod_{j=1}^k g_j(n)f_j 
    \end{equation}
    exists in $L^2(\mu) $.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
By \cite[Lemma 3.5]{Zorin-Kranich16}, it is enough to prove that for every F\o lner sequence $\Phi=(\Phi_N)_N$, the limit in \cref{eq-limit-correlation-and-functions} for averages along $\Phi$ exists. Fix a F\o lner sequence $\Phi=(\Phi_N)_N$. For $N,M\in \mathbb{N}$, we have that 
    \begin{align*}
        &\left\lVert\mathcal{A}_{N,M} \mathcal{D}(n) \cdot  \prod_{j=1}^k g_j(n)f_j    \right\rVert_{L^2(\mu)}^2 \\
        &= \int \left| \int  u_0(y) \cdot \left(  \mathcal{A}_{N,M}  \prod_{j=1}^k h_j(n)u_j(y) \cdot  \prod_{j=1}^k g_j(n)f_j(x)   \right) d\mu(y)  \right|^2 d\mu(x)\\
        &\leq \left\lVertu_0\right\rVert_{L^2(\mu)}^2 \int  \int  \left|  \mathcal{A}_{N,M}  \prod_{j=1}^k h_j(n)u_j(y) \cdot  \prod_{j=1}^k g_j(n)f_j(x)   \right|^2  d\mu(y) d\mu(x),
    \end{align*}
    where we used the Cauchy-Schwarz inequality in the last step. We observe that 
    \begin{align*}
         \int  \int  \left|  \mathcal{A}_{N,M}  \prod_{j=1}^k h_j(n)u_j(y) \cdot  \prod_{j=1}^k g_j(n)f_j(x)   \right|^2  d\mu(y) d\mu(x)&= \left\lVertA_{N,M}  \prod_{j=1}^k (h_j\times g_j)(n)\left(u_j\otimes f_j\right)\right\rVert_{L^2(\mu\otimes \mu)}^2
    \end{align*}
    where we denote by $(h_j\times g_j)(n)$ the action of $G\times G$ in the product $X\times X$. Since clearly $h_j\times g_j$ is a polynomial sequence in the nilpotent group $G\times G$, we have by Walsh \cite[Theorem 5.1]{Walsh12} that 
    $$\lim_{\min\{N,M\}\to \infty}\left\lVertA_{N,M}  \prod_{j=1}^k (h_j\times g_j)(n)(u_j\otimes f_j)\right\rVert_{L^2(\mu\otimes \mu)}=0, $$
    concluding that 
    $$ N\mapsto \frac{1}{|\Phi_N|} \sum_{n\in \Phi_N} \mathcal{D}(n) \cdot  \prod_{j=1}^k g_j(n)f_j$$
    is a Cauchy sequence in $L^2(\mu)$, and thus it converges.
<!-- end proof-1 -->
<!-- proposition-2: body_md -->
\label{nil-plus-uniform-error}
Suppose, for every $\varepsilon>0$, a sequence $a(n)$ can be written as a sum of a generalized nilsequence $a_{\text{nil},\varepsilon}(n)$, a null-sequence $a_{\text{null},\varepsilon}(n)$, and a sequence $a_{\varepsilon}(n)$ which is uniformly bounded by $\varepsilon$. Then, $a(n)$ is the sum of a generalized nilsequence and a null-sequence.
<!-- end proposition-2 -->
<!-- lemma-1: body_md -->
\label{nil+nul-inequality}
    Let ${\mathcal{N}}$ be a generalized nilsequence and ${\mathbf{e}}$ be a null-sequence. Then
    $$\left\lVert{\mathcal{N}}\right\rVert_\infty\leq \left\lVert{\mathcal{N}}+{\mathbf{e}}\right\rVert_{\infty}. $$
