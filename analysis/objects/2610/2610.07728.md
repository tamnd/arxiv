<!-- theorem-1: body_md -->
\label{thm:main}
For every $n\ge1$, let $\mathcal C_n$ denote the class of isotropic
log-concave probability measures on $\mathbb R^n$. Then
\begin{equation}\label{eq:main-bound}
 \sup_{\mu\in\mathcal C_n}\ \sup_f
 \frac{\operatorname{Var}_\mu f}{\int|\nabla f|^2\,d\mu}
 \le25,
\end{equation}
where the inner supremum is over locally Lipschitz $f\in L^2(\mu)$
with nonzero finite gradient energy.

<!-- lemma-1: body_md -->
\label{lem:weighted-moment-map-energy}
Let $X$ be isotropic and log-concave, with density $e^{-V}$ on a bounded
open convex set $K$, where $V$ is smooth and convex on a neighborhood
of $\overline K$. Let $\nu(dy)=e^{-\phi(y)}\,dy$ be its moment-measure
representation and put $H=D^2\phi$. For every symmetric matrix $T$,
\begin{equation}\label{eq:weighted-moment-map-input}
 \int\operatorname{Tr}(THTH)\,d\nu
 \le2\operatorname{Tr}(T^2).
\end{equation}

<!-- lemma-2: body_md -->
\label{lem:sharp-third-pair-slot}
Let $X$ be an isotropic log-concave random vector in $\mathbb R^n$, and set
\[
 (S_k)_{ij}=\mathbb E[X_kX_iX_j],\qquad k=1,\ldots,n.
\]
Then, for every real symmetric matrix $T$,
\begin{equation}\label{eq:sharp-third-pair-slot}
 \sum_{k=1}^n\operatorname{Tr}(TS_kTS_k)
 \le4\operatorname{Tr}(T^2).
\end{equation}
Equivalently, the self-adjoint operator
$T\mapsto\sum_k S_kTS_k$ on symmetric matrices is bounded above by $4I$.
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
First assume the hypotheses of
Lemma~\ref{lem:weighted-moment-map-energy}, and define the symmetric
matrix field $\tau$ by
\[
 \tau(\nabla\phi(y))=D^2\phi(y).
\]
The mean identity in \cite[Equation~(11)]{ChenKlartag} gives
$\mathbb E\tau(X)=I$. Their third-moment identity
\cite[Lemma~3.7]{ChenKlartag}, written in the variable
$X=\nabla\phi(Y)$ with $Y\sim\nu$, is
\begin{equation}\label{eq:third-tensor-steinkernel-coefficient}
 \mathbb E[X_k(\tau(X)-I)]=\tfrac12S_k.
\end{equation}
The cited weighted estimate becomes
$\mathbb E\operatorname{Tr}(T\tau T\tau)\le2\operatorname{Tr}(T^2)$
under this same change of variables.

For $T\succeq0$, apply Bessel's inequality coordinate by coordinate
to the matrix-valued function $T^{1/2}(\tau-I)T^{1/2}$.
The functions $X_1,\ldots,X_n$ are orthonormal in $L^2$ by isotropy.
Using \eqref{eq:third-tensor-steinkernel-coefficient}, we obtain
\begin{align*}
 \frac14\sum_k\operatorname{Tr}(TS_kTS_k)
 &\le\mathbb E\operatorname{Tr}\bigl(T(\tau-I)T(\tau-I)\bigr)=\mathbb E\operatorname{Tr}(T\tau T\tau)-\operatorname{Tr}(T^2)
 \le\operatorname{Tr}(T^2).
\end{align*}
The equality uses $\mathbb E\tau=I$; the final inequality is
\eqref{eq:weighted-moment-map-input}.

For an arbitrary symmetric $T$, choose an orthonormal eigenbasis and
write its eigenvalues as $t_1,\ldots,t_n$. In this basis,
\[
 \sum_k\operatorname{Tr}(TS_kTS_k)
 =\sum_{k,i,j}t_it_j(S_k)_{ij}^2
 \le\sum_{k,i,j}|t_i||t_j|(S_k)_{ij}^2
 =\sum_k\operatorname{Tr}(|T|S_k|T|S_k).
\]
Apply the positive-semidefinite case to $|T|$ and use
$\operatorname{Tr}(|T|^2)=\operatorname{Tr}(T^2)$.

For a general isotropic log-concave law, use the approximating laws
from \citet[Lemma~A.1]{Letwin}. They satisfy the regularity assumptions
above, remain isotropic, and converge in every polynomial moment of
degree at most four. Their third-moment matrices therefore converge
entrywise to each $S_k$. For a fixed $T$, both sides of
\eqref{eq:sharp-third-pair-slot} pass to the limit, proving the assertion.

<!-- proposition-1: body_md -->
\label{prop:one-step-coefficient}
Fix $k\ge2$, write $j=k+1$, and let $h_1=1,h_2,\ldots,h_k$ be
positive constants such that, for every isotropic log-concave law,
\[
 \|B_a(f)\|^2\le h_a\|f\|_2^2
 \qquad(1\le a\le k).
\]
Let $h_j>0$ be a proposed next bound, and put
\[
 d_\ell=\sum_{a=2}^{\ell-1}
       \binom{\ell}{a}\sqrt{h_a h_{\ell-a+1}}
 \qquad(\ell=k,j),
\]
where an empty sum is zero. Every index on the right is at most $k$.
Choose $0<\alpha<1$ and a positive, decreasing, concave $C^2$ function
$\phi$ on a neighborhood of $[0,1]$, with $\phi(0)=1$. Set
\[
 A=\frac4{1-\alpha}-2,\qquad q=-\frac{\phi'}{\phi}.
\]
If, for $0\le u\le1$,
\begin{align}
 q(u)\left[
  \alpha\frac{h_j}{h_k}\phi(u)
  -\bigl(Ak^2+(A-3)k+2\bigr)u
  -\frac{2d_k}{\sqrt{h_k}}\sqrt u
 \right]
 >2j^2-j+2+\frac{2d_j}{\sqrt{h_j\phi(u)}},
 \label{eq:one-step-coefficient-condition}
\end{align}
then $\|B_j(f)\|^2\le h_j\|f\|_2^2$ for every isotropic
log-concave law.
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
Fix a dimension and suppose the proposed bound fails. Maximize
\[
 \frac{E_j}{g\phi(E_k/(h_kg))}
\]
over isotropic laws and nonzero test functions, as in the preceding
compactness argument. Because $\phi\le1$, the maximum is larger than
$h_j$. At a maximizing pair, normalize $g=1$ and write
\[
 x=E_j,\qquad y=E_k=h_ku,\qquad
 w=\frac{xq(u)}{h_k}.
\]
Then $x>h_j\phi(u)$. The maximum-principle inequality
\eqref{eq:barrier-maximum-inequality} and the stationarity identity
\eqref{eq:barrier-terminal-cancellation} apply unchanged.

Separate the terminal cumulant from each remainder:
\[
 R_\ell=\kappa_{\ell+1}(\cdot,B_1)+R_\ell^\circ.
\]
The contraction estimate
\eqref{eq:barrier-cumulant-contraction} bounds each summand of
$R_\ell^\circ$ by
$\sqrt{h_a h_{\ell-a+1}}$, and therefore
$\|R_\ell^\circ\|\le d_\ell$ for $\ell=k,j$.
Apply \eqref{eq:barrier-upper-energy} at rank $j$ and
\eqref{eq:barrier-adjustable-energy} with $\beta=\alpha$ at rank $k$.
The two terminal terms together cost at most $2x(1+uq)$ by the
test function variation. Cauchy--Schwarz is used only on the remaining
terms. We obtain
\begin{align*}
 w\alpha x-(2j^2-j)x
 -w\bigl(Ak^2+(A-3)k\bigr)y
 -2x(1+uq)-2\sqrt{x}\,d_j-2w\sqrt{y}\,d_k \leq 0
\end{align*}
Divide by $x>0$ and use $y=h_ku$:
\[
 q\left[\alpha\frac{x}{h_k}
       -\bigl(Ak^2+(A-3)k+2\bigr)u
       -\frac{2d_k}{\sqrt{h_k}}\sqrt u\right]
       -(2j^2-j+2)-\frac{2d_j}{\sqrt{x}} \leq 0
\]
Condition \eqref{eq:one-step-coefficient-condition} implies $q>0$.
Since $x>h_j\phi(u)$, the last displayed left-hand side is strictly
larger than the positive margin in that condition. This is a
contradiction. The next bound holds in each dimension, with the same
constant $h_j$.

<!-- theorem-2: body_md -->
\label{thm:geometric-coefficients}
For every isotropic log-concave probability measure,
\begin{equation}\label{eq:final-coefficient-tail}
 c_1=1,\qquad c_2^2\le2,\qquad
 c_d^2\le B K^{d-3}\quad(d\ge3),
 \qquad B=\frac{121}{4},\quad K=\frac{223}{10}.
\end{equation}

<!-- proof-3: body_md -->
The linear and quadratic bounds were proved in
\eqref{eq:barrier-linear-seed} and
\eqref{eq:barrier-quadratic-seed}. We first establish the cubic bound
and then apply Proposition~\ref{prop:one-step-coefficient} inductively.

\paragraph{The Cubic Coefficient.}
Take $k=2$, $j=3$, $h_2=8$, and the proposed bound $h_3=36B$.
The constants in Proposition~\ref{prop:one-step-coefficient} are
$d_2=0$ and $d_3=\binom32\sqrt{8\cdot8}=24$.
Consequently its criterion is
\begin{equation}\label{eq:short-cubic-criterion}
 q(u)\left[\frac92\alpha B\phi(u)-(6A-4)u\right]
 >17+\frac8{\sqrt{B\phi(u)}},\qquad
 A=\frac4{1-\alpha}-2,\quad q=-\frac{\phi'}{\phi}.
\end{equation}
Choose $\alpha=7/13$, so that $A=20/3$ and $6A-4=36$, and define
\[
 U_3(v)=\frac7{117}(1-v)(65v+1).
\]
Its derivatives and endpoint values are
\[
 U_3'(v)=\frac{448-910v}{117},\qquad
 U_3''(v)=-\frac{910}{117},\qquad U_3(1)=0,
\]
\[
 U_3(1/2)-1=\frac1{468}>0,\qquad
 U_3'(1/2)=-\frac7{117}<0.
\]
Thus $U_3$ is strictly decreasing and concave on $[1/2,1]$, and its
range contains $[0,1]$. Its inverse $\phi$ on $[0,1]$ satisfies
\[
 \phi(0)=1,\qquad \phi>1/2,\qquad
 \phi'\!<0,\qquad
 \phi''(u)=-\frac{U_3''(\phi(u))}{U_3'(\phi(u))^3}<0.
\]
The derivative of $U_3$ is nonzero at the preimages of both endpoints;
therefore the same inverse extends smoothly to a neighborhood of
$[0,1]$.
Multiplying the displayed expression for $U_3'$ by $18v+9/16$ and
expanding $U_3$ gives
\[
 \left(18v+\frac9{16}\right)U_3'(v)
 =36U_3(v)-\frac{7623}{104}v
 =36U_3(v)-\frac92\alpha Bv.
\]
Writing $u=U_3(v)$ and using $\phi'(u)=1/U_3'(v)$ yields
\begin{equation}\label{eq:short-cubic-inverse-identity}
 -\phi'(u)\left[\frac92\alpha B\phi(u)-36u\right]
 =18\phi(u)+\frac9{16}.
\end{equation}
For $v>0$, the inequality
$\sqrt v\le(2/3)v+3/8$ follows from
$(\sqrt v-3/4)^2\ge0$. Since $8/\sqrt B=16/11<3/2$,
\[
 17v+\frac8{\sqrt B}\sqrt v
 <17v+\frac32\left(\frac23v+\frac38\right)
 =18v+\frac9{16}.
\]
Use this strict inequality in
\eqref{eq:short-cubic-inverse-identity} and divide by $\phi(u)>0$.
The result is \eqref{eq:short-cubic-criterion}.
Proposition~\ref{prop:one-step-coefficient} now proves
$\|B_3(f)\|^2\le36B\|f\|_2^2$, equivalently $c_3^2\le B$.

\paragraph{The Convolution with One Exceptional Degree.}
For the subsequent induction define
\[
 C=\frac BK=\frac{605}{446},\qquad
 h_1=1,\quad h_2=8,\quad
 h_d=CK^{d-2}(d!)^2\quad(d\ge3).
\]
Then $h_3=36B$, so the cubic estimate supplies the initial case.
The useful normalized constants are
\[
 r=\sqrt{\frac CK}=\frac{55}{223}<\frac14,
 \qquad e=\sqrt{\frac2C},\qquad
 \delta=\frac{e^2-1}{2}=\frac{287}{1210}\in(0,1/2).
\]
The sequence $h_d$ is geometric after its factorial normalization,
except at degree two. We record how this single exception enters
its convolution. For every $\ell\ge3$, put
\[
 d_\ell=\sum_{a=2}^{\ell-1}
       \binom{\ell}{a}\sqrt{h_a h_{\ell-a+1}},\qquad
 C_\ell=\frac{\ell(\ell-1)}2-1.
\]
We claim that
\begin{equation}\label{eq:short-one-exception-remainder}
 d_\ell\le r\sqrt{h_\ell}
                  [C_\ell+\delta(\ell+1)]\qquad(\ell\ge3).
\end{equation}
At $\ell=3$, the sum is $24$ and $r\sqrt{h_3}=6C$.
Their ratio is $4/C=2e^2=C_3+4\delta$, proving the claim in that case.
For $\ell\ge4$, set $b'=\ell-a+1$, and write
$e_2=e$, $e_a=1$ for $a\ge3$. Then
\[
 h_a=CK^{a-2}(a!)^2e_a^2\quad(a\ge2),
\]
and cancellation of factorials gives
\begin{align*}
 \binom{\ell}{a}\sqrt{h_a h_{b'}}
 &=\frac{\ell!}{a!(\ell-a)!}
       CK^{(\ell-3)/2}a!b'!e_a e_{b'}\\
 &=r\sqrt{h_\ell}\,b'e_a e_{b'}.
\end{align*}
The weights $b'$ range from $2$ to $\ell-1$, so their sum is
$\ell(\ell-1)/2-1=C_\ell$.
The two splits containing degree two are distinct. Their weights are
$\ell-1$ and $2$, and each contains exactly one exceptional factor.
Thus
\[
 \frac{d_\ell}{r\sqrt{h_\ell}}
 =C_\ell+(\ell+1)(e-1)
 \le C_\ell+\delta(\ell+1),
\]
where the last inequality follows from
$\delta-(e-1)=(e-1)^2/2\ge0$.
This proves \eqref{eq:short-one-exception-remainder}.
At an induction step with $j=k+1$, every coefficient entering
$d_k$ and $d_j$ has degree at most $k$; the proposed constant $h_j$
appears in \eqref{eq:short-one-exception-remainder} only as an
algebraic normalization.

\paragraph{One Cubic Denominator for All Subsequent Degrees.}
Choose
\[
 \alpha=\frac{17}{33},\qquad A=\frac4{1-\alpha}-2=\frac{25}{4},
 \qquad P=\frac{446}{165},\qquad Q=\frac{281}{8415},
\]
and set
\[
 U(z)=Pz+Q-(P+Q)\left(\frac{17z+1}{18}\right)^3.
\]
Writing $w=(17z+1)/18$, we have
\[
 U'(z)=P-\frac{17}{6}(P+Q)w^2,\qquad
 U''(z)=-\frac{289}{54}(P+Q)w.
\]
In particular $U''<0$ for $z\ge4/7$. The identities
$P=4K/33$ and $Q=(P-1)/51$ give
\begin{equation}\label{eq:short-tail-inverse-identity}
 \left(\frac{17}{8}z+\frac18\right)U'(z)
 =\frac{51}{8}U(z)+\frac18-\alpha Kz.
\end{equation}
Indeed the cubic terms on the two sides are both
$-(51/8)(P+Q)w^3$, the coefficients of $z$ agree because
$(17/4)P=\alpha K$, and the constants agree because
$51Q=P-1$.
The rational endpoint evaluations are
\[
 U(1)=0,\qquad
 U(4/7)-1=\frac{32117}{36673560}>0,\qquad
 U'(4/7)=-\frac{20953}{476280}<0.
\]
Thus $U$ is strictly decreasing on $[4/7,1]$ and has an inverse
$\phi$ on $[0,1]$, with
\[
 \phi(0)=1,\qquad \phi>4/7,\qquad
 \phi'<0,\qquad \phi''=-\frac{U''(\phi)}{U'(\phi)^3}<0.
\]
As with $U_3$, the derivative is nonzero at both relevant endpoints,
so this inverse is smooth on a neighborhood of $[0,1]$.
The elementary inequality $\sqrt t\le(t+1)/2$ and $r<1/4$ imply
\[
 r\sqrt u\le\frac18(u+1),\qquad
 2z+r\sqrt z<\frac{17}{8}z+\frac18\quad(z>0).
\]
For $u=U(z)\in[0,1]$, the sign $U'(z)<0$ and
\eqref{eq:short-tail-inverse-identity} therefore give
\begin{align*}
 \alpha Kz-Au-r\sqrt u+(2z+r\sqrt z)U'(z) >\alpha Kz-\frac{51}{8}u-\frac18
       +\left(\frac{17}{8}z+\frac18\right)U'(z)=0.
\end{align*}
Set $q=-\phi'/\phi=-1/(zU'(z))>0$ and divide by $-zU'(z)>0$.
We obtain the positive limiting margin
\begin{equation}\label{eq:short-bulk-margin}
 \mathcal M_\infty(u)
 :=q(u)[\alpha K\phi(u)-Au-r\sqrt u]
       -2-\frac r{\sqrt{\phi(u)}}>0.
\end{equation}

\paragraph{Verification in Every Finite Degree.}
Suppose the coefficient bounds have been established through degree
$k\ge3$, and put $j=k+1$. Then $h_j/h_k=Kj^2$.
Use \eqref{eq:short-one-exception-remainder} for $d_k$ and $d_j$
in Proposition~\ref{prop:one-step-coefficient}.
The difference between the two sides of its criterion, divided by
$j^2$, is bounded below by
\begin{align*}
 \mathcal M_k(u)={}&
 q\left[\alpha K\phi
   -\frac{Ak^2+(A-3)k+2}{j^2}u
   -\frac{2r}{j^2}[C_k+\delta(k+1)]\sqrt u\right]\\
 &-2+\frac1j-\frac2{j^2}
   -\frac{2r}{j^2\sqrt\phi}[C_j+\delta(j+1)].
\end{align*}
Put $t=1/j$, so $0<t\le1/4$. The three coefficients depending on
rank expand as
\[
 \frac{Ak^2+(A-3)k+2}{j^2}
       =A-(A+3)t+5t^2,
\]
\[
 \frac{2[C_j+\delta(j+1)]}{j^2}
       =1+(2\delta-1)t+(2\delta-2)t^2,
 \qquad
 \frac{2[C_k+\delta(k+1)]}{j^2}
       =1+(2\delta-3)t.
\]
Substitution gives
\[
 \mathcal M_k-\mathcal M_\infty
   =t\mathsf B+t^2\mathsf C,
\]
where
\begin{align*}
 \mathsf B&=1+(A+3)qu+(1-2\delta)\frac r{\sqrt\phi}
                       +(3-2\delta)rq\sqrt u,\\
 \mathsf C&=-5qu+(2-2\delta)\frac r{\sqrt\phi}-2.
\end{align*}
Because $0<\delta<1/2$ and $q>0$, we have $\mathsf B>0$.
At the other endpoint of the interval for $t$,
\[
 \mathsf B+\frac14\mathsf C
 =\frac12+(A+\tfrac74)qu
   +(\tfrac32-\tfrac52\delta)\frac r{\sqrt\phi}
   +(3-2\delta)rq\sqrt u>0.
\]
The function $\mathsf B+t\mathsf C$ is affine in $t$, so it is
positive throughout $[0,1/4]$. Consequently
$\mathcal M_k>\mathcal M_\infty>0$.
Proposition~\ref{prop:one-step-coefficient} proves the next coefficient
bound. The cubic seed starts the induction, and therefore
$c_d^2\le CK^{d-2}=BK^{d-3}$ for every $d\ge3$.

<!-- lemma-3: body_md -->
\label{full-hodge}
For every $m\ge0$ and $F\in\mathcal K_{m+1}$,
\begin{equation}\label{full-hodge-bound}
 \|S_mF\|_2^2\ge\|\nabla F\|_2^2+a\|F\|_2^2.
\end{equation}
The coefficient of the curvature term does not depend on $m$.
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
Write $F=\nabla^{m+1}\psi$, $\psi\in C_c^\infty$, and put
$Y=\operatorname{div}_V F$.
The commutator $[\partial_i,\partial_j^*]=V_{ij}$ gives
\[
 \sum_{i,j}\langle\partial_i^*F_{Ii},\partial_j^*F_{Ij}\rangle_{L^2}
 =\sum_{i,j}\mathbb E[\partial_jF_{Ii}\,\partial_iF_{Ij}]
  +\sum_{i,j}\mathbb E[V_{ij}F_{Ii}F_{Ij}].
\]
Indeed, move $\partial_j^*$ to the first factor, commute
$\partial_j$ past $\partial_i^*$, and move $\partial_i^*$ back.
Since $\nabla F$ is symmetric, summation over $I$ yields
\begin{equation}\label{full-divergence-energy}
 \|P_{\mathcal C_m}Y\|_2^2
 =\|\nabla F\|_2^2+B_{\mathsf{K}}(F)-\|(I-P_{\mathcal C_m})Y\|_2^2,
 \qquad
 B_{\mathsf{K}}(F)=\sum_{I,i,j}\mathbb E[V_{ij}F_{Ii}F_{Ij}].
\end{equation}
For $m=0$ the projection is the identity and
$B_{\mathsf{K}}(F)\ge a\|F\|_2^2$, which proves the assertion. Assume $m\ge1$.
Differentiating $Y$ and using compatibility gives
\begin{equation}\label{full-divergence-curl}
 \partial_kY_{lI}-\partial_lY_{kI}
   =\sum_j(V_{kj}F_{lIj}-V_{lj}F_{kIj}),\qquad |I|=m-1.
\end{equation}
Every term differentiating $F$ cancels: both its second derivatives and
the first derivatives multiplied by $V_j$ are symmetric in $k,l$.
Only the displayed derivatives of $V_j$ remain.
For example, when $m=1$ and $D^2V$ is diagonal at the point,
the $kl$ component of this curl is $(V_{kk}-V_{ll})F_{kl}$.
Thus the correction is driven by differences between curvature
directions. The constrained complex below organizes all such
differences, so that their cost can be compared with the curvature
already present in \eqref{full-divergence-energy}.

\paragraph{A Complex Encoding Compatibility.}
On $\operatorname{Sym}^q\mathbb R^n$, use the orthonormal multiplicity
basis $e_\alpha$, $|\alpha|=q$, whose ordered components equal
$\sqrt{\alpha!/q!}$ on tuples of multiplicity $\alpha$ and vanish
otherwise. Define
\[
 a_i e_\alpha=\sqrt{\alpha_i}\,e_{\alpha-e_i},\qquad
 a_i^*e_\alpha=\sqrt{\alpha_i+1}\,e_{\alpha+e_i}.
\]
The first expression is zero if $\alpha_i=0$. These operators satisfy
$[a_i,a_j^*]=\delta_{ij}I$ and $\sum_i a_i^*a_i=qI$.
Contraction in one physical tensor slot is $a_i/\sqrt q$.
On exterior forms let $\varepsilon_i$ be wedge multiplication by $e_i$
and $\iota_i=\varepsilon_i^*$, so
$\varepsilon_i\iota_j+\iota_j\varepsilon_i=\delta_{ij}I$.
Set
\[
 b=\sum_i\varepsilon_i a_i,\qquad b^*=\sum_i a_i^*\iota_i.
\]
The commutation relations give
\begin{equation}\label{full-algebraic-complex}
 b^2=0,\qquad bb^*+b^*b=(q+p)I
 \quad\text{on }\operatorname{Sym}^q\mathbb R^n\otimes\Lambda^p\mathbb R^n.
\end{equation}
The second identity in \eqref{full-algebraic-complex} is the classical
polynomial de~Rham--Koszul homotopy identity; see, for example \citet[Theorem~3.1, Equation~(3.9)]{ArnoldFalkWinther}.
Under the identification $e_\alpha\leftrightarrow y^\alpha/\sqrt{\alpha!}$,
$b$ is exterior differentiation in the auxiliary variables $y$, and
$b^*$ is contraction with the radial vector field.
To verify the identity directly in our normalization, commute $a_i$ past
$a_j^*$ in $bb^*$ and use
$\iota_j\varepsilon_i=\delta_{ij}I-\varepsilon_i\iota_j$ in $b^*b$.
The mixed terms cancel, leaving
$\sum_i a_i^*a_i+\sum_i\varepsilon_i\iota_i=(q+p)I$.

Fix $q=m-1$ and put
\[
 \mathcal F_p=\ker b\subset\operatorname{Sym}^{m-1}\mathbb R^n\otimes\Lambda^p\mathbb R^n,
 \qquad P_p=\frac{bb^*}{m+p-1},\qquad p\ge1.
\]
Here $P_p$ is the orthogonal projection onto $\mathcal F_p$: $bb^*$ has range in
$\ker b$ and equals $(m+p-1)I$ there by
\eqref{full-algebraic-complex}. The map
\[
 I_m:\operatorname{Sym}^m\mathbb R^n\longrightarrow \mathcal F_1,
 \qquad I_mT=\frac{bT}{\sqrt m},
\]
is an isometry onto $\mathcal F_1$. Its isometry property follows from
$b^*b=mI$ on exterior degree zero. For $\omega\in \mathcal F_1$,
$bb^*\omega=m\omega$, proving surjectivity.

Exterior differentiation $d=\sum_i\varepsilon_i\partial_i$ satisfies
$bd+db=0$ and $d^2=0$. It therefore defines closed operators
$d_p:L^2(\mu;\mathcal F_p)\to L^2(\mu;\mathcal F_{p+1})$ on their maximal distributional
domains. Under $I_m$, $\ker d_1$ is precisely $\mathcal C_m$.
Lemma~\ref{full-complex-domains} identifies the adjoint as $P_pd^*$,
where $d^*=\sum_i\iota_i\partial_i^*$, and supplies a common smooth core
for the form
\[
 \ell(\omega)=\|d_2\omega\|_2^2+\|d_1^*\omega\|_2^2
             =\|d\omega\|_2^2+\|P_1d^*\omega\|_2^2.
\]
This is the energy of the constrained Hodge Laplacian. We use its
variational form, so no inverse differential operator is needed below.

\paragraph{The Energy of the Correction.}
Initially let $\omega\in C_c^\infty(\mathbb R^n;\mathcal F_2)$. Define
\[
 \mathsf{K}_{\rm ext}=\sum_{i,j}V_{ij}\varepsilon_i\iota_j,\qquad
 \mathsf{K}_{\rm sym}=\sum_{i,j}V_{ij}a_j^*a_i,\qquad
 c=\sum_i a_i\partial_i^*,\quad c^*=\sum_i a_i^*\partial_i.
\]
Expanding $dd^*+d^*d$ gives
\[
 dd^*+d^*d
 =\sum_{i,j}(\varepsilon_i\iota_j+\iota_j\varepsilon_i)
                    \partial_j^*\partial_i
   +\sum_{i,j}V_{ij}\varepsilon_i\iota_j
 =\sum_i\partial_i^*\partial_i+\mathsf{K}_{\rm ext}.
\]
Also $bd^*+d^*b=c$. Since $b\omega=0$ and
$I-P_1=b^*b/m$,
\[
 \|(I-P_1)d^*\omega\|_2^2=\frac1m\|c\omega\|_2^2.
\]
To check the norm identity, $b^*b/m$ is an orthogonal projection, so its
squared norm on a vector $z$ equals
$\langle z,b^*bz\rangle/m=\|bz\|^2/m$.
Consequently
\begin{equation}\label{full-hook-energy}
 \ell(\omega)=\|\nabla\omega\|_2^2
       +\mathbb E\langle\omega,\mathsf{K}_{\rm ext}\omega\rangle
       -\frac1m\|c\omega\|_2^2.
\end{equation}
A second commutator calculation gives
\[
 cc^*-c^*c
 =\sum_i\partial_i^*\partial_i
   +\sum_{i,j}a_j^*a_i(\partial_i^*\partial_j-\partial_j\partial_i^*)
 =\sum_i\partial_i^*\partial_i-\mathsf{K}_{\rm sym}.
\]
The row operator $(a_i^*)_i$ from a sum of symmetric degree-$m-1$
spaces to symmetric degree $m$ has squared norm $m$, because its product
with its adjoint is $\sum_i a_i^*a_i=mI$ on the target. Applied to the
components $\partial_i\omega$, this yields
$\|c^*\omega\|_2^2\le m\|\nabla\omega\|_2^2$. Thus
\[
 \|c\omega\|_2^2
 =\|c^*\omega\|_2^2-\|\nabla\omega\|_2^2
                      +\mathbb E\langle\omega,\mathsf{K}_{\rm sym}\omega\rangle
 \le(m-1)\|\nabla\omega\|_2^2
                      +\mathbb E\langle\omega,\mathsf{K}_{\rm sym}\omega\rangle.
\]
Substituting into \eqref{full-hook-energy} gives
\begin{equation}\label{full-hook-lower-bound}
 \ell(\omega)\ge\frac1m\|\nabla\omega\|_2^2
                    +\mathbb E\langle\omega,Q\omega\rangle,
 \qquad Q=P_2(\mathsf{K}_{\rm ext}-m^{-1}\mathsf{K}_{\rm sym})P_2.
\end{equation}

\paragraph{A Pointwise Completed Square.}
At a fixed point diagonalize $\mathsf{K}$, with eigenvalues $\kappa_i\ge a$,
and put $N=m+1$. Only pointwise algebra follows, so no derivatives of the chosen eigenbasis enter. Decompose by the total multiplicities $\alpha$ of the
symmetric and exterior indices, so $|\alpha|=N$.
In this block set $r_i=\sqrt{\alpha_i}$ and
$S=\sum_i\alpha_i\kappa_i$, restricting to indices with $\alpha_i>0$.
The map $b$ is wedge multiplication by $r$: inserting an exterior index
$i$ lowers its symmetric multiplicity and contributes $\sqrt{\alpha_i}$.
Therefore every element of $\mathcal F_2$ in this block has the unique form
\[
 \omega=r\wedge v,\qquad v\perp r,\qquad |\omega|^2=N|v|^2.
\]
Indeed, $r\wedge\omega=0$ and
$\iota_r(r\wedge\omega)+r\wedge\iota_r\omega=N\omega$ show that
$v=N^{-1}\iota_r\omega$ works; $\iota_r^2=0$ gives $v\perp r$.
If the block has only one active index, $\mathcal F_2$ is zero and the formulas
below have $v=0$.

On the exterior pair $ij$, the uncompressed coefficient of $Q$ is
$[N(\kappa_i+\kappa_j)-S]/m$. Expansion gives
\[
 \sum_{i<j}(\kappa_i+\kappa_j)(r_iv_j-r_jv_i)^2
 =S|v|^2+N\sum_i\kappa_iv_i^2
       -2(r\cdot v)\sum_i\kappa_ir_iv_i.
\]
The last term vanishes. Combining this equality with
$|r\wedge v|^2=N|v|^2$ proves
\begin{equation}\label{full-block-Q}
 \langle\omega,Q\omega\rangle=\frac{N^2}{m}\langle v,\mathsf{K}v\rangle
 \ge\frac{Na}{m}|\omega|^2.
\end{equation}
In particular the compressed operator $Q$ is positive. This is a
statement on $\ker b$; the uncompressed diagonal expression need not
be positive. It is also bounded, since $\mathsf{K}\preceq MI$.
The joint graph core in Lemma~\ref{full-complex-domains}, applied to
differences of approximants in \eqref{full-hook-lower-bound}, makes their
full gradients Cauchy. Closedness of the derivative and boundedness of
$Q$ therefore extend that bound to the full form domain of $\ell$.

Let $f=F_\alpha$ be the coefficient of $F$ in the normalized symmetric
rank-$N$ basis, and let $t$ be the corresponding block of $dI_mY$.
Equation~\eqref{full-divergence-curl} gives
\[
 t_{ij}=\frac{(\kappa_i-\kappa_j)\sqrt{\alpha_i\alpha_j}}
                  {\sqrt{Nm}}f,
 \qquad t=-\frac f{\sqrt{Nm}}r\wedge \mathsf{K}r.
\]
The denominator follows from two physical contractions, first
$a_j/\sqrt N$ on $F$ and then $a_i/\sqrt m$ in $I_m$.
The contribution of this block to the unintegrated form $B_{\mathsf{K}}$ is
$(S/N)f^2$: curvature acts in one tensor slot, and averaging that slot
over all $N$ slots gives the same quadratic form on a symmetric tensor.
Using $v\perp r$ and \eqref{full-block-Q}, we obtain the exact identity
\begin{align*}
 \frac SN f^2-2\langle t,\omega\rangle+\langle\omega,Q\omega\rangle=\left\langle
       \frac f{\sqrt N}r+\frac N{\sqrt m}v,
       \mathsf{K}\left(\frac f{\sqrt N}r+\frac N{\sqrt m}v\right)
       \right\rangle\ge a\left(f^2+\frac{N^2}{m}|v|^2\right)\ge af^2.
\end{align*}
Thus the correction can consume at most the curvature in excess of $a$.

Finally apply the variational projection identity of
Lemma~\ref{full-complex-domains} to $u=I_mY$:
\[
 \|(I-P_{\mathcal C_m})Y\|_2^2
 =\sup_{\omega\in\operatorname{Dom}(d_1^*)\cap\operatorname{Dom}(d_2)}
       \{2\langle dI_mY,\omega\rangle_{L^2}-\ell(\omega)\}.
\]
Drop the nonnegative gradient term in
\eqref{full-hook-lower-bound}, use the completed square in every block,
and integrate. Each expression inside this supremum is at most
$B_{\mathsf{K}}(F)-a\|F\|_2^2$. Equation~\eqref{full-divergence-energy} and
\eqref{full-adjoint-projection} prove \eqref{full-hodge-bound}.

 \int\langle\nabla^q\phi,\nabla^q\psi\rangle\,d\mu=0
 \quad\text{for every }\psi\in C_c^\infty.
\end{equation}
The left side is initially a distributional pairing.

Write $w=Z^{-1}e^{-V}$. Equation
\eqref{eq:dense-divergence-weak-equation} means
$\sum_I\partial_I(w\partial_I\phi)=0$, where $I$ runs over ordered
$q$-tuples. Its leading term is $w\Delta^q\phi$; every other term
contains a derivative of $\phi$ of order at most $2q-1$.
Since $\phi\in H^{q-1}_{\rm loc}$, these remaining terms belong locally
to $H^{-q}$. The function $w$ is smooth and strictly positive, so
$\Delta^q\phi\in H^{-q}_{\rm loc}$.
For $\zeta\in C_c^\infty$, the operator
$[\Delta^q,\zeta]$ has order at most $2q-1$. Consequently
$\Delta^q(\zeta\phi)\in H^{-q}(\mathbb R^n)$, or, in Fourier variables,
\[
 \int\frac{|\xi|^{4q}}{(1+|\xi|^2)^q}
                   |\widehat{\zeta\phi}(\xi)|^2\,d\xi<\infty.
\]
For $|\xi|\ge1$ the weight is bounded below by a positive constant
times $(1+|\xi|^2)^q$. For $|\xi|\le1$, integrability follows from
$\zeta\phi\in H^{q-1}$. Thus $\phi\in H^q_{\rm loc}$.

Choose $\eta\in C_c^\infty$ with $0\le\eta\le1$ and $\eta=1$ on
$B_1$, and put $\eta_R(x)=\eta(x/R)$. Local Sobolev approximation

\eqref{eq:dense-divergence-weak-equation}. The product rule gives
\[
 \nabla^q(\eta_R^{2q}\phi)
     =\eta_R^{2q}\nabla^q\phi+R_R,
 \qquad
 |R_R|\le C_q\eta_R^q
                   \sum_{j=1}^qR^{-j}|\nabla^{q-j}\phi|.
\]
Indeed, a term with $j\le q$ derivatives on the product of $2q$
copies of $\eta_R$ leaves at least $2q-j\ge q$ undifferentiated
copies, while the differentiated factors contribute $C_qR^{-j}$.

 A_R=\|\eta_R^q\nabla^q\phi\|_2,
 \qquad
 b_R=C_q\sum_{j=1}^qR^{-j}\|\nabla^{q-j}\phi\|_2.

<!-- lemma-5: body_md -->
\label{lem:radius-iteration}
Let $R:\mathcal H\to\mathcal H$ and $L:\mathcal E\to\mathcal H$
be bounded operators. Suppose $a,K,C>0$, $m_0\ge1$, and
\begin{equation}\label{op-hypothesis}
 R^*R\preceq g_a(RR^*+LL^*),\qquad
 g_a(t)=\frac{t}{1+at},\qquad
 \|R^mL\|^2\le CK^{m-1}\quad(m\ge m_0).
\end{equation}
Then $r(R)^2\le K$. Moreover, for every unit vector $f$, putting
$u_m=\|(R^*)^mf\|^2$, one has
\[
 u_{m+1}\ge\frac{u_m^2}{u_{m-1}}-\|R^mL\|^2
 \quad\text{for }m\ge1\text{ whenever }u_m>0.
\]
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
Assume $\mathcal H\ne\{0\}$; the zero space is trivial.
We first prove that the spectral radius $r(R)$ satisfies
\begin{equation}\label{op-radius}
 r(R)^2\le K.
\end{equation}
Complexification preserves the norms and the inequality in
\eqref{op-hypothesis}, so we may work over complex Hilbert spaces. If
$r(R)=0$, there is nothing to prove at this step. Otherwise choose
$\lambda\in\sigma(R^*)$ with $|\lambda|=r(R)>0$, and set
$s=|\lambda|^2$. Such a point exists because the spectrum of a bounded
operator on a nonzero complex Hilbert space is nonempty and compact
\cite[Chapter~III, Section~6.2]{Kato}, and $\sigma(R^*)$ is the
complex conjugate of $\sigma(R)$.

There are unit vectors $v_h$ such that
\begin{equation}\label{op-approximate-eigenvector}
 \|(R^*-\lambda I)v_h\|\longrightarrow0.
\end{equation}
Here is the resolvent argument. Put $z_h=(1+h^{-1})\lambda$.
Since $|z_h|>r(R)$, $R^*-z_hI$ is invertible. The factorization
\[
 R^*-\lambda I=(R^*-z_hI)
 \bigl[I+(z_h-\lambda)(R^*-z_hI)^{-1}\bigr]
\]
shows that
$|z_h-\lambda|\|(R^*-z_hI)^{-1}\|\ge1$; otherwise the second
factor would be invertible by its convergent Neumann series, contrary
to $\lambda\in\sigma(R^*)$. Choose a unit vector $w_h$ for which
\[
 \|(R^*-z_hI)^{-1}w_h\|
 \ge\tfrac12\|(R^*-z_hI)^{-1}\|,
\]
and normalize this resolvent image to obtain $v_h$. It follows that
\[
 \|(R^*-\lambda I)v_h\|
 \le\frac{2}{\|(R^*-z_hI)^{-1}\|}+|z_h-\lambda|
 \longrightarrow0,
\]
which proves \eqref{op-approximate-eigenvector}.

After passing to a subsequence, write
$\|L^*v_h\|^2\to p\ge0$. The approximate eigenvector relation gives
\[
 \|R^*v_h\|^2\to s,
 \qquad
 \liminf_h\|Rv_h\|^2\ge s.
\]
Indeed, the second assertion follows from
$\|Rv_h\|\ge|\langle v_h,Rv_h\rangle|
=|\langle R^*v_h,v_h\rangle|$.
For any bounded positive operator $T$ and unit vector $v$, its spectral
measure at $v$ is a probability measure. The scalar function $g_a$ is
concave, so Jensen's inequality gives
\[
 \langle v,g_a(T)v\rangle\le g_a(\langle v,Tv\rangle).
\]
Apply this with $T=RR^*+LL^*$ and $v=v_h$. By
\eqref{op-hypothesis},
\[
 \|Rv_h\|^2
 \le g_a\bigl(\|R^*v_h\|^2+\|L^*v_h\|^2\bigr).
\]
Taking limits yields $s\le g_a(s+p)<a^{-1}$. Rearranging gives
\begin{equation}\label{op-positive-observation}
 p\ge\frac{as^2}{1-as}>0.
\end{equation}

For each fixed integer $m\ge m_0$,
\[
 (R^*)^m-\lambda^m I
 =\sum_{\ell=0}^{m-1}
 \lambda^{m-1-\ell}(R^*)^\ell(R^*-\lambda I).
\]
Equation \eqref{op-approximate-eigenvector} therefore implies
$L^*(R^*)^mv_h-\lambda^mL^*v_h\to0$. Taking norms and using the
observation bound of degree $m+1$ gives
\[
 s^mp\le\|L^*(R^*)^m\|^2
       =\|R^mL\|^2\le C K^{m-1}.
\]
If $s>K$, this inequality would imply
$p\le(C/K)(K/s)^m\to0$, contradicting
\eqref{op-positive-observation}. Thus $s\le K$, proving
\eqref{op-radius}.


Since $g_a(t)\le t$, the comparison gives
$R^*R\preceq RR^*+LL^*$. If $u_m>0$, then $u_{m-1}>0$ and
Cauchy--Schwarz gives
\begin{align*}
 u_m^2
 &=|\langle (R^*)^{m-1}f,R(R^*)^mf\rangle|^2\\
 &\le u_{m-1}\|R(R^*)^mf\|^2\\
 &\le u_{m-1}\bigl(u_{m+1}+\|L^*(R^*)^mf\|^2\bigr)\\
 &\le u_{m-1}\bigl(u_{m+1}+\|R^mL\|^2\bigr).
\end{align*}
Division by $u_{m-1}$ proves the recurrence.

 r_j=\frac{z_j}{z_{j-1}},\qquad

\]

$r_j>1+\sqrt{q^2-2\varepsilon}$.

 \|h_\varepsilon-h_\delta\|_{L^2(B_R)}
 \le C_R\|f_\varepsilon-f_\delta\|_{L^2(B_R)}.
\]
To verify the use of the mean on $B_1$ rather than $B_R$, put
$h=h_\varepsilon-h_\delta$ and let $h_R$ be its mean on $B_R$.
Since the mean of $h$ on $B_1$ is zero,
$|h_R|\le |B_1|^{-1/2}\|h-h_R\|_{L^2(B_R)}$; applying the ball
Poincar\'e inequality to $h-h_R$ proves the displayed estimate.
Thus $h_\varepsilon$ converges in $W^{1,2}(B_R)$ for every $R$ to a
locally Sobolev primitive $h$ of $f$.

If $f\in L^2(\mu)$, this primitive has an $L^2(\mu)$ representative
modulo constants. Choose a median of $h$ to be zero and set
$h_N=\max(-N,\min(h,N))$. The median of $h_N$ is zero. For a random
variable $Z$ with median zero, $(\mathbb EZ)^2\le\operatorname{Var}Z$:
if $\mathbb EZ\ge0$, then
$\mathbb EZ\le\mathbb E[Z\mathbf1_{\{Z>0\}}]$ and
$\mathbb P(Z>0)\le1/2$, so Cauchy--Schwarz gives
$(\mathbb EZ)^2\le\mathbb EZ^2/2$; the other case follows by replacing
$Z$ with $-Z$. Applying this to $h_N$ and then the scalar Poincar\'e
inequality gives
\[
 \mathbb E h_N^2\le2\operatorname{Var}h_N
 \le2C_P\mathbb E|\nabla h_N|^2
 \le2C_P\mathbb E|f|^2.
\]
Here $h_N$ belongs to $W^{1,2}(\mu)$ because it is bounded and its
weak gradient is bounded by $|f|$.
Fatou's lemma proves $h\in L^2(\mu)$. Subtracting its mean and applying
the scalar inequality once more gives
$\|h-\mathbb Eh\|_2^2\le C_P\|f\|_2^2$.

Apply this construction to $(F_{iI})_i$ for each ordered $r$-tuple $I$,
and choose each primitive $U_I$ to have mean zero. Symmetry of $F$
shows that permuting $I$ does not change the gradient of $U_I$;
uniqueness of a centered primitive therefore makes $U$ symmetric.
Also $\partial_iU_{jI}=F_{ijI}=F_{jiI}=\partial_jU_{iI}$, so $U$ is
compatible. Summing the scalar inequalities proves the tensor bound.
Any two such primitives differ by a constant tensor, whose zero mean
forces it to vanish.

Iterating the construction, every $F\in\mathcal C_s$ can be written
as $F=\nabla^s\phi$ with $\phi\in W^{s,2}(\mu)$ and every lower
primitive centered. Choose a smooth cutoff $\chi_R$, equal to one on
$B_R$, supported in $B_{2R}$, and satisfying
$|\nabla^j\chi_R|\le C_jR^{-j}$.
The product rule and the square integrability of the derivatives of
$\phi$ show that $\chi_R\phi\to\phi$ in $W^{s,2}(\mu)$: the
undifferentiated cutoff terms converge by dominated convergence, and
each remaining term has norm at most
$C_jR^{-j}\|\nabla^{s-j}\phi\|_2$.
Mollify each compactly supported $\chi_R\phi$. Equivalence of weighted
and unweighted Sobolev norms on its compact support gives smooth
compactly supported approximations in all required norms. A diagonal
sequence proves density of $\mathcal K_s$ in $\mathcal C_s$.
For $s=0$, cutoff and mollification directly give density in $L^2(\mu)$.

Centering these approximants gives a dense subset of
$\operatorname{Dom}(D_r)$ in $\mathcal G_r$. Distributional derivatives
are closed in $L^2(\mu)$, so $D_r$ is closed. The primitive construction
proves surjectivity, injectivity, and the stated bounded inverse.

\label{lem:static-regular-approximation}
Let $\mu$ be a full-dimensional isotropic log-concave probability measure on $\mathbb R^n$. There are isotropic probability measures $\mu_\varepsilon$, $0<\varepsilon\le1$, with smooth potentials $\widetilde W_\varepsilon$ such that
\[
 0<a_\varepsilon I\preceq D^2\widetilde W_\varepsilon
       \preceq a_\varepsilon^+I<\infty.
\]
Every derivative of $\widetilde W_\varepsilon$ has polynomial growth. Moreover, $\mu_\varepsilon$ converges weakly to $\mu$, and
\[
 \int p\,d\mu_\varepsilon\longrightarrow\int p\,d\mu
\]
for every polynomial $p$. The construction uses only minimization, convolution of the potential with a compactly supported smooth kernel, and affine changes of variables.
<!-- end lemma-10 -->
<!-- proof-12: body_md -->
Write $d\mu=Z^{-1}e^{-V(x)}\,dx$, where $V:\mathbb R^n\to(-\infty,\infty]$ is proper, lower semicontinuous, and convex, and $0<Z<\infty$. The domain of $V$ has nonempty interior because $\mu$ is full-dimensional. We may use the lower semicontinuous representative: a convex potential is continuous on the interior of its domain, and changing its boundary values does not change the measure. A full-dimensional convex set has boundary of Lebesgue measure zero. Indeed, after intersecting its closure with a sufficiently large closed ball, contraction toward an interior point by a factor $1-\delta$ sends this bounded convex set into its interior. The boundary therefore has measure at most $[1-(1-\delta)^n]$ times the volume of that bounded set. Letting $\delta$ decrease to zero proves this for the bounded intersection. The portion of the original boundary inside the ball is contained in the boundary of the intersection. Increasing the ball proves the assertion.

We first establish a common tail bound for the approximants. Choose a finite level $L$ for which $\{V\le L\}$ contains a ball. Such a level exists: choose a cube whose vertices belong to the interior of the domain, and use convexity to bound $V$ on that cube by the largest of its values at the vertices. The sublevel set $K_L=\{V\le L\}$ is closed and satisfies
\[
 |K_L|\le e^L\int e^{-V(x)}\,dx=e^LZ<\infty.
\]
It is bounded. To justify this last statement, if a convex set containing a ball of radius $r>0$ also contains points arbitrarily far from the center of that ball, it contains cones with a fixed $(n-1)$-dimensional base and arbitrarily large height. The volumes of these cones tend to infinity. This contradicts $|K_L|<\infty$; in dimension one, the same argument is the assertion that an unbounded interval has infinite length.

Lower semicontinuity on the compact set $K_L$ gives a minimizer $x_*$ of $V$ there, with finite value $m=V(x_*)$. It is a global minimizer because $V>L$ outside $K_L$. The sublevel set $K_{m+1}$ is also bounded: it is contained in a sublevel set containing a ball, and the preceding finite-volume argument applies to that larger set. Choose $R>0$ so that $K_{m+1}$ is contained in the open ball $B(x_*,R)$. If $r=|x-x_*|\ge R$ and $V(x)<\infty$, convexity at $x_*+(R/r)(x-x_*)$ gives
\[
 m+1<V\!\left(x_*+\frac Rr(x-x_*)\right)
 \le m+\frac Rr\bigl(V(x)-m\bigr).
\]
Thus $V(x)>m+r/R$. For $r<R$, the inequality $V(x)\ge m$ gives $V(x)\ge m+r/R-1$. The same lower bounds hold when $V(x)=\infty$. Consequently there are constants $c>0$ and $C<\infty$ such that
\begin{equation}\label{eq:static-linear-tail}
 V(x)\ge c|x|-C\qquad(x\in\mathbb R^n).
\end{equation}
For example, one may take $c=1/R$ and $C=1-m+c|x_*|$.

For $\varepsilon>0$, define the Moreau envelope
\[
 V^\varepsilon(x)=\inf_{y\in\mathbb R^n}
       \left\{V(y)+\frac{|x-y|^2}{2\varepsilon}\right\}.
\]
The minimized function is lower semicontinuous and coercive by \eqref{eq:static-linear-tail}, and it is strictly convex on its domain because of its quadratic term. It therefore has a unique minimizer, denoted $p_\varepsilon(x)$. Set
\[
 g_\varepsilon(x)=\frac{x-p_\varepsilon(x)}{\varepsilon}.
\]
For completeness, we verify the differentiability and Hessian bounds needed below. Put $p=p_\varepsilon(x)$. For $z$ with $V(z)<\infty$ and $0<s<1$, minimality at $p$ and convexity of $V$ imply
\[
 0\le s\bigl(V(z)-V(p)\bigr)
      -\frac{s}{\varepsilon}\langle x-p,z-p\rangle
      +\frac{s^2}{2\varepsilon}|z-p|^2.
\]
Divide by $s$ and let $s$ decrease to zero. The result is
\begin{equation}\label{eq:static-subgradient}
 V(z)\ge V(p)+\langle g_\varepsilon(x),z-p\rangle.
\end{equation}
It also holds when $V(z)=\infty$. Apply this inequality at $x,x'$ with $z=p_\varepsilon(x'),p_\varepsilon(x)$, respectively, and add. Writing $p'=p_\varepsilon(x')$ and $g'=g_\varepsilon(x')$ gives
\[
 \langle g_\varepsilon(x)-g',p-p'\rangle\ge0.
\]
Since $x-x'=p-p'+\varepsilon(g_\varepsilon(x)-g')$, it follows that
\[
 |x-x'|^2
 =|p-p'|^2+2\varepsilon\langle p-p',g_\varepsilon(x)-g'\rangle
       +\varepsilon^2|g_\varepsilon(x)-g'|^2
 \ge\varepsilon^2|g_\varepsilon(x)-g'|^2.
\]
Thus $g_\varepsilon$ is $\varepsilon^{-1}$-Lipschitz.

Using $p_\varepsilon(x)$ as a competitor in the minimization at $x+h$ gives
\[
 V^\varepsilon(x+h)-V^\varepsilon(x)
 \le\langle g_\varepsilon(x),h\rangle+\frac{|h|^2}{2\varepsilon}.
\]
Using $p_\varepsilon(x+h)$ as a competitor at $x$ gives the reverse bound
\[
 V^\varepsilon(x+h)-V^\varepsilon(x)
 \ge\langle g_\varepsilon(x+h),h\rangle-\frac{|h|^2}{2\varepsilon}.
\]
The Lipschitz estimate for $g_\varepsilon$ shows that the difference between either bound and $\langle g_\varepsilon(x),h\rangle$ is $O(|h|^2/\varepsilon)$. Hence $V^\varepsilon$ is continuously differentiable and $\nabla V^\varepsilon=g_\varepsilon$. It is convex: for $0\le s\le1$, insert $(1-s)p_\varepsilon(x)+sp_\varepsilon(x')$ as a competitor at $(1-s)x+sx'$ and apply convexity to both $V$ and the squared norm. Convexity together with the first of the preceding bounds yields
\begin{equation}\label{eq:static-envelope-remainder}
 0\le V^\varepsilon(x+h)-V^\varepsilon(x)
          -\langle\nabla V^\varepsilon(x),h\rangle
 \le\frac{|h|^2}{2\varepsilon}.
\end{equation}

Choose a nonnegative, even function $\chi\in C_c^\infty(\mathbb R^n)$, supported in the unit ball, with $\int\chi=1$. Put $\chi_\varepsilon(y)=\varepsilon^{-n}\chi(y/\varepsilon)$ and define
\[
 W_\varepsilon(x)
   =(V^\varepsilon*\chi_\varepsilon)(x)+\frac\varepsilon2|x|^2.
\]
Convolution with this smooth compactly supported kernel makes $W_\varepsilon$ smooth. Adding \eqref{eq:static-envelope-remainder} at $h$ and $-h$ gives
\[
 0\le V^\varepsilon(x+h)+V^\varepsilon(x-h)-2V^\varepsilon(x)
       \le\frac{|h|^2}{\varepsilon}.
\]
Integrate against $\chi_\varepsilon$, set $h=sv$ for a unit vector $v$, divide by $s^2$, and let $s$ tend to zero. We obtain
\begin{equation}\label{eq:static-hessian-bounds}
 \varepsilon I\preceq D^2W_\varepsilon(x)
       \preceq(\varepsilon^{-1}+\varepsilon)I.
\end{equation}

We also record the derivative growth explicitly. Since $\nabla V^\varepsilon$ is Lipschitz,
\[
 |\nabla V^\varepsilon(x)|
       \le|\nabla V^\varepsilon(0)|+\varepsilon^{-1}|x|.
\]
Its convolution has the same linear-growth property. Thus $\nabla W_\varepsilon$ has linear growth, and $W_\varepsilon$ has at most quadratic growth. For a multi-index $\alpha$ with $|\alpha|\ge2$, the kernel identities
\[
 \int D^\alpha\chi_\varepsilon(y)\,dy=0,
 \qquad
 \int y_iD^\alpha\chi_\varepsilon(y)\,dy=0
       \quad(1\le i\le n)
\]
follow by integration by parts, using compact support. Therefore
\[
 \begin{split}
 D^\alpha(V^\varepsilon*\chi_\varepsilon)(x)
 =\int\bigl[V^\varepsilon(x-y)-V^\varepsilon(x)
       +\langle\nabla V^\varepsilon(x),y\rangle\bigr]
        D^\alpha\chi_\varepsilon(y)\,dy.
 \end{split}
\]
Equation \eqref{eq:static-envelope-remainder} bounds its absolute value by
\[
 \frac1{2\varepsilon}\int |y|^2
                 |D^\alpha\chi_\varepsilon(y)|\,dy<\infty,
\]
uniformly in $x$. The added quadratic has constant second derivatives and zero derivatives of order at least three. This proves the required polynomial growth of every derivative.

We next prove convergence and control the tails uniformly. For every $x$, $V^\varepsilon(x)$ tends to $V(x)$ as $\varepsilon$ decreases to zero, with the value $\infty$ allowed. Here are the details. The global lower bound $V\ge m$ and the minimizing identity give
\[
 \frac{|x-p_\varepsilon(x)|^2}{2\varepsilon}
       \le V^\varepsilon(x)-m.
\]
If $V(x)<\infty$, the competitor $y=x$ gives $V^\varepsilon(x)\le V(x)$, so $p_\varepsilon(x)\to x$. Lower semicontinuity and $V^\varepsilon(x)\ge V(p_\varepsilon(x))$ then give
\[
 V(x)\le\liminf_{\varepsilon\downarrow0}V^\varepsilon(x)
       \le\limsup_{\varepsilon\downarrow0}V^\varepsilon(x)\le V(x).
\]
If $V(x)=\infty$ but $V^\varepsilon(x)$ were bounded along a sequence decreasing to zero, the same minimizing identity would give $p_\varepsilon(x)\to x$ and bounded values of $V(p_\varepsilon(x))$, contradicting lower semicontinuity.

Because $\chi_\varepsilon$ has mean zero, convexity gives
\[
 (V^\varepsilon*\chi_\varepsilon)(x)\ge V^\varepsilon(x).
\]
At an interior point of the domain, the opposite estimate
\[
 (V^\varepsilon*\chi_\varepsilon)(x)
       \le\sup_{|y|\le\varepsilon}V(x-y)
\]
follows from $V^\varepsilon\le V$. Continuity of a finite convex function on the interior of its domain implies that the right side tends to $V(x)$. Consequently $W_\varepsilon(x)\to V(x)$ there. At every point where $V(x)=\infty$, the lower estimate gives $W_\varepsilon(x)\to\infty$. Since the boundary of the domain has measure zero, these statements prove
\[
 e^{-W_\varepsilon(x)}\longrightarrow e^{-V(x)}
       \quad\text{for Lebesgue-almost every }x.
\]

From \eqref{eq:static-linear-tail} and the triangle inequality,
\[
 \begin{split}
 V^\varepsilon(x)
 &\ge c|x|-C+\inf_{y\in\mathbb R^n}
       \left\{\frac{|x-y|^2}{2\varepsilon}-c|x-y|\right\}\\
 &=c|x|-C-\frac{\varepsilon c^2}{2}.
 \end{split}
\]
The support of $\chi_\varepsilon$ is contained in $B(0,\varepsilon)$, so for $0<\varepsilon\le1$,
\[
 W_\varepsilon(x)
       \ge c|x|-C-c\varepsilon-\frac{\varepsilon c^2}{2}
       \ge c|x|-C-c-\frac{c^2}{2}.
\]
Thus $e^{-W_\varepsilon(x)}$ is bounded by a constant, independent of $\varepsilon\le1$, times $e^{-c|x|}$. Every polynomial is integrable against this majorant. Dominated convergence proves
\[
 Z_\varepsilon:=\int e^{-W_\varepsilon(x)}\,dx\longrightarrow Z,
 \qquad
 Z_\varepsilon^{-1}\int p(x)e^{-W_\varepsilon(x)}\,dx
       \longrightarrow\int p\,d\mu
\]
for every polynomial $p$. Applying the same argument to bounded continuous functions proves weak convergence of the probabilities $\widehat\mu_\varepsilon$ with densities $Z_\varepsilon^{-1}e^{-W_\varepsilon}$.

Let $m_\varepsilon$ and $A_\varepsilon$ be the mean and covariance of $\widehat\mu_\varepsilon$. The moment convergence gives $m_\varepsilon\to0$ and $A_\varepsilon\to I$. Each $A_\varepsilon$ is positive definite because $\widehat\mu_\varepsilon$ has a positive density on all of $\mathbb R^n$. Define $\mu_\varepsilon$ to be the law of
\[
 Y_\varepsilon=A_\varepsilon^{-1/2}(X_\varepsilon-m_\varepsilon),
       \qquad X_\varepsilon\sim\widehat\mu_\varepsilon.
\]
It is centered and has covariance $I$. Its potential, up to an additive normalizing constant, is
\[
 \widetilde W_\varepsilon(y)
       =W_\varepsilon(m_\varepsilon+A_\varepsilon^{1/2}y).
\]
The chain rule and \eqref{eq:static-hessian-bounds} give
\[
 \varepsilon\lambda_{\min}(A_\varepsilon)I
       \preceq D^2\widetilde W_\varepsilon(y)
       \preceq(\varepsilon^{-1}+\varepsilon)
                    \lambda_{\max}(A_\varepsilon)I.
\]
Affine substitution preserves the polynomial growth of its derivatives.

Finally, $A_\varepsilon^{-1/2}\to I$ by the spectral theorem and continuity of $s\mapsto s^{-1/2}$ near $1$. Expanding a polynomial $p(A_\varepsilon^{-1/2}(x-m_\varepsilon))$ gives a finite sum of monomials whose coefficients converge to those of $p(x)$. The moment convergence for $\widehat\mu_\varepsilon$ therefore proves the asserted moment convergence for $\mu_\varepsilon$. If $f$ is bounded and continuous, then $f(A_\varepsilon^{-1/2}(x-m_\varepsilon))\to f(x)$ pointwise. The common exponential majorant just proved, together with $Z_\varepsilon\to Z>0$, allows another application of dominated convergence. Hence $\mu_\varepsilon$ converges weakly to $\mu$ as well.

\label{lem:static-finite-energy}\label{finite-energy-extension}
Let $\mu$ be an absolutely continuous probability measure on $\mathbb R^n$, and let $0\le C<\infty$. Assume
\begin{equation}\label{eq:static-smooth-poincare}
 \int u^2\,d\mu-\left(\int u\,d\mu\right)^2
       \le C\int|\nabla u|^2\,d\mu
       \qquad\bigl(u\in C_c^\infty(\mathbb R^n)\bigr).
\end{equation}
If $f:\mathbb R^n\to\mathbb R$ is locally Lipschitz and
\[
 E:=\int|\nabla f|^2\,d\mu<\infty,
\]
then $f\in L^2(\mu)$ and
\begin{equation}\label{eq:static-full-poincare}
 \int\left|f-\int f\,d\mu\right|^2\,d\mu
       \le C\int|\nabla f|^2\,d\mu.
\end{equation}
Here local Lipschitz regularity is required on the ambient space $\mathbb R^n$. For a measure on a smaller affine hull, the same statement applies in coordinates on that affine hull.
If the density of $\mu$ is smooth and strictly positive on $\mathbb R^n$, the conclusion also holds for every $f\in W_{\mathrm{loc}}^{1,2}(\mathbb R^n)$ whose weak gradient has finite $L^2(\mu)$ norm.

We first extend \eqref{eq:static-smooth-poincare} to compactly supported Lipschitz functions. Let $u:\mathbb R^n\to\mathbb R$ have compact support and Lipschitz constant $L$. Use the nonnegative smooth kernel $\chi_\varepsilon$ from the proof of Lemma~\ref{lem:static-regular-approximation}, and put $u_\varepsilon=u*\chi_\varepsilon$. Then $u_\varepsilon\in C_c^\infty(\mathbb R^n)$ and
\[
 |u_\varepsilon(x)-u(x)|
       \le \int L|y|\chi_\varepsilon(y)\,dy\le L\varepsilon.
\]
A Lipschitz function has a weak gradient agreeing with its pointwise gradient almost everywhere, with $|\nabla u|\le L$. Differentiating convolution in the weak sense gives
\[
 \nabla u_\varepsilon=(\nabla u)*\chi_\varepsilon,
       \qquad |\nabla u_\varepsilon|\le L.
\]
The Lebesgue differentiation theorem implies $\nabla u_\varepsilon\to\nabla u$ at Lebesgue-almost every point. Absolute continuity of $\mu$ gives the same convergence $\mu$-almost everywhere. The bound $|\nabla u_\varepsilon|\le L$, and the fact that $\mu$ is a probability measure, permit dominated convergence in the energy integral. Uniform convergence of $u_\varepsilon$ gives convergence of its first and second moments. Applying \eqref{eq:static-smooth-poincare} to $u_\varepsilon$ and taking the limit proves it for $u$.

This argument does not require a lower bound for the density of $\mu$ near the boundary of its support. The function being convolved is defined on all of $\mathbb R^n$; its gradients are bounded independently of $\varepsilon$, and convergence is first established Lebesgue-almost everywhere.

Next, let $h:\mathbb R^n\to\mathbb R$ be bounded and locally Lipschitz, with $\nabla h\in L^2(\mu)$. Choose $\eta\in C_c^\infty(\mathbb R^n)$ with $0\le\eta\le1$, equal to one on the unit ball and zero outside the ball of radius two. Set $\eta_R(x)=\eta(x/R)$ and  $ h_R=\eta_R h.$ The function $h_R$ is compactly supported and Lipschitz. To justify the latter assertion from local Lipschitz regularity, cover the compact ball $\overline B(0,3R)$ by finitely many neighborhoods on which $h$ is Lipschitz. A Lebesgue number for this cover allows each line segment in the ball to be partitioned into pieces lying in one neighborhood. Summing the local estimates bounds $|h(x)-h(y)|$ by the largest of the finitely many Lipschitz constants times $|x-y|$. Multiplication by the smooth cutoff preserves a Lipschitz bound on this ball, and the product vanishes outside $B(0,2R)$, which gives a global bound.

Dominated convergence gives $h_R\to h$ in $L^2(\mu)$. Moreover, $\nabla h_R-\nabla h=(\eta_R-1)\nabla h+h\nabla\eta_R.$ The first term tends to zero in $L^2(\mu)$ by dominated convergence. The second satisfies
\[
 \|h\nabla\eta_R\|_{L^2(\mu)}
       \le \frac{\|h\|_\infty\|\nabla\eta\|_\infty}{R}
       \longrightarrow0.
\]
We may therefore pass to the limit in the compact-support inequality for $h_R$, obtaining \eqref{eq:static-full-poincare} for every such bounded $h$.

It remains to prove integrability of a finite-energy function $f$. Choose a finite median $m$ of $f$, so that
\[
 \mu(f\le m)\ge\tfrac12,\qquad \mu(f\ge m)\ge\tfrac12.
\]
A finite median exists because $f$ is real-valued: it can be defined as the infimum of the real numbers $s$ for which $\mu(f\le s)\ge1/2$. For $N>0$, let
\[
 f_N=\max\{-N,\min\{f-m,N\}\}.
\]
These functions are bounded and locally Lipschitz, and their gradients satisfy $|\nabla f_N|\le|\nabla f|$ almost everywhere. The gradient bound follows from the fact that the clipping map is $1$-Lipschitz: at every point where $f$ and $f_N$ are differentiable, take difference quotients in each direction and use this Lipschitz bound. Both functions are differentiable almost everywhere. The bounded-function case consequently gives
\[
 \int|f_N-a_N|^2\,d\mu\le CE,\qquad
       a_N=\int f_N\,d\mu.
\]
The number zero is a median of $f_N$. If $a_N\ge0$, then on the set $\{f_N\le0\}$ one has $|f_N-a_N|\ge a_N$, and that set has probability at least $1/2$. If $a_N<0$, use instead the set $\{f_N\ge0\}$. In either case,
\[
 \frac12|a_N|^2\le\int|f_N-a_N|^2\,d\mu\le CE.
\]
It follows that
\[
 \int f_N^2\,d\mu
       =\int|f_N-a_N|^2\,d\mu+|a_N|^2\le3CE.
\]
Since $f_N\to f-m$ pointwise, Fatou's lemma gives $f-m\in L^2(\mu)$, hence $f\in L^2(\mu)$. Finally, $|f_N|\le|f-m|$ and $|f_N-(f-m)|\le|f-m|$, so dominated convergence gives $f_N\to f-m$ in $L^2(\mu)$. Their variances therefore converge to the variance of $f$. Taking the limit in their variance bound proves \eqref{eq:static-full-poincare}.

We finish with the Sobolev assertion. Write $d\mu=\rho\,dx$, with $\rho$ smooth and strictly positive. On each fixed compact set $K$, both $\rho$ and $\rho^{-1}$ are bounded, so the weighted and unweighted $L^2$ norms are equivalent there. If $u$ is compactly supported and belongs to $W_{\mathrm{loc}}^{1,2}(\mathbb R^n)$, its mollifications converge to $u$ in unweighted $W^{1,2}(\mathbb R^n)$ and have support in a fixed enlarged compact set. To see the convergence directly, apply
\[
 \|v*\chi_\varepsilon-v\|_{L^2(dx)}
 \le\int\chi_\varepsilon(y)
               \|v(\,\cdot-y)-v\|_{L^2(dx)}\,dy
       \longrightarrow0
\]
to $v=u$ and to each weak derivative of $u$, using continuity of translations in $L^2(dx)$. The bounds on $\rho$ then give convergence in the weighted function and gradient norms. This proves \eqref{eq:static-smooth-poincare} for compactly supported Sobolev functions. Bounded Sobolev functions are treated by the same cutoffs $h_R=\eta_Rh$ and the same two energy estimates as above. Finally, Sobolev clipping preserves local $W^{1,2}$ regularity and satisfies $|\nabla f_N|\le|\nabla f|$ almost everywhere. One can obtain this bound by replacing the clipping map by smooth maps with derivatives bounded by one: the chain rule for a smooth map follows by mollifying $f$ on compact sets, and weak lower semicontinuity of the gradient norm on each measurable subset gives the stated bound in the clipping limit. The median, Fatou, and $L^2$ convergence arguments above consequently apply without change and prove the Sobolev assertion.
