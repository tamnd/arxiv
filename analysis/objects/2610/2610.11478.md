<!-- theorem-1: body_md -->
\label{thm:MainA} 
Let $({\mathit{M}},\mathsf{d},\mathfrak{m})$ be a locally compact length metric measure space with $\mathfrak{m}(M)<\infty$ which is infinitesimally Hilbertian. Assume the Cheeger energy ${\sf Ch}$ admits a heat kernel and the induced intrinsic distance coincides with $\mathsf{d}$. Moreover, let $({\mathit{M}},\mathsf{d},\mathfrak{m})$ satisfy the bounded interpolation property. Lastly, assume there exist a Borel measure $\kappa$ and $n\in [1,\infty)$ such that $({\mathit{M}},\mathsf{d},\mathfrak{m})$ satisfies ${\mathrm{BE}}(-\kappa,n)$ and there exist $T>0$ and $\delta \in [0, 1\wedge \frac{1}{2|n-2|})$ such that
\begin{equation*}
\mathrm{k}_T:=\mathrm{k}_T({\mathit{M}},\mathsf{d},\mathfrak{m},\kappa) \le \delta \, \cdot 
\end{equation*}
Then there exist constants $c=c(\delta,n)\geq 1$ and $N = N(\delta,n) \geq n$ such that $(M,\mathsf{d},\mathfrak{m})$ satisfies a doubling property and an $L^2$-Poincaré inequality at scale $(T/(2\delta N))\wedge (\operatorname{diam}({\mathit{M}})/c)$ and dimension $N$.

\label{th:MainB}
Consider $n \in [1,+\infty)$, $T \in (0,+\infty)$ and  $\delta \in [0, 1\wedge \frac{1}{2|n-2|})$.

\begin{enumerate}
    \item Let $\mathcal{M}(n,T,\delta)$ be the collection of isometry classes of pointed metric measures spaces $(M,\mathsf{d},\mathfrak{m},o)$ satisfying the assumptions of Theorem \ref{thm:MainA}. Then $\mathcal{M}(n,T,\delta)$ is precompact in the pointed measured Gromov--Hausdorff topology.

\label{Def:BIP}
    We say that $(M,\mathsf{d},\mathfrak{m})$ satisfies the bounded interpolation property (or is BIP, for brevity) if for any $q \in (1,+\infty)$ there exists $C_q : (0,+\infty) \to [1,+\infty)$ such that for any $\mu_0,\mu_1 \in \mathcal{P}(M)$ that are absolutely continuous with respect to $\mathfrak{m}$ with bounded densities $\rho_0,\rho_1$, if $\mathrm{diam}(\mathrm{supp}(\mu_0) \cup \mathrm{supp}(\mu_1) ) < D$ for some $D>0$, then there exists $\pi \in \mathrm{OptGeo}_q(\mu_0,\mu_1)$ such that for any $t \in [0,1]$,
    \[
    (e_t)_\# \pi  \ll \mathfrak{m} \quad \text{with} \quad \left\| \rho_t \right\|_\infty \le C_q(D)   \max \left( \left\| \rho_0\right\|_\infty,\left\| \rho_1 \right\|_\infty \right)
    \]
    where $\rho_t$ is the density of $(e_t)_\# \pi$ with respect to $\mathfrak{m}$.

For  an infinitesimally Hilbertian  space $ (M,\mathsf{d},\mathfrak{m})$, $\kappa \in \mathcal{M}_0(M,\mathsf{d},\mathfrak{m})$ and $t>0$, set

<!-- definition-5: body_md -->
\label{Def:BE}
Let $ (M,\mathsf{d},\mathfrak{m})$ be an infinitesimally Hilbertian space. For any $\kappa \in \mathcal{M}_0(M,\mathsf{d},\mathfrak{m})$  and $n \in [1,+\infty]$, we say that $(M,\mathsf{d},\mathfrak{m})$ satisfies $\mathrm{BE}(-\kappa,n)$ if
\begin{equation*}
\Gamma_2[f,\phi] \ge \frac{1}{n} \int_M \phi \,  (\Delta f)^2 \,\mathrm{d} \mathfrak{m} - \int_M \phi  \Gamma(f) \,\mathrm{d} \kappa,
\end{equation*}
for every $f \in \mathscr{D}(\Delta)$ with $\Delta f \in W^{1,2}(M,\mathsf{d},\mathfrak{m})$, and for every $\mathfrak{m}$-a.e.~non-negative $\phi \in \mathscr{D}(\Delta^{2\kappa})$ with $\phi,\Delta^{2\kappa} \phi \in L^{\infty}(M,\mathfrak{m})$, where
\begin{align*}
    \Gamma_2[f,\phi] := -\frac{1}{2}\int_{\mathit{M}}\Gamma(\Gamma(f),\phi)\,\mathrm{d}\mathfrak{m} - \int_{\mathit{M}}\Gamma(f,\Delta f)\,\phi\,\mathrm{d}\mathfrak{m}.
\end{align*}

\begin{remark}
    If $\kappa$ is absolutely continuous with respect to $\mathfrak{m}$ with Radon--Nikodym derivative $\mathfrak{m}$-a.e.~equal to a constant $K \in \mathbb{R}$, then the inequality defining ${\mathrm{BE}}(-\kappa,n)$ rewrites as the classical Bakry--\'Emery curvature-dimension condition \begin{equation*}
\Gamma_2[f,\phi] \ge  \int_M \phi  \left( \frac{(\Delta f)^2}{n}- K\Gamma(f)\right)\,\mathrm{d} \mathfrak{m}.
\end{equation*}
\end{remark}

\label{lem:HKtimechange}
    Let $(M,\mathsf{d},\mathfrak{m})$ be an infinitesimally Hilbertian length space admitting a heat kernel $p$. Consider $w \in L^\infty({\mathit{M}},\mathfrak{m})$  continuous $\mathfrak{m}$-a.e.~on $M$. Then $(M,\mathsf{d}_w,\mathfrak{m}_w)$ admits a heat kernel $p^w$ as well.
<!-- end lemma-3 -->
<!-- proof-1: body_md -->
By \cite[Lemma A.3.10]{CF}, the resolvent of the time-changed Dirichlet form is given by
\begin{align*}
    R_\alpha f(x) = \mathbf{E}^x\left[ \int_0^\infty e^{-\alpha A_t}\,f(X_t)\,\mathrm{d} A_t \right].
\end{align*}
Since the positive continuous additive functional is given by $A_t = \int_0^t w(X_s)\,\mathrm{d} s$, the previous term writes as
\begin{align*}
    R_\alpha f(x) = \mathbf{E}^x \left[\int_0^\infty e^{-\alpha \int_0^t w(X_s)\,\mathrm{d} s} f(X_t)\,w(X_t)\,\mathrm{d} t \right].
\end{align*}
Now, define a set function $B\mapsto R_\alpha 1_B(x)$. Let $B$ with $\mathfrak{m}_w[B]=0$ (hence $\mathfrak{m}[B]=0$). Then
\begin{align*}
    R_\alpha 1_B(x) = \mathbf{E}^x\left[\int_0^\infty e^{-\alpha\int_0^t w(X_s)\,\mathrm{d} s} 1_B(X_t)\,w(X_t)\,\mathrm{d} t\right] & \leq C \mathbf{E}^x\left[\int_0^\infty e^{\alpha Ct}\,1_B(X_t)\,\mathrm{d} t\right] \\
    & = C\int_0^\infty e^{\alpha Ct}\,\mathsf{P}_t1_B(x)\,\mathrm{d} t=0
\end{align*}
because
\begin{align*}
    \mathsf{P}_t1_B(x) = \int_Bp_t(x,y)\,\mathrm{d} \mathfrak{m}(y) = 0.
\end{align*}
This means that the resolvent is absolutely continuous with respect to $\mathfrak{m}_w$, and therefore, the heat kernel  $p^w$ exists.

<!-- lemma-4: body_md -->
\label{lem:gaugefunction} 
    Let $(M,\mathsf{d},\mathfrak{m})$ be an infinitesimally Hilbertian length space satisfying $\mathfrak{m}(M)<+\infty$. For $T>0$ and $n \in [1,+\infty)$, assume there exists $\kappa \in \mathscr{K}^-_{T,c(n)}(M,\mathsf{d},\mathfrak{m})$ such that $(M,\mathsf{d},\mathfrak{m})$ satisfies $\mathrm{BE}(-\kappa,n)$. Suppose there exists $\beta>0$ such that
\begin{align*}
    \mathrm{k}_T(M,\mathsf{d},\mathfrak{m},\kappa) \leq 1-\mathrm{e}^{-\beta T} < c(n).
\end{align*}
Then there exists an $\mathfrak{m}$-a.e.~continuous function $v \in \mathscr{D}(\Delta^{\kappa})$ satisfying $\smash{1\leq v\leq 2\,\mathrm{e}^{\beta T}}$ $\mathfrak{m}$-a.e.~and
\begin{align*}
    -\Delta^{\kappa}v +2\,\beta\,v = 2\,\beta \qquad \text{$\mathfrak{m}$-a.e.}
\end{align*}
<!-- end lemma-4 -->
<!-- proof-2: body_md -->
Set
\[
v := 2\beta \int_0^\infty \mathrm{e}^{-2\beta t}\,\mathsf{P}_t^{\kappa}1_{\mathit{M}}\,\mathrm{d} t.
\]
Since $\mathfrak{m}({\mathit{M}})<+\infty$, the heat semigroup $(\mathsf{P}_t)_{t\geq 0}$ is conservative, i.e., $\mathsf{P}_t1_{\mathit{M}}=1_{\mathit{M}}$, $t\geq0$.
By the Feynman-Kac formula, we have $$\mathsf{P}_t^{\kappa}1_{\mathit{M}}\geq \mathsf{P}_t1_{\mathit{M}}=1_{\mathit{M}}$$ for any $t>0$, so that $v\geq1$ $\mathfrak{m}$-a.e. In order to estimate the supremum norm, note that the Myadera-Voigt perturbation theorem \cite{Voigt-77} implies 
\[
\|\mathsf{P}_t^{\kappa}\|_{\infty,\infty}\leq \mathrm{e}^{\beta (T+t)}, \qquad t>0,
\]
where $\|A\|_{\infty,\infty}$ denotes the norm of the bounded operator $A$ in $L^\infty({\mathit{M}},\mathfrak{m})$.
Hence, we obtain
\begin{align*}
\|v\|_\infty
&\leq 2\beta\int_0^\infty e^{-2\beta t}\| \mathsf{P}_t^{\kappa}1_{\mathit{M}}\|_\infty \,\mathrm{d} t
\\
&\leq 2\beta\int_0^\infty \mathrm{e}^{-2\beta t}\| \mathsf{P}_t^{\kappa}\|_{\infty,\infty} \,\mathrm{d} t 
\\
&\leq 
2\beta \int_0^\infty \mathrm{e}^{\beta(T- t)} \,\mathrm{d} t
\\
&=2\mathrm{e}^{\beta T}.
\end{align*}
Since $\mathfrak{m}(X)<\infty$, we have $1_{\mathit{M}}\in L^2(X,\mathfrak{m})$ and therefore $v\in\mathscr{D}(\Delta^{\kappa})\subset W^{1,2}(M,\mathsf{d},\mathfrak{m})$, where the last inclusion follows since $\kappa$ is in the extended Kato class. Hence, $v$ solves the desired equation $\mathfrak{m}$-almost everywhere. By \cite[Theorem~2.1.3]{FukushimaYoshidaTakeda}, $v \in W^{1,2}(M,\mathsf{d},\mathfrak{m})$ possesses a quasi-continuous modification, which therefore is $\mathfrak{m}$-almost everywhere continuous.

Set $\mathrm{k}:=\mathrm{k}_T(M,\mathsf{d},\mathfrak{m},\kappa)$. If $\mathrm{k}=0$, then $({\mathit{M}},\mathsf{d},\mathfrak{m})$ satisfies the Bakry--\'Emery curvature-dimension condition $\mathrm{BE}(0,n)$, and the statement is trivial. Thus, we assume that $\mathrm{k}>0$.
Set \[
\hat N \coloneqq \frac{(n-2)(N-2)}{N-n},\quad
\lambda \coloneqq \begin{cases}
    \frac{1}{2\mathrm{k}}& \colon n= 2,\\
    \hat N&\colon n\neq 2,
\end{cases}
\quad\text{and}\quad \beta \coloneqq  \frac{1}T\ln\left(\frac1{1-\lambda\mathrm{k}} \right)>0.
\]
As the definition of $\beta$ implies $
\mathrm{k}= (1-\mathrm{e}^{-\beta T})/\lambda$, we have 
\[
\mathrm{k}_T(M,\mathsf{d},\mathfrak{m},\lambda\kappa)=\lambda\mathrm{k}= 1-\mathrm{e}^{-\beta T}.
\]
By Lemma~\ref{lem:gaugefunction}, there exists a quasi-continuous $v\in \mathscr{D}(\Delta^{\lambda \kappa})\subset W^{1,2}(M,\mathsf{d},\mathfrak{m})$ such that 
$\smash{1\leq v\leq 2\,\mathrm{e}^{\beta T}}$ and $- \Delta^{\lambda\kappa}v +2\,\beta\,v = 2\,\beta$ $\mathfrak{m}$-a.e. In particular, we have $\Gamma(v)\in L^1(M,\mathfrak{m})$. We set $$w:=\frac{1}{\lambda}\ln v.$$ 
Then $0\leq w\leq \frac{\ln(2)+\beta T}{\lambda}$ $\mathfrak{m}$-a.e. Therefore, Proposition~\ref{thm:timechange} yields that, for every $f\in\mathrm{Test}_\infty({\mathit{M}})$ and every $\mathfrak{m}$-a.e.~non-negative $\phi\in\mathrm{Test}_c({\mathit{M}})$,
\begin{align*}
    \Gamma_2^w[f,\phi]  \geq & - \hat{N}\int_{\mathit{M}} \,\Gamma(w)\,\Gamma^w(f)\,\phi\,\mathrm{d}\mathfrak{m} + \int_{\mathit{M}}\Gamma(w,\Gamma^w(f)\,\phi)\,\mathrm{d}\mathfrak{m} \\
     & - \int_{\mathit{M}}\Gamma^w(f)\,\phi\,\mathrm{d} \kappa + \frac{1}{N}\int_{\mathit{M}}(\Delta^wf)^2\,\phi\,\mathrm{d}\mathfrak{m}_w.
\end{align*}
In order to obtain the desired Bakry-\'Emery curvature-dimension condition, we are left with bounding from below the sum of the first three integrals in the previous right-hand side.

Expanding the first integral using the definition of $w=\frac{1}{\lambda}\ln v$ yields 
\[
    - \hat{N}\int_{\mathit{M}} \,\Gamma(w)\,\Gamma^w(f)\,\phi\,\mathrm{d}\mathfrak{m}
    =- \frac{\hat{N}}{\lambda}\int_{\mathit{M}} \,\frac{\Gamma(v)}{v}\,\frac{\Gamma^w(f)\,\phi}{\lambda v}\,\mathrm{d}\mathfrak{m} \text.
\]

For the second integral, we first claim $\Gamma^w(f)\phi = e^{-2w}\Gamma(f)\phi\in W^{1,2}(M,\mathsf{d},\mathfrak{m})\cap L^\infty(M,\mathfrak{m})$. Boundedness is straightforward since each factor belongs to $L^\infty({\mathit{M}},\mathfrak{m})$, where this property for $\smash{e^{-2w}}$ follows from \Cref{lem:gaugefunction}. Since bounded Sobolev functions form an algebra, it remains to show that each factor belongs to $W^{1,2}({\mathit{M}},\mathsf{d},\mathfrak{m})$. For $\phi$ and $\Gamma(f)$, this follows directly from the definition of $\mathrm{Test}({\mathit{M}},\mathsf{d},\mathfrak{m})$. The inclusion $e^{-2w}\in W^{1,2}({\mathit{M}},\mathsf{d},\mathfrak{m})$ follows from the chain rule, finiteness of $\mathfrak{m}$, and the property $w\in W^{1,2}({\mathit{M}},\mathsf{d},\mathfrak{m})\cap L^\infty({\mathit{M}},\mathfrak{m})$ implied by \Cref{lem:gaugefunction}. Therefore, the Leibniz rule gives
\begin{multline*}
    \int_{\mathit{M}}\Gamma(w,\Gamma^w(f)\,\phi)\,\mathrm{d}\mathfrak{m}
    =\int_{\mathit{M}}\frac{1}{\lambda v}\Gamma(v,\Gamma^w(f)\,\phi)\,\mathrm{d}\mathfrak{m} 
    \\=
    \int_{\mathit{M}}\Gamma\left( v,\frac{\Gamma^w(f)\,\phi}{\lambda v}\right)\,\mathrm{d}\mathfrak{m} 
     -\int_{\mathit{M}} \Gamma\left( v,\frac{1}{\lambda v}\right)\Gamma^w(f)\phi\,\mathrm{d}\mathfrak{m}
    \\
    =\int_{\mathit{M}}\Gamma\left( v,\frac{\Gamma^w(f)\,\phi}{\lambda v}\right)\,\mathrm{d}\mathfrak{m} 
     +\int_{\mathit{M}} \frac{\Gamma(v)}{v}\frac{\Gamma^w(f)\phi}{\lambda v}\,\mathrm{d}\mathfrak{m}.
\end{multline*}
Using $\lambda\geq \hat N$ and $\Gamma(v)/v\geq 0$ $\mathfrak{m}$-a.e.~to drop the first integral in the second line, we obtain
\begin{multline*}
    - \hat{N}\int_{\mathit{M}} \,\Gamma(w)\,\Gamma^w(f)\,\phi\,\mathrm{d}\mathfrak{m} + \int_{\mathit{M}}\Gamma(w,\Gamma^w(f)\,\phi)\,\mathrm{d}\mathfrak{m}
     - \int_{\mathit{M}}\Gamma^w(f)\,\phi\,\mathrm{d} \kappa
     \\
     =
    \left(1 - \frac{\hat{N}}{\lambda}\right)\int_{\mathit{M}} \,\frac{\Gamma(v)}{v}\,\frac{\Gamma^w(f)\,\phi}{\lambda v}\,\mathrm{d}\mathfrak{m} 
     + \int_{\mathit{M}}\Gamma\left( v,\frac{\Gamma^w(f)\,\phi}{\lambda v}\right) \,\mathrm{d}\mathfrak{m} 
    - \int_{\mathit{M}} \frac{\Gamma^w(f)\,\phi}{\lambda v} \lambda v\,\mathrm{d} \kappa
     \\
     \geq 
   \int_{\mathit{M}}\Gamma\left( v,\frac{\Gamma^w(f)\,\phi}{\lambda v}\right) \,\mathrm{d}\mathfrak{m} 
    - \int_{\mathit{M}} \frac{\Gamma^w(f)\,\phi}{\lambda v} \lambda v\,\mathrm{d} \kappa
    =\int \frac{\Gamma^w(f)\phi}{\lambda v}\left[-\Delta^{-\lambda\kappa}v\right]\,\mathrm{d} \mathfrak{m}.
\end{multline*}
Since $ -\Delta^{\lambda\kappa}v=2\beta-2\beta v=2\beta(1-v)$ and $1\leq v\leq 2e^{\beta T}$ $\mathfrak{m}$-a.e., we obtain, using $\frac{\Gamma^w(f)\phi}{\lambda v}\in \mathrm{Test}({\mathit{M}})$ 
and $\frac{\Gamma^w(f)\phi}{\lambda v}\geq 0$ $\mathfrak{m}$-a.e.,
\[
\int \frac{\Gamma^w(f)\phi}{\lambda v}\left[-\Delta^{-\lambda\kappa}v\right]\,\mathrm{d} \mathfrak{m}
=
\frac{2\beta}\lambda\int \Gamma^w(f)\phi\left[\frac{1}v-1\right]\,\mathrm{d} \mathfrak{m}
\geq -\frac{\beta}\lambda (2-e^{-\beta T})
\int \Gamma^w(f)\phi\,\mathrm{d} \mathfrak{m}.
\]
We are left with estimating the constant on the right-hand side. 
Since  $\lambda \mathrm{k}=1-e^{-\beta T}$ and $\beta= \frac{1}T\ln\left(\frac1{1-\lambda\mathrm{k}} \right)$ we have
\[
\frac{\beta}\lambda (2-e^{-\beta T})
= \frac{1}T\ln\left(\frac1{1-\lambda\mathrm{k}} \right)\left(\mathrm{k}+\frac{1}{\lambda}\right).
\]
In order to estimate the right-hand side, we  distinguish three cases. Recall $\hat N=\tfrac{(n-2)(N-2)}{N-n}$.
\\
\textbf{Case 1:} $n>2$. We have $N=n+3(n-2)^2\mathrm{k}$, $\lambda=\hat N$, and $\mathrm{k}< 1\wedge \tfrac{1}{2(n-2)}$, so that 
\[
\lambda \mathrm{k}=\hat N\mathrm{k}
=
\frac{1}{3}+(n-2)\mathrm{k}\leq \frac{5}{6},\qquad \frac{1}{\lambda}=\frac{3(n-2)^2\mathrm{k}}{(n-2+3(n-2)^2\mathrm{k})(n-2)}\le 3\mathrm{k}.
\]
\textbf{Case 2:} $n<2$. We have $N=n+2(n-2)^2\mathrm{k}$, $\lambda=\hat N$, and $\mathrm{k}<1\wedge \tfrac{1}{2(2-n)}$, so that, since $n-2<0$,
\[
\lambda \mathrm{k}=\hat N\mathrm{k}
=
\frac{1}{2}+(n-2)\mathrm{k}\leq \frac{1}{2},\qquad \frac{1}{\lambda}=\frac{2(2-n)^2\mathrm{k}}{(2(n-2)^2\mathrm{k}+(2-n))(2-n)}\le 2\mathrm{k}.
\]
\textbf{Case 3:} $n=2$. We have $N=n+2\mathrm{k}$, $\lambda=1/2\mathrm{k}$, and $\mathrm{k}< 1$, such that
\[
\lambda \mathrm{k}=\frac{1}{2},
\qquad \frac{1}{\lambda}= 2\mathrm{k}.
\]
Hence, we always have $\lambda\mathrm{k}\leq 5/6$ and $1/\lambda\leq 3\mathrm{k}$. Therefore,
\[
 \frac{1}T\ln\left(\frac1{1-\lambda\mathrm{k}} \right)\left(\mathrm{k}+\frac{1}{\lambda}\right)
\leq 
\frac{1}T\ln\left(\frac1{1-5/6} \right)\left(\mathrm{k}+3\mathrm{k}\right)
=\frac{8\mathrm{k}}{T},
\]
and the claim follows.

Consider $f \in L^1(M,\mathfrak{m}_w)$ such that $f \ge 0$ $\mathfrak{m}_w$-a.e.~in $M$. Since $\mathfrak{m}$ has a finite measure and $w$ is essentially bounded below away from zero, the measure $\mathfrak{m}_w$ has a finite measure as well, so for any $\epsilon>0$, the function $f_\epsilon = f + \epsilon$ still belongs to $L^1(M,\mathfrak{m}_w)$. For $z \in M$ and $\tau \ge 0$, set $$u_\epsilon (z,\tau) := \mathsf{P}^w_\tau f_\epsilon (z)$$ 
and notice that for any given $\tau >0$ it holds that $u_\epsilon(z,\tau)>0$ for $\mathfrak{m}_w$-a.e.~$z \in M$.  By the BIP property of the original space $(M,\mathsf{d},\mathfrak{m})$, given $y,z \in M$ and $r>0$, there exists $\pi^r \in \mathrm{OptGeo}(\mathfrak{m}_0^r,\mathfrak{m}_1^r)$ concentrated on $\mathsf{d}$-geodesics, where $$\mathfrak{m}_0^r =\mathfrak{m}(B_r(y))^{-1} 1_{B_r(y)}\mathfrak{m}  \quad \text{and} \quad  \mathfrak{m}_1^r=\mathfrak{m}(B_r(z))^{-1} 1_{B_r(z)}\mathfrak{m}.$$  Consider $s<t$ in $(0,+\infty)$. Acting as in \cite[(4.11)]{GarofaloMondino-13} with the metric measure space $(M,\mathsf{d}_w,\mathfrak{m}_w)$,
\begin{align*}
    \int \ln \left( \frac{u_\epsilon(\gamma_1,s)}{u_\epsilon(\gamma_0,t)}\right) \,\mathrm{d} \pi^r(\gamma) & \le \int \int_0^1 |d(\ln u_\epsilon(\cdot,t + \tau(s-t))) |_{w}(\gamma_\tau) |\dot{\gamma}|_{w}(\tau) \,\mathrm{d} \tau \,\mathrm{d} \pi^r(\gamma) \\
    & \phantom{=} - (t-s) \int \int_0^1 \left( \frac{\partial}{\partial \tau}  \ln u_\epsilon \right) (\gamma_\tau,t + \tau(s-t)) \,\mathrm{d} \tau \,\mathrm{d} \pi^r(\gamma).
\end{align*}
Apply the Li-Yau inequality from Proposition \ref{prop:LiYau} at time $\tau$ to the function $f_\epsilon$. Divide the resulting inequality by $e^{2K\tau/3}u_\epsilon^2$ to obtain  
\begin{equation*}
- \left(\frac{\partial}{\partial \tau} \ln u_\epsilon\right) (z,\tau)
\leq -e^{2K\tau/3} |d(\ln u_\epsilon(\cdot,\tau) )|_{w}^2(z) - \frac{NK}{3}\frac{e^{2K\tau/3}}{1-e^{2K\tau/3}} \, \cdot 
\end{equation*}
Together with Young's inequality with $\eta >0$, this gives
\begin{align*}
    \int \ln \left( \frac{u_\epsilon(\gamma_1,s)}{u_\epsilon(\gamma_0,t)}\right) \,\mathrm{d} \pi^r(\gamma) 
    & 
    \le \frac{\eta}{2}\int \int_0^1 |d(\ln u_\epsilon(\cdot,t + \tau(s-t)) )|_{w}^2(\gamma_\tau)  \,\mathrm{d} \tau \,\mathrm{d} \pi^r(\gamma) \\
    &\phantom{=}  + \frac{1}{2\eta} \int \int_0^1 |\dot{\gamma}|_{w}^2(\tau) \,\mathrm{d} \tau \,\mathrm{d} \pi^r(\gamma)\\
    & \phantom{=} - (t-s) \int \int_0^1 e^{-2K(t + \tau(s-t))/3}|d(\ln u_\epsilon(\cdot,t + \tau(s-t)) )|_{w}^2(\gamma_\tau) \,\mathrm{d} \tau \,\mathrm{d} \pi^r(\gamma) \\
    & \phantom{=}- (t-s) \int \int_0^1 \frac{NK}{3}\frac{e^{2K(t + \tau(s-t))/3}}{1-e^{2K(t + \tau(s-t))/3}}  \,\mathrm{d} \tau \,\mathrm{d} \pi^r(\gamma).
\end{align*}
Since
\[
- (t-s) e^{-2K(t + \tau(s-t))/3} \le - (t-s) e^{-2Kt/3}
\]
we may choose $\eta=2(t-s)\mathrm{e}^{2Kt/3}$ to cancel out the first and third terms in the previous right-hand side. We get
\begin{align*}
    \int \ln \left( \frac{u_\epsilon(\gamma_1,s)}{u_\epsilon(\gamma_0,t)}\right) \,\mathrm{d} \pi^r(\gamma) & \le 
    \frac{\mathrm{e}^{-2Kt/3}}{4(t-s)} \int \int_0^1 |\dot{\gamma}|_{w}^2(\tau) \,\mathrm{d} \tau \,\mathrm{d} \pi^r(\gamma)
    \\
    &\phantom{=}- (t-s) \int \int_0^1 \frac{N K}{3}\frac{e^{2K(t + \tau(s-t))/3}}{1-e^{2K(t + \tau(s-t))/3}}  \,\mathrm{d} \tau \,\mathrm{d} \pi^r(\gamma).\nonumber
\end{align*}
In order to conclude, note that
\begin{align*}
\lim\limits_{r \downarrow 0} \int \int_0^1 |\dot{\gamma}|_{w}^2(t) \,\mathrm{d} t  \,\mathrm{d} \pi^r (\gamma) & = \lim\limits_{r \downarrow 0} \int \int_0^1 e^{-2w(\gamma(t))}|\dot{\gamma}|^2(t) \,\mathrm{d} t  \,\mathrm{d} \pi^r (\gamma)\\
& \le e^{-2 \|w\|_\infty} \lim\limits_{r \downarrow 0} \int \int_0^1 |\dot{\gamma}|^2(t) \,\mathrm{d} t  \,\mathrm{d} \pi^r (\gamma) \\
& = e^{-2 \|w\|_\infty} \mathsf{d}^2(y,z).
\end{align*}
Moreover, as pointed out in \cite{GarofaloMondino-13}, a direct computation shows that
\[
-(t-s) \int_0^1 \frac{N K}{3}\frac{e^{2K(t + \tau(s-t))/3}}{1-e^{2K(t + \tau(s-t))/3}}  \,\mathrm{d} \tau  = \frac{N}{2} \ln \left( \frac{1-e^{-2Kt/3}}{1-e^{-2Ks/3}} \right). 
\]
Therefore, letting $r \to 0$, we obtain
\[
\ln \left( \frac{u_\epsilon(y,s)}{u_\epsilon(z,t)}\right) \le \frac{e^{-2 \|w\|_\infty} \mathsf{d}^2(y,z)}{4(t-s)e^{2Kt/3}} + \frac{N}{2} \ln \left( \frac{1-e^{2Kt/3}}{1-e^{2Ks/3}} \right).
\]
We get the conclusion by using that $e^{-2 \|w\|_\infty} \mathsf{d}^2(y,z) \le \mathsf{d}_w^2(y,z)$ and then by letting $\epsilon$ tend to $0$.

\textbf{Step 1.} We prove that for all non-negative functions $f \in L^\infty({\mathit{M}}, \mathfrak{m}_w)$, $\varphi \in \mathrm{Test}_\infty(M,\mathsf{d}_w,\mathfrak{m}_w)$, and all $\tau,t>0$,
\begin{align*}
    \tau\int\Phi^w(t)\varphi\,\mathrm{d} \mathfrak{m}_w-(\tau+t)\int\Phi^w(0)\varphi\,\mathrm{d} \mathfrak{m}_w
    & \geq-C_1\int [\Delta^w P_t^wf]\varphi\,\mathrm{d}\mathfrak{m}_w  -\frac{N}{8}C_2\int [P_t^wf]\varphi\,\mathrm{d} \mathfrak{m}_w
    \end{align*}
    where
    \[
    C_1 =  C_1(t,K,\tau):= t(1+2K(\tau+t)) \quad \text{and} \quad C_2 =  C_2(t,K,\tau):= (1+2K(\tau+t))^2\ln\left(1+\frac{t}{\tau}\right).
    \]

We apply Proposition \ref{prop:gauge} with $a : (0,t) \to \mathbb{R}_+$ and $\gamma : (0,t) \to \mathbb{R}$ defined by
    \[
    a(s)=\tau+t-s, \qquad \gamma(s)=-\frac{N}{4a(s)}(1+2Ka(s)).
    \]
    Observe that
    \[
    a'=-1, \qquad \frac{4a\gamma}{N}=-(1+2Ka),\qquad \frac{2a\gamma^2}{N}=\frac{N}{8a}(1+2Ka)^2.
    \]
    In particular,
    \[
    a' - \frac{4a\gamma}{N} - 2 K a = 0.
    \]
    In virtue of the dominated convergence theorem, we assume that $f \ge \delta$ $\mathfrak{m}_w$-a.e.~for some $\delta>0$. Integrating the inequality from Proposition \ref{prop:gauge} from $0$ to $t$ with the previous $a$ and $\gamma$, we obtain
    
    \begin{align*}
    \tau\int\Phi^w(t)\varphi\,\mathrm{d} \mathfrak{m}_w-(\tau+t)\int\Phi^w(0)\varphi\,\mathrm{d} \mathfrak{m}_w & 
    \geq -\int_0^t(1+2Ka(s))\,\mathrm{d} s\int [\Delta^w P_t^wf]\varphi\,\mathrm{d}\mathfrak{m}_w\\
    & \phantom{=} \, -\frac{N'}{8}\int_0^T\frac{(1+2Ka(s))^2}{a(s)}\,\mathrm{d} s\int [P_t^wf]\varphi\,\mathrm{d} \mathfrak{m}_w.

\[
\int_0^t(1+2Ka(s))\,\mathrm{d} s\leq t(1+2K(\tau+t)) = C_1 \]
and \begin{align*} \phantom{=} \int_0^t\frac{(1+2Ka(s))^2}{a(s)}\,\mathrm{d} s

\end{align*}
we get the desired result.



\textbf{Step 2.} For any $h\in L^\infty({\mathit{M}},\mathfrak{m}_w)$ such that $|dh|_{w}^2\le 1$ $\mathfrak{m}_w$-a.e., any $\lambda,t >0$,
\[
h \le \psi(\lambda,t) + \Phi(\lambda,t)
\]

\[
\psi(\lambda,t)=\frac{1}{\lambda}\ln \mathsf{P}_t^w\mathrm{e}^{\lambda h}
\]

\[
\Phi(\lambda,t) = \int_0^t \lambda \eta (\lambda^2 t) + \frac{N}{8 \lambda} \ln \left( 1 + \frac{1}{\eta(\lambda^2 t)} \right) \left( \frac{1}{t} + 2 K (\eta(\lambda^2t )+1)\right) \,\mathrm{d} t, \quad \eta(x) = \frac{1}{2} \left( \sqrt{1 + \frac{1}{2x}}- 1\right).

\[
\mathsf{P}_t^wh=\frac{1}{\lambda}\mathsf{P}_t^w\ln \mathrm{e}^{\lambda h}
\leq \frac{1}{\lambda}\ln \mathsf{P}_t^w\mathrm{e}^{\lambda h}=\psi(\lambda,t) \quad \text{$\mathfrak{m}_w$-a.e.~on $M$.}
\]
Apply Step 1 to the function 
$f = \mathrm{e}^{\lambda h}$ to obtain
\begin{multline*}
    \tau\lambda^2\int P_t^w(\mathrm{e}^{\lambda h}|dh|_w^2)\varphi\,\mathrm{d} \mathfrak{m}_w
    -(\tau+t)\lambda^2\int \mathrm{e}^{\lambda \psi(\lambda,t)}|d\psi(\lambda,t)|_w^2\varphi\,\mathrm{d} \mathfrak{m}_w \\
        \geq -C_1\int [\Delta^w P_t^w\mathrm{e}^{\lambda h}]\varphi\,\mathrm{d}\mathfrak{m}_w -\frac{N}{8}C_2\int \mathrm{e}^{\lambda \psi(\lambda,t)}\varphi\,\mathrm{d} \mathfrak{m}_w.

Note that the second term on the left-hand side is non-negative. We can therefore drop it to obtain a weaker inequality. Since $\mathsf{P}_t^w$ is positivity-preserving,
\[
\mathsf{P}_t^w(\mathrm{e}^{\lambda h}|d h|_w^2)\leq \mathsf{P}_t^w\mathrm{e}^{\lambda h}
=\mathrm{e}^{\lambda\psi(\lambda,t)}.\]
Morever,
\[
\Delta^w \mathsf{P}_t^w\mathrm{e}^{\lambda h} = \left. \frac{\partial }{\partial t} \right|_{s=t} \mathsf{P}_s^w\mathrm{e}^{\lambda h} = \lambda \mathrm{e}^{\lambda \psi(\lambda,t)}\frac{\partial\psi}{\partial t}(\lambda, t).

Thus the above inequality yields
\[
    \tau\lambda^2\int \mathrm{e}^{\lambda\psi(\lambda,t)}\varphi\,\mathrm{d} \mathfrak{m}_w
        \geq -C_1\int \lambda \mathrm{e}^{\lambda \psi(\lambda,t)}\frac{\partial\psi}{\partial t}(\lambda, t) \varphi\,\mathrm{d}\mathfrak{m}_w -\frac{N}{8}C_2\int \mathrm{e}^{\lambda \psi(\lambda,t)}\varphi\,\mathrm{d} \mathfrak{m}_w.

Rearrange to get
\[
    \int \left( \frac{\partial\psi}{\partial t}(\lambda,t)  +\frac{\tau \lambda}{C_1} + \frac{NC_2}{8\lambda C_1} \right) C_1 \lambda\mathrm{e}^{\lambda \psi(\lambda,t)}\varphi\,\mathrm{d} \mathfrak{m}_w\geq0 .

\[
   0  \leq  \frac{\partial\psi}{\partial t}(\lambda,t)  + \frac{\tau \lambda}{C_1} + \frac{NC_2}{8\lambda C_1} = \frac{\partial\psi}{\partial t}(\lambda,t)  + \frac{\tau \lambda}{t(1+2K(\tau+t))} + \frac{N}{8\lambda}\frac{(1+2K(\tau+t))}{t}\ln\left(1+\frac{t}{\tau}\right) \qquad \text{$\mathfrak{m}_w$-a.e.}
\]
Choose
\[
\tau = t \eta (\lambda^2 t)
\]
to get
\[
    \frac{\partial\psi}{\partial t} (\lambda,t)
    +\frac{\lambda \eta (\lambda^2 t)}{1 + 2Kt(1+\eta(\lambda^2 t))} + \frac{N}{8 \lambda} \ln \left( 1 + \frac{1}{\eta(\lambda^2 t)} \right) \left( \frac{1}{t} + 2 K (\eta(\lambda^2t )+1)\right)
     \geq0.
   \]
   Since $1 + 2Kt(1+\eta(\lambda^2 t)) \ge 1$, we get
   \[
    \frac{\partial\psi}{\partial t} (\lambda,t)
    +G(\lambda,t)
     \geq0
   \]
   where
   \[
   G(\lambda,t) \coloneqq \lambda \eta (\lambda^2 t) + \frac{N}{8 \lambda} \ln \left( 1 + \frac{1}{\eta(\lambda^2 t)} \right) \left( \frac{1}{t} + 2 K (\eta(\lambda^2t)+1)\right).
   \]
 Integrating this inequality from $0\leq \epsilon\leq t$ yields 
\[\lambda\psi(\lambda,t)+\lambda\int_{\epsilon}^t G(\lambda,s)\,\mathrm{d} s\geq \lambda \psi(\lambda,\epsilon)\geq \lambda \mathsf{P}_{\epsilon}^wh\]
$\mathfrak{m}_w$-almost everywhere. In particular, since $\mathsf{P}_\epsilon^wh$ converges in $L^2({\mathit{M}},\mu^w)$ to $h$ as $\epsilon\to0$, there is a subsequence $(\epsilon_n)_n$ such that $\mathsf{P}_{\epsilon_n}^wh\to h$, $n\to\infty$, $\mathfrak{m}_w$-almost everywhere. Hence
\[
h=
\lim_{n\to \infty}\mathsf{P}_{\epsilon_n}^w h
\leq \psi(\lambda,t)+\Phi(\lambda,t)
\]
almost everywhere, as claimed.

\textbf{Step 3.} For $\mathfrak{m}_w$-a.e.~$y\in B_w(x,r/2)$, we have  \[
\mathsf{P}_t^w(1_B)(y)\geq \mathrm{e}^{-\lambda r/2}(\mathrm{e}^{-\lambda \Phi(\lambda,t)(y)}-\mathrm{e}^{-\lambda r/2 }).
\]
Apply the previous step to $h=-\mathsf{d}_w(x,\cdot)\wedge r$ where $x \in M$ and $r \in (0,1/\sqrt{K N}]$. Then $h\in L^\infty({\mathit{M}},\mathfrak{m}_w)$ with $|dh|_w \le 1$ $\mathfrak{m}_w$-a.e.~and we clearly have
\[
\mathrm{e}^{\lambda h}\leq \mathrm{e}^{-\lambda r}1_{B^c}+1_B
\]
where $B=B_w(x,r)$.  Apply $\mathsf{P}_t^w$ for $t>0$ to get 
\[
\mathrm{e}^{\lambda \psi(\lambda, t)}=\mathsf{P}_t^w\mathrm{e}^{\lambda h}\leq \mathrm{e}^{-\lambda r }+\mathsf{P}_t^w(1_B) \qquad \text{$\mathfrak{m}_w$-a.e.}
\]
Since
\[
h\geq -r/2 \qquad \text{on}\  B_w(x,r/2),
\]
we have for $\mathfrak{m}_w$-a.e.~$y\in B_w(x,r/2)$
\[
\mathrm{e}^{-\lambda r/2}\leq \mathrm{e}^{\lambda h(y)}\leq \mathrm{e}^{\lambda\psi(\lambda,t)(y)}\mathrm{e}^{\lambda \Phi(\lambda,t)(y)}
\]
and thus
\[
\mathsf{P}_t^w(1_B)(y)\geq \mathrm{e}^{\lambda \psi(\lambda, t)(y)}-\mathrm{e}^{-\lambda r }
\geq \mathrm{e}^{-\lambda r/2-\lambda \Phi(\lambda,t)(y)}-\mathrm{e}^{-\lambda r }
=\mathrm{e}^{-\lambda r/2}(\mathrm{e}^{-\lambda \Phi(\lambda,t)(y)}-\mathrm{e}^{-\lambda r/2 }).
\]

\textbf{Step 4.} Finally, we conclude by finding the constants $A\in(0,1)$ and $C>0$. Choose $\lambda=1/r$ and $t=Ar^2$ to obtain for $\mathfrak{m}_w$-a.e.~$y\in B_w(x,r/2)$
\[
\mathsf{P}_{Ar^2}^w(1_B)(y)
\geq\mathrm{e}^{- 1/2}(\mathrm{e}^{-(1/r) \Phi(1/r,Ar^2)(y)}-\mathrm{e}^{- 1/2 }).
\]
In order to bound the right-hand side from below, we estimate $F(1/r,Ar^2)$ from above as follows. Write
\[
    \frac{1}{r} \Phi(1/r,Ar^2)=\frac{1}{r}\int_0^{Ar^2}G(1/r,\tau)\,\mathrm{d} \tau
    \\
    =S_1+S_2+S_3
    \]
    where
    \[S_1=\frac{1}{r^2}\int_0^{Ar^2}
    \eta\left( \frac{t}{r^2}\right)\,\mathrm{d} t
    ,\qquad S_2=\frac{N}{8}\int_0^{Ar^2}\frac{1}{t}\ln\left(1+\frac{1}{\eta\left( \frac{t}{r^2}\right)}\right) \,\mathrm{d} t,\]
    and 
    \[
     S_3=\frac{2KN}{8}\int_0^{Ar^2}\bigg( \eta\left( \frac{t}{r^2}\right) +1 \bigg) \ln\left(1+\frac{1}{\eta\left( \frac{t}{r^2}\right)}\right)  \,\mathrm{d} t.
\]
The substitution $s=t/r^2$ yields 
\[
S_1=\int_{0}^{A}
    \eta(s) \,\mathrm{d} s
    ,\qquad S_2=\frac{N}{8}\int_0^A \ln \left(1 + \frac{1}{\eta(s)} \right) \frac{\,\mathrm{d} s}{s},
\]
and 
\[
     S_3=\frac{2KN}{8}r^2\int_0^{A}\bigg( \eta\left( s\right) +1 \bigg) \ln\left(1+\frac{1}{\eta\left(s\right)}\right)  \,\mathrm{d} s
\]
Since $\eta(s)\sim 1/\sqrt s$ as $s\to 0$, the integrals appearing in $S_1$, $S_2$, and $S_3$ are finite for $A>0$ and independent of $N$ and $K$. Recall that $N=n+3(1\vee (n-2)^2) \mathrm{k}\leq n+3\frac{(1\vee (n-2)^2)}{1\wedge (n-2)} =:\omega(n) $. Hence, for any $\epsilon>0$, there is a constant $A\in(0,1)$ such that 
\[
S_1\leq \epsilon, \qquad S_2\leq \epsilon \omega(n), \qquad \text{and}\qquad S_3\leq \epsilon KN r^2.
\]
Hence, for $\mathfrak{m}_w$-a.e.~$y\in B_w(x,r/2)$,
\[
\mathsf{P}_{Ar^2}^w(1_B)(y)\geq \mathrm{e}^{\lambda \psi(\lambda, t)(y)}-\mathrm{e}^{-\lambda r }
\geq \mathrm{e}^{-1/2}(\mathrm{e}^{-(1+\omega(n))\epsilon -\epsilon KNr^2)}-\mathrm{e}^{-1/2 }).
\]
This yields the claim.

    & =p_{2s}^w(y,y)^\frac{1}{2}V_w(x,\sqrt{t})^\frac{1}{2}.
    \end{align*}

    \[
    \frac{c^2}{V_w(x,\sqrt{t})}\le p^w_{At/4}(y,y)
    \leq p_t^w(y,z)

    \]
    for $\mathfrak{m}_w$-a.e.~$z,y\in B_w(x,\sqrt t/2)$ and all $t\leq 4/(KN)$. Estimating the factors on the right-hand side further, we obtain
    \[
    \mathrm{e}^{\frac{\mathsf{d}_w(z,y)^2}{4(t-At/4)\mathrm{e}^{2Kt/3}}}\leq \mathrm{e}^{\frac{1}{(4-A)\mathrm{e}^{2Kt/3}}}\leq 1
    \]
    and, by the mean-value theorem,
    \[
    \frac{\mathrm{e}^{2Kt/3}-1}{\mathrm{e}^{KAt/6}-1}
    =\frac{2}{A}\frac{\mathrm{e}^{2Kt/3}-1}{2Kt/3-0}\frac{KAt/6-0}{\mathrm{e}^{KAt/6}-1}
    \leq

<!-- theorem-4: body_md -->
\label{thm:main1}
    Let $({\mathit{M}},\mathsf{d},\mathfrak{m})$ be a locally compact length infinitesimally Hilbertian space with strongly regular Dirichlet form ${\sf Ch}$ and intrinsic metric~$\mathsf{d}$. Assume  $\mathfrak{m}({\mathit{M}})<\infty$ and  $({\mathit{M}},\mathsf{d},\mathfrak{m})$ satisfies ${\mathrm{BE}}(-\kappa,n)$ for some $\kappa \in \mathcal{M}_0(M,\mathsf{d},\mathfrak{m})$ such that
\[\mathrm{k}_T= \mathrm{k}_T(M,\mathsf{d},\mathfrak{m},\kappa)<1\wedge \frac{1}{2|c-2|}.\]
  \begin{enumerate}
        \item
    The space $({\mathit{M}},\mathsf{d},\mathfrak{m})$ satisfies the local doubling property and a family of local weak Poincar\'e inequalities. More precisely, there exist constants $c_1=c_1(\mathrm{k}_T,n,\mu)\geq 1$, $c_2=c_2(\mathrm{k}_T,n)\geq 1$ and $N=N(\mathrm{k}_T,n)\geq n$ such that, for all $0<r\leq R\leq (4/(KN))\wedge (\operatorname{diam}{\mathit{M}}/c_2)$, we have
    \[
    \frac{V(x,R)}{V(x,r)}\leq c_1\left(\frac{R}{r}\right)^{N},\qquad \int_{B(x,r)}|\varphi-\bar\varphi|^2\,\mathrm{d}\mathfrak{m}\leq c_1r^2\int_{B(x,c_2 r)}\Gamma(\varphi)\,\mathrm{d}\mathfrak{m}.
    \]
    In particular, $N\to n$ when $\mathrm{k}_T\to 0$.
     \item The space $({\mathit{M}},\mathsf{d},\mathfrak{m})$ satisfies a scale-invariant parabolic Harnack inequality. More precisely, there exist constants $c_1=c_1(\mathrm{k}_T,N,\mu)>0$ and $c_2=c_2(\mathrm{k}_T,N,\mu)>0$ such that, for all $x\in M$, $t\in\operatorname{\mathbb{R}}$, $r\in(0,(4/(KN))\wedge (\operatorname{diam}{\mathit{M}}/c_2)]$, and all $u$ solving $\partial_tu=\Delta u $ weakly in $B(x,2r)$, we have 
    \[
    \sup_{[t-r^2,t-3r^2/4]\times B(x,r)} u \leq c_1 \inf_{[t-r^2/4,t]\times B(x,r)} u.
    \]
    \item The heat kernel $p$ of $({\mathit{M}},\mathsf{d},\mathfrak{m})$ satisfies double-sided Gaussian heat kernel bounds. More precisely, there exist constants $c_1=c_1(\mathrm{k}_T,n)\ge 1$ and $c_2=c_2(\mathrm{k}_T,n)\geq 1$ such that, for $\mathfrak{m}$-a.e.~$x,y\in {\mathit{M}}$ and $0<t\leq (4/(KN))\wedge (\operatorname{diam}{\mathit{M}}/c_2)$:
    \[
    \frac{c_1^{-1}}{V(x,\sqrt{t})}\mathrm{e}^{-c_1\frac{d(x,y)^2}{t}
    }
    \leq p_t(x,y)
    \leq 
    \frac{c_1}{V(x,\sqrt{t})}\mathrm{e}^{-\frac{d(x,y)^2}{c_1t}
    }.
    \]
    \end{enumerate}
