\label{cor:linfty-fully-discrete-rate}

If \(\alpha C_Bh^{-2}\le q<1\) for fixed \(q\), where \(C_B\) is the mesh-family stability constant, independent of \(h\), from Lemma~\ref{lem:linfty-stab}, then, for sufficiently small \(h\),
\begin{equation*}
\|\gamma_h^\delta-\gamma\|_{L^\infty(W_0)}\le C\bigl(\eta_\phi(h)+\mathfrak c_hh^{(1-\theta_N)\sigma_I}+\delta+\alpha h^{-1}+h^{\beta_c}\bigr).
\end{equation*}
The ideal-profile error satisfies \(\eta_\phi(h)\le C\mathfrak c_h\). In particular, if \(0<\alpha(h)\le c_\alpha h^2\) with a fixed constant \(c_\alpha>0\) satisfying \(c_\alpha C_B<1\), then
\begin{equation}\label{eq:linfty_rate_sim}
\|\gamma_h^\delta-\gamma\|_{L^\infty(W_0)}\le C\bigl(\mathfrak c_h+\delta+h^{\beta_c}\bigr).
\end{equation}
Consequently, \(\gamma_h^\delta\to\gamma\) in \(L^\infty(W_0)\) whenever \(h\to0\) and \(\delta(h)\to0\). \label{prop:mesh-resolved-concentration}
Assume the hypotheses of Lemma \ref{lem:uniform-crz-concentrating}, Assumption \ref{ass:interp-concentrating-data}, and \eqref{eq:concentration-index-choice}. Then, for all sufficiently small \(h\), the function \(g_{i,h}\) in \eqref{eq:def-gih} is well defined, belongs to \(V_h^W\), and satisfies
\[
\|g_{i,h}\|_{W^s(\mathbb{R}^d)}=1.
\]
Define
\begin{equation}\label{eq:def-rh-ellh}
r_h := \max_{1\le i\le N_h} \sup_{z\in\operatorname{supp}(g_{i,h})}|z-x_i|,
\qquad \ell_h := \max_{1\le i\le N_h} \|g_{i,h}\|_{L^2(\mathbb{R}^d)}.
\end{equation}
Then
\begin{align}
r_h &\le C\bigl(N(h)^{-1}+h\bigr), \label{eq:rh-rate} \\ \ell_h &\le C\left(
N(h)^{-s} + h^{s+\sigma_I}N(h)^{\sigma_I} \right).
\label{eq:ellh-rate}
\end{align}
The constants may depend on \(d,s,k,\sigma_I,W_0,W\) and on the shape-regularity of the mesh family, but are independent of \(h\), \(i\), \(N_h\), and \(N(h)\).

By \eqref{eq:interp-concentrating-Hs} and \eqref{eq:phi-Hst-rate}, we have

\label{lem:exact-galerkin-dn-consistency}

\[
 |\mathcal D_h^{\rm G}(f,g)-\langle\Lambda_\gamma f,g\rangle|
 \le C\kappa_{G,h}^2\|f\|_{W^s(\mathbb R^d)}\|g\|_{W^s(\mathbb R^d)},

\]

<!-- lemma-4: body_md -->
\label{lem:fractional-cutoff-tail}
Let \(x_i\in\overline{W_0}\), let \(0<r\le t/4\) for some $t>0$, and let
\(\chi_{i,t}\in C_c^\infty(B(x_i,t))\) satisfy
\[
0\le \chi_{i,t}\le 1,
\qquad
\chi_{i,t}\equiv 1
\quad\text{on }B(x_i,t/2).
\]
Then
\begin{equation*} 
\sup_{x\in B(x_i,r)}\int_{\mathbb{R}^d}|\nabla^s\chi_{i,t}(x,y)|^2\,dy \le Ct^{-2s}, 
\end{equation*}
where \(C>0\) depends only on \(d\) and \(s\).
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Fix \(x\in B(x_i,r)\). Since \(r\le t/4\), we have 
\(\chi_{i,t}(x)=1\). The integrand vanishes for \(y\in B(x_i,t/2)\). On the remaining set,
\( |x-y|\ge t/4\).
Therefore, using the definition of the fractional gradient and \(0\le \chi_{i,t}\le1\),
\[
\int_{\mathbb{R}^d}|\nabla^s\chi_{i,t}(x,y)|^2\,dy
\le
C\int_{\{|x-y|\ge t/4\}}\frac{dy}{|x-y|^{d+2s}}
\le
C t^{-2s}.
\]
This completes the proof.

<!-- theorem-3: body_md -->
\label{thm:uniform-concentration}
Assume \( 0<\gamma_0 \le \gamma(x) \le \gamma_1 <\infty\) for almost every \(x\in\mathbb{R}^d\) and \(\sqrt\gamma\in C(\overline W)\). Let \(u^e_{g_i}\) be the exact weak solution satisfying \eqref{eq:weakstate-gi} with exterior data \(g_{i,h}\) defined by \eqref{eq:def-gih}. Let \(r_h\) and \(\ell_h\) be defined by \eqref{eq:def-rh-ellh}. For all sufficiently small \(h\), assume \(4r_h<d_W\). Then for every \(t\in[4r_h,d_W]\),
\begin{equation}\label{eq:uniform-concentration-t}
\eta_c(h)
\le
C\left(
\omega_{\sqrt\gamma}(t)
+
t^{-2s}\ell_h^2
+
\ell_h^2
\right),
\end{equation}
where
\[
\omega_{\sqrt\gamma}(t)
:=
\sup_{\substack{x,y\in\overline W\\ |x-y|\le t}}
|\sqrt\gamma(x)-\sqrt\gamma(y)|.
\]
The constant \(C>0\) is independent of \(h\), \(i\), \(N_h\), and \(t\).
<!-- end theorem-3 -->
<!-- proof-5: body_md -->
Fix \(i\in\{1,\dots,N_h\}\), and for a moment, write
\( u_i:=u^e_{g_i}\) and \(g_i:=g_{i,h}\).
We split
\begin{equation*} 
|B_\gamma(u_i,u_i)-\gamma(x_i)| \le |B_\gamma(u_i,u_i)-B_\gamma(g_i,g_i)|+|B_\gamma(g_i,g_i)-\gamma(x_i)|.
\end{equation*}
For the first term, let \(w_i = u_i-g_i \in \widetilde H^s(\Omega)\); then \(B_\gamma(u_i,w_i) =0\) and 
\[
B_\gamma(g_i,g_i) = B_\gamma(u_i - w_i, u_i - w_i) = B_\gamma(u_i, u_i) + B_\gamma(w_i, w_i),
\]
which, together with the boundedness of \(B_\gamma\) and the exterior energy estimate \eqref{eq:crz-exterior-energy}, implies
\begin{equation}\label{eq:eta-c-first-term}
|B_\gamma(u_i,u_i)-B_\gamma(g_i,g_i)| = B_\gamma(w_i, w_i) \leq C \|w_i\|^2_{H^s(\mathbb{R}^d)} = C \|u_i- g_i\|^2_{H^s(\mathbb{R}^d)}\leq C \|g_i\|^2_{L^2(W)} \leq C\ell_h^2.
\end{equation}
We now estimate \(B_\gamma(g_i,g_i)-\gamma(x_i)\). Since \(
\|g_i\|_{W^s(\mathbb{R}^d)}^2 = \|g_i\|_{L^2(\mathbb{R}^d)}^2 + \|\nabla^s g_i\|_{L^2(\mathbb{R}^{2d})}^2
= 1\), we have \( \|\nabla^s g_i\|_{L^2(\mathbb{R}^{2d})}^2 = 1-\|g_i\|_{L^2(\mathbb{R}^d)}^2\). Therefore,
\begin{align}
B_\gamma(g_i,g_i)-\gamma(x_i) = \iint_{\mathbb{R}^{2d}}
\bigl(\gamma(x)^{1/2}\gamma(y)^{1/2}-\gamma(x_i)\bigr)
|\nabla^s g_i(x,y)|^2\,dx\,dy - \gamma(x_i)\|g_i\|_{L^2(\mathbb{R}^d)}^2 .
\label{eq:Bgg-minus-gamma}
\end{align}
The last term in \eqref{eq:Bgg-minus-gamma} is bounded by \(C\ell_h^2\). Since \(\sqrt{\gamma}\) is bounded,
\[
|\gamma(x)^{1/2}\gamma(y)^{1/2}-\gamma(x_i)|
\le
C\bigl(|\gamma(x)^{1/2}-\gamma(x_i)^{1/2}|+|\gamma(y)^{1/2}-\gamma(x_i)^{1/2}|\bigr).
\]
By symmetry, it is enough to estimate
\[
J_i := \iint_{\mathbb{R}^{2d}}
|\gamma(y)^{1/2}-\gamma(x_i)^{1/2}|\,|\nabla^s g_i(x,y)|^2\,dx\,dy.
\]
Fix \(t\in[4r_h,d_W]\). Since \(t\le d_W\), we have \(B(x_i,t)\subset\subset W\). Choose
\(\chi_{i,t}\in C_c^\infty(B(x_i,t))\) such that
\[
0\le \chi_{i,t}\le1,
\qquad
\chi_{i,t}\equiv1
\quad\text{on }B(x_i,t/2).
\]
Since \(\operatorname{supp}(g_i)\subset\overline{B(x_i,r_h)}\) and \(r_h\le t/4\), we have \(g_i\chi_{i,t}=g_i\). Using the product rule for the fractional gradient, we get
\[
|\nabla^s g_i(x,y)|^2 = |\nabla^s (g_i\chi_{i,t})(x,y)|^2 \le
2|\chi_{i,t}(y)|^2|\nabla^s g_i(x,y)|^2 + 2|g_i(x)|^2|\nabla^s\chi_{i,t}(x,y)|^2.
\]
Hence \(J_i\le 2J_{i,1}+2J_{i,2}\), where
\[
J_{i,1} := \iint_{\mathbb{R}^{2d}} |\gamma(y)^{1/2}-\gamma(x_i)^{1/2}|\,|\chi_{i,t}(y)|^2|\nabla^s g_i(x,y)|^2\,dx\,dy
\]
and
\[
J_{i,2} := \iint_{\mathbb{R}^{2d}} |\gamma(y)^{1/2}-\gamma(x_i)^{1/2}|\,|g_i(x)|^2|\nabla^s\chi_{i,t}(x,y)|^2\,dx\,dy.
\]
For \(J_{i,1}\), the cutoff implies \(y\in B(x_i,t)\). Hence \(|\gamma(y)^{1/2}-\gamma(x_i)^{1/2}|\le \omega_{\sqrt\gamma}(t)\), and therefore
\[
J_{i,1} \le
\omega_{\sqrt\gamma}(t)
\|\nabla^s g_i\|_{L^2(\mathbb{R}^{2d})}^2 \le \omega_{\sqrt\gamma}(t).
\]
For \(J_{i,2}\), boundedness of \(\sqrt{\gamma}\), the support of \(g_i\), and
Lemma \ref{lem:fractional-cutoff-tail} give
\[
J_{i,2}
\le
C
\int_{\operatorname{supp}(g_i)}
|g_i(x)|^2
\left(
\int_{\mathbb{R}^d}|\nabla^s\chi_{i,t}(x,y)|^2\,dy
\right)dx \le C t^{-2s}\|g_i\|_{L^2(\mathbb{R}^d)}^2 \le C t^{-2s}\ell_h^2.
\]
Thus
\[
J_i \le C\left( \omega_{\sqrt\gamma}(t)+t^{-2s}\ell_h^2 \right).
\]
The same bound holds for the term containing \(|\gamma(x)^{1/2}-\gamma(x_i)^{1/2}|\). Combining this with
\eqref{eq:Bgg-minus-gamma}, we obtain
\[
|B_\gamma(g_i,g_i)-\gamma(x_i)|
\le
C\left(
\omega_{\sqrt\gamma}(t) + t^{-2s}\ell_h^2 + \ell_h^2
\right).
\]
Taking the maximum over \(i=1,\dots,N_h\) proves \eqref{eq:uniform-concentration-t}.

\label{cor:eta-c-explicit-rate}

\Psi_{\beta_c,s}(r):=
\begin{cases} r^{\beta_c},&\beta_c<2s,\\ r^{2s}(1+|\log r|),&\beta_c=2s,\\ r^{2s},&\beta_c>2s. 
\end{cases} 
\end{equation*}

\begin{equation*}
\eta_c(h)\le C\mathfrak c_h.
\end{equation*}

\end{equation}

Write \(K_h=\mathsf M_h+\mathsf S_h\), where \(
(\mathsf M_h)_{ij}:=\int_{W_0}\psi_i\psi_j\,dx\), and 
\((\mathsf S_h)_{ij}:=\int_{W_0}\nabla\psi_i\cdot\nabla\psi_j\,dx\).

\[

= \int_{W_0}\psi_i\sum_{j=1}^{N_h}\psi_j\,dx = \int_{W_0}\psi_i\,dx = \mu_i .
\]

\[
|\operatorname{supp}\psi_i|\le Ch^d, \qquad \|\nabla\psi_i\|_{L^\infty(\operatorname{supp}\psi_i)}\le Ch^{-1}, \qquad
\mu_i\ge ch^d,
\]
and only a uniformly bounded number of basis functions overlap with \(\psi_i\). Therefore

\[

\]

\[

\le \sum_{m=0}^{\infty}q^m = \frac1{1-q}.
\]
This proves \eqref{eq:sign-free-smoother-stability}.

\label{ass:liouville-regularity}
Throughout the Liouville--Cauchy reconstruction we assume the following.

\begin{enumerate}[label=(\roman*)]
\item \textbf{Ellipticity, normalization, and Liouville admissibility.}
There exist constants \(0<\gamma_0\le\gamma_1<\infty\) such that \(\gamma_0\le\gamma\le\gamma_1\) almost everywhere in \(\mathbb R^d\) and  \( m_\gamma \in \widetilde H^s(\Omega_R)\cap H^{s,d/s}(\mathbb R^d)\). In particular, \(m_\gamma=0\) almost everywhere in \(\mathbb R^d\setminus\Omega_R\), equivalently \(\gamma\equiv1\) in \(\mathbb R^d\setminus\Omega_R\). The \(H^{s,d/s}\)-regularity is the hypothesis used in the weak fractional Liouville reduction.

\item \textbf{\(L^\infty\)-representative of the Liouville potential on the computational domain.} The restriction of the weak Sobolev multiplier \(q_\gamma\) to \(\Omega_R\) is represented by a function in \(L^\infty(\Omega_R)\), still denoted by \(q_\gamma\). More precisely, \(
\langle q_\gamma,\varphi\rangle =\int_{\Omega_R}q_\gamma\varphi\,dx\) for all \(\varphi\in C_c^\infty(\Omega_R)\). We let \( Q_\gamma=q_\gamma|_\Omega\in L^\infty(\Omega)\) and \(q_\gamma|_{W}\in L^\infty(W)\). 

\item \textbf{Exterior \(C^1\)-regularity.}
The conductivity has a \(C^1\)-representative on an open neighborhood
\(\mathcal O_0\) of \(\overline{W_0}\), with
\(\mathcal O_0\Subset W\). Consequently,
\[
    a_\gamma:=\gamma^{-1/2}
    \in W^{1,\infty}(W_0)
    \cap
    \bigcap_{U\Subset W_0}\mathcal M_s(U).
\]
This exterior \(C^1\)-regularity is used only to verify the discrete multiplier consistency estimate \eqref{eq:eps-a-def} below.
\end{enumerate}

<!-- lemma-7: body_md -->
\label{lem:discrete-multiplier-consistency}
Let \(W_*\) and \(\sigma_h\) be chosen as above, and assume Assumption~\ref{ass:liouville-regularity}. Let \(\widehat\Gamma_h^\delta\) be the local mollification of \(\Gamma_h^\delta\) on \(W_*\) with a nonnegative standard mollifier of radius \(\sigma_h\), and define \(a_{h,\sigma}^\delta:=(\widehat\Gamma_h^\delta)^{-1/2}\).  Then, for every \(U\in\{W_a,W_b,W_c\}\), there holds
\begin{equation}\label{eq:eps-a-def}
    \varepsilon_U^a(h,\delta)
    :=
    \sup_{0\ne\phi_h\in V_h^U}
    \frac{
        \|I_h^U(a_{h,\sigma}^\delta\phi_h)-a_\gamma\phi_h\|_{H^s(\mathbb R^d)}
    }{
        \|\phi_h\|_{H^s(\mathbb R^d)}
    }
    \longrightarrow0 .
\end{equation}
More precisely,
\begin{equation}\label{eq:eps-a-bound}
    \varepsilon_U^a(h,\delta)
    \le
    C\left(
        e_h^\delta+\omega_\gamma(\sigma_h)
        +\frac{e_h^\delta}{\sigma_h}
        +\omega_{\nabla\gamma}(\sigma_h)
        +h^{1-s}
    \right),
\end{equation}
where \(\omega_\gamma\) and \(\omega_{\nabla\gamma}\) denote the moduli of
continuity of \(\gamma\) and \(\nabla\gamma\) on \(\overline{W_0}\). The
constant \(C\) is independent of \(h\), \(\delta\), and \(U\).

<!-- proof-13: body_md -->
For \(x\in W_*\), the choice of \(\sigma_h\) gives
\(B(x,\sigma_h)\subset W_0\). Hence the mollification uses only exterior
values where \(\Gamma_h^\delta\) is known. Since the mollifier is nonnegative
and \(\Gamma_h^\delta\in[\gamma_0/2,2\gamma_1]\), we have \( \gamma_0/2\le \widehat\Gamma_h^\delta\le 2\gamma_1 \quad\text{on }W_* \).
Moreover, \( \|\widehat\Gamma_h^\delta-\gamma\|_{L^\infty(W_*)} \le e_h^\delta+\omega_\gamma(\sigma_h)\). Differentiating the mollification and adding and subtracting \(\nabla(\rho_{\sigma_h}*\gamma)=\rho_{\sigma_h}*\nabla\gamma\), we obtain
\[
\|\nabla\widehat\Gamma_h^\delta-\nabla\gamma\|_{L^\infty(W_*)}
\le
\|\nabla\rho_{\sigma_h}\|_{L^1}
\|\Gamma_h^\delta-\gamma\|_{L^\infty(W_0)}
+
\omega_{\nabla\gamma}(\sigma_h) \le C\frac{e_h^\delta}{\sigma_h}
+

\]
since \(\|\nabla\rho_{\sigma_h}\|_{L^1}=C\sigma_h^{-1}\). Thus \(\widehat\Gamma_h^\delta\to\gamma\) in \(W^{1,\infty}(W_*)\). Since
\(t\mapsto t^{-1/2}\) is smooth on \([\gamma_0/2,2\gamma_1]\),
\[
    \eta_a(h,\delta)
    := \|a_{h,\sigma}^\delta-a_\gamma\|_{W^{1,\infty}(W_*)} \le

        +\omega_{\nabla\gamma}(\sigma_h)
    \right).
\]

\[
\begin{aligned}
\|I_h^U(\omega\phi_h)-\omega\phi_h\|_{L^2(\mathbb R^d)}
&\le
Ch\|\nabla\omega\|_{L^\infty(W_*)}
\|\phi_h\|_{L^2(\mathbb R^d)},\\
\|I_h^U(\omega\phi_h)-\omega\phi_h\|_{H^1(\mathbb R^d)}
&\le
C\|\nabla\omega\|_{L^\infty(W_*)}
\|\phi_h\|_{L^2(\mathbb R^d)}.
\end{aligned}
\]
For \(U=W_c\), the same estimates hold by the explicit construction of the local interpolant \(I_h^c\) on the enriched space; see \eqref{eq:tail-enriched-superapprox}. Interpolation between \(L^2\) and \(H^1\) therefore yields, uniformly for \(U\in\{W_a,W_b,W_c\}\),

\le
C h^{1-s}
\|\omega\|_{W^{1,\infty}(W_*)}
\|\phi_h\|_{H^s(\mathbb R^d)} .
\]

\[
\|I_h^U(a_{h,\sigma}^\delta\phi_h) -a_{h,\sigma}^\delta\phi_h\|_{H^s(\mathbb R^d)}
\le Ch^{1-s}\|\phi_h\|_{H^s(\mathbb R^d)} .
\]
Since \(U\Subset W_*\), choose \(\zeta_U\in C_c^\infty(W_*)\) with \(\zeta_U=1\) on \(U\). Multiplication by a \(W^{1,\infty}(\mathbb R^d)\)-function is bounded on \(H^s(\mathbb R^d)\) for \(0<s<\min\{1,d/2\}\). Applying this to \(\zeta_U(a_{h,\sigma}^\delta-a_\gamma)\) gives
\[

\le
C\|a_{h,\sigma}^\delta-a_\gamma\|_{W^{1,\infty}(W_*)}

\]
Combining the last two bounds proves \eqref{eq:eps-a-bound}. The choice of \(\sigma_h\) above gives \(e_h^\delta\to0\), \(\sigma_h\to0\), and \(e_h^\delta/\sigma_h\to0\). \le C(\mathrm{r}_h(e)+Mh^s)\|\phi\|_{H^{2s}(\mathbb R^d)}.
\end{aligned}
\]
Taking the supremum over \(\|\phi\|_{H^{2s}(\mathbb R^d)}=1\) yields \(\|\zeta(-\Delta)^s e\|_{H^{-2s}(\mathbb R^d)}\le C(\mathrm{r}_h(e)+Mh^s)\).
Set $w=(1-\chi)e$. The map $(-\Delta)^s:L^2\to H^{-2s}$ is bounded, and $w$ vanishes on $B_{3r}$. Hence, for $\vartheta\le1$,
\[
\begin{aligned}
\|\zeta(-\Delta)^sw\|_{H^{-2s}}\le C(\mathrm{r}_h(e)+Mh^s+\|\chi e\|_{L^2(\mathbb{R}^{d})})\le CM\vartheta^{1/2},\qquad
\|\zeta(-\Delta)^sw\|_{L^2(\mathbb{R}^{d})}\le CM.
\end{aligned}
\]
The second estimate follows from the separated kernel; interpolation gives $\|(-\Delta)^sw\|_{H^{-s}(B_r)}\le CM\vartheta^{1/4}$. Also, $\|w\|_{H^s(\mathbb R^d)}\le CM$ while $\operatorname{supp}w\Subset B_L$ for a concentric ball $B_L$ containing $\overline{\Omega_R}$. If $d\ge2$, $D:=B_L\setminus\overline{B_{5r/2}}$ is a bounded Lipschitz domain containing $\operatorname{supp}w$ and disjoint from $B_r$. For sufficiently small \(\vartheta\), Proposition~6.1 of \cite{Ghosh2020uniqueness}, with \(s'=0\), yields
\[
\|w\|_{L^2(\mathbb{R}^{d})}\le CM[\log(c\vartheta^{-1/4})]^{-\sigma}\le CM[\log(2+\vartheta^{-1})]^{-\sigma},
\]
since the two logarithms are comparable in this range.
For \(d=1\), the interval \(B_L\setminus\overline{B_{5r/2}}\) has two components. We adapt the continuation argument in the proof of \cite[Prop.~6.1]{Ghosh2020uniqueness} using the local estimates of \cite{RulandSalo2020low}. Choose \(I_-:=(-L-r/2,-2r)\) and \(I_+:=(2r,L+r/2)\), which together contain \(\operatorname{supp}w\), and let \(\widetilde w\) be the Caffarelli--Silvestre extension of the whole function \(w\). By \cite[Lems.~4.1--4.2]{RulandSalo2020low},
\[
\|\widetilde w(\cdot,y)\|_{L^2(\mathbb R)}\le\|w\|_{L^2(\mathbb R)}\le CM,\qquad \|y^{(1-2s)/2}\nabla\widetilde w\|_{L^2(\mathbb R\times(0,\infty))}\le CM.
\]
Its trace vanishes on \(B_r\), and its conormal derivative there has \(H^{-s}(B_r)\) norm at most \(CM\vartheta^{1/4}\). The local boundary-to-bulk estimate \cite[Prop.~5.13]{RulandSalo2020low} therefore bounds the weighted \(L^2\) norm on a fixed interior ball by \(CM\vartheta^c\), for some \(c>0\).
For an auxiliary height \(0<\varrho<1/2\), cover \((I_-\cup I_+)\times(\varrho,1)\) by at most \(C\varrho^{-2}\) balls whose radii are one fifth of their centre heights. Chains from the initial ball first reach a fixed height, move horizontally, and then descend with geometrically decreasing heights. Each chain uses at most \(C(1+|\log\varrho|)\) balls, whose fourfold enlargements remain in the upper half-plane and in a fixed-height strip. Iterating \cite[Prop.~5.4]{RulandSalo2020low}, as in the proof of \cite[Thm.~5.5]{RulandSalo2020low}, bounds the weighted norm on each covering ball by \(CM\vartheta^{c\varrho^\kappa}\), for some \(\kappa>0\). The \(L^2\) contraction above also gives \(\|y^{(1-2s)/2}\widetilde w\|_{L^2(\mathbb R\times(0,\varrho))}\le CM\varrho^{1-s}\). Summing squared norms over the cover and adding the thin-layer bound yields
\[
\sum_{\pm}\|y^{(1-2s)/2}\widetilde w\|_{L^2(I_\pm\times(0,1))}\le CM\bigl(\varrho^{-1}\vartheta^{c\varrho^\kappa}+\varrho^{1-s}\bigr).
\]
Choose \(\varrho=[\log(\vartheta^{-1})]^{-1/(2\kappa)}\) for sufficiently small \(\vartheta\). Since \(\vartheta^{c\varrho^\kappa}=e^{-c\sqrt{\log(\vartheta^{-1})}}\), the preceding estimate and comparison of the logarithms give
\[
\sum_{\pm}\|y^{(1-2s)/2}\widetilde w\|_{L^2(I_\pm\times(0,1))}\le CM[\log(2+\vartheta^{-1})]^{-\mu},\qquad \mu:=\frac{1-s}{2\kappa}>0.
\]
For \(0<t<1/2\), the weighted vertical trace estimate gives
\[
\|w\|_{L^2(I_\pm)}\le C\left(t^sM+t^{-(1-s)}\|y^{(1-2s)/2}\widetilde w\|_{L^2(I_\pm\times(0,2t))}\right).
\]
Taking \(t=[\log(2+\vartheta^{-1})]^{-\mu}\) for sufficiently small \(\vartheta\) and summing over the two intervals yields \(\|w\|_{L^2(\mathbb R)}\le CM[\log(2+\vartheta^{-1})]^{-\mu s}\).
Thus, in both cases, adding $\|\chi e\|_{L^2(\mathbb{R}^{d})}\le CM\vartheta^{1/2}$ proves \eqref{eq:discrete-cauchy-logarithmic} for small $\vartheta$; $\|e\|_{L^2(\mathbb{R}^{d})}\le M$ covers the remaining values.

The logarithmic modulus in \eqref{eq:final-interior-logarithmic} contains every propagated error through $E_h^{\rm in}$. \label{lem:abstract-shifted-tikhonov}
Let \(\mathsf X,\mathsf Y\) be real Hilbert spaces, let \(E\subset\mathsf X\) be finite dimensional, and let \(T:E\to\mathsf Y\) be linear. For \(b\in\mathsf Y\), \(x_c\in E\), and \(\alpha>0\), the functional
\[
J_b(x):=\tfrac12\|Tx-b\|_{\mathsf Y}^2+\tfrac\alpha2\|x-x_c\|_{\mathsf X}^2,\qquad x\in E,
\]
has a unique minimizer \(x_b\). It satisfies
\begin{equation}\label{eq:abstract-tikhonov-normal}
(Tx_b,Tv)_{\mathsf Y}+\alpha(x_b-x_c,v)_{\mathsf X}=(b,Tv)_{\mathsf Y}\qquad(v\in E).
\end{equation}
For any basis \(\{e_i\}\) of \(E\), the variational matrix \(H_{ij}=(Te_j,Te_i)_{\mathsf Y}+\alpha(e_j,e_i)_{\mathsf X}\) is symmetric positive definite. For every \(x^\dagger\in E\),
\begin{equation}\label{eq:abstract-tikhonov-comparison}
\|Tx_b-b\|_{\mathsf Y}^2+\alpha\|x_b-x_c\|_{\mathsf X}^2
\le\|Tx^\dagger-b\|_{\mathsf Y}^2+\alpha\|x^\dagger-x_c\|_{\mathsf X}^2.
\end{equation}
In particular,
\begin{equation}\label{eq:abstract-tikhonov-apriori}
\|Tx_b-b\|_{\mathsf Y}^2+\alpha\|x_b-x_c\|_{\mathsf X}^2\le\|b-Tx_c\|_{\mathsf Y}^2.
\end{equation}
For fixed \(T,x_c,\alpha\), the minimizers associated with \(b_1,b_2\in\mathsf Y\) satisfy
\begin{equation}\label{eq:abstract-tikhonov-data-stability}
\|x_{b_1}-x_{b_2}\|_{\mathsf X}\le\frac{\|b_1-b_2\|_{\mathsf Y}}{2\sqrt\alpha},\qquad
\|T(x_{b_1}-x_{b_2})\|_{\mathsf Y}\le\|b_1-b_2\|_{\mathsf Y}.
\end{equation}
If \(E\ne\{0\}\), set \(\sigma:=\inf_{0\ne v\in E}\|Tv\|_{\mathsf Y}/\|v\|_{\mathsf X}\ge0\). Then
\begin{align}\label{eq:abstract-tikhonov-comparator}
\|x_b-x^\dagger\|_{\mathsf X}
\le \min\left\{\frac1{2\sqrt\alpha},\frac1{\sqrt{\sigma^2+\alpha}}\right\}\|b-Tx^\dagger\|_{\mathsf Y}+\frac{\alpha}{\sigma^2+\alpha}\|x^\dagger-x_c\|_{\mathsf X}.
\end{align}
For \(\ell\in E^*\), the equation \((Te,Tv)_{\mathsf Y}+\alpha(e,v)_{\mathsf X}=\ell(v)\), \(v\in E\), has a unique solution satisfying
\begin{equation}\label{eq:abstract-tikhonov-load-stability}
\|e\|_{\mathsf X}\le\frac{\|\ell\|_{E^*}}\alpha,\qquad
\|Te\|_{\mathsf Y}\le\frac{\|\ell\|_{E^*}}{2\sqrt\alpha},
\end{equation}
where the dual norm on \(E^*\) is induced by \(\mathsf X\).

Finally, let \(\widetilde x\) be the minimizer with \((T,b)\) replaced by \((\widetilde T,\widetilde b)\), keeping \(E,x_c,\alpha\) fixed. With \(r=Tx_b-b\) and \(\ell_r(v)=(r,(\widetilde T-T)v)_{\mathsf Y}\),
\begin{equation}\label{eq:abstract-tikhonov-operator-perturbation}
\|\widetilde x-x_b\|_{\mathsf X}
\le\frac{\|\widetilde b-b-(\widetilde T-T)x_b\|_{\mathsf Y}}{2\sqrt\alpha}
+\frac{\|\ell_r\|_{E^*}}\alpha.
\end{equation}

The Hessian \(a_\alpha(u,v):=(Tu,Tv)_{\mathsf Y}+\alpha(u,v)_{\mathsf X}\) satisfies \(a_\alpha(v,v)\ge\alpha\|v\|_{\mathsf X}^2\). Hence the strictly convex functional has a unique minimizer, its first variation gives \eqref{eq:abstract-tikhonov-normal}, and \(\mathbf c^\top H\mathbf c=a_\alpha(\sum_i c_ie_i,\sum_i c_ie_i)>0\) for \(\mathbf c\ne0\). Minimality proves \eqref{eq:abstract-tikhonov-comparison}; choosing \(x^\dagger=x_c\) gives \eqref{eq:abstract-tikhonov-apriori}.

Subtracting two normal equations and testing with \(e=x_{b_1}-x_{b_2}\) yields \(\|Te\|_{\mathsf Y}^2+\alpha\|e\|_{\mathsf X}^2\le D\|Te\|_{\mathsf Y}\), where \(D=\|b_1-b_2\|_{\mathsf Y}\). The inequalities \(Dt-t^2\le D^2/4\) and \(\|Te\|_{\mathsf Y}^2\le D\|Te\|_{\mathsf Y}\) prove \eqref{eq:abstract-tikhonov-data-stability}. For \(B=T^*T\), the normal equation also gives
\[
x_b-x^\dagger=(B+\alpha I)^{-1}T^*(b-Tx^\dagger)-\alpha(B+\alpha I)^{-1}(x^\dagger-x_c).
\]
The eigenvalues of \(B\) are at least \(\sigma^2\). Moreover, for \(t\ge\sigma\), \(t/(t^2+\alpha)\le\min\{1/(2\sqrt\alpha),1/\sqrt{\sigma^2+\alpha}\}\). Spectral decomposition therefore proves \eqref{eq:abstract-tikhonov-comparator}. The same positive Hessian gives unique solvability of the load equation. Testing with \(e\) gives \(\|Te\|_{\mathsf Y}^2+\alpha\|e\|_{\mathsf X}^2\le\|\ell\|_{E^*}\|e\|_{\mathsf X}\); dropping the first term and completing the square in \(\|e\|_{\mathsf X}\) prove \eqref{eq:abstract-tikhonov-load-stability}.

For \(e=\widetilde x-x_b\), subtraction of the normal equations gives
\[
(\widetilde Te,\widetilde Tv)_{\mathsf Y}+\alpha(e,v)_{\mathsf X}
=(\widetilde b-b-(\widetilde T-T)x_b,\widetilde Tv)_{\mathsf Y}-\ell_r(v).
\]
Split \(e\) into the solutions associated with the two terms on the right. Apply \eqref{eq:abstract-tikhonov-data-stability} to the first and \eqref{eq:abstract-tikhonov-load-stability} to the second, both with operator \(\widetilde T\). Their sum gives \eqref{eq:abstract-tikhonov-operator-perturbation}.
