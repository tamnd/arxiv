<!-- theorem-3: body_md -->
\label{thm:equality case}
Let $X$ be a minimal projective $3$-fold with $\kappa(X)=2$ such that $p_g(X)\geq 2$ and \[
\operatorname{Ivol}(X)= \frac{3}{4}p_g(X)-\frac{5}{4}.
\] 
Then one of the following statements holds:
\begin{enumerate}

 \item $p_g(X)=3$ and $|K_X|$ is base point free and induces a morphism \[\Phi_{|K_X|}: X\to \mathbb{P}^{2}\] which coincides with the Iitaka fibration of $X$;

 \item $p_g(X)\geq 3$ and the following properties hold:
\begin{enumerate}
\item $X$ is Gorenstein and $h^1(X, \mathcal{O}_{X})=h^2(X, \mathcal{O}_{X})=0$;
\item $|2K_X|$ is base point free with $h^0(X, 2K_X)=3p_g(X)-3$ and induces a morphism \[\Phi_{|2K_X|}: X\to T_2\subseteq \mathbb{P}^{3p_g(X)-4}\] where $T_2$ is a surface of minimal degree $\deg T_2=3p_g(X)-5$ and $\Phi_{|2K_X|}$ coincides with the Iitaka fibration of $X$;

\item $|K_X|$ is composed with a pencil of surfaces $G$ with \[\operatorname{Ivol}(G)=\begin{cases}\frac{1}{2}  \text{ or } \frac34 &\text{if } p_g(X)=3;\\
                                                       
\frac{1}{2} &\text{if } p_g(X)\geq 4.
\end{cases}\]
\end{enumerate}
\end{enumerate}

<!-- definition-1: body_md -->
Let $X$ be a normal projective variety with at worst terminal singularities.
\begin{enumerate}
 \item The {\it geometric genus} of $X$ is defined as 
\[
p_g(X)\coloneqq h^0(X,K_X).
\]
\item The {\it Kodaira dimension} $\kappa(X)$ of $X$ is defined as follows: if $|mK_X|\neq \emptyset$ for some positive integer $m$, then 
\begin{equation*}
 \kappa(X)\coloneqq 
 \max \{\dim\Phi_{|mK_X|}(X)\mid m\in\mathbb{Z}_{\geq1}\},
\end{equation*}
where $\Phi_{|mK_X|}$ is the rational map defined by $|mK_X|$; otherwise, $\kappa(X)\coloneqq -\infty$. 
\end{enumerate}
<!-- end definition-1 -->
<!-- definition-2: body_md -->
Let $X$ be a normal projective variety with at worst terminal singularities and with $\kappa(X)=\kappa> 0$.
The {\it Iitaka volume} $\operatorname{Ivol}(X)$ of $X$ is defined as
\begin{equation}
\operatorname{Ivol}(X)=\limsup_{m\to +\infty} \frac{h^0(X,mK_X)}{m^{\kappa}/\kappa!}. \label{Ivol}
\end{equation}
<!-- end definition-2 -->
<!-- definition-3: body_md -->
Let $X$ be a normal projective variety with at worst terminal singularities such that $\kappa(X)> 0$ and $K_X$ is semiample. Then there is a fibration 
\[
f:X\to Z
\]
and 
an ample $\mathbb{Q}$-divisor $L$ on $Z$ such that $K_X\sim_{\mathbb{Q}}f^*L$. In this case,
we have $\dim Z=\kappa(X)$ and
\[
\operatorname{Ivol}(X)=L^{\dim Z}.
\]
The fibration $f:X\to Z$ is called the {\it Iitaka fibration} of $X$.
<!-- end definition-3 -->
<!-- setting-1: body_md -->
\label{setting}Let $X$ be a minimal projective $3$-fold with 
$\kappa(X)>0$ and let $f:X\to Z$ be the Iitaka fibration with $K_X\sim_{\mathbb{Q}} f^*L$ for some ample $\mathbb{Q}$-divisor $L$ on $Z$. 
For a positive integer $m$ with $h^0(X, mK_X)\geq 2$,
consider the $m$-canonical map 
\[
\Phi_{|mK_X|}:X\dashrightarrow T_m\subseteq \mathbb{P}^{h^0(X, mK_X)-1}
\]
defined by the linear system $|mK_X|$, where $T_m$ is the closure of its image.
As $mK_X$ is numerically trivial and effective on a general fiber of $f$, it is linearly trivial on a general fiber of $f$. Hence
$\Phi_{|mK_X|}$ factors through $f$. We will construct a commutative diagram as follows: 
\begin{equation*}
\begin{tikzcd}

<!-- lemma-3: body_md -->
\label{lem:compare KX KG}
Let $X$ be a minimal projective $3$-fold.
Let $\phi:X'\to X$ be a projective birational morphism where $X'$ is terminal, such that there is a fibration $g:X'\to \Gamma$ where $\Gamma$ is a smooth projective curve. 
Let $G$ be a general fiber of $g$ and $\sigma:G\to G_0$ the morphism onto its minimal model. 

\begin{enumerate}
 \item If $p_g(\Gamma)>0$, then $\phi^*K_X|_G=\sigma^*K_{G_0}$;


 \item If $p_g(\Gamma)=0$ and $\phi^*K_X-bG$ is $\mathbb{Q}$-effective for some positive rational number $b$, then 
$ \phi^*K_X|_G-\frac{b}{b+1}\sigma^*K_{G_0}$ is 
$\mathbb{Q}$-effective.


 
 \item If $p_g(X)\geq 2$ and $\operatorname{Mov} |\lfloor \phi^*K_X\rfloor|$ is base-point free and induces the fibration $g$, then 
\[
p_g(X) \phi^*K_X|_G\geq (p_g(X)-1) \sigma^*K_{G_0}.
\]
To be more precise, there exist effective divisors $D\in |K_X|$ and $D_0\in |K_{G_0}|$ such that 
\[
p_g(X) \phi^*D|_G-(p_g(X)-1) \sigma^*D_0
\]
is an effective $\mathbb{Q}$-divisor.
\end{enumerate}
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
(1) By \cite{hacon2007}, the fibers of $\phi$ are rationally chain connected, which implies that $g$ descends to a morphism $g_0: X\to \Gamma$.
As $K_X$ is nef, a general fiber of $g_0$ is a minimal projective surface, which is just the minimal model $G_0$ of $G$. Hence we have $K_X|_{G_0}=K_{G_0}$ by the adjunction, which gives the desired equality by pullback.

\medskip

By (1), Assertion (3) holds automatically when 
$p_g(\Gamma)>0$. 
From now on, we may assume that $p_g(\Gamma)=0$.
By taking higher models, it is clear that the assertions are independent of the choice of $\phi: X'\to X$.
So we may assume that 
$\phi$ is the birational projective morphism constructed in \cite[Proposition~3.1]{CHJ25} satisfying that 
\begin{align}
 \phi^*(K_X+\phi_*G)-K_{X'}-G=\widetilde{E}\geq 0\label{eq: K+G>0}
\end{align}
is an effective $\phi$-exceptional $\mathbb{Q}$-divisor.

\medskip



(2) By assumption, $\phi^*K_X-bG$ is $\mathbb{Q}$-effective. In particular, $K_X-b\phi_*G$ is $\mathbb{Q}$-effective. 
So by \eqref{eq: K+G>0}, 
$(1+\frac1b)\phi^*K_X-K_{X'}-G$
is $\mathbb{Q}$-effective.
Hence $ (1+\frac1b)\phi^*K_X|_G- \sigma^*K_{G_0}$ is 
$\mathbb{Q}$-effective by restricting on $G$.

\medskip

(3) 
Write $d=p_g(X)-1$, we have $K_X=d\phi_*G+\Delta$, where $\Delta$ is the fixed part of $|K_X|$. Take a general fiber $G'$ of $g$ different from $G$, 
then $D=d\phi_*G'+\Delta\in|K_X|$. Write $K_{X'}=\phi^*K_X+E_{\phi}$, then $\phi^*D+E_{\phi}\in |K_{X'}|$ and $(\phi^*D+E_{\phi})|_{G}\in |K_{G}|$.
Take $D_0=\sigma_*((\phi^*D+E_{\phi})|_{G})\in |K_{G_0}|$, then 
\begin{align}
(\phi^*D+E_{\phi})|_{G}-\sigma^*D_0=K_{G}-\sigma^*K_{G_0}\geq 0 \label{eq:phiD+E-D>0}
\end{align}
is an effective $\mathbb{Q}$-divisor. On the other hand, by \eqref{eq: K+G>0}, we have
\[
\phi^*(D+\phi_*G')=\phi^*D+E_{\phi}+G'+\widetilde{E}.
\]
Hence
\[
(d+1)\phi^*D\geq d\phi^*(D+\phi_*G')=d(\phi^*D+E_{\phi}+G'+\widetilde{E})\geq d(\phi^*D+E_{\phi}).
\]
Restricting both sides on $G$, we conclude by \eqref{eq:phiD+E-D>0} that 
\[
(d+1)\phi^*D|_{G}-d\sigma^*D_0\geq 0.
\]
This concludes the proof.

\label{prop:Noehter pencil}

Keep the notation in Setting~\ref{setting}. 
Recall we have the following commutative diagram:
\begin{equation*}
\begin{tikzcd}

\end{tikzcd}

Let $F$ be a general fiber of $g'_m$, which is a smooth projective curve by Bertini's theorem. Note that $f': X'\to Z'$ induces an elliptic fibration $f_G: G\to F$ between general fibers of $g_m$ and $g'_m$.

$M_m \equiv aG$ and $L'_m\equiv a F$ with \begin{align}

\end{align}

\end{align}

$f_G^*(\phi'^*L|_F)=\phi^*K_X|_G$.
So $f_G^*(\phi'^*L|_F)-\frac{a}{a+m}\sigma^*f_0^*L_0$ is $\mathbb{Q}$-effective, which implies that

\deg(\phi'^*L|_F)\geq \frac{a}{a+m}\deg(L_0)=\frac{a}{a+m} \operatorname{Ivol}(G_0)= \frac{a}{a+m} \operatorname{Ivol}(G).\label{eq: deg L >= IvolG}
\end{align}
So we get the desired inequality by combining with \eqref{eq: a>p-1} and \eqref{eq:L2>kLF}.

\label{lem: Ivol S<1}
 Let $G$ be a smooth projective surface with $\kappa(G)=1$. 
 Assume that $p_g(G)>0$ and $\operatorname{Ivol}(G)<1$. Then the following hold:
 \begin{enumerate}
 \item $p_g(G)=1$, $h^1(G, \mathcal{O}_{G})=0$, $h^0(G, 2K_G)=2$, and $\operatorname{Ivol}(G)=1-\frac{1}{m_1}$ for an integer $m_1\geq 2$;
 
 \item the Iitaka fibration of $G_0$ has exactly one multiple fiber $m_1R$ with multiplicity $m_1$ and $(m_1-1)R\in |K_{G_0}|$ is the unique element, where $G_0$ is the minimal model of $G$. 
 \end{enumerate}
<!-- end lemma-4 -->
<!-- proof-7: body_md -->
Without loss of generality, we may assume that $G$ is minimal. 
Denote by $f_G:G\to F$ the Iitaka fibration.
By Kodaira's canonical bundle formula \cite[Theorem~V.12.1]{BHPV2004}, we have 
\begin{align}
K_{G}\sim f_G^*\left(K_F+L_F+\sum_{i=1}^r\frac{m_i-1}{m_i}P_i\right),\label{eq:cbf elliptic fib}
\end{align}
where $L_F$ is a divisor on $F$ such that $\mathcal{O}_F(L_F)=(R^1{f_G}_*\mathcal{O}_{G})^{\vee}$, $P_i$ is a prime divisor on $F$ and $f_G^{-1}(P_i)$ is a multiple fiber of $f_G$ with multiplicity $m_i$ for each $1\leq i\leq r$. By the relative duality 
and \cite[Proposition~V.12.2, Theorem~III.18.2]{BHPV2004}, we have \[\chi(G, \mathcal{O}_{G})=\deg L_F=\deg {f_G}_*\omega_{G/F}\geq 0.\] In particular, 
\[
\operatorname{Ivol}(G)=2p_g(F)-2+\chi(G, \mathcal{O}_{G})+\sum_{i=1}^r\frac{m_i-1}{m_i}.
\]
Since $\operatorname{Ivol}(G)<1$, we have $p_g(F)\leq 1$. 


Suppose that $p_g(F)=1$. Then we have $\chi(G, \mathcal{O}_G)=0$ and $r=1$ by $\operatorname{Ivol}(G)<1$. 
Note that $h^1(G, \mathcal{O}_{G})=1+p_g(G)-\chi(G, \mathcal{O}_G)\geq 2$, so by \cite[Lemma~1.6]{Serrano1991}, we have $h^1(G, \mathcal{O}_{G})=2=p_g(F)+1$. But we get a contradiction by \cite[Proposition~5.6]{Serrano1996} as $f_G$ cannot have exactly one multiple fiber, namely, $r\neq 1$.


Suppose that $p_g(F)=0$. 
                                                                                                                     
Then by \cite[Lemma~1.6]{Serrano1991}, either $h^1(G, \mathcal{O}_{G})=p_g(F)=0$, or $\chi(G, \mathcal{O}_{G})=0$ and $h^1(G, \mathcal{O}_{G})=p_g(F)+1=1$.
In the latter case, 
\[0=\chi(G, \mathcal{O}_{G})=p_g(G)-h^1(G, \mathcal{O}_{G})+1=p_g(G)>0,\] which is absurd.
So we have $h^1(G, \mathcal{O}_{G})=p_g(F)=0$ and $\chi(G, \mathcal{O}_{G})=p_g(G)+1\geq 2$. Then $\operatorname{Ivol}(G)<1$ implies that \[\chi(G, \mathcal{O}_{G})=2, \quad \operatorname{Ivol}(G)=\frac{m_1-1}{m_1}, \quad r=1 .\]
In particular, $f_G$ has exactly one multiple fiber $m_1R$ with multiplicity $m_1$. Then by \eqref{eq:cbf elliptic fib}, $K_G\sim (m_1-1)R$ and $2K_G\sim (2m_1-2)R$, which implies that $p_g(G)=1$ and $h^0(G, 2K_G)=2$.

<!-- lemma-5: body_md -->
\label{lem:P2 >= 3pg-3}
Let $X$ be a minimal projective $3$-fold with $\kappa(X)=2$. 
Assume that $p_g(X)\geq 2$ and that $|K_X|$ is composed with a pencil of surfaces $G$ with $p_g(G)=1$ and $h^1(G, \mathcal{O}_{G})=0$. Then
\[
h^0(X, 2K_X)\geq 3p_g(X)-3.
\]
The equality holds only if $X$ is Gorenstein and $h^1(X, \mathcal{O}_{X})=h^2(X, \mathcal{O}_X)=0$.
<!-- end lemma-5 -->
<!-- proof-8: body_md -->
First, we show that $h^2(X, 2K_X)=0$. 
The proof is well-known to experts (see \cite[Lemma~8.5]{Miyaoka1987}), but we reproduce it here for completeness. 
Take a general sufficiently ample Cartier divisor $S$ on $X$, we may assume that $S$ is a smooth projective surface by Bertini's theorem as $X$ has only isolated singularities by \cite[Corollary~5.18]{Kollar-Mori-1998}. 
By the exact sequence 
\[
 H^1(S,K_S+K_X|_S)\to H^2(X, 2K_X)\to H^2(X, 2K_X+S),
\]
it suffices to show that \[H^1(S,K_S+K_X|_S)=H^2(X, 2K_X+S)=0.\] 
It is clear that $H^2(X, 2K_X+S)=0$
by the Serre vanishing 
since $S$ is sufficiently ample. 
On the other hand,
denote by $f: X\to Z$ the Iitaka fibration of $X$ with $K_X\sim_\mathbb{Q}f^*L$, then the induced morphism $f_S: S\to Z$ is generically finite and 
$K_X|_S\sim_\mathbb{Q} f_S^*L$ is nef and big on $S$, which implies that $H^1(S,K_S+K_X|_S)=0$ by the Kawamata--Viehweg vanishing theorem \cite[Theorem~1-2-3]{KMM87}. Therefore, $h^2(X, 2K_X)=0$.
In particular, $h^0(X, 2K_X)\geq \chi(X, \mathcal{O}_X(2K_X)).$

As $K_X^3=0$, by Reid's Riemann--Roch formula \cite[Corollary~(10.3)(5)]{Reid-1987-YPG}, we have 
\begin{align}
\chi(X, \mathcal{O}_X(2K_X))\geq-3\chi(X, \mathcal{O}_X)=3\chi(X, \omega_X).\label{eq: 2K=3O}
\end{align}
So it suffices to show that $\chi(X, \omega_X)\geq p_g(X)-1$. 

Keep the notation in Setting~\ref{setting} for $m=1$ with the subscript omitted. Take $G$ to be a general fiber of $g: X'\to \Gamma$. 
By \cite[Theorem~2.1]{Kollar1986-I}, for each $i\geq 0$, $R^ig_*\omega_{X'}$ is torsion free, hence locally free since $\Gamma$ is a smooth curve. In particular, since $p_g(G)=1$ and $h^1(G, \mathcal{O}_{G})=0$, $g_*\omega_{X'}$ is a line bundle and $R^1g_*\omega_{X'}=0$ by \cite[Corollary~III.12.9]{Hartshorne1977}. 
Also we have 
$R^2g_*\omega_{X'}=\omega_{\Gamma}$ by the relative duality. 
Then by \cite[Corollary~3.2]{Kollar1986-II}, 
we have
\begin{align*}
\chi(X, \omega_X)=\chi(X', \omega_{X'})=& \chi(\Gamma, g_*\omega_{X'})-\chi(\Gamma, R^1g_*\omega_{X'})+\chi(\Gamma, R^2g_*\omega_{X'})\\
=& \chi(\Gamma, g_*\omega_{X'})+\chi(\Gamma, \omega_\Gamma)\\
=&\deg(g_*\omega_{X'})\\
\geq & p_g(X)-1.
\end{align*}
Here the last equality follows from the Riemann--Roch formula and the inequality follows from Lemma~\ref{lem:deg>=h0-1}. 

Finally, if the equality holds, then the equality in \eqref{eq: 2K=3O} holds, which implies that $X$ is Gorenstein by \cite[Corollary~(10.3)(5)]{Reid-1987-YPG} again. Also $\deg(g_*\omega_X)=p_g(X)-1$ implies that $p_g(\Gamma)=0$ by Lemma~\ref{lem:deg>=h0-1} and $h^1(X, \mathcal{O}_{X})=h^2(X, \mathcal{O}_X)$. 
A standard spectral sequence argument (see \cite[\S 2.6]{EXPII}) shows that
\[
h^1(X, \mathcal{O}_X)=p_g(\Gamma)+h^1(\Gamma, R^1g_*\omega_{X'})=0. \qedhere
\]

<!-- proof-9: body_md -->
Keep the notation in Setting~\ref{setting} for $m=1$ with the subscript omitted. Take $G$ to be a general fiber of $g: X'\to \Gamma$. 
By Remark~\ref{rem:pencil}, $\kappa(G)=1$. 
Also $p_g(X)>0$ implies that $p_g(G)>0$ by the adjunction formula $K_{X'}|_G=K_G$. 
So $G$ satisfies the conditions of Lemma~\ref{lem: Ivol S<1}. 

Write $p_g=p_g(X)$ for simplicity. By Remark~\ref{rem:pencil}, we may write 
\[\phi^*K_X\sim_{{\mathbb Q}} \sum_{j=1}^{a}G_j+E\equiv aG+E\]
where $a\geq p_g-1$, each $G_j$ is a general fiber of $g$ with \[\sum_{j=1}^{a}G_j=M_1\in \operatorname{Mov}|\lfloor\phi^*K_X\rfloor|,\] and $E$ is an effective ${\mathbb Q}$-divisor on $X'$. 
\begin{claim}\label{claim1}
 The ${\mathbb Q}$-divisor 
\[
N\coloneq \phi^*K_X-\frac{1}{p_g-1}E-G
\]
is semiample and $mN-G$ is $\mathbb{Q}$-effective for a sufficiently large integer $m$. 
\end{claim}
\begin{proof}
 If $p_g(\Gamma)=0$, then by Remark~\ref{rem:pencil}, we have $a=p_g-1$ and that 
 \[N\sim_\mathbb{Q}\left(1-\frac1{p_g-1}\right)\phi^*K_X\]
 is a semiample $\mathbb{Q}$-divisor. 

 If $p_g(\Gamma)>0$, then by Remark~\ref{rem:pencil}, we have $a>p_g-1$ and
 \[N\sim_\mathbb{Q}\left(1-\frac1{p_g-1}\right)\phi^*K_X+\frac{1}{p_g-1}(M_1-(p_g-1)G)\]
is a semiample $\mathbb{Q}$-divisor, as $\phi^*K_X$ and $M_1-(p_g-1)G$ are both semiample. Here recall that $M_1-(p_g-1)G$ is the pull-back of an ample $\mathbb{Q}$-divisor on $\Gamma$. 

Finally, in either case, $mN-G$ is $\mathbb{Q}$-effective for a sufficiently large integer $m$ as $\phi^*K_X-G$ is $\mathbb{Q}$-effective.
\end{proof}
After replacing $X'$ by a higher model, we may assume that $N$ has simple normal crossing support. 
By Claim~\ref{claim1} and \cite[Theorem~6-1-6]{KMM87}, the natural map
\begin{align*}
& H^1\left(X',K_{X'}+\left\lceil \phi^*K_X-\frac1{p_g-1}E-G \right\rceil\right)\\\to &H^1\left(X',K_{X'}+\left\lceil \phi^*K_X-\frac1{p_g-1}E \right\rceil\right)
\end{align*}
is injective, which implies that the natural map
\begin{align}\label{eq:K+K-tE}
H^0\left(X',K_{X'}+\left\lceil \phi^*K_X-\frac1{p_g-1}E\right\rceil\right)\to H^0(G,D)
\end{align}
is surjective,
where
\[D\coloneqq \left(K_{X'}+\left\lceil \phi^*K_X-\frac1{p_g-1}E\right\rceil\right)|_G=K_G+\left\lceil\phi^*K_X|_G-\frac1{p_g-1}E|_G\right\rceil,\] here 
the restriction on $G$
commutes with the roundup 
since $G$ is general.

Let $\sigma:G\to G_0$ be the morphism to the minimal model. By Lemma~\ref{lem:compare KX KG}(3), 
we know that $\phi^*K_X|_G-\frac{p_g-1}{p_g}\sigma^*K_{G_0}$ is an effective $\mathbb{Q}$-divisor where we regard $\phi^*K_X$ and $\sigma^*K_{G_0}$ as well-chosen effective divisors as in Lemma~\ref{lem:compare KX KG}(3). 
Then
\begin{align*}
&\phi^*K_X|_G-\frac1{p_g-1}E|_G\geq \left(1-\frac1{p_g-1}\right)\phi^*K_X|_G\\
\geq &\left(1-\frac{1}{p_g-1}\right)\frac{p_g-1}{p_g}\sigma^*K_{G_0}= \left(1-\frac{2}{p_g}\right)\sigma^*K_{G_0}. 
\end{align*}
So we have
\begin{align}\label{eq D>2K+K}
D\geq K_G+\left\lceil \left(1-\frac{2}{p_g}\right)\sigma^*K_{G_0}\right\rceil=2\sigma^*K_{G_0}+K_{G/G_0}-\left\lfloor\frac{2}{p_g}\sigma^*K_{G_0}\right\rfloor.
\end{align}
By Lemma~\ref{lem: Ivol S<1}, $K_{G_0}\sim R$ where $2R$ is the unique multiple fiber of the Iitaka fibration of $G_0$. Here we used the condition $\operatorname{Ivol}(G)<\frac23$ to guarantee that $m_1=2$ in Lemma~\ref{lem: Ivol S<1}.
By Kodaira's classification of singular fibers \cite[V.7]{BHPV2004}, $R$ is of type $I_0$, $I_1$ or $I_b$ with $b\geq2$, which means that $R$ is an analytically normal crossing divisor.
Hence by \cite[Example~9.2.13]{lazarsfeld2004-II} and $p_g\geq 3$, we have 
\[
\sigma_*\mathcal{O}_G\left(K_{G/G_0}-\left\lfloor\frac{2}{p_g}\sigma^*R\right\rfloor\right)=\mathcal{J}\left(\frac{2}{p_g}R\right)=\mathcal{O}_{G_0}.
\]
Therefore, by the projection formula and Lemma~\ref{lem: Ivol S<1}, \eqref{eq D>2K+K} implies that 
\[
h^0(G, D)\geq h^0(G_0, 2K_{G_0})=2.
\]

Suppose that $|2K_X|$ is composed with a pencil of surfaces, then it is composed with the same pencil as $|K_X|$, which induces the same fibration $g:X'\to \Gamma$. But by the surjectivity of \eqref{eq:K+K-tE} and $h^0(G, D)\geq 2$, we know that $\operatorname{Mov}|\lfloor\phi^*(2K_X)\rfloor|$ induces a non-trivial movable linear system on $G$, a contradiction.

For integers $p\geq 0$ and $n\geq 2$, there is an infinite sequence of smooth minimal projective $n$-folds $\{X_i\}$ with $\kappa(X_i)=1$ and $p_g(X_i)=p$ such that $\lim_{i\to \infty}\operatorname{Ivol}(X_i)=\infty$.

<!-- proof-16: body_md -->
By \cite[Theorem~4.16]{CHL2025}, there exists a sequence of minimal projective surfaces $\{S_i\}$ such that $p_g(S_i)=p$, $\kappa(S_i)=1$, and $\lim_{i\to \infty}\operatorname{Ivol}(S_i)=\infty$. So these provide examples for $n=2$. If $n>2$,
let $A$ be an Abelian variety of dimension $n-2$ and let $X_i\coloneqq A\times S_i$. 
Then $X_i$ is a smooth minimal projective $n$-fold with $\kappa(X_i)=1$.
By the K\"unneth formula, we have $p_g(X_i)=p$ and
\[
\operatorname{Ivol}(X_i)=\limsup_{m\to \infty} \frac{h^0(A,\omega_A^{\otimes m})\cdot h^0(S_i,\omega_{S_i}^{\otimes m})}{m}= \operatorname{Ivol}(S_i).
\]
This gives the desired sequence of examples.
