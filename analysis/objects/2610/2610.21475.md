<!-- proposition-1: body_md -->
\label{prop-f-ent}
Suppose that $\omega_+\in\Sigma_+$ is such that, for some $\Psi_+\in{\mathcal B}_\mathrm{b}$, one has 
$\omega_+\in{\mathcal S}_\mathrm{eq}(\Psi_+)$ and $E(\Psi_+)\in{\mathfrak C}(\Phi)$. Then 
\[
s(\omega_+)=\max\{s(\nu)\mid\nu\in{\mathcal S}_{\omega,\Phi}\}.
\]
<!-- end proposition-1 -->
<!-- head-1: body_md -->
Let $\nu\in{\mathcal S}_{\omega,\Phi}$. Since $E(\Psi_+)\in{\mathfrak C}(\Phi)$, the definition of
${\mathcal S}_{\omega,\Phi}$ and the fact that $\omega_+\in\Sigma_+$ give
\[
\nu(E(\Psi_+))=\omega(E(\Psi_+))=\omega_+(E(\Psi_+)).
\]
By the Gibbs variational principle, $s(\nu)-\nu(E(\Psi_+))\leq P(\Psi_+)$. 
On the other hand, since $\omega_+\in{\mathcal S}_\mathrm{eq}(\Psi_+)$, $P(\Psi_+)=s(\omega_+)-\omega_+(E(\Psi_+))$. 
Combining the last three relations yields $s(\nu)\leq s(\omega_+)$. 
Since $\omega_+\in{\mathcal S}_{\omega,\Phi}$, the maximum is attained at
$\omega_+$. \hfill$\Box$
<!-- end head-1 -->

$\Psi_+\in{\mathcal B}_\mathrm{sd}$. If the pair $(\Phi,\Psi_+)$ has the

\label{thm:dGvp}
Let $\omega_+\in\Sigma_+(\omega,\Phi)$ be such that
$\omega_+\in{\mathcal S}_\mathrm{eq}(\Psi_+)$ with $[\Psi_+]\in\mathcal{C}_\mathrm{phys}(\Phi)$. Then, one has
\[
\inf_{\nu\in{\mathcal S}_\mathrm{phys}(\Phi)}s(\omega|\nu)=s(\omega_+)-s(\omega),
\]
and the set of minimizers is precisely ${\mathcal S}_\mathrm{eq}(\Psi_+)$.

\label{thm-fin-r-mar}
Suppose that $d=1$, that $\Phi,\Psi_0\in{\mathcal B}_{\rm f}$,
$\omega\in{\mathcal S}_\mathrm{eq}(\Psi_0)$, and that $(\omega,\Phi)$ has the property of
finite-range restoration of locality. Then, the following hold:

\item $\Sigma_+(\omega,\Phi)=\{\omega_+\}$.
\item If $\omega_+\in{\mathcal S}_\mathrm{eq}(\Psi_+)$ with $\Psi_+\in{\mathcal B}_{\rm f}$, then
$[\Psi_+]\in\mathcal{C}_\mathrm{phys}(\Phi)$.
\item The dynamical Gibbs variational principle holds and $\omega_+$ is its
unique minimizer.
\item $s(\omega_+)>s(\omega)$.
\end{romanenumerate}

<!-- proposition-2: body_md -->
\label{prop:vanish_on_average}
For $A\in{\mathfrak A}_\mathrm{sa}$, the following conditions are equivalent. 
\begin{romanenumerate}
\item $A\in{\mathfrak I}$.
\item $\displaystyle\lim_{\{\Lambda\}}\|m_\Lambda(A)\|=0$.
\item $\varphi(A)=0$ for all $\varphi\in{\mathfrak A}^\ast_I$.
\item $\omega(A)=0$ for all $\omega\in{\mathcal S}_\mathrm{I}$.
\end{romanenumerate}
<!-- end proposition-2 -->
<!-- head-2: body_md -->
We shall use the following notation and properties, $C$ denoting a subset of a Banach space $B$,
and $D$ a subset of its dual $B^\ast$. The annihilators of $C$ and $D$ are, respectively,
\begin{align*}
C^\perp&\coloneq\{f\in B^\ast\mid f(x)=0\text{ for all }x\in C\},\\[4pt]
\leftindex^\perp D&\coloneq\{x\in B\mid f(x)=0\text{ for all }f\in D\}.
\end{align*}
We note that $C^\perp$ is a $\mathrm{w}^\ast\!\!\mathrm{-closed}$ subspace of $B^\ast$ and $\leftindex^\perp D$
a norm-closed subspace of $B$. We recall also that $\leftindex^\perp(C^\perp)$ is the
norm closure of the linear span of $C$ while $(\leftindex^\perp D)^\perp$ is the
$\mathrm{w}^\ast\!\!\mathrm{-closure}$ of the linear span of $D$; see~\cite[Theorem~4.7]{Rudin1973}.

\medskip\noindent\textbf{(i)} $\Rightarrow $\textbf{(ii)} For any $A\in{\mathfrak A}$ 
and $a\in{\mathbbm Z}^d$ one has
$$
\left\|m_\Lambda\left(A-\varphi^a(A)\right)\right\|
\le\frac1{|\Lambda|}\sum_{x\in\Lambda\Delta(\Lambda+a)}\|\varphi^x(A)\|
\le\frac{|\Lambda\Delta(\Lambda+a)|}{|\Lambda|}\|A\|=O(|\Lambda|^{-1/d}),
$$
and since $m_\Lambda$ is a linear map such that $\|m_\Lambda(A)\|\le\|A\|$, 
the result follows from an $\epsilon/3$ argument.

\medskip\noindent\textbf{(ii)} $\Rightarrow $\textbf{(iii)} Immediately follows 
from the fact that $\varphi(A)=\varphi(m_\Lambda(A))$. 

\medskip\noindent\textbf{(iii)} $\Rightarrow $\textbf{(i)} 
By assumption $A\in\leftindex^\perp{\mathfrak A}_I^\ast$. Observing that ${\mathfrak I}_0^\perp\subset{\mathfrak A}_I^\ast$, 
we conclude ${\mathfrak I}=\leftindex^\perp({\mathfrak I}_0^\perp)\supset\leftindex^\perp{\mathfrak A}_I^\ast\ni A$.

\medskip\noindent\textbf{(iii)} $\Leftrightarrow $ \textbf{(iv)} Since
${\mathfrak A}^\ast_I$ is the linear span of ${\mathcal S}_\mathrm{I}$,
one has $\leftindex^\perp{\mathcal S}_I=\leftindex^\perp{\mathfrak A}_I^\ast$.\hfill$\Box$

<!-- proposition-3: body_md -->
\label{prop:seminorm_interpretation}
For any $A\in{\mathfrak A}_\mathrm{sa}$, one has
\begin{gather}
\lim_{\{\Lambda\}}\left\|m_\Lambda(A)\right\|=\|A\|_{{\mathfrak A}_\mathrm{sa}/{\mathfrak I}},\label{eq:overI}\\
\lim_{\{\Lambda\}}\left\|m_\Lambda(A)\right\|_{{\mathfrak A}_\mathrm{sa}/{\mathbbm R}}=\|A\|_{{\mathfrak A}_\mathrm{sa}/({\mathfrak I}+{\mathbbm R})}\label{eq:overIplusR}.
\end{gather}
<!-- end proposition-3 -->
<!-- head-3: body_md -->
Let $A\in{\mathfrak A}_\mathrm{sa}$. On the one hand,
\begin{equation}
m_\Lambda(A)=A+\frac1{|\Lambda|}\sum_{x\in\Lambda}\left(\varphi^x(A)-A\right)=A+A_\Lambda,
\label{eq:mLambdaDecomp}
\end{equation}
where $A_\Lambda\in{\mathfrak I}$. It follows that $\|m_\Lambda(A)\|=\|A+A_\Lambda\|\ge\|A\|_{{\mathfrak A}_\mathrm{sa}/{\mathfrak I}}$, 
and hence
\begin{equation}
\liminf_{\{\Lambda\}}\|m_\Lambda(A)\|\ge\|A\|_{{\mathfrak A}_\mathrm{sa}/{\mathfrak I}}.
\label{eq:overIlow}
\end{equation}
On the other hand, for any $\epsilon>0$ there exists $B_\epsilon\in{\mathfrak I}$ such that 
$\|A-B_\epsilon\|<\|A\|_{{\mathfrak A}_\mathrm{sa}/{\mathfrak I}}+\epsilon$. We deduce that
$$
\|m_\Lambda(A)\|\le\|m_\Lambda(A-B_\epsilon)\|+\|m_\Lambda(B_\epsilon)\|
\le\|A-B_\epsilon\|+\|m_\Lambda(B_\epsilon)\|
\le\|A\|_{{\mathfrak A}_\mathrm{sa}/{\mathfrak I}}+\epsilon+\|m_\Lambda(B_\epsilon)\|,
$$
and hence, by Proposition~\ref{prop:vanish_on_average},
$$
\limsup_{\{\Lambda\}}\|m_\Lambda(A)\|\le\|A\|_{{\mathfrak A}_\mathrm{sa}/{\mathfrak I}}+\epsilon.
$$
Letting $\epsilon\downarrow0$, the last estimate combined with~\eqref{eq:overIlow}
gives~\eqref{eq:overI}.

To prove~\eqref{eq:overIplusR} we derive from~\eqref{eq:mLambdaDecomp}
$$
\|m_\Lambda(A)\|_{{\mathfrak A}_\mathrm{sa}/{\mathbbm R}}=\inf_{b\in{\mathbbm R}}\|A+A_\Lambda-b\mathbbm{1}\|
\ge\inf_{B\in{\mathfrak I},b\in{\mathbbm R}}\|A-B-b\mathbbm{1}\|=\|A\|_{{\mathfrak A}_\mathrm{sa}/({\mathfrak I}+{\mathbbm R})},
$$
and hence
\begin{equation}
\liminf_{\{\Lambda\}}\|m_\Lambda(A)\|_{{\mathfrak A}_\mathrm{sa}/{\mathbbm R}}\ge\|A\|_{{\mathfrak A}_\mathrm{sa}/({\mathfrak I}+{\mathbbm R})}.
\label{eq:overIplusRlow}
\end{equation}
To obtain the opposite estimate, we note that for  any $\epsilon>0$ 
there exists $B_\epsilon\in{\mathfrak I}$ and $b_\epsilon\in{\mathbbm R}$ such that 
$\|A-B_\epsilon-b_\epsilon\mathbbm{1}\|<\|A\|_{{\mathfrak A}_\mathrm{sa}/({\mathfrak I}+{\mathbbm R})}+\epsilon$. It follows that
\begin{align*}
	\|m_\Lambda(A)\|_{{\mathfrak A}_\mathrm{sa}/{\mathbbm R}}
	&=\inf_{b\in{\mathbbm R}}\|m_\Lambda(A-B_\epsilon-b_\epsilon\mathbbm{1})+m_\Lambda(B_\epsilon)+(b_\epsilon-b)\mathbbm{1}\|\\[4pt]
	&\le\inf_{b\in{\mathbbm R}}\left(\|A-B_\epsilon-b_\epsilon\mathbbm{1}\|+\|m_\Lambda(B_\epsilon)\|+|b_\epsilon-b|\right)\\[4pt]
	&\le\|A\|_{{\mathfrak A}_\mathrm{sa}/({\mathfrak I}+{\mathbbm R})}+\epsilon+\|m_\Lambda(B_\epsilon)\|,
\end{align*}
and hence, invoking again Proposition~\ref{prop:vanish_on_average},
$$
\limsup_{\{\Lambda\}}\|m_\Lambda(A)\|_{{\mathfrak A}_\mathrm{sa}/{\mathbbm R}}\le\|A\|_{{\mathfrak A}_\mathrm{sa}/({\mathfrak I}+{\mathbbm R})}+\epsilon.
$$
Combining this estimate with~\eqref{eq:overIplusRlow} and letting 
$\epsilon\downarrow0$ yields~\eqref{eq:overIplusR}.
\hfill$\Box$

\label{prop:ru-R}
\begin{romanenumerate}
\item $R:{\mathcal B}_\mathrm{b}\to{\mathfrak A}_\mathrm{sa}$ is a surjective linear contraction such that $R({\mathcal B}_\mathrm{f})={\mathfrak A}_\mathrm{sa}\cap{\mathfrak A}_\mathrm{loc}$.
\item For all $\Phi\in{\mathcal B}_\mathrm{b}$, $R(\Phi)\equiv E(\Phi)\,(\mathrm{mod}\,{\mathfrak I})$.
\item The map $\hat R:{\mathcal B}_\mathrm{b}/\mathop{\mathrm{Ker}}\nolimits(R)\ni\Phi+\mathop{\mathrm{Ker}}\nolimits(R)\mapsto R(\Phi)\in{\mathfrak A}_\mathrm{sa}$ is an isometric isomorphism.

<!-- head-4: body_md -->
\textbf{(i)} Setting
$$
T_{X,Y}=\begin{cases*}
	1&\text{if $Y$ is a translate of $X$; }\\[4pt]
	0&\text{otherwise},
\end{cases*}
$$
for $\Phi\in{\mathcal B}_\mathrm{b}$, one has
\begin{equation}
\begin{split}
	\|R(\Phi)\|\le\sum_{c(X)=0}\|\Phi(X)\|
	&=\sum_{c(X)=0}\frac{1}{|X|}\sum_{x\in X}\|\Phi(X-x)\|
	=\sum_{c(X)=0}\sum_{Y\ni0}\frac{\|\Phi(Y)\|}{|Y|}T_{X,Y}\\[4pt]
	&=\sum_{Y\ni0}\frac{\|\Phi(Y)\|}{|Y|}\sum_{c(X)=0}T_{X,Y}
	=\|\Phi\|_\mathrm{b},
\end{split}
\label{eq:Rtrick}
\end{equation}
where, in the last equality, we used the fact that $Y$ has a unique translate centered 
at $0$. This shows that $R$ is a linear contraction from ${\mathcal B}_\mathrm{b}$ to ${\mathfrak A}_\mathrm{sa}$. Clearly 
$R({\mathcal B}_\mathrm{f})\subset{\mathfrak A}_\mathrm{sa}\cap{\mathfrak A}_\mathrm{loc}$. Moreover, let $A\in{\mathfrak A}_\mathrm{sa}\cap{\mathfrak A}_\mathrm{loc}$ and choose a finite set
$Y\supset\mathop{\mathrm{supp}}\nolimits(A)$ centered at $0$. Define the translation-invariant finite-range
interaction $\Phi_A$ by
\[
\Phi_A(Y+a)=\varphi^a(A),\qquad a\in{\mathbbm Z}^d,
\]
and $\Phi_A(X)=0$ for all other $X\in{\mathcal F}$. Since $Y$ has a unique translate
centered at $0$, $R(\Phi_A)=A$. Moreover, exactly $|Y|$ translates of $Y$ contain
the origin, and hence $\|\Phi_A\|_\mathrm{b}=\|A\|$. Thus,
$R({\mathcal B}_\mathrm{f})={\mathfrak A}_\mathrm{sa}\cap{\mathfrak A}_\mathrm{loc}$ and for any $A\in{\mathfrak A}_\mathrm{sa}\cap{\mathfrak A}_\mathrm{loc}$, one has 
\begin{equation}
\|A\|=\inf\{\|\Phi\|_\mathrm{b}\mid \Phi\in{\mathcal B}_\mathrm{b},R(\Phi)=A\}.
\label{eq:Rbound}
\end{equation}
Now, for any $A\in{\mathfrak A}_\mathrm{sa}$, there is a sequence $(A_n)_{n\in{\mathbbm N}}$ in ${\mathfrak A}_\mathrm{sa}\cap{\mathfrak A}_\mathrm{loc}$ 
such that $\sum_n\|A_n\|<\infty$, and $A=\sum_nA_n$. By the above construction,
there is a sequence $(\Phi_n)_{n\in{\mathbbm N}}$ in ${\mathcal B}_\mathrm{f}$ such that 
$R(\Phi_n)=A_n$ and $\|\Phi_n\|_\mathrm{b}=\|A_n\|$ for all $n\in{\mathbbm N}$. It follows that 
$\Phi=\sum_n\Phi_n\in{\mathcal B}_\mathrm{b}$ and $R(\Phi)=A$. 

\medskip\noindent\textbf{(ii)} Repeating the argument in~\eqref{eq:Rtrick}, we derive 
$\omega(R(\Phi))=\omega(E(\Phi))$ for all $\omega\in{\mathcal S}_\mathrm{I}$, and the claim
follows from Proposition~\ref{prop:vanish_on_average}.

\medskip\noindent\textbf{(iii)} With ${\mathcal K}=\mathop{\mathrm{Ker}}\nolimits(R)$, the quotient ${\mathcal Q}={\mathcal B}_\mathrm{b}/{\mathcal K}$ equipped
with the quotient norm $\|\,{\bm\cdot}\,\|_{\mathcal Q}$ is a Banach space. We denote by $\pi$ the
canonical surjection ${\mathcal B}_\mathrm{b}\to{\mathcal Q}$. Since ${\mathfrak A}_\mathrm{sa}=\mathop{\mathrm{Ran}}\nolimits(R)$ is closed, the map
$\hat R:\pi(\Phi)\mapsto R(\Phi)$ is an isomorphism, {\sl i.e.,}
a bounded linear bijection from ${\mathcal Q}$ onto ${\mathfrak A}_\mathrm{sa}$, with a bounded inverse. 
For $\Phi\in{\mathcal B}_\mathrm{f}$, \eqref{eq:Rbound} can be written 
as
$$
\|\hat R\circ\pi(\Phi)\|=\inf\{\|\Phi+\Psi\|_\mathrm{b}\mid\Psi\in{\mathcal K}\}=\|\pi(\Phi)\|_{\mathcal Q},
$$
and since $\pi({\mathcal B}_f)$ is dense in ${\mathcal Q}$, this identity extends by continuity to $\Phi\in{\mathcal B}_\mathrm{b}$.\hfill$\Box$

\label{thm:Gibbs}
Assume that $\Phi\in{\mathcal B}_\mathrm{b}$.
\begin{romanenumerate}
\item {\bf Specific energy.}
For any $\omega\in{\mathcal S}_\mathrm{I}$, one has 
\[
\lim_{\{\Lambda\}}\frac1{|\Lambda|}\omega(H_\Lambda(\Phi))={\mathcal e}_\Phi(\omega).
\]
\item {\bf Pressure.}
The limit 
\begin{equation}
P(\Phi)\coloneq\lim_{\{\Lambda\}}\frac1{|\Lambda|}\log\mathop{\mathrm{tr}}\nolimits_{{\mathcal H}_\Lambda}\left(\mathrm{e}^{-H_\Lambda(\Phi)}\right)
\label{eq:PDef}
\end{equation}
exists and is finite. The map ${\mathcal B}_\mathrm{b}\ni\Phi \mapsto P(\Phi)$ is convex,
continuous, and satisfies
\[
|P(\Phi)-P(\Psi)|\leq\|\Phi-\Psi\|_\mathrm{b}.
\]
\item {\bf Variational principle.} The pressure $P$ satisfies
\begin{equation}
P(\Phi)=\sup_{\omega\in{\mathcal S}_\mathrm{I}}\left(s(\omega)-{\mathcal e}_\Phi(\omega)\right).
\label{eq:GibbsPrince}
\end{equation}
The set of maximizers
\[
{\mathcal S}_\mathrm{eq}(\Phi)\coloneq\left\{\omega\in{\mathcal S}_\mathrm{I}\mid P(\Phi)=s(\omega)-{\mathcal e}_\Phi(\omega)\right\}
\]
is a non-empty, convex, compact subset of ${\mathcal S}_\mathrm{I}$. It is a Choquet simplex
and a face of ${\mathcal S}_\mathrm{I}$. Its elements are called \textbf{equilibrium states}
for the interaction $\Phi$ and its extreme points are $\varphi$-ergodic.

\item {\bf Dual variational principle.} For any $\omega\in{\mathcal S}_\mathrm{I}$, one has
\begin{equation} 
s(\omega)=\inf_{\Psi\in{\mathcal B}_\mathrm{b}}\left(P(\Psi)+{\mathcal e}_\Psi(\omega)\right). 
\label{eq:dual-var}
\end{equation}
\end{romanenumerate}

\label{def:wg}
A state $\omega\in{\mathcal S}_\mathrm{I}$ is called \textsl{weak Gibbs} for the
interaction $\Phi\in{\mathcal B}_\mathrm{b}$ if there exist constants $c_\Lambda>0$ satisfying

\label{eq:mil-er}
\end{equation}

$$

\leq c_\Lambda+\log\omega_{\Phi,\Lambda}^\mathrm{c}.
$$

<!-- proposition-9: body_md -->
\label{prop:BsStuf}
If $\Phi\in{\mathcal B}_\mathrm{s}$, then the following hold. 
\begin{romanenumerate}
                                                    
\item $\|W_\Lambda(\Phi)\|\le|\Lambda|\,\|\Phi\|_\mathrm{s}$.
\item $\displaystyle\lim_\Lambda\frac1{|\Lambda|}\|W_\Lambda(\Phi)\|=0$.
\item $\displaystyle\lim_{\{\Lambda\}}\left\|\frac1{|\Lambda|}U_\Lambda(\Phi)-m_\Lambda(E(\Phi))\right\|=0$.
\end{romanenumerate}
<!-- end proposition-9 -->
<!-- head-8: body_md -->
\textbf{(i)} For $\Phi\in{\mathcal B}_\mathrm{s}$, we have
\begin{align*}
\|W_\Lambda(\Phi)\|
&\leq\sum_{\genfrac{}{}{0pt}{1}{X\cap\Lambda\neq\emptyset}{X\cap\Lambda^c\neq\emptyset}}\|\Phi(X)\|
\leq\sum_{x\in\Lambda}\sum_{\genfrac{}{}{0pt}{1}{X\ni x}{X\cap\Lambda^c\neq\emptyset}}\|\Phi(X)\|\\
&=\sum_{x\in\Lambda}\sum_{\genfrac{}{}{0pt}{1}{X\ni0}{(X+x)\cap\Lambda^c\neq\emptyset}}\|\Phi(X)\|\\
&=\sum_{X\ni0}\|\Phi(X)\|
\left|\left\{x\in\Lambda\mid(X+x)\cap\Lambda^c\neq\emptyset\right\}\right|\\
&\leq|\Lambda|\sum_{X\ni0}\|\Phi(X)\|
=|\Lambda|\,\|\Phi\|_\mathrm{s}.
\end{align*}

\medskip\noindent\textbf{(ii)} Dividing the preceding estimate before its last
inequality by $|\Lambda|$ gives
\[
\frac1{|\Lambda|}\|W_\Lambda(\Phi)\|
\leq\sum_{X\ni0}\|\Phi(X)\|
\frac{\left|\left\{x\in\Lambda\mid(X+x)\cap\Lambda^c\neq\emptyset\right\}\right|}{|\Lambda|}.
\]
For fixed $X\ni0$, the set in the numerator is contained in
\[
\left\{x\in\Lambda\mid\mathop{\mathrm{dist}}\nolimits(x,\Lambda^c)\leq\oslash(X)\right\}.
\]
Its relative cardinality therefore tends to zero along every van~Hove net,
while it is bounded by one. Since
$\sum_{X\ni0}\|\Phi(X)\|=\|\Phi\|_\mathrm{s}<\infty$, dominated convergence
yields Part~\textbf{(ii)}.

\medskip\noindent\textbf{(iii)} Part~\textbf{(ii)} applies in particular along
centered cubes, and the assertion follows from Corollary~\ref{cor:Abel} and
$U_\Lambda(\Phi)=H_\Lambda(\Phi)+W_\Lambda(\Phi)$. \hfill$\Box$

\label{thm:GibbsII}
\begin{romanenumerate}
\item There is a unique $e_0\in(e_\mathrm{min},e_\mathrm{max})$ such that 
${\mathcal s}(e_0)=\max\{{\mathcal s}(e)\mid e\in{\mathbbm R}\}=\log\mathop{\mathrm{dim}}\nolimits{\mathcal H}$, the function ${\mathcal s}$ being
monotone increasing on the interval $[e_\mathrm{min},e_0]$ and monotone decreasing
on the interval $[e_0,e_\mathrm{max}]$.
\item The parametric form of the Gibbs variational principle~\eqref{eq:AizLieb1}
admits the following dual principle: For any $e\in(e_\mathrm{min},e_\mathrm{max})$, 
\begin{equation}
{\mathcal s}(e)=\min_{\beta\in{\mathbbm R}}\left({\mathcal p}(\beta)+\beta e\right),
\label{eq:smalldualGibbs}
\end{equation}
the minimum being achieved at $\beta=\beta_\ast(e)$, i.e.,
\begin{equation}
{\mathcal s}(e)={\mathcal p}(\beta_\ast(e))+\beta_\ast(e)e.
\label{eq:AizLieb}
\end{equation}
Moreover, if $\nu\in{\mathcal S}_\mathrm{I}$ is such that ${\mathcal e}_\Phi(\nu)=e$, then ${\mathcal s}(e)=s(\nu)$ iff
$\nu\in{\mathcal S}_\mathrm{eq}(\beta_\ast(e)\Phi)$.
\end{romanenumerate}
<!-- end theorem-22 -->
<!-- head-13: body_md -->
\textbf{(i)} The tracial state on ${\mathfrak A}$ is the unique maximizer of the
specific entropy $s$, \cite[Theorem~IV.2.4]{Simon1993}. Thus setting $e_0=\mathop{\mathrm{tr}}\nolimits(E(\Phi))$,
we have ${\mathcal s}(e_0)=\max\{s(\nu)\mid\nu\in{\mathcal S}_\mathrm{I}\}=\log\mathop{\mathrm{dim}}\nolimits{\mathcal H}$. The function $\beta_\ast={\mathcal s}'$
being monotone decreasing, we have $\beta_\ast(e)\ge0$ for $e\in[e_\mathrm{min},e_0]$,
and $\beta_\ast(e)\le0$ for $e\in[e_0,e_\mathrm{max}]$.

\medskip\noindent\textbf{(ii)}
Since $\hat{\mathcal s}$ is convex, it is the Legendre--Fenchel transform of ${\mathcal p}$, 
which translates into Relation~\eqref{eq:smalldualGibbs}.
A well known property of the Legendre--Fenchel 
transform~\cite[Theorem~23.5]{Rockafellar1970},\footnote{$(\partial f)(x)$ 
denotes the subdifferential of $f$ at $x$, \textsl{i.e.,\ } the interval 
$[(\partial^-f)(x),(\partial^+f)(x)]$.}
$$
-e\in(\partial{\mathcal p})(\beta)
\quad\Leftrightarrow\quad {\mathcal p}(\beta)-{\mathcal s}(e)=-\beta e
\quad\Leftrightarrow\quad\beta=\beta_\ast(e),
$$
yields Relation~\eqref{eq:AizLieb}. Finally, it follows from definition~\eqref{eq:csDef}
that $s(\nu)\le{\mathcal s}(e)$ for any $\nu\in{\mathcal S}_\mathrm{I}$ such that ${\mathcal e}_\Phi(\nu)=e$,
and Relation~\eqref{eq:AizLieb} yields that
$$
s(\nu)={\mathcal s}(e)={\mathcal p}(\beta_\ast(e))+\beta_\ast(e)e=P(\beta_\ast(e)\Phi)+\beta_\ast(e){\mathcal e}_\Phi(\nu)
$$
iff $\nu\in{\mathcal S}_\mathrm{eq}(\beta_\ast(e)\Phi)$.
\hfill$\Box$

We say that the pair $(\Phi,\Psi)\in{\mathcal B}_\mathrm{sd}\times{\mathcal B}_\mathrm{sd}$ has the \textbf{(SD)}\footnote{SD stands for surface-dynamics.}
property if the following hold for all $t$ in a neighborhood of $0$.
\begin{romanenumerate}
\item For every finite $\Lambda\in{\mathcal F}$, the surface energy
\[
W_\Lambda(\Psi|\Phi,t)
=
\lim_{\widetilde\Lambda}
\big(H_{\widetilde\Lambda}(\Psi|\Phi,t)-H_\Lambda(\Psi|\Phi,t)
-H_{\widetilde\Lambda\setminus\Lambda}(\Psi|\Phi,t)\big)
\]
exists in norm, where $\widetilde\Lambda$ runs over the finite supersets of
$\Lambda$, directed by inclusion.
\item The surface energies vanish in the van Hove sense,
\[
\lim_\Lambda\frac{1}{|\Lambda|}\|W_\Lambda(\Psi|\Phi,t)\|=0.
\]
\item One has, again in the van Hove sense,
\[
\lim_\Lambda\frac1{|\Lambda|}\left\|\tau_\Phi^{-t}(H_\Lambda(\Psi))-H_\Lambda(\Psi|\Phi,t)\right\|=0.
\]
\end{romanenumerate}

<!-- theorem-23: body_md -->
Suppose that $(\Phi, \Psi)$ has the \preSD{} property and that for all $s,t\in{\mathbbm R}$, 
\begin{equation}
\tau_\Phi^t \circ \tau_\Psi^s=\tau_\Psi^s\circ \tau_\Phi^t.
\label{eq:DynCommut}
\end{equation}
Then $\Psi$ is $\tau_\Phi$-invariant, i.e., $[\Psi]\in{\mathfrak C}(\Phi)$.
\label{thm:main1-const}
<!-- end theorem-23 -->
<!-- head-16: body_md -->
Since
\[
\mathrm{e}^{\mathrm{i} sH_\Lambda(\Psi|\Phi,t)}
=\mathrm{e}^{-\mathrm{i} tH_\Lambda(\Phi)}\mathrm{e}^{\mathrm{i} sH_\Lambda(\Psi)}\mathrm{e}^{\mathrm{i} tH_\Lambda(\Phi)},
\]
one has
\begin{equation}
\lim_{\{\Lambda\}}\mathrm{e}^{\mathrm{i} sH_\Lambda(\Psi|\Phi,t)}A\mathrm{e}^{-\mathrm{i} sH_\Lambda(\Psi|\Phi,t)}
=\tau_\Phi^{-t}\circ \tau_\Psi^s\circ \tau_\Phi^{t}(A)
\label{eq:Borodin}
\end{equation}
for all $A\in{\mathfrak A}$, uniformly for $s$ and $t$ in compacts. Our assumption~\eqref{eq:DynCommut} 
yields that the right-hand side of the last identity is just $\tau_\Psi^s(A)$. 
Since $[H_{\Lambda^\prime\setminus\Lambda}(\Psi|\Phi,t),A]=0$ whenever $A\in{\mathfrak A}_\Lambda$, 
the \preSD{} property further gives that, for such $A$ and all $t$ in a neighborhood of $0$,
\[
\lim_{\Lambda^\prime}[ H_{\Lambda^\prime}(\Psi|\Phi,t),A]
=[H_\Lambda(\Psi|\Phi,t)+W_\Lambda(\Psi|\Phi,t),A].
\]
Hence, on the one hand
\begin{align*}
\tau_\Psi^s(A)
&=\lim_{\Lambda^\prime}\mathrm{e}^{\mathrm{i} sH_{\Lambda^\prime}(\Psi|\Phi,t)}A\mathrm{e}^{-\mathrm{i} sH_{\Lambda^\prime}(\Psi|\Phi,t)}\\[4pt]
&=A+\mathrm{i}\,\lim_{\Lambda^\prime}\int_0^s\mathrm{e}^{\mathrm{i} uH_{\Lambda^\prime}(\Psi|\Phi,t)}
[H_{\Lambda^\prime}(\Psi|\Phi,t),A]\mathrm{e}^{-\mathrm{i} uH_{\Lambda^\prime}(\Psi|\Phi,t)}\mathrm{d} u\\[4pt]
&=A+\mathrm{i}\int_0^s\tau_\Psi^u([H_\Lambda(\Psi|\Phi,t)+ W_\Lambda(\Psi|\Phi,t),A])\mathrm{d} u,
\end{align*}
and on the other hand
\[
\tau_\Psi^s(A)=A+\mathrm{i}\int_0^s\tau_\Psi^u([H_\Lambda(\Psi)+W_\Lambda(\Psi),A])\mathrm{d} u,
\]
which allows us to conclude that
\[
[H_\Lambda(\Psi|\Phi,t)+ W_\Lambda(\Psi|\Phi,t),A]=[H_\Lambda(\Psi)+W_\Lambda(\Psi),A].
\]
Taking the partial normalized trace over ${\mathfrak A}_{\Lambda^c}$ yields
\[
[ H_\Lambda(\Psi|\Phi,t)-H_\Lambda(\Psi)+\mathop{\mathrm{tr}}\nolimits_{\Lambda^c}(W_\Lambda(\Psi|\Phi,t)-W_\Lambda(\Psi)),A]=0, 
\]
and so 
\begin{equation}
H_\Lambda(\Psi|\Phi,t)-H_\Lambda(\Psi)+\mathop{\mathrm{tr}}\nolimits_{\Lambda^c}(W_\Lambda(\Psi|\Phi,t)-W_\Lambda(\Psi))=C_\Lambda(t)\mathbbm{1}
\label{eq:sosot}
\end{equation}
for some $C_\Lambda(t)\in{\mathbbm R}$. Since $\mathop{\mathrm{tr}}\nolimits(H_\Lambda(\Psi|\Phi,t)-H_\Lambda(\Psi))=0$, we deduce
\[
\mathop{\mathrm{tr}}\nolimits(W_\Lambda(\Psi|\Phi,t)-W_\Lambda(\Psi))=C_\Lambda(t).
\]
Hence, by Proposition~\ref{prop:BsStuf}(ii) and the \preSD{} property,
\[
\lim_\Lambda\frac{|C_\Lambda(t)|}{|\Lambda|}=0.
\]
Going back to~\eqref{eq:sosot} and using contractivity of the partial normalized
trace, we obtain, for any $\nu\in{\mathcal S}_\mathrm{I}$ and $t$ in a neighborhood of $0$,
\[
\lim_\Lambda\frac1{|\Lambda|}
\left|\nu(H_\Lambda(\Psi|\Phi,t)-H_\Lambda(\Psi))\right|=0.
\]
In particular, the corresponding centered-cube limit vanishes. Combining this
with Part~\textbf{(iii)} of the \preSD{} property and
Theorem~\ref{thm:Gibbs}(i) gives
\[
\nu\circ\tau_\Phi^{-t}(E(\Psi))
=\lim_{\{\Lambda\}}\frac1{|\Lambda|}\nu\circ\tau_\Phi^{-t}(H_\Lambda(\Psi))
=\lim_{\{\Lambda\}}\frac1{|\Lambda|}\nu(H_\Lambda(\Psi|\Phi,t))
=\nu(E(\Psi)).
\]
Thus a neighborhood of $0$ is contained in ${\mathcal T}_\Phi(E(\Psi))$ and it follows that
${\mathcal T}_\Phi(E(\Psi))={\mathbbm R}$. \hfill$\Box$

\label{thm:main2-const}
Let $\Phi,\Psi\in{\mathcal B}_\mathrm{sd}$ and suppose that $[\Psi]\in{\mathfrak C}(\Phi)$. If either
$(\Phi,\Psi)$ has the \preSDplus{} property or $\tau_\Phi$ preserves the
specific entropy, then for all $s,t\in{\mathbbm R}$, 
$$
\tau_\Phi^t \circ \tau_\Psi^s=\tau_\Psi^s\circ \tau_\Phi^t.
$$
<!-- end theorem-24 -->
<!-- head-18: body_md -->
We first assume that $(\Phi,\Psi)$ has the \preSDplus{} property.
By the group property, it suffices to show that $\alpha=\tau_\Psi$ for small
enough $t$. Although it may happen that $\Psi_t\not\in{\mathcal B}_\mathrm{b}$,
the pressure and the specific energy associated with the local Hamiltonian family
$H(\Psi|\Phi,t)$ are well-defined. In the formulas below we denote these quantities,
with a slight abuse of notation, by $P(\Psi_t)$ and ${\mathcal e}_{\Psi_t}$. This is precisely
the local-Hamiltonian formalism used in the companion paper~\cite{Jaksic2026b}. It is immediate from the
definitions~\eqref{eq:PDef} and~\eqref{eq:ha-t} that
\begin{equation} 
P(\Psi_t)=P(\Psi),
\label{eq:pres-t}
\end{equation} 
while Part~\textbf{(iii)} of the \preSD{} property and $E(\Psi)$ being a constant of the motion for $\tau_\Phi$ 
ensure that for any $\omega\in{\mathcal S}_\mathrm{I}$ 
\begin{equation}
{\mathcal e}_{\Psi_t}(\omega)=
\lim_{\{\Lambda\}}\frac1{|\Lambda|}\omega(H_\Lambda(\Psi|\Phi,t))=\lim_{\{\Lambda\}}\frac1{|\Lambda|}\omega\circ\tau_\Phi^{-t}(H_\Lambda(\Psi))
=\omega\circ\tau_\Phi^{-t}(E(\Psi))=\omega(E(\Psi))={\mathcal e}_{\Psi}(\omega).
\label{eq:energy-t} 
\end{equation}
We now appeal to Araki--Moriya's thermodynamic formalism. We shall use the terminology and
notation of~\cite{Araki2003} and refer the reader to this article for more details.
It follows from the \preSD{} property that $\Psi_t$ is a translation covariant general potential, 
as defined in~\cite[Section~5.5 and~8.1]{Araki2003}. More precisely, Conditions $(\Phi-\mathrm{a})$ 
and $(\Phi-\mathrm{b})$, \textsl{i.e.,\ } $\Psi_t(X)^\ast=\Psi_t(X)\in{\mathfrak A}_X$ for $X\in{\mathcal F}$, with 
$\Psi_t(\emptyset)=0$, are obviously satisfied. From the formula
$$
\Psi_t(X)=\sum_{Y\subset X}(-1)^{|X|-|Y|}H_Y(\Psi|\Phi,t)
$$
one easily infers Condition $(\Phi-\mathrm{f})$, translation covariance.
Condition $(\Phi-\mathrm{c})$, which expresses evenness in the fermionic
setting, is omitted for quantum spin systems. Finally, since for
$\Lambda\subset\Lambda'\in{\mathcal F}$,
$$
\sum_{\genfrac{}{}{0pt}{1}{X\subset\Lambda'}{X\cap\Lambda\neq\emptyset}}\Psi_t(X)
=H_{\Lambda'}(\Psi|\Phi,t)-H_{\Lambda'\setminus\Lambda}(\Psi|\Phi,t),
$$
it follows from Part~\textbf{(i)} of the \preSD{} property that
$$
\lim_{\Lambda'}\sum_{\genfrac{}{}{0pt}{1}{X\subset\Lambda'}{X\cap\Lambda\neq\emptyset}}\Psi_t(X)
=H_\Lambda(\Psi|\Phi,t)+W_\Lambda(\Psi|\Phi,t),
$$
which yields Condition $(\Phi-\mathrm{e})$. Part~(ii) of the \preSD{} property
provides the surface-energy assumption of~\cite[Proposition~14.1]{Araki2003}.
Moreover, invariance of the trace under finite-volume unitary conjugation and
finite-range approximation in ${\mathcal B}_\mathrm{b}$ give
$$
c=\lim_\Lambda\frac1{|\Lambda|}\mathop{\mathrm{tr}}\nolimits(H_\Lambda(\Psi|\Phi,t))
=\lim_\Lambda\frac1{|\Lambda|}\mathop{\mathrm{tr}}\nolimits(H_\Lambda(\Psi))=\mathop{\mathrm{tr}}\nolimits(E(\Psi)).
$$
By Part~(ii) of the \preSD{} property, the same trace limit is obtained with
$H_\Lambda(\Psi|\Phi,t)+W_\Lambda(\Psi|\Phi,t)$ in place of
$H_\Lambda(\Psi|\Phi,t)$. Thus, \cite[Proposition~14.1]{Araki2003} applies:
there exists a translation covariant standard potential, \cite[Definition~5.10 and Section~8.1]{Araki2003} 
$\widetilde\Psi_t$ such that, for any $\omega\in{\mathcal S}_\mathrm{I}$,
\begin{equation}
P(\widetilde\Psi_t)=P(\Psi_t)+c,\qquad {\mathcal e}_{\widetilde\Psi_t}(\omega)={\mathcal e}_{\Psi_t}(\omega)-c.
\label{eq:AMchanges}
\end{equation}
By~\cite[Section~11 and Proposition~14.1]{Araki2003}, Relations~\eqref{eq:pres-t}, \eqref{eq:energy-t} 
and~\eqref{eq:AMchanges} give that $\widetilde\Psi_t$, $\Psi_t$ and $\Psi$ share the same equilibrium states.
The \preSDplus{} property ensures that for $t$ in a neighborhood of $0$ the $C^\ast$-dynamics $\alpha=\tau_\Phi^{-t}\circ\tau_\Psi\circ\tau_\Phi^{t}$
satisfies properties~(II), (III) and~(IV) of~\cite[Section~1]{Araki2003}. It then follows 
from~\cite[Theorems~A and~B]{Araki2003} that $\tau_\Psi$ and $\alpha$ share the same KMS states. 
Hence, uniqueness of the modular group, Theorem~\ref{thm:israel}~\textbf{(iii)}, yields that
$\alpha=\tau_\Psi$ for small enough $t$.

Suppose now that $\tau_\Phi$ preserves the specific entropy. Choose
$\eta\in{\mathcal S}_\mathrm{eq}(\Psi)$.  Conservation of the specific entropy and the assumption
$[\Psi]\in{\mathfrak C}(\Phi)$ give, for all $t\in{\mathbbm R}$,
\[
s(\eta_t)-\eta_t(E(\Psi))
=s(\eta)-\eta(E(\Psi))=P(\Psi).
\]
Thus $\eta_t\in{\mathcal S}_\mathrm{eq}(\Psi)$, and Theorem~\ref{thm:KMSeq} implies that
$\eta_t$ is a $1$-KMS state for $\tau_\Psi$. By conjugation, $\eta_t$ is also a $1$-KMS state for
$\tau_\Phi^{-t}\circ\tau_\Psi\circ\tau_\Phi^t$. It follows from Theorem~\ref{thm:israel}~\textbf{(iii)} that
\[
\tau_\Phi^{-t}\circ\tau_\Psi^s\circ\tau_\Phi^t=\tau_\Psi^s
\]
for all $s,t\in{\mathbbm R}$. \hfill$\Box$

<!-- proposition-14: body_md -->
Suppose that $\Phi,\Psi\in{\mathcal B}_\mathrm{sd}$. Then
\begin{equation}
[\Psi]\in{\mathfrak C}(\Phi)\quad\Leftrightarrow\quad [\Phi]\in{\mathfrak C}(\Psi).
\label{eq:t-nice}
\end{equation}
<!-- end proposition-14 -->
<!-- head-19: body_md -->
We use a bilinear version of the argument in the proof
of~\cite[Theorem~2.8(2)]{Jaksic2023a}.

We first note that
\[
\sum_{X\ni0}\frac1{|X|}
\sum_{Y:\,Y\cap X\neq\emptyset}
\|[\Phi(Y),\Psi(X)]\|
\leq 2\|\Phi\|_{\mathrm s}\|\Psi\|_{\mathrm s},
\]
since $\sum_{Y:\,Y\cap X\neq\emptyset}\|\Phi(Y)\|
\leq |X|\|\Phi\|_{\mathrm s}$. The finite partial sums of
$E(\Psi)=\sum_{X\ni0}\Psi(X)/|X|$ belong to ${\mathfrak A}_\mathrm{loc}$ and converge
in norm. The estimate shows that their images under $\delta_\Phi$
also converge in norm. By closedness of $\delta_\Phi$, it follows that
$E(\Psi)\in\mathop{\mathrm{Dom}}\nolimits(\delta_\Phi)$ and that $\delta_\Phi(E(\Psi))$ is given
by the corresponding absolutely convergent commutator series.
Similarly, $E(\Phi)\in\mathop{\mathrm{Dom}}\nolimits(\delta_\Psi)$.

Fix $\nu\in{\mathcal S}_\mathrm{I}$. Absolute convergence and translation invariance give
\begin{align*}
\nu(\delta_\Phi(E(\Psi)))
&=\sum_{X\ni0}\sum_{Y:\,Y\cap X\neq\emptyset}
\frac1{|X|\,|Y|}\sum_{y\in Y}
\nu\bigl(\mathrm{i}[\Phi(Y-y),\Psi(X-y)]\bigr)\\
&=\sum_{Y\ni0}\frac1{|Y|}
\sum_{X:\,X\cap Y\neq\emptyset}
\nu\bigl(\mathrm{i}[\Phi(Y),\Psi(X)]\bigr)\\
&=-\nu(\delta_\Psi(E(\Phi))).
\end{align*}
For the second equality, set $X'=X-y$ and $Y'=Y-y$. Then
$0\in Y'$ and $-y\in X'$. For each fixed pair $(X',Y')$, there
are exactly $|X'|$ possible values of $y$, cancelling the factor
$1/|X'|$.

Suppose now that $[\Psi]\in{\mathfrak C}(\Phi)$. Differentiating the conservation
identity at $t=0$ gives $\nu(\delta_\Phi(E(\Psi)))=0$ for every
$\nu\in{\mathcal S}_\mathrm{I}$. The preceding identity therefore yields
$\nu(\delta_\Psi(E(\Phi)))=0$ for every $\nu\in{\mathcal S}_\mathrm{I}$.
Since $\nu\circ\tau_\Psi^t\in{\mathcal S}_\mathrm{I}$, we obtain
\[
\frac{\mathrm{d}}{\mathrm{d} t}\nu\circ\tau_\Psi^t(E(\Phi))
=(\nu\circ\tau_\Psi^t)(\delta_\Psi(E(\Phi)))=0.
\]
Thus $[\Phi]\in{\mathfrak C}(\Psi)$. The reverse implication follows by
interchanging $\Phi$ and $\Psi$. \hfill$\Box$

\label{prop:wd-pres}
Suppose that $\tau_\Phi$ preserves the specific entropy, and let
$(\rho,\Psi)\in{\mathcal S}_\mathrm{I}\times{\mathcal B}_\mathrm{b}$ be a regular pair. Then the following
statements are equivalent.
\begin{romanenumerate}
\item $s(\nu \circ \tau_\Phi^t|\rho)=s(\nu|\rho)$ for all $\nu\in{\mathcal S}_\mathrm{I}$
and $t\in{\mathbbm R}$.
\item $[\Psi]\in{\mathfrak C}(\Phi)$.
\end{romanenumerate}
<!-- end proposition-15 -->
<!-- head-20: body_md -->
The pair $(\rho,\Psi)$ being regular, one has
$s(\nu|\rho)=-s(\nu)+{\mathcal e}_\Psi(\nu)+P(\Psi)$ for all $\nu\in{\mathcal S}_\mathrm{I}$.
Since $\tau_\Phi$ preserves the specific entropy, we further have
\begin{align*}
s(\nu|\rho)
&=-s(\nu\circ\tau_\Phi^t)+{\mathcal e}_\Psi(\nu\circ\tau_\Phi^t)+P(\Psi)
+\nu\left(E(\Psi)-\tau_\Phi^t(E(\Psi))\right)
\\[4pt]
&=s(\nu\circ\tau_\Phi^t|\rho)
+\nu\left(E(\Psi)-\tau_\Phi^t(E(\Psi))\right).
\end{align*}
Thus, \textbf{(i)} holds iff $E(\Psi)\approx\tau_\Phi^t(E(\Psi))$
for all $t\in{\mathbbm R}$. By the definition of ${\mathfrak C}(\Phi)$, this is precisely
\textbf{(ii)}. \hfill$\Box$

$s(\omega_+)=s(\omega_+')$. Since $[\Psi_+]\in{\mathfrak C}(\Phi)$,
\[
{\mathcal e}_{\Psi_+}(\omega_+')={\mathcal e}_{\Psi_+}(\omega)={\mathcal e}_{\Psi_+}(\omega_+).
\]
Using regularity of $(\omega_+,\Psi_+)$ we therefore obtain
\[
s(\omega_+'|\omega_+)
=-s(\omega_+')+{\mathcal e}_{\Psi_+}(\omega_+')+P(\Psi_+)
=-s(\omega_+)+{\mathcal e}_{\Psi_+}(\omega_+)+P(\Psi_+)=0.
\]
This proves Part~\textbf{(i)}. 

For Part~\textbf{(ii)}, let $\omega_+'\in{\mathcal S}_{\mathrm{req},+}(\Phi)$. By
Part~\textbf{(i)}, $s(\omega_+'|\omega_+)=0$. 
Since $(\omega_+,\Psi_+)$ is regular, this is equivalent to
$\omega_+'\in{\mathcal S}_\mathrm{eq}(\Psi_+)$. 
\hfill$\Box$

<!-- proposition-19: body_md -->
\label{prop:DGVP-state}
Let $\omega_+\in\Sigma_+(\omega,\Phi)\cap{\mathcal S}_\mathrm{phys}(\Phi)$ and choose
$\Psi_+\in{\mathcal B}_\mathrm{sd}\cap{\mathcal B}_\mathrm{R}$ such that
$[\Psi_+]\in{\mathfrak C}(\Phi)$ and $\omega_+\in{\mathcal S}_\mathrm{eq}(\Psi_+)$. Then, the following hold.
\begin{romanenumerate}
\item
\[
\inf_{\nu\in{\mathcal S}_\mathrm{phys}(\Phi)}s(\omega|\nu)
=s(\omega_+)-s(\omega).
\]
\item The set of minimizers is precisely ${\mathcal S}_\mathrm{eq}(\Psi_+)$. In particular, the
minimizer is unique iff ${\mathcal S}_\mathrm{eq}(\Psi_+)=\{\omega_+\}$.
\item If $\Sigma_+(\omega,\Phi)\subset{\mathcal S}_\mathrm{phys}(\Phi)$ and
${\mathcal S}_\mathrm{eq}(\Psi_+)=\{\omega_+\}$, then
$\Sigma_+(\omega,\Phi)=\{\omega_+\}$.
\end{romanenumerate}
<!-- end proposition-19 -->
<!-- head-27: body_md -->
Since ${\mathcal S}_\mathrm{phys}(\Phi)\subset{\mathcal S}_\mathrm{req}(\Phi)$, Proposition~\ref{prop:fr}
gives
\[
s(\omega|\nu)\ge s(\omega_+)-s(\omega)
\]
for $\nu\in{\mathcal S}_\mathrm{phys}(\Phi)$. Moreover, $[\Psi_+]\in{\mathfrak C}_\mathrm{req}(\Phi)$, and hence the same
proposition gives
\[
s(\omega|\omega_+)=s(\omega_+)-s(\omega).
\]
Since $\omega_+\in{\mathcal S}_\mathrm{phys}(\Phi)$, this proves Part~\textbf{(i)}.

$\nu\in{\mathcal S}_\mathrm{eq}(\Psi_+)$ is a minimizer by Proposition~\ref{prop:fr}. This proves

then gives $\omega_+'\in{\mathcal S}_\mathrm{eq}(\Psi_+)=\{\omega_+\}$. \hfill$\Box$

<!-- theorem-31: body_md -->
\label{thm:main-ate}
Suppose that $\beta_\mathrm{eq} \Phi\in{\mathcal B}_\mathrm{R}$ and that there exists
$\omega_\mathrm{eq}\in{\mathcal S}_\mathrm{eq}(\omega,\Phi)$ such that
\[
\omega(C)=\omega_\mathrm{eq}(C)
\]
provided $[C]\in\mathfrak{C}_\mathrm{phys}(\Phi)$. Let $\omega_+\in\Sigma_+(\omega,\Phi)$. Then the following statements are
equivalent:
\begin{romanenumerate}
\item $\omega_+\in{\mathcal S}_\mathrm{eq}(\omega,\Phi)$;
\item $\omega_+\in{\mathcal S}_\mathrm{phys}(\Phi)$.
\end{romanenumerate}
<!-- end theorem-31 -->
<!-- head-32: body_md -->
Suppose first that $\omega_+\in{\mathcal S}_\mathrm{eq}(\omega,\Phi)$. Then
$\omega_+\in{\mathcal S}_\mathrm{eq}(\beta_\mathrm{eq}\Phi)$. Since $E(\Phi)$ is a constant of the
motion and $\beta_\mathrm{eq}\Phi\in{\mathcal B}_\mathrm{sd}\cap{\mathcal B}_\mathrm{R}$, it follows that
$\omega_+\in{\mathcal S}_\mathrm{phys}(\Phi)$.

Conversely, assuming that $\omega_+\in{\mathcal S}_\mathrm{phys}(\Phi)$, one can choose 
$\Psi_+\in{\mathcal B}_\mathrm{sd}\cap{\mathcal B}_\mathrm{R}$ such that $[\Psi_+]\in{\mathfrak C}(\Phi)$ and
$\omega_+\in{\mathcal S}_\mathrm{eq}(\Psi_+)$. By the assumption on $\omega_\mathrm{eq}$ and
conservation of $E(\Psi_+)$,
\[
\omega_\mathrm{eq}(E(\Psi_+))
=\omega(E(\Psi_+))
=\omega_+(E(\Psi_+)).
\]
Hence
\begin{align*}
0\le s(\omega_\mathrm{eq}|\omega_+)
&=-s(\omega_\mathrm{eq})+\omega_\mathrm{eq}(E(\Psi_+))+P(\Psi_+)\\[1mm]
&=s(\omega_+)-s(\omega_\mathrm{eq}).
\end{align*}
Proposition~\ref{prop:st-pres} gives the reverse inequality, and therefore
$s(\omega_\mathrm{eq}|\omega_+)=0$. Regularity of $(\omega_+,\Psi_+)$ yields
$\omega_\mathrm{eq}\in{\mathcal S}_\mathrm{eq}(\Psi_+)$. Since also
$\omega_\mathrm{eq}\in{\mathcal S}_\mathrm{eq}(\beta_\mathrm{eq}\Phi)$ and
$\Psi_+,\beta_\mathrm{eq}\Phi\in{\mathcal B}_\mathrm{sd}$, Theorem~\ref{thm:phys-eq} gives $\Psi_+\sim\beta_\mathrm{eq}\Phi$. 
It follows that $\omega_+\in{\mathcal S}_\mathrm{eq}(\beta_\mathrm{eq}\Phi)$. Finally,
${\mathcal e}_\Phi(\omega_+)={\mathcal e}_\Phi(\omega)$ by energy conservation, and hence
$\omega_+\in{\mathcal S}_\mathrm{eq}(\omega,\Phi)$.\vskip-8pt \hfill$\Box$
