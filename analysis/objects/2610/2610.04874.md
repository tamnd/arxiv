<!-- theorem-1: body_md -->
\label{thm: low}
    If $n\ge 4m^{2}$, the AM-GM inequality~\eqref{eq: rr} holds on the lower bound and
    \begin{equation*}
        \mathbf W_{m,n} \succeq -\left(1 - \frac{1}{2m}\right)\mathbf I.
    \end{equation*}
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm: both}
    If $n \ge 2 \lceil m/2 \rceil^2$, the AM-GM inequality~\eqref{eq: rr} holds on both bounds.
<!-- end theorem-2 -->
<!-- remark-1: body_md -->
This relaxed lower-bound problem is closely related to finding approximate Chebyshev nodes (extrema are nodes of second kind) from an equispaced grid. These nodes are known as ``mock-Chebyshev''. It is a well-studied topic in numerical analysis~\cite{coppersmith1992growth,platte2011impossibility,adcock2019optimal}. The first gap between Chebyshev extrema is $1 - \cos(\pi/m) \sim \frac{\pi^2}{2m^2}$, which needs to be resolved by the grid size $\frac{2}{n}$ for a stable interpolation. This gives a conjectured optimal constant $C_{\ast} = 4/\pi^2$. The stability of interpolation is precisely characterized by $\sum_{i=0}^m |a_i|$.
<!-- end remark-1 -->
<!-- lemma-1: body_md -->
\label{lem: sep}
If $n > m^2$, then $0 = s_0 < s_1 < \dots < s_m$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Clearly, the sequence is non-decreasing. If $s_j = s_{j+1}$ for some $0\le j\le m-1$, then
    \begin{equation*}
        \left| \frac{n(1 - c_j)}{2} - \frac{n(1 - c_{j+1})}{2} \right| \le 1 \Longrightarrow |c_j - c_{j+1}|\le \frac{2}{n}.
    \end{equation*}
    On the other hand,

\label{lem: iter est}
   Let $1\le j \le L-1$. Suppose that

Let $\mathcal D = u \frac{d}{du}$, then it is a self-adjoint operator on $\mathbb{H}_{\rm coeff}$ since $\mathcal D u^l = l u^l$. We consider the operator $$\mathcal H p :=  (u -1)^{\sharp}\mathcal D (L - \mathcal D) (u -1) p, $$ it satisfies
\begin{equation}\label{eq: tri}
\begin{aligned}
    \mathcal H (u-1)^{l} &= (u^{-1}-1) \left( (L-1) u (l+1) (u-1)^{l} - (l+1)l u^2 (u-1)^{l-1} \right) \\
    &= -(l+1)(u-1)^{l} ((L-l-1) (u-1) - l)\\
    &= l(l+1)(u-1)^{l} - (l+1)(L-l-1) (u-1)^{l+1},
\end{aligned}
\end{equation} 
when $l = L-1$, the last term vanishes. Thus, $\mathcal H$ is self-adjoint on $\mathbb{H}_{\rm coeff}$ and preserves $V_j$. The calculation~\eqref{eq: tri} indicates the representation under $V_j$ is triangular. The eigenvalues restricted on $V_j$ are the diagonal values $l(l+1)$, $j\le l \le L-1$. Therefore,
\begin{equation*}
    \langle {\varphi, \mathcal H \varphi} \rangle_{\rm coeff} \ge j(j+1) \|\varphi\|_{\rm coeff}^2.  
\end{equation*}
Moreover, 
\begin{equation*}
\begin{aligned}
    \langle {\varphi, \mathcal H \varphi} \rangle_{\rm coeff}  &= \langle {\varphi, (u -1)^{\sharp}\mathcal D (L- \mathcal D) (u -1)  \varphi} \rangle_{\rm coeff} \\
    &= \langle {(u -1) \varphi, \mathcal D (L- \mathcal D) (u -1)  \varphi} \rangle_{\rm coeff}.
\end{aligned}
\end{equation*}
The eigenvalues of $ \mathcal D (L - \mathcal D)$ are $l(L - l)\le \frac{L^2}{4}$ for $0\le l\le L$ with eigenvectors as the monomial $\{ u^l \}_{l=0}^L$. In particular, the monomials $u^0$ and $u^L$ are suppressed completely. Therefore, 
\begin{equation*}
j(j+1) \|\varphi\|_{\rm coeff}^2 \le \frac{L^2}{4} \|(u-1)\varphi\|_{\rm coeff, 1}^2 .
\end{equation*}

<!-- lemma-10: body_md -->
\label{lem: c1}
    For any $s$, let $Z = \sum_{i=1}^n z_i$, with $z_i\in\{0, 1\}$, then 
    \begin{equation}
        \begin{aligned}
            \int_{\{0, 1\}^n} (Z - s) f(\mathbf z) d\nu_s(\mathbf z) &= 0,\qquad &&\forall f\in\mathbb P_{n-1}.\\
            \int_{\{0,1\}^n} \prod_{j\in S} z_j \prod_{i\in T}(1-z_i) d\nu_s(\mathbf z) & = \frac{(s)_{|S|} (n-s)_{|T|}}{(n)_{|S|+|T|}},\qquad &&\forall S\cap T =\emptyset. 
        \end{aligned}
    \end{equation}
<!-- end lemma-10 -->
<!-- proof-10: body_md -->
Consider a monomial $z_{j_1}\cdots z_{j_r}$, where $j_1, \dots, j_r$ distinct. Then,
    \begin{equation*}
    \begin{aligned}
             \int_{\{0, 1\}^n} Z z_{j_1}\cdots z_{j_r} d\nu_s &= r\frac{(s)_r}{(n)_r}  + (n-r) \frac{(s)_{r+1}}{(n)_{r+1}} \\
             &= r\frac{s(s-1)\cdots (s-r+1)}{n(n-1)\cdots (n-r+1)}  + (s-r) \frac{s(s-1)\cdots (s-r+1)}{n(n-1)\cdots (n-r+1)}\\
             &= s \frac{(s)_r}{(n)_r} =  s\int_{\{0, 1\}^n} z_{j_1}\cdots z_{j_r} d\nu_s,
    \end{aligned}
    \end{equation*}
    The first decomposition holds because exactly $n-r$ indices from $Z$ increase the degree from $r$ to $r+1$. The remaining indices keep the degree.  

    For all integer $s = 0,\dots, n$. Both sides of the second equation give the probability that a subset of size $s$ contains $S$ and excludes $T$. Treating both sides as polynomials in $s$, their degrees are at most $|S|+|T|\le n-1$. Thus, the polynomials are identical.
