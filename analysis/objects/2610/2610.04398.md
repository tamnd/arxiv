\label{thm:A}
Let $r,m,n\in\mathbb N$, $1\le p\le\infty$, and put
\[
 d:=\min\{r,m,n\},
 \qquad
 a(p):=\left|\frac1p-\frac12\right|.
\]
Then
\begin{equation}\label{eq:main-bottleneck}
 B_d^{\mathbb K}d^{a(p)}
 \le
 \Gamma_p^{\mathbb K}(r;m,n)
 \le
 d^{a(p)}\Gamma_2^{\mathbb K}(r;m,n)
 \le
 \kappa_G^{\mathbb K}(d)d^{a(p)}.
\end{equation}
More precisely, for every $A\in\mathbb K^{m\times n}$,
\begin{equation}\label{eq:Hilbert-saturation}
 G_2^{(r)}(A)=G_2^{(d)}(A),
\end{equation}
and
\begin{equation}\label{eq:pointwise-bottleneck}
 G_p^{(r)}(A)
 \le
 d^{a(p)}G_2^{(d)}(A)
 =d^{a(p)}G_2^{(r)}(A).
\end{equation}

\label{thm:B}
For every $d\in\mathbb N$,
\begin{equation}\label{eq:Bd-lower}
 B_d^{\mathbb K}\ge\beta_d^{\mathbb K}.
\end{equation}
Moreover,
\begin{equation}\label{eq:beta-rate}
 \beta_d^{\mathbb C}=1,
 \qquad
 1-\beta_d^{\mathbb R}=O(d^{-5/6}).
\end{equation}
Therefore, for all $r,m,n\in\mathbb N$ and all $1\le p\le\infty$, with $d=\min\{r,m,n\}$,
\begin{equation}\label{eq:uniform-rectangular}
 \beta_d^{\mathbb K}d^{a(p)}
 \le
 \Gamma_p^{\mathbb K}(r;m,n)
 \le
 \kappa_G^{\mathbb K}(d)d^{a(p)}.
\end{equation}
If $\mathbb K=\mathbb C$, the lower factor in \eqref{eq:uniform-rectangular} is exactly $1$ for every $d$. If $\mathbb K=\mathbb R$ and $d$ is a Hadamard order, it is also exactly $1$. In particular,
\begin{equation}\label{eq:rectangular-order}
 \Gamma_p^{\mathbb K}(r;m,n)
 \asymp
 \min\{r,m,n\}^{|1/p-1/2|}
\end{equation}
with constants independent of $r,m,n$ and $p$.

\label{thm:C}
Let $r,m,n\in\mathbb N$.

\medskip
\noindent
\textup{(i)} For every $1\le p\le\infty$, over either scalar field,
\begin{equation}\label{eq:rectangular-lower-plus}
\Gamma_p^\mathbb K(r;m,n)
\ge
\sup_{1\le k\le k_+}
k^{-1/p}\mathcal R_{\mathbb K}(m,k).
\end{equation}
Equivalently,
\begin{equation}\label{eq:rectangular-lower-plus-normalized}
\Gamma_p^\mathbb K(r;m,n)
\ge
\sup_{1\le k\le k_+}
\frac{\mathcal R_{\mathbb K}(m,k)}{\sqrt{k}}\,
k^{\,1/2-1/p}.
\end{equation}

\medskip
\noindent
\textup{(ii)} For every $1\le p\le\infty$, over either scalar field,
\begin{equation}\label{eq:rectangular-lower-minus}
\Gamma_p^\mathbb K(r;m,n)
\ge
\sup_{1\le k\le k_-}
k^{-1/p'}\mathcal R_{\mathbb K}(n,k),
\end{equation}
or equivalently
\begin{equation}\label{eq:rectangular-lower-minus-normalized}
\Gamma_p^\mathbb K(r;m,n)
\ge
\sup_{1\le k\le k_-}
\frac{\mathcal R_{\mathbb K}(n,k)}{\sqrt{k}}\,
k^{\,1/p-1/2}.
\end{equation}
Consequently, for every $1\le p\le\infty$,
\begin{equation}\label{eq:rectangular-lower-envelope}
\Gamma_p^\mathbb K(r;m,n)
\ge
\max\left\{
\sup_{1\le k\le k_+}k^{-1/p}\mathcal R_{\mathbb K}(m,k),
\sup_{1\le k\le k_-}k^{-1/p'}\mathcal R_{\mathbb K}(n,k)
\right\}.
\end{equation}

\medskip
\noindent
\textup{(iii)} Suppose that $\mathbb K=\mathbb C$. If $2\le p\le\infty$, then
\begin{equation}\label{eq:rectangular-upper-plus}
\Gamma_p^\mathbb C(r;m,n)
\le
\Gamma_2^\mathbb C(r;m,n)^{2/p}
\,
\mathcal R_{\mathbb C}(m,k_+)^{\,1-2/p}.
\end{equation}
Equivalently,
\begin{equation}\label{eq:rectangular-upper-plus-ratio}
\Gamma_p^\mathbb C(r;m,n)
\le
\Gamma_2^\mathbb C(r;m,n)
\left(
\frac{\mathcal R_{\mathbb C}(m,k_+)}
     {\Gamma_2^\mathbb C(r;m,n)}
\right)^{2a(p)}.
\end{equation}

\medskip
\noindent
\textup{(iv)} Suppose that $\mathbb K=\mathbb C$. If $1\le p\le2$, then
\begin{equation}\label{eq:rectangular-upper-minus}
\Gamma_p^\mathbb C(r;m,n)
\le
\mathcal R_{\mathbb C}(n,k_-)^{\,2/p-1}
\,
\Gamma_2^\mathbb C(r;m,n)^{\,2-2/p},
\end{equation}
or, equivalently,
\begin{equation}\label{eq:rectangular-upper-minus-ratio}
\Gamma_p^\mathbb C(r;m,n)
\le
\Gamma_2^\mathbb C(r;m,n)
\left(
\frac{\mathcal R_{\mathbb C}(n,k_-)}
     {\Gamma_2^\mathbb C(r;m,n)}
\right)^{2a(p)}.
\end{equation}

\label{thm:D}
Let $(r_N)$, $(m_N)$ and $(n_N)$ be sequences in $\mathbb N$ such
that
\[
d_N:=\min\{r_N,m_N,n_N\}\longrightarrow\infty,
\]
and let $p_N\in[1,\infty]$. Put
\begin{equation}\label{eq:tauN}
\tau_N:=
\left|\frac1{p_N}-\frac12\right|\log d_N.
\end{equation}
Then the following statements hold.

\begin{enumerate}[label=\textup{(\roman*)}]
\item The constants stay uniformly bounded precisely when
$(\tau_N)$ is bounded:
\[
\sup_N\Gamma_{p_N}^{\mathbb K}(r_N;m_N,n_N)<\infty
\quad\Longleftrightarrow\quad
\sup_N\tau_N<\infty.
\]

\item If $\tau_N\to0$, then
\begin{equation}\label{eq:subcritical-KG}
\Gamma_{p_N}^{\mathbb K}(r_N;m_N,n_N)
\longrightarrow K_G^{\mathbb K}.
\end{equation}

\item If $\tau_N\to\infty$, then
\begin{equation}\label{eq:supercritical-log}
\frac{\log\Gamma_{p_N}^{\mathbb K}(r_N;m_N,n_N)}
{\tau_N}
\longrightarrow1.
\end{equation}

\item Suppose that $\tau_N\to\tau\in[0,\infty)$, and write
\[
\underline B_{\mathbb K}:=\liminf_{d\to\infty}B_d^{\mathbb K}.
\]
Then
\begin{equation}\label{eq:critical-envelope}
\max\{K_G^{\mathbb K},e^\tau\underline B_{\mathbb K}\}
\le
\liminf_N\Gamma_{p_N}^{\mathbb K}(r_N;m_N,n_N)
\le
\limsup_N\Gamma_{p_N}^{\mathbb K}(r_N;m_N,n_N)
\le K_G^{\mathbb K}e^\tau.
\end{equation}
Moreover, $\underline B_{\mathbb K}\ge1$.
\end{enumerate}

Thus uniform boundedness is equivalent to
\[
\left|\frac1{p_N}-\frac12\right|
=O\!\left(\frac1{\log d_N}\right).
\]
More specifically, the transition window $\tau_N\asymp1$
corresponds to
\begin{equation}\label{eq:critical-scale}
\left|\frac1{p_N}-\frac12\right|
\asymp\frac1{\log d_N}.
\end{equation}
When $p_N\to2$, the latter condition is equivalent to
$|p_N-2|\asymp1/\log d_N$.

\label{prop:endpoint}
For both scalar fields,
\begin{equation}\label{eq:endpoint-rho}
 \Gamma_\infty^{\mathbb K}(d;d,d)
=\rho(\ell_1^d(\mathbb K),\ell_1^d(\mathbb K)),
\end{equation}
For $d\ge1$ set
\begin{equation}\label{eq:Md-def}
 M_d:=2^{-d}\sum_{\varepsilon\in\{-1,1\}^d}
 \left|\varepsilon_1+\cdots+\varepsilon_d\right|.
\end{equation}
Over $\mathbb R$,
\begin{equation}\label{eq:endpoint-real}
 B_d^{\mathbb R}
 =\frac1{\sqrt d}\pi_1^{(d)}(I_{\ell_1^d})
 \le
 \frac{\sqrt d}{M_d},
\end{equation}
and therefore
\begin{equation}\label{eq:endpoint-real-limsup}
 \limsup_{d\to\infty}B_d^{\mathbb R}
 \le\sqrt{\frac\pi2}.
\end{equation}
Over $\mathbb C$,
\begin{equation}\label{eq:endpoint-complex}
 1\le B_d^{\mathbb C}\le\frac2{\sqrt\pi}.
\end{equation}

\label{prop:complex-profile}
For every $d\ge2$, the function $F_d$ is even, convex and $1$-Lipschitz on $[-L_d,L_d]$. For $0\le s\le L_d$ set
\[
 H_d(s):=F_d(s)-s.
\]
Then $H_d$ is convex and nonincreasing, with
\[
 H_d(0)=\log\Gamma_2^{\mathbb C}(d;d,d),
 \qquad
 H_d(L_d)=\log B_d^{\mathbb C}.
\]
Therefore,
\begin{align}
 \max\{\log\Gamma_2^{\mathbb C}(d;d,d),\,|s|+\log B_d^{\mathbb C}\}
 &\le F_d(s)\label{eq:profile-lower}\\
 &\le |s|
 +\left(1-\frac{2|s|}{\log d}\right)
   \log\Gamma_2^{\mathbb C}(d;d,d)
 +\frac{2|s|}{\log d}\log B_d^{\mathbb C}.
 \label{eq:profile-upper}
\end{align}
Every sequence $d_k\to\infty$ has a subsequence along which $F_{d_k}$ converges locally uniformly on $\mathbb R$ and $B_{d_k}^{\mathbb C}\to B\in[1,2/\sqrt\pi]$. Every such limit satisfies
\begin{equation}\label{eq:profile-limit}
 \max\{\log K_G^{\mathbb C},|s|+\log B\}
 \le F(s)
 \le \log K_G^{\mathbb C}+|s|.
\end{equation}

\label{lem:Lewis-reduction}
Let $A\in\mathbb K^{m\times n}$ and $1\le p\le\infty$. If
$s=\min\{r,m\}$, then
\begin{equation}\label{eq:Lewis-x}
G_p^{(r)}(A)\le s^{a(p)}G_2^{(s)}(A).
\end{equation}
Likewise, if $t=\min\{r,n\}$, then
\begin{equation}\label{eq:Lewis-y}
G_p^{(r)}(A)\le t^{a(p)}G_2^{(t)}(A).
\end{equation}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Fix admissible vectors
$x_1,\ldots,x_m\in B_{\ell_p^r}$ and
$y_1,\ldots,y_n\in B_{\ell_{p'}^r}$, and put
\[
E=\operatorname{span}\{x_1,\ldots,x_m\},
\qquad q=\dim E.
\]
If $q=0$, the expression to be estimated is zero. Otherwise
$q\le s$, and \eqref{eq:Lewis-distance} gives an isomorphism
$U:E\to\ell_2^q$ which can be normalized so that
\begin{equation}\label{eq:U-normalization}
\left\lVert U\right\rVert\le1,
\qquad
\left\lVert U^{-1}\right\rVert\le q^{a(p)}.
\end{equation}
Set $u_i=Ux_i$. Then $\left\lVert u_i\right\rVert_2\le1$.

For each $j$, restriction of the pairing with $y_j$ defines a
functional on $E$:
\[
\varphi_j(x)=\left\langle x,y_j\right\rangle.
\]
Hölder's inequality gives
$\left\lVert \varphi_j\right\rVert_{E^*}\le\left\lVert y_j\right\rVert_{p'}\le1$.
Consequently,
\[
\left\lVert \varphi_j\circ U^{-1}\right\rVert_{(\ell_2^q)^*}
\le\left\lVert U^{-1}\right\rVert\le q^{a(p)}.
\]
Identify $(\ell_2^q)^*$ with $\ell_2^q$ using the bilinear
pairing in \eqref{eq:G-rmn}. This identification is isometric
over both scalar fields. We may therefore write
\[
\varphi_j(U^{-1}u)=\left\langle u,v_j\right\rangle
\quad(u\in\ell_2^q),
\qquad
\left\lVert v_j\right\rVert_2\le q^{a(p)}.
\]
The vectors $w_j=q^{-a(p)}v_j$ belong to $B_{\ell_2^q}$.
It follows that
\begin{align*}
\left|\sum_{i=1}^m\sum_{j=1}^n
a_{ij}\left\langle x_i,y_j\right\rangle\right|
&=
q^{a(p)}
\left|\sum_{i=1}^m\sum_{j=1}^n
a_{ij}\left\langle u_i,w_j\right\rangle\right|\\
&\le q^{a(p)}G_2^{(q)}(A)\\
&\le s^{a(p)}G_2^{(s)}(A).
\end{align*}
The last step uses $q\le s$ and zero padding. Taking the
supremum over the original vectors proves \eqref{eq:Lewis-x}.

To obtain \eqref{eq:Lewis-y}, apply \eqref{eq:Lewis-x} to
$A^T$ with exponent $p'$. Indeed,
$G_p^{(r)}(A)=G_{p'}^{(r)}(A^T)$ and $a(p')=a(p)$.

<!-- proof-4: body_md -->
Put $d=\min\{r,m,n\}$. One of the two quantities
$\min\{r,m\}$ and $\min\{r,n\}$ equals $d$. Applying the
corresponding estimate from Lemma \ref{lem:Lewis-reduction}
gives, for every $A\in\mathbb K^{m\times n}$,
\begin{equation}\label{eq:A-pointwise-first}
G_p^{(r)}(A)\le d^{a(p)}G_2^{(d)}(A).
\end{equation}

At $p=2$, this says $G_2^{(r)}(A)\le G_2^{(d)}(A)$.
The reverse inequality follows by embedding $\ell_2^d$ into
$\ell_2^r$. Hence
\[
G_2^{(r)}(A)=G_2^{(d)}(A),
\]
which proves \eqref{eq:Hilbert-saturation} and completes
\eqref{eq:pointwise-bottleneck}. Dividing
\eqref{eq:A-pointwise-first} by $\left\lVert A\right\rVert_{\infty,1}$ and taking
the supremum over nonzero $A$ yields
\begin{equation}\label{eq:A-upper-refined}
\Gamma_p^{\mathbb K}(r;m,n)
\le d^{a(p)}\Gamma_2^{\mathbb K}(r;m,n)
\le d^{a(p)}\kappa_G^{\mathbb K}(d).
\end{equation}

For the lower bound, zero padding gives
\begin{equation}\label{eq:square-padding}
\Gamma_p^{\mathbb K}(r;m,n)\ge\Gamma_p^{\mathbb K}(d;d,d).
\end{equation}
Suppose first that $p\ge2$. If
$x_i\in B_{\ell_\infty^d}$ and $y_j\in B_{\ell_1^d}$,
then
\[
\left\lVert d^{-1/p}x_i\right\rVert_p\le1,
\qquad
\left\lVert y_j\right\rVert_{p'}\le1.
\]
Thus every configuration admissible for the square
$p=\infty$ problem yields, after rescaling the $x_i$,
an admissible configuration for the $p$ problem. Taking
suprema gives
\begin{align}
\Gamma_p^{\mathbb K}(d;d,d)
&\ge d^{-1/p}\Gamma_\infty^{\mathbb K}(d;d,d)\notag\\
&=B_d^{\mathbb K}d^{1/2-1/p}
 =B_d^{\mathbb K}d^{a(p)}.
\label{eq:A-lower-pge2}
\end{align}
For $p\le2$, use Lemma \ref{lem:duality} and apply the same
argument to $p'\ge2$; the square dimensions remain unchanged.
Together with \eqref{eq:square-padding}, this proves
\eqref{eq:main-bottleneck}.

\label{lem:rectangular-infinity}
Let $A=(a_{ij})\in\mathbb K^{m\times n}$ and $r\in\mathbb N$. Then
\begin{equation}\label{eq:rectangular-infinity-exact}
G_\infty^{(r)}(A)
=
\max_{\substack{
\sigma:\{1,\ldots,n\}\to\{1,\ldots,r\}\\
\zeta_1,\ldots,\zeta_n\in\mathbb T_{\mathbb K}}}
\sum_{i=1}^{m}\sum_{k=1}^{r}
\left|
\sum_{\substack{1\le j\le n\\ \sigma(j)=k}}
a_{ij}\zeta_j
\right|.
\end{equation}
In particular, if $r\ge n$, then
\begin{equation}\label{eq:rectangular-infinity-saturation}
G_\infty^{(r)}(A)
=
\sum_{i=1}^{m}\sum_{j=1}^{n}|a_{ij}|.
\end{equation}
Thus, for $A\in\mathbb K^{d\times d}$,
\begin{equation}\label{eq:G-infinity-exact}
G_\infty^{(d)}(A)
=
\sum_{i,j=1}^{d}|a_{ij}|.
\end{equation}

<!-- proof-6: body_md -->
Fix $y_1,\ldots,y_n\in B_{\ell_1^r}$ and set
\[
b_{ik}:=\sum_{j=1}^{n}a_{ij}(y_j)_k
\qquad(1\le i\le m,\ 1\le k\le r).
\]
Since the coordinates of the vectors
$x_i\in B_{\ell_\infty^r}$ can be chosen independently,
\[
\sup_{x_1,\ldots,x_m\in B_{\ell_\infty^r}}
\left|\sum_{i=1}^{m}\sum_{k=1}^{r}x_{ik}b_{ik}\right|
=
\sum_{i=1}^{m}\sum_{k=1}^{r}|b_{ik}|.
\]
Indeed, the upper bound is the triangle inequality. For the
reverse inequality, take $x_{ik}=\overline{b_{ik}}/|b_{ik}|$
when $b_{ik}\ne0$ in the complex case, and the corresponding
sign in the real case; choose any scalar of modulus one when
$b_{ik}=0$. Consequently,
\[
G_\infty^{(r)}(A)
=
\max_{y_1,\ldots,y_n\in B_{\ell_1^r}}
\sum_{i=1}^{m}\sum_{k=1}^{r}
\left|\sum_{j=1}^{n}a_{ij}(y_j)_k\right|.
\]

The function on the right is continuous and convex in each
$y_j$ separately. Its maximum is attained because the domain
is compact. Starting from a maximizing tuple, maximize
successively over each $y_j$ while keeping the other variables
fixed. A convex function on the finite-dimensional $\ell_1^r$
unit ball has a maximizer at an extreme point. These extreme
points are precisely $\zeta e_k$, where
$\zeta\in\mathbb T_{\mathbb K}$ and $1\le k\le r$. Each successive
replacement preserves the global maximum. We may therefore
take $y_j=\zeta_j e_{\sigma(j)}$ for all $j$, which gives
\eqref{eq:rectangular-infinity-exact}.

The triangle inequality bounds the right-hand side of
\eqref{eq:rectangular-infinity-exact} by
$\sum_{i,j}|a_{ij}|$. If $r\ge n$, choose $\sigma$ injective;
then every inner sum has at most one term and this upper
bound is attained. This proves
\eqref{eq:rectangular-infinity-saturation}, of which
\eqref{eq:G-infinity-exact} is the square case.

\label{prop:endpoint-compression}
Let $r,m,n\in\mathbb N$ and put $k=\min\{r,n\}$. Then
\begin{equation}\label{eq:endpoint-compression}
\Gamma_\infty^\mathbb K(r;m,n)

\Gamma_1^\mathbb K(r;m,n)

<!-- proof-7: body_md -->
For a matrix $C=(c_{i\ell})\in\mathbb K^{m\times k}$, the canonical
identifications for $\ell_1$ tensor products give
\begin{equation}\label{eq:matrix-tensor-ratio}
\rho\bigl(\ell_1^m(\mathbb K),\ell_1^k(\mathbb K)\bigr)
=
\sup_{C\ne0}
\frac{\sum_{i=1}^m\sum_{\ell=1}^k|c_{i\ell}|}
     {\left\lVert C\right\rVert_{\infty,1}}.
\end{equation}

Fix $A=(a_{ij})\in\mathbb K^{m\times n}$. By
Lemma~\ref{lem:rectangular-infinity}, there are an assignment
$\sigma:\{1,\ldots,n\}\to\{1,\ldots,r\}$ and phases
$\zeta_j\in\mathbb T_\mathbb K$ such that
\[
G_\infty^{(r)}(A)
=
\sum_{i=1}^m\sum_{\ell=1}^r|b_{i\ell}|,
\qquad
b_{i\ell}:=
\sum_{\sigma(j)=\ell}a_{ij}\zeta_j.
\]
At most $k$ columns of $B=(b_{i\ell})$ are nonzero. Deleting
the remaining columns does not change either
$\sum_{i,\ell}|b_{i\ell}|$ or $\left\lVert B\right\rVert_{\infty,1}$.

Furthermore,
\begin{equation}\label{eq:compressed-matrix-norm}
\left\lVert B\right\rVert_{\infty,1}\le\left\lVert A\right\rVert_{\infty,1}.

\label{rem:rectangular-endpoint-saturation}
Let $\omega=e^{2\pi i/3}$ and consider
\[
A=
\begin{pmatrix}
1&1&1\\
1&\omega&\omega^2
\end{pmatrix}
\in\mathbb C^{2\times3}.
\]
Then
\begin{equation}\label{eq:rectangular-endpoint-example}
G_\infty^{(2)}(A)=2+2\sqrt3
<6=G_\infty^{(3)}(A).
\end{equation}
Indeed, the two rows of $A$ are orthogonal and have squared
Euclidean norm $3$, so $AA^*=3I_2$. By
\eqref{eq:rectangular-infinity-exact}, an assignment of the three
columns to two coordinates either places all three in one group
or places two in one group and the remaining column in the other.

In the first case, for $z\in\mathbb T_{\mathbb C}^3$,
\[
\|Az\|_1
\le\sqrt2\,\|Az\|_2
\le\sqrt2\,\|A\|_{2\to2}\|z\|_2
=3\sqrt2
<2+2\sqrt3.
\]
In the second case, the singleton column contributes $2$.
If $a_j,a_k$ are the two columns in the other group, then
$\|a_j\|_2^2=\|a_k\|_2^2=2$ and $|a_j^*a_k|=1$, where $a_j^*a_k$
denotes the usual Hermitian inner product on $\mathbb C^2$. Hence, for
$\zeta_j,\zeta_k\in\mathbb T_{\mathbb C}$,
\[
\|\zeta_j a_j+\zeta_k a_k\|_1
\le\sqrt2\,\|\zeta_j a_j+\zeta_k a_k\|_2
\le\sqrt2\,\sqrt{4+2}=2\sqrt3.
\]
This bound is attained by grouping the first two columns and
choosing $\zeta_1=1$, $\zeta_2=e^{-i\pi/3}$: both coordinates
of $\zeta_1a_1+\zeta_2a_2$ have modulus $\sqrt3$.
This proves the first equality in
\eqref{eq:rectangular-endpoint-example}. The second follows
from \eqref{eq:rectangular-infinity-saturation}, since $r=3=n$.

On the other hand, Theorem~\ref{thm:A} gives
$G_2^{(2)}(A)=G_2^{(3)}(A)$, because
$\min\{2,m,n\}=2$. Thus the Hilbertian stabilization at the
effective dimension need not be a pointwise stabilization of
$G_\infty^{(r)}(A)$.

\label{lem:Hadamard-truncation}
Let $h\ge d$ and suppose that there exists a real Hadamard matrix $H$ of order $h$. If $A$ is any $d\times d$ corner of $H$, then
\begin{equation}\label{eq:corner-infty1}
 \left\lVert A\right\rVert_{\infty,1}\le d\sqrt h.
\end{equation}
Therefore,
\begin{equation}\label{eq:corner-B}
 B_d^{\mathbb R}\ge\sqrt{\frac dh}.
\end{equation}

 &\le \sqrt d\,\left\lVert Az\right\rVert_2

For the real field, apply Lemma \ref{lem:Hadamard-truncation} with
$h=h_+(d)$ to obtain
\begin{equation}\label{eq:real-hplus-lower}
B_d^{\mathbb R}\ge\sqrt{\frac{d}{h_+(d)}}=\beta_d^{\mathbb R}.
\end{equation}
To make the quantitative estimate explicit, write the Hadamard orders
in increasing order as $h_1<h_2<\cdots$ and define the Hadamard gap
function $\gamma(x):=\max\{h_{i+1}-h_i:h_i\le x\}$ for $x\ge0$.
Livinskyi's asymptotic existence result, in the form recorded by
Brent, Osborn and Smith \cite{BrentOsbornSmith2016}, implies that the gaps
between consecutive Hadamard orders satisfy
\begin{equation}\label{eq:Hadamard-gap-function}
\gamma(x)=O(x^{1/6})\qquad\text{as }x\to\infty.
\end{equation}
If $d$ is not itself a Hadamard order, choose consecutive Hadamard
orders $h_i<d<h_{i+1}$. Then $h_{i+1}=h_+(d)$ and
\[
0\le h_+(d)-d\le h_{i+1}-h_i\le\gamma(d)=O(d^{1/6}).
\]
If $d$ is already a Hadamard order, then $h_+(d)=d$ and $h_+(d)-d=0$.
Thus, unconditionally,
\begin{equation}\label{eq:Hadamard-gap-rate}
h_+(d)-d=O(d^{1/6}).
\end{equation}
Setting $x_d:=(h_+(d)-d)/d$, we have $x_d=O(d^{-5/6})$. Since $x_d\ge0$,
\begin{align*}
1-\beta_d^{\mathbb R}
&=1-(1+x_d)^{-1/2}\\
&=\frac{(1+x_d)-1}{\sqrt{1+x_d}\,(1+\sqrt{1+x_d})}\\
&=\frac{x_d}{\sqrt{1+x_d}\,(1+\sqrt{1+x_d})}\\
&\le\frac{x_d}{2}
=O(d^{-5/6}).
\end{align*}
If $d$ is a Hadamard order, $h_+(d)=d$ and \eqref{eq:real-hplus-lower}
gives $B_d^{\mathbb R}\ge1$ identically.

immediately yields the universal lower bound $\beta_d^{\mathbb R}\ge 1/\sqrt2$

\end{align*}

\begin{align*}
 M_{2m}
 &=2^{-2m}\sum_{k=0}^{2m}|2m-2k|\binom{2m}{k}\\
 &=2^{1-2m}\sum_{k=0}^{m-1}(2m-2k)\binom{2m}{k}\\
 &=2^{2-2m}\sum_{k=0}^{m-1}(m-k)\binom{2m}{k}.
\end{align*}

\[

\begin{align}

 &=2^{2-2m}m\binom{2m-1}{m-1}\notag\\
