  Let $f$ be twice continuously differentiable and satisfy~\eqref{eq:lip_hess}. Then, for every $x,y\in\mathcal{X}$,

      \bigl|f(y)&-f(x)-\left\langle\nabla f(x),y-x\right\rangle

<!-- head-1: body_md -->
We use the potential
\begin{equation*}
  E_t = A_t(f(w_t)-f^*)+\frac{\lambda_t}{2}\|{z_t-x^*} \|^2.
\end{equation*}
Taking expectation and using H\"older inequality gives
\begin{equation*}
  \mathbb{E} \left[ f(w_T) - f^\ast \right] \leq \mathbb{E} \left[ E_T A_T^{-1} \right] \leq \left( \mathbb{E} E_T^2 \right)^{1/2} \left( \mathbb{E} A_T^{-2} \right)^{1/2}. 
\end{equation*}
The damping rule makes $A_T$ depend on the stochastic
derivatives. In the proof, we show that the following bounds hold
\begin{equation*}
  \left(\mathbb{E} E_T^2\right)^{1/2}\lesssim\lambda_TD^2,
  \qquad
  \left(\mathbb{E} A_T^{-2}\right)^{1/2}\lesssim\tfrac{D}{T^{7/2}}.
\end{equation*}
They give the rate of Theorem~\ref{thm:optimal_convergence}.

Using the damping condition~\eqref{eq:damping_condition},
we combine the optimality conditions of the two subproblems
in lines~\ref{lin:cubic_sub_acc} and~\ref{lin:avg_sub_acc}
with convexity to obtain
\begin{equation}
  \label{eq:potential_one_step_main}
  E_t+\tfrac{a_t\tau_t}{8\alpha_t}\|{s_t} \|^2
  \leq E_{t-1}+\tfrac{D^2}{2}(\lambda_t-\lambda_{t-1})+\delta_t+\ell_t,
\end{equation} 
where $s_t=y_t-v_t$. 
The last two terms on the right are due to the stochastic oracle errors. As in the analysis of the basic method, for the $\widetilde{g}$'s error we use the decomposition
\begin{equation*}
    z_t-x^*=(z_t-z_{t-1})+(z_{t-1}-x^*).
\end{equation*}
The term involving $z_t-z_{t-1}$ is bounded using Young's inequality
and the diameter bound, and included in $\delta_t$, together with
the errors of $g_t$ and $H_t$. The remaining term
\begin{equation*}
    \ell_t
    =
    -a_t\left\langle
    \widetilde g_t-\nabla f(y_t),
    z_{t-1}-x^*
    \right\rangle
\end{equation*}
is kept linear. Since $a_t$, $y_t$, and $z_{t-1}$ are fixed before the gradient $\widetilde g_t$ is queried, unbiasedness gives $\mathbb{E}\ell_t=0$.
Summing~\eqref{eq:potential_one_step_main}, we get
\begin{equation}
  \label{eq:potential_telescope_main}
  E_T + \frac{1}{8} \mathop{\textstyle\sum}_{t=1}^T \frac{a_t \tau_t}{\alpha_t}\|s_t\|^2 \leq \frac{\lambda_T D^2}{2} + \mathop{\textstyle\sum}_{t=1}^T \delta_t + \mathop{\textstyle\sum}_{t=1}^T \ell_t.
\end{equation}
Thus, to bound $\left( \mathbb{E} E_T^2 \right)^{1/2}$, we need to estimate second moments of the sums of $\delta$'s and $\ell$'s. Since $\ell_t$ has zero mean conditionally on the past, the cross terms vanish and
\begin{equation*}
  \mathbb{E} \left( \mathop{\textstyle\sum}_{t=1}^T \ell_t \right)^2 = \mathop{\textstyle\sum}_{t=1}^T \mathbb{E} \ell_t^2 \leq \mathop{\textstyle\sum}_{t=1}^T  D^2 \sigma_1^2 \mathbb{E} a_t^2\lesssim \sigma_1^2 T^6 \stackrel{\eqref{eq:acc_params}}{\lesssim} \lambda_T^2 D^4,
\end{equation*}
where $\tau_t \geq \beta_t$ and~\eqref{eq:damping_condition} give
$a_t\leq t^{3/2}/(\alpha_tD)$. The contraction bound $\|{s_t} \|\leq\alpha_tD$
and the regularization parameters~\eqref{eq:acc_params}
also yield a second-moment bound of order
$\lambda_T^2D^4$ for $\mathop{\textstyle\sum}_{t=1}^T\delta_t$ (see Lemma~\ref{lem:optimal_error_moments}). Consequently, 
\begin{equation}
  \label{eq:potential_rate}
  \left( \mathbb{E}  E_T^2 \right) ^{1/2} \lesssim \lambda_T D^2.
\end{equation} 
It remains to show the bound on $\left( \mathbb{E} A_T^{-2} \right)^{1/2}$.  By~\eqref{eq:A_relation_main}, there are two types of iterations. If all iterations $t=2,\ldots,T$ are undamped, then
\begin{equation*}
  A_T^{-1} = A_1^{-1}\mathop{\textstyle\prod}_{t=2}^T (1-\alpha_t) \stackrel{\eqref{eq:acc_params}}{\lesssim} \frac{D}{T^4}.
\end{equation*}
On damped iterations, $a_t=\lambda_t/(\alpha_t\tau_t)$, so
\begin{equation*}
  A_t^{-1}\leq(1-\alpha_t)A_{t-1}^{-1}+\alpha_t^3\left(\tfrac{D}{t^{3/2}}+\|{s_t} \|\right),
\end{equation*}
while at undamped iterations the last term is absent. Multiplying by $t(t+1)(t+2)(t+3)$ and summing gives
\begin{equation*}
  A_T^{-1}\lesssim\frac{D}{T^{7/2}}
  +\frac{1}{T^4}\sum_{\substack{t\leq T\\a_t<\alpha_tA_t}}\frac{\|{s_t} \|}{\alpha_t}.
\end{equation*}
On damped iterations,~\eqref{eq:damping_condition}
holds with equality, so the squared-step term
in~\eqref{eq:potential_one_step_main} is
$\lambda_t\|{s_t} \|^2/(8\alpha_t^2)$.
A weighted summation of the normalized one-step bound
controls the second moment of the sum above by
$\mathcal{O}(D^2T)$  (see Lemma~\ref{lem:optimal_movement}).
Substituting this estimate and using Minkowski inequality yields
\begin{equation*}
  \left(\mathbb{E} A_T^{-2}\right)^{1/2}
  \lesssim\frac{D}{T^{7/2}}.
\end{equation*}
Combining this bound with~\eqref{eq:potential_rate} and using the definition of $\lambda_t$ yields the desired rate. 

The full proof is given in Appendix~\ref{sec:proof_optimal}.

\label{lem:basic_one_step}
  Let $f$ be convex and satisfy Assumption~\ref{as:lip_hess}, and let $M\geq 0$. For the iterates of Algorithm~\ref{alg:basic_method}, with $0<\alpha_t\leq1$, $\eta_t\geq0$, and $H_t\succeq0$, we have
  \begin{equation}
    \label{eq:basic_one_step}
    \begin{aligned}
      f(x_{t+1})-f^*
      \leq{}& (1-\alpha_t)\left( f(x_t)-f^*\right) \\
      &+\frac{\alpha_t\eta_t}{2}\left(
        \|{u_t-x^*} \|^2-\|{u_{t+1}-x^*} \|^2-\|{u_{t+1}-u_t} \|^2
      \right) \\
      &+\alpha_t\left\langle\nabla f(x_t)-g_t,u_{t+1}-x^*\right\rangle
      +\alpha_t^2D^2\|{H_t-\nabla^2 f(x_t)} \|
      +\frac{\max\left\lbrace M,L_2\right\rbrace}{3}\alpha_t^3D^3.
    \end{aligned}
  \end{equation}

<!-- proof-1: body_md -->
Optimality of $u_{t+1}$ over $\mathcal{X}$ gives
  \begin{equation*}
    \left\langle\nabla\omega_t^M(u_{t+1}),x-u_{t+1}\right\rangle\geq0,
    \qquad x\in\mathcal{X}.
  \end{equation*}
  Since $H_t\succeq0$, the model $\omega_t^M$ is convex and, when $\eta_t>0$, is $\eta_t$-strongly convex. Therefore,
  \begin{equation*}
    \omega_t^M(x)
    \geq\omega_t^M(u_{t+1})
      +\left\langle\nabla\omega_t^M(u_{t+1}),x-u_{t+1}\right\rangle
      +\frac{\eta_t}{2}\|{x-u_{t+1}} \|^2
    \geq\omega_t^M(u_{t+1})+\frac{\eta_t}{2}\|{x-u_{t+1}} \|^2.
  \end{equation*}
  Rearranging yields
  \begin{equation}
    \label{eq:basic_model_comparison}
    \omega_t^M(u_{t+1})
    \leq\omega_t^M(x)-\frac{\eta_t}{2}\|{u_{t+1}-x} \|^2,
    \qquad x\in\mathcal{X}.
  \end{equation}

  By Lemma~\ref{lem:taylor_remainder} with $x=x_t$ and $y=x_{t+1}$, using $x_{t+1}-x_t=\alpha_t(u_{t+1}-x_t)$, we obtain
  \begin{equation}
    \label{eq:basic_model_upper}
    \begin{aligned}
      f(x_{t+1})
      &\leq f(x_t)+\alpha_t\left\langle\nabla f(x_t),u_{t+1}-x_t\right\rangle \\
      &\quad+\frac{\alpha_t^2}{2}\left\langle\nabla^2 f(x_t)(u_{t+1}-x_t),u_{t+1}-x_t\right\rangle
      +\frac{L_2\alpha_t^3}{6}\|{u_{t+1}-x_t} \|^3 \\
      &\stackrel{\eqref{eq:cubic_model}}{=} f(x_t)+\alpha_t\omega_t^M(u_{t+1})
      -\frac{\alpha_t\eta_t}{2}\|{u_{t+1}-u_t} \|^2 
      +\alpha_t\left\langle\nabla f(x_t)-g_t,u_{t+1}-x_t\right\rangle \\
      &\quad+\frac{\alpha_t^2}{2}\left\langle\left(\nabla^2 f(x_t)-H_t\right)(u_{t+1}-x_t),u_{t+1}-x_t\right\rangle 
      +\frac{(L_2-M)\alpha_t^3}{6}\|{u_{t+1}-x_t} \|^3 \\
      &\leq f(x_t)+\alpha_t\omega_t^M(u_{t+1})
      -\frac{\alpha_t\eta_t}{2}\|{u_{t+1}-u_t} \|^2 
      +\alpha_t\left\langle\nabla f(x_t)-g_t,u_{t+1}-x_t\right\rangle \\
      &\quad+\frac{\alpha_t^2D^2}{2}\|{H_t-\nabla^2 f(x_t)} \| +\frac{(L_2-M)_{+}}{6}\alpha_t^3D^3\\
      &\stackrel{\eqref{eq:basic_model_comparison}}{\leq} f(x_t)+\alpha_t\omega_t^M(x)
      -\frac{\alpha_t\eta_t}{2}\left(\|{u_{t+1}-x} \|^2+\|{u_{t+1}-u_t} \|^2\right)
      +\alpha_t\left\langle\nabla f(x_t)-g_t,u_{t+1}-x_t\right\rangle \\
      &\quad +\frac{\alpha_t^2D^2}{2}\|{H_t-\nabla^2 f(x_t)} \| +\frac{(L_2-M)_+}{6}\alpha_t^3D^3,
    \end{aligned}
  \end{equation}
  where the second inequality uses $\|{u_{t+1}-x_t} \|\leq D$ and
  \begin{equation*}
    (L_2-M)\|{u_{t+1}-x_t} \|^3\leq(L_2-M)_+D^3,
    \qquad (L_2-M)_+=\max\left\lbrace L_2-M,0\right\rbrace.
  \end{equation*}

  Next, $(1-\alpha_t)x_t+\alpha_tx\in\mathcal{X}$ since $0<\alpha_t\leq1$. By Lemma~\ref{lem:taylor_remainder} at this point
  \begin{equation}
    \begin{aligned}
      \label{eq:basic_comparator_taylor}
      f\left((1-\alpha_t)x_t+\alpha_tx\right) 
      \geq{} & f(x_t)+\alpha_t\left\langle\nabla f(x_t),x-x_t\right\rangle\\
      &+\frac{\alpha_t^2}{2}\left\langle\nabla^2 f(x_t)(x-x_t),x-x_t\right\rangle
      -\frac{L_2\alpha_t^3}{6}\|{x-x_t} \|^3.
    \end{aligned}
  \end{equation}
  Therefore, by~\eqref{eq:basic_comparator_taylor} and the definition of $\omega_t^M(x)$~\eqref{eq:cubic_model},
  \begin{equation}
    \label{eq:basic_comparator_bound}
    \begin{aligned}
      f(x_t)+\alpha_t\omega_t^M(x)
      \stackrel{\eqref{eq:cubic_model}}{=}& f(x_t)+\alpha_t\left\langle\nabla f(x_t),x-x_t\right\rangle
      +\frac{\alpha_t^2}{2}\left\langle\nabla^2 f(x_t)(x-x_t),x-x_t\right\rangle\\
      &+\alpha_t\left\langle g_t-\nabla f(x_t),x-x_t\right\rangle\\
      &+\frac{\alpha_t^2}{2}\left\langle\left( H_t-\nabla^2 f(x_t)\right)(x-x_t),x-x_t\right\rangle\\
      &+\frac{M\alpha_t^3}{6}\|{x-x_t} \|^3
      +\frac{\alpha_t\eta_t}{2}\|{x-u_t} \|^2\\
      \stackrel{\eqref{eq:basic_comparator_taylor}}{\leq}& f\left((1-\alpha_t)x_t+\alpha_tx\right)
      +\frac{\alpha_t\eta_t}{2}\|{x-u_t} \|^2
      +\alpha_t\left\langle g_t-\nabla f(x_t),x-x_t\right\rangle\\
      &+\frac{\alpha_t^2}{2}\left\langle\left( H_t-\nabla^2 f(x_t)\right)(x-x_t),x-x_t\right\rangle
      +\frac{M+L_2}{6}\alpha_t^3\|{x-x_t} \|^3\\
      \leq{}& f\left((1-\alpha_t)x_t+\alpha_tx\right)
      +\frac{\alpha_t\eta_t}{2}\|{x-u_t} \|^2
      +\alpha_t\left\langle g_t-\nabla f(x_t),x-x_t\right\rangle\\
      &+\frac{\alpha_t^2D^2}{2}\|{H_t-\nabla^2 f(x_t)} \|
      +\frac{M+L_2}{6}\alpha_t^3D^3,
    \end{aligned}
  \end{equation}
  where the last inequality uses $\|{x-x_t} \|\leq D$.
  Substituting~\eqref{eq:basic_comparator_bound} into~\eqref{eq:basic_model_upper} with $x=x^*$ and using the convexity
  \begin{equation*}
    f\left((1-\alpha_t)x_t+\alpha_tx^*\right)
    \leq(1-\alpha_t)f(x_t)+\alpha_tf^*
  \end{equation*}
  together with $M+L_2+(L_2-M)_+=L_2+\max\left\lbrace M,L_2\right\rbrace\leq2\max\left\lbrace M,L_2\right\rbrace$ yields~\eqref{eq:basic_one_step}.

Let $f$ be convex. Let Assumptions~\ref{as:lip_hess},~\ref{as:stoch_grad},~\ref{as:unbiased_grad}, and~\ref{as:stoch_hess} hold. Then Algorithm~\ref{alg:basic_method} with $M\geq0$ and parameters~\eqref{eq:basic_reg} satisfies, for every integer $T\geq1$,
  \begin{equation}
    \label{eq:basic_rate}
      \mathbb{E}\left[ f(x_T)-f^*\right]
      \leq\frac{9\max\left\lbrace M,L_2\right\rbrace D^3}{(T+1)(T+2)} 
      +\frac{9\sigma_2D^2}{2(T+2)}+\frac{9\sigma_1D}{4\sqrt T}.
  \end{equation}
  In particular, if $M\geq0$ and $M=\mathcal{O}(L_2)$ with an absolute implied constant, then
  \begin{equation*}
    \mathbb{E}\left[ f(x_T)-f^*\right]
    =\mathcal{O}\left(\frac{\sigma_1D}{\sqrt T} + \frac{\sigma_2D^2}{T} + \frac{L_2D^3}{T^2}\right).
  \end{equation*}

<!-- proof-2: body_md -->
Suppose first that $\sigma_1>0$, so that $\eta_t>0$. By~\eqref{eq:unbiased_grad} we have
  \begin{equation*}
    \mathbb{E}\left[\left\langle\nabla f(x_t)-g_t,u_t-x^*\right\rangle
      \mid x_t,u_t\right]=0.
  \end{equation*}
  Splitting $u_{t+1}-x^*=(u_{t+1}-u_t)+(u_t-x^*)$ and applying Young's inequality gives
  \begin{gather*}
    \mathbb{E}\left[\left\langle\nabla f(x_t)-g_t,u_{t+1}-x^*\right\rangle
      -\frac{\eta_t}{2}\|{u_{t+1}-u_t} \|^2\right] 
    =\mathbb{E}\left[\left\langle\nabla f(x_t)-g_t,u_{t+1}-u_t\right\rangle
      -\frac{\eta_t}{2}\|{u_{t+1}-u_t} \|^2\right] \\
    \leq\frac{1}{2\eta_t}\mathbb{E}\|{g_t-\nabla f(x_t)} \|^2
    \stackrel{\eqref{eq:stoch_grad}}{\leq}\frac{\sigma_1^2}{2\eta_t}.
  \end{gather*}
  Also, by~\eqref{eq:stoch_hess},
  \begin{equation*}
    \mathbb{E}\|{H_t-\nabla^2 f(x_t)} \|
    \leq\left(\mathbb{E}\|{H_t-\nabla^2 f(x_t)} \|^2\right)^{1/2}
    \leq\sigma_2.
  \end{equation*}
  Taking expectations in~\eqref{eq:basic_one_step} and substituting these bounds yields
  \begin{equation}
    \label{eq:basic_expected_step}
    \begin{aligned}
      \mathbb{E}\left[ f(x_{t+1})-f^*\right]
      \leq{}&(1-\alpha_t)\mathbb{E}\left[ f(x_t)-f^*\right] 
      +\frac{\alpha_t\eta_t}{2}\left(
        \mathbb{E}\|{u_t-x^*} \|^2-\mathbb{E}\|{u_{t+1}-x^*} \|^2
      \right)  \\
      &+\frac{\max\left\lbrace M,L_2\right\rbrace}{3}\alpha_t^3D^3
      +\alpha_t^2\sigma_2D^2+\frac{\alpha_t\sigma_1^2}{2\eta_t}.
    \end{aligned}
  \end{equation}
  Multiplying~\eqref{eq:basic_expected_step} by $(t+1)(t+2)(t+3)$ and substituting $\alpha_t=3/(t+3)$ gives
  \begin{gather*}
      (t+1)(t+2)(t+3)\mathbb{E}\left[ f(x_{t+1})-f^*\right]
      -t(t+1)(t+2)\mathbb{E}\left[ f(x_t)-f^*\right] \\
      \leq\frac{3}{2}(t+1)(t+2)\eta_t
        \left(\mathbb{E}\|{u_t-x^*} \|^2-\mathbb{E}\|{u_{t+1}-x^*} \|^2\right) \\
      +9\max\left\lbrace M,L_2\right\rbrace D^3\frac{(t+1)(t+2)}{(t+3)^2}
      +9\sigma_2D^2\frac{(t+1)(t+2)}{t+3}
      +\frac{3\sigma_1^2}{2}\frac{(t+1)(t+2)}{\eta_t}.
  \end{gather*}
  Now sum it over $t=0,\ldots,T-1$. The left-hand side telescopes:
  \begin{gather*}
    \sum_{t=0}^{T-1}\left(
      (t+1)(t+2)(t+3)\mathbb{E}\left[ f(x_{t+1})-f^*\right]
      -t(t+1)(t+2)\mathbb{E}\left[ f(x_t)-f^*\right]
    \right)\\
    =T(T+1)(T+2)\mathbb{E}\left[ f(x_T)-f^*\right],
  \end{gather*}
  where we used $\alpha_0=1$. Next,
  \begin{equation}
    \label{eq:basic_summed}
    \begin{aligned}
      T(T+1)(T+2)\mathbb{E}\left[ f(x_T)-f^*\right]
      \leq{}&\frac{3}{2}\sum_{t=0}^{T-1}(t+1)(t+2)\eta_t
        \left(\mathbb{E}\|{u_t-x^*} \|^2-\mathbb{E}\|{u_{t+1}-x^*} \|^2\right) \\
      &+9\max\left\lbrace M,L_2\right\rbrace D^3\sum_{t=0}^{T-1}\frac{(t+1)(t+2)}{(t+3)^2}\\
      &+9\sigma_2D^2\sum_{t=0}^{T-1}\frac{(t+1)(t+2)}{t+3}
      +\frac{3\sigma_1^2}{2}\sum_{t=0}^{T-1}\frac{(t+1)(t+2)}{\eta_t}.
    \end{aligned}
  \end{equation}
  Let us consider the first sum in inequality above. 
  \begin{gather*}
    \sum_{t=0}^{T-1}(t+1)(t+2)\eta_t
      \left(\mathbb{E}\|{u_t-x^*} \|^2-\mathbb{E}\|{u_{t+1}-x^*} \|^2\right) \\
    =2\eta_0\mathbb{E}\|{u_0-x^*} \|^2
    +\sum_{t=0}^{T-2}\left((t+2)(t+3)\eta_{t+1}-(t+1)(t+2)\eta_t\right)
      \mathbb{E}\|{u_{t+1}-x^*} \|^2\\
    -T(T+1)\eta_{T-1}\mathbb{E}\|{u_T-x^*} \|^2 \\
    \leq D^2\left[2\eta_0+
      \sum_{t=0}^{T-2}\left((t+2)(t+3)\eta_{t+1}-(t+1)(t+2)\eta_t\right)\right] \\
   =T(T+1)\eta_{T-1}D^2 \stackrel{\eqref{eq:basic_reg}}{=}\sigma_1D T(T+1)\sqrt T.
  \end{gather*}
  The remaining sums satisfy
  \begin{gather*}
    \sum_{t=0}^{T-1}\frac{(t+1)(t+2)}{(t+3)^2}\leq T,
    \qquad
    \sum_{t=0}^{T-1}\frac{(t+1)(t+2)}{t+3}\leq\sum_{t=0}^{T-1}(t+1)=\frac{T(T+1)}{2}, \\
    \sum_{t=0}^{T-1}\frac{(t+1)(t+2)}{\eta_t}
    \stackrel{\eqref{eq:basic_reg}}{=}\frac{D}{\sigma_1}\sum_{t=0}^{T-1}(t+2)\sqrt{t+1} 
    \leq\frac{D\sqrt T}{\sigma_1}\sum_{t=0}^{T-1}(t+2)
    =\frac{D\sqrt T T(T+3)}{2\sigma_1}.
  \end{gather*}
  Substituting into~\eqref{eq:basic_summed} and dividing by $T(T+1)(T+2)$, we obtain
  \begin{gather*}
    \mathbb{E}\left[ f(x_T)-f^*\right]
    \leq\frac{9\max\left\lbrace M,L_2\right\rbrace D^3}{(T+1)(T+2)}
    +\frac{9\sigma_2D^2}{2(T+2)}
    +\frac{3\sigma_1D\sqrt T}{2(T+2)}
    +\frac{3\sigma_1D\sqrt T(T+3)}{4(T+1)(T+2)} \\
    =\frac{9\max\left\lbrace M,L_2\right\rbrace D^3}{(T+1)(T+2)}
    +\frac{9\sigma_2D^2}{2(T+2)}
    +\frac{3\sigma_1D\sqrt T(3T+5)}{4(T+1)(T+2)}.
  \end{gather*}
  Since $T(3T+5)\leq3(T+1)(T+2)$, this proves~\eqref{eq:basic_rate}.

\label{lem:optimal_geometry}
  For the iterates of Algorithm~\ref{alg:optimal_method_appendix} and every $t\geq1$,
  \begin{equation}
    \label{eq:optimal_geometry}
    s_t=\alpha_t(u_t-z_{t-1}),

  The weights satisfy
  \begin{equation}
    \label{eq:optimal_cap}

    +\frac{a_t}{\alpha_tA_t}y_t.

  \begin{equation}
    \label{eq:optimal_model_condition}
    \left\langle g_t+H_ts_t+\tau_ts_t,x-u_t\right\rangle\geq0,
    \quad x\in\mathcal{X}.

By~\eqref{eq:optimal_cap}, $0<a_t/(\alpha_tA_t)\leq1$. Thus, by~\eqref{eq:optimal_averaging} and convexity,
\begin{gather*}
  A_tf(w_t)
  =A_tf\left(\left(1-\frac{a_t}{\alpha_tA_t}\right) w_{t-1}
  +\frac{a_t}{\alpha_tA_t}y_t\right)
  \leq\left( A_t-\frac{a_t}{\alpha_t}\right) f(w_{t-1})
  +\frac{a_t}{\alpha_t}f(y_t).
\end{gather*}
Using $A_t-A_{t-1}=a_t$, we obtain
\begin{gather*}
  A_t(f(w_t)-f^*)-A_{t-1}(f(w_{t-1})-f^*)\\
  \leq\left( A_t-\frac{a_t}{\alpha_t}-A_{t-1}\right) f(w_{t-1})
  +\frac{a_t}{\alpha_t}f(y_t)-a_tf^*\\
  =\frac{a_t}{\alpha_t}
  \left( f(y_t)-(1-\alpha_t)f(w_{t-1})-\alpha_tf^*\right)\\
  \leq\frac{a_t}{\alpha_t}
  \left\langle\nabla f(y_t),y_t-(1-\alpha_t)w_{t-1}-\alpha_tx^*\right\rangle
  =a_t\left\langle\nabla f(y_t),u_t-x^*\right\rangle.
\end{gather*}

  Hence,
  \begin{equation}
    \label{eq:optimal_convexity}
    A_t(f(w_t)-f^*)\leq A_{t-1}(f(w_{t-1})-f^*)
    +a_t\left\langle\nabla f(y_t),u_t-x^*\right\rangle.

  The optimality condition for~\eqref{eq:optimal_updates}, multiplied by $a_t>0$, gives
  \begin{equation*}
    \left\langle a_t\widetilde g_t+\lambda_t(z_t-z_{t-1}),x^*-z_t\right\rangle\geq0.
  \end{equation*}
  Since
  \begin{equation*}
    \|{z_{t-1}-x^*} \|^2
    =\|{z_t-x^*} \|^2+\|{z_t-z_{t-1}} \|^2
    -2\left\langle z_t-z_{t-1},z_t-x^*\right\rangle,
  \end{equation*}
  we obtain
  \begin{gather*}
    a_t\left\langle\widetilde g_t,z_t-x^*\right\rangle
    \leq-\lambda_t\left\langle z_t-z_{t-1},z_t-x^*\right\rangle
    =\frac{\lambda_t}{2}
    \left(\|{z_{t-1}-x^*} \|^2-\|{z_t-x^*} \|^2-\|{z_t-z_{t-1}} \|^2\right).
  \end{gather*}
  Rearranging,
  \begin{equation}
    \label{eq:three_dist_bound}
    \frac{\lambda_t}{2}
    \left( \|{z_t-x^*} \|^2 - \|{z_{t-1}-x^*} \|^2 \right) \leq a_t\left\langle\widetilde g_t,x^* - z_t \right\rangle -\frac{\lambda_t}{2}\|{z_t-z_{t-1}} \|^2.
  \end{equation}

  \begin{gather*}
    E_t-E_{t-1}
    \stackrel{\eqref{eq:optimal_energy}}{=}
    A_t(f(w_t)-f^*)-A_{t-1}(f(w_{t-1})-f^*)\\
    +\frac{\lambda_t}{2}\left(\|{z_t-x^*} \|^2-\|{z_{t-1}-x^*} \|^2\right)
    +\frac{\lambda_t-\lambda_{t-1}}{2}\|{z_{t-1}-x^*} \|^2\\
    \stackrel{\eqref{eq:optimal_convexity}}{\leq}a_t\left\langle\nabla f(y_t),u_t-x^*\right\rangle

    -a_t\left\langle\widetilde g_t-\nabla f(y_t),z_t-x^*\right\rangle

    a_t\left\langle\nabla f(y_t)-g_t-H_ts_t-\tau_ts_t,u_t-z_t\right\rangle

      \frac{a_t}{\alpha_t}\left\langle h,s_t\right\rangle
      &+\frac{2a_t}{3\alpha_t\tau_t}\|{h} \|^2
      =\frac{2a_t}{3\alpha_t\tau_t}\|{h+\tau_ts_t} \|^2
      -\frac{a_t}{3\alpha_t}\left\langle h+\tau_ts_t,s_t\right\rangle
      -\frac{a_t\tau_t}{3\alpha_t}\|{s_t} \|^2\\

      \left(\frac{2a_t}{3\alpha_t\tau_t}+\frac{a_t}{3\alpha_t\tau_t}\right)\|{h+\tau_ts_t} \|^2
      +\left(\frac{a_t\tau_t}{12\alpha_t}-\frac{a_t\tau_t}{3\alpha_t}\right)\|{s_t} \|^2
      =\frac{a_t}{\alpha_t\tau_t}\|{h+\tau_ts_t} \|^2
      -\frac{a_t\tau_t}{4\alpha_t}\|{s_t} \|^2.
    \end{aligned}
  \end{equation}

    \label{eq:optimal_square_bound}
    a_t\left\langle h,u_t-z_t\right\rangle-\frac{3\lambda_t}{8}\|{z_t-z_{t-1}} \|^2
    \leq-\frac{a_t\tau_t}{4\alpha_t}\|{s_t} \|^2
    +\frac{a_t}{\alpha_t\tau_t}\|{h+\tau_ts_t} \|^2.
  \end{equation}

      -\frac{3\lambda_t}{8}\|{z_t-z_{t-1}} \|^2\\
      \leq-\frac{a_t\tau_t}{4\alpha_t}\|{s_t} \|^2
      +\frac{a_t}{\alpha_t\tau_t}\|{\nabla f(y_t)-g_t-H_ts_t} \|^2.

    \begin{gather*}
      \|{\nabla f(y_t)-g_t-H_ts_t} \|^2\\

      +4\|{g_t-\nabla f(v_t)} \|^2\\
      \stackrel{\eqref{eq:optimal_taylor}}{\leq}
      \frac{\tau_t^2}{8}\|{s_t} \|^2
      +4\|{H_t-\nabla^2 f(v_t)} \|^2\|{s_t} \|^2
      +4\|{g_t-\nabla f(v_t)} \|^2.

      \end{aligned}
    \end{equation}

    \end{aligned}

    \stackrel{\eqref{eq:optimal_cap}}{\leq}
    \frac{\lambda_t}{\alpha_t^2\tau_t^2}
    \leq\frac{\lambda_t}{\alpha_t^2\beta_t^2}
    \stackrel{\eqref{eq:optimal_reg}}{=}
    \frac{\lambda_t}{\alpha_t^2(\lambda_tD/t^{3/2})^2}
    =\frac{t^3}{\alpha_t^2\lambda_tD^2},\\

    \leq\frac{D}{\alpha_t},\qquad

    \frac{t^{3/2}}{\alpha_tD}D
    =\frac{t^{3/2}}{\alpha_t},\\
    \frac{2a_t^2}{\lambda_t}

    =\frac{2t^3}{\alpha_t^2\lambda_tD^2}.
  \end{gather*}
  Substituting these coefficient bounds into~\eqref{eq:optimal_energy_combined} proves~\eqref{eq:optimal_one_step}.

\label{lem:optimal_error_moments}

  \begin{equation}

    \mathbb{E}\left[\delta_t^2\mid\mathcal F_{t-1}\right]
    \leq\frac{3D^2\sigma_2^2}{\alpha_t^2}
    +\frac{6\sigma_1^2t^3}{\alpha_t^2}.

    =\frac{\lambda_TD^2}{2}+\sum_{t=1}^T\mathbb{E}\delta_t

\end{equation}
Dividing by $T(T+1)(T+2)(T+3)\geq T^4$ proves~\eqref{eq:optimal_inverse_step_sum}.

<!-- proof-10: body_md -->
Dividing~\eqref{eq:optimal_one_step} by $\lambda_t$, taking expectations, and using $\mathbb{E}\ell_t=0$, we get
  \begin{equation}
    \label{eq:Vt_bound}
    \frac18\mathbb{E} V_t
    \leq\frac{\mathbb{E} E_{t-1}}{\lambda_t}-\frac{\mathbb{E} E_t}{\lambda_t}
    +\frac{D^2}{2}\left(1-\frac{\lambda_{t-1}}{\lambda_t}\right)
    +\frac{\mathbb{E}\delta_t}{\lambda_t}.
  \end{equation}
  We bound the third term first. By the definition of $\lambda_t$ in~\eqref{eq:optimal_reg}, for $t\geq2$,
  \begin{gather*}
    \lambda_{t-1}
    =\frac{\sigma_1t^3}{D^2}\left(1-\frac1t\right)^3
    +\frac{\sigma_2t^{3/2}}{D}\left(1-\frac1t\right)^{3/2}+2M\\
    \geq\left(1-\frac1t\right)^3
    \left(\frac{\sigma_1t^3}{D^2}+\frac{\sigma_2t^{3/2}}{D}+2M\right)
    =\left(1-\frac1t\right)^3\lambda_t.
  \end{gather*}
  Since $\lambda_t\geq\lambda_{t-1}$,
  \begin{equation*}
    0\leq1-\frac{\lambda_{t-1}}{\lambda_t}
    \leq\frac3t-\frac3{t^2}+\frac1{t^3}\leq\frac3t.
  \end{equation*}
  The last bound also holds at $t=1$ because $\lambda_0=\lambda_1$. 
  
  Next, we bound the last term in~\eqref{eq:Vt_bound}
  \begin{equation*}
    \frac{\mathbb{E}\delta_t}{\lambda_t}
    \stackrel{\eqref{eq:optimal_error_budget} }{\leq}\frac{\sigma_2D\sqrt t+2\sigma_1t^2}{\lambda_t}
    \leq\frac{2D^2}{t}
    \frac{\sigma_2t^{3/2}/D+\sigma_1t^3/D^2}{\lambda_t}
    \stackrel{\eqref{eq:optimal_reg}}{\leq}\frac{2D^2}{t}.
  \end{equation*}
  Substituting these bounds into~\eqref{eq:Vt_bound} and using $\lambda_t\geq\lambda_{t-1}$ gives
  \begin{equation}
    \label{eq:optimal_normalized_step}
    \frac18\mathbb{E} V_t
    \leq\frac{\mathbb{E} E_{t-1}}{\lambda_{t-1}}-\frac{\mathbb{E} E_t}{\lambda_t}
    +\frac{7D^2}{2t}.
  \end{equation}
  Multiplying~\eqref{eq:optimal_normalized_step} by $\sqrt t$ and summing gives
  \begin{gather*}
    \frac18\mathbb{E}\sum_{t=1}^T\sqrt t V_t
    \leq\sum_{t=1}^T\sqrt t\left(
      \frac{\mathbb{E} E_{t-1}}{\lambda_{t-1}}-\frac{\mathbb{E} E_t}{\lambda_t}\right)
    +\frac{7D^2}{2}\sum_{t=1}^T\frac1{\sqrt t}\\
    =\frac{E_0}{\lambda_0}
    +\sum_{t=1}^{T-1}(\sqrt{t+1}-\sqrt t)\frac{\mathbb{E} E_t}{\lambda_t}
    -\sqrt T\frac{\mathbb{E} E_T}{\lambda_T}
    +\frac{7D^2}{2}\sum_{t=1}^T\frac1{\sqrt t}\\
    \stackrel{\eqref{eq:optimal_energy_moments},\eqref{eq:optimal_inverse_sqrt_sum}}{\leq}
    \frac{D^2}{2}+\frac{5D^2}{2}(\sqrt T-1)
    +\frac{7D^2}{2}(2\sqrt T-1)
    =\frac{D^2}{2}(19\sqrt T-11)
    \leq\frac{19D^2}{2}\sqrt T.
  \end{gather*}
  This proves the first bound in~\eqref{eq:optimal_movement_bounds}. By Cauchy--Schwarz,
  \begin{gather*}
    \left(\sum_{t=1}^T\sqrt{V_t}\right)^2
    =\left(\sum_{t=1}^Tt^{-1/4}\left( t^{1/4}\sqrt{V_t}\right)\right)^2
    \leq\left(\sum_{t=1}^T\frac1{\sqrt t}\right)
          \left(\sum_{t=1}^T\sqrt t V_t\right),\\
    \mathbb{E}\left(\sum_{t=1}^T\sqrt{V_t}\right)^2
    \stackrel{\eqref{eq:optimal_inverse_sqrt_sum}}{\leq}
    (2\sqrt T-1)\mathbb{E}\sum_{t=1}^T\sqrt t V_t
    \leq(2\sqrt T-1)76D^2\sqrt T
    \leq152D^2T.
  \end{gather*}
