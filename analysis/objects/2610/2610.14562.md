<!-- remark-2: body_md -->
\label{remark:Cgamma-verification}
We provide sufficient conditions for \Cref{assumption:Cgamma} under entropy regularization and Tsallis regularization.
Let $r,s\in[1,\infty)$, and let
$\pi^*$ be a finite minimizer of $\mathcal{F}_\varepsilon^{\mu,\nu}$.
Fix $(x_0,y_0)\in\mathcal{X}\times\mathcal{Y}$.
Assume
$\mu\in\mathcal{P}_{2rs}(\mathcal{X})$ and $\nu\in\mathcal{P}_{2rs}(\mathcal{Y})$, and set
\[
w(x,y):=
1+d_{\mathcal{X}}(x,x_0)^{rs}+d_{\mathcal{Y}}(y,y_0)^{rs},
\qquad
S:=\sup_{\eta\in\Pi(\mu,\nu)}\int w^2\,d\eta.
\]
Then
\begin{align}\label{inequality_gamma_bound_assumption_verificaion}
                              
S\le
3\left(
1+\int d_{\mathcal{X}}(x,x_0)^{2rs}\,d\mu(x)
+\int d_{\mathcal{Y}}(y,y_0)^{2rs}\,d\nu(y)
\right)<\infty.
\end{align}
(i) Entropy regularization. It suffices to require
$
\varepsilon
>\kappa
=2^{rs-1}S.
$

\noindent
(ii) Tsallis regularization with $1<\alpha\le2$.
$
\varepsilon
>\kappa
=
\frac{2^{rs-1}}{\alpha-1}
\left(\int w^2\,d(\mu\otimes\nu)\right)^{\alpha-1}S^{2-\alpha}
$
would be sufficient.

\noindent
(iii) Tsallis regularization with $\alpha>2$ on compact supports. 
It suffices to require $\varepsilon \geq 5M$
with $M:=
\max\left\{
\operatorname{diam}(\operatorname{supp}\mu)^{rs},
\operatorname{diam}(\operatorname{supp}\nu)^{rs}
\right\}$. Under this setting,
$
\varepsilon>\kappa
:=
\frac{\alpha M}{2(\alpha-1)}
\left(1-\frac{4M}{\varepsilon}\right)^{-(\alpha-2)/(\alpha-1)}.
$

<!-- lemma-3: body_md -->
\label{lemma:strong-convexity-tv}
Suppose \Cref{assumption:regularizaion} and  \Cref{assumption:Cgamma} hold. If
$\pi\in\Pi(\mu,\nu)$ and $\pi^*$ is the optimizer of $S_{r,s,\varepsilon}$
and $h:\mathcal{X}\times\mathcal{Y}\to[0,\infty)$ is measurable with
$\int_{\mathcal{X}\times\mathcal{Y}}h^2\,d(\mu\otimes\nu+\pi^*+\pi)<\infty$,
then
\begin{align}\label{eq:objective-tv}
\|h(\pi&-\pi^*)\|_{\mathrm{TV}}^{\beta+2}\nonumber\\
&\le
\frac{ (\beta+1)(\beta+2)3^{\beta/2}((\lambda_0+\lambda_1)\vee\lambda_2)}{\varepsilon-\kappa}
\|h\|_{L^2(\mu\otimes\nu+\pi^*+\pi)}^{\beta+2}
\bigl(\mathcal{F}_\varepsilon^{\mu,\nu}(\pi)
-
\mathcal{F}_\varepsilon^{\mu,\nu}(\pi^*)\bigr).
\end{align}
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
Without loss of generality, we assume $\mathcal{F}_\varepsilon^{\mu,\nu}(\pi) < \infty$. Otherwise  the inequality to show is trivial.
We first show that
\begin{equation}\label{eq:coercivity}
\mathcal{F}_\varepsilon^{\mu,\nu}(\pi)-\mathcal{F}_\varepsilon^{\mu,\nu}(\pi^*)
\ge
(\varepsilon-\kappa)B^{\mu,\nu}_\varphi(\pi\mid\pi^*).
\end{equation}
For $t\in[0,1]$, let $\pi_t:=(1-t)\pi^*+t\pi$.
Then
\begin{equation}\label{eq:energy-expansion}
\mathcal{E}_\Gamma(\pi_t)
=(1-t)\mathcal{E}_\Gamma(\pi^*)+t\mathcal{E}_\Gamma(\pi)
-t(1-t)\mathcal{E}_\Gamma(\pi-\pi^*).
\end{equation}
Write $f_*:=d\pi^*/d(\mu\otimes\nu)$, $f:=d\pi/d(\mu\otimes\nu)$, and
$f_t:=(1-t)f_*+tf$. The convexity gap
\[
J_t
:=(1-t)\,\mathsf{D}_\varphi(\pi^*,\mu\otimes\nu)+t\,\mathsf{D}_\varphi(\pi,\mu\otimes\nu)
-\mathsf{D}_\varphi(\pi_t,\mu\otimes\nu)
\]
is finite and nonnegative. Combining its defining identity with
\eqref{eq:energy-expansion} and using optimality gives
\begin{equation}\label{eq:near-optimal}
\mathcal{F}_\varepsilon^{\mu,\nu}(\pi)-\mathcal{F}_\varepsilon^{\mu,\nu}(\pi^*)
\ge
(1-t)\mathcal{E}_\Gamma(\pi-\pi^*)+\varepsilon\frac{J_t}{t},
\qquad 0<t<1.
\end{equation}
The integrand defining $J_t$ is nonnegative. Hence Fatou's lemma,
the definition of $B^{\mu,\nu}_\varphi$, and \eqref{eq:near-optimal} imply
\begin{align*}
B^{\mu,\nu}_\varphi(\pi\mid\pi^*)
&=
\int_{\mathcal{X}\times\mathcal{Y}}\lim_{t\downarrow0}
\frac{(1-t)\varphi(f_*)+t\varphi(f)-\varphi(f_t)}{t}\,d\mu\otimes\nu\\
&\le\liminf_{t\downarrow0}\frac{J_t}{t}
\le
\frac{
\mathcal{F}_\varepsilon^{\mu,\nu}(\pi)-\mathcal{F}_\varepsilon^{\mu,\nu}(\pi^*)-\mathcal{E}_\Gamma(\pi-\pi^*)
}{\varepsilon}
<\infty.
\end{align*}
Thus the Bregman divergence is finite, and rearranging gives
\[
\mathcal{F}_\varepsilon^{\mu,\nu}(\pi)-\mathcal{F}_\varepsilon^{\mu,\nu}(\pi^*)
\ge
\mathcal{E}_\Gamma(\pi-\pi^*)+\varepsilon B^{\mu,\nu}_\varphi(\pi\mid\pi^*)
\ge
(\varepsilon-\kappa)B^{\mu,\nu}_\varphi(\pi\mid\pi^*),
\]
where the last step uses \eqref{eq:Cgamma}.
This proves \eqref{eq:coercivity}.
Next, \Cref{assumption:regularizaion} gives, for $u>0$,
\[
\varphi''(u)
\ge
\frac{u^{\beta}}
{\lambda_0+\lambda_1u^{\beta}
+\lambda_2u^{\beta+1}}
\ge
\frac{u^{\beta}}
{((\lambda_0+\lambda_1)\vee\lambda_2)(1+u)^{\beta+1}}.
\]
Take $a,b>0$, it holds that
\[
\int_0^1(1-s)((1-s)b+sa)^\beta\,ds
\ge
\frac{(a\vee b)^\beta}{(\beta+1)(\beta+2)}.
\]
Indeed, if \(a\ge b\), then $(1-s)b+sa\ge sa$,
and hence
\[
\int_0^1(1-s)((1-s)b+sa)^\beta\,ds
\ge
a^\beta\int_0^1(1-s)s^\beta\,ds
=
\frac{a^\beta}{(\beta+1)(\beta+2)}.
\]
If \(b\ge a\), then $(1-s)b+sa\ge (1-s)b$,
so
\[
\int_0^1(1-s)((1-s)b+sa)^\beta\,ds
\ge
b^\beta\int_0^1(1-s)^{\beta+1}\,ds
=
\frac{b^\beta}{\beta+2}
\ge
\frac{b^\beta}{(\beta+1)(\beta+2)}.
\]
Taylor's formula therefore yields
\begin{align}\label{eq:pointwise-bregman-lower}
b_\varphi(a\mid b)
&=(a-b)^2\int_0^1(1-s)
\varphi''((1-s)b+sa)\,ds
\\
&\ge
\frac{(a-b)^2(a \vee b)^{\beta}}
{(\beta+1)(\beta+2)((\lambda_0+\lambda_1)\vee\lambda_2)
(1+a \vee b)^{\beta+1}}\nonumber\\
&\ge
\frac{|a-b|^{\beta+2}}
{(\beta+1)(\beta+2)((\lambda_0+\lambda_1)\vee\lambda_2)
(1+a+b)^{\beta+1}}\nonumber.
\end{align}
The same inequality holds for \(a,b\ge0\): apply
\eqref{eq:pointwise-bregman-lower} to \(a+\delta,b+\delta\) and let
\(\delta\downarrow0\). By the boundary definition of \(b_\varphi\), $b_\varphi(a+\delta\mid b+\delta)\to b_\varphi(a\mid b)$,
with convergence to \(+\infty\) allowed when
\(\varphi'_+(0)=-\infty\).
Applying \eqref{eq:pointwise-bregman-lower} to $f,f_*$ and using
H\"older's inequality with
$k:=\beta+2$ and $k':=k/(k-1)\le2$ yields
\begin{align}
&\|h(\pi-\pi^*)\|_{\mathrm{TV}}^k
=\left(\int_{\mathcal{X}\times\mathcal{Y}}h|f-f_*|\,d\mu\otimes\nu\right)^k
\notag\\
&\le
\left(\int_{\mathcal{X}\times\mathcal{Y}}
\frac{|f-f_*|^k}{(1+f+f_*)^{k-1}}\,d\mu\otimes\nu\right)
\left(\int_{\mathcal{X}\times\mathcal{Y}}h^{k'}(1+f+f_*)\,d\mu\otimes\nu\right)^{k-1}
\notag\\
&\le
(\beta+1)(\beta+2)((\lambda_0+\lambda_1)\vee\lambda_2)
B^{\mu,\nu}_\varphi(\pi\mid\pi^*)
\left(\int_{\mathcal{X}\times\mathcal{Y}}h^{k'}\,d(\mu\otimes\nu+\pi+\pi^*)\right)^{k-1}
\notag\\
&\le
(\beta+1)(\beta+2)3^{\beta/2}((\lambda_0+\lambda_1)\vee\lambda_2)
\left(\int_{\mathcal{X}\times\mathcal{Y}}h^2\,d(\mu\otimes\nu+\pi+\pi^*)\right)^{k/2}
B^{\mu,\nu}_\varphi(\pi\mid\pi^*).
\label{eq:bregman-tv}
\end{align}
The last inequality uses that $\mu\otimes\nu+\pi+\pi^*$ has total mass $3$:
\[
\left(\int_{\mathcal{X}\times\mathcal{Y}}h^{k'}\,d(\mu\otimes\nu+\pi+\pi^*)\right)^{k-1}
\le
3^{\beta/2}
\left(\int_{\mathcal{X}\times\mathcal{Y}}h^2\,d(\mu\otimes\nu+\pi+\pi^*)\right)^{k/2}.
\]
Combining \eqref{eq:bregman-tv} with
\eqref{eq:coercivity} proves \eqref{eq:objective-tv}.

\mathcal{F}_\varepsilon^{\mu,\nu} (\pi) -  \mathcal{F}_\varepsilon^{{\tilde\mu,\tilde\nu}} (\widetilde\pi^*)

\end{align}

     \leq

     \Lambda \Delta.

W_q(\pi^*,\pi)

\le

=2^{1/q-1/p}\Delta.
\end{equation}

\eqref{eq:Wq-second}, proves \eqref{eq:main-bound}.

\label{thm:gw-duality}
Let $\mu\in\mathcal{P}(\mathbb{R}^{d_x})$ and $\nu\in \mathcal{P}(\mathbb{R}^{d_y})$ be compactly supported. For every $M\ge M_{\mu,\nu}$,
\begin{align}\label{eq:gw-variational-duality}
&\quad S_{2,2,\varepsilon}(\mu,\nu) -\mathsf{GW}^1(\mu,\nu)\\
=&
\min_{A\in\mathbb{R}^{d_x\times d_y}}
\left\{32\|A\|_F^2+\mathsf {OT}_{\varepsilon}^{c_A}(\mu,\nu)\right\}
=
\min_{A\in\mathcal D_M}
\left\{32\|A\|_F^2+\mathsf {OT}_{\varepsilon}^{c_A}(\mu,\nu)\right\}.\nonumber
\end{align} Equivalently,
\begin{equation}
\begin{split}
S_{2,2,\varepsilon}(\mu,\nu)-\mathsf{GW}^1(\mu,\nu)
=
&\min_{A\in\mathcal D_M}
\max_{\substack{f\in \mathcal{C}(\mathcal{X})\\g\in \mathcal{C}(\mathcal{Y})}}
\biggl\{
32\|A\|_F^2+\int f\, d\mu+\int g\,d\nu \\
&-\varepsilon\int
\psi\!\left(
\frac{f(x)+g(y)-c_A(x,y)}{\varepsilon}
\right)\,d(\mu\otimes\nu)(x,y)
\biggr\}.
\end{split}
\label{eq:gw-potential-duality}
\end{equation}
For each \(A\in\mathbb{R}^{d_x\times d_y}\), the inner ROT problem admits a unique optimizer \(\pi_A\).
The GW minimizers and the outer minimizers in either matrix domain,
correspond bijectively through
\begin{equation}
\pi^*=\pi_{A^*},
\qquad
A^*=\frac12\int xy^\top\,d\pi^*(x,y).
\label{eq:optimal-matrix-moment}
\end{equation}

The optimums related to regularized GW are attained because of the lower semicontinuity of $\pi$ and $A$.
For each fixed \(A\), the ROT problem has a unique primal optimizer \(\pi_A\), and its dual maximum is attained \cite[Proposition~2.3]{alberto.gonzalez-marcel.nutz_austin.stromme2026}.
The minimum in \eqref{eq:gw-variational-duality} is attained due to continuity of the objective with respect to $A$ on the compact set $\mathcal{D}$.
For $\pi\in\Pi(\mu,\nu)$, write $a_\mu:=\int x\,d\mu(x), b_\nu:=\int y\,d\nu(y), B_\pi:=\int xy^\top\,d\pi(x,y)$.
Then $\langle A,B_\pi\rangle_F
 =\int\langle A,xy^\top\rangle_F\,d\pi
 =\int x^\top Ay\,d\pi$.
Expanding squared distances, noting that $\iint\langle x,x'\rangle\langle y,y'\rangle\,d\pi\,d\pi = \|B_\pi\|^2_F$ and then integrating gives
\begin{equation}
\mathcal{E}_{\Gamma}(\pi)
=\mathsf{GW}^{1}(\mu,\nu)
-4\int\|x\|^2\|y\|^2\,d\pi
-8\|B_\pi\|_F^2
+8\int\bigl(
a_\mu^\top x\|y\|^2
+b_\nu^\top y\|x\|^2
\bigr)\,d\pi.
\label{eq:distortionexpansion}
\end{equation}

and then substituting this attained fixed-$A$ dual representation into \eqref{eq:gw-variational-duality} proves \eqref{eq:gw-potential-duality}.

<!-- proof-6: body_md -->
We divide the proof into three steps.

\noindent
\textit{Step 1: mean absolute error bound.}
We first prove
\begin{align}\label{eq:main-l1}
\mathbb{E}\bigl|S_{2,2,\varepsilon}(\mu_n,\nu_n)-S_{2,2,\varepsilon}(\mu,\nu)\bigr|\,\,\,
\le C n^{-1/2},
\end{align}
for some constant $C\geq0$ depending on  $\varphi,R_x,d_x,R_y,d_y,\varepsilon$. 
By \Cref{lem:empirical-reduction}, it suffices to bound $|\mathsf{GW}^1(\mu_n,\nu_n)
-
\mathsf{GW}^1(\mu,\nu)|, |
V_\varepsilon(\mu_n,\nu_n)
-
V_\varepsilon(\mu,\nu)|, \|\mu_nx\|$ and $
\|\nu_ny\|$.
Define $ S_n:=\sup_{h\in\mathcal{H}}|((\mu_n\otimes\nu_n) - (\mu\otimes\nu) )h|$. 
Hoeffding's inequality for two-sample U-statistics \cite{hoeffding1963probability} implies that $Z_h:=((\mu_n\otimes\nu_n) - (\mu\otimes\nu) )h$ has sub-Gaussian increments with respect to \(n^{-1/2}\|\cdot\|_\infty\).
Then \Cref{lem:covering} combined with chaining \cite[Theorem 5.22]{Wainwright_2019} gives
\begin{equation}
 \mathbb{E} S_n\le C n^{-1/2}.
 \label{eq:Sn-bound}
\end{equation}
Abbreviate
\[
 \mathrm{OT}_A:=\mathsf {OT}_{\varepsilon}^{c_A}(\mu,\nu),\quad
 \widehat{\mathrm{OT}}_A:=\mathsf {OT}_{\varepsilon}^{c_A}(\mu_n,\nu_n),\quad
 V:=V_\varepsilon (\mu,\nu),\quad V_n:=V_\varepsilon(\mu_n,\nu_n).
\]
Theorem~\ref{thm:gw-duality} guarantees a population minimizer $A^*\in \mathcal D$. Strong duality for ROT gives $\mathrm{OT}_{A^*}=(\mu\otimes\nu) H_{A^*}$.
Define 
\begin{equation*}
 Z_{{A^*},n}:=((\mu_n\otimes\nu_n) -(\mu\otimes\nu) )H_{A^*},\qquad
 R_{{A^*},n}:=\widehat{\mathrm{OT}}_{A^*}-(\mu_n\otimes\nu_n)  H_{A^*}\ge0.
\end{equation*}
The inequality is empirical duality.
                   
Thus \Cref{proposition:ROT_bias_parametric_bound} gives
\begin{equation}
 \mathbb{E} R_{{A^*},n}
 =\mathbb{E}\widehat{\mathrm{OT}}_{A^*}-\mathrm{OT}_{A^*}
 \le 
 C n^{-1/2}.
 \label{eq:remainder-mean}
\end{equation}
Evaluation at $A^*$ gives
\[
 V_n-V\le {\widehat{\mathsf {OT}}}_{A^*}
 -
 \mathsf {OT}_{A^*}
 =Z_{A^*,n}+R_{A^*,n}\le S_n+R_{A^*,n}.
\]
For the opposite direction, take any $A\in\mathcal D$ with associated integrand $H_A\in\mathcal H$.  Empirical duality and the definition of $S_n$ imply
\[
 \widehat{\mathrm{OT}}_A
 \ge (\mu_n\otimes\nu_n)H_A
 \ge (\mu\otimes\nu)H_A-S_n
 = \mathrm{OT}_A-S_n.
\]
Taking infimum over $A\in\mathcal D$ yields
\[
 V_n
 =\inf_{A\in\mathcal D}
   \bigl\{32\|A\|_F^2+\widehat{\mathrm{OT}}_A\bigr\}
 \ge
 \inf_{A\in\mathcal D}
   \bigl\{32\|A\|_F^2+\mathrm{OT}_A\bigr\}-S_n
 =V-S_n.
\]
As a consequence, we know that
\begin{equation}
 |V_n-V|\le R_{A^*,n}+S_n.
 \label{eq:one-sided-transfer}
\end{equation}
By 
\Cref{eq:Sn-bound,eq:remainder-mean,eq:one-sided-transfer},
\begin{equation}
 \mathbb{E}|V_n-V|
 \le
 C
 n^{-1/2}.
 \label{eq:outer-rate}
\end{equation}
Cauchy-Schwarz inequality gives that
\begin{align}\label{eqn:moment_bound}
\mathbb E|m_2(\mu_n)-m_2(\mu)|
\le \sqrt{\frac{\operatorname{Var}(\|X_1\|^2)}{n}}
\le \frac{R_x^2}{\sqrt n},
\qquad
\mathbb E|m_2(\nu_n)-m_2(\nu)|
\le \frac{R_y^2}{\sqrt n}.
\end{align}
For $h(x,x'):=\|x-x'\|^4 \leq 16R_x^4$, set $Q:=\iint h\,d(\mu\otimes\mu)$, $Q_n:=\frac1{n^2}\sum_{1\leq i,j\leq n} h(X_i,X_j)$.
Since $h(x,x)=0$, we have $\mathbb E Q_n=(1-1/n)Q$.
Independence gives
\[
\begin{aligned}
\operatorname{Var}(Q_n)
&=
\frac1{n^4}\sum_{1\leq i,j,k,l\leq n}
\operatorname{Cov}\bigl(h(X_i,X_j),h(X_k,X_l)\bigr)\\
&\le
\frac{(16R_x^4)^2}{n^4}\sum_{1\leq i,j,k,l\leq n}
\left(
\mathbf 1_{\{i=k\}}+\mathbf 1_{\{i=l\}}
+\mathbf 1_{\{j=k\}}+\mathbf 1_{\{j=l\}}
\right)
=
\frac{4(16R_x^4)^2}{n}.\\
\mathbb E|Q_n-Q|
&\le \mathbb E|Q_n- \mathbb{E} Q_n| + |\mathbb{E} Q_n - Q|
\le\sqrt{\operatorname{Var}(Q_n)}+\frac Qn
\le\frac{48R_x^4}{\sqrt n}.
\end{aligned}
\]
The same bounds hold for $\nu_n$, with $R_x$ replaced by $R_y$.
At the same time, note that
\begin{align*}
|m_2(\mu_n)m_2(\nu_n)-m_2(\mu)m_2(\nu)|
\le 
R_y^2|m_2(\mu_n)-m_2(\mu)|
+
R_x^2|m_2(\nu_n)-m_2(\nu)|
\end{align*}
and it follows that
\begin{align}\label{eqn:GW1_bound}
 \mathbb{E}|\mathsf{GW}^1(\mu_n,\nu_n)-\mathsf{GW}^1(\mu,\nu)|\le 
 C n^{-1/2}.
\end{align}
At the same time, Jensen's inequality implies that
\begin{align}\label{eqn:empirical_moment_bound}
\mathbb E\|\mu_nx\|
\le
\biggl(
\mathbb E\Bigl\|\frac1n\sum_{i=1}^nX_i\Bigr\|^2
\biggr)^{1/2}
=
\sqrt{\frac{m_2(\mu)}n}
\le\frac{R_x}{\sqrt n},
\qquad
\mathbb E\|\nu_ny\|

\begin{align}\label{eqn:verification_bound_distortion}
0\le\Gamma_{r,s}((x,y),(x',y'))
&\le
d_{\mathcal{X}}(x,x')^{rs}+d_{\mathcal{Y}}(y,y')^{rs}\\
&\le
2^{rs-1}\bigl(w(x,y)+w(x',y')-2\bigr)
\le
2^{rs-1}w(x,y)w(x',y').\nonumber
\end{align}
Under the standing moment assumptions,
$\int w\,d\eta<\infty$ for every $\eta\in\Pi(\mu,\nu)$,

\begin{equation}
\label{eq:Cgamma-weighted-energy}
\mathcal{E}_\Gamma(\theta-\pi^*)
\ge
-2^{rs-2}\|w(\theta-\pi^*)\|_{\mathrm{TV}}^2.
\end{equation} 
Actually, for $\sigma:=\theta-\pi^*$, the inequality

=
\mathcal{E}_\Gamma(\sigma^+)+\mathcal{E}_\Gamma(\sigma^-)
-2\iint\Gamma(z,z')\,d\sigma^+(z)\,d\sigma^-(z')
\notag\\
&\ge
-2^{rs}
\left(\int w\,d\sigma^+\right)
\left(\int w\,d\sigma^-\right)
\ge
-2^{rs-2}\left(\int w\,d|\sigma|\right)^2
=
-2^{rs-2}\|w(\theta-\pi^*)\|_{\mathrm{TV}}^2,
\end{align*}
where $\sigma=\sigma^+-\sigma^-$ is the Jordan decomposition and the last inequality uses $4ab\le(a+b)^2$.

\Cref{lemma:GW-linearization} to regard $\pi^*$ as an optimizer

\[
c_*(x,y):=2\int\Gamma((x,y),(x',y'))\,d\pi^*(x',y'),
\]

\[
\pi^*\in
\operatorname*{argmin}_{\theta\in\Pi(\mu,\nu)}
\left\{
\int c_*\,d\theta
+\varepsilon \,\mathsf D_{\varphi}(\theta,\mu\otimes\nu)
\right\}.
\]
Bounded-cost convex-regularized transport duality
\cite{alberto.gonzalez-marcel.nutz_austin.stromme2026}
provides bounded potentials $u,v$ such that
$$f_*(x,y)=
\left(
\frac{u(x)+v(y)-c_*(x,y)}{\varepsilon}
\right)_+^{1/(\alpha-1)}$$
with 
$\operatorname{ess\,osc}_{\nu}v\le2M$
and then
$
\operatorname{ess\,osc}_{y\sim\nu}
\bigl(u(x)+v(y)-c_*(x,y)\bigr)\le4M.
$
Equation $\int f_*(x,y) d\nu(y)=1$ for
$f_*$ implies
$
\operatorname*{ess\,sup}_{y\sim\nu}
\bigl(u(x)+v(y)-c_*(x,y)\bigr)\ge\varepsilon.
$

\begin{equation}
\label{eq:Cgamma-Tsallis-density}
f_*\ge
\Bigl(1-\frac{4M}{\varepsilon}\Bigr)^{\frac{1}{\alpha-1}}
\end{equation}
Note that $f_t \geq (1-t)f_*$,  using \eqref{eq:Cgamma-Tsallis-density}, the integral Taylor formula \eqref{eq:Cgamma-bregman-integral} and Fubini's theorem yield
\begin{align*}
B^{\mu,\nu}_{\varphi}&(\theta\mid\pi^*)
\geq
(\alpha-1)
\Bigl(1-\frac{4M}{\varepsilon}\Bigr)^{\frac{\alpha-2}{\alpha-1}}
\int_0^1
(1-t)^{\alpha-1} \int  (f-f_*)^{2}
\,d(\mu\otimes\nu)\,dt\\
&=
\frac{\alpha-1}{\alpha}
\Bigl(1-\frac{4M}{\varepsilon}\Bigr)^{\frac{\alpha-2}{\alpha-1}}
\int(f-f_*)^2\,d(\mu\otimes\nu)
\ge
\frac{\alpha-1}{\alpha}
\Bigl(1-\frac{4M}{\varepsilon}\Bigr)^{\frac{\alpha-2}{\alpha-1}}
\|\theta-\pi^*\|_{\mathrm{TV}}^2.
\end{align*}
We apply Cauchy--Schwarz with $\mu\otimes\nu$ for the last inequality.
Since $0\le\Gamma\le M$, the Jordan decomposition as above gives
\[
\mathcal{E}_\Gamma(\theta-\pi^*)
\ge-\frac M2\|\theta-\pi^*\|_{\mathrm{TV}}^2
\ge-\kappa B^{\mu,\nu}_{\varphi}(\theta\mid\pi^*).
\]
Finally, $\varepsilon\ge5M$ implies
\[
\frac{\kappa}{\varepsilon}
\le
\frac{\alpha}{2(\alpha-1)}\,5^{-1/(\alpha-1)}
<1,
\]
which completes the verification.

Throughout the proof, $C$ denotes a finite positive constant depending on  $R_x,d_x,R_y,d_y,\varphi,\varepsilon$.
Its value may change from one display to the next.

We first establish the following claim: there exist a finite
integer $J\ge1$ satisfying \eqref{eq:covering-J-bound} and fixed
continuous linear functionals
$\tau_j:\mathcal C(\mathcal X)\to\mathbb{R}$, $j=1,\ldots,J$, satisfying
\[
 |\tau_j(h)|\le\|h\|_\infty
 \qquad\text{for every }h\in\mathcal C(\mathcal X),
\]
such that for every $(A,f,g),(A',f',g')\in\mathcal{S}$,
\begin{equation}
 \|H_{A,f,g}-H_{A',f',g'}\|_\infty
 \le C\bigl(
 \|A-A'\|_F^{\frac{1}{2(1+\beta)}}
 +\|\tau(f)-\tau(f')\|_{\ell^\infty_J}^{\frac{1}{1+\beta}}
 \bigr),
\label{eq:integrand-stability}
\end{equation}
where $\tau(f)=(\tau_1(f),\ldots,\tau_J(f))$, $\|a\|_{\ell^\infty_J}=\max_{1\le j\le J}|a_j|, a\in\mathbb{R}^J$.
Assuming this claim, define
\[
 \Theta
 :=
 \{(A,\tau(f)):(A,f,g)\in\mathcal{S}\}
 \subset\mathbb{R}^{d_xd_y+J}.
\]
This set is bounded because $\mathcal D$ is bounded and $|\tau_j(f)|\le B$ under the normalization. 
Here, we identify \(\mathbb R^{d_xd_y+J}\) with \(\mathbb R^{d_x\times d_y}\times\mathbb R^J\) and equip it with the Euclidean norm
$
|(A,a)|
:=
\left(\|A\|_F^2+\sum_{j=1}^J |a_j|^2\right)^{1/2}.
$
By \eqref{eq:integrand-stability}, $z\mapsto H_z$ is a well-defined map from $\Theta$ onto $\mathcal{H}$.







Choose $R_\Theta\ge1$ such that
$\Theta\subset B(0,R_\Theta)$. On this bounded set,
\eqref{eq:integrand-stability} implies
\[
 \|H_z-H_{z'}\|_\infty
 \le L_\Theta|z-z'|^{\frac{1}{2(1+\beta)}}
\]
for some $L_\Theta\ge1$. 
A parameter net with centers in $\Theta$ therefore gives
\begin{align}
 N(\eta,\mathcal{H},\|\cdot\|_\infty)
 \le
 N\bigl((\eta/L_\Theta)^{2(1+\beta)},
        \Theta,|\cdot|\bigr)
 \le
 \left[
  1+2R_\Theta(L_\Theta/\eta)^{2(1+\beta)}
 \right]^{d_xd_y+J}.
 \label{eq:volumetric-cover}
\end{align}
The final inequality follows from a standard volumetric packing argument applied to a maximal separated subset of $\Theta$; see \cite[Section~4.2]{vershynin2019high}. 
Also note that
$\sup_{H\in\mathcal{H}}\|H\|_\infty
\le2B+\varepsilon\sup_I|\psi|<\infty$.
Equation~\eqref{eq:entropy-bound} follows.
Thus, it remains to prove \eqref{eq:integrand-stability}.

\textit{Step 1: Partition.}
Every element $(A,f,g)$ belonging to \(\mathcal S\) satisfies \(\|f\|_\infty,\|g\|_\infty\le B\) under the prescribed normalization, and inherits the respective coordinatewise Lipschitz constants $24R_xR_y^2$ and $24R_x^2R_y$, respectively.
The interval $I$ contains all dual potential related arguments below.
Write $M=\max\{1,\sup_I\psi'\}$, and we retain $L,\kappa=\kappa_{I,1/2}$ as above.
Take $(A,f,g),(A',f',g')\in\mathcal{S}$ and put
\[
 u=f'-f,\qquad v=g'-g,\qquad
 \xi=\frac{f\oplus g-c_A}{\varepsilon},
 \qquad \rho=\psi'(\xi).
\]
Then $0\le\rho\le M$. Define
\begin{equation*}
 Q(u,v)
 :=
 (\mu\otimes\nu)
 \bigl[\mathbf 1_{\{\rho\ge1/2\}}(u\oplus v)^2\bigr].
\end{equation*}
Fubini's theorem and the first-order condition \eqref{eqn:schrodinger_system} give
\begin{align*}
 (\mu\otimes\nu)[\rho(u\oplus v)]
 &=
 \int_{\mathcal X}
 u(x)
 \int_{\mathcal Y}\rho(x,y)\,d\nu(y)
d\mu(x)+
 \int_{\mathcal Y}
 v(y)
 \int_{\mathcal X}\rho(x,y)\,d\mu(x)
 d\nu(y)\\
 &=
 \int u(x)\,d\mu(x)+\int v(y)\,d\nu(y)
 =
 (\mu\otimes\nu)(u\oplus v).
\end{align*}
Consequently, writing $\Phi_A=\Phi_{\mu,\nu}^{c_A}$,
\begin{align*}
 \Phi_A(f,g)-\Phi_A(f',g')
 =
 \varepsilon(\mu\otimes\nu)
 B_\psi\!\left(
 \xi+\frac{u\oplus v}{\varepsilon},\xi
 \right)\ge
 \frac{\kappa}{\varepsilon}Q(u,v).
\end{align*}
The optimality of $(f',g')$ for $c_{A'}$ yields
\begin{align*}
 \Phi_A(f,g)-\Phi_A(f',g')
                                                                                              
 \le
 2M\|c_A-c_{A'}\|_\infty,
\end{align*}
since
\[
 |\Phi_A(f,g)-\Phi_{A'}(f,g)|
 \le M\|c_A-c_{A'}\|_\infty,
 \qquad
 |\Phi_A(f',g')-\Phi_{A'}(f',g')|
 \le M\|c_A-c_{A'}\|_\infty.
\]
It follows that
\begin{equation}
 Q(u,v)
 \le
 \frac{2M\varepsilon}{\kappa}
 \|c_A-c_{A'}\|_\infty.
 \label{eq:active-gap-bound}
\end{equation}
For $a\in\{1/2,3/4\}$, set $m_a:=\frac{1-a}{M-a}$.
For any fixed $x\in\mathcal X$, the first-order condition \eqref{eqn:schrodinger_system} implies
\[
 1=\int\rho(x,y)\,d\nu(y)
 \le
 a\bigl(1-\nu\{\rho(x,\cdot)\ge a\}\bigr)
 +M\nu\{\rho(x,\cdot)\ge a\}.
\]
Hence
\begin{equation}
 \nu\{\rho(x,\cdot)\ge a\}\ge m_a,
 \qquad a\in\{1/2,3/4\}.
 \label{eq:active-mass_x}
\end{equation}
Analogously, for $y\in\mathcal Y$,
\begin{equation}
 \mu\{\rho(\cdot,y)\ge a\}\ge m_a,
 \qquad a\in\{1/2,3/4\}.
 \label{eq:active-mass_y}
\end{equation}
The common Lipschitz bounds \eqref{eq:gw-spatial-cost-bounds} and H\"older continuity
\eqref{eq:holder-psi} give
\begin{align} \label{eq:density-modulus}
 \sup_{y\in\mathcal Y}
 |\rho(x,y)-\rho(x',y)|
&\le L\varepsilon^{-\frac{1}{1+\beta}}
\sup_{y\in\mathcal Y}
\left|
f(x)-f(x')-c_A(x,y)+c_A(x',y)
\right|^{\frac{1}{1+\beta}}\nonumber\\
&\le L\varepsilon^{-\frac{1}{1+\beta}}
\sup_{y\in\mathcal Y}
 \left(
|f(x)-f(x')|
+|c_A(x,y)-c_A(x',y)|
\right)^{\frac{1}{1+\beta}}\\
&\le
 L\varepsilon^{-\frac{1}{1+\beta}}
 \bigl(48R_xR_y^2\|x-x'\|\bigr)^{\frac{1}{1+\beta}}.\nonumber
\end{align}



The right side of \eqref{eq:density-modulus} is at most
$1/4$ whenever $\|x-x'\|\le r_0:=
\frac{\varepsilon}
{48R_xR_y^2(4L)^{1+\beta}}$.
We hence choose a maximal $r_0/2$-separated set
$\{x_1,\ldots,x_{N_0}\}\subset\mathcal X$.
Its open balls of radius $r_0$ cover $\mathcal X$.
The balls $B(x_j,r_0/4)$ are pairwise disjoint and lie in
$B(0,R_x+r_0/4)\subset\mathbb{R}^{d_x}$. Comparing their volumes gives
$ N_0\le
 \left(1+\frac{4R_x}{r_0}\right)^{d_x}.$
Form a disjoint Borel partition subordinate to the cover
$\{B(x_j,r_0)\}_{j=1}^{N_0}$ and discard zero-$\mu$-mass cells.
Relabel the remaining cells as $E_1,\ldots,E_J$, retaining their
associated centers $x_j$. Then
\[
 E_j\subset B(x_j,r_0),\qquad
 \mu(E_j)>0,\qquad
 \mu\biggl(\,\bigcup_{j=1}^J E_j\biggr)=1,\qquad
 1\le J\le N_0.
\]
This proves \eqref{eq:covering-J-bound}.
Based on this specific partition, define the fixed functionals
\begin{equation*}
\tau_j(h):=\frac1{\mu(E_j)}\int_{E_j}h\,d\mu,
 \qquad
 h\in\mathcal C(\mathcal X),\quad j=1,\ldots,J.
\end{equation*}
They are linear and continuous, with
$|\tau_j(h)|\le\|h\|_\infty$.

$
 F_j=\{y\in\mathcal Y:\rho(x_j,y)\ge3/4\}.
$
By \eqref{eq:active-mass_x},
\begin{equation}\label{eqn:nu(Fj)_lower-bound}
\nu(F_j)\ge m_{3/4}.
\end{equation}
For $x\in E_j$ and $y\in F_j$,
\eqref{eq:density-modulus} gives
$\rho(x,y)\ge3/4-1/4=1/2$. Thus
\begin{equation}
 E_j\times F_j\subset\{\rho\ge1/2\}.
 \label{eq:active-rectangles}
\end{equation}

\textit{Step 2: $L^2$ perturbation bound.}
For every fixed $y\in\mathcal Y$ the exact variance decomposition
gives
\begin{align}\label{eqn:Ejdecomposition}
 \int_{E_j}[u(x)+v(y)]^2\,d\mu(x)
 =
 \int_{E_j}|u(x)-\tau_j(u)|^2\,d\mu(x)+
 \mu(E_j)[\tau_j(u)+v(y)]^2.
\end{align}
Integrating over $F_j$, summing over the disjoint cells, and using
\Cref{eqn:nu(Fj)_lower-bound,eq:active-rectangles,eqn:Ejdecomposition}, we obtain
\begin{align*}
 Q&(u,v)
 \ge
 \sum_{j=1}^J
 \int_{F_j}\int_{E_j}
 |u(x)+v(y)|^2\,d\mu(x)d\nu(y)\\
 &\ge
 \sum_{j=1}^J
 \int_{F_j}\int_{E_j}
 |u(x)-\tau_j(u)|^2\,d\mu(x)d\nu(y)\ge
 m_{3/4}\sum_{j=1}^J
 \int_{E_j}|u(x)-\tau_j(u)|^2\,d\mu(x).
\end{align*}
Because the cell masses sum to one,
\begin{align}
 \|u\|_{L^2(\mu)}^2
 =
 \sum_{j=1}^J\int_{E_j}|u(x)|^2\,d\mu(x)
 &\le
 \sum_{j=1}^J
 \int_{E_j}|u(x)-\tau_j(u)|^2\,d\mu(x)
 +\|\tau(f')-\tau(f)\|_{\ell^\infty_J}^2
 \nonumber\\
 &\le
 \frac1{m_{3/4}}Q(u,v)
 +
 \|\tau(u)\|_{\ell^\infty_J}^2.
 \label{eq:first-potential-l2}
\end{align}
For each $y\in\mathcal Y$, put
$G_y=\{x\in\mathcal X:\rho(x,y)\ge1/2\}$.
Equation~\eqref{eq:active-mass_y} yields
$\mu(G_y)\ge m_{1/2}$.
Integrating
$|v(y)|^2\le2|u(x)+v(y)|^2+2|u(x)|^2$
over $G_y$ gives
\[
 m_{1/2}|v(y)|^2
 \le
 2\int_{G_y}|u(x)+v(y)|^2\,d\mu(x)
 +2\int_{G_y}|u(x)|^2\,d\mu(x).
\]
Integrating over $y\in\mathcal Y$ and using Fubini's theorem gives
\begin{align} \label{eq:second-potential-l2}
 \|v\|_{L^2(\nu)}^2
 &\le
 \frac2{m_{1/2}}
 \biggl(
 \int_{\mathcal Y}\int_{G_y}
 |u(x)+v(y)|^2\,d\mu(x)d\nu(y)
 +
 \int_{\mathcal Y}\int_{G_y}
 |u(x)|^2\,d\mu(x)d\nu(y)
 \biggr)
 \nonumber\\
 &=
 \frac2{m_{1/2}}
 \biggl(
 Q(u,v)
 +
 \int_{\mathcal X}|u(x)|^2
 \nu\{y:\rho(x,y)\ge1/2\}\,d\mu(x)
 \biggr)\\
 &\le
 \frac2{m_{1/2}}
 \bigl(Q(u,v)+\|u\|_{L^2(\mu)}^2\bigr).\nonumber
\end{align}
Combining \eqref{eq:active-gap-bound},
\eqref{eq:first-potential-l2}, and
\eqref{eq:second-potential-l2} proves
\begin{equation}
 \|u\|_{L^2(\mu)}+\|v\|_{L^2(\nu)}
 \le
 C\bigl(\|c_A-c_{A'}\|_\infty^{1/2}
 +
 \|\tau(u)\|_{\ell^\infty_J}
 \bigr).
 \label{eq:both-potentials-l2}
\end{equation}


\textit{Step 3: From $L^2$ perturbation bound to uniform perturbation bound.}
The identity
$
 (\psi'(s)-\psi'(t))(s-t)
 =
 B_\psi(s,t)+B_\psi(t,s)
$
and convexity, together with \eqref{eq:active-curvature}, imply
\begin{equation}
 |\psi'(s)-\psi'(t)|\ge\kappa|s-t|
 \quad\text{if }s,t\in I
 \text{ and }\psi'(t)\ge1/2.
 \label{eq:inverse-slope}
\end{equation}
Since $\psi'$ is nondecreasing, the difference
$
 \psi'\!\left(
 \frac{f'(x)+g(y)-c_A(x,y)}{\varepsilon}
 \right)
 -
 \psi'\!\left(
 \frac{f(x)+g(y)-c_A(x,y)}{\varepsilon}
 \right)
$
has the same sign as $f'(x)-f(x)$.
Thus, \eqref{eq:inverse-slope} and the first-order condition \eqref{eqn:schrodinger_system} indicate that
\begin{align} \label{eq:scalar-inverse}
 &\quad
 \left|
 \int
 \psi'\!\left(
 \frac{f'(x)+g(y)-c_A(x,y)}{\varepsilon}
 \right)d\nu(y)-1
 \right|
 \nonumber\\
 &=
 \int
 \left|
 \psi'\!\left(
 \frac{f'(x)+g(y)-c_A(x,y)}{\varepsilon}
 \right)
 -
 \psi'\!\left(
 \frac{f(x)+g(y)-c_A(x,y)}{\varepsilon}
 \right)
 \right|d\nu(y)
\\
 &\ge
 \int_{\{y,\rho(x,y)\ge1/2\}}
 \left|
 \psi'\!\left(
 \frac{f'(x)+g(y)-c_A(x,y)}{\varepsilon}
 \right)
 -
 \psi'\!\left(
 \frac{f(x)+g(y)-c_A(x,y)}{\varepsilon}
 \right)
 \right|d\nu(y)\nonumber\\
 &\ge
 \frac{\kappa m_{1/2}}{\varepsilon}|u(x)|\nonumber.
\end{align}
At the same time, first-order condition \eqref{eqn:schrodinger_system} for $c_{A'}$ gives 
$
 \int
 \psi'\!\left(
 \frac{f'(x)+g'(y)-c_{A'}(x,y)}{\varepsilon}
 \right)d\nu(y)=1.
$
Using \eqref{eq:holder-psi}, one has
\begin{align*}
 \Bigl|
 \int
 \psi'\Bigl(
 \frac{f'(x)+g(y)-c_A(x,y)}{\varepsilon}
 \Bigr)d\nu(y)-1
 \Bigr|
 &\le
 \frac{L}{\varepsilon^{\frac{1}{1+\beta}}}
 \int
 (|v(y)|+\|c_A-c_{A'}\|_\infty)^{\frac{1}{1+\beta}}
 \,d\nu(y)\\
 &\le
 L\varepsilon^{-\frac{1}{1+\beta}}
 \bigl(
 \|v\|_{L^2(\nu)}^{\frac{1}{1+\beta}}
 +\|c_A-c_{A'}\|_\infty^{\frac{1}{1+\beta}}
 \bigr).
\end{align*}
We used $(r+s)^{\frac{1}{1+\beta}}\le r^{\frac{1}{1+\beta}}+s^{\frac{1}{1+\beta}}$ for $r,s\ge0$
and
$\int|v|^{\frac{1}{1+\beta}}\,d\nu
\le
\left(\int|v|^2\,d\nu\right)^{\frac{1}{2(1+\beta)}}
=
\|v\|_{L^2(\nu)}^{\frac{1}{1+\beta}}.
$
Together with \eqref{eq:scalar-inverse}, this yields
\begin{align}
 \|u\|_\infty
 &\le
 \frac{L\varepsilon^{\frac{\beta}{1+\beta}}}{\kappa m_{1/2}}
 \bigl(
 \|v\|_{L^2(\nu)}^{\frac{1}{1+\beta}}
 +\|c_A-c_{A'}\|_\infty^{\frac{1}{1+\beta}}
 \bigr).
 \label{eq:scalar-upgrade_u}
\end{align}
Analogously,
\begin{align}
 \|v\|_\infty
 &\le
 \frac{L\varepsilon^{\frac{\beta}{1+\beta}}}{\kappa m_{1/2}}
 \bigl(
 \|u\|_{L^2(\mu)}^{\frac{1}{1+\beta}}
 +\|c_A-c_{A'}\|_\infty^{\frac{1}{1+\beta}}
 \bigr).
 \label{eq:scalar-upgrade_v}
\end{align}
Since $\|c_A-c_{A'}\|_\infty\le 40 R_x^2 R_y^2$ by \Cref{lem:gw-cost-bounds}, equations
\eqref{eq:both-potentials-l2}--\eqref{eq:scalar-upgrade_v}
imply
\begin{equation}
 \|u\|_\infty+\|v\|_\infty
 \le
 C\bigl(
 \|c_A-c_{A'}\|_\infty^{\frac{1}{2(1+\beta)}}+\|\tau(u)\|_{\ell^\infty_J}^{\frac{1}{1+\beta}}
 \bigr).
 \label{eq:potential-uniform-stability}
\end{equation}
The function $\psi$ is $M$-Lipschitz on $I$. Hence,
\begin{align*}
 \|H_{A',f',g'}-H_{A,f,g}\|_\infty
 &\le
 (1+M)(\|u\|_\infty+\|v\|_\infty)
 +M\|c_A-c_{A'}\|_\infty\\
 &\le
 C\bigl(
 \|c_A-c_{A'}\|_\infty^{\frac{1}{2(1+\beta)}}+\|\tau(u)\|_{\ell^\infty_J}^{\frac{1}{1+\beta}}
 \bigr).
\end{align*}
Combining with Lemma \ref{lem:gw-cost-bounds}, this proves
\eqref{eq:integrand-stability}.

We use the notations introduced in the proof of \Cref{thm:gw-duality}. For every $\pi\in\Pi(\lambda,\eta)$,
$\|B_\pi\|_F\le R_xR_y$, so $B_\pi/2\in \mathcal D$.
The minimum defining $V_\varepsilon(\lambda,\eta)$ exists by the same
compactness and continuity argument as in
Theorem~\ref{thm:gw-duality}. By \eqref{eq:distortionexpansion}, the completion of the square
used in \eqref{eq:gw-completed-square}, and $B_\pi/2\in \mathcal D$,
taking infima over $A$ and $\pi$ in either order gives
\[
\mathsf{GW}^1(\lambda,\eta)+V_\varepsilon(\lambda,\eta)
=
\inf_{\pi\in\Pi(\lambda,\eta)}
\left\{
\mathcal F_\varepsilon^{\lambda,\eta}(\pi)
-
\Bigl( 8a_\lambda^\top\int x\|y\|^2\,d\pi
+
8b_\eta^\top\int y\|x\|^2\,d\pi
\Bigr)
\right\}.
\]
Moreover, uniformly over $\pi\in\Pi(\lambda,\eta)$,
\[
\Bigl|
8a_\lambda^\top\int x\|y\|^2\,d\pi
+
8b_\eta^\top\int y\|x\|^2\,d\pi
\Bigr|
\le
8R_xR_y^2\|a_\lambda\|
+
8R_x^2R_y\|b_\eta\|.
\]
Comparison of the infima therefore yields
\[
\begin{aligned}
\left|
S_{2,2,\varepsilon}(\lambda,\eta)
-\mathsf{GW}^1(\lambda,\eta)
-V_\varepsilon(\lambda,\eta)
\right|
\le
8R_xR_y^2\|a_\lambda\|
+
8R_x^2R_y\|b_\eta\|.
\end{aligned}
\]
Apply this estimate with $(\lambda,\eta)=(\mu_n,\nu_n)$.
Since $\mu$ and $\nu$ are centered and
$M_{\mu,\nu}\le R_xR_y$, equation
\eqref{eq:gw-variational-duality}, with $M=R_xR_y$, gives
\[
S_{2,2,\varepsilon}(\mu,\nu)
=
\mathsf{GW}^1(\mu,\nu)+V_\varepsilon(\mu,\nu).
\]
The triangle inequality now proves
\eqref{eq:empirical-gw-reduction}.
