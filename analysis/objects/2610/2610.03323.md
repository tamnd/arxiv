<!-- theorem-3: body_md -->
\label{thm:highgirth}
Fix $k\geq 2$ and $r\geq 3$. Let $F$ be a non-empty ${\mathcal G}^{(r)}_{k}$-free $r$-graph. 
                                                                                                     
Then,
$$\liminf_{n\rightarrow \infty}\frac{f^{(r)}(n;rk-2k+2,k)}{n^2}\geq \frac{|F|}{2\,|P_{\le \lfloor k/2\rfloor}(F)|},$$
where we define $P_{\le t}(F):=\{xy\in {V(F)\choose 2}: C_{F}(xy)\cap [t]\not=\varnothing\}$ to consist of all pairs $xy$ of vertices of $F$ such that $C_{F}(xy)$ contains some $i$ with $1\leq i\leq t$.

<!-- lemma-1: body_md -->
\label{lem:k9-support}
Fix an integer $d\ge 3$. For every prime power $q$, there exist pairwise disjoint sets $A_{i,a}$ for $(i,a)\in[d]\times \{0,1\}$, each of size $q^d$, a graph $\Gamma$ on $\bigcup_{i\in[d]}(A_{i,0}\cup A_{i,1})$, and a set $M\subseteq\mathcal{L}_1\times\cdots\times\mathcal{L}_d$, where
\[
\mathcal{L}_i:= \left\{\{u,v\}:u\in A_{i,0},\ v\in A_{i,1}\right\},\quad \text{for $i \in [d]$},
\]
with the following properties.
\begin{enumerate}[label=(\roman*)]
	\item\label{it:0} The set $M$ is a \emph{matching}, that is, for all distinct $d$-tuples $(L_1,\dots,L_d),(L_1',\dots,L_d')\in M$ and every $i\in [d]$ it holds that $L_i\not=L_i'$.
    \item\label{it:1} $|M|=q^{2d}$.
    \item\label{it:2} If $(L_1,\ldots,L_d)\in M$, then $\{uv:u\in L_i,\ v\in L_j\}\subseteq E(\Gamma)$ for every $1\le i<j\le d$.
    \item\label{it:3} $|E(\Gamma)|\le 4\binom{d}{2}q^{2d-1}$.
\end{enumerate}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Let $V:=\mathbb{F}_q^{2d}$ and fix a basis $e_1,\ldots,e_d,f_1,\ldots,f_d$ of $V$. For each $i\in[d]$, define two $d$-dimensional subspaces of $V$:
$$
\begin{aligned}
K_{i,0}&:=\langle e_i\rangle+\langle f_j:j\in[d]\setminus\{i\}\rangle,\\
K_{i,1}&:=\langle f_i\rangle+\langle e_j:j\in[d]\setminus\{i\}\rangle.
\end{aligned}
$$

For each $i\in[d]$ and $a\in\{0,1\}$, let $A_{i,a}$ be a copy of the quotient space $V/K_{i,a}$. We assume that all the sets $A_{i,a}$ are pairwise disjoint and write $\pi_{i,a}:V\to A_{i,a}$ for the quotient map. For every $i\in[d]$, we have $K_{i,0}\cap K_{i,1}=\{0\}$. Indeed, a vector in $K_{i,0}$ has possible $e$-coordinate only in the direction $e_i$ and possible $f$-coordinates only in the directions $f_j$ with $j\ne i$, while a vector in $K_{i,1}$ has possible $f$-coordinate only in the direction $f_i$ and possible $e$-coordinates only in the directions $e_j$ with $j\ne i$. Hence no non-zero vector can occur in both subspaces.

For each $t\in V$ and $i\in[d]$, we define the pair 
\[
L_i(t):=\left\{\pi_{i,0}(t),\pi_{i,1}(t)\right\}.
\]
Define 
                                  
 \[
 M:=\{(L_1(t),\ldots,L_d(t)):t\in V\}\subseteq \mathcal{L}_1\times\cdots\times\mathcal{L}_d.
 \]
 Since $K_{i,0}\cap K_{i,1}=\{0\}$, the linear map $t\mapsto (\pi_{i,0}(t),\pi_{i,1}(t))$ is injective. Since we assumed that $A_{i,0}\cap A_{i,1}=\emptyset$, the map $t\to \{\pi_{i,0}(t),\pi_{i,1}(t)\}$ is also injective. 
 Thus, for each $i$, the pairs $L_i(t)$, $t\in V$, are all distinct and  $M$ is a matching. Clearly, $|M|=|V|=q^{2d}$. This shows Items~\ref{it:0} and~\ref{it:1}.

Next, we define the support graph. For every $1\le i<j\le d$, every $a,b\in\{0,1\}$, and every $t\in V$, put the pair $\{\pi_{i,a}(t),\pi_{j,b}(t)\}$ into $E(\Gamma)$. By construction, if $(L_1(t),\ldots,L_d(t))\in M$, then $\{uv:u\in L_i(t),\ v\in L_j(t)\}\subseteq E(\Gamma)$ for every $1\le i<j\le d$. This proves Item~\ref{it:2}.

We now bound $|E(\Gamma)|$. Fix $i<j$ and $a,b\in\{0,1\}$. The number of distinct pairs of the form $\{\pi_{i,a}(t),\pi_{j,b}(t)\}$, where $t\in V$, is the size of the image of the linear map $t\mapsto(\pi_{i,a}(t),\pi_{j,b}(t))$. Its kernel is $K_{i,a}\cap K_{j,b}$, and this intersection contains a non-zero vector whenever $i\ne j$. Indeed, if $a=b=0$, then, since $d\ge3$, there exists $h\in[d]\setminus\{i,j\}$, and $f_h\in K_{i,0}\cap K_{j,0}$. Similarly, if $a=b=1$, then $e_h\in K_{i,1}\cap K_{j,1}$. If $(a,b)=(0,1)$, then $e_i\in K_{i,0}\cap K_{j,1}$, while if $(a,b)=(1,0)$, then $f_i\in K_{i,1}\cap K_{j,0}$.

Thus $\dim(K_{i,a}\cap K_{j,b})\ge1$, so the image of the linear map $t\mapsto(\pi_{i,a}(t),\pi_{j,b}(t))$ has size at most $q^{2d-1}$. Summing over the $4\binom{d}{2}$ choices of $(i,j,a,b)$ gives $|E(\Gamma)|\le4\binom{d}{2}q^{2d-1}$.

<!-- lemma-2: body_md -->
\label{lem:k9-free}
The $3$-graph $F_{d,q}$ is ${\mathcal G}^{(3)}_{9}$-free.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Since two distinct triples span at least four vertices, it suffices to prove that every set of $h$ edges of $F_{d,q}$ spans at least $h+2$ vertices for $3\le h\le8$, and every set of nine edges spans at least $12$ vertices.

Let $S\subseteq E(F_{d,q})$ be a set of $h$ edges, where $3\le h\le9$. For $t\in V$ and $i\in[d]$, call the index $(t,i)$ \textit{active} if $S$ contains at least one of the two edges $L_i(t)x_t$ and $L_i(t)y_t$, and call it \textit{double active} if $S$ contains both edges. Let $c$ be the number of active indices and $a$ the number of double active indices. Then $h=c+a$, since $a$ of the $c$ active indices
contribute two edges to $S$ and the remaining $c-a$ contribute
one edge each.

For each $i\in[d]$, set $T_i:=\{t\in V:(t,i)\text{ is active}\}$ and $m_i:=|T_i|$. Let $I:=\{i\in[d]:m_i>0\}$, $s:=|I|$, and $m_{\max}:=\max_{i\in I}m_i$. Thus $\sum_{i\in I}m_i=c$.

We first lower-bound the number of core vertices used by $S$. For a positive integer $m\le9$, define $g(m):=\min\{u+v:u,v\in\mathbb N,\ uv\ge m\}$. We compute $g(m)$ for $m \in [9]$:
$$
\begin{array}{c|ccccccccc}
m&1&2&3&4&5&6&7&8&9\\
\hline
g(m)&2&3&4&4&5&5&6&6&6
\end{array}
$$
Fix $i\in I$. Since the map $t\mapsto L_i(t)$ is injective, the $m_i$ pairs $L_i(t)$ with $t\in T_i$ are distinct pairs between $A_{i,0}$ and $A_{i,1}$. If their endpoints occupy $u_i$ vertices in $A_{i,0}$ and $v_i$ vertices in $A_{i,1}$, then $m_i\le u_iv_i$. Hence these $m_i$ pairs use at least $g(m_i)$ core vertices. Since the sets $A_{i,a}$ are pairwise disjoint for different $(i,a)$, the total number of core vertices used by $S$ is at least $\sum_{i\in I}g(m_i)$.

We next lower-bound the number of private vertices used by $S$. Call a block $Q_t$ \emph{active} if it contains at least one edge from $S$. Let $B:=\{t\in V:S\cap Q_t\ne\emptyset\}$ be the set of active blocks. Since some coordinate $i$ satisfies $m_i=m_{\max}$, there are at least $m_{\max}$ active blocks, so $|B|\ge m_{\max}$. Let $B_2$ be the set of blocks containing at least one double active index. Every block can contain double active indices only in its active coordinates, at most $s$ of them; so $|B_2|\ge\lceil a/s\rceil$. Each active block contributes at least one of its two private vertices $x_t,y_t$, and every block in $B_2$ contributes both private vertices. Therefore $v(S)\ge\sum_{i\in I}g(m_i)+m_{\max}+\lceil a/s\rceil$.

It remains to minimize this lower bound for $h\le9$. For fixed $c$ and $a$, define
$$
\nu(c,a):=
\min_{\substack{s\ge1,\ m_1,\ldots,m_s\ge1\\
                m_1+\cdots+m_s=c}}
\left(
   \sum_{\rho=1}^s g(m_\rho)
   +\max_{\rho\in[s]}m_\rho
   +\left\lceil\frac as\right\rceil
\right).
$$
This is a relaxation of the actual situation, since we do not impose $s\le d$, and therefore it still gives a valid lower bound. Since $a=h-c$ and $a\le c$, the possible values of $c$ are $\lceil h/2\rceil,\lceil h/2\rceil+1,\ldots,h$. Substituting the displayed values of $g$, we obtain the following table (the values of the second column are listed in this order of $c$):
$$
\begin{array}{c|c|c}
h&
\{\nu(c,h-c):\lceil h/2\rceil\le c\le h\}&
\min_{\lceil h/2\rceil\le c\le h}\nu(c,h-c)\\
\hline
             
3&6,\ 7&6\\
4&6,\ 8,\ 8&6\\
5&8,\ 9,\ 10&8\\
6&8,\ 9,\ 11,\ 11&8\\
7&10,\ 11,\ 12,\ 12&10\\
8&10,\ 11,\ 12,\ 13,\ 12&10\\
9&12,\ 12,\ 13,\ 13,\ 14&12
\end{array}
$$

Consequently, if $3\le h\le8$, then $v(S)\ge h+2>h+1$, so $S$ is not an $h^-$-configuration. If $h=9$, then $v(S)\ge12>11$, so $S$ is not a $9$-configuration. Thus $F_{d,q}$ contains no $\ell^-$-configuration for $3\le\ell\le8$ and no $9$-configuration. Hence $F_{d,q}$ is ${\mathcal G}^{(3)}_{9}$-free.

<!-- lemma-3: body_md -->
\label{lem:k9-claimed-pairs}
The claimed-pair set of $F_{d,q}$ satisfies $|P_{\le 4}(F_{d,q})|\le(5d+1)N+|E(\Gamma)|$.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Set $R:=P_{1}(F_{d,q})\cup\{x_ty_t:t\in V\}\cup E(\Gamma)$. We first show that $P_{\le 4}(F_{d,q})\subseteq R$.
Now let $uv\inP_{\le 4}(F_{d,q})$. Then for some $p\in\{1,2,3,4\}$ there are $p$ distinct edges $e_1,\ldots,e_p\in E(F_{d,q})$ such that 
\begin{equation}\label{eq:cl-pair}
\left|\{u,v\}\cup\bigcup_{\alpha=1}^p e_\alpha\right|\le p+2.
\end{equation}
We distinguish the four possible values of $p$.

If $p=1$, then the right-hand side of~(\ref{eq:cl-pair}) is equal to $3$, so both $u$ and $v$ lie in $e_1$. Thus $uv\inP_{1}(F_{d,q})\subseteq R$.

Suppose $p=2$. Note that two distinct triples have union of size at least $4$.
                                                     
Hence $|e_1\cup e_2|=4$ and $\{u,v\}\subseteq e_1\cup e_2$. Thus $e_1$ and $e_2$ share exactly two vertices. In $F_{d,q}$ this can happen only when, for some $t\in V$ and $i\in[d]$, $\{e_1,e_2\}=\{L_i(t)x_t,L_i(t)y_t\}$. Indeed, two edges can share two core vertices only if they contain the same core pair $L_i(t)$, and the injectivity of $t\mapsto L_i(t)$ then forces the same $t$ and the same $i$; the distinctness of the two edges implies that the private vertices must be $x_t$ and $y_t$. Also, two distinct edges cannot share one private vertex and one core vertex: the private vertex determines the block $t$, and inside one block the different core pairs $L_i(t)$ are disjoint. Therefore $e_1\cup e_2=L_i(t)\cup\{x_t,y_t\}$. Every pair inside this $4$-set is in the usual $2$-shadow or is the private pair $\{x_t,y_t\}$. Hence $uv\in R$.

Suppose $p=3$. The proof of Lemma~\ref{lem:k9-free} shows that any three edges of $F_{d,q}$ span at least $6$ vertices, whereas the right-hand side of~(\ref{eq:cl-pair}) is $5$. So this case cannot occur.

It remains to consider $p=4$. Lemma~\ref{lem:k9-free} shows that any four edges of $F_{d,q}$ span at least $6$ vertices. Since the right-hand side of~(\ref{eq:cl-pair}) is now $6$, we have $\left|\cup_{\alpha=1}^4e_\alpha\right|=6$ and $\{u,v\}\subseteq\bigcup_{\alpha=1}^4e_\alpha$. We claim that the four edges must be two complete diamonds in one block.

Use the notation from the proof of Lemma~\ref{lem:k9-free} for $S:=\{e_1,e_2,e_3,e_4\}$. Thus $c$ is the number of active indices and $a$ is the number of double active indices. Since $4=c+a$ and $a\le c$, the only possibilities are $(c,a)=(4,0),(3,1),(2,2)$. If $(c,a)=(4,0)$ or $(3,1)$, then the table in the proof of Lemma~\ref{lem:k9-free} shows that we have at least $8$ vertices. Hence equality with $6$ vertices is possible only when $(c,a)=(2,2)$.

In this case both active indices are double. If the two active indices have the same coordinate, then the lower bound is at least $g(2)+2+2=7$, which is impossible. Therefore they have two different coordinates. Since the total number of vertices is exactly $6$, equality must also hold in the private-vertex lower bound, so there is only one active block. Thus there exist $t\in V$ and distinct $i,j\in[d]$ such that
$$
\{e_1,e_2,e_3,e_4\}=\{L_i(t)x_t,L_i(t)y_t,L_j(t)x_t,L_j(t)y_t\}.
$$
Consequently, $\cup_{\alpha=1}^4e_\alpha=L_i(t)\cup L_j(t)\cup\{x_t,y_t\}$. All pairs inside this $6$-set are of one of the following types: usual $2$-shadow pairs, the private pair $x_ty_t$, or cross-pairs between $L_i(t)$ and $L_j(t)$. The first type lies in $P_{1}(F_{d,q})$, the second lies in $\{x_ty_t:t\in V\}$, and the cross-pairs lie in $E(\Gamma)$ by Lemma~\ref{lem:k9-support}. Hence $uv\in R$.

We have shown that $P_{\le 4}(F_{d,q})\subseteq R$.
Note that $|P_{1}(F_{d,q})|\le5dN$. Indeed, for fixed $t\in V$ and $i\in[d]$, the two edges $L_i(t)x_t$ and $L_i(t)y_t$ contribute to the usual $2$-shadow only the following five pairs: the core pair $L_i(t)$, the two pairs joining $x_t$ to the vertices of $L_i(t)$, and the two pairs joining $y_t$ to the vertices of $L_i(t)$. Summing over all $t\in V$ and $i\in[d]$ gives at most $5dN$ pairs.
Therefore
$$
\begin{aligned}
|P_{\le 4}(F_{d,q})|
&\le|P_{1}(F_{d,q})|+\bigl|\{x_ty_t:t\in V\}\bigr|+|E(\Gamma)| \le(5d+1)N+|E(\Gamma)|.
\end{aligned}
$$
This finishes the proof of the lemma.

Note that there may be infinitely many sequences with the $\mathcal{R}$-property and thus infinitely many constraints in~\eqref{lp:S1...Sl}. However, we set $w_{S}:=0$ for every $S$ such that the constant sequence $S^{(k)}:=(S,\dots,S)$ of length $k$ has the $\mathcal{R}$-property

<!-- theorem-4: body_md -->
\label{thm:lp-ub}
Let $r\ge3$ and $k\ge3$, and let $\mathcal{R}$ be a collection of merging rules $\ifthenelse{\equal{A}{}}{B}{A|B}$ with $A, B \subseteq [k-1]$. If $t>0$ is a feasible value of the linear program with constraints \eqref{lp:F}--\eqref{lp:S}, then 
\[
\pi(r,k)\le\frac1{2t}.
\]
<!-- end theorem-4 -->
<!-- proof-5: body_md -->
Suppose $G$ is an arbitrary ${\mathcal G}^{(r)}_{k}$-free $n$-vertex $r$-graph, and let
$\mathcal{M}'^G_{\mathcal{R}}$
be its final cluster partition. Recall that $\mathcal{M}'^G_{\mathcal{R}} \subseteq \mathfrak F_{\mathcal{R}}(r,k)$, and hence, up to isomorphism, each cluster $F \in \mathcal{M}'^G_{\mathcal{R}}$ appears in the linear program.

For every final cluster $F\in \mathcal{M}'^G_{\mathcal{R}}$ and every pair $xy\in\binom{V(F)}2$, define the \emph{weight} of $xy$ contributed by $F$ to be $w_{C_F^{<k}(xy)}$. We then define
\[
w(F):=\sum_{xy\in\binom{V(F)}2}w_{C_F^{<k}(xy)},
\qquad F\in \mathcal{M}'^G_{\mathcal{R}},
\]
and
\[
w(xy):=\sum_{\substack{F\in \mathcal{M}'^G_{\mathcal{R}}\\xy\in\binom{V(F)}2}}w_{C_F^{<k}(xy)},
\qquad xy\in\binom{V(G)}2.
\]

We claim that
\begin{align}
    w(xy)&\le1 &&\text{for every }xy\in\binom{V(G)}2, \label{eq:one}\\
    w(F)&\ge t|F| &&\text{for every }F\in \mathcal{M}'^G_{\mathcal{R}}. \label{eq:weight}
\end{align}
Indeed,~\eqref{eq:weight} follows directly from~\eqref{lp:F}.

To prove~\eqref{eq:one}, fix a pair $xy\in\binom{V(G)}2$ and enumerate all final clusters containing $xy$ as $F_1,\dots,F_m$. If $m=0$, then $w(xy)=0$, so assume $m\ge1$. Set
\[
S_i:=C_{F_i}^{<k}(xy),\qquad i\in[m].
\]
We show that $S_1,\dots,S_m$ has the $\mathcal{R}$-property. Clearly $0\in S_i$ for every $i$, so condition (i) is true. Moreover, no two distinct final clusters are $\mathcal{R}$-mergeable, so condition~\ref{it:R-property} follows from the maximality of the merging procedure.

It remains to verify condition~\ref{it:sum-property}. Suppose for a contradiction that $k\in S_1+\cdots+S_m$, and choose $s_i\in S_i$ such that $\sum_{i=1}^m s_i=k$. For every $i$ with $s_i>0$, the cluster $F_i$ contains $s_i$ distinct edges which, together with $xy$, use at most 
$(r-2)s_i+2$ vertices. Since the clusters are edge-disjoint and all these witnesses contain the common pair $xy$, their union consists of $k$ distinct edges on at most
\[
2+\sum_{i=1}^m(r-2)s_i=(r-2)k+2=rk-2k+2
\]
vertices. This is a $k$-configuration in $G$, a contradiction. Thus $S_1,\dots,S_m$ has the $\mathcal{R}$-property and the constraints in~\eqref{lp:S1...Sl} give
$
w(xy)=\sum_{i=1}^m w_{S_i}\le1.
$
This proves~\eqref{eq:one}.

Consequently,
\[
|G|=\sum_{F\in \mathcal{M}'^G_{\mathcal{R}}}|F|
\le t^{-1}\sum_{F\in \mathcal{M}'^G_{\mathcal{R}}}w(F)
=t^{-1}\sum_{xy\in\binom{V(G)}2}w(xy)
\le t^{-1}\binom n2.
\]
Since $G$ was arbitrary, Lemma~\ref{lem:ub} and $\binom n2 = (1+o(1)) n^2/2$ imply $\pi(r,k)\le1/(2t)$.
