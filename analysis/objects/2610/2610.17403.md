<!-- lemma-1: body_md -->
\label{lem:esteq}
{\rm (a)} For every $z>0$, $n(z)-z\,d(z)=W(\log z)/(Ns_1s_2)$. Hence, whenever $d(z)>0$ (i.e., some posterior draw lies in the set $A$ defined in {\rm (b)}), $z$ is a fixed point of $T$ if and only if $W(\log z)=0$.
{\rm (b)} $W$ is continuous and non-increasing, with $W'(c)=-\sum_{\text{all draws}}\pi_c(1-\pi_c)$. It is strictly decreasing as soon as one draw lies in $A=\{f>0,\,g>0\}$.
{\rm (c)} Define $\widehat{c}:=\sup\{c\in\mathbb{R}:W(c)>0\}\in[-\infty,\infty]$ and $\widehat{\mathcal{Z}}=e^{\widehat{c}}$. If both samples meet $A$, then $\widehat{c}$ is the unique root of $W$ and $\widehat{\mathcal{Z}}$ is the unique positive fixed point of $T$.
{\rm (d)} $W$ is the derivative of the concave log-likelihood $c\mapsto\sum_i\log\pi_c(\tilde\theta_{i})+\sum_j\log\{1-\pi_c(\theta^{\ast}_{j})\}$ and $\widehat{c}$ is the maximum-likelihood estimate of the intercept of a logistic regression of the sample label (proposal $=1$) on the offset $\lambda$ \citep{geyer_estimating_1994,kong_theory_2003}.

<!-- definition-1: body_md -->
\label{def:overlap}
Let $m=s_1 p+s_2 g$ and $\pi^{\star}=s_2 g/m$. Define
\[
H:=\mathbb{E}_m\big[\pi^{\star}(1-\pi^{\star})\big]=s_1s_2\int\frac{pg}{s_1 p+s_2 g},\qquad
I:=\frac{H}{s_1s_2}\in[0,1],\qquad
\beta:=1-I,
\]
and
\[
n_{\mathrm{eff}}:=NH\beta=Ns_1s_2\,I(1-I),\qquad \sigma_{N}^2:=\frac{\beta}{NH}=\frac{1}{Ns_1s_2}\Big(\frac1I-1\Big).
\]
<!-- end definition-1 -->
<!-- lemma-2: body_md -->
\label{lem:overlap}
Let $\varepsilon_B:=\int\min(s_1 p,s_2 g)$ be the Bayes error for classifying a draw from $m$ as coming from $p$ or from $g$, and let $\chi^2(p\|g)=\int p^2/g-1$.
\begin{enumerate}[label={\rm (\alph*)}]
\item $H=s_1\mathbb{E}_p[\pi^{\star}]=s_2\mathbb{E}_g[1-\pi^{\star}]$.
\item $\tfrac12\varepsilon_B\leH\le\varepsilon_B$.
\item $I\ge\{1+s_1\chi^2(p\|g)\}^{-1}$ and $I\ge\{1+s_2\chi^2(g\|p)\}^{-1}$.
\item $\beta=s_1s_2\int(p-g)^2/m$. In particular $\beta=0$ if and only if $p=g$ a.e.
\item $\operatorname{Var}(W(\log\mathcal{Z}))=n_{\mathrm{eff}}$ under independent sampling, and $\sigma_{N}^2=n_{\mathrm{eff}}/(NH)^2$.
\end{enumerate}

<!-- lemma-3: body_md -->
\label{lem:mcse-structure}
For independent draws,
\[
\widehat{\sigma}_{\delta}^2=\frac{\widehat{n}_{\mathrm{eff}}}{(N\widehat{H}_0)^2},\qquad
\widehat{n}_{\mathrm{eff}}:=S_2\widehat V_{\mathrm{prop}}+S_1\widehat V_{\mathrm{post}},\qquad
N\widehat{H}_0:=\sum_{j=1}^{S_1}\widehat\pi_j=\sum_{i=1}^{S_2}(1-\widehat\pi_i).
\]
Moreover $\widehat{H}_0$ and $\widehat{H}:=N^{-1}\sum_{\text{all draws}}\widehat\pi_k(1-\widehat\pi_k)$ both estimate $H$.

<!-- lemma-4: body_md -->
\label{lem:meanvar}
Let $W_t:=W(c^{\star}+t)$, $\mu(t):=\mathbb{E} W_t$ and $v(t):=\operatorname{Var}(W_t)$. Under \cref{asmp:iid}:
\begin{enumerate}[label={\rm (\alph*)}]
\item $\mu(t)=-N(e^t-1)\,\mathbb{E}_m\!\big[\pi^{\star}(1-\pi^{\star})/\{1+(e^t-1)\pi^{\star}\}\big]$. Hence $\operatorname{sign}\mu(t)=-\operatorname{sign}t$ and $NH(1-e^{-|t|})\le|\mu(t)|\le NH(e^{|t|}-1)$.
\item $|\mu(t)+NH t|\le NH(e^{|t|}-1-|t|)$.
\item $v(0)=n_{\mathrm{eff}}$, $v(t)\le NH\cosh^2(t/2)$, and $e^{-2|t|}v(0)\le v(t)\le e^{2|t|}v(0)$.
\end{enumerate}

<!-- theorem-1: body_md -->
\label{thm:concentration}
(a) For every $t\in\mathbb{R}$ and every joint law of the draws, $\{\Delta\le t\}=\{W(c^{\star}+t)\le0\}$, equivalently $\{\Delta>t\}=\{W(c^{\star}+t)>0\}$. Moreover $\{\Delta\ge t\}\subseteq\{W(c^{\star}+t)\ge0\}$, with equality whenever some draw lies in $A=\{f>0,g>0\}$.
(b) Under \cref{asmp:iid}, for all $t>0$,
\[
\mathbb{P}(\Delta\ge t)\le e^{-NH\kappa(t)},\qquad \mathbb{P}(\Delta\le -t)\le e^{-NH\kappa(t)}.
\]

<!-- corollary-1: body_md -->
\label{cor:samplesize}
Under \cref{asmp:iid}, if $NH\ge16\log(2/\delta)$, then with probability at least $1-\delta$,
\[
|\log\widehat{\mathcal{Z}}-\log\mathcal{Z}|\le2\sqrt{\log(2/\delta)/(NH)}.
\]
Consequently, relative accuracy $t\in(0,1/2]$ with confidence $1-\delta$ is guaranteed once $N\ge 8\log(2/\delta)/(t^2\varepsilon_B)$, where $\varepsilon_B$ is the Bayes error of \cref{lem:overlap}.
<!-- end corollary-1 -->
<!-- theorem-2: body_md -->
\label{thm:BE}
Under \cref{asmp:iid},
\[
\sup_{x\in\mathbb{R}}\Big|\mathbb{P}\big(\log\widehat{\mathcal{Z}}-\log\mathcal{Z}\le x\,\sigma_{N}\big)-\Phi(x)\Big|\le\frac{3.1}{\sqrt{n_{\mathrm{eff}}}}.
\]

<!-- remark-1: body_md -->
\label{rem:nearperfect}
If $g\approx p$, then $\beta\to0$ and $n_{\mathrm{eff}}\to0$ even though the estimator is nearly exact ($\sigma_{N}\to0$). The theorem then says little, because it measures the error in units of the vanishing $\sigma_{N}$. The bound uses $|\pi-\mathbb{E}\pi|\le1$ to control third moments, which is loose when the responsibilities barely fluctuate. The regime of practical concern is poor overlap, where $\beta\approx1$ and $n_{\mathrm{eff}}\approx NH$.

<!-- theorem-3: body_md -->
\label{thm:mcse}
Assume \cref{asmp:iid} with $S_1,S_2\ge2$, and $\delta\in(0,1)$ with $n_{\mathrm{eff}}\ge64\log(12/\delta)$, then with probability at least $1-\delta$,
\[
\Big|\log\frac{\widehat{\sigma}_{\delta}}{\sigma_{N}}\Big|\le10\sqrt{\frac{\log(12/\delta)}{n_{\mathrm{eff}}}}.
\]
Let $\mathcal C_\alpha=[\log\widehat{\mathcal{Z}}-z_{1-\alpha/2}\widehat{\sigma}_{\delta},\ \log\widehat{\mathcal{Z}}+z_{1-\alpha/2}\widehat{\sigma}_{\delta}]$. If $n_{\mathrm{eff}}\ge350$, then
\[
\big|\mathbb{P}(\log\mathcal{Z}\in\mathcal C_\alpha)-(1-\alpha)\big|\le\frac{7.2}{\sqrt{n_{\mathrm{eff}}}}+2z_{1-\alpha/2}(e^{r}-1)\,\phi\big(z_{1-\alpha/2}e^{-r}\big),\qquad r=10\sqrt{\frac{\log(12\sqrt{n_{\mathrm{eff}}})}{n_{\mathrm{eff}}}}.
\]
In particular, the coverage error is $O(\sqrt{\logn_{\mathrm{eff}}/n_{\mathrm{eff}}})$ uniformly over all $(p,g)$.

<!-- proposition-2: body_md -->
\label{prop:onesided}
Let $M_{\mathrm{post}}:=\sum_{j=1}^{S_1}\pi^{\star}(\theta^{\ast}_{j})$, $M_{\mathrm{prop}}:=\sum_{i=1}^{S_2}\{1-\pi^{\star}(\tilde\theta_{i})\}$ and $\varsigma:=S_1/(S_1-1)+S_2/(S_2-1)$.
\begin{enumerate}[label={\rm (\alph*)}]
\item For any draws, independent or not, $\widehat{n}_{\mathrm{eff}}\le\varsigma\,N\widehat{H}_0\le\varsigma\max\{M_{\mathrm{post}},M_{\mathrm{prop}}\}$.
\item Under \cref{asmp:iid}, $M_{\mathrm{post}}$ and $M_{\mathrm{prop}}$ have mean $NH$, and for every $x\ge NH$,
\[
\mathbb{P}(\widehat{n}_{\mathrm{eff}}\ge\varsigma x)\le2e^{-NH}\Big(\frac{eNH}{x}\Big)^{x}.
\]
Moreover, for every $\delta\in(0,1)$, with probability at least $1-\delta$,
\[
\widehat{n}_{\mathrm{eff}}\le\varsigma\Big(\sqrt{NH}+\sqrt{\tfrac23\log(2/\delta)}\Big)^2.
\]
\item If $I\le1/2$, then $NH\le2n_{\mathrm{eff}}$. If $NH\le1$ and $S_1,S_2\ge100$, then $\mathbb{P}(\widehat{n}_{\mathrm{eff}}\ge10)\le0.04$.
\end{enumerate}

<!-- theorem-4: body_md -->
\label{thm:sharp}
Let $\varepsilon\in(0,1/2]$ and $S\ge1$. Let $\Theta=\{0,1,2\}$ with $p=(1-\varepsilon)\delta_0+\varepsilon\delta_1$ and $g=\varepsilon\delta_1+(1-\varepsilon)\delta_2$, and take $S_1=S_2=S$. Then $H=\varepsilon/4$, $\beta=1-\varepsilon$, $n_{\mathrm{eff}}=S\varepsilon(1-\varepsilon)/2$, and $\Delta=\log(n_{\mathrm{prop}}/n_{\mathrm{post}})$ whenever both counts are positive, where $n_{\mathrm{post}},n_{\mathrm{prop}}\sim\mathrm{Bin}(S,\varepsilon)$ are the independent numbers of draws landing on the overlap point. Moreover:
\begin{enumerate}[label={\rm (\alph*)}]
\item $\mathbb{P}(\widehat{\mathcal{Z}}\in\{0,\infty\})\ge e^{-4NH}$;
\item if $n_{\mathrm{eff}}\ge2$, then $\sup_x|\mathbb{P}(\Delta\le x\sigma_{N})-\Phi(x)|\ge0.04/\sqrt{n_{\mathrm{eff}}}$.
\end{enumerate}

<!-- remark-2: body_md -->
\label{rem:ISvsbridge}
\citet{chatterjee_sample_2018} show that importance sampling from $g$ to $p$ needs about $\exp\{\mathrm{KL}(p\|g)\}$ draws (see \citet{jarzynski_rare_2006} for the analogous law for exponential work averages). For the optimal bridge the corresponding requirement is $N\asymp1/H\asymp1/\varepsilon_B$. For two Gaussians with a mean shift $\|\mu\|$, $\mathrm{KL}=\|\mu\|^2/2$ while $\varepsilon_B\approx e^{-\|\mu\|^2/8}$ up to polynomial factors, so the exponent improves by a factor of four. The corresponding asymptotic comparison of relative variances, $e^{\|\mu\|^2}$ for importance sampling against $\|\mu\|e^{\|\mu\|^2/8}$ for the optimal bridge, is due to MW, Equation (6.1), (6.4). Since $H\le\varepsilon_B$ (\cref{lem:overlap}(b)), the requirement grows exponentially in the dimension whenever the Bayes error decays exponentially, which quantifies the known curse of dimensionality of bridge sampling \citep{wang_warp_2022}. For BAR, \citet{reinhardt_small_2021} found empirically that the sample size per state at which BAR's mean squared error falls below that of the linear estimator (with the constant in BAR set to its exact value) follows $n\approx b\,\Omega^{-a}$, with $a=0.77$--$0.83$ and $b=6.0$--$8.5$. Here $\Omega$ is our $I$, and their test systems have $s_1=s_2=1/2$. Translated into our units, $n_{\mathrm{eff}}=(n/2)\Omega(1-\Omega)$ along their fitted crossovers lies between about $0.5$ and $2.5$ for all four systems and all $\Omega$ between $4\times10^{-4}$ and $1/2$, while $n$ itself varies by a factor of $200$--$300$ over that range. By our reading of their Figure~2(b,c), the actual mean squared error of BAR also departs from Bennett's asymptotic formula at $n_{\mathrm{eff}}\approx2$, for both $n=20$ and $n=1000$. These observations are consistent with our finding (\cref{sec:exp-universality}) that the transition to asymptotic behaviour is governed by $n_{\mathrm{eff}}$ rather than by $n$ or $I$ separately. Constant $n_{\mathrm{eff}}$ would appear over their range of $\Omega$ as a power law with exponent about $0.9$. The slightly smaller fitted exponents indicate that their crossover value of $n_{\mathrm{eff}}$ decreases slowly as the overlap decreases, which is plausible since their criterion is a comparison with another estimator rather than a reliability criterion.

<!-- assumption-2: body_md -->
\label{asmp:mcmc}
The proposal draws are i.i.d.\ from $g$ and independent of the posterior draws. The posterior draws $\theta^{\ast}_{1},\dots,\theta^{\ast}_{S_1}$ are consecutive states of a Markov chain with transition kernel $P$ that is reversible with respect to $p$ and started at stationarity, $\theta^{\ast}_{1}\sim p$. As an operator on $L^2_0(p)=\{u\in L^2(p):\mathbb{E}_pu=0\}$, $P$ has spectrum in $[-1+\gamma_-,1-\gamma]$ for some $\gamma\in(0,1]$ and $\gamma_-\in(0,2-\gamma]$. We write $\gamma_{*}=\min(\gamma,\gamma_-)$ and assume $H>0$.
<!-- end assumption-2 -->
<!-- definition-2: body_md -->
\label{def:mcmc-overlap}
Let
\[
v_P:=S_2\operatorname{Var}_g(\pi^{\star})+S_1\sigma_P^2(\pi^{\star}),\qquad \sigma_{N,P}^2:=\frac{v_P}{(NH)^2},\qquad \tau_W:=\frac{v_P}{n_{\mathrm{eff}}},\qquad n_{\mathrm{eff}}^{P}:=\frac{n_{\mathrm{eff}}}{\tau_W}=\frac{n_{\mathrm{eff}}^2}{v_P}.
\]
<!-- end definition-2 -->
<!-- theorem-5: body_md -->
\label{thm:mcmc-conc}
Assume \cref{asmp:mcmc}. Let $b_P:=5/\gamma$ if $\gamma<1$ and $b_P:=1/3$ if $\gamma=1$, and
\[
\kappa_P(t):=\frac{(1-e^{-t})^2}{2c_{+}\cosh^2(t/2)+2b_P(1-e^{-t})},\qquad t\ge0.
\]
\begin{enumerate}[label={\rm (\alph*)}]
\item For all $t>0$, $\mathbb{P}(\Delta\ge t)\le e^{-NH\kappa_P(t)}$ and $\mathbb{P}(\Delta\le-t)\le e^{-NH\kappa_P(t)}$.
\item $\mathbb{P}(\widehat{\mathcal{Z}}\in\{0,\infty\})\le e^{-NH}+e^{-\gamma NH(1-1/S_1)}$.
\item Suppose instead that the chain is started from a law $q$ with $\chi^2(q\|p)<\infty$, and that its first $t_0$ states are discarded. Then every event $E$ determined by the draws satisfies $\mathbb{P}_q(E)\le\mathbb{P}_p(E)+\tfrac12(1-\gamma_{*})^{t_0}\chi^2(q\|p)^{1/2}$ and $\mathbb{P}_q(E)\le\{1+(1-\gamma_{*})^{2t_0}\chi^2(q\|p)\}^{1/2}\mathbb{P}_p(E)^{1/2}$, where $\mathbb{P}_p$ refers to the stationary chain.
\end{enumerate}
<!-- end theorem-5 -->
<!-- theorem-6: body_md -->
\label{thm:mcmc-BE}
Assume \cref{asmp:mcmc} with $S_1\ge2$, and let $\zeta_P:=(c_{+}/c_{-})^{1/2}$. Then
\[
\sup_{x\in\mathbb{R}}\big|\mathbb{P}(\Delta\le x\,\sigma_{N,P})-\Phi(x)\big|\le\frac{C_P}{\sqrt{v_P}}+0.63(1+\zeta_P)\,\sigma_{N,P}+e^{-NH\kappa_P(1/(4+4\zeta_P))}+\frac{0.12}{\gamma_{*}}\,e^{-\gamma_{*}S_1/2},
\]
where $C_P$ depends only on $(\gamma,\gamma_{*},\gamma_-)$ and is given in \cref{eq:CP}. If the chain is positive, then $C_P\le83/\gamma$.

<!-- theorem-7: body_md -->
\label{thm:mcmc-sharp}
In the two-point family of \cref{thm:sharp}, let the posterior draws be the stationary Markov chain on $\{0,1\}$ with $P(0,1)=\varepsilon\gamma$ and $P(1,0)=(1-\varepsilon)\gamma$, where $\gamma\in(0,1]$. This chain is positive and satisfies \cref{asmp:mcmc} with right and absolute spectral gap $\gamma$. Moreover $\tau_W=1/\gamma$, $v_P=n_{\mathrm{eff}}/\gamma$ and $n_{\mathrm{eff}}^{P}=\gamman_{\mathrm{eff}}$, and:
\begin{enumerate}[label={\rm (\alph*)}]
\item $\mathbb{P}(\widehat{\mathcal{Z}}\in\{0,\infty\})\ge(1-\varepsilon)(1-\varepsilon\gamma)^{S-1}\ge\tfrac12e^{-4\gamma NH}$;
\item if $\varepsilon\le1/4$ and $S=\lceil1/(\varepsilon\gamma)\rceil$, then $\sup_x|\mathbb{P}(\Delta\le x\sigma_{N,P})-\Phi(x)|\ge0.15$, while $v_P\ge3/(8\gamma^2)$. Consequently no bound of the form $C/\sqrt{v_P}$ holds for all reversible chains with spectral gap $\gamma$ unless $C\ge0.09/\gamma$.
\end{enumerate}

<!-- theorem-8: body_md -->
\label{thm:mcmc-mcse}
Assume \cref{asmp:mcmc} with $S_1,S_2\ge2$. Let $\widehat\omega^2$ be any estimate of $\sigma^2_P(\pi^{\star})$, and let $\varepsilon_\omega\in[0,1/2]$ and $\delta_\omega\in[0,1]$ be such that $\mathbb{P}(|\widehat\omega^2/\sigma_P^2(\pi^{\star})-1|>\varepsilon_\omega)\le\delta_\omega$. Let $\delta\in(0,1)$, $L=\log(12/\delta)$, $\ell_P=c_{+} L/(NH)$ and $r_P=2\sqrt{L/v_P}+\varepsilon_\omega$. If $\ell_P\le1/64$ and $r_P\le1/2$, then with probability at least $1-\delta-\delta_\omega$,
\[
\Big|\log\frac{\widehat{\sigma}_{\delta,P}}{\sigma_{N,P}}\Big|\le8.4\sqrt{\ell_P}+1.4\,r_P=:r .
\]
Consequently the intervals $\mathcal C^P_\alpha=[\log\widehat{\mathcal{Z}}-z_{1-\alpha/2}\widehat{\sigma}_{\delta,P},\ \log\widehat{\mathcal{Z}}+z_{1-\alpha/2}\widehat{\sigma}_{\delta,P}]$ satisfy
\[
\big|\mathbb{P}(\log\mathcal{Z}\in\mathcal C^P_\alpha)-(1-\alpha)\big|\le2B_P+\delta+\delta_\omega+2z_{1-\alpha/2}(e^{r}-1)\,\phi\big(z_{1-\alpha/2}e^{-r}\big),
\]
where $B_P$ is the right-hand side of \cref{thm:mcmc-BE}.

<!-- proposition-3: body_md -->
\label{prop:orth}
Let $h_{\zeta,\vartheta}=\{s_1 f+s_2\zeta g_\vartheta\}^{-1}$ for $\zeta>0$ and a proposal $g_\vartheta$ from any family.
\begin{enumerate}[label={\rm (\alph*)}]
\item For all $(\zeta,\vartheta)$ with $H(\vartheta)>0$: $\mathbb{E}_{g_\vartheta}[f h_{\zeta,\vartheta}]-\mathcal{Z}\,\mathbb{E}_p[g_\vartheta h_{\zeta,\vartheta}]=0$. Hence every derivative of this population moment with respect to the nuisance quantities $(\zeta,\vartheta)$ vanishes identically.
\item The population fixed-point map is constant: $T_{\mathrm{pop}}(\zeta):=\mathbb{E}_g[f h_\zeta]/\mathbb{E}_p[g h_\zeta]=\mathcal{Z}$ for all $\zeta>0$. The population MW iteration therefore terminates after one step from any starting value.
\item The moment map $\Psi(z)=\mathbb{E}_g[fh_z]-z\mathbb{E}_p[gh_z]$, in which $z$ appears both as the target and inside the bridge, factorizes as $\Psi(z)=(1-z/\mathcal{Z})K(z)$ with $K(z)=\int fg/(s_1 f+s_2 zg)>0$. It has the unique root $\mathcal{Z}$, and $\Psi'(\mathcal{Z})=-K(\mathcal{Z})/\mathcal{Z}=-\mathbb{E}_p[g h_\mathcal{Z}]$.
\end{enumerate}

<!-- theorem-9: body_md -->
\label{thm:iteration}
Suppose both samples meet $A=\{f>0,g>0\}$.
\begin{enumerate}[label={\rm (\alph*)}]
\item $\varphi'(x)=\mathsf A(x)-\mathsf B(x)$, where
\[
\mathsf A(x)=\frac{\sum_j\pi_x(\theta^{\ast}_{j})^2}{\sum_j\pi_x(\theta^{\ast}_{j})}\qquad\text{and}\qquad \mathsf B(x)=\frac{\sum_i\pi_x(\tilde\theta_{i})\{1-\pi_x(\tilde\theta_{i})\}}{\sum_i\{1-\pi_x(\tilde\theta_{i})\}}
\]
are weighted averages of responsibilities in $(0,1)$. The weights are those of the bridge-weighted posterior and proposal samples, whose population versions coincide.
\item $L:=\sup_x|\varphi'(x)|<1$. The iteration $x_{t+1}=\varphi(x_t)$ converges to $\widehat{c}$ from every starting value, with $|x_t-\widehat{c}|\le L^t|x_0-\widehat{c}|$ and $|x_t-\widehat{c}|\le|x_{t+1}-x_t|/(1-L)$.
\item At the solution,
\[
T'(\widehat{\mathcal{Z}})=\varphi'(\widehat{c})=1-\frac{\widehat{H}}{\widehat{H}_0}=-\frac{\sum_{k}(L_k-\widehat\pi_k)\,\widehat\pi_k}{\sum_{j}\widehat\pi_j}.
\]
\item Under \cref{asmp:iid}, if $NH\ge64\log(6/\delta)$, then with probability at least $1-\delta$,
\[
|\log\{1-T'(\widehat{\mathcal{Z}})\}|\le5.9\sqrt{\log(6/\delta)/(NH)}.
\]
\end{enumerate}

<!-- theorem-10: body_md -->
\label{thm:proposal}
Assume that the posterior draws are i.i.d.\ from $p$, that $s_1$ is fixed as $N\to\infty$, that $\widehat\vartheta\xrightarrow{\,\Pr\,}\vartheta_0$ ($\widehat\vartheta$ may be computed from the same posterior draws), and:
\begin{enumerate}[label={\rm(P\arabic*)}]
    \item On a compact neighbourhood of $\vartheta_0$ the maps $\vartheta\mapsto\log g_\vartheta(\theta)$ and $\vartheta\mapsto\log\{g_\vartheta(\tau_\vartheta(\epsilon))/f(\tau_\vartheta(\epsilon))\}$ are finite and Lipschitz, with Lipschitz constants square-integrable under $p$ and $\nu$ respectively (this requires $g_\vartheta>0$ $p$-a.s.\ and $f>0$ on the support of $g_\vartheta$);
    \item $H(\vartheta_0)>0$ and $\vartheta\mapstoH(\vartheta)$ is continuous at $\vartheta_0$.
\end{enumerate}
Then $\sqrt N\{\widehat{c}(\widehat\vartheta)-\widehat{c}(\vartheta_0)\}\xrightarrow{\,\Pr\,}0$. If moreover $\beta(\vartheta_0)>0$ (i.e., $g_{\vartheta_0}\neq p$), then $(\widehat{c}(\widehat\vartheta)-\log\mathcal{Z})/\sigma_{N}(\vartheta_0)\Rightarrow\mathcal N(0,1)$ and the delta-method MCSE computed with $g_{\widehat\vartheta}$ is consistent for $\sigma_{N}(\vartheta_0)$, whether or not the sample is split. If instead $g_{\vartheta_0}=p$, then $\widehat{c}(\vartheta_0)=\log\mathcal{Z}$ and the error of $\widehat{c}(\widehat\vartheta)$ is $o_\mathbb{P}(N^{-1/2})$.

<!-- remark-3: body_md -->
\label{rem:secondorder}
(i) When $\widehat\vartheta$ is computed from the same posterior draws, the $o_\mathbb{P}(N^{-1/2})$ remainder in \cref{thm:proposal} has non-zero mean. Expanding one order further, this mean is $(NH)^{-1}\mathbb{E}[\partial_\vartheta\{W_N-\mathbb{E} W_N\}(c^{\star},\vartheta_0)\cdot(\widehat\vartheta-\vartheta_0)]-H^{-1}\nabla_\varthetaH(\vartheta_0)\cdot\mathbb{E}[(\widehat{c}-c^{\star})(\widehat\vartheta-\vartheta_0)]$, up to smaller terms; the first term is typically the dominant one. For fixed $k$ this bias is $O(N^{-1})$, so its ratio to the MCSE is $O(N^{-1/2})$, with a constant that grows with $k$. This is the ``correlation-induced bias'' found by simulation in \citet{wong_properties_2020}, including its growth with dimension: a Gaussian proposal with full covariance has $k=d+d(d+1)/2$ parameters. With splitting or cross-fitting, $\widehat\vartheta$ is independent of the draws used in $W_N$, and both terms have mean zero at this order. Splitting is therefore advisable when the proposal has many parameters relative to the effective overlap; otherwise it only wastes draws. \Cref{sec:exp-proposal} quantifies this: with $2d$ fitted parameters and $S_1=S_2=2000$, the same-draw bias grows from $0.3$ to $1.1$ MCSEs as $d$ increases from $2$ to $32$, while splitting removes it.
(ii) A natural proposal-fitting diagnostic is $R_g^2=(\partial_\vartheta\widehat{\mathcal{Z}})\Sigma_\vartheta(\partial_\vartheta\widehat{\mathcal{Z}})^\top/\widehat{\sigma}_{\delta}^2$, where $\Sigma_\vartheta$ is the covariance of $\widehat\vartheta$ and $\partial_\vartheta\widehat{\mathcal{Z}}$ is computed by differentiating $T$ while \emph{holding the proposal draws fixed}. That derivative omits the score term $\mathbb{E}_{g_\vartheta}[(1-\pi)\partial_\vartheta\log g_\vartheta]$ generated by the dependence of the proposal draws on $\vartheta$. Its expectation is therefore of order one instead of zero, and $R_g$ converges to a positive constant unrelated to the variance of $\widehat{\mathcal{Z}}$. The pathwise derivative, computed with common random numbers, is $O_\mathbb{P}((NH)^{-1/2})$, and the implied variance contribution is of smaller order than $\sigma_{N}^2$ (\cref{sec:exp-proposal}). The regime is different if the proposal draws are held fixed and $\vartheta$ is re-estimated from those same draws. First-order effects are then real, and they can even reduce the variance, as for importance sampling with an estimated sampler \citep{henmi_importance_2007} and for NCE \citep{uehara_analysis_2018}. This may be a route to improving bridge sampling, but it does not measure the variance of the standard estimator.
(iii) For the same reason, a jackknife that refits $\vartheta$ within each deletion must regenerate the proposal draws from the same base variates. Refitting while holding the proposal draws fixed inflates the jackknife variance spuriously (\cref{sec:exp-proposal}).

<!-- remark-4: body_md -->
\label{rem:calib-power}
Part (b) separates the two effects of a sampler error $h$ on the overlap region. The bias of $\log\widehat{\mathcal{Z}}$ is governed by the mean $\mathbb{E}_{m_\star}[h]$, the test by the covariance $\operatorname{Cov}_{m_\star}(\pi^{\star},h)$. Under local alternatives $\tilde p=p(1+h/\sqrt N)$, the expansions in the proofs show that $t_\mathcal{T}$ converges to a noncentral $t_{G-1}$ law with noncentrality $\sqrt{H}\,\operatorname{Cov}_{m_\star}(\pi^{\star},h)/\mathrm{sd}_{m_\star}(\pi^{\star})$, while the bias is $-\mathbb{E}_{m_\star}[h]\sqrt{H/\beta}$ MCSEs. The test therefore detects samplers whose error varies with $\pi^{\star}$ on the overlap region, such as samplers that misweight modes that the proposal covers unequally (\cref{sec:exp-failure}). It is blind to errors that are nearly constant there, and part (c) shows that this is not a weakness of the test: the responsibilities depend on the draws only through $f/g$, and an error whose average over the level sets of $f/g$ is constant on $A$ leaves no trace in them, although it biases $\log\widehat{\mathcal{Z}}$ by $\log\{p(A)/\tilde p(A)\}$. The typical case is a mode that the sampler misses and the proposal does not cover. In \cref{app:repro}, such a mode carrying $30\%$ of the posterior mass produces a bias of $21$ MCSEs and zero interval coverage, while $t_\mathcal{T}$ rejects at the nominal rate. Errors confined to $A^c$ that leave $\tilde p(A)=p(A)$ do not affect $\widehat{\mathcal{Z}}$ at all.

\label{thm:jackknife}

\label{rem:thresholds}
(i) The threshold $\chi^2_{G-1,1-\alpha}/(G-1)$ for $\mathcal{V}_{\mathrm{jack}}$ is correct in the fixed-$G$ regime. Under the pooled law, $\mathbb{E}[L-\tilde\pi]=\widetilde F(\tilde c)=0$ and $\mathbb{E}[(L-\tilde\pi)\tilde\pi]=s_2\mathbb{E}_g[\tilde\pi(1-\tilde\pi)]-s_1\mathbb{E}_{\tilde p}[\tilde\pi^2]$, which gives the second. The expansion in the proof of \cref{thm:calib-null}, carried out around $\tilde c$ and $\mathcal{T}_\infty$, shows that $\mathcal{T}$ and the $\mathcal{T}_{(-g)}$ are asymptotically linear with bounded influence functions. Hence $\widehat{\mathrm{se}}_{\mathrm{jack}}(\mathcal{T})=O_\mathbb{P}(N^{-1/2})$, and $|t_\mathcal{T}|\xrightarrow{\,\Pr\,}\infty$ when $\mathcal{T}_\infty\ne0$.

(b) $\widetilde F(c)=F(c)-\epsilons_1\mathbb{E}_p[h\pi_c]$, where $F$ is the population function under $p$, with $F(c^{\star})=0$ and $F'(c^{\star})=-H$. Since $s_1\mathbb{E}_p[h\pi^{\star}]=\mathbb{E}_m[\pi^{\star}(1-\pi^{\star})h]=H\mathbb{E}_{m_\star}[h]$, the implicit function theorem gives $\tilde c-c^{\star}=-\epsilon\mathbb{E}_{m_\star}[h]+O(\epsilon^2)$. Let $\mathcal N(c,\epsilon)=s_1\mathbb{E}_{\tilde p}[\pi_c^2]-s_2\mathbb{E}_g[\pi_c(1-\pi_c)]$ be the numerator of $\mathcal{T}_\infty$ at $c$. It vanishes at $(c^{\star},0)$, its $\epsilon$-derivative there is $s_1\mathbb{E}_p[h{\pi^{\star}}^2]=H\mathbb{E}_{m_\star}[\pi^{\star} h]$, and its $c$-derivative there is $2\mathbb{E}_m[{\pi^{\star}}^2(1-\pi^{\star})^2]-\mathbb{E}_m[{\pi^{\star}}^2(1-\pi^{\star})(1-2\pi^{\star})]=\mathbb{E}_m[{\pi^{\star}}^2(1-\pi^{\star})]=H\mathbb{E}_{m_\star}[\pi^{\star}]$. Hence $\mathcal N(\tilde c,\epsilon)=\epsilonH\{\mathbb{E}_{m_\star}[\pi^{\star} h]-\mathbb{E}_{m_\star}[\pi^{\star}]\mathbb{E}_{m_\star}[h]\}+O(\epsilon^2)$, and dividing by $\widetilde H_0=H+O(\epsilon)$ proves the claim.

(c) Responsibilities depend on a draw only through $f/g$, and they equal $0$ for posterior draws outside $A$ and $1$ for proposal draws outside $A$. Integrating the condition over $A$ gives $\tilde p(A)=\lambda p(A)$. Let $p''=\lambda p$ on $A$ and $p''=\tilde p$ on $A^c$, a probability density. On $A$ the laws of $f/g$ under $\tilde p$ and under $p''$ both equal $\lambda$ times its law under $p$, and both put mass $\tilde p(A^c)$ outside $A$, so the responsibilities of i.i.d.\ draws from $\tilde p$ and from $p''$ have the same joint law. Let $f''=(\mathcal{Z}/\lambda)p''$. Then $f''=f$ on $A$, and $f''=0=f$ where $g>0$ and $p=0$. Hence responsibilities computed from $f''$ coincide with those computed from $f$ at every point where a draw can fall, and the problem $(p'',f'',g)$ is correctly specified with evidence $\int f''=\mathcal{Z}/\lambda$. The bridge estimator is consistent in that problem, so $\widehat{\mathcal{Z}}\to\mathcal{Z}/\lambda$.
