<!-- theorem-1: body_md -->
\label{thm:local}
There exist constants $\eta_{\mathrm{loc}}>0,$
such that whenever
\[h\boldsymbol{\theta}  \in B(\mathbf{k}_1, \eta_{\mathrm{loc}})
\quad \text{or}\quad 
h\boldsymbol{\theta}  \in B(\mathbf{k}_2, \eta_{\mathrm{loc}})\]
we have
\begin{equation*}
|1-Z(h\boldsymbol{\theta})|\asymp  \rho_h(\boldsymbol{\theta}).
\end{equation*}
<!-- end theorem-1 -->
<!-- remark-1: body_md -->
\label{rem:local}
We note that the conclusion of Theorem~\ref{thm:local} remains valid if the condition \(h\boldsymbol{\theta}  \in B(\mathbf{k}_1, \eta_{\mathrm{loc}})
\text{~or~}
h\boldsymbol{\theta}  \in B(\mathbf{k}_2, \eta_{\mathrm{loc}})\) is replaced by \(|1 - Z(h\boldsymbol{\theta})| \le \eta_{\mathrm{loc}}\).
<!-- end remark-1 -->
<!-- definition-1: body_md -->
Let $\lambda > 0$. We say that $\boldsymbol{\theta} \in \mathbb{T}^2$ is of \emph{$\lambda$-Diophantine type} if for every $\varepsilon > 0$, there exists a constant $c(\boldsymbol{\theta},\varepsilon ) > 0$ such that 
\[
\rho_h(\boldsymbol{\theta}) \ge c(\boldsymbol{\theta},\varepsilon )\cdot |h|^{-(\lambda+\varepsilon)} \quad (|h| \ge 1).
\]
<!-- end definition-1 -->
<!-- theorem-2: body_md -->
\label{LI}
Suppose  $0 < \lambda < 1$, and   $\boldsymbol{\theta} = (\sigma, \delta)$ is of $\lambda$-Diophantine type. Then
\[
1, ~\sigma,~ \delta
 \text{ are linearly independent over } \mathbb Q.
\]
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{full}
For $\lambda \ge \frac{2}{3}$, the set
\[
D_\lambda := \left\{ \boldsymbol{\theta} \in \mathbb{T}^2\colon \boldsymbol{\theta} \text{ is of } \lambda\text{-Diophantine type} \right\}
\]
has full Lebesgue measure in \(\mathbb{T}^2\).
<!-- end theorem-3 -->
<!-- theorem-4: body_md -->
\label{thm:intro-deterministic-PPC}
If $\boldsymbol{\theta} = (\sigma, \delta)$ is of $2/3$-Diophantine type, then the Bernoulli $\alpha\beta$-orbit $\{t_n(\boldsymbol{\omega})\}$ has PPC with probability one, where the probability is taken over the Bernoulli sequence $(\xi_i)_{i \ge 1}$.
<!-- end theorem-4 -->
<!-- theorem-5: body_md -->
\label{thm:intro-smooth-moments}
Let \(\boldsymbol{\theta}\) satisfy the \(2/3\)-Diophantine condition. Then for  \(0 < \varepsilon < 1/18\) and a real-valued even function \(f \in C_c^\infty(\mathbb R)\), we have that
\[
\mathbb{E} R_N(f, \{t_n(\boldsymbol{\omega})\}) \to \int_{\mathbb R} f(x) \,\mathrm{d} x \text{ as }N\to\infty,\]
and 
\[\operatorname{Var}\bigl(R_N(f, \{t_n(\boldsymbol{\omega})\})\bigr) \ll_{f,\boldsymbol{\theta}} N^{-\varepsilon}.
\]
<!-- end theorem-5 -->
<!-- remark-2: body_md -->
Put \(N_j=\lfloor j^K\rfloor\) with \(K\varepsilon>1\). Theorem~\ref{thm:intro-smooth-moments} implies
\[
\sum_{j=1}^{\infty}\operatorname{Var}\bigl(R_{N_j}(f,\{t_n(\boldsymbol{\omega})\})\bigr)<\infty,
\]
and Chebyshev's inequality together with the first Borel--Cantelli lemma yields
\[
R_{N_j}(f,\{t_n(\boldsymbol{\omega})\})\longrightarrow \int_{\mathbb{R}}f(x)\,\mathrm{d} x
\]
with probability one. 

Since \(N_{j+1}/N_j\to1\), we may pass from this subsequence to all \(N\). To this end, we approximate the indicator function \(\mathbf{1}_{[-s,s]}\) by smooth compactly supported functions, verify the interpolation step for such smooth functions, and then pass to the limit (see \cite[Section 7]{TY23} for the standard interpolation argument). This proves Theorem~\ref{thm:intro-deterministic-PPC}.
<!-- end remark-2 -->
<!-- theorem-6: body_md -->
\label{thm:ap}
Let $\sigma, \delta$ be real algebraic numbers such that $1, \sigma, \delta$ are linearly independent over $\mathbb Q$. Then $\boldsymbol{\theta} = (\sigma, \delta) \in \mathbb{T}^2$ is of ${2}/{3}$-Diophantine type.
<!-- end theorem-6 -->
<!-- head-1: body_md -->
In particular, take
\[
\boldsymbol{\theta}=(\sigma,\delta)=(\sqrt{3},\sqrt{2}),\quad 
\alpha=\sqrt{3}-\sqrt{2},\quad 
\beta=\sqrt{3}+\sqrt{2},
\]
where all quantities are understood modulo \(1\). The conditions of Theorem~\ref{thm:intro-deterministic-PPC} are then satisfied.

<!-- lemma-1: body_md -->
\label{FTK}
For \(z\ne1\),
\begin{equation*}
\mathcal{F}_N(z)
=\frac z{1-z}-\frac{z(1-z^N)}{N(1-z)^2}.
\end{equation*}
Moreover, uniformly for \(|z|\le1\),
\begin{equation*}

|\mathcal{F}_N(z)|
\ll
\min\left\{
N,\frac1{|1-z|}
\right\},

\end{equation*}
 with the convention that $\frac{1}{|1-z|} = +\infty$ when $z = 1$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
For \(z\ne1\), a direct computation yields                                
   
                                                 
       
                                                                      
   
       
\[
\mathcal{F}_N(z)
=
\frac{(N-1)z-Nz^2+z^{N+1}}
{N(1-z)^2}
=
\frac z{1-z}
-
\frac{z(1-z^N)}
{N(1-z)^2},
\]
and
                                                    
\[
\sum_{r=1}^{N-1}(N-r)z^r
=
\sum_{k=1}^{N-1}\sum_{r=1}^{k}z^r.
\]
From this, we obtain that                       
\[\left|\mathcal{F}_N(z)\right|\le \frac{1}{N}\sum_{k=1}^{N-1}\sum_{r=1}^{k}1=\frac{N-1}{2} \quad 
\text{and} \quad 
\left|\mathcal{F}_N(z)\right|\le\frac{1}{N}\sum_{k=1}^{N-1}\frac{2}{|1-z|}=\frac{2}{|1-z|}.\]

<!-- lemma-2: body_md -->
\label{smooth}
Let \(N\in\mathbb{N}\) be sufficiently large that \(1/N+1/N^2<\eta_{\mathrm{loc}}\). Then there exists a smooth cutoff function \(g_N\in C^\infty(\mathbb{T}^2)\) such that
\[
\left|
\frac{1}{N}
\sum_{h\in\mathbb{Z}}
\widehat{f}\!\left(\frac{h}{N}\right)
\mathbf{1}_N(h\boldsymbol{\theta})
\frac{Z(h\boldsymbol{\theta})^m}{1-Z(h\boldsymbol{\theta})}
-
\frac{1}{N}
\sum_{h\in\mathbb{Z}}
\widehat{f}\!\left(\frac{h}{N}\right)
g_N(h\boldsymbol{\theta})
\frac{Z(h\boldsymbol{\theta})^m}{1-Z(h\boldsymbol{\theta})}
\right|
\ll_M
N^{(1-1/\lambda)M+1/\lambda},
\]
where the implied constant depends on the parameter \(M\) governing the rapid decay of \(\widehat{f}\).
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Let \(\psi \in C^\infty(\mathbb R)\) be a fixed function satisfying
\[
\psi(t) =
\begin{cases}
0, & t \le 0, \\
1, & t \ge 1,
\end{cases}
\quad \text{ and }
0 \le \psi(t) \le 1 \quad (t \in \mathbb R).
\]
Define
\begin{equation}\label{ap}
 g_N(\mathbf{x}) := \psi\!\left( \frac{N |1 - Z(\mathbf{x})| - 1}{1/N} \right)
= \psi\!\left( N^2 |1 - Z(\mathbf{x})| - N \right).   
\end{equation}
Then \(g_N \in C^\infty(\mathbb T^2)\) and
\[
g_N(\mathbf{x}) =
\begin{cases}
1, & |1 - Z(\mathbf{x})| \ge \dfrac{1}{N} + \dfrac{1}{N^2}, \\[6pt]
0, & |1 - Z(\mathbf{x})| \le \dfrac{1}{N} - \dfrac{1}{N^2},
\end{cases}
\]
with \(0 \le g_N \le 1\) everywhere. In the transition region
\[
T_N:=\left\{\mathbf{x}\in\mathbb{T}^{2}:\frac{1}{N}-\frac{1}{N^2}<|1-Z(\mathbf{x})|<\frac{1}{N}+\frac{1}{N^2}\right\},
\]
we have \(|g_N(\mathbf{x})-\mathbf{1}_N(\mathbf{x})|\le1\); outside this region, \(g_N(\mathbf{x})=\mathbf{1}_N(\mathbf{x})\).

Since \(1/N+1/N^2<\eta_{\mathrm{loc}}\), Remark~\ref{rem:local} implies that the difference between the sharp and smooth cutoffs is supported only in the transition region \(T_N\). Consequently,
\begin{align*}
&{}~~~~\left|
\frac{1}{N}
\sum_{h \in \mathbb Z}
\widehat f\!\left(\frac{h}{N}\right)
\mathbf{1}_N(h\boldsymbol{\theta})
\frac{Z(h\boldsymbol{\theta})^m}{1 - Z(h\boldsymbol{\theta})}
-
\frac{1}{N}
\sum_{h \in \mathbb Z}
\widehat f\!\left(\frac{h}{N}\right)
g_N(h\boldsymbol{\theta})
\frac{Z(h\boldsymbol{\theta})^m}{1 - Z(h\boldsymbol{\theta})}
\right|
\\
& = \frac{1}{N}
\left|
\sum_{h : h\boldsymbol{\theta} \in T_N}
\widehat f\!\left(\frac{h}{N}\right)
\Big(\mathbf{1}_N(h\boldsymbol{\theta})-g_N(h\boldsymbol{\theta})\Big)
\frac{Z(h\boldsymbol{\theta})^m}{1 - Z(h\boldsymbol{\theta})}
\right|
\\
&\ll_M
\frac{1}{N}
\sum_{h : |h| \gg N^{1/\lambda}}
\left( 1 + \frac{|h|}{N} \right)^{-M}
\cdot N
\ll_M
N^{(1 - 1/\lambda)M + 1/\lambda}.
\end{align*}

<!-- lemma-3: body_md -->
\label{Ab} We have 
\begin{equation*}
\lim_{N\to\infty}\int_{\mathbb{T}^{2}}\Phi_{N,m}(\mathbf{x})\,\mathrm{d} \mathbf{x}=0.
\end{equation*}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Consider the untruncated Abel function
\begin{equation*}
G_{N,m}(\mathbf{x})
=
\frac{Z(\mathbf{x})^m}{1-Z(\mathbf{x})}.
\end{equation*}

We claim that \(G_{N,m} \in L^1(\mathbb T^2)\). Indeed, away from the two exceptional points \((0,0)\) and \((1/2,1/2)\), the function \(G_{N,m}\) is bounded. Near \((0,0)\), the Taylor expansion
\[
|1 - Z(\mathbf{x})| \asymp |x| + y^2
\]
yields
\[
|G_{N,m}(\mathbf{x})| \ll \frac{1}{|x| + y^2}.
\]
                                   
The singularity \(1/(|x|+y^2)\) is locally integrable near \((0,0)\) in \(\mathbb{T}^2\). By translation symmetry, the same local integrability holds near \((1/2,1/2)\). Hence \(G_{N,m}\in L^1(\mathbb{T}^2)\).

For almost every fixed \(y\), we have \(|\cos(2\pi y)|<1\), hence \(|Z(\mathbf{x})|<1\) almost everywhere. Expanding \(1/(1-Z)\) as a geometric series gives
\[
G_{N,m}(\mathbf{x})
=
\frac{Z(\mathbf{x})^m}{1-Z(\mathbf{x})}
=
\sum_{r=0}^\infty Z(\mathbf{x})^{m+r}
=
\sum_{r=m}^\infty Z(\mathbf{x})^r.
\]
Therefore,
\[
G_{N,m}(x,y)
=
\sum_{r=m}^\infty e(rx)\cos(2\pi y)^r.
\]

For each fixed \(y\) with \(|\cos(2\pi y)|<1\), termwise integration in \(x\) gives
\[
\int_0^1 G_{N,m}(x,y)\,dx
=
\sum_{r=m}^\infty \cos(2\pi y)^r \int_0^1 e(rx)\,\mathrm{d} x
=
0,
\]
since \(\int_0^1 e(rx)\,\mathrm{d} x=0\) for \(r\ge1\). Since \(G_{N,m}\in L^1(\mathbb{T}^2)\), Fubini's theorem yileds                                                     
\[
\int_{\mathbb{T}^2} G_{N,m}(\mathbf{x})\,\mathrm{d} \mathbf{x}
=
\int_0^1 \left(\int_0^1 G_{N,m}(x,y)\,\mathrm{d} x\right)\,\mathrm{d} y
=
0.
\]
Recalling that \(\Phi_{N,m}=g_NG_{N,m}\), we deduce that
\[
\int_{\mathbb{T}^2} \Phi_{N,m}(\mathbf{x})\,\mathrm{d}{x}
=
\int_{\mathbb{T}^2} \bigl(g_N(\mathbf{x})-1\bigr)G_{N,m}(\mathbf{x})\,\mathrm{d}{x}.
\]
The integrand on the right-hand side is supported in the region
\[
\left\{\mathbf{x}\in\mathbb{T}^2:|1-Z(\mathbf{x})|\le\frac{1}{N}+\frac{1}{N^2}\right\}.
\]
By the local comparison estimate in Theorem~\ref{thm:local}, we have
\[
\left|
\int_{\mathbb{T}^2} \Phi_{N,m}(\mathbf{x})\,\mathrm{d} {x}
\right|
\ll
\iint_{|x|+y^2\le 2/N}
\frac{\,\mathrm{d} x\,\mathrm{d} y}{|x|+y^2}
\ll N^{-1/2}.
\]
Letting \(N\to\infty\) completes the proof.

<!-- lemma-4: body_md -->
\label{0}
We have
\begin{equation*}

\lim_{N\to\infty}\frac1N
\sum_{h\in\mathbb Z}
\widehat f\left(\frac hN\right)
\Phi_{N,m}(h\boldsymbol{\theta})=0.

\end{equation*}
Removing the \(h=0\) term from the sum does not affect the limit.
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
As usual, for \(\mathbf{x}=(x_1,x_2)\in\mathbb{T}^2\) and \(\mathbf{n}=(n_1,n_2)\in\mathbb{Z}^2\), set \(\mathbf{n}\cdot\mathbf{x}:=n_1x_1+n_2x_2\). Since \(\Phi_{N,m}\in C^\infty(\mathbb{T}^2)\), it has an absolutely convergent Fourier series
\[
\Phi_{N,m}(\mathbf{x})
=
\sum_{\mathbf{n}\in\mathbb{Z}^2} c_{\mathbf{n}} e(\mathbf{n}\cdot\mathbf{x}),
\qquad
\sum_{\mathbf{n}\in\mathbb{Z}^2} |c_{\mathbf{n}}|<\infty,
\]
with
\[
c_{\mathbf{0}}=\int_{\mathbb{T}^2}\Phi_{N,m}(\mathbf{x})\,\mathrm{d} \mathbf{x}.
\]
Therefore,
\begin{align*}
\frac{1}{N}
\sum_{h\in\mathbb{Z}}
\widehat{f}\!\left(\frac{h}{N}\right)
\Phi_{N,m}(h\boldsymbol{\theta})
&=
\sum_{\mathbf{n}\in\mathbb{Z}^2} c_{\mathbf{n}}
\left[
\frac{1}{N}
\sum_{h\in\mathbb{Z}}
\widehat{f}\!\left(\frac{h}{N}\right)
e\bigl(h\mathbf{n}\cdot\boldsymbol{\theta}\bigr)
\right]\\
&=
c_{\mathbf{0}}\cdot\frac{1}{N}
\sum_{h\in\mathbb{Z}}
\widehat{f}\!\left(\frac{h}{N}\right)
+
\sum_{\mathbf{n}\neq\mathbf{0}} c_{\mathbf{n}}
\left[
\frac{1}{N}
\sum_{h\in\mathbb{Z}}
\widehat{f}\!\left(\frac{h}{N}\right)
e\bigl(h\mathbf{n}\cdot\boldsymbol{\theta}\bigr)
\right].
\end{align*}

By the elementary bound \(\frac{1}{N}\sum_{h}|\widehat{f}(h/N)|\ll_f1\) and Lemma~\ref{Ab}, the first term satisfies
\[
\lim_{N\to\infty}
c_{\mathbf{0}}\cdot\frac{1}{N}
\sum_{h\in\mathbb{Z}}
\widehat{f}\!\left(\frac{h}{N}\right)
=0.
\]

Now fix \(\mathbf{n}\neq\mathbf{0}\). By Theorem~\ref{LI}, we have
\[
\omega(\mathbf{n}):=\mathbf{n}\cdot\boldsymbol{\theta}
=n_1\sigma+n_2\delta\notin\mathbb{Z}.
\]
Applying Poisson summation gives
\[
\frac{1}{N}
\sum_{h\in\mathbb{Z}}
\widehat{f}\!\left(\frac{h}{N}\right)e(h\omega(\mathbf{n}))
=
\sum_{k\in\mathbb{Z}} f\bigl(N(k+\omega(\mathbf{n}))\bigr).
\]
Since \(\omega(\mathbf{n})\notin\mathbb{Z}\), we have
\[
\operatorname{dist}(\omega(\mathbf{n}),\mathbb{Z})>0.
\]

Let \(s \in \mathbb{N}\). For \(N\) sufficiently large that \((1/N)^{1/8}<\eta_{\mathrm{loc}}\), where \(\eta_{\mathrm{loc}}>0\) is the constant from Theorem~\ref{thm:local}, we have
\[\frac{1}{N^2}
\sum_{h \in \mathbb Z \setminus \{0\}}
\left| \widehat f\!\left(\frac{h}{N}\right) \right|^s
\cdot
\frac{1}{|1 - z_h|^2}\ll N^{-\frac{1}{3}}\]

<!-- proof-10: body_md -->
For $0<\varepsilon <{1}/{18},$  we fix $\lambda= {2}/{3}+\varepsilon$ and  
\(M \ge (2\lambda(1+\varepsilon) + 1/3)/\varepsilon.\) 

As in the expectation estimate in the previous section, we split the relevant sum into two parts, one over the near-exceptional set and the other over the regular set. Here the definitions of these two sets are slightly different from those used before; the near-exceptional set and the regular set are defined respectively by
\[
\Omega_N^{(1)}
:=
\left\{
h \in \mathbb Z \setminus \{0\} : |1 - z_h| < \frac{1}{N^{1/8}}
\right\}~~\text{and } ~\Omega_N^{(2)}
:=
\left\{
h \in \mathbb Z \setminus \{0\} : |1 - z_h| \ge \frac{1}{N^{1/8}}
\right\}.
\]
We then have
\[\frac{1}{N^2}
\sum_{h \in \mathbb Z \setminus \{0\}}
\left| \widehat f\!\left(\frac{h}{N}\right) \right|^2
\cdot
\frac{1}{|1 - z_h|^2}
=
\frac{1}{N^2}
\sum_{h \in \Omega_N^{(1)}}
+
\frac{1}{N^2}
\sum_{h \in \Omega_N^{(2)}}
=:
S_{1,N}
+S_{2,N}.
\]
        
              
              
                                      
                                                     
     
                     
  
   
             
                           
                                                     
     
                     
 
             
                           
                                                     
     
                     
  
    
       
         
            
   

\paragraph{Estimate of $S_{1,N}$.}  By Theorem \ref{thm:local} and Remark \ref{rem:local},  we obtain 
\[S_{1,N}\asymp\frac{1}{N^2}
\sum_{h \in \Omega_N^{(1)}}
\left| \widehat f\!\left(\frac{h}{N}\right) \right|^s
\cdot
\frac{1}{\rho_h(\boldsymbol{\theta})^2}.\]
Splitting the sum according to \(|h| \le N^{1+\varepsilon}\) and \(|h| > N^{1+\varepsilon}\), we obtain
\begin{align*}
S_{1,N}
&\ll_{\boldsymbol{\theta}}
\frac{1}{N^2}
\sum_{\substack{h \in \Omega_N^{(1)} \\ 0 < |h| \le N^{1+\varepsilon}}}
\left|\widehat f\!\left(\frac{h}{N}\right) \right|^s
\cdot
\frac{1}{\rho_h(\boldsymbol{\theta})^2}
+
\frac{1}{N^2}
\sum_{\substack{h \in \Omega_N^{(1)} \\ |h| > N^{1+\varepsilon}}}
\left(1 + \frac{|h|}{N}\right)^{-sM}
|h|^{2\lambda}\\
&\ll_{\boldsymbol{\theta},M}
\frac{1}{N^2}
\sum_{\substack{h \in \Omega_N^{(1)} \\ 0 < |h| \le N^{1+\varepsilon}}}
\left|\widehat f\!\left(\frac{h}{N}\right) \right|^s
\cdot
\frac{1}{\rho_h(\boldsymbol{\theta})^2}+N^{-sM\varepsilon+2\lambda(1+\varepsilon)}\\
&\le\frac{1}{N^2} \sum_{j=0}^{\left\lceil (1+\varepsilon)\log N \right\rceil} \sum_{2^{j}\le |h|<2^{j+1}}\frac{\mathbf{1}_{\{h\colon \rho_h(\boldsymbol{\theta})\ll \frac{1}{N} \}}}{\rho_h(\boldsymbol{\theta})^2}+N^{-sM\varepsilon+2\lambda(1+\varepsilon)}.\end{align*}
It remains to control the contribution from the first sum.
For $k\in\mathbb{N},$ we define 
\[\mathcal B_{j,k}:=\{h\in\mathbb{Z}\colon 2^{j}\le |h|<2^{j+1}, 2^{-(k+1)}\le\rho_h(\boldsymbol{\theta})<2^{-k} \}.\]
Since $\rho_h(\boldsymbol{\theta})\gg|h|^{-\lambda}$ and  $2^{-j\lambda}\ll2^{-k}\ll (\frac{1}{N})^{1/8}$,  
\begin{align*}
\sum_{2^{j}\le |h|<2^{j+1}}\frac{\mathbf{1}_{\{h\colon \rho_h(\boldsymbol{\theta})\ll \frac{1}{N} \}}}{\rho_h(\boldsymbol{\theta})^2}\asymp\sum_{k: 2^{-j\lambda}\ll2^{-k}\ll (\frac{1}{N})^{1/8}}~\sum_{h\in \mathcal B_{j,k}}2^{2k}
=\sum_{k: 2^{-j\lambda}\ll2^{-k}\ll (\frac{1}{N})^{1/8}} \# \mathcal B_{j,k}\cdot 2^{2k}.
    \end{align*}

A key observation is the following: for  \(h_1, h_2 \in \mathcal B_{j,k}\) with \(h_1 \neq h_2\),                                                          
\[
|h_1 - h_2|^{-\lambda}
\ll \rho_{h_1 - h_2}(\boldsymbol{\theta})
\ll \rho_{h_1}(\boldsymbol{\theta}) + \rho_{h_2}(\boldsymbol{\theta})
< 2^{-(k-1)}.
\]
Indeed, the first inequality follows from the Diophantine condition, the second from the triangle inequality for \(\rho_h\), and the third from the defining property of \(\mathcal B_{j,k}\). 

Consequently,
\[
|h_1 - h_2| \gg 2^{k/\lambda},
\]
which implies that the elements of \(\mathcal B_{j,k}\) are separated by at least a distance of order \(2^{k/\lambda}\). Since \(\mathcal B_{j,k} \subset [-2^{j+1}, 2^{j+1}]\), we obtain
\[
\#  \mathcal B_{j,k}
\ll
\frac{2^{j}}{2^{k/\lambda}}+1.
\]
It follows that 
\begin{align*}
\sum_{2^{j}\le |h|<2^{j+1}}\frac{\mathbf{1}_{\{h\colon \rho_h(\boldsymbol{\theta})\ll \frac{1}{N} \}}}{\rho_h(\boldsymbol{\theta})^2}
\ll \sum_{k: 2^{-j\lambda}\ll2^{-k}\ll (\frac{1}{N})^{1/8}}  2^{2k}(\frac{2^{j}}{2^{k/\lambda}}+1)\asymp2^{2j\lambda} .
    \end{align*}

 Therefore we conclude that
\[S_{1,N}\ll_{\boldsymbol{\theta},M}\frac{1}{N^2} \sum_{j=0}^{\left\lceil (1+\varepsilon)\log N \right\rceil} 2^{2j\lambda}+ N^{-sM\varepsilon+2\lambda(1+\varepsilon)}\ll N^{-2+2\lambda(1+\varepsilon)}+N^{-sM\varepsilon+2\lambda(1+\varepsilon)}\ll N^{-\frac{1}{3}}.\]





\paragraph{Estimate of $S_{2,N}$.} We deduce that 
\begin{align*}
S_{2,N}
\le \frac{1}{N^{2}}\left(\sum_{1\le |h|<N^{1+\varepsilon}} N^{\frac{1}{4}}+\sum_{ |h|\ge N^{1+\varepsilon}} (1+\frac{|h|}{N})^{-2M}N^{\frac{1}{4}}\right)\ll N^{-\frac{1}{3}}.
\end{align*}

We treat the case \(X=z_{h_1}\); the other case is analogous.

Suppose for contradiction that one of \(\varpi\) and \(z_{h_1}\) has modulus \(>1-N^{-4r}\). Then the triangle inequality implies that both \(|\varpi|>1-2N^{-4r}\) and \(|z_{h_1}|>1-2N^{-4r}\).

\medskip
We distinguish the cases according to the value of \(\varpi\).

\medskip
\noindent\textbf{Case 1: \(\varpi=z_{h_1-h_2}\).}\quad Then
\[
|z_{h_1-h_2}-z_{h_1}|\le N^{-4r}
\quad\text{and}\quad
|z_{h_1}|,\ |z_{h_1-h_2}|\ge 1-2N^{-4r}.
\]
In this case, we may write
\[
\cos(2\pi h_1\delta)=\varepsilon_1+O(N^{-4r}),
\qquad
\cos(2\pi(h_1-h_2)\delta)=\varepsilon_2+O(N^{-4r})
\]
for some \(\varepsilon_1,\varepsilon_2=\pm1\).
Since cosine decays quadratically near its extrema,
\[
2\pi h_1\delta=k_1\pi+O(N^{-2r}),
\qquad
2\pi(h_1-h_2)\delta=k_2\pi+O(N^{-2r}),
\]
for some integers \(k_1,k_2\). Subtracting yields
\[
2\pi h_2\delta=(k_1-k_2)\pi+O(N^{-2r}),
\]
hence
\[
\cos(2\pi h_2\delta)=\varepsilon_1\varepsilon_2+O(N^{-4r}).
\]

On the other hand, from \(|z_{h_1-h_2}-z_{h_1}|\le N^{-4r}\), we obtain
\[
|e(-h_2\sigma)\cos(2\pi(h_1-h_2)\delta)-\cos(2\pi h_1\delta)|\le N^{-4r}.
\]
Substituting the cosine estimates,
\[
|e(-h_2\sigma)\varepsilon_2-\varepsilon_1|
\le N^{-4r}+O(N^{-4r})
\ll N^{-4r}.
\]
Thus
\[
e(-h_2\sigma)=\varepsilon_1\varepsilon_2+O(N^{-4r}).
\]
Combining this with the estimate for \(\cos(2\pi h_2\delta)\),
\[
z_{-h_2}
=
e(-h_2\sigma)\cos(2\pi h_2\delta)
=
\bigl(\varepsilon_1\varepsilon_2+O(N^{-4r})\bigr)
\bigl(\varepsilon_1\varepsilon_2+O(N^{-4r})\bigr)
=
1+O(N^{-4r}).
\]
When \(N\) is sufficiently large, this contradicts \(|1-z_{-h_2}|\ge N^{-r}\).

\medskip
\noindent\textbf{Case 2: \(\varpi  = z_{h_1} z_{-h_2}\).} Then
        
                     
  
                                                 
      
                                                  
  
                                                                                                                      
   
\[
|z_{h_1}z_{-h_2}-z_{h_1}|\le N^{-4r},
\]
and since \(|z_{h_1}|\le1\),
\[
|1-z_{-h_2}|\le N^{-4r},
\]
which contradicts \(|1-z_{-h_2}|>N^{-r}\) for \(N\) sufficiently large.

<!-- lemma-12: body_md -->
\label{HX}
Suppose that, for $i=1,2,\ldots, s$,
\[
\quad |X_i|\le 1-N^{-4r}.
\]
Let $\{m_i: 1\le i\le  s\}\subset \mathbb{N}$ and
\(H(t)=\sum_{n= 0}^{\infty}h_n t^n\). We have 
\[
\left|
[t^N]H(t)\prod_{i=1}^{s}(1-X_it)^{-m_i}
\right|
\ll
 N^{4r(m_1+\cdots+m_s)}
\sup_{0\le n\le N}|h_n|.
\]
<!-- end lemma-12 -->
<!-- proof-12: body_md -->
Expand each factor \((1-X_it)^{-m_i}\) into its negative-binomial series:
\[
(1-X_it)^{-m_i}
=\sum_{n_i= 0}^{\infty}\binom{n_i+m_i-1}{m_i-1}X_i^{n_i}t^{n_i}.
\]
Thus, comparing coefficients of \(t^N\), we obtain
\[
[t^N]H(t)\prod_{i=1}^{s}(1-X_it)^{-m_i}
=
\sum_{\substack{n_1,\dots,n_s\ge 0\\ n_1+\cdots+n_s\le N}}
h_{N-n_1-\cdots-n_s}
\prod_{i=1}^{s}\binom{n_i+m_i-1}{m_i-1}X_i^{n_i}.
\]
Taking absolute values and using \(|h_{N-n_1-\cdots-n_s}|\le \sup_{0\le n\le N}|h_n|\), we get
\[
\left|
[t^N]H(t)\prod_{i=1}^{s}(1-X_it)^{-m_i}
\right|
\le
\sup_{0\le n\le N}|h_n|
\sum_{\substack{n_1,\dots,n_s\ge 0\\ n_1+\cdots+n_s\le N}}
\prod_{i=1}^{s}\binom{n_i+m_i-1}{m_i-1}|X_i|^{n_i}.
\]
Enlarging the constraint \(n_1+\cdots+n_s\le N\) to the product ranges \(0\le n_i\le N\) for each \(i\), we obtain
\[
\left|
[t^N]H(t)\prod_{i=1}^{s}(1-X_it)^{-m_i}
\right|
\le
\sup_{0\le n\le N}|h_n|
\prod_{i=1}^{s}
\sum_{n_i=0}^{N}\binom{n_i+m_i-1}{m_i-1}|X_i|^{n_i}.
\]
Since for each fixed \(i\),
\[
\sum_{n_i=0}^{N}\binom{n_i+m_i-1}{m_i-1}|X_i|^{n_i}\le (1-|X_i|)^{-m_i}
\le N^{4rm_i},
\]
we conclude that
\[
\left|
[t^N]H(t)\prod_{i=1}^{s}(1-X_it)^{-m_i}
\right|
\le
N^{4r(m_1+\cdots+m_s)}
\sup_{0\le n\le N}|h_n|,
\]
as required.

\label{df}
Let \(A_0 \in \overline{\mathbb D}\) and \(0 < \eta \le \eta_{\mathrm{loc}}\), where \(\eta_{\mathrm{loc}} > 0\) is the sufficiently small absolute constant from Theorem~\ref{thm:local}. Assume that \(q_1 \neq q_2 \in \mathbb Z \setminus \{0\}\) satisfy
\[
|1 - A_0 z_{-q_i}| \le \eta, \quad i = 1, 2.
\]

\[
\rho_{q_1 - q_2}(\boldsymbol{\theta}) \ll \eta.

The implicit constant is uniform in \(A_0\).

By assumption,
\[
|1 - A_0 z_{-q_i}| \le \eta,
\]
so \(|A_0| \ge 1 - \eta\). Since \(\eta \le \eta_{\mathrm{loc}} < 1\), we have \(A_0 \neq 0\), and we may write
\[
A_0 = r e(\phi), \quad 1 - \eta \le r \le 1, \quad \phi \in \mathbb T.
\]

Because \(r \le 1\), the inequality \(|1 - A_0 z_{-q_i}| \le \eta\) implies
\(
|\cos(2\pi q_i\delta)| \ge 1 - \eta.
\)
Since \(|\cos(2\pi x)|\) attains its maximum \(1\) only at \(x \in \{0, 1/2\} \pmod 1\), with quadratic decay near these points, there exist for each \(i\) some \(\varepsilon_i \in \{0, 1\}\) and \(v_i \in \mathbb R\) such that
\[
q_i\delta = \frac{\varepsilon_i}{2} + v_i \pmod 1, \quad |v_i| \le C_1\eta^{1/2},
\]
for an absolute constant \(C_1 > 0\). Hence
\[
\cos(2\pi q_i\delta) = e\!\left(\frac{\varepsilon_i}{2}\right)\cos(2\pi v_i),
\]
and therefore
\[
A_0 z_{-q_i} = r\cos(2\pi v_i) \, e\!\left(\phi - q_i\sigma + \frac{\varepsilon_i}{2}\right).
\]
Since \(r\cos(2\pi v_i) = 1 + O(\eta)\), the inequality \(|1 - A_0 z_{-q_i}| \le \eta\) yields
\[
\left|1 - \widetilde r_i \, e(t_i)\right| \le C_2\eta,
\]
where
\[
\widetilde r_i := r\cos(2\pi v_i), \quad t_i := \phi - q_i\sigma + \frac{\varepsilon_i}{2},
\]
and \(1 - C_2\eta \le \widetilde r_i \le 1\).

We now derive the bound on \(\|t_i\|_{\mathbb T}\) directly. The identity
\[
|1 - \widetilde r_i e(t_i)|^2
=
(1 - \widetilde r_i)^2 + 2\widetilde r_i \bigl(1 - \cos(2\pi t_i)\bigr)
\]
together with \(\widetilde r_i \ge 1/2\) (which holds after decreasing \(\eta_{\mathrm{loc}}\) if necessary) and \(|1 - \widetilde r_i e(t_i)| \le C_2\eta\) gives
\[
1 - \cos(2\pi t_i) \ll_{C_2} \eta^2.
\]
Since \(1 - \cos(2\pi t_i) \asymp \|t_i\|\) uniformly on \(\mathbb T\), we conclude
\[
\|t_i\|\le C_3\eta
\]
for some absolute constant \(C_3 > 0\). Hence there exists \(u_i \in \mathbb R\) such that
\[
q_i\sigma = \phi + \frac{\varepsilon_i}{2} + u_i \pmod 1, \quad |u_i| \le C_3\eta.
\]

Combining the two coordinates,
\[
q_i\boldsymbol{\theta} = \kappa_{\varepsilon_i}(\phi) + (u_i, v_i), \quad i = 1, 2,
\]
where the center \(\kappa_{\varepsilon_i}(\phi)\) is defined by
\[
\kappa_{\varepsilon_i}(\phi)
:=
\left(\phi + \frac{\varepsilon_i}{2}, \frac{\varepsilon_i}{2}\right) \pmod 1.
\]
Thus \(\kappa_0(\phi) = (\phi, 0)\) and \(\kappa_1(\phi) = (\phi + 1/2, 1/2)\), which are precisely the translates of the two exceptional points in
\[
\mathcal K = \left\{(0,0), \left(\tfrac{1}{2}, \tfrac{1}{2}\right)\right\}.
\]

We now compare the two centers. If \(\varepsilon_1 = \varepsilon_2\), then \(\kappa_{\varepsilon_1}(\phi) - \kappa_{\varepsilon_2}(\phi) = (0,0)\). If \(\varepsilon_1 \neq \varepsilon_2\), then
\[
\kappa_{\varepsilon_1}(\phi) - \kappa_{\varepsilon_2}(\phi) = \left(\frac{1}{2}, \frac{1}{2}\right) \pmod 1.
\]
Thus in all cases, \(\kappa_{\varepsilon_1}(\phi) - \kappa_{\varepsilon_2}(\phi) \in \mathcal K\). Since \(\mathcal K\) is a subgroup of \(\mathbb T^2\) and every element of \(\mathcal K\) has zero difference frequency, the two centers contribute nothing to \(\rho_{q_1 - q_2}\). Consequently,
\[
\rho_{q_1 - q_2}(\boldsymbol{\theta})
\ll
\operatorname{dist}_{\mathcal K}\bigl((u_1 - u_2, v_1 - v_2)\bigr)
\ll
|u_1 - u_2| + (v_1 - v_2)^2.
\]
Using \(|u_i| \le C_3\eta\) and \(|v_i| \le C_1\eta^{1/2}\), we obtain
\[
|u_1 - u_2| \le 2C_3\eta \ll \eta,
\quad 
(v_1 - v_2)^2 \le (2C_1\eta^{1/2})^2 = 4C_1^2\eta \ll \eta.
\]
Therefore,
\[
\rho_{q_1 - q_2}(\boldsymbol{\theta}) \ll \eta.
\]
This completes the proof.

We now apply Schmidt's Subspace Theorem \cite{S89} to prove the required lower bound. Recall that the theorem states the following: if $L_0, \dots, L_{n-1}$ are linearly independent linear forms in $n$ variables with algebraic coefficients, then for any $\varepsilon > 0$, the integer solutions $\mathbf{x} \in \mathbb Z^n \setminus \{0\}$ satisfying
\[
|L_0(\mathbf{x}) \cdots L_{n-1}(\mathbf{x})| < |\mathbf{x}|^{-\varepsilon}
\]
lie in a finite union of proper rational subspaces of $\mathbb R^n$.




For \(i=1,2\), choose
\(p_i(q)\in\mathbb Z\) such that
\[
|q\alpha_i-p_i(q)|
=
\|q\alpha_i\|.
\]
Define the linearly independent algebraic linear forms
\[
L_0(x_0, x_1, x_2) = x_0,~
L_1(x_0, x_1, x_2) = \alpha_1 x_0 - x_1, ~ 
L_2(x_0, x_1, x_2) = \alpha_2 x_0 - x_2,
\]
and set
\[
\mathbf{x}_q := (q, p_1(q), p_2(q)) \in \mathbb Z^3.
\]
A direct computation gives
\begin{equation*}
\prod_{i=0}^{2} |L_i(\mathbf{x}_q)| = q \, \|q\alpha_1\| \, \|q\alpha_2\|.
\end{equation*}
Moreover, we claim that
\[
H(\mathbf{x}_q) := \max\{1, q, |p_1(q)|, |p_2(q)|\} \asymp_{\alpha_1, \alpha_2} q.
\]
To prove the claim, we assume for contradiction that for infinitely many $q$,
\[
\prod_{i=0}^{2}|L_i(\mathbf x_q)| < H(\mathbf{x}_q)^{-\varepsilon}
\]
for some $\varepsilon > 0$. Then the Schmidt's Subspace Theorem ($n=3$)
implies that all such vectors $\mathbf{x}_q$ lie in a finite union of proper rational subspaces of $\mathbb R^3$. Hence at least one proper rational subspace contains infinitely many of the $\mathbf{x}_q$.
We   show that this is impossible. 

Suppose there exist integers $A, B, C$, not all zero, such that
\[
A q + B p_1(q) + C p_2(q) = 0
\]
for infinitely many $q$. Dividing by $q$ and taking the limit along this infinite subsequence, we obtain
\[
A + B \lim_{q  } \frac{p_1(q)}{q} + C \lim_{q } \frac{p_2(q)}{q} = 0.
\]
Since $p_i(q)/q \to \alpha_i$, this gives
\[
A + B\alpha_1 + C\alpha_2 = 0,
\]
contradicting the assumed linear independence of $1, \alpha_1, \alpha_2$ over $\mathbb Q$.                                                                                           



Consequently, for all sufficiently large $q$,
\begin{equation*}
q \, \|q\alpha_1\| \|q\alpha_2\| \gg_{\alpha_1, \alpha_2, \varepsilon} q^{-\varepsilon}.
\end{equation*}
The finitely many remaining small $q$ are handled by adjusting the implicit constant, and the case of negative $q$ follows from the symmetry $\| - q\alpha_i\|= \|q\alpha_i\|$.

For $\varepsilon_0 \in \mathbb{Q}^{+}$ (the set of positive rational numbers), define the auxiliary set
\[
D_\lambda^*(\varepsilon_0) := \left\{ \boldsymbol{\theta} \in \mathbb{T}^2 : \rho_h(\boldsymbol{\theta}) \ge |h|^{-(\lambda+\varepsilon_0)} \text{ for all } h \in \mathbb{Z}\setminus\{0\} \right\}.

\]
Hence it suffices to show that $\bigcap_{\varepsilon_0\in\mathbb{Q}^{+}} D_\lambda^*(\varepsilon_0)$ has full Lebesgue measure, or equivalently, that the complement of $D_\lambda^*(\varepsilon_0)$ has measure zero for each $\varepsilon_0 \in\mathbb{Q}^{+}$.

\begin{align*}
D_\lambda^*(\varepsilon_0)^c
&= \left\{ \boldsymbol{\theta} \in \mathbb{T}^2 : \rho_h(\boldsymbol{\theta}) < |h|^{-(\lambda+\varepsilon_0)} \text{ for infinitely many } h \in \mathbb{Z}\setminus\{0\} \right\} \\
&= \bigcap_{H=1}^{\infty} \bigcup_{\substack{h \in \mathbb{Z} \\ |h| \ge H}} \mathcal{E}_h,
\end{align*}
where
\[
\mathcal{E}_h := \left\{ \boldsymbol{\theta} \in \mathbb{T}^2 : \rho_h(\boldsymbol{\theta}) < |h|^{-(\lambda+\varepsilon_0)} \right\}.
\]

\[
\mathcal{E}_h \subset
\left\{ \boldsymbol{\theta} \in \mathbb{T}^2 : \|h\sigma\| + \|h\delta\|^2 \ll |h|^{-(\lambda+\varepsilon_0)} \right\}
\cup
\left\{ \boldsymbol{\theta} \in \mathbb{T}^2 : \left\|h\sigma-\frac12\right\| + \left\|h\delta-\frac12\right\|^2 \ll |h|^{-(\lambda+\varepsilon_0)} \right\}.
\]
Consider the first set in the above union. On the torus, the conditions $\|h\sigma\| < r$ and $\|h\delta\| < r^{1/2}$ define regions of measure $\asymp r$ and $\asymp r^{1/2}$, respectively. Hence its measure is

\operatorname{Leb}(\mathcal{E}_h) \ll |h|^{-3(\lambda+\varepsilon_0)/2}.

\operatorname{Leb}\left( D_\lambda^*(\varepsilon_0)^c \right) = 0.
