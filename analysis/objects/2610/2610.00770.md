<!-- definition-1: body_md -->
We define $\mathcal{X}^*$ as the adjoint of $\mathcal{X}$ in $L^2_+(\mathbb{R})$. 
\label{def:Xstar}
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
Let $u_0\in H^2(\mathbb{R})$ be given real-valued initial data, and let $u\in C(\mathbb{R},H^2(\mathbb{R}))$ be the solution to \eqref{eq:BO} with initial condition \eqref{eq:IC}. Then for every $t\in\mathbb{R}$, the analytic continuation of $x\mapsto \Pi u(t,x)$ as a function of $x\in\mathbb{R}$ to $z=x+\mathrm{i} y\in \mathbb{C}_+$ satisfies 
\begin{equation}
    \Pi u(t,z)=\frac{1}{2\pi\mathrm{i}}I_+\left[(\mathcal{X}^*-2t\mathcal{L}_{u_0}-z\,\mathrm{Id})^{-1}\Pi u_0\right],\quad z\in\mathbb{C}_+.
\label{eq:explicitformula}
\end{equation}
\label{thm:explicitformula}
<!-- end theorem-1 -->
<!-- remark-1: body_md -->
It is instructive to consider how the explicit formula \eqref{eq:explicitformula} returns the initial data $u(0,x)=u_0(x)$.  Setting $t=0$ in \eqref{eq:explicitformula} yields
\begin{equation}
    \Pi u(0,z)=\frac{1}{2\pi\mathrm{i}}I_+\left[(\mathcal{X}^*-z\,\mathrm{Id})^{-1}\Pi u_0\right],
    \label{eq:Piu(0)}
\end{equation}
or in other words, using $y\in\mathbb{R}$ as the independent variable of functions $f\in L^2_+(\mathbb{R})$,
\begin{equation}
    \Pi u(0,z)=\frac{1}{2\pi\mathrm{i}}I_+[f],\quad \mathcal{X}^*f(y)-zf(y)=\Pi u_0(y),\quad z\in\mathbb{C}_+.
\end{equation}
Using \eqref{eq:Xstar-action-x} to express the 
action of $\mathcal{X}^*$ we get the equation $(y-z)f(y)-\mathfrak{a}[f]=\Pi u_0(y)$, where $\mathfrak{a}[f]=-I_+[f]/(2\pi\mathrm{i})$ is a constant depending linearly on $f$. 
Solving explicitly for $f(y)$, we obtain
\begin{equation}
    f(y)=\frac{\Pi u_0(y)+\mathfrak{a}[f]}{y-z}.
    \label{eq:f-t-zero}
\end{equation}
It remains to determine $\mathfrak{a}[f]$.  Since the parameter $z$ is in the upper half-plane $\mathbb{C}_+$ and $f\in L^2_+(\mathbb{R})$ must extend to an analytic function of its argument $y$ in the same half-plane, where $\Pi u_0$ is also analytic, it is necessary that the constant $\mathfrak{a}[f]$ takes the $z$-dependent value $\mathfrak{a}[f]=-\Pi u_0(z)$ to cancel the root of the denominator in \eqref{eq:f-t-zero}.  Thus 
\begin{equation}
\Pi u(0,z)=\frac{1}{2\pi\mathrm{i}}I_+[f]=-\mathfrak{a}[f]=\Pi u_0(z)    
\end{equation}
holds for all $z\in\mathbb{C}_+$, which then implies that $u(0,\diamond)=u_0(\diamond)$ in $L^2(\mathbb{R})$. 

An interesting observation is that upon replacing $\Pi u(0,z)$ on the left-hand side of \eqref{eq:Piu(0)} with the equivalent expression $g(z):=\Pi u_0(z)$ and likewise replacing $\Pi u_0$ with $g\in L_+^2(\mathbb{R})$ on the right-hand side yields the identity
\begin{equation}
    g(z)=\frac{1}{2\pi\mathrm{i}}I_+\left[(\mathcal{X}^*-z\mathrm{Id})^{-1}g\right],\quad g\in L_+^2(\mathbb{R}), \quad z\in\mathbb{C}_+.
\label{eq:Cauchy-generalization}
\end{equation}
In fact, this is equivalent to the familiar Cauchy integral formula
\begin{equation}
    g(z)=\frac{1}{2\pi\mathrm{i}}\int_\mathbb{R}\frac{g(x)\,\mathrm{d} x}{x-z},\quad g\in L^2_+(\mathbb{R}),\quad z\in\mathbb{C}_+.
    \label{eq:Cauchy}
\end{equation}
Indeed, we may give a direct proof that the right-hand sides of \eqref{eq:Cauchy-generalization} and of \eqref{eq:Cauchy} agree.
By the definition of $I_+$, the right-hand side of \eqref{eq:Cauchy-generalization} reads
\begin{equation}
    \frac{1}{2\pi\mathrm{i}}\lim_{\xi\downarrow 0}\hat{f}(\xi),\quad \text{where}\quad\mathcal{X}^*f-zf=g.
\end{equation}
Taking the Fourier transform of $\mathcal{X}^*f-zf=g$ using \eqref{eq:X-star-hat} gives the first-order differential equation
\begin{equation}
    \mathrm{i}\frac{\mathrm{d} \widehat{f}}{\mathrm{d}\xi}(\xi)-z\widehat{f}(\xi)=\widehat{g}(\xi),\quad\xi>0.
\end{equation}
Using the integrating factor $\mathrm{e}^{\mathrm{i} z\xi}$, which is in $L^2(0,+\infty)$ as a function of $\xi$ because $z\in\mathbb{C}_+$, we obtain
\begin{equation}
    \mathrm{i}\frac{\mathrm{d}}{\mathrm{d}\xi}\left(\mathrm{e}^{\mathrm{i} z\xi}\widehat{f}(\xi)\right)=\mathrm{e}^{\mathrm{i} z\xi}\widehat{g}(\xi),\quad \xi>0.
\end{equation}
For $z\in\mathbb{C}_+$ and $\widehat{g}\in L^2(0,+\infty)$, the right-hand side is in $L^1(0,+\infty)$ as a function of $\xi$, so we have
\begin{equation}
    \widehat{f}(\xi)=\mathrm{i}\mathrm{e}^{-\mathrm{i} z\xi}\int_{\xi}^{+\infty}\mathrm{e}^{\mathrm{i} z\zeta}\widehat{g}(\zeta)\,\mathrm{d}\zeta,\quad \xi>0.
\end{equation}
The right-hand side of \eqref{eq:Cauchy-generalization} therefore reads
\begin{equation}
    \frac{1}{2\pi\mathrm{i}}\lim_{\xi\downarrow 0}\widehat{f}(\xi)=\frac{1}{2\pi}\int_0^{+\infty}\mathrm{e}^{\mathrm{i} z\zeta}\widehat{g}(\zeta)\,\mathrm{d}\zeta = \frac{1}{2\pi} \langle \widehat{g},\mathrm{e}^{-\mathrm{i} z^*\diamond}\rangle.
\end{equation}
But $\mathrm{e}^{-\mathrm{i} z^*\xi}1_{\ge 0}(\xi)$ is the Fourier transform of $x\mapsto -[2\pi\mathrm{i}(x-z^*)]^{-1}\in L_+^2(\mathbb{R})$, so coming back from the Fourier side using \eqref{eq:inner-product-convention},
\begin{equation}
    \frac{1}{2\pi\mathrm{i}}\lim_{\xi\downarrow 0}\widehat{f}(\xi)=\left\langle g,-\frac{1}{2\pi\mathrm{i} (\diamond-z^*)}\right\rangle=\frac{1}{2\pi\mathrm{i}}\int_\mathbb{R}\frac{g(x)\,\mathrm{d} x}{x-z},
\end{equation}
proving directly that \eqref{eq:Cauchy-generalization} is the same as \eqref{eq:Cauchy}.

<!-- theorem-2: body_md -->
\label{thm:inversion-formula-rational}
Fix $\varepsilon>0$ and let $u_0$ be of the form \eqref{eq:PN-rational-IC}.  Then for every $x\in\mathbb{R}$ and $t>0$, the solution $u$ of the Cauchy problem \eqref{eq:BO}--\eqref{eq:IC} satisfies $u(t,x)=2\mathrm{Re}(\Pi u(t,x))$ with
\begin{equation}
    \Pi u(t,x)=\frac{\det(\mathbf{A}(t,x))}{\det(\mathbf{B}(t,x))}.
    \label{eq:Pi-u-fraction}
\end{equation}
An equivalent formula is
\begin{equation}
\Pi u(t,x)=\mathrm{i}\varepsilon\frac{\partial}{\partial x}\log\left(\det(\mathbf{B}(t,x))\prod_{n=1}^N\mathrm{e}^{\mathrm{i} (p_n-x)^2/(4\varepsilon t)}\right).
    \label{eq:Pi-u-derivative}
\end{equation}
To solve the Cauchy problem for $t<0$, we may use the symmetry $u(-t,x)=u(t,-x)$, for which we have the initial data $u_0(-x)$ that is also of the form \eqref{eq:PN-rational-IC} with pole and residue data explicitly related to that of $u_0(x)$.
<!-- end theorem-2 -->
<!-- remark-2: body_md -->
\label{rem:continuity-at-t=0+}
The exponential factor $\mathrm{e}^{-\mathrm{i} h(z)/\varepsilon}$ appearing in the integrals defining the elements of the matrices $\mathbf{A}(t,x)$ (see \eqref{eq:Amatrix}) and $\mathbf{B}(t,x)$ (see \eqref{eq:Bmatrix}) is undefined at $t=0$, according to \eqref{eq:PN-exph}.  Of course by arguments independent of the rational assumption we have already shown how the explicit formula \eqref{eq:explicitformula} returns $\Pi u(0,z)=\Pi u_0(z)$ for all $z\in \mathbb{C}_+$ (and hence also in the limit $z\to x$ for almost every $x\in\mathbb{R}$) when $t=0$, but it is useful to see how the determinantal formula \eqref{eq:Pi-u-fraction} valid for rational data $u_0$ behaves in the limit $t\to 0$ to verify directly the continuity of $t\mapsto u(t,x)$ for each fixed $x\in\mathbb{R}$ at $t=0$.  

We assume that $t>0$ and compute the limit of the right-hand side of \eqref{eq:Pi-u-fraction} as $t\downarrow 0$ for fixed $x\in\mathbb{R}$ and $\varepsilon>0$.  
For this limit, it is best to isolate the factor $\mathrm{e}^{-\mathrm{i} (z-x)^2/(4\varepsilon t)}$ in $\mathrm{e}^{-\mathrm{i} h(z)/\varepsilon}$, since it is responsible for the concentration in the integrand as $t\downarrow 0$.  
The only critical point of the exponent function $-\mathrm{i} (z-x)^2/(4\varepsilon t)$ is $z=x\in\mathbb{R}$, and it is simple.  To analyze the ratio of determinants of the matrices $\mathbf{A}(t,x)$ and $\mathbf{B}(t,x)$ in the limit $t\downarrow 0$, it is first useful to apply some row and column operations to the matrices that leave the determinants unchanged.  We first apply row operations on rows $j=2,\dots,N+1$ to replace the integration contour $C_{j-1}$ whenever $j-1$ is a non-exceptional index with a contour $\widetilde{C}_{j-1}$ that encircles only one pole $p_{j-1}$ in the counterclockwise sense and tends to $z=\infty\mathrm{e}^{3\pi\mathrm{i}/4}$ in both directions.  If $j-1$ is exceptional, then we just take $\widetilde{C}_{j-1}=C_{j-1}$.  Then, in the matrix $\mathbf{A}(t,x)$, we use column operations to replace the function $u_0(z)$ in the integrand of the first column with $\Pi u_0(z)$ (since $u_0(z)-\Pi u_0(z)$ is a linear combination of $(z-p_n)^{-1}$ for $n=1,\dots,N$).  Thus we obtain two modified matrices $\widetilde{\mathbf{A}}(t,x)$ and $\widetilde{\mathbf{B}}(t,x)$ with the same determinants as $\mathbf{A}(t,x)$ and $\mathbf{B}(t,x)$ respectively.

Then, using the method of steepest descent, it is not difficult to see that each of the $N+2$ distinct integrals appearing in any given row of both matrices $\widetilde{\mathbf{A}}(t,x)$ and $\widetilde{\mathbf{B}}(t,x)$, all of which correspond to the same contour of integration, will be dominated in the limit $t\downarrow 0$ by contributions from the neighborhood of exactly one point.  For the first row, corresponding to the contour $C_0$, this is the simple critical point $z=x$.  For row $j$, $j=2,\dots,N+1$, corresponding to the contour $\widetilde{C}_{j-1}$, this is the point $z=p_{j-1}$.  Although the details of the form of the leading term are different depending on whether the dominating point is a critical point or a pole $z=p_{j-1}$, and in the latter case whether the integrand is integrable at the pole or not, these details produce nonzero factors that are common to every element of the same row of both $\widetilde{\mathbf{A}}(t,x)$ and $\widetilde{\mathbf{B}}(t,x)$.  Hence they will cancel from the ratio of determinants in \eqref{eq:Pi-u-fraction} for $\Pi u(t,x)$.  The relative sizes of the renormalized integrals in any given row are then determined by the order of the integrand at the dominating point.  Thus, for some nonzero common factor $\alpha_1(t)\neq 0$, the first rows of $\widetilde{\mathbf{A}}(t,x)$ and of $\widetilde{\mathbf{B}}(t,x)$ can be written as
\begin{equation}
\begin{split}
\left(\widetilde{A}_{11},\dots,\widetilde{A}_{1,N+1}\right) &= \alpha_1(t)\left[\left(\Pi u_0(x),\frac{1}{x-p_1},\dots,\frac{1}{x-p_N}\right)+o(1)\right]\\
\left(\widetilde{B}_{11},\dots,\widetilde{B}_{1,N+1}\right)&=\alpha_1(t)\left[\left(1,\frac{1}{x-p_1},\dots,\frac{1}{x-p_N}\right)+o(1)\right]
\end{split}
\end{equation}
in the limit $t\downarrow 0$.
For row $j$, $j=2,\dots,N+1$, because $u_0(z)$ has been replaced by $\Pi u_0(z)$ which is analytic in the upper half-plane, the most singular element appears in column $j$ and therefore there exists some nonzero factor $\alpha_j(t)\neq 0$ such that
\begin{equation}
    \begin{split}
        \left(\widetilde{A}_{j1},\dots,\widetilde{A}_{j,N+1}\right)&=\alpha_j(t)\left[\left(0,\dots,0,1,0,\dots,0\right)+o(1)\right]\\
        \left(\widetilde{B}_{j1},\dots,\widetilde{B}_{j,N+1}\right)&=\alpha_j(t)\left[\left(0,\dots,0,1,0,\dots,0\right)+o(1)\right]
    \end{split}
\end{equation}
as $t\downarrow 0$, where the $1$ appears in column $j$ on the right-hand side in both cases.  Note that the $o(1)$ terms in the first column here are different expressions for $\widetilde{\mathbf{A}}(t,x)$ and for $\widetilde{\mathbf{B}}(t,x)$, but they are both relatively small compared to the identical diagonal elements.
Therefore, defining a $N$-component row vector by $\mathbf{v}^\top:=((x-p_1)^{-1},\dots,(x-p_N)^{-1})$, and
letting $\mathbf{0}$ denote the $N$-component column vector of zeros, in the limit $t\downarrow 0$,
\begin{equation}
u(t,x)= 2\mathrm{Re}\left(\frac{\det(\widetilde{\mathbf{A}}(t,x))}{\det(\widetilde{\mathbf{B}}(t,x))}\right) = 2\mathrm{Re}\left(\frac{\displaystyle \begin{vmatrix} \Pi u_0(x) & \mathbf{v}^\top\\
\mathbf{0} & \mathbb{I}\end{vmatrix}+o(1)}{\displaystyle\begin{vmatrix} 1 & \mathbf{v}^\top\\\mathbf{0} & \mathbb{I}\end{vmatrix}+o(1)}\right) = 2\mathrm{Re}\left(\Pi u_0(x)\right) + o(1).
\end{equation}
This verifies directly that the solution formula of Theorem~\ref{thm:inversion-formula-rational} is continuous from the right at $t=0$, for each $x\in\mathbb{R}$ and $\varepsilon>0$.

<!-- remark-3: body_md -->
When $\epsilon>0$ and $t>0$, all of the contour integrals in the elements of $\mathbf{A}(t,x)$ and $\mathbf{B}(t,x)$ are absolutely convergent for all $x\in\mathbb{C}$ and hence $\det(\mathbf{A}(t,x))$ and $\det(\mathbf{B}(t,x))$ are both entire functions of $x\in\mathbb{C}$. 
In particular, $x\mapsto \Pi u(t,x)$ is meromorphic for every $t>0$, and according to \eqref{eq:Pi-u-derivative} every pole (necessarily in the lower half-plane $\mathbb{C}_-$) is simple with residue that is a positive integer multiple of $\mathrm{i}\varepsilon$.  Since for $x\in\mathbb{R}$ we have $u(t,x)=\Pi u(t,x)+\Pi u(t,x)^*=\Pi u(t,x)+\Pi u(t,x^*)^*$, it follows by Schwarz reflection that $x\mapsto u(t,x)$ is a Schwarz-symmetric meromorphic function for all $t>0$ with simple poles in complex-conjugate pairs, all having residues that are positive integer multiples of $\mathrm{i}\varepsilon$, and having no finite accumulation point.  Naturally the same is true for $t=0$ except that the meromorphic function degenerates to the rational one $u_0$.  However for $t>0$, $x\mapsto u(t,x)$ is generally no longer rational, unless $\mathrm{i} c_n/\varepsilon$ is a positive integer $m_n>0$ for every $n=1,\dots,N$ (in which case $u(t,x)$ is an exact $M$-soliton solution of \eqref{eq:BO} where $M=m_1+\cdots + m_N$ as discussed in Section~\ref{sec:examples}).  
In the general case that $p_n\in\mathbb{C}_+$ is a pole of $u_0$ with a non-quantized residue $c_n\not\in -\mathrm{i}\varepsilon\mathbb{Z}_{>0}$, the behavior of $u(t,x)$ for $x\approx p_n$ is necessarily very complicated in the limit $t\downarrow 0$.  Note that the analysis carried out in Remark~\ref{rem:continuity-at-t=0+} is not valid for $x$ in the complex plane near a pole in $\mathbb{C}_-$ of $x\mapsto\Pi u(t,x)$.

f_k(z):=\frac{R_k(p_k)}{R_k(z)},
\quad z\in\mathbb{C}_+.
\label{eq:fk-define}
\end{equation}
Conversely, $F$ is spanned by eigenfunctions $\varphi$ of $\mathcal{L}_{u_0}$.

<!-- proof-1: body_md -->
Let $\varphi\in L_+^2(\mathbb{R})$ be an eigenfunction of $\mathcal{L}_{u_0}$ associated to the eigenvalue $\lambda<0$. Then using the identity 
\begin{equation}
\Pi \left(\frac{\varphi}{\diamond-p}\right)(z)=\frac{\varphi(z)-\varphi(p)}{z-p},\quad p\in \mathbb{C}_+,    
\end{equation}
in $\mathcal{L}_{u_0}\varphi=\lambda\varphi$ leads to
\begin{equation}
\varepsilon \varphi'(z)-\mathrm{i}(\lambda+u_0)\varphi(z)=-\varepsilon\sum_{n=1}^N\frac{\mathrm{i} v_n}{z-p_n},
\quad v_n:=\varphi(p_n).
\end{equation}
Then using the integrating factor $\mathrm{e}^{-\mathrm{i}\lambda z/\varepsilon}R(z)$ gives
\begin{equation}\label{eq:phi-multisol}
(\mathrm{e}^{-\mathrm{i}\lambda z/\varepsilon}R(z)\varphi(z))'=-\mathrm{e}^{-\mathrm{i}\lambda z/\varepsilon}R(z)\sum_{n=1}^N\frac{v_n}{z-p_n}.
\end{equation}
Since the left-hand side is the derivative of a function meromorphic in $\mathbb{C}_+$, the residues of the right-hand side at all poles $p_k$, $k=1,\dots,N$, must vanish.  
To calculate those residues, first use \eqref{eq:Rk}, 
to obtain the partial fraction expansion
\begin{equation}
R(z)\sum_{n=1}^N\frac{v_n}{z-p_n}
	=\sum_{n=1}^N\frac{\alpha_n}{(z-p_n)^2}+\frac{\beta_n}{(z-p_n)},
    \label{eq:R-sum-partial-fractions}
\end{equation}
where 
\begin{equation}
\alpha_n
	:=R_n(p_n)v_n,
\quad
\beta_n
	:=R_n'(p_n)v_n+R_n(p_n)\mathop{\sum_{m=1}^N}_{m\neq n}\frac{v_m}{p_n-p_m}.
\label{eq:alphanbetan}
\end{equation}
Then, we must impose the conditions
\begin{equation}
0
	=\mathop{\mathrm{Res}}_{z=p_k}\left[-\mathrm{e}^{-\mathrm{i}\lambda z/\varepsilon}R(z)\sum_{n=1}^N\frac{v_n}{z-p_n}\right]=\mathrm{e}^{-\mathrm{i}\lambda p_k/\varepsilon}\left(\frac{\mathrm{i}\lambda}{\varepsilon} \alpha_k-\beta_k\right),\quad k=1,\dots,N.
    \label{eq:zero-residues}
\end{equation}
For $\lambda<0$, we then integrate \eqref{eq:phi-multisol} to obtain
\begin{equation}
    R(z)\varphi(z)=\int_z^{+\mathrm{i}\infty}\mathrm{e}^{\mathrm{i}\lambda(z-\zeta)/\varepsilon}R(\zeta)\sum_{n=1}^N\frac{v_n}{\zeta-p_n}\,\mathrm{d}\zeta + K\mathrm{e}^{\mathrm{i}\lambda z/\varepsilon},
\end{equation}
where $K$ is an integration constant and the integral is independent of any path avoiding the points $p_1,\dots,p_N\in\mathbb{C}_+$ by \eqref{eq:zero-residues}.  Since $\varphi\in L_2^+(\mathbb{R})$ and $R(z)\to 1$ as $z\to\infty$, by letting $z\to +\mathrm{i}\infty$ we see that $K=0$.  The right-hand side is therefore a meromorphic function of $z$ with poles only at $\{p_1,\dots,p_N\}$.  By \eqref{eq:R-sum-partial-fractions}--\eqref{eq:alphanbetan}, the residue of $R(z)\varphi(z)$ at $p_n$ is $\alpha_n=R_n(p_n)v_n$.  One can also check by expanding the integral for large $z$ that $R(z)\varphi(z)\to 0$ as $z\to\infty$ in all directions so 
\begin{equation}
    R(z)\varphi(z)=\sum_{n=1}^N\frac{R_n(p_n)v_n}{z-p_n}\implies \varphi(z)=\sum_{n=1}^Nv_nf_n(z),\quad f_n(z):=\frac{R_n(p_n)}{(z-p_n)R(z)}=\frac{R_n(p_n)}{R_n(z)}.
    \label{eq:varphi-sum}
\end{equation}
Therefore any eigenfunction $\varphi\in L_+^2(\mathbb{R})$ of $\mathcal{L}_{u_0}$ for $u_0=\Pi u_0 + (\Pi u_0)^*$ with $\Pi u_0$ given by \eqref{eq:rational-soliton} belongs to $F$.  On the other hand, it it is known \cite[Proposition 4.3]{Sun2020} that $\mathcal{L}_{u_0}$ has exactly $N$ simple negative eigenvalues $\lambda_1<\cdots<\lambda_N<0$ with linearly independent eigenfunctions $\varphi_j\in L^2_+(\mathbb{R})$.  It follows that $F=\mathrm{span}\{\varphi_j\}_{j=1}^N$.

$F$ is an invariant space for $\mathcal{L}_{u_0}$ and for the operator $\mathcal{X}^*$.  The matrix of $\mathcal{L}_{u_0}\vert_F$ in the basis $\{f_k\}_{k=1}^N$ is $-\mathbf{M}$.  The matrix of $\mathcal{X}^*\vert_F$ in the same basis is $\mathbf{D}-\mathbf{1r}^\top$.
\label{lem:L-and-Xstar-fs}

<!-- proof-2: body_md -->
Since $F=\mathrm{span}\{\varphi_j\}_{j=1}^N$ where $\varphi_j\in L_+^2(\mathbb{R})$ with $\varphi_j\neq 0$ satisfies $\mathcal{L}_{u_0}\varphi_j=\lambda_j\varphi_j$, 
then it is obvious that $F$ is invariant under $\mathcal{L}_{u_0}$.  Also, combining \eqref{eq:zero-residues} with \eqref{eq:alphanbetan} and comparing with \eqref{def:M} shows that the vector
    $\mathbf{v}^{(j)}:=[v_1^{(j)}, v_2^{(j)}, \cdots, v_N^{(j)}]^\top$, $v_n^{(j)}=\varphi_j(p_n)$, must satisfy the equation
\begin{equation}
\mathbf{M}\mathbf{v}^{(j)}=-\lambda_j \mathbf{v}^{(j)}.
	\label{eq:M-lax}
\end{equation}
But by \eqref{eq:varphi-sum}, $\mathbf{v}^{(j)}$ is the vector of coordinates of $\varphi_j$ in the basis $\{f_n\}_{n=1}^N$.  This proves that $-\mathbf{M}$ is the matrix of $\mathcal{L}_{u_0}\vert_F$ in that basis.

Since by \eqref{eq:fk-define} and \eqref{eq:Rn-define}, $f_n(\diamond)\in\mathrm{dom}(\mathcal{X}^*)$ with $\mathfrak{a}[f_n]=R_n(p_n)$ according to \eqref{eq:domain-Xstar-alternate}, applying the action in \eqref{eq:Xstar-action-x} gives
\begin{equation}
\begin{aligned}
    \mathcal{X}^*f_n(z)=zf_n(z)-\mathfrak{a}[f_n]
    &=p_nf_n(z) + \frac{R_n(p_n)}{R(z)}\left[1-R(z)\right]\\
    &=p_nf_n(z)-\frac{R_n(p_n)}{R(z)}\sum_{k=1}^N\frac{R_k(p_k)}{z-p_k}\\
    &=p_nf_n(z)-R_n(p_n)\sum_{k=1}^Nf_k(z)
    \end{aligned}
\end{equation}
using the expansion \eqref{eq:R} of $R(z)$ in partial fractions on the second line.  Therefore $F$ is also invariant under $\mathcal{X}^*$.  Since then
\begin{equation}
    \mathcal{X}^*\sum_{n=1}^Nv_nf_n(z) = \sum_{n=1}^Np_nv_nf_n(z) - \sum_{k=1}^N\left(\sum_{n=1}^NR_n(p_n)v_n\right)f_k(z),
\end{equation}
comparing with \eqref{eq:D-define} and \eqref{eq:r1-define} shows that the matrix of $\mathcal{X}^*\vert_F$ in the basis $\{f_n\}_{n=1}^N$ is $\mathbf{D}-\mathbf{1r}^\top$.

<!-- proof-3: body_md -->
According to \cite{Wu16a}, for every eigenvalue  $\lambda=\lambda_\mathrm{s}<0$ of $\mathcal{L}_{u_0}$ we have $\dim(\mathrm{Ker}(\mathcal{L}_{u_0}-\lambda_\mathrm{s}\mathrm{Id}))=1$, and by \cite[Section 3.3]{MillerW16a}, $\mathrm{Ker}(\mathbf{E}(v_\mathrm{s}))$ is isomorphic to $\mathrm{Ker}(\mathcal{L}_{u_0}-\lambda_\mathrm{s}\mathrm{Id})$ for $v_\mathrm{s}=-2\lambda_\mathrm{s}$, so there is a nonzero vector $\boldsymbol{\phi}\in\mathbb{C}^N$ such that $\mathrm{Ker}(\mathbf{E}(v_\mathrm{s}))=\mathrm{span}\{\boldsymbol{\phi}\}$.  It then follows from the theory of holomorphic matrix pencils (see \cite[Section 2.2]{KollarM14} for instance) that the two conditions $E(v_\mathrm{s})=E'(v_\mathrm{s})=0$ imply the existence of a vector $\boldsymbol{\psi}\in\mathbb{C}^N$ such that
\begin{equation}
    \mathbf{E}(v_\mathrm{s})\boldsymbol{\psi} + \mathbf{E}^{\prime}(v_\mathrm{s})\boldsymbol{\phi}=\mathbf{0}.
\end{equation}
Now differentiation with respect to $v$ gives $E_{jk}^{\prime}(v)=\frac{1}{2}\mathrm{i} E_{jk}^{[1]}(v)$, so using \eqref{eq:PN-D1} we can write the equation on $\boldsymbol{\psi}$ in the form
\begin{equation}
    \mathbf{E}(v_\mathrm{s})\boldsymbol{\psi} +\frac{1}{2}\mathrm{i} \mathbf{f}(v_\mathrm{s})\mathbf{1}^\top\boldsymbol{\phi} + \frac{1}{2}\mathrm{i}\mathbf{E}(v_\mathrm{s})\mathrm{diag}(p_1,\dots,p_N)\boldsymbol{\phi}=\mathbf{0}.
    \label{eq:PN-psi-vector-equation}
\end{equation}
An eigenfunction $\Phi\in \mathrm{Ker}(\mathcal{L}_{u_0}-\lambda_\mathrm{s}\mathrm{Id})$ can be normalized so that $x\Phi(x)\to 1$ as $x\to\infty$ in $\mathbb{C}_+$, and the isomorphism with $\mathrm{Ker}(\mathbf{E}(v_\mathrm{s}))$ developed in \cite[Section 3.3]{MillerW16a} shows that the corresponding nonzero nullvector $\boldsymbol{\phi}$ satisfies
$\mathbf{1}^\top\boldsymbol{\phi}=\lambda_\mathrm{s}\neq 0$.  Therefore the existence of $\boldsymbol{\psi}$ solving \eqref{eq:PN-psi-vector-equation} is equivalent to the existence of $\mathbf{a}\in\mathbb{C}^N$ given explicitly by
\begin{equation}
    \mathbf{a}:= \frac{2\mathrm{i} \boldsymbol{\psi}-\mathrm{diag}(p_1,\dots,p_N)\boldsymbol{\phi}}{\mathbf{1}^\top\boldsymbol{\phi}}
\end{equation}
that satisfies $\mathbf{E}(v_\mathrm{s})\mathbf{a}=\mathbf{f}(v_\mathrm{s})$.  We will show that existence of such $\mathbf{a}\in\mathbb{C}^N$ leads to a contradiction, implying that $E(v_\mathrm{s})$ and $E'(v_\mathrm{s})$ cannot both vanish.

To this end, consider the function $g(x)$ defined by the formula
\begin{equation}
    g(x):=\mathrm{e}^{\mathrm{i} h^\infty(x;v_\mathrm{s})}\int_{\mathrm{e}^{3\pi\mathrm{i}/4}\infty}^x\mathrm{e}^{-\mathrm{i} h^\infty(z;v_\mathrm{s})}\left(\sum_{k=1}^N\frac{a_k}{z-p_k}-1\right)\,\mathrm{d} z.
\end{equation}
The hypothesis that $\mathbf{E}(v_\mathrm{s})\mathbf{a}=\mathbf{f}(v_\mathrm{s})$ implies that $g$ is the boundary value of a function analytic in the upper half $x$-plane (see \cite[Section 3.2]{MillerW16a}).  In fact, $g\in L^\infty_+(\mathbb{R})$, i.e., it is bounded on the closed upper half-plane, and one can verify that $g(x)\to -\mathrm{i}\lambda_\mathrm{s}^{-1}$ as $x\to\infty$ in $\mathbb{C}_+$.  Therefore also $u_0g\in L^2(\mathbb{R})$.  We then compute
\begin{equation}
    (\mathcal{L}_{u_0}-\lambda_\mathrm{s}\mathrm{Id})g(x)=-\mathrm{i} g'(x)-\Pi(u_0g)(x)-\lambda_\mathrm{s} g(x).
\end{equation}
Since $g\in L_+^\infty(\mathbb{R})$, we have
\begin{equation}
    \Pi(u_0g)(x)=u_0(x)g(x)-\sum_{k=1}^N\frac{c_kg(p_k)}{x-p_k}.
\end{equation}
The second term depends on $g$ through its analytic continuation to the points $\{p_k\}\subset\mathbb{C}_+$, but it is a rational function of $x$.  Therefore,
\begin{equation}
\begin{split}
    (\mathcal{L}_{u_0}-\lambda_\mathrm{s}\mathrm{Id})g(x)&=\mathrm{e}^{\mathrm{i} h^\infty(x;v_\mathrm{s})}\left[\mathrm{e}^{-\mathrm{i} h^\infty(x;v_\mathrm{s})}\left(-\mathrm{i} g'(x)-u_0(x)g(x)-\lambda_\mathrm{s}g(x)\right)\right]+\sum_{k=1}^N\frac{c_kg(p_k)}{x-p_k}\\
    &=-\mathrm{i}\mathrm{e}^{\mathrm{i} h^\infty(x;v_\mathrm{s})}\frac{\mathrm{d}}{\mathrm{d} x}\left[\mathrm{e}^{-\mathrm{i} h^\infty(x;v_\mathrm{s})}g(x)\right] + \sum_{k=1}^N\frac{c_kg(p_k)}{x-p_k}\\
    &=-\mathrm{i}\left(\sum_{k=1}^N\frac{a_k}{x-p_k}-1\right) + \sum_{k=1}^N\frac{c_kg(p_k)}{x-p_k}.
\end{split}
\end{equation}
On the other hand, since by hypothesis $g\in L_+^\infty(\mathbb{R})$, the left-hand side has to be the boundary value for $x\in\mathbb{R}$ of a function analytic in the upper half $x$-plane, which implies that 
$c_kg(p_k)=\mathrm{i} a_k$ for all $k$, and hence $g(x)$ satisfies
\begin{equation}
    (\mathcal{L}_{u_0}-\lambda_\mathrm{s}\mathrm{Id})g(x)=\mathrm{i},
\label{eq:PN-g-equation}
\end{equation}
a constant nonzero function.  

We can write $g(x)=-\mathrm{i}\lambda_\mathrm{s}^{-1} + h(x)$, where $h\in L_+^2(\mathbb{R})$.  Now for $\epsilon>0$, let $\chi_\epsilon\in L_+^2(\mathbb{R})$ be defined by
\begin{equation}
    \chi_\epsilon(x):=\frac{1}{1-\mathrm{i}\epsilon x},\quad\epsilon>0.
    \label{eq:PN-chi-epsilon-def}
\end{equation}
Note that $\chi_\epsilon(x)\to 1$ as $\epsilon\downarrow 0$ for every $x\in\mathbb{R}$ and that $|\chi_\epsilon(x)|\le 1$.
Recalling the eigenfunction $\Phi\in \mathrm{Ker}(\mathcal{L}_{u_0}-\lambda_\mathrm{s}\mathrm{Id})$ normalized so that 
$\Phi(x)=x^{-1}+O(x^{-2})$ as $x\to\infty$ in $\mathbb{C}_+$, multiplying \eqref{eq:PN-g-equation} by $\Phi(x)^*\chi_\epsilon(x)$ and integrating over $\mathbb{R}$ implies that
\begin{equation}
    \langle\chi_\epsilon (\mathcal{L}_{u_0}-\lambda_\mathrm{s}\mathrm{Id})g,\Phi\rangle = \mathrm{i}\langle\chi_\epsilon,\Phi\rangle,\quad\epsilon>0.
\end{equation}
Using $g(x)=-\mathrm{i}\lambda_\mathrm{s}^{-1}+h(x)$, where the action of $\mathcal{L}_{u_0}-\lambda_\mathrm{s}\mathrm{Id}$ on the constant function $1$ is well defined:
\begin{equation}\label{eq:Lax-one}
    (\mathcal{L}_{u_0}-\lambda_\mathrm{s}\mathrm{Id})1(x)=-\Pi u_0(x)-\lambda_\mathrm{s},
\end{equation} 
we get that $h$ is in the domain of $\mathcal{L}_{u_0}$, and
\begin{equation}
\langle\chi_\epsilon (\mathcal{L}_{u_0}-\lambda_\mathrm{s}\mathrm{Id})h,\Phi\rangle -\mathrm{i} \lambda_\mathrm{s}^{-1}\langle\chi_\epsilon(\mathcal{L}_{u_0}-\lambda_\mathrm{s}\mathrm{Id})1,\Phi\rangle=\mathrm{i}\langle\chi_\epsilon,\Phi\rangle,\quad\epsilon>0.
\label{eq:PN-inner-product-equation}
\end{equation}
Using~\eqref{eq:Lax-one}, we arrive at the identity
\begin{equation}
    \langle\chi_\epsilon(\mathcal{L}_{u_0}-\lambda_\mathrm{s}\mathrm{Id})h,\Phi\rangle +\mathrm{i}\lambda_\mathrm{s}^{-1}\langle\chi_\epsilon\Pi u_0,\Phi\rangle = 0,\quad\epsilon>0.
\end{equation}
Because $h$ is in the domain of $\mathcal{L}_{u_0}$ while $\Phi$ and $\Pi u_0$ are both in $L_+^2(\mathbb{R})$, a dominated convergence argument shows that the limit $\epsilon\downarrow 0$ can be taken in both terms, replacing $\chi_\epsilon$ by its pointwise limit $1$ and yielding
\begin{equation}
    \langle (\mathcal{L}_{u_0}-\lambda_\mathrm{s}\mathrm{Id})h,\Phi\rangle  +\mathrm{i}\lambda_\mathrm{s}^{-1}\langle \Pi u_0,\Phi\rangle=0.
\label{eq:after-dominated-convergence}
\end{equation}
Now since $\mathcal{L}_{u_0}-\lambda_\mathrm{s}\mathrm{Id}$ is self adjoint we have
\begin{equation}
    \langle (\mathcal{L}_{u_0}-\lambda_\mathrm{s}\mathrm{Id})h,\Phi\rangle=\langle h,(\mathcal{L}_{u_0}-\lambda_\mathrm{s}\mathrm{Id})\Phi\rangle = \langle h,0\rangle = 0
\end{equation}
because $\mathcal{L}_{u_0}\Phi = \lambda_\mathrm{s}\Phi$, and since $-\mathrm{i}\lambda_\mathrm{s}\neq 0$, \eqref{eq:after-dominated-convergence} gives
\begin{equation}
    \langle\Pi u_0,\Phi\rangle = 0.
    \label{eq:PN-penultimate-identity}
\end{equation}
Then, using
$\langle \Pi u_0,\Phi\rangle = \langle u_0,\Pi\Phi\rangle = \langle u_0,\Phi\rangle$, \eqref{eq:PN-penultimate-identity} becomes
\begin{equation}
    \int_\mathbb{R} \Phi(x)u_0(x)\,\mathrm{d} x = 0
\end{equation}
because $u_0$ is real valued.  On the other hand using the normalization $\Phi(x)=x^{-1}+O(x^{-2})$ and the singular integral representation \eqref{eq:Pi-singular-integral} of $\Pi$ to expand the Lax equation $\mathcal{L}_{u_0}\Phi=\lambda_\mathrm{s}\Phi$ as $x\to\infty$ in $\mathbb{C}_+$ shows that (see also \cite[Lemma 2.5 and (2.30)]{Wu16a})
\begin{equation}
    \int_{\mathbb{R}}\Phi(x)u_0(x)\,\mathrm{d} x = 2\pi\mathrm{i}\lambda_\mathrm{s}\neq 0.
\end{equation}
This is the desired contradiction.  Hence the hypothesis that there exists $\mathbf{a}\in \mathbb{C}^N$ such that $\mathbf{E}(v_\mathrm{s})\mathbf{a}=\mathbf{f}(v_\mathrm{s})$ is false, and simplicity of the roots of the determinant $E(v)$ is proved.

\label{prop:sigma1/sigma0}

Recall $\mathbf{1}:=(1,1,\dots,1)^\top$ and set $\mathbf{p}:=(p_1,p_2,\dots,p_N)^\top$.
Then note that if $E(v)\neq 0$, by Cramer's rule,
\begin{equation}
    \frac{\Sigma_0(v)}{E(v)} = \mathbf{1}^\top\mathbf{w}(v)\quad\text{and}\quad\frac{v\Sigma_1(v)}{E(v)}=\mathbf{1}^\top\mathbf{x}(v)+\mathbf{p}^\top\mathbf{y}(v),
\end{equation}
where $\mathbf{w}(v)$, $\mathbf{x}(v)$, $\mathbf{y}(v)$ are the unique solutions of 
\begin{equation}
    \mathbf{E}(v)\mathbf{w}(v)=\mathbf{f}(v),\quad
    \mathbf{E}(v)\mathbf{x}(v)=\mathbf{g}(v),\quad
    \mathbf{E}(v)\mathbf{y}(v)=\mathbf{h}(v),\quad E(v)\neq 0.
\end{equation}
Therefore also
\begin{equation}
    \frac{\Sigma_1(v)}{\Sigma_0(v)} = \frac{\mathbf{1}^\top\mathbf{x}(v)+\mathbf{p}^\top\mathbf{y}(v)}{v\mathbf{1}^\top\mathbf{w}(v)},\quad E(v)\neq 0.
\end{equation}
Since $\Sigma_j(v)$ are analytic in $v$ at $v=v_\mathrm{s}>0$ and $\Sigma_0(v_\mathrm{s})\neq 0$, we can take the limit to obtain
\begin{equation}
    \frac{\Sigma_1(v_\mathrm{s})}{\Sigma_0(v_\mathrm{s})} = \frac{1}{v_\mathrm{s}}\lim_{v\to v_\mathrm{s}}\frac{\mathbf{1}^\top\mathbf{x}(v)+\mathbf{p}^\top\mathbf{y}(v)}{\mathbf{1}^\top\mathbf{w}(v)}.
\label{eq:PN-Sigma1Sigma0-limit}
\end{equation}

Since $E(v)=\det(\mathbf{E}(v))$ has a simple zero at $v=v_\mathrm{s}$, the vectors $\mathbf{w}(v)$, $\mathbf{x}(v)$, and $\mathbf{y}(v)$ have at worst simple poles at $v=v_\mathrm{s}$.  For instance $\mathbf{w}(v)$ has a Laurent expansion of the form
\begin{equation}
    \mathbf{w}(v)=\frac{1}{v-v_\mathrm{s}}\mathbf{w}^{[-1]} + \mathbf{w}^{[0]} + O(v-v_\mathrm{s}),
\end{equation}
and using the Taylor expansions
\begin{equation}
    \begin{split}
        \mathbf{E}(v)&=\mathbf{E}(v_\mathrm{s}) + (v-v_\mathrm{s})\mathbf{E}^{\prime}(v_\mathrm{s}) + O((v-v_\mathrm{s})^2)\\
        \mathbf{f}(v)&=\mathbf{f}(v_\mathrm{s}) + O(v-v_\mathrm{s}),
    \end{split}
\end{equation}
we substitute into $\mathbf{E}(v)\mathbf{w}(v)=\mathbf{f}(v)$ to obtain
\begin{equation}
    \mathbf{E}(v_\mathrm{s})\mathbf{w}^{[-1]}=\mathbf{0},
\label{eq:PN-w-leading}
\end{equation}
\begin{equation}
    \mathbf{E}(v_\mathrm{s})\mathbf{w}^{[0]}=\mathbf{f}(v_\mathrm{s}) -\mathbf{E}^{\prime}(v_\mathrm{s})\mathbf{w}^{[-1]},
\label{eq:PN-w-correction}
\end{equation}
and so on.  Clearly \eqref{eq:PN-w-leading} implies that we need $\mathbf{w}^{[-1]}\in \mathrm{Ker}(\mathbf{E}(v_\mathrm{s}))=\mathrm{span}\{\boldsymbol{\phi}\}$.  Normalizing $\boldsymbol{\phi}$ by assuming that $\mathbf{1}^\top\boldsymbol{\phi}=\lambda_\mathrm{s}=-\frac{1}{2}v_\mathrm{s}$, we may therefore write $\mathbf{w}^{[-1]}=\tau\boldsymbol{\phi}$ for some scalar $\tau$.  To determine $\tau$, let $\mathbf{m}$ be a nonzero left nullvector of $\mathbf{E}(v_\mathrm{s})$, i.e., $\mathbf{m}^\top\mathbf{E}(v_\mathrm{s})=\mathbf{0}^\top$.  Then the solvability condition for existence of $\mathbf{w}^{[0]}$ is obtained by multiplying \eqref{eq:PN-w-correction} on the left by $\mathbf{m}^\top$, leading to
\begin{equation}
\tau=\frac{\mathbf{m}^\top\mathbf{f}(v_\mathrm{s})}{\mathbf{m}^\top\mathbf{E}^{\prime}(v_\mathrm{s})\boldsymbol{\phi}}\implies \mathbf{w}(v)=\frac{1}{v-v_\mathrm{s}}\frac{\mathbf{m}^\top\mathbf{f}(v_\mathrm{s})}{\mathbf{m}^\top\mathbf{E}^{\prime}(v_\mathrm{s})\boldsymbol{\phi}}\boldsymbol{\phi}+O(1).
\end{equation}
Similarly, one finds that 
\begin{equation}
    \mathbf{x}(v)=\frac{1}{v-v_\mathrm{s}}\frac{\mathbf{m}^\top\mathbf{g}(v_\mathrm{s})}{\mathbf{m}^\top\mathbf{E}^{\prime}(v_\mathrm{s})\boldsymbol{\phi}}\boldsymbol{\phi}+O(1),\quad\mathbf{y}(v)=\frac{1}{v-v_\mathrm{s}}\frac{\mathbf{m}^\top\mathbf{h}(v_\mathrm{s})}{\mathbf{m}^\top\mathbf{E}^{\prime}(v_\mathrm{s})\boldsymbol{\phi}}\boldsymbol{\phi}+O(1).
\end{equation}
By $E_{jk}^{\prime}(v)=\frac{1}{2}\mathrm{i} E_{jk}^{[1]}(v)$ and \eqref{eq:PN-D1} we have
\begin{equation}
    \mathbf{m}^\top\mathbf{E}^{\prime}(v_\mathrm{s})\boldsymbol{\phi} =\frac{1}{2}\mathrm{i}\mathbf{m}^\top\mathbf{f}(v_\mathrm{s})\mathbf{1}^\top\boldsymbol{\phi} +\frac{1}{2}\mathrm{i}\mathbf{m}^\top\mathbf{E}(v_\mathrm{s})\mathrm{diag}(p_1,\dots,p_N)\boldsymbol{\phi} = -\frac{1}{4}\mathrm{i} v_\mathrm{s} \mathbf{m}^\top\mathbf{f}(v_\mathrm{s}),
\end{equation}
where we used $\mathbf{m}^\top\mathbf{E}(v_\mathrm{s})=\mathbf{0}^\top$ and $\mathbf{1}^\top\boldsymbol{\phi}=-\frac{1}{2}v_\mathrm{s}$.  The condition $\mathbf{m}^\top\mathbf{f}(v_\mathrm{s})=0$ is equivalent to the existence of $\mathbf{a}\in\mathbb{C}^N$ such that $\mathbf{E}(v_\mathrm{s})\mathbf{a}=\mathbf{f}(v_\mathrm{s})$, and it was shown in the proof of simplicity of zeros of $E(v)$ that no such $\mathbf{a}$ can exist.  
Therefore, cancelling nonzero common denominators, \eqref{eq:PN-Sigma1Sigma0-limit} becomes
\begin{equation}
    \frac{\Sigma_1(v_\mathrm{s})}{\Sigma_0(v_\mathrm{s})}=\frac{1}{v_\mathrm{s}}\frac{\mathbf{m}^\top\mathbf{g}(v_\mathrm{s})\mathbf{1}^\top\boldsymbol{\phi} + \mathbf{m}^\top\mathbf{h}(v_\mathrm{s})\mathbf{p}^\top\boldsymbol{\phi}}{\mathbf{m}^\top\mathbf{f}(v_\mathrm{s})\mathbf{1}^\top\boldsymbol{\phi}}=\frac{v_\mathrm{s}\mathbf{m}^\top\mathbf{g}(v_\mathrm{s})-2\mathbf{m}^\top\mathbf{h}(v_\mathrm{s})\mathbf{p}^\top\boldsymbol{\phi}}{v_\mathrm{s}^2\mathbf{m}^\top\mathbf{f}(v_\mathrm{s})}.
\label{eq:PN-Sigma1-Sigma0-no-limit}
\end{equation}

We focus first on the term $\mathbf{m}^\top\mathbf{h}(v_\mathrm{s})$.  Because $\mathbf{m}^\top\mathbf{E}(v_\mathrm{s})=\mathbf{0}^\top$, we can add some terms to $\mathbf{h}(v_\mathrm{s})$ without changing the value of $\mathbf{m}^\top\mathbf{h}(v_\mathrm{s})$:
\begin{equation}
    \mathbf{m}^\top\mathbf{h}(v_\mathrm{s}) = \mathbf{m}^\top\left(\mathbf{E}(v_\mathrm{s})\mathbf{c} +\mathbf{h}(v_\mathrm{s})\right),\quad \mathbf{c}:=(c_1,\dots,c_N)^\top.
\end{equation}
Using \eqref{eq:PN-Dp-define} and \eqref{eq:PN-g-and-h}, the components of $\mathbf{E}(v_\mathrm{s})\mathbf{c} +\mathbf{h}(v_\mathrm{s})$ are
\begin{equation}
\begin{split}
    \left(\mathbf{E}(v_\mathrm{s})\mathbf{c} +\mathbf{h}(v_\mathrm{s})\right)_j&=\int_{C_j}\mathrm{e}^{-\mathrm{i} h^\infty(z;v_\mathrm{s})}\left(\sum_{n=1}^N\frac{c_n}{z-p_n} + \sum_{n=1}^N\frac{c_n^*}{z-p_n^*}\right)\,\mathrm{d} z \\ &= \int_{C_j}\mathrm{e}^{-\mathrm{i} h^\infty(z;v_\mathrm{s})}u_0(z)\,\mathrm{d} z,\quad j=1,\dots,N,
\end{split}
\end{equation}
where on the second line we used \eqref{eq:PN-rational-IC}.  Since $(h^{\infty})^{\prime}(z;v_\mathrm{s})=u_0(z)-\frac{1}{2}v_\mathrm{s}$ and 
\begin{equation}
    \int_{C_j}\mathrm{e}^{-\mathrm{i} h^\infty(z;v_\mathrm{s})}(h^{\infty})^{\prime}(z;v_\mathrm{s})\,\mathrm{d} z=0,\quad j=1,\dots,N,
\end{equation}
we can also write
\begin{equation}
    \left(\mathbf{E}(v_\mathrm{s})\mathbf{c} +\mathbf{h}(v_\mathrm{s})\right)_j=\frac{1}{2}v_\mathrm{s}\int_{C_j}\mathrm{e}^{-\mathrm{i} h^\infty(z;v_\mathrm{s})}\,\mathrm{d} z = \frac{1}{2}v_\mathrm{s}f_j(v_\mathrm{s})
\end{equation}

\label{lem:detD-nonzero}

<!-- proof-7: body_md -->
We show that every $\lambda>0$ such that $\det(\mathbf{E}^\pm(\lambda))=0$ is an eigenvalue of $\mathcal{L}_{u_0}$, in contradiction with the fact that all eigenvalues are negative \cite{Wu16a}. The rest of the proof is very similar to the proof of~\cite[Proposition 1]{MillerW16a}, therefore we only recall the ideas.
The eigenvalue equation $\mathcal{L}_{u_0}g=\lambda g$ can be written as an ordinary differential equation on the complex $x$-plane:
\begin{equation}
-\mathrm{i} g'(x)-u_0(x)g(x)-\lambda g(x)=-\sum_{k=1}^N\frac{c_ng(p_n)}{x-p_n}.
\label{eq:spectral-ODE}
\end{equation}
Integrating yields two particular solutions $g=g_\pm(x)$ that have to satisfy
\begin{equation}
g_\pm(x)
    =\mathrm{e}^{\mathrm{i} h^{\infty}(x;-2\lambda)}\int_{\ell_0^\pm(x)}\mathrm{e}^{-\mathrm{i} h^{\infty}(x;-2\lambda)}\sum_{n=1}^N\frac{v_n^\pm(\lambda)}{z-p_n}\mathrm{d} z,\quad v_n^\pm(\lambda)=-\mathrm{i} c_ng_\pm(p_n),\quad n=1,\dots,N,
\end{equation}
where $\ell_0^\pm(x)$ is a contour from $z=\infty\mathrm{e}^{-\mathrm{i}\pi/4}\in\Omega^\pm$  to $x\in\mathbb{C}_+$. As in the proof of~\cite[Corollary 1]{MillerW16a}, one can show that $g_\pm$ is analytic on the upper half-plane if and only if
\begin{equation}\label{eq:Uj-f}
\int_{U_j^\pm}\mathrm{e}^{-\mathrm{i} h^{\infty}(x;-2\lambda)}\sum_{n=1}^N\frac{v_n^\pm(\lambda)}{z-p_n}\mathrm{d} z
    =0,
    \quad j=1,\dots,N.
\end{equation}
If this holds for some nonzero vector $\mathbf{v}^\pm(\lambda)=(v_1^\pm(\lambda),\dots,v_N^\pm(\lambda))^\top\in\mathbb{C}^N$, one can check that $g_\pm\in L_+^2(\mathbb{R})$ defines an eigenfunction of $\mathcal{L}_{u_0}$ associated to the eigenvalue $\lambda>0$, which is impossible.
It remains to observe that if $\lambda>0$, the existence of $\mathbf{v}^\pm(\lambda)\neq\mathbf{0}$ satisfying \eqref{eq:Uj-f} is equivalent to the assertion that $\det(\mathbf{E}^\pm(\lambda))=0$.

\quad j=1,\dots, N.
\end{equation}
Using this claim, it only remains to solve the system of equations~\eqref{eq:m-holo} thanks to Cramer's rule to get~\eqref{eq:wk}. 


The proof of the claim follows that of~\cite[Proposition 1]{MillerW16a}, so we only recall the ideas. Fixing $j=1,\dots,N$, we require that $m_-(\cdot,\lambda)$ be analytic at $p_j$. We first assume that $\operatorname{Re}(\mathrm{i} c_j)<0$. Then $\mathrm{e}^{-\mathrm{i} h^\infty(z;-2\lambda)}$ vanishes to some positive fractional order at $z=p_j$ but is not single-valued there (because $j$ is not an exceptional index); therefore 
for $x$ in the vicinity of $p_j$:
\begin{multline}
m_-(x,\lambda)
	=K_-\mathrm{e}^{\mathrm{i}  h^\infty(x;-2\lambda)}
    \left(1 	+\int_{\ell_0^+(p_j)} \mathrm{e}^{-\mathrm{i} h^\infty(z;-2\lambda)}\sum_{n=1}^N \frac{w_n^-(\lambda)}{z-p_n}\mathrm{d} z\right)\\
	+K_- 
    \mathrm{e}^{\mathrm{i}  h^\infty(x;-2\lambda)}\int_{p_j}^x\mathrm{e}^{-\mathrm{i} h^\infty(z;-2\lambda)}\sum_{n=1}^N \frac{w_n^-(\lambda)}{z-p_n}\mathrm{d} z.
\end{multline}
It turns out that the last term on the right hand side is analytic at $p_j$. However, the first term in the right hand side is analytic at $p_j$ if and only if 
\begin{equation}
1	+\int_{\ell_0^+(p_j)} \mathrm{e}^{-\mathrm{i} h^\infty(z;-2\lambda)}\sum_{n=1}^N \frac{w_n^-(\lambda)}{z-p_n}\mathrm{d} z
 	=0.
    \label{eq:first-pass}
\end{equation}
Now we introduce contours $V_j^\pm$ analogous to $U_j^\pm$ except that $V_j^\pm$ encircles only the pole $z=p_j$, and we define $\mathrm{e}^{-\mathrm{i} h^\infty(z;-2\lambda)}$ on these contours similarly as for $U_j^\pm$.  Then for $G(z)=(z-p_n)^{-1}$, $n=1,\dots,N$, we have the identity
\begin{equation}
    \int_{V_j^\pm}\mathrm{e}^{-\mathrm{i} h^\infty(z;-2\lambda)}G(z)\,\mathrm{d} z = (1-\mathrm{e}^{2\pi c_j})\int_{\ell_0^\pm(p_j)}\mathrm{e}^{-\mathrm{i} h^\infty(z;-2\lambda)} G(z)\,\mathrm{d} z,
\end{equation}
so \eqref{eq:first-pass} can be written equivalently as
\begin{equation}
    1-\mathrm{e}^{2\pi c_j} + \int_{V_j^+}\mathrm{e}^{-\mathrm{i} h^\infty(z;-2\lambda)}\sum_{n=1}^N\frac{w_n^-(\lambda)}{z-p_n}\,\mathrm{d} z = 0.
    \label{eq:second-pass}
\end{equation}
In the case $\operatorname{Re}(\mathrm{i} c_j)\geq 0$, we get the same conclusion after first integrating by parts.  Now we relate integration over $U_j^\pm$ to integration over $V_k^\pm$, $k=1,\dots,j$:
\begin{equation}
\begin{aligned}
    \int_{U_j^\pm}\mathrm{e}^{-\mathrm{i} h^\infty(z;-2\lambda)}\sum_{n=1}^N\frac{w_n^-(\lambda)}{z-p_n}\,\mathrm{d} z &= \sum_{k=1}^j\mathrm{e}^{2\pi(c_j+c_{j-1}+\cdots+c_{k+1})}\int_{V_k^\pm}\mathrm{e}^{-\mathrm{i} h^\infty(z;-2\lambda)}\sum_{n=1}^N\frac{w_n^-(\lambda)}{z-p_n}\,\mathrm{d} z \\
    &=-\sum_{k=1}^j\mathrm{e}^{2\pi(c_j+c_{j-1}+\cdots+c_{k+1})}(1-\mathrm{e}^{2\pi c_k})\\
    &=-(1-\mathrm{e}^{2\pi(c_1+\cdots+c_j)}),
\end{aligned}
\end{equation}
where on the second line we used \eqref{eq:second-pass} and on the third line we noted a telescoping sum.  This establishes \eqref{eq:m-holo} and completes the proof.

<!-- proof-12: body_md -->
Taking into account that $\mathbf{E}^-(\lambda)=\mathbf{E}^+(\lambda)+\mathbf{dq}(\lambda)^\top$ which follows from \eqref{eq:Uj+-} in Remark~\ref{rem:integral-identities}, showing \eqref{eq:vk-diff} is equivalent to proving that $S_2=-S_1$, where $S_1,S_2$ are defined as follows.
Using the expression~\eqref{eq:ypm-n} for $y_n^+(\lambda)$ and \eqref{eq:vk+-} for $y_n^-(\lambda)$, and using~\eqref{eq:wk} for $w_n^-(\lambda)$,
\begin{equation}S_1:=\det(\mathbf{E}^+\mathop{\longleftarrow}\limits^{n}\mathbf{b}^+)\det(\mathbf{E}^++\mathbf{d}\mathbf{q}^\top)
	-\det(\mathbf{E}^++\mathbf{d}\mathbf{q}^\top \mathop{\longleftarrow}\limits^{n} \mathbf{b}^+ +q_0 \mathbf{d})\det(\mathbf{E}^+)
\end{equation} and 
\begin{equation}S_2:=\det(\mathbf{A}^\infty_f)\det(\mathbf{E}^+\mathop{\longleftarrow}\limits^{n} \mathbf{d}),
\end{equation}
where we suppress the dependence on $\lambda$ for simplicity of notation.

Further simplifying the notation by writing $\mathbf{E}$ and $\mathbf{b}$ for $\mathbf{E}^+$ and $\mathbf{b}^+$ respectively, we have from Cramer's rule
\begin{equation}
\det(\mathbf{E} \mathop{\longleftarrow}\limits^{n} \mathbf{b})
	=\det(\mathbf{E})(\mathbf{E}^{-1}\mathbf{b})_n,
\end{equation}
where $(\mathbf{E}^{-1}\mathbf{b})_n$ denotes element $n$ of the vector $\mathbf{E}^{-1}\mathbf{b}$. 
Moreover, applying the matrix determinant lemma:
\begin{equation}\label{eq:detD}
\det(\mathbf{E}+\mathbf{d}\mathbf{q}^T)
	=\det(\mathbf{E})(1+\mathbf{q}^T \mathbf{E}^{-1}\mathbf{d}),
\end{equation}
the first term in the definition of $S_1$ therefore can be written as
\begin{equation}
\det(\mathbf{E} \mathop{\longleftarrow}\limits^{n} \mathbf{b})\det(\mathbf{E}+\mathbf{d}\mathbf{q}^T)
	=\det(\mathbf{E})^2 (\mathbf{E}^{-1}\mathbf{b})_n (1+\mathbf{q}^T \mathbf{E}^{-1}\mathbf{d}).
\end{equation}


Now looking at the second term in $S_1$, another application of Cramer's rule gives
\begin{equation}
\det(\mathbf{E}+\mathbf{d} \mathbf{q}^\top \mathop{\longleftarrow}\limits^{n} \mathbf{b}+q_0 \mathbf{d})
	=\det(\mathbf{E}+\mathbf{d}\mathbf{q}^\top)((\mathbf{E}+\mathbf{d}\mathbf{q}^\top)^{-1}(\mathbf{b}+q_0 \mathbf{d}))_n.
\end{equation}
Moreover, by the Sherman–Morrison formula, 
\begin{equation}
(\mathbf{E}+\mathbf{d}\mathbf{q}^\top)^{-1}
	=\mathbf{E}^{-1}-\frac{\mathbf{E}^{-1}\mathbf{d}\mathbf{q}^\top\mathbf{E}^{-1}}{1+\mathbf{q}^\top\mathbf{E}^{-1}\mathbf{d}}.
\end{equation}
Also using again~\eqref{eq:detD}, we find
\begin{equation}
\begin{aligned}
\det(\mathbf{E}+\mathbf{d} \mathbf{q}^\top \mathop{\longleftarrow}\limits^{n} \mathbf{b}+q_0 \mathbf{d})
	&=\det(\mathbf{E})((1+\mathbf{q}^\top\mathbf{E}^{-1}\mathbf{d})\mathbf{E}^{-1}(\mathbf{b}+q_0\mathbf{d})-\mathbf{E}^{-1}\mathbf{d}\mathbf{q}^\top\mathbf{E}^{-1}(\mathbf{b}+q_0\mathbf{d}))_n\\
	&=\det(\mathbf{E})\left(
	(1+\mathbf{q}^\top \mathbf{E}^{-1}\mathbf{d})(\mathbf{E}^{-1}\mathbf{b})_n+(q_0
	- \mathbf{q}^\top \mathbf{E}^{-1}\mathbf{b})(\mathbf{E}^{-1} \mathbf{d})_n
	\right).
\end{aligned}
\end{equation}
Finally, we obtain a simplied form of $S_1$:
\begin{equation}
S_1
	=-\det(\mathbf{E})^2
	(q_0-\mathbf{q}^\top \mathbf{E}^{-1}\mathbf{b}) (\mathbf{E}^{-1}\mathbf{d})_n.
\end{equation}

Now we turn to $S_2$.
Applying Cramer's rule to the second factor gives
\begin{equation}
\det(\mathbf{E} \mathop{\longleftarrow}\limits^{n} \mathbf{d} )=\det(\mathbf{E})(\mathbf{E}^{-1}\mathbf{d})_n.
\end{equation}
Then we expand the determinant in the first factor of $S_2$ along the first row and apply Cramer's rule to get
\begin{equation}
\begin{aligned}
\det(\mathbf{A}^\infty_f)
	&=q_0\det(\mathbf{E})-\sum_{n=1}^Nq_n\det(\mathbf{E} \mathop{\longleftarrow}\limits^{n}\mathbf{b})\\
	&=\det(\mathbf{E})\left(q_0-\sum_{n=1}^N q_n(\mathbf{E}^{-1}\mathbf{b})_n\right)\\
	&=\det(\mathbf{E})(q_0-\mathbf{q}^\top \mathbf{E}^{-1}\mathbf{b}).
\end{aligned}
\end{equation}
We conclude that 
\begin{equation}
S_2
	=\det(\mathbf{E})^2(\mathbf{E}^{-1} \mathbf{d})_n(q_0-\mathbf{q}^\top \mathbf{E}^{-1} \mathbf{b})
	=-S_1.
\end{equation}

\begin{remark}[Reflectionless potentials]
The $N$-soliton initial data $u_0$ considered in Section~\ref{sec:examples} and Appendix~\ref{app:Nsoliton} has $c_1=c_2=\cdots = c_N=-\mathrm{i}$.  In this case, $\mathrm{e}^{-\mathrm{i} h^\infty(z;-2\lambda)}$ has simple zeros at $p_1^*,\dots,p_N^*$, so the integrand in each of the first row entries of $\mathbf{A}^\infty_{\Pi u_0}(\lambda)$ is analytic in $\mathbb{C}_-$ and exponentially decaying as $z\to\infty\mathrm{e}^{-\mathrm{i}\pi/4}$ for $\lambda>0$.  It follows that the first row entries of $\mathbf{A}^\infty_{\Pi u_0}(\lambda)$ all vanish identically for $\lambda>0$, so $\widetilde{\Pi u_0}^\pm(\lambda)=0$ and $\beta(\lambda)=0$, consistent with the terminology of $u_0$ being a reflectionless potential.
\end{remark}
