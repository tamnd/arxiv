<!-- theorem-1: body_md -->
\label{thm:intro-uniformity}
Let $s\geqslant1$ be an integer and $\varepsilon>0$. Uniformly for $X\geqslant3$ and
$$
X^{3/5+\varepsilon}\leqslant H\leqslant X,
$$
one has\footnote{See Section~\ref{subsec:notation} for the definition of the Gowers norm}
\begin{equation}\label{eq:intro-gowers-theorem}
\left\lVert \Lambda-\Lambda_X^\sharp\right\rVert_{U^s(X,X+H]}=o_{s,\varepsilon}(1).
\end{equation}

<!-- theorem-2: body_md -->
\label{thm:discorrelation}
Let $d,D \in \mathbb{N}$, let $\varepsilon>0$, $\delta\in(0,1/2)$, $X\geqslant 3$, and
\begin{equation*}
  X^{3/5+\varepsilon}\leqslant H\leqslant X^{1-\varepsilon}.
\end{equation*}
Let $G/\Gamma$ be a filtered nilmanifold\footnote{The symbols relating to nilsequences have the same meanings as in \cite[Section~2.3]{MSTT}.} of degree at most $d$, dimension at most $D$, and complexity at most $1/\delta$, and let $F\colon G/\Gamma\to\mathbb C$ be a Lipschitz function of norm at most $1/\delta$. Then, for every $A>0$,
\begin{align}
  \sup_{g\in\operatorname{Poly}(\mathbb Z\to G)} \left\lvert\sum_{X<n\leqslant X+H}\mu(n)\overline{F(g(n)\Gamma)}\right\rvert^{\,*} &\ll_{A,\varepsilon,d,D} \delta^{-O_{d,D}(1)}\frac{H}{\log^A X},\label{eq:mobius-discor}\\
  \sup_{g\in\operatorname{Poly}(\mathbb Z\to G)} \left\lvert\sum_{X<n\leqslant X+H}(\Lambda(n)-\Lambda_X^\sharp(n))\overline{F(g(n)\Gamma)}\right\rvert^{\,*} &\ll_{A,\varepsilon,d,D} \delta^{-O_{d,D}(1)}\frac{H}{\log^A X}.\label{eq:lambda-discor}
\end{align}

<!-- corollary-1: body_md -->
\label{cor:polynomial-phases}
Let $\varepsilon>0$, $d\geqslant 1$, $A>0$, $X\geqslant 3$, and $X^{3/5+\varepsilon}\leqslant H\leqslant X^{1-\varepsilon}$. For every polynomial $P\in\mathbb R[Y]$ of degree at most $d$,
\begin{equation}\label{eq:mu-poly}
  \left\lvert\sum_{X<n\leqslant X+H}\mu(n)e(P(n))\right\rvert \ll_{A,d,\varepsilon} \frac{H}{\log^A X}.
\end{equation}
Moreover, we have
\begin{equation}\label{eq:lambda-poly}
  \left\lvert\sum_{X<n\leqslant X+H}\Lambda(n)e(P(n))\right\rvert \ll_{A,d,\varepsilon} \frac{H}{\log^A X},
\end{equation}
unless, writing
\begin{equation*}
  P(\lfloor X\rfloor +m)=\alpha_0+\alpha_1m+\cdots+\alpha_dm^d,
\end{equation*}
there exists an integer $1\leqslant q\leqslant \log^{O_{A,d,\varepsilon}(1)} X$ such that
\begin{equation}\label{eq:major-arc-poly}
  \max_{1\leqslant j\leqslant d} H^j\left\lVert q\alpha_j\right\rVert_{\mathbb R/\mathbb Z}
  \leqslant \log^{O_{A,d,\varepsilon}(1)}X.
\end{equation}

<!-- proof-1: body_md -->
Polynomial phases are nilsequences on the filtered torus $\mathbb R/\mathbb Z$ with filtration degree $d$, so \eqref{eq:mu-poly} follows directly from Theorem~\ref{thm:discorrelation}. For \eqref{eq:lambda-poly}, write $\Lambda=(\Lambda-\Lambda_X^\sharp)+\Lambda_X^\sharp$. The first term is covered by Theorem~\ref{thm:discorrelation}; the second is a standard major arc calculation for the Cram\'er model \eqref{eq:lambda-sharp}, giving a negligible contribution unless \eqref{eq:major-arc-poly} holds. This is the argument of \cite[Corollary~1.3]{MSTT} with $5/8$ replaced by $3/5$.

<!-- theorem-7: body_md -->
\label{thm:intro-linear}
Let $\varepsilon>0$ and $X^{3/5+\varepsilon}\leqslant H\leqslant X^{1-\varepsilon}$. Let $d,t\geqslant1$ be  integers and let $L\geqslant1$. Let $\Psi=(\psi_1,\ldots,\psi_t)$ be a system of affine linear forms $\psi_i\colon\mathbb Z^d\to\mathbb Z$ of the shape
$$
  \psi_i(\mathbf n)=\dot\psi_i\cdot \mathbf n+\psi_i(0),
  \qquad \dot\psi_i\in\mathbb Z^d,
$$
with $0<\|\dot\psi_i\|_\infty\leqslant L$ and $\left\lvert \psi_i(0)\right\rvert\leqslant LX$, and suppose that $\dot\psi_i$ and $\dot\psi_j$ are linearly independent over $\mathbb Q$ whenever $i\neq j$. Let $K\subset(X,X+H]^d$ be a convex body. Extend $\Lambda$ by zero to the nonpositive integers. Then,
\begin{equation}\label{eq:intro-linear-theorem}
  \sum_{\mathbf n\in K\cap\mathbb Z^d}\prod_{i=1}^t\Lambda(\psi_i(\mathbf n))
  =\beta_\infty\prod_p\beta_p+o_{d,t,L,\varepsilon}(H^d),
\end{equation}
where
$$
  \beta_\infty=\operatorname{vol}\bigl(K\cap\Psi^{-1}(\mathbb R_{>0}^t)\bigr)
$$
and
$$
  \beta_p=\mathbb E_{\mathbf n\in(\mathbb Z/p\mathbb Z)^d}\prod_{i=1}^t \frac{p}{p-1}\1_{\psi_i(\mathbf n)\not\equiv0\pmod p}.
$$

<!-- lemma-1: body_md -->
\label{le:MVHuGM}
Let $B \geqslant 1$ be fixed. Let $\beta>0$, $X\geqslant3$, and
\begin{equation*}
  A(s)=\sum_{X^\beta<n\leqslant 2X^\beta}a(n) n^{-s},\qquad |a(n)|\leqslant d_B(n)\log^B X.
\end{equation*}
Let $T = X^\tau$ with $\tau \in (0, 1]$.
Then the following statements hold:

\begin{enumerate}[label=(\roman*)]
  \item For every $\nu \geqslant 0$ and every fixed $k \in \mathbb{N}$, we have
\begin{equation}\label{eq:hm-large-values-bound}
  \mathcal N_A(\nu ; [T, 2T])\ll X^{\mathcal{K}(\beta k,\nu;\tau)+o(1)}.
\end{equation}

  \item There is a constant $C = C(B) > 0$ such that the following holds. Let 
  \[
  a(n) = \mathbf{1}_{n \in I} \chi(n) \log^r n,
  \]
   where $I \subseteq (X^\beta, 2X^\beta]$ is an interval, $\chi$ is a Dirichlet character of modulus $\ll \log^B X$ and $r \in \{0, 1\}$. Let $\mathcal{T} \subseteq [T, 2T]$ be such that
  \begin{equation} \label{eq:4thmomcond}
|A(1/2+it)| \geqslant \frac{X^{\beta/2}}{T} \log^C X
\end{equation}
for every $t \in \mathcal{T}$. Then, if $\tau>1/\sqrt{\log X}$, for every $\nu \geqslant 0$ we have
  \begin{equation} \label{eq:fourthMomentN}
    \mathcal{N}_A(\nu; \mathcal{T}) \ll X^{\tau + 2\beta(2\nu-1) +o(1)}.
  \end{equation}
\end{enumerate}

\textbf{Proof of (i).} We first note that it suffices to prove
  \begin{equation} \label{eq:LVk=1}
    \mathcal{N}_A(\nu; [T, 2T])
    \ll X^{\mathcal{K}(\beta, \nu; \tau) +o(1)}.
  \end{equation}
  Indeed, let $k\in\mathbb N$ be fixed and write
  \[
    A(s)^k=\sum_{n\in \mathbb{N}}\frac{c_k(n)}{n^s}.
  \]
  The coefficients are supported on $N^k<n\leqslant2^kN^k$, where
  $N=X^\beta$, and satisfy
  \[
    |c_k(n)|\leqslant d_{Bk}(n) \log^{Bk} X.
  \]
  Split
  \[
    A(s)^k=\sum_{\ell=0}^{k-1}A_{k,\ell}(s),\qquad  A_{k,\ell}(s)=\sum_{2^\ell N^k<n\leqslant2^{\ell+1}N^k}\frac{c_k(n)}{n^s}.
  \]
  Let $\mathcal U\subseteq[T,2T]$ be a $1$-separated set on which
  \[
    |A(1/2+it)|\geqslant X^{\beta(1/2-\nu)}.
  \]
  Then, for every $t\in\mathcal U$, at least one
  $\ell\in\{0,\ldots,k-1\}$ satisfies
\[
|A_{k,\ell}(1/2+it)|\geqslant \frac1k X^{k\beta(1/2-\nu)}.
\]
  We partition $\mathcal U$ according to the least such $\ell$. Since
  $2^{\ell+1}N^k=X^{k\beta+O_k(1/\log X)}$, applying
  \eqref{eq:LVk=1} to each $A_{k,\ell}$, with the parameters changed
  by $O_{B,k,\beta,\nu}(1/\log X)$, gives
\[
    \#\mathcal U\ll_{B,k}X^{\mathcal K(k\beta,\nu;\tau)+o(1)},
\]
  where we used that $\mathcal K$ is piecewise linear. This proves the
  reduction to \eqref{eq:LVk=1}.

     We now turn to the proof of \eqref{eq:LVk=1}. Recall that $T = X^\tau$. If $\mathcal{K}(\beta, \nu; \tau) = \tau$, then $X^{\mathcal{K}(\beta, \nu; \tau)} = X^\tau = T$ and the claim is trivial. We may therefore assume henceforth that
      \begin{equation} \label{eq:Ksimp}
         \mathcal{K}(\beta, \nu; \tau) = \max\left\{2\beta \nu, \mathcal{K}_0(\beta, \nu; \tau)\right\} \leqslant \tau.
       \end{equation}
       
Set $N=X^\beta$ and $V=N^{1/2-\nu}$, and let
$\mathcal U\subseteq[T,2T]$ be any $1$-separated set such that
\[
  |A(1/2+it)|\geqslant V\qquad\text{for }t\in\mathcal U.
\]
Write
$$
  G\coloneqq \sum_{N<n\leqslant2N}\frac{|a(n)|^2}{n}\ll \log^{2B} X N^{-1}\sum_{N<n \leqslant 2N} d_B(n)^2\ll \log^{O_B(1)} X.
$$
By the second moment large values theorem \cite[Theorem~9.7]{IwaniecKowalski}, one has
\begin{equation}\label{eq:IK-thm-97-use}
  \#\mathcal{U}\ll (N+T)GV^{-2}\log(2N) \ll (X^{2\beta\nu}+X^{\tau+\beta(2\nu-1)})X^{o(1)}.
\end{equation}
By Huxley's large values estimate in the form of \cite[Corollary~9.9]{IwaniecKowalski}, one has
\begin{equation}\label{eq:huxley-use}
  \#\mathcal{U}\ll \left(GNV^{-2}+G^3N TV^{-6}\right) \log^6 (2T) \ll (X^{2\beta \nu}+X^{\tau+\beta(6\nu-2)})X^{o(1)}.
\end{equation}

Next we apply the large value results of Guth and
Maynard~\cite{GuthMaynard}, together with the refinement due to Chen,
Gupta, and Li~\cite{CGCL}. Let $C_B$ be sufficiently large and set
\[
  A^\ast(it)\coloneqq N^{1/2-C_B/\log\log N}A(1/2+it)=\sum_{N<n\leqslant2N}b(n)n^{-it},
\]
where
\[
  b(n)=\frac{a(n)}{N^{C_B/\log\log N}}\left(\frac Nn\right)^{1/2}.
\]
The standard bound for the divisor function shows that if $C_B$ is large enough, then $|b(n)|\leqslant1$ for all $n$. Moreover, for
$t\in\mathcal U$,
\[
  |A^\ast(it)|\geqslant V^\ast,\qquad V^\ast\coloneqq N^{1-\nu-C_B/\log\log N}.
\]

Suppose first that $\beta\geqslant5\tau/6$. Then, for all sufficiently
large $X$,
\[
  N=X^\beta\geqslant(2T)^{2/3}.
\]
We may therefore apply
\cite[Theorem~1.1]{CGCL}, with $q=q_1=1$ and height $2T$, to the

\label{lem:sparse-mean-square}
Let $N, T \geqslant 2$ and
$$
  A(s)=\sum_{N<n\leqslant2N}a(n)n^{-s}
$$
with $a(n) \in \mathbb{C}$. Let $\mathcal{U}\subset[-T,T]$ be $1$-separated. Then, for $\eta \in (0, 1/2]$,
\begin{equation*}
  \sum_{t\in\mathcal{U}}|A(it)|^2
  \ll_\eta
  \left(N+|\mathcal{U}| \cdot T^{\frac{9}{2} \cdot \eta^{3/2}} N^{1-\eta}\log^{2/3} T\right) \sum_{N < n \leqslant 2N} |a(n)|^2.
\end{equation*}

<!-- proof-3: body_md -->
The proof is a slight variant of the proof of the Hal\'asz--Montgomery mean value theorem (see e.g.~\cite[Theorem 9.6]{IwaniecKowalski}). For a similar variant for prime-supported Dirichlet polynomials, see~\cite[Lemma 4.4]{MRII}.

  By the duality principle (see e.g.~\cite[Section 7.1]{IwaniecKowalski}), it suffices to show that, for any complex coefficients $c(t)$, one has
\begin{equation}\label{eq:sparse-mean-square-discrete2}
\sum_{N < n \leqslant 2N}\left| \sum_{t\in\mathcal{U}}c(t) n^{-it}\right|^2 \ll \left(N+|\mathcal{U}| T^{\frac{9}{2}\eta^{3/2}} N^{1-\eta}\log^{2/3} T\right) \sum_{t \in \mathcal{U}} |c(t)|^2.
\end{equation}
Let $w$ be a smooth non-negative compactly supported function such that $w(x) = 1$ for $x \in [1,2]$. Its Mellin transform
\[
  \widetilde{w}(z)=\int_0^\infty w(x)x^{z-1}\,\textnormal{d} x
\]
satisfies, for $1/2\leqslant\sigma\leqslant2$,
\begin{equation}\label{eq:mellin-decay-sparse}
  \widetilde w(\sigma+it)\ll(1+|t|)^{-10}.
\end{equation}

By inserting this smooth weight $w$, we have
\begin{align} \label{eq:HM1ststep}
\sum_{N < n \leqslant 2N}\left| \sum_{t\in\mathcal{U}}c(t) n^{-it}\right|^2 \leqslant \sum_{n} w\left(\frac{n}{N} \right)\left| \sum_{t\in\mathcal{U}}c(t) n^{-it}\right|^2 = \sum_{t_1, t_2 \in \mathcal{U}} c(t_1)\overline{c(t_2)} \sum_{n} w\left(\frac{n}{N} \right) n^{-i(t_1-t_2)}.
\end{align}
For the inner summation, Mellin inversion on a line $c>1$ gives
\[
w\left(\frac{n}{N} \right) =\frac{1}{2\pi i} \int_{c-i\infty}^{c+i\infty} \widetilde w (z) N^zn^{-z} \,\textnormal{d} z.
\]
Hence
\[
  \sum_{n} w\left(\frac{n}{N} \right) n^{-i(t_1-t_2)} =\frac1{2\pi i}\int_{c-i\infty}^{c+i\infty}\widetilde w(z)N^z\zeta(z+i(t_1-t_2))\,\textnormal{d} z.
\]
Moving the contour to $\operatorname{Re}(z)=1-\eta$ crosses a pole at $z=1+i(t_2-t_1)$. This gives
\[
 \sum_{n} w\left(\frac{n}{N} \right) n^{-i(t_1-t_2)} = \widetilde w(1+i (t_2-t_1)) N^{1+ i(t_2-t_1)} +\frac{1}{2\pi i} \int_{1-\eta-i\infty}^{1-\eta+i\infty} \widetilde w(z)N^z\zeta(z+i(t_1-t_2))\,\textnormal{d} z.
\]
We bound the integral using \eqref{eq:mellin-decay-sparse} and a result of Ford~\cite[Theorem 1]{Ford02}, which for $|t|\geqslant3$ and $\sigma \in [1/2, 1]$ gives
\[
  |\zeta(\sigma+it)| \leqslant 100 (|t|+1)^{\frac{9}{2}(1-\sigma)^{3/2}} \log^{2/3} (|t|+2).
  \]
For $|t|\leqslant3$, we instead use
$|\zeta(1-\eta+it)|\ll_\eta 1$. This leads to
\begin{align*}
 &\left|
 \int_{1-\eta-i\infty}^{1-\eta+i\infty}
 \widetilde w(z)N^z\zeta(z+i(t_1-t_2))\,\textnormal{d} z
 \right|\\
 &\leqslant N^{1-\eta}\int_{-\infty}^{\infty}
 |\widetilde w(1-\eta+iu)|
 |\zeta(1-\eta+i(u+t_1-t_2))|\,\,\textnormal{d} u\\
 &\ll_\eta N^{1-\eta}\int_{-\infty}^{\infty}
 \frac{(|u+t_1-t_2|+1)^{\frac92\eta^{3/2}}
 \log^{2/3}(|u+t_1-t_2|+2)}
 {(1+|u|)^{10}}\,\,\textnormal{d} u\\
 &\ll_\eta N^{1-\eta}
 (1+|t_1|+|t_2|)^{\frac92\eta^{3/2}}
 \log^{2/3}(2+|t_1|+|t_2|).
\end{align*}
Thus, one concludes from~\eqref{eq:HM1ststep} and the above estimates that
\begin{align*}
\sum_{N < n \leqslant 2N}\left| \sum_{t\in\mathcal{U}}c(t) n^{-it}\right|^2 &\ll N \sum_{t_1,t_2\in \mathcal U} |c(t_1) \overline{c(t_2)}| \cdot|\widetilde w(1+i (t_2-t_1))| \\
&+ N^{1-\eta} \sum_{t_1,t_2\in \mathcal U} |c(t_1)\overline{c(t_2)}| (1+|t_1|+|t_2|)^{\frac{9}{2}\eta^{3/2}} \log^{2/3}(2+|t_1|+|t_2|).
\end{align*}
The claim~\eqref{eq:sparse-mean-square-discrete2} now follows from the inequality $|c(t_1)\overline{c(t_2)}| \leqslant (|c(t_1)|^2+|c(t_2)|^2)/2$, the rapid decay of $\widetilde{w}$ (see~\eqref{eq:mellin-decay-sparse}) and the assumption that $\mathcal{U}$ is $1$-separated.

<!-- lemma-8: body_md -->
\label{lem:gowers-gluing}
Let $s \in \mathbb{N}$. Let  $a \in \mathbb{R}$ and $H\geqslant 2L\geqslant 4$, and write $I=(a,a+H]$. Then, for every $f\colon\mathbb Z\to\mathbb C$,
\[
  \|f\|_{U^s(I)}\ll_s\max_{\substack{I'\subseteq I\text{ an interval}\\ L/2<|I'|\leqslant L}}
  \|f\|_{U^s(I')}.
\]
<!-- end lemma-8 -->
<!-- proof-8: body_md -->
Set
\[
  m=\left\lceil\frac{H}{L}\right\rceil,\qquad \ell=\frac{H}{m}.
\]
Then $L/2<\ell\leqslant L$. Partition $I$ into the consecutive intervals
\[
  I_j=(a+j\ell,a+(j+1)\ell],\qquad 0\leqslant j<m,
\]
and write $f_j=f\1_{I_j}$.

For a tuple $\boldsymbol{i}=(i_\omega)_{\omega\in\{0,1\}^s}$, with each $i_\omega\in\{0,\ldots,m-1\}$, set
\[
  \Lambda_{\boldsymbol{i}} =\sum_{x,h_1,\ldots,h_s\in\mathbb Z} \prod_{\omega\in\{0,1\}^s}\mathcal C^{|\omega|}f_{i_\omega}(x+\omega_1h_1+\cdots+\omega_sh_s).
\]
By multilinearity, we have
\begin{equation} \label{eq:f1InormLambdai}
  \|f\1_I\|_{U^s(\mathbb Z)}^{2^s}=\sum_{\boldsymbol{i}}\Lambda_{\boldsymbol{i}}.
\end{equation}

We first count the tuples $\boldsymbol{i}$ for which
$\Lambda_{\boldsymbol{i}}$ can be nonzero. Suppose that such a tuple $\boldsymbol{i}$ is fixed and let $x,h_1,\ldots, h_s$ be such that $\prod_{\omega\in \{0,1\}^s}\mathcal{C}^{|\omega|}f_{i_{\omega}}(x+\omega_1h_1+\cdots+\omega_sh_s)\neq 0$. Writing, for $\omega \in \{0, 1\}^s$,
\[
  n_\omega=x+\omega_1h_1+\cdots+\omega_sh_s,
\]
we have
\[
  n_\omega=(1-|\omega|)n_{\boldsymbol{0}}+\sum_{r=1}^s\omega_r n_{\boldsymbol{e}_r}.
\]
The condition $n_\omega\in I_{i_\omega}$ gives
\[
  i_\omega=(1-|\omega|)i_{\boldsymbol{0}}+\sum_{r=1}^s\omega_r i_{\boldsymbol{e}_r}+O_s(1).
\]
Therefore, once
\[
  i_{\boldsymbol{0}},i_{\boldsymbol{e}_1},\ldots,i_{\boldsymbol{e}_s}
\]
have been chosen, there are only $O_s(1)$ possibilities for every
remaining $i_\omega$. Hence at most $O_s(m^{s+1})$
tuples $\boldsymbol{i}$ contribute.

For each such tuple, the Gowers--Cauchy--Schwarz inequality gives
\begin{equation} \label{eq:afterGCS}
  |\Lambda_{\boldsymbol{i}}|\leqslant \prod_{\omega\in\{0,1\}^s} \|f_{i_\omega}\|_{U^s(\mathbb Z)}.
\end{equation}

Writing
\begin{equation*}
  \eta=
  \max_{\substack{I'\subseteq I\text{ an interval}\\ L/2<|I'|\leqslant L}}
  \|f\|_{U^s(I')},
\end{equation*}
we have
\[
  \|f_j\|_{U^s(\mathbb Z)}\leqslant \eta\|\1_{I_j}\|_{U^s(\mathbb Z)}\ll_s \eta\ell^{(s+1)/2^s}.
\]
Combining this with~\eqref{eq:afterGCS} yields
\[
  |\Lambda_{\boldsymbol{i}}|\ll_s \eta^{2^s}\ell^{s+1}.
\]
Summing over the $O_s(m^{s+1})$ possible tuples $\boldsymbol{i}$ that contribute non-trivially to~\eqref{eq:f1InormLambdai}, we obtain
\begin{equation} \label{eq:F1Iest}
  \|f\1_I\|_{U^s(\mathbb Z)}^{2^s}\ll_s\eta^{2^s}m^{s+1}\ell^{s+1}=\eta^{2^s}H^{s+1}.
\end{equation}
Finally,
\begin{equation} \label{eq:1Iest}
  \|\1_I\|_{U^s(\mathbb Z)}^{2^s}\asymp_s H^{s+1}.
\end{equation}
Dividing~\eqref{eq:F1Iest} by~\eqref{eq:1Iest} and taking $2^s$-th roots proves the claim.

 \left|\sum_{X<n\leqslant X+H}h(n)n^{iT}\right|^*&\gg_{d,D,A,\varepsilon}H\log^{-A_1} X.

We argue by induction on the dimension of $G/\Gamma$. The zero-dimensional case is immediate, since a function with mean zero on a one-point nilmanifold vanishes.

\begin{equation}\label{eq:absolute-mass-f}

\end{equation}
Indeed, for $f=\mu$ this is immediate, while for $f=\Lambda-\Lambda_X^\sharp$ it follows from Shiu's bound and the definition of $\Lambda_X^\sharp$. Let $A_1$ be sufficiently large in terms of $d,D,A,\varepsilon,C_0$ to absorb any logarithmic losses that follow.

 F=\sum_{\xi\in\Xi}F_\xi+O(\log^{-E} X),\qquad |\Xi|+\max_{\xi\in\Xi}\|F_\xi\|_{\mathrm{Lip}}\ll \log^{O_{d,D,A,E}(1)} X,
\]
where $F_\xi$ oscillates with central frequency $\xi$. For each $\xi\in\Xi$, define
\[
 m_\xi=\int_{G/\Gamma}F_\xi,\qquad F_\xi^0=F_\xi-m_\xi.
\]
Then $F_\xi^0$ has mean zero; if $\xi\ne0$, Haar invariance gives $m_\xi=0$, while the zero-frequency component factors through the quotient by the centre. The error term is controlled by \eqref{eq:absolute-mass-f}, the constants $m_\xi$ by the major arc estimate of \cite[Theorem~3.1(i)]{MSTT}, and every component invariant under a positive-dimensional central subgroup by the induction hypothesis on the corresponding quotient. \textbf{Case 2: $\beta_1 < \tau_1/2$}.
Define
\[
  F_{-} (t) = R_{\beta_2}(t) + R_{\beta_3}(t) + R_{2\beta_1}(t)/2 -t.
\]
For $t \in [p(\beta_2),r(\beta_2)]$, the slopes of $R_{\beta_2}$, $R_{\beta_3}$, and $R_{2\beta_1}/2$ are at most $1/4$, $1/2$, and $1/4$, respectively, so the slope of $F_-$ is $\leqslant 0$ in this interval. For $t \in [r(\beta_2),\tau_1]$, the first two slopes equal $1/2$ and the slope of $R_{2\beta_1}$ is non-negative, so the slope of $F_-$ is $\geqslant 0$ in this interval. Therefore $F_-$ attains its minimum at $r(\beta_2)$, and the minimum is
\begin{align*}
  F_{-} (r(\beta_2)) &= \frac{r(\beta_2)-\tau_1+\beta_2}{2} + \frac{r(\beta_2)-\tau_1+\beta_3}{2}+\frac{R_{2\beta_1} (r(\beta_2))}{2}-r(\beta_2)\\
  &= \frac12 -\frac{\beta_1}{2}-\tau_1 +\frac{R_{2\beta_1} (r(\beta_2))}{2}.
\end{align*}

Thus it suffices to show that
\begin{align}\label{eq:claim-beta1}
 R_{2\beta_1} (r(\beta_2)) > 2\tau_1+\beta_1-1+2\varepsilon^{10}
\end{align}
Consider this for a fixed $\beta_1 < \tau_1/2$. Now $R_{2\beta_1}(t)$ is increasing in $t$ while $r(\beta_2)$ is decreasing in $\beta_2$. Thus $R_{2\beta_1}(r(\beta_2))$ is minimal when $\beta_2$ is maximal. Since $\beta_2 \leqslant \beta_3$ and $\beta_1 + \beta_2 + \beta_3 = 1$, we have $\beta_2 \leqslant (1-\beta_1)/2$. Thus it suffices to show~\eqref{eq:claim-beta1} for $\beta_2 = (1-\beta_1)/2$. For $\beta_1 < \tau_1/2$, we have $(1-\beta_1)/2 \geqslant 5\tau_1/6$. Thus, by definition of $r(\beta)$ it suffices to show that
\begin{equation} \label{eq:R2bet1bound}
 H(\beta_1) := R_{2\beta_1}\left(\frac{3\tau_1}{2}+\frac{\beta_1-1}{2}\right) - (2\tau_1+\beta_1-1+2\varepsilon^{10}) > 0.
\end{equation} 
Looking carefully at~\eqref{eq:Rbtaulow},~\eqref{eq:Rbtaulowmid},~\eqref{eq:Rbtauupmid},~\eqref{eq:Rbtauup}, we see that the slope of $H(\beta_1)$ is $\leqslant 0$ when the argument of $R_{2\beta_1}$ is $< r(2\beta_1)$ and $\geqslant 0$ when the argument of $R_{2\beta_1}$ is $\geqslant r(2 \beta_1)$. Thus it suffices to consider the case
  \begin{equation} \label{eq:r2beta1eq}
    \frac{3\tau_1}{2}+\frac{\beta_1-1}{2} = r(2\beta_1)
  \end{equation}
  
    If we had $2 \beta_1 \leqslant 5\tau_1/6$, it would follow that
  \[
    r(2\beta_1) = \tau_1 - \frac{4\beta_1}{5}.
  \]
In this case
\[
  \frac{3\tau_1}{2}+\frac{\beta_1-1}{2} - r(2\beta_1) = \frac{\tau_1}{2} -\frac{1}{2} +\frac{13}{10}\beta_1 \leqslant \frac{25}{24}\tau_1 - \frac{1}{2} < 0.
  \]
Thus~\eqref{eq:r2beta1eq} does not have a solution with $2\beta_1 \leqslant 5\tau_1/6$ and $H(\beta_1)$ is non-increasing in this range.

For $2 \beta_1 > 5\tau_1/6$, by the definition of $r(\beta)$,
    \[
    \frac{3\tau_1}{2}+\frac{\beta_1-1}{2} = \frac{3\tau_1}{2} - 2\beta_1 \iff \beta_1 = \frac{1}{5}
  \]
  and
  \[
    R_{2/5}\left(\frac{3\tau_1}{2}+\frac{\frac{1}{5}-1}{2}\right) = \frac{\frac{3\tau_1}{2}-\frac{2}{5}-\tau_1+\frac{2}{5}}{2} = \frac{\tau_1}{4}. 
  \]
  Thus
  \[
    H(1/5) = \frac{\tau_1}{4}-(2\tau_1 + \frac{1}{5} -1 + 2\varepsilon^{10}) = -\frac{7\tau_1}{4} + \frac{4}{5} -2\varepsilon^{10}.
  \]
  Recalling $\tau_1 = 16/35-\varepsilon$, we see that $H(1/5) = \frac{7}{4}\varepsilon-2\varepsilon^{10}> 0$ and the claim follows.

Set $\tau^*=2/5-\varepsilon$ and, for convenience, set $P_\beta(t) =P_{\beta,\tau^\ast}(t)$ and $R_\beta(t) =R_{\beta,\tau^\ast}(t)$. Arguing as in the beginning of the proof of Lemma~\ref{le:16/35comput}, it suffices to show that
  \begin{align} \label{eq:3/5-reduc-tau1}
F(t) := P_{\beta_1}(t) + P_{\beta_2}(t) + P_{\beta_3}(t) -t > \varepsilon^{10}
\end{align}
for all $t\in [\varepsilon^6/2, \tau^\ast]$. Note that, for all $\beta \in \{\beta_1, \beta_2, \beta_3\}$,
\begin{equation} \label{eq:Pbetatlow3/5}
P_\beta(t) \geqslant \max \Big\{R_{\beta} (t), \frac{R_{2\beta} (t)}{2}\Big\}.
\end{equation}
We may assume that $\beta_2\leqslant\beta_3$. Since
$\beta_2\geqslant1/5$ and $\beta_1\geqslant2/5$, we have
$\beta_1,2\beta_2\geqslant\tau^*$. We therefore deduce from~\eqref{eq:Pbetatlow3/5} and the definition
of $R_\beta(t)$ that 
\[
P_{\beta_1} (t) \geqslant R_{\beta_1}(t)=\frac{t}{2} \qquad \text{ and }\qquad P_{\beta_2}(t) \geqslant \frac{R_{2\beta_2} (t)}{2}=\frac{t}{4}.
\]
Therefore,
\[
P_{\beta_1}(t) + P_{\beta_2} (t) + P_{\beta_3} (t) -t \geqslant P_{\beta_3} (t) -\frac{t}{4}\geqslant R_{\beta_3}(t)-\frac{t}{4}.
\]
The remaining task is to prove that for all $t\in [\varepsilon^6/2, \tau^*]$ we have
\begin{align} \label{eq:function-T}
R_{\beta_3}(t)-\frac{t}{4} > \varepsilon^{10}.
\end{align}
First of all, we notice that
\begin{equation} \label{eq:beta3simplow}
  \beta_3\geqslant (1-\beta_1)/2 \geqslant 1/4\geqslant \tau^*/2.
\end{equation}
If $R_{\beta_3}(t)=t/2$, then~\eqref{eq:function-T} is immediate. In particular we can concentrate on the range $\beta_3 \in [1/4, \tau^\ast].$ Looking at the slopes of $R_{\beta}(t)$ in~\eqref{eq:Rbtaulowmid} and~\eqref{eq:Rbtauupmid}, we see that the left-hand side of~\eqref{eq:function-T} is minimal when $t = r(\beta_3)$ with
\begin{align}\label{eq:return-slope2/5}
 r(\beta) = 
 \begin{cases}
   \tau^\ast -2\beta/5, \qquad &\text{ if } \beta \leqslant 5\tau^\ast/6;\\
 	3\tau^\ast/2 -\beta, &\text{ if } 5\tau^\ast/6 \leqslant \beta \leqslant \tau^\ast,
 \end{cases}	
\end{align}
and
\[
  R_{\beta_3}(r(\beta_3))-\frac{r(\beta_3)}{4} = \frac{r(\beta_3)-\tau^\ast+\beta_3}{2}-\frac{r(\beta_3)}{4} = \frac{r(\beta_3)-2\tau^\ast+2\beta_3}{4}.
  \]
Now, for $\beta_3 \in [5\tau^\ast/6, \tau^\ast]$, we obtain
\[
  R_{\beta_3}(r(\beta_3))-\frac{r(\beta_3)}{4} = \frac{\beta_3 -\tau^\ast/2}{4} \geqslant \frac{\tau^\ast}{12} > \varepsilon^{10}.
\]
and for $\beta_3 \in [1/4, 5\tau^\ast/6]$, we obtain
\[
R_{\beta_3}(r(\beta_3))-\frac{r(\beta_3)}{4} =\frac{-\tau^* +8\beta_3/5}{4} \geqslant \frac{\varepsilon}{4} >\varepsilon^{10},
\]
which completes the proof.
