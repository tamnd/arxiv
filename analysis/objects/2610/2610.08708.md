<!-- theorem-1: body_md -->
\label{thm:pointwise-anticoncentration}
Let $F$ be a multilinear polynomial of degree $d \geqslant 1$ with complex coefficients. Then, for some absolute constant $C$,
\begin{equation*}
    \mathbb{P}\big[F(\xi_1, \ldots, \xi_n) = 0\big] \leqslant \frac{C 2^d \sqrt{d}}{\sqrt{r(F)}}\,.
\end{equation*}
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:small-ball-anticoncentration}
Let $F$ be a multilinear polynomial of degree $d \geqslant 1$ with complex coefficients. Then, for every $\gamma > 0$ and some absolute constant $C$,
\begin{equation*}
    \mathbb{P}\big[|F(\xi_1, \ldots, \xi_n)| \leqslant \gamma\big] \leqslant \frac{C 2^d \sqrt{d}}{\sqrt{r_{\gamma}(F)}}\,.
\end{equation*}

Naturally, one can ask about analogues of \cref{thm:pointwise-anticoncentration,thm:small-ball-anticoncentration} for polynomials of i.i.d.\ random variables that are not Rademacher. When these random variables are ``not too close to being constant'', one can reduce this question to the Rademacher case via a conditioning argument analogous to \cite[Section 11]{kwan-sauermann}.

<!-- theorem-3: body_md -->
\label{thm:pointwise-anticoncentration-robust}
Let $F$ be a multilinear polynomial of degree $d \geqslant 1$ with complex coefficients. Then, for some absolute constant $C_1$,
\begin{equation*}
    \mathbb{P}\big[F(\xi_1, \ldots, \xi_n) = 0\big] \leqslant \frac{C_1 2^d d}{\sqrt{s(F)}}\,.
\end{equation*}

<!-- remark-2: body_md -->
The polynomial Littlewood--Offord problem has found a number of applications to other questions in combinatorics and computer science, and hence our results have immediate implications for some of them. We briefly mention two such applications:
\begin{itemize}[leftmargin=2em]
    \item \emph{Polynomial approximations of the parity function.} Razborov and Viola~\cite{razborov-viola} proposed a ``challenge'' to construct an \emph{explicit} Boolean function that cannot be approximated by polynomials of degree at most $\log_2 n$. They suggested the parity function as a candidate, and proved that $\mathbb{P}[F(\xi_1, \ldots, \xi_n) = \prod_{i=1}^n \xi_i] \leqslant 1/2$ for every polynomial $F$ of degree at most $\frac{1}{2} \log_2 \log_2 n$ and sufficiently large $n$. Meka, O.\ Nguyen, and Vu~\cite{meka-nguyen-vu} extended this result to polynomials of degree at most $\frac{\log n}{15 \log \log n}$. Plugging the bound given by \cref{thm:pointwise-anticoncentration} into the approach of Razborov and Viola extends it further to polynomials of degree at most $c \log_2 n$ for any fixed $c < 1/2$: indeed, if $r(F) \geqslant n/\log_2 n$ then the desired bound follows from \cref{thm:pointwise-anticoncentration} applied to $F-1$ and $F+1$; otherwise, we can condition on the outcomes of $n/2$ variables to reduce the degree of our polynomial and iterate the argument.
    \item \emph{Edge-statistics in large hypergraphs.} The connection between this topic and the polynomial Littlewood--Offord problem was first observed by Kwan, Sudakov, and Tran~\cite{kwan-sudakov-tran}, and later used in \cite{grebennikov-kwan-edge-statistics,jain-kwan-mubayi-tran,kwan-sauermann}. Using this connection, Jain, Kwan, Mubayi, and Tran~\cite{jain-kwan-mubayi-tran} proved that the fraction of $k$-vertex subsets in a large $r$-uniform hypergraph that induce exactly $\ell \leqslant \frac{1}{2}\binom{k}{r}$ edges is at most $(\ell/k^{r-1})^{-1/2 + o(1)}$. Our results remove the ``$o(1)$-loss'', yielding the optimal bound of $O_r((\ell/k^{r-1})^{-1/2})$.
\end{itemize}

\label{lemma:linear-algebra}
    Let $\mathcal{L}$ be a family of subsets of $[n]$, and denote $V_{\mathcal{L}} = \operatorname{span}_{\mathbb{R}}\{x^S : S \in \mathcal{L}\}$. For a function $f: \{-1, 1\}^n \to \mathbb{R}$, also define the vector space $$K_{\mathcal{L}}(f) = \{h \in V_{\mathcal{L}} : fh \in V_{\mathcal{L}}\}\,.$$ Then,
    \begin{equation*}
    \sum_{\substack{S_1 \notin \mathcal{L} \\ S_2 \in \mathcal{L}}} \hat{f}(S_1 \triangle S_2)^2 \leqslant \big(|\mathcal{L}| - \dim K_{\mathcal{L}}(f)\big) \cdot \|f\|^2_{\infty}\,.
    \end{equation*}
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
Consider the linear map from the vector space of functions $\{-1, 1\}^n \to \mathbb{R}$ to itself given by $\varphi \mapsto f \varphi$.
    Let $A$ be the matrix of this map in the basis $(x^S)_{S \subseteq [n]}$ (with rows and columns indexed by subsets of $[n]$). Then, the entries of $A$ can be expressed in terms of the Fourier coefficients of $f$ as 
    \begin{equation*}
    A_{S_1, S_2} = \mathbb{E}\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left[f(\xi) \xi^{S_1} \cdot \xi^{S_2}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right] = \hat{f}(S_1 \triangle S_2)\,.
    \end{equation*}
    Denote $\mathcal{L}^c = \{S \subseteq [n]: S \notin \mathcal{L}\}$, and let $B = A[\mathcal{L}^c, \mathcal{L}]$ be the submatrix of $A$ with rows indexed by $\mathcal{L}^c$ and columns indexed by $\mathcal{L}$. Let $\sigma_1 \geqslant \ldots \geqslant \sigma_{\operatorname{rank}(B)} > 0$ be the nonzero singular values of $B$. Using that the largest singular value equals the operator norm $\|B\|$, we conclude that 
    \begin{equation} \label{eq:sum-of-squares}
    \sum_{\substack{S_1 \notin \mathcal{L} \\ S_2 \in \mathcal{L}}} \hat{f}(S_1 \triangle S_2)^2 = \sum_{\substack{S_1 \notin \mathcal{L} \\ S_2 \in \mathcal{L}}} B_{S_1, S_2}^2 = \operatorname{tr}(B^T B) = \sum_{i=1}^{\operatorname{rank}(B)} \sigma_i^2 \leqslant \operatorname{rank}(B) \cdot \|B\|^2.
    \end{equation}
    It remains to bound the rank and the operator norm of $B$. Indeed, since $B$ is a submatrix of $A$, 
    \begin{equation} \label{eq:norm-of-B}
    \|B\| \leqslant \|A\| = \sup_{\|\varphi\|_2 = 1} \|f \varphi\|_2 = \|f\|_{\infty}\,,
    \end{equation}
    and, since the kernel of $B$ can be identified with the set of functions $h \in V_{\mathcal{L}}$ such that $fh \in V_{\mathcal{L}}$,
    \begin{equation} \label{eq:rank-of-B}
    \operatorname{rank}(B) = |\mathcal{L}| - \dim \ker(B) = |\mathcal{L}| - \dim K_{\mathcal{L}}(f)\,.
    \end{equation}
    Substituting \cref{eq:norm-of-B,eq:rank-of-B} into \cref{eq:sum-of-squares} gives the desired bound.

<!-- proof-2: body_md -->
By rescaling, we may assume that $\|f\|_{\infty} = 1$. For an integer $k$, denote 
    $$\mathcal{L}_k = \{S \subseteq [n]: |S| \leqslant k\}, \qquad V_k = \operatorname{span}_{\mathbb{R}}\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left\{x^S : S \in \mathcal{L}_k\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right\}, \qquad K_k(f) = \{h \in V_k : fh \in V_k\}\,.$$
    For each $0 \leqslant k \leqslant n-1$, \cref{lemma:linear-algebra} gives 
    \begin{equation}  \label{eq:apply-linear-algebra}
    \sum_{\substack{|S_1| > k \\ |S_2| \leqslant k}} \hat{f}(S_1 \triangle S_2)^2 \leqslant |\mathcal{L}_k| - \dim K_k(f).
    \end{equation}
    Clearly, $\dim V_k = |\mathcal{L}_k| = \sum_{\ell = 0}^k \binom{n}{\ell}$. To get a lower bound on the dimension of $K_k(f)$, we recall that $f = p/q$ where $p$ and $q$ are polynomials of degree at most $d$. Since $q(x) \neq 0$ for each $x \in \{-1, 1\}^n$, multiplication by $q$ is invertible, and thus the subspace $qV_{k-d} = \{qh : h \in V_{k-d}\} \subseteq V_k$ has the same dimension as $V_{k-d}$. Furthermore, 
    $$fqV_{k-d} = \{fqh : h \in V_{k-d}\} = \{ph : h \in V_{k-d}\} \subseteq V_k\,,$$
    and therefore 
    \begin{equation*}
    \dim K_k(f) \geqslant \dim (qV_{k-d}) = \dim V_{k-d} = \sum_{0 \leqslant \ell \leqslant k-d} \binom{n}{\ell}.
    \end{equation*}
    Substituting this into \cref{eq:apply-linear-algebra}, summing over $k = 0, 1, \ldots, n-1$, and dividing by $2^n$ yields
    \begin{equation} \label{eq:sum-over-k}
    \frac{1}{2^n}\sum_{k = 0}^{n-1} \sum_{\substack{|S_1| > k \\ |S_2| \leqslant k}} \hat{f}(S_1 \triangle S_2)^2 \leqslant \frac{1}{2^n}\sum_{k = 0}^{n-1} \sum_{\ell = \max(k-d+1,0)}^k \binom{n}{\ell} \leqslant \frac{d}{2^n} \sum_{\ell = 0}^n \binom{n}{\ell} = d\,.
    \end{equation}
    By exchanging the order of summation, we can rewrite the left-hand side of \cref{eq:sum-over-k} as
    \begin{equation} \begin{aligned} \label{eq:exchange-sum}
    \frac{1}{2^n}\sum_{k = 0}^{n-1} \sum_{\substack{|S_1| > k \\ |S_2| \leqslant k}} \hat{f}(S_1 \triangle S_2)^2 &= \sum_{S \subseteq [n]} \mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(\frac{1}{2^n} \sum_{T \subseteq [n]} \max\Big\{|T| - |S \triangle T|, 0\Big\}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right) \hat{f}(S)^2\,.
    \end{aligned} \end{equation}
    The coefficient of $\hat{f}(S)^2$ in the last sum is the expected value of $\max\big\{|T| - |S \triangle T|, 0\big\}$ for a uniformly random $T \subseteq [n]$. Writing $T = \{i \in [n]: \xi_i = 1\}$, we have 
    \begin{equation*}
    \mathbb{E}\max\Big\{|T| - |S \triangle T|, 0\Big\} = \mathbb{E}\max\Big\{\sum_{i \in S} \xi_i, 0\Big\} = \lambda_{|S|}\,.
    \end{equation*}
    Combined with \cref{eq:sum-over-k} and \cref{eq:exchange-sum}, this gives the desired bound.

If $\mathbb{P}\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left[|G(\xi_1, \ldots, \xi_d)| \leqslant \tfrac{1}{2}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right] = 0$, then we are done. Otherwise, there is a point $x_- \in \{-1, 1\}^d$ such that $|G(x_-)| \leqslant \tfrac{1}{2}$. In this case, we will show that 
    \begin{equation} \label{eq:sb-i-0}
    \frac{64 \cdot 2^d}{d+1} \operatorname{Inf}[g] \geqslant 1 \geqslant \mathbb{P}\big[|G(\xi_1, \ldots, \xi_d)| \leqslant \tfrac{1}{2}\big].
    \end{equation}
    Since 
    \begin{equation*}
    \max_{x \in \{-1, 1\}^d} |G(x)| \geqslant \mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left|\mathbb{E}\big[G(\xi_1, \ldots, \xi_d) \xi_1 \cdots \xi_d\big]\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right| = |a_{[d]}| \geqslant 1\,,
    \end{equation*}
    there also exists a point $x_+ \in \{-1, 1\}^d$ such that $|G(x_+)| \geqslant 1$. So, 
    \begin{equation} \label{eq:sb-i-1}
    |g(x_-) - g(x_+)| = \mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left|\frac{1}{1 + |G(x_-)|^2} - \frac{1}{1 + |G(x_+)|^2}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right| \geqslant \frac{4}{5} - \frac{1}{2} > \frac{1}{4}.
    \end{equation}
    On the other hand, by Cauchy--Schwarz and \cref{fact:influence-fourier},
    \begin{equation}\begin{aligned} \label{eq:sb-i-2}
    (g(x_-) - g(x_+))^2 &= \mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(\sum_{S \subseteq [d]} \hat{g}(S) (x_-^S - x_+^S)\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right)^2 \leqslant \mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(\sum_{\substack{S \subseteq [d],S \neq \emptyset}} |S| \hat{g}(S)^2\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right) \mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(\sum_{\substack{S \subseteq [d], S \neq \emptyset}} \frac{(x_-^S - x_+^S)^2}{|S|}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right) \\
    &\leqslant \frac{1}{4} \operatorname{Inf}[g] \cdot 4 \sum_{\substack{S \subseteq [d] \\ S \neq \emptyset}} \frac{1}{|S|} = \operatorname{Inf}[g] \cdot \sum_{k = 1}^d \frac{1}{k} \binom{d}{k}\,. 
    \end{aligned} \end{equation}
    Finally, since $\tfrac{1}{k+1} \binom{d}{k} = \frac{1}{d+1} \binom{d+1}{k+1}$, we have
    \begin{equation} \label{eq:sb-i-3}
    \sum_{k = 1}^d \frac{1}{k} \binom{d}{k} \leqslant 2 \sum_{k = 1}^d \frac{1}{k+1} \binom{d}{k} = \frac{2}{d+1} \sum_{k = 1}^d \binom{d+1}{k+1} \leqslant \frac{4 \cdot 2^d}{d+1}\,.
    \end{equation}
    Together, \cref{eq:sb-i-1,eq:sb-i-2,eq:sb-i-3} imply \cref{eq:sb-i-0}, completing the proof.

By rescaling, we may assume that $\gamma = 1$. By the definition of $r = r_{1}(F)$, there exist pairwise disjoint sets $S_1, \ldots, S_r \subseteq [n]$ of size $d$ such that $|a_{S_i}| \geqslant 1$ for all $i \in [r]$. By conditioning on the outcomes of $\xi_j$ for all $j \notin S_1 \cup \ldots \cup S_r$, we may also assume that $r = n/d$.  
    
    Our goal is to bound the probability that $|F(\xi_1, \ldots, \xi_n)| \leqslant 1$. In fact, it is sufficient to show that
    \begin{equation} \label{eq:smaller-ball}
    \mathbb{P}\big[|F(\xi_1, \ldots, \xi_n)| \leqslant \tfrac{1}{2}\big] = O\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(\frac{2^d d}{\sqrt{n}}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right).
    \end{equation}
    Indeed, one can cover the disk of radius $1$ by a constant number of disks of radius $\tfrac{1}{2}$ (say, centered at the points $z_1, \ldots, z_{M} \in \mathbb{C}$), and then apply \cref{eq:smaller-ball} to the polynomials $F - z_1, \ldots, F - z_{M}$.

    Denote $f = \frac{1}{1 + |F|^2}$. Since $|F|^2 = F \cdot \overline{F}$ is a polynomial of degree at most $2d$ with real coefficients, we can apply \cref{thm:influence} to $f$ and obtain that 
    \begin{equation} \label{eq:small-ball-1}
    \operatorname{Inf}[f] = O\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(d \sqrt{n}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right).
    \end{equation}
    For each $i \in [r]$, let $F_i$ be the (random) polynomial in the variables $(x_j)_{j \in S_i}$ obtained from $F$ by fixing the values of $x_j$ to $\xi_j$ for all $j \notin S_i$, and denote $f_i = \frac{1}{1 + |F_i|^2}$. Since $|a_{S_i}| \geqslant 1$ and the coefficient of $x^{S_i}$ in $F_i$ is exactly $a_{S_i}$, \cref{lemma:anticoncentration-influence} yields
    \begin{equation} \label{eq:small-ball-2}
    \mathbb{P}\big[|F(\xi_1, \ldots, \xi_n)| \leqslant \tfrac{1}{2}\big] = \mathbb{E}_i\Big[\mathbb{P}\big[|F_i((\xi_j)_{j \in S_i})| \leqslant \tfrac{1}{2}\big]\Big] = O\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(\frac{2^d}{d} \mathbb{E}_i\big[\operatorname{Inf}[f_i]\big]\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right),
    \end{equation}
    where $\mathbb{E}_i$ denotes the expectation over the random variables $(\xi_j)_{j \notin S_i}$. Finally, from the definition of total influence, we have $\operatorname{Inf}[f] = \sum_{i=1}^r \mathbb{E}_i[\operatorname{Inf}[f_i]]$ with $r=n/d$. Together with \cref{eq:small-ball-1} and \cref{eq:small-ball-2}, this implies that
    \begin{equation*}
        \frac{n}{d} \cdot \mathbb{P}\big[|F(\xi_1, \ldots, \xi_n)| \leqslant \tfrac{1}{2}\big] = O\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(2^d \sqrt{n}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right),
    \end{equation*}
    which is exactly the desired bound \cref{eq:smaller-ball}.

We argue by induction on $d$ that the desired bound holds with $C_1 = \sqrt{2} C$, where $C$ is the constant in \cref{thm:pointwise-anticoncentration}. If $r(F) \geqslant \frac{1}{2d} s(F)$, then this bound follows from \cref{thm:pointwise-anticoncentration}. Otherwise, let $S_1, \ldots, S_{r(F)}$ be pairwise disjoint sets of size $d$ such that $a_{S_i} \neq 0$ for all $i$. Then, by the definition of $r(F)$, $a_S = 0$ for all sets $S \subseteq [n]$ of size $d$ that do not intersect the set $I = S_1 \cup \ldots \cup S_{r(F)}$. Condition on an arbitrary outcome of the random variables $(\xi_i)_{i \in I}$, and note that the resulting polynomial $F_0$ in the remaining variables $(x_i)_{i \notin I}$ has degree at most $d-1$. It suffices to show that the probability (with respect to the randomness of $(\xi_i)_{i \notin I}$) that $F_0((\xi_i)_{i \notin I}) = 0$ is at most $\frac{C_1 2^d d}{\sqrt{s(F)}}$. We distinguish the following two cases:

    \textbf{Case 1:} $F_0$ is a constant polynomial. Since $F_0$ is obtained from $F$ by fixing the values of $|I| = d \cdot r(F) < \frac{1}{2}s(F) < s(F)$ variables, this constant is not zero, and thus 
    \begin{equation*}
    \mathbb{P}\big[F_0((\xi_i)_{i \notin I}) = 0\big] = 0\,.
    \end{equation*}
    
    \textbf{Case 2:} $1 \leqslant \deg(F_0) \leqslant d-1$. Again, since $F_0$ is obtained from $F$ by fixing the values of $|I|$ variables, we have $s(F_0) \geqslant s(F) - |I| > \frac{1}{2}s(F)$. Therefore, by the induction hypothesis,
    \begin{equation*}
    \mathbb{P}\big[F_0((\xi_i)_{i \notin I}) = 0\big] \leqslant \frac{C_1 2^{d-1} (d-1)}{\sqrt{s(F_0)}} \leqslant \frac{C_1 2^d d}{\sqrt{s(F)}}\,. \qedhere
    \end{equation*}
