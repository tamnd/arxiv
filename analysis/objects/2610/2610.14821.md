For Shannon entropy,
\begin{align*}
\mathcal{H}^S(v)=\frac{1}{2}\ln(2\pi ev),\qquad
f^S(v)=\frac{v}{2},\qquad
p^\xi(v)=\frac{2\kappa}{v}. \refstepcounter{equation}\tag{\theequation}\label{shannon_entropy_specification}
\end{align*}
Thus, in the Shannon benchmark, auxiliary precision is inversely proportional to $v$.

For the Tsallis entropy with $\rho>1$,\footnote{Generalized models of rational inattention likewise use alternative entropy-based costs, including Tsallis entropy, to capture different information-processing frictions \citep{caplin2022rationally}.}
\begin{align*}
\mathcal{H}^{\rho}(v)&=\frac{1}{\rho-1}\left[1-\rho^{-1/2}(2\pi v)^{(1-\rho)/2}\right],\\
f^{\rho}(v)&=\frac{1}{2}\rho^{-1/2}(2\pi)^{(1-\rho)/2}v^{(3-\rho)/2},\\
p^\xi(v)&=2\kappa\rho^{1/2}(2\pi)^{(\rho-1)/2}v^{(\rho-3)/2}. \refstepcounter{equation}\tag{\theequation}\label{tsallis_entropy_specification}
\end{align*}
The Shannon specification is recovered as $\rho\to1$. Economically, $\rho$ determines how a fixed attention capacity translates into the precision of the auxiliary signal as posterior uncertainty changes: under full capacity, $p^\xi(v)$ increases as $v$ decreases for $1<\rho<3$, is independent of $v$ for $\rho=3$, and increases with $v$ for $\rho>3$. 
Both specifications satisfy \eqref{entropy_regular}.

\label{proposition_31}
 Let $(t,x,m,v)\in[0,T)\times\mathcal{O}$ and $\kappa\in[0,\infty)$ be fixed. Then the attentive investor's optimal trading strategy is given by
\begin{align*}
\pi_s^{\ast,AI}=e^{-r(T-s)}\frac{m_s^{AI}-r}{\gamma\sigma^2}\frac{\sigma^2}{\sigma^2+(T-s)v_s^{AI}},\qquad s\in[t,T]. \refstepcounter{equation}\tag{\theequation}\label{optimal_strategy}
\end{align*}
Moreover, the corresponding value function is given by
\begin{align*}
V^{AI}(t,x,m,v;\kappa)=-\frac{1}{\gamma}\exp\left(-\gamma e^{r\left( T-t \right)}x-a(t,v)(m-r)^2-b^{AI}(t,v;\kappa)\right), \refstepcounter{equation}\tag{\theequation}\label{value_function}
\end{align*}
where
\begin{align*}
a(t,v)&=\frac{T-t}{2\left( \sigma^2+(T-t)v\right)}, \refstepcounter{equation}\tag{\theequation}\label{a_explicit}\\
b^{AI}(t,v;\kappa)&=\int_t^T
\left(\frac{(v_s^{AI})^2}{\sigma^2}
+\frac{\kappa(v_s^{AI})^2}{f(v_s^{AI})}\right)
a(s,v_s^{AI})ds. \refstepcounter{equation}\tag{\theequation}\label{b_integral}
\end{align*}

<!-- theorem-1: body_md -->
\label{theorem_42}
Let $(t,x,m,v)\in[0,T)\times\mathcal{O}$, $\mu\in\mathbb{R}$, and $\kappa\in[0,\infty)$ be fixed. Set
\begin{align*} \lambda_s^\kappa:=q_s^{AI}v_s^{AI}\left(\frac{1}{\sigma^2}+\frac{\kappa}{f(v_s^{AI})}\right),\qquad q_s^{AI}&:=\frac{\sigma^2}{\sigma^2+(T-s)v_s^{AI}}, \qquad s\in[t,T]. \refstepcounter{equation}\tag{\theequation}\label{q_ai}
\end{align*}
Let $B(s;\kappa)$ and $C(s;\kappa)$ be the solutions of the terminal-value ODE system
\begin{align*}
\frac{d}{ds}B(s;\kappa)&=-2\lambda_s^\kappa\left( a(s,0)-B(s;\kappa)\right)+2q_s^{AI}v_s^{AI}\lambda_s^\kappa\left( a(s,0)-B(s;\kappa)\right)^2,\qquad B(T;\kappa)=0,\\
\frac{d}{ds}C(s;\kappa)&=-q_s^{AI}v_s^{AI}\lambda_s^\kappa\left( a(s,0)-B(s;\kappa)\right),\qquad C(T;\kappa)=0. \refstepcounter{equation}\tag{\theequation}\label{ode_system_correction_fluctuation}
\end{align*}
Then $B(t;\kappa)\in(0,a(t,0))$, $C(t;\kappa)\in(0,\frac{1}{2})$, and the scenario-conditional total value of attention satisfies
\begin{align*}
P^{AI}_{SC}(\mu,\kappa)=\frac{e^{-r(T-t)}}{\gamma}\left[\underbrace{B(t;\kappa)\left(\Delta^{NI}\right)^2}_{\text{correction benefit}}-\underbrace{C(t;\kappa)}_{\text{fluctuation cost}}\right]. \refstepcounter{equation}\tag{\theequation}\label{correction_fluctuation_decomposition}
\end{align*}
Consequently,
\begin{align*}
P^{AI}_{SC}(\mu,\kappa)\gtreqqless0\Longleftrightarrow B(t;\kappa)\left(\Delta^{NI}\right)^2\gtreqqless C(t;\kappa). \refstepcounter{equation}\tag{\theequation}\label{general_scenario_performance_condition}
\end{align*}

<!-- proof-10: body_md -->
Write $v_s^\kappa:=v_s^{\mathrm{AI}}$ and $q_s^\kappa:=q_s^{\mathrm{AI}}$, and define
\begin{align*}
A_\kappa&:=a(t,0)-B(t;\kappa),\\
E_\kappa&:=\frac{1}{2\sigma^2}\int_t^T q_s^\kappa v_s^\kappa ds,\\
w_0&:=q_t^{\mathrm{NI}}v>0.
\end{align*}
By \eqref{b_capacity_representation} and \eqref{b_ai_infty},
\begin{align*}
E_\kappa=b^{\mathrm{AI},\infty}(t,v)-b^{\mathrm{AI}}(t,v;\kappa).
\end{align*}
Moreover, \eqref{ex_ante_attention_value_aggreagated} and the identity
\begin{align*}
1+2vB(t;\kappa)=\left( 1+\frac{(T-t)v}{\sigma^2}\right)(1-2w_0A_\kappa)
\end{align*}
yield
\begin{align*}
E_\kappa=C(t;\kappa)-\frac{1}{2}\ln(1-2w_0A_\kappa).
\end{align*}
Since $0<A_\kappa<a(t,0)$ and $C(t;\kappa)>0$ by Theorem \ref{theorem_42}, we obtain
\begin{align*}
0<C(t;\kappa)&\le E_\kappa,\\
0<A_\kappa&\le\frac{1-e^{-2E_\kappa}}{2w_0} \le\frac{E_\kappa}{w_0}.
\end{align*}
By \eqref{variance_perfect_learning}, $v_s^\kappa\to0$ as $\kappa\to\infty$ for every $s>t$. Furthermore,
\begin{align*}
0<q_s^\kappa v_s^\kappa\le v, \qquad s\in[t,T].
\end{align*}
Dominated convergence therefore implies $E_\kappa\to0$. The preceding bounds give
\begin{align*}
A_\kappa\to0, \qquad C(t;\kappa)\to0,
\end{align*}
which proves both assertions.

<!-- lemma-2: body_md -->
\label{lemma_a3_aux}
Let $(t,x,m,v)\in[0,T)\times\mathcal{O}$, $\mu\in\mathbb{R}$, and $\kappa\in[0,\infty)$ be fixed. Let $\widetilde{\mathbb{P}}^{\mu}$ be the probability measure satisfying
\begin{align*}
\left.\frac{d\widetilde{\mathbb{P}}^{\mu}}{d\mathbb{P}^\mu}\right|_{\mathcal{F}^{AI}_T}
=\exp\left(-\int_t^T\frac{q_s^{AI}(m_s^{AI}-r)}{\sigma}dW_s^1-\frac{1}{2}\int_t^T\left(\frac{q_s^{AI}(m_s^{AI}-r)}{\sigma}\right)^2ds\right). \refstepcounter{equation}\tag{\theequation}\label{tilted_measure}
\end{align*}
Then the attentive investor's scenario-conditional expected utility satisfies
\begin{align*}
J^{AI}(t,x,m,v;\mu,\kappa)=V^{FI}(t,x;\mu)\widetilde{\mathbb{E}}^{\mu}_{t,x,m,v}\left[\exp\left(\frac{1}{2\sigma^2}\int_t^T\left(\Delta_s^{AI}\right)^2ds\right)\right], \refstepcounter{equation}\tag{\theequation}\label{j_ai_presentation}
\end{align*}
where 
\begin{align*}
    \Delta_s^{AI}&:=q_s^{AI}(m_s^{AI}-r)-(\mu-r), \qquad s\in[t,T]. \refstepcounter{equation}\tag{\theequation}\label{delta_ai}
\end{align*}

<!-- proof-11: body_md -->
For fixed $\kappa$, set
\begin{align*}
h_s:=\frac{q_s^{AI}(m_s^{AI}-r)}{\sigma},
\qquad
\theta:=\frac{\mu-r}{\sigma}.
\end{align*}
Because $h$ is affine in the linear-Gaussian state with bounded deterministic coefficients, the Bene\v{s} linear-growth criterion \citep{klebaner2014stochastic} shows that the stochastic exponential in \eqref{tilted_measure} is a true martingale. Under $\widetilde{\mathbb{P}}^{\mu}$, completing the square gives
\begin{align*}
J^{AI}
={}&-\frac{1}{\gamma}e^{-\gamma e^{r(T-t)}x-\frac{1}{2}\theta^2(T-t)}
\widetilde{\mathbb{E}}^{\mu}_{t,x,m,v}
\left[\exp\left(\frac{1}{2}\int_t^T(h_s-\theta)^2ds\right)\right]\\
={}&V^{FI}(t,x;\mu)
\widetilde{\mathbb{E}}^{\mu}_{t,x,m,v}
\left[\exp\left(\frac{1}{2\sigma^2}
\int_t^T\left(\Delta_s^{AI}\right)^2ds\right)\right],
\end{align*}
which proves \eqref{j_ai_presentation}. It remains to verify finiteness of the exponential moment in \eqref{j_ai_presentation}. Under each $\mathbb{P}^{\vartheta}$, $m^{AI}$ is a Gaussian process whose covariance is independent of $\vartheta$. The parameter $\vartheta$ changes only its mean. Thus, by \eqref{scenario_conditional_wealth} and \eqref{optimal_strategy}, 
$\mathbb{E}^{\vartheta}_{t,x,m,v}[\exp(-\gamma X_T^{\pi^{*,AI}})]$
is a Gaussian quadratic exponential with a common quadratic kernel. Its prior average equals the finite Bayesian moment $-\gamma V^{AI}(t,x,m,v;\kappa)$, so it is finite for almost every $\vartheta$ under the prior. Since Gaussian quadratic integrability depends only on the common quadratic kernel, it is finite for every $\vartheta$, in particular for $\mu$. Hence $J^{AI}$, and therefore the exponential moment in
\eqref{j_ai_presentation}, is finite.

<!-- proof-12: body_md -->
For use throughout the appendix, we define the total posterior-variance decay rate by
\begin{align*}
G_\kappa(y):=\frac{y^2}{\sigma^2}+\frac{\kappa y^2}{f(y)},\qquad y>0. \refstepcounter{equation}\tag{\theequation}\label{variance_decay_rate}
\end{align*}
Thus, the variance ODE in \eqref{filter_ai_compact} is
$dv_s^{AI}=-G_\kappa(v_s^{AI})ds$. Using
\eqref{wealth_learning_investor} and \eqref{filter_ai_compact}, define for a
smooth function $w=w(s,x,z,y)$ the second-order differential operator
\begin{align*}
\mathcal{L}^\pi w
:={}&\left( rx+\pi(z-r)\right) w_x-G_\kappa(y)w_y
+\frac{1}{2}\pi^2\sigma^2w_{xx}+\pi yw_{xz}
+\frac{1}{2}G_\kappa(y)w_{zz}.
\end{align*}
The dynamic programming principle for \eqref{portfolio_problem} leads to the HJB equation
\begin{align*}
0=W_s+\sup_{\pi\in\mathbb{R}}\mathcal{L}^\pi W,
\qquad W(T,x,z,y)=U(x). \refstepcounter{equation}\tag{\theequation}\label{hjb_equation}
\end{align*}


                                                                                                                                    

For each $(s,y)\in[0,T)\times\mathbb{R}_+$, let $(v_u)_{u\in[s,T]}$ be the
deterministic solution of
\begin{align*}
dv_u=-G_\kappa(v_u)du,\qquad v_s=y. \refstepcounter{equation}\tag{\theequation}\label{variance_characteristic}
\end{align*}
Condition \eqref{entropy_regular} implies that $G_\kappa(u)/u$ is bounded on

&\longrightarrow-\frac{(1-y)v}{\sigma^2},\\
\frac{\widetilde{\mathbb{E}}^{\mu}_{t,x,m,v}
\left[ m_{t+\tau y}^{AI}-m\right]}{\tau}
&\longrightarrow-y\left[\frac{G_\kappa(v)}{v}(m-\mu)
+\frac{v}{\sigma^2}(m-r)\right],\\
\frac{\widetilde{\mathbb{E}}^{\mu}_{t,x,m,v}
\left[\left( m_{t+\tau y}^{AI}-m\right)^2\right]}{\tau}
&\longrightarrow yG_\kappa(v). \refstepcounter{equation}\tag{\theequation}\label{short_horizon_filter_limits}
\end{align*}
Expanding the square in \eqref{delta_ai} and applying dominated
convergence to \eqref{short_horizon_filter_limits} give
\begin{align*}
&\lim_{\tau\downarrow0}\frac{1}{\tau^2}\Bigg\{
\widetilde{\mathbb{E}}^{\mu}_{t,x,m,v}
\left[\int_t^{t+\tau}\left(\Delta_s^{AI}\right)^2ds\right]
-\tau(m-\mu)^2\Bigg\}\\
&=-\frac{2v}{\sigma^2}(m-\mu)(m-r)
-\frac{G_\kappa(v)}{v}(m-\mu)^2+\frac{G_\kappa(v)}{2}. \refstepcounter{equation}\tag{\theequation}\label{integrated_gap_limit}
\end{align*}
Gaussian fourth-moment bounds and local exponential integrability also give
\begin{align*}
\lim_{\tau\downarrow0}\frac{1}{\tau^2}
\operatorname{Var}_{\widetilde{\mathbb{P}}^{\mu}}
\left(\int_t^{t+\tau}\left(\Delta_s^{AI}\right)^2ds\right)&=0,\\
\lim_{\tau\downarrow0}\frac{1}{\tau^2}\Bigg\{
K_\kappa(\tau)-\widetilde{\mathbb{E}}^{\mu}_{t,x,m,v}
\left[\int_t^{t+\tau}\left(\Delta_s^{AI}\right)^2ds\right]\Bigg\}&=0. \refstepcounter{equation}\tag{\theequation}\label{short_horizon_cumulant_control}
\end{align*}
Subtracting \eqref{integrated_gap_limit} at $\kappa=0$ and using
\eqref{variance_decay_rate} and \eqref{short_horizon_cumulant_control} yield
\begin{align*}
\lim_{\tau\downarrow0}\frac{K_\kappa(\tau)-K_0(\tau)}{\tau^2}
=\frac{\kappa v}{2f(v)}\left( v-2(m-\mu)^2\right). \refstepcounter{equation}\tag{\theequation}\label{short_horizon_K_difference}
\end{align*}
Together, \eqref{ce_short_horizon} and \eqref{short_horizon_K_difference} give
\begin{align*}
\lim_{\tau\downarrow0}\frac{P^{AI}_{SC}(\mu,\kappa;\tau)
-P^{PO}_{SC}(\mu;\tau)}{\tau^2}
=-\frac{\kappa v}{4\gamma\sigma^2f(v)}
\left( v-2(m-\mu)^2\right),
\end{align*}
which proves \eqref{short_horizon_ai}.

For the price observer, set
$q:=\frac{\sigma^2}{\sigma^2+\tau v}$ and write $C(q):=C(t;0)$. Then \eqref{correction_fluctuation_decomposition} and \eqref{b_c_po} give
\begin{align*}
P^{PO}_{SC}(\mu;\tau)
=\frac{e^{-r\tau}}{\gamma}
\left[\frac{(1-q)a(t,0)}{2-q}\left(\Delta^{NI}\right)^2-C(q)\right]. \refstepcounter{equation}\tag{\theequation}\label{price_observer_short_horizon_exact}
\end{align*}
The required elementary limits are
\begin{align*}
\lim_{\tau\downarrow0}\frac{1-q}{\tau}
&=\frac{v}{\sigma^2},&
\lim_{\tau\downarrow0}\Delta^{NI}&=m-\mu,&
\lim_{q\uparrow1}\frac{C(q)}{(1-q)^2}&=\frac{1}{4}. \refstepcounter{equation}\tag{\theequation}\label{price_observer_short_horizon_limits}
\end{align*}
Combining \eqref{price_observer_short_horizon_exact}--\eqref{price_observer_short_horizon_limits}
with $a(t,0)=\frac{\tau}{2\sigma^2}$ gives
\begin{align*}
\lim_{\tau\downarrow0}\frac{P^{PO}_{SC}(\mu;\tau)}{\tau^2}
=-\frac{v}{4\gamma\sigma^4}\left( v-2(m-\mu)^2\right),
\end{align*}
which proves \eqref{short_horizon_po}. If
$(m-\mu)^2<\frac{v}{2}$, both limits are strictly negative, so the asserted
ordering holds for all sufficiently small $\tau>0$.

For positive quantities depending on $\kappa$, write $x_\kappa\asymp y_\kappa$ when $x_\kappa/y_\kappa$ is bounded above and bounded away from zero for all sufficiently large $\kappa$. Comparisons involving $u$ are uniform over the stated range of $u$. All comparison constants may depend on the fixed state, benchmark, and model parameters but not on $\kappa$ or $\mu$.

Set $\tau:=T-t$, $u:=s-t\in[0,\tau]$, and suppress fixed initial arguments. The Shannon and Tsallis specifications can be written jointly as
\begin{align*}
f^\rho(y)=c_\rho y^{1-\beta_\rho},\qquad
\beta_\rho:=\frac{\rho-1}{2},\qquad
c_\rho:=\frac{1}{2}\rho^{-1/2}(2\pi)^{(1-\rho)/2},
\end{align*}
where $\rho=1$ gives the Shannon specification. Write $v_\kappa(u):=v_{t+u}^{AI}$ and $q_\kappa(u):=q_{t+u}^{AI}$. The variance equation becomes
\begin{align*}
\frac{d}{du}v_\kappa(u)
=-\frac{v_\kappa(u)^2}{\sigma^2}
-\frac{\kappa}{c_\rho}v_\kappa(u)^{1+\beta_\rho},
\qquad v_\kappa(0)=v.
\refstepcounter{equation}\tag{\theequation}\label{power_variance_ode}
\end{align*}
If $\beta_\rho=0$, scalar comparison gives
\begin{align*}
v_\kappa(u)\asymp\exp\left(-\frac{\kappa u}{c_\rho}\right),
\qquad u\in[0,\tau].
\end{align*}
If $\beta_\rho>0$, differentiating $v_\kappa^{-\beta_\rho}$ gives
\begin{align*}
\frac{d}{du}v_\kappa(u)^{-\beta_\rho}
=\frac{\beta_\rho\kappa}{c_\rho}
+\frac{\beta_\rho}{\sigma^2}v_\kappa(u)^{1-\beta_\rho}.
\end{align*}
The second term is bounded when $\beta_\rho\leq1$ and sublinear in $v_\kappa^{-\beta_\rho}$ when $\beta_\rho>1$. Scalar comparison on $[0,\tau]$ therefore yields
\begin{align*}
v_\kappa(u)\asymp(1+\kappa u)^{-\frac{1}{\beta_\rho}},
\qquad u\in[0,\tau].
\refstepcounter{equation}\tag{\theequation}\label{power_variance_comparison}
\end{align*}
All comparison constants may depend on $(\tau,v,\sigma,\rho)$ but not on $\kappa$ or $u$. Hence, for every $p>0$,
\begin{align*}
\int_0^\tau v_\kappa(u)^pdu
\asymp
\begin{cases}
\kappa^{-1},&\beta_\rho=0\text{ or }0<\beta_\rho<p,\\
\frac{\ln(1+\kappa)}{\kappa},&\beta_\rho=p,\\
\kappa^{-\frac{p}{\beta_\rho}},&\beta_\rho>p.
\end{cases}
\refstepcounter{equation}\tag{\theequation}\label{variance_integral_rates}
\end{align*}
As in the proof of Theorem \ref{theorem_42}, write $A(s;\kappa):=a(s,0)-B(s;\kappa)$, and set
\begin{align*}
w_\kappa(u):=q_\kappa(u)v_\kappa(u),\qquad
\lambda_\kappa(u):=\lambda_{t+u}^\kappa.
\end{align*}
Since $q_t^{NI}\leq q_\kappa(u)\leq1$, $w_\kappa(u)\asymp v_\kappa(u)$ uniformly. The first equation in \eqref{AC_system} and variation of constants give
\begin{align*}
A(t;\kappa)=\frac{1}{2\sigma^2}\int_0^\tau
\exp\left(-2\int_0^u\lambda_\kappa(y)\left[1-w_\kappa(y)A(t+y;\kappa)\right] dy\right) du.
\refstepcounter{equation}\tag{\theequation}\label{A_variation_constants}
\end{align*}
By \eqref{correction_fluctuation_decomposition} and \eqref{p_sc_limit},
\begin{align*}
P_{SC}^{AI,\infty}(\mu)-P_{SC}^{AI}(\mu,\kappa)
=
\frac{e^{-r(T-t)}}{\gamma}
\left[
A(t;\kappa)\left(\Delta^{NI}\right)^2+C(t;\kappa)
\right].
\end{align*}
By Lemma \ref{lemma_a1}, $C(t;\kappa)\to0$ as $\kappa\to\infty$. Moreover, \eqref{BC_positive} implies
\begin{align*}
0\leq\int_0^u w_\kappa(y)\lambda_\kappa(y)A(t+y;\kappa)dy\leq C(t;\kappa)\to0,
\qquad u\in[0,\tau].
\end{align*}
Hence, the nonlinear contribution to the exponent in \eqref{A_variation_constants} converges uniformly to zero. Direct differentiation of
\begin{align*}
\frac{1}{w_\kappa(u)}
=
\frac{1}{v_\kappa(u)}
+
\frac{\tau-u}{\sigma^2}
\end{align*}
and \eqref{power_variance_ode} gives
\begin{align*}
\frac{d}{du}w_\kappa(u)
&=-\frac{\kappa w_\kappa(u)^2}{f^\rho(v_\kappa(u))},\\
\lambda_\kappa(u)
&=\frac{w_\kappa(u)}{\sigma^2}
-\frac{d}{du}\ln w_\kappa(u).
\end{align*}
Consequently,
\begin{align*}
\exp\left(-2\int_0^u\lambda_\kappa(y)dy\right)
=
\left(\frac{w_\kappa(u)}{w_\kappa(0)}\right)^2
\exp\left(-\frac{2}{\sigma^2}\int_0^u w_\kappa(y)dy\right).
\end{align*}
The last exponential is bounded above and bounded away from zero uniformly. Since $w_\kappa(0)=q_t^{NI}v$ is independent of $\kappa$, it follows from \eqref{A_variation_constants} that
\begin{align*}
A(t;\kappa)\asymp\int_0^\tau v_\kappa(u)^2du.
\refstepcounter{equation}\tag{\theequation}\label{A_variance_integral}
\end{align*}

The identity established in the proof of Lemma \ref{lemma_a1} gives
\begin{align*}
\frac{1}{2\sigma^2}\int_0^\tau w_\kappa(u)du
=
C(t;\kappa)
-\frac{1}{2}\ln\left(1-2w_\kappa(0)A(t;\kappa)\right).
\refstepcounter{equation}\tag{\theequation}\label{C_variance_identity}
\end{align*}
The left-hand side has the order of $\int_0^\tau v_\kappa(u)du$, while the logarithmic term has the order of $A(t;\kappa)$. If $\beta_\rho\geq1$, \eqref{variance_integral_rates} and \eqref{A_variance_integral} show that $A(t;\kappa)$ has strictly smaller order than $\int_0^\tau v_\kappa(u)du$. Therefore,
\begin{align*}
C(t;\kappa)\asymp\int_0^\tau v_\kappa(u)du.
\end{align*}

If $0\leq\beta_\rho<1$, restrict the integral representation in \eqref{BC_positive} to
\begin{align*}
u\in\left[\frac{c_1}{\kappa},\frac{c_2}{\kappa}\right]
\end{align*}
for fixed $0<c_1<c_2$. On this interval, scalar comparison gives $w_\kappa(u)$ bounded away from zero, $\lambda_\kappa(u)\asymp\kappa$, and $A(t+u;\kappa)\asymp\kappa^{-1}$. This gives the matching lower bound, while \eqref{C_variance_identity} gives the upper bound. Hence
\begin{align*}
C(t;\kappa)\asymp\int_0^\tau v_\kappa(u)du
\refstepcounter{equation}\tag{\theequation}\label{C_variance_integral}
\end{align*}
for every $\beta_\rho\geq0$.

Applying \eqref{variance_integral_rates} with $p=2$ and $p=1$, and substituting $\beta_\rho=(\rho-1)/2$, gives
\begin{align*}
A(t;\kappa)&\asymp
\begin{cases}
\kappa^{-1},&1\leq\rho<5,\\
\frac{\ln(1+\kappa)}{\kappa},&\rho=5,\\
\kappa^{-4/(\rho-1)},&\rho>5,
\end{cases}\\
C(t;\kappa)&\asymp r_\rho(\kappa).
\refstepcounter{equation}\tag{\theequation}\label{coefficient_high_capacity_rates}
\end{align*}
In particular, $A(t;\kappa)$ is bounded above by a positive multiple of $r_\rho(\kappa)$ for all sufficiently large $\kappa$.

Finally, as $\mu$ ranges over $\mathbb{R}$, $z:=\left(\Delta^{NI}\right)^2$ ranges over $[0,\infty)$. By \eqref{correction_fluctuation_decomposition} and \eqref{p_sc_limit},
\begin{align*}
\sup_{\mu\in\mathbb{R}}
\frac{
P_{SC}^{AI,\infty}(\mu)-P_{SC}^{AI}(\mu,\kappa)
}{
P_{SC}^{AI,\infty}(\mu)-P_{SC}^{AI}(\mu,\bar\kappa)
}
&=
\sup_{z\geq0}
\frac{A(t;\kappa)z+C(t;\kappa)}
{A(t;\bar\kappa)z+C(t;\bar\kappa)}
=
\max\left\{
\frac{A(t;\kappa)}{A(t;\bar\kappa)},
\frac{C(t;\kappa)}{C(t;\bar\kappa)}
\right\}.
\end{align*}
Because $t<T$ and $\bar\kappa$ is finite, Theorem \ref{theorem_42} gives
\begin{align*}
A(t;\bar\kappa)>0,\qquad C(t;\bar\kappa)>0.
\end{align*}
Equation \eqref{coefficient_high_capacity_rates} implies that $C(t;\kappa)$ is bounded above and below by positive multiples of $r_\rho(\kappa)$, while $A(t;\kappa)$ is bounded above by a positive multiple of $r_\rho(\kappa)$. Dividing by the fixed positive benchmark coefficients and taking the maximum yields constants $0<K_1\leq K_2<\infty$ and $\kappa_0>\bar\kappa$ for which \eqref{voi_high_capacity_rates} holds.

Since $r_\rho(\kappa)\to0$, the upper bound in \eqref{voi_high_capacity_rates} is smaller than one for all sufficiently large $\kappa$. Hence,
\begin{align*}
P_{SC}^{AI,\infty}(\mu)-P_{SC}^{AI}(\mu,\kappa)
<
P_{SC}^{AI,\infty}(\mu)-P_{SC}^{AI}(\mu,\bar\kappa)
\end{align*}
uniformly in $\mu$, which proves \eqref{eventual_capacity_dominance}.
