<!-- theorem-1: body_md -->
\label{thm:basic}
		Under \eqref{eq:standing}, the following conclusions hold.
		\begin{enumerate}[label=\textup{(\roman*)}]
			\item $G_{\varphi,\Omega}\in\operatorname{PSH}(\Omega)\cap L^\infty_{\mathrm{loc}}(\Omega)$, it is

		\end{enumerate}

<!-- definition-1: body_md -->
\label{def:deformation}
		A neighbourhood deformation of $\overline\Omega$ is a family $\Phi_t$, $0\le t<t_0$,
		of biholomorphic maps defined on one fixed open neighbourhood $N$ of $\overline\Omega$,
		with $\Phi_0=\mathrm{id}$ and $\Phi_t\to\mathrm{id}$ locally uniformly on $N$.
		The family is continuous in $(t,z)$. For each $z\in\Omega$ we also require that
		$\Phi_t^{-1}(z)$ is defined for sufficiently small $t$ and is real analytic
		in $t$ near $0$. The identity curve is allowed.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:cluster}
		The boundary cluster set of the deformation is
		\begin{equation}\label{eq:cluster}
			S_\Phi\coloneqq\{\xi:\exists\,t_j\downarrow0,\ 
			y_j\in\overline\Omega\cap\Phi_{t_j}(\partial\Omega),\ y_j\to\xi\}.
		\end{equation}
		It is a compact subset of $\partial\Omega$ and may be empty.

<!-- theorem-2: body_md -->
\label{thm:continuity}
		Assume \eqref{eq:standing} and let $\Phi_t$ be a neighbourhood deformation
		as in Definition~\ref{def:deformation}. Write $B\coloneqq A\cap\partial\Omega$ and
		$D_t\coloneqq\Phi_t(\Omega)$. Assume the following two conditions.
		\begin{enumerate}[label=\textup{(D\arabic*)}]
			\item\label{cond:obstacle}
			Whenever $t_j\downarrow0$ and $x_j\in\Omega$ tend to a point of $B$, with
			$y_j\coloneqq\Phi_{t_j}(x_j)\in\overline\Omega\setminus A$, one has
			\begin{equation}\label{eq:compatibility}
				\limsup_{j\to\infty}\bigl(\varphi(x_j)-\varphi(y_j)\bigr)\le0.
			\end{equation}
			Here $\varphi(y_j)$ is finite; if $x_j\in A$, the corresponding difference is
			$+\infty$. Thus no expression $+\infty-(+\infty)$ occurs.
			\item\label{cond:barrier}
			$S_\Phi\cap A=\varnothing$, and every point of $S_\Phi$ admits a continuous
			plurisubharmonic barrier.
		\end{enumerate}
		Then
		\begin{equation}\label{eq:density}
			G_{\varphi,\Omega}(z)=\sup_{v\in\mathcal C_{\varphi}(\Omega)}v(z),\qquad z\in\Omega,
		\end{equation}
		and $G_{\varphi,\Omega}$ is continuous on $\Omega$. Moreover, if
		$f_j\in C(\partial\Omega)$ increases pointwise to $\varphi$ on the whole boundary,
		then
		\[
		\operatorname{PB}_{f_j,\Omega}\uparrow G_{\varphi,\Omega}\quad\text{on }\Omega,
		\]
		and the convergence is locally uniform. If the barriers at the points of
		$S_\Phi$ are neighbourhood barriers, then \eqref{eq:density} remains valid
		with $\mathcal N_{\varphi}(\Omega)$ in place of $\mathcal C_{\varphi}(\Omega)$.
		No inclusion between distinct $D_t$, or between $D_t$ and $\Omega$, is
		required. The ambient neighbourhood need not be pseudoconvex or Runge.

\label{thm:hull}

			K&\coloneqq\{(z,w):z\in\overline\Omega,\ |w|\le e^{-\varphi(z)}\},\\

			\widehat{K}_{U\times\mathbb C}=\widehat{K_\partial}_{U\times\mathbb C}.

		\]

		\cite[Lemma 2.5]{DiGreen}. Let $H\in\mathcal O(U\times\mathbb C)$ with
		$\|H\|_{L_f}\le1$ and write
		$H(z,w)=\sum_{m=0}^{\infty}H_m(z)w^m$. Cauchy's estimate gives
		\[
		|H_m(z)|\le e^{mf(z)}\quad(z\in\partial D,\ m\ge1),
		\]

		\[
		m^{-1}\log|H_m|\le\operatorname{PB}_{f,D}\quad\text{on }D.
		\]

		\[
		|H(z,w)|\le\sum_{m=0}^{\infty}r^m=(1-r)^{-1}.
		\]

		and take $N$th roots. Letting $N\to\infty$ gives $|H(z,w)|\le1$.
		Closure gives the limiting fibre circle. Scaling $H$ (or first dividing by
		$\|H\|+\delta$) establishes the hull inclusion.

		It gives $m^{-1}\log|H_m|\le\psi$ on $D$; at points where $\psi=+\infty$ the
		inequality is automatic. By the definition of the obstacle envelope, the
		left side is at most $G_{\psi,D}$. The zero section
		$\overline D\times\{0\}$ lies in $K_\psi$, so the normalization
		$\|H\|_{K_\psi}\le1$ also gives $|H_0|\le1$ on $D$.
		The same powers argument now proves (ii). Neither assertion requires
		holomorphic convexity of $\overline D$.

We first verify that boundary-supported Jensen measures exist, without
		assuming regularity of the boundary. Let $\omega_z$ be the harmonic measure
		at $z$ for the bounded domain $\Omega$, viewed as a probability measure on
		$\partial\Omega$. The submean (or stopped Brownian-motion) inequality for a
		continuous subharmonic function on $\overline\Omega$ gives
		$v(z)\le\int_{\partial\Omega}v\,d\omega_z$ for every
		$v\in\operatorname{PSH}(\Omega)\cap C(\overline\Omega)$. Thus
		$\omega_z\in J_z^{c,\partial}$, and its integral of $f$ is finite.
		
		Extend $f$ to the lower semicontinuous, bounded-below function
		$\widetilde f$ on $\overline\Omega$ which equals $f$ on $\partial\Omega$ and $+\infty$
		on $\Omega$. Edwards' theorem for lower semicontinuous extended-real data,
		applied to the cone $\operatorname{PSH}(\Omega)\cap C(\overline\Omega)$, gives the infimum over its
		full Jensen class on $\overline\Omega$; see \cite{Edwards} and compare
		\cite[Proposition~2.1]{DW}. The boundary measure $\omega_z$ shows that
		this infimum is finite. The full Jensen class is weakly compact, and the
		integral of $\widetilde f$ is lower semicontinuous, so the infimum is
		attained. Every minimizer has finite integral and hence gives zero mass to
		$\Omega$. Conversely, a boundary-supported Jensen measure for the cone is
		exactly a member of $J_z^{c,\partial}$. This proves \eqref{eq:Edwards}.

		measures on the compact set $\partial\Omega$, because every test function has a
		continuous boundary trace. It is therefore compact.

		\eqref{eq:polarhull}. If a plurisubharmonic function on a neighbourhood of
		$\overline B$ is $-\infty$ on $B_A$, its restriction to the closed axis disk

		the reverse inclusion. Hence
		\[
		\mathcal H_B(B_A)=\{(0,z_2):|z_2|<1\}.
		\]

\label{ex:curved}
		Let $B\subset\mathbb C^2$ be the unit ball, with ambient domain $U=\mathbb C^2$. Choose $0<a<1$ and
		$\varepsilon\in\mathbb C$ with $|\varepsilon|<1/2$. Set
		\[
		f_\pm(z_1)\coloneqq\varepsilon z_1^2\pm a,\qquad
		F(z_1,z_2)\coloneqq(z_2-\varepsilon z_1^2)^2-a^2,\qquad \varphi\coloneqq-\log|F|.
		\]
		The two graphs are disjoint everywhere. The explicit polynomial
		biholomorphisms
		\begin{equation}\label{eq:curveddeformation}
			\Phi_t(z_1,z_2)\coloneqq
			\big((1+t)z_1,\ z_2+\varepsilon((1+t)^2-1)z_1^2\big)
		\end{equation}
		satisfy $F\circ\Phi_t=F$. Their inverse is obtained by first dividing the
		first coordinate by $1+t$, then subtracting the displayed quadratic term.
		Thus the obstacle compatibility is exact.
		
		For $\rho\coloneqq|z_1|^2+|z_2|^2-1$ and forward velocity
		$(z_1,2\varepsilon z_1^2)$, the outward normal expression at a point
		$\xi\in\{F=0\}\cap\partial B$ is
		\begin{align*}
			h(\xi)&=|\xi_1|^2+\operatorname{Re}(2\varepsilon\xi_1^2\overline{\xi_2})\\
			&\ge(1-2|\varepsilon|)|\xi_1|^2>0.
		\end{align*}
		The last strict inequality holds because $\xi_1=0$ on these graphs would
		force $\xi_2=\pm a$, which lies inside the ball, not on its boundary.
		Proposition~\ref{prop:transverse} proves $S_\Phi\cap\{F=0\}=\varnothing$.
		In fact, if $s\coloneqq(1+t)^2-1>0$ and $y\in\partial B$, then
		\[
		\|\Phi_t(y)\|^2-1
		\ge s(1-2|\varepsilon|)|y_1|^2.
		\]
		Thus the only crossing points in the closed ball are the fixed points
		$y_1=0$, and $S_\Phi=\{(0,w):|w|=1\}$ is genuinely nonempty.
		All boundary points of the ball have neighbourhood barriers, and
		$\overline B$ is polynomially convex. Hence the residual envelope is
		continuous, the disk-bundle hull identities hold, and the obstacle-dominated
		solution is unique.

<!-- proposition-6: body_md -->
\label{prop:tuple}
		Let $F\coloneqq(F_1,\ldots,F_q)$ be holomorphic on a neighbourhood of $\overline\Omega$, and
		assume that $F$ is not identically zero on the connected neighbourhood in question. Put
		\[
		\varphi\coloneqq-\log\|F\|+H,
		\]
		where $H\in C(\overline\Omega;\mathbb R)$ is superharmonic on $\Omega$. Suppose $\Phi_t$ is a
		neighbourhood deformation and that on a fixed neighbourhood of $\overline\Omega$
		there are holomorphic $q\times q$ matrices $M_t$ such that
		\begin{equation}\label{eq:matrixcovariance}
			F(\Phi_t(x))=M_t(x)F(x),\qquad
			\sup_{x\in\overline\Omega}\|M_t(x)-I\|\longrightarrow0.
		\end{equation}
		If $S_\Phi$ avoids $\{F=0\}\cap\partial\Omega$ and admits the barriers of
	Theorem~\ref{thm:continuity}, then $G_{\varphi,\Omega}$ is continuous and the monotone
	approximation conclusion of that theorem holds. The nonmonotone conclusion of
	Corollary~\ref{cor:completepolar} also applies. The same assertion holds
		for a finite positive linear combination of such logarithmic obstacles,
		provided each tuple satisfies \eqref{eq:matrixcovariance} for the same
		family and the cluster set avoids the union of their zero sets.

<!-- proof-29: body_md -->
The function $\log\|F\|$ is plurisubharmonic, so $\varphi$ satisfies
		\eqref{eq:standing}. The singular set is exactly $\{F=0\}\cap\overline\Omega$.
		Write $\eta_t\coloneqq\sup_{\overline\Omega}\|M_t-I\|$. For small $t$, $\eta_t<1/2$ and
		\[
		(1-\eta_t)\|F(x)\|\le\|F(\Phi_t(x))\|
		\le(1+\eta_t)\|F(x)\|.
		\]
		In particular, the common zero set is preserved in both directions wherever
		this identity is used. At nonzero fibres,
		\[
		\left|\log\frac{\|F(\Phi_t(x))\|}{\|F(x)\|}\right|
		\le \max\{-\log(1-\eta_t),\log(1+\eta_t)\}\longrightarrow0.
		\]
		On the overlap $x\in\Omega$, $\Phi_t(x)\in\overline\Omega$, uniform continuity of $H$
		gives $H(x)-H(\Phi_t(x))\to0$ uniformly. Thus (D1) follows, with the
		zero-fibre convention included. Moreover,
		$\{F=0\}=\{\log\|F\|=-\infty\}$ is complete pluripolar on the
		connected neighbourhood under consideration. Hence Theorem~\ref{thm:continuity}
		and Corollary~\ref{cor:completepolar} give the stated continuity and
		approximation conclusions. For
		finitely many tuples, sum the estimates with their positive coefficients.
		The union of the corresponding zero sets is again contained in a complete
		pluripolar set, for instance the $-\infty$ locus of the sum of the functions
		$\log\|F^{(\ell)}\|$, and the total error still tends to zero. This proves
		the stated extension.

<!-- remark-4: body_md -->
Equation~\eqref{eq:matrixcovariance} is a sufficient hypothesis, not a
		consequence of mere invariance of the common zero set. The distinction is
		already visible for $F(z_1,z_2)\coloneqq(z_1^2,z_2)$ and
		$\Phi_t(z_1,z_2)\coloneqq(z_1,z_2+t z_1)$. The common zero set is $\{0\}$ and is
		preserved. But, for a fixed nonzero $t$, the term $t z_1$ is not in the
		ideal $(z_1^2,z_2)$, so the second transformed component cannot be expressed
		as a holomorphic linear combination of the original components near $0$.
		Also, along $(z_1,0)$, the quotient
		$\|F(\Phi_t(z_1,0))\|/\|F(z_1,0)\|$ is unbounded as $z_1\to0$.
		In the reduced principal-divisor case, Lemma~\ref{lem:divisor} supplies
		the required unit factor.
