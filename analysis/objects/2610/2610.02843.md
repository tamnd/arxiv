<!-- theorem-1: body_md -->
\label{cor:pw-eisl} For the signal model \eqref{eq:tit-unitary-modulation} with independent symbols satisfying \eqref{eq:pw-symbol-moments}, the following statements hold.
\begin{enumerate}
\item { For a prescribed lag \(1\le k<N\), write \(N=qk+h\), where \(q\ge1\) and \(0\le h<k\). The exact minimum is

\begin{equation}
 \min_{\mathbf U^{\mathrm H}\mathbf U=\mathbf I_N}\operatorname{\acs*{ESL}}_k(\mathbf U)
 =(N-k)+(\mu_4-2)\left(N-2k+\frac{h}{q+1}+\frac{k-h}{q}\right).
 \label{eq:tit-optimal-single-esl}
\end{equation}

It is attained by the unitary matrix \(\mathbf U_{\rm blk}\) constructed in \eqref{eq:pw-block-competitor-matrix}, which applies a normalized \ac{IDFT} separately on each chain of sample indices spaced by \(k\). Its improvement over the full \ac{IDFT} \(\mathbf V\) is

\begin{equation}
 \operatorname{\acs*{ESL}}_k(\mathbf V)
 -\min_{\mathbf U^{\mathrm H}\mathbf U=\mathbf I_N}\operatorname{\acs*{ESL}}_k(\mathbf U)
 =(2-\mu_4)\frac{h(k-h)}{Nq(q+1)}.
 \label{eq:tit-single-esl-gain}
\end{equation}

Consequently, the full \ac{IDFT} minimizes the \ac{ESL} at lag \(k\) if and only if \(k\mid N\).}
\item Every unitary \(\mathbf U\) and every \(1\le r<N\) satisfy \begin{equation} \mathcal E_r(\mathbf U)\ge\sum_{k=1}^{r}(N-k)+\frac{\mu_4-2}{N}\sum_{k=1}^{r}(N-k)^2=\mathcal E_r(\mathbf V). \label{eq:tit-optimal-partial-eisl} \end{equation} For any fixed \(1\le r<N\), equality holds in \eqref{eq:tit-optimal-partial-eisl} if and only if \begin{equation} u_{m,n}=\frac{e^{\mathrm i(\alpha_n+m\theta)}}{\sqrt N}e^{2\pi\mathrm i m\sigma(n)/N},\qquad 0\le m,n<N, \label{eq:tit-optimal-modulation-family} \end{equation} where \(\theta,\alpha_n\in\mathbb R\) and \(\sigma\) is a permutation of \(\{0,\ldots,N-1\}\).
\end{enumerate}

<!-- corollary-1: body_md -->
\label{cor:tit-simultaneous-prefix}
Under the assumptions of Theorem~\ref{cor:pw-eisl}, fix \(1\le r<N\). There exists a unitary matrix \(\mathbf U_*\) satisfying
\begin{equation}
 \operatorname{\acs*{ESL}}_k(\mathbf U_*)
 =\min_{\mathbf U^{\mathrm H}\mathbf U=\mathbf I_N}
    \operatorname{\acs*{ESL}}_k(\mathbf U),\qquad 1\le k\le r,
 \label{eq:tit-simultaneous-prefix}
\end{equation}
if and only if
\begin{equation}
 k\mid N\qquad\text{for every }k\in\{1,\ldots,r\}.
 \label{eq:tit-simultaneous-divisibility}
\end{equation}
When \(N\ge3\), no unitary matrix simultaneously minimizes the \ac{ESL} at all positive lags.

Let \(\mathcal{C}_1,\ldots,\mathcal{C}_s\) be the connected components of the undirected graph with incidence matrix \(\mathbf A_{\mathcal K}\). By \cite[Theorem~2.3 and Lemma~2.4]{bapat2010graphs},
\begin{equation}
 \operatorname{rank}(\mathbf A_{\mathcal K})=N-s,
 \label{eq:pw-solvability-rank}
\end{equation}
and \eqref{eq:balance_matrix} is solvable for all
\(\ell\in\mathcal K\) if and only if
\begin{equation}
 \sum_{i\in\mathcal{C}_a}\eta_{\ell,i}=0,
 \qquad a=1,\ldots,s,\ \ell\in\mathcal K.
 \label{eq:pw-solvability-component-balance}
\end{equation}
Since every edge length is divisible by \(d\), each component must lie in a single residue class modulo \(d\).  Since all \(d\) classes are nonempty, we have \(s\ge d\).

For necessity, let us assume that \eqref{eq:pw-solvability-component-balance} holds.  For each \(\ell\in\mathcal K\), a component \(\mathcal{C}_a\) is a union of complete chains with step \(\ell\).  Each chain contributes one vertex to each boundary interval in \eqref{eq:pw-boundary-coefficients}, including a chain with only one vertex.  Let \(h_\ell(\mathcal{C}_a)\) be the number of these chains, then
\begin{equation}
 0=\sum_{i\in\mathcal{C}_a}\eta_{\ell,i}
 =2h_\ell(\mathcal{C}_a)-\frac{2\ell|\mathcal{C}_a|}{N}
 \quad\Longrightarrow\quad
 \frac{\ell|\mathcal{C}_a|}{N}\in\mathbb Z.
 \label{eq:pw-solvability-chain-integrality}
\end{equation}
Because \(d\) is the greatest common divisor of the allowed lags, it can be represented as an integer linear combination of these lags, and thus \eqref{eq:pw-solvability-chain-integrality} gives
$$
\frac{d|\mathcal{C}_a|}{N}=\sum_{\ell\in\mathcal{K}} \alpha_\ell \frac{\ell|\mathcal{C}_a|}{N}\in\mathbb Z_+,
$$  
for some integers $\{\alpha_{\ell}\}$. Hence
\begin{equation}
 |\mathcal{C}_a|\ge\frac Nd,\qquad
 N=\sum_{a=1}^{s}|\mathcal{C}_a|\ge\frac{sN}{d}
 \quad\Longrightarrow\quad s\le d.
 \label{eq:pw-solvability-component-count}
\end{equation}
Together with \(s\ge d\), \eqref{eq:pw-solvability-component-count}
forces \(s=d\) and \(|\mathcal{C}_a|=N/d\) for every \(a\).

required rank.

Conversely, suppose \(d\mid N\) and the rank is \(N-d\). By \eqref{eq:pw-solvability-rank}, \(s=d\), so each component is an entire residue class modulo \(d\).  Such a class has \(N/d\) vertices. \begin{equation}

 =\frac{2\ell}{d}-\frac{2\ell}{N}\frac Nd=0,
 \qquad a=1,\ldots,d,\ \ell\in\mathcal K.
 \label{eq:pw-solvability-sufficient-balance}
\end{equation}
Equation~\eqref{eq:pw-solvability-sufficient-balance} verifies
the criterion \eqref{eq:pw-solvability-component-balance},
which proves sufficiency.

\textbf{An estimate for a vector on one chain.} For every \(m\ge1\) and \(\mathbf z\in\mathbb C^m\), we first establish

\begin{equation}
 |\gamma_1(\mathbf z)|^2\le\|\mathbf z\|_2^2
 \left[\left(1-\frac1{m^2}\right)\|\mathbf z\|_2^2
       -\left(1-\frac1m\right)(|z_0|^2+|z_{m-1}|^2)\right],
 \label{eq:tit-chain-vector-bound}
\end{equation}

where \(\gamma_1(\mathbf z)=\sum_{i=0}^{m-2}\overline{z_i}z_{i+1}\), with the empty sum equal to zero when \(m=1\). Both sides of \eqref{eq:tit-chain-vector-bound} vanish when \(m=1\) or \(\mathbf z=0\). Otherwise, normalize the magnitudes by

\begin{equation}
 x_i=\frac{\sqrt m\,|z_i|}{\|\mathbf z\|_2},\qquad p_i=x_i^2,
 \qquad \sum_{i=0}^{m-1}p_i=m.
 \label{eq:tit-chain-normalization}
\end{equation}

Use \(E_1,H_1,b_1,V_1\) from \eqref{eq:pw-EH-definition}, \eqref{eq:pw-A-definition}, and \eqref{eq:pw-V-definition} with ambient dimension \(m\). The boundary expression \eqref{eq:pw-boundary-form} and telescoping give

\begin{equation}
 b_1=p_0+p_{m-1}-2
 =\sum_{i=0}^{m-2}\left(1-\frac{2(i+1)}m\right)(p_i-p_{i+1}).
 \label{eq:tit-chain-boundary-expansion}
\end{equation}

The largest absolute coefficient in \eqref{eq:tit-chain-boundary-expansion} is \(1-2/m\). The triangle inequality and \eqref{eq:pw-variation-bound} therefore imply

\textbf{An upper bound for an arbitrary unitary matrix.} For each column \(\mathbf u_n\) of a unitary \(\mathbf U\), denote its entries on chain \(\mathcal C_j\), in chain order, by

\begin{equation}
 \mathbf z_{j,n}=(u_{j,n},u_{j+k,n},\ldots,u_{j+(m_j-1)k,n})^{\mathrm T}
 \in\mathbb C^{m_j}.
 \label{eq:tit-chain-column-restriction}
\end{equation}

The chains partition the rows. Column and row normalization in \eqref{eq:tit-unitary-modulation} therefore give

\begin{subequations}
\begin{align}
 \sum_{j=0}^{k-1}\|\mathbf z_{j,n}\|_2^2&=1,
 \label{eq:tit-chain-column-norm}\\
 \sum_{n=0}^{N-1}\|\mathbf z_{j,n}\|_2^2&=m_j,
 \label{eq:tit-chain-power-sum}\\

 \label{eq:tit-chain-endpoint-sum}
\end{align}

Equation~\eqref{eq:tit-chain-endpoint-sum} also holds when \(m_j=1\), in which case the same row is counted twice. Every sample pair at lag \(k\) is adjacent within one chain.  |\gamma_k(\mathbf u_n)|^2

       \left(1-\frac1{m_j^2}\right)\|\mathbf z_{j,n}\|_2^2

       (|u_{j,n}|^2+|u_{j+(m_j-1)k,n}|^2)\right].
 \label{eq:tit-chain-column-bound}
\end{align}

 &=\sum_{j=0}^{k-1}\frac{(m_j-1)^2}{m_j}.
 \label{eq:tit-arbitrary-single-upper-bound}

 \label{eq:pw-path-last-vertex}
\end{equation}
The interval in \eqref{eq:pw-path-last-vertex} contains \(\ell\) consecutive integers, so there is a unique choice. The congruence and \(0\le a<\ell\) give \(a\le v_{ab}<b\). Choose
\begin{equation}
 \mathcal{P}_{ab}=(a,a+\ell,\ldots,v_{ab},b),\qquad
 r-\ell+1\le b-v_{ab}\le r.
 \label{eq:pw-q-path-choice}
\end{equation}
When \(v_{ab}=a\), this is just the direct path, otherwise all steps before the last have length \(\ell\). Since \(\ell\le r\), every path in \eqref{eq:pw-q-path-choice} satisfies \eqref{eq:pw-correction-path-choice}.

Next we compute the edge reuse times. Fix an allowed edge \((i,i+k)\). If it is the last edge of a path, we have \(b=i+k\). When \(b\le r\), it is a direct edge and hence \(a=i\). When \(b>r\), \eqref{eq:pw-path-last-vertex} determines \(a\) as the unique integer in \(\{0,\ldots,\ell-1\}\) congruent to \(i\) modulo \(\ell\). Thus at most one chosen path uses this edge as its last step, namely
\begin{equation}
 \bigl|\{(a,b)\in\mathcal{Q}_\ell:
 (i,i+k)\text{ is the last edge of }\mathcal{P}_{ab}\}\bigr|\le1.
 \label{eq:pw-last-edge-count}
\end{equation}
Every edge used before the last step has length \(\ell\). If \((i,i+\ell)\) is such an edge, its initial vertex again determines the starting vertex \(a\) modulo \(\ell\). For this fixed \(a\), there is at most one path per terminal vertex \(b\), and $b$ satisfies \(r<b<N-r-1\) by \eqref{eq:pw-correction-endpoints} because every path with \(b\le r\) has only one edge. Writing \(x_+=\max\{x,0\}\), the number of valid $b$ is at most
\begin{equation}
 \bigl|\{b\in\mathbb Z:r<b<N-r-1\}\bigr|=(N-2r-2)_+.
 \label{eq:pw-initial-step-terminals}
\end{equation}
Combining \eqref{eq:pw-last-edge-count} and \eqref{eq:pw-initial-step-terminals} gives \(c_{i,\ell}^{(\ell)}\le1+(N-2r-2)_+ =\max\{1,N-2r-1\}\). For \(k\ne\ell\), the edge can only occur as a last step, so \eqref{eq:pw-last-edge-count} gives \(c_{i,k}^{(\ell)}\le1\). This proves \eqref{eq:pw-correction-count-bound}. 

Finally, by \eqref{eq:pw-correction-endpoints}, all the vertices in every permissible path have indices less than \(N-r-1\), which proves the stated vanishing of the edge reuse times.

Assume equality in \eqref{eq:pw-prefix-bound}. By
\eqref{eq:pw-deficit-definition}, this means
\(B_{\{1,\ldots,r\}}(\mathbf U)=0\).
We determine the absolute values of the entries of \(\mathbf U\),
then their phases.

\par\medskip\noindent
\textbf{All entries of \(\mathbf U\) have absolute value \(N^{-1/2}\).}
For any nonnegative vector \(\mathbf x\) with
\(\sum_{i=0}^{N-1}x_i^2=N\), the row and column bounds
\eqref{eq:pw-unified-prefix-budgets} allow us to apply
Lemma~\ref{lem:pw-contraction} with
\(\rho_{N,r}=\max\{r,N-2\}/N<1\).
Its conclusion \eqref{eq:pw-boundary-contraction}, followed by
\eqref{eq:pw-variation-bound}, gives
\begin{equation}
 \begin{aligned}
 \sum_{k=1}^{r}(E_kH_k-b_k^2)
 &\ge\sum_{k=1}^{r}E_kH_k
       -\rho_{N,r}^{\,2}\sum_{k=1}^{r}V_k^2\\
 &\ge(1-\rho_{N,r}^{\,2})\sum_{k=1}^{r}E_kH_k\ge0.
 \end{aligned}
 \label{eq:pw-strict-profile}
\end{equation}
Apply \eqref{eq:pw-strict-profile} to each vector
\(\mathbf x^{(n)}\) defined in \eqref{eq:pw-normalized-profile}.
In the gap decomposition \eqref{eq:pw-deficit-decomposition},
the sum over \(k\) in the first term is nonnegative for each
column \(n\), by \eqref{eq:pw-strict-profile}. Every summand in
the second term is nonnegative by \eqref{eq:pw-phase-relaxation}.
Since \(B_{\{1,\ldots,r\}}(\mathbf U)=0\), these nonnegative
quantities must all be zero. Thus, for every \(0\le n<N\),
\begin{equation}
 \begin{gathered}
 \sum_{k=1}^{r}
 \left[E_k^{(n)}H_k^{(n)}-(b_k^{(n)})^2\right]=0,~\gamma_k(\mathbf x^{(n)})^2
 =N^2|\gamma_k(\mathbf u_n)|^2,~ 1\le k\le r.
 \end{gathered}
 \label{eq:pw-zero-gap-conditions}
\end{equation}
Because \(1-\rho_{N,r}^{\,2}>0\), equations
\eqref{eq:pw-strict-profile} and \eqref{eq:pw-zero-gap-conditions}
imply \(E_1^{(n)}H_1^{(n)}=0\).
On the other hand, \eqref{eq:pw-EH-definition} and \eqref{eq:pw-profile-normalization} give $ H_1^{(n)} =\sum_{i=0}^{N-2}(x_i^{(n)}+x_{i+1}^{(n)})^2>0$. It follows that
\begin{equation}
 E_1^{(n)}=\sum_{i=0}^{N-2}(x_i^{(n)}-x_{i+1}^{(n)})^2=0.
 \label{eq:pw-adjacent-magnitudes-equal}
\end{equation}
Equation~\eqref{eq:pw-adjacent-magnitudes-equal} makes all adjacent
entries of \(\mathbf x^{(n)}\) equal. Their common value is \(1\)
by \eqref{eq:pw-profile-normalization}. Therefore,
\begin{equation}
 x_i^{(n)}=1,\qquad |u_{i,n}|=N^{-1/2},
 \qquad 0\le i,n<N.
 \label{eq:pw-basis-equality-magnitudes}
\end{equation}

\par\medskip\noindent
\textbf{Within each column, successive entries have a constant phase difference.}
Taking \(k=1\) in \eqref{eq:pw-zero-gap-conditions} and using
\eqref{eq:pw-basis-equality-magnitudes}, we obtain
\begin{equation}
 \left|\sum_{q=0}^{N-2}\overline{u_{q,n}}u_{q+1,n}\right|
 =\frac{N-1}{N}
 =\sum_{q=0}^{N-2}|\overline{u_{q,n}}u_{q+1,n}|.
 \label{eq:pw-adjacent-triangle-equality}
\end{equation}
Equality in the triangle inequality in
\eqref{eq:pw-adjacent-triangle-equality} means that all products
\(\overline{u_{q,n}}u_{q+1,n}\) have the same argument.
Their absolute values are \(1/N\) by
\eqref{eq:pw-basis-equality-magnitudes}. Hence, for some real \(\theta_n\),
\begin{equation}
 \overline{u_{q,n}}u_{q+1,n}=N^{-1}e^{\mathrm i\theta_n}
\Longrightarrow
 u_{q+1,n}=e^{\mathrm i\theta_n}u_{q,n},\qquad
 0\le q<N-1.
 \label{eq:pw-successive-entry-ratio}
\end{equation}
Write \(u_{0,n}=N^{-1/2}e^{\mathrm i\alpha_n}\) with \(\alpha_n\in\mathbb R\). Repeatedly applying \eqref{eq:pw-successive-entry-ratio} gives
\begin{equation}
 u_{q,n}=N^{-1/2}e^{\mathrm i\alpha_n+\mathrm i q\theta_n},
 \qquad 0\le q,n<N.
 \label{eq:pw-geometric-columns}
\end{equation}

\par\medskip\noindent
\textbf{Orthogonality determines the possible values of \(\theta_n\).}
For distinct columns \(m\ne n\), equations
{\eqref{eq:tit-unitary-modulation}} and \eqref{eq:pw-geometric-columns} give
\begin{equation}
 0=\mathbf u_m^{\mathrm H}\mathbf u_n
 =\frac{e^{\mathrm i(\alpha_n-\alpha_m)}}N
   \sum_{q=0}^{N-1}e^{\mathrm i q(\theta_n-\theta_m)}.
 \label{eq:pw-column-orthogonality}
\end{equation}
If \(e^{\mathrm i(\theta_n-\theta_m)}=1\), the sum in \eqref{eq:pw-column-orthogonality} would be \(N\), causing a contradiction. The formula for a finite geometric sum therefore yields
\begin{equation}
 0=\frac{1-e^{\mathrm i N(\theta_n-\theta_m)}}
        {1-e^{\mathrm i(\theta_n-\theta_m)}}
 \quad\Longrightarrow\quad
 e^{\mathrm i N(\theta_n-\theta_m)}=1.
 \label{eq:pw-phase-difference-roots}
\end{equation}
Equation~\eqref{eq:pw-phase-difference-roots} shows that, modulo \(2\pi\), each \(\theta_n\) is one of \(\theta_0,\theta_0+2\pi/N,\ldots,\theta_0+2\pi(N-1)/N\). The values \(\theta_n\) are distinct modulo \(2\pi\), because equal values would contradict \eqref{eq:pw-column-orthogonality}. Thus each of these \(N\) possibilities occurs once, and there is a permutation \(\sigma\) of \(\{0,\ldots,N-1\}\) such that
\begin{equation}
 \theta_n\equiv\theta_0+\frac{2\pi\sigma(n)}N\pmod{2\pi}.
 \label{eq:pw-common-phase-values}
\end{equation}
\begingroup
Combining \eqref{eq:pw-geometric-columns} and \eqref{eq:pw-common-phase-values} gives
\begin{equation}
 u_{q,n}=\frac{e^{\mathrm i(\alpha_n+q\theta_0)}}{\sqrt N}
             e^{2\pi\mathrm i q\sigma(n)/N},\qquad 0\le q,n<N.
 \label{eq:tit-equality-matrix-form}
\end{equation}
Equation~\eqref{eq:tit-equality-matrix-form} is precisely \eqref{eq:tit-optimal-modulation-family}, with \(\theta=\theta_0\).
\par\endgroup
