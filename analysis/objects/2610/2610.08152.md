<!-- assumption-1: body_md -->
\label{ass:movement}
Let $G_{\mathrm{mv}}=(V,E_{\mathrm{mv}})$ be a fixed connected unweighted graph on the vertex set $V$ with $\left\lvert V \right\rvert=n$, and let
$d(u,v):=\min_{\pi:u\leadsto v}\left\lvert \pi \right\rvert$
denote its shortest-path distance, where $\pi$ ranges over paths in $G_{\mathrm{mv}}$ and $\left\lvert \pi \right\rvert$ is the number of edges. All distances, $P$-hop neighborhoods, Voronoi cells, and matching distances in this paper refer to $G_{\mathrm{mv}}$.
<!-- end assumption-1 -->
<!-- remark-1: body_md -->
\label{rem:metric_scope}
The bottleneck matching distance of Definition~\ref{def:matching_distance} treats sensors as interchangeable (only asks whether the old and new positions can be matched with small displacement). It does not separate sensor types or account for trajectory collisions, travel times, or edge-capacity constraints. These limitations motivate trajectory-related extensions discussed in Section~\ref{sec:conclusion} for future work.
<!-- end remark-1 -->
<!-- definition-1: body_md -->
\label{def:optimizers}
Write $\operatorname*{arg\,max} F_t$ for the set of configurations maximizing $F_t$ over all $m$-subsets; it is nonempty because there is a finite possible family. A reference maximizer $S_t^{\star}\in\operatorname*{arg\,max} F_t$ is any fixed element of this set, relative to which all margins are defined. No tie-breaking is needed at time $t+1$; the results hold for every $S\in\operatorname*{arg\,max} F_{t+1}$.
<!-- end definition-1 -->
<!-- lemma-1: body_md -->
\label{lem:nondegenerate}
If $m\le K$ then some $m$-subset $S$ has $\det C_t(S)>0$; consequently $d^{\star}_t:=\det C_t(S_t^{\star})=e^{F_t(S_t^{\star})}>0$ for every reference maximizer.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Since $U_t$ has rank $K\ge m$,   it contains $m$ linearly independent rows, indexed by some $S$. Then $U_{t,S}\in\mathbb{R}^{m\times K}$ has rank $m$ and $C_t(S)=U_{t,S}U_{t,S}^\top$ is positive definite. Any maximizer thus has objective value at least $F_t(S)>-\infty$
<!-- end proof-1 -->
<!-- definition-2: body_md -->
\label{def:matching_distance}
For $m$-sets $S=\{s_1,\dots,s_m\}$ and $S'=\{s_1',\dots,s_m'\}$, put
\begin{equation}
\begin{split}
    d_{\mathrm{match}}(S,S')&:=\min_{\pi:\{1,\ldots,m\}\to\{1,\ldots,m\}\ \mathrm{bijective}}\\
    &\qquad\max_{1\le i\le m} d\bigl(s_i,s'_{\pi(i)}\bigr)
\end{split}
    \label{eq:dmatch_def}
\end{equation}
with $d$ as the distance of the fixed movement graph $G_{\mathrm{mv}}$. This is the smallest radius within which every old sensor can be matched bijectively to a new sensor. The value is independent of the ordering chosen for $S$ and $S'$.
<!-- end definition-2 -->
<!-- proposition-1: body_md -->
\label{prop:matching_metric}
$d_{\mathrm{match}}$ is a metric on the family of $m$-subsets of $V$.
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
The minimum in \eqref{eq:dmatch_def} is attained because there are only finitely many bijections between two $m$-element sets. Nonnegativity and symmetry thus follow directly from the corresponding properties of $d$. If $d_{\mathrm{match}}(S,S')=0$, then some bijection pairs each $s_i$ with a vertex at graph distance zero (hence with $s_i$ itself) so $S=S'$. Conversely, $d_{\mathrm{match}}(S,S)=0$. For the triangle inequality, let $S=\{s_i\}$, $T=\{t_j\}$, and $R=\{r_k\}$, and choose bijections $\pi$ and $\sigma$ attaining $d_{\mathrm{match}}(S,T)$ and $d_{\mathrm{match}}(T,R)$, respectively. Composing gives us
\begin{eqnarray*}
  d\bigl(s_i,r_{\sigma(\pi(i))}\bigr) 
  &\le& d\bigl(s_i,t_{\pi(i)}\bigr)+d\bigl(t_{\pi(i)},r_{\sigma(\pi(i))}\bigr) \\
  &\le& d_{\mathrm{match}}(S,T)+d_{\mathrm{match}}(T,R)
\end{eqnarray*}
for every $i$. Taking the maximum over $i$ and then the minimum over bijections from $S$ to $R$ gives us the claim.
<!-- end proof-2 -->
<!-- definition-3: body_md -->
\label{def:margin}
Fix a reference maximizer $S_t^{\star}\in\operatorname*{arg\,max} F_t$. For an integer $P\ge0$ define
\begin{equation}
  M_t(P):=\min_{\substack{\left\lvert S \right\rvert=m\\ d_{\mathrm{match}}(S,S_t^{\star})>P}}\bigl[F_t(S_t^{\star})-F_t(S)\bigr],
  \label{eq:margin_def}
\end{equation}
with convention $M_t(P)=+\infty$ when the minimum is over an empty family. Equivalently, among all configurations that cannot be reached from $S^\star_t$ by moving each sensor at most $P$ hops, $M_t(P)$ measures how much worse the best one is under the old objective.
<!-- end definition-3 -->
<!-- lemma-2: body_md -->
\label{lem:compression}
For every $m$-subset $S$,
\[\left\lVert C_t(S)-C_{t+1}(S) \right\rVert_2\le\rho_t.\]
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
$C_t(S)-C_{t+1}(S)=R_S(\Pi_t-\Pi_{t+1})R_S^\top$. Since $R_S$ has orthonormal rows, $\left\lVert R_S \right\rVert_2=\left\lVert R_S^\top \right\rVert_2=1$, and submultiplicativity gives the bound.
<!-- end proof-3 -->
<!-- lemma-3: body_md -->
\label{lem:logdet}
Let $X,Y\in\mathbb{R}^{m\times m}$ be symmetric with $X\succ0$ and $c:=\lambda_{\min}(X)>0$. If
\begin{equation}
  \left\lVert Y-X \right\rVert_2\le r\le\frac{c}{2},
  \label{eq:logdet_smallness}
\end{equation}
then $Y\succ0$ and
\begin{equation}
  \left\lvert \log\det Y-\log\det X \right\rvert\le\frac{2mr}{c}.
  \label{eq:logdet_bound}
\end{equation}
<!-- end lemma-3 -->
<!-- proof-4: body_md -->
Write $E:=Y-X$. By Weyl's inequality \cite[Cor.~4.3.15]{hornjohnson2013}, $\lambda_{\min}(Y)\ge c-r\ge c/2>0$, so $Y\succ0$. Since $X\succ0$ it has a symmetric positive-definite square root, and we may set  $ B:=X^{-1/2}EX^{-1/2}$ so that 
$  Y=X+E=X^{1/2}(I+B)X^{1/2}.$
 The matrix $B$ is symmetric because $E$ and $X^{-1/2}$ are, hence its eigenvalues $\nu_1,\dots,\nu_m$ are real. Taking determinants,
\begin{eqnarray*}
  \log\det Y-\log\det X&=&\log\det(I+B)\\
  &=&\sum_{i=1}^m\log(1+\nu_i).
\end{eqnarray*}
By submultiplicativity, $\left\lVert B \right\rVert_2\le\left\lVert X^{-1/2} \right\rVert_2^2\left\lVert E \right\rVert_2=\left\lVert E \right\rVert_2/c\le r/c\le1/2$, so $\left\lvert \nu_i \right\rvert\le1/2$ for every $i$. Then, for $\left\lvert x \right\rvert\le\tfrac12$ the scalar inequality
\begin{equation}
\begin{gathered}
  \left\lvert \log(1+x) \right\rvert\le2\left\lvert x \right\rvert
\end{gathered}
\label{eq:scalar_log}
\end{equation}
holds because for $x\in[0,1/2]$, one has $\log(1+x)\le x\le 2\left\lvert x \right\rvert$, while for $x\in[-1/2,0]$,
\begin{eqnarray*}
  \left\lvert \log(1+x) \right\rvert&=&-\log(1+x)\\
  &=&\int_{1+x}^{1}\frac{dt}{t}\le 2(-x)\\
  &=&2\left\lvert x \right\rvert,
\end{eqnarray*}
since $t\ge1/2$ throughout the interval of integration. Therefore
\begin{eqnarray*}
  \left\lvert \log\det Y-\log\det X \right\rvert &\le& \sum_{i=1}^m\left\lvert \log(1+\nu_i) \right\rvert\\
  &\le&2\sum_{i=1}^m\left\lvert \nu_i \right\rvert\\
  &\le&2m\left\lVert B \right\rVert_2\\
  &\le&\frac{2mr}{c}.
\end{eqnarray*}

This proves \eqref{eq:logdet_bound}.
<!-- end proof-4 -->
<!-- proposition-2: body_md -->
\label{prop:auto}
Let $\Pi$ be an orthogonal projector, $S$ an $m$-subset, and $C=(\Pi)_{SS}$. Then $0\preceq C\preceq I_m$, and if $\det C>0$ then
\begin{equation}
  \lambda_{\min}(C)\ge\det C.
  \label{eq:auto}
\end{equation}
<!-- end proposition-2 -->
<!-- proof-5: body_md -->
Since $0\preceq\Pi\preceq I_n$, congruence by $R_S$ gives $0\preceq C\preceq I_m$, so every eigenvalue of $C$ lies in $[0,1]$. Writing $\det C=\prod_{i=1}^m\lambda_i(C)$ and bounding every factor other than the smallest by $1$ gives us \eqref{eq:auto}.
<!-- end proof-5 -->
<!-- theorem-1: body_md -->
\label{thm:main}
Assume $m\le K$ and Assumption~\ref{ass:movement}. Let $S_t^{\star}\in\operatorname*{arg\,max} F_t$ be a reference maximizer and define
 $ \lambda^{\star}_t:=\lambda_{\min}\bigl(C_t(S_t^{\star})\bigr),$ and
  $d^{\star}_t:=\det C_t(S_t^{\star})>0 .$ 
Let $r$ satisfy $\rho_t\le r$, and define
\begin{gather*}
\eta_{t,1}:=\frac{2mr}{\lambda^{\star}_t},\text{ and }
\eta_{t,2}:=\frac{2mr\,e^{\eta_{t,1}}}{d^{\star}_t},\\\text{ and }
B_t:=\eta_{t,1}+\eta_{t,2}.
\end{gather*}
Suppose the two smallness conditions

\begin{equation}
\begin{gathered}
r\le\frac{\lambda^{\star}_t}{2},
\qquad
r\le\frac{d^{\star}_t e^{-\eta_{t,1}}}{2}.
\end{gathered}
\label{eq:smallness}
\end{equation}

hold. Then:
\begin{enumerate}[label=(\alph*),leftmargin=*]
\item Objective loss. For every $S\in\operatorname*{arg\,max} F_{t+1}$,
\begin{equation}
  F_t(S_t^{\star})-F_t(S)\le B_t.
  \label{eq:objective_loss}
\end{equation}
\item Movement. For every integer $P\ge0$,
\begin{equation}
\begin{gathered}
  M_t(P)>B_t
  \Longrightarrow\quad d_{\mathrm{match}}(S_t^{\star},S)\le P\\
  \text{ for every }S\in\operatorname*{arg\,max} F_{t+1}.
\end{gathered}
\label{eq:main_implication}
\end{equation}
\end{enumerate}
<!-- end theorem-1 -->
<!-- proof-6: body_md -->
By Lemma~\ref{lem:nondegenerate}, $d^{\star}_t>0$, and by Proposition~\ref{prop:auto}, $\lambda^{\star}_t\ged^{\star}_t>0$.

First, we control the drift at the reference maximizer. By Lemma~\ref{lem:compression}, $\left\lVert C_{t+1}(S_t^{\star})-C_t(S_t^{\star}) \right\rVert_2\le\rho_t\le r$, and by the first condition in \eqref{eq:smallness}, $r\le\lambda^{\star}_t/2$. Lemma~\ref{lem:logdet} with $X=C_t(S_t^{\star})$, $Y=C_{t+1}(S_t^{\star})$, and $c=\lambda^{\star}_t$ gives $C_{t+1}(S_t^{\star})\succ0$ and
\begin{equation}
  \left\lvert F_{t+1}(S_t^{\star})-F_t(S_t^{\star}) \right\rvert\le\frac{2mr}{\lambda^{\star}_t}=\eta_{t,1}.
  \label{eq:step1}
\end{equation}
Next, let $S\in\operatorname*{arg\,max} F_{t+1}$. By optimality at time $t+1$ and \eqref{eq:step1},
\[
  F_{t+1}(S)\ge F_{t+1}(S_t^{\star})\ge F_t(S_t^{\star})-\eta_{t,1}=\logd^{\star}_t-\eta_{t,1},
\]
and so $\det C_{t+1}(S)\ged^{\star}_t e^{-\eta_{t,1}}>0$. The matrix $C_{t+1}(S)$ is positive semidefinite with positive determinant, hence positive definite, and Proposition~\ref{prop:auto} applied to $\Pi_{t+1}$ gives us
\begin{equation}
  \lambda_{\min}\bigl(C_{t+1}(S)\bigr)\ge\det C_{t+1}(S)\ged^{\star}_t e^{-\eta_{t,1}}.
  \label{eq:step2}
\end{equation}
Now compare the new maximizer back to time $t$. By Lemma~\ref{lem:compression}, $\left\lVert C_t(S)-C_{t+1}(S) \right\rVert_2\le\rho_t\le r$, and the second condition in \eqref{eq:smallness} together with \eqref{eq:step2} gives
\[
  r\le\frac{d^{\star}_t e^{-\eta_{t,1}}}{2}\le\frac{\lambda_{\min}\bigl(C_{t+1}(S)\bigr)}{2}.
\]
Lemma~\ref{lem:logdet}, applied with $X=C_{t+1}(S)$ and $Y=C_t(S)$, therefore gives
\begin{eqnarray}
  \left\lvert F_t(S)-F_{t+1}(S) \right\rvert
  &\le&\frac{2mr}{\lambda_{\min}(C_{t+1}(S))}\nonumber\\
  &\le&\frac{2mr\,e^{\eta_{t,1}}}{d^{\star}_t}
  =\eta_{t,2}.
  \label{eq:step3}
\end{eqnarray}
Combining \eqref{eq:step1}, optimality of $S$ for $F_{t+1}$, and \eqref{eq:step3}:
\begin{align*}
&F_t(S_t^{\star})-F_t(S)
  =\underbrace{\bigl[F_t(S_t^{\star})-F_{t+1}(S_t^{\star})\bigr]}_{\le\;\eta_{t,1}}\\
&\qquad \qquad \qquad \quad+\underbrace{\bigl[F_{t+1}(S_t^{\star})-F_{t+1}(S)\bigr]}_{\le\;0}\\
&\qquad \qquad \qquad \quad+\underbrace{\bigl[F_{t+1}(S)-F_t(S)\bigr]}_{\le\;\eta_{t,2}} \le B_t,
\end{align*}
which is \eqref{eq:objective_loss}.

Finally, assume $M_t(P)>B_t$, let $S\in\operatorname*{arg\,max} F_{t+1}$, and suppose for the sake of contradiction that $d_{\mathrm{match}}(S_t^{\star},S)>P$. Then $S$ is feasible in the minimum defining $M_t(P)$, so $F_t(S_t^{\star})-F_t(S)\geM_t(P)>B_t$, contradicting \eqref{eq:objective_loss}. Therefore, $d_{\mathrm{match}}(S_t^{\star},S)\le P$ which proves the statement.
<!-- end proof-6 -->
<!-- corollary-1: body_md -->
\label{cor:simple}
Under the hypotheses of Theorem~\ref{thm:main} except \eqref{eq:smallness}, suppose instead that
\begin{equation}
\begin{gathered}

<!-- lemma-4: body_md -->
\label{lem:dk}
Under Assumption~\ref{ass:spectral}, the rank-$K$ spectral projector $\Pi_{t+1}$ is uniquely determined and
\begin{equation}
  \rho_t=\left\lVert \Pi_t-\Pi_{t+1} \right\rVert_2\le\frac{\varepsilon}{\delta-\varepsilon}\le\frac{4\varepsilon}{3\delta}\leq\frac{4\varepsilon}{\delta}.
  \label{eq:dk_bound}
\end{equation}
<!-- end lemma-4 -->
<!-- proof-9: body_md -->
Write $L_{t+1}=L_t+\Delta$ with $\left\lVert \Delta \right\rVert_2\le\varepsilon$. By Weyl's inequality \cite[Cor.~4.3.15]{hornjohnson2013},
\[
  \lambda_{K+1}(L_{t+1})-\lambda_K(L_{t+1})\ge\delta-2\varepsilon>0,
\]
so the rank-$K$ spectral projector of $L_{t+1}$ is well defined. Let $U_t^\perp$ consist of eigenvectors of $L_t$ for $\lambda_{K+1}(L_t),\dots,\lambda_n(L_t)$, with $(U_t^\perp)^\top L_tU_t^\perp=\Lambda_t^\perp$, and let $U_{t+1}$ contain the first $K$ eigenvectors of $L_{t+1}$, with $L_{t+1}U_{t+1}=U_{t+1}\Lambda_{t+1}$. Put $X:=(U_t^\perp)^\top U_{t+1}$. Then
\begin{eqnarray*}
  \Lambda_t^\perp X-X\Lambda_{t+1}
  &=&(U_t^\perp)^\top L_tU_{t+1}\\
  &&-(U_t^\perp)^\top L_{t+1}U_{t+1}\\
  &=&-(U_t^\perp)^\top \Delta U_{t+1}.
\end{eqnarray*}
Every eigenvalue of $\Lambda_t^\perp$ is at least $\lambda_{K+1}(L_t)$ and every eigenvalue of $\Lambda_{t+1}$ is at most $\lambda_K(L_{t+1})\le\lambda_K(L_t)+\varepsilon$, so the two groups of eigenvalues are separated by at least $\delta-\varepsilon>0$. The Davis--Kahan $\sin\Theta$ theorem \cite{davis1970rotation,stewart1990matrix} gives
\[
  \left\lVert X \right\rVert_2\le\frac{\left\lVert (U_t^\perp)^\top \Delta U_{t+1} \right\rVert}{\delta-\varepsilon}\le\frac{\varepsilon}{\delta-\varepsilon}.
\]
For orthogonal projectors of equal rank, $\left\lVert \Pi_t-\Pi_{t+1} \right\rVert_2=\left\lVert (I-\Pi_t)\Pi_{t+1} \right\rVert_2$ \cite[Thm.~I.5.5]{stewart1990matrix}, and
\begin{eqnarray*}
  (I-\Pi_t)\Pi_{t+1}&=&U_t^\perp\bigl[(U_t^\perp)^\top U_{t+1}\bigr]U_{t+1}^\top\\
  &=&U_t^\perp XU_{t+1}^\top .
\end{eqnarray*}
Multiplication by matrices with orthonormal columns preserves the nonzero singular values, so $\left\lVert \Pi_t-\Pi_{t+1} \right\rVert_2=\left\lVert X \right\rVert_2$. And so finally, $\varepsilon<\delta/4$ gives $\delta-\varepsilon>3\delta/4$, which gives the last two inequalities in \eqref{eq:dk_bound}.

<!-- lemma-5: body_md -->
\label{lem:sturm}
Let $L$ be the Laplacian of a path on $n$ vertices with strictly positive edge weights. Then the eigenvalues of $L$ are simple,
 $ 0=\lambda_1<\lambda_2<\cdots<\lambda_n,$
and, after deleting the zero coordinates,
an eigenvector for $\lambda_j$ has exactly $j-1$ sign changes. In particular, a nonconstant eigenvector with exactly one sign change belongs to $\lambda_2$.
<!-- end lemma-5 -->
<!-- proof-11: body_md -->
A positively weighted path Laplacian is a real symmetric tridiagonal (Jacobi) matrix with nonzero off-diagonal entries $-w_i<0$. For any $\lambda$, deleting the first row and last column of $L-\lambda I$ leaves an
upper-triangular matrix with diagonal entries $-w_1,\dots,-w_{n-1}\ne0$, so $L-\lambda I$ has
rank at least $n-1$; every eigenvalue therefore has a one-dimensional eigenspace, and since $L$
is symmetric, its eigenvalues are simple. Fiedler's oscillation theorem \cite[Thm.~2.3]{fiedler1975acyclic} states that, after deleting zero coordinates, an eigenvector for $\lambda_j$ has exactly $j-1$ sign changes; see also \cite{fiedler1975property}. Therefore an eigenvector with exactly one sign change corresponds to $\lambda_2$.

<!-- theorem-2: body_md -->
\label{thm:impossibility}
There is no function $g:(0,\infty)^2\to\mathbb{R}_{\ge0}$ satisfying both:
\begin{enumerate}[label=(\alph*),leftmargin=*]
\item for every fixed $\delta>0$, $\displaystyle\lim_{\varepsilon\to 0^+}g(\varepsilon,\delta)=0$.
\item for every $\varepsilon>0$ and every pair of connected weighted-graph Laplacians $L_t,L_{t+1}$ on a common vertex set and common edge set with $\left\lVert L_t-L_{t+1} \right\rVert_2\le\varepsilon$ and $\delta_t:=\lambda_3(L_t)-\lambda_2(L_t)>0$, every reference maximizer $S_t^{\star}$ and every $S\in\operatorname*{arg\,max} F_{t+1}$ of the one-sensor, $K=2$ objective satisfy $d_{\mathrm{match}}(S_t^{\star},S)\le g(\varepsilon,\delta_t)$.
\end{enumerate}
In more precise words, for every $n\ge4$, there is a positively weighted path $L_t$ on $n$ vertices with fixed baseline eigengap $\delta_n>0$ and a sequence of positively weighted path Laplacians $L^{(k)}$ on the same edge set such that $\varepsilon_k:=\left\lVert L^{(k)}-L_t \right\rVert_2\to 0^+$, while the optimizer moves a matching distance of $n-1$. The same holds under any fixed tie-breaking rule at time $t$.
<!-- end theorem-2 -->
<!-- proof-12: body_md -->
Assume for the sake of contradiction that there exists a function
$g:(0,\infty)^2\to\mathbb{R}_{\ge0}$ satisfying both (a) and (b).
We construct a sequence of perturbations that violates this property.

Set $n\ge4$. For $1\le i\le n$, define
\[
a_i:=n+1-2i,
\qquad
q_i:=\Bigl(i-\frac{n+1}{2}\Bigr)^2-\frac{n^2-1}{12}.
\]
Then set $u_i(\theta):=a_i+\theta q_i.$ Both $a$ and $q$ have zero sum, the first by antisymmetry $a_{n+1-i}=-a_i$ and the second because $\sum_{i=1}^n\bigl(i-\frac{n+1}2\bigr)^2=\frac{n(n^2-1)}{12}$. Hence $u(\theta)\perp\mathbf1$ for every $\theta$. Write $A_i(\theta):=\sum_{j\le i}u_j(\theta)$
for the partial sums. At $\theta=0$,
$a_i-a_{i+1}=2\;$ for $1\le i\le n-1,$ and $A_i(0)=i(n-i)>0$ for $1\le i\le n-1,$ and 
\begin{gather*}
  a_1=n-1>0>1-n=a_n,\\
  \left\lvert a_1 \right\rvert=\left\lvert a_n \right\rvert=n-1>\max_{2\le i\le n-1}\left\lvert a_i \right\rvert=n-3 .
\end{gather*}
$u_i(\theta)$, their consecutive differences, and the partial sums $A_i(\theta)$ depend continuously on $\theta$. Therefore, all strict inequalities that hold at $\theta=0$ also hold for sufficiently small $\left\lvert \theta \right\rvert$. And thus, there exists $\theta_0>0$ such that, whenever $\left\lvert \theta \right\rvert<\theta_0$:
\begin{enumerate}[label=(\roman*)]
\item $u_1(\theta)>u_2(\theta)>\cdots>u_n(\theta)$;
\item $A_i(\theta)>0$ for $1\le i\le n-1$;
\item $u_1(\theta)>0>u_n(\theta)$;
\item $\max_i\left\lvert u_i(\theta) \right\rvert$ is attained only at $i\in\{1,n\}$;
\item $u_1(\theta)$ and $u_n(\theta)$ have the same signs as $a_1$ and $a_n$.
\end{enumerate}
We can take $\theta_0$ to be the minimum of the finitely many positive radii by these continuity arguments.

For $\left\lvert \theta \right\rvert<\theta_0$, define the path edge weights
\[
w_i(\theta):=
\frac{A_i(\theta)}{u_i(\theta)-u_{i+1}(\theta)},
\text{  for } 1\le i\le n-1.
\]
By (i) and (ii), the numerator and denominator are positive, so $w_i(\theta)>0$. Let $L(\theta)$ be the corresponding weighted-path Laplacian. Since $w_i(\theta)\bigl(u_i(\theta)-u_{i+1}(\theta)\bigr)=A_i(\theta)$,
\begin{align*}
  \bigl(L(\theta)u(\theta)\bigr)_1&=u_1(\theta),\\
  \bigl(L(\theta)u(\theta)\bigr)_i&=u_i(\theta)\qquad (2\le i\le n-1),\\
  \bigl(L(\theta)u(\theta)\bigr)_n&=u_n(\theta),
\end{align*}
where the last equality uses $\sum_{j=1}^n u_j(\theta)=0$. Thus $L(\theta)u(\theta)=u(\theta)$, so $u(\theta)$ is an eigenvector with eigenvalue $1$. By (i) and (iii), after deleting a possible zero coordinate, $u(\theta)$ has exactly one sign change. Lemma~\ref{lem:sturm} therefore gives $\lambda_2(L(\theta))=1$ for all $\left\lvert \theta \right\rvert<\theta_0$. Since the eigenvalues are simple, $\delta_n:=\lambda_3\bigl(L(0)\bigr)-1>0.$

For $K=2$ the low-frequency projector of $L(\theta)$ is
\begin{gather*}
\Pi(\theta)=\frac1n\mathbf1\,\mathbf1^\top+\widehat u(\theta)\widehat u(\theta)^\top,~ {\rm and ~}
  \widehat u(\theta)=\frac{u(\theta)}{\left\lVert u(\theta) \right\rVert_2},
\end{gather*}
because $\lambda_1=0$ has eigenvector $\mathbf1/\sqrt n$ and $\lambda_2=1$ has eigenvector $\widehat u(\theta)$. With one sensor, $m=1\le K$ and
\[
  F(\{i\})=\log\Pi(\theta)_{ii}=\log\Bigl(\tfrac1n+\widehat u_i(\theta)^2\Bigr),
\]
so maximizing $F$ over single vertices is equivalent to maximizing $\left\lvert u_i(\theta) \right\rvert$.

At $\theta=0$, the maximizers are $\{1\}$ and $\{n\}$, by $\left\lvert a_1 \right\rvert=\left\lvert a_n \right\rvert>\left\lvert a_i \right\rvert$ for interior $i$. For $\theta\ne0$ small, property (v) gives $\left\lvert u_1(\theta) \right\rvert=a_1+\theta q_1$ and $\left\lvert u_n(\theta) \right\rvert=a_1-\theta q_1$. This then gives
\begin{gather*}
  \left\lvert u_1(\theta) \right\rvert-\left\lvert u_n(\theta) \right\rvert=2\theta q_1,\\
  q_1=q_n=\frac{(n-1)(n-2)}{6}>0 .
\end{gather*}
By (iv) no interior vertex can be maximal, so the unique maximizer is $\{1\}$ when $\theta>0$ and $\{n\}$ when $\theta<0$. Set $L_t:=L(0)$ (whose baseline eigengap is $\delta_t=\delta_n$). For sufficiently small $\theta\ne0$, $L(\theta)\ne L(0)$: otherwise $u(0)$ and $u(\theta)$ would both lie in the eigenspace of $L(0)$ for the simple eigenvalue $1$, and thus be proportional, forcing $q$ to be a multiple of $a$.

This is impossible because $q\ne0$ is symmetric, $q_{n+1-i}=q_i$, whereas $a\ne0$ is antisymmetric, $a_{n+1-i}=-a_i$. Since each $w_i(\theta)$ is a rational function with nonvanishing denominator on $(-\theta_0,\theta_0)$, $L(\theta)\to L(0)$ in operator norm as $\theta\to0$. Therefore we may choose a sequence $\theta_k<0$ tending to zero such that the positive numbers
\[
  \varepsilon_k:=\left\lVert L(\theta_k)-L(0) \right\rVert_2
\]
are strictly decreasing. Then $\varepsilon_k\to 0^+$. Put $L^{(k)}:=L(\theta_k)$. For every $k$, the perturbed objective has the unique maximizer $\{n\}$. Choose the reference maximizer $S_t^{\star}=\{1\}$, since $\{1\}\in\operatorname*{arg\,max} F_t$. Then
\begin{gather*}
  d_{\mathrm{match}}(S_t^{\star},S)=d(1,n)=n-1\\
  \text{ for the unique }S\in\operatorname*{arg\,max} F_{t+1},
\end{gather*}
because $G_{\mathrm{mv}}$ is the unweighted path. Property (b) of a hypothetical $g$ would give $n-1\le g(\varepsilon_k,\delta_n)$ for every $k$, contradicting (a) as $k\to\infty$.

Finally, if a fixed rule selects one maximizer at time $t$, choose the sign of the perturbation opposite to the endpoint selected by the rule; the same construction produces movement $n-1$.

  d(x,q_j)&\ge& d(q_i,q_j)-d(x,q_i)\\
  &>&2P-P=P\ge d(x,q_i),
\end{eqnarray*}
so $q_i$ is strictly closer to $x$ than any other $q_j$, that is, $x\in W_i^{<}$. If $\pi$ is a bijection with $d(q_i,r_{\pi(i)})\le P$ for all $i$, then the previous argument gives $r_{\pi(i)}\in W_i^{<}$, which together with $d(q_i,r_{\pi(i)})\le P$ is \eqref{eq:compatible}.

<!-- lemma-6: body_md -->
\label{lem:over_logdet}

Let $c_S:=\lambda_{\min}(H_t(S))>0$ and let $r\ge\rho_t$ satisfy $r\le c_S/4$. Then $H_{t+1}(S)$ is positive definite and
\begin{equation}
  \left\lvert F_t^{\mathrm{ov}}(S)-F_{t+1}^{\mathrm{ov}}(S) \right\rvert\le\frac{4Kr}{c_S}.
  \label{eq:over_logdet}
\end{equation}
<!-- end lemma-6 -->
<!-- proof-20: body_md -->
Set $A_t:=\Pi_tD_S\Pi_t=U_tH_t(S)U_t^\top$, so the nonzero eigenvalues of $A_t$ are the same as those of $H_t(S)$, and likewise at time $t+1$. From
\[
  A_t-A_{t+1}=(\Pi_t-\Pi_{t+1})D_S\Pi_t+\Pi_{t+1}D_S(\Pi_t-\Pi_{t+1})
\]
and $\left\lVert D_S \right\rVert_2\le1$, $\left\lVert \Pi_t \right\rVert_2=\left\lVert \Pi_{t+1} \right\rVert_2=1$ we get $\left\lVert A_t-A_{t+1} \right\rVert_2\le2\rho_t\le2r$. Order eigenvalues nonincreasingly. By Weyl's inequality, $\left\lvert \lambda_j(A_{t+1})-\lambda_j(A_t) \right\rvert\le2r$ for all $j$. For $1\le j\le K$ we have $\lambda_j(A_t)\ge c_S$ and $r\le c_S/4$, so $2r\le c_S/2$, so $\lambda_j(A_{t+1})\ge c_S/2>0.$ Since $\operatorname{rank}A_{t+1}\le K$, these are exactly the nonzero eigenvalues of $A_{t+1}$, hence the eigenvalues of $H_{t+1}(S)$, which is thus positive definite. Setting
\[  x_j:=\frac{\lambda_j(A_{t+1})-\lambda_j(A_t)}{\lambda_j(A_t)}, \text{ and }\left\lvert x_j \right\rvert\le\frac{2r}{c_S}\le\frac12,\]
and using $\det H_t(S)=\prod_{j\le K}\lambda_j(A_t)$, $\det H_{t+1}(S)=\prod_{j\le K}\lambda_j(A_{t+1})$ together with \eqref{eq:scalar_log},
\begin{eqnarray*}
  \left\lvert F_{t+1}^{\mathrm{ov}}(S)-F_t^{\mathrm{ov}}(S) \right\rvert&=&\Bigl|\sum_{j=1}^K\log(1+x_j)\Bigr| \\
  &\le&2\sum_{j=1}^K\left\lvert x_j \right\rvert\le\frac{4Kr}{c_S}.
\end{eqnarray*} This proves \eqref{eq:over_logdet}.

<!-- theorem-4: body_md -->
\label{thm:over}

Assume $m\ge K$ and Assumption~\ref{ass:movement}. Let $r\ge\rho_t$ satisfy
\begin{equation}
\begin{gathered}
  r\le\frac{\sigma^{\star,\mathrm{ov}}_t}{8},~ {\rm and}~
  \eta^{\star,\mathrm{ov}}_t:=\frac{4Kr}{\sigma^{\star,\mathrm{ov}}_t}\le\log2 .
\end{gathered}
\label{eq:over_smallness}
\end{equation}
Then every $S\in\operatorname*{arg\,max} F_{t+1}^{\mathrm{ov}}$ satisfies
\[
F_t^{\mathrm{ov}}(S_t^{\star})-F_t^{\mathrm{ov}}(S)\le3\eta^{\star,\mathrm{ov}}_t.
\]
Consequently, for every integer $P\ge0$, one has that
\begin{equation}
\begin{gathered}
  M_t^{\mathrm{ov}}(P)>3\eta^{\star,\mathrm{ov}}_t=\frac{12Kr}{\sigma^{\star,\mathrm{ov}}_t}\\
  \Longrightarrow
  d_{\mathrm{match}}(S_t^{\star},S)\le P
  \quad\\
  \text{for every }S\in\operatorname*{arg\,max} F_{t+1}^{\mathrm{ov}}.
\end{gathered}
\label{eq:over_implication}
\end{equation}
Under Assumption~\ref{ass:spectral} one may take $r=4\varepsilon/\delta$, in which case the hypotheses read $4\varepsilon/\delta\le\sigma^{\star,\mathrm{ov}}_t/8$ and $16K\varepsilon/(\sigma^{\star,\mathrm{ov}}_t\delta)\le\log2$, and the threshold becomes $48K\varepsilon/(\sigma^{\star,\mathrm{ov}}_t\delta)$.
<!-- end theorem-4 -->
<!-- proof-21: body_md -->
Since $0\preceq H_t(S_t^{\star})\preceq I_K$, we have $\lambda_{\min}(H_t(S_t^{\star}))\ge\det H_t(S_t^{\star})=\sigma^{\star,\mathrm{ov}}_t$,
so $r\le\sigma^{\star,\mathrm{ov}}_t/8\le\lambda_{\min}(H_t(S_t^{\star}))/4$. Lemma~\ref{lem:over_logdet} at $S=S_t^{\star}$ gives
\[
\left\lvert F_{t+1}^{\mathrm{ov}}(S_t^{\star})-F_t^{\mathrm{ov}}(S_t^{\star}) \right\rvert\le\frac{4Kr}{\sigma^{\star,\mathrm{ov}}_t}=\eta^{\star,\mathrm{ov}}_t .
\]
Let $S\in\operatorname*{arg\,max} F_{t+1}^{\mathrm{ov}}$. By optimality and this bound,
$F_{t+1}^{\mathrm{ov}}(S)\ge F_{t+1}^{\mathrm{ov}}(S_t^{\star})\ge\log\sigma^{\star,\mathrm{ov}}_t-\eta^{\star,\mathrm{ov}}_t$, so
$\det H_{t+1}(S)\ge\sigma^{\star,\mathrm{ov}}_t e^{-\eta^{\star,\mathrm{ov}}_t}\ge\sigma^{\star,\mathrm{ov}}_t/2$ because $\eta^{\star,\mathrm{ov}}_t\le\log2$. Since
$0\preceq H_{t+1}(S)\preceq I_K$, one has that
\[
\lambda_{\min}(H_{t+1}(S))\ge\frac{\sigma^{\star,\mathrm{ov}}_t}{2},
\] and thus
\[r\le\frac{\sigma^{\star,\mathrm{ov}}_t}{8}\le\frac{\lambda_{\min}(H_{t+1}(S))}{4}.
\]
Lemma~\ref{lem:over_logdet} with the two times exchanged therefore gives
\begin{eqnarray}
\left\lvert F_t^{\mathrm{ov}}(S)-F_{t+1}^{\mathrm{ov}}(S) \right\rvert&\le&\frac{4Kr}{\lambda_{\min}(H_{t+1}(S))}\nonumber\\&\le&\frac{8Kr}{\sigma^{\star,\mathrm{ov}}_t}=2\eta^{\star,\mathrm{ov}}_t .\nonumber
\end{eqnarray}
Combining the two bounds with optimality of $S$ at time $t+1$,
\begin{align*}
F_t^{\mathrm{ov}}(S_t^{\star})-F_t^{\mathrm{ov}}(S)
&=\bigl[F_t^{\mathrm{ov}}(S_t^{\star})-F_{t+1}^{\mathrm{ov}}(S_t^{\star})\bigr]\\
&\quad+\bigl[F_{t+1}^{\mathrm{ov}}(S_t^{\star})-F_{t+1}^{\mathrm{ov}}(S)\bigr]\\
&\quad+\bigl[F_{t+1}^{\mathrm{ov}}(S)-F_t^{\mathrm{ov}}(S)\bigr]\\
&\le\eta^{\star,\mathrm{ov}}_t+0+2\eta^{\star,\mathrm{ov}}_t=3\eta^{\star,\mathrm{ov}}_t .
\end{align*}
If $d_{\mathrm{match}}(S_t^{\star},S)>P$, then $S$ is feasible in the minimum defining $M_t^{\mathrm{ov}}(P)$, so
$F_t^{\mathrm{ov}}(S_t^{\star})-F_t^{\mathrm{ov}}(S)\geM_t^{\mathrm{ov}}(P)>3\eta^{\star,\mathrm{ov}}_t$, a contradiction; this proves
\eqref{eq:over_implication}. Under Assumption~\ref{ass:spectral}, Lemma~\ref{lem:dk} shows that
$r=4\varepsilon/\delta$ satisfies $r\ge\rho_t$, which gives the Laplacian specialization.

<!-- theorem-5: body_md -->
\label{thm:reg}
Assume Assumption~\ref{ass:movement}.
Let $r\ge\rho_t$ with $r/\tau\le1/2$. Then for every integer $P\ge0$, and for every $S\in\operatorname*{arg\,max} F_{t+1}^\tau$, we have
\begin{equation}
\begin{gathered}
  M_t^{\tau}(P)>\frac{4mr}{\tau}
  \Longrightarrow\quad d_{\mathrm{match}}\bigl(S_t^{\star,\tau},S\bigr)\le P.
\end{gathered}
\label{eq:reg_implication}
\end{equation}
Under Assumption~\ref{ass:spectral} it suffices that $4\varepsilon/(\tau\delta)\le1/2$ and $M_t^{\tau}(P)>16m\varepsilon/(\tau\delta)$.
<!-- end theorem-5 -->
<!-- proof-23: body_md -->
Let $S\in\operatorname*{arg\,max} F_{t+1}^\tau$. By Lemma~\ref{lem:reg} applied at $S_t^{\star,\tau}$ and at $S$, and by optimality of $S$ at time $t+1$,
\begin{align*}
  F_t^\tau(S_t^{\star,\tau})-F_t^\tau(S)
  &=\bigl[F_t^\tau(S_t^{\star,\tau})-F_{t+1}^\tau(S_t^{\star,\tau})\bigr]\\
  &\quad+\bigl[F_{t+1}^\tau(S_t^{\star,\tau})-F_{t+1}^\tau(S)\bigr]\\
  &\quad+\bigl[F_{t+1}^\tau(S)-F_t^\tau(S)\bigr]\\
  &\le\eta_t^\tau+0+\eta_t^\tau=2\eta_t^\tau=\frac{4mr}{\tau}.
\end{align*}
If $d_{\mathrm{match}}(S_t^{\star,\tau},S)>P$ the margin definition forces the opposite strict inequality. The Laplacian statement follows from Lemma~\ref{lem:dk} with $r=4\varepsilon/\delta$.
