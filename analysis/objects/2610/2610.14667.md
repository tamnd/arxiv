<!-- definition-1: body_md -->
\label{def:narrowdomain}
We say that a domain $\Omega \subset \mathbb{R}^n$ is $(m,M)$-thin if
 
\begin{align*}                        
\Omega \subset \{x = (x_1,y_1,\dots,y_\ell,z_1,\dots,z_m) \in \mathbb{R}^n : x_1 > 0, |z| < M\}.
 
\end{align*}
Moreover, for an $(m,M)$-thin domain $\Omega$ we use the notation $\Sigma_L = \overline \Omega \cap \{x : x_1 = L\}$ for $L \geq 0$, and $\Gamma^+ = \partial \Omega \cap \{x_1 > 0\}$ so that $\partial \Omega = \Sigma_0 \cup \Gamma^+$. Finally, we call the domain 
$
\{x=(x_1,y,z)\in\mathbb R^n:x_1>0,\ |z|<M\}
$
the maximal $(m,M)$-thin domain.
<!-- end definition-1 -->
<!-- lemma-1: body_md -->
\label{le:comp-weak-some-sence}
Let $\Omega$ be a bounded domain and assume the monotonicity in \eqref{eq:ass_drift_super}. Let $u \in USC(\overline{\Omega})$ be a subsolution (in some weak sense)
and $v$ a classical strict supersolution of \eqref{eq:main} in $\Omega$, continuous on $\overline{\Omega}$.
If $u \leq v$ on $\partial \Omega$ then $u < v$ in $\Omega$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Assume that $v$ is a classical strict supersolution and also, by way of contradiction, that $u \geq v$ somewhere in $\Omega$. 
By $USC$ the function $u-v$ then attains a maximum $\geq 0$ at some point $x_0 \in \Omega$.
Since $v \in C^2(\Omega)$, $u - v$ has a maximum at $x_0$, and $u$ is a viscosity subsolution, it follows by definition that
 
\begin{align}\label{eq:johej-lemma1}
F(x_0, u(x_0), Dv(x_0), D^2 v(x_0)) \leq 0.
\end{align}
 
But since $v$ is a classical strict supersolution, 
we have 
 
$
F(x,v(x), Dv(x), D^2v(x)) > 0                                           
$
 
whenever $x \in \Omega$, 
and as $u(x_0) \geq v(x_0)$ it follows from the monotonicity in \eqref{eq:ass_drift_super} that
 
$$
F(x_0, u(x_0), Dv(x_0), D^2 v(x_0)) \geq F(x_0, v(x_0), Dv(x_0), D^2 v(x_0)) > 0.
$$
 
This contradicts \eqref{eq:johej-lemma1} and hence we have proved the lemma.
<!-- end proof-1 -->
<!-- theorem-1: body_md -->
\label{th:exp-decay-pipe}
Let $\Omega$ be an $(m,M)$-thin domain,  
assume that \eqref{eq:ass_drift_super} holds with 
$\lambda=\lambda(x_1), \Lambda = \Lambda(x_1)$,      
                                  
and
 
$$
\Phi(x,r,p) \leq C \widehat \Phi(x_1) \lambda(x_1) \quad \text{whenever} \quad (x,r,p) \in \Omega\times \mathbb{R} \times \mathbb{R}^n
$$
 
for some positive, nonincreasing $\widehat\Phi \in C^2([0,\infty))$
satisfying $\widehat \Phi'' > 0$ on $[0,\infty)$ and $\widehat \Phi(0) = 1$.
If $u$ is a viscosity subsolution of \eqref{eq:main} in $\Omega$, 
satisfying $u \leq C_1$ on $\overline \Omega$ and $u \leq 0$ on $\Gamma^+$,  
then for every $\bar \epsilon > 0$ there exists $A = A(\bar \epsilon, C, C_1, m, M) > 0$ such that
 
\begin{align*}
\frac{\widehat\Phi''(x_1)}{\widehat\Phi (x_1)} \leq 
\left(\kappa - \bar \epsilon \right) \frac{\lambda(x_1)}{\Lambda(x_1)}  \quad \text{for all}\quad x_1 > 0
\quad \implies 
\quad  u(x) \leq A\, \widehat \Phi(x_1)                              
\end{align*}
  
whenever $x \in  \Omega$ and 
where $\kappa$ is as in \eqref{eq:kappa}.
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{cor:exp-decay-pipe}
Let $\Omega$, $\lambda/\Lambda$, $\widehat\Phi$, $u$ and $\kappa$ be as in 
Theorem \ref{th:exp-decay-pipe} and assume \eqref{eq:ass_drift_super} with $\Phi \leq 0$. 
Suppose also that the boundary growth estimate in \eqref{eq:for_corr} holds for $u$. 
Then there exists $A = A(\bar c, C_1, M, m) > 0$ such that
 
\begin{align*}
            
\frac{\widehat\Phi''(x_1)}{\widehat\Phi (x_1)} \leq  \kappa\, \frac{\lambda(x_1)}{\Lambda(x_1) } \quad \text{for all}\quad x_1 > 0
\quad \implies             
\quad  u(x) \leq A\, \widehat \Phi(x_1)
\end{align*}
  
whenever $x \in  \Omega$.
<!-- end corollary-1 -->
<!-- lemma-2: body_md -->
\label{le:supersol}
Suppose that \eqref{eq:ass_drift_super} holds and that there are positive $C^2(E)$-functions $f = f(x_1)$, $g = g(y)$ and $h = h(z) = \bar h(|z|)$ in $E$ such that
$f'' \geq 0$, $D^2 g  \geq 0$, $\Delta h \leq 0$, $\bar h$ is non-increasing. If
 
\begin{align*}   
   - \frac{\Delta_z h}{h} > \frac{\Phi(x,\hat u, D\hat u)}{ \lambda\, \hat u} +\frac{\Lambda}{\lambda}  \left( \frac{f''}{f}  +  \frac{\Delta_y g}{g}  \right)\quad \text{in} \quad E,
\end{align*}
 
where $\hat u(x) = f(x_1) g(y) h(z)$, 
then $\hat u$ is a strict classical supersolution of \eqref{eq:main} in $E$.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
We will prove that the two inequalities
 
\begin{align}\label{eq:helplemmaeq1}
   -& \Lambda\left( \left(D^2\hat u\right)_{11}^+ + \text{Tr}\left(D^2 \hat u\right)^+_y  +\left(\text{Tr}\left(D^2 \hat u\right)_z\right)^+ \right) \notag\\
&+ \lambda\left( \left(D^2\hat u\right)_{11}^- + \text{Tr}\left(D^2 \hat u\right)^-_y +\left(\text{Tr}\left(D^2 \hat u\right)_z\right)^- \right)  \notag\\
 
&\geq  -\Lambda   \left(f'' g h + f \Delta_y g h \right) -  \lambda f g \Delta_z h, 
\end{align}
 
and, whenever $|D\hat u| \neq 0$,  
 
\begin{align}\label{eq:helplemmaeq2}
- \Lambda \operatorname{Tr} D^2\hat u + (\Lambda-\lambda)
 \frac{\langle D^2\hat u D\hat u, D\hat u\rangle}{|D\hat u|^2}
 \geq  -\Lambda   \left(f'' g h + f \Delta_y g h \right) -  \lambda f g \Delta_z h
\end{align}
 
hold in $E$. 
The result then follows by \eqref{eq:ass_drift_super}, \eqref{eq:operatorexpression} and the inequality in the lemma because 
 
\begin{align*}
F(x,\hat u,D\hat u,D^2\hat u) &\geq \mathcal P_{\lambda,\Lambda}(D\hat u,D^2\hat u) - \Phi(x,\hat u, D\hat u)  \\
&\geq  -\Lambda   \left(f'' g h + f \Delta_y g h \right) -  \lambda f g \Delta_z h  - \Phi(x,\hat u, D\hat u) > 0,
\end{align*}
 
and thus $\hat u$ is a strict classical supersolution of \eqref{eq:main} in $E$.
By sign assumptions, we have
 
$$
\left(D^2 \hat u\right)^+_{11} = f'' g h \quad \text{and} \quad 
\left(D^2 \hat u\right)^-_{11} = 0,
$$
 
and since $D^2 g  \geq 0$ it also follows that
 
$$
\text{Tr}\left(D^2 \hat u\right)^+_y =  f \Delta g_y h \quad \text{and} \quad \text{Tr}\left(D^2 \hat u\right)^-_y = 0.
$$
 
Moreover, 
 
$$
\left(\text{Tr}\left(D^2 \hat u\right)_z\right)^+ = f g \left(\Delta_z h\right)^+ = 0 \quad \text{and} \quad \left(\text{Tr}\left(D^2 \hat u\right)_z\right)^- = -f g \Delta_z h.
$$
 
Insertion of these equalities in \eqref{eq:helplemmaeq1} establishes its inequality.
 
To prove \eqref{eq:helplemmaeq2} we first claim that 
 
\begin{align}\label{eq:claimforlemma}
\frac{\Delta_\infty\hat u}{|D\hat u|^2}
\geq                 
fg\,\Delta_z h \quad\text{in}\quad E,
\end{align}
 
in which $\Delta_\infty\hat u=\langle D^2\hat uD\hat u,D\hat u\rangle$, giving
 
\begin{align*}
-& \Lambda \operatorname{Tr} D^2\hat u + (\Lambda-\lambda)
 \frac{\langle D^2\hat u D\hat u, D\hat u\rangle}{|D\hat u|^2}
=
-\Lambda\Delta\hat u + (\Lambda-\lambda) \frac{\Delta_\infty\hat u}{|D\hat u|^2}\notag\\
&\geq 
-\Lambda\bigl(f''gh + fh\,\Delta_y g + fg\,\Delta_z h\bigr) + (\Lambda-\lambda)fg\,\Delta_z h\\
& =
 -\Lambda   \left(f'' g h + f \Delta_y g h \right) -  \lambda f g \Delta_z h,
\end{align*}
 
which proves \eqref{eq:helplemmaeq2}. 
 
To establish claim \eqref{eq:claimforlemma}, we write
 
\begin{align*}
\Delta_\infty \hat u
={}&\hat u_{x_1}^2\hat u_{x_1x_1}
+\sum_{i,j=1}^{\ell}\hat u_{y_i} \hat u_{y_j}\hat u_{y_i y_j}
+\sum_{i,j=1}^{m}\hat u_{z_i}\hat u_{z_j}\hat u_{z_i z_j}\\
&+2\sum_{i=1}^{\ell}\hat u_{x_1}\hat u_{y_i}\hat u_{x_1y_i}
+2\sum_{j=1}^{m}\hat u_{x_1}\hat u_{z_j}\hat u_{x_1z_j} 
+2\sum_{i=1}^{\ell}\sum_{j=1}^{m}
  \hat u_{y_i}\hat u_{z_j}\hat u_{y_i z_j}\\
 
={}&f''(f')^2g^3h^3
+f^3h^3\langle D_y^2g\,D_yg,D_yg\rangle
+f^3g^3\langle D_z^2h\,D_zh,D_zh\rangle\\
&+2f(f')^2gh^3|D_yg|^2
+2f(f')^2g^3h|D_zh|^2
+2f^3gh|D_yg|^2|D_zh|^2.
\end{align*}
 
From our choice of $f,g$ and $h$, 
we see that $f''(f')^2g^3h^3 \geq 0$, 
$\langle D_y^2g\,D_yg,D_yg\rangle  \geq 0$, 
and that the last three terms are also non-negative.
 
Since $h$ is radial, $D_z h$ is an eigenvector of $D_z^2h$
with eigenvalue $\bar h''$. Thus
 
$$
\Delta_\infty \hat u \geq f^3g^3\langle D_z^2h\,D_zh,D_zh\rangle = f^3g^3\bar h'' (\bar h')^2,
$$
 
and
$\Delta_z h = \bar h'' + \frac{m-1}{|z|} \bar h'$,                 
giving, since $\bar h' \leq 0$, that
 
$$
\bar h'' (\bar h')^2 = \left(\Delta_z h -\frac{m-1}{|z|} \bar h'\right) (\bar h')^2 \geq  (\bar h')^2 \Delta_z h
$$
 
whenever $|z| \neq 0$ (when $|z| = 0$ both sides of the inequality vanish). 
Hence,
 
$$
\Delta_\infty \hat u \geq  f^3 g^3  (\bar h')^2 \Delta_z h \geq 
f g |D\hat u|^2 \Delta_z h,
$$
 
where the last inequality follows since $\Delta_z h \leq 0$ and $|D_z \hat u| \leq |D\hat u|$. 
This proves claim \eqref{eq:claimforlemma} and therefore the proof of Lemma \ref{le:supersol} is complete.

We choose $\hat u_L(x) = f(x_1) g(y) h(z)$ where
 
\begin{align*}
f(x_1) = A \left(\widehat\Phi(x_1) -\alpha_L\left(\frac{x_1}{L}\right)^2 + \frac{x_1}{L} \right) \quad \text{and} \quad
g(y) = 1 + \epsilon_L \sum_{i = 1}^l y_i^2,
\end{align*}
 
for some small $0 < \alpha_L, \epsilon_L$ to be chosen later, 
and observe that then the inequality in Lemma \ref{le:supersol} is implied in $\Omega \cap\{\mathbb{R}^n : 0 < x_1 < L\}$ by
 
\begin{align}\label{eq:johejsan1}   
   - \frac{\Delta_z h}{h} \geq \frac{\Phi(x,\hat u_L, D\hat u_L)}{ \lambda\, \hat u_L} +\frac{\Lambda}{\lambda}  \left( \frac{ \widehat\Phi'' - \frac{2 \alpha_L}{L^2}}{\widehat \Phi}  +  2 l \epsilon_L  \right).
\end{align}
 









\noindent
To find the `best' $h = h(z) = \bar h(|z|)$ we solve for positive solutions of
 
\begin{align}\label{eq:MM+}
\Delta_z h(z) + \kappa^+ \,h(z) = 0\quad \text{in} \quad \{z\in \mathbb{R}^m : |z| < M^+\} \quad\text{with}\quad \bar h(M^+) = 0,                                  
\end{align}
 
in which $M^+ \geq M$ will be specified later. 
The well known solutions are 
 
\begin{align*}                          
h(z)=c |z|^{-(m-2)/2}
J_{(m-2)/2}\!\left(\sqrt{\kappa^+}|z|\right),
\end{align*}
 
where $c$ is chosen so that \(\bar h(0)=1\), 
and $J_\nu$ denotes the Bessel function of the first kind of order $\nu$.
The solution $h$ is $C^{\infty}$ and the boundary conditions and positivity imply that $\kappa^+$ is given by \eqref{eq:kappa} but with $M$ replaced by $M^+$. 






For each \(L>0\), choose \(\alpha_L>0\) sufficiently small that
\(f''\geq0\) on \(0<x_1<L\); this is possible since
\(\widehat\Phi''>0\) on \([0,L]\). We then choose \(\epsilon_L>0\)
so that
 
\begin{align}\label{eq:jojo}
    \frac{ \widehat\Phi''(x_1) - \frac{2\alpha_L}{L^2}}{\widehat \Phi(x_1)}  +  2 l \epsilon_L  \leq 
      \frac{ \widehat\Phi''(x_1)}{\widehat \Phi(x_1)},
\end{align}
 
and since \(\widehat\Phi\) is positive, nonincreasing, and satisfies
\(\widehat\Phi(0)=1\), this inequality is equivalent to
\(2lL^2\epsilon_L\leq 2\alpha_L/\widehat\Phi(x_1)\). Thus, when
\(\ell>0\), we may take \(\epsilon_L=\alpha_L/(\ell L^2)\). 
When \(\ell=0\), we set \(\epsilon_L=0\).
 
Therefore, by \eqref{eq:jojo}, \eqref{eq:johejsan1} and our choice of $h$ it is enough to prove
 
\begin{align}\label{eq:johejsan}   
   \kappa^+ \geq \frac{\Phi(x,\hat u_L, D\hat u_L)}{ \lambda\, \hat u_L} +\frac{\Lambda}{\lambda}  \frac{ \widehat\Phi''(x_1)}{\widehat \Phi(x_1)}.
\end{align}
 












          
We now handle the lower-order terms.  
Take $A = A(\bar \epsilon, C, m, M) \geq \frac{2 C}{\bar \epsilon \bar h(M)}$ and observe that then
 
\begin{align}\label{eq:choice_source}
\frac{\Phi(x,\hat u_L, D\hat u_L)}{ \lambda\, \hat u_L} \leq \frac{C}{A \bar h(M)} \leq \frac{\bar \epsilon}2,
\end{align}
 
and thus \eqref{eq:johejsan} is implied by
 
\begin{align*}
\frac{\lambda(x_1)}{\Lambda(x_1)}\left(\kappa^+ - \frac{\bar \epsilon}2 \right) \geq \frac{\widehat\Phi''(x_1)}{\widehat \Phi(x_1)}. 
\end{align*}
 
Now we can, by taking $M^+$ close enough to $M$, ensure that $\kappa^+ - \bar\epsilon/2 \geq \kappa -\bar\epsilon$ so that it is enough to prove
 
\begin{align}\label{eq:ineq_theorem}
\frac{\lambda(x_1)}{\Lambda(x_1)} \left(\kappa - {\bar \epsilon} \right) \geq    \frac{\widehat\Phi''(x_1)}{\widehat \Phi(x_1)}, \quad \text{where} \quad \kappa = \left(\frac{j_{\frac{(m-2)}{2}, 1} }{M}\right)^2.
\end{align}
 
This is the inequality in the theorem, and we have now proved that it implies 
the inequality in Lemma \ref{le:supersol}. 
This gives us a family of strict classical supersolutions $\hat u_L$ of \eqref{eq:main} in $\Omega \cap\{\mathbb{R}^n : 0 < x_1 < L\}$, whenever $L > 0$. 





We now perform the comparison argument, and to do so we add a boundary in the `open' $y$-directions if $\ell > 0$:  
For each $L > 0$ we construct the domain
 
\begin{align*}
&\Omega_L = \Omega \cap \{\mathbb{R}^n : 0 < x_1 < L, |y|^2 < \omega_L(x_1), |z| < M\}, \notag\\        
&\text{where} \quad  
\omega_L(x_1) = A f(x_1)^{-1} \epsilon_L^{-1},
\end{align*}
 
and claim that for $A$ large enough, independent of $L$, 
 
\begin{align}\label{eq:bchatu}
    \hat u_L \geq C_1 \quad \text{on}\quad \partial \Omega_L \setminus \Gamma^+.                                       
\end{align}
 
We assume that claim \eqref{eq:bchatu} holds for the moment 
and prove it later. 
Observe that $\omega_L \to \infty$ uniformly as $L \to \infty$, and therefore $\Omega_L \to \Omega$ in the limit $L \to \infty$.  
Moreover, since $u \leq C_1$ on $\overline \Omega$, $u\leq 0$ on $\Gamma^+$ by assumption, 
and $\hat u_L \geq 0$ on $\overline \Omega_L$ by construction, 
claim \eqref{eq:bchatu} and Lemma \ref{le:comp-weak-some-sence} allow us to compare $u$ with $\hat u_L$ for any $L > 0$. 
In particular, we conclude 
 
\begin{align*}
    \hat u_L \geq u \quad \text{on} \quad \partial\Omega_L \quad \text{giving}\quad  \hat u_L \geq u \quad \text{in} \quad \Omega_L.
\end{align*}
 
Now, since $L>0$ is arbitrary, $\Omega_L \to \Omega$ as $L \to \infty$, we have
 
\begin{align*}
u (x) \leq \inf_{L>x_1} \hat u_L(x) \leq  A \widehat \Phi(x_1) \quad \text{whenever}\quad x \in \Omega,
\end{align*}
 
which is the desired estimate.
 
To finish the proof, it only remains to establish the claim \eqref{eq:bchatu}. 
To do so, observe that 
 
\begin{align*}
  \left(\partial \Omega_L \setminus \Gamma^+\right) \subset \left(\Sigma_0 \cup  \Sigma_L \cup \{\mathbb{R}^n : |y|^2 = \omega_L\}\right).  
\end{align*}
 
Since $g \geq  1$ and $\bar h(M) \leq h$ on $\overline \Omega_L$, we have
 
\begin{align*}
\hat u_L(x) \geq \bar h(M) f(x_1) = A \bar h(M)\left(\widehat\Phi(x_1) - \alpha_L\left(\frac{x_1}{L}\right)^2 + \frac{x_1}{L} \right)  \quad \text{whenever} \quad  x \in \overline \Omega_L.  
\end{align*}
 
This gives $\hat u_L \geq A \bar h(M)/2 \geq C_1$ on $\Sigma_0 \cup \Sigma_L$ if $A$ is large enough and if we assume, as we may, 
that $\alpha_L \leq \frac{1}{2}$. 
In particular, we now take $A$ according to
 
\begin{align}\label{eq:choice_boundary}
A = A(\bar \epsilon, C, C_1, m, M) \geq \frac{2}{\bar h(M)} \max\left\{ C_1, \frac{C}{\bar \epsilon}\right\}.
\end{align}
 
It remains to ensure that $\hat u_L \geq C_1$ on $\{\mathbb{R}^n : |y|^2 = \omega_L\}$, but this follows by construction since on the set 
 
\begin{align*}
 |y|^2  = \omega_L(x_1) =  A f(x_1)^{-1} \epsilon_L^{-1} \quad \text{for}\quad x_1 \in  (0,L), 
\end{align*}
 
it holds that
 
\begin{align*}
\hat u_L \geq \bar h(M) f(x_1) \left(1 + \epsilon_L |y|^2\right) =  \bar h(M) f(x_1) \left(1 + \frac{A}{f(x_1)}\right) \geq C_1 \geq u.                                      
\end{align*}
 
The proof of Theorem \ref{th:exp-decay-pipe} is complete.

Following the proof of Theorem \ref{th:exp-decay-pipe} but with $\Phi\equiv0$ we see that we can take $\bar \epsilon = 0$ and  ignore \eqref{eq:choice_source}. 
Therefore, we can take $M^+ = M$ and thereby obtain the desired family of strict supersolutions whenever \eqref{eq:ineq_theorem} holds with $\bar\epsilon = 0$.

We next update the choice of $A$ in \eqref{eq:choice_boundary}, 
and to do so we assume in the following that $\Omega$ is a maximal $(m,M)$-thin domain since any smaller allowed domain $\Omega$ makes the situation easier. 
This time we have $\bar h(M) = 0$, i.e. the barrier vanishes on the boundary where $|z| = M$. 
 
The subsolution $u$ may jump immediately from 0 to $C_1$ on $\Sigma_0$, 
making it impossible to dominate by $\hat u_L$ on that set. 
Therefore, we replace it by $\Sigma_1$ and make comparison in the corresponding slightly smaller set. 
 
To dominate the subsolution $u$ near $|z| = M$ on $\Sigma_1$ and $\Sigma_L$ we need assumption \eqref{eq:for_corr} and the fact that the Bessel function has nonvanishing gradient near its first zero. 
In particular, there exist $\delta = \delta(m,M) > 0$ and $\hat c = \hat c(m,M) > 0$ such that
 
\begin{align}\label{eq:besselgradient}
    |J_\nu'(r)| \geq \hat c > 0
    \quad \text{whenever} \quad
    r \in (j_{\nu,1}-\delta,\, j_{\nu,1}+\delta).
\end{align}
 
By \eqref{eq:besselgradient}, assumption \eqref{eq:for_corr}, $u \leq C_1$, 
and by the construction of $\hat u_L$ it follows that there exists 
$A = A(\bar c, C_1, m, M)$ such that 
 
\begin{align}\label{eq:corollaryboundaryextra}
\hat u_L = f(x_1) g(y) h(z) \geq A\left( \widehat\Phi(x_1) -\alpha_L\left(\frac{x_1}{L}\right)^2 + \frac{x_1}{L} \right) h(z) \geq u \quad \text{on} \quad \Sigma_1 \cup \Sigma_L.
\end{align}
 
Indeed, to prove \eqref{eq:corollaryboundaryextra} on $\Sigma_1$ we bound $\widehat\Phi(1)$ from below, depending only on $m,M$: 
Since $\widehat\Phi$ is positive and convex, 
we have $\widehat\Phi'\le0$ and
$\widehat\Phi'\to0$. Multiplying
$\widehat\Phi''\le\kappa\widehat\Phi$ by $\widehat\Phi'\le0$
and integrating from $x_1$ to infinity gives
 
\[
    (\widehat\Phi'(x_1))^2\le\kappa\,\widehat\Phi(x_1)^2.
\]
 
Thus $\widehat\Phi'/\widehat\Phi\ge-\sqrt{\kappa}$,
and integration over $[0,1]$, using $\widehat\Phi(0)=1$, yields
 
\[
    \widehat\Phi(1)\ge e^{-\sqrt{\kappa}}.
\]
 
To ensure $\hat u_L \geq u$ on the part of the boundary of $\Omega_L$ where $|y|^2  = \omega_L(x_1)$ we note that on the set
 
\begin{align*}
 |y|^2  = \omega_L(x_1) =  A f(x_1)^{-1} \epsilon_L^{-1} \quad \text{for}\quad x_1 \in  (0,L), 
\end{align*}
 
it holds that
 
\begin{align*}
\hat u_L = f(x_1) \left(1 + \epsilon_L |y|^2\right)  h(z) = f(x_1)  \left(1 + \frac{A}{f(x_1)}\right)h(z) \geq A h(z) \geq u.                                      
\end{align*}
 

To finish the proof it only remains to clarify that the replacement of $\Sigma_0$ with $\Sigma_1$ does not affect the result.
Indeed, now we only obtain
 
$
u \leq A\widehat\Phi(x_1)
$
for $x_1 \geq 1$.
                                                                                                                                                                                                                                                    
However, by increasing $A$ if necessary so that
$A\geq C_1e^{\sqrt{\kappa}}$, we obtain $u(x)\leq C_1\leq A\widehat\Phi(1)
\leq A\widehat\Phi(x_1)$ for $0<x_1<1$. 
This completes the proof.

Let $\Omega$ and $\kappa$ be as in Theorem \ref{th:exp-decay-pipe}. 
Assume $p\in(1,2]$ and that $u$ is a $p$-subharmonic function in $\Omega$, 
satisfying $u \leq C_1$ on $\overline \Omega$ and $u \leq 0$ on $\Gamma^+$. 
Then there exists $A = A(p, n, C_1, m, M) > 0$ such that 
 
\begin{align*}
 u(x) \leq A\,  e^{-x_1 \sqrt{\kappa\, (p-1)}}
\end{align*}
  
whenever $x \in  \Omega$.

<!-- proof-5: body_md -->
Set \(\lambda=p-1\) and \(\Lambda=1\). For \(q\neq0\),
the second term in the minimum defining
\(\mathcal P_{p-1,1}(q,X)\) is
 
\begin{align*}
-\operatorname{Tr}X + (2-p)\frac{\langle Xq,q\rangle}{|q|^2} = -\widetilde\Delta_p(q,X),
\end{align*}
 
in which the normalized $p$-Laplace operator $\widetilde\Delta_p u$ is defined via
 
$$
\Delta_pu
=\operatorname{div}\!\left(|Du|^{p-2}Du\right)
=|Du|^{p-2}\left(
\Delta u+(p-2)\frac{\langle D^2u\,Du,Du\rangle}{|Du|^2}
\right)
=|Du|^{p-2}\widetilde\Delta_pu.
$$
 
Thus, our class of PDEs covers the normalized $p$-Laplace equation. 
Moreover, from \cite[Theorem 4.1]{JJ12} it follows that any bounded viscosity supersolution of the $p$-Laplace equation, $1 < p \leq 2$, is also a weak supersolution to the same equation, and in the definition of viscosity solutions, one can omit testing at points where the gradient vanishes.
This, together with a well known comparison principle for weak sub- and supersolutions of the $p$-Laplace equation, ensures that Corollary \ref{cor:exp-decay-pipe} applies once we ensure the boundary estimate in \eqref{eq:for_corr}.
Indeed, such boundary growth rate is well known for $p$-subharmonic functions with $\bar c$ depending only on $p,n$, when the domain is convex (such as a  maximal $(m,M)$-thin domain) see \cite{AKSZ07, J15}. 
An application of Corollary \ref{cor:exp-decay-pipe} with $\Lambda/\lambda$ constant now gives
 
\begin{align*}
 u(x) \leq A\,  e^{-\sqrt{\frac{\lambda}{\Lambda}\kappa}\,x_1} 
 =  A e^{-x_1 \sqrt{(p-1) \kappa}}
\end{align*}
  
whenever $x \in  \Omega$, which is the desired estimate.
