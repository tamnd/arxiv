<!-- theorem-1: body_md -->
\label{thm:main}
For every $d\in\{5,6,7,8\}$,
\begin{equation}\label{eq:sharp-constants}
 \lambda_d=\frac{(d-2)^2}{4}.
\end{equation}
Equivalently, for every $u\in\mathcal D_0(\mathbb Z^d)$,
\begin{equation}\label{eq:main-hardy}
 \mathcal E_d(u)\ge\frac{(d-2)^2}{4}\mathcal H_d(u).
\end{equation}
For $d=5,6$, \eqref{eq:main-hardy} holds for every finitely supported
$u:\mathbb Z^d\to\mathbb C$, without imposing $u(0)=0$.  In every stated admissible class
the coefficient is sharp, and equality is not attained by a nonzero finitely
supported function.
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{cor:transition}
For every integer $d\ge3$,
\[
 \lambda_d=\frac{(d-2)^2}{4}=C_d
 \quad\Longleftrightarrow\quad
 3\le d\le8.
\]
In particular, dimension nine is the first dimension in which the continuous
Hardy coefficient fails to be sharp on $\mathbb Z^d$.
<!-- end corollary-1 -->
<!-- proof-1: body_md -->
The equality in dimensions $5\le d\le8$ is Theorem~\ref{thm:main}.
The values $\lambda_3=1/4$ and $\lambda_4=1$ are known from
\cite{Alpay2026,Lizama2026}, whereas \cite{Lizama2026} proves
$\lambda_d<C_d$ for every $d\ge9$.
<!-- end proof-1 -->

<!-- proposition-2: body_md -->
\label{prop:upper}
For every integer $d\ge3$, there is a sequence of nonzero
$u_k\in\mathcal D_0(\mathbb Z^d)$ such that
\[
 \frac{\mathcal E_d(u_k)}{\mathcal H_d(u_k)}\longrightarrow C_d.
\]
In particular, $\lambda_d\le C_d$.
<!-- end proposition-2 -->
<!-- remark-2: body_md -->
Our result concerns the best coefficient multiplying the fixed weight
$|x|^{-2}$.  We do not claim that the weights constructed below are critical
or pointwise optimal Hardy weights.
<!-- end remark-2 -->
<!-- lemma-1: body_md -->
\label{lem:edge-square}
For $a>0$ and $z,w\in\mathbb C$,
\begin{equation}\label{eq:two-point-square}
 |z-w|^2=(1-a)|z|^2+(1-a^{-1})|w|^2
       +\left|a^{1/2}z-a^{-1/2}w\right|^2.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
The last square is
$a|z|^2+a^{-1}|w|^2-2\operatorname{Re}(z\overline w)$.
Adding the other two terms leaves
$|z|^2+|w|^2-2\operatorname{Re}(z\overline w)=|z-w|^2$.
<!-- end proof-2 -->
<!-- proposition-3: body_md -->
\label{prop:edge}
Let $a_{xy}>0$ be reciprocal on every nearest-neighbor edge of $\mathbb Z^d$,
and define
\[
 W(x)=2d-\sum_{y\sim x}a_{xy}.
\]
For every finitely supported $u:\mathbb Z^d\to\mathbb C$,
\begin{equation}\label{eq:edge}
 \mathcal E_d(u)-\sum_xW(x)|u(x)|^2
 =\sum_{(x,y)\in\mathcal E_+^{(d)}}
   \left|\sqrt{a_{xy}}u(x)-a_{xy}^{-1/2}u(y)\right|^2.
\end{equation}
<!-- end proposition-3 -->
<!-- proof-3: body_md -->
Apply \eqref{eq:two-point-square} to each forward edge.
At a fixed vertex $v$, the edges $(v,v+e_j)$ contribute
$\sum_j(1-a_{v,v+e_j})|u(v)|^2$. The edges $(v-e_j,v)$ contribute
$\sum_j(1-a_{v,v-e_j})|u(v)|^2$, where reciprocity has been used
for the second endpoint. The total coefficient is therefore
\[
 \sum_{j=1}^d\bigl[(1-a_{v,v+e_j})+(1-a_{v,v-e_j})\bigr]=W(v).
\]
Only edges meeting the finite support of $u$ contribute, so all sums
are finite and the regrouping is legitimate. The remaining terms are
the edge squares in \eqref{eq:edge}.
<!-- end proof-3 -->
<!-- proposition-4: body_md -->
\label{prop:punctured}
Assign positive reciprocal factors to edges with both endpoints in $X_d$.
Set
\begin{equation}\label{eq:punctured-weight}
 V(x)=2d-\sum_{\substack{y\sim x\\y\ne0}}a_{xy},\qquad x\in X_d.
\end{equation}
For every $u\in\mathcal D_0(\mathbb Z^d)$,
\begin{equation}\label{eq:punctured}
 \mathcal E_d(u)-\sum_{x\ne0}V(x)|u(x)|^2
 =\sum_{\substack{(x,y)\in\mathcal E_+^{(d)}\\x,y\ne0}}
 \left|\sqrt{a_{xy}}u(x)-a_{xy}^{-1/2}u(y)\right|^2.
\end{equation}
<!-- end proposition-4 -->
<!-- proof-4: body_md -->
Apply the two-point identity only to edges with nonzero endpoints.
If $x\sim0$, the remaining origin edge contributes $|u(x)|^2$,
since $u(0)=0$. Thus its diagonal contribution is retained in $2d$
but no factor for that edge is subtracted in \eqref{eq:punctured-weight}.
Regrouping the other terms as in Proposition~\ref{prop:edge} proves
\eqref{eq:punctured}.
<!-- end proof-4 -->
<!-- remark-3: body_md -->
If $a_{xy}=H(y)/H(x)$ for a positive vertex function $H$, the factors
multiply to one around every closed path. The two propositions require positivity and reciprocity, but not this
closed-path condition. We use this fact when modifying the field in
Section~\ref{sec:local}.

<!-- lemma-2: body_md -->
\label{lem:gluing}
Let $S\subset X_d$ be finite, and let $a^0$ be a positive reciprocal
field on edges with nonzero endpoints. Suppose its punctured weight
satisfies $V^0(x)\ge\kappa/|x|^2$ for $x\in X_d\setminus S$.
For $x\in S$, define
\begin{equation}\label{eq:core-diagonal}
 D_x=2d-\frac{\kappa}{|x|^2}
     -\sum_{\substack{y\sim x\\y\in X_d\setminus S}}a^0_{xy},
\end{equation}
and
\begin{equation}\label{eq:core-operator}
 (Mh)(x)=D_xh(x)-\sum_{\substack{y\sim x\\y\in S}}h(y).
\end{equation}
If $h:S\to(0,\infty)$ satisfies $Mh\ge0$, there is a positive
reciprocal field with punctured weight at least $\kappa/|x|^2$
at every $x\ne0$.
<!-- end lemma-2 -->
<!-- proof-5: body_md -->
Set $a_{xy}=h(y)/h(x)$ when both endpoints lie in $S$, and retain
$a^0_{xy}$ on every other edge. In particular every edge crossing the
boundary of $S$ remains unchanged. Each exterior vertex therefore
retains all its incident factors. For $x\in S$, its new weight satisfies
\[
 V(x)-\frac{\kappa}{|x|^2}
 =D_x-\sum_{\substack{y\sim x\\y\in S}}\frac{h(y)}{h(x)}
 =\frac{(Mh)(x)}{h(x)}\ge0.
\]
The field is positive and reciprocal by construction.
<!-- end proof-5 -->
<!-- lemma-3: body_md -->
\label{lem:minorant}
Let $n>0$, let $d\ge1$ be an integer, and let
$\alpha,\beta,\kappa\in\mathbb R$.
Suppose $f:[0,n]\to\mathbb R$ satisfies
\[
 f(s)\ge\alpha+\beta s
 \quad\text{for every }s\in[0,n],
 \qquad
 d\alpha+n\beta\ge\frac{\kappa}{n}.
\]
Then
\[
 \sum_{j=1}^d f(s_j)\ge\frac{\kappa}{n}
 \quad\text{whenever }s_j\ge0,\quad
 \sum_{j=1}^d s_j=n.
\]
In particular, it is sufficient that
\begin{equation}\label{eq:minorant}
 f(s)\ge f(0)+\frac{s}{n}
 \left(\frac{\kappa}{n}-df(0)\right),
 \qquad 0\le s\le n.
\end{equation}
If \eqref{eq:minorant} is strict for $s>0$, the resulting
summed inequality is strict.
<!-- end lemma-3 -->
<!-- proof-6: body_md -->
Since the $s_j$ are nonnegative and sum to $n$, each belongs
to $[0,n]$. Applying the assumed bound
$f(s_j)\ge\alpha+\beta s_j$ for each $j$ and summing gives
\[
 \sum_{j=1}^d f(s_j)
 \ge d\alpha+\beta\sum_{j=1}^d s_j
 =d\alpha+n\beta
 \ge\frac{\kappa}{n}.
\]
For \eqref{eq:minorant}, take
\[
 \alpha=f(0),
 \qquad
 \beta=\frac1n\left(\frac{\kappa}{n}-df(0)\right),
\]
so that $d\alpha+n\beta=\kappa/n$.
Finally, at least one $s_j$ is positive because their sum
is $n>0$. If \eqref{eq:minorant} is strict for positive
arguments, at least one summand inequality is strict,
and hence so is their sum.
<!-- end proof-6 -->
<!-- lemma-4: body_md -->
\label{lem:quadratic-positivity}
For $d\in\{5,6,7\}$, take $r=(d-2)/4$ in
\eqref{eq:general-coefficients}. Then
\[
 \mathsf A(n)s^2+\mathsf B(n)s+\mathsf G(n)>0
 \qquad\text{for all real }n\ge L_d,\quad s\ge0,
\]
with the thresholds in \eqref{eq:thresholds}.
<!-- end lemma-4 -->
<!-- proof-7: body_md -->
Appendix~\ref{app:polynomials} gives positive constants

 \frac{\mathsf A s^2+\mathsf B s+\mathsf G}{\theta_d}
 =\frac{\mathsf A}{\theta_d}
       \left(s-\frac n4-c_d\right)^2
   +\eta_d Q_d(t)s+T_d(t).
\end{equation}
Since $3/4\le r\le3/2$ and $n\ge L_d\ge2$,
\[

 \ge(r+1)\left(4n^2-n-\frac{13}{4}\right)>0.

Here
$4n^2-n-13/4=4(n-2)^2+15(n-2)+43/4$.
Since $q>0$, we obtain $\mathsf A=16q^2H>0$.
For $t,s\ge0$, the first two terms on the right of

\label{lem:d8-factor-positive}

 &\ge S^4-\frac{1327}{256}S^3+\frac{121}{16}S^2
       -\frac{61}{8}S-\frac58
       +\left(\frac{125}{48}S^2-\frac{13}{4}S+\frac14\right)\delta^2\\
 &\ge S^3(S-6).
\end{align*}
For the second inequality, the coefficient of $\delta^2$ is positive

\[

\[
 125(S-2)^2+344(S-2)+200,
 \qquad

Also
\[

 =\delta\left(\frac32S^3-\frac{817}{128}S^2
                 +\frac{31}{8}S+\frac{23}{4}+15\delta^2\right).

 \frac{27}{n}+\frac{82}{n^3}+\frac{65}{n^5}+\frac{21}{n^7}<5,

 \frac{\ell(n)-n^8(n-14)}{n^7}
 \le-\frac92n+73+\frac2{n^2}+\frac{158}{n^4}+\frac{90}{n^6}
 <-\frac92\cdot17+73+3<0.

 \frac{U_0(n)}{26837967175680}

   =n^2(n^2-2n-1)-1>0.
\end{align*}
Thus $a_3<0$, $a_4>0$, and $K>0$; for the last assertion use
$n^2-17n+12=t^2+17t+12>0$.

Put

 \rho=\frac{\ell_0^2}{4k_0b_0}
      =\frac{47628756003}{40674421760}<\frac65.
\]
By \eqref{eq:d8-quartic-identities},

 4KB-L^2=\frac{64k_0b_0n^2}{117649}
             \bigl(k(n)b(n)-\rho\ell(n)^2\bigr).
\]
All lower bounds used in the product are positive. Hence
\eqref{eq:d8-three-bounds} implies
\begin{align*}
 k(n)b(n)-\rho\ell(n)^2
 &\ge\frac12n^{17}(n^2-17n+12)-\frac65n^{16}(n-14)^2\\
 &=\frac{n^{16}}{10}(5n^3-97n^2+396n-2352)\\
 &=\frac{n^{16}}{10}(5t^3+158t^2+1433t+912)>0.
\end{align*}
In particular, $4KB-L^2>0$.

The exact Taylor identity for the quartic is
\[
 p_n(v)=B+L(v-\theta)
 +\bigl[a_2+a_3(v+2\theta)
         +a_4(v^2+2\theta v+3\theta^2)\bigr](v-\theta)^2.

For $0\le v\le1$, the expression in brackets minus $K$ is

\]
because $a_3<0$ and $a_4>0$. Therefore

 p_n(v)\ge

 +\frac{4KB-L^2}{4K}>0.
\]
Taking $v=s/n$ proves the lemma.

<!-- proof-13: body_md -->
Fix $d\ge3$, put $\alpha=(d-2)/2$, and choose a nonzero real-valued
$\phi\in C_c^\infty((0,1))$. For $L>1$, define
\begin{equation}\label{eq:phi-L}
 \varphi_L(x)=|x|^{-\alpha}
              \phi\left(\frac{\log|x|}{L}\right),\qquad x\ne0,
\end{equation}
and set $\varphi_L(0)=0$. The function is supported in an annulus
separated from the origin and vanishes near both boundaries of that
annulus. Its extension is therefore in $C_c^\infty(\mathbb R^d)$.

For a radial function $\varphi(r)=r^{-\alpha}g(\log r)$, polar
coordinates and $2\alpha=d-2$ give
\begin{align}
 \int_{\mathbb R^d}\frac{|\varphi(x)|^2}{|x|^2}\,\mathrm d x
 &=\omega_{d-1}\int_\mathbb R|g(t)|^2\,\mathrm d t,\label{eq:radial-mass}\\
 \int_{\mathbb R^d}|\nabla\varphi(x)|^2\,\mathrm d x
 &=\omega_{d-1}\int_\mathbb R|g'(t)-\alpha g(t)|^2\,\mathrm d t,
 \label{eq:radial-energy}
\end{align}
where $t=\log r$ and $\omega_{d-1}=|\mathbb S^{d-1}|$.
For \eqref{eq:phi-L}, $g(t)=\phi(t/L)$. The mixed term vanishes because
\[
 \operatorname{Re}\int_\mathbb R g'(t)\overline{g(t)}\,\mathrm d t
 =\frac12\int_\mathbb R (|g(t)|^2)'\,\mathrm d t=0.
\]
Changing variables from $t$ to $t/L$ yields
\begin{align}
 \int_{\mathbb R^d}\frac{|\varphi_L|^2}{|x|^2}\,\mathrm d x
 &=\omega_{d-1}L\int_0^1|\phi|^2,\label{eq:continuous-mass}\\
 \int_{\mathbb R^d}|\nabla\varphi_L|^2\,\mathrm d x
 &=\omega_{d-1}L\left(C_d\int_0^1|\phi|^2
               +\frac1{L^2}\int_0^1|\phi'|^2\right).
 \label{eq:continuous-energy}
\end{align}
Consequently,
\begin{equation}\label{eq:continuous-quotient}
 \frac{\int_{\mathbb R^d}|\nabla\varphi_L|^2\,\mathrm d x}
      {\int_{\mathbb R^d}|\varphi_L|^2/|x|^2\,\mathrm d x}
 =C_d+\frac1{L^2}
       \frac{\int_0^1|\phi'|^2}{\int_0^1|\phi|^2}.
\end{equation}

We now pass to the lattice. For fixed $L$ and $N\in\mathbb N$, set
\begin{equation}\label{eq:u-LN}
 u_{L,N}(x)=\varphi_L(x/N),\qquad x\in\mathbb Z^d.
\end{equation}
This is finitely supported and vanishes at the origin.
The difference quotients
\[
 N\bigl(\varphi_L(y+e_j/N)-\varphi_L(y)\bigr)
 =\int_0^1\partial_j\varphi_L(y+t e_j/N)\,\mathrm d t
\]
converge uniformly to $\partial_j\varphi_L(y)$.
Their supports lie in one compact set depending on $L$ but not on $N$.
Writing the scaled energy as a Riemann sum therefore gives
\begin{align}
 N^{2-d}\mathcal E_d(u_{L,N})
 &=\frac1{N^d}\sum_{x\in\mathbb Z^d}\sum_{j=1}^d
 \left|N\bigl(\varphi_L((x+e_j)/N)-\varphi_L(x/N)\bigr)\right|^2
 \notag\\
 &\longrightarrow \int_{\mathbb R^d}|\nabla\varphi_L(y)|^2\,\mathrm d y.
 \label{eq:energy-riemann}
\end{align}
Indeed, the difference between this sum and the Riemann sum of
$\sum_j|\partial_j\varphi_L|^2$ is bounded by the uniform error
times $N^{-d}$ times the number of grid points in a fixed compact set;
the last factor is bounded uniformly in $N$.
Likewise,
\begin{align}
 N^{2-d}\mathcal H_d(u_{L,N})
 &=\frac1{N^d}\sum_{x\ne0}
       \frac{|\varphi_L(x/N)|^2}{|x/N|^2}\longrightarrow\int_{\mathbb R^d}\frac{|\varphi_L(y)|^2}{|y|^2}\,\mathrm d y.
 \label{eq:mass-riemann}
\end{align}
The mass integrand extends continuously by zero near the origin and is
compactly supported. Its integral is strictly positive, so the sampled
function is nonzero for sufficiently large $N$, and its quotient
converges to \eqref{eq:continuous-quotient}.

For every integer $L\ge2$, choose $N(L)$ large enough that $u_{L,N(L)}$
is nonzero and its quotient differs from \eqref{eq:continuous-quotient}
by at most $L^{-2}$. With $u_L=u_{L,N(L)}$, the quotient tends to $C_d$.
This proves Proposition~\ref{prop:upper}.

\label{rem:d4-check}
Set $d=4$ in the quadratic field \eqref{eq:base-field}. 
For $r=1/2$, the coefficients $\mathsf A,\mathsf B,\mathsf G$
in \eqref{eq:general-coefficients} satisfy
$\mathsf A(n),\mathsf G(n)>0$ for $n\ge2$ and
$\mathsf B(n)>0$ for $n\ge8$.
Indeed, the expansions of $\mathsf A$ and $\mathsf G$
in powers of $n-2$, and of $\mathsf B$ in powers of $n-8$,
have strictly positive coefficients.
For the remaining integer values, direct substitution gives
\[
 4\mathsf A(n)\mathsf G(n)-\mathsf B(n)^2
 \ge \frac{90737675}{4}>0,
 \qquad n=2,\ldots,7.
\]
Completing the square for these six values, and using
coefficient positivity for $n\ge8$, therefore gives
\[
 \mathsf A(n)s^2+\mathsf B(n)s+\mathsf G(n)>0
 \qquad\text{for integers }n\ge2,\quad s\ge0.
\]
The same
affine-minorant computation gives
\[
 W_4^0(x)>\frac1{|x|^2},
 \qquad |x|^2\ge2,
\]
while the first-shell calculation in Section~\ref{sec:local} gives
\[
 W_4^0(0)=\frac{32}{11},
 \qquad
 W_4^0(e_1)-1=\frac{499}{1645}>0.
\]
The full-lattice decomposition therefore yields $\mathcal E_4(u)\ge\mathcal H_4(u)$, with
strict inequality for nonzero finitely supported $u$.  Proposition~\ref{prop:upper}
gives the reverse bound for the sharp coefficient, so $\lambda_4=1$.  This
recovers the result of \cite{Lizama2026} by a different method.
