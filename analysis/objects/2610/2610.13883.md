<!-- thm-2: body_md -->
\label{thm:profile}
For $H\ge H_0$ let $\boldsymbol{m}_H\in H^2(\mathbb{R}^3;\mathbb{S}^2)$ be a minimizer of $I_1(H)$. Then the following holds.
\begin{enumerate}[label=(\roman*), ref=(\roman*)]
\item\label{item:profileTrivial} As $H\to\infty$,
\begin{align*}
   \| \boldsymbol{m}_H- \boldsymbol{\hat{e}}_{3} \|_{H^1(\mathbb{R}^3)} \to 0
    \qquad\text{while}\qquad
    \|\Delta\boldsymbol{m}_H\|_{L^2(\mathbb{R}^3)}\to\infty .
\end{align*}
In particular $(\boldsymbol{m}_H-\boldsymbol{\hat{e}}_{3})_{H\ge H_0}$ is unbounded in $H^2(\mathbb{R}^3;\mathbb{R}^3)$.
\item\label{item:profileLimit} For every sequence $H_j\to\infty$ there exist translations $y_j\in\mathbb{R}^3$, a subsequence (not relabelled) and a map $\boldsymbol{u}\in H^2(\mathbb{R}^3;\mathbb{S}^2)$ such that
\begin{align*}
    \boldsymbol{u}_j
    :=
    \boldsymbol{m}_{H_j}\bigl(H_j^{-\frac14}(\,\cdot\,-y_j)\bigr)
    \longrightarrow
    \boldsymbol{u}
    \quad\text{strongly in } H^2(\mathbb{R}^3;\mathbb{R}^3)
\end{align*}
(and thus uniformly) with $Q[\boldsymbol{u}]=1$ and
\[
 \mathcal{E}^\infty(\boldsymbol{u}) \;=\; \inf\big\{\mathcal{E}^\infty[\boldsymbol{m}] : \boldsymbol{m}\in H^2(\mathbb{R}^3;\mathbb{S}^2),\; Q[\boldsymbol{m}]=1\big\}.
\]
\end{enumerate}

<!-- lemma-1: body_md -->
\label{lemma:scaling}
Let $\boldsymbol{m}\in H^2(\mathbb{R}^3;\mathbb{S}^2)$. Then there exists $\theta\in [0,1]$ (depending on $\boldsymbol{m}$) such that
\begin{align}
    \label{eq:scalingFormula}
  \inf_{\lambda>0}\mathcal{E}_H[\boldsymbol{m}^{\lambda}]
=
\frac{\sqrt{2}}{3}
\frac{\sqrt{\theta^2+12H}-2\theta}
{\sqrt{\sqrt{\theta^2+12H}-\theta}}
\,
\|\boldsymbol{m}-\boldsymbol{\hat{e}}_{3}\|_{L^2}^{\frac12}
\|\Delta\boldsymbol{m}\|_{L^2}^{\frac32}.
\end{align}
In particular
\begin{align*}
      \frac{\sqrt{2}}{3}\frac{\sqrt{1+12H}-2}{\sqrt{\sqrt{1+12H}-1}}\,\|\boldsymbol{m}-\boldsymbol{\hat{e}}_{3}\|_{L^2}^{\frac12}\|\Delta\boldsymbol{m}\|_{L^2}^{\frac32}\le \inf_{\lambda>0}\mathcal{E}_H[\boldsymbol{m}^{\lambda}]\le \frac{\sqrt{2}}{3}(12H)^{\frac14}\,\|\boldsymbol{m}-\boldsymbol{\hat{e}}_{3}\|_{L^2}^{\frac12}\|\Delta\boldsymbol{m}\|_{L^2}^{\frac32}.
\end{align*}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Define for $\lambda>0$, $\boldsymbol{m}^{\lambda}(x)=\boldsymbol{m}(\lambda x)$ and as in \eqref{eq:scaling}
\begin{align*}
\mathcal{E}_H[\boldsymbol{m}^{\lambda}] &=  \frac{\lambda \| \Delta \boldsymbol{m} \|_{L^2}^2}{2} - \frac{ \|\nabla \boldsymbol{m}\|_{L^2}^2}{2\lambda} + \frac{H\|\boldsymbol{m}-\boldsymbol{\hat{e}}_{3}\|_{L^2}^2}{2\lambda^3}\\
&=\frac12
\|\boldsymbol{m}-\boldsymbol{\hat{e}}_{3}\|_{L^2}^{\frac12}
\|\Delta\boldsymbol{m}\|_{L^2}^{\frac32}
\left(
y-\frac{\theta}{y}+\frac{H}{y^3}
\right),
\end{align*}
where (assuming $\boldsymbol{m} \neq \boldsymbol{\hat{e}}_{3}$)
\begin{align*}
y
=
\lambda
\frac{\|\Delta\boldsymbol{m}\|_{L^2}^{\frac12}}
{\|\boldsymbol{m}-\boldsymbol{\hat{e}}_{3}\|_{L^2}^{\frac12}},
\qquad
\theta
=
\frac{\|\nabla\boldsymbol{m}\|_{L^2}^{2}}
{\|\boldsymbol{m}-\boldsymbol{\hat{e}}_{3}\|_{L^2}\|\Delta\boldsymbol{m}\|_{L^2}} \in [0,1].
\end{align*}
The minimum in $y>0$ is attained at
\begin{align*}
y =\sqrt \frac{\sqrt{\theta^2+12H}-\theta}{2}.
\end{align*}
Since $\inf_{\lambda>0} \mathcal{E}_H[\boldsymbol{m}^\lambda]$ is decreasing in $\theta$, the desired estimates follow.

<!-- lemma-2: body_md -->
\label{lemma:scaling_invariant_topological_bound}
Let $\boldsymbol{m}\in H^2(\mathbb{R}^3;\mathbb{S}^2)$. Then
\begin{align*}
    \|\boldsymbol{m}-\boldsymbol{\hat{e}}_{3}\|_{L^2}^{\frac12}
    \|\Delta\boldsymbol{m}\|_{L^2}^{\frac32}
    \ge
    2^4 3^{\frac38}\pi^2 |Q[\boldsymbol{m}]|^{\frac34}.
\end{align*}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
We use the Whitehead representation
\begin{align*}
    Q[\boldsymbol{m}]
    =
    \frac{1}{16\pi^2}
    \int_{\mathbb{R}^3}
    \boldsymbol{A}\cdot\boldsymbol{F}
    \mathrm{d}x,
    \qquad
    \nabla \times \boldsymbol{A}=\boldsymbol{F},
    \qquad
    \nabla \cdot \boldsymbol{A}=0.
\end{align*}
By Hölder's inequality,
\begin{align*}
    |Q[\boldsymbol{m}]|
    \le
    \frac{1}{16\pi^2}
    \|\boldsymbol{A}\|_{L^6}
    \|\boldsymbol{F}\|_{L^{\frac65}}.
\end{align*}
In Coulomb gauge and by the sharp Sobolev inequality on $\mathbb{R}^3$ (cf. \cite{Talenti1976BestSobolevConstant}),
\begin{align*}
    \|\boldsymbol{A}\|_{L^6}
    \le
    \frac{2^{\frac23}}{\sqrt{3}\pi^{\frac23}}
    \|\nabla\boldsymbol{A}\|_{L^2}
    =
    \frac{2^{\frac23}}{\sqrt{3}\pi^{\frac23}}
    \|\boldsymbol{F}\|_{L^2}.
\end{align*}
and Hölder interpolation $L^1\cap L^2\subset L^{\frac65}$ yield
\begin{align*}
    |Q[\boldsymbol{m}]| \le \frac{2^{\frac23}}{16\sqrt{3}\pi^{\frac83}} \|\boldsymbol{F}\|_{L^1}^{\frac23} \|\boldsymbol{F}\|_{L^2}^{\frac43}.
\end{align*}
As in the Vakulenko-Kapitanskii-type estimate used by Ward \cite{Ward1999Hopfions}, and with $ \boldsymbol{m}\cdot\Delta\boldsymbol{m}=-|\nabla\boldsymbol{m}|^2$, one has
\begin{align*}
    |\boldsymbol{F}| \le \frac12|\nabla\boldsymbol{m}|^2 \quad \text{and} \quad |\boldsymbol{F}| \le \frac12|\Delta\boldsymbol{m}|.
\end{align*}
Hence, we obtain the intermediate estimate
\begin{align}
    \label{eq:intermediateVK}
    |Q[\boldsymbol{m}]|^{\frac34}
    \le
    \frac{1}{2^4 3^{\frac38}\pi^2}
    \|\nabla\boldsymbol{m}\|_{L^2}
    \|\Delta\boldsymbol{m}\|_{L^2}.
\end{align}
Finally, using integration by parts yields
\begin{align*}
|Q[\boldsymbol{m}]|^{\frac34} \le \frac{1}{2^4 3^{\frac38}\pi^2} \|\boldsymbol{m}-\boldsymbol{\hat{e}}_{3}\|_{L^2}^{\frac12} \|\Delta\boldsymbol{m}\|_{L^2}^{\frac32}.
\end{align*}

<!-- lemma-3: body_md -->
\label{lemma:loc}
Let $B \subset \mathbb{R}^3$ be measurable and let $\boldsymbol{m} \in H^2(\mathbb{R}^3;\mathbb{S}^2)$. There exist constants $C_0, C_1 > 0$, independent of $\boldsymbol{m}$ and $B$, such that
\begin{align*}
\int_B |\boldsymbol{F} \cdot \boldsymbol{A}|  \mathrm{d}x  &\le C_0 \mathcal{E}_H[\boldsymbol{m}] |B|^{\frac13},\\[0.5em]
\int_B |\boldsymbol{F} \cdot \boldsymbol{A}|  \mathrm{d}x  &\le C_1 \mathcal{E}_H[\boldsymbol{m}]^{\frac76} \mathcal{E}_H[\boldsymbol{m};B]^{\frac16},
\end{align*}
where $\boldsymbol{A}\in L^6(\mathbb{R}^3;\mathbb{R}^3)$ is chosen in Coulomb gauge, i.e. $\nabla \cdot \boldsymbol{A} = 0$.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Using the same estimates as in the proof of Lemma~\ref{lemma:scaling_invariant_topological_bound}, applied locally on $B$ where appropriate, we obtain
\begin{align*}
\int_B |\boldsymbol{F}\cdot\boldsymbol{A}|\mathrm{d}x
\lesssim \|\nabla \boldsymbol{m}\|_{L^2(B)}^{\frac43}\|\nabla \boldsymbol{m} \|_{L^4(B)}^{\frac23}\|\nabla \boldsymbol{m}\|_{L^4(\mathbb{R}^3)}^2.
\end{align*}
Thus 
\begin{align*}
    \|\nabla \boldsymbol{m}\|_{L^2(B)}^{\frac43} \le c |B|^{\frac13} \| \nabla \boldsymbol{m}\|_{L^4(B)}^{\frac43} \quad \text{ and } \quad  \|\nabla \boldsymbol{m}\|_{L^4(\mathbb{R}^3)}^4 \lesssim \mathcal{E}_H[\boldsymbol{m}]
\end{align*}
yield the first inequality. To obtain the second estimate, we use 
\begin{align*}
\| \nabla \boldsymbol{m}\|_{L^4(B)}^{\frac23}\lesssim \mathcal{E}_H[\boldsymbol{m};B]^{\frac16} \quad \text{ and } \quad \| \nabla \boldsymbol{m}\|_{L^2(\mathbb{R}^3)}^2 \lesssim \mathcal{E}_H[\boldsymbol{m}].
\end{align*}

<!-- lemma-4: body_md -->
\label{lemma:energygrowth}
There exists a constant $C>0$ such that for all $k\in\mathbb{Z}$ and $H>\tfrac14$,
\begin{align*}
I_k(H) \le CH^{\frac14}|k|^{\frac34}.
\end{align*}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
We may assume $k>0$. Choose $\ell\in\mathbb{N}$ such that
\begin{align*}
\ell^{2} \le k < (\ell+1)^{2}.
\end{align*}
By \cite{LinYang2004}, there exist maps $\overline{\boldsymbol{m}},\boldsymbol{m}_{0}\in H^{2}(\mathbb{R}^{3};\mathbb{S}^{2})$ satisfying
\begin{align*}
\overline{\boldsymbol{m}}=\boldsymbol{\hat{e}}_{3} \text{ on } \mathbb{R}^{3}\setminus B_{\sqrt{\ell}}, \quad Q[\overline{\boldsymbol{m}}]=\ell^{2}, \\
\boldsymbol{m}_{0}=\boldsymbol{\hat{e}}_{3} \text{ on } \mathbb{R}^{3}\setminus B_{1},\quad Q[\boldsymbol{m}_{0}]=1,
\end{align*}
and a constant $C>0$ such that
\begin{align*}
\|\nabla^{2}\overline{\boldsymbol{m}}\|_{L^{\infty}} \le C, \quad \|\nabla^{2}\boldsymbol{m}_{0}\|_{L^{\infty}} \le C.
\end{align*}
Define $\boldsymbol{m}\in H^{2}(\mathbb{R}^{3};\mathbb{S}^{2})$ by setting $\boldsymbol{m}=\overline{\boldsymbol{m}}$ on $B_{\sqrt{\ell}}$ and placing $(k-\ell^{2})$ translated copies of $\boldsymbol{m}_{0}$ at points $x_{1},\dots,x_{k-\ell^{2}}$ with
\begin{align*}
|x_{i}-x_{j}|\gg 3,\quad |x_{i}|\gg\sqrt{\ell}+1.
\end{align*}
Exploiting the quasi-locality of the Hopf index density (in Coulomb gauge), the supports may be chosen sufficiently far apart so that
\begin{align*}
Q[\boldsymbol{m}] = \ell^{2} + (k-\ell^{2}) = k.
\end{align*}
Lemma~\ref{lemma:scaling}, together with Young's inequality, then yields
\begin{align*}
   I_k(H) \le  \inf_{\lambda >0}\mathcal{E}_H[\boldsymbol{m}^{\lambda}] \lesssim H^{\frac14}\int_{\mathbb{R}^3} |\nabla^2 \boldsymbol{m}|^2 + |\boldsymbol{m} - \boldsymbol{\hat{e}}_{3}|^2 \mathrm{d}x .
\end{align*}
Using disjointness of supports and the uniform bounds on $\nabla^{2}\overline{\boldsymbol{m}}$ and $\nabla^{2}\boldsymbol{m}_{0}$, together with $|B_{\sqrt{\ell}}|=|B_1|  \ell^{\frac32}$ and $0 \le k-\ell^{2}\le 2\ell+1$, we obtain
\begin{align*}
I_k(H) \lesssim H^{\frac14}(\ell^{\frac32} + \ell)\lesssim H^{\frac14} k^{\frac34}.
\end{align*}

\label{lemma:vanishing}
Let $(\boldsymbol{m}_n)_{n \in \mathbb{N}}$ be a minimizing sequence for $I_k(H)$ with $k\in\mathbb Z\setminus\{0\}$.

<!-- lemma-6: body_md -->
\label{cut-offconstruction}
There exists a $\varepsilon_0 > 0$ such that if for $0<\varepsilon<\varepsilon_0$, $R \geq 1$ and $\boldsymbol{m} \in H^2(\mathbb{R}^3;\mathbb{S}^2)$ we have
\begin{align*}
\int_{B_{4R}\setminus B_R} |\boldsymbol{m}- \boldsymbol{\hat{e}}_{3}|^2 + |\Delta \boldsymbol{m}|^2 \mathrm{d}x \le \varepsilon,
\end{align*}
then we can find $\boldsymbol{m}^{(1)}$ and $\boldsymbol{m}^{(2)}$ with the following properties:
\begin{align*}
&\boldsymbol{m}^{(1)} = \boldsymbol{m} \text{ on } B_{2R}  &&\boldsymbol{m}^{(2)} = \boldsymbol{m} \text{ on } \mathbb{R}^3 \setminus B_{2R} \\
&\boldsymbol{m}^{(1)} = \boldsymbol{\hat{e}}_{3} \text{ on } \mathbb{R}^3 \setminus B_{4R} &&\boldsymbol{m}^{(2)} = \boldsymbol{\hat{e}}_{3} \text{ on }  B_{R} \\
\end{align*}	
and, furthermore, there is a constant $c>0$ with
\begin{align*}
\lVert \boldsymbol{m}^{(1)} - \boldsymbol{\hat{e}}_{3} \rVert_{H^2(\mathbb{R}^3 \setminus B_{2R})}^2 + \lVert \boldsymbol{m}^{(2)} - \boldsymbol{\hat{e}}_{3} \rVert_{H^2(B_{2R})}^2 \le c \varepsilon.
\end{align*}
<!-- end lemma-6 -->
<!-- proof-6: body_md -->
We construct only $\boldsymbol{m}^{(1)}$. The construction of $\boldsymbol{m}^{(2)}$ is analogous.
By the interior elliptic estimate on annuli, applied after scaling, we have
\begin{align*}
    \|\boldsymbol{m}-\boldsymbol{\hat{e}}_{3}\|_{H^2(B_{3R}\setminus B_{2R})}^2
    \le
    C
    \int_{B_{4R}\setminus B_R}
    |\boldsymbol{m}-\boldsymbol{\hat{e}}_{3}|^2+|\Delta\boldsymbol{m}|^2 \mathrm{d}x ,
\end{align*}
where $C$ is independent of $R\geq 1$. Here one covers $B_{3R}\setminus B_{2R}$ by balls adapted to the scale $R$ and applies the standard interior elliptic estimate on slightly larger balls. The overlap is uniformly bounded, so the constant is independent of $R\geq 1$. Hence, by the assumption,
\begin{align*}
    \|\boldsymbol{m}-\boldsymbol{\hat{e}}_{3}\|_{H^2(B_{3R}\setminus B_{2R})}^2
    \le C\varepsilon .
\end{align*}
By Sobolev embedding, after decreasing $\varepsilon_0$ if necessary, we may assume
\begin{align*}
    |\bar m|\le \frac{1}{\sqrt{2}},
    \qquad
    m_3\geq \frac{1}{\sqrt{2}}
    \qquad
    \text{on } B_{3R}\setminus B_{2R},
\end{align*}
where $\bar m=(m_1,m_2)$. Thus on this annulus
\begin{align*}
    \boldsymbol{m}
    =
    \bigl(\bar m,\sqrt{1-|\bar m|^2}\,\bigr).
\end{align*}

Let $\eta\in C^\infty([0,\infty))$ satisfy
\begin{align*}
    0\le \eta\le 1,
    \qquad
    \eta(t)=1 \text{ for } t\le 2
    \quad \text{ and } \quad
    \eta(t)=0 \text{ for } t\geq 3 .
\end{align*}
Set
\begin{align*}
    \eta_R(x)=\eta\left(\frac{|x|}{R}\right).
\end{align*}
Then
\begin{align*}
    |\nabla^j\eta_R|\le C R^{-j},
    \qquad
    j=1,2 .
\end{align*}
We define
\begin{align*}
    \boldsymbol{m}^{(1)}(x)
    =
    \begin{cases}
        \boldsymbol{m}(x),
        & x\in B_{2R},
        \\
        \left(
        \eta_R(x)\bar m(x),
        \sqrt{1-|\eta_R(x)\bar m(x)|^2}
        \right),
        & x\in B_{3R}\setminus B_{2R},
        \\
        \boldsymbol{\hat{e}}_{3},
        & x\in \mathbb{R}^3\setminus B_{3R}.
    \end{cases}
\end{align*}
This is well defined because $|\eta_R\bar m|\le 1/\sqrt{2}$. Moreover
\begin{align*}
    \boldsymbol{m}^{(1)}=\boldsymbol{m} \text{ on } B_{2R},
    \qquad
    \boldsymbol{m}^{(1)}=\boldsymbol{\hat{e}}_{3} \text{ on } \mathbb{R}^3\setminus B_{3R}.
\end{align*}
Let
\begin{align*}
    \Psi(u)=\bigl(u,\sqrt{1-|u|^2}\,\bigr).
\end{align*}
On $\{|u|\le 1/\sqrt{2}\}$ the map $\Psi$ has bounded $C^2$ norm. Since
$\boldsymbol{m}^{(1)}=\boldsymbol{\hat{e}}_{3}$ on $\mathbb{R}^3\setminus B_{3R}$, we have
\begin{align*}
    \|\boldsymbol{m}^{(1)}-\boldsymbol{\hat{e}}_{3}\|_{H^2(\mathbb{R}^3\setminus B_{2R})}^2
    =
    \|\boldsymbol{m}^{(1)}-\boldsymbol{\hat{e}}_{3}\|_{H^2(B_{3R}\setminus B_{2R})}^2 
\end{align*}
and $\boldsymbol{m}^{(1)} - \boldsymbol{\hat{e}}_{3} = \Psi(\eta_R \bar m) - \Psi(0)$ on $B_{3R}\setminus B_{2R}$. Therefore, using the bounds on $\eta_R$ and $R\geq 1$, we have
\begin{align*}
    \|\boldsymbol{m}^{(1)}-\boldsymbol{\hat{e}}_{3}\|_{H^2(\mathbb{R}^3\setminus B_{2R})}^2
    &\le
    C\|\eta_R\bar m\|_{H^2(B_{3R}\setminus B_{2R})}^2 \le
    C\|\bar m\|_{H^2(B_{3R}\setminus B_{2R})}^2
    \\
    &\le
    C\|\boldsymbol{m}-\boldsymbol{\hat{e}}_{3}\|_{H^2(B_{3R}\setminus B_{2R})}^2 \le C\varepsilon .
\end{align*}
This proves the desired estimate for $\boldsymbol{m}^{(1)}$.

<!-- lemma-7: body_md -->
\label{lemma:dichotomy}
Let $H>\frac14$ and $k \in \mathbb{Z} \setminus \{0\}$. Assume further there exists a minimizing sequence $(\boldsymbol{m}_n)_{n \in \mathbb{N}}$ for $I_k(H)$ such that condition~\ref{case:dichotomy} dichotomy is satisfied. Then there exists $\ell \in \mathbb{Z} \setminus\{0,k\}$ with
\begin{align*}
I_k(H) \geq I_\ell(H) + I_{k-\ell}(H).
\end{align*}
<!-- end lemma-7 -->
<!-- proof-7: body_md -->
Suppose that dichotomy~\ref{case:dichotomy} occurs for $(e(\boldsymbol{m}_n))_{n\in \mathbb{N}}$. Let $0 < \lambda < 1$ be the $\lambda$ described in dichotomy and fix $\varepsilon > 0$ to find $R, R_n, y_n$ as stated in~\ref{case:dichotomy}. Assume further $R>1$ and by translation invariance $y_n = 0$, then 
\begin{align*}
|\lambda\mathcal{E}_H[\boldsymbol{m}_n] - \mathcal{E}_H[\boldsymbol{m}_n;B_R]| < \varepsilon \text{ and } |(1-\lambda)\mathcal{E}_H[\boldsymbol{m}_n] - \mathcal{E}_H[\boldsymbol{m}_n;\mathbb{R}^3 \setminus B_{R_n}]| < \varepsilon.
\end{align*}
Without loss of generality $R_n > 4R$ since $R_n \to \infty$, then we conclude
\begin{align*}
&|(1-\lambda)\mathcal{E}_H[\boldsymbol{m}_n] - \mathcal{E}_H[\boldsymbol{m}_n;\mathbb{R}^3 \setminus B_{4R}] | < 3 \varepsilon \text{ and} \\
&\mathcal{E}_H[\boldsymbol{m}_n;B_{4R} \setminus B_R] < 2 \varepsilon.
\end{align*}
Both statements can be shown by showing that $\mathcal{E}_H[\boldsymbol{m}_n;B_{R_n} \setminus B_R] < 2\varepsilon$, a consequence of
\begin{align*}
\mathcal{E}_H[\boldsymbol{m}_n;B_{R_n} \setminus B_R] 
&\le |\mathcal{E}_H[\boldsymbol{m}_n;B_{R_n}] - \lambda \mathcal{E}_H[\boldsymbol{m}_n]| + |\lambda \mathcal{E}_H[\boldsymbol{m}_n] - \mathcal{E}_H[\boldsymbol{m}_n;B_R]|\\
&= |(1-\lambda)\mathcal{E}_H[\boldsymbol{m}_n] - \mathcal{E}_H[\boldsymbol{m}_n;\mathbb{R}^3 \setminus B_{R_n}] | \\
&+ |\lambda \mathcal{E}_H[\boldsymbol{m}_n] - \mathcal{E}_H[\boldsymbol{m}_n;B_R]|
\end{align*}
and~\ref{case:dichotomy} dichotomy. In particular, this shows that in the dichotomy case the radius $R$ can be replaced by a radius $R'\in(R,R_n)$, and the same conclusions remain valid. Since $R_n\to\infty$, we may therefore enlarge $R$ in the proof whenever needed. In particular, we assume $R\ge 1$. Then, by \eqref{eq:densityCoercivity}, we get that
\begin{align*}
\int_{B_{4R}\setminus B_R} |\boldsymbol{m} - \boldsymbol{\hat{e}}_{3}|^2 + |\Delta \boldsymbol{m}|^2 \mathrm{d}x< c_0 \varepsilon.
\end{align*}
Let $\boldsymbol{m}^{(1)}_n,\boldsymbol{m}^{(2)}_n\in H^2(\mathbb{R}^3;\mathbb{S}^2)$ be the maps provided by Lemma~\ref{cut-offconstruction}, such that
\begin{align*}
&\boldsymbol{m}^{(1)}_n = \boldsymbol{m}_n \text{ on } B_R ,&& \boldsymbol{m}^{(2)}_n = \boldsymbol{m}_n \text{ on } \mathbb{R}^3 \setminus B_{4R}, \\
&\boldsymbol{m}^{(1)}_n = \boldsymbol{\hat{e}}_{3} \text{ on } \mathbb{R}^3 \setminus B_{4R}  , &&\boldsymbol{m}^{(2)}_n = \boldsymbol{\hat{e}}_{3} \text{ on } B_R.
\end{align*}
From the same Lemma and $\mathcal{E}_H[\boldsymbol{m}_n;B_{4R} \setminus B_R] < c\varepsilon$ we get that 
\begin{align*}
&\lVert \boldsymbol{m}^{(1)}_n - \boldsymbol{\hat{e}}_{3} \rVert_{H^2(\mathbb{R}^3 \setminus B_R)}^2 + \lVert \boldsymbol{m}^{(2)}_n - \boldsymbol{\hat{e}}_{3} \rVert_{H^2(B_{4R})}^2\\ 
&=  \lVert \boldsymbol{m}^{(1)}_n - \boldsymbol{\hat{e}}_{3} \rVert_{H^2(B_{4R} \setminus B_R)}^2 + \lVert \boldsymbol{m}^{(2)}_n - \boldsymbol{\hat{e}}_{3} \rVert_{H^2(B_{4R} \setminus B_R)}^2 < c \varepsilon . 
\end{align*}
This implies 
\begin{align*}
\mathcal{E}_H[\boldsymbol{m}^{(1)}_n;B_{4R} \setminus B_R] < c \varepsilon \text{ and } \mathcal{E}_H[\boldsymbol{m}^{(2)}_n;B_{4R} \setminus B_R] < c \varepsilon.
\end{align*}
Let $\boldsymbol{F}_n$ and $\boldsymbol{F}_n^{(j)}$ denote the fields corresponding to $\boldsymbol{m}_n$ and $\boldsymbol{m}_n^{(j)}$, respectively. Choose potentials in Coulomb gauges
\begin{align*}
\nabla\times \boldsymbol{A}_n=\boldsymbol{F}_n,
\qquad
\nabla\times \boldsymbol{A}_n^{(j)}=\boldsymbol{F}_n^{(j)}
\end{align*}
for $j=1,2$. Then
\begin{align*}
\nabla\times\bigl(\boldsymbol{A}_n-\boldsymbol{A}_n^{(1)}-\boldsymbol{A}_n^{(2)}\bigr)
=
\boldsymbol{F}_n-\boldsymbol{F}_n^{(1)}-\boldsymbol{F}_n^{(2)},
\qquad
\nabla\cdot\bigl(\boldsymbol{A}_n-\boldsymbol{A}_n^{(1)}-\boldsymbol{A}_n^{(2)}\bigr)=0.
\end{align*}
We compute
\begin{align*}
&16\pi^2
\left|Q[\boldsymbol{m}_n]-Q[\boldsymbol{m}_n^{(1)}]-Q[\boldsymbol{m}_n^{(2)}]\right|
\\
&\le
\left|
\int_{\mathbb{R}^3}
\bigl(\boldsymbol{F}_n-\boldsymbol{F}_n^{(1)}-\boldsymbol{F}_n^{(2)}\bigr)\cdot\boldsymbol{A}_n
\mathrm{d}x
\right|
+
\left|
\int_{\mathbb{R}^3}
\boldsymbol{F}_n^{(1)}\cdot
\bigl(\boldsymbol{A}_n-\boldsymbol{A}_n^{(1)}-\boldsymbol{A}_n^{(2)}\bigr)
\mathrm{d}x
\right|
\\
&+
\left|
\int_{\mathbb{R}^3}
\boldsymbol{F}_n^{(2)}\cdot
\bigl(\boldsymbol{A}_n-\boldsymbol{A}_n^{(1)}-\boldsymbol{A}_n^{(2)}\bigr)
\mathrm{d}x
\right|
+
\left|
\int_{\mathbb{R}^3}\boldsymbol{F}_n^{(1)}\cdot\boldsymbol{A}_n^{(2)}\mathrm{d}x
\right|
+
\left|
\int_{\mathbb{R}^3}\boldsymbol{F}_n^{(2)}\cdot\boldsymbol{A}_n^{(1)}\mathrm{d}x
\right|.
\end{align*}
Since $\boldsymbol{F}_n-\boldsymbol{F}_n^{(1)}-\boldsymbol{F}_n^{(2)}$ is supported in $B_{4R}\setminus B_R$, the first three terms are estimated by Hölder and Sobolev as
\begin{align*}
\left|
\int_{\mathbb{R}^3}
\bigl(\boldsymbol{F}_n-\boldsymbol{F}_n^{(1)}-\boldsymbol{F}_n^{(2)}\bigr)\cdot\boldsymbol{A}_n
\mathrm{d}x
\right|
&\le cM^{\frac12}\varepsilon^{\frac56},
\\
\left|
\int_{\mathbb{R}^3}
\boldsymbol{F}_n^{(j)}\cdot
\bigl(\boldsymbol{A}_n-\boldsymbol{A}_n^{(1)}-\boldsymbol{A}_n^{(2)}\bigr)
\mathrm{d}x
\right|
&\le cM^{\frac56}\varepsilon^{\frac12}
\end{align*}
for $j=1,2$. It remains to estimate the mixed terms. We only consider the first one. Since $\boldsymbol{F}_n^{(1)}=0$ outside $B_{4R}$,
\begin{align*}
\left|
\int_{\mathbb{R}^3}\boldsymbol{F}_n^{(1)}\cdot\boldsymbol{A}_n^{(2)}\mathrm{d}x
\right|
&\le
\left|
\int_{B_R}\boldsymbol{F}_n^{(1)}\cdot\boldsymbol{A}_n^{(2)}\mathrm{d}x
\right|
+
\left|
\int_{B_{4R}\setminus B_R}\boldsymbol{F}_n^{(1)}\cdot\boldsymbol{A}_n^{(2)}\mathrm{d}x
\right|
\\
&\le
\left|
\int_{B_R}\boldsymbol{F}_n^{(1)}\cdot\boldsymbol{A}_n^{(2)}\mathrm{d}x
\right|
+
cM^{\frac12}\varepsilon^{\frac56}.
\end{align*}
For $x\in B_R$ we use the Biot--Savart representation only for the contribution of
$\boldsymbol{F}_n^{(2)}$ outside $B_R$. The part coming from $B_{4R}\setminus B_R$ gives
$cM^{\frac56}\varepsilon^{\frac12}$. For the part coming from
$\mathbb{R}^3\setminus B_{4R}$ we use $|x-y|\ge 3R$ and obtain
\begin{align*}
\left|
\int_{B_R}\boldsymbol{F}_n^{(1)}\cdot\boldsymbol{A}_n^{(2)}\mathrm{d}x
\right|
&\le
cM^{\frac56}\varepsilon^{\frac12}
+
\frac{c}{R^2}
\|\boldsymbol{F}_n^{(1)}\|_{L^1(\mathbb{R}^3)}
\|\boldsymbol{F}_n^{(2)}\|_{L^1(\mathbb{R}^3)}
\\
&\le
cM^{\frac56}\varepsilon^{\frac12}
+
\frac{cM^2}{R^2}.
\end{align*}
The same estimate holds for
$\int_{\mathbb{R}^3}\boldsymbol{F}_n^{(2)}\cdot\boldsymbol{A}_n^{(1)}\mathrm{d}x$. Hence
\begin{align*}
\left|Q[\boldsymbol{m}_n]-Q[\boldsymbol{m}_n^{(1)}]-Q[\boldsymbol{m}_n^{(2)}]\right|
\le
cM^{\frac12}\varepsilon^{\frac56}
+
cM^{\frac56}\varepsilon^{\frac12}
+
\frac{cM^2}{R^2}.
\end{align*}
Choosing $\varepsilon>0$ small and $R$ large, the right-hand side is smaller than $1$. Since all three Hopf invariants are integers, we obtain
\begin{align}\label{eq:chargesplit}
Q[\boldsymbol{m}_n]=Q[\boldsymbol{m}_n^{(1)}]+Q[\boldsymbol{m}_n^{(2)}].
\end{align}

\begin{align*}
|Q[\boldsymbol{m}^{(1)}_n]| \le c \mathcal{E}_H[\boldsymbol{m}^{(1)}_n]^{\frac43} \le c(\mathcal{E}_H[\boldsymbol{m}_n] + \mathcal{E}_H[\boldsymbol{m}^{(1)}_n;B_{4R} \setminus B_R])^{\frac43} \le c(M+\varepsilon)^{\frac43}.
\end{align*}

If $\ell=k$, then $I_k(H) \le \mathcal{E}_H[\boldsymbol{m}^{(1)}_n]$ for every $n \in \mathbb{N}$. In particular
\begin{align*}
I_k(H) &\le \mathcal{E}_H[\boldsymbol{m}^{(1)}_n] = \mathcal{E}_H[\boldsymbol{m}_n;B_R] + \mathcal{E}_H[\boldsymbol{m}^{(1)}_n; B_{4R} \setminus B_R]\\
&\le |\lambda \mathcal{E}_H[\boldsymbol{m}_n] - \mathcal{E}_H[\boldsymbol{m}_n;B_R]| + \lambda \mathcal{E}_H[\boldsymbol{m}_n] + \mathcal{E}_H[\boldsymbol{m}^{(1)}_n; B_{4R} \setminus B_R]\\
&\le \lambda \mathcal{E}_H[\boldsymbol{m}_n]  + c \varepsilon \to \lambda I_k(H) + c \varepsilon.
\end{align*}
Choosing $\varepsilon < \frac{1-\lambda}{c}I_k(H)$ contradicts this case. If $\ell=0$ then $Q[\boldsymbol{m}^{(2)}_n]=k$, hence we would derive a similar contradiction. Therefore $\ell \in \mathbb{Z} \setminus \{ 0 , k \}$ and we compute (for $n \in \mathbb{N}$ large enough)
\begin{align*}
\mathcal{E}_H[\boldsymbol{m}_n] &= \lambda \mathcal{E}_H[\boldsymbol{m}_n]  + (1 - \lambda ) \mathcal{E}_H[\boldsymbol{m}_n] \\
&\geq \mathcal{E}_H[\boldsymbol{m}_n;B_R] + \mathcal{E}_H[\boldsymbol{m}_n ;\mathbb{R}^3 \setminus B_{4R}] - c \varepsilon \\
&= \mathcal{E}_H[\boldsymbol{m}^{(1)}_n] + \mathcal{E}_H[\boldsymbol{m}^{(2)}_n] -\mathcal{E}_H[\boldsymbol{m}^{(1)}_n;B_{4R}\setminus B_{R}] - \mathcal{E}_H[\boldsymbol{m}^{(2)}_n;B_{4R}\setminus B_{R}]-   c \varepsilon  \\
&\geq \mathcal{E}_H[\boldsymbol{m}^{(1)}_n] + \mathcal{E}_H[\boldsymbol{m}^{(2)}_n] - c \varepsilon\\
&\geq I_\ell(H) + I_{k-\ell}(H) - c \varepsilon\\
\end{align*}
and with $n \to \infty$
\begin{align*}
I_k(H) \geq I_\ell(H) + I_{k-\ell}(H) - c \varepsilon.
\end{align*}
Although $\ell$ may depend on $\varepsilon$, it is integer-valued and uniformly bounded. Thus, along a sequence $\varepsilon_j\to 0$, we may assume that $\ell$ is constant. Letting $j\to\infty$ proves the claim.

<!-- prop-1: body_md -->
\label{prop:existence}
Let $H>\frac14$ and $k \in \mathbb{Z}$. If for all $\ell \in \mathbb{Z} \setminus \{ 0,k \}$ 
\begin{align*}
I_k(H) < I_\ell(H) + I_{k-\ell}(H),
\end{align*}
then $I_k(H)$ is attained.
<!-- end prop-1 -->
<!-- proof-8: body_md -->
Let $(\boldsymbol{m}_n)_{n \in \mathbb{N}} \subset H^2(\mathbb{R}^3;\mathbb{S}^2)$ be a minimizing sequence and assume without loss of generality $k \neq 0$. Using Lemma~\ref{lemma:vanishing} and Lemma~\ref{lemma:dichotomy}, $(e(\boldsymbol{m}_n))_{n \in \mathbb{N}}$, up to some subsequence, can only satisfy the compactness condition~\ref{case:compactness}. By translation invariance of $\mathcal E$ and $Q$ we may assume the concentrating centers are at the origin, i.e. for every $\varepsilon>0$ there exists $R>0$ such that
\begin{align} \label{eq:tightness}
\int_{\mathbb R^3\setminus B_R} e(\boldsymbol{m}_n)\mathrm{d}x \le \varepsilon
\quad \text{for all large } n .
\end{align}
The coercivity estimate \eqref{eq:densityCoercivity}, together with the elliptic estimate on $\mathbb{R}^3$, gives
\begin{align*}
\|\boldsymbol{m}_n - \boldsymbol{\hat{e}}_{3}\|_{H^2(\mathbb R^3)}^2 \le C \mathcal{E}_H[\boldsymbol{m}_n] \le C M .
\end{align*}
Since $H^2$ is reflexive there is a map $\boldsymbol{m}:\mathbb{R}^3 \to \mathbb{R}^3$ such that $\boldsymbol{m}-\boldsymbol{\hat{e}}_{3}\in H^2(\mathbb R^3;\mathbb R^3)$ with
\begin{align*}
\boldsymbol{m}_n - \boldsymbol{\hat{e}}_{3} \rightharpoonup \boldsymbol{m} - \boldsymbol{\hat{e}}_{3} \quad \text{in } H^2(\mathbb R^3) .
\end{align*}
By Rellich on bounded sets together with \eqref{eq:tightness}, we also have
\begin{align*}
\boldsymbol{m}_n - \boldsymbol{\hat{e}}_{3} \to \boldsymbol{m} - \boldsymbol{\hat{e}}_{3} \quad \text{in } H^1(\mathbb R^3).
\end{align*}
and hence, up to a subsequence,
\begin{align*}
\boldsymbol{m}_n \to \boldsymbol{m} \quad \text{and} \quad \nabla \boldsymbol{m}_n \to \nabla \boldsymbol{m} \quad \text{almost everywhere in } \mathbb R^3 .
\end{align*}
In particular $\boldsymbol{m}(x)\in \mathbb S^2$ almost everywhere, hence $\boldsymbol{m}\in H^2(\mathbb R^3;\mathbb S^2)$. Interpolating the strong $H^1$ convergence with the weak $H^2$ convergence also yields
\begin{align*}
    \|\boldsymbol{m}_n - \boldsymbol{m}\|_{H^s(\mathbb{R}^3)} \le\|\boldsymbol{m}_n - \boldsymbol{m}\|_{H^1(\mathbb{R}^3)}^{2-s}  \|\boldsymbol{m}_n - \boldsymbol{m}\|_{H^2(\mathbb{R}^3)}^{s-1}\to 0
\end{align*}
as $n\to \infty$ for every $1<s<2$. Furthermore, for $s>\frac32$, we conclude uniform convergence, hence $Q[\boldsymbol{m}]=Q[\boldsymbol{m}_n]=k$ for large $n \in \mathbb{N}$. Lastly, the energy is weakly lower semicontinuous
\begin{align*}

\label{EnDec}
Let $H>\frac14$. For every $k \in \mathbb{Z}\setminus\{0\}$ there exist $J \in \mathbb{N}$ and $d_{1},\dots,d_{J} \in \Sigma_H$ with $k=\sum_{i=1}^{J} d_{i}$ such that

<!-- end lemma-8 -->
<!-- proof-10: body_md -->
Fix $k \in \mathbb{Z}$. If $k \in \Sigma_H$ the statement is immediate with $J=1$ and $d_{1}^{(1)}=k$. Assume $k \notin \Sigma_H$. By \eqref{eq:SigmaH} there exist nonzero integers $d^{(2)}_{1},d^{(2)}_{2}$ with
\begin{align*}
k = d^{(2)}_{1} + d^{(2)}_{2}, \qquad I_k(H) \ge I_{d^{(2)}_{1}}(H) + I_{d^{(2)}_{2}}(H).
\end{align*}
If both $d^{(2)}_{1}$ and $d^{(2)}_{2}$ lie in $\Sigma_H$, the proof is complete. Otherwise, choose the smallest $i \in \{1,2\}$ with $d^{(2)}_{i}\notin \Sigma_H$. We then apply \eqref{eq:SigmaH} again to decompose $d^{(2)}_{i}$ into two nonzero integers. After $J$ refinement steps this yields a finite family of nonzero integers
\begin{align*}
\left( d^{(J)}_{i} \right)_{1 \le i \le J}
\end{align*}
such that
\begin{align*}
k = \sum_{i=1}^{J} d^{(J)}_{i}, \qquad
I_k(H) \ge  \sum_{i=1}^{J} I_{d^{(J)}_i}(H).
\end{align*}
Using
\begin{align*}
I_d(H) \ge C(H)|d|^{\frac34} \ge C(H) \quad \text{for every nonzero } d \in \mathbb{Z},
\end{align*}
we infer
\begin{align*}
I_k(H) \ge JC(H).
\end{align*}
By Lemma~\ref{lemma:energygrowth}, $I_k(H) \le C|k|^{\frac34}$, which implies
\begin{align*}
J \le \frac{I_k(H)}{C(H)}  \le \frac{C}{C(H)}|k|^{\frac34}.
\end{align*}
Thus the refinement terminates after finitely many steps, which proves the claim.
