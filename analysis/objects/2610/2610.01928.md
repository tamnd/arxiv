\label{thm:main}
Let $p\in [1,2]$ and $\alpha\in (0,1]$. Let $\lambda,\mu$ be probability measures on $\mathbb{R}^d$ admitting $C^{0,\alpha}$-densities satisfying $0<c^{-1}\leq \lambda,\mu\leq c$ on their support for some $c>0$. Assume $\,{\rm supp }\,\lambda$ and $\,{\rm supp }\,\mu$ are $C^{1,\alpha}$-domains. Suppose the optimal transport map  $T\colon \,{\rm supp }\,\lambda \to \,{\rm supp }\,\mu$ between $\lambda$ and $\mu$ is a bi-$C^{1,\alpha}$ map.  Consider $\pi_\varepsilon\in \Pi(\lambda,\mu)$ an optimal transport plan for regularised transport with regularisation parameter $\varepsilon^2>0$ and regularisation function $h_p$ with potentials $\phi_\varepsilon$ and $\psi_\varepsilon$. There is $\varepsilon_1>0$ such that if $\varepsilon\leq \varepsilon_1$, then $\phi_\varepsilon\in W^{2,\infty}(\,{\rm supp }\,\lambda)$, $\psi_\varepsilon\in W^{2,\infty}(\,{\rm supp }\,\mu)$ and there is $C>0$, independent of $\varepsilon$, such that
\begin{align*}
C^{-1}\textup{id} \leq \nabla^2 \phi_\varepsilon \leq C \textup{id}\quad \text{ a.e. on } \,{\rm supp }\,\lambda,\\
C^{-1}\textup{id} \leq \nabla^2 \psi_\varepsilon \leq C \textup{id}\quad \text{ a.e. on } \,{\rm supp }\,\mu.
\end{align*}
If $p\in [1,2)$, then $\phi_\varepsilon\in C^2(\,{\rm supp }\,\lambda)$, $\psi_\varepsilon\in C^2(\,{\rm supp }\,\mu)$. 

Furthermore if $p\in(1,2]$,  with $R_c:= \varepsilon^\frac 2 {d(p-1)+2}$, for every\footnote{$A\simeq B$ means $A\lesssim B$ and $B\lesssim A$, see Section \ref{sec:notation}.} $x\in \,{\rm supp }\,\lambda$, $y\in \,{\rm supp }\,\mu$,
\begin{gather*}
B\left(\frac 1 2\nabla \phi_\varepsilon(x),C^{-1}R_c\right)\cap \,{\rm supp }\,\mu\subset \mathcal S_x\subset B\left(\frac 1 2\nabla \phi_\varepsilon(x),C R_c\right)\\
B(\frac{1}{2}\nabla \psi_\varepsilon(y),C^{-1}R_c)\cap \,{\rm supp }\,\lambda\subset \mathcal T_y\subset B(\frac{1}{2}\nabla \psi_\varepsilon(y),C R_c)\\
\left|B\left(\frac 1 2\nabla \phi_\varepsilon(x),C^{-1}R_c\right)\cap \,{\rm supp }\,\mu\right|\simeq |B(C^{-1}R_c)| \simeq \left|B\left(\frac 1 2\nabla \psi_\varepsilon(y), C^{-1}R_c\right)\cap \,{\rm supp }\,\lambda\right|.
\end{gather*}

We first show \eqref{c52}. Label the $2d$ open faces of $Q$ as $F_1^\pm,\dots,F_d^\pm$, where $F_i^- = \{x_i=0\}\cap\partial Q$ and $F_i^+ = \{x_i=1\}\cap\partial Q$ (with the exception for $F_1^+ = \{x_1=a\}\cap \partial Q$). Writing $f$ as a sum of its restrictions to each open face and by linearity, it suffices to assume that $f$ is supported on a single face, say on	$F_1^- = \{0\}\times (0,1)^{d-1}$. Hence, we may identify $f$ with a Lipschitz function $f:(0,1)^{d-1}\to\mathbb{R}$. It then suffices to show that
	\begin{align*}
		\|\phi\|_{C^{1,\alpha}([0,a/2]\times [0,1]^{d-1}) }\lesssim \|f\|_{C^{\alpha}}
	\end{align*}
which combined with	interior Schauder estimates as well as the standard energy estimate ${\|\phi\|_{L^2(Q)}\lesssim \|f\|_{L^2}}$ completes the proof, see e.g. \cite{GT}.

\medskip
Let $\tilde Q:=(0,a)\times (-1,2)^{d-1}$. Extend $\phi$ and $f$ by even reflection. More precisely,	define $\sigma: (-1,2)\to (0,1)$ by  $\sigma(t)=|t|$ for $t\in(-1,1] $ and $ \sigma(t)=2-t$ for $t\in (1,2) $. Define

\tilde\phi(x_1,\dots,x_d):= \phi\big(x_1,\,\sigma(x_2),\dots,\sigma(x_d)\big)
\end{align*}
and similarly for $\tilde f$. Note that $\|\tilde f\|_{W^{1,\infty}((-1,2)^{d-1})}\leq \|f\|_{W^{1,\infty}((0,1)^{d-1})}$ and
\begin{align}
	-\Delta\tilde\phi=c\text{ on }\tilde Q,\quad
	\nabla\tilde\phi\cdot\nu=\tilde f\text{ on }\partial\tilde Q.
\end{align}
Let $\Omega$ be a smooth domain compactly supported in 
$\overline{\tilde Q} $ such that $ [0,a/2]\times [0,1]^{d-1} $ is compactly supported in $\bar\Omega$. As $\Omega$ is a smooth domain, by boundary Schauder theory and energy estimate, see \cite{GT},
\begin{align}
	\|\phi\|_{C^{1,\alpha}([0,a/2]\times [0,1]^{d-1}) }\leq \|\tilde\phi\|_{C^{1,\alpha}(\bar\Omega)}\lesssim \|\nabla\phi\|_{L^2(\tilde Q)}+ \|\bar f\|_{C^{\alpha}(\partial \tilde Q)}\lesssim \|f\|_{C^\alpha}
\end{align}
which concludes the proof.

\label{sec:basicPropertiesPotentials}	
Let $\lambda,\mu$ be compactly supported, absolutely continuous
probability measures on $\mathbb R^d$. Assume $h$ satisfies Assumption \ref{ass:h} and consider the problems \eqref{eq:regularisedProblem} and \eqref{eq:dualProblem}. Then the following statements hold.

\begin{enumerate}
\item The problem \eqref{eq:regularisedProblem} admits a unique minimiser
$\pi_\varepsilon$, and its dual \eqref{e:ROT-dual} admit a maximising pair
$(f_\varepsilon,g_\varepsilon)$. It holds
\begin{align*}
\frac{d\pi_\varepsilon}{dP}(x,y) &= (h')^{-1}\left(\frac{\xi(x,y)}{\varepsilon^2}\right)\\
\,{\rm supp }\,\pi_\varepsilon &= \{(x,y)\in \,{\rm supp }\,\lambda\times \,{\rm supp }\,\mu\colon \xi(x,y)\geq \varepsilon^2 h'(0)\}.
\end{align*}
In particular, there is no duality gap.
\item The  potentials $(\phi_\varepsilon,\psi_\varepsilon)$ admit convex Lipschitz representatives on their respective
supports, satisfying
\[
\mbox{Lip}(\phi_\varepsilon;\mbox{supp}\,\lambda)
+
\mbox{Lip}(\psi_\varepsilon;\mbox{supp}\,\mu)
\leq C,
\]
where $C$ depends only on the marginals and is independent of
$\varepsilon$. Moreover, $(\phi_\varepsilon,\psi_\varepsilon)$ may be extended to convex Lipschitz functions on $\mathbb{R}^d$ satisfying for every $x,y\in \mathbb{R}^d$,
\[
\int \frac{d\pi_\varepsilon}{dP}(x,y)\,d\mu(y)=1,
\qquad
\int \frac{d\pi_\varepsilon}{dP}(x,y)\,d\lambda(x)=1
\]

\item In the case $h=h_p$ with $p\in [1,2]$, it holds
\begin{equation}\label{eq:gradPhiPower}
\nabla\phi_\varepsilon(x)
= \begin{cases}
2\frac{\int_{\mathcal S_x} y\,\xi(x,y)^{\frac{2-p}{p-1}}\,d\mu(y)}
{\int_{\mathcal S_x}
\xi(x,y)^{\frac{2-p}{p-1}}\,d\mu(y)}\quad& \text{ if } p>1\\\
2\int y\,
\exp\!\left(\frac{\xi(x,y)}{\varepsilon^2}-1\right)\,d\mu(y). \quad&\text{ if } p=1.
\end{cases}
\end{equation}
Analogous formulas hold for $\nabla\psi_\varepsilon$.
\end{enumerate}

<!-- proof-2: body_md -->
The statements on $\,{\rm supp }\,\lambda\times \,{\rm supp }\,\mu$ follow from \cite[Proposition 18 and Lemmas 19--21]{GvalaniKoch1};
continuity extends the marginal identities to every point of the supports.

We now detail how to obtain the extensions to $\mathbb{R}^d$. Extend $\phi_\varepsilon$ by imposing its second marginal identity at every $x\in\mathbb R^d$. Keeping the original values of $\psi_\varepsilon$ on
$\operatorname{supp}\mu$, for each fixed $x$ we seek $t$ satisfying
$F(x,t)=1$, where
\[
F(x,t):=\int (h')^{-1}\left(\frac{2\langle x,y\rangle - t-\psi_\varepsilon(y)}{\varepsilon^2}\right)\,d\mu(y).
\]
The second marginal identity gives $F(x,\phi_\varepsilon(x))=1$ for $x\in \,{\rm supp }\,\lambda$. 
For each fixed $x$, boundedness of $y$ and $\psi_\varepsilon(y)$ on
$\operatorname{supp}\mu$ gives
\[
\lim_{t\to-\infty}F(x,t)=+\infty,
\qquad \lim_{t\to+\infty}F(x,t)=0.
\]
The function $F(x,\cdot)$ is continuous and strictly decreasing while positive.
Define $\phi_\varepsilon(x)$ as its unique solution of $F(x,t)=1$. Uniqueness preserves the original values on $\operatorname{supp}\lambda$. Define $\psi_\varepsilon$ symmetrically from the original values of $\phi_\varepsilon$ on $\operatorname{supp}\lambda$.
With $\xi(x,y)=\langle x, y\rangle-\phi_\varepsilon(x)-\psi_\varepsilon(y)$, we now have
\begin{equation}\label{eq:globalRow}
\int (h')^{-1}\left(\frac{\xi_\varepsilon(x,y)}{\varepsilon^2}\right)\,d\mu(y)=1
\end{equation}
and the analogous result for every $y\in\mathbb R^d$. The convexity argument
in the proof of \cite[Lemma 10]{GvalaniKoch1} relies only on \eqref{eq:globalRow} and its analogous result for $y\in \mathbb{R}^d$. Consequently, the extensions are convex and thus automatically Lipschitz on $\mathbb{R}^d$. The gradient formula now follows as in \cite[Proposition 4.2]{GonzalezGvalaniKoch}.

<!-- proof-3: body_md -->
The proof proceeds in a number of steps.

\medskip\noindent\textbf{Step 1. } Two facts about optimal plans for ROT.
First, recall monotonicity of $\,{\rm supp }\,\pi$: for $(x,y),(x',y')$ write
\begin{align*}
	\Delta=\Delta(x,y,x',y')
	:=|x-y'|^2+|x'-y|^2-|x-y|^2-|x'-y'|^2,
\end{align*}
Then  for  any $(x,y),\,(x',y')\in\,{\rm supp }\pi$ with $\Delta\leq0$, (see \cite[Equation (7)]{GvalaniKoch1},
\begin{align}\label{c44}
	(h')^{-1}\left(\frac{-\Delta}{2\epsilon^2}\right)\leq \max\left(\frac{d\pi}{dP}(x',y),\frac{d\pi}{dP}(x,y')\right).
\end{align}
The second fact claims existence of points with bounded density:  Let $(x,y)\in \,{\rm supp }\pi$ and $B\subset\mathbb{R}^d\times\mathbb{R}^d$ be such that $\pi(B)>0$, then 
\begin{align}\label{c43}
	\pi\left(\{(x',y')\in B\mid \max\big(\frac{d\pi}{dP}(x,y'),\frac{d\pi}{dP}(x',y) \big)\}\leq \frac{2}{\pi(B)} \right)>0.
\end{align}
Indeed,  it suffices to show that $\pi(B_1)<\frac{\pi(B)}{2}$ and $\pi(B_2)<\frac{\pi(B)}{2}$ where
\begin{align*}
	B_1:=\{(x',y')\in B\mid \frac{d\pi}{dP}(x',y)>\frac{2}{\pi(B)} \},\quad B_2:=\{(x',y')\in B\mid \frac{d\pi}{dP}(x,y')>\frac{2}{\pi(B)}\}.
\end{align*}
We next show that $\pi(B_1)<\frac{\pi(B)}{2}$ while the argument for $B_2$ is the same. Let $A_1\subset\mathbb{R}^d$ be the projection to the first marginal of $B_1$ so that $\pi(B_1)\leq \pi(A_1\times\mathbb{R}^d)\leq\lambda(A_1)$. Now the conclusion follows from the optimality condition
\begin{align*}
	1=\int \frac{d\pi}{dP}(x',y)\,d\lambda(x')\geq\int_{A_1}\frac{d\pi}{dP}(x',y)\,d\lambda(x')>\frac{2}{\pi(B)}\lambda(A_1)\geq\frac{2}{\pi(B)}\pi(B_1).
\end{align*}

\medskip\noindent\textbf{Step 2. } Proof of (i).
Fix $(x,y)\in\,{\rm supp }\pi$ with $x\in\Omega_0\cap Q_R$ and $e\in\mathbb{S}^{d-1}$ with $\langle e, e_1\rangle\geq \theta$. Let $0<r<\frac{1}{1+C_0}$ and $\bar r>0$ to be fixed below. We also use the shorthand notation
\begin{align*}
	v:=y-x-b(x),\qquad v':=y'-x'-b(x').
\end{align*}
We define 
\begin{align*}
	B_{C_0rR}&:=B_{C_0rR}(x+rRe)\overset{\eqref{c40}}{\subset} \Omega_0\cap Q_{2R},\\
	B(x,y)&:=\Bigl\{(x',y')\in B_{C_0rR}\times\mathbb{R}^d\mid \
	|y'-x'-b(x')|\le \bar r R,\quad
	\langle x'-(x+rRe),v\rangle\geq 0\Bigr\}.
\end{align*}
\medskip\noindent\textit{Substep 1. } Estimate of $\Delta$. We show that for any $(x',y')\in B(x,y)$,
\begin{align}\label{c41}
	-\Delta\geq rR \langle y-x-b(x),e\rangle\quad \mbox{provided}\quad \langle y-x-b(x),e\rangle\geq 2(L_1+1)(C_0+1)^2(r+\bar r)R.
\end{align}

For $(x',y')\in B(x,y)$, we have
\begin{align*}
	\Delta=2\langle x-x',\,y-y'\rangle
	=2\langle x-x',\,v-v'\rangle+2\langle x-x',\,(x-x')+(b(x)-b(x'))\rangle.
\end{align*}
Noting $x-x'=(x-x'+rRe)-rRe$ in the first term and estimating the last term using that $b$ is Lipschitz, we obtain that
\begin{align*}
	\Delta\leq 2\langle x-x'+rRe,\,v-v'\rangle+2\langle -rRe,\,v-v'\rangle+2(L_1+1)(C_0+1)^2
	(rR^2).
\end{align*}
Next, by definition of $B(x,y)$,
\begin{align*}
	\langle x-x'+rRe,\,v\rangle\le0,\qquad -\langle x-x'+rRe,\,v'\rangle\le C_0 r\bar r R^2,\\
	2\langle -rRe,\,v-v'\rangle=-2rR\langle v,e\rangle+2rR\langle e,v'\rangle\le-2rR\langle v,e\rangle+2r\bar rR^2.
\end{align*}
Collecting estimates gives
\begin{align*}
	\Delta\leq\ -2rR\langle v,e\rangle+2(C_0+1)r\bar rR^2+2(L+1)(1+C_0)^2(rR)^2,
\end{align*}
from which \eqref{c41} follows.

\medskip\noindent\textit{Substep 2. } Lower bound on $\pi(B(x,y))$. We show that with an appropriate choice of $r,\bar r$, 
\begin{align*}
	\pi(B(x,y))\sim (rR)^d.
\end{align*}
Let $H:=\{z\mid\langle z-(x+rRe),v\rangle\ge0\}$, so
$|B\cap H|=\omega_d(C_0rR)^d/2$. Since $B\cap H\subseteq\Omega_{0}\cap B_{2R}$, we have
\begin{align*}
	\pi\big((B\cap H)\times\mathbb{R}^d\big)=\lambda(B\cap H)\geq D\omega_d (C_0rR)^d/2.
\end{align*}
Thus by Markov's inequality,
\begin{align*}
	\pi(B_{C_0rR}\times\mathbb{R}^d\backslash B(x,y))\leq\frac{1}{(\bar rR)^2}\int_{B_{C_0rR}\times\mathbb{R}^d}|v'|^2\,d\pi
	\ \le\ \frac{(2R)^{d+2}E}{(\bar rR)^2}
	\ =\ \frac{2^{d+2}R^d E}{\bar r^2}.
\end{align*}
Choosing $\bar r$ to be
\begin{equation}\label{c42}
	\bar r=\Bigl(\frac{2^{\,d+4}}{D\,C_0^{\,d}\,\omega_d}\Bigr)^{1/2} r^{-d/2}E^{1/2},
\end{equation}
we obtain that $2^{d+2}R^d E/\bar r^2 \leq D\omega_d(C_0rR)^d/4$. Hence, we find that
\begin{align*}
	\pi\bigl(B(x,y)\bigr)\geq
	\frac{1}{4} D C_0^d\omega_d (rR)^d \gtrsim (rR)^d .
\end{align*}
The opposite inequality is obvious.

\medskip\noindent\textit{Substep 3. } Conclusion. By Substep 1 and 2, we know that whenever $(x,y)$ satisfies \eqref{c41} with the choice of $r,\bar r$ also satisfying \eqref{c42}, by \eqref{c44} and \eqref{c43} we can find $(x',y')\in B(x,y)$ such that for some $C<\infty$,
\begin{align}\label{c45}
	(rR)^d(h')^{-1}\left(\frac{rR(y-x-b(x))\cdot e}{2\epsilon^2} \right)\leq C.
\end{align}

We begin by choosing
\begin{align*}
	\bar r=E^{\frac 1 2},\quad\mbox{hence by \eqref{c42}}\quad r=K_1(d,D):=\left(\frac{2^{d+4}}{DC_0^dw_d} \right)^{\frac{1}{d}}.
\end{align*}
Therefore, if we assume
\begin{align}\label{c46}
	\langle y-x-b(x),e\rangle\geq 2(L_1+1)(C_0+1)^2(K_1+E^{\frac 1 2})R\geq K_2rR,
\end{align}
with $K_2=K_2(d,L_1):=2(L_1+1)(C_0+1)^2$, then \eqref{c45} holds, and consequently by monotonicity of $(h')^{-1}$ and the fact that $R\geq R_c$, we see that 
\begin{align*}
	(rR_c)^d(h')^{-1}\left(\frac{K_2(rR_c)^2}{2\epsilon^2} \right)\leq C.
\end{align*}
Now we choose $R_c$ such that it satisfies
\begin{align*}
	(rR_c)^d(h')^{-1}\left(\frac{K_2(rR_c)^2}{2\epsilon^2} \right)=2C.
\end{align*}
This provides a contradiction and we conclude that the opposite of \eqref{c46} must be true:
\begin{align}\label{c47}
	\langle y-x-b(x)), e\rangle\leq K_2 (K_1+E^{\frac 1 2})R\quad\text{for}\quad R\geq R_c.
\end{align}

If $E\geq 1$, \eqref{c47} completes the proof. Hence, we may assume that $E\leq 1$ and that \eqref{c47} holds. We choose now
\begin{align*}
	r&=\frac{(y-x-b(x))\cdot e}{R}K_3\mbox{ with } K_3=\min\{\frac{1}{K_2(1+K_1)(1+C_0)},\frac{1}{4K_1(L_1+1)(C_0+1)^2} \},\\
	\bar r&=K_1^{\frac d 2}\left(\frac{(y-x-b(x))\cdot e}{R} \right)^{-\frac d 2}E^{\frac{1}{2}}.
\end{align*}
Now we assume that \eqref{c41} holds for $(x,y)$, then the choice of $K_3$ ensures that $r\leq \frac{1}{1+C_0}$ and that if \eqref{c41} holds, then 
\begin{align*}
	\frac{\langle y-x-b(x),e\rangle}{R}\geq K_4(d,D,L_1) E^{\frac{1}{d+2}}\text{ with } K_3:=\left(\frac{2(L_1+1)(C_0+1)^2K_1^{\frac d 2}}{1-2(L_1+1)(C_0+1)^2K_1K_3} \right)^{\frac{2}{d+2}}.
\end{align*}
Therefore, by \eqref{c45} we have that
\begin{align*}
	K_3^{d}((y-x-b(x))\cdot e)^{d}(h')^{-1}\left(\frac{K_3\big((y-x-b(x))\cdot e\big)^2}{2\epsilon^2} \right)\leq C.
\end{align*}
Consequently, if we define $\tau(\epsilon)$ to be the solution of
\begin{align*}
	K_3^d (\tau(\epsilon))^{d}(h')^{-1}\left(\frac{K_3(\tau(\epsilon))^2}{2\epsilon^2} \right)=C,
\end{align*}
we obtain by monotonicity that
\begin{align*}
	\langle y-x-b(x),e\rangle\leq \tau(\epsilon),
\end{align*}
which finishes the proof.

\medskip\noindent\textbf{Step 3. } Proof of (ii). Let $(x,y)\in\,{\rm supp }\pi\cap (Q_R\cap \Omega_0)\times\mathbb{R}^d$ be given. We first observe that we may assume  $E\leq1$. Indeed, if $E\ge1$ the proof of \cite[Theorem 6]{GvalaniKoch1} applies without change: for $r<\frac 1 3$ and $x\in \Omega_0\cap Q_R$ satisfying \eqref{c48},
\begin{align*}
B_{rR}\left(x+2rR\frac{y-x-b(x)}{|y-x-b(x)|}\right)\subset\Omega_0\cap B_{2R},
\end{align*}
since  by \eqref{c48}, $x$ is of distance at least $RE^{1/2}\ge R$ away from $\partial\Omega_0$. By exactly the same proof as \cite[large transport regime in Theorem 6]{GvalaniKoch1}, it follows that
\begin{align*}
\frac{|y-x-b(x)|}{R}\lesssim E^{\frac{1}{2}}\quad\text{whenever}\quad E\geq 1.
\end{align*}

We may therefore assume $E\leq 1$ in the following. We also set
\begin{align*}
\delta:=RL_1+R\,\max\left(E^{\frac1{d+2}},\frac{\tau(\epsilon)}{R}\right)\mbox{ so that assumption \eqref{c48} is }\langle x,e_1\rangle\ge\delta.
\end{align*}
 For $r,\bar r$ to be chosen, define 
\begin{align*}
B_{rR}&:=B_{rR}\left(x+2rR\frac{y-x-b(x)}{|y-x-b(x)|}\right),\\
B(x,y)&:=\{(x',y')\in B\times \mathbb{R}^d\mid |y'-x'-b(x')|\leq\bar rR \}.
\end{align*}
We shall always chose $r$ such that $r<\min\{\frac{1}{3},\frac{1}{3}\max(E^{\frac{1}{d+2}},\frac{\tau}{R})\}$ so that $B_{rR}\subset Q_{2R}\cap \Omega_0$.

The computations of \cite[Step 2, Theorem 6]{GvalaniKoch1} give
\begin{align}
	&\Delta\le -rR|y-x-b(x)|\quad \forall\,(x',y')\in B(x,y)\label{c49}
    \end{align}
    as long as $|y-x-b(x)|\ge 18(L_1+1)rR+6\bar rR$. Further,
    \begin{align}
    \bar r=\Bigl(\frac{2\cdot 3^{\,d+2}}{D\,\omega_d}\Bigr)^{1/2}r^{-d/2}E^{1/2} 	\quad\Rightarrow\quad \pi\left(B(x,y)\right)\gtrsim(rR)^d. \label{c50}
\end{align}
As before, combining \eqref{c44} and \eqref{c43}, for any $r,\bar r$ and $(x,y)$ satisfying the above two constraints, we have
\begin{equation}\label{c51}
	(rR)^d(h')^{-1}\left(\frac{rR|y-x-b(x)|}{2\epsilon^2}\right) \le C 
\end{equation}
for some $C<\infty$.
Also let $P:=\dfrac{1}{54(L_1+1)}$.

\medskip\noindent\textit{Substep 1. } The regime $E^\frac 1 {d+2}\geq \frac{\tau(\epsilon)}{R}$. In this regime, we show that
\begin{align*}
|y-x-b(x)|\leq (1+K_5)RE^{\frac{1}{d+2}}\quad\text{where}\quad K_5:=6\Bigl(\dfrac{2\cdot3^{d+2}}{D\omega_d}\Bigr)^{1/2}\bigl(54(L_1+1)\bigr)^{d/2}.
\end{align*}

Assume for contradiction that the above is not true, then we choose $r,\bar r$ in \eqref{c50} to be
\begin{align*}
r=\frac{E^{\frac1{d+2}}}{54(L_1+1)},\quad r=\frac{K_5}{6}E^{\frac1{d+2}}.
\end{align*}
This choice as well the contradicting assumption implies that
\begin{align*}
|y-x-b(x)|\geq (1+K_5)RE^{\frac{1}{d+2}}\geq 18(L_1+1)rR+6\bar rR,
\end{align*}
which means that \eqref{c51} holds. Noting that $rR=PE^{\frac 1{d+2}}R$ combined with the assumption that $|y-x-b(x)|\geq (1+K_5)E^{\frac{1}{d+2}}R$ allows us to conclude
\begin{align*}
(PRE^{\frac1{d+2}})^d
(h')^{-1}\left(\frac{P(1+K)(RE^{\frac1{d+2}})^2}{2\epsilon^2}\right)\leq C.
\end{align*}
Since we are in the regime $RE^{\frac1{d+2}}\ge\tau(\epsilon)$ and
$z\mapsto (Pz)^d(h')^{-1}\!\bigl(\tfrac{P(1+K)z^2}{2\varepsilon^2}\bigr)$ is increasing, we find that
\begin{align*}
(P\tau(\epsilon))^d\,
(h')^{-1}\left(\frac{P(1+K)\tau(\epsilon)^2}{2\epsilon^2}\right) \le C .
\end{align*}
Now we define  $\tau(\epsilon)$ to be the unique solution of
\begin{align*}
(P\tau(\epsilon))^d\,
(h')^{-1}\left(\frac{P(1+K)\tau(\epsilon)^2}{2\epsilon^2}\right)=2C.
\end{align*}
This provides the desired contradiction.

\medskip\noindent\textit{Substep 2. }. The regime  $\frac{\tau(\epsilon)}{R}\geq E^{\frac1{d+2}}$. In this regime, we claim that
\begin{align*}
|y-x-b(x)|\leq (1+K_6)\tau(\epsilon)\quad\text{for some}\quad K_6=K_6(d,D,L_1)
\end{align*}
Indeed, the proof argues via contradiction and can be obtained by verbatim replacing $E^{\frac{1}{d+2}}$ by $\frac{\tau(\epsilon)}{R}$ in the proof of Step 1. We omit the details.

Finally, we remark that in the case $h=h_p$, explicit calculations easily show that one may take $R_c\simeq R^\frac 2 {d(p-1)+2}$.

&\int_{\#_R}|y-x|^2d\pi-\int_{\tilde Q_R}|\nabla\phi^r|^2\\
\lesssim& \left(\tau^\frac 1 2 E(\pi,4)^\frac 1 2+D(4)^\frac 1 2+D_{bdy}(4T)^\frac 1 2+\Lambda^\frac 1 2\left(\frac 1 {\min(\hat \kappa_{\lambda,R},\hat \kappa_{\mu,R})}\int_{\tilde Q_R} |\nabla \phi|^2\right)^\frac  1 2\right)\\
&\times\left(\tau^\frac 1 2 E(\pi,4)^\frac 1 2+D(4)^\frac 1 2+D_{bdy}(4T)^\frac 1 2+\Lambda^\frac 1 2+\left(\frac 1 {\min(\hat \kappa_{\lambda,R},\hat \kappa_{\mu,R})}\int_{\tilde Q_R} |\nabla \phi|^2\right)^\frac  1 2-\left(\int_{\tilde Q_R}|\nabla \phi^r|^2\right)^\frac 1 2\right)
\end{align}
Recall the energy estimate \eqref{energy}. Standard mollification estimates give
\begin{align}
\int_{\tilde Q_R}|\nabla\phi^r-\nabla\phi|^2\lesssim r(E(\pi,4)+D(4T)),\label{c14}
\end{align}
In combination with elementary estimates, this allows us to deduce
\begin{align*}
	&\int_{\#_R}|y-x|^2d\pi-\int_{\tilde Q_R}|\nabla\phi^r|^2\\
    \lesssim& \tau E(\pi,4)+D(4)+D_{bdy}(4)+\Lambda\\
    &+C(\tau) \left(1-\frac 1 {\min(\hat \kappa_{\lambda,R},\hat \kappa_{\mu,R})}\right)\int_{\tilde Q_R} |\nabla \phi|^2+C(\tau)\int_{\tilde Q_R} |\nabla \phi-\nabla \phi^r|^2\\
    \stackrel{\eqref{eq:goodData}}\lesssim& (\tau+r)E(\pi,4)+C_\tau\left(D(4)+D_{bdy}(4T)+\Lambda\right).
\end{align*}

\medskip\noindent\textbf{Step 3. } The second term.
We write \eqref{c4} as follows
\begin{align*}
\int_{\#_R}&\int_{\sigma}^{\tau}|\nabla\phi^r(X(t))|^2dtd\pi-\int_{\tilde Q_R}|\nabla\phi^r|^2\\
&=\int_{\#_R}\int_{0}^{1}I(X(t)\in\tilde Q_R)|\nabla\phi^r(X(t))|^2dtd\pi-\int_{\tilde Q_R}|\nabla\phi^r|^2dx=I
\end{align*}
Then observe that
\begin{align*}
	I(X(t)\in\tilde Q_R)\leq I\left(X(t)\in\tilde Q_R,\:\exists s\in[0,1]\: X(s)\notin\partial(\tilde Q_R)\right)+I\left(X(t)\in\tilde Q_R,\forall t\in[0,1]\right)
\end{align*}
so that we may estimate
\begin{align*}
I\lesssim& \sup_{\tilde Q_R}|\nabla\phi^r|^2\pi(\partial(\#_R))+[|\nabla\phi^r|^2]_{\alpha,\tilde Q_R}\int_{\#_R}|y-x|^\alpha d\pi\\
\stackrel{\eqref{c15},\eqref{c16}}\lesssim& \left(\frac 1 {r^{d-1}}(\pi(\partial(\#_R))+\frac 1 {r^{d-1+\alpha}} \left(\int_{\#_R} |y-x|^2 d\pi\right)^\frac \alpha 2\right) (E(\pi,4)+D(4)+D_{bdy}(4)) \\
\stackrel{\eqref{eq:goodCrossingMeasure}}\lesssim& \left(\frac{\tau}{r^{d-1}}+\frac{E(\pi,4)^\alpha}{r^{d-1+\alpha}}\right) E(\pi,4)+D(4)+D_{bdy}(4).
\end{align*}


\medskip\noindent\textbf{Step 4. } The third term.
By \eqref{c12} and \eqref{c13}, we need to estimate
\begin{align*}
&\int_{\tilde Q_R}\phi^r(-d(\mu-\lambda))+\int_{\partial\tilde Q_R}\phi^r(\bar g^r_R-\bar f^r_R+\hat g_{1,R}-\hat g_{0,r}-d(g_R-f_R))\\
&=\left[\int_{\tilde Q_R}\phi^r(\kappa_{\lambda,R}\mathbf{1}_{\Omega_0}-\kappa_{\mu,R}\mathbf{1}_{\Omega_1})+\int_{\partial\tilde Q_R}\phi^r(\hat g_{1,R}-\hat g_{0,R})\right]\\
&+\int_{\tilde Q_R}\phi^r(-\kappa_{\lambda,R}\mathbf{1}_{\Omega_0}+\kappa_{\mu,R}\mathbf{1}_{\Omega_1}-d(\mu-\lambda))+\int_{\partial\tilde Q_R}\phi^r(\bar g^r_R-\bar f^r_R-d(g_R-f_R))\\
&=:I+II+III.
\end{align*}
Let us first estimate $I$. Using the normalizing condition $\int_{\tilde Q_R}\phi^r=0$ and that $\tilde Q_R=\hat Q_R\cup (Q_R\cap\{-D_{bdy}(4)^{\frac 1 2}< x_1\leq D_{bdy}(4)^{\frac 1 2}\})$, we can rewrite the first term in $I$ as follows,
 
\begin{align*}
&\int_{\tilde Q_R}\phi^r(\kappa_{\lambda,R}\mathbf{1}_{\Omega_0}-\kappa_{\mu,R}\mathbf{1}_{\Omega_1})=\int_{\hat Q_R}\phi^r(\kappa_{\lambda,R}-\kappa_{\mu,R})+\int_{-D_{bdy}(4)^{\frac 1 2}}^{D_{bdy}(4)^{\frac 1 2}}\int_{Q_R'}\phi^r(\kappa_{\lambda,R}\mathbf{1}_{\Omega_0}-\kappa_{\mu,R}\mathbf{1}_{\Omega_1})\\
=&\int_{-D_{bdy}(4)^{\frac 1 2}}^{D_{bdy}(4)^{\frac 1 2}}\int_{ Q_R'}\phi^r(\kappa_{\lambda,R}-\kappa_{\mu, R})+\int_{-D_{bdy}(4)^{\frac 1 2}}^{D_{bdy}(4)^{\frac 1 2}}\int_{Q_R'}\phi^r(\kappa_{\lambda,R}\mathbf{1}_{\Omega_0}-\kappa_{\mu,R}\mathbf{1}_{\Omega_1})
\end{align*}
 
where $Q_R':=(-R,R)^{d-1}$ is the $(d-1)$-dimensional cube. Thus we have that $I=I_1+I_2$ where 
\begin{align*}
I_1&:=\int_{-D_{bdy}(4)^{\frac 1 2}}^{D_{bdy}(4)^\frac 1 2}\int_{ Q_R'}\phi^r(\kappa_{\lambda,R}-\kappa_{\mu, R}),\\
I_2&:=\int_{-D_{bdy}(4)^{\frac 1 2}}^{D_{bdy}(4)^{\frac 1 2}}\int_{Q_R'}\phi^r(\kappa_{\lambda,R}\mathbf{1}_{\Omega_0}-\kappa_{\mu,R}\mathbf{1}_{\Omega_1})+\int_{\partial\tilde Q_R}\phi^r(\hat g_{1,R}-\hat g_{0,R}).
\end{align*}
We now estimate $I_1$. Note that $\sup_{\tilde Q_R}|\phi^r|\lesssim \sup_{\tilde Q_R}|\nabla\phi^r|$, therefore,
\begin{align*}
|I_1|&\lesssim  D_{bdy}(4)^\frac 1 2|\kappa_{\lambda, R}-\kappa_{\mu, R}|\sup_{\tilde Q_R}|\nabla\phi^r|\\
&\overset{\eqref{eq:goodData2},\eqref{c15}}{\lesssim} D_{bdy}(4) (D(4))^{\frac 1 2}\cdot\frac{1}{r^{\frac{d-1}{2}}}(E(\pi,4)+D(4)+D_{bdy}(4T))^{\frac 1 2}\\
&\lesssim \frac{\tau}{r^{\frac{d-1}{2}}}E(\pi,4)+C_\tau (D(4)+D_{bdy}(4T)).
\end{align*}
To estimate $I_2$,  we recall the Poincar\'e inequality in a thin rectangle: for a function $ f:Q_R\cap\{-\delta \leq x_1\leq \delta\}\to \mathbb{R} $ such that $ f(-\delta,x')\equiv 0 $, it holds that
\begin{align}\label{c17}
	\|f\|_{L^2(Q_R\cap\{-\delta\leq x_1\leq \delta\})}\lesssim \delta \|\nabla f\|_{L^2(Q_R\cap\{-\delta\leq x_1\leq \delta\})}.
\end{align} 
Then we have by definition of $\hat g_{1,R}$ and $\hat g_{0,R}$, that is \eqref{hatg-def},
\begin{align*}
I_2&=\int_{-D_{bdy}^{\frac 1 2}}^{D_{bdy}^{\frac 1 2}}\int_{Q_R'}\phi^r(\kappa_{\lambda,R}\mathbf{1}_{\Omega_0}-\kappa_{\mu,R}\mathbf{1}_{\Omega_1})+\int_{\partial\tilde Q_R}\phi^r(\hat g_{1,R}-\hat g_{0,R})\\
&=\int_{-D_{bdy}(4)^{\frac 1 2}}^{D_{bdy}(4)^{\frac 1 2}}\int_{Q_R'}\phi^r(x_1,x')(\kappa_{\lambda,R}\mathbf{1}_{\Omega_0}-\kappa_{\mu,R}\mathbf{1}_{\Omega_1})\\
&+\int_{-D_{bdy}(4)^{\frac 1 2}}^{D_{bdy}(4)^{\frac 1 2}}\int_{Q_R'}\phi^r(-D_{bdy}(4)^{\frac 1 2},x')(\kappa_{\mu,R}\mathbf{1}_{\Omega_1}-\kappa_{\lambda,R}\mathbf{1}_{\Omega_0})
\end{align*}
Using $|\kappa_{\lambda,R}\mathbf{1}_{\Omega_0}-\kappa_{\mu,R}\mathbf{1}_{\Omega_1}|\leq 2 $, this implies
\begin{align*}
|I_2|&\leq 2\int_{-D_{bdy}(4)^{\frac 1 2}}^{D_{bdy}(4)^{\frac 1 2}}\int_{Q_R'}|\phi^r(-D_{bdy}(4)^{\frac 1 2},x')-\phi^r(x_1,x')|dx_1dx'\\
&\lesssim D_{bdy}(4)^{\frac{1}{4}}\left(\int_{-D_{bdy}(4)^{\frac 1 2}}^{D_{bdy}(4)^{\frac 1 2}}\int_{Q_R'}|\phi^r(-D_{bdy}(4)^{\frac 1 2},x')-\phi^r(x_1,x')|^2 \right)^{\frac{1}{2}}\\
&\overset{\eqref{c17},\eqref{energy}}{\lesssim} D_{bdy}(4)^{\frac 3 4}(\int_{\tilde Q_R}|\nabla\phi^r|^2)^{\frac{1}{2}}\\
&\lesssim D_{bdy}(4)^{\frac{3}{4}}(E(\pi,4)+D(4)+D_{bdy}(4T))^{\frac{1}{2}}\leq \tau E(\pi,4)+C_{\tau}(D(4)+D_{bdy}(4T)).
\end{align*}
Combining the estimates for $I_1$ and $I_2$, we obtain that 
\begin{align*}
|I|\lesssim\frac{\tau}{r^{\frac{d-1}{2}}}E(\pi,4)+C_\tau (D(4)+D_{bdy}(4T)).
\end{align*}

The control of $II$ and $III$ follows similarly as in \cite{KochOtto} by exploiting regularity of $\phi^r$. For $II$, we let $\pi$ be optimal for $W_2(\kappa_{\mu,R}dx\llcorner\Omega_{1,R},\mu\llcorner Q_R)$, then 
\begin{align*}
\int_{\tilde Q_R}\phi^r(\kappa_{\mu,R}\mathbf{1}_{\Omega_1}-d\mu)=&\int(\phi^r(x)-\phi^r(y))d\pi\leq \sup_{\tilde Q_R}|\nabla\phi^r|\int|x-y|d\pi\\
\lesssim& \sup_{\tilde Q_R}|\nabla\phi^r|W_2(\kappa_{\mu,R}dx\llcorner\Omega_{1,R},\mu\llcorner Q_R)\\
\overset{\eqref{c15},\eqref{eq:goodData2}}{\lesssim}&\frac{1}{r^{\frac{d-1}{2}}}(E(\pi,4)^{\frac{1}{2}}+D(4)^{\frac{1}{2}}+D_{bdy}(4T)^\frac 1 2)(D(4)^{\frac{1}{2}}+D_{bdy}(4T)^\frac 1 2)
\end{align*}
With similar considerations with $\mu$ replaced by $\lambda$, we obtain
\begin{align*}
II\lesssim& \frac{1}{r^{\frac{d-1}{2}}}(E(\pi,4)^{\frac{1}{2}}+D(4)^{\frac{1}{2}}+D_{bdy}(4T)^\frac 1 2)(D(4)^{\frac{1}{2}}+D_{bdy}(4)^\frac 1 2)\\
\lesssim& \frac{\tau}{r^{\frac{d-1}{2}}}E(\pi,4)+C_\tau(D(4)+D_{bdy}(4T)).
\end{align*}
We next estimate $III$. Letting $\pi$ be optimal for $W_2(\bar g,g)$, we see by similar arguments as for $II$,
\begin{align*}
\int_{\partial\tilde Q_R}\phi^r(\bar g^r-dg)&=\int_{\partial\tilde Q_R}\phi^r(\bar g^r-\bar g)+\int_{\partial\tilde Q_R}\bar g-dg\\
&\lesssim \left(r\int_{\tilde Q_R}|\nabla\phi^r|^2\int_{\partial\tilde Q_R}\bar g^2 \right)^{\frac{1}{2}}+\sup_{\tilde Q_R}|\nabla\phi^r|W_2(\bar g,g)(\int_{\partial\tilde Q_R}\bar g)^{\frac{1}{2}}\\
&\overset{\eqref{c15},\eqref{eq:goodWasserstein},\eqref{eq:goodBoundaryData}}{\lesssim}\left(r^{\frac{1}{2}}+\frac{\tau}{r^{\frac{d-1}{2}}}\right)(E(\pi,4)+D(4)+D_{bdy}(4T)).
\end{align*}
The same argument shows that the above is also true with $g$ replaced by $f$, hence 
\begin{align*}
III\lesssim \left(r^{\frac{1}{2}}+\frac{\tau}{r^{\frac{d-1}{2}}}\right)(E(\pi,4)+D(4)+D_{bdy}(4T)).
\end{align*}
Combining the estimates for $I,II$ and $III$ above, we obtain the estimate of \eqref{c5},
\begin{align}\label{c20}
\int_{\tilde Q_R}&\phi(-\Delta\phi-d(\mu-\lambda))+\int_{\partial \tilde Q_R}\phi(\nabla\phi\cdot\nu-d(g-f))\notag\\
&\lesssim (r^{\frac{1}{2}}+\frac{\tau}{r^{\frac{d-1}{2}}})E(\pi,4)+\frac{C_{\tau}}{r^{\frac{d-1}{2}}}(D(4)+D_{bdy}(4T)).
\end{align}

\medskip\noindent\textbf{Step 5. } Conclusion.
We combine the estimates \eqref{c3}-\eqref{c5} obtained from previous sections to see that 
\begin{align*}
\int_{\#_R}\int_{\sigma}^{\tau}|y-x-\nabla\phi^r(X(t))|^2dtd\pi&\lesssim \left(\frac{\tau+E(\pi,4)}{r^{d-1+\alpha}}+r^{\frac{1}{2}}\right)E(\pi,4)+\frac{C_\tau}{r^{d-1+\alpha}}(D(4)+D_{bdy}(4T)).
\end{align*}
To finish the proof, we need to estimate the LHS and then choose $\tau$ and $r$. We begin by noting that
\begin{align}\label{c26}
(\tau-\sigma)(x,y)=1\text{ for any }(x,y)\notin \partial(\#_R).
\end{align}
Therefore we have that
\begin{align*}
&\int_{\#_1}|y-x-\nabla\phi^r(x)|^2d\pi\\
&\leq\int_{\#_1\backslash \partial(\#_R)}|y-x-\nabla\phi^r(x)|^2d\pi+\int_{\partial(\#_R)}2|y-x|^2+2|\nabla\phi^r(x)|^2d\pi\\
&\overset{\eqref{c26}}{\lesssim} \int_{\#_1}\int_{\sigma}^{\tau}|y-x-\nabla\phi^r(X(t))|^2dtd\pi+\int_{\#_1}\int_{\sigma}^{\tau}|\nabla\phi^r(X(t))-\nabla\phi^r(X(0))|^2dtd\pi\\
&\quad\quad+\frac{\tau}{r^{d-1}}(E(\pi,4)+D(4)+D_{bdy}(4T))
\end{align*}
where for the last line we also used \eqref{eq:goodCrossing}, \eqref{eq:goodCrossingMeasure} and \eqref{c15}. We next estimate
\begin{align*}
&\int_{\#_1}\int_{\sigma}^{\tau}|\nabla\phi^r(X(t))-\nabla\phi^r(X(0))|^2dtd\pi\lesssim [|\nabla\phi^r|^2]_{\alpha,\tilde Q_{\frac{3}{2}}} \int |y-x|^{2\alpha} d\pi\\
&\overset{\eqref{c16}}{\lesssim}\frac{1}{r^{d-1+\alpha}}\left(E(\pi,4)+D(4)+D_{bdy}(4T)\right)^{1+\alpha}.
\end{align*}
Collecting estimates, we conclude that
\begin{align}\label{eq:finalEstimateApproximation}
\int_{\#_1}&|y-x-\nabla\phi^r(x)|^2\,\mathrm{d}\pi\lesssim(\frac{\tau+E(\pi,4)^\alpha}{r^{d-1+\alpha}}+r^{\frac{1}{2}})E(\pi,4)+\frac{C_{\epsilon_1}}{r^{d-1+\alpha}}(D(4)+D_{bdy}(4T)).
\end{align}

Introduce now
\begin{align*}
-\Delta\tilde\phi^r = \hat\kappa_{\mu,R}-\hat\kappa_{\lambda,R}\mbox{ in }\tilde Q_R,\quad \nabla\tilde\phi^r\cdot\nu = h^r\mbox{ on }\partial\tilde Q_R,
\end{align*}
where $h^r=\bar g_R^r-\bar f^r_R+T_r(\hat g_{1,R}-\hat g_{0,R})$. Here for any function $f$ defined on $F=\partial\tilde Q_R \cap \{x_1 = - D_{bdy}^\frac 1 2\}$, we define
\begin{align*}
    T_r(f)(x) = f^r(x)-\langle\nabla f^r(0),x\rangle.
\end{align*}
Note that $\nabla' T_r(f)(0)=0$. We note that since $\nabla' \hat g_{1,r}=\nabla' \hat g_{0,R}=0$,
\begin{align*}
\|h^r\|_{L^2(F)}\lesssim& \|\hat g_{1,R}^r-\hat g_{0,R}^r\|_{L^2(F)}+r^\alpha ([\nabla' g_{0,R}]_{\alpha,Q'_R}+[\nabla' g_{1,R}]_{\alpha,Q'_R}).
\end{align*}
By standard properties of mollification and using the definition \eqref{hatg-def} of $\hat g_{0,R},\hat g_{1,R}$ as well as Assumption \ref{ass:tangencyLocal}, we deduce
\begin{align}
\|h^r\|_{L^2(F)}\lesssim& \|\hat g_{1,R}^r-\hat g_{0,R}^r\|_{L^2(F)}+r^\alpha ([\nabla' g_{0,R}]_{\alpha,Q'_R}+[\nabla' g_{1,R}]_{\alpha,Q'_R})\nonumber\\
\lesssim& r \left(\|\hat g_{0,R}\|_{C^{0,1}(\partial \tilde Q_R)}+\|\hat g_{1,R}\|_{C^{0,1}(\partial \tilde Q_R)}\right)+r^\alpha ([\nabla' g_{0,R}]_{\alpha,Q'_R}+[\nabla' g_{1,R}]_{\alpha,Q'_R})\nonumber\\
\lesssim r^\alpha D'_{bdy}(R).\label{eq:difficultFace}
\end{align}

By triangle inequality,
\begin{align*}
&\int_{\#_1} |y-x-\nabla \tilde \phi^r|^2\mathrm d\pi \\
=& \int_{\#_1} |y-x-\nabla \phi^r|^2\mathrm d \pi+\int_{\#_1} \langle \nabla \phi^r-\nabla \tilde \phi^r,2(y-x)-\nabla \phi^r-\nabla \tilde \phi^r\rangle\mathrm d \pi\\
\leq& \int_{\#_1} |y-x-\nabla \phi^r|^2 \mathrm d \pi+\|\nabla \phi^r-\nabla \tilde \phi^r\|_{L^2(\tilde Q_R)}\left(E(\pi,4)+\|\nabla \phi^r\|_{L^2(\tilde Q_R)}^2+\|\nabla \tilde\phi^r\|_{L^2(\tilde Q_R)}^2\right)^\frac 1 2.
\end{align*}

By energy estimate and \eqref{eq:difficultFace},
\begin{align*}
\|\nabla \phi^r-\nabla \tilde \phi^r\|_{L^2(\tilde Q_R)}\lesssim& \|h^r\|_{L^2(\partial \tilde Q_R)}\lesssim r^\alpha D'_{bdy}(R).
\end{align*}

In combination with \eqref{energy}, we have shown
\begin{align*}
\int_{\#_1} |y-x-\nabla \tilde \phi^r|^2\lesssim r D_{bdy}(E(\pi,4)+D(4)+D_{bdy}'(4T))
\end{align*}
and it follows that \eqref{eq:finalEstimateApproximation} holds with $\tilde \phi$ in place of $\phi$. Choosing first $r\ll 1$ sufficiently small and then ensuring $\frac{\tau+E(\pi,4)^\alpha}{r^{d-1+\alpha}}\leq \frac{\tau+\varepsilon_1^\alpha}{r^{d-1+\alpha}}\ll 1$, we conclude the proof.

\medskip\noindent\textbf{Step 6. } Proof of \eqref{e:phi-C2alpha}. By Lemma \ref{lem:regularity}, in particular \eqref{c53} applied to the equation \eqref{c12}, as well as \eqref{eq:difficultFace}, we have 
\begin{align*}
\|\tilde\phi\|_{C^{3}(\tilde Q_1)}\lesssim \|\bar f^r_R\|^2_{C^2(\partial \tilde Q_R)} +\|\bar g^r_R\|^2_{C^2(\partial \tilde Q_R)}+\|\hat g_{0,R}^r\|^2_{C^{2}(Q_R')}+\|\hat g_{1,R}^r\|_{C^{2}(Q_R')}^2+D_{bdy}'(R).
\end{align*}
With $r$ being fixed relative to $\tau$ in the last step, standard mollification estimates give 
\begin{align*}
\|\bar f^r_R\|^2_{C^\alpha(\partial \tilde Q_R)} +\|\bar g^r_R\|^2_{C^\alpha(\partial \tilde Q_R)}\leq C\int_{\partial \tilde Q_R}\bar f_R^2+\bar g_R^2\overset{\eqref{eq:goodBoundaryData}}{\leq} C (E(\pi,4)+D(4)+D_{bdy}(4T))
\end{align*}
where $C=C(\tau,d,\alpha)$. On the other hand, similar estimates in combination with the definition \eqref{hatg-def} of $\hat g_{0,R},\hat g_{1,R}$ as well as Assumption \ref{ass:tangencyLocal} give
\begin{align*}
\|\hat g_{0,R}^r\|^2_{C^{2}(Q_R')}+\|\hat g_{1,R}^r\|_{C^{2}(Q_R')}^2\lesssim&\|\hat g_{0,R}\|^2_{C^{1,\alpha}(Q_R')}+\|\hat g_{1,R}\|_{C^{1,\alpha}(Q_R')}^2\\
\lesssim& D_{bdy}(R)+ [\nabla g_0]^2_{\alpha,Q_4'}+ [\nabla g_1]^2_{\alpha,Q_4'}\\
\lesssim& [\nabla g_0]^2_{\alpha,Q_4'}+ [\nabla g_1]^2_{\alpha,Q_4'},
\end{align*}
where the last inequality again follows from Assumption \ref{ass:tangencyLocal}.
Combining the above two estimates finishes the proof of \eqref{e:phi-C2alpha}.

\label{one step improvement}
Assume $\lambda,\mu\in \mathscr M(\mathbb{R}^d)$ admit $C^{0,\alpha}$-densities satisfying $\lambda(0)=\mu(0)=1$. Suppose $\pi\in \Pi(\lambda,\mu)$ satisfies assumptions (A),(B1) and (C). For any $\beta \in (0, 1)$ there exists $\varepsilon_1>0$ such that if $E(\pi,R)+\mathscr D(R,\lambda,R)\leq \varepsilon_1$, then there exist constants $\theta \in (0, 1]$ and $C_\beta \ge 1$ depending only on $d, \alpha, \beta, r$, symmetric $B \in \mathbb{R}^{d \times d}$, $b \in \mathbb{R}^d$ and $\gamma>0$ satisfying
\begin{equation}\nonumber
|B - Id|^2 + \frac{|b|^2}{R^2} \lesssim E(\pi,R) + \mathscr D(R,\lambda,\mu),
\end{equation}
such that with $s=(A,b,\gamma,1)$ and $\pi_s$ defined as in Section \ref{sec:changeOfCoordinates},
\begin{align*}
E(\pi_s,\theta R)\leq \theta^{2\beta} E(\pi,R)+C_\beta \left(\mathscr D(R,\lambda,\mu)+ \Lambda\right)\\
\mathscr D(\theta R,\lambda_s,\mu_s)\leq \theta^{2\alpha}\left(1+C\left(E(\pi,R)^\frac 1 2+\mathscr D(R,\lambda,\mu)^\frac 1 2\right)\right)\mathscr D(R,\lambda,\mu).

\mathscr D(r_k,\lambda_k,\mu_k)\leq& \theta^{2\alpha}\left(1+C\left(E_{k-1}+^\frac 1 2 + \mathscr D(r_{k-1},\lambda_{k-1},\mu_{k-1})^\frac 1 1\right)\right)\mathscr D(r_{k-1},\lambda_{k-1},\mu_{k-1}).
\end{align}

Here $\pi_k$ is a minimiser of regularised optimal transport between $\lambda_k$ and $\mu_k$ with regularisation parameter $\prod_{i=0}^k \gamma_k \varepsilon^2$.

Before justifying the application of Corollary \ref{cor:oneStepRegularised}, we claim that \eqref{eq:IterationUnjustified} implies for $r_k \gg R_c$ that there is $C>0$ independent of $k$ such that
\begin{align}\label{eq:claimIteration1}
r_k^{-2\alpha}\mathscr D(r_k,\lambda_k,\mu_k) \leq (1+\theta^{k\alpha}+C r_{k-1}^{-2}R_c^2)r_{k-1}^{-2\alpha}\mathscr D(r_{k-1},\lambda_{k-1},\mu_{k-1})\\
\label{eq:claimIteration2}
\sup_{i\leq k} \theta^{-2i\alpha} E_i\leq E(\pi,R)+C\left( \mathscr D(R,\lambda,\mu)+\frac{R_c^2}{\theta^{2k\alpha} r_k^2}\right).
\end{align}



We prove \eqref{eq:claimIteration1} by induction. The kase $k=1$ is clear, so suppose \eqref{eq:claimIteration1} holds for $k=1, \ldots, K-1$. By the induction hypothesis, we find
\begin{align*}
   \mathscr D(r_{k-1},\lambda_{K-1},\mu_{K-1}) \leq \theta^{2\alpha {K-1}}\prod_{k=1}^{K-2} (1+\theta^{\alpha k}+C r_{k-1}^{-1}R_c)\mathscr D(R,\lambda,\mu).
\end{align*}
Now note for $r_K \gg R_c$, i.e. $K \ll \frac{\log(R_c/R)}{\theta}$,
\begin{align*}
\prod_{k-1}^{K-2} \left(1+\theta^{\alpha k}+\frac{\varepsilon}{C\theta^{k-1}R}\right)\leq& \prod_{k-1}^{K-2} \left(1+\theta^{\alpha k}+\frac{R_c}{C\theta^{K-1}R}\right)\\
\leq& \prod_{k-1}^{k_K} (1+\theta^{2\alpha k})\prod_{k_K+1}^{K-2} \left(1+\frac{2C R_c}{\theta^{K-1}R}\right).
\end{align*}
Here $k_K = \max\{k\colon \theta^{\alpha k}\geq C(\theta^{K-1}R)^{-1} \varepsilon\}$. The first product clearly admits a bound independent of $k_K$. Regarding the second bound, as long as $\theta^k R\geq C_0 R_c$,
\begin{align*}
\left(1+\frac{2C R_c}{\theta^{K-1}R}\right)^K\leq \left(1+\frac{2C}{C_0}\right)^\frac{\log(C_0R_c/R)}{\log \theta}.
\end{align*}
By elementary calculations, this expression is bounded independently of $K$ if $C_0$ is chosen sufficiently large. It follows that there is $C>0$, independent of $K$ such that as long as $\theta^k R \gg R_c$,
\begin{align}\label{eq:uniformDataBound}
    \mathscr D(r_{K-1},\lambda_{K-1},\mu_{K-1})\leq C<+\infty.
\end{align}
Inserting this into the last equation of \eqref{eq:IterationUnjustified} and using the induction hypothesis, we deduce \eqref{eq:claimIteration1} for $K$.

We turn to proving \eqref{eq:claimIteration2} for $K$. Using \eqref{eq:uniformDataBound}, the induction hypothesis and \eqref{eq:IterationUnjustified}, we may ensure that
\begin{align*}
    \sup_{1\leq i\leq K} \theta^{-(2i\alpha)} E_i\leq& \theta^{2(\beta-\alpha)}\left(\sup_{1\leq i\leq K-1} \theta^{-2i\alpha} E_i+E(\pi,R)\right)+C \mathscr D(R,\lambda,\mu)+\frac{C R_c^2}{ \theta^{2K\alpha} r_K^2}\\
    \leq& E(\pi,R)+C\mathscr D(R,\lambda,\mu)+\frac{C R_c^2}{\theta^{2K\alpha} r_K^2}.
\end{align*}
To obtain the last line, we used that $\beta>\alpha$ and potentially decreased the choice of $\theta$. Thus \eqref{eq:claimIteration2} is established. We note that \eqref{eq:claimIteration2} implies that
\begin{align}\label{eq:uniformBoundEnergy}
    E_k\leq C\left(E(\pi,R)+\mathscr D(R,\lambda,\mu)+\frac{C R_c^2}{r_K^2}\right).
\end{align}

This almost justifies the iterated application of Corollary \ref{cor:oneStepRegularised} in \eqref{eq:IterationUnjustified}. However, at each stage the regularisation parameter changes. Thus the proof is complete, once we prove that the regularisation parameter remains comparable to $\varepsilon^2$. Thus, it suffices to establish for $K \ll \frac{\log(R_c/R)}{\theta}$,
\begin{align}\label{eq:iterationFinalClaim}
\left|\prod_{i=1}^K \gamma_k-1\right|\ll 1.
\end{align}
Indeed, with $k_K$ as above,
\begin{align*}
\left|\prod_{i=1}^K \gamma_k-1\right|\leq \prod_{j=1}^{k_K} |\gamma_j|\left(\sum_{i=1}^{k_K} |\gamma_i-1|^2\right)^\frac 1 2
\end{align*}
Using \eqref{eq:IterationUnjustified}, \eqref{eq:uniformDataBound} and \eqref{eq:uniformBoundEnergy}, we find
\begin{align*}
\prod_{j=1}^{k_K} |\gamma_j|^2 \leq& \prod_{j=1}^{k_K} \left(1+E_{j-1}+\mathscr D(r_{j-1},\lambda_{j-1},\mu_{j-1})+\frac{R_c^2}{r_{j-1}^2}\right)\\
\leq& \prod_{j=1}^{k_K} \left(1+\theta^{2 k_K\alpha}+C \sum_{i=1}^{j-1} \frac{R_c^2}{r_i^2}+\theta^{2\alpha(j-1)}+\frac{R_c^2}{r_{j-1}^2}\right)\\
\leq& \prod_{j=1}^{k_K} \left(1+\theta^{2k\alpha}+\frac{C R_c^2}{\theta^{2(j-1)}R^2}\right)\leq C<+\infty.
\end{align*}
Similarly, using \eqref{eq:IterationUnjustified} and \eqref{eq:claimIteration2},
\begin{align*}
\sum_{i=1}^{k_K} |\gamma_k-1|^2 \lesssim& \sum_{i=1}^{k_K} E_{i-1}+\mathscr D(r_{i-1},\lambda_{i-1},\mu_{i-1})+\frac{R_c^2}{r_{i-1}^{2}}\\
\lesssim& \sum_{i=1}^{k_K} \varepsilon_1 \theta^{2(i-1)\alpha}+\frac{R_c^2}{r_{i-1}^2}\lesssim \varepsilon_1+\frac{R_c^2}{r_{k_K}^2}\lesssim \varepsilon_1+C_0^{-2}.
\end{align*}
Enlarging $C_0$ if necessary, this shows \eqref{eq:iterationFinalClaim}.

Similar estimates show that with $B_k=\prod_{i=1}^{k}B_i$ and $b_k=\sum_{i=1}^k \Gamma_{i-1}^{-1} M_{i-1}^{-1} b_i$ where $M_i = B_i\cdots B_1$ and $\Gamma_i=\gamma_i\cdots \gamma_1$, we have for $k\leq k_K$,
\begin{align*}
|A_k-\textup{id}|^2+\frac{|b_k|}{R}\lesssim E(\pi,R)+\mathscr D(R,\lambda,\mu)+\frac{R_c^2}{r_k^2}.
\end{align*}

For $k\leq k_K$, rewriting the estimate for $E_{k}$ in \eqref{eq:IterationUnjustified} in terms of $\pi$, it is an easy calculation to see that
\begin{align*}
    \frac 1 {r^{d+2}} \int_{(Q_r(0)\times \mathbb{R}^d)\cup (\mathbb{R}^d\times Q_r(b_k))} |y-A_k x-b_k|^2 d\pi \lesssim E_p(\pi,R)+\mathscr D(R,\lambda,\mu)+\frac{R_c^2}{r_k^2}.
\end{align*}
Filling in the gaps between $r_k$ and $r_{k+1}$ in a routine fashion this completes the proof.

\label{lem:convexHullGeometry}
Suppose $\Omega\subset \mathbb{R}^d$ is a $C^{1,\alpha}$-domain and $\omega$ is a convex domain. Let $\alpha\in (0,1)$. There are $C=C(\Omega),\delta=\delta(\Omega)>0$ such that if $\textup{diam}(\omega\cap\Omega)\leq \delta$, then for $x\in \textup{conv}(\omega\cap \Omega)$,
\begin{align*}
d(x,\Omega)\leq C\textup{diam}(\omega\cap\Omega)^{1+\alpha}.
\end{align*}
Moreover, if $\omega \cap \Omega$ contains a ball of radius $c \textup{diam}(\omega \cap \Omega)$ for some $c>0$, then the estimate can be improved to
\begin{align*}
d(x,\omega\cap\Omega)\leq C\textup{diam}(\omega\cap\Omega)^{1+\alpha}
\end{align*}
for some $C=C(\Omega,c)$.
<!-- end lemma-9 -->
<!-- proof-17: body_md -->
Set $d= \textup{diam}(\omega\cap\Omega)$. We ensure $d<r_0$, where $r_0$ is the radius of a $C^{1,\alpha}$-boundary chart of $\Omega$. Then, we may assume $d(\omega\cap\Omega,\partial \Omega)<d$, as else $\textup{conv}(\omega\cap\Omega)\subset \Omega$ and the claim automatically holds.

After a change of coordinates, we may restrict ourselves to the situation where
\begin{align*}
\{(x',x_d)\colon x'\in B_{r_0}', r_0>x_d>g(x')\} \supset \omega\cap \Omega
\end{align*}
with $g(0)=0$, $\nabla g(0)=0$ and $[\nabla g]_{C^{0,\alpha}}\leq L$.

For $x'\in B_{r_0}'$, set $I(x') = \{t\colon (x',t)\in \omega\}$. As $\omega$ is convex, $I(x')$ is an interval and we write $I(x') = [l(x'),u(x')]$. Note that $l(x')$ is a convex function on the convex set where $I(x')$ is non-empty. In particular, we find
\begin{align*}
\Omega\cap \omega = \{(x',t)\colon x'\in B_{r_0}', t\in [\max(l(x'),g(x')), u(x')]\}.
\end{align*}

Let now $x_i = (x_i',t_i)\in \Omega \cap \omega$ and consider $\bar x = \sum \lambda_i x_i$, where $\sum \lambda_i = 1$. We may assume that $\bar x \not \in \Omega$, as else $d(\bar x,\omega\cap\Omega)=0$. Introduce $\bar x' = \sum \lambda_i x_i'$ and $\bar t = \sum \lambda_i t_i$. As $\omega$ is convex, $\bar t\in I(\bar x')$. Noting $t_i \geq \max(l(x_i'),g(x_i'))$, we estimate
\begin{align*}
\bar t \geq \sum \lambda_i \max(l(x_i'),g(x_i')).
\end{align*}
By convexity of $l$,
\begin{align}\label{eq:convexpart}
\sum \lambda_i l(x_i') \geq l(\bar x').
\end{align}
We claim that
\begin{align}\label{eq:claim}
\sum \lambda_i g(x_i')\geq g\left(\bar x'\right)- C L d^{1+\alpha}.
\end{align}
By Taylor-expansion, for some absolute constant $C>0$,
\begin{align*}
g(x_i') = g(\bar x')+\langle \nabla g(\bar x'),x_i'-\bar x'\rangle+R_i,\quad |R_i|\leq C L |x_i'-\bar x'|^{1+\alpha}.
\end{align*}
Consequently,
\begin{align*}
\sum \lambda_i g(x_i')=& g(\bar x')+\sum \langle \nabla g(\bar x'),\lambda_i(x_i'-\bar x')\rangle + \sum_i \lambda_i R_i\\
=& g(\bar x')+\sum_i \lambda_i R_i\geq g(\bar x')- C L d^{1+\alpha}.
\end{align*}
Combining \eqref{eq:convexpart} and \eqref{eq:claim}, we deduce
\begin{align}\label{eq:bargLower}
\bar t \geq \max(l(\bar x'), g(\bar x')-C L d^{1+\alpha})\geq \max(l(\bar x'), g(\bar x'))-C L d^{1+\alpha}.
\end{align}
Set $\tilde x = (\bar x', \max(l(\bar x'), g(\bar x'))$. Recall that $\bar t< g(\bar x')$ since $\bar x\notin\Omega$ and that by definition $\tilde x\in \Omega$. Therefore, 
\begin{align*}
d(\bar x,\Omega)\leq |\bar x-\tilde x| = \max(l(\bar x'), g(\bar x'))-\bar t\leq C L d^{1+\alpha}.
\end{align*}


For the moreover part, suppose $B_\rho(y_0)\subset \Omega\cap\omega$, where $\rho=c\,{\rm diam}(\omega\cap\Omega)$. Consider $z=y_0+\frac{\rho} 2 e_d$. Note $z=(z',t_z)\in \Omega\cap\omega$ and $t_z-g(z')\geq \frac{\rho} 2$. As $\omega$ is convex, the segment connecting $\bar x$ and $z$ is contained in $\omega$. Consider $(p',t_p)=p:=\bar x+s(z-\bar x)\in \omega$ for $s\in [0,1]$. Now by \eqref{eq:claim},
\begin{align*}
g(p')\leq s g(z')+(1-s) g(\bar x')+C L d^{1+\alpha}.
\end{align*}
We deduce
\begin{align*}
t_p-g(p')\geq& s (t_z-g(z))+(1-s)(\bar t-g(\bar x'))-C L d^{1+\alpha}\\
\stackrel{\eqref{eq:bargLower}}\geq& \frac{s \rho} 4-2C L d^{1+\alpha}
\end{align*}
Choose $s=K d^\alpha$ for a sufficiently large constant $K>0$ to ensure $t_p-g(p')> 0$ and hence $p\in \Omega \cap \omega$. Moreover, since $|y_0-\bar x|\leq 2 d$,
\begin{align*}
d(\bar x, \omega\cap \Omega)\leq |p-\bar x|\leq s (|y_0-\bar x|+\frac{\rho} 2)\lesssim  d^{1+\alpha}.
\end{align*}
This concludes the proof.

\label{lem:smallScaleGeometric}
Let $p\in (1,2]$. Assume $\,{\rm supp }\,\lambda,\,\,{\rm supp }\,\mu$ are $C^{1,\alpha}$-domains. Suppose $\pi$ solves $OT_{\varepsilon,h_p}(\lambda,\mu)$. There are $C_0,\varepsilon_1>0$ such that if $E(\pi,C_0 R_c)+\varepsilon+C_0^{-1}+\varepsilon\leq \varepsilon_1$, then there is $C_1>0$, independent of $\varepsilon$, such that for $x\in B_{C_0 R_c/6}, \,y\in B_{C_0 R_c/6}$,

\end{align*}

By symmetry, it suffices to consider $x\in B_{C_0 R_c/3}$. The second inclusion is an immediate consequence of Section \ref{sec:Linfty}. Moreover, by Proposition \ref{prop:smallScale}, $\phi_\varepsilon,\psi_\varepsilon$ are $W^{2,\infty}$-functions on $B_{C_0 R_c/3}$. Increasing $C_0$ if necessary, we deduce that for some $C>0$, independent of $\varepsilon$ and $C_0$,
\begin{align}\label{eq:Lipschitz}
\|\nabla\phi_\varepsilon\|_{W^{1,\infty}\left(\cup_{y\in B_{C_0 R_c/6}}\mathcal T_y\right)}+\|\nabla\psi_\varepsilon\|_{W^{1,\infty}\left(\cup_{x\in B_{C_0 R_c/6}}\mathcal S_x\right)}\leq C.
\end{align}

Note that by Jensen's inequality, with $\frac 1 p+\frac 1 q=1$,
\begin{align*}
\xi(x,\frac{1}{2} \nabla \phi_\varepsilon(x)) \geq \left(\int_{\mathcal S_x} \xi(x,y)^{q-2}\mathrm d\mu(y)\right)^{-1}\int_{\mathcal S_x} \xi(x,y)^{q-1}\mathrm d \mu(y).
\end{align*}

We estimate,
\begin{align*}
\int_{\mathcal S_x} \xi(x,y)^{q-2}\mathrm d \mu(y)\lesssim \max_{y\in \mathcal S_x} \xi(x,y)^{q-2} |\mathcal S_x|.
\end{align*}
Consequently, using the optimality condition and that we already established $|\mathcal S_x|\lesssim R_c^d$,
\begin{align*}
\xi\left(x,\frac{1}{2}\nabla \phi_\varepsilon(x)\right)\gtrsim& \frac{\varepsilon^{2(q-1)}}{R_c^d\max_{y\in \mathcal S_x} \xi(x,y)^{q-2}}.
\end{align*}


Finally, we use the concavity of $\xi$ to estimate for $y\in \mathcal S_x$,
\begin{align*}
   \xi(x,y) \geq  \xi(x,y_m)-\langle \nabla_y \xi(x,y),y_m-y\rangle.
\end{align*}
where $y_m$ is chosen so that $\xi(x,y_m) = \xi_m:=\max_{y\in \mathcal S_x} \xi(x,y)$. Now, for any $z\in \mathcal T_y$,
\begin{align*}
|\nabla_y \xi(x,y)|=2\left|x-\frac 1 2\nabla \psi_\varepsilon(y)\right|\leq  2|z-x|+2\left|z-\frac 1 2\nabla \psi_\varepsilon(y)\right|\leq 2\textup{diam}(\mathcal T_y)+2\left|z-\frac 1 2\nabla \psi_\varepsilon(y)\right|.
\end{align*}

As $\frac 1 2 \nabla \psi_\varepsilon(y)\in \textup{conv}(\mathcal T_y)$, $d\left(\frac 1 2 \nabla \psi_\varepsilon(y),\mathcal T_y\right)\leq \textup{diam}(\mathcal T_y)\lesssim R_c$. Taking infimums over $z$, we thus deduce $|\nabla_y \xi(x,y)|\lesssim R_c$ and hence for some $C>0$,
\begin{align*}
    \xi(x,y)\geq \xi(x,y_m)-C R_c |y_m-y|.
\end{align*}
Inserting this estimate into the Schr\"odinger equation and using the interior cone property, we deduce for $\lambda$-a.~e.~$x$,
\begin{align*}
q^{q-1}\varepsilon^{2(q-1)}=& \int_{\mathcal S_x} \xi(x,y)^{q-1}\mathrm d\mu(y)\geq \int (\xi(x,y_m)-C R_c |y_m-y|)_+^{q-1}\mathrm d\mu(y)\\
\gtrsim& \int (\xi(x,y_m)-C R_c t)_+^{q-1}t^{d-1}\mathrm d t\gtrsim \frac{\xi(x,y_m)^{q+d-1}}{R_c^d}
\end{align*}
Combining estimates, we have shown
\begin{align*}
    \xi\left(x,\frac 1 2\nabla \phi_\varepsilon(x)\right)\geq C\frac{\varepsilon^{2(q-1)}}{R_c^d \varepsilon^\frac{2(q-1)(q-2)}{q+d-1}R_c^\frac{d(q-2)}{q+d-1}}=C R_c^2.
\end{align*}

\begin{align*}

\end{align*}

By Lemma \ref{lem:convexHullGeometry}, applied to $\omega = \{y\colon \xi(x,y)\geq 0\}$ and $\Omega = \,{\rm supp }\,\mu$ we may find $y_0\in \,{\rm supp }\,\mu$ such that $\left|y_0-\frac 1 2 \nabla \phi_\varepsilon(x)\right|\leq c R_c^{1+\alpha}$. \begin{align*}
\left|x-\frac 1 2 \nabla \psi_\varepsilon(y_0)\right|\leq& \left| x-\frac 1 2 \nabla \psi_\varepsilon(y)\right|+C |y-y_0|\\
\leq&\left| x-\frac 1 2 \nabla \psi_\varepsilon(y)\right|+C \left|y-\nabla \phi_\varepsilon(x)\right|+\left|\frac 1 2\nabla \phi_\varepsilon(x)-y_0\right|
\end{align*}
We have already shown that  $\left| x-\frac 1 2 \nabla \psi_\varepsilon(y)\right|\lesssim R_c$. A symmetric argument gives $\left| y-\frac 1 2 \nabla \phi_\varepsilon(x)\right|\lesssim R_c$. We deduce that for some $c>0$,
\begin{align*}
\xi(x,y_0)\geq C R_c^2 - c R_c^{2+\alpha}.
\end{align*}
In particular, $\xi(x,y_m)\geq \xi(x,y_0)\geq \frac{C} 2 R_c^2$ for a sufficiently small choice of $\varepsilon_1$. Since for $y\in\mathcal S_x$, due to Section \ref{sec:Linfty} and Lemma \ref{lem:convexHullGeometry},

\end{align*}
we conclude that there is $c>0$ such that $B(y_m,2c R_c)\cap \,{\rm supp }\,\mu\subset \mathcal S_x$. Consequently,
\begin{align*}
    B\left(\frac 1 2\nabla \phi_\varepsilon(x),c R_c\right)\cap \,{\rm supp }\,\mu\subset \mathcal S_x,
\end{align*}
and
\begin{align*}

\end{align*}
This completes the proof.

Note that $A$ is symmetric. In particular, this shows that choosing $\tau>0$ sufficiently small, we can ensure $\#_{\tau R_0}\subset Q(\#_{R_0})$, where $Q$ is the affine transformation associated to $s$.

Further, since $T$ is a $C^{1,\alpha}$ map,
\begin{align*}
\int_{Q_{R_0}} |T(x)-A^2x-b|^2 \mathrm d \lambda(x)\lesssim R^{d+2+2\alpha}.
\end{align*}
Consequently, applying also Lemma \ref{prop:EnergyConvergence} and Lemma \ref{lem:EnergyConvergence},
\begin{align*}
\int_{\#_{R_0}} |y-A^2x-b|^2 \mathrm d\pi_\varepsilon\lesssim& \int_{\#_{R_0}} |y-T(x)|^2+\int_{\#_{R_0}} |T(x)-A^2x-b|^2 \mathrm d\pi_\varepsilon(x)\\
\lesssim& R_c^2+ R_0^{d+2+2\alpha}.
\end{align*}
Moreover, note that due to Proposition \ref{prop:EnergyConvergence} ,
\begin{align*}
\varepsilon^2\int_{\#_{R_0}} h_p\left(\frac{\mathrm d \pi_\varepsilon}{\mathrm d P}\right)\lesssim R_c^2.
\end{align*}
Consequently,
\begin{align*}
E_p(\pi_{s},\tau R_0)\lesssim \frac{R_c^2}{R_0^{d+2}} + R_0^{2\alpha}.
\end{align*}
Further, by \cite[Corollary 5.4.]{Wiesel}, there exists $\delta(\varepsilon)\to 0$ as $\varepsilon\to 0$ such that
\begin{align*}
\sup_{(x,y)\in \,{\rm supp }\pi_\varepsilon}|y-T(x)|\leq \delta(\varepsilon).
\end{align*}
Reducing first $R_0$ and ensuring $\varepsilon$ is sufficiently small, we may ensure, independently of $p$,
\begin{align}\label{eq:smallnessEnergy}
E_p(\tau R_0,\pi_{s})+\mathscr D(\tau R,\lambda_{s},\mu_{s})\leq \varepsilon_2,
\end{align}
as well as
\begin{align}\label{eq:topological}
\sup_{(x_s,y_s)\in  \,{\rm supp }\pi_{s}\cap \#_{\tau R_0}} \frac{|x_s-y_s|}{\tau R_0}\leq 1.
\end{align}
We recall that $\pi_s$ solves $OT_{\kappa^\frac{p-1} 2 \gamma^{\frac 1 2} \varepsilon,h_p}(\lambda_s,\mu_s)$. Note that $\kappa^\frac{p-1} 2 \gamma^{\frac 1 2}\varepsilon\simeq \varepsilon$.

Set $R_1 = \tau R_0$. 

\medskip\noindent\textbf{Step 1. } Treatment of the range $\max(C R_c,2R)\leq r\leq R_1$ by boundary regularity. 

In light of \eqref{eq:smallnessEnergy}, \eqref{eq:topological}, the fact that $\lambda_s(0)=\mu_s(0)=1$ and Assumption \ref{ass:tangency} holds in $Q_{\tau R}$, if $\varepsilon_2$ is sufficiently small, we are in a position to apply Corollary \ref{cor:oneStepRegularised}. We then find $C,C_2>0$, such that for $C R_c\leq r\leq R_1$, there exists $b_r\in \mathbb{R}^d$ and $A_r\in \mathbb{R}^{d\times d}$ symmetric with
\begin{align*}
\frac{|b_r|^2}{R_1^2}+|A_r-\textup{id}|^2\leq C_2 \varepsilon_2+\frac{C_2} C\\
\frac 1 {r^{d+2}}\int_{(Q_r(0)\times \mathbb{R}^d)\cup (\mathbb{R}^d \times Q_r(b_r))}|y-A_r x-b_r|^2 \mathrm d \pi_{s}\leq C_2 \varepsilon_2+\frac{C_2 R_c^2}{r^2}\leq C_2 \varepsilon_2+\frac{C_2}{C}.
\end{align*}

\medskip\noindent\textbf{Step 2. } Treatment of the range $C R_c \leq r \leq 2R$ by interior regularity. In light of Step 1, we find $b\in \mathbb{R}^d$ and $A\in \mathbb{R}^{d\times d}$ symmetric such that
\begin{align*}
\frac 1 {(2R)^{d+2}}\int_{(Q_{2R}(0)\times \mathbb{R}^d)\cup (\mathbb{R}^d\times Q_{2R}(b))}|y-A x-b|^2 \mathrm d \pi_{s}\leq C_2 \varepsilon_2+\frac{C_2} C\\
\frac{|b|^2}{R_1^2}+|A-\textup{id}|^2\leq C_2 \varepsilon_2+\frac{C_2} C.
\end{align*}
Note that with $z=Q_1(x_0)$, reducing $\tau$ if necessary, $(Q_{\tau R}(z)\times \mathbb{R}^d)\cup (\mathbb{R}^d\times Q_{\tau R}(z+b))\subset (Q_{2R}(0)\times \mathbb{R}^d)\cup (\mathbb{R}^d\times Q_{2R}(b))$ and
\begin{align*}
\frac 1 {(\tau R)^{d+2}}\int_{(Q_{\tau R}(z)\times \mathbb{R}^d)\cup (\mathbb{R}^d\times Q_{\tau R}(z+b))}|y-Ax-b|^2 \mathrm d \pi_{s}\leq \tau^{-(d+2)} C_2 \varepsilon_2+\frac{C_2 }{ C\tau^{d+2}}.
\end{align*}

In order to apply the interior large-scale-regularity result \cite[Theorem 15]{GvalaniKoch1}, we now apply a further affine change of coordinates $s'$ to $\pi_s$, $\lambda_s$ and $\mu_s$, obtaining $\pi_{s'}$, $\lambda_{s'}$ and $\mu_{s'}$. Set
$s'= \left(A,b,\frac{\lambda_s(z)^\frac 1 d}{\mu_s(z)^\frac 1 d},\lambda_s(z)^{-1}\right)$. Then $\lambda_s(z)=\mu_s(z')=1$. Decreasing $\varepsilon_2$ and increasing $C$ if necessary, we can further ensure $\#_{\tau R/2}\subset Q'((Q_{\tau R}(z)\times \mathbb{R}^d)\cup (\mathbb{R}^d\times Q_{\tau R}(z+b)))$, where $Q'$ is the affine transformation associated to $s'$. Hence
\begin{align}\label{eq:interiorFinal}
\int_{\#_{\tau R}}|x-y|^2 \mathrm d \pi_{s'}\lesssim \varepsilon_2+\frac 1 C.
\end{align}
Moreover, the $L^\infty$-bound Corollary \ref{cor:Linfty} applied to $\pi_s$ at scale $R$ gives
\begin{align*}
\sup_{(x_s,y_s)\in  \,{\rm supp }\pi_{s}\cap \#_{R/2}} \frac{|x_s-y_s|}{R}\lesssim_\tau \delta\left(\varepsilon_2,\frac 1 C\right),
\end{align*}
where $\delta\left(\varepsilon_2,\frac 1 C\right)\to 0$ as $\varepsilon_2,\frac 1 C \to 0$. Consequently, decreasing $\varepsilon_2$ and increasing $C$ further, if necessary, we are now able to apply \cite[Theorem 15]{GvalaniKoch1} to find $b'\in \mathbb{R}^d$ and $A'\in \mathbb{R}^{d\times d}$ symmetric such that
\begin{align*}
\frac 1 {(CR_c)^{d+2}}\int_{(Q_{C R_c}(0)\times \mathbb{R}^d)\cup (\mathbb{R}^d\times Q_{C R_c}(b'))}|y-A' x-b'|^2 \mathrm d \pi_{s'}\lesssim \varepsilon_2+\frac{1} C\\
\frac{|b'|^2}{R_1^2}+|A'-\textup{id}|^2\lesssim \varepsilon_2+\frac{1} C.
\end{align*}

\medskip\noindent\textbf{Step 3. } Conclusion. Combining Step 1 and Step 2 with a suitably large choice of $C$, we find $R_{fin}=c R_c$ for some $c>0$ such that for some $b_{fin}\in \mathbb{R}^d$ and $A_{fin}\in \mathbb{R}^{d\times d}$ symmetric, and $\tilde s=s$ or $\tilde s=s'$,
\begin{align*}
\frac 1 {R_{fin}^{d+2}}\int_{(Q_{R_{fin}}(0)\times \mathbb{R}^d)\cup (\mathbb{R}^d\times Q_{R_{fin}}(b_{fin}))}|y-A_{fin}x-b_{fin}|^2 \mathrm d \pi_{\tilde s}\lesssim \varepsilon_2\\
\frac{|b_{fin}|^2}{(CR_c)^2}+|A_{fin}-\textup{id}|^2\lesssim \varepsilon_2.
\end{align*}
Reducing $\varepsilon_2$ if necessary, we find,  arguing analogously as for \eqref{eq:interiorFinal},
\begin{align*}
\frac 1{ (\tau R_{fin})^{d+2}} \int_{\#_{\tau R_{fin}}} |y-x|^2 \mathrm d \pi_{\hat s}\lesssim \varepsilon_2,    
\end{align*}
for some change of coordinates $\hat s=(\hat A,\hat b,\hat \gamma,\hat \kappa)$ with
\begin{align*}
    |\hat A|+|\hat A^{-1}|+|\hat \gamma|+|\hat \kappa|\lesssim 1.
\end{align*}
Let $\phi_{\varepsilon,\hat s}, \psi_{\varepsilon,\hat s}$ be the potentials for $\pi_{\hat s}$. 
Proposition \ref{prop:smallScale} shows that 
\begin{align*}
    \|\nabla^2 \phi_{\varepsilon,\hat s}\|_{L^\infty(B_{C R_c/3})}+\|\nabla^2 \psi_{\varepsilon,\hat s}\|_{L^\infty(B_{C R_c/3})}\lesssim 1.
\end{align*}

By Proposition \ref{lem:smallScaleGeometric}, there is $\tilde C>0$ such that for $x\in B_{C R_c/6}$, $y\in B_{C R_c/6}$,
\begin{align*}
B\left(\frac 1 2\nabla \phi_{\varepsilon,\hat s}(x), \tilde C^{-1} R_c\right)\cap \,{\rm supp }\,\mu_{\hat s}\subset \mathcal S_x\subset B\left(\frac 1 2\nabla \phi_{\varepsilon,\hat s}(x),\tilde C R_c\right).
\end{align*}

Returning to original variables and recalling that $p$ was arbitrary, this completes the proof.

\medskip\noindent\textbf{Step 4. }. The case $p=1$. The case $p=1$ proceeds as the case $p>1$ with the exception of obtaining \eqref{eq:topological} at the initial scale $R_1$. In order to apply Proposition \ref{prop:largeScale} at scale $R_1$, it suffices to ensure
\begin{align}\label{eq:largeEntropic}
R_1^{-(d+2)}\int_{\#_{R_1}\cap \{|x-y|\geq R_1\}} |x-y|^2\mathrm d \pi_s  \leq \varepsilon_1.
\end{align}

Denote by $T_s$ the optimal transport map between $\lambda_s$ and $\mu_s$. 
By Proposition \ref{prop:EnergyConvergence} and \eqref{lem:EnergyConvergence},
\begin{align*}
\int |x-T_s^{-1}(y)|^2+|y-T_s(x)|^2 \,d\pi_s\lesssim \varepsilon^2.
\end{align*}
For $(x,y)\in \#_{R_1}$ with $|x-y|\geq R_1$, by triangle inequality, we have if $x\in Q_{R_1}$,
\begin{align*}
|x-y|\leq |x-T_s(x)|+|y-T_s(x)|\leq c R_1^{1+\alpha}+|y-T_s(x)|\leq 2 |y-T_s(x)|.
\end{align*}
To obtain the last inequality, we used that $T_s(0)=0$ and $\nabla T_s(0)=\textup{id}$ in addition to the $C^{1,\alpha}$-regularity of $T_s$. If $y\in Q_{R_1}$, a similar estimate gives
\begin{align*}
|x-y|\leq |y-T_s^{-1}(y)|+|x-T_s^{-1}(y)|\leq  2 |x-T_s^{-1}(y)|.
\end{align*}
Consequently,
\begin{align*}
\int_{\#_{R_1}\cap \{|x-y|\geq R_1\}} |x-y|^2\mathrm d \pi_s  \lesssim \int |x-T_s^{-1}(y)|^2+|y-T_s(x)|^2\mathrm d \pi_s\lesssim \varepsilon^2.
\end{align*}
Reducing $\varepsilon$ if necessary, this gives \eqref{eq:largeEntropic}.
