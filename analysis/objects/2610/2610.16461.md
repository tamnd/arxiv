\thlabel{th:main}
Let $M\subset\mathbb{R}^N$ be a closed $C^{1,1}$ submanifold of dimension $1\leq m\leq N-1$. If $n=N-m$ then
\begin{align}
    \lim_{\rho\to0^+}\rho^{s-n}\operatorname{Per}_s^N(M_\rho)
    =\operatorname{Per}_s^n(B_1^n)\,\text{\usefont{OMS}{cmsy}{m}{n}H} ^m(M).\label{eq:thm-main}
\end{align}
<!-- end thm-1 -->
<!-- thm-2: body_md -->
\thlabel{cor:pointsets}
Let $P\geq2$ be an integer and let $M=\{a_1,\dots,a_P\}$ consist of pairwise distinct points of $\mathbb{R}^N$. Then, for $0<\rho<\frac12\min_{i\neq j}\vert a_i-a_j\vert$,
\begin{align}\label{eq:cor:pointsets}
    \operatorname{Per}_s^N(M_\rho)
    ={}&\operatorname{Per}_s^N(B_1^N)\text{\usefont{OMS}{cmsy}{m}{n}H} ^0(M)\rho^{N-s}\\
    &-\gamma(N,s)\big(\text{\usefont{OMS}{cmsy}{m}{n}H} ^N(B_1^N)\big)^2\rho^{2N}
    \sum_{\substack{1\leq i,j\leq P\\i\neq j}}
    \vert a_i-a_j\vert^{-(N+s)}
    {}_3F_2\!\left(\begin{matrix}
    \frac{N+s}{2},\ 1+\frac{s}{2},\ \frac{N+1}{2}\\
    \frac N2+1,\ N+1
    \end{matrix};\frac{4\rho^2}{\vert a_i-a_j\vert^2}\right).\nonumber
\end{align}
Here ${}_3F_2$ denotes the generalised hypergeometric function.

<!-- prop-1: body_md -->
\thlabel{4gkQqROo} Let \(M\subset \mathbb{R}^N\) be a closed \(C^{1,1}\) submanifold of dimension \(m\) with \(1\leq m \leq N-1\) and let \(n=N-m\). Then there exist functions \(c_i \in L^\infty(N^1M)\) (depending on \(M\)) for \(i=0,\dots,m\) such that for all non-negative measurable functions \(h:\mathbb{R}^N \to \mathbb{R}\) and all \(0<\rho < \operatorname{reach}(M)\), we have
\begin{align*}
    \int_{M_\rho} h(z) \mathop{}\!\mathrm d \text{\usefont{OMS}{cmsy}{m}{n}H} ^N_z &= \sum_{i=0}^m\int_0^\rho \int_M \int_{N_x^1M} h(x+t\nu) t^{n-1+i} c_i(x,\nu) \mathop{}\!\mathrm d \text{\usefont{OMS}{cmsy}{m}{n}H} ^{n-1}_\nu \mathop{}\!\mathrm d \text{\usefont{OMS}{cmsy}{m}{n}H} ^m_x \mathop{}\!\mathrm d t .
\end{align*}
Furthermore, \(c_0=1\).

<!-- prop-2: body_md -->
\thlabel{HmhFO2hM} Let \(M\subset \mathbb{R}^N\) be a closed \(C^{1,1}\) manifold of dimension \(m\) with \(1\leq m\leq N-1\) and let \(n=N-m\). There exist \(R,\delta>0\), depending only on \(M\), such that the following holds. Suppose
\begin{align*}
    0<\rho<\delta, \qquad x\in M, \qquad \nu\in N_x^1M, \qquad \tau\in[0,1], \qquad t\in[1,R/\rho], \qquad j\in\{0,\dots,m\}.
\end{align*}
Define
\begin{align*}
 I_j^{(\rho)}(x,\nu,t,\tau) := \int_M \int_{N_y^1M} \frac{c_j(y,\eta)}{\vert x-y+\rho t\nu-\rho\tau\eta\vert^{N+s}}\mathop{}\!\mathrm d \text{\usefont{OMS}{cmsy}{m}{n}H} ^{n-1}_\eta\mathop{}\!\mathrm d \text{\usefont{OMS}{cmsy}{m}{n}H} ^m_y .
\end{align*}
Then, for \((t,\tau)\neq(1,1)\),
\begin{align}
    \big \vert I_j^{(\rho)}(x,\nu,t,\tau)\big \vert
    \leq C\bigg(\frac{1}{\rho^{n+s}}\int_{N_x^1M}\frac{\mathop{}\!\mathrm d \text{\usefont{OMS}{cmsy}{m}{n}H} ^{n-1}_\eta}{\vert t\nu-\tau\eta\vert^{n+s}}+1\bigg), \label{7qouPfas}
\end{align}

By translating and rotating, we may assume without loss of generality that \(x=0\in M\), that \(T_0M=\mathbb{R}^m\times\{0_n\}\), and that \(\nu=(0,\omega)\) for some \(\omega\in\mathbb{S}^{n-1}\). \end{align*}

Let \(R,\delta>0\) be as in \thref{HmhFO2hM}. After possibly taking them smaller, assume that \(0<\delta<R<\operatorname{reach}(M)\). Then, for all \(0<\rho<\delta\),
\begin{align*}
\gamma(N,s)^{-1}\operatorname{Per}_s^N(M_\rho)
&=\int_{\mathbb{R}^N\setminus M_\rho}\int_{M_\rho}\frac{\mathop{}\!\mathrm d y\mathop{}\!\mathrm d x}{\vert x-y\vert^{N+s}}
=I_1(\rho)+I_2(\rho),
\end{align*}
where
\begin{align*}
I_1(\rho)&=\int_{\mathbb{R}^N\setminus M_R}\int_{M_\rho}\frac{\mathop{}\!\mathrm d y\mathop{}\!\mathrm d x}{\vert x-y\vert^{N+s}},\\
I_2(\rho)&=\int_{M_R\setminus M_\rho}\int_{M_\rho}\frac{\mathop{}\!\mathrm d y\mathop{}\!\mathrm d x}{\vert x-y\vert^{N+s}}.
\end{align*}

For \(I_1(\rho)\), we claim that
\begin{align}
I_1(\rho)\leq C\rho^n \qquad\text{for all }0<\rho<\delta. \label{ybvOyJ8W}
\end{align}
Indeed, \thref{4gkQqROo} with \(h=1\) gives \(\vert M_\rho\vert\leq C\rho^n\). Let \(\widetilde R>0\) be such that \(M\subset B_{\widetilde R/2}\). Then
\begin{align}
    I_1(\rho)
    &=\int_{B_{\widetilde R}\setminus M_R}\int_{M_\rho}\frac{\mathop{}\!\mathrm d y\mathop{}\!\mathrm d x}{\vert x-y\vert^{N+s}}
    +\int_{(\mathbb{R}^N\setminus M_R)\setminus B_{\widetilde R}}\int_{M_\rho}\frac{\mathop{}\!\mathrm d y\mathop{}\!\mathrm d x}{\vert x-y\vert^{N+s}}. \label{tJoywSjD}
\end{align}
Since \(\operatorname{dist}(\mathbb{R}^N\setminus M_R,M_\rho)=R-\rho\), the first integral is bounded by \(C\rho^n\). After taking \(\delta<\widetilde R/4\), we have \(M_\rho\subset B_{3\widetilde R/4}\), and therefore the second integral is also bounded by
\begin{align*}
    C\vert M_\rho\vert\int_{\mathbb{R}^N\setminus B_{\widetilde R}}\frac{\mathop{}\!\mathrm d x}{\vert x\vert^{N+s}}\leq C\rho^n.
\end{align*}
This proves~\eqref{ybvOyJ8W}; in particular, \(\rho^{s-n}I_1(\rho)\to0\).

For \(I_2(\rho)\), by \thref{4gkQqROo},
\begin{align*}
I_2(\rho)
&=\sum_{i,j=0}^m\int_\rho^R\int_M\int_{N_x^1M}\int_0^\rho\int_M\int_{N_y^1M}
\frac{t^{n-1+i}\tau^{n-1+j}c_i(x,\nu)c_j(y,\eta)}{\vert x-y+t\nu-\tau\eta\vert^{N+s}}\\
&\hspace{15em}\mathop{}\!\mathrm d\text{\usefont{OMS}{cmsy}{m}{n}H} ^{n-1}_\eta\mathop{}\!\mathrm d\text{\usefont{OMS}{cmsy}{m}{n}H} ^m_y\mathop{}\!\mathrm d\tau\mathop{}\!\mathrm d\text{\usefont{OMS}{cmsy}{m}{n}H} ^{n-1}_\nu\mathop{}\!\mathrm d\text{\usefont{OMS}{cmsy}{m}{n}H} ^m_x\mathop{}\!\mathrm d t.
\end{align*}
The changes of variables \(t\mapsto\rho t\) and \(\tau\mapsto\rho\tau\) yield
\begin{align*}
I_2(\rho)=\sum_{i,j=0}^m\rho^{2n+i+j}A_{ij}^{(\rho)},
\end{align*}
where
\begin{align*}
A_{ij}^{(\rho)}
&=\int_1^{R/\rho}\int_M\int_{N_x^1M}\int_0^1 t^{n-1+i}\tau^{n-1+j}c_i(x,\nu)I_j^{(\rho)}(x,\nu,t,\tau)\\
&\hspace{16em}\mathop{}\!\mathrm d\tau\mathop{}\!\mathrm d\text{\usefont{OMS}{cmsy}{m}{n}H} ^{n-1}_\nu\mathop{}\!\mathrm d\text{\usefont{OMS}{cmsy}{m}{n}H} ^m_x\mathop{}\!\mathrm d t.
\end{align*}
By~\eqref{7qouPfas} and the boundedness of \(c_i\),
\begin{align*}
\vert A_{ij}^{(\rho)}\vert
&\leq C\int_1^{R/\rho}\int_M\int_{N_x^1M}\int_0^1 t^{n-1+i}\tau^{n-1+j}

\end{align*}

\begin{align*}
&\int_1^{R/\rho}\int_{N_x^1M}\int_0^1\int_{N_x^1M}
\frac{t^{n-1+i}\tau^{n-1+j}}{\rho^{n+s}\vert t\nu-\tau\eta\vert^{n+s}}
\mathop{}\!\mathrm d\text{\usefont{OMS}{cmsy}{m}{n}H} ^{n-1}_\eta\mathop{}\!\mathrm d\tau\mathop{}\!\mathrm d\text{\usefont{OMS}{cmsy}{m}{n}H} ^{n-1}_\nu\mathop{}\!\mathrm d t\\

\int_{\mathbb{R}^n\setminus B_1^n}\int_{B_1^n}\frac{\mathop{}\!\mathrm d\text{\usefont{OMS}{cmsy}{m}{n}H} ^n_y\mathop{}\!\mathrm d\text{\usefont{OMS}{cmsy}{m}{n}H} ^n_x}{\vert x-y\vert^{n+s}}
\leq C\rho^{-n-s-i}.
\end{align*}

\begin{align*}
\int_1^{R/\rho}\int_{N_x^1M}\int_0^1 t^{n-1+i}\tau^{n-1+j}\mathop{}\!\mathrm d\tau\mathop{}\!\mathrm d\text{\usefont{OMS}{cmsy}{m}{n}H} ^{n-1}_\nu\mathop{}\!\mathrm d t\leq C\rho^{-n-i}.
\end{align*}
Therefore,
\begin{align*}
    \vert A_{ij}^{(\rho)}\vert\leq C\rho^{-n-s-i}.
\end{align*}

\begin{align}
\rho^{s-n}I_2(\rho)
=\sum_{i,j=0}^m\rho^{n+s+i+j}A_{ij}^{(\rho)}
=\sum_{i=0}^m\rho^{n+s+i}A_{i0}^{(\rho)}+O(\rho). \label{Li7JuxU0}
\end{align}

Since \(c_0=1\),
\begin{align*}
\rho^{n+s+i}A_{i0}^{(\rho)}
&=\int_1^{+\infty}\int_M\int_{N_x^1M}\int_0^1

&\hspace{17em}\mathop{}\!\mathrm d\tau\mathop{}\!\mathrm d\text{\usefont{OMS}{cmsy}{m}{n}H} ^{n-1}_\nu\mathop{}\!\mathrm d\text{\usefont{OMS}{cmsy}{m}{n}H} ^m_x\mathop{}\!\mathrm d t.
\end{align*}
By~\eqref{7qouPfas}, the absolute value of the integrand is bounded by
\begin{align*}
C\bigg(&R^i t^{n-1}\tau^{n-1}\int_{N_x^1M}\frac{\mathop{}\!\mathrm d\text{\usefont{OMS}{cmsy}{m}{n}H} ^{n-1}_\eta}{\vert t\nu-\tau\eta\vert^{n+s}}

\end{align*}
This upper bound is integrable over
\begin{align*}
\Omega=\{(x,\nu,t,\tau):x\in M,\ \nu\in N_x^1M,\ t\in[1,+\infty),\ \tau\in[0,1]\},
\end{align*}
because its first term is a constant multiple of the fractional perimeter integral of \(B_1^n\) (see the computation below) and its second term is elementary.

Furthermore, by~\eqref{QylvaPee}, for almost every \((x,\nu,t,\tau)\in\Omega\),
\begin{align*}
&\lim_{\rho\to0^+}\rho^{n+s+i}\chi_{[1,R/\rho]}(t)c_i(x,\nu)t^{n-1+i}\tau^{n-1}I_0^{(\rho)}(x,\nu,t,\tau)\\
&\qquad=\frac{\gamma(n,s)}{\gamma(N,s)}\delta_{i0}t^{n-1}\tau^{n-1}

\end{align*}
Therefore, sending \(\rho\to0^+\) in~\eqref{Li7JuxU0} and applying the dominated convergence theorem, we obtain
\begin{align*}
\lim_{\rho\to0^+}\rho^{s-n}I_2(\rho)
&=\frac{\gamma(n,s)}{\gamma(N,s)}\int_M\int_{N_x^1M}\int_1^{+\infty}\int_0^1\int_{N_x^1M}
\frac{t^{n-1}\tau^{n-1}}{\vert t\nu-\tau\eta\vert^{n+s}}\\
&\hspace{16em}\mathop{}\!\mathrm d\text{\usefont{OMS}{cmsy}{m}{n}H} ^{n-1}_\eta\mathop{}\!\mathrm d\tau\mathop{}\!\mathrm d t\mathop{}\!\mathrm d\text{\usefont{OMS}{cmsy}{m}{n}H} ^{n-1}_\nu\mathop{}\!\mathrm d\text{\usefont{OMS}{cmsy}{m}{n}H} ^m_x\\
&=\frac{\gamma(n,s)}{\gamma(N,s)}\text{\usefont{OMS}{cmsy}{m}{n}H} ^m(M)
\int_{\mathbb{R}^n\setminus B_1^n}\int_{B_1^n}\frac{\mathop{}\!\mathrm d\text{\usefont{OMS}{cmsy}{m}{n}H} ^n_y\mathop{}\!\mathrm d\text{\usefont{OMS}{cmsy}{m}{n}H} ^n_x}{\vert x-y\vert^{n+s}}\\
&=\gamma(N,s)^{-1}\operatorname{Per}_s^n(B_1^n)\text{\usefont{OMS}{cmsy}{m}{n}H} ^m(M).
\end{align*}
Together with \(\rho^{s-n}I_1(\rho)\to0\), this proves \thref{th:main}.

Let \(u:\mathbb{R}^N\to \mathbb{R}\) be defined by \(u(x)=x_1\). Then, on \(\partial B_1\), \(\vert \nabla_{\partial B_1}u(x)\vert  = \sqrt{1-x_1^2}\), so the co-area formula on \(\partial B_1\) yields \begin{align*}

        \int_{\partial B_1} h(x_1) \mathop{}\!\mathrm d \text{\usefont{OMS}{cmsy}{m}{n}H} ^{N-1}_x &=  \text{\usefont{OMS}{cmsy}{m}{n}H} ^{N-2} \big (\partial B_1^{N-1}\big )\int_{-1}^1 h(t) \big(1-t^2 \big )^{ \frac{N-3}2   }\mathop{}\!\mathrm d t .
    \end{align*} Finally, the change of variable \(t\to 1-2t\) yields \begin{align*}
        \int_{\partial B_1} h(x_1) \mathop{}\!\mathrm d \text{\usefont{OMS}{cmsy}{m}{n}H} ^{N-1}_x &=  2^{N-2} \text{\usefont{OMS}{cmsy}{m}{n}H} ^{N-2} \big (\partial B_1^{N-1}\big )\int_0^1 h(1-2t) t^{ \frac{N-3}2   }(1-t)^{ \frac{N-3}2   }\mathop{}\!\mathrm d t \\

\end{align*} Finally, we show that \(c_\alpha\) has the form~\eqref{eq:c-alpha}. As at the start of the proof of~\thref{prop:interaction-balls}, we make the change of variable \(y=x-z\) in the inner integral, swap the order of integration, then use polar coordinates to obtain \begin{align*}
    c_\alpha &= \frac1{(2\alpha)!}\int_{B_1}\int_{B_1(x)} z^{2\alpha}\mathop{}\!\mathrm d z \mathop{}\!\mathrm d x \\
    &= \frac1{(2\alpha)!} \int_{B_2} \vert B_1 \cap B_1(z)\vert  z^{2\alpha}\mathop{}\!\mathrm d z \\
    &= \frac1{(2\alpha)!} \bigg ( \int_0^2 r^{N+2\vert \alpha\vert  -1} \vert B_1 \cap B_1(r e_1 )\vert \mathop{}\!\mathrm d r \bigg ) \bigg (  \int_{\partial B_1 }  \omega ^{2\alpha}\mathop{}\!\mathrm d \text{\usefont{OMS}{cmsy}{m}{n}H} ^{N-1}_\omega \bigg ) .
\end{align*} From \cite[Equation 12]{weyl_volume_1939}, we have \begin{align*}
    \int_{\partial B_1 }  \omega ^{2\alpha}\mathop{}\!\mathrm d \text{\usefont{OMS}{cmsy}{m}{n}H} ^{N-1}_\omega  &= \frac{\text{\usefont{OMS}{cmsy}{m}{n}H} ^{N-1}(\partial B_1) \prod_{\alpha_k\neq 0, k=1}^N (2\alpha_k - 1)!! }{(N+2\vert \alpha \vert -2) \cdots (N+2) N}. 
\end{align*} Then, using that \((2m-1)!! = \frac{(2m)!}{2^m m!}\), we have \begin{align*}
    \prod_{\alpha_k\neq 0, k=1}^N (2\alpha_k - 1)!! = \prod_{\alpha_k\neq 0, k=1}^N \frac{(2\alpha_k )!}{2^{\alpha_k} (\alpha_k)!} =\prod_{k=1}^N \frac{(2\alpha_k )!}{2^{\alpha_k} (\alpha_k)!} = \frac{(2\alpha)!}{2^{\vert \alpha \vert }\alpha! }
\end{align*} and \begin{align*}
    (N+2\vert \alpha \vert -2) \cdots (N+2) N = \frac{2^{\vert \alpha \vert } N }{N+ 2 \vert \alpha \vert } \cdot \bigg ( \frac N2 +\vert \alpha \vert \bigg  ) \cdots \bigg ( \frac N2 +1 \bigg ) = \frac{2^{\vert \alpha \vert } N }{N+ 2 \vert \alpha \vert } \bigg ( \frac N 2 +1\bigg )_{\vert \alpha \vert},
\end{align*} so \begin{align*}
    \int_{\partial B_1 }  \omega ^{2\alpha}\mathop{}\!\mathrm d \text{\usefont{OMS}{cmsy}{m}{n}H} ^{N-1}_\omega  &= \frac{ (N+2\vert \alpha \vert) \text{\usefont{OMS}{cmsy}{m}{n}H} ^{N-1}(\partial B_1) (2\alpha)!  }{2^{2\vert \alpha \vert } N  \cdot \big ( \frac N 2 +1\big )_{\vert \alpha \vert}\cdot  \alpha! }= \frac{ (N+2\vert \alpha \vert) \text{\usefont{OMS}{cmsy}{m}{n}H} ^N(B_1) (2\alpha)!  }{2^{2\vert \alpha \vert }   \big ( \frac N 2 +1\big )_{\vert \alpha \vert} \cdot \alpha! } . 
\end{align*} Furthermore, from~\eqref{eq:volume}, we have \begin{align*}
   \int_0^2 r^{N+2 \vert \alpha \vert  -1} \vert B_1 \cap B_1(r e_1 )\vert \mathop{}\!\mathrm d r &= 2 \text{\usefont{OMS}{cmsy}{m}{n}H} ^{N-1}\big ( B_1^{N-1} \big ) \int_0^2 \int_{r/2}^1r^{N+2 \vert \alpha \vert  -1}  (1-t^2)^{\frac{N-1}2} \mathop{}\!\mathrm d t \mathop{}\!\mathrm d r \\
   &=2 \text{\usefont{OMS}{cmsy}{m}{n}H} ^{N-1}\big ( B_1^{N-1} \big ) \int_0^1 \int_0^{2 t }r^{N+2 \vert \alpha \vert  -1}  (1-t^2)^{\frac{N-1}2} \mathop{}\!\mathrm d r  \mathop{}\!\mathrm d t \\
   &= \frac{2^{N+2\vert \alpha \vert +1} \text{\usefont{OMS}{cmsy}{m}{n}H} ^{N-1}\big ( B_1^{N-1} \big ) }{N+2\vert \alpha \vert }\int_0^1 t^{N+2 \vert \alpha \vert }  (1-t^2)^{\frac{N-1}2}  \mathop{}\!\mathrm d t .
\end{align*} Then the change of variable \(t\to t^{1/2}\) gives \begin{align*}
    \int_0^2 r^{N+2 \vert \alpha \vert  -1} \vert B_1 \cap B_1(r e_1 )\vert \mathop{}\!\mathrm d r &= \frac{2^{N+2\vert \alpha \vert } \text{\usefont{OMS}{cmsy}{m}{n}H} ^{N-1}\big ( B_1^{N-1} \big ) }{N+2\vert \alpha \vert }\int_0^1 t^{ \frac{N+2 \vert \alpha \vert-1}2  }  (1-t)^{\frac{N-1}2}  \mathop{}\!\mathrm d t \\
    &= \frac{2^{N+2\vert \alpha \vert } \text{\usefont{OMS}{cmsy}{m}{n}H} ^{N-1}\big ( B_1^{N-1} \big )  \Gamma \big ( \frac{N+2\vert \alpha \vert +1}2 \big )  \Gamma \big ( \frac{N+1}2 \big )  }{(N+2\vert \alpha \vert) \Gamma  (N+\vert \alpha \vert +1  )  }\\
    &= \frac{2^{N+2\vert \alpha \vert } \text{\usefont{OMS}{cmsy}{m}{n}H} ^{N-1}\big ( B_1^{N-1} \big )  \big ( \frac{N+1}2 \big )_{\vert \alpha \vert }  \cdot  \Gamma \big ( \frac{N+1}2 \big )  ^2 }{(N+2\vert \alpha \vert) (N+1)_{\vert \alpha \vert }\cdot  \Gamma  (N +1  )  } 
\end{align*} using the identities \(\int_0^1 t^{a-1}(1-t)^{b-1}\mathop{}\!\mathrm d t = \mathrm B(a,b) = \frac{\Gamma(a)\Gamma(b)}{\Gamma(a+b)}\) and \(\Gamma(x+k)=(x)_k \cdot \Gamma(x)\). Collecting the preceding identities, we obtain \begin{align*}
    c_\alpha &= \frac1{(2\alpha)!} \bigg ( \frac{2^{N+2\vert \alpha \vert } \text{\usefont{OMS}{cmsy}{m}{n}H} ^{N-1}\big ( B_1^{N-1} \big )  \big ( \frac{N+1}2 \big )_{\vert \alpha \vert }  \cdot  \Gamma \big ( \frac{N+1}2 \big )  ^2 }{(N+2\vert \alpha \vert) (N+1)_{\vert \alpha \vert }\cdot  \Gamma  (N +1  )  }  \bigg  )\bigg ( \frac{ (N+2\vert \alpha \vert) \text{\usefont{OMS}{cmsy}{m}{n}H} ^N(B_1) (2\alpha)!  }{2^{2\vert \alpha \vert }   \big ( \frac N 2 +1\big )_{\vert \alpha \vert} \cdot \alpha! }\bigg  )\\
    &=  \frac{2^{N } \text{\usefont{OMS}{cmsy}{m}{n}H} ^{N-1}\big ( B_1^{N-1} \big )\text{\usefont{OMS}{cmsy}{m}{n}H} ^N(B_1)     \cdot  \Gamma \big ( \frac{N+1}2 \big )  ^2 }{   \Gamma  (N +1  )  }  \cdot \frac{\big ( \frac{N+1}2 \big )_{\vert \alpha \vert }  }{ (N+1)_{\vert \alpha \vert } \cdot \big ( \frac N 2 +1\big )_{\vert \alpha \vert} \cdot \alpha! }.
\end{align*} Finally, using~\eqref{eq:Gamma-calc}, we obtain~\eqref{eq:c-alpha}, as desired.
