<!-- assumption-1: body_md -->
\label{ass:population}
Globally on $\mathbb R^{d_x}\times\mathbb R^{d_y}$, $g(x,\cdot)$ is $\mu$-strongly convex, $f(x,\cdot)$ is $L_0$-Lipschitz, and both full gradients are jointly $L_1$-Lipschitz. The blocks $g_{xy},g_{yy}$ exist and are jointly $L_2$-Lipschitz. At the supplied starting pair,
\[
 \Phi(x_0)-\inf\Phi\le\Delta,\qquad \left\|y_0-y^*(x_0)\right\|\le R_0.
\]
<!-- end assumption-1 -->
<!-- definition-1: body_md -->
\label{def:profile}
Fix $M,\Lambda>0$ and $\beta\ge0$, independently of both order and accuracy, and set
\begin{equation}\label{eq:profile}
 L_j=M\Lambda^j(j!)^\beta,\qquad j\ge3.
\end{equation}
For $p\in\mathbb N\cup\{\infty\}$, let $\mathcal F_{p,\beta}$ satisfy Assumption~\ref{ass:population} and~\eqref{eq:high} with~\eqref{eq:profile}. For $p=\infty$, all positive integers $q$ are included. The baseline constant $L_2$ remains separately fixed.
<!-- end definition-1 -->
<!-- assumption-2: body_md -->
\label{ass:oracle}
At an adaptively chosen $(x,y)$, a call selects $a\in\{f,g\}$ and returns a full gradient sample $G_a$ with, conditionally on the past and the query,
\begin{equation}\label{eq:oracle-model}
 \mathbb E G_a=\nabla a(x,y),\qquad
 \mathbb E\left\|G_a-\nabla a(x,y)\right\|^2\le\sigma^2.
\end{equation}
Independent fresh copies are available; a sample is evaluated only once. No function values, Hessians, Hessian--vector products, exact lower solutions, or exact hypergradients are supplied. Individual sample losses need not have uniform smoothness or convexity. Every sample in every batch is charged.
<!-- end assumption-2 -->
<!-- theorem-1: body_md -->
\label{thm:uniform}
Under~\eqref{eq:base-budgets} and Definition~\ref{def:profile}, there exist $c,C,\epsilon_0>0$, depending only on the fixed budgets and $(M,\Lambda,\beta)$, such that \emph{simultaneously} for every $p\in\mathbb N\cup\{\infty\}$ and $0<\epsilon<\epsilon_0$,
\begin{equation}\label{eq:uniform-law}
 c\epsilon^{-4}\mathcal D_{p,\beta}(\epsilon)^2\le Q^*_{p,\beta}(\epsilon)
 \le C\epsilon^{-4}\mathcal D_{p,\beta}(\epsilon)^2.
\end{equation}
The constants and accuracy threshold are independent of $p$ and dimension. The lower bound holds against unrestricted adaptive randomized algorithms, already with $d_y=1$, exact $f$ gradients, and a globally unbiased $g$ oracle that samples the gradients of two fixed smooth losses. Its initial inner error is zero, and
\[
 d_x=O\!\left(\epsilon^{-2}\log(1/\epsilon)\right)
\]
suffices, with an implied constant independent of $p$.
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{thm:fixed}\label{thm:main}
For every fixed $p\ge1$ and every such finite positive list,
\begin{equation}\label{eq:fixed-law}
 Q_p^*(\epsilon;L_3,\ldots,L_{p+1})=\Theta_p(\epsilon^{-4-2/p})
\end{equation}
for all sufficiently small $\epsilon$. Constants and the threshold in this statement may depend on that fixed list and on $p$. The same rate holds if all $f$ gradients are exact.
<!-- end corollary-1 -->
<!-- corollary-2: body_md -->
\label{cor:infty}
Put $t=\log(1/\epsilon)$. For fixed $\beta>0$ and sufficiently small $\epsilon$, uniformly in $p$,
\begin{equation}\label{eq:phase}
 Q^*_{p,\beta}(\epsilon)=
 \begin{cases}
 \Theta_\beta(p^{2\beta}\epsilon^{-4-2/p}),&p\le t/\beta,\\
 \Theta_\beta(\epsilon^{-4}t^{2\beta}),&p\ge t/\beta.
 \end{cases}
\end{equation}
For $\beta=0$, $Q^*_{p,0}(\epsilon)=\Theta(\epsilon^{-4-2/p})$ uniformly over finite $p$, and $Q^*_{\infty,0}(\epsilon)=\Theta(\epsilon^{-4})$. In particular,
\begin{equation}\label{eq:analytic-law}
 Q^*_{\infty,1}(\epsilon)=\Theta\!\left(\epsilon^{-4}\log^2(1/\epsilon)\right).
\end{equation}
All these lower bounds remain valid with exact upper gradients.
<!-- end corollary-2 -->
<!-- lemma-1: body_md -->
\label{lem:cheb}
Let $n\ge1$ be odd, $t_j=\cos(j\pi/n)$ for $j=0,\ldots,n$, and let
$c_0=c_n=2$, $c_j=1$ otherwise. Put $s_n=(-1)^{(n-1)/2}$. For any radius $h>0$ define
\begin{equation}\label{eq:cheb-weights}
 d_j=\frac{s_n(-1)^j}{n h c_j t_j^2}.
\end{equation}
No $t_j$ is zero. For every vector-valued polynomial $P$ of degree at most $n$,
\begin{equation}\label{eq:cheb-identities}
 \sum_{j=0}^n d_jP(ht_j)=P'(0),\qquad
 \sum_{j=0}^n d_j=0,\qquad
 W:=\sum_{j=0}^n|d_j|=\frac nh.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Let $U_{n-1}$ be the polynomial defined by
$U_{n-1}(\cos\theta)=\sin(n\theta)/\sin\theta$, and put
$w(t)=(t^2-1)U_{n-1}(t)$. Its roots are exactly the $t_j$.
For odd $n$, $w$ is even, $w(0)=-s_n$, and
$w'(t_j)=n c_j(-1)^j$. Differentiating the Lagrange cardinal polynomial
$w(t)/((t-t_j)w'(t_j))$ at zero gives \eqref{eq:cheb-weights} after rescaling by $h$. This proves polynomial exactness and the zero-sum identity. Finally, the polynomial $T_n$ defined by $T_n(\cos\theta)=\cos(n\theta)$ has
$T_n(t_j)=(-1)^j$ and $T_n'(0)=n s_n$.
All terms $d_j(-1)^j$ have sign $s_n$, so exactness applied to $T_n(t/h)$ yields $W=n/h$.

<!-- lemma-3: body_md -->
\label{lem:batch}
Suppose a squared-error recursion satisfies
\[
 E_{k+1}\le qE_k+d+\frac{A}{B_k},\qquad 0<q<1,
\]
with $E_0\le D^2$. For a target $u>0$, assume $d/(1-q)\le u^2/4$.
Choose the deterministic length
\[
 J=\max\left\{1,\left\lceil
 \frac{\log(4(D+1)^2/u^2)}{-\log q}\right\rceil\right\},
\]
which has $q^JD^2\le u^2/4$, and let
\begin{equation}\label{eq:geobatches}
 B_k=\max\left\{1,\left\lceil
 \frac{4A}{u^2(1-\sqrt q)}q^{(J-1-k)/2}
 \right\rceil\right\},\qquad 0\le k<J.
\end{equation}
Then $E_J\le3u^2/4$ and
\[
 \sum_{k=0}^{J-1}B_k\le J+\frac{4A}{u^2(1-\sqrt q)^2}.
\]
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Unroll the recursion. The initial term is at most $u^2/4$ and the bias term is at most $d/(1-q)\le u^2/4$.
For the noise term, \eqref{eq:geobatches} implies
\[
 \sum_{k=0}^{J-1}\frac{Aq^{J-1-k}}{B_k}
 \le\frac{u^2(1-\sqrt q)}4\sum_{k=0}^{J-1}q^{(J-1-k)/2}
 \le u^2/4.
\]
Summing the ceiling bounds proves the cost assertion. If $A=0$, take $B_k=1$ and omit the noise term.

<!-- lemma-4: body_md -->
\label{lem:B-bounds}
There are absolute constants $C_1,C_2$ such that, for every $T$ and $z$,
\begin{equation}\label{eq:B-bounds}
 |B_T(z)|\le B_0,\qquad \left\|\nabla B_T(z)\right\|\le C_1,\qquad
 \left\|\nabla^2 B_T(z)\right\|\le C_2.
\end{equation}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
The support of $\ell$ and of each of its derivatives is contained in a region where $a$ is identically zero. Once a nonzero $\ell(z_i)$ occurs, every later prefix vanishes. Thus at most one summand in \eqref{eq:B} is nonzero. For the first two derivatives used below, a zero gate in this region has all derivatives zero. If a summand has a nonzero derivative contribution, its index lies in the support of a derivative of $\ell$; the gate at that index is locally zero and annihilates every later contribution. Thus at most one index contributes to the value, gradient, or Hessian at a given point. It remains to bound the first two derivatives of an arbitrary prefix product $P=\prod_{j=1}^m a(z_j)$ uniformly in $m$.

Put $M_a=\left\|a''\right\|_\infty$. Nonnegativity of $a$ and $1-a$, applied to their quadratic Taylor upper bounds, gives
\[
 |a'(t)|^2\le2M_a\min\{a(t),1-a(t)\}\le4M_a a(t)(1-a(t)).
\]
Write $C_a=4M_a$. The elementary identity/inequality
\begin{equation}\label{eq:exact-one}
 \sum_{j=1}^m(1-a_j)\prod_{k\ne j}a_k\le1,\qquad a_j\in[0,1],
\end{equation}
is the probability of exactly one failure in independent Bernoulli trials with success probabilities $a_j$. It implies
\[
 \left\|\nabla P\right\|^2
 \le C_a\sum_j a_j(1-a_j)\prod_{k\ne j}a_k^2\le C_a.
\]
For positive $a_j$, put $b_j=a'_j/a_j$. The off-diagonal part of $\nabla^2P$ is
$P[bb^\top-\operatorname{diag}(b_j^2)]$. Its operator norm is at most
\[
 2P\sum_j b_j^2\le 2C_a\sum_j(1-a_j)\prod_{k\ne j}a_k\le2C_a.
\]
The diagonal part has norm at most $M_a$. The bounds extend to zero $a_j$ by continuity. Multiplying $P$ by the one possibly active $\ell(z_i)$ gives, for example,
\[
 C_1=B_0\sqrt{C_a}+\left\|\ell'\right\|_\infty,
\quad C_2=B_0(M_a+2C_a)+2\sqrt{C_a}\left\|\ell'\right\|_\infty+\left\|\ell''\right\|_\infty.
\]
This proves the lemma, including at boundaries of the support where all relevant cutoff derivatives vanish.

<!-- lemma-5: body_md -->
\label{lem:chain}
There are absolute constants $C_A,C_G,C_L$, independent of $T$, with
\begin{align}
 |A_T(z)|&\le C_A T,& \left\|\nabla A_T(z)\right\|&\le C_G\sqrt T,&
 \left\|\nabla^2 A_T(z)\right\|&\le C_L.\label{eq:A-bounds}
\end{align}
Moreover:
\begin{enumerate}[label=(\roman*)]
\item If $|z_i|\le\delta$, then $\partial_i A_T(z)=0$, irrespective of the other coordinates.
\item If $k$ is the first index with $|z_k|<1$, then
\begin{equation}\label{eq:frontier-gradient}
 \partial_k H_T(z)\le-1.
\end{equation}
\item For $0\le k\le T$, if $|z_j|\le\delta$ for all $j>k$, the vector $\nabla A_T(z)$ is supported in the first $k$ coordinates and, on those coordinates, depends only on $z_1,\ldots,z_k$. Under the same condition, $B_T(z)$ and $\nabla B_T(z)$ depend only on the first $\min\{T,k+1\}$ coordinates and the gradient is supported there.
\end{enumerate}
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
Each term of $A_T$ involves two adjacent coordinates and fixed bounded scalar functions. Its Hessian is tridiagonal with uniformly bounded entries; the row-sum bound proves \eqref{eq:A-bounds}.
Because $r,r',\Psi,\Psi'$ are nonnegative, every partial derivative of $A_T$ is nonpositive. Its self-link contribution is
\[
 -\Psi(z_{i-1})r'(z_i)-\Psi(-z_{i-1})r'(-z_i)
\]
(with the first term $-r'(z_1)$ at $i=1$), and its next-link contribution is
\[
 -\Psi'(z_i)r(z_{i+1})-\Psi'(-z_i)r(-z_{i+1}).
\]
Both vanish when $|z_i|\le\delta$, proving (i).

For (ii), every preceding gate equals one. The preceding self-link therefore contributes exactly $-r'(z_k)$, since $r'$ is even. Splitting \eqref{eq:B} at $k$ gives
\[
 \partial_k B_T(z)=-\ell'(z_k)-a'(z_k)\mathcal S,
\]
where $\mathcal S$ is a suffix prefix-sum of the same type, and $|\mathcal S|\le B_0$ by the argument for Lemma~\ref{lem:B-bounds}.
When $|z_k|\le1/8$, $\ell'=1$ and $a'=0$, and all contributions from $A_T$ are nonpositive. When $1/8\le|z_k|<1$, use $r'=M_{\rm ch}$ and \eqref{eq:M-choice}. This proves \eqref{eq:frontier-gradient}.

For (iii), the case $k=T$ is immediate. Suppose $k<T$. $r$ is constant and $r'$ is zero on the tail interval, and both $\Psi$ and $\Psi'$ vanish there. The derivative in coordinate $k$ may involve $r(\pm z_{k+1})$, but these are the same fixed constant throughout that interval. For $B_T$, the first tail gate $a(z_{k+1})$ and all its derivatives vanish. All later summands and gradient components are therefore zero. The remaining terms use at most $z_{k+1}$.

<!-- lemma-6: body_md -->
\label{lem:wrapper}
Uniformly in $T,d,U$,
\begin{align}
 |\mathcal B_U|&\le B_0,& \left\|\nabla\mathcal B_U\right\|&\le C_B,&
 \left\|\nabla^2\mathcal B_U\right\|&\le C_{BB},\label{eq:wrappedB}\\
 \left\|\nabla^2\mathcal A_U\right\|&\le C_{AA}.\label{eq:wrappedA}
\end{align}
If some coordinate of $U^\top\rho(w)$ has absolute value below one, then
\begin{equation}\label{eq:wrapped-gradient}
 \left\|\nabla\mathcal H_U(w)\right\|\ge1/2.
\end{equation}
Also $\mathcal H_U(0)-\inf\mathcal H_U\le C_\Delta T$ for a fixed $C_\Delta>0$.
<!-- end lemma-6 -->
<!-- proof-6: body_md -->
Writing $\gamma=(1+\left\|w\right\|^2/\mathcal R^2)^{-1/2}$ gives
\[
 D\rho(w)=\gamma I-\frac{\gamma^3}{\mathcal R^2}ww^\top,
 \qquad \left\|D\rho\right\|\le1,\qquad \left\|D^2\rho\right\|\le C/\mathcal R.
\]
The chain rule, Lemma~\ref{lem:B-bounds}, and \eqref{eq:A-bounds} prove \eqref{eq:wrappedB}--\eqref{eq:wrappedA}; the factor $\sqrt T$ in $\left\|\nabla A_T\right\|$ is canceled by $\mathcal R$.

Let $v=U^\top\rho(w)$ and $q=U\nabla H_T(v)$, so $\left\|q\right\|\le C_H\sqrt T$. Then
\begin{equation}\label{eq:wrapper-grad-formula}
 \nabla\mathcal H_U(w)=\gamma q+
 w\left(\eta_0-\frac{\gamma^3}{\mathcal R^2}\left\langle w,q\right\rangle\right).
\end{equation}
If $\left\|w\right\|\ge\mathcal R/2$, the triangle inequality applied before expanding $D\rho$ gives
\[
 \left\|\nabla\mathcal H_U(w)\right\|\ge\eta_0\left\|w\right\|-\left\|q\right\|
 \ge C_H\sqrt T\ge1.
\]
Otherwise $\gamma\ge2/\sqrt5$. Choose the first $k$ with $|v_k|<1$. By Lemma~\ref{lem:chain}, $q\cdot u_k\le-1$, and
$|w\cdot u_k|=|v_k|/\gamma\le\sqrt5/2$. The absolute value of the parenthesized scalar in \eqref{eq:wrapper-grad-formula} is at most
\[
 \frac1{16}+\frac{(\mathcal R/2)C_H\sqrt T}{\mathcal R^2}
 =\frac9{128}.
\]
Hence the $u_k$ component is at most
$-2/\sqrt5+(\sqrt5/2)(9/128)<-1/2$.
Finally, $A_T$ is bounded by $C_AT$, $B_T$ is bounded, and the quadratic is nonnegative. The gap claim follows, after enlarging $C_\Delta$ to cover every $T\ge1$.

<!-- lemma-7: body_md -->
\label{lem:cap-new}
If $u$ is uniform on the unit sphere in an $m_0$-dimensional Euclidean space and $z$ is fixed with $\left\|z\right\|\le R$, then
\[
 \mathbb P\{|\left\langle u,z\right\rangle|>a\}\le2\exp\left(-\frac{m_0a^2}{2R^2}\right).
\]
<!-- end lemma-7 -->
<!-- proof-7: body_md -->
To compute the spherical moments, write a standard Gaussian vector in polar form $Z=Ru$. Its rotationally invariant density factors into a radial density and the uniform surface measure, so $R$ and $u$ are independent. One-dimensional Gaussian integration by parts gives $\mathbb E Z_1^{2k}=(2k-1)!!$, while integrating the radial density $r^{m_0-1}e^{-r^2/2}$ gives $\mathbb E R^{2k}=m_0(m_0+2)\cdots(m_0+2k-2)$. Dividing and rotating proves
$\mathbb E[\left\langle u,e\right\rangle^{2k}]=(2k-1)!!/[m_0(m_0+2)\cdots(m_0+2k-2)]$
for a unit vector $e$. Odd moments vanish by symmetry. Expanding the exponential therefore gives
$\mathbb E e^{t\left\langle u,z\right\rangle}\le e^{t^2\left\|z\right\|^2/(2m_0)}$.
Exponential Markov bounds for the positive and negative tails, optimized in $t$, prove the claim.

<!-- proposition-1: body_md -->
\label{prop:decoupling}
Under the baseline population assumptions with $L_1=\mu$, one has
\[
 g_{yy}(x,y)=\mu I,\qquad g_{xy}(x,y)=0,\qquad
 g_y(x,y)=\mu y+c
\]
for a vector $c$ independent of $x$ and $y$. In particular, $y^*(x)=-c/\mu$ is constant.
<!-- end proposition-1 -->
<!-- proof-9: body_md -->
For any unit lower direction $v$, differentiating the full-gradient Lipschitz bound in that direction gives
\[
 \left\|(g_{xy}v,g_{yy}v)\right\|\le\mu.
\]
Strong convexity gives $\left\langle v,g_{yy}v\right\rangle\ge\mu$, hence $\left\|g_{yy}v\right\|\ge\mu$. Equality must hold throughout. Consequently $g_{xy}v=0$ and $g_{yy}v=\mu v$ for every unit $v$. The blocks are continuous by assumption, so the identities hold everywhere. Integrating them gives the claimed affine lower gradient.

<!-- proposition-2: body_md -->
\label{prop:boundary-rate}
Fix positive baseline budgets except set $L_1=\mu$, and let $R_0\ge0$. Impose either a fixed finite positive derivative list or the envelope of Definition~\ref{def:profile}, including $p=\infty$. Under the same fresh-gradient oracle and dimension convention, the minimax complexity is $\Theta(\epsilon^{-4})$ for all sufficiently small $\epsilon$.
<!-- end proposition-2 -->
<!-- proof-10: body_md -->
Write $\bar y=-c/\mu$. At $(x_0,y_0)$, take $M$ fresh samples and estimate
\[
 \widehat y=y_0-\frac1{\mu M}\sum_{i=1}^M G_{g,y}(x_0,y_0;i).
\]
Then $\mathbb E\left\|\widehat y-\bar y\right\|^2\le\sigma^2/(\mu^2M)$. Set
$d=\min\{1,\epsilon/(16L_1)\}$ and choose $M=\max\{1,\lceil\sigma^2/(\mu^2d^2)\rceil\}$. For the random but subsequently fixed objective $\widehat h(x)=f(x,\widehat y)$,
\begin{align*}
 \operatorname{Lip}(\nabla\widehat h)&\le L_1,\\
 \mathbb E[\widehat h(x_0)-\inf\widehat h]&\le\Delta+2L_0d=:D_h,\\
 \mathbb E\left\|\nabla\widehat h(x)-\nabla\Phi(x)\right\|&\le L_1d.
\end{align*}
The last estimate holds also at a data-dependent $x$, because its pointwise bound $L_1\left\|\widehat y-\bar y\right\|$ is uniform in $x$. The gap estimate follows from $|f(x,\widehat y)-f(x,\bar y)|\le L_0\left\|\widehat y-\bar y\right\|$, uniformly in $x$.

Run normalized stochastic gradient descent on $\widehat h$ with fresh batches of size $S_h=\max\{1,\lceil256\sigma^2/\epsilon^2\rceil\}$, step length $\eta_h=\epsilon/(4L_1)$, and $K_h=\max\{1,\lceil8D_h/(\eta_h\epsilon)\rceil\}$ steps. Conditional expected gradient-estimation error is at most $\epsilon/16$. The descent inequality proved in Appendix~\ref{app:upper}, now averaged also over $\widehat y$, gives
\[
 \frac1{K_h}\sum_{k<K_h}\mathbb E\left\|\nabla\widehat h(x_k)\right\|
 \le\frac{D_h}{K_h\eta_h}+\frac\epsilon8+\frac{L_1\eta_h}{2}
 \le\frac{3\epsilon}{8}.
\]
Adding the uniform gradient discrepancy of at most $\epsilon/16$ verifies the target for an independently uniform iterate. The fixed cost is $M+K_hS_h=O(\epsilon^{-4})$.

Use the split chain and wrapper of Appendices~\ref{sec:split}--\ref{sec:wrapper}. Choose a fixed $\tau>0$ sufficiently small that $\tau(C_{AA}+C_{BB})\le L_1$. For small $s$, put $\lambda=s/\tau$, $A=s^2/\tau$, and $T=\lfloor\Delta\tau/(2C_\Delta s^2)\rfloor$. Set
