<!-- definition-1: body_md -->
Let $m$ and $n$ be nonnegative integers. The tree $T(2^m1^n)$ is defined as the $m$-level perfect binary tree with a copy of the path $P_n$ dangling from each of its leaves (see \Cref{fig:T}).
<!-- end definition-1 -->
<!-- definition-2: body_md -->
Given $p, q \in E[[x]]$, the series $p$ \emph{is absorbed by} $q$ (or $q$ \emph{absorbs} $p$) if $p+q =q$. In such a case, we write $p \mathrel{\unlhd} q$.
<!-- end definition-2 -->
<!-- proposition-1: body_md -->
\label{p:Aabs} Let $p, q, r, s \in E[[x]]$. Then:
  \begin{enumerate}
  \item\label{p:Aabs1} $p\mathrel{\unlhd} q$ implies $p+r\mathrel{\unlhd} q+r$ and $pr\mathrel{\unlhd} qr$;
  \item\label{p:Aabs1a} $p\mathrel{\unlhd} q$ and $r\mathrel{\unlhd} s$ imply $p+r\mathrel{\unlhd} q+s$ and $pr\mathrel{\unlhd} qs$;
  \item\label{p:Aabs2} if $q=p+r$ then $p\mathrel{\unlhd} q$;
  \item\label{p:Aabs3} if $p\mathrel{\unlhd} q$ and $r\mathrel{\unlhd} q$ then $p+r\mathrel{\unlhd} q$;
    \item\label{p:Aabs4} if $p\mathrel{\unlhd} q$ then $p\mathrel{\unlhd} r+q$.
  \end{enumerate}
<!-- end proposition-1 -->
<!-- definition-3: body_md -->
\label{def:2}
  \begin{enumerate}
  \item The \emph{Jacobsthal sequence} $(J_m)_{m \geq 0}$ is defined by the recurrence
    \[
      J_m = J_{m-1} + 2J_{m-2}, \quad m \geq 2,
    \]
    with the initial conditions $J_0 = 0$ and $J_1 = 1$.
  \item For any integer $m \geq 0$, let $K_m = k^{2^m} x^{J_m} \in E[[x]]$.
  \end{enumerate}
<!-- end definition-3 -->
<!-- definition-4: body_md -->
The parity functions $\operatorname{od} \colon \mathbb{N} \to \{0,1\}$ and $\operatorname{ev} \colon \mathbb{N} \to \{0,1\}$ are defined by
\[
\operatorname{od}(m)=
\begin{cases}
  1, &\text{if $m$ is odd;}\\
  0, &\text{if $m$ is even,}
\end{cases} \quad \text{and} \quad \operatorname{ev}(m)=1-\operatorname{od}(m).
\]
<!-- end definition-4 -->
<!-- definition-5: body_md -->
Let $(p_m)_{m\geq 0}$ be a sequence of polynomials in $E[[x]]$, defined recursively by
  \[
    p_m = K_{m-2}^{2\operatorname{od}(m)}p_{m-1} + K_{m-1}^{\operatorname{ev}(m)}(1+K_{m-2})p_{m-2}, \quad m \geq 2,
  \]
  with initial conditions $p_0=0$ and $p_1=1$.
<!-- end definition-5 -->
<!-- lemma-1: body_md -->
\label{l:triv} 
If $m\geq 2$, then $K_{m-2}^{2\operatorname{od}(m)}p_{m-1}$ is absorbed by $p_m$.
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{l:red} 
For any $m\geq 1$ and $u\geq 0$, $p_u$ is absorbed by $p_m p_u$.
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
Since the constant term of $p_m$ equals $1$, it follows that
  \[
    p_m p_u + p_u = (p_m + 1)p_u = p_m p_u.
  \]
<!-- end proof-1 -->
<!-- lemma-3: body_md -->
\label{l:xp}
  \begin{enumerate}
   \item \label{l:xp0} If $m\geq 2$, then $(1+x)p_m = p_m + K_m/k^4$. 
   \item \label{l:xp1} If $m\geq 3$, then $p_m + x p_{m-1} = p_m$.
  \end{enumerate}
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
\begin{enumerate}
    \item We proceed by induction on $m$. The base cases $m=2$ and $m=3$ follow from straightforward calculations. For $m\geq 4$, the inductive step proceeds as follows: If $m$ is even, the inductive definition of $p_m$, the induction hypothesis, and Eqs.~\eqref{eq:Ks} ensure that
\begin{align*}
&(1+x)p_m=\left(K_{m-1}p_{m-2}+\frac{K_{m-1}}{k^4}\right)\\
&\qquad\qquad\quad+\left(K_{m-1}K_{m-2}p_{m-2}+\frac{K_{m-1}K_{m-2}}{k^4}\right)+p_{m-1}+\frac{K_m}{k^4}\\
&= K_{m-1}p_{m-2}+K_{m-1}K_{m-2}p_{m-2}+p_{m-1}+\frac{K_m}{k^4}=p_m+\frac{K_m}{k^4},
\end{align*}   
   since the constant term of $p_{m-2}$ equals $1$. A similar argument applies when $m$ is odd.
 
\item Again, the proof proceeds by induction on $m$. The base case $m=3$ follows from straightforward calculations. Assume $m\geq 4$. If $m$ is even, the inductive definition of $p_m$ gives
  \begin{align*}
    p_m+&xp_{m-1}=(1+x)p_{m-1}+K_{m-1}(1+K_{m-2})p_{m-2},\\
                &= p_{m-1}+\left(\frac{K_{m-1}}{k^4}+K_{m-1}p_{m-2}\right)\\
                &\qquad\qquad\qquad\qquad +K_{m-1}K_{m-2}p_{m-2}, \quad \text{(by \Cref{l:xp} \Cref{l:xp0})}\\
    &= p_{m-1}+K_{m-1}p_{m-2}+K_{m-1}K_{m-2}p_{m-2}=p_m,
  \end{align*}
  since the constant term of $p_{m-1}$ equals $1$. Therefore, $p_m+x p_{m-1}= p_m$. If $m$ is odd, the recursive definitions of $p_m$ and $p_{m-1}$ yield
  \begin{align*}
    &p_{m}+xp_{m-1}=K_{m-2}^{2}p_{m-1}+K_{m-2}p_{m-2}
                  +(1+x)p_{m-2}\\&\qquad\qquad\qquad\qquad\qquad\qquad\qquad\qquad+xK_{m-2}(1+K_{m-3})p_{m-3}\\
    &= K_{m-2}^{2}p_{m-1}+K_{m-2}p_{m-2}
                   +p_{m-2}\\&\qquad+\frac{K_{m-2}}{k^4}+xK_{m-2}(1+K_{m-3})p_{m-3}. \quad \text{(by \Cref{l:xp} \Cref{l:xp0})}
  \end{align*} 
Since $K_{m-2}p_{m-2}+K_{m-2}/k^4= K_{m-2}p_{m-2}$ (because the constant term of $p_{m-2}$ equals $1$), it follows that
\begin{align*}
p_m+xp_{m-1}= K_{m-2}^2p_{m-1}+p_{m-2}+&K_{m-2}p_{m-2}\\&+xK_{m-2}(1+K_{m-3})p_{m-3},
\end{align*}
where $xK_{m-2}(1+K_{m-3})p_{m-3}$ is absorbed by $K_{m-2}p_{m-2}$ because
\begin{align*}
p_{m-2}+x(1+K_{m-3})p_{m-3}&= (p_{m-2}+xp_{m-3})+K_{m-4}^2p_{m-3}\\
&= p_{m-2}+K_{m-4}^2p_{m-3}= p_{m-2},
\end{align*}
by applying Eqs.~\eqref{eq:Ks} (which implies $xK_{m-3}=K_{m-4}^2$), the induction hypothesis and \Cref{l:triv}. Therefore, $p_m+xp_{m-1}= p_m$.
\end{enumerate}
<!-- end proof-2 -->
<!-- lemma-4: body_md -->
\label{l:pKs}  
   If $m \geq 2$ is even, then
    \[
      p_m+K_mp_m+K_{m-1}^2p_m= p_m+K_mp_m+\frac{K_m^2}{k^4}.
    \]
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
From Eqs.~\eqref{eq:Ks}, $K_{m-1}^2=xK_m$. Thus, \Cref{l:xp} \Cref{l:xp0} leads to
  \begin{align*}
    p_m+K_mp_m+K_{m-1}^2p_m&=p_m+K_mp_m(1+x)= p_m+K_m\left(p_m+\frac{K_m}{k^4}\right).
  \end{align*}
<!-- end proof-3 -->
<!-- lemma-5: body_md -->
\label{l:Kmk4}
  \begin{enumerate}
  \item\label{l:Kmk4i} If $m \geq 4$ is even, then
  \[
    p_m + K_{m-1} p_{m-1} = p_m + \frac{K_m}{k^4}.
  \]
  \item\label{l:Kmk4ii} If $m \geq 5$ is odd, then
  \[
    p_m + K_{m-2}^3 p_{m-2} = p_m + \frac{K_m}{k^4}.
  \]
  \end{enumerate}
<!-- end lemma-5 -->
<!-- proof-4: body_md -->
\begin{enumerate}
  \item 
  The base case $m=4$ follows from straightforward calculations. For $m\geq 6$, the recursive definition of $p_{m-1}$ allows us to write
  \begin{align*}

  \end{multline}
  where nearly every squared term $p_{m-j}^2$ ($j=1,2$) in this expression can be simplified by the induction hypothesis. For instance, the first summand $p_{m-1}^2$ reduces to $p_{m-1}$ because $K_{m-1}p_{m-1}$ can be added to the right-hand side of \Cref{eq:pm2ev}, as it is absorbed by $K_{m-1}p_{m-1}p_{m-2}$ according to \Cref{l:red}. \begin{multline*}
  p_m^2 = p_{m-1} + K_{m-1}(1+K_{m-2})p_{m-1}p_{m-2} \\

\end{multline*}

\begin{multline}\label{eq:pm2evB}
  p_m^2 + K_m p_m = p_m + K_m p_m + K_{m-1}^2 K_{m-2}^2 p_{m-2}^2 \\
  + K_{m-1}^2 K_{m-2} p_{m-2} + K_{m-1}^2 p_{m-2}.
\end{multline}
To obtain the result, it suffices to prove that the last three summands on the right-hand side of \Cref{eq:pm2evB} are absorbed by $K_m p_m$.         \deg(t_{m-1,n}^2) - \deg(x t_{m-2,n}^4) = 1.
      \]
  \end{enumerate}
  Combining these degree relations with the multiplicative and almost-additive properties in \eqref{eq:R} of the reflection operator $R$ completes the proof.

<!-- definition-6: body_md -->
If $p, q \in E[[x]]$ satisfy $\pi_w(p) = \pi_w(q)$, we write
  \[
    p = q \pmod{x^w}.
  \]
<!-- end definition-6 -->
<!-- lemma-16: body_md -->
\label{l:2} 
  Let $w$ be a fixed positive integer. Then,
  \[
    R[I(P_{2k}, x)] = kS \pmod{x^w} \quad \text{and} \quad R[I(P_{2k+1}, x)] = S \pmod{x^w}.
  \]
<!-- end lemma-16 -->
<!-- proof-19: body_md -->
From \cite[Prop.~1(iv)]{Hopkins}\footnote{There is an obvious misprint in this reference: the term $n-k-1$ should be $n-k+1$.}, we obtain
  \[
    R[I(P_{2k}, x)] = \sum_{j=0}^k \tbinom{k+j+1}{2j+1} x^j, \quad R[I(P_{2k+1}, x)] = \sum_{j=0}^{k+1} \tbinom{k+j+1}{2j} x^j,
  \]
  where, for any $0 \leq j < w$, the factorial formula for binomial coefficients implies
  \[
    \tbinom{k+j+1}{2j+1} \in \Theta(k^{2j+1}), \quad \tbinom{k+j+1}{2j} \in \Theta(k^{2j}),
  \]
  from which the result follows.

<!-- theorem-3: body_md -->
\label{thm:main}
  Let $w$ be a fixed positive integer. For every $m \geq 1$ and odd integer $n = 2k+1$, 
  \[
    R[t_{m,n}] = p_m + k^{2^m} x^{J_m} \sum_{j=0}^\infty k^{2j} x^j \pmod{x^w}.
  \]
<!-- end theorem-3 -->
<!-- proof-20: body_md -->
We proceed by induction on $m$. For $m=1$, \Cref{l:2} proves that
  \begin{align*}
    R[t_{1,n}] &= R[I(P_{2k+3}; x)] 
              = S \pmod{x^w} \\
               &= p_1 + K_1 S.
  \end{align*}
  For $m=2$, applying \Cref{thm:rev}, \Cref{l:2,l:SIdem} gives
  \begin{align*}
    R[t_{2,n}] &= R[t_{1,n}]^2 + x R[t_{0,n}]^4 
               = S + \sum_{j=0}^{\infty} k^{2j+4} x^{j+1} \pmod{x^w} \\
               &= 1 + \sum_{j=1}^{\infty} k^{2j+2} x^j 
               = p_2 + K_2 \sum_{u=0}^\infty k^{2u} x^u.
  \end{align*}
Now assume $m \geq 2$. If $m$ is even, then the induction hypothesis, \Cref{thm:rev}, and \Cref{l:4} \Cref{l:4iii} ensure that
  \begin{align*}
    R[t_{m+1,n}] &= R[t_{m-1,n}]^4 + x R[t_{m,n}]^2 \\
                 &= (p_{m-1} + K_{m-1} S)^4 + x(p_m + K_m S)^2 \pmod{x^w} \\
                 &= \sum_{j=0}^3 K_{m-1}^j p_{m-1}^{4-j} + K_{m-1}^4 S + x p_m^2 + x K_m p_m + K_{m+1} S,
  \end{align*}
  where $K_{m-1}^4 S \mathrel{\unlhd} K_{m+1} S$ since $K_{m-1}^4 = x K_{m+1}$ and $xS \mathrel{\unlhd} S$.
Thus,
  \begin{align*}
    R[t_{m+1,n}] &= x \sum_{j=0}^1 K_m^j p_m^{2-j} + \sum_{j=0}^3 K_{m-1}^j p_{m-1}^{4-j} + K_{m+1} S \pmod{x^w}\\
                 &= x \sum_{j=0}^1 K_m^j p_m + \sum_{j=0}^3 K_{m-1}^j p_{m-1} + K_{m+1} S \quad {\text{(by \Cref{c:NoPow})}} \\
                 &= p_{m+1} + \frac{K_{m+1}}{k^4} + K_{m+1} S \quad{\text{(by \Cref{l:Alst})} }\\
                 &= p_{m+1} + K_{m+1} S.
  \end{align*}
  The case where $m$ is odd is handled analogously.
<!-- end proof-20 -->
<!-- lemma-17: body_md -->
\label{l:degp_m}
  For all $m \geq 1$, $\deg p_m = J_m - 1$.
<!-- end lemma-17 -->
<!-- proof-21: body_md -->
Since $p_1 = p_2 = 1$ and $J_1 - 1 = J_2 - 1 = 0$, the result holds for the base cases $m=1, 2$. Assume $m \geq 2$. If $m+1$ is even, then
  \[
    p_{m+1} = p_m + K_m p_{m-1} + K_m K_{m-1} p_{m-1},
  \]
  and applying the induction hypothesis yields
  \[
    \deg p_{m+1} = \max \{ J_m - 1, J_m + J_{m-1} - 1, \underbrace{J_m + 2J_{m-1}}_{J_{m+1}} - 1 \} = J_{m+1} - 1.
  \]
  The odd case follows symmetrically.
<!-- end proof-21 -->
<!-- lemma-18: body_md -->
\label{l:breaks}
  For all $m \geq 3$, $p_m$ has at least $J_{m-3}$ asymptotic breaks in log-concavity.
<!-- end lemma-18 -->
<!-- proof-22: body_md -->
The base cases hold directly: $p_3 = 1 + k^2 x + k^4 x^2$ has $J_0 = 0$ asymptotic breaks in log-concavity, and $p_4 = 1 + k^2 x + k^4 x^2 + k^8 x^3 + k^{12} x^4$ breaks log-concavity at index $2$, giving $J_1 = 1$ break. For odd $m > 4$, we have $p_m = p_{m-2} + K_{m-2} p_{m-2} + K_{m-2}^2 p_{m-1}$. By \Cref{l:degp_m}, these three summands have disjoint degree ranges ($0$ to $J_{m-2}-1$, $J_{m-2}$ to $2J_{m-2}-1$, and $2J_{m-2}$ to $J_m-1$, respectively), so their breaks do not interfere. Therefore, by the induction hypothesis, $p_m$ inherits at least $2J_{m-5} + J_{m-4} = J_{m-3}$ asymptotic breaks in log-concavity. The case where $m$ is even is analogous.
<!-- end proof-22 -->
<!-- corollary-3: body_md -->
\label{c:final}
  For any $m \geq 3$ and sufficiently large odd integer $n$, the independence polynomial of the spherically symmetric tree $T(2^m 1^n)$ has at least $J_{m-3}$ breaks in log-concavity.
<!-- end corollary-3 -->
<!-- proof-23: body_md -->
Let $R[t_{m,n}] = \sum_{j=0}^\infty c_j x^j$ and $p_m = \sum_{j=0}^{J_m-1} a_j x^j$, where $c_j, a_j \in \mathbb{N}[k]$. Applying \Cref{thm:main} with $w \geq J_m$ yields $c_j \in \Theta(a_j)$ for $j = 0, \dots, J_m - 1$. Consequently, every asymptotic break in log-concavity of $p_m$ guaranteed by \Cref{l:breaks} yields a break in log-concavity of $R[t_{m,n}]$ for sufficiently large $k$, since \Cref{eq:asympBrk} implies $c_{u-1} c_{u+1} - c_u^2 \in \Theta(a_{u-1} a_{u+1})$.
<!-- end proof-23 -->
