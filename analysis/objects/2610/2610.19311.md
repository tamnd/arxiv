<!-- lthm-1: body_md -->
\label{thm:intro-main}
Let $G$ be a nontrivial finite abelian group and let $K/\mathbb{Q}$ be a tamely ramified Galois extension with $\operatorname{Gal}(K/\mathbb{Q})\simeq G$. Then the shape of $K$ is the similarity class of the fixed lattice $L_G$ equipped with the metric
\[
        \bigoplus_{\omega\in\Omega(G)} f_\omega(K)Q_\omega.
\]
In particular, the shape of $K$ is determined by the conductors of the nontrivial rational characters of $G$.
<!-- end lthm-1 -->
<!-- lthm-2: body_md -->
\label{thm:intro-distribution}
Let $\ell$ be the smallest prime divisor of $|G|$, and put $a_0=|G|(1-1/\ell)$ and $\alpha_G=1/a_0$. Let $\mathscr F_G^{\rm mark}(X)$ be the marked tame $G$-extensions of absolute discriminant at most $X$. Then
\[
       |\mathscr F_G^{\rm mark}(X)|\sim C_G^{\rm mark}X^{\alpha_G}(\log X)^{s_G-1}
\]
for an explicit constant $C_G^{\rm mark}>0$. Moreover, the rescaled logarithmic parameters $a_0\lambda(K,\iota)/\log X$ become uniformly distributed on the simplex $\mathscr P_G$.

\label{def:shape-number-field}

denoted by $\operatorname{sh}(K)$, is the similarity class of the projected Minkowski lattice

<!-- lemma-1: body_md -->
\label{lem:character-lines-orthogonal}
Let $B$ be a $G$-invariant Hermitian form on $\mathbb{C}[G]$. Then the complex character lines $\mathbb{C} e_\chi$ and $\mathbb{C} e_\psi$ are orthogonal whenever $\chi\neq\psi$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Consider nonzero vectors $v_\chi\in \mathbb{C} e_\chi$ and $v_\psi\in \mathbb{C} e_\psi$. For every $g\in G$, we have that $gv_\chi=\chi(g)v_\chi$ and $gv_\psi=\psi(g)v_\psi$.
We find that
\begin{align*}
        B(v_\chi,v_\psi)
        &=B(gv_\chi,gv_\psi)\\
        &=B(\chi(g)v_\chi,\psi(g)v_\psi)\\
        &=\chi(g)\overline{\psi(g)}B(v_\chi,v_\psi).
\end{align*}
Since $\chi\neq\psi$, there is a $g\in G$ for which $\chi(g)\overline{\psi(g)}\neq1$. The displayed identity then gives $B(v_\chi,v_\psi)=0$.

<!-- proposition-1: body_md -->
\label{prop:hilbert-speiser}
With this notation, the following assertions hold:
\begin{enumerate}
    \item $K/\mathbb{Q}$ is tamely ramified if and only if $N$ is odd and squarefree.
    \item If the equivalent conditions in (1) hold, the Gaussian period
\begin{equation}\label{eq:alphaK}
        \alpha_K
        :=
        \operatorname{Tr}_{M/K}(\zeta_N)
        =
        \sum_{h\in H}h(\zeta_N)
\end{equation}
is a normal integral basis generator for $K/\mathbb{Q}$.
\item Under the same assumption,
\begin{equation}\label{eq:trace-alpha}
        \operatorname{Tr}_{K/\mathbb{Q}}(\alpha_K)=\mu(N)\in\{\pm1\}.
\end{equation}
\end{enumerate}
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
Assertions (1) and (2) are well known, see
\cite[Theorem~8.4]{JohnstonGaloisModules}. We find that
\begin{align*}
        \operatorname{Tr}_{K/\mathbb{Q}}(\alpha_K)
        &=
        \operatorname{Tr}_{K/\mathbb{Q}}\left(\operatorname{Tr}_{M/K}(\zeta_N)\right)\\
        &=
        \operatorname{Tr}_{M/\mathbb{Q}}(\zeta_N)=\mu(N);
\end{align*}
cf. \cite[\S8.3]{ApostolANT}.  Under the conditions in (1), $N$ is squarefree and $\mu(N)=\pm1$.
This proves \eqref{eq:trace-alpha}.

<!-- lemma-2: body_md -->
\label{lem:imprimitive-gauss}
Let $\chi$ be a Dirichlet character modulo a squarefree integer $N$, and
suppose that $\chi$ is induced by a primitive character $\chi^*$ of conductor
$f$.  Writing $N=fm$, note that $(f,m)=1$. Then we find that
\begin{equation}\label{eq:imprimitive-gauss}
        \tau_N(\chi)
        =
        \mu(m)\chi^*(m)\tau_f(\chi^*).
\end{equation}
In particular,
\begin{equation}\label{eq:imprimitive-gauss-abs}
        |\tau_N(\chi)|^2=f.
\end{equation}
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
Since $N$ is squarefree and $f\mid N$, the integers $f$ and
$m=N/f$ are coprime.  Choose integers $\overline m$ and $\overline f$
satisfying
\[
        m\overline m\equiv1\pmod f,
        \qquad
        f\overline f\equiv1\pmod m.
\]
By the Chinese remainder theorem, a residue class $a$ modulo $N=fm$ is
determined by the pair
\[
        (u,v)=(a\bmod f,a\bmod m),
\]
and one may choose the representative
\[
        a\equiv
        um\overline m+vf\overline f
        \pmod{fm}.
\]
Consequently,
\begin{align*}
        e^{2\pi ia/N}
        &=
        e^{2\pi i u\overline m/f}
        e^{2\pi i v\overline f/m}\\
        &=
        \zeta_f^{u\overline m}
        \zeta_m^{v\overline f}.
\end{align*}

If $a$ is a unit modulo $N$, then both $u$ and $v$ are units modulo
$f$ and $m$, respectively.  Since $\chi$ is induced from $\chi^*$, we have
\[
        \chi(a)=\chi^*(u).
\]
It follows that
\begin{align}
        \tau_N(\chi)
        &=
        \sum_{a\in(\mathbb{Z}/N\mathbb{Z})^\times}
        \chi(a)\zeta_N^a\notag\\
        &=
        \left(
        \sum_{u\in(\mathbb{Z}/f\mathbb{Z})^\times}
        \chi^*(u)\zeta_f^{u\overline m}
        \right)
        \left(
        \sum_{v\in(\mathbb{Z}/m\mathbb{Z})^\times}
        \zeta_m^{v\overline f}
        \right).
        \label{eq:gauss-factorization}
\end{align}

We evaluate the two factors separately.  Since $\overline m$ is a unit
modulo $f$, multiplication by $\overline m$ permutes the reduced residue
classes modulo $f$.  The standard twisting identity for Gauss sums therefore
gives
\begin{align*}
        \sum_{u\in(\mathbb{Z}/f\mathbb{Z})^\times}
        \chi^*(u)\zeta_f^{u\overline m}
        &=
        \overline{\chi^*}(\overline m)\tau_f(\chi^*).
\end{align*}
Since
\[
        m\overline m\equiv1\pmod f,
\]
we have
\[
        \chi^*(\overline m)=\chi^*(m)^{-1}.
\]
Character values have absolute value one, and hence
\[
        \overline{\chi^*}(\overline m)
        =
        \chi^*(m).
\]

For the second factor in \eqref{eq:gauss-factorization}, multiplication by
the unit $\overline f$ permutes $(\mathbb{Z}/m\mathbb{Z})^\times$.  Therefore
\begin{align*}
        \sum_{v\in(\mathbb{Z}/m\mathbb{Z})^\times}
        \zeta_m^{v\overline f}
        &=
        \sum_{v\in(\mathbb{Z}/m\mathbb{Z})^\times}\zeta_m^v\\
        &=
        c_m(1)
        =
        \mu(m),
\end{align*}
where the last identity is the Ramanujan-sum evaluation recalled above.
Combining the two factors gives
\[
        \tau_N(\chi)
        =
        \mu(m)\chi^*(m)\tau_f(\chi^*),
\]
which proves \eqref{eq:imprimitive-gauss}.

For completeness, we also recall why a primitive Gauss sum has absolute
value $\sqrt f$.  Let $\eta$ be a primitive Dirichlet character modulo $f$
and, for $n\in\mathbb{Z}$, put
\[
        S(n)
        :=
        \sum_{a\bmod f}
        \eta(a)e^{2\pi ian/f}.
\]
If $(n,f)=1$, then multiplication by $n$ permutes the reduced residue
classes modulo $f$.  Making the substitution $a\mapsto n^{-1}a$ gives
\[
        S(n)
        =
        \overline{\eta(n)}\tau_f(\eta).
\]
If $(n,f)>1$, then $S(n)=0$.  Indeed, put $d=(n,f)$.  Since $\eta$ is
primitive, it cannot be trivial on the kernel of the reduction map from
$(\mathbb{Z}/f\mathbb{Z})^\times$ to $(\mathbb{Z}/(f/d)\mathbb{Z})^\times$.  Hence there exists a unit
$b\bmod f$ satisfying
\[
        b\equiv1\pmod{f/d},
        \qquad
        \eta(b)\neq1.
\]
The first congruence implies
\[
        bn\equiv n\pmod f.
\]
Replacing $a$ by $ba$ in the definition of $S(n)$ therefore gives
\[
        S(n)=\overline{\eta(b)}S(n),
\]
and since $\eta(b)\neq1$, it follows that $S(n)=0$.

Now apply Parseval's identity to the finite Fourier transform of the function
$a\mapsto\eta(a)$ on $\mathbb{Z}/f\mathbb{Z}$.  We obtain
\[
        \sum_{n\bmod f}|S(n)|^2
        =
        f\sum_{a\bmod f}|\eta(a)|^2
        =
        f\varphi(f).
\]
There are exactly $\varphi(f)$ residue classes $n\bmod f$ with
$(n,f)=1$, and on each of them
\[
        |S(n)|=|\tau_f(\eta)|.
\]
Consequently
\[
        \varphi(f)|\tau_f(\eta)|^2
        =
        f\varphi(f),
\]
and therefore
\begin{equation}\label{eq:primitive-gauss-absolute}
        |\tau_f(\eta)|^2=f.
\end{equation}
This is the classical absolute-value formula for primitive Gauss sums; see
also \cite[Theorem~8.15]{ApostolANT}.

Applying \eqref{eq:primitive-gauss-absolute} to $\eta=\chi^*$, and using
\[
        |\mu(m)|=|\chi^*(m)|=1,
\]
gives \eqref{eq:imprimitive-gauss-abs}.

<!-- theorem-1: body_md -->
\label{thm:main}
Let $G$ be a nontrivial finite abelian group of order $n$, and let $K/\mathbb{Q}$ be a
tamely ramified $G$-extension of conductor $N$.  Put
\[
        \alpha_K
        =
        \operatorname{Tr}_{\mathbb{Q}(\zeta_N)/K}(\zeta_N).
\]
Then the following statements hold.
\begin{enumerate}
\item
The conjugates of $\alpha_K$ form a normal integral basis of $\mathcal{O}_K$.

\item
Under the resulting identification
\[
        \iota_K:\mathbb{Z}[G]\xrightarrow{\sim}\mathcal{O}_K,
\]
the scaled projected lattice $\mathcal{O}_K^\perp$ is identified with the fixed
lattice $L_G$.

\item
With respect to the rational decomposition
\[
        \ker(\varepsilon)\otimes_\mathbb{Z}\mathbb{R}
        =
        \bigoplus_{\omega\in\Omega(G)}
        V_\omega\otimes_\mathbb{Q}\mathbb{R},
\]
the pulled-back Minkowski form is
\begin{equation}\label{eq:BK-block-decomposition}
        B_K|_{\ker(\varepsilon)\otimes\mathbb{R}}
        =
        \bigoplus_{\omega\in\Omega(G)}
        f_\omega(K)Q_\omega.
\end{equation}
Consequently,
\begin{equation}\label{eq:shape-main}
        \operatorname{sh}(K)
        =
        \operatorname{sh}\left(
        L_G,
        \bigoplus_{\omega\in\Omega(G)}
        f_\omega(K)Q_\omega
        \right).
\end{equation}
This is to say that when $\mathcal{O}_K^\perp$ is identified with $L_G$ the shape is given by the form
\[ \bigoplus_{\omega\in\Omega(G)}
        f_\omega(K)Q_\omega.\]
\end{enumerate}
<!-- end theorem-1 -->
<!-- proof-4: body_md -->
The first assertion is Proposition~\ref{prop:hilbert-speiser}. Choose a marking $\iota:\operatorname{Gal}(K/\mathbb{Q})\xrightarrow{\sim}G$.
Together with the normal integral basis generator $\alpha_K$, this gives the
$\mathbb{Z}[G]$-module isomorphism $\iota_K:\mathbb{Z}[G]\xrightarrow{\sim}\mathcal{O}_K$ defined by $a\mapsto a\alpha_K$. By Proposition~\ref{prop:hilbert-speiser},
\[
        \operatorname{Tr}_{K/\mathbb{Q}}(\alpha_K)=\mu(N),
\]
and the calculation in \eqref{eq:trace-iota}--\eqref{eq:projected-group-ring}
shows that
\[
        n\iota_K(a)
        -
        \operatorname{Tr}_{K/\mathbb{Q}}\bigl(\iota_K(a)\bigr)
        =
        \iota_K\bigl(na-\varepsilon(a)N_G\bigr)
\]
and hence $\mathcal{O}_K^\perp=\iota_K(L_G)$ which proves the second assertion.

Equations \eqref{eq:B0-vchi} and \eqref{eq:BK-vchi} give $B_0(v_\chi,v_\chi)=n$ and $B_K(v_\chi,v_\chi)=nf_\chi(K)$. The character lines are pairwise orthogonal for both $B_0$ and $B_K$. The conductor $f_\chi(K)$ is constant on a Galois orbit
$\omega\subseteq\widehat G$. Therefore we obtain
\[
        B_K|_{\ker(\varepsilon)\otimes_\mathbb{Z}\mathbb{R}}
        =
        \bigoplus_{\omega\in\Omega(G)}
        f_\omega(K)Q_\omega
\]
and this proves the third assertion. The shape of $K$ is the similarity class of $\mathcal{O}_K^\perp$ equipped with the metric $B_K$.  Identifying $\mathcal{O}_K^\perp$ with $L_G$ gives
\eqref{eq:shape-main}.

<!-- proposition-2: body_md -->
\label{prop:conductor-inertia}
For $\omega\in\Omega(G)$ and $\chi\in\omega$,
\begin{equation}\label{eq:conductor-inertia}
       f_\omega(K)=
       \prod_{\substack{C\in\mathcal{C}(G)\\\chi|_C\neq1}}q_C(K).
\end{equation}
Equivalently, if $x_C=\log q_C(K)$ and $y_\omega=\log f_\omega(K)$, then $y=A_Gx$.
<!-- end proposition-2 -->
<!-- proof-5: body_md -->
Fix a finite prime $p$.  Since $K/\mathbb{Q}$ is tame, the wild inertia group at $p$ is trivial.  For a one-dimensional character $\chi$ of $G$, the local Artin conductor exponent is therefore
\[
        a_p(\chi)=\dim(\mathbb{C})-\dim(\mathbb{C}^{I_p})
        =\begin{cases}
        0,&\chi|_{I_p}=1,\\
        1,&\chi|_{I_p}\neq1.
        \end{cases}
\]
This can also be read directly in Dirichlet-character language: a tame local character is either unramified, in which case $p$ does not divide its conductor, or nontrivial on tame inertia, in which case its conductor contains $p$ to the first power and no higher power.

It follows that the primitive conductor of $\chi$ is
\begin{equation}\label{eq:fchi-local-product}
       f_\chi=
       \prod_{\substack{p\text{ ramified in }K\\\chi|_{I_p}\neq1}}p.
\end{equation}
Now fix a nontrivial cyclic subgroup $C\leq G$.  By definition,
\[
        q_C(K)=\prod_{I_p=C}p.
\]
The primes in \eqref{eq:fchi-local-product} are partitioned by their inertia subgroups.  Grouping them according to this partition gives
\[
       f_\chi=
       \prod_{\substack{C\in\mathcal{C}(G)\\\chi|_C\neq1}}q_C(K),
\]
which is \eqref{eq:conductor-inertia}.

The condition $\chi|_C\neq1$ is invariant under Galois conjugation of $\chi$.  Indeed, if $\chi'$ is a Galois conjugate of $\chi$, then $\chi'=\chi^a$ for some integer $a$ relatively prime to the order of $\chi$.  If $\chi(c)$ has order $e$, then $(a,e)=1$, so $\chi(c)^a=1$ if and only if $\chi(c)=1$.  Hence the product depends only on the orbit $\omega$, as claimed.  Finally, taking logarithms of the product gives
\[
        \log f_\omega(K)=
        \sum_{C\in\mathcal{C}(G)}a_{\omega,C}\log q_C(K),
\]
which is exactly the matrix relation $y=A_Gx$.

<!-- theorem-2: body_md -->
\label{thm:incidence}
Let $G$ be a nontrivial finite abelian group, and put
$m_G=|\mathcal{C}(G)|=|\Omega(G)|$. Then
\[
        A_G\in\mathrm{GL}_{m_G}(\mathbb{Q}).
\]
In particular, for every tamely ramified $G$-extension $K/\mathbb{Q}$
with a fixed marking $\iota:\operatorname{Gal}(K/\mathbb{Q})\xrightarrow{\sim}G$,
the vectors
\[
        x(K,\iota)
        :=\bigl(\log q_C(K,\iota)\bigr)_{C\in\mathcal{C}(G)}\quad \text{and}\quad
        y(K,\iota)
        :=\bigl(\log f_\omega(K,\iota)\bigr)_{\omega\in\Omega(G)}
\]
satisfy $y(K,\iota)=A_Gx(K,\iota)$. Moreover,
\[
        \operatorname{sh}(K)
        =\Phi_G\bigl([A_Gx(K,\iota)]\bigr).
\]
Consequently,
\[
        \bigl\{
          \operatorname{sh}(K):
          K/\mathbb{Q} \text{ is tamely ramified and }
          \operatorname{Gal}(K/\mathbb{Q})\simeq G
        \bigr\}
        \subseteq \Phi_G(E_G).
\]
<!-- end theorem-2 -->
<!-- proof-6: body_md -->
We first prove that $A_G$ is nonsingular. Include the trivial
character and the trivial subgroup by setting
\[
        \Omega^+(G)=\Omega(G)\cup\{1\}\quad \text{and}\quad
        \mathcal{C}^+(G)=\mathcal{C}(G)\cup\{1\}.
\]
Here the additional element of $\Omega^+(G)$ denotes the orbit
consisting of the trivial character. Define
\[
        B^+=(b_{\omega,C})_{
          \omega\in\Omega^+(G),\,C\in\mathcal{C}^+(G)},
      \quad \text{where}\quad
        b_{\omega,C}
        =
        \begin{cases}
        1,&\chi|_C=1,\\
        0,&\chi|_C\neq1,
        \end{cases}
\]
where $\chi$ is any member of $\omega$.

It is easy to see that this definition does not depend on the choice of
$\chi$. Indeed, if $\chi$ has order $d$, then every character in its Galois
orbit has the form $\chi^a$ with $(a,d)=1$. For every $g\in G$,
the value $\chi(g)$ is a $d$th root of unity, and therefore
\[
        \chi(g)^a=1
        \quad\Longleftrightarrow\quad
        \chi(g)=1.
\]
Thus $\chi$ and $\chi^a$ have the same kernel. In particular,
their restrictions to $C$ are either both trivial or both
nontrivial.

We shall also use the equality
\[
        |\Omega^+(G)|=|\mathcal{C}^+(G)|=m_G+1.
\]
Indeed, the Galois orbit of a character $\chi$ consists precisely
of the generators of the cyclic subgroup $\langle\chi\rangle$
of $\widehat G$. Consequently, $\Omega^+(G)$ is in bijection
with the cyclic subgroups of $\widehat G$, including the trivial
subgroup. To compare their number with that for $G$, write
\[
        G\simeq C_{d_1}\times\cdots\times C_{d_r}.
\]
The character group of $C_{d_i}$ is cyclic of order $d_i$, so
$\widehat G$ has the same invariant factors as $G$. Hence $G$
and $\widehat G$ are isomorphic, although no particular
isomorphism is needed here. They therefore have the same number
of cyclic subgroups. In particular, $B^+$ is square.

For $\omega\in\Omega^+(G)$, let $V_\omega$ be the corresponding
irreducible rational representation, and write
\[
        \theta_\omega=\sum_{\chi\in\omega}\chi
\]
for its character. Since $G$ is abelian, its complex irreducible
representations are one-dimensional, and
\[
        V_\omega\otimes_\mathbb{Q}\mathbb{C}
        \simeq
        \bigoplus_{\chi\in\omega}\mathbb{C} e_\chi,
        \qquad
        g e_\chi=\chi(g)e_\chi.
\]
In particular,
\[
        \dim_\mathbb{Q} V_\omega=|\omega|.
\]
One can see this decomposition explicitly by choosing
$\chi\in\omega$ of order $d$: the representation $V_\omega$
is realized on $\mathbb{Q}(\zeta_d)$, with $g\in G$ acting by
multiplication by $\chi(g)$. The embeddings of $\mathbb{Q}(\zeta_d)$
into $\mathbb{C}$ give the character lines indexed by $\omega$.

Taking $C$-invariants commutes with extension of scalars from
$\mathbb{Q}$ to $\mathbb{C}$. Indeed, the averaging operator
\[
        \frac{1}{|C|}\sum_{c\in C}c
\]
is a projection onto the invariant subspace over either field.
If $b_{\omega,C}=1$, then every character in $\omega$ is trivial
on $C$, and every summand $\mathbb{C} e_\chi$ is fixed by $C$.
If $b_{\omega,C}=0$, none of these character lines has a
nonzero $C$-fixed vector. Consequently,
\begin{equation}\label{eq:fixed-dim-incidence}
        \dim_\mathbb{Q}(V_\omega^C)
        =(\dim_\mathbb{Q} V_\omega)b_{\omega,C}.
\end{equation}

For each $C\in\mathcal{C}^+(G)$, let
\[
        \rho_C:=\operatorname{Ind}_C^G(1)
\]
be the character of the rational permutation representation
$\mathbb{Q}[G/C]$. Frobenius reciprocity gives, for every
$\chi\in\widehat G$,
\[
        \langle\rho_C,\chi\rangle_G
        =
        \langle 1,\chi|_C\rangle_C
        =
        \frac{1}{|C|}
        \sum_{c\in C}\overline{\chi(c)}
        =
        \begin{cases}
        1,&\chi|_C=1,\\
        0,&\chi|_C\neq1.
        \end{cases}
\]
The last equality follows from the character sum over $C$:
the sum is $|C|$ for the trivial character and zero for a
nontrivial character. Thus
\[
        \rho_C
        =
        \sum_{\substack{\chi\in\widehat G\\\chi|_C=1}}\chi
        =
        \sum_{\omega\in\Omega^+(G)}
        b_{\omega,C}\theta_\omega.
\]
It follows that the column of $B^+$ indexed by $C$ is the
coefficient vector of $\rho_C$ in the basis of irreducible
rational characters.

For completeness, these coefficients are compatible with
\eqref{eq:fixed-dim-incidence} as follows. Orthogonality of
the complex irreducible characters gives
\[
        \langle\theta_\omega,\theta_{\omega'}\rangle_G
        =
        \begin{cases}
        |\omega|,&\omega=\omega',\\
        0,&\omega\neq\omega'.
        \end{cases}
\]
Therefore
\[
        \langle\theta_\omega,\rho_C\rangle_G
        =
        |\omega|b_{\omega,C}
        =
        \dim_\mathbb{Q}(V_\omega^C).
\]
Thus the coefficient of $\theta_\omega$ in $\rho_C$ is
$b_{\omega,C}$, while the dimension of the fixed subspace is
$|\omega|b_{\omega,C}$.

We now prove directly that the characters $\rho_C$, with
$C\in\mathcal{C}^+(G)$, are linearly independent. Enumerate the cyclic
subgroups as
\[
        C_1,\ldots,C_{m_G+1},
        \qquad
        |C_1|\leq\cdots\leq|C_{m_G+1}|,
\]
and choose a generator $g_i$ of each $C_i$. For the trivial
subgroup, take $g_i=1$. Since $G$ is abelian, an element
$g\in G$ fixes a coset $hC_j$ if and only if
\[
        ghC_j=hC_j
        \quad\Longleftrightarrow\quad
        g\in C_j.
\]
Thus $g$ fixes every coset when $g\in C_j$, and fixes no coset
otherwise. The permutation character consequently satisfies
\[
        \rho_{C_j}(g_i)
        =
        \begin{cases}
        [G:C_j],&C_i\subseteq C_j,\\
        0,&C_i\not\subseteq C_j.
        \end{cases}
\]

Consider the square evaluation matrix
\[
        M=\bigl(\rho_{C_j}(g_i)\bigr)_
          {1\leq i,j\leq m_G+1}.
\]
If $i>j$, then $|C_i|\geq|C_j|$. Inclusion
$C_i\subseteq C_j$ would force equality of the two subgroups
when their orders are equal, and is impossible when
$|C_i|>|C_j|$. Since the enumeration has no repetitions,
$M_{i,j}=0$ for $i>j$. Hence $M$ is upper triangular, with
diagonal entries
\[
        M_{i,i}=[G:C_i].
\]
It follows that
\[
        \det M=\prod_{i=1}^{m_G+1}[G:C_i]\neq0.
\]
Any linear relation among the characters $\rho_{C_j}$,
evaluated at $g_1,\ldots,g_{m_G+1}$, would therefore have all
its coefficients equal to zero.

There are $m_G+1$ such permutation characters and $m_G+1$
irreducible rational characters. Hence the $\rho_C$ form a
basis of the rational character space. This verifies directly
the basis assertion obtained from Artin induction in the
present abelian setting; cf.\ \cite[Chapter~9]{Serre1977}.
Their coefficient matrix $B^+$ is therefore nonsingular
over $\mathbb{Q}$.

Order the trivial character and the trivial subgroup first
in $B^+$. The trivial character is trivial on every subgroup,
and every character is trivial on the trivial subgroup.
Consequently,
\[
        B^+
        =
        \begin{pmatrix}
        1&\mathbf1^{\mathrm t}\\
        \mathbf1&B
        \end{pmatrix},
\]
where $B$ is indexed by $\Omega(G)$ and $\mathcal{C}(G)$, and
$\mathbf1$ is the column vector of $m_G$ ones. By the
definition of $A_G$,
\[
        A_G=\mathbf1\mathbf1^{\mathrm t}-B.
\]
Subtracting the first row of $B^+$ from each of its remaining
rows gives
\[
        \begin{pmatrix}
        1&\mathbf1^{\mathrm t}\\
        0&B-\mathbf1\mathbf1^{\mathrm t}
        \end{pmatrix}
        =
        \begin{pmatrix}
        1&\mathbf1^{\mathrm t}\\
        0&-A_G
        \end{pmatrix}.
\]
These row operations preserve the determinant, so
\[
        \det(B^+)=\det(-A_G)
        =(-1)^{m_G}\det(A_G).
\]
Since $\det(B^+)\neq0$, we obtain $\det(A_G)\neq0$.
The entries of $A_G$ are integers, and hence its inverse has
rational entries. This proves
\[
        A_G\in\mathrm{GL}_{m_G}(\mathbb{Q}).
\]

Now fix a marked tame $G$-extension $(K,\iota)$. By
Proposition~\ref{prop:conductor-inertia}, for every
$\omega\in\Omega(G)$,
\[
        f_\omega(K,\iota)
        =
        \prod_{C\in\mathcal{C}(G)}
        q_C(K,\iota)^{a_{\omega,C}}.
\]
Each $q_C(K,\iota)$ is positive, with value $1$ when no
ramified prime has inertia subgroup $C$. Taking logarithms
therefore gives
\[
        \log f_\omega(K,\iota)
        =
        \sum_{C\in\mathcal{C}(G)}
        a_{\omega,C}\log q_C(K,\iota).
\]
These identities, one for each $\omega$, are exactly
\[
        y(K,\iota)=A_Gx(K,\iota).
\]
In particular, nonsingularity also gives
\[
        x(K,\iota)=A_G^{-1}y(K,\iota).
\]

Finally, recall that
\[
        E_G=\mathbb{R}^{\Omega(G)}/\mathbb{R}\mathbf1
\]
and that $\Phi_G$ is defined by
\[
        \Phi_G([y])
        =
        \operatorname{sh}\left(
        L_G,
        \bigoplus_{\omega\in\Omega(G)}
        e^{y_\omega}Q_\omega
        \right).
\]
This expression depends only on $[y]$: replacing $y$ by
$y+c\mathbf1$ multiplies the entire quadratic form by $e^c$,
which leaves its similarity class unchanged. For
$y=y(K,\iota)$, we have
$e^{y_\omega}=f_\omega(K,\iota)$. Theorem~\ref{thm:main}
therefore gives
\[
        \operatorname{sh}(K)
        =
        \operatorname{sh}\left(
        L_G,
        \bigoplus_{\omega\in\Omega(G)}
        f_\omega(K,\iota)Q_\omega
        \right)
        =
        \Phi_G([y(K,\iota)])
        =
        \Phi_G([A_Gx(K,\iota)]).
\]
Every tame extension whose Galois group is isomorphic to
$G$ admits a marking, and the displayed equality places its
shape in $\Phi_G(E_G)$. This proves the final inclusion.

\label{lem:tame-discriminant}
Let $p$ be ramified in a tame $G$-extension $K/\mathbb{Q}$, and suppose $I_p=C$. Then
\[
        v_p(\operatorname{disc}(K))=a_C.
\]
Consequently
\begin{equation}\label{eq:disc-inertia}
        |\operatorname{disc}(K)|=
        \prod_{C\in\mathcal{C}(G)}q_C(K)^{a_C}.
\end{equation}

<!-- proof-7: body_md -->
Fix a prime $\mathfrak p$ of $K$ above $p$.  Since $K/\mathbb{Q}$ is Galois, the ramification index $e=e(\mathfrak p/p)$, the residue degree $f=f(\mathfrak p/p)$, and the number $g$ of primes above $p$ are independent of the choice of $\mathfrak p$, and
\[
        n=[K:\mathbb{Q}]=efg.
\]
Moreover the inertia subgroup $I_p$ has order $e$, so $e=d_C$.

Let $K_{\mathfrak p}/\mathbb{Q}_p$ be the corresponding local extension.  For a finite Galois extension of local fields, the exponent of the different is
\[
        d(\mathfrak p/p)=\sum_{i\geq0}(|G_i|-1),
\]
where $G_i$ are the lower ramification groups.  Tameness means $G_1=1$, while $G_0=I_p$ has order $e$.  Hence
\[
        d(\mathfrak p/p)=e-1.
\]
The exponent of $p$ in the global discriminant is obtained by taking the norm of the global different.  Each of the $g$ primes above $p$ contributes residue degree $f$, and therefore
\begin{align*}
        v_p(\operatorname{disc}(K))
        &=\sum_{\mathfrak p\mid p}f(\mathfrak p/p)
          d(\mathfrak p/p)\\
        &=gf(e-1)\\
        &=\frac{n}{e}(e-1)\\
        &=n\left(1-\frac1e\right)
         =a_C.
\end{align*}
This proves the local formula.  Finally, every ramified prime belongs to exactly one inertia class $C$, and
\[
        q_C(K)=\prod_{I_p=C}p.
\]
Multiplying the local contributions over all ramified primes and grouping the primes according to $C$ gives
\[
        |\operatorname{disc}(K)|
        =\prod_{C\in\mathcal{C}(G)}q_C(K)^{a_C}.
\]

\label{lem:tame-parametrization}
Marked tame $G$-extensions of $\mathbb{Q}$ are parametrized by admissible tuples $\mathbf q$, together with, for every prime $p\mid q_C$, a surjection $\mathbb{F}_p^\times\twoheadrightarrow C$, subject to the generation condition

<!-- proof-8: body_md -->
Let $(K,\iota)$ be a marked tame $G$-extension and let $N$ be the conductor of $K$. Proposition~\ref{prop:hilbert-speiser} shows that $N$ is odd and squarefree.  Put $M=\mathbb{Q}(\zeta_N)$.  Restriction of automorphisms gives a surjection
\[
        \operatorname{Gal}(M/\mathbb{Q})\longrightarrow\operatorname{Gal}(K/\mathbb{Q}),
\]
and after composing with the marking $\iota$ we obtain a surjective homomorphism
\begin{equation}\label{eq:KW-map}
        \rho:(\mathbb{Z}/N\mathbb{Z})^\times\simeq\operatorname{Gal}(M/\mathbb{Q})
        \longrightarrow G.
\end{equation}
Conversely, the fixed field of the kernel of any surjective map of the form \eqref{eq:KW-map} is a marked abelian extension with Galois group $G$.  We now describe exactly what tameness and squarefreeness of $N$ do to this map.

Because $N$ is squarefree, the Chinese remainder theorem gives a direct product decomposition
\begin{equation}\label{eq:CRT-local-factors}
        (\mathbb{Z}/N\mathbb{Z})^\times
        \simeq\prod_{p\mid N}(\mathbb{Z}/p\mathbb{Z})^\times
        =\prod_{p\mid N}\mathbb{F}_p^\times.
\end{equation}
For a fixed prime $p\mid N$, the factor $\mathbb{F}_p^\times$ in \eqref{eq:CRT-local-factors} is exactly the inertia subgroup at $p$ in the full cyclotomic extension $M/\mathbb{Q}$.  Indeed,
\[
        M=\mathbb{Q}(\zeta_p)\,\mathbb{Q}(\zeta_{N/p}).
\]
The first factor is totally ramified at $p$ with ramification group $(\mathbb{Z}/p\mathbb{Z})^\times$, while the second factor is unramified at $p$ because its discriminant is supported on the primes dividing $N/p$.  Since the two factors are linearly disjoint, the inertia subgroup of $M/\mathbb{Q}$ at $p$ is precisely the $p$-factor in \eqref{eq:CRT-local-factors}.  Passing to the quotient $G$ shows that the image of this factor under $\rho$ is the inertia subgroup $I_p$ of $K/\mathbb{Q}$.

Let this image be $C$.  The restriction of $\rho$ to the $p$-factor is then a surjection
\[
        \rho_p:\mathbb{F}_p^\times\twoheadrightarrow C.
\]
Since $\mathbb{F}_p^\times$ is cyclic of order $p-1$, a surjection onto the cyclic group $C$ of order $d_C$ exists if and only if
\[
        d_C\mid p-1,
\]
or equivalently $p\equiv1\pmod{d_C}$.  If this divisibility holds, choose a generator $g_p$ of $\mathbb{F}_p^\times$.  A homomorphism is determined by $\rho_p(g_p)$, and it is surjective exactly when this image is a generator of $C$.  There are $\varphi(d_C)$ such generators.  Thus there are exactly $\varphi(d_C)$ local surjections with prescribed image $C$.

Group the ramified primes according to their inertia subgroups. For each $C\in\mathcal{C}(G)$ set
\[
        q_C=\prod_{\substack{p\mid N\\\rho(\mathbb{F}_p^\times)=C}}p.
\]
Each prime divisor of $N$ appears in one and only one of these products.  Hence every $q_C$ is squarefree and the integers $q_C$ are pairwise coprime.  The preceding paragraph shows that every prime divisor of $q_C$ is congruent to $1$ modulo $d_C$.  Thus the tuple is admissible.

Conversely, suppose an admissible tuple $\mathbf q=(q_C)_C$ is given.  Put
\[
        N=\prod_C q_C.
\]
For every prime $p\mid q_C$, choose one of the $\varphi(d_C)$ surjections
\[
        \rho_p:\mathbb{F}_p^\times\twoheadrightarrow C\hookrightarrow G.
\]
Using the product decomposition \eqref{eq:CRT-local-factors}, the product of these local maps defines a homomorphism
\[
        \rho:(\mathbb{Z}/N\mathbb{Z})^\times\longrightarrow G.
\]
Its image is generated by the images of the local factors, namely by the cyclic subgroups $C$ for which $q_C>1$.  Consequently $\rho$ is surjective if and only if
\[
        \langle C:q_C>1\rangle=G.
\]
When this condition holds, the fixed field of $\ker(\rho)$ has conductor exactly $N$: every $p\mid N$ has nontrivial local image, so it ramifies, and because the extension is tame its conductor exponent is one.  Thus no prime can be removed from $N$.  The resulting extension is therefore a tame marked $G$-extension with the prescribed inertia tuple.

The choices at distinct primes are independent.  For a fixed admissible tuple satisfying the generation condition, the number of collections of local surjections is therefore
\[
        \prod_C\varphi(d_C)^{\omega(q_C)}=w(\mathbf q),
\]
which proves the asserted weight formula.

Finally, let $K$ be an unmarked field with $\operatorname{Gal}(K/\mathbb{Q})\simeq G$.  The set of isomorphisms
\[
        \operatorname{Gal}(K/\mathbb{Q})\xrightarrow{\sim}G
\]
is a torsor under $\operatorname{Aut}(G)$, and hence contains exactly $|\operatorname{Aut}(G)|$ elements.  Changing the marking generally permutes the labels on the cyclic subgroups and therefore changes the labeled tuple.  For this reason division by $|\operatorname{Aut}(G)|$ is legitimate only after one has summed over a collection stable under changing the marking.  All passages from marked to unmarked fields below will be made in this way.

<!-- lemma-5: body_md -->
\label{lem:zell-factorization}
There is a neighbourhood $U$ of $1$ and a holomorphic nonvanishing function $h_\ell$ on $U$ such that
\begin{equation}\label{eq:zell-factorization}
        Z_\ell(z):=
        \prod_{p\equiv1\;({\rm mod}\,\ell)}
        (1-p^{-z})^{-(\ell-1)}
        =\frac{h_\ell(z)}{z-1}
\end{equation}
for $z\in U$ with $\Re(z)>1$, and hence by continuation on $U\setminus\{1\}$.
<!-- end lemma-5 -->
<!-- proof-9: body_md -->
Let $\mathcal X_\ell$ denote the full group of Dirichlet characters modulo the prime $\ell$.  It has $\ell-1$ elements.  For $\Re(z)>1$ define
\[
        P_\ell(z)=\prod_{\chi\in\mathcal X_\ell}L(z,\chi).
\]
We compare its Euler factors with those of $Z_\ell(z)$.

If $p\neq\ell$ and $p\equiv1\pmod\ell$, then the class of $p$ is the identity in $(\mathbb{Z}/\ell\mathbb{Z})^\times$, so $\chi(p)=1$ for every $\chi\in\mathcal X_\ell$.  The $p$-Euler factor of $P_\ell$ is therefore
\[
        \prod_{\chi\in\mathcal X_\ell}(1-\chi(p)p^{-z})^{-1}
        =(1-p^{-z})^{-(\ell-1)},
\]
which is exactly the $p$-Euler factor occurring in $Z_\ell$.

Suppose instead that $p\neq\ell$ and $p\not\equiv1\pmod\ell$.  There is no $p$-factor in $Z_\ell$, while the reciprocal of the $p$-factor of $P_\ell$ is
\[
        R_p(z):=\prod_{\chi\in\mathcal X_\ell}(1-\chi(p)p^{-z}).
\]
Expanding this finite product, the coefficient of $p^{-z}$ is
\[
        -\sum_{\chi\in\mathcal X_\ell}\chi(p).
\]
Character orthogonality on the finite group $(\mathbb{Z}/\ell\mathbb{Z})^\times$ gives
\[
        \sum_{\chi\in\mathcal X_\ell}\chi(p)=0
\]
because the residue class of $p$ is not the identity.  Consequently
\[
        R_p(z)=1+O_\ell(p^{-2\Re(z)}),
\]
locally uniformly in $z$.  At the prime $p=\ell$, every Dirichlet $L$-function modulo $\ell$ omits the Euler factor, and $Z_\ell$ also omits $\ell$, so there is no additional factor.

It follows that the quotient
\[
        R_\ell(z):=\frac{Z_\ell(z)}{P_\ell(z)}
\]
can be represented, initially for $\Re(z)>1$, by an Euler product whose local factors are $1+O_\ell(p^{-2\Re(z)})$.  Hence this product converges absolutely and locally uniformly in every closed half-plane
\[
        \Re(z)\geq\frac12+\delta
\]
with $\delta>0$.  In particular, $R_\ell(z)$ extends to a holomorphic function in a neighbourhood of $z=1$.  Shrinking the neighbourhood if necessary, it is nonzero there: the absolutely convergent Euler product is nonzero at $1$, and holomorphic functions which are nonzero at a point remain nonzero on a sufficiently small neighbourhood.

 The principal character $\chi_0$ modulo $\ell$ satisfies
\[
        L(z,\chi_0)=\zeta(z)(1-\ell^{-z}),
\]
so it has a simple pole at $z=1$.  Every nonprincipal character modulo the prime $\ell$ is primitive, and Dirichlet's nonvanishing theorem gives
\[
        L(1,\chi)\neq0.
\]
Thus
\[
        P_\ell(z)
        =\frac{u_\ell(z)}{z-1}
\]
in a neighbourhood of $1$, where $u_\ell$ is holomorphic and nonzero.  Multiplying by the holomorphic nonzero factor $R_\ell(z)$ yields
\[
        Z_\ell(z)=\frac{h_\ell(z)}{z-1}\quad \text{and}\quad h_\ell(z)=R_\ell(z)u_\ell(z),
\]
\hypersetup{linkcolor=black,citecolor=black,urlcolor=black}and $h_\ell$ is holomorphic and nonvanishing near $1$. Moreover, $h_\ell(1)>0$: it is the nonzero limit of $(z-1)Z_\ell(z)>0$ as real $z$ decreases to $1$. This proves \eqref{eq:zell-factorization}.

The factorization also supplies the continuation required for the Tauberian theorem. Every nonprincipal Dirichlet $L$-function in $P_\ell$ is entire, and its principal factor has only the pole at $1$. The product $R_\ell$ is holomorphic for $\Re z>1/2$. Consequently $Z_\ell=R_\ell P_\ell$ is meromorphic in that half-plane, and its only pole there is the simple pole at $1$. In particular, it is holomorphic at every other point of the line $\Re z=1$.

\begin{equation}\label{eq:HQ-local-factor}

Only finitely many primes divide $Q$, so these factors cause no convergence problem.  At the point $(1,\ldots,1)$ every local factor is positive. We compute the mixed moments of the normalized logarithms and compare them with the moments of normalized Lebesgue measure on $\Delta_{s-1}$. The moment with all exponents zero gives the total count.

For a multi-index $\mathbf k=(k_1,\ldots,k_s)\in\mathbb{Z}_{\geq0}^s$, define $|\mathbf k|=k_1+\cdots+k_s$ and
set
\begin{equation}\label{eq:Mk-def}
        M_{\mathbf k}(Y;Q)
        =\sum_{q_1\cdots q_s\leq Y}
        w(\mathbf q)\prod_{i=1}^s(\log q_i)^{k_i},
\end{equation}
where the sum runs over ordered tuples $(q_1,\ldots,q_s)$
of pairwise coprime positive squarefree integers such that
$(q_1\cdots q_s,Q)=1$ and every prime divisor of
$q_1\cdots q_s$ is congruent to $1$ modulo $\ell$.
For $\Re(z)>1$, absolute convergence allows us to differentiate the multiple Dirichlet series term by term.  After differentiating in the $i$-th variable $k_i$ times and then restricting to the diagonal $z_1=\cdots=z_s=z$, we obtain
\begin{align}\label{eq:Dk-def}
        D_{\mathbf k,Q}(z)
        &:=\sum_{\mathbf q}
        \frac{w(\mathbf q)\prod_i(\log q_i)^{k_i}}
        {(q_1\cdots q_s)^z}\notag\\
        &=(-1)^{|\mathbf k|}
        \left.
        \frac{\partial^{|\mathbf k|}}
        {\partial z_1^{k_1}\cdots\partial z_s^{k_s}}
        \mathcal F_Q(\mathbf z)
        \right|_{z_1=\cdots=z_s=z}.
\end{align}

Insert the factorization from Lemma~\ref{lem:FQ-factorization}:
\[
        \mathcal F_Q(\mathbf z)
        =C_Q(\mathbf z)\prod_{i=1}^s(z_i-1)^{-1}.
\]
The highest possible pole order in \eqref{eq:Dk-def} occurs when, for every $i$, all $k_i$ derivatives in the variable $z_i$ fall on the factor $(z_i-1)^{-1}$.  Since
\[
        \frac{d^{k_i}}{dz_i^{k_i}}(z_i-1)^{-1}
        =(-1)^{k_i}k_i!(z_i-1)^{-k_i-1},
\]
the signs cancel the prefactor $(-1)^{|\mathbf k|}$ in \eqref{eq:Dk-def}.  The resulting leading term is
\[
        C_Q(1,\ldots,1)
        \left(\prod_{i=1}^sk_i!\right)
        (z-1)^{-s-|\mathbf k|}.
\]
If at least one derivative falls on $C_Q$, then at least one polar factor receives fewer derivatives, and the total pole order drops by at least one.  Because $C_Q$ and all of its derivatives are holomorphic near $(1,\ldots,1)$, we therefore have
\begin{equation}\label{eq:moment-pole}
        D_{\mathbf k,Q}(z)
        =c_{\ell,s}(Q)
        \left(\prod_{i=1}^sk_i!\right)
        (z-1)^{-s-|\mathbf k|}
        +O_{\mathbf k,Q}\left((z-1)^{-s-|\mathbf k|+1}\right).
\end{equation}

The coefficients of $D_{\mathbf k,Q}$ are nonnegative, and its series converges absolutely for $\Re z>1$. To apply the Ikehara--Delange theorem, we also check continuation along the rest of the boundary. By Lemma~\ref{lem:FQ-factorization}, $\mathcal F_Q$ is meromorphic in the product domain $\Re z_i>1/2$, with possible poles only on $z_i=1$. Partial differentiation preserves this property: the differentiated factors remain holomorphic away from these coordinate hyperplanes. Restricting the resulting expressions to the diagonal gives a meromorphic continuation of $D_{\mathbf k,Q}$ to $\Re z>1/2$ whose only possible pole is at $1$. Thus it is holomorphic at every other point of the full boundary line $\Re z=1$.

Near $1$, equation \eqref{eq:moment-pole} can be written as $D_{\mathbf k,Q}(z)=H_{\mathbf k,Q}(z)(z-1)^{-\rho}$ with $\rho=s+|\mathbf k|$, where $H_{\mathbf k,Q}$ is holomorphic and $H_{\mathbf k,Q}(1)=c_{\ell,s}(Q)\prod_i k_i!>0$. All the hypotheses of the Tauberian statement preceding Lemma~\ref{lem:box-counting} are now satisfied. Applying it with this pole order and leading coefficient gives
\begin{equation}\label{eq:moment-asymptotic}
        M_{\mathbf k}(Y;Q)
        \sim c_{\ell,s}(Q)
        \frac{\prod_i k_i!}{(s+|\mathbf k|-1)!}
        Y(\log Y)^{s+|\mathbf k|-1}.
\end{equation}
For $\mathbf k=\mathbf0$, this is precisely \eqref{eq:product-count}.

We now turn the moment asymptotics into weak convergence.  Put $T=\log Y$ and let $S_Q(Y)$ denote the total sum in \eqref{eq:product-count}.  Define a probability measure $\mu_Y$ on
\[
        \Sigma_s:=\{\mathbf x\in\mathbb{R}_{\geq0}^s:x_1+\cdots+x_s\leq1\}
\]
by
\[
        \mu_Y
        =\frac1{S_Q(Y)}
        \sum_{\substack{q_1\cdots q_s\leq Y\\q_i\text{ as above}}}
        w(\mathbf q)
        \delta_{(\log q_1/T,\ldots,\log q_s/T)}.
\]
The support lies in $\Sigma_s$ because the product constraint is equivalent to
\[
        \frac{\log q_1}{T}+\cdots+\frac{\log q_s}{T}\leq1.
\]
For the monomial $\mathbf x^{\mathbf k}=x_1^{k_1}\cdots x_s^{k_s}$, equations \eqref{eq:product-count} and \eqref{eq:moment-asymptotic} give
\begin{align*}
        \int_{\Sigma_s}\mathbf x^{\mathbf k}\,d\mu_Y(\mathbf x)
        &=\frac{M_{\mathbf k}(Y;Q)}{T^{|\mathbf k|}S_Q(Y)}\\
        &\longrightarrow
        \frac{(s-1)!\prod_i k_i!}
        {(s+|\mathbf k|-1)!}.
\end{align*}

Let $\mu_\Delta$ be normalized Lebesgue measure on the outer face
\[
        \Delta_{s-1}=\{\mathbf t\in\mathbb{R}_{\geq0}^s:t_1+\cdots+t_s=1\}.
\]
With the coordinates $t_1,\ldots,t_{s-1}$ and $t_s=1-\sum_{i<s}t_i$, ordinary Lebesgue measure on this simplex has total mass $1/(s-1)!$.  Thus
\[
        d\mu_\Delta=(s-1)!\,d\mathbf t.
\]
The Dirichlet beta integral yields
\[
        \int_{\Delta_{s-1}}\prod_{i=1}^st_i^{k_i}\,d\mathbf t
        =\frac{\prod_i k_i!}{(s+|\mathbf k|-1)!},
\]
and hence
\[
        \int\mathbf t^{\mathbf k}\,d\mu_\Delta(\mathbf t)
        =\frac{(s-1)!\prod_i k_i!}{(s+|\mathbf k|-1)!}.
\]
The limiting moments of $\mu_Y$ are therefore exactly the moments of $\mu_\Delta$.

Since $\Sigma_s$ is compact, polynomials in the coordinate functions form a subalgebra of $C(\Sigma_s)$ which contains the constants and separates points.  By the Stone--Weierstrass theorem they are uniformly dense in $C(\Sigma_s)$.  Convergence of all polynomial moments therefore implies
\[
        \mu_Y\Longrightarrow\mu_\Delta
\]
weakly.

If $F$ is continuous on $\Sigma_s$, weak convergence gives
\[
        \frac1{S_Q(Y)}
        \sum_{q_1\cdots q_s\leq Y}w(\mathbf q)
        F\left(\frac{\log q_1}{T},\ldots,\frac{\log q_s}{T}\right)
        \longrightarrow
        (s-1)!\int_{\Delta_{s-1}}F(\mathbf t)\,d\mathbf t.
\]
Multiplying by the asymptotic
\[
        S_Q(Y)\sim\frac{c_{\ell,s}(Q)}{(s-1)!}Y(\log Y)^{s-1}
\]
gives \eqref{eq:simplex-count} with an additive error. This formulation also applies when the limiting integral is zero, or when $F$ takes both positive and negative values.

Affine independence means that the only relation
\[
        \sum_{C\in\mathcal{M}(G)}r_Cv_C=0
\]
whose coefficients satisfy
\[
        \sum_{C\in\mathcal{M}(G)}r_C=0
\]
is the trivial relation. Suppose such a relation is given, and write $\mathbf r=(r_C)_{C\in\mathcal{M}(G)}$.  By definition $v_C=\pi(A_C)$, and the minimal columns of $A_G$ form the matrix $B_G$.  Hence
\[
        \pi(B_G\mathbf r)=0.
\]
The kernel of $\pi:\mathbb{R}^{\Omega(G)}\to E_G$ is exactly the line $\mathbb{R}\mathbf1$.  There is therefore a scalar $c\in\mathbb{R}$ such that
\begin{equation}\label{eq:BGr-c1}
        B_G\mathbf r=c\mathbf1.
\end{equation}
We consider separately the cases $G[\ell]\neq G$ and $G[\ell]=G$.

Assume first that $G$ is not elementary abelian of exponent $\ell$.  Then $G[\ell]\neq G$.  Indeed, if every element of $G$ were killed by $\ell$, the finite abelian group $G$ would be a vector space over $\mathbb{F}_\ell$ and hence isomorphic to $(C_\ell)^r$.  Thus the quotient $G/G[\ell]$ is a nontrivial finite abelian group.  It has a nontrivial complex character; inflating such a character to $G$ gives a character $\chi$ which is trivial on $G[\ell]$ and hence on every subgroup $C\in\mathcal{M}(G)$.  The rational Galois orbit of $\chi$ therefore determines a row of $B_G$ all of whose entries are zero.  Reading this row in \eqref{eq:BGr-c1} gives $c=0$.

We are left with $B_G\mathbf r=0$.  The columns of $B_G$ are a subset of the columns of the invertible square matrix $A_G$.  Every subset of the columns of an invertible matrix is linearly independent, so $\mathbf r=0$.

Now suppose that $G=(C_\ell)^r$.  In this case every nontrivial cyclic subgroup has order $\ell$, so $B_G=A_G$.  A nontrivial character has kernel a hyperplane $H\subset\mathbb{F}_\ell^r$.  There are
\[
        \frac{\ell^r-1}{\ell-1}
\]
lines in $\mathbb{F}_\ell^r$ and
\[
        \frac{\ell^{r-1}-1}{\ell-1}
\]
lines in $H$.  Therefore exactly
\[
        \frac{\ell^r-\ell^{r-1}}{\ell-1}=\ell^{r-1}
\]
lines are not contained in $H$.  By the definition of $A_G$, every row therefore contains $\ell^{r-1}$ entries equal to $1$.  Hence
\[
        A_G\mathbf1=\ell^{r-1}\mathbf1.
\]
Since $A_G$ is invertible,
\[
        A_G^{-1}\mathbf1=\ell^{1-r}\mathbf1.
\]
Equation \eqref{eq:BGr-c1} now becomes $A_G\mathbf r=c\mathbf1$, and applying $A_G^{-1}$ gives
\[      \mathbf r=c\ell^{1-r}\mathbf1.
\]
But the coefficients of an affine relation also satisfy $\sum_Cr_C=0$.  A constant vector has zero coordinate sum only when it is zero.  Therefore $c=0$ and $\mathbf r=0$.

In both cases the only affine relation is trivial.  Thus the points $v_C$ are affinely independent, and the convex hull of $s_G$ such points has dimension $s_G-1$.

        |\mathscr F_G^{\rm mark}(X)|
        \sim C_G^{\rm mark}X^{\alpha_G}(\log X)^{s_G-1}.
\end{equation}
Moreover, for every bounded continuous function $F$ on $E_G$,
\begin{align}\label{eq:shape-limit-marked}
&\frac1{|\mathscr F_G^{\rm mark}(X)|}
\sum_{(K,\iota)\in\mathscr F_G^{\rm mark}(X)}
F\left(\frac{a_0}{\log X}\lambda(K,\iota)\right)\notag\\
&\qquad\longrightarrow
(s_G-1)!\int_{\Delta_{s_G-1}}
F\left(\sum_{C\in\mathcal{M}(G)}t_Cv_C\right)d\mathbf t.
\end{align}
Thus the marked rescaled logarithmic conductor parameters become uniformly distributed on $\mathscr P_G$.

For unmarked fields one has
\begin{equation}\label{eq:total-G-count}
        |\mathscr F_G(X)|
        \sim C_GX^{\alpha_G}(\log X)^{s_G-1},
       \quad \text{where}\quad C_G=\frac{C_G^{\rm mark}}{|\operatorname{Aut}(G)|}.
\end{equation}
For unmarked fields, the rescaled canonical classes $a_0\overline\lambda(K)/\log X$ are distributed according to the pushforward of normalized Lebesgue measure on $\mathscr P_G$ under the quotient map $E_G\to E_G/\operatorname{Aut}(G)$. Equivalently, \eqref{eq:shape-limit-marked} also holds for unmarked fields whenever $F$ is $\operatorname{Aut}(G)$-invariant.

Write
\[
        \mathcal{C}(G)=\mathcal{M}(G)\sqcup\mathcal{C}_{>}(G),
\]
where $\mathcal{C}_{>}(G)$ consists of the cyclic subgroups of order strictly larger than $\ell$.  Fix for the moment an admissible tuple in the higher coordinates,
\[
        \mathbf q_{>}=(q_C)_{C\in\mathcal{C}_{>}(G)}.
\]
We use the notation
\begin{equation}\label{eq:higher-notation}
        Q_{>}=
        \prod_{C\in\mathcal{C}_{>}(G)}q_C,
        \qquad
        D_{>}=
        \prod_{C\in\mathcal{C}_{>}(G)}q_C^{a_C}\quad \text{and}\quad
        w_{>}=
        \prod_{C\in\mathcal{C}_{>}(G)}
        \varphi(d_C)^{\omega(q_C)}.
\end{equation}
Here $Q_{>}$ is the product of the primes in the higher coordinates, $D_{>}$ is their contribution to the discriminant, and $w_{>}$ counts the corresponding choices of local surjections. Pairwise coprimality requires the minimal coordinates to be coprime to $Q_{>}$.

For every $C\in\mathcal{M}(G)$ one has $a_C=a_0$.  Thus the discriminant formula of Lemma~\ref{lem:tame-discriminant} becomes
\[
        D_{>}
        \left(\prod_{C\in\mathcal{M}(G)}q_C\right)^{a_0}
        \leq X.
\]
Since $\alpha_G=1/a_0$, this is equivalent to
\begin{equation}\label{eq:Y-higher}
        \prod_{C\in\mathcal{M}(G)}q_C
        \leq Y_{\mathbf q_{>}}\quad \text{and}\quad
        Y_{\mathbf q_{>}}
        :=\left(\frac{X}{D_{>}}\right)^{\alpha_G}.
\end{equation}
If $D_{>}>X$ there is no contribution.  Otherwise $Y_{\mathbf q_{>}}\geq1$.

The minimal coordinates have a uniform arithmetic description.  Every prime occurring in them is congruent to $1$ modulo $\ell$, every local surjection has multiplicity
\[
        \varphi(\ell)=\ell-1,
\]
and the primes must avoid $Q_{>}$.

The collection of all order-$\ell$ subgroups of $G$ generates $G[\ell]$.  Consequently a necessary condition for the fixed higher tuple to occur in the leading term is
\begin{equation}\label{eq:higher-generation}
        \left\langle
        G[\ell],
        C:q_C>1,\ C\in\mathcal{C}_{>}(G)
        \right\rangle=G.
\end{equation}
If this condition fails, adjoining all the minimal inertia subgroups to the active higher subgroups still gives a proper subgroup of $G$. Thus no choice of minimal coordinates makes the map in Lemma~\ref{lem:tame-parametrization} surjective.

Assume now that \eqref{eq:higher-generation} holds.  If every minimal coordinate is nontrivial, then every subgroup in $\mathcal{M}(G)$ occurs as an inertia image.  These subgroups generate $G[\ell]$, and together with the active higher subgroups they generate $G$ by \eqref{eq:higher-generation}.  Thus every such tuple gives a surjective global map.

We must check that allowing some minimal coordinate to equal $1$ does not change the leading term.  Fix $C_0\in\mathcal{M}(G)$ and impose $q_{C_0}=1$.  Dropping the generation condition gives an upper bound, and the remaining $s_G-1$ minimal coordinates satisfy the same product constraint.  Lemma~\ref{lem:box-counting}, applied with $s_G-1$ coordinates, gives
\[
        O_G\left(
        Y_{\mathbf q_{>}}
        (\log 2Y_{\mathbf q_{>}})^{s_G-2}
        \right)
\]
when $s_G\geq2$.  There are only $s_G$ choices of a missing coordinate, so the same estimate holds for the union of all tuples with at least one missing minimal coordinate.  If $s_G=1$, the only tuple with the unique minimal coordinate absent is $q_C=1$, and its contribution is $O(1)$, which is again lower order than the main term $\asymp Y_{\mathbf q_{>}}$. Therefore, for a fixed higher tuple, imposing the generation condition changes the minimal-coordinate count by $o\bigl(Y_{\mathbf q_{>}}(\log Y_{\mathbf q_{>}})^{s_G-1}\bigr)$.

Lemma~\ref{lem:minimal-counting} now gives, for a fixed higher tuple satisfying \eqref{eq:higher-generation},
\begin{align}\label{eq:fixed-higher-main-detailed}
N_{\mathbf q_{>}}^{\rm mark}(X)
&=
w_{>}
\frac{c_{\ell,s_G}(Q_{>})}{(s_G-1)!}
Y_{\mathbf q_{>}}
(\log Y_{\mathbf q_{>}})^{s_G-1}
\bigl(1+o_{\mathbf q_{>}}(1)\bigr).
\end{align}
For a fixed higher tuple,
\[
        \log Y_{\mathbf q_{>}}
        =\alpha_G\log X-\alpha_G\log D_{>}
        =\alpha_G\log X+O_{\mathbf q_{>}}(1),
\]
and
\[
        Y_{\mathbf q_{>}}
        =X^{\alpha_G}D_{>}^{-\alpha_G}.
\]
Hence \eqref{eq:fixed-higher-main-detailed} is equivalent to
\begin{align}\label{eq:fixed-higher-main}
N_{\mathbf q_{>}}^{\rm mark}(X)
&=
\frac{w_{>}c_{\ell,s_G}(Q_{>})}
{(s_G-1)!D_{>}^{\alpha_G}}
X^{\alpha_G}(\alpha_G\log X)^{s_G-1}
\bigl(1+o_{\mathbf q_{>}}(1)\bigr).
\end{align}

The fixed-tuple asymptotic cannot be summed over infinitely many tuples without a uniform tail estimate, so we truncate first.  Fix $M\geq1$ and sum only over admissible higher tuples satisfying \eqref{eq:higher-generation} and
\[
        D_{>}\leq M.
\]
There are only finitely many such tuples: every prime dividing $Q_{>}$ divides the integer $D_{>}$, so only primes at most $M$ may occur.  The error terms in \eqref{eq:fixed-higher-main} can therefore be summed term by term over the truncated collection.

We next prove that the series defining the leading constant converges absolutely. If $C\in\mathcal{C}_{>}(G)$, then $a_C>a_0$, and therefore
\begin{equation}\label{eq:strict-higher-exponent}
        \alpha_Ga_C=\frac{a_C}{a_0}>1.
\end{equation}
Drop all coprimality and generation conditions.  The resulting majorant factors over the finitely many higher cyclic subgroups:
\begin{align*}
        \sum_{\mathbf q_{>}}
        \frac{w_{>}}{D_{>}^{\alpha_G}}

        \prod_{C\in\mathcal{C}_{>}(G)}
        \prod_{p\equiv1\;({\rm mod}\,d_C)}
        \left(1+
        \frac{\varphi(d_C)}{p^{\alpha_Ga_C}}
        \right).
\end{align*}
Each Euler product converges by \eqref{eq:strict-higher-exponent}.  Hence
\begin{equation}\label{eq:higher-series-converges}
        \sum_{\mathbf q_{>}}
        \frac{w_{>}}{D_{>}^{\alpha_G}}<\infty.
\end{equation}
Lemma~\ref{lem:FQ-factorization} also gives the uniform bound
\[
        c_{\ell,s_G}(Q_{>})\leq c_{\ell,s_G}(1).

Thus the same series remains convergent after inserting the factor $c_{\ell,s_G}(Q_{>})$.

        X^{\alpha_G}(\log 2X)^{s_G-1}
        \sum_{D_{>}>M}
        \frac{w_{>}}{D_{>}^{\alpha_G}}.
\end{equation}
After division by $X^{\alpha_G}(\log X)^{s_G-1}$, the right-hand side tends to zero as $M\to\infty$, uniformly for large $X$, by \eqref{eq:higher-series-converges}.  We may therefore first let $X\to\infty$ for the finite truncated collection and then let $M\to\infty$.

Putting that prime into the corresponding coordinate produces an admissible higher tuple satisfying \eqref{eq:higher-generation}, and its summand in \eqref{eq:CG} is positive.

which gives \eqref{eq:total-G-count} and

        C_G=\frac{C_G^{\rm mark}}{|\operatorname{Aut}(G)|}.
\]
If $F$ is $\operatorname{Aut}(G)$-invariant, its value is unchanged when the marking is replaced by $\alpha\circ\iota$.  The marked numerator is then exactly $|\operatorname{Aut}(G)|$ times the corresponding unmarked numerator.  Dividing numerator and denominator by this common factor gives the same limit for unmarked fields.  Thus the rescaled classes $a_0\overline\lambda(K)/\log X$ in $E_G/\operatorname{Aut}(G)$ have the pushforward of the simplex measure as their limiting distribution.
