<!-- theorem-1: body_md -->
\label{mainresult}
There exist a closed oriented hyperbolic three-manifold $(\Sigma,g_0)$,
real analytic negatively curved metrics $g_j\to g_0$ in $C^\infty$,
and acyclic unitary
line characters $\chi_j\to\mathbf1$ such that $Z_{g_j,\chi_j}$ is
holomorphic at zero, and
\begin{equation}\label{mainratio}
 \frac{|Z_{g_j,\chi_j}(0)|}{T_{\chi_j}(\Sigma)^2}\longrightarrow0.
\end{equation}
Thus \eqref{Friedidentity} fails for all sufficiently large $j$.

<!-- lemma-1: body_md -->
\label{holom}
There exist $\rho,\varepsilon>0$ such that, for $|z|<\rho$ and
$\tau\in[0,\varepsilon]$, the horizontal resonance spaces
$\mathcal U^k_{z,\tau}$, $0\leq k\leq4$, and the full resonance spaces $\mathcal U^k_{z,\tau}\oplus\alpha_\tau\wedge\mathcal U^{k-1}_{z,\tau},$ with $
 0\leq k\leq5,$
inside $\gamma_{\rm cut}$ have the ranks in~\eqref{ranks}.
Here $\mathcal U^{-1}_{z,\tau}=\mathcal U^5_{z,\tau}=0$.

Bases of these spaces can be chosen so that the matrices representing
the restrictions of $B_{z,\tau}$ have entries holomorphic in $z$ and
$C^1$ in $\tau$. The entries and their $\tau$-derivatives are jointly
continuous on
$\{|z|\leq\rho'\}\times[0,\varepsilon]$ for every $0<\rho'<\rho$.
The $\tau$-derivatives are also holomorphic in $z$.

The coefficients of the characteristic polynomials of these
restrictions have the same regularity. For the horizontal restriction
$A_k(z,\tau)=B_{z,\tau}|_{\mathcal U^k_{z,\tau}}$, the polynomial is
$P_k(s,z,\tau)=\det(s\operatorname{Id}-A_k(z,\tau))$.
Each characteristic coefficient $a(z,\tau)$ satisfies $a(-z,\tau)=a(z,\tau)$ and $
 a(\bar z,\tau)=\overline{a(z,\tau)}.$
In particular, these coefficients are real for real $z$.

<!-- proof-1: body_md -->
We take the common spaces $\mathcal H_0,\ldots,\mathcal H_4$ from
\cite[Appendix B.2--B.6]{CD}, on which the continued resolvent is bounded.
Their inclusions are continuous:
\[
 \mathcal H_4\hookrightarrow\mathcal H_3\hookrightarrow\mathcal H_2
       \hookrightarrow\mathcal H_1\hookrightarrow\mathcal H_0.
\]
$\mathcal H_{j+1}$ has one more order of Sobolev regularity than $\mathcal H_j$.
In each degree the untwisted resolvents
$R_{0,\tau}(s)=(L_{0,\tau}+s)^{-1}$ are uniformly bounded on the
contour at every level. The twist is a bounded multiplier:
\[
 L_{z,\tau}=L_{0,\tau}+zV_\tau,
 \qquad V_\tau=i\pi^*\omega(X_\tau)
 \quad\text{\cite[(8.9)]{CD}}.
\]
We choose one $\rho$ so that $\|zV_\tau R_{0,\tau}(s)\|\leq1/2$
on all these levels, uniformly in $\tau$ and $s\in\gamma_{\rm cut}$.
We write $R_{z,\tau}(s)=(L_{z,\tau}+s\operatorname{Id})^{-1}$ for the twisted
resolvent. The norm bound gives the convergent series
$(\operatorname{Id}+zV_\tau R_{0,\tau}(s))^{-1}
 =\sum_{n=0}^{\infty}(-zV_\tau R_{0,\tau}(s))^n.$
This gives the resolvent formula:
\begin{equation}\label{Neumann}
 R_{z,\tau}(s)=R_{0,\tau}(s)
                  (\operatorname{Id}+zV_\tau R_{0,\tau}(s))^{-1}
\end{equation}
The resolvent is holomorphic throughout the chosen disk in the character parameter. For two nearby real metric
parameters $\tau$ and $\tau'$, the identity
\[
 R_{z,\tau}-R_{z,\tau'}
 =R_{z,\tau}(L_{z,\tau'}-L_{z,\tau})R_{z,\tau'}
\]
gives continuity $\mathcal H_{j+1}\to\mathcal H_j$ for $0\leq j\leq3$,
locally uniformly in $z$.
For differentiability, \cite[Appendix B.5, (B.6)]{CD} gives operator bounds
between spaces whose Sobolev regularity differs by three orders.
On each chosen space, the multiplier $V_\tau$ depends smoothly on $\tau$. Applying the same resolvent identity to difference
quotients gives
\[
 \frac{R_{z,\tau'}-R_{z,\tau}}{\tau'-\tau}
 =-R_{z,\tau'}
     \frac{L_{z,\tau'}-L_{z,\tau}}{\tau'-\tau}R_{z,\tau}.
\]
The difference quotient of $L_{z,\tau}$ converges in the operator norm
for maps from $\mathcal H_{j+1}$ to $\mathcal H_j$. The resolvents are also
continuous in the operator norm for maps from $\mathcal H_{j+1}$ to $\mathcal H_j$.
We can therefore pass to the limit in the product as an operator from
$\mathcal H_3$ to $\mathcal H_0$, obtaining
\begin{equation}\label{metricresolvent}
 \partial_\tau R_{z,\tau}(s)
 =-R_{z,\tau}(s)(\partial_\tau L_{0,\tau}
                    +z\partial_\tau V_\tau)R_{z,\tau}(s)
 \quad\text{as maps }\mathcal H_3\to\mathcal H_0.
\end{equation}
The derivative is jointly continuous and locally uniformly holomorphic
in $z$.
Consequently the full-cut projection $\Pi_{z,\tau}$ and its first
moment $\mathsf T_{z,\tau}$, defined by
\begin{equation}\label{xp:projector-moment}
 \begin{aligned}
 \Pi_{z,\tau}&=\frac1{2\pi i}
     \int_{\gamma_{\rm cut}} R_{z,\tau}(s)\,ds \text{ and }
 \mathsf T_{z,\tau}=\frac1{2\pi i}
     \int_{\gamma_{\rm cut}} sR_{z,\tau}(s)\,ds
     =B_{z,\tau}\Pi_{z,\tau},
 \end{aligned}
\end{equation}
have the same regularity. Untwisted ranks persist by
\cite[Appendix B.6]{CD}. To check rank persistence under twisting,
we write $\|\cdot\|_j$ for the operator norm on $\mathcal H_j$ and choose
constants $K_R,K_V>0$ such that, on every chosen level,
\[
 \|R_{0,\tau}(s)\|_j\leq K_R,
 \qquad\|V_\tau R_{0,\tau}(s)\|_j\leq K_V.
\]
The series and the contour length $2\pi\lambda_{\rm cut}$ give
\[
 \|\Pi_{z,\tau}-\Pi_{0,\tau}\|_j
 \leq\lambda_{\rm cut}K_R\frac{|z|K_V}{1-|z|K_V}<1,
\]
after shrinking $\rho$ again. Two projections at distance less than
one have the same finite rank.
Let $N$ be the dimension of the full resonance space in the chosen degree. Choose fixed smooth
forms $f_1,\ldots,f_N$ and linear functionals
$\varphi_1,\ldots,\varphi_N$ given by pairing with fixed smooth dual
forms, so that the matrix $F$ below is invertible at the origin:
\begin{equation}\label{xp:frame-matrices}
 F_{ij}=\varphi_i(\Pi f_j),\qquad (M_B)_{ij}=\varphi_i(\mathsf T f_j)
\end{equation}
The vectors $\Pi f_j$ then form a basis of the cut near the origin.
The matrix of $B$ in this basis is $F^{-1}M_B$, because
\[
 B\Pi f_j=\sum_{l=1}^N\Pi f_l(F^{-1}M_B)_{lj}.
\]
We write $N_h$ for the horizontal rank in that degree. For the horizontal
cut, we repeat this construction with the projection
$(\operatorname{Id}-\alpha_\tau\wedge\iota_{X_\tau})\Pi$ and with $N_h$ smooth
forms $f_j$ and $N_h$ linear functionals $\varphi_i$, chosen as above.
After shrinking $\rho,\varepsilon$, the matrices used to construct the full
and horizontal bases remain invertible throughout the parameter set. Fix $0<\rho_*<\rho$ and put
$\mathcal P=\{(z,\tau):|z|\le\rho_*,\ 0\le\tau\le\varepsilon\}$.
The inverse frame matrices are uniformly bounded on $\mathcal P$.
The horizontal projection commutes with $\Pi$. We set
$\mathcal U^{-1}=\mathcal U^5=0$. The contact splitting gives
\[
 \dim\operatorname{ran}(\Pi|_{\text{degree }k})
 =\dim\mathcal U^{k-1}+\dim\mathcal U^k\qquad(0\leq k\leq5).
\]
Since the full resonance spaces have constant dimensions, this identity
shows by induction on $k$, starting with $\mathcal U^{-1}=0$, that the horizontal
dimensions are constant as well. In the bases constructed above, the
matrices representing $B$ have entries holomorphic in $z$ and $C^1$ in $\tau$.

We denote an orbit's transverse symplectic return map by $P_\gamma$,
its length by $\ell_\gamma=\ell_{g_\tau}(\gamma)$, and the integral
of the harmonic form along its projection by
$I_\gamma=\int_{\pi\gamma}\omega$.
The degree-$k$ dynamical determinant in a right half-plane is
\[
 \mathscr D_k(s,z,\tau)=\exp\left(-\sum_{\gamma\in\mathcal G^\#_{g_\tau}}
 \sum_{j\geq1}\frac{e^{-sj\ell_\gamma-izjI_\gamma}}{j}
 \frac{\operatorname{tr}(\Lambda^kP_\gamma^j)}{|\det(\operatorname{Id}-P_\gamma^j)|}\right).
\]
The reversed orbit $\bar\gamma$ has
$I_{\bar\gamma}=-I_\gamma$ and return map
similar to $P_\gamma^{-1}$. A symplectic map preserves the transverse
two-form. In particular, its determinant is one and its inverse is
similar to its transpose. Thus
\[
 \operatorname{tr}(\Lambda^kP_\gamma^{-j})=\operatorname{tr}(\Lambda^kP_\gamma^j),\qquad
 |\det(\operatorname{Id}-P_\gamma^{-j})|=|\det(\operatorname{Id}-P_\gamma^j)|.
\]
Pairing the two orientations gives $e^{-izjI_\gamma}+e^{izjI_\gamma}=2\cos(zjI_\gamma).$
Hence the determinant satisfies
\[
 \mathscr D_k(s,-z,\tau)=\mathscr D_k(s,z,\tau),\qquad
 \overline{\mathscr D_k(s,z,\tau)}
   =\mathscr D_k(\bar s,\bar z,\tau).
\]
These identities continue meromorphically. Apply
\cite[Section~5, Theorem~4]{DG} to the complex bundle
$\Lambda^k\operatorname{Ann}(X_\tau)\otimes L_{\chi_z}$ with lift $L_{z,\tau}$.
See also \cite[Remark~3]{DGRS}. Here $L_{\chi_z}$ is the flat character line.
The logarithmic derivative has only simple poles, with residues equal to
the algebraic resonance multiplicities. Integrating shows that these
multiplicities are the zero orders of the determinant, also for complex $z$
and perturbed metrics.
The zeros inside the contour therefore determine the monic cut
polynomial, including multiplicities, and give the asserted symmetries.
The full-form characteristic polynomials are products of those in
the two adjacent horizontal degrees, so they have the same symmetries.

<!-- proposition-1: body_md -->
\label{prop:crossing}
After shrinking $\rho_*,\varepsilon$, the cut polynomials factor on
$\mathcal P$ as
\begin{align}
 P_1(s)&=(s-c)(s-t),\label{P1}\\
 P_2(s)&=(s-c)(s-t)Q(s),\qquad Q(s)=s^2-e_1s+e_2.\label{P2}
\end{align}
The coefficients $e_1,e_2$ are even and holomorphic in $z$, $C^1$ in
$\tau$, and real for real $z$. They are the sum and product of the
two remaining middle-degree roots, counted with algebraic multiplicity.
Moreover,
\begin{equation}\label{twoaxisfactor}
 e_1(z,\tau)=z^2\tau E(z,\tau),
\end{equation}
where $E$ is continuous and bounded on $\mathcal P$.
For every sufficiently small real $\theta\ne0$, the function
$\tau\mapsto t(\theta,\tau)$ has a unique zero
$\sigma_\theta\in(0,\varepsilon)$, and
\begin{equation}\label{cross}
 t(\theta,\sigma_\theta)=0,\qquad
 \frac{\sigma_\theta}{\theta^2}\longrightarrow\frac\kappa{2r}
 \quad(\theta\to0).
\end{equation}
At this crossing, $c(\theta,\sigma_\theta)<0$ and
$e_1(\theta,\sigma_\theta)=O(\theta^4)$.
Writing $V^k\subset\mathcal U^k_{\theta,\sigma_\theta}$ for the horizontal
generalized zero-resonance space, we have $(\dim V^0,\ldots,\dim V^4)=(0,1,m,1,0),$ for $ m\in\{2,3\},$
and, with all coefficients evaluated at $(\theta,\sigma_\theta)$,
\begin{equation}\label{two}
 e_2=0,\qquad m=3\Longleftrightarrow e_1=0,
 \qquad m=2\Longleftrightarrow e_1\ne0.
\end{equation}

<!-- proof-2: body_md -->
We define operators on horizontal forms by $D=d_z-\alpha_\tau\wedge L$ and $ W=d\alpha_\tau\wedge.$
They satisfy
\begin{equation}\label{Cartan}
 [B,D]=0,\qquad D^2=WB.
\end{equation}
Here $D$ is the horizontal part of $d_z$. It does not in general
square to zero. Indeed, for a horizontal form $u$,
\[
 d_zu=Du+\alpha_\tau\wedge Lu,\qquad
 0=d_z^2u=D^2u+WLu,
\]
where the terms containing $\alpha_\tau$ cancel because $D$ and $L$
commute. Thus $D^2u=0$ on zero eigenvectors of $B$.
The two-form $d\alpha_\tau$ is nondegenerate on the four-dimensional
contact plane $\ker\alpha_\tau$. Its wedge maps, called the contact
Lefschetz maps, commute with $B$ and give isomorphisms
\[
 W:\mathcal U^1\longrightarrow\mathcal U^3,\qquad
 W^2:\mathcal U^0\longrightarrow\mathcal U^4.
\]
For small real $\theta\ne0$, pullback acyclicity gives
$H^0(M;\pi^*\chi_\theta)=0$. Hence \cite[Lemma~7.4]{DGRS}
excludes a scalar resonance at zero, so $c\ne0$. For a nonzero scalar
eigenvector $f$ with $Bf=cf$, we have
\[
 D^2f=cWf\ne0,\qquad Df\ne0,\qquad B(Df)=cDf.
\]
Hence $c$ is also a degree-one root. Since the degree-one space has
dimension two and trace $c+t$, this proves \eqref{P1}, first for real
nonzero characters and then by holomorphy everywhere.
Using evenness, \eqref{taxis}, and Cauchy's formula, we define quotient
functions $C_c$ and $C_t$ by
\begin{equation}\label{ctfactors}
 c=z^2C_c,\qquad t=t(0,\tau)+z^2C_t,
 \qquad C_c(0,0)=C_t(0,0)=-\kappa/2.
\end{equation}
Here and below the quotients are holomorphic in $z$ and $C^1$ in
$\tau$, locally uniformly in $z$. For any even such function $a_0$ with
$a_0(0,\tau)=0$, and fixed $\rho_*<a<\rho$,
\begin{equation}\label{Cauchyquotient}
 \frac{a_0(z,\tau)}{z^2}
 =\frac1{2\pi i}\int_{|w|=a}
       \frac{a_0(w,\tau)}{w^2(w-z)}\,dw,\qquad |z|<a.
\end{equation}
Differentiating under the integral proves the stated regularity of the
extended quotient and its derivative with respect to the metric parameter. Shrink $\rho_*,\varepsilon$ so that $C_c$ is
nonzero on $\mathcal P$.

When $A_1$ is invertible, $D:\mathcal U^1\to\mathcal U^2$ is injective. Indeed,
\[
 Du=0\ \Longrightarrow\ D^2u=WBu=0
 \ \Longrightarrow\ Bu=0\ \Longrightarrow\ u=0.
\]
Here the second implication uses the isomorphism $W$, and the last
uses invertibility of $A_1$. Since $BD=DB$, the restriction of $B$ to
$D(\mathcal U^1)$ has characteristic polynomial $P_1$. It follows that $P_1$
divides $P_2$.
The remainder in monic polynomial division is holomorphic in $z$
and $C^1$ in $\tau$. For fixed small $\tau>0$, $t(0,\tau)\ne0$,
and \eqref{ctfactors} shows that $c(\cdot,\tau)$ is not identically zero.
Thus $A_1$ is invertible on a dense set of $z$. At $\tau=0$, the same
conclusion follows from \eqref{hyper}. The remainder vanishes, giving \eqref{P2}. The quotient
coefficients inherit the regularity and symmetries of Lemma~\ref{holom}.
Shrink $\varepsilon$ below the untwisted splitting bound $\varepsilon_0$
established in Appendix~\ref{app:inputs}.
The hyperbolic identities and the untwisted algebraic multiplicities at zero
in degrees one and two from~\cite[Theorem~3 and Corollary~4.1]{CDDP}
give, respectively,
\begin{equation}\label{qhyper}
 e_1(z,0)=0,\quad e_2(z,0)=-c(z,0)^2,
 \qquad e_1(0,\tau)=e_2(0,\tau)=0.
\end{equation}
For the hyperbolic axis, division of \eqref{hyper} by \eqref{P1} gives
\[
 Q(s,z,0)=(s-c(z,0))(s+c(z,0))=s^2-c(z,0)^2.
\]
For $z=0$ and small $\tau>0$, the horizontal generalized
zero-resonance dimensions are $(1,1,3,1,1)$.
The map $D$ sends eigenvectors for the remaining degree-one root
$t(0,\tau)\ne0$ to nonzero degree-two eigenvectors with the same eigenvalue. The ranks therefore give
\[
 P_1(s,0,\tau)=s(s-t(0,\tau)),\quad
 P_2(s,0,\tau)=s^3(s-t(0,\tau)),\quad Q(s,0,\tau)=s^2.
\]
Continuity gives the same coefficient identities at $\tau=0$.
Applying \eqref{Cauchyquotient} to $e_1$ gives
\begin{equation}\label{factors}
 e_1=z^2A_e,\qquad A_e(z,0)=0.
\end{equation}
The fundamental theorem of calculus, using $A_e(z,0)=0$, gives the
two-axis factorization \eqref{twoaxisfactor} of~\cite{Lawton}, with
$E(z,\tau)=\int_0^1\partial_\tau A_e(z,v\tau)\,dv$ continuous and
bounded on $\mathcal P$.

For real $\theta$ and $\tau\geq0$, equations \eqref{taxis} and
\eqref{ctfactors} give
\begin{equation}\label{crossingbalance}
 t(\theta,\tau)=r\tau-\frac\kappa2\theta^2
                   +o(\tau+\theta^2)
\end{equation}
as $(\theta,\tau)\to(0,0)$.
By Lemma~\ref{holom}, shrink $\rho_*,\varepsilon$ so that
$\partial_\tau t(\theta,\tau)>r/2$ for real $(\theta,\tau)\in\mathcal P$.
Thus $t(0,\tau)\geq r\tau/2$.
Choose $K_*>0$ with $|C_t|\leq K_*$ on $\mathcal P$
and restrict $\theta$ so that $4K_*\theta^2/r<\varepsilon$. Hence
\[
 t(\theta,0)=c_0(\theta)<0,
 \qquad t(\theta,4K_*\theta^2/r)\geq K_*\theta^2>0.
\]
Continuity and strict monotonicity give a unique zero $\sigma_\theta$ of
$\tau\mapsto t(\theta,\tau)$, with
$0<\sigma_\theta\leq4K_*\theta^2/r$.
Using this bound in~\eqref{crossingbalance} gives $0=r\frac{\sigma_\theta}{\theta^2}
       -\frac\kappa2+o(1).$ This proves \eqref{cross}.
Equation \eqref{twoaxisfactor} now gives
$e_1(\theta,\sigma_\theta)=O(\theta^4)$.
The scalar root $c$ stays negative, since $c=\theta^2C_c$ and
$C_c\to-\kappa/2$.

Fix a crossing and suppress the parameters. Since $c\ne0$, we have
$V^0=V^4=0$, and the degree-one zero is simple. Lefschetz gives
$\dim V^1=\dim V^3=1$ and $L|_{V^1}=L|_{V^3}=0$.
Set $m=\dim V^2$.
We take $\alpha=\alpha_{\sigma_\theta}$ and set $V^{-1}=V^5=0$.
The full generalized zero-resonance complex has spaces
\[
 \mathcal V^k=V^k\oplus\alpha\wedge V^{k-1}\qquad(0\leq k\leq5).
\]
For $u\in V^k$ and $v\in V^{k-1}$, its differential is
\begin{equation}\label{xp:full-zero-complex}
 d_\theta(u+\alpha\wedge v)
   =Du+Wv+\alpha\wedge(Lu-Dv).
\end{equation}
The resonant de Rham theorem~\cite[Theorem 2.1]{DR} identifies the
cohomology of this full generalized zero-resonance complex with the
twisted de Rham cohomology of $M$. Pullback acyclicity therefore
makes the full complex exact. Exactness does not require its spaces
to vanish, nor does it imply that $L$ vanishes on generalized zero
states. We write $D_k$
for $D|_{V^k}$. Since $V^0=V^4=0$, its end terms are
$\mathcal V^1=V^1$ and $\mathcal V^4=\alpha\wedge V^3$. For
$u\in V^3$ and $v\in V^2$, \eqref{xp:full-zero-complex} reduces to $d_\theta(u+\alpha\wedge v)=-\alpha\wedge D_2v.$
In degree one, $L|_{V^1}=0$ gives
\[
 d_\theta|_{\mathcal V^1}=D_1,\qquad
 \ker D_1=\operatorname{im}(d_\theta|_{\mathcal V^0})=0.
\]
In degree four, $\mathcal V^5=0$ gives
\[
 \alpha\wedge D_2(V^2)=\mathcal V^4=\alpha\wedge V^3,
 \qquad D_2(V^2)=V^3.
\]
Thus $D_1$ is injective and $D_2$ is surjective. Also
\[
 D_2D_1=WB|_{V^1}=0.
\]
Consequently
\[
 1=\dim D_1(V^1)\leq\dim\ker D_2=m-1.
\]
The nonzero root $c$ occupies at least one dimension of the rank-four
middle cut. Therefore $2\leq m\leq3.$
Since $c\ne0$ and $t=0$, the zero multiplicities satisfy
\[
 m=\operatorname{ord}_{s=0}P_2
   =1+\operatorname{ord}_{s=0}Q.
\]
It follows that $Q(0)=e_2=0$, so $Q(s)=s(s-e_1)$, proving \eqref{two}.

\label{cutvariation}
Fix small real $\theta\ne0$. Let $\Pi_{\theta,\tau}$ be any full
resonance cut whose fixed contour remains free of resonances throughout
the parameter interval, and set $C^\bullet_{\theta,\tau}=\operatorname{ran}\Pi_{\theta,\tau}$.
We write $\mathcal T_{\rm fin}$ for its contact-chirality torsion
\cite[Section~3.3]{CD} and define
\[
 H(\theta,\tau)=\operatorname{str}_{C^\bullet_{\theta,\tau}}
       (\Pi_{\theta,\tau}\alpha_\tau\wedge\iota_{\dot X_\tau}).
\]
Then
\[
 \partial_\tau\log\mathcal T_{\rm fin}(\tau)=-H(\theta,\tau).
\]
Nonzero resonances inside the cut are allowed.
For the cut of Section~\ref{sec:spectral}, we also have
\begin{equation}\label{Gvariation}
 \frac{G(0,\theta,\tau)}{G(0,\theta,0)}
 =\exp\left(\int_0^\tau H(\theta,v)\,dv\right).
\end{equation}
For this cut, the finite trace $H$ extends continuously and boundedly
to $\theta=0$. In particular the ratio tends to one uniformly as
$\tau\to0$ and $\theta\to0$ through nonzero real characters.

We fix $\theta$ and write
$C_\tau=C^\bullet_{\theta,\tau}$ and
$U^k_\tau=C^k_\tau\cap\ker\iota_{X_\tau}$.
We suppress metric subscripts and use Chaubet--Dang's convention
$v\wedge\alpha=(-1)^{k-1}\alpha\wedge v$ for horizontal $(k-1)$-forms.
For a horizontal form $u$, the contact splitting and Lefschetz map are
\[
 C^k_\tau=U^{k-1}_\tau\wedge\alpha_\tau\oplus U^k_\tau,
 \qquad \ell_\tau u=u\wedge d\alpha_\tau,
 \qquad U^{-1}_\tau=U^5_\tau=0.
\]
For $k=0,1,2$, $f\in U^{k-1}$, and $g\in U^k$, we define the contact
chirality $\Gamma$, a degree-reversing involution, by
\begin{equation}\label{xp:contact-chirality}
 \Gamma(f\wedge\alpha+g)
 =\ell^{2-k}g\wedge\alpha+\ell^{3-k}f.
\end{equation}
In higher degrees it is the inverse. Thus
\[
 \Gamma(C^k_\tau)=C^{5-k}_\tau,\qquad\Gamma^2=\operatorname{Id}.
\]
The powers of $\ell$ denote the corresponding Lefschetz maps.
The inverses used below are restricted to the ranges of these maps.

The generalized zero-resonance part is acyclic by
\cite[Theorem~2.1]{DR}. On the sum of the
nonzero generalized resonance spaces, $L$ is invertible and
\[
 d_\theta(\iota_XL^{-1})+(\iota_XL^{-1})d_\theta=\operatorname{Id}.
\]
Thus the whole cut $C_\tau$ is acyclic. The spectral projector commutes with $d_\theta$,
$\iota_X$, wedging by $\alpha$, and the Lefschetz maps.
Thus $\Gamma$ preserves the full resonance space.

Write $\det E$ for the top exterior power of a finite-dimensional space.
The determinant line of the complex is
\[
 \det C_\tau=\bigotimes_{k=0}^5(\det C^k_\tau)^{(-1)^k},
\]

degrees $0,1,2$ and their $\Gamma$-partners in degrees $5,4,3$.
Their alternating tensor product, with the sign of~\cite[Section~3.3]{CD},
is choice-independent: paired rescalings cancel. By acyclicity, the canonical torsion
isomorphism takes values in $\det H^\bullet(C_\tau)=\mathbb C$.
It sends this tensor to the nonzero scalar $\mathcal T_{\rm fin}(\tau)$.
The differentiability of the spectral projectors established in
\cite[Proposition~7.2 and Appendix~B]{CD} implies that the finite-dimensional
matrices depend differentiably on $\tau$. Here the projectors act from a
stronger anisotropic space to a weaker one.

 \Pi_*=\Pi_{\theta,\tau_*},\qquad
 \dot\Pi_*=(\partial_\tau\Pi_{\theta,\tau})|_{\tau=\tau_*}.

On the fixed space $C=C_{\tau_*}$, set

 S_\tau=\Pi_{\theta,\tau}|_C,
 \qquad \mathsf F_\tau=\Pi_*S_\tau:C\to C.

They are invertible near $\tau_*$. Since the projections commute
with $d_\theta$, the differential in the fixed coordinates is unchanged $S_\tau^{-1}d_\theta S_\tau=d_\theta|_C.$
We define the chirality in these coordinates by

 \widetilde\Gamma_\tau
 =S_\tau^{-1}\Gamma_\tau S_\tau
 =\mathsf F_\tau^{-1}\Pi_*\Gamma_\tau S_\tau.

 \Pi_*\dot\Pi_*\Pi_*=0,
 \qquad \mathsf F_{\tau_*}=\operatorname{Id},\qquad\dot{\mathsf F}_{\tau_*}=0.

Using $\Pi_*\Gamma_{\tau_*}=\Gamma_{\tau_*}\Pi_*$, we obtain

 \dot{\widetilde\Gamma}_{\tau_*}
 =\Pi_*\dot\Gamma_{\tau_*}|_C
   +\Pi_*\Gamma_{\tau_*}\dot\Pi_*|_C
 =\Pi_*\dot\Gamma_{\tau_*}|_C.

   =\sum_{k=0}^{2}(-1)^{k+1}\operatorname{tr}_{C^k}(\Pi\Gamma\dot\Gamma).
 \end{aligned}
\end{equation}

 H=-h_0+h_1-h_2+h_3-h_4=-2h_0+2h_1-h_2.

 h_3=h_1,\qquad h_4=h_0.

 \mathcal T_{\rm dyn}(\theta,\tau)

 Z_{\rm hyp}(\theta):=Z(0,\theta,0)
       =-G(0,\theta,0)c_0(\theta)^2\ne0.

We fix a crossing $(\theta,\sigma_\theta)$ from Proposition~\ref{prop:crossing}.

 Z(s,\theta,\sigma_\theta)
 &=G(s,\theta,\sigma_\theta)(s-c)(s-e_1),\\

 \end{aligned}

 =\frac{\sigma_\theta}{C_c(\theta,0)}

 =O(\theta^2).

 \longrightarrow\frac\kappa{2r}\frac{E(0,0)}{-\kappa/2}
