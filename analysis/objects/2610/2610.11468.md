<!-- introtheorem-1: body_md -->
Let $G/F$ be a smooth algebraic group of dimension $d$. If $d>0$, choose an analytic chart $z:U\xrightarrow{\sim}V\subset F^d$ on an open neighborhood of the identity $e$, with $z(e)=0$. There are integers $\kappa_1,\ldots,\kappa_{n-1}\ge0$ and $\kappa_n\ge1$ such that all balls $\mathfrak{p}_{i_1+\kappa_1,\ldots,i_n+\kappa_n}^{\,d}$ with $(i_1,\ldots,i_n)\in\mathcal I$ lie in $V$, and
\[
H^z_{i_1,\ldots,i_n}:=z^{-1}(\mathfrak{p}_{i_1+\kappa_1,\ldots,i_n+\kappa_n}^{\,d}),
\qquad (i_1,\ldots,i_n)\in\mathcal I,
\]
are subgroups. Their left cosets are called \emph{congruence cylinders}. The ring
\[
\mathcal{R}_z(G(F))=\operatorname{ring}\{gH^z_{i_1,\ldots,i_n}:g\in G(F),\ (i_1,\ldots,i_n)\in\mathcal I\}
\]
has a unique finitely additive left-invariant measure $\mu^0_{G,z}:\mathcal{R}_z(G(F))\to\mathbb R_n$ with
\[
\boxed{\ \mu^0_{G,z}(gH^z_{i_1,\ldots,i_n})=q^{-di_1}\prod_{\ell=2}^nX_\ell^{di_\ell}.\ }
\]
Every nonempty measurable set has strictly positive measure. For $d=0$, counting measure on finite subsets has these properties.

<!-- definition-1: body_md -->
\label{def:vector-valuation}
For $d\ge1$ and $x=(x_1,\ldots,x_d)\in F^d$, define
\[
v(x)=\min_{1\le s\le d}v(x_s)\in\mathbb{Z}^n\cup\{+\infty\},
\qquad v(0)=+\infty.
\]
The minimum is taken in the right lexicographic order.
<!-- end definition-1 -->
<!-- proposition-1: body_md -->
\label{prop:ideal}
The valuation ideals satisfy $\mathfrak{p}_{i_1,\ldots,i_n}\mathfrak{p}_{j_1,\ldots,j_n}\subseteq\mathfrak{p}_{i_1+j_1,\ldots,i_n+j_n}$. For $(j_1,\ldots,j_n)\ge(i_1,\ldots,i_n)$,
\begin{equation}\label{eq:idealindex}
[\mathfrak{p}_{i_1,\ldots,i_n}:\mathfrak{p}_{j_1,\ldots,j_n}]=
\begin{cases}
q^{j_1-i_1},&(j_2,\ldots,j_n)=(i_2,\ldots,i_n),\\
\infty,&(j_2,\ldots,j_n)>(i_2,\ldots,i_n).
\end{cases}
\end{equation}
In particular, $\mathfrak{p}_{i_1,\ldots,i_n}/\mathfrak{p}_{i_1+1,i_2,\ldots,i_n}\simeq(k,+)$ and $[\mathfrak{p}_{i_1,\ldots,i_n}:\mathfrak{p}_{i_1+m,i_2,\ldots,i_n}]=q^m$ for $m\ge0$.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
For $a\in\mathfrak{p}_{i_1,\ldots,i_n}$ and $b\in\mathfrak{p}_{j_1,\ldots,j_n}$,
\[
v(ab)=v(a)+v(b)\ge(i_1+j_1,\ldots,i_n+j_n).
\]
At a fixed level $(i_2,\ldots,i_n)$, multiplication by $(t_1^{i_1}\cdots t_n^{i_n})^{-1}$ and successive reduction give
\[
\mathfrak{p}_{i_1,\ldots,i_n}/\mathfrak{p}_{i_1+m,i_2,\ldots,i_n}
\simeq\mathcal{O}/t_1^m\mathcal{O}
\simeq\mathcal{O}_{F_1}/\pi_1^m\mathcal{O}_{F_1}.
\]
The last quotient has $q^m$ elements. At a strictly larger level,
\[
\begin{aligned}
(j_2,\ldots,j_n)>(i_2,\ldots,i_n)
&\ \Longrightarrow\
\mathfrak{p}_{j_1,\ldots,j_n}\subseteq\mathfrak{p}_{i_1+m,i_2,\ldots,i_n}
\quad(m\ge0),\\
[\mathfrak{p}_{i_1,\ldots,i_n}:\mathfrak{p}_{j_1,\ldots,j_n}]
&\ge q^m\quad(m\ge0)
\ \Longrightarrow\
[\mathfrak{p}_{i_1,\ldots,i_n}:\mathfrak{p}_{j_1,\ldots,j_n}]=\infty.
\end{aligned}
\]

<!-- lemma-1: body_md -->
\label{lem:estimate}
There are integers $\kappa_1,\ldots,\kappa_{n-1}\ge0$ and $\kappa_n\ge1$ such that, for every pair of depths satisfying
\[
(i_1,\ldots,i_n)\ge(\kappa_1,\ldots,\kappa_n),\qquad
(j_1,\ldots,j_n)\ge(\kappa_1,\ldots,\kappa_n),
\]
the coordinate expressions are defined on the indicated balls and satisfy
\begin{equation}\label{eq:estimate}
\begin{aligned}
R(\mathfrak{p}_{i_1,\ldots,i_n}^{\,d},\mathfrak{p}_{j_1,\ldots,j_n}^{\,d})&\subseteq\mathfrak{p}_{i_1+j_1-\kappa_1,\ldots,i_n+j_n-\kappa_n+1}^{\,d},\\
S(\mathfrak{p}_{i_1,\ldots,i_n}^{\,d})&\subseteq\mathfrak{p}_{2i_1-\kappa_1,\ldots,2i_n-\kappa_n+1}^{\,d},\\
C(\mathfrak{p}_{i_1,\ldots,i_n}^{\,d},\mathfrak{p}_{j_1,\ldots,j_n}^{\,d})&\subseteq\mathfrak{p}_{i_1+j_1-\kappa_1,\ldots,i_n+j_n-\kappa_n+1}^{\,d}.
\end{aligned}
\end{equation}

All valuation inequalities use the right lexicographic order. Coordinatewise,
\[
x\in\mathfrak{p}_{\alpha_1,\ldots,\alpha_n}^{\,d}
\ \Longleftrightarrow\
v(x_s)\ge(\alpha_1,\ldots,\alpha_n)\ (1\le s\le d)
\ \Longleftrightarrow\
v(x)\ge(\alpha_1,\ldots,\alpha_n).
\]
Shrink the chart so that all three coordinate maps are analytic. The group identities give
\[
\begin{aligned}
\Phi(x,0)=x,\quad \Phi(0,y)=y
&\ \Longrightarrow\ R(x,0)=R(0,y)=0,\\
[g,e]=[e,h]=e
&\ \Longrightarrow\ C(x,0)=C(0,y)=0.
\end{aligned}
\]
For $I=(I_1,\ldots,I_d)\in\mathbb{Z}_{\ge0}^d$, write $x^I=\prod_r x_r^{I_r}$ and $|I|=\sum_r I_r$. Uniqueness of power-series coefficients yields
\[
R_s=\sum_{I,J}a_{s,I,J}x^Iy^J,\quad
C_s=\sum_{I,J}c_{s,I,J}x^Iy^J
\quad\Longrightarrow\quad
\begin{cases}
a_{s,I,0}=a_{s,0,J}=0,\\
c_{s,I,0}=c_{s,0,J}=0.
\end{cases}
\]

For inversion, $\iota(0)=0$ and $S_s(x)=\iota_s(x)+x_s$. With $\delta_{sr}$ the Kronecker delta,
\[
\frac{\partial\Phi_s}{\partial x_r}(0,0)
=\frac{\partial\Phi_s}{\partial y_r}(0,0)=\delta_{sr}.
\]
Differentiating $\Phi_s(x,\iota(x))=0$ componentwise gives
\[
\begin{aligned}
0&=\frac{\partial\Phi_s}{\partial x_r}(0,0)
+\sum_{t=1}^d\frac{\partial\Phi_s}{\partial y_t}(0,0)
 \frac{\partial\iota_t}{\partial x_r}(0)\\
&=\delta_{sr}+\frac{\partial\iota_s}{\partial x_r}(0),\\
\frac{\partial S_s}{\partial x_r}(0)
&=\frac{\partial\iota_s}{\partial x_r}(0)+\delta_{sr}=0.
\end{aligned}
\]
Thus, including the vanishing constant terms,
\[
\begin{aligned}
R_s(x,y)&=\sum_{|I|,|J|\ge1}a_{s,I,J}x^Iy^J,\\
C_s(x,y)&=\sum_{|I|,|J|\ge1}c_{s,I,J}x^Iy^J,\\
S_s(x)&=\sum_{|K|\ge2}b_{s,K}x^K
\qquad(1\le s\le d).
\end{aligned}
\]

For the coefficient estimates below, $h_\gamma$ denotes a coefficient of one of these component series, according to
\[
(h_\gamma,\gamma,m)=
\begin{cases}
(a_{s,I,J},(I,J),|I|+|J|),&\text{for }R_s,\\
(c_{s,I,J},(I,J),|I|+|J|),&\text{for }C_s,\\
(b_{s,K},K,|K|),&\text{for }S_s.
\end{cases}
\]
Thus $m=|\gamma|\ge2$ is the total degree. The component index $s$ and the choice of series are suppressed in the notation $h_\gamma$.

Choose $\rho_0\in F^\times$ so that $D_0=(\rho_0\mathcal{O}_F)^d\subset V$, the series for $R$ and $C$ converge on $D_0\times D_0$, and the series for $S$ converge on $D_0$. Taking these closed polydiscs inside the respective convergence domains gives a common coefficient bound
\[
\exists B\ge0\quad
v_n(h_\gamma\rho_0^m)\ge-B
\qquad\text{for all component coefficients}.
\]

Fix $L\ge1$ and set $\rho=t_n^L\rho_0$, $D=(\rho\mathcal{O}_F)^d\subset D_0$. Then
\[
\begin{aligned}
v_n(h_\gamma\rho^m)&\ge-B+mL,\\
v_n(h_\gamma\rho^{m-2})&\ge-B+mL-2v_n(\rho)
\xrightarrow[m\to\infty]{}+\infty.
\end{aligned}
\]
Let $\mathcal B$ consist of the nonzero members of
\[
\left\{
a_{s,I,J}\rho^{|I|+|J|-2},\
c_{s,I,J}\rho^{|I|+|J|-2},\
b_{s,K}\rho^{|K|-2}
\right\}.
\]
Choose $M_0\ge2$ with $M_0L>B+2v_n(\rho)$. The preceding bound implies
\[
\begin{aligned}
\mathcal E&:=\{\beta\in\mathcal B:v_n(\beta)\le0\},\\
\mathcal E&\subseteq\{\text{normalized coefficients of degree at most }M_0\},\\
\#\mathcal E&<\infty.
\end{aligned}
\]
Define the component constants by
\[
\begin{aligned}
\kappa_s
&=\max\bigl(\{0\}\cup\{-v_s(\beta):\beta\in\mathcal E\}\bigr)
\qquad(1\le s<n),\\
\kappa_n
&=1+\max\bigl(\{0,v_n(\rho)\}\cup\{-v_n(\beta):\beta\in\mathcal E\}\bigr).
\end{aligned}
\]
These maxima exist even when $\mathcal E=\varnothing$, and
\[
\kappa_s\ge0\quad(1\le s<n),\qquad
\kappa_n\ge1,\qquad \kappa_n>v_n(\rho).
\]
The extra $1$ in $\kappa_n$ leaves one outer valuation step in the estimates below. In particular,
\[
\begin{cases}
v(\beta)\ge(-\kappa_1,\ldots,-\kappa_{n-1},1-\kappa_n),&\beta\in\mathcal E,\\
v(\beta)>(0,\ldots,0)\ge(-\kappa_1,\ldots,-\kappa_{n-1},1-\kappa_n),
&\beta\in\mathcal B\setminus\mathcal E.
\end{cases}
\]
Since $v(0)=+\infty$, all coefficients satisfy
\[
\begin{aligned}
v(a_{s,I,J}\rho^{|I|+|J|-2})&\ge(-\kappa_1,\ldots,-\kappa_{n-1},1-\kappa_n),\\
v(c_{s,I,J}\rho^{|I|+|J|-2})&\ge(-\kappa_1,\ldots,-\kappa_{n-1},1-\kappa_n),\\
v(b_{s,K}\rho^{|K|-2})&\ge(-\kappa_1,\ldots,-\kappa_{n-1},1-\kappa_n).
\end{aligned}
\]
These constants depend only on the component series and $\rho$.

Suppose $v(x),v(y)\ge v(\rho)$ and put $u=x/\rho$, $w=y/\rho$ coordinatewise. Then
\[
\begin{aligned}
v(u_s)&=v(x_s)-v(\rho)\ge0,\qquad v(w_s)\ge0,\\
u,w&\in\mathcal{O}^d,\qquad
x,y\in(\rho\mathcal{O})^d\subseteq D.
\end{aligned}
\]
Thus the series converge, and every monomial in $u,w$ has nonnegative valuation.

For $|I|,|J|\ge1$, choose $p,r$ with $I_p,J_r\ge1$. With $e_p,e_r$ the standard multiindices,
\[
x^I=x_p\rho^{|I|-1}u^{I-e_p},\qquad
y^J=y_r\rho^{|J|-1}w^{J-e_r}.
\]
Consequently,
\[
\begin{aligned}
a_{s,I,J}x^Iy^J
&=(a_{s,I,J}\rho^{|I|+|J|-2})x_py_ru^{I-e_p}w^{J-e_r},\\
v(a_{s,I,J}x^Iy^J)
&=v(a_{s,I,J}\rho^{|I|+|J|-2})+v(x_p)+v(y_r)\\
&\qquad+v(u^{I-e_p})+v(w^{J-e_r})\\
&\ge v(x)+v(y)-(\kappa_1,\ldots,\kappa_{n-1},\kappa_n-1).
\end{aligned}
\]
The identical factorization for $c_{s,I,J}$ gives
\[
v(c_{s,I,J}x^Iy^J)\ge v(x)+v(y)-(\kappa_1,\ldots,\kappa_{n-1},\kappa_n-1).
\]

For $|K|\ge2$, choose $p,r$ so that
\[
K-e_p-e_r\in\mathbb{Z}_{\ge0}^d:
\qquad
\begin{cases}
r=p,&K_p\ge2,\\
r\ne p,\ K_p,K_r\ge1,&\text{otherwise}.
\end{cases}
\]
Then
\[
\begin{aligned}
b_{s,K}x^K
&=(b_{s,K}\rho^{|K|-2})x_px_ru^{K-e_p-e_r},\\
v(b_{s,K}x^K)
&\ge v(x_p)+v(x_r)-(\kappa_1,\ldots,\kappa_{n-1},\kappa_n-1)\\
&\ge2v(x)-(\kappa_1,\ldots,\kappa_{n-1},\kappa_n-1).
\end{aligned}
\]
The inequalities include zero terms by the convention $v(0)=+\infty$.

For the depths in the statement, $\kappa_n>v_n(\rho)$ gives
\[
\begin{aligned}
(i_1,\ldots,i_n)&\ge(\kappa_1,\ldots,\kappa_n)>v(\rho),\\
(j_1,\ldots,j_n)&\ge(\kappa_1,\ldots,\kappa_n)>v(\rho).
\end{aligned}
\]
Hence, for $x\in\mathfrak{p}_{i_1,\ldots,i_n}^{\,d}$ and $y\in\mathfrak{p}_{j_1,\ldots,j_n}^{\,d}$,
\[
\begin{aligned}
v(a_{s,I,J}x^Iy^J),\ v(c_{s,I,J}x^Iy^J)
&\ge(i_1+j_1-\kappa_1,\ldots,i_n+j_n-\kappa_n+1),\\
v(b_{s,K}x^K)&\ge(2i_1-\kappa_1,\ldots,2i_n-\kappa_n+1).
\end{aligned}
\]
The same threshold also gives
\[
\begin{aligned}
&(i_1+j_1-\kappa_1,\ldots,i_n+j_n-\kappa_n+1)\\
&\hspace{2em}>\max\{(i_1,\ldots,i_n),(j_1,\ldots,j_n)\},\\
&(2i_1-\kappa_1,\ldots,2i_n-\kappa_n+1)>(i_1,\ldots,i_n).
\end{aligned}
\]

It remains to take limits. Each valuation ideal is additive:
\[
z_\ell\in\mathfrak{p}_{\alpha_1,\ldots,\alpha_n}\ (1\le\ell\le M)
\ \Longrightarrow\
v\left(\sum_{\ell=1}^M z_\ell\right)
\ge\min_\ell v(z_\ell)\ge(\alpha_1,\ldots,\alpha_n).
\]
It is also closed in the outer topology. Indeed, for the open subgroup
$H=t_n^{\alpha_n+1}\mathcal{O}_F$,
\[
H\subseteq\mathfrak{p}_{\alpha_1,\ldots,\alpha_n},
\qquad
F\setminus\mathfrak{p}_{\alpha_1,\ldots,\alpha_n}
=\bigcup_{a\notin\mathfrak{p}_{\alpha_1,\ldots,\alpha_n}}(a+H),
\]
so the complement is open.

Let $R_{s,\le M},C_{s,\le M},S_{s,\le M}$ denote the corresponding truncations by total degree. The termwise bounds yield
\[
\begin{aligned}
R_{s,\le M}(x,y),C_{s,\le M}(x,y)
&\in\mathfrak{p}_{i_1+j_1-\kappa_1,\ldots,i_n+j_n-\kappa_n+1},\\
S_{s,\le M}(x)&\in\mathfrak{p}_{2i_1-\kappa_1,\ldots,2i_n-\kappa_n+1}.
\end{aligned}
\]
Taking outer limits in these closed ideals gives
\[
\begin{aligned}
R_s(x,y),C_s(x,y)
&\in\mathfrak{p}_{i_1+j_1-\kappa_1,\ldots,i_n+j_n-\kappa_n+1},\\
S_s(x)&\in\mathfrak{p}_{2i_1-\kappa_1,\ldots,2i_n-\kappa_n+1}
\qquad(1\le s\le d).
\end{aligned}
\]
Combining the componentwise estimates for all $1\le s\le d$ proves the three inclusions in \eqref{eq:estimate}.
Increasing any component constant weakens the coefficient lower bounds and raises the depth threshold in the right lexicographic order, so the conclusions remain valid.

<!-- proposition-2: body_md -->
\label{prop:deep}
Choose $\kappa_1,\ldots,\kappa_n$ as in Lemma~\ref{lem:estimate}. For the depths
\[
(i_1,\ldots,i_n)\ge(\kappa_1,\ldots,\kappa_n),
\]
the sets $K_{i_1,\ldots,i_n}=z^{-1}(\mathfrak{p}_{i_1,\ldots,i_n}^{\,d})$ are nested subgroups with intersection $\{e\}$. For every such depth and every integer $m\ge0$,
\begin{equation}\label{eq:deepindex}
K_{i_1,\ldots,i_n}/K_{i_1+1,i_2,\ldots,i_n}\simeq(k^d,+),\qquad
[K_{i_1,\ldots,i_n}:K_{i_1+m,i_2,\ldots,i_n}]=q^{dm}.
\end{equation}
Each $K_{i_1+m,i_2,\ldots,i_n}$ is normal in $K_{i_1,\ldots,i_n}$. For any two depths in this family,
\begin{equation}\label{eq:commutator}
[K_{i_1,\ldots,i_n},K_{j_1,\ldots,j_n}]\subseteq K_{i_1+j_1-\kappa_1,\ldots,i_n+j_n-\kappa_n+1}.
\end{equation}
For such depths with $(j_1,\ldots,j_n)\ge(i_1,\ldots,i_n)$ and $(j_2,\ldots,j_n)>(i_2,\ldots,i_n)$, the index $[K_{i_1,\ldots,i_n}:K_{j_1,\ldots,j_n}]$ is infinite.
<!-- end proposition-2 -->
<!-- proof-3: body_md -->
For an allowed depth $(i_1,\ldots,i_n)$ and every $m\ge0$, right lexicographic order and $n\ge2$ give
\[
\begin{aligned}
&(2i_1-\kappa_1,\ldots,2i_n-\kappa_n+1)\\
&\hspace{2em}\ge(i_1,\ldots,i_{n-1},i_n+1)
>(i_1+m,i_2,\ldots,i_n).
\end{aligned}
\]
Consequently,
\[
\mathfrak{p}_{2i_1-\kappa_1,\ldots,2i_n-\kappa_n+1}^{\,d}
\subseteq\mathfrak{p}_{i_1+m,i_2,\ldots,i_n}^{\,d}
\subseteq\mathfrak{p}_{i_1,\ldots,i_n}^{\,d}.
\]
For $g,h\in K_{i_1,\ldots,i_n}$, put $x=z(g)$ and $y=z(h)$. Lemma~\ref{lem:estimate} yields
\[
\begin{aligned}
z(e)&=0\in\mathfrak{p}_{i_1,\ldots,i_n}^{\,d},\\
R(x,y),S(x)&\in\mathfrak{p}_{i_1+m,i_2,\ldots,i_n}^{\,d}\qquad(m\ge0),\\
z(gh)&=x+y+R(x,y)\in\mathfrak{p}_{i_1,\ldots,i_n}^{\,d},\\
z(g^{-1})&=-x+S(x)\in\mathfrak{p}_{i_1,\ldots,i_n}^{\,d}.
\end{aligned}
\]
Thus every $K_{i_1,\ldots,i_n}$ is a subgroup. Nesting and separatedness follow from
\[
\begin{aligned}
(j_1,\ldots,j_n)\ge(i_1,\ldots,i_n)
&\ \Longrightarrow\
\mathfrak{p}_{j_1,\ldots,j_n}^{\,d}\subseteq\mathfrak{p}_{i_1,\ldots,i_n}^{\,d}\\
&\ \Longrightarrow\
K_{j_1,\ldots,j_n}\subseteq K_{i_1,\ldots,i_n},\\
\bigcap_{r\ge0}\mathfrak{p}_{\kappa_1,\ldots,\kappa_{n-1},\kappa_n+r}^{\,d}
&\subseteq\bigcap_{r\ge0}(t_n^{\kappa_n+r}\mathcal{O}_F)^d=\{0\}.
\end{aligned}
\]
Since $z(e)=0$ and $z$ is injective, the intersection of all these subgroups is $\{e\}$.

Fix an allowed depth and $m\ge0$. Increasing the first component preserves the threshold:
\[
(i_1+m,i_2,\ldots,i_n)\ge(i_1,\ldots,i_n)
\ge(\kappa_1,\ldots,\kappa_n).
\]
Define
\[
\begin{aligned}
\theta_m:K_{i_1,\ldots,i_n}
&\longrightarrow
\mathfrak{p}_{i_1,\ldots,i_n}^{\,d}/\mathfrak{p}_{i_1+m,i_2,\ldots,i_n}^{\,d},\\
g&\longmapsto z(g)+\mathfrak{p}_{i_1+m,i_2,\ldots,i_n}^{\,d}.
\end{aligned}
\]
The remainder inclusion above holds for this $m$, so
\[
\begin{aligned}
\theta_m(gh)
&=z(g)+z(h)+R(z(g),z(h))
  +\mathfrak{p}_{i_1+m,i_2,\ldots,i_n}^{\,d}\\
&=\theta_m(g)+\theta_m(h).
\end{aligned}
\]
Every target class has a representative $x\in\mathfrak{p}_{i_1,\ldots,i_n}^{\,d}$, and $g=z^{-1}(x)$ belongs to $K_{i_1,\ldots,i_n}$ with $\theta_m(g)=x+\mathfrak{p}_{i_1+m,i_2,\ldots,i_n}^{\,d}$. Thus $\theta_m$ is surjective. Its kernel is determined by
\[
g\in\ker\theta_m
\ \Longleftrightarrow\ z(g)\in\mathfrak{p}_{i_1+m,i_2,\ldots,i_n}^{\,d}
\ \Longleftrightarrow\ g\in K_{i_1+m,i_2,\ldots,i_n}.
\]
Hence $K_{i_1+m,i_2,\ldots,i_n}$ is normal as the kernel of a group homomorphism. The first isomorphism theorem and the successive reductions in Proposition~\ref{prop:ideal} give
\[
\begin{aligned}
K_{i_1,\ldots,i_n}/K_{i_1+m,i_2,\ldots,i_n}
&\simeq\bigl(\mathfrak{p}_{i_1,\ldots,i_n}/\mathfrak{p}_{i_1+m,i_2,\ldots,i_n}\bigr)^d\\
&\simeq\bigl(\mathcal{O}_{F_1}/\pi_1^m\mathcal{O}_{F_1}\bigr)^d.
\end{aligned}
\]
For $m=1$ the last group is $(k^d,+)$; for every $m\ge0$ it has $(q^m)^d=q^{dm}$ elements. This proves \eqref{eq:deepindex} and all the asserted normality statements. For $m=0$, both quotients are trivial and the index is $1$.

For $g\in K_{i_1,\ldots,i_n}$ and $h\in K_{j_1,\ldots,j_n}$, Lemma~\ref{lem:estimate} gives
\[
z([g,h])=C(z(g),z(h))
\in\mathfrak{p}_{i_1+j_1-\kappa_1,\ldots,i_n+j_n-\kappa_n+1}^{\,d}.
\]
The target depth remains above the threshold, since
\[
\begin{aligned}
&(i_1+j_1-\kappa_1,\ldots,i_n+j_n-\kappa_n+1)\\
&\hspace{2em}\ge(\kappa_1,\ldots,\kappa_{n-1},\kappa_n+1)
>(\kappa_1,\ldots,\kappa_n).
\end{aligned}
\]
Thus the target is a subgroup, and
\[
\langle[g,h]:g\in K_{i_1,\ldots,i_n},\
h\in K_{j_1,\ldots,j_n}\rangle
\subseteq K_{i_1+j_1-\kappa_1,\ldots,i_n+j_n-\kappa_n+1},
\]
which is \eqref{eq:commutator}.

Finally, if $(j_2,\ldots,j_n)>(i_2,\ldots,i_n)$, then for every $m\ge0$,
\[
(j_1,\ldots,j_n)>(i_1+m,i_2,\ldots,i_n),
\qquad
K_{j_1,\ldots,j_n}\subseteq K_{i_1+m,i_2,\ldots,i_n}.
\]
The induced surjection of left-coset sets gives
\[
[K_{i_1,\ldots,i_n}:K_{j_1,\ldots,j_n}]\ge q^{dm}
\quad(m\ge0).
\]
Since $q>1$ and $d>0$, the index is infinite.

<!-- proposition-3: body_md -->
\label{prop:extension}
Assume that the following two conditions hold.
\begin{enumerate}
\item For every pair of depths
$(i_1,\ldots,i_n),(j_1,\ldots,j_n)\in J$
with $(j_1,\ldots,j_n)\ge(i_1,\ldots,i_n)$,
\[
[P_{i_1,\ldots,i_n}:P_{j_1,\ldots,j_n}]<\infty
\quad\Longleftrightarrow\quad
(j_2,\ldots,j_n)=(i_2,\ldots,i_n).
\]
\item For every pair of depths $(i_1,i_2,\ldots,i_n),(j_1,i_2,\ldots,i_n)\in J$ with $i_1\le j_1$,
\begin{equation}\label{eq:compatibility}
m_B(U_{i_1})=[P_{i_1,\ldots,i_n}:P_{j_1,i_2,\ldots,i_n}]\,m_B(U_{j_1}).
\end{equation}
\end{enumerate}
For formal variables $Y_2,\ldots,Y_n$, the assignment
\begin{equation}\label{eq:Yweight}
\nu(hP_{i_1,\ldots,i_n})=m_B(U_{i_1})\prod_{\ell=2}^nY_\ell^{i_\ell}
\end{equation}
extends uniquely to a finitely additive left-invariant measure on $\mathcal{R}(P)$ with values in $\mathbb{R}((Y_2))\cdots((Y_n))$. Every nonempty ring set has strictly positive measure.

For $D=hP_{i_1,\ldots,i_n}$, let $w(D)$ be the value in \eqref{eq:Yweight} and set
$\operatorname{lev}(D)=(i_2,\ldots,i_n)$. Equal cylinders have equal underlying subgroups, and
\[
\begin{aligned}
P_{i_1,\ldots,i_n}=P_{j_1,\ldots,j_n}
&\ \Longrightarrow\ (i_2,\ldots,i_n)=(j_2,\ldots,j_n),\\
[P_{i_1,\ldots,i_n}:P_{j_1,\ldots,j_n}]=1
&\ \Longrightarrow\ m_B(U_{i_1})=m_B(U_{j_1})
\ \Longrightarrow\ i_1=j_1.
\end{aligned}
\]
The last implication uses strict decrease of the compact open subgroups and their Haar volumes. Thus $w$ and $\operatorname{lev}$ are well-defined.

For intersecting cylinders, choose $a$ in their intersection:
\[
D=aH,\quad E=aL,\quad
H\subseteq L\ \text{or}\ L\subseteq H
\ \Longrightarrow\
D\subseteq E\ \text{or}\ E\subseteq D.
\]

Consider a finite partition $D=\bigsqcup_{r=1}^ND_r$, with underlying subgroups $H,H_r$. For $a\in D_r$,
\[
D_r=aH_r\subseteq aH=D\ \Longrightarrow\ H_r\subseteq H.
\]
We prove $[H:H_r]<\infty$ by induction on $N$. The case $N=1$ is immediate. For $N>1$, put
\[
L=\max_{\subseteq}\{H_1,\ldots,H_N\}.
\]
Each piece lies in one $L$-coset. A piece with subgroup $L$ is an entire $L$-coset, so $L=H$ would force that piece to equal $D$. Hence
\[
2\le[H:L]\le N.
\]
For each $L$-coset $aL\subseteq D$,
\[
aL=\bigsqcup_{\{r:D_r\subseteq aL\}}D_r,
\qquad
1\le\#\{r:D_r\subseteq aL\}<N.
\]
The induction hypothesis therefore gives
\[
[L:H_r]<\infty,\qquad
[H:H_r]=[H:L][L:H_r]<\infty
\ \Longrightarrow\
\operatorname{lev}(D_r)=\operatorname{lev}(D).
\]

Write
\[
D=hP_{i_1,\ldots,i_n},\qquad
D_r=h_rP_{j_{r,1},i_2,\ldots,i_n},\qquad
a\ge\max\{i_1,j_{1,1},\ldots,j_{N,1}\}.
\]
The closure property of $J$ ensures $(a,i_2,\ldots,i_n)\in J$. Set $M=P_{a,i_2,\ldots,i_n}$,
$w_M=m_B(U_a)\prod_{\ell=2}^nY_\ell^{i_\ell}$. Counting $M$-cosets and using \eqref{eq:compatibility},
\[
\begin{aligned}
[H:M]&=\sum_{r=1}^N[H_r:M],\\
w(D)&=[H:M]w_M
=\sum_{r=1}^N[H_r:M]w_M
=\sum_{r=1}^Nw(D_r).
\end{aligned}
\]

For a shell, prescribe
\begin{equation}\label{eq:shell}
S=D\setminus\bigsqcup_{\ell=1}^mD_\ell,\qquad D_\ell\subsetneq D,
\end{equation}
\begin{equation}\label{eq:shellweight}
\nu(S)=w(D)-\sum_{\ell=1}^m w(D_\ell),
\end{equation}
where the holes are disjoint and $m=0$ is allowed. The partition identity gives
\[
S=\varnothing
\ \Longrightarrow\
D=\bigsqcup_{\ell=1}^mD_\ell
\ \Longrightarrow\
w(D)-\sum_{\ell=1}^m w(D_\ell)=0.
\]

For a finite family of sets, $\operatorname{Max}_{\subseteq}$ denotes the subfamily of inclusion-maximal members. For a finite cylinder family $\mathcal F$, define its immediate descendants and shells by
\[
\begin{aligned}
\mathcal H_{\mathcal F}(C)
&=\operatorname{Max}_{\subseteq}\{E\in\mathcal F:E\subsetneq C\},\\
S_C^{\mathcal F}
&=C\setminus\bigsqcup_{E\in\mathcal H_{\mathcal F}(C)}E,\\
W_{\mathcal F}(C)
&=w(C)-\sum_{E\in\mathcal H_{\mathcal F}(C)}w(E).
\end{aligned}
\]
Nesting gives disjoint decompositions
\[
\bigcup_{C\in\mathcal F}C=\bigsqcup_{C\in\mathcal F}S_C^{\mathcal F},
\qquad
C=\bigsqcup_{\substack{E\in\mathcal F\\E\subseteq C}}S_E^{\mathcal F}.
\]
Thus every set in $\operatorname{ring}\mathcal F$ is a union of these shells.

Let $\mathcal F\subseteq\mathcal G$ be finite cylinder families. For $C\in\mathcal G$, each $D\in\mathcal G$ with $D\subsetneq C$ has a unique immediate parent among the members of $\mathcal G$ contained in $C$: the cylinders strictly containing $D$ and contained in $C$ form a nonempty finite chain, whose smallest member is that parent. Thus each such $w(D)$ occurs exactly once in the double sum below, and
\[
\begin{aligned}
\sum_{\substack{E\in\mathcal G\\E\subseteq C}}W_{\mathcal G}(E)
&=\sum_{\substack{E\in\mathcal G\\E\subseteq C}}w(E)
-\sum_{\substack{E\in\mathcal G\\E\subseteq C}}
 \sum_{D\in\mathcal H_{\mathcal G}(E)}w(D)\\
&=\sum_{\substack{E\in\mathcal G\\E\subseteq C}}w(E)
-\sum_{\substack{D\in\mathcal G\\D\subsetneq C}}w(D)
=w(C).
\end{aligned}
\]
This identity follows directly from the weight definitions.

For $C\in\mathcal F$, put
\[
\mathcal G_C=
\{E\in\mathcal G:E\subseteq C,\ 
 E\nsubseteq D\text{ for every }D\in\mathcal H_{\mathcal F}(C)\}.
\]
The $\mathcal G$-shells partition $C$. Each old hole $D\in\mathcal H_{\mathcal F}(C)$ is the union of exactly those shells indexed by $E\subseteq D$. Removing these disjoint holes gives
\[
S_C^{\mathcal F}=\bigsqcup_{E\in\mathcal G_C}S_E^{\mathcal G}.
\]
Applying the preceding weight identity to $C$ and to each old hole gives
\[
\begin{aligned}
W_{\mathcal F}(C)
&=w(C)-\sum_{D\in\mathcal H_{\mathcal F}(C)}w(D)\\
&=\sum_{\substack{E\in\mathcal G\\E\subseteq C}}W_{\mathcal G}(E)
-\sum_{D\in\mathcal H_{\mathcal F}(C)}
 \sum_{\substack{E\in\mathcal G\\E\subseteq D}}W_{\mathcal G}(E)\\
&=\sum_{E\in\mathcal G_C}W_{\mathcal G}(E).
\end{aligned}
\]
An empty shell has weight zero by the finite partition identity already proved. Empty shells may therefore be omitted from these decompositions and sums.

For $S\in\operatorname{ring}\mathcal F$, define
\[
\nu_{\mathcal F}(S)
=\sum_{\substack{C\in\mathcal F\\
\varnothing\ne S_C^{\mathcal F}\subseteq S}}W_{\mathcal F}(C).
\]
Each nonempty $\mathcal G$-shell contained in $S$ occurs in exactly one of the disjoint refinements above. Summing their weights consequently gives
\[
\nu_{\mathcal F}(S)
=\sum_{\substack{E\in\mathcal G\\
\varnothing\ne S_E^{\mathcal G}\subseteq S}}W_{\mathcal G}(E)
=\nu_{\mathcal G}(S).
\]
Two representations therefore agree on their common refinement:
\[
\nu_{\mathcal F_1}(S)
=\nu_{\mathcal F_1\cup\mathcal F_2}(S)
=\nu_{\mathcal F_2}(S).
\]
This defines $\nu$ on $\mathcal{R}(P)$. On a common shell decomposition,
\[
S\cap T=\varnothing
\ \Longrightarrow\
\nu(S\bigsqcup T)=\nu(S)+\nu(T).
\]
Moreover, $w(aD)=w(D)$ implies $\nu(aS)=\nu(S)$. Formula \eqref{eq:shellweight} forces every extension to have these values, proving uniqueness.

For positivity, take a nonempty shell $S$ with parent $D=hP_{i_1,\ldots,i_n}$. Let $L_\ell$ underlie $D_\ell$, and set
\[
T=\{\ell:\operatorname{lev}(D_\ell)=(i_2,\ldots,i_n)\}.
\]
Choose $a$ at least as large as the first indices of the parent and these holes; set $M=P_{a,i_2,\ldots,i_n}$. The number of parent $M$-cosets left by these holes is
\[
b=[P_{i_1,\ldots,i_n}:M]-\sum_{\ell\in T}[L_\ell:M]\ge1,
\]
since $S\ne\varnothing$. Thus
\[
\nu(S)=
b\,m_B(U_a)\prod_{r=2}^nY_r^{i_r}
-\sum_{\ell\notin T}w(D_\ell).
\]
Every term in the last sum has level strictly greater than $(i_2,\ldots,i_n)$. The leading real coefficient is therefore
\[
\operatorname{lc}(\nu(S))=b\,m_B(U_a)>0
\ \Longrightarrow\ \nu(S)>0.
\]
Finally, $S=\bigsqcup_{r=1}^NS_r$ with nonempty shells gives
$\nu(S)=\sum_{r=1}^N\nu(S_r)>0$.

\label{lem:image}
Let $f:W\to F^d$ be analytic on an open neighborhood of $0$, with $f(0)=0$ and $L=f'(0)\in\operatorname{GL}_d(F)$. There are integers $\kappa_1,\ldots,\kappa_{n-1}\ge0$ and $\kappa_n\ge1$ such that, for every depth

the ball $\mathfrak{p}_{i_1,\ldots,i_n}^{\,d}$ lies in $W$ and

f(\mathfrak{p}_{i_1,\ldots,i_n}^{\,d})=L\mathfrak{p}_{i_1,\ldots,i_n}^{\,d}.
\end{equation}

\qquad \left|\det f'(a)\right|_n=\left|\det L\right|_n.

Since $\det f'(0)=\det L\ne0$, choose a neighborhood $V_0$ of $0$ and a common outer polydisc on which $a+h\in W$ and $f'(a)$ is invertible. The maps
\[
\begin{aligned}
g_a(h)&=f'(a)^{-1}\bigl(f(a+h)-f(a)\bigr)=h+E_a(h),\\
g_a(0)&=0,\qquad g_a'(0)=\operatorname{id}_{F^d},\\
E_a(0)&=0,\qquad E_a'(0)=0
\end{aligned}
\]
are jointly analytic in $(a,h)$ for $a\in V_0$ and small $h$. Each component of $E_a(h)$ has only terms of degree at least two in $h$.

The identities
\[
u^r-v^r=(u-v)\sum_{t=0}^{r-1}u^{r-1-t}v^t
\qquad(r\ge1)
\]
and coordinatewise telescoping give, on a smaller common convergence polydisc,
\[
E_{a,s}(h)-E_{a,s}(k)
=\sum_{r=1}^d(h_r-k_r)b_{sr}(a,h,k),
\qquad b_{sr}(a,0,0)=0.

The factor series are jointly analytic; after one factor $h_r-k_r$ is removed, their terms still have positive total degree in $(h,k)$. By joint continuity, shrink $V_0$ to a neighborhood $V$ of $0$ and choose integers $\kappa_1,\ldots,\kappa_{n-1}\ge0$ and $\kappa_n\ge1$ such that, for every $a\in V$ and $h,k\in\mathfrak{p}_{\kappa_1,\ldots,\kappa_n}^{\,d}$,
\[

a+\mathfrak{p}_{\kappa_1,\ldots,\kappa_n}^{\,d}&\subseteq W,\\
b_{sr}(a,h,k)&\in t_n^2\mathcal{O}_F\qquad(1\le s,r\le d).
\end{aligned}

The constants are independent of $a$. Since
\[
v(b_{sr}(a,h,k))>(0,\ldots,0,1),

the valuation inequality for finite sums gives

v(E_a(h)-E_a(k))
\ge v(h-k)+(0,\ldots,0,1)\qquad(h\ne k).

Use the real-valued outer absolute value and maximum coordinate norm

|u|_{\mathrm{out}}=2^{-v_n(u)}\quad(u\ne0),\qquad
|0|_{\mathrm{out}}=0,\qquad
\|h\|_{\mathrm{out}}=\max_{1\le s\le d}|h_s|_{\mathrm{out}}.
\]
The same coefficient bound gives, with $c=|t_n|_{\mathrm{out}}^2<1$,

\|E_a(h)-E_a(k)\|_{\mathrm{out}}

The case $h=0$ follows directly from $E_a(0)=0$.

\|h^{(s)}-h^{(r)}\|_{\mathrm{out}}

\end{aligned}

Thus the iterates are Cauchy. Since $F^d$ is outer-complete and the ball is outer-closed, they converge to some $h$ in the same ball. Continuity of $T_{a,b}$ gives
\[

\ \Longleftrightarrow\ h+E_a(h)=b
\ \Longleftrightarrow\ g_a(h)=b.

Zero terms have valuation $+\infty$; an empty block is omitted. Since $U$ preserves the minimum coordinate valuation,
\begin{equation}\label{eq:minvaluation}
\min_{s,t}v\bigl(\iota_H(x)_{st}-(I_m)_{st}\bigr)=v(x).
\end{equation}
The equality includes $x=0$.

Choose the chart with $\iota_H(W)=H(F)\cap\Omega$, where $\Omega$ is an ambient identity neighborhood. Choose component constants as in Lemma~\ref{lem:estimate}, enlarged so that
\[
\mathfrak{p}_{\kappa_1,\ldots,\kappa_n}^{\,d}\subset W,\qquad
I_m+M_m(\mathfrak{p}_{\kappa_1,\ldots,\kappa_n})\subset\Omega.
\]
For $(i_1,\ldots,i_n)\ge(\kappa_1,\ldots,\kappa_n)$, nesting places the corresponding coordinate ball in $W$ and the matrix ball in $\Omega$. For $z=\iota_H^{-1}$ and $g\in H(F)\cap\Omega$, equation \eqref{eq:minvaluation} gives
\[
\begin{aligned}
g\in I_m+M_m(\mathfrak{p}_{i_1,\ldots,i_n})
&\ \Longleftrightarrow\
\min_{s,t}v(g_{st}-(I_m)_{st})\ge(i_1,\ldots,i_n)\\
&\ \Longleftrightarrow\
v(z(g))\ge(i_1,\ldots,i_n)\\
&\ \Longleftrightarrow\
z(g)\in\mathfrak{p}_{i_1,\ldots,i_n}^{\,d}.
\end{aligned}
\]
This proves \eqref{eq:matrixidentity} for every depth in the statement. For $m_0\ge0$,
\[
(i_1+m_0,i_2,\ldots,i_n)
\ge(i_1,\ldots,i_n)\ge(\kappa_1,\ldots,\kappa_n),
\]
so both depths remain in this family. Proposition~\ref{prop:deep} therefore gives subgroups with
\[
[C^H_{i_1,\ldots,i_n}:C^H_{i_1+m_0,i_2,\ldots,i_n}]=q^{dm_0}.
\]
The analytic-group construction in the proof of Theorem~\ref{thm:main}, with these component shifts and \eqref{eq:absnormalization}, gives the absolute cylinder measure on the ring generated by this family. For any fixed $c\in(\mathbb R_n)_{>0}$, the left-invariant measure $\mu_H=c\,\mu^{\mathrm{abs}}_{H(F),z}$ satisfies
\[
\begin{aligned}
\mu_H(hC^H_{i_1,\ldots,i_n})
&=c\,\mu^{\mathrm{abs}}_{H(F),z}(hC^H_{i_1,\ldots,i_n})\\
&=c\,q^{-di_1}\prod_{\ell=2}^nX_\ell^{di_\ell},
\end{aligned}
\]
as required.

\label{ex:norm-one-torus}
Assume $\operatorname{char}F\ne2$ and choose a nonsquare $D\in F^\times$. The norm-one torus for $F(\sqrt D)/F$ has the realization

T(F)=\left\{\begin{pmatrix}a&Dx\\x&a\end{pmatrix}:a^2-Dx^2=1\right\}.

The derivative with respect to $a$ is $2$ at $(a,x)=(1,0)$, so there is a unique analytic solution $a=a(x)$ near the identity. Since
\[
a(x)-1=\frac{Dx^2}{a(x)+1},
\]
$x$ is an identity coordinate. Choose its component constants as in Proposition~\ref{prop:deep}. For $(i_1,\ldots,i_n)\ge(\kappa_1,\ldots,\kappa_n)$,

K^T_{i_1,\ldots,i_n}=\left\{\begin{pmatrix}a(x)&Dx\\x&a(x)\end{pmatrix}:x\in\mathfrak{p}_{i_1,\ldots,i_n}\right\},
\qquad
\mu_T(gK^T_{i_1,\ldots,i_n})=q^{-i_1}\prod_{\ell=2}^nX_\ell^{i_\ell}.
\]

\label{ex:heisenberg}
Write the Heisenberg group as
\[
h(x,y,w)=\begin{pmatrix}1&x&w\\0&1&y\\0&0&1\end{pmatrix},
\qquad \mathcal H(F)=\{h(x,y,w):x,y,w\in F\}.
\]
Its multiplication is
\[
h(x,y,w)h(x',y',w')=h(x+x',y+y',w+w'+xy').
\]
Inversion and the commutator are
\[
\begin{aligned}
h(x,y,w)^{-1}&=h(-x,-y,xy-w),\\
[h(x,y,w),h(x',y',w')]&=h(0,0,xy'-x'y).
\end{aligned}
\]
For $i_n>0$, the set $K^{\mathcal H}_{i_1,\ldots,i_n}=\{h(x,y,w):x,y,w\in\mathfrak{p}_{i_1,\ldots,i_n}\}$ is a subgroup because $\mathfrak{p}_{2i_1,\ldots,2i_n}\subset\mathfrak{p}_{i_1,\ldots,i_n}$. The bilinear commutator gives an explicit instance of Lemma~\ref{lem:estimate}:
\[
[K^{\mathcal H}_{i_1,\ldots,i_n},K^{\mathcal H}_{j_1,\ldots,j_n}]
\subseteq K^{\mathcal H}_{i_1+j_1,\ldots,i_n+j_n}
\qquad(i_n,j_n>0).
\]
The center $Z=\{h(0,0,w):w\in F\}$ is the kernel of the quotient map $\pi:\mathcal H(F)\to F^2$, $\pi(h(x,y,w))=(x,y)$. In absolute normalization, the volume and fiber integral are
\[
\begin{gathered}
\mu_{\mathcal H}(K^{\mathcal H}_{i_1,\ldots,i_n})=q^{-3i_1}\prod_{\ell=2}^nX_\ell^{3i_\ell},
\\
\pi_!\mathbf{1}_{K^{\mathcal H}_{i_1,\ldots,i_n}}
=q^{-i_1}\prod_{\ell=2}^nX_\ell^{i_\ell}\mathbf{1}_{\mathfrak{p}_{i_1,\ldots,i_n}^{\,2}},
\end{gathered}
\]
as in Proposition~\ref{prop:disintegration}.

\label{ex:affine-group}
The affine group is
\[
A=F^\times\ltimes F,\qquad (a,b)(a',b')=(aa',b+ab').
\]
At positive outer depth, the coordinates $(a-1,b)$ give
\[
K^A_{i_1,\ldots,i_n}=\{(a,b):a-1,b\in\mathfrak{p}_{i_1,\ldots,i_n}\},
\qquad \mu_A(K^A_{i_1,\ldots,i_n})=q^{-2i_1}\prod_{\ell=2}^nX_\ell^{2i_\ell}.
\]
A left-invariant form is $\omega=a^{-2}\,da\wedge db$. Since $\delta_A(a,b)=a$, right translation by $g_0=(a_0,b_0)$ gives
\[
\lambda_\omega(Sg_0)=\left|a_0\right|_n^{-1}\lambda_\omega(S).
\]
In particular, if $a_0=t_1^{j_1}\cdots t_n^{j_n}$, the factor is
\[
q^{j_1}\prod_{\ell=2}^nX_\ell^{-j_\ell}.
\]

\label{ex:abelian-varieties}
Let $\mathcal A/F$ be an abelian variety of dimension $h>0$. Choose identity coordinates $z_{\mathcal A}$ with values in $F^h$ and component constants as in Proposition~\ref{prop:deep}. For $(i_1,\ldots,i_n)\ge(\kappa_1,\ldots,\kappa_n)$, set
\[
K^{\mathcal A}_{i_1,\ldots,i_n}
:=z_{\mathcal A}^{-1}(\mathfrak{p}_{i_1,\ldots,i_n}^{\,h}).
\]
For $P\in\mathcal A(F)$ and $m\ge0$, this gives
\[
\mu_{\mathcal A}(P+K^{\mathcal A}_{i_1,\ldots,i_n})=q^{-hi_1}\prod_{\ell=2}^nX_\ell^{hi_\ell},
\qquad
[K^{\mathcal A}_{i_1,\ldots,i_n}:K^{\mathcal A}_{i_1+m,i_2,\ldots,i_n}]=q^{hm}.
\]
The measure is bi-invariant because $\mathcal A(F)$ is commutative. For an elliptic curve $\mathcal E/F$ with identity $O$, the formal group gives an identity coordinate $z$ with $z(O)=0$ \cite[Chapter~IV]{silverman}; the formulas above apply with $h=1$. These give non-affine instances of Theorem~\ref{thm:main}.
