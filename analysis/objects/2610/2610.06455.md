 \big\|(A_Nf)_{N\in\mathbb N}\big\|_

<!-- lemma-2: body_md -->
\label{sup>inf}
Let $(f_n)_n\subset L_0(\mathcal A)_+$. Then, for every $\lambda>0$,
\[
 \Phi_{\sup}\left(
 \mathop{\mathrm{sup}^{+}}\limits_n f_n>\lambda
 \right)
 \geq
 \Phi_{\inf}\left(
 \mathop{\mathrm{sup}^{+}}\limits_n f_n>\lambda
 \right).
\]
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
By Zorn's lemma, choose a maximal pairwise orthogonal family
$(e_k)_k$ such that, for every $k$, there exists an index $n(k)$
satisfying
\[
 e_kf_{n(k)}e_k>\lambda e_k.
\]
Since $\varphi$ is finite and faithful, the nonzero members of this
family form an at most countable collection. Put
\[
 e=\left(\bigvee_ke_k\right)^\perp.
\]
We claim that
\[
 ef_ne\leq\lambda e
 \qquad\text{for every }n.
\]
Indeed, suppose that this fails for some $n_0$. Then the spectral
projection
\[
 e'
 =
 \mathbf 1_{(\lambda,\infty)}(ef_{n_0}e)
\]
is a nonzero subprojection of $e$ and satisfies
\[
 e'f_{n_0}e'>\lambda e'.
\]
Since $e'$ is orthogonal to every $e_k$, adjoining $e'$ to the
family $(e_k)_k$ contradicts maximality.

Consequently, $e$ is admissible in the definition of
$\Phi_{\inf}$, and hence
\[
 \Phi_{\inf}\left(
 \mathop{\mathrm{sup}^{+}}\limits_n f_n>\lambda
 \right)
 \leq\varphi(e^\perp)
 =\sum_k\varphi(e_k)
 \leq
 \Phi_{\sup}\left(
 \mathop{\mathrm{sup}^{+}}\limits_n f_n>\lambda
 \right).
\]

<!-- proposition-2: body_md -->
\label{prop:borel-cantelli}
Let $\mathcal M$ be a $\mathrm{II}_1$ factor and let
$\mathcal U$ be a subgroup of the unitary group of $\mathcal M$
such that
\[
 \mathcal U' \cap\mathcal M=\mathbb C1.
\]
Let $(e_k)_{k\geq1}$ be a sequence of projections in
$ \mathcal A=L_\infty(G)\overline\otimes\mathcal M$ such that
\[
 \sum_{k\geq1}\varphi(e_k)=\infty.
\]
Then there exist $(g_k)_{k\geq1}\subset G$ and
$(u_k)_{k\geq1}\subset\mathcal U$ such that, for every $\ell\geq1$,
\[
 \bigvee_{k\geq\ell}
 \alpha_{g_k}(u_k^*e_ku_k)
 =1.
\]

<!-- proof-2: body_md -->
By homogeneity, replacing $A_r$ by $B^{-1}A_r$, we may assume that
$B=1$. By decomposing an arbitrary element into four positive
parts, it is enough to prove the required estimate for
$f\in L_p(\mathcal A)_+$.

In view of Lemma \ref{sup>inf}, it suffices to prove that
\[
 \sup_{\lambda>0}
 \lambda^p
 \Phi_{\sup}\left(
 \mathop{\mathrm{sup}^{+}}\limits_{r\in\mathcal I}A_r(f)>\lambda
 \right)
 \lesssim1
\]
whenever $f\in L_p(\mathcal A)_+$ and $\|f\|_p\leq1$.

Define
\[
 M(\lambda)
 :=
 \sup_{\substack{f\in L_p(\mathcal A)_+\\\|f\|_p\leq1}}
 \lambda^p
 \Phi_{\sup}\left(
 \mathop{\mathrm{sup}^{+}}\limits_{r\in\mathcal I}A_r(f)>\lambda
 \right).
\]
Since $\varphi(1)=1$, we have
\[
 M(\lambda)\leq\lambda^p.
\]
Thus, if $M$ is unbounded, then necessarily
\[
 \limsup_{\lambda\to\infty}M(\lambda)=\infty.
\]

Assume, toward a contradiction, that $M$ is unbounded. The standard
selection argument used in \cite[p.~78]{Bur64} provides sequences
$(a_k)_{k\geq1}$ and $(\lambda_k)_{k\geq1}$ of positive numbers such
that
\[
 \lambda_k\longrightarrow\infty,
 \qquad
 \sum_{k\geq1}a_k^p\leq1,
\]
and
\[
 \sum_{k\geq1}
 \left(\frac{a_k}{\lambda_k}\right)^p
 M\left(\frac{\lambda_k}{a_k}\right)
 =\infty.
\]

For every $k$, choose $f_k\in L_p(\mathcal A)_+$ with
$\|f_k\|_p\leq1$ such that
\[
 \Phi_{\sup}\left(
 \mathop{\mathrm{sup}^{+}}\limits_{r\in\mathcal I}A_r(f_k)>
 \frac{\lambda_k}{a_k}
 \right)
 \geq
 \left(\frac{a_k}{\lambda_k}\right)^p
 M\left(\frac{\lambda_k}{a_k}\right)-2^{-k}.
\]
It follows that
\begin{equation}\label{est1}
 \sum_{k\geq1}
 \Phi_{\sup}\left(
 \mathop{\mathrm{sup}^{+}}\limits_{r\in\mathcal I}A_r(f_k)>
 \frac{\lambda_k}{a_k}
 \right)
 =\infty.
\end{equation}

For every $k$ we may choose a pairwise
orthogonal family $(e_{k,i})_i$ such that
\[
 \sum_i\varphi(e_{k,i})
 \geq
 \Phi_{\sup}\left(
 \mathop{\mathrm{sup}^{+}}\limits_{r\in\mathcal I}A_r(f_k)>
 \frac{\lambda_k}{a_k}
 \right)-2^{-k},
\]
and such that, for every $i$, there exists
$r_{k,i}\in\mathcal I$ satisfying
\begin{equation}\label{est2}
 e_{k,i}A_{r_{k,i}}(f_k)e_{k,i}
 >
 \frac{\lambda_k}{a_k}e_{k,i}.
\end{equation}
Set
\[
 e_k=\sum_i e_{k,i}.
\]
Equation \eqref{est1} then gives
\begin{equation}\label{est3}
 \sum_{k\geq1}\varphi(e_k)=\infty.
\end{equation}

We now enlarge the underlying von Neumann algebra. Let
\[
 \mathcal N=\mathcal M*L(\mathbb F_2)
\]
be the reduced free product with respect to the given traces. Then
$\mathcal N$ is a finite factor containing $\mathcal M$ as a
trace-preserving von Neumann subalgebra; see
\cite{TakesakiIII}. Let
\[
 \mathcal N^\infty
 =
 \overline{\bigotimes}_{k\geq1}\mathcal N,
 \qquad
 \mathcal A^\infty
 =
 L_\infty(G)\overline\otimes\mathcal N^\infty.
\]

Let
\[
 \Pi_k
 =
 \operatorname{id}_{L_\infty(G)}\otimes\pi_k:
 \mathcal A\longrightarrow\mathcal A^\infty
\]
be the trace-preserving embedding into the $k$-th tensor coordinate.
Thus, for elementary tensors,
\[
 \Pi_k(f\otimes m)
 =
 f\otimes
 1\otimes\cdots\otimes1\otimes m\otimes1\otimes\cdots,
\]
where $m$ occupies the $k$-th coordinate. Put
\[
 e_k'=\Pi_k(e_k),
 \qquad
 e_{k,i}'=\Pi_k(e_{k,i}).
\]
Because the noncommutative parts of these projections lie in
distinct tensor coordinates and their $G$-components belong to the
commutative algebra $L_\infty(G)$, all the projections
$(e_{k,i}')_{k,i}$ commute whenever their first indices are
different.

Since $\Pi_k$ commutes with every $A_r$, equation \eqref{est2}
implies
\begin{equation}\label{est4}
 e_{k,i}'A_{r_{k,i}}(\Pi_k(f_k))e_{k,i}'
 >
 \frac{\lambda_k}{a_k}e_{k,i}'.
\end{equation}
Moreover, \eqref{est3} gives
\[
 \sum_{k\geq1}\varphi^\infty(e_k')=\infty.
\]

Let $\mathcal U_0$ be the subgroup of the unitary group of
$\mathcal N^\infty$ consisting of finite tensor products:
\[
 \mathcal U_0
 :=
 \left\{
 \prod_{j\in F}\pi_j(v_j):
 F\subset\mathbb N\text{ finite},
 \ v_j\in\mathcal U(\mathcal N)
 \right\}.
\]
Since $\mathcal N$ is a factor,
\[
 \mathcal U_0'\cap\mathcal N^\infty=\mathbb C1.
\]
Applying Proposition \ref{prop:borel-cantelli} to $(e_k')_k$, we
obtain sequences $(g_k)_k\subset G$ and $(u_k)_k\subset\mathcal U_0$
such that
\begin{equation}\label{est5}
 \bigvee_{k\geq\ell}
 \alpha_{g_k}(u_k^*e_k'u_k)
 =1
 \qquad\text{for every }\ell\geq1.
\end{equation}

Define
\[
 e_k''
 :=
 \alpha_{g_k}(u_k^*e_k'u_k),
 \qquad
 e_{k,i}''
 :=
 \alpha_{g_k}(u_k^*e_{k,i}'u_k).

It follows that the projections $e_k''$, and hence all
$e_{k,i}''$ with distinct first indices, commute.

 A_r\alpha_{g_k}=\alpha_{g_k}A_r,

Since $1<p\leq2$, the noncommutative space
$L_p(\mathcal A^\infty)$ has Rademacher type $p$; see
\cite[Corollary~5.5]{PX03} and also \cite{Fack87}. Therefore,
\begin{equation*}
 \left\|
 \sum_{k\geq1}\varepsilon_ka_k\gamma_k

\end{equation*}

\[
 \widetilde{\mathcal A}
 =
 \mathcal A^\infty
 \overline\otimes L_\infty(\Omega_\varepsilon)
\]
to the self-adjoint element
\[

\]
By the order description of the

\[

\]

 \mathcal D_0
 =

\]
and set
\[
 \mathcal D

 \mathcal D_0
 \overline\otimes L_\infty(\Omega_\varepsilon).
\]
Here,

the weak-operator closure of the unital $*$-algebra generated by
these projections.
The algebra $\mathcal D_0$ is commutative. Indeed, for each fixed
$k$, the projections $(e_{k,i}'')_i$ are pairwise orthogonal. If
$k\neq\ell$, then $e_{k,i}''$ and $e_{\ell,j}''$ belong to distinct
tensor coordinates and therefore commute. Consequently, $\mathcal D$
is commutative.

\[
 \mathbb E:\widetilde{\mathcal A}\longrightarrow\mathcal D

be the trace-preserving conditional expectation. Applying
$\mathbb E$ to the preceding inequalities gives
\[
 -\mathbb E(b)

 \leq

\]
for every $r\in\mathcal I$. Notice that
$\mathbb E(A_r(\gamma_k))$ does not depend on the Rademacher
variables.

\cite[Lemma~1, p.~78]{Bur64} and Fubini's theorem,
\[
 \limsup_{k\to\infty}
 \sup_{r\in\mathcal I}
 \bigl|a_k\mathbb E(A_r(\gamma_k))\bigr|
 <\infty
\qquad\text{almost everywhere}.
\]

$\mathbb E$ is $\mathcal D$-bimodular, applying $\mathbb E$ to
\eqref{est7} gives
\begin{equation}\label{est9}
 e_{k,i}''
 \mathbb E(A_{r_{k,i}}(\gamma_k))
 e_{k,i}''
 \geq
 \frac{\lambda_k}{a_k}e_{k,i}''.
\end{equation}

\[
 \sup_{r\in\mathcal I}
 \bigl|a_k\mathbb E(A_r(\gamma_k))\bigr|
 \geq\lambda_ke_{k,i}''
 \qquad\text{for every }i.
\]
The projections $(e_{k,i}'')_i$ are pairwise orthogonal and satisfy
$e_k''=\sum_i e_{k,i}''$. Hence
\[
 \sup_{r\in\mathcal I}
 \bigl|a_k\mathbb E(A_r(\gamma_k))\bigr|
 \geq\lambda_k\sum_i e_{k,i}''
 =\lambda_ke_k''.
\]
By \eqref{est5},
\[
 \bigvee_{k\geq\ell}e_k''=1
 \qquad\text{for every }\ell\geq1.
\]
Since $\lambda_k\to\infty$, we obtain
\[
 \limsup_{k\to\infty}
 \sup_{r\in\mathcal I}
 \bigl|a_k\mathbb E(A_r(\gamma_k))\bigr|
 =\infty
 \qquad\text{almost everywhere},
\]
which is a contradiction. This proves
\eqref{Lambdap8 bound}.

\eqref{holder 2} for self-adjoint $h_1,h_2$.

\emph{Step 2: the general case.}
Write
$h_1=|h_1^*|u_1^*$ and $h_2=u_2|h_2|$. The
Cauchy--Schwarz inequality gives
\begin{align*}
 &\left|\tau\left(\sum_k\gamma_kf_kh_1f_kh_2\right)\right|\\
 &\quad\leq
 \left[
 \tau\left(\sum_kf_k|h_1^*|f_k|h_2|\right)
 \right]^{1/2}
 \left[
 \tau\left(
 \sum_kf_ku_2|h_2|u_2^*f_ku_1|h_1^*|u_1^*
 \right)
 \right]^{1/2}.
\end{align*}
Applying Step 1 to both factors proves \eqref{holder 2} in general.

\emph{Step 3: reduction of the general case to $d=2$.}
It remains to consider $d\geq3$.
Let $(\gamma_k)_k$ be independent random variables uniformly
distributed over the {$(d-1)$}-th roots of unity. Then
\[
 \mathbb E(\gamma_k^\ell)=0
 \quad(1\leq\ell\leq d-2),
 \qquad
 \mathbb E(\gamma_k^{d-1})=1.
\]
Independence gives
\begin{align*}
 \sum_k f_kh_1f_kh_2\cdots f_kh_d=
 \mathbb E\left[
 \left(\sum_k\gamma_kf_kh_1f_kh_2\right)
 \prod_{i=3}^d\left(\sum_j\gamma_jf_j\right)h_i
 \right].
\end{align*}
For every choice of the scalars $\gamma_j$, the positive
factorization used above gives
\[
 \left\|\sum_j\gamma_jf_j\right\|_r
 \leq\left\|\sum_jf_j\right\|_r.
\]
Define $s$ by
\[
 \frac1s
 =
 \frac1{r_2}
 +\frac{d-2}{r}
 +\sum_{i=3}^d\frac1{r_i}.
\]
The hypothesis of the lemma gives
\[
 \frac1r+\frac1{r_1}+\frac1s=1.
\]
Thus we may apply \eqref{holder 2} with
\[
 h_2\left(\sum_j\gamma_jf_j\right)h_3
 \cdots
 \left(\sum_j\gamma_jf_j\right)h_d
\]
as its second operator. H\"older's inequality then yields
\[
 \left|
 \tau\left(\sum_k f_kh_1f_kh_2\cdots f_kh_d\right)
 \right|
 \leq
 \left\|\sum_kf_k\right\|_r^{d-1}
 \prod_{i=1}^d\|h_i\|_{r_i},
\]
which is the desired conclusion.

For clarity, we divide the proof into two parts.  First, we estimate
the contribution of the rational centers which occur in
$L_{2^k}^{(\le2^s)}$ but not in $\widehat M_{s,k}$, using the decay
of $S(\theta)$ and Lemma~\ref{lem:packet-v2}.  We then estimate the
error caused by the different cutoffs at the common centers, using
the separation of the rational centers, the support properties of
the cutoffs, and the decay estimate \eqref{eq:wdec-v2}.

\emph{Step 1: the contribution of the additional rational centers.}

sum defining $L_{2^k}^{(\le2^s)}$ but not in $\widehat M_{s,k}$.  Set
\[
 \Theta_s
 =

 \frac aq\in\mathbb Q\cap\mathbb T:(a,q)=1,\ q\mid\mathcal Q_{2^s}
 \right\}, \quad\mathcal Q_{2^s}=(2^s)!^{A\lfloor\log 2^s\rfloor}
\]
and

 =

 \frac aq\in\mathbb Q\cap\mathbb T:(a,q)=1,\ q\mid Q_s
 \right\},

By the definition of the sets $\Gamma_u$,
\[
 \{0\}\cup

 =\Theta_s.

$L_{2^k}^{(\le2^s)}$, whereas $\Lambda_s$ is the set of centers
occurring in $\widehat M_{s,k}$.  Since
\[
 Q_s=(2^s)!\mid\mathcal Q_{2^s},
\]
we have $\Lambda_s\subset\Theta_s$.

When $a/Q_s$ reduces to $a'/q$, the definition of the exponential
sum gives
\[
 S(a/Q_s)=S(a'/q).
\]
Consequently, the sum defining $\widehat M_{s,k}$ may be indexed by the reduced
fractions in $\Lambda_s$, with coefficient $S(\theta)$ at the center
$\theta$.  It remains in this step to estimate the centers in
$\Theta_s\setminus\Lambda_s$.

Let $\theta=a/q\in\Theta_s\setminus\Lambda_s$ be written in reduced
form.  Then
\[
 q\mid\mathcal Q_{2^s},
 \qquad
 q\nmid(2^s)!.
\]
For a prime $\ell$ and a positive integer $n$, we write
\[
 v_\ell(n)
 =
 \max\{\nu\in\mathbb N\cup\{0\}:\ell^\nu\mid n\}.
\]
Thus $v_\ell(n)$ is the exponent of $\ell$ in the prime
factorization of $n$.  We shall also use the standard formula
\[
 v_\ell(n!)
 =
 \sum_{\nu\ge1}
 \left\lfloor\frac{n}{\ell^\nu}\right\rfloor.
\]

Since $q|\mathcal Q_{2^s}$ its prime factor are smaller than $2^s$, so
if $q\nmid(2^s)!$, there is a prime $\ell\leq2^s$ such that
\[
 v_\ell(q)>v_\ell((2^s)!).
\]
We claim that this implies
\[
 4^s\lesssim q.
\]
If $\ell>2^{s-1}$, then $v_\ell((2^s)!)=1$, and therefore
$q\ge\ell^2\ge2^{2s-2}$.  If $\ell\le2^{s-1}$, put
\[
 m=\left\lfloor\frac{2^s}{\ell}\right\rfloor\ge2.
\]
Since $v_\ell((2^s)!)\ge m$, the integer $q$ is divisible by
$\ell^{m+1}$.  Moreover, $2^s<(m+1)\ell$, and hence
\[
 \frac{q}{4^s}
 \ge
 \frac{\ell^{m+1}}{4^s}
 >

 \frac{2^{m-1}}{(m+1)^2}

&\qquad\lesssim2^{C_0s+C_1 2^{\kappa s}-4^s}\|f\|_2.
\end{align*}
Since $\kappa<2$, we have $2^{\kappa s}=o(4^s)$.  It follows that
\[

 \lesssim2^{-C_{-1} s

 \lesssim2^{-C_{-1}s}\|f\|_2.

\emph{Step 3: conclusion.}
By \eqref{eq:comparison-decomposition-v2}, the triangle inequality
gives

 \left(
 \mathcal T_{L_{2^k}^{(\leq2^s)}-\widehat M_{s,k}}f

 \right\|_{L_2(\mathcal A_{\mathbb Z}(\mathcal M);\ell_\infty)}

 &\qquad\lesssim \left(2^{-\delta_ds}+2^{-C_{-1}s}\right)\|f\|_2
\lesssim2^{-\gamma's}\|f\|_2,
\end{align*}
where
\[ \gamma'=\min\{\delta_0,C_{-1}\}>0.
\]
This proves \eqref{eq:comparison-v2}.

We first prove the maximal inequality \eqref{mthm1}.
 It suffices to prove
\eqref{thm11}, namely,
\begin{equation}\label{thm111}
   \big\|(f*K_{2^k})_{k\in\mathbb N}\big\|_
   {L_p(\mathcal A_{\mathbb Z}(\mathcal N);\ell_\infty)}
   \lesssim_{p} \|f\|_{L_p(\mathcal A_{\mathbb Z}(\mathcal N))}.
\end{equation}

Since each convolution operator $f\mapsto f*K_{2^k}$ is positive,
the standard decomposition of an arbitrary element into four
positive elements reduces the proof to the case $f\geq0$.  We first
consider $1<p<2$.  The finitely many indices $k<4$ are bounded by
$C\|f\|_p$, since $\|f*K_{2^k}\|_p\leq\|f\|_p$.  Thus it remains to
consider $k\geq4$.

Given $k\in[4^s,4^{s+1})$, where $s$ is a positive integer,
\begin{equation*}
    f*K_{2^k}=\Big(f*K_{2^k}-f*M_{s,k}\Big)+\sum_{s^\prime=1}^sf*\Big(M_{s^\prime,k}-M_{s^\prime-1,k}\Big).
\end{equation*}
Here, $M_{0,k}:=0$.
For $k\ge4$, put $s(k)=\lfloor\log_4k\rfloor$. Then
$$\Big(\sum_{s^\prime=1}^{s(k)}f*\big(M_{s^\prime,k}-M_{s^\prime-1,k}\big)\Big)_{k\geq 4}=\sum_{s^\prime=1}^{\infty}\Big(1_{k\geq 4^{s'}}f*\big(M_{s^\prime,k}-M_{s^\prime-1,k}\big)\Big)_{k\geq 4}.$$
Consequently,
\begin{align}\label{Key-es2}
    \big\|(f*K_{2^k})_{k\in\mathbb N}\big\|_
    {L_p(\mathcal A_{\mathbb Z}(\mathcal N);\ell_\infty)}
    \lesssim(I)+(II),
    \end{align}
where
\begin{equation*}
    (I)=\sum_{s=1}^\infty
    \big\|\big(f*(K_{2^k}-M_{s,k})\big)_{4^s\leq k<4^{s+1}}\big\|_
    {L_p(\mathcal A_{\mathbb Z}(\mathcal N);\ell_\infty)}
\end{equation*}
and
\begin{equation*}
    (II)=\sum_{s^\prime=1}^\infty
    \big\|\big(f*(M_{s^\prime,k}-M_{s^\prime-1,k})\big)_{k\ge4^{s^\prime}}\big\|_
    {L_p(\mathcal A_{\mathbb Z}(\mathcal N);\ell_\infty)}.
\end{equation*}
Choose $p_0$ such that $1<p_0<p$.  By Lemma \ref{lem:M-p-v2},
\begin{equation}\label{Key-es3}
    \big\|(f*M_{s^\prime,k})_{k\ge4^{s^\prime}}\big\|_
    {L_{p_0}(\mathcal A_{\mathbb Z}(\mathcal N);\ell_\infty)}
    \lesssim\|f\|_{p_0}.
\end{equation}Hence, by the triangle inequality and \eqref{Key-es3},
\begin{equation}\label{Key-es5}
    \big\|\big(f*(M_{s^\prime,k}-M_{s^\prime-1,k})\big)_{k\ge4^{s^\prime}}\big\|_
    {L_{p_0}(\mathcal A_{\mathbb Z}(\mathcal N);\ell_\infty)}
    \lesssim\|f\|_{p_0}.
\end{equation}
The interval $[4^s,4^{s+1})$ is the union of
$[4^s,2\cdot4^s)$ and $[2\cdot4^s,4\cdot4^s)$.  Applying
Proposition \ref{Key1-es} with $N_0=2^{4^s}$ and
$N_0^2=2^{2\cdot4^s}$, respectively, and absorbing the endpoints into
the constant, we obtain the following estimate, since
$\log\log(2^{4^s})\simeq s$:
\begin{equation}\label{Key-es4}
    \big\|(f*K_{2^k})_{4^s\leq k<4^{s+1}}\big\|_
    {L_{p_0}(\mathcal A_{\mathbb Z}(\mathcal N);\ell_\infty)}
    \lesssim s\|f\|_{p_0}.
\end{equation}
Moreover, \eqref{Key-es3} and \eqref{Key-es4} give
\begin{equation}\label{Key-es6}
    \big\|\big(f*(K_{2^k}-M_{s,k})\big)_{4^s\leq k<4^{s+1}}\big\|_
    {L_{p_0}(\mathcal A_{\mathbb Z}(\mathcal N);\ell_\infty)}
    \lesssim s\|f\|_{p_0}.
\end{equation}
By Lemma \ref{lem:L2decay-v2}, for $s^\prime\geq2$ we have
\begin{equation}\label{Key-es7}
   \big\|\big(f*(M_{s^\prime,k}-M_{s^\prime-1,k})\big)_{k\ge4^{s^\prime}}\big\|_
   {L_2(\mathcal A_{\mathbb Z}(\mathcal N);\ell_\infty)}
   \lesssim2^{-\delta s^\prime}\|f\|_{2}.
\end{equation}
The same lemma also gives, for every $s\geq1$,
\begin{equation}\label{Key-es8}
     \big\|\big(f*(K_{2^k}-M_{s,k})\big)_{4^s\leq k<4^{s+1}}\big\|_
     {L_2(\mathcal A_{\mathbb Z}(\mathcal N);\ell_\infty)}
     \lesssim 2^{-\delta s}\|f\|_{2}.
\end{equation}
The case $s^\prime=1$ in \eqref{Key-es7} follows from Lemma
\ref{lem:M-p-v2} with $p=2$, after increasing the implicit constant.

Let $0<\theta<1$ be determined by
\[
 \frac1p=\frac{1-\theta}{p_0}+\frac\theta2.
\]
Interpolating \eqref{Key-es5} with \eqref{Key-es7}, we obtain
\begin{equation}\label{Key-es9}
   \big\|\big(f*(M_{s^\prime,k}-M_{s^\prime-1,k})\big)_{k\ge4^{s^\prime}}\big\|_
   {L_p(\mathcal A_{\mathbb Z}(\mathcal N);\ell_\infty)}
   \lesssim2^{-\delta_ps^\prime}\|f\|_{p}.
\end{equation}
Similarly, interpolation between \eqref{Key-es6} and
\eqref{Key-es8} first gives
\begin{equation}\label{Key-es10}
     \big\|\big(f*(K_{2^k}-M_{s,k})\big)_{4^s\leq k<4^{s+1}}\big\|_
     {L_p(\mathcal A_{\mathbb Z}(\mathcal N);\ell_\infty)}
     \lesssim s^{1-\theta}2^{-\theta\delta s}\|f\|_{p}
     \lesssim 2^{-\delta_ps}\|f\|_{p},
\end{equation}
where one may take any $0<\delta_p<\theta\delta$.  Combining
\eqref{Key-es2}, \eqref{Key-es9}, and \eqref{Key-es10}, and summing
the resulting geometric series, we deduce
\begin{equation*}
     \big\|(f*K_{2^k})_{k\in\mathbb N}\big\|_
     {L_p(\mathcal A_{\mathbb Z}(\mathcal N);\ell_\infty)}
     \lesssim \|f\|_p.
\end{equation*}

For $p=2$, the same conclusion follows directly from
\eqref{Key-es2}, \eqref{Key-es7}, and \eqref{Key-es8}.  Finally, for
$2<p<\infty$, interpolate the $L_2$ estimate with the evident bound
\[
 \big\|(f*K_{2^k})_{k\in\mathbb N}\big\|_
 {L_\infty(\mathcal A_{\mathbb Z}(\mathcal N);\ell_\infty)}
 \leq \|f\|_\infty.
\]
This proves \eqref{thm111} for every $1<p<\infty$, and hence
establishes \eqref{mthm1}.

Since the maximal inequality holds in $L_p$, to prove the b.a.u. convergence for $f\in L_p$, it suffices to do it only when $f\in L_2$.  To do so, by the  the discussion in
\cite[Section~4.1]{HongWang+arXiv2026}, it suffices to prove an analog of arguments of
\cite[Proposition~4.4]{HongWang+arXiv2026}. More precisely,  it suffices to show that for  fixed $0<\varepsilon<1/4$, there is a sequence $c_J\to 0$ such that  with
\[
 \mathbb D_\varepsilon
 =
 \{\lfloor(1+\varepsilon)^k\rfloor:k\geq1\},
 \qquad
 I_i=\mathbb D_\varepsilon\cap[N_i,N_{i+1}),
\]
where $N_i\in\mathbb D_\varepsilon$ and $N_{i+1}>2N_i$ and any $f\in L_2$:
\begin{equation}\label{eq:quantitative-convergence}
 \frac1{J+1}\sum_{i=0}^J
 \left\|
 \bigl(f*K_N-f*K_{N_i}\bigr)_{N\in I_i}
 \right\|_{L_2(\mathcal A_{\mathbb Z}(\mathcal N);\ell_\infty)}^2
 \leq c_J\|f\|_2^2.
\end{equation}

Fix $s\geq1$. Decompose, using \eqref{eq:LN-v2} and \eqref{eq:defLNs},
\[
 \widehat K_N
 =
 (\widehat K_N-L_N)
 +L_N^{(\leq2^s)}
 +(L_N-L_N^{(\leq2^s)}).
\]
Let $T_N^{(1)}$, $T_{s,N}^{(2)}$, and $T_{s,N}^{(3)}$
be the operators with these three multipliers, respectively.
Then
\[
 f*K_N
 =
 T_N^{(1)}f+T_{s,N}^{(2)}f+T_{s,N}^{(3)}f.
\]
For the first term, \eqref{eq:scalar-error-v2},
and Plancherel's theorem give
\begin{align*}
 \sum_{i=0}^J
 \left\|
 \bigl(T_N^{(1)}f-T_{N_i}^{(1)}f\bigr)_{N\in I_i}
 \right\|_{L_2(\mathcal A_{\mathbb Z}(\mathcal N);\ell_\infty)}^2
 \lesssim_{\varepsilon,P}
 \left(
 \sum_{N\in\mathbb D_\varepsilon}(\log N)^{-2\sigma}
 \right)\|f\|_2^2
 \lesssim_P\|f\|_2^2,
\end{align*}
since $\sigma>1/2$.
For the third term, summing the bounds in
Lemma~\ref{lem:layer-v2} over dyadic $u>2^s$ yields
\[
 \sum_{i=0}^J
 \left\|
 \bigl(T_{s,N}^{(3)}f-T_{s,N_i}^{(3)}f\bigr)_{N\in I_i}
 \right\|_{L_2(\mathcal A_{\mathbb Z}(\mathcal N);\ell_\infty)}^2
 \lesssim_P
 (J+1)2^{-2\gamma s}\|f\|_2^2.
\]
It remains to estimate $T_{s,N}^{(2)}$. Its multiplier is a
finite sum of terms
\[
 S(\theta)w_N(\alpha-\theta)\chi(\alpha-\theta),
\]
where $\theta$ and $\chi$ are independent of $N$.
We compare $w_N$ with the corresponding expression for the
leading monomial $a_dt^d$:
\[
 w_N^{(0)}(\beta)
 :=
 \frac1N\int_0^N
 \exp(-2\pi i\beta a_dt^d)\,dt.
\]
 Estimate \eqref{eq:wdec-v2} gives
\[
 |w_N(\beta)-w_N^{(0)}(\beta)|
 \lesssim_P
 \min\bigl\{
 |\beta|N^{d-1},(N^d|\beta|)^{-1/d}
 \bigr\},
 \qquad \beta\neq0.
\]
Consequently,
\[
 \|w_N-w_N^{(0)}\|_\infty
 \lesssim_P N^{-1/(d+1)}.
\]
For each fixed $\theta$ and $\chi$,
\cite[Corollary~4.8 and Remark~4.9]{HongWang+arXiv2026}
gives the quantitative maximal estimate over the intervals
$I_i$ for the multiplier
\[
 S(\theta)w_N^{(0)}(\alpha-\theta)\chi(\alpha-\theta),
\]
with a constant independent of $J$ and $(N_i)$. For the difference $w_N-w_N^{(0)}$, the same
calculation as for $T_N^{(1)}$, we have
\[
 \left(
 \sum_{N\in\mathbb D_\varepsilon}
 \|w_N-w_N^{(0)}\|_\infty^2
 \right)\|f\|_2^2
 \lesssim_{P}
 \left(
 \sum_{N\in\mathbb D_\varepsilon}N^{-2/(d+1)}
 \right)\|f\|_2^2
 \lesssim_{\varepsilon,P}\|f\|_2^2.
\]
Combining this bound with the estimate for $w_N^{(0)}$
and summing over the finitely many terms, we obtain
\[
 \sum_{i=0}^J
 \left\|
 \bigl(T_{s,N}^{(2)}f-T_{s,N_i}^{(2)}f\bigr)_{N\in I_i}
 \right\|_{L_2(\mathcal A_{\mathbb Z}(\mathcal N);\ell_\infty)}^2
 \lesssim_{\varepsilon,P,s} \|f\|_2^2,
\]
where the constant is independent of $J$ and $(N_i)$.

Combining these three bounds, we obtain
\[
 \frac1{J+1}\sum_{i=0}^J
 \left\|
 \bigl(f*K_N-f*K_{N_i}\bigr)_{N\in I_i}
 \right\|_{L_2(\mathcal A_{\mathbb Z}(\mathcal N);\ell_\infty)}^2
 \leq
 \left(
 \frac{C_{\varepsilon,P,s}}{J+1}
 +C_P2^{-2\gamma s}
 \right)\|f\|_2^2.
\]
Letting $J\to\infty$ and then $s\to\infty$ proves
\eqref{eq:quantitative-convergence}.
The desired b.a.u. convergence now follows from the discussion in
\cite[Section~4.1]{HongWang+arXiv2026}.
This completes the proof.

As in  \cite{Bur64}, if the conclusion does not hold then we may find a sequence $\lambda_k\to \infty$, $a_k>0$ with $\sum a_k=1$ and $G_k=(g_{k,n})_n\in \mathcal C$ such that $\sum_{k=1}^\infty \Phi_{\rm inf}(\sup^+_na_kg_{k,n}>\lambda_k)=\infty$ and the $(G_k)$ are tensor independent.

  Let $\mathcal M_k= \mathcal M_{G_k}$ for $k\geq 1$, we can see them in a bigger $(\mathcal M,\tau)$ as they are independent. There are conditional expectations $\mathbb E_k : \mathcal M \to \mathcal M_k$. We will also use that
  independence forces the spaces
$L_2(\mathcal M_k)\cap 1^\bot$ to be orthogonal in $L_2(\mathcal M)$.

  Let $X=(x_n)_n=\sum_{k=1}^\infty
  a_kG_k$, that is $x_n=\sum_{k=1}^\infty a_kg_{k,n}$.
  To get a contraction with the finiteness condition, we
  want to show that for any projection $p\in \mathcal
  M$ with $\tau(p)=t>0$ we have $\sup_n \|px_np\|=\infty$.

 Set
  $f_k=\mathbb E_k(p)$, then $f_k-t$ has mean 0 and by independence
  $\sum_{k=1}^\infty \|f_k-t\|_2^2\leq \| p-t\|_2^2<\infty$.

  Put $p_k=1_{f_k\geq t/2}$ and $p_k^\bot=1-p_k=1_{t-f_k> t/2}$. We have $t^2/4\tau(p_k^\bot)\leq \|f_k-t\|^2$, thus $\sum_k \tau(p_k^\bot)<\infty$. This obviously implies that $I=\{k\geq 1 \;|\; \tau(p_k^\bot)<\Phi_{\rm inf}(\sup^+_na_kg_{k,n}>\lambda_k)\}$ is infinite.

  Now, since $x_n\geq a_kg_{k,n}$, we have
  \begin{equation*}\|x_n^{1/2}p\|_\infty\geq
  \|(a_kg_{k,n})^{1/2}p\|_\infty\geq \| \mathbb E_k((a_kg_{k,n})^{1/2}p)\|_\infty=\|(a_kg_{k,n})^{1/2}f_k\|_\infty.\end{equation*}
  But $f_k\geq t/2 p_k$ and they commute so $t/2p_k=f_kc$ for some contraction $c$
  and we conclude, for all $k,n\geq 1$
  \begin{equation}\label{normxn}\|px_np\|_\infty\geq t^2/4\|p_k a_k g_{k,n}p_k\|_\infty.\end{equation}

  Fix $k\geq 1$, if for all $n\geq 1$, we have
  $\|p_k a_k g_{k,n}p_k\|_\infty\leq \lambda_k$, then by definition of $\Phi_{\inf}$ in \eqref{phi_inf},  $\Phi_{\rm inf}(\sup^+_na_kg_{k,n}>\lambda_k)\leq \tau(p_k^\bot)$ and thus $k\notin I$. It follows that $\sup_n \|p_ka_k g_{k,n}p_k\|\geq \lambda_k$ for $k\in I$. Since $I$ is infinite and $\lambda_k\to \infty$, \eqref{normxn} yields $\sup_n \|px_np\|=\infty$ and the proof is over.
