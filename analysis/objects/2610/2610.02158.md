<!-- theorem-1: body_md -->
\label{intro:theorem1}
Under our standing assumptions and for admissible parameter choices, the diffusion \eqref{intro:sde} admits the unique invariant probability measure $\mu_{\beta,\delta}$ in \eqref{intro:joint}. There exist a Lyapunov weight $V\geq1$ and constants $C,c>0$ such that, for every initial law $\mu_0$ satisfying $\mu_0(V)<\infty$,
\[
    \bigl\|\mathcal{L}(X_t,P_t)-\mu_{\beta,\delta}\bigr\|_{\mathrm{TV},V}
    \leq Ce^{-ct}\mu_0(V),
    \qquad t\geq0,
\]
where $\|\cdot\|_{\mathrm{TV},V}$ denotes weighted total variation distance.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{intro:theorem2}
Under our standing assumptions, for admissible parameter choices and a sufficiently small fixed stepsize $\lambda>0$, the explicit scheme \eqref{intro:algorithm} has polynomial moments of every order uniformly over all iterations, provided its initial state has a finite expectation of the exponential Lyapunov weight used in the discrete analysis. In particular, for every deterministic initial state $(X_0,P_0)=(x_0,p_0)$ and every $q\geq1$, there exists $C_q<\infty$, independent of $k$, such that
\[
    \sup_{k\geq0}\mathbb{E}\left[
        \|X_k\|_{\mathrm{F}}^q+\|P_k\|_{\mathrm{F}}^q
    \right]\leq C_q.
\]
<!-- end theorem-2 -->
<!-- remark-1: body_md -->
The scalar map in \eqref{taming} coincides with the relativistic momentum to velocity relation
\begin{align}
    v(s) = \frac{s}{\sqrt{m^2 + s^2 / c^2}},
    \label{Lorentz}
\end{align}
for a unit rest mass $m = 1$ and a speed limit $c = \delta^{-1/2}$. Taking the limit $c \to \infty$ while holding $m = 1$ recovers the classical velocity $v(s) = s$. Conversely, the massless limit $m \to 0$ with $c = 1$ maps each positive singular value to one. This recovers Muon's orthogonalization, while zero singular values remain zero.
<!-- end remark-1 -->
<!-- assumption-1: body_md -->
\label{ass:U_smooth}
The potential $U:\mathbb{R}^{n\times m}\to\mathbb{R}$ belongs to
$C^{\infty}(\mathbb{R}^{n\times m};\mathbb{R})$.
<!-- end assumption-1 -->
<!-- assumption-2: body_md -->
\label{ass:U_curvature}
The Hessian satisfies the $L_0/L_1$ growth bound. There exist
$L_0,L_1\geq0$ such that
\begin{equation}
    \|\nabla^2U(x)\|_{\mathrm{F}\to\mathrm{F}}
    \leq L_0+L_1|\nabla U(x)|,
    \qquad \forall x\in\mathbb{R}^{n\times m}.
    \label{eq:U_curvature}
\end{equation}
<!-- end assumption-2 -->
<!-- remark-2: body_md -->
\label{rem:U_gradient_bounds}
Under Assumptions~\ref{ass:U_smooth} and~\ref{ass:U_curvature},
the gradient of $U$ is locally Lipschitz.
<!-- end remark-2 -->
<!-- assumption-3: body_md -->
\label{ass:U_dissipativity}
The gradient is $2$-dissipative. There exist $w_1>0$ and
$w_2\geq0$ such that
\begin{equation}
    \langle x,\nabla U(x)\rangle
    \geq w_1|x|^2-w_2,
    \qquad \forall x\in\mathbb{R}^{n\times m}.
    \label{eq:U_dissipativity}
\end{equation}
<!-- end assumption-3 -->
<!-- remark-3: body_md -->
\label{rem:U_coercivity}
Under Assumptions~\ref{ass:U_smooth} and~\ref{ass:U_dissipativity},
there exists $C_U\geq0$ such that
\begin{equation}
    U(x)\geq\frac{w_1}{4}|x|^2-C_U,
    \qquad
    |\nabla U(x)|\geq w_1|x|-\frac{w_2}{|x|}
    \quad (x\neq0).
    \label{eq:U_coercive_bounds}
\end{equation}
Consequently, $U$ is coercive and $|\nabla U(x)|\to\infty$ as
$|x|\to\infty$.
<!-- end remark-3 -->
<!-- remark-4: body_md -->
\label{rem:U_normalization}
Since $U$ is continuous and coercive, it attains a global minimum.
Adding a constant changes neither the dynamics nor the normalized
Gibbs measure. Henceforth, without loss of generality, we set
$\min_xU(x)=0$, so that $U\geq0$.
Moreover, \eqref{eq:U_coercive_bounds} implies
\begin{equation}
    0<Z_\beta
    :=\int_{\mathbb{R}^{n\times m}}e^{-\beta U(x)}\,dx
    <\infty,
    \qquad \forall\beta>0.
    \label{eq:U_Gibbs_integrability}
\end{equation}
<!-- end remark-4 -->
<!-- condition-1: body_md -->
\label{harris_drift}\textit{(Geometric drift)}\\
    There exist a measurable $\mathcal{V}:\mathbb{R}^{2n\times m}\to[0,\infty)$ and constants $\varrho\in(0,1)$, $C_{\mathcal{V}}>0$ such that
    \begin{align}
        R\mathcal{V}\leq\varrho\mathcal{V}+C_{\mathcal{V}}.\label{harris_drift_ineq}
    \end{align}
<!-- end condition-1 -->
<!-- condition-2: body_md -->
\label{harris_minor}\textit{(Minorization)}\\
    For some $R_0>2C_{\mathcal{V}}/(1-\varrho)$, the sublevel set $C=\{\mathcal{V}\leq R_0\}$ satisfies
    \begin{align}
        R(z,\cdot)\geq\alpha_C\nu_C(\cdot),\ \ \forall z\in C,\label{harris_minor_ineq}
    \end{align}
    for some $\alpha_C\in(0,1)$ and a probability measure $\nu_C$.
<!-- end condition-2 -->
<!-- theorem-3: body_md -->
\label{harris}
    Let Conditions \ref{harris_drift} and \ref{harris_minor} hold. Fix $\alpha_0\in(0,\alpha_C)$, $\varrho_0\in(\varrho+2C_{\mathcal{V}}/R_0,1)$ and set
    \begin{align*}
        \theta=\dfrac{\alpha_0}{C_{\mathcal{V}}},\ \ \bar{\alpha}=\max\left\{1-(\alpha_C-\alpha_0),\dfrac{2+R_0\theta\varrho_0}{2+R_0\theta}\right\}\in(0,1).
    \end{align*}
    Then for all probability measures $\mu_1,\mu_2$,
    \begin{align}
        \|\mu_1R-\mu_2R\|_{\mathrm{TV},\theta}\leq\bar{\alpha}\|\mu_1-\mu_2\|_{\mathrm{TV},\theta}.\label{harris_contraction}
    \end{align}
    Consequently, $R$ admits a unique invariant probability measure $\pi_R$, with $\pi_R(\mathcal{V})<\infty$, and for every $\mu_0$ with $\mu_0(\mathcal{V})<\infty$ and $n\geq0$,
    \begin{align}
        \|\mu_0R^n-\pi_R\|_{\mathrm{TV},\theta}\leq e^{-n\log\bar{\alpha}}\left(2+\theta\mu_0(\mathcal{V})+\theta\pi_R(\mathcal{V})\right).\label{harris_ergodic}

<!-- corollary-1: body_md -->
\label{cor:SDE_moments}

\begin{align}
    \sup_{t\geq0}\mathbb{E}\left[|X_t|^r+|P_t|^r\right]
    \leq C_r\left(\mathbb{E}[\Psi(X_0,P_0)]+\frac{C_{\mathcal{K}}}{\eta}\right)<\infty.
    \label{eq:SDE_uniform_moments}
\end{align}
<!-- end corollary-1 -->

    \leq e^{-\eta t}\mathbb{E}[\Psi(X_0,P_0)]
    +\frac{C_{\mathcal{K}}}{\eta}(1-e^{-\eta t}),
    \qquad t\geq0.
\]
The asserted uniform polynomial moment bounds follow from the exponential domination by $\Psi$.

\label{thm:discrete_uniform_moments}

    \leq C_r\left(1+\mathbb{E}[\mathcal{V}(X_0,P_0)]\right)<\infty.
    \label{eq:discrete_uniform_moments}
\end{align}
In particular, this conclusion holds for every deterministic initial state.

The discrete Lyapunov estimate gives contraction outside the compact sublevel set
\[
    \mathcal{C}=\left\{(x,p):K_{\delta}(p)+K_{\delta}\bigl(y(x,p)\bigr)\leq R_0\right\}.
\]
Boundedness of the one-step Lyapunov expectation on $\mathcal{C}$ therefore yields
\[
    \mathbb{E}[\mathcal{V}_{n+1}\mid\mathcal{F}_n]
    \leq e^{-\eta\lambda}\mathcal{V}_n+B_{\lambda}
\]
for some $B_{\lambda}<\infty$. Iterating this inequality bounds $\mathbb{E}[\mathcal{V}_n]$ uniformly in $n$.

The coercivity of $U$ controls all polynomial powers of $|x|$. Taking expectations proves the asserted bounds.

<!-- lemma-3: body_md -->
\label{CharConvex}\textit{(Characterization of Convexity, \citet{Lewis1995})}\\
Let $A\in\mathbb{R}^{m\times n}$ and $\sigma:\mathbb{R}^{m\times n}\to\mathbb{R}^d$ (with $d=\min(m,n)$) have components the singular values of $A$. Let $f:\mathbb{R}^d\to(-\infty,+\infty]$ be absolutely symmetric. The corresponding unitarily invariant function $f(\sigma)$ is convex on $\mathbb{R}^{m\times n}$ if and only if $f$ is convex.
<!-- end lemma-3 -->
<!-- lemma-4: body_md -->
\label{HessianBound}
    For every $p\in\mathbb{R}^{m\times n}$, with $\sigma_1(p)$ denoting the largest spectral value of $p$, the following bound holds
    \begin{align}
        0\prec(1+\lambda\sigma^2_1(p))^{-3/2}I\preceq\nabla^2K_{\lambda}(p)\preceq I
    \end{align}
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
We define the auxiliary function $F(p)=\frac{1}{2}|p|^2-K_{\lambda}(p)$. In view of \eqref{kinEnergy} we obtain the representation
    \begin{align*}
        F(p)=\sum_{i\leq d}\dfrac{\sigma^2_i(p)}{2}-k(\sigma_i(p))=\sum_{i\leq d}f(\sigma_i(p)), \ f(s):=\dfrac{s^2}{2}-\dfrac{1}{\lambda}\left(\sqrt{1+\lambda s^2}-1\right).
    \end{align*}
    We have $\forall s\in\mathbb{R}$ that $f'(s)=s-s(1+\lambda s^2)^{-1/2}\Rightarrow f''(s)=1-(1+\lambda s^2)^{-3/2}\geq 0$, thus $f$ is a convex function and symmetric by construction, hence Lemma \ref{CharConvex} implies the composition $f(\sigma)$ is also convex. Since $F$ is defined as the sum of convex functions, it follows immediately that $F$ is convex. Hence $\nabla^2 F(p)\succeq0\Leftrightarrow I-\nabla^2 K_{\lambda}(p)\succeq0$. Define the positive constant $c=(1+\lambda R^2)^{-3/2}$, such that $R>|p|_2=\sigma_1(p)$. Then one defines the auxiliary function 
    \begin{align*}
       G(p)=\sum_{i\leq d}g(\sigma_i(p)),\ g(s)=\begin{cases}
            k(s)-\frac{c}{2}s^2,\ &|s|\leq R\\
            +\infty ,\ &|s|>R
        \end{cases}
    \end{align*}
    and concludes $G$ is convex. Since $G(p)=K_{\lambda}(p)-\frac{c}{2}|p|^2$ on $\{\sigma_1(p)\leq R\}$ and $R$ was chosen to be arbitrary, one recovers the statement.

<!-- lemma-5: body_md -->
\label{boundLaplacian}
    For every $p \in \mathbb{R}^{m \times n}$ the following bound holds
    \begin{align}
        \left|\Delta_p T_\lambda(p)\right| \leq (m+n)d^{1/2}\lambda^{1/2}.
    \end{align}
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
Since $K_{\lambda}$ is smooth, partial derivatives commute, hence $\Delta_pT_{\lambda}=\nabla_{p}\Delta_pK_{\lambda}$. Moreover matrices with pairwise distinct, positive singular values are dense, thus by continuity it suffices to consider such $p$ without loss of generality. Let $p=U\Sigma V^{\top}$, the Hessian of the spectral map \eqref{taming} acting on $A\in\mathbb{R}^{m\times n}$ admits the closed form [Theorem 2.2. in Christian Erwin Kümmerle, Understanding and Enhancing Data Recovery Algorithms, TU Munchen 2019]
\begin{align*}
    \nabla_p ^2K_{\lambda}(p)[A]=U\left(\begin{matrix}
    H_1\odot S(A_1)+H_2\odot S^*(A_1)\\
    H_3\odot A_2
    \end{matrix}\right)V^{\top},
\end{align*}
where $UAV^{\top}=[A_1;A_2]$ such that $A_1\in\mathbb{R}^{n\times n}$. Additionally one defines the symmetrization and anti-symmetrization operators, $S(B)=(1/2)(B+B^{\top})$ and $S^*(B)=(1/2)(B-B^{\top})$ respectively.
\begin{align*}
    H_1(i,j)=\tau^{[1]}(\sigma_i,\sigma_j), H_2(i,j)=\tau^{[1]}(\sigma_i,-\sigma_j), \text{and } H_3(i,j)=\tau^{[1]}(\sigma_j,0) 
\end{align*}
Now taking the trace, one recovers the Laplacian in the following form
\begin{align}
    \Delta_p K_{\lambda}(p)&=\sum_{i=1}^nH_1(i,i)+\sum_{1\leq i<j\leq n}\left(H_1(i,j)+H_2(i,j)\right)+\sum_{i=1}^{m-n}\sum_{j=1}^n H_3(i,j)\nonumber\\
&=\sum_{i\leq d}\tau'(\sigma_i)
    +\sum_{i<j\leq d}\left[\tau^{[1]}(\sigma_i,\sigma_j)+\tau^{[1]}(\sigma_i,-\sigma_j)\right]
    +(m-d)\sum_{i\leq d}\tau^{[1]}(\sigma_i,0)\label{laplaceSVD}
\end{align}
Notice how $\Delta_pK_{\lambda(p)}$ is essentially a smooth function of the singular values, namely $C(\sigma_1,\ldots,\sigma_d)$. Moreover it is symmetric and even in each argument, since $\tau'$ is even and, the oddness of $\tau$ gives $\tau^{[1]}(-x,y)=\tau^{[1]}(x,-y)$. Therefore the gradient formula for spectral functions [Theorem 3.1 \citet{Lewis1995}] applies, so that
\begin{align*}
    \nabla_{p}\Delta_pK_{\lambda}(p)=\nabla_pC(\sigma_1(p),\ldots,\sigma_d(p))=U\left(\text{Diag}\nabla_{\sigma} C(\sigma_1(p),\ldots,\sigma_d(p))\right)V^{\top}.
\end{align*}
Thus by the unitary invariance of the Frobenius norm one obtains
\begin{align*}
    |\Delta_pT_{\lambda}(p)|=|\nabla_{p}\Delta_pK_{\lambda}(p)|=\Big(\sum_{i=1}^d(\partial_{\sigma_i}C)^2\Big)^{1/2}
    \leq\sqrt{d}\,\max_{1\leq i\leq d}|\partial_{\sigma_i}C|.
\end{align*}
Since $\tau''(x)=-3\lambda x(1+\lambda x^2)^{-5/2}$, one has the elementary bound $|\tau''(x)|\leq\sqrt{\lambda}$ for all $x\in\mathbb{R}$. Differentiating the finite first difference once gives $\partial_x\tau^{[1]}(x,y)=(\tau'(x)-\tau^{[1]}(x,y))(x-y)^{-1}$. Next, one applies Taylor's theorem to obtain $\tau^{[1]}(x,y)=\tau'(x)+\frac{1}{2}\tau''(\xi)(y-x)$ for some $\xi\in(x,y)$ without the loss of generalization. Therefore we have that $\partial_x\tau^{[1]}(x,y)=\frac{1}{2}\tau''(\xi)$. Collecting the off-diagonal terms in \eqref{laplaceSVD} yields
\begin{align*}
    |\partial_{\sigma_i}C|\leq \sqrt{\lambda}+2(n-1)\dfrac{\sqrt{\lambda}}{2}+(m-n)\dfrac{\sqrt{\lambda}}{2}=\dfrac{m+n}{2}\sqrt{\lambda}.
\end{align*}
and one concludes with the statement of the Lemma.

<!-- lemma-6: body_md -->
\label{lem_master_equation}
Let $\mathcal{A}$ be the generator in \eqref{s3_gen}, and let
$\Psi$ be the Lyapunov function defined in \eqref{s3_Psi}, with
$a\in(0,1)$ and $b>0$. Define
\[
    \phi(x,p):=-\frac{\mathcal{A}\Psi(x,p)}{\Psi(x,p)}.
\]
Then
\begin{align}
    \phi(x,p)
    &=a(1-a)\beta\gamma\langle T_{\delta}(p),p\rangle
      +(1-2a)b\gamma\langle G(x),p\rangle\nonumber\\
    &\quad-b\langle T_{\delta}(p),\mathcal{J}_G(x)^\top p\rangle
      +b\langle\nabla U(x),G(x)\rangle-a\gamma nm\nonumber\\
    &\quad-b\gamma\beta^{-1}\langle\Gamma_{\delta}(p),G(x)\rangle
      -b^2\gamma\beta^{-1}
      \langle G(x),M_{\delta}(p)G(x)\rangle,
      \label{master1}
\end{align}
where $\mathcal{J}_G(x)^\top$ denotes the adjoint of the derivative
of $G$ with respect to the Frobenius inner product.
<!-- end lemma-6 -->
<!-- proof-7: body_md -->
Let $\psi=\log\Psi$, by direction calculations we get 
\begin{align}
    \nabla_x\psi&=a\beta\nabla_xH+b\nabla_x\langle G(x),p\rangle=a\beta\nabla U(x)+b\mathcal{J}_G^\top p,\nonumber\\
    \nabla_p\psi&=a\beta\nabla_pH+b\nabla_p\langle G(x),p\rangle=a\beta T_{\lambda}(p)+bG(x),\label{calc1}\\
    \Delta_p\psi&=a\beta\operatorname{Tr}(\nabla_p^2H)=a\beta\operatorname{Tr}(\mathcal{J}_{T_{\delta}})=a\beta \text{div}(T_{\delta}),\nonumber
\end{align}
where $[\mathcal{J}_G^\top p]_{k\ell}=\sum_{i=1}^n\sum_{j=1}^m\frac{[\partial G(x)]_{ij}}{\partial x_{k\ell}}p_{ij}$ and $\text{div}(T_{\delta})=\sum_{i=1}^n\sum_{j=1}^m\frac{\partial[T_{\delta}(p)]_{ij}}{\partial p_{ij}}$. Moreover, we have
\begin{align}
    \sum_{j=1}^m\sum_{i,k=1}^n[M_{\delta}(p)]_{ik}\dfrac{\partial^2K_{\delta}(p)}{\partial p_{ij}\partial p_{kj}}=\sum_{j=1}^m\sum_{i,k=1}^n[M_{\delta}(p)]_{ik}\dfrac{\partial[ T_{\delta}(p)]_{kj}}{\partial p_{ij}}.\nonumber
\end{align}
Since we know that $A\frac{\partial B}{\partial p_{ij}}=\frac{\partial}{\partial p_{ij}}(AB)-\frac{\partial A}{\partial p_{ij}}B$, we rewrite the latter term inside the sum
\begin{align}
    \sum_{j=1}^m\sum_{i,k=1}^n\left(\dfrac{\partial}{\partial p_{ij}}\left([M_{\delta}(p)]_{ik}[T_{\delta}(p)]_{kj}\right)-\dfrac{\partial[M_{\delta}(p)]_{ik}}{\partial p_{ij}}[T_{\delta}(p)]_{kj}\right).\nonumber
\end{align}
Recall the identity $M_{\delta}(p)T_{\delta}(p)=p$. One notices that the sum over $k$ in the first term corresponds to the matrix multiplication of $M_{\delta},T_{\delta}$. Hence we obtain
\begin{align}
    \sum_{j=1}^m\sum_{i=1}^n&\dfrac{\partial p_{ij}}{\partial p_{ij}}-\sum_{j=1}^m\sum_{i,k=1}^n\dfrac{\partial[M_{\delta}(p)]_{ik}}{\partial p_{ij}}[T_{\delta}(p)]_{kj}\nonumber\\
    =&nm-\sum_{j=1}^m\sum_{i,k=1}^n\dfrac{\partial[M_{\delta}(p)]_{ik}}{\partial p_{ij}}[T_{\delta}(p)]_{kj}.\nonumber
\end{align}
Moreover, by definition we have $[\Gamma_{\delta}(p)]_{kj}:=\sum_{i=1}^n\frac{\partial[M_{\delta}(p)]_{ki}}{\partial p_{ij}}$. Using the fact that $M_{\delta}$ is a symmetric matrix, we further simplify to
\begin{align}
    nm-\sum_{j=1}^m\sum_{k=1}^n[\Gamma_{\delta}(p)]_{kj}[T_{\delta}(p)]_{kj}=nm-\langle\Gamma_{\delta}(p),T_{\delta}(p)\rangle.\label{calc00}
\end{align}
Applying the generator of Eq.\eqref{s3_gen} to $\psi$ and substituting from \eqref{calc1} and \eqref{calc00} yields
\begin{align}
    \mathcal{L}\psi&=a\beta\langle T_{\lambda}(p),\nabla U(x)\rangle+b\langle T_{\lambda}(p),\mathcal{J}_G^\top p\rangle-a\beta\langle \nabla U(x),T_{\lambda}(p)\rangle-b\langle \nabla U(x), G(x) \rangle\nonumber\\
    &-a\beta\gamma\langle p,T_{\lambda}(p)\rangle-b\gamma\langle p, G(x) \rangle+a\gamma\langle \Gamma_{\delta}(p),T_{\lambda}(p)\rangle+b\gamma\beta^{-1}\langle \Gamma_{\delta}(p), G(x) \rangle\nonumber\\
    &+a\gamma nm-a\gamma\langle\Gamma_{\delta}(p),T_{\delta}(p)\rangle.\nonumber
\end{align}
The terms $\pm\langle\Gamma_{\delta}(p),T_{\delta}(p)\rangle$ and $\pm\langle T_{\lambda}(p),\nabla U(x)\rangle$ cancel out, leading to
\begin{align}
    \mathcal{L}\psi=&a\gamma nm+b\langle T_{\lambda}(p),\mathcal{J}_G^\top p\rangle-b\langle \nabla U(x), G(x) \rangle\nonumber\\
    &-a\beta\gamma\langle p,T_{\lambda}(p)\rangle-b\gamma\langle p, G(x) \rangle+b\gamma\beta^{-1}\langle \Gamma_{\delta}(p), G(x) \rangle.\label{calc2}
\end{align}
Moreover, in view of $M_{\delta}(p)T_{\delta}(p)=p$ and the identity
\begin{align}
    \sum_{j=1}^m\sum_{i,k=1}^n[M_{\delta}(p)]_{ik}\dfrac{\partial f}{\partial p_{ij}}\dfrac{\partial f}{\partial p_{kj}}=\langle \nabla_pf,M_{\delta}(p)\nabla_p f\rangle,\nonumber
\end{align}
one obtains the following equivalence for $f=H(x,p)$
\begin{align}
    \sum_{j=1}^m\sum_{i,k=1}^n[M_{\delta}(p)]_{ik}\dfrac{\partial H}{\partial p_{ij}}\dfrac{\partial H}{\partial p_{kj}}=\langle T_{\delta}(p),p\rangle.\label{calc3}
\end{align}
Therefore through Eq.\eqref{calc2}-\eqref{calc3}, we recover the master equation
\begin{align}
    \phi&=-\dfrac{\mathcal{A}\Psi}{\Psi}=-\mathcal{A}\psi-\dfrac{\gamma}{\beta}\langle\nabla_p \psi,M_{\delta}(p)\nabla_p \psi\rangle\nonumber\\
    &=-a\gamma nm-b\langle T_{\lambda}(p),\mathcal{J}_G^\top p\rangle+b\langle \nabla U(x), G(x) \rangle\nonumber\\
    &+a\beta\gamma\langle p,T_{\lambda}(p)\rangle+b\gamma\langle p, G(x) \rangle-b\gamma\beta^{-1}\langle \Gamma_{\delta}(p), G(x) \rangle\nonumber\\
    &-a^2\beta\gamma\langle T_{\delta}(p),p\rangle-2ab\gamma\langle G(x) ,p\rangle-b^2\gamma\beta^{-1}\langle G(x),M_{\delta}(p)G(x)\rangle.\nonumber
\end{align}
Rearranging the latter equation proves Eq.\eqref{master1}.

By setting $b=0$ in \eqref{calc2}--\eqref{calc3}, we obtain the energy identities
\begin{align}
    \mathcal{A}H
    &=\gamma\left(\frac{nm}{\beta}

<!-- lemma-7: body_md -->
\label{lem_lyapunov_drift}
Suppose that Assumptions \ref{ass:U_smooth},
\ref{ass:U_curvature} and \ref{ass:U_dissipativity} hold,
and normalize $U$ so that $U\geq0$.
Let $\Psi$ be defined by \eqref{s3_Psi}, and fix
$\delta,\beta,\gamma>0$ and $a,q\in(0,1)$.
Choose $b>0$ such that
\begin{align}
    b\leq\min\left\{
    \frac{a\beta}{2\sqrt{\delta}},\,
    \frac{qa(1-a)\beta\gamma}{2B_{\delta}},\,
    \beta\sqrt{\frac{qa(1-a)}{2\delta}}
    \right\}.
    \label{b_restriction}
\end{align}
If $B_{\delta}=0$, the middle restriction is omitted.

For any $\eta>0$, define
\begin{align*}
    \Lambda_q&:=(1-q)a(1-a)\beta\gamma,\\
    R&:=\frac{\gamma J+\eta}{\Lambda_q},
\end{align*}
and
\begin{align}
    y:=\max\left\{1,\frac{\sqrt{2}(\gamma J+\eta)}{b}\right\},
    \qquad
    r:=\frac{y+\sqrt{y^2+4w_1w_2}}{2w_1}.
    \label{x_radius}
\end{align}
Let
\[
    \mathcal{K}
    :=\{(x,p):S_{\delta}(p)\leq R,\ |x|\leq r\},
    \qquad
    U_r:=\sup_{|x|\leq r}U(x).
\]
Then $\Psi\geq1$, the set $\mathcal{K}$ is compact, and
\begin{align}
    \mathcal{A}\Psi
    \leq-\eta\Psi+C_{\mathcal{K}}\mathbb{I}_{\mathcal{K}},
    \label{LypCond}
\end{align}
where
\[
    C_{\mathcal{K}}
    :=(\eta+\gamma J)\exp\left(
    a\beta U_r+(a\beta+b\sqrt{\delta})R
    +\frac{2bd}{\sqrt{\delta}}\right).
\]
In particular, the drift condition holds for any fixed
$\delta,\beta,\gamma>0$ after choosing $b>0$ sufficiently small.
The constants and the compact set may depend on $\delta$.
<!-- end lemma-7 -->
<!-- proof-8: body_md -->
In view of \eqref{b_restriction}, we have
\begin{align*}
    bB_{\delta}+\dfrac{b^2\gamma\delta}{\beta}
    \leq qa(1-a)\beta\gamma.
\end{align*}
Therefore, setting
$\Lambda_q=(1-q)a(1-a)\beta\gamma>0$, we obtain
\begin{align}
    \phi\geq\Lambda_qS_{\delta}(p)
    +b\dfrac{|\nabla U(x)|^2}{\sqrt{1+|\nabla U(x)|^2}}
    -\gamma J.
    \label{phi_bound_coercive}
\end{align}

\noindent Moreover, the first restriction in \eqref{b_restriction}
ensures that the normalization in \eqref{s3_Psi} is valid. Indeed,
\begin{align}
    |p|\leq\sum_{i=1}^d\sigma_i(p)
    \leq\sqrt{\delta}K_{\delta}(p)
    +\dfrac{d}{\sqrt{\delta}},
    \label{p_K_bound}
\end{align}
and hence
\begin{align*}
    \log\Psi
    &=a\beta U(x)+a\beta K_{\delta}(p)
    +b\langle G(x),p\rangle+\dfrac{bd}{\sqrt{\delta}}\\
    &\geq a\beta U(x)
    +(a\beta-b\sqrt{\delta})K_{\delta}(p)
    \geq0.
\end{align*}
Consequently, $\Psi\geq1$.

\noindent\textbf{Domain (I)} Fix $\eta>0$ and introduce
\begin{align*}

    \qquad
    \Omega=\{p\in\mathbb{R}^{n\times m}:S_{\delta}(p)\geq R\}.
\end{align*}
If $p\in\Omega$, dropping the second non-negative term in
\eqref{phi_bound_coercive} yields
\begin{align}
    \phi\geq\Lambda_qR-\gamma J=\eta>0.
    \label{phi_bound3}
\end{align}
Notice that the threshold $R$ is chosen according to the constants
in the drift inequality. In view of \eqref{S_delta_coercive},
this is possible for any fixed $\delta>0$, without imposing
a smallness condition on $\delta$.

\noindent\textbf{Domain (II)} Suppose that $p\notin\Omega$.
Due to the dissipativity of $U(x)$, for $|x|>0$ we have
\begin{align*}
    |x||\nabla U(x)|
    &\geq\langle x,\nabla U(x)\rangle
    \geq w_1|x|^2-w_2,\\
    |\nabla U(x)|
    &\geq w_1|x|-\dfrac{w_2}{|x|}.
\end{align*}
Recall that
\begin{align*}
    y=\max\left\{1,\dfrac{\sqrt{2}(\gamma J+\eta)}{b}\right\},
    \qquad
    r=\dfrac{y+\sqrt{y^2+4w_1w_2}}{2w_1}.
\end{align*}
Then $|\nabla U(x)|\geq y\geq1$ whenever $|x|\geq r$.
Since $t^2/\sqrt{1+t^2}\geq t/\sqrt{2}$ for all $t\geq1$,
dropping the first non-negative term in
\eqref{phi_bound_coercive} yields
\begin{align}
    \phi
    \geq\dfrac{b}{\sqrt{2}}|\nabla U(x)|-\gamma J
    \geq\dfrac{by}{\sqrt{2}}-\gamma J
    \geq\eta>0.
    \label{phi_bound4}
\end{align}

\noindent\textbf{Domain (III)} Finally, when $p\notin\Omega$
and $|x|\leq r$, we land on the compact set
\begin{align*}
    \mathcal{K}
    =\{(x,p):S_{\delta}(p)\leq R,\ |x|\leq r\}.
\end{align*}
Indeed, \eqref{S_delta_coercive} gives
$|p|\leq\sqrt{\delta}R+\delta^{-1/2}$ on $\mathcal{K}$.
From \eqref{phi_bound_coercive}, one also obtains
$\phi\geq-\gamma J$. Combining the three domains,
we therefore get
\begin{align}
    \mathcal{A}\Psi
    \leq-\eta\Psi+(\eta+\gamma J)
    \left(\sup_{\mathcal{K}}\Psi\right)
    \mathbb{I}_{\mathcal{K}}.
    \label{drift_before_constant}
\end{align}

\noindent To quantify the constant, write
$U_r=\sup_{|x|\leq r}U(x)<\infty$. Notice that
\begin{align*}
    K_{\delta}(p)
    &=\dfrac{1}{\delta}\sum_{i=1}^d
    \left(\sqrt{1+\delta\sigma_i^2(p)}-1\right)\\
    &\leq\sum_{i=1}^d
    \dfrac{\sigma_i^2(p)}{\sqrt{1+\delta\sigma_i^2(p)}}
    =S_{\delta}(p).
\end{align*}
Consequently, using \eqref{p_K_bound}, for
$(x,p)\in\mathcal{K}$ we have
\begin{align*}
    \log\Psi(x,p)
    &\leq a\beta U_r+a\beta K_{\delta}(p)
    +b|p|+\dfrac{bd}{\sqrt{\delta}}\\
    &\leq a\beta U_r+(a\beta+b\sqrt{\delta})R
    +\dfrac{2bd}{\sqrt{\delta}}.
\end{align*}
If $U(x)\leq L(1+|x|^{\nu})$, one may additionally use
$U_r\leq L(1+r^{\nu})$.

\noindent Collecting the above bounds yields
\begin{align*}
    \mathcal{A}\Psi
    \leq-\eta\Psi+C_{\mathcal{K}}\mathbb{I}_{\mathcal{K}},
    \qquad \eta,C_{\mathcal{K}}>0,
\end{align*}
where
\begin{align*}
    C_{\mathcal{K}}
    :=(\eta+\gamma J)\exp\left(
    a\beta U_r+(a\beta+b\sqrt{\delta})R
    +\dfrac{2bd}{\sqrt{\delta}}\right).
\end{align*}
Thus, for any fixed $\delta,\beta,\gamma>0$ and $a\in(0,1)$,
a sufficiently small choice of $b>0$ gives the required
Lyapunov drift condition \eqref{LypCond}. The constants and
the compact set may depend on $\delta$.

\label{velocity_profile}
    Let $T>0$, $x_0,x_1\in\mathbb{R}^{n\times m}$ with $|x_1-x_0|_2<\delta^{-1/2}T$, and $q_0,q_1\in\mathcal{Q}_{\delta}$. Then there exists $\psi\in C^{\infty}([0,T];\mathcal{Q}_{\delta})$ such that
    \begin{align*}
        \psi(0)=q_0,\ \ \psi(T)=q_1,\ \ \int_0^{\top}\psi(t)dt=x_1-x_0.
    \end{align*}
<!-- end lemma-8 -->
<!-- proof-9: body_md -->
Let $\rho=\sqrt{\delta}|x_1-x_0|_2/T\in[0,1)$ and define the smooth step function:
    \begin{align}
        f(s)=\begin{cases}
            0\ &,\text{for } s\le 0\\
            \dfrac{e^{-1/s}}{e^{-1/s}+e^{-1/(1-s)}} &,\text{for } 0< s< 1\\
            1\ &,\text{for } s\geq 1
        \end{cases}\label{smooth_step}
    \end{align}
    It is well known that $f\in C^{\infty}(\mathbb{R};[0,1])$ and that all its derivatives vanish at $s=0$ and $s=1$. Moreover, $f(s)+f(1-s)=1$, hence $\int_0^1f(s)ds=1/2$. Now let us set the parameters:
    \begin{align*}
        \ell=\dfrac{(1-\rho)T}{4},\ \text{and } \psi^{*}=\dfrac{(x_1-x_0)-\ell(q_0+q_1)/2}{T-\ell}.
    \end{align*}
    Then we define $\psi:[0,T]\to\mathbb{R}^{n\times m}$ as follows:
    \begin{align}
        \psi(t)=\begin{cases}
            q_0+f(t/\ell)(\psi^{*}-q_0)&,0\leq t\leq\ell\\
            \psi^{*}&,\ell\leq t\leq T-\ell\\
            \psi^{*}+f((t-T+\ell)/\ell)(q_1-\psi^{*})&,T-\ell\leq t\leq T.
        \end{cases}\label{velocity_def}
    \end{align}
    Since all derivatives of $f$ vanish at $0$ and $1$, the three pieces glue smoothly, i.e. $\psi\in C^{\infty}([0,T];\mathbb{R}^{n\times m})$ with $\psi(0)=q_0$ and $\psi(T)=q_1$. Moreover, in view of $\int_0^1f(s)ds=1/2$, one obtains
    \begin{align*}
        \int_0^{\top}\psi(t)dt&=\ell\dfrac{q_0+\psi^{*}}{2}+(T-2\ell)\psi^{*}+\ell\dfrac{\psi^{*}+q_1}{2}\\
        &=\ell\dfrac{q_0+q_1}{2}+(T-\ell)\psi^{*}=x_1-x_0.
    \end{align*}
    It remains to show that $\psi$ takes values in $\mathcal{Q}_{\delta}$. Since $|q_0|_2,|q_1|_2<\delta^{-1/2}$, $|x_1-x_0|_2=\rho\delta^{-1/2}T$ and $\ell>0$, we have
    \begin{align*}
        |\psi^{*}|_2\leq\dfrac{|x_1-x_0|_2+\ell(|q_0|_2+|q_1|_2)/2}{T-\ell}<\dfrac{\rho\delta^{-1/2}T+\ell\delta^{-1/2}}{T-\ell}=\delta^{-1/2}\dfrac{1+3\rho}{3+\rho}\leq\delta^{-1/2},
    \end{align*}
    where the last inequality follows from $\rho<1$. Hence $\psi^{*}\in\mathcal{Q}_{\delta}$. Finally, since $f$ takes values in $[0,1]$, each $\psi(t)$ is a convex combination of $q_0,\psi^{*},q_1\in\mathcal{Q}_{\delta}$. Since $\mathcal{Q}_{\delta}$ is convex, it follows that $\psi(t)\in\mathcal{Q}_{\delta}$ for all $t\in[0,T]$.

\label{support1}
    Let $\mu$ be a Borel probability measure on a separable metric space $E$. Then the set
    \begin{align*}
        F=\{e\in E:\mu(\mathcal{B}_{\epsilon}(e))>0,\ \forall\epsilon>0\}
    \end{align*}

Let $x\in F^c$, then there exists $\epsilon>0$ such that $\mu(\mathcal{B}_{\epsilon}(x))=0$. Now choose arbitrary $x'\in\mathcal{B}_{\epsilon}(x)$ and let $0<\epsilon'\leq\epsilon-\mathrm{d}(x,x')$. For any $x''\in\mathcal{B}_{\epsilon'}(x')$ it follows that $\mathrm{d}(x'',x)\leq\mathrm{d}(x'',x')+\mathrm{d}(x',x)<\epsilon$, i.e. $\mathcal{B}_{\epsilon'}(x')\subseteq\mathcal{B}_{\epsilon}(x)$. This in turn implies that $\mu(\mathcal{B}_{\epsilon'}(x'))=0$, hence $x'\in F^c$. Therefore $\mathcal{B}_{\epsilon}(x)\subseteq F^c$, so $F^c$ is an open set covered by $\mu$-null balls. Moreover, since $E$ is separable, by Lindelof's Lemma, any open cover can be reduced to a countable subcover. Therefore, there exists a sequence $\{\mathcal{B}_{\epsilon_k}(x_k)\}_{k=1}^{\infty}$ of $\mu$-null balls that cover $F^c$. Then,

<!-- proposition-1: body_md -->
\label{minorization}
    For every nonempty compact set $C\subseteq\mathbb{R}^{2n\times m}$, there exist $\bar{T}_C>0$, $\alpha_C>0$ and a probability measure $\nu_C$ on $\mathscr{B}(\mathbb{R}^{2n\times m})$, such that
    \begin{align}
        R_{\bar{T}_C}(z,A)\geq\alpha_C\nu_C(A),\ \ \forall z\in C,\ A\in\mathscr{B}(\mathbb{R}^{2n\times m}).\label{minorization_ineq}
    \end{align}
    If $\text{int}C\neq\emptyset$, then $\nu_C$ may be chosen such that $\nu_C(C)=1$.
<!-- end proposition-1 -->
<!-- proof-15: body_md -->
First, suppose that $\text{int}C\neq\emptyset$. Choose $z_*=(x_*,p_*)\in\text{int}C$, fix $s>0$ and let $\epsilon>0$ be such that $\overline{\mathcal{B}_{\epsilon}(z_*)}\subset\text{int}C$. In view of Lemma \ref{support3}, we have $z_*\in\text{supp}R_s(z_*,\cdot)=\{(x,p):|x-x_*|_2\leq\delta^{-1/2}s\}$, i.e. $z_*$ is reachable from itself at time $s$. Therefore, by Lemma \ref{support1} and Eq.\eqref{kernelDensity}, one obtains
    \begin{align*}
        0<R_s(z_*,\mathcal{B}_{\epsilon}(z_*))=\int_{\mathcal{B}_{\epsilon}(z_*)}p_s(z_*,w)dw.
    \end{align*}
    Hence there exists $w_*\in\mathcal{B}_{\epsilon}(z_*)$ such that $p_s(z_*,w_*)>0$. Since the density $p_s$ is jointly continuous, there exist $\epsilon_0,\epsilon_1>0$ with $\overline{\mathcal{B}_{\epsilon_0}(z_*)},\overline{\mathcal{B}_{\epsilon_1}(w_*)}\subset\text{int}C$, and a constant $c_0>0$, such that
    \begin{align}
        p_s(v,w)\geq c_0,\ \ \forall v\in\mathcal{B}_{\epsilon_0}(z_*),\ w\in\mathcal{B}_{\epsilon_1}(w_*).\label{density_lower}
    \end{align}
    Next, we fix a single entrance time $\tau$ for the whole set $C$, namely
    \begin{align}
        \tau\geq T_C=\sqrt{\delta}\sup_{(x,p)\in C}|x_*-x|_2.\label{entrance_time}
    \end{align}
    Corollary \ref{accessibility} with target $z_1=z_*$ and $\epsilon=\epsilon_0$ yields $R_{\tau}(z,\mathcal{B}_{\epsilon_0}(z_*))>0$ for all $z\in C$. Moreover, $p_{\tau}$ is jointly continuous and hence bounded on the compact set $C\times\overline{\mathcal{B}_{\epsilon_0}(z_*)}$. Thus, by dominated convergence, the map $z\mapsto R_{\tau}(z,\mathcal{B}_{\epsilon_0}(z_*))=\int_{\mathcal{B}_{\epsilon_0}(z_*)}p_{\tau}(z,w)dw$ is continuous on $C$. Since $C$ is compact, the minimum is attained, i.e.
    \begin{align*}
        \kappa_C:=\min_{z\in C}R_{\tau}(z,\mathcal{B}_{\epsilon_0}(z_*))>0.
    \end{align*}
    Now let $A\in\mathscr{B}(\mathbb{R}^{2n\times m})$. The Chapman-Kolmogorov equation and Eq.\eqref{density_lower} yield, for all $z\in C$:
    \begin{align*}
        R_{\tau+s}(z,A)&=\int_{\mathbb{R}^{2n\times m}}R_s(v,A)R_{\tau}(z,dv)\geq\int_{\mathcal{B}_{\epsilon_0}(z_*)}R_s(v,A)R_{\tau}(z,dv)\\
        &\geq\int_{\mathcal{B}_{\epsilon_0}(z_*)}\left(\int_{A\cap\mathcal{B}_{\epsilon_1}(w_*)}p_s(v,w)dw\right)R_{\tau}(z,dv)\\
        &\geq c_0\text{Leb}(A\cap\mathcal{B}_{\epsilon_1}(w_*))R_{\tau}(z,\mathcal{B}_{\epsilon_0}(z_*))\geq c_0\kappa_C\text{Leb}(A\cap\mathcal{B}_{\epsilon_1}(w_*)).
    \end{align*}
    Therefore, setting
    \begin{align*}
        \bar{T}_C=\tau+s,\ \ \nu_C(A)=\dfrac{\text{Leb}(A\cap\mathcal{B}_{\epsilon_1}(w_*))}{\text{Leb}(\mathcal{B}_{\epsilon_1}(w_*))},\ \ \alpha_C=c_0\kappa_C\text{Leb}(\mathcal{B}_{\epsilon_1}(w_*))>0,
    \end{align*}
    one recovers Eq.\eqref{minorization_ineq}. Since $\mathcal{B}_{\epsilon_1}(w_*)\subset C$, we additionally get $\nu_C(C)=1$. For an arbitrary compact set $C$, there exists $r>0$ such that $C\subseteq\overline{\mathcal{B}_r(0)}$. The closed ball has nonempty interior, hence its minorization restricts to $C$, with $\bar{T}_C,\alpha_C,\nu_C$ being those of $\overline{\mathcal{B}_r(0)}$.

    K_{\delta}((1-\gamma\lambda)Y_n+\Delta_n)-K_{\delta}(Y_n)\leq-\gamma\lambda K_{\delta}(Y_n)+L_{\delta}|\Delta_n|\label{inq_kinetic1}.
\end{equation}
Moreover by the definition of $y$ and Remark \ref{SpectralProperties}.4, we obtain:
\begin{equation}
    |\nabla U(X_n)|\leq\gamma|Y_n-P_n|\leq \gamma|Y_n|+\gamma|P_n|\leq \gamma\delta^{1/2}K_{\delta}(Y_n)+\gamma\delta^{1/2}K_{\delta}(P_n)+2d\gamma\delta^{-1/2}.\label{inq_kinetic2}
\end{equation}
Recall the $L_0/L_1$-condition of the potential, i.e. $|\nabla^2U(x)|\leq L_0+L_1|\nabla U(x)|,\ \forall x\in\mathbb{R}^{m\times n}$. We define the auxiliary function $\phi(t)=L_0+L_1|\nabla U(x_t)|$ with $x_t=x+ty$, so that we can apply Grownwall's Lemma later. Additionally we define the upper-right Duni derivative:
\begin{equation}
    D^+\phi(t)=\limsup_{\epsilon\to 0}\dfrac{\phi(t+\epsilon)-\phi(t)}{\epsilon}
\end{equation}
Since $U$ is semi-convex, the Alexandrov Hessian exists a.e. and due to $\nabla U$ being locally Lipschitz, the fundamental theorem of calculus yields:
\begin{align}
    |\nabla U(x_{t+\epsilon})-\nabla U(x_t)|& \leq\int_t^{t+\epsilon}|\nabla^2U(x_s)[y]|ds\leq\int_t^{t+\epsilon}|\nabla^2U(x_s)|| y|ds\leq\int_t^{t+\epsilon}|y|(L_0+L_1|\nabla U(x_s)|)ds.\label{inq_kinetic3}
\end{align}
Now the reverse triangular inequality gives:
\begin{align}
    \phi(t+\epsilon)-\phi(t)&\leq L_1 |\nabla U(x_{t+\epsilon})-\nabla U(x_t)|\leq  |y|L_1 \int_t^{t+\epsilon}\phi(s)ds,\nonumber
\end{align}
Dividing by $\epsilon$, and taking the $\limsup$ as $\epsilon\to 0$ yields:
\begin{align}
    D^+\phi(t)\leq L_1|y|\phi(t)&\Rightarrow \phi(t)\leq \phi(0)\exp[{L_1t|y|}]\nonumber\\
    \Rightarrow(L_0+L_1|\nabla U(x_t)|)&\leq\exp[L_1t|y|](L_0+L_1|\nabla U(x)|),\label{gronwall1}
\end{align}
where the second to last step follows from Gronwall's Lemma. In view of Eq.\eqref{inq_kinetic3}, setting $\epsilon=1$, $t=0$, $x=X_n$, and $y=\lambda T_{\delta}(P_n)$, we get:
\begin{align}
    |\nabla U(X_{n+1})-\nabla U(X_n)|=|\nabla U(X_n+\lambda T_{\delta}(P_n))-\nabla U(X_n)|\leq e^{\lambda L_1 L_{\delta}}\lambda L_{\delta}(L_0+L_1|\nabla U(X_n)|).
\end{align}
Finally, Lemma [?] provides the bound $\text{div}[M_{\delta}(P_n)]\leq(d+1/2)\sqrt{d\delta}$. Combining the former result with the inequalities \eqref{inq_kinetic1},\eqref{inq_kinetic2} and \eqref{inq_kinetic3} yields:
\begin{align}
    K_{\delta}((1-\gamma\lambda)Y_n+\Delta_n)-K_{\delta}(Y_n)\leq-\gamma\lambda K_{\delta}(Y_n)+\hat{C}_2(K_{\delta}(Y_n)+K_{\delta}(P_n))+\hat{C}_1.\label{inq_kinetic4}
\end{align}

We start by providing two elementary bounds:
\begin{align}
    \text{Tr}(M_{\delta})&=\sum_{i=1}^m\sqrt{1+\delta \sigma_i^2(p)}=m+\delta\sum_{i=1}
^m\left(\dfrac{1}{\delta}\sqrt{1+\delta\sigma^2_i(p)}-1\right)=m+\delta K_{\delta}(p),\label{M_bound1}\\
|M_{\delta}(p)|_2&=\sqrt{1+\delta\sigma_1^2(p)}\leq 1+\delta^{1/2}\sigma_1(p)\leq1+\delta^{1/2}|p|\leq1+d+\delta K_{\delta}(p)\label{M_bound2}.
\end{align}
Since $T_{\delta}$ is $1-$Lipschitz (see Remark \ref{SpectralProperties}.2), we obtain the following by f.t.c.:
\begin{align}
    K_{\delta}(x+h)-K(x)&=\int_0^1\langle T_{\delta}(x+th),h\rangle dt\nonumber\\
    K_{\delta}(x+h)-K(x)-\langle T_{\delta}(x),h\rangle&=\int_0^1\langle T_{\delta}(x+th)-T(x),h\rangle dt\nonumber\\
    &\leq\int_0^1|h||T_{\delta}(x+th)-T_{\delta}(x)|dt\nonumber
\end{align}
Therefore,
\begin{align}
     K_{\delta}(x+h)-K(x)-\langle T_{\delta}(x),h\rangle\leq \int_0^1 t|DT_{\delta}|_2|h|^2dt\leq \dfrac{1}{2}|h|^2.\label{inq_kinetic7}
\end{align}
Next, we apply Eq.\eqref{inq_kinetic7} with $x=B_n$ and $h=\Xi_{n+1}$, to yield:
\begin{align}
    \mathbb{E}\left[K_{\delta}(B_n+\Xi_{n+1})-K_{\delta}(B_n)|\mathcal{F}_n\right]&=\mathbb{E}\left[\langle T_{\delta}(B_n),\Xi_{n+1}\rangle|\mathcal{F}_n\right]\nonumber\\&+\mathbb{E}\left[K_{\delta}(B_n+\Xi_{n+1})-K_{\delta}(B_n)-\langle T_{\delta}(B_n),\Xi_{n+1}\rangle |\mathcal{F}_n\right]\nonumber\\
    &\leq (1/2)\mathbb{E}\left[|\Xi_{n+1}|^2|\mathcal{F}_n\right]\leq (1/2)\text{Tr}(\Sigma)\nonumber\\
    &\leq \gamma\lambda\beta^{-1}(m+\delta K_{\delta}(P_n))
\end{align}
Finally, let us define the auxiliary function $\psi(\zeta)=K_{\delta}(x+A\zeta)-K_{\delta}(x)$. We apply Lemma \ref{GaussianConce} for the conditional expectation $\mathbb{E}[\cdot|\mathcal{F}_n]$ with $\psi(\zeta)=K_{\delta}(x+A\zeta)-K_{\delta}(x)$ for $\zeta=\xi_{n+1}$, $A=\Sigma^{1/2}$ and $x=B_n$. First, we show that $\psi$ is indeed Lipschitz:
\begin{align}
    |\psi(\zeta_1)-\psi(\zeta_2)|&\leq |K_{\delta}(x+A\zeta_1)-K_{\delta}(x+A\zeta_2)|\leq |T_{\delta}|_2|A(\zeta_1-\zeta_2)|.
\end{align}
Moreover, due to Eq.\eqref{M_bound2}, we have that $|\Sigma^{1/2}_n|_2\leq\sqrt{2\gamma\lambda\beta^{-1}}|M(P_n)|_2^{1/2}\leq \left(2\gamma\lambda\beta^{-1}[1+d+\delta K_{\delta}(P_n)]\right)^{1/2}$. Therefore, applying Lemma \ref{GaussianConce} for $\tau=\eta w$, we obtain:
\begin{align}
    \log\mathbb{E}^{\mathcal{F}_n}\left[e^{\eta w[K_{\delta}(B_n+\Xi_{n+1})-K_{\delta}(B_n)]}\right]&\leq \eta w\mathbb{E}^{\mathcal{F}_n}\left[K_{\delta}(B_n+\Xi_{n+1})-K_{\delta}(B_n)\right]+\dfrac{\eta^2w^2d}{2\delta}\dfrac{2\gamma\lambda}{\beta}(1+d+\delta K_{\delta}(P_n))\nonumber\\
    &\leq \dfrac{\eta w}{2}\dfrac{2\gamma\lambda}{\beta}d(m+\delta K_{\delta}(P_n))+\dfrac{\eta^2w^2d\gamma\lambda}{\delta\beta}(1+d+\delta K_{\delta}(P_n))\nonumber\\
    &\leq \eta\lambda A_3K_{\delta}(P_n)+\eta\lambda \hat{C}_4.\label{T2_bound}
\end{align}
where the second to last inequality follows from the elementary bound of Eq.\eqref{M_bound1}.
