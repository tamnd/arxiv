 such that

<!-- proposition-5: body_md -->
\label{prop:entropy}  Let $\Phi\in{\mathcal B}^{\mathop{\oslash}\nolimits^{\gamma}\!\!\!\!}$ with $\gamma>2d$ and $\omega\in{\mathcal S}_\mathrm{I}$. Let $\{\Lambda\}$ be an increasing family of cubes centered
at the origin that exhausts ${\mathbbm Z}^d$. Then, for any $t$,  
$$
\lim_{\{\Lambda\}}\frac1{|\Lambda|}
\left(S(\omega_{t,\Lambda})-S(\omega_{\Lambda,t})\right)=0.
$$
<!-- end proposition-5 -->
<!-- head-8: body_md -->
Fix $\delta\in(d,\gamma-d)$, choose $\{\Lambda'\}$ satisfying
Properties~(i)--(ii) of Lemma~\ref{lem:boxes}, and set
$\Lambda''=\Lambda\setminus\Lambda'$.  It suffices to show that for some constants $c_1, c_2$ and $c_3$,
\begin{equation}
\left|S(\omega_{t,\Lambda})-S(\omega_{\Lambda,t})\right|
\le c_1|\Lambda'|^2\left(1+\mathop{\mathrm{dist}}\nolimits(\Lambda',\Lambda^c)\right)^{-\delta}
+c_2|\Lambda''|+c_3.
\label{eq:bound_h}
\end{equation}
Indeed,  after division by $|\Lambda|$,
the right-hand side tends to zero by Lemma~\ref{lem:boxes},
proving the result. We proceed to  prove~\eqref{eq:bound_h}.


Denote by $\omega_{\Lambda,t|\Lambda'}$ the restriction of
$\omega_{\Lambda,t}$ to ${\mathfrak A}_{\Lambda'}$.
The Araki--Lieb inequality~\cite{Araki1970} gives  
\begin{align*}
S(\omega_{t,\Lambda'})-S(\omega_{t,\Lambda''})
\le\left|S(\omega_{t,\Lambda'})-S(\omega_{t,\Lambda''})\right|
&\le S(\omega_{t,\Lambda})\le S(\omega_{t,\Lambda'})+S(\omega_{t,\Lambda''}),\\[4pt]
S(\omega_{\Lambda,t|\Lambda'})-S(\omega_{\Lambda,t|\Lambda''})
\le\left|S(\omega_{\Lambda,t|\Lambda'})-S(\omega_{\Lambda,t|\Lambda''})\right|
&\le S(\omega_{\Lambda,t})
\le S(\omega_{\Lambda,t|\Lambda'})+S(\omega_{\Lambda,t|\Lambda''}).
\end{align*}
It follows that
\begin{align*}
S(\omega_{t,\Lambda})-S(\omega_{\Lambda,t})&\le S(\omega_{t,\Lambda'})+S(\omega_{t,\Lambda''})-S(\omega_{\Lambda,t|\Lambda'})+S(\omega_{\Lambda,t|\Lambda''})\\
&\le\left|S(\omega_{t,\Lambda'})-S(\omega_{\Lambda,t|\Lambda'})\right|+S(\omega_{t,\Lambda''})+S(\omega_{\Lambda,t|\Lambda''}),
\end{align*}
and similarly
\begin{align*}
S(\omega_{\Lambda,t})-S(\omega_{t,\Lambda})&\le S(\omega_{\Lambda,t|\Lambda'})+S(\omega_{\Lambda,t|\Lambda''})-S(\omega_{t,\Lambda'})+S(\omega_{t,\Lambda''})\\
&\le\left|S(\omega_{t,\Lambda'})-S(\omega_{\Lambda,t|\Lambda'})\right|+S(\omega_{t,\Lambda''})+S(\omega_{\Lambda,t|\Lambda''}).
\end{align*}
Thus, 
\begin{equation}
\left|S(\omega_{t,\Lambda})-S(\omega_{\Lambda,t})\right|
\leq\left|S(\omega_{t,\Lambda'})-S(\omega_{\Lambda,t|\Lambda'})\right|+S(\omega_{t,\Lambda''})+S(\omega_{\Lambda,t|\Lambda''}),
\label{eq:box_outer00}
\end{equation}
where, with $c_2=2\log\mathop{\mathrm{dim}}\nolimits\,{\mathcal H}$,
\begin{equation}
S(\omega_{t,\Lambda''})+ S(\omega_{\Lambda,t|\Lambda''})\leq c_2|\Lambda''|.
\label{eq:box_outer}
\end{equation}
For these estimates, see also~\cite[Proof of Proposition~2.1]{Wreszinski2019}
and~\cite[Eq.~(56)]{Wreszinski2022}.

For the remaining term in~\eqref{eq:box_outer00}, we use
an argument of Fannes~\cite{Fannes1973}.
 Set $M=\mathop{\mathrm{dim}}\nolimits\,{\mathcal H}_{\Lambda'}=\mathrm{e}^{h|\Lambda'|}$, where $h=\log\mathop{\mathrm{dim}}\nolimits\,{\mathcal H}$, and denote by
$(\alpha_l)_{l\in\llbracket1,M\rrbracket}$ and $(\beta_l)_{l\in\llbracket1,M\rrbracket}$ 
the eigenvalues of $\omega_{t,\Lambda'}$ and
$\omega_{\Lambda,t|\Lambda'}$, respectively, counted with multiplicity
and arranged in increasing order.
Set $\epsilon_l=\beta_l-\alpha_l$, and note that, by~\cite[Theorem~6.11]{Kato1966},
\begin{equation}
a_{\Lambda,\Lambda'}\coloneq\sum_{l\in\llbracket1,M\rrbracket}|\epsilon_l|
\le\|\omega_{t,\Lambda'}-\omega_{\Lambda,t|\Lambda'}\|_1\le2,
\label{eq:abound}
\end{equation}
where $\|\,{\bm\cdot}\,\|_1$ denotes the trace norm. Set ${\mathfrak s}(x)\coloneq-x\log x$ for $x\in(0,1]$, with ${\mathfrak s}(0)=0$.
By~\cite[Lemma~2]{Fannes1973}, for $x,y\in[0,1]$ we have
$$
|{\mathfrak s}(x)-{\mathfrak s}(y)|\leq{\mathfrak s}(|x-y|)+2|x-y|, 
$$
which implies that
\begin{equation}
\left|S(\omega_{t,\Lambda'})-S(\omega_{\Lambda,t|\Lambda'})\right|
\le\sum_{l\in\llbracket1,M\rrbracket}{\mathfrak s}(|\epsilon_l|)+4.
\label{eq:sum_eps0}
\end{equation}
If $a_{\Lambda,\Lambda'}=0$, then the left-hand side
of~\eqref{eq:sum_eps0} vanishes. Otherwise, invoking the concavity of the
logarithm, we get
\begin{align*}
\sum_{l\in\llbracket1,M\rrbracket}{\mathfrak s}(|\epsilon_l|)
&=a_{\Lambda,\Lambda'}
\sum_{\substack{l\in\llbracket1,M\rrbracket\\\epsilon_l\neq0}}
\frac{|\epsilon_l|}{a_{\Lambda,\Lambda'}}
\log\frac1{|\epsilon_l|}
\leq

\label{lem:Ruelle_chains}
For $K\subset{\mathbbm Z}^d$, $X\in{\mathcal F}$, $A\in{{\mathfrak{A}}_{\mathrm{loc}}}$, and $n\geq1$, one has
$$

$$
and
$$
\sum_{\boldsymbol{Y}\in{\mathcal F}^n }\|D_{\Psi|_K}(\boldsymbol{Y})A\|\leq\mathrm{e}^{\rho|K\cap\mathop{\mathrm{supp}}\nolimits A|}\frac{n!}{\Theta_\Psi^n}\|A\|.

Set $X=\mathop{\mathrm{supp}}\nolimits A$. By~\eqref{eq:need_chains3},
\begin{equation}
\sum_{\boldsymbol{Y}\in{\mathcal F}^n}\|D_{\Psi|_K}(\boldsymbol{Y})A\|
=\sum_{\boldsymbol{Y}\in{\mathcal C}^K_n(X)}\|D_\Psi(\boldsymbol{Y})A\|
\leq2^n\|A\|\sum_{\boldsymbol{Y}\in{\mathcal C}_n^K(X)}
\prod_{i=1}^n\|\Psi(Y_i)\|.
\label{eq:Ruelle_est0}
\end{equation}
To bound the last sum, we follow~\cite[Eq.~(6.6)]{Ruelle1969}.
Let $\boldsymbol{Y}=(Y_1,\ldots,Y_n)\in{\mathcal C}_n^K(X)$, set $Q_1\coloneq X\cap K$ and, for 
$i\in\llbracket2,n\rrbracket$, $Q_i\coloneq(X\cap K)\cup\bigcup_{\ell=1}^{i-1}Y_\ell$.
Since $|Q_i|\leq|X\cap K|+\sum_{\ell=1}^{i-1}(|Y_\ell|-1)$, we have
$$
\sum_{Y_i\cap Q_i\neq\emptyset}\|\Psi(Y_i)\|\leq|Q_i|\sum_{Z_i\ni0}\|\Psi(Z_i)\|
\leq\left(|X\cap K|+\sum_{\ell=1}^{i-1}(|Y_\ell|-1)\right)\sum_{k_i\ge0}
\sum_{\genfrac{}{}{0pt}{1}{Z_i\ni 0}{|Z_i|=k_i+1}}\|\Psi(Z_i)\|.
$$
Hence,
\begin{align*}
\sum_{\boldsymbol{Y}\in{\mathcal C}_n^K(X)}\prod_{i=1}^n\|\Psi(Y_i)\|
&\leq
\sum_{Y_1\colon Y_1\cap Q_1\neq\emptyset}\cdots
\sum_{Y_n\colon Y_n\cap Q_n\neq\emptyset}
\prod_{i=1}^n\|\Psi(Y_i)\|\\
&\leq\sum_{(k_1,\ldots,k_n)\in{\mathbbm N}^n}\prod_{i=1}^n
\left((|X\cap K|+k_1+\ldots+k_{i-1})\sum_{\genfrac{}{}{0pt}{1}{Z_i\ni0}{|Z_i|=k_i+1}}\|\Psi(Z_i)\|\right).
\end{align*}
The inequality
$$
\prod_{i=1}^n(|X\cap K|+k_1+\ldots+k_{i-1})\leq
(|X\cap K|+k_1+\ldots+k_n)^n
\le\mathrm{e}^{\rho|X \cap K|}\frac{n!}{\rho^n} \prod_{i=1}^n \mathrm{e}^{\rho k_i}
$$
gives
$$
\sum_{\boldsymbol{Y}\in{\mathcal C}_n^K(X)}\prod_{i=1}^n\|\Psi(Y_i)\|
\leq\frac{n!}{\rho^n}\mathrm{e}^{\rho|X\cap K|}
\left(\sum_{Z\ni0}\mathrm{e}^{\rho(|Z|-1)}\|\Psi(Z)\|\right)^n
\leq n!\,\mathrm{e}^{\rho|X\cap K|}\left(\frac{\|\Psi\|_\rho}{\rho}\right)^n. 
$$
This proves the first bound. The second follows
from~\eqref{eq:Ruelle_est0}. \hfill$\Box$

(L+1)\|\Theta\|_\mathrm{s}.
\]
Since every non-empty $X$ with $\mathop{\oslash}\nolimits(X)\leq L$ is contained in exactly
$N_L(X)$ intervals of the form ${\mathbbm I}_L^x$, we further have
\[
\sum_{x\in{\mathbbm Z}}D_{\widehat{\Theta}_K}(x)A
=
\sum_{x\in{\mathbbm Z}}
\sum_{\substack{\emptyset\neq X\subset K\cap{\mathbbm I}_L^x}}
\frac{\mathrm{i}[\Theta(X),A]}{N_L(X)}
=
\sum_{\substack{\emptyset\neq X\subset K\\X\cap\mathop{\mathrm{supp}}\nolimits A\neq\emptyset}}
\mathrm{i}[\Theta(X),A]
=
\delta_{\Theta|_K}(A),
\]
which proves~\eqref{eq:restricted_block_decomposition}.
\hfill$\Box$

$\Lambda=\llbracket a,b\rrbracket\subset{\mathbbm Z}$, the maps

Set $n_0\coloneq\max\{R,R'\}$. 
For $X\in{\mathcal F}$, $K\subset{\mathbbm Z}$, and $m\geq0$, define
$$
Q_K^{(m)}(X)
\coloneq
\delta_{\Phi|_K}^m(\Psi(X))
=
\sum_{\boldsymbol{Y}\in{\mathcal C}_m^K(X)}D_\Phi(\boldsymbol{Y})\Psi(X).
$$
If $\Psi(X)\neq0$, choose an interval $J_X$ of length $n_0+1$ containing
$X$. Lemma~\ref{lem:HHhat} gives
\begin{equation}
Q_K^{(m)}(X)
=
\sum_{\boldsymbol{y}\in{\mathcal C}'_{R',m}(J_X)}
D_{\widehat{\Phi}_K}(\boldsymbol{y})\Psi(X).
\label{eq:Q_block_expansion}
\end{equation}
For an interval $\Lambda\subset{\mathbbm Z}$, set
$$
B_\Lambda^{(m)}(X)
\coloneq
\sum_{\boldsymbol{Y}\in{\mathcal C}_m(X,\partial\Lambda)}
D_\Phi(\boldsymbol{Y})\Psi(X).
$$
By~\eqref{eq:need_chains4},
\begin{equation}
B_\Lambda^{(m)}(X)
=
\begin{cases}
Q_{\mathbbm Z}^{(m)}(X)-Q_\Lambda^{(m)}(X)
&\text{if }X\subset\Lambda;\\[4pt]
Q_{\mathbbm Z}^{(m)}(X)-Q_{\Lambda^c}^{(m)}(X)
&\text{if }X\subset\Lambda^c;\\[4pt]
Q_{\mathbbm Z}^{(m)}(X)
&\text{if }X\cap\Lambda\neq\emptyset\neq X\cap\Lambda^c.
\end{cases}
\label{eq:boundary_Q_decomposition}
\end{equation}
For $|t|<\Theta_\Phi$, Lemma~\ref{lem:WG_formula} and
\eqref{eq:boundary_Q_decomposition} give
\begin{equation}
W_\Lambda(t)
=
\sum_{m\geq0}\frac{(-t)^m}{m!}
\sum_{X\in{\mathcal F}}B_\Lambda^{(m)}(X).
\label{eq:formula_Wt_1d}
\end{equation}
We proceed to  show that the series on the right-hand side converges
locally uniformly in $t\in{\mathbbm C}$ and hence defines an entire ${\mathfrak A}$-valued
function.

If $B_\Lambda^{(m)}(X)\neq0$ and $\Psi(X)\neq0$, then
$$
X\cap{\mathcal T}_m^\Lambda\neq\emptyset,
\qquad
{\mathcal T}_m^\Lambda
\coloneq
\left\{
x\in{\mathbbm Z}\mid
\mathop{\mathrm{dist}}\nolimits(\{x\},\partial\Lambda)\leq R+mR'
\right\}.
$$
There exists a constant $c>0$, independent of $\Lambda$ and $m$, such that
\begin{equation}
|{\mathcal T}_m^\Lambda|
\leq
c(1+m).
\label{eq:one_dim_tube_bound}
\end{equation}
Moreover, Lemma~\ref{lem:delta_Araki} and
\eqref{eq:Q_block_expansion} give
$$
\|Q_K^{(m)}(X)\|
\leq
C_\Phi^m
|{\mathcal C}'_{R',m}(n_0)|
\|\Psi(X)\|,
$$
and hence
$$
\|B_\Lambda^{(m)}(X)\|
\leq
2C_\Phi^m
|{\mathcal C}'_{R',m}(n_0)|
\|\Psi(X)\|.
$$
It follows that
\begin{equation}
\sum_{m\geq0}\frac{|t|^m}{m!}
\sum_{X\in{\mathcal F}}\|B_\Lambda^{(m)}(X)\|
\leq
2c\|\Psi\|_\mathrm{s}
\sum_{m\geq0}
(1+m)
\frac{(C_\Phi|t|)^m}{m!}
|{\mathcal C}'_{R',m}(n_0)|
<\infty,
\label{mar-phy-bb}
\end{equation}
where the last assertion follows from Lemma~\ref{lem:HHhat}(i). The same estimates apply to the Taylor series of
$W_{\Lambda,\widetilde\Lambda}(t)$, uniformly in
$\widetilde\Lambda\supset\Lambda$. Since the interactions have finite
range, each coefficient agrees with the corresponding coefficient
in~\eqref{eq:formula_Wt_1d} once $\widetilde\Lambda$ is sufficiently
large. The  bound~\eqref{mar-phy-bb} gives locally uniform convergence on
${\mathbbm C}$ to the right-hand side of~\eqref{eq:formula_Wt_1d}.
For real $t$, Proposition~\ref{prop:formula_W} identifies this limit
with $W_\Lambda(t)$. Thus~\eqref{eq:formula_Wt_1d} holds for every
$t\in{\mathbbm R}$ and defines an entire ${\mathfrak A}$-valued extension of
$t\mapsto W_\Lambda(t)$.




Fix $a_0>0$ and set $g\coloneq G(t,a_0,n_0)$. 
For $|\mathop{\mathrm{Im}}\nolimits z|<a_0$, Proposition~\ref{prop:arakis_bound} and
\eqref{eq:Q_block_expansion} give
\begin{equation}

|{\mathcal C}'_{R',m}(n_0)|

\|\Psi(X)\|,
\label{eq:Q_alpha_bound}
\end{equation}

\eqref{eq:formula_Wt_1d},
\eqref{eq:boundary_Q_decomposition},
\eqref{eq:one_dim_tube_bound}, and

\frac{(C_\Phi g|t|)^m}{m!}
|{\mathcal C}'_{R',m}(n_0)|.
\end{align*}

\sum_{X\in{\mathcal F}}\alpha^z(B_\Lambda^{(m)}(X))

arbitrary, the series provides  the required entire extension.
