{ For \(a,b\in\mathbb R\), write \(a\wedge b:=\min\{a,b\}\),
\(a\vee b:=\max\{a,b\}\), and \(a^+:=a\vee0\).

Its subspace \(C_b^2(U)\) consists of functions that, together with

Similarly, \(C_b^\infty(U)\) denotes the space of smooth functions

<!-- definition-1: body_md -->
\label{def:coordinatewise}
	Assume \eqref{eq:boundary-compatible} holds.
	We call \(Z=(Z_t)_{t\ge0}\), with \(Z_t=(X_t,Y_t)\),
	a \emph{coordinatewise process} (CWP)
	if it satisfies the following:
	Before \(\tau_0\wedge\tau_\infty\), it follows
	\eqref{eq:model}. On \(\{\tau_0<\tau_\infty\}\),
	if \(Y\) reaches zero first, then \(Y\) remains zero and
	the surviving coordinate solves
	\[
	\begin{aligned}
		X_t={}&X_{\tau_0}
		-\int_{\tau_0}^t b_{10}X_s^{r_{10}}\,\mathrm d s
		+\int_{\tau_0}^t\sqrt{2b_{11}X_s^{r_{11}}}\,\mathrm d B_1(s)\\
		&+\int_{\tau_0}^t\int_0^\infty
		\int_0^{b_{12}X_{s-}^{r_{12}}}
		z\,\widetilde N_1(\,\mathrm d s,\,\mathrm d z,\,\mathrm d u).
	\end{aligned}
	\]
	If \(X\) reaches zero first, then \(X\) remains zero and
	the surviving coordinate solves
	\[
	\begin{aligned}
		Y_t={}&Y_{\tau_0}
		-\int_{\tau_0}^t b_{20}Y_s^{r_{20}}\,\mathrm d s
		+\int_{\tau_0}^t\sqrt{2b_{21}Y_s^{r_{21}}}\,\mathrm d B_2(s)\\
		&+\int_{\tau_0}^t\int_0^\infty
		\int_0^{b_{22}Y_{s-}^{r_{22}}}
		z\,\widetilde N_2(\,\mathrm d s,\,\mathrm d z,\,\mathrm d u).
	\end{aligned}
	\]
	If both coordinates are zero, the process remains at the origin.
	On \(\{\tau_\infty\le\tau_0\}\), it is killed at
	\(\tau_\infty\). For a boundary initial state, set
	\(\tau_0=0\) and start the corresponding axis dynamics
	at time zero.

<!-- definition-2: body_md -->
\label{def:absorbed}
We call \(Z^{\tau_0}=(Z_t^{\tau_0})_{t\ge0}\)
a \emph{source-stopped process} (SSP)
if it satisfies the following: for an initial state in
\(E^\circ\), it follows \eqref{eq:model} up to
\(\tau_0\wedge\tau_\infty\), remains at \(Z_{\tau_0}\)
after \(\tau_0\) on \(\{\tau_0<\tau_\infty\}\), and is
killed at \(\tau_\infty\) on \(\{\tau_\infty\le\tau_0\}\).
For an initial state \((x_0,y_0)\in\partial E\), set
\(\tau_0=0\) and \(Z_t^{\tau_0}=(x_0,y_0)\) for all
\(t\ge0\).

 (p/r)\wedge(q/s),&r+s>1,\\

 \end{cases}

 \!\!\!&\!\!\!& +c_1\int_0^t\int_0^\infty\int_0^{b_{22}Y_{s-}^{r_{22}}}
 z\,\widetilde N_2(\,\mathrm d s,\,\mathrm d z,\,\mathrm d u).

\end{eqnarray*}

 \right]

\label{lem:time-average-limits}
Let \(D=E\) or \(E^\circ\), and let \((Q_t)_{t\ge0}\)
be the transition semigroup of a conservative weak Feller
Markov process with c\`adl\`ag paths in \(D\).
For \(z\in D\), define
\[
\nu_T^z:=\frac1T\int_0^T Q_s(z,\cdot)\,\mathrm d s.
\]
Then every weak subsequential limit of \(\nu_T^z\) as
\(T\to\infty\) is invariant for \((Q_t)_{t\ge0}\).
If \(\{\nu_T^z:T\ge1\}\) is tight on \(D\), an invariant
probability exists. If, in addition, the invariant probability
is unique, say \(\pi\), then
\(\nu_T^z\Rightarrow\pi\) as \(T\to\infty\).
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
We use the Krylov--Bogoliubov argument; see
\cite[Theorem~3.1.1, p.~21]{DaPratoZabczyk1996}.
For \(t\ge0\) and \(f\in C_b(D)\), the semigroup property gives
\begin{eqnarray*}
 \left|\int_D Q_tf\,\,\mathrm d\nu_T^z-\int_D f\,\,\mathrm d\nu_T^z\right|
 =\frac1T\left|
 \int_T^{T+t}Q_sf(z)\,\mathrm d s-\int_0^tQ_sf(z)\,\mathrm d s
 \right| \le\frac{2t\|f\|_\infty}{T}.
 \end{eqnarray*}
If \(\nu_{T_k}^z\Rightarrow\pi\) for some \(T_k\to\infty\),
the weak Feller property allows passage to the limit,
yielding
\[
 \int_D Q_tf\,\,\mathrm d\pi=\int_D f\,\,\mathrm d\pi.
\]
Thus \(\pi\) is invariant. Tightness and Prokhorov's theorem
give such a subsequential limit. If the invariant probability
is unique, every weak subsequential limit equals \(\pi\). Then by tightness, we have  \(\nu_T^z\Rightarrow\pi\). The result follows.

		+c\mathbb E_{x_0,y_0}\int_0^{t\wedge\tau_0}
		W(Z_s)\,\mathrm d s \le H(x_0,y_0)+C\mathbb E_{x_0,y_0}(t\wedge\tau_0).

\begin{eqnarray*}

\end{aligned}

\end{equation}

\end{align*}

	+\frac{\kappa_1}{B+\kappa_1}x^{\kappa_2}y^{-B}

	\bigl(x^{\kappa_2}y^{-B}\bigr)^{\kappa_1/(B+\kappa_1)} 
	=x^{-\eta} }

and 
\begin{eqnarray*}
 \frac{\kappa_2}{A+\kappa_2}x^{-A}y^{\kappa_1}
 +\frac{A}{A+\kappa_2}x^{\kappa_2}y^{-B}
  \ge y^{-\zeta}.
\end{eqnarray*}
Consequently,

 \ge x^{-\eta}\vee y^{-\zeta}

\begin{equation}

 \label{eq:cooperative-cutoff-coercivity}
\end{equation}

 \label{eq:cooperative-power-absorption}
\end{equation}

	Let \(Z^{(m)}\) and \(\widetilde Z^{(m)}\) be two solutions
	of \eqref{eq:imm-truncated-model} driven by the same noises, and let $\Delta X^{(m)}:=X^{(m)}-\widetilde X^{(m)},$ $
	\Delta Y^{(m)}:=Y^{(m)}-\widetilde Y^{(m)}$ and $D_t^{(m)}:=|\Delta X_t^{(m)}|+|\Delta Y_t^{(m)}|.$
	By the same zero-local-time criterion, one can check that
	\(L_t^0(\Delta X^{(m)})=L_t^0(\Delta Y^{(m)})=0\).
	Applying Tanaka's formula and taking expectations, as in
	\cite[Proposition~3.1]{LiMytnik2011}, we have
	\[
	\mathbb E D_t^{(m)}\le D_0^{(m)}
	+C_m\int_0^t\mathbb E D_s^{(m)}\,\mathrm d s.
	\]
It follows that $\mathbb E D_t^{(m)}\le\mathrm e^{C_mt}D_0^{(m)}$.
	By \cite[Theorems~1 and~2]{BarczyLiPap2015},
	\eqref{eq:imm-truncated-model} therefore has a pathwise unique
	adapted strong c\`adl\`ag solution for every \(m\).
	
	For later use, the same argument applies to two solutions
	\(Z=(X,Y)\) and \(\widetilde Z=(\widetilde X,\widetilde Y)\)
	of \eqref{eq:model}, driven by the same noises, with initial
	states \((x_0,y_0)\) and \((\widetilde x_0,\widetilde y_0)\).
	For
	\(R>x_0\vee y_0\vee\widetilde x_0\vee\widetilde y_0\), let
	\[
	\sigma_R:=\inf\{t\ge0:
	X_t\vee Y_t\vee\widetilde X_t\vee\widetilde Y_t\ge R\},
	\qquad D_t:=|X_t-\widetilde X_t|+|Y_t-\widetilde Y_t|.
	\]
	By Gronwall's inequality, it gives that
	\begin{equation}
		\mathbb E D_{t\wedge\sigma_R}\le\mathrm e^{C_Rt}D_0,
		\qquad t\ge0.
		\label{eq:stopped-L1-continuity}
	\end{equation}
	In particular, pathwise uniqueness holds up to the explosion
	time. 
As in the proof of \Cref{prop:axis-realization},
we construct all \(Z^{(m)}\) with the same initial state and
driving noises, and $\rho_n^{(m)}:=\inf\{t\ge0:
X_t^{(m)}\vee Y_t^{(m)}\ge n\}$ for $m\ge n>x_0\vee y_0.$
By \eqref{eq:stopped-L1-continuity},
\(\rho_n^{(m)}=\rho_n^{(n)}\) and the stopped solutions
are consistent. The same patching argument shows that $
\tau_\infty:=\lim_{n\to\infty}\rho_n^{(n)}$ and $Z_t:=Z_t^{(n)}$ if $t<\rho_n^{(n)}$
defines an \(E\)-valued strong c\`adl\`ag solution of
\eqref{eq:model} on \([0,\tau_\infty)\). Then
pathwise uniqueness follows from
\eqref{eq:stopped-L1-continuity} by letting \(R\to\infty\).
