\label{prop:exact-decomp}
Let $\widehat\pi_{\bm{n}}$ be the estimator above, with $\bm{U} := \widehat{\bm{V}}_r^{\bm{C}\top}\bm{Z}$. Under Assumption~2.1 of Part~I and the integrability condition of \textup{(M)}, every term below is finite and, for each realization of the estimation sample,
\begin{equation}
\begin{aligned}
  \mathcal{D}_{\mathrm{KL}}(\pi_{\bm{n}}\,\|\,\widehat\pi_{\bm{n}})
  \;=\;
  &\underbrace{\textstyle\sum_{i=1}^d \mathcal{D}_{\mathrm{KL}}(\pi_i\,\|\,\widehat\pi_i) \;+\; R_N \;+\; \log\widehat Z_{\mathrm{comp}}}_{\textstyle \mathcal E_{\rm marg}\ \text{(the marginals)}}
  \;+\;
  \underbrace{\mathcal{D}_{\mathrm{KL}}\bigl(\pi_{\bm{n}}\,\|\,\pi_{\bm{n}}(\,\cdot\,;\widehat{\bm{V}}_r^{\bm{C}})\bigr)}_{\textstyle \mathcal E_{\rm proj}\ \text{(the subspace)}}
  \\[4pt]
  &\;+\;
  \underbrace{\mathcal{D}_{\mathrm{KL}}\bigl(\pi_{\bm{U}}\,\|\,\widehat\pi_{\bm{U}}\bigr)}_{\textstyle \mathcal E_{\rm dens}\ \text{(the reduced density)}},
\end{aligned}
  \label{eq:kl-three-way}
\end{equation}
where $R_N := \mathbb{E}_{\pi_{\bm{n}}}[\log\widehat c^{Z}(\mathcal{Z})] - \mathbb{E}_{\pi_{\bm{n}}}[\log\widehat c^{Z}(\widehat\mathcal{Z})]$ is the difference between the expected log copula factors evaluated at the exact and estimated rank transforms. The normalizing constant $\widehat Z_{\mathrm{comp}}$ is defined above, and $\mathcal E_{\rm proj} = \mathbb{E}_{\pi_{\bm{U}}}[\mathcal{D}_{\mathrm{KL}}(\pi_{\bm{W}\mid\bm{U}}\,\|\,\gamma_{d-r})]$.

<!-- proposition-2: body_md -->
\label{prop:stage1-cert}
Let $\widehat{\bm{V}}_r^{\bm{C}}$ be the estimated Stage-1 basis of \textup{(3.5)} and $\widehat{\bm{P}}_r := \widehat{\bm{V}}_r^{\bm{C}}\widehat{\bm{V}}_r^{\bm{C}\top}$. Under Assumption~2.1 of Part~I, for every score field $\mathcal{S}' \in L^2(\pi_{\bm{Z}};\mathbb{R}^d)$ with second-moment matrix $\bm{C}' := \mathbb{E}_{\pi_{\bm{Z}}}[\mathcal{S}'\mathcal{S}'^{\top}]$,
\begin{equation}
  \sqrt{2\,\mathcal E_{\rm proj}}
  \;\le\;
  \|\mathcal{S} - \mathcal{S}'\|_{L^2(\pi_{\bm{Z}})} \;+\; \sqrt{\operatorname{Tr}\!\bigl((\bm{I}-\widehat{\bm{P}}_r)\bm{C}'\bigr)}.
  \label{eq:finite-sample-kl}
\end{equation}
For $\mathcal{S}' = \mathcal{S}$, the score-error term in \eqref{eq:finite-sample-kl} vanishes. Squaring the remaining inequality and dividing by two yields the oracle bound
\begin{equation}
  \mathcal E_{\rm proj} \;=\; \mathcal{D}_{\mathrm{KL}}\!\bigl(\pi_{\bm{n}}\,\|\,\pi_{\bm{n}}(\,\cdot\,;\widehat{\bm{V}}_r^{\bm{C}})\bigr)
  \;\le\;
  C_1 \;:=\; \tfrac12\operatorname{Tr}\!\bigl((\bm{I}-\widehat{\bm{P}}_r)\bm{C}\bigr),
  \label{eq:stage1-cert}
\end{equation}
The oracle bound uses half the trace of $\bm{C}$ over the discarded subspace. For comparison, let $\bm{P}_r^{\bm{C}} := \bm{V}_r^{\bm{C}}\bm{V}_r^{\bm{C}\top}$ project onto the top-$r$ eigenspace of $\bm{C}$. Subtracting and adding the trace at this projector separates the rank and suboptimality contributions:
\begin{equation}
  C_1 \;=\; \underbrace{\tfrac12\textstyle\sum_{i>r}\lambda_i(\bm{C})}_{\text{rank}} \;+\; \underbrace{\tfrac12\bigl[\operatorname{Tr}(\bm{P}_r^{\bm{C}}\bm{C}) - \operatorname{Tr}(\widehat{\bm{P}}_r\bm{C})\bigr]}_{\Delta_{\rm subopt}\,\ge\,0},
  \label{eq:stage1-split}
\end{equation}
where the rank term is fixed by $\bm{C}$ and $r$. The remaining term $\Delta_{\rm subopt}$ is of order $\|\sin\Theta(\widehat{\bm{V}}_r^{\bm{C}},\bm{V}_r^{\bm{C}})\|_F^2$. It is zero whenever $\mathrm{span}(\widehat{\bm{V}}_r^{\bm{C}}) = \mathrm{span}(\bm{V}_r^{\bm{C}})$, and only in this case if $\lambda_r(\bm{C}) > \lambda_{r+1}(\bm{C})$.

<!-- theorem-1: body_md -->
\label{thm:stage1-rate}
Let $\widehat{\bm{V}}_r^{\bm{C}}$ be the estimated Stage-1 basis of \textup{(3.5)}, let $\widehat{\bm{P}}_r := \widehat{\bm{V}}_r^{\bm{C}}\widehat{\bm{V}}_r^{\bm{C}\top}$, and let $\lambda_1(\bm{C}_\Lambda) \ge \lambda_2(\bm{C}_\Lambda) \ge \cdots$ be the eigenvalues of $\bm{C}_\Lambda$. Under Assumption~2.1 of Part~I,
\begin{equation}
\begin{aligned}
  \sqrt{2\,\mathcal{D}_{\mathrm{KL}}\!\bigl(\pi_{\bm{n}}\,\|\,\pi_{\bm{n}}(\,\cdot\,;\widehat{\bm{V}}_r^{\bm{C}})\bigr)}
  \;\le\;
  &\underbrace{\vphantom{\sqrt{\textstyle\sum_{i>r}\lambda_i(\bm{C}_\Lambda)}}\|\mathcal{S} - \mathcal{S}_\Lambda\|_{L^2(\pi_{\bm{Z}})}}_{\text{Hermite multi-index truncation}}
  \;+\;
  \underbrace{\sqrt{\textstyle\sum_{i>r}\lambda_i(\bm{C}_\Lambda)}}_{\text{subspace rank truncation}}
  \\[2pt]
  &\;+\;
  \underbrace{\sqrt{\operatorname{Tr}\!\bigl((\bm{P}_r^{\Lambda\star}-\widehat{\bm{P}}_r)\bm{C}_\Lambda\bigr)}}_{\text{subspace statistical estimation}}.
\end{aligned}
  \label{eq:stage1-threeterm}
\end{equation}
If the moment and regularity conditions \textup{(A1)}--\textup{(A3)} of \Cref{app:sample-proof} hold as well, then the third term satisfies
\begin{equation}
  \operatorname{Tr}\!\bigl((\bm{P}_r^{\Lambda\star}-\widehat{\bm{P}}_r)\bm{C}_\Lambda\bigr)
  \;\le\;
  2r\,\|\widehat{\bm{C}} - \bm{C}_\Lambda\|_{\rm op}
  \;=\;
  O_p(N^{-1/2}) \;+\; O(\tau_N)
  \label{eq:stage1-estrate}
\end{equation}
at the Tikhonov regularization strength $\tau_N$ of \textup{(A3)} \cite{EnglHankeNeubauer1996}.

<!-- theorem-2: body_md -->
\label{thm:stage2-kl}
Let $\widehat\pi_{\bm{U}}$ be an exactly normalized Stage-2 density satisfying a logarithmic Sobolev inequality with $C_{\rm LSI}(\widehat{\bm{\theta}}^{r})>0$, as ensured by the global constraints in \Cref{prop:lsi-holds}. Let $\mathcal{L}_r^\star$ be the Fisher projection over $\Lambda_r$, and $\mathcal J_{\rm est}(\widehat{\bm{\theta}}^{r}):=\|\nabla(\mathcal{L}_r^\star-\mathcal{L}_r(\cdot;\widehat{\bm{\theta}}^{r}))\|_{L^2(\pi_{\bm{U}})}^2$. Under \textup{(H1)--(H2)} and the population score-matching regularity of \Cref{app:bisc-ridge}, for every realization,
\begin{equation}
\mathcal{D}_{\mathrm{KL}}(\pi_{\bm{U}}\,\|\,\widehat\pi_{\bm{U}})
\le\frac{\mathcal J_{\rm trunc}+\mathcal J_{\rm est}(\widehat{\bm{\theta}}^{r})}{2C_{\rm LSI}(\widehat{\bm{\theta}}^{r})}.
\label{eq:stage2-kl}
\end{equation}
Here $\mathcal J_{\rm trunc}\le C_\star K^{-(2s-1)}\|\mathcal{L}_r\|_{H^s_{\gamma}}^2$, with
$K:=\min\{|\alpha|_1\ge1:\alpha\notin\Lambda_r,\ \vartheta_\alpha\ne0\}$, and $\mathcal J_{\rm trunc}=0$ when this set is empty. Under the hypotheses of \Cref{thm:ccsm-rate} and \textup{(H3)}: $C_{\rm LSI}(\widehat{\bm{\theta}}^{r})\ge c_{\rm LSI}>0$ almost surely,
\begin{equation}
\mathbb{E}[\mathcal{D}_{\mathrm{KL}}(\pi_{\bm{U}}\,\|\,\widehat\pi_{\bm{U}})]
\le\mathcal B_{\rm dens}+O(N^{-1}),\qquad
\mathcal B_{\rm dens}:=
\frac{\mathcal J_{\rm trunc}+\mathcal E_{\rm est}+16\mathbb{E}[d_N^2\mathbf1_{\mathcal G}]}{2c_{\rm LSI}}.
\label{eq:stage2-kl-mean}
\end{equation}
where $\mathcal E_{\rm est}$, the event $\mathcal G$ and the constraint distance $d_N$ are those of \Cref{thm:ccsm-rate}. For constants chosen from a finite grid, assume that every pair satisfies the hypotheses with a common $c_{\rm LSI}$. Summing $\mathcal E_{\rm est}$ and the constraint term over the grid makes \eqref{eq:stage2-kl-mean} valid for any selection rule.

<!-- proposition-3: body_md -->
\label{prop:lsi-holds}
Let $\widehat\pi_{\bm{U}} = \widehat Z_r^{-1}\exp\bigl(\textstyle\sum_{\alpha\in\Lambda_r}\widehat\theta^r_\alpha H_\alpha\bigr)\,\gamma_r$, and let $K_{\rm eff} \ge 4$ be the largest even integer at most $K_2$. Assume the coefficients satisfy the constraints of the Stage-2 solve (\csname Cref@appendix@name\endcsname~E of Part~I). In particular, coefficients of degree $K_2$ vanish when $K_2$ is odd, and the leading form $F[\widehat{\bm{\theta}}^{r}](\bm{v}) := \sum_{|\alpha|_1 = K_{\rm eff}}\widehat\theta^r_\alpha\,\bm{v}^\alpha/\sqrt{\alpha!}$ satisfies $F[\widehat{\bm{\theta}}^{r}](\bm{v}) \le -\varepsilon$ for every $\|\bm{v}\| = 1$. Then $\widehat\pi_{\bm{U}}$ is a probability measure and satisfies a logarithmic Sobolev inequality with a constant $C_{\rm LSI} > 0$.

\label{prop:apriori-rates}
Compute the Stage-2 coefficients from exact-transform observations independent of the sample used to compute $\widehat{\bm{V}}_r^{\bm{C}}$. <!-- corollary-1: body_md -->
\label{cor:apost-den}
Assume, conditionally on the estimation sample: \textup{(D1)} $\pi_{\bm{U}}$ belongs to the class $\mathcal F_{r,\theta}$ of \cite[\S2]{BerrettSamworthYuan2019}, with $\alpha>r$ moments and smoothness $\beta>r/2$; \textup{(D2)} $\mathrm{Var}_{\pi_{\bm{U}}}\log\pi_{\bm{U}}<\infty$ and $\mathrm{Var}_{\pi_{\bm{U}}}\log\widehat\pi_{\bm{U}}<\infty$; \textup{(D3)} $\bm{x}_1,\dots,\bm{x}_M$ are i.i.d.\ $\pi_{\bm{n}}$ and independent of the estimation sample. Take $w^{(k)}\in\mathcal W^{(k)}$ with $\sup_k\|w^{(k)}\|<\infty$ and $k=k_M$ as in \cite[Thm.~1]{BerrettSamworthYuan2019}. Then $\widehat{\mathcal E}_{\rm dens}\to\mathcal E_{\rm dens}$ in probability, and
\begin{equation}
  \sqrt M\,\bigl(\widehat{\mathcal E}_{\rm dens} - \mathcal E_{\rm dens}\bigr)
  \;\xrightarrow{\;d\;}\;
  \mathcal N\bigl(0,\ \sigma_{\rm den}^2\bigr),
  \qquad
  \sigma_{\rm den}^2 \;=\; \mathrm{Var}_{\pi_{\bm{U}}}\!\Bigl[\log\frac{\pi_{\bm{U}}(\bm{U})}{\widehat\pi_{\bm{U}}(\bm{U})}\Bigr],
  \label{eq:apost-clt}
\end{equation}
the variance of the oracle log-ratio estimator. For each marginal divergence, the corresponding one-dimensional hypotheses give the same conclusion. Minkowski's inequality also gives
\[
\sigma_{\rm den}\le\sqrt{\mathrm{Var}_{\pi_{\bm{U}}}\log\pi_{\bm{U}}}
+\sqrt{\mathrm{Var}_{\pi_{\bm{U}}}\log\widehat\pi_{\bm{U}}}.
\]
The variance estimator of \cite[\S2]{BerrettSamworthYuan2019} and the sample variance of $\log\widehat\pi_{\bm{U}}(\bm{u}_m)$ estimate the first and the second variance consistently. We report the sum of their square roots divided by $\sqrt M$ as an estimate of an upper bound on the asymptotic standard error.

<!-- corollary-2: body_md -->
\label{cor:aposteriori}
Divide the $N$ samples into equal halves. Fit the rank transform and the Stage-1 coefficients $\widehat{\bm{\theta}}^{d}$ on the first half, and write $\bm{Z}^{(1)},\dots,\bm{Z}^{(N_2)}$, $N_2 := N/2$, for the transformed second half. Evaluate $\widehat\mathcal{S}(\bm{Z}^{(n)})$ to form the empirical second-moment matrix $\widehat{\bm{C}}$ in \textup{(3.5)}. From this matrix, compute the top-$r$ eigenprojector $\widehat{\bm{P}}_r$ and trailing eigenvalue sum $\sum_{i>r}\widehat\lambda_i(\widehat{\bm{C}})$. Also compute $\widehat m_4 := N_2^{-1}\sum_n\|\widehat\mathcal{S}(\bm{Z}^{(n)})\|_2^4$.
Conditional on the first half, let $\bm{Z}$ be an independent residual transformed by the fitted rank map. The corresponding population moments are $\widetilde\bm{C}(\widehat{\bm{\theta}}^{d}) := \mathbb{E}[\widehat\mathcal{S}(\bm{Z})\,\widehat\mathcal{S}(\bm{Z})^\top]$ and $m_4 := \mathbb{E}[\|\widehat\mathcal{S}(\bm{Z})\|_2^4]$, both conditional on the first half. The range of the fitted transform is bounded, so the fixed polynomial $\widehat\mathcal{S}$ has a finite fourth moment for every noise law. For any $\delta\in(0,1)$, with probability at least $1-\delta$ over the second half,
\begin{equation}
  \operatorname{Tr}\!\bigl((\bm{I}-\widehat{\bm{P}}_r)\,\widetilde\bm{C}(\widehat{\bm{\theta}}^{d})\bigr)
  \;\le\;
  \textstyle\sum_{i>r}\widehat\lambda_i(\widehat{\bm{C}}) \;+\; \sqrt{d-r}\,\sqrt{\frac{m_4}{N_2\,\delta}}.
  \label{eq:aposteriori}
\end{equation}
Replacing the unknown $m_4$ by its conditionally unbiased estimate $\widehat m_4$ changes the probability statement: \eqref{eq:aposteriori} holds with probability at least $1-\delta-o(1)$ over both halves as $N_2\to\infty$, provided $m_8/m_4^2=O_p(1)$ on $m_4>0$, where $m_8:=\mathbb{E}[\|\widehat\mathcal{S}(\bm{Z})\|_2^8]$. On $m_4=0$, the score is almost surely zero, as are both sides of \eqref{eq:aposteriori}.

<!-- proposition-5: body_md -->
\label{prop:hier-trunc}
Let $\Lambda \subset \Lambda'$ be Stage-1 multi-index sets, so $\mathcal{F}_\Lambda \subseteq \mathcal{F}_{\Lambda'}$. Write $\mathcal{S}_\Lambda, \mathcal{S}_{\Lambda'}$ for the $L^2(\pi_{\bm{Z}})$ projections of $\mathcal{S}$ onto these spaces, and set $\eta := \|\mathcal{S}_{\Lambda'} - \mathcal{S}_\Lambda\|_{L^2(\pi_{\bm{Z}})}$. Orthogonality gives $\|\mathcal{S} - \mathcal{S}_\Lambda\|^2_{L^2(\pi_{\bm{Z}})} = \|\mathcal{S} - \mathcal{S}_{\Lambda'}\|^2_{L^2(\pi_{\bm{Z}})} + \eta^2$. Under the saturation assumption \textup{(SAT)}: $\|\mathcal{S} - \mathcal{S}_{\Lambda'}\|_{L^2(\pi_{\bm{Z}})} \le \beta\,\|\mathcal{S} - \mathcal{S}_\Lambda\|_{L^2(\pi_{\bm{Z}})}$ for some $\beta \in [0,1)$,
\begin{equation}
  \eta^2 \;\le\; \|\mathcal{S} - \mathcal{S}_\Lambda\|^2_{L^2(\pi_{\bm{Z}})} \;\le\; \frac{\eta^2}{1-\beta^2}.
  \label{eq:hier-trunc}
\end{equation}
In exact rank-Gaussianized coordinates, estimate the coefficient differences between the two sets on two disjoint subsamples, obtaining $\Delta\widehat{\bm{\theta}}^{d}_1$ and $\Delta\widehat{\bm{\theta}}^{d}_2$. On a third subsample, compute the Gram matrix $\widehat\bm{A}_{\Lambda'}$ for the gradients $\nabla H_\alpha$, $\alpha\in\Lambda'$. If all three subsample sizes are proportional to $N$ and both sets satisfy \textup{(A1)}--\textup{(A3)}, the cross-fitted estimate \cite{ChernozhukovEtAl2018} $\widehat\eta_\times^2:=(\Delta\widehat{\bm{\theta}}^{d}_1)^\top\widehat\bm{A}_{\Lambda'}\Delta\widehat{\bm{\theta}}^{d}_2$ satisfies $\widehat\eta_\times^2=\eta^2+O_p(N^{-1/2})+O(\tau_N)$.

<!-- proof-1: body_md -->
Apply the Gaussian-reference bound \textup{(2.5)} to the basis $\widehat{\bm{V}}_r^{\bm{C}}$:
\[
  2\,\mathcal{D}_{\mathrm{KL}}(\pi_{\bm{n}}\,\|\,\pi_{\bm{n}}(\,\cdot\,;\widehat{\bm{V}}_r^{\bm{C}}))
  \;\le\;
  \operatorname{Tr}\!\bigl((\bm{I} - \widehat{\bm{P}}_r)\bm{C}\bigr)
  \;=\;
  \|(\bm{I} - \widehat{\bm{P}}_r)\mathcal{S}\|_{L^2(\pi_{\bm{Z}})}^2,
\]
where the equality uses $\bm{C} = \mathbb{E}_{\pi_{\bm{Z}}}[\mathcal{S}\mathcal{S}^\top]$ and the idempotence of $\bm{I}-\widehat{\bm{P}}_r$. Division by two proves \eqref{eq:stage1-cert}. For any score field $\mathcal{S}'$ with second-moment matrix $\bm{C}' := \mathbb{E}_{\pi_{\bm{Z}}}[\mathcal{S}'\mathcal{S}'^\top]$, apply the triangle inequality in $L^2(\pi_{\bm{Z}};\mathbb{R}^d)$. The orthogonal projector $\bm{I}-\widehat{\bm{P}}_r$ does not increase norms, and $\|(\bm{I}-\widehat{\bm{P}}_r)\mathcal{S}'\|_{L^2(\pi_{\bm{Z}})}^2 = \operatorname{Tr}((\bm{I}-\widehat{\bm{P}}_r)\bm{C}')$. Hence $\|\mathcal{S} - \mathcal{S}'\|_{L^2} + \sqrt{\operatorname{Tr}((\bm{I}-\widehat{\bm{P}}_r)\bm{C}')}$ bounds $\|(\bm{I}-\widehat{\bm{P}}_r)\mathcal{S}\|_{L^2}$, proving \eqref{eq:finite-sample-kl} as in \cite[Thm.~3.3]{BaptistaBrennanMarzouk2025}. The choice $\mathcal{S}' = \widehat\mathcal{S}$ proves \eqref{eq:aposteriori-kl} of \Cref{cor:aposteriori}.

By Part~I's \csname Cref@corollary@name\endcsname~2.3, $\operatorname{Tr}((\bm{I}-\bm{P}_r^{\bm{C}})\bm{C}) = \sum_{i>r}\lambda_i(\bm{C})$. Subtracting this minimum from the trace proves the split \eqref{eq:stage1-split} and the nonnegativity $\Delta_{\rm subopt} \ge 0$. Apply the eigengap inequality of Part~I (\csname Cref@proposition@name\endcsname~4.1, which is \cite[Lem.~4.2]{VuLei2013}) with $\bm{C}$ in place of $\bm{C}_\Lambda$. It gives uniqueness of the maximizer when $\lambda_r(\bm{C}) > \lambda_{r+1}(\bm{C})$, as well as $\Delta_{\rm subopt} = O(\|\sin\Theta(\widehat{\bm{V}}_r^{\bm{C}},\bm{V}_r^{\bm{C}})\|_F^2)$. For the truncated score, Part~I's \csname Cref@proposition@name\endcsname~4.1 states that $\bm{P}_r^{\Lambda\star}$ minimizes $\operatorname{Tr}((\bm{I}-\bm{P})\bm{C}_\Lambda)$ over rank-$r$ orthogonal projectors. Its minimum is $\sum_{i>r}\lambda_i(\bm{C}_\Lambda)$, so
\begin{equation}
  \operatorname{Tr}\!\bigl((\bm{I}-\widehat{\bm{P}}_r)\bm{C}_\Lambda\bigr) \;=\; \textstyle\sum_{i>r}\lambda_i(\bm{C}_\Lambda) \;+\; \operatorname{Tr}\!\bigl((\bm{P}_r^{\Lambda\star}-\widehat{\bm{P}}_r)\bm{C}_\Lambda\bigr),
  \label{eq:stage1-decomp}
\end{equation}
where the second summand is nonnegative. Apply $\sqrt{a+b}\le\sqrt a+\sqrt b$ to \eqref{eq:stage1-decomp} and use \eqref{eq:finite-sample-kl} with $\mathcal{S}' = \mathcal{S}_\Lambda$ to obtain \eqref{eq:stage1-threeterm}. For \eqref{eq:stage1-estrate}, Weyl's inequality \cite[Cor.~III.2.6]{Bhatia1997}, the Ky Fan trace extremum \cite[Prob.~III.6.11]{Bhatia1997}, and a rank-$r$ trace bound give $\operatorname{Tr}((\bm{P}_r^{\Lambda\star}-\widehat{\bm{P}}_r)\bm{C}_\Lambda) \le 2r\|\widehat{\bm{C}}-\bm{C}_\Lambda\|_{\rm op}$. Equation \eqref{eq:opnorm-rate} of \Cref{lem:score-cov-rate} bounds this quantity by $O_p(N^{-1/2}) + O(\tau_N)$ without an eigengap assumption. Assumption \textup{(A4)} enters only the $\sin\Theta$ rate \eqref{eq:sintheta-rate}, which is tested in Supplement~\ref{sm:rate-real}.

Conditional on the first half, the rank transform and $\widehat{\bm{\theta}}^{d}$ are fixed. The scores $\widehat\mathcal{S}(\bm{Z}^{(n)})$, $n = 1, \dots, N_2$, from the second half are i.i.d.\ and satisfy $\mathbb{E}[\widehat\mathcal{S}\widehat\mathcal{S}^\top] = \widetilde\bm{C}(\widehat{\bm{\theta}}^{d})$. Since $\sum_{i>r}\widehat\lambda_i(\widehat{\bm{C}}) = \operatorname{Tr}((\bm{I}-\widehat{\bm{P}}_r)\widehat{\bm{C}})$,
\[
  \operatorname{Tr}\!\bigl((\bm{I}-\widehat{\bm{P}}_r)\widetilde\bm{C}(\widehat{\bm{\theta}}^{d})\bigr) \;=\; \textstyle\sum_{i>r}\widehat\lambda_i(\widehat{\bm{C}}) + \operatorname{Tr}\!\bigl((\bm{I}-\widehat{\bm{P}}_r)(\widetilde\bm{C}(\widehat{\bm{\theta}}^{d}) - \widehat{\bm{C}})\bigr).
\]
Because the orthogonal projector $\bm{I}-\widehat{\bm{P}}_r$ has rank $d-r$, von Neumann's trace inequality \cite{Mirsky1975} bounds the second term by the sum of the $d-r$ largest singular values of $\widetilde\bm{C}(\widehat{\bm{\theta}}^{d})-\widehat{\bm{C}}$. Cauchy--Schwarz bounds that sum by $\sqrt{d-r}\,\|\widetilde\bm{C}(\widehat{\bm{\theta}}^{d})-\widehat{\bm{C}}\|_F$. These inequalities apply to each realized $\widehat{\bm{P}}_r$. The centered summands $\widehat\mathcal{S}(\bm{Z}^{(n)})\widehat\mathcal{S}(\bm{Z}^{(n)})^\top - \widetilde\bm{C}(\widehat{\bm{\theta}}^{d})$ are i.i.d.\ and mean-zero, so
\[
  \mathbb{E}[\|\widehat{\bm{C}} - \widetilde\bm{C}(\widehat{\bm{\theta}}^{d})\|_F^2] \;=\; N_2^{-1}\bigl(\mathbb{E}[\|\widehat\mathcal{S}(\bm{Z})\|_2^4] - \|\widetilde\bm{C}(\widehat{\bm{\theta}}^{d})\|_F^2\bigr) \;\le\; m_4/N_2,
\]
and Chebyshev's inequality applied to $\|\widehat{\bm{C}}-\widetilde\bm{C}(\widehat{\bm{\theta}}^{d})\|_F^2$ gives $\|\widehat{\bm{C}}-\widetilde\bm{C}(\widehat{\bm{\theta}}^{d})\|_F \le \sqrt{m_4/(N_2\delta)}$ with probability at least $1-\delta$. This proves \eqref{eq:aposteriori}. The moment $m_4$ is finite for every noise law, because the first-half rank transform has bounded range and the fixed polynomial $\widehat\mathcal{S}$ is bounded on it.

To replace the moment by $\widehat m_4$, condition again on the first half and first suppose $m_4>0$. Since $\mathrm{Var}(\widehat m_4/m_4)=(m_8/m_4^2-1)/N_2$, the moment-ratio condition implies $\widehat m_4/m_4=1+O_p(N_2^{-1/2})$ over both halves. Let $\epsilon\in(0,1)$. On the event $\widehat m_4\ge(1-\epsilon)m_4$, failure of \eqref{eq:aposteriori} with $\widehat m_4$ implies $\|\widehat{\bm{C}}-\widetilde\bm{C}(\widehat{\bm{\theta}}^{d})\|_F>\sqrt{(1-\epsilon)m_4/(N_2\delta)}$. The preceding bound puts this conditional probability at most $\delta/(1-\epsilon)$. With $\epsilon=N_2^{-1/4}$, Chebyshev's inequality bounds the conditional probability of the complementary event by $(m_8/m_4^2-1)N_2^{-1/2}$. Adding the two probabilities and averaging over the first half bounds the failure probability by $\delta+o(1)$. If $m_4=0$, the second-half scores vanish almost surely. Then $\widehat{\bm{C}}$, $\widetilde\bm{C}(\widehat{\bm{\theta}}^{d})$ and $\widehat m_4$ vanish, and both sides of \eqref{eq:aposteriori} are zero.

<!-- proof-4: body_md -->
The exact rank-transform change of variables and the estimator definition in \Cref{sec:estimator} give $\pi_{\bm{n}}(\bm{x}) = c^{Z}(\mathcal{Z}(\bm{x}))\prod_i\pi_i(x_i)$ and $\widehat Z_{\mathrm{comp}}\,\widehat\pi_{\bm{n}}(\bm{x}) = \widehat c^{Z}(\widehat\mathcal{Z}(\bm{x}))\prod_i\widehat\pi_i(x_i)$, with $0 < \widehat Z_{\mathrm{comp}} < \infty$ (Part~I \cite{CAS_partI}). To justify integration, check each logarithm under $\pi_{\bm{n}}$. The function $|\log c^{Z}| = |\mathcal{L}|$ grows polynomially by \textup{(T2)} (Part~I \cite{CAS_partI}), while the Gaussian marginals of $\pi_{\bm{Z}}$ supply all polynomial moments. The fitted $\log\widehat c^{Z}$ is a degree-$K_2$ Stage-2 polynomial in $\widehat{\bm{V}}_r^{\bm{C}\top}\bm{z}$, minus the log-normalizer. It is bounded above by the Stage-2 constraint (Part~I \cite{CAS_partI}) and grows at most polynomially below. For the marginal logarithms, $\mathbb{E}_{\pi_{\bm{n}}}[|\log(\pi_i/\widehat\pi_i)|] < \infty$ follows from \textup{(M)} (\Cref{sec:estimator}).

\emph{The exact identity.} Pointwise, $\log(\pi_{\bm{n}}/\widehat\pi_{\bm{n}}) = [\log c^{Z}(\mathcal{Z}) - \log\widehat c^{Z}(\widehat\mathcal{Z})] + \sum_i\log(\pi_i/\widehat\pi_i) + \log\widehat Z_{\mathrm{comp}}$. Since $x_i \sim \pi_i$ under $\pi_{\bm{n}}$, taking expectations gives the marginal contribution $\sum_i\mathcal{D}_{\mathrm{KL}}(\pi_i\,\|\,\widehat\pi_i)$, and the constant term remains $\log\widehat Z_{\mathrm{comp}}$. Split the copula contribution by adding and subtracting $\mathbb{E}_{\pi_{\bm{n}}}[\log\widehat c^{Z}(\mathcal{Z})]$. The transform $\mathcal{Z}$ pushes $\pi_{\bm{n}}$ forward to $\pi_{\bm{Z}}$, so the first difference equals $\mathbb{E}_{\pi_{\bm{Z}}}[\log(\pi_{\bm{Z}}/\gamma_d) - \log(\widehat\pi_{\bm{Z}}/\gamma_d)] = \mathcal{D}_{\mathrm{KL}}(\pi_{\bm{Z}}\,\|\,\widehat\pi_{\bm{Z}})$. Here $\widehat\pi_{\bm{Z}} := \widehat c^{Z}\gamma_d$ is normalized by $\widehat Z_r$ (\Cref{sec:apost-density}). The second difference is $R_N$. Conditional on the estimation sample, the factorization $\widehat\pi_{\bm{Z}}(\bm{z}) = \widehat c_r^{U}(\bm{u})\gamma_r(\bm{u})\,\gamma_{d-r}(\bm{w})$ holds in the coordinates $(\bm{u},\bm{w}) = (\widehat{\bm{V}}_r^{\bm{C}\top}\bm{z},\ \widehat{\bm{V}}_\perp^\top\bm{z})$. The KL chain rule in these coordinates gives
\[
  \mathcal{D}_{\mathrm{KL}}(\pi_{\bm{Z}}\,\|\,\widehat\pi_{\bm{Z}}) \;=\; \mathcal{D}_{\mathrm{KL}}(\pi_{\bm{U}}\,\|\,\widehat\pi_{\bm{U}}) + \mathbb{E}_{\pi_{\bm{U}}}[\mathcal{D}_{\mathrm{KL}}(\pi_{\bm{W}\mid\bm{U}}\,\|\,\gamma_{d-r})],
\]
where the second term is $\mathcal{D}_{\mathrm{KL}}(\pi_{\bm{n}}\,\|\,\pi_{\bm{n}}(\,\cdot\,;\widehat{\bm{V}}_r^{\bm{C}}))$ by Part~I's reduced-form identity \cite{CAS_partI}. Because the identity holds for any fixed subspace, it holds at each realized $\widehat{\bm{V}}_r^{\bm{C}}$. Every term is finite by the integrability checks above.

\emph{The conditional rate of $R_N$.} For the fixed finite dictionary, the polynomial mean-value formula and $\|\widehat{\bm{V}}_r^{\bm{C}\top}\|_{\rm op}=1$ give a deterministic constant $C_\Lambda$ such that
\[
|\widehat\mathcal{L}_r(\widehat{\bm{V}}_r^{\bm{C}\top}\bm{z})-\widehat\mathcal{L}_r(\widehat{\bm{V}}_r^{\bm{C}\top}\bm{z}')|
\le C_\Lambda\|\widehat{\bm{\theta}}^{r}\|\,(1+\|\bm{z}\|+\|\bm{z}'\|)^{K_2-1}\|\bm{z}-\bm{z}'\|.
\]
With $W_N$ denoting the inner expectation in \eqref{eq:transform-moment}, Cauchy--Schwarz over the estimation sample and \textup{(T)} yield the following bound. No independence between $\widehat{\bm{\theta}}^{r}$ and $\widehat\mathcal{Z}$ is needed:
\[
\begin{aligned}
\mathbb{E}|R_N|&\le C_\Lambda\mathbb{E}[\|\widehat{\bm{\theta}}^{r}\|W_N^{1/2}]
\le C_\Lambda(\mathbb{E}\|\widehat{\bm{\theta}}^{r}\|^2)^{1/2}(\mathbb{E} W_N)^{1/2}\\
&=O(\sqrt d\,(\log N)^{K_2/2}N^{-1/2}).
\end{aligned}
\]
Combining this bound with \textup{(M)} proves \eqref{eq:Delta-marg}, and Markov's inequality gives the $O_p$ statement. Beyond the sample range, the estimated distribution functions of Part~I take the values $1/[2(N+1)]$ and $1-1/[2(N+1)]$. Thus $\|\widehat\mathcal{Z}\|_\infty\le z_N:=\Phi^{-1}(1-1/[2(N+1)])$ and $\|\widehat{\bm{V}}_r^{\bm{C}\top}\widehat\mathcal{Z}\|\le\sqrt d\,z_N$, with $z_N^2\le2\log(N+1)$.

\emph{Expectation and assembly.} Take expectations over the estimation sample in the exact identity to obtain $\mathbb{E}[\mathcal{D}_{\mathrm{KL}}(\pi_{\bm{n}}\,\|\,\widehat\pi_{\bm{n}})] = \mathbb{E}[\mathcal E_{\rm marg}] + \mathbb{E}[\mathcal E_{\rm proj}] + \mathbb{E}[\mathcal E_{\rm dens}]$. The expectation of this conditional bound, together with the marginal and projection bounds, proves \eqref{eq:composite-kl}.

<!-- theorem-3: body_md -->
\label{thm:ccsm-rate}
Fix a finite dictionary $\Lambda_r$ and an exact-transform basis. Assume
\textup{(H1)--(H2)}, independent observations from $\pi_{\bm{U}}$,
$\bm{A}^\star\succ0$, and the population score-matching identity stated above.
Let $\bm\theta^{r,\star}$ solve the population normal equations.
Fix $C_{\rm samp},C_{\rm curv}>0$ and use the regularizer of
\textup{(3.7)}. Let $\mathcal C_N$ be a measurable, nonempty,
closed convex constraint set, possibly depending on the same observations,
and let $\widehat{\bm{\theta}}^{r}$ minimize the regularized quadratic over $\mathcal C_N$.
Assume $\widehat\bm{M}\succ0$ almost surely and
$\sup_{N\ge N_0}\mathbb{E}\|\widehat{\bm{\theta}}^{r}\|^4<\infty$ for some fixed $N_0$.
The set $\mathcal C_N$ includes all imposed constraints, including centering.

Write $a:=\lambda_{\min}(\bm{A}^\star)$,
$L:=\|\bm{A}^\star\|_{\rm op}$,
$t_k:=\operatorname{Tr}(\bm{A}^{\star,(k,k)})$, and let $\bm{D}_N$ be diagonal, with entry
\[
 \rho_k:=C_{\rm samp}t_k/N+C_{\rm curv}Lk^2
 \quad\hbox{on the degree-$k$ class.}
\]
Let $\bm\Sigma_N:=\mathrm{Cov}(\mathbf r_N)$ and define
\[
 \mathcal G:=\{\tfrac12\bm{A}^\star\preceq\widehat\bm{A}_r
                                     \preceq2\bm{A}^\star\},
 \qquad
 d_N:=\inf_{\beta\in\mathcal C_N}
           \|\beta-\bm\theta^{r,\star}\|_{\widehat\bm{M}}.
\]
Then, for $N\ge N_0$,
\begin{equation}
\begin{aligned}
 \mathbb{E}[\mathcal J(\widehat{\bm{\theta}}^{r})]-\mathcal J_{\rm trunc}
 &\le\mathcal E_{\rm est}+16\mathbb{E}[d_N^2\mathbf1_{\mathcal G}]+O(N^{-1}),\\
 \mathcal E_{\rm est}&:=16\operatorname{Tr}(\bm{D}_N^{-1}\bm\Sigma_N)
 +\frac{64}{a}\|\bm{D}_N\bm\theta^{r,\star}\|^2.
\end{aligned}
\label{eq:ccsm-rate}
\end{equation}
The constant in $O(N^{-1})$ depends on the dictionary, on $\pi_{\bm{U}}$ and on the fourth-moment bound.

In particular, if $t_k\ge C_-kp_k$ and
$(\bm\Sigma_N)_{\alpha\alpha}\le v_0 k^2/N$
for $\alpha\in\Lambda^{(k)}$, then
\begin{equation}
 16\operatorname{Tr}(\bm{D}_N^{-1}\bm\Sigma_N)
 \le 16v_0\min\left\{
       \frac{\sum_{k:p_k>0}k}{C_{\rm samp}C_-},
       \frac{|\Lambda_r|}{NC_{\rm curv}L}\right\}.
 \label{eq:V-piecewise}
\end{equation}
Here $C_->0$ exists for the fixed dictionary when $\bm{A}^\star\succ0$,
and one may take
$v_0=C_\star(2+2K_b^2K_{\max}\|\bm\theta^{r,\star}\|^2)$
from the per-mode noise calculation.
Writing $\bm\theta^{r,\star,(k)}$ for the degree-$k$ coefficient block, the other term satisfies
\begin{equation}
 \frac{64}{a}\|\bm{D}_N\bm\theta^{r,\star}\|^2
 \le\frac{128}{a}\left[
  \frac{C_{\rm samp}^2}{N^2}
       \sum_k t_k^2\|\bm\theta^{r,\star,(k)}\|^2
 + C_{\rm curv}^2 L^2
       \sum_k k^4\|\bm\theta^{r,\star,(k)}\|^2\right].
 \label{eq:B-bound}
\end{equation}
The two bounds in \eqref{eq:V-piecewise} give the sampling scales $1/C_{\rm samp}$ and $1/(NC_{\rm curv})$ associated with the two parts of the regularizer.

<!-- lemma-3: body_md -->
\label{lem:lsi-pd-leading}
Let $V:\mathbb{R}^r\to\mathbb{R}$ be a polynomial of even degree $K \ge 4$ whose degree-$K$ homogeneous part $G$ is positive-definite, and let $Z := \int e^{-V} < \infty$. Then $\mu := Z^{-1}e^{-V}\,d\bm{u}$ satisfies a tight logarithmic Sobolev inequality.
<!-- end lemma-3 -->
<!-- proof-11: body_md -->
Positive definiteness of $G$ gives $g_0 := \min_{\|\bm{v}\|=1}G(\bm{v})>0$ and hence $G(\bm{u}) \ge g_0\|\bm{u}\|^K$. The lower-degree terms are dominated at large $\|\bm{u}\|$, so $V(\bm{u}) \ge \tfrac{g_0}{2}\|\bm{u}\|^{K}$. Thus condition (3.14.1) of \cite{CattiauxGuillinWangWu2009} holds, and $Z < \infty$. Euler's identity $\bm{u}\cdot\nabla G = K\,G$ implies that $g_1 := \min_{\|\bm{v}\|=1}|\nabla G(\bm{v})| > 0$, and homogeneity then yields $|\nabla G(\bm{u})| \ge g_1\|\bm{u}\|^{K-1}$. Since $\nabla V = \nabla G + O(\|\bm{u}\|^{K-2})$ and $\Delta V = O(\|\bm{u}\|^{K-2})$, it follows that $|\nabla V|^2 \ge \tfrac{g_1^2}{2}\|\bm{u}\|^{2(K-1)}$ and $|\Delta V| \le c_3\|\bm{u}\|^{K-2}$ for large $\|\bm{u}\|$. Set $G_{\max} := \max_{\|\bm{v}\|=1}G(\bm{v})$, so that $V(\bm{u}) \le 2G_{\max}\|\bm{u}\|^{K}$ for large $\|\bm{u}\|$. Because $K \ge 4$, the exponents satisfy $2(K-1) > K$. For each fixed $a_0 \in (0,1)$, choose $R$ large enough that, whenever $\|\bm{u}\|$ exceeds this radius,
\[
(1-a_0)|\nabla V|^2 - \Delta V \ge \tfrac{(1-a_0)g_1^2}{4}\|\bm{u}\|^{2(K-1)} \ge 2G_{\max}\|\bm{u}\|^{K} \ge V(\bm{u}),
\]
where the middle inequality follows from the comparison of exponents above. This is condition (3.14.2) with $\eta(u) = u$, after choosing their constant $b_0$ to cover the finite deficit on $\{\|\bm{u}\| \le R\}$. Condition (3.14.3) follows from $V/|\nabla V|^2 = O(\|\bm{u}\|^{2-K}) \to 0$. Both $|\nabla V|$ and $\nabla^2 V$ grow at most polynomially in $\bm{u}$ and are therefore bounded by $e^{K'V}$ outside a ball, as required by the growth condition in their Thm.~3.17. That theorem and Rem.~3.18 of \cite{CattiauxGuillinWangWu2009} give a defective logarithmic Sobolev inequality. Conditions (3.14.1) and (3.14.2) also give a Poincar\'e inequality \cite{BakryBartheCattiauxGuillin2008}. Rothaus's lemma \cite{Rothaus1985} then gives a tight logarithmic Sobolev inequality.

<!-- remark-1: body_md -->
\label{rem:gaussian-copula}
Suppose $\pi_{\bm{Z}} = \gamma(0,\bm\Sigma^{\bm{Z}})$, where $\bm\Sigma^{\bm{Z}} := \mathbb{E}_{\pi_{\bm{Z}}}[\bm{Z}\otimes\bm{Z}]$ is a correlation matrix. Its log-copula relative to $\gamma_d$ is the quadratic $g^\star(\bm{z}) = -\tfrac12\bm{z}^\top \bm{M}\bm{z} + \mathrm{const}$, with $\bm{M} := (\bm\Sigma^{\bm{Z}})^{-1} - \bm{I}$. This is a combination of degree-$2$ Hermite polynomials and a constant, since $z_iz_j = H_{e_i+e_j}$ ($i\ne j$) and $z_i^2 = \sqrt2 H_{2e_i} + 1$.

For any $\Lambda = \Lambda_{K,q}$ with $K,q\ge2$, the space $\mathcal{F}_\Lambda$ contains every symmetric linear vector field $\bm{z}\mapsto\bm S\bm{z}$ with $\bm S=\bm S^\top$. It therefore contains $\mathcal{S}(\bm{z})=-\bm{M}\bm{z}$. Hence $\mathcal{S}_\Lambda=\mathcal{S}$, the coefficient vector $\theta^\star$ is supported on degree $2$, and $\bm{C}=\bm{M}\,\bm\Sigma^{\bm{Z}}\bm{M}$. Fisher consistency \cite[Thm.~2, Cor.~3]{Hyvarinen2005} identifies $\theta^\star$ as the unique infinite-sample minimizer of the strictly convex quadratic objective.

By the closed form of the solve and the strong law of large numbers, $\widehat\theta \to \theta^\star$ a.s.\ and $\widehat{\bm{C}} \to \bm{C}$ in operator norm as $\tau_N \to 0$. At the fixed strength $\tau_N \asymp \kappa\|\widehat{\bm{A}}\|_{\rm op}$ used here, the bias term of \Cref{lem:score-cov-rate} is $O(\tau_N)$ and does not vanish, but it does not visibly change the measured rate of \Cref{fig:rate-sm} over the sample sizes used.
