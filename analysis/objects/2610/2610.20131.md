 Let $\mathfrak{M}_{{\mathcal P}}$ be a solenoidal manifold with base $M_0$ defined by a group chain ${\mathcal G}$ in $\Gamma = \pi_1(M_0, x_0)$. Then there is a well-defined characteristic map
\begin{equation}\label{eq-chi}
\chi \colon {\rm Mod}(\mathfrak{M}_{{\mathcal P}} ; \widehat{x})  \to {\rm Comm}(\Gamma) \ .
\end{equation}

<!-- thm-9: body_md -->
\cite[Corollary~5.2]{RT1972b}\label{thm-RT}
     Let $M_0$ be a closed connected manifold, and  $\mathfrak{M}_{{\mathcal P}}$ be a solenoidal manifold over $M_0$ defined by a presentation ${\mathcal P}$.  Given a self-homeomorphism  $h \colon \mathfrak{M}_{{\mathcal P}} \to \mathfrak{M}_{{\mathcal P}}$ and ${\varepsilon} > 0$,  then  $h$ is ${\varepsilon}$-homotopic to a continuous map   induced by a \emph{shuffle map} of ${\mathcal P}$. That is, there exists    an increasing sequence   $i_{\ell} \geq \ell$, and continuous maps $\varphi_{i_\ell} \colon M_{i_{\ell}} \to M_{\ell}$  so that the ``zig-zag'' diagram \eqref{eq-shuffle} commutes
 \begin{align} \label{eq-shuffle}
 \xymatrix{ M_{0}   & \ar[l] M_{i_0} \ar[ld]^{\varphi_{i_0}}& \ar[l] M_{i_1} \ar[ld]^{\varphi_{i_1}} & \ar[l] M_{i_2} \ar[ld]^{\varphi_{i_2}}  & \ar[l] \cdots \\
M_0 & \ar[l] M_1  & \ar[l] M_2 & \ar[l] M_3   & \ar[l]\cdots } 
\end{align}
and the self-map   ${\widehat{\varphi}} \colon \mathfrak{M}_{{\mathcal P}} \to \mathfrak{M}_{{\mathcal P}}$ induced by the collection $\{\varphi_{i_{\ell}} \mid \ell \geq 0\}$ is ${\varepsilon}$-homotopic to $h$. Moreover, for each $\ell \geq 0$, the map $\varphi_{i_{\ell}}$ induces a monomorphism between the fundamental groups of $M_{i_{\ell}}$ and $M_{\ell}$ whose image has finite index.
<!-- end thm-9 -->
<!-- head-2: body_md -->
For each $\ell \geq 0$, there is the proper covering map $p_0^{\ell} = p_{\ell} \circ \cdots \circ p_1 \colon M_{\ell} \to M_0$, and the projection map 
  $\widehat{q}_{\ell} \colon \mathfrak{M}_{{\mathcal P}} \to M_{\ell}$  onto the factor $M_{\ell}$ in \eqref{eq-invlimit}.  
  Let $\widehat{x} \in \mathfrak{M}_{{\mathcal P}}$ be a choice of basepoint, and  set $x_{\ell} = \widehat{q}_{\ell}(\widehat{x}) \in M_{\ell}$  and set $\Gamma_0 = \pi_1(M_0, x_0)$. For each $\ell > 0$,  $\Gamma_{\ell}$ is the image of $(p_0^{\ell})_{\#} \colon \pi_1(M_{\ell}, x_{\ell}) \to \Gamma_0$, so ${\mathcal G} = \{\Gamma_{\ell} \mid \ell \geq 0 \}$ is a group chain in $\Gamma_0$.
  
  Let $d_{M_0}$ denote a Riemannian metric on $M_0$ and let $\delta_0 > 0$ be sufficiently small so  that, for all $y \in M_0$, the closed metric ball $B_{M_0}(y, \delta_0) \subset M_0$ of radius $\delta_0$ is convex. In particular, $\delta_0$ is less than the injectivity radius of $M_0$. It follows that for each $y \in M_0$ and $y' \in  B_{M_0}(y, \delta_0)$ there is an isomorphism $\iota_{y',y} \colon \pi_1(M_0, y') \cong \pi_1(M_0, y)$ induced by concatenation with a geodesic path  in $B_{M_0}(y, \delta_0)$  from $y'$ to $y$.
  
  
  
   Finally, since the projection $\widehat{q}_0 \colon \mathfrak{M}_{{\mathcal P}} \to M_0$  is uniformly continuous, we can choose  ${\varepsilon} >0$   sufficiently small so that  $\widehat{q}_0(B_{\mathfrak{M}}(\widehat{y}, {\varepsilon})) \subset B_{M_0}(y, \delta_0/3)$ for all $\widehat{y} \in \widehat{q}_0^{-1}(y)$, where $B_{\mathfrak{M}}(\widehat{y},{\varepsilon})$ is a ball around $\widehat{y} \in \mathfrak{M}_{\mathcal P}$ with respect to the metric $d_\mathfrak{M}$.
  
   
  Given $h \in {\rm Homeo}(\mathfrak{M}_{{\mathcal P}}; \widehat{x})$, we first define $\chi(h) \in {\rm Comm}(\Gamma)$. We then   show that $\chi(h)$ is independent of choices, and depends only on the basepoint fixed homotopy class of $h$.
   
Let $h \colon \mathfrak{M}_{{\mathcal P}} \to \mathfrak{M}_{{\mathcal P}}$, then by Theorem~\ref{thm-RT} there exists an increasing  sequence $i_{\ell} \geq \ell$, $\ell \geq 0$,  and maps $\varphi_{i_\ell} \colon M_{i_{\ell}} \to M_\ell$  so that the shuffle diagram \eqref{eq-shuffle} commutes, and for  
 the map  ${\widehat{\varphi}} \colon \mathfrak{M}_{{\mathcal P}} \to \mathfrak{M}_{{\mathcal P}}$ defined by the maps $\{\varphi_{i_{\ell}} \mid \ell \geq 0\}$, we have that $h$ and ${\widehat{\varphi}}$ are ${\varepsilon}$-homotopic.  Let $\widehat{y} = {\widehat{\varphi}}(\widehat{x})$ and set $y_{\ell} = \widehat{q}_{\ell}(\widehat{y}) \in M_{\ell}$.
Then we have the diagram:
  \begin{align} \label{eq-defchi}
 \xymatrix{  \pi_1(  M_{i_0} , x_{i_0}) \ar[d]_{(p_0^{i_0})_{\#}} \ar[r]^{({\varphi}_{i_0})_{\#}} &   \pi_1(M_{0} ,y_0)  \ar[r]_{\cong}^{\iota_{y',y}}  &  \pi_1(M_0 , x_0)   \\
\pi_1(M_0 , x_0) &  &   } 
\end{align}
where $(p_0^{i_0})_{\#} \colon  \pi_1(  M_{i_0} , x_{i_0})  \to \Gamma_{i_0} \subset \Gamma_0$ is an isomorphism onto its image $\Gamma_{i_0}$.
Let $h_{\#} \colon \Gamma_{i_0} \to \Gamma_0$ be  the commensurator defined by the inverse of $(p_0^{i_0})_{\#}$ followed by   $\iota_{y',y} \circ ({\varphi}_{i_0})_{\#}$.  Define  $\chi(h) \in {\rm Comm}(\Gamma_0)$ to be the commensurator class of $h_{\#}$.
 
Next, we show that $\chi(h)$ is independent of the choice of an approximation to $h$ by an induced map. 
Suppose there are given two sequences $\{\varphi_{i_\ell} \colon M_{i_{\ell}} \to M_{\ell} \mid \ell \geq 0\}$ and $\{\psi_{j_\ell} \colon M_{j_{\ell}} \to M_{\ell} \mid \ell \geq 0\}$,   so that the shuffle diagram \eqref{eq-shuffle} commutes for each.
  Moreover,   assume that   the induced maps   ${\widehat{\varphi}} \colon \mathfrak{M}_{{\mathcal P}} \to \mathfrak{M}_{{\mathcal P}}$ and  ${\widehat{\psi}} \colon \mathfrak{M}_{{\mathcal P}} \to \mathfrak{M}_{{\mathcal P}}$   are both ${\varepsilon}$-homotopic to $h$. Thus ${\widehat{\varphi}}$ and ${\widehat{\psi}}$ are $2{\varepsilon}$-homotopic.

    Let $\widehat{y} = {\widehat{\varphi}}(\widehat{x})$ then set  $y_{k} = \widehat{q}_{k}(\widehat{y}) \in M_{k}$ for   $k \geq 0$. Likewise, let
      $\widehat{z} = {\widehat{\psi}}(\widehat{x})$ and set  $z_{k} = \widehat{q}_{k}(\widehat{z}) \in M_{k}$ for   $k \geq 0$.
      Choose an index $k_0 > \max\{i_0, j_0\}$. 
  Then we have two maps:
 \begin{eqnarray*}
\Psi_{{\widehat{\varphi}}} \colon & \mathfrak{M}_{{\mathcal P}} \stackrel{\widehat{q}_{k_0}}{\longrightarrow} M_{k_0} \stackrel{p^{k_0}_{i_0}}{\longrightarrow} M_{i_{0}} \stackrel{{\varphi}_{i_0}}{\longrightarrow}   M_0\\
\Psi_{{\widehat{\psi}}}  \colon & \mathfrak{M}_{{\mathcal P}} \stackrel{\widehat{q}_{k_0}}{\longrightarrow} M_{k_0} \stackrel{p^{k_0}_{j_0}}{\longrightarrow} M_{j_{0}} \stackrel{\psi_{j_0}}{\longrightarrow}   M_0   ~ .
\end{eqnarray*}
Now, given $\alpha \in \pi_1(M_{k_0}, x_{k_0})$ choose a  closed curve $\gamma$ in $M_{k_0}$ with basepoint $x_{k_0}$ representing $\alpha$.

Let $\widetilde{\gamma}$ be the lift of $\gamma$ to a curve in   the leaf $L_{\widehat{x}}$ in $\mathfrak{M}_{{\mathcal P}}$ starting at $\widehat{x}$ and which covers $\gamma$. 

 

The images   $\Psi_{{\widehat{\varphi}}}(\widetilde{\gamma})$ and $\Psi_{{\widehat{\psi}}}(\widetilde{\gamma})$ in   $M_0$  of the curve  $\widetilde{\gamma}$ in   $\mathfrak{M}_{{\mathcal P}}$ then lie within $2\delta_0/3$ of each other. By our choice of $\delta_0$,  the  images of the  curves   lie in a convex neighborhood of each other, and hence are homotopic. Thus, they determine the same equivalence class in $\pi_1(M_0, x_0)$. This implies that the definition of $\chi(h)$ on elements of $\Gamma_{k_0}$ agrees using either approximation ${\widehat{\varphi}}$ or ${\widehat{\psi}}$, and $\chi(h)$ does not depend on the choice of an approximating homotopy.

Finally, let $\{h_t \colon \mathfrak{M}_{{\mathcal P}} \to \mathfrak{M}_{{\mathcal P}} \mid 0 \leq t \leq 1\} \subset {\rm Homeo}(\mathfrak{M}_{{\mathcal P}};  \widehat{x})$ be an isotopy. Let ${\varepsilon} > 0$ be as above, then there exists $\lambda > 0$ such that   $| s-t| < \lambda$ implies $d_{\mathfrak{M}}[h_s , h_t] < {\varepsilon}$, where $d_{\mathfrak{M}}[\cdot , \cdot]$ denotes the sup-norm on homeomorphisms induced by $d_{\mathfrak{M}}$.  Then for $n \gg 0$ so that $1/n < \lambda$ we form the sequence of isotopic homeomorphisms $\{h_{i/n} \mid 0 \leq i \leq n\}$ with $d_{\mathfrak{M}}[h_{i/n} , h_{(i+1)/n}] < {\varepsilon}$.
 As above, for each $i$ choose an induced tower map ${\widehat{\varphi}}_i$ which approximates $h_{i/n}$.  Then ${\widehat{\varphi}}_i$ and ${\widehat{\varphi}}_{i+1}$ are $3{\varepsilon}$-homotopic, and so $\chi({\widehat{\varphi}}_i)$ and $\chi({\widehat{\varphi}}_{i+1})$ determine the same class in ${\rm Comm}(\Gamma)$. It follows that 
 $\chi({\widehat{\varphi}}_0)$ and $\chi({\widehat{\varphi}}_n)$ determine the same class in ${\rm Comm}(\Gamma)$.  This shows that $\chi(h)$ depends only on the isotopy class of $h$ hence $\chi$ descends to  a map on ${\rm Mod}(\mathfrak{M}_{{\mathcal P}}, \widehat{x})$. This completes the proof of Theorem~\ref{thm-chi}.
 \endproof

{\rm
 The conclusion of Theorem~\ref{thm-chi} can be reformulated in terms of pro-categories.
Theorem~\ref{thm-RT} implies that a homeomorphism $h \colon \mathfrak{M}_{{\mathcal P}} \to \mathfrak{M}_{{\mathcal P}}$ induces a map $\{ {\varphi}_{i_{\ell}} \}$ in the pro-category of spaces. \label{def-isomorphism}
Odometer actions $({\mathfrak{X}}_1, \Gamma_1, \Phi_1)$ and $({\mathfrak{X}}_2, \Gamma_2, \Phi_2)$ are said to be \emph{isomorphic}  if there is a homeomorphism $h \colon {\mathfrak{X}}_1 \to {\mathfrak{X}}_2$ and group isomorphism $\Theta \colon \Gamma_1 \to \Gamma_2$ such that 
\begin{equation}\label{eq-isomorphism}
\Phi_1(g) = h^{-1} \circ \Phi_2(\Theta(g)) \circ h   \in   {\rm Homeo}({\mathfrak{X}}_1) \   \textrm{for  all} \ g \in \Gamma_1 \ .
\end{equation}
The map  $h$ is said to be an \emph{isomorphism} when    ${\mathfrak{X}}_1 = {\mathfrak{X}}_2$ and $\Gamma_1 = \Gamma_2$. In this case,   $\Theta \in {\rm Aut}(\Gamma)$ can be viewed as a ``uniform time change'' for the action.

The isomorphism $h$ is an \emph{automorphism}   if ${\mathfrak{X}}_1 = {\mathfrak{X}}_2$, $\Gamma_1 = \Gamma_2$, and $\Theta \colon \Gamma \to \Gamma$ is the identity.

{\mathfrak{I}}({\mathfrak{X}}, \Gamma, \Phi) = \{ h \in {\rm Homeo}({\mathfrak{X}})  \mid h ~ {\rm is \ an \ isomorphism \ of \ the \ action}\} \ .

{\mathfrak{I}}_0({\mathfrak{X}}, \Gamma, \Phi) = \{ h  \in {\rm Homeo}({\mathfrak{X}})  \mid h ~ {\rm is \ an \ automorphism \ of \ the \ action}\} \ .

   Let  $({\mathfrak{X}}, \Gamma, \Phi)$   be an odometer action, and $x \in {\mathfrak{X}}$. 
   \begin{enumerate}
\item  A \emph{dynamical commensurator} at $x$  is a homeomorphism $h_U \colon U \to V$, where $U$ and $V$ are adapted subsets with $x \in U \cap V$, such that $h_U(x) = x$, and $h_U$ induces an isomorphism $\Theta_U \colon {\mathcal H}_U \to {\mathcal H}_V$.

{\rm Comm}({\mathfrak{X}}, \Gamma, \Phi, x) = \{  h_U \colon U \to V  \mid x \in U \cap V ,  \ h_U \  {\rm is \ a \ commensurator}  \} \slash  \mathrel{\stackrel{\makebox[0pt]{\mbox{\normalfont\tiny c}}}{\sim}}  $$
The equivalence class of $h_U \colon U \to V$ is denoted by $[h_U,U,V]$.
\end{enumerate}

\label{lem-coherentqa}
A topologically free  odometer  $({\mathfrak{X}}, \Gamma, \Phi)$ is coherent.

<!-- head-4: body_md -->
If $({\mathfrak{X}}, \Gamma, \Phi)$ is topologically free, then for every adapted set $U \subset {\mathfrak{X}}$, the restricted holonomy action map $\Phi_U \colon \Gamma_U \to Homeo(U)$ has trivial kernel. Conversely, assume that the action   is  coherent, and let $\gamma \in \Gamma$ be such that there exists a non-empty open set $V \subset {\rm Fix}(\gamma)$. By assumption, the kernel of $\Phi_U$ is a torsion subgroup of $\Gamma_U$ hence a torsion subgroup of $\Gamma$. Thus, $\gamma$ is the identity, and so ${\rm Iso}(\Phi)$ is a meager set. 
\endproof

\label{ex-product-lqa-coherent}
{\rm The first example   is that of an action which is effective and coherent, but not topologically free.  It is   a special case of \cite[Example~7.6]{DHL2016a}.

Let $({\mathfrak{X}}_0, \Gamma_0, \Phi_0)$ be a free odometer. For example, this can be a standard odometer action of $\Gamma_0 = {\mathbb Z}$ on the profinite completion $\widehat{{\mathbb Z}}$.

The action of  $\Gamma_{U_h} = \Gamma_0 \times hKh^{-1}$ preserves $U_h$ and the kernel of the action map $\Phi_{U_h} \colon \Gamma_{U_h} \to {\rm Homeo}(U_h)$ is the finite group $\{e\} \times hKh^{-1}$. It follows that this action is effective and coherent.

<!-- prop-1: body_md -->
\label{prop-nilpotentcoherent}
 If $\Gamma$   is  nilpotent, then  an  effective odometer       $({\mathfrak{X}}, \Gamma, \Phi)$     is   topologically free.
<!-- end prop-1 -->
<!-- head-5: body_md -->
Let $U \subset {\mathfrak{X}}$ be an adapted set and let $h\in \Gamma_U$ such that $\Phi_U(h)$ is the identity on $U$. We claim that $\Phi(h)$ acts as the identity on $X_{{\mathcal G}}$.  We proceed by induction on the upper central series
$$\{e\} = \Gamma_{(0)} \subset \Gamma_{(1)} \subset \cdots \subset \Gamma_{(k)} = \Gamma$$
So $\Gamma_{(1)}$ is the center of $\Gamma$, and $\Gamma_{(i+1)} = \{g \in \Gamma \mid [g,h] \in \Gamma_{(i)} \ \forall h \in \Gamma\}$. 
 



Let $g \in \Gamma_{(1)}$, then for $y \in U$, we have 
$h \cdot g \cdot y = g \cdot h \cdot y = g \cdot y$. Thus $\Phi(h)$ acts as the identity on the union of the clopen sets $\{g \cdot U \mid g \in \Gamma_{(1)} \}$.

For $1 \leq i < k$, assume that $\Phi(h)$ acts as the identity on the union of the clopen sets $\{g\cdot U \mid g \in \Gamma_{(i)} \}$.
Let $g \in \Gamma_{(i+1)}$ then for $y \in  U$, we have $[g,h] = g^{-1}h^{-1}g h \in \Gamma_{(i)}$ and
$$h \cdot g \cdot y = h \cdot    g \cdot [g,h]  \cdot y = h \cdot g \cdot g^{-1}h^{-1}g h \cdot y =  g \cdot h \cdot y = g \cdot y.$$ 

 

Thus, $\Phi(h)$ acts as the identity on the union of the clopen sets $\{g\cdot U \mid g \in \Gamma_{(i+1)} \}$.
It follows by finite induction that $\Phi(h)$ acts as the identity on   the union of the clopen sets $\{g \cdot U \mid g \in \Gamma\}$,  which is all of ${\mathfrak{X}}$. Since the action is effective, $h = e$. Thus for every $g \in \Gamma$ the set ${\rm Fix}(g)$ is nowhere dense, and ${\rm Iso}(\Phi)$ is meager.
  \endproof

\label{lem-lifting}

      Thus we have   $h_1 =  \rho_{{\mathcal G}}(\overline{\gamma})   \circ h_2$ and so 
\begin{eqnarray}
(\widehat{g}_1, h_1)  = (\widehat{g}_1 , \rho_{{\mathcal G}}(\overline{\gamma})   \circ h_2)  & = &  (\widehat{g}_1 , C(\gamma^{-1})   \circ h_2)  \\ 
& \sim_{\zeta}  &  ( \rho_{{\mathcal G}}(\overline{\gamma}^{-1})( \widehat{g}_1) , h_2) \nonumber \\
& =  &  ( \widehat{g}_1 \cdot \overline{\gamma}^{-1} , h_2) =  (\widehat{g}_1 \cdot   \widehat{g}_1^{-1}   \widehat{g}_2 , h_2)   =  (\widehat{g}_2 , h_2)  \ .\nonumber
\end{eqnarray}
It follows that $\overline{\mu}_{{\mathcal G}}$ is injective, as claimed. 
     \endproof

<!-- thm-14: body_md -->
\label{thm-normalize}
Let $\mathfrak{M}_{\Phi}$ be a solenoidal manifold. 
Assume the baseleaf $L_{\widehat{x}} \subset \mathfrak{M}_{\Phi}$ is simply connected. Then there is an inclusion of topological spaces
\begin{equation}\label{eq-tau21}
\overline{\mu}_{{\mathcal G}} \colon   {\mathfrak{A}}_{{\mathcal G}} \times_{\zeta}   {\rm Mod}(\mathfrak{M}_{\Phi} ; \widehat{x})     \to {\rm Mod}(\mathfrak{M}_{\Phi})  \ .
\end{equation}
<!-- end thm-14 -->
<!-- head-15: body_md -->
By Proposition~\ref{prop-nonhomogeneous}, the map 
 $\overline{\mu}_{{\mathcal G}} \colon   {\mathfrak{A}}_{{\mathcal G}} \times_{\zeta} {\rm Homeo}(\mathfrak{M}_{\Phi} ; L_{\widehat{x}})    \longrightarrow {\rm Homeo}(\mathfrak{M}_{\Phi})$
 is injective. We need to show that the same is true for the induced map on isotopy classes of maps.
 
 
We first extend the map $\overline{\mu}_{{\mathcal G}}$ to an action of ${\mathfrak{A}}_{{\mathcal G}}$ on ${\rm Mod}(\mathfrak{M}_{\Phi} ; L_{\widehat{x}})$.
That is, me must show that  the action of $\overline{\mu}_{{\mathcal G}}$ preserves the isotopy equivalence classes. 
 Let     $h_t \in {\rm Homeo}(\mathfrak{M}_{\Phi} ; L_{\widehat{x}})$, for $0 \leq t \leq 1$, be an isotopy between $h_0$ and $h_1$, then consider a lift to a family of maps 
  \begin{equation}
\widetilde{h}_t \colon ( \widetilde{M}_0 \times X_{{\mathcal G}} ; \widetilde{M}_0 \times \{x\})  \to (\widetilde{M}_0 \times X_{{\mathcal G}}  ; \widetilde{M}_0 \times \{x\})   \quad , \quad 0 \leq t \leq 1  \ . 
\end{equation}

The space $X_{{\mathcal G}}$ is totally disconnected, so for each $\widehat{y} \in \mathfrak{M}_{\Phi}$ the trace curve $t \mapsto h_t(\widehat{y})$, for $0 \leq t \leq 1$, of the isotopy is a leafwise curve. Thus, for $\widehat{g} \in {\mathfrak{A}}_{{\mathcal G}}$ the action of $\rho_{{\mathcal G}}(\widehat{g})$ on $X_{{\mathcal G}}$ permutes these trace curves, and hence preserves the   isotopy equivalence relation between the maps $h_0$ and $h_1$.
It follows that there is  a well-defined map on the isotopy classes  
\begin{equation}\label{eq-modinduced}
\mu_{{\mathcal G}} \colon  {\mathfrak{A}}_{{\mathcal G}} \times {\rm Mod}(\mathfrak{M}_{\Phi} ; L_{\widehat{x}})   \to {\rm Mod}(\mathfrak{M}_{\Phi})  \ .
\end{equation}
For each $\widehat{g} \in {\mathfrak{A}}_{{\mathcal G}}$, the map $\rho_{{\mathcal G}}(\widehat{g})$ is uniformly continuous on $\mathfrak{M}_{\Phi}$,  and therefore acts uniformly continuous on ${\rm Homeo}(\mathfrak{M}_{\Phi})$. Thus we obtain a continuous action by  $ {\mathfrak{A}}_{{\mathcal G}} $ which descends to an action on the quotient space
\begin{equation}\label{eq-actionmap}
\overline{\mu}_{{\mathcal G}} \colon   {\mathfrak{A}}_{{\mathcal G}} \times_{\zeta}   {\rm Mod}(\mathfrak{M}_{\Phi} ; L_{\widehat{x}})     \to  {\rm Mod}(\mathfrak{M}_{\Phi})  \ .
\end{equation}
We next show that the  map \eqref{eq-actionmap}  is injective.  
Let  $h, h' \in {\rm Homeo}(\mathfrak{M}_{\Phi} ; L_{\widehat{x}})$ and $\widehat{g}, \widehat{g}' \in {\mathfrak{A}}_{{\mathcal G}}$  be such that   $\rho_{{\mathcal G}}(\widehat{g})(h)$ and $\rho_{{\mathcal G}}(\widehat{g}')(h')$ are    isotopic. 
Then for $\widehat{g}'' =   \widehat{g}' \cdot \widehat{g}^{-1}$,  the maps $h$ and $\rho_{{\mathcal G}}(\widehat{g}'') \circ h'$ are isotopic in ${\rm Homeo}(\mathfrak{M}_{\Phi})$, and we need to show they are isotopic in ${\rm Homeo}(\mathfrak{M}_{\Phi} ; L_{\widehat{x}})$. 

 

Let    $H \colon \mathfrak{M}_{\Phi} \times [0,1] \to \mathfrak{M}_{\Phi}$ be an isotopy with  
$$h =  H |_{t=0}  \colon \mathfrak{M}_{\Phi} \times \{0\} \to \mathfrak{M}_{\Phi} \quad {\rm  and} \quad  \rho_{{\mathcal G}}(\widehat{g}'') \circ h' =  H |_{t=1}  \colon \mathfrak{M}_{\Phi} \times \{1\} \to \mathfrak{M}_{\Phi}  \  .$$
The map $h$ preserves the leaf $L_{\widehat{x}}$, thus  $h(\widehat{x}) \in L_{\widehat{x}}$ and hence  the trace $H(\widehat{x}, t)$  is a path in $L_{\widehat{x}}$. 

Let $h_t(\widehat{y}) = H(\widehat{y}, t)$ for $\widehat{y} \in \mathfrak{M}_{\Phi}$, which defines $h_t \in {\rm Homeo}(\mathfrak{M}_{\Phi})$.  We have that $h_t(\widehat{x}) = H(\widehat{x}, t) \in L_{\widehat{x}}$ for $0 \leq t \leq 1$, so $h_t$ is an isotopy in ${\rm Homeo}(\mathfrak{M}_{\Phi} ; L_{\widehat{x}})$. 
 It follows that 
the map $\overline{\mu}_{{\mathcal G}}$ in \eqref{eq-actionmap} is injective.

   
Finally, by   Proposition~\ref{prop-baseleaf} we have  ${\rm Mod}_{BLF}(\mathfrak{M}_{\Phi}) \cong {\rm Mod}(\mathfrak{M}_{\Phi} ; \widehat{x})$ so that Theorem~\ref{thm-normalize} is proven. 
\endproof

<!-- lemma-12: body_md -->
\cite[Corollary~5.4]{DHL2016a} \label{lem-cnn}
 The normal core $\bigcap_{\gamma \in \Gamma} \ \  \overline{\gamma}^{-1} \ {\mathfrak{D}}_{{\mathcal G}} \ \overline{\gamma} \ = \ \{e\}$ is trivial.

<!-- head-17: body_md -->
Recall that     ${\mathfrak{D}}_{{\mathcal G}}$ is identified with a subgroup of ${\rm Homeo}(X_{{\mathcal G}})$, via the left action of ${\mathfrak{G}}_{{\mathcal G}}$ on the space, so each non-trivial  $\widehat{h} \in {\mathfrak{D}}_{{\mathcal G}}$ acts non-trivially on $X_{{\mathcal G}} \cong {\mathfrak{G}}_{{\mathcal G}}/{\mathfrak{D}}_{{\mathcal G}}$.

\endproof
