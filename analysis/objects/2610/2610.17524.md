<!-- lemma-1: body_md -->
\label{lemma: exterior determination for n=1 s=1/2}
 Let $x_0\in W$ be a common Lebesgue point of
$a_1$ and $a_2$. If
\begin{equation}
    \label{eq:equal-DN-ext-1d}
    \Lambda_{\gamma_1}f|_W=\Lambda_{\gamma_2}f|_W
    \qquad\text{for all }f\in C_c^\infty(W),
\end{equation}
then
\[
    \gamma_1(x_0)=\gamma_2(x_0).
\]
Consequently, $\gamma_1=\gamma_2$ a.e. in $W$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
First, let us choose an open bounded interval $W_0\Subset W$ such that
\[
    x_0\in W_0\quad\text{and}\quad \operatorname{dist}(W_0,\Omega)>0.
\]
Choose $\psi\in C_c^\infty((-1,1))$, not identically zero, and, for $N>0$, set
\begin{equation}
\label{eq:def_psiN}
    \psi_N(x)=A\psi(N(x-x_0)),
\end{equation}
where $A>0$ is selected so that
\begin{equation}
\label{eq: normalization}
    \frac{c_{1,1/2}}{2}\int_{\mathbb R^2}
    \frac{|\psi_N(x)-\psi_N(y)|^2}{|x-y|^2}\,dx\,dy=1
\end{equation}
for every $N >0$. This is possible because the left hand side is independent of
$N$, by the change of variables $X=N(x-x_0)$, $Y=N(y-x_0)$. Moreover,
\begin{equation}
\label{eq:L2-vanishing-psiN}
    \|\psi_N\|_{L^2(\mathbb R)}^2
    =
    N^{-1}A^2\|\psi\|_{L^2(\mathbb R)}^2
    \to 0.
\end{equation}
Since $\psi\in C_c^\infty((-1,1))$ and $W_0$ is an open interval containing $x_0$, we have $\psi_N\in C_c^\infty(W_0)$ for all sufficiently large $N>0$.

Let $u_{j,N}\in H^{1/2}({\mathbb R})$ be the unique solution of $L_{\gamma_j} u=0$ in $\Omega$ with exterior value $f=\psi_N$. The exterior
energy estimate \eqref{eq:energy-estimate-frac-cond} and $\operatorname{dist}(W_0,\Omega)>0$, together with the support conditions $\operatorname{supp}(u_{j,N}-\psi_N)\subset \overline\Omega$ and $\operatorname{supp} \psi_N\subset W_0$, imply that
\begin{equation}
\label{eq:energy-estimate-ext-1d}
    \|u_{j,N}-\psi_N\|_{H^{1/2}(\mathbb R)}
    \leq C\|\psi_N\|_{L^2(W_0)}
    \to 0;
\end{equation}
see \cite[Lemma~3.1]{RZ2022LowReg}. Thus, the definition of the DN map $\Lambda_\gamma$ \eqref{eq: DN map}, $u_{j,N}-\psi_N\in \widetilde H^{1/2}(\Omega)$, and $L_{\gamma_j}u_{j,N}=0$ in $\Omega$ yield
\begin{equation}
\label{eq:DN-decomp}
    \begin{split}
    &\langle\Lambda_{\gamma_j}\psi_N,\psi_N\rangle
    =
    B_{\gamma_j}(u_{j,N},u_{j,N})\\
    &\quad = B_{\gamma_j}(\psi_N,\psi_N)+2B_{\gamma_j}(u_{j,N}-\psi_N,\psi_N)+B_{\gamma_j}(u_{j,N}-\psi_N,u_{j,N}-\psi_N)\\
    &\quad = B_{\gamma_j}(\psi_N,\psi_N)+B_{\gamma_j}(u_{j,N}-\psi_N,\psi_N)\\
    &\quad =
    B_{\gamma_j}(\psi_N,\psi_N)+o(1)
    \end{split}
\end{equation}
as $N\to\infty$. In the last step, we used the continuity of the bilinear form $B_\gamma$, \eqref{eq:energy-estimate-ext-1d}, \eqref{eq: normalization}, and \eqref{eq:L2-vanishing-psiN}.

We claim that
\begin{equation}
\label{eq:convergence-psiN-energy}
    B_{\gamma_j}(\psi_N,\psi_N)\to\gamma_j(x_0)\quad \text{as }N\to\infty.
\end{equation}
For the time bing, let us fix $j=1,2$, and set $a\vcentcolon =a_j$, $a_0\vcentcolon =a(x_0)$. Since $a\in L^{\infty}({\mathbb R})$ its precise representative is bounded,
\begin{equation}
\label{eq:product-estimate}
    |a(x)a(y)-a_0^2|
    \leq
    C\bigl(|a(x)-a_0|+|a(y)-a_0|\bigr).
\end{equation}
Using \eqref{eq: normalization} and symmetry in $x,y$, we get
\begin{equation}
\label{eq: estimate bilinear form}
\begin{aligned}
    \big|B_{\gamma_j}(\psi_N,\psi_N)-a_0^2\big|
    &\leq
    C\int_{\mathbb R}
    |a(z)-a_0|K_N(z)\,dz,
\end{aligned}
\end{equation}
where
\[
    K_N(z)
    :=
    \int_{\mathbb R}
    \frac{|\psi_N(x)-\psi_N(z)|^2}{|x-z|^2}\,dx .
\]
By scaling and \eqref{eq:def_psiN},
\begin{equation}
\label{eq: def KN}
    K_N(z)=N K(N(z-x_0))\quad\text{with}\quad
    K(Z):=A^2\int_{\mathbb R}
    \frac{|\psi(X)-\psi(Z)|^2}{|X-Z|^2}\,dX .
\end{equation}
The kernel $K$ belongs to $L^1(\mathbb R)$, because $C_c^{\infty}({\mathbb R})\subset H^{1/2}({\mathbb R})$.
    
   
                                                                                                 
   
                                                                                                                                                                        
                 
                               
                                                                                                
               
By the normalization of $\psi_N$, \eqref{eq: normalization}, we have
\[
    m_K \vcentcolon =\int_{\mathbb R}K(Z)\,dZ=\frac{2}{c_{1,1/2}}.
\]
In particular
$(K_N)_{N>0}$ is an $L^1$-approximate
identity with total mass $m_K$. More explicitly, for every $\delta>0$, by Lebesgue's dominated convergence theorem, we deduce that
\begin{equation}
\label{eq: vanishing at infinity}
    \int_{|z-x_0|>\delta}K_N(z)\,dz
    =
    \int_{|Z|>N\delta}K(Z)\,dZ
    \to 0.
\end{equation}
Since $x_0$ is a Lebesgue point of $a$, we obtain the convergence
\begin{equation}
\label{eq: Lebesgue point argument}
    \int_{\mathbb R}|a(z)-a_0|K_N(z)\,dz\to 0\quad \text{as}\quad N\to\infty.
\end{equation}
Next we demonstrate this in detail. First, \eqref{eq: def KN} and the change of variables $Z=N(z-x_0)$ yield
\[
\int_{\mathbb R}|a(z)-a_0|K_N(z)\,dz
=
\int_{\mathbb R}|a(x_0+Z/N)-a_0|K(Z)\,dZ.
\]
Now we split the integral into a near part and a tail, $\int_{\mathbb R}
=
\int_{|Z|\le R}
+
\int_{|Z|>R}$, for some fixed $R>0$. For the tail, since $a$ is bounded, we get
\[
    \int_{|Z|>R}|a(x_0+Z/N)-a_0|K(Z)\,dZ
\le
2\|a\|_{L^\infty({\mathbb R})}\int_{|Z|>R}K(Z)\,dZ.
\]
Because $K\in L^1({\mathbb R})$, for any $\varepsilon>0$, 
\begin{equation}
\label{eq: model tail estimate}
    \int_{|Z|>R}|a(x_0+Z/N)-a_0|K(Z)\,dZ <\varepsilon
\end{equation}
for sufficiently large $R>0$. Next, we show that for every fixed $R>0$, we have
\begin{equation}
\label{eq:convergence-bulk-part}
    \int_{|Z|\le R}|a(x_0+Z/N)-a_0|K(Z)\,dZ \to 0
\end{equation}
as $N\to\infty$. Since $K\in L^1([-R,R])$ is nonnegative, there exists a step function\footnote{For any set $A\subset{\mathbb R}^n$, $\chi_A$ denotes the characteristic function of $A$.}
\begin{equation}
\label{eq:step-fct}
    S(Z)=\sum_{m=1}^M c_m \chi_{I_m}(Z)
\end{equation}
such that
\begin{equation}
\label{eq: approximation by step function}
    \|K\chi_{[-R,R]}-S\|_{L^1} < \varepsilon,
\end{equation}
where, without loss of generality, we may assume that each $I_m$ is a bounded interval contained in $[-R,R]$. 

Indeed, by the standard mollification argument, there exists a nonnegative function $H\in C([-R,R])$ such that
\[
    \|K\chi_{[-R,R]}-H\|_{L^1([-R,R])}<\varepsilon/2.
\]
Using the uniform continuity of $H$ on $[-R,R]$, we may choose a partition $-R=t_0<t_1<\cdots<t_M=R$ of $[-R,R]$ such that
\[
    |H(x)-H(y)| < \frac{\varepsilon}{4R},
\]
whenever $x,y\in [t_{m-1},t_{m}]$ for some $m=1,\ldots,M$. Now, define the step function
\[
    S(x)\vcentcolon =\sum_{m=1}^M H(\xi_m)\chi_{[t_{m-1},t_m)}(x),
\]
where $\xi_m\in [t_{m-1},t_m)$, $m=1,\ldots,M$, which has by construction the form \eqref{eq:step-fct} with
\[
    c_m\vcentcolon = H(\xi_m)\quad\text{and}\quad I_m\vcentcolon = [t_{m-1},t_m),\quad\text{for }m=1,\ldots,M.
\]
Then
\[
    \|H-S\|_{L^1([-R,R])}
\le
\sum_{m=1}^M\int_{t_{m-1}}^{t_m} |H(\xi_m)-H(x)|\,dx
\leq 
\varepsilon/2.
\]
Therefore we deduce that the step function $S$ satisfies
\[
    \|K\chi_{[-R,R]}-S\|_{L^1([-R,R])}
\le
\|K\chi_{[-R,R]}-H\|_{L^1([-R,R])}+\|H-S\|_{L^1([-R,R])}
<
\varepsilon,
\]
which proves the approximation property \eqref{eq: approximation by step function}.

Therefore, we may now write 
\begin{equation}
\label{eq: near part}
    \int_{|Z|\le R}|a(x_0+Z/N)-a_0|K(Z)\,dZ=\sum_{m=1}^M c_m \int_{I_m}|a(x_0+Z/N)-a_0|\,dZ+\mathcal{O}(\varepsilon).
\end{equation}
We show now that each integral in \eqref{eq: near part} goes to zero as $N\to\infty$. Fix $m=1,\ldots,M$ and apply the change of variables $z=x_0+Z/N$, so that
\[
    \int_{I_m} |a(x_0+Z/N)-a_0|\,dZ
=
N\int_{x_0+I_m/N}|a(z)-a_0|\,dz.
\]
The interval $x_0+I_m/N$ has length $|I_m|/N$ and lies inside $[x_0-R/N, x_0+R/N]$.
Therefore,
\[
\begin{split}
    N\int_{x_0+I_m/N}|a(z)-a_0|\,dz
&\le
N\int_{x_0-R/N}^{x_0+R/N}|a(z)-a_0|\,dz\\
&=2R \cdot
\frac{1}{2R/N}
\int_{x_0-R/N}^{x_0+R/N}|a(z)-a_0|\,dz.
\end{split}
\]
Since $x_0\in W_0$ is a Lebesgue point of $a$, we have
\[
    \frac{1}{2R/N}
\int_{x_0-R/N}^{x_0+R/N}|a(z)-a_0|\,dz\to 0\quad \text{as }N\to\infty.
\]
Since there are finitely many pieces in \eqref{eq: near part}, we get
\[
    \limsup_{N\to\infty}\int_{|Z|\le R}|a(x_0+Z/N)-a_0|K(Z)\,dZ=\mathcal{O}(\varepsilon).
\]
Therefore \eqref{eq: model tail estimate} ensures that
\[
    \limsup_{N\to\infty}\int_{\mathbb R}|a(z)-a_0|K_N(z)\,dz=\mathcal{O}(\varepsilon).
\]
Finally let $\varepsilon \to 0$ to establish \eqref{eq: Lebesgue point argument}.

Consequently, \eqref{eq: estimate bilinear form} yields
\[
    B_{\gamma_j}(\psi_N,\psi_N)\to a_j(x_0)^2=\gamma_j(x_0)\quad\text{as }N\to\infty
\]
and hence proves the asserted convergence \eqref{eq:convergence-psiN-energy}.

Now, the decomposition \eqref{eq:DN-decomp} and the equality of DN maps \eqref{eq:equal-DN-ext-1d} demonstrate that
\[
    \gamma_1(x_0)=\lim_{N\to\infty}\langle\Lambda_{\gamma_1}\psi_N,\psi_N\rangle
    =
    \lim_{N\to\infty}\langle\Lambda_{\gamma_2}\psi_N,\psi_N\rangle =\gamma_2(x_0).
\]
Since almost every point of $W$ is a common Lebesgue point of $a_1,a_2$, the assertion
follows.

<!-- lemma-2: body_md -->
\label{lemma: equality of weighted solutions}
Assume that $a_1=a_2$ a.e.~in $W$. Let $V\Subset W$ be a nonempty open
interval and assume that the exterior value $f\in C_c^\infty(W)$ vanishes in a neighborhood of $V$. Suppose that
\begin{equation}
\label{eq: equality of DNs weighted}
    \Lambda_{\gamma_1}f |_W=\Lambda_{\gamma_2}f|_W.
\end{equation}
For $j=1,2$, let $u_j^f\in H^{1/2}({\mathbb R})$ be the unique solution to the Dirichlet problem
\begin{equation}
\begin{cases}
\label{eq:Dirichlet-frac-cond-1d}
    L_{\gamma_j} u=0 &\text{in }\Omega,\\
    u=f &\text{in }\Omega_e,
\end{cases}
\end{equation}
then
\[
    a_1u_1^f=a_2u_2^f
    \qquad\text{a.e. in }\mathbb R .
\]
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Let $h\in L^2({\mathbb R})$ be given by
\[
    h\vcentcolon =a_1u_1^f-a_2u_2^f.
\]
Since $u_j^f=f$ in $\Omega_e$, $f=0$ near $V$, and $a_1=a_2$ on
$W\supset V$, we have
\[
    h=0 \qquad\text{in }V.
\]
Next, let us recall that $(-\Delta)^{1/2} u=\mathcal{F}^{-1}( |\xi| \hat{u})$ may be defined for any $u\in L_{1/2}$ in a distributional sense and, in particular, for $h\in L^2({\mathbb R})$ (see the discussion following \eqref{eq: multiplier frac Lap}). Moreover, since $h=0$ in $V$ and $L^2({\mathbb R}^n)\subset L_{1/2}$, \cite[Proposition~2.1.4]{silvestrephdthesis} demonstrates that $(-\Delta)^{1/2}$ is a continuous function in $V$ and it is given by the integral
\begin{equation}
\label{eq:half-lap-1d}
    (-\Delta)^{1/2}h(x)=c_{1,1/2}\int_{{\mathbb R}}\frac{h(x)-h(y)}{|x-y|^2}\,dy=- c_{1,1/2}\int_{{\mathbb R}\setminus V}\frac{h(y)}{|x-y|^2}\,dy
\end{equation}
for $x\in V$. Consequently, $(-\Delta)^{1/2} h\in L^1_{\text{loc}}(V)$.

Next, let $\varphi\in C_c^\infty(V)$. The equality of the DN maps, \eqref{eq: equality of DNs weighted}, gives
\begin{equation}
    B_{\gamma_1}(u_1^f,\varphi)=B_{\gamma_2}(u_2^f,\varphi).
\end{equation}
Since $u_j^f=0$ in $V$, $\operatorname{supp}\varphi\subset V$, and
$a_1=a_2=:a$ a.e.~in $V$, the kernel formula \eqref{eq: fractional bilinear form} reduces this identity to
\begin{equation}
\label{eq: integral identity}
    \int_V a(x)\varphi(x)
    \int_{\mathbb R\setminus V}\frac{h(y)}{|x-y|^2}\,dy\,dx=0.
\end{equation}
Indeed,
\begin{equation}
\label{eq: difference bilinear forms}
\begin{split}
    0&= \frac{2}{c_{1,1/2}}(B_{\gamma_1}(u_1^f,\varphi)-B_{\gamma_2}(u_2^f,\varphi))\\
    & =\int_{{\mathbb R}^2}\left\{a_1(x)a_1(y)(u_1^f(x)-u_1^f(y))-a_2(x)a_2(y)(u_2^f(x)-u_2^f(y))\right\}\\
    &\qquad \quad\times \frac{\varphi(x)-\varphi(y)}{|x-y|^2}\,dxdy.
\end{split}
\end{equation}
Now, we decompose ${\mathbb R}^2$ as
\[
    {\mathbb R}^2=(V\times V)\cup (V^c\times V^c)\cup (V\times V^c)\cup (V^c\times V).
\]
On the one hand, since $a\vcentcolon =a_1=a_2$ a.e.~in $V$, $u_1^f=u_2^f=f$ in $W$, and $f=0$ in $V$, the integrand of \eqref{eq: difference bilinear forms} vanishes in $V\times V$, and on the other hand it vanishes in $V^c\times V^c$, because $\varphi\in C_c^{\infty}(V)$. Thus, using that $a\vcentcolon = a_1=a_2$ in $V$, we get
\[
\begin{split}
    0&=\int_{V} a(x)\int_{V^c}\left(\left\{a_1(y)(u_1^f(x)-u_1^f(y))-a_2(y)(u_2^f(x)-u_2^f(y))\right\}\frac{\varphi(x)}{|x-y|^2}\,dy\right)dx\\
    &\quad -\int_{V}a(y)\int_{V^c}\left(\left\{a_1(x)(u_1^f(x)-u_1^f(y))-a_2(x)(u_2^f(x)-u_2^f(y))\right\}\frac{\varphi(y)}{|x-y|^2}\,dx\right)dy\\
    &=2\int_{V} a(x)\int_{V^c}\left(\left\{a_1(y)(u_1^f(x)-u_1^f(y))-a_2(y)(u_2^f(x)-u_2^f(y))\right\}\frac{\varphi(x)}{|x-y|^2}\,dy\right)dx.
\end{split}
\]
Since $u_j^f=f=0$ in $V$, for $x\in V$ and $y\in V^c$, we have
\[
\begin{aligned}
    &a(x)\left\{
        a_1(y)(u_1^f(x)-u_1^f(y))
        -a_2(y)(u_2^f(x)-u_2^f(y))
    \right\}\\
    &\qquad =
    -a(x)\bigl(a_1(y)u_1^f(y)-a_2(y)u_2^f(y)\bigr)\\
    &\qquad =
    -a(x)h(y).
\end{aligned}
\]
Thus, we obtain
\[
\begin{split}
    0&=\int_V a(x)\varphi(x)
    \int_{V^c}\frac{h(y)}{|x-y|^2}\,dy\,dx,
\end{split}
\]
which proves \eqref{eq: integral identity}. 

Since $(-\Delta)^{1/2} h \in L^1_{\text{loc}}(V)$, $\varphi\in C_c^{\infty}(V)$, and $0<\lambda_0^{1/2}\leq a\in L^{\infty}({\mathbb R})$, it follows from \eqref{eq:half-lap-1d} and \eqref{eq: integral identity} that
\[
    (-\Delta)^{1/2}h(x) = -c_{1,1/2}\int_{\mathbb R}\frac{h(y)}{|x-y|^2}\,dy=0
    \quad\text{a.e. in }V,
\]
and hence in $\mathscr{D}^{\prime}(V)$. Together with $h=0$ a.e.~in $V$, this is precisely
\[
    (-\Delta)^{1/2}h=0\quad\text{in }V,
    \qquad
    h=0\quad\text{in }V.
\]
Since $h\in L^2(\mathbb R)$, the usual unique continuation property for the
half-Laplacian gives $h=0$ in $\mathbb R$; see Theorem~\ref{thm:UCP-energy-spaces}.

<!-- proof-3: body_md -->
Let
\begin{equation}
\label{eq: Poisson extension}
    G(x,y)=P_y*g(x),\qquad y>0,
\end{equation}
be the Poisson extension of $g$ to the upper half-plane, where the Poisson kernel $P_y$ is given by
\[
    P_y(x)\vcentcolon =\frac{1}{\pi}\frac{y}{x^2+y^2}
\]
for $x\in {\mathbb R}$, $y>0$. By \cite[Chapters~1~\&~3]{stein2016singular}, the functions $P_y$ and $G$ have the following properties:
\begin{enumerate}[(i)]
    \item\label{item: properties of P} For each $y>0$ and $1\leq p\leq \infty$, $P_y\in L^p({\mathbb R})$, and its Fourier transform is 
    \begin{equation}
    \label{eq: Fourier transform of Py}
        \widehat{P_y}(\xi)=e^{-|\xi|y},\quad y>0.
    \end{equation}
    \item\label{item: harmonic G} $P$ and $G$ are harmonic functions in ${\mathbb R}_+^2$.
    \item\label{item: boundedness of G} $G\in L^{\infty}({\mathbb R}^2_+)$ with 
    \begin{equation}
        \|G\|_{L^{\infty}({\mathbb R}^2_+)}\leq \|g\|_{L^{\infty}({\mathbb R})}.
    \end{equation}
    \item\label{item: convergence} For a.e. $x\in{\mathbb R}$, $G(x,y)\to g(x)$ as $y\downarrow 0$.
\end{enumerate} 
Now, let $I'\Subset I$ be a nonempty open interval. Since $g$ vanishes in $I$, the convolution $G$ is given by
\begin{equation}
\label{eq: representation of G}
    G(x,y)=\frac{1}{\pi}\int_{{\mathbb R}\setminus I}\frac{y}{(x-z)^2+y^2}g(z)\,dz
\end{equation}
for $x\in I'$, $y>0$. Since $|x-z|\geq \delta>0$ for all $x\in I'$ and $z\in {\mathbb R}\setminus I$, we infer that $G$ is smooth up to the boundary portion
$I'\times\{0\}\subset \partial {\mathbb R}^2_+$, that is, $G\in C^{\infty}(I'\times [0,\kappa))$, $\kappa>0$. 

To this end, let $\delta\vcentcolon=
    \operatorname{dist}(\overline{I'},\mathbb R\setminus I)>0$
and fix $\kappa>0$. For every pair of integers $k,\ell\geq0$,
there exists $C_{k,\ell,I',\delta,\kappa}>0$ such that
\[
    \left|
    \partial_x^k\partial_y^\ell
    \frac{y}{(x-z)^2+y^2}
    \right|
    \leq
    \frac{C_{k,\ell,I',\delta,\kappa}}{1+|z|^2}
\]
for $x\in\overline{I'}$, $|y|\leq\kappa$, and
$z\in\mathbb R\setminus I$.
Indeed, the identity
\[
    \frac{y}{(x-z)^2+y^2}
    =
    \frac{1}{2i}
    \left(
        \frac{1}{x-z-iy}-\frac{1}{x-z+iy}
    \right)
\]
shows that, for $k+\ell\geq1$,
\[
    \left|
    \partial_x^k\partial_y^\ell
    \frac{y}{(x-z)^2+y^2}
    \right|
    \leq
    C_{k,\ell}
    \bigl((x-z)^2+y^2\bigr)^{-(k+\ell+1)/2}.
\]
Since $|x-z|\geq\delta$, these derivatives are uniformly bounded when $z$ remains bounded. Since $I'$ is bounded, for sufficiently large $|z|$ we also have $|x-z|\geq |z|/2$, uniformly in $x\in\overline{I'}$. Thus the derivatives decay at least as $|z|^{-2}$. For $k=\ell=0$, the same decay follows from
\[
    \left|\frac{y}{(x-z)^2+y^2}\right|
    \leq \frac{\kappa}{|x-z|^2}.
\]

As $g\in L^\infty(\mathbb R)$, these bounds provide an integrable majorant for every differentiated integrand. Consequently, differentiation under the integral sign shows that
\[
    F(x,y)\vcentcolon=
    \frac{1}{\pi}\int_{\mathbb R\setminus I}
    \frac{y}{(x-z)^2+y^2}g(z)\,dz,
    \qquad (x,y)\in I'\times(-\kappa,\kappa),
\]
belongs to $C^\infty(I'\times(-\kappa,\kappa))$. For $y>0$, it agrees with $G$, and $F(x,0)=0$. Therefore $G$ is smooth up to $I'\times\{0\}$ and $G(x,0)=0$ for every $x\in I'$.

Using the same reasoning as in the proof of Lemma~\ref{lemma: equality of weighted solutions}, we conclude from the conditions $g\in L^{\infty}({\mathbb R})$ and $g=0$ in $I$ that $(-\Delta)^{1/2}g$ is continuous in $I$ and, moreover, $(-\Delta)^{1/2}g(x)$, $x\in I$, can be computed via the singular integral representation of $(-\Delta)^{1/2}$ (with $c_{1,1/2}=1/\pi$).
Additionally, $G\in C^{\infty}(I'\times [0,\kappa))$, for any $\kappa>0$, and the identity $g=(-\Delta)^{1/2}g=0$ in $I$ ensure that
\[
\begin{split}
    -\partial_yG(x,0)&=-\left.\partial_y\right|_{y=0}G(x,y)\\
    &=-\frac{1}{\pi}\left.\partial_y\right|_{y=0}\int_{{\mathbb R}\setminus I}\frac{y}{(x-z)^2+y^2}g(z)\,dz\\
    &=-\frac{1}{\pi}\int_{{\mathbb R}\setminus I}\frac{1}{(x-z)^2}g(z)\,dz\\
    &=\frac{1}{\pi}\int_{{\mathbb R}}\frac{g(x)-g(z)}{(x-z)^2}\,dz\\
    &=(-\Delta)^{1/2} g(x)=0
\end{split}
\]
for $x\in I'$. Thus $G$ has vanishing Cauchy data on $I'\times\{0\}$.

Since $G$ is harmonic \ref{item: harmonic G} and smooth up to $I'\times\{0\}$,
the identity $\partial_y^2G=-\partial_x^2G$ gives, for all
integers $k,m\geq0$,
\[
\begin{aligned}
    \partial_x^k\partial_y^{2m}G(x,0)
    &=(-1)^m\partial_x^{k+2m}G(x,0)=0,\\
    \partial_x^k\partial_y^{2m+1}G(x,0)
    &=(-1)^m\partial_x^{k+2m}\partial_yG(x,0)=0.
\end{aligned}
\]
Thus all derivatives of $G$ vanish on $I'\times\{0\}$.
Consequently, the zero extension
\[
    \widetilde G(x,y)\vcentcolon=
    \begin{cases}
        G(x,y),& x\in I',\ 0\leq y<\kappa,\\
        0,& x\in I',\ -\kappa<y<0
    \end{cases}
\]
belongs to $C^\infty(I'\times(-\kappa,\kappa))$ and is harmonic
there. Since this rectangle is connected and $\widetilde G$
vanishes on its lower half, real analyticity implies that
$\widetilde G=0$ throughout the rectangle.
In particular, $G$ vanishes on the nonempty open set
$I'\times(0,\kappa)$.

Applying unique continuation for harmonic functions in the
connected upper half-plane, we obtain
$G\equiv0$ in $\mathbb R^2_+$. Property~\ref{item: convergence}
then yields $g=0$ almost everywhere in $\mathbb R$.

$W$. Next, choose nonempty open intervals $U,V\Subset W$ with $\overline U\cap\overline V=\emptyset$, and let the exterior condition
$f\in C_c^\infty(W)$ satisfy
\[
    f\geq c_f>0 \quad\text{in }U,
    \qquad
    f=0 \quad\text{in a neighborhood of }\overline V.
\]
By Lemma~\ref{lemma: equality of weighted solutions}, we have
\[
    w\vcentcolon =a_1u_1^f=a_2 u_2^f
    \qquad\text{in }\mathbb R.
\]
Thus,
\[
    a_j(x)a_j(y)(u_j^f(x)-u_j^f(y))
    =
    a_j(y)w(x)-a_j(x)w(y)
\]
for a.e.~$x,y\in{\mathbb R}$ and all $j=1,2$. Next, let us define the function $\delta_a\in L^\infty ({\mathbb R})$ by
\[
    \delta_a(x)\vcentcolon =a_1(x)-a_2(x),\quad x\in{\mathbb R}.

Hence, for any $\varphi\in C_c^\infty (U)$, \eqref{eq: equality DN maps uniqueness n=1, s=1/2} yields

\label{theorem:bounded-fractional-ucp}
Let $V\subset\mathbb R^n$ be a nonempty open set, and assume that $g\in L_s$. If $g$ satisfies
\[
    g=0 \quad\text{a.e.~in } V,
    \qquad
    (-\Delta)^s g=0 \quad\text{in } \mathscr{D}^{\prime}(V),
\]
then $g=0$ a.e.~in $\mathbb R^n$.
<!-- end theorem-4 -->
<!-- proof-5: body_md -->
We establish the claim in five steps.\\

\noindent\textbf{Step 1:} \textit{Caffarelli--Silvestre extension.}

We denote by
\begin{equation}
\label{eq: CS extension}
    U(x,y)\vcentcolon = P^s_y*g(x),\quad (x,y)\in{\mathbb R}^{n+1}_+,
\end{equation}
the Caffarelli--Silvestre (CS) extension of $g$. Here 
\[
P^s_y(x)\vcentcolon =p_{n,s}\frac{y^{2s}}{(|x|^2+y^2)^{(n+2s)/2}}
\]
is the fractional Poisson kernel and the constant $p_{n,s}>0$ is chosen such that 
\begin{equation}
\label{eq: normalization fractional poisson}
    \int_{{\mathbb R}^n}P^s_y(x)\,dx=1.
\end{equation}
Since $g\in L_s$, the convolution \eqref{eq: CS extension} is well-defined for every $y>0$. Indeed, for fixed $x\in{\mathbb R}^n$ and $y>0$, this follows from $g\in L_s\subset L^1_{\rm loc}({\mathbb R}^n)$, $P_y^s\in L^\infty_{\mathrm{loc}}({\mathbb R}^n)$ and the tail estimate
\begin{equation}
\label{eq: tail estimate}
    P_y^s(x-z)\leq C_{n,s}\frac{y^{2s}}{1+|z|^{n+2s}},
\end{equation}
whenever $|z|\geq R_x\vcentcolon = \max(1,2|x|)$. Indeed, this yields the pointwise estimate
\begin{equation}
\label{eq:pointwise-estiamte}
    \begin{split}
        &|P^s_y\ast g(x)|\leq \int_{{\mathbb R}^n}P^s_y(x-z)|g(z)|\,dz\\
        &\leq \|P^s_y(x-z)\|_{L^{\infty}(B_{R_x})}\int_{B_{R_x}}|g(z)|\,dz+C_{n,s}y^{2s}\int_{B^c_{R_x}}\frac{|g(z)|}{1+|z|^{n+2s}}\,dz\\
        &\leq \|P^s_y\|_{L^{\infty}(B_{R_x}(x))}\|g\|_{L^1(B_{R_x})}\,dz+C_{n,s}y^{2s}\|g\|_{L_s}\\
        &\leq p_{n,s}y^{-n}\|g\|_{L^1(B_{R_x})}\,dz+C_{n,s}y^{2s}\|g\|_{L_s}<\infty
    \end{split}
\end{equation}
for every $x\in{\mathbb R}^n$ and every $y>0$.
Hence, it remains to show \eqref{eq: tail estimate}. The condition $|z|\geq 2|x|$ yields
\[
    |x-z|\geq |z|-|x|\geq \frac{|z|}{2}.
\]
Consequently,
\[
    |x-z|^2+y^2\geq |x-z|^2\geq \frac{|z|^2}{4}.
\]
Hence
\[
\begin{split}
    P_y^s(x-z)
    &=
    p_{n,s}\frac{y^{2s}}{(|x-z|^2+y^2)^{(n+2s)/2}}  \leq
    C_{n,s}y^{2s}|z|^{-n-2s},
\end{split}
\]
where the constant $C_{n,s}>0$ only depends on $n$ and $s$. Since $|z|\geq 1$, we have
\[
    |z|^{-n-2s}
    \leq \frac{2}{1+|z|^{n+2s}}.
\]
Thus, \eqref{eq: tail estimate} follows.

Next, we show that $U\in L^{\infty}_{\rm loc}({\mathbb R}^{n+1}_+)$, that is, for every compact set $K\Subset\mathbb R^n$ and all $0<r<R<\infty$,
\begin{equation}
\label{eq:uniform-bound-CS}
    \|U\|_{L^{\infty}(K\times (r,R))}\leq CR^{2s}(r^{-n-2s}\|g\|_{L^1_{\rm loc}({\mathbb R}^n)}+\|g\|_{L_s})<\infty.
\end{equation}
To this end, let us define
\begin{equation}
\label{eq: radius RK}
    R_K\vcentcolon = \max(1,2 \ell_K)\quad \text{with }\ell_K\vcentcolon = \sup_{x\in K}|x|.
\end{equation}
Similar to the computation \eqref{eq:pointwise-estiamte}, we observe that \eqref{eq: tail estimate} and $0<r<R<\infty$ yield
\[
\begin{split}
    |U(x,y)|&\leq \int_{B_{R_K}}P^s_y(x-z)|g(z)|\,dz+\int_{B^c_{R_K}}P^s_y(x-z)|g(z)|\,dz\\
    &\leq C R^{2s}r^{-n-2s}\|g\|_{L^1(B_{R_K})}+CR^{2s}\|g\|_{L_s}
\end{split}
\]
for a.e.~$x\in K$ and $r<y<R$. This proves \eqref{eq:uniform-bound-CS}.

Next, we observe that $P^s_y$, $y>0$, has the following additional properties (see \cite[Lemma~7.1]{KRZ-2023}):
\begin{enumerate}[(i)]
    \item\label{item: scaling} For any $y>0$, $P^s_y(x)=y^{-n}P^s_1(x/y)$.
    \item\label{item: decreasing} For every $y>0$, $P^s_y(x)$ is decreasing in $|x|$ and $P^s_y\in L^p({\mathbb R}^n)$, $1\leq p\leq \infty$.
    \item\label{item: Dirac property} For any $\delta >0$, we have
    \[
        \int_{{\mathbb R}^n\setminus B_\delta}P^s_y(x)\,dx\to 0\quad \text{as }y\to 0.
    \]
    \item\label{item: s harmonic} The fractional Poisson kernel is of class $C^{\infty}({\mathbb R}^{n+1}_+)$ and solves the degenerate elliptic equation
    \[
        \operatorname{div}(y^{1-2s}\nabla P^s)=0\quad\text{in }{\mathbb R}^{n+1}_+.
    \]
\end{enumerate}
The properties \ref{item: scaling}--\ref{item: decreasing} follow by direct computation. Combining \ref{item: scaling} and
\[
    \|P^s_1\|_{L^1({\mathbb R}^n)}=\|P^s_y\|_{L^1({\mathbb R}^n)}=1
\]
(see \eqref{eq: normalization fractional poisson}) yields \ref{item: Dirac property}. The statement \ref{item: s harmonic} follows again by direct computation. We also note that the degenerate divergence operator can be rewritten as
\begin{equation}
\label{eq: degenerate divergence operator}
    \operatorname{div}(y^{1-2s}\nabla u)=y^{1-2s}\left(\Delta_{x,y}+\frac{1-2s}{y}\partial_y\right)u.
\end{equation}

The above properties show in particular that $(P^s_y)_{y>0}$ is an approximation of the identity in the following sense. For any
$g\in L_s$,
\begin{equation}
\label{eq: trace of U is g}
    U(\cdot,y)=P_y^s*g\to g
    \quad\text{a.e.~in }{\mathbb R}^n
\end{equation}
as $y\downarrow 0$. Moreover, when $g$ additionally satisfies $g\in L^{p}_{\rm loc}({\mathbb R}^n)$ for some $1\leq p<\infty$, then 
\begin{equation}
\label{eq: Lp trace of U is g}
    U(\cdot,y)=P_y^s*g\to g
    \quad\text{in }L^p_{\operatorname{loc}}(\mathbb R^n).
\end{equation}
                                                                                                                                                                                                         

To see the above approximation properties of the fractional Poisson kernel, let us first note that, for any $f\in L^p({\mathbb R}^n)$, $1\leq p<\infty$, we have
\begin{equation}
\label{eq: approximate identity property Lp}
    P_y^s*f\to f\quad\text{in }L^p({\mathbb R}^n),
\end{equation}
as $y\downarrow 0$. This follows from the $L^p$-continuity of translations, that is
\[
    \|f(\cdot +h)-f\|_{L^p({\mathbb R}^n)}\to 0
\]
as $|h| \to 0$ and the properties $P^s_y \geq 0$, \eqref{eq: normalization fractional poisson}, \ref{item: Dirac property} (see \cite[Satz~2.15]{alt1985lineare}). Next, suppose that $g\in L^p_{\rm loc}({\mathbb R}^n)\cap L_s$ for some $1\leq p <\infty$. Let $r>0$. Fix radii $\rho,R>0$ such that $r<\rho<R$ and choose a cutoff function $\eta \in C_c^\infty(\mathbb R^n)$ satisfying
\[
    0\leq \eta\leq 1,\quad \left.\eta\right|_{\overline{B_\rho}}=1,\quad\text{and}\quad \left.\eta\right|_{B_{R}^c}=0.
\]
Then
\begin{equation}
\label{eq:near-tail-decomp-CS}
    P_y^s*g-g
    =
    P_y^s*(\eta g)-\eta g
    +
    P_y^s*((1-\eta)g)
    \quad\text{in }B_{\rho}.
\end{equation}
By \eqref{eq: approximate identity property Lp} and $\eta g\in L^p(\mathbb R^n)$, we have
\[
    P_y^s*(\eta g)-\eta g\to 0\quad \text{in }L^p({\mathbb R}^n).
\]
Next let us show that the tail term goes to zero as $y\to 0$. Without loss of generality, we may assume that $\rho\geq \max(1,2r)$ and hence \eqref{eq: tail estimate} applies for any $x\in B_r$ and $|z|\geq \rho$. Thus, $\eta=1$ on $B_\rho$ yields
\begin{equation}
\label{eq:conv-tail-CS}
\begin{split}
    |P_y^s*((1-\eta)g)(x)|&\leq \int_{B_{\rho}^c}P^s_y(x-z)|(1-\eta(z))g(z)|\,dz\\
    &
    \leq
    C y^{2s}
    \int_{\mathbb R^n}
    \frac{|g(z)|}{1+|z|^{n+2s}}\,dz
    \to 0
\end{split}
\end{equation}
uniformly for $x\in B_r$. Thus $P_y^s*g\to g$ in $L^p(B_r)$, proving
\eqref{eq: Lp trace of U is g} whenever
$g\in L^p_{\mathrm{loc}}(\mathbb R^n)\cap L_s$. For the almost-everywhere convergence in \eqref{eq: trace of U is g}, we apply the same local/tail decomposition \eqref{eq:near-tail-decomp-CS}. More concretely, suppose that $x$ is a Lebesgue point of $g\in L_s$ and choose $r>0$ such that $x\in B_r$. Let $\eta \in C_c^{\infty}({\mathbb R}^n)$ be a cutoff function as above. The tail tends to zero locally uniformly by \eqref{eq:conv-tail-CS}. Next, we show 
\begin{equation}
\label{eq:Lebesgue-point-CS-conv}
    (P^s_y\ast (\eta g)-\eta g)(x)\to 0,\quad y\downarrow 0.
\end{equation}
For convenience, set $h\vcentcolon = \eta g\in L_s \cap L^1({\mathbb R}^n)$. One easily sees that $x$ is again a Lebesgue point for $h$. The function
\[
    p^s_y(t)\vcentcolon=
    p_{n,s}\frac{y^{2s}}{(t^2+y^2)^{(n+2s)/2}}.\quad t>0,
\]
satisfies
\begin{equation}
\label{eq:decay-deriv}
    0\leq -\frac{dp^s_y}{dt}(t)
    =(n+2s)p_{n,s}
    \frac{y^{2s}t}{(t^2+y^2)^{(n+2s+2)/2}}
   =o(t^{-n-s-1})\quad\text{as }t\to\infty.
\end{equation}
Consequently, \eqref{eq: normalization fractional poisson}, the polar decomposition \cite[Section~3.4.4]{evans-mt}, and integration by parts give
\[
\begin{aligned}
    |(P_y^s*h)(x)-h(x)|
    &\leq
    \int_{\mathbb R^n}|h(x-z)-h(x)|p^s_y(|z|)\,dz\\
    &=\int_{0}^\infty p^s_y(t)\left( \int_{\partial B_t}|h(x-z)-h(x)|\,d\mathcal H^{n-1}(z)\right)\,dt\\
    &=-\int_0^\infty \frac{dp^s_y}{dt}(t)\left(\int_{B_t}|h(x-z)-h(x)|\,dz\right)\,dt.
\end{aligned}
\]
From the second to third line we used \eqref{eq:decay-deriv}, 
\[
    \fint_{B_t}|h(x-z)-h(x)|\,dz\to 0
\]
because $x$ is a Lebesgue point, and the fact that for every $k\in L^1_{\rm loc}({\mathbb R}^n)$ one has
\[
    \frac{d}{dt}\int_{B_t} k\,dx=\int_{\partial B_t} k\,d\mathcal{H}^{n-1}
\]
for a.e.~$t>0$. Let $\varepsilon>0$. Since $x$ is a Lebesgue point, there exists $\delta>0$ such that
\[
    \int_{B_t}|h(x-z)-h(x)|\,dz\leq\varepsilon |B_t|,
    \qquad 0<t\leq\delta.
\]
The normalization of $P_y^s$ and Tonelli's theorem imply
\[
    -\int_0^\infty\frac{dp^s_y}{dt}(t)|B_t|\,dt
    =\int_{\mathbb R^n}P_y^s(z)\,dz=1.
\]
Hence
\[
    -\int_0^\delta\frac{dp^s_y}{dt}(t)\left(\int_{B_t}|h(x-z)-h(x)|\,dz\right)\,dt
    \leq\varepsilon.
\]
For $t\geq\delta$, we use
\[
    \int_{B_t}|h(x-z)-h(x)|\,dz\leq
    \|h\|_{L^1(\mathbb R^n)}+|h(x)|\,|B_t|,
    \qquad
    -\frac{dp^s_y}{dt}(t)\leq C_{n,s}y^{2s}t^{-n-2s-1},
\]
to obtain
\[
\begin{split}
    &-\int_\delta^\infty \frac{dp^s_y}{dt}(t)\left(\int_{B_t}|h(x-z)-h(x)|\,dz\right)\,dt\\
    &\leq C_{n,s}y^{2s}
    \left(
        \|h\|_{L^1(\mathbb R^n)}\delta^{-n-2s}
        +|h(x)|\delta^{-2s}
    \right)
    \longrightarrow0
\end{split}
\]
as $y\downarrow0$. Therefore
\[
    \limsup_{y\downarrow0}|(P_y^s*h)(x)-h(x)|
    \leq\varepsilon.
\]
Letting $\varepsilon\downarrow0$ yields
$(P_y^s*h)(x)\to h(x)$, which verifies \eqref{eq:Lebesgue-point-CS-conv}.
Together with the previously established local uniform
convergence $P_y^s*((1-\eta)g)\to0$ on $B_r$, this proves
\[
    (P_y^s*g)(x)\to g(x)\quad \text{as }y\downarrow 0.
\]
Finally, since almost every point is a Lebesgue point of $g\in L^1_{\mathrm{loc}}(\mathbb R^n)$, the assertion \eqref{eq: trace of U is g} follows.

Next, we show that $U$ solves
\begin{equation}
\label{eq: CS extension PDE}
    \operatorname{div}(y^{1-2s}\nabla U)=0
    \quad\text{in }\mathbb R^{n+1}_+.
\end{equation}
By the elementary kernel estimates, see for instance
\cite[Appendix~B, estimate~(78)]{colombo2020generalized}, one has
\begin{equation}
\label{eq: estimates Poisson kernel}
    |\partial_y^k P_y^s(x)|
    \leq
    C_{n,s,k} y^{-k}P_y^s(x),
    \qquad k\in\mathbb N,
\end{equation}
and
\[
    |\partial_x^\alpha P_y^s(x)|
    \leq
    C_{\alpha,n,s}
    \frac{y^{2s}}
    {(|x|^2+y^2)^{(n+2s+|\alpha|)/2}}.
\]
In particular,
\begin{equation}
\label{eq: estimates Poisson kernel Laplacian}
    |\Delta_x P_y^s(x)|
    \leq
    C_{n,s} y^{-2}P_y^s(x).
\end{equation}
Thus, \eqref{eq: CS extension PDE} is a consequence of \ref{item: s harmonic}, \ref{item: decreasing}, \eqref{eq: degenerate divergence operator}, \eqref{eq: estimates Poisson kernel}, \eqref{eq: estimates Poisson kernel Laplacian},
and Lebesgue's dominated convergence theorem.

\medskip

\noindent\textbf{Step 2:} \textit{$H^1_{\rm loc}$-Regularity of CS extension.}

Next, let us record the local information needed at boundary points of $V\times \{0\}$. Fix
$V'\Subset V$. Since $g=0$ a.e.~in $V$, one may write
\[
    U(x,y)
    =
    \int_{\mathbb R^n\setminus V}
    P^s_y(x-z)g(z)\,dz ,\quad x\in{\mathbb R}^n.
\]
Let $K\Subset V'\Subset V$ and set
\begin{equation}
\label{eq: K distances}
    d_K\vcentcolon =\operatorname{dist}(K,\mathbb R^n\setminus V)>0.
\end{equation}
We assert that for every multi-index $\alpha\in{\mathbb N}^n_0$, one has
\begin{equation}
\label{eq: tangential derivatives of frac poisson}
    |\partial_x^\alpha P^s_y(x-z)|
    \leq
    C_{\alpha,K}
    \frac{y^{2s}}{(1+|x-z|^2+y^2)^{(n+2s+|\alpha|)/2}},
\end{equation}
for all $x\in K,\ y>0$, and $z\in \mathbb R^n\setminus V$.

First assume that $|\alpha|=0$. Since $K\Subset V$ implies that $1\leq d_K^{-1}|x-z|$ for $x\in K$, $z\in {\mathbb R}^n\setminus V$, we have
\begin{equation}
\label{eq: singularity pos distance}
    1+|x-z|^2+y^2 \leq C_K(|x-z|^2+y^2)
\end{equation}
for all $x\in K$, $y>0$, and $z\in\mathbb R^n\setminus V$. Thus, for $x\in K$ and $z\in{\mathbb R}^n\setminus V$, we can estimate
\[
    P^s_y(x-z)\leq C_{K}\frac{y^{2s}}{(1+|x-z|^2+y^2)^{(n+2s)/2}},
\]
which yields \eqref{eq: tangential derivatives of frac poisson} with $\alpha=0$.

Next, we consider the case $|\alpha|\geq 1$. Set
\[
    w=x-z,\quad q(w,y)=|w|^2+y^2,\quad F(t)=t^{-\beta},\quad
    \beta=\frac{n+2s}{2}.
\]
Then
\[
    P_y^s(x-z)=p_{n,s}y^{2s}F(q(w,y)).
\]
Since the $x$-derivatives are the same as the $w$-derivatives, it is enough
to estimate $\partial_w^\alpha (F\circ q)$. 

By the multivariable Fa\`a di Bruno formula,
\[
    \partial_w^\alpha(F\circ q)(w,y)
    =
    \sum_{k=1}^{|\alpha|}
    F^{(k)}(q(w,y))
    \sum_{\substack{\gamma^1+\cdots+\gamma^k=\alpha\\ |\gamma^j|\geq 1}}
    C_{\alpha,\gamma^1,\ldots,\gamma^k}
    \prod_{j=1}^k \partial_w^{\gamma^j}q(w,y).
\]
Here the expression simplifies because $q(w,y)=|w|^2+y^2$ is quadratic:
\[
    \partial_w^\gamma q=0\quad\text{if }|\gamma|\geq 3,
\]
while
\[
    |\partial_w^\gamma q|\leq C|w|
    \quad\text{if }|\gamma|=1,
    \qquad
    |\partial_w^\gamma q|\leq C
    \quad\text{if }|\gamma|=2.
\]
Moreover,
\[
    F^{(k)}(t)=C_{\beta,k}t^{-\beta-k}.
\]
Thus every nonzero term in the Fa\`a di Bruno expansion is bounded by
\[
    C q^{-\beta-k}|w|^\ell,
\]
where $\ell$ is the number of first-order derivatives of $q$ appearing in
that term. If $m$ is the number of second-order derivatives of $q$, then
\[
    k=\ell+m,
    \qquad
    |\alpha|=\ell+2m.
\]
Using $|w|^\ell\leq q(w,y)^{\ell/2}$, we obtain
\[
    q^{-\beta-k}|w|^\ell
    \leq
    q^{-\beta-k+\ell/2}
    =
    q^{-\beta-(\ell+2m)/2}
    =
    q^{-\beta-|\alpha|/2}.
\]
Consequently,
\begin{equation}
\label{eq: local bound x derivatives}
    |\partial_x^\alpha P_y^s(x-z)|
    \leq
    C_\alpha y^{2s}
    (|x-z|^2+y^2)^{-(n+2s+|\alpha|)/2}.
\end{equation}
Therefore, taking into account \eqref{eq: singularity pos distance}, we obtain
\begin{equation}
    |\partial_x^\alpha P^s_y(x-z)|
    \leq
    C_{\alpha,K}
    \frac{y^{2s}}{(1+|x-z|^2+y^2)^{(n+2s+|\alpha|)/2}} .
\end{equation}
This completes the verification of \eqref{eq: tangential derivatives of frac poisson}.

Fix $y>0$. Multiplying \eqref{eq: tangential derivatives of frac poisson} by $|g(z)|$ gives an integrable majorant of $z\mapsto \partial_x^\alpha P^s_y(x-z)g(z)$ locally uniformly for $x\in K$. Let $R_K$ be defined as in \eqref{eq: radius RK}. Then the proof of \eqref{eq: tail estimate} shows that
\begin{equation}
        \frac{1}{(1+|x-z|^2+y^2)^{(n+2s+|\alpha|)/2}} \leq \frac{C}{1+|z|^{n+2s}}
\end{equation}
for all $x\in K$, $y>0$, and $|z|\geq R_K$. Hence,
\begin{equation}
\label{eq: near far field decomposition}
\begin{split}
    &\int_{{\mathbb R}^n}\frac{|g(z)|}{(1+|x-z|^2+y^2)^{(n+2s+|\alpha|)/2}} \,dz\\
    &= \int_{B_{R_K}}\frac{|g(z)|}{(1+|x-z|^2+y^2)^{(n+2s+|\alpha|)/2}} \,dz+\int_{B_{R_K}^c}\frac{|g(z)|}{(1+|x-z|^2+y^2)^{(n+2s+|\alpha|)/2}} \,dz\\
    &\leq C\left(\|g\|_{L^1(B_{R_K})}+\int_{B_{R_K}^c}\frac{|g(z)|}{1+|z|^{n+2s}}\,dz\right)<\infty.
\end{split}
\end{equation}
Hence dominated convergence allows us
to pass tangential derivatives under the integral sign,
\[
    \partial_x^\alpha U(x,y)
    =
    \int_{\mathbb R^n\setminus V}
    \partial_x^\alpha P^s_y(x-z)g(z)\,dz .
\]
The same argument gives, for $x\in K$ and $y>0$,
\begin{equation}
\label{eq: local estimates}
    |U(x,y)|\leq Cy^{2s},\qquad
    |\nabla_x U(x,y)|\leq Cy^{2s},\qquad
    |\partial_y U(x,y)|\leq Cy^{2s-1}
\end{equation}
where the constant depends on $K$, on the $L^1_{\rm loc}$-norm of $g$, and
on the $L_s$-norm of $g$. For $|\alpha|\le1$, the estimate above gives
\[
    |\partial_x^\alpha U(x,y)|
    \leq
    C y^{2s}
    \int_{\mathbb R^n\setminus V}
    \frac{|g(z)|}{(1+|x-z|^2)^{(n+2s+|\alpha|)/2}}\,dz.
\]
The last integral is uniformly bounded for $x\in K$, by the estimate \eqref{eq: near far field decomposition}. This allows to obtain the first two estimates in \eqref{eq: local estimates}. 

Similarly, using \eqref{eq: estimates Poisson kernel} with $k=1$ and \eqref{eq: tangential derivatives of frac poisson} with $\alpha=0$, we have
\begin{equation}
\label{eq: uniform bound y derivative poisson}
    |\partial_yP_y^s(x-z)|\leq Cy^{-1}P^s_y(x-z)
    \leq
    C_K \frac{y^{2s-1}}{(1+|x-z|^2)^{(n+2s)/2}}
\end{equation}
for $x\in K$, $z\in {\mathbb R}^n\setminus V$, and $y>0$. Hence
\begin{equation}
\label{eq: uniform bound y derivative U}
    |\partial_y U(x,y)|
    \leq
    C_K y^{2s-1}
    \int_{\mathbb R^n\setminus V}
    \frac{|g(z)|}{(1+|x-z|^2)^{(n+2s)/2}}\,dz
    \leq C_K y^{2s-1},
\end{equation}
where the last integral is finite uniformly for $x\in K$. This again follows from \eqref{eq: near far field decomposition}.


As a consequence of \eqref{eq: local estimates}, we obtain
\begin{equation}
\label{eq: pointwise L2 bounds}
    y^{1-2s}|U|^2\leq Cy^{1+2s},\qquad
    y^{1-2s}|\nabla_x U|^2\leq Cy^{1+2s},\qquad
    y^{1-2s}|\partial_y U|^2\leq Cy^{2s-1},
\end{equation}
which are integrable near $y=0$ because $s>0$. Away from $y=0$, the
Poisson extension is smooth. Therefore
\begin{equation}
\label{eq: local regularity CS extension}
    U\in H^1_{\mathrm{loc}}(V'\times[0,\infty);y^{1-2s}dxdy).
\end{equation}

\medskip 

\noindent\textbf{Step 3:} \textit{Vanishing of conormal derivative.}
We claim that, for every $\varphi\in C_c^\infty(V')$,
\begin{equation}
\label{eq: vanishing conormal derivative}
    \lim_{y\downarrow 0}
    \int_{\mathbb R^n} y^{1-2s}\partial_y U(x,y)\varphi(x)\,dx
    =
    C_{n,s}\,\langle (-\Delta)^s g,\varphi\rangle
    =0,
\end{equation}
where 
\begin{equation}
\label{eq: Cs extension constant}
    C_{n,s}\vcentcolon =-2^{1-2s}\frac{\Gamma(1-s)}{\Gamma(s)}.
\end{equation}
Equivalently, $U$ has vanishing weighted conormal derivative on
$V'\times\{0\}$ in the weak sense. 

Let $K\vcentcolon = \operatorname{supp}\varphi$. By \eqref{eq: uniform bound y derivative U}, for $x\in K$ and $0<y<1$,
\[
    |y^{1-2s}\partial_yU(x,y)\varphi(x)|
    \leq
    C_K|\varphi(x)|.
\]
Since $\varphi\in C_c^\infty(V')$, the right hand side belongs to
$L^1(\mathbb R^n)$. Thus, once the pointwise limit of
$y^{1-2s}\partial_yU(x,y)$ is identified for $x\in K$, dominated
convergence may be applied in the $x$-integral.

For any $w\neq 0$ and $y>0$, we have
\[
\begin{split}
    y^{1-2s}\partial_y P_y^s(w)
    &=
    p_{n,s}
    \left[
        2s\frac{1}{(|w|^2+y^2)^{(n+2s)/2}}
        -(n+2s)\frac{y^2}{(|w|^2+y^2)^{(n+2s+2)/2}}
    \right].
\end{split}
\]
Thus,
\[
    y^{1-2s}\partial_y P^s_y(x-z)g(z)\to 2s p_{n,s}\frac{g(z)}{|x-z|^{n+2s}},\quad (y\to 0),
\]
for $x\in K$ and $z\in{\mathbb R}^n\setminus V$. Now, since $y^{1-2s}\partial_y P^s_y(x-z)g(z)$, $0<y<1$, is uniformly bounded by an integrable function (see \eqref{eq: near far field decomposition}), using dominated convergence, we may pass the $y\to 0$ limit into the $z$-integral and use the above pointwise limit to obtain
\[
\begin{split}
    \lim_{y\downarrow0}
    y^{1-2s}\partial_y U(x,y)
    &=
    2s p_{n,s}
    \int_{\mathbb R^n\setminus V}
    \frac{g(z)}{|x-z|^{n+2s}}\,dz
\end{split}
\]
for every $x\in K$. Consequently,
\[
\begin{split}
    \lim_{y\downarrow0}
    \int_{\mathbb R^n}
    y^{1-2s}\partial_y U(x,y)\varphi(x)\,dx
    &=
    2s p_{n,s}
    \int_{V'}
    \varphi(x)
    \int_{\mathbb R^n\setminus V}
    \frac{g(z)}{|x-z|^{n+2s}}\,dz\,dx .
\end{split}
\]
On the other hand, since $g=0$ in $V$, for $x\in V'$ the singular
integral defining $(-\Delta)^s g$ is nonsingular and equals
\[
    (-\Delta)^s g(x)
    =
    c_{n,s}\operatorname{p.v.}
    \int_{\mathbb R^n}
    \frac{g(x)-g(z)}{|x-z|^{n+2s}}\,dz
    =
    -c_{n,s}
    \int_{\mathbb R^n\setminus V}
    \frac{g(z)}{|x-z|^{n+2s}}\,dz .
\]
The last integral is continuous in $x\in V'$. Since, by assumption,
$(-\Delta)^s g=0$ in $\mathscr{D}^{\prime}(V)$, this continuous representative must
vanish in $V'$. Therefore
\[
    \lim_{y\downarrow0}
    \int_{\mathbb R^n}
    y^{1-2s}\partial_y U(x,y)\varphi(x)\,dx
    =
    -\frac{2s p_{n,s}}{c_{n,s}}
    \langle (-\Delta)^s g,\varphi\rangle
    =
    0 .
\]
Finally, the value of 
\[
    C_{n,s}\vcentcolon = -\frac{2s p_{n,s}}{c_{n,s}},
\]
given in \eqref{eq: Cs extension constant}, follows from \eqref{eq: constant fractional Lap} and 
\[
    p_{n,s}=\frac{\Gamma(n/2+s)}{\pi^{n/2}\Gamma(s)}.
\]
\medskip 

\noindent\textbf{Step 4:}
\textit{Local vanishing of the CS extension.}

Fix $x_0\in V$ and choose $r>0$ such that
$\overline{B_{2r}(x_0)}\subset V$.
Set $D=B_{2r}(x_0)$ and $Q=D\times(0,2r)$. Let $K=\overline{D}$ and choose an open set $V'\subset {\mathbb R}^n$ such that $D\Subset V'\Subset V$. The estimates \eqref{eq: local estimates} and the regularity assertion \eqref{eq: local regularity CS extension} from Step~2, applied with $K=\overline D$ and $V'$ as above, give
\[
    U\in H^1(Q;y^{1-2s}\,dx\,dy)\cap L^\infty(Q).
\]
Moreover,
\[
    \operatorname{div}(y^{1-2s}\nabla U)=0
    \quad\text{in }Q.
\]

We first identify the weighted Sobolev trace.
For $0<t<y<2r$, the fundamental theorem of calculus and
weighted Cauchy--Schwarz imply
\[
\begin{aligned}
    \|U(\cdot,y)-U(\cdot,t)\|_{L^2(D)}
    &\leq
    \left(\int_t^y\tau^{2s-1}\,d\tau\right)^{1/2}
    \left(
        \int_t^y\int_D
        \tau^{1-2s}|\partial_\tau U|^2\,dx\,d\tau
    \right)^{1/2}\\
    &=
    \left(\frac{y^{2s}-t^{2s}}{2s}\right)^{1/2}
    \left(
        \int_t^y\int_D
        \tau^{1-2s}|\partial_\tau U|^2\,dx\,d\tau
    \right)^{1/2}.
\end{aligned}
\]
Hence $U(\cdot,y)$ has an $L^2(D)$ limit as $y\downarrow0$.
This slice limit realizes the weighted Sobolev trace. The resulting trace map is continuous in the weighted
$H^1$ norm and agrees with boundary restriction on smooth
functions. Since \eqref{eq: local estimates} also gives
\[
    \|U(\cdot,y)\|_{L^2(D)}
    \leq C|D|^{1/2}y^{2s}\to 0,
\]
the trace is zero.

We next verify the weak formulation of the zero conormal
condition. Let
$\Phi\in C_c^\infty(D\times(-2r,2r))$.
Integration by parts in $D\times(\varepsilon,2r)$ gives
\[
    \int_{D\times(\varepsilon,2r)}
    y^{1-2s}\nabla U\cdot\nabla\Phi\,dx\,dy
    =
    -\int_D
    \varepsilon^{1-2s}\partial_yU(x,\varepsilon)
    \Phi(x,\varepsilon)\,dx.
\]
Write $\Phi(x,\varepsilon)=\Phi(x,0)
+(\Phi(x,\varepsilon)-\Phi(x,0))$.
By Step~3, the boundary integral with $\Phi(x,0)$ tends
to zero. By \eqref{eq: uniform bound y derivative U},
the remaining term is bounded in absolute value by
\[
    C\|\Phi(\cdot,\varepsilon)-\Phi(\cdot,0)\|_{L^1(D)}
    \to 0.
\]
Since $U\in H^1(Q;y^{1-2s}\,dx\,dy)$ and
$y^{1-2s}$ is integrable near $y=0$, we may also pass
to the limit on the left. Thus
\[
    \int_Q y^{1-2s}\nabla U\cdot\nabla\Phi\,dx\,dy=0.
\]
This establishes the zero conormal condition in the weak
sense, including test functions that meet the flat boundary.

The restriction of $U$ to
\[
    B_{2r}^+((x_0,0))
    \vcentcolon=
    B_{2r}((x_0,0))\cap\mathbb R^{n+1}_+
\]
therefore satisfies the hypotheses of
\cite[Proposition~2.2]{ruland2015unique}.
After translation and rescaling, that proposition yields
\[
    U=0\quad\text{in }B_{2r}^+((x_0,0)).
\]
In particular,
\[
    U=0\quad\text{in }B_r(x_0)\times(0,r),
\]
since this cylinder is contained in the half-ball.

\medskip 

\noindent\textbf{Step 5:} \textit{Global vanishing and recovery of the trace.}

It remains to pass from local vanishing of $U$ to global vanishing of $g$.
From the preceding step, there exists a nonempty open set $\mathcal O\subset \mathbb R^{n+1}_+$ such that
\[
    U=0\quad\text{in }\mathcal O.
\]
We show that this forces $U\equiv0$ in the whole upper half-space ${\mathbb R}^{n+1}_+$.

To this end, let
\[
    \mathcal Z
    =
    \bigl\{(x,y)\in\mathbb R^{n+1}_+:
    \text{ there exists }\rho>0\text{ such that }
    U=0\text{ in }B_\rho(x,y)\bigr\}.
\]
Then $\mathcal Z$ is nonempty and open. We claim that $\mathcal Z$ is also
closed in $\mathbb R^{n+1}_+$. Let $(x_j,y_j)\in\mathcal Z$ and suppose
\[
    (x_j,y_j)\to (x_\ast,y_\ast)\in\mathbb R^{n+1}_+.
\]
Choose $\rho>0$ such that
\[
    B_{2\rho}(x_\ast,y_\ast)\Subset \mathbb R^{n+1}_+.
\]
Since $(x_j,y_j)\in\mathcal Z$, there exists $\rho_j>0$ such that
$U=0$ in $B_{\rho_j}(x_j,y_j)$. Reducing $\rho_j$ if necessary, and using
$(x_j,y_j)\in B_\rho(x_\ast,y_\ast)$ for $j$ large enough, this gives a nonempty open subset of
$B_{2\rho}(x_\ast,y_\ast)$ on which $U$ vanishes. 

Inside $B_{2\rho}(x_\ast,y_\ast)$, the weight $y^{1-2s}$ is bounded above
and below by positive constants, and is smooth. Hence $U$ solves a uniformly
elliptic equation with smooth coefficients in this ball:
\[
    \operatorname{div}(y^{1-2s}\nabla U)=0.
\]
By the usual interior unique continuation property for uniformly elliptic
equations with (bounded) smooth coefficients, vanishing in a nonempty open subset of
$B_{2\rho}(x_\ast,y_\ast)$ implies
\[
    U=0\quad\text{in }B_{2\rho}(x_\ast,y_\ast).
\]
In particular, $(x_\ast,y_\ast)\in\mathcal Z$. Thus $\mathcal Z$ is both
open and closed in the connected set $\mathbb R^{n+1}_+$. Since
$\mathcal Z\neq\emptyset$, we conclude
\[
    U=0\quad\text{in }\mathbb R^{n+1}_+.
\]

Finally, since $g\in L_s\subset L^1_{\mathrm{loc}}(\mathbb R^n)$,
the local approximate identity property
\eqref{eq: Lp trace of U is g}, with $p=1$, gives
\[
    P_y^s*g\to g
    \quad\text{in }L^1_{\mathrm{loc}}(\mathbb R^n).
\]
Since $P_y^s*g=U(\cdot,y)=0$ for every $y>0$, we conclude
that $g=0$ almost everywhere in $\mathbb R^n$.

\label{cor:UCP-Bessel-potential-spaces}
Let $0<s<1$, $r\in\mathbb R$, $1\leq p<\infty$,
and let $V\subset\mathbb R^n$ be a nonempty open set.
If $u\in H^{r,p}(\mathbb R^n)$ satisfies
\[
    u=0,\qquad (-\Delta)^s u=0
    \quad\text{in }\mathcal D'(V),
\]
then $u=0$ in $\mathbb R^n$.
<!-- end corollary-1 -->
<!-- proof-6: body_md -->
Let $\rho\in C_c^\infty(B_1)$ be a nonnegative, even
mollifier with integral one, and set
\[
    \rho_\varepsilon(x)=\varepsilon^{-n}\rho(x/\varepsilon),
    \qquad u_\varepsilon=\rho_\varepsilon*u.
\]
Writing $J=(1-\Delta)^{1/2}$ and $v=J^r u\in L^p$,
we have
\[
    u_\varepsilon=(J^{-r}\rho_\varepsilon)*v.
\]
Since $J^{-r}\rho_\varepsilon\in\mathcal S(\mathbb R^n)$,
Young's inequality gives $u_\varepsilon\in L^p\cap L^\infty$.
In particular, $u_\varepsilon\in L_s$.

Choose a nonempty open set $V'\Subset V$.
For $0<\varepsilon<\operatorname{dist}(\overline{V'},V^c)$,
local vanishing and commutation with mollification imply
\[
    u_\varepsilon=0,\qquad
    (-\Delta)^s u_\varepsilon
    =\rho_\varepsilon*((-\Delta)^s u)=0
    \quad\text{in }V'.
\]
Theorem~\ref{theorem:bounded-fractional-ucp} therefore gives
$u_\varepsilon=0$ in $\mathbb R^n$.
Passing to the limit as $\varepsilon\downarrow0$ in
$\mathcal D'(\mathbb R^n)$ yields $u=0$.

<!-- lemma-4: body_md -->
\label{lemma: exterior determination general}
Let $x_0\in W$ be a common Lebesgue point of $a_1$ and $a_2$. Assume that
\begin{equation}
    \label{eq:equal-DN-exterior-det}\Lambda^s_{\gamma_1}f|_W=\Lambda^s_{\gamma_2}f|_W
    \qquad\text{for all }f\in C_c^\infty(W).
\end{equation}
Then
\[
    \gamma_1(x_0)=\gamma_2(x_0).
\]
Consequently, $\gamma_1=\gamma_2$ a.e.~in $W$.
<!-- end lemma-4 -->
<!-- proof-7: body_md -->
Since the construction is very similar to the one of Lemma~\ref{lemma: exterior determination for n=1 s=1/2}, we limit ourselves to the main steps. Choose a bounded open set $W_0\Subset W$ such that $x_0\in W_0$ and
$\operatorname{dist}(W_0,\Omega)>0$. Let $\psi\in C_c^\infty(B_1(0))$, $\psi\not\equiv0$, and set
\[
    \psi_N(x)=A N^{(n-2s)/2}\psi(N(x-x_0)),\quad N>0,
\]
where $A>0$ is chosen so that
\[
    \frac{c_{n,s}}{2}
    \iint_{\mathbb R^n\times\mathbb R^n}
    \frac{|\psi_N(x)-\psi_N(y)|^2}{|x-y|^{n+2s}}\,dxdy=1.
\]
This normalization is independent of $N$ by scaling. Moreover,
\[
    \|\psi_N\|_{L^2(\mathbb R^n)}^2
    =
    A^2N^{-2s}\|\psi\|_{L^2(\mathbb R^n)}^2
    \to0.
\]
For $N>0$ sufficiently large, $\psi_N\in C_c^\infty(W_0)$.

Let $u_{j,N}\in H^s({\mathbb R}^n)$ be the unique solution to the fractional conductivity equation $L_{\gamma_j}u=0$ in $\Omega$ with exterior value $\psi_N$. By the exterior
energy estimate from \cite[Lemma~3.1]{RZ2022LowReg},
\[
    \|u_{j,N}-\psi_N\|_{H^s(\mathbb R^n)}
    \leq C\|\psi_N\|_{L^2(W_0)}
    \to 0,\quad \text{as }N\to\infty.
\]
The computation \eqref{eq:DN-decomp} shows that
\[
    \langle \Lambda^s_{\gamma_j}\psi_N,\psi_N\rangle
    =
    B^s_{\gamma_j}(\psi_N,\psi_N)+o(1)\quad \text{as }N\to\infty.
\]
As in the one dimensional case, we claim that
\[
    B^s_{\gamma_j}(\psi_N,\psi_N)\to \gamma_j(x_0).
\]
Fix $j$ and write $a=a_j$, $a_0=a(x_0)$. Our starting point is again
\[
\begin{split}
    |B^s_{\gamma_j}(\psi_N,\psi_N)-a_0^2|
    &\leq
    C\int_{\mathbb R^n}|a(z)-a_0|K_N(z)\,dz,
\end{split}
\]
which follows from the estimate \eqref{eq:product-estimate}. Here we put
\[
    K_N(z):=
    \int_{\mathbb R^n}
    \frac{|\psi_N(x)-\psi_N(z)|^2}{|x-z|^{n+2s}}\,dx .
\]
By scaling, we get
\[
    K_N(z)=N^nK(N(z-x_0)),\quad \text{with}\quad K(Z)\vcentcolon =
    A^2\int_{\mathbb R^n}
    \frac{|\psi(X)-\psi(Z)|^2}{|X-Z|^{n+2s}}\,dX .
\]
Moreover $K\in L^1(\mathbb R^n)$, since $\psi\in C_c^\infty(\mathbb R^n)
\subset H^s(\mathbb R^n)$, and there holds
\[
    \int_{\mathbb R^n}K(Z)\,dZ=\frac{2}{c_{n,s}}.
\]
After changing variables, we have
\[
    \int_{\mathbb R^n}|a(z)-a_0|K_N(z)\,dz
    =
    \int_{\mathbb R^n}|a(x_0+Z/N)-a_0|K(Z)\,dZ.
\]
For fixed $R,M>0$, set $\widehat{K}_M=\min\{K,M\}$. Then
\[
\begin{aligned}
    &\int_{\mathbb R^n}|a(x_0+Z/N)-a_0|K(Z)\,dZ\\
    &\quad = \int_{B_R}|a(x_0+Z/N)-a_0|K(Z)\,dZ+\int_{{\mathbb R}^n\setminus B_R}|a(x_0+Z/N)-a_0|K(Z)\,dZ\\
    &\quad\leq
    M N^n\int_{B_{R/N}(x_0)}|a(z)-a_0|\,dz+2\|a\|_{L^\infty}\int_{B_R}(K-\widehat K_M)\,dZ\\
    &\qquad+
    2\|a\|_{L^\infty}
    \int_{\mathbb R^n\setminus B_R}K\,dZ.
\end{aligned}
\]
For fixed $R,M$, the first term tends to zero, since $x_0$ is a Lebesgue point of $a$ and therefore
\[
    N^n\int_{B_{R/N}(x_0)}|a(z)-a_0|\,dz
    =
    |B_R|\fint_{B_{R/N}(x_0)}|a(z)-a_0|\,dz
    \to 0,
\]
as $N\to\infty$. Taking the upper limit as $N\to\infty$, then letting
$M\to\infty$ and finally $R\to\infty$, we obtain
\[
    \int_{\mathbb R^n}|a(z)-a_0|K_N(z)\,dz\longrightarrow0,
\]
because $K\in L^1(\mathbb R^n)$.

Consequently,
\[
    B^s_{\gamma_j}(\psi_N,\psi_N)
    \to a_j(x_0)^2=\gamma_j(x_0),
    \qquad j=1,2.
\]
Combining this convergence with the DN energy decomposition
and \eqref{eq:equal-DN-exterior-det}, we obtain
\[
\begin{aligned}
    \gamma_1(x_0)
    &=\lim_{N\to\infty}
      \langle\Lambda^s_{\gamma_1}\psi_N,\psi_N\rangle\\
    &=\lim_{N\to\infty}
      \langle\Lambda^s_{\gamma_2}\psi_N,\psi_N\rangle
      =\gamma_2(x_0).
\end{aligned}
\]
Since almost every point of $W$ is a common Lebesgue point
of $a_1$ and $a_2$, it follows that
$\gamma_1=\gamma_2$ almost everywhere in $W$.

<!-- lemma-5: body_md -->
\label{lemma: equality weighted solutions general}
Assume that $a_1=a_2$ a.e.~in $W$. Let $V\Subset W$ be a nonempty open
set and assume that the exterior value $f\in C_c^\infty(W)$ vanishes in a neighborhood of $V$. Suppose that
\begin{equation}
\label{eq: equality of DNs weighted nd}
    \Lambda^s_{\gamma_1}f |_W=\Lambda^s_{\gamma_2}f|_W.
\end{equation}
For $j=1,2$, let $u_j^f\in H^{s}({\mathbb R}^n)$ be the unique solution to the Dirichlet problem
\begin{equation}
\begin{cases}
\label{eq:Dirichlet-frac-cond-nd}
    L_{\gamma_j} u=0 &\text{in }\Omega,\\
    u=f &\text{in }\Omega_e,
\end{cases}
\end{equation}
then
\[
    a_1u_1^f=a_2u_2^f
    \qquad\text{a.e. in }\mathbb R^n .
\]
<!-- end lemma-5 -->
<!-- proof-8: body_md -->
Again we focus on the main differences to the one-dimensional case, established in Lemma~\ref{lemma: equality of weighted solutions}. Throughout the proof, we set
\[
    h=a_1u_1^f-a_2u_2^f\in L^2({\mathbb R}^n)\subset L_s.
\]
Since $u_j^f=f$ in $\Omega_e$, $f=0$ in a neighborhood of $\overline V$, and $a_1=a_2$ in $W$,
we have
\[
    h=0\quad\text{a.e. in }V.
\]
Since $h=0$ in $V$ and $h\in L_s$, \cite[Proposition~2.1.4]{silvestrephdthesis} demonstrates that $(-\Delta)^{s}$ is a continuous function in $V$ and it is given by the integral
\[
    (-\Delta)^s h(x)
    =
    -c_{n,s}\int_{\mathbb R^n \setminus V}
    \frac{h(y)}{|x-y|^{n+2s}}\,dy 
\]
for $x\in V$. Consequently, $(-\Delta)^s h\in L^1_{\rm loc}(V)$.

Let $a$ denote the common restriction of $a_1$ and $a_2$ to $V$. Since the derivation of formula \eqref{eq: integral identity} is independent of the dimension $n$ and fractional exponent $s$, we again infer from \eqref{eq: equality of DNs weighted nd} that 
\[
    \int_V a(x)\varphi(x)
    \int_{\mathbb R^n \setminus V}
    \frac{h(y)}{|x-y|^{n+2s}}\,dy\,dx=0
\]
for every $\varphi\in C_c^\infty(V)$. Since $a\geq \lambda_0^{1/2}>0$ and the inner integral belongs to $L^1_{\operatorname{loc}}(V)$, we obtain
\[
    (-\Delta)^s h=0\quad\text{in }\mathscr{D}^{\prime}(V).
\]
Together with $h=0$ in $V$, the unique continuation property for the
fractional Laplacian (Theorem~\ref{thm:UCP-energy-spaces} with $r=0$) gives
\[
    h=0\quad\text{a.e. in }\mathbb R^n.
\]
Thus $a_1u_1^f=a_2u_2^f$ a.e.~in $\mathbb R^n$.

<!-- theorem-5: body_md -->
\label{thm: uniqueness Linfty general}
Let $0<s<1$ and $n\geq 1$. Let $\Omega\subset\mathbb R^n$ be open bounded set, and assume that $W\subset\Omega_e$ is a nonempty open set. Let $\gamma_1,\gamma_2\in L^\infty(\mathbb R^n)$ be uniformly elliptic. If
\[
    \Lambda^s_{\gamma_1}f|_W=\Lambda^s_{\gamma_2}f|_W
    \qquad\text{for all }f\in C_c^\infty(W),
\]
then
\[
    \gamma_1=\gamma_2
    \qquad\text{a.e. in }\mathbb R^n.
\]
<!-- end theorem-5 -->
<!-- proof-9: body_md -->
By exterior determination (Lemma~\ref{lemma: exterior determination general}), we know that $\gamma_1=\gamma_2$ a.e.~in
$W$. Next, choose nonempty open sets $U,V\Subset W$ with $\overline U\cap\overline V=\emptyset$, and let the exterior condition
$f\in C_c^\infty(W)$ satisfy
\[
    f\geq c_f>0 \quad\text{in }U,
    \qquad
    f=0 \quad\text{in a neighborhood of }\overline V.
\]
By Lemma~\ref{lemma: equality weighted solutions general}, we have
\[
    w:=a_1u_1^f=a_2u_2^f
    \qquad\text{a.e. in }\mathbb R^n.
\]
In particular, since $u_j^f=f$ in $U\subset\Omega_e$,
\[
    w=a_1f=a_2f\geq \lambda_0^{1/2}c_f>0
    \qquad\text{a.e. in }U.
\]
The rest of the proof of Theorem~\ref{thm:uniqueness-1d} is independent of the dimension $n$ or the fractional exponent $s$ and therefore we only point out the main points. Set
\[
    \delta_a:=a_1-a_2.
\]
Let $\varphi\in C_c^\infty(U)$. Since
\[
    a_j(x)a_j(y)(u_j^f(x)-u_j^f(y))
    =
    a_j(y)w(x)-a_j(x)w(y),
\]
the equality of the DN maps gives
\[
    0=
    \iint_{\mathbb R^n\times\mathbb R^n}
    \frac{(\delta_a(y)w(x)-\delta_a(x)w(y))(\varphi(x)-\varphi(y))}
    {|x-y|^{n+2s}}\,dxdy .
\]
Because $\delta_a=0$ a.e. in $U$ and $\operatorname{supp}\varphi\subset U$, this reduces,
by symmetry, to
\[
    \int_U \varphi(x)w(x)
    \int_{\mathbb R^n \setminus U}
    \frac{\delta_a(y)}{|x-y|^{n+2s}}\,dy\,dx=0.
\]
The inner integral belongs to $L^1_{\operatorname{loc}}(U)$, by \cite[Proposition~2.1.4]{silvestrephdthesis}. Because
$w\geq \lambda_0^{1/2}c_f>0$ a.e. in $U$, we get
\[
    \int_{\mathbb R^n \setminus U}
    \frac{\delta_a(y)}{|x-y|^{n+2s}}\,dy=0
    \qquad\text{for a.e. }x\in U.
\]
Equivalently,
\[
    (-\Delta)^s\delta_a=0\quad\text{in }\mathscr{D}^{\prime}(U),
    \qquad
    \delta_a=0\quad\text{a.e. in }U.
\]
Since $\delta_a\in L^\infty(\mathbb R^n)$, Theorem~\ref{theorem:bounded-fractional-ucp}
implies
\[
    \delta_a=0\quad\text{a.e. in }\mathbb R^n.
\]
Thus $a_1=a_2$ a.e. in $\mathbb R^n$, and consequently
\[
    \gamma_1=\gamma_2\quad\text{a.e. in }\mathbb R^n.
\]
This completes the proof.
