\label{thm:1}
Let $u=u(t,x)$ be the solution to \eqref{eq:1}. Suppose $2n/(n-1)\leq p<\infty$ and $f$ is a radial function in $\mathbb{H}^n$. Let $\gamma<1/p$. Then for $T\geq 0$
\begin{align}\label{eq:4.1}
\left(\int_T^{T+1}\|u(t,\cdot)\|^p_p\mathrm{d} t\right)^{\frac1p}\leq C_{p,\gamma}\, \mathrm{e}^{(n-1)\left(\frac12-\frac1p\right)T}\left\|(-\Delta_{\mathbb{H}^n})^{(\alpha(p,n)-\gamma)/2}f\right\|_p 
\end{align}
with $\alpha(p,n)$ as in \eqref{e1.1}.
Moreover, the exponential part $\mathrm{e}^{(n-1)(1/2-1/p)T}$ of the bound is optimal.

<!-- lemma-2: body_md -->
\label{lem:3.1}
\begin{asparaenum}[\rm (i)]
\item\label{lem:3.1(i)} For $0<r\leq 1$,
\begin{align*}
\varphi_s(r)=(\sinh r)^{-\frac{n-1}2}\left(\mathrm{e}^{\mathrm{i} sr}a(s,r)+\mathrm{e}^{-i sr}a(-s,r)\right),
\end{align*}
where $a(\cdot,r)\inS^{-(n-1)/2}$ uniformly in $0<r\leq 1$.
\item\label{lem:3.1(ii)} For $r\geq 1/2$,
\begin{align*}
\varphi_s(r)=(\sinh r)^{-\frac{n-1}2}\left(\mathrm{e}^{\mathrm{i} sr}\mathbf{c}(s)\,b(s,r)+\mathrm{e}^{-\mathrm{i} sr}\mathbf{c}(-s)\,b(-s,r)\right)\!,
\end{align*}
where $b(\cdot,r)\in S^0$ uniformly in $r\geq 1/2$. In particular, if $s\geq 2$, then
\begin{align*}
\varphi_s(r)&=\frac{2^{\frac{n-1}2}}{\sqrt{\uppi}}\Gamma\!\left(\frac n2\right)(\sinh r)^{-\frac{n-1}2}s^{-\frac{n-1}2}\cos\!\left(sr-\frac{(n-1)\uppi}4\right)\\
&\qquad+(\sinh r)^{-\frac{n-1}2}\big(\mathrm{e}^{\mathrm{i} sr}a(s,r)+\mathrm{e}^{-\mathrm{i} sr}a(-s,r)\big),
\end{align*}
where $a(\cdot,r)\inS^{-(n+1)/2}$ uniformly in $r\geq 1/2$.
\end{asparaenum}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
To show \eqref{lem:3.1(i)} we shall use the following asymptotic expansion \cite[Theorem2.1]{ST78} for $0<r\leq 1$:
\begin{align}\label{eq:7.0}
\varphi_s(r)&=\frac{\Gamma\!\left(\frac n2\right)}{\Gamma\!\left(\frac{n-1}2\right)}(\sinh r)^{-\frac{n-1}2}\sum_{\ell=0}^\infty 2^{\frac{n-2}2+\ell}\Gamma\!\left(\frac{n-1}2+\ell\right)c_\ell(r)\\
&\hphantom{=\frac{\Gamma(\frac n2)}{\Gamma(\frac{n-1}2)}(\sinh r)^{-\frac{n-1}2}\sum_{\ell=0}^\infty}\ \ \ \ \times r^{\frac{n-1}2+2\ell}(sr)^{-\frac{n-2}2-\ell}J_{\frac{n-2}2+\ell}(sr),\notag
\end{align}
where $J_\mu$ is the Bessel function of order $\mu$, and the coefficients $c_\ell$ satisfy
\begin{align*}
c_0\equiv 1,\quad |c_\ell(t)|\leq CR^{-\ell}\ (\ell\in\mathbb{N}^*)
\end{align*}
for some constant $R>1$. It's well-known that the Bessel function satisfies the following asymptotic expansion:
\begin{align*}
\rho^{-\mu}J_\mu(\rho)=\mathrm{e}^{\mathrm{i}\rho}\psi_{-\mu-1/2}(\rho)+\mathrm{e}^{-\mathrm{i}\rho}\psi_{-\mu-1/2}(-\rho)\quad\text{for }\rho>0,
\end{align*}
where $\psi_{-\mu-1/2}\inS^{-\mu-1/2}$. Since for $\ell\in\mathbb{N}$,
\begin{align*}
r^{\frac{n-1}2+2\ell}\psi_{-\frac{n-1}2-\ell}(r\cdot)\inS^{-\frac{n-1}2-\ell}\quad\text{uniformly in }0<r\leq 1,
\end{align*}
in view of \eqref{eq:7.0}, for $r$ as above the spherical function $\varphi_s(r)$ satisfies the desired asymptotic expansion.\smallskip

The first assertion in \eqref{lem:3.1(ii)} follows readily from the Harish-Chandra expansion \cite[(8) and (10)]{APV12}:
\begin{align}\label{eq:7.01}
\varphi_s(r)=(2\sinh r)^{-\frac{n-1}2}\left(\mathrm{e}^{\mathrm{i} sr}\mathbf{c}(s)\sum_{\ell=0}^\infty\frac{\gamma_\ell(s)}{\mathrm{e}^{2\ell r}}+\mathrm{e}^{-\mathrm{i} sr}\mathbf{c}(-s)\sum_{\ell=0}^\infty\frac{\gamma_\ell(-s)}{\mathrm{e}^{2\ell r}}\right)\!,
\end{align}
where the coefficients $\gamma_\ell$ satisfy
\begin{align}\label{eq:3-3.3}
\gamma_0\equiv 1,\quad|\partial^j_s\gamma_\ell(s)|\leq C_j\ell^\nu (1+|s|)^{-j-1}\ (\ell\in\mathbb{N}^*,\ j\in\mathbb{N},\ s\in\mathbb{R}^*)
\end{align}
for some constant $\nu>0$. The last assertion of \eqref{lem:3.1(ii)} is indeed \cite[Proposition 3.1]{WY25}.

\label{prop:3.1}
 Suppose $2\leq p<2n/(n-1)$ and $f$ is a radial function in $\mathbb{H}^n$, and $T>0$. Then for any $\delta>0$
\begin{align}\label{eq:3-1}
\left(\int_T^{T+1}\left\|(-\Delta_{\mathbb{H}^n})^{-\frac\delta2}\cos(tD)(f)\right\|^p_p\mathrm{d} t\right)^{\frac1p}\lesssim_{p,\delta} \mathrm{e}^{(n-1)\big(\frac12-\frac1p\big)T}\|f\|_p.
\end{align}

<!-- proof-3: body_md -->
By Proposition~\ref{prop2.3}, it suffices to show that for every $0<\delta<1/2$,
\begin{align}\label{eq:3-1r}
\left(\int_T^{T+1}\left\|\left(-\Delta_{\mathbb{H}^n}+\frac{1}{10}\operatorname{Id}\right)^{-\frac\delta2}\cos(tD)(f)\right\|^p_p\mathrm{d} t\right)^{\frac1p}\lesssim_{p,\delta} \mathrm{e}^{(n-1)\big(\frac12-\frac1p\big)T}\|f\|_p.
\end{align}

We define
\begin{align*}
k_{\delta,t}(r):=\mathscr{F}^{-1}\!\left[\left((\cdot)^2+\frac{(n-1)^2}4+\frac1{10}\right)^{-\frac\delta2}\cos(t\cdot)\right](r)
\end{align*}
such that
\begin{align*}
\left(-\Delta_{\mathbb{H}^n}+\frac{1}{10} \operatorname{Id} \right)^{-\frac\delta2}\cos(tD)(f)(x)=\int_{\mathbb{H}^n}f(y)\,k_{\delta,t}(d(x,y))\,\mathrm{d} y,
\end{align*}
and decompose the kernel $k_{\delta,t}(d(x,y))$ into $K_{\delta,t}(x,y)+E_{\delta,t}(x,y)$, where
\begin{align*}
K_{\delta,t}(x,y)=\1_{\{y\in\mathbb{H}^n:|R(x)-R(y)|\leq t+1\}}(y)\,k_{\delta,t}(d(x,y)).
\end{align*}
We further define the operator
\begin{align*}
\mathscr{K}_{\delta,t}(f)(x):=\int_{\mathbb{H}^n}f(y)\,K_{\delta,t}(x,y)\,\mathrm{d} y,\quad\mathscr{E}_{\delta,t}(f)(x):=\int_{\mathbb{H}^n}f(y)\,E_{\delta,t}(x,y)\,\mathrm{d} y.
\end{align*}

We first deal with the operator $\mathscr{K}_{\delta,t}$. Since $f$ is radial, we write $|f(x)|=h(R(x))$ for some function $h$ defined on $[0,\infty)$. We also set $f_0(r):=(\sinh r)^{\frac{n-1}p}h(r)$ such that
\begin{align}
\|f_0\|_{L^p(\mathbb{R}_+)}=C\|f\|_p.
\end{align}
By \eqref{eq:2-2} and \eqref{eq:2-3}, we write in the Cartan coordinates
\begin{align*}
\mathscr{K}_{\delta,t}(f)(x)&=C\int_0^\infty f(r)\,\1_{\{r\in\mathbb{R}_+:|R(x)-r|\leq t+1\}}(r)\,(\sinh r)^{n-1}\\
&\hphantom{=C\int_0^\infty}\ \ \times\left(\int_{\mathbb{K}}k_{\delta,t}(d(\mathfrak{a}_{R(x)},\mathfrak{k}\mathfrak{a}_r))\,\mathrm{d}\mathfrak{k}\right)\mathrm{d} r.
\end{align*}
Therefore, in view of \eqref{eq:2-4}, \eqref{eq:11} and \eqref{eq:11.2},
\begin{align}\label{eq:3-2}
\|\mathscr{K}_{\delta,t}(f)\|_p^p\lesssim\int_0^\infty\left(\int_0^\infty f_0(r)\,\kappa_{\delta,t}(R,r)\,\mathrm{d} r\right)^p\mathrm{d} R,
\end{align}
where
\begin{align*}
\kappa_{\delta,t}(R,r)&=\1_{\{(R,r)\in(\mathbb{R}_+)^2:|R-r|\leq t+1\}}(R,r)\,(\sinh r)^{(n-1)\left(1-\frac1p\right)}(\sinh R)^{\frac{n-1}p}\\
&\qquad\times\left|\vphantom{\int_0^\infty \left(s^2+\frac{(n-1)^2}4\right)^{-\frac\delta2}\cos(st)\,\varphi_s(R)\,\varphi_s(r)\,|\mathbf{c}(s)|^{-2}\mathrm{d} s}\right.\underbrace{\int_0^\infty \left(s^2+\frac{(n-1)^2}4+\frac1{10}\right)^{-\frac\delta2}\cos(st)\,\varphi_s(R)\,\varphi_s(r)\,|\mathbf{c}(s)|^{-2}\mathrm{d} s}_{\mathbf{I}(R,r,t)}\left.\vphantom{\int_0^\infty \left(s^2+\frac{(n-1)^2}4+\frac1{10}\right)^{-\frac\delta2}\cos(st)\,\varphi_s(R)\,\varphi_s(r)\,|\mathbf{c}(s)|^{-2}\mathrm{d} s}\right|.
\end{align*}
In view of Lemmas \ref{lem:3.0} and \ref{lem:3.1},
\begin{align*}
\mathbf{I}(R,r,t)&=(\sinh r\cdot \sinh R)^{-\frac{n-1}2}\sum_{\iota_1,\iota_2\in\{-1,1\}}\int_{\mathbb{R}}\mathrm{e}^{\mathrm{i} s(t+\iota_1R+\iota_2r)}a_{\delta,\iota_1,\iota_2}(s,R,r)\,\mathrm{d} s,
\end{align*}
where $a_{\delta,\pm1,\pm1}(\cdot,R,r)\inS^{-\delta}$ uniformly in $R,r>0$. By the classical estimates on Fourier transform of symbols \cite[p. 241]{Ste93},
\begin{align*}
|\mathbf{I}(R,r,t)|\lesssim (\sinh r)^{-\frac{n-1}2}(\sinh R)^{-\frac{n-1}2}\sum_{\iota_1,\iota_2\in\{-1,1\}}A_\delta(|t+\iota_1R+\iota_2r|)
\end{align*}
for
\begin{align*}
A_\delta(\tau)=\tau^{\delta-1}(1+|\tau|)^{-100},
\end{align*}
whence
\begin{align*}
\kappa_{\delta,t}(R,r)&\lesssim\1_{\{(R,r)\in(\mathbb{R}_+)^2:|R-r|\leq t+1\}}(R,r)\,\left(\frac{\sinh r}{\sinh R}\right)^{(n-1)\left(\frac12-\frac1p\right)}\\
&\qquad\times\sum_{\iota_1,\iota_2\in\{-1,1\}}A_\delta(|t+\iota_1R+\iota_2r|),
\end{align*}
Now we integrate on both sides of \eqref{eq:3-2} with respect to $t$ over $[T,T+1]$ to get
\begin{align}
&\int_T^{T+1}\|\mathscr{K}_{\delta,t}(f)\|_p^p\mathrm{d} t\notag\\
&\quad\lesssim \mathrm{e}^{(n-1)\left(\frac p2-1\right)T}\int_0^1R^{(n-1)\left(1-\frac p2\right)}\int_T^{T+1}\left(\int_0^\infty f_0(r)\,\widetilde{\kappa}_{\delta,R}(t,r)\,\mathrm{d} r\right)^p\mathrm{d} t\,\mathrm{d} R\label{eq:19}\\
&\quad\qquad+\int_T^{T+1}\int_1^\infty\left(\int_0^\infty f_0(r)\,\kappa_{\delta,t}(R,r)\,\mathrm{d} r\right)^p\,\mathrm{d} R\,\mathrm{d} t\notag
\end{align}
provided $p\geq 2$, where
\begin{align*}
\widetilde{\kappa}_{\delta,R}(t,r):=\1_{[0,R+t+1]}(r)\sum_{\iota_1,\iota_2\in\{-1,1\}}A_\delta(|t+\iota_1R+\iota_2r|).
\end{align*}

On the one hand, for $R\in(0,1)$ and $\delta>0$, 
\begin{align*}
\sup_{t\in[T,T+1]}\int_0^\infty \widetilde{\kappa}_{\delta,R}(t,r)\,\mathrm{d} r+\sup_{r\in(0,\infty)}\int_T^{T+1}\widetilde{\kappa}_{\delta,R}(t,r)\,\mathrm{d} t\leq C_\delta.
\end{align*}
On the other hand, for $\delta$ as above, $t\in[T,T+1]$ and $p\geq 2$,
\begin{align*}
\sup_{R\in[1,\infty)}\int_0^\infty \kappa_{\delta,t}(R,r)\,\mathrm{d} r+\sup_{r\in(0,\infty)}\int_1^\infty \kappa_{\delta,t}(R,r)\,\mathrm{d} R\lesssim_\delta \mathrm{e}^{(n-1)\left(\frac12-\frac1p\right)T}.
\end{align*}
By Schur's test, we conclude that for all $\delta,p$ as above and $1\leq q\leq \infty$,
\begin{align*}
\sup_{R\in(0,1)}\int_T^{T+1}\left(\int_0^\infty f_0(r)\,\widetilde{\kappa}_{\delta,R}(t,r)\,\mathrm{d} r\right)^q\mathrm{d} t\lesssim_\delta \|f_0\|_{L^q(\mathbb{R}_+)}^q
\end{align*}
and
\begin{align*}
\sup_{t\in[T,T+1]}\int_1^\infty\left(\int_0^\infty f_0(r)\,\kappa_{\delta,t}(R,r)\,\mathrm{d} r\right)^q\,\mathrm{d} R\lesssim_\delta \mathrm{e}^{q(n-1)\left(\frac12-\frac1p\right)T}\|f_0\|_{L^q(\mathbb{R}_+)}^q.
\end{align*}
Therefore, according to \eqref{eq:19},
\begin{align}\label{eq:3-13}
\left(\int_T^{T+1}\|\mathscr{K}_{\delta,t}(f)\|_p^p\mathrm{d} t\right)^{\frac1p}\lesssim_{p,\delta} \mathrm{e}^{(n-1)\left(\frac12-\frac1p\right)T}\|f_0\|_{L^p(\mathbb{R}_+)}\sim\mathrm{e}^{(n-1)\left(\frac12-\frac1p\right)T} \|f\|_p
\end{align}
provided $2\leq p<2n/(n-1)$ and $\delta>0$.\medskip

It remains to handle the error term $\mathscr{E}_{\delta,t}$. Indeed, on the one hand, note that the triangle inequality implies that $E_{\delta,t}$ is supported in $\{(x,y):d(x,y)>t+1\}$. Hence by the argument proving \cite[(3.9)]{Ion00},
\begin{align*}
|E_{\delta,t}(x,y)|\lesssim e_{\delta,t}(d(x,y))
\end{align*}
for
\begin{align*}
e_{\delta,t}(r)=\1_{(t+1,\infty)}(r)\,\mathrm{e}^{\frac{n-1}2t}\mathrm{e}^{-(n-1)r}(r-t)^{-2},
\end{align*}
whence
\begin{align*}
\|\mathscr{E}_{\delta,t}(f)(x)\|_{L^\infty_{x,t}(\mathbb{H}^n\times[T,T+1])}\leq \|e_{\delta,t}\|_1\|f\|_\infty\lesssim \mathrm{e}^{\frac{n-1}2T}\|f\|_\infty
\end{align*}
in view of Schur's test and \eqref{eq:2-4}. On the other hand, according to \eqref{eq:9.1} and \eqref{eq:3-13} with $p=2$,
\begin{align*}
&\|\mathscr{E}_{\delta,t}(f)(x)\|_{L^2_{x,t}(\mathbb{H}^n\times[T,T+1])}\\
&\ \ \leq \sup_{t\in[T,T+1]}\left\|\left(-\Delta_{\mathbb{H}^n}+\frac{1}{10}\operatorname{Id}\right)^{-\frac\delta2}\cos(tD)(f)\right\|_2+\left(\int_T^{T+1}\|\mathscr{K}_{\delta,t}(f)\|_2^2\mathrm{d} t\right)^{\frac12}\lesssim \|f\|_2.
\end{align*}
Therefore, by interpolation, for all $2\leq p\leq \infty$,
\begin{align*}
\left(\int_T^{T+1}\|\mathscr{E}_{\delta,t}(f)\|^p_p\mathrm{d} t\right)^{\frac1p}\lesssim \mathrm{e}^{(n-1)\left(\frac12-\frac1p\right)T}\|f\|_p.
\end{align*}
This, combined with \eqref{eq:3-13}, yields the desired estimate \eqref{eq:3-1r}. The proof of Proposition~\ref{prop:3.1} is complete.

Recall that the solution to \eqref{eq:1} can be written as $u(t,x)=\cos(tD)(f)(x)$. We shall firstly use Theorem \ref{thmA:A} and Proposition \ref{prop:3.1} to establish the estimate
\begin{align}\label{eq:21}
\int_T^{T+1}\left\|(-\Delta_{\mathbb{H}^n})^{-\frac\alpha2}\cos(tD)(f)\right\|_p^p\mathrm{d} t\lesssim_{\alpha,p}\mathrm{e}^{p(n-1)\big(\frac12-\frac1p\big)T}\|f\|_p^p
\end{align}
for $p\to \infty$ and $p\to (2n/(n-1))^-$, respectively, and then use interpolation. In the sequel we fix $\eta\in\mathbb{R}$.

On the one hand, for $10000\leq p_1<\infty$ we have
\begin{align*}
&\int_T^{T+1}\left\|(-\Delta_{\mathbb{H}^n})^{-\frac{n-1}2\left(\frac12-\frac1{p_1}\right)-\frac{\mathrm{i}\eta}2}\cos(tD)(f)\right\|_{p_1}^{p_1}\mathrm{d} t\\
&\qquad\lesssim_{p_1} \mathrm{e}^{(n-1)\big(\frac12-\frac1{p_1}\big)T}\left\|(-\Delta_{\mathbb{H}^n})^{-\frac{\mathrm{i}\eta}2}f\right\|_{p_1}^{p_1}\leq C_\eta \mathrm{e}^{p_1(n-1)\big(\frac12-\frac1{p_1}\big)T}\|f\|_{p_1}^{p_1}
\end{align*}
with $C_\eta\lesssim\mathrm{e}^{c\eta^2}$, where the first inequality follows from Theorem \ref{thmA:A} and the last inequality follows from Proposition ~\ref{prop2.3}. Hence \eqref{eq:21} holds with $p=p_1$, $\operatorname{Re}\alpha=(n-1)(1/2-1/p_1)$ and the implicit constant of at most $\mathrm{e}^{c(\operatorname{Im}\alpha)^2}$ growth on $\operatorname{Im}\alpha$.

On the other hand, for $2\leq p_2<2n/(n-1)$ and $\delta>0$ we have
\begin{align*}
&\int_T^{T+1}\left\|(-\Delta_{\mathbb{H}^n})^{-\frac\delta2-\frac{\mathrm{i}\eta}2}\cos(tD)(f)\right\|_{p_2}^{p_2}\mathrm{d} t\\
&\ \ \lesssim_{p_2,\delta} \mathrm{e}^{(n-1)\big(\frac12-\frac1{p_2}\big)T}\left\|(-\Delta_{\mathbb{H}^n})^{-\frac{\mathrm{i}\eta}2}f\right\|_{p_2}^{p_2}\leq C_\eta\,\mathrm{e}^{p_2(n-1)\big(\frac12-\frac1{p_2}\big)T}\|f\|_{p_2}^{p_2}
\end{align*}
with $C_\eta\lesssim\mathrm{e}^{c\eta^2}$, where the first inequality follows from Proposition \ref{prop:3.1} and the last inequality follows from Proposition ~\ref{prop2.3}. Hence \eqref{eq:21} holds with $p=p_2$, $\operatorname{Re}\alpha=\delta$ and the implicit constant of at most $\mathrm{e}^{c(\operatorname{Im}\alpha)^2}$ growth on $\operatorname{Im}\alpha$.

Using the interpolation of analytic family of operators (see e.g. \cite{Sag69}), we obtain that \eqref{eq:21} holds for $p_2\leq p\leq p_1$ and
\begin{align*}
\alpha=(n-1)\left(\frac12-\frac1{p_1}\right)\frac{\frac1{p_2}-\frac1p}{\frac1{p_2}-\frac1{p_1}}+\delta\frac{\frac1{p_1}-\frac1p}{\frac1{p_1}-\frac1{p_2}}=(n-1)\left(\frac12-\frac1p\right)-\gamma,
\end{align*}
where
\begin{align*}
\gamma=(n-1)\left(\frac1{p_1}-\frac1p\right)+(n-1)\left(\frac12-\frac1{p_1}\right)\frac{\frac1p-\frac1{p_1}}{\frac1{p_2}-\frac1{p_1}}-\delta\frac{\frac1{p_1}-\frac1p}{\frac1{p_1}-\frac1{p_2}}<\frac1p.
\end{align*}
Consequently, \eqref{eq:21} holds with $2n/(n-1)\leq p<\infty$ and $\alpha=(n-1)(1/2-1/p)-\gamma$ with $\gamma<1/p$, as desired. This completes the proof of \eqref{eq:4.1}.


Now we show the optimality of the exponential part $\mathrm{e}^{(n-1)(1/p-1/2)T}$ of the bound in \eqref{eq:4.1} of Theorem~\ref{thm:1}. In particular, we shall show a somewhat stronger result as follows.
\begin{proposition}\label{prop:6}
For all $2\leq p<\infty$, $T\geq 0$ and $\alpha\in\mathbb{R}$, if
\begin{align}\label{eq:16.1}
\left(\int_T^{T+1}\|\cos(tD)(f)\|^p_p\mathrm{d} t\right)^{\frac1p}\leq C_p(T)\,\|(-\Delta)^\alpha f\|_p
\end{align}
holds for all radial functions $f$ on $\mathbb{H}^n$, then we must have
\begin{align*}
C_p(T)\gtrsim\mathrm{e}^{(n-1)\big(\frac12-\frac1p\big)T}.
\end{align*}
\end{proposition}
\begin{proof}
Instead of considering \eqref{eq:16.1}, we shall focus on another estimate
\begin{align}\label{eq:16.2}
\left(\int_T^{T+1}\left\|(-\Delta_{\mathbb{H}^n})^{\frac18}\cos(tD)(f)\right\|^p_p\mathrm{d} t\right)^{\frac1p}\leq C_p(T)\,\|(-\Delta_{\mathbb{H}^n})^{\alpha'} f\|_p\quad\text{for }\alpha'=\alpha+\frac18
\end{align}
equivalent to \eqref{eq:16.1}.

Let $c>0$ be a sufficiently small but fixed constant to be determined later, $h\in C^\infty_c(\mathbb{R}_+)$ such that $0\leq h\leq 1$ and
\begin{align*}
h(r)=\begin{cases}
1 & \text{on }\big[\frac12+5c,\frac12+6c\big],\smallskip\\
0 & \text{outside }[\frac12+4c,\frac12+7c].
\end{cases}
\end{align*}
We set ad radial function $f_T(x):=h(R(x)-T)$ on $\mathbb{H}^n$. Then it follows from the multiplier theorem (see e.g. \cite[Theorem 3]{CS74} or \cite[Theorem A]{Tay89}) that if we set $k:=\max\{0,\lceil\alpha'\rceil\}$, then
\begin{align}\label{eq:16.1.1}
\|(-\Delta_{\mathbb{H}^n})^{\alpha'}f_T\|_p\lesssim \|(-\Delta_{\mathbb{H}^n})^kf_T\|_p\lesssim \|h\|_{C^N}\|\mathbf{1}_{[T+\frac12+4c,T+\frac12+7c]}(R(\cdot))\|_p\lesssim \mathrm{e}^{\frac{n-1}pT},
\end{align}
where the second inequalities follows from the fact that
\begin{align*}
\Delta_{\mathbb{H}^n} f_T(x)=\frac{\partial^2h(R(\cdot)-T)}{\partial r^2}(R(x))+(n-1)\coth(R(x))\frac{\partial h(R(\cdot)-T)}{\partial r}(R(x)),
\end{align*}
see e.g. \cite[(3.85)]{Gri09}.

Next we need to bound
\begin{align*}
\left(\int_T^{T+1}\left\|(-\Delta_{\mathbb{H}^n})^{\frac18}\cos(tD)(f_T)\right\|^p_p\mathrm{d} t\right)^{\frac1p}
\end{align*}
from below. To do this, we denote by $k_t$ the convolution kernel of the operator $(-\Delta_{\mathbb{H}^n})^{1/8}\cos(tD)$ and let $\eta$ be an even smooth function such that
\begin{align*}
\eta\equiv \begin{cases}
1 & \text{outside }[-3,3],\\
0 & \text{on }[-2,2].
\end{cases}
\end{align*}
Then according to \eqref{eq:9.0}, Lemmas \ref{lem:3.0}\eqref{lem:3.0(iii)} and \ref{lem:3.1}\eqref{lem:3.1(ii)}, for $t\in[T+1/2,T+1/2+c]$,
\begin{align}
k_t(x)&=\frac{2^{n-3}\Gamma\!\left(\frac n2\right)}{\uppi^{\frac n2+1}}\int_0^\infty \left(s^2+\frac{(n-1)^2}4\right)^{\frac18}\cos(ts)\,\varphi_s(R(x))\,|\mathbf{c}(s)|^{-2}\mathrm{d} s\notag\\
&=C_n\,(\sinh R(x))^{-\frac{n-1}2}\!\left[\vphantom{\int_0^\infty \eta(s)\,s^{\frac n2-\frac14}\cos(ts)\cos\!\left(sR(x)-\frac{(n-1)\uppi}4\right)\mathrm{d} s}\right.\!\!\underbrace{\int_0^\infty \eta(s)\,s^{\frac n2-\frac14}\cos(ts)\cos\!\left(sR(x)-\frac{(n-1)\uppi}4\right)\mathrm{d} s}_{\widetilde{k}_t(x)}\label{eq:16.3}\\
&\hphantom{C_n\mathrm{e}^{-\frac{n-1}2R(x)}}\ \ +\underbrace{\int_0^\infty \cos(ts)\left(\mathrm{e}^{\mathrm{i} sR(x)}a(s,R(x))+\mathrm{e}^{-\mathrm{i} sR(x)}a(-s,R(x))\right)\mathrm{d} s}_{e_t(x)}\left.\vphantom{\int_0^\infty \cos(ts)\left(\mathrm{e}^{\mathrm{i} sR(x)}a(s,R(x))+\mathrm{e}^{-\mathrm{i} sR(x)}a(-s,R(x))\right)\mathrm{d} s}\!\!\right]\!,\notag
\end{align}
where $a(\cdot,r)\in S^{n/2-5/4}$ uniformly in $r\geq 1/2$.

To handle $\widetilde{k}_t$, for $r>0$ and $\gamma\notin2\mathbb{Z}+1$, in view of \cite[Theorem 2.4.6]{Gra14}, we have
\begin{align}\label{e6.001}
\int_0^\infty\cos(sr)\,s^\gamma\mathrm{d} s=\frac{\Gamma\!\left(\frac{\gamma+1}2\right)}{\Gamma\!\left(-\frac\gamma2\right)}2^\gamma\sqrt{\uppi}\,r^{-1-\gamma}.
\end{align}
Taking derivatives with respect to $r$ on both sides of \eqref{e6.001} and then replacing $\gamma +1$ by $\gamma$, in view of \cite[p. 3, (6) and (7)]{EMOT53}, we obtain that for $\gamma\notin2\mathbb{Z}$,
\begin{align*}
\int_0^\infty\sin(sr)\,s^\gamma\mathrm{d} s=-\frac{\Gamma\!\left(\frac{\gamma+1}2\right)}{\Gamma\!\left(-\frac\gamma2\right)}\cot\!\left(\frac{\gamma\uppi}2\right)2^\gamma\sqrt{\uppi}\,r^{-1-\gamma}.
\end{align*}
Since
\begin{align*}
&\cos\!\left(sr-\frac{(n-1)\uppi}4\right)\cos(ts)\\
&\quad=\frac12\!\left[\cos\!\left((r-t)s-\frac{(n-1)\uppi}4\right)+\cos\!\left((r+t)s-\frac{(n-1)\uppi}4\right)\right],
\end{align*}
we have
\begin{align*}

\end{align*}
for $t<r\leq t+10c$ with $c$ small enough. In addition,
\begin{align*}
\int_0^\infty (1-\eta(s))\,s^{\frac n2-\frac14}\cos(ts)\cos\!\left(sr-\frac{(n-1)\uppi}4\right)\mathrm{d} s=\mathrm{O}(1).
\end{align*}
Hence
\begin{align}\label{eq:16.4}
\widetilde{k}_t(x)=C\,(R(x)-t)^{-\frac n2-\frac34}+\mathrm{O}(1)
\end{align}
for $t\in[T+1/2,T+1/2+c]$ and $t<R(x)\leq t+10c$. On the other hand, by the classical estimates on Fourier transform of symbols \cite[p. 241]{Ste93}, for $c$ small enough and $t,x$ as above we have
\begin{align}\label{eq:16.5}
|e_t(x)|\lesssim (R(x)-t)^{\frac14-\frac n2}.
\end{align}

It follows from \eqref{eq:16.3}--\eqref{eq:16.5} that for $t\in[T+1/2,T+1/2+c]$ and $x\in B_c(\0)$ with $c$ sufficiently small,
\begin{align*}
&\!\!\!\!\left|(-\Delta_{\mathbb{H}^n})^{\frac18}\cos(tD)(f_T)(x)\right|\\
&=|f_T*k_t(x)|\\
&\geq |f_T*[(\sinh R(\cdot))^{-\frac{n-1}2}\widetilde{k}_t](x)|-|f_T*[(\sinh R(\cdot))^{-\frac{n-1}2}e_t](x)|\\
&\gtrsim \int_{B_{T+\frac12+6c}(\0)\backslash B_{T+\frac12+5c}(\0)}(\sinh d(x,y))^{-\frac{n-1}2}(d(x,y)-t)^{-\frac n2-\frac34}\mathrm{d} y\\
&\qquad -\int_{B_{T+\frac12+7c}(\0)\backslash B_{T+\frac12+4c}(\0)}(\sinh d(x,y))^{-\frac{n-1}2}\left(\mathrm{O}(1)+(d(x,y)-t)^{\frac14-\frac n2}\right)\mathrm{d} y\\
&\sim \mathrm{e}^{\frac{n-1}2T}c^{1-\frac n2-\frac34}\left(1-c\right)\gtrsim \mathrm{e}^{\frac{n-1}2T},
\end{align*}
whence
\begin{align*}
&\!\!\!\!\left(\int_T^{T+1}\left\|(-\Delta_{\mathbb{H}^n})^{\frac18}\cos(tD)(f_T)\right\|^p_p\mathrm{d} t\right)^{\frac1p}\\
&\geq \left(\int_{T+\frac12}^{T+\frac12+c}\left\|(-\Delta_{\mathbb{H}^n})^{\frac18}\cos(tD)(f_T)\right\|^p_{L^p(B_c(\0))}\mathrm{d} t\right)^{\frac1p}\gtrsim \mathrm{e}^{\frac{n-1}2T}.
\end{align*}
This combined with \eqref{eq:16.1.1} yields that if \eqref{eq:16.2} holds for all radial functions $f$, then we must have
\[
C_p(T)\gtrsim \mathrm{e}^{\frac{n-1}2T}\mathrm{e}^{-\frac{n-1}pT}=\mathrm{e}^{(n-1)\big(\frac12-\frac1p\big)T}.\qedhere
\]
Hence, the proof of Theorem~\ref{thm:1} is complete.
\end{proof}

\label{prop:5}
Suppose $2\leq p<3$ and $f$ is a radial function in $\mathbb{H}^3$. Then for $T\geq 0$
\begin{align*}
\left(\int_T^{T+1}\|\cos(tD)(f)\|^p_p\mathrm{d} t\right)^{\frac1p}\lesssim_p \mathrm{e}^{2\big(\frac12-\frac1p\big)T}\|f\|_p.\medskip
\end{align*} 
In particular, $u(t,\cdot)\in L^p(\mathbb{H}^3)$ for almost all times $t$, provided that $f\in L^p(\mathbb{H}^3)$ is a radial function and $2\leq p<3$.
<!-- end proposition-4 -->
<!-- proof-6: body_md -->
Since $f$ is radial, we write $f(x)=h(R(x))$ for some function $h$ defined on $[0,\infty)$ and extend the domain of $h$ from $[0,\infty)$ to $\mathbb{R}$ by setting $h\equiv 0$ on $(-\infty,0)$. We also set $f_0(r):=(\sinh r)^{\frac2p}h(r)$. In addition, letting $n=3$, \eqref{eq:7} and \eqref{eq:8} become
\begin{align}\label{eq:A1}
\varphi_s(r)=\frac1{\sinh r}\frac{\sin sr}s\quad\text{and}\quad\mathbf{c}(s)=-\frac\mathrm{i} s,
\end{align}
respectively. In view of \eqref{eq:2-4}, \eqref{eq:10} and \eqref{eq:11},
\begin{align}\label{eq:A12}
\|\cos(tD)(f)\|_p^p=\int_0^\infty\left|\int_0^\infty f_0(r)\,\kappa_t(R,r)\,\mathrm{d} r\right|^p\mathrm{d} R,
\end{align}
where
\begin{align}
\kappa_t(R,r)&=C\left(\frac{\sinh r}{\sinh R}\right)^{1-\frac2p}\int_0^\infty \cos st\,\sin sR\,\sin sr\,\mathrm{d} s\notag\\
&=C\left(\frac{\sinh r}{\sinh R}\right)^{1-\frac2p}\int_{\mathbb R}\big(\mathrm{e}^{\mathrm ist}+\mathrm{e}^{-\mathrm ist}\big)\big(\mathrm{e}^{\mathrm is(r+R)}-\mathrm{e}^{\mathrm is(r-R)}\big)\mathrm d s\notag\\
&=C\left(\frac{\sinh r}{\sinh R}\right)^{1-\frac2p}\big(\delta_0(t+r+R)+\delta_0(r+R-t)-\delta_0(t+r-R)\label{eq:A13}\\
&\hphantom{=C\left(\frac{\sinh r}{\sinh R}\right)^{1-\frac2p}}\ \ \quad -\delta_0(r-R-t)\big)\notag.
\end{align}

Finally, by substituting \eqref{eq:A13} into \eqref{eq:A12}, for $t>0$ we have
\begin{align}
\|\cos(tD)(f)\|_p^p&\lesssim\int_0^\infty(\sinh R)^{2-p}\left[|f_0(t-R)|^p(\sinh(t-R))^{p-2}\right.\label{eq:A14}\\
&\hphantom{\lesssim\int_0^\infty(\sinh R)^{2-p}[}\ \ \left.+|f_0(R-t)|^p(\sinh(R-t))^{p-2}\right.\notag\\
&\hphantom{\lesssim\int_0^\infty(\sinh R)^{2-p}[}\ \ \left.+|f_0(t+R)|^p(\sinh(t+R))^{p-2}\right]\mathrm{d} R.\medskip\notag
\end{align}

In view of \eqref{eq:A14}, we can write
\begin{align*}
\int_T^{T+1}\|\cos(tD)(f)\|^p_p\mathrm{d} t\lesssim \mathbf{J}_1+\mathbf{J}_2+\mathbf{J}_3,
\end{align*}
where
\begin{align*}
\mathbf{J}_1&=\int_T^{T+1}\int_0^t |f_0(t-R)|^p(\sinh(t-R))^{p-2}(\sinh R)^{2-p}\mathrm{d} R\,\mathrm{d} t\\
&\lesssim \mathrm{e}^{(p-2)T}\int_0^{T+1}|f_0(r)|^p\left(\int_{\max\{r,T\}}^{T+1}(\sinh(t-r))^{2-p}\mathrm{d} t\right)\mathrm{d} r\\
&\lesssim_p\mathrm{e}^{2p\left(\frac12-\frac1p\right)T}\|f_0\|_{L^p(\mathbb{R}_+)}^p=C\,\mathrm{e}^{2p\left(\frac12-\frac1p\right)T}\|f\|_p^p
\end{align*}
since $p<3$,
\begin{align*}
\mathbf{J}_2&=\int_T^{T+1}\int_t^\infty |f_0(R-t)|^p(\sinh(R-t))^{p-2}(\sinh R)^{2-p}\mathrm{d} R\,\mathrm{d} t\\
&\leq \int_T^{T+1}\int_0^\infty |f_0(r)|^p(\sinh(r+t))^{p-2}(\sinh(r+t))^{2-p}\mathrm{d} r\,\mathrm{d} t\\
&=\|f_0\|_{L^p(\mathbb{R}_+)}^p\lesssim \|f\|_p^p
\end{align*}
since $p\geq 2$, and
\begin{align*}
\mathbf{J}_3&=\int_T^{T+1}\int_0^\infty |f_0(t+R)|^p(\sinh(t+R))^{p-2}(\sinh R)^{2-p}\mathrm{d} R\,\mathrm{d} t\\
&\leq \mathrm{e}^{(p-2)T}\int_T^{T+1}\int_0^\infty |f_0(R+t)|^p\left(\frac{\mathrm{e}^R}{\sinh R}\right)^{p-2}\mathrm{d} R\,\mathrm{d} t\\
&\lesssim \mathrm{e}^{(p-2)T}\left(\int_0^1R^{2-p}\left(\int_{T+R}^{T+R+1}|f_0(r)|^p\mathrm{d} r\right)\mathrm{d} R+\int_T^{T+1}\int_t^\infty|f_0(r)|^p\mathrm{d} r\,\mathrm{d} t\right)\\
&\lesssim_p\mathrm{e}^{2p\left(\frac12-\frac1p\right)T}\|f_0\|_{L^p(\mathbb{R}_+)}^p\lesssim\mathrm{e}^{2p\left(\frac12-\frac1p\right)T}\|f\|_p^p
\end{align*}
since $p<3$. This yields the desired estimate. The proof of Proposition~\ref{prop:5} is complete.
