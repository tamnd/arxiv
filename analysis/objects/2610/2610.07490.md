<!-- corollary-1: body_md -->
\label{cor:energy}
For every fixed integer $m\geq2$,
$$
 t_{m,n}=2m-1-(1+o(1))\log_n \frac{(2m-1)^{m-1/2}}{(2m)^{m-1}}
 \qquad(n\to\infty).
$$
In particular,
$$
 t_{2,n}=3-(1+o(1))\log_n\frac{3\sqrt3}{4}.
$$

<!-- theorem-1: body_md -->
\label{thm:main}
For every fixed integer $m\geq2$,
$$
 \lim_{n\rightarrow\infty} 2m\log n\Big(\frac1{p_m}-\frac1{q_m(n)}\Big) = \log C_m.
$$
More precisely, for every $0<\varepsilon<1$, the following statements hold:
\begin{enumerate}[label=\textup{(\roman*)}]
\item For every $n$ sufficiently large in terms of $m,\varepsilon$ and every function $f: \mathbb{Z}\rightarrow\mathbb{C}$ with $|\mathrm{supp} f| \leq n$, we have  $\|\widehat{f}\|_{L^{2m}} \leq \|f\|_{\ell^q}$, where $q$ is defined by
$$
 \frac1q=\frac1{p_m}
 -\frac{(1-\varepsilon)\log C_m}{2m\log n}.
$$
\item For every $n$ sufficiently large in terms of $m,\varepsilon$, there exists a function $f:\mathbb{Z}\to[0,\infty)$ supported on exactly $n$ consecutive integers such that $\|\widehat{f}\|_{L^{2m}} > \|f\|_{\ell^q}$, where $q$ is defined by 
$$
 \frac1q=\frac1{p_m}
 -\frac{(1+\varepsilon)\log C_m}{2m\log n}.
$$
\end{enumerate}

\label{lem:rigidity}
Let $f\in\ell^p(\mathbb{Z})$. If
$$
 \|f^{*m}\|_{\ell^2}=\|f\|_{\ell^p}^m,
$$
then $f$ is supported on at most one point.

<!-- lemma-2: body_md -->
\label{lem:local}
Fix $m\geq2$ and $\lambda>0$. Let $n$ be sufficiently large in terms of $m,\lambda$, and define $q$ by
$$
 \frac1q=\frac1p-\frac\lambda{2m\log n}.
$$
Let $f: \mathbb{Z}\rightarrow \mathbb{C}$ be a function with $|\mathrm{supp} f| \leq n$, and write $f=f(0)\1_{\{0\}}+h$. Suppose that
$$
 \|h\|_{\ell^p}\leq\delta |f(0)|
$$
for some $\delta > 0$ sufficiently small in terms of $m,\lambda$. Then
$$
 \|f^{*m}\|_{\ell^2}\leq\|f\|_{\ell^q}^m.
$$
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
The argument follows \cite[Section 4.3]{shao-cubes}.
If $f(0)=0$, then $h=0$ and the conclusion is trivial. We may therefore
assume that $f(0)\neq 0$. Set $x=|f(0)|$, $y=\|h\|_{\ell^p}$, and $t=y/x\leq\delta$. From the decomposition
$f=f(0)\1_{\{0\}}+h$ we obtain
$$
 f^{*m} = f(0)^m\1_{\{0\}}+mf(0)^{m-1}h
       +\sum_{r=2}^m\binom mr f(0)^{m-r}h^{*r}.
$$
The first two terms are orthogonal in $\ell^2$ because $h(0)=0$.
Since $\|h\|_{\ell^2}\leq\|h\|_{\ell^p}=y$, we have
$$
 \|f(0)^m\1_{\{0\}}+mf(0)^{m-1}h\|_{\ell^2}^2
 =x^{2m}+m^2x^{2m-2}\|h\|_{\ell^2}^2
 \leq x^{2m}(1+m^2t^2).
$$
For $1\leq r\leq m$, set $p_r=2r/(2r-1)\geq p$.
By Young's inequality \eqref{eq:young-general} and monotonicity \eqref{eq:interpolation-monotone}, we have
$$
 \|h^{*r}\|_{\ell^2}
 \leq\|h\|_{\ell^{p_r}}^r
 \leq\|h\|_{\ell^p}^r=y^r.
$$
Combining these estimates with the triangle inequality, we obtain
$$
\begin{aligned}
 \|f^{*m}\|_{\ell^2}
 &\leq  \|f(0)^m\1_{\{0\}}+mf(0)^{m-1}h\|_{\ell^2} + \sum_{r=2}^m\binom{m}{r} x^{m-r}\|h^{*r}\|_{\ell^2} \\
 &\leq x^m\Big(\sqrt{1+m^2t^2}
                      +\sum_{r=2}^m\binom mr t^r\Big).
\end{aligned}
$$
Since $\sqrt{1+m^2t^2}\leq1+m^2t^2/2$ and
$t^r\leq t^2$ for $r\geq2$ and $0\leq t\leq1$, we have
\begin{equation} \label{eq:local-upper}
 \|f^{*m}\|_{\ell^2} \leq x^m\Big(1 + \Big(\frac{m^2}{2} + 2^m\Big) t^2\Big).
\end{equation}
On the other hand, by H\"{o}lder's inequality (see \eqref{eq:support-norms}) we have
$$
 \|h\|_{\ell^q}\geq n^{1/q-1/p}\|h\|_{\ell^p}=e^{-\lambda/(2m)}y.
$$
Consequently,
\begin{equation}\label{eq:local-lower}
 \|f\|_{\ell^q}^m
 =\Big(x^q+\|h\|_{\ell^q}^q\Big)^{m/q} \geq x^m\Big(1+e^{-\lambda q/(2m)}t^q\Big)^{m/q}.
\end{equation}
We now compare the right-hand sides of \eqref{eq:local-upper} and \eqref{eq:local-lower}. Since $\delta$ is sufficiently small in terms of $m,\lambda$, we can ensure that
$$
 \Big(\frac{m^2}{2}+2^m\Big)\delta^{2-q}
 \leq e^{-\lambda q/(2m)}.
$$
Since $t \leq \delta$ and, by the definition of $q$, $m/q\geq1$
for sufficiently large $n$, it follows that
$$
1 + \Big(\frac{m^2}{2} + 2^m\Big) t^2 \leq 1 +  e^{-\lambda q/(2m)} t^q \leq \Big(1+e^{-\lambda q/(2m)}t^q\Big)^{m/q}.
$$
This completes the proof.

<!-- lemma-3: body_md -->
\label{lem:kappa}
For $m\geq2$ we have
$$
 \sum_{h\in\mathbb{Z}}\kappa_m(h)=1,
 \qquad \sum_{h\in\mathbb{Z}}h^2\kappa_m(h)=\frac m{6}.
$$
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
For each integer $r\geq1$, set
$$
 K_r=\chi^{*r}*\widetilde\chi^{*r},\qquad
 \widetilde\chi(x)=\chi(-x).
$$
Then $\kappa_m(h)=K_m(h)$ and $K_r=K_1^{*r}$, where
$$
 K_1(x)=\max(1-|x|,0).
$$
Since $K_1$ is even and has integral $1$, the same is true for each $K_r$.
Write $\{x\} = x-\lfloor x\rfloor$. For any $x$, only $h=\lfloor x\rfloor$
and $h=\lfloor x\rfloor+1$ can contribute to the following sums, so
\begin{align*}
 & \sum_{h\in\mathbb{Z}}K_1(h-x)=(1-\{x\}) + \{x\} = 1,\\
 & \sum_{h\in\mathbb{Z}}h^2K_1(h-x)
 =(1-\{x\})\lfloor x\rfloor^2
     +\{x\}(\lfloor x\rfloor+1)^2=x^2+\{x\}(1-\{x\}).
\end{align*}
For $m \geq 2$ we have
$$
 \sum_{h\in\mathbb{Z}}K_m(h-x)
 =\int_{\mathbb{R}}K_{m-1}(y)
       \Big(\sum_{h\in\mathbb{Z}}K_1(h-x-y)\Big)\,dy=\int_{\mathbb{R}}K_{m-1}(y)\,dy=1.
$$
Setting $x=0$ proves the first identity in the lemma.

For the second moment, first note that $\int_\mathbb{R} x^2K_1(x)\,dx=1/6$ by direct computation. For $r\geq2$ we have
$$
\begin{aligned}
  \int_\mathbb{R} x^2K_r(x)\,dx
 &=\int_{\mathbb{R}^2}(y+z)^2K_{r-1}(y)K_1(z)\,dy\,dz \\
 &= \Big(\int_{\mathbb{R}}y^2K_{r-1}(y)\,dy\Big)
       \Big(\int_{\mathbb{R}}K_1(z)\,dz\Big)+\Big(\int_{\mathbb{R}}K_{r-1}(y)\,dy\Big)
       \Big(\int_{\mathbb{R}}z^2K_1(z)\,dz\Big)\\
 &\qquad + 2\Big(\int_{\mathbb{R}}yK_{r-1}(y)\,dy\Big)\Big(\int_{\mathbb{R}}zK_1(z)\,dz\Big) \\
 &=\int_\mathbb{R} y^2K_{r-1}(y)\,dy+\frac16,
 \end{aligned}
$$
where the mixed term vanishes since each $K_r$ is even.
It follows by induction that $\int_\mathbb{R} x^2K_r(x)\,dx=r/6$
for each $r$.

For $m\geq2$, we therefore obtain
\begin{align*}
 \sum_{h\in\mathbb{Z}}h^2\kappa_m(h)
 &=\int_\mathbb{R} K_{m-1}(x)
                   \sum_{h\in\mathbb{Z}}h^2K_1(h-x)\,dx\\
 &=\int_\mathbb{R} K_{m-1}(x) \Big(x^2 + \{x\}(1-\{x\})\Big)\,dx \displaybreak[1]\\
 &=\frac{m-1}{6}
       +\int_\mathbb{R} \{x\}(1-\{x\})K_{m-1}(x)\,dx\\
 &=\frac{m-1}{6}
       +\int_0^1u(1-u)\sum_{j\in\mathbb{Z}}K_{m-1}(j+u)\,du\\
 &=\frac{m-1}{6}+\int_0^1u(1-u)\,du=\frac m6.
\end{align*}
This completes the proof.

<!-- proposition-1: body_md -->
\label{prop:identity}
Let $v:\mathbb{Z}\to\mathbb{C}$ be finitely supported and let $V$ be defined by
\eqref{eq:step}.
Then
$$
 \|v^{*m}\|_{\ell^2}^2-\|V^{*m}\|_{L^2}^2
 =\frac12\sum_{h\in\mathbb{Z}}\kappa_m(h)
                    \|\Delta_h v^{*m}\|_{\ell^2}^2.
$$
In particular,
$$
 0\leq\|v^{*m}\|_{\ell^2}^2-\|V^{*m}\|_{L^2}^2
 \leq\frac m{12}\|\Delta_1 v^{*m}\|_{\ell^2}^2.
$$
<!-- end proposition-1 -->
<!-- proof-3: body_md -->
Write $a=v^{*m}$. Expanding the convolution in \eqref{eq:step},
we obtain
$$
 V^{*m}(x)=\sum_{k\in\mathbb{Z}}a(k)\chi^{*m}(x-k).
$$
Hence
$$
 \|V^{*m}\|_{L^2}^2 = \langle V^{*m}, V^{*m}\rangle
 =\sum_{k,\ell\in\mathbb{Z}}a(k)\overline{a(\ell)}
      \int_\mathbb{R}\chi^{*m}(x-k)\chi^{*m}(x-\ell)\,dx
$$
By setting $h=\ell-k$ and recalling the definition of $\kappa_m(h)$, we obtain
$$
 \|V^{*m}\|_{L^2}^2 = \sum_{h\in\mathbb{Z}}\kappa_m(h)
                     \sum_{k\in\mathbb{Z}}a(k)\overline{a(k+h)}.
$$
For each $h$, let $T_ha$ be the function defined by $T_ha(k) = a(k+h)$, so that $\Delta_ha = T_ha-a$. Taking real parts of both sides of the equation above, we obtain
$$
 \|V^{*m}\|_{L^2}^2 = \sum_{h\in\mathbb{Z}}\kappa_m(h) \cdot \operatorname{Re}\langle a,T_ha\rangle.
$$
By translation invariance, we have $\|T_ha\|_{\ell^2} = \|a\|_{\ell^2}$  and hence
$$
\|\Delta_ha\|_{\ell^2}^2 = \langle T_ha-a,T_ha-a\rangle = 2\|a\|_{\ell^2}^2 - 2\operatorname{Re}\langle a,T_ha\rangle.
$$
It follows that
$$
 \|V^{*m}\|_{L^2}^2
 =\sum_{h\in\mathbb{Z}}\kappa_m(h)
       \Big(\|a\|_{\ell^2}^2-\frac12\|\Delta_ha\|_{\ell^2}^2\Big)
 =\|a\|_{\ell^2}^2
       -\frac12\sum_{h\in\mathbb{Z}}\kappa_m(h)\|\Delta_ha\|_{\ell^2}^2,
$$
where the second equality follows from
$\sum_h\kappa_m(h)=1$ in Lemma \ref{lem:kappa}. This proves the first assertion of the proposition.

For $h\geq1$, telescoping gives
$$
 \Delta_ha(k)=\sum_{j=0}^{h-1}\Delta_1a(k+j).
$$
The triangle inequality gives
$\|\Delta_ha\|_{\ell^2}\leq |h|\|\Delta_1a\|_{\ell^2}$ for
$h\geq1$. Since $\Delta_{-h}a(k)=-\Delta_ha(k-h)$, the same bound holds for all $h \in \mathbb{Z}$. Hence
$$
 0\leq\|v^{*m}\|_{\ell^2}^2-\|V^{*m}\|_{L^2}^2
  \leq\frac12\Big(\sum_{h\in\mathbb{Z}}h^2\kappa_m(h)\Big)
                         \|\Delta_1a\|_{\ell^2}^2
   =\frac m{12}\|\Delta_1a\|_{\ell^2}^2,
$$
where we used Lemma \ref{lem:kappa} in the last equality.
This completes the proof.

<!-- proposition-2: body_md -->
\label{prop:comparison}
Let $v:\mathbb{Z}\to[0,\infty)$ be finitely supported and suppose that
$$
 \|\Delta_1v\|_{\ell^1}\leq4\|v\|_{\ell^\infty}.
$$
Then
$$
 \|v^{*m}\|_{\ell^2}^2
 \leq C_m^{-1}\|v\|_{\ell^p}^{2m}
 +\frac{4m}{3}\|v\|_{\ell^\infty}^{2p}
                 \|v\|_{\ell^p}^{2m-2p}.
$$
<!-- end proposition-2 -->
<!-- proof-4: body_md -->
Let $V$ be the interpolation in \eqref{eq:step}. Since
$\|V\|_{L^p}=\|v\|_{\ell^p}$, the sharp Euclidean inequality
\eqref{eq:young-euclidean} gives
$$
 \|V^{*m}\|_{L^2}^2\leq C_m^{-1}\|v\|_{\ell^p}^{2m}.
$$
By Proposition \ref{prop:identity}, it remains to estimate
$\|\Delta_1(v^{*m})\|_{\ell^2}$. Since
$\Delta_1(v^{*m})=(\Delta_1v)*v^{*(m-1)}$, an application of
Young's inequality yields
$$
 \|\Delta_1(v^{*m})\|_{\ell^2}
 \leq \|\Delta_1v\|_{\ell^1}\cdot\|v^{*(m-1)}\|_{\ell^2}.
$$
Set $r=2(m-1)/(2m-3)$. By Young's convolution inequality \eqref{eq:young-general} again, we have
$$
 \|v^{*(m-1)}\|_{\ell^2}
 \leq\|v\|_{\ell^r}^{m-1}.
 $$
Since $r>p$ and $(m-1)/r=(m-p)/p$, the interpolation
inequality \eqref{eq:interpolation-infinity} gives
$$
\|v\|_{\ell^r}^{m-1}
 \leq\|v\|_{\ell^\infty}^{p-1}\cdot\|v\|_{\ell^p}^{m-p}.
$$
Combining the three inequalities above, we obtain
\begin{align*}
 \|\Delta_1(v^{*m})\|_{\ell^2}
 &\leq\|\Delta_1v\|_{\ell^1}\cdot
           \|v\|_{\ell^\infty}^{p-1}\cdot\|v\|_{\ell^p}^{m-p}\\
 &\leq4\|v\|_{\ell^\infty}^{p}\cdot\|v\|_{\ell^p}^{m-p}.
\end{align*}
Substituting this bound into Proposition \ref{prop:identity} completes the proof.

<!-- lemma-4: body_md -->
\label{lem:tail-product}
Let $g$ be a nonnegative balanced function in $\ell^p(\mathbb{Z})$.
For an integer $R\geq1$, let $I_R = [-R,R] \cap \mathbb{Z}$ and define
$$
 S_R=\|g\1_{I_R}\|_{\ell^1},\qquad
 M_R=\sup_{x\notin I_R}g(x).
$$
Then $S_RM_R^{p-1}\to0$ as $R\to\infty$.
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
List the values of $g$ in decreasing order as
$a_1\geq a_2\geq\cdots\geq0$. Since $g$ is balanced, it suffices to prove that
$$
 \Big(\sum_{i\leq N}a_i\Big)a_{N+1}^{p-1}\longrightarrow0
$$
as $N\rightarrow\infty$.
Monotonicity  gives
$$
 \frac{N}{2}a_{N+1}^p\leq \sum_{i>\lfloor N/2\rfloor}a_i^p.
$$
Since $g \in \ell^p(\mathbb{Z})$, the right-hand side above tends to $0$ as $N\rightarrow\infty$, and hence so does the left-hand side.
By H\"older's inequality,
\begin{align*}
 \Big(\sum_{i\leq N}a_i\Big)a_{N+1}^{p-1}
 &\leq N^{1-1/p}\Big(\sum_{i\leq N}a_i^p\Big)^{1/p}
                         a_{N+1}^{p-1}\\
 &\leq\|g\|_{\ell^p}(Na_{N+1}^p)^{1-1/p}\longrightarrow0.
\end{align*}
This completes the proof.

<!-- proposition-3: body_md -->
\label{prop:energy-limit}
Let $\{f_j\}$ be a sequence of finitely supported, nonnegative, balanced functions with
$\|f_j\|_{\ell^p}=1$. Suppose that $f_j(x)\to g(x)$ for every
$x\in\mathbb{Z}$, and set $\rho=\|g\|_{\ell^p}^p$. Then
$$
 \limsup_{j\to\infty}\|f_j^{*m}\|_{\ell^2}^2
 \leq\rho^s+C_m^{-1}(1-\rho)^s.
$$
<!-- end proposition-3 -->
<!-- proof-6: body_md -->
For an integer $R\geq1$, let $I_R = [-R,R]\cap\mathbb{Z}$ and write
$$
 u_{j,R}=f_j\1_{I_R},\qquad
 v_{j,R}=f_j\1_{\mathbb{Z}\setminus I_R},\qquad
 g_R=g\1_{I_R}.
$$
Set
$$
 \rho_R=\|g_R\|_{\ell^p}^p,
 \qquad S_R=\|g_R\|_{\ell^1},
 \qquad M_R=\sup_{x\notin I_R}g(x).
$$
Thus $f_j = u_{j,R} + v_{j,R}$ and we have
$$
f_j^{*m} = u_{j,R}^{*m} + v_{j,R}^{*m} + \sum_{1 \leq b \leq m-1} \binom{m}{b}u_{j,R}^{*(m-b)}*v_{j,R}^{*b}.
$$
We first show that convolutions involving both $u_{j,R}$ and
$v_{j,R}$ make a negligible contribution in $\ell^2$-norm in the limit. Let $1\leq b\leq m-1$ and set $p_b=2b/(2b-1)>p$. By Young's convolution inequality \eqref{eq:young-general} and interpolation \eqref{eq:interpolation-infinity}, we get
\begin{align*}
 \|u_{j,R}^{*(m-b)}*v_{j,R}^{*b}\|_{\ell^2}
 &\leq\|u_{j,R}\|_{\ell^1}^{m-b}
                         \|v_{j,R}\|_{\ell^{p_b}}^b\\
 &\leq\|u_{j,R}\|_{\ell^1}^{m-b}
        \|v_{j,R}\|_{\ell^\infty}^{b-bp/p_b}
        \|v_{j,R}\|_{\ell^p}^{bp/p_b}\\
 &\leq \|u_{j,R}\|_{\ell^1}^{m-b}
        \|v_{j,R}\|_{\ell^\infty}^{b-bp/p_b}.
\end{align*}
As $j\to\infty$, we have $\|u_{j,R}\|_{\ell^1} \rightarrow S_R$ and $\|v_{j,R}\|_{\ell^{\infty}} \rightarrow M_R$ for fixed $R$, since the functions $f_j$ and $g$ are all balanced. More explicitly, $\|v_{j,R}\|_{\ell^\infty}=f_j(R+1)$ and $M_R=g(R+1)$. The identity $b-bp/p_b=(m-b)(p-1)$ therefore gives the required tail bound. By applying Lemma
\ref{lem:tail-product}, we obtain
$$
 \limsup_{j\to\infty}
 \|u_{j,R}^{*(m-b)}*v_{j,R}^{*b}\|_{\ell^2}
 \leq(S_RM_R^{p-1})^{m-b}=o_{R\to\infty}(1).
$$
It follows that
$$
 \limsup_{j\to\infty} \|f_j^{*m}\|_{\ell^2} \leq \limsup_{j\to\infty} \|u_{j,R}^{*m} + v_{j,R}^{*m}\|_{\ell^2} + o_{R\rightarrow\infty}(1).
$$
Since $\|u_{j,R}^{*m}\|_{\ell^2}$ and
$\|v_{j,R}^{*m}\|_{\ell^2}$ are at most $1$ by Young's
inequality, squaring both sides gives
$$
 \limsup_{j\to\infty}\|f_j^{*m}\|_{\ell^2}^2
 \leq\limsup_{j\to\infty}\|u_{j,R}^{*m}\|_{\ell^2}^2
       +\limsup_{j\to\infty}\|v_{j,R}^{*m}\|_{\ell^2}^2+2\limsup_{j\to\infty}
         \langle u_{j,R}^{*m},v_{j,R}^{*m}\rangle
       +o_{R\to\infty}(1).
$$
For the inner product, we have
$$
|\langle u_{j,R}^{*m},v_{j,R}^{*m}\rangle| \leq \|u_{j,R}^{*m}\|_{\ell^1}\|v_{j,R}^{*m}\|_{\ell^\infty} = \|u_{j,R}\|_{\ell^1}^m\cdot\|v_{j,R}^{*m}\|_{\ell^\infty}.
$$
Set $r=m/(m-1)>p$. By Young's inequality  \eqref{eq:young-general} and interpolation \eqref{eq:interpolation-infinity}, we have
$$
 \|v_{j,R}^{*m}\|_{\ell^\infty}
 \leq\|v_{j,R}\|_{\ell^r}^m\leq\|v_{j,R}\|_{\ell^\infty}^{m-mp/r}
        \|v_{j,R}\|_{\ell^p}^{mp/r}\leq\|v_{j,R}\|_{\ell^\infty}^{m-mp/r}.
$$
Note that $m-mp/r = m(p-1)$.
By applying Lemma \ref{lem:tail-product}, we obtain
$$
 \limsup_{j\to\infty}
 |\langle u_{j,R}^{*m},v_{j,R}^{*m}\rangle|
 \leq(S_RM_R^{p-1})^m=o_{R\to\infty}(1).
$$
For the $\ell^2$-norm of $u_{j,R}^{*m}$, note that
$u_{j,R}^{*m}\to g_R^{*m}$ pointwise and all these functions
are supported on the fixed interval $[-mR,mR]\cap\mathbb{Z}$. Hence
$$
\lim_{j\rightarrow\infty} \|u_{j,R}^{*m}\|_{\ell^2}^2 = \|g_R^{*m}\|_{\ell^2}^2 \leq \|g_R\|_{\ell^p}^{2m} = \rho_R^{s}.
$$
It follows that
\begin{equation}\label{eq:split-energy}
 \limsup_{j\to\infty}\|f_j^{*m}\|_{\ell^2}^2 \leq
 \rho_R^{s}
 +\limsup_{j\to\infty}\|v_{j,R}^{*m}\|_{\ell^2}^2
 +o_{R\to\infty}(1).
\end{equation}
For the tail $v_{j,R}$, since it is decreasing on $(R, +\infty)$ and increasing on $(-\infty,-R)$, we have $\|\Delta_1v_{j,R}\|_{\ell^1} \leq 4\|v_{j,R}\|_{\ell^{\infty}}$. Hence by Proposition \ref{prop:comparison} we obtain
$$
\|v_{j,R}^{*m}\|_{\ell^2}^2 \leq C_m^{-1}\|v_{j,R}\|_{\ell^p}^{2m} + \frac{4m}{3}\|v_{j,R}\|_{\ell^{\infty}}^{2p} \|v_{j,R}\|_{\ell^p}^{2m-2p}.
$$
Since $\|v_{j,R}\|_{\ell^p}^p = 1 - \|u_{j,R}\|_{\ell^p}^p \rightarrow 1-\rho_R$, we get
$$
 \limsup_{j\to\infty}\|v_{j,R}^{*m}\|_{\ell^2}^2
 \leq C_m^{-1}(1-\rho_R)^s
       +\frac{4m}{3}M_R^{2p}(1-\rho_R)^{s-2}.
$$
Since $M_R = o_{R\rightarrow\infty}(1)$, we conclude by substituting the above inequality into \eqref{eq:split-energy} that
$$
 \limsup_{j\to\infty}\|f_j^{*m}\|_{\ell^2}^2 \leq \rho_R^s + C_m^{-1}(1-\rho_R)^s + o_{R\rightarrow\infty}(1).
 $$
Finally, letting $R\rightarrow\infty$ completes the proof since $\rho_R\rightarrow\rho$.

<!-- proposition-4: body_md -->
\label{prop:q-limit}
Under the hypotheses of Proposition \ref{prop:energy-limit},
suppose also that $|\mathrm{supp} f_j|\leq n_j$ and $n_j\to\infty$.
Fix $\lambda>0$ and, for all sufficiently large $j$, define $q_j$ by
$$
 \frac1{q_j}=\frac1p-\frac{\lambda}{2m\log n_j}.
$$
Then
$$
 \liminf_{j\to\infty}\|f_j\|_{\ell^{q_j}}^{2m}
 \geq\big(\rho+e^{-\lambda/s}(1-\rho)\big)^s.
$$
<!-- end proposition-4 -->
<!-- proof-7: body_md -->
For an integer $R\geq1$, write $I_R = [-R,R]\cap\mathbb{Z}$ and
$\rho_R=\sum_{x\in I_R}g(x)^p$ (as in the proof of Proposition
\ref{prop:energy-limit}). Since $f_j(x)\to g(x)$ and $q_j\to p$, we have
$$
 \lim_{j\rightarrow\infty} \sum_{x\in I_R}f_j(x)^{q_j} = \rho_R.
$$
Now consider the contribution from $x \notin I_R$. Since there are at most $n_j$ nonzero terms, H\"{o}lder's inequality (see \eqref{eq:support-norms}) gives
$$
 \sum_{x\notin I_R} f_j(x)^{p}
 \leq n_j^{1-p/q_j}
       \Big(\sum_{x\notin I_R}f_j(x)^{q_j}\Big)^{p/q_j},
$$
which leads to
$$
 \sum_{x\notin I_R}f_j(x)^{q_j}
 \geq n_j^{1-q_j/p}
       \Big(\sum_{x\notin I_R}f_j(x)^p\Big)^{q_j/p}.
$$
Since $\sum_xf_j(x)^p=1$, we have
$$
 \lim_{j\rightarrow\infty} \sum_{x\notin I_R}f_j(x)^p
 = \lim_{j\rightarrow\infty} \Big(1-\sum_{x\in I_R}f_j(x)^p\Big)=1-\rho_R.
$$
The definition of $q_j$ gives
$$
 \frac{q_j}{p}=\Big(1-\frac{\lambda}{s\log n_j}\Big)^{-1},
 \qquad \Big(\frac{q_j}{p}-1\Big)\log n_j
 =\frac{\lambda/s}{1-\lambda/(s\log n_j)}
 \longrightarrow\frac\lambda s.
$$
Hence $\lim_{j\rightarrow\infty} n_j^{1-q_j/p} = e^{-\lambda/s}$. Combining the estimates
above and using $q_j\to p$, we obtain
$$
 \liminf_{j\to\infty}\sum_{x\notin I_R}f_j(x)^{q_j}
 \geq e^{-\lambda/s}(1-\rho_R).
$$
Adding the contribution from $I_R$ gives
$$
 \liminf_{j\to\infty}\sum_xf_j(x)^{q_j}
 \geq\rho_R+e^{-\lambda/s}(1-\rho_R).
$$
Raising both sides to the exponent $2m/q_j$ and using $2m/q_j\to s$ gives 
$$
 \liminf_{j\to\infty} \|f_j\|_{\ell^{q_j}}^{2m}
 \geq\Big(\rho_R+e^{-\lambda/s}(1-\rho_R)\Big)^{s}.
$$
Letting $R\to\infty$ completes the proof.

<!-- proposition-5: body_md -->
\label{prop:tensorization}
For integers $m,n\geq2$, one has
$$
 t_{m,n}=\frac{2m}{q_m(n)}.
$$
<!-- end proposition-5 -->
<!-- proof-8: body_md -->
\cite[Proposition 21]{dios-pont-et-al} shows  the relation
$$
 t_{m,n}=\frac{2m}{q_m'(n)},
$$
where $q_m'(n)$ is the supremum of all $q \geq 1$ for which \eqref{eq:HY-threshold} holds for all functions $f: \{0,1,\cdots,n-1\}\rightarrow\mathbb{R}$. Let $q = q_m'(n)$. We prove that \eqref{eq:HY-threshold} holds for all complex-valued functions $f$ with $|\mathrm{supp} f| \leq n$. Indeed, by replacing $f$ by $|f|$, we may assume that $f$ takes nonnegative real values. By the rearrangement inequality, we may assume that $f$ is supported on an interval of $n$ consecutive integers. By translation invariance, we may assume that $f$ is supported on $\{0,1,\cdots,n-1\}$. This proves that $q_m(n) = q_m'(n)$ as desired.
