<!-- head-1: body_md -->
We consider nonlinear dynamics with both parametric and exogenous worst-case uncertainty,
\begin{equation}\label{eq:uncert_param_system}
x_{k+1} = f(x_k,u_k;\theta) + F(x_k)w_k,
\end{equation}
where $x_k \in \mathcal{X} \subseteq \mathbb{R}^{n_x}$ is the system state, $u_k \in \mathcal{U}\subseteq \mathbb{R}^{n_u}$ is the control input, and $\theta \in \Theta_0 \subseteq\mathbb{R}^{n_{\theta}}$ is an unknown parameter (e.g., mass) contained in an initial zonotopic uncertainty set $\Theta_0$. The map $f:\mathbb{R}^{n_x}\times\mathbb{R}^{n_u}\times\mathbb{R}^{n_{\theta}} \to \mathbb{R}^{{n_x}}$ defines the nonlinear dynamics, while $F:\mathbb{R}^{{n_x}} \to \mathbb{R}^{{n_x} \times {n_x}}$ scales the bounded disturbance $w_k \in  \mathcal{B}^{n_x}_\infty$. Because $\theta$ is unknown at planning time, the controller begins from an initial estimate $\hat{\theta}_0$ and refines it online as measurements are collected, yielding $\hat{\theta}_k$ at time $k$. We seek a robust-adaptive policy with (i) a feedback component that tracks the planned state trajectory and (ii) a feedforward component that exploits updated parameter estimates as uncertainty is reduced. The policy must robustly satisfy all state and input constraints despite residual parametric uncertainty and process noise. We therefore seek to solve the following robust-adaptive nonlinear optimal control problem (RANOCP):
\begin{subequations} \label{eq:ranocp}
    \begin{align}
        \hspace{-20mm}\min_{\pi(\cdot)} \ &J\left(x_1,\dots,x_N, \pi(\cdot)\right) \\
        \text{s.t.} \ & x_{k+1} = f(x_k, u_k;\hat\theta_k) + E(x_k,u_k,\hat\theta_k;\theta) \nonumber\\ &\qquad\qquad+ F(x_k)w_k, \quad \forall k \in [N],
        \\
                          & x_0 = \bar{x}_0, \quad\hat\theta_0 = \hat\theta,\\ 
                          & \hat\theta_k = m_k(\hat\theta_0, r_1, \dots, r_k; \theta), \quad\forall k\in[N]\\
                          & u_k = \pi_k(x_{0},\dots,x_{k}, \hat\theta_0, \dots,\hat\theta_k), \quad \forall k \in [N], \\
                          & g_k(x_k, u_k) \leq 0,  \,\,\forall w_1,\dots,w_k \in \mathcal{B}^{n_x}_\infty,\nonumber\\
                          & \quad\qquad\qquad\qquad \forall \theta \in \Theta_k,\,\forall k \in [N], \label{eq:robust_nocp_con_running}\\
                          & g_N(x_N) \leq 0, \,\, \forall w_1,\dots,w_N \in \mathcal{B}^{n_x}_\infty,\,\, \forall \theta \in \Theta_N,\label{eq:robust_nocp_con_terminal}
    \end{align}
\end{subequations}
\looseness-1where $\pi:=\{\pi_i\}_{i=0}^{N-1}$ is a sequence of causal control policies, $m_k:\mathbb{R}^{{n_{\theta}}}\times(\mathbb{R}^{{n_x}})^k \times\mathbb{R}^{{n_{\theta}}}\to \mathbb{R}^{{n_{\theta}}}$ is a measurement function based on the history of residual measurements $r_k$, $\bar{x}_0\in\mathbb{R}^{{n_x}}$ is the initial state, and $\hat{\theta}, \theta\in\mathbb{R}^{{n_{\theta}}}$ are the initial parameter guess and true parameter, respectively. The set $\Theta_k$ denotes the parameter uncertainty set at step $k$, with $\hat{\theta}_k\in\Theta_k$ the parameter estimate. The functions $g_k:\mathbb{R}^{n_x}\times\mathbb{R}^{n_u}\rightarrow\mathbb{R}^{n_c}$ denote stagewise state-input constraints, while $g_N:\mathbb{R}^{n_x}\rightarrow\mathbb{R}^{n_c^f}$ denotes the terminal constraint. Lastly, $E:\mathbb{R}^{{n_x}}\times\mathbb{R}^{{n_u}}\times\mathbb{R}^{{n_{\theta}}}\times\mathbb{R}^{{n_{\theta}}}\rightarrow\mathbb{R}^{n_x}$ represents the residual dynamics error from an incorrect parameter estimate.

<!-- proposition-1: body_md -->
\label{prop:disturbance_expression}
Let $e_0 = G_0\xi_0$, where $\xi_0 \in \mathcal{B}^{p_0}_\infty$ represents the initial parameter-error realization, with $p_0$ generators, and define the Kalman gain product
\begin{equation}
\Psi_{j,\ell} =
\begin{cases}
\textstyle \prod_{m=\ell}^{j-1}
(I_{{n_{\theta}}}-K_m\Pi_m), & j>\ell,\\
I_{{n_{\theta}}}, & j=\ell,\\
0_{{n_{\theta}}\times{n_{\theta}}}, & \text{otherwise}.
\end{cases}
\end{equation}
Then the parameter error $e_j$ and estimate shift $\Delta\hat{\theta}_j$ (Eqs. \eqref{eq:param_error_w_bound}-\eqref{eq:param_shift}) can be expressed in terms of the initial parameter error and disturbances $w_\ell$ for $\ell\le j$:
\begin{align}
e_j
&= \Psi_{j,0}G_0\xi_0
-\textstyle\sum_{\ell=0}^{j-1}
\Psi_{j,\ell+1}K_\ell W_\ell w_\ell,
\label{eq:param_error_dist}\\
\Delta\hat{\theta}_j
&= K_j\Pi_j\Psi_{j,0}G_0\xi_0
+K_jW_jw_j
\nonumber\\
&\qquad
-\textstyle\sum_{\ell=0}^{j-1}
K_j\Pi_j\Psi_{j,\ell+1}K_\ell W_\ell w_\ell.
\label{eq:param_shift_dist}
\end{align}

<!-- theorem-1: body_md -->
\label{thm:adaptive_gains}
Define the adaptive disturbance gains
\begin{align}
\hspace{-7pt}\bm{M}_{k,\xi}
&=
\textstyle\sum_{j=0}^{k-1}
\left(
\mathbf{\tilde\Phi}_{k,j}E_j
+\mathbf{\bar\Phi}_{k,j}K_j\Pi_j
\right)\mathbf{\Psi}_{j,0}G_0, \\
\hspace{-7pt}\bm{M}_{k,j}
&=\mathbf{\tilde\Phi}_{k,j}F_j
+
\mathbf{\bar\Phi}_{k,j}K_jW_j
\nonumber\\&-\textstyle\sum_{t=j+1}^{k-1}
\left(
\mathbf{\tilde\Phi}_{k,t}E_t
+\mathbf{\bar\Phi}_{k,t}K_t\Pi_t
\right)\mathbf{\Psi}_{t,j+1}K_jW_j,
\end{align}
where $\mathbf{\Phi}_{k,j}:=\begin{bmatrix}\mathbf{\tilde\Phi}_{k,j}&\mathbf{\bar\Phi}_{k,j}\end{bmatrix}$, with $\mathbf{\tilde\Phi}_{k,j} \in\mathbb{R}^{({n_x}+{n_{\theta}}+{n_u})\times{n_x}}$ and $\mathbf{\bar\Phi}_{k,j} \in\mathbb{R}^{({n_x}+{n_{\theta}}+{n_u})\times{n_{\theta}}}$. Then the state and control deviations satisfy
\begin{equation}\label{eq:adaptive_state_disturbance}
\begin{bmatrix}
\Delta s_k\\ \Delta u_k
\end{bmatrix}
=\bm{M}_{k,\xi}\xi_0+\textstyle\sum_{j=0}^{k-1}\bm{M}_{k,j}w_j.
\end{equation}

<!-- corollary-1: body_md -->
\label{cor:robust_constraint_satisfaction}
Robust constraint satisfaction is ensured if, for all $k \in [N]$,
\begin{equation}\label{eq:robust_constraint_combined}
g_k(z_k, v_k) + \beta_k \le 0_{n_c}, \quad
g_N(z_N) + \beta_N \le 0_{n_c^f},
\end{equation}
where, for all $k \in [N+1]$,
\begin{equation}\label{eq:beta_definition}
\beta_k := \|P_k\bm{M}_{k,\xi}\|_{1,r}
+\textstyle\sum_{j=0}^{k-1}\|P_k\bm{M}_{k,j}\|_{1,r}
+b^\textrm{lin}_k,
\end{equation}
and $b_k^\textrm{lin}$ is the correction factor from \cite[Eq. 22]{zhan2026robustly} accounting for nonlinear constraints.

<!-- remark-1: body_md -->
The resulting A-SLS controller is compatible with the recursive-feasibility construction of \cite{leeman2025guaranteed}. In particular, \eqref{eq:fastsls_problem} can be augmented with a robust positively invariant terminal set and associated terminal controller, together with the shifted nominal and auxiliary trajectory construction in \cite[Eqs. 27-29]{leeman2025guaranteed}. The adaptive disturbance representation in \eqref{eq:adaptive_state_disturbance} and nonlinear linearization-error bound in \eqref{eq:linearization_error} can then be treated as the bounded prediction uncertainty. Under the corresponding assumptions, initial feasibility implies recursive feasibility and robust constraint satisfaction.
