<!-- remark-1: body_md -->
Identifying the two binary expansions of each dyadic point, and also
	the endpoints in the circle case, recovers the standard piecewise-linear
	realizations of $\mathsf F$ on $[0,1]$ and $\mathsf T$ on $\mathbf{R}/\mathbf{Z}$;
	see \cite{CannonFloydParry1996,CannonFloyd2011}.
<!-- end remark-1 -->
<!-- definition-1: body_md -->
\label{def:smashed}
	The \emph{smashed boundary} is
	\[
	\partial_I^\#\mathcal{F}:=\partial_I\mathcal{F}/\!\sim,
	\qquad
	\varepsilon\sim\eta
	\ \Longleftrightarrow\
	\operatorname{ev}(\varepsilon)=\operatorname{ev}(\eta).
	\]
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:left-flip}
On the set of ends of the Farey tree, the \emph{flip at $WL$} is defined as the map
\[
 \phi_{WL}(X)=
 \begin{cases}
  WB^2Y,&X=WBY,\\
  WBAY,&X=WABY,\\
  WAY,&X=WA^2Y,\\
  X,&X\notin\mathcal{C}(W).
 \end{cases}
\]
Its inverse is the \emph{flip at $WL^2$}:
\[
 \phi_{WL^2}(X)=
 \begin{cases}
  WBY,&X=WBBY,\\
  WABY,&X=WBAY,\\
  WAAY,&X=WAY,\\
  X,&X\notin\mathcal{C}(W).
 \end{cases}
\]
<!-- end definition-2 -->
<!-- proposition-1: body_md -->
\label{prop:flip-descends}
Every flip descends to an orientation-preserving homeomorphism of $\partial_I^{\#}\mathcal{F}$.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
The nontrivial identifications are
	\[
	PAB^\infty\sim PBA^\infty
	\quad(P\text{ admissible}),
	\qquad
	B^\infty\sim SA^\infty,
	\qquad
	A^\infty\sim SB^\infty.
	\]
	The last two pairs represent $0$ and $\infty$, respectively.
	The defining substitutions of each flip and its inverse
	preserve these identifications, so they induce mutually inverse
	homeomorphisms of the quotient.
	
	Within its supporting cylinder, a left flip preserves
	the order of the prefix lists:
	\[
	(WB,WAB,WAA)\longmapsto(WBB,WBA,WA).
	\]
	The base flip cyclically permutes its four initial
	cylinders. Thus every flip preserves cyclic order
	on the quotient.
<!-- end proof-1 -->
<!-- definition-3: body_md -->
\label{def:shuffle}
On the set of ends of the Farey tree, the \emph{shuffle at $W$} is defined as the local transposition $\sigma_W$ given by
\[
 \sigma_W(WAY)=WBY,\qquad
 \sigma_W(WBY)=WAY,
\]
and $\sigma_W(X)=X$ for $X\notin\mathcal{C}(W)$.
<!-- end definition-3 -->
<!-- remark-2: body_md -->
In fact, the composition
 \[
\bar\kappa\circ\overline{\operatorname{ev}}^{-1}:  \mathbb{P}^1(\mathbf{R})\longrightarrow\mathcal B_{\mathrm{circ}}
 \]
 is the circular Minkowski question-mark function; it sends rational points
 to dyadic points.
<!-- end remark-2 -->
<!-- theorem-1: body_md -->
\label{thm:flip-conjugacy}
Let
\[
	G_{\mathrm{flip}} :=  \left\langle \phi_I,\phi_{WL},\phi_{WL^2} : W\text{ admissible}  \right\rangle.
\]
The group $G_{\mathrm{flip}}$ is isomorphic to Thompson's group $\mathsf T$ in its prefix-code realization. More precisely, 
\begin{equation}\label{eq:flip-generators}
 X_0=\phi_I\phi_{SL^2}\phi_I^3\phi_{SL^2}\phi_I^3,
 \qquad X_1=\phi_{SL},\qquad Y=\phi_I^2.
\end{equation}
<!-- end theorem-1 -->
<!-- proof-2: body_md -->
The flip $\phi_I$ acts by
	\[
	 00 \xi \mapsto01\xi\mapsto 10\xi\mapsto 11\xi \mapsto00\xi.
	\]
	Taking $W=I$ in \cref{def:left-flip} gives
	\begin{equation}\label{eq:phi-L-binary}
		\phi_L:
		\quad
		00\xi\mapsto000\xi,\qquad
		010\xi\mapsto001\xi,\qquad
		011\xi\mapsto01\xi,
		\qquad 1\xi\mapsto1\xi.
	\end{equation}
	Thus $\phi_L$ is supported on $0\mathcal B$.  Taking $W=S$ gives
	\begin{equation}\label{eq:phi-SL-binary}
		\phi_{SL}:
		\quad
		0\xi\mapsto0\xi,\qquad
		10\xi\mapsto100\xi,\qquad
		110\xi\mapsto101\xi,\qquad
		111\xi\mapsto11\xi.
	\end{equation}
	Its inverse is the right flip
	\begin{equation}\label{eq:phi-SL2-binary}
		\phi_{SL^2}:
		\quad
		0\xi\mapsto0\xi,\qquad
		100\xi\mapsto10\xi,\qquad
		101\xi\mapsto110\xi,\qquad
		11\xi\mapsto111\xi.
	\end{equation}
	More generally, every flip has a cyclic-order-preserving prefix table, so we deduce $G_{\mathrm{flip}}\leq\mathsf T$.  For the reverse
	inclusion, we calculate the standard generators in flip coordinates. Indeed, $X_1=\phi_{SL}$, $\phi_I^2$ exchanges $0\xi$ and $1\xi$, and hence
	$Y=\phi_I^2$.  Finally, composing this four-cycle with the right-flip table \eqref{eq:phi-SL2-binary}, from right to left, gives

<!-- proposition-2: body_md -->
\label{prop:flip-shuffle-relations}
	The following identities hold among the boundary flips and shuffles.
	
	\smallskip
	\noindent
	\emph{Flip--flip relations.}  A right flip is the inverse of the
	corresponding left flip:
	\begin{equation}\label{eq:flip-inverse}
		\phi_{UL^2}=\phi_{UL}^{-1}.
	\end{equation}
	If $\mathcal{C}(U)\cap\mathcal{C}(V)=\varnothing$, then
	\begin{equation}\label{eq:flip-disjoint-commutation}
		[\phi_{UL},\phi_{VL}]=1.
	\end{equation}
	For nested cylinders one has
	\begin{align}
		\phi_{UL}\phi_{(UBR)L}\phi_{UL^2}
		&=\phi_{(UBBR)L}, \label{eq:flip-transport-B}\\
		\phi_{UL}\phi_{(UABR)L}\phi_{UL^2}
		&=\phi_{(UBAR)L}, \label{eq:flip-transport-AB}\\
		\phi_{UL}\phi_{(UAAR)L}\phi_{UL^2}
		&=\phi_{(UAR)L}. \label{eq:flip-transport-AA}
	\end{align}
	Finally, the pentagon, or splitting, relation is
	\begin{equation}\label{eq:flip-pentagon}
		\phi_{UL}^{\,2}
		=\phi_{(UB)L}\phi_{UL}\phi_{(UA)L}.
	\end{equation}
	
	\smallskip
	\noindent
	\emph{Shuffle--shuffle relations.}  Every shuffle is an involution:
	\begin{equation}\label{eq:shuffle-involution}
		\sigma_U^2=1.
	\end{equation}
	If $\mathcal{C}(U)\cap\mathcal{C}(V)=\varnothing$, then
	\begin{equation}\label{eq:shuffle-disjoint-commutation}
		[\sigma_U,\sigma_V]=1,
	\end{equation}
	whereas for nested cylinders one has
	\begin{equation}\label{eq:shuffle-transport}
		\sigma_U\sigma_{UBR}\sigma_U=\sigma_{UAR}.
	\end{equation}
	
	\smallskip
	\noindent
	\emph{Mixed relations.}  If $\mathcal{C}(U)\cap\mathcal{C}(V)=\varnothing$, then
	\begin{equation}\label{eq:mixed-disjoint-commutation}
		[\phi_{UL},\sigma_V]=1.
	\end{equation}
	Conjugation of a shuffle by a flip gives
	\begin{align}
		\phi_{UL}\sigma_{UBR}\phi_{UL^2}
		&=\sigma_{UBBR}, \label{eq:mixed-flip-transport-B}\\
		\phi_{UL}\sigma_{UABR}\phi_{UL^2}
		&=\sigma_{UBAR}, \label{eq:mixed-flip-transport-AB}\\
		\phi_{UL}\sigma_{UAAR}\phi_{UL^2}
		&=\sigma_{UAR}. \label{eq:mixed-flip-transport-AA}
	\end{align}
	Conversely, conjugation by a shuffle interchanges the two descendant
	cylinders:
	\begin{equation}\label{eq:mixed-shuffle-transport}
		\sigma_U\phi_{(UBR)L}\sigma_U=\phi_{(UAR)L}.
	\end{equation}
	The remaining same-support interaction is the hexagon relation
	\begin{equation}\label{eq:flip-shuffle-hexagon}
		\phi_{UL}\sigma_U\phi_{UL}
		=\sigma_{UB}\phi_{UL}\sigma_{UA}.
	\end{equation}

	UBR,\qquad UABR,\qquad UAAR

	UBBR,\qquad UBAR,\qquad UAR,

\label{lem:finite-block-decomposition}
	We have:
	\begin{enumerate}
		\item $\mathscr E[C_K]$ and $\mathscr F[C_K]$ are maximal pairwise
		unlinked families of internal pairs of $C_K$;		
		\item $\mathscr E[C_K]\cap\mathscr F[C_K]=\varnothing$;
		
		\item the finite differences decompose as
		\[
		\mathscr E\setminus\mathscr F
		=
		\bigsqcup_K\mathscr E[C_K],
		\qquad
		\mathscr F\setminus\mathscr E
		=
		\bigsqcup_K\mathscr F[C_K].
		\]
	\end{enumerate}
	
	In particular,
	\begin{equation}\label{eq:defect-cardinalities-equal}
		|\mathscr E\setminus\mathscr F|
		=
		|\mathscr F\setminus\mathscr E|
		=
		\sum_K\bigl(|C_K|-3\bigr).
	\end{equation}
	The decomposition can be computed from the finite symmetric difference and
	the common-neighbour relation.
<!-- end lemma-2 -->
<!-- proof-6: body_md -->
Let $K$ have $k$ edges.  By construction, the internal members of
	$\mathscr E$ in $C_K$ are precisely the labels of the edges of $K$.
	Therefore
	\[
	|\mathscr E[C_K]|=k=|C_K|-3.
	\]
	In particular, the families $\mathscr E[C_K]$ partition
	$\mathscr E\setminus\mathscr F$.
	
	There is no pair belonging to both $\mathscr E[C_K]$ and $\mathscr F[C_K]$: a common internal pair would be an internal member of $\mathscr E$, hence a label of $K$, hence not in $\mathscr F$.
		
	Let $f\in\mathscr F\setminus\mathscr E$.  By maximality of $\mathscr E$,
	the pair $f$ is linked with some $e\in\mathscr E$.  Since $f$ is unlinked
	with every member of $\mathscr F$, the pair $e$ cannot belong to
	$\mathscr F$.  Hence
	\[
	e\in\mathscr E\setminus\mathscr F.
	\]
	Let $K$ be the component containing the edge labelled by $e$.  Every member
	of $\partial K$ belongs to $\mathscr F$, so $f$ is unlinked with
	$\partial K$.  Since $f$ is linked with an internal pair of $C_K$, the
	cyclic order implies that $f$ is itself internal in $C_K$.  Moreover, $f$ is internal in no other block $C_{K'}$: the pair $e$ is not internal in $C_{K'}$, since the internal members of $\mathscr E$ in $C_{K'}$ are the labels of $K'$, and $e$ is unlinked with $\partial K'$; hence $e$ is unlinked with every internal pair of $C_{K'}$.  Since also $\mathscr F[C_K]\cap\mathscr E\subseteq\mathscr E[C_K]\cap\mathscr F[C_K]=\varnothing$, the families $\mathscr F[C_K]$ partition $\mathscr F\setminus\mathscr E$.
	
                                                                          
                                                                        
                                                                                       
                                                                            
                                                                          
  
	
	Both restrictions are maximal among pairwise unlinked internal
	pairs of $C_K$.  Indeed, let
	$\mathscr H\in\{\mathscr E,\mathscr F\}$.
	Since $\partial K\subseteq\mathscr H$, a member of $\mathscr H$
	linked with an internal pair of $C_K$ must itself be internal
	in $C_K$.  Hence an additional internal pair unlinked with
	$\mathscr H[C_K]$ would be unlinked with all of $\mathscr H$,
	contradicting maximality.

	cyclically ordered points has $r-3$ members.  Indeed, an internal pair
	divides the cyclic list into two shorter lists, and induction gives
	\[
	(r_1-3)+(r_2-3)+1=r-3,
	\qquad r_1+r_2=r+2.
	\]
	Consequently,
	\[
	|\mathscr E[C_K]|
	=
	|\mathscr F[C_K]|
	=
	|C_K|-3.
	\]
	Summing over the components proves
	\eqref{eq:defect-cardinalities-equal}.

\label{lem:block-fan-reduction}
	Let $C=(v_1,\ldots,v_r)$, with $r\geq4$, and let $\mathscr M$ be a maximal
	pairwise unlinked family of internal pairs of $C$.  There is a sequence of
	\[
	r-3-d_{\mathscr M}(v_1)
	\]
	diagonal exchanges, all supported inside $C$ and fixing $\partial C$, which

	\[
	\mathscr M\cup\partial C
	\]
	into the fan $\mathscr F_C(v_1)$.  In particular, the sequence contains at

	$\mathscr M\cup\partial C$.  Suppose that this list does not contain every

	which the interval from $u$ to $v$ not containing $v_1$ contains a point of

	$\mathscr M$.  Indeed, a pair linked with $\langle u,v\rangle$ would either
	be linked with one of $\langle v_1,u\rangle$ and
	$\langle v_1,v\rangle$, or would join $v_1$ to a point strictly between
	$u$ and $v$, contradicting the choice of consecutive neighbours.

\label{thm:smashed-boundary-algorithm}

		encoded length $O(s^2)$, and is computed in $O(s^3)$ time under the
		computational convention above.
	\end{enumerate}

	$\psi_N,\ldots,\psi_1$ and then the table of $t$ to $\infty$ and $0$.  Write
	$m_t(\vec e_0)=\left(\frac ac,\frac bd\right)$, choosing primitive integer
	representatives with $ad-bc=1$.  The matrix
	\[
	M=\begin{pmatrix}a&b\\ c&d\end{pmatrix}\in\operatorname{\mathrm{SL}_2(\mathbf{Z})}
	\]
	sends $\vec e_0=(\infty,0)$ to $m_t(\vec e_0)$, so, by simple transitivity of
	the $\operatorname{\mathrm{PSL}_2(\mathbf{Z})}$-action on oriented Farey edges, its class in $\operatorname{\mathrm{PSL}_2(\mathbf{Z})}$ is $m_t$.
	Applying the Euclidean algorithm to $M$, followed by reduction using
	$\operatorname{\mathrm{PSL}_2(\mathbf{Z})}=\langle S,L\mid S^2=L^3=1\rangle$, produces the reduced free-product
	normal form of $m_t$.  Finally, $t\psi_1\cdots\psi_N=m_t$ gives
	\eqref{eq:smashed-boundary-algorithm-output}, and
	\eqref{eq:modular-generators-as-flips} expresses it entirely in boundary flips.
	
	For a disagreement block $C$ with $r$ points, the two
	fan reductions together use at most $2(r-3)$ exchanges.
	By \eqref{eq:defect-cardinalities-equal},
	\[
	N(t)\leq2\sum_C(|C|-3)=2k_-(t),
	\qquad k_-(t)=k_+(t).
	\]
	Every missing standard pair must be inserted, and each
	exchange inserts only one pair. Hence
	$k_-(t)\leq N_{\min}(t)\leq N(t)$, proving
	\eqref{eq:main-factor-two-bound}.
	
	The fixed ordering conventions determine the exchanges
	from $\mathscr E_t$. Starting with $\theta_0=t$, the
	recursion then determines every $\psi_j$ and $m_t$.
	Thus the output is independent of the input table.
	
	The complete output and running-time bounds follow from
	\eqref{eq:complete-bound-lambda},
	\cref{lem:residual-modular-length,cor:T-input-complexity}
	below.

<!-- remark-3: body_md -->
One may instead insert missing standard pairs successively.
	Give a missing pair $\alpha$ its preferred orientation
	$(x,y)$. Among the pairs $\langle u,v\rangle\in\mathscr E$
	linked with $\alpha$, with $u\in(x,y)$ and $v\in(y,x)$,
	choose the least $u$ and, among ties, the greatest $v$.
	Any member of $\mathscr E$ linked with $\langle x,u\rangle$
	or $\langle x,v\rangle$ would have its first endpoint
	before $u$, or share $u$ and have its second endpoint
	after $v$, contradicting this choice.
	Maximality therefore makes $x$ a common neighbour of
	$u,v$. If $z$ is their other common neighbour, exchange
	$\langle u,v\rangle$ for $\langle x,z\rangle$.
	This decreases the finite number of pairs linked with
	$\alpha$ by one; repetition inserts $\alpha$ by maximality.
	No standard pair is removed, and the number of nonstandard
	pairs never increases. Starting from $\mathscr E_t$, the
	procedure therefore uses at most $k_-(t)k_+(t)$ exchanges.

<!-- lemma-5: body_md -->
\label{lem:address-growth}
	Let $x\in\mathscr R$.
	\begin{enumerate}
		\item If $\phi$ is a flip, then $\ell(\phi(x))\leq\ell(x)+2$.
		\item If $h=h_{P,Q,\tau} \in \mathsf{T}$, then $\ell(h(x))\leq\ell(x)+\max_i|q_i|$.
		\item If $W$ is an admissible prefix with binary code $w$, then the
		larger of $\ell(W(0))$ and $\ell(W(\infty))$ equals $|w|$, and the
		reduced word of $W$ in $S,L,L^2$ has at most $2|w|-1$ letters.
	\end{enumerate}
<!-- end lemma-5 -->
<!-- proof-10: body_md -->
Let $ud^\infty$, with $d\in\{0,1\}$ and $|u|=\ell(x)$, represent $x$.
	
	(1) In binary coordinates, $\phi_{WL}$ acts on $w\mathcal B$ by
	$w0\xi\mapsto w00\xi$, $w10\xi\mapsto w01\xi$, $w11\xi\mapsto w1\xi$, and
	$\phi_{WL^2}$ is its inverse; both fix the complement of $w\mathcal B$.
	Thus a prefix $p$ is replaced by a prefix $q$ with $|q|\leq|p|+1$.  If
	$|u|\geq|p|$, the image of $ud^\infty$ has a representative $vd^\infty$ with
	$|v|\leq|u|+1$.  If $|u|<|p|$, then $p=ud^{\,|p|-|u|}$, and $ud^\infty$ is
	fixed except for $w10\,0^\infty\mapsto w01\,0^\infty$ and
	$w01\,1^\infty\mapsto w10\,1^\infty$, where $\ell$ grows by one.  The base
	flip $\phi_I^{\pm1}$ replaces prefixes of length $2$ by prefixes of length
	$2$, so $\ell$ grows by at most two.
	
	(2) If $ud^\infty\in p_i\mathcal B$, then $h(ud^\infty)=q_{\tau(i)}vd^\infty$
	with $|v|=\max\{|u|-|p_i|,0\}$.
	
	(3) Write $W=S^\delta C_1\cdots C_k$ with $C_i\in\{A,B\}$, so that
	$|w|=k+1$.  The points $W(0)$ and $W(\infty)$ are represented by $w0^\infty$
	and $w1^\infty$.  Hence $\ell(W(0))$ and $\ell(W(\infty))$ are the lengths of
	$w$ with its terminal $0$'s, respectively $1$'s, removed, and one of them is
	$|w|$.  The reduced word $S^\delta L^{i_1}S\cdots L^{i_k}S$ has $\delta+2k\leq2|w|-1$ letters.

<!-- corollary-2: body_md -->
\label{cor:T-input-complexity}
	For a cyclic prefix table of size $s$, the deterministic marked reduction
	produces $O(s)$ flip factors.  Its explicitly encoded output has length
	$O(s^2)$, and a direct implementation on finite $A$--$B$ prefix words has
	running time $O(s^3)$.
<!-- end corollary-2 -->
<!-- proof-12: body_md -->
By \eqref{eq:defect-table-bound} and
	\eqref{eq:main-factor-two-bound},
	$N(t)\leq k_-(t)+k_+(t)\leq2s$.
	By \cref{lem:residual-modular-length}, $q\leq10s+1$.
	Hence \eqref{eq:complete-bound-lambda} yields
	\begin{equation}\label{eq:complete-linear-table-bound}
		\Lambda_{\mathsf T}(t)\leq N(t)+5q=O(s).
	\end{equation}
	
	Every point handled during the computation has $\ell=O(s)$.  Indeed, the
	endpoints of the pairs in $\mathscr A(P)\cup\mathscr A(Q)$ from the proof of \cref{lem:effective-finite-support} and their standard common neighbours have $\ell\leq s+1$; by \cref{lem:address-growth}, applying
	$t^{\pm1}$ adds at most $s$, each flip adds at most $2$, and at most
	$N(t)\leq2s$ flips are applied to any point.  By
	\cref{lem:address-growth}(3), every $\psi_j$ has a supporting address of
	length $O(s)$, while the modular part uses only $\phi_I$, $\phi_{SL}$ and
	$\phi_{SL^2}$.  Hence the encoded output length is
	\begin{equation}\label{eq:T-encoded-output-bound}
		O\bigl(N(t)\,s+q\bigr)=O(s^2).
	\end{equation}
	
	The initial finite-defect computation also takes $O(s^3)$
	time. There are $O(s)$ candidate pairs in $\mathscr A(P)\cup\mathscr A(Q)$, and their endpoints
	and required standard common neighbours have addresses
	of length $O(s)$ and integer coordinates of $O(s)$ bits.
	Each required conversion, Farey-condition test, and
	standard common-neighbour computation takes $O(s^2)$
	time with schoolbook arithmetic.
	
	For the running time, there are $O(s)$ such pairs, defect pairs and
	block incidences, each described by words of length $O(s)$.  Constructing
	and ordering the incidence forest uses $O(s^2)$ comparisons of such words,
	hence $O(s^3)$ time.  The fan stage performs $O(s)$ exchanges, each scanning
	$O(s)$ pairs of its block, which costs $O(s^3)$.  Computing $e_j^\circ$
	applies $t^{-1}$ and at most $j-1$ flips to two points with $\ell=O(s)$,
	each application costing $O(s)$; all pull-backs cost
	$O(N(t)^2s)=O(s^3)$.  Computing $m_t(\vec e_0)$ costs $O(N(t)s)$.  Each turn
	at most doubles the entries of the corresponding matrix, so the Euclidean
	reduction involves integers of $O(s)$ bits and costs $O(s^2)$ with
	schoolbook arithmetic; writing the output costs $O(s^2)$.  Thus the total
	running time is $O(s^3)$, without a unit-cost assumption on long addresses
	or rational boundary points.

<!-- theorem-3: body_md -->
\label{prop:V-effective-algorithm}
Given a finite prefix table for $v\in\mathsf V$, there is a finite algorithm which returns
a finite word in flips and the basic shuffle $\sigma_S$ representing $v$.
Apart from finite permutation sorting and word operations, it invokes
$\mathcal A_{\mathsf T}$ three times.
<!-- end theorem-3 -->
<!-- proof-13: body_md -->
Write the input table as
\[
 v=h_{P,Q,\tau},\qquad
 P=\{p_1<\cdots<p_n\},\qquad
 Q=\{q_1<\cdots<q_n\},
\]
so that
\[
 v(p_i\xi)=q_{\tau(i)}\xi.
\]
If $n<3$, repeatedly expand the row with lexicographically
least source prefix until the table has three rows.
Henceforth $P,Q,\tau$ and $n$ refer to this expanded table.
This preliminary step increases the input size by at most
an absolute constant.

Use the canonical maximal prefix code
\[
 C_n=\{c_1<\cdots<c_n\},\qquad
 c_j=1^{j-1}0\quad(1\leq j<n),\qquad
 c_n=1^{n-1}.
\]
For example, $C_3=\{0,10,11\}$ and
$C_4=\{0,10,110,111\}$.  Define the cyclic prefix replacements
\[
 a_P(c_i\xi)=p_i\xi, \qquad
 a_Q(c_i\xi)=q_i\xi.
\]
They belong to $\mathsf T$.  Define also the cylinder permutation
\[
 \pi_\tau(c_i\xi)=c_{\tau(i)}\xi.
\]
Evaluation on each $p_i\mathcal B$ gives the factorization
\begin{equation}\label{eq:V-canonical-factorization}
 v=a_Q\pi_\tau a_P^{-1}.
\end{equation}
Compute
\[
 A_P:=\mathcal A_{\mathsf T}(a_P),\qquad
 A_Q:=\mathcal A_{\mathsf T}(a_Q).
\]

Next let $\rho_n\in\mathsf T$ be the cyclic shift of $C_n$,
\[
 \rho_n(c_i\xi)=c_{i+1}\xi\quad(1\leq i<n),
 \qquad
 \rho_n(c_n\xi)=c_1\xi,
\]
and make the third and final call
\[
 R_n:=\mathcal A_{\mathsf T}(\rho_n).
\]
Let $W_n$ be the admissible Farey prefix whose binary code is
$1^{n-2}$; in particular, $W_3=S$.  The shuffle
\[
 z_n:=\sigma_{W_n}
\]
exchanges the last two cylinders
$c_{n-1}\mathcal B$ and $c_n\mathcal B$ and fixes all other cylinders of
$C_n$.  It can be written using only the basic shuffle $\sigma_S$.  Indeed,
$X_0^{-(n-3)}$ sends $10\mathcal B$ and $11\mathcal B$ to those last two
cylinders, and therefore
\begin{equation}\label{eq:canonical-last-shuffle}
 z_n=X_0^{-(n-3)}\sigma_SX_0^{n-3}.
\end{equation}
Here $X_0$ is already the explicit flip word in
\eqref{eq:flip-generators}.

Let $s_i=(i\ i+1)$ for $1\leq i<n$, acting on the cylinders of $C_n$, and
choose the representative $k_i\in\mathbf{Z}$ of least absolute value satisfying
\[
 k_i\equiv i-(n-1)\pmod n.
\]
If the two representatives have the same absolute value, choose the
nonnegative one.  Thus $2|k_i|\leq n$.
Conjugating \eqref{eq:canonical-last-shuffle} by the appropriate cyclic
shift gives
\begin{equation}\label{eq:canonical-adjacent-transposition}
 s_i=\rho_n^{k_i}z_n\rho_n^{-k_i}.
\end{equation}

Finally, use bubble sort to write
\[
 \tau=s_{i_r}\cdots s_{i_1}.
\]
Here $r=\operatorname{inv}(\tau)\leq n(n-1)/2$.  Equations
\eqref{eq:canonical-last-shuffle} and
\eqref{eq:canonical-adjacent-transposition} give the explicit word
\begin{equation}\label{eq:V-algorithm-output}
 \boxed{
 \begin{aligned}
 v={}&A_Q
 \bigl(R_n^{k_{i_r}}z_nR_n^{-k_{i_r}}\bigr)
 \cdots
 \bigl(R_n^{k_{i_1}}z_nR_n^{-k_{i_1}}\bigr)
 A_P^{-1},\\
 z_n={}&X_0^{-(n-3)}\sigma_SX_0^{n-3}.
 \end{aligned}}
\end{equation}
Every symbol on the right side is now a flip or the shuffle $\sigma_S$.
The factorization \eqref{eq:V-canonical-factorization} proves correctness,
and termination follows from the termination of
$\mathcal A_{\mathsf T}$ and the finiteness of the sorting procedure.
The use of finite cylinder permutations is standard in the prefix-code
description of $\mathsf V$; see
\cite[\S6]{CannonFloydParry1996} and
\cite[\S2.1]{BleakQuick2017}.

	\begin{enumerate}
		\item[\textup{(i)}] \emph{Output bound.} The word produced by the
		flip--shuffle reduction satisfies

			\Lambda_{\mathsf V}(v)
			\leq
			\Lambda_{\mathsf T}(a_P)+\Lambda_{\mathsf T}(a_Q)
			+r\bigl(
			n\Lambda_{\mathsf T}(\rho_n)+10(n-3)+1
			\bigr).
		\end{equation}

		\[
		\Lambda_{\mathsf V}(v)=O(s_{\mathsf V}^{5}).
		\]

		length is $O(s_{\mathsf V}^{7})$.
		
		\item[\textup{(ii)}] \emph{Running time.} A direct implementation which
		writes the fully expanded output runs in
		\[
		O(s_{\mathsf V}^{7})
		\]
		time.  In particular, the flip--shuffle reduction is polynomial-time.
	\end{enumerate}
