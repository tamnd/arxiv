<!-- theorem-1: body_md -->
\label{thm:main}
For every $d\ge2$ and $1\le q\le\infty$, set $r$ by $1/r=1+1/q$ (where $1/\infty=0$). There exist nonnegative $f\in C_c^\infty(\mathbb R^d)$ and $g\in C^\infty(\mathbb R^d)\cap L^q(\mathbb R^d)$ such that
\begin{equation}\label{eq:failure}
\mathcal M_{\mathrm{lac}}(f,g)\notin L^r(\mathbb R^d).
\end{equation}
In particular neither $\mathcal M_{\mathrm{lac}}$ nor $\mathcal M_{\mathrm{full}}$ is bounded from $L^1\times L^q$ to $L^r$. The same assertions hold with the two inputs interchanged.
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
Choose $f\in C_c^\infty(\mathbb R^d)$ nonnegative, supported in $B(0,1)$, with $f\ge1$ on $B(0,1/2)$. Choose $\eta\in C^\infty([0,\infty))$ with $0\le\eta\le1$, $\eta=0$ on $[0,3]$ and $\eta=1$ on $[4,\infty)$.

Suppose first $1\le q<\infty$. Select $\alpha$ with
\begin{equation}\label{eq:alpha}
\frac1q<\alpha\le\frac1r=1+\frac1q
\end{equation}
and define the smooth radial function, set equal to zero near the origin,
\begin{equation}\label{eq:g}
g(u)=\eta(|u|)|u|^{-d/q}(\log|u|)^{-\alpha}.
\end{equation}
Since $\alpha q>1$,
\begin{equation}\label{eq:gq}
\int_{\mathbb R^d}|g(u)|^q\,du\lesssim 1+
\int_4^\infty\frac{d\rho}{\rho(\log\rho)^{\alpha q}}<\infty.
\end{equation}
For $q=\infty$, set $g\equiv1$ and $\alpha=0$.

For $k\ge6$, write $t=2^k$ and consider the disjoint shells and integration balls
\begin{equation}\label{eq:shells}
D_k=\left\{x:\frac t2\le|x|\le\frac{3t}{5}\right\},\qquad
Y_{t,x}=\left\{y:\left|y-\frac xt\right|\le\frac1{2t}\right\}.
\end{equation}
When $x\in D_k$ and $y\in Y_{t,x}$, we have $|x-ty|\le1/2$ and $|y|\le3/5+1/(2t)<5/8$. Hence $f(x-ty)\ge1$ and $\sqrt{1-|y|^2}>3/4$. For every $\omega\in S^{d-1}$, the triangle inequalities imply
\begin{equation}\label{eq:geometry}
\frac{3t}{20}\le\left|x-t\sqrt{1-|y|^2}\,\omega\right|
\le\frac{8t}{5}.
\end{equation}
Since $t\ge64$, the lower endpoint is greater than $4$. The cutoff in \eqref{eq:g} is therefore identically one throughout this region. In the case $q<\infty$, monotonicity of the radial tail gives
\[
g\bigl(x-t\sqrt{1-|y|^2}\,\omega\bigr)
\ge\frac{(8t/5)^{-d/q}}{(\log(8t/5))^{\alpha}}
\gtrsim t^{-d/q}k^{-\alpha}.
\]
The same inequality holds with $q=\infty$, $\alpha=0$. The weight in \eqref{eq:coarea} is at least $(39/64)^{(d-2)/2}$ on $Y_{t,x}$, whose volume equals $|B(0,1)|(2t)^{-d}$. Restricting the integral to that ball thus gives
\begin{equation}\label{eq:lower}
\mathcal M_{\mathrm{lac}}(f,g)(x)\ge\mathcal A_{2^k}(f,g)(x)
\gtrsim t^{-d(1+1/q)}k^{-\alpha}
=2^{-kd/r}k^{-\alpha},\qquad x\in D_k.
\end{equation}
The shells are pairwise disjoint and satisfy
$|D_k|=|B(0,1)|((3/5)^d-(1/2)^d)2^{kd}$. Consequently
\[
\int_{\mathbb R^d}\mathcal M_{\mathrm{lac}}(f,g)(x)^r\,dx
\gtrsim\sum_{k\ge6}|D_k|\,2^{-kd}k^{-\alpha r}
\asymp\sum_{k\ge6}k^{-\alpha r}=\infty,
\]
by \eqref{eq:alpha}; for $q=\infty$ the same calculation uses $r=1$ and $\alpha=0$. Finally, the averages are symmetric in their inputs and $\mathcal M_{\mathrm{full}}\ge\mathcal M_{\mathrm{lac}}$ pointwise.

<!-- proposition-1: body_md -->
\label{prop:smooth}
For every $d\ge2$ and $1\le q\le\infty$ there exist nonnegative $f,g_N\in C_c^\infty(\mathbb R^d)$, $N\ge7$, for which
\begin{equation}\label{eq:quantitative}
\frac{\displaystyle\Big\|\max_{6\le k\le N}|\mathcal A_{2^k}(f,g_N)|\Big\|_{L^r}}
{\|f\|_{L^1}\|g_N\|_{L^q}}\gtrsim_{d,q} N,
\qquad\frac1r=1+\frac1q.
\end{equation}
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
Take the preceding $f$, and let $\chi_N\in C_c^\infty(\mathbb R^d)$ be a radial cutoff with $0\le\chi_N\le1$, equal to one on $4\le|u|\le 2^{N+1}$ and supported in $2<|u|<2^{N+2}$. Set $g_N(u)=\chi_N(u)|u|^{-d/q}$, where $|u|^{-d/\infty}=1$. For finite $q$, polar coordinates yield $\|g_N\|_q^q\lesssim\int_2^{2^{N+2}}d\rho/\rho\lesssim N$; for $q=\infty$ its supremum norm is at most one. For $6\le k\le N$, \eqref{eq:geometry} lies in the region where $\chi_N=1$. The lower bound \eqref{eq:lower} holds with $\alpha=0$, so the $r$th power of the output quasi-norm is at least $c(N-5)$. Dividing the resulting estimate by $\|f\|_1\|g_N\|_q$ gives $(N-5)^{1/r}/N^{1/q}\gtrsim N$, since $1/r-1/q=1$.
<!-- end proof-2 -->
<!-- lemma-1: body_md -->
\label{lem:tail}
Let $1\le q<\infty$, $1/r=1+1/q$, and $\alpha\ge0$. For the function $g_\alpha$ in \eqref{eq:g}, with exponent $\alpha$, one has
\begin{equation}\label{eq:lorentztail}
g_\alpha\in L^{q,s}\ \Longleftrightarrow\ \alpha s>1\quad(0<s<\infty),
\qquad g_\alpha\in L^{q,\infty}\quad(\alpha\ge0).
\end{equation}
For the fixed $f$ of Section~\ref{sec:proof},
\begin{equation}\label{eq:negativetail}
\mathcal M_{\mathrm{lac}}(f,g_\alpha)\notin L^{r,t}\quad\hbox{whenever }0<t<\infty\ \hbox{and }\alpha t\le1.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-3: body_md -->
For large $|u|$ the radial function $g_\alpha$ is decreasing and its decreasing rearrangement satisfies
$g_\alpha^*(v)\asymp v^{-1/q}(\log v)^{-\alpha}$ as $v\to\infty$. Its boundedness and vanishing near the origin handle bounded $v$. This proves \eqref{eq:lorentztail} by direct integration.

For any $\alpha\ge0$, the geometric argument leading to \eqref{eq:lower} remains valid, independently of whether $g_\alpha$ belongs to $L^q$. Put $H=\mathcal M_{\mathrm{lac}}(f,g_\alpha)$ and write $m_k=|D_k|=c_d'2^{kd}$, with $c_d'>0$. By \eqref{eq:lower} and the definition of rearrangement,
$H^*(m_k/2)\gtrsim 2^{-kd/r}k^{-\alpha}$.
Fix $a,b$ with $2^{-d}b<a<b<1/2$. The intervals $I_k=[am_k,bm_k]$ are disjoint, and monotonicity gives
$H^*(v)\gtrsim2^{-kd/r}k^{-\alpha}$ for $v\in I_k$.
Therefore
\[
\int_0^\infty[v^{1/r}H^*(v)]^t\,\frac{dv}{v}
\gtrsim\sum_{k\ge6}k^{-\alpha t},
\]
which diverges under the stated condition.
<!-- end proof-3 -->
<!-- theorem-2: body_md -->
\label{thm:lorentz}
Let $d\ge2$, $d/(d-1)<q<\infty$ and $1/r=1+1/q$. For $0<s,t\le\infty$, each of the maps
\begin{equation}\label{eq:iff}
\mathcal M_{\mathrm{full}},\ \mathcal M_{\mathrm{lac}}:\quad L^1(\mathbb R^d)\times L^{q,s}(\mathbb R^d)
\longrightarrow L^{r,t}(\mathbb R^d)
\end{equation}
is bounded if and only if $s\le t$.
<!-- end theorem-2 -->
<!-- proof-4: body_md -->
When $s\le t$, \eqref{eq:product}--\eqref{eq:holderLorentz}, followed by the embedding $L^{r,s}\hookrightarrow L^{r,t}$, prove both estimates. Conversely, if $t<s<\infty$, choose $1/s<\alpha\le1/t$. Lemma~\ref{lem:tail} provides an input $g_\alpha\in L^{q,s}$ for which the lacunary output fails to belong to $L^{r,t}$. The same construction works for $s=\infty$ and finite $t$ upon choosing $0\le\alpha\le1/t$. Because $\mathcal M_{\mathrm{full}}\ge\mathcal M_{\mathrm{lac}}$ on these nonnegative inputs, both bounds fail.
<!-- end proof-4 -->
<!-- proposition-2: body_md -->
\label{prop:multi}
If $1\le q_j\le\infty$ for $2\le j\le m$ and $1/r=1+\sum_{j=2}^m1/q_j$, there are nonnegative $f_1\in C_c^\infty(\mathbb R^d)$ and $f_j\in C^\infty(\mathbb R^d)\cap L^{q_j}(\mathbb R^d)$ such that
$\mathcal M_{m,\mathrm{lac}}(f_1,\ldots,f_m)\notin L^r(\mathbb R^d)$. Hence every strong-type H\"older estimate with an $L^1$ input fails for the lacunary and full multilinear spherical maximal operators.

<!-- proof-5: body_md -->
Disintegration with respect to the first $d$ coordinates gives
\begin{equation}\label{eq:multislice}
\begin{split}
\int_{S^{md-1}}F(y_1,y')\,d\sigma_{md-1}
={}&C_{m,d}\int_{B(0,1)}(1-|y_1|^2)^{((m-1)d-2)/2}\\[-3pt]
&\times\int_{S^{(m-1)d-1}}F\bigl(y_1,\sqrt{1-|y_1|^2}\,\omega\bigr)
\,d\sigma_{(m-1)d-1}(\omega)\,dy_1,
\end{split}
\end{equation}
where $y'=(y_2,\ldots,y_m)$, $\omega=(\omega_2,\ldots,\omega_m)$, and
\[
C_{m,d}=\frac{\Gamma(md/2)}{\pi^{d/2}\Gamma((m-1)d/2)}.
\]
The normalization follows by the same beta integral as \eqref{eq:coarea}. Fix $f_1=f$ from Section~\ref{sec:proof}. Set $P=\sum_{j=2}^m1/q_j$. For each finite $q_j$, choose $\alpha_j>1/q_j$ so that
\begin{equation}\label{eq:choosemulti}
A:=\sum_{j:q_j<\infty}\alpha_j<1+P=1/r,
\end{equation}
and let $f_j(u)=\eta(|u|)|u|^{-d/q_j}(\log|u|)^{-\alpha_j}$, extended by zero near the origin. If $q_j=\infty$ take $f_j\equiv1$. Condition \eqref{eq:choosemulti} is possible by choosing positive excesses whose sum is smaller than one; all these inputs belong to their stated spaces by \eqref{eq:gq}.

Fix $x\in D_k$, $t=2^k$, and restrict $y_1$ to $Y_{t,x}$ in \eqref{eq:multislice}. The marginal weight is bounded from below and this ball has measure comparable to $t^{-d}$. On the residual sphere impose
\[
E_x=\{\omega:\omega_j\cdot x\le0\text{ for }2\le j\le m\}.
\]
Sign changes $\omega_j\mapsto-\omega_j$ preserve residual surface measure; the boundaries have zero measure and so $\sigma(E_x)=2^{-(m-1)}$. Writing $b=\sqrt{1-|y_1|^2}\le1$, for every $\omega\in E_x$ and $j\ge2$ we have
\[
\frac t2\le|x-tb\omega_j|\le\frac{3t}{5}+t=\frac{8t}{5}.
\]
In particular all finite-$q_j$ cutoffs equal one on this region and
$f_j(x-tb\omega_j)\gtrsim t^{-d/q_j}k^{-\alpha_j}$; the same bound with zero exponents holds for constant inputs. Hence
\[
\mathcal M_{m,\mathrm{lac}}(f_1,\ldots,f_m)(x)\gtrsim
 t^{-d(1+P)}k^{-A}=2^{-kd/r}k^{-A}\quad(x\in D_k).
\]
Integrating its $r$th power over disjoint shells gives $\sum_{k\ge6}k^{-Ar}=\infty$ by \eqref{eq:choosemulti}. Symmetry permits the $L^1$ entry in any position, and domination gives the full maximal conclusion.
