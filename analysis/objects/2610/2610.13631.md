<!-- definition-1: body_md -->
A randomized mechanism \( \mathcal{M} \) satisfies \((\epsilon, \delta)\)-differential privacy if, for all adjacent datasets \( D \) and \( D' \) differing in one element, and for all measurable sets \( S \),
\[
\mathbb{P}[\mathcal{M}(D) \in S] \le e^\epsilon \, \mathbb{P}[\mathcal{M}(D') \in S] + \delta.
\]
<!-- end definition-1 -->
<!-- proposition-1: body_md -->
\label{prop:client_update}
Given the updated global model \(M^*\), the client update \(M_g^*\) is obtained by solving the following convex optimization problem:
\begin{equation}
\label{eq:orig_cs}
\begin{aligned}
\mathcal{J}_g(M,\mu_g): \quad \min_{M_g,\, \lambda_g,\, \{s_n\}} \quad
& \lambda_g\, r_g + \frac{1}{N_g}\sum_{n=1}^{N_g} s_n + \frac{\rho}{2}\,\|M_g - M + \mu_g\|_F^2 \\
\text{s.t.} \quad
& \max_{c \in [K]} \Bigl[ v_{(c)}^\top (M_g x_n - y_n) + 1 \Bigr] - y_n^\top M_g x_n \le s_n, \quad \forall n \in [N_g], \\
& \max_{c \in [K]} \big[ v_{(c)}^\top (M_g x_n - y) + 1 \big] - y^\top M_g x_n - \kappa\,\lambda_g \le s_n, \\
& \qquad \forall n \in [N_g],\ \forall y \in \mathcal{Y},\ y \neq y_n, \\
& \lambda_g \ge \big\| (v_{(i)} - v_{(j)})^\top M_g \big\|_*, \quad \forall i,j \in [K],
\end{aligned}
\end{equation}
where \(\|\cdot\|_*\) denotes the dual of the norm used in the transportation cost \(d(\xi,\xi')\) in \eqref{eq:cost_function}.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
The proof is given in Appendix~\ref{app:proof_prop_client}.
<!-- end proof-1 -->

<!-- lemma-1: body_md -->
\label{lem:sensitivity_Mg}
Let \(D_g\) and \(D_g'\) be two adjacent local datasets that differ in exactly one sample. Assume that the feature vectors satisfy \(\|x_n\|_2 \le L\) for all \(n\), and let \(M_g^t(D_g)\) denote the client update obtained from the proximal linearized subproblem at iteration \(t\). Then the \(\ell_2\)-sensitivity of the released client update satisfies
\[
\max_{D_g \sim D_g'}
\left\|M_g^t(D_g)-M_g^t(D_g')\right\|_F
\le
\frac{2\sqrt{2}\,L}{N_g\left(\rho+\frac{1}{\eta_g}\right)}.
\]

<!-- theorem-1: body_md -->
\label{thm:dp_client_update}
Under the assumptions of Lemmas~\ref{lem:sensitivity_Mg} and
\ref{lem:sensitivity_lambda}, consider the perturbed variables
\[
\widetilde{M}_g^t=M_g^t+\xi_g^t,
\qquad
\widetilde{\lambda}_g^t
=\lambda_g^t+\zeta_g^t,
\]
where \(\xi_g^t\sim\mathcal{MN}_{K,P}(0,I_K,\sigma_{M,g}^2 I_P)\) and
\(\zeta_g^t\sim\mathcal{N}(0,\sigma_{\lambda,g}^2)\), with
\[
\sigma_{M,g} =
\frac{2\sqrt{2}L}
{N_g\left(\rho+\frac{1}{\eta_g}\right)}
\frac{\sqrt{2\ln(1.25/\delta_M)}}{\epsilon_M},
\]
and
\[
\sigma_{\lambda,g} =
\frac{\kappa\nu_g}{N_g}
\frac{\sqrt{2\ln(1.25/\delta_\lambda)}}{\epsilon_\lambda}.
\]
Then \(\widetilde{M}_g^t\) satisfies
\((\epsilon_M,\delta_M)\)-differential privacy and
\(\widetilde{\lambda}_g^t\) satisfies
\((\epsilon_\lambda,\delta_\lambda)\)-differential privacy.

<!-- theorem-3: body_md -->
\label{thm:ergodic_convergence_dp_fed_dro}
Assume that, for each client \(g=1,\dots,G\) and round \(t=1,\dots,T\), the local objective \(h_g^{t}\) is convex in \(z_g\), and that, over the finite training horizon \(T\), its subgradients along the trajectory of Algorithm~\ref{alg:dp_fed_dro} are bounded as \(\|\partial_{M_g}(h_g-R_g)\|_F\le\sqrt{2}\,L\), \(|\partial_{\lambda_g}(h_g-R_g)|\le S:=\max\{r_g,\kappa-r_g\}\), \(\|\partial_{M_g}R_g\|_F\le\sqrt{2}\,C\), and \(|\partial_{\lambda_g}R_g|\le V\), for constants \(L,C,V>0\). Let \(c_M,c_{\lambda,g}>0\) satisfy \(\|M^\star\|_F\le c_M\) and \(|\lambda_g^\star|\le c_{\lambda,g}\) for all \(g\), and let the step sizes be chosen as
\begin{equation}
\eta_g := \frac{c_M}{\sqrt{T\left(2(L+C)^2+\frac{16KPL^2\ln(1.25/\delta_M)}{N_g^2\epsilon_M^2}\right)}},
\qquad
\nu_g := \frac{c_{\lambda,g}}{\sqrt{T\left((S+V)^2+\frac{2\kappa^2\ln(1.25/\delta_\lambda)}{N_g^2\epsilon_\lambda^2}\right)}}.
\label{eq:eta_nu_choice_final}
\end{equation}
Define
\begin{equation}
M_{1,T} := c_M \sum_{g=1}^G \alpha_g \Bigl( 2(L+C)^2 + \frac{16KPL^2\ln(1.25/\delta_M)}{N_g^2\epsilon_M^2} \Bigr)^{1/2} + \sum_{g=1}^G \alpha_g\, c_{\lambda,g} \Bigl( (S+V)^2 + \frac{2\kappa^2\ln(1.25/\delta_\lambda)}{N_g^2\epsilon_\lambda^2} \Bigr)^{1/2},
\label{eq:M1_final}
\end{equation}
and
\begin{equation}
M_2 := \frac{\rho}{2}c_M^2 + \frac{\beta^2}{2\rho}.
\label{eq:M2_final}
\end{equation}
Then, for any \(T\ge 1\) and any \(\beta>0\), we have
\begin{equation}
\mathbb{E}\Biggl[ \frac{1}{T}\sum_{t=1}^{T}\sum_{g=1}^G \alpha_g \Bigl( h_g^{t}(\widetilde z_g^{\,t-1})-h_g^{t}(z_g^\star) \Bigr) + \beta\sum_{g=1}^G \alpha_g \|\bar{\widetilde M}_g^{\,T}-\bar M^{\,T}\|_F \Biggr] \le \frac{M_{1,T}}{\sqrt{T}} + \frac{M_2}{T}.
\label{eq:final_theorem_fixed}
\end{equation}

For this dataset, we examine the privacy--utility tradeoff and convergence behavior. Figure~\ref{fig:privacy_tradeoff} shows the effect of the privacy hyperparameters on model performance, with the cumulative budget $\bar{\epsilon}$ on the top axis. At the smallest per-round budget, $\epsilon=0.01$, the injected noise dominates the signal and all private models perform poorly with large variability across seeds. As $\epsilon$ increases, all three private models improve, since larger values of $\epsilon$ correspond to weaker privacy guarantees and lower injected noise, and DP-Fed DRO and DP-Fed SGD approach the non-private Fed DRO model. DP-Fed ERM, in contrast, saturates early and does not benefit from larger budgets. DP-Fed DRO is the best private method at every budget where the models are informative, with the clearest margin at moderate budgets, $\epsilon\in\{0.05,0.1\}$, where the noise is significant but does not overwhelm the signal. DP-Fed DRO is also markedly more robust to the choice of $\delta$ than DP-Fed SGD, whose performance degrades noticeably as $\delta$ tightens. As a result, the gap between the two is largest at the tightest values of $\delta$. This reflects the ability of the robust formulation to absorb the larger noise level induced by a smaller $\delta$ with little loss in performance.

Given the updated global model \(M^\star\), client \(g\) updates its local model by minimizing the augmented local Lagrangian with respect to \((M_g,\lambda_g)\). By the same problem separability argument used in Proposition~1 of \citet{ibrahim2024federated}, this update is obtained from
\begin{align}
\inf_{M_g}\; \mathcal L_{\rho,g}(M_g,M^\star,\mu_g)
&= \inf_{M_g}\; \sup_{\mathbb Q_g\in \mathcal A^{(g)}_{r_g,1,d}(\Xi)} \mathbb E^{\mathbb Q_g}\!\left[\ell_{\mathrm{CS}}(M_g;\xi)\right] + \frac{\rho}{2}\|M_g-M^\star+\mu_g\|_F^2
\label{eq:prop41_pf_1}\\
&= \inf_{M_g,\lambda_g\ge 0} \bigg\{ \lambda_g r_g + \frac{1}{N_g}\sum_{n=1}^{N_g} \sup_{\xi\in\Xi} \bigl( \ell_{\mathrm{CS}}(M_g;\xi)-\lambda_g d(\xi,\hat\xi_n) \bigr) \notag\\
&\qquad + \frac{\rho}{2}\|M_g-M^\star+\mu_g\|_F^2 \bigg\}.
\label{eq:prop41_pf_2}
\end{align}

Introducing slack variables \(\{s_n\}_{n=1}^{N_g}\), we obtain
\begin{equation}
\label{eq:prop41_pf_3}
\begin{aligned}
&\sup_{\mathbb Q_g\in\mathcal A^{(g)}_{r_g,1,d}(\Xi)} \mathbb E^{\mathbb Q_g}\!\left[\ell_{\mathrm{CS}}(M_g;\xi)\right] \\
&\qquad = \begin{aligned}[t]
\inf_{\lambda_g,\{s_n\}} \quad & \lambda_g r_g + \frac{1}{N_g}\sum_{n=1}^{N_g}s_n \\
\text{s.t.}\quad & \sup_{\xi\in\Xi} \left( \ell_{\mathrm{CS}}(M_g;\xi) - \lambda_g d(\xi,\hat\xi_n) \right) \le s_n, \quad \forall n\in[N_g], \quad \lambda_g\ge 0.
\end{aligned}
\end{aligned}
\end{equation}

Since the inner supremum in \eqref{eq:prop41_pf_3} has the same Wasserstein robust structure as in the centralized multiclass DRO formulation, Theorem~1 in \citet{ibrahim2024wasserstein} applies here with \(M\) replaced by \(M_g\). Therefore, the constraints in \eqref{eq:prop41_pf_3} are equivalently rewritten as the following:
\begin{align}
\ell_{\mathrm{CS}}(M_g;x_n,y_n) &\le s_n,
\qquad \forall n\in[N_g],
\label{eq:prop41_pf_4}
\\
\ell_{\mathrm{CS}}(M_g;x_n,y)-\kappa\lambda_g &\le s_n,
\qquad
\forall n\in[N_g],\ \forall y\in\mathcal Y,\ y\neq y_n,
\label{eq:prop41_pf_5}
\\
\lambda_g &\ge \|(v_{(i)}-v_{(j)})^\top M_g\|_*,
\quad i,j\in[K].
\label{eq:prop41_pf_6}
\end{align}

Substituting \eqref{eq:prop41_pf_4}--\eqref{eq:prop41_pf_6}
into the clientwise augmented Lagrangian subproblem yields the
optimization problem stated in Proposition~\ref{prop:client_update}.

Consider the proximal linearized client subproblem in
\eqref{eq:prox_linearized_lagrangian}. To make the dependence
on the local dataset explicit, denote the nonsmooth term in
\eqref{eq:def_f_local} by $f(M_g,\lambda_g;D_g).$
The first-order optimality condition for the update \(M_g^t(D_g)\)
with respect to \(M_g\) is
\begin{equation}
0 \in \partial_{M_g} f\!\left( \widetilde M_g^{t-1}, \widetilde{\lambda}_g^{t-1}; D_g \right) + \rho\left( M_g^t(D_g)-M^{t-1}+\mu_g^{t-1} \right) + \frac{1}{\eta_g} \left( M_g^t(D_g)-\widetilde M_g^{t-1} \right).
\label{eq:sens_pf_optimality}
\end{equation}
Equivalently,
\begin{equation}
\left( \rho+\frac{1}{\eta_g} \right) M_g^t(D_g) = - \partial_{M_g} f\!\left( \widetilde M_g^{t-1}, \widetilde{\lambda}_g^{t-1}; D_g \right) + \rho\left( M^{t-1}-\mu_g^{t-1} \right) + \frac{1}{\eta_g} \widetilde M_g^{t-1}.
\label{eq:sens_pf_update}
\end{equation}

Let \(D_g\sim D_g'\) be adjacent local datasets differing in
exactly one sample, indexed by \(j\). We condition on the
algorithmic state available at the beginning of iteration
\(t\), including the previous perturbed variables and server
broadcasts. Thus, \(M^{t-1}\), \(\mu_g^{t-1}\),
\(\widetilde M_g^{t-1}\),
\(\widetilde{\lambda}_g^{t-1}\), and
\(\tau_g^{t-1}\) are fixed when comparing the current update
under \(D_g\) and \(D_g'\). In particular, \(\tau_g^{t-1}\)
is computed from the previous perturbed variables according to
Algorithm~\ref{alg:dp_fed_dro}.

Under this fixed algorithmic state, the quadratic terms and the
linearized Bertsekas penalty contribution are identical for
the two adjacent datasets. Moreover, the robust-loss
contributions associated with all unchanged samples are
identical. Hence, the only difference in
\eqref{eq:sens_pf_update} arises from the single changed sample
in the empirical robust-loss average. Therefore,
\begin{equation}
M_g^t(D_g)-M_g^t(D_g') = -\left(\rho+\frac{1}{\eta_g}\right)^{-1} \frac{1}{N_g} \Bigl( \partial_{M_g}\ell_j^{(g)}(D_g) - \partial_{M_g}\ell_j^{(g)}(D_g') \Bigr),
\label{eq:sens_pf_difference}
\end{equation}
where
\(\partial_{M_g}\ell_j^{(g)}(D_g)\) and
\(\partial_{M_g}\ell_j^{(g)}(D_g')\)
denote selected subgradients of the robust-loss contribution
associated with the differing sample under \(D_g\) and \(D_g'\),
respectively.

For the multiclass Crammer--Singer loss, any active \(M_g\)-subgradient has the form
\begin{equation}
\left(v_{(c)}-y\right)x^\top
\label{eq:sens_pf_subgradient_form}
\end{equation}
for an active class \(c\neq y\). At points where multiple
classes are active, the subdifferential is the convex hull of
such matrices.

Using $\|ab^\top\|_F=\|a\|_2\|b\|_2,$
the one-hot encoding property
$\|v_{(c)}-y\|_2=\sqrt{2},$
and the bounded-feature assumption \(\|x\|_2\leq L\), we obtain
\begin{equation}
\left\|
\partial_{M_g}\ell_j^{(g)}(D_g)
\right\|_F
\leq \sqrt{2}L,
\qquad
\left\|
\partial_{M_g}\ell_j^{(g)}(D_g')
\right\|_F
\leq \sqrt{2}L.
\label{eq:sens_pf_subgradient_bound}
\end{equation}

Applying the triangle inequality to
\eqref{eq:sens_pf_difference} and then using
\eqref{eq:sens_pf_subgradient_bound} yields
\begin{align}
\left\| M_g^t(D_g)-M_g^t(D_g') \right\|_F
&\leq \left( \rho+\frac{1}{\eta_g} \right)^{-1} \frac{1}{N_g} \Bigl( \left\| \partial_{M_g}\ell_j^{(g)}(D_g) \right\|_F + \left\| \partial_{M_g}\ell_j^{(g)}(D_g') \right\|_F \Bigr) \notag\\
&\leq \frac{2\sqrt{2}L}{N_g\left(\rho+1/\eta_g\right)}.
\label{eq:sens_pf_final}
\end{align}
Taking the maximum over all adjacent datasets
\(D_g\sim D_g'\) yields the desired sensitivity bound.

For any client \(g\), let \(h_g^t(\cdot):=h_g(\cdot;\tau_g^{t-1})\). Since \(h_g^t\) is convex in \(z_g\), we have the subgradient inequality

\le

+\rho\langle \widetilde M_g^t-M_g,\mu_g^t\rangle_F .
\label{eq:lem1_app_2}

\left\langle h_g^{t\prime}(\widetilde z_g^{\,t-1}),
\,\widetilde z_g^{\,t-1}-\widetilde z_g^t \right\rangle
+\left\langle h_g^{t\prime}(\widetilde z_g^{\,t-1})+\rho\bar\mu_g^t,
\,\widetilde z_g^t-z_g \right\rangle
\\
=
\left\langle
h_g^{t\prime}(\widetilde z_g^{\,t-1})
-c_k\bar\xi_g^t
-c_\ell\bar\zeta_g^t,\,
\widetilde z_g^{\,t-1}-\widetilde z_g^t
\right\rangle
\\
\qquad
-c_k\langle \xi_g^t,\,
M_g-\widetilde M_g^{\,t-1}\rangle_F
-c_\ell\zeta_g^t
\bigl(\lambda_g-\widetilde\lambda_g^{\,t-1}\bigr)
\\
\qquad
+\left\langle
h_g^{t\prime}(\widetilde z_g^{\,t-1})
+\rho\bar\mu_g^t
-c_k\bar\xi_g^t
-c_\ell\bar\zeta_g^t,\,
\widetilde z_g^t-z_g
\right\rangle .
\label{eq:lem1_app_4}
\end{multline}


Using the scaled-dual update
\(\bar\mu_g^t=\bar\mu_g^{t-1}+(\widetilde M_g^t-M^t,0)\)
and the decomposition
\[
(\widetilde M_g^t-M^t,0)
=
(\widetilde M_g^t-M^{t-1},0)
+
(M^{t-1}-M^t,0),
\]
we obtain
\begin{multline}
\left\langle
h_g^{t\prime}(\widetilde z_g^{\,t-1})
-c_k\bar\xi_g^t
-c_\ell\bar\zeta_g^t,\,
\widetilde z_g^{\,t-1}-\widetilde z_g^t
\right\rangle
\\
-c_k\langle \xi_g^t,\,
M_g-\widetilde M_g^{\,t-1}\rangle_F
-c_\ell\zeta_g^t
\bigl(\lambda_g-\widetilde\lambda_g^{\,t-1}\bigr)
\\
+\left\langle
h_g^{t\prime}(\widetilde z_g^{\,t-1})
+\rho\bar\mu_g^t
-c_k\bar\xi_g^t
-c_\ell\bar\zeta_g^t,\,
\widetilde z_g^t-z_g
\right\rangle
\\
=
\left\langle
h_g^{t\prime}(\widetilde z_g^{\,t-1})
-c_k\bar\xi_g^t
-c_\ell\bar\zeta_g^t,\,
\widetilde z_g^{\,t-1}-\widetilde z_g^t
\right\rangle
\\
-c_k\langle \xi_g^t,\,
M_g-\widetilde M_g^{\,t-1}\rangle_F
-c_\ell\zeta_g^t
\bigl(\lambda_g-\widetilde\lambda_g^{\,t-1}\bigr)
\\
+\left\langle
h_g^{t\prime}(\widetilde z_g^{\,t-1})
+\rho \bar\mu_g^{\,t-1}
+\rho(\widetilde M_g^t-M^{t-1},0)
-c_k\bar\xi_g^t
-c_\ell\bar\zeta_g^t,\,
\widetilde z_g^t-z_g
\right\rangle
\\
+\rho\langle
\widetilde M_g^t-M_g,\,
M^{t-1}-M^t
\rangle_F .
\label{eq:lem1_app_5}
\end{multline}


We now upper bound the last term in \eqref{eq:lem1_app_5} by rewriting the inner product as a difference of squared Frobenius norms.  
Using the identity
\begin{equation}
\|A-C\|_F^2 = \|A-B\|_F^2+\|B-C\|_F^2+2\langle A-B,\;B-C\rangle_F,
\label{eq:basic_frob_expand}
\end{equation}
we first apply \eqref{eq:basic_frob_expand} with $
(A,B,C)=(M_g,M^t,M^{t-1}), $
which gives
\begin{equation}
\|M_g-M^{t-1}\|_F^2
=
\|M_g-M^t\|_F^2+\|M^t-M^{t-1}\|_F^2
+2\langle M_g-M^t,\;M^t-M^{t-1}\rangle_F.
\label{eq:lem1_app_sq1}
\end{equation}
Next, apply \eqref{eq:basic_frob_expand} with
$(A,B,C)=(\widetilde M_g^t,M^t,M^{t-1}),$
which yields
\begin{equation}
\|\widetilde M_g^t-M^{t-1}\|_F^2
=
\|\widetilde M_g^t-M^t\|_F^2+\|M^t-M^{t-1}\|_F^2
+2\langle \widetilde M_g^t-M^t,\;M^t-M^{t-1}\rangle_F.
\label{eq:lem1_app_sq2}
\end{equation}
Subtracting \eqref{eq:lem1_app_sq1} from \eqref{eq:lem1_app_sq2}, the common term
\(\|M^t-M^{t-1}\|_F^2\) cancels, and we obtain
\begin{equation}
2\langle \widetilde M_g^t-M_g,\;M^{t-1}-M^t\rangle_F
=
\|M_g-M^{t-1}\|_F^2-\|M_g-M^t\|_F^2
+\|\widetilde M_g^t-M^t\|_F^2-\|\widetilde M_g^t-M^{t-1}\|_F^2.
\label{eq:lem1_app_sq3}
\end{equation}
Multiplying by \(\rho/2\) and dropping the final nonpositive term gives
\begin{equation}
\rho\langle \widetilde M_g^t-M_g,\;M^{t-1}-M^t\rangle_F
\le
\frac{\rho}{2}
\Big(
\|M_g-M^{t-1}\|_F^2-\|M_g-M^t\|_F^2
\Big)
+\frac{\rho}{2}\|\widetilde M_g^t-M^t\|_F^2.
\end{equation}
Finally, using the scaled dual update
\[
\mu_g^t-\mu_g^{t-1}=\widetilde M_g^t-M^t,
\]
we conclude that
\begin{equation}
\rho\langle \widetilde M_g^t-M_g,\;M^{t-1}-M^t\rangle_F
\le
\frac{\rho}{2}
\Big(
\|M_g-M^{t-1}\|_F^2-\|M_g-M^t\|_F^2
\Big)
+\frac{\rho}{2}\|\mu_g^t-\mu_g^{t-1}\|_F^2 .
\label{eq:lem1_app_6}
\end{equation}

Set $c_k:=\rho+\frac{1}{\eta_g},$ and $c_\ell:=\frac{1}{\nu_g}.$
We now split the second-to-last term in \eqref{eq:lem1_app_5}
into its \(M\)- and \(\lambda\)-components. Since $\bar\mu_g^{\,t-1}=(\mu_g^{t-1},0),\bar\xi_g^t=(\xi_g^t,0), $ and $ \bar\zeta_g^t=(0,\zeta_g^t),$

Recall that
\begin{multline}
\widehat L_{\rho,g}^t(M_g,\lambda_g)
=
f(\widetilde M_g^{t-1},\widetilde\lambda_g^{t-1})
+
\left\langle \partial_{M_g} f(\widetilde M_g^{t-1},\widetilde\lambda_g^{t-1}),\, M_g-\widetilde M_g^{t-1}\right\rangle_F
\\
+
\left\langle \partial_{\lambda_g} f(\widetilde M_g^{t-1},\widetilde\lambda_g^{t-1}),\, \lambda_g-\widetilde\lambda_g^{t-1}\right\rangle
+
\lambda_g r_g
+
\frac{\rho}{2}\left\|M_g-M^{t-1}+\mu_g^{t-1}\right\|_F^2
\\
+
\frac{1}{2\eta_g}\left\|M_g-\widetilde M_g^{t-1}\right\|_F^2
+
\frac{1}{2\nu_g}\left(\lambda_g-\widetilde\lambda_g^{t-1}\right)^2 .
\end{multline}

The term $f(\widetilde M_g^{t-1},\widetilde\lambda_g^{t-1})$ is constant with respect to $(M_g,\lambda_g)$ and therefore does not affect coercivity.

We first expand the terms depending on $M_g$. We have
\begin{equation}
\left\langle \partial_{M_g} f(\widetilde M_g^{t-1},\widetilde\lambda_g^{t-1}),\, M_g-\widetilde M_g^{t-1}\right\rangle_F
=
\left\langle \partial_{M_g} f(\widetilde M_g^{t-1},\widetilde\lambda_g^{t-1}),\, M_g\right\rangle_F
\\
-
\left\langle \partial_{M_g} f(\widetilde M_g^{t-1},\widetilde\lambda_g^{t-1}),\, \widetilde M_g^{t-1}\right\rangle_F ,
\end{equation}
and
\begin{equation}
\frac{1}{2\eta_g}\left\|M_g-\widetilde M_g^{t-1}\right\|_F^2
=
\frac{1}{2\eta_g}\|M_g\|_F^2
-
\frac{1}{\eta_g}\left\langle M_g,\widetilde M_g^{t-1}\right\rangle_F
\\
+
\frac{1}{2\eta_g}\left\|\widetilde M_g^{t-1}\right\|_F^2 ,
\end{equation}
while
\begin{equation}
\frac{\rho}{2}\left\|M_g-M^{t-1}+\mu_g^{t-1}\right\|_F^2
=
\frac{\rho}{2}\|M_g\|_F^2
-
\rho\left\langle M_g,\, M^{t-1}-\mu_g^{t-1}\right\rangle_F
\\
+
\frac{\rho}{2}\left\|M^{t-1}-\mu_g^{t-1}\right\|_F^2 .
\end{equation}

Therefore, all terms independent of \(M_g\) can be grouped into \(B_g^t\), and
\(\widehat L_{\rho,g}^t(M_g,\lambda_g)\) can be written as
\begin{equation}
\left(\frac{1}{2\eta_g}+\frac{\rho}{2}\right)\|M_g\|_F^2
+
\Bigl\langle
\partial_{M_g} f(\widetilde M_g^{t-1},\widetilde\lambda_g^{t-1})
-
\frac{1}{\eta_g}\widetilde M_g^{t-1}
\\
-
\rho\bigl(M^{t-1}-\mu_g^{t-1}\bigr),
\, M_g
\Bigr\rangle_F
+
B_g^t.
\end{equation}

Next, we expand the terms depending on $\lambda_g$. We have
\begin{equation}
\left\langle \partial_{\lambda_g} f(\widetilde M_g^{t-1},\widetilde\lambda_g^{t-1}),\, \lambda_g-\widetilde\lambda_g^{t-1}\right\rangle
=
\partial_{\lambda_g} f(\widetilde M_g^{t-1},\widetilde\lambda_g^{t-1})\,\lambda_g
\\
-
\partial_{\lambda_g} f(\widetilde M_g^{t-1},\widetilde\lambda_g^{t-1})\,\widetilde\lambda_g^{t-1},
\end{equation}
and
\begin{equation}
\frac{1}{2\nu_g}\left(\lambda_g-\widetilde\lambda_g^{t-1}\right)^2
=
\frac{1}{2\nu_g}\lambda_g^2
-
\frac{1}{\nu_g}\widetilde\lambda_g^{t-1}\lambda_g
+

Define
\[
H(v_g^t):=(\mu_g^t,\,0,\,-\mu_g^t,\,-(\widetilde M_g^t-M^t)).
\]
By direct expansion of the stacked inner product, we have
\begin{equation}
\langle v_g^t-v_g,\;H(v_g^t)\rangle
=
\langle \mu_g^t,\widetilde M_g^t-M_g\rangle_F
-\langle \mu_g^t,M^t-M\rangle_F
+\langle \mu_g-\mu_g^t,\widetilde M_g^t-M^t\rangle_F .
\label{eq:thm6_pf_1}
\end{equation}

To account for the update of $\tau_g$, write
\[
h_g^{t}(\cdot):=h_g(\cdot\,;\tau_g^{t-1})
\]
for the local objective used at round $t$, which is convex in $z_g$ for each fixed $\tau_g^{t-1}$.
Since $H$ is linear and monotone
\begin{equation}
\sum_{g=1}^G \alpha_g \langle \bar v_g^T-v_g,\;H(\bar v_g^T)\rangle
\le
\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g
\langle v_g^t-v_g,\;H(v_g^t)\rangle .
\label{eq:thm6_pf_3}
\end{equation}

Adding the per-round objective gaps to both sides of \eqref{eq:thm6_pf_3} and using
\eqref{eq:thm6_pf_1}, we obtain
\begin{multline}
\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g
\Bigl(h_g^{t}(\widetilde z_g^{\,t-1})-h_g^{t}(z_g)\Bigr)
+\rho\sum_{g=1}^G \alpha_g
\langle \bar v_g^T-v_g,\;H(\bar v_g^T)\rangle
\\
\le
\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g
\Bigl(
h_g^{t}(\widetilde z_g^{\,t-1})-h_g^{t}(z_g)
+\rho \langle v_g^t-v_g,\;H(v_g^t)\rangle
\Bigr)
\\
=
\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g
\Bigl(
h_g^{t}(\widetilde z_g^{\,t-1})-h_g^{t}(z_g)
+\rho\langle \mu_g^t,\widetilde M_g^t-M_g\rangle_F
\\
\qquad
-\rho\langle \mu_g^t,M^t-M\rangle_F
+\rho\langle \mu_g-\mu_g^t,\widetilde M_g^t-M^t\rangle_F
\Bigr)
\\
=
\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g
\Bigl(
h_g^{t}(\widetilde z_g^{\,t-1})-h_g^{t}(z_g)
+\rho\langle \mu_g^t,\widetilde M_g^t-M_g\rangle_F
\\
\qquad
+\rho\langle \mu_g-\mu_g^t,\widetilde M_g^t-M^t\rangle_F
\Bigr),
\label{eq:thm6_pf_4}
\end{multline}
where the last equality uses $\sum_{g=1}^G\alpha_g\mu_g^t=0$.


Applying Lemma 3 to each summand on the right-hand side of
\eqref{eq:thm6_pf_4} yields
\begin{multline}
\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g
\Bigl(h_g^{t}(\widetilde z_g^{\,t-1})-h_g^{t}(z_g)\Bigr)
+\rho\sum_{g=1}^G \alpha_g
\langle \bar v_g^T-v_g,\;H(\bar v_g^T)\rangle
\\
\le
\frac{1}{T}\sum_{t=1}^T
\sum_{g=1}^G \alpha_g
\Bigg[
\frac{\eta_g}{2}
\left\|
[h_g^{t\prime}(\widetilde z_g^{\,t-1})]_M
-\left(\rho+\frac{1}{\eta_g}\right)\xi_g^t
\right\|_F^2
+\frac{\nu_g}{2}
\left|
[h_g^{t\prime}(\widetilde z_g^{\,t-1})]_\lambda
-\frac{1}{\nu_g}\zeta_g^t
\right|^2
\\
\qquad
-\left(\rho+\frac{1}{\eta_g}\right)
\langle \xi_g^t,\,
M_g-\widetilde M_g^{\,t-1}\rangle_F
-\frac{1}{\nu_g}\,\zeta_g^t
\bigl(\lambda_g-\widetilde\lambda_g^{\,t-1}\bigr)
\\
\qquad
+\frac{1}{2\eta_g}
\Bigl(
\|M_g-\widetilde M_g^{\,t-1}\|_F^2
-\|M_g-\widetilde M_g^t\|_F^2
\Bigr)
\\
\qquad
+\frac{1}{2\nu_g}
\Bigl(
(\lambda_g-\widetilde\lambda_g^{\,t-1})^2
-(\lambda_g-\widetilde\lambda_g^t)^2
\Bigr)
\\
\qquad
+\frac{\rho}{2}
\Bigl(
\|M_g-M^{t-1}\|_F^2
-\|M_g-M^t\|_F^2
\Bigr)
+\frac{\rho}{2}
\Bigl(
\|\mu_g-\mu_g^{t-1}\|_F^2
-\|\mu_g-\mu_g^t\|_F^2
\Bigr)
\Bigg].
\label{eq:thm6_pf_5}
\end{multline}


Since the step sizes \(\eta_g\) and \(\nu_g\) are constant in \(t\), the
difference-of-squares terms in \eqref{eq:thm6_pf_5} telescope exactly, and
dropping the nonpositive terms
\(-\|M_g-\widetilde M_g^T\|_F^2\), \(-(\lambda_g-\widetilde\lambda_g^T)^2\),
\(-\|M_g-M^T\|_F^2\), and \(-\|\mu_g-\mu_g^T\|_F^2\) gives
\begin{multline}
\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g
\Bigl(h_g^{t}(\widetilde z_g^{\,t-1})-h_g^{t}(z_g)\Bigr)
+\rho\sum_{g=1}^G \alpha_g
\langle \bar v_g^T-v_g,\;H(\bar v_g^T)\rangle
\\
\le
\sum_{g=1}^G \alpha_g
\Biggl[
\frac{1}{T}\sum_{t=1}^T \frac{\eta_g}{2}
\left\|
[h_g^{t\prime}(\widetilde z_g^{\,t-1})]_M
-\left(\rho+\frac{1}{\eta_g}\right)\xi_g^t
\right\|_F^2
+\frac{1}{T}\sum_{t=1}^T \frac{\nu_g}{2}
\left|
[h_g^{t\prime}(\widetilde z_g^{\,t-1})]_\lambda
-\frac{1}{\nu_g}\zeta_g^t
\right|^2
\\
\qquad
-\frac{1}{T}\sum_{t=1}^T
\left(\rho+\frac{1}{\eta_g}\right)
\langle \xi_g^t,\;M_g-\widetilde M_g^{\,t-1}\rangle_F
-\frac{1}{T}\sum_{t=1}^T
\frac{1}{\nu_g}\,\zeta_g^t\bigl(\lambda_g-\widetilde\lambda_g^{\,t-1}\bigr)
\\
\qquad
+\frac{1}{2T\eta_g}\|M_g-\widetilde M_g^0\|_F^2
+\frac{1}{2T\nu_g}(\lambda_g-\widetilde\lambda_g^0)^2
+\frac{\rho}{2T}\|M_g-M^0\|_F^2
+\frac{\rho}{2T}\|\mu_g-\mu_g^0\|_F^2
\Biggr].
\label{eq:thm6_pf_6}
\end{multline}


Next, choose the primal comparison point to be an optimal solution,
\[
z_g=z_g^\star=(M_g^\star,\lambda_g^\star),
\qquad
M_g^\star=M^\star \text{ for all } g,
\qquad
M=M^\star .
\]
With this choice, \eqref{eq:thm6_pf_6} becomes
\begin{multline}
\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g
\Bigl(h_g^{t}(\widetilde z_g^{\,t-1})-h_g^{t}(z_g^\star)\Bigr)
+\rho\sum_{g=1}^G \alpha_g
\langle \bar v_g^T-v_g,\;H(\bar v_g^T)\rangle
\\
\le
\sum_{g=1}^G \alpha_g
\Biggl[
\frac{1}{T}\sum_{t=1}^T \frac{\eta_g}{2}
\left\|
[h_g^{t\prime}(\widetilde z_g^{\,t-1})]_M
-\left(\rho+\frac{1}{\eta_g}\right)\xi_g^t
\right\|_F^2
+\frac{1}{T}\sum_{t=1}^T \frac{\nu_g}{2}
\left|
[h_g^{t\prime}(\widetilde z_g^{\,t-1})]_\lambda
-\frac{1}{\nu_g}\zeta_g^t
\right|^2
\\
\qquad
-\frac{1}{T}\sum_{t=1}^T
\left(\rho+\frac{1}{\eta_g}\right)
\langle \xi_g^t,\;M^\star-\widetilde M_g^{\,t-1}\rangle_F
-\frac{1}{T}\sum_{t=1}^T
\frac{1}{\nu_g}\,\zeta_g^t\bigl(\lambda_g^\star-\widetilde\lambda_g^{\,t-1}\bigr)
\\
\qquad
+\frac{1}{2T\eta_g}\|M^\star-\widetilde M_g^0\|_F^2
+\frac{1}{2T\nu_g}(\lambda_g^\star-\widetilde\lambda_g^0)^2
+\frac{\rho}{2T}\|M^\star-M^0\|_F^2
+\frac{\rho}{2T}\|\mu_g-\mu_g^0\|_F^2
\Biggr],
\label{eq:thm6_pf_7}
\end{multline}
where \(v_g=(M^\star,\lambda_g^\star,M^\star,\mu_g)\).




Take the initialization
\[
M^0=0,
\qquad
\widetilde M_g^0=0,
\qquad
\widetilde\lambda_g^0=0,
\qquad
\mu_g^0=0
\quad \text{for all } g.
\]


Then \eqref{eq:thm6_pf_7} simplifies to
\begin{multline}
\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g
\Bigl(h_g^{t}(\widetilde z_g^{\,t-1})-h_g^{t}(z_g^\star)\Bigr)
+\rho\sum_{g=1}^G \alpha_g
\langle \bar v_g^T-v_g,\;H(\bar v_g^T)\rangle
\\
\le
\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g \frac{\eta_g}{2}
\left\|
[h_g'(\widetilde z_g^{\,t-1})]_M
-\left(\rho+\frac{1}{\eta_g}\right)\xi_g^t
\right\|_F^2
+\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g \frac{\nu_g}{2}
\left|
[h_g'(\widetilde z_g^{\,t-1})]_\lambda
-\frac{1}{\nu_g}\zeta_g^t
\right|^2
\\
\qquad
-\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g
\left(\rho+\frac{1}{\eta_g}\right)
\langle \xi_g^t,\;M^\star-\widetilde M_g^{\,t-1}\rangle_F
-\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g
\frac{1}{\nu_g}\,\zeta_g^t\bigl(\lambda_g^\star-\widetilde\lambda_g^{\,t-1}\bigr)
\\
\qquad
+\frac{\|M^\star\|_F^2}{T}\sum_{g=1}^G \alpha_g \frac{1}{2\eta_g}
+\frac{1}{T}\sum_{g=1}^G \alpha_g \frac{(\lambda_g^\star)^2}{2\nu_g}
+\frac{\rho}{2T}\|M^\star\|_F^2
+\frac{\rho}{2T}\sum_{g=1}^G \alpha_g \|\mu_g\|_F^2 .
\label{eq:thm6_pf_8}
\end{multline}


We now simplify the term \(\rho \langle \bar v_g^T-v_g,\;H(\bar v_g^T)\rangle\).
For a comparison point \(v_g=(M,\lambda_g,M,\mu_g)\), we have
\begin{multline}
\langle \bar v_g^T-v_g,\;H(\bar v_g^T)\rangle
=
\langle \bar\mu_g^T,\bar{\widetilde M}_g^T-M\rangle_F
-\langle \bar\mu_g^T,\bar M^T-M\rangle_F
-\langle \bar\mu_g^T-\mu_g,\bar{\widetilde M}_g^T-\bar M^T\rangle_F
\\
=
\langle \mu_g,\bar{\widetilde M}_g^T-\bar M^T\rangle_F.
\label{eq:thm6_pf_9}
\end{multline}
Therefore, the left-hand side of \eqref{eq:thm6_pf_8} equals
\begin{equation}
\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g
\Bigl(h_g^{t}(\widetilde z_g^{\,t-1})-h_g^{t}(z_g^\star)\Bigr)
+\rho\sum_{g=1}^G \alpha_g
\langle \mu_g,\bar{\widetilde M}_g^T-\bar M^T\rangle_F .
\label{eq:thm6_pf_10}
\end{equation}

Now restrict the comparison dual variable to the ball
\[
\{\mu_g:\|\mu_g\|_F\le \beta/\rho\}.
\]
Let
\[
d_g^T:=\bar{\widetilde M}_g^T-\bar M^T.
\]
By Cauchy--Schwarz,
\begin{equation}
\max_{\|\mu_g\|_F\le \beta/\rho}
\rho\langle \mu_g,d_g^T\rangle_F
\le
\rho \max_{\|\mu_g\|_F\le \beta/\rho}
\|\mu_g\|_F\,\|d_g^T\|_F
=
\beta \|d_g^T\|_F.
\label{eq:thm6_pf_11}
\end{equation}


The bound is tight by choosing
\[
\hat\mu_g = \frac{\beta}{\rho}\frac{d_g^T}{\|d_g^T\|_F}
\qquad \text{when } d_g^T\neq 0,
\]
hence
\begin{equation}
\max_{\|\mu_g\|_F\le \beta/\rho}
\rho\langle \mu_g,\bar{\widetilde M}_g^T-\bar M^T\rangle_F
=
\beta \|\bar{\widetilde M}_g^T-\bar M^T\|_F.
\label{eq:thm6_pf_12}
\end{equation}

Consequently, using \eqref{eq:thm6_pf_10} and \eqref{eq:thm6_pf_12},
\begin{multline}
\max_{\substack{\|\mu_g\|_F\le \beta/\rho\\ g=1,\dots,G}}
\Biggl[
\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g
\Bigl(h_g^{t}(\widetilde z_g^{\,t-1})-h_g^{t}(z_g^\star)\Bigr)
+\rho\sum_{g=1}^G \alpha_g
\langle \bar v_g^T-v_g,\;H(\bar v_g^T)\rangle
\Biggr]
\\
=
\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g
\Bigl(h_g^{t}(\widetilde z_g^{\,t-1})-h_g^{t}(z_g^\star)\Bigr)
+\beta\sum_{g=1}^G \alpha_g
\|\bar{\widetilde M}_g^T-\bar M^T\|_F .
\label{eq:thm6_pf_13}
\end{multline}
We now maximize the last term on the right-hand side of \eqref{eq:thm6_pf_8}
over the same ball. Since \(\|\mu_g\|_F\le \beta/\rho\) and \(\sum_{g=1}^G\alpha_g=1\),
\begin{equation}
\max_{\substack{\|\mu_g\|_F\le \beta/\rho\\ g=1,\dots,G}}
\frac{\rho}{2T}\sum_{g=1}^G \alpha_g \|\mu_g\|_F^2
\le
\frac{\rho}{2T}\sum_{g=1}^G \alpha_g \frac{\beta^2}{\rho^2}
=
\frac{\beta^2}{2\rho T}.
\label{eq:thm6_pf_14}
\end{equation}


Since \eqref{eq:thm6_pf_8} holds for every \(\mu_g\), we may take the maximum
over \(\|\mu_g\|_F\le\beta/\rho\), \(g=1,\dots,G\), on both sides. Using
\eqref{eq:thm6_pf_13} for the left-hand side and \eqref{eq:thm6_pf_14} for the
only \(\mu_g\)-dependent term on the right-hand side, we obtain
\begin{multline}
\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g
\Bigl(h_g^{t}(\widetilde z_g^{\,t-1})-h_g^{t}(z_g^\star)\Bigr)
+\beta\sum_{g=1}^G \alpha_g
\|\bar{\widetilde M}_g^T-\bar M^T\|_F
\\
\le
\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g \frac{\eta_g}{2}
\left\|
[h_g^{t\prime}(\widetilde z_g^{\,t-1})]_M
-\left(\rho+\frac{1}{\eta_g}\right)\xi_g^t
\right\|_F^2
+\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g \frac{\nu_g}{2}
\left|
[h_g^{t\prime}(\widetilde z_g^{\,t-1})]_\lambda
-\frac{1}{\nu_g}\zeta_g^t
\right|^2
\\
\qquad
-\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g
\left(\rho+\frac{1}{\eta_g}\right)
\langle \xi_g^t,\;M^\star-\widetilde M_g^{\,t-1}\rangle_F
-\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g
\frac{1}{\nu_g}\,\zeta_g^t\bigl(\lambda_g^\star-\widetilde\lambda_g^{\,t-1}\bigr)
\\
\qquad
+\frac{\|M^\star\|_F^2}{T}\sum_{g=1}^G \alpha_g \frac{1}{2\eta_g}
+\frac{1}{2T}\sum_{g=1}^G \alpha_g
\frac{(\lambda_g^\star)^2}{\nu_g}
+\frac{\rho}{2T}\|M^\star\|_F^2
+\frac{\beta^2}{2\rho T}.
\label{eq:thm6_pf_15}
\end{multline}


We now take expectations on both sides of \eqref{eq:thm6_pf_15}. By construction
of Algorithm~2, for each \(g\) and \(t\), the perturbations \(\xi_g^t\) and
\(\zeta_g^t\) are drawn independently of each other and of all past iterates, and satisfy
\[
\mathbb{E}[\xi_g^t]=0,\qquad
\mathbb{E}\|\xi_g^t\|_F^2=KP\,\sigma_{M,g}^2,\qquad
\mathbb{E}[\zeta_g^t]=0,\qquad
\mathbb{E}(\zeta_g^t)^2=\sigma_{\lambda,g}^2 .
\]
Since \(M^\star-\widetilde M_g^{\,t-1}\) and \(\lambda_g^\star-\widetilde\lambda_g^{\,t-1}\)
depend only on past iterates, while \(\xi_g^t\) and \(\zeta_g^t\) are independent of the
past and have zero mean, it follows that
\begin{equation}
\mathbb{E}\langle \xi_g^t,\; M^\star-\widetilde M_g^{\,t-1}\rangle_F=0,
\qquad
\mathbb{E}\bigl[\zeta_g^t\,(\lambda_g^\star-\widetilde\lambda_g^{\,t-1})\bigr]=0.
\label{eq:thm6_pf_16}
\end{equation}

Recall that, at round \(t\),
\[
h_g^{t}(z_g)=\lambda_g r_g+\frac{1}{N_g}\sum_{n=1}^{N_g}L_n(M_g,\lambda_g)+R_g(M_g,\lambda_g;\tau_g^{t-1}),
\qquad z_g=(M_g,\lambda_g),
\]
where
\[
L_n(M_g,\lambda_g)
:=
\max_{y\in\mathcal Y}
\Bigl(
\ell_{\mathrm{CS}}(M_g;x_n,y)-\kappa\lambda_g\,\mathbf{1}_{\{y\neq y_n\}}
\Bigr),
\]
and
\[
R_g(M_g,\lambda_g;\tau_g^{t-1})
:=
\frac{1}{2\gamma}
\Bigl(
\max\{0,\tau_g^{t-1}+\gamma(\phi(M_g)-\lambda_g)\}^2-(\tau_g^{t-1})^2
\Bigr).
\]

Next, define
\[
a_g^{t-1}:=[h_g^{t\prime}(\widetilde z_g^{\,t-1})]_M,
\qquad
b_g^{t-1}:=[h_g^{t\prime}(\widetilde z_g^{\,t-1})]_\lambda,
\qquad
c_g:=\rho+\frac{1}{\eta_g}.
\]
By direct evaluation of the selected subgradients of the robust loss term,
\[
\|\partial_{M_g}L_n(M_g,\lambda_g)\|_F \le \sqrt{2}\,L,
\qquad
\partial_{\lambda_g}L_n(M_g,\lambda_g)\in[-\kappa,0].
\]
Since \([h_g^{t\prime}(\widetilde z_g^{\,t-1})]_\lambda
= r_g+\frac{1}{N_g}\sum_{n=1}^{N_g}\partial_{\lambda_g}L_n+\partial_{\lambda_g}R_g\),
the first two terms together lie in \([r_g-\kappa,\,r_g]\), so that
\[
\Bigl|\,r_g+\frac{1}{N_g}\sum_{n=1}^{N_g}\partial_{\lambda_g}L_n\Bigr|
\le S:=\max\{r_g,\ \kappa-r_g\}.
\]


Moreover, since
\(\partial_{\lambda_g}R_g=-\max\{0,\tau_g^{t-1}+\gamma(\phi(M_g)-\lambda_g)\}\) and
\(\partial_{M_g}R_g=\max\{0,\tau_g^{t-1}+\gamma(\phi(M_g)-\lambda_g)\}\,\partial\phi(M_g)\),
using \(\phi(M_g)\le\sqrt{2}\,\|M_g\|_F\) and \(\|\partial\phi(M_g)\|_F\le C_\phi\) gives
\[
\|\partial_{M_g}R_g(M_g,\lambda_g;\tau_g^{t-1})\|_F
\le
\bigl(|\tau_g^{t-1}|+\gamma\sqrt{2}\,\|M_g\|_F+\gamma[-\lambda_g]_+\bigr)C_\phi,
\]
and
\[
|\partial_{\lambda_g}R_g(M_g,\lambda_g;\tau_g^{t-1})|
\le
|\tau_g^{t-1}|+\gamma\sqrt{2}\,\|M_g\|_F+\gamma[-\lambda_g]_+ .
\]

Further, Lemma~\ref{lem:bounded_traj} shows that the trajectory generated by Algorithm~2 remains finite
over the training horizon, so that the penalty subgradients evaluated along it are
bounded. Accordingly, we assume that these bounds are uniform: there exist constants
\(C,V>0\) such that
\[
\bigl\|\partial_{M_g}R_g(\widetilde M_g^{\,t-1},\widetilde\lambda_g^{\,t-1};\tau_g^{t-1})\bigr\|_F
\le \sqrt{2}\,C,
\qquad
\bigl|\partial_{\lambda_g}R_g(\widetilde M_g^{\,t-1},\widetilde\lambda_g^{\,t-1};\tau_g^{t-1})\bigr|
\le V,
\]
for all \(g=1,\dots,G\) and \(t=1,\dots,T\).

Therefore,
\begin{equation}
\|a_g^{t-1}\|_F \le \sqrt{2}\,(L+C),
\qquad
|b_g^{t-1}| \le S+V.
\label{eq:thm6_pf_17}
\end{equation}

Expanding the square and using that \(\xi_g^t\) is zero-mean and independent of
\(a_g^{t-1}\), which depends only on past iterates, gives
\begin{equation}
\mathbb E\|a_g^{t-1}-c_g\xi_g^t\|_F^2
=
\mathbb E\|a_g^{t-1}\|_F^2
-2c_g\,\mathbb E\langle a_g^{t-1},\xi_g^t\rangle_F
+c_g^2\,\mathbb E\|\xi_g^t\|_F^2
=
\mathbb E\|a_g^{t-1}\|_F^2 + c_g^2 KP\,\sigma_{M,g}^2 ,
\end{equation}
and likewise, since \(\zeta_g^t\) is zero-mean and independent of \(b_g^{t-1}\),
\begin{equation}
\mathbb E\Bigl|b_g^{t-1}-\frac{1}{\nu_g}\zeta_g^t\Bigr|^2
=
\mathbb E|b_g^{t-1}|^2+\frac{\sigma_{\lambda,g}^2}{\nu_g^2} .
\end{equation}
Using \eqref{eq:thm6_pf_17}, we obtain
\begin{equation}
\mathbb E\|a_g^{t-1}-c_g\xi_g^t\|_F^2
\le
2(L+C)^2 + c_g^2 KP\,\sigma_{M,g}^2,
\qquad
\mathbb E\Bigl|b_g^{t-1}-\frac{1}{\nu_g}\zeta_g^t\Bigr|^2
\le
(S+V)^2+\frac{\sigma_{\lambda,g}^2}{\nu_g^2} .
\label{eq:thm6_pf_18}
\end{equation}


Taking expectations in \eqref{eq:thm6_pf_15} and using
\eqref{eq:thm6_pf_16}, \eqref{eq:thm6_pf_17}, and \eqref{eq:thm6_pf_18}, we get
\begin{multline}
\mathbb E\Biggl[
\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g
\Bigl(h_g^{t}(\widetilde z_g^{\,t-1})-h_g^{t}(z_g^\star)\Bigr)
+\beta\sum_{g=1}^G \alpha_g
\|\bar{\widetilde M}_g^T-\bar M^T\|_F
\Biggr]
\\
\le
\sum_{g=1}^G \alpha_g
\frac{\eta_g}{2}
\Biggl(
2(L+C)^2+\left(\rho+\frac{1}{\eta_g}\right)^2 KP\,\sigma_{M,g}^2
\Biggr)
+
\sum_{g=1}^G \alpha_g
\frac{\nu_g}{2}
\Biggl(
(S+V)^2+\frac{\sigma_{\lambda,g}^2}{\nu_g^2}
\Biggr)
\\
+\frac{\|M^\star\|_F^2}{T}\sum_{g=1}^G \alpha_g \frac{1}{2\eta_g}
+\frac{1}{2T}\sum_{g=1}^G \alpha_g
\frac{(\lambda_g^\star)^2}{\nu_g}
+\frac{\rho}{2T}\|M^\star\|_F^2
+\frac{\beta^2}{2\rho T}.
\label{eq:thm6_pf_19}
\end{multline}

Now substitute the DP-calibrated Gaussian variances from Theorem~1,
\[
\sigma_{M,g}^2
=
\frac{8L^2}{N_g^2\left(\rho+\frac{1}{\eta_g}\right)^2}
\frac{2\ln(1.25/\delta_M)}{\epsilon_M^2},
\qquad
\sigma_{\lambda,g}^2
=
\frac{\kappa^2\nu_g^2}{N_g^2}
\frac{2\ln(1.25/\delta_\lambda)}{\epsilon_\lambda^2}.
\]
Then
\begin{equation}
\left(\rho+\frac{1}{\eta_g}\right)^2 KP\,\sigma_{M,g}^2
=
KP\cdot \frac{8L^2}{N_g^2}\cdot \frac{2\ln(1.25/\delta_M)}{\epsilon_M^2},
\qquad
\frac{\sigma_{\lambda,g}^2}{\nu_g^2}
=
\frac{\kappa^2}{N_g^2}\cdot\frac{2\ln(1.25/\delta_\lambda)}{\epsilon_\lambda^2}.
\label{eq:thm6_pf_20}
\end{equation}

Substituting \eqref{eq:thm6_pf_20} into \eqref{eq:thm6_pf_19} yields
\begin{multline}
\mathbb E\Biggl[
\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g
\Bigl(h_g^{t}(\widetilde z_g^{\,t-1})-h_g^{t}(z_g^\star)\Bigr)
+\beta\sum_{g=1}^G \alpha_g
\|\bar{\widetilde M}_g^T-\bar M^T\|_F
\Biggr]
\\
\le
\sum_{g=1}^G \alpha_g
\frac{\eta_g}{2}
\left(
2(L+C)^2
+
KP\cdot \frac{8L^2}{N_g^2}\cdot \frac{2\ln(1.25/\delta_M)}{\epsilon_M^2}
\right)
+
\sum_{g=1}^G \alpha_g
\frac{\nu_g}{2}
\left(
(S+V)^2
+
\frac{\kappa^2}{N_g^2}\cdot\frac{2\ln(1.25/\delta_\lambda)}{\epsilon_\lambda^2}
\right)
\\
+\frac{\|M^\star\|_F^2}{T}\sum_{g=1}^G \alpha_g \frac{1}{2\eta_g}
+\frac{1}{2T}\sum_{g=1}^G \alpha_g
\frac{(\lambda_g^\star)^2}{\nu_g}
+\frac{\rho}{2T}\|M^\star\|_F^2
+\frac{\beta^2}{2\rho T}.
\label{eq:thm6_pf_21}
\end{multline}

For constants \(c_M,c_{\lambda,g}>0\) satisfying \(\|M^\star\|_F\le c_M\) and
\(|\lambda_g^\star|\le c_{\lambda,g}\) for all \(g\), define the \(M_g\)-block and
\(\lambda_g\)-block step sizes as
\begin{equation}
\begin{aligned}
\eta_g
&:=
\frac{c_M}
{
\sqrt{
T\left(
2(L+C)^2
+
KP\cdot \frac{8L^2}{N_g^2}\cdot \frac{2\ln(1.25/\delta_M)}{\epsilon_M^2}
\right)
}
},
\\
\nu_g
&:=
\frac{c_{\lambda,g}}
{
\sqrt{
T\left(
(S+V)^2
+
\frac{\kappa^2}{N_g^2}\cdot\frac{2\ln(1.25/\delta_\lambda)}{\epsilon_\lambda^2}
\right)
}
}.
\end{aligned}
\label{eq:thm6_pf_22}
\end{equation}
For convenience, define
\begin{equation}
D_{M,g}
:=
\sqrt{
2(L+C)^2
+
KP\cdot \frac{8L^2}{N_g^2}\cdot \frac{2\ln(1.25/\delta_M)}{\epsilon_M^2}
},
\qquad
D_{\lambda,g}
:=
\sqrt{
(S+V)^2
+
\frac{\kappa^2}{N_g^2}\cdot\frac{2\ln(1.25/\delta_\lambda)}{\epsilon_\lambda^2}
}.
\label{eq:thm6_pf_23}
\end{equation}

Substituting \eqref{eq:thm6_pf_22} into \eqref{eq:thm6_pf_21} and using
\(\|M^\star\|_F^2\le c_M^2\) and \((\lambda_g^\star)^2\le c_{\lambda,g}^2\), we obtain
\begin{multline}
\mathbb E\Biggl[
\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g
\Bigl(h_g^{t}(\widetilde z_g^{\,t-1})-h_g^{t}(z_g^\star)\Bigr)
+\beta\sum_{g=1}^G \alpha_g
\|\bar{\widetilde M}_g^T-\bar M^T\|_F
\Biggr]
\\
\le
\frac{1}{\sqrt T}\sum_{g=1}^G \alpha_g\, c_M D_{M,g}
+
\frac{1}{\sqrt T}\sum_{g=1}^G \alpha_g\, c_{\lambda,g} D_{\lambda,g}
+
\frac{\rho}{2T}c_M^2
+\frac{\beta^2}{2\rho T}.
\label{eq:thm6_pf_24}
\end{multline}


Defining
\begin{equation}
M_{1,T}
:=
\sum_{g=1}^G \alpha_g\, c_M D_{M,g}
+
\sum_{g=1}^G \alpha_g\, c_{\lambda,g} D_{\lambda,g},
\qquad
M_2
:=
\frac{\rho}{2}c_M^2+\frac{\beta^2}{2\rho},
\label{eq:thm6_pf_26}
\end{equation}
we can rewrite (122) as
\begin{multline}
\mathbb E\Biggl[
\frac{1}{T}\sum_{t=1}^T \sum_{g=1}^G \alpha_g
\Bigl(h_g^{t}(\widetilde z_g^{\,t-1})-h_g^{t}(z_g^\star)\Bigr)
+\beta\sum_{g=1}^G \alpha_g
\|\bar{\widetilde M}_g^T-\bar M^T\|_F
\Biggr]
\\
\le
\frac{1}{\sqrt T}
\Biggl[
\sum_{g=1}^G \alpha_g\, c_M D_{M,g}
+
\sum_{g=1}^G \alpha_g\, c_{\lambda,g} D_{\lambda,g}
\Biggr]
+
\frac{1}{T}
\Bigl(
\frac{\rho}{2}c_M^2+\frac{\beta^2}{2\rho}
\Bigr)
=
\frac{M_{1,T}}{\sqrt T}+\frac{M_2}{T}.
\label{eq:thm6_pf_27}
\end{multline}
This proves the claimed convergence bound.
