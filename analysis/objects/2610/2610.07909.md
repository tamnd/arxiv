<!-- lemma-1: body_md -->
\label{lem:composition}
Suppose that $\tau = \theta(c,\delta) \in \Theta_\delta$ and $\theta' = \theta(d,\Delta)\in \Theta_\Delta$. Then 
\[ \mathcal A_\tau\circ \mathcal A_{\theta'} = \mathcal A_{\theta}  \]
for $\theta = \theta(c+\delta\circ d, \delta\circ \Delta)$. Consequently, $L_{\tau} L_{\theta'} = L_\theta$ and $$L_\tau(T_{\theta'}) = T_\theta.$$
Moreover, 
\[ \mathcal{A}_\tau\big(N_{(\delta_1 \delta_2 R)^{-1}}(H_{\theta'})\big) = N_{R^{-1}}(H_{\theta}) \]
and 
\[ \mathcal{A}_\tau\big(\theta' \times \mathbb{R} \big) = \theta \times \mathbb{R}. \]
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
The identity $\mathcal A_\tau\circ \mathcal A_{\theta'} = \mathcal A_{\theta}$ is just a restatement of \eqref{eqn:composition}. In particular, this yields 
\[ M_\tau M_{\theta'} = M_{\theta}. \]
Taking inverse transpose, we obtain $L_{\tau} L_{\theta'} = L_\theta$. 
Applying both sides to $Q_1$, we obtain $L_\tau(T_{\theta'}) = T_\theta$. 

For $(\xi,\eta,s)\in\mathbb{R}^3$, write $(u,v,w)=\mathcal{A}_\tau(\xi,\eta,s)$. Then $u=c_1+\delta_1\xi$, $v=c_2+\delta_2 \eta$ and 
\[ w= uv +  \delta_1\delta_2 (s-\xi\eta).\]
Therefore,
\[ (\xi,\eta) \in \theta' \text{ and } |s- \xi\eta| \leq (\delta_1\delta_2 R)^{-1},\]
if and only if 
\[ (u,v) \in \theta \text{ and } |w- uv| \leq R^{-1}. \]
Thus, $\mathcal{A}_\tau\big(N_{(\delta_1 \delta_2 R)^{-1}}(H_{\theta'})\big) = N_{R^{-1}}(H_{\theta})$. 

The identity $\mathcal{A}_\tau\big(\theta' \times \mathbb{R} \big) = \theta \times \mathbb{R}$ follows from the first two components of $\mathcal{A}_\tau$.

\label{lem:trivialestimate}

\begin{align*}
    \int_X \bigg|\sum_{T\in\mathbb T}f_T\bigg|^2w
    &\lesssim_N

For each $T$, the Cauchy--Schwarz inequality implies
\[
    \frac1{|T|}\int_X w\chi_T
    \lesssim \|w\|_{L^2_{\mathrm{avg}}(T)},

where we used $\int_{X}\chi_T \leq \| \chi_T \|_{L^1} \lesssim|T|$.

    &\lesssim
    R
    \left(
\sum_{\theta\in \Theta_{\mathrm{square}}(R^{-1})}\sum_{T\in \mathbb{T}_\theta}
        \|f_T\|_2^2\|w\|_{L^2_{\mathrm{avg}}(T)}^2
    \right)^{1/2}\left(\sum_{\theta\in \Theta_{\mathrm{square}}(R^{-1})}\sum_{T\in \mathbb{T}_\theta}\|f_T\|_2^2\right)^{1/2}.\\
    &\lesssim
    R 
    \Bigg(
        \sum_{ \theta\in\Theta_{\mathrm{square}}(R^{-1})}
        \sum_{T\in\mathbb T_\theta}
        \|f_T\|_{L^2}^2\|w\|_{L^2_{\mathrm{avg}}(T)}^2
    \Bigg)^{1/2}\|f\|_{L^2},
\end{aligned}
\]
where we used \eqref{eqn:orthogonalWavePacket} and Plancherel's theorem in the second inequality. Note that $\Theta_{\mathrm{square}}(R^{-1})\subset\Theta(R^{-1})$. Thus $C(R)\lesssim  R$ as claimed.

<!-- proof-6: body_md -->
Recall that \(\tau_1,\tau_2\in\Theta_{\mathrm{square}}(K^{-2})\) for some dyadic $K\in [1,R^{1/3}]$. Then $c(\tau_j) \in (K^{-1}\mathbb{Z} \times K^{-1}\mathbb{Z})\cap [-1,1)^2$ for $j=1,2$.
Write
\[
    d=c(\tau_2)-c(\tau_1) \in (K^{-1}\mathbb{Z} \times K^{-1}\mathbb{Z}).
\]
Let $\delta=(\delta_1,\delta_2)$ be a pair of dyadic numbers such that 
\[ |d_j|<\delta_j\leq2|d_j|,\qquad j=1,2. \]
We set
\[ \Delta=\delta^{-1}\circ (K^{-1},K^{-1}).\] 
Then, for $j=1,2$,  
\[\Delta_j=\delta_j^{-1}K^{-1} \leq 1/4\]
as $\delta_j^{-1}K^{-1}$ is dyadic and $\delta_j^{-1}K^{-1} < |d_j|^{-1}K^{-1} \leq 1/2$, where the second inequality follows from the assumption that $\tau_1,\tau_2$ are $K^{-1}$-transverse.

Define $g$ by 
\[ \widehat{g}(\zeta) = \delta_1\delta_2 \widehat{f}\bigl(\mathcal{A}_\tau(\zeta)\bigr)1_{[-1,1]^2\times \mathbb{R}}(\zeta), \;\; \text{ where } \; \tau = \theta(c(\tau_1), \delta).\]
Then 
\[ \operatorname{supp} \widehat{g} \subset N_{R_0^{-1}}(H), \;\; \text{ where } \;\; R_0 = \delta_1\delta_2 R \]
by \Cref{lem:composition}. Moreover,  
\[ \tau_j \times \mathbb{R} = \mathcal{A}_\tau (\sigma_j \times \mathbb{R}),  \]
where
\[ \sigma_j=\theta\big(\delta^{-1}\circ( c(\tau_j)-c(\tau_1)), \Delta \big).\]


Let 
\[ c:=c(\sigma_2) = \delta^{-1}\circ d \in \Delta_1\mathbb{Z} \times \Delta_2 \mathbb{Z},\] 
so that 
\[ \sigma_1 = \theta(0,\Delta), \; \text{ and } \; \sigma_2 = \theta(c,\Delta).\]
Since $\delta_j\leq 2|d_j|$,
\[ |c_j| \geq 1/2, \;\;\; j=1,2. \]
Clearly, $\sigma_1 \in \Theta_{\Delta}$. Since $\delta_j>|d_j|$, we have  $c \in (-1,1)^2$, thus $\sigma_2 \in \Theta_{\Delta}$. 
In particular, $\sigma_j \subset [-1,1]^2$ for each $j=1,2$, which yields
\begin{align*}
\widehat{g_{\sigma_j}}(\zeta)
    =
   \widehat{g}(\zeta)  1_{\sigma_j \times \mathbb{R}}(\zeta) 
    = \delta_1\delta_2  \widehat{f}(\mathcal A_\tau(\zeta))  1_{\tau_j \times \mathbb{R}}\bigl(
    \mathcal{A}_\tau(\zeta)\bigr) = \delta_1\delta_2  \widehat{f_{\tau_j}}(\mathcal A_\tau(\zeta)). 
\end{align*}
Plancherel's theorem gives
\begin{equation}\label{eq:bilinear-affine-L2}
    \|f_{\tau_j}\|_2=
     \|g_{\sigma_j}\|_2.
\end{equation}
Moreover, by Fourier inversion, 
\[
    |f_{\tau_j}(x)|
    =
    \delta_1\delta_2 |g_{\sigma_j}(L_\tau^{-1} x)|.
\] 
Thus, a change of variables yields
\begin{equation}\label{eq:bilinear-affine-integral}
    \int_{\mathbb R^3}|f_{\tau_1}f_{\tau_2}|w
    =
    \int_{\mathbb R^3}|g_{\sigma_1}g_{\sigma_2}|\widetilde w, \text{ where }  \widetilde w:=w\circ L_\tau.
\end{equation}

We now apply \Cref{prop:bilinear-separated} to $g_{\sigma_1}$ and $g_{\sigma_2}$ at scale $R_0=\delta_1\delta_2R$. The only remaining hypothesis to check is $\Delta_j \geq R_0^{-1/2}$, or equivalently, $\delta_j^{-2}K^{-2} \delta_1\delta_2 R \geq 1$. Since $\delta_j \in (2K^{-1}, 2]$, we have
\[ \delta_j^{-2} K^{-2} \delta_1\delta_2 R \geq K^{-3} R \geq 1. \]
In the second inequality, we used the assumption that $K\leq R^{1/3}$. 
Thus, by \Cref{prop:bilinear-separated},  
\[
\begin{split}
\int_{\mathbb R^3}|g_{\sigma_1}g_{\sigma_2}|\widetilde w
\lesssim_\varepsilon{}&
R_0^\varepsilon
\prod_{j=1}^2
\Bigg(
\sum_{\substack{
\theta'\in\Theta_{\mathrm{square}}(R_0^{-1}) \\
\theta' \subset \sigma_j}}
\sum_{T'\in\mathbb T_{\theta'}}
\|g_{T'}\|_2^2
\|\widetilde w\|_{L^2_{\mathrm{avg}}(T')}^2
\Bigg)^{1/4}
\|g_{\sigma_j}\|_2^{1/2}.
\end{split}
\]
By applying \eqref{eq:weight-scaling}
(with $K_1K_2=(\delta_1\delta_2)^{-1}$),
\eqref{eq:bilinear-affine-L2} and
\eqref{eq:bilinear-affine-integral}, we obtain
\[
\begin{split}
\int_{\mathbb R^3}|f_{\tau_1}f_{\tau_2}|w
\lesssim_\varepsilon{}&
R^\varepsilon
\prod_{j=1}^2
\Bigg(
\sum_{\substack{
\theta\in\Theta(R^{-1}) \\ \theta\subset \tau_j
}}
\sum_{T\in\mathbb T_\theta}
\|f_T\|_2^2
\|w\|_{L^2_{\mathrm{avg}}(T)}^2
\Bigg)^{1/4}
\|f_{\tau_j}\|_2^{1/2}.
\end{split}
\] This proves \Cref{prop:bilinear}.

<!-- theorem-5: body_md -->
\label{lem:bilinear-refined-decoupling}
Let \(\Omega_1,\Omega_2 \subset [-1,1]^2\) be transverse squares.
For \(j=1,2\), let
\[
    f_j=\sum_{T\in\mathbb W_j}f_T
\]
be a sum of scale-\(R\) wave packets, where 
\[\mathbb{W}_j \subset \bigcup_{\substack{\theta \in \Theta_{\mathrm{square}}(R^{-1}) \\ \theta \subset \Omega_j}} \mathbb{T}_\theta.\]
Let $B_R$ denote a ball of radius $R$, and let $X\subset B_R$ be a union of pairwise disjoint
 balls \(Q\) of radius $R^{1/2}$.
Then, 
\begin{equation}\label{eq:bilinear-refined-decoupling}
 \int_X|f_1f_2|^2
 \lessapprox
 \prod_{j=1}^2
 \bigg( M_j
    \sum_{T\in\mathbb W_j}
    \|f_T\|_{L^4(\mathbb R^3)}^4
 \bigg)^{1/2},
\end{equation}
where  
\[ M_j = \max_{Q\subset X} M(Q,\mathbb{W}_j),\qquad j=1,2. \]
Here $A \lessapprox B$ denotes $A\leq C_\varepsilon R^{\varepsilon} B$  for some constant $C_\varepsilon>0$ for any $\varepsilon>0$.

<!-- proposition-6: body_md -->
\label{prop:extension-equivalence}
Estimates \eqref{eq:extension-MTthm}  and  \eqref{eqn:equivL2} are equivalent.
<!-- end proposition-6 -->
<!-- proof-7: body_md -->
We first verify the implication  \eqref{eqn:equivL2} $\implies$ 
\eqref{eq:extension-MTthm}. Choose
\(\psi\in C_c^\infty((-1,1))\) such that
\[
    |\check\psi(t)|\gtrsim1
    \qquad\text{whenever }|t|\leq1.
\]
Given \(g\in L^2([-1,1]^2)\), define \(F\) by
\[
    \widehat F(\xi,\eta,\tau)
    :=
    R\psi\bigl(R(\tau-\xi\eta)\bigr)g(\xi,\eta).
\]
Since \(\operatorname{supp}\psi\subset(-1,1)\), we have
\(\operatorname{supp}\widehat F\subset N_{R^{-1}}(H)\). Fourier inversion in the
\(\tau\)-variable gives
\[
    F(x)
    =
    \check\psi(x_3/R)Eg(x).
\]
For \(x\in B_R\), we have \(|x_3/R|\leq1\), and hence
\[
    |Eg(x)|\lesssim|F(x)|.
\]
Plancherel's theorem and the change of variables
\(s=R(\tau-\xi\eta)\) give
\[
\begin{aligned}
 \|F\|_2^2=\int
   R^2\left|\psi\bigl(R(\tau-\xi\eta)\bigr)\right|^2
   |g(\xi,\eta)|^2\,d\xi\,d\eta d\tau =R\|\psi\|_2^2\|g\|_2^2.
\end{aligned}
\]
Applying \eqref{eqn:equivL2} to \(F\), we obtain
\[
\begin{split}
\int_{B_R} |Eg|^2w \lesssim \int |F|^2w \lesssim_\varepsilon
 R^\varepsilon
 \max_{\substack{
       \theta\in\Theta(R^{-1})\\
       F_\theta\neq0}}
 \sup_{T\parallel T_\theta}
 \left(
    \frac1{|T|}\int_Tw^2
 \right)^{1/2}
 \|F\|_2^2.
\end{split}
\]
Note that the condition \(F_\theta\neq0\) implies \(g_\theta\neq0\). Additionally, $|T|^{-1/2} \| F\|_{2}^2 \sim \| g\|_{2}^2$ since $|T|\sim R^2$ and $\| \psi\|_2\sim 1$.
Therefore,
\[
 \max_{\substack{
       \theta\in\Theta(R^{-1})\\
       F_\theta\neq0}}
 \sup_{T\parallel T_\theta}
 \left(
    \frac1{|T|}\int_Tw^2
 \right)^{1/2} \| F\|_{2}^2
 \lesssim
 \max_{\substack{
       \theta\in\Theta(R^{-1})\\
       g_\theta\neq0}}
 \sup_{T\parallel T_\theta}
 \left(
   \int_Tw^2
 \right)^{1/2} \| g\|_{2}^2.
\]
Consequently, we obtain \eqref{eq:extension-MTthm}.

Conversely, assume \eqref{eq:extension-MTthm}, and let \(F\) satisfy
\(\operatorname{supp}\widehat F\subset N_{R^{-1}}(H)\). For
\(|s|\leq R^{-1}\), define
\[
    g_s(\xi,\eta)
    :=
    \widehat F(\xi,\eta,\xi\eta+s).
\]
Fourier inversion gives
\[
    F(x)
    =
    \int_{|s|\leq R^{-1}}
    e^{2\pi isx_3}Eg_s(x)\,ds.
\]
By Cauchy--Schwarz in \(s\),
\[
    |F(x)|^2
    \lesssim
    R^{-1}
    \int_{|s|\leq R^{-1}}|Eg_s(x)|^2\,ds.
\]
Integrating against \(w\) and applying Fubini's theorem, we get
\[
    \int_{B_R}|F|^2w
    \lesssim
    R^{-1}
    \int_{|s|\leq R^{-1}}
    \int_{B_R}|Eg_s(x)|^2w(x)\,dx\,ds.
\]
We apply \eqref{eq:extension-MTthm} to each \(g_s\). We first compare
the frequency supports of \(g_s\) and \(F\). Let
\(\theta=\theta(c,(\delta_1,\delta_2))\in\Theta(R^{-1})\), and
suppose that \((g_s)_\theta\neq0\). Then there exists $(\xi,\eta) \in \theta$ such that 
\[
    \widehat F(\xi,\eta,\xi\eta+s)\neq0.
\]
Hence \(F_\theta\neq 0\).
Therefore,
\[
 \max_{\substack{
       \theta\in\Theta(R^{-1})\\
       (g_s)_\theta\neq0}}
 \sup_{T\parallel T_\theta}
 \left(\int_Tw^2\right)^{1/2}
 \leq
 \max_{\substack{
       \theta\in\Theta(R^{-1})\\
       F_\theta\neq0}}
 \sup_{T\parallel T_\theta}
 \left(\int_Tw^2\right)^{1/2}.
\]
Applying \eqref{eq:extension-MTthm} to each \(g_s\), and using the
preceding bound, we obtain
\[
\begin{split}
 \int_{B_R}|F|^2w
 \lesssim_\varepsilon{}R^\varepsilon
 R^{-1}
 \max_{\substack{
       \theta\in\Theta(R^{-1})\\
       F_\theta\neq0}}
 \sup_{T\parallel T_\theta}
 \left(
    \int_Tw^2
 \right)^{1/2}
 \int_{|s|\leq R^{-1}}\|g_s\|_2^2\,ds.
\end{split}
\]
By the definition of \(g_s\),
\[
\begin{aligned}
 \int_{|s|\leq R^{-1}}\|g_s\|_2^2\,ds
 &=
 \int_{|s|\leq R^{-1}}
 \int_{[-1,1]^2}
 \left|
    \widehat F(\xi,\eta,\xi\eta+s)
 \right|^2
 \,d\xi\,d\eta ds\\
 &=
 \int_{\substack{
       (\xi,\eta)\in[-1,1]^2\\
       |\tau-\xi\eta|\leq R^{-1}}}
 |\widehat F(\xi,\eta,\tau)|^2
 \,d\xi\,d\eta d\tau=
 \|\widehat F\|_2^2
 =
 \|F\|_2^2.
\end{aligned}
\]
Here the third equality follows from
\(\operatorname{supp}\widehat F\subset N_{R^{-1}}(H)\), and the last equality
follows from Plancherel's theorem.
Finally, we recall that $|T| \sim R^2$. Combining these estimates, we obtain
\[
 \int_{B_R}|F|^2w
 \lesssim_\varepsilon
 R^\varepsilon
 \max_{\substack{
       \theta\in\Theta(R^{-1})\\
       F_\theta\neq0}}
 \sup_{T\parallel T_\theta}
 \left(
    \frac1{|T|}\int_Tw^2
 \right)^{1/2}
 \|F\|_2^2.
\]
The passage from this local estimate to the estimate on
\(\mathbb R^3\) is the same as at the end of the proof of
\Cref{prop:bilinear-separated}. This proves
\eqref{eqn:equivL2}.
