<!-- theorem-1: body_md -->
\label{vol com thm}
Let $\Sigma$ be an $n$-dimensional compact connected smooth manifold ($n \geq 3$) with weakly convex boundary $\partial\Sigma$, and let $0 \leq \gamma \leq \frac{n-1}{n-2}$, $\lambda > 0$. Assume that there exists a positive smooth function $\Phi \in C^{\infty}(\Sigma)$ satisfying
\begin{align*}
\begin{cases}
\Phi Ric^{\Sigma}_{-} - \gamma\Delta_{\Sigma}\Phi \geq (n-1)\lambda \Phi \,\, {\rm in} \,\, \Sigma, \\
\frac{\partial \Phi}{\partial\nu} = 0, \,\, {\rm on} \,\, \partial\Sigma,
\end{cases}
\end{align*}
where $Ric^{\Sigma}_{-}$ denotes the smallest eigenvalue of the Ricci curvature of $\Sigma$ and $\nu$ is the unit conormal vector of $\partial\Sigma$ in $\Sigma$. If $\Phi$ attains its minimum on $\partial\Sigma$, then
\begin{align} \label{vol com ine}
{\rm Vol}(\Sigma) \leq \lambda^{-\frac{n}{2}}{\rm Vol}(\mathbb{S}^{n}_{+}).
\end{align}
Moreover, if equality holds in (\ref{vol com ine}), then $\Sigma$ is isometric to the round hemisphere of radius $\lambda^{-\frac{1}{2}}$.

<!-- theorem-2: body_md -->
\label{main thm}
Let $M$ be an $(n+1)$-dimensional Riemannian manifold $(n \geq 3)$ with weakly convex boundary $\partial M$. Assume that $Ric^{M} \geq 0$ and $\inf_{M}R^{M} > 0$. Let $\Sigma$ be a compact connected two-sided, properly embedded, volume-minimizing free boundary hypersurface satisfying $Ric^{\Sigma} = \frac{R^{\Sigma}}{n}g_{\Sigma}$. Then
\begin{align} \label{ine vol}
{\rm Vol}(\Sigma)^{\frac{2}{n}}\inf_{M}R^{M} \leq n(n-1){\rm Vol}(\mathbb{S}^{n}_{+})^{\frac{2}{n}},
\end{align}
where $\mathbb{S}^{n}_{+}$ denotes the round hemisphere. Moreover, if equality holds in (\ref{ine vol}), then, up to scaling, $\Sigma$ is isometric to $\mathbb{S}^{n}_{+}$ and a neighborhood of $\Sigma$ in $M$ isometric to $\mathbb{S}^{n}_{+} \times (-\epsilon,\epsilon)$ with the product metric.

<!-- proposition-1: body_md -->
\label{prop 3.1}
Let $M$ be an $(n+1)$-dimensional Riemannian manifold with weakly convex boundary $\partial M$. Assume that $Ric^{M} \geq 0$ and $\inf_{M}R^{M} > 0$. Let $\Sigma$ be a compact connected two-sided, properly embedded, stable minimal free boundary hypersurface satisfying $Ric^{\Sigma} = \frac{R^{\Sigma}}{n}g_{\Sigma}$. Then
\begin{align} \label{ine 3.1}
{\rm Vol}(\Sigma)^{\frac{2}{n}}\inf_{M}R^{M} \leq n(n-1){\rm Vol}(\mathbb{S}^{n}_{+})^{\frac{2}{n}},
\end{align}
where $\mathbb{S}^{n}_{+}$ denotes the unit round hemisphere. Moreover, if equality holds in (\ref{ine 3.1}), then, up to scaling, $\Sigma$ is isometric to $\mathbb{S}^{n}_{+}$.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Since $\Sigma$ is a stable minimal free boundary hypersurface, we have
\begin{align*}
\begin{aligned}
0 \leq& \int_{\Sigma}\vert \nabla^{\Sigma}\varphi \vert^{2} dv - \int_{\Sigma}\varphi^{2}\left(Ric^{M}(N,N) + \vert h^{\Sigma} \vert^{2}\right) dv \\
&- \int_{\partial\Sigma}h^{\partial M}(N,N)\varphi^{2}ds
\end{aligned}
\end{align*}
for every $\varphi \in C^{\infty}(\Sigma)$. Taking $\varphi = 1$, we obtain
\begin{align*}
\begin{aligned}
0 \leq& -\int_{\Sigma}\left(Ric^{M}(N,N) + \vert h^{\Sigma} \vert^{2}\right)dv - \int_{\partial\Sigma} h^{\partial M}(N,N) ds \\
\leq& - \int_{\Sigma}\vert h^{\Sigma} \vert^{2} dv \leq 0,
\end{aligned}
\end{align*}
where we used $Ric^{M} \geq 0$ and the weakly convexity of $\partial M$. Hence, $Ric^{M}(N,N) = 0$ and $h^{\Sigma} = 0$ on $\Sigma$, and $h^{\partial M} (N,N) = 0$ along $\partial\Sigma$. Moreover, since $\Sigma$ meets $\partial M$ orthogonally, we have
\begin{align*}
h^{\partial\Sigma}(V,W) = h^{\partial M}(V,W)
\end{align*}
for all $V,W \in T(\partial\Sigma)$. This implies that 
\begin{align} \label{ine 3.2}
h^{\partial\Sigma} \geq 0 \quad \text{along } \partial\Sigma. 
\end{align}
On the other hand, the Gauss equation gives
\begin{align*}
Ric^{M}(N,N) = \frac{1}{2}\left(R^{M} - R^{\Sigma} + H^{2} - \vert h^{\Sigma} \vert^{2}\right).
\end{align*}
Since $\Sigma$ is minimal, $Ric^{M}(N,N) = 0$, and $h^{\Sigma} = 0$, we obtain $R^{M} = R^{\Sigma}$ on $\Sigma$. Therefore, using $Ric^{\Sigma} = \frac{R^{\Sigma}}{n}g_{\Sigma}$, the smallest eigenvalue $Ric^{\Sigma}_{-}$ of the Ricci curvature of $\Sigma$ satisfies
\begin{align*}
Ric^{\Sigma}_{-} = \frac{R^{\Sigma}}{n} = \frac{R^{M}}{n} \geq \frac{\inf_{M}R^{M}}{n}.
\end{align*}
Set $\lambda:= \frac{\inf_{M}R^{M}}{n(n-1)} > 0$. Then 
\begin{align*}
Ric^{\Sigma}_{-} \geq (n-1)\lambda.
\end{align*}
Taking $\Phi = 1$ and $\gamma = 0$, we have
\begin{align*}
\Phi Ric^{\Sigma}_{-} - \gamma\Delta_{\Sigma}\Phi = Ric^{\Sigma}_{-} \geq (n-1)\lambda\Phi
\end{align*}
in $\Sigma$, and
\begin{align*}
\frac{\partial\Phi}{\partial\nu} = 0
\end{align*}
along $\partial\Sigma$. Moreover, since $\Phi = 1$, it attains its minimum on $\partial\Sigma$. From (\ref{ine 3.2}), all the hypotheses of Theorem \ref{vol com thm} are satisfied. Hence,
\begin{align*}
{\rm Vol}(\Sigma) \leq \lambda^{-\frac{n}{2}}{\rm Vol}(\mathbb{S}^{n}_{+}).
\end{align*}
Equivalently,
\begin{align*}
\begin{aligned}
{\rm Vol}(\Sigma)^{\frac{2}{n}} &\leq \lambda^{-1}{\rm Vol}(\mathbb{S}^{n}_{+})^{\frac{2}{n}} \\
&= \frac{n(n-1)}{\inf_{M}R^{M}}{\rm Vol}(\mathbb{S}^{n}_{+})^{\frac{2}{n}},
\end{aligned}
\end{align*}
which proves (\ref{ine 3.1}). Finally, suppose that equality holds in (\ref{ine 3.1}). Then equality holds in (\ref{vol com ine}) of Theorem \ref{vol com thm}. By the rigidity statement in Theorem \ref{vol com thm}, $\Sigma$ is isometric to the round hemisphere of radius $\lambda^{-\frac{1}{2}}$. Therefore, up to scaling, $\Sigma$ is isometric to $\mathbb{S}^{n}_{+}$.

<!-- proposition-2: body_md -->
\label{foliation}
Let $M$ be an $(n+1)$-dimensional Riemannian manifold with boundary $\partial M$. Let $\Sigma$ be a compact connected two-sided, properly embedded, free boundary hypersurface. If $\Sigma$ is totally geodesic, $Ric^{M}(N,N) = 0$ on $\Sigma$, and $h^{\partial M}(N,N) = 0$ along $\partial\Sigma$, then there exists $\epsilon > 0$ and a smooth function $w: \Sigma \times (-\epsilon,\epsilon) \rightarrow \mathbb{R}$ such that, for every $t \in (-\epsilon,\epsilon)$, the set $\Sigma_{t} = \left\{\phi(x,w(x,t)) \vert x \in \Sigma \right\}$ is a free boundary hypersurface with constant mean curvature $H(t)$. Moreover,
\begin{align*}
w(x,0) = 0, \,\, \int_{\Sigma}\left( w(x,t) - t \right) dv = 0, \,\, {\rm and} \,\, \frac{\partial w}{\partial t}(x,0) = 1 \,\,.
\end{align*}
In particular, for some smaller $\epsilon$, the family $\{\Sigma_{t}\}_{t \in (-\epsilon,\epsilon)}$ forms a foliation of a neighborhood of $\Sigma$ in $M$.
<!-- end proposition-2 -->
<!-- proof-of-main-thm-1: body_md -->
The rigidity of Proposition \ref{prop 3.1} implies that, up to scaling, $\Sigma$ is isometric to the round hemisphere $\mathbb{S}^{n}_{+}$. The same proposition also yields $h^{\Sigma} = 0$, $Ric^{M}(N,N) = 0$ on $\Sigma$, and $h^{\partial M}(N,N) = 0$ along $\partial\Sigma$. Hence, Proposition \ref{foliation} provides a free boundary CMC foliation $\{\Sigma_{t}\}_{t\in (-\epsilon,\epsilon)}$ of a neighborhood of $\Sigma$. Let $G: \Sigma \times (-\epsilon,\epsilon) \rightarrow M$ given by $G_{t}(x) = \phi(x,w(x,t))$, be a parametrization of this foliation. Denote by $N_{t}$ a unit normal vector field along $\Sigma_{t}$, and define the \textit{lapse function} by $\rho_{t} = \langle \partial_{t}G, N_{t} \rangle$. Since $\rho_{0} = 1$ and $\Sigma$ is compact, we may assume that $\rho_{t} > 0$ for every $t \in (-\epsilon,\epsilon)$. It is well known that
\begin{align} \label{eq 4.1}
\begin{cases}
H'(t) = -\Delta_{\Sigma_{t}}\rho_{t} - \left(Ric^{M}(N_{t},N_{t}) + \vert h^{\Sigma_{t}} \vert^{2}\right)\rho_{t} \quad &\text{on } \Sigma_{t} ,\\
\frac{\partial\rho_{t}}{\partial\nu_{t}} = h^{\partial M}(N_{t},N_{t})\rho_{t} \quad &\text {along } \partial\Sigma_{t},
\end{cases}
\end{align}
where $\nu_{t}$ denotes unit conormal of $\partial\Sigma_{t}$ in $\Sigma_{t}$ (see \cite{AM, BC}). Multiplying (\ref{eq 4.1}) by $\rho^{-1}_{t}$, we obtain
\begin{align} \label{ine 4.2}
\begin{aligned}
H'(t)\rho^{-1}_{t} &= -\rho^{-1}_{t}\Delta_{\Sigma_{t}}\rho_{t} - \left(Ric^{M}(N_{t},N_{t}) + \vert h^{\Sigma_{t}} \vert^{2}\right) \\
&\leq -\rho^{-1}_{t}\Delta_{\Sigma_{t}}\rho_{t},
\end{aligned}
\end{align}
where we used $Ric^{M} \geq 0$. Integrating (\ref{ine 4.2}) over $\Sigma_{t}$, we obtain
\begin{align} \label{eq 4.3}
\begin{aligned}
H'(t)\int_{\Sigma_{t}}\rho^{-1}_{t} dv &\leq - \int_{\Sigma_{t}} \rho^{-1}_{t}\Delta_{\Sigma_{t}}\rho_{t} \, dv \\
&= -\int_{\partial\Sigma_{t}}\frac{\partial\rho_{t}}{\partial\nu_{t}}\frac{1}{\rho_{t}} \, ds - \int_{\Sigma_{t}}\frac{\vert \nabla^{\Sigma_{t}}\rho_{t} \vert^{2}} {\rho^{2}_{t}} \, dv \\
&= -\int_{\partial\Sigma_{t}}h^{\partial M}(N_{t},N_{t}) \, ds - \int_{\Sigma_{t}}\frac{\vert \nabla^{\Sigma_{t}}\rho_{t} \vert^{2}}{\rho^{2}_{t}} \, dv \\
&\leq -\int_{\Sigma_{t}} \frac{\vert \nabla^{\Sigma_{t}}\rho_{t} \vert^{2}}{\rho^{2}_{t}} \, dv \leq 0,
\end{aligned}
\end{align}
where in the last inequality we have used the weak convexity of $\partial M$. Since $\rho_{t} > 0$, it follows from (\ref{eq 4.3}) that $H'(t) \leq 0$. Since $H(0) = 0$, we obtain
\begin{align} \label{ine 4.3}
H(t) \leq 0 \leq H(-t) \quad \text{for all } t \in [0,\epsilon).
\end{align}
On the other hand, since each $\Sigma_{t}$ is a free boundary hypersurface, the first variation formula of volume gives
\begin{align} \label{ine 4.4}
\frac{d}{dt}{\rm Vol}(\Sigma_{t}) = H(t)\int_{\Sigma_{t}}\rho_{t} \, dv.
\end{align}
Since $\rho_{t} > 0$, it follows from (\ref{ine 4.3}) and (\ref{ine 4.4}) that
\begin{align*}
{\rm Vol}(\Sigma_{t}) \leq {\rm Vol}(\Sigma) \quad \text{for all } t \in (-\epsilon,\epsilon).
\end{align*}
Since $\Sigma$ is volume-minimizing, we have ${\rm Vol}(\Sigma_{t}) = {\rm Vol}(\Sigma)$ for every $t \in (-\epsilon,\epsilon)$. It follows from (\ref{ine 4.4}), together with $\rho_{t} > 0$, that $H(t) = 0$ for all $t \in (-\epsilon,\epsilon)$. Consequently, $H'(t) = 0$. Returning to (\ref{eq 4.3}), we obtain
\begin{align*}
0 \leq -\int_{\partial\Sigma_{t}}h^{\partial M}(N_{t},N_{t}) \, ds - \int_{\Sigma_{t}}\frac{\vert \nabla^{\Sigma_{t}}\rho_{t} \vert^{2}}{\rho^{2}_{t}} \, dv \leq 0.
\end{align*}
Therefore, $h^{\partial M}(N_{t},N_{t}) = 0$ along $\partial\Sigma_{t}$, and $\nabla^{\Sigma_{t}}\rho_{t} = 0$ on $\Sigma_{t}$. Thus $\rho_{t}$ is constant on each $\Sigma_{t}$. Hence, (\ref{eq 4.1}) reduces to
\begin{align*}
 0 = -\left(Ric^{M}(N_{t},N_{t}) + \vert h^{\Sigma_{t}} \vert^{2}\right)\rho_{t}.
\end{align*}
Since $\rho_{t} > 0$, $Ric^{M} \geq 0$, and $\vert h^{\Sigma_{t}} \vert^{2} \geq 0$, we conclude that $Ric^{M}(N_{t},N_{t}) = 0$ and $h^{\Sigma_{t}} = 0$ on every $\Sigma_{t}$. We next show that the unit normal vector field $N_{t}$ is parallel along $\Sigma_{t}$. Let $(x_{1}, \cdots, x_{n})$ be local coordinates on $\Sigma$. Since $\Sigma_{t}$ is totally geodesic, $\nabla_{\partial_{x_{i}}G}N_{t} = 0$. Furthermore, since $\vert N_{t} \vert = 1$, 
\begin{align*}
\left\langle \nabla_{\partial_{t}G}N_{t}, N_{t} \right\rangle = 0,
\end{align*}
and hence $\nabla_{\partial_{t}G}N_{t}$ is tangent to $\Sigma_{t}$. Since $\rho_{t}$ is constant on $\Sigma_{t}$, we have
\begin{align*}
\begin{aligned}
0 = \partial_{x_{i}}\rho_{t} &= \left\langle \nabla_{\partial_{x_{i}}G}N_{t}, \partial_{t}G \right\rangle + \left\langle N_{t}, \nabla_{\partial_{x_{i}}G}\partial_{t}G \right\rangle \\
&= \left\langle N_{t}, \nabla_{\partial_{t}G} \partial_{x_{i}}G \right\rangle \\
&= - \left\langle \nabla_{\partial_{t}G}N_{t}, \partial_{x_{i}}G \right\rangle
\end{aligned}
\end{align*}
(see also \cite{AM, BC, CR}). Thus $N_{t}$ is a parallel vector field. It follows that the integral curves of $N_{t}$ are geodesic. Consider the exponential map $E(t,x) := {\rm exp}_{x}\left(tN(x)\right)$ for $x \in \Sigma$. Since $N_{t}$ is parallel, its flow is given by the exponential map $E$, which is an isometry for all $t \in (-\epsilon,\epsilon)$. Therefore, we conclude that, up to scaling, a neighborhood of $\Sigma$ in $M$ is isometric to $\mathbb{S}^{n}_{+} \times (-\epsilon,\epsilon)$.

<!-- corollary-1: body_md -->
Let $M$ be a connected, complete $(n+1)$-dimensional Riemannian manifold $(n \geq 3)$ with weakly convex boundary $\partial M$. Assume that $Ric^{M} \geq 0$ and $\inf_{M}R^{M} > 0$. Let $\Sigma$ be a compact connected two-sided, properly embedded, free boundary hypersurface satisfying $Ric^{\Sigma} = \frac{R^{\Sigma}}{n}g_{\Sigma}$. Assume, in addition, that $\Sigma$ is volume-minimizing in its isotopy class. If equality holds in (\ref{ine vol}), then, up to scaling, the universal cover $\tilde{M}$ of $M$ is isometric to $\mathbb{S}^{n}_{+} \times \mathbb{R}$.

<!-- proof-2: body_md -->
Since $\Sigma$ is volume-minimizing in its isotopy class, it is locally volume-minimizing. By the Proposition \ref{prop 3.1} and the assumed equality of (\ref{ine vol}), up to scaling, $\Sigma$ is isometric to $\mathbb{S}^{n}_{+}$. Applying the global conclusion of Theorem 4.2 with constant density in \cite{CR}, we obtain a Riemannian covering $G: \Sigma \times \mathbb{R} \rightarrow M$. Since $\Sigma$ is a round hemisphere, $\Sigma \times \mathbb{R}$ is simply connected. Thus $G$ is the universal covering of $M$. Consequently, up to scaling, the universal cover of $M$ is isometric to $\mathbb{S}^{n}_{+} \times \mathbb{R}$ with the product metric.
