<!-- theorem-1: body_md -->
\label{thm:main}
Every promised 3-colorable graph on $n$ vertices has a proper coloring with
$O(n^{4/23})$ colors that can be found in randomized polynomial time, with
success probability at least $1-n^{-10}$.
<!-- end theorem-1 -->
<!-- head-1: body_md -->
For a graph $G=(V,E)$, write $n=|V|$, let $G[S]$ denote the
subgraph induced by $S\subseteq V$, and let $N_G(v)$ denote the
open neighborhood of $v$. We write $\Delta(G)$ for the maximum
degree and use $\Delta$ for a specified upper bound on $\Delta(G)$.
All vector norms are Euclidean, and $\langle x,y\rangle$ denotes the
Euclidean inner product. Throughout, $g$ is a standard Gaussian
vector in the ambient space, and $\mathbb E$ and $\operatorname{Var}$ denote expectation
and variance. For $Z\sim N(0,1)$, define $\Phi(t):=\Pr[Z\le t]$ and $\overline\Phi(t):=1-\Phi(t)=\Pr[Z\ge t]$. We write $x_+:=\max\{x,0\}$, and all logarithms are natural.
The notation $\widetilde O$ suppresses polylogarithmic factors.
Unless stated otherwise, asymptotic notation refers to
$n\to\infty$, with the recursion depth and all auxiliary constants
held fixed. Moreover, all the neighborhoods we consider are open neighborhoods. Unless explicitly restricted,
a root-correlation shell with root $i$ consists of vertices in
$V\setminus\{i\}$ of the original input graph. All analytic and
algorithmic correlation buckets use one fixed partition of width
$1/\log\Delta$. \\
<!-- end head-1 -->
<!-- definition-1: body_md -->
\label{def:vector-coloring}
For a real number $\kappa\ge2$, a \emph{vector $\kappa$-coloring}
of $G=(V,E)$ assigns unit vectors $(x_j)_{j\in V}$ such that
$\langle x_j,x_k\rangle\le-1/(\kappa-1)$ for every $jk\in E$.
The coloring is \emph{strict} if equality holds on every edge.
<!-- end definition-1 -->
<!-- lemma-1: body_md -->
\label{lem:kms-background}
Given a vector $\kappa$-coloring of an $m$-vertex graph with maximum
degree at most $\Delta\ge2$, a suitable threshold in
Algorithm~\ref{alg:kms} gives an independent set $I$ with
$\mathbb E|I|\ge\widetilde\Omega(m\Delta^{-(\kappa-2)/\kappa})$.
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{lem:sos-background}
An exact feasible symmetrized three-round solution gives a strict
vector 3-coloring $(v_j)$ and consistent local distributions on proper
colorings of triples, with uniform one-vertex marginals.
For a root $i$, write $q_j=\langle v_i,v_j\rangle$. Then
$\Pr[j=\mathsf R\mid i=\mathsf R]=(1+2q_j)/3$,
so $q_j\ge-1/2$ and $q_j+q_k\le1/2$ whenever $jk\in E$.
<!-- end lemma-2 -->
<!-- definition-2: body_md -->
\label{def:gaussian-cover}
For $s\in\mathbb R$ and $\delta\in(0,1]$, a finite family
$X=(r_j)_{j\in J}$ is an \emph{$(s,\delta)$-cover} if

\label{lem:chen-background}

<!-- lemma-8: body_md -->
\label{lem:variance}
Let $F(g)=\max_{1\le j\le m}\langle g,u_j\rangle$, where $g$ is standard Gaussian,
$m\ge2$, and $\|u_j\|\le r$. Write $\mu=\mathbb E F$ and $v=\operatorname{Var}(F)$.
Then
\begin{equation}\label{eq:variance}
 v\le r^2-\frac{\mu^2}{2\log m}.
\end{equation}
For $x\ge0$, $\Pr(F\le\mu-x)\le\exp(-x^2/(2v))$, with the usual
interpretation when $v=0$.
<!-- end lemma-8 -->
<!-- proof-1: body_md -->
The function $F$ is convex and $r$-Lipschitz. Chen's Theorems~1 and~3
\cite{Chen} give convexity of
$\Lambda(\lambda)=\lambda^{-1}\log\mathbb E e^{\lambda F}$,
continuously extended at zero, and the stated variance-scale lower tail.
The expansion
$\log\mathbb E e^{\lambda F}=\lambda\mu+\lambda^2v/2+o(\lambda^2)$
as $\lambda\to0$ gives $\Lambda(0)=\mu$ and
$\Lambda'(0)=v/2$, so convexity implies
$\Lambda(\lambda)\ge\mu+\lambda v/2$ for $\lambda>0$.
For such $\lambda$, we have
\begin{align*}
 \mathbb E e^{\lambda F}
   &=\mathbb E\max_{1\le j\le m}e^{\lambda\langle g,u_j\rangle}\\
   &\le\sum_{j=1}^m\mathbb E e^{\lambda\langle g,u_j\rangle}\\
   &=\sum_{j=1}^m\exp\!\left(\frac{\lambda^2\|u_j\|^2}{2}\right)\le m\exp\!\left(\frac{\lambda^2r^2}{2}\right),
    \end{align*}
    and $\mu+\frac{\lambda v}{2}
   \le\Lambda(\lambda)
    \le\frac{\log m}{\lambda}+\frac{\lambda r^2}{2}$ with $v\le r^2-\frac{2\mu}{\lambda}
                  +\frac{2\log m}{\lambda^2}$.
Since $\mathbb E F\ge\mathbb E\langle g,u_1\rangle=0$, we have $\mu\ge0$.
For $\mu>0$, substituting $\lambda=2\log(m)/\mu$ gives
\begin{align*}v &\le r^2-\mu^2/\log m+\mu^2/(2\log m)\\
&=r^2-\mu^2/(2\log m).\end{align*} For $\mu=0$, taking $\lambda\to\infty$
gives $v\le r^2$, and both cases give \eqref{eq:variance}.

<!-- lemma-9: body_md -->
\label{lem:compose}
Fix constants $L>0$ and $A\ge0$. Suppose a family of unit vectors $(x_j)_{j\in J}$ is an
$(Lt,t^{-O(1)})$-cover, with $|J|\le\Delta^{A+o(1)}$. For each $j$, let
$U_j\subset x_j^\perp$ be a fixed inner $(\sqrt3t-o(t),t^{-O(1)})$-cover,
with at most $\Delta$ vectors and a uniform norm upper bound $r_j\le1$.
Then there is a joint witness, on an event of probability $1/\mathrm{poly}(t)$, satisfying $\langle g,x_j\rangle\ge Lt$ and
\begin{align*}
 \max_{u\in U_j}\langle g,u\rangle
 &\ge\left(\sqrt3-
 \sqrt{\left(r_j^2-\frac1{1+c_*}\right)_+}
 \sqrt{(DA-L^2)_+}-o(1)\right)t.
\end{align*}
The error is uniform when $L$ ranges over a fixed compact subset of
$(0,\infty)$ and $A$ ranges over a fixed bounded interval, provided
the cover-probability and inner-threshold bounds are uniform.
<!-- end lemma-9 -->
<!-- proof-2: body_md -->
Let $F_j(g)=\max_{u\in U_j}\langle g,u\rangle$,
$\mu_j=\mathbb E F_j$, and $v_j=\operatorname{Var}(F_j)$. Set
$H_j=\{\langle g,x_j\rangle\ge Lt\}$, $H=\bigcup_{j\in J}H_j$,
$\delta=\Pr(H)$, and
$B_t=\log(2|J|\overline\Phi(Lt)/\delta)$. Since the $x_j$ are unit vectors, we have that $ B_t\ge\log2$, and 
\begin{align*}
 t^{-O(1)}&\le\delta
       \le\sum_{j\in J}\Pr(H_j)=|J|\overline\Phi(Lt).
\end{align*}
Let $E_j=\{F_j<\mu_j-\sqrt{2v_jB_t}\}$.
Lemma~\ref{lem:variance} gives $\Pr(E_j)\le e^{-B_t}$ when
$v_j>0$; when $v_j=0$, the strict inequality makes $E_j$ null.
Since $U_j\subset x_j^\perp$, the event $E_j$ is independent of $H_j$.
Therefore
\begin{align*}
 \Pr\!\left(\bigcup_{j\in J}(H_j\cap E_j)\right)
   &\le\sum_{j\in J}\Pr(H_j\cap E_j)\\
   &=\sum_{j\in J}\Pr(H_j)\Pr(E_j)\\
   &\le |J|\overline\Phi(Lt)e^{-B_t}=\frac\delta2,
   \end{align*}
   and
   \begin{align*}
 \Pr\!\left(\bigcup_{j\in J}(H_j\cap E_j^c)\right)
   &\ge\Pr(H)-\Pr\!\left(\bigcup_{j\in J}(H_j\cap E_j)\right)\ge\frac\delta2.
\end{align*}
With probability at least $\delta/2$, some $j$ satisfies both
$\langle g,x_j\rangle\ge Lt$ and $F_j\ge\mu_j-\sqrt{2v_jB_t}$.
To bound this threshold, we use Gaussian-tail asymptotics and $|J|$ to get
\begin{align*}
 B_t
   &=\log|J|+\log\overline\Phi(Lt)+\log(2/\delta)\\
   &\le(A+o(1))\log\Delta-\frac{L^2t^2}{2}+O(\log t)\\
   &=\frac{DA-L^2}{2}t^2+o(t^2)
    \\
    &\le\frac{(DA-L^2)_+}{2}t^2+o(t^2).
\end{align*}
Put $\sigma_j^2=(r_j^2-1/(1+c_*))_+$ and $Q=(DA-L^2)_+$.
The mean estimate and \eqref{eq:normvariance} allow a uniform
$\eta_t\ge0$, tending to zero, such that
$\mu_j\ge(\sqrt3-\eta_t)t,
 v_j\le\sigma_j^2+\eta_t$, and
$2B_t\le(Q+\eta_t)t^2$, and \begin{align*}
 \sqrt{2v_jB_t}
   &\le t\sqrt{(\sigma_j^2+\eta_t)(Q+\eta_t)}\\
   &\le t(\sigma_j+\sqrt{\eta_t})(\sqrt Q+\sqrt{\eta_t})\\
   &=t\bigl(\sigma_j\sqrt Q+o(1)\bigr).\end{align*} Thus, we have that
$\mu_j-\sqrt{2v_jB_t}\ge\bigl(\sqrt3-\sigma_j\sqrt Q-o(1)\bigr)t$. The error is uniform because $\sigma_j$ and $Q$ are bounded, including
when either is zero. So, desired threshold holds with
probability at least $\delta/2=t^{-O(1)}$.

<!-- lemma-10: body_md -->
\label{lem:shell-rounding}
Fix a root $i$ and $m$ vertices $H\subseteq V\setminus\{i\}$ with
$|\langle v_i,v_j\rangle-q|\le h$, where $q\in[-1/2,1]$ and $h=o(1)$.
From the symmetrized three-round SoS solution one can compute, in
polynomial time, an independent set in $G[H]$ of expected size
\[
 m\Delta^{-\rho(q)-o(1)},
\]
with $\rho$ given in \eqref{eq:rho}. The bound is uniform in $q$,
including under the inverse-polynomial SDP errors described in
Section~\ref{sec:algorithm}.

<!-- proof-3: body_md -->
Let $v_{\{(i,\mathsf R),(j,a)\}}$ be the SoS vector for the indicated
partial color assignment, with $a\in\{\mathsf R,\mathsf G,\mathsf B\}$.
Since $\Pr[i=\mathsf R]=1/3$, the conditioned vectors
$w_{j,a}:=\sqrt3\,v_{\{(i,\mathsf R),(j,a)\}}$ satisfy $\|w_{j,a}\|^2=\Pr[j=a\mid i=\mathsf R]$ with 
$\langle w_{j,a},w_{k,b}\rangle=\Pr[j=a,\ k=b\mid i=\mathsf R]$ and $\sum_a\|w_{j,a}\|^2=1$, where $\langle w_{j,a},w_{j,b}\rangle=0$ for $a\neq b$.

For an edge $jk$, the same-color inner products vanish. In particular,
with $p_j=\|w_{j,\mathsf R}\|^2=(1+2q_j)/3$, we have $\langle w_{j,\mathsf R},w_{k,\mathsf R}\rangle=0$, with $\Pr[j\ne\mathsf R,\ k\ne\mathsf R\mid i=\mathsf R]=1-p_j-p_k$ and $\langle w_{j,\mathsf G},w_{k,\mathsf B}\rangle
 +\langle w_{j,\mathsf B},w_{k,\mathsf G}\rangle
=1-p_j-p_k$.

 
For $q\le0$, we use the construction of \cite[Lemma~3.2]{BHL}: let $x_j\coloneqq \frac{w_{j,\mathsf G}-w_{j,\mathsf B}}{\sqrt{1-p_j}}$. Then,
\begin{align*}
 \|x_j\|^2 &=\left\|\frac{w_{j,\mathsf G}-w_{j,\mathsf B}}{\sqrt{1-p_j}}\right\|^2=\frac{\|w_{j,\mathsf G}\|^2+\|w_{j,\mathsf B}\|^2}{1-p_j}
 =1,
 \end{align*}
 and
\begin{align*} \langle x_j,x_k\rangle
 &=-\frac{1-p_j-p_k}{\sqrt{(1-p_j)(1-p_k)}}=-\frac{(1-4q)/3}{(2-2q)/3}+O(h)
=-\frac{1-4q}{2-2q}+O(h).
\end{align*}
Its parameter and rounding loss are given here by $\kappa_-(q)=1+\frac{2-2q}{1-4q}
 =\frac{3-6q}{1-4q}$ and $\frac{\kappa_-(q)-2}{\kappa_-(q)}=\frac{(3-6q)-2(1-4q)}{3-6q}
 =\frac{1+2q}{3-6q}$, where all denominators are bounded below uniformly for
$q\in[-1/2,0]$, and the limiting parameter at $q=-1/2$ is two.

For $0\le q\le1/4$, project off the root. On an edge $jk$, let 
$x_j:=\frac{v_j-q_jv_i}{\sqrt{1-q_j^2}}$ where $\|x_j\|^2=1$. Then,
\begin{align*}
 \langle x_j,x_k\rangle
 &=\frac{\langle v_j,v_k\rangle-q_jq_k}
         {\sqrt{(1-q_j^2)(1-q_k^2)}}=-\frac{1/2+q^2}{1-q^2}+O(h)
         \end{align*} and we have that 
$\kappa_0(q)=1+\frac{1-q^2}{1/2+q^2}
 =\frac{3}{1+2q^2}$ and $\frac{\kappa_0(q)-2}{\kappa_0(q)}
 =1-\frac{2(1+2q^2)}{3}
 =\frac{1-4q^2}{3}$.

For $1/16\le q\le1/4$, we also use the construction of
\cite[Lemma~3.4]{BHL}. Assign signs $+1$ to $\mathsf R$ and $-1$ to
the other two colors, and set
$y_j:=w_{j,\mathsf R}-w_{j,\mathsf G}-w_{j,\mathsf B}$.
Orthogonality at each vertex gives $\|y_j\|^2=1$.
On an edge, exactly one endpoint has color $\mathsf R$ with conditional
probability $p_j+p_k$, and both avoid it with probability $1-p_j-p_k$.
Therefore
\begin{align*}
 \langle y_j,y_k\rangle
 &=(1-p_j-p_k)-(p_j+p_k)\\
 &=1-2p_j-2p_k
 =-\frac{1+8q}{3}+O(h),
 \end{align*} with
$\kappa_+(q)=1+\frac{3}{1+8q}
 =\frac{4+8q}{1+8q}$, and 
$\frac{\kappa_+(q)-2}{\kappa_+(q)}=\frac{(4+8q)-2(1+8q)}{4+8q}
 =\frac{1-4q}{2+4q}$. Now, apply KMS to each available construction and keep the larger returned
set. A construction with loss $\ell$ gives
$\mathbb E|I|\ge\widetilde\Omega(m\Delta^{-\ell-O(h)})
=m\Delta^{-\ell-o(1)}$. Choosing the smaller available loss gives
the first three branches of \eqref{eq:rho}.

For $q\ge1/4$, the same $y_j$ construction gives
\begin{align*}
 \langle y_j,y_k\rangle
 &=-\frac{1+8q}{3}+O(h)\le-1+O(h),\end{align*} with $
 \kappa=2+O(h)$ and $\frac{\kappa-2}{\kappa}=O(h)$. 
 This yields $\mathbb E|I|\ge m\Delta^{-o(1)}$.
If every $q_j>1/4$, the local constraint instead shows that $H$ itself
is independent: an edge would require both
$q_j+q_k>1/2$ and $q_j+q_k\le1/2$.

If moment residuals are bounded by $\xi$, the bound remains uniform. Conditioning divides
by $1/3+O(\xi)$, and every further denominator used above has a fixed
positive lower bound in its stated range. After normalization, the
edge correlations and vector parameters therefore change by
$O(h+\xi)$. For each construction this gives
\begin{align*}
 \kappa_{\rm used}=\kappa(q)+O(h+\xi) \implies \frac{\kappa_{\rm used}-2}{\kappa_{\rm used}}=\frac{\kappa(q)-2}{\kappa(q)}+O(h+\xi),\end{align*} and so
$\mathbb E|I|\ge m\Delta^{-\rho(q)-o(1)}$.
Near parameter two, choose a valid parameter at least
$2+O(h+\xi)$, and at exactly two, use bipartite coloring. The losses agree at $q=0$ and $q=1/4$, and the projection loss gives the minimum on both sides of $q=1/16$. So, the same estimate holds
uniformly for buckets crossing the boundaries.\looseness=-1

<!-- lemma-11: body_md -->
\label{lem:lift}
Let $L>0$. Suppose retained endpoints $j$ form a cover of root directions
$p_j=(v_j-q_jv_i)/\sqrt{1-q_j^2}$ at threshold $Lt$, where
$q_j=\langle v_i,v_j\rangle$ belongs to a bucket represented by
$q\in[-1/2,3/5]$. Under the size assumptions above, some retained
successor shell has correlation $z$ satisfying
\begin{equation}\label{eq:feasible}
 z\ge-\tfrac12,\quad q+z\le\tfrac12,\quad
 (2z+q)^2\le3d(1-q^2),
\end{equation}
up to vanishing errors, and forms a cover at threshold
$(M(q,L,z)-o(1))t$, where
\begin{equation}\label{eq:lift}
 M(q,L,z)=
 \frac{(1+2qz)L
 -\sqrt{K(q)-L^2}\sqrt{3d(1-q^2)-(2z+q)^2}
 +3\sqrt{1-q^2}}
 {2\sqrt{1-q^2}\sqrt{1-z^2}}.
\end{equation}
Here \eqref{eq:lift} is used on continuing states $L^2\le K(q)$;
positive parts and vanishing slack cover boundary perturbations.

<!-- proof-4: body_md -->
Use actual endpoint and successor correlations $q,z$ until the
bucketing step. Put $s=\sqrt{1-q^2}$ and $p=p_j$, so
$v_j=qv_i+sp$. The reverse root direction
$v_{ji}:=(v_i-qv_j)/s=sv_i-qp$ is a unit vector perpendicular to
$v_j$, whether or not $i$ and $j$ are adjacent. For a neighbor $k$ of $j$ in $G'$, write
$r:=v_{jk}$ and $z:=\langle v_i,v_k\rangle$. Since
$v_k=-v_j/2+\sqrt3r/2$ and $r\perp v_j$, we obtain $\langle v_i,r\rangle=\frac{2z+q}{\sqrt3}$, with
\begin{align*}
 \delta:=\langle r,v_{ji}\rangle
 &=\frac{\langle r,v_i\rangle-q\langle r,v_j\rangle}s
 =\frac{2z+q}{\sqrt3s},
 \end{align*} and
 $u:=r-\delta v_{ji}$ with $\|u\|^2=1-\delta^2$ and $
 \langle u,v_j\rangle=0, \langle u,v_{ji}\rangle=0$.
 
The pairs $(v_j,v_{ji})$ and $(v_i,p)$ span the same plane, so
$u\perp v_i,p$. Substituting $r=\delta(sv_i-qp)+u$ yields
\begin{align*}
 v_k
 &=-\frac12(qv_i+sp)
   +\frac{\sqrt3}{2}\bigl(\delta sv_i-\delta qp+u\bigr)\\
 &=\left(-\frac q2+\frac{\sqrt3\delta s}{2}\right)v_i
   -\left(\frac s2+\frac{\sqrt3\delta q}{2}\right)p
   +\frac{\sqrt3}{2}u,\\
\end{align*} Moreover, we would have
$-\frac q2+\frac{\sqrt3\delta s}{2}=z$ and $\frac s2+\frac{\sqrt3\delta q}{2}
 =\frac{s^2+q(2z+q)}{2s}
 =\frac{1+2qz}{2s}$. Hence
\begin{equation}\label{eq:geometry}
 v_k-zv_i=-\frac{1+2qz}{2s}p+\frac{\sqrt3}{2}u.
\end{equation}

Every $j$ retains an outgoing cover in $G'$. For a fixed $j$,
let $F_j(g):=\max_{k\in N_{G'}(j)}\langle g,v_{jk}\rangle$ and let
$\widehat F_j(g):=\max_{k\in N_{G'}(j)}\langle g,u_{jk}\rangle$, where
$u_{jk}$ is the projection just defined. Since $|\delta|\le1$,
\begin{align*}
 \widehat F_j(g)&\ge F_j(g)-|\langle g,v_{ji}\rangle|,\\
 \Pr\!\left[\widehat F_j(g)\ge\sqrt3t-o(t)-\log t\right]
 &\ge\Pr\!\left[F_j(g)\ge\sqrt3t-o(t)\right]
       -\Pr[|\langle g,v_{ji}\rangle|>\log t]\\
 &\ge t^{-C_j}-2e^{-(\log t)^2/2}
 =t^{-O(1)}.
\end{align*}
Pruning bounds the $C_j$ uniformly. The estimate applies separately to
each parent, and $\log t=o(t)$ leaves threshold $\sqrt3t-o(t)$.
Select a heavy $z$ bucket in each family, then a heavy outer class with
the same selected bucket. Each inner family still has at most $\Delta$
vectors and cover probability $t^{-O(1)}$. We can now replace the
correlations by bucket representatives, with the uniform errors proved
above. Writing $\widehat F_j$ for the selected maximum,
\eqref{eq:normvariance} gives $0\le\operatorname{Var}(\widehat F_j)\le1-\delta^2-\frac1{1+c_*}+o(1)
 =d-\delta^2+o(1)$ where $\delta^2\le d+o(1)$, and $(2z+q)^2\le3d(1-q^2)+o(1)$.
 
The remaining constraints $z\ge-1/2$ and $q+z\le1/2$ come from
the local triple distribution. Together these are \eqref{eq:feasible}.

By Gaussian symmetry, the restricted outer family $(-p_j)$ is also a
cover. Its size is at most $\Delta^{K(q)/D+o(1)}$, and its selected
inner family lies in $p_j^\perp$. Lemma~\ref{lem:compose} therefore
provides a joint witness, on an event of probability $t^{-O(1)}$, with $ \langle g,-p\rangle\ge Lt$, $\langle g,u\rangle\ge\left(\sqrt3-
          \sqrt{d-\delta^2}\sqrt{K(q)-L^2}-o(1)\right)t$, and $d-\delta^2=\frac{3d(1-q^2)-(2z+q)^2}{3(1-q^2)}$.
          
Positive parts and vanishing slack are understood at the boundaries.
For feasible successors, \eqref{eq:domain} below gives
$q,z\in[-1/2,3/5]$, and consequently $1+2qz\ge2/5>0$.
Both favorable evaluations can therefore be inserted into
\eqref{eq:geometry}. With
$p_k:=(v_k-zv_i)/\sqrt{1-z^2}$, this gives
\begin{align*}
 \frac{\langle g,p_k\rangle}t
 &=\frac{1+2qz}{2s\sqrt{1-z^2}}\,\frac{\langle g,-p\rangle}t
   +\frac{\sqrt3}{2\sqrt{1-z^2}}\,\frac{\langle g,u\rangle}t\\
 &\ge\frac{(1+2qz)L+3s
       -\sqrt3s\sqrt{d-\delta^2}\sqrt{K(q)-L^2}}
       {2s\sqrt{1-z^2}}-o(1)\\
 &=\frac{(1+2qz)L+3\sqrt{1-q^2}
       -\sqrt{3d(1-q^2)-(2z+q)^2}\sqrt{K(q)-L^2}}
       {2\sqrt{1-q^2}\sqrt{1-z^2}}-o(1)\\
 &=M(q,L,z)-o(1).
\end{align*}
Therefore, we have shown that the successor shell has the required cover. Its direction
depends only on the root and endpoint, so repeated paths do not increase
the number of directions beyond the shell size, and all the successors remain
in $G'$. Since we established feasibility of the norm before normalization, the bound below keeps $z$ a fixed distance from $1$.

<!-- proof-5: body_md -->
We verify both assertions by rational interval arithmetic on the compact
domain. The verifier and its recorded output are available at
\url{https://github.com/emiletimothy/3coloringVerifier.git}, and the accompanying README gives the command needed to reproduce the verification.

Now, set $Q=2^{120}$ and round rational operations outward using
$\operatorname{down}(x)=Q^{-1}\lfloor Qx\rfloor$ and
$\operatorname{up}(x)=Q^{-1}\lceil Qx\rceil$, so
$\operatorname{down}(x)\le x\le\operatorname{up}(x)$.
For $x\ge0$, compute $k=\lfloor\sqrt{\lfloor Q^2x\rfloor}\rfloor$
by integer square root. Then $k^2\le\lfloor Q^2x\rfloor\le Q^2x<(k+1)^2$ and $\frac{k}{Q}\le\sqrt{x}<\frac{k+1}{Q}$. If $k^2=Q^2x$, use $k/Q$ for both endpoints. Apply these rules
outward to interval endpoints. A positive lower endpoint certifies
positivity on the whole box; otherwise subdivide.

For initialization, the domain $[0,d]\times[-1/2,3/5]$ in the seed
parameter $a$ and the first successor correlation $z$ is covered by an
$8\times8$ grid; boxes violating a necessary feasibility constraint
are discarded, and unresolved boxes are bisected adaptively. Boxes
proving an immediate terminal violation or a first threshold above
$T$ are accepted directly. The remaining boxes are handled by the
following uniform second-lift bound.\\


For a state $(q,L)$, put $s=\sqrt{1-q^2}$, $A=\sqrt{3d}\,s$,
$W=\sqrt{K(q)-L^2}$, and $u=2z+q$. The denominator is
$V=2s\sqrt{1-z^2}=s\sqrt{4-(u-q)^2}$.
Feasibility gives $u^2\le A^2$ and $0<V\le2$, and
\begin{align*}
 H:=V\bigl(M(q,L,z)-T\bigr)
 &= (1+2qz)L+3s-W\sqrt{A^2-u^2}-TV\\
 &= s^2L+3s+qLu-W\sqrt{A^2-u^2}
      -Ts\sqrt{4-(u-q)^2}.
\end{align*}
For $\ell\in\mathbb R$, write
$qLu=(qL-\ell)u+\ell(u-q)+\ell q$ and apply Cauchy--Schwarz twice:
\begin{align*}
 (qL-\ell)u-W\sqrt{A^2-u^2}
 &=\langle (qL-\ell,-W),(u,\sqrt{A^2-u^2})\rangle\\
 &\ge-\sqrt{(qL-\ell)^2+W^2}
          \sqrt{u^2+(A^2-u^2)}\\
 &=-A\sqrt{(qL-\ell)^2+W^2}\end{align*} and \begin{align*}
 \ell(u-q)-Ts\sqrt{4-(u-q)^2}
 &=\langle (\ell,-Ts),(u-q,\sqrt{4-(u-q)^2})\rangle\\
 &\ge-\sqrt{\ell^2+T^2s^2}
          \sqrt{(u-q)^2+4-(u-q)^2}\\
 &=-2\sqrt{\ell^2+T^2s^2}.
\end{align*}
Combining these inequalities gives
\begin{equation}\label{eq:dual}
\begin{aligned}
 H\ge{}&s^2L+3s+\ell q
       -A\sqrt{(qL-\ell)^2+W^2}\\
       &\hspace{28mm}-2\sqrt{\ell^2+T^2s^2}.
\end{aligned}
\end{equation}

The bound holds for every feasible successor and every $\ell$.
On each box choose any rational $\ell$ and verify the bound by exact
interval arithmetic. A poor choice may require further subdivision,
but cannot cause a false acceptance.

Before this check, replace the first-lift threshold by its positive
dyadic lower endpoint. This is valid because, with
$B=\sqrt{3d(1-q^2)-(2z+q)^2}$ and
$V=2\sqrt{1-q^2}\sqrt{1-z^2}$, we have that for all $0\leq L < \sqrt{K(q)}$,
\begin{align*}
 \frac{\partial M}{\partial L}(q,L,z)
 &=\frac{1+2qz+B L/\sqrt{K(q)-L^2}}{V}>0,
\end{align*}
where $1+2qz\ge2/5$, and continuity extends monotonicity to the
endpoint. Thus $0<\underline L_1\le L_1$ implies
$M(z,L_1,w)\ge M(z,\underline L_1,w)>T$ for every feasible second
successor $w$. Subdivision terminates with smallest threshold margin
at least $2\times10^{-7}$.
For drift, retain $s,B,V$ above and set $a_0=1+2qz-V$, $K=K(q)$.
Then
\begin{align*}
 H(L):=V\bigl(M(q,L,z)-L\bigr)
 =a_0L-B\sqrt{K-L^2}+3s,
 \end{align*} and $H'(L)=a_0+\frac{BL}{\sqrt{K-L^2}}$ and $H''(L)=\frac{BK}{(K-L^2)^{3/2}}\ge0$, for $T\le L  < \sqrt K$.
 
In particular, when $K>T^2$, convexity gives the supporting-line bound
\begin{align*}
 H(L)&\ge H(T)+H'(T)(L-T)\\
 &\ge H(T)+\min\{0,H'(T)\}(\sqrt K-T)
 \qquad(T\le L\le\sqrt K).
\end{align*}
A second, global lower bound follows from Cauchy--Schwarz:
\begin{align*}
 H(L)-3s
 &=\langle (a_0,-B),(L,\sqrt{K-L^2})\rangle\\
 &\ge-\sqrt{a_0^2+B^2}\sqrt{L^2+K-L^2}
 =-\sqrt K\sqrt{a_0^2+B^2}.
\end{align*}
Thus the verifier may use the larger of these two lower bounds.
When $H'(T)\ge0$, the first reduces to $H(L)\ge H(T)$.
When $K=T^2$, the interval consists of the single point $L=T$,
which is evaluated directly; when $K<T^2$, no continuing state is
present. Boxes where a denominator enclosure contains zero are
handled by direct endpoint bounds or subdivision. These inequalities
treat $L$ universally, leaving only $q,z$ as subdivision variables.
A finite adaptive subdivision of the $(q,z)$-domain verifies
$H(L)-V/1000>10^{-7}$ on every nonempty accepted box, with smallest
numerator margin at least $2\times10^{-5}$. To enclose $K$, the function
$\rho$ is evaluated at the interval endpoints and at zero when present;
$\rho$ is continuous, increases up to $q=0$, and decreases thereafter.
Both expressions defining $K$ are increasing in $c$, so endpoint
evaluation also encloses the entire stated parameter interval. This
proves both assertions.

The initialization and drift checks terminate with $8{,}140$
and $355$ final boxes, respectively, including boxes excluded
by necessary feasibility constraints.
The certified initialization threshold margin exceeds
$2.55\times10^{-7}$, and the certified lower bound on
$V(M(q,L,z)-L-1/1000)$ exceeds $2.83\times10^{-5}$.
Both checks cover their full stated domains and leave no
unresolved boxes.

<!-- lemma-12: body_md -->
\label{lem:app-persistent-pruning}
Suppose an $n$-vertex graph has symmetric nonnegative edge weights
$\mu_i(j)=\mu_j(i)$, with row sums at most one and
\[
 \sum_{\{i,j\}\in E}\mu_i(j)\ge n/8.
\]
For every vertex $i$, suppose unit directions $r_{ij}$ form an outgoing
packing at threshold $s=(\sqrt3-o(1))t$ with weights $\mu_i$.
Assume $|N(i)|\le\Delta\le\overline\Phi(s)^{-(1+c_{\rm pr})}$.
Put $\varepsilon=1/\log\log t$, $\tau=\varepsilon^2$, and
\[
 \lambda=\frac{c_{\rm pr}}{1+c_{\rm pr}}
       \left(1+\frac{C_1}{\log s}\right),\qquad
 p=\exp[-C_2(\log s)^2],
\]
where $C_1,C_2>0$ depend only on $c_{\rm pr}$.
For sufficiently large $n$, there is a nonempty retained graph $G'$
such that every retained vertex $i$ has
\begin{equation}\label{eq:app-row}
 \mu_i(N_{G'}(i))\ge1/8-O(\varepsilon),
\end{equation}
and every oriented retained edge $i\to j$ has
\begin{equation}\label{eq:app-twowalk}
 \mu_j\{k\in N_{G'}(j):
       -\varepsilon\le\langle r_{ji},r_{jk}\rangle\le\lambda\}
 \ge\tau/2.
\end{equation}
<!-- end lemma-12 -->
<!-- proof-6: body_md -->
Apply \cite[Theorem 4.11]{BHL} at each vertex. It gives
$X_i\subseteq N(i)$ with lost mass at most $1/\log s$ and, for every
$h\in X_i$,
\begin{equation}\label{eq:app-spread}
 \mu_i\{k\in X_i:\langle r_{ih},r_{ik}\rangle\ge\lambda\}\le p.
\end{equation}
Rows with total mass at most $1/\log s$ may be discarded entirely.
Delete an edge unless both endpoints retain it, and let $E_0$ be the
remaining edge set. Symmetry of the weights gives the explicit charge
\begin{align*}
 \sum_{\{i,j\}\in E\setminus E_0}\mu_i(j)
 &\le\sum_i\sum_{j\in N(i)\setminus X_i}\mu_i(j)\\
 &\le\sum_i\frac1{\log s}
   =\frac n{\log s}.
\end{align*}
An edge discarded by both endpoints may be counted twice on the right,
which only increases this upper bound. Every subsequent neighborhood is
a subset of its $X_i$, so~\eqref{eq:app-spread} continues to hold for
every retained neighbor.

Set $B_t=\frac1{\log s}+\tau(1+1/\varepsilon)$ and $r_0=\frac18-2B_t$. For large $n$, $r_0>0$. We repeatedly apply either of the following operations whenever possible:
\begin{enumerate}
\item Delete a vertex whose current incident weight is less than $r_0$.
\item If a current oriented edge $i\to j$ satisfies
 \[
 E_{ij}:=\{k\in N_{\rm current}(j):
                 \langle r_{ji},r_{jk}\rangle\ge-\varepsilon\},
 \qquad \mu_j(E_{ij})<\tau,
 \]
 delete every edge $jk$ with $k\in E_{ij}$.
\end{enumerate}
The second operation removes the triggering edge $ij$ itself, since
$\langle r_{ji},r_{ji}\rangle=1$. If $i_1,\ldots,i_q$ successively trigger that
operation at a fixed center $j$, a later triggering edge survived every
earlier deletion. Consequently, for $a<b$, we have that $i_b\notin E_{i_a j}
 \implies \langle r_{ji_a},r_{ji_b}\rangle<-\varepsilon$, and so
\begin{align*}
 0\le\left\|\sum_{a=1}^q r_{ji_a}\right\|^2
 &=\sum_{a=1}^q\|r_{ji_a}\|^2
   +2\sum_{1\le a<b\le q}\langle r_{ji_a},r_{ji_b}\rangle\\
 &\le q-2\varepsilon\binom q2
   \\
   &=q\bigl(1-\varepsilon(q-1)\bigr).
\end{align*}
For $q\ge1$, division by $q$ yields $q\le1+1/\varepsilon$; the bound also
holds if no operation is triggered at $j$. If $q_j$ is the number of
operations at center $j$, their total deletion charge is therefore
\begin{align*}
 W_{\rm edge}
 &\le\sum_j\sum_{a=1}^{q_j}\mu_j(E_{i_a j})\\
 &\le\sum_j q_j\tau
\\
&\le n\tau(1+1/\varepsilon).
\end{align*}
Here each $E_{i_a j}$ is evaluated in the graph at the time of its
deletion. Charging a vertex deletion only for edges still present at
that time similarly gives $W_{\rm vertex}\le nr_0$.

Every operation deletes a vertex or an edge, so the process terminates.
If no edge remained, all the initial weight would have been deleted,
and the three charges would give
\begin{align*}
 \frac n8
 &\le\sum_{\{i,j\}\in E}\mu_i(j)\\
 &\le\frac n{\log s}+n\tau(1+1/\varepsilon)+nr_0\\
 &=n(B_t+r_0)
   \\
   &=n\left(\frac18-B_t\right)
   <\frac n8,
\end{align*}
which is a contradiction. At termination, every retained vertex satisfies $\mu_i(N_{G'}(i))
 \ge r_0
   =\frac18-\frac2{\log s}-2\varepsilon^2-2\varepsilon
   =\frac18-O(\varepsilon)$.
   For a final oriented edge $i\to j$, the second operation is no longer
applicable, so $\mu_j(E_{ij})\ge\tau$. Also $i\in X_j$, and
\eqref{eq:app-spread} bounds the mass with inner product above $\lambda$.
For sufficiently large $n$, $p\le\tau/2$, whence
\begin{align*}
 \mu_j\{k\in N_{G'}(j):-\varepsilon\le\langle r_{ji},r_{jk}\rangle\le\lambda\} &=\mu_j(E_{ij})
   -\mu_j\{k\in N_{G'}(j):\langle r_{ji},r_{jk}\rangle>\lambda\}\\
 &\ge\tau-p\\ &\ge\frac\tau2.
\end{align*}
This proves~\eqref{eq:app-row} and~\eqref{eq:app-twowalk} in the same
final graph, with all their supporting neighbors still present.

<!-- lemma-13: body_md -->
\label{lem:weighted}
Let $(x_j)_{j\in J}$ be unit vectors with an $(s,\delta)$-packing
$\mu$ of total mass exactly $\delta>0$, and discard zero-mass indices.
Let $p=\overline\Phi(s)$, $\pi_j=\mu(j)/\delta$, and $a_j=\log\frac{p}{\mu(j)}$ with $\overline a=\sum_j\pi_j a_j
             =H(\pi)+\log\frac p\delta$. 
For each $j$, let $F_j(g)$ be the maximum of a fixed finite family
of Gaussian linear forms with coefficient vectors in $x_j^\perp$,
and write $m_j=\mathbb E F_j$ and $v_j=\operatorname{Var}(F_j)$.
For $0<\epsilon<1$, with probability at least $(1-\epsilon)\delta$
there is an index $j$ such that $\langle g,x_j\rangle\ge s$ and
\[F_j(g)\ge m_j-\sqrt{2v_j\bigl(a_j+\log(1/\epsilon)\bigr)}.\]
If also $m_j\ge m$, $v_j\le v$, and
$0<\eta<1-\epsilon$, then with probability at least
$(1-\eta-\epsilon)\delta$ there is a joint witness satisfying $\langle g,x_j\rangle\ge s$ and
\[F_j(g)\ge m-\sqrt{2v\bigl(\overline a/\eta+\log(1/\epsilon)\bigr)}.
\]
<!-- end lemma-13 -->
<!-- proof-7: body_md -->
The singleton packing inequalities and the identity
$\mu(j)=\delta\pi_j$ give $0<\mu(j)\le\Pr[\langle g,x_j\rangle\ge s]=p$ and $a_j=\log\frac p{\mu(j)}\ge0$. So, we have
\begin{align*}
\overline{a} &=\sum_j\pi_j\log\frac p{\delta\pi_j}\\
 &=\sum_j\pi_j\log\frac1{\pi_j}
   +\left(\sum_j\pi_j\right)\log\frac p\delta
   \\ &=H(\pi)+\log\frac p\delta.
\end{align*}
Define the outer-hit and inner-failure events by $A_j=\{\langle g,x_j\rangle\ge s\}$ and $ B_j=\left\{F_j(g)<m_j-\sqrt{2v_j\bigl(a_j+\log(1/\epsilon)\bigr)}\right\}$.
       
For fixed $j$, the coefficient vectors defining $F_j$ lie in
$x_j^\perp$. Thus $F_j$ depends only on the projection of $g$ onto
$x_j^\perp$, which is independent of $\langle g,x_j\rangle$. If $v_j>0$,
Lemma~\ref{lem:variance} therefore yields
\begin{align*}
 \Pr(A_j\cap B_j)
 &=\Pr(A_j)\Pr(B_j)\\
 &\le p\exp\!\left[-\frac{2v_j(a_j+\log(1/\epsilon))}{2v_j}\right]\\
 &=\epsilon p e^{-a_j}
   =\epsilon\mu(j).
\end{align*}
If $v_j=0$, then $F_j=m_j$ almost surely, so $B_j$ has probability
zero and the same bound holds. For a one-member inner family, the
tail estimate is the ordinary one-dimensional Gaussian estimate.

The outer packing gives $\Pr(\bigcup_j A_j)\ge\delta$. Moreover,
every outcome in $(\bigcup_j A_j)\setminus\bigcup_j(A_j\cap B_j)$
has an outer witness whose inner inequality succeeds. Consequently
\begin{align*}
 \Pr\!\left[\bigcup_j(A_j\cap B_j^c)\right]
 &\ge\Pr\!\left[\bigcup_j A_j\right]
       -\Pr\!\left[\bigcup_j(A_j\cap B_j)\right]\\
 &\ge\delta-\sum_j\Pr(A_j\cap B_j)\\
 &\ge\delta-\epsilon\sum_j\mu(j)
   =(1-\epsilon)\delta.
\end{align*}

For the uniform threshold, first suppose $\overline a>0$, and retain
$J_\eta=\{j:a_j\le\overline a/\eta\}$. The mass of the discarded
indices is bounded directly by
\begin{align*}
 \sum_{j\notin J_\eta}\mu(j)
 &=\delta\sum_{j\notin J_\eta}\pi_j\\
 &\le\delta\,\frac\eta{\overline a}
                  \sum_{j\notin J_\eta}\pi_j a_j\\
 &\le\delta\,\frac\eta{\overline a}
                  \sum_j\pi_j a_j
   =\eta\delta,
\end{align*} and \begin{align*}
 \Pr\!\left[\bigcup_{j\in J_\eta}A_j\right]
 &\ge\sum_{j\in J_\eta}\mu(j)
   \ge(1-\eta)\delta.
\end{align*}
If $\overline a=0$, positivity of the $\pi_j$ and nonnegativity of
the $a_j$ imply $a_j=0$ for every $j$; take $J_\eta=J$ instead.
In either case, on $B_j^c$ for a retained index,
\begin{align*}
 F_j(g)
 &\ge m_j-\sqrt{2v_j\bigl(a_j+\log(1/\epsilon)\bigr)}\\
 &\ge m-\sqrt{2v\bigl(\overline a/\eta+\log(1/\epsilon)\bigr)}.
\end{align*}
Subtracting the failures over the retained set gives
\begin{align*}
 \Pr\!\left[\bigcup_{j\in J_\eta}(A_j\cap B_j^c)\right]
 &\ge\Pr\!\left[\bigcup_{j\in J_\eta}A_j\right]
       -\sum_{j\in J_\eta}\Pr(A_j\cap B_j)\\
 &\ge(1-\eta)\delta
       -\epsilon\sum_{j\in J_\eta}\mu(j)\\
 &\ge(1-\eta-\epsilon)\delta,
\end{align*}
which is the second assertion.
