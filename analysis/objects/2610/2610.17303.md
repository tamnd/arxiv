<!-- theorem-1: body_md -->
\label{cor:two-exponents}
Let $2\le m<n$ be integers and let $A\in\boldsymbol B(\mathcal H)$. Then $A$ is
quasinormal if and only if
\[
  A^{*k}A^k=(A^*A)^k,
  \qquad k\in\{m,n\}.
\]
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:all-pairs}
Let $2\le m<n$ be integers and let $A\in\boldsymbol B(\mathcal H)$. Then $A$ is
quasinormal if and only if
\begin{equation}\label{eq:main-hypotheses}
  A^{*m}A^m\ge(A^*A)^m,
  \qquad
  A^{*n}A^n\le(A^*A)^n.
\end{equation}
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{hans}
Let $A\in\boldsymbol B(\mathcal H)$ be positive,
$T\in\boldsymbol B(\mathcal H)$ be a contraction, and
$f\colon[0,\infty)\to\mathbb R$ be a continuous operator monotone
function such that $f(0)\ge0$. Then
\begin{equation}\label{Han-inq}
  T^*f(A)T\le f(T^*AT).
\end{equation}
<!-- end theorem-3 -->
<!-- theorem-4: body_md -->
\label{bblem}
Let $T,B\in\boldsymbol B(\mathcal H)$, where $B\ge0$, and let
$\alpha,\beta$ be distinct positive real numbers. If $T^*T\le B$
and
\[
  T^*B^sT=B^{s+1},
  \qquad s\in\{\alpha,\beta\},
\]
then $B=T^*T$ and $T$ is quasinormal.
<!-- end theorem-4 -->
<!-- lemma-1: body_md -->
\label{lem:power-interpolation}
Let $B,T\in\boldsymbol B(\mathcal H)$, where $B\ge0$, and let $a,b,m>0$. If
$T^*T\le B^a$ and $T^*B^mT\le B^b$, then
\begin{equation}\label{eq:power-interpolation}
  T^*B^sT\le B^{(1-s/m)a+(s/m)b},
  \qquad 0<s\le m.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Fix $0<s\le m$ and put $\theta=s/m$.
Let $\#_\theta$ denote the weighted operator geometric mean,
extended to all positive operators by continuity from above.
The transformer inequality and monotonicity give
\begin{align*}
  T^*B^sT
  &=T^*(I\#_\theta B^m)T\\
  &\le (T^*T)\#_\theta(T^*B^mT)\\
  &\le B^a\#_\theta B^b
   =B^{(1-\theta)a+\theta b}.
\end{align*}
This proves \eqref{eq:power-interpolation}.
<!-- end proof-1 -->
<!-- proof-2: body_md -->
For $\varepsilon>0$, set $B_\varepsilon=B+\varepsilon I$ and
$Q_\varepsilon=TB_\varepsilon^{-a/2}$. Then
\[
  Q_\varepsilon^*Q_\varepsilon
  \le B_\varepsilon^{-a}B^a\le I,
\]
so $Q_\varepsilon$ is a contraction. Moreover,
\[
  Q_\varepsilon^*B^mQ_\varepsilon
  \le B_\varepsilon^{-a}B^b
  \le B_\varepsilon^{\,b-a}.
\]
Hence, by Hansen's inequality and the
L\"owner--Heinz inequality,
\[
  Q_\varepsilon^*B^sQ_\varepsilon
  \le \bigl(Q_\varepsilon^*B^mQ_\varepsilon\bigr)^{s/m}
  \le B_\varepsilon^{(b-a)s/m},
  \qquad 0<s\le m.
\]
Conjugating by $B_\varepsilon^{a/2}$ gives
\[
  T^*B^sT
  \le B_\varepsilon^{(1-s/m)a+(s/m)b}.
\]
Since $(1-s/m)a+(s/m)b>0$, letting $\varepsilon\downarrow0$
and using norm continuity of positive powers yields
\eqref{eq:power-interpolation}.
<!-- end proof-2 -->
<!-- proposition-1: body_md -->
\label{thm:consecutive}
Let $m\ge2$ be an integer  and let $A\in\boldsymbol B(\mathcal H)$. If $A^{*m}A^m=(A^*A)^m$ holds and
$A^{*(m+1)}A^{m+1}\le(A^*A)^{m+1}$, then $A$ is quasinormal.
<!-- end proposition-1 -->
<!-- proof-3: body_md -->
Let $A=U|A|$ be the polar decomposition of $A$, and define
$\Phi\colon \boldsymbol B(\mathcal H)\to \boldsymbol B(\mathcal H)$ by
$\Phi(T)=U^*TU$. By hypothesis,
$A^*(A^*A)^mA\le (A^*A)^{m+1}$, and hence
\[
|A|\Phi((A^*A)^m)|A|
   \le (A^*A)^{m+1}.
\]
Since $\Phi((A^*A)^m)$ vanishes on ${\mathscr N}(A^*A)$ and its range is contained in $\overline{\mathscr{R}(A^*A)}$, conjugating the above inequality by $(A^*A+\varepsilon I)^{-1/2}$ and passing to the limit as $\varepsilon$ tends to zero gives
\begin{align*}
    \Phi((A^*A)^m)\le (A^*A)^m.
\end{align*}
Therefore, applying Hansen's inequality to the operator monotone function
$f(t)=t^{s/m}$, and then using the operator monotonicity of this function, we obtain
\begin{equation}\label{eq:consecutive-comparison}
  \Phi((A^*A)^s)
  \le \Phi((A^*A)^m)^{s/m}
  \le (A^*A)^s,
  \qquad 0<s\le m.
\end{equation}
Multiplying on both sides by $|A|$ yields
$A^*(A^*A)^sA\le (A^*A)^{s+1}$. Since
$A^{*(j+1)}A^{j+1}=A^*(A^{*j}A^j)A$, 
the estimate above yields, by induction,
\[
A^{*(j+1)}A^{j+1}
\le A^*(A^*A)^jA
\le (A^*A)^{j+1},
\qquad 1\le j<m.
\]
Hence
\[
A^{*j}A^j\le (A^*A)^j,
\qquad 1\le j\le m.
\]
On the other hand, $A^{*m}A^m=(A^*A)^m,
$ gives
\[
|A|\Phi(A^{*(m-1)}A^{m-1})|A|
   =(A^*A)^m.
\]
The same regularization argument yields
$\Phi(A^{*(m-1)}A^{m-1})=(A^*A)^{m-1}$. Consequently,
\begin{equation}\label{eq:consecutive-fixed-moments}
\begin{aligned}
  (A^*A)^{m-1}
  &=\Phi(A^{*(m-1)}A^{m-1}) \\
  &\le \Phi((A^*A)^{m-1})
   \le \Phi((A^*A)^m)^{(m-1)/m}
   \le (A^*A)^{m-1}.
\end{aligned}
\end{equation}
Thus equality holds throughout. In particular,
$\Phi((A^*A)^s)=(A^*A)^s$ for $s=m-1,m$, and hence
$A^*(A^*A)^sA=(A^*A)^{s+1}$ for these two distinct positive exponents.
The conclusion follows from \Cref{bblem}, applied with
$T=A$ and $B=A^*A$.
<!-- end proof-3 -->
<!-- proof-4: body_md -->
The forward implication follows from Embry's characterization.
For the converse, by homogeneity we may and shall assume that $\|A\|\le1$.
Put $k=n-m$ and, for $1\le r\le n$, define
\begin{equation}\label{eq:optimal-exponents}
  p_r=\max\bigl\{
    p\in[1,r]: A^{*r}A^r\le(A^*A)^p
  \bigr\}.
\end{equation}
The maxima in \eqref{eq:optimal-exponents} are well defined. Indeed, since $A$ is a contraction, one has $A^{*r}A^r\le A^*A$, so the corresponding sets are nonempty.
 Moreover, the map $p\longmapsto (A^*A)^p$
is norm continuous on $[1,r]$. Hence each set in
\eqref{eq:optimal-exponents} is compact, and therefore its maximum is
attained. Consequently,
\[
  p_1=1,
  \qquad
  1\le p_r\le r,
  \qquad 1\le r\le n.
\]


We first establish two inequalities for the optimal exponents:
\begin{align}
  p_j
  &\ge
  \frac{(m-1)p_{j-1}+p_{m+j-1}}{m},
  &&2\le j\le k,
  \label{eq:optimal-feedback}\\
  p_{k+\ell}
  &\ge p_k+p_\ell,
  &&1\le\ell\le m.
  \label{eq:optimal-additivity}
\end{align}

Fix $2\le j\le k$. From the first inequality in
\eqref{eq:main-hypotheses} and the definition of
$p_{m+j-1}$, we obtain
\[
  (A^{j-1})^*(A^*A)^mA^{j-1}
  \le A^{*(m+j-1)}A^{m+j-1}
  \le (A^*A)^{p_{m+j-1}}.
\]
Applying \Cref{lem:power-interpolation} with
$T=A^{j-1}$, $B=A^*A$, $a=p_{j-1}$,
$b=p_{m+j-1}$, and $s=1$, yields
\[
  A^{*j}A^j
  \le
  (A^*A)^{((m-1)p_{j-1}+p_{m+j-1})/m}.
\]
Since
\[
  \frac{(m-1)p_{j-1}+p_{m+j-1}}{m}
  \le
  \frac{(m-1)(j-1)+(m+j-1)}{m}
  =j,
\]
the definition of $p_j$ gives
\eqref{eq:optimal-feedback}.

Next, let $1\le\ell\le m$. Since $k=n-m$, the two
inequalities in \eqref{eq:main-hypotheses} imply
\[
  (A^k)^*(A^*A)^mA^k
  \le A^{*n}A^n
  \le (A^*A)^n.
\]
As $0<p_\ell\le\ell\le m$, another application of
\Cref{lem:power-interpolation}, now with
$T=A^k$, $a=p_k$, $b=n$, and $s=p_\ell$, gives
\begin{align*}
  A^{*(k+\ell)}A^{k+\ell}
  &=(A^k)^*(A^{*\ell}A^\ell)A^k\\
  &\le (A^k)^*(A^*A)^{p_\ell}A^k\\
  &\le (A^*A)^{\,p_k+(n-p_k)p_\ell/m}\\
  &\le (A^*A)^{p_k+p_\ell}.
\end{align*}
Here the last inequality follows from
$n-p_k\ge n-k=m$ and $0\le A^*A\le I$.
Since $p_k+p_\ell\le k+\ell$, this proves
\eqref{eq:optimal-additivity}.

Set
\[
  \theta=\min_{1\le r\le n}\frac{p_r}{r}.
\]
We claim that $\theta=1$. Suppose otherwise, and let $j$ be the
smallest index for which $p_j=\theta j$. Since $p_1=1$, we have
$j\ge2$.

If $j\le k$, then $p_{j-1}>\theta(j-1)$, whereas
$p_{m+j-1}\ge\theta(m+j-1)$. Hence
\eqref{eq:optimal-feedback} gives
\[
  p_j
  \ge
  \frac{(m-1)p_{j-1}+p_{m+j-1}}{m}
  >
  \frac{(m-1)\theta(j-1)+\theta(m+j-1)}{m}
  =\theta j=p_j,
\]
a contradiction.

If $j>k$, then both $k$ and $j-k$ are strictly smaller than $j$,
and $1\le j-k\le m$. By the minimality of $j$,
\[
  p_k>\theta k,
  \qquad
  p_{j-k}>\theta(j-k).
\]
Using \eqref{eq:optimal-additivity}, we obtain
\[
  p_j
  \ge p_k+p_{j-k}
  >
  \theta k+\theta(j-k)
  =\theta j=p_j,
\]
again a contradiction. Thus $\theta=1$.

It follows that $p_r=r$ for every $1\le r\le n$. In particular,
\[
  A^{*m}A^m\le(A^*A)^m
  \quad\text{and}\quad
  A^{*(m+1)}A^{m+1}\le(A^*A)^{m+1}.
\]
The first of these, together with the first inequality in
\eqref{eq:main-hypotheses}, gives
$A^{*m}A^m=(A^*A)^m$. The conclusion now follows from
\Cref{thm:consecutive}.
