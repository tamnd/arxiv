<!-- definition-1: body_md -->
\label{def:packing_profile}
For a finite Borel measure $\mu$ on $\mathbb{R}^n$ and $s>0$, set
$$F_s^\mu(x,r)=\int_{\mathbb{R}^n}\min\{1,\,r^s|x-y|^{-s}\}\,d\mu(y),$$
and define the \emph{$s$-dimensional packing dimension profile} of $\mu$
by
$$\dim_P^s\mu=\sup\left\{t\geq 0 : \liminf_{r\to 0}\frac{F_s^{\mu}(x,r)}{r^{t}}=0
\ \text{ for }\mu\text{-a.e.\ }x\in\mathbb{R}^n\right\}.$$
<!-- end definition-1 -->
<!-- remark-1: body_md -->
\label{rem:profiles}
Let $\mu$ be a nonzero finite Borel measure on $\mathbb{R}^n$ and $s>0$.
\begin{enumerate}[(a)]
\item
$\liminf_{r\to0}r^{-t}F_s^\mu(x,r)=0$ for $\mu$-a.e.\ $x$ and every
$t<\dim_P^s\mu$.
\item If $0<s_1\leq s_2$, then
$\min\{1,r^{s_2}|z|^{-s_2}\}\leq\min\{1,r^{s_1}|z|^{-s_1}\}$, so that
$F_{s_2}^\mu\leq F_{s_1}^\mu$ and therefore
$\dim_P^{s_1}\mu\leq\dim_P^{s_2}\mu$.
\item $\dim_P^s\mu\leq\min\{s,\dim_P\mu\}$. Indeed,
$\mu(B(x,r))\leq F_s^\mu(x,r)$, so that \eqref{eq:hu} gives
$\dim_P^s\mu\leq\dim_P\mu$; and for $r\in(0,1)$ we have
$F_s^\mu(x,r)\geq r^s\int\min\{1,|x-y|^{-s}\}\,d\mu(y)$, where the
integral is positive, so that $\dim_P^s\mu\leq s$. Together with (b) and
\eqref{eq:FHcor3}, this gives $\dim_P^s\mu=\dim_P\mu$ for every $s\geq n$.
\item If $0<r\leq u$, then $\mathbf{1}_{B(0,u)}(z)\leq(u/r)^s\min\{1,r^s|z|^{-s}\}$
for every $z$, and therefore
\begin{equation}\label{eq:ball_profile}
\mu\bigl(B(x,u)\bigr)\leq\Bigl(\frac{u}{r}\Bigr)^{s}F_s^\mu(x,r)\qquad\text{for every }x\in\mathbb{R}^n.
\end{equation}
\item \emph{(Layer-cake formula.)} Let $\psi:[0,\infty)\to[0,1]$ be a
continuous nonincreasing function which is constant on $[0,r]$ for some
$r>0$, is piecewise continuously differentiable, and tends to $0$ at
infinity. Then $\psi(v)=\int_v^\infty(-\psi'(u))\,du$ for every $v\geq0$,
and by Tonelli's theorem
\begin{equation}\label{eq:layer_general}
\int\psi(|x-y|)\,d\mu(y)=\int_0^\infty(-\psi'(u))\,\mu\bigl(B(x,u)\bigr)\,du
\qquad\text{for every }x\in\mathbb{R}^n.
\end{equation}
Applied to $\psi(u)=\min\{1,r^su^{-s}\}$ this gives
\begin{equation}\label{eq:layer_profile}
F_s^\mu(x,r)=s\,r^s\int_r^\infty u^{-s-1}\mu\bigl(B(x,u)\bigr)\,du .
\end{equation}
\end{enumerate}

<!-- lemma-1: body_md -->
\label{lem:frostman}
If $0<d<\dim_H\mu$, then for $\mu$-almost every $x$ there is
$c_1=c_1(x)$ such that $\mu(B(x,r))\leq c_1r^d$ for all $r>0$. If $d=0$
this holds with $c_1=1$.
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{lem:regular}
Let $E\subset\mathbb{R}^n$ be a compact set and let $\dim_A E<\lambda\leq n$.
Then there exist $t\in(\dim_A E,\lambda)$, a compact set $F\subset\mathbb{R}^n$
with $E\subset F$, a Borel probability measure $\nu$ with $\operatorname{supp}\nu=F$, and
a constant $c\geq1$ such that
\begin{equation}\label{eq:nu_regular}
c^{-1}r^{t}\leq\nu\bigl(B(x,r)\bigr)\quad\text{for } x\in F,\ 0<r\leq1,
\qquad
\nu\bigl(B(x,r)\bigr)\leq c\,r^{t}\quad\text{for } x\in\mathbb{R}^n,\ 0<r\leq1.
\end{equation}
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
The statement is invariant under translations and contracting
dilations, up to the value of $c$, so we may assume that
$E\subset[0,1)^n$. Let $s=\dim_A E$, $\varepsilon:=\lambda-s>0$ and
$u:=s+\varepsilon/2<n$. For $j\in\mathbb{N}_0$, let $\mathcal{D}_j$
denote the family of half-open dyadic subcubes of $[0,1)^n$ of side
length $2^{-j}$, and for a bounded set $A$ let $\mathcal{N}_r(A)$ denote
the number of cubes in the mesh of side length $r$, oriented at the
origin, that intersect $A$. By the definition of the Assouad dimension,
there exists $C\geq 1$ such that for all $0<r<R$ and all $x\in E$,
\begin{equation}\label{eq:assouad_bound}
\mathcal{N}_r\bigl(E \cap Q(x,R)\bigr)\leq C \left(\frac{R}{r}\right)^{u},
\end{equation}
where $Q(x,R)$ denotes the cube of side length $R$ centred at $x$.

Fix a positive integer $g$ large enough such that
\begin{equation}\label{def:gap}
    M := \left\lceil C\,2^{u}\,2^{gu} \right\rceil < 2^{gn}
    \qquad\text{and}\qquad
    \frac{u+1+\log_2 C}{g}< \frac{\varepsilon}{2},
\end{equation}
and define $t:=\frac{\log_2 M}{g}$. Since $C\geq 1$, we have
$M\geq 2^{gu}$ and $M\leq C\,2^u\,2^{gu}+1\leq 2C\,2^u\,2^{gu}$, whence
$$s<u\leq t\leq u+\frac{u+1+\log_2 C}{g}<u+\frac{\varepsilon}{2}=\lambda;$$
moreover $t<n$, since $M<2^{gn}$. Set $r_k:=2^{-kg}$ for $k\geq 0$, and
note that
\begin{equation}\label{eq:exact_scaling}
    M^{-k}=r_k^{\,t}\qquad\text{for every }k\geq 0.
\end{equation}

\emph{Construction of $F$.} Let $\mathcal{F}_0:=\{[0,1)^n\}$. Suppose
that for some $k\geq 0$ we have constructed a family of dyadic cubes
$\mathcal{F}_k\subset\mathcal{D}_{kg}$ such that
$E\subset F_k:=\bigcup_{Q\in\mathcal{F}_k}Q$. For each
$Q\in\mathcal{F}_k$, partition $Q$ into its $2^{gn}$ dyadic children in
$\mathcal{D}_{(k+1)g}$, retain all children $Q'$ with
$E\cap Q'\neq\emptyset$, and then add further children of $Q$, chosen
arbitrarily, until exactly $M$ children of $Q$ have been selected. This
is always possible: on the one hand, if $Q\cap E\neq\emptyset$, pick
$x\in Q\cap E$; since $Q\subset Q(x,2r_k)$, inequality
\eqref{eq:assouad_bound} gives
\begin{equation}\label{ineq:children}
    \mathcal{N}_{r_{k+1}}(E\cap Q)
    \leq \mathcal{N}_{r_{k+1}}\bigl(E\cap Q(x,2r_k)\bigr)
    \leq C\,2^u\left(\frac{r_k}{r_{k+1}}\right)^{u}\leq M,
\end{equation}
so all children that intersect $E$ can be retained; on the other hand,
by \eqref{def:gap} we have $M<2^{gn}$, so there are always enough
children available to complete the selection. Let $\mathcal{F}_{k+1}$ be
the family of all selected children of all cubes of $\mathcal{F}_k$, and
$F_{k+1}:=\bigcup_{Q\in\mathcal{F}_{k+1}}Q$. Since every cube of
$\mathcal{D}_{(k+1)g}$ meeting $E$ is a child of some cube of
$\mathcal{F}_k$ (because $E\subset F_k$), we get $E\subset F_{k+1}$, and by
construction $F_{k+1}\subset F_k$ and each cube of $\mathcal{F}_k$
contains exactly $M$ cubes of $\mathcal{F}_{k+1}$; in particular
$\#\mathcal{F}_k=M^k$. Define
\[
F := \bigcap_{k=0}^{\infty} \overline{F_k},
\]
a compact set containing $E$. Since every cube of $\mathcal{F}_k$ has
exactly $M$ selected children in $\mathcal{F}_{k+1}$, setting
\[
\nu(Q):=M^{-k}\qquad \text{for every }Q\in\mathcal{F}_k
\]
defines a consistent mass distribution, which by
\cite[Proposition 1.7]{falconer2013fractal} extends to a Borel probability
measure $\nu$ with $\operatorname{supp}\nu\subset F$. In particular $\nu(F_k)=1$, so
that every cube of $\mathcal{D}_{kg}$ which does not belong to
$\mathcal{F}_k$ is $\nu$-null.

\emph{Upper bound.} Let $x\in\mathbb{R}^n$ and $0<r<1$, and choose $k\geq 0$ such
that $r_{k+1}\leq r<r_k$. Since $B(x,r)\subset Q(x,2r_k)$ and a cube of
side length $2r_k$ meets at most $3^n$ cubes of $\mathcal{D}_{kg}$, each
of which has $\nu$-mass at most $M^{-k}$, we obtain from
\eqref{eq:exact_scaling} and $r_k=2^g r_{k+1}\leq 2^g r$ that
$$\nu(B(x,r))\leq 3^n M^{-k}=3^n r_k^{\,t}\leq 3^n\,2^{gn}\, r^{\,t};$$
for $r=1$ the bound $\nu(B(x,1))\leq 1$ is trivial.

\emph{Lower bound.} Let $x\in F$ and $0<r\leq 1$, and choose $k\geq 0$
with $r_{k+1}\leq \frac{r}{2\sqrt n}<r_k$. Since
$x\in\overline{F_{k+1}}$, which is a finite union of closed cubes, there
exists $Q'\in\mathcal{F}_{k+1}$ with $x\in\overline{Q'}$; as $Q'$ has
diameter $\sqrt n\, r_{k+1}$, we get
$Q'\subset B(x,2\sqrt n\, r_{k+1})\subset B(x,r)$. Hence, using
\eqref{eq:exact_scaling}, $r_{k+1}>2^{-g}\frac{r}{2\sqrt n}$ and $t<n$,
$$\nu(B(x,r))\geq \nu(Q')=r_{k+1}^{\,t}
\geq \left(\frac{2^{-g}}{2\sqrt n}\right)^{n} r^{\,t}.$$
In particular $F\subset\operatorname{supp}\nu$, so that $\operatorname{supp}\nu=F$, and taking
$c:=\max\{3^n 2^{gn},\,(2^{g+1}\sqrt n)^{n}\}$ completes the proof.

<!-- lemma-3: body_md -->
\label{lem:growth}
Let $\mu$ be a Borel probability measure on $\mathbb{R}^n$, let $\lambda>N_\mu$
and let $a\in(0,1)$. Then for $\mu$-almost every $x$ there exists
$\rho_0(x)\in(0,1)$ such that
\begin{equation}\label{eq:ball_growth}
\mu\bigl(B(x,r)\bigr)\leq\left(\frac{4r}{\rho}\right)^{\lambda}\mu\bigl(B(x,\rho)\bigr)
\qquad\text{whenever } 0<\rho<\rho_0(x)\ \text{ and }\ \rho^{a}\leq r\leq 1 .
\end{equation}
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
Since $4r/\rho\geq 1$ in \eqref{eq:ball_growth}, a larger exponent only
weakens the inequality. If $\lambda>n$, this is
\cite[Corollary 2.3]{falconer1996packing}, stated there with the exponent
$n(1+\varepsilon)$, $\varepsilon=\lambda/n-1$; it holds for every Borel
probability measure on $\mathbb{R}^n$. Hence we may assume that
$N_\mu<\lambda\leq n$, so that $E:=\operatorname{supp}\mu$ is compact and
$\dim_A E<\lambda$. Let $t\in(\dim_A E,\lambda)$, $F\supset E$, $\nu$ and
$c\geq1$ be given by Lemma~\ref{lem:regular}, and fix $\eta>0$ with
$t(1+\eta)\leq\lambda$. It suffices to prove \eqref{eq:ball_growth} with
the exponent $t(1+\eta)$ in place of $\lambda$.

For nonnegative integers $p<q$ define
$$A_{p,q}:=\bigl\{x\in E:\ \mu(B(x,2^{-p}))>2^{(q-p)t(1+\eta)}\,\mu(B(x,2^{-q}))\bigr\}.$$
We claim that
\begin{equation}\label{eq:dyadic_bound}
\mu(A_{p,q})\leq 4^{t}c^{2}\,2^{-(q-p)t\eta}.
\end{equation}
First observe that for every $x\in\mathbb{R}^n$,
\begin{equation}\label{eq:local_bound}
\mu\bigl(A_{p,q}\cap B(x,2^{-q-1})\bigr)\leq
2^{(p-q)t(1+\eta)}\,\mu\bigl(B(x,2^{-p+1})\bigr).
\end{equation}
Indeed, if the left-hand side is zero there is nothing to prove;
otherwise pick $y\in A_{p,q}\cap B(x,2^{-q-1})$. Any two points of
$B(x,2^{-q-1})$ lie within distance $2^{-q}$ of each other, so
$A_{p,q}\cap B(x,2^{-q-1})\subset B(y,2^{-q})$, and since $y\in A_{p,q}$
means $\mu(B(y,2^{-q}))<2^{(p-q)t(1+\eta)}\mu(B(y,2^{-p}))$, we get
\begin{align*}
    \mu\bigl(A_{p,q}\cap B(x,2^{-q-1})\bigr)&\leq\mu(B(y,2^{-q}))
    <2^{(p-q)t(1+\eta)}\mu(B(y,2^{-p}))\\
    &\leq 2^{(p-q)t(1+\eta)}\mu(B(x,2^{-p+1})).
\end{align*}
Now we estimate $\mu(A_{p,q})$. By the lower bound in
\eqref{eq:nu_regular}, valid at every point of $E\subset F$,
$$\mu(A_{p,q})\leq c\,2^{(q+1)t}\int_{A_{p,q}}\nu\bigl(B(x,2^{-q-1})\bigr)\,d\mu(x).$$
Writing $\nu(B(x,2^{-q-1}))=\int\mathbf{1}_{\{|x-y|\leq 2^{-q-1}\}}\,d\nu(y)$
and applying Fubini's theorem and \eqref{eq:local_bound},
\begin{align*}
\int_{A_{p,q}}\nu\bigl(B(x,2^{-q-1})\bigr)\,d\mu(x)
&=\int \mu\bigl(A_{p,q}\cap B(y,2^{-q-1})\bigr)\,d\nu(y)\\
&\leq 2^{(p-q)t(1+\eta)}\int\mu\bigl(B(y,2^{-p+1})\bigr)\,d\nu(y).
\end{align*}
Applying Fubini's theorem once more and then the upper bound in
\eqref{eq:nu_regular} (for $p=0$ simply use $\nu(B(x,2))\leq1\leq c\,2^t$),
$$\int\mu\bigl(B(y,2^{-p+1})\bigr)\,d\nu(y)
=\int\nu\bigl(B(x,2^{-p+1})\bigr)\,d\mu(x)
\leq c\,2^{t}\,2^{-pt}.$$
Combining,
$$\mu(A_{p,q})\leq c\,2^{(q+1)t}\, 2^{(p-q)t(1+\eta)}\, c\,2^{t}\,2^{-pt}
= 4^{t}c^{2}\,2^{-(q-p)t\eta},$$
which proves \eqref{eq:dyadic_bound}.

Now, for $q_0\in\mathbb{N}$ let
$$
M_{q_0}:=\mu\Bigl(\bigcup\bigl\{A_{p,q}\,:\,p,q\in\mathbb{N}_0,\ 0\leq p\leq aq,\ q\geq q_0\bigr\}\Bigr).
$$
By \eqref{eq:dyadic_bound}, using $q-p\geq q-aq=(1-a)q$ and summing the
geometric series in $p$,
$$M_{q_0}\leq 4^{t}c^{2}\sum_{q=q_0}^{\infty}\sum_{p=0}^{\lfloor aq\rfloor}
2^{-(q-p)t\eta}
\leq 4^{t}c^{2}\,\frac{2^{t\eta}}{2^{t\eta}-1}\sum_{q=q_0}^{\infty}2^{-q(1-a)t\eta}
=c_{1}\,2^{-q_{0}(1-a)t\eta},$$
where $c_{1}$ depends only on $a,\eta,t$ and $c$.

Finally, let $\rho_{0}:=2^{-q_{0}-1}$, and suppose that $x\in E$,
$0<\rho<\rho_{0}$, $\rho^{a}\leq r\leq 1$ and
$$\mu(B(x,r))>\left(\frac{4r}{\rho}\right)^{t(1+\eta)}\mu(B(x,\rho)).$$
Choose integers $p,q$ with $2^{-p-1}<r\leq 2^{-p}$ and
$2^{-q}\leq\rho<2^{-q+1}$. Then $p\geq0$, $2^{-p}\geq r\geq\rho^{a}\geq
2^{-aq}$ gives $p\leq aq$, and $2^{-q}\leq\rho<2^{-q_{0}-1}$ gives
$q>q_{0}$. Moreover $\frac{4r}{\rho}\geq\frac{4\cdot 2^{-p-1}}{2^{-q+1}}
=2^{q-p}$, so
$$
2^{(q-p)t(1+\eta)}\mu(B(x,2^{-q}))\leq \left(\tfrac{4r}{\rho}\right)^{t(1+\eta)}\mu(B(x,\rho))
<\mu(B(x,r))\leq \mu(B(x,2^{-p})),
$$
that is, $x\in A_{p,q}$. Hence the set of points $x\in E$ for which
\eqref{eq:ball_growth} fails, with $t(1+\eta)$ in place of $\lambda$, for
some $0<\rho<\rho_{0}$ and $\rho^{a}\leq r\leq 1$, has $\mu$-measure at
most $M_{q_{0}}\leq c_{1}2^{-q_{0}(1-a)t\eta}$. These exceptional sets
decrease as $q_{0}\to\infty$ and their measures tend to $0$; hence their
intersection is $\mu$-null, which means that for $\mu$-a.e.\ $x$ there
exists $\rho_{0}(x)\in(0,1)$ such that
$\mu(B(x,r))\leq(4r/\rho)^{t(1+\eta)}\mu(B(x,\rho))$ for all
$0<\rho<\rho_{0}(x)$ and $\rho^{a}\leq r\leq 1$.

<!-- definition-2: body_md -->
\label{def:pid}
Let $\theta\in(0,1]$, $0\leq t\leq n$ and $r\in(0,1)$. For $z\in\mathbb{R}^n$ let
\[
\phi^{t,n}_{r,\theta}(z)=\min\bigl\{1,\ r^t|z|^{-t},\ r^{\theta(n-t)+t}|z|^{-n}\bigr\}
=\begin{cases}
1, & |z|\leq r,\\
r^t|z|^{-t}, & r<|z|\leq r^\theta,\\
r^{\theta(n-t)+t}|z|^{-n}, & |z|>r^\theta,
\end{cases}
\]
and write $\phi^{t,n}_{r,\theta}(u)$, $u\geq0$, for its value at any $z$
with $|z|=u$. For a nonzero finite Borel measure $\nu$ on $\mathbb{R}^n$ let
$$F^\nu_{t,n,\theta}(x,r)=\int\phi^{t,n}_{r,\theta}(x-y)\,d\nu(y),$$
and define the \emph{$\theta$-packing intermediate dimension} of $\nu$ by
\[
\dim_{P,\theta}\nu=\sup\Big\{t\in[0,n] : \liminf_{r\to0}r^{-t}F^\nu_{t,n,\theta}(x,r)=0\ \text{ for $\nu$-a.e.\ } x\in\mathbb{R}^n\Big\}.
\]
Since $0<\phi^{t,n}_{r,\theta}\leq1$, the definition makes sense for every
finite Borel measure, and its value does not change if $\nu$ is
multiplied by a positive constant.

<!-- remark-3: body_md -->
\label{rem:basic}
Let $\nu$ be a nonzero finite Borel measure on $\mathbb{R}^n$,
$\theta\in(0,1]$, $0\leq t\leq n$ and $r\in(0,1)$.
\begin{enumerate}[(a)]
\item If $0\leq t'\leq t$, then
\[
r^{-t'}\phi^{t',n}_{r,\theta}\leq r^{\theta(t-t')}\,r^{-t}\phi^{t,n}_{r,\theta}.
\]
Consequently, $\liminf_{r\to0}r^{-t}F^\nu_{t,n,\theta}(x,r)=0$ for
$\nu$-a.e.\ $x$ and every $t<\dim_{P,\theta}\nu$.
\item If $\theta\leq\theta'\leq1$, then
$\phi^{t,n}_{r,\theta'}\leq\phi^{t,n}_{r,\theta}$. Hence
$\theta\mapsto\dim_{P,\theta}\nu$ is nondecreasing on $(0,1]$.
\item $\phi^{t,n}_{r,1}(z)=\min\{1,r^n|z|^{-n}\}$, so that
$F^\nu_{t,n,1}=F^\nu_n$ and $\dim_{P,1}\nu=\dim_P^n\nu=\dim_P\nu$ by
\eqref{eq:FHcor3}.
\item If $u\in[r,r^\theta]$, then
$\mathbf{1}_{B(0,u)}\leq (u/r)^t\,\phi^{t,n}_{r,\theta}$, and therefore
\begin{equation}\label{eq:ballkernel}
\nu(B(x,u))\leq \Big(\frac ur\Big)^t F^\nu_{t,n,\theta}(x,r)\qquad\text{for every } x\in\mathbb{R}^n.
\end{equation}
\item For every $x\in\mathbb{R}^n$,
\begin{equation}\label{eq:layer}
F^\nu_{t,n,\theta}(x,r)=t\,r^t\int_r^{r^\theta}u^{-t-1}\nu(B(x,u))\,du+n\,r^{\theta(n-t)+t}\int_{r^\theta}^\infty u^{-n-1}\nu(B(x,u))\,du .
\end{equation}
\end{enumerate}

<!-- theorem-1: body_md -->
\label{thm:windows}
Let $\nu$ be a nonzero finite Borel measure on $\mathbb{R}^n$ and let
$\theta\in(0,1]$.
\begin{enumerate}[(i)]
\item If $0\leq t<\dim_{P,\theta}\nu$, then for $\nu$-almost every $x$
and every $\varepsilon>0$ there are arbitrarily small $\rho\in(0,1)$ such
that
\[
\nu(B(x,r))\leq \varepsilon\,r^t\qquad\text{for all } r\in[\rho,\rho^\theta].
\]
\item $\dim_{P,\theta}\nu$ is the supremum of the numbers $t\geq0$ with
the following property: for $\nu$-almost every $x$, for every
$\varepsilon>0$ and every $r_0>0$ there is $\rho\in(0,r_0)$ such that
$\nu(B(x,r))\leq\varepsilon r^t$ for all $r\in[\rho,\rho^\theta]$.
\item $\dim_{P,\theta}\nu$ is also the supremum of the numbers $t\geq0$
such that, for $\nu$-almost every $x$, there are arbitrarily small
$\rho\in(0,1)$ with $\nu(B(x,r))\leq r^t$ for all $r\in[\rho,\rho^\theta]$.
\end{enumerate}
<!-- end theorem-1 -->
<!-- proof-3: body_md -->
Statements (i) and (ii) do not change if $\nu$ is multiplied by a
positive constant, so in their proof we may assume that $\nu$ is a
probability measure.

(i) By Remark~\ref{rem:basic}(a), for $\nu$-almost every $x$ we have
$\liminf_{r\to0}r^{-t}F^\nu_{t,n,\theta}(x,r)=0$. For such $x$ and
$\varepsilon>0$ there are arbitrarily small $\rho\in(0,1)$ with
$F^\nu_{t,n,\theta}(x,\rho)\leq\varepsilon\rho^t$, and by
\eqref{eq:ballkernel}, $\nu(B(x,r))\leq(r/\rho)^t\varepsilon\rho^t=\varepsilon r^t$
for every $r\in[\rho,\rho^\theta]$.

(ii) Let $t^*$ denote the supremum in (ii). By (i),
$\dim_{P,\theta}\nu\leq t^*$. Conversely, let $t>0$ have the property in
(ii). Then $\liminf_{r\to0}r^{-t}\nu(B(x,r))=0$ for $\nu$-a.e.\ $x$, so
that $t\leq\dim_P\nu\leq n$ by \eqref{eq:hu} and \eqref{eq:chain}. Let
$0\leq t'<t$; we show that $t'\leq\dim_{P,\theta}\nu$. Fix
$\lambda\in(n,n+t-t')$ and $a\in(0,1)$ with $(1-a)\lambda<t-t'$, and let
$\rho_0(x)$ be given by Lemma~\ref{lem:growth} for these parameters
(note that $\lambda>n\geq N_\nu$). Fix $x$ in the intersection of the two
sets of full measure, let $\varepsilon\in(0,1)$, and choose $\rho\in(0,1)$
with $\rho^\theta<\rho_0(x)$ and $\nu(B(x,r))\leq\varepsilon r^t$ for all
$r\in[\rho,\rho^\theta]$. By \eqref{eq:layer}, with $t'$ in place of $t$,
we have $F^\nu_{t',n,\theta}(x,\rho)=J_1+J_2+J_3+J_4$, where $J_1$ is the
first term in \eqref{eq:layer} and $J_2,J_3,J_4$ are the parts of the
second term corresponding to the intervals $[\rho^\theta,\rho^{a\theta}]$,
$[\rho^{a\theta},1]$ and $[1,\infty)$. The window condition gives
\[
J_1\leq t'\varepsilon\rho^{t'}\int_\rho^{\rho^\theta}u^{t-t'-1}\,du\leq\frac{t'}{t-t'}\,\varepsilon\rho^{t'} .
\]
For $u\in[\rho^\theta,\rho^{a\theta}]$, applying \eqref{eq:ball_growth} at
the scale $\rho^\theta$ and then the window condition at $r=\rho^\theta$,
$$\nu(B(x,u))\leq\nu(B(x,\rho^{a\theta}))\leq 4^\lambda\rho^{-(1-a)\theta\lambda}\,\nu(B(x,\rho^\theta))\leq4^\lambda\varepsilon\,\rho^{\theta t-(1-a)\theta\lambda},$$
so that, since $\int_{\rho^\theta}^\infty u^{-n-1}du=\rho^{-\theta n}/n$,
\[
J_2\leq 4^\lambda\varepsilon\,\rho^{t'+\theta(t-t')-(1-a)\theta\lambda}\leq4^\lambda\varepsilon\rho^{t'} .
\]
For $u\in[\rho^{a\theta},1]$, in the same way,
$\nu(B(x,u))\leq(4u/\rho^\theta)^\lambda\nu(B(x,\rho^\theta))\leq4^\lambda\varepsilon u^\lambda\rho^{\theta(t-\lambda)}$,
and since $\lambda>n$,
\[
J_3\leq n\,\rho^{\theta(n-t')+t'}\,4^\lambda\varepsilon\,\rho^{\theta(t-\lambda)}\int_0^1u^{\lambda-n-1}\,du
=\frac{n4^\lambda}{\lambda-n}\,\varepsilon\,\rho^{t'+\theta(t-t')-\theta(\lambda-n)}\leq\frac{n4^\lambda}{\lambda-n}\,\varepsilon\rho^{t'} .
\]
Finally, $\nu(B(x,u))\leq1$ gives $J_4\leq\rho^{t'+\theta(n-t')}$.
Therefore
\[
F^\nu_{t',n,\theta}(x,\rho)\leq C\varepsilon\rho^{t'}+\rho^{t'+\theta(n-t')},\qquad
C=\frac{t'}{t-t'}+4^\lambda+\frac{n4^\lambda}{\lambda-n},
\]
where $C$ depends neither on $\varepsilon$ nor on $\rho$. Since $t'<n$
and $\rho$ can be taken arbitrarily small, we get
$\liminf_{r\to0}r^{-t'}F^\nu_{t',n,\theta}(x,r)\leq C\varepsilon$, and
since $\varepsilon$ was arbitrary, this $\liminf$ vanishes. This holds
for $\nu$-almost every $x$, so $t'\leq\dim_{P,\theta}\nu$. Letting
$t'\to t$ gives $t\leq\dim_{P,\theta}\nu$, and therefore
$t^*\leq\dim_{P,\theta}\nu$.

(iii) Let $t^\circ$ denote the supremum in (iii). By (i) with
$\varepsilon=1$, $\dim_{P,\theta}\nu\leq t^\circ$. Conversely, let $t$
have the property in (iii) and $t'<t$. For $r\in[\rho,\rho^\theta]$ we have
$\nu(B(x,r))\leq r^{t-t'}r^{t'}\leq\rho^{\theta(t-t')}r^{t'}$, and
$\rho^{\theta(t-t')}\to0$ as $\rho\to0$, so $t'$ has the property in
(ii). Hence $t'\leq\dim_{P,\theta}\nu$, and $t^\circ\leq\dim_{P,\theta}\nu$.

<!-- lemma-5: body_md -->
\label{lem:floor}
Let $\mu$ be a Borel probability measure on $\mathbb{R}^n$, and write
$P=\dim_P\mu$ and $N=N_\mu$. Then, for every $\theta\in(0,1]$,
\begin{equation}\label{eq:floor}
\dim_{P,\theta}\mu\ \geq\ \sigma_\theta(P):=\frac{\theta\,N\,P}{N-(1-\theta)P},
\end{equation}
where $\sigma_\theta(P):=0$ if $P=0$. Note that if $P>0$, then
$N-(1-\theta)P\geq\theta N>0$.
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
If $P=0$ there is nothing to prove, so assume $P>0$. Let $0<p<P$,
$\lambda>N$ and
\[
0<t<\frac{\theta\lambda p}{\lambda-(1-\theta)p}.
\]
Since $\theta\lambda\leq\lambda-(1-\theta)p$, we have $t<p$. A direct
computation shows that the condition on $t$ is equivalent to
$t(\lambda-p)<\theta p(\lambda-t)$, that is,
\[
\frac{\lambda-p}{\lambda-t}<\frac{\theta p}{t}.
\]
Fix $\kappa$ strictly between these two numbers and fix $a\in(0,1)$ with
$(1-a)\lambda<p-t$. Then the three exponents
\[
e_1=p-\frac{\kappa t}{\theta},\qquad e_2=p-(1-a)\lambda-t,\qquad e_3=\kappa(\lambda-t)-(\lambda-p)
\]
are positive. Since $p<\dim_P\mu$, by \eqref{eq:hu} for $\mu$-almost
every $x$ there are arbitrarily small $\delta$ with
$\mu(B(x,\delta))\leq\delta^p$, and Lemma~\ref{lem:growth} applies with
the parameters $\lambda$ and $a$. Fix such an $x$, let $\varepsilon>0$
and $r_0>0$, and choose $\delta\in(0,\rho_0(x))$, where $\rho_0(x)$ is
given by Lemma~\ref{lem:growth}, with $\mu(B(x,\delta))\leq\delta^p$,
$\delta^{\kappa/\theta}<r_0$ and
$4^\lambda\max\{\delta^{e_1},\delta^{e_2},\delta^{e_3}\}\leq\varepsilon$.
Put $\rho=\delta^{\kappa/\theta}$, so that the window is
$[\rho,\rho^\theta]=[\delta^{\kappa/\theta},\delta^\kappa]$, and let
$r\in[\rho,\rho^\theta]$.

If $r\leq\delta$ then, since $r\geq\delta^{\kappa/\theta}$,
\[
\mu(B(x,r))\leq\mu(B(x,\delta))\leq\delta^p=\delta^{e_1}\delta^{\kappa t/\theta}\leq\delta^{e_1}\,r^t .
\]
If $\delta<r\leq\delta^a$ then, by \eqref{eq:ball_growth} with radius
$\delta^a$,
\[
\mu(B(x,r))\leq\mu(B(x,\delta^a))\leq4^\lambda\delta^{(a-1)\lambda}\delta^p=4^\lambda\delta^{e_2}\delta^{t}\leq4^\lambda\delta^{e_2}\,r^t .
\]
If $\delta^a<r\leq\delta^\kappa$ then $r\leq1$ and, by
\eqref{eq:ball_growth} and $r\leq\delta^\kappa$,
\[
\mu(B(x,r))\leq4^\lambda\Big(\frac r\delta\Big)^\lambda\delta^p=4^\lambda r^t\,r^{\lambda-t}\delta^{p-\lambda}\leq4^\lambda r^t\delta^{\kappa(\lambda-t)+p-\lambda}=4^\lambda\delta^{e_3}\,r^t .
\]
Hence $\mu(B(x,r))\leq\varepsilon r^t$ for all $r\in[\rho,\rho^\theta]$,
and $\rho<r_0$. By Theorem~\ref{thm:windows}(ii), $t\leq\dim_{P,\theta}\mu$.
Letting $t\to\theta\lambda p/(\lambda-(1-\theta)p)$, then $p\to P$ and
$\lambda\to N$ gives the claim; the expression is continuous there
because $N-(1-\theta)P\geq\theta N>0$.

<!-- definition-3: body_md -->
\label{def:qH}
The \emph{quasi-Hausdorff dimension} of a nonzero finite Borel measure
$\nu$ on $\mathbb{R}^n$ is
$$\dim_{qH}\nu:=\lim_{\theta\to0}\dim_{P,\theta}\nu=\inf_{\theta\in(0,1]}\dim_{P,\theta}\nu .$$
<!-- end definition-3 -->
<!-- remark-5: body_md -->
\label{rem:qH}
\begin{enumerate}[(a)]
\item By Lemma~\ref{lem:compare},
$\dim_H\nu\leq\dim_{qH}\nu\leq\dim_P\nu$. Thus the packing intermediate
dimensions range from the packing dimension, attained at $\theta=1$, to
the quasi-Hausdorff dimension, which is at least the Hausdorff
dimension. The name is chosen by analogy with the quasi-Assouad
dimension, which is defined as a limit of the Assouad spectrum
\cite{fraser2018new}, and with the quasi-Hausdorff dimensions of sets
introduced in \cite{angelini2025critical}, which are the limits as
$\theta\to0$ of the intermediate dimensions of
\cite{falconer2020intermediate}; the latter coincide with the Hausdorff
dimension at $\theta=0$, but may be discontinuous there.
\item By Lemma~\ref{lem:floor}, $\dim_{qH}\mu=\dim_P\mu$ whenever
$\dim_P\mu=N_\mu$; for instance, if $\dim_P\mu=n$, or if $\operatorname{supp}\mu$ is
compact and $\dim_P\mu=\dim_A(\operatorname{supp}\mu)$. Thus the first inequality in
(a) is strict for every measure with $\dim_H\mu<\dim_P\mu=n$. The second inequality in (a) can also be strict, see
Examples \ref{ex:constant} and ~\ref{ex:FM} below.
\item By Proposition~\ref{prop:lipschitz}, $\dim_{qH}\nu_f\leq\dim_{qH}\nu$
for every Lipschitz map $f:\operatorname{supp}\nu\to\mathbb{R}^m$.
\end{enumerate}

<!-- theorem-2: body_md -->
\label{thm:threshold}
Let $\mu$ be a Borel probability measure on $\mathbb{R}^n$ and let $0<s\leq n$.
Then, for every $\theta\in(0,1]$,
\begin{equation}\label{eq:two_sided}
(1-\theta)\min\{s,\dim_{P,\theta}\mu\}\ \leq\ \dim_P^{s}\mu\ \leq\ (1-\theta)\,s+\theta\min\{s,\dim_{P,\theta}\mu\}.
\end{equation}
Consequently,
\begin{equation}\label{eq:qH_bounds}
\min\{s,\dim_{qH}\mu\}\ \leq\ \dim_P^{s}\mu\ \leq\ \min\{s,\dim_P\mu\},
\end{equation}
and
\begin{equation}\label{eq:threshold}
\dim_P^{s}\mu=s\quad\Longleftrightarrow\quad s\leq\dim_{qH}\mu .
\end{equation}
<!-- end theorem-2 -->
<!-- proof-6: body_md -->
We first prove the lower bound in \eqref{eq:two_sided}. For
$0\leq t\leq s$, $r\in(0,1)$ and $z\in\mathbb{R}^n$ we have
\begin{equation}\label{eq:kernel_comparison}
\min\{1,r^s|z|^{-s}\}\leq\phi^{t,n}_{r,\theta}(z)+r^{t(1-\theta)}.
\end{equation}
Indeed, if $|z|\leq r^\theta$, then
$\min\{1,r^s|z|^{-s}\}\leq\min\{1,r^t|z|^{-t}\}=\phi^{t,n}_{r,\theta}(z)$,
and if $|z|>r^\theta$, then
$\min\{1,r^s|z|^{-s}\}\leq r^{s(1-\theta)}\leq r^{t(1-\theta)}$.
Integrating \eqref{eq:kernel_comparison} with respect to $\mu$,
\begin{equation}\label{eq:comparison_profiles}
F_s^\mu(x,r)\leq F^\mu_{t,n,\theta}(x,r)+r^{t(1-\theta)}\qquad\text{for all }x\in\mathbb{R}^n,\ r\in(0,1).
\end{equation}
Let $0<t<\min\{s,\dim_{P,\theta}\mu\}$. By Remark~\ref{rem:basic}(a), for
$\mu$-a.e.\ $x$ there exists a sequence $r_k\to0$ with
$F^\mu_{t,n,\theta}(x,r_k)\leq r_k^t$, and \eqref{eq:comparison_profiles}
gives $F_s^\mu(x,r_k)\leq2\,r_k^{t(1-\theta)}$. Now observe that $F_s^\mu$
does not depend on $t$; therefore, for every $t'<t(1-\theta)$,
$$r_k^{-t'}F_s^{\mu}(x,r_k)\leq 2\,r_k^{\,t(1-\theta)-t'}
\longrightarrow 0\qquad (k\to\infty),$$
so that $\liminf_{r\to0}r^{-t'}F_s^\mu(x,r)=0$ for $\mu$-a.e.\ $x$, and
$t'\leq\dim_P^s\mu$. Letting $t'\to t(1-\theta)$ and then
$t\to\min\{s,\dim_{P,\theta}\mu\}$ gives the lower bound.

For the upper bound in \eqref{eq:two_sided}, since $\dim_P^s\mu\leq s$ by
Remark~\ref{rem:profiles}(c), it suffices to show that
\begin{equation}\label{eq:pid_from_profile}
\dim_{P,\theta}\mu\ \geq\ s-\frac{s-\dim_P^s\mu}{\theta},
\end{equation}
which is equivalent to $\dim_P^s\mu\leq(1-\theta)s+\theta\dim_{P,\theta}\mu$.
Let $0\leq t<\dim_P^s\mu$ and set $t':=s-(s-t)/\theta\leq t$; we may
assume that $t'>0$. By Remark~\ref{rem:profiles}(a), for $\mu$-a.e.\ $x$
and every $\varepsilon>0$ there are arbitrarily small $r\in(0,1)$ with
$F_s^\mu(x,r)\leq\varepsilon r^t$. For such $r$ and every $u\in[r,r^\theta]$,
\eqref{eq:ball_profile} gives
$$\mu(B(x,u))\leq\Bigl(\frac ur\Bigr)^{s}F_s^\mu(x,r)\leq\varepsilon\,u^{s}r^{t-s}
=\varepsilon\,u^{t'}\,u^{s-t'}r^{t-s}\leq\varepsilon\,u^{t'}r^{\theta(s-t')+t-s}=\varepsilon\,u^{t'},$$
where we used $u\leq r^\theta$, $s-t'\geq0$ and $\theta(s-t')=s-t$. Thus
$t'$ has the property in Theorem~\ref{thm:windows}(ii), and therefore
$t'\leq\dim_{P,\theta}\mu$. Letting $t\to\dim_P^s\mu$ gives
\eqref{eq:pid_from_profile}.

As $\theta\downarrow0$, $\dim_{P,\theta}\mu$ decreases to
$\dim_{qH}\mu$, so the lower bound in \eqref{eq:two_sided} gives
$\dim_P^s\mu\geq\min\{s,\dim_{qH}\mu\}$; the upper bound in
\eqref{eq:qH_bounds} is Remark~\ref{rem:profiles}(c). Finally, if
$s\leq\dim_{qH}\mu$, then \eqref{eq:qH_bounds} gives $\dim_P^s\mu=s$;
and if $s>\dim_{qH}\mu$, we may choose $\theta\in(0,1]$ with
$\dim_{P,\theta}\mu<s$, and then \eqref{eq:two_sided} gives
$\dim_P^s\mu\leq(1-\theta)s+\theta\dim_{P,\theta}\mu<s$.

<!-- corollary-1: body_md -->
\label{cor:kahane_profiles}
Let $\mu$ be a Borel probability measure on $\mathbb{R}^n$. The following are
equivalent:
\begin{enumerate}[(i)]
\item $\dim_{qH}\mu=\dim_P\mu$;
\item $\dim_P^s\mu=\min\{s,\dim_P\mu\}$ for every $s>0$;
\item $\dim_P\mu=0$, or $\dim_P^{s}\mu=s$ for $s=\dim_P\mu$.
\end{enumerate}

<!-- proposition-2: body_md -->
\label{prop:blocks}
Let $0<s\leq n$. Let $d$ satisfy either $0<d<d_H$ or $d=d_H=0$, and
assume $d<s$. Let $\theta\in(0,1]$, let $t$ satisfy $d<t<P_\theta$ and
$t\neq s$, and let $\lambda>N$ with $\lambda\neq s$. Then for $\mu$-almost
every $x$ there is $c=c(x)>0$ such that
\[
F_s^\mu(x,h)\ \leq\ c\,h^{\beta_\theta(d,t,\lambda)}
\]
for arbitrarily small values of $h>0$.
<!-- end proposition-2 -->
<!-- proof-8: body_md -->
By \eqref{eq:layer_profile}, $F_s^\mu(x,h)=s\,J_x(h)$, where
$J_x(h):=h^s\int_h^\infty u^{-s-1}\mu(B(x,u))\,du$, so we estimate
$J_x(h)$.

For fixed $d$ and $\lambda$ the function
$t\mapsto\beta_\theta(d,t,\lambda)$ is continuous and strictly increasing
on $(d,\lambda)$. Indeed, the second term in \eqref{eq:beta} equals
\[
\frac{s}{1+\theta\,\dfrac{s-d}{\lambda-d}\Big(\dfrac{\lambda}{t}-1\Big)} ,
\]
and the minimum of two continuous strictly increasing functions is
continuous and strictly increasing. Hence there is $t'\in(t,P_\theta)$
with $t'\neq s$ and $\beta_\theta(d,t',\lambda)>\beta_\theta(d,t,\lambda)$.
Let $w'$ be the number \eqref{eq:w} computed with $t'$, so that
$t'/w'=\beta_\theta(d,t',\lambda)$. Fix $a\in(0,1)$ with
$a\geq(\lambda-t')/(\lambda-d)$ and so close to $1$ that
$\eta=\theta(1-a)\lambda$ satisfies $(t'-\eta)/w'\geq\beta_\theta(d,t,\lambda)$.
Since $h^{e'}\leq h^e$ for $h\in(0,1)$ and $e'\geq e$, it suffices to prove
the following statement. \emph{For every admissible $t$ and every
$a\in(0,1)$ with $a\geq(\lambda-t)/(\lambda-d)$, for $\mu$-almost every
$x$ there is $c=c(x)$ such that}
\begin{equation}\label{eq:goal}
J_x(h)\leq c\,h^{(t-\eta)/w},\qquad \eta=\theta(1-a)\lambda,
\end{equation}
\emph{for arbitrarily small values of $h$.}

Now we will estimate the measure on balls centered in typical points.

By Lemma~\ref{lem:frostman},
Theorem~\ref{thm:windows}(i) and Lemma~\ref{lem:growth}, for $\mu$-almost
every $x$ the following hold simultaneously.
\begin{enumerate}
\item[(F)] $\mu(B(x,r))\leq c_1r^d$ for all $r>0$, with $c_1=c_1(x)$.
\item[(V)] There are arbitrarily small $\rho\in(0,1)$ with $\mu(B(x,r))\leq r^t$ for all $r\in[\rho,\rho^\theta]$.
\item[(D)] $\mu(B(x,r))\leq(4r/\varrho)^\lambda\mu(B(x,\varrho))$ for $0<\varrho<\rho_0(x)$ and $\varrho^a\leq r\leq1$.
\end{enumerate}
Fix such an $x$ and a number $\rho$ as in (V) with $\rho^\theta<\rho_0(x)$.
Applying (D) with $\varrho=\rho^\theta$, the top of the window, and using
$\mu(B(x,\rho^\theta))\leq\rho^{\theta t}$, which is (V) at
$r=\rho^\theta$, we obtain
\begin{enumerate}
\item[(L)] $\mu(B(x,r))\leq4^\lambda\big(r/\rho^\theta\big)^\lambda\rho^{\theta t}$ for all $r\in[\rho^{a\theta},1]$.
\end{enumerate}
The only selective condition is (V). Indeed, (F) holds at every scale and
(D) holds for every $\varrho<\rho_0(x)$ (Remark~\ref{rem:uniform}), so
(F), (V) and (L) hold simultaneously along a family of arbitrarily small
scales $\rho$, without matching two sequences of scales. In particular,
\begin{equation}\label{eq:topblock}
\mu(B(x,r))\leq\mu(B(x,\rho^{a\theta}))\leq4^\lambda\rho^{-\theta(1-a)\lambda}\rho^{\theta t}=4^\lambda\rho^{\theta t-\eta}\qquad\text{for } r\leq\rho^{a\theta}.
\end{equation}

First, suppose that $\lambda>s$. Put
\[
h=\rho^{w},\qquad R=\rho^{\theta(\lambda-t)/(\lambda-d)} .
\]
We distinguish two regimes. In \emph{regime II}, $t+\theta E\geq s$, we
have $h^s=\rho^{t+\theta E}$. In \emph{regime I}, $t+\theta E<s$, we have
$h=\rho$. Regime I can only occur if $t<s$ and $\theta<1$; indeed, $t>s$
implies $t+\theta E>s$, and for $\theta=1$ the identity \eqref{eq:Eid}
gives $t+E-s>0$ because $\lambda>s$ and $t>d$. The cut points satisfy
\[
h\ \leq\ \rho\ \leq\ \rho^\theta\ \leq\ \rho^{a\theta}\ \leq\ R\ \leq\ 1 .
\]
Here $h\leq\rho$ because $w\geq1$, $\rho^\theta\leq\rho^{a\theta}$
because $a<1$, and $\rho^{a\theta}\leq R$ is equivalent to
$a\geq(\lambda-t)/(\lambda-d)$, the same condition on $a$ as in
\cite[(2.4)]{falconer1996packing}. We shall use the inequality
\begin{equation}\label{eq:key}
h^s\rho^{\theta(t-s)}\leq\rho^t .
\end{equation}
If $t<s$ it follows from $h\leq\rho$, since
$\rho^{s+\theta(t-s)}=\rho^{t+(1-\theta)(s-t)}\leq\rho^t$. If $t>s$ we are
in regime II and $h^s\rho^{\theta(t-s)}=\rho^{t+\theta(E+t-s)}\leq\rho^t$
by \eqref{eq:Eid}. Each contribution below is of the form
$h^s\int_Iu^{-s-1}\mu(B(x,u))\,du$ over the indicated interval $I$.

We will break the integral over $5$ different blocks as follows.


\emph{Block 1, $[h,\rho]$, monotonicity from the foot of the window
(regime II only).} By (V) at $r=\rho$,
$\mu(B(x,u))\leq\mu(B(x,\rho))\leq\rho^t$ for $u\leq\rho$, and the lower
end dominates:
\[
I_1=h^s\int_h^\rho u^{-s-1}\mu(B(x,u))\,du\leq\rho^t\,h^s\int_h^\infty u^{-s-1}\,du=\frac1s\,\rho^t .
\]
In regime I we have $h=\rho$ and this block is absent.

\emph{Block 2, $[\rho,\rho^\theta]$, the window.} By (V) and $t\neq s$,
\[
I_2\leq h^s\int_\rho^{\rho^\theta}u^{t-s-1}\,du\leq\frac{h^s}{|t-s|}\max\{\rho^{t-s},\rho^{\theta(t-s)}\}\leq\frac{\rho^t}{|t-s|}.
\]
If $t<s$ the maximum is $\rho^{t-s}$ and we use $h\leq\rho$; if $t>s$ the
maximum is $\rho^{\theta(t-s)}$ and we use \eqref{eq:key}.

\emph{Block 3, $[\rho^\theta,\rho^{a\theta}]$, junction.} By
\eqref{eq:topblock} and \eqref{eq:key},
\[
I_3\leq4^\lambda\rho^{\theta t-\eta}\,h^s\int_{\rho^\theta}^\infty u^{-s-1}\,du=\frac{4^\lambda}{s}\,\rho^{-\eta}\,h^s\rho^{\theta(t-s)}\leq\frac{4^\lambda}{s}\,\rho^{t-\eta}.
\]
The loss $\rho^{-\eta}$ disappears as $a\to1$.

\emph{Block 4, $[\rho^{a\theta},R]$, the quasi-regular range.} By (L) and
$\lambda>s$ the upper end dominates:
\[
I_4\leq4^\lambda\rho^{\theta(t-\lambda)}\,h^s\int_0^Ru^{\lambda-s-1}\,du=\frac{4^\lambda}{\lambda-s}\,h^s\rho^{\theta(t-\lambda)}R^{\lambda-s}=\frac{4^\lambda}{\lambda-s}\,h^s\rho^{-\theta E},
\]
because $\theta(t-\lambda)+\theta(\lambda-t)(\lambda-s)/(\lambda-d)=-\theta(\lambda-t)(s-d)/(\lambda-d)=-\theta E$.

\emph{Block 5, $[R,\infty)$, the Frostman range.} By (F) and $d<s$ the
integral converges, and
\[
I_5\leq c_1h^s\int_R^\infty u^{d-s-1}\,du=\frac{c_1}{s-d}\,h^sR^{d-s}=\frac{c_1}{s-d}\,h^s\rho^{-\theta E}.
\]

Now, the choice of $R$ equalizes Blocks 4 and 5; it is the
solution of $\rho^{\theta(t-\lambda)}R^{\lambda-s}=R^{d-s}$. The choice of
$h$ equalizes their common value $h^s\rho^{-\theta E}$ with $\rho^t$. In
regime II, $h^s\rho^{-\theta E}=\rho^{t+\theta E-\theta E}=\rho^t$. In
regime I, $h=\rho$ and $h^s\rho^{-\theta E}=\rho^{s-\theta E}\leq\rho^t$,
because $s-\theta E>t$. Therefore, in both regimes,
\[
J_x(h)\leq I_1+\dots+I_5\leq c\,\rho^{t-\eta}=c\,h^{(t-\eta)/w},\qquad
c=\frac1s+\frac1{|t-s|}+\frac{4^\lambda}s+\frac{4^\lambda}{\lambda-s}+\frac{c_1}{s-d},
\]
since $\rho=h^{1/w}$. The constant depends on $x$ only through $c_1$ and the result follows.

Now, suppose that $\lambda<s$. 

Then
$t<P_\theta\leq N<\lambda<s$, $w=1$, and we put $h=\rho$.

\emph{Block 2, $[\rho,\rho^\theta]$.} As in the case $s<\lambda$, $I_2\leq\rho^t/(s-t)$.

\emph{Block 3, $[\rho^\theta,\rho^{a\theta}]$.} By \eqref{eq:topblock},
$I_3\leq(4^\lambda/s)\rho^{-\eta}\rho^{s+\theta(t-s)}\leq(4^\lambda/s)\rho^{t-\eta}$.

\emph{Block 4, $[\rho^{a\theta},1]$.} Now $\lambda<s$, so the lower end
dominates. By (L),
\[
I_4\leq4^\lambda\rho^{\theta(t-\lambda)}\rho^s\int_{\rho^{a\theta}}^\infty u^{\lambda-s-1}\,du=\frac{4^\lambda}{s-\lambda}\,\rho^{s+\theta(t-\lambda)+a\theta(\lambda-s)}\leq\frac{4^\lambda}{s-\lambda}\,\rho^{t},
\]
because $s+\theta(t-\lambda)+a\theta(\lambda-s)=t+(1-\theta)(s-t)+\theta(1-a)(s-\lambda)\geq t$.

\emph{Block 5, $[1,\infty)$.} Since $\mu(B(x,r))\leq1$,
$I_5\leq\rho^s\int_1^\infty u^{-s-1}\,du=\rho^s/s\leq\rho^t/s$.

Therefore $J_x(h)\leq c\,\rho^{t-\eta}=c\,h^{t-\eta}$ with
$c=1/(s-t)+4^\lambda/s+4^\lambda/(s-\lambda)+1/s$. Here neither (F) nor
the number $d$ is used.

\medskip
In both cases $\rho$ can be taken arbitrarily small, hence so can
$h=\rho^w$, and \eqref{eq:goal} is proved.

 \emph{The case $t=s$.} It is excluded only to avoid a logarithm in
Block 2. Since $t\mapsto\beta_\theta(d,t,\lambda)$ is continuous, this
exclusion costs nothing.

<!-- lemma-6: body_md -->
\label{lem:algebra}
Let $d_H<s\leq n$, $\lambda>d_H$, $\theta\in(0,1]$ and $0<t\leq\lambda$.
\begin{enumerate}[(i)]
\item $\displaystyle\frac{1}{G_\lambda(\theta,t)}=\frac1s+\frac{\theta(s-d_H)}{s(\lambda-d_H)}\Big(\frac\lambda t-1\Big)$. In particular $G_\lambda(\theta,t)$ is increasing in $t$ and nonincreasing in $\theta$.
\item $G_\lambda(1,t)\leq t$ if and only if $(\lambda-s)(t-d_H)\geq0$,
and the same holds with strict inequalities. If $\lambda\leq s$ and
$t\geq d_H$, then $G_\lambda(\theta,t)\geq t$ for every $\theta$.
\item If $P\leq\lambda$, then $G_\lambda(1,P)=\dfrac{P\,(1-d_H/\lambda)}{1+(1/s-1/\lambda)P-d_H/s}$; in particular $G_n(1,P)=\alpha_{FM}(s)$.
\item The function $\lambda\mapsto G_\lambda(1,P)$ is nonincreasing on
$[P,\infty)$, and strictly decreasing if $d_H<P$.
\item If $\lambda\geq\max\{s,P\}$, then $d\mapsto G_\lambda(1,P)$,
computed with $d$ in place of $d_H$, is nondecreasing on $[0,s)$.
\end{enumerate}
<!-- end lemma-6 -->
<!-- proof-9: body_md -->
(i) Divide the denominator of \eqref{eq:G} by $st(\lambda-d_H)$.
(ii) By (i), $G_\lambda(1,t)\leq t$ if and only if
$t(\lambda-d_H)+(s-d_H)(\lambda-t)\geq s(\lambda-d_H)$, and the
difference of the two sides equals $(\lambda-s)(t-d_H)$. If $\lambda\leq s$
and $t\geq d_H$ this gives $G_\lambda(1,t)\geq t$, and
$G_\lambda(\theta,t)\geq G_\lambda(1,t)$ by (i). (iii) Divide numerator
and denominator of $G_\lambda(1,P)$ by $s\lambda$ and use
$P(\lambda-d_H)+(s-d_H)(\lambda-P)=\lambda s+\lambda P-sP-\lambda d_H$.
(iv) By (i),

\label{thm:FMimproved}
Let $\mu$ be a Borel probability measure on $\mathbb{R}^n$ and let $0<s\leq n$
with $\dim_H\mu<s$. Then
\begin{equation}\label{eq:FMimproved}
\dim_P^s\mu\ \geq\ \sup_{0<\theta\leq1}\Lambda_\theta,\qquad
\Lambda_\theta=\min\Big\{P_\theta,\ \frac{s\,P_\theta\,(N-d_H)}{P_\theta(N-d_H)+\theta(s-d_H)(N-P_\theta)}\Big\},
\end{equation}
where $d_H=\dim_H\mu$, $P_\theta=\dim_{P,\theta}\mu$ and $N=N_\mu$. If
$N=d_H$, then $P_\theta=d_H$ for all $\theta$ and $\Lambda_\theta$ is to
be read as $d_H$.
<!-- end theorem-3 -->
<!-- proof-10: body_md -->
In the notation \eqref{eq:G} with $d=d_H$,
$\Lambda_\theta=\min\{P_\theta,G_N(\theta,P_\theta)\}$. Recall that
$d_H\leq P_\theta\leq P\leq N$.

First suppose that there exists $\theta$ with $P_\theta=d_H$. By
Theorem~\ref{thm:threshold} and Lemma~\ref{lem:compare},
$\dim_P^s\mu\geq\min\{s,\dim_{qH}\mu\}\geq\min\{s,d_H\}=d_H$. If
$P_\theta=d_H<N$, then
$G_N(\theta,d_H)=sd_H/(d_H+\theta(s-d_H))\geq d_H$, so that
$\Lambda_\theta=d_H$, and if $N=d_H$ then $\Lambda_\theta=d_H$ by
convention. Hence these values of $\theta$ are covered.

Now, fix $\theta$ with $P_\theta>d_H$. Choose $d$ with
$0<d<d_H$, or $d=0$ if $d_H=0$; choose $t$ with $d<t<P_\theta$ and
$t\neq s$; and choose $\lambda>N$ with $\lambda\neq s$, and with
$\lambda<s$ if $N<s$. By Proposition~\ref{prop:blocks}, for $\mu$-almost
every $x$ there are $c(x)>0$ and $h_k\downarrow0$ with
$F_s^\mu(x,h_k)\leq c(x)\,h_k^{\beta_\theta(d,t,\lambda)}$. Hence
$\liminf_{h\to0}h^{-\beta'}F_s^\mu(x,h)=0$ for every
$\beta'<\beta_\theta(d,t,\lambda)$, and therefore
$\dim_P^s\mu\geq\beta_\theta(d,t,\lambda)$.

Now, if $N<s$, then
$\beta_\theta(d,t,\lambda)=t$ (see after \eqref{eq:beta}); letting
$t\uparrow P_\theta$ gives $\dim_P^s\mu\geq P_\theta$, and
$P_\theta=\Lambda_\theta$ by Lemma~\ref{lem:algebra}(ii). If $N\geq s$,
we let $d\uparrow d_H$, $t\uparrow P_\theta$ and $\lambda\downarrow N$
along sequences with $s<\lambda$, $d<t$ and $t\neq s$. The function
$(d,t,\lambda)\mapsto\beta_\theta(d,t,\lambda)$ is continuous at
$(d_H,P_\theta,N)$, because the denominator
$t(\lambda-d)+\theta(s-d)(\lambda-t)$ is positive there (note that
$N\geq P_\theta>d_H$), and its value at this point is $\Lambda_\theta$.
Hence $\dim_P^s\mu\geq\Lambda_\theta$.

Finally, taking the supremum over $\theta\in(0,1]$ completes the
proof.

<!-- remark-8: body_md -->
\label{rem:FFS}
Falconer, Fraser and Shmerkin \cite[Theorem 2.2]{falconer2021assouad}
proved that if $F\subset\mathbb{R}^n$ is a Borel set, $s\in(0,n]$ and
$\theta\in(0,1)$, then
$$\dim_P^{s}F\ \geq\ \dim_P F-\max\bigl\{0,\ \overline{\dim}_A^{\theta}F-s,\
(\dim_A F-s)(1-\theta)\bigr\},$$
where $\overline{\dim}_A^{\theta}F\leq\dim_A F$ denotes the upper
Assouad spectrum \cite{fraser2020assouad}. Since the maximum above never
exceeds $\max\{0,\dim_A F-s\}$, this gives
\begin{equation}\label{eq:FFS_assouad_form}
\dim_P^{s}F\ \geq\ \dim_P F-\max\{0,\ \dim_A F-s\},
\end{equation}
which we refer to as the \emph{Assouad dimension form} of their theorem.
In \cite{falconer2021assouad} the profiles of sets are defined through
capacities; for integer exponents this definition coincides with the one
of Section~\ref{sec:prelim}, since both quantities equal the packing
dimension of $\pi_V F$ for $\gamma_{n,m}$-almost every $V\in G(n,m)$, see
\cite[Theorem 10]{falconer1997packing} and
\cite[Theorem 2.1]{falconer2021assouad}.

Corollary~\ref{cor:assouad} is not an adaptation of their argument, which
relies on the capacity formulation of the box dimension profiles, in which
the profile of a set is defined through an infimum over all measures
supported on it. This allows one to choose, at each scale $r$, a measure
adapted to that scale, and their proof takes it to be the normalized
counting measure on a maximal $r$-separated subset of $F$, the packing
case being deduced afterwards by a Baire category argument. In the present
setting no such freedom is available: the measure $\mu$ is given and is
the same at every scale, and the profile is defined through the behaviour
of its own potentials at $\mu$-almost every point. Our argument relies
instead on the regular measure of Lemma~\ref{lem:regular}, which replaces
the Lebesgue measure in the argument of Falconer and Mattila.

On the other hand, Corollary~\ref{cor:assouad} yields
\eqref{eq:FFS_assouad_form} for analytic sets. Indeed, let $E\subset\mathbb{R}^n$
be an analytic set. For every $\mu\in\mathcal{M}_c^+(E)$ we have
$\dim_A(\operatorname{supp}\mu)\leq\dim_A E$, so that
$$\dim_P^s\mu\geq\dim_P\mu-\max\{0,\ \dim_A E-s\},$$
and taking the supremum over $\mu\in\mathcal{M}_c^+(E)$, together with
\eqref{eq:dimP_sets} and the definition of $\dim_P^s E$, gives
$\dim_P^s E\geq\dim_P E-\max\{0,\ \dim_A E-s\}$.

\label{prop:criterion}
Let $\theta\in(0,1)$ and $t\in(0,N]$.
\begin{enumerate}[(a)]
\item $G_N(\theta,t)\geq G_N(1,P)$ if and only if $t\geq\sigma_\theta(P)$,
and the same holds with strict inequalities and with equalities. The
threshold $\sigma_\theta(P)$ depends neither on $s$ nor on $d_H$.
\item Assume $s<N$ and $P<N$, and put $G=G_N(1,P)=\Lambda_1$. Then
$d_H<G<P$, the function $\theta\mapsto\sigma_\theta(P)$ is increasing
with $\sigma_1(P)=P$, and the number
\[
\theta^\dagger=\frac{G\,(N-P)}{P\,(N-G)}\in(0,1)
\]
satisfies $\sigma_{\theta^\dagger}(P)=G$. Moreover $\Lambda_\theta>\Lambda_1$
if and only if $P_\theta>\max\{\sigma_\theta(P),G\}$, and for
$\theta\in[\theta^\dagger,1)$ this condition reduces to
$P_\theta>\sigma_\theta(P)$.
\item If $N\leq s$ or $P=N$, then $\Lambda_\theta\leq\Lambda_1$ for every
$\theta\in(0,1]$.
\end{enumerate}
<!-- end proposition-4 -->

(a) By Lemma~\ref{lem:algebra}(i), after cancelling the positive factor
$(s-d_H)/(s(N-d_H))$, the inequality $G_N(\theta,t)\geq G_N(1,P)$ is
equivalent to $\theta(N/t-1)\leq N/P-1$, that is, to
$\theta P(N-t)\leq t(N-P)$, that is, to $t(N-P+\theta P)\geq\theta NP$.
The strict version and the version with equalities are identical.

(b) $G<P$ follows from Lemma~\ref{lem:algebra}(ii), since $N>s$ and

$G_N(1,d_H)=d_H$. The derivative of $\theta\mapsto\sigma_\theta(P)$ is
$NP(N-P)/(N-P+\theta P)^2>0$. Solving $\theta NP=G(N-P+\theta P)$ gives
$\theta^\dagger$, and $0<\theta^\dagger<1$ because $0<G<P$. Finally,
$\Lambda_\theta>G$ means $P_\theta>G$ and $G_N(\theta,P_\theta)>G$, and

$\theta\geq\theta^\dagger$ we have $\sigma_\theta(P)\geq G$.

(c) If $N\leq s$, then $\Lambda_\theta=P_\theta\leq P=\Lambda_1$ by

$P=N$, then $\sigma_\theta(P)=P\geq P_\theta$, and (a) gives
$G_N(\theta,P_\theta)\leq G_N(1,P)$, so that
$\Lambda_\theta\leq\min\{P,G_N(1,P)\}=\Lambda_1$.

<!-- definition-4: body_md -->
\label{def:random_holder}
Let $(\Omega,\mathcal{F},\mathbb{P})$ be a probability space, let
$K\subset\mathbb{R}^n$ be a Borel set, $m\in\mathbb{N}$ and $0<\alpha\leq1$. A
family $\{f_\omega:K\to\mathbb{R}^m\}_{\omega\in\Omega}$ of continuous maps such
that $(\omega,x)\mapsto f_\omega(x)$ is
$\mathcal{F}\otimes\mathcal{B}(K)$-measurable is a \emph{random
$\alpha$-H\"{o}lder map} from $K$ to $\mathbb{R}^m$ if the following two
conditions hold:
\begin{enumerate}
\item[(H1)] for $\mathbb{P}$-almost every $\omega$ and every
$\beta\in(0,\alpha)$, $f_\omega$ is $\beta$-H\"{o}lder continuous on every
compact subset of $K$;
\item[(H2)] there exists $c>0$ such that, for all $x,y\in K$ and $r>0$,
$$\mathbb{P}\bigl(\{\omega\,:\,|f_\omega(x)-f_\omega(y)|\leq r\}\bigr)\leq
c\,\min\bigl\{1,\ r^{m}|x-y|^{-\alpha m}\bigr\}.$$
\end{enumerate}
<!-- end definition-4 -->

paths recalled in Section~\ref{sec:prelim}. 
Moreover, it was proved in \cite[Theorem 3.4]{Burrell} that $X$ satisfies (H2). Hence $X$ is a random

<!-- proposition-5: body_md -->
\label{prop:holder_upper}
Let $\mu$ be a Borel probability measure on $\mathbb{R}^n$, let $m\in\mathbb{N}$
and $0<\beta\leq1$, and let $f:\operatorname{supp}\mu\to\mathbb{R}^m$ be $\beta$-H\"{o}lder
continuous. Then
$$\dim_P\mu_f\ \leq\ \frac{1}{\beta}\,\dim_P^{\beta m}\mu .$$
Consequently, if $0<\alpha\leq1$ and $f$ is $\beta$-H\"{o}lder continuous
for every $\beta\in(0,\alpha)$, then
$\dim_P\mu_f\leq\frac{1}{\alpha}\dim_P^{\alpha m}\mu$.
<!-- end proposition-5 -->
<!-- proof-15: body_md -->
Let $c>0$ be such that $|f(x)-f(y)|\leq c\,|x-y|^{\beta}$ for all
$x,y\in\operatorname{supp}\mu$. Then, for $x,y\in\operatorname{supp}\mu$ and $r>0$,
$$\min\bigl\{1,\ r^m|x-y|^{-\beta m}\bigr\}\leq\min\bigl\{1,\ c^m r^m|f(x)-f(y)|^{-m}\bigr\}
\leq C\min\bigl\{1,\ r^m|f(x)-f(y)|^{-m}\bigr\},$$
with $C=\max\{1,c^m\}$, and integrating with respect to $\mu(y)$ we
obtain, by the definition of the image measure,
\begin{equation}\label{eq:holder_kernels}
F^\mu_{\beta m}\bigl(x,r^{1/\beta}\bigr)\leq C\,F^{\mu_f}_m\bigl(f(x),r\bigr)\qquad\text{for every }x\in\operatorname{supp}\mu .
\end{equation}
Let $u>\dim_P^{\beta m}\mu/\beta$. Since $\beta u>\dim_P^{\beta m}\mu$,
the set
$$A:=\Bigl\{x\in\operatorname{supp}\mu\,:\,\liminf_{\rho\to0}\rho^{-\beta u}F^\mu_{\beta m}(x,\rho)>0\Bigr\}$$
has positive $\mu$-measure, and for $x\in A$, by
\eqref{eq:holder_kernels},
\begin{align*}
\liminf_{r\to0}r^{-u}F^{\mu_f}_m\bigl(f(x),r\bigr)&\geq C^{-1}\liminf_{r\to0}r^{-u}F^\mu_{\beta m}\bigl(x,r^{1/\beta}\bigr)\\
&=C^{-1}\liminf_{\rho\to0}\rho^{-\beta u}F^\mu_{\beta m}(x,\rho)>0 .
\end{align*}
Hence
$A\subset f^{-1}\bigl(\{w:\liminf_{r\to0}r^{-u}F^{\mu_f}_m(w,r)>0\}\bigr)$,
the latter set has positive $\mu_f$-measure, and by
Remark~\ref{rem:profiles}(a), $\dim_P^m\mu_f\leq u$. As $\mu_f$ is a
measure on $\mathbb{R}^m$, \eqref{eq:FHcor3} gives $\dim_P\mu_f=\dim_P^m\mu_f\leq u$,
and since $u>\dim_P^{\beta m}\mu/\beta$ was arbitrary, the first claim
follows. If $f$ is $\beta$-H\"{o}lder continuous for every
$\beta\in(0,\alpha)$, then by Remark~\ref{rem:profiles}(b),
$\dim_P\mu_f\leq\dim_P^{\beta m}\mu/\beta\leq\dim_P^{\alpha m}\mu/\beta$
for every $\beta<\alpha$, and letting $\beta\to\alpha$ completes the
proof.

<!-- proposition-6: body_md -->
\label{prop:random_lower}
Let $(\Omega,\mathcal{F},\mathbb{P})$ be a probability space, $\mu$ a
Borel probability measure on $\mathbb{R}^n$, $m\in\mathbb{N}$ and
$0<\alpha\leq1$. Let $\{f_\omega:\operatorname{supp}\mu\to\mathbb{R}^m\}_{\omega\in\Omega}$ be
a family of continuous maps such that $(\omega,x)\mapsto f_\omega(x)$ is
$\mathcal{F}\otimes\mathcal{B}(\operatorname{supp}\mu)$-measurable and (H2) holds with
$K=\operatorname{supp}\mu$. Then
$$\dim_P\mu_{f_\omega}\ \geq\ \frac{1}{\alpha}\,\dim_P^{\alpha m}\mu\qquad\text{for }\mathbb{P}\text{-almost every }\omega\in\Omega.$$
<!-- end proposition-6 -->
<!-- proof-16: body_md -->
Let $0<t<\dim_P^{\alpha m}\mu$. By Remark~\ref{rem:profiles}(a), for
$\mu$-a.e.\ $x$ and every $\varepsilon>0$ there exists a sequence
$\rho_k\to0$ such that $F^\mu_{\alpha m}(x,\rho_k)<\varepsilon\,\rho_k^{t}$.
Fix $x$ in the corresponding full-measure set and put
$r_k:=\rho_k^{\alpha}$. By Fubini's theorem and (H2), for every $r>0$,
\begin{align*}
\int r^{-t/\alpha}\mu_{f_\omega}\bigl(B(f_\omega(x),r)\bigr)\,d\mathbb{P}(\omega)
&=r^{-t/\alpha}\int\mathbb{P}\bigl(\{\omega:|f_\omega(x)-f_\omega(y)|\leq r\}\bigr)\,d\mu(y)\\
&\leq c\,r^{-t/\alpha}\int\min\bigl\{1,\ r^m|x-y|^{-\alpha m}\bigr\}\,d\mu(y)\\
&=c\,r^{-t/\alpha}F^\mu_{\alpha m}\bigl(x,r^{1/\alpha}\bigr).
\end{align*}
Along the sequence $\{r_k\}$ the right-hand side is at most
$c\,\varepsilon$; hence, by Fatou's lemma,
$$\int\liminf_{r\to0}\, r^{-t/\alpha}\mu_{f_\omega}\bigl(B(f_\omega(x),r)\bigr)\,d\mathbb{P}(\omega)\leq c\,\varepsilon.$$
Note that the left-hand side no longer involves the sequence $\{r_k\}$,
which depended on $x$ and $\varepsilon$. Since $\varepsilon>0$ was
arbitrary, for $\mathbb{P}$-a.e.\ $\omega$ we have
$\liminf_{r\to0}r^{-t/\alpha}\mu_{f_\omega}(B(f_\omega(x),r))=0$. This
holds for $\mu$-a.e.\ $x$, so by Fubini's theorem, for $\mathbb{P}$-a.e.\
$\omega$ it holds for $\mu$-a.e.\ $x$. For such an $\omega$, the set
$Z_\omega$ of points $w\in\mathbb{R}^m$ with
$\liminf_{r\to0}r^{-t/\alpha}\mu_{f_\omega}(B(w,r))=0$ satisfies
$\mu_{f_\omega}(Z_\omega)=\mu(f_\omega^{-1}(Z_\omega))=1$, and therefore
$t/\alpha\leq\dim_P\mu_{f_\omega}$ by \eqref{eq:hu}. Letting
$t\to\dim_P^{\alpha m}\mu$ along a countable sequence and intersecting
the corresponding almost sure events completes the proof.

<!-- corollary-4: body_md -->
\label{cor:images_threshold}
Let $\mu$ be a Borel probability measure on $\mathbb{R}^n$, $m\in\mathbb{N}$ and
$0<\alpha\leq1$.
\begin{enumerate}[(a)]
\item If $f:\operatorname{supp}\mu\to\mathbb{R}^m$ is $\beta$-H\"{o}lder continuous for every
$\beta\in(0,\alpha)$ and $\dim_P\mu_f=m$, then $\alpha m\leq\dim_{qH}\mu$.
\item If $\operatorname{supp}\mu$ is compact and $\{f_\omega\}$ is a random
$\alpha$-H\"{o}lder map from $\operatorname{supp}\mu$ to $\mathbb{R}^m$, then
$$\dim_P\mu_{f_\omega}=m\ \text{ almost surely}\quad\Longleftrightarrow\quad\alpha m\leq\dim_{qH}\mu ,$$
and if $\alpha m>\dim_{qH}\mu$, then $\dim_P\mu_{f_\omega}<m$ almost
surely.
\end{enumerate}

(a) By Proposition~\ref{prop:holder_upper} and
Remark~\ref{rem:profiles}(c),
$m=\dim_P\mu_f\leq\dim_P^{\alpha m}\mu/\alpha\leq m$, so that
$\dim_P^{\alpha m}\mu=\alpha m$. Since $\dim_P^{\alpha m}\mu\leq\dim_P\mu\leq n$,
we have $\alpha m\leq n$, and \eqref{eq:threshold} gives
$\alpha m\leq\dim_{qH}\mu$.

(b) By Theorem~\ref{thm:images},

For every $\beta\in(0,1)$ there exists a Borel probability measure $\mu$ with compact support in $\mathbb{R}$ such that

<!-- proof-22: body_md -->
Let $\beta\in(0,1)$. For integers $u<v$ we write
$(u,v]=\{k\in\mathbb{N}:u<k\leq v\}$, and we let
\[
  S_\beta=\{k\in\mathbb{N}:\ \lfloor\beta k\rfloor>\lfloor\beta(k-1)\rfloor\},
\]
so that
$\#\bigl(S_\beta\cap(u,v]\bigr)=\lfloor\beta v\rfloor-\lfloor\beta u\rfloor$
for all integers $0\leq u<v$. Let $a_1=1$ and define recursively, for $j\geq 1$,
\[
  b_j=(j+1)\,a_j,\qquad
  c_j=\Bigl\lceil\frac{b_j}{1-\beta}\Bigr\rceil,\qquad
  a_{j+1}=(j+1)\,c_j .
\]
Then $a_j<b_j<c_j<a_{j+1}$, $a_j/b_j\to 0$ and $c_j/a_{j+1}\to 0$. Let
\[
  F=\bigcup_{j\geq 1}\Bigl((b_j,c_j]\ \cup\ \bigl((c_j,a_{j+1}]\cap S_\beta\bigr)\Bigr),
\]
let $(\varepsilon_k)_{k\geq 1}$ be independent random variables such that
$\varepsilon_k$ is uniformly distributed on $\{0,1\}$ if $k\in F$ and
$\varepsilon_k=0$ if $k\notin F$, and let $\mu$ be the distribution of
$X=\sum_{k\geq 1}\varepsilon_k 2^{-k}$. Thus the binary digits of $X$ vanish
on $[1,a_1]$ and on the blocks $(a_j,b_j]$, they are free on the blocks
$(b_j,c_j]$, and on the blocks $(c_j,a_{j+1}]$ they are free exactly at the
indices in $S_\beta$.

Let $\kappa(m)=\#(F\cap[1,m])$ be the number of free digits among the first
$m$, and let $\mathcal{D}_m$ be the family of dyadic intervals
$[i2^{-m},(i+1)2^{-m})$, $i\in\mathbb{Z}$.\\
Since
$\mathbb{N}\setminus F$ is infinite, every realization of
$(\varepsilon_k)$ has infinitely many zeros, so it is the binary expansion
of $X$ that does not end in an infinite string of ones. Hence, if
$0\leq i<2^m$ and $i=\sum_{k=1}^m d_k2^{m-k}$ with $d_k\in\{0,1\}$, then
$X\in[i2^{-m},(i+1)2^{-m})$ if and only if $\varepsilon_k=d_k$ for
$k=1,\dots,m$, and by independence
\[
  \mu\bigl([i2^{-m},(i+1)2^{-m})\bigr)=\prod_{k=1}^{m}\mathbb{P}(\varepsilon_k=d_k).
\]
Consequently $\mu(I)\in\{0,2^{-\kappa(m)}\}$ for every $I\in\mathcal{D}_m$.
Let $K=\{\sum_{k\in F}d_k2^{-k}: d_k\in\{0,1\}\}$, then $K$ is a compact set with
$\mu(K)=1$. If $x\in K$ and $I_m(x)$ denotes the interval of $\mathcal{D}_m$
containing $x$, then $\mu(I_m(x))=2^{-\kappa(m)}$. If
$2^{-m-1}\leq r<2^{-m}$, then $I_{m+1}(x)\subset B(x,r)$, and since $B(x,r)$
meets at most three intervals of $\mathcal{D}_m$ and 
$\kappa(m+1)\leq\kappa(m)+1$, we obtain
\begin{equation}\label{eq:exb-balls}
  2^{-\kappa(m)-1}\leq\mu(B(x,r))\leq 3\cdot 2^{-\kappa(m)}
  \qquad (x\in K,\ 2^{-m-1}\leq r<2^{-m}).
\end{equation}

 By construction,
\[
  \kappa(m)=
  \begin{cases}
    \kappa(a_j), & a_j\leq m\leq b_j,\\[2pt]
    \kappa(a_j)+m-b_j, & b_j\leq m\leq c_j,\\[2pt]
    \kappa(c_j)+\lfloor\beta m\rfloor-\lfloor\beta c_j\rfloor,
      & c_j\leq m\leq a_{j+1}.
  \end{cases}
\]
Since $b_j\leq(1-\beta)c_j<b_j+1$, we have
$\kappa(c_j)\geq c_j-b_j\geq\beta c_j$, and therefore
\begin{equation}\label{eq:exb-lower}
  \kappa(m)\geq\beta m-1\qquad (c_j\leq m\leq a_{j+1}).
\end{equation}
In the other direction, since $\kappa(a_j)\leq a_j$ and
$(1-\beta)m\leq(1-\beta)c_j<b_j+1$ for $m\leq c_j$, the last two lines of
the formula above give $\kappa(m)\leq\beta m+a_j+2$ for
$b_j<m\leq a_{j+1}$, by using that $\kappa(c_j)=\kappa(a_j)+c_j-b_j$. As $\kappa$ is constant on $[a_{j+1},b_{j+1}]$, then $\kappa(m)/m$ is at most $\kappa(a_{j+1})/a_{j+1}$ there, and it
follows that
\begin{equation}\label{eq:exb-upper}
  \frac{\kappa(m)}{m}\leq\beta+\eta_j\quad\text{for all } m>b_j,
  \qquad\text{where }\
  \eta_j=\sup_{i\geq j}\frac{a_i+2}{b_i}\xrightarrow[j\to\infty]{}0 .
\end{equation}
Finally, $\kappa(b_j)=\kappa(a_j)\leq a_j$, and $(b_j,c_j]\subset F$ with
$c_j-b_j\geq\beta c_j\to\infty$.

Now, we will prove that $\dim_H\mu=0$ and $\dim_P\mu=\beta$.

Since $E:=\operatorname{supp}\mu$ is covered
by the $2^{\kappa(b_j)}$ intervals of $\mathcal{D}_{b_j}$ with positive
measure. Hence, for every $s>0$,
\[
  \mathcal{H}^s_{2^{-b_j}}(E)\leq 2^{\kappa(b_j)-sb_j}
  \leq 2^{-b_j(s-a_j/b_j)}\xrightarrow[j\to\infty]{}0,
\]
so $\dim_H E=0$ and therefore $\dim_H\mu=0$. Now let $x\in E$. If
$t<\beta$, then \eqref{eq:exb-balls} with $r=2^{-a_{j+1}}$ (that is,
$m=a_{j+1}-1$) and \eqref{eq:exb-lower} give
\[
  r^{-t}\mu(B(x,r))\leq 3\cdot 2^{\beta+1}\,2^{-(\beta-t)a_{j+1}}
  \xrightarrow[j\to\infty]{}0 .
\]
If $t>\beta$, choose $j$ with $\eta_j<t-\beta$; then \eqref{eq:exb-balls}
and \eqref{eq:exb-upper} give

$$r^{-t}\mu(B(x,r))\geq 2^{tm-\kappa(m)-1}\geq 2^{(t-\beta-\eta_j)m-1}$$

whenever $2^{-m-1}\leq r<2^{-m}$ and $m>b_j$, and the right-hand side tends
to infinity as $r\to 0$. Since this holds at every point of $E$ we have $\dim_P\mu=\beta$.\\

Now we will show that  $\dim_{P,\theta}\mu\geq\beta$ for every $\theta\in(0,1]$.
Fix $\theta\in(0,1]$ and $s<\beta$, and let
$\delta_j=2^{-c_j-1}$. If $r\in[\delta_j^{1/\theta},\delta_j]$ and
$2^{-m-1}\leq r<2^{-m}$, then
\[
  c_j\leq m<\frac{c_j+1}{\theta}\leq a_{j+1}
\]
for all sufficiently large $j$, because $c_j/a_{j+1}\to 0$. Hence, by
\eqref{eq:exb-balls} and \eqref{eq:exb-lower}, for all $x\in E$ and
$r\in[\delta_j^{1/\theta},\delta_j]$ we have
\[
  r^{-s}\mu(B(x,r))\leq 2^{s(m+1)}\cdot 3\cdot 2^{-\beta m+1}
  \leq 12\cdot 2^{-(\beta-s)c_j}.
\]
Given $\epsilon>0$ and $\delta_0>0$, taking $\delta=\delta_j$ with $j$ large
enough we obtain
$\dim_{P,\theta}\mu\geq s$ for every $s<\beta$, that is,
$\dim_{P,\theta}\mu\geq\beta$.

Finally we will show that $\dim_A\mu=\dim_A E=1$. Let $x\in E$, $R=2^{-b_j}$ and
$r=2^{-c_j}$. The interval $I_{b_j}(x)$ is contained in $B(x,R)$, and, since
$(b_j,c_j]\subset F$, each of the $2^{c_j-b_j}$ intervals of
$\mathcal{D}_{c_j}$ contained in $I_{b_j}(x)$ has positive measure and hence
meets $E$. Since every ball of radius $r$ meets at most three of them,
\[
  N_r\bigl(E\cap B(x,R)\bigr)\geq\tfrac13\,2^{c_j-b_j}=\tfrac13\,\frac{R}{r},
\]
where $N_r(A)$ is the least number of balls of radius $r$ needed to cover
$A$. Similarly, \eqref{eq:exb-balls} and $b_j\notin F$ give
\[
  \frac{\mu(B(x,R))}{\mu(B(x,r))}
  \geq\frac{2^{\kappa(c_j-1)-\kappa(b_j-1)-1}}{3}
  =\frac{2^{c_j-b_j-2}}{3}=\frac{1}{12}\,\frac{R}{r}.
\]
As $c_j-b_j\to\infty$, it follows that $\dim_A E\geq 1$ and
$\dim_A\mu\geq 1$. The inequality $\dim_A E\leq 1$ is trivial, and
\eqref{eq:exb-balls} also gives $\mu(B(x,R))\leq 12\,(R/r)\,\mu(B(x,r))$ for
all $x\in E$ and $0<r<R<1$, so $\dim_A\mu\leq 1$.

Example~\ref{ex:constant}, and let $\nu=\mu^{\otimes n}$ be the product of
$n$ copies of $\mu$, that is, the distribution of $(X_1,\dots,X_n)$, where

\end{gather*}

of Example~\ref{ex:constant}, let $x=(x_1,\dots,x_n)$ be a point of $E^n$,
and let $2^{-\ell-1}\leq r<2^{-\ell}$. Since
\begin{gather*}
B(x,r)\subset\prod_{i=1}^nB(x_i,r)\subset B(x,\sqrt n\,r),
\end{gather*}
we have 
\[
c^{-1}2^{-n\kappa(\ell)}\leq\nu(B(x,r))\leq c\,2^{-n\kappa(\ell)},
\]
where $c\geq1$ depends only on $n$. This is \eqref{eq:exb-balls} with
$\kappa$ replaced by $n\kappa$, and the rest of the verification of
Example~\ref{ex:constant} applies with dyadic cubes in place of dyadic
intervals; only the constants change.

Now let $1\leq m<n$ be an integer. For every $V\in G(n,m)$ we have
$\dim_P\nu_V\leq\min\{\dim_P\nu,m\}$. Since $\dim_{P,\theta}\nu=\dim_P\nu$
for every $\theta\in(0,1]$ we have the reverse
inequality for $\gamma_{n,m}$-almost every $V$. Therefore, for
$\gamma_{n,m}$-almost every $V\in G(n,m)$,
\begin{equation}\label{eq:constant-Rn-proj}
\dim_P\nu_V=\min\{\dim_P\nu,m\}=\min\{n\beta,m\}.
\end{equation}
This is the exact analogue, for packing dimension, of the Hausdorff dimension projection

$\dim_H\nu=0$, the lower bound of Falconer and Mattila

\[
\dim_P\nu_V\geq\frac{\dim_P\nu}{1+\bigl(\frac1m-\frac1n\bigr)\dim_P\nu}
=\frac{mn\beta}{m+(n-m)\beta},
\]
and the right-hand side is strictly smaller than $\min\{n\beta,m\}$. Hence, for $\nu$, Theorem~\ref{thm:FMimproved} gives the exact

\label{ex:FM}
Let $0<d<p<n$ and let $\mu$ be the measure of
\cite[Example 5.1]{falconer1996packing}, for which $\dim_H\mu=d$ and
$\dim_P\mu=p$. We claim that
\begin{equation}\label{eq:example}
\dim_{P,\theta}\mu=\max\Big\{d,\ \frac{\theta np}{n-(1-\theta)p}\Big\}\qquad\text{for every }\theta\in(0,1],
\end{equation}
so that $\dim_{qH}\mu=\dim_H\mu=d<\dim_P\mu$, and the lower bound
\eqref{eq:floor}, with $N=n$, is attained for every $\theta$ for which it
is at least $d$.

We verify \eqref{eq:example}. Let $F=\operatorname{supp}\mu\subset[0,1]^n$, let
$r_k<R_k$ be the side lengths of the construction, with $R_k^p=r_k^d$, and
let $m_k$ be the integers defined in \cite[(5.1)]{falconer1996packing}. The
proof of \cite[Example 5.1(a)]{falconer1996packing} shows that there are
constants $c\geq1$ and $k_0$ such that, for $x\in F$ and $k\geq k_0$,
\begin{align}
c^{-1}r_k^d\leq\mu(B(x,u))&\leq c\,r_k^d
  &&\text{for } \sqrt n\,r_k\leq u\leq R_k,\label{eq:FM-plateau}\\
c^{-1}u^nR_{k+1}^{p-n}\leq\mu(B(x,u))&\leq c\,u^nR_{k+1}^{p-n}
  &&\text{for } R_{k+1}<u\leq\sqrt n\,r_k.\label{eq:FM-ramp}
\end{align}
That is, $\mu(B(x,u))$ is essentially constant for
$\sqrt n\,r_k\leq u\leq R_k$, and it grows like $u^n$ for
$R_{k+1}<u\leq\sqrt n\,r_k$. Moreover, by \cite[(5.2)]{falconer1996packing}
and $R_k^p=r_k^d$ we have $R_{k+1}^p=r_k^d/m_k$, while
$m_k>r_k^{-n(p-d)/(n-p)}$ by \cite[(5.1)]{falconer1996packing}; hence
\begin{equation}\label{eq:FM-scales}
R_{k+1}<r_k^{\frac{n-d}{n-p}}\qquad\text{for every } k.
\end{equation}

\emph{Upper bound.} Since $\dim_{P,\theta}\mu\leq\dim_P\mu=p$, we may assume
that $\theta<1$, so that $\max\{d,\theta np/(n-(1-\theta)p)\}<p$. Fix $e$
with $\max\{d,\theta np/(n-(1-\theta)p)\}<e<p$ and a constant $C\geq1$. Then
$\frac{n-p}{n-e}<1<\frac pe$, and
\begin{equation}\label{eq:FM-exponents}
\frac{n-d}{n-e}>1,\qquad \frac pd>\frac pe,\qquad
\frac{n-p}{n-e}>\frac{\theta p}{e};
\end{equation}
the first two inequalities hold because $e>d$, and the third one is
equivalent to $e>\theta np/(n-(1-\theta)p)$. For $k\geq k_0$ let
\[
I_k=\Big[(cC)^{\frac1{n-e}}R_{k+1}^{\frac{n-p}{n-e}},\ (cC)^{-\frac1e}R_k^{\frac pe}\Big].
\]
We claim that, for every $x\in F$ and every sufficiently large $k$,
\begin{equation}\label{eq:FM-bad}
\mu(B(x,u))\geq C\,u^e\qquad\text{for all } u\in I_k .
\end{equation}
Indeed, by \eqref{eq:FM-scales} and the first two inequalities in
\eqref{eq:FM-exponents}, for $k$ large we have
\[
(cC)^{\frac1{n-e}}R_{k+1}^{\frac{n-p}{n-e}}
\leq(cC)^{\frac1{n-e}}r_k^{\frac{n-d}{n-e}}\leq\sqrt n\,r_k
=\sqrt n\,R_k^{\frac pd}\leq(cC)^{-\frac1e}R_k^{\frac pe}\leq R_k .
\]
If $u\in I_k$ and $u\geq\sqrt n\,r_k$, then \eqref{eq:FM-plateau} gives
$\mu(B(x,u))\geq c^{-1}r_k^d=c^{-1}R_k^p\geq Cu^e$. If $u\in I_k$ and
$u\leq\sqrt n\,r_k$, then $u>R_{k+1}$, because $\frac{n-p}{n-e}<1$ and
$cC\geq1$, so \eqref{eq:FM-ramp} gives
\[
\mu(B(x,u))\geq c^{-1}u^e\,u^{n-e}R_{k+1}^{p-n}
\geq c^{-1}u^e\,(cC)R_{k+1}^{n-p}R_{k+1}^{p-n}=C\,u^e .
\]
This proves \eqref{eq:FM-bad}. On the other hand, the third inequality in
\eqref{eq:FM-exponents} gives, for $k$ large,
\begin{equation}\label{eq:FM-gap}
\Big((cC)^{-\frac1e}R_{k+1}^{\frac pe}\Big)^{\theta}
\geq(cC)^{\frac1{n-e}}R_{k+1}^{\frac{n-p}{n-e}} .
\end{equation}
Fix $k_1\geq k_0$ such that \eqref{eq:FM-bad} and \eqref{eq:FM-gap} hold for
all $k\geq k_1$, let $x\in F$ and let
$0<\rho<(cC)^{-1/e}R_{k_1}^{p/e}$. Choose $k\geq k_1$ such that
$(cC)^{-1/e}R_{k+1}^{p/e}<\rho\leq(cC)^{-1/e}R_k^{p/e}$. If $\rho\in I_k$,
we take $u=\rho$. Otherwise $\rho<u$, where
$u=(cC)^{1/(n-e)}R_{k+1}^{(n-p)/(n-e)}$ is the left endpoint of $I_k$, and
by \eqref{eq:FM-gap},
\[
\rho<u\leq\Big((cC)^{-\frac1e}R_{k+1}^{\frac pe}\Big)^{\theta}<\rho^\theta .
\]
In both cases $[\rho,\rho^\theta]$ contains a scale $u\in I_k$, so that
$\mu(B(x,u))\geq Cu^e$ by \eqref{eq:FM-bad}. Since this holds for every
$x\in F$, every sufficiently small $\rho$ and every $C\geq1$, the window
condition of Theorem~\ref{thm:windows} with exponent $e$ fails at every
point of $F$, and therefore $\dim_{P,\theta}\mu\leq e$. Letting $e$ decrease
to $\max\{d,\theta np/(n-(1-\theta)p)\}$ gives the upper bound in
\eqref{eq:example}.

\emph{Lower bound.} It follows from Lemmas~\ref{lem:compare} and
\ref{lem:floor}, since $\sigma_\theta(p)$ is nonincreasing as a function of
$N$ and $N_\mu\leq n$.

In particular, $\dim_A(\operatorname{supp}\mu)=n$. Indeed, if $N=\dim_A(\operatorname{supp}\mu)$ were
smaller than $n$, Lemma~\ref{lem:floor} would give, for $\theta<1$ close
to $1$,
\[
\dim_{P,\theta}\mu\geq\frac{\theta Np}{N-(1-\theta)p}
>\frac{\theta np}{n-(1-\theta)p}=\dim_{P,\theta}\mu .
\]

<!-- theorem-5: body_md -->
\label{thm:sets_profile}
Let $E\subset\mathbb{R}^n$ be an analytic set and let $0<s\leq n$. Then
$$\dim_P^s E=s\quad\Longleftrightarrow\quad s\leq\lim_{\theta\to0}\dim_{P,\theta}E .$$
<!-- end theorem-5 -->
<!-- proof-23: body_md -->
Write $L:=\lim_{\theta\to0}\dim_{P,\theta}E$. Suppose first that $s\leq L$,
and fix $\theta\in(0,1)$ and $\delta>0$. Since $\dim_{P,\theta}E\geq L\geq s$,
there exists $\mu\in\mathcal{M}_c^+(E)$, which we may normalize to be a
probability measure, with $\dim_{P,\theta}\mu\geq s-\delta$. By
Theorem~\ref{thm:threshold} applied to $\mu$,
$$\dim_P^{s}\mu\geq(1-\theta)\min\{s,\dim_{P,\theta}\mu\}\geq(1-\theta)(s-\delta),$$
and therefore $\dim_P^{s}E\geq(1-\theta)(s-\delta)$. Since
$\theta\in(0,1)$ and $\delta>0$ were arbitrary, $\dim_P^{s}E\geq s$, and
the reverse inequality follows from Remark~\ref{rem:profiles}(c).

Conversely, suppose that $\dim_P^s E=s$, and fix $\theta\in(0,1]$ and
$\varepsilon>0$. There exists $\mu\in\mathcal{M}_c^+(E)$, normalized to be
a probability measure, with $\dim_P^s\mu>s-\varepsilon$, and by the upper
bound in \eqref{eq:two_sided},
$$s-\varepsilon<\dim_P^s\mu\leq(1-\theta)\,s+\theta\min\{s,\dim_{P,\theta}\mu\},$$
that is, $\dim_{P,\theta}E\geq\dim_{P,\theta}\mu>s-\varepsilon/\theta$.
Letting $\varepsilon\to0$ gives $\dim_{P,\theta}E\geq s$ for every
$\theta\in(0,1]$, and therefore $L\geq s$.

<!-- theorem-6: body_md -->
\label{thm:sets_images}
Let $E\subset\mathbb{R}^n$ be an analytic set, $m\in\mathbb{N}$ and
$0<\alpha\leq1$, and let $\{f_\omega:\mathbb{R}^n\to\mathbb{R}^m\}$ be a random
$\alpha$-H\"{o}lder map from $\mathbb{R}^n$ to $\mathbb{R}^m$. Then
\begin{equation}\label{eq:sets_xiao}
\dim_P f_\omega(E)=\frac{1}{\alpha}\,\dim_P^{\alpha m}E\qquad\text{almost surely}.
\end{equation}
Consequently,
$$\dim_P f_\omega(E)=m\ \text{ almost surely}\quad\Longleftrightarrow\quad\alpha m\leq\lim_{\theta\to0}\dim_{P,\theta}E,$$
and otherwise $\dim_P f_\omega(E)<m$ almost surely.
<!-- end theorem-6 -->
<!-- proof-24: body_md -->
For every $\mu\in\mathcal{M}_c^+(E)$, normalized to be a probability
measure, the restrictions of the maps $f_\omega$ to the compact set
$\operatorname{supp}\mu$ form a random $\alpha$-H\"{o}lder map from $\operatorname{supp}\mu$ to $\mathbb{R}^m$.

Fix $\omega$ in the almost sure event of (H1). For every
$\mu\in\mathcal{M}_c^+(E)$, the map $f_\omega$ is $\beta$-H\"{o}lder
continuous on $\operatorname{supp}\mu$ for every $\beta\in(0,\alpha)$, so that
Proposition~\ref{prop:holder_upper} gives
$\dim_P\mu_{f_\omega}\leq\dim_P^{\alpha m}\mu/\alpha\leq\dim_P^{\alpha m}E/\alpha$,
and Lemma~\ref{lem:xiao_sets} yields
$\dim_P f_\omega(E)\leq\dim_P^{\alpha m}E/\alpha$.

For the reverse inequality, choose $\mu_j\in\mathcal{M}_c^+(E)$ with
$\dim_P^{\alpha m}\mu_j\to\dim_P^{\alpha m}E$. By
Proposition~\ref{prop:random_lower}, for each $j$, almost surely
$\dim_P(\mu_j)_{f_\omega}\geq\dim_P^{\alpha m}\mu_j/\alpha$. Since
$(\mu_j)_{f_\omega}$ is supported on the compact set
$f_\omega(\operatorname{supp}\mu_j)\subset f_\omega(E)$, we have
$\dim_P f_\omega(E)\geq\dim_P(\mu_j)_{f_\omega}$, and intersecting the
countably many almost sure events yields
$\dim_P f_\omega(E)\geq\dim_P^{\alpha m}E/\alpha$ almost surely. This
proves \eqref{eq:sets_xiao}.

Finally, $\dim_P f_\omega(E)=m$ almost surely if and only if
$\dim_P^{\alpha m}E=\alpha m$. If $\alpha m\leq n$, by
Theorem~\ref{thm:sets_profile} this holds if and only if
$\alpha m\leq\lim_{\theta\to0}\dim_{P,\theta}E$; if $\alpha m>n$, neither
condition holds, since $\dim_P^{\alpha m}E\leq\dim_P E\leq n$ by
Remark~\ref{rem:profiles}(c) and \eqref{eq:dimP_sets}, and
$\lim_{\theta\to0}\dim_{P,\theta}E\leq\dim_P E\leq n$. In the remaining
case, $\dim_P^{\alpha m}E<\alpha m$ and $\dim_P f_\omega(E)<m$ almost
surely.
