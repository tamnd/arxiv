\label{thm:local-linearization}
Assume Condition~\ref{cond:A} holds with $q_0>q_H$. Fix
$0<T_0<T<\infty$, $q_H<q\le q_0$, and $1\le p\le q/2$.  
\begin{itemize}
\item[(i)] Fix $0<\delta<\delta_t(q)$. There exist $C<\infty$ and
$\varepsilon_0>0$ such that
\begin{equation}\label{t:eq:main-moment}
 \sup_{T_0\le t\le T}\sup_{x\in\mathbb R}
 \lambda(x)^{1/q}\left\|R^{\mathrm t}(t,x,\varepsilon)\right\|_{p}
 \le C|\varepsilon|^{H/2+\delta},\qquad 0<|\varepsilon|\le \varepsilon_0.
\end{equation} \item[(ii)]
 
Fix $0<\delta<\delta_x^*$. There exist $C<\infty$ and
$\varepsilon_0>0$ such that
\begin{equation}\label{eq:main}
 \sup_{T_0\le t\le T}\sup_{x\in\mathbb R}
 \lambda(x)^{1/q}\left\|R^{\mathrm x}(t,x,\varepsilon)\right\|_{p}
 \le C|\varepsilon|^{H+\delta},\qquad 0<|\varepsilon|\le \varepsilon_0.
\end{equation}

\end{itemize}

<!-- proposition-1: body_md -->
\label{thm:oscillations}
Assume that Condition~\ref{cond:A} holds with $q_0>q_H$.
For each fixed $(t,x)\in(0,\infty)\times\mathbb R$, the following
Khinchin and Chung LILs hold almost surely.
\begin{itemize}
\item[(i)] If $\delta_{x,\mathrm{pw}}>0$, then
\begin{align}
 \limsup_{h\downarrow0}
 \frac{u(t,x+h)-u(t,x)}
      {h^H\sqrt{2\log\log(1/h)}}
 =&\, \frac{|\sigma(u(t,x))|}{\sqrt2},
 \label{eq:spatial-lil}\\
 \liminf_{r\downarrow0}\sup_{0\le h\le r}
 \frac{ |u(t,x+h)-u(t,x)|}
      {(r/\log\log(1/r))^H}
 =&\, \frac{b_H^H}{\sqrt2}|\sigma(u(t,x))|.

<!-- lemma-1: body_md -->
\label{lem:initial-smoothing}
\textup{(i)} For the weight function $\lambda$
defined by \eqref{eq weight}, we have
\begin{equation}\label{eq:weight-ratio}
\frac{\lambda(x)}{\lambda(y)}
\le C_H(1+|x-y|)^{2(1-H)},
\qquad x,y\in\mathbb R.
\end{equation}
\textup{(ii)} Let $0<T_0<T<\infty$,
$2\le q<\infty$, $1\le p\le q$, and $j=0,1,2$.
If $f\in L^q_\lambda(\Omega\times\mathbb R)$, then
\begin{equation}\label{eq:msmooth}
\sup_{T_0\le t\le T}\sup_{x\in\mathbb R}
\lambda(x)^{1/q}
\left\|\partial_x^j(G_t*f)(x)\right\|_p
\le
C_{T_0,T,q,H,j}
\left\|f\right\|_{L^q_\lambda(\Omega\times\mathbb R)}.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Part (i) follows from the proof of
\cite[Lemma~2.5]{HW2022}.

For (ii), we first consider deterministic $f$ and put
$q'=q/(q-1)$. 
By H\"older's inequality and (i), we have
$$
\begin{aligned}
&\int_\mathbb R
|\partial_x^jG_t(x-y)|\,|f(y)|\mathrm d y\\
\le&\,
\left\|f\right\|_{L^q_\lambda}
\left(
\int_\mathbb R
|\partial_x^jG_t(z)|^{q'}
\lambda(x-z)^{-q'/q}\mathrm d z
\right)^{1/q'}\\
\le&\,
C\lambda(x)^{-1/q}\left\|f\right\|_{L^q_\lambda}
\left(
\int_\mathbb R
|\partial_x^jG_t(z)|^{q'}
(1+|z|)^{2(1-H)q'/q}\mathrm d z
\right)^{1/q'}\\
\le&\,
C_{T_0,T,q,H,j}
\lambda(x)^{-1/q}\left\|f\right\|_{L^q_\lambda},
\end{aligned}
$$
where the last inequality follows from the uniform Gaussian
decay of the heat-kernel derivatives for $t\in[T_0,T]$.

 For each fixed $x\in\mathbb R$, the Gaussian bounds give
$$
\sup_{\substack{t\in[T_0,T]\\ |a|\le1}}
|\partial_x^mG_t(x+a-y)|
\le C e^{-c|x-y|^2},
\qquad 0\le m\le j.
$$
Since $e^{-c|x-y|^2}|f(y)|$ is integrable, we may differentiate
the convolution by differentiating the heat kernel inside the
integral. The preceding estimate then proves \eqref{eq:msmooth}
for deterministic $f$. 
 

For random $f\in L^q_\lambda(\Omega\times\mathbb R)$, we have
$f(\omega,\cdot)\in L^q_\lambda(\mathbb R)$ almost surely.
Applying the preceding estimate pathwise and using $p\le q$,
we obtain
$$
\begin{aligned}
\left\|\partial_x^j(G_t*f)(x)\right\|_p
&\le
C\lambda(x)^{-1/q}
\left\|\left\|f\right\|_{L^q_\lambda(\mathbb R)}\right\|_p\\
&\le
C\lambda(x)^{-1/q}
\left(\mathbb E\left\|f\right\|_{L^q_\lambda(\mathbb R)}^q\right)^{1/q}\\
&=
C\lambda(x)^{-1/q}
\left\|f\right\|_{L^q_\lambda(\Omega\times\mathbb R)}.
\end{aligned}
$$
This proves \eqref{eq:msmooth}.

Fix $r_0>0$ and $q_H<q\le q_0$, and let $k$ be

<!-- proof-2: body_md -->
Fix $x\in\mathbb R$.
By \eqref{eq:umom} and Lemma~\ref{lem:initial-smoothing}(ii),
$$
\left\|u(r_0,x-z)-(G_{r_0}*u_0)(x-z)\right\|_2\le C\lambda(x-z)^{-1/q}.
$$
Using \eqref{eq:stochastic-isometry}, \eqref{eq solution nonlinear},  
and  
Lemma~\ref{lem:initial-smoothing}(i), we have 
$$
\begin{aligned}
&\int_\mathbb R |k(z)|
\left(
\mathbb E\int_0^{r_0}
\left\|
G_{r_0-s}(x-z-\cdot)\sigma(u(s,\cdot))
\right\|_{\mathcal H}^2\mathrm d s
\right)^{1/2}\mathrm d z\\
 = &\,
\int_\mathbb R |k(z)|\left\|u(r_0,x-z)-(G_{r_0}*u_0)(x-z)\right\|_2\mathrm d z\\
 \le&\,
C\lambda(x)^{-1/q}
\int_\mathbb R |k(z)|(1+|z|)^{2(1-H)/q}\mathrm d z
<\infty,
\end{aligned}
$$
where finiteness follows from the rapid decay of $k$.
Thus the convolution on the left-hand side
is well defined in $L^2(\Omega)$, and stochastic Fubini's theorem
yields the stated identity.

It suffices to prove the estimates for $2\le p\le q$,
since the bounds for lower moments then follow from Jensen's inequality.
Fix $s,t\in[T_0,T]$ and $x,y\in\mathbb R$.
By symmetry, we may assume $s\le t$.
Splitting the stochastic convolution at time $T_0/2$, we write
\begin{equation}\label{eq:initial-sharp}
u(t,x)=(G_t*u_0)(x)+I(t,x)+J(t,x),
\end{equation}
where
$$
\begin{aligned}
I(t,x):=&\,
\int_0^{T_0/2}\int_\mathbb R
G_{t-r}(x-z)\sigma(u(r,z))W(\mathrm d r,\mathrm d z),\\
J(t,x):=&\,
\int_{T_0/2}^t\int_\mathbb R
G_{t-r}(x-z)\sigma(u(r,z))W(\mathrm d r,\mathrm d z).
\end{aligned}
$$
We estimate the increments of these three terms separately.


Since $\partial_tG_t=\partial_x^2G_t$, the fundamental theorem
of calculus, Minkowski's inequality, and
Lemma~\ref{lem:initial-smoothing}(ii) yield
\begin{equation}\label{eq:initial-sharp-expanded}
\begin{aligned}
&\left\|(G_t*u_0)(x)-(G_s*u_0)(y)\right\|_p\\
\le&\,
\int_s^t
\left\|\partial_x^2(G_r*u_0)(x)\right\|_p\mathrm d r
+|x-y|\int_0^1
\left\|\partial_x(G_s*u_0)(y+\theta(x-y))\right\|_p\mathrm d\theta\\
\le&\,
C\lambda(x)^{-1/q}|t-s|
+
C|x-y|\int_0^1
\lambda(y+\theta(x-y))^{-1/q}\mathrm d\theta\\
\le&\,
C\bigl(\lambda(x)^{-1/q}+\lambda(y)^{-1/q}\bigr)
\bigl(|t-s|+|x-y|\bigr).
\end{aligned}
\end{equation}

 

Set
$$
f(z):=u(T_0/2,z)-(G_{T_0/2}*u_0)(z).
$$
By Lemma~\ref{lem:semigroup-fubini},
$$
I(t,x)-I(s,y)
=
(G_{t-T_0/2}*f)(x)-(G_{s-T_0/2}*f)(y).
$$
Moreover, \eqref{eq:msmooth} and \eqref{eq:umom} imply
$$
\left\|f(z)\right\|_p\le C\lambda(z)^{-1/q},
\qquad z\in\mathbb R.
$$
By Minkowski's inequality, the preceding bound on $f$,
\eqref{eq:weight-ratio}, and the Gaussian decay of the
heat-kernel derivatives, we obtain
$$
\begin{aligned}
 \left\|(\partial_aG_a*f)(v)\right\|_p
+\left\|(G_a'*f)(v)\right\|_p 
\le&\,
\int_\mathbb R
\bigl(
|\partial_aG_a(v-z)|+|G_a'(v-z)|
\bigr)\left\|f(z)\right\|_p\,\mathrm d z\\
\le&\, C\lambda(v)^{-1/q},
\end{aligned}
$$
uniformly for $a\in[T_0/2,T]$ and $v\in\mathbb R$.
Here $G_a'$ denotes the spatial derivative of $G_a$.
Applying the fundamental theorem of calculus to the heat kernels,
as in \eqref{eq:initial-sharp-expanded}, yields
\begin{equation}\label{eq:early-stochastic-sharp}
\left\|I(t,x)-I(s,y)\right\|_p
\le
C\bigl(\lambda(x)^{-1/q}+\lambda(y)^{-1/q}\bigr)
\bigl(|t-s|+|x-y|\bigr).
\end{equation}

\le&\,

\bigl[\mathcal N_{1/2-H,p}\sigma(u(r,\cdot))(z)\bigr]^2

\end{aligned}

\eqref{eq:umom} and \eqref{eq:ufrac}, yields
\begin{equation}\label{eq:sigma-bounds}

+
\mathcal N_{1/2-H,p}\sigma(u(r,\cdot))(z)
\le C\lambda(z)^{-1/q},
\end{equation}
uniformly for $r\in[T_0/2,T]$.

where the last inequality uses $0<|h|\le1$ and $2H<1$.

\le&\,
C\lambda(x)^{-2/q}\varepsilon^{H+2\vartheta},
\end{aligned}

$H+2\vartheta<1/2$.

Part (i) follows by the same grid interpolation argument, using the temporal moment and H\"older estimates. It gives a version of \cite[Lemma~5.1]{Qian2026} that holds uniformly over compact time intervals.



Fix $t>0$, compact intervals $J,\widetilde J\subset\mathbb R$ with
$J\subset\operatorname{int}(\widetilde J)$, and
$0<\delta_x<\delta_{x,\mathrm{pw}}$. Set $P:=q_0/2$.
By \eqref{eq:delta-x-pw}, we can choose $m>1$ such that
$$
 \frac{H+\delta_x}{H-1/q_0}
 <m<
 \frac{P(\delta_x^*-\delta_x)+1}{2}.
$$
The strict inequalities allow us to choose
$\delta_x<\delta_1<\delta_2<\delta_x^*$ and
$0<\zeta<H-1/q_0$ such that
\begin{equation}\label{eq:pathwise-parameter-choice}
 m\zeta>H+\delta_x,
 \qquad
 P(\delta_2-\delta_1)+1-2m>0.
\end{equation} 

Since $\lambda$ is bounded away from zero on $\widetilde J$,
Theorem~\ref{thm:local-linearization}(ii), applied with $q=q_0$
and $p=P$, yields
\begin{equation}\label{eq:pathwise-space-moment}
 \|R^{\mathrm x}(t,x,y-x)\|_{P}
 \le C|y-x|^{H+\delta_2}
\end{equation}
for all $x,y\in\widetilde J$ with $|y-x|$ sufficiently small.

  Combining   \eqref{eq:U-increment-Lp} and \eqref{eq:exact-holder} with $p=q=q_0$, and using the compactness of $\widetilde J$, we obtain 
\[
 \mathbb E|u(t,x)-u(t,y)|^{q_0}
 +\mathbb E|U(t,x)-U(t,y)|^{q_0}
 \le C|x-y|^{q_0H},
 \qquad x,y\in \widetilde J.
\]

Since $q_0H>1$, Kolmogorov's continuity theorem
(see, e.g., \cite[p.~10]{DKMNX}) yields continuous versions of
$u(t,\cdot)$ and $U(t,\cdot)$,  whose sample paths are almost surely
$\zeta$-H\"older continuous on $\widetilde J$ for the chosen
$\zeta\in(0,H-1/q_0)$.  Their H\"older continuity and boundedness on $\widetilde J$,
together with the Lipschitz continuity of $\sigma$, imply
\begin{equation}\label{eq:pathwise-space-interpolation}
\begin{aligned}
  |R^{\mathrm x}(t,x,y-x)-R^{\mathrm x}(t,\bar x,\bar y-\bar x)| \le C(\omega)\bigl(|x-\bar x|^\zeta+|y-\bar y|^\zeta\bigr),
 \quad x,y,\bar x,\bar y\in \widetilde J,
\end{aligned}
\end{equation}
where $C(\omega)<\infty$ almost surely.

 Let $r_n=2^{-n}$, $a_n=r_n^m$, and
$$
 \mathcal G_n=(a_n\mathbb Z)^2\cap(\widetilde J\times\widetilde J).
$$
The interval $\widetilde J$ contains $O(a_n^{-1})$ grid points $x$. For each such $x$, the interval $[x-2r_n,x+2r_n]$ contains
$O(r_n/a_n)$ grid points $y$.  Hence,
\begin{equation}\label{eq:pathwise-grid-count}
 \#\{\,(x,y)\in\mathcal G_n:|y-x|\le2r_n\,\}
 \le Cr_na_n^{-2}=Cr_n^{1-2m}.
\end{equation}



 For all sufficiently large $n$,  by Markov's
inequality,   \eqref{eq:pathwise-space-moment},
and \eqref{eq:pathwise-grid-count},  we have 
\begin{align*}
 \mathbb P\left\{
 \max_{\substack{(x,y)\in\mathcal G_n\\|y-x|\le2r_n}}
 |R^{\mathrm x}(t,x,y-x)|>r_n^{H+\delta_1}
 \right\} 
 \le &\,
 r_n^{-P(H+\delta_1)}
 \sum_{\substack{(x,y)\in\mathcal G_n\\|y-x|\le2r_n}}
 \|R^{\mathrm x}(t,x,y-x)\|_{P}^P\\
 \le &\,  Cr_n^{P(\delta_2-\delta_1)+1-2m}.
\end{align*}
By \eqref{eq:pathwise-parameter-choice}, the exponent in the last term 
is positive, so these probabilities are summable. The Borel--Cantelli 
lemma therefore yields, almost surely, for all sufficiently large $n$,
\begin{equation}\label{eq:pathwise-grid-bound}
 \max_{\substack{(x,y)\in\mathcal G_n\\|y-x|\le2r_n}}
 |R^{\mathrm x}(t,x,y-x)|\le r_n^{H+\delta_1}.
\end{equation} 

Since $J\subset\operatorname{int}(\widetilde J)$ and $a_n=o(r_n)$,
for all sufficiently large $n$ and all $x,y\in J$ with
$|y-x|\le r_n$, we can choose $(\bar x,\bar y)\in\mathcal G_n$
such that
$$
 |x-\bar x|+|y-\bar y|\le2a_n,
 \qquad
 |\bar y-\bar x|\le r_n+2a_n\le2r_n.
$$
 
Combining \eqref{eq:pathwise-space-interpolation} and
\eqref{eq:pathwise-grid-bound} yields, almost surely,
\[
 \sup_{\substack{x,y\in J\\|y-x|\le r_n}}
 |R^{\mathrm x}(t,x,y-x)|
 \le C(\omega)r_n^{m\zeta}+r_n^{H+\delta_1}
 =o(r_n^{H+\delta_x}).
\]
Since the supremum is nondecreasing in $r$ and $r_n/r_{n+1}=2$,
the estimate extends to arbitrary $r\downarrow0$, proving
\eqref{eq:spatial-pathwise} for the  chosen $J$ and $\delta_x$. 


Taking the intersection over all $k\ge1$ and
$\gamma\in\mathbb Q\cap(0,\delta_{x,\mathrm{pw}})$,
we obtain a single event of probability one on which
\eqref{eq:spatial-pathwise} holds with $J=J_k:=[-k,k]$
and $\delta_x=\gamma$ for every such pair $(k,\gamma)$.
Now fix any compact interval $J\subset\mathbb R$ and
$0<\delta_x<\delta_{x,\mathrm{pw}}$.
Choose $k\ge1$ such that $J\subset J_k$ and
a rational number $\gamma$ satisfying
$\delta_x<\gamma<\delta_{x,\mathrm{pw}}$.
On the same probability-one event, the estimate on $J_k$
with exponent $\gamma$ also holds on $J$.
Since
\[
r^{H+\gamma}=o(r^{H+\delta_x})
\qquad\text{as }r\downarrow0,
\]
the desired assertion follows simultaneously for all
such $J$ and $\delta_x$.
 
  
 The proof is complete.

\label{lem:schwartz}
Let $0<H<1/2$. For every Schwartz function $f$ on $\mathbb R$,
\begin{equation}\label{eq:schwartz}
Q_Hf(z)
:=
\int_\mathbb R |f(z+h)-f(z)|^2|h|^{2H-2}\mathrm d h
\le
C_{f,H}(1+|z|)^{2H-2},
\quad z\in\mathbb R.
\end{equation}

For $|z|\le2$, we apply the mean-value theorem when $|h|\le1$
and use the boundedness of $f$ when $|h|>1$ to obtain 
$$
\begin{aligned}
Q_Hf(z)
\le&\,
\left\|f'\right\|_\infty^2
\int_{|h|\le1}|h|^{2H}\mathrm d h
+
4\left\|f\right\|_\infty^2
\int_{|h|>1}|h|^{2H-2}\mathrm d h\\
\le&\, C_{f,H}.
\end{aligned}
$$

Now let $|z|>2$ and split the integral at $|h|=|z|/2$.

$$
|f(z+h)-f(z)|
\le
C_f(1+|z|)^{-1}(|h|\wedge1),
$$
and hence
$$
\int_{|h|\le|z|/2}
|f(z+h)-f(z)|^2|h|^{2H-2}\mathrm d h
\le
C_{f,H}(1+|z|)^{-2}.
$$
On the complementary region, using
$|a-b|^2\le2|a|^2+2|b|^2$, we obtain 
$$
\begin{aligned}
 \int_{|h|>|z|/2}
|f(z+h)-f(z)|^2|h|^{2H-2}\mathrm d h 
\le&\,
C_H|z|^{2H-2}\left\|f\right\|_{L^2(\mathbb R)}^2
+
C_H|f(z)|^2|z|^{2H-1}\\
\le&\,
C_{f,H}(1+|z|)^{2H-2}.
\end{aligned}
$$
Since
$$
(1+|z|)^{-2}\le(1+|z|)^{2H-2},
$$
the desired estimate follows.

\begin{itemize}

\item[(i)]
\begin{equation}\label{eq:localmoment}

\le

\begin{cases}
1+R^{2\gamma+2H-1},
& 2\gamma+2H-1>0,\\
1+\log R,
& 2\gamma+2H-1=0,\\
1,
& 2\gamma+2H-1<0.

\le

\end{equation}

\end{itemize}

\begin{align}\label{eq F-integ}
     \int_\mathbb R F_H(w)\mathrm d w<\infty.
\end{align}

We first show that
\begin{equation}\label{eq:FH-pointwise}
 F_H(w)\le C|w|^{2H-2},

\end{equation}
For \(0<s\le1\),
\[
 Q_H[D_s(\cdot,1)](w)
 \le 2Q_HG_s(w+1)+2Q_HG_s(w).
\]
By the scaling identity
\[
 Q_HG_s(w)
 =
 s^{H-3/2}Q_HG_1(w/\sqrt s)
\]
and Lemma~\ref{lem:schwartz}, we obtain, for \(|w|>2\),
\[
 Q_H[D_s(\cdot,1)](w)
 \le C|w|^{2H-2}s^{-1/2}.
\]
Consequently,
\[
 \int_0^1 Q_H[D_s(\cdot,1)](w)\mathrm d s
 \le C|w|^{2H-2}.
\]


We also note the scaling relation
\begin{equation}\label{eq:heat-kernel-derivative-scaling}
G_s'(y)=s^{-1}G_1'(y/\sqrt{s}),
\qquad s>0,\ y\in\mathbb R.
\end{equation}
For $s\ge1$, we have
$$
D_s(\cdot,1)=\int_0^1G_s'(\cdot+v)\,\mathrm d v.
$$
Jensen's inequality, \eqref{eq:heat-kernel-derivative-scaling},
and Lemma~\ref{lem:schwartz} therefore yield  
$$
Q_H[D_s(\cdot,1)](w)
\le
Cs^{H-5/2}
\left(1+\frac{|w|}{\sqrt{s}}\right)^{2H-2}.
$$
 
Hence, for \(|w|>2\),
\[
\begin{aligned}
\int_1^\infty Q_H[D_s(\cdot,1)](w)\mathrm d s
\le &\, 
C|w|^{2H-2}
\int_1^{|w|^2/4}s^{-3/2}\mathrm d s
+
C\int_{|w|^2/4}^\infty s^{H-5/2}\mathrm d s\\
\le &\, 
C|w|^{2H-2}.
\end{aligned}
\]
This proves \eqref{eq:FH-pointwise}.

\begin{aligned}
\int_{|w|\le R}|w|^{2\gamma}F_H(w)\mathrm d w

\end{aligned}
\]
The three cases in \eqref{eq:localmoment}  follow by considering whether
\(2\gamma+2H-1\) is positive, zero, or negative.

\le &\, 
CR^{a+2H-1}.
\end{aligned}
\] 

The case $1\le R<2$ follows from \eqref{eq F-integ}
and the estimate above with $R=2$, after enlarging $C$.
This proves \eqref{eq:weightedtail}.

\mathrm d h\mathrm d z\mathrm d r\\

C

\varepsilon^{2H+2\gamma}+\varepsilon,

& \gamma=\frac12-H.

D_{\varepsilon^2s}(\varepsilon w+\varepsilon\ell,\varepsilon)-D_{\varepsilon^2s}(\varepsilon w,\varepsilon)

\varepsilon^{-1}\bigl[D_s(w+\ell,1)-D_s(w,1)\bigr].

|D_r(z+h,\varepsilon)-D_r(z,\varepsilon)|^2

\end{equation}

\le
\int_\mathbb R F_H(w)(1+|w|)^a\mathrm d w.

The integral on the right is finite: on $|w|\le1$ this follows from

For (i), the scaling $r=\varepsilon^2s$ and $z=\varepsilon w$ gives
$$
\begin{aligned}
 \int_0^\infty\int_\mathbb R
|D_r(z,\varepsilon)|^2(1+|z|)^a\mathrm d z\mathrm d r
=&\,
\varepsilon\int_0^\infty\int_\mathbb R
|D_s(w,1)|^2(1+\varepsilon|w|)^a\mathrm d w\mathrm d s\\
\le&\,
\varepsilon\int_0^\infty\int_\mathbb R
|D_s(w,1)|^2(1+|w|)^a\mathrm d w\mathrm d s.
\end{aligned}
$$
It therefore suffices to show that
\begin{equation}\label{eq:single-energy-target}
\int_0^\infty\int_\mathbb R
|D_s(w,1)|^2(1+|w|)^a\mathrm d w\mathrm d s<\infty.
\end{equation}
For $0<s\le1$,
$$
|D_s(w,1)|^2
\le
2\bigl(G_s(w+1)^2+G_s(w)^2\bigr),
$$
and Gaussian decay yields
$$
\int_\mathbb R
|D_s(w,1)|^2(1+|w|)^a\mathrm d w
\le
Cs^{-1/2}.
$$
For $s\ge1$, using Jensen's inequality 
and  \eqref{eq:heat-kernel-derivative-scaling}, we have
$$
\begin{aligned}
\int_\mathbb R|D_s(w,1)|^2(1+|w|)^a\mathrm d w
\le&\,
\int_0^1\int_\mathbb R
|G_s'(w+c)|^2(1+|w|)^a\mathrm d w\mathrm d c\\
\le&\,
Cs^{(a-3)/2}
\int_\mathbb R|G_1'(z)|^2(1+|z|)^a\mathrm d z\\
\le&\,
Cs^{(a-3)/2}.
\end{aligned}
$$
Since $a<1$, both bounds are integrable over their respective ranges,
which proves \eqref{eq:single-energy-target}.

For (ii), using $|z|^\gamma\wedge1\le|z|^\gamma$ and the
fundamental theorem of calculus, we obtain
$$
\begin{aligned}
\int_\mathbb R
|D_\rho(z,\varepsilon)|(1+|z|)^b(|z|^\gamma\wedge1)\mathrm d z
\le&\,
\varepsilon\int_0^1\int_\mathbb R
|G_\rho'(z+c\varepsilon)|(1+|z|)^b|z|^\gamma\mathrm d z\mathrm d c.
\end{aligned}
$$
Since $\varepsilon\le\sqrt{\rho}\le1$ and $0\le c\le1$,
$$
(1+|z|)^b\le C(1+|z+c\varepsilon|)^b,
\qquad
|z|^\gamma\le|z+c\varepsilon|^\gamma+\varepsilon^\gamma.
$$
Using the change of  variables $y=z+c\varepsilon$ and 
\eqref{eq:heat-kernel-derivative-scaling}, we obtain

C\varepsilon\int_\mathbb R

C\varepsilon\left(

\right)\\
\le&\,
C\varepsilon\rho^{-(1-\gamma)/2}.
\end{aligned}
$$ 

The proof is complete.
