<!-- theorem-1: body_md -->
\label{thm-1-1}
There exist absolute constants $C_1,C_2\ge1$ with the following property. For every $\varepsilon\in(0,1)$ and all integers $d,N\ge2$, there exists an explicitly constructed finite set $W_{d,\varepsilon}(N)\subset\mathbb T^d$ such that
\begin{equation}\label{eq:generalcard-1-4}
 |W_{d,\varepsilon}(N)|
 \le \bigl(C_1(1+\varepsilon^{-1})\bigr)^{d-1}N^{1+\varepsilon}
\end{equation}
and
\begin{equation}\label{eq:generalnorm-1-5}
 \|f\|_\infty
 \le \bigl(C_2(1+\varepsilon^{-1})\bigr)^{d-1}
 \|f\|_{W_{d,\varepsilon}(N)},
 \qquad f\in\mathcal T(N,d).
\end{equation}
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:main}
Let $d,N\ge2$, $\varepsilon\in(0,1)$, and $\eta\in(0,1)$. There exist constants $C(d,\varepsilon)>0$ and $B_{d,\varepsilon}>0$ with
$
 B_{d,\varepsilon}\le
 \exp\bigl(C_d\varepsilon^{-d}\bigr)$ such that,  whenever
\begin{equation}\label{eq:m-random-intro}
 m\ge C(d,\varepsilon)\log\frac2\eta\,N^{1+\varepsilon},
\end{equation}
the inequality
\begin{equation}\label{eq:random-norming-intro}
 \|f\|_\infty\le
 B_{d,\varepsilon} \|f\|_{X_m}
\end{equation}
holds simultaneously for all $f\in\mathcal T(N,d)$ with probability at least $1-\eta$.
<!-- end theorem-2 -->
<!-- remark-1: body_md -->
The deterministic and random results have the same exponent $1+\varepsilon$ in $N$, but their norming constants have very different dependence on \(\varepsilon\).
<!-- end remark-1 -->
<!-- lemma-1: body_md -->
\label{lem:combination}
Let $P_j:C(\mathbb T)\to C(\mathbb T)$, $j=0,1,\ldots$, be a sequence of linear operators, and set
\[
 D_0:=P_0,
 \qquad
 D_j:=P_j-P_{j-1},\quad j\ge1.
\]
Then, for every integer $n\ge d-1$,
\begin{equation}\label{eq:abstractcombination}
 \sum_{|\nu|_1\le n}
 D_{\nu_1}^{(1)}\cdots D_{\nu_d}^{(d)}
 =\sum_{j=0}^{d-1}(-1)^j\binom{d-1}{j}
 \sum_{|\nu|_1=n-j}
 P_{\nu_1}^{(1)}\cdots P_{\nu_d}^{(d)},
\end{equation}
where all sums are taken over $\nu\in\mathbb Z_+^d$.

Fix $\varepsilon\in(0,1)$ and set

 \qquad
 q:=N^{1/r},
 \qquad
 M_j:=\lceil q^j\rceil,
 \quad 0\le j\le r.
\]
Then $M_0=1$ and $M_r=N$. Define
\[
 T_j:=V_{M_j},\quad 0\le j\le r,
 \qquad
 T_j:=T_r=V_N,\quad j>r,
\]
and put
\[
 \Delta_0:=T_0,
 \qquad
 \Delta_j:=T_j-T_{j-1},\quad j\ge1.
\]
Since $T_r$ reproduces every trigonometric polynomial of degree at most $N$, we have
\begin{equation}\label{decom}
 g=T_rg=\sum_{j=0}^r\Delta_jg
   =\sum_{j=0}^\infty\Delta_jg,
 \qquad g\in\mathcal T([-N,N]).
\end{equation}
Moreover, $\Delta_j=0$ for $j>r$, and the reproduction property of $V_n$ gives
\begin{equation}\label{3-5:eq}
 \Delta_j e^{\mathrm i kt}=0
 \qquad\text{whenever }j\ge 1\text{ and }|k|\le q^{j-1}.
\end{equation}

For $\nu=(\nu_1,\ldots,\nu_d)\in\mathbb Z_+^d$, define
\[
 T_\nu:=T_{\nu_1}^{(1)}\cdots T_{\nu_d}^{(d)},
 \qquad
 \Delta_\nu:=\Delta_{\nu_1}^{(1)}\cdots\Delta_{\nu_d}^{(d)}.
\]
Applying \eqref{decom} in each variable yields
\begin{equation}\label{3-7-0}
 f=\sum_{\nu_1=0}^r\cdots\sum_{\nu_d=0}^r\Delta_\nu f
  =\sum_{\nu\in\mathbb Z_+^d}\Delta_\nu f,
 \qquad f\in\mathcal T(N,d).
\end{equation}

Let
\[

\]
We claim that
\begin{equation}\label{3-7}
 \Delta_\nu f=0\   \ \text{whenever}
 \qquad f\in\mathcal T(N,d)\  \ \text{and}\  \  |\nu|_1>L.
\end{equation}
 Assume that $\Delta_\nu f\neq 0$ for some
$f\in\mathcal T(N,d)$. Then there must exist a vector   $\boldsymbol{k}=(k_1,\ldots,k_d)\in\Gamma(N,d)$
such that
\[
 \Delta_\nu e^{\mathrm i\langle\boldsymbol{k},\boldsymbol{x}\rangle}=\prod_{j=1}^d  \Delta_{\nu_j}[ e^{\mathrm i k_j x_j }] \neq 0.
\]
By \eqref{3-5:eq}, this implies that

$$\max\{|k_j|, 1\}\ge 1 > q^{-1}=q^{v_j-1}.$$  Consequently, in either case, 
\[ \max\{|k_j|,1\}> q^{v_j-1},\   \  j=1,\ldots, d.\]
Multiplying these inequalities over $j=1,\ldots, d$ and using $\boldsymbol{k}\in\Gamma(N,d)$, we obtain

 \le N=q^r.

\]

\begin{equation}\label{eq:generalcombination}
 f=\sum_{j=0}^{d-1}(-1)^j\binom{d-1}{j}

\end{equation}

 &\le 2^{d-1}A^d\binom{r+2d-2}{d-1}

 \qquad a>b\ge1,

\begin{align*}
 2^{d-1}A^d\binom{r+2d-2}{d-1}
 &\le 2^{d-1}A^d

 &\le \bigl(6eA^2(1+\varepsilon^{-1})\bigr)^{d-1}.
\end{align*}

 |W|\ge

 =\frac1{4^d}\prod_{j=1}^d\bigl(e^{\mathrm i x_j}+e^{-\mathrm i x_j}+2\bigr).
\]

\[
 F(\boldsymbol{w}-\boldsymbol{a})\ge B^{-1}.
\]

 E_B:=\{\boldsymbol{x}\in[-\pi,\pi)^d:F(\boldsymbol{x})\ge B^{-1}\}.
\]

\[
 [-\pi,\pi)^d=\bigcup_{\boldsymbol{w}\in W}(\boldsymbol{w}-E_B),
\]

\begin{equation}\label{4-3}
 |W|\mu_d(E_B)\ge1.
\end{equation}

\[
 \cos (t/2)\le e^{-t^2/8},
\]
it follows that
\[
 F(\boldsymbol{x})\le\exp\left(-\frac{|\boldsymbol{x}|^2}{4}\right),
 \qquad \boldsymbol{x}\in[-\pi,\pi)^d.
\]
Hence,
\[
 E_B\subset\{\boldsymbol{x}\in[-\pi,\pi)^d:|\boldsymbol{x}|^2\le4\log B\}.

 \left(\frac{2e\log B}{\pi d}\right)^{d/2}.
\end{equation}
Combining \eqref{4-3} and \eqref{eq:EB-measure} proves \eqref{eq:HC-lower}.

Choose a maximal $\delta$-separated subset $\mathcal N$ of $\mathcal B$ in the uniform norm. Since $V$ has real dimension $n_0$, the standard volumetric estimate gives

 \le\left(1+\frac2\delta\right)^{n_0}
 =(1+4B)^{n_0}
 \le(5B)^{n_0}.
\]
By maximality, $\mathcal N$ is also a $\delta$-net of $\mathcal B$. Furthermore, by Lemma~\ref{thm:remez},

 \qquad g\in\mathcal N.

 X_m\cap S_g(B)\ne\varnothing
 \quad\Longleftrightarrow\quad
 \lVert g\rVert_\infty\le B\lVert g\rVert_{X_m}.
\end{equation}
Let $E$ be the event that $X_m\cap S_g(B)\ne\varnothing$ for every $g\in\mathcal N$.

\[ E^c= \bigcup_{g\in\mathcal N} \Bigl\{ X_m\cap S_g(B)=\varnothing \Bigr\}=\bigcup_{g\in\mathcal N} \Bigl\{ \xi^j\notin S_g(B),\  \ \forall 1\leq j\leq m\Bigr\}.\]

 \bigl(1-\mu_d(S_g(B))\bigr)^m\le(5B)^{n_0}\left(1-\frac1{m_0}\right)^m\\
 &\le\exp\left(n_0\log(5B)-\frac{m}{m_0}\right)

\label{lem:leverage}
Let $K_V$ be the reproducing kernel of $V$ in $L^2(\Omega,\mu)$. Then
\begin{equation}\label{6-10a}
 K_V(x,x)\le Kr,
 \qquad x\in\Omega,
 \qquad\text{and}\qquad
 \dim V\le Kr.
\end{equation}
In particular,
\begin{equation}\label{8-4-Nik}
 \lVert F\rVert_\infty\le\sqrt{Kr}\,\lVert F\rVert_2,
 \qquad F\in V.
\end{equation}
<!-- end lemma-4 -->
<!-- proof-8: body_md -->
We first claim that
\begin{equation}\label{6-11a}
 \langle P_VF,F\rangle
 \le\sum_{j=1}^K\langle P_jF,F\rangle,
 \qquad F\in L^2(\Omega,\mu).
\end{equation}
Indeed, by \eqref{6-4} and \eqref{6-9},
\begin{align*}
 \langle P_VF,F\rangle
 &=\sum_{\omega\ne0}\lVert \Pi_\omega F\rVert_2^2\\
 &\le\sum_{j=1}^K\sum_{\omega:\,\omega_j=1}
 \lVert \Pi_\omega F\rVert_2^2
 =\sum_{j=1}^K\langle P_jF,F\rangle.
\end{align*}

The reproducing kernel of $H_j$ is
\[
 K_{H_j}(x,y)
 =r_j\sum_{A\in\mathcal P_j}\mathbf1_A(x)\mathbf1_A(y),
\]
and therefore
\begin{equation}\label{6-12-a}
 K_{H_j}(x,x)=r_j,
 \qquad x\in\Omega.
\end{equation}
For $u\in\Omega$, define $e_u:=|\Omega|\mathbf1_{\{u\}}$. Then
\[
 \langle F,e_u\rangle=F(u),
 \qquad F\in L^2(\Omega,\mu),
\]
and
\[
 K_V(u,u)=\langle P_Ve_u,e_u\rangle.
\]
Using \eqref{6-11a} and \eqref{6-12-a}, we obtain
\[
 K_V(u,u)
 \le\sum_{j=1}^K\langle P_je_u,e_u\rangle
 =\sum_{j=1}^Kr_j
 \le Kr.
\]
This proves the first assertion in \eqref{6-10a}.

Let $\varphi_1,\ldots,\varphi_n$ be an orthonormal basis of $V$, where $n=\dim V$. Since
\[
 K_V(x,y)=\sum_{s=1}^n\varphi_s(x)\overline{\varphi_s(y)},
\]
we have
\[
 n=\int_\Omega K_V(x,x)\,d\mu(x)\le Kr.
\]
Finally, the reproducing property and the Cauchy--Schwarz inequality give
\begin{align*}
 |F(x)|
 &\le\lVert F\rVert_2
 \left(\int_\Omega|K_V(x,y)|^2\,d\mu(y)\right)^{1/2}\\
 &=\lVert F\rVert_2\sqrt{K_V(x,x)}
 \le\sqrt{Kr}\,\lVert F\rVert_2.
\end{align*}
Taking the maximum over $x\in\Omega$ proves \eqref{8-4-Nik}.

<!-- lemma-7: body_md -->
\label{lem:emp}
Assume that the frame event \eqref{eq:frame} occurs with $0<\delta\le1/2$. Then the following statements hold.
\begin{enumerate}[\rm (i)]
 \item The form $\langle\cdot,\cdot\rangle_m$ is an inner product on $V$, and $\widehat P_j|_V$ is the orthogonal projection of $(V,\langle\cdot,\cdot\rangle_m)$ onto $H_j$.
 \item For every $F\in V$,
 \[
  \lVert \widehat P_jF\rVert_\infty\le\lVert F\rVert_{\Omega_m},
  \qquad
  \lVert \widehat SF\rVert_\infty\le\lVert F\rVert_{\Omega_m}.
 \]
 Consequently,
 \[
  \lVert \widehat P_j\rVert_{\infty\to\infty}\le1,
  \qquad
  \lVert \widehat S\rVert_{\infty\to\infty}\le1.
 \]
 \item For every $F\in V$,
 \begin{equation}\label{6-17c}
  \lVert (\widehat P_j-P_j)F\rVert_2\le2\delta\lVert F\rVert_2.
 \end{equation}
 Consequently,
 \[
  \lVert \widehat S-S\rVert_{2\to2}\le2\delta,
  \qquad
  \lVert \widehat S\rVert_{2\to2}\le2.
 \]
\end{enumerate}
<!-- end lemma-7 -->
<!-- proof-10: body_md -->
The lower estimate in \eqref{eq:frame} shows that $\langle\cdot,\cdot\rangle_m$ is positive definite on $V$. Hence, $\langle\cdot,\cdot\rangle_m$  is an inner product on $V$.  For each $A\in {\mathcal P}_j$, let $\phi_A=\sqrt{\frac m {N_A}}\1_A$. A straightforward calculation shows that $\{ \phi_A:\  A\in{\mathcal P}_j\}$
is an orthonormal basis of $(H_j, \langle\cdot,\cdot\rangle_m)$, and
\[ \widehat P_jF= \sum_{A\in{\mathcal P}_j} \langle F,\phi_A\rangle_m\phi_A\in H_j.\]
Hence, $\widehat P_j|_V$ is the orthogonal projection of $(V,\langle\cdot,\cdot\rangle_m)$ onto $H_j$.
This proves (i).

If $x\in A\in\mathcal P_j$, then
\[
 |\widehat P_jF(x)|
 =\left|\frac1{N_A}\sum_{i:\,u_i\in A}F(u_i)\right|
 \le\lVert F\rVert_{\Omega_m}.
\]
Taking the maximum over $x$ proves the first estimate in (ii), and averaging over $j$ gives the estimate for $\widehat S$. Since $\lVert F\rVert_{\Omega_m}\le\lVert F\rVert_\infty$, the operator-norm estimates follow.

It remains to prove (iii). Put
\[
 G:=(\widehat P_j-P_j)F\in H_j.
\]
The lower frame estimate gives
\begin{equation}\label{6-18b}
 (1-\delta)\lVert G\rVert_2^2
 \le\lVert G\rVert_{L^2(\Omega_m)}^2.
\end{equation}
Write
\[ G=(\widehat P_j F-F) +(F-P_jF).\]
Since $F-\widehat P_jF$ is orthogonal to $H_j$ with respect to $\langle\cdot,\cdot\rangle_m$, while $F-P_jF$ is orthogonal to $H_j$ in $L^2(\Omega,\mu)$, we have
\begin{align}
 \lVert G\rVert_{L^2(\Omega_m)}^2
 &=\langle F-P_jF,G\rangle_m\notag\\
 &=\langle F-P_jF,G\rangle_m
   -\langle F-P_jF,G\rangle.
 \label{6-18}
\end{align}
Using \eqref{6-16a} and the fact that $P_j$ is an orthogonal projection, we obtain
\[
 \lVert G\rVert_{L^2(\Omega_m)}^2
 \le\delta\lVert F-P_jF\rVert_2\lVert G\rVert_2
 \le\delta\lVert F\rVert_2\lVert G\rVert_2.
\]
Together with \eqref{6-18b}, this gives
\[
 \lVert G\rVert_2=\lVert (\widehat P_j-P_j)F\rVert
 \le\frac{\delta}{1-\delta}\lVert F\rVert_2
 \le2\delta\lVert F\rVert_2,
\]
which proves \eqref{6-17c}. Averaging over $j$ yields
\[
 \lVert \widehat S-S\rVert_{2\to2}\le2\delta.
\]
Finally, since $\lVert S\rVert_{2\to2}\le1$, we have
\[
 \lVert \widehat S\rVert_{2\to2}
 \le\lVert S\rVert_{2\to2}+\lVert \widehat S-S\rVert_{2\to2}
 \le1+2\delta\le2.
\]

\label{lem:poly}

as an operator on $V$. Then

 \lVert \Lambda\rVert_{2\to2}\le C_K\delta,
 \qquad

 \qquad
 \lVert Q_K(\widehat S)\rVert_{\infty\to\infty}\le M_K.
\]

Set $A:=\widehat S$ and $B:=S$, regarded as operators on $V$. By Lemma~\ref{lem:emp},
\[
 \lVert A\rVert_{2\to2}\le2,
 \qquad
 \lVert B\rVert_{2\to2}\le1,
 \qquad
 \lVert A-B\rVert_{2\to2}\le2\delta.
\]
By Lemma~\ref{lem:annih}(iii), $h_K(B)=0$ on $V$. Therefore,
\begin{equation}\label{6-21a}
 \Lambda=h_K(A)-h_K(B)
 =\sum_{j=1}^Kb_j(A^j-B^j).
\end{equation}
For $1\le j\le K$, the identity
\[
 A^j-B^j
 =\sum_{i=0}^{j-1}A^{j-i-1}(A-B)B^i

gives
\[
 \lVert A^j-B^j\rVert_{2\to2}
 \le2\delta\sum_{i=0}^{j-1}2^{j-i-1}
 \le2^{j+1}\delta.
\]
Substituting this estimate into \eqref{6-21a} and using Lemma~\ref{lem:annih}(ii), we obtain
\[
 \lVert \Lambda\rVert_{2\to2}
 \le\delta\sum_{j=1}^K2^{j+1}|b_j|
 \le2^{K+1}M_K\delta
 =C_K\delta.

 \le M_K,

and similarly,

\label{lem:amplify}
Let $B>1$ and $0<\eta<\eta_0<1$. Then
\begin{equation}\label{8-5}
 m_*(N,d,B,\eta)
 \le\left\lceil\frac{\log(1/\eta)}{\log(1/\eta_0)}\right\rceil
 m_*(N,d,B,\eta_0).
\end{equation}
<!-- end lemma-16 -->
<!-- proof-18: body_md -->
Put
\[
 m_0:=m_*(N,d,B,\eta_0),
 \qquad
 k:=\left\lceil\frac{\log(1/\eta)}{\log(1/\eta_0)}\right\rceil.
\]
Let $\xi^1, \xi^2, \ldots$ be iid random points drawn uniformly from $\mathbb T^d$.
Divide the first $km_0$ sample points into the independent blocks
\[
 X^{(j)}:=\{\xi^{(j-1)m_0+1},\ldots,\xi^{jm_0}\},
 \qquad 1\le j\le k.
\]
For each $j$, let $E_j$ be the event that $X^{(j)}$ is not a $B$-norming set for $\mathcal T(N,d)$. Then $\mathbf{P}(E_j)\le\eta_0$, and the events $E_1,\ldots,E_k$ are independent. Hence
\[
 \mathbf{P}(E_1\cap\cdots\cap E_k)\le\eta_0^k\le\eta.
\]
Let $X=\bigcup_{j=1}^k X^{(j)}$. If at least one of the blocks $X^{(j)}$ is a $B$-norming set, then their union  $X$ is also a $B$-norming set.  It follows that
 \begin{align*}{\mathbf P}& \Bigl\{  \|f\|_\infty\leq B\|f\|_{X}, \   \ \forall f\in\mathcal T(N,d)\Bigr\}\\
 &\ge {\mathbf P} \Bigl( \bigcup_{j=1}^k E_j^c\Bigr)={\mathbf P} \Bigl[\big(\bigcap_{j=1}^k E_j\big)^c\Bigr]=1-\mathbf{P}(E_1\cap\cdots\cap E_k)\\
  &\ge 1-\eta_0^k\ge 1-\eta.
 \end{align*}
 This means that  $X$ is a $B$-norming set for $\mathcal T(N, d)$ with probability at least $1-\eta$, and hence proves \eqref{8-5}.
