\label{defweak}
	We define the bilinear form $B\,:\, V\times V\to \mathbb{R}$ by
	$$B(\phi,v)=-\omega \phi(0) \overline{v(0)}+\int_\Omega \partial_x \phi\, \overline{\partial_xv}\,dx ,\qquad \phi,v\in V.$$
	We say that $\phi\in V$ is a weak solution to the problem $(\mathcal{S}_{\omega,\lambda})$ given $\lambda\in\mathbb{R}$ if and only if the following integral identity holds:
	$$ B(\phi,v)=\lambda( \phi, v)_{L^2_c},\qquad v\in V. $$

<!-- theorem--1: body_md -->
\label{spectralth}
	\textit{The spectrum $\sigma(A)$ is countable (discrete), compact, real, positive. Moreover, $0$ belongs to $\sigma(A)$ and the point spectrum $\sigma_p(A)=\sigma(A)\setminus\{0\}$ consists only of eigenvalues $\{\lambda_n\}_{n\in\mathbb{N}}$ of $A$ satisfying $\lambda_1\geqslant \lambda_2 \geqslant \lambda_3\geqslant \dots > 0$ and $\lambda_n\to 0$ as $n\to +\infty$. There is an orthonormal basis for $L^2_c$ of corresponding eigenfunctions $\{\phi_n\}_{n\in\mathbb{N}}\subset V$ such that
		$$A = \sum\limits_{n=1}^{\infty} \lambda_n (\cdot ,\phi_n )_{L^2_c}\, \phi_n .$$
		The series on the right hand side converges in the operator norm of $\mathscr{B}(L^2_c)$.}

<!-- definition-2: body_md -->
The spectrum of the operator $P$ is defined by
	$$\sigma(P) = \mathbb{C}\setminus \{\lambda\in\mathbb{C}\mid P-\lambda I\,:\, V\to V^*\text{ is a topological isomorphism}\}.$$
	A spectrum element $\lambda\in \sigma(P)$ is called an \textit{eigenvalue of $P$} if the forward problem $(\mathcal{S}_{\omega,\lambda})$ has a nontrivial weak solution. In this case, we say that this solution is an eigenfunction corresponding to $\lambda$. The set of all eigenvalues of $P$ is called point spectrum of $P$ and denoted by $\sigma_{p}(P)$.

<!-- proposition-4: body_md -->
\label{eigveigf}
	\textit{The point spectrum of $P$ consists of reciprocals of eigenvalues of $A$:
		$$\sigma_p(P)=\{\lambda >0 : \lambda^{-1}\in\sigma_p(A)\}.$$
		Moreover, for any $\lambda\in\sigma_p(P)$, the set of all weak solutions to $(\mathcal{S}_{\omega,\lambda})$ is exactly the eigenspace $\mathrm{Ker}(A-\lambda^{-1}I)\subset V$. }
<!-- end proposition-4 -->
<!-- proof-1: body_md -->
If $\lambda = 0$, then $P-\lambda I = P$ is a topological isomorphism, so $0\not\in\sigma(P)$.
	
	Suppose $\lambda\ne 0$. Let $\phi\in V\setminus\{0\}$ be a nontrivial weak solution to $(\mathcal{S}_{\omega,\lambda})$, that is, $B(\phi,v) =\lambda ( \phi,v )_{L^2_c}$ for any $v\in V$. Then, we get
	$$ B(\phi,v) =\lambda ( \phi,v )_{L^2_c} = \lambda B(A[\phi],v) ,\qquad v\in V.$$
	It entails that $B(\phi-\lambda A[\phi],v) = 0$ for any $v\in V$. Fix $v=\phi-\lambda A[\phi]$, then the coercivity estimate implies $0=B(v,v)\geqslant C\|v\|_{H^1(\Omega)}^2$, which, in turn, gives   
	\begin{equation*}\label{aphi_eig}
		A[\phi]=\frac{1}{\lambda}\phi.
	\end{equation*}
	
	For other direction, pick an eigenfunction $\phi\in V\setminus\{0\}$ of $A$ corresponding to an eigenvalue $\lambda\ne 0$ and repeat the steps above backwards. Thus, a function $\phi\in V\setminus\{0\}$ is a weak solution to $(\mathcal{S}_{\omega,\lambda})$ if and only if it is an eigenfunction of $A$ corresponding to the eigenvalue $\lambda^{-1}$.
	
	Now, pick $\lambda\in\sigma_p(P)$, so $\lambda^{-1}\in\sigma_p(A)$. Then, clearly, $\mathrm{Ker}(A-\lambda^{-1}I)\ne\{0\}$. This implies that any $\phi\in\mathrm{Ker}(A-\lambda^{-1}I)$ is a weak solution to $(\mathcal{S}_{\omega,\lambda})$.

<!-- lemma-1: body_md -->
\label{pndformula} \textit{
	Let $0\leqslant\omega <1$. The spectral projector $\mathbb{P}_{\omega}^{\, n}$ admits the Bochner integral formula:
	\begin{equation}\label{pnd1}
		\mathbb{P}_{\omega}^{\, n}[f]=\frac{1}{2\pi i}\,\int_{\Gamma_n}(P_\omega-z)^{-1}[f]\,dz,\qquad f\in L^2_c,
	\end{equation}
	where $\Gamma_n\subset\mathbb{C}$ is a contour oriented clockwise of sufficiently small diameter enclosing only $n$-th eigenvalue $\lambda_n(\omega)$ such that $\Gamma_n\cap\sigma_p(P_{\omega})=\varnothing$. }
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
To begin with, notice that the function $\lambda\mapsto (\lambda-z)^{-1}$, $z\in\Gamma_n$ is bounded on the spectrum $\sigma(P_\omega)$, since 
	$$\inf\left\{|\lambda-z|:\lambda\in\sigma(P_\omega),\,z\in\Gamma_n\right\}=\mathrm{dist}(\sigma(P_\omega),\Gamma_n)>0,$$
	where the positivity follows from the fact that $\sigma(P_\omega)$ is closed and $\Gamma_n$ is compact. Also, this function is Borel measurable due to continuity in $\lambda\in\mathbb{R}\setminus\Gamma_n$. Therefore, applying the functional calculus (see e.g., Chapter~XII, Section~2, Theorem~6 in~\cite{dunford_schwartz} or Theorem~VIII.5 in~\cite{reedsimon}) to the function $\lambda\mapsto (\lambda-z)^{-1}$ gives the identity 
	\begin{equation}\label{pm1upb}
		\|(P_\omega - z)^{-1}\|_{\mathscr{B}(L^2_c)}=\mathrm{dist}(\sigma(P_\omega),z)^{-1},\qquad z\in\Gamma_n,
	\end{equation}
	and, moreover, for any $f,g\in L^2_c$ and all $z\in\Gamma_n$,
	$$(P_\omega - z)^{-1}=\int\limits_{\sigma(P_\omega)}\frac{\mathbb{P}_\omega(d\lambda)}{\lambda-z},\quad ((P_\omega - z)^{-1}f,g)_{L^2_c} = \int\limits_{\sigma(P_\omega)}\frac{\mu^\omega_{f,g}(d\lambda)}{\lambda-z} .$$
	
	Now, we show that the Bochner integral in (\ref{pnd1}) is well-defined. The function $z\mapsto (P_\omega-z)^{-1}$ is continuous in operator norm of $\mathscr{B}(L^2_c)$ on $\Gamma_n$, which follows from the identity
	$$(P_\omega-z)^{-1}-(P_\omega-z_0)^{-1}=(z-z_0)(P_\omega-z)^{-1}(P_\omega-z_0)^{-1},\qquad z,z_0\in\Gamma_n.$$
	Whence, for any $f\in L^2_c$ the map $z\mapsto (P_\omega-z)^{-1}f$ is Bochner measurable. Furthermore, since the contour $\Gamma_n$ is compact, continuity of the resolvent on~$\Gamma_n$ implies uniform continuity in $z\in\Gamma_n$. In particular, the upper bound (\ref{pm1upb}) gives Bochner integrability of $z\mapsto (P_\omega-z)^{-1}f$ for a fixed $f\in L^2_c$ due to the following upper bound,
	$$\int\limits_{\Gamma_n}\|(P_\omega-z)^{-1}f\|_{L^2_c}\,dz\leqslant \frac{2\pi\|f\|_{L^2_c}}{\mathrm{dist}(\sigma(P_\omega),\Gamma_n)},\qquad f\in L^2_c.$$
	
	We are ready to prove the identity (\ref{pnd1}). Fix arbitrary $f,g\in L^2_c$. Since the linear functional $(\cdot,g)_{L^2_c}$ is bounded, it commutes with Bochner integrals, and, therefore,
	$$\left(\,\int_{\Gamma_n}(P_\omega-z)^{-1}f\,dz,g\right)_{L^2_c} = \int\limits_{\Gamma_n}((P_\omega-z)^{-1}f,g)_{L^2_c}\,dz = \int\limits_{\Gamma_n}\int\limits_{\sigma(P_\omega)}\frac{\mu^\omega_{f,g}(d\lambda)}{\lambda-z}\,dz.$$
	The modulus of the latter term is finite due to boundedness of the integrand on $\Gamma_n\times\sigma(P_\omega)$ and finiteness of $|\mu^\omega_{f,g}(\mathbb{R})|=|(f,g)_{L^2_c}|<+\infty$. Hence, by the Fubini-Tonelli theorem one can exchange the order of integration. Cauchy's integral formula entails that $\int_{\Gamma_n}(\lambda-z)^{-1}\,dz$ equals $2\pi i$ whenever $\lambda$ is in the interior of the contour $\Gamma_n$ and $0$ otherwise. And again, since $\sigma(P_\omega)$ is discrete and only $\lambda_n(\omega)$ is inside $\Gamma_n$ by the definition of the contour, we obtain
	$$\frac{1}{2\pi i}\left(\,\int_{\Gamma_n}(P_\omega-z)^{-1}f\,dz,g\right)_{L^2_c}=\mu^\omega_{f,g}(\{\lambda_n(\omega)\})=(\mathbb{P}_{\omega}^{\, n}[f],g)_{L^2_c}.$$
	This concludes the lemma.

<!-- proposition-5: body_md -->
\label{lem1}
	\textit{For $n\in \mathbb{N}$, the spectral projector $\mathbb{P}_{\omega}^{\, n}$ is analytic in $\omega$ on the interval $[0,1)$.}
<!-- end proposition-5 -->
<!-- proof-3: body_md -->
Let the trace operator $\gamma\,:\,V\to V^*$ be defined via $\langle\gamma[\phi],v\rangle=\phi(0)v(0)$, so $\gamma$ is bounded. Due to Definition~\ref{deffr} the operators $P_\omega$ can be understood as operators $V\to V^*$ via $\langle P_\omega[\phi],v\rangle = B_\omega(\phi,v)$, $\phi,v\in V$. We notice that $P_\omega=P-\omega\gamma$ holds in $\mathscr{B}(V,V^*)$ and for any $z\in\mathbb{C}\setminus\sigma(P_\omega)$ the resolvent operator $(P_\omega-z)^{-1}$ belongs to $\mathscr{B}(V^*,V)$. 
	
	In the view of Chapter~VII, Section~4.2, Remark~4.6, in~\cite{kato}, we show that the resolvent $(P_\omega-z)^{-1}\,:\, V^*\to V$ is analytic in $\omega$ near any fixed $\omega_0\in [0,1)$, for any fixed $z\in\mathbb{C}\setminus\sigma(P_{\omega_0})$. Indeed, writing $P_\omega=P_{\omega_0}-(\omega-\omega_0)\gamma$,
	$$(P_\omega-z)^{-1}=\bigl(I-(\omega-\omega_0)(P_{\omega_0}-z)^{-1}\gamma\bigr)^{-1}(P_{\omega_0}-z)^{-1},$$
	and the Neumann series
	$$\bigl(I-(\omega-\omega_0)(P_{\omega_0}-z)^{-1}\gamma\bigr)^{-1}=I+\sum_{k\in\mathbb{N}}(\omega-\omega_0)^k\bigl((P_{\omega_0}-z)^{-1}\gamma\bigr)^k$$
	converges in $\mathscr{B}(V)$ for $|\omega-\omega_0|<\|(P_{\omega_0}-z)^{-1}\gamma\|^{-1}_{\mathscr{B}(V)}$. Whence, $(P_\omega-z)^{-1}$ inherits analyticity in a neighborhood of $\omega_0$ in $\mathbb{C}$ as a power series convergent in operator norm~$\mathscr{B}(V^*,V)$. Since $[0,1)$ is connected, on overlapping neighborhoods the power series agree by uniqueness of analytic continuation, see the Identity Principle in Chapter~II, Section~5, Proposition~5.7, in~\cite{mujica}. 
	
	Consider the composition $\imath (P_\omega-z)^{-1}\jmath$, where $\imath\,:\,V\hookrightarrow L^2_c$ and $\jmath\,:\,L^2_c\hookrightarrow V^*$ are continuous embeddings from the Gelfand triple (\ref{gelfand}). This operator coincides with the resolvent $(P_\omega-z)^{-1}\in\mathscr{B}(L^2_c)$, since both solve the same variational problem and the solution is unique by the Lax-Milgram theorem. Composing holomorphic operators with bounded operators preserves holomorphy, see Chapter II, Section~5, Exercise 5A in~\cite{mujica}, and it yields that for $z\in\mathbb{C}\setminus\bigcup_{\omega\in[0,1)}\sigma(P_\omega)$ the resolvent $(P_\omega-z)^{-1}$ depends analytically on $\omega$ on the whole interval $[0,1)$.
	
	Analyticity of $\mathbb{P}_{\omega}^{\, n}$ in $\omega$ is a consequence of Lemma~\ref{pndformula} and analyticity of the resolvent $(P_\omega-z)^{-1}$ under the Bochner integral sign. Indeed, fix $\omega_0\in [0,1)$ and choose a contour $\Gamma_n\subset \mathbb{C}\setminus\sigma(P_{\omega_0})$ enclosing $\lambda_n(\omega_0)\in\sigma(P_{\omega_0})$. Analyticity of the resolvent implies continuity of $\omega\mapsto\|(P_\omega-z)^{-1}\|_{\mathscr{B}(L^2_c)}$ on $\Gamma_n$. Hence, the radius of convergence of the Neumann series above, taken at $\omega_0$, is bounded from below by $\delta_n=\inf\{\|(P_{\omega_0}-z)^{-1}\gamma\|^{-1}_{\mathscr{B}(V)}:z\in\Gamma_n\}>0$, so there is an interval $I\subset[0,1)$ containing $\omega_0$, the same for all $z\in\Gamma_n$, on which $\Gamma_n\subset\mathbb{C}\setminus\sigma(P_\omega)$ and the resolvent $(P_\omega-z)^{-1}$ is analytic in $\omega$ for every $z\in\Gamma_n$. Shrinking $I$ if necessary, the relation (\ref{pm1upb}) and the compactness of $\Gamma_n$ give
	$$M=\sup\left\{\|(P_\omega-z)^{-1}\|_{\mathscr{B}(L^2_c)}:z
	\in\Gamma_n,\omega\in I\right\}< +\infty .$$
	
	Fix $f,g\in L^2_c$ and $z\in\Gamma_n$. Then, the function $\omega\mapsto ((P_{\omega}-z)^{-1}f,g)_{L^2_c}$ is analytic, since it is the evaluation of the analytic resolvent against the fixed bounded bilinear form $(\cdot,\cdot)_{L^2_c}$. Also, this function is uniformly bounded,
	$$|((P_{\omega}-z)^{-1}f,g)_{L^2_c}|\leqslant M \|f\|_{L^2_c}\|g\|_{L^2_c},\qquad \omega\in I,\,z\in\Gamma_n.$$
	That is, for each fixed $f$ and $g$, this function is dominated by an integrable function in $z\in\Gamma_n$. Hence, by the classical complex analysis we obtain that for all $f,g\in L^2_c$,
	$$(\mathbb{P}_{\omega}^{\, n}[f],g)_{L^2_c}=\frac{1}{2\pi i}\,\int\limits_{\Gamma_n}((P_\omega-z)^{-1}f,g)_{L^2_c}\,dz$$
	is analytic in $\omega\in I$. Furthermore, the map $\omega\mapsto\mathbb{P}_{\omega}^{\, n}$ is locally bounded in operator norm due to
	$$\|\mathbb{P}_{\omega}^{\, n}\|_{\mathscr{B}(L^2_c)}\leqslant \frac{|\Gamma_n|}{2\pi}\sup\left\{\|(P_\omega-z)^{-1}\|_{\mathscr{B}(L^2_c)}:z
	\in\Gamma_n\right\} \lesssim M,\qquad \omega\in I.$$
	A standard characterization of analytic operators yields that the spectral projector $\mathbb{P}_{\omega}^{\, n}$ is analytic in $I$. Finally, since the choice of $\omega_0$ is arbitrary, covering $[0,1)$ by overlapping intervals implies analyticity of the spectral projector $\mathbb{P}_{\omega}^{\, n}$ on all of $[0,1)$.

<!-- lemma-2: body_md -->
\label{uniqS}
	\textit{Let $0\leqslant\omega < 1$. For any $\lambda\in \sigma_p(P_\omega)$, $\mathrm{dim}\,E_\lambda(P_\omega)=1$, that is, all eigenvalues of $P_\omega$ are simple.}
<!-- end lemma-2 -->
<!-- proof-4: body_md -->
By the discussion following Definition~\ref{deffr}, $E_\lambda(P_\omega)=E_\lambda(P_F)$ and $\mathrm{Dom}(P_F)\subset H^2(\Omega)$. Let $\phi\in E_\lambda(P_F)$. By the one-dimensional Sobolev embedding $H^2(\Omega)\hookrightarrow C^1(\overline\Omega)$, the values $\phi(1)$ and $\phi'(1)$ are well defined, and $\phi(1)=0$ since $\phi\in V$. Moreover, the eigenvalue equation $P_F[\phi]=\lambda\phi$ is the differential equation of $(\mathcal{S}_{\omega,\lambda})$, a linear second-order equation with continuous coefficients on $\overline\Omega$, so by uniqueness for its Cauchy problem a solution with $\phi(1)=\partial_x\phi(1)=0$ vanishes identically.

Consider the linear map
$$E_\lambda(P_F)\to\mathbb{R},\qquad\phi\mapsto\partial_x\phi(1).$$
If $\partial_x\phi(1)=0$, then $\phi(1)=\partial_x\phi(1)=0$ and so $\phi=0$. Hence the map is injective and $\dim E_\lambda(P_F)\leqslant1$. Since $\lambda$ is an eigenvalue, $\dim E_\lambda(P_\omega)=1$.
<!-- end proof-4 -->
<!-- definition-4: body_md -->
Consider the family of impedance spectral problems $(\mathcal{S}_\omega)$ for $\omega\in [0,1)$. Lemma~\ref{uniqS} allows us to define for each $\omega$ the family of pairs 
	$$\{(\lambda_n(\omega),\phi_{n,\omega})\}_{n\in \mathbb{N}}\subset \mathbb{R}_+\times \widetilde{V},$$ 
	where $\lambda_n(\omega)\in\sigma_p(P_\omega)$ and $\phi_{n,\omega}\in E_{\lambda_n(\omega)}(P_\omega)\cap \widetilde{V}$ are an eigenvalue and the corresponding unique eigenfunction of $P_\omega$. Let us omit the index $0$ in pairs $\{(\lambda_n(0),\phi_{n,0})\}_{n\in\mathbb{N}}$ and write simply $\{(\lambda_n,\phi_n)\}_{n\in \mathbb{N}}$. Note that the elements of the spectrum $\sigma_p(P_\omega)$ are discrete with no accumulation point, meaning that for a fixed $\omega$ all eigenvalues $\lambda_n(\omega)$ are isolated. Moreover, for any $n\in\mathbb{N}$ and any $\omega\in[0,1)$ one has $\mathrm{Ran}(\mathbb{P}_{\omega}^{\, n}) = \mathrm{span}\{\phi_{n,\omega}\}$.

<!-- proposition-6: body_md -->
\label{lem2}
	\textit{For each $n\in \mathbb{N}$, the eigenvalue $\lambda_n(\omega)$ and the eigenfunction $\phi_{n,\omega}$ are analytic in $\omega$ for $0\leqslant\omega <1$.}
<!-- end proposition-6 -->
<!-- proof-5: body_md -->
Fix arbitrary $\omega_0\in [0,1)$. Because $\mathbb{P}^{\,n}_{\omega_0}$ is the nonzero spectral projector onto 1 dimensional eigenspace, we fix a function $v\in\widetilde{V}$ such that $\mathbb{P}^{\,n}_{\omega_0} [v]\ne 0$. By Proposition~\ref{lem1}, the map $p(\omega)=(\mathbb{P}_{\omega}^{\, n} [v],\mathbb{P}_{\omega}^{\, n} [v])_{L^2_c}$ is analytic in $\omega\in [0,1)$, since the function $v$ is independent of $\omega$. Notice that $p(\omega_0) > 0$. By continuity of $p$, there is an interval $I$ containing $\omega_0$ such that $p(\omega) > 0$ for any $\omega\in I$. Therefore, the function
	$$\omega\mapsto\frac{1}{\|\mathbb{P}_{\omega}^{\, n} [v]\|_{L^2_c}} = \frac{1}{\sqrt{p(\omega)}}$$
	is analytic on $I$, since a nonvanishing analytic function has a locally analytic logarithm and an analytic reciprocal. Moreover, Proposition~\ref{lem1} implies that $\mathbb{P}_{\omega}^{\, n}[v]$ is analytic in $\omega\in[0,1)$, since $v$ is independent of $\omega$. We define 
	$$\widetilde{\phi}_{n,\omega}=\frac{\mathbb{P}_{\omega}^{\, n} [v]}{\|\mathbb{P}_{\omega}^{\, n} [v]\|_{L^2_c}},\qquad n\in\mathbb{N},\;\omega\in I.$$
	As the product of two analytic functions is analytic, it follows that $\widetilde{\phi}_{n,\omega}\in\mathrm{Ran}(\mathbb{P}_{\omega}^{\, n})$ is analytic on $I$. Also, for any $\omega\in I$, $\widetilde{\phi}_{n,\omega}\in\mathrm{span}\{\phi_{n,\omega}\}$ and $\|\widetilde{\phi}_{n,\omega}\|_{L^2_c}=1$. 
	
	Now we note that at $\omega=\omega_0$, either $\widetilde{\phi}_{n,\omega_0}=\phi_{n,\omega_0}$ or $\widetilde{\phi}_{n,\omega_0}=-\phi_{n,\omega_0}$. By the obtained above analyticity, $\widetilde{\phi}_{n,\omega}(0)$ has the same sign as $\widetilde{\phi}_{n,\omega_0}(0)$ on a subinterval $\widetilde{I}\subset I$. Without loss of generality, let $\widetilde{\phi}_{n,\omega_0}=\phi_{n,\omega_0}$. Thus, for any $\omega\in\widetilde{I}$, the eigenfunction $\widetilde{\phi}_{n,\omega}$ belongs to $E_{\lambda_n(\omega)}(P_\omega)\cap \widetilde{V}$. Recall that Lemma~\ref{uniqS} implies that for all $\omega\in\widetilde{I}$ there is exactly one eigenfunction of $P_\omega$ corresponding to $\lambda_n(\omega)$ in the energy class $E_{\lambda_n(\omega)}(P_\omega)\cap \widetilde{V}$. This entails  
	$$\widetilde{\phi}_{n,\omega}=\phi_{n,\omega},\qquad \omega\in\widetilde{I},$$
	which gives analyticity of $\omega\mapsto\phi_{n,\omega}$ on the subinterval $\widetilde{I}$. Finally, because the choice of $\omega_0\in [0,1)$ is arbitrary, we repeat this argument for any basepoint $\omega_0$ and get analyticity on overlapping intervals.
	
	Analyticity of the eigenvalues is a straightforward consequence of the following identity,
	$$\lambda_n(\omega)=B_\omega(\phi_{n,\omega},\phi_{n,\omega}),$$
	and the fact that for any fixed $\phi, v\in V$, the function $\omega\mapsto B_\omega(\phi,v)$ depends linearly on $\omega\in [0,1)$.

<!-- proposition-7: body_md -->
\label{gbeig}
	\textit{For each $n\in \mathbb{N}$ the following formula holds:}
	$$\lambda^{\prime}_n(\omega) =-|\phi_{n,\omega}(0)|^2,\qquad \omega\in (0,1).$$
<!-- end proposition-7 -->
<!-- proof-6: body_md -->
Fix arbitrary $n\in \mathbb{N}$ and $\omega\in (0,1)$. The eigenfunction $\phi_{n,\omega}$ satisfies $(\mathcal{S}_{\omega,\lambda_n(\omega)})$ in the weak sense, 
	$$( \partial_x\phi_{n,\omega} , \partial_x v)_{L^2(\Omega)} - \omega \phi_{n,\omega}(0)v(0) = \lambda_n(\omega) (\phi_{n,\omega}, v)_{L^2_c},\quad v\in V.$$
	Note that $\frac{d}{d\omega}\phi_{n,\omega}\in V$, so taking $v=\frac{d}{d\omega}\phi_{n,\omega}$ turns the identity above into
	\begin{equation}\label{eq1212}
		\left( \partial_x\phi_{n,\omega} , \partial_x \frac{d\phi_{n,\omega}}{d\omega}\right)_{L^2(\Omega)}  - \omega \phi_{n,\omega}(0)\frac{d\phi_{n,\omega}}{d\omega}(0) = \lambda_n(\omega) \left(\phi_{n,\omega},\frac{d\phi_{n,\omega}}{d\omega}\right)_{L^2_c}.
	\end{equation}
	Moreover, since $\phi_{n,\omega}\in \widetilde{V}$, one can pick $v=\phi_{n,\omega}$, which gives
	\begin{equation}\label{eq111}
		\| \partial_x\phi_{n,\omega} \|^2_{L^2(\Omega)} - \omega |\phi_{n,\omega}(0)|^2 = \lambda_n(\omega) \|\phi_{n,\omega}\|^2_{L^2_c}.
	\end{equation}
	
	Now, let $b^\omega[v]=B_\omega(v,v)$ be a quadratic form on $V$; this form is bounded, since $B_\omega$ is bounded. Also, the quadratic form $q^\omega=\lambda_n(\omega)\|\cdot\|^2_{L^2_c}$ is bounded. We calculate the needed Gateaux derivatives; for any $h\in V$, $$D_h\left(b^\omega[v]\right) = 2(\partial_xv , \partial_x h)_{L^2(\Omega)}  -  2\omega v(0)h(0),$$ $$D_h\left(q^\omega[v]\right) = 2\lambda_n(\omega)(v,h)_{L^2_c}.$$ 
	Finally, the functionals $\omega\mapsto b^\omega[\phi_{n,\omega}]$ and $\omega \mapsto q^\omega[\phi_{n,\omega}]$ are analytic in $\omega$ as a composition of analytic functions due to Proposition~\ref{lem2}. The identity (\ref{eq111}) turns into $b^\omega[\phi_{n,\omega}]= q^\omega[\phi_{n,\omega}]$. Differentiate it in the Gateaux sense applying the chain rule,	
	$$2\left(\partial_x\phi_{n,\omega} , \partial_x \frac{d\phi_{n,\omega}}{d\omega}\right)_{L^2(\Omega)}  -  |\phi_{n,\omega}(0)|^2 - 2\omega \phi_{n,\omega}(0)\frac{d\phi_{n,\omega}}{d\omega}(0) =$$
	$$=\lambda^{\prime}_n(\omega) \|\phi_{n,\omega}\|^2_{L^2_c} + 2\lambda_n(\omega) \left(\phi_{n,\omega}, \frac{d\phi_{n,\omega}}{d\omega}\right)_{L^2_c} .$$
	Hence, upon combining the latter, identity (\ref{eq1212}) multiplied by $2$, and the fact that $\|\phi_{n,\omega}\|_{L^2_c}=1$, we obtain the required formula.

Recall that eigenfunctions $\{\phi_n\}_{n\in\mathbb{N}}$ form a complete orthonormal basis in $L^2_c$. We claim that the coefficients $a_k^N(t)$ of the Galerkin approximations $u^N$ approximate the Fourier coefficients of $u$. Indeed, $u^N$ tends to $u$ in $C([0,T];L^2_c)$, and, therefore, for a fixed $k\in\{1,\dots,N\}$ the identity 
	$$(u^N(t),\phi_k)_{L^2_c}=\sum_{n=1}^N a_n^N(t)(\phi_n,\phi_k)_{L^2_c}=a_k^N(t),\qquad t\in [0,T],$$
	implies that $a_k^N(t)\to (u(t),\phi_k)_{L^2_c}$ as $N\to +\infty$ for any $t\in [0,T]$. Then, the Fourier series expansion of $u(t)$ for a fixed $t>0$ is given by
	$$u(t)=\sum_{n\in \mathbb{N}} u_n(t)\,\phi_n,\quad \text{with }\; u_n(t)=(u(t),\phi_n)_{L^2_c}. $$
	For any $t\in[0,T]$ the series converges in $L^2_c$. Moreover, the energy estimate of Section~\ref{fpee} applies to the Galerkin approximations $u_N(t)=\sum_{n\leqslant N}u_n(t)\phi_n$ uniformly in $N$, so $\|u_N(t)\|_{H^1(\Omega)}\lesssim\|f\|_{L^2(0,T)}$ for almost every $t\in[0,T]$. Together with the convergence in $L^2_c$ this gives $u_N(t)\rightharpoonup u(t)$ weakly in $H^1(\Omega)$ for almost every $t\in[0,T]$.
	
	Recall that the Dirac delta $\delta_0$ defined by $\langle\delta_0,v\rangle=v(0)$ is bounded in $V$. By~the definition of $\Lambda$ we see that $\Lambda[f](t)=\langle\delta_0,u(t)\rangle$, $f\in L^2(0,T)$. Since $u_N(t)\rightharpoonup u(t)$ weakly in $H^1(\Omega)$ and $u_N(t)\in V$ for every $N$, the weak limit stays in~$V$, and therefore
	$\langle\delta_0,u_N(t)\rangle\to\langle\delta_0,u(t)\rangle$ for a.e. $t\in[0,T]$. Thus, $\delta_0$ may be passed through the infinite sum, which gives
	\begin{equation}\label{NDrepr}
		\Lambda[f](t) = \sum_{n\in \mathbb{N}} u_n(t)\,\phi_n(0),\qquad \text{a.e. }t\in [0,T].
	\end{equation}
	
	Fix arbitrary $n\in\mathbb{N}$ and suppose $f\in H^1_{\diamond}(0,T)$. Section \ref{fpee} implies that $u_n\in H^2(0,T)$. By the definition of the weak solutions, the Fourier coefficients must obey
	$$\partial_t^2u_n(t)+ \lambda_n(0) u_n(t) = -f(t)\phi_n(0),\quad \text{a.e. }t\in [0,T],$$
	with the initial conditions $u_n(0)=\partial_tu_n(0)=0$. Whence, the standard ODE theory gives
	$$u_n(t) = -\phi_n(0)\int\limits_0^t  \frac{\sin(\tau\sqrt{\lambda_n(0)})}{\sqrt{\lambda_n(0)}}f(t-\tau)\,d\tau ,\qquad \text{a.e. }t\in [0,T].$$
	Substituting $u_n(t)$ in formula (\ref{NDrepr}) yields
	\begin{equation*}\label{lamh1d}
		\Lambda[f](t) = -\sum_{n\in \mathbb{N}}|\phi_n(0)|^2\, S_{\lambda_n(0)}[f](t) ,\qquad \text{a.e. }t\in [0,T],
	\end{equation*}
	and Corollary~\ref{gbeig0} entails the required formula~(\ref{NDreprl}).

	Finally, on $H^1_\diamond(0,T)$ the operator $\tilde{\Lambda}\,:\,f\mapsto\sum_{n\in\mathbb{N}}\lambda_n'(0)\, S_{\lambda_n(0)}[f]$ coincides with $\Lambda$ by the identity (\ref{NDreprl}) proved above for $H^1_\diamond(0,T)$ functions. It is therefore bounded with respect to $L^2(0,T)$ norm, that is, $\|\tilde{\Lambda}[f]\|_{L^2(0,T)}\lesssim\|f\|_{L^2(0,T)}$, $f\in H^1_\diamond(0,T)$. Since $H^1_\diamond(0,T)$ is dense in $L^2(0,T)$, it extends uniquely to an operator in $\mathscr{B}(L^2(0,T))$, and the extension is $\Lambda$. Indeed, given $f\in L^2(0,T)$, choose $\{f_k\}_{k\in\mathbb{N}}\subset H^1_\diamond(0,T)$ with $f_k\to f$ in $L^2(0,T)$; then
	$$\tilde{\Lambda}[f_k]=\sum_{n\in\mathbb{N}}\lambda_n'(0)\,S_{\lambda_n(0)}[f_k]=\Lambda[f_k]\to\Lambda[f] \qquad\text{in }L^2(0,T),$$
	both sides converging because $\Lambda$ is bounded. This concludes the proof.

The first identity is classical, see \cite{blag2,blag1}; we indicate the argument in the present setting. Let $w(t,s)=(u^f(t),u^h(s))_{L^2_c}$. Two integrations by parts in $x$, using $u^f(t,1)=u^h(s,1)=0$ and the Neumann conditions at $x=0$, give
$$(\partial_t^2-\partial_s^2)w=\Lambda[f](t)h(s)-f(t)\Lambda[h](s),\qquad t,s>0,$$
with $w=\partial_tw=0$ on $\{t=0\}$ and $w=\partial_sw=0$ on $\{s=0\}$. Solving this $(1+1)$-dimensional wave equation by d'Alembert and evaluating at $(T,T)$ integrates the right-hand side over the characteristic triangle $\{(t,s):s>t>0,\ t+s<2T\}$, which is precisely the kernel of $J$; collecting terms yields $K=R\Lambda RJ-J\Lambda$.

For the second identity, set $F(t)=(u^f(t),1)_{L^2_c}$. Integrating the equation~in~$x$,
$$F''(t)=\int\limits_0^1\partial_x^2u^f(t,x)\,dx=\partial_xu^f(t,1)-f(t).$$

\noindent\textit{Step 1: existence of a minimizer.}
Let $f\in H(r)$. Define the energy functional
$$E(f) = (f,Kf)_{L^2(0,2T)} - 2(f,B1)_{L^2(0,2T)}+\|f\|^2_{\alpha,\beta}.$$
By the finite speed of propagation, the solution $u^f$ to $(\mathcal{W}_f)$ is such that $u^f(T)$ is supported in $\Omega(r)$, see Section \ref{fpee}. Using the Blagove\v{s}\v{c}enski\u{\i} identities, Theorem~\ref{blagid}, we get

Pick a minimizing sequence $\{f_k\}_{k\in\mathbb{N}}\subset H(r)$, that is, $E(f_k)\to\inf\{E(f):f\in H(r)\}$ as $k\to +\infty$. This implies
$$\|f_k\|^2_{\alpha,\beta}\leqslant E(f_k) + \|\mathds{1}_{\Omega(r)}\|^2_{L^2_c},\qquad k\in\mathbb{N},$$
so the sequence $\{f_k\}_{k\in\mathbb{N}}$ is bounded in $H(r)$. Passing to a subsequence, still denoted by $\{f_k\}_{k\in\mathbb{N}}$, we may assume that $f_k\rightharpoonup f_\infty$ weakly in $H(r)$ for some $f_\infty\in H(r)$.

$$E(f_\infty)\leqslant\liminf\limits_{k\to +\infty} E(f_k)= \inf \{E(f):f\in H(r)\}.$$
This yields that $f_\infty$ is a minimizer of $E(f)$.

Let $f$ be a minimizer of $E$ and $h\in H(r)$. Since $E$ is the sum of a continuous quadratic form and a bounded linear functional on $H(r)$, the map $s\mapsto E(f+sh)$ is a polynomial of degree two in $s$, so $D_hE(f)$ exists and vanishes. \smallskip
\noindent\textit{Step 1: the spectral data determine the Blagove\v{s}\v{c}enski\u{\i} operator.}
By Proposition~\ref{1mainres}, the Neumann-to-Dirichlet map $\Lambda$ is built from the sequence $\{\lambda_n(0),\lambda_n'(0)\}_{n\in\mathbb{N}}$ alone. Hence $\Lambda$ is a function of the spectral data. The Blagove\v{s}\v{c}enski\u{\i} operator is defined in terms of $\Lambda$ by $K=R\Lambda RJ-J\Lambda$, where $J$ and $R$ are known time-domain operators. Consequently, $K$ is a function of the spectral data as well.

\smallskip
\noindent\textit{Step 2: the operator $K$ determines the travel-time volume function.}
By Proposition~\ref{2mainres}, for each $r\in[0, \tau(1)]$ and $\alpha,\beta>0$, the regularized control
$$f_{\alpha,\beta,r}=(\mathcal{P}_rK\mathcal{P}_r+\alpha I+\beta S)^{-1}\mathcal{P}_rB1$$
is uniquely determined by $K$. Moreover, $u^{f_{\alpha,\beta,r}}(T)\to\mathds{1}_{\Omega(r)}$ in $L^2_c$ uniformly with respect to $r\in[0, \tau(1)]$ as $\alpha,\beta\to+0$. The Blagove\v{s}\v{c}enski\u{\i} identities give
$$(f_{\alpha,\beta,r},B1)_{L^2(0,2T)}=\bigl(u^{f_{\alpha,\beta,r}}(T),1\bigr)_{L^2_c}\to(\mathds{1}_{\Omega(r)},1)_{L^2_c}, \qquad \alpha,\beta\to +0,$$
uniformly in $r\in[0, \tau(1)]$. Hence the travel-time volume function
\[
V(r)=\int_{\Omega(r)}\frac{dx}{c^2(x)}= (\mathds{1}_{\Omega(r)},1)_{L^2_c}
\]
is uniquely determined by the spectral data.

\smallskip
\noindent\textit{Step 3: the volume function determines the wave speed.}
Recall that the travel time coordinate is $\tau(x)=\int_0^x c(y)^{-1}\,dy$. Since $c(x)>0$, the map $\tau$ is strictly increasing, and we denote its inverse by $\chi=\tau^{-1}$. Thus $\chi(0)=0$, $\chi(\tau(x))=x$, and $\tau(\chi(t))=t$. Differentiating the identity $\tau(\chi(t))=t$ gives
$$\chi'(t)=\frac{1}{\tau'(\chi(t))}=c(\chi(t)).$$
On the other hand, by changing variables $x=\chi(t)$ in the definition of $V$, we obtain
$$V(r)=\int_0^{\chi(r)}\frac{dx}{c^2(x)}=\int_0^r\frac{\chi'(t)}{c^2(\chi(t))}\,dt=\int_0^r\frac{dt}{c(\chi(t))},\quad r\in [0,\tau(1)].$$
The integrand above is continuous on $[0,\tau(1)]$ and bounded from below and above, so $V\in C^1([0,\tau(1)])$. Hence,
$$c(\chi(t))=\frac{1}{V'(t)},\qquad t\in[0,\tau(1)].$$
Furthermore, $\chi(t)=\int_0^t c(\chi(s))\,ds=\int_0^t \frac{1}{V'(s)}\,ds$ for $t\in[0,\tau(1)]$.

Thus both the coordinate change $t\mapsto \chi(t)$ and the value $c(\chi(t))$ are determined by $V$. Consequently, for $x=\chi(t)$ we recover
$$c(x)=\frac{1}{V'(\tau(x))},\qquad x\in\overline\Omega.$$
Therefore the values of $V$ on $[0,\tau(1)]$ determine $\chi$ on $[0,\tau(1)]$ and $c$ on $\overline\Omega$. 

\smallskip
\noindent\textit{Step 4: conclusion.}
Combining the steps, the spectral data determine wave speeds,
$$\left\{\lambda_n(0),\lambda_n^{\prime}(0)\right\}_{n\in\mathbb{N}}\mapsto\Lambda\mapsto K\mapsto V\mapsto c(x).$$
Let now $c_1,c_2\in\mathcal{C}$ have the same spectral data $\{\lambda_n(0),\lambda_n^{\prime}(0)\}_{n\in\mathbb{N}}$, and write $\tau_i,\chi_i,V_i$ for the corresponding travel time, its inverse and the travel-time volume. By Step~1, the two wave speeds give the same $K$, hence the same controls $f_{\alpha,\beta,r}$, so by Step~2 we get $V_1=V_2$ on $[0,r_\ast]$, where $r_\ast=\min\{\tau_1(1),\tau_2(1)\}$. By Step~3, $\chi_1=\chi_2$ on $[0,r_\ast]$ and $c_1(\chi_1(t))=c_2(\chi_2(t))$ for $t\in [0,r_\ast]$, since both functions are continuous. Thereby, $c_1=c_2$ on $[0,\chi_1(r_\ast)]$. Notice that $\chi_i(r_\ast)=1$ for at least one $i$, so $\chi_1(r_\ast)=\chi_2(r_\ast)=1$. Hence, $c_1=c_2$ on $[0,1]=\overline\Omega$. Thus $\Phi$ is injective; it is surjective by the definition $\mathcal{D}=\Phi(\mathcal{C})$, so $\Phi:\mathcal{C}\to\mathcal{D}$ is bijective.

For any $c \in \mathcal{C}$, the $n$-th eigenvalue is characterized by the min-max principle, namely
$$\lambda_n^c = \min_{\dim E = n} \,\max_{v \in E\setminus\{0\}} \frac{B(v,v)}{\|v\|_{L^2_c}^2},$$
where the minimum is taken over all $n$-dimensional subspaces $E \subset V$; note that $V$ does not depend on $c$. Since $C_{min}\leqslant c\leqslant C_{max}$ on $\Omega$, one has
$$C_{max}^{-2}\|v\|_{L^2(\Omega)}^2\leqslant \|v\|_{L^2_c}^2\leqslant C_{min}^{-2}\|v\|_{L^2(\Omega)}^2,\qquad v\in V.$$
Recall that $\lambda_n^1=\left(\left(n-\frac{1}{2}\right)\pi\right)^2$ the eigenvalues of $(\mathcal{S})$ for $c\equiv 1$. Comparing the Rayleigh quotients termwise and taking the min-max gives the uniform two-sided bound $C_{min}^2\,\lambda_n^1\leqslant\lambda_n^c\leqslant C_{max}^2\,\lambda_n^1$ for any $n\in\mathbb{N}$ and $c\in\mathcal{C}$. Thereby, $\lambda_n^c\asymp n^2$ with constants depending only on the admissible class $\mathcal{C}$ definition.

By the mean value theorem,
$$|c_1(x)^{-2} - c_2(x)^{-2}| \leqslant 2C_{min}^{-3} |c_1(x) - c_2(x)| \leqslant 2C_{min}^{-3} \|c_1 - c_2\|_{C^2(\overline\Omega)}.$$
Consequently, using $\|v\|_{L^2(\Omega)}^2 \leqslant C_{max}^2 \|v\|_{L^2_{c_2}}^2$ we obtain
$$\left| \|v\|_{L^2_{c_1}}^2 - \|v\|_{L^2_{c_2}}^2 \right| = \left| \int_\Omega (c_1^{-2} - c_2^{-2}) |v|^2 \, dx \right| \leqslant C\|c_1 - c_2\|_{C^2(\overline\Omega)} \|v\|_{L^2_{c_2}}^2,$$
where $C = 2C_{min}^{-3} C_{max}^2$. Dropping the absolute value on the left yields $\|v\|_{L^2_{c_1}}^2 \leqslant \big(1 + C \|c_1 - c_2\|_{C^2(\overline\Omega)}\big) \|v\|_{L^2_{c_2}}^2$, and dividing gives
$$\frac{B(v,v)}{\|v\|_{L^2_{c_1}}^2} \geqslant \frac{B(v,v)}{\big(1 + C \|c_1 - c_2\|_{C^2(\overline\Omega)}\big) \|v\|_{L^2_{c_2}}^2},\qquad v\in V\setminus\{0\}.$$
Taking the min-max over all subspaces $E \subset V$ preserves this inequality for the eigenvalues,
$$ \lambda_n^{c_1} \geqslant \frac{\lambda_n^{c_2}}{1 + C\|c_1 - c_2\|_{C^2(\overline\Omega)}}.$$
Hence, $\lambda_n^{c_2}-\lambda_n^{c_1}\leqslant C \lambda_n^{c_1} \|c_1-c_2\|_{C^2(\overline\Omega)}$. By complete symmetry, we also get $\lambda_n^{c_1} - \lambda_n^{c_2} \leqslant C  \lambda_n^{c_2}\|c_1-c_2\|_{C^2(\overline\Omega)}$. Thus, we obtain
$$|\lambda_n^{c_1} - \lambda_n^{c_2}|\leqslant C\max(\lambda_n^{c_1}, \lambda_n^{c_2}) \|c_1-c_2\|_{C^2(\overline\Omega)}.$$
The uniform upper bound $\lambda_n^c\leqslant C_{max}^2\,\lambda_n^1\lesssim n^2$ entails the required estimate.

<!-- lemma-4: body_md -->
\label{eigderabs}
	\textit{There is a constant $C_{abs}>0$ such that $|(\lambda_n^c)'(0)|\leqslant C_{abs}$ for any $n\in\mathbb{N}$ and for any $c\in\mathcal{C}$.}
<!-- end lemma-4 -->
<!-- proof-13: body_md -->
Consider the following coordinate change $\tau(x) = \int_0^x c(y)^{-1}\,dy$ and $x=\tau^{-1}$. The differentials and derivatives transform as follows:
$$d\tau = \frac{dx}{c(x)},\quad dx = c(x(\tau))\,d\tau,\quad \partial_x = \frac{1}{c(x(\tau))}\partial_\tau.$$
Let $\phi$ be a weak solution to $(\mathcal{S})$, then we get that under the coordinate change the identity $(\partial_x\phi,\partial_xv)_{L^2(\Omega)}=\lambda (\phi,v)_{L^2_c}$, $v\in V$, transforms into
$$\int\limits_0^{\tau(1)}\frac{1}{\tilde{c}(\tau)}\partial_\tau\tilde{\phi}(\tau)\partial_\tau\tilde{v}(\tau)\,d\tau = \lambda \int\limits_0^{\tau(1)}\frac{1}{\tilde{c}(\tau)}\tilde{\phi}(\tau)\tilde{v}(\tau)\,d\tau,$$
where $\tilde{\phi}(\tau)=\phi(x(\tau))$, $\tilde{v}(\tau)=v(x(\tau))$, and $\tilde{c}(\tau)=c(x(\tau))$. Now we define the transformed eigenfunction $\psi(\tau)$ by rescaling $\phi(x)$, namely $\psi(\tau) = \tilde{c}(\tau)^{-1/2}\tilde{\phi}(\tau)$. Notice that $\psi(\tau(1)) = c(1)^{-1/2}\phi(1)=0$. Now, we evaluate
$$\partial_\tau\tilde{\phi}(\tau)=\partial_\tau\left(\tilde{c}(\tau)^{1/2}\psi(\tau)\right)= \tilde{c}(\tau)^{1/2}\left(\partial_\tau\psi(\tau)+\frac{\partial_\tau \tilde{c}(\tau)}{2\tilde{c}(\tau)}\psi(\tau)\right).$$
Similarly, defining $\eta(\tau) = \tilde{c}(\tau)^{-1/2}\tilde{v}(\tau)$ gives exactly the same identity. Insert all of the above into the identity:
$$\int\limits_0^{\tau(1)}\left(\partial_\tau\psi(\tau)+\frac{\partial_\tau \tilde{c}(\tau)}{2\tilde{c}(\tau)}\psi(\tau)\right)\left(\partial_\tau\eta(\tau)+\frac{\partial_\tau \tilde{c}(\tau)}{2\tilde{c}(\tau)}\eta(\tau)\right)\,d\tau = \lambda \int\limits_0^{\tau(1)}\psi(\tau)\eta(\tau)\,d\tau.$$
Expanding the product gives three terms:
$$\int\limits_0^{\tau(1)}\partial_\tau\psi\partial_\tau\eta\,d\tau + \int\limits_0^{\tau(1)} \left(\frac{\partial_\tau \tilde{c}}{2\tilde{c}}\right)^2 \psi\eta\,d\tau + \int\limits_0^{\tau(1)}\frac{\partial_\tau \tilde{c}}{2\tilde{c}}\partial_\tau(\psi\eta)\,d\tau. $$
Integrating the latter by parts gives
$$\left(\frac{\partial_\tau \tilde{c}}{2\tilde{c}}\psi\eta\right)\Bigg|_0^{\tau(1)} -\int\limits_0^{\tau(1)}\partial_\tau\left(\frac{\partial_\tau \tilde{c}}{2\tilde{c}}\right)\psi\eta\,d\tau.$$
The first term above vanishes due to the boundary conditions and the fact that $\partial_\tau\tilde{c}(0)=0$. Hence, we obtain
$$\int\limits_0^{\tau(1)}\partial_\tau\psi\partial_\tau\eta\,d\tau + \int\limits_0^{\tau(1)} \left(\left(\frac{\partial_\tau \tilde{c}}{2\tilde{c}}\right)^2 - \partial_\tau\left(\frac{\partial_\tau \tilde{c}}{2\tilde{c}}\right)\right)\psi\eta\,d\tau  =\lambda \int\limits_0^{\tau(1)}\psi\eta\,d\tau.$$
Moreover, the following 
$$\partial_\tau\psi(\tau)=c(x)\partial_x\left(c(x)^{-1/2}\phi(x)\right)= c(x)^{1/2}\partial_x\phi(x)-\frac{1}{2}c(x)^{-1/2}c'(x)\phi(x)$$
shows that the condition $\partial_x\phi(0)=0$ turns into $\partial_\tau\psi(0)=0$, since $c'(0)=0$ for any $c\in\mathcal{C}$. This matches the canonical weak formulation of the Schrödinger equation:
$$-\partial_\tau^2\psi + q\psi = \lambda\psi,\quad q=\frac{1}{4}\left(\frac{\partial_\tau \tilde{c}}{\tilde{c}}\right)^2 - \frac{1}{2}\partial_\tau\left(\frac{\partial_\tau \tilde{c}}{\tilde{c}}\right).$$

The potential $q$ is continuous and uniformly bounded due to the formula $q=(\frac14 (c')^2 -\frac12 cc'')(x(\tau))$, namely $\|q\|_{L^\infty(0, \tau(1))} \leqslant \frac34 C_{max}^2=: Q < \infty$. By variation of parameters, any solution to the Schrödinger equation satisfies the Volterra integral equation:
$$\psi_n(\tau)= \phi_n(0)\cos(\tau\sqrt{\lambda_n^c}) + S_{\lambda_n^c}[q \psi_n](\tau) .$$
Now fix arbitrary $c\in\mathcal{C}$. Note that $\|\cos(\cdot\sqrt{\lambda_n^c})\|_{L^2(0,\tau(1))}^2=\frac12 \tau(1) +(4\sqrt{\lambda_n^c})^{-1}\sin(2\tau(1)\sqrt{\lambda_{n}^c}) \geqslant \frac14 \tau(1) \geqslant(4C_{max})^{-1}=:\varepsilon$ for any $n\in\mathbb{N}$, since if $\tau(1)\sqrt{\lambda_{n}^c}\leqslant 1 <\frac\pi2$, the value of the sine function is nonnegative, and if $\tau(1)\sqrt{\lambda_{n}^c}> 1$, the absolute value of the sine function is at most $1$. Then the inequality
$$|\phi_n(0)|\cdot\|\cos(\cdot\sqrt{\lambda_n^c})\|_{L^2(0,\tau(1))}\leqslant\|\psi_n\|_{L^2(0,\tau(1))}+\|S_{\lambda_n^c}[q \psi_n]\|_{L^2(0,\tau(1))}$$
implies that there is a constant $C_{abs} >0$ depending only on $C_{min}$ and $C_{max}$ such that
$$|\phi_n(0)|^2\lesssim \frac{1}{\varepsilon}\left(1+\frac{\tau(1)Q}{\sqrt{\lambda_n^c}}\right)^2\|\psi_n\|^2_{L^2(0,\tau(1))}\leqslant C_{abs}$$
due to $\|\psi_n\|_{L^2(0,\tau(1))}=\|\phi_n\|_{L^2_c}=1$ and the uniform lower bound $\lambda^c_n\geqslant C^2_{min}\lambda^1_1=(C_{min}\pi/2)^2$ from the proof of Lemma~\ref{weylsn2}. The required follows from Corollary \ref{gbeig0}.

<!-- lemma-5: body_md -->
\label{eigderdiff0}
	\textit{For any $c_1,c_2\in\mathcal{C}$, $|(\lambda_n^{c_1})'(0) - (\lambda_n^{c_2})'(0)|\lesssim \|c_1-c_2\|_{C^2(\overline\Omega)}$ for every $n\in\mathbb{N}$.}
<!-- end lemma-5 -->
<!-- proof-14: body_md -->
Fix $c_1,c_2\in\mathcal{C}$, $n\in\mathbb{N}$, and write $\delta=\|c_1-c_2\|_{C^2(\mathbb{R})}$, $\lambda_i=\lambda_n^{c_i}$, $i=1,2$. By Corollary~\ref{gbeig0},
$$(\lambda_n^{c_1})'(0) - (\lambda_n^{c_2})'(0) =(\phi_n^{c_2}(0) - \phi_n^{c_1}(0))\, (\phi_n^{c_1}(0) + \phi_n^{c_2}(0)),$$
and Lemma~\ref{eigderabs} gives $|\phi_n^{c_1}(0) + \phi_n^{c_2}(0)|\leqslant 2\sqrt{C_{abs}}$. Hence it suffices to show
\begin{equation}\label{eq:trace-diff}
	|\phi_n^{c_2}(0) - \phi_n^{c_1}(0)|\lesssim \delta .
\end{equation}

\smallskip
\noindent\textit{Step 1: common domain and normalized eigenfunctions.}
We place the eigenfunctions of $c_1$ and $c_2$ on a common interval, normalize them at $\tau=0$ instead of in $L^2$, and record uniform bounds on the normalized solutions and their derivatives.

For $c\in\mathcal{C}$ the travel time $\tau_c(x)=\int_0^x c^{-1}(y)\,dy$ is defined for all $x\in\mathbb{R}$ and its inverse $\chi_c$ is defined on $\mathbb{R}$, since $c\equiv 1$ on $\mathbb{R}\setminus\Omega$. Consequently the potential
$$q_c=\Big(\tfrac14 (c')^2-\tfrac12\, c\, c''\Big)\circ\chi_c$$
from the proof of Lemma~\ref{eigderabs} is continuous on $[0,+\infty)$, $\|q_c\|_{L^\infty(0,+\infty)}\leqslant Q$, and $q_c\equiv 0$ on $[\tau_c(1),+\infty)$ due to $c'(1)=c''(1)=0$. Set $L=C_{min}^{-1}$; then $\tau_c(1)\leqslant L$ for every $c\in\mathcal{C}$. Throughout the proof all functions of $\tau$ are considered on the common interval $[0,L]$. Let $v_c\in C^2[0,L]$ be the solution of the initial value problem
$$-\partial_\tau^2 v_c + q_c v_c=\lambda_n^c v_c \ \text{ on } (0,L),\qquad v_c(0)=1,\quad v_c'(0)=0 .$$
Let $\phi_n^c$ and $\psi_n^c$ be as in the proof of Lemma~\ref{eigderabs}; recall that $\psi_n^c(0)=\phi_n^c(0)>0$ and $\partial_\tau\psi_n^c(0)=0$. Uniqueness for the initial value problem yields $v_c=\psi_n^c/\phi_n^c(0)$ on $[0,\tau_c(1)]$. By variation of parameters, $v_c$ satisfies the Volterra equation
\begin{equation}\label{eq:volterra}
	v_c(\tau) = \cos(\tau\sqrt{\lambda_n^c}) + S_{\lambda_n^c}[q_c v_c](\tau),\qquad \tau\in[0,L].
\end{equation}
Note that for every $\lambda>0$ and $0<\tau\leqslant L$,
\begin{equation}\label{eq:S-op-bound}
	\|S_\lambda\|_{L^1(0,\tau)\to L^\infty(0,\tau)}\leqslant\lambda^{-1/2}.
\end{equation}
Hence \eqref{eq:volterra} gives $|v_c(\tau)|\leqslant 1+Q(\lambda_n^c)^{-1/2}\|v_c\|_{L^1(0,\tau)}$ for $\tau\in[0,L]$, and Gr\"onwall's inequality together with Lemma~\ref{weylsn2} yields
\begin{equation}\label{eq:v-bound}
	\|v_c\|_{L^\infty(0,L)}\leqslant e^{QL/\sqrt{\lambda_1^c}}\leqslant e^{2Q/(\pi C_{min}^2)}=:M_0 .
\end{equation}
Differentiating \eqref{eq:volterra} by the Leibniz rule, we get
$$v_c'(\tau)=-\sqrt{\lambda_n^c}\,\sin(\tau\sqrt{\lambda_n^c})+\int_0^\tau\cos\big((\tau-t)\sqrt{\lambda_n^c}\big)\,q_c(t)v_c(t)\,dt,$$
and hence, by \eqref{eq:v-bound},
\begin{equation}\label{eq:vprime-bound}
	\|v_c'\|_{L^\infty(0,L)}\leqslant \sqrt{\lambda_n^c}+LQM_0\lesssim n .
\end{equation}
Finally, by \eqref{eq:S-op-bound}, \eqref{eq:v-bound} and Lemma~\ref{weylsn2},
\begin{equation}\label{eq:rem-bound}
	\|S_{\lambda_n^c}[q_cv_c]\|_{L^\infty(0,L)}\leqslant (\lambda_n^c)^{-1/2}LQM_0\lesssim n^{-1}.
\end{equation}

\smallskip
\noindent\textit{Step 2: comparison of the coordinate changes and potentials.}
Write $\tau_i=\tau_{c_i}$ and $\chi_i=\chi_{c_i}$, $i=1,2$. We show that the travel-time coordinates differ by $O(\delta)$, and split $q_{c_2}-q_{c_1}$ into a part of size $O(\delta)$ and a derivative of a function of size $O(\delta)$; this splitting is needed for the integration by parts in Step~3.

For $x\in\overline{\Omega}$, $|\tau_1(x)-\tau_2(x)|\leqslant\int_0^1|c_1^{-1}-c_2^{-1}|\,dy\leqslant C_{min}^{-2}\delta$, and the difference $\tau_1-\tau_2$ is constant on $[1,+\infty)$ and vanishes on $(-\infty,0]$, so $\|\tau_1-\tau_2\|_{L^\infty(\mathbb{R})}\leqslant C_{min}^{-2}\delta$. Since $\chi_1'=c_1\circ\chi_1\leqslant C_{max}$, for $\tau\geqslant 0$ and $x=\chi_2(\tau)$,
$$|\chi_1(\tau)-\chi_2(\tau)|=|\chi_1(\tau_2(x))-\chi_1(\tau_1(x))|\leqslant C_{max}\|\tau_1-\tau_2\|_{L^\infty(\mathbb{R})}\leqslant C_{max}C_{min}^{-2}\delta .$$
Write $q_{c_i}=a_i^2-\partial_\tau a_i$ with $a_i=\tfrac12 c_i'\circ\chi_i\in C^1[0,L]$, and set $b=a_2-a_1$. Since $c_1'$ is $C_{max}$-Lipschitz, for $\tau\in[0,L]$,
$$2|b(\tau)|\leqslant |c_2'(\chi_2(\tau))-c_1'(\chi_2(\tau))|+|c_1'(\chi_2(\tau))-c_1'(\chi_1(\tau))|\leqslant \delta + C_{max}^2 C_{min}^{-2}\delta,$$
so $\|b\|_{L^\infty(0,L)}\lesssim\delta$. Moreover $\|a_1+a_2\|_{L^\infty(0,L)}\leqslant C_{max}$, hence
$$\|a_2^2-a_1^2\|_{L^\infty(0,L)}\leqslant\|b\|_{L^\infty(0,L)}\|a_1+a_2\|_{L^\infty(0,L)}\lesssim\delta .$$
Finally, $b(0)=\tfrac12(c_2'(0)-c_1'(0))=0$.

\smallskip
\noindent\textit{Step 3: the difference $w=v_{c_2}-v_{c_1}$.}
Subtracting the Volterra equations \eqref{eq:volterra} gives, on $[0,L]$,
\begin{align*}
w ={}& \cos(\cdot\,\sqrt{\lambda_2}) - \cos(\cdot\,\sqrt{\lambda_1})+ (S_{\lambda_2} - S_{\lambda_1})[q_{c_2} v_{c_2}] \\
&+ S_{\lambda_1}\big[(q_{c_2} - q_{c_1}) v_{c_2}\big] + S_{\lambda_1}[q_{c_1} w] .
\end{align*}
We estimate the first three terms in $L^\infty(0,L)$.

First, by the mean value theorem and Lemma~\ref{weylsn2}, for $\tau\in[0,L]$,
\begin{equation}\label{eq:cos-diff}
|\cos(\tau\sqrt{\lambda_2}) - \cos(\tau\sqrt{\lambda_1})| \leqslant \tau|\sqrt{\lambda_2}-\sqrt{\lambda_1}|= \frac{\tau |\lambda_2 - \lambda_1|}{\sqrt{\lambda_1}+\sqrt{\lambda_2}} \lesssim n\delta .
\end{equation}

Second, for $s\in[0,2L]$ the function $\lambda\mapsto \frac{\sin(s\sqrt\lambda)}{\sqrt\lambda}$ satisfies
$$\Big|\partial_\lambda\frac{\sin(s\sqrt{\lambda})}{\sqrt\lambda}\Big| = \Big|\frac{s \cos(s\sqrt{\lambda})}{2\lambda} - \frac{\sin(s\sqrt{\lambda})}{2\lambda^{3/2}}\Big|\leqslant \frac{L}{\lambda}+\frac{1}{2\lambda^{3/2}},$$
and hence, by the mean value theorem in $\lambda$ and Lemma~\ref{weylsn2},
\begin{equation}\label{eq:kernel-diff}
\sup_{s\in[0,2L]}\Big|\frac{\sin(s\sqrt{\lambda_2})}{\sqrt{\lambda_2}}-\frac{\sin(s\sqrt{\lambda_1})}{\sqrt{\lambda_1}}\Big|
\leqslant |\lambda_2-\lambda_1|\sup_{\lambda\in[\min\lambda_i,\max\lambda_i]}\Big(\frac{L}{\lambda}+\frac{1}{2\lambda^{3/2}}\Big).
\end{equation}
The latter is bounded by a constant multiple of $n^{-2}\cdot n^2\delta=\delta$. In particular $\|S_{\lambda_2}-S_{\lambda_1}\|_{L^1(0,L)\to L^\infty(0,L)}\lesssim\delta$, and together with \eqref{eq:v-bound},
\begin{align*}
\|(S_{\lambda_2} - S_{\lambda_1})[q_{c_2} v_{c_2}]\|_{L^\infty(0,L)}& \leqslant \|S_{\lambda_2}-S_{\lambda_1}\|_{L^1(0,L)\to L^\infty(0,L)}\,\|q_{c_2}v_{c_2}\|_{L^1(0,L)}\\
& \lesssim LQM_0\,\delta .
\end{align*}

Third, $q_{c_2}-q_{c_1}=(a_2^2-a_1^2)-\partial_\tau b$. By Step 2, \eqref{eq:v-bound} and \eqref{eq:S-op-bound},
\begin{align*}
	\|S_{\lambda_1}[(a_2^2-a_1^2) v_{c_2}]\|_{L^\infty(0,L)} & \leqslant \lambda_1^{-1/2}\|(a_2^2-a_1^2)v_{c_2}\|_{L^1(0,L)}\\
	& \leqslant L\lambda_1^{-1/2}M_0\|a_2^2-a_1^2\|_{L^\infty(0,L)}\lesssim n^{-1}\delta .
\end{align*}
For the remaining part we integrate by parts. For $\tau\in[0,L]$,
\begin{align*}
S_{\lambda_1}[b'\,v_{c_2}](\tau)=&\int_0^\tau \frac{\sin((\tau-t)\sqrt{\lambda_1})}{\sqrt{\lambda_1}}\,b'(t)\,v_{c_2}(t)\,dt\\ 
=&\Big[\frac{\sin((\tau-t)\sqrt{\lambda_1})}{\sqrt{\lambda_1}}\,b(t)\,v_{c_2}(t)\Big]_{t=0}^{t=\tau}\\ 
&-\int_0^\tau b(t)\,\partial_t\Big(\frac{\sin((\tau-t)\sqrt{\lambda_1})}{\sqrt{\lambda_1}}v_{c_2}(t)\Big)\,dt .
\end{align*}
The boundary term vanishes, since $\sin 0=0$ and $b(0)=0$. Furthermore, by \eqref{eq:v-bound} and \eqref{eq:vprime-bound}, for $0\leqslant t\leqslant\tau\leqslant L$,
\begin{align*}
\partial_t\Big(\frac{\sin((\tau-t)\sqrt{\lambda_1})}{\sqrt{\lambda_1}}v_{c_2}(t)\Big)
=&-\cos((\tau-t)\sqrt{\lambda_1})\,v_{c_2}(t)\\ &+\frac{\sin((\tau-t)\sqrt{\lambda_1})}{\sqrt{\lambda_1}}\,v_{c_2}'(t).
\end{align*}
The modulus of the latter is upper bounded by $M_0+\lambda_1^{-1/2}(\sqrt{\lambda_2}+LQM_0)\lesssim 1$, where $\sqrt{\lambda_2/\lambda_1}\leqslant C_{max}/C_{min}$ by Lemma~\ref{weylsn2}. Hence
$$\|S_{\lambda_1}[b'\, v_{c_2}]\|_{L^\infty(0,L)}\lesssim L\|b\|_{L^\infty(0,L)}\lesssim\delta .$$

Finally, by \eqref{eq:S-op-bound} on $(0,\tau)$, $|S_{\lambda_1}[q_{c_1} w](\tau)|\leqslant Q\lambda_1^{-1/2}\|w\|_{L^1(0,\tau)}$ for $\tau\in[0,L]$. Collecting the estimates, for $\tau\in[0,L]$, $|w(\tau)|\lesssim n\,\delta + Q\lambda_1^{-1/2}\|w\|_{L^1(0,\tau)}$, and Gr\"onwall's inequality together with $QL/\sqrt{\lambda_1}\leqslant 2Q/(\pi C_{min}^2)$ yields
\begin{equation}\label{eq:w-bound}
	\|v_{c_2}-v_{c_1}\|_{L^\infty(0,L)}\lesssim n\delta .
\end{equation}
By \eqref{eq:S-op-bound} and \eqref{eq:w-bound}, $\|S_{\lambda_1}[q_{c_1}w]\|_{L^\infty(0,L)}\leqslant\lambda_1^{-1/2}LQ\|w\|_{L^\infty(0,L)}\lesssim\delta$, and hence the three estimates above give
\begin{equation}\label{eq:rem-diff}
	\big\|S_{\lambda_2}[q_{c_2}v_{c_2}]-S_{\lambda_1}[q_{c_1}v_{c_1}]\big\|_{L^\infty(0,L)}\lesssim\delta .
\end{equation}

\smallskip
\noindent\textit{Step 4: boundary traces.}
Since $\|\psi_n^c\|_{L^2(0,\tau_c(1))}=\|\phi_n^c\|_{L^2_c}=1$ and $v_c=\psi_n^c/\phi_n^c(0)$ on $[0,\tau_c(1)]$, we have
$$N_c:=\|v_c\|_{L^2(0,\tau_c(1))}=\frac{1}{\phi_n^c(0)},\qquad C_{abs}^{-1/2}\leqslant N_c\leqslant \sqrt{L}\,M_0 ,$$
by Lemma~\ref{eigderabs} and \eqref{eq:v-bound}. Denote $N_i=N_{c_i}$. Then
$$|\phi_n^{c_2}(0)-\phi_n^{c_1}(0)|=\frac{|N_1-N_2|}{N_1N_2}= \frac{|N_1^2-N_2^2|}{N_1N_2(N_1+N_2)}\leqslant \frac{C_{abs}^{3/2}}{2}\,|N_1^2-N_2^2| .$$
Assume without loss of generality that $\tau_1(1)\geqslant\tau_2(1)$. Then
\begin{equation}\label{eq:N-split}
|N_1^2-N_2^2|
\leqslant \Big|\int_0^{\tau_2(1)} (v_{c_1}^2 - v_{c_2}^2) \, d\tau\Big| + \int_{\tau_2(1)}^{\tau_1(1)} |v_{c_1}|^2 \, d\tau ,
\end{equation}
where the second term is upper bounded by $M_0^2|\tau_1(1)-\tau_2(1)|\leqslant M_0^2C_{min}^{-2}\delta$ by Step 2. For the first term we substitute \eqref{eq:volterra} for $v_{c_1}$ and $v_{c_2}$ and expand:
\begin{align*}
v_{c_1}^2-v_{c_2}^2={}&\cos^2(\cdot\sqrt{\lambda_1})-\cos^2(\cdot\sqrt{\lambda_2})\\
&+2\big(\cos(\cdot\sqrt{\lambda_1})-\cos(\cdot\sqrt{\lambda_2})\big)S_{\lambda_1}[q_{c_1}v_{c_1}]\\
&+2\cos(\cdot\sqrt{\lambda_2})\big(S_{\lambda_1}[q_{c_1}v_{c_1}]-S_{\lambda_2}[q_{c_2}v_{c_2}]\big)\\
&+\big(S_{\lambda_1}[q_{c_1}v_{c_1}]-S_{\lambda_2}[q_{c_2}v_{c_2}]\big)\big(S_{\lambda_1}[q_{c_1}v_{c_1}]+S_{\lambda_2}[q_{c_2}v_{c_2}]\big).
\end{align*}
The first line is integrated explicitly: since $\cos^2(\tau\sqrt\lambda)=\tfrac12(1+\cos(2\tau\sqrt\lambda))$,
$$\int\limits_0^{\tau_2(1)}\big(\cos^2(\tau\sqrt{\lambda_1})-\cos^2(\tau\sqrt{\lambda_2})\big)\,d\tau
=\frac14\Big(\frac{\sin(2\tau_2(1)\sqrt{\lambda_1})}{\sqrt{\lambda_1}}-\frac{\sin(2\tau_2(1)\sqrt{\lambda_2})}{\sqrt{\lambda_2}}\Big),$$
and $\tau_2(1)\leqslant L$, so its absolute value is bounded by a constant multiple of $\delta$ due to \eqref{eq:kernel-diff}. The remaining three lines are bounded in $L^\infty(0,\tau_2(1))$ by \eqref{eq:cos-diff} and \eqref{eq:rem-bound}, by \eqref{eq:rem-diff}, and by \eqref{eq:rem-diff} and \eqref{eq:rem-bound}, respectively:
$$n\delta\cdot n^{-1}=\delta,\qquad \delta,\qquad \delta\cdot n^{-1}\leqslant\delta .$$
Integrating over $(0,\tau_2(1))$ and inserting into \eqref{eq:N-split} gives $|N_1^2-N_2^2|\lesssim\delta$. This proves \eqref{eq:trace-diff} and the lemma.

\label{hiddenreg}
\textit{There is $L>0$ depending only on $C_{min}$, $C_{max}$ and $T$ such that $\|\Lambda^c\|_{L^2(0,2T)\to H^1(0,2T)}\leqslant L$ for all $c\in\mathcal{C}$.}
<!-- end lemma-6 -->
<!-- proof-16: body_md -->
By density of $H^1_0(0,2T)$ in $L^2(0,2T)$ and boundedness of $\Lambda^c$ on $L^2(0,2T)$, it suffices to show $\|\partial_t\Lambda^c[f]\|_{L^2(0,2T)}\lesssim\|f\|_{L^2(0,2T)}$ for $f\in H^1_0(0,2T)$. The proof relies on energy estimates on $(0,s)\times\Omega$ for some $s\geqslant 2T$. Applied on the time interval $(0,2T)$, they would require $s=2T$ and hence would contain terms evaluated at the fixed time $s=2T$, for which the energy estimate of Section~\ref{fpee} provides no bound. We therefore work on $(0,4T)$ and choose $s\in[2T,4T]$.

Fix $f\in H^1_0(0,2T)$ and extend it by zero to $(0,4T)$; the extension belongs to $H^1_\diamond(0,4T)$. Let $u\in H^2((0,4T)\times\Omega)$ be the strong solution of $(\mathcal{W}_f)$ on $(0,4T)$ given by Section~\ref{fpee}; by uniqueness, its restriction to $(0,2T)$ is the solution of the original problem, and $\partial_t\Lambda^c[f](t)=\partial_tu(t,0)$ on $(0,2T)$. The energy estimate on $(0,4T)$ reads
\begin{equation}\label{eq:hr-energy}
\|\partial_xu\|_{L^\infty(0,4T;L^2(\Omega))}+\|\partial_tu\|_{L^2((0,4T)\times\Omega)}\lesssim\|f\|_{L^2(0,2T)}.
\end{equation}

Fix $s\in(0,4T]$ and set $m(x)=1-x$. Multiply the equation $c^{-2}\partial_t^2u-\partial_x^2u=0$ by $m\,\partial_xu$ and integrate over $(0,s)\times\Omega$. Since $u\in H^2((0,4T)\times\Omega)$, the products $c^{-2}m\,\partial_tu\,\partial_xu$, $c^{-2}m|\partial_tu|^2$ and $m|\partial_xu|^2$ belong to $W^{1,1}((0,4T)\times\Omega)$. Hence the integrations by parts below are justified. Writing $c^{-2}m\,\partial_t^2u\,\partial_xu=\partial_t\big(c^{-2}m\,\partial_tu\,\partial_xu\big)-\tfrac12c^{-2}m\,\partial_x\big(|\partial_tu|^2\big)$ and using $u(0)=\partial_tu(0)=0$, and ${c(0)=1}$, we get
\begin{align*}
	\int\limits_0^s\!\!\int\limits_\Omega c^{-2}m\,\partial_t^2u\,\partial_xu\,dx\,dt=&\int\limits_\Omega c^{-2}m\,\partial_tu(s)\,\partial_xu(s)\,dx+\frac12\int\limits_0^s|\partial_tu(t,0)|^2\,dt\\
	&+\frac12\int\limits_0^s\!\!\int\limits_\Omega\partial_x(c^{-2}m)|\partial_tu|^2\,dx\,dt .
\end{align*}
Similarly, $m\,\partial_x^2u\,\partial_xu=\tfrac12\partial_x\big(m|\partial_xu|^2\big)+\tfrac12|\partial_xu|^2$ and $\partial_xu(t,0)=f(t)$ give
$$-\int\limits_0^s\!\!\int\limits_\Omega m\,\partial_x^2u\,\partial_xu\,dx\,dt=\frac12\int\limits_0^s|f|^2\,dt-\frac12\int\limits_0^s\!\!\int\limits_\Omega|\partial_xu|^2\,dx\,dt .$$
Adding the two identities, the left-hand side vanishes, whence
\begin{align*}
	\frac12\int\limits_0^s|\partial_tu(t,0)|^2\,dt=&-\int\limits_\Omega c^{-2}m\,\partial_tu(s)\,\partial_xu(s)\,dx-\frac12\int\limits_0^s\!\!\int\limits_\Omega\partial_x(c^{-2}m)|\partial_tu|^2\,dx\,dt\\
	&-\frac12\int\limits_0^s|f|^2\,dt+\frac12\int\limits_0^s\!\!\int\limits_\Omega|\partial_xu|^2\,dx\,dt .
\end{align*}
Dropping the nonpositive term $-\frac12\int_0^s|f|^2\,dt$, we obtain
\begin{align}
\frac12\int\limits_0^s|\partial_tu(t,0)|^2\,dt\leqslant &\Bigg|\int\limits_\Omega c^{-2}m\,\partial_tu(s)\,\partial_xu(s)\,dx\Bigg|\nonumber\\
&+\Big(\frac{C_{max}}{C_{min}^3}+\frac{1}{2C_{min}^2}\Big)\int\limits_0^s\!\!\int\limits_\Omega|\partial_tu|^2\,dx\,dt\nonumber\\
&+\frac12\int\limits_0^s\!\!\int\limits_\Omega|\partial_xu|^2\,dx\,dt,\label{eq:hr-multiplier}
\end{align}
where $|\partial_x(c^{-2}m)|=|2c'c^{-3}m+c^{-2}|\leqslant 2C_{max}C_{min}^{-3}+C_{min}^{-2}$ since $0\leqslant m\leqslant1$.

Denote $J(s)=\int_\Omega c^{-2}m\,\partial_tu(s)\,\partial_xu(s)\,dx$, $s\in(0,4T]$. By the Cauchy--Schwarz inequality in $x$ and then in $s$, together with \eqref{eq:hr-energy},
\begin{align*}
\int\limits_{2T}^{4T}|J(s)|\,ds&\leqslant C_{min}^{-2}\int\limits_{2T}^{4T}\|\partial_tu(s)\|_{L^2(\Omega)}\|\partial_xu(s)\|_{L^2(\Omega)}\,ds\\
&\lesssim\|\partial_tu\|_{L^2((0,4T)\times\Omega)}\|\partial_xu\|_{L^\infty(0,4T;L^2(\Omega))}\lesssim\|f\|^2_{L^2(0,2T)} .
\end{align*}
Hence there is $s_0\in[2T,4T]$ such that $|J(s_0)|\leqslant\frac{1}{2T}\int_{2T}^{4T}|J(s)|\,ds\lesssim\|f\|^2_{L^2(0,2T)}$ (otherwise $|J|>\frac{1}{2T}\int_{2T}^{4T}|J(s)|\,ds$ on $[2T,4T]$, and integrating over $[2T,4T]$ gives a contradiction). The remaining two terms on the right-hand side of \eqref{eq:hr-multiplier} are bounded, for every $s\in(0,4T]$, by
$$\Big(\frac{C_{max}}{C_{min}^3}+\frac{1}{2C_{min}^2}\Big)\|\partial_tu\|^2_{L^2((0,4T)\times\Omega)}+2T\|\partial_xu\|^2_{L^\infty(0,4T;L^2(\Omega))}\lesssim\|f\|^2_{L^2(0,2T)}$$
due to \eqref{eq:hr-energy}. Since $s_0\geqslant 2T$, the estimate \eqref{eq:hr-multiplier} with $s=s_0$ yields
$$\|\partial_tu(\cdot,0)\|^2_{L^2(0,2T)}\leqslant\int\limits_0^{s_0}|\partial_tu(t,0)|^2\,dt\lesssim\|f\|^2_{L^2(0,2T)} .$$
Since $\partial_t\Lambda^c[f]=\partial_tu(\cdot,0)$ on $(0,2T)$, this gives $\|\partial_t\Lambda^c[f]\|_{L^2(0,2T)}\lesssim\|f\|_{L^2(0,2T)}$ with a constant depending only on $C_{min}$, $C_{max}$ and $T$.

<!-- lemma-7: body_md -->
\label{interpolateA}
\textit{Let $A:L^2(0,2T)\to H^1(0,2T)$ be bounded. Then}
$$\|A\|_{L^2(0,2T)\to L^2(0,2T)}\lesssim\|A\|^{1/2}_{L^2(0,2T)\to H^{-1}(0,2T)}\,\|A\|^{1/2}_{L^2(0,2T)\to H^{1}(0,2T)} .$$
<!-- end lemma-7 -->
<!-- proof-17: body_md -->
For $v\in H^1_0(0,2T)$ the inequality $\|v\|^2_{L^2(0,2T)}\leqslant\|v\|_{H^{-1}(0,2T)}\|v\|_{H^1(0,2T)}$ is immediate, since $v$ may be tested against itself in the duality between $H^{-1}(0,2T)$ and $H^1_0(0,2T)$. For $v\in H^1(0,2T)$ this argument is unavailable, as $v$ need not vanish at the endpoints, and we test $v$ against a truncation instead.

Let $v\in H^1(0,2T)$ and $h\in(0,T)$, and let $\zeta_h:[0,2T]\to[0,1]$ be Lipschitz, equal to one on $[h,2T-h]$, vanishing at the endpoints and linear in between, so that $|\zeta_h'|\leqslant h^{-1}$ and $\zeta_h'$ is supported in $E_h=(0,h)\cup(2T-h,2T)$, with $|E_h|=2h$. Since $\zeta_hv\in H^1_0(0,2T)$ and $1-\zeta_h$ vanishes outside $E_h$,
\begin{align*}
\|v\|^2_{L^2(0,2T)}&=\int_{E_h}v^2(1-\zeta_h)\,dt+\int_0^{2T}v\,\zeta_hv\,dt\\
&\leqslant 2h\|v\|^2_{L^\infty(0,2T)}+\|v\|_{H^{-1}(0,2T)}\|\zeta_hv\|_{H^1(0,2T)} .
\end{align*}
By the product rule,
\begin{equation}\label{prrulzethv}
	\|\zeta_hv\|_{H^1(0,2T)}\lesssim\|v\|_{H^1(0,2T)}+h^{-1}\|v\|_{L^2(E_h)}\lesssim\|v\|_{H^1(0,2T)}+h^{-1/2}\|v\|_{L^\infty(0,2T)}.
\end{equation}
By the Gagliardo--Nirenberg inequality on a bounded interval, 
$$\|v\|_{L^\infty(0,2T)}\lesssim\|v\|^{1/2}_{L^2(0,2T)}\|v'\|^{1/2}_{L^2(0,2T)}+\|v\|_{L^2(0,2T)},$$
and squaring this, together with $\|v\|^2_{L^2(0,2T)}\leqslant\|v\|_{L^2(0,2T)}\|v\|_{H^1(0,2T)}$, gives $\|v\|^2_{L^\infty(0,2T)}\lesssim\|v\|_{L^2(0,2T)}\|v\|_{H^1(0,2T)}$. Inserting this into \eqref{prrulzethv} and both into the first display for $\|v\|^2_{L^2(0,2T)}$, we obtain
\begin{align*}
\|v\|^2_{L^2(0,2T)}\lesssim &\, h\,\|v\|_{L^2(0,2T)}\|v\|_{H^1(0,2T)}\\
& +\|v\|_{H^{-1}(0,2T)}\Bigl(\|v\|_{H^1(0,2T)}+h^{-1/2}\|v\|^{1/2}_{L^2(0,2T)}\|v\|^{1/2}_{H^1(0,2T)}\Bigr).
\end{align*}
Assume $v\neq0$ and take $h=\rho\,\|v\|_{L^2(0,2T)}/\|v\|_{H^1(0,2T)}$ with a constant $\rho\in(0,T]$; this is admissible, since the quotient does not exceed one. The first term on the right becomes $\rho\,\|v\|^2_{L^2(0,2T)}$ up to the implicit constant, so for $\rho$ small enough it is absorbed into the left-hand side; and $h^{-1/2}\|v\|^{1/2}_{L^2(0,2T)}\|v\|^{1/2}_{H^1(0,2T)}=\rho^{-1/2}\|v\|_{H^1(0,2T)}$. Hence
$$\|v\|^2_{L^2(0,2T)}\lesssim\|v\|_{H^{-1}(0,2T)}\|v\|_{H^1(0,2T)},\qquad v\in H^1(0,2T),$$
with a constant depending only on $T$. Let $v=A[f]$, $f\in L^2(0,2T)$, so taking the supremum over $\|f\|_{L^2(0,2T)}\leqslant1$ gives the required estimate.

<!-- proposition-8: body_md -->
\label{NDstab}
\textit{The map $\mathcal{D}\ni\Phi(c)\mapsto\Lambda^c\in\mathscr{B}(L^2(0,2T))$ is H\"older continuous with every exponent $\theta<\frac12$, that is, for any $\theta\in(0,\frac12)$ and $c_1,c_2\in\mathcal{C}$,
$$\|\Lambda^{c_1}-\Lambda^{c_2}\|_{L^2(0,2T)\to L^2(0,2T)}\lesssim_\theta\|\Phi(c_1)-\Phi(c_2)\|_{\ell^\infty_2\times\ell^\infty_0}^{\theta},$$
where the constant depends on $\theta$ and is not bounded as $\theta\to\frac12$. Moreover, if $\varepsilon=\|\Phi(c_1)-\Phi(c_2)\|_{\ell^\infty_2\times\ell^\infty_0}\leqslant\frac12$, then
$$\|\Lambda^{c_1}-\Lambda^{c_2}\|_{L^2(0,2T)\to L^2(0,2T)}\lesssim\big(\varepsilon\log(1/\varepsilon)\big)^{1/2}.$$}
<!-- end proposition-8 -->
<!-- proof-18: body_md -->
Set $\varepsilon=\|\Phi(c_1)-\Phi(c_2)\|_{\ell^\infty_2\times\ell^\infty_0}$. By \eqref{eq:Phi-norm}, $|\lambda^{c_1}_n-\lambda^{c_2}_n|\leqslant n^2\varepsilon$ and $|(\lambda^{c_1}_n)'(0)-(\lambda^{c_2}_n)'(0)|\leqslant\varepsilon$ for all $n\in\mathbb{N}$. Besides this, we use only $|(\lambda^{c_i}_n)'(0)|\leqslant C_{abs}$ from Lemma~\ref{eigderabs} and $\lambda^{c_i}_n\asymp n^2$ from Lemma~\ref{weylsn2}.

We first show
\begin{equation}\label{eq:Hminus1}
\|\Lambda^{c_1}-\Lambda^{c_2}\|_{L^2(0,2T)\to H^{-1}(0,2T)}\lesssim\varepsilon\log(1/\varepsilon),\qquad\varepsilon\leqslant\tfrac12 .
\end{equation}
Fix $f\in L^2(0,2T)$ and $\varphi\in H^1_0(0,2T)$ with $\|\varphi\|_{H^1(0,2T)}\leqslant1$. Write $S_\lambda[f](t)=\int_0^t k_\lambda(t-s)f(s)\,ds$ with $k_\lambda(\rho)=\frac{\sin(\rho\sqrt\lambda)}{\sqrt\lambda}$, and set
$$g_\lambda(\rho)=\int\limits_0^\rho k_\lambda(\zeta)\,d\zeta=\frac{1-\cos(\rho\sqrt\lambda)}{\lambda},\qquad \rho\geqslant0,$$
so that $g_\lambda'=k_\lambda$ and $g_\lambda(0)=0$. The integral $\int_0^{2T}S_\lambda[f](t)\varphi(t)\,dt$ is a double integral over the triangle $\{(s,t):0<s<t<2T\}$. By Fubini's theorem, integrating first in $t$, which for fixed $s$ ranges over $(s,2T)$, then substituting $\rho=t-s$ in the inner integral, and finally integrating by parts in $\rho$, where the boundary terms vanish since $g_\lambda(0)=0$ and $\varphi(2T)=0$, we get
\begin{align}
\int\limits_0^{2T}S_\lambda[f](t)\varphi(t)\,dt&=\int\limits_0^{2T}f(s)\int\limits_s^{2T}k_\lambda(t-s)\varphi(t)\,dt\,ds\nonumber\\
&=\int\limits_0^{2T}f(s)\int\limits_0^{2T-s}k_\lambda(\rho)\varphi(\rho+s)\,d\rho\,ds\nonumber\\
&=-\int\limits_0^{2T}f(s)\int\limits_0^{2T-s}g_\lambda(\rho)\,\varphi'(\rho+s)\,d\rho\,ds .\label{eq:S-ibp}
\end{align}
Let $\lambda,\mu\asymp n^2$. Since $|g_\lambda|\leqslant 2\lambda^{-1}$, $\|g_\lambda\|_{L^\infty(0,2T)}\lesssim n^{-2}$ and $\|g_\lambda-g_\mu\|_{L^\infty(0,2T)}\lesssim n^{-2}$. On the other hand, for fixed $\rho\in[0,2T]$ the map $\lambda\mapsto g_\lambda(\rho)$ is $C^1$ on $(0,+\infty)$ with
$$|\partial_\lambda g_\lambda(\rho)|=\Big|-\frac{1-\cos(\rho\sqrt\lambda)}{\lambda^2}+\frac{\rho\sin(\rho\sqrt\lambda)}{2\lambda^{3/2}}\Big|\leqslant\frac{2}{\lambda^2}+\frac{T}{\lambda^{3/2}}.$$
By the mean value theorem, $g_\lambda(\rho)-g_\mu(\rho)=\partial_\lambda g_{\tilde\lambda}(\rho)\,(\lambda-\mu)$ for some $\tilde\lambda$ between $\lambda$ and $\mu$; as $\tilde\lambda\asymp n^2$, we have $|\partial_\lambda g_{\tilde\lambda}(\rho)|\lesssim n^{-3}$. Together with the bound $\|g_\lambda-g_\mu\|_{L^\infty(0,2T)}\lesssim n^{-2}$ above,
$$\|g_\lambda-g_\mu\|_{L^\infty(0,2T)}\lesssim\min\{n^{-2},\,n^{-3}|\lambda-\mu|\}.$$
Now take $\lambda=\lambda^{c_1}_n$ and $\mu=\lambda^{c_2}_n$, so that $|\lambda-\mu|\lesssim n^2\varepsilon$ and the second bound becomes $\min\{n^{-2},\,n^{-1}\varepsilon\}$. 

We now return to \eqref{eq:S-ibp}. For $s\in(0,2T)$ the inner integral satisfies
$$\Bigg|\int\limits_0^{2T-s}g_\lambda(\rho)\,\varphi'(\rho+s)\,d\rho\Bigg|\leqslant\|g_\lambda\|_{L^\infty(0,2T)},$$
since $\|\varphi'\|_{L^1(0,2T)}\leqslant\|\varphi\|_{H^1(0,2T)}$. Hence, 
$$\Bigg|\int\limits_0^{2T}S_\lambda[f](t)\varphi(t)\,dt\Bigg|\lesssim\|g_\lambda\|_{L^\infty(0,2T)}\|f\|_{L^2(0,2T)} .$$
Subtracting \eqref{eq:S-ibp} for $\mu$ from \eqref{eq:S-ibp} for $\lambda$ replaces $g_\lambda$ by $g_\lambda-g_\mu$ in the inner integral, and the same argument gives
$$\Bigg|\int\limits_0^{2T}\big(S_\lambda-S_\mu\big)[f](t)\varphi(t)\,dt\Bigg|\lesssim\|g_\lambda-g_\mu\|_{L^\infty(0,2T)}\|f\|_{L^2(0,2T)} .$$
Taking the supremum over $\varphi$ and inserting the bounds on $g_\lambda$ and $g_\lambda-g_\mu$ for $\lambda=\lambda^{c_1}_n$, $\mu=\lambda^{c_2}_n$, we obtain
\begin{align}
\|S_{\lambda^{c_1}_n}[f]\|_{H^{-1}(0,2T)}&\lesssim n^{-2}\|f\|_{L^2(0,2T)},\nonumber\\
\big\|\big(S_{\lambda^{c_1}_n}-S_{\lambda^{c_2}_n}\big)[f]\big\|_{H^{-1}(0,2T)}&\lesssim\min\{n^{-2},\,n^{-1}\varepsilon\}\,\|f\|_{L^2(0,2T)} .\label{eq:S-Hminus1}
\end{align}

By Proposition~\ref{1mainres}, $\Lambda^{c_1}[f]-\Lambda^{c_2}[f]=I_1+I_2$, where
$$I_1=\sum_{n\in\mathbb{N}}\big((\lambda^{c_1}_n)'(0)-(\lambda^{c_2}_n)'(0)\big)\,S_{\lambda^{c_1}_n}[f],\quad I_2=\sum_{n\in\mathbb{N}}(\lambda^{c_2}_n)'(0)\,\big(S_{\lambda^{c_1}_n}-S_{\lambda^{c_2}_n}\big)[f].$$
The first bound in \eqref{eq:S-Hminus1} gives $\|I_1\|_{H^{-1}(0,2T)}\lesssim\varepsilon\sum_{n\in\mathbb{N}} n^{-2}\|f\|_{L^2(0,2T)}\lesssim\varepsilon\|f\|_{L^2(0,2T)}$. For $I_2$, let $N=\lceil\varepsilon^{-1}\rceil$ and use the second bound in \eqref{eq:S-Hminus1} with $n^{-1}\varepsilon$ for $n\leqslant N$ and with $n^{-2}$ for $n>N$:
\begin{align*}
\|I_2\|_{H^{-1}(0,2T)}&\lesssim C_{abs}\Big(\varepsilon\sum_{n\leqslant N}\frac1n+\sum_{n>N}\frac1{n^2}\Big)\|f\|_{L^2(0,2T)}\\
&\lesssim\big(\varepsilon(1+\log N)+N^{-1}\big)\|f\|_{L^2(0,2T)}\\
&\lesssim\varepsilon\log(1/\varepsilon)\|f\|_{L^2(0,2T)},
\end{align*}
where the last step uses $\varepsilon\leqslant\frac12$. This proves \eqref{eq:Hminus1}. By Lemma~\ref{hiddenreg}, the operator $A=\Lambda^{c_1}-\Lambda^{c_2}$ maps $L^2(0,2T)$ into $H^1(0,2T)$ with $\|A\|_{L^2(0,2T)\to H^1(0,2T)}\leqslant 2L$, so Lemma~\ref{interpolateA} applies and, together with \eqref{eq:Hminus1}, gives
$$\|\Lambda^{c_1}-\Lambda^{c_2}\|_{L^2(0,2T)\to L^2(0,2T)} \lesssim\big(\varepsilon\log(1/\varepsilon)\big)^{1/2},$$
which is the second estimate of the proposition.

Finally, fix $\theta\in(0,\frac12)$. For $\varepsilon\leqslant\frac12$, the second estimate and $\varepsilon\log(1/\varepsilon)\leqslant\big(e(1-2\theta)\big)^{-1}\varepsilon^{2\theta}$ give the first estimate with a constant proportional to $(1-2\theta)^{-1/2}$, which is unbounded as $\theta\to\frac12$. For $\varepsilon>\frac12$, Lemma~\ref{hiddenreg} and $H^1(0,2T)\hookrightarrow L^2(0,2T)$ give
$$\|\Lambda^{c_1}-\Lambda^{c_2}\|_{L^2(0,2T)\to L^2(0,2T)}\leqslant2L\leqslant2^{1+\theta}L\,\varepsilon^\theta .$$
The proposition is proven.

Write $w=c_1-c_2$ and $A=\|w\|_{C^2(\overline\Omega)}$. Since $\|c_j\|_{C^3(\mathbb{R})}\leqslant C_{max}$, $j=1,2$, we have $A\leqslant 2C_{max}$ and

Let $\eta\in(0,1)$ and $x,y\in\overline\Omega$, $x\neq y$. Bounding the increment of $w''$
in two ways, by its supremum and by the mean value theorem, gives
$$|w''(x)-w''(y)|\leqslant\min\bigl\{2A,\ 2C_{max}|x-y|\bigr\}
\leqslant(2A)^{1-\eta}\bigl(2C_{max}|x-y|\bigr)^{\eta},$$
where the last step uses $\min\{a,b\}\leqslant a^{1-\eta}b^{\eta}$ for $a,b\geqslant0$.
Dividing by $|x-y|^\eta$ and taking the supremum,
$$\sup_{x\neq y}\frac{|w''(x)-w''(y)|}{|x-y|^{\eta}}\leqslant
2(2C_{max})^{\eta}A^{1-\eta}.$$
Moreover $A=A^{1-\eta}A^{\eta}\leqslant(2C_{max})^{\eta}A^{1-\eta}$, so
$$\|w\|_{C^{2,\eta}(\overline\Omega)}\leqslant 3(2C_{max})^{\eta}A^{1-\eta}\lesssim A^{1-\eta},$$
the constant being bounded by $3\max\{1,2C_{max}\}$, hence independent of $\eta$.

By Theorem~\ref{5mainres}, $A\lesssim_\theta\varepsilon^{\theta}$ for
$\theta\in(0,\frac\kappa2)$, where
$\varepsilon=\|\Phi(c_1)-\Phi(c_2)\|_{\ell^\infty_2\times\ell^\infty_0}$. Raising this to
the power $1-\eta\in(0,1)$ and combining with the previous display gives $\|w\|_{C^{2,\eta}(\overline\Omega)}\lesssim_\theta\varepsilon^{\theta(1-\eta)}$. The case $\eta=0$ is Theorem~\ref{5mainres} itself.
