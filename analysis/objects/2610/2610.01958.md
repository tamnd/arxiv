<!-- definition-1: body_md -->
A measurable map $u:X\to Y$ belongs to $L^p(X;Y)$, relative to the chosen
base point $y_0$, if
\begin{equation*}
 M_p(u):=\int_X d_Y(u(x),y_0)^p\,\,d\mu(x)<\infty.
\end{equation*}
For $u,v\in L^p(X;Y)$ we set
\begin{equation*}
 d_p(u,v):=
 \left(\int_X d_Y(u(x),v(x))^p\,\,d\mu(x)\right)^{1/p}.
\end{equation*}
<!-- end definition-1 -->
<!-- lemma-1: body_md -->
\label{lem:truncation}
For every $u\in L^p(X;Y)$ there exists a sequence $u_j\in L^p(X;Y)$ such
that
\begin{enumerate}[label=\textup{(\roman*)}]
\item $u_j=y_0$ outside a compact set $K_j\subset X$;
\item $d_Y(u_j(x),y_0)\le j$ for every $x\in X$;
\item $d_p(u_j,u)\to0$.
\end{enumerate}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
The finite Borel measure
$d_Y(u(x),y_0)^p\mu$ is Radon.  Hence for every $j\in\mathbb{N}$ one can choose a compact
$K_j\subset X$ such that
\[
 \int_{X\setminus K_j}d_Y(u(x),y_0)^p\,\,d\mu<2^{-j}.
\]
Define
\[
 u_j(x):=
 \begin{cases}
  u(x),&x\in K_j\text{ and }d_Y(u(x),y_0)\le j,\\
  y_0,&\text{otherwise}.
 \end{cases}
\]
Then (i) and (ii) are immediate.  Moreover,
\begin{align*}
 d_p(u_j,u)^p
 &=\int_{\{u_j\ne u\}}d_Y(u(x),y_0)^p\,\,d\mu(x)\\
 &\le
 \int_{X\setminus K_j}d_Y(u(x),y_0)^p\,\,d\mu(x)
 +\int_{\{d_Y(u(x),y_0)>j\}}d_Y(u(x),y_0)^p\,\,d\mu(x).
\end{align*}
The first term tends to zero by construction and the second tends to zero by
dominated convergence.  Hence (iii) follows.
<!-- end proof-1 -->
<!-- definition-2: body_md -->
\label{def:bounded-marginal-coupling}
Let $a,b\in L^\infty_+(X,\mu)$. A positive Borel measure $\Pi$ on
$X\times X$ is called an \emph{$(a,b)$-marginal coupling} of $\mu$ if
\begin{equation*}
 (\pi_1)_\#\Pi=a\mu,
 \qquad
 (\pi_2)_\#\Pi=b\mu,
\end{equation*}
where $\pi_1(x,y)=x$ and $\pi_2(x,y)=y$. Equivalently, for every
nonnegative Borel function $\varphi$,
\[
 \int_{X\times X}\varphi(x)\,\,d\Pi(x,y)
 =\int_X a\varphi\,\,d\mu,
 \qquad
 \int_{X\times X}\varphi(y)\,\,d\Pi(x,y)
 =\int_X b\varphi\,\,d\mu.
\]
<!-- end definition-2 -->
<!-- definition-3: body_md -->
\label{def:escape-rectangles}
Let $(\Pi_\alpha)_{\alpha}$ be a family of positive Borel measures on $X\times X$ indexed by a directed set\footnote{A partially ordered set $(I,\preceq)$ is called a \emph{directed set} if, for every
$i,j\in I$, there exists $k\in I$ such that
\[
i\preceq k
\qquad\text{and}\qquad
j\preceq k.
\]}. We say that $(\Pi_\alpha)_{\alpha}$ \emph{escapes from bounded rectangles} if, for every bounded Borel
set $K\subset X$,
\begin{equation}\label{eq:escape-rectangles}
 \Pi_\alpha(K\times K)\longrightarrow0,
\end{equation}
where the limit is understood along the directed set.
<!-- end definition-3 -->
<!-- remark-1: body_md -->
\label{compact}
Since $X$ is proper, it is equivalent to test \eqref{eq:escape-rectangles}
on compact sets. Indeed, if $K\subset X$ is bounded and Borel, then
$\overline K$ is compact and
\[
0\leq \Pi_{\alpha}(K\times K)\leq \Pi_{\alpha}(\overline{K}\times \overline{K}).
\]
Thus, if $\Pi_{\alpha}(C\times C)\to 0$ for every compact $C\subset X$,
then $\Pi_{\alpha}(K\times K)\to 0$.  The converse is immediate because
every compact subset of a metric space is bounded and Borel.

<!-- lemma-2: body_md -->
\label{lem:coupling-stability}
Let $\Pi$ be an $(a,b)$-marginal coupling of $\mu$. Then for every
$u,v\in L^p(X;Y)$,
\begin{equation*}
 \lvert J_\Pi(u)^{1/p}-J_\Pi(v)^{1/p}\rvert
 \le \bigl(\|a\|_\infty^{1/p}+\|b\|_\infty^{1/p}\bigr)d_p(u,v).
\end{equation*}

<!-- proof-2: body_md -->
The reverse triangle inequality in 
$L^p(X\times X,\Pi)$ gives
\begin{align*}
 &\lvert J_\Pi(u)^{1/p}-J_\Pi(v)^{1/p}\rvert\\
 &\qquad\le
 \left(
 \int_{X\times X}
 \lvert d_Y(u(x),u(y))-d_Y(v(x),v(y))\rvert^p\,\,d\Pi
 \right)^{1/p}.
\end{align*}
By the metric triangle inequality,
\[
 \lvert d_Y(u(x),u(y))-d_Y(v(x),v(y))\rvert
 \le d_Y(u(x),v(x))+d_Y(u(y),v(y)).
\]
Minkowski's inequality and the marginal identities imply
\begin{align*}
 &\left(
 \int_{X\times X}
 \lvert d_Y(u(x),u(y))-d_Y(v(x),v(y))\rvert^p\,\,d\Pi
 \right)^{1/p}\\
 &\quad\le
 \left(\int d_Y(u(x),v(x))^p\,\,d\Pi\right)^{1/p}
 +\left(\int d_Y(u(y),v(y))^p\,\,d\Pi\right)^{1/p}\\
 &\quad\le
 \|a\|_\infty^{1/p}d_p(u,v)
 +\|b\|_\infty^{1/p}d_p(u,v).
\end{align*}
This concludes the proof.

<!-- proposition-1: body_md -->
\label{prop:compact-decoupling}
Let $\Pi$ be an $(a,b)$-marginal coupling and suppose that $u:X\to Y$
satisfies $u=y_0$ outside a bounded Borel set $K$ and
$d_Y(u(x),y_0)\le L$ for every $x\in X$. Then
\begin{equation}\label{eq:compact-decomposition}
 J_\Pi(u)=\int_X(a+b)d_Y(u,y_0)^p\,\,d\mu+R_\Pi(u;K),
\end{equation}
where
\begin{equation}\label{eq:R-compact}
 R_\Pi(u;K)=
 \int_{K\times K}
 \Bigl[d_Y(u(x),u(y))^p
 -d_Y(u(x),y_0)^p-d_Y(u(y),y_0)^p\Bigr]\,\,d\Pi
\end{equation}
and
\begin{equation}\label{eq:R-bound}
 \lvert R_\Pi(u;K)\rvert\le 2^{p+1}L^p\Pi(K\times K).
\end{equation}
<!-- end proposition-1 -->
<!-- proof-3: body_md -->
Since $u=y_0$ on $K^c$, we get
\[
 d_Y(u(x),u(y))^p-d_Y(u(x),y_0)^p-d_Y(u(y),y_0)^p=0
 \quad\text{for }(x,y)\notin K\times K.
\]
Hence
\begin{align*}
 J_\Pi(u)
 &=\int_{X\times X}
 \bigl(d_Y(u(x),y_0)^p+d_Y(u(y),y_0)^p\bigr)\,\,d\Pi\\
 &\quad+\int_{K\times K}
 \bigl[d_Y(u(x),u(y))^p-d_Y(u(x),y_0)^p-d_Y(u(y),y_0)^p\bigr] \,\,d\Pi.
\end{align*}
The first integral equals
$\int_X(a+b)d_Y(u,y_0)^p\,\,d\mu$ by the marginal identities. On
$K\times K$, the triangle inequality gives
\[
 d_Y(u(x),u(y))^p\le2^pL^p.
\]
Thus the absolute value of the bracket in \eqref{eq:R-compact} is bounded
by $2^{p+1}L^p$, which gives \eqref{eq:R-bound}.

<!-- theorem-1: body_md -->
\label{thm:abstract-decoupling}
Assume that $Y$ contains at least two points. Let
$a_\alpha,b_\alpha,a_\infty,b_\infty\in L^\infty_+(X,\mu)$, indexed by a
directed set, and suppose that
\begin{equation}\label{eq:marginal-assumptions}
 \sup_\alpha\bigl(\|a_\alpha\|_\infty+\|b_\alpha\|_\infty\bigr)<\infty,
 \qquad
 a_\alpha\stackrel{*}{\rightharpoonup}a_\infty,
 \qquad
 b_\alpha\stackrel{*}{\rightharpoonup}b_\infty
 \quad\text{in }L^\infty(X,\mu).
\end{equation}
Here weak-* convergence means convergence of the integrals against every
function in $L^1(X,\mu)$.
Let $(\Pi_\alpha)_\alpha$ be a family of
$(a_\alpha,b_\alpha)$-marginal couplings. Then the following statements are
equivalent:
\begin{enumerate}[label=\textup{(\roman*)}]
\item \label{item:escape-iff}
$\Pi_\alpha$ escapes from bounded rectangles.
\item \label{item:allmaps-iff}
For every $u\in L^p(X;Y)$,
\begin{equation}\label{eq:abstract-decoupling}
 J_{\Pi_\alpha}(u)
 \longrightarrow
 \int_X(a_\infty+b_\infty)d_Y(u,y_0)^p\,\,d\mu.
\end{equation}
\item \label{item:twovalued-iff}
Fix $y_1\in Y\setminus\{y_0\}$.  For every bounded Borel set
$K\subset X$, the map
\[
 u_K(x):=\begin{cases}y_1,&x\in K,\\y_0,&x\notin K,\end{cases}
\]
verifies \eqref{eq:abstract-decoupling}.
\end{enumerate}
<!-- end theorem-1 -->
<!-- proof-4: body_md -->
We first prove \ref{item:escape-iff}$\Rightarrow$\ref{item:allmaps-iff}.
Let $u_j$ be the truncations from Lemma~\ref{lem:truncation}.  For fixed $j\in \mathbb{N}$,
$u_j$ equals $y_0$ outside a compact set $K_j$ and $d_Y(u_j(x),y_0)\leq j$ on $X$.  By
Proposition~\ref{prop:compact-decoupling},
\[
 J_{\Pi_\alpha}(u_j)
 =\int_X(a_\alpha+b_\alpha)d_Y(u_j,y_0)^p\,\,d\mu+R_{\alpha,j},
 \qquad
 |R_{\alpha,j}|\le C_{p,j}\Pi_\alpha(K_j\times K_j)\to0.
\]
Here $C_{p,j}:=2^{p+1}j^p$, and the convergence follows from
Remark~\ref{compact}.
Since $d_Y(u_j,y_0)^p\in L^1(X,\mu)$, weak-* convergence gives
\begin{equation*}
 J_{\Pi_\alpha}(u_j)
 \longrightarrow
 \int_X(a_\infty+b_\infty)d_Y(u_j,y_0)^p\,\,d\mu.
\end{equation*}
By Lemma~\ref{lem:coupling-stability} and
\eqref{eq:marginal-assumptions}, there is a constant $C>0$ independent of
$\alpha$ and $j$ such that
\begin{equation*}
 |J_{\Pi_\alpha}(u)^{1/p}-J_{\Pi_\alpha}(u_j)^{1/p}|
 \le C d_p(u,u_j).
\end{equation*}
Set
\[
 J_\infty(v):=\int_X(a_\infty+b_\infty)d_Y(v,y_0)^p\,\,d\mu.
\]
The reverse triangle inequality in
$L^p(X,(a_\infty+b_\infty)\mu;Y)$ gives
\[
 |J_\infty(u)^{1/p}-J_\infty(u_j)^{1/p}|
 \le
 \|a_\infty+b_\infty\|_\infty^{1/p}d_p(u,u_j).
\]
Now
\begin{align*}
&|J_{\Pi_\alpha}(u)^{1/p}-J_\infty(u)^{1/p}|\\
&\leq |J_{\Pi_\alpha}(u)^{1/p}-J_{\Pi_\alpha}(u_j)^{1/p}|\\
&\quad+|J_\infty(u_j)^{1/p}-J_\infty(u)^{1/p}|\\
&\quad+|J_{\Pi_\alpha}(u_j)^{1/p}-J_\infty(u_j)^{1/p}|.
\end{align*}
Taking first the limit in $\alpha$ and then letting $j\to\infty$ proves
\[
J_{\Pi_\alpha}(u)^{1/p}\longrightarrow J_\infty(u)^{1/p}.
\]
Thus \eqref{eq:abstract-decoupling} holds.

The implication \ref{item:allmaps-iff}$\Rightarrow$\ref{item:twovalued-iff}
is immediate because $\mu(K)<\infty$ for bounded $K$, hence
$u_K\in L^p(X;Y)$.

It remains to prove
\ref{item:twovalued-iff}$\Rightarrow$\ref{item:escape-iff}. Set
\[
c:=d_Y(y_0,y_1)>0.
\]
By the definition of $u_K$,
\begin{align*}
J_{\Pi_\alpha}(u_K)
&=c^p\bigl[
   \Pi_\alpha(K\times K^c)
   +\Pi_\alpha(K^c\times K)
  \bigr]\\
&=c^p\bigl[
   \Pi_\alpha(K\times X)
   +\Pi_\alpha(X\times K)
   -2\Pi_\alpha(K\times K)
  \bigr]\\
&=c^p\bigl[
   \int_K(a_\alpha+b_\alpha)\,\,d\mu
   -2\Pi_\alpha(K\times K)
  \bigr].
\end{align*}
On the other hand,
\[
M_p(u_K)=c^p\mu(K).
\]
Therefore, assumption~$(iii)$ and weak-* convergence, tested against
$\mathbf 1_K\in L^1(X,\mu)$, give
\[
c^p\bigl[
   \int_K(a_\alpha+b_\alpha)\,\,d\mu
   -2\Pi_\alpha(K\times K)
  \bigr]
\longrightarrow
c^p\int_K(a_\infty+b_\infty)\,\,d\mu.
\]
It follows that
\[
2c^p\Pi_\alpha(K\times K)
=
c^p\int_K(a_\alpha+b_\alpha)\,\,d\mu-J_{\Pi_\alpha}(u_K)
\longrightarrow 0.
\]
Since $c>0$, we conclude that
\[
\Pi_\alpha(K\times K)\longrightarrow0,
\]
and hence $(\Pi_\alpha)_\alpha$ escapes from bounded rectangles.

\label{def:metric-tail-law}
Let $u:X\to Y$ be Borel measurable with $u|_\Omega\in L^p(\Omega;Y)$ and
$\lambda_{\varepsilon,x}\in\mathcal P_p(Y)$ for almost every $x\in \Omega$.  A measure
$\lambda_\infty\in\mathcal P_p(Y)$ is called an \emph{$L^p$ target tail law} for the family
$K_\varepsilon$ if
\begin{equation}\label{eq:integrated-Wp-tail}
 \int_\Omega
 W_p(\lambda_{\varepsilon,x},\lambda_\infty)^p
 \,\,d\mu(x)\longrightarrow0\quad \mbox{as}\ \varepsilon \to 0^+
\end{equation}
If, in addition,                                                            
            
\begin{equation*}
 \operatorname*{ess\,sup}_{\substack{x\in\Omega\\m_\varepsilon(x)>0}}
 \frac{K_\varepsilon(x,B)}{m_\varepsilon(x)}
 \longrightarrow0 \quad \mbox{as}\ \varepsilon \to 0^+
 \qquad \forall\text{ bounded Borel }B\subset X,
\end{equation*}
we say that $\lambda_\infty$ is a \emph{target law \emph{at infinity}}.

<!-- theorem-2: body_md -->
\label{thm:metric-tail-law}
Let $u:X\to Y$ be Borel measurable with $u|_\Omega\in L^p(\Omega;Y)$.  Assume
that $\lambda_{\varepsilon,x}\in\mathcal P_p(Y)$ a.e. $x\in \Omega$, that
\eqref{eq:integrated-profile-tail} holds for some
$\lambda_\infty\in\mathcal P_p(Y)$, and that
\begin{equation*}
 \sup_\varepsilon\|m_\varepsilon\|_{L^\infty(\Omega)}<\infty,
 \qquad
 m_\varepsilon\longrightarrow m
 \quad\text{in }L^\infty(\Omega).
\end{equation*}
Then
\begin{equation*}
 \begin{aligned}
 &\lim_{\varepsilon\to0^+}
 \int_\Omega\int_Xd_Y(u(x),u(y))^p
 \,K_\varepsilon(x,\,d y)\,\,d\mu(x)\\
 &\qquad=
 \int_\Omega m(x)
 \int_Yd_Y(u(x),z)^p\,\,d\lambda_\infty(z)
 \,\,d\mu(x).
 \end{aligned}
\end{equation*}
<!-- end theorem-2 -->
<!-- proof-6: body_md -->
For a.e. $x\in\Omega$, the maps $a\mapsto P_{\varepsilon,x}(a)$ and
$a\mapsto P_\infty(a)$ are $1$-Lipschitz.  Indeed, this follows from the
reverse triangle inequality in $L^p$ applied to the functions
$z\mapsto d_Y(a,z)$ and $z\mapsto d_Y(b,z)$.  Let $(u_j)_j$ be a sequence
of finite-valued Borel maps such that
\begin{equation}\label{eq:simple-map-approximation}
 \int_\Omega d_Y(u_j(x),u(x))^p\,\,d\mu(x)\longrightarrow0.
\end{equation}
Such a sequence exists because $Y$ is separable and
$u|_\Omega\in L^p(\Omega;Y)$.

If $u_j$ takes the values $a_{j,1},\dots,a_{j,N_j}$, then
\eqref{eq:integrated-profile-tail}, applied to this finite family, gives
\[
 P_{\varepsilon,\cdot}(u_j(\cdot))
 \longrightarrow P_\infty(u_j(\cdot))
 \quad\text{in }L^p(\Omega).
\]
The Lipschitz estimate therefore yields
\begin{align*}
 &\|P_{\varepsilon,\cdot}(u(\cdot))-P_\infty(u(\cdot))\|_{L^p(\Omega)}\\
 &\quad\le
 2\|d_Y(u,u_j)\|_{L^p(\Omega)}
 +\|P_{\varepsilon,\cdot}(u_j(\cdot))
       -P_\infty(u_j(\cdot))\|_{L^p(\Omega)}.
\end{align*}
First letting $\varepsilon\to0$ and then $j\to\infty$ proves
\begin{equation}\label{eq:Aeps-Ainf-Lp}
 A_\varepsilon:=P_{\varepsilon,\cdot}(u(\cdot))
 \longrightarrow
 A_\infty:=P_\infty(u(\cdot))
 \quad\text{in }L^p(\Omega).
\end{equation}
Moreover,
\[
 A_\infty(x)
 \le d_Y(u(x),y_0)+P_\infty(y_0),
\]
so $A_\infty\in L^p(\Omega)$ and
$\sup_\varepsilon\|A_\varepsilon\|_{L^p(\Omega)}<\infty$ for small
$\varepsilon$.

We claim that
\begin{equation*} 
 A_\varepsilon^p\longrightarrow A_\infty^p
 \quad\text{in }L^1(\Omega).
\end{equation*}
For $p=1$ this is exactly \eqref{eq:Aeps-Ainf-Lp}.  If $p>1$, then using
$|a^p-b^p|\le p|a-b|(a^{p-1}+b^{p-1})$ and the H\"older inequality we get
\begin{align*}
 \|A_\varepsilon^p-A_\infty^p\|_{L^1}
 &\le p\|A_\varepsilon-A_\infty\|_{L^p}
 \bigl(\|A_\varepsilon\|_{L^p}^{p-1}
       +\|A_\infty\|_{L^p}^{p-1}\bigr)\longrightarrow0.
\end{align*}
We claim that
\begin{align}\label{eq:kernel-profile-identity}
 \int_Xd_Y(u(x),u(y))^pK_\varepsilon(x,\,d y)
 =m_\varepsilon(x)A_\varepsilon(x)^p.
 \end{align}
Indeed, if $m_{\varepsilon}(x)=0$ then $K_\varepsilon(x,\cdot)=0$ and the equality trivially follows. If $m_{\varepsilon}(x)>0$ then
\begin{align*}
\int_Xd_Y(u(x),u(y))^pK_\varepsilon(x,\,d y)&=m_{\varepsilon}(x)\int_Xd_Y(u(x),u(y))^p\frac{K_\varepsilon(x,\,d y)}{m_{\varepsilon}(x)}\\
&=m_{\varepsilon}(x)\int_Y d_Y(u(x),z)^p\,\,d\lambda_{\varepsilon,x}(z)\\
&=m_{\varepsilon}(x)A_{\varepsilon}(x)^p.
\end{align*}
Consequently,
\begin{align*}
 &\left|\int_\Omega m_\varepsilon A_\varepsilon^p\,\,d\mu
 -\int_\Omega mA_\infty^p\,\,d\mu\right|\\
 &\qquad\le
 \|m_\varepsilon\|_\infty
 \|A_\varepsilon^p-A_\infty^p\|_{L^1}
 +\|m_\varepsilon-m\|_\infty
 \|A_\infty^p\|_{L^1}\longrightarrow0.
\end{align*}
Together with \eqref{eq:kernel-profile-identity}, this concludes the proof.

which tends to zero.  In particular, the condition is automatic for
globally defined fractional Sobolev maps
$u\in W^{s_0,p}(\mathbb R^d;Y)$, understood through the metric Gagliardo
energy.

<!-- lemma-4: body_md -->
\label{lem:far-angular-equidistribution}
For $\theta\in \mathbb{S}^{d-1}$ set
\[
 w_{s,R,x}(\theta)
 :=s\int_R^\infty
 \frac{r^{d-1}}{|r\theta-x|^{d+sp}}\,\,d r.
\]
Then, uniformly for $x\in\Omega$ and $\theta\in \mathbb{S}^{d-1}$,
\begin{equation}\label{eq:w-uniform}
 w_{s,R,x}(\theta)=\frac{R^{-sp}}p+O(s)
 \qquad(s\downarrow0).
\end{equation}
Consequently
\begin{equation}\label{eq:m-far-uniform}
 m_{s,R}\longrightarrow\frac{\sigma_{d-1}}p
 \quad\text{in }L^\infty(\Omega),
\end{equation}
and the normalized angular densities
\[
 g_{s,R,x}(\theta):=\frac{w_{s,R,x}(\theta)}{m_{s,R}(x)}
\]
satisfy
\begin{equation}\label{eq:g-angular-uniform}
 \sup_{x\in\Omega,\,\theta\in \mathbb{S}^{d-1}}
 \left|g_{s,R,x}(\theta)-\frac1{\sigma_{d-1}}\right|
 \longrightarrow0.
\end{equation}
<!-- end lemma-4 -->
<!-- proof-10: body_md -->
Since $R>2M_\Omega$, for $r\ge R$ we have $|x|/r\le1/2$.  For
$0<s\le1$ the functions
$a\mapsto|\theta-a|^{-d-sp}$ have gradients uniformly bounded on
$|a|\le1/2$, uniformly also in $\theta\in \mathbb{S}^{d-1}$.  Hence, by the mean value theorem applied to the function $a\to |a-\theta|^{-d-sp}$ along the segment joining $0$ to $x/r$, we get
\[
 \left||\theta-x/r|^{-d-sp}-1\right|
 \le C\frac{|x|}{r}.
\]
Therefore,
\begin{align*}
 \left|w_{s,R,x}(\theta)-s\int_R^\infty r^{-1-sp}\,\,d r\right|
 &\le C M_\Omega s\int_R^\infty r^{-2-sp}\,\,d r\\
 &\le C_R s.
\end{align*}
and \eqref{eq:w-uniform} follows since
\[
 s\int_R^\infty r^{-1-sp}\,\,d r=\frac{R^{-sp}}p.
\]
Integrating in $\theta$ and using the coarea formula proves
\eqref{eq:m-far-uniform}.  Dividing \eqref{eq:w-uniform} by the mass, which
is bounded away from zero for small $s$, gives
\eqref{eq:g-angular-uniform}.

<!-- lemma-5: body_md -->
\label{lem:directional-Wp-tail}
Let $u\in L^p_{\mathrm{loc}}(\mathbb R^d;Y)$ satisfy
\eqref{eq:uniform-directional-tail}.  Let $\lambda_{s,R,x}$ be the
normalized target law associated with \eqref{eq:far-cutoff-kernel}.  Then
$\lambda_{s,R,x}\in\mathcal P_p(Y)$ and, for every $a\in Y$,
\begin{equation}\label{eq:directional-Wp-uniform}
 \sup_{x\in\Omega}
 \left|
 \left(\int_Yd_Y(a,z)^p\,\,d\lambda_{s,R,x}(z)\right)^{1/p}
 -\left(\int_Yd_Y(a,z)^p\,\,d\lambda_\infty(z)\right)^{1/p}
 \right|
 \longrightarrow0.
\end{equation}
<!-- end lemma-5 -->
<!-- proof-11: body_md -->
For $a\in Y$ set
\[
 N_{s,R,x}(a)
 :=s\int_{|y|>R}
 \frac{d_Y(a,u(y))^p}{|x-y|^{d+sp}}\,\,d y.
\]
Since $R>1$, local $L^p$ integrability gives
\[
 s\int_{1<|y|\le R}
 \frac{d_Y(a,u(y))^p}{|y|^{d+sp}}\,\,d y\longrightarrow0.
\]
Consequently, \eqref{eq:uniform-directional-tail} implies
\begin{equation}\label{eq:origin-profile-cutoff}
 N_{s,R,0}(a)
 \longrightarrow
 L(a):=\frac1p\int_{\mathbb S^{d-1}}
 d_Y(a,u_\infty(\theta))^p
 \,\,d\mathcal H^{d-1}(\theta).
\end{equation}

We next show that translating the centre by a bounded vector does not alter
this limit.  Fix $A>R$.  On $R<|y|\le A$, both $|y|$ and $|x-y|$ are
bounded away from zero, uniformly for $x\in\Omega$.  Hence local
$L^p$ integrability gives
\begin{equation}\label{eq:A-normalized-vanish}
 \sup_{x\in\Omega}s\int_{R<|y|\le A}d_Y(a,u(y))^p
 \left||x-y|^{-d-sp}-|y|^{-d-sp}\right|\,\,d y
 \longrightarrow0.
\end{equation}
For $|y|>A>2M_\Omega$, the mean value theorem yields, uniformly for
$0<s\le1$,
\[
 \left||x-y|^{-d-sp}-|y|^{-d-sp}\right|
 \le C\frac{M_\Omega}{|y|}|y|^{-d-sp}.
\]
By \eqref{eq:origin-profile-cutoff}, there exists $s_a\in(0,1)$ such that
$\sup_{0<s<s_a}N_{s,R,0}(a)<\infty$.  It follows that
\begin{equation}\label{eq:B-normalized-estimate}
 \limsup_{s\downarrow0}\sup_{x\in\Omega}
 |N_{s,R,x}(a)-N_{s,R,0}(a)|
 \le \frac{C M_\Omega}{A}
 \sup_{0<s<s_a}N_{s,R,0}(a).
\end{equation}
Letting $A\to\infty$ and using \eqref{eq:origin-profile-cutoff}, we obtain
\begin{equation}\label{eq:shifted-profile-uniform}
 \sup_{x\in\Omega}|N_{s,R,x}(a)-L(a)|\longrightarrow0.
\end{equation}
In particular, the case $a=y_0$ shows that
$\lambda_{s,R,x}\in\mathcal P_p(Y)$ for small $s$.  Finally,
\[
 \int_Yd_Y(a,z)^p\,\,d\lambda_{s,R,x}(z)
 =\frac{N_{s,R,x}(a)}{m_{s,R}(x)}.
\]
Combining \eqref{eq:shifted-profile-uniform} with
\eqref{eq:m-far-uniform}, and observing that
\[
 \int_Yd_Y(a,z)^p\,\,d\lambda_\infty(z)
 =\frac1{\sigma_{d-1}}
 \int_{\mathbb S^{d-1}}d_Y(a,u_\infty(\theta))^p
 \,\,d\mathcal H^{d-1}(\theta),
\]
proves \eqref{eq:directional-Wp-uniform}.

<!-- theorem-4: body_md -->
\label{thm:directional-metric-MS}
Let $1\le p<\infty$, let $u\in L^p_{\mathrm{loc}}(\mathbb R^d;Y)$, and let
$u_\infty:\mathbb S^{d-1}\to Y$ be Borel with
$d_Y(u_\infty(\cdot),y_0)\in L^p(\mathbb S^{d-1})$.  Assume
$\mathrm{EP}_p$, namely \eqref{eq:uniform-directional-tail}.  Suppose that
for some $s_0\in(0,1)$,
\begin{equation*}
 \mathcal E_{s_0,p}^\Omega(u)<\infty.
\end{equation*}
Then
\begin{equation}\label{eq:directional-metric-MS}
 \lim_{s\downarrow0}\frac{s}{2}\mathcal E_{s,p}^\Omega(u)
 =\frac1p\int_\Omega\int_{\mathbb{S}^{d-1}}
 d_Y(u(x),u_\infty(\theta))^p
 \,\,d\mathcal H^{d-1}(\theta)\,\,d x.
\end{equation}
<!-- end theorem-4 -->
<!-- proof-12: body_md -->
Fix $R>\max\{1,2M_\Omega\}$.  By symmetry,
\begin{align*}
 \frac{s}{2}\mathcal E_{s,p}^\Omega(u)
 &=\frac{s}{2}\iint_{\Omega\times\Omega}
 \frac{d_Y(u(x),u(y))^p}{|x-y|^{d+sp}}\,\,d y\,\,d x\quad (=:\mathcal{A}_s)\\
 &\quad+s\int_\Omega\int_{\Omega^c\cap B_R}
 \frac{d_Y(u(x),u(y))^p}{|x-y|^{d+sp}}\,\,d y\,\,d x\quad (=:\mathcal{B}_s)\\
 &\quad+s\int_\Omega\int_{\Omega^c\cap \{|y|>R\}}
 \frac{d_Y(u(x),u(y))^p}{|x-y|^{d+sp}}\,\,d y\,\,d x.\quad (=:\mathcal{C}_s)
\end{align*}
Now, splitting at $|x-y|=1$ and using $s<s_0$ on the short-range part,
\begin{align*}
\mathcal A_s
&\le \frac{s}{2}\mathcal E_{s_0,p}^\Omega(u)
 +2^{p-1}s\,|\Omega|
 \int_\Omega d_Y(u(x),y_0)^p\,\,d x
 \longrightarrow0.
\end{align*}
Similarly, with $D_R:=\Omega^c\cap B_R$,
\begin{align*}
\mathcal B_s
&\le s\mathcal E_{s_0,p}^\Omega(u)
 +2^{p-1}s\,|\Omega|
 \int_{D_R}d_Y(u(y),y_0)^p\,\,d y\\
&\quad
 +2^{p-1}s\,|D_R|
 \int_\Omega d_Y(u(x),y_0)^p\,\,d x
 \longrightarrow0.
\end{align*}
For the last term use the kernel \eqref{eq:far-cutoff-kernel}.  By
Lemma~\ref{lem:far-angular-equidistribution},
$m_{s,R}\to\sigma_{d-1}/p$ in $L^\infty(\Omega)$, and, in particular,
\[
\sup_{0<s<s_*}\|m_{s,R}\|_{L^\infty(\Omega)}<\infty
\]
for some $s_*>0$.  Lemma~\ref{lem:directional-Wp-tail} gives the
$p$-energy-profile convergence \eqref{eq:integrated-profile-tail}, in fact
uniformly in $x$.  Therefore Theorem~\ref{thm:metric-tail-law}
gives\footnote{Since $R>2M_\Omega$, we have
$\{|y|>R\}\subset\Omega^c$ and, up to the null sphere $\partial B_R$,
$\Omega^c=(\Omega^c\cap B_R)\cup\{|y|>R\}$.}
\begin{align*}
 \lim_{s\downarrow 0} \mathcal C_s&=\lim_{s\downarrow0}s\int_\Omega\int_{\{|y|>R\}}
 \frac{d_Y(u(x),u(y))^p}{|x-y|^{d+sp}}\,\,d y\,\,d x\\
 &=\frac{\sigma_{d-1}}p
 \int_\Omega\int_Yd_Y(u(x),z)^p\,\,d\lambda_\infty(z)\,\,d x\\
 &=\frac1p\int_\Omega\int_{\mathbb{S}^{d-1}}
 d_Y(u(x),u_\infty(\theta))^p
 \,\,d\mathcal H^{d-1}(\theta)\,\,d x.
\end{align*}
Combining the three contributions proves \eqref{eq:directional-metric-MS}.

\label{prop:anisotropic-directional-MS}
Let $K\subset\mathbb R^n$ be an origin-symmetric convex body with nonempty
interior, and let $\|\cdot\|_K$ be its Minkowski norm. Set
\[
 S_K:=\{\omega\in\mathbb R^n:\|\omega\|_K=1\},
\]
and let $\sigma_K$ be the polar measure on $S_K$, characterized by
\begin{equation}\label{eq:K-polar-formula}
 \int_{\mathbb R^n} f(z)\,\,d z
 =
 \int_0^\infty\int_{S_K}
 f(r\omega)\,\,d\sigma_K(\omega)\,r^{n-1}\,\,d r
\end{equation}
for every nonnegative Borel function $f$. In particular,
\begin{equation*}
 \sigma_K(S_K)=n|K|.
\end{equation*}

Let $(Y,d_Y,y_0)$ be a Polish pointed metric space,
let $\Omega\subset\mathbb R^n$ be bounded and Borel, and let
$u\in L^p_{\rm loc}(\mathbb R^n;Y)$, where $1\le p<\infty$.
Assume that there exists a Borel map
\[
 u_\infty:S_K\to Y,
 \qquad
 \int_{S_K}d_Y(u_\infty(\omega),y_0)^p
 \,\,d\sigma_K(\omega)<\infty,
\]
such that the anisotropic energy-profile condition
\begin{equation}\label{eq:anisotropic-energy-profile}
 s\int_{\{\|y\|_K>1\}}
 \frac{d_Y(u(y),a)^p}{\|y\|_K^{n+sp}}\,\,d y
 \longrightarrow
 \frac1p\int_{S_K}d_Y(a,u_\infty(\omega))^p\,\,d\sigma_K(\omega)
 \qquad\text{for every }a\in Y
\end{equation}
holds.

Define
\begin{equation*}
 \mathcal E_{s,p}^{\Omega,K}(u)
 :=
 \iint_{Q_\Omega}
 \frac{d_Y(u(x),u(y))^p}
      {\|x-y\|_K^{n+sp}}
 \,\,d y\,\,d x,
 \qquad
 Q_\Omega
 :=
 (\mathbb R^n\times\mathbb R^n)
 \setminus(\Omega^c\times\Omega^c).
\end{equation*}
If
\begin{equation}\label{eq:anisotropic-positive-order}
 \mathcal E_{s_0,p}^{\Omega,K}(u)<\infty
\end{equation}
for some $s_0\in(0,1)$, then
\begin{equation}\label{eq:anisotropic-directional-MS}
 \lim_{s\downarrow0}
 \frac{s}{2}\mathcal E_{s,p}^{\Omega,K}(u)
 =
 \frac1p
 \int_\Omega\int_{S_K}
 d_Y\bigl(u(x),u_\infty(\omega)\bigr)^p
 \,\,d\sigma_K(\omega)\,\,d x.
\end{equation}

Set
\[
 M_{\Omega,K}:=\sup_{x\in\Omega}\|x\|_K
\]
and fix $R>\max\{1,2M_{\Omega,K}\}$. For $x\in\Omega$ consider the far kernel
\[
 K^K_{s,R}(x,\,d y)
 :=
 s\,\mathbf 1_{\{\|y\|_K>R\}}
 \|x-y\|_K^{-n-sp}\,\,d y.
\]
Using the polar formula \eqref{eq:K-polar-formula}, its angular density is
\[
 w^K_{s,R,x}(\omega)
 :=
 s\int_R^\infty
 \frac{r^{n-1}}
      {\|r\omega-x\|_K^{n+sp}}
 \,\,d r.
\]
Since
\[
 \|r\omega-x\|_K
 =
 r\|\omega-x/r\|_K
\]
and
\[
 \left|
 \|\omega-x/r\|_K-1
 \right|
 \le \frac{\|x\|_K}{r}
 \le\frac12,
\]
the function $t\mapsto t^{-n-sp}$ is uniformly Lipschitz on the relevant
compact interval, uniformly for $0<s\le1$. Hence
\[
 \left|
 \|\omega-x/r\|_K^{-n-sp}-1
 \right|
 \le C\frac{\|x\|_K}{r},
\]
and therefore, uniformly for $x\in\Omega$ and $\omega\in S_K$,
\begin{equation}\label{eq:anisotropic-w-asymptotic}
 w^K_{s,R,x}(\omega)
 =
 s\int_R^\infty r^{-1-sp}\,\,d r+O(s)

 \frac{R^{-sp}}p+O(s).
\end{equation}

\[
 m^K_{s,R}(x):=K^K_{s,R}(x,\mathbb R^n),

\begin{equation}\label{eq:anisotropic-far-mass}

\end{equation}

 \frac1{\sigma_K(S_K)}=\frac1{n|K|}.
\]

 \lambda_\infty^K
 :=

 \left(
 \frac{\sigma_K}{n|K|}
 \right).
\end{equation*}
The assumed $p$-integrability of $u_\infty$ gives
$\lambda_\infty^K\in\mathcal P_p(Y)$.

 \,\,d y\,\,d x

 \frac{d_Y(u(x),u(y))^p}{\|x-y\|_K^{n+sp}}
 \,\,d y\,\,d x

\eqref{eq:anisotropic-profile-uniform} give
\begin{align*}

 \\

\label{prop:Bernoulli-tail-perimeter}
Let $\Omega\subset\mathbb R^n$ be bounded and open and let
$E\subset\mathbb R^n$ be measurable.  For $0<s<1$ write
\[
 L_s(A,B)
 :=
 \int_A\int_B\frac{1}{|x-y|^{n+s}}\,\,d y\,\,d x
\]
and
\[
 \operatorname{Per}_s(E;\Omega)
 :=
 L_s(E\cap\Omega,E^c\cap\Omega)
 +L_s(E\cap\Omega,E^c\cap\Omega^c)
 +L_s(E\cap\Omega^c,E^c\cap\Omega).
\]
Assume that
\[
 \operatorname{Per}_{s_0}(E;\Omega)<\infty
\]
for some $s_0\in(0,1)$ and that the limit
\begin{equation}\label{eq:alpha-E-tail}
 \alpha(E)
 :=
 \lim_{s\downarrow0}
 s\int_{E\cap B_1^c}\frac{\,d y}{|y|^{n+s}}
\end{equation}
exists.  Set
\[
 \sigma_{n-1}:=\mathcal H^{n-1}(S^{n-1}),
 \qquad
 \vartheta_E:=\frac{\alpha(E)}{\sigma_{n-1}}\in[0,1].
\]

\begin{equation}\label{eq:DFPV-Bernoulli-limit}
 \lim_{s\downarrow0}
 s\,\operatorname{Per}_s(E;\Omega)
 =
 \sigma_{n-1}

 (1-\vartheta_E)|E\cap\Omega|
 +
 \vartheta_E|E^c\cap\Omega|
 \right].
\end{equation}

\begin{equation}\label{eq:Bernoulli-tail-law}
 \lambda_\infty^E
 :=
 (1-\vartheta_E)\delta_0+\vartheta_E\delta_1,
\end{equation}
then
\begin{equation}\label{eq:DFPV-tail-law-form}
 \lim_{s\downarrow0}
 s\,\operatorname{Per}_s(E;\Omega)
 =

 |\mathbf 1_E(x)-z|
 \,\,d\lambda_\infty^E(z)\,\,d x.
\end{equation}

formula of Dipierro--Figalli--Palatucci--Valdinoci

$u\in L^p(X;Y)$ and suppose that

 \beta sp\,t^{-\beta sp-1}\mathbf 1_{(1,\infty)}(t)\,\,d t.

\[

 =1-T^{-\beta sp}\longrightarrow0,

\end{equation}

<!-- proof-21: body_md -->
For $t>0$ define
\[
\,d\Pi_t(x,y)
:=
h_t(x,y)\,\,d\mu(x)\,\,d\mu(y).
\]
The symmetry and sub-Markov property of the heat kernel imply that both
marginals of $\Pi_t$ are $m_t\mu$ and $0\le m_t\le1$. Moreover,

 \quad\text{in }L^\infty(M,\mu).

<!-- lemma-6: body_md -->
\label{lem:Carnot-translation-decoupling}
Let $(\nu_\alpha)_\alpha$ be Borel probability measures on $G$ such that
$\nu_\alpha(K)\to0$ for every compact set $K\subset G$. Then, for every
$u\in L^p(G;Y)$,
\[
 \int_G F_u(h)\,\,d\nu_\alpha(h)
 \longrightarrow 2M_p(u).
\]
<!-- end lemma-6 -->
<!-- proof-25: body_md -->
Associate with $\nu_\alpha$ the coupling $\Pi_\alpha$ defined by
\[
 \int_{G\times G}\Phi(x,y)\,\,d\Pi_\alpha(x,y)
 :=
 \int_G\int_G \Phi(x,xh)\,\,d x\,\,d\nu_\alpha(h)
\]
for every nonnegative Borel function $\Phi$. Since every Carnot group is
unimodular, both marginals of $\Pi_\alpha$ are Haar measure. Moreover, if
$K\subset G$ is bounded, then $\overline K^{-1}\overline K$ is compact and
\[
 \Pi_\alpha(K\times K)
 \le |K|\,\nu_\alpha(\overline K^{-1}\overline K)
 \longrightarrow0.
\]
Thus $\Pi_\alpha$ escapes from bounded rectangles. If $Y$ has at least two
points, Theorem~\ref{thm:abstract-decoupling} applies with
$a_\alpha=b_\alpha=1$; the one-point case is trivial. Since
\[
 J_{\Pi_\alpha}(u)=\int_G F_u(h)\,\,d\nu_\alpha(h),
\]
the conclusion follows.

<!-- theorem-6: body_md -->
\label{thm:Carnot-MS}
Let $1\le p<\infty$ and $u\in L^p(G;Y)$. Assume that, for all sufficiently
small $s>0$,
\[
 \int_{\{N_G(h)<1\}}
 \frac{F_u(h)}
      {N_G(h)^{Q+sp}}
 \,\,d h<\infty
\]
and that
\begin{equation}\label{eq:Carnot-minimal-short}
 \lim_{s\downarrow0}
 s\int_{\{N_G(h)<1\}}
 \frac{F_u(h)}
      {N_G(h)^{Q+sp}}
 \,\,d h
 =0.
\end{equation}
Then
\begin{equation}\label{eq:Carnot-MS}
 \lim_{s\downarrow0}
 s\mathcal E_{s,p}(u)
 =
 \frac{2\sigma_G(S_G)}{p}M_p(u).
\end{equation}

In particular, \eqref{eq:Carnot-MS} holds whenever
\begin{equation}\label{eq:Carnot-positive-order}
 \mathcal E_{\theta,p}(u)<\infty
\end{equation}
for some $\theta>0$.
<!-- end theorem-6 -->
<!-- proof-26: body_md -->
For $s>0$ define
\begin{equation}\label{eq:nu-s-Carnot}
 \,d\nu_s(h)
 :=
 \frac{sp}{\sigma_G(S_G)}
 \frac{\mathbf 1_{\{N_G(h)>1\}}}
      {N_G(h)^{Q+sp}}
 \,\,d h.
\end{equation}
By the polar formula,
\[
 \int_G\,d\nu_s
 =
 \frac{sp}{\sigma_G(S_G)}\sigma_G(S_G)
 \int_1^\infty r^{-1-sp}\,\,d r
 =1.
\]
Thus $\nu_s$ is a probability measure. Moreover, for every $R>1$,
\[
 \nu_s(\{1<N_G\le R\})
 =
 sp\int_1^Rr^{-1-sp}\,\,d r
 =
 1-R^{-sp}
 \longrightarrow0
\]
as $s\downarrow0$. Since every compact subset of $G$ is contained in some
ball, $(\nu_s)_s$ escapes to infinity.

Split \eqref{eq:Carnot-energy-F} into the regions $N_G(h)<1$ and
$N_G(h)>1$. By \eqref{eq:Carnot-minimal-short}, the normalized local
contribution vanishes. For the far part, \eqref{eq:nu-s-Carnot} gives
\[
 s\int_{\{N_G(h)>1\}}
 \frac{F_u(h)}{N_G(h)^{Q+sp}}\,\,d h
 =
 \frac{\sigma_G(S_G)}{p}
 \int_G F_u(h)\,\,d\nu_s(h).
\]
Lemma~\ref{lem:Carnot-translation-decoupling} yields
\[
 \int_G F_u(h)\,\,d\nu_s(h)
 \longrightarrow2M_p(u),
\]
and \eqref{eq:Carnot-MS} follows.

It remains to prove the last assertion. Assume that
\eqref{eq:Carnot-positive-order} holds for some $\theta>0$. For
$0<s<\theta$ and $N_G(h)<1$,
\[
 N_G(h)^{-Q-sp}
 \le
 N_G(h)^{-Q-\theta p}.
\]
Consequently,
\[
 0
 \le
 s\int_{\{N_G(h)<1\}}
 \frac{F_u(h)}{N_G(h)^{Q+sp}}\,\,d h
 \le
 s\mathcal E_{\theta,p}(u)
 \longrightarrow0.
\]
Thus \eqref{eq:Carnot-minimal-short} holds and the first part of the theorem
applies.

\label{cor:Carnot-to-Carnot}
Let $u:G\to H$ satisfy
\[
 \int_G N_H(u(x))^p\,\,d x<\infty.
\]
Assume the short-range condition
\[
 \lim_{s\downarrow0}
 s\int_{\{N_G(h)<1\}}
 \int_G
 \frac{N_H(u(x)^{-1}u(xh))^p}
      {N_G(h)^{Q+sp}}
 \,\,d x\,\,d h
 =0.
\]
Then
\begin{equation}\label{eq:Carnot-to-Carnot-MS}
 \begin{aligned}
 &\lim_{s\downarrow0}
 s\int_G\int_G
 \frac{N_H(u(x)^{-1}u(y))^p}
      {N_G(x^{-1}y)^{Q+sp}}
 \,\,d y\,\,d x
 \\
 &\qquad=
 \frac{2\sigma_G(S_G)}{p}
 \int_G N_H(u(x))^p\,\,d x.
 \end{aligned}
\end{equation}
In particular, the conclusion holds if the same double integral is finite for
one positive order $\theta>0$.

<!-- corollary-10: body_md -->
\label{cor:Carnot-heat-MS}
Let $p_t$ be the heat kernel of a homogeneous sub-Laplacian on $G$. For a
pointed metric space $(Y,d_Y,y_0)$, $1\le p<\infty$, and $u\in L^p(G;Y)$,
set
\begin{equation*}
 J_p^G(t,u)
 :=\int_G\int_G d_Y(u(g),u(h))^p
 p_t(g^{-1}h)\,\,d h\,\,d g
\end{equation*}
and
\begin{equation*}
 \mathscr N_{s,p}^{G,h}(u)^p
 :=\int_0^\infty t^{-sp/2-1}J_p^G(t,u)\,\,d t.
\end{equation*}
If $\mathscr N_{s_0,p}^{G,h}(u)<\infty$ for some $s_0>0$, then
\begin{equation}\label{eq:Carnot-heat-MS}
 \lim_{s\downarrow0}s\mathscr N_{s,p}^{G,h}(u)^p
 =\frac4p\int_Gd_Y(u(g),y_0)^p\,\,d g.
\end{equation}
<!-- end corollary-10 -->
<!-- proof-28: body_md -->
Define
\[
 \,d\Pi_t(g,h):=p_t(g^{-1}h)\,\,d g\,\,d h.
\]
The heat kernel is conservative and symmetric. Haar measure on a Carnot
group is unimodular, so for every $g,h\in G$,
\[
 \int_Gp_t(g^{-1}h)\,\,d h=1,
 \qquad
 \int_Gp_t(g^{-1}h)\,\,d g=1.
\]
Thus both marginals of $\Pi_t$ are Haar measure.

The time-one heat kernel is bounded. Its scaling gives
\[
 \|p_t\|_{L^\infty(G)}
 =t^{-Q/2}\|p_1\|_{L^\infty(G)}.
\]
Consequently, for every bounded Borel set $K\subset G$,
\[
 \Pi_t(K\times K)
 \le |K|^2\|p_t\|_{L^\infty(G)}
 =|K|^2t^{-Q/2}\|p_1\|_{L^\infty(G)}
 \longrightarrow0.
\]
The couplings therefore escape from bounded rectangles, and
Corollary~\ref{cor:transition-decoupling} yields
\[
 J_p^G(t,u)\longrightarrow2M_p(u).
\]
The positive-order assumption implies the short-time condition by
Proposition~\ref{prop:Besov-positive-order}. Theorem~\ref{thm:Abel-semigroup}
with $\beta=1/2$ then gives \eqref{eq:Carnot-heat-MS}.

<!-- remark-12: body_md -->
\label{rem:GT-comparison}
For scalar-valued maps, Corollary~\ref{cor:Carnot-heat-MS} coincides with
the small-order result of Garofalo--Tralli
\cite[Theorem~1.5]{GarofaloTralli2024}.  With their notation,
\[
 \mathcal N_{s,p}^{\mathbb G}(f)^p
 :=\int_0^\infty t^{-sp/2-1}
 \int_{\mathbb G}P_t\bigl(|f-f(g)|^p\bigr)(g)\,\,d g\,\,d t,
\]
and for $1\le p<\infty$ they prove
\[
 \lim_{s\downarrow0}s\,\mathcal N_{s,p}^{\mathbb G}(f)^p
 =\frac4p\|f\|_{L^p(\mathbb G)}^p
\]
for $f$ belonging to one positive heat-Besov class.  A central ingredient of
their proof is an integral decoupling property of the Carnot heat kernel.

In the language of the present paper, that decoupling property says that the
measures
\[
 \,d\Pi_t(g,h)=p_t(g^{-1}h)\,\,d g\,\,d h
\]
escape every bounded rectangle as $t\to\infty$.  Conservation and
self-adjointness give two Haar marginals of mass one.  Hence
Theorem~\ref{thm:abstract-decoupling} yields
$J_p(t,f)\to2\|f\|_p^p$, and the Abel factor associated with
$t^{-sp/2-1}$ is $2/p$, producing the constant $4/p$. The proof above
isolates these two steps in the abstract coupling and Abelian statements.
