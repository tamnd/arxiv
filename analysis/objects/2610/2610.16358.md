<!-- definition-1: body_md -->
\label{def:ex}
{\rm
Let $G$ and $T$ be two graphs and $\mathcal{F}$ be a family of graphs.
\begin{itemize}
\item[(i)] Define $\operatorname{ex}(G, T, \mathcal{F})$ as the maximum number of copies of $T$ in an $\mathcal{F}$-free subgraph of $G$.
\item[(ii)] Define $\operatorname{ex}'(G, T, \mathcal{F})$ as the maximum number of copies of $T$ in an $\mathcal{F}$-free spanning subgraph of $G$.
\end{itemize}
}
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
{\normalfont (\cite{AlSh22})}\label{thm:AlSh22-1}
Let $T$ be a fixed graph and let $\mathcal{F}$ be a family of graphs such that there is a graph $F\in \mathcal{F}$ which is a subgraph of a blow-up of $T$.
Then there is a fixed $\varepsilon=\varepsilon(T, \mathcal{F})>0$ such that $\operatorname{ex}(G, T, \mathcal{F})$ can be calculated in polynomial time within an additive error $n^{v(T)-\varepsilon}$ for any graph $G$ on $n$ vertices.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
{\normalfont (\cite{AlSh22})}\label{thm:AlSh22-2}
Let $k>m\geq 2$ be integers.
Then for every $\varepsilon>0$, approximating $\operatorname{ex}(G, K_m, K_k)$ within an additive error $n^{m-\varepsilon}$ is NP-hard for an input graph $G$ on $n$ vertices.
<!-- end theorem-2 -->
<!-- conjecture-1: body_md -->
{\normalfont (\cite{AlSh22})}\label{conj:AlSh22}
Let $T$ be a graph with $e(T)> 0$, and $\mathcal{F}$ be a nonempty family such that no $F\in \mathcal{F}$ is a subgraph of a blow-up of $T$.
Then for every $\varepsilon>0$, approximating $\operatorname{ex}(G, T, \mathcal{F})$ within an additive error $n^{v(T)-\varepsilon}$ is NP-hard for an input graph $G$ on $n$ vertices.
<!-- end conjecture-1 -->
<!-- theorem-3: body_md -->
\label{thm:bipar}
Let $T$ be a bipartite graph with $e(T)> 0$, $\mathcal{F}$ be a nonempty family such that no $F\in \mathcal{F}$ is a subgraph of a blow-up of $T$, and $\varepsilon>0$.
Then for each of the two parameters $\operatorname{ex}(G, T, \mathcal{F})$ and $\operatorname{ex}'(G, T, \mathcal{F})$, approximation within additive error $n^{v(T)-\varepsilon}$ is NP-hard for an input graph $G$ on $n$ vertices.

<!-- proposition-1: body_md -->
\label{prop:ex}
Let $T$ be a fixed graph with $e(T)>0$, and let $\mathcal{F}$ be a fixed nonempty family of graphs with $e(F)>0$ for every $F\in \mathcal{F}$.
Then the following statements hold.
\begin{itemize}
\item[{\rm (i)}] If $\mathcal{F}$ is finite, then $0\leq \operatorname{ex}(G, T, \mathcal{F})-\operatorname{ex}'(G, T, \mathcal{F})=O_{T,\mathcal{F}}(1)$ for sufficiently large graph $G$.
\item[{\rm (ii)}] If every member of $\mathcal{F}$ has no isolated vertices, then $\operatorname{ex}(G, T, \mathcal{F})=\operatorname{ex}'(G, T, \mathcal{F})$ for every graph $G$.
\end{itemize}
<!-- end proposition-1 -->
<!-- head-1: body_md -->
(i) By the definition, we have $\operatorname{ex}(G, T, \mathcal{F})-\operatorname{ex}'(G, T, \mathcal{F})\geq 0.$
Now let $M=\max_{F\in \mathcal{F}}v(F)$, and we show that $$\operatorname{ex}(G, T, \mathcal{F})-\operatorname{ex}'(G, T, \mathcal{F})\leq \mathcal{N}(K_{M-1}, T)$$ for every graph $G$, which implies the conclusion since $\mathcal{N}(K_{M-1}, T)$ depends only on $T$ and $\mathcal{F}$.
Let $H$ be an $\mathcal{F}$-free subgraph of $G$ with $\mathcal{N}(H,T)=\operatorname{ex}(G, T, \mathcal{F})$.
If $v(H)\leq M-1$, then $\operatorname{ex}(G, T, \mathcal{F})-\operatorname{ex}'(G, T, \mathcal{F})\leq \operatorname{ex}(G, T, \mathcal{F})=\mathcal{N}(H,T)\leq \mathcal{N}(K_{M-1}, T)$.
Now assume $v(H)\geq M$.
Let $H'$ be the spanning subgraph of $G$ obtained from $H$ by adding the missing vertices as isolated vertices.
We claim that $H'$ is also $\mathcal{F}$-free.
Indeed, if $H'$ contains a copy of some $F\in \mathcal{F}$, then every nonisolated vertex of $F$ must be mapped into $H$.
Since $v(H)\geq M\geq v(F)$, the isolated vertices of $F$ can also be mapped to the unused vertices of $H$.
This implies that $H$ contains a copy of $F$, a contradiction.
Thus $\operatorname{ex}'(G, T, \mathcal{F})\geq \mathcal{N}(H',T)\geq \mathcal{N}(H,T)=\operatorname{ex}(G, T, \mathcal{F})$, so $\operatorname{ex}(G, T, \mathcal{F})=\operatorname{ex}'(G, T, \mathcal{F})$ in this case.
The result follows.

(ii) It suffices to show that $\operatorname{ex}(G, T, \mathcal{F})\leq \operatorname{ex}'(G, T, \mathcal{F})$ if every member of $\mathcal{F}$ has no isolated vertices.
Let $H$ be an $\mathcal{F}$-free subgraph of $G$ with $\mathcal{N}(H,T)=\operatorname{ex}(G, T, \mathcal{F})$.
Extend $H$ to a spanning subgraph $H'$ of $G$ by adding the missing vertices as isolated vertices.
Since no $F\in \mathcal{F}$ contains isolated vertices, $H'$ is also $\mathcal{F}$-free.
Thus $\operatorname{ex}'(G, T, \mathcal{F})\geq \mathcal{N}(H',T)\geq \mathcal{N}(H,T)=\operatorname{ex}(G, T, \mathcal{F})$.
This completes the proof.
            
\hfill$\blacksquare$

<!-- theorem-7: body_md -->
\label{thm:transfer}
Let $T$ be a fixed graph with $e(T)>0$, $\mathcal{F}$ be a nonempty family of graphs, and $\varepsilon>0$.
Let $\mathcal{G}$ be a class of input graphs,
and suppose that $z=z(G)$ is a graph parameter that is NP-hard to approximate within an additive error $g(n)$ for $G\in \mathcal{G}$.
Suppose that for every $n$-vertex graph $G\in \mathcal{G}$, one can construct in polynomial time a graph $Z=Z(G)$ so that
$$L+Sz-a \leq \operatorname{ex}'(Z, T, \mathcal{F})\leq \operatorname{ex}(Z, T, \mathcal{F})\leq L+Sz+b,$$
where $L=L(G)\geq 0, S=S(G)>0$ are polynomial-time computable rational numbers, and $a,b$ are nonnegative errors.
If $$a+b+v(Z)^{v(T)-\varepsilon}\leq g(n)S,$$ then it is NP-hard to approximate $\operatorname{ex}(Z, T, \mathcal{F})$ or $\operatorname{ex}'(Z, T, \mathcal{F})$ within an additive error $v(Z)^{v(T)-\varepsilon}$ for $Z\in \{Z(G)\colon\, G\in\mathcal{G}\}$.
<!-- end theorem-7 -->
<!-- head-2: body_md -->
It suffices to consider $\operatorname{ex}(Z, T, \mathcal{F})$, since the proof for $\operatorname{ex}'(Z, T, \mathcal{F})$ is essentially the same.
Suppose that there is a polynomial-time algorithm which returns a number $x$ satisfying $|x-\operatorname{ex}(Z, T, \mathcal{F})|\leq v(Z)^{v(T)-\varepsilon}$.
Let $z'=\frac{x-L}{S}$.
Since $L+Sz-a \leq \operatorname{ex}(Z, T, \mathcal{F})\leq L+Sz+b$ and $a+b+v(Z)^{v(T)-\varepsilon}\leq g(n)S$, we have
\begin{align*}
  |z'-z| = &~\left|\frac{x-L}{S}-z\right| = \frac{1}{S}\left|x-(L+Sz)\right| \leq \frac{1}{S}\big(\left|x-\operatorname{ex}(Z, T, \mathcal{F})\right|+\left|\operatorname{ex}(Z, T, \mathcal{F})-(L+Sz)\right|\big)\\
   \leq &~\frac{1}{S}\left(v(Z)^{v(T)-\varepsilon}+a+b\right) \leq g(n),
\end{align*}
where the first inequality follows from the triangle inequality.
This yields a polynomial-time approximation for $z$ within an additive error $g(n)$, which is assumed to be NP-hard.
Therefore, it is NP-hard to approximate $\operatorname{ex}(Z, T, \mathcal{F})$ within an additive error $v(Z)^{v(T)-\varepsilon}$ for $Z\in \{Z(G)\colon\, G\in\mathcal{G}\}$.
            
\hfill$\blacksquare$

<!-- corollary-2: body_md -->
\label{cor:transfer}
Let $T$ be a fixed graph with $e(T)>0$, $\mathcal{F}$ be a nonempty family of graphs with $k=\min_{F\in \mathcal{F}}\chi(F)\geq 3$, and $\varepsilon>0$.
Let $\mathcal{G}$ be a class of input graphs such that computing $z=\operatorname{cut}_{k-1}(G)$ is NP-hard for inputs $G\in \mathcal{G}$.
Suppose that for every $n$-vertex graph $G\in \mathcal{G}$, one can construct in polynomial time a graph $Z=Z(G)$ so that
$$L+Sz-a \leq \operatorname{ex}'(Z, T, \mathcal{F})\leq \operatorname{ex}(Z, T, \mathcal{F})\leq L+Sz+b,$$
where $L=L(G)\geq 0, S=S(G)>0$ are polynomial-time computable rational numbers, and $a,b$ are nonnegative errors.
If $$a+b+v(Z)^{v(T)-\varepsilon}< \frac{S}{2},$$ then it is NP-hard to approximate $\operatorname{ex}(Z, T, \mathcal{F})$ or $\operatorname{ex}'(Z, T, \mathcal{F})$ within an additive error $v(Z)^{v(T)-\varepsilon}$ for $Z\in \{Z(G)\colon\, G\in\mathcal{G}\}$.
<!-- end corollary-2 -->
<!-- head-3: body_md -->
For each $G\in \mathcal{G}$, define $g(G)\colonequals S^{-1}\left(a+b+v(Z)^{v(T)-\varepsilon}\right)$.
Let $\mathcal{G}(n)\colonequals \{G\in \mathcal{G}\colon\, v(G)=n\}$ and $g(n)\colonequals \max_{G\in \mathcal{G}(n)}g(G)$, with $g(n)\colonequals 0$ if $\mathcal{G}(n)=\emptyset$.
By the assumption, we have $0\leq g(n)<1/2$ for every $n$.
Since $z=\operatorname{cut}_{k-1}(G)$ is an integer, if we can approximate $z$ within an additive error $g(n)<1/2$, then we can determine $z$ exactly by rounding to the nearest integer.
Hence, approximating $z$ within an additive error $g(n)$ is as hard as determining $z$ exactly, and thus NP-hard.
Now the result follows from Theorem~\ref{thm:transfer}.

<!-- lemma-3: body_md -->
\label{le:leaf-hom-1}
For any graph $G$, bipartite graph $Q$ and positive integer $x$, we have
$$\hom(Q, L_x(G))=\sum\limits_{\substack{I\subseteq V(Q),  \\ \text{\rm $I$ is independent}}}x^{|I|}\hom(Q_I, G).$$
<!-- end lemma-3 -->
<!-- head-5: body_md -->
Fix a homomorphism $\varphi$ from $Q$ to $L_x(G)$.
Let $I\subseteq V(Q)$ be the set of vertices mapped to the set $V(L_x(G))\setminus V(G)$ of newly added leaves.
Then $I$ is an independent set of $Q$ since $V(L_x(G))\setminus V(G)$ is an independent set of $L_x(G)$.
For any nonisolated vertex $v\in I$, all vertices of $N_Q(v)$ must map to the anchor of $\varphi(v)$.
This gives precisely the identifications in the definition of $Q_I$.
For any isolated vertex $v\in I$, the anchor of $\varphi(v)$ can be viewed as $\psi(v')$ for some homomorphism $\psi$ from $Q_I$ to $G$.
Once we fix a homomorphism $\psi$ from $Q_I$ to $G$, for each $v\in I$ we can choose $\varphi(v)$ independently from the $x$ leaves attached to $\psi(v')$.
This is a bijection, so the number of homomorphisms with this fixed set $I$ is $x^{|I|}\hom(Q_I,G)$.
Summing over independent sets $I$ proves the identity.
            
\hfill$\blacksquare$

<!-- lemma-4: body_md -->
\label{le:QI}
Let $Q$ be a bipartite graph and $I$ be an independent set of $Q$.
Then
\begin{itemize}
\item[{\rm (i)}] $v(Q_I)\leq v(Q)$;
\item[{\rm (ii)}] $Q_I$ is bipartite.
\end{itemize}
<!-- end lemma-4 -->
<!-- head-6: body_md -->
(i) In the construction of $Q_I$, each isolated vertex of $I$ is replaced by one isolated vertex, while every nonisolated vertex of $I$ is deleted.
Let $I'$ be the set of nonisolated vertices of $I$.
Then $v(Q_I)\leq v(Q)-|I'|\leq v(Q)$.

(ii) Let $A, B$ be the partite sets of $Q$.
For any nonisolated vertex $v\in I\cap A$, we have $N_Q(v)\subseteq B$, so after identifying $N_Q(v)$ we still obtain a bipartite graph.
Similarly, for any nonisolated vertex $v\in I\cap B$, we have $N_Q(v)\subseteq A$, so after identifying $N_Q(v)$ we still obtain a bipartite graph.
Therefore, even after taking the transitive closure of all these identifications, the resulting graph is bipartite.
Moreover, the new vertices corresponding to isolated vertices can be placed in either part.
Hence, $Q_I$ is a bipartite graph.
            
\hfill$\blacksquare$

<!-- lemma-6: body_md -->
\label{le:pQ}
Let $Q$ be a bipartite graph.
\begin{itemize}
\item[{\rm (i)}] For any independent set $I$, we have $p(Q_I)\leq p(Q)$.
\item[{\rm (ii)}] Among all independent sets $I$ for which $e(Q_I)>0$, choose one $I$ of maximum cardinality. Then $p(Q_I)<p(Q)$ whenever $p(Q)\geq 3$.
\end{itemize}
<!-- end lemma-6 -->
<!-- head-8: body_md -->
(i) In the construction of $Q_I$, all nonisolated vertices of $I$ (if any) are deleted.
At each identification step, if the resulting vertex $v'$ is nonisolated, then at least one of the vertices being identified was nonisolated immediately before that step.
Thus identification cannot increase the number of nonisolated vertices.
Moreover, adding isolated vertices does not affect this number.
Hence, we have $p(Q_I)\leq p(Q)$.

(ii) Let $I$ be an independent set of maximum cardinality subject to $e(Q_I)>0$.
We claim that $I$ contains a nonisolated vertex.
Suppose for a contradiction that all vertices in $I$ are isolated vertices.
Then all edges of $Q$ are contained in $Q-I$.
Choose an edge $uv\in E(Q)$.
Since $p(Q)\ge3$, there exists a nonisolated vertex $w\notin\{u,v\}$.
Since every vertex of $I$ is isolated, we have $w\notin I$, and $I\cup\{w\}$ is independent.
Note that $uv\in E(Q-(I\cup \{w\}))$.
Since $Q$ is bipartite, we have $|N_Q(w)\cap \{u,v\}|\leq 1$, so the image of $uv$ remains an edge of $Q_{I\cup\{w\}}$.
Then $e(Q_{I\cup \{w\}})>0$, and this contradicts the maximality of $I$.
Hence, $I$ contains a nonisolated vertex.
Since all nonisolated vertices of $I$ are deleted in the construction of $Q_I$, we have $p(Q_I)<p(Q)$.
            
\hfill$\blacksquare$

<!-- lemma-7: body_md -->
\label{le:K2}
Let $T$ be a bipartite graph with $e(T)>0$.
Among all sequences $(I_1,\ldots,I_s)$ defined by $($\ref{eq:Ti}$)$ for which $e(T_s)>0$, choose one with lexicographically largest exponent vector.
Then $p(T_s)=2$, and consequently, $T_s\cong K_2\cup qK_1$ for some integer $q\geq 0$.

<!-- head-9: body_md -->
First, there exists at least one sequence $(I_1,\ldots,I_s)$ satisfying $e(T_s)>0$.
Indeed, we may take $I_1=\cdots=I_s=\emptyset$.
Since $Q_{\emptyset}=Q$ for every graph $Q$, we have $T_0=T_1=\cdots=T_s=T$, and hence $e(T_s)=e(T)>0$.
Since there are only finitely many possible sequences, there exists one with lexicographically largest exponent vector.
Fix such a sequence $(I_1,\ldots,I_s)$.
Since $e(T_s)>0$, we have $e(T_i)>0$ for each $i\in \{0, \ldots, s\}$.

\begin{claim}\label{cl:K2}
For each $i\in \{1, \ldots, s\}$, the set $I_i$ has maximum cardinality among all independent sets $I\subseteq V(T_{i-1})$ for which $e((T_{i-1})_I)>0$.
\end{claim}

\begin{proof}
For a contradiction, suppose that for some $j$ there is an independent set $J\subseteq V(T_{j-1})$ such that $e((T_{j-1})_J)>0$ and $|J|>|I_j|$.
Let $I'_i\colonequals I_i$ for $1\leq i\leq j-1$, $I'_j\colonequals J$, and $I'_i\colonequals \emptyset$ for $j+1\leq i\leq s$.
Write $T_0', \ldots, T_s'$ for the graphs produced by $(I_1', \ldots, I_s')$.
Then $e(T'_s)=e(T'_j)=e((T'_{j-1})_J)=e((T_{j-1})_J)>0$.
However, the exponent vector of $(I'_1,\ldots,I'_s)$ is lexicographically larger than that of $(I_1,\ldots,I_s)$, a contradiction.
\end{proof}

We now show that $p(T_s)=2$.
Since $e(T_s)>0$, we have $p(T_s)\geq 2$.
Suppose for a contradiction that $p(T_s)\geq 3$.
Then $p(T_0)\geq p(T_1)\geq \cdots \geq p(T_s)\geq 3$ by Lemma~\ref{le:pQ}~(i).
By Lemma~\ref{le:pQ}~(ii) and Claim~\ref{cl:K2}, we have $p(T_i)<p(T_{i-1})$ whenever $p(T_{i-1})\geq 3$.
Since these quantities are integers, we have $p(T_i)\leq p(T_{i-1})-1$ whenever $p(T_{i-1})\geq 3$.
Then $p(T_s)\leq p(T_0)-s \leq v(T)-(v(T)-2)=2$, a contradiction.
Hence, we have $p(T_s)=2$.
Since $T_s$ is a simple graph with exactly two nonisolated vertices and $e(T_s)>0$, these two vertices are adjacent and all other vertices are isolated.
Therefore, $T_s\cong K_2\cup qK_1$ for some integer $q\geq 0$.
The proof of Lemma~\ref{le:K2} is complete.
            
\hfill$\blacksquare$

\end{align*}

\end{align*}

Since $v(Z)\leq v(T)n^2AB$$, A=n^{2v(T)-2}$ and $B=n^{\beta}$, where $\beta$ is fixed once $T, \mathcal{F},\varepsilon$ are fixed, we have $v(Z)=O(n^{2v(T)+\beta})$.
Hence, $Z$ can be constructed in polynomial time.
Since $T$ is fixed, $L=\mathcal{N}(Z_0,T)$ and $X$ can be computed in polynomial time by enumerating copies of $T$ in $Z_0$ and $Z_0+e$ and checking the conditions defining $X$.
Thus $L\geq 0$ and $S=XB^2>0$ are polynomial-time computable integers.
Now by Corollary~\ref{cor:transfer}, each of the two approximation problems is NP-hard even when the input graph is restricted to $\{Z(G)\colon\, G\in \mathcal{G}_{k,N^{\ast}}\}$.
Consequently, the same holds on $\{Z(G)\colon\, G\in \mathcal{G}_k\}$ and on arbitrary host graphs.
This completes the proof of Lemma~\ref{lem:symmetric-rooted-transfer}.
            
\hfill$\blacksquare$

\label{cl:multipar-T-1}
Every copy of $T$ in $Z$ contains at most $q$ auxiliary vertices.

<!-- proof-3: body_md -->
Let $T'$ be a copy of $T$ in $Z$, and assume that $T'$ contains an auxiliary vertex.
First, assume that $T'$ contains auxiliary vertices corresponding to at least two edges of $G$.
Since auxiliary vertices corresponding to distinct edges of $G$ are nonadjacent, all auxiliary vertices of $T'$ must belong to the same partite set of $T'$.

Consequently, the auxiliary vertices lie in at most $r-2$ partite sets of $T'$.
Therefore, the number of auxiliary vertices of $T'$ is at most $s_3+\cdots +s_r=q$.

<!-- claim-4: body_md -->
\label{cl:multipar-T-2}
For each copy of $T$ containing exactly $q$ auxiliary vertices, all its auxiliary vertices belong to one $C_{uv}$.
Moreover, for each $uv\in E(G)$, the number of copies of $T$ containing exactly $q$ auxiliary vertices of $C_{uv}$ is $S$.

<!-- proof-4: body_md -->
Let $T'$ be a copy of $T$ containing exactly $q$ auxiliary vertices in $Z$.
We first show that all its auxiliary vertices belong to one $C_{uv}$.
Suppose for a contradiction that $T'$ contains auxiliary vertices corresponding to at least two edges of $G$.
Then by the same argument as Claim~\ref{cl:multipar-T-1}, all the $q$ auxiliary vertices of $T'$ must belong to the same partite set of $T'$.
For $r\geq4$, the number of such vertices is at most $s_r= q-(s_3+\cdots+s_{r-1})< q$, a contradiction.
For $r=3$, we have $q=s_3$, so all the $q$ auxiliary vertices of $T'$ must belong to the partite set of size $s_3$.
Let $u_1v_1$ and $u_2v_2$ be two edges of $G$ such that $T'$ contains auxiliary vertices from $C_{u_1v_1}$ and $C_{u_2v_2}$.
Every vertex in the other two partite sets is a source vertex adjacent to these auxiliary vertices, so it must belong to a cluster corresponding to a vertex of $\{u_1,v_1\}\cap\{u_2,v_2\}$.
Since the two edges $u_1v_1$ and $u_2v_2$ are distinct, there is at most one vertex in $\{u_1,v_1\}\cap\{u_2,v_2\}$.
However, every cluster is an independent set, so it cannot contain vertices from two partite sets of $T'$, a contradiction.
Hence, all auxiliary vertices of $T'$ belong to one $C_{uv}$.

Fix $uv\in E(G)$, and suppose that all $q$ auxiliary vertices of $T'$ lie in $C_{uv}$.
Since each independent set $C_{uv}^i$ meets at most one partite set of $T'$, these vertices lie in at most $r-2$ partite sets, whose total size is at most $s_3+ \cdots +s_r=q$.
Thus they constitute exactly $r-2$ entire partite sets of sizes $s_3, \ldots, s_r$, one contained in each of $C_{uv}^3, \ldots, C_{uv}^r$.
The remaining two partite sets consist of source vertices adjacent to all these auxiliary vertices, so they lie in $C_u\cup C_v$.
Since $C_u$ and $C_v$ are independent sets, one partite set lies entirely in $C_u$ and the other entirely in $C_v$.
Consequently, every such copy is obtained uniquely by choosing subsets of sizes $a_1, \ldots, a_r$ in $C_u, C_v, C_{uv}^3, \ldots, C_{uv}^r$, respectively, for some $(a_1, \ldots, a_r)\in\mathcal{P}$.
Conversely, every such choice gives a copy of $T$.
Therefore, the number of copies of $T$ containing exactly $q$ auxiliary vertices of $C_{uv}$ is
$$\sum_{\mathbf{a}\in \mathcal{P}}{|C_u|\choose a_1}{|C_v|\choose a_2}\prod_{i=3}^{r}{|C_{uv}^{i}| \choose a_i}=\sum_{\mathbf{a}\in \mathcal{P}}{B\choose a_1}{B\choose a_2}\prod_{i=3}^{r}{AB \choose a_i}=S.$$
This completes the proof.

<!-- claim-6: body_md -->
\label{cl:multipar-upper}
There exist constants $c=c(T,\mathcal{F})>0$, $C=C(T,\mathcal{F})>0$ and $\eta=\eta(T,\mathcal{F})>0$ such that
$$\operatorname{ex}(Z, T, \mathcal{F})\leq Sz +S \frac{cv(Z)^{w-\eta}}{A^{w-k}B^{w}} +Cn^{v(T)+q-1}A^{q-1}B^{v(T)},$$ where $w=k+(r-2){k\choose 2}$.
<!-- end claim-6 -->
<!-- proof-6: body_md -->
Let $J$ be an $\mathcal F$-free subgraph of $Z$ with $\mathcal{N}(J,T)=\operatorname{ex}(Z, T, \mathcal{F})$.
Let $Y(J)$ denote the set of copies of $T$ in $J$ containing exactly $q$ auxiliary vertices.
By Claim~\ref{cl:multipar-T-3}, it suffices to show that for some constants $c=c(T,\mathcal{F})>0$ and $\eta=\eta(T,\mathcal{F})>0$, we have
\begin{equation}\label{eq:multipar-YJ}
    |Y(J)| \leq Sz +S \frac{cv(Z)^{w-\eta}}{A^{w-k}B^{w}},
\end{equation}
where $w=k+(r-2){k\choose 2}$.

Define a random spanning subgraph $R_J$ of $G$ on the vertex set $V(G)$ as follows.
For every $u\in V(G)$, choose a vertex $u'\in C_u$ uniformly at random.
For every $uv\in E(G)$ and every $i\in\{3,\ldots,r\}$, choose a vertex $w_{uv}^i\in C_{uv}^i$ uniformly at random, with all choices mutually independent.
Put $uv\in E(R_J)$ if and only if $u', v', w_{uv}^3, \ldots, w_{uv}^r$ are vertices of $J$ and induce a copy of $K_r$ in $J$.

For $uv\in E(G)$ and $\mathbf a=(a_1,\ldots,a_r)\in\mathcal P$, let $Y_{uv,\mathbf a}(J)$ consist of the copies in $Y(J)$ having $a_1$ vertices in $C_u$, $a_2$ vertices in $C_v$, and $a_i$ vertices in $C_{uv}^i$ for each $3\leq i\leq r$.
By Claim~\ref{cl:multipar-T-2}, we have $|Y(J)|=\sum_{uv\in E(G)}\sum_{\mathbf{a}\in \mathcal{P}}|Y_{uv,\mathbf{a}}(J)|$.
We now prove an upper bound on $|Y_{uv,\mathbf{a}}(J)|$.
Choose uniformly and independently an $a_1$-subset of $C_u$, an $a_2$-subset of $C_v$, and an $a_i$-subset of $C_{uv}^{i}$ for every $i\in\{3,\ldots,r\}$.
Then from each chosen subset, choose one representative vertex uniformly at random.
For any one of the sets $C_u,C_v,C_{uv}^3,\ldots,C_{uv}^r$, if its size is $M$ and the chosen subset has size $a$, then each vertex is the representative with probability $\frac{a}{M}\cdot\frac{1}{a}=\frac{1}{M}$.
Thus the representatives are independent and uniformly distributed over $C_u, C_v, C_{uv}^3, \ldots, C_{uv}^r$, respectively, exactly as in the definition of $R_J$.
Moreover, if the chosen subsets form a copy of $T$ in $J$, then the $r$ representative vertices form a copy of $K_r$ in $J$.
Therefore, we have
$|Y_{uv,\mathbf{a}}(J)| \leq {B\choose a_1}{B\choose a_2}\prod_{i=3}^{r}{AB \choose a_i} \Pr[uv\in E(R_J)],$
and thus
$$|Y(J)|=\sum_{uv\in E(G)}\sum_{\mathbf{a}\in \mathcal{P}}|Y_{uv,\mathbf{a}}(J)| \leq \sum_{uv\in E(G)}\sum_{\mathbf{a}\in \mathcal{P}}{B\choose a_1}{B\choose a_2}\prod_{i=3}^{r}{AB \choose a_i} \Pr[uv\in E(R_J)] = S\cdot \mathbb{E}[e(R_J)].$$
Now to prove Inequality~(\ref{eq:multipar-YJ}), it suffices to show that
\begin{equation}\label{eq:multipar-eRJ}
    \mathbb{E}[e(R_J)] \leq z +\frac{cv(Z)^{w-\eta}}{A^{w-k}B^{w}}.
\end{equation}

Since deleting one edge from each copy of $K_k$ in $R_J$ yields a $K_k$-free subgraph of $G$, we have $\operatorname{ex}(G, K_2, K_k)\geq e(R_J)-\mathcal{N}(R_J, K_k)$.
Combining with Equality~(\ref{eq:multipar-z}), we obtain
$e(R_J)\leq \operatorname{ex}(G, K_2, K_k)+\mathcal{N}(R_J, K_k)=z+\mathcal{N}(R_J, K_k).$
Taking expectations, we have
\begin{equation}\label{eq:multipar-eRJ+}
    \mathbb{E}[e(R_J)] \leq z+\mathbb{E}[\mathcal{N}(R_J, K_k)].
\end{equation}
It remains to show that $\mathbb{E}[\mathcal{N}(R_J, K_k)]\leq  \frac{cv(Z)^{w-\eta}}{A^{w-k}B^{w}}.$

Let $F_0\in \mathcal{F}$ with $\chi(F_0)=k$.
Then $J$ is $F_0$-free.
Let $W$ be the graph obtained from $K_k$ by adding, for every edge of $K_k$, $r-2$ new vertices and adding all edges so that these $r-2$ vertices together with the two endvertices of the edge form a copy of $K_r$.
All the new vertices are distinct, and no edges are added between vertices corresponding to distinct edges of $K_k$.
Then $v(W)=k+(r-2)\binom{k}{2}=w.$
Moreover, since $\chi(F_0)=k$ and $K_k\subseteq W$, $F_0$ is a subgraph of a sufficiently large blow-up of $W$.
By Lemma~\ref{le:AlSh16-degen}, there exist constants $c_1>0$ and $\eta>0$ such that $\mathcal{N}(J,W)\leq c_1v(Z)^{w-\eta}.$

For every copy $K$ of $K_k$ in $G$, let $\mathbf{1}_K$ be the indicator random variable of the event that $K$ is a subgraph of $R_J$.
Then $\mathcal{N}(R_J,K_k)=\sum_{K}\mathbf{1}_K,$ where the sum is taken over all copies $K$ of $K_k$ in $G$.
Hence, we have $$\mathbb{E}[\mathcal{N}(R_J,K_k)]=\sum\nolimits_{K}\Pr[K\subseteq R_J]$$ by linearity of expectation.
Fix a copy $K$ of $K_k$ in $G$.
By the definition of $R_J$, in order that $K$ be contained in $R_J$, we need to choose one vertex from $C_u$ for every $u\in V(K)$, and for every $uv\in E(K)$, we need to choose one vertex from each of $C_{uv}^3, \ldots, C_{uv}^r$. The total number of possible choices of these $k+(r-2)\binom{k}{2}=w$ vertices is $B^k(AB)^{w-k}=A^{w-k}B^w.$
All these choices occur with the same probability $\frac{1}{A^{w-k}B^w}$.
If $K\subseteq R_J$, then the $w$ chosen vertices form a copy of $W$ in $J$.
Let $M_K$ be the number of choices of these $w$ vertices for which $K\subseteq R_J$.
Then $\Pr[K\subseteq R_J]=\frac{M_K}{A^{w-k}B^w},$ so
$$\mathbb{E}[\mathcal{N}(R_J,K_k)]=\sum\nolimits_{K}\Pr[K\subseteq R_J]=\sum\nolimits_{K}\frac{M_K}{A^{w-k}B^w}=\frac{1}{A^{w-k}B^w}\sum\nolimits_{K} M_K.$$

Since $W$ is a fixed graph depending only on $r$ and $k$, there is a constant $c_{r,k}>0$ such that every copy of $W$ in $J$ can arise from at most $c_{r,k}$ choices of the $w$ vertices counted in $\sum_K M_K$.
Thus $\sum_K M_K \leq c_{r,k}\mathcal{N}(J,W).$
Combining with $\mathcal{N}(J,W)\leq c_1v(Z)^{w-\eta}$, we have
$$\mathbb{E}[\mathcal{N}(R_J,K_k)]=\frac{1}{A^{w-k}B^w}\sum\nolimits_{K} M_K\leq \frac{1}{A^{w-k}B^w}c_{r,k} \mathcal{N}(J,W)\leq \frac{c_{r,k}c_1v(Z)^{w-\eta}}{A^{w-k}B^{w}} = \frac{cv(Z)^{w-\eta}}{A^{w-k}B^{w}},$$
where $c=c_{r,k}c_1$ depends only on $T$ and $\mathcal{F}$.
This together with (\ref{eq:multipar-eRJ+}) implies (\ref{eq:multipar-eRJ}), and hence (\ref{eq:multipar-YJ}).
The proof of Claim~\ref{cl:multipar-upper} is complete.

\label{cl:multipar-lower}
$\operatorname{ex}'(Z, T, \mathcal{F})\geq Sz.$

<!-- proof-7: body_md -->
Let $H$ be a spanning subgraph of $G$ satisfying $\chi(H)\leq k-1$ and $e(H)=z$.
Let $Z_0$ be the spanning subgraph of $Z$ obtained by deleting all edges of $G(B)$.
Then $Z_0+H(B)$ is also a spanning subgraph of $Z$.
We shall prove that $Z_0+H(B)$ is $\mathcal{F}$-free.

We first show that $\chi(Z_0+H(B))\leq k-1$.
Fix a proper $(k-1)$-coloring of $H$, and assign every vertex of $C_u$ the color of $u$.
For each edge $uv\in E(G)$, whether the colors of $u$ and $v$ are distinct or equal, there are at least $k-1-2\geq r-2$ available colors.
So we can assign pairwise distinct colors, different from the colors of $u$ and $v$, to $C_{uv}^3, \ldots, C_{uv}^r$.
Since auxiliary vertices corresponding to distinct edges of $G$ are nonadjacent, this yields a proper $(k-1)$-coloring of $Z_0+H(B)$.
Hence, $\chi(Z_0+H(B))\leq k-1$.

Since $\min_{F\in \mathcal{F}}\chi(F)=k$, the graph $Z_0+H(B)$ is $\mathcal{F}$-free, so $\operatorname{ex}'(Z, T, \mathcal{F})\geq \mathcal{N}(Z_0+H(B),T).$

Therefore,
$\operatorname{ex}'(Z, T, \mathcal{F})\geq \mathcal{N}(Z_0+H(B),T)\geq Se(H)=Sz.$
