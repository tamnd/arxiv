<!-- thm-1: body_md -->
\label{thm:dougall}
Let $a,b,c,d$ be complex numbers satisfying $\Re(c+d-a-b)>1$, and assume that both sides of the following identity are well-defined. Then
\[
\sum_{n\in\mathbb{Z}}\frac{(a,b)_n}{(c,d)_n}=\frac{\Gamma(c)\Gamma(d)\Gamma(1-a)\Gamma(1-b)\Gamma(c+d-a-b-1)}{\Gamma(c-a)\Gamma(d-a)\Gamma(c-b)\Gamma(d-b)}
\]
holds.
<!-- end thm-1 -->
<!-- thm-2: body_md -->
\label{thm:main}
The following identity holds in $\mathcal{Z}_{\mathcal{F}}[[a,b,c,d]]/(ab-cd)$:
\begin{multline*}
1+ab\zeta^{}_{\mathcal{F}}\left(\frac 1{1-y(a+b)-yxab}yx\frac 1{1+y(c+d)-yxcd}\right)\\
=\exp\bigg(\sum_{2\leq n}\frac{Z_{\mathcal{F}}(n)}{n}(a^n+b^n+c^n+d^n+(a+b+c+d)^n\\
\qquad\qquad\qquad -(a+c)^n-(a+d)^n-(b+c)^n-(b+d)^n)\bigg).
\end{multline*}

\label{thm:sub}
Let $k$ and $r$ be positive integers, and let $i$ be an integer with $1\leq i\leq r$. Then we have
\begin{align*}
\sum_{\substack{1\leq k_1,\dots,k_r\\k_1,\dots,k_{i-1},k_{i+1},\dots,k_r:\mathrm{odd}\\k_i:\mathrm{even}\\k_1+\cdots+k_r=k}}\zeta^{}_{\mathcal{F}}(k_1, \ldots, k_r)\in\mathbb{Q}[Z_{\mathcal{F}}(3), Z_{\mathcal{F}}(5),\dots].
\end{align*}

<!-- lem-1: body_md -->
\label{lem:truncated}
Let $N$ be a positive integer. The following identity of formal power series in $a,b,c,d$ holds:
\[
\sum_{n=1}^{N-1}\frac{(a,b)_n}{(1-c,1-d)_n}=ab\zeta^{}_{N}\left(\frac 1{1-y(a+b)-yxab}yx\frac 1{1+y(c+d)-yxcd}\right)\frac{(N-1)!^2}{(1-c,1-d)_{N-1}}.
\]
<!-- end lem-1 -->
<!-- proof-1: body_md -->
A direct calculation gives
\begin{align*}
&\sum_{n=1}^{N-1}\frac{(a,b)_n}{(1-c,1-d)_n}\frac{(1-c,1-d)_{N-1}}{(N-1)!^2}\\
&=ab\sum_{n=1}^{N-1}\left(\prod_{j=1}^{n-1}\left(1+\frac{a+b}j+\frac{ab}{j^2}\right)\right)\frac 1{n^2}\prod_{l=n+1}^{N-1}\left(1-\frac{c+d}l+\frac{cd}{l^2}\right)\\
&=ab\sum_{n=1}^{N-1}\zeta^{}_{N}\left(\frac 1{1-y(a+b)-yxab}\right)\frac 1{n^2}\zeta^{}_{n,N}\left(\frac 1{1+y(c+d)-yxcd}\right)\\
&=ab\zeta^{}_{N}\left(\frac 1{1-y(a+b)-yxab}yx\frac 1{1+y(c+d)-yxcd}\right).
\end{align*}
This completes the proof.

<!-- cor-1: body_md -->
\label{cor:infinite}
The following identity of formal power series in $a,b,c,d$ holds:
\[
\sum_{0<n}\frac{(a,b)_n}{(1-c,1-d)_n}=ab\zeta^*\left(\frac 1{1-y(a+b)-yxab}yx\frac 1{1+y(c+d)-yxcd}\right)\Gamma^*(1-c)\Gamma^*(1-d).
\]
<!-- end cor-1 -->
<!-- proof-2: body_md -->
Define the $\mathbb{Q}$-linear signed reversal map $\rho:\mathfrak{H}^1\to\mathfrak{H}^1$ by
\[
\rho(1_{\mathfrak{H}})=1_{\mathfrak{H}},\qquad
\rho(yx^{k_1-1}\cdots yx^{k_r-1})
=(-1)^{k_1+\cdots+k_r}
yx^{k_r-1}\cdots yx^{k_1-1}.
\]
We extend $\rho$ coefficientwise to $\mathfrak{H}^1[[a,b,c,d]]$. Put
\[
P=\frac{1}{1-y(a+b)-yxab},
\qquad
Q=\frac{1}{1+y(c+d)-yxcd}.
\]
Expanding $P$ and $Q$ as formal power series and splitting the defining sum for $\zeta^{}_{\mathcal{S}}(PyxQ)$ according to whether the cut occurs before or after the distinguished factor $yx$, we obtain
\[
\zeta^{}_{\mathcal{S}}(PyxQ)
=\zeta^*(PyxQ)\zeta^*(\rho(Q))
 +\zeta^*(P)\zeta^*(\rho(PyxQ)).
\]
Since
\[
\rho(Q)=\frac{1}{1-y(c+d)-yxcd}
\]
and
\[
\rho(PyxQ)
=\frac{1}{1-y(c+d)-yxcd}
 yx\frac{1}{1+y(a+b)-yxab},
\]
multiplying the preceding identity by $ab$ and using $ab=cd$ in the second term gives


\begin{align*}
&ab\zeta^{}_{\mathcal{S}}\left(\frac 1{1-y(a+b)-yxab}yx\frac 1{1+y(c+d)-yxcd}\right)\\
&=ab\zeta^*\left(\frac 1{1-y(a+b)-yxab}yx\frac 1{1+y(c+d)-yxcd}\right)\zeta^*\left(\frac 1{1-y(c+d)-yxcd}\right)\\
&\qquad+cd\zeta^*\left(\frac 1{1-y(a+b)-yxab}\right)\zeta^*\left(\frac 1{1-y(c+d)-yxcd}yx\frac 1{1+y(a+b)-yxab}\right).
\end{align*}
Since
\begin{align*}
\zeta^{}_{N}\left(\frac 1{1-y(a+b)-yxab}\right)&=\prod_{n=1}^{N-1}\left(1+\frac{a+b}{n}+\frac{ab}{n^2}\right)\\
&=\frac{(1+a,1+b)_{N-1}}{(N-1)!^2},
\end{align*}
harmonic regularization yields
\begin{align*}
\zeta^*\left(\frac 1{1-y(a+b)-yxab}\right)&=\frac{1}{\Gamma^*(1+a)\Gamma^*(1+b)}.
\end{align*}
Hence, combining the identity above with Corollary \ref{cor:infinite} yields
\begin{align*}
&ab\zeta^{}_{\mathcal{S}}\left(\frac 1{1-y(a+b)-yxab}yx\frac 1{1+y(c+d)-yxcd}\right)\\
&=\frac 1{\Gamma^*(1+c)\Gamma^*(1-c)\Gamma^*(1+d)\Gamma^*(1-d)}\sum_{0<n}\frac{(a,b)_n}{(1-c,1-d)_n}\\
&\qquad+\frac 1{\Gamma^*(1+a)\Gamma^*(1-a)\Gamma^*(1+b)\Gamma^*(1-b)}\sum_{0<n}\frac{(c,d)_n}{(1-a,1-b)_n}.
\end{align*}
In $\mathcal{Z}_{\mathcal{S}}$,
\[
\frac 1{\Gamma^*(1+X)\Gamma^*(1-X)}=\frac{\sin\pi X}{\pi X}\equiv 1\pmod{\pi^2\mathcal{Z}}.
\]
Hence, we obtain
\begin{align*}
&ab\zeta^{}_{\mathcal{S}}\left(\frac 1{1-y(a+b)-yxab}yx\frac 1{1+y(c+d)-yxcd}\right)\\
&=\sum_{0<n}\frac{(a,b)_n}{(1-c,1-d)_n}+\sum_{0<n}\frac{(c,d)_n}{(1-a,1-b)_n}\\
&=\sum_{n\in\mathbb{Z}}\frac{(a,b)_n}{(1-c,1-d)_n}-1\\
&=\frac{\Gamma(1-a)\Gamma(1-b)\Gamma(1-c)\Gamma(1-d)\Gamma(1-a-b-c-d)}{\Gamma(1-a-c)\Gamma(1-a-d)\Gamma(1-b-c)\Gamma(1-b-d)}-1.
\end{align*}
The last equality follows from Theorem \ref{thm:dougall}. Applying the identity 
\begin{align*}
    \Gamma(1-x)=\exp\left(\gamma x+\sum_{2\leq n}\frac{\zeta(n)}{n}x^n\right)
\end{align*}
to the above expression yields the desired result.

<!-- prop-1: body_md -->
\label{prop:G_p 1-cocyle}
We obtain the following equality
\begin{align}\label{eq:G_p 1-cocyle}
\overline{G}_p(x,y;u,v)\overline{G}_p(x+y,z;u,v)=\overline{G}_p(y,z;u,v)\overline{G}_p(x,y+z;u,v).
\end{align}
<!-- end prop-1 -->
<!-- proof-3: body_md -->
Set $\eta\coloneqq-\frac{v}{u}, E\coloneqq \mathbb{F}_p(u,v)[t]/(t^p-\eta)$ and define a $\mathbb{F}_p(u,v)$-linear map $\theta\colon E\to E$ by $\theta \coloneqq t\frac{d}{dt}$. Since $\{1, t, \ldots, t^{p-1}\}$ is a $\mathbb{F}_p(u,v)$-basis of $E$ and $\theta(t^j)=jt^j \ (1\le j \le p-1)$, we have $\mathrm{Ker}\,\theta=\mathbb{F}_p(u,v)$.
For an indeterminate $X$ and a non-negative integer $0 \le n \le p-1$, set
\begin{align*}
h_n(X)
\coloneqq \frac{(Xu)_n}{(1-Xv)_n} \in \mathbb{F}_p(u,v)(X), \quad
H_X(t)\coloneqq \sum_{n=0}^{p-1}h_n(X)t^{n} \in \mathbb{F}_p(u,v)(X)[t].
\end{align*}
Then we have
\begin{align*}
(1-t)\theta H_X(t)-X(v+ut)H_X(t)
&=\sum_{n=0}^{p-1}(n-Xv)h_n(X)t^n-\sum_{n=0}^{p-1}(n+Xu)h_n(X)t^{n+1}\\
&=-Xv-(p-1+Xu)h_{p-1}(X)t^p.
\end{align*}
Moreover, we have
\begin{align*}
(p-1+Xu)h_{p-1}(X)
&=\frac{(Xu)_p}{(1-Xv)_{p-1}}
=\frac{Xu-X^pu^p}{1-X^{p-1}v^{p-1}}\\
&\equiv Xu \bmod{X^p}. 
\end{align*}
Since $t^p=\eta=-v/u$ and $1-t$ is invertible in $E$, we have 
\begin{align*}
\theta H_X(t)\equiv XR(t)H_X(t) \bmod{X^p},
\end{align*}
where $R(t)\coloneqq\frac{v+ut}{1-t}$. Then, putting $X=x,y,x+y$, we have the following equalities in $E[x,y]/(x,y)^p$:
\begin{align*}
\theta H_x(t)\equiv xR(t)H_x(t),\quad \theta H_y(t)\equiv yR(t)H_y(t), \quad
\theta H_{x+y}(t)\equiv (x+y)R(t)H_{x+y}(t).
\end{align*}
By differentiating above, we have
\begin{align*}
\theta\left(\frac{H_x(t)H_y(t)}{H_{x+y}(t)}\right)\equiv 0 \bmod{(x,y)^p}.
\end{align*}
From above, there exists $C_p(x,y) \in E[x,y]/(x,y)^{N+1}$ satisfying 
\begin{align}\label{eq:H_xH_y=CH_{x+y}}
H_x(t)H_y(t)=C_p(x,y)H_{x+y}(t).
\end{align}
By comparing the constant term in \eqref{eq:H_xH_y=CH_{x+y}} and using $t^p=\eta=-v/u$, we have
\begin{align*}
C_p(x,y)=1+\eta\sum_{n=1}^{p-1}h_n(x)h_{p-n}(y).
\end{align*}
Moreover, since
\begin{align*}
\eta h_{p-n}(y)
=\frac{1-y^{p-1}u^{p-1}}{1-y^{p-1}v^{p-1}}\frac{(yv)_n}{(1-yu)_n}
\equiv \frac{(yv)_n}{(1-yu)_n} \bmod{y^p},
\end{align*}
we see that
\begin{align}\label{eq:C_p=G_p}
C_p(x,y)=\overline{G}_p(x,y;u,v).
\end{align}
Substituting \eqref{eq:C_p=G_p} into \eqref{eq:H_xH_y=CH_{x+y}}, we have 
\begin{align}\label{eq:H_xH_y=G_pH_{x+y}}
H_x(t)H_y(t)=\overline{G}_p(x,y;u,v)H_{x+y}(t).
\end{align}
Therefore, using \eqref{eq:H_xH_y=G_pH_{x+y}} to rewrite
\begin{align*}
    (H_x(t)H_y(t))H_z(t)=H_x(t)(H_y(t)H_z(t))
\end{align*}
into $H_{x+y+z}(t)$, we complete the proof.

<!-- prop-2: body_md -->
\label{prop:L_n}
We have
\begin{align*}
L_n(x,y;u,v)=\frac{B_{p-n}}{n^2}((x+y)^n-x^n-y^n)((u+v)^n-u^n-v^n).
\end{align*}
<!-- end prop-2 -->
<!-- proof-4: body_md -->
Taking $\log$ of both sides of \eqref{eq:G_p 1-cocyle} and comparing the degree $n$ part, we have
\begin{align*}
L_n(x,y;u,v)+L_n(x+y,z;u,v)=L_n(y,z;u,v)+L_n(x,y+z;u,v).
\end{align*}
Since $\overline{G}_p(x,0;u,v)=\overline{G}_p(0,y;u,v)=1$ by the definition of $\overline{G}_p$, we have $L_n(x,0;u,v)=L_n(0,y;u,v)=0$.
Let $\partial_2$ denote the differentiation with respect to the second variable. Since
\begin{align*}
\partial_2L_n(x+y,z;u,v)=\partial_2L_n(y,z;u,v)+\partial_2L_n(x,y+z;u,v),
\end{align*}
by substituting $z=0$, we have
\begin{align}\label{eqref:del2}
\partial_2L_n(x,y;u,v)=\partial_2L_n(x+y,0;u,v)-\partial_2L_n(y,0;u,v).
\end{align}
Since $L_n(x,y)$ is a homogeneous polynomial of degree $n$ with respect to $x,y$, we can write $L_n$ as follows:
\begin{align*}
L_n(x,y;u,v)=\sum_{j=0}^{n}A_j(u,v)x^{n-j}y^j,
\end{align*}
where $A_j(u,v) \in \mathbb{F}_p[u,v]$. Thus, we have $\partial_2L_n(x,0;u,v)=A_1(u,v)x^{n-1}$. Substituting this equality into \eqref{eqref:del2} and rewriting $A_1(u,v)$ as $A(u,v)$, we have
\begin{align*}
\frac{\partial L_n}{\partial y}(x,y;u,v)=A(u,v)((x+y)^{n-1}-y^{n-1}).
\end{align*}
Therefore, since $n<p$, we have
\begin{align*}
L_n(x,y;u,v)=\frac{A(u,v)}{n}((x+y)^n-x^n-y^n).
\end{align*}
Let $[x^{k_1}_1\cdots x^{k_n}_n](f)$ be the coefficient of $x^{k_1}_1\cdots x^{k_n}_n$ in a polynomial $f(x_1, \ldots, x_n)$. Put $\lambda_{p,n}\coloneqq[x^{n-1}y](A(x,y)) \in \mathbb{F}_p$. Since $L_n$ is a bi-homogeneous polynomial of degree $n$ with respect to $x,y$, and also to $u,v$ and $\overline{G}_p(x,y;u,v)=\overline{G}_p(u,v;x,y)$, we have
\begin{align}\label{eq:L_n(x,y;u,v)=L_n(u,v;x,y)}
L_n(x,y;u,v)=L_n(u,v;x,y).
\end{align}
Comparing the coefficient of $x^{n-1}y$ in \eqref{eq:L_n(x,y;u,v)=L_n(u,v;x,y)}, we see that
\begin{align*}
A(u,v)=\frac{\lambda_{p,n}}{n}((u+v)^n-u^n-v^n),
\end{align*}
which leads to
\begin{align*}
L_n(x,y;u,v)=\frac{\lambda_{p,n}}{n^2}((x+y)^n-x^n-y^n)((u+v)^n-u^n-v^n).
\end{align*}

Finally, we prove $\lambda_{p,n}=B_{p-n}$. By the expression above, we have $\lambda_{p,n}=[x^{n-1}yu^{n-1}v](L_n)=[x^{n-1}yu^{n-1}v](\log\overline{G}_p)$. Moreover, since $[x^{n-1}yu^{n-1}v](\overline{G}_p-1)^j=0$ for $j\ge2$, we see that $[x^{n-1}yu^{n-1}v](\log\overline{G}_p)=[x^{n-1}yu^{n-1}v](\overline{G}_p)$.
By the definition of $\overline{G}_p$, we have
\begin{align*}
[x^{n-1}yu^{n-1}v](\overline{G}_p)
&=[x^{n-1}yu^{n-1}v]\sum_{m=1}^{p-1}\frac{xyuv}{m^2}\frac{\prod_{j=1}^{m-1}(1+\frac{xu}{j})(1+\frac{yv}{j})}{\prod_{j=1}^{m}(1-\frac{xv}{j})(1-\frac{yu}{j})}\\
&=\sum_{m=1}^{p-1}\frac{1}{m^2}[x^{n-2}u^{n-2}]\prod_{j=1}^{m-1}\left(1+\frac{xu}{j}\right)\\
&=\zeta^{}_{p}(\{1\}^{n-2},2)\equiv B_{p-n} \pmod{p}.
\end{align*}
The last congruence comes from \cite[Theorem 4.5]{hht}.
This completes the proof.

Since
\begin{align*}
((x+y)^n-x^n-y^n)((u+v)^n-u^n-v^n)=Q_n(xu,yv,xv,yu)=\varphi(Q_n(a,b,c,d)),

By the injectivity of $\varphi$ and $\overline{\varphi}_N$, we see that 
\begin{align*}
F_{p}(a,b,c,d)=\exp\left(\sum_{n=2}^{N}\frac{B_{p-n}}{n^2}Q_n(a,b,c,d)\right),
\end{align*}
which is \eqref{eq:enough_to_show}. This completes the proof of Theorem \ref{thm:main} with $\mathcal{F}=\mathcal{A}$.

For an indeterminate $s$, put
\begin{align*}
\alpha_{\pm}(s)\coloneqq \frac{s\pm\sqrt{s^2+4}}{2} \in \mathbb{Q}[[s]]
\end{align*}
and for indeterminates $u,v, t$, put $(a,b,c,d)=(\alpha_+(u)t, \alpha_-(u)t, -\alpha_+(v)t,-\alpha_-(v)t)$. Then we have an injection
\begin{align*}
\mathcal{Z}_{\mathcal{F}}[[a,b,c,d]]/(ab-cd) \to \mathcal{Z}_{\mathcal{F}}[[u,v,t]].
\end{align*}
For a non-negative integer $n$, set
\begin{align*}
q_n(u,v) \coloneq Q_n(\alpha_+(u),\alpha_-(u),-\alpha_+(v),-\alpha_-(v)),
\end{align*}
which is an element in $\mathbb{Q}[u,v]$ by the fundamental theorem of symmetric polynomials. From Theorem \ref{thm:main}, we have
\begin{align}\label{eq:Phi_calF}
\begin{split}
t^2\zeta^{}_{\mathcal{F}}\biggl(\frac{1}{1-uty+t^2yx}yx\frac{1}{1-vty+t^2yx}\biggr)&=1-\exp\biggl(\sum_{2\le n}\frac{Z_{\mathcal{F}}(n)}{n}q_n(u,v)t^n\biggr).
\end{split}
\end{align}

Let $*$ denote the harmonic product. A calculation similar to that in the proof of Lemma \ref{lem:truncated} gives
\begin{align*}
&\zeta^{}_{N}\left(\frac 1{1-uty+yxt^2}yx\frac 1{1-vty+yxt^2}\right)\\
&=\sum_{n=1}^{N-1}\left(\prod_{k=1}^{n-1}\left(1+\frac{ut}{k}-\frac{t^2}{k^2}\right)\right)\frac 1{n^2}\prod_{k=n+1}^{N-1}\left(1+\frac{vt}{k}-\frac{t^2}{k}\right)\\
&=\left(\prod_{n=1}^{N-1}\left(1-\frac{t^2}{n^2}\right)\right)\sum_{n=1}^{N-1}\left(\prod_{k=1}^{n-1}\left(1+\frac{utk}{k^2-t^2}\right)\right)\frac 1{n^2-t^2}\prod_{k=n+1}^{N-1}\left(1+\frac{vtk}{k^2-t^2}\right)\\
&=\zeta^{}_{N}\left(\frac 1{1+yxt^2}\right)\zeta^{}_{N}\left(\frac 1{1-uty\frac 1{1-x^2t^2}}yx\frac 1{1-x^2t^2}\frac 1{1-vty\frac 1{1-x^2t^2}}\right)\\
&=\zeta^{}_{N}\left(\frac 1{1+yxt^2}*\frac 1{1-uty\frac 1{1-x^2t^2}}yx\frac 1{1-x^2t^2}\frac 1{1-vty\frac 1{1-x^2t^2}}\right).
\end{align*}
Here, the last equality follows from the harmonic relation for finite multiple harmonic sums. By the linear independence of multiple harmonic sums (see Yamamoto \cite[Theorem 3.1]{y}), we have
\begin{align*}
\frac 1{1-uty+yxt^2}yx\frac 1{1-vty+yxt^2}=\frac 1{1+yxt^2}*\frac 1{1-uty\frac 1{1-x^2t^2}}yx\frac 1{1-x^2t^2}\frac 1{1-vty\frac 1{1-x^2t^2}}.
\end{align*}


Since $\zeta^{}_{\mathcal{F}}(k)=0$ for every positive integer $k$, the harmonic relation  (see Kaneko \cite[(8.3) and Proposition 9.2]{k2}) gives
\[
\zeta^{}_{\mathcal{F}}\left(\frac 1{1+yxt^2}\right)=1.
\]
Hence, we obtain
\begin{align*}
&t^2\zeta^{}_{\mathcal{F}}\left(\frac 1{1-uty+yxt^2}yx\frac 1{1-vty+yxt^2}\right)\\
&=t^2\zeta^{}_{\mathcal{F}}\left(\frac 1{1-uty\frac 1{1-x^2t^2}}yx\frac 1{1-x^2t^2}\frac 1{1-vty\frac 1{1-x^2t^2}}\right)\\
&=\sum_{1\leq i\leq r}u^{i-1}v^{r-i}t^{r+1}\zeta^{}_{\mathcal{F}}\left(\left(y\frac 1{1-x^2t^2}\right)^{i-1}yx\frac 1{1-x^2t^2}\left(y\frac 1{1-x^2t^2}\right)^{r-i}\right)\\
&=\sum_{\substack{1\leq i\leq r<k\\r\not\equiv k\,\mathrm{mod}\,2}}u^{i-1}v^{r-i}t^{k}\sum_{\substack{1\leq k_1,\dots,k_r\\k_1,\dots,k_{i-1},k_{i+1},\dots,k_r:\mathrm{odd}\\k_i:\mathrm{even}\\k_1+\cdots+k_r=k}}\zeta^{}_{\mathcal{F}}(k_1,\dots,k_r).
\end{align*}
Therefore, from \eqref{eq:Phi_calF}, we obtain the desired formula.

\end{align*}
hold.

<!-- lem-3: body_md -->
\label{lem:key}
The following identity holds in $\mathfrak{H}[[t]]$:
\[
\left(\frac 1{1-yx\frac{t^2}{1-xt}}\mathbin{\mathcyr{sh}} \frac 1{1-yt}\right)*\frac 1{1+yt}=\frac 1{1-y^2xt^3}.
\]
<!-- end lem-3 -->
<!-- proof-7: body_md -->
By Lemma \ref{lem:ikz}, we obtain
\begin{align*}
\left(\frac 1{1-yx\frac{t^2}{1-xt}}\mathbin{\mathcyr{sh}}\frac 1{1-yt}\right)*\frac 1{1+yt}&=\Phi_{-t}^*\left(\Phi^{\mathbin{\mathcyr{sh}}}_t\left(\frac 1{1-y\frac{t^2}{1-xt}x}\right)\frac 1{1-yt}\right)\frac 1{1+yt}\\
&=\Phi_{-t}^*\left(\frac 1{1-\frac 1{1-yt}y\frac{t^2}{1-\frac 1{1-yt}xt}\frac 1{1-yt}x}\frac 1{1-yt}\right)\frac 1{1+yt}\\
&=\Phi_{-t}^*\left(\frac 1{1-yt-y\frac{t^2}{1-(x+y)t}x}\right)\frac 1{1+yt}\\
&=\frac 1{1-\frac{y}{1+yt}(1-xt)t-\frac{y}{1+yt}(1-xt)\frac{t^2}{1-\frac 1{1+yt}(x+y)t}x}\frac 1{1+yt}\\
&=\frac 1{1+yt-y(1-xt)t-y(1+yt)xt^2}\\
&=\frac 1{1-y^2xt^3}.
\end{align*}
This completes the proof.

<!-- proof-8: body_md -->
Using Lemma \ref{lem:key} and the shuffle and harmonic relations for finite multiple zeta values (see Kaneko \cite[(8.3) and Theorem 8.1]{k2}), and for symmetric multiple zeta values (see Kaneko \cite[Proposition 9.2 and Theorem 9.6]{k2}), we obtain
\begin{align*}
\sum_{0\leq n}\zeta^{}_{\mathcal{F}}(\{1,2\}^n)t^{3n}&=\zeta^{}_{\mathcal{F}}\left(\frac 1{1-y^2xt^3}\right)\\
&=\zeta^{}_{\mathcal{F}}\left(\left(\frac 1{1-yx\frac{t^2}{1-xt}}\mathbin{\mathcyr{sh}} \frac 1{1-yt}\right)*\frac 1{1+yt}\right)\\
&=\zeta^{}_{\mathcal{F}}\left(\frac 1{1-yx\frac{t^2}{1-xt}}\mathbin{\mathcyr{sh}} \frac 1{1-yt}\right)\\
&=\zeta^{}_{\mathcal{F}}\left(\frac 1{1-yx\frac{t^2}{1-xt}}\frac 1{1+yt}\right)\\
&=1+\zeta^{}_{\mathcal{F}}\left(\frac 1{1-yx\frac{t^2}{1-xt}}yx\frac{t^2}{1-xt}\frac 1{1+yt}\right).
\end{align*}

On the other hand, Lemma \ref{lem:truncated} gives
\begin{align*}
&\zeta^{}_{N}\left(\frac 1{1-yx\frac{t^2}{1-xt}}yx\frac{t^2}{1-xt}\frac 1{1+yt}\right)\\
&=t^2\sum_{0\leq m,r}(-1)^rt^{2m+r}\zeta^{}_{N}\left(\left(yx\frac{1}{1-xt}\right)^{m+1}y^r\right)\\
&=t^2\sum_{0\leq m,r}(-1)^rt^{2m+r}\sum_{0<j_1<\cdots<j_m<n<l_1<\cdots<l_r<N}\frac 1{j_1(j_1-t)}\cdots\frac 1{j_m(j_m-t)}\frac 1{n(n-t)}\frac 1{l_1\cdots l_r}\\
&=t^2\sum_{0<n<N}\left(\prod_{j=1}^{n-1}\left(1+\frac{t^2}{j(j-t)}\right)\right)\frac 1{n(n-t)}\prod_{j=n+1}^{N-1}\left(1-\frac tj\right)\\
&=\sum_{0<n<N}\frac{(\omega t,\omega^2 t)_{n}}{(1-t)_n^2}\frac{(1-t)_{N-1}}{(N-1)!}\\
&=t^2\zeta^{}_{N}\left(\frac 1{1+yt-yxt^2}yx\frac 1{1+2yt-yxt^2}\right)\zeta^{}_{N}\left(\frac 1{1+yt}\right)^{-1},
\end{align*}
where $\omega\coloneqq e^{\frac{2\pi i}3}$. Hence, we have
\[
\zeta^{}_{N}\left(\frac 1{1-yx\frac{t^2}{1-xt}}yx\frac{t^2}{1-xt}\frac 1{1+yt}*\frac 1{1+yt}\right)=t^2\zeta^{}_{N}\left(\frac 1{1+yt-yxt^2}yx\frac 1{1+2yt-yxt^2}\right).
\]

By the linear independence of multiple harmonic sums (see Yamamoto \cite[Theorem 3.1]{y}), we have
\[
\frac 1{1-yx\frac{t^2}{1-xt}}yx\frac{t^2}{1-xt}\frac 1{1+yt}*\frac 1{1+yt}=t^2\frac 1{1+yt-yxt^2}yx\frac 1{1+2yt-yxt^2}.
\]
Hence, 
\begin{align*}\label{eq:toKaneko}
1+\zeta^{}_{\mathcal{F}}\left(\frac 1{1-yx\frac{t^2}{1-xt}}yx\frac{t^2}{1-xt}\frac 1{1+yt}\right)&=1+t^2\zeta^{}_{\mathcal{F}}\left(\frac 1{1+yt-yxt^2}yx\frac 1{1+2yt-yxt^2}\right).
\end{align*}
Put $a=\omega t, b=\omega^2t, c=d=t$ in Theorem \ref{thm:main}. 
We have
\begin{align*}
&1+t^2\zeta^{}_{\mathcal{F}}\left(\frac 1{1+yt-yxt^2}yx\frac 1{1+2yt-yxt^2}\right)\\
&=\exp\biggl(3\sum_{2\le n}\frac{Z_{\mathcal{F}}(n)}{n}(\omega^{2n}+\omega^n+1)t^{n}\biggr)\\
&=\exp\biggl(3\sum_{0<n}\frac{Z_{\mathcal{F}}(3n)}{n}t^{3n}\biggr).
\end{align*}
Note that for the case $\mathcal{F}=\mathcal{A}$, we consider the identity above in the ring $(\mathcal{Z}_{\mathcal{A}}\otimes_{\mathbb{Q}}\mathbb{Q}(\omega))[[t]]$.

Kaneko \cite[Example 7.4 and 9.4]{k2} gives
\[
\zeta^{}_{\mathcal{F}}(1,3n-1)=3nZ_{\mathcal{F}}(3n). \label{eq:FDZV}
\]
Therefore, replacing $t^3$ by $t$, we obtain Theorem \ref{thm:one-two}.
