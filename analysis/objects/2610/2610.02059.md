<!-- thm-1: body_md -->
\label{theo1}
Let $\rho, \mu$ and $\nu$ be arbitrary functions in  $C^1(0,\infty)$.
For the system \eqref{eq: Main_system}, there exists a class of entropy/entropy-flux pairs $( \eta, \mathcal{F} )$ which is 
given by
\begin{equation}\label{eq:entropyclass}
\begin{array}{rcl}
 \eta[f, b, g, q]&=& \rho(fb)+\sqrt{fb}\mu\left(\dfrac{b}{\sqrt{fb}}\right)+\sqrt{gq}\nu\left(\dfrac{q}{\sqrt{gq}}\right)+\dfrac{1}{fb+gq},\\[3ex]
 \mathcal{F}[f, b, g, q]&=&\psi(fb)+(fb)^{3/2} \mu\left(\dfrac{b}{\sqrt{fb}}\right)+(\sqrt{gq}) \nu\left(\dfrac{q}{\sqrt{gq}}\right)(fb+gq)\\
&&\hspace{3.5 cm}-\ln \left((fb+gq)^{3/2}\right)-\dfrac{fb}{2(fb+gq)}.
\end{array}
\end{equation}
The function $\psi$ is determined from $\psi'(w)=\dfrac{3}{2}  w\rho'(w)$.
<!-- end thm-1 -->
<!-- lemma-1: body_md -->
For any $\mathbf{U}\in \mathcal{U}$, the following pair of functions form an entropy-entropy pair:
\begin{equation}\label{Eq:Ent-EntFl pair}
\eta= \frac{1}{|fb|}+ \frac{1}{fb+gq},\qquad \mathcal{F}= -\frac{1}{2}\left(\frac{fb}{fb+gq} +3\ln(|fb|^{\sigma(b)})+3\ln(fb+gq)\right),             
\end{equation}
where $\sigma(b)$ is the sign of $b$ and $\eta$ is a strictly convex function for $\mathbf{U}\in \mathcal{U}$. 
In particular, for smooth solutions of the hyperbolic system \eqref{eq: Main_system}, we have 
\begin{align}
    \frac{\partial}{\partial t}\eta+\frac{\partial}{\partial x}\mathcal{F}=0.
\end{align}

<!-- proof-1: body_md -->
By choosing $\rho(u)=1/|u|$, $\mu=0$, $\nu=0$, it is easy to see that the function $\eta$ belongs to the general class of entropies of the system \eqref{eq: Main_system} along with the associated flux $\mathcal{F} $ as defined in \eqref{eq:entropyclass}. Therefore, we directly proceed to prove that the entropy $\eta$ is strictly convex for all $\mathbf{U}\in \mathcal{U}$. We separate the proof of convexity for $\eta$ into two different cases depending upon the sign of $b$.\\\\
\textbf{Case 1. $\mathbf{b>0}$}\\
For this case, we directly compute the Hessian matrix of $\eta$, which gives
    \begin{align*}
   H_{\eta}&= \left(\begin{array}{cccc}2\left(\frac{b^2}{(bf+gq)^3}+\frac{1}{f^3 b}\right)&\frac{bf-gq}{(bf+gq)^3}+\frac{1}{b^2f^2}&\frac{2bq}{(bf+gq)^3}&\frac{2bg}{(bf+gq)^3}\\ \frac{bf-gq}{(bf+gq)^3}+\frac{1}{b^2f^2}&2\left(\frac{f^2}{(bf+gq)^3}+\frac{1}{fb^3}\right)&\frac{2fq}{(bf+gq)^3}&\frac{2fg}{(bf+gq)^3}\\\frac{2bq}{(bf+gq)^3}&\frac{2fq}{(bf+gq)^3}&\frac{2q^2}{(bf+gq)^3}&\frac{gq-bf}{(bf+gq)^3}\\\frac{2bg}{(bf+gq)^3}&\frac{2fg}{(bf+gq)^3}&\frac{gq-bf}{(bf+gq)^3}&\frac{2g^2}{(bf+gq)^3}\end{array}\right),\\
   \end{align*}
Thus we have
\begin{align*}
   \mathbf{r_1}^\top H_{\eta}\mathbf{r_1}&= \frac{2f}{b(bf+gq)^2}+\frac{2}{b^3f},\\
   \mathbf{r_2}^\top H_{\eta}\mathbf{r_2}&=\frac{3(3gq-bf)\left(3g^2q^2+2bf(gq-fb)\right)}{8b^3f^3q^2(bf+gq)},\\
   \mathbf{r_3}^\top H_{\eta}\mathbf{r_3}&=\frac{2g}{q(bf+gq)^2},\\
   \mathbf{r_4}^\top H_{\eta}\mathbf{r_4}&=\frac{2g(3gq-bf)}{q(bf+gq)^3},
\end{align*}
   which are all positive for $\mathbf{U}\in \mathcal{U}$.\\\\
 \textbf{Case 2. $\mathbf{b<0}$}\\
For this case, we directly compute the Hessian matrix of $\eta$, which gives
   \begin{align*}
   \left(\begin{array}{cccc}
   \frac{2b^2}{(bf+gq)^3}-\frac{2}{bf^3}&\frac{bf-gq}{(bf+gq)^3}-\frac{1}{b^2f^2}&\frac{2bq}{(bf+gq)^3}&\frac{2bg}{(bf+gq)^3}\\ \frac{bf-gq}{(bf+gq)^3}-\frac{1}{b^2f^2}&\frac{2f^2}{(bf+gq)^3}-\frac{2}{b^3f}&\frac{2fq}{(bf+gq)^3}&\frac{2fg}{(bf+gq)^3}\\\frac{2bq}{(bf+gq)^3}&\frac{2fq}{(bf+gq)^3}&\frac{2q^2}{(bf+gq)^3}&\frac{gq-bf}{(bf+gq)^3}\\\frac{2bg}{(bf+gq)^3}&\frac{2fg}{(bf+gq)^3}&\frac{gq-bf}{(bf+gq)^3}&\frac{2g^2}{(bf+gq)^3}\\\end{array}\right),\\
    \end{align*}
Thus we have
\begin{align*}
   \mathbf{r_1}^\top H_{\eta}\mathbf{r_1}&= -\frac{2gq(2fb+gq)}{bf(bf+gq)^2},\\
   \mathbf{r_2}^\top H_{\eta}\mathbf{r_2}&=-\frac{3g(3gq-bf)(2bf+3gq)}{8b^3f^3q(bf+gq)},\\
   \mathbf{r_3}^\top H_{\eta}\mathbf{r_3}&=\frac{2g}{q(bf+gq)^2},\\
   \mathbf{r_4}^\top H_{\eta}\mathbf{r_4}&=\frac{2g(3gq-bf)}{q(bf+gq)^3},
\end{align*}
   which are all positive for $\mathbf{U}\in \mathcal{U}$.\\

<!-- thm-3: body_md -->
\label{thm:entropy-conservative-fd}
Assume that the numerical flux
$
\textbf{F}^{\mathrm{num}}_{i+\frac12}
=
\textbf{F}^{\mathrm{num}}(\textbf{U}_i,\textbf{U}_{i+1})
$
is consistent with the physical flux $F(\textbf{U})$ and satisfies Tadmor's entropy-conservation condition
$
\left[\!\left[\textbf{V}\right]\!\right]_{i+\frac12}
\cdot
\textbf{F}^{\mathrm{num}}_{i+\frac12}
=
\left[\!\left[\Psi\right]\!\right]_{i+\frac12}.
$
Then the semi-discrete FD scheme
\eqref{eq_semi} is entropy conservative, i.e.,
\begin{equation}\label{eq:entropy-equality-fd}
\frac{\mathrm d}{\mathrm dt}\eta(\textbf{U}_i)
+
\frac{
\widehat{\mathcal{F}}_{i+\frac12}
-
\widehat{\mathcal{F}}_{i-\frac12}
}{\Delta x}
=0,
\end{equation}
where the numerical entropy flux is defined by
\begin{equation}\label{eq:numerical-entropy-flux-fd}
\widehat{\mathcal{F}}_{i+\frac12}
=
\left\{\!\!\left\{\textbf{V}\right\}\!\!\right\}_{i+\frac12}
\cdot
\textbf{F}^{\mathrm{num}}_{i+\frac12}
-
\left\{\!\!\left\{\Psi\right\}\!\!\right\}_{i+\frac12}.
\end{equation}
