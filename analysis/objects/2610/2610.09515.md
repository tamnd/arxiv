<!-- theorem-1: body_md -->
For any $1<p<\infty$ we have the inequality
$$ \|M\|_{L^p(w)\to L^p(w)}\leq \frac{p}{p-1-d(p,[w]_{A_p})}.$$
The constant on the right is the best possible: for any $c\geq 1$ and $\varepsilon>0$ there is an $A_p$ weight $w$ satisfying $[w]_{A_p}\leq c$ such that
$$ \|M\|_{L^p(w)\to L^p(w)}\geq \frac{p}{p-1-d(p,c)}-\varepsilon.$$

<!-- definition-1: body_md -->
Suppose that  $(X,\mu)$ is a nonatomic probability space. 
A set $\mathcal{T}$ of measurable subsets of $X$ will be called a tree if the following conditions are satisfied:
\begin{itemize}
\item[(i)] $X\in \mathcal{T}$ and for every $Q\in\mathcal{T}$ we have $\mu(Q)>0$.

\smallskip

\item[(ii)] For every $Q \in \mathcal{T}$ there is a  finite subset $C(Q) \subset \mathcal{T}$ containing at least two elements such that

(a) the elements of $C(Q)$ are pairwise disjoint subsets of $Q$,

(b) $ Q = \bigcup_{Q'\in C(Q)}Q'$.

\smallskip

\item[(iii)] $\mathcal{T} = \bigcup_{n\geq 0} \mathcal{T}^n$, where $\mathcal{T}^0=\{X\}$ and $\mathcal T^{n+1} = \bigcup_{Q\in\mathcal{T}^n} C(Q)$.

\smallskip

\item[(iv)] We have $\lim_{n\to\infty}\sup_{Q\in\mathcal{T}^n} \mu(Q) = 0$. 
\end{itemize}

<!-- theorem-2: body_md -->
\label{mainthm}
Let $(X,\mu)$ be an arbitrary nonatomic probability space with a tree structure $\mathcal{T}$. Then for any vector $\vec{P}=(p_1,p_2,\ldots,p_m)\in (1,\infty)^m$ with $p^{-1}=p_1^{-1}+p_2^{-1}+\ldots+p_m^{-1}$ and any vector $\vec{w}\in A_{\vec{P}}(X,\mathcal{T})$ we have the inequality
\begin{equation}\label{mainin}
\|\mathcal{M}_{X,\mathcal{T}}\|_{L^{p_1}(w_1)\times L^{p_2}(w_2)\times \ldots \times L^{p_m}(w_m)\to L^p(v_{\vec{w}})}\leq  \prod_{j=1}^m \frac{p_j}{p_j-1-\alpha(p,[\vec{w}]_{A_{\vec{P}}})}.
\end{equation}
The constant on the right is the best possible: for any $c\geq 1$ and $\varepsilon>0$ there is a weight $\vec{w}$ satisfying $[\vec{w}]_{A_{\vec{P}}}\leq c$ such that
\begin{equation}\label{rever}
 \|\mathcal{M}_{X,\mathcal{T}}\|_{L^{p_1}(w_1)\times L^{p_2}(w_2)\times \ldots \times L^{p_m}(w_m)\to L^p(v_{\vec{w}})}>\prod_{j=1}^m \frac{p_j}{p_j-1-\alpha(p,c)}-\varepsilon.
 \end{equation}

<!-- theorem-3: body_md -->
\label{Hardythm}
Let $\vec{P}=(p_1,p_2,\ldots,p_m)\in (1,\infty)^m$ and $p^{-1}=p_1^{-1}+p_2^{-1}+\ldots+p_m^{-1}$. Assume further that  $\vec{w}=(w_1,w_2,\ldots,w_m)$ is a vector of positive, locally integrable functions on $(0,\infty)$. Suppose that the function $v_{\vec{w}}=\prod_{j=1}^m w_j^{p/p_j}$ satisfies
$$ [\vec{w}]_{A_{\vec{P}}}:=\sup \langle v_{\vec{w}}\rangle_I \prod_{j=1}^m \langle w_j^{1-p_j'}\rangle_I^{p/p_j'}<\infty,$$
where the supremum is taken over all intervals  $I=[0,t]$, $t\in \mathbb{R}$. Then for any vector $\vec{f}=(f_1,f_2,\ldots,f_m)$ with $f_j\in L^{p_j}(w_j)$, we have
\begin{equation}\label{Hardy}
\begin{split}
\left(\int_0^\infty \prod_{j=1}^m \left(\int_0^t |f_j(s)|\mbox{d}s\right)^p\frac{v_{\vec{w}}(t)}{t^{mp}}\mbox{d}t\right)^{1/p} & \leq \prod_{j=1}^m \frac{p_j\|f_j\|_{L^{p_j}(w_j)}}{p_j-1-\alpha(p,[\vec{w}]_{A_{\vec{P}}})}.
\end{split}
\end{equation}
The constant $\prod_{j=1}^m p_j/(p_j-1-\alpha(p,[\vec{w}]_{A_{\vec{P}}}))$ is the best possible.

<!-- lemma-1: body_md -->
\label{propB1}
For any $s=(s_1,s_2,\ldots,s_m)\in (0,\infty)^m$ and any $u\geq s^{-p/\underline{p}'}$, we have
\begin{equation}\label{maj}
 B(s;t,u)\geq t^ps^{-p/\underline{p}'}-C^ps^{p/\underline{p}}.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Clearly, we have $\partial_u B\geq 0$ (see the formula for $B$). Therefore, all we need is the estimate
$$  B(s;t,s^{-p/\underline{p}'})\geq t^ps^{-p/\underline{p}'}-C^ps^{p/\underline{p}}.
$$
To this end, by the very definition of $B$, we must show that
$$ \frac{t^ps^{-p/\underline{p}'}}{\alpha}+(s^*)^{p/\underline{p}}\beta\left(\frac{s_1}{s_1^*},\frac{s_2}{s_2^*},\ldots,\frac{s_m}{s_m^*}\right)+C^p s^{p/\underline{p}}\geq 0$$
for any $s_1^*,\,s_2^*,\,\ldots,\,s_m^*>0$ with $C s_1^*s_2^*\ldots s_m^*=t$. Let us divide throughout by $mpC^p(s^*)^{p/\underline{p}}/\alpha$ and substitute $r_j=s_j/s_j^*$ to obtain the equivalent form
$$ \frac{r^{-p/\underline{p}'}}{pm}+\sum_{j=1}^m \frac{\alpha+1-\alpha p_j'}{mp_j'} r_j+\frac{\alpha}{pm}r^{p/\underline{p}}\geq 1.$$
This bound follows immediately from Young's inequality, because of the two identities
$$ \frac{1}{pm}+\sum_{j=1}^m \frac{\alpha+1-\alpha p_j'}{mp_j'} +\frac{\alpha}{pm}=1$$
and
$$ \big(r^{-p/\underline{p}'}\big)^{1/pm}\cdot \prod_{j=1}^m r_j^{(\alpha+1-\alpha p_j')/mp_j'} \cdot \big(r^{p/\underline{p}}\big)^{\alpha/pm}=1.$$
The proof is complete.

<!-- lemma-2: body_md -->
\label{propB2}
Suppose that $(s;t,u)\in (0,\infty)^m\times (0,\infty)\times (0,\infty)$ satisfies $ s_1s_2\ldots s_m\geq t$ and $ u\leq cs^{-p/\underline{p}'}$. Then
\begin{equation}\label{monoto}
 B(s;t,u)\geq B(s;s_1s_2\ldots s_m,u).
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
By the very definition of $B$, we have
\begin{equation}\label{partial}
\partial_u B(s;t,u)=(\alpha+1)t^p/\alpha\leq (\alpha+1)(s_1s_2\ldots s_m)^p/\alpha=\partial_u B(s;s_1s_2\ldots s_m,u),
\end{equation}
and hence it is enough to establish \eqref{monoto} for the largest value of $u$, i.e., for $u=cs^{-p/\underline{p}'}$. First we handle the right-hand side: we have
\begin{equation}\label{zero}
 B(s;s_1s_2\ldots s_m,cs^{-p/\underline{p}'})\leq 0.
\end{equation}
To see this, we pick $s^*_j=s_j(1+\alpha(1-p_j'))$, $j=1,\,\,2,\,\ldots,\,m$. Then
$$ Cs_1^*s_2^*\ldots s_m^*=Cs_1s_2\ldots s_m\cdot \prod_{j=1}^m(1+\alpha(1-p_j'))=s_1s_2\ldots s_m,$$
so by the very definition of $B$,
$$ B(s;s_1s_2\ldots s_m,cs^{-p/\underline{p}'})\leq \frac{\alpha+1}{\alpha}(s_1s_2\ldots s_m)^p\cdot cs^{-p/\underline{p}'} +(s^*)^{p/\underline{p}}\beta\left(\frac{s_1}{s_1^*},\frac{s_2}{s_2^*},\ldots, \frac{s_m}{s_m^*}\right).$$
However, by \eqref{defalfa} and \eqref{defCe} we compute that
$$ (s^*)^{p/\underline{p}}=s^{p/\underline{p}}\prod_{j=1}^m \frac{(1+\alpha(1-p_j'))^p}{(1+\alpha(1-p_j'))^{p/p_j'}}=s^{p/\underline{p}}\cdot \frac{c(\alpha+1)}{C^p}$$
and
\begin{align*}
 \beta\left(\frac{s_1}{s_1^*},\frac{s_2}{s_2^*},\ldots, \frac{s_m}{s_m^*}\right)&=pC^p\left[\sum_{j=1}^m \left(\frac{\alpha+1}{\alpha p_j'}-1\right)\frac{1}{1+\alpha(1-p_j')}-\frac{m}{\alpha}\right]\\
 &=\frac{pC^p}{\alpha}\left(\sum_{j=1}^m \frac{1}{p_j'}-m\right)=-\frac{C^p}{\alpha}.
\end{align*}
Combining the above observations, we obtain \eqref{zero}; thus, to complete the proof, it is enough to prove that
$ B(s;t,cs^{-p/\underline{p}'})\geq 0.$ By the definition of $B$, we need to show the estimate
$$ \frac{\alpha+1}{\alpha}t^p\cdot cs^{-p/\underline{p}'}+(s^*)^{p/\underline{p}}\beta\left(\frac{s_1}{s_1^*},\frac{s_2}{s_2^*},\ldots, \frac{s_m}{s_m^*}\right)\geq 0$$
for any sequence $s_1^*$, $s_2^*$, $\ldots$, $s_m^*$ of positive numbers satisfying $Cs_1^*s_2^*\ldots s_m^*=t$. Recall that by \eqref{defalfa},
$$ c(\alpha+1)=\prod_{j=1}^m (1+\alpha(1-p_j'))^{-p/p_j'}.$$
The substitution $r_j=(1+\alpha(1-p_j'))s_j/s_j^*$ transforms the desired bound into
$$ \frac{r^{-p/\underline{p}'}}{pm}+\sum_{j=1}^m \frac{r_j}{mp_j'}\geq 1,$$
which follows directly from Young's inequality.

<!-- remark-1: body_md -->
The inequality \eqref{zero} and the bound $\partial_u B\geq 0$ (see the formula for $B$) imply that for any $s_1,s_2,\ldots,s_m>0$ and any $ s^{-p/\underline{p}'}\leq u\leq cs^{-p/\underline{p}'}$ we have
\begin{equation}\label{init}
B(s_1,s_2,\ldots,s_m;s_1s_2\ldots s_m,u)\leq 0.
\end{equation}

<!-- theorem-4: body_md -->
Suppose that $(X,\mu)$ is a probability space with a tree structure $\mathcal{T}$.
Fix $c\geq 1$ and let $\vec{w}$ be a vector of weights on $X$, satisfying $[\vec{w}]_{A_{\vec{P}}}\leq c$. Then for any $R\in \mathcal{T}$, the vector $\vec{\sigma}=(\sigma_1,\sigma_2,\ldots,\sigma_m)$ of dual weights satisfies
\begin{equation}\label{testing}
 \int_R \mathcal{M}(\vec{\sigma}\chi_R)^p v_{\vec{w}}\mbox{d}\mu\leq C^p_{\vec{P},c}\int_R \prod_{j=1}^m \sigma_j^{p/p_j}\mbox{d}\mu .
\end{equation}
<!-- end theorem-4 -->
<!-- proof-3: body_md -->
It is convenient to split the reasoning into a few intermediate parts.

\smallskip

\emph{Step 1. An auxiliary notation.}
Fix $w$ and $R$ as in the statement and let $n_0$ be the unique integer such that $R\in \mathcal{T}^{n_0}$. We introduce the auxiliary functional sequences $(x_n)_{n\geq n_0}$, $(y_n)_{n\geq n_0}$ and $(z_n)_{n\geq n_0}$ as follows: for any $n\geq n_0$ and $\omega\in R$,
\begin{align*}
 x_n(\omega)&=(x_{n1}(\omega),x_{n2}(\omega),\ldots,x_{nm}(\omega))=\big(\langle \sigma_1\rangle_{Q_n(\omega)},\langle \sigma_2\rangle_{Q_n(\omega)},\ldots,\langle \sigma_m\rangle_{Q_n(\omega)}\big)\\
 y_n(\omega)&=\max_{n_0\leq k\leq n}\Big(x_{k1}(\omega)x_{k2}(\omega)\ldots x_{km}(\omega)\Big),\\
z_n(\omega)&=\langle v_{\vec{w}}\rangle_{Q_n(\omega)},
\end{align*}
where $Q_n(\omega)$ is the unique element of $\mathcal{T}^n$ which contains $\omega$. There is a nice probabilistic interpretation of these sequences: $(x_n)_{n\geq n_0}$ and $(z_n)_{n\geq n_0}$ are martingales induced by the filtration $(\mathcal{T}^n)_{n\geq n_0}$ , with the terminal variables equal to $(\sigma_1,\sigma_2,\ldots,\sigma_m)$ and $v_{\vec{w}}$, respectively; in addition, $(y_n)_{n\geq n_0}$ is the maximal process associated with the sequence of products $(x_{k1}x_{k2}\ldots x_{km})_{k\geq n_0}$. Observe that by Lebesgue's differentiation theorem (or rather, by Doob's martingale convergence theorem), we have $x_{nj}\to \sigma_j$, $y_n\to \mathcal{M}(\vec{\sigma}\chi_R)$ and $z_n\to v_{\vec{w}}$ almost surely as $n\to \infty$.

\smallskip

\emph{Step 2. Monotonicity.} Now we will show that the sequences $(x_n)_{n\geq n_0}$, $(y_n)_{n\geq n_0}$ and $(z_n)_{n\geq n_0}$ combine nicely with the Bellman function $B$ defined previously. More specifically, we will prove that the sequence
$$ \left(\int_R B(x_n;y_n,z_n)\mbox{d}\mu\right)_{n\geq n_0}$$
is nonincreasing. To see this, fix an integer $n\geq n_0$, let $Q$ be an arbitrary element of $\mathcal{T}^n$ and let $Q_1$, $Q_2$, $\ldots$, $Q_\ell$ be the collection of all children of $Q$ in $\mathcal{T}^{n+1}$. The functions $x_n$, $y_n$, $z_n$ are constant on $Q$, while $x_{n+1}$, $y_{n+1}$ and $z_{n+1}$  are constant on each $Q_j$. It is easy to check that these constant values satisfy
\begin{equation}\label{constan}
 \langle x_n\rangle_Q=\sum_{j=1}^\ell \frac{|Q_j|}{|Q|} \langle x_{n+1}\rangle_{Q_j},\qquad \langle z_n\rangle_Q =\sum_{j=1}^\ell \frac{|Q_j|}{|Q|}\langle z_{n+1}\rangle_{Q_{j}}.
\end{equation}
(This is nothing but the martingale property, expressed in analytic terms).  Next, observe that for any $j$ we have
\begin{equation}\label{interm}
 \int_{Q_j} B(x_{n+1};y_{n+1},z_{n+1})\mbox{d}\mu\leq \int_{Q_j} B(x_{n+1};y_n,z_{n+1})\mbox{d}\mu.
\end{equation}
Indeed, if $y_{n+1}=y_n$ on $Q_j$, then there is nothing to prove; on the other hand, if $y_{n+1}\neq y_n$ on $Q_j$, then by the definition of the sequence $y$ we must have $y_n<y_{n+1}=x_{(n+1)1}x_{(n+1)2}\ldots x_{(n+1)m}$. Furthermore, by the $A_p$ condition we have
$$ z_{n+1}(\omega)=\langle v_{\vec{w}}\rangle_{Q_{n+1}(\omega)}\leq c\prod_{j=1}^m \langle \sigma_j\rangle_{Q_{n+1}(\omega)}^{-p/p_j'}=c x_{(n+1)1}^{-p/p_1'}(\omega)x_{(n+1)2}^{-p/p_2'}(\omega)\ldots x_{(n+1)m}^{-p/p_m'}(\omega). $$
By Lemma \ref{propB2}, this implies $B(x_{n+1};y_{n+1},z_{n+1})\leq  B(x_{n+1};y_n,z_{n+1})$ on $Q_j$ and hence \eqref{interm} follows. Summing over $j$, we thus obtain
$$  \int_{Q} B(x_{n+1};y_{n+1},z_{n+1})\mbox{d}\mu\leq \int_{Q} B(x_{n+1};y_n,z_{n+1})\mbox{d}\mu.$$
Now, for a fixed $t$, the function $(s,u)\to B(s;t,u)$, being the infimum of affine functions, is concave. Therefore, by \eqref{constan}, the right-hand side above does not exceed  $ \int_{Q} B(x_{n};y_n,z_{n})\mbox{d}\mu.$
Summing over all $Q\in \mathcal{T}^n$ contained in $R$, we obtain the aforementioned monotonicity.

\smallskip

\emph{Step 3. Completion of the proof.} By the previous step, we obtain that for any $n\geq n_0$ we have
\begin{equation}\label{land}
 \int_R B(x_{n};y_n,z_n)\mbox{d}\mu\leq \int_R B(x_{n_0};y_{n_0},z_{n_0})\mbox{d}\mu.
\end{equation}
Let us inspect the expression on the right. By the very definition of $y$, we have $y_{n_0}=x_{n_01}x_{n_02}\ldots x_{n_0m}$; furthermore,  by the $A_{\vec{P}}$ condition, we obtain $z_{n_0}\leq c\prod_{j=1}^m x_{n_0j}^{-p/p_j'}$. Consequently, we have $
 B(x_{n_0};y_{n_0},z_{n_0})\leq 0$, by virtue of \eqref{init}.
The left-hand side of \eqref{land} is handled by Lemma \ref{propB1}: we have
$$ B(x_{n};y_n,z_n)\geq y_n^px_{n1}^{-p/p_1'}x_{n2}^{-p/p_2'}\ldots x_{nm}^{-p/p_m'}-C^px_{n1}^{p/p_1}x_{n2}^{p/p_2}\ldots x_{nm}^{p/p_m}.$$Putting all these fact together, we get
$$ \int_R y_n^px_{n1}^{-p/p_1'}x_{n2}^{-p/p_2'}\ldots x_{nm}^{-p/p_m'}\mbox{d}\mu\leq C^p \int_R \prod_{j=1}^m x_{nj}^{p/p_j}\mbox{d}\mu.$$
It remains to let $n\to \infty$ and carry out an appropriate limiting procedure. Recall that we have the almost sure convergence $x_{nj}\to \sigma_j$ and $z_n\to v_{\vec{w}}$. Therefore, the left-hand side above can be handled by Fatou's lemma:
$$ \int_R \mathcal{M}(\vec{\sigma}\chi_R)^p v_{\vec{w}}\mbox{d}\mu\leq \liminf_{n\to\infty} \int_R y_n^px_{n1}^{-p/p_1'}x_{n2}^{-p/p_2'}\ldots x_{nm}^{-p/p_m'}\mbox{d}\mu.$$
To deal with the right hand side, we will apply Lebesgue's dominated convergence theorem: we have $\prod_{j=1}^m x_{nj}^{p/p_j}\to \prod_{j=1}^m \sigma_j^{p/p_j}$ almost surely, so all we need is a suitable majorant. The inclusion $\vec{w}\in A_{\vec{P}}$ implies $\sigma_j\in A_{mp_j'}$ (cf. \cite{LOPTT}), which gives that $\sigma_j$, and hence also the maximal function $M\sigma_j$, belongs to $L^r$ for some $r>1$. Thus, by the H\"older inequality, the majorant $\prod_{j=1}^m \Big(\sup_{n\geq n_0}x_{nj}\Big)^{p/p_j}$ is integrable and the assertion follows.

<!-- theorem-5: body_md -->
\label{multilinear-carleson}
Suppose that a sequence $(a_Q)_{Q\in \mathcal{T}}$ of nonnegative numbers satisfies the following condition: for any $R\in \mathcal{T}$ we have the estimate
\begin{equation*}\label{Carle}
 \sum_{Q\subseteq R} a_Q \leq K^p\int_R \prod_{j=1}^m \sigma_j^{p/p_j}\mbox{d}\mu.
\end{equation*}
Then for any vector $\vec{f}=(f_1,f_2,\ldots,f_m)$ such that $f_j\in L^{p_j}(\sigma_j)$ for all $j$, we have
\begin{equation*}\label{innu}
 \left(\sum_{Q\in \mathcal{T}} a_Q \prod_{j=1}^m \langle f_j\rangle_{Q,\sigma_j}^p\right)^{1/p}\leq K\prod_{j=1}^m p_j'\|f_j\|_{L^{p_j}(\sigma_j)},
\end{equation*}
where the notation $\langle \psi \rangle_{Q,\sigma_j}$ denotes the average of $\psi$ over $Q$ with respect to $\sigma_j\mbox{d}\mu$.
<!-- end theorem-5 -->
<!-- proof-4: body_md -->
Fix a vector $\vec{w}\in A_{\vec{P}}(X,\mathcal{T})$ and pick any sequence $\vec{f}=(f_1,f_2,\ldots,f_m)$ of functions on $X$ such that $f_j\in L^{p_j}(w_j)$ for each $j$. Clearly, we may assume that $f_j>0$ for all $j$. Let $\varepsilon>0$ be a fixed parameter.                                                                                                                                                                                                                                                                                                                                                                         
By the definition of the multilinear maximal operator, for any $\omega\in X$ there is a set $\mathcal{Q}(\omega)$ containing $\omega$ such that
$$ \mathcal{M}\vec{f}(\omega)\leq (1+\varepsilon)\langle f_1\rangle_{\mathcal{Q}(\omega)}\langle f_2\rangle_{\mathcal Q(\omega)}\ldots \langle f_m\rangle_{\mathcal Q(\omega)}.$$
Of course, such a set $\mathcal Q(\omega)$ need not be unique: to avoid ambiguity, we take $\mathcal{Q}(\omega)$ belonging to $\mathcal{T}^n$ with $n$ as small as possible. Now, for any $Q\in \mathcal{T}$, we define $ E(Q)=\{\omega\in X\,:\,\mathcal{Q}(\omega)=Q\}$
and let
$$a_Q=\prod_{j=1}^m \langle \sigma_j\rangle_Q^p\cdot v_{\vec{w}}(E(Q)).$$
By the very definition, we have $E(Q)\subseteq Q$ and the sets $E(Q)$ corresponding to different $Q$'s are disjoint. Furthermore, we have $a_Q\leq (\mathcal{M}(\vec{\sigma}\chi_Q)(\omega))^pv_{\vec{w}}(E(Q))$ for $\omega\in Q$.
Consequently, for any $R\in \mathcal{T}$ we get
$$ \sum_{Q\subseteq  R}a_Q\leq \int_R (\mathcal{M}(\vec{\sigma}\chi_Q))^pv_{\vec{w}}\mbox{d}\mu\leq C^p\int_R \prod_{j=1}^m \sigma_j^{p/p_j}\mbox{d}\mu,$$
where the second inequality is due to \eqref{testing}.  Thus, by the Carleson embedding theorem (i.e. Theorem \ref{multilinear-carleson}) applied to the vector $\overrightarrow{f\sigma^{-1}}=(f_1\sigma_1^{-1},f_2\sigma_2^{-1},\ldots,f_m\sigma_m^{-1})$, we obtain
$$ \left(\sum_{Q\in \mathcal{T}} a_Q \prod_{j=1}^m \langle f_j\sigma_j^{-1}\rangle_{Q,\sigma_j}^p\right)^{1/p}\leq C\prod_{j=1}^m p_j'\|f_j\sigma_j^{-1}\|_{L^{p_j}(\sigma_j)}=C\prod_{j=1}^m p_j'\|f_j\|_{L^{p_j}(w_j)}.$$
Now, by the definition of $a_Q$ and the identity $\langle f_j\sigma_j^{-1}\rangle_{Q,\sigma_j}=\langle f_j\rangle_Q\langle \sigma_j\rangle_Q^{-1}$, we get
\begin{align*}
\sum_{Q\in \mathcal{T}} a_Q \prod_{j=1}^m \langle f_j\sigma_j^{-1}\rangle_{Q,\sigma_j}^p&=\sum_{Q\in \mathcal{T}}v_{\vec{w}}(E(Q))\prod_{j=1}^m \langle f_j\rangle_Q^p\geq (1+\varepsilon)^{-1}\int_X (\mathcal{M}\vec{f})^pv_{\vec{w}}\mbox{d}\mu.
\end{align*}
Combining the two estimates above yields the claim, since $\varepsilon$ was arbitrary.

<!-- proof-5: body_md -->
It is enough to show the localized estimate
\begin{equation}\label{local}
 \left(\int_0^T \prod_{j=1}^m \left(\int_0^t |f_j(s)|\mbox{d}s\right)^p\frac{v_{\vec{w}}(t)}{t^{mp}}\mbox{d}t \right)^{1/p} \leq \prod_{j=1}^m \frac{p_j\|f_j\|_{L^{p_j}([0,T],w_j)}}{p_j-1-\alpha(p,[\vec{w}]_{A_{\vec{P}}})}
 \end{equation}
for any $T>0$, and we may restrict ourselves to nonnegative functions $f_j$. This inequality will follow from the estimate \eqref{mainin}, thanks to a discretization argument. Fix $\vec{w}$, $\vec{f}$ as in the statement, let $\delta\in (0,1)$ be a small number and distinguish the points $t_n=(1-\delta)^nT$, $n=0,\,1,\,2,\,\ldots$. Consider the space $X=(0,T]$ equipped with the normalized Lebesgue measure $\mu=|\cdot|/T$. Introduce the tree $\mathcal{T}$ by the following inductive procedure:

\smallskip

(i) We let $\mathcal{T}^0=\{(0,T]\}$;

(ii) If $\mathcal{T}^j=\{(0,t_j],I_1,I_2,\ldots,I_k\}$ for some subintervals $I_1,\,I_2,\,\ldots,\, I_k$, then
$$ \mathcal{T}^{j+1}=\{(0,t_{j+1}],(t_{j+1},t_j],I_{1-},I_{1+},I_{2-},I_{2+},\ldots,I_{k-},I_{k+}\}.$$

\smallskip

\noindent Here $I_-$ and $I_+$ denote the left and the right half of $I$, respectively.
Next, let $\tilde{\sigma}_j,\,\tilde{f_j}:X\to (0,\infty)$ be the conditional expectations of $\sigma_j=w_j^{1-p_j'}$ and $f_j$ with respect to the $\sigma$-algebra generated by $(t_1,t_0]$, $(t_2,t_1]$, $\ldots$, i.e.,
\begin{equation}\label{condexp}
 \tilde{\sigma}_j=\sum_{n=1}^\infty\langle \sigma_j\rangle_{(t_n,t_{n-1}]}\chi_{(t_n,t_{n-1}]},\quad \quad \tilde{f}_{j}=\sum_{n=1}^\infty\langle f_j\rangle_{(t_{n},t_{n-1}]}\chi_{(t_{n},t_{n-1}]}.
\end{equation}
Finally, set $\tilde{w}_j=\tilde \sigma_j^{1-p_j}$. Then the vector weight $\vec{\tilde{w}}=(\tilde{w}_1,\tilde{w}_2,\ldots,\tilde{w}_m)$ satisfies the $A_{\vec{P}}$ condition with respect to the above tree $\mathcal{T}$ and $[\vec{\tilde{w}}]_{A_{\vec{P}}}\leq [\vec{w}]_{A_{\vec{P}}}$. To see this, fix an arbitrary element $Q$ of $\mathcal{T}$. There are two possibilities: either we have $Q=(0,t_n]$ for some $n$, or $Q$ is entirely contained in some interval $(t_{n-1},t_n]$. If the second possibility occurs, then $\vec{\tilde{w}}$ is constant on $Q$ and hence
$$ \langle v_{\vec{\tilde{w}}}\rangle_Q\prod_{j=1}^m \langle \tilde{w}^{1-p_j'}_j\rangle_Q^{p/p_j'}=1.$$
In the case $Q=(0,t_n]$ for some $n$, we note that for any $k\geq 1$ we have
$$  \langle v_{\vec{\tilde{w}}}\rangle_{(t_{k},t_{k-1}]}=\prod_{j=1}^m \langle \tilde{w}^{1-p_j'}_j\rangle_{(t_{k},t_{k-1}]}^{-p/p_j'}=\prod_{j=1}^m \langle \tilde{\sigma}_j\rangle_{(t_{k},t_{k-1}]}^{-p/p_j'}=\prod_{j=1}^m \langle {\sigma}_j\rangle_{(t_{k},t_{k-1}]}^{-p/p_j'}\leq \langle v_{\vec{w}}\rangle_{(t_{k},t_{k-1}]},$$
where the last estimate follows from H\"older's inequality. Summing over all $k\geq n+1$, we obtain $\langle v_{\vec{\tilde{w}}}\rangle_Q\leq \langle v_{\vec{w}}\rangle_Q$ and hence
$$ \langle v_{\vec{\tilde{w}}}\rangle_Q\prod_{j=1}^m \langle \tilde{w}^{1-p_j'}_j\rangle_Q^{p/p_j'}\leq
 \langle v_{\vec{w}}\rangle_Q\prod_{j=1}^m \langle \tilde{\sigma}_j\rangle_Q^{p/p_j'}= \langle v_{\vec{w}}\rangle_Q\prod_{j=1}^m \langle {\sigma}_j\rangle_Q^{p/p_j'}\leq [\vec{w}]_{A_{\vec{P}}}.$$
This establishes the $A_{\vec{P}}$ property and thus we may apply the estimate \eqref{mainin} to $\vec{\tilde{f}}$ and $\vec{\tilde{w}}$.
Note that by the very definition of the multilinear maximal operator, if $t\in (t_{n},t_{n-1}]$, then automatically $t\in (0,t_{n-1}]\in \mathcal{T}$ and hence
$$ \mathcal{M}\vec{\tilde{f}}(t)\geq \prod_{j=1}^m \left(\frac{1}{\mu((0,t_{n-1}])}\int_{(0,t_{n-1}]} \tilde{f_j}\mbox{d}\mu\right)=\frac{1}{t_{n-1}^m}\prod_{j=1}^m \left(\int_0^{t_{n-1}} f_j(s)\mbox{d}s\right).$$
Furthermore, by the convexity of the function $(a,b)\mapsto a^{p_j}b^{1-p_j}$ on $[0,\infty)^2$, we get
$$ \|\tilde{f}_j\|_{L^{p_j}(\tilde{w}_j)}^{p_j}=\int_X \tilde{f}_j^{p_j}\tilde{\sigma}_j^{1-p_j}\mbox{d}\mu\leq \int_X f_j\sigma_j^{1-p_j}\mbox{d}\mu=\frac{1}{T}\|f_j\|_{L^{p_j}([0,T],w_j)}^{p_j}.$$
Therefore, the estimate \eqref{mainin} yields
\begin{align*}
 \left\{\sum_{n=1}^\infty \left[\frac{1}{t_{n-1}^m}\prod_{j=1}^m \left(\int_0^{t_{n-1}}\!\! f_j\right)\right]^p \!\!\!\langle v_{\vec{\tilde{w}}}\rangle_{(t_{n},t_{n-1}]} (t_{n-1}-t_{n})\right\}^{1/p}\!\!\!\!\leq
 \prod_{j=1}^m \frac{p_j\|f_j\|_{L^{p_j}([0,T],\tilde{w}_j)}}{p_j-1-\alpha(p,[\vec{w}]_{A_{\vec{P}}})}
 \end{align*}
 and letting $\delta\to 0$ gives \eqref{local}, by Fatou's lemma.
