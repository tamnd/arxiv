<!-- theorem-1: body_md -->
\label{mf:clt}
Under Assumption~\ref{mf:assumptions}, let $0<\alpha<1$ and set
$\eta^N:=\sqrt N(\mu^N-\mu)$. Then
\begin{equation}\label{mf:stable-clt}
 \eta^N\xRightarrow[N\to\infty]{\mathrm{st}(\mathcal G_T^0)}\eta
 \quad\text{in }D([0,T];H^{-s}).
\end{equation}
The path space carries the Skorokhod $J_1$ topology. Conditionally on
$\mathcal G_T^0$, the limit is centered Gaussian, starts from the
$\mu_0$-Brownian bridge, and satisfies, for $\phi\in C^\infty(\mathbf T^d)$,
\begin{align}
 \eta_t(\phi)
 &=\eta_0(\phi)+M_t(\phi)
   +\int_0^t\eta_r(\mathcal B_{\mu_r}\phi)\,\mathrm d r
 \label{mf:limit-equation}\\
 &\quad+
 \int_{[0,t]\times\mathbf R^d\times[0,\bar f]}
 \mathbf 1_{\{u\le\mu_{r-}(f)\}}
 \eta_{r-}\bigl(\phi(\cdot+z)-\phi\bigr)
 \mathcal N^0(\,\mathrm d r,\,\mathrm d z,\,\mathrm d u).
 \notag
\end{align}
Here $M$ is a continuous, conditionally centered Gaussian $H^{-s}$-valued
martingale, conditionally independent of $\eta_0$, with
\begin{equation}\label{mf:bracket}
 \left\langle M(\phi),M(\psi)\right\rangle_t
 =\int_0^t\mu_r(\nabla\phi^\top a\nabla\psi)\,\mathrm d r.
\end{equation}
The jump sum is absolutely convergent for smooth tests. For almost

 d_{\mathrm{BL}}(K_N,K)\xrightarrow[N\to\infty]{}0
 \quad\text{in probability}.

 +\sup_{t\le T}W_1(\mu_t^N,\widetilde\mu_t^N)
 =O_{\mathbf P}(N^{-1/(2\alpha)}).
\end{equation}

\label{mf:prescribed}
Under Assumption~\ref{mf:assumptions}, put $r:=s-2>d/2+1$.
For every $p\ge2$,
\begin{equation}\label{mf:sampling-moments}
 \sup_N\mathbf E\left[\sup_{t\le T}
 \|\sqrt N(\widetilde\mu_t^N-\mu_t^\ell)\|_{H^{-r}}^p\right]
 \le C_{p,T},
\end{equation}
with $C_{p,T}$ independent of $\ell$. The fields $\zeta^N$ converge in
$C([0,T];H^{-s})$ to the continuous Gaussian solution of the co-moving
linearized equation. Hence
$\sqrt N(\widetilde\mu^N-\mu^\ell)$ converges in
$D([0,T];H^{-s})$ to its translation by $\ell$.
If $\ell$ is measurable with respect to an environment independent of
the initial sample and Brownian motions, these conclusions hold for
conditional laws for almost every environment.

<!-- proof-1: body_md -->
Choose $p_*>\max\{p,2\}$. Couple $Y^{i,N}$ with i.i.d. nonlinear
copies $\bar Y^i$ of law $\nu$, using the same initial states and
Brownian motions. In paired Euclidean lifts, put
\[
 Z_{j,t}^N:=\frac1N\sum_{i=1}^N g_{j,t}(\bar Y_t^i)-\nu_t(g_{j,t}),
 \qquad
 \Delta_N(t):=\max_{i\le N}\sup_{u\le t}|Y_u^{i,N}-\bar Y_u^i|.
\]
Since the Brownian terms cancel,
\begin{align*}
 |\nu_t^N(g_{j,t})-\nu_t(g_{j,t})|
 &\le\|g_j\|_{\mathrm{Lip}}\Delta_N(t)+|Z_{j,t}^N|,\\
 \Delta_N(t)&\le C\int_0^t\Delta_N(u)\,\mathrm d u
                  +C\sum_{j=1}^q\int_0^t|Z_{j,u}^N|\,\mathrm d u,\qquad
 \Delta_N(T)\le C_T\sum_{j=1}^q\int_0^T|Z_{j,u}^N|\,\mathrm d u.
\end{align*}
The summands defining $Z_{j,t}^N$ are centered, bounded, and i.i.d., so
\begin{equation}\label{mf:empirical-moment}
 \mathbf E\left[|Z_{j,t}^N|^{p_*}\right]\le C_{p_*}N^{-p_*/2}.
\end{equation}
Jensen's inequality gives
\begin{equation}\label{mf:particle-comparison}
 \mathbf E\left[\Delta_N(T)^{p_*}\right]
 \le C_{p_*,T}\sum_{j=1}^q\int_0^T\mathbf E\left[|Z_{j,u}^N|^{p_*}\right]\,\mathrm d u
 \overset{\eqref{mf:empirical-moment}}{\le} C_{p_*,T}N^{-p_*/2}.
\end{equation}
For the independent-copy empirical field
$\Gamma_t^N:=N^{-1/2}\sum_{i=1}^N(\delta_{\bar Y_t^i}-\nu_t)$,
bounded drift and Brownian increments imply
\[
 \mathbf E\left[|\bar Y_t^i-\bar Y_u^i|^{p_*}\right]
 \le C_{p_*,T}|t-u|^{p_*/2}.
\]
For centered i.i.d. variables $\Xi_i$ in a real Hilbert space,
symmetrization and the Hilbert-space moment inequality yield
\[
 \mathbf E\left[\left\|N^{-1/2}\sum_{i=1}^N\Xi_i\right\|^{p_*}\right]
 \le C_{p_*}\left\{(\mathbf E\left[\|\Xi_1\|^2\right])^{p_*/2}
              +N^{1-p_*/2}\mathbf E\left[\|\Xi_1\|^{p_*}\right]\right\}.
\]
For the centered $H^{-r}$-valued Dirac increments, this gives
\begin{equation}\label{mf:independent-increments}
 \mathbf E\left[\|\Gamma_t^N-\Gamma_u^N\|_{H^{-r}}^{p_*}\right]
 \overset{\eqref{mf:dirac-bound}}{\le} C_{p_*,T}|t-u|^{p_*/2},\qquad
 \mathbf E\left[\|\Gamma_0^N\|_{H^{-r}}^{p_*}\right]\le C_{p_*}.
\end{equation}
For $t_{n,k}:=Tk2^{-n}$, dyadic approximation and Minkowski give
\begin{align*}
 \left\|\sup_{t\le T}\|\Gamma_t^N\|_{H^{-r}}\right\|_{L^{p_*}}
 &\le\|\Gamma_0^N\|_{L^{p_*}(H^{-r})}
 +\sum_{n\ge0}\left(\sum_{k=1}^{2^n}
 \mathbf E\left[\|\Gamma_{t_{n,k}}^N-\Gamma_{t_{n,k-1}}^N\|_{H^{-r}}^{p_*}\right]
 \right)^{1/p_*}\\
 &\overset{\eqref{mf:independent-increments}}{\le}
 C+C\sqrt T\sum_{n\ge0}2^{-n(1/2-1/p_*)}<\infty.
\end{align*}
Finally,
\[
 \sup_{t\le T}\|\zeta_t^N\|_{H^{-r}}
 \overset{\eqref{mf:dirac-bound}}{\le}
 \sup_{t\le T}\|\Gamma_t^N\|_{H^{-r}}
       +C_r\sqrt N\Delta_N(T).
\]
Together with \eqref{mf:particle-comparison} and translation isometry,
this proves \eqref{mf:sampling-moments}.
Define the co-moving generator
\[
 \widehat{\mathcal A}_{t,\nu}:\left|\begin{aligned}
 H^s&\longrightarrow H^{s-2}\\
 \phi&\longmapsto \widehat B_t(\cdot,\nu)\cdot\nabla\phi
                         +\tfrac12a:D^2\phi.
 \end{aligned}\right.
\]
It\^o's formula gives the exact $H^{-s}$ fluctuation drift
\begin{equation}\label{mf:exact-drift}
 \widehat{\mathcal A}_{t,\nu_t^N}^*\zeta_t^N
 -\sum_{j=1}^q\zeta_t^N(g_{j,t})\nabla\cdot(c_{j,t}^N\nu_t),
\end{equation}
where the vector field $c_{j,t}^N:\mathbf T^d\to\mathbf R^d$ is
\[
 c_{j,t}^N:\left|\begin{aligned}
 \mathbf T^d&\longrightarrow \mathbf R^d\\
 y&\longmapsto \displaystyle\int_0^1
 \partial_{m_j}B\bigl(y+\ell_t,
 m(\tau_{\ell_t}[(1-\theta)\nu_t+\theta\nu_t^N])\bigr)\,\mathrm d\theta.
 \end{aligned}\right.
\]
The martingale part is
\[
 \widehat M_t^N(\phi)
 :=\frac1{\sqrt N}\sum_{i=1}^N\int_0^t
 \nabla\phi(Y_u^{i,N})^\top\sigma\,\mathrm d W_u^i,\qquad
 \left\langle \widehat M^N(\phi),\widehat M^N(\psi)\right\rangle_t
 =\int_0^t\nu_u^N(\nabla\phi^\top a\nabla\psi)\,\mathrm d u.
\]
For $r>d/2+1$, Sobolev embedding gives
\[
 |\zeta(g_{j,t})|\le C\|\zeta\|_{H^{-r}},\qquad
 \|\nabla\cdot(c_{j,t}^N\nu_t)\|_{H^{-r}}\le C,
\]
and the covariance trace per unit time is bounded by
\begin{equation}\label{mf:trace-bound}
 4\pi^2\sum_{k\in\mathbf Z^d}(1+|k|^2)^{-r}k^\top ak<\infty.
\end{equation}
Thus $\widehat M^N$ is a continuous $H^{-r}$-valued martingale,
and the drift in \eqref{mf:exact-drift} has $H^{-s}$ norm at most
$C\|\zeta_t^N\|_{H^{-r}}$. For stopping times $\theta\le T$ and
$0\le h\le\delta$,
\[
 \mathbf E\left[\|\zeta_{(\theta+h)\wedge T}^N-\zeta_\theta^N\|_{H^{-s}}^2\right]
 \le C_T(\delta+\delta^2).
\]
The moment bound, compact embedding $H^{-r}\hookrightarrow H^{-s}$,
and these increment estimates give tightness with continuous limits.
The Hilbert-space initial CLT gives the $\mu_0$-Brownian bridge.
For $\phi\in C^\infty(\mathbf T^d)$, define
\[
 h_{j,t}^\phi:\left|\begin{aligned}
 \mathbf T^d&\longrightarrow \mathbf R\\
 y&\longmapsto \partial_{m_j}B(y+\ell_t,m(\tau_{\ell_t}\nu_t))
                       \cdot\nabla\phi(y).
 \end{aligned}\right.
\]
Taylor's formula in the moment vector yields
\begin{gather*}
 \sqrt N\{\nu_t^N(\widehat{\mathcal A}_{t,\nu_t^N}\phi)
                    -\nu_t(\widehat{\mathcal A}_{t,\nu_t}\phi)\}
 =\zeta_t^N(\widehat{\mathcal A}_{t,\nu_t}\phi)
   +\sum_{j=1}^q\zeta_t^N(g_{j,t})\nu_t(h_{j,t}^\phi)
   +R_N(t,\phi),\\
 \begin{aligned}
 |R_N(t,\phi)|
 &\le C_\phi\left\{\|\zeta_t^N\|_{H^{-r}}
                  |m(\tau_{\ell_t}(\nu_t^N-\nu_t))|
           +\sqrt N|m(\tau_{\ell_t}(\nu_t^N-\nu_t))|^2\right\}\\
 &\le C_\phi N^{-1/2}\|\zeta_t^N\|_{H^{-r}}^2.
 \end{aligned}
\end{gather*}
Hence
\[
 \mathbf E\left[\int_0^T|R_N(t,\phi)|\,\mathrm d t\right]
 \le C_\phi N^{-1/2}\int_0^T\mathbf E\left[\|\zeta_t^N\|_{H^{-r}}^2\right]\,\mathrm d t
 \overset{\eqref{mf:sampling-moments}}{\le}C_{\phi,T}N^{-1/2}.
\]
For $H:=\nabla\phi^\top a\nabla\psi$,
\begin{align*}
 \mathbf E\left[\sup_{t\le T}\left|\int_0^t(\nu_u^N-\nu_u)(H)\,\mathrm d u\right|\right]
 &\le\|H\|_{\mathrm{Lip}}T\mathbf E\left[\Delta_N(T)\right]
 +\int_0^T\mathbf E\left[\left|\frac1N\sum_iH(\bar Y_u^i)-\nu_u(H)\right|\right]\,\mathrm d u\\
 &\overset{\eqref{mf:particle-comparison}}{\le}C_{\phi,\psi,T}N^{-1/2}.
\end{align*}
The joint CLT for the initial field and martingale therefore gives a centered
Gaussian martingale $\widehat M$, independent of the initial bridge,
with covariance
\[
 \left\langle \widehat M(\phi),\widehat M(\psi)\right\rangle_t
 =\int_0^t\nu_u(\nabla\phi^\top a\nabla\psi)\,\mathrm d u
\]
\citep{FernandezMeleard1997,EthierKurtz1986,Rebolledo1980}.
Every limit solves
\[
 \zeta_t(\phi)=\zeta_0(\phi)+\widehat M_t(\phi)
 +\int_0^t\left\{\zeta_u(\widehat{\mathcal A}_{u,\nu_u}\phi)
       +\sum_j\zeta_u(g_{j,u})\nu_u(h_{j,u}^\phi)\right\}\,\mathrm d u.
\]
We prove uniqueness by backward duality. For $0\le u\le v\le T$,
let
\[
 P_{u,v}:\left|\begin{aligned}
 C^1(\mathbf T^d)&\longrightarrow C^1(\mathbf T^d)\\
 \phi&\longmapsto [y\longmapsto\mathbf E_{u,y}\left[\phi(\bar Y_v)\right]],
 \end{aligned}\right.
\qquad
 \widehat{\mathcal K}_u:\left|\begin{aligned}
 C^1(\mathbf T^d)&\longrightarrow C^1(\mathbf T^d)\\
 \phi&\longmapsto \displaystyle\sum_jg_{j,u}\nu_u(h_{j,u}^\phi).
 \end{aligned}\right.
\]
Here $\mathbf E_{u,y}$ denotes expectation for the diffusion with fixed law flow $\nu$
started from $y$ at time $u$. Both operators are uniformly bounded
on $C^1$. For $\phi\in C^\infty(\mathbf T^d)$ and $t\le T$, solve
\begin{equation}\label{mf:backward}
 v_u=P_{u,t}\phi+\int_u^tP_{u,w}\widehat{\mathcal K}_w v_w\,\mathrm d w,
 \qquad 0\le u\le t.
\end{equation}
The $n$th Picard term is bounded in $C^1$ by
$C_T^{n+1}(t-u)^n\|\phi\|_{C^1}/n!$. Differentiating the diffusion
flow and the integral equation gives
\[
 \sup_{u\le t}\|v_u\|_{C^k}\le C_T\|\phi\|_{C^k},
 \qquad 1\le k\le\lceil s\rceil+2.
\]
Consequently,
\[
 v\in L^\infty([0,t];H^{s+2})\cap W^{1,\infty}([0,t];H^s),\qquad
 \partial_uv_u=-(\widehat{\mathcal A}_{u,\nu_u}
                      +\widehat{\mathcal K}_u)v_u
 \quad\text{for a.e. }u,\qquad v_t=\phi.
\]
The derivative follows from the integral evolution identity for
$P_{u,w}$ and requires only measurable time dependence.
Extend $v$ constantly beyond $[0,t]$ and smooth it in time and space.
The resulting tests $v^{(n)}$ satisfy
\[
 \|v^{(n)}-v\|_{L^1(H^{s+2})}
 +\|v^{(n)}-v\|_{C(H^s)}
 +\|\partial_uv^{(n)}-\partial_uv\|_{L^1(H^s)}\longrightarrow0.
\]
Apply the product rule to $\zeta_u(v_u^{(n)})$.
The drift passes to the limit through the uniform operator bound
$H^{s+2}\to H^s$, and the martingale through It\^o isometry and
$H^s\hookrightarrow C^1$. For any càdlàg $H^{-s}$-valued
solution, the same limit argument applies after localizing its bounded path norm.
Thus
\[
 \zeta_t(\phi)=\zeta_0(v_0)+\int_0^t\,\mathrm d\widehat M_u(v_u).
\]
This identifies the law, proves Gaussianity and uniqueness, and gives
convergence of the full sequence. Finally,
\[
 \sup_{t\le T}\|\tau_{\ell_t}z_n(t)-\tau_{\ell_t}z(t)\|_{H^{-s}}
 =\sup_{t\le T}\|z_n(t)-z(t)\|_{H^{-s}}
\]
for $z_n,z\in C([0,T];H^{-s})$. Translating back therefore gives
convergence in the original frame. Conditioning on an independent environment
fixes $\ell$ and leaves the private-input law unchanged.

\label{mf:strip}
Let $a,b$ be predictable $[0,\bar f]$-valued processes in a filtration
preserving the compensator $\,\mathrm d t\,\nu_\alpha(\,\mathrm d z)\,\mathrm d u$. Define
\begin{align*}
 A_t&:=\int_0^t|a_r-b_r|\,\mathrm d r,\qquad
 V_t:=\int_{[0,t]\times\mathbf R^d\times[0,\bar f]}
 |z|\,|\mathbf 1_{\{u\le a_r\}}-\mathbf 1_{\{u\le b_r\}}|
 \mathcal N^0(\,\mathrm d r,\,\mathrm d z,\,\mathrm d u),\\
 K_t&:=\int_{[0,t]\times\mathbf R^d\times[0,\bar f]}
 z(\mathbf 1_{\{u\le a_r\}}-\mathbf 1_{\{u\le b_r\}})
 \mathcal N^0(\,\mathrm d r,\,\mathrm d z,\,\mathrm d u).
\end{align*}
For every stopping time $\tau\le T$ and $x,v>0$,
\begin{equation}\label{mf:strip-estimate}
 \mathbf P(V_\tau>x,\ A_\tau\le v)\le Cvx^{-\alpha},
 \qquad \sup_{t\le\tau}|K_t|\le V_\tau.
\end{equation}
The estimate is uniform for mark measures $\nu_N$ satisfying
\begin{equation}\label{mf:tail-bounds}
 \nu_N(|z|>x)\le Cx^{-\alpha},\qquad
 \int_{|z|\le x}|z|\nu_N(\,\mathrm d z)\le Cx^{1-\alpha},
 \qquad x>0.
\end{equation}

Let $\omega_{d-1}$ be the surface area of the unit sphere in $\mathbf R^d$.
Polar coordinates give
\begin{align*}
 \nu_\alpha(|z|>x)
 &=c_{d,\alpha}\omega_{d-1}\int_x^\infty r^{-1-\alpha}\,\mathrm d r
   =\frac{c_{d,\alpha}\omega_{d-1}}\alpha x^{-\alpha},\\
 \int_{|z|\le x}|z|\nu_\alpha(\,\mathrm d z)
 &=c_{d,\alpha}\omega_{d-1}\int_0^x r^{-\alpha}\,\mathrm d r
   =\frac{c_{d,\alpha}\omega_{d-1}}{1-\alpha}x^{1-\alpha}.
\end{align*}
Fix $\varepsilon>0$ and stop at
$\widehat\tau:=\tau\wedge\inf\{t:A_t>v+\varepsilon\}\wedge T$.
Let $B_x$ count disagreement marks with $|z|>x/2$ up to
$\widehat\tau$, and let $S_x$ sum the sizes of the remaining
disagreement marks. Since
\[
 \int_0^{\bar f}|\mathbf 1_{\{u\le a_r\}}-\mathbf 1_{\{u\le b_r\}}|\,\mathrm d u
 =|a_r-b_r|,
\]
continuity of $A$ and predictable compensation imply
\begin{equation}\label{mf:strip-compensation}
 \mathbf E\left[B_x\right]\le(v+\varepsilon)\nu_\alpha(|z|>x/2),\qquad
 \mathbf E\left[S_x\right]\le(v+\varepsilon)\int_{|z|\le x/2}|z|\nu_\alpha(\,\mathrm d z).
\end{equation}
On $\{A_\tau\le v\}$, stopping has no effect before $\tau$.
Consequently,
\begin{align*}
 \mathbf P(V_\tau>x,A_\tau\le v)
 &\le\mathbf P(B_x\ge1)+\mathbf P(S_x>x)\\
 &\le\mathbf E\left[B_x\right]+x^{-1}\mathbf E\left[S_x\right]\\
 &\overset{\eqref{mf:strip-compensation}}{\le}(v+\varepsilon)
 \left[\nu_\alpha(|z|>x/2)
       +x^{-1}\int_{|z|\le x/2}|z|\nu_\alpha(\,\mathrm d z)\right]\\
 &\overset{\eqref{mf:tail-bounds}}{\le} C(v+\varepsilon)x^{-\alpha}.
\end{align*}
Let $\varepsilon\downarrow0$. The bound $\sup_{t\le\tau}|K_t|\le V_\tau$
is pathwise, and the calculation uses only \eqref{mf:tail-bounds}.

\label{mf:bootstrap}
Let $\varepsilon_N\to0$, $R_N=O_{\mathbf P}(\varepsilon_N)$, and let
$(A^N,V^N)$ start from zero, with $V^N\ge0$ and $A^N$ adapted,
continuous, and nondecreasing. Suppose \eqref{mf:strip-estimate}
holds uniformly for stopping times of the same filtration and
\begin{equation}\label{mf:bootstrap-hypothesis}
 A_t^N\le R_N+CV_t^N,\qquad t\le T.
\end{equation}
Then
\begin{equation}\label{mf:bootstrap-rates}
 A_T^N=O_{\mathbf P}(\varepsilon_N),\qquad
 V_T^N=O_{\mathbf P}(\varepsilon_N^{1/\alpha})
       =o_{\mathbf P}(\varepsilon_N).
\end{equation}
The conclusion also holds for an almost surely finite random $C$
independent of $N$.

Replace $C$ by $1\vee C$. For $M>0$, set
$\tau_N:=\inf\{t:A_t^N\ge2M\varepsilon_N\}\wedge T$.
On $\{A_T^N>2M\varepsilon_N,\ R_N\le M\varepsilon_N\}$,
continuity and \eqref{mf:bootstrap-hypothesis} give
\[
 A_{\tau_N}^N=2M\varepsilon_N,\qquad
 CV_{\tau_N}^N\overset{\eqref{mf:bootstrap-hypothesis}}{\ge}
 A_{\tau_N}^N-R_N\ge M\varepsilon_N.
\]
Therefore
\begin{align*}
 \mathbf P(A_T^N>2M\varepsilon_N)
 &\le\mathbf P(R_N>M\varepsilon_N)

                   A_{\tau_N}^N\le2M\varepsilon_N\right)\\
 &\overset{\eqref{mf:strip-estimate}}{\le}\mathbf P(R_N>M\varepsilon_N)
      +C'(M\varepsilon_N)^{1-\alpha},\\
 \mathbf P(V_T^N>R\varepsilon_N^{1/\alpha})
 &\overset{\eqref{mf:strip-estimate}}{\le}
 \mathbf P(A_T^N>2M\varepsilon_N)+C'MR^{-\alpha},\qquad R>0.

 =\varepsilon_N^{1/\alpha-1}
      \frac{V_T^N}{\varepsilon_N^{1/\alpha}}

<!-- proof-4: body_md -->
Above each cutoff, there are finitely many candidate translations, with
Lipschitz diffusions between them. Between candidates the reference law
solves the usual Lipschitz McKean--Vlasov equation. At each candidate
its pre-jump law fixes the acceptance label. This constructs both
cutoff systems. The discarded variation satisfies
\[
 J_T^\varepsilon:=
 \int_{[0,T]\times\{|z|\le\varepsilon\}\times[0,\bar f]}
 |z|\mathcal N^0(\,\mathrm d t,\,\mathrm d z,\,\mathrm d u),\;
 \mathbf E\left[J_T^\varepsilon\right]
 =T\bar f\int_{|z|\le\varepsilon}|z|\nu_\alpha(\,\mathrm d z)
 =\frac{T\bar f c_{d,\alpha}\omega_{d-1}}{1-\alpha}
        \varepsilon^{1-\alpha},\quad
 J_T^\varepsilon\downarrow0\quad\text{a.s.}
\]
Couple cutoffs $\varepsilon'<\varepsilon$ synchronously, and let
\[
 D_t:=\max_{i\le N}\sup_{u\le t}
 d_{\mathbf T^d}(X_u^{i,N,\varepsilon},X_u^{i,N,\varepsilon'}),\quad
 A_t:=\int_0^t|\mu_r^{N,\varepsilon}(f)
                    -\mu_r^{N,\varepsilon'}(f)|\,\mathrm d r.
\]
Let $V_t$ be the disagreement variation of their acceptance labels
on $|z|>\varepsilon$. The omitted marks have total size at most
$J_T^\varepsilon$, so synchronous coupling gives
\[
 D_t\le J_T^\varepsilon+V_t+C\int_0^tD_r\,\mathrm d r,\qquad
 A_t\le\|f\|_{\mathrm{Lip}}\int_0^tD_r\,\mathrm d r.
\]
Hence
\begin{equation}\label{mf:cutoff-comparison}
 D_t+A_t\le C_T(J_T^\varepsilon+V_t),\qquad t\le T,
\end{equation}
and
\[
 \sup_{u\le t}|L_u^{N,\varepsilon}-L_u^{N,\varepsilon'}|
 \le J_T^\varepsilon+V_t.
\]
Since $J_T^\varepsilon=O_{\mathbf P}(\varepsilon^{1-\alpha})$,
lemmas~\ref{mf:strip} and \ref{mf:bootstrap} give, uniformly in
$\varepsilon'<\varepsilon$,
\[
 A_T=O_{\mathbf P}(\varepsilon^{1-\alpha}),\qquad
 V_T=O_{\mathbf P}(\varepsilon^{(1-\alpha)/\alpha}),\qquad
 D_T=O_{\mathbf P}(\varepsilon^{1-\alpha}).
\]
Thus the particle and signal paths are Cauchy in probability for the
uniform metric. For the reference, use
\[
 D_t:=\mathbf E\left[\sup_{u\le t}
 d_{\mathbf T^d}(\bar X_u^\varepsilon,\bar X_u^{\varepsilon'})
 \Bigm|\mathcal G_T^0\right].
\]
The cutoff flows depend only on past master atoms and independent
private input, so this conditional quantity is $\mathcal G_t^0$-measurable,
bounds $\sup_{u\le t}W_1(\mu_u^\varepsilon,\mu_u^{\varepsilon'})$,
and satisfies the same Gronwall estimate. The reference flows and
signals are therefore Cauchy as well.
Choose a subsequence with summable error probabilities. Its paths
converge uniformly almost surely to adapted càdlàg limits. Lipschitz
continuity lets us pass to the limit in the drift integrals. For each
$\delta>0$, compensation gives
\[
 \mathbf E\left[\int_{[0,T]\times\{|z|>\delta\}\times[0,\bar f]}
 \mathbf 1_{\{u=\mu_{t-}(f)\}}\mathcal N^0(\,\mathrm d t,\,\mathrm d z,\,\mathrm d u)\right]=0,
\]
and the same holds for the empirical threshold. Hence no candidate
above $\delta$ lies on the limiting threshold. There are finitely many
such candidates, and uniform convergence makes all their acceptance
labels agree eventually. Marks below $\delta$ contribute at most
$J_T^\delta\to0$, which identifies the limiting jump sums.
To identify the conditional law, construct every cutoff copy from the
same independent initial state and Brownian motion. Boundedness of the
torus distance and the conditional coupling bound imply that these
copies are Cauchy in $L^1$ for the uniform distance. Their limit
$\bar X$ has an adapted càdlàg version. For bounded Lipschitz $\phi$
and fixed $t$,
\[
 \mathbf E\left[\left|\mathbf E\left[\phi(\bar X_t^\varepsilon)-\phi(\bar X_t)
                   \Bigm|\mathcal G_t^0\right]\right|\right]
 \le\mathbf E\left[|\phi(\bar X_t^\varepsilon)-\phi(\bar X_t)|\right]\longrightarrow0.
\]
Also $\mu_t^\varepsilon(\phi)\to\mu_t(\phi)$ in $L^1$, proving
$\mu_t=\mathcal L(\bar X_t\mid\mathcal G_t^0)$. Independence of future master
increments gives the same kernel when conditioning on $\mathcal G_T^0$.
A countable determining class, rational times, and càdlàg versions
make this identity simultaneous in $t$. This constructs solutions,
and the Cauchy property gives convergence of the full cutoff family.

\label{rk:hall-sufficient}
Suppose $\overline F$ is continuous and strictly decreasing for all
sufficiently large arguments and, for some $\epsilon>0$,

Set $x_v:=(c/v)^{1/\alpha}$ and $\delta_v:=Q(v)/x_v-1$.
Regular variation gives $\delta_v\to0$. Eventual continuity and strict
monotonicity give $\overline F(Q(v))=v$ for small $v$, hence
\[
 1\overset{\eqref{rk:strong-hall}}{=}(1+\delta_v)^{-\alpha}
 \left\{1+dx_v^{-\rho}(1+\delta_v)^{-\rho}
           +O(x_v^{-\rho-\epsilon})\right\},\qquad 0=-\alpha\delta_v+dx_v^{-\rho}
    +O(\delta_v^2+x_v^{-\rho}|\delta_v|+x_v^{-\rho-\epsilon}).

Absorbing the terms containing $|\delta_v|$ first gives

 =N^{1-a}\int_{v_0}^\infty k(v)^2\,\mathrm d v\xrightarrow[N\to\infty]{}0.

 \qquad A_t^N\le\|f'\|_\infty\int_0^tD_r^N\,\mathrm d r.
\]

\[
 D_t^N+A_t^N
 \le C_T\left(\sup_{v\le t}|R_v^N|+V_t^N\right).
\]
Applying Lemma~\ref{mf:bootstrap} yields
\[
 A_T^N=O_{\mathbf P}(b_N),\qquad
 V_T^N=O_{\mathbf P}(b_N^{1/\alpha})=o_{\mathbf P}(b_N).
\]

 \begin{cases}
 N^{1/2-\rho/\alpha},&\rho<1-\alpha/2,\\
 N^{(\alpha-1)/\alpha}\sqrt{1+\log N},&\rho=1-\alpha/2,\\

 \end{cases}

\]

\[

 =\sup_{\|\phi\|_{H^s}\le1}|(\lambda-\lambda')(\phi)| \le\sup_{\|\phi\|_{H^s}\le1}\|\phi'\|_\infty W_1(\lambda,\lambda')

\]

\eqref{mf:kernel-convergence} completes the proof.

\emph{Signal comparison.}
Set $D_N:=r_N^{-1}(g_N-g)$ and
$\iota_t^N(u):=\mathbf 1_{\{u\le\mu_{t-}^{N,\mathrm{mic}}(f)\}}$.
In this proof, omitted mark domains are
$(0,\infty)\times\{-1,1\}\times[0,\bar f]$. Split
\begin{align*}
 L_t^{N,\mathrm{mic}}-L_t&=F_t^N+K_t^N,\qquad
 F_t^N:=\int_{(0,t]}e(g_N-g)(y)\iota_v^N(u)
                  \mathcal P^0(\,\mathrm d v,\,\mathrm d y,\,\mathrm d e,\,\mathrm d u),\\
 K_t^N&:=\int_{(0,t]}eg(y)(\iota_v^N-\iota_v)(u)
                  \mathcal P^0(\,\mathrm d v,\,\mathrm d y,\,\mathrm d e,\,\mathrm d u).
\end{align*}
Let $V_t^N:=\int_{(0,t]}g(y)|\iota_v^N-\iota_v|(u)\,\mathrm d\mathcal P^0$
and $A_t^N:=\int_0^t|\mu_{u-}^{N,\mathrm{mic}}(f)-\mu_{u-}(f)|\,\mathrm d u$.
Then $\|K^N\|_{\infty,[0,t]}\le V_t^N$.
Here $b_N=r_N$. The predictable-threshold bound
\eqref{rk:rank-error-probability} gives
$\|F^N\|_\infty=O_{\mathbf P}(r_N)$.

By Lemma~\ref{mf:prescribed}, $R_N=O_{\mathbf P}(N^{-1/2})$, and synchronous
coupling gives
\[
 A_t^N\le C_T\{R_N+\|F^N\|_{\infty,[0,t]}+V_t^N\}.
\]
Since $N^{-1/2}\le r_N$, lemmas~\ref{mf:strip} and \ref{mf:bootstrap} yield
\begin{equation}\label{cr:strip-rate}
 A_T^N=O_{\mathbf P}(r_N),\qquad
 V_T^N=O_{\mathbf P}(r_N^{1/\alpha})=o_{\mathbf P}(r_N).
\end{equation}
Since
\[
 \sup_{t\le T}|\mu_t^{N,\mathrm{mic}}(f)-\mu_t(f)|
 \le R_N+C_T(\|F^N\|_\infty+V_T^N),
\]
we also obtain
\begin{equation}\label{cr:threshold-rate}
 \sup_{t\le T}|\mu_t^{N,\mathrm{mic}}(f)-\mu_t(f)|=O_{\mathbf P}(r_N).
\end{equation}

a positive minimum rank almost surely. Predictable compensation gives
\[
 \mathbf E\left[\int_{(0,T]\times(0,R]\times\{-1,1\}\times[0,\bar f]}
       \mathbf 1_{\{u=\mu_{v-}(f)\}}\,\mathrm d\mathcal P^0\right]=0.
\]

with $\Delta_R:=1$ when $K_R=0$. Their labels coincide whenever
$\sup_{t\le T}|\mu_t^{N,\mathrm{mic}}(f)-\mu_t(f)|<\Delta_R$.

On it, the uniform error between their rescaled sums is at most
$\sum_{j=1}^{K_R}|D_N(y_j)-h(y_j)|\to0$.

\[
 \mathbf E\left[\|r_N^{-1}F^{N,>R}\|_\infty^2\right]
       \le4T\bar f\int_R^\infty D_N(y)^2\,\mathrm d y,\qquad

       \le4T\bar f\int_R^\infty h(y)^2\,\mathrm d y.

The superscript $>R$ denotes restriction to those ranks. Since

 &\le2\int_R^\infty|D_N(y)-h(y)|^2\,\mathrm d y

 &\overset{\eqref{rk:inverse-l2}}{\le}
 2\int_R^\infty h(y)^2\,\mathrm d y\xrightarrow[R\to\infty]{}0,

is continuous into vector $J_1$: the uniform distance between two

plus $\|z_N-Z\|_\infty$. Then
\[
 \mathbf E\left[1\wedge\varepsilon_N\Bigm|\mathcal G_T^0\right]
           \xrightarrow[N\to\infty]{\mathbf P}0,
 \qquad

field and this approximation is bounded by a constant times
$\mathbf E\left[1\wedge\varepsilon_N\Bigm|\mathcal G_T^0\right]$. This proves
\eqref{cr:critical-limit} and its joint version. Below criticality, the
same argument gives the conditional Dirac kernel at $\chi$.

\begin{equation}\label{rk:three-regimes}

The chain event times, mapped by $A$, form a countable random set

surely no jump of $L$ is a chain event, simultaneously for all $N$.

\label{clk:cauchy}
Let $\alpha=1$, $t>0$, and $\phi\in C_b^1(\mathbf R)$. Define
\[
 \begin{gathered}
 a:\left|\begin{aligned}
 \mathbf R&\longrightarrow \mathbf R^m\\
 \ell&\longmapsto (\phi(y_a+\ell))_{a=1}^m,
 \end{aligned}\right.
 \qquad
 d:\left|\begin{aligned}
 \mathbf R&\longrightarrow \mathbf R\\
 \ell&\longmapsto \displaystyle\sum_a p_{t,a}\phi'(y_a+\ell),
 \end{aligned}\right.\\[1ex]
 \kappa(\ell):=a(\ell)^\top c_t,\qquad
 r(\ell):=a(\ell)^\top P_ta(\ell)-\kappa(\ell)^2/v_t\ge0.
 \end{gathered}
\]
Then, stably relative to $\sigma(S)$,
\[
 \sqrt N\bigl(\mu_t^N(\phi)-\mu_t^L(\phi)\bigr)
       \xRightarrow[N\to\infty]{}Y:=U+d(L_t)Z,
 \qquad U:=a(L_t)^\top G_t,\qquad Z:=|H_t|\xi,
\]
where $\xi$ is a standard Cauchy variable independent of $(G,H,S)$.
Conditionally on $L$, $U$ and $Z$ are independent if and only if
$\kappa(L_t)=0$ almost surely, and
\begin{equation}\label{clk:cauchy-cf}
 \mathbf E\left[e^{iuY}\Bigm| L\right]
 =e^{-u^2r(L_t)/2}
   \mathbf E_{h\sim\mathcal N(0,v_t)}\left[
       \exp\left\{\frac{iu\kappa(L_t)}{v_t}h-|u d(L_t)|\,|h|\right\}\right].
\end{equation}
At time $t$, the Gaussian empirical field and $Z$ are conditionally
independent if and only if $c_t=0$. These criteria concern $Z$ itself.
When $d(L_t)=0$, the observed clock correction vanishes.

For each $a$, the fundamental theorem of calculus gives
\[
 \phi(y_a+\ell+z)-\phi(y_a+\ell)
 =z\phi'(y_a+\ell)
  +z\int_0^1\{\phi'(y_a+\ell+\theta z)-\phi'(y_a+\ell)\}\,\mathrm d\theta.
\]
The last term is $o(|z|)$, uniformly for $\ell$ in compact sets.
Apply Proposition~\ref{clk:tangent}, using tightness of $L_t$ and
$\sqrt N(L_t^N-L_t)$, to obtain the scalar limit.
Conditional Gaussian regression gives
\begin{equation}\label{clk:gaussian-regression}
 U=\frac{\kappa(L_t)}{v_t}H_t+R,\qquad
 R\perp H_t\mid L,\qquad R\mid L\sim\mathcal N(0,r(L_t)).
\end{equation}
The residual $R$ is also conditionally independent of $\xi$.
Integrating over $R$, $\xi$, and $H_t$ in turn gives
\[
 \begin{aligned}
 \mathbf E\left[e^{iuY}\Bigm| L,H_t\right]
 &\overset{\eqref{clk:gaussian-regression}}{=}
 e^{iu\kappa(L_t)H_t/v_t}\mathbf E\left[e^{iuR}\Bigm| L\right] \times\mathbf E\left[e^{iu d(L_t)|H_t|\xi}\Bigm| L,H_t\right]\\
 &=\exp\left\{-\tfrac12u^2r(L_t)
       +\frac{iu\kappa(L_t)}{v_t}H_t-|u d(L_t)|\,|H_t|\right\}.
 \end{aligned}
\]
Since $H_t\mid L\sim\mathcal N(0,v_t)$, averaging proves

$U\perp(H_t,\xi)\mid L$, hence $U\perp Z\mid L$.
Conversely,
\[
 \mathbf E\left[e^{iuZ}\Bigm| U,L\right]=\mathbf E\left[e^{-|u||H_t|}\Bigm| U,L\right].
\]
If $U\perp Z\mid L$, uniqueness of Laplace transforms implies
$U\perp|H_t|\mid L$. The Gaussian fourth-moment identity then gives

 0=\operatorname{Cov}(U^2,H_t^2\mid L)=2\kappa(L_t)^2.
\]
Finally, the map $g\mapsto\sum_ag_a\delta_{y_a+L_t}$ is injective
because the locations are distinct. Equivalently, smooth test functions
can take any prescribed values at these locations. The scalar criterion
for all such tests is $a^\top c_t=0$ for every $a\in\mathbf R^m$, namely $c_t=0$.

<!-- proof-19: body_md -->
The integrable characteristic function gives the density
\[
 f_U(x)=\pi^{-1}\operatorname{Re}\int_0^\infty
                  e^{ixt-t^\alpha}\,\mathrm d t.
\]
For $x>0$, rotate the contour into the first quadrant, using the
principal branch of $t^\alpha$. The small arc vanishes with its radius.
On the large arc,
$|e^{ixt-t^\alpha}|=e^{-x|t|\sin\theta-|t|^\alpha\cos(\alpha\theta)}$
and $\cos(\alpha\theta)\ge\cos(\pi\alpha/2)>0$. Hence
\[
 f_U(x)=-\pi^{-1}\operatorname{Im}\int_0^\infty
       e^{-xy}\exp\{-e^{i\pi\alpha/2}y^\alpha\}\,\mathrm d y.
\]
For $z\in\mathbf C$ with $\operatorname{Re}z\ge0$,
\begin{equation}\label{clk:exponential-remainder}
 e^{-z}-1+z=z^2\int_0^1(1-u)e^{-uz}\,\mathrm d u,\qquad |e^{-z}-1+z|
 \le|z|^2\int_0^1(1-u)e^{-u\operatorname{Re}z}\,\mathrm d u
 \le|z|^2/2.
\end{equation}
Using $z=e^{i\pi\alpha/2}y^\alpha$ and
$\int_0^\infty e^{-xy}y^a\,\mathrm d y=\Gamma(a+1)x^{-a-1}$ gives
\begin{equation}\label{clk:stable-density-bound}
 \begin{aligned}
 \left|f_U(x)-\frac{\Gamma(\alpha+1)\sin(\pi\alpha/2)}\pi
                         x^{-1-\alpha}\right| &\le\frac1\pi\int_0^\infty e^{-xy}
 \left|e^{-e^{i\pi\alpha/2}y^\alpha}-1+e^{i\pi\alpha/2}y^\alpha\right|\,\mathrm d y\\
 &\overset{\eqref{clk:exponential-remainder}}{\le}
 \frac1{2\pi}\int_0^\infty e^{-xy}y^{2\alpha}\,\mathrm d y
 =\frac{\Gamma(2\alpha+1)}{2\pi}x^{-1-2\alpha}.
 \end{aligned}
\end{equation}
Integrate from $x$ to infinity and use symmetry:
\[
 \begin{aligned}
 \left|2\int_x^\infty f_U(y)\,\mathrm d y
       -\frac{2\Gamma(\alpha+1)\sin(\pi\alpha/2)}{\pi\alpha}x^{-\alpha}\right|
 &\overset{\eqref{clk:stable-density-bound}}{\le}
 \frac{\Gamma(2\alpha+1)}\pi\int_x^\infty y^{-1-2\alpha}\,\mathrm d y\\
 &=\frac{\Gamma(2\alpha+1)}{2\pi\alpha}x^{-2\alpha}
 =\frac{\Gamma(2\alpha)}\pi x^{-2\alpha}.
 \end{aligned}
\]
The identity $\Gamma(a+1)=a\Gamma(a)$ proves
\eqref{clk:stable-tail-bound}.

\label{clk:same-law}
Fix $T>0$, $0<\alpha<1$, $\lambda>0$, $0<|\varepsilon|<1$, and $s>1/2$.
Let $(Y_i)_{i\ge1}$ be independent on $\mathbf T$ with common law
$\nu(\,\mathrm d x):=(1+\varepsilon\cos(2\pi x))\,\mathrm d x$, and set
$\nu^N:=N^{-1}\sum_{i=1}^N\delta_{Y_i}$.
There are rank and grid couplings, denoted by $\mathrm r$ and $\mathrm g$,
in which $L^N$ is a compound Poisson process of rate $N\lambda$ with
marks $N^{-1/\alpha}U$, where $U$ is as in Lemma~\ref{clk:stable-tail},
independent of $(Y_i)_{i\ge1}$.
In each construction, let $L_t:=S_{\lambda t}$ for a standard symmetric
$\alpha$-stable process and set
$\mu_t^N:=\tau_{L_t^N}\nu^N$, $\mu_t:=\tau_{L_t}\nu$.
For each fixed $N$, the particle systems $(Y_i+L^N)_{i=1}^N$ have
identical laws. Their joint laws with $L$, and their joint realizations
across $N$, depend on the coupling.
In the rank coupling,
\begin{equation}\label{clk:same-law-rank}
 \sqrt N(\mu^{N,\mathrm r}-\mu^{\mathrm r})
 \xRightarrow[N\to\infty]{}\Gamma,\qquad
 \Gamma_t:=\tau_{L_t}\mathcal B,
 \quad\text{in }D([0,T];H^{-s}(\mathbf T)),
\end{equation}
where $\mathcal B$ is the $\nu$-Brownian bridge of
\eqref{rk:bridge-covariance}, independent of $S$.
In the grid coupling $L_t^{N,\mathrm g}:=S_{K_t^N/N}$, where $K^N$ is a
rate-$N\lambda$ Poisson process independent of $S$, the same limit holds
at every finite collection of deterministic times. Moreover,
\begin{equation}\label{clk:same-law-grid-center}
 \sqrt N(\mu^{N,\mathrm g}-\tau_{L^{N,\mathrm g}}\nu)
       \xRightarrow[N\to\infty]{}\Gamma
       \quad\text{in }D([0,T];H^{-s}(\mathbf T)).
\end{equation}
All these limits are stable relative to $\sigma(S)$ in their respective
constructions. Nevertheless, for every deterministic $a_N\to\infty$,
\begin{equation}\label{clk:same-law-obstruction}
 \sup_{t\le T}a_N\|\mu_t^{N,\mathrm g}-\mu_t^{\mathrm g}\|_{H^{-s}}
                 \longrightarrow\infty\quad\text{almost surely}.
\end{equation}
Thus the grid-coupled differences normalized by $a_N$ are not
$J_1$-tight at the limiting center.
