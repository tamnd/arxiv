<!-- proposition-1: body_md -->
\label{prop1}
  Suppose that $u\in\mathcal{S}'(\mathbb R^d)$ is a solution to $(\Delta-k^2(1+q))u=0$ in
  $B_R^c(0).$ If $u\in L^2(B_R^c(0))$ then $u$ vanishes in this set.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
As $u\in L^2(B_R^c(0)),$ the condition in~\eqref{eqn37.101} is
  satisfied for both $+$ and $-,$ which then implies {\color{black} by the
    propagation of singularities as noted above} \color{black} that
  $\operatorname{WF}_{3\operatorname{sc}}(u)=\emptyset.$ Therefore $u$ is Schwartz in a
  neighborhood of infinity. The proof in~\cite{VasyAsterisque} that
  such a solution vanishes faster than any exponential is local at
  infinity. Moreover, the Carleman estimate {\color{black}
    in~\cite{VasyAsterisque}} \color{black} showing that a
  super-exponentially vanishing solution is actually zero in a
  neighborhood of infinity is also local at infinity. Hence $u$
  vanishes in an open set. The classical unique continuation theorems
  for second order elliptic operators then imply that $u$ vanishes in
  $B_R^c(0).$

finitely many negative eigenvalues $\{\lambda_l(\tau)\}.$ A simple
comparison argument using the min-max principle shows that there is an
integer $J$ so that, for $|\tau|\leq \tau_{\max},$ there are at most $J$

 First assume that $L_{\tau,j}$ does not have resonances at
 $\sqrt{\lambda}.$ In this case we only need to consider intersections between

   see~\cite{Kato}, to conclude that there are analytic functions
   $(\lambda(\tau),u(\tau)),$ such that
 \begin{equation}
   L_{\tau}u(\tau)=\lambda(\tau)u(\tau)\text{ with }\|u(\tau)\|^2_{L^2}=1.
 \end{equation}

 \begin{equation}\label{eqn110.223}

   \langle \widetilde{Q}^{\epsilon_j,\mu_j}_{10}u(\tau),u(\tau)\rangle
   =-4\operatorname{Im}\langle \check{\xi}_{1\tau},d\mathfrak l\lrcorner\breve{\xi}_{\tau}\rangle,
 \end{equation}

     |\dot{\lambda}(\tau)|<&4\|d\mathfrak l\|_{L^{\infty}}\|\check{\xi}_{1\tau}\|(\|\check{\xi}_{2\tau}\|
     +\|\check{\xi}_{3\tau}\|)\\
     &<2\sqrt{2}\|d\mathfrak l\|_{L^{\infty}}.
     \end{split}
 \end{equation}

 $\lambda_l(\tau)=\lambda-\tau^2.$ A simple compactness
 argument, combined with the bound on the maximum number of
 eigenvalues, for any allowable $\tau,$ and the fact that eigenvalue
 branches can only appear, or disappear at $\lambda=0,$ and therefore
 have analytic representations,
 finishes the proof in the non-resonance case.

 Now suppose that $L_{\tau,j}$ has a resonance at $\sqrt{\lambda};$

 \end{equation}

 \end{equation}

\label{IdMourreConst}
 Suppose $\lambda\in\mathbb{R}$. We say that the {\em ideal Mourre constant} at the level of an eigenvalue
$\lambda'=\lambda-|\xi_a|^2$ of $H^a(y_a,\xi_a)$ with eigenprojector $E$ is $M^a_{(y_a,\xi_a)}(\lambda)$
\begin{equation}\label{eq:ideal-Mourre-constant}
M^a_{(y_a,\xi_a)}(\lambda)=2
(\lambda-\lambda')+C^a_{(y_a,\xi_a)}
\end{equation}
where $C^a_{(y_a,\xi_a)}(\lambda)$ is the largest constant (i.e.\ the
$\sup$ of all constants) such that
\begin{equation}\label{eq:ideal-Mourre-constant-2}
\lambda-\lambda'=|\xi_a|^2\Rightarrow
|\xi_a|E\Big[\frac{\xi_a}{|\xi_a|}\partial_{\xi_a},H^a\Big]E\geq C^a_{(y_a,\xi_a)}(\lambda) E.
\end{equation}
In case $\lambda'=\lambda-|\xi_a|^2$ is not an eigenvalue of
$H^a(y_a,\xi_a)$, we set $C^a_{(y_a,\xi_a)}(\lambda),$ and
$M^a_{(y_a,\xi_a)}(\lambda)$ to be $+\infty$.


We say that the ideal Mourre constant at the level of a subsystem
$a\in\mathcal{A}$ at energy $\lambda$ is
\begin{equation}\label{Ma_def}
M^a(\lambda)=\inf_{(y_a,\xi_a)}M^a_{(y_a,\xi_a)}(\lambda),
\end{equation}
understood to be $+\infty$ if $\lambda-|\xi_a|^2\notin \sigma_{\operatorname{pp}}(H^a(y_a,\xi_a))$ for all
$(y_a,\xi_a)$.

\label{def:microlocal-Mourre-loss}
We say that the effective microlocal Mourre loss at energy $\lambda$
at $(y_a,\xi_a)$, $\delta^a(y_a,\xi_a;\lambda)$, is given as follows: let
$\lambda'=\lambda-|\xi_a|^2$  with spectral
projector $E$ of $H^a(y_a,\xi_a)$ to $\lambda'$.
\begin{enumerate}
  \item
  If $E$ is finite rank (including if $\lambda'$ is not an eigenvalue) we set $\delta^a(y_a,\xi_a;\lambda)=0$,
  \item
$E$ is infinite rank and
\begin{equation}\label{eq:Mourre-strongly-adapted-compact}
E\Big[\frac{\xi_a}{|\xi_a|}\partial_{\xi_a},H^a\Big]E
\end{equation}
is compact on $L^2(X^a)$, we set $\delta^a(y_a,\xi_a;\lambda)=0$.
\item
  Otherwise $\delta^a(y_a,\xi_a;\lambda)$ is the inf of constants so that
\begin{equation}\label{eq:Mourre-strongly-adapted}
\Big\||\xi_a|E\Big[\frac{\xi_a}{|\xi_a|}\partial_{\xi_a},H^a\Big]E\Big\|\leq  \delta^a(y_a,\xi_a;\lambda).
\end{equation}
\end{enumerate}

We say that the effective Mourre loss at the level of a subsystem
$a\in\mathcal{A}$ at energy $\lambda$ is
$$
\delta^a(\lambda)=\sup_{(y_a,\xi_a)}\delta^a(y_a,\xi_a;\lambda),
$$
understood to be $0$ if $\lambda-|\xi_a|^2\notin \sigma_{\operatorname{pp}}(H^a(y_a,\xi_a))$ for all
$(y_a,\xi_a)$.

<!-- theorem-3: body_md -->
\label{Mourre+}
For all $\varepsilon>0$ there is $\delta>0$ such that for all smooth functions
$\phi\in{\mathcal C}^{\infty}_c((\lambda-\delta,\lambda+\delta);[0,1]),$ we have
\begin{equation}\label{eq:Mourre-0}
\phi(H)i[A,H]\phi(H)\geq {\color{black}(M'_1(\lambda)-\delta'_1(\lambda)-\varepsilon)} \color{black} \phi(H)^2+K',
\end{equation}
where $K'\in\Psi_{\mathrm{Sc}}^{-\infty,1}(\bar X,\mathcal{C})$ is a compact operator.

For the proof we follow the approach taken in~\cite[Chapter
    12]{VasyAsterisque}, which reduces the proof of the global
  estimate~\eqref{eqn128.222} to proving uniform estimates, at
  infinity in $\overline{\mathbb R}^3_{\operatorname{WG}}$ for the indicial operators,
  $i\widehat{N}_{a,0}([A,H])(\tau),$ where $a=\operatorname{mf}$ or $\operatorname{ff}_j,\,j=1,\dots,m.$
  The argument with $a=\operatorname{mf}$ is trivial because $H$ equals the
  Euclidean Laplacian in a neighborhood of $\operatorname{mf},$ and therefore
  $i[A,H]=2H.$ We focus on the estimates along the front faces.

  Recall that in a neighborhood of $\operatorname{ff}_j$ the coefficients
  $\epsilon_j,\mu_j$ depend only variables orthogonal to the channel
  direction. We fix a value of $j$ and introduce coordinates so that
  the channel is $\{x_1>r,\, x_2=x_3=0\}.$ To simplify the notation we
  also drop the $j$-subscripts. As coordinates near the
  front face we use
  \begin{equation}
    \rho=\frac{1}{x_1},\,(Y_1,Y_2)=(x_2,x_3).
  \end{equation}
We observe that
\begin{equation}\label{eqn158.502}
  \begin{split}
    i\widehat{N}_{\operatorname{ff},0}([A,H])(\tau)&=[\widehat{N}_{\operatorname{ff},0}(H)(\tau),Y\cdot\partial_{Y}]+2\tau^2+
    \tau\widetilde{Q}^{\epsilon,\mu}_{10}\\
    &=[\widehat{N}_{\operatorname{ff},0}(H)(\tau),Y\cdot\partial_{Y}]+\tau\partial_{\tau}\widehat{N}_{\operatorname{ff},0}(H)(\tau).
    \end{split}
  \end{equation}
  Because of the term $\tau\widetilde{Q}^{\epsilon,\mu}_{10},$ this differs from
  the usual case.  Another important difference is the fact that the
  point spectrum of $\widehat{N}_{\operatorname{ff},0}(H)(\tau)$ depends on $\tau.$ The
  family gap, defined in~\eqref{eqn119.224}, satisfies
  \begin{equation}
  \Gamma_0=  \Gamma(\lambda,\delta_0)>0;
  \end{equation}
  we assume that $\delta<\delta_0.$

  We let $\hat{H}_{\operatorname{ff}}(\tau)=\widehat{N}_{\operatorname{ff},0}(H)(\tau),$ which is given
  in~\eqref{eqn48.61}:
  \begin{equation}
   \hat{H}_{\operatorname{ff}}(\tau)= \Delta^{(2)}\otimes\operatorname{Id}_{\Lambda^1\mathbb R^3}+\widetilde{Q}_1^{\epsilon,\mu}(\tau)+
    \widetilde{Q}_0^{\epsilon,\mu}+\omega^2(\kappa_0^2-\kappa^2),
  \end{equation}
  where $\Delta^{(2)}=-[\partial_{Y_1}^2+\partial_{Y_2}^2].$ Using the results
  in Appendix C of~\cite{VasyAsterisque} it suffices to show that
  there is a $C>0$ for which there exists a $\delta>0$ so that, if
  $\operatorname{supp}\phi\subset (\lambda-\delta,\lambda+\delta),$ then
  \begin{multline}\label{eqn144.224}
    \phi(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)
        \left([\hat{H}_{\operatorname{ff}}(\tau),Y\cdot\partial_Y]+2\tau^2+
          \tau\widetilde{Q}^{\epsilon,\mu}_{10}\right)
        \phi(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)\\\geq
        C \phi^2(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)
  \end{multline}
  holds for all $\tau.$ The operator $\phi(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)$
  depends only on the spectrum of $\hat{H}_{\operatorname{ff}}(\tau)$ in the interval
  $(\lambda-\tau^2-\delta, \lambda-\tau^2+\delta).$ It therefore
  follows that from Proposition~\ref{prop2.224} that, for $\delta<\delta_0,$
  there is a $\tau'_{\max},$ so that
  \begin{equation}
      |\tau|>\tau'_{\max}\Rightarrow \phi(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)=0,
  \end{equation}
and therefore, for $|\tau|>\tau'_{\max},$~\eqref{eqn144.224} is
immediate. Because the family gap, $\Gamma_0,$ is non-zero, if
$2\delta<\Gamma_0,$ then $\operatorname{supp}\phi(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)$ lies
either in the continuous spectrum of $\hat{H}_{\operatorname{ff}}(\tau),$ or in the
point spectrum, but not both. The arguments in the 2 cases are
somewhat different. We first consider the continuous spectrum; in this
case $\tau^2<\lambda+\delta.$

Given $\epsilon>0,$ with $\phi(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)$ supported in
the continuous spectrum, the fact that there is a $\delta>0,$ so that
 \begin{multline}\label{eqn146.224}
    \phi(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)
        \left([\hat{H}_{\operatorname{ff}}(\tau),Y\cdot\partial_Y]+2\tau^2\right)
        \phi(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)\\>
        2(\lambda-\frac{\epsilon}{2}) \phi(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)^2
 \end{multline}
 is the classical 2-body estimate
 from~\cite{FroeseHerbst1,FroeseHerbst2}. Only the term
 \begin{equation}\label{eqn149.334}
   \tau q_0(\tau)= \tau\phi(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)
        \widetilde{Q}^{\epsilon,\mu}_{10}
        \phi(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)
  \end{equation}
requires further consideration. Because the support of
$\phi(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)$ lies in the continuous spectrum, we
claim that as $\delta\to 0,$ the norm of this operator tends to
zero. We prove this only in the case of the free space Laplacian.

Let $Q'=\widetilde{Q}^{\epsilon,\mu}_{10};$ this is a compactly supported,
smooth matrix valued function, and let $\widehat{Q}'$ denote its Fourier
transform. For simplicity we replace $\phi(t)$ with
$\chi^{\delta}(t)=\chi_{[\lambda-\delta,\lambda+\delta]}(t).$ The
projection operator then takes the form
\begin{equation}
 \chi^{\delta}(\Delta+\tau^2)\psi=\mathcal F^{-1}\left[\chi_{D_{\tau}}(\xi)\widehat{\psi}(\xi)\right],
\end{equation}
where $D_{\tau}=B_{\sqrt{(\lambda+\delta-\tau^2)_+}}(0)\setminus
B_{\sqrt{(\lambda-\delta-\tau^2)_+}}(0),$ and $t_+=\max\{t,0\}.$ Using
the convolution theorem we see that
\begin{equation}
  \mathcal F(q_0(\tau)\psi)(\xi)=\chi_{D_{\tau}}(\xi)\int_{\mathbb R^2}\widehat{Q}'(\eta)\chi_{D_{\tau}}(\xi-\eta)\widehat{\psi}(\xi-\eta)d\eta.
\end{equation}

The Cauchy-Schwarz inequality shows that
\begin{equation}
  \left|\langle\chi^{\delta}(\Delta+\tau^2)\psi,q_0(\tau)\psi\rangle\right|\leq
  \|\chi^{\delta}(\Delta+\tau^2)\psi\|\|q_0(\tau)\psi\|.
\end{equation}
The Parseval theorem and the Cauchy-Schwarz inequality imply that
\begin{equation}
  \begin{split}
    \|q_0(\tau)\psi\|^2\leq &\|Q'\|_{L^1}^2\int
    \chi_{D_{\tau}}(\xi)
    \left[\int_{\mathbb R^2}\chi_{D_{\tau}}(\xi-\eta)d\eta\right]\left[\int
     \chi_{D_{\tau}}(\xi-\eta) |\widehat{\psi}(\xi-\eta)|^2d\eta\right] d\xi\\
  \leq &
  |D_{\tau}|^2\|Q'\|^2_{L^1}\|\chi^{\delta}(\Delta+\tau^2)\psi\|^2.
  \end{split}
\end{equation}
For bounded $\tau,$ the area  $|D_{\tau}|<m\delta,$ for a fixed
constant, $m,$ and therefore
\begin{equation}
  \left|\langle\chi^{\delta}(\Delta+\tau^2)\psi,\tau q_0(\tau)\psi\rangle\right|\leq
  m\delta|\tau||Q'\|_{L^1} \|\chi^{\delta}(\Delta+\tau^2)\psi\|^2.
\end{equation}
By choosing $\delta$ so that
$$m\delta\sqrt{\lambda+\delta}\|Q'\|_{L^1}<\epsilon,$$ this term can
be absorbed into the right hand side of~\eqref{eqn146.224}, proving
the existence of $\delta$ in these cases.

Recalling that $\delta<\delta_0,$ we are left to consider $|\tau|\in
(\sqrt{\lambda+\delta},\tau'_{\max}),$ with
$(\lambda-\delta-\tau^2,\lambda+\delta-\tau^2)$ intersecting only the
point spectrum of $\hat{H}_{\operatorname{ff}}(\tau).$ Recall that all the point spectrum of
these operators is negative and is locally parameterized by analytic
functions $\{(\lambda_l(\tau),v_l(\tau)):l=1,\dots, N_{\tau}\},$ where
$N_{\tau}<J,$ for $|\tau|\leq |\tau'_{\max}|.$ The maximum number of
eigenvalues in an interval of length $2\delta$ is denoted by
$N^{\delta}_{\max}.$

As before we simplify the
discussion by working with the projections defined by
$\chi^{\delta}(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2).$
A new difficulty that arises in the present case is that the point
spectrum of  $\hat{H}_{\operatorname{ff}}(\tau)$ depends on $\tau.$ If we make the
assumption that the eigenvalues of $\hat{H}_{\operatorname{ff}}(\tau)$ never cross, then
we can choose $\delta>0$ so that each interval
$(\lambda-\delta-\tau^2,\lambda+\delta-\tau^2)$ contains at most one
eigenvalue. It then follows from the virial theorem that
\begin{equation}\label{eqn170.502}
   \chi^{\delta}(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)
        [\hat{H}_{\operatorname{ff}}(\tau),Y\cdot\partial_Y]
        \chi^{\delta}(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)=0,
\end{equation}
therefore we are left to show that there is a positive constant $c$ so
that
\begin{equation}\label{eqn156.334}
  2\tau^2\chi^{\delta}(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)+\tau
  q_0(\tau)>c\chi^{\delta}(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2),
\end{equation}
see~\eqref{eqn149.334}. As $\tau^2>\lambda\in\mathcal I_M(c_0)$ it follows that 
there is a positive number $c_0$ so that
\begin{equation}\label{eqn202.700}
 \tau(2\tau+\dot{\lambda}(\tau))>c_0.
\end{equation}


                                                                
                                                                                
                                                           
                   
                                                                                
                 
                                                          
                                                                    
                   
                                              
                                  
                                            
                 
                                                                      
                                            
                                            

                                                                            
                                                   
                                                 
                                                          
                   
                                                                                
                                                             
                                                                                           
                         
                                                                     
                                                                              
                                     
                                                                 
                 
                                                  
                                                                         
                                                                                  
                   
                                                                     
                 


We now need to consider what happens if the interval
$(\lambda-\tau^2-\delta,\lambda-\tau^2+\delta)$ contains several
different eigenvalues. We first estimate of the contribution of the
$\widetilde{Q}^{\epsilon,\mu}_{10}$-term. As above
$\{(\lambda_j(\tau),v_j(\tau)):j=1,\dots,n\}$ are the eigenpairs with
$\lambda_j(\tau)\in(\tau^2-\lambda-\delta,\tau^2-\lambda+\delta);$ we
know that $n<N^{\delta}_{\max}.$ Since we are only considering a
single $\tau$ at a time, the functions $\{v_j(\tau)\}$ are an
orthonormal set, and therefore
\begin{equation}\label{eqn203.700}
  \langle \partial_{\tau}(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)v_j,v_l\rangle=
  (2\tau+\dot{\lambda_j}(\tau))\delta_{jl}+(\lambda_j-\lambda_l)\langle\dot{v}_j,v_l\rangle.
\end{equation}
From which it follows that there is a constant, $M,$ so that
\begin{equation}
  \chi^{\delta}(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)
 \tau
 \partial_{\tau}(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)\chi^{\delta}(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)>
    \tau[ c_0-\delta M]\chi^{\delta}(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2).
\end{equation}
Decreasing $\delta$ if needed, gives the desired lower bound.


We cannot use the virial theorem to estimate the left hand side
of~\eqref{eqn170.502}, but there is an approximate virial theorem that we
can use instead. We need to bound the
operator
\begin{multline}
  F_{\delta,\tau}=\chi^{\delta}(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)
        [\hat{H}_{\operatorname{ff}}(\tau),Y\cdot\partial_Y]
        \chi^{\delta}(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)=\\
        \chi^{\delta}(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)
        [\hat{H}_{\operatorname{ff}}(\tau)-\lambda_0,Y\cdot\partial_Y]
        \chi^{\delta}(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2).
\end{multline}
Using the eigenpairs from above, the projection operator is given by
\begin{equation}
  \chi^{\delta}(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)\psi=\sum_{l=1}^n\langle
  \psi,v_l\rangle  v_l,
\end{equation}
and
\begin{equation}
 Y\cdot\partial_Y (\hat{H}_{\operatorname{ff}}(\tau)-\lambda_0)
 (\chi^{\delta}(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)\psi=
 \sum_{l=1}^n(\lambda_l-\lambda_0)\langle
  \psi,v_l\rangle Y\cdot\partial_{Y} v_l.
\end{equation}

A computation shows that
\begin{equation}
  \langle\chi^{\delta}(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)\psi,
  F_{\delta,\tau}\psi\rangle=
  -2\sum_{k,l=1}^{n}(\lambda_l-\lambda_0)\langle\psi,v_l\rangle
  \langle\psi,v_k\rangle\langle Y\cdot\partial_Y v_k,v_l\rangle
\end{equation}
Since all of the eigenvalues are less than $-\Gamma_0,$ the
exponential decay estimates for eigenfunctions show that there is
constant $M$ independent of $|\tau|\in [\lambda,\tau_{\max}]$ and the
particular eigenfunction so that
\begin{equation}\label{eqn165.334}
  \|Y\cdot\partial_Y v_k\|_{L^2}\leq M.
\end{equation}
The Cauchy-Schwarz inequality then shows that, with another constant, $M',$
\begin{equation}\label{eqn166.334} 
  |\langle\chi^{\delta}(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)\psi,
  F_{\delta,\tau}\psi\rangle|\leq M'\delta
  \|\chi^{\delta}(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)\psi\|_{L^2}^2.
\end{equation}
Decreasing $\delta$ further completes the proof of the theorem.

<!-- remark-17: body_md -->
The assumption that $\Gamma_0>0$ allows us to avoid analyzing what
  happens if $\operatorname{supp}\phi(\hat{H}_{\operatorname{ff}}(\tau)+\tau^2)$ contains both
  continuous and point spectrum. It is easy to see that the
  approximate virial theorem,~\eqref{eqn166.334}, extends to the case
  $\Gamma_0=0.$ In particular, the estimate~\eqref{eqn165.334} does
  not require $\Gamma_0>0.$ Using the following lemma, we can extend
  our proof of this estimate to the case that $\Gamma_0=0.$ We leave the details of
  this extension to the reader; a proof of the Mourre estimate in much
  greater generality is provided in
  Appendix~\ref{MourreEstGenCase}.
                                       
                                                                     
  \begin{lemma}\label{lem1}
    Let $q$ be a  first order operator whose coefficients are
    supported in $B_R(0)$. There is a constant $C>0,$ so that if
    $v\in L^2(\mathbb R^2)$ is a solution to
    \begin{equation}\label{eqn168.334}
      (-\Delta+q)v=-\lambda v,
    \end{equation}
    for a $\lambda>0,$ then
    \begin{equation}
      \|r\partial_r v\|^2_{L^2(B_1^c(0))}\leq 4\|v\|^2_{L^2}+C\|\nabla v\|^2_{L^2(B_R(0))}.
    \end{equation}
    The constant, $C,$ is independent of $\lambda.$
  \end{lemma}
  \begin{remark}
    These estimates hold very generally for eigenfunctions of $N$-body
    Schr\"odinger operators, though the proof in the general case is
    considerably more technical.
  \end{remark}
  \begin{proof}
  As $q$ is compactly
 supported, the fact that $\lambda>0$ implies that $v$ and all of its
 derivatives are exponentially decaying. Let
 $\chi\in\mathcal{C}^{\infty}(\mathbb R)$ be a monotone increasing function that is
 zero for $r<1/2,$ and $1$ for $r\geq 1.$ Integrating by parts we see
 that
 \begin{equation}
  \int_0^{2\pi} \int_{0}^{\infty}|\chi^{\frac
    12}(r)r\partial_rv(r,\theta)|^2rdrd\theta=
  -\int_0^{2\pi} \int_{0}^{\infty}\bar{v}\partial_r[\chi(r)r^3\partial_rv(r,\theta)]drd\theta.
 \end{equation}
 Expanding the derivative, using the equation~\eqref{eqn168.334} and integrating by parts in
 $\theta$ we see that
 \begin{multline}
  \int_0^{2\pi} \int_{0}^{\infty}|\chi^{\frac
    12}(r)r\partial_rv(r,\theta)|^2rdrd\theta=-\int_{0}^{2\pi}\int_0^{\infty}
  \chi(r)[|v_{\theta}|^2+\lambda r^3|v|^2]drd\theta\\
  -\int_{0}^{2\pi}\int_0^{\infty}[2\chi(r)r^2\bar{v}\partial_rv+\chi_r
    r^3\bar{v}\partial_rv
    +\chi(r)r^3\bar{v}q v]drd\theta.
 \end{multline}
 The first integral on the right is non-positive, and the last two
 terms in the next integral are compactly supported. Applying the
 Cauchy-Schwarz and arithmetic-geometric mean inequality to the
 remaining term gives the conclusion of the lemma.
  \end{proof}

<!-- remark-19: body_md -->
The statement $iE[w^bD_{w^b},H^b]E=0$ is formally a consequence of
\begin{equation*}
iE[w^bD_{w^b},H^b]E=iEw^b D_{w^b}(H^b-\lambda-|\xi_b|^2)E
-iE(H^b-\lambda-|\xi_b|^2)w^b D_{w^b}E=0,
\end{equation*}
but this requires justification since $w^b D_{w^b}$ is not bounded
on $L^2(X^b)$. In fact, by elliptic regularity
(namely using $E=E\phi(\hat H_b)$), the only issue is
the lack of decay of $w^b D_{w^b}$ at infinity, but the computation is
justified by replacing $w^b D_{w^b}$ by $\chi_0(|w^b|/C) w^bD_{w^b}$,
$\chi_0\inC^\infty_c(\mathbb{R})$ identically $1$ near $0$, and observing that
\begin{equation*}
\phi(\hat H_b)[\chi_0(w^b/C),H]w^bD_{w^b}
\end{equation*}
is uniformly bounded and
goes to $0$ strongly as $C\to\infty$.

\label{thm:non-product-prop-sing-WG}
Suppose that $H$ is a wave guide 3-body operator, so $C_a$ for $a\neq
0,1$ is discrete. Suppose also that $\lambda=k^2>0$ and that the ideal
Mourre constant is $M'_1(\lambda)>0$. Suppose also that $(H-\lambda)u$
is Schwartz. Then
\begin{enumerate}
\item
  Propagation of
singularities holds, i.e.\ $\operatorname{WF}_{3\operatorname{sc}}(u)$ is a
union of maximally extended GBBs, and the same holds for all weighted
Sobolev wave front sets, $W^{m,l}_{3\operatorname{sc}}(u).$
\item
  The high regularity radial estimates hold: if $R^+_{k^2,\operatorname{mf}}$ and
  $\cup_jR^+_{k^2,\operatorname{ff}_j}$ are disjoint from
  $\operatorname{WF}_{3\operatorname{sc}}^{*,-1/2+\delta}(u)$ for some $\delta>0$, they are also disjoint from
  $\operatorname{WF}_{3\operatorname{sc}}(u)$. The same holds with $R^+_{k^2,\operatorname{mf}}$ and
  $\cup_jR^+_{k^2,\operatorname{ff}_j}$ replaced by $R^-_{k^2,\operatorname{mf}}$ and
  $\cup_jR^-_{k^2,\operatorname{ff}_j}.$
\item
  The low regularity radial estimates hold: if a punctured
  neighborhood of $R^+_{k^2,\operatorname{mf}}$ and
  $\cup_jR^+_{k^2,\operatorname{ff}_j}$ is disjoint from
  $\operatorname{WF}_{3\operatorname{sc}}^{*,l}(u)$ for some $l<-1/2$, then so is $R^+_{k^2,\operatorname{mf}}$ and
  $\cup_jR^+_{k^2,\operatorname{ff}_j}$. The same holds for $R^+_{k^2,\operatorname{mf}}$ and
  $\cup_jR^+_{k^2,\operatorname{ff}_j}$ replaced by $R^-_{k^2,\operatorname{mf}}$ and
  $\cup_jR^-_{k^2,\operatorname{ff}_j}$.
\end{enumerate}
