\emph{(Depth zero, \Cref{thm PS in terms of new})}
     Let $\pi$ be a depth-zero irreducible cuspidal representation of $\operatorname{GL}_n(F)$. Then, the Pa\v{s}k\={u}nas-Stevens Whittaker function $W_{\pi,\psi_1}^\mathrm{PS}\in \mathcal{W}(\pi,\psi_1)$ admits the following integral representation in terms of the Whittaker newform $W_{\pi,\psi_0}^\mathrm{new}\in\mathcal{W}(\pi,\psi_0)$:

    for an appropriately normalized Haar measure $dx$ on $U_n(\mathcal{O}_F)K_n^1.$

\label{prop: non-vanishing for general n}
    Let $\pi$ be a depth-zero irreducible cuspidal representation of $G_n$. Then, 
    $$\int_{K_n^1}W_{\pi,\psi_0}^\mathrm{new}(\Sigma_n^{-1} k \Sigma_n)\ dk \neq 0.$$
    \begin{proof}
        Recall the intertwining map $\mathbf{Y}_1\in \mathrm{Hom}_{G_n}(\pi,\mathrm{Ind}_{U_n(F)}^{G_n}\psi_1)$ of \Cref{thm PS main thm}. By Frobenius reciprocity, we will regard $\mathbf{Y}_1$ as an element of $\mathrm{Hom}_{U_n(F)}(\pi,\psi_1).$ Then, the assignment 
        $$\mathbf{Y}_0:f\mapsto \mathbf{Y}_1(\Sigma_n \cdot f),\ f\in \pi\simeq \mathrm{c}\text{-}\mathrm{Ind}_{Z_nK_n}^{G_n}\tilde{\tau},$$
        is an element of $\mathrm{Hom}_{U_n(F)}(\pi,\psi_0).$ By uniqueness of the Whittaker model, uniqueness of the newform, and the fact that the Whittaker newform is supported at the identity, we must have that, up to normalization by a non-zero scalar, 
    $$W_{\pi_,\psi_0}^\mathrm{new}(g)=\mathbf{Y}_0(g\cdot f^\mathrm{new}), \mathrm{for\ all}\ g\in G_n,$$
    where $f^\mathrm{new}$ denotes the unique, up to scalars, newform in the model $\mathrm{c}\text{-}\mathrm{Ind}_{Z_nK_n}^{G_n}\tilde{\tau}$. Therefore, 
    \begin{align*}
        \int_{K_n^1}W_{\pi,\psi_0}^\mathrm{new}(\Sigma_n^{-1} k \Sigma_n)\ dk&=\int_{K_n^1}\mathbf{Y}_0(\Sigma_n^{-1}k \Sigma_n \cdot f^\mathrm{new})\ dk\\
        &=\int_{K_n^1} \mathbf{Y}_1(k \Sigma_n \cdot f^\mathrm{new})\ dk\\
        &=\mathbf{Y}_1(\mathbf{f})
    \end{align*}
    where
    $$\mathbf{f}:=\int_{K_n^1} k\Sigma_n\cdot f^\mathrm{new}\ dk \in \pi\simeq \mathrm{c}\text{-}\mathrm{Ind}_{Z_nK_n}^{G_n}\tilde{\tau}.$$
    By construction, the element $\mathbf{f}$ lies in the space of $K_n^1$-invariants $\pi^{K_n^1}$. We now need the following lemma regarding this space:
    \begin{lem}\label{lem K_n^1 invariants}
        With $\pi\simeq \mathrm{c}\text{-}\mathrm{Ind}_{Z_nK_n}^{G_n}\tilde{\tau}$ as above, there is a $K_n/K_n^1\simeq \operatorname{GL}_n(k_F)$-equivariant isomorphism of representations
        $$\iota:\pi^{K_n^1}\rightarrow \tau,\ f\mapsto f(1)$$
        with inverse given by
        $$\iota^{-1}:\tau{\rightarrow}\pi^{K_n^1},\ x\mapsto \left(f_{[x]}:g\mapsto \begin{dcases}
            \tilde{\tau}(g)x,\ &\mathrm{if}\ g\in Z_nK_n\\
            0,\ &\mathrm{if}\ g\not\in Z_nK_n
        \end{dcases}\right).$$
        \begin{proof}
            The only thing that is not immediately clear is that the support of an element in $\pi^{K_n^1}$ is contained in $Z_nK_n$. This follows by Mackey theory and cuspidality of $\tau.$ See for example the proof of \cite[Theorem $6.2$]{PrasadRaghuram2008}.
        \end{proof}
    \end{lem}
   Using the fact that $\tilde{\tau}(k)$ acts trivially for $k\in K_n^1$, we see that, up to a non-zero scalar, $\mathbf{f}(1)=f^\mathrm{new}(\Sigma_n)=f_{\left[f^\mathrm{new}(\Sigma_n)\right]}(1)$, where now $f_{\left[f^\mathrm{new}(\Sigma_n)\right]}\in \pi^{K_n^1}$ and thus by \Cref{lem K_n^1 invariants} we see that $\mathbf{f}=f_{\left[f^\mathrm{new}(\Sigma_n)\right]}$ as elements of $\pi$. Thus we are reduced to evaluating the quantity $$\mathbf{Y}_1(f_{\left[f^\mathrm{new}(\Sigma_n)\right]}).$$
   To do this, we will use the following explicit expression for $f^\mathrm{new}(\Sigma_n)$ obtained in \cite{girsch2025newformscuspidalrepresentations}:
   \begin{thm}\emph{(\cite[Theorem $4.1(3)$]{girsch2025newformscuspidalrepresentations})}\label{thm: f^new interms of Bessel}
       With $\pi\simeq \mathrm{c}\text{-}\mathrm{Ind}_{Z_nK_n}^{G_n}\tilde{\tau}$ as above, 
       $$f^\mathrm{new}(\Sigma_n)=\sum_{\overline{b}\in B_{n-1}^\mathrm{op}(k_F)}\overline{b}\cdot B_{\tau,\overline{\psi_1}}$$
       where $B_{\tau,\overline{\psi_1}}$ is the Bessel function from \emph{\Cref{sec modular Bessel functions}}.
   \end{thm}
   Combining \Cref{thm: f^new interms of Bessel} with the fact that $f_{[B_{\tau,\overline{\psi_1}}]}$ is nothing more than the Bessel vector $\mathcal{B}_{\tilde{\tau},\psi_1}$, we have that up to a non-zero scalar,
   \begin{align*}
       \mathbf{Y}_1(f_{\left[f^\mathrm{new}(\Sigma_n)\right]})&=\sum_{\overline{b}\in B_{n-1}^\mathrm{op}(k_F)}\mathbf{Y}_1(b\cdot \mathcal{B}_{\tilde{\tau},\psi_1})\\
       &=\sum_{\overline{b}\in B_{n-1}^\mathrm{op}(k_F)} B_{\tau,\psi_1}(\overline{b})
   \end{align*}
   where the second equality follows from \Cref{thm PS main thm}. Finally, we observe that the sum $\sum_{\overline{b}\in B_{n-1}^\mathrm{op}(k_F)} B_{\tau,\psi_1}(\overline{b})$ is non-zero by \cite[\S $2.2$(B$1$)]{girsch2025newformscuspidalrepresentations}. This concludes the proof of \Cref{prop: non-vanishing for general n}.
    \end{proof}

\label{prop: main in minimax}

    \begin{align}\label{eq: integral in q.}\int_{H^1}W_{\pi,\psi_0}^\mathrm{new}(\Sigma_{m,n}^{-1}h \Sigma_{m,n})\theta_{\Psi_m}(h^{-1})\ dh\neq 0.\end{align}
    \begin{proof}
        Start by noting that $\theta_{\Psi_m}(h^{-1})=\theta(h^{-1})$ for all $h\in H^1$ by definition. By \Cref{thm: new interms of PS minimax}, we have \begin{align}\label{eq: beginning of proof.}\nonumber\int_{H^1}W_{\pi,\psi_0}^\mathrm{new}(\Sigma_{m,n}^{-1}h \Sigma_{m,n})\theta_{\Psi_m}(h^{-1})\ dh&=\int_{H^1}\int_{K_n(n(m+1))}W_{\pi,\Psi_m}^\mathrm{PS}(h \Sigma_{m,n}k \Sigma_{m,n}^{-1})\theta(h^{-1})\ dk\ dh\\

        We now prove the following technical lemma regarding the auxiliary integral $\mathbf{I}$. 
        \begin{lem}\label{lem: lemma about I}
            The following hold:
            \begin{enumerate}
            \item The integral $\mathbf{I}$ satisfies the bi-equivariance property
            $$\mathbf{I}(g_1gg_2)=\theta_{\Psi_m}(g_1)\theta_{\Psi_m}(g_2)\mathbf{I}(g),\ \mathrm{for\ all}\ g_1,g_2\in (U_n(F)\cap J^1)H^1,\ g\in G_n. $$
                \item Let $g\in G_n.$
                $$\mathbf{I}(g)\neq 0\ \implies\ g\in I_{G_n}(\theta_{\Psi_m}):=\left\{g \in G_n\ |\ \substack{\theta_{\Psi_m}(\gamma)=\theta_{\Psi_m}(g\gamma g^{-1})\ \\ \forall\ \gamma \in (U_n(F)\cap J^1)H^1\cap g^{-1}(U_n(F)\cap J^1)H^1 g}\right\}.$$
                \item The support $\mathrm{supp}(\mathbf{I})\subseteq \mathbf{J}$, where the subgroup $\mathbf{J}\subseteq G_n$ is given at the beginning of \emph{\Cref{sec: simple chars}}.
            \end{enumerate}
            \begin{proof}
                We start with part $(1)$. Let $g_1,g_2\in (U_n(F)\cap J^1)H^1.$ Since $W_{\pi,\Psi_m}^\mathrm{PS}$ lies in the $((U_n(F)\cap J^1)H^1,\theta_{\Psi_m})$-isotypic subspace of $\mathcal{W}(\pi,\Psi_m)$, we immediately see that $\mathbf{I}(g_1gg_2)=\theta_{\Psi_m}(g_2)\mathbf{I}(g_1g).$ Write $g_1=n_1h_1$ with $n_1\in U_n(F)\cap J^1$ and $h_1\in H^1$.                 &=\theta(h_1)\int_{H^1}W_{\pi,\Psi_m}^\mathrm{PS}(hn_1g)\theta(h^{-1})\ dh\\

                &=\theta_{\Psi_m}(g_1)\mathbf{I}(g).\end{align*}

                 For part $(2),$ assume that $\mathbf{I}(g)\neq 0$ for $g\in G_n$. Let $\gamma$ be any element of $(U_n(F)\cap J^1)H^1\cap g^{-1}(U_n(F)\cap J^1)H^1g$. Then, we have the chain of equalities

                     &=\mathbf{I}(g\gamma g^{-1} g)\\
                     &=\theta_{\Psi_m}(g\gamma g^{-1})\mathbf{I}(g),

    \end{proof}

\label{lem: lemma about I}
            The following hold:
            \begin{enumerate}
            \item The integral $\mathbf{I}$ satisfies the bi-equivariance property
            $$\mathbf{I}(g_1gg_2)=\theta_{\Psi_m}(g_1)\theta_{\Psi_m}(g_2)\mathbf{I}(g),\ \mathrm{for\ all}\ g_1,g_2\in (U_n(F)\cap J^1)H^1,\ g\in G_n. $$
                \item Let $g\in G_n.$
                $$\mathbf{I}(g)\neq 0\ \implies\ g\in I_{G_n}(\theta_{\Psi_m}):=\left\{g \in G_n\ |\ \substack{\theta_{\Psi_m}(\gamma)=\theta_{\Psi_m}(g\gamma g^{-1})\ \\ \forall\ \gamma \in (U_n(F)\cap J^1)H^1\cap g^{-1}(U_n(F)\cap J^1)H^1 g}\right\}.$$
                \item The support $\mathrm{supp}(\mathbf{I})\subseteq \mathbf{J}$, where the subgroup $\mathbf{J}\subseteq G_n$ is given at the beginning of \emph{\Cref{sec: simple chars}}.
            \end{enumerate}
            \begin{proof}
                We start with part $(1)$. Let $g_1,g_2\in (U_n(F)\cap J^1)H^1.$ Since $W_{\pi,\Psi_m}^\mathrm{PS}$ lies in the $((U_n(F)\cap J^1)H^1,\theta_{\Psi_m})$-isotypic subspace of $\mathcal{W}(\pi,\Psi_m)$, we immediately see that $\mathbf{I}(g_1gg_2)=\theta_{\Psi_m}(g_2)\mathbf{I}(g_1g).$ Write $g_1=n_1h_1$ with $n_1\in U_n(F)\cap J^1$ and $h_1\in H^1$. Then, 
                \begin{align*}\mathbf{I}(g_1g)&=\int_{H^1} W_{\pi,\Psi_m}^\mathrm{PS}(hn_1h_1g)\theta(h^{-1})\ dh\\
                &=\theta(n_1h_1 n_1^{-1})\int_{H^1}W_{\pi,\Psi_m}^\mathrm{PS}(hn_1g)\theta(h^{-1})\ dh\\
                &=\theta(h_1)\int_{H^1}W_{\pi,\Psi_m}^\mathrm{PS}(hn_1g)\theta(h^{-1})\ dh\\
                &=\theta(h_1)\Psi_m(n_1)\int_{H^1}W_{\pi,\Psi_m}^\mathrm{PS}(hg)\theta((n_1hn_1^{-1})^{-1})\ dh\\
                &=\theta_{\Psi_m}(g_1)\mathbf{I}(g).\end{align*}
                 The second equality follows from the fact that $n_1h_1n_1^{-1}\in H^1$ together with a change of variables $h\rightsquigarrow h(n_1h_1n_1^{-1})$; the third equality follows, again, from the fact that $n_1h_1n_1^{-1}\in H^1$, and the fact that $\theta=\theta_{\Psi_m}|_{H^1}$; the fourth equality follows from the left equivariance of the $\Psi_m$-Whittaker function and a change of variables $h\rightsquigarrow n_1^{-1}hn_1\in H^1$, and the last equality follows, again, from the fact that $\theta=\theta_{\Psi_m}|_{H^1}$. This concludes the proof of part $(1)$.

                 For part $(2),$ assume that $\mathbf{I}(g)\neq 0$ for $g\in G_n$. Let $\gamma$ be any element of $(U_n(F)\cap J^1)H^1\cap g^{-1}(U_n(F)\cap J^1)H^1g$. Then, we have the chain of equalities 
                 \begin{align*}
                     \theta_{\Psi_m}(\gamma)\mathbf{I}(g)&=\mathbf{I}(g\gamma )\\
                     &=\mathbf{I}(g\gamma g^{-1} g)\\
                     &=\theta_{\Psi_m}(g\gamma g^{-1})\mathbf{I}(g),
                 \end{align*}
                 where the first equality follows from part $(1)$, and the last equality follows from the fact that $g\gamma g^{-1} \in (U_n(F)\cap J^1)H^1$ together with part $(1).$ Since by assumption $\mathbf{I}(g)\neq 0$, we get that $\theta_{\Psi_m}(\gamma)=\theta_{\Psi_m}(g\gamma g^{-1})$ for all $\gamma \in (U_n(F)\cap J^1)H^1\cap g^{-1}(U_n(F)\cap J^1)H^1g$, which in turn implies that $g\in I_{G_n}(\theta_{\Psi_m})$ as claimed.

But by \cite[($3.3.2$)]{BushnellKutzko}, applied to this integral-depth minimax setting, we have that $I_{G_n}(\theta)=\mathbf{J}$. This proves part $(3)$ and concludes the proof of the lemma.
            \end{proof}

\label{thm test vector depth zero}
    Let $\pi$ be a depth-zero irreducible cuspidal representation of $G_n$, and $\chi$ an unramified character of $F^\times$. Consider the following input data:
    \begin{itemize}
        \item The Whittaker newform $W_{\chi \pi,\psi_0}^\mathrm{new}\in \mathcal{W}(\chi\pi,\psi_0)$,
        \item The average $\widetilde{W}_{\pi^\vee,\psi_0^{-1}}^\mathrm{new}\in \mathcal{W}(\pi^\vee,\psi_0^{-1})$ of the dual Whittaker newform, 
$$\widetilde{W}_{\pi^\vee,\psi_0^{-1}}^\mathrm{new}:=\int_{U_n(\mathcal{O}_F)K_n^1} W_{\pi^\vee,\psi_0^{-1}}^\mathrm{new}( (-)\Sigma_n^{-1}y\Sigma_n) \psi_1(y)\ dy,$$
        \item The Schwartz function $\Phi_n:=\mathbf{1}_{\mathcal{O}_F^n\cdot \Sigma_n}\in \mathcal{S}(F^n)$.
    \end{itemize}
    Then, up to appropriate normalization of the quotient measure $dg$ appearing in \eqref{eq: RS integral}, we have 
    $$L(\chi\pi,\pi^\vee,s)=Z(W_{\chi \pi,\psi_0}^\mathrm{new},\widetilde{W}_{\pi^\vee,\psi_0^{-1}}^\mathrm{new}, \Phi_n,s). $$
\begin{proof}
    For ease of notation, we will assume that $\chi$ is the trivial character. The proof for general unramified $\chi$ is identical upon carrying over all the unramified twists of Whittaker functions. 

    Let $W_{\pi,\psi_1}^\mathrm{PS}\in \mathcal{W}(\pi,\psi_1)$ and $W_{\pi^\vee,\psi_1^{-1}}^\mathrm{PS}\in \mathcal{W}(\pi^\vee, \psi_1^{-1})$ be the associated depth-zero Paskunas-Stevens Whittaker functions already studied in \Cref{sec: depth zero}. Let $\Phi_0:=\mathbf{1}_{\mathcal{O}_F^n}$. We start the proof by applying the main result of \cite{TestvectorsRobNadir} (i.e. \Cref{thm: main theorem of KurinczukMatringe}) to this depth-zero setting. In particular, \Cref{thm: main theorem of KurinczukMatringe} tells us that up to appropriate normalization of the quotient measure $dg$ appearing in \eqref{eq: RS integral}, we have 
        $$L(\pi, \pi^\vee,s)=Z(W_{\pi,\psi_1}^\mathrm{PS},W_{\pi^\vee,\psi_1^{-1}}^\mathrm{PS}, \Phi_0).$$
Now, we combine this with the definition of the zeta integral \eqref{eq: RS integral} and \Cref{thm PS in terms of new} to obtain the following: 
\begin{align*}
L(\pi,\pi^\vee,s)=&\int_{U_n(F)\backslash \operatorname{GL}_n(F)} \left(\int_{U_n(\mathcal{O}_F)K_n^1} W_{\pi,\psi_0}^\mathrm{new}(\Sigma_n^{-1}gx \Sigma_n)\psi_1(x^{-1})\ dx\right)\\
    &\times\left(\int_{U_n(\mathcal{O}_F)K_n^1} W_{\pi^\vee,\psi_0^{-1}}^\mathrm{new}(\Sigma_n^{-1} gy \Sigma_n) \psi_1^{-1}(y^{-1})\ dy\right)\Phi_0(e_ng)|\det(g)|^s\ dg.
\end{align*} 
Then, the change of variables $g\rightsquigarrow gx$, together with the fact that $U_n(\mathcal{O}_F)K_n^1\subseteq K_n$ stabilizes $\Phi_0$, imply that
\begin{align*}
L(\pi,\pi^\vee,s)=&\int_{U_n(F)\backslash \operatorname{GL}_n(F)} W_{\pi,\psi_0}^\mathrm{new}(\Sigma_n^{-1}g\Sigma_n)\Phi_0(e_ng)|\det(g)|^s\\
&\times\int_{U_n(\mathcal{O}_F)K_n^1}\int_{U_n(\mathcal{O}_F)K_n^1} W_{\pi^\vee,\psi_0^{-1}}^\mathrm{new}(\Sigma_n^{-1} gx^{-1}y \Sigma_n) \psi_1(x^{-1}y)\ dx\ dy\ dg.
\end{align*}
Now, the change of variables $y\rightsquigarrow x^{-1}y$ followed by the change of variables $g\rightsquigarrow \Sigma_n^{-1} g \Sigma_n$ (\textit{cf.} \cite[\S $2.12(1)$]{jacquet1983rankin}) and the fact that $e_n\Sigma_n=e_n$ imply that 
\begin{align*}
    L(\pi,\pi^\vee,s)&=\mathrm{vol}(U_n(\mathcal{O}_F)K_n^1,dx)\delta_{B_n}(\Sigma_n^{-1})\int_{U_n(F)\backslash \operatorname{GL}_n(F)} W_{\pi,\psi_0}^\mathrm{new}(g) \\
    &\times\left( \int_{U_n(\mathcal{O}_F)K_n^1} W_{\pi^\vee,\psi_0^{-1}}^\mathrm{new}( g\Sigma_n^{-1}y \Sigma_n) \psi_1(y)\ dy\right)\Phi_n(e_ng)|\det(g)|^s\ dg.\\
    &=\int_{U_n(F)\backslash \operatorname{GL}_n(F)} W_{\pi,\psi_0}^\mathrm{new}(g)\widetilde{W}_{\pi^\vee,\psi_0^{-1}}^\mathrm{new}(g) \Phi_n(e_ng)|\det(g)|^s\ dg \\
    &\times \mathrm{vol}(U_n(\mathcal{O}_F)K_n^1,dx)\delta_{B_n}(\Sigma_n^{-1}),
\end{align*}
where $\delta_{B_n}$ denotes the modulus character of $B_n.$
Thus, up to appropriate normalization of the quotient measure $dg$, we have 
$$L(\pi,\pi^\vee,s)=Z(W_{\pi,\psi_0}^\mathrm{new},\widetilde{W}_{\pi^\vee,\psi_0^{-1}}^\mathrm{new},\Phi_n,s)$$
as claimed.
\end{proof}
