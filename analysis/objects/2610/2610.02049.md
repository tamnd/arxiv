<!-- definition-1: body_md -->
A  
  \ifstrempty{polarized algebra}{}{\ifmmodepolarized algebra\else\emph{polarized algebra}\fi} 
	\IfValueTF{i}{\index{i}}{\index{polarized algebra}} 
s a pair $(A_+,A_-)$ of associative (not necessarily unital) algebras together with a linear map
$\mu:A_+\otimes A_-\to A=A_+\oplus A_-$
such that
\begin{enumerate}
\item the projection $\mu_+:A_+\otimes A_-\to A_+$ is a morphism of left $A_+$-modules.
\item the projection $\mu_-:A_+\otimes A_-\to A_-$ is a morphism of right $A_-$-modules.
\end{enumerate}
We will usually refer to $A=A_+\oplus A_-$ as a polarized algebra
and denote $\mu(a, b)$ by $ab$.
We assume that $A_+$ is unital and $\mu(1,a)=a$ for all $a\in A_-$.
Then the normally ordered product\xspace~\cref{nop} induces a linear map
\[\N:\@ifstar{\bigoplus\nolimits}\bigoplus_{n\ge0}A^{\otimes n}\to A.\]

<!-- lemma-4: body_md -->
\label{ta=0}
Let $a\in A_n$ and $b\in A$.
Then
\begin{enumerate}
\item $\ta(ab)=q^n\cdot\ta(ba)$.
\item
If $n\ne0$ and $[a,b]=0$, then
$\ta(ab)=\ta(a)=0$.
\end{enumerate}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
\noindent\textbf{(1)}\xspace
We can assume that $b\in A_{-n}$, hence $bq^\mathbf d=q^n\cdot q^\mathbf d b$.
Therefore
\[\ta(ab)
=\frac{\tr(q^\mathbf d ab)}{\tr(q^\mathbf d)}
=\frac{\tr(bq^\mathbf d a)}{\tr(q^\mathbf d)}
=q^n\frac{\tr(q^\mathbf d ba)}{\tr(q^\mathbf d)}
=q^n\ta(ba).
\]

\noindent\textbf{(2)}\xspace
We have $\ta(ab)=q^n\ta(ba)=q^n\ta(ab)$, hence $\ta(ab)=0$.
For $b=1$, we obtain $\ta(a)=0$.
<!-- end proof-4 -->
<!-- lemma-5: body_md -->
\label{ta-exp}
Let $a\in A_k$ and $b\in A_{-k}$ for $k\ne0$ be such that $c=[b,a]\in K$.
Let $d\in A$ commute with $a,b$.
Then
\[\ta(e^{sa} e^{tb} d)
=\exp\mleft(\frac{q^kstc}{1-q^{k}}\mright)\ta(d)\in K(\hspace{-2.5pt}(q)\hspace{-2.5pt})[\![s,t]\!].\]
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
If $m\ne n$, then $\deg(a^mb^n)=k(m-n)\ne0$
and $a^mb^n$ commutes with $d$,
hence $\ta(a^mb^nd)=0$ by \cref{ta=0}.
We have $b^na=ab^n+ncb^{n-1}$, since $\ad_a(b^n)=-ncb^{n-1}$.
Therefore $x_n=\ta(a^nb^nd)$ satisfies
\[x_n
=q^k\ta(a^{n-1}b^nda)
=q^k\mleft(\ta(a^nb^nd)+nc \ta(a^{n-1}b^{n-1}d)\mright)
=q^k\mleft(x_n+ncx_{n-1}\mright).
\]
This implies $x_n=n\frac{q^kc}{1-q^k}x_{n-1}$.

<!-- lemma-12: body_md -->
\label{residue ident}
For $f\in\cC_r$ we have
an equality in $\cM(U_{r-1})$
\[

=\@ifstar{\@ifstar{\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum_{i=0}^{r-1}\Res_{x=qz_i}f\,\frac{dx}{x}.

<!-- proof-13: body_md -->
Let $u=(z_1,\dots,z_{r-1},q)\in U_{r-1}$.
Choose $\left\lvertq\right\rvert<R<\left\lvertz_{r-1}\right\rvert$, so that
\[\mleft\{x\col\left\lvertx\right\rvert=R\mright\}\subset A_u.\]
Since $f\in\cC_r\subset\cO(U_r)$,
the function $f(u,-)$ is holomorphic on $A_u$,
hence
\[
\frac{1}{2\pi i}\int_{\left\lvertx\right\rvert=R}f(u,x)\frac{dx}{x}
=[x^0]f(u,-).
\]
Likewise, since $T_rf\in\cC_r$,
\[
\frac{1}{2\pi i}\int_{|y|=R}(T_rf)(u,y)\frac{dy}{y}
=
[x^0]T_rf(u,-).
\]
After the change of variables \(x=qy\), we obtain
\[
\frac{1}{2\pi i}\int_{|y|=R}(T_rf)(u,y)\frac{dy}{y}
=
\frac{1}{2\pi i}\int_{|x|=|q|R}f(u,x)\frac{dx}{x}.
\]
Therefore
\[
[x^0](1-T_r)f(u)
=\frac{1}{2\pi i}\int_{|x|=R}f(u,x)\frac{dx}{x}
-\frac{1}{2\pi i}\int_{|x|=|q|R}f(u,x)\frac{dx}{x}.

$\left\lvertq\right\rvert^2<\left\lvertq\right\rvert^n\left\lvertz_i\right\rvert<\left\lvertz_i\right\rvert$, hence $n=1$.
The points $x=qz_i$ lie in the interior of the annulus.

\label{rem:super qm}
\Cref{adm fields:th,qm for adm oper:th} remain valid for a quadratic
superspace $H$ and its super Fock space, with $q$-traces replaced by
super $q$-traces, provided that the fixed field $X_0$ defining
$W=\,\mathchoice{\scole{1.2}}{\scole{1.2}}{\scole{1}}{\scole{1}}^{X_0}\mathchoice{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}\,$ is even.

we introduce a supercommuting auxiliary variable $s_{ij}$ with
$p(s_{ij})=p(X_{ij})$.

apply without change. The required normally ordered monomials are
recovered by extracting the coefficients of the corresponding ordered
monomials in the variables $s_{ij}$.

the same scalar functions $\Phi_k$ and $F_{2k}$ as above, multiplied

\cref{adm fields:th,qm for adm oper:th} carry over to the super setting.

\[\,\mathchoice{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}\@ifstar{\@ifstar{\@ifstar{\@ifstar{\t@prod\nolimits}\t@prod\nolimits}\@ifstar{\t@prod\nolimits}\t@prod\nolimits}\@ifstar{\@ifstar{\t@prod\nolimits}\t@prod\nolimits}\@ifstar{\t@prod\nolimits}\t@prod\nolimits}\@ifstar{\@ifstar{\@ifstar{\t@prod\nolimits}\t@prod\nolimits}\@ifstar{\t@prod\nolimits}\t@prod\nolimits}\@ifstar{\@ifstar{\t@prod\nolimits}\t@prod\nolimits}\@ifstar{\t@prod\nolimits}\t@prod*_{i=1}^r B_{n_i}(D)A_{a^i}(z)\mathchoice{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}\,
=z^{\@ifstar{\@ifstar{\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum_i n_i}
\,\mathchoice{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}\@ifstar{\@ifstar{\@ifstar{\@ifstar{\t@prod\nolimits}\t@prod\nolimits}\@ifstar{\t@prod\nolimits}\t@prod\nolimits}\@ifstar{\@ifstar{\t@prod\nolimits}\t@prod\nolimits}\@ifstar{\t@prod\nolimits}\t@prod\nolimits}\@ifstar{\@ifstar{\@ifstar{\t@prod\nolimits}\t@prod\nolimits}\@ifstar{\t@prod\nolimits}\t@prod\nolimits}\@ifstar{\@ifstar{\t@prod\nolimits}\t@prod\nolimits}\@ifstar{\t@prod\nolimits}\t@prod*_{i=1}^r \dd_z^{(n_i-1)}Y(a^i,z)\mathchoice{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}\,
=z^d Y(v,z)
\]
for $v=a^1_{-n_1}\cdots a^r_{-n_r}\one$ and $d=n_1+\dots+n_r$.
This implies that the admissible operator
\[[z^0]\,\mathchoice{\scolD{1.2}}{\scolD{1.2}}{\scolD{1}}{\scolD{1}}^{k_1}A_{a^1}(z)\cdots D^{k_r}A_{a^r}(z)\mathchoice{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}\,
\]
is a linear combination of operators $o(v)$.

<!-- corollary-3: body_md -->
For all operators $f$ in the algebra $\mleft\langleo(V)\mright\rangle_{alg}\subset\iend_0(V)$ generated by $o(V)$, the normalized $q$-trace $\ta(f)$ is quasi-modular.
Moreover, if $v_i\in V_{d_i}$ for $i\in[r]$, then
$\ta(o(v_1)\dots o(v_r))$ is quasi-modular of weight at most $\@ifstar{\@ifstar{\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum_i d_i$.
<!-- end corollary-3 -->
<!-- remark-14: body_md -->
Dong--Mason--Nagatomo \cite{dong_quasi}
prove for the Heisenberg vertex operator algebra that
\[
\ta(o(v))=\eta(\rho)^c Z_V(v,\rho)
\]
is quasi-modular.
Thus their theorem gives precisely the quasi-modularity of our
normalized $q$-trace for a single zero-mode insertion.
Our corollary gives the corresponding statement for arbitrary products
\[
\ta\mleft(o(v_1)\cdots o(v_r)\mright)\in\QM(1,\bQ).
\]
This is a normalized zero-mode $r$-point correlator,
equal to the simultaneous constant term of the normalized $r$-point function
\[[z_1^0\dots z_r^0]
\frac{\tr\mleft(q^{L_0}\@ifstar{\@ifstar{\@ifstar{\prod\nolimits}\prod\nolimits}\@ifstar{\prod\nolimits}\prod\nolimits}\@ifstar{\@ifstar{\prod\nolimits}\prod\nolimits}\@ifstar{\prod\nolimits}\prod_iY(z_i^{L_0}v_i,z_i)\mright)}{\tr\mleft(q^{L_0}\mright)}.

<!-- proposition-2: body_md -->
\label{BO bosonization}
Under the boson--fermion correspondence, we have
\[
\cT(s)=\frac1{\varsigma(s)} [z^0]\,\mathchoice{\scole{1.2}}{\scole{1.2}}{\scole{1}}{\scole{1}}^{X_s(z)}\mathchoice{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}\,,
\]
where
\[
\varsigma(s)=e^{s/2}-e^{-s/2},\qquad
X_s(z)
=\@ifstar{\@ifstar{\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum_{n\ne0}\frac{\varsigma(ns)}n a_nz^{-n}.
\]
<!-- end proposition-2 -->
<!-- proof-25: body_md -->
Let
\[
\psi(z)=\@ifstar{\@ifstar{\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum_{r\in\bZ+\frac12}\psi_rz^r,
\qquad
\psi^*(z)=\@ifstar{\@ifstar{\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum_{r\in\bZ+\frac12}\psi_r^*z^{-r}
\]
be the fermionic fields.
The standard boson--fermion correspondence gives
\[
\psi(e^{s/2}z)\psi^*(e^{-s/2}z)
=
\frac1{\varsigma(s)}
\,\mathchoice{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}\exp\left(
\@ifstar{\@ifstar{\@ifstar{\@ifstar{\t@sum\nolimits}\t@sum\nolimits}\@ifstar{\t@sum\nolimits}\t@sum\nolimits}\@ifstar{\@ifstar{\t@sum\nolimits}\t@sum\nolimits}\@ifstar{\t@sum\nolimits}\t@sum\nolimits}\@ifstar{\@ifstar{\@ifstar{\t@sum\nolimits}\t@sum\nolimits}\@ifstar{\t@sum\nolimits}\t@sum\nolimits}\@ifstar{\@ifstar{\t@sum\nolimits}\t@sum\nolimits}\@ifstar{\t@sum\nolimits}\t@sum_{n\ne0}\frac{\varsigma(ns)}n a_nz^{-n}
\right)\mathchoice{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}\,.
\]
Taking the constant coefficient in $z$ gives
\[
\frac1{\varsigma(s)}
[z^0]\,\mathchoice{\scole{1.2}}{\scole{1.2}}{\scole{1}}{\scole{1}}^{X_s(z)}\mathchoice{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}\,
=
\@ifstar{\@ifstar{\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum_{r\in\bZ+\frac12}e^{sr}\psi_r\psi_r^*.
\]
The occupied positions of $|\la\rangle$ are
$r_i=\la_i-i+\frac12$ for $i\ge1$.
Hence
\[
\@ifstar{\@ifstar{\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum_{r\in\bZ+\frac12}e^{sr}\psi_r\psi_r^*|\la\rangle
=\mleft(\@ifstar{\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\@ifstar{\sum\nolimits}\sum\nolimits}\@ifstar{\sum\nolimits}\sum_{i\ge1}e^{s(\la_i-i+\frac12)}\mright) |\la\rangle
=W_\la(s)|\la\rangle.
\]
The result follows from the characterization of $\cT(s)$ above.

<!-- corollary-4: body_md -->
\label{BO operators admissible}
For every $k\ge1$, the Bloch--Okounkov operator $\widetilde D_k$ is admissible,
of weight at most $k+1$.
<!-- end corollary-4 -->
<!-- proof-26: body_md -->
The coefficient of $s^{2m+1}$ in $X_s(z)$ is a scalar multiple of
$D^{2m}A(z)$, which has weight $2m+1$.
Hence the coefficient of $s^N$ in
$\,\mathchoice{\scole{1.2}}{\scole{1.2}}{\scole{1}}{\scole{1}}^{X_s(z)}\mathchoice{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1.2}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}{\scalebox{1}{\ensuremath\col}}\,$ is a linear combination of normally ordered
monomials of weight $N$.
Multiplication by $\varsigma(s)^{-1}$ shows that the coefficient of $s^k$
in $\cT(s)$ has weight at most $k+1$.
