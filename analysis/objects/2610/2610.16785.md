<!-- theorem-1: body_md -->
\label{thm:main-approximation}
Given a non-negative submodular
function through a value oracle and a matroid through an independence oracle, there is a deterministic algorithm that 
returns an independent set $S$ satisfying
\[
    f(S)\ge0.401f(\operatorname{OPT}).
\]
The algorithm uses polynomially many value- and independence-oracle queries.

<!-- lemma-1: body_md -->
\label{lem:base-exchange-matching}
Let $A$ and $B$ be bases of a matroid. There exists a bijection
$h:A\to B$ such that $h(u)=u$ for every $u\in A\cap B$ and
$B-h(u)+u$ is a base for every $u\in A$.

<!-- theorem-2: body_md -->
\label{thm:eme-rounding}
Let $y$ be an EME vector satisfying
$\operatorname{Mar}(y)\in P(\mathcal{M})$.  There is a deterministic algorithm that
returns an independent set $S\in\mathcal{I}$ such that
\[
  f(S)\ge F(y).
\]
It uses $O(n^2 2^{\operatorname{frac}(y)})$ value-oracle queries and
$O(n^5\log^2 n)$ independence-oracle queries.

<!-- lemma-3: body_md -->
\label{box:guarantee}
For every base $Z$, every set $Q\subseteq Z$, and every $\rho>0$, the
set $S=\operatorname{DeterministicBox}(Z,\rho)$ satisfies
\[
    f(S)
    \ge
    \frac{
        2\rho f(Q)+f(\varnothing)+\rho^2f(Z)
    }{(\rho+1)^2}.
\]

<!-- proof-1: body_md -->
Fix $Q\subseteq Z$ for the analysis.  For every maintained state
$(X_j,Y_j)$, define
\[
    Q_j=(Q\cup X_j)\cap Y_j.
\]
Initially $Q_j=Q$, and after every element has been processed,
$Q_j=X_j=Y_j$.

Consider an iteration that processes $u$.  If $u\notin Q$, deleting $u$
from $Y_j$ leaves $Q_j$ unchanged, whereas adding $u$ to $X_j$ replaces
$Q_j$ by $Q_j+u$.  Since $Q_j\subseteq Y_j-u$, submodularity gives
\[
    f(Q_j)-f(Q_j+u)
    \le f(Y_j-u)-f(Y_j)
    =b_j.
\]
Hence, the aggregate decrease in the value of the comparison sets is at
most $\sum_jw_jp_jb_j$, which is bounded by the left-hand side of
\eqref{box:add-constraint}.

If $u\in Q$, adding $u$ to $X_j$ leaves $Q_j$ unchanged, whereas deleting
$u$ from $Y_j$ replaces $Q_j$ by $Q_j-u$.  Since
$X_j\subseteq Q_j-u$, submodularity gives
\[
    f(Q_j)-f(Q_j-u)
    \le f(X_j+u)-f(X_j)
    =a_j.
\]
The aggregate decrease is therefore at most
$\sum_jw_j(1-p_j)a_j$, which is bounded by the left-hand side of
\eqref{box:delete-constraint}.

Let
\[
    F_X=\sum_jw_jf(X_j),
    \qquad
    F_Y=\sum_jw_jf(Y_j),
    \qquad
    F_Q=\sum_jw_jf(Q_j)
\]
denote the three weighted values at the beginning of an iteration.  The
increase in $F_X$ during this iteration is
$\sum_jw_jp_ja_j$, and the increase in $F_Y$ is
$\sum_jw_j(1-p_j)b_j$.  Combining the preceding case analysis with
\eqref{box:add-constraint} and \eqref{box:delete-constraint} shows that
the decrease in $F_Q$ is at most
\[
    \frac{1}{2\rho}\,\Delta F_X
    +\frac{\rho}{2}\,\Delta F_Y.
\]

Initially,
\[
    (F_X,F_Y,F_Q)
    =\bigl(f(\varnothing),f(Z),f(Q)\bigr).
\]
At termination, $X_j=Y_j=Q_j$ in every state, so all three weighted
values are equal; denote their common value by $F_{\mathrm{out}}$.
Summing the per-iteration inequality therefore yields
\[
    f(Q)-F_{\mathrm{out}}
    \le
    \frac{F_{\mathrm{out}}-f(\varnothing)}{2\rho}
    +\frac{\rho}{2}
       \bigl(F_{\mathrm{out}}-f(Z)\bigr).
\]
Multiplying by $2\rho$ and rearranging gives
\[
    (\rho+1)^2F_{\mathrm{out}}
    \ge
    2\rho f(Q)+f(\varnothing)+\rho^2f(Z).
\]
The algorithm returns a maximum-value set in the final state support, whose
value is at least $F_{\mathrm{out}}$.  This proves the lemma.

<!-- lemma-4: body_md -->
\label{lem:extended-local-search}
Every move performed by Algorithm~\ref{alg:extended-local-search} increases
$H_\lambda$ by more than $\eta$.  Moreover, the algorithm performs fewer than
\[
    \frac{(1+\lambda\ell)f(\operatorname{OPT})}{\eta}
    \le
    \frac{4rn(1+\lambda\ell)}{\varepsilon}
\]
moves and terminates at an $\eta$-locally optimal partitioned base.  In particular, for
$0\le\lambda\le\lambda_{\max}$, the number of iterations is polynomial.
<!-- end lemma-4 -->
<!-- proof-2: body_md -->
Fix a feasible exchange at the current partitioned base
$\mathcal T=(T_1,\ldots,T_\ell)$.  Let $v\in T_k$ be the element removed from its
current block, let $u\notin T$ be the entering element, and let $T_j$ be its
chosen destination block.  The underlying base changes from $T$ to $T-v+u$,
so submodularity gives
\begin{align}
    f(T-v+u)-f(T)
    =f(u\mid T-v)-f(v\mid T-v)
    \ge f(u\mid T)-f(v\mid T-v).
    \label{dir:exchange-f-change}
\end{align}
Suppose first that $j\ne k$.  The operation updates
$T_k\leftarrow T_k-v$ and $T_j\leftarrow T_j+u$, while every other block is
unchanged.  Therefore
\begin{align*}
    \Delta W
    &=g(T_k-v)-g(T_k)+g(T_j+u)-g(T_j)=-g(v\mid T_k-v)+g(u\mid T_j).
\end{align*}
Combining this identity with \eqref{dir:exchange-f-change} yields
\[
    \Delta H_\lambda
    \ge\Gamma_{\lambda,\mathcal T}(u,v,j).
\]

Suppose now that $j=k$.  The entering element is assigned to the same block
from which $v$ is removed, so the operation updates
$T_k\leftarrow T_k-v+u$ and leaves all other blocks unchanged.  Hence
\begin{align*}
    \Delta W
    &=g(T_k-v+u)-g(T_k)\\
    &=g(u\mid T_k-v)-g(v\mid T_k-v)\\
    &\ge g(u\mid T_k)-g(v\mid T_k-v),
\end{align*}
where the inequality follows from $T_k-v\subseteq T_k$ and submodularity.
Since $\lambda\ge0$, equation~\eqref{dir:exchange-f-change} again gives
\[
    \Delta H_\lambda
    \ge\Gamma_{\lambda,\mathcal T}(u,v,k).
\]

For a reassignment of $v\in T_k$ to $T_j$, the underlying base does not
change.  Hence the $f$-term is unchanged, and
\begin{align*}
    \Delta H_\lambda
    &=\lambda\bigl(
        g(T_j+v)-g(T_j)+g(T_k-v)-g(T_k)
      \bigr)\\
    &=\lambda\bigl(
        g(v\mid T_j)-g(v\mid T_k-v)
      \bigr)\\
    &=\Lambda_{\lambda,\mathcal T}(v,j).
\end{align*}
Thus every move selected by the algorithm increases $H_\lambda$ by more than
$\eta$.

It remains to bound the number of iterations.  At initialization, $\mathcal T$ is
a partition of the all-dummy base.  Since both functions ignore dummy
elements,
\[
    H_\lambda(\mathcal T)=f(\varnothing)\ge0.
\]
For any partitioned base, its underlying base becomes an independent set after
the dummy elements are removed, and hence
\[
    f(q(\mathcal T))\le f(\operatorname{OPT}).
\]
Each block $T_b$ is also independent, so \eqref{dir:g-bounds} gives
\[
    g(T_b)-g(\varnothing)\le f(\operatorname{OPT}).
\]
Consequently,
\[
    W(\mathcal T)\le\ell f(\operatorname{OPT})
    \qquad\text{and}\qquad
    H_\lambda(\mathcal T)\le(1+\lambda\ell)f(\operatorname{OPT}).
\]
Because every iteration increases $H_\lambda$ by more than $\eta$, the number
of iterations is strictly smaller than
\[
    \frac{(1+\lambda\ell)f(\operatorname{OPT})}{\eta}
    \le
    \frac{4rn(1+\lambda\ell)}{\varepsilon},
\]
where the inequality uses
$\eta=\varepsilon f(S_{\max})/(4r)$ and
$f(\operatorname{OPT})\le n f(S_{\max})$.  The direction oracle only calls the local search
with $0\le\lambda\le\lambda_{\max}$, so this is a polynomial bound.  Finally,
when the algorithm terminates, no feasible exchange has gain greater than
$\eta$ and no reassignment has gain greater than $\eta$.  This is precisely
the definition of $\eta$-local optimality.

<!-- lemma-5: body_md -->
\label{lem:weighted-exchange}
Let $\mathcal T=(T_1,\ldots,T_\ell)$ be the output of
Algorithm~\ref{alg:extended-local-search} on input $(f,g,\lambda)$, and put
$T=q(\mathcal T)$.
Then
\begin{equation}
\begin{aligned}
    &f(T\cup\operatorname{OPT})+f(T\cap\operatorname{OPT})-2f(T)+
      \lambda\left[
        \left(1-\frac1\ell\right)g(\operatorname{OPT})-g(\varnothing)
        -\left(1+\frac1\ell\right)W(\mathcal T)
      \right]\\
    &\hspace{15mm}\le
      \left(1+\frac1\ell\right)
      \frac{\varepsilon f(S_{\max})}{4}
      +\frac{f(T)}{\ell}.
\end{aligned}
\label{dir:weighted-exchange-bound}
\end{equation}
<!-- end lemma-5 -->
<!-- proof-3: body_md -->
By Lemma~\ref{lem:base-exchange-matching}, there is a bijection
$h:\operatorname{OPT}\to T$ such that $T-h(u)+u$ is a base for every $u\in\operatorname{OPT}$, and
$h(u)=u$ for every $u\in\operatorname{OPT}\cap T$.  For $u\in\operatorname{OPT}\setminus T$, define
\[
    A_u=\frac1\ell\sum_{j=1}^{\ell}\Gamma(u,h(u),j).
\]
For $u\in\operatorname{OPT}\cap T_k$, define
\[
    A_u=\frac1\ell
        \sum_{\substack{j=1\\j\ne k}}^{\ell}\Lambda(u,j).
\]
The omitted destination $j=k$ is the null action, whose gain is zero.  Thus
both definitions average over all $\ell$ destination blocks.  Set
\[
    \Sigma=\sum_{u\in\operatorname{OPT}}A_u=\Sigma_f+\lambda\Sigma_g,
\]
where $\Sigma_f$ is the total contribution of the $f$-terms and $\Sigma_g$
is the coefficient of $\lambda$.  Every non-null term is a feasible tested
move and is at most $\eta$ at termination; the null term is zero.  Since
$|\operatorname{OPT}|=r$,
\begin{equation}
    \Sigma\le r\eta
    =\frac{\varepsilon f(S_{\max})}{4}.
    \label{dir:averaged-moves}
\end{equation}

\emph{Contribution of $f$.}
Reassignment and null actions do not change the underlying base, and the
$f$-part of an exchange gain is independent of its destination block.
Therefore
\[
\begin{aligned}
    \Sigma_f
    &=\sum_{u\in\operatorname{OPT}\setminus T}f(u\mid T)
      -\sum_{u\in\operatorname{OPT}\setminus T}f(h(u)\mid T-h(u))=\sum_{u\in\operatorname{OPT}\setminus T}f(u\mid T)
      -\sum_{v\in T\setminus\operatorname{OPT}}f(v\mid T-v),
\end{aligned}
\]
where the second equality uses that $h$ maps $\operatorname{OPT}\setminus T$ bijectively
onto $T\setminus\operatorname{OPT}$.
Adding the elements of $\operatorname{OPT}\setminus T$ successively shows that the first
sum is at least $f(T\cup\operatorname{OPT})-f(T)$.  Deleting the elements of
$T\setminus\operatorname{OPT}$ successively shows that the second sum is at most
$f(T)-f(T\cap\operatorname{OPT})$: when $v$ is deleted, its current context is contained
in $T-v$.  Consequently,
\begin{equation}
    \Sigma_f\ge f(T\cup\operatorname{OPT})+f(T\cap\operatorname{OPT})-2f(T).
    \label{dir:f-contribution}
\end{equation}

\emph{Contribution of $g$.}
For $v\in T_k$, put $d_v=g(v\mid T_k-v)$.  Expanding the tested gains and
using the bijectivity of $h$ gives
\begin{equation}
\begin{aligned}
    \Sigma_g={}&
    \frac1\ell\sum_{u\in\operatorname{OPT}\setminus T}\sum_{j=1}^{\ell}
        \bigl[g(u\mid T_j)-d_{h(u)}\bigr]+\frac1\ell\sum_{k=1}^{\ell}
        \sum_{u\in\operatorname{OPT}\cap T_k}
        \sum_{\substack{j=1\\j\ne k}}^{\ell}
        \bigl[g(u\mid T_j)-d_u\bigr]\\
    ={}&\frac1\ell\sum_{j=1}^{\ell}\sum_{u\in\operatorname{OPT}}g(u\mid T_j)
        -\sum_{v\in T}d_v
        +\frac1\ell\sum_{u\in\operatorname{OPT}\cap T}d_u.
\end{aligned}
\label{dir:g-exact}
\end{equation}
In the second equality, the final sum corrects the terms with
$u\in\operatorname{OPT}\cap T_k$ and $j=k$: including this destination in the double sum
contributes $g(u\mid T_k)-d_u=-d_u$, whereas the corresponding null action
has value zero.  Successive deletion within each block gives
\begin{equation}
\begin{aligned}
    \sum_{v\in T}d_v
    &=\sum_{j=1}^{\ell}\sum_{v\in T_j}g(v\mid T_j-v)\le\sum_{j=1}^{\ell}\bigl[g(T_j)-g(\varnothing)\bigr]
     =W(\mathcal T).
\end{aligned}
    \label{dir:g-deletion}
\end{equation}
To see the inequality, fix an ordering $v_1,\ldots,v_m$ of $T_j$ and let
$R_i=T_j\setminus\{v_1,\ldots,v_i\}$.  Since
$R_i\subseteq T_j-v_i$, submodularity gives
\[
    g(v_i\mid R_i)\ge g(v_i\mid T_j-v_i),
\]
and summing over $i$ telescopes to
$g(T_j)-g(\varnothing)$.  Applying this argument to every block proves
\eqref{dir:g-deletion}.

Similarly, order $\operatorname{OPT}=\{u_1,\ldots,u_r\}$ and set
$O_i=\{u_1,\ldots,u_i\}$.  Submodularity gives
$g(u_i\mid T_j)\ge g(u_i\mid T_j\cup O_{i-1})$; summing these inequalities
gives
\begin{equation}
    \sum_{u\in\operatorname{OPT}}g(u\mid T_j)
    \ge g(\operatorname{OPT}\cup T_j)-g(T_j).
    \label{dir:g-insertion}
\end{equation}
Because the blocks are pairwise disjoint, repeated submodularity also gives
\begin{equation}
    \sum_{j=1}^{\ell}g(\operatorname{OPT}\cup T_j)
    \ge(\ell-1)g(\operatorname{OPT})+g(\operatorname{OPT}\cup T)
    \ge(\ell-1)g(\operatorname{OPT}).
    \label{dir:g-disjoint-unions}
\end{equation}
The first inequality follows from the equivalent form of
submodularity
\[
    g(A)+g(B)\ge g(A\cup B)+g(A\cap B).
\]
Indeed, when $A=\operatorname{OPT}\cup\bigcup_{i<j}T_i$ and
$B=\operatorname{OPT}\cup T_j$, pairwise disjointness of the blocks gives
$A\cap B=\operatorname{OPT}$.  Applying this inequality successively for
$j=2,\ldots,\ell$ yields the first inequality in
\eqref{dir:g-disjoint-unions}; its second inequality uses $g\ge0$.  Applying
\eqref{dir:g-deletion}--\eqref{dir:g-disjoint-unions} to
\eqref{dir:g-exact}, and using
$\sum_jg(T_j)=\ell g(\varnothing)+W(\mathcal T)$, yields
\begin{equation}
\begin{aligned}
    \Sigma_g
    &\ge\frac1\ell\sum_{j=1}^{\ell}
        \bigl[g(\operatorname{OPT}\cup T_j)-g(T_j)\bigr]
        -W(\mathcal T)
        +\frac1\ell\sum_{u\in\operatorname{OPT}\cap T}d_u\\
    &\ge\left(1-\frac1\ell\right)g(\operatorname{OPT})-g(\varnothing)
        -\left(1+\frac1\ell\right)W(\mathcal T)
        +\frac1\ell\sum_{u\in\operatorname{OPT}\cap T}d_u.
\end{aligned}
    \label{dir:g-lower-bound}
\end{equation}

\emph{Correction for common elements.}
Combining \eqref{dir:f-contribution} and \eqref{dir:g-lower-bound} gives
\begin{equation}
\begin{aligned}
    \Sigma\ge{}&f(T\cup\operatorname{OPT})+f(T\cap\operatorname{OPT})-2f(T)\\
    &+\lambda\left[
        \left(1-\frac1\ell\right)g(\operatorname{OPT})-g(\varnothing)
        -\left(1+\frac1\ell\right)W(\mathcal T)
      \right]\\
    &+\frac{\lambda}{\ell}\sum_{u\in\operatorname{OPT}\cap T}d_u.
\end{aligned}
\label{dir:g-common-correction}
\end{equation}
The last sum cannot be discarded because the deletion marginals $d_u$ need
not be non-negative.

Write $a_v=f(v\mid T-v)$ and $a_v^+=\max\{a_v,0\}$.  For every real element
$v\in T_k$, there is an unused dummy element $e$ such that $T-v+e$ is a base.
Both $f$ and $g$ ignore $e$, so local optimality for this exchange gives
\[
    -a_v-\lambda d_v\le\eta,
    \qquad\text{and hence}\qquad
    \lambda d_v\ge-a_v-\eta\ge-a_v^+-\eta.
\]
For a dummy element, $a_v=d_v=0$.  Since $|\operatorname{OPT}\cap T|\le r$, it follows that
\begin{equation}
    \lambda\sum_{u\in\operatorname{OPT}\cap T}d_u
    \ge-\sum_{v\in T}a_v^+-r\eta
    \ge-f(T)-\frac{\varepsilon f(S_{\max})}{4}.
    \label{dir:negative-correction}
\end{equation}
Indeed, let $P=\{v\in T:a_v>0\}$ and delete the elements of $P$ successively.
At each deletion, the current context is contained in $T-v$, so submodularity
implies that the actual loss is at least $a_v$.  Therefore
\[
    \sum_{v\in T}a_v^+
    =\sum_{v\in P}a_v
    \le f(T)-f(T\setminus P)
    \le f(T),
\]
where the last inequality follows from non-negativity.

Substituting \eqref{dir:negative-correction} into
\eqref{dir:g-common-correction} yields
\[
\begin{aligned}
    \Sigma\ge{}&f(T\cup\operatorname{OPT})+f(T\cap\operatorname{OPT})-2f(T)\\
    &+\lambda\left[
        \left(1-\frac1\ell\right)g(\operatorname{OPT})-g(\varnothing)
        -\left(1+\frac1\ell\right)W(\mathcal T)
      \right]\\
    &-\frac{f(T)}{\ell}
      -\frac{\varepsilon f(S_{\max})}{4\ell}.
\end{aligned}
\]
Comparing this lower bound with \eqref{dir:averaged-moves} and rearranging
proves \eqref{dir:weighted-exchange-bound}.

\label{thm:direction-oracle}
Assume \eqref{dir:g-bounds} and $g(\operatorname{OPT})\ge V$.  Then
Algorithm~\ref{alg:direction-oracle} does not return the all-dummy fallback.
It returns a direction distribution $D$ supported on at most two partitioned
bases such that, for $\mathcal T=(T_1,\ldots,T_\ell)\sim D$ and $T=q(\mathcal T)$,
\begin{align}
    \mathbb E_D\bigl[
        f(T\cup\operatorname{OPT})+f(T\cap\operatorname{OPT})-2f(T)
    \bigr]
    &\le2\varepsilon f(\operatorname{OPT}),
    \label{dir:averaged-stationarity}\\
    \mathbb E_D W(\mathcal T)
    &\ge V-g(\varnothing)
       -\left(\frac4\ell+\varepsilon\right)f(\operatorname{OPT}).
    \label{dir:averaged-growth}
\end{align}
Moreover, $\mathbb E_D W(\mathcal T)\ge b$, with equality whenever the returned
distribution uses both endpoints with positive weight.

<!-- proof-4: body_md -->
Every local-search output satisfies \eqref{dir:weighted-exchange-bound}.
For any such output with underlying base $T$, its right-hand side is bounded
uniformly by
\begin{equation}
\begin{aligned}
    \left(1+\frac1\ell\right)\frac{\varepsilon f(S_{\max})}{4}
      +\frac{f(T)}{\ell}
    &\le\left(1+\frac1\ell\right)
       \frac{\varepsilon f(S_{\max})}{4}
       +\frac{f(\operatorname{OPT})}{\ell}\\
    &\le\frac{11}{40}\varepsilon f(\operatorname{OPT})
       +\varepsilon f(\operatorname{OPT})
     =\frac{51}{40}\varepsilon f(\operatorname{OPT}).
\end{aligned}
\label{dir:local-search-error}
\end{equation}
Here we used $f(S_{\max})\le f(\operatorname{OPT})$, $\ell\ge10$, and
$1/\ell\le\varepsilon$.  The hypothesis $g(\operatorname{OPT})\ge V$ and the
definition of $b$ also give
\begin{equation}
\begin{aligned}
    &\left(1-\frac1\ell\right)g(\operatorname{OPT})-g(\varnothing)
       -\left(1+\frac1\ell\right)b=\left(1-\frac1\ell\right)(g(\operatorname{OPT})-V)
       +\varepsilon f(S_{\max})
     \ge\varepsilon f(S_{\max}).
\end{aligned}
\label{dir:threshold-slack}
\end{equation}

If $W(\mathcal T_l)\ge b$ at $\lambda_l=0$, the oracle returns
$D=\{(\mathcal T_l,1)\}$.  Equation~\eqref{dir:local-search-error} then gives
\[
\begin{aligned}
    &\mathbb E_D\bigl[f(T\cup\operatorname{OPT})+f(T\cap\operatorname{OPT})-2f(T)\bigr]\\
    &\qquad=f(q(\mathcal T_l)\cup\operatorname{OPT})+f(q(\mathcal T_l)\cap\operatorname{OPT})
      -2f(q(\mathcal T_l))
      \le\frac{51}{40}\varepsilon f(\operatorname{OPT})
      <2\varepsilon f(\operatorname{OPT}),
\end{aligned}
\]
where the weighted exchange inequality is used at $\lambda_l=0$.
This proves \eqref{dir:averaged-stationarity} for the single-output case.
Otherwise $W(\mathcal T_l)<b$.  Suppose that the output $\mathcal T_h$ at
$\lambda_{\max}$ also had $W(\mathcal T_h)<b$.  Since the coefficient of
$W$ in \eqref{dir:weighted-exchange-bound} is negative,
\eqref{dir:threshold-slack} implies
\[
\begin{aligned}
    &\left(1-\frac1\ell\right)g(\operatorname{OPT})-g(\varnothing)
       -\left(1+\frac1\ell\right)W(\mathcal T_h)>\left(1-\frac1\ell\right)g(\operatorname{OPT})-g(\varnothing)
       -\left(1+\frac1\ell\right)b
     \ge\varepsilon f(S_{\max}).
\end{aligned}
\]
Non-negativity of $f$ gives
$f(q(\mathcal T_h)\cup\operatorname{OPT})+f(q(\mathcal T_h)\cap\operatorname{OPT})
-2f(q(\mathcal T_h))\ge-2f(\operatorname{OPT})$.
Consequently, the left-hand side of
\eqref{dir:weighted-exchange-bound} for $\mathcal T_h$ is greater than
\[
\begin{aligned}
    -2f(\operatorname{OPT})+\lambda_{\max}\varepsilon f(S_{\max})
    &=-2f(\operatorname{OPT})+4n f(S_{\max})\\
    &\ge2n f(S_{\max})
     >\frac{51}{40}\varepsilon f(\operatorname{OPT}),
\end{aligned}
\]
where $f(\operatorname{OPT})\le n f(S_{\max})$ and $0<\varepsilon\le1/10$.
This contradicts \eqref{dir:local-search-error}.  Thus
$W(\mathcal T_h)\ge b$, so the all-dummy fallback is not taken.

At each bisection step, the midpoint output replaces the endpoint on the
same side of $b$.  Hence \eqref{dir:bracketing-invariant} persists, and
after $k$ steps the interval has length $\lambda_{\max}/2^k$; choosing
$k\ge\lceil\log_2(\lambda_{\max}/\tau)\rceil$ ensures
$\lambda_h-\lambda_l\le\tau$.  In particular,
$W(\mathcal T_l)<b\le W(\mathcal T_h)$, so the algorithm's weight
satisfies
\begin{equation}
\begin{aligned}
    p
    &=\frac{W(\mathcal T_h)-b}
            {W(\mathcal T_h)-W(\mathcal T_l)}\in[0,1],\\
    pW(\mathcal T_l)+(1-p)W(\mathcal T_h)&=b.
\end{aligned}
    \label{dir:exact-mixing}
\end{equation}

We next put the two exchange inequalities at the common multiplier
$\lambda_l$.  From \eqref{dir:g-bounds}, each block satisfies
$-g(\varnothing)\le g(T_j)-g(\varnothing)\le f(\operatorname{OPT})$ and
$g(\varnothing)\le f(\operatorname{OPT})$.  Since $\operatorname{OPT}$ is feasible, the same
assumption gives $g(\operatorname{OPT})\le g(\varnothing)+f(\operatorname{OPT})\le2f(\operatorname{OPT})$.
Therefore
\[
    |W(\mathcal T)|\le\ell f(\operatorname{OPT})
\]
and, by the triangle inequality,
\begin{equation}
\begin{aligned}
    &\left|\left(1-\frac1\ell\right)g(\operatorname{OPT})-g(\varnothing)
       -\left(1+\frac1\ell\right)W(\mathcal T)\right|\\
    &\quad\le\left(1-\frac1\ell\right)2f(\operatorname{OPT})+f(\operatorname{OPT})
       +\left(1+\frac1\ell\right)\ell f(\operatorname{OPT})\\
    &\quad\le(\ell+4)f(\operatorname{OPT})
      \le(\ell+4)n f(S_{\max}).
\end{aligned}
\label{dir:multiplier-coefficient-bound}
\end{equation}
The low-endpoint weighted exchange inequality has multiplier $\lambda_l$.
For the high endpoint, subtracting the difference
$(\lambda_h-\lambda_l)$ times the bracketed coefficient and applying
\eqref{dir:multiplier-coefficient-bound} gives
\begin{equation}
\begin{aligned}
    &f(q(\mathcal T_h)\cup\operatorname{OPT})+f(q(\mathcal T_h)\cap\operatorname{OPT})
      -2f(q(\mathcal T_h))\\
    &\quad+\lambda_l\left[
       \left(1-\frac1\ell\right)g(\operatorname{OPT})-g(\varnothing)
       -\left(1+\frac1\ell\right)W(\mathcal T_h)\right]\\
    &\le\frac{51}{40}\varepsilon f(\operatorname{OPT})
       +(\lambda_h-\lambda_l)(\ell+4)n f(S_{\max})\\
    &\le\frac{51}{40}\varepsilon f(\operatorname{OPT})
       +\tau(\ell+4)n f(S_{\max})\\
    &=\frac{51}{40}\varepsilon f(\operatorname{OPT})
       +\frac{\varepsilon f(S_{\max})}{4},
\end{aligned}
\label{dir:common-multiplier}
\end{equation}
Multiply the low- and high-endpoint inequalities by $p$ and $1-p$ and add.
By \eqref{dir:exact-mixing}, the weighted average of their bracketed
coefficients replaces $W(\mathcal T_l)$ and $W(\mathcal T_h)$ by $b$.
Thus
\begin{equation}
\begin{aligned}
    &\mathbb E_D\bigl[f(T\cup\operatorname{OPT})+f(T\cap\operatorname{OPT})-2f(T)\bigr]\\
    &\quad+\lambda_l\left[
      \left(1-\frac1\ell\right)g(\operatorname{OPT})-g(\varnothing)
      -\left(1+\frac1\ell\right)b\right]\\
    &\le\frac{51}{40}\varepsilon f(\operatorname{OPT})
       +(1-p)\frac{\varepsilon f(S_{\max})}{4}\\
    &\le\frac{61}{40}\varepsilon f(\operatorname{OPT})
     <2\varepsilon f(\operatorname{OPT}).
\end{aligned}
\label{dir:mixed-stationarity}
\end{equation}
By \eqref{dir:threshold-slack} and $\lambda_l\ge0$, the bracketed term
multiplied by $\lambda_l$ is non-negative.  Dropping it
proves \eqref{dir:averaged-stationarity} for the two-endpoint return.

Finally, the first return has $W(\mathcal T_l)\ge b$, while the mixed return
has $\mathbb E_D W(\mathcal T)=b$ by \eqref{dir:exact-mixing}.  Thus in every
non-fallback case,
\begin{equation}
\begin{aligned}
    \mathbb E_D W(\mathcal T)
    &\ge b\\
    &=V-g(\varnothing)
      -\frac{2V-g(\varnothing)+\ell\varepsilon f(S_{\max})}{\ell+1}\\
    &\ge V-g(\varnothing)
      -\left(\frac4\ell+\varepsilon\right)f(\operatorname{OPT}).
\end{aligned}
\label{dir:growth-calculation}
\end{equation}
For the last inequality, $V\le g(\operatorname{OPT})\le2f(\operatorname{OPT})$,
$g(\varnothing)\ge0$, and $f(S_{\max})\le f(\operatorname{OPT})$ imply
\[
    \frac{2V-g(\varnothing)+\ell\varepsilon f(S_{\max})}{\ell+1}
    \le\frac{4f(\operatorname{OPT})+\ell\varepsilon f(\operatorname{OPT})}{\ell+1}
    \le\left(\frac4\ell+\varepsilon\right)f(\operatorname{OPT}).
\]
Only the upper bound on $V$ is used, so this argument also covers
$V<0$.  Equation~\eqref{dir:growth-calculation} proves
\eqref{dir:averaged-growth}, as well as the stated bound by $b$.

\label{thm:guided-greedy}
Let $\mathcal Z$ be a guide distribution supported on at most two bases, and let
$(\mathcal C,\mathcal G)$ be the output of
Algorithm~\ref{alg:guided-greedy}.  At least one of the following holds:
\begin{enumerate}[label=(\roman*)]
\item There is a feasible candidate $S\in\mathcal C$ such that
\begin{equation}
\begin{aligned}
f(S)\ge{}&
  \bigl[(2-t_s)e^{t_s-1}-e^{-1}\bigr]f(\operatorname{OPT})\\
&-\bigl[e^{t_s-1}-e^{-1}\bigr]
  \mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cap Z)\\
&-\left[(2-t_s)e^{t_s-1}
  -\left(2+\frac{(1-t_s)^2}{2}\right)e^{-1}\right]
  \mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cup Z)\\
&-15\varepsilon f(\operatorname{OPT}).
\end{aligned}
\label{greedy:successful-bound}
\end{equation}
\item There is a guide distribution $\mathcal Z'\in\mathcal G$ that is
$\varepsilon$-stationary and satisfies
\begin{equation}
    \mathbb E_{Z'\sim\mathcal Z'}f(\operatorname{OPT}\cup Z')
    <\mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cup Z)-\varepsilon f(\operatorname{OPT}).
    \label{greedy:improved-guide}
\end{equation}
\end{enumerate}

<!-- lemma-6: body_md -->
\label{lem:greedy-feasibility}
For every state $\omega\in\Omega_i$,
$\operatorname{Mar}(y_i^\omega)\in P(\mathcal{M})$ and
$F(y_i^\omega)\le f(\operatorname{OPT})$.  Moreover, $g_i$ is non-negative and
submodular, satisfies
\[
    g_i(S)=g_i(S\cap\mathcal N_0)
    \qquad\text{for every }S\subseteq\mathcal N,
\]
and obeys \eqref{dir:g-bounds}.
<!-- end lemma-6 -->
<!-- proof-6: body_md -->
Fix a state after $i$ rounds, and let $\mathcal T^{\omega,j}$ be its selected
partitioned base in round $j<i$.  Within a round the blocks are disjoint,
and the sets appended to the list are subsets of those blocks.  Hence an
element is included by that round with probability at most $\delta$.
The union bound gives, coordinatewise,
\begin{equation}
    \operatorname{Mar}(y_i^\omega)
    \le\delta\sum_{j<i}\mathbf 1_{q(\mathcal T^{\omega,j})}
    \in P(\mathcal{M}).
    \label{greedy:state-feasibility}
\end{equation}
The vector on the right is a convex combination of base indicators and zero,
since $i\delta\le1$; downward closure of $P(\mathcal{M})$ proves the first claim.
Theorem~\ref{thm:eme-rounding} then yields
$F(y_i^\omega)\le f(\operatorname{OPT})$.

By the probabilistic definition of EME, each summand of
\eqref{greedy:auxiliary} is an expectation of the set function
$S\mapsto f(A\cup(S\setminus Z_{\omega,i}))$ over fixed realized sets $A$.
This function is non-negative and submodular, and its value at $S$ equals its
value at $S\cap\mathcal N_0$.  The weighted sum $g_i$ therefore has the same
properties.  Also
\[
    g_i(\varnothing)=\sum_{\omega\in\Omega_i}w_\omega F(y_i^\omega)
    \le f(\operatorname{OPT}).
\]
If $S$ is feasible, then $S\setminus Z_{\omega,i}$ is feasible and,
by submodularity and non-negativity,
\[
    f\bigl(A\cup(S\setminus Z_{\omega,i})\bigr)
    \le f(A)+f(S\setminus Z_{\omega,i})
    \le f(A)+f(\operatorname{OPT}).
\]
Averaging this inequality proves the remaining bound in
\eqref{dir:g-bounds}.

<!-- lemma-7: body_md -->
\label{lem:greedy-one-round}
Every round satisfies
\begin{equation}
\begin{aligned}
    &\sum_{\omega\in\Omega_{i+1}}w_\omega F(y_{i+1}^\omega)\ge
      \sum_{\omega\in\Omega_i}w_\omega F(y_i^\omega)
      +\delta\mathbb E_{\mathcal T\sim D_i}W(\mathcal T)
      -2\ell^2\delta^2f(\operatorname{OPT}).
\end{aligned}
\label{greedy:one-round}
\end{equation}
If $g_i(\operatorname{OPT})\ge V_i$, then the guide distribution induced by $D_i$ is
$\varepsilon$-stationary and
\begin{equation}
\begin{aligned}
    &\sum_{\omega\in\Omega_{i+1}}w_\omega F(y_{i+1}^\omega)\ge
      (1-\delta)\sum_{\omega\in\Omega_i}w_\omega F(y_i^\omega)
      +\delta V_i-6\varepsilon\delta f(\operatorname{OPT}).
\end{aligned}
\label{greedy:successful-round}
\end{equation}
<!-- end lemma-7 -->
<!-- proof-7: body_md -->
Write the list of an old state $\omega$ as
$\mathcal L_i^\omega=((A_{\omega,1},p_{\omega,1}),\ldots,
(A_{\omega,m_\omega},p_{\omega,m_\omega}))$.
For a fixed $\mathcal T=(T_1,\ldots,T_\ell)$, the list in
\eqref{greedy:state-update} gives the exact expansion
\[
\begin{aligned}
F(y_{i+1}^{(\omega,\mathcal T)})
={}&\sum_{J\subseteq[m_\omega]}
   \left(\prod_{a\in J}p_{\omega,a}
         \prod_{a\in[m_\omega]\setminus J}(1-p_{\omega,a})\right)\\
&\quad\cdot\sum_{H\subseteq[\ell]}
   \delta^{|H|}(1-\delta)^{\ell-|H|}
   f\left(\bigcup_{a\in J}A_{\omega,a}
      \cup\bigcup_{b\in H}(T_b\setminus Z_{\omega,i})\right).
\end{aligned}
\]
Since $f$ is non-negative, the terms with $|H|\ge2$ can be dropped.
The terms with $H=\varnothing$ sum to
$(1-\delta)^\ell F(y_i^\omega)$.  For each $b\in[\ell]$, the term with
$H=\{b\}$ is
\[
\begin{aligned}
&\delta(1-\delta)^{\ell-1}
  \sum_{J\subseteq[m_\omega]}
  \left(\prod_{a\in J}p_{\omega,a}
        \prod_{a\in[m_\omega]\setminus J}(1-p_{\omega,a})\right)
  f\left(\bigcup_{a\in J}A_{\omega,a}
      \cup(T_b\setminus Z_{\omega,i})\right)\\
&\qquad=\delta(1-\delta)^{\ell-1}
  F\bigl(y_i^\omega\vee e_{T_b\setminus Z_{\omega,i}}\bigr).
\end{aligned}
\]
Multiplying by the successor weight $w_\omega D_i(\mathcal T)$, summing over
$\omega$ and $\mathcal T$, and using \eqref{greedy:auxiliary} yields
\begin{equation}
\begin{aligned}
    &\sum_{\omega\in\Omega_{i+1}}w_\omega F(y_{i+1}^\omega)\ge (1-\delta)^\ell
      \sum_{\omega\in\Omega_i}w_\omega F(y_i^\omega)
      +\delta(1-\delta)^{\ell-1}
       \mathbb E_{\mathcal T\sim D_i}\sum_{b=1}^{\ell}g_i(T_b).
\end{aligned}
\label{greedy:zero-singleton}
\end{equation}
By \eqref{dir:block-gain}, subtracting the first two terms on the
right-hand side of \eqref{greedy:one-round} from
\eqref{greedy:zero-singleton} leaves at least
\[
\begin{aligned}
    &\bigl[(1-\delta)^\ell-1+\ell\delta\bigr]g_i(\varnothing)-\delta\bigl[1-(1-\delta)^{\ell-1}\bigr]
      \mathbb E_{\mathcal T\sim D_i}\sum_{b=1}^{\ell}g_i(T_b).
\end{aligned}
\]
The first term is non-negative.  By Lemma~\ref{lem:greedy-feasibility},
$g_i(T_b)\le2f(\operatorname{OPT})$, while
$1-(1-\delta)^{\ell-1}\le(\ell-1)\delta$.  The remaining loss is
therefore at most $2\ell(\ell-1)\delta^2 f(\operatorname{OPT})$, proving
\eqref{greedy:one-round}.

If $g_i(\operatorname{OPT})\ge V_i$, Theorem~\ref{thm:direction-oracle} gives the
stationarity assertion and
\[
    \mathbb E_{\mathcal T\sim D_i}W(\mathcal T)
    \ge V_i-g_i(\varnothing)
       -\left(\frac4\ell+\varepsilon\right)f(\operatorname{OPT}).
\]
Substitution into \eqref{greedy:one-round}, followed by
$1/\ell\le\varepsilon$ and $2\ell^2\delta\le\varepsilon$,
gives \eqref{greedy:successful-round}.

<!-- lemma-8: body_md -->
\label{lem:greedy-trajectory}
The following inequalities hold for every run, without assuming
$g_j(\operatorname{OPT})\ge V_j$ in any round.  For $i<i_s$,
\begin{equation}
\begin{aligned}
    g_i(\operatorname{OPT})\ge{}&f(\operatorname{OPT})-\mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cap Z)-\bigl(1-(1-\delta)^i\bigr)
      \mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cup Z).
\end{aligned}
\label{greedy:pre-switch-trajectory}
\end{equation}
For $i\ge i_s$,
\begin{equation}
\begin{aligned}
    g_i(\operatorname{OPT})\ge{}&(1-\delta)^{i-i_s}f(\operatorname{OPT})\\
    &-\bigl((1-\delta)^{i-i_s}-(1-\delta)^i\bigr)
      \mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cup Z)\\
    &+\delta(1-\delta)^{i-1}
      \sum_{j=i_s}^{i-1}
      \mathbb E_{\mathcal T\sim D_j}f(\operatorname{OPT}\cup q(\mathcal T)).
\end{aligned}
\label{greedy:post-switch-trajectory}
\end{equation}

<!-- proof-8: body_md -->
\emph{Before the switch, $i<i_s$.}
Fix $\omega\in\Omega_i$.  Every set in $\mathcal L_i^\omega$ is disjoint
from $Z_\omega$.  Each element belongs to at most one block per round,
and the listed blocks are activated independently with probability $\delta$.
Consequently,
\[
    \operatorname{Mar}_u(y_i^\omega)=0\quad(u\in Z_\omega),
    \qquad
    \operatorname{Mar}_u(y_i^\omega)\le1-(1-\delta)^i\quad(u\in\mathcal N).
\]
Let $B_t^\omega=\{u\in\mathcal N:\operatorname{Mar}_u(y_i^\omega)>t\}$.
By Lemma~\ref{lem:lovasz-marginal-lower-bound},
\[
\begin{aligned}
F\bigl(y_i^\omega\vee e_{\operatorname{OPT}\setminus Z_\omega}\bigr)
&\ge\int_0^1
  f\bigl((\operatorname{OPT}\setminus Z_\omega)\cup B_t^\omega\bigr)\,dt\\
&=\int_0^{1-(1-\delta)^i}
  f\bigl((\operatorname{OPT}\setminus Z_\omega)\cup B_t^\omega\bigr)\,dt
  +(1-\delta)^i f(\operatorname{OPT}\setminus Z_\omega),
\end{aligned}
\]
where the equality uses $B_t^\omega=\varnothing$ for
$t\ge1-(1-\delta)^i$.
Since $B_t^\omega\cap Z_\omega=\varnothing$, submodularity gives
\[
\begin{aligned}
f\bigl((\operatorname{OPT}\setminus Z_\omega)\cup B_t^\omega\bigr)
\ge{}&f(\operatorname{OPT}\setminus Z_\omega)
  +f(\operatorname{OPT}\cup Z_\omega\cup B_t^\omega)
  -f(\operatorname{OPT}\cup Z_\omega)\\
\ge{}&f(\operatorname{OPT}\setminus Z_\omega)-f(\operatorname{OPT}\cup Z_\omega),
\end{aligned}
\]
where the second inequality follows from non-negativity.  Substituting this
bound into the integral yields
\[
\begin{aligned}
F\bigl(y_i^\omega\vee e_{\operatorname{OPT}\setminus Z_\omega}\bigr)
\ge{}&f(\operatorname{OPT}\setminus Z_\omega)
  -\bigl[1-(1-\delta)^i\bigr]f(\operatorname{OPT}\cup Z_\omega)\\
\ge{}&f(\operatorname{OPT})-f(\operatorname{OPT}\cap Z_\omega)
  -\bigl[1-(1-\delta)^i\bigr]f(\operatorname{OPT}\cup Z_\omega).
\end{aligned}
\]
The last inequality uses submodularity and $f(\varnothing)\ge0$.
Taking the weighted sum over $\omega\in\Omega_i$, whose guide bases have
distribution $\mathcal Z$, proves \eqref{greedy:pre-switch-trajectory}.

\emph{After the switch, $i\ge i_s$.}
Fix a state $\omega\in\Omega_i$.  For each preceding round $j<i$, write
its selected partitioned base as
\[
    \mathcal T^{\omega,j}=(T_1^{\omega,j},\ldots,T_\ell^{\omega,j}),
    \qquad B_b^{\omega,j}=T_b^{\omega,j}\setminus Z_{\omega,j}.
\]
By the list representation of $y_i^\omega$, all listed blocks are activated
independently with probability $\delta$.  Hence
\begin{equation}
\begin{aligned}
F(y_i^\omega\vee e_{\operatorname{OPT}})
={}&\sum_{H_0,\ldots,H_{i-1}\subseteq[\ell]}
  \left[\prod_{j=0}^{i-1}
    \delta^{|H_j|}(1-\delta)^{\ell-|H_j|}\right]\cdot f\left(\operatorname{OPT}\cup
  \bigcup_{j=0}^{i-1}\bigcup_{b\in H_j}B_b^{\omega,j}\right).
\end{aligned}
\label{greedy:block-activation-expansion}
\end{equation}
For any fixed set $A$, the blocks $B_1^{\omega,j},\ldots,B_\ell^{\omega,j}$
are pairwise disjoint.  The union of $A$ and the independently activated
blocks has marginal probabilities $1$ on $A$, $\delta$ on
$(\bigcup_b B_b^{\omega,j})\setminus A$, and zero elsewhere.
Lemma~\ref{lem:lovasz-marginal-lower-bound} therefore gives
\begin{equation}
\begin{aligned}
\sum_{H\subseteq[\ell]}
  \delta^{|H|}(1-\delta)^{\ell-|H|}
  f\left(A\cup\bigcup_{b\in H}B_b^{\omega,j}\right)&\ge\int_0^\delta
  f\left(A\cup\bigcup_{b=1}^{\ell}B_b^{\omega,j}\right)\,dt
  +\int_\delta^1 f(A)\,dt\\
&=\delta f\left(A\cup\bigcup_{b=1}^{\ell}B_b^{\omega,j}\right)
  +(1-\delta)f(A).
\end{aligned}
\label{greedy:all-or-none}
\end{equation}
Applying \eqref{greedy:all-or-none} successively to the sums over
$H_0,\ldots,H_{i-1}$ in \eqref{greedy:block-activation-expansion}, we obtain
\begin{equation}
\begin{aligned}
F(y_i^\omega\vee e_{\operatorname{OPT}})
\ge{}&\sum_{J\subseteq\{0,\ldots,i-1\}}
  \delta^{|J|}(1-\delta)^{i-|J|}\cdot f\left(\operatorname{OPT}\cup\bigcup_{j\in J}
  \bigl(q(\mathcal T^{\omega,j})\setminus Z_{\omega,j}\bigr)\right).
\end{aligned}
\label{greedy:round-activation-expansion}
\end{equation}

We now bound the terms in \eqref{greedy:round-activation-expansion}.
The term $J=\varnothing$ contributes $(1-\delta)^i f(\operatorname{OPT})$.
For $\varnothing\ne J\subseteq\{0,\ldots,i_s-1\}$, the union of the
activated sets, denoted by $Y$, is disjoint from $Z_\omega$.
By submodularity and non-negativity,
\[
\begin{aligned}
f(\operatorname{OPT}\cup Y)
&\ge f(\operatorname{OPT})+f(\operatorname{OPT}\cup Y\cup Z_\omega)-f(\operatorname{OPT}\cup Z_\omega)\\
&\ge f(\operatorname{OPT})-f(\operatorname{OPT}\cup Z_\omega).
\end{aligned}
\]
The total weight of these terms is
$(1-\delta)^{i-i_s}\bigl[1-(1-\delta)^{i_s}\bigr]$.
For $J=\{j\}$ with $i_s\le j<i$, we have $Z_{\omega,j}=\varnothing$,
so the corresponding term is
\[
    \delta(1-\delta)^{i-1}
    f\bigl(\operatorname{OPT}\cup q(\mathcal T^{\omega,j})\bigr).
\]
Discarding all remaining terms by non-negativity gives
\begin{equation}
\begin{aligned}
F(y_i^\omega\vee e_{\operatorname{OPT}})
\ge{}&(1-\delta)^i f(\operatorname{OPT})\\
&+(1-\delta)^{i-i_s}\bigl[1-(1-\delta)^{i_s}\bigr]
  \bigl[f(\operatorname{OPT})-f(\operatorname{OPT}\cup Z_\omega)\bigr]\\
&+\delta(1-\delta)^{i-1}
  \sum_{j=i_s}^{i-1}f\bigl(\operatorname{OPT}\cup q(\mathcal T^{\omega,j})\bigr).
\end{aligned}
\label{greedy:state-trajectory-bound}
\end{equation}

Finally, the state-update rule assigns total weight $D_j(\mathcal T)$ to the
histories selecting $\mathcal T$ in round $j$, and subsequent updates preserve
that total weight.  The guide distribution is likewise preserved.  Thus
\[
    \sum_{\omega\in\Omega_i}w_\omega f(\operatorname{OPT}\cup Z_\omega)
    =\mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cup Z),
\]
and, for every $j<i$,
\[
    \sum_{\omega\in\Omega_i}w_\omega
      f\bigl(\operatorname{OPT}\cup q(\mathcal T^{\omega,j})\bigr)
    =\mathbb E_{\mathcal T\sim D_j}f\bigl(\operatorname{OPT}\cup q(\mathcal T)\bigr).
\]
Since $Z_{\omega,i}=\varnothing$, we have
$g_i(\operatorname{OPT})=\sum_{\omega\in\Omega_i}w_\omega F(y_i^\omega\vee e_{\operatorname{OPT}})$.
Taking the weighted sum of \eqref{greedy:state-trajectory-bound} over
$\omega\in\Omega_i$ and collecting coefficients proves
\eqref{greedy:post-switch-trajectory}.

<!-- lemma-9: body_md -->
\label{lem:greedy-thresholds}
Define the exact-value threshold
\begin{equation}
V_i^\star=
\begin{cases}
\begin{aligned}
    &f(\operatorname{OPT})-\mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cap Z)\\
    &\quad-\bigl(1-(1-\delta)^i\bigr)
      \mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cup Z),
\end{aligned}&i<i_s,\\[3pt]
\begin{aligned}
    &(1-\delta)^{i-i_s}f(\operatorname{OPT})\\
    &\quad-\bigl[(1-\delta)^{i-i_s}
      -(1-\delta)^i(1+\delta(i-i_s))\bigr]
      \mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cup Z),
\end{aligned}&i\ge i_s.
\end{cases}
\label{greedy:ideal-threshold}
\end{equation}
For a correctly guessed run, $V_i\ge V_i^\star-8\varepsilon f(\operatorname{OPT})$
for every $i$.  Moreover, $V_i\le V_i^\star$ before the switch and
$V_i\le V_i^\star-2\varepsilon f(\operatorname{OPT})$ after it.
<!-- end lemma-9 -->
<!-- proof-9: body_md -->
Before the switch, subtract the first branch of
\eqref{greedy:ideal-threshold} from \eqref{greedy:threshold}:
\[
\begin{aligned}
V_i-V_i^\star={}&\widehat v-f(\operatorname{OPT})
 +\bigl(\mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cap Z)-\widehat v_\cap\bigr)\\
&+\bigl(1-(1-\delta)^i\bigr)
  \bigl(\mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cup Z)-\widehat v_\cup\bigr)
 -4\varepsilon\widehat v.
\end{aligned}
\]
Equation~\eqref{greedy:correct-guesses} bounds this difference below by
$-5\varepsilon f(\operatorname{OPT})$ and above by
$2\varepsilon\widehat v-4\varepsilon\widehat v\le0$.

After the switch, the coefficient of the union-value term in
\eqref{greedy:threshold} is
\[
    (1-\delta)^{i-i_s}
       -(1-\delta)^i(1+\delta(i-i_s)).
\]
Its absolute value is at most one: each of the two terms lies in $[0,1]$,
and the second is at most
$e^{-i\delta}e^{\delta(i-i_s)}\le1$.
Thus \eqref{greedy:correct-guesses} gives
\[
    -8\varepsilon f(\operatorname{OPT})
    \le V_i-V_i^\star
    \le-5\varepsilon\widehat v
    \le-2\varepsilon f(\operatorname{OPT}),
\]
where the final inequality uses
$\widehat v\ge(1-\varepsilon)f(\operatorname{OPT})$ and
$\varepsilon\le1/10$.

<!-- lemma-10: body_md -->
\label{lem:greedy-failure}
In a correctly guessed run, suppose that $i$ is the first round with
$g_i(\operatorname{OPT})<V_i$.  Then $i\ge i_s$, and some $i_s\le j<i$ has
\begin{equation}
    \mathbb E_{\mathcal T\sim D_j}f(\operatorname{OPT}\cup q(\mathcal T))
    <\mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cup Z)-\varepsilon f(\operatorname{OPT}).
    \label{greedy:failure-decrease}
\end{equation}
The guide distribution induced by $D_j$ is $\varepsilon$-stationary.
<!-- end lemma-10 -->
<!-- proof-10: body_md -->
The pre-switch bounds
\eqref{greedy:pre-switch-trajectory} and
Lemma~\ref{lem:greedy-thresholds} imply $g_i(\operatorname{OPT})\ge V_i$ whenever
$i<i_s$, so the first failure occurs after the switch.
Suppose, toward a contradiction, that every $i_s\le j<i$ satisfies the
reverse weak inequality to \eqref{greedy:failure-decrease}.  Subtracting
\eqref{greedy:ideal-threshold} from
\eqref{greedy:post-switch-trajectory} then gives
\[
\begin{aligned}
g_i(\operatorname{OPT})-V_i^\star
&\ge\delta(1-\delta)^{i-1}
  \sum_{j=i_s}^{i-1}
  \left[\mathbb E_{\mathcal T\sim D_j}f(\operatorname{OPT}\cup q(\mathcal T))
    -(1-\delta)\mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cup Z)\right]\\
&\ge-\delta(1-\delta)^{i-1}(i-i_s)
       \varepsilon f(\operatorname{OPT})
 \ge-\varepsilon f(\operatorname{OPT}).
\end{aligned}
\]
Here the second inequality uses non-negativity of the union value and the
last uses $\delta(i-i_s)\le1$.  But
$V_i\le V_i^\star-2\varepsilon f(\operatorname{OPT})$ by
Lemma~\ref{lem:greedy-thresholds}, contradicting
$g_i(\operatorname{OPT})<V_i$.  Hence \eqref{greedy:failure-decrease} holds for some
$j<i$.  Since $i$ was the first failed round, $g_j(\operatorname{OPT})\ge V_j$.
Theorem~\ref{thm:direction-oracle} therefore makes the guide distribution
induced by $D_j$ $\varepsilon$-stationary.

<!-- corollary-2: body_md -->
\label{cor:main-successful-run}
Suppose $r\ge1$ and $f(S_{\max})>0$.
Among the calls visited by Algorithm~\ref{alg:main-deterministic}, there
is one with an $\varepsilon$-stationary guide distribution whose correctly
guessed run satisfies $g_i(\operatorname{OPT})\ge V_i$ for every $0\le i<1/\delta$.
<!-- end corollary-2 -->
<!-- proof-11: body_md -->
The initial guide distribution is $\varepsilon$-stationary by
Corollary~\ref{cor:initial-guide}, since
$\varepsilon f(S_{\max})/4\le2\varepsilon f(\operatorname{OPT})$.
At each visited call, the guess enumeration includes a correctly guessed
run by \eqref{greedy:correct-guesses}.  If this run has no failed threshold,
the desired conclusion holds.  Otherwise, Lemma~\ref{lem:greedy-failure}
provides an $\varepsilon$-stationary guide distribution
$\mathcal Z'\in\mathcal G$ satisfying
\[
    \mathbb E_{Z'\sim\mathcal Z'}f(\operatorname{OPT}\cup Z')
    <\mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cup Z)-\varepsilon f(\operatorname{OPT}).
\]
Whenever the depth limit has not been reached, the main algorithm visits
the call with guide $\mathcal Z'$ because it explores every member of
$\mathcal G$.

Follow such a child in the analysis whenever the selected correctly
guessed run has a failed threshold.  After $d\ge1$ descents, the guide
distribution $\mathcal Z^{(d)}$ on this path would satisfy
\[
\begin{aligned}
    0
    &\le\mathbb E_{Z\sim\mathcal Z^{(d)}}f(\operatorname{OPT}\cup Z)\\
    &<\mathbb E_{Z\sim\mathcal Z_0}f(\operatorname{OPT}\cup Z)
      -d\varepsilon f(\operatorname{OPT})\\
    &\le(2-d\varepsilon)f(\operatorname{OPT}).
\end{aligned}
\]
The last inequality follows from submodularity, non-negativity, and
$f(Z)\le f(\operatorname{OPT})$ for every base $Z$.  Since $f(\operatorname{OPT})>0$ and
$d_{\max}=\lceil2/\varepsilon\rceil+1$, these inequalities are impossible
for $d=d_{\max}$.  Hence, before the depth limit, a visited call has an
$\varepsilon$-stationary guide and a correctly guessed run with no failed
threshold.

<!-- lemma-11: body_md -->
\label{lem:greedy-completed-run}
In a correctly guessed run, if $g_i(\operatorname{OPT})\ge V_i$ for every
$0\le i<1/\delta$, then the final states satisfy the right-hand side of
\eqref{greedy:successful-bound} with
$\sum_{\omega\in\Omega_{1/\delta}}w_\omega F(y_{1/\delta}^\omega)$
in place of $f(S)$.
<!-- end lemma-11 -->
<!-- proof-12: body_md -->
Lemma~\ref{lem:greedy-one-round} and
$V_i\ge V_i^\star-8\varepsilon f(\operatorname{OPT})$ yield
\[
\begin{aligned}
    \sum_{\omega\in\Omega_{i+1}}w_\omega F(y_{i+1}^\omega)\ge(1-\delta)
      \sum_{\omega\in\Omega_i}w_\omega F(y_i^\omega)
      +\delta V_i^\star-14\varepsilon\delta f(\operatorname{OPT}).
\end{aligned}
\]
Iterating, using $F(0)=f(\varnothing)\ge0$ and the fact that the
geometric weights sum to at most one, gives
\begin{equation}
\begin{aligned}
    \sum_{\omega\in\Omega_{1/\delta}}w_\omega F(y_{1/\delta}^\omega)\ge\delta\sum_{i=0}^{1/\delta-1}
       (1-\delta)^{1/\delta-1-i}V_i^\star
       -14\varepsilon f(\operatorname{OPT}).
\end{aligned}
\label{greedy:threshold-convolution}
\end{equation}

To compare this finite sum with the coefficients in
\eqref{greedy:successful-bound}, put $t=i_s\delta$ and, for
$s\in[0,1]$, define
\[
v_t(s)=
\begin{cases}
\begin{aligned}
    &f(\operatorname{OPT})-\mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cap Z)\\
    &\quad-(1-e^{-s})\mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cup Z),
\end{aligned}&s<t,\\[3pt]
\begin{aligned}
    &e^{-(s-t)}f(\operatorname{OPT})\\
    &\quad-\bigl[e^{-(s-t)}-e^{-s}(1+s-t)\bigr]
      \mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cup Z),
\end{aligned}&s\ge t.
\end{cases}
\]
The function $v_t$ is the continuous counterpart of
\eqref{greedy:ideal-threshold} at the actual switch time $t$.  For every
integer $0\le k\le1/\delta$,
\begin{equation}
    0\le e^{-k\delta}-(1-\delta)^k\le\delta.
    \label{greedy:power-error}
\end{equation}
Indeed,
$0\le-\log(1-\delta)-\delta\le\delta^2/[2(1-\delta)]$;
use $1-e^{-x}\le x$, $k\delta\le1$, and $\delta\le1/2$.
Equation~\eqref{greedy:power-error} and
$\mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cap Z)\le f(\operatorname{OPT})$,
$\mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cup Z)\le2f(\operatorname{OPT})$ imply
\begin{equation}
    |V_i^\star-v_t(i\delta)|\le7\delta f(\operatorname{OPT}),
    \qquad |V_i^\star|\le4f(\operatorname{OPT}).
    \label{greedy:threshold-approximation}
\end{equation}
To verify the first bound, write $m=i-i_s$ after the switch.  Before the
switch, only the coefficient of the union expectation changes, so the error
is at most $2\delta f(\operatorname{OPT})$.  After the switch, the coefficient of
$f(\operatorname{OPT})$ changes by at most $\delta$, while
\[
\begin{aligned}
\left|(1-\delta)^m-(1-\delta)^i(1+\delta m)
  -e^{-m\delta}+e^{-i\delta}(1+\delta m)\right|\le\delta+(1+\delta m)\delta\le3\delta.
\end{aligned}
\]
Since the union expectation is at most $2f(\operatorname{OPT})$, the total post-switch
error is at most $7\delta f(\operatorname{OPT})$.  The bound $|V_i^\star|\le4f(\operatorname{OPT})$
follows directly before the switch; after the switch, both
$(1-\delta)^m$ and $(1-\delta)^i(1+\delta m)$ lie in $[0,1]$.
Also, \eqref{greedy:power-error} and $e^\delta-1\le2\delta$ give
\[
    \left|(1-\delta)^{1/\delta-1-i}-e^{i\delta-1}\right|
    \le3\delta.
\]
Indeed, with $k=1/\delta-1-i$, the left-hand side is at most
$|e^{-k\delta}-(1-\delta)^k|+(e^\delta-1)e^{i\delta-1}\le3\delta$.
The function $v_t$ has absolute value at most $4f(\operatorname{OPT})$ before the switch
and at most $3f(\operatorname{OPT})$ after it.  Its derivative with respect to $s$ has
absolute value at most $2f(\operatorname{OPT})$ before the switch and at most
$5f(\operatorname{OPT})$ after it.  Hence $s\mapsto e^{s-1}v_t(s)$ is
$8f(\operatorname{OPT})$-Lipschitz on each side.  Since $t=i_s\delta$ lies on the
summation grid, summing the standard $L\delta^2/2$ error over the
$1/\delta$ grid intervals bounds the left Riemann-sum error by
$4\delta f(\operatorname{OPT})$.  The weight error contributes at most
$3\delta\cdot4f(\operatorname{OPT})=12\delta f(\operatorname{OPT})$ after summation, and the threshold
error contributes at most $7\delta f(\operatorname{OPT})$.  Combining these estimates with
\eqref{greedy:threshold-approximation} gives
\begin{equation}
\begin{aligned}
    \left|\delta\sum_{i=0}^{1/\delta-1}
       (1-\delta)^{1/\delta-1-i}V_i^\star
       -\int_0^1e^{s-1}v_t(s)\,ds\right|\le(12+7+4)\delta f(\operatorname{OPT})
      =23\delta f(\operatorname{OPT}).
\end{aligned}
\label{greedy:integral-error}
\end{equation}

Integrating the two branches directly gives
\begin{equation}
\begin{aligned}
\int_0^1 e^{s-1}v_t(s)\,ds
={}&\bigl[(2-t)e^{t-1}-e^{-1}\bigr]f(\operatorname{OPT})\\
&-\bigl[e^{t-1}-e^{-1}\bigr]
  \mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cap Z)\\
&-\left[(2-t)e^{t-1}
  -\left(2+\frac{(1-t)^2}{2}\right)e^{-1}\right]
  \mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cup Z).
\end{aligned}
\label{greedy:continuous-integral}
\end{equation}
Finally, $0\le t-t_s<\delta$.  On the interval where the two switch
times assign different branches, the absolute values of the branches sum
to at most $7f(\operatorname{OPT})$ by the branch bounds above.  To check the second
part, write $U=\mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cup Z)$.  For $s\ge\sigma$, the
post-switch branch satisfies
\[
    \frac{\partial v_\sigma(s)}{\partial\sigma}
    =e^{\sigma-s}f(\operatorname{OPT})-\bigl(e^{\sigma-s}+e^{-s}\bigr)U,
\]
whose absolute value is at most $f(\operatorname{OPT})+2U\le5f(\operatorname{OPT})$.
Thus the interval on which the branch choices differ contributes at most
$7(t-t_s)f(\operatorname{OPT})$, and their common post-switch interval contributes at
most $5(t-t_s)f(\operatorname{OPT})$.  Hence replacing $t$ by $t_s$ in
\eqref{greedy:continuous-integral} changes its value by at most
$12\delta f(\operatorname{OPT})$.  Equations~\eqref{greedy:threshold-convolution}
and~\eqref{greedy:integral-error} therefore yield the desired expression
with error $(14\varepsilon+35\delta)f(\operatorname{OPT})$.  Since
$\delta\le\varepsilon/(2\ell^2)$ and $\ell\ge10$,
$35\delta<\varepsilon$, proving the stated
$15\varepsilon f(\operatorname{OPT})$ error.

\label{lem:box-distribution}
Let $\mathcal Z$ be an $\varepsilon$-stationary guide distribution, and let
$\mathcal C_{\mathrm{box}}$ contain the output of
$\operatorname{DeterministicBox}(Z,\rho)$ for every positive-weight base

\ge{}\frac{4\rho+\rho^2}{2(\rho+1)^2}

       \mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cup Z)
       -\varepsilon f(\operatorname{OPT}).
\end{aligned}
\label{analysis:box-distribution-bound}
\end{equation}

Apply Lemma~\ref{box:guarantee} to each support base with
$Q=\operatorname{OPT}\cap Z$.  The largest returned value is at least the weighted
average of those values, so
\[
\begin{aligned}
\max_{S\in\mathcal C_{\mathrm{box}}}f(S)
\ge{}\frac{2\rho}{(\rho+1)^2}
       \mathbb E_{Z\sim\mathcal Z}f(\operatorname{OPT}\cap Z)

 +\frac{\rho^2}{(\rho+1)^2}\mathbb E_{Z\sim\mathcal Z}f(Z).
\end{aligned}
\]
