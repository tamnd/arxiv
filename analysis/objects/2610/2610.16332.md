<!-- theorem-1: body_md -->
\label{thm:main}
The totally nonnegative wonderful compactification
$\overline{G_{\geqslant0}}$, equipped with its cell decomposition,
is a regular CW complex. In particular,
$\overline{G_{\geqslant0}}$ is homeomorphic to a closed ball
of dimension $\dim G$.
<!-- end theorem-1 -->

The wonderful compactification and its totally nonnegative part
admit a positive Bruhat atlas.
<!-- end theorem-2 -->

<!-- proposition-1: body_md -->
\label{combinatorial_atlas}
The restriction of $\varphi$ to $\mathcal {Q}$ is an order-reversing embedding with convex image
$$\{\tilde w\in\tilde W\big|\tilde w\leq r_1^\flat r_2^\sharp c_{I^\natural}(r_1^{-1}w_0)^\sharp (r_2^{-1}w_0)^\flat\text{ for some }r_1,r_2\in W\}.$$


\begin{proof}
Denote $$\mathcal {C} := \{\tilde w\in\tilde W\big|\tilde w\leq r_1^\flat r_2^\sharp c_{I^\natural}(r_1^{-1}w_0)^\sharp (r_2^{-1}w_0)^\flat,\text{ for some }r_1,r_2\in W\}.$$
Note that $\mathcal {C}$ is a union of some Bruhat lower ideals and thus is convex. It is sufficient to show that $\varphi(\mathcal {Q}) = \mathcal {C}$. 

Suppose that $(J;x,y;v,w)\in\mathcal {Q}$, we have $y^J\leq w^J$ and $v\leq x$. There exist $r_1,r_2\in W$ such that $v\leq r_1\leq x\leq xy_J^{-1}$ and $y^J\leq r_2\leq w^J\leq w$. Note that $c_{(I\setminus J)^\natural}$ commutes with $r^\flat$ and $r^\sharp$ for any $r\in W_J$. We may write
$$\varphi(J;x,y;v,w) = v^\flat (y^J)^\sharp c_{(I\setminus J)^\natural}(y_Jx^{-1}w_0)^\sharp (w^{-1}w_0)^\flat,$$
and we have
$\varphi(J;x,y;v,w)\leq r_1^\flat r_2^\sharp c_{I^\natural}(r_1^{-1}w_0)^\sharp (r_2^{-1}w_0)^\flat$. Thus, $\varphi(\mathcal {Q})\subseteq\mathcal {C}$.

Conversely, for $\tilde w\in\mathcal {C}$, we see that there exist $(J;x,y;v,w)\in\mathcal {A}\diamond\mathcal {A}$ such that $\varphi(J;x,y;v,w) = \tilde w$. Suppose that $\varphi(J;x,y;v,w)\leq
r_1^\flat r_2^\sharp c_{I^\natural}
(r_1^{-1}w_0)^\sharp(r_2^{-1}w_0)^\flat$. Then $v\leq r_1^J\leq x$ and $y^J\leq r_2^J\leq w^J$. We have $y^J\leq w^J$ and $v\leq x$. Therefore, $(J;x,y;v,w)\in\mathcal {Q}$. We conclude that $\varphi(\mathcal {Q}) = \mathcal {C}$.
\end{proof}

Let $G = PGL_2$ and $I = \{1\}$. The atlas group $\tilde G = SL_4$ is of type $A_3$, and $\tilde I = \{1^\sharp,1^\natural,1^\flat\}$, where $1^\natural$ is the middle vertex in the Dynkin diagram of $\tilde G$. In this example, we write down explicitly the atlas maps.

An element in $\mathcal O_{(1,1)}$ can be written as $(y_{1}(a),x_1(-c))\cdot\pi(b)$. We have $u_1 = y_1(a)$ and $u_2 = x_1(-c)$. Then $\iota(\tau_{1,-}(u_2))^{-1} = \sigma_{1,+}(u_1) = e$, $\tau_{1,+}(u_2)^{-1} = x_1(c)$ and $\iota(\sigma_{1,-}(u_1)) = x_1(a)$. The atlas map is
\[
\phi_{(1,1)}(y_1(a),x_1(-c);b)
=\left\{
\begin{array}{ll}
   \tilde y_{1^\natural}(b^{-1})\tilde \alpha^\vee_{1^\natural}(b)\tilde x_{1^\flat}(c)\tilde x_{1^\sharp}(a)\dot{\tilde s}_{1^\sharp}\dot{\tilde s}_{1^\flat}\tilde{B}^+/\tilde{B}^+, &\text{if }b\neq0, \\
    \dot{\tilde s}_{1^\natural}\tilde x_{1^\flat}(c)\tilde x_{1^\sharp}(a)\dot{\tilde s}_{1^\sharp}\dot{\tilde s}_{1^\flat}\tilde{B}^+/\tilde{B}^+, &\text{if }b=0.
\end{array}
\right.
\]

Using the identity $\tilde y_{1^\natural}(b^{-1})\tilde\alpha^\vee_{1^\natural}(b) = \tilde x_{1^\natural}(b)\dot{\tilde s}_{1^\natural}\tilde x_{1^\natural}(b^{-1})$, we see that the two expressions are compatible when $b 
   \ifbool{@display}{\longrightarrow}{\rightarrow} 
    0$.



An element in $\mathcal O_{(s_1,1)}$ can be written as $(x_1(a)\dot s_1,x_1(-c))\cdot\pi(b)$. We have $u_1 = x_1(a)$ and $u_2 = x_1(-c)$. <!-- proposition-2: body_md -->
\label{image_atlas_map}
For $r_1,r_2\in W$, the atlas map 
\[
\phi_{(r_1,r_2)}:\mathcal {O}_{(r_1,r_2)}
 
   \ifbool{@display}{\longrightarrow}{\rightarrow} 
   
\mathring{\tilde{\mathcal B}}_{\varphi(\xi_{(r_1,r_2)})}
\]
is an isomorphism
\begin{proof}
From the identity $\tilde y_{i^\natural}(a^{-1})\tilde\alpha_{i^\natural}(a) = \tilde x_{i^\natural}(a)\dot{\tilde s}_{i^\natural}\tilde x_{i^\natural}(a^{-1})$, we see that if we define
\[
n(\underline{a}) = \prod_{i\in I}\tilde x_{i^\natural}(a)\dot{\tilde s}_{i^\natural},
\]
then $\phi_{(r_1,r_2)}(u_1,u_2;\underline{a})$ can be written as
\[
\bigl(\iota(\tau_{r_2,-}(u_2)\dot r_2)\bigr)^{-1,\sharp}\; \bigl(\sigma_{r_1,+}(u_1)\dot r_1 \bigr)^\flat\:n(\underline{a})\; \bigl(\tau_{r_2,+}(u_2)\dot r_2\bigr)^{-1,\flat}\; \bigl(\iota(\sigma_{r_1,-}(u_1)\dot r_1 )\bigr)^\sharp\;\dot w_0^\sharp \dot w_0^\flat\tilde B^+/\tilde B^+ . \bigr.
\]
Therefore $\phi_{(r_1,r_2)}$ is algebraic.

We construct the inverse map. Since \[
\ell(r_1^\flat r_2^\sharp c_{I^\natural}(r_1^{-1}w_0)^\sharp(r_2^{-1}w_0)^\flat) = \ell(r_1)+\ell(r_2)+\ell(c_{I^\natural})+\ell(r_1^{-1}w_0)+\ell(r_2^{-1}w_0),
\]
every element in $\tilde x\tilde B^+/\tilde B^+\in \mathring{\tilde{\mathcal B}}_{\varphi(\xi_{(r_1,r_2)})}$ can be uniquely written as 
\[\tilde x\tilde B^+/\tilde B^+=
(g_1\dot r_1)^\flat (g_2\dot r_2)^\sharp n(\underline{a})(\dot r_1^{-1}h_1)^\sharp(\dot r_2^{-1}h_2)^\flat\dot w_0^\sharp\dot w_0^\flat\tilde B^+/\tilde B^+
\]
for some $g_1\in \dot r_1U^-\dot r_1^{-1}\cap U^+$, $g_2\in\dot r_2U^-\dot r_2^{-1}\cap U^+$, $\underline{a}\in\mathbb C^I$, $h_1\in \dot r_1U^+\dot r_1^{-1}\cap U^+$, and $h_2\in \dot r_2U^+\dot r_2^{-1}\cap U^+$.

The morphisms \[
(\sigma_{r_1,+},\sigma_{r_1,-}):(\dot r_1U^-\dot r_1^{-1}\cap U^+)\times (\dot r_1U^-\dot r_1^{-1}\cap U^-) 
   \ifbool{@display}{\longrightarrow}{\rightarrow} 
   \dot r_1U^-\dot r_1^{-1},
\]
\[
(\tau_{r_2,-},\tau_{r_2,+}):(\dot r_2U^+\dot r_2^{-1}\cap U^-)\times (\dot r_2U^+\dot r_2^{-1}\cap U^+) 
   \ifbool{@display}{\longrightarrow}{\rightarrow} 
   \dot r_2U^+\dot r_2^{-1}
\]
are invertible. Let \[u_1 = (\sigma_{r_1,-},\sigma_{r_1,+})^{-1}(g_1,\iota(h_1)),\enspace u_2 = (\tau_{r_2,-},\tau_{r_2,+})^{-1}(\iota(g_2)^{-1},h_2^{-1}).\]
It is easy to check that $\phi_{(r_1,r_2)}^{-1}(\tilde x\tilde B^+/\tilde B^+) = (u_1,u_2;\underline{a})$. The conclusion follows.
\end{proof}

<!-- definition-1: body_md -->
Let $X=\bigsqcup_{\gamma}X_\gamma$ be a stratification of a real
algebraic variety, or of an ind-variety with finite-dimensional strata. Let $X_{\geqslant0}\subseteq X(\mathbb R)$ be a
closed connected subset, and set
$X_{\gamma,>0}=X_{\geqslant0}\cap X_\gamma(\mathbb R)$.
We call $X_{\geqslant0}=\bigsqcup_\gamma X_{\gamma,>0}$ a
\emph{remarkable polyhedral space} if the following conditions
hold:
\begin{enumerate}
    \item (Connected components) $X_{\gamma,>0}$ is either empty or a connected component of $X_\gamma(\mathbb R)$;
    \item (Cell structure) If $X_{\gamma,>0} \neq \emptyset$, then $X_{\gamma,>0} \simeq\mathbb R_{>0}^{d_\gamma}$ is a semi-algebraic cell, where $d_\gamma=\dim_{\mathbb R} X_\gamma$;
    \item (Cellular decomposition) The Hausdorff closure $\overline{X_{\gamma,>0}}$ is the disjoint union of totally nonnegative cells $\bigsqcup_{\gamma'\leq\gamma} X_{\gamma',>0}$, where $\gamma'\leq\gamma$ if $X_{\gamma'}\subseteq\overline{X_{\gamma}}$.
\end{enumerate}
We say that $X_{\geqslant0}$ has the \emph{regularity property} if,
for every nonempty $X_{\gamma,>0}$, there is a homeomorphism from a closed ball to $\overline{X_{\gamma,>0}}$ that takes the interior onto $X_{\gamma,>0}$.

<!-- proposition-4: body_md -->
\label{positive_atlas}
Suppose that $\{\phi_f\}$ is a positive atlas on
$(M,\mathcal Y,M_{\geqslant0})$. Then every $M_{y,>0}$ is nonempty,
and
$M_{\geqslant0}=\bigsqcup_{y\in\mathcal Y}M_{y,>0}$ is a
remarkable polyhedral space. Moreover, for every $y$, the
Hausdorff closure $\overline{M_{y,>0}}$ is a topological
manifold with boundary
\[
\partial\overline{M_{y,>0}}
   =\bigsqcup_{y'<y}M_{y',>0}.
\]
<!-- end proposition-4 -->
<!-- proof-4: body_md -->
We denote $\bar\phi_f(y)$ a Bruhat interval of $\tilde W$ such that $\phi_f(\mathring M_y\cap U_f) =\mathring{\tilde\mathcal {B}}_{\bar\phi_f(y)}$. For $y\in\mathcal {Y}$, we take $f\in F$ such that $\mathring M_y\cap U_f\neq\emptyset$. By assumption, we have $\phi_f(M_{y_0,>0}) =\tilde\mathcal {B}_{\bar\phi_f(y_0),>0}$. Since $\phi_f$ is a stratified embedding, we have
$$\phi_f(M_{y,>0})=\phi_f\left(\overline{M_{y_0,>0}}\bigcap\mathring M_y\right) = \overline{\phi_f(M_{y_0,>0})}\bigcap \mathring{\tilde\mathcal {B}}_{\bar\phi_f(y)} = \tilde\mathcal {B}_{\bar\phi_f(y),>0}.$$
Therefore, $M_{y,>0}$ is a connected component of $\mathring M_y\cap U_f$ and a semi-algebraic cell. We also have $$\dim M_{y,>0} = \dim \tilde\mathcal {B}_{\bar\phi_f(y)}=\dim(\mathring M_y\cap U_f).$$
Since $\{U_f\}_{f\in F}$ is an open covering of $M$, we have $\dim\mathring M_y(\mathbb R) = \dim M_{y,>0}$.

Since $U_f$ is open in $M$, we also have $M_{y,>0}$ is open in $\mathring M_y(\mathbb R)$. Since $M_{y,>0} = \mathring M_y\cap M_{\geqslant0}$ is closed, $M_{y,>0}$ is a connected component of $\mathring M_y(\mathbb R)$.

For $y,y'\in\mathcal {Y}$ such that $y'\leq y$, since $\{U_f\}_{f\in F}$ is an open covering of $M$, there is an $f\in F$ such that $U_f\cap\mathring M_{y'}\neq\emptyset$. Since $\mathring M_{y'}$ is contained in the Zariski closure of $\mathring M_y$, we have $U_f\cap\mathring M_y\neq\emptyset$. 
Since $\tilde\mathcal {B}_{\geqslant0}$ is a remarkable polyhedral space, we have 
$$\phi_f(M_{y',>0})\subseteq \phi_f\left(\overline{M_{y,>0}}\bigcap U_f\right),$$
which shows that $M_{y',>0}\subseteq\overline{M_{y,>0}}$. Since $\overline{\mathring{M}}_y = \bigsqcup_{y'\leq y}\mathring{M}_{y'}$, we have $\overline{M_{y,>0}}\subseteq\bigsqcup_{y'\leq y}M_{y',>0}$. This verifies the closure relation condition.

Finally, since $\overline{\phi_f(M_{y,>0})}$ is a closed ball with interior $\phi_f(M_{y,>0})$, the collection $\left(\overline{M_{y,>0}}\cap U_f,\phi_f\right)_{f\in F}$ gives an Euclidean chart on $\overline{M_{y,>0}}$, making it a topological manifold, with $M_{y,>0}$ lying in the interior. For $y'<y$, we have 
$$\phi_f(M_{y',>0})\subseteq\partial\overline{\phi_f(M_{y,>0})} = \partial\left(\phi_f\left(\overline{M_{y,>0}}\bigcap U_f\right)\right). $$
Therefore, $\partial\overline{M_{y,>0}} = \bigsqcup_{y'<y} M_{y',>0}$.

<!-- lemma-5: body_md -->
\label{levi_nonnegative}
Suppose that $v,r,w\in W^J$ such that $v\leq r\leq w$. 
Let $a=\text{Res}_J(v,r)$ and $b=\text{Res}_J(r,w)$.
For $g\in G_{\mathbf v_+,\mathbf w,>0}$, we have $\pi_J(\dot r^{-1}g)\in U_{b,>0}^-T_{>0}U_{a^{-1},>0}^+$.

\begin{proof}
We prove the statement by induction on $l(w)$. The base case where $l(w) = 0$ is trivial. 

Let $\mathbf w = (s_{i_1},s_{i_2},\cdots, s_{i_n})$, and write $g = g_1g_2\cdots g_n$ as in the definition of $G_{\mathbf v_+,\mathbf w,>0}$. Set $w' = s_{i_1}w$, $r' = s_{i_1}\circ_l r$, $v' = s_{i_1}\circ_l v$, $a' = \text{Res}_J(v',r')$, and $b' = \text{Res}_J(r',w')$. Then $v',r',w'\in W^J$ and $v'\leq r'\leq w'$.

We divide the computation into several cases.
\begin{enumerate}[label = (\roman*)]
    \item $s_{i_1}r>r$ and $s_{i_1}r\in W^J$. Then $r' = r$ and $v\leq r\leq s_{i_1}w$. This forces $g_1\in y_{i_1}(\mathbb R_{>0})$ and $b' = b$. In this case, we have $\dot r^{-1}y_{i_1}(\mathbb R_{>0})\dot r\subseteq U_{P_J^-}$, and 
    $$\dot r^{-1}G_{\mathbf v_+,\mathbf w,>0}\subseteq U_{P_J^-}\left(\dot r^{\prime,-1}G_{\mathbf v_+,\mathbf w',>0}\right).$$


    \item $s_{i_1}r = rs_j$ for some $j\in J$. We still have $r' = r$, $v\leq r\leq s_{i_1}w$, and $g_1\in y_{i_1}(\mathbb R_{>0})$. Now $b = s_j*b'$. In this case, we have $\dot r^{-1}y_{i_1}(\mathbb R_{>0})\dot r = y_j(\mathbb R_{>0})$. Here, the sign is preserved according to \cite[Lemma 1.3.12]{He05}. Then
    $$\dot r^{-1}G_{\mathbf v_+,\mathbf w,>0}\subseteq U^-_{s_j,>0}\left(\dot r^{\prime,-1}G_{\mathbf v_+,\mathbf w',>0}\right).$$

    
    \item $s_{i_1}r<r$ and $g_1 = \dot s_{i_1}$. Then $r' = s_{i_1}r\in W^J$, $v' = s_{i_1}v$, and $a' = a$, $b' = b$. We have 
        $$\dot r^{-1}G_{\mathbf v_+,\mathbf w,>0}=\dot r^{\prime,-1}G_{\mathbf v'_+,\mathbf w',>0}.$$


    \item $s_{i_1}r<r$, $g_1= y_{i_1}(m)$ for $m>0$, and $s_{i_1}v\in W^J$. In this case, $r'=s_{i_1}r\in W^J$, $a' = a$, and $b' = b$.
    
    Using the identity $$y_{i_1}(m) = x_{i_1}(m^{-1})\dot s_{i_1}x_{i_1}(m)\alpha^\vee_{i_1}(m),$$
    we have $$\dot r^{-1}gB^+/B^+ = (\dot r^{-1}x_{i_1}(m^{-1})\dot r)\dot r^{\prime,-1}x_{i_1}(m)g'B^+/B^+,$$
    for some $g'\in G_{\mathbf v_+,\mathbf w',>0}$. By assumption, $\dot r^{-1}x_{i_1}(m^{-1})\dot r\in U_{P_J^-}$.
    
    Let $w_{J,0}$ denote the longest element in $W_J$. Since $v,v',w'\in W^J$, we have $$G_{\mathbf{vw_{J,0}}_+,\mathbf{w'w_{J,0}},>0} = G_{\mathbf v_+,\mathbf{w'},>0}\dot w_{J,0},\enspace G_{\mathbf{v'w_{J,0}}_+,\mathbf{w'w_{J,0}},>0} = G_{\mathbf v'_+,\mathbf{w'},>0}\dot w_{J,0}.$$ By \cite[Proposition 5.2]{BH21a}, we have
    \begin{equation*}
        x_{i_1}(m)g'\dot w_{J,0}B^+/B^+ = h\dot w_{J,0}B^+/B^+,\text{ for some }h\in G_{\mathbf{v'}_+,\mathbf{w'},>0}.
    \end{equation*}
    Since $w'\in W^J$, we may consider the map 
    \[
    \begin{aligned}
    \pi^{w'w_{J,0}}_{w'}:\quad
    U^+\dot w'\dot w_{J,0}B^+/B^+
       &\longrightarrow U^+\dot w'B^+/B^+,\\
    u\dot w'\dot w_{J,0}B^+/B^+
       &\longmapsto u\dot w'B^+/B^+.
    \end{aligned}
    \]
    as in \cite[\S4.3]{MR04}. Since $x_{i_1}(m)\in B^+$, we have
    \begin{align*}
      x_{i_1}(m)g'B^+/B^+ &= \pi^{w'w_{J,0}}_{w'}(x_{i_1}(m)g'\dot w_{J,0}B^+/B^+) \nonumber \\
      &=\pi^{w'w_{J,0}}_{w'}(h\dot w_{J,0}B^+/B^+)\nonumber \\
      &= hB^+/B^+.\
    \end{align*}
    It follows that we have 
    $$h^{-1}x_{i_1}(m)g'\in B^+\cap \dot w_{J,0}B^+\dot w_{J,0}^{-1} = TU_{P_J^+}.$$
    Write $x_{i_1}(m)g' = htu$, where $t\in T$ and $u\in U_{P_J^+}$. We show that $t\in T_{>0}$.


    For this purpose, we may assume that $G$ is simply-laced, because for groups of other types the result can be obtained from folding. Let $\lambda$ be a regular dominant weight, $V_\lambda$ the corresponding irreducible highest weight module and $\eta_\lambda$ a highest weight vector. Then we have
    $$x_{i_1}(m)g'\eta_\lambda =\lambda(t) h\eta_\lambda.$$
    Since $x_{i_1}(m)g'B^+\in\mathcal B_{\geqslant0}$ and $hB^+\in\mathcal B_{\geqslant0}$, the vectors $x_{i_1}(m)g'\eta_\lambda$ and $h\eta_\lambda$ are both nonnegative linear combinations of canonical basis \cite[Proposition 8.17 (b)]{L94}, and we have $\lambda(t)>0$. Since $\lambda$ is arbitrary, we have $t\in T_{>0}$.

    In conclusion, we have
    $$\dot r^{-1}G_{\mathbf v_+,\mathbf w,>0}\subseteq U_{P_J^-}\left(\dot r^{\prime,-1}G_{\mathbf {v'}_+,\mathbf{w'},>0}\right)T_{>0}U_{P_J^+}.$$
    

    \item $s_{i_1}r<r$, $g_1\in y_{i_1}(\mathbb R_{>0})$, and $s_{i_1}v=vs_j$ for $j\in J$. In this case, $r' = s_{i_1}r\in W^J$, $v' = v$, $a = s_j*a'$, and $b' = b$. 

    Similar to (iv), we have

By \Cref{combinatorial_atlas} and the thinness of $\tilde W$, every interval of length $2$ in $\mathcal Q$ contains exactly four elements. It remains to check that for $q\in\hat{\mathcal Q}$ with $\operatorname{rank}(q)=2$, there are exactly two rank-$1$ elements below $q$. Three cases occur:
\begin{itemize}

\end{itemize}

Then $\xi_{(r_1,r_2)}\lessdot q'$, $\xi_{(r_1,r_2')}\lessdot q'$, and $\xi_{(r_1,r_2')}\triangleleft_{\min(\mathcal {Q})}\xi_{(r_1,r_2)}$. It remains to show that $q'\leq_{\mathcal {Q}}q$, which we obtain from
    $$v = r_1\leq xy_J^{-1},\enspace yy_J^{-1} = y^J\leq r_2'\leq r_2\leq w.$$
\end{itemize}
This completes the proof.

\label{CL-shellable}
The chain-edge labeling $\lambda$ on $\hat\mathcal {Q}$ is CL. As a consequence, $\hat\mathcal {Q}$ is shellable.

\begin{proof}
We first consider a rooted interval $[q_1,q_2]_r$ where $q_1\neq\hat 0$. By \Cref{combinatorial_atlas}, this is isomorphic to a Bruhat interval. By our construction, the labeling is CL from the result of Dyer \cite[\S3]{Dy92}.

Now, we consider an interval $[\hat 0,q]$ where $q = (J;x,y;v,w)$. Here, the root is trivial. By our definition of $\triangleleft_{\min(\mathcal {Q})}$, the lexicographically minimal chain passes through
$\xi_{(v,y^J)}$. Since the labeling is CL over the interval $[\xi_{(v,y^J)},q]$, the lexicographical minimal chain 
$$c = (\xi_{(v,y^J)}\lessdot_\mathcal {Q} q_1\lessdot_{\mathcal {Q}}q_2\lessdot_\mathcal {Q}\cdots\lessdot_\mathcal {Q} q)$$ 
is increasing, and the first label in this chain is $(\xi_{(v,y^J)},\varphi(q_1)\varphi(\xi_{(v,y^J)})^{-1})$, by our definition of $\prec_{(v,y^J)}$ and \Cref{diamond_labels}, we have
$$(\xi_{(v,y^J)},\eta_{(v,y^J)})\prec(\xi_{(v,y^J)},\varphi(q_1)\varphi(\xi_{(v,y^J)})^{-1}).$$
Therefore the maximal chain $(\hat 0,c)$ is lexicographically minimal and increasing. Moreover, there is only one increasing chain that passes through $\xi_{(v,y^J)}$.

Now we suppose that a maximal chain
$$(\hat 0\lessdot_{\hat\mathcal {Q}}\xi_{(r_1,r_2)}\lessdot_{\hat\mathcal {Q}} q_1'\lessdot_{\hat\mathcal {Q}}q_2'\lessdot_{\hat\mathcal {Q}}\cdots\lessdot_{\hat\mathcal {Q}} q)$$
is increasing but $(r_1,r_2)\neq (v,y^J)$. Since $\lambda$ defines a CL-labeling on $[\xi_{(r_1,r_2)},q]$, we know that 
$$c' =(\xi_{(r_1,r_2)}\lessdot_{\hat\mathcal {Q}} q_1'\lessdot_{\hat\mathcal {Q}}q_2'\lessdot_{\hat\mathcal {Q}}\cdots\lessdot_{\hat\mathcal {Q}} q)$$
is lexicographically minimal. However, by \Cref{decreasing_label}, there is $q''_1\in\mathcal {Q}$ and $\xi_{(r_1',r_2')}\in\min(\mathcal {Q})$ such that $\xi_{(r_1,r_2)}\lessdot_{\mathcal {Q}}q''_1\leq_{\mathcal {Q}} q$, $\xi_{(r_1',r_2')}\lessdot_\mathcal {Q} q''_1$ and $\xi_{(r_1',r_2')}\triangleleft_{\min(\mathcal {Q})}\xi_{(r_1,r_2)}$. By \Cref{diamond_labels}, we have $\varphi(q_1'')\varphi(\xi_{(r_1,r_2)})^{-1}\prec_{(r_1,r_2)}\eta_{(r_1,r_2)}$. Since $c'$ is lexicographically minimal, we have 
$$\varphi(q_1')\varphi(\xi_{(r_1,r_2)})^{-1}\preceq_{(r_1,r_2)}\varphi(q_1'')\varphi(\xi_{(r_1,r_2)})^{-1}\prec_{(r_1,r_2)}\eta_{(r_1,r_2)},$$
and thus, $(\xi_{(r_1,r_2)},\varphi(q_1')\varphi(\xi_{(r_1,r_2)})^{-1})\prec (\xi_{(r_1,r_2)},\eta_{(r_1,r_2)})$. This is a contradiction, and a maximal chain passing through $\xi_{(r_1,r_2)}$ with $(r_1,r_2)\neq(v,y^J)$ can never be increasing. This completes the proof.
\end{proof}

\label{shellable_regular}
Let $X$ be a regular CW complex with face poset $P$. If
$P\sqcup\{\hat0,\hat1\}$, obtained by adjoining a minimum and a
maximum, is graded, thin, and shellable, then $X$ is homeomorphic to

<!-- theorem-8: body_md -->
\label{thm:poincare}
Let $X$ be a compact $n$-dimensional topological manifold with
boundary such that
\begin{enumerate}
    \item $\partial X$ is homeomorphic to an $(n-1)$-dimensional sphere;
    \item $X\setminus\partial X$ is homeomorphic to an
    $n$-dimensional open ball.
\end{enumerate}
Then $X$ is homeomorphic to an $n$-dimensional closed ball $D^n$.

<!-- proposition-9: body_md -->
\label{boundary_regular}
Let $(X=\bigsqcup X_\gamma,X_{\geqslant0})$ be a remarkable polyhedral
space, and write $X_{\gamma,>0}=X_\gamma\cap X_{\geqslant0}$.
Let $P$ be the closure poset of its \emph{nonempty} nonnegative
cells, and put $\hat P=P\sqcup\{\hat0\}$. Suppose that
\begin{enumerate}
    \item for every $\gamma\in P$, the interval $[\hat0,\gamma]$
    is finite, thin, and shellable;
    \item for every $\gamma\in P$, the Hausdorff closure
    $\overline{X_{\gamma,>0}}$ is a compact topological manifold
    with boundary
    \[
    \partial\overline{X_{\gamma,>0}}
      =\bigsqcup_{\gamma'\in P,\,\gamma'<\gamma}X_{\gamma',>0}.
    \]
\end{enumerate}
Then $X_{\geqslant0}$ has the regularity property.

We induct on $n=\dim X_{\gamma,>0}$. If $n=0$, the cell and its closure are a single point.

Now we consider general $\gamma$. By induction hypothesis, $\partial \overline{X_{\gamma,>0}}$ is a regular CW complex, with face poset $[\hat 0,\gamma]\setminus\{\hat 0,\gamma\}$. It then follows from condition (1) and \Cref{shellable_regular} that $\partial\overline{X_{\gamma,>0}}$ is homeomorphic to a sphere of codimension $1$ in $\overline{X_{\gamma,>0}}$. Finally, it follows from \Cref{thm:poincare} that $\overline{X_{\gamma,>0}}$ is homeomorphic to a closed ball with interior $X_{\gamma,>0}$. This completes the proof.
