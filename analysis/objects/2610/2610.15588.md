<!-- theorem-1: body_md -->
\label{thm:global}
Assume \eqref{eq:viscosity} and $\gamma\geq1$. Let
\begin{equation}\label{eq:initial-regularity}
q_0\in H^3(\Omega),\qquad
u_0\in H^3(\Omega)\cap H^1_0(\Omega).
\end{equation}
Let $\phi_0\in\dot H^1(\Omega)$ solve
\begin{equation}\label{eq:initial-poisson}
\Delta\phi_0=q_0\quad\text{in }\Omega,\qquad
\partial_3\phi_0=0\quad\text{on }\partial\Omega.
\end{equation}
Suppose that the initial data satisfy the compatibility condition
\begin{equation}\label{eq:compatibility}
\left.
\left[
\frac{\mu\Delta u_0+(\mu+\lambda)\nabla\operatorname{div} u_0}{\rho_0}
-u_0\cdot\nabla u_0
-\gamma\rho_0^{\gamma-2}\nabla q_0+\nabla\phi_0
\right]\right|_{\partial\Omega}=0.
\end{equation}
There exist constants $\varepsilon_*>0$ and $C>0$ such that, if
\begin{equation}\label{eq:smallness}
\varepsilon_0
:=\|q_0\|_{H^3}+\|u_0\|_{H^3}+\|\nabla\phi_0\|
\leq\varepsilon_*,
\end{equation}
then the problem \eqref{eq:NSP}, \eqref{eq:data}, with
$\phi(t)\in\dot H^1(\Omega)$, admits a unique global strong solution $(\rho,u,\phi)$. For every $T>0$, this solution satisfies
\begin{equation}\label{eq:regularity}
\begin{aligned}
&q\in C([0,T];H^3),\qquad
  u\in C([0,T];H^3\cap H^1_0)\cap L^2(0,T;H^4),\\
&\nabla\phi\in C([0,T];H^3),\qquad
  q_t\in C([0,T];H^2),\\
&u_t\in C([0,T];H^1_0)\cap L^2(0,T;H^2),\qquad
  u_{tt}\in L^2(0,T;L^2),
\end{aligned}
\end{equation}
and
\begin{equation}\label{eq:density-bounds}
\frac12\leq\rho(t,x)\leq\frac32,
\qquad (t,x)\in[0,\infty)\times\Omega.
\end{equation}
Moreover,
\begin{equation}\label{eq:global-estimate}
\begin{aligned}
&\sup_{t\geq0}\bigl(
\|q(t)\|_{H^3}^2+\|u(t)\|_{H^3}^2
+\|\nabla\phi(t)\|_{H^3}^2
+\|q_t(t)\|_{H^2}^2+\|u_t(t)\|_{H^1}^2\bigr)\\
&\quad+\int_0^\infty\bigl(
\|q\|_{H^3}^2+\|\nabla u\|_{H^3}^2
+\|\nabla u_t\|_{H^1}^2+\|q_t\|_{H^2}^2\bigr)\,\mathrm{d} t
\leq C\varepsilon_0^2.
\end{aligned}
\end{equation}

<!-- theorem-2: body_md -->
\label{thm:decay}
Under the assumptions of Theorem~\ref{thm:global}, suppose in addition
that, for some $\frac12\leq s<1$,
\begin{equation}\label{eq:negative-data}
M_{s,0}:=\|\Lambda_y^{-s}(q_0,u_0,\nabla\phi_0)\|<\infty.
\end{equation}
Then the global solution satisfies
\begin{equation}\label{eq:solution-decay}
\begin{aligned}
&\|q(t)\|_{H^3}+\|u(t)\|_{H^3}
+\|\nabla\phi(t)\|_{H^3}\\
&\qquad+\|q_t(t)\|_{H^2}+\|u_t(t)\|_{H^1}
\leq C_s(1+t)^{-s/2},\qquad t\geq0,
\end{aligned}
\end{equation}
and
\begin{equation}\label{eq:dissipation-tail}
\int_t^\infty
\bigl(\|q\|_{H^3}^2+\|\nabla u\|_{H^3}^2
+\|\nabla u_t\|_{H^1}^2+\|q_t\|_{H^2}^2\bigr)(\tau)\,\mathrm{d}\tau
\leq C_s(1+t)^{-s}.
\end{equation}
The constant $C_s$ may depend on $\mu,\lambda,\gamma,s,\varepsilon_0$
and $M_{s,0}$, but is independent of $t$. No smallness of $M_{s,0}$
is required.

<!-- remark-1: body_md -->
The smallness condition in Theorem~\ref{thm:global} includes the $L^2$
norm of the initial electric field. This low-order norm is retained in
the energy, while higher derivatives of the potential are controlled
by the density through the Poisson equation. The compatibility condition
\eqref{eq:compatibility} means that $u_t(0)$ has zero trace.
The additional assumption \eqref{eq:negative-data} controls tangential
low frequencies. By the two-dimensional Hardy--Littlewood--Sobolev
inequality, a sufficient condition for \eqref{eq:negative-data} is
\[
q_0,\ u_0,\ \nabla\phi_0
\in L^2\bigl(\mathbb R_+;L^{2/(1+s)}(\mathbb R^2)\bigr).
\]

<!-- lemma-1: body_md -->
\label{lem:poisson}
If $\phi\in\dot H^1(\Omega)$ solves
$\Delta\phi=q$ and $\partial_3\phi=0$ on $\partial\Omega$, then
\begin{equation}\label{eq:poisson-estimate}
\|\nabla^{m+2}\phi\|\leq C\|\nabla^m q\|,
\qquad m=0,1,2.
\end{equation}
For a time-dependent solution, we also have
\begin{equation}\label{eq:poisson-time}
\|\nabla\phi_t\|_{L^6}\leq C\|q_t\|.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Extend $\phi$ evenly across $x_3=0$. Since
$\partial_3\phi=0$ there, the reflected equation has no boundary
source. The $L^2$ boundedness of the Fourier multipliers
$\xi_i\xi_j/|\xi|^2$ yields
$\|\nabla^2\phi\|\leq C\|q\|$.
Tangential derivatives satisfy the same Neumann condition, so
the same argument applies to $\partial_y^\beta\phi$.
The remaining normal derivatives follow from
\[
\partial_3^2\phi=q-\Delta_y\phi.
\]
This proves \eqref{eq:poisson-estimate} without any boundary
condition on $q$. Applying the case $m=0$ to $\phi_t$ and using
the homogeneous Sobolev inequality gives
\[
\|\nabla\phi_t\|_{L^6}
\leq C\|\nabla^2\phi_t\|
\leq C\|q_t\|.
\]

<!-- lemma-2: body_md -->
\label{lem:elliptic}
For $z=0$ on $\partial\Omega$, the Lam\'e operator satisfies
\begin{equation}\label{eq:lame-estimate}
\|\nabla^2z\|\leq C\|Az\|,
\qquad
\|\nabla^3z\|\leq C\|\nabla Az\|.
\end{equation}
If $(z,\pi)$ solves
\begin{equation}\label{eq:stokes-problem}
-\mu\Delta z+\nabla\pi=g,\qquad
\operatorname{div} z=h,\qquad z|_{\partial\Omega}=0,
\end{equation}
then
\begin{equation}\label{eq:stokes-estimate}
\|\nabla^{m+2}z\|+\|\nabla^{m+1}\pi\|
\leq C\bigl(\|\nabla^m g\|+\|\nabla^{m+1}h\|\bigr),
\qquad m=0,1,2.
\end{equation}
Here the estimates are understood in the corresponding homogeneous
Sobolev spaces, and $\pi$ is determined up to a constant.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Since the tangential derivatives of $z$ vanish on the boundary,
\[
(Az,-\Delta_yz)
=\sum_{i=1}^2a(\partial_i z,\partial_i z).
\]
It follows that $\|\nabla\nabla_y z\|\leq C\|Az\|$.
The component equations for $Az$ then give the estimate for
$\partial_3^2z$, since their normal coefficients are $\mu$ and
$2\mu+\lambda$. Applying this argument to the tangential
derivatives of $z$, and differentiating the component equations
once in $x_3$, proves the second estimate in
\eqref{eq:lame-estimate}.

The estimate \eqref{eq:stokes-estimate} is the homogeneous
Dirichlet estimate for the Stokes system in the half-space; see
\cite[Chapter IV, Section 3]{Galdi2011}. It follows by taking the
Fourier transform in the tangential variables and estimating the
decaying solutions of the resulting ordinary differential system.
Tangential differentiation preserves the zero boundary value.
The remaining normal derivatives are recovered from the component
equations and the divergence equation. This gives the estimates for
$m=0,1,2$ in the homogeneous norms stated above.

<!-- lemma-3: body_md -->
\label{lem:tangential-negative}
Let $\frac12\leq s<1$. For $f\in H^1(\Omega)$ and $g\in L^2(\Omega)$,
\begin{equation}\label{eq:negative-product}
\|\Lambda_y^{-s}(fg)\|
\leq C_s\|f\|^{s-\frac12}
\|\nabla f\|^{\frac32-s}\|g\|.
\end{equation}
Moreover, whenever the norms on the right-hand side are finite,
\begin{equation}\label{eq:tangential-interpolation}
\|f\|\leq
\|\Lambda_y^{-s}f\|^{1/(1+s)}
\|\nabla_y f\|^{s/(1+s)}.
\end{equation}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
For smooth $f$ of compact support in $\overline\Omega$, the tangential
Fourier transform gives, for every $x_3\geq0$,
\[
\begin{aligned}
|\xi'|\,|\widehat f(\xi',x_3)|^2
&=-\int_{x_3}^\infty|\xi'|\,
\partial_z|\widehat f(\xi',z)|^2\,\mathrm{d} z\\
&\leq\int_{x_3}^\infty
\bigl(|\xi'|^2|\widehat f(\xi',z)|^2
+|\partial_z\widehat f(\xi',z)|^2\bigr)\,\mathrm{d} z.
\end{aligned}
\]
After integration in $\xi'$, the two-dimensional homogeneous Sobolev
inequality yields
\[
\sup_{x_3>0}\|f(\cdot,x_3)\|_{L_y^4}
\leq C\|\nabla f\|.
\]
Similarly,
\[
\sup_{x_3>0}\|f(\cdot,x_3)\|_{L_y^2}
\leq C\|f\|^{1/2}\|\partial_3f\|^{1/2}.
\]
Interpolation between these two bounds gives
\[
\|f\|_{L_{x_3}^\infty L_y^{2/s}}
\leq C_s\|f\|^{s-\frac12}\|\nabla f\|^{\frac32-s}.
\]
Applying the two-dimensional Hardy--Littlewood--Sobolev inequality on
each horizontal slice, followed by H\"older's inequality, we obtain
\[
\|\Lambda_y^{-s}(fg)\|
\leq C_s\|fg\|_{L_{x_3}^2L_y^{2/(1+s)}}
\leq C_s\|f\|_{L_{x_3}^\infty L_y^{2/s}}\|g\|.
\]
This proves \eqref{eq:negative-product}; approximation gives the result
for the stated classes of functions. Finally,
\eqref{eq:tangential-interpolation} follows from H\"older's inequality
in the tangential Fourier variables and $x_3$.

<!-- lemma-4: body_md -->
\label{lem:local}
Let $q_0\in H^3$, $u_0\in H^3\cap H^1_0$, and let
$\phi_0\in\dot H^1(\Omega)$ satisfy \eqref{eq:initial-poisson}.
Assume \eqref{eq:viscosity}, \eqref{eq:compatibility},
$\gamma\geq1$, and $\inf_\Omega(1+q_0)>0$.
Then there exists $T_0>0$ such that the problem
\eqref{eq:NSP}, \eqref{eq:data}, with
$\phi(t)\in\dot H^1(\Omega)$, has a unique strong solution
with the regularity \eqref{eq:regularity} on $[0,T_0]$.
If its maximal existence time $T^*$ is finite, then either
\[
\liminf_{t\uparrow T^*}\inf_{x\in\Omega}(1+q(t,x))=0
\]
or
\[
\sup_{0\leq t<T^*}
\bigl(\|q(t)\|_{H^3}+\|u(t)\|_{H^3}
+\|\nabla\phi(t)\|\bigr)=\infty.
\]
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
We use the local iteration for positive-density compressible
viscous flow with the homogeneous Dirichlet condition, as in
\cite{MatsumuraNishida1983}. We indicate how the Poisson equation
is included in this construction. Along the continuity equation,
the electric field is determined by
\[
\nabla\phi(t)=\nabla\phi_0
-\int_0^t\mathbb Q\bigl((1+q)u\bigr)(s)\,\mathrm{d} s.
\]
This identity preserves the Poisson equation and the homogeneous
Neumann condition in their weak form. Moreover,
\[
\|\nabla\phi(t)\|
\leq\|\nabla\phi_0\|
+\int_0^t\|(1+q)u\|(s)\,\mathrm{d} s,
\qquad
\|\nabla\phi(t)\|_{H^3}
\leq C\bigl(\|\nabla\phi(t)\|+\|q(t)\|_{H^2}\bigr).
\]
The second estimate follows from Lemma~\ref{lem:poisson}.
For two such triples $(q^{(i)},u^{(i)},\phi^{(i)})$, $i=1,2$,
with the same initial data and a common $H^3$ bound for $q^{(i)}$
and $u^{(i)}$, the $L^2$ boundedness of $\mathbb Q$ gives
\[
\begin{aligned}
\|\nabla(\phi^{(1)}-\phi^{(2)})(t)\|
&\leq\int_0^t
\|(1+q^{(1)})u^{(1)}-(1+q^{(2)})u^{(2)}\|(s)\,\mathrm{d} s\\
&\leq C\int_0^t
\bigl(\|u^{(1)}-u^{(2)}\|+\|q^{(1)}-q^{(2)}\|\bigr)(s)\,\mathrm{d} s.
\end{aligned}
\]
Here $C$ depends on the common $H^3$ bound. Applying
Lemma~\ref{lem:poisson} to the difference of the Poisson equations
also yields
\[
\|\nabla(\phi^{(1)}-\phi^{(2)})\|_{H^3}
\leq C\bigl(
\|\nabla(\phi^{(1)}-\phi^{(2)})\|
+\|q^{(1)}-q^{(2)}\|_{H^2}\bigr).
\]
These estimates give the required local Lipschitz bound for the electric
force. Thus the transport estimates for $q$ and the Dirichlet parabolic
estimates for $u$
give a uniform bound and convergence on a sufficiently short
time interval. The compatibility condition gives
$u_t(0)\in H^1_0$ and hence the stated time regularity.
The existence time depends only on a positive lower bound for
$1+q_0$ and an upper bound for
$\|q_0\|_{H^3}+\|u_0\|_{H^3}+\|\nabla\phi_0\|$.
Restarting the same construction proves the continuation
criterion.

<!-- lemma-5: body_md -->
\label{lem:basic}
Let $G(1)=G'(1)=0$ and $G''(\rho)=p'(\rho)/\rho$. Define
\begin{equation}\label{eq:E0}
\mathcal E_0(t)=\int_\Omega
\left(\frac12\rho|u|^2+G(\rho)+\frac12|\nabla\phi|^2\right)\,\mathrm{d} x.
\end{equation}
Then
\begin{equation}\label{eq:basic-energy}
\frac{\,\mathrm{d}}{\,\mathrm{d} t}\mathcal E_0+a(u,u)=0,
\end{equation}
and $\mathcal E_0$ is equivalent to
$\|q\|^2+\|u\|^2+\|\nabla\phi\|^2$ under
\eqref{eq:apriori-assumption}.
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
The definition of $G$ implies
\[
\rho G'(\rho)-G(\rho)=p(\rho)-p(1).
\]
The continuity equation therefore yields
\[
\partial_tG(\rho)+\operatorname{div}(G(\rho)u)
+(p(\rho)-p(1))\operatorname{div} u=0.
\]
Taking the inner product of the momentum equation in
\eqref{eq:NSP} with $u$ and adding the integral of this identity,
we obtain
\[
\frac{\,\mathrm{d}}{\,\mathrm{d} t}\int_\Omega
\left(\frac12\rho|u|^2+G(\rho)\right)\,\mathrm{d} x
+a(u,u)=(\rho u,\nabla\phi).
\]
Here the continuity equation gives the kinetic energy derivative,
and the pressure terms cancel. The boundary fluxes vanish since $u=0$. By \eqref{eq:field-time} and the
self-adjointness of $\mathbb Q$,
\[
\frac12\frac{\,\mathrm{d}}{\,\mathrm{d} t}\|\nabla\phi\|^2
=-(\rho u,\nabla\phi).
\]
Adding these equalities proves \eqref{eq:basic-energy}.
Finally, Taylor's formula gives
\[
G(1+q)=\gamma q^2\int_0^1(1-s)(1+sq)^{\gamma-2}\,\mathrm{d} s,
\]
which is bounded above and below by positive multiples of $q^2$
for $|q|\leq1/2$. This proves the asserted equivalence.

<!-- lemma-6: body_md -->
\label{lem:time}
Set
\begin{equation}\label{eq:E1}
\mathcal E_1(t)=\frac12
\bigl(\gamma\|q_t\|^2+\|u_t\|^2+\|\nabla\phi_t\|^2\bigr).
\end{equation}
Then
\begin{equation}\label{eq:time-energy}
\frac{\,\mathrm{d}}{\,\mathrm{d} t}\mathcal E_1+c\|\nabla u_t\|^2\leq CN(t)D(t).
\end{equation}
<!-- end lemma-6 -->
<!-- proof-6: body_md -->
Differentiating the first two equations of \eqref{eq:perturbation}
in time and testing them with $\gamma q_t$ and $u_t$, respectively,
gives
\begin{equation}\label{eq:time-identity}
\frac{\,\mathrm{d}}{\,\mathrm{d} t}\mathcal E_1+a(u_t,u_t)
=(F_t,u_t)+\gamma((qu)_t,\nabla q_t)
-(\nabla\phi_t,(qu)_t).
\end{equation}
Here $u_t=0$ and $(qu)_t=0$ on the boundary. Differentiating
\eqref{eq:field-time} and using the orthogonality of $\mathbb Q$, we have
\[
\frac12\frac{\,\mathrm{d}}{\,\mathrm{d} t}\|\nabla\phi_t\|^2
=-(u_t,\nabla\phi_t)-((qu)_t,\nabla\phi_t),
\]
which supplies the electric part of \eqref{eq:time-identity}.

We estimate $(F_t,u_t)$ using \eqref{eq:Ft}. For its principal
viscous term, integration by parts yields
\begin{equation}\label{eq:time-viscous}
\begin{aligned}
\left(\frac{q}{1+q}Au_t,u_t\right)
={}&\int_\Omega\frac{q}{1+q}
\left(\mu|\nabla u_t|^2+(\mu+\lambda)|\operatorname{div} u_t|^2\right)\,\mathrm{d} x\\
&+\mu\sum_{i,j=1}^3\int_\Omega
\partial_j\left(\frac{q}{1+q}\right)
\partial_j u_{t,i}\,u_{t,i}\,\mathrm{d} x\\
&+(\mu+\lambda)\int_\Omega
\operatorname{div} u_t\,\nabla\left(\frac{q}{1+q}\right)\cdot u_t\,\mathrm{d} x.
\end{aligned}
\end{equation}
The right-hand side is bounded in absolute value by
\[
C\|q\|_{L^\infty}\|\nabla u_t\|^2
+C\|\nabla q\|_{L^3}\|\nabla u_t\|\|u_t\|_{L^6}
\leq CN(t)\|\nabla u_t\|^2.
\]
We used $\|u_t\|_{L^6}\leq C\|\nabla u_t\|$ in the last step.
The other viscous term satisfies
\[
\left|\left(\frac{q_t}{(1+q)^2}Au,u_t\right)\right|
\leq C\|q_t\|_{L^3}\|Au\|\|u_t\|_{L^6}
\leq CN(t)D(t).
\]
Integration by parts in the pressure term and H\"older's
inequality give
\begin{align*}
\left|\left(\nabla[((1+q)^{\gamma-2}-1)q_t],u_t\right)\right|
&\leq C\|q\|_{L^\infty}\|q_t\|\|\nabla u_t\|
\leq CN(t)D(t),\\
|(u_t\cdot\nabla u,u_t)|
&\leq\|u_t\|_{L^3}\|\nabla u\|\|u_t\|_{L^6}
\leq CN(t)D(t),\\
|(u\cdot\nabla u_t,u_t)|
&=\frac12|(\operatorname{div} u,|u_t|^2)|\\
&\leq C\|\nabla u\|\|u_t\|_{L^3}\|u_t\|_{L^6}
\leq CN(t)D(t).
\end{align*}
Consequently, $|(F_t,u_t)|\leq CN(t)D(t)$.
For the continuity term, we write $(qu)_t=q_tu+qu_t$ and obtain
\begin{align*}
|(q_tu,\nabla q_t)|
&=\frac12|(\operatorname{div} u,q_t^2)|
\leq C\|\nabla u\|_{L^\infty}\|q_t\|^2
\leq CN(t)D(t),\\
|(qu_t,\nabla q_t)|
&\leq\|q\|_{L^3}\|u_t\|_{L^6}\|\nabla q_t\|
\leq CN(t)D(t).
\end{align*}
Finally, Lemma~\ref{lem:poisson} gives
\begin{align*}
|(\nabla\phi_t,(qu)_t)|
&\leq\|\nabla\phi_t\|_{L^6}
\bigl(\|q_t\|\|u\|_{L^3}+\|q\|\|u_t\|_{L^3}\bigr)\\
&\leq CN(t)\bigl(\|q_t\|^2+\|q\|^2\bigr)
\leq CN(t)D(t).
\end{align*}
Substituting these bounds into \eqref{eq:time-identity} proves
\eqref{eq:time-energy}.

\label{lem:density}
The density satisfies
\begin{equation}\label{eq:density-energy}
\frac{\,\mathrm{d}}{\,\mathrm{d} t}(u,\nabla q)+c\|q\|_{H^1}^2
\leq C\|\nabla u\|_{H^1}^2+CN(t)D(t).
\end{equation}

<!-- proof-7: body_md -->
Taking the inner product of the momentum equation in
\eqref{eq:perturbation} with $\nabla q$ and using the continuity
equation, we find
\begin{equation}\label{eq:density-identity}
\begin{aligned}
\frac{\,\mathrm{d}}{\,\mathrm{d} t}(u,\nabla q)
+\gamma\|\nabla q\|^2+\|q\|^2
={}&\|\operatorname{div} u\|^2-(Au,\nabla q)+(F,\nabla q)\\
&+(\operatorname{div}(qu),\operatorname{div} u).
\end{aligned}
\end{equation}
In this calculation,
\[
(u,\nabla q_t)=-(\operatorname{div} u,q_t),\qquad
(\nabla\phi,\nabla q)=-\|q\|^2,
\]
because $u\cdot n=0$ and $\partial_n\phi=0$. Moreover,

|(Au,\nabla q)|
&\leq\frac\gamma2\|\nabla q\|^2+C\|\nabla^2u\|^2,\\

|(\operatorname{div}(qu),\operatorname{div} u)|
&\leq C\bigl(\|u\|_{L^\infty}\|\nabla q\|
+\|q\|_{L^\infty}\|\nabla u\|\bigr)\|\nabla u\|
\leq CN(t)D(t).
\end{align*}
Substitution into \eqref{eq:density-identity} leaves positive
multiples of $\|q\|^2$ and $\|\nabla q\|^2$ on the left-hand
side and proves \eqref{eq:density-energy}.

<!-- lemma-8: body_md -->
\label{lem:gradient-time}
Define
\begin{equation}\label{eq:E2}
\mathcal E_2(t)
=\frac12a(u_t,u_t)+(\mathbb Q u,Au)-\frac12a(u,u).
\end{equation}
Then
\begin{equation}\label{eq:gradient-time-energy}
\frac{\,\mathrm{d}}{\,\mathrm{d} t}\mathcal E_2+c\|\nabla^2u_t\|^2

\end{equation}

<!-- proof-8: body_md -->
Direct integration by parts gives
\begin{equation}\label{eq:electric-boundary}
\begin{aligned}
(\nabla\phi_t,Au_t)
={}&\mu(\nabla^2\phi_t,\nabla u_t)
+(\mu+\lambda)(q_t,\operatorname{div} u_t)\\
&-\mu\int_{\partial\Omega}
\nabla_y\phi_t\cdot\partial_n u_{t,y}\,\mathrm{d} y.
\end{aligned}
\end{equation}
The last integral need not vanish, since the no-slip condition does not
force $\partial_n u_{t,y}$ to vanish on the boundary.
To retain its contribution, we apply $\mathbb P$ to the momentum equation and use
\eqref{eq:field-time}:
\[
\mathbb P u_t=-\mathbb P Au+\mathbb P F,\qquad
\mathbb Q u_t=u_t+\mathbb P Au-\mathbb P F,\qquad
\nabla\phi_t=-\mathbb Q u-\mathbb Q(qu).
\]
It follows that
\begin{equation}\label{eq:projection-energy-identity}
\begin{aligned}
(\nabla\phi_t,Au_t)
={}&-\frac{\,\mathrm{d}}{\,\mathrm{d} t}(\mathbb Q u,Au)
+(\mathbb Q u_t,Au)-(\mathbb Q(qu),Au_t)\\
={}&\frac{\,\mathrm{d}}{\,\mathrm{d} t}
\left[-(\mathbb Q u,Au)+\frac12a(u,u)\right]
+\|\mathbb P Au\|^2\\
&-(\mathbb P F,Au)-(\mathbb Q(qu),Au_t).
\end{aligned}
\end{equation}
Here $(u_t,Au)=a(u,u_t)$ since $u_t=0$ on the boundary.

Testing the time derivative of the momentum equation with $Au_t$
and substituting \eqref{eq:projection-energy-identity}, we obtain
\begin{equation}\label{eq:gradient-time-identity}
\begin{aligned}
\frac{\,\mathrm{d}}{\,\mathrm{d} t}\mathcal E_2+\|Au_t\|^2
={}&\|\mathbb P Au\|^2-(\mathbb P F,Au)-(\mathbb Q(qu),Au_t)\\
&-\gamma(\nabla q_t,Au_t)+(F_t,Au_t).
\end{aligned}
\end{equation}
Since
\[
\|\mathbb Q(qu)\|\leq\|u\|_{L^\infty}\|q\|,
\]
the $L^2$ boundedness of the projections and
\eqref{eq:products}--\eqref{eq:Ft-bound} give
\begin{align*}
\|\mathbb P Au\|^2&\leq C\|\nabla^2u\|^2,\\
|(\mathbb P F,Au)|&\leq\|F\|\|Au\|\leq CN(t)D(t),\\
|(\mathbb Q(qu),Au_t)|+|(F_t,Au_t)|
&\leq\epsilon\|Au_t\|^2+C_\epsilon N(t)^2D(t).
\end{align*}
For the pressure term, \eqref{eq:qt-estimates} implies
\[
\gamma|(\nabla q_t,Au_t)|
\leq\epsilon\|Au_t\|^2
+C_\epsilon\bigl(\|\nabla u\|_{H^1}^2
+N(t)^2\|q\|_{H^2}^2\bigr).
\]
Choosing $\epsilon$ sufficiently small absorbs the two terms
containing $\|Au_t\|^2$. Since $N(t)\leq\delta\leq1$ and
$\|\nabla^2u_t\|\leq C\|Au_t\|$ by
Lemma~\ref{lem:elliptic}, these bounds prove
\eqref{eq:gradient-time-energy}.

<!-- lemma-9: body_md -->
\label{lem:tangential}
Under \eqref{eq:apriori-assumption},
\begin{equation}\label{eq:tangential-energy}
\begin{aligned}
&\frac12\frac{\,\mathrm{d}}{\,\mathrm{d} t}
\sum_{1\leq|\beta|\leq3}
\left(\gamma\|\partial_y^\beta q\|^2
+\|\partial_y^\beta u\|^2
+\|\nabla\partial_y^\beta\phi\|^2\right)\\
&\qquad+c\sum_{1\leq|\beta|\leq3}
\|\nabla\partial_y^\beta u\|^2\leq CN(t)D(t).
\end{aligned}
\end{equation}
<!-- end lemma-9 -->
<!-- proof-9: body_md -->
Tangential derivatives preserve the boundary conditions
$\partial_y^\beta u=0$ and
$\partial_n\partial_y^\beta\phi=0$. Applying
$\partial_y^\beta$ to \eqref{eq:perturbation}, taking the
corresponding inner products, and using \eqref{eq:field-time},
we obtain
\begin{equation}\label{eq:tangential-identity}
\begin{aligned}
&\frac12\frac{\,\mathrm{d}}{\,\mathrm{d} t}
\left(\gamma\|\partial_y^\beta q\|^2
+\|\partial_y^\beta u\|^2
+\|\nabla\partial_y^\beta\phi\|^2\right)
+a(\partial_y^\beta u,\partial_y^\beta u)\\
&\quad=(\partial_y^\beta F,\partial_y^\beta u)
-\gamma(\partial_y^\beta\operatorname{div}(qu),\partial_y^\beta q)
-(\nabla\partial_y^\beta\phi,\partial_y^\beta(qu)).
\end{aligned}
\end{equation}
For the continuity term, write
\begin{equation}\label{eq:tangential-transport}
\begin{aligned}
\partial_y^\beta\operatorname{div}(qu)
={}&u\cdot\nabla\partial_y^\beta q
+q\operatorname{div}\partial_y^\beta u\\
&+[\partial_y^\beta,u\cdot\nabla]q
+[\partial_y^\beta,q]\operatorname{div} u.
\end{aligned}
\end{equation}
The zero boundary flux and \eqref{eq:sobolev-commutator} give
\begin{align*}
(u\cdot\nabla\partial_y^\beta q,\partial_y^\beta q)
&=-\frac12(\operatorname{div} u,|\partial_y^\beta q|^2),\\
\|[\partial_y^\beta,u\cdot\nabla]q\|
+\|[\partial_y^\beta,q]\operatorname{div} u\|
&\leq C\bigl(\|\nabla u\|_{L^\infty}\|q\|_{H^3}
+\|\nabla q\|_{L^\infty}\|\nabla u\|_{H^2}\bigr)
\leq CN(t)D(t)^{1/2}.
\end{align*}
Together with
\[
|(q\operatorname{div}\partial_y^\beta u,\partial_y^\beta q)|
\leq\|q\|_{L^\infty}\|\operatorname{div}\partial_y^\beta u\|
\|\partial_y^\beta q\|\leq CN(t)D(t),
\]
these estimates bound the continuity term in
\eqref{eq:tangential-identity} by $CN(t)D(t)$.
The convective term in $F$ is treated by the same integration:
\begin{align*}
|(\partial_y^\beta(u\cdot\nabla u),\partial_y^\beta u)|
&\leq\frac12\|\operatorname{div} u\|_{L^\infty}\|\partial_y^\beta u\|^2
+\|[\partial_y^\beta,u\cdot\nabla]u\|
\|\partial_y^\beta u\|\\
&\leq CN(t)\|\nabla u\|_{H^2}^2\leq CN(t)D(t).
\end{align*}

For the nonlinear pressure, integration by parts gives
\begin{equation}\label{eq:pressure-primitive}
\begin{aligned}
&\left(-\partial_y^\beta
\left[\gamma\bigl((1+q)^{\gamma-2}-1\bigr)\nabla q\right],
\partial_y^\beta u\right)\\
&\qquad=
\left(\partial_y^\beta
\int_0^q\gamma\bigl((1+s)^{\gamma-2}-1\bigr)\,\mathrm{d} s,
\operatorname{div}\partial_y^\beta u\right).
\end{aligned}
\end{equation}
The primitive vanishes to second order at $q=0$. Hence the
composition and product estimates imply
\[
\left\|\partial_y^\beta
\int_0^q\gamma\bigl((1+s)^{\gamma-2}-1\bigr)\,\mathrm{d} s\right\|
\leq C\|q\|_{L^\infty}\|q\|_{H^3}
\leq CN(t)\|q\|_{H^3}.
\]
Multiplication by $\|\operatorname{div}\partial_y^\beta u\|$ bounds
\eqref{eq:pressure-primitive} by $CN(t)D(t)$.

For the viscous term in $F$, expand
\begin{equation}\label{eq:tangential-viscous}

+\sum_{0<\zeta\leq\beta}\binom\beta\zeta

\end{equation}
The first term is integrated once, with zero boundary contribution,

\left|\left(\frac{q}{1+q}A\partial_y^\beta u,
\partial_y^\beta u\right)\right|
&\leq C\left\|\frac{q}{1+q}\right\|_{L^\infty}
\|\nabla\partial_y^\beta u\|^2\\
&\quad+C\left\|\nabla\left(\frac{q}{1+q}\right)\right\|_{L^3}
\|\nabla\partial_y^\beta u\|\|\partial_y^\beta u\|_{L^6}
\leq CN(t)\|\nabla\partial_y^\beta u\|^2.
\end{align*}

&\quad\leq

\|\nabla^{m+1}u\|\|\partial_y^\beta u\|,& |\zeta|=1,\\[1mm]

\|\nabla^m u\|_{L^6}\|\partial_y^\beta u\|,& |\zeta|=2,
\end{cases}
\end{align*}
and both bounds are at most $CN(t)D(t)$ by \eqref{eq:sobolev}.

\begin{align*}
&\left|\left(
\partial_y^\beta\left(\frac{q}{1+q}\right)Au,
\partial_y^\beta u\right)\right|\\
&\qquad\leq
\left\|\partial_y^\beta\left(\frac{q}{1+q}\right)\right\|
\|Au\|_{L^3}\|\partial_y^\beta u\|_{L^6}
\leq CN(t)D(t).
\end{align*}
Here the coefficient norms are bounded by $CN(t)$, while
$\|Au\|_{L^3}\leq C\|\nabla u\|_{H^2}$ and
$\|\partial_y^\beta u\|_{L^6}
\leq C\|\nabla\partial_y^\beta u\|$.

\[

\leq C\|q\|_{H^{|\beta|-1}},\qquad 1\leq|\beta|\leq3.
\]
Using the product estimate for derivatives of positive order,
\[

\leq C\bigl(\|u\|_{L^\infty}\|q\|_{H^3}
+\|q\|_{L^\infty}\|\nabla u\|_{H^2}\bigr)
\leq CN(t)D(t)^{1/2},
\]
we also have
$|(\nabla\partial_y^\beta\phi,\partial_y^\beta(qu))|\leq CN(t)D(t)$.
Substituting these bounds into \eqref{eq:tangential-identity}
and summing over $\beta$ proves \eqref{eq:tangential-energy}.

Here $I$ denotes the identity matrix,
$(\nabla v)_{ij}=\partial_jv_i$, and
$(\operatorname{div} S)_i=\sum_j\partial_jS_{ij}$.
The last line of \eqref{eq:decay-stress} accounts for
$Am-Au=A(qu)$. The boundary condition is $m=0$ on $\partial\Omega$.

Lemma~\ref{lem:poisson} and the smallness of $N(t)$ give
\[
\|m\|+\|u\|+\|\nabla\phi\|\leq CN(t),\qquad
\|\nabla m\|+\|\nabla u\|+\|\nabla^2\phi\|
\leq CD(t)^{1/2}.
\]
For the momentum, this follows from
$\nabla m=(1+q)\nabla u+u\otimes\nabla q$.
By Lemma~\ref{lem:tangential-negative},
\begin{equation}\label{eq:negative-stress}
\|\Lambda_y^{-s}S\|
\leq C_sN(t)^{s+\frac12}D(t)^{(\frac32-s)/2},
\qquad \frac12\leq s<1.
\end{equation}
Indeed, the convective and electric terms satisfy
\[
\begin{aligned}
\|\Lambda_y^{-s}(m\otimes u)\|
&\leq C_s\|u\|^{s-\frac12}
\|\nabla u\|^{\frac32-s}\|m\|,\\
\|\Lambda_y^{-s}(\nabla\phi\otimes\nabla\phi)\|
&\leq C_s\|\nabla\phi\|^{s+\frac12}
\|\nabla^2\phi\|^{\frac32-s}.
\end{aligned}
\]
For the pressure remainder, use the factorization
\[
p(1+q)-p(1)-\gamma q
=q\int_0^1\bigl(p'(1+\theta q)-p'(1)\bigr)\,\mathrm{d}\theta.
\]
The $L^2$ norm of the second factor is bounded by $C\|q\|$.
Applying \eqref{eq:negative-product} to these two factors gives the
bound $C_s\|q\|^{s+\frac12}\|\nabla q\|^{\frac32-s}$.
For the viscous corrections, write
$\nabla(qu)=q\nabla u+u\otimes\nabla q$ and use
\[
\begin{aligned}
\|\Lambda_y^{-s}(q\nabla u)\|
&\leq C_s\|q\|^{s-\frac12}
\|\nabla q\|^{\frac32-s}\|\nabla u\|,\\
\|\Lambda_y^{-s}(u\otimes\nabla q)\|
&\leq C_s\|u\|^{s-\frac12}
\|\nabla u\|^{\frac32-s}\|\nabla q\|.
\end{aligned}
\]
In these bounds the last factor is controlled by $CN(t)$, while the
factor with exponent $\frac32-s$ is controlled by
$CD(t)^{(\frac32-s)/2}$. This proves \eqref{eq:negative-stress}.

Define the negative-order energy
\[
\mathcal L_s(t)=\frac12\left(
\gamma\|\Lambda_y^{-s}q(t)\|^2
+\|\Lambda_y^{-s}m(t)\|^2
+\|\Lambda_y^{-s}\nabla\phi(t)\|^2\right).
\]
The operators $\Lambda_y^{-s}$ and $\mathbb Q$ commute, since the
characterization \eqref{eq:projection-definition} is invariant under
tangential translations. Apply $\Lambda_y^{-s}$ to
\eqref{eq:conservative-decay}, and take the inner products with
$\gamma\Lambda_y^{-s}q$, $\Lambda_y^{-s}m$ and
$\Lambda_y^{-s}\nabla\phi$, respectively.
The pressure terms cancel after integration by parts, and the electric
contribution to the fluid energy is
$(\Lambda_y^{-s}\nabla\phi,\Lambda_y^{-s}m)$ on the right-hand side.
On the other hand, since $\mathbb Q$ is an orthogonal projection and
$\mathbb Q\nabla\phi=\nabla\phi$,
\[
\begin{aligned}
\frac12\frac{\,\mathrm{d}}{\,\mathrm{d} t}\|\Lambda_y^{-s}\nabla\phi\|^2
&=-(\mathbb Q\Lambda_y^{-s}m,\Lambda_y^{-s}\nabla\phi)\\
&=-(\Lambda_y^{-s}m,\Lambda_y^{-s}\nabla\phi).
\end{aligned}
\]
Adding these identities cancels the electric terms. Since
$\Lambda_y^{-s}m=0$ on $\partial\Omega$, we obtain
\[
\frac{\,\mathrm{d}}{\,\mathrm{d} t}\mathcal L_s
+a(\Lambda_y^{-s}m,\Lambda_y^{-s}m)
=-(\Lambda_y^{-s}S,\nabla\Lambda_y^{-s}m).
\]
Using $(\epsilon^2-\Delta_y)^{-s/2}$ in the same calculation,
we apply Young's inequality and \eqref{eq:negative-stress}, integrate
in time, and let $\epsilon\downarrow0$ by monotone convergence to obtain
\begin{equation}\label{eq:negative-energy-integral}
\begin{aligned}
\mathcal L_s(t)
+c\int_0^t\|\nabla\Lambda_y^{-s}m(\tau)\|^2\,\mathrm{d}\tau
\leq \mathcal L_s(0)
+C_s\int_0^tN(\tau)^{2s+1}D(\tau)^{\frac32-s}\,\mathrm{d}\tau.
\end{aligned}
\end{equation}
The right-hand side is finite on each finite interval by
\eqref{eq:decay-energy}. Moreover,
\begin{equation}\label{eq:negative-momentum-transfer}
\|\Lambda_y^{-s}(qu)\|
\leq C_s\|q\|^{s-\frac12}
\|\nabla q\|^{\frac32-s}\|u\|
\leq C_sN(t)^2.
\end{equation}
Thus \eqref{eq:negative-data} implies
$\mathcal L_s(0)\leq C_s(M_{s,0}^2+\varepsilon_0^4)$.
Since $u=m-qu$, we also have
\[
\begin{aligned}
\|\Lambda_y^{-s}u(t)\|
&\leq \|\Lambda_y^{-s}m(t)\|
+\|\Lambda_y^{-s}(qu)(t)\|
\leq \sqrt{2\mathcal L_s(t)}+C_sN(t)^2,\\
\|\Lambda_y^{-s}\nabla\phi(t)\|
&\leq\sqrt{2\mathcal L_s(t)}.
\end{aligned}
\]

We next relate a uniform bound for $\mathcal L_s$ to the decay of the
full energy. Suppose that $\sup_{t\geq0}\mathcal L_s(t)<\infty$.
The preceding bounds then control both negative norms uniformly.
The regularity in Theorem~\ref{thm:global} also gives
$u,\nabla\phi\in H^1$, so \eqref{eq:tangential-interpolation} applies
to each of them. Together with Lemma~\ref{lem:poisson}, it yields
\begin{equation}\label{eq:decay-low-order}
\begin{aligned}
\|u\|^2+\|\nabla\phi\|^2
&\leq C_s\bigl(\mathcal L_s(t)+\varepsilon_0^4\bigr)^{1/(1+s)}
\bigl(\|\nabla_yu\|^2
+\|\nabla_y\nabla\phi\|^2\bigr)^{s/(1+s)}\\
&\leq C_sD(t)^{s/(1+s)}.
\end{aligned}
\end{equation}
Here and in the following conditional decay estimate, the constants
may also depend on $\sup_{t\geq0}\mathcal L_s(t)$.
The $L^2$ norm of $u_t$ is estimated from the momentum equation.
In fact, directly from \eqref{eq:F},
\[
\|F\|\leq CN(t)\bigl(\|\nabla u\|_{H^1}+\|\nabla q\|\bigr),
\]
and consequently
\[
\|u_t\|^2
\leq C\bigl(\|\nabla u\|_{H^1}^2
+\|q\|_{H^1}^2+\|\nabla\phi\|^2\bigr).
\]
It follows that
\[
N(t)^2\leq C\bigl(
\|q\|_{H^3}^2+\|\nabla u\|_{H^2}^2+\|\nabla u_t\|^2
+\|u\|^2+\|\nabla\phi\|^2\bigr).
\]
The sum of the first three terms is bounded both by $D(t)$ and by
$CN(t)^2\leq C\varepsilon_0^2$. Hence it is bounded by
$C\varepsilon_0^{2/(1+s)}D(t)^{s/(1+s)}$.
Together with \eqref{eq:decay-low-order}, this proves
$\mathcal E\leq C_sD(t)^{s/(1+s)}$. We can therefore use
\eqref{eq:decay-energy} to obtain
\[
\frac{\,\mathrm{d}}{\,\mathrm{d} t}\mathcal E+c_s\mathcal E^{1+1/s}\leq0,
\]
and integration gives
\begin{equation}\label{eq:energy-decay-implication}
\mathcal E(t)\leq C_s(1+t)^{-s},\qquad
N(t)\leq C_s(1+t)^{-s/2},
\end{equation}
whenever $\mathcal L_s$ is uniformly bounded.

We now establish the required uniform bound.
For $\frac12<s<1$, Plancherel's theorem and H\"older's inequality give
\[
\begin{aligned}
M_{1/2,0}
&=\|\Lambda_y^{-1/2}(q_0,u_0,\nabla\phi_0)\|\\
&\leq
\|(q_0,u_0,\nabla\phi_0)\|^{1-\frac1{2s}}
M_{s,0}^{\frac1{2s}}
\leq \varepsilon_0^{1-\frac1{2s}}M_{s,0}^{\frac1{2s}}<\infty.
\end{aligned}
\]
For $s=\frac12$, this finiteness is precisely
\eqref{eq:negative-data}. In either case,
\eqref{eq:negative-momentum-transfer} at $t=0$ implies
$\mathcal L_{1/2}(0)\leq C(M_{1/2,0}^2+\varepsilon_0^4)$.
Applying \eqref{eq:negative-energy-integral} with exponent $\frac12$
and using \eqref{eq:decay-energy}, we obtain
\[
\sup_{t\geq0}\mathcal L_{1/2}(t)
\leq \mathcal L_{1/2}(0)+C\int_0^\infty N(t)^2D(t)\,\mathrm{d} t
\leq C(M_{1/2,0}^2+\varepsilon_0^4).
\]
Equation \eqref{eq:energy-decay-implication} with exponent $\frac12$
therefore yields
\begin{equation}\label{eq:half-order-decay}
N(t)\leq C_s(1+t)^{-1/4},
\end{equation}
where $C_s$ depends only on the parameters and initial norms listed
in Theorem~\ref{thm:decay}. This proves the desired decay when
$s=\frac12$.
For $\frac12<s<1$, we use \eqref{eq:half-order-decay} and
H\"older's inequality in time to obtain
\[
\begin{aligned}
\int_0^\infty N(t)^{2s+1}D(t)^{\frac32-s}\,\mathrm{d} t
&\leq
\left(\int_0^\infty D(t)\,\mathrm{d} t\right)^{\frac32-s}
\left(\int_0^\infty
N(t)^{(2s+1)/(s-\frac12)}\,\mathrm{d} t\right)^{s-\frac12}\\
&<\infty,
\end{aligned}
\]
since $(2s+1)/(4(s-\frac12))>1$ for $\frac12<s<1$.
Equation \eqref{eq:negative-energy-integral} proves that
$\mathcal L_s$ is uniformly bounded. We may thus apply
\eqref{eq:energy-decay-implication} with the given value of $s$.

Finally, Lemma~\ref{lem:poisson} and \eqref{eq:qt-estimates} imply
\[
\|\nabla\phi\|_{H^3}+\|q_t\|_{H^2}\leq CN(t).
\]
This proves \eqref{eq:solution-decay}. Integrating
\eqref{eq:decay-energy} from $t$ to infinity gives
\[
\int_t^\infty D(\tau)\,\mathrm{d}\tau\leq C\mathcal E(t)
\leq C_s(1+t)^{-s}.
\]
Moreover, \eqref{eq:qt-estimates} and $N(t)\leq\delta\leq1$ give,
for almost every $t\geq0$,
\[
\|q_t\|_{H^2}^2
\leq C\|\nabla u\|_{H^2}^2+CN(t)^2\|q\|_{H^3}^2
\leq CD(t).
\]
This proves \eqref{eq:dissipation-tail}.
