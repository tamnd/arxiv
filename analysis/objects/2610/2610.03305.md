\label{Thm: PMP}
    Let $\gamma:[0,T]\to{\mathcal {AA}}$ be a geodesic and let $\bar{u}(\cdot)=(\bar{u_1}(\cdot),\bar{u_2}(\cdot))$ be the corresponding minimal control, i.e., for a.e. $s\in[0,T]$,
    \begin{equation*}
        \dot{\gamma}(s)=\bar{u_1}(s)U_{\gamma(s)}+\bar{u_2}(s)V_{\gamma(s)},\quad \ell(\gamma)=\int_0^T \sqrt{\bar{u_1}^2(s)+\bar{u_2}^2(s)}\,ds=d_{\mathcal {AA}}(\gamma(0),\gamma(T)),
    \end{equation*}
    with $\sqrt{\bar{u_1}^2(s)+\bar{u_2}^2(s)}=1$ a.e. on $[0,T]$. Then there exists a curve $\mu:[0,T] \to{\rm T}^*{\mathcal {AA}}$ with initial data $\mu(0)$ identified with $(\gamma(0),\xi(0))$, such that
    \begin{equation}\label{eq: Hamilton eq}
        \dot{\mu}(s)=\sum_{i=1}^2\bar{u_i}(s)\overrightarrow{h_i}(\mu(s)),\quad\text{a.e. }s\in [0,T],
    \end{equation}
and such that one of the following conditions is satisfied:
\begin{align}
    h_i(\mu(s))&=\bar{u_i}(s), \;\forall s\in[0,T],\;\forall i\in\{1,2\}, \label{eq: Normal extr}\tag{N}\\
    h_i(\mu(s))&=0, \;\forall s\in[0,T],\;\forall i\in\{1,2\} \label{eq: Abnormal extr}\tag{A}.
\end{align}
Moreover, in case \eqref{eq: Abnormal extr} one has $\mu(s)\neq0$ for all $s\in[0,T]$.
<!-- end thm-1 -->
<!-- rem-1: body_md -->
An extremal trajectory $\gamma:[0,T]\to{\mathcal {AA}}$ which satisfies condition \eqref{eq: Normal extr} (respectively \eqref{eq: Abnormal extr}) is called \textit{normal trajectory} (respectively \textit{abnormal trajectory}). If an extremal trajectory satisfies only \eqref{eq: Abnormal extr}, then it is said to be a \textit{strictly abnormal trajectory}.

        \item $X,Y$ are connected,

\label{thm: Geodesics of AA}
Let \eqref{eq: initial data} be the initial data, with $\eta_0^2 + (\xi_0+\tau_0)^2 = 1$. The sub-Riemannian geodesics $\gamma:[0,T]\to{\mathcal {AA}}$ are given by the following cases:
\begin{enumerate}
    \item if $\tau_0=0$ and $\eta_0=0$, then $\xi_0=\pm 1$ and the geodesic is the straight line $\gamma(s)=(1, \xi_0 s,\xi_0 s)$;
    \item if $\tau_0=0$ and $\eta_0\neq 0$, then $\gamma(s)=\left(e^{\eta_0 s}, \frac{\xi_0}{\eta_0}\left(e^{\eta_0s}-1\right),\xi_0 s\right)$;
    \item if $\tau_0\neq0$, then $\gamma(s)=(\lambda(s),t(s),a(s))$ is given by:
     \begin{equation}
        \begin{aligned}\label{AAGeod3}
        \lambda(s) &= \frac{\sin\phi(s)-\xi_0}{\tau_0},\\
        t(s) &= \frac{\eta_0-\cos\phi(s)}{\tau_0},\\
        a(s) &= \phi(s) - \phi_0 + \xi_0 s,
    \end{aligned}
    \end{equation}
    where $\phi(0)=\phi_0 $ such that $\cos\phi_0 = \eta_0$ and $\sin\phi_0 = \xi_0+\tau_0$, and $\phi:[0,T]\to{{\mathbb R}}$ is a smooth function which satisfies the o.d.e. $\dot{\phi}(s) = \sin\phi(s) - \xi_0,\,\forall s\in(0,T)$.
    
    Explicitly, \begin{itemize}
    \item if $|\xi_0| < 1$ and $\xi_0\neq 0$, then
    \begin{align}\label{eq: phi 1}
        \phi(s) &= 2\arctan \left( \frac{1}{\xi_0} + \frac{\sqrt{1-\xi_0^2}}{\xi_0} \tanh \left( \frac{(s+c_1)\sqrt{1-\xi_0^2}}{2} \right) \right), 
    \end{align}
  \item if $\xi_0= 0$, then
 \begin{align}\label{eq: phi 2}
     \phi (s) &=2 \arctan \left( \pm e^s c_1' \right),
 \end{align}

    
    \item if $\xi_0 < -1$, then
    \begin{align}\label{eq: phi 3}
        \phi(s) &= 2\arctan \left( \frac{1}{\xi_0} - \frac{\sqrt{\xi_0^2-1}}{\xi_0} \tan \left( \frac{(s+c_2)\sqrt{\xi_0^2-1}}{2} \right) \right) ,
    \end{align}
    \item if $\xi_0 = -1$, then
    \begin{align}\label{eq: phi 4}
        \phi(s) &= 2\arctan \left( -\frac{2+s+c_3}{s+c_3} \right),
    \end{align}
\end{itemize}
for some constants $c_1, c_1', c_2, c_3 \in \mathbb{R}$ given by
 \begin{align*}
    c_1 &= \frac{2}{\sqrt{1-\xi_0^2}} \mathrm{arctanh}\left( \frac{ \xi_0(\xi_0+\tau_0)-(1+\eta_0) }{ (1+\eta_0)\sqrt{1-\xi_0^2} } \right),\\
    c_1'&=\ln \left| \frac{\xi_0+\tau_0}{1+\eta_0} \right| ,\\
    c_2&=\frac{2}{\sqrt{\xi_0^2-1}} \left( \arctan\left( \frac{ 1+\eta_0-\xi_0(\xi_0+\tau_0) }{ (1+\eta_0)\sqrt{\xi_0^2-1} } \right) \right), \\
    c_3&= -\frac{2(1+\eta_0)} {\eta_0+\tau_0}.
\end{align*}
\end{enumerate}
The cases for $\xi_0 \geq 1$ are recovered by the action of the reflection $j:{\mathcal {AA}}\to{\mathcal {AA}}$.

We begin our proof by looking at system \eqref{eq: ode_system} which we rewrite as follows
\begin{align}\label{eq: ode_system_1}
\begin{cases}
    \dot \lambda(s) = \lambda(s) \cos \phi(s),\\
    \dot t(s) = \lambda(s) \sin \phi(s),\\
    \dot a(s) = \sin \phi(s),\\
    \dot{\xi_1}(s)=-\cos\phi(s)\xi_1(s)-\tau_0\sin\phi(s),\\
    \xi_2(s)=\tau_0,\\
    \xi_3(s)=\xi_0,
\end{cases}\hspace{-.4cm},\quad s\in(0,T).
\end{align}
First we consider the case $\tau_0=0$. The fifth and sixth equations of system \eqref{eq: ode_system_1}, together with definitions \eqref{eq: h1, h_2 applied}, imply that 
\begin{equation}\label{eq: h2}
    h_2(s)=\xi_0 , \quad\forall s\in[0,T].
\end{equation}
If also $\eta_0=0$, the assumption on the initial data $\xi(0)$ yields $\xi_0=\pm 1$. Thus \eqref{eq: phi} imposes that 
\begin{equation*}
    \sin\phi(s)=\xi_0,\,\cos\phi(s)=0, \quad\forall s \in[0,T],
\end{equation*}
and system \eqref{eq: ode_system_1} gives the curve 
\begin{equation*}
    \gamma(s)=(1, \xi_0 s,\xi_0 s),\quad s\in[0,T]. 
\end{equation*}
If $\eta_0\neq0$, the initial data conditions for $\gamma(0)$ and $\xi(0)$ together with \eqref{eq: h2} imply that 
\begin{equation}\label{eq: h1}
    h_1(s)=\eta_0 ,\quad \forall s\in[0,T].
\end{equation}
Thanks to \eqref{eq: phi} and \eqref{eq: h1}, the fourth equation of system \eqref{eq: ode_system_1} becomes the o.d.e.
\begin{equation}\label{eq: ode xi_1}
    \begin{aligned}
  \dot{\xi_1}(s)&=-\eta_0\,\xi_1(s),\quad s\in(0,T),\\
  \xi_1(0)&=\eta_0,
\end{aligned}
\end{equation}
which gives 
\begin{equation*}
    \xi_1(s)=\eta_0 e^{-\eta_0 s},\quad s\in[0,T].
\end{equation*}
The latter expression combined with \eqref{eq: h1, h_2 applied} and \eqref{eq: h1} gives that
\begin{equation}\label{eq: lambda deg}
    \lambda(s)=e^{\eta_0 s},\quad s\in[0,T].
\end{equation}
Using \eqref{eq: phi}, \eqref{eq: h1}, \eqref{eq: h2} and \eqref{eq: lambda deg} we integrate the second and third equations from system \eqref{eq: ode_system_1} and obtain
\begin{equation*}
    \gamma(s)=\left(e^{\eta_0 s},\frac{\xi_0}{\eta_0}(e^{\eta_0s}-1),\xi_0 s\right), \quad s\in[0,T].
\end{equation*}
Now we assume $\tau_0\neq0$. From \eqref{eq: phi}, \eqref{eq: h1, h_2 applied}, the first and the last three equations of system \eqref{eq: ode_system_1} we obtain the o.d.e.
\begin{equation}\label{eq: ode phi}
    \dot{\phi}(s)=\sin\phi(s)-\xi_0,\quad s\in (0,T),
\end{equation} 
where $\phi(0)=\phi_0$ satisfies
\begin{equation*}
    \cos\phi_0=\eta_0 \text{ and } \sin\phi_0=\xi_0+\tau_0.
\end{equation*}
 Thus, integrating $\dot{a}(s) = \sin\phi(s) = \dot{\phi}(s) + \xi_0$, we have $a(s) = \phi(s) - \phi_0 + \xi_0 s$.

For $\lambda(s)$, notice from \eqref{eq: ode_system_1} that:
$$ \frac{\dot{\lambda}(s)}{\lambda(s)} = \cos\phi(s) = \frac{\dot{\phi}(s)\cos\phi(s)}{\sin\phi(s) - \xi_0}. $$
Integrating the latter with respect to $s\in[0,T]$ yields 
 \begin{equation}\label{eq: lambda}
     \lambda(s) = \frac{\sin\phi(s) - \xi_0}{\sin\phi_0 - \xi_0} = \frac{\sin\phi(s) - \xi_0}{\tau_0},\quad s\in[0,T].
 \end{equation}
For $t(s)$, we recall \eqref{eq: h1, h_2 applied} and system \eqref{eq: ode_system_1} and get:
$$ \dot{t}(s) = \lambda(s)\,\sin\phi(s) = \frac{(\sin\phi(s)-\xi_0)\sin\phi(s)}{\tau_0} = \frac{\dot{\phi}(s)\sin\phi(s)}{\tau_0}. $$
Integrating with respect to $s\in[0,T] $ gives $t(s) = \frac{\cos\phi_0 - \cos\phi(s)}{\tau_0}$. To sum up, for $\tau_0\neq0$ the geodesic $\gamma$ is given by
\begin{equation}\label{eq: gamma tau not 0}
    \gamma(s)=\left(\frac{\sin\phi(s) - \xi_0}{\tau_0},\frac{\cos\phi_0 - \cos\phi(s)}{\tau_0},\phi(s) - \phi_0 + \xi_0 s\right),\quad s\in[0,T].
\end{equation}
We are going to solve the o.d.e. \eqref{eq: ode phi} which we rewrite as \begin{equation}\label{eq: angle ode} 
\frac{\dot{\phi}(s)}{\sin\phi(s) - \xi_0} = 1, \quad s \in(0,T).
\end{equation} Using the half-angle substitution $x(s)=\tan\left(\frac{\phi(s)}{2}\right)$ and the associated identities relating the parameter $x$ with $\sin \phi$ and $\cos \phi$, the o.d.e. \eqref{eq: angle ode} becomes:
\begin{equation}\label{eq: phi4}
   -\frac{2}{\xi_0} \frac{\dot{x}(s)}{\left( x(s)-\frac{1}{\xi_0} \right)^2+1-\frac{1}{\xi_0^2}}=1,\quad s\in(0,T).
\end{equation}
Without loss of generality we assume $\tau_0>0$. We observe that the o.d.e. \eqref{eq: ode phi} and \eqref{eq: lambda} show that $\tau_0>0$ is equivalent to $\dot{\phi}(s)>0$, $\forall s\in(0,T)$.
We recall that assumption $\eta_0^2 + (\xi_0 + \tau_0)^2 = 1$ gives $|\xi_0 + \tau_0|\leq 1$ and since $\tau_0 > 0$, we must have $\xi_0 < 1$. Thus, we consider three distinct ranges for $\xi_0$: $\xi_0 \in (-1, 1)$, $\xi_0 < -1$, and $\xi_0 = -1$.
Solving the o.d.e. \eqref{eq: phi4} yields the explicit solutions \eqref{eq: phi 1}, \eqref{eq: phi 2},\eqref{eq: phi 3} and \eqref{eq: phi 4}.

In the same spirit, if we assume $\tau_0<0$, it guarantees $\dot{\phi}(s)<0$, $\forall s\in(0,T)$. Thus, we must have $\xi_0>-1$ splitting into three distinct cases for $\xi_0$: $\xi_0\in (-1,1),\, \xi_0>1$ and $\xi_0=1$. Working similarly as in case $\tau_0>0$, one can find that the solution is the reflection of the trajectory $\gamma(s)$ given by \eqref{eq: gamma tau not 0}, i.e. the curve $j(\gamma(s))=\left(\lambda(s),-t(s),-a(s) \right)$.

Without loss of generality, we assume $\sin\phi_0-\xi_0>0$. Indeed, one may recover the case $\sin\phi_0-\xi_0<0$ by means of the reflection $j$. We consider the following distinction:\\
\noindent{} \textit{Case I:} If $\xi_0<-1$, by the proof of Theorem \ref{thm: Geodesics of AA}, the following conditions are satisfied:
    $$\dot{\phi}(s)=\sin\phi(s)-\xi_0, \qquad \dot{\phi}(0)=\tau_0.$$

Since $\sin\phi(s)-\xi_0\geq-1-\xi_0>0$, we find that $\dot{\phi}(s)>0$, for all $s>0$. Therefore, $\phi(s)$ is strictly-increasing. Using \eqref{eq : Txi_02} and the half-angle substitution $z=\tan\frac{u}{2}$, we calculate:
$$T_{\xi_0}=\frac{2\pi}{\sqrt{\xi_0^2-1}}.$$

\noindent{} \textit{Case II:} If $-1\leq \xi_0\leq1$, the equation $\sin\phi=\xi_0$, has a solution $\phi^*$ between the interval $(\phi_0, \phi_0+2\pi)$. Let $S$ be the time such that $\phi(S)=\phi_0+2\pi$. Due to continuity of $\phi(s)$, there is an $s^*\in (0,S)$, such that $\phi(s^*)=\phi^*$. Hence, it is $\dot{\phi}(s^*)=0$.

In addition, the constant function $\phi(s)=\phi^*$ is also a solution of the o.d.e. \eqref{eq: angle ode}. $$\phi_0<\phi(s)<\phi^*<\phi_0+2\pi, $$
for every finite $s>0$. Then, $\phi$ cannot complete a full rotation in finite time and combining with \eqref{eq : Txi0}, it is $T_{\xi_0}=+\infty$.

Finally, if $\sin\phi_0-\xi_0=0$, then the o.d.e. \eqref{eq: ode phi} reduces to $\dot{\phi}(s)=0$. Thus $\phi(s)\equiv \phi_0$ is a solution. Then, again by \eqref{eq : Txi0}, we obtain $T_{\xi_0}=+\infty$.

    \gamma(s)=\exp_{{\rm e}}(\xi_0,\phi_0,s)=\left(\lambda(s),t(s),a(s)\right), \quad s\in [0,T],
\end{equation}    
    is the solution from Theorem \ref{thm: Geodesics of AA}, where $\hat{p}=\gamma(T)$ denotes its endpoint and $\phi(s)$ is the corresponding angular solution. 
    In what follows we construct pairs $(\varepsilon_i,\sigma_i), \, i\in\{1,2,3\}$ that are non-trivial symmetries with the purpose of determining $\rm Max$.\\
    We define the mappings $\varepsilon_i : C_{{\rm e}}\times {{\mathbb R}}_+\to C_{\rm e}\times {{\mathbb R}}_+, \, i\in\{1,2,3\},$
    given by
    \begin{align}
        \varepsilon_1(\xi_0,\phi_0,T)&=(-\xi_0,-\phi_0,T),\label{e1}\\ \varepsilon_2(\xi_0,\phi_0,T)&=(-\xi_0,\phi(T)+\pi,T),\quad (\xi_0,\phi_0,T)\in C_{\rm e}\times{{\mathbb R}}_+. \notag\\ \varepsilon_3(\xi_0,\phi_0,T)&=(\xi_0,\pi-\phi(T),T),\notag
    \end{align}
Further, the mappings $\sigma_i : {\mathcal {AA}}\to {\mathcal {AA}}, \, i\in\{1,2,3\},$ are given by $$\sigma_1(p)=j(p), \quad \sigma_2(p)=p^{-1}, \quad \sigma_3(p)=j(p^{-1}),\quad p\in{\mathcal {AA}}.  $$
Recalling the curve $\gamma$ given in \eqref{eq: gamma Max}, one can verify that the curves
$$\gamma_1(\cdot)=j(\gamma(\cdot)), \quad \gamma_2(\cdot)=\gamma(T)^{-1}\cdot \gamma(T-\cdot), \quad \gamma_3(\cdot)=j(\gamma_2(\cdot)),$$
correspond to the initial covectors $(-\xi_0,-\phi_0), \, (-\xi_0,\phi(T)+\pi), \, (\xi_0,\pi-\phi(T))$, respectively. Therefore, by the uniqueness of the solutions, we write
\begin{align*}
    \gamma_1(s)&=\exp_{\rm e}(-\xi_0,-\phi_0,s)\\
    \gamma_2(s)&=\exp_{\rm e}(-\xi_0,\phi(T)+\pi,s)\\
    \gamma_3(s)&=\exp_{\rm e}(\xi_0,\pi-\phi(T),s).
\end{align*}
Computing at $s=T>0$, we obtain
$\gamma_i(T)=\sigma_i(\gamma(T))$ for $i\in\{1,2,3\}$. In other words,
$$\exp_{\rm e}\left(\varepsilon_i(\xi_0,\phi_0,T)\right)=\sigma_i\left(\exp_{\rm e}(\xi_0,\phi_0,T)\right), \quad i\in\{1,2,3\}.$$
Since $T>0$ was assumed arbitrary, the pairs $(\varepsilon_i,\sigma_i), \, i\in\{1,2,3\}$ satisfy the symmetry definition (\ref{eq: symmetry}). 

Now, for some $T_0>0$, a Maxwell point $p_0=\gamma(T_0)$ is generated by $(\varepsilon_i,\sigma_i), \, i\in\{1,2,3\}$ and it must satisfy 
\begin{equation*}\label{max123}
\varepsilon_i(\xi_0,\phi_0,T_0)\neq (\xi_0,\phi_0,T_0) \quad \text{and} \quad \sigma_i(p_0)=p_0,\;  \forall i\in\{1,2,3\}.
\end{equation*}
We study each case separately:


    \noindent{} \textit{Case I:} We have
    $$\gamma_1(T_0)=\gamma(T_0)\iff (\lambda(T_0),-t(T_0),-a(T_0))=(\lambda(T_0),t(T_0),a(T_0))\iff t(T_0)=a(T_0)=0,$$
which, after combining the last fact with the second and the third o.d.e of system (\ref{eq: ode_system}), leads to $\sin\phi(s)\equiv 0,$ for all $s\in (0,T_0)$. Therefore, $\phi(s)\equiv 0$ or $\phi(s)\equiv \pi$ for all $s\in (0,T_0)$, which yields $\gamma_1(s)=\gamma(s)=(e^{\pm s},0,0)$, for all $s\in (0,T_0)$, thus contradicting the definition of a Maxwell point. 

\noindent{} \textit{Case II:} It holds
$$\gamma_2(T_0)=\gamma(T_0)\iff (\lambda(T_0)^{-1},-t(T_0)\cdot\lambda(T_0)^{-1},-a(T_0))=(\lambda(T_0),t(T_0),a(T_0)).$$
Thus $\lambda(T_0)=1, \, t(T_0)=0, \, a(T_0)=0$. Following the same argument of the first case, we have $\gamma(s)=(e^{\pm s},0,0)$, which leads to a contradiction since $\gamma(T_0)=(e^{\pm T_0},0,0)\neq (1,0,0)$.

\noindent{} \textit{Case III:} For the third symmetry, we have 
$$\gamma_3(T_0)=\gamma(T_0)\iff (\lambda(T_0)^{-1},t(T_0)\cdot \lambda(T_0)^{-1},a(T_0))=(\lambda(T_0),t(T_0),a(T_0)).$$
The last condition holds if and only if $\lambda(T_0)=1$. Assuming that $\sin\phi_0-\xi_0\neq 0$, by the expression of $\lambda(s)$ in (\ref{AAGeod3}), we obtain
$$\phi(T_0)=\phi_0+2k\pi \quad \text{or} \quad\phi(T_0)=\pi-\phi_0+2k\pi, \quad k\in\mathbb{Z}.$$
If $k=0$ we do not get any Maxwell points. If $k\neq0$ we have $\gamma_3(s)=\gamma(s)$ for all $s\in (0,T_0).$ Then $\phi(T_0)=\phi_0+2k\pi, \quad k\in \mathbb{Z}\setminus\{0\}$. Therefore $T_0=2k \,T_{\xi_0}=\frac{2k\pi}{\sqrt{\xi_0^2-1}}>0, \, k\in\mathbb{Z}\setminus\{0\}$ is equivalent to $ |\xi_0|>1$, which produces Maxwell points of the form $(1,0,a), \, a\neq0$, i.e. $(1,0,a)\in\mathcal{L}$.\\ 
If $\sin\phi_0-\xi_0=0$, using the solutions of Theorem \ref{thm: Geodesics of AA} again, we do not have Maxwell points.

Combining Cases $I-III$, we find that the non-trivial symmetries $(\varepsilon_i,\sigma_i), \, i\in\{1,2,3\}$ generate Maxwell points on $\mathcal{L}$. In other words, $\mathrm{Max}\subset \mathcal{L}$, which completes the proof.

By definition, for a conjugate time $s>0$ we have that the tangent map of the exponential map
$
D\,\exp_{\rm e}\big|_{(\xi_0,\phi_0,s)}:{\rm T}_{(\xi_0,\phi_0,s)}C_{\rm e}\times {{\mathbb R}}_+\to{\rm T}_{\gamma(s)}{\mathcal {AA}}
$ is not surjective.
Following the reasoning from \cite[Section 3.1.3]{BR-CCmetrics}, we have that the latter condition is equivalent to the linear dependence of the vectors
                  
$\gamma(s)^{-1}\cdot
\frac{\partial\exp_{\rm e}}{\partial\xi_0},
\quad    
\gamma(s)^{-1}\cdot
\frac{\partial\exp_{\rm e}}{\partial\phi_0},
\quad
\gamma(s)^{-1}\cdot
\frac{\partial\exp_{\rm e}}{\partial s},
$
                
with $\gamma(s)=\exp_{\rm e}(\xi_0,\phi_0,s)$. Hence, the conjugate times are determined by
$
 J=0,
$
where $J$ is the Jacobian determinant of the matrix representing $D\exp_{\rm e}\big|_{(\xi_0,\phi_0,s)}$ with respect to the bases $\left\{ \partial_{\xi_0},\partial_{\phi_0},\partial_s\right\}\rvert_{(\xi_0,\phi_0,s)}$ and $\left\{U,V,W\right\}\rvert_{\gamma(s)}.$

Thus, the conjugate times are the positive solutions of
\begin{equation}\label{eq: J=0}
     J=0 .
\end{equation}
We now assume $\sin\phi_0-\xi_0>0$, we shall consider the cases $\sin\phi_0-\xi_0<0$ and $\sin\phi_0-\xi_0=0$ at the end of the proof.

\noindent \textit{Case I:} $\xi_0<-1$. We let
$
\kappa=\sqrt{\xi_0^2-1}$ and
$
\Theta=\frac{\kappa s}{2}.
$
Using the half-angle substitution
$z=\tan\frac{u}{2}$ in (\ref{arclength}),
we obtain
$$
s
=
\int_{\tan(\phi_0/2)}^{\tan(\phi/2)}
\frac{2\,dz}{2z-\xi_0(1+z^2)}.
$$
Since
$
2z-\xi_0(1+z^2)
=
-\frac{\kappa^2+(1-\xi_0z)^2}{\xi_0},
$
we find
$$
s
=
\frac{2}{\kappa}
\left[
\arctan\left(
\frac{1-\xi_0\tan(u/2)}{\kappa}
\right)
\right]_{u=\phi_0}^{u=\phi}.
$$
Therefore,
\begin{equation}\label{eq: Theta}
    \Theta
=
\arctan\left(
\frac{1-\xi_0\tan(\phi/2)}{\kappa}
\right)
-
\arctan\left(
\frac{1-\xi_0\tan(\phi_0/2)}{\kappa}
\right) +k\pi,\quad k\in\mathbb{Z}.
\end{equation}
When
\begin{equation}\label{eq: tat hyp}
    1+\left(
\frac{1-\xi_0\tan(\phi/2)}{\kappa}
\right)\left(
\frac{1-\xi_0\tan(\phi_0/2)}{\kappa}
\right)\neq0,
\end{equation}
we rewrite \eqref{eq: Theta} using the identity
$
\tan(\arctan x-\arctan y+k\pi)
=
\frac{x-y}{1+xy},
$ for all $k\in\mathbb Z$ and for all $x,y\in{{\mathbb R}}$ such that $1+xy\neq 0$,
resulting in
\begin{equation*}
\tan\Theta
=
\frac{
\kappa
\left(
\tan\frac{\phi}{2}-\tan\frac{\phi_0}{2}
\right)
}{
\tan\frac{\phi}{2}+\tan\frac{\phi_0}{2}
-
\xi_0
\left(
1+\tan\frac{\phi}{2}\tan\frac{\phi_0}{2}
\right)
}.
\end{equation*} 
By means of trigonometric identities, $\tan\Theta$ simplifies to
$\tan\Theta
=
\frac{
\kappa\sin\left(\frac{\phi-\phi_0}{2}\right)
}{
\sin\left(\frac{\phi+\phi_0}{2}\right)
-
\xi_0\cos\left(\frac{\phi-\phi_0}{2}\right)
}.
$
We rewrite the latter identity as
\begin{equation}\label{eq: J1}
\sin\left(\frac{\phi-\phi_0}{2}\right)
=
\frac{\tan\Theta}{\kappa}
\left(
\sin\left(\frac{\phi+\phi_0}{2}\right)
-
\xi_0\cos\left(\frac{\phi-\phi_0}{2}\right)
\right).
\end{equation}
Also, we observe that
\begin{align}\label{eq: J2}
(\sin\phi-\xi_0)(\sin\phi_0-\xi_0)
&=
\sin\phi\sin\phi_0
-\xi_0(\sin\phi+\sin\phi_0)
+\xi_0^2
\nonumber\\
&=
\left(
\sin\frac{\phi+\phi_0}{2}
-
\xi_0\cos\frac{\phi-\phi_0}{2}
\right)^2
+
(\xi_0^2-1)
\sin^2\frac{\phi-\phi_0}{2}.
\end{align}
Since $\kappa^2=\xi_0^2-1$ and
$s=\frac{2\Theta}{\kappa},
$
we have
\begin{align}\label{eq: J3}
&2\sin\frac{\phi-\phi_0}{2}
+
s\left(
\xi_0\cos\frac{\phi-\phi_0}{2}
-
\sin\frac{\phi+\phi_0}{2}
\right)
=
\frac{2}{\kappa}
\left(
\sin\frac{\phi+\phi_0}{2}
-
\xi_0\cos\frac{\phi-\phi_0}{2}
\right)
(\tan\Theta-\Theta).
\end{align}
Substituting \eqref{eq: J1}, \eqref{eq: J2} and \eqref{eq: J3} into the expression for $J$ given in Lemma \ref{lem: J}, yields:
\begin{equation*}\label{eq: newJ}
     J
=
\frac{4}{(\xi_0^2-1)^2}
\sin\Theta
\bigl(
\sin\Theta-\Theta\cos\Theta
\bigr).
\end{equation*}
From a straightforward computation we deduce that assumption \eqref{eq: tat hyp} does not hold if and only if $ \cos\Theta=0$. In this case we have $\sin\Theta=\pm1$ and combined with the continuity of $J$, leads to $J=\frac{4}{(\xi_0^2-1)^2}>0
$.





To sum up, we have
$
 J=0$ if and only if $
\sin\Theta=0
\text{ or }
\tan\Theta=\Theta.
$
The first positive zero of $\sin\Theta$ is $\Theta=\pi$. By the definition of $\Theta$ and $\kappa$
this yields the candidate conjugate time:
$$
s=\frac{2\pi}{\kappa}=\frac{2\pi}{\sqrt{\xi_0^2-1}}.
$$
It remains to verify that the second factor does not vanish at an earlier
time. Since $\Theta(s)$ is strictly increasing and $\Theta(0)=0$, we have
$$
0<s<\frac{2\pi}{\kappa}
\quad\Longleftrightarrow\quad
0<\Theta<\pi.
$$
We define the function
$
g(\Theta)=\sin\Theta-\Theta\cos\Theta.$
For $\Theta\in (0,\pi)$, we have
$
g'(\Theta)=\Theta\sin\Theta>0,
$
 $\Theta\in(0,\pi)$.
Since $g(0)=0$, it follows that
$
g(\Theta)>0$
for all $\Theta\in (0,\pi)$.
Moreover, $\sin\Theta>0$ for all $\Theta\in(0,\pi)$. Therefore,
$
 J>0
\text{ for all } s\in \left(0, \frac{2\pi}{\sqrt{\xi_0^2-1}}\right),
$
implying that no conjugate time occurs before $\Theta=\pi$. Consequently,
$\Theta=\pi$ gives the first solution of \eqref{eq: J=0}, and
$T_{\mathrm{conj}}(\xi_0,\phi_0)
=
\frac{2\pi}{\sqrt{\xi_0^2-1}},$
for $
\xi_0<-1.$

\noindent \textit{Case II:} $|\xi_0|<1$. Recalling that we assumed 
$\sin\phi_0>\xi_0$, we let
$\mu=\sqrt{1-\xi_0^2}$ and $\widetilde{\Theta}=\frac{\mu s}{2}
$.

As in the first case, we use (\ref{arclength}) and the half-angle substitution $z=\tan\frac{u}{2}$ to calculate:

$$
s=
\int_{\tan(\phi_0/2)}^{\tan(\phi/2)}
\frac{2\,dz}{2z-\xi_0(1+z^2)}.
$$
For $\xi_0\neq0$, the integration yields
\begin{equation}\label{eq: s1}
    s=
\frac{1}{\mu}
\left[
\log\left|
\frac{\mu+\xi_0\tan(u/2)-1}
{\mu-\xi_0\tan(u/2)+1}
\right|
\right]_{u=\phi_0}^{u=\phi}.
\end{equation}
For $\xi_0=0$, the integral is computed separately:
\begin{equation}\label{eq: s2}
    s=
\int_{\phi_0}^{\phi}\frac{du}{\sin u}
=
\left[
\log\left|\tan\frac{u}{2}\right|
\right]_{u=\phi_0}^{u=\phi}.
\end{equation}
We rewrite both \eqref{eq: s1} and \eqref{eq: s2} using
the hyperbolic trigonometric identity $
\tanh(x-y)
=
\frac{\tanh x-\tanh y}
{1-\tanh x\tanh y},
$ for all $x,y\in {{\mathbb R}}$, and obtain in both cases:
\begin{equation}\label{eq: Jh0}
    \tanh\widetilde{\Theta}
=
\frac{
\mu\sin\left(\frac{\phi-\phi_0}{2}\right)
}{
\sin\left(\frac{\phi+\phi_0}{2}\right)
-
\xi_0\cos\left(\frac{\phi-\phi_0}{2}\right)
}.
\end{equation}
We rewrite the latter identity as
\begin{equation}\label{eq: Jh1}
    \sin\frac{\phi-\phi_0}{2}
=
\frac{\tanh\widetilde{\Theta}}{\mu}
\left(
\sin\frac{\phi+\phi_0}{2}
-
\xi_0\cos\frac{\phi-\phi_0}{2}
\right).
\end{equation}
After making use of \eqref{eq: Jh0}, the following identity holds:
\begin{align}\label{eq: Jh2}
(\sin\phi-\xi_0)(\sin\phi_0-\xi_0)
&=
\left(
\sin\frac{\phi+\phi_0}{2}
-
\xi_0\cos\frac{\phi-\phi_0}{2}
\right)^2
(1-\tanh^2\widetilde{\Theta})
=
\frac{
\left(
\sin\frac{\phi+\phi_0}{2}
-
\xi_0\cos\frac{\phi-\phi_0}{2}
\right)^2
}{
\cosh^2\widetilde{\Theta}
}.
\end{align}
Moreover, since
$
s=\frac{2\widetilde{\Theta}}{\mu}$ and
$\xi_0^2-1=-\mu^2,$
we have
\begin{align}\label{eq: Jh3}
&2\sin\frac{\phi-\phi_0}{2}
+
s\left(
\xi_0\cos\frac{\phi-\phi_0}{2}
-
\sin\frac{\phi+\phi_0}{2}
\right)
=
\frac{2}{\mu}
\left(
\sin\frac{\phi+\phi_0}{2}
-
\xi_0\cos\frac{\phi-\phi_0}{2}
\right)
(\tanh\widetilde{\Theta}-\widetilde{\Theta}).
\end{align}
Substituting \eqref{eq: Jh1}, \eqref{eq: Jh2} and \eqref{eq: Jh3} into the expression for $J$ given in Lemma \ref{lem: J}, yields
$$
J
=
\frac{4}{\mu^4}
\tanh\widetilde{\Theta}\,\cosh^2\widetilde{\Theta}
(\widetilde{\Theta}-\tanh\widetilde{\Theta}).
$$
Finally, using
$
\widetilde{\Theta}-\tanh\widetilde{\Theta}
=
\frac{\widetilde{\Theta}\cosh\widetilde{\Theta}-\sinh\widetilde{\Theta}}{\cosh\widetilde{\Theta}}
$
and 
 $\mu^4=(1-\xi_0^2)^2$, the latter equation becomes
\begin{equation}
J=\frac{4}{(1-\xi_0^2)^2}
\sinh\widetilde{\Theta}
\bigl(
\widetilde{\Theta}\cosh\widetilde{\Theta}-\sinh\widetilde{\Theta}
\bigr).\label{detJ2}
\end{equation}
Since $\mu>0$ and $s>0$, we have $\widetilde{\Theta}>0$, and consequently
$\sinh\widetilde{\Theta}>0$.
To determine the sign of the second factor, we define
$
G(\widetilde{\Theta})
=
\widetilde{\Theta}\cosh\widetilde{\Theta}-\sinh\widetilde{\Theta}, \;{\Theta}>0.
$
Then,
$G(0)=0$
and
$
G'(\widetilde{\Theta})
=
\widetilde{\Theta}\sinh\widetilde{\Theta}>0
,$ 
for all $\widetilde{\Theta}>0.$
Thus,
$G(\widetilde{\Theta})>0$ for all
$\widetilde{\Theta}>0.$
Consequently,
$
 J>0,
\text{ for all } s>0.
$
Hence, the exponential map has no critical point at any positive time and
there is no finite conjugate time, i.e. 
$T_{\mathrm{conj}}(\xi_0,\phi_0)
=
+\infty,
\text{ for }
-1<\xi_0<1.$




\noindent \textit{Case III:} $\xi_0=-1$. The expression (\ref{detJ2}) obtained in Case II appears singular at $\xi_0=-1$.
This singularity is removable. Indeed, the Hamiltonian vector field is
smooth, and its flow depends smoothly on the initial covector. Consequently,
$\exp_{\rm e}$, its differential, and the determinant $J$ depend
continuously on $\xi_0$. Because we assumed $\sin\phi_0-\xi_0>0$, at $\xi_0=-1,$ we have $\sin\phi_0+1>0$. This, also holds for $\xi_0>-1$ sufficiently close to $-1$, which means that (\ref{detJ2}) can be used by taking the limit at $\xi_0\to-1^+$:
$$
J(-1,\phi_0,s)
=
\lim_{\xi_0\to-1^+}J(\xi_0,\phi_0,s).
$$
The Taylor expansions
$
\sinh\widetilde{\Theta}=\widetilde{\Theta}+O(\widetilde{\Theta}^3)$ and  
$\widetilde{\Theta}\cosh\widetilde{\Theta}-\sinh\widetilde{\Theta}
=
\frac{\widetilde{\Theta}^3}{3}+O(\widetilde{\Theta}^5),
$ for $\widetilde{\Theta}\to0$, together with the setting for Case II given by $\mu=\sqrt{1-\xi_0^2}$ and $
\widetilde{\Theta}=\frac{\mu s}{2}$ (since $\mu\to 0^+$, as $\xi_0\to -1^+$), 
imply
$$
J(-1,\phi_0,s)
=
\lim_{\mu\to0^+}
\frac{4}{\mu^4}
\sinh\widetilde{\Theta}
\bigl(
\Theta\cosh\widetilde{\Theta}-\sinh\widetilde{\Theta}
\bigr)
=
\frac{s^4}{12}.
$$
Thus, we conclude that
$
J(-1,\phi_0,s)>0
$ for all $s>0$. In other words,
$
T_{\mathrm{conj}}(-1,\phi_0)=+\infty.
$
To complete the proof, we consider the case $\sin\phi_0-\xi_0<0$. We recall the induced reflection \eqref{e1} on the costate $C_{\rm e}$, given by $\varepsilon_1(\xi_0,\phi_0,s)=(-\xi_0,-\phi_0,s)$. We have the following composition of tangent maps, induced by the symmetry $(\varepsilon_1,j)$:
$$\left(D\,\exp_{\rm e}\right)_{(-\xi_0,-\phi_0,s)}\circ D\,{\varepsilon_1}_{(\xi_0,\phi_0,s)}=D\,j_{\exp_{\rm e}(\xi_0,\phi_0,s)}\circ \left(D\,\exp_{\rm e} \right)_{(\xi_0,\phi_0,s)}.$$
Since $D\varepsilon_1$ and $Dj$ are invertible, it follows that
$\mathrm{rank}
\left(D\exp_{\rm e}\rvert_{(\xi_0,\phi_0,s)}\right)=\mathrm{rank}
\left(D\exp_{\rm e}\rvert_{(-\xi_0,-\phi_0,s)}\right).$

while the interval $[-\pi\sinh R,\pi\sinh R]$ parametrizes each generating curve of the flask, which stops being a geodesic when its projection completes a full hyperbolic circle.
