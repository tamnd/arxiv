<!-- theorem-1: body_md -->
\label{thm:main}
Equations \eqref{eq:frequency}--\eqref{eq:inverse} define a unique
nondecreasing sequence $A:\mathbb N\to\mathbb N$ that assumes every positive integer.
Its increments belong to $\{0,1\}$, and
\begin{equation}\label{eq:mainasymptotic}
 A(n)\sim Cn^\alpha,\qquad
 F(m)\sim CKm^\varphi=C^{-\varphi}m^\varphi.
\end{equation}
For every fixed $0<\gamma<1$,
\begin{equation}\label{eq:rateintro}
 A(n)=Cn^\alpha\bigl(1+O_\gamma((\log n)^{-\gamma})\bigr).
\end{equation}
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:arithmetic}
For every fixed $0<\gamma<1$, uniformly as $m\to\infty$,
\begin{equation}\label{eq:Bintro}
 B(m)=C\sigma_\alpha(m)
 \bigl(1+O_\gamma((\log m)^{-\gamma})\bigr).
\end{equation}
Let $(V_p)_p$ be independent random variables indexed by the primes,
with
\[
 \mathbb P(V_p=j)=(1-p^{-1})p^{-j}\qquad(j=0,1,\ldots),
 \qquad Z=\prod_p\left(\sum_{j=0}^{V_p}p^{-j\alpha}\right).
\]
This product is finite almost surely, with $\mathbb E Z=\zeta(\varphi)$.
For every bounded continuous $h:[0,\infty)\to\mathbb R$,
\begin{align*}
 \lim_{M\to\infty}\frac1M\sum_{m\le M}h(R_m)
 &=\mathbb E[h(Z)],\\
 \lim_{N\to\infty}\frac1N\sum_{n\le N}h(R_{A(n)})
 &=\frac{\mathbb E[Zh(Z)]}{\zeta(\varphi)}.
\end{align*}
Thus uniform sampling of block labels gives the Euler-product law,
whereas uniform sampling of sequence positions gives its size-biased law.

<!-- proposition-1: body_md -->
\label{prop:construction}
There is exactly one sequence satisfying
\eqref{eq:frequency}--\eqref{eq:inverse}. Its initial two blocks are
$1$ and $2,2,2$. For every $m\ge3$, the values $A(d)$ with $d\mid m$
are already known before block $m$ is appended.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Set $A(1)=1$. The first block has length $B(1)=1$. The next position
must therefore carry the value $2$, which fixes $A(2)=2$ and gives
$B(2)=A(1)+A(2)=3$. Hence $F(2)=4$.

Suppose the blocks up to $m-1$ have been constructed. Every block
$j\ge2$ has length at least $A(j)\ge2$, so
\begin{equation}\label{eq:causality}
 F(m-1)\ge1+2(m-2)=2m-3\ge m\qquad(m\ge3).
\end{equation}
Thus position $m$, and all its divisor positions, are already defined.
The finite positive integer $B(m)=\sum_{d\mid m}A(d)$ can be calculated,
and appending exactly $B(m)$ copies of $m$ extends the sequence without
altering any previous term. The resulting endpoints tend to infinity,
so every position is eventually assigned. Each block is nonempty and
finite. This proves existence, slow growth, and the occurrence of every
positive integer.

In any solution the initial two blocks are forced. The same argument
then forces every subsequent block, proving uniqueness.
<!-- end proof-1 -->
<!-- proposition-2: body_md -->
\label{prop:primecode}
If $m\mid n$ and $m<n$, then $B(m)<B(n)$. Moreover, for every $m\ge2$,
\begin{equation}\label{eq:primecode}
 m\text{ is prime}\quad\Longleftrightarrow\quad B(m)=A(m)+1.
\end{equation}
If $m$ is composite, then $B(m)-A(m)\ge3$; in particular, the difference
$B(m)-A(m)$ is never $2$.
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
When $m$ properly divides $n$, every divisor of $m$ is a divisor of $n$,
and $n$ itself occurs only in the latter divisor sum. Hence
\[
 B(n)-B(m)=\sum_{\substack{d\mid n\\d\nmid m}}A(d)>0.
\]
For $m\ge2$,
\[
 B(m)-A(m)-1=
 \sum_{\substack{d\mid m\\1<d<m}}A(d).
\]
The sum is empty exactly when $m$ is prime. If it is nonempty, each of
its terms is at least $2$, proving the stronger composite bound.

<!-- definition-1: body_md -->
Let $w:\mathbb N\to\mathbb Z_{\ge0}$ satisfy $w(1)=1$. A
\emph{$w$-weighted self-describing sequence} has multiplicities
\begin{equation}\label{eq:kernel}
 B_w(m)=\sum_{d\mid m}w(m/d)A_w(d),\qquad
 F_w(m)=\sum_{j\le m}B_w(j),
\end{equation}
and is the increasing-block inverse of $F_w$ as in
\eqref{eq:inverse}. Write
\begin{equation}\label{eq:W}
 W(s)=\sum_{q\ge1}\frac{w(q)}{q^s}
\end{equation}
where the series converges.
<!-- end definition-1 -->
<!-- theorem-3: body_md -->
\label{thm:kernel}
For every such $w$, the weighted sequence exists uniquely. If
$W(\varphi)<\infty$, then
\begin{equation}\label{eq:kernelasymptotic}
 A_w(n)\sim C_w n^\alpha,\qquad
 C_w=\left(\frac{\varphi}{W(\varphi)}\right)^{\alpha^2},
\end{equation}
and
\begin{equation}\label{eq:Fkernel}
 F_w(m)\sim \frac{C_w W(\varphi)}{\varphi}m^\varphi.
\end{equation}
Neither monotonicity nor boundedness of the multiplicities is assumed.
<!-- end theorem-3 -->
<!-- lemma-1: body_md -->
\label{lem:envelope}
If $W(\varphi)<\infty$, there is $M\ge2$, depending on $w$, such that
\begin{equation}\label{eq:envelope}
 M^{-1}n^\alpha\le A_w(n)\le Mn^\alpha
 \qquad(n\ge1).
\end{equation}
<!-- end lemma-1 -->
<!-- proof-3: body_md -->
Choose $M$ large enough that
\begin{equation}\label{eq:Mchoice}
 M^{1-\alpha}\ge W(\varphi)^\alpha,
 \qquad 1+(M\varphi)^\alpha\le M.
\end{equation}
The first condition is exactly
$(MW(\varphi))^{-\alpha}\ge M^{-1}$, which will close the lower induction
bound; the second closes the upper bound. Both hold for all sufficiently
large $M$ because $\alpha<1$. The initial cases $n=1,2$
are immediate. Suppose the bounds hold at every index below $n\ge3$,
and let $m=A_w(n)$. By \eqref{eq:strict}, all values $A_w(d)$ with
$d\le m$ are covered by the induction hypothesis. Hence
\[
 n\le F_w(m)\le M T_w(m)\le M W(\varphi)m^\varphi,
\]
which yields
\[
 m\ge(MW(\varphi))^{-\alpha}n^\alpha\ge M^{-1}n^\alpha.
\]
For the upper bound use only the $q=1$ contribution:
\[
 n>F_w(m-1)\ge M^{-1}\sum_{d\le m-1}d^\alpha
 \ge\frac{(m-1)^\varphi}{M\varphi}.
\]
Consequently,
\[
 m<1+(M\varphi n)^\alpha
 \le\bigl(1+(M\varphi)^\alpha\bigr)n^\alpha\le Mn^\alpha.
\]
This completes a strong induction on positions, rather than a formal
induction on the still-unknown growth law.
<!-- end proof-3 -->
<!-- proof-4: body_md -->
Existence has been established. Define
\[
 \ell=\liminf_{n\to\infty}\frac{A_w(n)}{n^\alpha},
 \qquad
 L=\limsup_{n\to\infty}\frac{A_w(n)}{n^\alpha}.
\]
Lemma~\ref{lem:envelope} gives $0<\ell\le L<\infty$.
For any $\varepsilon>0$, all sufficiently large $d$ satisfy
\[
 (\ell-\varepsilon)d^\alpha\le A_w(d)
 \le(L+\varepsilon)d^\alpha.
\]
For every fixed $D$, equation \eqref{eq:wsmall} gives explicitly
\[
 \sum_{d\le D}A_w(d)\sum_{q\le m/d}w(q)=o(m^\varphi).
\]
The same estimate holds with $A_w(d)$ replaced by $d^\alpha$.
Thus omitting these finitely many divisor positions changes both
the clock and its comparison sum by $o(m^\varphi)$.
Equation \eqref{eq:Tlimit} now implies
\begin{equation}\label{eq:Fenvelopes}
 \ell K_w\le\liminf_m\frac{F_w(m)}{m^\varphi}
 \le\limsup_m\frac{F_w(m)}{m^\varphi}\le LK_w.
\end{equation}
To make the inversion explicit, fix $0<\varepsilon<\ell K_w$.
For all sufficiently large $n$, with $m=A_w(n)$,
\[
 (\ell K_w-\varepsilon)(m-1)^\varphi<n
 \le(LK_w+\varepsilon)m^\varphi.
\]
The threshold is uniform because the power envelope gives $m\to\infty$.
Hence
\[
 (LK_w+\varepsilon)^{-\alpha}
 \le\frac{A_w(n)}{n^\alpha}
 <(\ell K_w-\varepsilon)^{-\alpha}+n^{-\alpha}.
\]
Taking lower and upper limits, then letting $\varepsilon\downarrow0$, gives
\begin{equation}\label{eq:contraction}
 \ell\ge(LK_w)^{-\alpha},\qquad
 L\le(\ell K_w)^{-\alpha}.
\end{equation}
Dividing the second bound by the first shows
\[
 1\le\frac L\ell\le\left(\frac L\ell\right)^\alpha.
\]
Because $0<\alpha<1$, this forces $L=\ell=c$. Both inequalities
in \eqref{eq:contraction} then force
$c=(cK_w)^{-\alpha}$. Solving gives
\[
 c=K_w^{-\alpha/(1+\alpha)}=K_w^{-\alpha^2}=C_w.
\]
Finally, substituting the resulting limit into
\eqref{eq:Fenvelopes} proves \eqref{eq:Fkernel}.

$c=(\varphi/(cW(\varphi)))^\alpha$, not
$c=(\varphi/W(\varphi))^\alpha$. The unknown leading coefficient occurs
on both sides. Its correct exponent is
$\alpha/(1+\alpha)=\alpha^2=2-\varphi$.

\label{lem:Terror}
As $x\to\infty$,
\begin{equation}\label{eq:Terror}
 T_1(x)=Kx^\varphi+O(x).
\end{equation}
<!-- end lemma-2 -->
<!-- proof-5: body_md -->
The estimate
$\sum_{d\le y}d^\alpha=y^\varphi/\varphi+O(y^\alpha)$, valid for $y\ge1$, gives

 T_1(x)=\frac{x^\varphi}{\varphi}\sum_{q\le x}q^{-\varphi}
       +O\left(x^\alpha\sum_{q\le x}q^{-\alpha}\right).

The error is $O(x)$ because $\alpha<1$. Extending the first sum to

<!-- lemma-3: body_md -->
\label{lem:clock}
For every fixed $0<\theta<1$ there are constants $D_\theta,m_\theta>0$
such that, for every integer $m\ge m_\theta$,
\begin{equation}\label{eq:clockerror}
 \left|\frac{F(m)}{CKm^\varphi}-1\right|
 \le E(m^\theta)+D_\theta m^{-\alpha(1-\theta)}.
\end{equation}
<!-- end lemma-3 -->
<!-- proof-6: body_md -->
Split the error in the exact identity
\[
 F(m)=C T_1(m)+C\sum_{d\le m}d^\alpha u(d)\lfloor m/d\rfloor
\]
at $d=m^\theta$. Boundedness of $u$ gives, for the small divisors,
\[
 \left|\sum_{d\le m^\theta}d^\alpha u(d)\lfloor m/d\rfloor\right|
 \ll m\sum_{d\le m^\theta}d^{\alpha-1}
 \ll m^{1+\alpha\theta}.
\]
The remaining sum has absolute value at most $E(m^\theta)T_1(m)$.
After division by $Km^\varphi$, Lemma~\ref{lem:Terror} proves the result.
The extra factor $1+O(m^{-\alpha})$ multiplying $E(m^\theta)$ is
absorbed into the remainder because $E$ is bounded.

<!-- lemma-4: body_md -->
\label{lem:logcontraction}
Suppose $E:[1,\infty)\to[0,\infty)$ is bounded and, for all sufficiently
large $x$,
\[
 E(x)\le qE(x^b)+Dx^{-\eta},
 \qquad 0<b,q<1,\quad D,\eta>0.
\]
If $\gamma>0$ and $qb^{-\gamma}<1$, then
$E(x)=O((\log x)^{-\gamma})$.
<!-- end lemma-4 -->
<!-- proof-7: body_md -->
Set $J(t)=t^\gamma E(e^t)$ and $\rho=qb^{-\gamma}<1$. Above a fixed
threshold $T>0$,
\[
 J(t)\le\rho J(bt)+D t^\gamma e^{-\eta t}.
\]
Choose $k$ so that $b^kt\in[bT,T]$ and $b^jt>T$ for $j<k$.
Iterating gives
\[
 J(t)\le \rho^kJ(b^kt)
 +D\sum_{j=0}^{k-1}\rho^j(b^jt)^\gamma e^{-\eta b^jt}.
\]
Boundedness of $E$ controls $J$ on $[bT,T]$, while
$\sup_{s\ge bT}s^\gamma e^{-\eta s}<\infty$ bounds each factor in the
sum apart from $\rho^j$. The geometric sum is at most $(1-\rho)^{-1}$,
uniformly in $t$ and $k$. Thus $J$ is bounded.

<!-- theorem-4: body_md -->
\label{thm:rate}
For every fixed $0<\gamma<1$,
\[
 \frac{A(n)}{Cn^\alpha}-1=O_\gamma((\log n)^{-\gamma}).
\]
<!-- end theorem-4 -->
<!-- proof-8: body_md -->
Let $m=A(n)$. The power envelope gives
\[
 B(m)\ll\sigma_\alpha(m)
 =m^\alpha\sum_{q\mid m}q^{-\alpha}
 \le m^\alpha\sum_{q\le m}q^{-\alpha}\ll m.
\]
Since $0\le F(m)-n<B(m)$, replacing $F(m)$ by $n$ in
Lemma~\ref{lem:clock} changes the normalized error by $O(m^{-\alpha})$.
We may therefore write $n=CKm^\varphi(1+v_n)$, where
\[
 |v_n|\le E(m^\theta)+O_\theta(m^{-\alpha(1-\theta)}).
\]
In particular $v_n\to0$. The normalization $C(CK)^\alpha=1$ gives
$u(n)=(1+v_n)^{-\alpha}-1$. For any fixed $q>\alpha$, the derivative of
$(1+v)^{-\alpha}$ is bounded in absolute value by $q$ on a sufficiently
small neighborhood of zero. Uniformly for all sufficiently large $n$,
\begin{equation}\label{eq:rawcontraction}
 |u(n)|\le qE(m^\theta)+O_{\theta,q}(m^{-\alpha(1-\theta)}).
\end{equation}

Fix $\gamma\in(0,1)$. Choose $\theta<1$ close enough to $1$,
$b<\alpha\theta$ close enough to $\alpha$, and $q>\alpha$ close enough
to $\alpha$, with $0<b,q<1$, so that
\begin{equation}\label{eq:qb}
 qb^{-\gamma}<1.
\end{equation}
Such choices exist because the limiting value is
$\alpha^{1-\gamma}<1$. The power envelope implies $m\asymp n^\alpha$,
hence eventually $m^\theta\ge n^b$. Using monotonicity of $E$ and taking
the supremum over integers $n\ge\lceil x\rceil$ in
\eqref{eq:rawcontraction} yields
\begin{equation}\label{eq:tailcontraction}
 E(x)\le qE(x^b)+Dx^{-\eta},\qquad
 \eta=\alpha^2(1-\theta)>0,
\end{equation}
for all sufficiently large real $x$. Here $D$ and the threshold
depend only on the chosen parameters and the fixed sequence.
Lemma~\ref{lem:logcontraction} proves the claim.

<!-- proposition-3: body_md -->
\label{prop:uniform}
For every $0<\gamma<1$,
\begin{equation}\label{eq:uniform}
 B(m)=C\sigma_\alpha(m)
 \left(1+O_\gamma((\log m)^{-\gamma})\right).
\end{equation}
In particular, if
\begin{equation}\label{eq:RZ}
 R_m=\frac{B(m)}{Cm^\alpha},\qquad
 Z_m=\sigma_{-\alpha}(m)=\sum_{d\mid m}d^{-\alpha},
\end{equation}
then $R_m/Z_m\to1$.
<!-- end proposition-3 -->
<!-- proof-9: body_md -->
The qualitative assertion follows by fixing $D$, applying
$A(d)=Cd^\alpha(1+o(1))$ uniformly for $d>D$, and observing that
the error from the finitely many $d\le D$ is $O_D(1)$, whereas
$\sigma_\alpha(m)\ge m^\alpha$.

For the stated rate choose $\delta=\alpha/(2\varphi)>0$. Divisors
$d\le m^\delta$ contribute an error at most
\[
 O\left(\sum_{d\le m^\delta}d^\alpha\right)=O(m^{\alpha/2}),
\]
using the power envelope. On the remaining divisors,
Theorem~\ref{thm:rate} bounds the relative summand error by
$O_\gamma((\log m)^{-\gamma})$. Divide their sum by
$C\sigma_\alpha(m)$, and use
$\sigma_\alpha(m)=m^\alpha\sigma_{-\alpha}(m)$. The first part then
contributes $O(m^{-\alpha/2})=o((\log m)^{-\gamma})$.

<!-- corollary-1: body_md -->
\label{cor:primes}
For a fixed prime $p$ and a fixed positive integer $q$,
\begin{align}
 \lim_{\substack{r\to\infty\\r\ \mathrm{prime}}}R_r&=1,
       \label{eq:prime}\\
 \lim_{j\to\infty}R_{p^j}&=\frac1{1-p^{-\alpha}},
       \label{eq:primepower}\\
 \lim_{\substack{r\to\infty\\r\ \mathrm{prime},\ r\nmid q}}R_{qr}
       &=\sigma_{-\alpha}(q).\label{eq:fixedpart}
\end{align}
Along primorials $P(y)=\prod_{p\le y}p$, one has
\begin{equation}\label{eq:primorial}
 R_{P(y)}\sim\prod_{p\le y}(1+p^{-\alpha})\longrightarrow\infty.
\end{equation}
Consequently \eqref{eq:liminfintro} holds.
<!-- end corollary-1 -->
<!-- proof-10: body_md -->
Apply Proposition~\ref{prop:uniform} and the finite Euler product
for $\sigma_{-\alpha}$. For example,
$Z_{qr}=\sigma_{-\alpha}(q)(1+r^{-\alpha})$ when $r\nmid q$.
The primorial product diverges already by comparison with
$\prod_{p\le y}(1+1/p)$. For completeness, the latter is
\[
 \prod_{p\le y}(1-p^{-2})\prod_{p\le y}(1-p^{-1})^{-1}.

<!-- theorem-6: body_md -->
\label{thm:distribution}
The product \eqref{eq:randomZ} converges to a finite random variable
almost surely. For every bounded continuous $h:[0,\infty)\to\mathbb R$,
\begin{equation}\label{eq:weak}
 \lim_{M\to\infty}\frac1M\sum_{m\le M}h(R_m)=\mathbb E[h(Z)].
\end{equation}
Equivalently, the empirical distributions converge weakly to the law
of $Z$. All positive integer moments converge. For an integer $r\ge1$,
\begin{equation}\label{eq:allmoments}
 \lim_{M\to\infty}\frac1M\sum_{m\le M}R_m^r
 =\mathbb E[Z^r]
 =\prod_p\left[(1-p^{-1})
   \sum_{j=0}^{\infty}\frac{(1+p^{-\alpha}+\cdots+p^{-j\alpha})^r}{p^j}\right].
\end{equation}
In particular,
\begin{align}
 \mathbb E Z&=\zeta(\varphi),\label{eq:mean}\\
 \mathbb E Z^2&=\frac{\zeta(\varphi)^2\zeta(2\varphi-1)}{\zeta(2\varphi)},
       \label{eq:second}\\
 \operatorname{Var} Z&=\zeta(\varphi)^2
 \left(\frac{\zeta(2\varphi-1)}{\zeta(2\varphi)}-1\right)>0.
       \label{eq:variance}
\end{align}
Here $2\varphi-1=\sqrt5$.

Realize the independent $V_p$ on their countable product probability
space and, for every $d=\prod p^{v_p(d)}$, set
$I_d=\mathbf 1_{\{V_p\ge v_p(d)\ \text{for all }p\}}$. Then
$\mathbb E I_d=1/d$. Monotone convergence gives
\[
 \mathbb E\left[\sum_{d\ge1}d^{-\alpha}I_d\right]
 =\sum_{d\ge1}d^{-1-\alpha}=\zeta(\varphi)<\infty.
\]
The nonnegative sum is therefore finite almost surely. Expanding
finite-prime products and passing monotonically to the limit shows
that it equals $Z$ in \eqref{eq:randomZ}.

For a finite cutoff $D$, the arithmetic approximation
\[
 Z_{m,D}=\sum_{\substack{d\le D\\d\mid m}}d^{-\alpha}
\]
is periodic. Uniform residues modulo $\operatorname{lcm}(1,\ldots,D)$, and the
Chinese remainder theorem, identify its limiting distribution with
that of $Z_D=\sum_{d\le D}d^{-\alpha}I_d$. Its arithmetic tail is
uniformly small in mean:
\begin{equation}\label{eq:tailZ}
 \frac1M\sum_{m\le M}(Z_m-Z_{m,D})
 =\sum_{D<d\le M}d^{-\alpha}\frac{\lfloor M/d\rfloor}{M}
 \le\sum_{d>D}d^{-1-\alpha}.
\end{equation}
The probabilistic tail has the same upper bound. First testing
bounded Lipschitz functions and then letting $D\to\infty$ proves
weak convergence for $Z_m$. Also
$M^{-1}\sum_{m\le M}Z_m\le\zeta(\varphi)$. Since $R_m/Z_m\to1$, it
follows that
\begin{equation}\label{eq:RZLone}
 \frac1M\sum_{m\le M}|R_m-Z_m|\longrightarrow0.
\end{equation}
To justify the transfer for every bounded continuous test function,
let $J_M$ be uniform on $\{1,\ldots,M\}$. For every $\varepsilon>0$,
Markov's inequality gives
\[
 \mathbb P(|R_{J_M}-Z_{J_M}|>\varepsilon)
 \le\frac{1}{\varepsilon M}\sum_{m\le M}|R_m-Z_m|
 \longrightarrow0.
\]
The already established convergence $Z_{J_M}\Rightarrow Z$ gives
tightness. Together with this estimate it also gives tightness of
$R_{J_M}$. On a fixed compact interval, a continuous test function is
uniformly continuous, so the difference of its expectations tends to
zero after the tails have been made arbitrarily small. Hence
$R_{J_M}\Rightarrow Z$, without requiring uniform continuity on
the entire half-line.

For integer moments expand the divisor sums. The natural dominating
series is
\begin{equation}\label{eq:lcmseries}
 \sum_{d_1,\ldots,d_r\ge1}
 \frac{(d_1\cdots d_r)^{-\alpha}}{\operatorname{lcm}(d_1,\ldots,d_r)}.
\end{equation}
It equals the Euler product in \eqref{eq:allmoments}. Indeed, for
each prime the factor is the expectation of
$(\sum_{j=0}^{V_p}p^{-j\alpha})^r$. This factor is
$1+O_r(p^{-1-\alpha})$, since $\mathbb P(V_p\ge1)=p^{-1}$ and
$\sum_{j=0}^{V_p}p^{-j\alpha}=1+O(p^{-\alpha})$ uniformly in
$V_p\ge1$ for large $p$. Hence the product and
\eqref{eq:lcmseries} converge.

The arithmetic $r$th moment is the same nonnegative sum with
$1/\operatorname{lcm}(d_1,\ldots,d_r)$ replaced by
$\lfloor M/\operatorname{lcm}(d_1,\ldots,d_r)\rfloor/M$. Dominated convergence
proves moment convergence for $Z_m$, uniformly bounded in $M$.
Write $R_m=Z_m(1+\varepsilon_m)$, where $\varepsilon_m\to0$.
For $m$ large enough that $|\varepsilon_m|\le1/2$, the mean value theorem
gives
\[
 |R_m^r-Z_m^r|
 \le c_r|\varepsilon_m|Z_m^r,\qquad
 c_r=r(3/2)^{r-1}.
\]
Let $K_r=\sup_{M\ge1}M^{-1}\sum_{m\le M}Z_m^r<\infty$.
For every sufficiently large fixed $D$,
\[
 \frac1M\sum_{m\le M}|R_m^r-Z_m^r|
 \le O_{D,r}(M^{-1})+
 c_rK_r\sup_{m>D}|\varepsilon_m|.
\]
Taking $M\to\infty$ and then $D\to\infty$ transfers the moment limit
to $R_m$.

For $r=2$, write $d=ga$, $e=gb$, $(a,b)=1$. Formula
\eqref{eq:lcmseries} becomes
\[
 \sum_{g\ge1}g^{-2\alpha-1}
 \sum_{(a,b)=1}a^{-\alpha-1}b^{-\alpha-1}
 =\zeta(2\alpha+1)\frac{\zeta(\alpha+1)^2}{\zeta(2\alpha+2)}.
\]
This proves \eqref{eq:second}; \eqref{eq:mean} was already obtained.
The variance is positive since $2\varphi-1<2\varphi$ and $\zeta(s)$ is
strictly decreasing for real $s>1$.

<!-- proposition-4: body_md -->
\label{prop:fullsupport}
The topological support of the limiting law is exactly $[1,\infty)$.
In fact, $Z>1$ almost surely, although $1$ remains in the support.

For each prime put $a_p=\log(1+p^{-\alpha})$. Then $a_p\to0$ and

elementary greedy subsum
argument therefore shows that the finite subsums of the $a_p$ are dense

product obtained by prescribing $V_p\in\{0,1\}$ at finitely many primes,
an event of positive probability.

 \le2p^{-\alpha}\mathbf 1_{\{V_p\ge1\}}.

<!-- theorem-7: body_md -->
\label{thm:singular}
The law of $Z$ is singular continuous: it has no atoms and is
concentrated on a set of Lebesgue measure zero.
<!-- end theorem-7 -->
<!-- proof-14: body_md -->
Set
\[
 X_p=\log\left(\sum_{j=0}^{V_p}p^{-j\alpha}\right),\qquad
 Y=\log Z=\sum_pX_p.
\]
By Theorem~\ref{thm:distribution}, this sum of independent discrete
random variables converges almost surely. The Jessen--Wintner
pure-type theorem \cite[Theorem~35]{jessenwintner} therefore makes
its law either discrete, singular continuous, or absolutely continuous;
the additive-function formulation is also stated in
\cite[\S13(iv), p.~720]{erdoswintner}.

The values of $X_p$ strictly increase with $V_p$. Its largest atom is
therefore the one at zero, of mass
\[
 d_p=\sup_{x\in\mathbb R}\mathbb P(X_p=x)=\mathbb P(V_p=0)=1-\frac1p.
\]
L\'evy's continuity criterion for a convergent infinite convolution of
discrete laws \cite[Th\'eor\`eme~XIII]{levy}, as reproduced in
\cite[\S13(v), p.~721]{erdoswintner}, excludes atoms precisely when
the product of their largest atom masses is zero. Here
\[
 \prod_p d_p=0,\qquad\text{since}\qquad
 \sum_p(1-d_p)=\sum_p\frac1p=\infty.
\]
Thus the law of $Y$ has no atoms.

To exclude absolute continuity, we adapt the covering argument of
Erd\H{o}s \cite[\S3]{erdossmoothness}, retaining the full prime-power
model. Put $f(n)=\log\sigma_{-\alpha}(n)$ and, for $y\ge2$, let
\[
 M_y=\prod_{p\le y}p^{V_p},\qquad
 Y_y=\log\sigma_{-\alpha}(M_y),\qquad
 c_y=\prod_{p\le y}(1-p^{-1}).
\]
For every integer $m$ whose prime factors are at most $y$,
independence gives $\mathbb P(M_y=m)=c_y/m$ exactly. Every
$m\le y^{\alpha/4}$ has this property. Mertens' product formula
\cite{mertens} and the harmonic-sum asymptotic therefore give
\[
 \mathbb P(M_y\le y^{\alpha/4})
 =c_y\sum_{m\le y^{\alpha/4}}\frac1m
 \longrightarrow e^{-\gamma_E}\frac{\alpha}{4}>0,
\]
where $\gamma_E$ is Euler's constant. On the other hand,
\[
 0\le\mathbb E(Y-Y_y)\ll\sum_{p>y}p^{-1-\alpha}\ll y^{-\alpha}.
\]
Markov's inequality shows that $Y-Y_y\le y^{-\alpha/2}$ except on
an event of probability $O(y^{-\alpha/2})$. Consequently the set
\[
 H_y=\bigcup_{m\le y^{\alpha/4}}
 [\,f(m),\,f(m)+y^{-\alpha/2}\,]
\]
has Lebesgue measure at most $y^{-\alpha/4}\to0$, whereas
$\liminf_{y\to\infty}\mathbb P(Y\in H_y)>0$. This contradicts absolute
continuity, whose integrable density would give uniformly vanishing
mass to sets of vanishing Lebesgue measure. The pure-type theorem
therefore makes the law of $Y$ singular continuous.
Exponentiation preserves atoms and maps null sets to null sets on
each bounded interval, so the same conclusion holds for $Z$.

\label{thm:residues}
Fix $q\ge2$ and a residue class $a\pmod q$. Then
\begin{equation}\label{eq:residuedensitydef}
 \delta_{q,a}:=\lim_{N\to\infty}\frac1N
 \#\{n\le N:A(n)\equiv a\pmod q\}
\end{equation}
exists and equals
\begin{equation}\label{eq:residuedensity}
 \delta_{q,a}=\frac1q\sum_{g\mid\gcd(a,q)}g^{1-\varphi}
 \prod_{p\mid q/g}(1-p^{-\varphi}),
\end{equation}

totient function, then
\begin{equation}\label{eq:divisibilitydensity}
 \operatorname{dens}\{n:q\mid A(n)\}
 =\delta_{q,0}=\frac1q\sum_{d\mid q}\frac{\phi_{\!E}(d)}{d^\varphi}.
\end{equation}

divisibility of $A(n)$ by fixed coprime moduli is asymptotically
independent. For example,

 \operatorname{dens}\{n:A(n)\text{ is even}\}
 &=\frac{1+2^{-\varphi}}2=0.6628895560\ldots,\label{eq:evendensity}\\
 \operatorname{dens}\{n:A(n)\text{ is odd}\}
 &=\frac{1-2^{-\varphi}}2=0.3371104439\ldots.\label{eq:odddensity}
\end{align}

$q^{-1}\prod_{p\mid q}(1-p^{-\varphi})$.

More explicitly, put
\[
 S_D=\sum_{d,e>D}\frac{d^{-\alpha}e^{-\alpha}}{\operatorname{lcm}(d,e)}
 \longrightarrow0.
\]
The unshifted mean square tail is at most $S_D$ for every $M$, and
the shifted one is at most $(M+h)S_D/M$, by enlarging the summation
interval to $1\le m\le M+h$. Cauchy--Schwarz therefore permits passage
to the untruncated $Z_m$.
For the final transfer write $R_m=Z_m(1+\varepsilon_m)$, with
$\varepsilon_m\to0$. Then
\[
 |R_mR_{m+h}-Z_mZ_{m+h}|
 \le\bigl(|\varepsilon_m|+|\varepsilon_{m+h}|
       +|\varepsilon_m\varepsilon_{m+h}|\bigr)Z_mZ_{m+h}.
\]
If $K_2=\sup_{L\ge1}L^{-1}\sum_{m\le L}Z_m^2<\infty$, Cauchy--Schwarz
gives
\[
 \frac1M\sum_{m\le M}Z_mZ_{m+h}
 \le K_2\sqrt{1+h/M}.
\]
Put $e_D=\sup_{m>D}|\varepsilon_m|$. Splitting off the first $D$ terms,
the averaged product difference is at most
$O_{D,h}(M^{-1})+(2e_D+e_D^2)K_2\sqrt{1+h/M}$.
Taking $M\to\infty$ and then $D\to\infty$ proves the transfer to $R_m$.

For arbitrary $h$, write $d=ga$, $e=gb$, with $(a,b)=1$.
The compatibility condition becomes $g\mid h$, and
$\operatorname{lcm}(d,e)=gab$. Thus
\[
 \mathcal C(h)=
 \sum_{g\mid h}g^{-2\alpha-1}\sum_{(a,b)=1}(ab)^{-\varphi}.
\]
The inner sum has Euler product
\[
 \sum_{(d,e)=1}(de)^{-\varphi}
 =\prod_p\frac{1+p^{-\varphi}}{1-p^{-\varphi}}
 =\frac{\zeta(\varphi)^2}{\zeta(2\varphi)}.
\]
This proves \eqref{eq:closedshift} and its specialization at $h=1$.
For fixed $h$, the shifted first and second Ces\`aro moments have the
same limits as the unshifted ones: the finitely many boundary terms are
$M^{o(1)}$ by Theorem~\ref{thm:maximal}, and their contribution after
division by $M$ vanishes.
Subtracting the square of the mean in \eqref{eq:mean}, then dividing by
the variance in \eqref{eq:variance}, proves
\eqref{eq:adjacentcovariance}--\eqref{eq:adjacentcorrelation}.

<!-- proposition-7: body_md -->
\label{prop:iterates}
For every fixed integer $k\ge1$ and every $0<\gamma<1$,
\begin{equation}\label{eq:iterateasymptotic}
 A^{\circ k}(n)=
 C^{(1-\alpha^k)/(1-\alpha)}n^{\alpha^k}
 \left(1+O_{k,\gamma}((\log n)^{-\gamma})\right).
\end{equation}
The only fixed points of $A$ are $1$ and $2$. For $n\ge2$, let
\begin{equation}\label{eq:hittingdef}
 h(n)=\min\{k\ge0:A^{\circ k}(n)=2\}.
\end{equation}
Then
\begin{equation}\label{eq:hittingtime}
 h(n)=\frac{\log\log n}{\log\varphi}+O(1).
\end{equation}
<!-- end proposition-7 -->
<!-- proof-21: body_md -->
Formula \eqref{eq:iterateasymptotic} follows by induction from
Theorem~\ref{thm:rate}. For fixed $j$,
$\log A^{\circ j}(n)\asymp\log n$, so the relative error remains
$O_{j,\gamma}((\log n)^{-\gamma})$, while the exponent of $C$ obeys
$s_{j+1}=1+\alpha s_j$ and hence
$s_j=1+\alpha+\cdots+\alpha^{j-1}$.

For the hitting time, the power envelope gives a constant $M\ge2$ such
that
\[
 M^{-1}x^\alpha\le A(x)\le Mx^\alpha\qquad(x\ge1).
\]
If $x_j=A^{\circ j}(n)$, iteration of the logarithmic inequalities gives
\begin{equation}\label{eq:iteratelogenvelope}
 \left|\log x_j-\alpha^j\log n\right|
 \le\frac{\log M}{1-\alpha}.
\end{equation}
Since $A(x)<x$ for $x\ge3$ and $A(x)\ge2$ for $x\ge2$, every orbit
starting at $n\ge2$ reaches $2$, and no integer greater than $2$ is fixed.
Putting $j=h(n)$ in \eqref{eq:iteratelogenvelope} shows that
$\alpha^j\log n=O(1)$, which gives the lower bound in
\eqref{eq:hittingtime}. Conversely, after
$\log\log n/\log\varphi+O(1)$ iterations the upper half of
\eqref{eq:iteratelogenvelope} places $x_j$ in a fixed finite set. Strict
descent then requires only a bounded number of further steps. This proves
the matching upper bound.

<!-- theorem-11: body_md -->
\label{thm:zetalayer}
The variables $Q_M$ converge in total variation to the zeta-distributed
integer $Q$ given by
\begin{equation}\label{eq:zetalayerlaw}
 \mathbb P(Q=q)=\frac{q^{-\varphi}}{\zeta(\varphi)}\qquad(q\ge1).
\end{equation}
Its tail and moments satisfy
\begin{align}
 \mathbb P(Q>y)&\sim\frac{y^{-\alpha}}{\alpha\zeta(\varphi)},
       \label{eq:zetalayertail}\\
 \mathbb E Q^t&=\frac{\zeta(\varphi-t)}{\zeta(\varphi)}\qquad(t<\alpha).
       \label{eq:zetalayermoments}
\end{align}
The moments of order $t\ge\alpha$, including the mean, are infinite.
For every $t<\alpha$, the corresponding moments of $Q_M$ converge to
\eqref{eq:zetalayermoments}.
<!-- end theorem-11 -->
<!-- proof-22: body_md -->
Exact counting and \eqref{eq:floor} give
\begin{equation}\label{eq:zetalayerfinite}
 \mathbb P(Q_M=q)=\frac{\sum_{d\le M/q}A(d)}{F(M)}.
\end{equation}
Since
\[
 \sum_{d\le x}A(d)\sim\frac C\varphi x^\varphi,\qquad
 F(M)\sim\frac{C\zeta(\varphi)}\varphi M^\varphi,
\]
the right side tends to $q^{-\varphi}/\zeta(\varphi)$ for every fixed $q$.
Both sides of the limiting assertion are probability mass functions, so
pointwise convergence implies total-variation convergence. The power
envelope also gives the uniform bound
$\mathbb P(Q_M=q)\ll q^{-\varphi}$. Dominated convergence therefore proves moment
convergence for $t<\varphi-1=\alpha$. Finally, summing the power-law mass in
\eqref{eq:zetalayerlaw} proves \eqref{eq:zetalayertail} and the remaining
moment assertions.

\label{prop:abelian}
For every $0<\gamma<1$, as $\varepsilon\downarrow0$,
\begin{align}
 \varepsilon D_A(\varphi+\varepsilon)
 &=C+O_\gamma(\varepsilon^\gamma),\label{eq:abelA}\\
 \varepsilon D_B(\varphi+\varepsilon)
 &=C\zeta(\varphi)+O_\gamma(\varepsilon^\gamma).
       \label{eq:abelB}
\end{align}
<!-- end proposition-8 -->
<!-- proof-23: body_md -->
Theorem~\ref{thm:rate} gives
\[
 D_A(\varphi+\varepsilon)
 =C\zeta(1+\varepsilon)
 +O_\gamma\left(1+\sum_{n\ge3}
 \frac{1}{n^{1+\varepsilon}(\log n)^\gamma}\right).
\]
The sum is $O_\gamma(\varepsilon^{\gamma-1})$, by the integral
substitution $t=\log x$. Since
$\varepsilon\zeta(1+\varepsilon)=1+O(\varepsilon)$, this proves
\eqref{eq:abelA}. Multiplication by
$\zeta(\varphi+\varepsilon)=\zeta(\varphi)+O(\varepsilon)$ proves
\eqref{eq:abelB}.
