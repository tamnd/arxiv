\label{thm:RogersIntegrationFormula} 
If $f_1, f_2 : \mathbb{R}^n \rightarrow \mathbb{R}$ are non-negative functions with compact support, and $n \geq 3$, then
\begin{align*}
    \int_{\mathsf{X}_n} 
    \sum_{\substack{z_1, z_2 \in \mathbb{Z}^n \\ z_1, z_2 \neq 0}}
    f_1 (z_1 g) f_2 (z_2 g) \, d g 
    &= 
    \widehat{f}_1 (0) \widehat{f}_2 (0) 
    + \frac{1}{\zeta(n)}\sum_{\substack{j\geq 1 \\ k \neq 0}} \int_{\mathbb{R}^{n}}f_1(jz)f_2(kz) \, dz.
\end{align*}

<!-- theorem-2: body_md -->
\label{thm:main}
Let $n \geq 2$. Let $\mathsf{Y}_{2n}$ be equipped with the right $\mathrm{Sp}(2n,\mathbb{R})$-invariant probability measure $\mu$, and let $f:(\mathbb{R}^{2n})^2\rightarrow\mathbb{R}$ be a Borel measurable function. Then
\begin{align*}
\int_{\mathsf{Y}_{2n}}\sum_{z_1,z_2\in\mathbb{Z}^{2n}\setminus \{0\}} f(z_1g,z_2g) \, d\mu 
= 
\frac{1}{\zeta(2n)} \sum_{\substack{j \geq 1 \\ k \in \mathbb{Z}}} 
\int_{\mathbb{R}^{2n}}\int_{L(k,z_1)}f(jz_1,z_2) \, d z_2 \, \frac{dz_1}{\|z_1\|} \\
+ \, \frac{1}{\zeta(2n)}\sum_{\substack{j \geq 1 \\ k \neq 0}}\int_{\mathbb{R}^{2n}}f(jz,kz) \, dz,
\end{align*}
where $J=\left( \begin{smallmatrix} & I_n \\ -I_n & \end{smallmatrix}\right)$ is the matrix for the symplectic form used to define $\mathrm{Sp}(2n)$, and 
\begin{align*}
L(k,z_1) 
= 
\mathrm{span}_\mathbb{R}(z_1J)^\perp +\frac{kz_1J}{\|z_1\|^2}
= 
\big\{ z_2 \in \mathbb{R}^{2n} : z_2 \cdot z_1 J = k \big\}.
\end{align*}

<!-- theorem-3: body_md -->
\label{thm:main-Schwartz}
Consider the special case where $f(z_1,z_2) = f_1(z_1)f_2(z_2)$ is the product of two functions $f_1, f_2 : \mathbb{R}^{2n} \rightarrow \mathbb{C}$ which are either
\begin{enumerate}
    \item in the Schwartz class, 
    \item compactly supported and Riemann integrable,

\begin{align*}
S_{x,y}:=\{(\tilde x_1,\tilde y_1,\tilde x_2, \tilde y_2)\subseteq\mathbb{F}_p^{2n}: (\tilde x_1,x_1;y_1,\tilde y_1)J(\tilde x_2,x_2;y_2,\tilde y_2) \equiv 0\ (\mathrm{mod}\ p)\}.

From the construction of $T_{x,y}$ it can be straightforwardly checked that $T_{x,y}$ indeed maps into $S_{x,y}$. We need show that it also surjects.

Let $(\tilde x_1,\tilde y_1,\tilde x_2,\tilde y_2) \in S_{x,y}$. The first and second rows of $\tilde D^\intercal $ can be chosen so that $-x_1\tilde D^\intercal =\tilde x_1$ and $-x_2\tilde D^\intercal =\tilde x_2$. (\tilde x_1,x_1;y_1,\tilde y_1)J(\tilde x_2,x_2;y_2,\tilde y_2) \equiv 0\ (\mathrm{mod}\ p)

<!-- proposition-4: body_md -->
\label{prop:bessel}
We have
\begin{align*}
\Delta(w) = 2\sum_{m=1}^\infty\left(\frac{w}{m}\right)^nJ_n(2\pi mw),
\end{align*}
where $J_\nu$ is the Bessel function of the first kind.
<!-- end proposition-4 -->
<!-- proof-6: body_md -->
Let $f_w(x) = V_{2n-1}(w^2-x^2)_+^{n-1/2}$. By the Poisson summation formula,
\begin{align*}
\sum_{k\in\mathbb{Z}}f_w(k)=\sum_{m\in\mathbb{Z}}\widehat{f}_w(m),
\end{align*}
so the desired statement will follow upon the computation of the Fourier coefficients $\widehat{f}_w(m)$ --- it will also turn out that $\widehat{f}_w$ decays quickly enough for the legitimate application of the Poisson summation formula. First notice
\begin{align*}
\widehat{f}_w(0)=\int V_{2n-1}(w^2-x^2)_+^{n-\frac{1}{2}}dx=V_{2n}w^{2n}.
\end{align*}

In general, since $f_w$ is even,
\begin{align*}
\widehat{f}_w(m) &= \int V_{2n-1}(w^2-x^2)_+^{n-\frac{1}{2}}e^{-2\pi i m x}dx\\
&=\int_{-w}^w V_{2n-1}(w^2-x^2)^{n-\frac{1}{2}}\cos(2\pi m x)dx,
\end{align*}
and $\widehat f_w(m)= \widehat f_w(-m)$. Hence we assume $m \geq 1$ for the rest of the proof.

Substituting $x=wt$, we have
\begin{align*}
\widehat f_w(m)=V_{2n-1}w^{2n}\int_{-1}^1 (1-t^2)^{n-\frac{1}{2}}\cos(2\pi mwt)dt.
\end{align*}
Recalling the Poisson integral for the Bessel function
\begin{align*}
\int_{-1}^1(1-t^2)^{\nu-\frac{1}{2}}\cos(zt)dt = \pi^\frac{1}{2}\Gamma\left(\nu+\frac{1}{2}\right)\left(\frac{2}{z}\right)^\nu J_\nu(z),
\end{align*}
the formula $V_{2n-1} = \pi^{n-1/2}/\Gamma(n+1/2)$, 
and taking $\nu=n, z=2\pi m w$,  we obtain
\begin{align*}
\widehat f_w(m)=\left(\frac{w}{m}\right)^nJ_n(2\pi mw).
\end{align*}
This completes the proof.

\label{thm:holm_improvement}
The average of \eqref{eq:holm_goal} over $\mathsf{Y}_{2n}$, as $N\rightarrow\infty$, converges to
\begin{align*}
2V+(\mathrm{error}(n)),
\end{align*}
where the absolute value of the error is no greater than
\begin{align*}
30V\frac{\zeta(\frac{3n}{2})\zeta(\frac{n}{2})}{\zeta(2n)}\frac{(2\pi)^\frac{n}{2}}{\Gamma(\frac{n}{2}+1)} + \frac{V}{n2^{2n}}.
\end{align*}

As a consequence, the variance $\sigma^2$ in \cite[Theorem 1.1]{Hol23} is $2+o_n(1)$.

<!-- proposition-5: body_md -->
\label{prop:2ndmmt_for_GH}
For $n \geq 3$ and $R>0$ satisfying $2\pi R^2 \geq 1$, \eqref{eq:2ndmmt_for_GH} equals
\begin{align*}
\mathrm{vol}(B)^2 + \mathrm{vol}(B)\left(\frac{2\zeta(2n-1)}{\zeta(2n)} + \mathrm{error}\right),
\end{align*}
where the error is $O(nR^{-\frac{8}{3}} + nR^{-2}2^{-n}).$
<!-- end proposition-5 -->
<!-- proof-7: body_md -->
Following Section \ref{subsect:application_discrepancy} and Proposition \ref{prop:bessel}, \eqref{eq:2ndmmt_for_GH} equals
\begin{align*}
\mathrm{vol}(B)^2 + \frac{2\zeta(2n-1)}{\zeta(2n)}\mathrm{vol}(B) + \frac{2S_{2n}}{\zeta(2n)}\sum_{j=1}^\infty\sum_{m=1}^\infty\int_0^{R^2/j} \left(\frac{w}{m}\right)^nJ_n(2\pi mw)\frac{dw}{w}.
\end{align*}
For the integral inside the double sum, we substitute $z=2\pi m w$ to rewrite it as
\begin{align}\label{eq:GH_1}
\frac{1}{(2\pi m^2)^n}\int_0^{y} z^{n-1}J_n(z)dz,
\end{align}
where $y=2\pi mR^2/j$.

We apply the formula \cite[10.22.10]{DLMF} with $\mu=n-1, \nu=n$. After a straightforward computation, \eqref{eq:GH_1} can be seen to be equal to
\begin{align*}
\frac{1}{(2\pi m^2)^n} y^{n-1}\sum_{k=0}^\infty(n+1+2k)\mathrm{B}(n,k+1)J_{n+1+2k}(y).
\end{align*}
where $\mathrm{B}$ is the beta function. By the result of \cite{Lan00} and Lemma \ref{lemma:funnysum} below, the above can be bounded in size by
\begin{align*}
\frac{3}{(2\pi m^2)^n}y^{n-1}\min(y^{-\frac{1}{3}},1).
\end{align*}

Recalling what $y$ was, and by seperating out the contribution from the $j=m=1$ term, provided that $2\pi R^2 \geq 1$ we have
\begin{align*}
\frac{2S_{2n}}{\zeta(2n)}\sum_{j,m=1}^\infty \frac{3}{(2\pi m^2)^n}y^{n-1}\min(y^{-\frac{1}{3}},1) &= O(nV_{2n}R^{2n-\frac{8}{3}} +nV_{2n}R^{2n-2}2^{-n}) \\
&=\mathrm{vol}(B)\cdot O(nR^{-\frac{8}{3}} + nR^{-2}2^{-n}).
\end{align*}
This completes the proof.

<!-- lemma-4: body_md -->
\label{lemma:funnysum}
For $n \geq 3$
$$\sum_{k=0}^\infty(n+1+2k)\mathrm{B}(n,k+1) = \frac{n}{n-2}.$$
For $n=2$ the sum diverges.
<!-- end lemma-4 -->
<!-- proof-8: body_md -->
First assume $n\geq 3$. Plugging in the integral formula for the beta function, we can rewrite the left-hand side as
\begin{align*}
\sum_{k=0}^\infty (n+1+2k)\int_0^1 t^{n-1}(1-t)^kdt.
\end{align*}
Changing the sum and the integral, we have
\begin{align*}
\int_0^1t^{n-1}\sum_{k=0}^\infty(n+1+2k)(1-t)^kdt.
\end{align*}
Recalling the facts
\begin{align*}
\sum_{k=0}^\infty x^k = \frac{1}{1-x} \mbox{ and } \sum_{k=0}^\infty kx^k = \frac{x}{(1-x)^2}
\end{align*}
for $|x|<1$, the above can be computed as
\begin{align*}
\int_0^1 t^{n-1}\left((n+1)t^{-1} +2(1-t)t^{-2}\right)dt 
&= \int_0^1 t^{n-1}\left((n-1)t^{-1} +2t^{-2}\right)dt \\
&= 1+\frac{2}{n-2} =\frac{n}{n-2}.
\end{align*}

In case $n=2$, the summand in question equals
\begin{align*}
\frac{3+2k}{(k+1)(k+2)},
\end{align*}
from which it is clear that the sum diverges.

<!-- theorem-6: body_md -->
\label{thm:symplectic_GH}
For $n$ sufficiently large, $\Lambda \in \mathsf{Y}_{2n}$ in a subset of measure at least $1-o_n(1)$ satisfies
\begin{align*}
1-\frac{\log\log n}{2n} \leq \left(\frac{V_{2n}}{2}\right)^{\frac{1}{2n}}\lambda_1(\Lambda) \leq 1+\frac{\log\log n}{2n}.
\end{align*}
<!-- end theorem-6 -->
<!-- proof-9: body_md -->
Let $\varepsilon(n) = (\log n)^{-1}$, and set $\mathrm{vol}(B) = 2\varepsilon(n)^{-1}$; note that this implies $\mathrm{error} = O(n^{-1/3})$, as defined in Proposition \ref{prop:2ndmmt_for_GH}.

Now by applying Proposition \ref{prop:2ndmmt_for_GH} to Proposition 27 of \cite{GSVV25}, we obtain the following statement. For $\Lambda\in \mathsf{Y}_{2n}$ in a subset of measure at least $1-o_n(1)$, we have
\begin{align*}
2\varepsilon(n) \leq V_{2n}\lambda_1(\Lambda)^{2n} \leq 2\varepsilon(n)^{-1}.
\end{align*}
By taking $2n$'th roots on all sides, and approximating $\varepsilon(n)^{\pm 1/2n}=\exp(\pm(\log\log n)/2n)$ by a Taylor expansion, the theorem follows.

<!-- theorem-7: body_md -->
\label{thm:lattice_smoothing_parameter}
    For any $n \geq 2$ and $s > 0$, the following statements hold:
    \begin{enumerate}
        \item The second moment over $\mathsf{Y}_{2n}$ of the (punctured) discrete Gaussian $\varrho_s$ equals
            \begin{align*}
                \int_{\mathsf{Y}_{2n}} \varrho_s (L \setminus \lbrace 0 \rbrace )^2 \, d \mu (L)
                =
                s^{2n} \Bigl( s^{2n} + \zeta(n) L(n, \chi) - \zeta(2n) + \tfrac12 \mathrm{Z}_s(n) \Bigr),
            \end{align*}
            where $\chi$ denotes the non-principal Dirichlet character (mod $4$), and $\mathrm{Z}_s$ denotes the (incomplete) Epstein zeta function
            \begin{align*}
                \mathrm{Z}_s (n) = 
                \sum_{\substack{x,y \in \mathbb{Z} \\ x,y \neq 0}}
                \frac{1}{(x^2 s^{-2} + y^2 s^2)^n}.
            \end{align*}
        \item Let $0 <\varepsilon <1$ be arbitrary, and let $\varepsilon^{-1/(2n)} < s < (\varepsilon/2)^{-1/n}$ be a fixed large number. Then, for any $n \geq 2$ there exists a subset $\Theta_{2n} \subseteq \mathsf{Y}_{2n}$ of measure $\mu(\Theta_{2n}) = 1 - O(2^{-n})$, such that all $L \in \Theta_{2n}$ satisfy $\varrho_{1/s}(L^P \setminus \lbrace 0 \rbrace) \leq \varepsilon$.\\[+0.5em] In other words, as $n \rightarrow \infty$, the smoothing parameter satisfies $\eta_\varepsilon (L) = 1 + O(1/n)$ on a set of measure $1-o(1)$. 
    \end{enumerate}
<!-- end theorem-7 -->
<!-- proof-10: body_md -->
Let $\varepsilon > 0$ and write $\Lambda' := \Lambda \setminus \lbrace 0 \rbrace$. To study $\eta_\varepsilon (\Lambda)$, we are required to study the discrete Gaussian $\varrho_{1/s}(\Lambda^P)$ on the dual lattice $\Lambda^P$ of $\Lambda$. However, for our metrical purpose, it is equivalent (and notationally simpler) to study $\varrho_{1/s}(\Lambda)$ instead, as the probability measure $\mu$ on $\mathsf{Y}_{2n}$ is invariant under the map $\Lambda \mapsto \Lambda^P$.  \\[+0.5em]
    For any $s > 0$, Siegel's mean value theorem over $\mathsf{Y}_{2n}$ gives $\mathbb{E} [\varrho_s (\Lambda')] = s^{2n}.$ To express the variance of $\varrho_s(\Lambda')$, we consider the function $A(2n) = \zeta(n) L(n, \chi) - \zeta(2n)$. (Note that the function $A(n)$ is the function $K(n)$ from \cite{PS26}.) By using Thm. \ref{thm:main-Schwartz} and a computation analogous to \cite{PS26}, we find that
    \begin{align}\label{discrete_Gaussian_variance}
        \mathrm{Var}(\varrho_s (\Lambda')) = 2 s^{2n} A(2n) + 
        \tfrac12 s^{2n} \mathrm{Z}_s(n),
    \end{align}
    where $\mathrm{Z}_s(n)$ arises from the new symplectic term in the second-moment formula. \par 
    We now replace $s$ by $1/s$ and use Chebyshev's inequality in combination with the trivial identity $\varrho_{s}(\Lambda) = 1 + \varrho_s (\Lambda')$ to obtain 
    \begin{align*}
    \mathbb{P} \Bigl( \varrho_{1/s}(\Lambda) > 1 + \varepsilon \Bigr)
    &=
    \mathbb{P} \Bigl( \varrho_{1/s}(\Lambda^\ast) - 1 -s^{-2n} > \varepsilon -s^{-2n}
    \Bigr)
    \\[0.5em] &\leq 
    \mathbb{P} \Bigl( \Big| \varrho_{1/s}(\Lambda') -s^{-2n} \Big| > 
    \varepsilon -s^{-2n} \Bigr)
    \\[0.5em] &\leq 
    \frac{\mathrm{Var}(\varrho_{1/s}(\Lambda '))}{(\varepsilon - s^{-2n})^2},
    \end{align*}
    where the final inequality holds if $\varepsilon - s^{-2n} > 0$.
    With \eqref{discrete_Gaussian_variance}, we arrive at the bound 
    \begin{align}\label{discrete_Gaussian_prob-bound}
        \mathbb{P} \Bigl( \varrho_{1/s}(\Lambda) > 1 + \varepsilon \Bigr)
        &\leq 
        \frac{2 A(2n) + 
        \tfrac{1}{2} \mathrm{Z}_{1/s}(n)}
        {s^{2n}(\varepsilon - s^{-2n})^2}.
    \end{align}
    \par Choose $s = s (\varepsilon)$ so that the denominator on the right-hand side of \eqref{discrete_Gaussian_prob-bound} is $2$; explicitly, this means
    \begin{align*}
    s(\varepsilon) = \Biggl( \frac{1 + \varepsilon + \sqrt{2 \varepsilon + 1}}{\varepsilon^2} \Biggr)^{\! \! 1/2n} < (4\varepsilon^{-2})^{1/(2n)} = (\varepsilon/2)^{-1/n} = 1 + O\bigl( \tfrac{1}{n} \bigr).
    \end{align*}
    Then in particular we have $0 < s(\varepsilon)^{-2n} < \varepsilon$, and 
    \begin{align*}
    \mathbb{P} \Bigl( \varrho_{1/s(\varepsilon)}(\Lambda^P) > 1 + \varepsilon \Bigr)
    =
    \mathbb{P} \Bigl( \varrho_{1/s(\varepsilon)}(\Lambda) > 1 + \varepsilon \Bigr)
    &\leq 
    A(2n) + \tfrac14 \mathrm{Z}_{1/s(\varepsilon)}(n).
    \end{align*}
    By definition of the smoothing parameter, we therefore have $\eta_\varepsilon (\Lambda) \leq s(\varepsilon)$ on a set $\Theta_{2n}$ of large measure, viz. 
    \begin{align}
        \mu (\Theta_{2n}) \geq 1 - A(2n) - \tfrac{1}{4} \mathrm{Z}_{1/s(\varepsilon)}(n).
    \end{align}
    It is relatively straightforward to prove the estimates
    \begin{align*}
    \mathrm{Z}_s(n) 
    \sim
    \frac{2}{(s^2+s^{-2})^n} \ll \frac{1}{2^n}
    \quad \text{and} \quad 
    A(2n) 
    \sim 
    \frac{1}{2^n}
    \qquad (\text{as } n \longrightarrow \infty).
    \end{align*}
    (The implied constant in the estimate for $\mathrm{Z}_s(n)$ is independent of $s$.) This proves that, for all $L$ in a large set, $\eta_\varepsilon(L)$ is at most $1 + O(1/n)$. \\[+0.5em]
    Finally, we show that $\eta_\varepsilon$ is also bounded from below: Observe that we have
    \begin{align*}
        \varrho_{1/s}(L^P) = s^{-2n} \varrho_s (L) \geq s^{-2n} \qquad (\text{for any } s > 0)
    \end{align*}
    by Poisson summation and the fact that $\varrho_s(0)$ is always $1$. This shows that
    \begin{align*}
        \Big\{ s > 0 \, \Big| \, \varrho_{1/s}(L^P) \leq 1+ \varepsilon \Big\}
        \subseteq 
        \Big\{ s > 0 \, \Big| \, s \geq (1+\varepsilon)^{-1/2n} \Big\}.
    \end{align*}
    Since $\eta_\varepsilon(L)$ is the infimum of the leftmost set, this inclusion immediately implies that
    \begin{align*}
        \eta_\varepsilon(L) \geq \inf \Big\{ s > 0 \, \Big| \, s \geq (1+\varepsilon)^{-1/2n} \Big\} 
        = 
        (1+\varepsilon)^{-1/2n} 
        = 
        1 + O\bigl( \tfrac{1}{n}\bigr).
    \end{align*}
    The final claim follows, and the proof is therefore concluded.
