<!-- theorem-1: body_md -->
\label{EKRth}
    Let $r$ and $n$ be integers such that $n \geq 2r$. Then $\alpha(G(n, r, 0)) = \binom{n-1}{r-1}$.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{alpha}
    For any fixed $r$ and $s$ such that $r > 2s + 1$ there exists $n_0(r,s)$ such that for every $n > n_0(r, s)$ we have 
    $$
        \alpha(G(n, r, s)) = \binom{n-s-1}{r-s-1}.
    $$
<!-- end theorem-2 -->
<!-- definition-1: body_md -->
Let $n$, $r$ and $s$ be integers and fix $p \in [0, 1]$. Then 
    $
        G_p(n, r, s)
    $
    is a random graph obtained from $G(n, r, s)$ by deleting each of its edges independently of the others with probability $1 - p$.
<!-- end definition-1 -->
<!-- theorem-3: body_md -->
Let $\varepsilon > 0$ be an arbitrary number, and let $n \geq 2r+1$. Then for every $p \geq (1 + \varepsilon)p_0$, where 
    $$
    p_0 = \begin{cases}
        3/4, & n = 2r + 1, \\
        \log\left(n\binom{n-1}{r}\right) / \binom{n-r-1}{r-1},& n > 2r+1,
    \end{cases}
    $$
    the graph $G_p(n,r,0)$ with high probability (w.h.p.) has independence number equal to $\binom{n-1}{r-1}$.
    
    Moreover the equality $\alpha(G_p(n,r,0)) = \binom{n-1}{r-1}$ w.h.p. does not hold for any $p \leq (1 - \varepsilon)p_0$.
<!-- end theorem-3 -->
<!-- theorem-4: body_md -->
\label{weakSt}
    Fix $r$ and $s$ such that $r > 2s + 1$. 
    Let
    $$
    p_0(n,r,s)=\frac{\ln n}{n^{r-2s-1}}.
    $$
 Then, as $n\to\infty$,
$$
\mathbf{P}\Bigl[\alpha\bigl(G_p(n,r,s)\bigr)
=\Theta\bigl(\alpha(G(n,r,s))\bigr)\Bigr]
\longrightarrow
\begin{cases}
1, & \text{if } p/p_0\ge 1,\\[1ex]
0, & \text{if } p/p_0\to 0.
\end{cases}
$$
<!-- end theorem-4 -->
<!-- theorem-5: body_md -->
\label{Gnr1}
    Fix $r \geq 5$. Then w.h.p. $\alpha(G_{1/2}(n,r,1)) = \alpha(G(n,r,1))$.
<!-- end theorem-5 -->
<!-- theorem-6: body_md -->
\label{main}
    Fix $s \geq 1$, $r > 2s+2$ and $\varepsilon \in (0, 1)$. Let $p_0(n)$, $p_1(n)$ be functions such that $$p_0(n) \leq (1-\varepsilon)\frac{\ln\left(\binom{n}{s+1}\binom{n-s-1}{r}\right)}{\binom{r}{s}\binom{n-r-s-1}{r-2s-1}},$$ 
    $$p_1(n) = \omega\left(\frac{\ln\left(\binom{n}{s+1}\binom{n-s-1}{r}\right)}{\binom{r}{s}\binom{n-r-s-1}{r-2s-1}}\right) = \omega\left(\frac{\ln n}{n^{r-2s-1}}\right).$$
    W.h.p. we have $\alpha(G_{p_0}(n,r,s)) > \alpha(G(n,r,s))$ and $\alpha(G_{p_1}(n,r,s)) = \alpha(G(n,r,s))$.
<!-- end theorem-6 -->
<!-- remark-1: body_md -->
Note that Theorem \ref{main} gives us the coarse probability threshold for the property $\alpha(G_p(n,r,s)) = \alpha(G(n,r,s))$. Moreover, it is likely that the value 
    $$
    \frac{\ln\left(\binom{n}{s+1}\binom{n-s-1}{r}\right)}{\binom{r}{s}\binom{n-r-s-1}{r-2s-1}}
    $$
    is an exact probability threshold for this property.
<!-- end remark-1 -->
<!-- remark-2: body_md -->
Another thing to note here is that we need a slightly stronger assumption on $r$ in comparison with Theorem \ref{weakSt}. While we believe that Theorem \ref{main} holds for $r = 2s+2$ as well, this case seems to be out of reach for our technique.
<!-- end remark-2 -->

Let $\mathcal{A}$ be an $r$-partite set of vertices of $G(n, r, s)$ with parts $X_1, \dotsc, X_r$. Consider a vertex $v$ from $\mathcal{A}$ and a set $J \subset [r]$. We call $v_J := v \cap \left(\bigcup_{j \in J} X_j\right)$ the narrowing of $v$ on $J$.

<!-- theorem-7: body_md -->
\label{struct}
    Let $\mathcal{A}$ be a set of vertices of $G(n,r,s)$. Fix an integer $a$ such that $a \geq 2r$. There exists $c = c(r, a) > 0$ (note that $c$ does not depend on $\mathcal{A}$) and $\mathcal{A}^{\prime} \subset \mathcal{A}$ such that
    \begin{enumerate}
        \item $\mathcal{A}^{\prime}$ is $r$-partite;

        \item $|
        \mathcal{A}^{\prime}| \geq c(r, a) |\mathcal{A}|$;

        \item For each $J \in \textrm{Int}(\mathcal{A}^{\prime})$ and each $v \in \mathcal{A}^{\prime}$ we have
        $$
            |\mathcal{A}^{\prime}(v_J)| \geq \max\left\{a, \frac{|\mathcal{A}^{\prime}|}{2rn^{|J|}}\right\}.
        $$

        \item For each $J \in \textrm{Int}(\mathcal{A}^{\prime})$ and each $v \in \mathcal{A}^{\prime}$ $C = \mathcal{A}^{\prime}(v_J)$ is $a$-diverse (that is, $|C(\{i\})| \leq \frac{1}{a}|C|$ for each $i \in [n]$).

        \item $\textrm{Int}(\mathcal{A}^{\prime})$ is closed under intersections.
    \end{enumerate}

<!-- definition-7: body_md -->
Let $\mathcal{F}$ be a family of sets. Define its $i$-shadow as the family of all $i$-element sets $U$ such that there is at least one set $F \in \mathcal{F}$ that is a superset of $U$. We denote the $i$-shadow of $\mathcal{F}$ as $\partial^{(i)}(\mathcal{F})$.

<!-- theorem-8: body_md -->
\label{KrKat}
    Let $\mathcal{F}$ be a family of $r$-element subsets. Let also $|\mathcal{F}| = \binom{x}{r} := \frac{x(x-1)\dotsc(x-r+1)}{r!}$, where $x$ may be non-integer. Then for each $s < r$ we have
    $$
        |\partial^{(s)}(\mathcal{F})| \geq \binom{x}{s}.

<!-- theorem-9: body_md -->
\label{shadow_for_ind}
    Let $\mathcal{A}$ be an independent set in $G(n,r,s)$, $r \geq 2s+1$. Then $|\partial^{(r-s-1)}(\mathcal{A})| \geq c(r)|\mathcal{A}|$ for some constant $c(r) > 0$.
<!-- end theorem-9 -->
<!-- definition-8: body_md -->
Let $\mathcal{M}$ be a family of subsets of $[r]$. Define its rank $\mathrm{rk}(\mathcal{M})$ as the minimum $l$ such that there exists $A \subset [r]$: $|A| = l$ and $A \nsubseteq M$ for any $M \in \mathcal{M} \setminus \{[r]\}$.
<!-- end definition-8 -->
<!-- lemma-1: body_md -->
\label{centre}
    Let $r$ and $s$ be two integers such that $r > 2s+2$. Let $\mathcal{M}$ be a family of subsets of $[r]$ with $\mathrm{rk}(\mathcal{M}) \geq r-s-1$. Assume that $\mathcal{M}$ is closed under intersection. Then one of the following holds.

    \begin{enumerate}
        \item There exists $M \in \mathcal{M}$ such that $|M| = s$;

        \item $\mathcal{M}$ consists of a set $M$ such that $|M| = s + 1$, all of its supersets and some of its subsets. In this case we call such $M$ the center of $\mathcal{M}$. 
    \end{enumerate}
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{rank}
    Let $\mathcal{A}$ be an $r$-partite set of vertices of $G(n, r, s)$ such that $|\mathcal{A}| > \binom{n}{r-s-2}$. Then $\mathrm{rk}(\textrm{Int}(\mathcal{A})) \geq r - s - 1$.
<!-- end lemma-2 -->
<!-- lemma-3: body_md -->
\label{KupSh}
    Consider a set $\mathcal{A}^{\prime}$ of vertices of $G(n,r,s)$. Let this set satisfy the properties from Theorem \ref{struct} with $a = 2r$. Then $\mathcal{A}^{\prime}$ is either an independent set or
    $$
        e(\mathcal{A}^{\prime}) \geq \frac{(r+s)|\mathcal{A}^{\prime}|^2}{4r|\partial^{(s)}(\mathcal{A})|}.
    $$
<!-- end lemma-3 -->
<!-- proof-1: body_md -->
Consider all sets formed by a star and one vertex that does not intersect with the center of the star. We have $\binom{n}{s+1}\binom{n-s-1}{r}$ such sets. Each one of those sets is independent in $G_{p_0}(n,r,s)$ with probability
    $$
        (1-p_0)^{\binom{r}{s}\binom{n-r-s-1}{r-2s-1}} = \exp\left(-\binom{r}{s}\binom{n-r-s-1}{r-2s-1}p_0 + o(1)\right) \geq \left(\binom{n}{s+1}\binom{n-s-1}{r}\right)^{-1+\varepsilon}(1+o(1)).
    $$
    Thus the expected number of such independent sets is at least
    $$
        \left(\binom{n}{s+1}\binom{n-s-1}{r}\right)^{\varepsilon}(1+o(1))
    $$
    and tends to infinity. 

    Now it only remains to prove that 
    $$
        \mathbf{E} X^2 = (1+o(1))(\mathbf{E} X)^2
    $$
    with $X$ being the number of independent sets of such form. First notice that 
    $$
        \mathbf{E} X^2 = \sum_{\mathcal{A}_1, \mathcal{A}_2}\mathbf{P}(\text{both } \mathcal{A}_1 \text{ and } \mathcal{A}_2 \text{ are independent in } G_{p_0}(n,r,s)),
    $$
    where we sum over all pairs of sets of the mentioned form. Let us denote the probability inside this sum by $\mathbf{P}(\mathcal{A}_1, \mathcal{A}_2)$.
    Now rewrite this sum in the following form
    $$
        \mathbf{E} X^2 = \sum_{S_1, S_2: |S_i| = s+1}\sum_{u_1, u_2:|u_i \cap S_i| = 0}\mathbf{P}(I(S_1, u_1), I(S_2, u_2)),
    $$
    where $I(S, v)$ is the set formed by the star with center $S$ and vertex $v$. 
    
    Let us now notice that if $u_1 \neq u_2$ the induced subgraphs $I(S_i, u_i)$ share no more than one edge. Indeed, in both of these graphs the only existing edges are edges of the form $(v, u_i)$, where $v$ belongs to the star with center $S_i$. Thus if two such edges coincide it must have both $u_1$ and $u_2$ as its endpoints and there is at most one such edge. Thus for any $u_1 \neq u_2$ and $S_1, S_2, |S_i| = s+1$ we have
    $$
        \mathbf{P}(I(S_1, u_1), I(S_2, u_2)) \leq (1-p_0)^{2\binom{r}{s}\binom{n-r-s-1}{r - 2s - 1} - 1} = (1+o(1))(1-p_0)^{2\binom{r}{s}\binom{n-r-s-1}{r - 2s - 1}},
    $$
    where $o(1)$ is uniform over all possible choices of $S_i$ and $u_i$.
    
    
                                                       
        
                                                                                                                                                                                                             
        
    Let us now consider a pair $S_1, S_2, |S_i| = s+1$ such that $|S_1 \cap S_2| =: i \neq s+1$, or equivalently, $S_1 \neq S_2$.
    We can easily bound $\mathbf{P}(I(S_1, u), I(S_2, u))$ for any $u$ not intersecting $S_1 \cup S_2$.
    \begin{multline*}
        \mathbf{P}(I(S_1, u), I(S_2, u)) = (1-p_0)^{2\binom{r}{s}\binom{n-r-s-1}{r - 2s - 1} - \binom{r}{s}\binom{n-r-2(s+1)+i}{r-s-2(s+1)+i}} \leq \\ \leq (1-p_0)^{2\binom{r}{s}\binom{n-r-s-1}{r - 2s - 1} - \binom{r}{s}\binom{n-r-s-2}{r-2s-2}} = \\ = (1-p_0)^{2\binom{r}{s}\binom{n-r-s-1}{r - 2s - 1}}\exp\left((p_0+p_0^2)\binom{r}{s}\binom{n-r-s-2}{r-2s-2}\right) = \\ =(1+o(1))(1-p_0)^{2\binom{r}{s}\binom{n-r-s-1}{r - 2s - 1}},
    \end{multline*}
    where $o(1)$ is uniform over all possible choices of $S_i$ and $u$.

    We can thus tell that
    \begin{multline*}
        \mathbf{E} X^2 \leq \sum_{S_1, S_2: |S_i| = s+1}\sum_{u_1, u_2: |u_i \cap S_i| = 0} (1+o(1))(1-p_0)^{2\binom{r}{s}\binom{n-r-s-1}{r - 2s - 1}} + \sum_{S: |S| = s+1}\sum_{u: |u \cap S| = 0}\mathbf{P}(I(S, u), I(S, u)) = \\ = (1+o(1))(\mathbf{E} X)^2 + \mathbf{E} X
    \end{multline*}
    and we are done since $\mathbf{E} X \to \infty \Rightarrow \mathbf{E} X = o((\mathbf{E} X)^2)$.

This proof goes along the lines of Kupavskii and Shubin paper \cite{ShKup} with slight modifications.

    In the proof we use the following notation $z := \frac{p_1n^{r-2s-1}}{\ln n}$ and $d(\mathcal{F}) := \min\limits_{S: |S| = s+1} |\mathcal{F}| - |\mathcal{F}[S]|$. Note that $z = \omega(1)$. 

    Let us take some set of vertices $\mathcal{F}$ such that $|\mathcal{F}| = \binom{n-s-1}{r-s-1} + 1$. Put $k := d(\mathcal{F})$. Fix $f = \min(z^{\frac{1}{5(s+2)}},\ln n)$.


    Before we start, let us note that it suffices to prove that each such $\mathcal{F}$ contains $\omega\left(\frac{k\ln n}{p_1}\right) = \omega\left(\frac{kn^{r-2s-1}}{z}\right)$ edges, since in this case the probability of at least one $\mathcal{F}$ being independent can be estimated by a trivial union bound. Indeed,
    \begin{multline*}
        \mathbf{P}\left(\exists \mathcal{F}: |\mathcal{F}| = \binom{n-s-1}{r-s-1} + 1, \mathcal{F} \text{ is independent in } G_{p_1}(n,r,s)\right) \leq \\ \leq  \sum_{k = 1}^{\binom{n-s-1}{r-s-1}}\binom{n}{s+1}\binom{\binom{n-s-1}{r-s-1}}{k-1}\binom{\binom{n}{r}}{k}(1-p_1)^{\omega\left(\frac{k\ln n}{p_1}\right)} = \\ = \sum_{k = 1}^{\binom{n-s-1}{r-s-1}}\binom{n}{s+1}\binom{\binom{n-s-1}{r-s-1}}{k-1}\binom{\binom{n}{r}}{k}\exp\{-\omega\left(k\ln n\right)\} \leq \\ \leq \sum_{k = 1}^{\binom{n-s-1}{r-s-1}}\exp\left(O(k\ln n)-\omega(k\ln n)\right) = o(1).
    \end{multline*}

    We start by repeatedly using Theorem \ref{struct} to decompose (most of) $\mathcal{F}$ into $r$-partite subsets. That is, 
    $$
    \mathcal{F} = \mathcal{F}_1 \sqcup \mathcal{F}_2 \sqcup \dotsc \sqcup \mathcal{F}_m \sqcup \mathcal{F}_{m+1},
    $$
    where $\mathcal{F}_i$, $i \leq m$ all satisfy the properties from Theorem \ref{struct}, $|\mathcal{F}_{i}| > \frac{c(r)}{f(n, r, s, p_1)}\binom{n}{r-s-1}$ and $|\mathcal{F}_{m+1}| \leq \frac{1}{f(n, r, s, p_1)}\binom{n}{r-s-1}$. Also note that these bounds imply that $m = O(f(n,r,s,p_1))$.

    Suppose there exists at least one $\mathcal{F}_i$, $i \leq m$ with an edge inside. Then the number of edges inside $\mathcal{F}$ is at least
    $$
        \frac{(r+s)\min_{i \leq m}|\mathcal{F}_i|^2}{4r\max_{i \leq m} |\partial^{(s)}(\mathcal{F}_i)|} \geq \frac{c(r,s)\binom{n}{r-s-1}^2}{f^2(n,r,s,p_1)\binom{n}{s}} \geq \Tilde{c}(r,s)\frac{n^{2r-3s-2}}{f^2(n, r, s, p_1)} = \omega\left(\frac{kn^{r-2s-1}}{z}\right)
    $$
    and we are done.

    We thus may assume that all $\mathcal{F}_i$ are independent.

    We can now notice that since $|\mathcal{F}_i| > \binom{n}{r-s-2}$, we can apply Lemma \ref{rank} to show that $\mathrm{rk}(\mathrm{Int}(\mathcal{F}_i)) \geq r-s-1$. By virtue of Lemma \ref{centre} we can say that either there is a set $A \in \mathrm{Int}(\mathcal{F}_i)$ such that $|A| = s$ or $\mathrm{Int}(\mathcal{F}_i)$ has a center. Assume the former holds. This means that there exist two vertices from $\mathcal{F}_i$ such that they share exactly $s$ elements. This contradicts the assumption that each $\mathcal{F}_i$ is independent. 
    
    We can thus assume that each of $\mathcal{F}_i$ has a center. This means that if we take the partition $X_{i,1} \sqcup \dotsc \sqcup X_{i, r}$ (recall that $\mathcal{F}_i$ is $r$-partite) there exists a set $M_i \subset [r]$, $|M_i| = s+1$ such that for each $v \in \mathcal{F}_i$ and each $j \in M_i$ we have $|X_{i,j} \cap v| = 1$. Let us define $\mathrm{center}(v)$ as $v|_{M_i}$. Thus $\mathrm{center}(v)$ is an $(s+1)$-element subset of $[n]$. 

    Now consider all $\mathrm{center}(v)$ for all $v \in \mathcal{F}_1 \sqcup \dotsc \sqcup \mathcal{F}_m$ and enumerate them as $C_1, \dotsc, C_q$. Also define $\mathcal{G}_j$, $j \leq q$ as follows.
    $$
        \mathcal{G}_j = \{v \in \mathcal{F}_1 \sqcup \mathcal{F}_2 \sqcup \dotsc \sqcup \mathcal{F}_m \mid \mathrm{center}(v) = C_j\}.
    $$

    The next thing to prove is the inequality
    \begin{equation}\label{sum_bound}
        \sum_{j = 1}^{q}|\partial^{(s)}(\mathcal{G}_j(C_j))| \leq m\binom{n}{s}.
    \end{equation}

    Consider the following sets $S_{i,j} = \partial^{(s)}((\mathcal{G}_j \cap \mathcal{F}_i)(C_j))$. If we can show that $S_{i,j_2} \cap S_{i,j_2} = \varnothing$ for $j_1 \neq j_2$, we are done, since 
    $$
        \sum_{j = 1}^{q}|\partial^{(s)}(\mathcal{G}_i(C_i))| \leq \sum_{j = 1}^{q}\sum_{i = 1}^{m}|\partial^{(s)}((\mathcal{G}_j \cap \mathcal{F}_i)(C_j))| \leq \sum_{i = 1}^{m}\binom{n}{s} = m\binom{n}{s}.
    $$

    To show that $S_{i,j_1} \cap S_{i,j_2} = \varnothing$ for $j_1 \neq j_2$ we only need to check that for any two vertices $u$ and $v$ from $\mathcal{F}_i$ such that $\mathrm{center}(u) \neq \mathrm{center}(v)$ we have $$|(u \setminus \mathrm{center}(u)) \cap (v \setminus \mathrm{center}(v))| \leq |(u \cap v) \setminus (\mathrm{center}(u) \cap \mathrm{center}(v))| < s.$$ 
    
    Assume that $|(u \cap v) \setminus (\mathrm{center}(u) \cap \mathrm{center}(v))| \geq s$. Let us take a set $Y \subset (u \cap v) \setminus (\mathrm{center}(u) \cap \mathrm{center}(v))$  such that $|Y| = s - |\mathrm{center}(u) \cap \mathrm{center}(v)|$. Since $M_i \cup \pi(Y)$ is a superset of $M_i$ by Lemma \ref{centre} we have that $(M_i \cup \pi(Y)) \in \mathrm{Int}(\mathcal{F}_i)$. Since $\mathrm{Int}(\mathcal{F}_i)$ is closed under intersection (property 5 in Theorem \ref{struct}), we get that $Z = (M_i \cup \pi(Y)) \cap \pi(u \cap v) \in \mathrm{Int}(\mathcal{F}_i)$. It remains to notice that $Z = \pi(Y) \sqcup \pi(\mathrm{center}(u) \cap \mathrm{center}(v))$ and thus $|Z| = s$. This means that there exist two vertices in $\mathcal{F}_i$ that intersect in exactly $s$ elements. Thus $\mathcal{F}_i$ is not independent which contradicts with our assumption. 

    Let us renumber the sets $\mathcal{G}_j$ in such a way that $|\partial^{(s)}(\mathcal{G}_1(C_1))| \geq |\partial^{(s)}(\mathcal{G}_2(C_2))| \dotsc$. We want to prove that the total size of the first $h = h(n,r,s,p_1) = \lceil f^{s+1}(n,r,s,p_1)\rceil$ sets $\mathcal{G}_j$ is almost the same as the total size of $\mathcal{F}$. More precisely, we prove that
    \begin{equation}\label{lb_on_sum_cG}
        \sum_{j = 1}^{h}|\mathcal{G}_j| = (1-o(1))\binom{n}{r-s-1}
    \end{equation}
    holds.

    Indeed, let us take $x_1 \geq \dotsc \geq x_q$ such that $|\partial^{(s)}(\mathcal{G}_j(C_j))| = \binom{x_j}{s}$. Note $x_j$ are not necessarily integer. From inequality \ref{sum_bound} we obtain that $|\partial^{(s)}(\mathcal{G}_{h+1})| \leq \frac{m}{h+1}\binom{n}{s}$. We thus have the following bound on $x_{h+1}$.
    \begin{multline*}
        \frac{(x_{h+1}-s+1)^s}{s!}\leq \binom{x_{h+1}}{s} \leq \frac{m}{h+1}\binom{n}{s} \leq \frac{mn^s}{(h+1)s!} \Rightarrow \\ \Rightarrow x_{h+1}-s+1 \leq \left(\frac{m}{h+1}\right)^{1/s}n = O\left(\frac{n}{f(n,r,s,p_1)}\right).
    \end{multline*}
    
    We also have the following bound that is trivial from Theorem \ref{KrKat}.
    $$
        |\mathcal{G}_j(C_j)| \leq \binom{x_j}{r-s-1}.
    $$
    Note that here we use the inequality $r - s - 1 \geq s$.

    The last observation we need is that the function 
    $$
        R(x) = \frac{\binom{x}{r-s-1}}{\binom{x}{s}} = \frac{s!}{(r-s-1)!}(x-s)\dotsc(x-r+s+2)
    $$
    is increasing for $x \geq r-s-2$ and so $R(x_j) \leq R(x_{h+1})$ for $j \geq h+1$. 

    Combining these bounds, we obtain that
    \begin{multline*}
        \sum_{j = h+1}^{q}|\mathcal{G}_j| = \sum_{j = h+1}^{q}|\mathcal{G}_j(C_j)| \leq \sum_{j = h+1}^{q}\binom{x_j}{r-s-1} = \sum_{j = h+1}^{q}\frac{\binom{x_j}{r-s-1}}{\binom{x_j}{s}}\binom{x_j}{s} = \\ = \sum_{j = h+1}^{q}\binom{x_j}{s}R(x_j) \leq R(x_{h+1})\sum_{j = h+1}^{q}\binom{x_j}{s} = R(x_{h+1})\sum_{j = h+1}^{q}|\partial^{(s)}(\mathcal{G}_j(C_j))| \leq m\binom{n}{s}R(x_{h+1}) \leq \\ \leq \frac{s!m}{(r-s-1)!}\binom{n}{s}x_{h+1}^{r-2s-1} = O\left(\frac{n^{r-s-1}}{f^{r-2s-2}}\right) = o(n^{r-s-1}).
    \end{multline*}
    
    It remains to notice that $$\sum_{j = 1}^{q}|\mathcal{G}_j| = |\mathcal{F}_1 \sqcup \dotsc \sqcup \mathcal{F}_{m}| = (1-o(1))\binom{n}{r-s-1}$$ and thus 
    $$
    \sum_{j = 1}^{h}|\mathcal{G}_j| = (1-o(1))\binom{n}{r-s-1} - \sum_{j = h+1}^{q}|\mathcal{G}_j| = (1-o(1))\binom{n}{r-s-1}.
    $$

    Our next step is to find subsets $\mathcal{G}^{\prime}_j$ of $\mathcal{G}_j$, $j \leq h$ that have some good properties and at the same time are almost of the same size as $\mathcal{G}_j$.

    Define $W = \bigcup_{j = 1}^{h} C_j$, $\tau = \min(z^{1/5},\ln^{s+2} n)$. We want to obtain $\mathcal{G}_1^{\prime}, \dotsc, \mathcal{G}_{h}^{\prime}$ that satisfy the following properties.

    \begin{enumerate}
        \item $\mathcal{G}^{\prime}_j \subseteq \mathcal{G}_j$ for each $j \in [h]$.
        
        \item 
        $$
        \sum_{j = 1}^{h} |\mathcal{G}_j| - \sum_{j = 1}^{h} |\mathcal{G}^{\prime}_j| = o(n^{r-s-1}).
        $$

        \item For any $j \in [h]$ and $G \in \mathcal{G}^{\prime}_j$, $G \cap W = C_j$.

        \item For any $j \in [h]$ and $X 
        \in \partial^{(s)}(\mathcal{G}^{\prime}_j(C_j))$, $|\mathcal{G}^{\prime}_j(C_j \cup X)| \geq \frac{n^{r-2s-1}}{\tau}$.

        \item For any $j \in [h]$ and $X 
        \in [n] \setminus W$, $|X| \leq s$, the family $\mathcal{G}^{\prime}_j(C_j \cup X)$ is $\tau$-diverse.
    \end{enumerate}

    To do so we first fulfill the third condition by removing every $v \in \mathcal{G}_j$ such that $v \cap W \neq C_j$. Since every such $v$ has at least $s+2$ vertices in $W$, we get that there are no more than 
    $$
        \binom{|W|}{s+2}\binom{n}{r-s-2} \leq |W|^{s+2}n^{r-s-2} \leq r^{s+2}h^{s+2}n^{r-s-2} = o(n^{r-s-1})
    $$
    removed vertices.

    To fulfill conditions 4 and 5 we perform the following algorithm. As long as there exists a pair $\mathcal{G}^{\prime}_j$ and $X \in [n]\setminus W$, $|X| \leq s$ violating condition 4 or 5, we remove all $v \in \mathcal{G}^{\prime}_j$ that contain $X$ as a subset.

    Note that if the pair $(\mathcal{G}^{\prime}_j, X)$ violates condition 4, we remove at most $\frac{n^{r-2s-1}}{\tau}$ vertices at this step. If this pair violates condition 5, we can see that there exists some $i \in [n] \setminus (C_j \cup X)$ such that $|\mathcal{G}^{\prime}_j(C_j \cup X \cup \{i\})| \geq \frac{1}{\tau}|\mathcal{G}^{\prime}_j(C_j \cup X)|$. Since $|\mathcal{G}^{\prime}_j(C_j \cup X \cup \{i\})| \leq \binom{n}{r-s-2-|X|}$, we get that at most
    $
        \tau\binom{n}{r-s-2-|X|}
    $
    vertices are removed at this step. As we can only perform the removal at most once for each pair $(\mathcal{G}^{\prime}_j, X)$, the total number of removed vertices is at most
    $$
    h\tau\sum_{x = 0}^{s}\binom{n}{x}\binom{n}{r-s-2-x} + \binom{n}{s}\frac{hn^{r-2s-1}}{\tau} \leq sh\tau n^{r-s-2} + \frac{hn^{r-s-1}}{\tau} = o(n^{r-s-1}).
    $$

    Thus after the end of this procedure the sets $\mathcal{G}^{\prime}_j$ satisfy all conditions 1--5.

    Let us now assume that $$\sum_{j = 1}^{h}|\partial^{(s)}(\mathcal{G}^{\prime}_j)| \geq \binom{n}{s}+h^22^s\binom{n}{s-1}+\frac{k}{n^{r-2s-1}\tau}.$$ Notice that for any $v \in \mathcal{G}^{\prime}_i$ and for any $X \subset [n]\setminus W$, $|X| = s - |C_i \cap C_j| =: x$ such that $X \subset v$ we have at least $|\mathcal{G}^{\prime}_j(X \cup C_j)| - \sum_{a \in G \setminus (X \cup C_j)} |\mathcal{G}^{\prime}_j(X \cup C_j \cup \{a\})|$ edges between $v$ and vertices from $\mathcal{G}^{\prime}_j$ such that the endpoints of these edges have common set precisely $X \cup (C_i \cap C_j)$. Using property 5 of $\mathcal{G}^{\prime}_j$ we can further bound the number of such edges by
    $$
    |\mathcal{G}^{\prime}_j(X \cup C_j)| - \sum_{a \in G \setminus (X \cup C_j)} |\mathcal{G}^{\prime}_j(X \cup C_j \cup \{a\})| \geq |\mathcal{G}^{\prime}_j(X \cup C_j)| - \frac{r}{\tau}|\mathcal{G}^{\prime}_j(X \cup C_j)| \geq \frac{\tau - r}{\tau}|\mathcal{G}^{\prime}_j(X \cup C_j)|
    $$

    Summing these inequalities over all pairs of $v \in \mathcal{G}^{\prime}_i$ and $X \subset [n]\setminus W$, $|X| = v$, we get that the number of edges inside $\mathcal{G}^{\prime}_i \cup \mathcal{G}^{\prime}_j$ is at least

    \begin{equation*}
        e(\mathcal{G}^{\prime}_i \cup \mathcal{G}^{\prime}_j) \geq \frac{\tau - r}{\tau}\sum_{X \subset [n]\setminus W,\\ |X| = x}|\mathcal{G}^{\prime}_i(C_i \cup X)||\mathcal{G}^{\prime}_j(C_j \cup X)|,
    \end{equation*}
    so the total number of edges inside $\mathcal{F} \supseteq (\mathcal{G}^{\prime}_1 \cup \dotsc \cup \mathcal{G}^{\prime}_h)$ is at least
    \begin{equation}
        \label{ineq1}
        e(\mathcal{F}) \geq \frac{\tau - r}{\tau}\sum_{i < j}\sum_{X \subset [n]\setminus W,\\ |X| = x}|\mathcal{G}^{\prime}_i(C_i \cup X)||\mathcal{G}^{\prime}_j(C_j \cup X)|
    \end{equation}

    At the same time we have the following trivial inequality
    \begin{equation}
        \label{ineq2}
        \sum_{X \subset [n]\setminus W, |X| = x}|\mathcal{G}^{\prime}_i(C_i \cup X)||\mathcal{G}^{\prime}_j(C_j \cup X)| \geq \frac{1}{\binom{r}{s}}\sum_{X \subset [n]\setminus W, |X| = s}|\mathcal{G}^{\prime}_i(C_i \cup X)||\mathcal{G}^{\prime}_j(C_j \cup X)|.
    \end{equation}
    
    Indeed, on the left hand side we count each pair of vertices $u, v$ such that $u \in \mathcal{G}^{\prime}_i$, $v \in \mathcal{G}^{\prime}_j$ and $|u \cap v| \geq s$ at least once, while on the right hand side we count each of these pairs at most $\binom{|u \cap v|}{s} \leq \binom{r}{s}$ times. Combining inequalities \ref{ineq1} and \ref{ineq2}, we obtain
    $$
    e(\mathcal{F}) \geq \frac{\tau-r}{\tau\binom{r}{s}}\sum_{i < j} \sum_{X \subset [n]\setminus W,\\ |X| = s} |\mathcal{G}^{\prime}_i(C_i \cup X)||\mathcal{G}^{\prime}_j(C_j \cup X)|.
    $$
    
    Let us now prove that under the assumption $\sum_{i}^{h}|\partial^{(s)}(\mathcal{G}^{\prime}_i)| \geq \binom{n}{s}+h^22^s\binom{n}{s-1}+\frac{k}{n^{r-2s-1}\tau}$ we have at least $\frac{k}{n^{r-2s-1}\tau}$ triples $(\mathcal{G}^{\prime}_i, \mathcal{G}^{\prime}_j, X)$, $X \subset [n] \setminus W$, $|X| = s$ such that $X \in \partial^{(s)}(\mathcal{G}^{\prime}_i) \cap \partial^{(s)}(\mathcal{G}^{\prime}_j)$. Indeed, 
    \begin{multline*}
       \sum_{i}^{h}|\partial^{(s)}(\mathcal{G}^{\prime}_i)| \leq \binom{n}{s} + \sum_{1 \leq i < j \leq h}|\partial^{(s)}(\mathcal{G}^{\prime}_i) \cap \partial^{(s)}(\mathcal{G}^{\prime}_j)| = \binom{n}{s} + \sum_{1 \leq i < j \leq h}\sum_{\substack{X \in \partial^{(s)}(\mathcal{G}^{\prime}_i) \cap \partial^{(s)}(\mathcal{G}^{\prime}_j)}}1 \leq \\ \leq \binom{n}{s} + \sum_{1 \leq i < j \leq h}\sum_{l = 1}^{|C_i \cap C_j|}\sum_{\substack{X \in \partial^{(s)}(\mathcal{G}^{\prime}_i) \cap \partial^{(s)}(\mathcal{G}^{\prime}_j),\\ |X \cap W| = l}}1 + \sum_{1 \leq i < j \leq h}\sum_{\substack{X \in \partial^{(s)}(\mathcal{G}^{\prime}_i) \cap \partial^{(s)}(\mathcal{G}^{\prime}_j),\\ |X \cap W| = l}}1 \leq \\ \leq \binom{n}{s} + h^22^s\binom{n}{s-1} + \sum_{1 \leq i < j \leq h}\sum_{\substack{X \in \partial^{(s)}(\mathcal{G}^{\prime}_i) \cap \partial^{(s)}(\mathcal{G}^{\prime}_j),\\ |X \cap W| = l}}1.
    \end{multline*}
    Using our assumption we see that 
    $$
    \sum_{1 \leq i < j \leq h}\sum_{\substack{X \in \partial^{(s)}(\mathcal{G}^{\prime}_i) \cap \partial^{(s)}(\mathcal{G}^{\prime}_j),\\ X \subset [n] \setminus W}}1 \geq \frac{k}{n^{r-2s-1}\tau},
    $$
    which is exactly what we need.

    Since there are at least $\frac{k}{n^{r-2s-1}\tau}$ triples $(\mathcal{G}^{\prime}_i, \mathcal{G}^{\prime}_j, X)$, $X \subset [n] \setminus W$, $|X| = s$ such that $X \in \partial^{(s)}(\mathcal{G}^{\prime}_i) \cap \partial^{(s)}(\mathcal{G}^{\prime}_j)$, we have that there are at least as many nonzero summands in 
    $$
        \sum_{i < j} \sum_{X \subset [n]\setminus W,\\ |X| = s} |\mathcal{G}^{\prime}_i(C_i \cup X)||\mathcal{G}^{\prime}_j(C_j \cup X)|.
    $$
    For each of these summands we know that both $|\mathcal{G}^{\prime}_i(C_i \cup X)|$ and $|\mathcal{G}^{\prime}_j(C_j \cup X)|$ are at least $\frac{n^{r-2s-1}}{\tau}$ due to property 4. This means that in this case we have
    $$
        e(\mathcal{F}) \geq \frac{\tau-r}{\binom{r}{s}\tau}\frac{k}{n^{r-2s-1}\tau}\left(\frac{n^{r-2s-1}}{\tau}\right)^2 = \omega\left(\frac{kn^{r-2s-1}}{\tau^4}\right).
    $$

    Using the fact that $\tau^4 = o\left(z\right)$ we get that the number of edges in $\mathcal{F}$ is $\omega\left(\frac{kn^{r-2s-1}}{z}\right)$ and so we are done.
    
    We may thus assume that $\sum_{i}^{h}|\partial^{(s)}(\mathcal{G}^{\prime}_i)| < \binom{n}{s}+h^22^s\binom{n}{s-1}+\frac{k}{n^{r-2s-1}\tau} = (1+o(1))\binom{n}{s}$.

    We are now going to prove that there exists $i \in [h]$ such that $|\mathcal{G}^{\prime}_i| = (1+o(1))\binom{n}{r-s-1}$ thus proving that almost all of the vertices from $\mathcal{F}$ are a part of one star.  

    Once again, let us renumber $\mathcal{G}^{\prime}_i$ in such a way that $|\mathcal{G}^{\prime}_1| \geq \dotsc \geq |\mathcal{G}^{\prime}_h|$. Furthermore let $y_1, \dotsc, y_h$ be real numbers such that $|\mathcal{G}^{\prime}_i| = \binom{y_i}{r-s-1}$. Clearly, $y_1 \geq \dotsc \geq y_h$.

    From Theorem \ref{KrKat} we have that
    \begin{equation*}
    \sum_{i = 1}^{h}\binom{y_i}{s} \leq \sum_{i = 1}^{h}|\partial^{(s)}(\mathcal{G}^{\prime}_i)| \leq (1+o(1))\binom{n}{s}.
    \end{equation*}

    Now recall that 
    $$
        \sum_{i = 1}^{h}|\mathcal{G}^{\prime}_i| \geq \sum_{i = 1}^{h}|\mathcal{G}_i| + o(n^{r-s-1}) \geq (1-o(1))\binom{n}{r-s-1},
    $$
    where the last inequality follows trivially from inequality \ref{lb_on_sum_cG}. On the other end, we have
    $$
    \sum_{i = 1}^{h}|\mathcal{G}^{\prime}_i| = \sum_{i = 1}^{h}\binom{y_i}{r-s-1} = \sum_{i =1}^{h}R(y_i)\binom{y_i}{s} \leq R(y_1)\sum_{i = 1}^{h}\binom{y_i}{s} \leq (1+o(1))\binom{n}{s}R(y_1).
    $$
    Combining these two bounds we obtain
    $$
        R(y_1) = \frac{\binom{y_1}{r-s-1}}{\binom{y_1}{s}} \geq (1-o(1))\frac{\binom{n}{r-s-1}}{\binom{n}{s}},
    $$
    from which $y_1 = (1-o(1))n$ follows trivially.

    Let us now return to the initial set $\mathcal{F}$. We know that the intersection of $\mathcal{F}$ with one of the stars (without loss of generality we assume that this star is $\left\{v \in G(n,r,s)\mid [s+1] \subset v\right\}$) is $(1-o(1))\binom{n}{r-s-1}$. Let us denote the set of vertices from this intersection by $\mathcal{A}$ and the set of other vertices from the star by $\mathcal{A}^{\prime}$. Furthermore let $\mathcal{B} := \mathcal{F} \setminus \mathcal{A}$. We know that $|\mathcal{B}| = k$ and $|\mathcal{A}| = \binom{n-s-1}{r-s-1}-k+1$.
    
    Let $\mathcal{B}^{\prime} = \{v \in \mathcal{B} \mid \deg_{\mathcal{A}}(v) \leq \frac{1}{2}\binom{n-r-s-1}{r-2s-1}\}$. Assume that $|\mathcal{B}^{\prime}| \leq \frac12|\mathcal{B}| = \frac12k$. Then $\mathcal{F}$ contains at least $\frac{k\binom{n-r-s-1}{r-2s-1}}{4} = \omega\left(\frac{kn^{r-2s-1}}{z}\right)$ edges and we are done. 
    
    It remains to consider the case where $|\mathcal{B}^{\prime}| > \frac{1}{2}k$. We start by considering all possible $Q \subset [s+1]$ and taking $Q_0$ such that the number of sets from $\mathcal{B}^{\prime}$ that intersect $[s+1]$ by exactly $Q_0$ is maximal. Let $\mathcal{B}^{\prime}_Q = \mathcal{B}^{\prime}[Q_0]$. We clearly have $|\mathcal{B}^{\prime}_Q| \geq \frac{1}{2^{s+1}}|\mathcal{B}^{\prime}| \geq \frac1{2^{s+2}}k$. 
    
    Let us now prove the inequality

    \begin{equation}
        \label{lst}
        k = |\mathcal{B}| \geq \frac{|\partial^{(s-|Q_0|)}(\mathcal{B}^{\prime}_{Q_0}(Q_0))|\binom{n-r-s-1+|Q_0|}{r-2s-1+|Q_0|}}{2\binom{r-s-1}{s-|Q_0|}}.
    \end{equation}

    Let us fix a pair $(v, S)$ such that $S \subset v$, $|S| = s-|Q_0|$, $v \in \mathcal{B}^{\prime}_{Q_0}$. We have at least $\binom{n-r-s-1+|Q_0|}{r-2s-1+|Q_0|}$ vertices $u \in \mathcal{A} \sqcup \mathcal{A}^{\prime}$ such that they intersect $v$ by exactly $S \cup Q_0$. Since the number of vertices between $v$ and $\mathcal{A}$ is less than 
    $$\frac{1}{2}\binom{n-r-s-1}{r-2s-1} \leq \frac{1}{2}\binom{n-r-s-1+|Q_0|}{r-2s-1+|Q_0|},$$
    at least half of these $u$ lie in $\mathcal{A}^{\prime}$. Summing over all $S \in \partial^{(s-|Q_0|)}(\mathcal{B}^{\prime}_{Q}(Q_0))$, we obtain
    $$
    \frac{1}{2}\binom{n-r-s-1+|Q_0|}{r-2s-1+|Q_0|}|\partial^{(s-|Q_0|)}(\mathcal{B}^{\prime})| \leq \sum_{S \in \partial^{(s-|Q_0|)}(\mathcal{B}^{\prime}_Q(Q_0))}|\mathcal{A}^{\prime}(S)| \leq \binom{r-s-1}{s-|Q_0|}|\mathcal{A}^{\prime}| < \binom{r-s-1}{s-|Q_0|}|\mathcal{B}|
    $$
    Now inequality \ref{lst} follows  immediately.
    
    
    Now apply Theorem \ref{struct} to $\mathcal{B}^{\prime}_Q(Q_0)$ with $a = 2(r-|Q_0|)$ (note that we are now considering vertices of $G(n-|Q_0|,r-|Q_0|,s-|Q_0|)$ graph). We obtain an $(r-|Q_0|)$-partite set $\mathcal{B}^{\prime\prime} \subset \mathcal{B}^{\prime}_Q(Q_0)$ such that $$|\mathcal{B}^{\prime\prime}| \geq c(r-|Q_0|, 2(r-|Q_0|))|\mathcal{B}^{\prime}_Q(Q_0)| \geq \frac{c(r,2r)}{2^{s+2}}k.$$

    Consider two options. The first option is that there is at least one edge inside $\mathcal{B}^{\prime\prime}$. Note that each edge inside $\mathcal{B}^{\prime\prime}$ corresponds to an edge inside $\mathcal{B}^{\prime}_Q$, as we add the same set of points to each of the vertices. We can then use Lemma \ref{KupSh} and inequality \ref{lst} to obtain that 
    $$
        e(\mathcal{B}^{\prime\prime}) \geq \frac{c|\mathcal{B}^{\prime\prime}|^2}{|\partial^{(s-|Q_0|)}(\mathcal{B}^{\prime\prime})|} \geq \frac{c|\mathcal{B}^{\prime\prime}|^2}{|\partial^{(s-|Q_0|)}(\mathcal{B}^{\prime})|} \geq \frac{c\binom{n-r-s-1}{r-2s-1+|Q_0|}|\mathcal{B}^{\prime\prime}|^2}{2\binom{r-s-1}{s-|Q_0|}|\mathcal{B}|} \geq c^{\prime}n^{r-2s-1+|Q_0|}k = \omega\left(\frac{kn^{r-2s-1}}{z}\right).
    $$
          
    It remains to consider the case where $\mathcal{B}^{\prime\prime}$ is independent. In this case we may apply Theorem \ref{shadow_for_ind} and get that 
    \begin{equation}
        \label{1}
        k = |\mathcal{B}| \leq C_0(r)|\mathcal{B}^{\prime\prime}| \leq C_1(r)|\partial^{(r-s-1)}(\mathcal{B}^{\prime\prime})|
    \end{equation}
    for some $C_1(r) > 0$. 
    
    Let $x$ be a real number such that $|\partial^{(r-s-1)}(\mathcal{B}^{\prime\prime})| = \binom{x}{r-s-1}$. Since $$|\partial^{(r-s-1)}(\mathcal{B}^{\prime\prime})| \leq \binom{r}{r-s-1}|\mathcal{B}^{\prime\prime}| = o(n^{r-s-1}),$$ we obtain that $x = o(n)$. At the same time 
    \begin{equation}
        \label{2}
        |\partial^{(s-|Q_0|)}(\mathcal{B}^{\prime}_Q(Q_0))| \geq |\partial^{(s-|Q_0|)}(\mathcal{B}^{\prime\prime})| = |\partial^{(s-|Q_0|)}(\partial^{(r-s-1)}(\mathcal{B}^{\prime\prime}))| \geq \binom{x}{s-|Q_0|}.
    \end{equation}
    Combining inequalities \ref{1}, \ref{2} and \ref{lst}, we get
    $$
    C_1(r)\binom{x}{r-s-1} \geq k \geq \frac{|\partial^{(s-|Q_0|)}(\mathcal{B}^{\prime}_Q(Q_0))|\binom{n-r-s-1+|Q_0|}{r-2s-1+|Q_0|}}{2\binom{r-s-1}{s-|Q_0|}} \geq \frac{\binom{x}{s-|Q_0|}\binom{n-r-s-1+|Q_0|}{r-2s-1+|Q_0|}}{2\binom{r-s-1}{s-|Q_0|}}
    $$
    and thus we have
    $$
        x^{r-s-1} \geq C_2x^{s-|Q_0|}n^{r-2s-1+|Q_0|},
    $$
    which contradicts with the fact that $x = o(n)$.
