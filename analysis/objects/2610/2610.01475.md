<!-- definition-1: body_md -->
\label{def:DiscFourier}
	Let $r>0$ and let $\omega \in [-\pi,\pi]$ denote the Fourier frequency, then the exponentially weighted discrete time Fourier transform of the sequence $\{\phi^n\}_{n\in\mathbb{Z}}$ is defined as
	\begin{equation}\label{eq:transform}
		\widehat{\phi}(r \Delta t + i \omega) := \sum_{n\in\mathbb{Z}} \phi^n e^{-n (r \Delta t+i\omega)}, \quad \omega \in [-\pi, \pi],
	\end{equation}
	whenever the sum converges.
<!-- end definition-1 -->
<!-- lemma-1: body_md -->
\label{lem:prepDisc2norm}
	Let $r > 0$ and $\boldsymbol{\phi} = \{\phi^n\}_{n\in\mathbb{Z}}$ be a sequence such that $ \sum_{n\in\mathbb{N}} (\phi^n)^2 < \infty$, and $\phi^n=0$ for $n<0$. Then the exponentially weighted discrete time Fourier exists, and we have
	\[\frac{1}{2 \pi} \int_{-\pi}^{\pi} |\widehat{\phi}(r\Delta t+i\omega)|^2 d\omega \leq \left\lVert\boldsymbol{\phi}\right\rVert_{l^2}^2.\]
<!-- end lemma-1 -->
<!-- theorem-1: body_md -->
\label{thm:superLinearEstFD}
	The error of the fully discrete iteration \eqref{eq:DNWRFullyDisc} on the time interval $[0, T]$, satisfies \begin{align}\label{eq:fullyDiscEst}
		\|\boldsymbol{h}^k\|_{l^2} \leq \inf_{r \in \mathbb{R}^+} e^{rT}\max_{\omega \in [-\pi,\pi]} |\hat{\rho}^{IE}_{FD}(r \Delta t + i \omega)^k| \left\lVert\boldsymbol{h}^0\right\rVert_{l^2}.
	\end{align}
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
As $r>0$, we have
	\begin{align*}
		\|\boldsymbol{h}^k\|_{l^2}^2 \leq
		e^{2 rT} \sum_{n=0}^{N} e^{-2 r \Delta t n} \left(h^{n, k}\right)^2 \leq e^{2 rT} \sum_{n=-\infty}^{\infty} \left( e^{- r \Delta t n}h^{n, k}\right)^2,
	\end{align*} where in the last inequality we use the extension of the sequence $\boldsymbol{h}^k$. By Lemma \ref{lem:prepDisc2norm} the exponentially weighted discrete time Fourier transform of $\boldsymbol{h}^k$ exists. Parseval's theorem then yields
	\begin{align*}
		e^{2rT} \sum_{n=-\infty}^{\infty} \left( e^{- r \Delta t n}h^{n, k}\right)^2
		&\leq e^{2rT} \frac{1}{2 \pi} \int_{-\pi}^{\pi} \left| \hat{h}^{k}(r \Delta t + i \omega) \right|^2  \\
		&\leq \frac{e^{2rT}}{2 \pi} \max_{\omega \in [-\pi,\pi]} \left|\hat{\rho}^{IE}_{FD}(r \Delta t + i\omega)\right|^{2k}\int_{-\pi}^{\pi} \left| \hat{h}^{0}(r \Delta t + i \omega) \right|^2  \\
		&\leq e^{2rT} \max_{\omega \in [-\pi,\pi]} \left|\hat{\rho}^{IE}_{FD}(r \Delta t + i\omega)\right|^{2k} \left\lVert\bf{h}^0\right\rVert_{l^2}^2,
	\end{align*}
	where we use Lemma \ref{lem:prepDisc2norm} in the last inequality.  Lastly, taking the square root on both sides and the infimum over all $r$ yields \eqref{eq:fullyDiscConvfac}.
<!-- end proof-1 -->
<!-- corollary-1: body_md -->
\label{cor:independentTEst}
	The error of the fully discrete iteration \eqref{eq:DNWRFullyDisc}, satisfies  \begin{align}\label{eq:independentTEst}
		\|\boldsymbol{h}^k\|_{l^2} \leq \max_{\omega \in [-\pi,\pi]} |\hat{\rho}^{IE}_{FD}(i \omega)|^k \left\lVert\boldsymbol{h}^0\right\rVert_{l^2}.
	\end{align}
<!-- end corollary-1 -->
<!-- remark-1: body_md -->
\label{rem:Tintuition}
	To give some intuition for the effect of $T$ on the estimate \eqref{eq:fullyDiscEst}, note that if $T$ is large, then $exp(rT)$ dominates in the fully discrete estimate \eqref{eq:fullyDiscEst}, and the optimal $r^*$ is close to zero, yielding linear convergence. However, if $T$ is small, then $exp(rT) \approx 1$, and the optimal $r^*$ is close to the $r$ minimizing \begin{equation}\label{def:supWfunc}
		F^{IE}_{FD}(r) = \max_{\omega \in [-\pi,\pi]}\left| \hat{\rho}^{IE}_{FD}\left(\frac{1-e^{-r \Delta t - i \omega}}{\Delta t}\right)\right|.
	\end{equation}
<!-- end remark-1 -->
<!-- theorem-2: body_md -->
\label{thm:assymptoticsRhoFD}
	If $|s_1|\gg 1$ and $|s_2| \gg 1 $, then \[\hat{\rho}_{IE}^{FD}(r \Delta t + i \omega) \approx 1 - \theta - \theta \frac{\alpha_1 \Delta x_1}{\alpha_2 \Delta x_2}.\]
	If $|s_1|\ll 1$ and $|s_2| \ll 1 $, then \begin{align*}
		\hat{\rho}_{FD}^{IE}(r \Delta t + i \omega) \approx \hat{\rho}^{IE}(r\Delta t+i\omega) = \\
		1 - \theta - \theta\sqrt{\frac{\alpha_1 \lambda_1}{\alpha_2 \lambda_2}} \coth\left(\sqrt{\frac{\alpha_1 (1-e^{- r \Delta t-i\omega})}{\lambda_1 \Delta t}}a\right)\tanh\left(\sqrt{\frac{\alpha_2 (1-e^{- r \Delta t-i\omega })}{\lambda_2 \Delta t}}b\right),
	\end{align*}
	where $\hat{\rho}^{IE}$ is given by \eqref{eq:timeDiscRho}.
<!-- end theorem-2 -->
<!-- proof-2: body_md -->
If $s_m \gg 1$ for $m=1,2$ then $r_{m,\pm}$ given in \eqref{eq:rm} can be approximated by \begin{align*}
		r_{m, +} &= 1 + s_m/2 + \sqrt{s_m} \sqrt{1 + s_m/4}  \\
		&\approx 1 + s_m/2 + \sqrt{s_m} \sqrt{s_m/4 } 
		\approx s_m, 
	\end{align*}
	and
	\begin{align*}
		r_{m, -} &= 1 + s_m/2 - \sqrt{s_m} \sqrt{1 + s_m/4}  \\ &\approx 1 + s_m/2 - \sqrt{s_m} \sqrt{s_m/4 } 
		\approx 1.
	\end{align*}
	Inserting the approximation into $f_m$, defined in \eqref{eq:fm}, yields 
	\[f_m = \frac{(r_{m,+})^{n_m}-(r_{m,-})^{n_m}}{(r_{m,+})^{n_m+1}-(r_{m,-})^{n_m+1}-(r_{m,+})^{n_m-1}+(r_{m,-})^{n_m-1}} \approx \frac{1}{s_m}, \]
	resulting in \[
	\hat{\rho}_{FD}^{IE} =  (1 -\theta)- \theta \frac{\lambda_1 \Delta x_2}{\lambda_2 \Delta x_1} \frac{f_2}{f_1} \approx 1 - \theta - \theta \frac{\alpha_1 \Delta x_1}{\alpha_2 \Delta x_2}.
	\] 
	
	For the second case where $s_1 \ll 1$ and $s_2 \ll 1$, $r_{m,\pm}$ can instead be approximated by \begin{align*}
		r_{m, \pm} = 1 + s_m/2 \pm \sqrt{s_m} \sqrt{1 + s_m/4} &\approx 1 + s_m/2 \pm \sqrt{s_m} \\ 
		&\approx 1 \pm \sqrt{s_m}.
	\end{align*}
	Next, we rewrite $f_m$, yielding 
	\begin{equation*}
		f_m = \frac{e^{n_m \ln(r_{m,+})}-e^{n_m  \ln(r_{m,-})} }{\left(e^{(n_m+1) \ln(r_{m,+})} - e^{(n_m+1)\ln(r_{m,-})}\right) + (e^{(n_m-1)\ln(r_{m,+})} - e^{(n_m-1)\ln(r_{m,-})})}.
	\end{equation*} 
	We insert the approximation of $r_{m,\pm}$ into $\ln(r_{m,\pm})$ and do a first order Taylor expansion in $\sqrt{s_m}$ yielding 
	\begin{equation*}
		\ln(r_{m,\pm}) \approx -\pm \sqrt{s_m}.
	\end{equation*} 
	Inserting this into $f_m$ and simplifying yields 
	\begin{align*}
		f_m \approx \frac{(e^{n_m\sqrt{s_m}} - e^{-n_m\sqrt{s_m}})}{(e^{n_m\sqrt{s_m}} + e^{-n_m\sqrt{s_m} }) (e^{\sqrt{s_m}} - e^{\sqrt{s_m}})}
		= \frac{\tanh\left(n_m \sqrt{s_m}\right)}{ e^{\sqrt{s_m}} - e^{\sqrt{s_m}}}.
	\end{align*} 
	 Approximating the exponential terms by $e^{\pm\sqrt{s_m}} \approx 1\pm\sqrt{s_m}$, yields
	 	\begin{align*}
	 	f_m &\approx \frac{\tanh\left(n_m \sqrt{s_m}\right)}{1 + \sqrt{s_m} - 1 + \sqrt{s_m}} = \frac{\tanh\left(n_m \sqrt{s_m}\right)}{2 \sqrt{s_m}}.
	 \end{align*} 
	Inserting the approximation of $f_m$ into $\hat{\rho}_{FD}^{IE}$ and simplifying yields the result.
<!-- end proof-2 -->

\label{rem:small}
	For $c_m\gg 1$, $m=1,2$, the spatial discretization does not affect the convergence behaviour: If $c_m\gg 1$  then $\left| s_m\right| \ll 1$, thus Theorem \ref{thm:assymptoticsRhoFD} yields that $\hat{\rho}_{FD}^{IE}$ is close to $\hat{\rho}^{IE}$, and hence that the fully discrete estimate~\eqref{eq:fullyDiscEst} is close to the time discrete estimate~\eqref{eq:timeDiscEst}.

<!-- remark-3: body_md -->
\label{rem:large}
	If $c_m \ll 1$, then we expect the fully discrete estimates to converge linearly with convergence factor 
	\begin{align}\label{eq:approx}
		\begin{aligned}
			\inf_{r \in \mathbb{R}^+} e^{rT} \max_{\omega \in [-\pi,\pi]} | \hat{\rho}^{IE}_{FD}(r \Delta t + i \omega)^k| &\approx \inf_{r \in \mathbb{R}^+} e^{rT} \max_{\omega \in [-\pi,\pi]}  \left|1 - \theta - \theta \frac{\alpha_1 \Delta x_1}{\alpha_2 \Delta x_2}\right|^k\\
			&= \left|1 - \theta - \theta \frac{\alpha_1 \Delta x_1}{\alpha_2 \Delta x_2}\right|^k,
		\end{aligned}
	\end{align}
	where we used Theorem \ref{thm:assymptoticsRhoFD} to obtain the approximation. This implies that space discretization plays an important role.
