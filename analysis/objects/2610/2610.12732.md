<!-- lemma-1: body_md -->
\label{lemma:paradox}
    Suppose $p(w) = p$ for all $w$. 
    \begin{enumerate}[label=(\roman*)]
        \item \textbf{The classroom is underenrolled.} The steady-state fraction of occupied licensed slots is
        \begin{equation} \label{eq:umax}
            U \;=\; \min\left\{\frac{\lambda p}{K\delta},\; \frac{p\gamma}{p\gamma+\delta}\right\} < 1.
        \end{equation}
        A positive fraction of slots can be empty at any moment, even with strictly positive waitlist length. 
        \item \textbf{Families experience long waiting times.} The expected waiting time for a family from joining the waitlist to receiving an offer is
        \begin{equation} \label{eq:queue_length}
            \mathbb{W} \;=\; \frac{p\gamma+\delta}{(K\gamma \delta - p\gamma \lambda - \delta \lambda)^{+}},
        \end{equation}
        with the convention that $\mathbb{W} = \infty$ when the denominator vanishes. 
        Only a fraction $p$ of listed families are active, so the waitlist overstates the number of active families.
    \end{enumerate}
<!-- end lemma-1 -->
<!-- proposition-1: body_md -->
\label{prop:paradox}
    Empty slots and long waits coexist. The expected number of empty slots is $K\,(1 - U) \;=\; \max\left\{K - \frac{\lambda p}{\delta},\; \frac{K\delta}{p\gamma+\delta}\right\} \;\geq\; \frac{K\delta}{p\gamma+\delta} \;>\; 0$, and the expected waiting time until receiving an offer satisfies $\mathbb{W} \geq 1/\mu_{\mathrm{eff}} > 0$. As requests arrive more frequently, the expected waiting time grows unboundedly while the number of empty slots never falls below $K\delta/(p\gamma+\delta)$.
<!-- end proposition-1 -->
<!-- lemma-2: body_md -->
\label{lemma:comp_statics}
    Suppose $p(w) = p$ for all $w$ and $\mathbb{W} < \infty$. Then $\partial U/\partial\gamma = 0$.

<!-- lemma-3: body_md -->
\label{lemma:alpha-u-wait}
For a given average acceptance probability $\alpha$, the steady-state fraction of occupied licensed slots and the expected waiting time for a family from joining the waitlist to receiving an offer are  
\begin{equation} \label{eq:alpha_u}
    U(\alpha) \;=\; \min\left\{\frac{\lambda\alpha}{K\delta},\; \frac{\alpha\gamma}{\alpha\gamma+\delta}\right\}, 
\end{equation}
and
\begin{equation} \label{eq:alpha_wait}
    \mathbb{W}(\alpha) \;=\; \frac{\alpha\gamma+\delta}{(K\gamma\delta - \alpha\gamma\lambda - \delta\lambda)^{+}}.
\end{equation}

<!-- theorem-1: body_md -->
\label{thm:fixed_point}
    The average acceptance probability $\alpha^*$ is the unique solution to~\eqref{eq:alpha_def}, equivalently, the unique solution to
    \begin{equation} \label{eq:alpha_fp}
        \alpha \;=\; \frac{p_0}{1 + \theta\,\mathbb{W}(\alpha)},
    \end{equation}
    and is strictly less than $p_0$.
<!-- end theorem-1 -->
<!-- proposition-2: body_md -->
\label{prop:amplified_paradox}
    Relative to the constant-$p$ benchmark with $p=p_0$, wait-dependent acceptance shortens average waiting time but lowers slot utilization: 
    \begin{align*}
        \mathbb{W}(\alpha^*) \leq \mathbb{W}(p_0), \qquad U(\alpha^*) < U(p_0).
    \end{align*}

<!-- lemma-4: body_md -->
\label{lemma:alpha_cs}
    $\partial\alpha^*/\partial\lambda \leq 0$, $\partial\alpha^*/\partial\theta \leq 0$, and $\partial\alpha^*/\partial\gamma \geq 0$. Moreover, $\partial \mathbb{W}(\alpha^*)/\partial \gamma \leq 0$ and $\partial U(\alpha^*) / \partial \gamma \geq 0$.

<!-- theorem-2: body_md -->
\label{thm:alpha_nu}
    For a verification frequency $\nu \geq 0$ and $\phi \in (0,1]$, the average acceptance probability $\alpha^*(\nu)$ is the unique solution to 
    \begin{equation} \label{eq:alpha_nu}
        \alpha \;=\; \frac{p_0\,\bigl(1 + \phi\nu\,\mathbb{W}(\nu)\bigr)}{1 + (\theta + p_0 \phi \nu) \, \mathbb{W}(\nu)}, 
    \end{equation}
    where $\mathbb{W}(\nu) \in (0,\infty)$ is the unique solution to 
    \begin{align}\label{eq:capacity_const-alt}
        \frac{\lambda p_0}{1+\theta \mathbb{W}(\nu)} \left(\frac{1}{\delta} + \frac{1}{\gamma}\right) + \frac{\lambda (1-p_0+\theta \mathbb{W}(\nu))}{(1+\theta \mathbb{W}(\nu))(1+\phi\nu\mathbb{W}(\nu))} \cdot \frac{1}{\gamma} = K. 
    \end{align}
    Furthermore, $\mathbb{W}(\nu)$ is strictly decreasing in $\nu$, and $\alpha^*(\nu)$ is strictly increasing in $\nu$, with $\alpha^*(0) = \delta(K\gamma-\lambda)/\lambda\gamma$ and $\lim_{\nu\to\infty}\alpha^*(\nu) = 1$.
<!-- end theorem-2 -->
<!-- lemma-5: body_md -->
\label{lemma:reverify_composition}
    Fix a verification frequency $\nu \geq 0$ and $\phi \in (0,1]$. The expected numbers of active and phantom families on the waitlist are
    \begin{equation} \label{eq:AP_nu}
       Q_a(\nu) = \frac{\lambda p_0  \mathbb{W}(\nu)}{1+\theta \mathbb{W}(\nu)}, \qquad  Q_p(\nu) = \frac{\lambda \mathbb{W}(\nu) (1-p_0 + \theta \mathbb{W}(\nu))}{(1+\theta \mathbb{W}(\nu))(1+\phi\nu \mathbb{W}(\nu))},
    \end{equation}
    and the steady-state fraction of occupied licensed slots is 
    \begin{align}\label{eq:U}
        U(\nu) = \frac{\lambda p_0}{K\delta} \frac{1}{1 + \theta \, \mathbb{W}(\nu)}. 
    \end{align}
    Both $Q_a(\nu)$ and $Q_p(\nu)$ strictly decrease in verification frequency $\nu$, with $\lim_{\nu\to\infty} Q_p(\nu) = 0$, while $U(\nu)$ strictly increases in $\nu$.
<!-- end lemma-5 -->
<!-- proposition-3: body_md -->
\label{prop:optimal_review}
    An optimal verification frequency $\nu^*$ exists and is unique.
    \begin{enumerate}[label=(\roman*)]
        \item Verification is beneficial, that is, $\nu^*>0$, if and only if the verification process is sufficiently accurate, namely, 
            \begin{equation} \label{eq:review_threshold}
                \phi \;>\; \phi_{\min} \;:=\; \frac{c_\nu}{V\,(1-\alpha^*(0))},
            \end{equation}
            where $V \;:=\; s + (r+s\delta) / \gamma$ is the value of removing one phantom family from the waitlist (that is, the wasted offer it saves, plus the occupancy it releases, each additional occupied slot earning $r$ and averting $\delta$ wasted offers per unit time). Otherwise, $\nu^*=0$.
        \item The optimal verification frequency $\nu^*$ is higher when the wasted-offer cost $s$ increases, the idle-slot cost $r$ increases, and the per-contact cost $c_\nu$ decreases. These comparative statics are strict while $\nu^* > 0$. 
    \end{enumerate}

<!-- lemma-6: body_md -->
\label{lemma:transition}
    \quad
    \begin{enumerate}
        \item [(i)] For the infant room, the average acceptance probability $\alpha_I^*$ is given by Theorem~\ref{thm:fixed_point}. 
        
        \item [(ii)] For the toddler room, (a) a fraction $g/(g + \lambda_T\alpha_T^*)$ of openings are filled by children moving up with acceptance probability one; (b) for the remaining openings, the average acceptance probability among external offers  $\alpha_T^*$ is given by Theorem~\ref{thm:fixed_point}, with expected waiting time
        \begin{equation} \label{eq:wait_T}
            \mathbb{W}_T(\alpha) \;=\; \frac{\alpha\gamma + \delta_T}{\left(\gamma\,(K_T\delta_T - g) - \alpha\gamma\lambda_T - \delta_T \lambda_T\right)^{+}}.
        \end{equation}
        (c) The steady-state fraction of occupied licensed slots is 
        \begin{align}\label{eq:U_T}
            U_T = \frac{g + \lambda_T \alpha_T^*}{K_T \delta_T}.
        \end{align}
        As the move-up rate into the toddler classroom $g$ increases, its average acceptance probability $\alpha_T^*$ strictly decreases and slot utilization $U_T$ strictly increases.
    \end{enumerate}

<!-- lemma-7: body_md -->
\label{lemma:pooling_alpha}
    Let $f \in [0,\hat{f}]$. 
    \begin{enumerate}[label=(\roman*)]
        \item \textbf{(Capacity Relief)} The infant room is characterized by Lemma~\ref{lemma:transition} (i) with $K_I$ replaced by $K_I+f$. As $f$ increases, its average acceptance probability $\alpha_I^*(f)$ strictly increases, expected wait $\mathbb{W}_I(f)$ strictly decreases, and the expected number of occupied infant slots $(K_I+f)U_I(f)$ strictly increases. 
        \item \textbf{(Move-Up Flow)} The toddler room is characterized by Lemma~\ref{lemma:transition} (ii) with $g$ replaced by $g(f)$. If $\sigma>0$, as $f$ increases, its average acceptance probability $\alpha_T^*(f)$ strictly decreases, and the expected number of occupied toddler slots $K_T U_T(f)$ strictly increases. If $\sigma=0$, both are constant. 
    \end{enumerate}

<!-- proposition-4: body_md -->
\label{prop:optimal_pooling}
    $TC(f)$ is strictly convex on $f \in [0,\hat{f}]$, where it has a unique minimizer. A global minimizer $f^*$ of $TC(f)$ exists.
    \begin{enumerate}[label=(\roman*)]
        \item $f^* > 0$ requires the first flexible toddler slot pays for itself, namely, 
        \begin{align}\label{eq:pooling_threshold}
            \lambda_I\,\alpha_I^{*\prime}(0)\,V_I \;+\; \lambda_T\,\alpha_T^{*\prime}(0)\,V_T \;>\; C'(0),
        \end{align}
        where $V_I \;:=\; s \;+\; r/\delta_I \;+\; \sigma\,r/\delta_T$ and $V_T \;:=\; s \;+\; r/\delta_T$ are the values of enrolling one infant and one toddler, respectively. Otherwise, $f^*<0$.\footnote{Negative values of $f$ lend infant slots to the toddler classroom. Infants retain access to all $K_I$ slots, while the toddler classroom behaves as one with $K_T - f$ slots. The infant classroom can lend at most its mean number of idle slots, namely, $K_I - \lambda_I \alpha_I^*(0) / \delta_I$.}
        \item When $f^*>0$, the optimal conversion toward infants $f^*$ is higher when the wasted-offer cost $s$ increases, the idle-slot cost $r$ increases, and the marginal flexibility cost $C'$ decreases. 
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                             
    \end{enumerate}

<!-- lemma-8: body_md -->
\label{lemma:alpha-star}
\quad
\begin{enumerate}
    \item [(i)] If $\lambda<K\gamma$, then
    \begin{align*}
        0<\alpha^*<\bar{\alpha} :=\frac{\delta(K\gamma-\lambda)}{\lambda\gamma},
        \qquad
        \lambda<\mu_{\mathrm{eff}}(\alpha^*),
        \qquad
        \mathbb{W}(\alpha^*)<\infty.
    \end{align*}

    \item [(ii)] If $\lambda\geq K\gamma$, then
    \begin{align*}
        \alpha^*=0,
        \qquad
        \lambda\geq\mu_{\mathrm{eff}}(\alpha^*),
        \qquad
        \mathbb{W}(\alpha^*)=\infty.
    \end{align*}
\end{enumerate}
<!-- end lemma-8 -->
<!-- head-1: body_md -->
Let $\Phi$ denote the right-hand side of~\eqref{eq:alpha_def}, or equivalently, the right-hand side of~\eqref{eq:alpha_fp}. We know that $\mathbb{W}(\alpha) < \infty$ if $\lambda<\mu_{\mathrm{eff}}(\alpha)$, and $\mathbb{W} = \infty$ if $\lambda \geq \mu_{\mathrm{eff}}(\alpha)$. Since $\Phi(\alpha) = p_0/(1+\theta\mathbb{W}(\alpha))$ with $p_0 > 0$ and $\theta > 0$, it follows that 
\begin{align}\label{eq:Phi-two-cases}
    \Phi(\alpha)>0
    \ \mbox{ if } \
    \lambda<\mu_{\mathrm{eff}}(\alpha), 
    \quad
    \Phi(\alpha)=0
    \ \mbox{ if } \
    \lambda\geq\mu_{\mathrm{eff}}(\alpha). 
\end{align}


\textit{(i)} $\lambda<K\gamma$. Since $\mu_{\mathrm{eff}}(0)=K\gamma>\lambda$, we have $\Phi(0)>0$, so zero cannot be a fixed point and thus $\alpha^*>0$. The fixed-point equation $\Phi(\alpha^*) = \alpha^*$ then gives $\Phi(\alpha^*)>0$, which, by~\eqref{eq:Phi-two-cases}, implies
\begin{align*}
    \lambda<\mu_{\mathrm{eff}}(\alpha^*)
    =\frac{K\gamma\delta}{\alpha^*\gamma+\delta}.
\end{align*}
Rearranging gives 
\begin{align*}
    \alpha^* < \frac{\delta(K\gamma-\lambda)}{\lambda\gamma} =: \bar{\alpha},
\end{align*}
and Lemma~\ref{lemma:alpha-u-wait}(ii) yields $\mathbb{W}(\alpha^*)<\infty$.

\textit{(ii)} $\lambda\geq K\gamma$. For every $\alpha\geq0$,
\begin{align*}
    \mu_{\mathrm{eff}}(\alpha)
    =\frac{K\gamma\delta}{\alpha\gamma+\delta}
    \leq K\gamma\leq\lambda.
\end{align*}
Hence, by~\eqref{eq:Phi-two-cases}, $\Phi\equiv0$. The fixed-point equation $\Phi(\alpha^*) = \alpha^*$ then gives $\alpha^*=0$. Consequently, $\mu_{\mathrm{eff}}(\alpha^*)=\mu_{\mathrm{eff}}(0) = K\gamma\leq\lambda$ and $\mathbb{W}(\alpha^*)=\infty$.
\qed
