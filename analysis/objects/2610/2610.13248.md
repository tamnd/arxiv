<!-- definition-1: body_md -->
\label{def:regular}
Let $T>0$.  A triple $(\rho,u,b)$ is a regular solution of
\eqref{eq:system}--\eqref{eq:farfield} on $[0,T]$ if the equations hold in
distributions, $\rho(t,x)>0$ at every finite $x$,
$\inf_{[0,T]\times\mathbb R}\rho=0$, and the following properties hold.

\smallskip
\noindent\emph{Sublinear regime $0<\delta<1$.}
\begin{align}
 &0<\rho^{\gamma-1}\in C([0,T];H^2),\qquad
   (\rho^{\delta-1})_x\in C([0,T];H^1),\label{eq:regdensity}\\
 &u\in C([0,T];H^2)\cap L^2(0,T;H^3),\nonumber\\
 &u_t\in C([0,T];L^2)\cap L^2(0,T;H^1),\nonumber\\
 &\rho^{(\delta-1)/2}u_x\in C([0,T];L^2),\qquad
   \rho^{(\delta-1)/2}u_{tx}\in L^2(0,T;L^2),\nonumber\\
 &\rho^{\delta-1}u_{xx}\in C([0,T];L^2)
       \cap L^2(0,T;H^1).\label{eq:regvelocity}
\end{align}
\smallskip
\noindent\emph{Linear regime $\delta=1$.}
In this case \eqref{eq:regdensity} is replaced by
\begin{equation}\label{eq:regdeltaone}
 0<\rho^{\gamma-1}\in C([0,T];H^2),\qquad
 (\log\rho)_x\in C([0,T];H^1),
\end{equation}
and, written without abbreviating the time interval,
\begin{equation}\label{eq:regvelocity-linear}
 u\in C([0,T];H^2)\cap L^2(0,T;H^3),\qquad
 u_t\in C([0,T];L^2)\cap L^2(0,T;H^1).
\end{equation}

\smallskip
\noindent\emph{Magnetic regularity common to both regimes.}
With $h=b/\rho$ and $k=\rho^{-1}h_x$,
\begin{align}
 h&\in L^\infty((0,T)\times\mathbb R),\qquad
 k\in C([0,T];H^2),\label{eq:regmagnetic}\\
 b&\in C([0,T];H^2\cap L^1),\qquad
 b_t\in C([0,T];H^1).\nonumber
\end{align}

\label{thm:global-sublinear}
Let $A,\alpha>0$ and assume
\begin{equation}\label{eq:parameters-sublinear}
 0<\delta<1,\qquad \gamma>1,
 \qquad \gamma\geq\delta+\frac12.
\end{equation}
Suppose that the initial density and velocity satisfy
\begin{equation}\label{eq:datafluid-sublinear}
 0<\rho_0\in L^1(\mathbb R),\qquad
 \rho_0^{\gamma-1}\in H^2(\mathbb R),\qquad u_0\in H^2(\mathbb R),
\end{equation}
and that
\begin{equation}\label{eq:datadelta}
 (\rho_0^{\delta-1})_x\in H^1(\mathbb R),\qquad
 u_{0x}=\rho_0^{(1-\delta)/2}g_1,\qquad
 \alpha u_{0xx}=\rho_0^{1-\delta}g_2
\end{equation}
for some $g_1,g_2\in L^2(\mathbb R)$.  Write $b_0=\rho_0h_0$ and assume
\begin{equation}\label{eq:datamagnetic-sublinear}
 \begin{split}
 h_0&\in L^\infty(\mathbb R;\mathbb R^m)
       \cap W^{1,1}_{\mathrm{loc}}(\mathbb R;\mathbb R^m),\\
 k_0&:=\rho_0^{-1}(h_0)_x\in H^2(\mathbb R;\mathbb R^m),
 \qquad \operatorname{supp}k_0\Subset\mathbb R.
 \end{split}
\end{equation}
The identity $(h_0)_x=\rho_0k_0$ in
\eqref{eq:datamagnetic-sublinear} is understood distributionally.  Because
$h_0$ is locally absolutely continuous and $\rho_0$ is positive at every
finite point, the quotient defining $k_0$ is well defined almost
everywhere.

Then the Cauchy problem \eqref{eq:system}--\eqref{eq:farfield}, with
$(\rho,u,b)|_{t=0}=(\rho_0,u_0,b_0)$, has a unique global classical
solution on $(0,\infty)\times\mathbb R$.  For each finite $T>0$ this solution has
the sublinear regularity in \cref{def:regular} and satisfies
\begin{align}
 &\rho\in C([0,T];L^1),
 \qquad (\rho^{\gamma-1})_t\in C([0,T];H^1),
 \label{eq:extra-sublinear}\\
 &h(t,X(t;x))=h_0(x),\qquad
 k(t,X(t;x))=k_0(x),
 \label{eq:transport-sublinear}
\end{align}
where the particle path is determined by
\begin{equation}\label{eq:flow-sublinear}
 X_t(t;x)=u(t,X(t;x)),\qquad X(0;x)=x.
\end{equation}
Moreover,
\begin{align}
 &t^{1/2}\rho^{(\delta-1)/2}u_{tx}
       \in L^\infty(0,T;L^2),
 \label{eq:timeweighted-sublinear}\\
 &t^{1/2}u_{txx}\in L^2(0,T;L^2),\qquad
 t^{1/2}u_{tt}\in L^2(0,T;L^2).\nonumber
\end{align}
No smallness assumption is imposed on any of the norms above.

<!-- lemma-1: body_md -->
\label{lem:commute}
Let $D_t=\partial_t+u\partial_x$ and $\mathcal D=\rho^{-1}\partial_x$.  If
$\rho_t+(\rho u)_x=0$, then
\begin{equation}\label{eq:commutator}
 [D_t,\mathcal D]f=0
\end{equation}
for every smooth scalar or vector function $f$.  If also
$b_t+(ub)_x=0$, then
\begin{equation}\label{eq:transporthk}
 D_t h=0,\qquad D_t k=0,\qquad h=b/\rho,\quad k=\mathcal D h.
\end{equation}
Consequently,
\begin{equation}\label{eq:hkmax}
 \lVert h(t)\rVert_{L^\infty}=\lVert h_0\rVert_{L^\infty},\qquad
 \lVert k(t)\rVert_{L^\infty}=\lVert k_0\rVert_{L^\infty}.
\end{equation}

<!-- proof-1: body_md -->
Expanding the divergence in the continuity equation gives
\begin{equation}\label{eq:materialrho}
 D_t\rho=\rho_t+u\rho_x=-\rho u_x,
 \qquad D_t(\rho^{-1})=\rho^{-1}u_x.
\end{equation}
The derivative $\partial_x$ does not commute with $D_t$.  Indeed, the
ordinary product rule gives
\begin{equation}\label{eq:materialfx}
 D_t f_x=f_{xt}+uf_{xx}=(D_t f)_x-u_xf_x.
\end{equation}
Combining \eqref{eq:materialrho} and \eqref{eq:materialfx}, we obtain
\begin{align*}
 D_t(\rho^{-1}f_x)
 &=D_t(\rho^{-1})f_x+\rho^{-1}D_t f_x\\
 &=\rho^{-1}u_xf_x
   +\rho^{-1}\bigl((D_t f)_x-u_xf_x\bigr)
 =\rho^{-1}(D_t f)_x=\mathcal D(D_t f).
\end{align*}
This is exactly the commutator identity \eqref{eq:commutator}.

The induction equation can be written
\begin{equation}\label{eq:materialb}
 D_t b=-bu_x.
\end{equation}
Since $h=b/\rho$, another application of the quotient rule gives the
cancellation
\[
 D_t h=\frac{D_t b}{\rho}-\frac b{\rho^2}D_t\rho
 =-\frac b\rho u_x+\frac b\rho u_x=0.
\]
Because $k=\mathcal D h$, the already proved commutator yields
$D_t k=\mathcal D(D_t h)=0$.

For completeness, let $X$ be the particle map
$X_t(t;a)=u(t,X(t;a))$, $X(0;a)=a$.  Differentiating it with respect to
$a$ gives
\begin{equation}\label{eq:flowjacobian}
 X_a(t;a)=\exp\left(\int_0^t
 u_x(s,X(s;a))\,\mathop{}\!\mathrm d s\right)>0.
\end{equation}
The two transport equations therefore imply
\begin{equation}\label{eq:hkcharacteristics}
 h(t,X(t;a))=h_0(a),\qquad k(t,X(t;a))=k_0(a).
\end{equation}
Taking suprema proves \eqref{eq:hkmax}.  Notice that this argument preserves
the $L^\infty$ norms, but not the $H^2$ norm of $k$; deformation of the
flow is accounted for separately in \cref{lem:kH2}.

<!-- lemma-2: body_md -->
\label{lem:kH2}
For every $T>0$,
\begin{equation}\label{eq:kH2bound}
 \sup_{0\leq t\leq T}\lVert k(t)\rVert_{H^2}
 \leq \lVert k_0\rVert_{H^2}
 \exp\left\{C\int_0^T
 \bigl(\lVert u_x\rVert_{L^\infty}+\lVert u_{xx}\rVert_{L^2}\bigr)\,\mathop{}\!\mathrm d t\right\}.
\end{equation}
Moreover, $\operatorname{supp}k(t)$ is the image of
$\operatorname{supp}k_0$ under the flow.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Starting from $k_t+uk_x=0$, differentiate once and twice in $x$:
\begin{align}
 (k_x)_t+u k_{xx}+u_xk_x&=0,
 \label{eq:kxtransport}\\
 (k_{xx})_t+u k_{xxx}+2u_xk_{xx}+u_{xx}k_x&=0.
 \label{eq:kxxtransport}
\end{align}
Taking the $L^2$ inner products of these three equations with
$k,k_x,k_{xx}$, respectively, and using
$-\int u f_x\cdot f=\frac12\int u_x|f|^2$, gives
\begin{align}
 \frac12\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\lVert k\rVert_{L^2}^2
 &=\frac12\int_\mathbb R u_x|k|^2\,\mathop{}\!\mathrm d x,\label{eq:k0energy}\\
 \frac12\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\lVert k_x\rVert_{L^2}^2
 &=-\frac12\int_\mathbb R u_x|k_x|^2\,\mathop{}\!\mathrm d x,\label{eq:k1energy}\\
 \frac12\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\lVert k_{xx}\rVert_{L^2}^2
 &=-\frac32\int_\mathbb R u_x|k_{xx}|^2\,\mathop{}\!\mathrm d x
   -\int_\mathbb R u_{xx}k_x\cdot k_{xx}\,\mathop{}\!\mathrm d x.\label{eq:k2energy}
\end{align}
For the last term in \eqref{eq:k2energy}, H\"older's inequality and the
one-dimensional embedding $H^1(\mathbb R)\hookrightarrow L^\infty(\mathbb R)$ give
\[
 \left|\int u_{xx}k_x\cdot k_{xx}\right|
 \leq\lVert u_{xx}\rVert_{L^2}\lVert k_x\rVert_{L^\infty}
       \lVert k_{xx}\rVert_{L^2}
 \leq C\lVert u_{xx}\rVert_{L^2}\lVert k\rVert_{H^2}^2.
\]
The other terms are bounded by
$\lVert u_x\rVert_{L^\infty}\lVert k\rVert_{H^2}^2$.  Adding the three identities yields
\[
 \frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\lVert k\rVert_{H^2}^2
 \leq C\bigl(\lVert u_x\rVert_{L^\infty}+\lVert u_{xx}\rVert_{L^2}\bigr)
       \lVert k\rVert_{H^2}^2.
\]
Integrating this scalar differential inequality and applying Gronwall's
lemma proves \eqref{eq:kH2bound}.  The support statement follows from the
transport formulas in \cref{lem:commute}: a characteristic that starts
outside $\operatorname{supp}k_0$ carries the value zero for all later times.

<!-- lemma-5: body_md -->
\label{lem:energy}
Every regular solution satisfies
\begin{equation}\label{eq:mass}
 \int_\mathbb R\rho(t,x)\,\mathop{}\!\mathrm d x=M_0:=\int_\mathbb R\rho_0\,\mathop{}\!\mathrm d x
\end{equation}
and
\begin{equation}\label{eq:energy}
 E(t)+\alpha\int_0^t\!\int_\mathbb R\rho^\delta u_x^2\,\mathop{}\!\mathrm d x\mathop{}\!\mathrm d s=E(0),
\end{equation}
where
\begin{equation}\label{eq:Edef}
 E(t)=\int_\mathbb R\left(\frac12\rho u^2
 +\frac{A}{\gamma-1}\rho^\gamma+\frac12|b|^2\right)\mathop{}\!\mathrm d x.
\end{equation}
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
We first perform the calculation for a smooth solution that decays rapidly;
the cutoff argument at the end makes it valid for a regular solution.  Put
\[
 p=A\rho^\gamma,\qquad e=\frac{A}{\gamma-1}\rho^\gamma,
 \qquad B=\frac12|b|^2.
\]
The continuity equation and the chain rule give the internal-energy identity
\begin{equation}\label{eq:internal-local}
 e_t+(ue)_x+pu_x=0.
\end{equation}
Indeed, after expanding $(ue)_x$, its left-hand side is
$\frac{A\gamma}{\gamma-1}\rho^{\gamma-1}
(\rho_t+u\rho_x)+\frac{A\gamma}{\gamma-1}\rho^\gamma u_x$,
which vanishes because $D_t\rho=-\rho u_x$.

Writing the induction equation as $b_t+ub_x+u_xb=0$ and taking its scalar
product with $b$ gives
\begin{equation}\label{eq:magnetic-local}
 B_t+(uB)_x+Bu_x=0.
\end{equation}
Next multiply the momentum equation by $u$.  The continuity equation turns
the first two terms into a kinetic-energy derivative:
\begin{equation}\label{eq:kinetic-local}
 \left(\frac12\rho u^2\right)_t
 +\left(\frac12\rho u^3\right)_x
 +up_x+uB_x=u(\alpha\rho^\delta u_x)_x.
\end{equation}
Use
\[
 up_x=(up)_x-pu_x,\quad uB_x=(uB)_x-Bu_x,\quad
 u(\alpha\rho^\delta u_x)_x
 =(\alpha\rho^\delta uu_x)_x-\alpha\rho^\delta u_x^2.
\]
Adding \eqref{eq:internal-local}--\eqref{eq:kinetic-local} cancels both
$pu_x$ and $Bu_x$ and yields the local identity
\begin{align}
 &\partial_t\left(\frac12\rho u^2+e+B\right)
 +\alpha\rho^\delta u_x^2+\partial_x\left[
 \frac12\rho u^3+u(e+p)+2uB-\alpha\rho^\delta uu_x
 \right]=0.\label{eq:total-local}
\end{align}
This line displays explicitly why the magnetic work from momentum is
cancelled by the magnetic energy from induction.

Let $\chi_R(x)=\chi(x/R)$, where $0\leq\chi\leq1$, $\chi=1$ on
$[-1,1]$, and $\chi=0$ outside $[-2,2]$.  Integrating continuity and
\eqref{eq:total-local} against $\chi_R$ leaves flux errors supported in
$R\leq|x|\leq2R$ and multiplied by $\lVert \chi_R'\rVert_{L^\infty}\leq C/R$.
The regularity in \cref{def:regular}, followed by approximation if needed,
makes those errors tend to zero as $R\to\infty$.  We obtain
\eqref{eq:mass} and \eqref{eq:energy}.

<!-- lemma-6: body_md -->
\label{lem:densityupper}
There is a constant $C_0$, depending only on $E(0)$, $M_0$, $\alpha$,
$\delta$, and $\lVert \rho_0\rVert_{L^\infty}$, such that
\begin{equation}\label{eq:rhoinfty}
 0<\rho(t,x)\leq C_0,\qquad (t,x)\in[0,\infty)\times\mathbb R.
\end{equation}
<!-- end lemma-6 -->
<!-- proof-6: body_md -->
Set $\xi(t,x)=\int_{-\infty}^x\rho u(t,y)\,\mathop{}\!\mathrm d y$.  The momentum
fluxes vanish at the left far field.  Integration from $-\infty$ to $x$
therefore gives
\begin{equation}\label{eq:xit}
 \xi_t=-\rho u^2-A\rho^\gamma-\frac12|b|^2
       +\alpha\rho^\delta u_x.
\end{equation}
Since $\xi_x=\rho u$, adding $u\xi_x=\rho u^2$ to
\eqref{eq:xit} cancels the convective term:
\begin{equation}\label{eq:Dtxi}
 D_t\xi=\alpha\rho^\delta u_x-A\rho^\gamma-\frac12|b|^2.
\end{equation}
On the other hand, \eqref{eq:materialrho} and the chain rule yield
\begin{equation}\label{eq:Dtrhodelta}
 D_t\left(\frac\alpha\delta\rho^\delta\right)
 =\alpha\rho^{\delta-1}D_t\rho=-\alpha\rho^\delta u_x.
\end{equation}
Adding \eqref{eq:Dtxi} and \eqref{eq:Dtrhodelta} proves
\eqref{eq:densityintro}.  If $X(t;x)$ is the particle that starts at $x$,
integration of this nonpositive derivative from $0$ to $t$ gives
\[
 \frac\alpha\delta\rho^\delta(t,X(t;x))
 \leq \frac\alpha\delta\rho_0^\delta(x)
      +|\xi_0(x)|+|\xi(t,X(t;x))|.
\]
For every $x$, Cauchy--Schwarz, mass conservation, and energy give
\[
 |\xi(t,x)|\leq\int_\mathbb R\rho|u|\,\mathop{}\!\mathrm d y
 \leq M_0^{1/2}\lVert \sqrt\rho\,u(t)\rVert_{L^2}
 \leq (2M_0E(0))^{1/2}.
\]
The same estimate holds at $t=0$.  Taking the $\delta$th root proves
\eqref{eq:rhoinfty}, with a constant independent of time.  No estimate of
$|b|^2$ was needed in this argument: magnetic pressure appears with a minus
sign in \eqref{eq:densityintro} and can only improve the upper bound.

<!-- lemma-7: body_md -->
\label{lem:moments}
For every $p\geq0$ and $T>0$,
\begin{equation}\label{eq:momentbound}
 \sup_{0\leq t\leq T}\int_\mathbb R\rho|u|^{p+2}\,\mathop{}\!\mathrm d x
 +\int_0^T\!\int_\mathbb R\rho^\delta|u|^p u_x^2\,\mathop{}\!\mathrm d x\mathop{}\!\mathrm d t
 \leq C_{T,p}.
\end{equation}
<!-- end lemma-7 -->
<!-- proof-7: body_md -->
We first take $p$ so that the multiplier is smooth; a standard regularization
of $|u|^p u$ removes this harmless restriction.  Multiply the conservative
momentum equation by $|u|^pu$ and integrate.  The continuity equation
combines the time and convection terms, while
$(|u|^pu)_x=(p+1)|u|^pu_x$.  Thus
\begin{align}
 &\frac1{p+2}\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}
    \int_\mathbb R\rho|u|^{p+2}\,\mathop{}\!\mathrm d x
 +\alpha(p+1)\int_\mathbb R\rho^\delta|u|^pu_x^2\,\mathop{}\!\mathrm d x
=(p+1)\int_\mathbb R
 \left(A\rho^\gamma+\frac12|b|^2\right)|u|^pu_x\,\mathop{}\!\mathrm d x.
 \label{eq:momentidentity}
\end{align}
Young's inequality with the dissipative factor
$\rho^{\delta/2}|u|^{p/2}u_x$ gives
\begin{align}
 \left|\int_\mathbb R A\rho^\gamma|u|^pu_x\,\mathop{}\!\mathrm d x\right|
 &\leq\varepsilon\int_\mathbb R\rho^\delta|u|^pu_x^2\,\mathop{}\!\mathrm d x
 +C_{\varepsilon,p}\int_\mathbb R\rho^{2\gamma-\delta}|u|^p\,\mathop{}\!\mathrm d x,
 \label{eq:momentpressure}\\
 \left|\int_\mathbb R\frac{|b|^2}{2}|u|^pu_x\,\mathop{}\!\mathrm d x\right|
 &\leq\varepsilon\int_\mathbb R\rho^\delta|u|^pu_x^2\,\mathop{}\!\mathrm d x
 +C_{\varepsilon,p}\int_\mathbb R\rho^{4-\delta}|u|^p\,\mathop{}\!\mathrm d x.
 \label{eq:momentmag}
\end{align}
Here we used $|b|=\rho|h|\leq\lVert h_0\rVert_{L^\infty}\rho$.

We spell out the last integrability step.  If $q>1$, split the line into
$\{|u|\leq1\}$ and $\{|u|>1\}$.  The density upper bound implies
\[
 \rho^q|u|^p
 \leq C\rho\quad\hbox{on }\{|u|\leq1\},
 \qquad
 \rho^q|u|^p
 \leq C\rho|u|^{p+2}\quad\hbox{on }\{|u|>1\}.
\]
Consequently,
\begin{equation}\label{eq:momentlower}
 \int_\mathbb R\rho^q|u|^p\,\mathop{}\!\mathrm d x
 \leq C\left(M_0+\int_\mathbb R\rho|u|^{p+2}\,\mathop{}\!\mathrm d x\right).
\end{equation}
Both exponents $2\gamma-\delta$ and $4-\delta$ are larger than one.
Choose $\varepsilon$ small enough to absorb the two dissipative terms in
\eqref{eq:momentidentity}, use \eqref{eq:momentlower}, and apply Gronwall's
lemma.  The initial moment is finite because $u_0\in H^2(\mathbb R)$ is bounded
and $\rho_0\in L^1$.  This proves \eqref{eq:momentbound} for every $p\geq0$.

<!-- lemma-8: body_md -->
\label{lem:BD}
Let $v$ be given by \eqref{eq:vintro}.  Then
\begin{align}
 &\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\int_\mathbb R
 \left(\frac12\rho|v|^2+\frac{A}{\gamma-1}\rho^\gamma
       +\frac12\rho^2|h|^2\right)\mathop{}\!\mathrm d x
 +\alpha A\gamma\int_\mathbb R\rho^{\gamma+\delta-3}\rho_x^2\,\mathop{}\!\mathrm d x+\alpha\int_\mathbb R |h|^2\rho^{\delta-1}\rho_x^2\,\mathop{}\!\mathrm d x
 \nonumber\\
 &=-\alpha\int_\mathbb R\rho^{\delta+1}(h\cdot k)\rho_x\,\mathop{}\!\mathrm d x.
 \label{eq:BDidentity}
\end{align}
Consequently, for every $T>0$,
\begin{align}
 &\sup_{0\leq t\leq T}\int_\mathbb R
 \left(\rho|v|^2+\rho^\gamma+|b|^2\right)\mathop{}\!\mathrm d x
 +\int_0^T\!\int_\mathbb R\rho^{\gamma+\delta-3}\rho_x^2\,\mathop{}\!\mathrm d x\mathop{}\!\mathrm d t
 +\int_0^T\!\int_\mathbb R |h|^2\rho^{\delta-1}\rho_x^2\,\mathop{}\!\mathrm d x\mathop{}\!\mathrm d t
 \leq C_T.
 \label{eq:BDbound}
\end{align}
<!-- end lemma-8 -->
<!-- proof-8: body_md -->
Put $w=\alpha\rho^{\delta-2}\rho_x$, so that $v=u+w$.  We first verify the
viscous cancellation instead of treating it as a formal identity.  From
\eqref{eq:materialrho} and \eqref{eq:materialfx},
\begin{equation}\label{eq:Dtrhox}
 D_t\rho_x=(D_t\rho)_x-u_x\rho_x
 =-\rho u_{xx}-2u_x\rho_x.
\end{equation}
Therefore
\begin{align*}
 \rhoD_t w
 &=\alpha\rhoD_t(\rho^{\delta-2}\rho_x)=-\alpha\rho^\delta u_{xx}
   -\alpha\delta\rho^{\delta-1}\rho_xu_x
 =-(\alpha\rho^\delta u_x)_x.
\end{align*}
In other words, continuity implies the exact identity
\begin{equation}\label{eq:visccancel}
 \rhoD_t(\alpha\rho^{\delta-2}\rho_x)
 =-(\alpha\rho^\delta u_x)_x.
\end{equation}
The nonconservative momentum equation is
$\rhoD_t u+(A\rho^\gamma+|b|^2/2)_x
=(\alpha\rho^\delta u_x)_x$.  Adding \eqref{eq:visccancel} cancels its
right-hand side and gives \eqref{eq:vrawintro}.

Multiply that equation by $v=u+w$ and integrate.  Continuity gives
\[
 \int_\mathbb R\rho v\cdotD_t v\,\mathop{}\!\mathrm d x
 =\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\int_\mathbb R\frac12\rho|v|^2\,\mathop{}\!\mathrm d x.
\]
The $u$ part of the pressure work is the positive time derivative of the
internal and magnetic energies, by \eqref{eq:internal-local} and
\eqref{eq:magnetic-local}.  For the $w$ part, integration by parts gives
\begin{align*}
 \int_\mathbb R\left(A\rho^\gamma+\frac12|b|^2\right)w_x\,\mathop{}\!\mathrm d x
 ={}&-\alpha A\gamma\int_\mathbb R
 \rho^{\gamma+\delta-3}\rho_x^2\,\mathop{}\!\mathrm d x\\
 &-\alpha\int_\mathbb R\rho^{\delta-2}\rho_x\,b\cdot b_x\,\mathop{}\!\mathrm d x.
\end{align*}
Combining these identities yields
\begin{align*}
 \frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\int_\mathbb R
 \left(\frac12\rho v^2+\frac{A}{\gamma-1}\rho^\gamma
 +\frac12|b|^2\right)\mathop{}\!\mathrm d x
 ={}&-\alpha A\gamma\int_\mathbb R
 \rho^{\gamma+\delta-3}\rho_x^2\,\mathop{}\!\mathrm d x
 -\alpha\int_\mathbb R\rho^{\delta-2}\rho_x\,b\cdot b_x\,\mathop{}\!\mathrm d x.
\end{align*}
To understand the magnetic term, use $b=\rho h$ and $h_x=\rho k$:
\begin{equation}\label{eq:bdmagneticexpand}
 b\cdot b_x=\rho|h|^2\rho_x+\rho^3h\cdot k.
\end{equation}
Thus
\[
 \rho^{\delta-2}\rho_x\,b\cdot b_x
 =|h|^2\rho^{\delta-1}\rho_x^2
  +\rho^{\delta+1}(h\cdot k)\rho_x,
\]
which converts the last integral into the final two terms in
\eqref{eq:BDidentity}.  The second one has no sign, but Young's inequality
gives
\begin{equation}\label{eq:BDyoung}
 \rho^{\delta+1}|h|\,|k|\,|\rho_x|
 \leq\frac12|h|^2\rho^{\delta-1}\rho_x^2
      +\frac12\rho^{\delta+3}|k|^2.
\end{equation}
The error on the right is uniformly bounded because
\begin{equation}\label{eq:bdremainder}
 \int_\mathbb R\rho^{\delta+3}|k|^2\,\mathop{}\!\mathrm d x
 \leq\lVert k_0\rVert_{L^\infty}^2
       \lVert \rho\rVert_{L^\infty}^{\delta+2}M_0.
\end{equation}
Absorb one half of the magnetic gradient term, integrate from $0$ to $t$,
and use \eqref{eq:rhoinfty}.  The initial effective-velocity energy is
finite by the corresponding assumptions on
$(\rho_0^{\delta-1})_x$ or $(\log\rho_0)_x$.  This proves
\eqref{eq:BDbound}.  In particular, the cross term is controlled; it is not
being discarded or assigned a favorable sign.

<!-- lemma-9: body_md -->
\label{lem:weightedpointwise}
For every finite $T>0$, the following estimates hold:
\begin{align}
 &\int_0^T\lVert \rho^{1/2}u\rVert_{L^\infty}\,\mathop{}\!\mathrm d t\leq C_T,
 &&0<\delta<1,\label{eq:rhouint-sublinear}\\
 &\int_0^T\lVert \rho^\iota u\rVert_{L^\infty}\,\mathop{}\!\mathrm d t\leq C_{T,\iota},
 &&\delta=1,\quad \frac12<\iota<1.
 \label{eq:rhouint-linear}
\end{align}
<!-- end lemma-9 -->
<!-- proof-9: body_md -->
We prove the two cases separately because they use different spatial
exponents.

\smallskip
\noindent\emph{Case $0<\delta<1$.}
Let $F=\rho^{1/2}u$ and set $p_\delta=2/(2-\delta)$.  A useful factorization
of its derivative is
\begin{equation}\label{eq:Fx-sublinear}
 F_x=\frac12
 \bigl(\rho^{\delta-3/2}\rho_x\bigr)
 \bigl(\rho^{1-\delta}u\bigr)
 +\rho^{(1-\delta)/2}\bigl(\rho^{\delta/2}u_x\bigr).
\end{equation}
The effective-velocity and basic energies imply
\begin{equation}\label{eq:rhograd-weighted}
 \lVert \rho^{\delta-3/2}\rho_x\rVert_{L^2}
 =\alpha^{-1}\lVert \sqrt\rho\,(v-u)\rVert_{L^2}\leq C_T.
\end{equation}
In H\"older's inequality choose the exponents $2$ and
$2/(1-\delta)$.  Since
\[
 \int_\mathbb R|\rho^{1-\delta}u|^{2/(1-\delta)}\,\mathop{}\!\mathrm d x
 =\int_\mathbb R\rho^2|u|^{2/(1-\delta)}\,\mathop{}\!\mathrm d x,
\]
the first factor is controlled by \cref{lem:moments} and the density upper
bound.  The second product in \eqref{eq:Fx-sublinear} is controlled because
$\rho^{(1-\delta)/2}\in L^{2/(1-\delta)}$ by mass conservation and
$\rho^{\delta/2}u_x\in L^2(0,T;L^2)$ by energy.  Hence
\begin{equation}\label{eq:Fx-sublinear-bound}
 \int_0^T\lVert F_x\rVert_{L^{p_\delta}}^2\,\mathop{}\!\mathrm d t\leq C_T.
\end{equation}
The one-dimensional Gagliardo--Nirenberg inequality reads
\begin{equation}\label{eq:GN-sublinear}
 \lVert F\rVert_{L^\infty}
 \leq C\lVert F\rVert_{L^2}^{1-\theta_\delta}
       \lVert F_x\rVert_{L^{p_\delta}}^{\theta_\delta},
 \qquad \theta_\delta=\frac1{1+\delta}.
\end{equation}
The $L^2$ norm of $F$ is bounded by the kinetic energy.  Integrating
\eqref{eq:GN-sublinear} in time and using
\eqref{eq:Fx-sublinear-bound} proves \eqref{eq:rhouint-sublinear}.

\smallskip
\noindent\emph{Case $\delta=1$.}
Fix $1/2<\iota<1$ and put $F=\rho^\iota u$.  This time
\begin{equation}\label{eq:Fx-linear}
 F_x=\iota(\rho^{-1/2}\rho_x)
       (\rho^{\iota-1/2}u)
 +\rho^{\iota-1/2}(\rho^{1/2}u_x).
\end{equation}
By the same effective-velocity estimate,
$\lVert \rho^{-1/2}\rho_x\rVert_{L^2}\leq C_T$.  Use H\"older with exponents
$2$ and $2/(2\iota-1)$.  The identities
\[
 \int_\mathbb R|\rho^{\iota-1/2}u|^{2/(2\iota-1)}\,\mathop{}\!\mathrm d x
 =\int_\mathbb R\rho|u|^{2/(2\iota-1)}\,\mathop{}\!\mathrm d x,
 \qquad
 \int_\mathbb R|\rho^{\iota-1/2}|^{2/(2\iota-1)}\,\mathop{}\!\mathrm d x=M_0
\]
and \cref{lem:moments} show that
\begin{equation}\label{eq:Fx-linear-bound}
 \int_0^T\lVert F_x\rVert_{L^{1/\iota}}^2\,\mathop{}\!\mathrm d t\leq C_{T,\iota}.
\end{equation}
Moreover, $\lVert F\rVert_{L^2}$ is controlled by kinetic energy and
\eqref{eq:rhoinfty}, since $2\iota-1>0$.  The corresponding
Gagliardo--Nirenberg inequality is
\[
 \lVert F\rVert_{L^\infty}
 \leq C\lVert F\rVert_{L^2}^{1-\theta_\iota}
       \lVert F_x\rVert_{L^{1/\iota}}^{\theta_\iota},
 \qquad \theta_\iota=\frac1{3-2\iota}.
\]
Time integration proves \eqref{eq:rhouint-linear}.

\label{lem:vinfty}
Under the parameter restrictions of \cref{thm:global-sublinear} or
\cref{thm:global-linear}, for every $T>0$,
\begin{equation}\label{eq:vinfty}
 \sup_{0\leq t\leq T}\lVert v(t)\rVert_{L^\infty}\leq C_T.
\end{equation}
<!-- end lemma-10 -->
<!-- proof-10: body_md -->
Set
\[
 a=\frac{A\gamma}{\alpha}\rho^{\gamma-\delta}
   +\frac{|h|^2}{\alpha}\rho^{2-\delta},
 \qquad G=\rho^2h\cdot k.
\]
Along $X(t;x)$, equation \eqref{eq:vequation} becomes the ordinary

\[
 \frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}V(t)+a(t,X(t;x))V(t)
 =a(t,X(t;x))U(t)-G(t,X(t;x)),
\]
where $V(t)=v(t,X(t;x))$ and $U(t)=u(t,X(t;x))$.  Its integrating-factor
formula is
\begin{align}
 V(t)={}&e^{-\int_0^t a(\tau,X(\tau;x))\,\mathop{}\!\mathrm d\tau}v_0(x) +\int_0^t e^{-\int_s^t a(\tau,X(\tau;x))\,\mathop{}\!\mathrm d\tau}
       (aU-G)(s,X(s;x))\,\mathop{}\!\mathrm d s.
 \label{eq:vvariation}
\end{align}
Because $a\geq0$, both exponential factors are at most one.

Suppose first that $0<\delta<1$.  With $\iota=1/2$,
\begin{equation}\label{eq:au-sublinear}
 a|u|\leq C\left(
 \rho^{\gamma-\delta-1/2}+\rho^{3/2-\delta}
 \right)|\rho^{1/2}u|.
\end{equation}
The two displayed density powers are nonnegative precisely because
$\gamma\geq\delta+1/2$ and $\delta<1$.  Thus the density upper bound and
\eqref{eq:rhouint-sublinear} make $a|u|$ integrable along every path,
uniformly in its starting point.

When $\delta=1$, choose
\begin{equation}\label{eq:iotachoice}
 \frac12<\iota<\min\{1,\gamma-1\}.
\end{equation}
Such a number exists exactly because $\gamma>3/2$.  Now
\begin{equation}\label{eq:au-linear}
 a|u|\leq C\left(
 \rho^{\gamma-1-\iota}+\rho^{1-\iota}
 \right)|\rho^\iota u|,
\end{equation}
and both powers are again nonnegative.  Use
\eqref{eq:rhouint-linear}.

Finally, in both regimes,
\[
 |G|\leq\lVert \rho\rVert_{L^\infty}^2
       \lVert h_0\rVert_{L^\infty}\lVert k_0\rVert_{L^\infty}\leq C_0.
\]
Also $v_0\in L^\infty$ because $u_0\in H^2$ and
$v_0-u_0=(\alpha/\delta)\psi_0$ with $\psi_0\in H^1$.  Taking the supremum
over starting points in \eqref{eq:vvariation} proves
\eqref{eq:vinfty}.  The calculation also shows transparently why the two
different lower bounds on $\gamma$ occur in the two main theorems.

Let $\mathcal M$ be defined by \eqref{eq:Mrepresentation} and put
\begin{equation}\label{eq:HG}
 H=\rho|h|^2,\qquad G=\rho^2h\cdot k.
\end{equation}
Then

 \mathcal M=H_x-G,\qquad D_t G=-2u_xG.
\end{equation}
Before $\rho_{xx}$ is known, the time-differentiated momentum estimate can

 \int_\mathbb R\mathcal M_tu_t\,\mathop{}\!\mathrm d x
 =-\int_\mathbb R H_tu_{tx}\,\mathop{}\!\mathrm d x-\int_\mathbb R G_tu_t\,\mathop{}\!\mathrm d x.
\end{equation}
On every interval on which the low-order fluid estimates and
$\lVert k\rVert_{H^2}$ are finite,
\begin{equation}\label{eq:HGtime}

 \leq C_T(1+\lVert u\rVert_{H^1}),

After $\psi_x\in L^2$ is known,
\begin{equation}\label{eq:Mxpsi}
 \lVert \mathcal M_x\rVert_{L^2}
 \leq C_T\bigl(1+\lVert \psi_x\rVert_{L^2}+\lVert k\rVert_{H^1}\bigr).
\end{equation}

Let $(\rho,u,b)$ be a regular solution on $[0,T]$ satisfying the assumptions

<!-- proof-12: body_md -->
We begin by deriving the equations used in the estimates.  Since
$\phi=C_\gamma\rho^{\gamma-1}$ with
$C_\gamma=A\gamma/(\gamma-1)$, equation \eqref{eq:materialrho} gives
$D_t\phi=-(\gamma-1)\phi u_x$.  Next,
$\psi=\delta\rho^{\delta-2}\rho_x$ and \eqref{eq:Dtrhox} give
\[
 D_t\psi=-\delta\psi u_x-\delta\rho^{\delta-1}u_{xx}.
\]
Finally, divide the momentum equation by $\rho$.  The identities
\[
 \rho^{-1}(A\rho^\gamma)_x=\phi_x,\qquad
 \rho^{-1}(\alpha\rho^\delta u_x)_x
 =\alpha\rho^{\delta-1}u_{xx}+\alpha\psi u_x
\]
and the definition of $\mathcal M$ give the third equation below.  Thus the
exact reformulated system is
\begin{align}
 \phi_t+u\phi_x+(\gamma-1)\phi u_x&=0,\label{eq:phieq}\\
 \psi_t+(u\psi)_x+(\delta-1)\psi u_x
 +\delta\rho^{\delta-1}u_{xx}&=0,\label{eq:psieq}\\
 u_t+uu_x+\phi_x
 &=\alpha\rho^{\delta-1}u_{xx}+\alpha\psi u_x-\mathcal M.
 \label{eq:ueq}
\end{align}
For $\delta=1$, \eqref{eq:psieq} reduces to
$\psi_t+(u\psi)_x+u_{xx}=0$.  The remainder of the proof is ordered so that
each derivative is estimated only after all quantities on its right-hand
side are known.  The purely hydrodynamic product estimates coincide with
\cite[Lemmas 3.8--3.13]{CaoLiZhu2022}; below we record their resulting
inequalities and derive every additional magnetic estimate explicitly.

\emph{Step 1: unweighted and first-order velocity estimates.}
By \cref{lem:Mlow,lem:vinfty},
\begin{equation}\label{eq:Mfirst}
 \sup_{0\leq t\leq T}\lVert \mathcal M(t)\rVert_{L^2}\leq C_T.
\end{equation}
For clarity, we write the first two energy calculations.  Expanding the
viscous derivative in momentum gives the equivalent form
\begin{equation}\label{eq:uexpanded}
 u_t+uu_x+A\gamma\rho^{\gamma-2}\rho_x
 =\alpha(\rho^{\delta-1}u_x)_x
  +\alpha\rho^{\delta-2}\rho_xu_x-\mathcal M.
\end{equation}
For $0<\delta<1$, multiply \eqref{eq:uexpanded} by $u$.  With
$q=\rho^{\delta-1}$,
\begin{align}
 \frac12\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\lVert u\rVert_{L^2}^2
 +\alpha\lVert q^{1/2}u_x\rVert_{L^2}^2
 ={}&-\int u^2u_x
 -A\gamma\int\rho^{\gamma-2}\rho_xu
 +\alpha\int\rho^{\delta-2}\rho_xu_xu
 -\int\mathcal M u.
 \label{eq:uL2identity}
\end{align}
The first integral is zero.  Since
\begin{equation}\label{eq:rhox-vu}
 \alpha\rho^{\delta-2}\rho_x=v-u,
\end{equation}
the pressure integral becomes
$-(A\gamma/\alpha)\int\rho^{\gamma-\delta}(v-u)u$.
The density powers are bounded by \eqref{eq:rhoinfty}, while $v$ is bounded
by \cref{lem:vinfty}.  Also, because $\delta-1<0$,
\begin{equation}\label{eq:qcoercive}
 q=\rho^{\delta-1}\geq
 \lVert \rho\rVert_{L^\infty}^{\delta-1}>0.
\end{equation}
This lower bound comes from an upper bound for $\rho$; it does not assume
that the density stays away from vacuum.  Using
\eqref{eq:rhox-vu}, \eqref{eq:qcoercive}, and Young's inequality, the last
three integrals in \eqref{eq:uL2identity} satisfy
\begin{align}
 \left|A\gamma\int\rho^{\gamma-2}\rho_xu\right|
 &\leq C_T(1+\lVert u\rVert_{L^2}^2),\label{eq:uL2pressure}\\
 \left|\alpha\int\rho^{\delta-2}\rho_xu_xu\right|
 &\leq\frac\alpha4\lVert q^{1/2}u_x\rVert_{L^2}^2
      +C_T\lVert u\rVert_{L^2}^2,\label{eq:uL2visccross}\\
 |\langle \mathcal M,u\rangle|
 &\leq C_T(1+\lVert u\rVert_{L^2}^2).
 \label{eq:newlow1}
\end{align}
Gronwall's lemma yields the zeroth-order part of \eqref{eq:firstorder}.

For the first derivative, multiply \eqref{eq:uexpanded} by $u_t$ and
integrate the principal viscous term by parts.  One obtains
\begin{align}
 \frac\alpha2\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\int q u_x^2\,\mathop{}\!\mathrm d x
 +\int u_t^2\,\mathop{}\!\mathrm d x
 ={}&\frac\alpha2\int q_tu_x^2\,\mathop{}\!\mathrm d x
 -\int uu_xu_t\,\mathop{}\!\mathrm d x
 -A\gamma\int\rho^{\gamma-2}\rho_xu_t\,\mathop{}\!\mathrm d x\nonumber\\
 &+\alpha\int\rho^{\delta-2}\rho_xu_xu_t\,\mathop{}\!\mathrm d x
 -\int\mathcal M u_t\,\mathop{}\!\mathrm d x.
 \label{eq:utfirstidentity}
\end{align}
The last term is genuinely new, but \eqref{eq:Mfirst} gives
\begin{equation}\label{eq:newlow2}
 |\langle \mathcal M,u_t\rangle|
 \leq\frac18\lVert u_t\rVert_{L^2}^2+C_T.
\end{equation}
For reference, the remaining four estimates are
\begin{align}
 \frac\alpha2\left|\int q_tu_x^2\right|
 &\leq C_T\bigl(1+\lVert q^{1/2}u_x\rVert_{L^2}^{10/3}\bigr)
       +\frac18\lVert u_t\rVert_{L^2}^2,\label{eq:qtestimate}\\
 \left|\int uu_xu_t\right|
 &\leq C_T\lVert q^{1/2}u_x\rVert_{L^2}^{3}
       +\frac18\lVert u_t\rVert_{L^2}^2,\label{eq:convutestimate}\\
 \left|A\gamma\int\rho^{\gamma-2}\rho_xu_t\right|
 &\leq C_T+\frac18\lVert u_t\rVert_{L^2}^2,\label{eq:pressutestimate}\\
 \left|\alpha\int\rho^{\delta-2}\rho_xu_xu_t\right|
 &\leq C_T(1+\lVert u_x\rVert_{L^2}^2)
       \lVert q^{1/2}u_x\rVert_{L^2}^2
       +\frac18\lVert u_t\rVert_{L^2}^2.\label{eq:crossutestimate}
\end{align}
Here \eqref{eq:qtestimate} follows from
$q_t+uq_x+(\delta-1)qu_x=0$; the other lines use
\eqref{eq:rhox-vu}, the one-dimensional inequality
$\lVert f\rVert_{L^\infty}^2\leq2\lVert f\rVert_{L^2}\lVert f_x\rVert_{L^2}$, and Young's
inequality.  The elliptic form of \eqref{eq:uexpanded} simultaneously gives
\begin{equation}\label{eq:uxxelliptic-low}
 \lVert q u_{xx}\rVert_{L^2}
 \leq C_T\bigl(1+\lVert u_t\rVert_{L^2}+\lVert u_x\rVert_{L^2}^{3/2}\bigr).
\end{equation}
Substitution of \eqref{eq:newlow2}--\eqref{eq:crossutestimate} into
\eqref{eq:utfirstidentity} can now be closed without assuming a small
initial norm.  To see this point explicitly, put
$y=\lVert q^{1/2}u_x\rVert_{L^2}^2$.  The zeroth-order estimate gives
$y\in L^1(0,T)$, while \eqref{eq:qcoercive} gives
$\lVert u_x\rVert_{L^2}^2\leq C_Ty$.  The largest powers on the right are
therefore
\[
 y^{5/3}=y\,y^{2/3},\qquad y^{3/2}=y\,y^{1/2},\qquad y^2=y\,y.
\]
Their second factors belong to $L^1(0,T)$ because $y\in L^1(0,T)$ and
the time interval is finite.  Thus the differential inequality has the
linear Gronwall form $y'\leq C_T+a(t)y$ with $a\in L^1(0,T)$, after the
$u_t$ terms have been absorbed.  Gronwall's lemma gives, for
$0<\delta<1$,
\begin{align}
 &\sup_{0\leq t\leq T}
 \left(\lVert u\rVert_{L^2}^2+
 \lVert \rho^{(\delta-1)/2}u_x\rVert_{L^2}^2\right)\nonumber\\
 &\quad+\int_0^T\left(
 \lVert \rho^{(\delta-1)/2}u_x\rVert_{L^2}^2
 +\lVert u_t\rVert_{L^2}^2+\lVert \rho^{\delta-1}u_{xx}\rVert_{L^2}^2
 +\lVert u_x\rVert_{L^\infty}\right)\mathop{}\!\mathrm d t\leq C_T.
 \label{eq:firstorder}
\end{align}
When $\delta=1$, $q=1$ and $q_t=0$.  The same calculation therefore becomes
shorter: all density weights in \eqref{eq:firstorder} are absent and the
first term on the right of \eqref{eq:utfirstidentity} vanishes.  The elliptic
form of \eqref{eq:ueq}, \eqref{eq:Mfirst}, and the one-dimensional
interpolation inequality also imply
\begin{equation}\label{eq:uxxintegral}
 \int_0^T\lVert u_{xx}\rVert_{L^2}^2\,\mathop{}\!\mathrm d t\leq C_T.
\end{equation}
No pointwise-in-time bound for $u_t$ is asserted at this stage.  Equations
\eqref{eq:firstorder}--\eqref{eq:uxxintegral} and \cref{lem:kH2} now yield
\begin{equation}\label{eq:kstrong}
 \sup_{0\leq t\leq T}\lVert k(t)\rVert_{H^2}\leq C_T.
\end{equation}

\emph{Step 2: $L^2$ effective velocity and low-order $\psi$.}
Set
\[
 a=\frac{A\gamma}{\alpha}\rho^{\gamma-\delta}
   +\frac{|h|^2}{\alpha}\rho^{2-\delta},
 \qquad G=\rho^2h\cdot k.
\]
Multiplication of \eqref{eq:vequation} by $v$ gives
\begin{equation}\label{eq:v2exact}
 \frac12\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\lVert v\rVert_{L^2}^2
 +\int_\mathbb R a|v|^2\,\mathop{}\!\mathrm d x
 =\frac12\int_\mathbb R u_x|v|^2\,\mathop{}\!\mathrm d x
  +\int_\mathbb R auv\,\mathop{}\!\mathrm d x-\int_\mathbb R Gv\,\mathop{}\!\mathrm d x.
\end{equation}
Here we used
$\int uvv_x=-\frac12\int u_xv^2$.  Since $a\in L^\infty$, $a\geq0$,
and $G\in L^2$, Young's inequality gives
\begin{equation}\label{eq:v2diff}
 \frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\lVert v\rVert_{L^2}^2
 \leq C_T(1+\lVert u_x\rVert_{L^\infty})\lVert v\rVert_{L^2}^2
      +C_T(1+\lVert u\rVert_{L^2}^2).
\end{equation}
The coefficient $\lVert u_x\rVert_{L^\infty}$ is integrable in time by
\eqref{eq:firstorder}, so Gronwall controls $v$ in $L^2$.  Since
$v-u=(\alpha/\delta)\psi$, this and the $L^2$ estimate for $u$ control
$\psi$ in $L^2$.  Moreover, \eqref{eq:qcoercive} (or $q=1$) gives
$u\in L^\infty(0,T;H^1)$ and hence $u\in L^\infty((0,T)\times\mathbb R)$.
Together with \eqref{eq:vinfty}, interpolation
$\lVert \psi\rVert_{L^4}^4\leq\lVert \psi\rVert_{L^\infty}^2
\lVert \psi\rVert_{L^2}^2$ proves
\begin{equation}\label{eq:v2psi4}
 \sup_{0\leq t\leq T}
 \left(\lVert v\rVert_{L^2}+\lVert \psi\rVert_{L^2}+\lVert \psi\rVert_{L^4}\right)\leq C_T,
\end{equation}
because $v-u=(\alpha/\delta)\psi$.

\emph{Step 3: pointwise $u_t$ and second derivatives.}
Differentiate \eqref{eq:ueq} in time and test by $u_t$.  The only term not
present in the hydrodynamic identity is $-\int\mathcal M_tu_t$.  Before
$\rho_{xx}$ is available, use \eqref{eq:Mttrick}.  By
\eqref{eq:Ht}--\eqref{eq:Gt}, \eqref{eq:firstorder},
\eqref{eq:kstrong}, and \eqref{eq:v2psi4},
\begin{align*}
 \left|\int H_tu_{tx}\,\mathop{}\!\mathrm d x\right|
 &\leq\lVert \rho^{(1-\delta)/2}H_t\rVert_{L^2}
       \lVert \rho^{(\delta-1)/2}u_{tx}\rVert_{L^2},\\
 \left|\int G_tu_t\,\mathop{}\!\mathrm d x\right|
 &\leq\lVert G_t\rVert_{L^2}\lVert u_t\rVert_{L^2}.
\end{align*}
The density upper bound controls the dual factor
$\rho^{(1-\delta)/2}$ in the first line.  Young's inequality therefore
gives
\begin{equation}\label{eq:Mtcontrol}
 \left|\int_\mathbb R\mathcal M_tu_t\,\mathop{}\!\mathrm d x\right|
 \leq\varepsilon\lVert \rho^{(\delta-1)/2}u_{tx}\rVert_{L^2}^2
 +C_{\varepsilon,T}\bigl(1+\lVert u_t\rVert_{L^2}^2\bigr)
\end{equation}
for $0<\delta<1$; omit the weight for $\delta=1$.  Notice that the bounded
object is the \emph{dual} factor $\rho^{(1-\delta)/2}$, not the singular
weight $\rho^{(\delta-1)/2}$ itself.

For completeness, the remaining terms lead to
\begin{align}
 &\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\lVert u_t\rVert_{L^2}^2
 +\lVert \rho^{(\delta-1)/2}u_{tx}\rVert_{L^2}^2\leq C_T\left[
 1+\bigl(1+\lVert u_{xx}\rVert_{L^2}^2\bigr)\lVert u_t\rVert_{L^2}^2
 +\lVert \rho^{\delta-1}u_{xx}\rVert_{L^2}^2\right].
 \label{eq:ut-differential}
\end{align}
To see where this comes from, the differentiated convection is
$u_tu_x+uu_{tx}$, the pressure term is integrated by parts as
$\int\phi_tu_{tx}$, and the differentiated $\psi u_x$ term is estimated
using \eqref{eq:v2psi4}.  The one-dimensional interpolation inequality and
Young's inequality absorb one quarter of the weighted $u_{tx}$ norm.  These
are exactly the hydrodynamic products in
\cite[Lemma 3.11]{CaoLiZhu2022}; \eqref{eq:Mtcontrol} shows that the
Lorentz force fits into the same differential inequality without using
$\rho_{xx}$.

At the initial time,
\begin{equation}\label{eq:utinitial}
 u_t(0)=-u_0u_{0x}-\phi_{0x}
 +\alpha\rho_0^{\delta-1}u_{0xx}
 +\alpha\psi_0u_{0x}-\mathcal M_0.
\end{equation}
In the sublinear regime, \eqref{eq:datadelta} gives
$\alpha\rho_0^{\delta-1}u_{0xx}=g_2$, while the factorization of $u_{0x}$
there makes $\psi_0u_{0x}$ square integrable.  In the $\delta=1$ regime,
the terms are controlled directly by $u_0\in H^2$ and
$(\log\rho_0)_x\in H^1$.  Moreover,
$\mathcal M_0\in L^2$ follows from \cref{lem:Mlow} (or directly from
\eqref{eq:Mrepresentation}).  Hence
$\limsup_{\tau\downarrow0}\lVert u_t(\tau)\rVert_{L^2}\leq C$.

Integrate \eqref{eq:ut-differential} over $(\tau,t)$, let
$\tau\downarrow0$, and use \eqref{eq:firstorder} to see that every
coefficient on its right is time integrable.  Gronwall's lemma, followed by
the elliptic estimate \eqref{eq:uxxelliptic-low}, yields
\begin{align}
 &\sup_{0\leq t\leq T}\left(
 \lVert u_t\rVert_{L^2}^2+\lVert u_{xx}\rVert_{L^2}^2
 +\lVert \rho^{\delta-1}u_{xx}\rVert_{L^2}^2\right)
+\int_0^T
 \lVert \rho^{(\delta-1)/2}u_{tx}\rVert_{L^2}^2\,\mathop{}\!\mathrm d t\leq C_T.
 \label{eq:secondorder}
\end{align}
Again the weights disappear when $\delta=1$.

\emph{Step 4: the coupled $(\phi_{xx},\psi_x,u_{xxx})$ estimate.}
Writing $q=\rho^{\delta-1}$, differentiation of the elliptic identity
\eqref{eq:ueq} gives
\begin{equation}\label{eq:uxxxexact}
 \alpha q u_{xxx}
 =u_{tx}+(uu_x)_x+\phi_{xx}
 -\alpha q_xu_{xx}-\alpha(\psi u_x)_x+\mathcal M_x,
 \qquad q_x=\frac{\delta-1}{\delta}\psi.
\end{equation}
We explain the absorption in this elliptic estimate.  First,
\[
 (\psi u_x)_x=\psi_xu_x+\psi u_{xx},\qquad
 (uu_x)_x=u_x^2+uu_{xx}.
\]
The terms containing $\psi_x$ are bounded by
$\lVert \psi_x\rVert_{L^2}\lVert u_x\rVert_{L^\infty}$.  The term with no known
$L^\infty$ bound for $\psi$ is handled by interpolation:
\begin{align}
 \lVert \psi u_{xx}\rVert_{L^2}
 &\leq\lVert \psi\rVert_{L^4}\lVert u_{xx}\rVert_{L^4}\nonumber\\
 &\leq C\lVert \psi\rVert_{L^4}\lVert u_{xx}\rVert_{L^2}^{3/4}
       \lVert u_{xxx}\rVert_{L^2}^{1/4}
 \leq\varepsilon\lVert u_{xxx}\rVert_{L^2}+C_{\varepsilon,T}.
 \label{eq:psiuxxabsorb}
\end{align}
Because $q_x=((\delta-1)/\delta)\psi$, the same estimate treats
$q_xu_{xx}$.  The convection terms are easier, since Step~3 controls
$u$ in $H^2$.  Finally, $q$ has the lower bound
\eqref{eq:qcoercive} when $0<\delta<1$ and equals one when $\delta=1$.
Choose $\varepsilon$ small enough to move the last term in
\eqref{eq:psiuxxabsorb} to the left.  The additional force
$\mathcal M_x$ is controlled by \eqref{eq:Mxpsi}; hence
\begin{align}
 &\lVert u_{xxx}\rVert_{L^2}+
 \lVert \rho^{\delta-1}u_{xxx}\rVert_{L^2}
\leq C_T\left(1+\lVert u_{tx}\rVert_{L^2}
 +\lVert \phi_{xx}\rVert_{L^2}+\lVert \psi_x\rVert_{L^2}\right).
 \label{eq:ellipticthird}
\end{align}
For $0<\delta<1$ one also obtains the intermediate
$\rho^{(\delta-1)/2}u_{xxx}$ bound.  Applying $\partial_x^2$ to
\eqref{eq:phieq} and $\partial_x$ to \eqref{eq:psieq}, and testing by
$\phi_{xx}$ and $\psi_x$, respectively, gives
\begin{align}
 \frac12\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\lVert \phi_{xx}\rVert_{L^2}^2
 ={}&-\left(\gamma+\frac12\right)
       \int_\mathbb R u_x|\phi_{xx}|^2\,\mathop{}\!\mathrm d x
 -(2\gamma-1)\int_\mathbb R u_{xx}\phi_x\phi_{xx}\,\mathop{}\!\mathrm d x
 -(\gamma-1)\int_\mathbb R\phi u_{xxx}\phi_{xx}\,\mathop{}\!\mathrm d x,
 \label{eq:phixxidentity}\\
 \frac12\frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}\lVert \psi_x\rVert_{L^2}^2
 ={}&-\left(\delta+\frac12\right)
       \int_\mathbb R u_x|\psi_x|^2\,\mathop{}\!\mathrm d x
 -(2\delta-1)\int_\mathbb R\psi u_{xx}\psi_x\,\mathop{}\!\mathrm d x
 -\delta\int_\mathbb R\rho^{\delta-1}u_{xxx}\psi_x\,\mathop{}\!\mathrm d x.
 \label{eq:psixidentity}
\end{align}
These identities follow only from the product rule and one integration by
parts.  For example,
\[
 \left|\int\phi u_{xxx}\phi_{xx}\right|
 \leq C_T\bigl(\lVert u_{xxx}\rVert_{L^2}^2
                +\lVert \phi_{xx}\rVert_{L^2}^2\bigr),
\]
and
\[
 \left|\int\psi u_{xx}\psi_x\right|
 \leq\lVert \psi\rVert_{L^4}\lVert u_{xx}\rVert_{L^4}
       \lVert \psi_x\rVert_{L^2}
 \leq\varepsilon\lVert u_{xxx}\rVert_{L^2}^2

\emph{Step 5: time weights.}
Only now do we estimate $\mathcal M_t$ directly.  Step~4 has supplied
$\psi_x$ and therefore $\rho_{xx}$ through \eqref{eq:rhoxxpsi}; attempting
this direct estimate in Step~3 would have been circular.  Differentiation of
\eqref{eq:Mrepresentation} gives
\begin{align}
 \mathcal M_t={}&2h\cdot h_t\,\rho_x+|h|^2\rho_{xt}
 +2\rho\rho_t h\cdot k+\rho^2h_t\cdot k
 +\rho^2h\cdot k_t,\label{eq:Mtexact}\\
 \rho_{xt}={}&-u\rho_{xx}-2u_x\rho_x-\rho u_{xx},
 \qquad h_t=-u\rho k,\qquad k_t=-uk_x.
 \label{eq:sourcetime}
\end{align}
Consequently,
\begin{equation}\label{eq:Mtstrong}
 \sup_{0\leq t\leq T}\lVert \mathcal M_t(t)\rVert_{L^2}\leq C_T.
\end{equation}
For instance,
$\lVert |h|^2\rho_{xt}\rVert_{L^2}$ is controlled by
\eqref{eq:sourcetime}, the $H^2$ bounds for $u$ and $\rho$, and the density
upper bound.  The factors $h_t=-u\rho k$ and $k_t=-uk_x$ are controlled by
$u\in H^2$, $h\in L^\infty$, and $k\in H^2$.  The remaining products in
\eqref{eq:Mtexact} have the same or lower order.

Differentiate \eqref{eq:ueq} in time and test by $u_{tt}$.  The actual new
term satisfies
\begin{equation}\label{eq:Mutt}
 |\langle \mathcal M_t,u_{tt}\rangle|
 \leq\tfrac14\lVert u_{tt}\rVert_{L^2}^2+C_T.
\end{equation}
The time-differentiated energy inequality therefore has the form
\begin{equation}\label{eq:uttenergy}
 \frac{\mathop{}\!\mathrm d}{\mathop{}\!\mathrm d t}
 \lVert \rho^{(\delta-1)/2}u_{tx}\rVert_{L^2}^2
 +\lVert u_{tt}\rVert_{L^2}^2
 \leq C_T\left(1+
 \lVert \rho^{(\delta-1)/2}u_{tx}\rVert_{L^2}^2
 +\lVert \mathcal M_t\rVert_{L^2}^2\right).
\end{equation}

$Y(t)=\lVert \rho^{(\delta-1)/2}u_{tx}(t)\rVert_{L^2}^2$ and let $Z$ denote the
$u_{tt}$ dissipation.  Multiply \eqref{eq:uttenergy} by $t$ and integrate.
The lower endpoint needs a short limiting argument because $Y(0)$ need not
be defined.  Since $Y\in L^1(0,T)$ by \eqref{eq:secondorder}, there is a
sequence $\tau_j\downarrow0$ such that
$\tau_jY(\tau_j)\to0$.  Integrating over $(\tau_j,t)$ gives
\[
 \int_{\tau_j}^t sY'(s)\,\mathop{}\!\mathrm d s
 =tY(t)-\tau_jY(\tau_j)-\int_{\tau_j}^tY(s)\,\mathop{}\!\mathrm d s.
\]
Letting $j\to\infty$ is the reason no value of $u_{tx}(0)$ is required.
The unweighted integral of $Y$ is already finite, and
\eqref{eq:Mtstrong} controls the remaining source.  Gronwall's lemma
therefore gives
\begin{align}
 &t\lVert \rho^{(\delta-1)/2}u_{tx}(t)\rVert_{L^2}^2
+\int_0^t s\left(
 \lVert u_{tt}\rVert_{L^2}^2+\lVert u_{txx}\rVert_{L^2}^2\right)\mathop{}\!\mathrm d s\leq C_T.
 \label{eq:timeweightcorrect}
\end{align}
To recover $u_{txx}$, differentiate the elliptic momentum identity in time:
\begin{align}
 \alpha\rho^{\delta-1}u_{txx}
 ={}&u_{tt}+(uu_x)_t+\phi_{tx}
 -\alpha(\rho^{\delta-1})_tu_{xx}
 -\alpha\psi_tu_x-\alpha\psi u_{tx}+\mathcal M_t.
 \label{eq:utxxexact}
\end{align}
The only additional right-hand side is $\mathcal M_t$, controlled by
\eqref{eq:Mtstrong}; every other factor has already been estimated in
Steps~3--4.  Since $\rho^{\delta-1}$ has the positive lower bound
\eqref{eq:qcoercive} for $0<\delta<1$, \eqref{eq:utxxexact} gives the
unweighted $u_{txx}$ norm appearing in \eqref{eq:timeweightcorrect}.  For
$\delta=1$ the coefficient is simply one.  The weighted
spatial estimates in \eqref{eq:regvelocity} follow from the same elliptic
identities; the magnetic factor
\begin{equation}\label{eq:weightedM}
 \rho^{1-\delta}\mathcal M
 =\frac{|h|^2}{\delta}\rho^{3-2\delta}\psi
  +\rho^{3-\delta}h\cdot k
\end{equation}
contains no negative power of $\rho$.  In fact, differentiating this
identity, using $h_x=\rho k$, and applying the density upper bound gives
\begin{equation}\label{eq:weightedMH1}
 \lVert \rho^{1-\delta}\mathcal M\rVert_{H^1}
 \leq C_T\bigl(1+\lVert \psi\rVert_{H^1}+\lVert k\rVert_{H^1}\bigr),
\end{equation}
because $3-2\delta\geq1$ and $3-\delta\geq2$.  Thus the weighted elliptic
recoveries introduce no singular coefficient.  Combining Steps 1--5 proves
\eqref{eq:apriori}.

\emph{Step 1: compatible positive-background data on $\mathbb R$.}
