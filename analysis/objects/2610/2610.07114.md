<!-- problem-1: body_md -->
\label{prob:main}
Let $\boldsymbol{\mu}, \boldsymbol{\nu} \in \mathbb{R}^d$ and let ${\mathbf{A}}, {\mathbf{B}} \in \mathbb{R}^{d \times d}$ be positive semidefinite.
Let $P = {\mathcal{N}}(\boldsymbol{\mu}, {\mathbf{A}})$ be a reference distribution, where $\boldsymbol{\mu}$, ${\mathbf{A}}$ are known, and let $\epsilon > 0$.
    Given $n$ i.i.d.\ samples $\{\boldsymbol{X}_i\}_{i \in [n]} \sim {\mathcal{N}} (\boldsymbol{\nu}, {\mathbf{B}})$, where $\boldsymbol{\nu}$, ${\mathbf{B}}$ are unknown, estimate~\eqref{eq:frechet_dist} with high probability up to additive error $\epsilon$.
<!-- end problem-1 -->
<!-- model-1: body_md -->
\label{model:one-side}
We are given $\boldsymbol{\mu} \in \mathbb{R}^d$, ${\mathbf{A}} \in \mathbb{S}_{\succeq \mathbf{0}}^{d \times d}$ inducing $P := {\mathcal{N}}(\boldsymbol{\mu}, {\mathbf{A}})$, and observe i.i.d.\ samples $
    \left\{\boldsymbol{X}_i\right\}_{i\in[n]}
    \sim_{\textup{i.i.d.}}
    Q:={\mathcal{N}}(\boldsymbol{\nu},{\mathbf{B}})$, where $\boldsymbol{\nu} \in \mathbb{R}^d$, ${\mathbf{B}} \in \mathbb{S}_{\succeq \mathbf{0}}^{d \times d}$ are unknown.
<!-- end model-1 -->
<!-- assumption-1: body_md -->
\label{assum:normalization}
Let $P={\mathcal{N}}(\boldsymbol{\mu},{\mathbf{A}})$ and $Q={\mathcal{N}}(\boldsymbol{\nu},{\mathbf{B}})$. We assume that
\begin{equation*}
    \mathbf{0}_{d \times d} \preceq{\mathbf{A}},{\mathbf{B}}\preceq\mathbf{I}_d,
    \qquad
    \left\lVert\boldsymbol{\mu}-\boldsymbol{\nu}\right\rVert_2\le1.
\end{equation*}
<!-- end assumption-1 -->
<!-- problem-2: body_md -->
\label{prob:FD_est}
Under Model~\ref{model:one-side},
suppose Assumption~\ref{assum:normalization} holds for $(P, Q)$. Given an accuracy parameter $\epsilon \in (0, 1)$, construct an estimator $\widehat \textup{d}$ such that
\begin{equation}
\label{eq:fid_estimation_goal}
    \mathbb{P}\left[\left|\widehat \textup{d}- \textup{d}_{\mathrm{FID}}(P,Q)\right|>\epsilon\right]
    \le \frac 1 3.
\end{equation}
<!-- end problem-2 -->
<!-- proposition-1: body_md -->
\label{prop:gaussian-w2-lower-bound}
For all $d \ge 1$ and $\epsilon \in (0, \frac 1 8)$, if $n < \frac{d}{128\epsilon^2}$, there is no estimator that solves Problem~\ref{prob:FD_est} under Model~\ref{model:one-side} and Assumption~\ref{assum:normalization}.
<!-- end proposition-1 -->
<!-- fact-1: body_md -->
\label{fact:wishart_1}
For every fixed ${\mathbf{K}} \in \mathbb{S}^{d \times d}$,
\begin{equation*}
    {\mathbb{E}}\left\lVert{\mathbf{H}}_m\right\rVert_{\mathrm F}^2
    =
    \frac{d(d+1)}m,
    \qquad
    {\mathbb{E}}\left \langle {\mathbf{K}},{\mathbf{H}}_m \right\rangle^2
    =
    \frac{2}{m}\left\lVert{\mathbf{K}}\right\rVert_{\mathrm F}^2.
\end{equation*}
Moreover, for any $\delta \in (0, 1)$, with probability $\ge 1-\delta$,
\begin{equation*}
    \left\lVert{\mathbf{H}}_m\right\rVert_{\mathrm F}^2
    \lesssim_\delta\frac{d^2}{m},
    \qquad
    \left|\left \langle {\mathbf{K}},{\mathbf{H}}_m \right\rangle\right|
    \lesssim_\delta
    \frac{\left\lVert{\mathbf{K}}\right\rVert_{\mathrm F}}{\sqrt m}.
\end{equation*}
<!-- end fact-1 -->
<!-- fact-2: body_md -->
\label{fact:wishart_2}
For every fixed $q\ge0$, if $m \gtrsim_q d$,
\begin{equation*}
    {\mathbb{E}}\left[
        \left\lVert{\mathbf{H}}_m\right\rVert_{\mathrm F}^2
        \left\lVert{\mathbf{H}}_m\right\rVert_{\mathrm{op}}^{2q}
    \right]
    \lesssim_q
    \frac{d^{q+2}}{m^{q+1}}.
\end{equation*}

<!-- lemma-1: body_md -->
\label{lem:multilinear-hs-second-moment}
Let ${\mathbf{Z}}_1,\ldots,{\mathbf{Z}}_j$ be independent random elements of ${\mathcal{H}}$.
Suppose that for every $a\in[j]$,
\begin{equation}
\label{eq:multilinear-input-second-moment}
    {\mathbb{E}}\left|\left\langle{\mathbf{M}}, {\mathbf{Z}}_a\right\rangle\right|^2
    \le v\left\lVert{\mathbf{M}}\right\rVert_{\textup{F}}^2 \text{ for every deterministic }{\mathbf{M}}\in{\mathcal{H}}.
\end{equation}
Then every deterministic $j$-linear form ${\mathcal{A}}:{\mathcal{H}}^j\to\mathbb{R}$
satisfies
\begin{equation}
\label{eq:multilinear-hs-second-moment}
    {\mathbb{E}}\left|{\mathcal{A}}({\mathbf{Z}}_1,\ldots,{\mathbf{Z}}_j)\right|^2
    \le v^j\left\lVert{\mathcal{A}}\right\rVert_{\textup{F}}^2.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Conditionally on the last $j - 1$ inputs ${\mathbf{Z}}_2,\ldots,{\mathbf{Z}}_j$, the resulting operator on ${\mathbf{Z}}_1$ is
\[\sum_{r \in [q]} {\mathcal{A}}({\mathbf{E}}_r,{\mathbf{Z}}_2,\ldots,{\mathbf{Z}}_j){\mathbf{E}}_r.\]
Independence and~\eqref{eq:multilinear-input-second-moment} then give
\begin{equation*}
    {\mathbb{E}}\left|{\mathcal{A}}({\mathbf{Z}}_1,\ldots,{\mathbf{Z}}_j)\right|^2
    \le
    v\sum_{r\in[q]}
    {\mathbb{E}}\left|{\mathcal{A}}({\mathbf{E}}_r,{\mathbf{Z}}_2,\ldots,{\mathbf{Z}}_j)\right|^2.
\end{equation*}
Repeating this argument for the remaining slots gives $v^j$ times the
sum in~\eqref{eq:multilinear-hs-definition}.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:first_order_expand}
Let ${\mathbf{A}}, {\mathbf{B}} \in \mathbb{S}_{\succeq \mathbf{0}}^{d \times d}$ and ${\mathbf{H}} \in \mathbb{S}^{d \times d}$ satisfy $\mathbf{I}_d + {\mathbf{H}} \in \mathbb{S}_{\succeq \mathbf{0}}^{d \times d}$. Then following the notation \eqref{eq:bh_def},
\[\textup{d}_{\mathrm{BW}}^2\left({\mathbf{A}}, {\mathbf{B}}_{{\mathbf{H}}}\right) = \textup{d}_{\mathrm{BW}}^2\left({\mathbf{A}}, {\mathbf{B}}\right) + \left\langle{\mathbf{B}} - {\mathbf{C}}, {\mathbf{H}}\right\rangle + \Phi_{{\mathbf{C}}}({\mathbf{H}}), \]
where
\begin{equation}
\label{eq:nonlinear-covariance-term}\begin{aligned}
    \Phi_{{\mathbf{C}}}({\mathbf{H}})
    &:= \textup{d}_{\mathrm{BW}}^2({\mathbf{C}}, {\mathbf{C}}^{1/2}(\mathbf{I}_d + {\mathbf{H}}){\mathbf{C}}^{1/2}) \\
    &= 
    \left\langle{\mathbf{C}}, {\mathbf{H}}\right\rangle
    -2\left(
        \textup{Tr}\left(
            \left({\mathbf{C}}(\mathbf{I}_d+{\mathbf{H}}){\mathbf{C}}\right)^{1/2}
        \right)
        -\textup{Tr}({\mathbf{C}})\right).
        \end{aligned}
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Directly by using the definition of $\textup{d}_{\mathrm{BW}}$, and applying \eqref{eq:gram_trace},
\begin{align*}\textup{d}_{\mathrm{BW}}^2\left({\mathbf{A}}, {\mathbf{B}}_{{\mathbf{H}}}\right) - \textup{d}_{\mathrm{BW}}^2\left({\mathbf{A}}, {\mathbf{B}}\right) &= \textup{Tr}\left({\mathbf{B}}_{{\mathbf{H}}} - {\mathbf{B}}\right) - 2\left(\textup{Tr}\left(({\mathbf{A}}^{1/2}{\mathbf{B}}_{{\mathbf{H}}}{\mathbf{A}}^{1/2})^{1/2}\right) - \textup{Tr}\left({\mathbf{C}}\right)\right) \\
&= \left\langle{\mathbf{B}}, {\mathbf{H}}\right\rangle + 2\textup{Tr}({\mathbf{C}}) - 2 \textup{Tr}\left(({\mathbf{A}}^{1/2}{\mathbf{B}}_{{\mathbf{H}}}{\mathbf{A}}^{1/2})^{1/2}\right).\end{align*}
Thus it is enough to show that
\[\textup{Tr}\left(({\mathbf{A}}^{1/2}{\mathbf{B}}_{{\mathbf{H}}}{\mathbf{A}}^{1/2})^{1/2}\right) = \textup{Tr}\left(
            \left({\mathbf{C}}(\mathbf{I}_d+{\mathbf{H}}){\mathbf{C}}\right)^{1/2}
        \right).\]
        This follows from ${\mathbf{X}}^\top{\mathbf{X}} = {\mathbf{Y}}^\top{\mathbf{Y}} = (\mathbf{I}_d + {\mathbf{H}})^{1/2}{\mathbf{C}}^2(\mathbf{I}_d + {\mathbf{H}})^{1/2}$ for $({\mathbf{X}}, {\mathbf{Y}}) := ({\mathbf{A}}^{1/2}{\mathbf{B}}^{1/2}(\mathbf{I}_d + {\mathbf{H}})^{1/2}, {\mathbf{C}}(\mathbf{I}_d + {\mathbf{H}})^{1/2})$, so ${\mathbf{X}}{\mathbf{X}}^\top = {\mathbf{A}}^{1/2}{\mathbf{B}}_{{\mathbf{H}}}{\mathbf{A}}^{1/2}$ and ${\mathbf{Y}}{\mathbf{Y}}^\top = {\mathbf{C}}(\mathbf{I}_d + {\mathbf{H}}){\mathbf{C}}$ have equal eigenvalues.
<!-- end proof-2 -->
<!-- fact-3: body_md -->
\label{fact:sqrt-resolvent}
    For any ${\mathbf{M}} \in \mathbb{S}_{\succeq \mathbf{0}}^{d \times d}$, we have 
\[
\label{eq:sqrt-resolvent}
    {\mathbf{M}}^{1/2}
    =
    \frac1\pi\int_0^\infty
    {\mathbf{M}}(t\mathbf{I}_d+{\mathbf{M}})^{-1}t^{-1/2}\mathrm dt.
\]
<!-- end fact-3 -->
<!-- proposition-2: body_md -->
\label{prop:trace-resolvent-expansion}
Let ${\mathbf{A}}, {\mathbf{B}} \in \mathbb{S}_{\succeq \mathbf{0}}^{d \times d}$ and ${\mathbf{H}} \in \mathbb{S}^{d \times d}$. For every $k \in \mathbb{N}$, define
\begin{equation}\label{eq:order_k}Q_{k, {\mathbf{C}}}({\mathbf{H}}) := \frac{2(-1)^k}{\pi} \int_0^\infty \textup{Tr}\left({\mathbf{R}}_t {\mathbf{T}}_t^k\right)t^{1/2}\textup{d} t,\end{equation}
following the notation \eqref{eq:bh_def}, \eqref{eq:rtdef}. Then for all $k \ge 2$,
\[|Q_{k, {\mathbf{C}}}({\mathbf{H}})| \le Q_{2, {\mathbf{C}}}({\mathbf{H}})\lVert{\mathbf{H}}\rVert_{\textup{op}}^{k - 2}.\]
Further, if $\left\lVert{\mathbf{H}}\right\rVert_{\textup{op}} < 1$,
\begin{equation}
\label{eq:trace-resolvent-series}
\begin{aligned}
\textup{d}_{\mathrm{BW}}^2\left({\mathbf{A}}, {\mathbf{B}}_{{\mathbf{H}}}\right) - \textup{d}_{\mathrm{BW}}^2\left({\mathbf{A}}, {\mathbf{B}}\right) = \left\langle{\mathbf{B}}-{\mathbf{C}}, {\mathbf{H}}\right\rangle + \sum_{k \ge 2} Q_{k, {\mathbf{C}}}({\mathbf{H}}),
\end{aligned}
\end{equation}
and
\[\left|R_{k, {\mathbf{C}}}({\mathbf{H}})\right| \le \frac{Q_{2, {\mathbf{C}}}({\mathbf{H}})\left\lVert{\mathbf{H}}\right\rVert_{\textup{op}}^{k - 1}}{1 - \left\lVert{\mathbf{H}}\right\rVert_{\textup{op}}},\text{ where } R_{k, {\mathbf{C}}}({\mathbf{H}}) := \sum_{k' > k} Q_{k', {\mathbf{C}}}({\mathbf{H}}). \]
<!-- end proposition-2 -->
<!-- lemma-3: body_md -->
\label{lem:first_two_terms}
In the setting of Proposition~\ref{prop:trace-resolvent-expansion}, let ${\mathbf{C}}$ have eigendecomposition ${\mathbf{U}}\mathbf{\Lambda}{\mathbf{U}}^\top$ where $\mathbf{\Lambda} = \textbf{\textup{diag}}\left(\boldsymbol{\lambda}\right)$, and let $\widetilde{{\mathbf{H}}} := {\mathbf{U}}^\top{\mathbf{H}}{\mathbf{U}}$. Then letting $\frac{ab}{a+b} := 0$ if $a = b = 0$,
\begin{equation}\label{eq:second_order}
Q_{1, {\mathbf{C}}}({\mathbf{H}}) = -\left\langle{\mathbf{C}}, {\mathbf{H}}\right\rangle,\quad 
Q_{2, {\mathbf{C}}}({\mathbf{H}}) = \frac 1 4\sum_{i \in [d]} \boldsymbol{\lambda}_i \widetilde{{\mathbf{H}}}_{ii}^2 + \sum_{1 \le i < j \le d} \frac{\boldsymbol{\lambda}_i\boldsymbol{\lambda}_j}{\boldsymbol{\lambda}_i + \boldsymbol{\lambda}_j}\widetilde{{\mathbf{H}}}_{ij}^2.\end{equation}
Moreover, following the notation \eqref{eq:nonlinear-covariance-term},
\begin{equation}\label{eq:2c_bound}0 \le Q_{2, {\mathbf{C}}}\left({\mathbf{H}}\right) \le \frac{\left\lVert{\mathbf{C}}\right\rVert_{\textup{op}}}{4}\left\lVert{\mathbf{H}}\right\rVert_{\textup{F}}^2.\end{equation}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
We first directly expand
\begin{equation}\label{eq:first_order_term}
\begin{aligned}\textup{Tr}\left({\mathbf{R}}_t {\mathbf{T}}_t\right) &= \left\langle{\mathbf{U}}\left(t\mathbf{I}_d + \mathbf{\Lambda}^2\right)^{-1}{\mathbf{U}}^\top, {\mathbf{U}}\left(\mathbf{\Lambda}\sqrt{\left(t\mathbf{I}_d + \mathbf{\Lambda}^2\right)^{-1}}\right)\widetilde{{\mathbf{H}}}\left(\mathbf{\Lambda}\sqrt{\left(t\mathbf{I}_d + \mathbf{\Lambda}^2\right)^{-1}}\right){\mathbf{U}}^\top\right\rangle \\
&= \left\langle\mathbf{\Lambda}^2\left(t\mathbf{I}_d + \mathbf{\Lambda}^2\right)^{-2}, \widetilde{{\mathbf{H}}}\right\rangle = \sum_{i \in [d]} \widetilde{{\mathbf{H}}}_{ii} \frac{\boldsymbol{\lambda}_i^2}{(t + \boldsymbol{\lambda}_i^2)^2}.
\end{aligned}
\end{equation}
Hence, exchanging the sum and integral, and using $\int_0^\infty \frac{t^{1/2}s^2}{(t + s^2)^2} \textup{d} t = \frac{s\pi}{2}$, gives
\begin{align*}
Q_{1, {\mathbf{C}}}({\mathbf{H}}) &= \frac{-2}{\pi} \int_0^\infty \textup{Tr}\left({\mathbf{R}}_t {\mathbf{T}}_t\right) t^{1/2}\textup{d} t   \\
&= -\frac 2 \pi \sum_{i \in [d]} \widetilde{{\mathbf{H}}}_{ii}\int_0^\infty \frac{t^{1/2}\boldsymbol{\lambda}_i^2}{(t + \boldsymbol{\lambda}_i^2)^2} \textup{d} t = -\frac 2 \pi \sum_{i \in [d]} \widetilde{{\mathbf{H}}}_{ii} \left(\frac{\boldsymbol{\lambda}_i\pi}{2}\right) = -\left\langle{\mathbf{C}}, {\mathbf{H}}\right\rangle.
\end{align*}
Next, a similar calculation to \eqref{eq:first_order_term} yields
\[\textup{Tr}\left({\mathbf{R}}_t {\mathbf{T}}_t^2\right) =  \sum_{(i, j) \in [d] \times [d]} \frac{\boldsymbol{\lambda}_i^2 \boldsymbol{\lambda}_j^2}{\left(t + \boldsymbol{\lambda}_i^2\right)^2\left(t+\boldsymbol{\lambda}_j^2\right)} \widetilde{{\mathbf{H}}}_{ij}^2.  \]
Thus, applying the identities $\int_0^\infty \frac{s^4t^{1/2}}{(t + s^2)^3} \textup{d} t = \frac{\pi s}{8}$ and $\int_0^\infty \frac{r^2s^2t^{1/2}}{(t + r^2)^2(t + s^2)} \textup{d} t = \frac{\pi rs^2}{2(r + s)^2}$ gives 
\begin{align*}
Q_{2, {\mathbf{C}}}({\mathbf{H}}) &= \frac 2 \pi \int_0^\infty \textup{Tr}({\mathbf{R}}_t{\mathbf{T}}_t^2) t^{1/2}\textup{d} t \\
&=  \frac 1 4 \sum_{i \in [d]} \boldsymbol{\lambda}_i \widetilde{{\mathbf{H}}}_{ii}^2 + \sum_{1 \le i < j \le d} \left(\frac{\boldsymbol{\lambda}_i\boldsymbol{\lambda}_j^2 + \boldsymbol{\lambda}_j\boldsymbol{\lambda}_i^2}{(\boldsymbol{\lambda}_i + \boldsymbol{\lambda}_j)^2}  \right)\widetilde{{\mathbf{H}}}_{ij}^2,
\end{align*}
proving \eqref{eq:second_order}. Finally, we have that $0 \le \boldsymbol{\lambda}_i \le \lVert{\mathbf{C}}\rVert_{\textup{op}}$ and $\frac{\boldsymbol{\lambda}_i\boldsymbol{\lambda}_j}{\boldsymbol{\lambda}_i + \boldsymbol{\lambda}_j} \le \frac{1}{2}\lVert{\mathbf{C}}\rVert_{\textup{op}}$. Applying these bounds termwise to \eqref{eq:second_order} and using $\lVert\widetilde{{\mathbf{H}}}\rVert_{\textup{F}}^2 = \lVert{\mathbf{H}}\rVert_{\textup{F}}^2$ gives the first claim in \eqref{eq:2c_bound}.
<!-- end proof-3 -->
<!-- proof-4: body_md -->
Because $\lVert{\mathbf{C}}{\mathbf{R}}_t^{1/2}\rVert_{\textup{op}} \le 1$,
\begin{equation}\label{eq:tt_bound}
\left\lVert{\mathbf{T}}_t\right\rVert_{\textup{op}} \le \left\lVert{\mathbf{H}}\right\rVert_{\textup{op}} \text{ for all } t \ge 0,
\end{equation}
so that for any $k \ge 2$,
\[\left|\textup{Tr}\left({\mathbf{R}}_t {\mathbf{T}}_t^k\right)\right| \le \left\lVert{\mathbf{T}}_t\right\rVert_{\textup{op}}^{k - 2}\textup{Tr}\left({\mathbf{R}}_t {\mathbf{T}}_t^2 \right)\le \left\lVert{\mathbf{H}}\right\rVert_{\textup{op}}^{k - 2} \textup{Tr}\left({\mathbf{R}}_t{\mathbf{T}}_t^2\right).\]
Integrating the above display gives $|Q_{k, {\mathbf{C}}}({\mathbf{H}})| \le Q_{2, {\mathbf{C}}}({\mathbf{H}})\lVert{\mathbf{H}}\rVert_{\textup{op}}^{k - 2}$. Next, assume that $\left\lVert{\mathbf{H}}\right\rVert_{\textup{op}} < 1$, so that \eqref{eq:tt_bound} also gives $\lVert{\mathbf{T}}_t\rVert_{\textup{op}} < 1$.
Then expanding $(\mathbf{I}_d + {\mathbf{T}}_t)^{-1} = \mathbf{I}_d + \sum_{k \ge 1} (-1)^k {\mathbf{T}}_t^k$,
\begin{align*}
\textup{Tr}\left(
            \left({\mathbf{C}}(\mathbf{I}_d+{\mathbf{H}}){\mathbf{C}}\right)^{1/2}
        \right)
        -\textup{Tr}({\mathbf{C}}) &= \frac 1 \pi \sum_{k \ge 1} (-1)^{k + 1} \int_0^\infty \textup{Tr}({\mathbf{R}}_t {\mathbf{T}}_t^k) t^{1/2} \textup{d} t \\
        &= -\frac{1}{2}\sum_{k \ge 1} Q_{k, {\mathbf{C}}}({\mathbf{H}}).
\end{align*}
Because Lemma~\ref{lem:first_two_terms} shows $Q_{1, {\mathbf{C}}}({\mathbf{H}}) = -\left\langle{\mathbf{C}}, {\mathbf{H}}\right\rangle$, $\Phi_{{\mathbf{C}}}({\mathbf{H}})$ defined in \eqref{eq:nonlinear-covariance-term} equals $\sum_{k \ge 2} Q_{k, {\mathbf{C}}}({\mathbf{H}})$. The claim \eqref{eq:trace-resolvent-series} now follows from Lemma~\ref{lem:first_order_expand}. Finally,
\[|R_{k, {\mathbf{C}}}({\mathbf{H}})| \le \sum_{k' > k} Q_{2, {\mathbf{C}}}({\mathbf{H}})\lVert{\mathbf{H}}\rVert_{\textup{op}}^{k' - 2} = \frac{Q_{2, {\mathbf{C}}}({\mathbf{H}})\left\lVert{\mathbf{H}}\right\rVert_{\textup{op}}^{k - 1}}{1 - \left\lVert{\mathbf{H}}\right\rVert_{\textup{op}}}.\]
<!-- end proof-4 -->
<!-- lemma-4: body_md -->
\label{lem:phi_bound}
Let ${\mathbf{C}} \in \mathbb{S}_{\succeq \mathbf{0}}^{d \times d}$ and ${\mathbf{H}} + \mathbf{I}_d \in \mathbb{S}_{\succeq \mathbf{0}}^{d \times d}$. Then following the notation \eqref{eq:nonlinear-covariance-term}, 
\[\Phi_{{\mathbf{C}}}({\mathbf{H}}) \le \left\lVert{\mathbf{C}}\right\rVert_{\textup{op}}\left\lVert{\mathbf{H}}\right\rVert_{\textup{F}}^2. \]
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
Taking ${\mathbf{M}} \gets {\mathbf{C}}^{1/2}$ and ${\mathbf{N}} \gets {\mathbf{C}}^{1/2}(\mathbf{I}_d + {\mathbf{H}})^{1/2}$ in \eqref{eq:bures-frobenius} implies
\begin{align*}\Phi_{{\mathbf{C}}}({\mathbf{H}}) &= \textup{d}_{\mathrm{BW}}^2\left({\mathbf{C}}, {\mathbf{C}}^{1/2}(\mathbf{I}_d + {\mathbf{H}}){\mathbf{C}}^{1/2}\right) \le \left\lVert{\mathbf{C}}^{1/2} - {\mathbf{C}}^{1/2}(\mathbf{I}_d + {\mathbf{H}})^{1/2}\right\rVert_{\textup{F}}^2 \\
&\le \left\lVert{\mathbf{C}}\right\rVert_{\textup{op}}\left\lVert\mathbf{I}_d - \left(\mathbf{I}_d + {\mathbf{H}}\right)^{1/2}\right\rVert_{\textup{F}}^2 \le \left\lVert{\mathbf{C}}\right\rVert_{\textup{op}}\left\lVert{\mathbf{H}}\right\rVert_{\textup{F}}^2, \end{align*}
where the last line used that $\mathbf{I}_d - (\mathbf{I}_d + {\mathbf{H}})^{1/2} \in \mathbb{S}^{d \times d}$, and $| 1- \sqrt{1 + x}| \le |x|$ for all $x \ge -1$.
<!-- end proof-5 -->
<!-- fact-4: body_md -->
\label{fact:rescaling}
In the setting of Model~\ref{model:one-side}, following the notation \eqref{eq:empirical}, we have $\widehat{\boldsymbol{\nu}}_n \sim {\mathcal{N}}(\nu, \frac 1 n {\mathbf{B}})$, $\widehat{\boldsymbol{\nu}}_n$ and $\widehat{{\mathbf{B}}}_n$ are independent, and for $m := n - 1$ and $\mathbf{I}_d + {\mathbf{H}}_m \sim \textup{Wish}(\frac 1 m \mathbf{I}_d, m)$,
\[\frac{n}{n-1}\widehat{{\mathbf{B}}}_n \overset{d}{=} {\mathbf{B}}^{1/2} \left(\mathbf{I}_d + {\mathbf{H}}_m\right){\mathbf{B}}^{1/2}.\]
<!-- end fact-4 -->
<!-- theorem-1: body_md -->
\label{thm:first-order-plugin-bias-variance}
Under Model~\ref{model:one-side} and Assumption~\ref{assum:normalization}, the plug-in estimator $\widehat{\textup{d}}_n$ in \eqref{eq:plugin} satisfies
\begin{equation}\label{eq:bias_variance_plugin}
    \left|\mathbb{E} [\widehat{\textup{d}}_n] - \textup{d}_{\mathrm{FID}}(P, Q)\right|
    =
    O\left(\frac{d^2}{n}\right),\quad \textup{Var}[\widehat{\textup{d}}_n] = O\left(\frac{d}{n} + \frac{d^2}{n^2}\right).
\end{equation}
Consequently, if $n \gtrsim \frac{d}{\epsilon^2} + \frac{d^2}{\epsilon}$, the plug-in estimator solves Problem~\ref{prob:FD_est}.
<!-- end theorem-1 -->
<!-- proof-6: body_md -->
Throughout, let $m := n - 1$, ${\mathbf{v}} := \boldsymbol{\nu} - \boldsymbol{\mu}$, and $\widehat{{\mathbf{v}}}_n := \widehat{\boldsymbol{\nu}}_n - \boldsymbol{\mu}$. By Lemma~\ref{lem:first_order_expand},
\[\widehat{\textup{d}}_n - \textup{d}_{\mathrm{FID}}(P, Q) = 2\left\langle{\mathbf{v}}, \widehat{{\mathbf{v}}}_n - {\mathbf{v}}\right\rangle + \left\lVert\widehat{{\mathbf{v}}}_n - {\mathbf{v}}\right\rVert_2^2 + \left\langle{\mathbf{B}} - {\mathbf{C}}, {\mathbf{H}}_m\right\rangle + \Phi_{{\mathbf{C}}}({\mathbf{H}}_m). \]
Moreover, by Fact~\ref{fact:rescaling}, $\widehat{{\mathbf{v}}}_n \sim {\mathcal{N}}({\mathbf{v}}, \frac 1 n {\mathbf{B}})$, so by taking expectations,
\begin{equation}\label{eq:decompose_diff}\mathbb{E}[\widehat{\textup{d}}_n] - \textup{d}_{\mathrm{FID}}(P, Q) = \frac{\textup{Tr}({\mathbf{B}})}{n} + \mathbb{E}[\Phi_{{\mathbf{C}}}({\mathbf{H}}_m)] \le \frac d n + \mathbb{E}[\Phi_{{\mathbf{C}}}({\mathbf{H}}_m)].\end{equation}
To bound the last term, Assumption~\ref{assum:normalization} implies $\left\lVert{\mathbf{C}}\right\rVert_{\textup{op}} \le 1$, so Lemma~\ref{lem:phi_bound} and Fact~\ref{fact:wishart_1} now give
\begin{equation}\label{eq:phi_expect_bound}\mathbb{E}[\Phi_{{\mathbf{C}}}({\mathbf{H}}_m)] \le \mathbb{E} \left\lVert{\mathbf{H}}_m\right\rVert_{\textup{F}}^2 = \frac{d(d+1)}{m} = O\left(\frac{d^2}{n}\right).\end{equation}
We next bound the variance. By Fact~\ref{fact:rescaling}, $\widehat{{\mathbf{v}}}_n$ and ${\mathbf{H}}_m$ are independent, so
\[\textup{Var}[\widehat{\textup{d}}_n] = \textup{Var}\left[2\left\langle{\mathbf{v}}, \widehat{{\mathbf{v}}}_n - {\mathbf{v}}\right\rangle + \left\lVert\widehat{{\mathbf{v}}}_n - {\mathbf{v}}\right\rVert_2^2\right] + \textup{Var}\left[\left\langle{\mathbf{B}} - {\mathbf{C}}, {\mathbf{H}}_m\right\rangle + \Phi_{{\mathbf{C}}}({\mathbf{H}}_m)\right].\]
By standard Gaussian moment identities (e.g., \cite{muirhead2009aspects}, Chapter 1), we have
\[\textup{Var}\left[2\left\langle{\mathbf{v}}, \widehat{{\mathbf{v}}}_n - {\mathbf{v}}\right\rangle + \left\lVert\widehat{{\mathbf{v}}}_n - {\mathbf{v}}\right\rVert_2^2\right] = \frac{4{\mathbf{v}}^\top {\mathbf{B}} {\mathbf{v}}}{n} + \frac{2\textup{Tr}({\mathbf{B}}^2)}{n^2} = O\left(\frac{1}{n} + \frac d {n^2}\right).\]
Moreover, using $(a + b)^2 \le 2a^2 + 2b^2$ and linearity of expectation gives
\begin{align*}\textup{Var}\left[\left\langle{\mathbf{B}} - {\mathbf{C}}, {\mathbf{H}}_m\right\rangle + \Phi_{{\mathbf{C}}}({\mathbf{H}}_m)\right] &\le 2\textup{Var}\left[\left\langle{\mathbf{B}} - {\mathbf{C}}, {\mathbf{H}}_m\right\rangle\right] + 2\textup{Var}\left[\Phi_{{\mathbf{C}}}({\mathbf{H}}_m)\right],
\end{align*}
and Fact~\ref{fact:wishart_1} shows
\[\textup{Var}\left[\left\langle{\mathbf{B}} - {\mathbf{C}}, {\mathbf{H}}_m\right\rangle\right] = \frac 2 m \left\lVert{\mathbf{B}} - {\mathbf{C}}\right\rVert_{\textup{F}}^2 = O\left(\frac d n\right).\]
It remains to control $\textup{Var}[\Phi_{{\mathbf{C}}}({\mathbf{H}}_m)]$. For ${\mathbf{G}} \in \mathbb{R}^{d \times m}$ with i.i.d.\ entries $\sim {\mathcal{N}}(0, 1)$, let
\[{\mathbf{H}}({\mathbf{G}}) := \frac 1 m {\mathbf{G}}{\mathbf{G}}^\top - \mathbf{I}_d,\quad {\mathbf{N}}({\mathbf{G}}) := \begin{pmatrix} \frac 1 {\sqrt m} {\mathbf{C}}^{1/2}{\mathbf{G}} & \0_{d \times \max(0, d-m)} \end{pmatrix}.\]
Then ${\mathbf{H}}_m \overset{d}{=} {\mathbf{H}}({\mathbf{G}})$, and ${\mathbf{N}}({\mathbf{G}}){\mathbf{N}}({\mathbf{G}})^\top = {\mathbf{C}}^{1/2}(\mathbf{I}_d + {\mathbf{H}}({\mathbf{G}})){\mathbf{C}}^{1/2}$. Now by \eqref{eq:bures-frobenius},
\[\sqrt{\Phi_{{\mathbf{C}}}({\mathbf{H}}({\mathbf{G}}))} = \textup{d}_{\mathrm{BW}}({\mathbf{C}}, {\mathbf{C}}^{1/2}(\mathbf{I}_d + {\mathbf{H}}({\mathbf{G}})){\mathbf{C}}^{1/2})  = \min_{{\mathbf{M}} \in \mathbb{R}^{d \times \max(m, d)}: {\mathbf{M}}{\mathbf{M}}^\top = {\mathbf{C}} } \left\lVert{\mathbf{M}} - {\mathbf{N}}({\mathbf{G}})\right\rVert_{\textup{F}}.\]
The triangle inequality then shows that for any ${\mathbf{G}}, {\mathbf{G}}' \in \mathbb{R}^{d \times m}$,
\[\left|\sqrt{\Phi_{{\mathbf{C}}}({\mathbf{H}}({\mathbf{G}}))} - \sqrt{\Phi_{{\mathbf{C}}}({\mathbf{H}}({\mathbf{G}}'))}\right| \le \left\lVert{\mathbf{N}}({\mathbf{G}}) - {\mathbf{N}}({\mathbf{G}}')\right\rVert_{\textup{F}} \le \frac 1 {\sqrt m}\left\lVert{\mathbf{G}} - {\mathbf{G}}'\right\rVert_{\textup{F}},\]
so ${\mathbf{G}} \to \sqrt{\Phi_{{\mathbf{C}}}({\mathbf{H}}({\mathbf{G}}))}$ has a Lipschitz constant bounded by $m^{-1/2}$. Thus,
\[\left\lVert\nabla_{{\mathbf{G}}} \Phi_{{\mathbf{C}}}({\mathbf{H}}({\mathbf{G}}))\right\rVert_{\textup{F}}^2 = 4\Phi_{{\mathbf{C}}}({\mathbf{H}}_m)\left\lVert\nabla_{{\mathbf{G}}} \sqrt{\Phi_{{\mathbf{C}}}({\mathbf{H}}({\mathbf{G}}))}\right\rVert_{\textup{F}}^2 \le \frac 4 m \Phi_{{\mathbf{C}}}({\mathbf{H}}({\mathbf{G}})).\]
The Gaussian Poincar\'e inequality (Section 3.7, \cite{BLM13}) now yields
\begin{equation}\label{eq:apply_pi}\textup{Var}\left[\Phi_{{\mathbf{C}}}({\mathbf{H}}({\mathbf{G}}))\right] \le \frac 4 m \mathbb{E}\left[ \Phi_{{\mathbf{C}}}({\mathbf{H}}({\mathbf{G}})) \right]. \end{equation}
Finally, applying ${\mathbf{H}}_m \overset{d}{=}{\mathbf{H}}({\mathbf{G}})$ and our earlier bound \eqref{eq:phi_expect_bound} shows $\textup{Var}[\Phi_{{\mathbf{C}}}({\mathbf{H}}_m)] = O(\frac{d^2}{n^2})$, concluding the proof of \eqref{eq:bias_variance_plugin}. The claim about Problem~\ref{prob:FD_est} follows from Chebyshev's inequality.

<!-- assumption-2: body_md -->
\label{assum:polynomial-expectation-expansion}
Let $\{\widehat{D}_n\}_{n \in \mathbb{N}}$ be a sequence of random variables. We assume that for every fixed $k \in \mathbb{N}$, there exist
${\mathbf{a}} \in \mathbb{R}^k$, $A_{k+1}\ge0$, and
$n_0(k) \in \mathbb{N}$ independent of $n$, such that for every $n\ge n_0(k)$,
\begin{equation}
\label{eq:polynomial-expectation-assumption}
    {\mathbb{E}}[\widehat{D}_n]
    =
    D+\sum_{i \in [k]} \frac{{\mathbf{a}}_i}{n^i}+R_{k,n},
    \quad
    \left|R_{k,n}\right|
    \le\frac{A_{k+1}}{n^{k+1}}.
\end{equation}

<!-- lemma-6: body_md -->
\label{lem:wishart-collision-expansion}
Under Assumption~\ref{assum:normalization}, and following the notation of \eqref{eq:order_k} and Fact~\ref{fact:rescaling},
for every fixed $r\ge2$, there exist coefficients
$b_{r,j}({\mathbf{C}})$, independent of $m$, such that
\begin{equation}
\label{eq:wishart-homogeneous-expansion}
    {\mathbb{E}} Q_{r,{\mathbf{C}}}({\mathbf{H}}_m)
    =
    \sum_{j=\lceil r/2\rceil}^{r-1}
    \frac{b_{r,j}({\mathbf{C}})}{m^j},
    \quad
    \left|b_{r,j}({\mathbf{C}})\right|
    \lesssim_r d^{j+1}.
\end{equation}
<!-- end lemma-6 -->
<!-- proof-9: body_md -->
Throughout, let $\{{\mathbf{g}}_a \in \mathbb{R}^d\}_{a \in [m]}$ have i.i.d.\ ${\mathcal{N}}(0, 1)$ entries, and let
\begin{equation*}
    {\mathbf{H}}_m
    =
    \frac1m\sum_{a \in [m]} {\mathbf{Y}}_a,
    \quad
    {\mathbf{Y}}_a
    :=
    {\mathbf{g}}_a{\mathbf{g}}_a^\top-\mathbf{I}_d.
\end{equation*}
Since $Q_{r,{\mathbf{C}}}$ is homogeneous of degree $r$, expanding the $r$
copies of ${\mathbf{H}}_m$ gives
\begin{equation}
\label{eq:homogeneous-sample-expansion}
    Q_{r,{\mathbf{C}}}({\mathbf{H}}_m)
    =
    \frac1{m^r}
    \sum_{a_1,\ldots,a_r\in[m]}
    \mathcal Q_{r,{\mathbf{C}}}
    ({\mathbf{Y}}_{a_1},\ldots,{\mathbf{Y}}_{a_r}),
\end{equation}
where $\mathcal Q_{r,{\mathbf{C}}}$ is multilinear in its arguments.
Concretely, it is obtained from the integral \eqref{eq:order_k} defining $Q_{r,{\mathbf{C}}}$
by replacing the $r$ copies of ${\mathbf{H}}$ in ${\mathbf{T}}_t^r$ by the corresponding
arguments ${\mathbf{Y}}_{a_1}, \ldots, {\mathbf{Y}}_{a_r}$.

Each tuple $(a_1,\ldots,a_r)$ of sample labels induces a partition
$\pi$ of $[r]$: positions $s,t\in[r]$ belong to the same block
precisely when $a_s=a_t$. For example, when $r=4$, a tuple
$(a,a,b,b)$ with $a\ne b$ induces
$\pi=\left\{\left\{1,2\right\},\left\{3,4\right\}\right\}$, whereas $(a,a,a,b)$ induces
$\left\{\left\{1,2,3\right\},\left\{4\right\}\right\}$.

If a sample label occurs only once,
conditioning on all other samples leaves an expression linear in
the corresponding ${\mathbf{Y}}_a$. Independence and
${\mathbb{E}}{\mathbf{Y}}_a=\mathbf{0}$ make this conditional expectation zero, so only partitions with every block having size at least two can contribute.

Let $\mathcal P_r^{\ge2}$ be the collection of such partitions of $[r]$.
For $\pi\in\mathcal P_r^{\ge2}$, let $\mathcal C_{\pi,{\mathbf{C}}}$ be the
expectation associated with any fixed assignment of distinct sample
labels to its blocks. For example,
\begin{equation*}
    \mathcal C_{\left\{\left\{1,2\right\},\left\{3,4\right\}\right\},{\mathbf{C}}}
    :=
    {\mathbb{E}}\mathcal Q_{4,{\mathbf{C}}}
    ({\mathbf{Y}}_1,{\mathbf{Y}}_1,{\mathbf{Y}}_2,{\mathbf{Y}}_2).
\end{equation*}
Since the ${\mathbf{Y}}_a$ are i.i.d., these quantities depend only on $\pi$
and ${\mathbf{C}}$, rather than the labels or $m$. If $\pi$ has $s=\left|\pi\right|$ blocks, there are exactly
\begin{equation*}
    (m)_s:= m(m-1)\cdots(m-s+1)
\end{equation*}
ways to assign $s$ distinct sample labels to its blocks. Hence
\begin{equation}
\label{eq:collision-partition-identity}
    {\mathbb{E}} Q_{r,{\mathbf{C}}}({\mathbf{H}}_m)
    =
    \frac1{m^r}
    \sum_{\pi\in\mathcal P_r^{\ge2}}
    (m)_{\left|\pi\right|}\,\mathcal C_{\pi,{\mathbf{C}}}.
\end{equation}

Every index appears at least twice, so
$\left|\pi\right|\le\lfloor \frac r 2\rfloor$. Moreover, $(m)_s$ is a degree-$s$
polynomial with no constant term. Thus $\frac{(m)_s}{m^r}$ contains only
powers $m^{-j}$ with $r-s\le j\le r-1$. Collecting equal powers
in~\eqref{eq:collision-partition-identity} gives the first claim in \eqref{eq:wishart-homogeneous-expansion}. 
To control the magnitude of the coefficients, define
\begin{equation}\label{eq:poly_def}
    p(z)
    := 
    \sum_{j= \lceil r/2 \rceil}^{r-1}
    \frac{b_{r,j}({\mathbf{C}})}{d^{j+1}}z^{j},
\end{equation}
Then $z^{-\lceil \frac r 2 \rceil} p(z)$ is a degree-$(L - 1)$ polynomial, where $L := \lfloor \frac r 2 \rfloor$. By construction,
\begin{equation}
\label{eq:normalized-homogeneous-polynomial}
    {\mathbb{E}} Q_{r,{\mathbf{C}}}({\mathbf{H}}_m)
    =
    dp\left(\frac dm\right).
\end{equation}
Thus, bounds on the coefficients of $p$ directly translate to bounds on the $b_{r, j}({\mathbf{C}})$. We bound the coefficients of $p$ by evaluating it at $L$ fixed interpolation points $\{m_i\}_{ i\in [L]}$. Choose an integer $c_r$ large enough such that if $m \ge c_r d$, Fact~\ref{fact:wishart_2} applies at $q \gets 0$ and $q \gets r$. Then for each $i \in [L]$, we set
\[m_i := ic_r d,\quad z_i := \frac{d}{m_i} = \frac 1 {ic_r}. \]
Then we have for each $i \in [L]$,
\begin{align*}
\left|\mathbb{E} Q_{r, {\mathbf{C}}}({\mathbf{H}}_{m_i})\right| &\le \frac 1 4 \mathbb{E}\left[\left\lVert{\mathbf{H}}_{m_i}\right\rVert_{\textup{F}}^2\left\lVert{\mathbf{H}}_{m_i}\right\rVert_{\textup{op}}^{r - 2}\right] \\
&\le \frac 1 4 \mathbb{E} \left\lVert{\mathbf{H}}_{m_i}\right\rVert_{\textup{F}}^2 + \frac 1 4 \mathbb{E}\left[\left\lVert{\mathbf{H}}_{m_i}\right\rVert_{\textup{F}}^2\left\lVert{\mathbf{H}}_{m_i}\right\rVert_{\textup{op}}^{2r}\right] \\
&\lesssim_r \frac{d^2}{m_i} + \frac{d^{r + 2}}{m_i^{r + 1}} \lesssim_r d,
\end{align*}
where the first inequality used Proposition~\ref{prop:trace-resolvent-expansion}, \eqref{eq:2c_bound}, and Assumption~\ref{assum:normalization}, the second inequality used $x^{r - 2} \le 1 + x^{2r}$ for all $x \ge 0$, the third inequality applied Fact~\ref{fact:wishart_2} at $q \gets 0$ and $q \gets r$, and the last substituted in our choice of $m_i$. Combined with \eqref{eq:normalized-homogeneous-polynomial}, this gives
\[\left|z_i^{-\lceil \frac r 2 \rceil}p(z_i)\right| \lesssim_r 1,\text{ for all } i \in [L].\]
Each interpolation point $z_i$ is a fixed nonzero constant depending only on $r$, so inverting the Vandermonde matrix in the $\{z_i\}_{i \in [L]}$ to recover coefficients incurs a blowup that is only a function of $r$. After shifting by $z^{\lceil \frac r 2\rceil}$, we conclude that every coefficient of $p$ is $\lesssim_r 1$ in magnitude.
Finally, \eqref{eq:poly_def} shows that each $b_{r, j}({\mathbf{C}})$ is $d^{j + 1}$ times larger than the degree-$j$ coefficient of $p$, yielding \eqref{eq:wishart-homogeneous-expansion}.

<!-- proof-10: body_md -->
We first claim that the following pointwise estimate holds for all realizations of ${\mathbf{H}}_m$:
\begin{equation}\label{eq:residual_event}
\left|\Phi_{{\mathbf{C}}}({\mathbf{H}}_m) - \sum_{r = 2}^{2k + 1} Q_{r, {\mathbf{C}}}({\mathbf{H}}_m)\right| \lesssim_k \left\lVert{\mathbf{H}}_m\right\rVert_{\textup{F}}^2 \left\lVert{\mathbf{H}}_m\right\rVert_{\textup{op}}^{2k}.
\end{equation}
We split our proof based on ${\mathcal{E}} := \{\lVert{\mathbf{H}}_m\rVert_{\textup{op}} \le \frac{1}{2}\}$. Under ${\mathcal{E}}$, \eqref{eq:residual_event} follows from Proposition~\ref{prop:trace-resolvent-expansion}, \eqref{eq:2c_bound}, and Assumption~\ref{assum:normalization}.

\[\left|Q_{r, {\mathbf{C}}}({\mathbf{H}}_m)\right| \le \frac 1 4 \left\lVert{\mathbf{H}}_m\right\rVert_{\textup{F}}^2\left\lVert{\mathbf{H}}_m\right\rVert_{\textup{op}}^{r - 2} \le 2^{2k} \left\lVert{\mathbf{H}}_m\right\rVert_{\textup{F}}^2 \left\lVert{\mathbf{H}}_m\right\rVert_{\textup{op}}^{2k}.\]

<!-- proof-11: body_md -->
We follow the notation of Lemmas~\ref{lem:wishart-collision-expansion} and~\ref{lem:truncation_ols}. 
For each $j \in [k]$, define
\[\beta_j({\mathbf{C}}) := \sum_{r = j+1}^{2j} b_{r, j}({\mathbf{C}}).\]
Then directly comparing with the statements of Lemmas~\ref{lem:wishart-collision-expansion} and~\ref{lem:truncation_ols} shows
\[\mathbb{E} \Phi_{{\mathbf{C}}}({\mathbf{H}}_m) = \sum_{j \in [k]} \frac{\beta_j({\mathbf{C}})}{m^j} + O_k\left(\frac{d^{k + 2}}{m^{k + 1}}\right),\quad |\beta_j({\mathbf{C}})| \lesssim_k d^{j + 1}.\]

We are now essentially done with the proof, except that Assumption~\ref{assum:polynomial-expectation-expansion} requires a polynomial in $\frac 1 n$ rather than in $\frac 1 m$. To fix this, we use the negative-binomial expansion identity:
\[\frac 1 {m^j} = \frac 1 {(n - 1)^j} =  \sum_{i=j}^\infty \binom{i-1}{j-1} \frac 1 {n^i}= \sum_{i = j}^k \binom{i-1}{j-1} \frac 1 {n^i} + O_k\left(\frac 1 {n^{k + 1}}\right),\]
so that Assumption~\ref{assum:polynomial-expectation-expansion} now holds with
\[{\mathbf{a}}_i := \mathbb{I}_{i = 1}\textup{Tr}({\mathbf{B}}) + \sum_{j \in [i]} \binom{i-1}{j-1} \beta_j({\mathbf{C}}),\quad A_{k + 1} = O(d^{k + 2}).\]
At this point substituting our bounds $\textup{Tr}({\mathbf{B}}) \le d$, $|\beta_j({\mathbf{C}})| \lesssim_k d^{j + 1}$ gives the conclusion.

<!-- lemma-10: body_md -->
\label{lem:symmetrized-trace-word-hs}
Let ${\mathbf{B}}_1,\ldots,{\mathbf{B}}_j\in\mathbb{R}^{d\times d}$, and define a symmetric $j$-linear form ${\mathcal{M}}: (\mathbb{R}^{d \times d})^j \to \mathbb{R}$ by
\begin{equation}
\label{eq:symmetrized-trace-word}
    {\mathcal{M}}({\mathbf{X}}_1,\ldots,{\mathbf{X}}_j)
    :=
    \frac1{j!}
    \sum_{\pi\in S_j}
    \textup{Tr}\left(
       {\mathbf{X}}_{\pi(1)}
        {\mathbf{B}}_1\cdots
        {\mathbf{X}}_{\pi(j)}{\mathbf{B}}_j
    \right),
\end{equation}
where $S_j$ is the set of permutations $\pi: [j] \to [j]$.
Then letting $(d)_j := d(d+1)\ldots(d + j - 1)$,
\begin{equation}
\label{eq:symmetrized-trace-word-hs}
    \left\lVert{\mathcal{M}}\right\rVert_{\textup{F}}^2
    \le
    \frac{(d)_j}{(j-1)!}
    \prod_{\ell \in [j]}
    \left\lVert{\mathbf{B}}_\ell\right\rVert_{\mathrm{op}}^2.
\end{equation}
<!-- end lemma-10 -->
<!-- proof-15: body_md -->
We set up two pieces of helper notation. Let $C_j \subseteq S_j$ be the set of $j$-\emph{cycles}, i.e., permutations consisting of a single cycle. Also, for a permutation $\pi \in S_j$, define $c_\pi \in S_j$ by $c_\pi(i) = \pi(\pi^{-1}(i) + 1)$ where indices are modulo $j$, i.e., the element of $\pi$ after $i$. For example, if $\pi = \{3, 2, 4, 1\}$, $c_\pi(4) = 1$.


Next, let ${\mathbf{G}}_1,\ldots,{\mathbf{G}}_j$ be independent $d\times d$ matrices with
i.i.d.\ entries $\sim {\mathcal{N}}(0, 1)$, so that
\begin{equation}
\label{eq:gaussian-probing-hs}
    \left\lVert{\mathcal{M}}\right\rVert_{\textup{F}}^2
    =
    {\mathbb{E}}\left|{\mathcal{M}}({\mathbf{G}}_1,\ldots,{\mathbf{G}}_j)\right|^2.
\end{equation}
Also, for each permutation $\pi\in S_j$, write
\begin{equation*}
    {\mathcal{M}}_\pi
    :=
    \textup{Tr}\left({\mathbf{G}}_{\pi(1)}
        {\mathbf{B}}_1\cdots
        {\mathbf{G}}_{\pi(j)}{\mathbf{B}}_j
    \right).
\end{equation*}
Thus, expanding \eqref{eq:gaussian-probing-hs} gives
\begin{equation}\label{eq:cross_terms}
\left\lVert{\mathcal{M}}\right\rVert_{\textup{F}}^2 = \frac 1 {(j!)^2} \sum_{\pi, \sigma} \mathbb{E}\left[{\mathcal{M}}_\pi{\mathcal{M}}_\sigma\right].
\end{equation}
We next bound $\mathbb{E}[{\mathcal{M}}_\pi {\mathcal{M}}_\sigma]$ for fixed $\pi, \sigma \in S_j$. Expanding traces entrywise,
\[{\mathcal{M}}_\pi = \sum_{(a, b) \in [d]^j \times [d]^j} \prod_{r \in [j]} [{\mathbf{G}}_r]_{a_r b_r} [{\mathbf{B}}_{\pi^{-1}(r)}]_{b_r a_{c_\pi(r)}}.\]
Thus, when expanding ${\mathcal{M}}_\pi {\mathcal{M}}_\sigma$ and taking expectations, all terms vanish except those which share all indices in $a, b$ due to the independent Gaussian entries. Therefore,
\begin{align*}
\mathbb{E}\left[{\mathcal{M}}_\pi{\mathcal{M}}_\sigma\right] = \sum_{a \in [d]^j} \prod_{r \in [j]} \left[{\mathbf{B}}_{\pi^{-1}(r)}^\top {\mathbf{B}}_{\sigma^{-1}(r)}\right]_{a_{c_\pi(r)}a_{c_\sigma(r)}} ,
\end{align*}
where we collapsed the $b$ indices. Next, let ${\mathbf{H}}_r := {\mathbf{B}}^\top_{\pi^{-1}(r)}{\mathbf{B}}_{\sigma^{-1}(r)}$ and $\rho := c_\pi^{-1}c_\sigma$, and re-index $x_r := a_{c_\pi(r)}$. The above display then becomes
\[\mathbb{E}\left[{\mathcal{M}}_\pi{\mathcal{M}}_\sigma\right] = \sum_{x \in [d]^j} \prod_{r \in [j]} [{\mathbf{H}}_r]_{x_rx_{\rho(r)}}.\]
This sum factors based on the cycle structure of $\rho$. Indeed, if $\rho = c_\tau$ is exactly a $j$-cycle, then the above display is $\textup{Tr}({\mathbf{H}}_{\tau(1)}{\mathbf{H}}_{\tau(2)}\ldots{\mathbf{H}}_{\tau(j)})$, and otherwise the terms factor independently. Thus,
\[\mathbb{E}\left[{\mathcal{M}}_\pi{\mathcal{M}}_\sigma\right] = \prod_{(r_1, r_2, \ldots, r_\ell) \text{ a cycle in } \rho} \textup{Tr}\left({\mathbf{H}}_{r_1}\ldots{\mathbf{H}}_{r_\ell}\right).\]
Now using $|\textup{Tr}({\mathbf{H}}_{r_1}\ldots{\mathbf{H}}_{r_\ell})| \le d\prod_{i \in [\ell]}\lVert{\mathbf{H}}_{r_i}\rVert_{\textup{op}}$ on each term above, we have
\[\left|\mathbb{E}[{\mathcal{M}}_\pi{\mathcal{M}}_\sigma]\right| \le d^{|\mathrm{cyc}(\rho)| } \prod_{r \in [j]} \left\lVert{\mathbf{H}}_r\right\rVert_{\textup{op}} \le d^{|\mathrm{cyc}(\rho)| } \prod_{r \in [j]} \left\lVert{\mathbf{B}}_r\right\rVert_{\textup{op}}^2, \]
where $\mathrm{cyc}(\rho)$ is the set of cycles that partition $\rho$. Plugging this back into \eqref{eq:cross_terms} then gives
\begin{align*}\left\lVert{\mathcal{M}}\right\rVert_{\textup{F}}^2 &\le \frac {1} {(j!)^2} \prod_{r \in [j]} \left\lVert{\mathbf{B}}_r\right\rVert_{\textup{op}}^2 \sum_{\pi, \sigma \in S_j} d^{|\mathrm{cyc}(c_\pi^{-1}c_\sigma)|} \\
&= \frac {1} {((j - 1)!)^2} \prod_{r \in [j]} \left\lVert{\mathbf{B}}_r\right\rVert_{\textup{op}}^2 \sum_{c, c' \in C_j} d^{|\mathrm{cyc}(c^{-1}c')|} \\
&\le \frac{1}{(j - 1)!} \prod_{r \in [j]} \left\lVert{\mathbf{B}}_r\right\rVert_{\textup{op}}^2 \sum_{\pi \in S_j} d^{|\mathrm{cyc}(\pi)|}.\end{align*}
The second line used that each cycle $c_\pi$ corresponds to $j$ distinct permutations $\pi$ (creating a factor of $j^2$), and the third line used that for each $c \in C_j$, we have $\sum_{c' \in C_j} d^{|\mathrm{cyc}(c^{-1}c')|} \le \sum_{\pi \in S_j} d^{|\mathrm{cyc}(\pi)|}$, and there are $(j - 1)!$ choices of the cycle $c$. The conclusion follows from the standard identity
\[\sum_{\pi \in S_j} d^{|\mathrm{cyc}(\pi)|} = (d)_j.\]

\label{lem:optimal-partial-contraction}
Under the event \eqref{eq:tmbd_def}, define ${\mathbf{E}}, \widetilde{{\mathbf{C}}}$ as in \eqref{eq:optimal-relative-localization}, let $k \ge 2$, and let $j \in [k]$. Then letting ${\mathcal{Q}}_{k, \widetilde{{\mathbf{C}}}}$ be the symmetric $k$-linear form associated with \eqref{eq:order_k},

For all $t>0$, we define for convenience
\begin{equation*}
    {\mathbf{R}}_t:=(t\mathbf{I}_d+\widetilde{\mathbf{C}}^2)^{-1},
    \quad
    {\mathbf{D}}_t:=\widetilde{\mathbf{C}}{\mathbf{R}}_t^{1/2}.
\end{equation*}
These matrices are all functions of $\widetilde{{\mathbf{C}}}$, so they all commute. Our starting point is manipulating the expression for ${\mathcal{Q}}_{k, \widetilde{{\mathbf{C}}}}$ into a form to which Lemma~\ref{lem:symmetrized-trace-word-hs} applies. First, from \eqref{eq:order_k} we have
\begin{equation*}
    Q_{k,\widetilde{\mathbf{C}}}({\mathbf{X}})
    =
    \frac{2(-1)^k}{\pi}
    \int_0^\infty
    t^{1/2}
    \textup{Tr}\left(
        {\mathbf{R}}_t
        \left({\mathbf{D}}_t{\mathbf{X}}{\mathbf{D}}_t\right)^k
    \right)\mathrm dt.
\end{equation*}
The expression above is obtained by setting ${\mathbf{X}}_1, \ldots, {\mathbf{X}}_k$ all equal in the following $k$-linear form:
\[({\mathbf{X}}_1, \ldots, {\mathbf{X}}_k) \to \frac{2(-1)^k}{\pi}
    \int_0^\infty
    t^{1/2}
    \textup{Tr}\left(
        {\mathbf{X}}_1 {\mathbf{D}}_t^2 {\mathbf{X}}_2 \ldots {\mathbf{D}}_t^2 {\mathbf{X}}_k
    {\mathbf{D}}_t^2 {\mathbf{R}}_t\right)\mathrm dt.\]
Therefore, to symmetrize the $k$-linear form, we explicitly have
\begin{equation}
\label{eq:polarized-Q-resolvent}
\begin{aligned}
    \mathcal Q_{k,\widetilde{\mathbf{C}}}[{\mathbf{X}}_1,\ldots,{\mathbf{X}}_k] =
    \frac{2(-1)^k}{\pi k!}
    \sum_{\pi \in S_k}
    \int_0^\infty
    t^{1/2}
    \textup{Tr}\left(
        {\mathbf{X}}_{\pi(1)} {\mathbf{D}}_t^2 {\mathbf{X}}_{\pi(2)} \ldots {\mathbf{D}}_t^2 {\mathbf{X}}_{\pi(k)}
    {\mathbf{D}}_t^2 {\mathbf{R}}_t
    \right)\mathrm dt.
\end{aligned}
\end{equation}
The expression \eqref{eq:polarized-Q-resolvent} is almost compatible with Lemma~\ref{lem:symmetrized-trace-word-hs}, except the free positions can change depending on which ${\mathbf{X}}_r$ copies are fixed to ${\mathbf{E}}$. For each $t > 0$ and possible set of free positions $P \in \binom{[k]}{j}$, we define a symmetric $j$-multilinear form
\[{\mathcal{M}}_{t, P}[{\mathbf{X}}_1, \ldots, {\mathbf{X}}_j] = \frac 1 {j!} \sum_{\pi \in S_j} \textup{Tr}\left({\mathbf{X}}_{\pi(1)} {\mathbf{B}}_{1, t, P} {\mathbf{X}}_{\pi(2)} {\mathbf{B}}_{2, t, P} \ldots {\mathbf{X}}_{\pi(j)} {\mathbf{B}}_{j, t, P}\right),\]
such that the ${\mathbf{B}}_{r, t, P}$ consist of intermediate copies of ${\mathbf{D}}_t$, ${\mathbf{R}}_t$, and ${\mathbf{E}}$ from the integrand \eqref{eq:polarized-Q-resolvent}, depending on the positions $P$. The only important property of these matrices is that
\begin{equation}\label{eq:opnorm_bound}\prod_{r \in [j]} \left\lVert{\mathbf{B}}_{r, t, P}\right\rVert_{\textup{op}} \le \rho^{k - j}\left\lVert{\mathbf{R}}_t{\mathbf{D}}_t^2\right\rVert_{\textup{op}},\end{equation}
because there are $k - j$ copies of ${\mathbf{E}}$ with $\lVert{\mathbf{E}}\rVert_{\textup{op}} \le \rho$, and $\lVert{\mathbf{D}}_t\rVert_{\textup{op}} \le 1$ so we eliminate $2(k - 1)$ copies of it. We next regroup the permutation sum in \eqref{eq:polarized-Q-resolvent} to apply Lemma~\ref{lem:symmetrized-trace-word-hs}:
\[\mathcal Q_{k,\widetilde{\mathbf{C}}}[{\mathbf{X}}_1,\ldots,{\mathbf{X}}_j, {\mathbf{E}}^{\otimes k - j}] =
    \frac{2(-1)^k}{\pi}
    \int_0^\infty
    t^{1/2} \frac 1 {\binom k j} \sum_{P \in \binom{[k]}{j}} {\mathcal{M}}_{t, P}[{\mathbf{X}}_1, \ldots, {\mathbf{X}}_j]\mathrm dt.\]
    To see where the coefficient of $\binom{k}{j}^{-1}$ comes from, observe that 
    \[\frac{(k - j)!}{k!} = \frac 1 {j!\binom k j},\]
    where the factor of $(k - j)!$ arises because this is the number of different labelings of the positions occupied by the ${\mathbf{E}}$, corresponding to the same multilinear form.
    Another way to see this is that the total mass on multilinear forms is $1$ in both expressions. Finally, Lemma~\ref{lem:symmetrized-trace-word-hs} and \eqref{eq:opnorm_bound} yield
    \begin{equation}\label{eq:last_integral}\left\lVert\mathcal Q_{k,\widetilde{\mathbf{C}}}[\cdot^{\otimes j}, {\mathbf{E}}^{\otimes k - j}]\right\rVert_{\textup{F}} \le \rho^{k - j}\left(\frac{(d)_j}{(j - 1)!}\right)^{1/2} \int_0^\infty t^{1/2} \left\lVert{\mathbf{R}}_t{\mathbf{D}}_t^2\right\rVert_{\textup{op}}\textup{d} t.\end{equation}
    Observe that since $\widetilde{{\mathbf{C}}} \in \mathbb{S}_{\succeq \mathbf{0}}^{d \times d}$ and $\lVert\widetilde{{\mathbf{C}}}\rVert_{\textup{op}} \in [0, 2]$, all eigenvalues of $\widetilde{{\mathbf{C}}}$ are in $[0, 2]$, so
    \[t^{1/2}\lVert{\mathbf{R}}_t {\mathbf{D}}_t^2\rVert_{\textup{op}} = t^{1/2} \left\lVert{\mathbf{R}}_t^2 \widetilde{{\mathbf{C}}}^2\right\rVert_{\textup{op}} \le \sup_{s \in [0, 2]}  \frac{s^2t^{1/2}}{(t + s^2)^2} \le \min\left\{\frac 1 {4\sqrt{t}}, \frac 4 {t^{1.5}}\right\},\]
    where we used $(t + s^2)^2 \ge \max\{4ts^2, t^2\}$.
    The conclusion follows because the integral in \eqref{eq:last_integral} is bounded by a universal constant after substituting the above display.

Now if ${\mathbf{C}}$ has eigendecomposition ${\mathbf{U}}\mathbf{\Lambda}{\mathbf{U}}^\top$ where $\mathbf{\Lambda} = \textbf{\textup{diag}}\left(\boldsymbol{\lambda}\right)$, let $\widetilde{{\mathbf{V}}} := {\mathbf{U}}^\top {\mathbf{V}}{\mathbf{U}}$ and $\widetilde{{\mathbf{X}}}_1({\mathbf{V}}) := {\mathbf{U}}^\top{\mathbf{X}}_1({\mathbf{V}}){\mathbf{U}}$. Then the diagonal of the first line of \eqref{eq:sylvester}, after changing basis, reads
\begin{equation}\label{eq:sylvester_basis}\mathbf{\Lambda} \widetilde{{\mathbf{X}}}_1({\mathbf{V}}) + \widetilde{{\mathbf{X}}}_1({\mathbf{V}})\mathbf{\Lambda} = \mathbf{\Lambda} \widetilde{{\mathbf{V}}}\mathbf{\Lambda} \implies 2\boldsymbol{\lambda}_j [\widetilde{{\mathbf{X}}}_1({\mathbf{V}})]_{jj} =\boldsymbol{\lambda}_j^2 \widetilde{{\mathbf{V}}}_{jj} \text{ for all } j \in [d].\end{equation}
Dividing by $\boldsymbol{\lambda}_j$ and summing over all $j \in [d]$ shows $2\textup{Tr}(\widetilde{{\mathbf{X}}}_1({\mathbf{V}})) = \textup{Tr}(\mathbf{\Lambda} \widetilde{{\mathbf{V}}})$, and then changing basis (which does not affect the trace) shows that \eqref{eq:linear_cancel} indeed vanishes. We note at this point that the entire proof extends straightforwardly to singular ${\mathbf{C}}$, by restricting to its span.

Finally, we discuss efficient computation. We first spend $O(d^\omega)$ time forming the eigendecomposition of ${\mathbf{C}}$. As justified in \eqref{eq:sylvester_basis}, each Sylvester equation ${\mathbf{C}}{\mathbf{X}} + {\mathbf{X}}{\mathbf{C}} = {\mathbf{R}}$ is solved entrywise by
\[\widetilde{{\mathbf{X}}}_{ij} = \frac{\widetilde{{\mathbf{R}}}_{ij}}{\boldsymbol{\lambda}_i + \boldsymbol{\lambda}_j},\]
after changing basis to $\widetilde{{\mathbf{X}}} = {\mathbf{U}}^\top {\mathbf{X}}{\mathbf{U}}$, $\widetilde{{\mathbf{R}}} = {\mathbf{U}}^\top {\mathbf{R}}{\mathbf{U}}$, where the entry is $0$ if $\boldsymbol{\lambda}_i = \boldsymbol{\lambda}_j = 0$. Thus, we work in the ${\mathbf{U}}$ basis and apply the above recursion $k$ times in total to \eqref{eq:sylvester}. Forming each of the matrices ${\mathbf{R}}$ on the right-hand side of \eqref{eq:sylvester} takes at most $O(kd^\omega)$ time, giving the claim.

\label{lem:lecam}
Let $P_0, P_1$ be probability distributions on the same measurable space $\Omega$, and let $\phi: \Omega \to \{0, 1\}$ be a (possibly randomized) function. Then,
\begin{equation}\label{eq:lecam}\mathbb{P}_{\omega \sim P_0}\left[\phi(\omega) = 1\right] + \mathbb{P}_{\omega \sim P_1}\left[\phi(\omega) = 0\right] \ge 1 - \mathrm{TV}\left(P_0, P_1\right).\end{equation}

\end{equation*}

        \frac{1}{
            \sqrt{1+2\tau}+1

    \quad
    \operatorname{Var}\left[\hat D_{I_i}\mid I_i\right]
    \lesssim
    \frac{d}{{\mathbf{n}}_i}

\]
Moreover, the latter variance bound admits the following refinement.
Writing $\boldsymbol{X}_a=\boldsymbol{\nu}+{\mathbf{B}}^{1/2}{\mathbf{g}}_a$ with
${\mathbf{g}}_a\sim_{\textup{i.i.d.}}{\mathcal{N}}(\mathbf{0},\mathbf{I}_d)$ and ${\mathbf{K}}:={\mathbf{B}}-{\mathbf{C}}$, define

    -

    =
    \frac1{{\mathbf{n}}_i}
    \sum_{a\in I_i}\psi({\mathbf{g}}_a)
    +
    \mathcal R_i,

    {\mathbb{E}}[\mathcal R_i^2\mid I_i]
    \lesssim

\[\widehat{\boldsymbol{\nu}}_n := \frac 1 n \sum_{i \in [n]} \boldsymbol{X}_i,\quad {\mathbf{S}}_m := \frac 1 {m} \sum_{i \in [n]}(\boldsymbol{X}_i - \widehat{\boldsymbol{\nu}}_n)(\boldsymbol{X}_i - \widehat{\boldsymbol{\nu}}_n)^\top\]

\[

\[\frac1 d \left(\mathbb{E}\left[\widehat{\textup{d}}_n\right] - D_0 \right) \to \frac{1}{2} \phi(\gamma), \]

\[\frac 1d \left(\mathbb{E}\left[\widehat{\textup{d}}_n\right] - D_0\right)  = \sum_{j \in [k]} A_j(d)\left(\frac d n\right)^j + O_k\left(\left(\frac d n\right)^{k + 1}\right), \]

It remains to identify the coefficients. The vectors $(A_1(d),\ldots,A_k(d))$ are uniformly bounded. Consider any convergent subsequence with limit $(A_1^\star,\ldots,A_k^\star)$. For any sufficiently small fixed $\gamma > 0$, take $n = \lceil d/\gamma\rceil$ and let $d \to \infty$ along this subsequence. The preceding limit and expansion give
\[\left|\sum_{j \in [k]} A_j^\star \gamma^j - \frac{1}{2} \phi(\gamma)\right| \lesssim_k \gamma^{k + 1}.\]
Since this holds for every sufficiently small $\gamma > 0$, the Taylor expansion of $\phi$ yields $A_j^\star = \frac{1}{2}\alpha_j$ for all $j \in [k]$. Thus all subsequential limits coincide, so $A_j(d) \to \frac{1}{2}\alpha_j > 0$. Recalling ${\mathbf{a}}_j = d^{j + 1}A_j(d)$ proves the claim.
