<!-- conjecture-1: body_md -->
\label{con:G}
Let $0\leq k\leq n$. For any convex body $K\subseteq\mathbb{R}^n$, one has
\begin{equation}\label{eq:godbersen}
 \operatorname{V}_n\bigl(K[k],-K[n-k]\bigr)\leq\binom{n}{k}\operatorname{V}_n(K).
\end{equation}
If $1\leq k\leq n-1$ and $K$ is full-dimensional, then equality holds if and only if $K$ is a simplex.
<!-- end conjecture-1 -->
<!-- theorem-1: body_md -->
\label{thm1.1}
Let $K\subseteq\mathbb{R}^n$ be a full-dimensional convex body and let $1\leq k\leq n-1$.
Equality in \eqref{eq:godbersen} holds if and only if $K$ is a simplex.
<!-- end theorem-1 -->
<!-- proposition-1: body_md -->
\label{prop1.3}
Let $1\leq k\leq n-1$. Assume $K\subseteq\mathbb{R}^n$ is a full-dimensional convex body.
Then $K$  attains equality in \eqref{eq:godbersen} if and only if for any two directions $u,v\in \mathbb{R}^n$ with $\dim T(K,u)\leq k$ and $\dim T(K,v)\leq n-k$, there exists a direction $w\in \mathbb{R}^n$ such that $T(K,w)$ contains both $T(K,u)$ and $T(K,v)$.
<!-- end proposition-1 -->
<!-- theorem-2: body_md -->
\label{thm:equivalent}
Let $K\subseteq\mathbb{R}^n$ be a full-dimensional convex body and fix an integer $1\leq k\leq n-1$.
The following are equivalent:
\begin{enuma}
\item  $K$ is a simplex.
\item For any two faces $F$ and $G$ of $K$ with  $\dim F\leq k-1$ and $\dim G\leq n-k-1$, there exists a proper face $H$ of $K$ containing both $F$ and $G$.
\item  For any two directions $u,v\in \mathbb{R}^n$ with $\dim T(K,u)\leq k$ and $\dim T(K,v)\leq n-k$, there exists a direction $w\in \mathbb{R}^n$ such that $T(K,w)$ contains both $T(K,u)$ and $T(K,v)$. 
\end{enuma}
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm:higherGodbersen}
Let $p\geq1$ and $0\leq k_0,\dots, k_p\leq n$ be integers such that $\sum_{i=0}^pk_i=n$. For any convex body $K\subseteq\mathbb{R}^n$ we have
\begin{equation}\label{eq:higher-godbersen}
 \operatorname{V}_{pn}\bigl(-\Delta_p K[n-k_0],\iota_1K[n-k_1],\ldots,
                    \iota_pK[n-k_p]\bigr)
 \leq\frac{(n!)^p}{(pn)!}\binom{n}{k_0,\ldots,k_p}\operatorname{V}_n(K)^p.
\end{equation}
If $k_i<n$ for all $i$ and $K$ is full-dimensional, then equality holds if and only if $K$ is a simplex.
<!-- end theorem-3 -->
<!-- remark-1: body_md -->
\label{rem:reduction}
If $k_i=n$ for some $i$, then both sides of \eqref{eq:higher-godbersen} are equal to $\frac{(n!)^p}{(pn)!}\operatorname{V}_n(K)^p$,
see \cite[Remark 3.5 (b)]{K25},
hence equality is attained for any $K$. Similarly, if $K$ is lower-dimensional, then both sides of \eqref{eq:higher-godbersen} are always equal to zero.

Furthermore, by \cite[Lemma 3.1]{FK26}, zero indices can be eliminated without changing whether equality holds. Thus, apart from the trivial cases above, it suffices to consider $k_0, \dots, k_p\geq1$, which in particular implies $p\leq n-1$.

\label{prop1.6}

<!-- theorem-4: body_md -->
\label{thm:higher_equivalent}
Let $K\subseteq\mathbb{R}^n$ be a full-dimensional convex body. Let $1\leq p\leq n-1$ and fix integers $k_0,\ldots,k_p\geq 1$ with $\sum_{i=0}^p k_i=n$.
Then the following are equivalent:
\begin{enuma}
\item $K$ is a simplex.
\item For any faces $F_0,\ldots,F_p$ of $K$ with $\dim F_i\leq k_i-1$, $i=0,\dots,p$, there exists a proper face $F$ of $K$ such that $F_i\subseteq F$ for all $i=0,\dots,p$.
\item For any $p+1$ directions $u_0,\dots, u_{p}\in \mathbb{R}^n$ with $\dim T(K,u_i)\leq k_i$, $i=0,\dots,p$, there exists a direction $w\in \mathbb{R}^n$ such that $T(K,u_i)\subseteq T(K,w)$ for all $i=0,\dots,p$.
\end{enuma}

<!-- lemma-1: body_md -->
\label{lem:minimal-face}
The face $F(K,x)$ is the unique face of $K$ containing $x$ in its relative interior.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
First, we have  $x\in \operatorname{relint} F(K,x)$. Indeed, if $x\in \operatorname{relbd} F(K,x)$, then we may choose a proper face $G$ of  $F(K,x)$ containing $x$. Then $G$ is a face of $K$ which is strictly contained in $F(K,x)$, contradicting the definition of $F(K,x)$.
    Now we show the uniqueness. 
    Let $F'$ be any face  with $x\in \operatorname{relint} F'$. Then $x\in F(K,x)\cap F'$. The definition of $F(K,x)$ implies $F(K,x)\subseteq F'$. Therefore, $F(K,x)$ is a face of $F'$. But if $F(K,x)$ is a proper face of $F'$, then  $x\in F(K,x)\subseteq \operatorname{relbd} F'$, contradicting the choice of $F'$. Hence $F' = F(K,x)$ as desired.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lemma2.1}
Let $F$ and $G$ be faces of $K\in\mathcal{K}^n$.
If $F\cap \operatorname{relint} G\neq  \emptyset$, then $G\subseteq F$.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Fix $x\in F\cap \operatorname{relint} G$.
For any $a\in G\setminus\{x\}$, there exists $b\in G$ such that $x\in(a,b)$.
Since $F$ is a face, we have $a,b\in F$.
Since $a$ was chosen arbitrarily in $G$, it follows that $G\subseteq F$.
<!-- end proof-2 -->
<!-- lemma-3: body_md -->
\label{lemma2.2}
Let $F$ and $G$ be faces of $K\in\mathcal{K}^n$, and let $u\in \operatorname{relint} F$ and $v\in \operatorname{relint} G$.
Then $(u,v)\subseteq \operatorname{relint} (F\vee G)$.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Let $x\in (u,v)$ and consider $H:=F(K,x)$. Lemma \ref{lem:minimal-face} implies $x\in \operatorname{relint} H$.
Since $x\in (u,v)$, we have $u,v\in H$.
Hence by Lemma \ref{lemma2.1} we have $F,G \subseteq H$.
Since $x\in F\vee G$, we have $H = F\vee G$.
Therefore $x\in \operatorname{relint} (F\vee G)$ and the claim follows.

<!-- lemma-4: body_md -->
\label{lem:fact}
Let $K\in\mathcal{K}^n_o$. There exist bijections
$$
\begin{aligned}
\{\text{proper faces of }K^\circ\}&\overset{\sim}{\longrightarrow}\{\text{proper touching cones of }K\},\\
\{\text{proper exposed faces of }K^\circ\}&\overset{\sim}{\longrightarrow}\{\text{proper normal cones of }K\},
\end{aligned}
$$
given by $F\mapsto\mathbb{R}_{\geq0}F$. Moreover, the map $T\mapsto T\cap\partial K^\circ$ gives the inverse to both maps above.

Using the first bijection of Lemma \ref{lem:fact}, we may immediately transfer all the properties of faces to touching cones.

<!-- lemma-5: body_md -->
\label{lem:exact}
Let $E_1,\dots,E_m\subseteq\mathbb{R}^n$ be linearly independent subspaces with $d_i:=\dim E_i\geq1$ and $\sum_{i=1}^m d_i=n$.
For convex bodies $K_i\subseteq E_i$, one has
\begin{align}
\label{eq:exact}
\binom{n}{d_1,\dots,d_m}\operatorname{V}_n(K_1[d_1],\dots,K_m[d_m])=|\det T|\prod_{i=1}^m\operatorname{V}_{d_i}^{E_i}(K_i),
\end{align}
where $T:\mathbb{R}^n\to\bigoplus_{i=1}^m E_i$ is given by $Tx=(\operatorname{P}_{E_i}x)_{i=1}^m$.
<!-- end lemma-5 -->
<!-- proof-4: body_md -->
By \eqref{eq:mv_covariance}, for $t_1,\dots,t_m\geq0$ we have
$$
\operatorname{V}_n\left(\sum_{i=1}^m t_iK_i\right)=|\det T|\prod_{i=1}^m t_i^{d_i}\operatorname{V}_{d_i}^{E_i}(K_i).
$$
Comparing the coefficients of $t_1^{d_1}\cdots t_m^{d_m}$ proves the claim.

<!-- lemma-6: body_md -->
\label{lem2.5}
Let $E_1,\dots,E_m\subseteq\mathbb{R}^n$ be linearly independent subspaces with $d_i:=\dim E_i\geq1$ and $\sum_{i=1}^m d_i=n$.
For any symmetric matrices $A_1,\dots,A_m\in\operatorname{Sym}(n)$ such that $A_i$ vanishes on $E_i^\perp$, $i=1,\dots,m$, one has
\begin{equation}
    \label{eq:MDprojection}
\binom{n}{d_1,\ldots,d_m}
\operatorname{D}_n(A_1[d_1],\ldots,A_ m[d_m])
=
|\det T|^2\prod_{i=1}^ m\det(A_i|_{E_i}),
\end{equation}
where $T:\mathbb{R}^n\to\bigoplus_{i=1}^ m E_i$ is given by $Tx=(P_{E_i}x)_{i=1}^ m$.
<!-- end lemma-6 -->
<!-- proof-5: body_md -->
The operator $T$ block diagonalizes
$\sum_{i=1}^m t_i A_i$ for any $t_1,\dots,t_m\in\mathbb{R}$. Hence
$$
\det\left(\sum_{i=1}^ m t_iA_i\right)
=
\det\left(T^t\operatorname{diag}(t_iA_i|_{E_i})_{i=1}^ m T\right)
=
|\det T|^2\prod_{i=1}^ m t_i^{d_i}\det(A_i|_{E_i}),
$$
where $\operatorname{diag}(t_iA_i|_{E_i})_{i=1}^ m$ is the block diagonal matrix where the $i$-th block is $t_iA_i|_{E_i}$.
The conclusion  now follows by comparing coefficients.

\label{lem3.1}

<!-- lemma-8: body_md -->
\label{lem2.11}
Fix an integer $1\leq k\leq n$.
Let $K,L,C_1,\dots, C_{n-k}\in\mathcal{K}^n$ be convex bodies such that $K\subseteq L$ and $\operatorname{V}_n(L[k],C_1,\dots,C_{n-k})>0$.
Denote by $B$ the unit Euclidean ball in $\mathbb{R}^n$.
Then
$$
\operatorname{V}_n(K[ k],C_1,\ldots,C_{n- k})=\operatorname{V}_n(L[ k],C_1,\ldots,C_{n- k})
$$
    holds  if and only if
    $$
    h_K=h_L\quad\text{on}\quad\operatorname{supp}\operatorname{S}_{B[ k-1],C_1,\ldots,C_{n- k}}.
    $$
<!-- end lemma-8 -->
<!-- proof-7: body_md -->
We first show the ``only if'' part by induction on $ k$.
    The base case is $ k=1$. By \eqref{eq:area-integral}, 
    $$
    \operatorname{V}_n(K,C_1,\dots,C_{n-1})-\operatorname{V}_n(L,C_1,\dots,C_{n-1})=
    \frac{1}{n}\int_{\mathbb S^{n-1}} (h_K-h_L) d\operatorname{S}_{C_1,\dots, C_{n-1}}.
    $$
    Since $K\subseteq L$, we have $h_K\leq h_L$. Therefore, the difference on the left-hand side vanishes if and only if $h_K=h_L$ on $\operatorname{supp} \operatorname{S}_{C_1,\dots, C_{n-1}}$.

    Now assume the conclusion holds for $ k-1$ and write $\mathcal C$ for the tuple $(C_1,\dots, C_{n-k}) $.
    By assumption and monotonicity of mixed volumes, we have that for all $1\leq i\leq  k-1$, 
    $  \operatorname{V}_n(K[i],L[ k-i],\mathcal C)  $
    is equal to the same positive constant.
    In particular,
    \begin{align*}        
    \operatorname{V}_n &(K,L,K[i-1],L[ k-i-1],\mathcal C)^2\\
    &=\operatorname{V}_n(K,K,K[i-1],L[ k-i-1],\mathcal C)\operatorname{V}_n(L,L,K[i-1],L[ k-i-1],\mathcal C).
    \end{align*}
    By Lemma \ref{lem3.1},  this implies $\operatorname{S}_{K,K[i-1],L[ k-i-1],\mathcal C} = \operatorname{S}_{L,K[i-1],L[ k-i-1],\mathcal C}$, and so by \eqref{eq:area-integral} we have 
    $$
    \operatorname{V}_n(K[i],L[ k-i-1], \mathcal{C}, B)=\operatorname{V}_n(K[i-1],L[ k-i], \mathcal{C}, B).
    $$
    Since this holds for all $1\leq i\leq  k-1$, we obtain $
    \operatorname{V}_n(K[ k-1], \mathcal{C}, B)=\operatorname{V}_n(L[ k-1], \mathcal{C}, B)$, and hence, by induction hypothesis, 
    $$
    h_K = h_L \quad \text{on} \quad \operatorname{supp} \operatorname{S}_{B[ k-2],\mathcal{C}, B} = \operatorname{supp} \operatorname{S}_{B[ k-1],\mathcal{C}}.
    $$
    
    For the ``if'' part, recall the following classical fact from \cite[Lemma 7.6.15]{Sch14}:
    $$
    \operatorname{supp} S_{C_1,C_2,\dots, C_{n-1}}\subseteq \operatorname{supp} S_{B, C_2,\dots, C_{n-1}}.
    $$
    This together with the assumption implies
    $$
    \operatorname{V}_n(K[ k],\mathcal C)=\operatorname{V}_n(K[ k-1],L,\mathcal C).
    $$
    Iterating the same argument then yields $\operatorname{V}_n(K[ k],\mathcal C)=\operatorname{V}_n(L[ k],\mathcal C)$.

<!-- proposition-3: body_md -->
\label{pro:higherGodbersen}
Let $p\geq1$ and $0\leq k_0,\dots, k_p\leq n$ be integers such that $\sum_{i=0}^pk_i=n$. For any $K\in\mathcal{K}^n$ we have
\begin{equation}
\label{eq:higherG}
 \operatorname{V}_{pn}\bigl(-\Delta_p K[n-k_0],\iota_1K[n-k_1],\ldots,
                    \iota_pK[n-k_p]\bigr)
 \leq\frac{(n!)^p}{(pn)!}\binom{n}{k_0,\ldots,k_p}\operatorname{V}_n(K)^p.
\end{equation}
<!-- end proposition-3 -->
<!-- proof-8: body_md -->
If $\operatorname{V}_n(K)=0$, then both sides of \eqref{eq:higherG} are equal to zero. We may thus assume  $\operatorname{V}_n(K)>0$. For brevity, we will denote $N:=(p+1)n$. Let $\widetilde\iota_0,\dots,\widetilde\iota_p:\mathbb{R}^n\to\mathbb{R}^N$ be the canonical inclusions. Write $\mathbb{R}^N=\mathbb{R}^n\times\mathbb{R}^{pn}$  and denote by $\pi:\mathbb{R}^N\to\mathbb{R}^{pn}$ the projection to the second factor. Consider further the linear map $\Phi:\mathbb{R}^N\to \mathbb{R}^N$ given by
$$
\Phi(x_0,\dots,x_p):=(x_0,x_1+x_0,\dots,x_p+x_0).
$$
Then we have $\det\Phi=1$ and $\Phi^{-1}(x_0,\dots,x_p)=(x_0,x_1-x_0,\dots,x_p-x_0)$. Moreover,
$$
\Phi\circ\widetilde\iota_0=\Delta_{p+1},\qquad \pi\circ\Phi^{-1}\circ\widetilde\iota_0=-\Delta_p,\qquad\text{and}\qquad \pi\circ\Phi^{-1}\circ\widetilde\iota_l=\iota_l,\quad l=1,\dots,p.
$$
Consequently, \eqref{eq:mv_covariance} and \eqref{eq:MVprojection} yield
\begin{align}
\label{eq:higherGproof1}
\begin{split}
    &\operatorname{V}_N\big(\Delta_{p+1}K[n],\widetilde\iota_0K[n-k_0],\dots,\widetilde\iota_pK[n-k_p]\big)\\
    &\quad=\operatorname{V}_N\big(\Phi(\widetilde\iota_0K)[n],\widetilde\iota_0K[n-k_0],\dots,\widetilde\iota_pK[n-k_p]\big)\\
    &\quad=\operatorname{V}_N\left(\widetilde\iota_0K[n],\Phi^{-1}(\widetilde\iota_0K)[n-k_0],\dots,\Phi^{-1}(\widetilde\iota_pK)[n-k_p]\right)\\
    &\quad={N\choose n}^{-1}\operatorname{V}_n(K)\,\operatorname{V}_{pn}\big(-\Delta_p K[n-k_0],\iota_1 K[n-k_1],\dots,\iota_p K[n-k_p]\big).
    \end{split}
\end{align}
On the other hand, using the monotonicity of mixed volumes, $K^{p+1}=\sum_{l=0}^p\widetilde\iota_lK$, and Lemma~\ref{lem:exact}, we have
\begin{align}
\label{eq:higherGproof2}
\begin{split}
    &\operatorname{V}_N\big(\Delta_{p+1}K[n],\widetilde\iota_0K[n-k_0],\dots,\widetilde\iota_pK[n-k_p]\big)\\
    &\quad\leq\operatorname{V}_N\big(K^{p+1}[n],\widetilde\iota_0K[n-k_0],\dots,\widetilde\iota_pK[n-k_p]\big)\\
    &\quad={n\choose k_0,\dots,k_p}\operatorname{V}_N\big(\widetilde\iota_0K[n],\dots,\widetilde\iota_pK[n]\big)\\
    &\quad={n\choose k_0,\dots,k_p}\frac{(n!)^{p+1}}{N!}\operatorname{V}_n(K)^{p+1}.
    \end{split}
\end{align}
Since $\operatorname{V}_n(K)>0$, \eqref{eq:higherG} follows at once by comparing \eqref{eq:higherGproof1} and \eqref{eq:higherGproof2}.

<!-- lemma-9: body_md -->
\label{lem:detP}
For nonzero $u, v\in\mathbb{R}^n$, we have
$$
\left|\det\left(P_{u^\perp}|_{v^\perp}\right)\right|
=
\frac{|\langle u,v\rangle|}{\|u\|\|v\|}.

<!-- lemma-10: body_md -->
\label{lem4.1}
Let $E\subseteq \mathbb{R}^n$ be a linear subspace, $C\subseteq E$ a convex body whose support function is $C^2$  on $E\setminus\{0\}$, and $u\in \mathbb S^{n-1}$ such that $P_E u\neq 0$. Consider the decomposition $u^\perp=(E\cap u^\perp)\oplus W$, where $W$ is the orthogonal complement of  $E\cap u^\perp$ inside $u^\perp$. Then
$$
D^2 h_C(u) = \frac{1}{\|P_E u\|}D^2_E h_C\left(\frac{P_E u}{\|P_E u\|}\right)\oplus \mathbf{0}_W,
$$
where $D^2_E h_C$ is first computed inside $E$ and then naturally embedded into $\mathbb{R}^n$.
<!-- end lemma-10 -->
<!-- proof-10: body_md -->
Since $C\subseteq E$, it holds that $h_C(x) = h_C(P_E x)$. Therefore, by the chain rule and the $(-1)$-homogeneity of $\nabla^2_E h_C$,
    \begin{equation}
        \label{eq20}
    \nabla^2 h_C(u) =  P_E^t \nabla^2_E h_C(P_E u) P_E =\frac{1}{\|P_E u\|}P_E^t \nabla^2_E h_C\left(\frac{P_E u}{\|P_E u\|}\right)P_E
    \end{equation}
    where $\nabla^2_E h_C$ is the Euclidean Hessian of $h_C$ computed inside $E$.
    Now restricting \eqref{eq20} to $u^\perp$, we obtain 
    $$
    D^2h_C(u) = \frac{1}{\|P_E u\|}P_{E\cap u^\perp}^t D^2_E h_C\left(\frac{P_E u}{\|P_E u\|}\right)P_{E\cap u^\perp},
    $$
    which is exactly the desired formula.

<!-- proposition-4: body_md -->
\label{prop:local_proj}
   For any convex bodies $C_i\subseteq E_i$, $i=0,\dots,p$, and any $u\in \mathbb S^{n-1}_+$ we have  
    \begin{align} \label{eq25}
    \begin{split}
    \binom{n-1}{d_0-1,\dots, d_p-1}  & \langle u , w\rangle d\operatorname{S}_{C_0[d_0-1],\dots, C_p[d_p-1]}(u)\\
    = &|\det J| \pi^* \left(\langle u_0 , w\rangle d\operatorname{S}^{E_0}_{C_0}(u_0)\otimes \cdots \otimes \langle u_p , w\rangle d\operatorname{S}^{E_p}_{C_p}(u_p)\right).
   \end{split}
    \end{align}
<!-- end proposition-4 -->
<!-- proof-11: body_md -->
We may assume that each $C_i$ is full-dimensional in $E_i$ and $h_{C_i}$ is $C^2$  in $E_i\setminus\{0\}$. The general case follows by approximation, using the weak continuity of mixed area measures with respect to Hausdorff convergence. For the proof we will denote $c_{d_0,\dots,d_p}:=\binom{n-1}{d_0-1,\ldots,d_p-1}$.

Let us first consider the case where the subspaces $V_i$ are mutually orthogonal. Then $|\det J|=1$. By \eqref{eq:MDrep},
for $u\in\mathbb S^{n-1}_+$ we have
$$
\frac{d\operatorname{S}_{C_0[d_0-1],\ldots,C_p[d_p-1]}}{d\mathcal H^{n-1}}(u)
=
\operatorname{D}_{n-1}\bigl(
D^2h_{C_0}(u)[d_0-1],\ldots,D^2h_{C_p}(u)[d_p-1]
\bigr).
$$
Applying Lemmas \ref{lem2.5} and \ref{lem4.1} to the right-hand side, we obtain
$$
c_{d_0,\dots,d_p}
\operatorname{D}_{n-1}\bigl(
D^2h_{C_0}(u)[d_0-1],\ldots,D^2h_{C_p}(u)[d_p-1]
\bigr)=
|\det T|^2
\prod_{i=0}^p
\frac{\det\left(
D^2_{E_i}h_{C_i}(u_i)
\right)}{\|P_{E_i}u\|^{d_i-1}},
$$
where $T:u^\perp\to\bigoplus_{i=0}^p(E_i\cap u^\perp) $ is defined by $Tx=(P_{E_i\cap u^\perp}x)_{i=0}^p$.
We claim that
\begin{align}
\label{eq:detT}
|\det T|=\frac{\langle u,w\rangle^p}{\prod_{i=0}^p\|P_{E_i}u\|}.
\end{align}
Indeed, the adjoint $T^t$ of $T$ is the addition map $T^t(x_0,\dots, x_p)=x_0+\cdots+ x_p$ and the linear map $P_{w^\perp}\circ T^t:\bigoplus_{i=0}^p(E_i\cap u^\perp)\to\bigoplus_{i=0}^p V_i$ is block diagonal. Since $w\in E_i$, the $i$-th block is the orthogonal projection $P_{w^\perp}:(P_{E_i}u)^\perp\to w^\perp$, where the orthogonal complements are taken inside $E_i$, and hence has determinant $|\det P_{w^\perp}|=\frac{\langle P_{E_i}u,  w\rangle}{\|P_{E_i}u\|}=\frac{\langle u,  w\rangle}{\|P_{E_i}u\|}$ by Lemma \ref{lem:detP}. Therefore, 
 $$
\left|\det(P_{w^\perp}\circ T^t)\right|=\prod_{i=0}^p\frac{\langle u,w\rangle}{\|P_{E_i}u\|}
=\frac{\langle u,w\rangle^{p+1}}{\prod_{i=0}^p\|P_{E_i}u\|}.
$$
At the same time, Lemma \ref{lem:detP} also directly implies
$$
\left|\det(P_{w^\perp}\circ T^t)\right|=\left|\det (P_{w^\perp}|_{u^\perp})\right||\det (T)| = \langle u,w\rangle|\det T|.
$$
Combining the two equations shows \eqref{eq:detT}. Consequently,
\begin{align}
\label{eq:LPF1}
c_{d_0,\dots,d_p} d\operatorname{S}_{C_0[d_0-1],\ldots,C_p[d_p-1]}(u)=\langle u,w\rangle^{2p}\prod_{i=0}^p\frac{\det\left(
D^2_{E_i}h_{C_i}(u_i)
\right)}{\|P_{E_i}u\|^{d_i+1}}d\mathcal H^{n-1}(u).
\end{align}
It remains to compute the Jacobian of $\pi$. To this end, for any $x\in u^\perp$ we have
$$
D\pi(u)x=\left(\frac{P_{E_i\cap u^\perp}x}{\|P_{E_i}u\|}\right)_{i=0}^p.
$$

Let $G'$ be a proper face of $G$. Then $\dim G'\leq n-k-1$ and $(I_{k-1,n-k-1})$ implies that $F\vee G'$ is proper. Hence, $F$ and $G$ also satisfy \eqref{eq:subface_property_G} and we may use Lemma \ref{lem:2cases}. If item (a) in Lemma~\ref{lem:2cases} holds, then we are done. 
Therefore, we assume from now on that item (b) holds. Choose an extreme point $y$ of $G$ and denote $D:=F\vee y$. We will also denote $S:=\operatorname{aff}(F,G)$. Then $F$ and $G$ are affinely independent, $K\cap S = \operatorname{conv} (F, G)$, and $D\cap S = \operatorname{conv}(F,y)$.

Property $(I_{k-1,n-k-1})$ implies that $D$ is proper. Moreover, since $F$ and $G$ are affinely independent, we have
\begin{align*}
    \dim S &= \dim F+\dim G+1 = d+n-k+1,\\
    \dim \operatorname{conv}(F,y) &= \dim F+1 = d+1,
\end{align*}
and therefore
\begin{equation}\label{eq13}
\dim D\leq \dim (D\cap S)+\operatorname{codim}(S) = 
\dim\operatorname{conv}(F,y)+n-\dim S=k.
\end{equation}
If $\dim D\leq k-1$, then taking $T=D$ in Lemma~\ref{lem:T} shows that $F\vee G$ is proper as desired. 


It remains to treat the case $\dim D=k$. In this case, \eqref{eq13} takes equality, which yields $\operatorname{aff} (D,S)=\mathbb{R}^n$. Since $F\subseteq D$ and $S=\operatorname{aff}(F,G)$, we have $\operatorname{aff}(D, G)=\mathbb{R}^n$.
Moreover, $D\cap G=\operatorname{conv}(F,y)\cap G=\{y\}$.
We may therefore apply Lemma~\ref{lemma4.5} to $D,G$, which  gives $K=\operatorname{conv}(D,G)$. Since $F$ is a proper face of $D$ and translating $K$ so that $y=0$, Lemma~\ref{lemma4.6} implies that there is a face $T$ of $K$ containing $F$, intersecting $G$, and satisfying $\dim T=d+1\leq k-1$. Applying Lemma~\ref{lem:T} to this $T$ shows that $F\vee G$ is proper.

\label{lem:without_subface}
    Let $2\leq k\leq n-1$ and $0\leq d\leq k-2$. Let $K\in\mathcal{K}^n$ be a full-dimensional convex body satisfying $(I_{k-1,n-k-1})$. Assume that $F$ and $G$ are faces of $K$ satisfying $\dim F=d$ and $\dim G=n-k$. Then $F\vee G$ is proper.
<!-- end lemma-16 -->
<!-- proof-18: body_md -->
We argue by strong induction on $d$. First, if $d=0$, then there is no proper face of $F$. Hence \eqref{eq:subface_property_F} holds trivially and the claim follows immediately from Lemma \ref{lem:with_subface}. Second, fix $d>0$ and assume the statement holds for $0,\dots,d-1$. For every proper face $F'$ of $F$, we have $\dim F'<d$ and hence the induction hypothesis implies that $F'\vee G$ is proper. Thus \eqref{eq:subface_property_F} holds and the claim follows again from  Lemma \ref{lem:with_subface}.

\label{prop4.8}
Let $K\in\mathcal{K}^n$ be a full-dimensional convex body and $2\leq k\leq n-1$. If $K$ satisfies $(I_{k-1,n-k-1})$, then it satisfies $(I_{k-2,n-k})$.

Take any faces $F,G$ of $K$ with $\dim F\leq k-2$ and $\dim G\leq n-k$. If $\dim G\leq n-k-1$, property $(I_{k-1,n-k-1})$ gives that $F\vee G$ is proper. Otherwise, $\dim G=n-k$, and the same conclusion follows from Lemma~\ref{lem:without_subface}, applied with $d=\dim F$. Since $F$ and $G$ were arbitrary, $K$ satisfies $(I_{k-2,n-k})$.

<!-- lemma-17: body_md -->
\label{lem:induction_base}
Let $K\in\mathcal{K}^n$ be a full-dimensional convex body. If $K$ satisfies $(I_{0,n-2})$, then $K$ is a simplex.
<!-- end lemma-17 -->
<!-- proof-20: body_md -->
Let $x$ be an arbitrary extreme point of $K$ and put $F:=\{x\}$. Choose $p\in\operatorname{int} K$ and let $b\in\partial K$ be the (unique) point such that $p\in[b,x]$. Denote $G:=F(K,b)$.
Then $F\vee G$ contains $p\in \operatorname{int} K$ and thus $F\vee G=K$.
Since $K$ satisfies $(I_{0,n-2})$,  we have  $\dim G=n-1$. Moreover, for any proper face $G'$ of $G$, one has $\dim G'\leq n-2$ and hence by $(I_{0,n-2})$, $F\vee G'$ is proper. This shows that \eqref{eq:subface_property_G} holds. Note also that \eqref{eq:subface_property_F} holds trivially since $F$ has no proper faces.
We may therefore apply Lemma~\ref{lem:2cases}. 
Since  $F\vee G=K$, item (a) of Lemma~\ref{lem:2cases} cannot hold. 
Hence $F$ and $G$ are affinely independent and one has
$$
K = K\cap \operatorname{aff}(F,G) =\operatorname{conv}(F,G) = \operatorname{conv} (G,x).
$$
We thus conclude that $K$ is a pyramid with base $G$ and apex $x$.
In particular, every extreme point of $K$ other than $x$ belongs to $G$ and so $x$ does not lie in the affine hull of the other extreme points.

Since $x$ was arbitrary, we get that every finite set of extreme points of $K$ is affinely independent and consequently $K$ has at most $n+1$ extreme points. Since $K$ is full-dimensional and equal to the convex hull of its extreme points, we conclude that $K$ is an $n$-simplex.

First, assume that $K$ satisfies (b) or, in other words, the property $(I_{k-1,n-k-1})$. Applying Proposition~\ref{prop4.8} repeatedly, this implies that $K$ satisfies $(I_{0,n-2})$. 
Now Lemma~\ref{lem:induction_base} shows that $K$ is a simplex.

Conversely, assume that $K$ is a simplex and $F,G$ are faces of $K$ with $\dim F\leq k-1$ and $\dim G\leq n-k-1$. Since $F$ and $G$ are simplices themselves, they together contain at most $k+(n-k)=n$ vertices of $K$. Hence there is a vertex $y\in K$ not contained in $F\cup G$. Next, translate $K$ so that it contains the origin in its interior and assume that $K^{\circ}$ satisfies (b). By Lemma \ref{lem:fact}, $F:=T(K, u)\cap\partial K^{\circ}$ and $G:=T(K, v)\cap\partial K^{\circ}$ are faces of $K^\circ$ satisfying $\dim F\leq k-1$ and $\dim G\leq n-k-1$. The touching cone $T(K, w)$ clearly contains both $T(K, u)$ and $T(K, v)$, and we conclude that $K$ satisfies (c).

Conversely, assume that $K$ satisfies (c), and let $F, G$ be proper faces of $K^{\circ}$ such that $\dim F\leq k-1$ and $\dim G\leq n-k-1$. By Lemma \ref{lem:fact}, there exist $u, v\in\mathbb S^{n-1}$ with $\mathbb R_{\geq0}F=T(K, u)$ and $\mathbb R_{\geq0}G=T(K, v)$. Then $\dim T(K, u)\leq k$ and $\dim T(K, v)\leq n-k$, and hence condition (c) in turn ensures that there exists some $w\in\mathbb S^{n-1}$ such that $T(K, w)$ contains both $T(K, u)$ and $T(K, v)$. By Lemma \ref{lem:fact}, $T(K, w)\cap\partial K^{\circ}$ is a proper face of $K^\circ$. Since it clearly contains both $F$ and $G$, $F\vee G$ is proper. We conclude that $K^{\circ}$ satisfies (b).

<!-- proposition-6: body_md -->
\label{pro:reduction_higher_godbersen}
    Let $K\in\mathcal{K}^n$ be a full-dimensional convex body. Let $f,g \geq 1$ be integers such that $f+g \geq n+1$. Then there exist proper faces $F,G$ of $K$ such that 
    \[\dim F \leq f-1, \qquad \dim G \leq g-1, \qquad\text{and}\qquad F \vee G = K.\]
<!-- end proposition-6 -->
<!-- proof-23: body_md -->
Without loss of generality we may assume $f+g = n+1$. 
    We will inductively construct two sequences  $(F_j)$ and $(G_j)$, $j=0,1,\dots$, of proper faces of $K$ satisfying $F_j\vee G_j=K$, $\dim F_j\leq f-1$ and \eqref{eq:subface_property_F},  \eqref{eq:subface_property_G} from Lemma \ref{lem:2cases}.
    
First, we may always write $K = x_1 \vee \dots \vee x_m $ where $m$ is an integer and $x_1,\dots,x_m$ are extreme points of $K$. Let us fix such a representation of $K$ with minimal $m$. We let $F_0 := \{x_1\}$ and $G_0$  be a minimal face of $x_2 \vee \dots \vee x_m$ such that $F_0\vee G_0=K$. It is clear that $F_0,G_0$ are faces of $K$ with the required properties.

Second, assume that $F_j,G_j\subseteq K$ are as above. If $\dim G_j\leq g-1$, we set $F_{j+1}:=F_j$ and $G_{j+1}:=G_j$. If $\dim G_j\geq g$, we proceed as follows: Pick a maximal proper face $\widetilde G\subseteq G_j$. Then there exists an extreme point $y\in G_j$ not contained in $\widetilde G$. We set $\widetilde F:=F_j\vee y$. By Lemma \ref{lem:2cases}, $F_j$ and $G_j$ are affinely independent and satisfy \eqref{eq:convFy}. The former implies
$$
\dim\operatorname{aff}(F_j, G_j) = \dim F_j + \dim G_j + 1,
$$
the latter yields $\operatorname{aff}(\widetilde F\cap\operatorname{aff}(F_j, G_j))=\operatorname{aff}(F_j,y)$. Consequently,
\begin{align*}
    \dim \widetilde F&\leq\dim\operatorname{aff}(\widetilde F\cap\operatorname{aff}(F_j,G_j))+\operatorname{codim}\operatorname{aff}(F_j,G_j)\\
    &=\dim F_j+1+n-(\dim F_j+\dim G_j+1)\\
    &=n-\dim G_j\\
    &\leq f-1,
    \end{align*}
where in the last step we used our assumptions $f+g=n+1$ and $\dim G_j\geq g$. Using the maximality of $\widetilde G$, we further have
$$
\widetilde F\vee \widetilde G=F_j\vee y\vee \widetilde G=F_j\vee G_j=K.
$$
We choose faces $F_{j+1}\subseteq \widetilde F$ and $G_{j+1}\subseteq \widetilde G$ satisfying $F_{j+1}\vee G_{j+1}=K$ and minimizing $\dim F_{j+1}+\dim G_{j+1}$. Then we have $\dim F_{j+1}\leq \dim\widetilde F\leq f-1$, and the minimality of $\dim F_{j+1}+\dim G_{j+1}$ implies that $F_{j+1},G_{j+1}$ satisfy \eqref{eq:subface_property_F} and \eqref{eq:subface_property_G}. Note that $F_{j+1},G_{j+1}$ are clearly proper faces of $K$ since $F_j,G_j$ were such.



 Observe that, if $\dim G_j\geq g$, we have $\dim G_{j+1}<\dim G_j$ since $\widetilde G\subseteq G_j$ was proper. Therefore, there exists $j_0\geq0$ such that $\dim G_{j_0}\leq g-1$ and we thus may take $F:=F_{j_0}$ and $G:=G_{j_0}$ to finish the proof.

<!-- definition-2: body_md -->
\label{def:higherIp}
Let $K\in\mathcal{K}^n$ and fix integers $p\geq1$ and $q_0,\dots,q_p\geq0$.
We say that $K$ satisfies property $(I_{q_0,\dots, q_p})$ if for any faces $F_0,\dots,F_p$ of $K$ with $\dim F_i\leq q_i$, $i=0,\dots,p$, the face $F_0\vee\cdots\vee F_p$ is proper.

<!-- lemma-18: body_md -->
\label{lem:p_reduction}
Let $K\in\mathcal{K}^n$, $p\geq2$, and $q_0,\dots,q_p\geq0$. If $K$ satisfies $(I_{q_0,\dots, q_p})$, then it satisfies $(I_{q_0,\dots,q_{p-2},q_{p-1}+q_p+1})$.
<!-- end lemma-18 -->
<!-- proof-24: body_md -->
Assume that $K$ satisfies $(I_{q_0,\dots, q_p})$ and consider any faces $F_0,\dots,F_{p-1}\subseteq K$ with
    $$\dim F_i\leq q_i,\quad i=0,\dots,p-2,\qquad\text{and}\qquad \dim F_{p-1}\leq q_{p-1}+q_p+1.
    $$
    By Proposition \ref{pro:reduction_higher_godbersen}, there exist faces $F,G\subseteq F_{p-1}$ such that $\dim F\leq q_{p-1}$, $\dim G\leq q_p$, and $F\vee G=F_{p-1}$. By $(I_{q_0,\dots, q_p})$, $
    F_0\vee\cdots\vee F_{p-1}=F_0\vee\cdots\vee F_{p-2}\vee F\vee G
    $
    is a proper face of $K$, and hence $K$ satisfies $(I_{q_0,\dots,q_{p-2},q_{p-1}+q_p+1})$.

First, assume that $K$ satisfies (b) or, in other words, the property $(I_{k_0-1,\dots, k_p-1})$. Applying Lemma \ref{lem:p_reduction} repeatedly and using $\sum_{i=0}^pk_i=n$, this implies that $K$ satisfies $(I_{k_0-1,n-k_0-1})$. 
Now Theorem \ref{thm:equivalent} implies that $K$ is a simplex.

Conversely, assume that $K$ is a simplex and $F_0,\dots,F_p$ are faces of $K$ with $\dim F_i\leq k_i-1$, $i=0,\dots,p$. Since $F_0,\dots,F_p$ are simplices themselves, they together contain at most $\sum_{i=0}^p k_i=n$ vertices of $K$. Hence there is a vertex $y\in K$ not contained in any $F_i$, $i=0,\dots,p$. Choosing $F\subseteq K$ to be the facet not containing $y$, (b) follows. 

Finally, translate $K$ so that it contains the origin in its interior. Then a straightforward generalization of the corresponding part of the proof of Theorem~\ref{thm:equivalent}, that is, an easy application of Lemma \ref{lem:fact}, shows that $K$ satisfies (c) if and only if $K^\circ$ satisfies (b). By the previously shown equivalence (a)$\iff$(b), we thus have that $K$ satisfies (c) if and only if $K^\circ$ is a simplex. Using the self-duality property of a simplex, we conclude that (a)$\iff$(c). This finishes the proof.
