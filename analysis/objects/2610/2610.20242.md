\label{conj-HLC}
	For any integer $k,\, \ell \geqslant 0$, let $h_1, \ldots, h_k, a_1, \ldots, a_{\ell}$ be distinct integers. Then we have 
	\begin{equation}\label{HLC-conj}
		\sum_{n \leqslant x} \mu(n+h_1) \cdots \mu(n+h_k) 
		\Lambda(n+a_1) \cdots \Lambda(n+a_{\ell})
		=\big(\mathfrak{S}_{k,\, \ell}+o(1)\big) x.
	\end{equation}

	$\sum_{n \leqslant x} \mu\bigl(n+h_1\bigr)  \Lambda(n) =o(x),$
or equivalently
\begin{align}\label{HLC-1-0}
	\sum_{p \leqslant x} \mu\bigl(p+h_1\bigr) =o\big(\pi(x)\big).
\end{align}

	\begin{equation*}
		\#\Big\{p\leqslant x:\ \mu(p+2)=\kappa\Big\}
		=\bigg(\frac{A_2}{2}+o(1)\bigg)\pi(x),

		=\frac12,
		\quad \kappa\in\{-1,\,1\}.

<!-- proposition-1: body_md -->
\label{prop:vau}
If $F$ is bounded on $(x,2x]$, then
\begin{equation}\label{eq:vaughan}
\begin{aligned}
\sum_{x<n\leqslant2x}\Lambda(n)F(n)
=T_1(F)-T_2(F)+T_3(F)+O\big(\|F\|_\infty Z\big),
\end{aligned}
\end{equation}
where
\begin{equation*} 
\begin{aligned}
T_1(F)&=\sum_{d\leqslant U}\mu(d)R_Z(d)
 \sum_{x<dk\leqslant2x}R_Z(k)(\log k)F(dk),\\
T_2(F)&=\sum_{\substack{d\leqslant U\\\ell\leqslant V}}
 \mu(d)\Lambda(\ell)R_Z(d\ell)
 \sum_{x<d\ell k\leqslant2x}R_Z(k)F(d\ell k),\\
T_3(F)&=\sum_{\substack{d>U,\, r>V\\x<dr\leqslant2x}}
 \mu(d)R_Z(d)c_V(r)R_Z(r)F(dr).
\end{aligned}
\end{equation*}
We refer to $T_1(F)$ and $T_2(F)$ as the Type I sums, and to $T_3(F)$ as the Type II sum.

<!-- proof-1: body_md -->
For every positive integer $n$, Vaughan's identity gives
	\begin{equation}\label{eq:vauid}
		\begin{aligned}
			\Lambda(n)
			=\Lambda(n)\mathbf{1}_{n\leqslant V}
			+\sum_{\substack{dk=n\\d\leqslant U}}\mu(d)\log k
			-\sum_{\substack{d\ell k=n\\d\leqslant U,\,\ \ell\leqslant V}}
			\mu(d)\Lambda(\ell)
			+\sum_{\substack{d\ell k=n\\d>U,\,\ \ell>V}}
			\mu(d)\Lambda(\ell).
		\end{aligned}
	\end{equation}
For $x<n\leqslant2x$, the first term of \eqref{eq:vauid}
is zero, since $V<x$. By the definition of $c_V(r)$ above, we multiply \eqref{eq:vauid} by $R_Z(n)F(n)$ and sum over $x<n\leqslant2x$, getting
\begin{equation}\label{eq:vaughan-sifted-sum}
	\sum_{x<n\leqslant2x}\Lambda(n)R_Z(n)F(n)
	=T_1(F)-T_2(F)+T_3(F).
\end{equation}
For a prime power $p^j$, we have $R_Z(p^j)=0$
if and only if $p\leqslant Z$. Hence
\begin{align*}
	\bigg|\sum_{x<n\leqslant2x}\Lambda(n)F(n)
	-\sum_{x<n\leqslant2x}\Lambda(n)R_Z(n)F(n)\bigg|
	=\bigg|\sum_{p\leqslant Z}
	\sum_{\substack{j\geqslant1\\x<p^j\leqslant2x}}
	(\log p)F(p^j)\bigg|\ll\|F\|_\infty Z.
\end{align*}
Combining this estimate with \eqref{eq:vaughan-sifted-sum}, we conclude \eqref{eq:vaughan}.

<!-- proposition-2: body_md -->
\label{prop:mel}
Let $\delta,\,B,\,C>0$ be fixed. Suppose that
$x^\delta\leqslant D\leqslant2x$ and $I\subseteq[D,2D]$ is an interval.
Let $\chi$ be a Dirichlet character modulo $q$ with $1\leqslant q\leqslant (\log x)^B$. Then for any $A>0$
\begin{equation}\label{eq:mellin}
	\sup_{|t|\leqslant x^C}
	\bigg|\sum_{d\in I}\mu(d)R_Z(d)\chi(d)d^{it}\bigg|
	\ll_{A,\,B,\,C,\,\delta}D L^{-A}.
\end{equation}
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
Let $\mathcal X\geqslant3$ and $1\leqslant r\leqslant\mathcal X$. Let $\alpha_0\in(0,1]$ be fixed.
Then for any character $\chi$ (mod $q$) with $q\leqslant (\log\mathcal X)^{A_0}$ and any $A_0>0$, one has (see \cite[Lemma 3.9(ii)]{MSTT})
\begin{equation}\label{eq:muinput}
\begin{aligned}
\sup_{|t|\leqslant\mathcal X}
 \sup_{I\subset[\mathcal X^{\alpha_0},\,2\mathcal X^{\alpha_0}]}
 \biggl|\sum_{n\in I}\frac{\mu(rn)\chi(n)}{n^{1/2+it}}\biggr|
 \ll_{\alpha_0,\,A_0}
 \mathcal X^{\alpha_0/2}(\log\mathcal X)^{-A_0}.
\end{aligned}
\end{equation}

Let $H$ satisfy $x^{\delta/2}\leqslant H\leqslant2x$.
We first consider the sum over subintervals of $[H,\,2H]$. 
Choose $R>\max\{1,\,2(C+1)/\delta\}$ depending only on
$\delta$ and $C$, and put
\begin{equation*}
	\mathcal X=H^R,\,\qquad \alpha_0=1/R.
\end{equation*}
Then $H=\mathcal X^{\alpha_0}$ and
$\mathcal X\geqslant x^{R\delta/2}>x^{C+1}$, so the range
$|t|\leqslant x^C$ is covered by that in \eqref{eq:muinput}.

For each fixed $A_0$, the implied constant in
\eqref{eq:muinput} is independent of $H$, because
$\alpha_0=1/R$ is fixed. Also,
$\log\mathcal X\asymp_{\delta,\,C}L$.

Increase $A_0$ to cover the modulus range and the required saving, replace $t$ by $-t$, and apply
partial summation only to the real weight $n^{1/2}$. This gives
\begin{equation}\label{eq:muplain}
\begin{aligned}
\sup_{|t|\leqslant x^C}\sup_{I\subset[H,\,2H]}
 \bigl|\sum_{n\in I}\mu(n)\chi(n)n^{it}\bigr|
 \ll_{A,\,B,\,C,\,\delta}H L^{-A}.
\end{aligned}
\end{equation}
The passage from \eqref{eq:muinput} to \eqref{eq:muplain} does not differentiate $n^{it}$, so it
incurs no loss in $|t|$.

The following identity
\begin{equation*}
\begin{aligned}
\mu(n)R_Z(n)=\sum_{\substack{a\mid n\\P^+(a)\leqslant Z}}\mu(n/a),
\end{aligned}
\end{equation*}
handles the indicator $R_Z(n)$. It can be checked prime by prime:
for $p\leqslant Z$, the right-hand side at $p^j$ is $1-1=0$ for $j\geqslant1$, whereas for $p>Z$ it is
$\mu(p^j)$. The sum on the left of \eqref{eq:mellin} therefore equals
\begin{equation}\label{eq:musum}
\begin{aligned}
\sum_{\substack{a\leqslant2D\\P^+(a)\leqslant Z}}\chi(a)a^{it}
\sum_{k\in I/a}\mu(k)\chi(k)k^{it}.                 
\end{aligned}
\end{equation}
For $a\leqslant D^{1/2}$, put $H=D/a$. Then
$I/a\subseteq[H,\,2H]$ and $x^{\delta/2}\leqslant H\leqslant2x$.
Applying \eqref{eq:muplain} with $A$ replaced by $A+1$,
we obtain
\begin{equation*}
	\sup_{|t|\leqslant x^C}
	\bigg|\sum_{k\in I/a}\mu(k)\chi(k)k^{it}\bigg|
	\ll_{A,\,B,\,C,\,\delta}\frac{D}{a}L^{-A-1}.
\end{equation*}
Moreover, one has
\begin{equation*}
	\sum_{P^+(a)\leqslant Z}\frac1a
	=\prod_{p\leqslant Z}(1-p^{-1})^{-1}
	\ll\log Z,
\end{equation*}
whence, by $\log Z=L^{0.24}$
\begin{align*}
	\sup_{|t|\leqslant x^C}
	\bigg|
	\sum_{\substack{a\leqslant D^{1/2}\\P^+(a)\leqslant Z}}
	\chi(a)a^{it}\sum_{k\in I/a}\mu(k)\chi(k)k^{it}
	\bigg|
	\ll	DL^{-A-1}
	\sum_{\substack{a\leqslant D^{1/2}\\P^+(a)\leqslant Z}}\frac1a
	\ll DL^{-A}.
\end{align*}

For $a>D^{1/2}$, take $\eta=1/\log Z$. Rankin's trick gives
\begin{equation}\label{eq:smtail}
\begin{aligned}
\sum_{\substack{a>D^{1/2}\\P^+(a)\leqslant Z}}\frac1a
\leqslant D^{-\frac{\eta}{2}}\prod_{p\leqslant Z}(1-p^{-1+\eta})^{-1}\ll \exp\biggl(-\frac{\log D}{2\log Z}\biggr)\log Z
\ll L^{0.24}\exp(-c_\delta L^{0.76}).
\end{aligned}
\end{equation}
For the second inequality we used
\begin{equation*}
\begin{aligned}
\log\frac{\prod_{p\leqslant Z}(1-p^{-1+\eta})^{-1}}
              {\prod_{p\leqslant Z}(1-p^{-1})^{-1}}
=\sum_{p\leqslant Z}\sum_{j\geqslant1}
       \frac{p^{-j(1-\eta)}-p^{-j}}j\ll\sum_{p\leqslant Z}\frac{p^\eta-1}p
 \ll\eta\sum_{p\leqslant Z}\frac{\log p}p\ll1,
\end{aligned}
\end{equation*}
since $\eta=1/\log Z$.
The inner sum
in \eqref{eq:musum} has at most $2D/a$ terms, so \eqref{eq:smtail} proves the required saving for
$a>D^{1/2}$.

<!-- proposition-3: body_md -->
\label{prop:rem}
	For $1\leqslant j\leqslant b$, let $\mathcal W_j$ be a weight of the form
	\eqref{eq:mark}, constructed from $K$ sets of primes as in
	\eqref{eq:prime-sets}. Define $m_j$ by \eqref{eq:markexp}
	with $\mathcal W$ replaced by $\mathcal W_j$.
	Assume that the prime sets for different weights have no primes in common. Then
	\begin{equation}\label{eq:004}
		\begin{aligned}
			\frac1x\sum_{x<p\leqslant2x}\log p
			\biggl(\frac1b\sum_{j=1}^b
			\frac{\mathcal W_j(p+2)}{m_j}-1\biggr)^2
			\leqslant\frac{C_{K,\,\rho}}b+o(1).
		\end{aligned}
	\end{equation}
	Here $b,K$ and all exponents are fixed before $x\to\infty$.
<!-- end proposition-3 -->
<!-- proof-3: body_md -->
Let $\mathcal F$ be either $\mathcal W_j$ or $\mathcal W_j\mathcal W_k$ with $j\ne k$. There are $K$ prime sets in the first case and $2K$ in the second. Denote their number by $s$ and write them as $\mathcal P_1,\ldots,\mathcal P_s$. Thus
	\begin{equation*}
		\mathcal F(n)=\prod_{i=1}^{s}\frac{\omega_i(n)\rho^{\omega_i(n)-1}}{V_i}.
	\end{equation*}
	
	Choosing $\ell$ distinct primes from $\{p\in\mathcal P_i:p\mid n\}$ is equivalent to choosing a squarefree divisor of $n$ with $\ell$ prime factors, all in $\mathcal P_i$. Hence
	\begin{equation*}
		\binom{\omega_i(n)}{\ell}=\sum_{\substack{d\mid n,\,\mu^2(d)=1\\p\mid d\Rightarrow p\in\mathcal P_i,\,\omega(d)=\ell}}1.
	\end{equation*}
	Differentiating $\rho^u=\sum_{\ell=0}^{u}\binom u\ell(\rho-1)^\ell$ gives
	\begin{equation}\label{eq:005}
		u\rho^{u-1}=\sum_{\ell=1}^{u}\ell(\rho-1)^{\ell-1}\binom u\ell\qquad(u\geqslant0),
	\end{equation}
	where $u$ is an integer and the sum is empty when $u=0$.
	
	For an integer $R\geqslant1$, define
	\begin{equation*}
		\mathcal F_R(n)=\prod_{i=1}^{s}\biggl\{\frac1{V_i}\sum_{\ell=1}^{R}\ell(\rho-1)^{\ell-1}\binom{\omega_i(n)}{\ell}\biggr\},
	\end{equation*}
	with $\binom{\omega_i(n)}{\ell}=0$ for $\ell>\omega_i(n)$. Subtracting this expression from the full expansion gives
	\begin{equation*}
		\mathcal F(n)-\mathcal F_R(n)=\sum_{\substack{\ell_1,\ldots,\ell_s\geqslant1\\\max_{1\leqslant i\leqslant s}\ell_i>R}}\prod_{i=1}^{s}\biggl\{\frac{\ell_i(\rho-1)^{\ell_i-1}}{V_i}\binom{\omega_i(n)}{\ell_i}\biggr\}.
	\end{equation*}
	
	We first estimate this difference averaged over integers. For each $i$, we have
	\begin{equation*}
		\sum_{\substack{\mu^2(d)=1,\,\omega(d)=\ell\\p\mid d\Rightarrow p\in\mathcal P_i}}\frac1d\leqslant\frac{V_i^\ell}{\ell!}.
	\end{equation*}
	Divisors supported on different $\mathcal P_i$ are coprime. Expanding the binomial coefficients and using $\lfloor3x/(d_1\cdots d_s)\rfloor\leqslant3x/(d_1\cdots d_s)$ therefore gives
	\begin{equation*}
		\sum_{n\leqslant3x}\prod_{i=1}^{s}\binom{\omega_i(n)}{\ell_i}\leqslant3x\prod_{i=1}^{s}\frac{V_i^{\ell_i}}{\ell_i!}.
	\end{equation*}
	Since $V_i=\log2+o(1)<1$ for sufficiently large $x$, we have
	\begin{equation*}
		\sum_{\ell\geqslant1}\frac{((1-\rho)V_i)^{\ell-1}}{(\ell-1)!}\leqslant e,\qquad \sum_{\ell>R}\frac{((1-\rho)V_i)^{\ell-1}}{(\ell-1)!}\leqslant\frac e{R!}.
	\end{equation*}
	In every omitted term, at least one $\ell_i$ exceeds $R$. Summing over the possible choices of this index yields
	\begin{equation}
		\begin{aligned}
			\sum_{n\leqslant3x}|\mathcal F(n)-\mathcal F_R(n)|&\leqslant3x\sum_{\substack{\ell_1,\ldots,\ell_s\geqslant1\\\max_{1\leqslant i\leqslant s}\ell_i>R}}\prod_{i=1}^{s}\frac{((1-\rho)V_i)^{\ell_i-1}}{(\ell_i-1)!}\\
			&\leqslant\frac{3se^s x}{R!}\ll_{K,\,\rho}\frac{x}{R!}.
		\end{aligned}
	\end{equation}
	
	We next estimate the difference over reduced residue classes. Let $P$ be the product of all primes used to define $\mathcal W_1,\ldots,\mathcal W_b$. Since $P$ is odd, the Chinese remainder theorem gives
	\begin{equation*}
		\frac1{\varphi(P)}\sum_{\substack{a\,(\textrm{mod}\, P)\\(a,\,P)=1}}\mathbf1_{d\mid a+2}=\frac1{\varphi(d)}\qquad(d\mid P).
	\end{equation*}
	For squarefree divisors supported on $\mathcal P_i$,
	\begin{equation*}
		\sum_{\substack{\mu^2(d)=1,\,\omega(d)=\ell\\p\mid d\Rightarrow p\in\mathcal P_i}}\frac1{\varphi(d)}\leqslant\frac1{\ell!}\biggl(\sum_{p\in\mathcal P_i}\frac1{p-1}\biggr)^\ell.
	\end{equation*}
	Since the prime sum on the right tends to $\log2<1$ and $V_i\gg1$, the corresponding tail satisfies
	\begin{equation*}
		\frac1{V_i}\sum_{\ell>R}\frac{\ell(1-\rho)^{\ell-1}}{\ell!}\biggl(\sum_{p\in\mathcal P_i}\frac1{p-1}\biggr)^\ell\leqslant\frac{e}{V_iR!}\ll\frac1{R!}.
	\end{equation*}
	Taking $R=0$ also bounds the full sum by an absolute constant. Applying these bounds to the expansion of $\mathcal F-\mathcal F_R$, with at least one $\ell_i>R$, gives
	\begin{equation*}
		\frac1{\varphi(P)}\sum_{\substack{a\,(\textrm{mod}\, P)\\(a,\,P)=1}}|\mathcal F(a+2)-\mathcal F_R(a+2)|\ll_{K,\,\rho}\frac1{R!}.
	\end{equation*}
	
	Now take $R=\lceil\log L\rceil$ and write $\mathcal F_R(n)=\sum_{d\mid n}c_d$. The disjointness of the prime sets gives a unique decomposition of each retained divisor into its factors supported on those sets. Consequently,
	\begin{equation*}
		d\leqslant\exp(4KRL^{0.2})=x^{o(1)},\qquad |c_d|\ll_{K,\,\rho}R^{2K}.
	\end{equation*}
	Every retained $d$ is squarefree and odd, so $d\mid p+2$ is the reduced residue class $p\equiv-2\,(\textrm{mod}\, d)$. The Bombieri--Vinogradov theorem, with sufficient logarithmic saving to absorb $R^{2K}$, gives
	\begin{equation*}
		\frac1x\sum_{x<p\leqslant2x}\log p\,\mathcal F_R(p+2)=\sum_d\frac{c_d}{\varphi(d)}+o(1).
	\end{equation*}
	The Chinese remainder theorem identifies the main term as
	\begin{equation*}
		\sum_d\frac{c_d}{\varphi(d)}=\frac1{\varphi(P)}\sum_{\substack{a\,(\textrm{mod}\, P)\\(a,\,P)=1}}\mathcal F_R(a+2).
	\end{equation*}
	
	Since $p+2\leqslant3x$ for $x<p\leqslant2x$ and sufficiently large $x$, the integer-average error is $O_{K,\,\rho}(L/R!)$ after multiplication by $\log p$ and division by $x$. Thus $L/R!\to0$ and the two remainder estimates give
	\begin{equation}\label{eq:rem-comparison}
		\frac1x\sum_{x<p\leqslant2x}\log p\,\mathcal F(p+2)=\frac1{\varphi(P)}\sum_{\substack{a\,(\textrm{mod}\, P)\\(a,\,P)=1}}\mathcal F(a+2)+o(1).
	\end{equation}
	
	By the definition of $m_j$ and the Chinese remainder theorem,
	\begin{equation*}
		\frac1{\varphi(P)}\sum_{\substack{a\,(\textrm{mod}\, P)\\(a,\,P)=1}}\mathcal W_j(a+2)=m_j.
	\end{equation*}
	For $j\ne k$, the prime sets defining the two weights are disjoint, so
	\begin{equation*}
		\frac1{\varphi(P)}\sum_{\substack{a\,(\textrm{mod}\, P)\\(a,\,P)=1}}\mathcal W_j(a+2)\mathcal W_k(a+2)=m_jm_k.
	\end{equation*}
	Combining these identities with \eqref{eq:rem-comparison}, together with the estimate $m_j\gg 1$, we deduce that for $j\ne k$
	\begin{equation*}
		\begin{aligned}
			\frac1x\sum_{x<p\leqslant2x}\log p\,\frac{\mathcal W_j(p+2)}{m_j}&=1+o(1),\\[1mm]
			\frac1x\sum_{x<p\leqslant2x}\log p\,\frac{\mathcal W_j(p+2)\mathcal W_k(p+2)}{m_jm_k}&=1+o(1).
		\end{aligned}
	\end{equation*}
	
	Moreover, $0\leqslant\mathcal W_j(n)\ll 1$ and $m_j\gg 1$ imply
	\begin{equation*}
		\biggl(\frac{\mathcal W_j(n)}{m_j}\biggr)^2\ll \frac{\mathcal W_j(n)}{m_j}.
	\end{equation*}
	The first-moment estimate therefore gives
	\begin{equation*}
		\frac1x\sum_{x<p\leqslant2x}\log p\biggl(\frac{\mathcal W_j(p+2)}{m_j}\biggr)^2\ll 1.
	\end{equation*}
	
	Finally, expanding the square and using the prime number theorem, we get
	\begin{equation*}
		\begin{aligned}
			\frac1x\sum_{x<p\leqslant2x}\log p\biggl(\frac1b\sum_{j=1}^{b}\frac{\mathcal W_j(p+2)}{m_j}-1\biggr)^2

Thus, with $\tau_3(m)$ denoting the number of ordered factorizations $m=d_1d_2d_3$,
	\begin{equation*}
		T_m\leqslant2\sum_{d_1d_2d_3=m}\frac m{d_1}\frac m{d_2}\frac m{d_3}=2m^2\tau_3(m).
	\end{equation*}
	
	For every unit $s$ modulo $m$, substitution in the defining sum gives $K_m(hs,\,ks^{-1})=K_m(h,\,k)$. The stabilizer consists of the units satisfying $s\equiv1\,(\textrm{mod}\,m/(h,\,k,\,m))$. Reduction between the corresponding unit groups is surjective, so there are exactly $\varphi(m/(h,\,k,\,m))$ distinct pairs $(hs,\,ks^{-1})$. Consequently,
	\begin{equation*}
		|K_m(h,\,k)|^4\leqslant\frac{2m^4\tau_3(m)}{\varphi(m/(h,\,k,\,m))}.
	\end{equation*}
	Using $\tau_3(m)\ll_\varepsilon m^\varepsilon$ and $m/\varphi(m)\ll_\varepsilon m^\varepsilon$, which follow by separating the finitely many small primes, we obtain
	\begin{equation}\label{eq:app-kloosterman}
		|K_m(h,\,k)|\ll_\varepsilon m^{3/4+\varepsilon}(h,\,k,\,m)^{1/4}.
	\end{equation}
	This includes even moduli and frequencies not coprime to the modulus.
	
	Suppose first that $d_0YM\leqslant N$. Choose integer representatives $g_0=\begin{pmatrix}u_0&c_0\\v_0&d'_0\end{pmatrix}$ and fix $u\equiv u_0\,(\textrm{mod}\,S)$. Let $\mathscr R_u$ be the set of pairs $(c,\,v)$ modulo $uS$ satisfying
	\begin{equation*}
		c\equiv c_0\,(\textrm{mod}\,S),\qquad v\equiv v_0\,(\textrm{mod}\,S),\qquad cv\equiv ud'_0-1\,(\textrm{mod}\,uS).
	\end{equation*}
	These congruences are equivalent to the prescribed matrix congruence once $d=\frac{1+cv}u$. The last congruence implies $(u,\,c)=(u,\,v)=1$. At primes dividing both $u$ and $S$, coprimality of $c$ and $u$ follows from $u_0d'_0-c_0v_0\equiv1\,(\textrm{mod}\,S)$; at the other primes dividing $u$, it excludes the proportion $1/p$ of possible lifts. For each admissible $c$, writing $v=v_0+Sj$ gives a unique $j$ modulo $u$. Therefore
	\begin{equation*}
		\#\mathscr R_u=u\prod_{\substack{p\mid u,\ p\nmid S}}\biggl(1-\frac1p\biggr).
	\end{equation*}
	
	We next show that the pairs in $\mathscr R_u$ are uniformly distributed in rectangles of the torus $(\mathbb R/\mathbb Z)^2$. Put $S_1=(u,\,S)$ and $S_2=\frac S{S_1}$, and choose $\gamma$ with $\gamma S_2\equiv1\,(\textrm{mod}\,uS_1)$. Since $S$ is squarefree, $(S_2,\,uS_1)=1$. Modulo $S_2$, both coordinates are fixed; modulo $uS_1$, they satisfy $cv\equiv ud'_0-1$ and $c\equiv c_0\,(\textrm{mod}\,S_1)$. The latter condition also forces $v\equiv v_0\,(\textrm{mod}\,S_1)$. The Chinese remainder theorem and additive orthogonality give
	\begin{equation*}
		\biggl|\sum_{(c,\,v)\in\mathscr R_u}e\biggl(\frac{hc+kv}{uS}\biggr)\biggr|\leqslant\frac1{S_1}\sum_{j\,(\textrm{mod}\,S_1)}\bigl|K_{uS_1}\bigl(\gamma h+ju,\,\gamma k(ud'_0-1)\bigr)\bigr|.
	\end{equation*}
	Both $\gamma$ and $ud'_0-1$ are coprime to $uS_1$. For $0<\max\{|h|,\,|k|\}\leqslant u^{0.02}$, we have
	\begin{equation*}
		(\gamma h+ju,\,\gamma k(ud'_0-1),\,uS_1)\leqslant S_1(h,\,k,\,u)\leqslant S_1u^{0.02}.
	\end{equation*}
	Applying \eqref{eq:app-kloosterman} with $\varepsilon=1/200$ we have for $0<\max\{|h|,\,|k|\}\leqslant u^{0.02}$
	\begin{equation}\label{eq:app-root-fourier}
		\biggl|\sum_{(c,\,v)\in\mathscr R_u}e\biggl(\frac{hc+kv}{uS}\biggr)\biggr|\ll u^{0.760}S_1^{1.005}\leqslant u^{0.760}S^2.
	\end{equation}
	
	To pass from \eqref{eq:app-root-fourier} to interval counts, take $\Delta=\frac1{100}u^{-0.01}$. For any interval $J$ in $\mathbb R/\mathbb Z$, enlargement or contraction by $2\Delta$, followed by convolution with a smooth probability density supported within distance $\Delta$, gives functions $f_J^-,\,f_J^+$ satisfying
	\begin{equation*}
		0\leqslant f_J^-\leqslant\mathbf1_J\leqslant f_J^+\leqslant1,\qquad \widehat f_J^\pm(0)=|J|+O(\Delta),\qquad |\widehat f_J^\pm(n)|\ll_j(1+\Delta|n|)^{-j}.
	\end{equation*}
	Here $\widehat f(n)=\int_0^1f(t)e(-nt)\,dt$. If $|J|<4\Delta$, take $f_J^-=0$; if the enlarged interval fills the circle, take $f_J^+=1$. The same bounds remain valid. For either choice of sign,
	\begin{equation*}
		\sum_{n\in\mathbb Z}|\widehat f_J^\pm(n)|\ll\Delta^{-1},\qquad \sum_{|n|>u^{0.02}}|\widehat f_J^\pm(n)|\ll\Delta^{-10}(u^{0.02})^{-9}.
	\end{equation*}
	Apply the products of these upper and lower functions to two torus intervals $J_1,\,J_2$. The zero frequency contributes $(\#\mathscr R_u)|J_1||J_2|+O(u\Delta)$. There are $O(u^{0.04})$ nonzero frequency pairs with $\max\{|h|,\,|k|\}\leqslant u^{0.02}$, while
	\begin{equation*}
		\sum_{\max\{|h|,\,|k|\}>u^{0.02}}|\widehat f_{J_1}^\pm(h)\widehat f_{J_2}^\pm(k)|\ll\Delta^{-11}(u^{0.02})^{-9}\ll u^{-0.07}.
	\end{equation*}
	Using the trivial bound $\#\mathscr R_u\leqslant u$ for these remaining frequencies, we obtain
	\begin{equation}\label{eq:app-root-discrepancy}
		\begin{aligned}
			\biggl|\sum_{(c,\,v)\in\mathscr R_u}\mathbf1_{J_1}\biggl(\frac c{uS}\biggr)\mathbf1_{J_2}\biggl(\frac v{uS}\biggr)-(\#\mathscr R_u)|J_1||J_2|\biggr|&\ll u^{0.99}+u^{0.80}S^2+u^{0.93}\ll u^{0.99}S^2.
		\end{aligned}
	\end{equation}
	The arguments of the indicators are taken modulo one. In particular, the error is independent of the lengths and locations of $J_1,\,J_2$.
	
	For fixed $u$, the restriction $\frac cu\in I_3$ corresponds to a torus interval of length $\frac{|I_3|}S$. Split the interval $NI_2$ for $v$ at multiples of $uS$; it has $O(\frac Nu+1)$ pieces, with total normalized length $\frac{N|I_2|}{uS}$. Thus \eqref{eq:app-root-discrepancy} gives
	\begin{equation*}
		\sum_{\substack{v,\,c\in\mathbb Z,\,0\leqslant c<u\\ \frac vN\in I_2,\,\frac cu\in I_3\\ (c,\,v)\,(\textrm{mod}\,uS)\in\mathscr R_u}}1=\frac{N|I_2||I_3|}{S^2}\prod_{\substack{p\mid u\\p\nmid S}}\biggl(1-\frac1p\biggr)+O\bigl((\frac Nu+1)u^{0.99}S^2\bigr).
	\end{equation*}
	Summing the errors over $u\in d_0YM I_1$ gives $O(d_0YMN(d_0YM)^{-0.01}S^2)$. To sum the main terms, expand the product into divisors. Since $(\ell,\,S)=1$, the conditions $\ell\mid u$ and $u\equiv u_0\,(\textrm{mod}\,S)$ specify one residue class modulo $\ell S$, whether or not $(u_0,\,S)=1$. Consequently,
	\begin{equation*}
		\begin{aligned}
			\sum_{\substack{\frac u{d_0YM}\in I_1\\u\equiv u_0\,(\textrm{mod}\,S)}}\prod_{\substack{p\mid u\\p\nmid S}}\biggl(1-\frac1p\biggr)&=\sum_{\substack{\ell\leqslant16d_0YM\\(\ell,\,S)=1}}\frac{\mu(\ell)}\ell\biggl(\frac{d_0YM|I_1|}{\ell S}+O(1)\biggr)\\
			&=\frac{d_0YM|I_1|}{S\zeta(2)}\prod_{p\mid S}(1-p^{-2})^{-1}+O\bigl(\log(2d_0YM)\bigr).
		\end{aligned}
	\end{equation*}
	Here extending the series in $\ell$ to infinity costs $O(1/S)$. Since
	\begin{equation*}
		|\mathrm{SL}_2(\mathbb Z/S\mathbb Z)|=\prod_{p\mid S}p(p^2-1)=S^3\prod_{p\mid S}(1-p^{-2}),
	\end{equation*}
	we obtain the main term in \eqref{eq:app-root-count}. The additional error is absorbed because
	\begin{equation*}
		\frac{N\log(2d_0YM)}{S^2}\ll d_0YMN(d_0YM)^{-0.01}S^2.
	\end{equation*}
	
	Finally suppose that $N<d_0YM$. Since $u,\,v>1$, the normalization $0\leqslant c<u$ implies $0<d<v$. The matrix
	\begin{equation*}
		\widetilde g=\begin{pmatrix}v&v-d\\u&u-c\end{pmatrix}\in\mathrm{SL}_2(\mathbb Z),\qquad \frac{v-d}{v}=1-\frac cu-\frac1{uv},
	\end{equation*}
	is the normalized completion of $(v,\,u)^T$, and $g\mapsto\widetilde g$ is an involution, also modulo $S$. Apply the case already proved with the two coordinate scales and intervals interchanged. The condition $\frac cu\in I_3$ is bounded above and below by requiring $\frac{v-d}{v}$ to lie in the interval $1-I_3$ enlarged or contracted by $\frac1{d_0YMN}$, with intersection with $[0,\,1]$. These changes alter the predicted main term by $O(1)$, which is absorbed by the error in \eqref{eq:app-root-count}. This proves the lemma for both orders of the coordinate scales.
