<!-- theorem-1: body_md -->
\label{thm:nonpositive}
 	Every biharmonic hypersurface in a space form $N^{n+1}(c)$ with $c\le0$ is minimal.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{cor:one-point}
Let $M^n$ be a biharmonic hypersurface in $\mathbb S^{n+1}$. If $H^2(p)\ge4(n-1)/[n(n+8)]$ at some point $p\in M^n$, then $M^n$ has constant mean curvature.
<!-- end theorem-2 -->
<!-- corollary-1: body_md -->
\label{cor:scalar-point}
Let $M^n$ be a biharmonic hypersurface in $\mathbb S^{n+1}$ {\rm ($n\ge5$)}. If $\operatorname{Scal}(p)\ge n(n^2+10n-20)/(n+8)$ at one point, then $M^n$ has constant mean curvature.
<!-- end corollary-1 -->
<!-- theorem-3: body_md -->
\label{thm:upper-classification}
Let $M^n$ be a biharmonic hypersurface in $\mathbb S^{n+1}$ {\rm ($n\ge4$)}. If $|H(p)|>(n-2)/n$ at one point $p$, then $M^n$ is locally congruent to an open part of $\mathbb S^n(1/\sqrt2)$. If $|H(p)|=(n-2)/n$ at one point $p$, then $M^n$ is locally congruent to an open part of
\begin{equation}\label{eq:Clifford-one}
 \mathbb S^1\!\left(\frac1{\sqrt2}\right)
 \times\mathbb S^{n-1}\!\left(\frac1{\sqrt2}\right)
 \subset\mathbb S^{n+1}.
\end{equation}
<!-- end theorem-3 -->
<!-- proposition-1: body_md -->
\label{prop:biharmonic}
The hypersurface $\phi:M^n\to N^{n+1}(c)$ is biharmonic if and only if
\begin{align}
 &\Delta H+HS=ncH,\label{eq:biharmonic-normal}\\
 &2A\nabla H+nH\nabla H=0,\label{eq:biharmonic-tangent}
\end{align}
where $S=\operatorname{trace} A^2$.
<!-- end proposition-1 -->
<!-- lemma-1: body_md -->
\label{lem:frame}
The principal curvature $\lambda$ is simple on $U\cap M_A$, that is, $\lambda_i\ne\lambda$ for $1\le i\le n-1$.
<!-- end lemma-1 -->
<!-- proposition-2: body_md -->
\label{prop:system}
Locally on $U\cap M_A$, the principal curvatures and connection coefficients satisfy
\begin{equation}\label{eq:system}
	\left\{
	\begin{aligned}
		\lambda''
		&=T\lambda'+\lambda(S-nc),\\
		\lambda_i'
		&=(\lambda_i-\lambda)a_i,
		\qquad\quad 1\le i\le n-1,\\
		a_i'
		&=a_i^2+\lambda\lambda_i+c,
		\qquad 1\le i\le n-1,
	\end{aligned}
	\right.
\end{equation}
where
\begin{equation}\label{eq:trace-def}
 \lambda=-\frac13\sum_{i=1}^{n-1}\lambda_i,\qquad
 T=\sum_{i=1}^{n-1}a_i,\qquad
 S=\lambda^2+\sum_{i=1}^{n-1}\lambda_i^2.
\end{equation}
<!-- end proposition-2 -->
<!-- proposition-3: body_md -->
\label{prop:compatibility}
Every solution of the last two equations of~\eqref{eq:system}, with~\eqref{eq:trace-def}, satisfies
\begin{equation}\label{eq:residual-polynomial}
 \lambda''-T\lambda'-\lambda(S-nc)=-\frac29\mathcal P.
\end{equation}
Consequently, the first equation in~\eqref{eq:system} is equivalent to $\mathcal P=0$ along the solution.
<!-- end proposition-3 -->
<!-- proof-1: body_md -->
Differentiating $-3\lambda'=G_{1,1}-\lambda T$, by~\eqref{eq:T-derivative} and~\eqref{eq:G-recursion}, we derive
\[
\begin{aligned}
 -3\lambda''
 &=G_{1,1}'-\lambda'T-\lambda T'\\
 &=2G_{1,2}-\lambda F_2+\lambda H_2+cH_1-\lambda'T\\
 &\quad-\lambda\bigl(F_2-3\lambda^2+(n-1)c\bigr)\\
 &=2G_{1,2}-2\lambda F_2-\lambda'T
   +\lambda H_2+3\lambda^3-(n+2)c\lambda.
\end{aligned}
\]
Using $S=H_2+\lambda^2$ and then $3\lambda'=\lambda T-G_{1,1}$, we obtain
\[
\begin{aligned}
 &\lambda''-T\lambda'-\lambda(S-nc)\\
 &=-\frac23G_{1,2}+\frac23\lambda F_2-\frac23T\lambda'
       -\frac43\lambda H_2-2\lambda^3+\frac{4n+2}{3}c\lambda\\
 &=-\frac29\bigl(3G_{1,2}-3\lambda F_2+\lambda T^2-TG_{1,1}\\
 &\quad+6\lambda H_2+9\lambda^3-3(2n+1)c\lambda\bigr).
\end{aligned}
\]
The last expression is $-2\mathcal P/9$, as required.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:analytic}
Let a smooth real solution of~\eqref{eq:system}--\eqref{eq:trace-def} be defined on a nonempty open interval $I_0$ containing $0$. After substituting the trace relation, extend the last two equations of~\eqref{eq:system} to their maximal interval $I\subset\mathbb R$, with $I_0\subset I$. Then the first equation of~\eqref{eq:system} holds on $I$.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Substituting $\lambda=-\frac13\sum_j\lambda_j$ into \eqref{eq:system} gives the autonomous system
\begin{equation}\label{eq:polynomial}
	\left\{
\begin{aligned}
 \lambda_i'&=\left(\lambda_i+\frac13\sum_{j=1}^{n-1}\lambda_j\right)a_i,\\
 a_i'&=a_i^2-\frac13\lambda_i\sum_{j=1}^{n-1}\lambda_j+c,
 \qquad 1\le i\le n-1,
\end{aligned}
	\right.
\end{equation}
where the right-hand sides of this system are polynomials in the $2n-2$ variables $\lambda_i,a_i$. The existence and uniqueness theorem gives a unique maximal solution on an open interval $I$ containing the original interval $I_0$. Since the vector field is polynomial, this solution is real analytic (see~\cite[Sections~2.6 and~4.1]{Teschl2012}).

Define $\lambda,T,S$ on $I$ by~\eqref{eq:trace-def}. These functions are analytic, as are all their derivatives. Consequently, the residual $\lambda''-T\lambda'-\lambda(S-nc)$ is real analytic on $I$. It vanishes on the original nonempty open interval $I_0$. The identity theorem for real analytic functions implies that it vanishes throughout the connected interval $I$, which yields the assertion.
<!-- end proof-2 -->
<!-- lemma-3: body_md -->
\label{lem:integrating}
The functions $\rho_i$ satisfy
\begin{align}
 &\bigl[(a_i^2+\lambda_i^2+c)\rho_i^2\bigr]'=0,
 \label{eq:pair-invariant}\\
 &\bigl[(\lambda-\lambda_i)\rho_i\bigr]'=\lambda'\rho_i,
 \label{eq:product-monotonicity}\\
 &\left(\lambda'\prod_{i=1}^{n-1}\rho_i\right)'
 =\lambda(S-nc)\prod_{i=1}^{n-1}\rho_i.
 \label{eq:weighted}
\end{align}
Moreover, $(a_i^2+\lambda_i^2+c)\rho_i^2\le1$ on $I$. In particular, for $c\le0$, one has
\begin{equation}\label{eq:D-nonpositive-bound}
\begin{aligned}
 &|a_i|\rho_i=|\rho_i'|\le1+\sqrt{-c}\,\rho_i,\\
 &|\lambda_i|\rho_i\le1+\sqrt{-c}\,\rho_i
\end{aligned}
\end{equation}
and for $c=1$, one has
\begin{equation}\label{eq:D-positive-bound}
 0<\rho_i\le1,\qquad |a_i|\rho_i\le1,\qquad
 |\lambda_i|\rho_i\le1.
\end{equation}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Using the last two equations of~\eqref{eq:system}, we compute
\[
\begin{aligned}
 (a_i^2+\lambda_i^2+c)'
 &=2a_i(a_i^2+\lambda\lambda_i+c)
   +2\lambda_i(\lambda_i-\lambda)a_i\\
 &=2a_i(a_i^2+\lambda_i^2+c).
\end{aligned}
\]
The terms containing $\lambda\lambda_i$ cancel. Since $\rho_i'=-a_i\rho_i$, it follows that
\[
\begin{aligned}
 \bigl[(a_i^2+\lambda_i^2+c)\rho_i^2\bigr]'
 &=2a_i(a_i^2+\lambda_i^2+c)\rho_i^2\\
 &\quad-2a_i(a_i^2+\lambda_i^2+c)\rho_i^2=0,
\end{aligned}
\]
which yields~\eqref{eq:pair-invariant}. Using \eqref{FU1} and the second equation of~\eqref{eq:system}, we obtain \eqref{eq:product-monotonicity}.
Since the conserved expression has initial value
\[
 \frac{a_i(0)^2+\lambda_i(0)^2+c}
 {1+a_i(0)^2+\lambda_i(0)^2+|c|}\le1,
\]
it follows from \eqref{eq:pair-invariant} that
$(a_i^2+\lambda_i^2+c)\rho_i^2\le 1$ and hence
\begin{equation}\label{eq:pair-bound}
 (a_i^2+\lambda_i^2)\rho_i^2\le1-c\rho_i^2.
\end{equation}
For $c\le0$, the bound $\sqrt{1-c\rho_i^2}\le1+\sqrt{-c}\,\rho_i$ implies~\eqref{eq:D-nonpositive-bound}. This argument uses only an upper bound for the conserved expression, so it remains valid when that expression is zero or negative. For $c=1$, equation~\eqref{eq:pair-bound} gives~\eqref{eq:D-positive-bound}.

Next, the equations for $\lambda_i$ and $\rho_i$ satisfy
\[
\begin{aligned}
 \bigl[(\lambda-\lambda_i)\rho_i\bigr]'
 &=\bigl(\lambda'-(\lambda_i-\lambda)a_i\bigr)\rho_i
   -(\lambda-\lambda_i)a_i\rho_i\\
 &=\lambda'\rho_i.
\end{aligned}
\]
Finally, differentiating the product yields
\[
 \left(\prod_{i=1}^{n-1}\rho_i\right)'
 =-T\prod_{i=1}^{n-1}\rho_i
\]
and hence
\[
\begin{aligned}
 \left(\lambda'\prod_{i=1}^{n-1}\rho_i\right)'
 &=(\lambda''-T\lambda')\prod_{i=1}^{n-1}\rho_i=\lambda(S-nc)\prod_{i=1}^{n-1}\rho_i,
\end{aligned}
\]
which gives~\eqref{eq:weighted}.

<!-- lemma-4: body_md -->
\label{lem:bounded-factors}
Let $I$ be the maximal interval of a solution of~\eqref{eq:polynomial}, and assume that the scalar equation in~\eqref{eq:system} holds on $I$. Write $t_*=\sup I$. There is no $t_0\in I$ such that, on $[t_0,t_*)$,
\begin{equation}\label{eq:bounded-hypotheses}
\begin{aligned}
 &\lambda>0,\qquad \lambda'>0,\\
 &\lambda'\prod_{i=1}^{n-1}\rho_i\ge\delta>0,\\
 &0<\rho_i\le C_i,\qquad 1\le i\le n-1,
\end{aligned}
\end{equation}
where $\delta,C_1,\ldots,C_{n-1}$ are constants.
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Suppose that~\eqref{eq:bounded-hypotheses} holds. By~\eqref{eq:pair-bound}, we find
\[
 (a_i^2+\lambda_i^2)\rho_i^2\le1+|c|C_i^2.
\]
Choose a constant $R>0$ larger than all $\sqrt{1+|c|C_i^2}$. Then
\begin{equation}\label{eq:R-bound}
 |a_i|\rho_i\le R,\qquad |\lambda_i|\rho_i\le R,
 \qquad 1\le i\le n-1.
\end{equation}
The bound for the product in~\eqref{eq:bounded-hypotheses} also gives
\begin{equation}\label{eq:lambda-positive-lower}
 \lambda'\ge\frac{\delta}{\prod_{i=1}^{n-1}\rho_i}
            \ge\frac{\delta}{\prod_{i=1}^{n-1}C_i}>0.
\end{equation}
We first bound all pairs $(\lambda_j,a_j)$ except one, and then use the trace relation to control the remaining pair.

At every $t\in[t_0,t_*)$, the equation $\sum_i\lambda_i=-3\lambda$ implies that at least one index $i$ satisfies $\lambda_i(t)\le-3\lambda(t)/(n-1)<0$. For that index, it follows that
\begin{equation}\label{eq:index-bound}
\begin{aligned}
 0<(\lambda-\lambda_i)\rho_i
 &=\left(1+\frac{\lambda}{|\lambda_i|}\right)|\lambda_i|\rho_i\\
 &\le\left(1+\frac{n-1}{3}\right)R
 =\frac{n+2}{3}R=:K.
\end{aligned}
\end{equation}
This index need not be independent of $t$. Choose an increasing sequence $t_k\to t_*$. There are only finitely many indices, so one index satisfies~\eqref{eq:index-bound} along an infinite subsequence. Relabel this index as $1$, and retain the notation $t_k$ for the subsequence.

By~\eqref{eq:product-monotonicity}, $(\lambda-\lambda_1)\rho_1$ is strictly increasing. For any $t<t_*$, choose $k$ with $t<t_k$. Then
\[
 (\lambda(t)-\lambda_1(t))\rho_1(t)
 \le(\lambda(t_k)-\lambda_1(t_k))\rho_1(t_k)\le K.
\]
Choose $t_1$ from this subsequence and put
$\varepsilon=(\lambda(t_1)-\lambda_1(t_1))\rho_1(t_1)>0$. Monotonicity now gives
\begin{equation}\label{eq:fixed-index}
 \varepsilon\le(\lambda-\lambda_1)\rho_1\le K,
 \qquad t_1\le t<t_*.
\end{equation}

For $j\ne1$, equation~\eqref{eq:bounded-hypotheses} implies
\[
 \frac1{\rho_j}\le\frac{\lambda'}{\delta}\prod_{k\ne j}\rho_k,
\]
which together with~\eqref{eq:R-bound} and \eqref{eq:product-monotonicity} yields
\begin{equation}\label{eq:log-control}
\begin{aligned}
 |(\log\rho_j)'|=|a_j|
 &\le\frac R{\rho_j}
 \le\frac{R\lambda'}\delta\prod_{k\ne j}\rho_k\\
 &\le\frac R\delta\left(\prod_{k\ne1,j}C_k\right)\lambda'\rho_1=C_{1j}\bigl[(\lambda-\lambda_1)\rho_1\bigr]',
\end{aligned}
\end{equation}
where $C_{1j}=R\delta^{-1}\prod_{k\ne1,j}C_k$. Unless specified otherwise, all these indices range from $1$ to $n-1$. Integrating~\eqref{eq:log-control} and using \eqref{eq:fixed-index}, we obtain
\[
\begin{aligned}
 \left|\log\frac{\rho_j(t)}{\rho_j(t_1)}\right|
 &\le\int_{t_1}^{t}|(\log\rho_j)'(\tau)|\,d\tau\\
 &\le C_{1j}\bigl((\lambda(t)-\lambda_1(t))\rho_1(t)-\varepsilon\bigr)\\
 &\le C_{1j}(K-\varepsilon).
\end{aligned}
\]
In particular, we have
\begin{equation}\label{eq:D-lower-other}
 \rho_j(t)\ge\rho_j(t_1)e^{-C_{1j}(K-\varepsilon)}>0,
 \qquad t_1\le t<t_*.
\end{equation}
The lower bound is independent of $t$. Equations~\eqref{eq:R-bound} and~\eqref{eq:D-lower-other} therefore bound both $|a_j|$ and $|\lambda_j|$ for every $j\ne1$.

Choose $C\ge0$ such that $|\sum_{j\ne1}\lambda_j|\le C$ on $[t_1,t_*)$. By the trace relation and~\eqref{eq:fixed-index},
\begin{equation}\label{eq:last-denominator}
0<\lambda-\lambda_1
 =4\lambda+\sum_{j\ne1}\lambda_j\le4\lambda+C.
\end{equation}
Both denominators in the following calculation are positive. Since $\lambda'>0$, equation~\eqref{eq:product-monotonicity} gives
\[
\begin{aligned}
 \frac{\lambda'}{4\lambda+C}
 \le\frac{\lambda'}{\lambda-\lambda_1}=\frac{\bigl[(\lambda-\lambda_1)\rho_1\bigr]'}
         {(\lambda-\lambda_1)\rho_1}.
\end{aligned}
\]
Integrating from $t_1$ to $t$ and applying~\eqref{eq:fixed-index}, we obtain
\begin{equation}\label{eq:lambda-log-bound}
\begin{aligned}
 \frac14\log\frac{4\lambda(t)+C}{4\lambda(t_1)+C}
 &\le\log\frac{(\lambda(t)-\lambda_1(t))\rho_1(t)}{\varepsilon}\\
 &\le\log\frac K\varepsilon.
\end{aligned}
\end{equation}
Consequently, we have
\[
 4\lambda(t)+C\le\bigl(4\lambda(t_1)+C\bigr)
                         \left(\frac K\varepsilon\right)^4.
\]
This is a uniform upper bound for $\lambda$. Inserting it into~\eqref{eq:last-denominator} and using~\eqref{eq:fixed-index}, we find
\[
\begin{aligned}
 \rho_1(t)
 &=\frac{(\lambda(t)-\lambda_1(t))\rho_1(t)}
         {\lambda(t)-\lambda_1(t)}\\
 &\ge\frac{\varepsilon}{4\lambda(t)+C}\\
 &\ge\frac{\varepsilon}
 {\bigl(4\lambda(t_1)+C\bigr)(K/\varepsilon)^4}>0.
\end{aligned}
\]
Thus $\rho_1$ also has a uniform positive lower bound. Equation~\eqref{eq:R-bound} bounds $|a_1|$ and $|\lambda_1|$, so every component of the polynomial system is bounded on $[t_1,t_*)$.

Suppose first that $t_*<\infty$. The right-hand sides of~\eqref{eq:polynomial} are bounded on the bounded range just obtained. Hence all $\lambda_i'$ and $a_i'$ are bounded on $[t_1,t_*)$. By integration, each $\lambda_i$ and $a_i$ is uniformly Lipschitz on this interval and has a finite limit as $t\uparrow t_*$. Use these limits as initial data at $t_*$. Local existence gives a solution on a neighborhood of $t_*$, and uniqueness identifies it with the original solution on the overlap. This extends the solution beyond $t_*$, contrary to maximality (see also~\cite[Corollary~2.15]{Teschl2012}).

It follows that $t_*=\infty$. However, integrating~\eqref{eq:lambda-positive-lower} gives
\[
 \lambda(t)\ge\lambda(t_1)
 +\frac{\delta}{\prod_{i=1}^{n-1}C_i}(t-t_1),
 \qquad t\ge t_1.
\]
The right-hand side tends to infinity, contradicting~\eqref{eq:lambda-log-bound}, which completes the proof of Lemma \ref{lem:bounded-factors}.

<!-- proposition-4: body_md -->
\label{prop:nonpositive-ODE}
Let $c\le0$. Every smooth real solution of~\eqref{eq:system}--\eqref{eq:trace-def} on a connected open interval satisfies $\lambda\equiv0$.
<!-- end proposition-4 -->
<!-- proof-5: body_md -->
Suppose that $\lambda$ is nonconstant. Then there exists a point at which $\lambda'\ne0$. The system is invariant under the simultaneous replacement of $\lambda,\lambda_i$ by $-\lambda,-\lambda_i$, and under the parameter reversal
\[
 (\lambda(t),\lambda_i(t),a_i(t))
 \longmapsto(\lambda(-t),\lambda_i(-t),-a_i(-t)).
\]
After translating the parameter and applying these transformations, we may assume
\[
 \lambda(0)>0,\qquad \lambda'(0)>0.
\]
Let us extend~\eqref{eq:polynomial} to its maximal interval $I$ and write $t_*=\sup I$. By Lemma~\ref{lem:analytic}, the scalar equation remains valid on $I$.

As long as $\lambda>0$, we have $S-nc\ge\lambda^2>0$. Equation~\eqref{eq:weighted} therefore shows that
\[
 \lambda'(t)\prod_{i=1}^{n-1}\rho_i(t)
 \ge \lambda'(0)\prod_{i=1}^{n-1}\rho_i(0)=:\delta>0.
\]
Since each $\rho_i$ is positive, $\lambda'>0$ as long as $\lambda>0$. Hence $\lambda$ cannot return to zero, and on the whole maximal forward interval,
\begin{equation}\label{eq:nonpositive-growth}
 \lambda(t)\ge\lambda(0)>0,\qquad
 \lambda'(t)>0,\qquad
 \lambda'(t)\prod_{i=1}^{n-1}\rho_i(t)\ge\delta>0.
\end{equation}

We first exclude finite-time breakdown. From~\eqref{eq:D-nonpositive-bound},
\[
 \rho_i'\le1+\sqrt{-c}\,\rho_i.
\]
Multiplying by $e^{-\sqrt{-c}\,t}$ and integrating gives
\[
 \rho_i(t)\le e^{\sqrt{-c}\,t}\bigl(\rho_i(0)+t\bigr),
 \qquad 0\le t<t_*.
\]
Thus, if $t_*<\infty$, every $\rho_i$ is bounded above on $[0,t_*)$. Together with~\eqref{eq:nonpositive-growth}, this contradicts Lemma~\ref{lem:bounded-factors}. Consequently,
\[
 t_*=\infty.
\]

We next prove that
\begin{equation}\label{eq:lambda-hyp-bound}
 0<\lambda(0)\le\lambda(t)\le\sqrt{-c},\qquad t\ge0.
\end{equation}
Suppose instead that $\lambda(t_0)>\sqrt{-c}$ at some $t_0\ge0$. Since $\lambda'>0$, the same inequality holds for all $t\ge t_0$. By~\eqref{eq:D-nonpositive-bound},
\[
 (\lambda-\sqrt{-c})\rho_i
 \le(\lambda-\lambda_i)\rho_i+1.
\]
Using~\eqref{eq:product-monotonicity}, we obtain
\[
\begin{aligned}
 \left(
 \frac{(\lambda-\lambda_i)\rho_i+1}{\lambda-\sqrt{-c}}
 \right)'
 &=\frac{\lambda'}{(\lambda-\sqrt{-c})^2}
   \bigl((\lambda_i-\sqrt{-c})\rho_i-1\bigr)\le0.
\end{aligned}
\]
Therefore, for $t\ge t_0$,
\[
\begin{aligned}
 0<\rho_i(t)
 &\le\frac{(\lambda(t)-\lambda_i(t))\rho_i(t)+1}
          {\lambda(t)-\sqrt{-c}}\\
 &\le\frac{(\lambda(t_0)-\lambda_i(t_0))\rho_i(t_0)+1}
          {\lambda(t_0)-\sqrt{-c}}.
\end{aligned}
\]
All the factors $\rho_i$ are then uniformly bounded above on $[t_0,\infty)$, again contradicting Lemma~\ref{lem:bounded-factors}. This shows~\eqref{eq:lambda-hyp-bound}. When $c=0$, the inequality is impossible, so it remains only to consider $c<0$.

Assume now that $c<0$. Summing the equations for $a_i$ and using $\sum_i\lambda_i=-3\lambda$, we obtain
\[
 T'=F_2-3\lambda^2+(n-1)c.
\]
By the Cauchy--Schwarz inequality, $F_2\ge T^2/(n-1)$, while~\eqref{eq:lambda-hyp-bound} gives $\lambda^2\le-c$. Hence
\begin{equation}\label{eq:T-barrier}
 T'\ge\frac{T^2}{n-1}+(n+2)c.
\end{equation}
Choose
\[
 B>\max\left\{|T(0)|,\sqrt{-(n-1)(n+2)c}\right\}.
\]
We claim that $T(t)>-B$ for all $t\ge0$. Otherwise, if $t_1$ is the first point with $T(t_1)=-B$, then $T'(t_1)\le0$, whereas~\eqref{eq:T-barrier} gives
\[
 T'(t_1)\ge\frac{B^2}{n-1}+(n+2)c>0,
\]
a contradiction.

Using $T+B>0$, $\lambda'>0$, $S\ge0$, and $c<0$, the scalar equation yields
\[
\begin{aligned}
 \lambda''+B\lambda'
 &=(T+B)\lambda'+\lambda(S-nc)\\
 &\ge -nc\,\lambda(0)>0.
\end{aligned}
\]
Multiplying by $e^{Bt}$ and integrating gives
\[
 \lambda'(t)\ge e^{-Bt}\lambda'(0)
 +\frac{-nc\,\lambda(0)}{B}(1-e^{-Bt}).
\]
Thus
\[
 \liminf_{t\to\infty}\lambda'(t)
 \ge\frac{-nc\,\lambda(0)}B>0.
\]
It follows that $\lambda(t)$ grows at least linearly for large $t$, contradicting the upper bound $\lambda(t)\le\sqrt{-c}$ in~\eqref{eq:lambda-hyp-bound}.

Therefore $\lambda$ must be constant. Substitution into the scalar equation gives $\lambda(S-nc)=0$. If $\lambda\ne0$, then $S-nc\ge\lambda^2>0$, which is impossible. Hence $\lambda\equiv0$.

<!-- proposition-5: body_md -->
\label{prop:positive-ODE}
If a smooth real solution of~\eqref{eq:system}--\eqref{eq:trace-def} with $c=1$ has nonconstant $\lambda$, then
\begin{equation}\label{eq:lambda-sphere-bound}
 \lambda^2<\frac{n(n-1)}{n+8}
\end{equation}
throughout its maximal analytic continuation.
<!-- end proposition-5 -->
<!-- proof-7: body_md -->
Apply Lemma~\ref{lem:analytic} and pass to the maximal interval $I$. Put $\lambda_*^2=n(n-1)/(n+8)$, where $\lambda_*>0$. We first prove the weak bound $\lambda^2\le\lambda_*^2$.

Suppose that $|\lambda|>\lambda_*$ at some point. The same inequality holds on a neighborhood. Since $\lambda$ is analytic and nonconstant, $\lambda'$ cannot vanish throughout this neighborhood. We may therefore choose a point at which $|\lambda|>\lambda_*$ and $\lambda'\ne0$. Using the sign change and parameter reversal from Section~\ref{sec:nonpositive}, and translating the parameter, we may assume $\lambda(0)>\lambda_*$ and $\lambda'(0)>0$. Equation~\eqref{eq:trace-Cauchy} gives
\begin{equation}\label{eq:sphere-coercivity}
\begin{aligned}
 S-n&\ge\frac{n+8}{n-1}\lambda^2-n\\
    &=\frac{n+8}{n-1}(\lambda^2-\lambda_*^2).
\end{aligned}
\end{equation}
As long as $\lambda>\lambda_*$, the right-hand side is positive. Equation~\eqref{eq:weighted} consequently gives
\[
 \lambda'(t)\prod_{i=1}^{n-1}\rho_i(t)
 \ge\lambda'(0)\prod_{i=1}^{n-1}\rho_i(0)>0.
\]
Thus $\lambda'>0$, and $\lambda$ cannot decrease to $\lambda_*$. These estimates hold throughout the maximal forward interval. By~\eqref{eq:D-positive-bound}, every $\rho_i$ is bounded above by $1$. All the hypotheses of Lemma~\ref{lem:bounded-factors} are therefore satisfied, a contradiction. This shows $\lambda^2\le\lambda_*^2$ on $I$.

We now exclude equality. After changing the sign if necessary, suppose that $\lambda(t_0)=\lambda_*$ at an interior point $t_0\in I$. The weak bound shows that $t_0$ is a maximum point of $\lambda$. Because $\lambda$ is analytic and nonconstant, the first nonzero term of $\lambda-\lambda_*$ at $t_0$ has even order and a negative coefficient. Hence there are an integer $k\ge1$ and a constant $d>0$ such that
\[
 \lambda(t)=\lambda_*-d(t-t_0)^{2k}
             +O\bigl(|t-t_0|^{2k+1}\bigr).
\]
The function $\lambda$ is positive near $t_0$. Combining the scalar equation with~\eqref{eq:sphere-coercivity}, we obtain
\begin{equation}\label{eq:sphere-contact}
 \lambda''-T\lambda'
 \ge\frac{n+8}{n-1}\lambda(\lambda^2-\lambda_*^2).
\end{equation}
The function $T$ is analytic and bounded near $t_0$. Differentiating the convergent Taylor expansion gives
\[
\begin{aligned}
 \lambda''-T\lambda'
 &=-2k(2k-1)d(t-t_0)^{2k-2}
    +O\bigl(|t-t_0|^{2k-1}\bigr),\\
 \frac{n+8}{n-1}\lambda(\lambda^2-\lambda_*^2)
 &=-2nd(t-t_0)^{2k}
    +O\bigl(|t-t_0|^{2k+1}\bigr).
\end{aligned}
\]
For $t>t_0$ sufficiently close to $t_0$, divide~\eqref{eq:sphere-contact} by $(t-t_0)^{2k-2}$ and let $t\to t_0^{+}$. The resulting inequality is $-2k(2k-1)d\ge0$, a contradiction. Thus the equality case is impossible, and~\eqref{eq:lambda-sphere-bound} follows.

<!-- theorem-4: body_md -->
\label{thm:sphere}
Let $M^n$ be a connected biharmonic hypersurface in $\mathbb S^{n+1}$. If the mean curvature is not constant, then
\begin{equation}\label{eq:spherical-global-estimates}
 H^2<\frac{4(n-1)}{n(n+8)}
 \qquad\text{on }M^n.
\end{equation}
If $n\ge5$, then in addition
\begin{equation}\label{eq:scalar-strict-bound}
 \operatorname{Scal}<\frac{n(n^2+10n-20)}{n+8}
 \qquad\text{on }M^n.
\end{equation}
<!-- end theorem-4 -->
<!-- proof-8: body_md -->
Suppose that $H$ is nonconstant and let $U=\{p\in M:\nabla H(p)\ne0\}$. At each point of $U\cap M_A$, Proposition~\ref{prop:system}  can be applied along an integral curve of $e_n$, and from Proposition~\ref{prop:positive-ODE} it follows that
\[
 H^2<\frac{4(n-1)}{n(n+8)}.
\]
Since $M_A$ is dense in $M^n$, the set $U\cap M_A$ is dense in $U$. Hence continuity yields
\begin{equation}\label{eq:weak-bound-U}
 H^2\le\frac{4(n-1)}{n(n+8)}\qquad\text{on }U.
\end{equation}

We next extend this bound to all of $M^n$. Define the open set
\[
 V=\left\{p\in M:H^2(p)>\frac{4(n-1)}{n(n+8)}\right\}.
\]
Suppose that $V\ne\varnothing$, and let $V_0$ be a connected component of $V$. Since $M$ is locally connected, $V_0$ is open in $M$. By~\eqref{eq:weak-bound-U}, $V\cap U=\varnothing$, so $H^2$ is constant on $V_0$. We use the boundary argument for connected components in~\cite{NistorOniciuc2019, BibiLoubeauOniciuc2021}.

If $p_0\in\partial_M V_0$, continuity gives $H^2(p_0)\ge4(n-1)/[n(n+8)]$. If the inequality were strict, there would be a connected neighborhood $B$ of $p_0$ contained in $V$. Since $B\cap V_0\ne\varnothing$, the maximality of the connected component $V_0$ would give $B\subset V_0$, contradicting $p_0\in\partial_M V_0$. Thus $H^2(p_0)=4(n-1)/[n(n+8)]$. On the other hand, $H^2$ is constant and strictly greater than this value on $V_0$, which contradicts continuity at $p_0$. Hence $\partial_M V_0=\varnothing$. Since $M$ is connected, $V_0=M$, contrary to the assumption that $H$ is nonconstant. Therefore
\begin{equation}\label{eq:weak-bound-M}
 H^2\le\frac{4(n-1)}{n(n+8)}\qquad\text{on }M^n.
\end{equation}

We now prove the global consequence of J.-H. Chen's estimate~\cite{ChenJH1993} independently of~\eqref{eq:weak-bound-M}. On $U$, the trace inequality~\eqref{eq:trace-Cauchy} gives
\begin{equation}\label{eq:global-S-estimate}
 S\ge\frac{n^2(n+8)}{4(n-1)}H^2.
\end{equation}
Define the open set
\[
 W=\left\{p\in M:S(p)<\frac{n^2(n+8)}{4(n-1)}H^2(p)\right\}.
\]
Suppose that $W\ne\varnothing$, and let $W_0$ be a connected component of $W$. Since $W\cap U=\varnothing$, $H^2$ is constant on $W_0$. This constant is positive, because $S\ge0$. The normal biharmonic equation therefore gives $S=n$ on $W_0$, and $S-n^2(n+8)H^2/[4(n-1)]$ is a fixed negative constant there. If $p_0\in\partial_M W_0$, continuity implies $p_0\in W$. A connected neighborhood of $p_0$ contained in $W$ and meeting $W_0$ must lie in $W_0$, again contradicting the definition of a boundary point. Thus $\partial_M W_0=\varnothing$. By connectedness, $W_0=M$, contrary to the assumption that $H$ is nonconstant. Hence $W=\varnothing$, and~\eqref{eq:global-S-estimate} holds on all of $M$.

It remains to make~\eqref{eq:weak-bound-M} strict. Set
\[
 u=\frac{4(n-1)}{n(n+8)}-H^2.
\]
Then $u$ is smooth and nonnegative. Since $\Delta=-\operatorname{div}\nabla$, the normal biharmonic equation gives $\operatorname{div}(\nabla H)=H(S-n)$. Hence
\begin{equation}\label{eq:strong-minimum}
\begin{aligned}
 \operatorname{div}(\nabla u)
 &=-2H\operatorname{div}(\nabla H)-2|\nabla H|^2\\
 &=-2H^2(S-n)-2|\nabla H|^2\\
 &\le-2H^2\left(\frac{n^2(n+8)}{4(n-1)}H^2-n\right)\\
 &=\frac{n^2(n+8)}{2(n-1)}H^2u.
\end{aligned}
\end{equation}
If $u(p)=0$ at some point, then $v=-u\le0$ attains its maximum value $0$ at $p$ and satisfies
\[
 \left(\operatorname{div}\nabla-\frac{n^2(n+8)}{2(n-1)}H^2\right)v\ge0.
\]
The strong maximum principle implies $u\equiv0$, so $H^2$ is constant, a contradiction. Thus the bound~\eqref{eq:spherical-global-estimates} is strict.

Finally, assume $n\ge5$. It follows from the Gauss equation and~\eqref{eq:global-S-estimate} that
\[
\begin{aligned}
 \operatorname{Scal}
 &=n(n-1)+n^2H^2-S\\
 &\le n(n-1)+\frac{3n^2(n-4)}{4(n-1)}H^2\\
 &<n(n-1)+\frac{3n(n-4)}{n+8}
 =\frac{n(n^2+10n-20)}{n+8},
\end{aligned}
\]
where the strict inequality follows from~\eqref{eq:spherical-global-estimates}. This yields~\eqref{eq:scalar-strict-bound}.

\label{prop:threshold-equality}
Let $M^n$ be a biharmonic hypersurface in $\mathbb S^{n+1}$. If $H^2(p)=4(n-1)/[n(n+8)]$ at one point, then
\begin{equation}\label{eq:threshold-constant}
 H^2\equiv\frac{4(n-1)}{n(n+8)},\qquad S\equiv n.
\end{equation}
For $n=4$, this equality holds if and only if $M$ is locally congruent to an open part of $\mathbb S^1(1/\sqrt2)\times\mathbb S^3(1/\sqrt2)\subset\mathbb S^5$. For $n\ge5$, equality in the scalar-curvature threshold, $\operatorname{Scal}(p)=n(n^2+10n-20)/(n+8)$, is equivalent to~\eqref{eq:threshold-constant}.

Note that Theorem~\ref{cor:one-point} gives constant nonzero $H$, and the normal equation gives $S=n$.

 &=\frac{n(n^2+10n-20)}{n+8},\\

\]
The converse follows from the same equation.

\label{prop:norm-equality}
Let $M^n$ be a connected biharmonic hypersurface in $\mathbb S^{n+1}$ with nonconstant mean curvature. Then
\[
 S>\frac{n^2(n+8)}{4(n-1)}H^2
\]
holds on $U\cap M_A$. Moreover, the equality set of~\eqref{eq:global-S-estimate} has empty interior. Consequently, the strict inequality holds on an open dense subset of $M^n$.

<!-- proof-13: body_md -->
Suppose that equality holds at a point of $U\cap M_A$. If $H=0$, then equality gives $S=0$, so every principal curvature vanishes, contradicting the fact that the distinguished eigenvalue $\lambda_n=-nH/2$ is simple on $U\cap M_A$. Thus $H\ne0$. Equality in~\eqref{eq:trace-Cauchy} yields~\eqref{eq:norm-equality-spectrum}. Since the number of distinct principal curvatures is locally constant on $M_A$, the hypersurface has exactly two distinct principal curvatures on a neighborhood of the point. The constant-mean-curvature theorem for this case~\cite{BMO2008} then contradicts $\nabla H\ne0$. This gives the first assertion.

Suppose next that equality holds on a nonempty open set $V$. If $\nabla H\ne0$ somewhere in $V$, then $V\cap U$ contains a nonempty open subset. Since $M_A$ is dense, this subset meets $M_A$, contradicting the first assertion. Hence $H$ is locally constant on $V$.  H^2=\frac{4(n-1)}{n(n+8)},
\]
contrary to Theorem~\ref{thm:sphere}, because $M$ is assumed to have nonconstant mean curvature. \label{prop:constant-S}
Let $M^n$ be a connected biharmonic hypersurface in $\mathbb S^{n+1}$. If $S$ is constant and $S\ge n$, then $M$ has constant mean curvature. Moreover, either $M$ is minimal or $S=n$.
<!-- end proposition-8 -->
<!-- proof-14: body_md -->
Suppose that $H$ is nonconstant. Choose an integral curve of a local principal field $e_n$ through a point of $U\cap M_A$, and apply Lemma~\ref{lem:analytic}. Let $S_0\ge n$ be the prescribed constant. The function $S-S_0$ is analytic along the extended polynomial solution and vanishes on the original interval. It therefore vanishes throughout the maximal interval $I$.

Let us apply the sign change and parameter reversal used in Section~\ref{sec:nonpositive} to arrange $\lambda(0)>0$ and $\lambda'(0)>0$. As long as $\lambda>0$, equation~\eqref{eq:weighted} gives
\[
 \left(\lambda'\prod_{i=1}^{n-1}\rho_i\right)'
 =\lambda(S_0-n)\prod_{i=1}^{n-1}\rho_i\ge0.
\]
Hence $\lambda'\prod_i\rho_i\ge\lambda'(0)\prod_i\rho_i(0)>0$. It follows that $\lambda'>0$ and $\lambda\ge\lambda(0)>0$ on the entire maximal forward interval. Since $\rho_i\le1$, Lemma~\ref{lem:bounded-factors} gives a contradiction. Thus $H$ is constant. The normal equation then reads $H(S_0-n)=0$, so $S_0=n$ when $M$ is nonminimal.
