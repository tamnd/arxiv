<!-- proposition-1: body_md -->
\label{Prop-Ou} Let $\phi:M^{m}\hookrightarrow L^m(\varepsilon)\times \mathbb{R}$ be a hypersurface and denote by $H$ its mean curvature function. Then the bitension field $\tau_2(\phi)=\tau_2$ is given by
$$
\tau_2=\left[\tau_2\right]^{\top}+\left[\tau_2\right]^{\perp}
$$
where
\begin{equation}\label{BHP1}
\begin{aligned}
\left[\tau_2\right]^{\top}= & -2 m A( \operatorname{grad} H)-m^2 H \operatorname{grad} H-2 m(m-1) \varepsilon H \cos \alpha\, T 
\end{aligned}
\end{equation}
is the tangential component, while
\begin{equation}\label{BHP2}
\begin{aligned}
\left[\tau_2\right]^{\perp} = &m \big\{-\Delta H-  H |A|^2 + \varepsilon  (m-1) H \sin^2\alpha\big\}\eta 
\end{aligned}
\end{equation}
is the normal component.
<!-- end proposition-1 -->
<!-- proposition-2: body_md -->
\label{Prop-CMC} Let $M^m$ be a {non-minimal} CMC hypersurface in $L^m(\varepsilon)\times{\mathbb R}$.
\begin{itemize}
\item[{\rm (i)}] If $M^m$ is biconservative, then it is a cylinder $M^{m-1}\times{\mathbb R}$, where $M^{m-1}$ is a non-minimal CMC hypersurface in $L^m(\varepsilon)$.
\item[{\rm (ii)}] If $M^m$ is biharmonic, then it is a cylinder $M^{m-1}\times{\mathbb R}$, where $M^{m-1}$ is a non-minimal CMC biharmonic hypersurface in $L^m(\varepsilon)$.
\end{itemize}

\label{Th-constant-angle}

Then \eqref{BC} becomes
\[
 (2\kappa+4H)\operatorname{grad} H=6HuT\,.
\]
By hypothesis $H\neq0$, and $u\neq0$. Therefore, either $T$ is identically zero and the hypersurface is a totally geodesic slice or, in a neighborhood $U$ of $M$, $T\neq 0$ and it is a principal direction. Since we are assuming that the hypersurface is non-minimal, we are left with $T\neq 0$ being a principal direction on $U$ open subset of $M$.
We put
\[
 E_4=\frac{T}{v},\qquad v=|T|>0\,.
\]
Then
\[
 AE_4=\mu E_4
\]
where $\mu$ is the principal curvature in the direction of $E_4$.
Since there are exactly two distinct principal curvatures, the
multiplicity $r$ of $\mu$ is one of
\[
 r=1,2,3.
\]
Let $\lambda\neq\mu$ denote the other principal curvature and let $ q=4-r$
be its multiplicity. Thus
\[
 \operatorname{spec}(A)
 =
 \{\underbrace{\mu,\ldots,\mu}_{r},
   \underbrace{\lambda,\ldots,\lambda}_{q}\}.
\]
Now, we set
\begin{equation}\label{xydef}
 x=\frac{\lambda}{v},
 \qquad
 y=\frac{\mu}{v}.
\end{equation}
Taking $X=E_4$ in \eqref{eq:XTheta} gives
\[
 E_4(u)=-\mu v
\]
and from $u^2+v^2=1$ we obtain
\begin{equation}\label{vprime}
 E_4(v)=u\mu=uyv.
\end{equation}
Next, we choose a unit
$\lambda$--principal vector $X$ orthogonal to $E_4$. Since
$X\perp T$, then \eqref{eq:nablaXT} gives
\[
 v\nabla_XE_4=u\lambda X.
\]
Putting $Y=E_4$ in the Codazzi equation
\[
 (\nabla_XA)Y-(\nabla_YA)X
 =
 -u\bigl(\langle Y,T\rangle X-\langle X,T\rangle Y\bigr)
\]
 and taking the scalar product with $X$ yields
\begin{equation}\label{eq:E4lambda}
 E_4(\lambda)
 =
 \frac{u\lambda}{v}(\mu-\lambda)+uv.
\end{equation}
Using $\lambda=xv$, $\mu=yv$ and \eqref{vprime} in \eqref{eq:E4lambda}, we obtain
\begin{equation}\label{xprime}
{E_4(x)=u(1-x^2).}
\end{equation}
For later use, introduce
\begin{equation}\label{zdef}
 z=\frac{E_4(y)}{u}.
\end{equation}
Since
\[
 \operatorname{tr}A=r\mu+q\lambda,
 \qquad
 |A|^2=r\mu^2+q\lambda^2,
\]
and $u^2=1-v^2$, using \eqref{eq:scal-general} we obtain
\begin{equation*}\label{scalar}
 \operatorname{Scal}_M=-12+v^2{\mathcal F}_r(x,y),
\end{equation*}
where
\begin{equation}\label{Fr}
 {\mathcal F}_r(x,y)
 =
 6+r(r-1)y^2+q(q-1)x^2+2rqxy.
\end{equation}
Assume now that $\operatorname{Scal}_M=S_0$ is constant. Differentiating
\[
 v^2{\mathcal F}_r(x,y)=S_0+12
\]
along $E_4$ and using
\eqref{vprime}, \eqref{xprime}, and \eqref{zdef}, we obtain, after dividing by the nonzero factor $uv^2$,
\begin{equation}\label{Ar}
 {\mathcal A}_r(x,y,z):=
 2y{\mathcal F}_r+(1-x^2)({\mathcal F}_r)_x+z({\mathcal F}_r)_y=0.
\end{equation}
The mean curvature can be written as
\[
 H=vh,
 \qquad
 h=h_r(x,y):=\frac{ry+qx}{4}.
\]
Using again \eqref{vprime}, \eqref{xprime}, and \eqref{zdef},
\begin{equation}\label{E4H}
 E_4(H)
 =
 uv\left(
 yh+\frac q4(1-x^2)+\frac r4z
 \right).
\end{equation}
Because $\operatorname{grad} H$ is parallel to $E_4$, the biconservative condition \eqref{BC} reduces to
\[
 (\mu+2H)E_4(H)-3Huv=0
\]
that, together with \eqref{E4H},  up to a nonzero factor $uv^2$, gives
\begin{equation}\label{Br}
 {\mathcal B}_r(x,y,z):=
 (y+2h)
 \left(
 yh+\frac q4(1-x^2)+\frac r4z
 \right)-3h=0.
\end{equation}
Thus, for each $r=1,2,3$, the functions $x,y,z$ satisfy the two
algebraic relations
\[
 {\mathcal  A}_r(x,y,z)=0,
 \qquad
 {\mathcal B}_r(x,y,z)=0.
\]
Eliminating $z$ from \eqref{Ar} and \eqref{Br} gives a polynomial
relation
\[
 {\mathcal P}_r(x,y)=0.
\]
Up to multiplication by a nonzero constant, these polynomials are
\begin{align}
 {\mathcal P}_1(x,y)
 &=
 x^4-x^3y-x^2y^2+5x^2+xy^3+4xy+3y^2,
\notag\\
 {\mathcal P}_2(x,y)
 &=
 x^4-3x^2y^2+5x^2+2xy^3+11xy+11y^2,
\notag\\
 {\mathcal P}_3(x,y)
 &=
 x^4+3x^3y-9x^2y^2+5x^2+5xy^3+32xy+71y^2.
\notag
\end{align}
Since ${\mathcal P}_r(x,y)$ is identically zero on $U$, differentiating along $E_4$
and dividing by $u\neq0$ gives
\begin{equation*}\label{Dr}
 {\mathcal D}_r(x,y,z):=
 (1-x^2)({\mathcal P}_r)_x+z({\mathcal P}_r)_y=0.
\end{equation*}
Eliminating $z$ from ${\mathcal A}_r(x,y,z)=0$ and ${\mathcal D}_r(x,y,z)=0$ gives a polynomial condition ${\mathcal G}_r(x,y)=0$. Then eliminating $y$
 from ${\mathcal P}_r(x,y)=0$ and ${\mathcal G}_r(x,y)=0$ produces a polynomial condition in $x$:
\[
 {\mathcal R}_r(x)=0.
\]
A direct expansion gives
\begin{align}
 {\mathcal R}_1(x)
 &=
 x^6(x^2-1)
 \bigl(150x^4-156x^2+55\bigr),
\notag\\
 {\mathcal R}_2(x)
 &=
 x^5(x^2-1)
 \bigl(
 26244x^{12}+15876x^{10}-257067x^8+94842x^6
 \notag\\[-1mm]
 &\hspace{35mm}
 +731421x^4-957352x^2+346060
 \bigr),
\notag\\
 {\mathcal R}_3(x)
 &=
 x^5(x^2-2)
 \Bigl(
 762048x^{12}-997488x^{10}-13970736x^8+28751214x^6
 \notag\\[-1mm]
 &\hspace{24mm}
 +55828234x^4
 -189031820x^2+131696125
 \Bigr).\notag
\end{align}
Now, the only fact needed is that each ${\mathcal R}_r$
is a nonzero polynomial. Therefore if $U$ is connected then $x$ is constant
on $U$. Equation \eqref{xprime}, together with $u\neq0$, then forces
\begin{equation*}\label{xpm1}
{x^2=1.}
\end{equation*}
This is the main rigidity step.
We now exclude  the multiplicities $r=2$ and $r=3$. Since $x^2=1$, a direct check shows that $x=\pm 1$ is not a solution of ${\mathcal R}_3(x)=0$, thus the value $r=3$ is not allowed. If $r=2$, then $x=\pm 1$ is a solution of ${\mathcal R}_2(x)=0$. 
But since $x$ is constant, the polynomial relation ${\mathcal P}_r(x,y)=0$ shows
that $y$ takes values in a finite set; hence $y$ is also constant on
the connected set $U$. Therefore from \eqref{zdef}
\[
 z=0.
\]
We now substitute $x=\pm1$ and $z=0$ into the original polynomials
${\mathcal A}_r(x,y,z)$ and ${\mathcal B}_r(x,y,z)$.
For $r=2$ we obtain, when $x=1$,
\[
 {\mathcal A}_2=4y(y+2)^2,
 \qquad
 {\mathcal B}_2=\frac12(y-1)(y+1)(2y+3),
\]
and, when $x=-1$,
\[
{\mathcal  A}_2=4y(y-2)^2,
 \qquad
 {\mathcal B}_2=\frac12(y-1)(y+1)(2y-3).
\]
In either case the two polynomials have no common real zero. Thus
also $r=2$ is not a possible value.
We conclude that
\begin{equation*}\label{r1}
 {r=1,\qquad q=3.}
\end{equation*}
Thus the principal curvature in the direction of $T$ is necessarily
simple.
We now determine the two principal curvatures. Since $x=\pm1$ is constant, the relation ${\mathcal P}_1(x,y)=0$ again implies that $y$ takes values in a finite set. Thus $y$ is constant on $U$, and \eqref{zdef} gives $z=0$. Substitution into ${\mathcal  A}_1={\mathcal B}_1=0$ gives
\[
(x,y)=(1,-2) \quad {\rm or} \quad (x,y)=(-1,2)\,.
\]
The two possibilities are exchanged by reversing the unit normal.
We therefore choose the orientation for which
\begin{equation*}\label{xyfinal}
 x=-1,\qquad y=2.
\end{equation*}
From \eqref{xydef}, we obtain
\[
 {
 \lambda=-v\quad\text{(multiplicity $3$)},\qquad
 \mu=2v\quad\text{(multiplicity $1$)}.
 }
\]
Substitution into \eqref{Fr} gives
\[
 {\mathcal F}_1(-1,2)
 =
 6\bigl(1+(-1)^2+(-1)\cdot2\bigr)=0.
\]
Hence
\begin{equation*}\label{scalminus12}
{\operatorname{Scal}_M=-12.}
\end{equation*}
Finally, since $T$ is principal and $u=\cos\alpha\neq0$, the hypersurface is
locally of Tojeiro type:
\[
 \phi(x,s)=\bigl(f_s(x),a(s)\bigr),
\]
where $f_s$ is a parallel family in ${\mathbb H}^4$. Put
\[
 \rho(s)=\frac{a'(s)}{\sqrt{1+a'(s)^2}},
\]
and recall that 
\begin{equation*}\label{vrho}
 {v=|T|=\rho.}
\end{equation*}
The principal curvatures of a Tojeiro hypersurface are
\[
 -\rho\,\kappa_1(s),\quad
 -\rho\,\kappa_2(s),\quad
 -\rho\,\kappa_3(s),\quad
 \rho'(s),
\]
where $\kappa_i(s)$ are the principal curvatures of
$f_s\subset{\mathbb H}^4$.
Since the $T$--principal curvature is simple and the other three
principal curvatures are all equal to $\lambda=-v=-\rho$, we have
\[
 -\rho\kappa_i=-\rho,\qquad i=1,2,3.
\]
As $\rho>0$,
\[
 \kappa_1=\kappa_2=\kappa_3=1.
\]
Therefore every $f_s$ is a totally umbilical hypersurface of
${\mathbb H}^4$ with principal curvature $1$, hence a horosphere.
Moreover,
\[
 \rho'=\mu=2v=2\rho.
\]
Thus
\begin{equation*}\label{rhoexp}
{\rho(s)=ce^{2s}}
\end{equation*}
for some $c>0$. The parameter $c>0$ does not produce geometrically distinct hypersurfaces. Indeed, after introducing the translated parameter
\[
\widetilde{s}=s+\frac{\log c}{2},
\]
one has
\[
\rho(\widetilde{s})=e^{2\widetilde{s}}.
\]
Thus the dependence on $c$ can be absorbed by a translation of the parameter $s$ along the parallel family of horospheres.

First, we state the classical Omori-Yau maximum principle in a form which is convenient for our purposes:
\begin{theorem}[\cite{MR215259,MR431040}, see also \cite{MR3445380}]\label{Th-Max-Princ} 

Assume that $M$ is a complete Riemannian manifold with Ricci curvature bounded from below. Let $u:M \to {\mathbb R} $ be a ${C}^2$ function. If $u$ is bounded from above,
then there exists a sequence of points $\{ p_k\}_{k \in {\mathbb N}} \subset M$ such that
\[
\Delta u(p_k) > -\,\frac{1}{k}\,, \quad \lim_{k \to \infty}|\operatorname{grad} u(p_k)|=0 \quad {\rm and}\quad \lim_{k \to \infty}u(p_k)= \sup_M u \,.
\]
\end{theorem}

\vspace{3mm}
The strategy of the proof is to apply the Omori-Yau maximum principle to the normal component of the bitension field.

First, using \eqref{BHP2}, we find:
\begin{eqnarray}\label{eq-DeltaH2<=0}\nonumber
\frac{1}{2}\Delta H^2&=&  H \Delta H -  |\operatorname{grad} H|^2 \\
&=&- [|A|^2 + (m-1)  \sin^2 \alpha \big ] H^2 -  |\operatorname{grad} H|^2 \leq 0 \,.
\end{eqnarray}
By assumption, $H^2$ is bounded from above. If ${\rm Ric}(M)$ is bounded from below, we can apply the maximum principle stated in Theorem~\ref{Th-Max-Princ} to the function $u=H^2$ and we deduce that there exists a sequence of points $\{ p_k\}_{k \in {\mathbb N}} \subset M^m$ such that
\[
\Delta H^2(p_k) > -\,\frac{1}{k} \quad {\rm and}\quad \lim_{k \to \infty}H^2(p_k)= \sup_M H^2 \,.
\]
Therefore, using \eqref{eq-DeltaH2<=0}, we deduce that 
\[
\lim_{k \to \infty}\Delta H^2(p_k) =0
\]
and so
\begin{equation}\label{eq:limite-utile}
\lim_{k \to \infty}- \left (\big [|A|^2 + (m-1)   \sin^2 \alpha \big ] H^2 \right )(p_k) =0\,.
\end{equation}
Next, we argue by contradiction and suppose that the hypersurface is not minimal. Thus
\begin{equation*}\label{eq-sup>0}
\sup_M H^2 >0\,.
\end{equation*}
Then, from \eqref{eq:limite-utile}, we infer
\begin{equation}\label{eq-max-princ-hyper}
\lim_{k \to \infty} |A|^2(p_k) =0\,.
\end{equation}

Next, we recall the general fact
\[
(m-1) H^2 \leq |A|^2 \,.
\]
This inequality, together with \eqref{eq-max-princ-hyper}, imply
\begin{equation*}\label{eq-1-limite}
\lim_{k \to \infty} H^2(p_k)=0\,,
\end{equation*}
a contradiction, as required to end the proof.

<!-- remark-2: body_md -->
The assumption on the Ricci curvature in Theorem~\ref{Th-Max-Princ} can be relaxed. Indeed, the validity of the Omori-Yau maximum principle can be ensured by either some function theoretic assumptions or by assuming a weaker lower bound on the Ricci curvature. We refer to \cite{MR3445380, MR2116555} for a thorough discussion on this topic. More precisely (see \cite[Theorem~2.5]{MR3445380}), in Theorem~\ref{Th-Max-Princ} and Theorem~\ref{Th-Max-Principle-application}, it suffices to assume
\begin{equation*}\label{eq-Ricci-bound}
{\rm Ric}(M)_x \geq - (m-1) G^2(r(x))\langle\,, \rangle_M \,,
\end{equation*}
where $r(x)$ denotes the distance function from a fixed point and $G$ is a $C^1$ function on $[0,+\infty)$ such that
\[
{\rm (i)} \, G(0)>0\, \quad {\rm (ii)}\, G'(t) \geq 0\, \quad {\rm (iii)}\, \frac{1}{G(t)} \not \in {\rm L}^1 ([0,+\infty))\,.
\]

<!-- remark-3: body_md -->
The method of proof of Theorem~\ref{Th-Max-Principle-application} does not naturally apply to the case of complete biharmonic hypersurfaces in ${\mathbb S}^m \times {\mathbb R}$. Nevertheless, we can prove a related result which requires different, somehow stronger assumptions. More specifically, we have:
{
\begin{proposition}\label{Prop-max-princ-sphere}
Let $\phi:M^{m}\hookrightarrow {\mathbb S}^m \times {\mathbb R}$ be a complete biharmonic hypersurface. Assume that
\[
 |A|^2 \geq m-1 
 \]
 and $H^2$ is of class $L^p$ for some $p>1$.
Then  $\phi:M^{m}\hookrightarrow {\mathbb S}^m \times {\mathbb R}$ is minimal.\end{proposition} 
\begin{proof} Using \eqref{BHP2} and the assumption $|A|^2 \geq m-1$ we have:
\begin{eqnarray*}
\frac{1}{2}\Delta H^2&=&  H \Delta H -  |\operatorname{grad} H|^2 \\
&=&- \big [|A|^2 - (m-1)  \sin^2 \alpha \big ] H^2 -  |\operatorname{grad} H|^2 \leq 0 \,.
\end{eqnarray*}
Now, since $H^2$ is of class $L^p$ for some $p>1$, a classical result of Yau \cite[Theorem 3]{MR417452} (see also \cite{MR662919}) implies that $H$ is constant. Assume that $H\neq 0$, then,  from Proposition~\ref{Prop-CMC} (ii),  $\phi:M^{m}\hookrightarrow {\mathbb S}^m \times {\mathbb R}$ is a cylinder $M^{m-1}\times{\mathbb R}$, where $M^{m-1}$ is a non-minimal CMC biharmonic hypersurface in ${\mathbb S}^m$. Now, in the cylinder $M^{m-1}\times{\mathbb R}$, $H\neq 0$ contradicts that $H^2$ is of class $L^p$ for some $p>1$.
\end{proof}
}






Finally, we point out that, according to the Proposition~4.1 of \cite{MR3107694}, a {\em compact} biharmonic hypersurface in ${\mathbb S}^m\times{\mathbb R}$ is a slice and, by the same argument, we conclude that there exists no compact biharmonic hypersurface in ${\mathbb H}^m\times{\mathbb R}$.
