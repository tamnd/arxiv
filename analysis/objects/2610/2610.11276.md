\label{main:reciprocity}
    Let $A$ and $B$ be finite-dimensional algebras over a splitting field and let $M$ be a skew $(A, B)$-bimodule.
    If $S$ is a simple right $A$-module and $T$ is a simple right $B$-module, then
    \[
        \left[\, {P(T)} \mid {S \otimes_A M} \,\right] = \left[\, {I(S)} \mid {\operatorname{Hom}_B(M, T)} \,\right].
    \]

\label{main:tensor-hom}
    Let $A$ and $B$ be finite-dimensional algebras over a field and let $M$ be a skew $(A, B)$-bimodule.
    If $X$ is a right $A$-module and $Y$ is a right $B$-module, then there exists an isomorphism of vector spaces
    \[
        \underline{\operatorname{Hom}}_B(X \otimes_A M, Y)
        \cong \overline{\operatorname{Hom}}_A(X, \operatorname{Hom}_B(M, Y))
    \]
    that is natural in $X$ and $Y$.

<!-- lem-1: body_md -->
\label{lem:tensor}
    Let $M$ be a skew $(A, B)$-bimodule.
    If $J$ is an injective right $A$-module, then $J \otimes_A M$ is a projective right $B$-module.
<!-- end lem-1 -->
<!-- proof-1: body_md -->
Since $J$ is injective,
    there exists a right $A$-module $K$ such that $J \oplus K \cong A^* \oplus \dotsb \oplus A^*$ by \cite[Lemma I.8.17]{SY11}.
    \index{$K$ : Injective Module}
    Then the right $B$-module $J \otimes_A M$ is a direct summand of
    \begin{align*}
        (J \otimes_A M) \oplus (K \otimes_A M)
        &\cong (J \oplus K) \otimes_A M \\
        &\cong (A^* \oplus \dotsb \oplus A^*) \otimes_A M \\
        &\cong (A^* \otimes_A M) \oplus \dotsb \oplus (A^* \otimes_A M).
    \end{align*}
    Since $A^* \otimes_A M$ is a projective right $B$-module, it is a direct summand of a free $B$-module and so is $J \otimes_A M$.
    Hence $J \otimes_A M$ is a projective right $B$-module.
<!-- end proof-1 -->
<!-- lem-2: body_md -->
\label{lem:hom}
    Let $M$ be a skew $(A, B)$-bimodule.
    If $Q$ is a projective right $B$-module, then $\operatorname{Hom}_B(M, Q)$ is an injective right $A$-module.
<!-- end lem-2 -->
<!-- proof-2: body_md -->
Since $Q$ is projective,
    there exists a right $B$-module $R$ such that $Q \oplus R \cong B \oplus \dotsb \oplus B$.
    \index{$R$ : Projective Module}
    Then the right $A$-module $\operatorname{Hom}_B(M, Q)$ is a direct summand of
    \begin{align*}
        \operatorname{Hom}_B(M, Q) \oplus \operatorname{Hom}_B(M, R)
        &\cong \operatorname{Hom}_B(M, Q \oplus R) \\
        &\cong \operatorname{Hom}_B(M, B \oplus \dots \oplus B) \\
        &\cong \operatorname{Hom}_B(M, B) \oplus \dots \oplus \operatorname{Hom}_B(M, B).
    \end{align*}
    Observe that $\operatorname{Hom}_B(M, B)$ is an injective right $A$-module since $M \otimes_B B^*$ is a projective left $A$-module,
    \cite[Proposition I.8.16(ii)]{SY11} and
    \[
        \operatorname{Hom}_F(M \otimes_B B^*, F)
        \cong \operatorname{Hom}_B(M, \operatorname{Hom}_F(B^*,F))
        \cong \operatorname{Hom}_B(M, B)
    \]
    as right $A$-modules.
    Since $\operatorname{Hom}_B(M, B)$ is an injective right $A$-module, it is a direct summand of the direct sum of finitely many copies of $A^*$ and so is $\operatorname{Hom}_B(M, Q)$ by \cite[Lemma I.8.17]{SY11}.
    Hence $\operatorname{Hom}_B(M, Q)$ is an injective right $A$-module.

    \index{$I(A)$ @ $\mathscr{I}(A)$ : Category}

<!-- lem-3: body_md -->
\label{lem:stable}
    If $Y$ is a right $B$-module and $T$ is a simple right $B$-module, then
    \[
        \underline{\operatorname{Hom}}_B(Y, T) \cong \operatorname{Hom}_B(\Omega^0(Y), T).
    \]
<!-- end lem-3 -->
<!-- proof-3: body_md -->
From the definition of $\underline{\operatorname{Hom}}$ and $\Omega^0$,
    \[
        \underline{\operatorname{Hom}}_B(Y, T) \cong \underline{\operatorname{Hom}}_B(\Omega^0(Y), T).
    \]
    Hence it suffices to prove that the canonical epimorphism
    \[
        \operatorname{Hom}_B(\Omega^0(Y), T) \to \underline{\operatorname{Hom}}_B(\Omega^0(Y), T)
    \]
    is a monomorphism.

    Let $\alpha \colon \Omega^0(Y) \to T$ factor through a projective right $B$-module $Q$ as $\alpha = \gamma \circ \beta$.
    \index{$\alpha$ : Homomorphism}
    \index{$\beta$ : Homomorphism}
    \index{$\gamma$ : Homomorphism}
    Assume that $\alpha$ is non-zero to obtain a contradiction.
    Let $\pi \colon P(T) \to T$ be the projective cover of $T$.
    \index{$\pi$ : Homomorphism}
    Since $Q$ is projective, there exists $\varphi \colon Q \to P(T)$ such that the following diagram commutes.
    \index{$\varphi$ : Homomorphism}
    \[
       \begin{tikzcd}[
            cells={nodes={text width=4em, align=center}}, row sep=6em, column sep=6em
            ]
            Q \arrow[r, dashed, "\varphi"] \arrow[rd, "\gamma" description] & P(T) \arrow[d, "\pi"] \\
            \Omega^0(Y) \arrow[u, "\beta"] \arrow[r, swap, "\alpha"]    & T
        \end{tikzcd}
    \]
    In particular, $\alpha$ factors through $P(T)$.
    Note that $\alpha$ is an epimorphism since $T$ is simple and $\alpha$ is non-zero.
    Since $\pi$ is an essential epimorphism, $\varphi \circ \beta \colon \Omega^0(Y) \to P(T)$ is also an epimorphism and hence splits.
    This shows that $P(T)$ is a direct summand of $\Omega^0(Y)$, a contradiction.
    Therefore $\alpha$ is zero, which completes the proof.

\label{lem:costable}
    If $S$ is a simple right $A$-module and $X$ is a right $A$-module, then
    \[
        \overline{\operatorname{Hom}}_A(S, X) \cong \operatorname{Hom}_A(S, \Sigma^0(X)).

<!-- end lem-4 -->
<!-- proof-4: body_md -->
From the definition of $\overline{\operatorname{Hom}}$ and $\Sigma^0$,
    \[
        \overline{\operatorname{Hom}}_A(S, X) \cong \overline{\operatorname{Hom}}_A(S, \Sigma^0(X)).
    \]
    Hence it suffices to prove that the canonical epimorphism
    \[
        \operatorname{Hom}_A(S, \Sigma^0(X)) \to \overline{\operatorname{Hom}}_A(S, \Sigma^0(X))
    \]
    is a monomorphism.

    Let $\alpha \colon S \to \Sigma^0(X)$ factor through an injective right $A$-module $J$ as $\alpha = \gamma \circ \beta$.
    Assume that $\alpha$ is non-zero to obtain a contradiction.
    Let $\iota \colon S \to I(S)$ be the injective hull of $S$.
    \index{$\iota$ : Homomorphism}
    Since $J$ is injective, there exists $\varphi \colon I(S) \to J$ such that the following diagram commutes.
    \[
       \begin{tikzcd}[
            cells={nodes={text width=4em, align=center}}, row sep=6em, column sep=6em
            ]
            I(S) \arrow[r, dashed, "\varphi"]                                                 & J \arrow[d, "\gamma"] \\
            S \arrow[u, "\iota"] \arrow[r, swap, "\alpha"] \arrow[ru, "\beta" description] & \Sigma^0(X)
        \end{tikzcd}
    \]
    In particular, $\alpha$ factors through $I(S)$.
    Note that $\alpha$ is a monomorphism since $S$ is simple and $\alpha$ is non-zero.
    Since $\iota$ is an essential monomorphism, $\gamma \circ \varphi \colon I(S) \to \Sigma^0(X)$ is also a monomorphism and hence splits.
    This shows that $I(S)$ is a direct summand of $\Sigma^0(X)$, a contradiction.
    Therefore $\alpha$ is zero, which completes the proof.

\label{rmk:self-injective}

<!-- proof-6: body_md -->
For a simple right $B$-module $V$ and a simple right $A$-module $U$, set
    \[
        n_{VS} = \left[\, {P(V)} \mid {S \otimes_A M} \,\right],
        \qquad
        m_{UT} = \left[\, {I(U)} \mid {\operatorname{Hom}_B(M, T)} \,\right].
    \]
    \index{$V$ : Simple Module}
    \index{$U$ : Simple Module}
    \index{$n_{VS}$ : Natural Number}
    \index{$m_{UT}$ : Natural Number}
    Our theorem reads $n_{TS} = m_{ST}$ with this notation.
    To prove this reciprocity, decompose $S \otimes_A M$ and $\operatorname{Hom}_B(M, T)$ as
    \begin{align*}
        S \otimes_A M &\cong \Omega^0(S \otimes_A M) \oplus \bigoplus_{V \in \operatorname{Irr}(B)} n_{VS} \, P(V), \\
        \operatorname{Hom}_B(M, T)  &\cong \Sigma^0(\operatorname{Hom}_B(M, T)) \oplus \bigoplus_{U \in \operatorname{Irr}(A)} m_{UT} \, I(U).
    \end{align*}
    Recall that the top of $P(V)$ is $V$ and the socle of $I(U)$ is $U$.
    From the top-socle adjunction \cite[Lemma 2.3]{Sa17}, Schur's lemma and the splitting assumption,
    \begin{align*}
        \dim \operatorname{Hom}_B(P(V), T) &= \dim \operatorname{Hom}_B(V, T) = \delta_{VT}, \\
        \dim \operatorname{Hom}_A(S, I(U)) &= \dim \operatorname{Hom}_A(S, U) = \delta_{SU}
    \end{align*}
    with the Kronecker delta.
    \index{$\delta_{VT}$ : Natural Number}
    Thus
    \begin{align}
        \dim \operatorname{Hom}_B(S \otimes_A M, T) &= \dim \operatorname{Hom}_B(\Omega^0(S \otimes_A M), T) + n_{TS}, \label{eq:n_TS} \\
        \dim \operatorname{Hom}_A(S, \operatorname{Hom}_B(M, T))  &= \dim \operatorname{Hom}_A(S, \Sigma^0(\operatorname{Hom}_B(M, T))) + m_{ST}.   \label{eq:m_ST}
    \end{align}
    On the one hand, the ordinary tensor-hom adjunction yields
    \begin{equation}
        \label{eq:whole}
        \dim \operatorname{Hom}_B(S \otimes_A M, T) = \dim \operatorname{Hom}_A(S, \operatorname{Hom}_B(M, T)).
    \end{equation}
    On the other hand, since $M$ is skew,
    \begin{align*}
        \operatorname{Hom}_B(\Omega^0(S \otimes_A M), T)
        &\cong \underline{\operatorname{Hom}}_B(S \otimes_A M, T)            & & \text{by \cref{lem:stable}} \\
        &\cong \overline{\operatorname{Hom}}_A(S, \operatorname{Hom}_B(M, T))             & & \text{by \cref{main:tensor-hom}} \\
        &\cong \operatorname{Hom}_A(S, \Sigma^0(\operatorname{Hom}_B(M, T)))     & & \text{by \cref{lem:costable}.}
    \end{align*}
    This isomorphism of vector spaces yields
    \begin{equation}
        \label{eq:part}
        \dim \operatorname{Hom}_B(\Omega^0(S \otimes_A M), T)
        = \dim \operatorname{Hom}_A(S, \Sigma^0(\operatorname{Hom}_B(M, T))).
    \end{equation}
    Hence
    \begin{align*}
        n_{TS}
        &= \dim \operatorname{Hom}_B(S \otimes_A M, T) - \dim \operatorname{Hom}_B(\Omega^0(S \otimes_A M), T) & & \text{by \eqref{eq:n_TS}} \\
        &= \dim \operatorname{Hom}_A(S, \operatorname{Hom}_B(M, T)) - \dim \operatorname{Hom}_A(S, \Sigma^0(\operatorname{Hom}_B(M, T)))    & & \text{by \eqref{eq:whole}, \eqref{eq:part}} \\
        &= m_{ST}                                                                   & & \text{by \eqref{eq:m_ST}.}
    \end{align*}
    This completes the proof.

<!-- cor-1: body_md -->
\label{cor:Frobenius}
    Let $A$ and $B$ be finite-dimensional Frobenius algebras over a splitting field and let $M$ be a perfect $(A, B)$-bimodule.
    Write $M^\vee = \operatorname{Hom}_B(M, B)$.
    If $S$ is a simple right $A$-module and $T$ is a simple right $B$-module, then
    \[
        \left[\, {P(T)} \mid {S \otimes_A M} \,\right] = \left[\, {I(S)} \mid {T \otimes_B M^\vee} \,\right].
    \]
<!-- end cor-1 -->
<!-- proof-7: body_md -->
Since $A$ and $B$ are Frobenius algebras, perfect bimodules are skew and $\operatorname{Hom}_B(M, T) \cong T \otimes_B M^\vee$ as right $A$-modules.
    Hence the assertion follows from \cref{main:reciprocity}.

<!-- cor-2: body_md -->
\label{cor:Koshitani}
    Let $A$ and $B$ be finite-dimensional symmetric algebras over a splitting field and let $M$ be a perfect $(A, B)$-bimodule.
    If $S$ is a simple right $A$-module and $T$ is a simple right $B$-module, then
    \[
        \left[\, {P(T)} \mid {S \otimes_A M} \,\right] = \left[\, {P(S)} \mid {T \otimes_B M^*} \,\right].
    \]
<!-- end cor-2 -->
<!-- proof-8: body_md -->
Since $A$ and $B$ are symmetric algebras, $I(S) \cong P(S)$ by \cite[Corollary IV.8.6(i)]{SY11} and $M^\vee \cong M^*$ by \cite[Corollary IV.8.6(ii)]{SY11}.
    Hence the assertion follows from \cref{cor:Frobenius}.

<!-- exm-1: body_md -->
Let $A$ and $B$ be finite-dimensional algebras over a splitting field $F$.
    Consider a simple right $A$-module $U$ and a simple right $B$-module $V$.
    Write $P(U)^\vee = \operatorname{Hom}_A(P(U), A)$.
    \index{$({-})^\vee$ : Functor}
    Then
    \[
        M = P(U)^\vee \otimes_F P(V)
    \] is a skew $(A, B)$-bimodule as
    \begin{align*}
        M \otimes_B B^*
        &\cong P(U)^\vee \otimes_F P(V) \otimes_B B^*
        \cong \dim (P(V) \otimes_B B^*) \, P(U)^\vee, \\
        A^* \otimes_A M
        &\cong A^* \otimes_A P(U)^\vee \otimes_F P(V)
        \cong \dim (A^* \otimes_A P(U)^\vee) \, P(V).
    \end{align*}
    For a simple right $A$-module $S$ and a simple right $B$-module $T$,
    \begin{align*}
        S \otimes_A M
        &\cong S \otimes_A P(U)^\vee \otimes_F P(V)   & & \\
        &\cong \operatorname{Hom}_A(P(U), S) \otimes_F P(V)         & & \text{by \cite[Corollary II.4.9]{SY11}} \\
        &\cong \operatorname{Hom}_A(U, S) \otimes_F P(V)            & & \\
        &\cong \delta_{US} \, P(V),                     & & \\
        \operatorname{Hom}_B(M, T)
        &\cong \operatorname{Hom}_B(P(U)^\vee \otimes_F P(V), T)    & & \\
        &\cong \operatorname{Hom}_F(P(U)^\vee, \operatorname{Hom}_B(P(V), T))     & & \\
        &\cong \operatorname{Hom}_F(P(U)^\vee, \operatorname{Hom}_B(V, T))        & & \\
        &\cong \delta_{VT} \, I(U)                      & & \text{by $(P(U)^\vee)^* \cong I(U)$.}
    \end{align*}
    Hence we obtain the equality of multiplicities
    \[
        \left[\, {P(T)} \mid {S \otimes_A M} \,\right] = \delta_{US}\delta_{VT} = \left[\, {I(S)} \mid {\operatorname{Hom}_B(M, T)} \,\right].
    \]

<!-- exm-2: body_md -->
Let $A$ and $B$ be finite-dimensional algebras over a splitting field $F$.
    Consider a semisimple right $A$-module $D$ and a semisimple right $B$-module $E$ defined by
    \[
        D = \bigoplus_{\substack{U \in \operatorname{Irr}(A)\\ \operatorname{Hom}_A(A^*, U) = 0}} U,
        \qquad
        E = \bigoplus_{\substack{V \in \operatorname{Irr}(B)\\ \operatorname{Hom}_B(V, B) = 0}} V.
    \]
    \index{$D$ : Semisimple Module}
    \index{$E$ : Semisimple Module}
    Then
    \[
        N = D^* \otimes_F E
    \]
    is a skew $(A, B)$-bimodule as
    \begin{align*}
        N \otimes_B B^*
        &\cong D^* \otimes_F E \otimes_B B^*
        \cong \dim (\operatorname{Hom}_B(E, B)) \, D^*
        = 0, \\
        A^* \otimes_A N
        &\cong A^* \otimes_A D^* \otimes_F E
        \cong \dim (\operatorname{Hom}_A(A^*, D)) \, E
        = 0.
    \end{align*}
    \index{$N$ : Bimodule}
    For a simple right $A$-module $S$ and a simple right $B$-module $T$,
    \begin{align*}
        S \otimes_A N
        &\cong S \otimes_A D^* \otimes_F E \\
        &\cong \operatorname{Hom}_A(S, D)^* \otimes_F E \\
        &\cong
        \begin{cases}
            E & \text{if $\operatorname{Hom}_A(A^*, S) = 0$} \\
            0 & \text{otherwise},
        \end{cases} \\
        \operatorname{Hom}_B(N, T)
        &\cong \operatorname{Hom}_B(D^* \otimes_F E, T) \\
        &\cong \operatorname{Hom}_F(D^*, \operatorname{Hom}_B(E, T)) \\
        &\cong
        \begin{cases}
            D & \text{if $\operatorname{Hom}_B(T, B) = 0$} \\
            0 & \text{otherwise}.
        \end{cases}
    \end{align*}
    Note that if $T$ is projective, then $\operatorname{Hom}_B(T, B) \cong \operatorname{Hom}_B(P(T), B) \neq 0$.
    Similarly, if $S$ is injective, then $\operatorname{Hom}_A(A^*, S) \cong \operatorname{Hom}_A(A^*, I(S)) \neq 0$.
    Hence we obtain the equality of multiplicities
    \[
        \left[\, {P(T)} \mid {S \otimes_A N} \,\right] = 0 = \left[\, {I(S)} \mid {\operatorname{Hom}_B(N, T)} \,\right].
    \]
