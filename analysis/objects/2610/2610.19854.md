Assume $S$ is a real-valued real analytic function near the origin in $\Bbb R^2$. Let $N_r(S)$ be the reduced Newton polyhedron of $S$. Then the following statements are true.

<!-- theorem-8: body_md -->
\label{Main-Theorem-Full-Range}
Assume $(m,n)$ is a vertex of $N_r(S)$ with $m,n\geq 1$. If $Rf$ is bounded from $L^{\frac{m+n}{m}}(\Bbb R^2)$ into $L^{\frac{m+n}{m}}_{\frac{1}{m+n}}(\Bbb R^2)$ for any cut-off function $\varphi\in C_0^{\infty}(\Bbb R^3)$, $\varphi(0)\neq 0$, supported in a sufficiently small neighborhood of the origin, then
$$\left\| |\partial_t|^{\frac{1}{m+n} } R_{m,n}f \right\|_{\frac{m+n}{m}}
\leq

for some constant $C$ and any $f\in C_0^{\infty}$.
<!-- end theorem-8 -->
<!-- proof-1: body_md -->
Without loss of generality, we assume that $S(x,0)$ and $S(0,y)$ vanish identically. Then $N_r(S)=N(S)$. Let $\varphi\in C_0^{\infty}(\Bbb R^3)$ with $\varphi(0)\neq 0$. For $\overrightarrow{\mu}=(\mu_1,\mu_2)$ with $\mu_1,\mu_2>0$. For any $f\in C_0^{\infty}(\Bbb R^2)$, we write $f_{\overrightarrow{\mu}}(t,x)=f(\mu_1t,\mu_2x)$. The definition of Radon transform $R$ gives
\begin{equation*}
Rf_{\overrightarrow{\mu}}(t,x)
=
\int_{-\infty}^{\infty}
f(\mu_1t + \mu_1 S(x,y),\mu_2y)
\varphi(t,x,y)dy.
\end{equation*}
For $\mu_3>0$, consider
\begin{equation*}
G(t,x)
=
\int_{-\infty}^{\infty}
f\big(t + \mu_1 S(x/\mu_3,y/\mu_2),y\big)
\varphi(t/\mu_1,x/\mu_3,y/\mu_2)dy.
\end{equation*}
Then we see that
$Rf_{\overrightarrow{\mu}}(t,x)
=
\mu_2^{-1}G(\mu_1t,\mu_3x).
$
For fixed positive numbers $\mu_1,\mu_2,\mu_3$, it follows from the assumption $f\in C_0^{\infty}$ that both $Rf_{\overrightarrow{\mu}}$ and $G$ are smooth with compact support. On the other hand, for $\delta>0$,
\begin{equation*}
|\partial_t|^{\delta}\left( Rf_{\overrightarrow{\mu}}(t,x) \right)
=
\mu_1^{\delta}\mu_2^{-1}
(|\partial_t|^{\delta}G)
(\mu_1t, \mu_3x).
\end{equation*}
By the inverse Fourier transform, one can see that
\begin{eqnarray}\label{inverseFT-FD-G}
& &
|\partial_t|^{\delta}G(t, x) \nonumber \\
&=&
\int_{\Bbb R^3}
e^{2\pi i t\xi}(2\pi|\xi|)^{\delta}
e^{2\pi i \eta \mu_1 S(\frac{x}{\mu_3},\frac{y}{\mu_2})}
(\mathcal{F}_1f)(\eta,y)~
\mu_1(\mathcal{F}_1\varphi)
 \Big(\mu_1(\xi-\eta),\frac{x}{\mu_3},\frac{y}{\mu_2}\Big)
d\eta dy d\xi.~~~
\end{eqnarray}
Since $(m,n)$ is an extreme point of the Newton polyhedron $N(S)$, we can take a line
$L:\alpha x+ \beta y=1$ in the plane, $\alpha>0$, $\beta>0$, such that $(m,n)\in L$ and $\alpha k+ \beta l>1$ for any point $(k,l)\in N(S)\backslash \{(m,n)\}$.

Take $\mu_1=\tau$, $\mu_2=\tau^{\beta}$, and $\mu_3=\tau^{\alpha}$ with $\tau\geq 1$. We are going to prove
\begin{equation}\label{limit-of-FD}
\lim_{\tau\rightarrow \infty}
|\partial_t|^{\delta}G(t, x)
=
c|\partial_t|^{\delta}
\left(
\int_{\Bbb R}f(t+a_{m,n}x^my^n,y)dy
\right)
\end{equation}
for all $f\in C_0^{\infty}$, where $c=\varphi(0,0,0)\neq 0$ and $a_{m,n}=\frac{\partial_x^m\partial_y^nS(0,0)}{m!n!}\neq 0$.

To show (\ref{limit-of-FD}), we first prove that the integral (\ref{inverseFT-FD-G}) with respect to $\eta$ is dominated by an integral function in $\xi,y$. More precisely, it is true that
\begin{eqnarray}\label{DM-Functions}
& &\sup
\left|
\int_{\Bbb R}
e^{2\pi i t\xi}(2\pi|\xi|)^{\delta}
e^{2\pi i \eta \mu_1 S(\frac{x}{\mu_3},\frac{y}{\mu_2})}
(\mathcal{F}_1f)(\eta,y)~
\mu_1(\mathcal{F}_1\varphi)
 \Big(\mu_1(\xi-\eta),\frac{x}{\mu_3},\frac{y}{\mu_2}\Big)
d\eta
\right|\nonumber \\
&\leq &
C\frac{|\xi|^{\delta}}{(1+\xi^2)(1+y^2)},
\end{eqnarray}
where the supremum is taken over $\mu_1\geq 1,\mu_2>,\mu_3>0$, and the constant $C$ is independent of $t$, $\xi$, and $y$.
It is clear that $|(\mathcal{F}_1f)(\eta,y)|\lesssim (1+\eta^2)^{-1}(1+y^2)^{-1}$ and $|(\mathcal{F}_1\varphi)(\eta,x,y)|\lesssim (1+\eta^2)^{-1}$. Then the left side of (\ref{DM-Functions}) is bounded by
\begin{eqnarray*}
& &
(2\pi|\xi|)^{\delta}
\sup_{\mu_1\geq 1}
\int_{\Bbb R}
\Big|(\mathcal{F}_1f)(\eta,y)\Big|
\left|\mu_1(\mathcal{F}_1\varphi)
\Big(\mu_1(\xi-\eta),\frac{x}{\mu_3},\frac{y}{\mu_2}\Big)\right|
d\eta\\
&\lesssim&
(2\pi|\xi|)^{\delta}
\sup_{\mu_1\geq 1}
\int_{\Bbb R}
\left(1+\eta^2\right)^{-1}\left(1+y^2\right)^{-1}
\mu_1\left(1+\mu_1^2|\xi-\eta|^2\right)d\eta.
\end{eqnarray*}
To prove (\ref{DM-Functions}), it suffices to show
\begin{equation*}
\sup_{\mu_1\geq 1}
\int_{\Bbb R}
\left( 1+ \eta^2\right)^{-1}
~\mu_1\left( 1+ \mu_1^2|\xi-\eta|^2\right)^{-1}d\eta
\lesssim
\left( 1+ \xi^2 \right)^{-1}.
\end{equation*}
Observe that the Fourier transform of $(1+x^2)^{-1}$ is $\pi e^{-2\pi|\xi|}$. Hence we obtain
\begin{equation*}
\int_{\Bbb R}
\left( 1+ \eta^2\right)^{-1}
~\mu_1\left( 1+ \mu_1^2|\xi-\eta|^2\right)^{-1}d\eta
=\pi\lambda(1+|\lambda \xi|^2)^{-1},~~~\lambda=(1+\mu_1^{-1})^{-1}.
\end{equation*}
Since $\frac12\leq \lambda \leq1$ is true for $\mu_1\geq 1$, the desired estimate follows. It should be pointed out that the above integral identity is not essential in our estimate. For large $\xi$, we can obtain the desired estimate by dividing the above integral into three parts $|\eta|\leq |\xi|/2$, $|\eta|\lesssim |\xi|$, and $|\eta|\geq 2|\xi|$. The details are omitted here. Hence the proof of (\ref{DM-Functions}) is complete.

On the other hand, for fixed $x,y$ and $f\in C_0^{\infty}$, it is clear that
\begin{eqnarray}\label{pointwise-limit}
\color{blue}& &
\lim_{\tau\rightarrow \infty}
\int_{\Bbb R}
e^{2\pi i \eta \tau S(\frac{x}{\tau^{\alpha}},\frac{y}{\tau^{\beta}})}
(\mathcal{F}_1f)(\eta,y)~
\tau(\mathcal{F}_1\varphi)
 \Big(\tau(\xi-\eta),\frac{x}{\tau^{\alpha}},
 \frac{y}{\tau^{\beta}}\Big)
{d\eta }   \nonumber\\
&&
=
ce^{2\pi i a_{m,n}\xi x^m y^n}
(2\pi|\xi|)^{\delta}
(\mathcal{F}_1f)(\xi,y),
\end{eqnarray}
where $c$ and $a_{m,n}$ are given below (\ref{limit-of-FD}).

By the Lebesgue dominated convergence theorem, we obtain
\begin{eqnarray*}
\lim_{\tau\rightarrow\infty}
|\partial_t|^{ \frac1{m+n} }G(t,x)
&=&
c\int_{\Bbb R^2}
e^{2\pi i t\xi}
(2\pi|\xi|)^{\frac1{m+n}}
e^{2\pi i a_{m,n} \xi x^m y^n}
(\mathcal{F}_1f)(\xi,y)dyd\xi\\
&=&
c|\partial_t|^{ \frac1{m+n} }
\left(\int_{\Bbb R}f(t+a_{m,n} x^m y^n,y)dy\right).
\end{eqnarray*}
By a simple scaling argument, we are able to prove the statement of the theorem.

$$\left\| |\partial_t|^{\delta} R_{m,n}f \right\|_{\frac{m+1}{m}}
\leq
C \left\| f \right\|_{\frac{m+1}{m}}$$
for all $f\in C_0^{\infty}(\Bbb R^2)$. The proof of this result is the same as that of $\delta=\frac{1}{m+1}$ and $n=1$. By scaling, $\delta=\frac{1}{m+1}$ is the only one case of interest in the theorem.

\end{equation*}
This implies
\begin{equation*}

=
\frac{\lambda_1^{\delta}}{\lambda_2}
(|\partial_t|^{\delta}R_{m,n}f)\left(\lambda_1t, \frac{\lambda_1^{1/m}}{\lambda_2^{n/m}} x \right).
\end{equation*}

\leq
C \|f_{\lambda}\|_{L^p},
\end{equation*}
then we would obtain
\begin{equation*}

\lambda_1^{-\frac1p}
\frac{\lambda_2^{n/(mp)}  }{ \lambda_1^{1/(mp)} }
=
\lambda_1^{-\frac1p}\lambda_2^{-\frac1p}.
\end{equation*}
Hence we get $p=\frac{m+n}m$ and $\delta=\frac1{m+n}$.}

<!-- lemma-3: body_md -->
\label{Lemma-upper-Lp-bound}
For any $1\leq p \leq \infty$, it is true that
$$\|R_jf\|_{L^p}
\lesssim
\left( \frac{2^{mj}}{z}  \right)^N~2^{-mj}~2^{(m+1)j/p}$$
for all nonnegative integers $N$.
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
By integration by parts, it is easy to see that
\begin{eqnarray*}
R_jf(t,x)
&=&
\Phi\left( \frac{x}{2^j} \right)
\int_{\Bbb R}\eta(t+x^my,y) e^{izy} dy \\
&=&
\Phi\left( \frac{x}{2^j} \right)
\left( \frac{1}{iz} \right)^N
\int_{\Bbb R}\eta(t+x^my,y) \frac{d^N}{dy^N}\Big(e^{izy} \Big) dy\\
&=&
\Phi\left( \frac{x}{2^j} \right)
\left(- \frac{1}{iz} \right)^N
\int_{\Bbb R}\frac{d^N}{dy^N}\Big(\eta(t+x^my,y) \Big) e^{izy} dy.
\end{eqnarray*}
Observe that
\begin{eqnarray*}
\frac{d^N}{dy^N}\Big(\eta(t+x^my,y) \Big)
&=&
\Big\{\Big((x^m,1)\cdot \nabla\Big)^N\eta\Big\}
(t+x^my,y)\\
&=&
\sum_{k=0}^N C_N^k x^{m(N-k)}\partial_1^{N-k}
\partial_2^k\eta(t+x^my,y),
\end{eqnarray*}
where $\nabla=(\partial_1,\partial_2)$ is the gradient operator in the plane. For fixed $t$ and $x\approx 2^j$, it follows from $\eta\in C_0^{\infty}(\Bbb R^2)$ that $\{y:\eta(t+x^my,y)\neq 0\}$ is contained in an interval with length less than a constant multiple of $2^{-mj}$.
This implies, for positive $j$,
$$
|R_jf(t,x)|
\lesssim
\left(\frac{2^{mj}}{z}\right)^N 2^{-mj}.
$$
Similarly, if $R_jf(t,x)\neq 0$, then it is necessarily that $|t|\lesssim 2^{mj}$ and $x\approx 2^j$. Hence for any $p\geq 1$, we obtain
\begin{equation*}
\|R_jf\|_{L^p}
\lesssim
\left(\frac{2^{mj}}{z}\right)^N 2^{-mj}
2^{(m+1)j/p}.
\end{equation*}
The proof of the lemma is complete.

<!-- lemma-4: body_md -->
\label{Lemma-comparable-Lp-lemma}
There exists a small number $\epsilon_0\in (0,1)$ such that, for sufficiently large $j\gg  1$ and $z=\epsilon_0~2^{mj}$, it is true that
\begin{equation}\label{comparable-Lp-norm}
\|\partial_t^NR_jf\|_{L^p}
\approx
2^{-mj}~2^{(m+1)j/p},~~~N=0,1,2,3,4,
\end{equation}
where the implicit constants are independent of $j$.

This estimate is also true for larger $N$ provided that $j\geq j(m,N)$, where $j(m, N)$ is a sufficiently large number depending on $m$ and $N$. For our purpose, the estimates for $N=0,1,2$ are enough.

For clarity, let $\epsilon_0=2^{-m-1}$. For $2^{\frac14} 2^{mj}+1\leq -t \leq 2^{\frac34} 2^{mj}-1$ and $2^{j-\frac1{4m}}\leq x \leq 2^{j+\frac1{4m}}$, the following estimate is true:
\begin{equation*}
\frac{-t-1}{x^m}\geq \frac{2^{\frac14} 2^{mj}}{2^{mj+\frac14}}=1,~~~
\frac{-t+1}{x^m}\leq \frac{2^{\frac34} 2^{mj}}{2^{mj-\frac14}}=2.
\end{equation*}
Observe that $\eta(X)=1$ for $X\in [-1,1]\times [1,2]$.
We claim that there exists a constant $C>0$, independent of $j$, such that
\begin{equation}\label{low-bound-Rjf}
|R_jf(t,x)|
\geq
C 2^{-mj}.
\end{equation}
In view of $\,{\rm supp}\,(f)\subseteq [-2,2]\times [\frac12,4]$, if $R_jf(t,x)\neq 0$ then $|t+x^my|\leq 2$. Hence, for $2^{j-1}\leq x \leq 2^{j+1}$,
$$
\left|y+\frac{t}{x^m} \right|
\leq
\frac{2}{x^m}
\leq
2^{-m(j-1)+1}.
$$
The oscillation term $e^{izy}$ can be removed essentially in $R_jf(t,x)$ for $t,x$ described as above. In fact, for $t\approx \epsilon_0 2^{mj}$ and $x\approx 2^j$, one sees that
\begin{equation*}
|R_jf(t,x)|
=
\left|
\Phi\left( \frac{x}{2^j} \right)
\int_{\Bbb R}\eta(t+x^my,y) e^{i\epsilon_0 2^{mj} (y+t/x^m)} dy
\right|
\geq
C
\Phi\left( \frac{x}{2^j} \right)
\int_{\Bbb R}\eta(t+x^my,y) dy,
\end{equation*}
where we have used the fact $\epsilon_0 2^{mj} |y+t/x^m|\leq 1$, and $\Phi,\eta$ are nonnegative functions. For those $y$ satisfying $|t+x^my|\leq 1$ and $1\leq y \leq 2$, our assumption on $\eta$ implies $\eta(t+x^my,y)=1$. Observe that
\begin{equation*}
|t+x^my|\leq 1
\Leftrightarrow
\frac{-1-t}{x^m}
\leq
y
\leq
\frac{1-t}{x^m}
\Rightarrow
1\leq y \leq 2
\end{equation*}
for $t,x$ described as above, i.e., $2^{\frac14} 2^{mj}+1\leq -t \leq 2^{\frac34} 2^{mj}-1$ and $2^{j-\frac1{4m}}\leq x \leq 2^{j+\frac1{4m}}$. This implies $|R_jf(t,x)|\geq C2^{-mj}$. Hence (\ref{comparable-Lp-norm}) is true for $N=0$.

&=&

\right)
(t+x^my,y)

\eta(t+x^my,y),
\end{eqnarray*}
where $\partial_k\eta(x_1,x_2)=\partial_{x_k}\eta(x_1,x_2)$
for $k=1,2$. It follows that
\begin{eqnarray*}
\partial_t^NR_jf(t,x)
&=&
\Phi\left( \frac{x}{2^j} \right)
\partial_t^N
\int_{\Bbb R}\eta(t+x^my,y) e^{izy} dy\\
&=&
\Phi\left( \frac{x}{2^j} \right)
x^{-Nm}

\Big((x^m,1)\cdot \nabla \Big)^N\eta
\right)
(t+x^my,y)e^{izy} dy\\

-

<!-- lemma-5: body_md -->
If all assumptions in Lemma \ref{Lemma-comparable-Lp-lemma} are true, then for all real numbers $0<\beta <2$, we have
$$\Big\||\partial_t|^{\beta}R_jf \Big\|_{L^p(\Bbb R^2)}
\approx
2^{-mj}~2^{(m+1)j/p}$$
for all $1<p<\infty$, where the implicit constants depend on $\beta$ and $p$.
<!-- end lemma-5 -->
<!-- proof-4: body_md -->
We shall prove the above statement by a standard complex interpolation argument. Define a multiplier $T_{y}g(x)=\left(|2\pi \xi|^{iy}\widehat{g}(\xi)\right)^{\vee}(x)$ for $y\in\Bbb R$ and $g\in C_0^{\infty}(\Bbb R)$. The Marcinkiewicz multiplier theorem implies that
$\|T_yg\|_{L^p}\leq C(p)(1+|y|)\|g\|_{L^p}$ for all $1<p<\infty$; this is also true with $T_yg$ replaced by $T_{-y}g$. By the Plancherel theorem, one has $\|T_yg\|_{L^2}=\|g\|_{L^2}$. The simple observation $T_yT_{-y}g=g$ implies
\begin{equation}\label{multiplier-comparable-Lp}
C(p)^{-1}(1+|y|)^{-1}\|g\|_{L^p}
\leq
\|T_yg\|_{L^p}
\leq
C(p)(1+|y|)\|g\|_{L^p},~~~1<p<\infty,
\end{equation}
where $C(p)$ is independent of $y$.

For $h\in L^{p'}(\Bbb R^2)\cap C_0^{\infty}(\Bbb R^2)$, $\|h\|_{L^{p'}}\leq 1$, let
\begin{eqnarray*}
G(z)
&=&
\int_{\Bbb R^2}|\partial_t|^{z}R_jf(t,x)
\overline{h(t,x)}dtdx\\
&=&
\int_{\Bbb R^2}|2\pi\xi|^{z}\mathcal{F}_1R_jf(\xi,x)
\overline{\mathcal{F}_1h(\xi,x)}d\xi dx
\end{eqnarray*}
for $z$ in the strip $S:\beta\leq \,{\rm Re}\,(z) \leq 4$.
Then $G$ is analytic in the interior of $S$. It is also bounded and continuous on $S$. By (\ref{multiplier-comparable-Lp}), we have $|G(z)|\leq C(p)(1+|y|)\||\partial_t|^{\beta}R_jf\|_{L^p}$ for $\,{\rm Re}\,(z)=\beta$, and $|G(z)|\leq C(p)(1+|y|)\|\partial_t^{4}R_jf\|_{L^p}$ for $\,{\rm Re}\,(z)=4$.
Choose an $h$ such that $G(2)\geq \frac12\|\partial_t^{2}R_jf\|_{L^p}$. Hence, by complex interpolation, we have
\begin{equation*}
\|\partial_t^{2}R_jf\|_{L^p}
\lesssim
\||\partial_t|^{\beta}R_jf\|_{L^p}^{\theta}
\|\partial_t^{4}R_jf\|_{L^p}^{1-\theta},
~~~\beta\theta+4(1-\theta)=2.
\end{equation*}
By Lemma \ref{Lemma-comparable-Lp-lemma}, we obtain $$\||\partial_t|^{\beta}R_jf\|_{L^p}\gtrsim 2^{-mj}2^{(m+1)j/p}.$$
By a similar complex interpolation in the strip $0\leq \,{\rm Re}\,(z) \leq 2$, we also have
\begin{equation*}
\||\partial_t|^{\beta}R_jf\|_{L^p}\lesssim
\|R_jf\|_{L^p}^{1-\frac{\beta}2}
\||\partial_t|^{2}R_jf\|_{L^p}^{\frac{\beta}2}
\lesssim
2^{-mj}2^{(m+1)j/p}.
\end{equation*}
 The proof of the lemma is complete.

\label{Lemma-concentration-property}
There is a sufficiently small number $\epsilon_0>0$ such that for $j\gg 1$ and $z=\epsilon_0~2^{mj}$, it is true that
$$
\Big\|\partial_t^{l}Rf \Big\|_{L^{ \frac{m+1}{m}  }(\Bbb R^2)}

where the implicit bounds are independent of $j$. Moreover, for any small $\epsilon>0$, there is an $M\gg 1$ such that, for $l=1,2,3,4,$

\end{equation}
where the choice of $M$ is independent of $j$.

<!-- remark-6: body_md -->
\label{remark-complex-interpolation}
As in the proof of Lemma \ref{Lemma-comparable-Lp-lemma}, we can choose $\epsilon_0=2^{-m-1}$. The above lemma tells us that each function $\partial_t^{l}Rf $ is essentially supported on the rectangle $\{(t,x): |t|\leq 2^{mj},~|x|\approx 2^j\}$. Also by lemma \ref{Lemma-comparable-Lp-lemma} and its proof, one can verify that  $\| Rf \|_{L^{ \frac{m+1}{m} }}=+\infty$. However, we shall prove that $\| |\partial_t|^w Rf \|_{L^{ \frac{m+1}{m} }}<\infty$ for any complex number $w$ with $\,{\rm Re}\,(w)>0$. Write
\begin{equation*}
|\partial_t|^w Rf(t,x)
=
\int_{\Bbb R}\int_{\Bbb R}
e^{2\pi i t\xi}
e^{2\pi i \xi x^m y}
(2\pi|\xi|)^{w}\mathcal{F}_1\eta(\xi,y)e^{izy}dyd\xi,~~~\,{\rm Re}\,(w)>0.
\end{equation*}
Since $\eta$ is smooth with compact support, $\mathcal{F}_1\eta(\xi,y)$ is rapidly decreasing in $\xi$, and has compact support in $y$. Hence $|\partial_t|^w Rf$ is bounded. By the stationary phase method, the integral with respect to $y$ can be bounded by
\begin{equation*}
\left|
\int_{\Bbb R}
e^{2\pi i t\xi}
e^{2\pi i \xi x^my} (2\pi|\xi|)^{w}\mathcal{F}_1\eta(\xi,y)e^{izy}dy
\right|
\leq
C |\xi|^{\,{\rm Re}\,(w)}(1+|\xi x^m|)^{-N},~~~N>1+\,{\rm Re}\,(w),
\end{equation*}
where the constant $C$ depends on $z$ and $N$, but is independent of $\xi$ and $x$. This implies
\begin{equation*}
||\partial_t|^w Rf(t,x)|
\leq
C\int_{\Bbb R}
|\xi|^{\,{\rm Re}\,(w)}(1+|\xi x^m|)^{-N} d\xi
\leq
C|x|^{-m(1+\,{\rm Re}\,(w))}.
\end{equation*}
Note that $|\partial_t|^w Rf$ is bounded. Hence $\big||\partial_t\big|^w Rf(t,x)|\leq C\left(1+|x|^{m(1+\,{\rm Re}\,(w))} \right)^{-1}.$ Similarly, apply the stationary phase method to the integral with $\xi$. Then
 \begin{eqnarray*}
|\partial_t|^w Rf(t,x)
&=&
\int_{\Bbb R}\int_{\Bbb R}
\left(\frac{1}{2\pi i t}\right)
\frac{d}{d\xi}(e^{2\pi i t\xi})
e^{2\pi i \xi x^m y}
(2\pi|\xi|)^{w}\mathcal{F}_1\eta(\xi,y)e^{izy}dyd\xi\\
&=&
\left(-\frac{1}{2\pi i t}\right)
\int_{\Bbb R}\int_{\Bbb R}
e^{2\pi i t\xi}
\frac{\partial}{\partial\xi}
\Big(e^{2\pi i \xi x^m y}
(2\pi|\xi|)^{w}
\mathcal{F}_1\eta(\xi,y)\Big)
e^{izy}dyd\xi.
\end{eqnarray*}
Here if the partial derivative $\partial_{\xi}$ is taken with respect to $e^{2\pi i \xi x^m y}$, then the term $2\pi i x^my$ will appear, and one shall apply the stationary phase method to the integral in $y$ to obtain decay $(1+|x|^{m(1+\,{\rm Re}\,(w))} )^{-1}$. Hence we can obtain $\big||\partial_t\big|^w Rf(t,x)|\leq C(1+|t| )^{-1}.$

Combining the decay in $x$ and $t$, we get the desired estimate
\begin{equation*}
\big||\partial_t\big|^w Rf(t,x)|
\leq C
\left(1+|t|+|x|^{m(1+\,{\rm Re}\,(w))} \right)^{-1}.
\end{equation*}
This shows $|\partial_t\big|^w Rf\in L^{ \frac{m+1}{m} }(\Bbb R^2)$ provided that $\,{\rm Re}\,(w)>0$.

By Lemma \ref{Lemma-comparable-Lp-lemma}, we get $
\|\partial_t^{l}Rf \|_{\frac{m+1}{m}  }\geq \|\partial_t^{l}R_jf \|_{ \frac{m+1}{m}}\geq C>0 $ for $l=1,2,3,4,$ where $C$ is independent of $j$. For $k\in \mathbb{Z}$, nonnegative integers $a,b$, and $|x|\approx 2^k$,
\begin{equation*}
\left|
\int_{\Bbb R}\partial_1^a\partial_2^b\eta(t+x^my,y)e^{izy}dy
\right|
\lesssim
2^{-mk}.
\end{equation*}
Also, if the above integral does not vanish, then $|t|\lesssim 2^{mk}$. Hence the following $L^p$ bound
\begin{equation}\label{L^p-estimate-dyadic-intervals}
\left\|
\int_{\Bbb R}\partial_1^a\partial_2^b\eta(t+x^my,y)e^{izy}dy
\right\|_{ L_{t,x}^{ (m+1)/m }(|x|\approx 2^k)}
\lesssim
1
\end{equation}
is true with the implicit upper bound independent of $k$.


Now we turn to the proof of (\ref{Lemma-Lp-outside-annulus}). For any $M\geq 1$, we use the stationary phase method to obtain, for $|x|\leq 2^{j-M}$,
\begin{eqnarray*}
\partial_t^{l}Rf(t,x)
&=&
\int_{\Bbb R}\partial_t^{l}\eta(t+x^my,y)e^{izy}dy\\
&=&
\left( \frac{1}{iz} \right)^N
\int_{\Bbb R}\partial_t^{l}\eta(t+x^my,y) \frac{d^N}{dy^N}\Big(e^{izy} \Big) dy\\
&=&
\left(- \frac{1}{iz} \right)^N
\int_{\Bbb R}\frac{d^N}{dy^N}\Big(\partial_t^{l}\eta(t+x^my,y) \Big) e^{izy} dy\\
&=&
\left(- \frac{1}{iz} \right)^N
\int_{\Bbb R}
\sum_{k=0}^NC_N^kx^{m(N-k)}\partial_1^{N-k}\partial_2^k
\partial_t^{l}\eta(t+x^my,y)  e^{izy} dy.
\end{eqnarray*}
Here $\partial_1=\partial_t$. On the one hand, $\|\partial_t^{l}Rf\|_{L^{\frac{m+1}m}(|x|\leq 1)  }\lesssim 2^{-mNj}$ is obvious. On the other hand, for $|x|\approx 2^k$, $0\leq k\leq j-M$, we use the above estimates to obtain
\begin{equation*}
 \|\partial_t^{l}Rf\|_{L^{\frac{m+1}m}(|x|\approx 2^k)  }\lesssim 2^{mN(k-j)}
 \end{equation*}
By the triangle inequality on $L^p$, we get
\begin{equation*}
\|\partial_t^{l}Rf\|_{L^{\frac{m+1}m}(t\in\Bbb R,~|x|\leq 2^{j-M})  }\lesssim 2^{-mMN}.
\end{equation*}
It should be pointed out that this estimate is also true for $l=0$.

For positive integers $l$, we shall follow the argument in the proof of Lemma \ref{Lemma-comparable-Lp-lemma}. For $|x|\approx 2^k$, $k\geq j+M$, there holds
\begin{eqnarray*}
\partial_t^lRf(t,x)
&=&
\partial_t^l
\int_{\Bbb R}\eta(t+x^my,y) e^{izy} dy\\
&=&
x^{-ml}
\int_{\Bbb R}\left(
\Big((x^m,1)\cdot \nabla \Big)^l\eta
\right)
(t+x^my,y)e^{izy} dy\\
&&
-
\sum_{r=1}^l C_l^r x^{-mr}\int_{\Bbb R}\partial_1^{l-r}\partial_2^r
\eta(t+x^my,y)e^{izy} dy\\
&=&
\left(
\frac{-iz}{x^m}
\right)^l
\int_{\Bbb R}
\eta(t+x^my,y)e^{izy} dy\\
&&
-
\sum_{r=1}^l C_l^r x^{-mr}\int_{\Bbb R}\partial_1^{l-r}\partial_2^r
\eta(t+x^my,y)e^{izy} dy.
\end{eqnarray*}
Hence, by (\ref{L^p-estimate-dyadic-intervals}),
\begin{equation*}
 \|\partial_t^{l}Rf\|_{L^{\frac{m+1}m}(|x|\approx 2^k)  }\lesssim 2^{m(j-k)l}+2^{-mk},~~~l=1,2,3,4.
\end{equation*}
By the triangle inequality, we obtain
\begin{equation*}
 \|\partial_t^{l}Rf\|_{L^{\frac{m+1}m}(|x|\geq 2^{j+M})  }\lesssim 2^{-mMl}+2^{-mj-mM}
\end{equation*}
for positive integers $l$, and the implicit constant is independent of $j,M$. The proof of the lemma is complete.

<!-- proof-6: body_md -->
Without loss of generality, assume $S(x,0)=S(0,y)=0$. We need only to show the statement (i), and the statement (ii) follows by duality.

By a contradiction argument, assume $A_L>B_L$ and there is a constant $C$ such that
\begin{equation*}
\left\|
|\partial_t|^{\delta}Rf
\right\|_{L^p}
\leq
C\|f\|_{L^p}
\end{equation*}
is true for some $\delta$ and $p$ given as in (i). For $\lambda>0$, let $f_{\lambda}(t,x)=f(\lambda t,x)$. The variable $x$ is unchanged. For any $\mu>0$, define
\begin{equation*}
R_{\lambda,\mu}f(t,x)
=
\int_{-\infty}^{\infty}
f\left(t+\lambda S\left(\frac{x}{\mu},y\right)\right)
\varphi\left(
\frac{t}{\lambda},\frac{x}{\mu},y
\right)
dy.
\end{equation*}
By dilation, $\||\partial_t|^{\delta}(Rf_{\lambda})\|_{L^p}
=\lambda^{\delta}\lambda^{-\frac1p}\mu^{-\frac1p}
\||\partial_t|^{\delta}(R_{\lambda,\mu}f)\|_{L^p}$.
Also, $\|f_{\lambda}\|_{L^p}=\lambda^{-\frac1p}\|f\|_{L^p}$.
By our assumption, there is a constant $C$ such that $\left\|
|\partial_t|^{\delta}Rf_{\lambda}
\right\|_{L^p}
\leq
C\|f_{\lambda}\|_{L^p}$ for all $f\in C_0^{\infty}(\Bbb R^2)$. Let $\lambda=\mu^{A_L+1}$. Then
\begin{equation}\label{Bound-dilated-RT}
\left\|
|\partial_t|^{\delta}R_{\lambda,\mu}f
\right\|_{L^p}
\leq
C\|f\|_{L^p},~~~\delta=\frac{1}{A_L+1}\cdot\frac{1}{p},
\end{equation}
where $C$ is independent of $\lambda,\mu$ and $f$.

By Taylor's expansion,
\begin{eqnarray*}
\lambda S\left(\frac{x}{\mu},y\right)
&=&
\lambda \sum_{\alpha,\beta}
c_{\alpha,\beta}
\left(\frac{x}{\mu}\right)^{\alpha}y^{\beta}\\
&=&
\sum_{\alpha=A_L+1,\beta\geq B_L+1}
c_{\alpha,\beta}
x^{\alpha}y^{\beta}
+
\lambda\sum_{\alpha\geq A_L+2,\beta\geq 0}
c_{\alpha,\beta}
\left(\frac{x}{\mu}\right)^{\alpha}y^{\beta}\\
&=&
\sum_{\alpha=A_L+1,\beta\geq B_L+1}
c_{\alpha,\beta}
x^{\alpha}y^{\beta}
+
x^{A_L+1}
\sum_{\alpha\geq A_L+2,\beta\geq 0}
c_{\alpha,\beta}
\left(\frac{x}{\mu}\right)^{\alpha-A_L-1}y^{\beta}.
\end{eqnarray*}
For $\lambda=\mu^{A_L+1}$, taking $\mu\rightarrow \infty$, the limit of $\lambda S(x/\mu,y)$ is $$\widetilde{S}(x,y)=\sum\limits_{\alpha=A_L+1,\beta\geq B_L+1}
c_{\alpha,\beta}
x^{\alpha}y^{\beta}.$$
By Fatou's lemma, it follows from (\ref{Bound-dilated-RT}) that
\begin{equation}\label{limit-standard-RT-Section3}
\left\|
\big|\partial_t\big|^{\delta}
\int_{-\infty}^{\infty}
f(t+\widetilde{S}(x,y),y)\varphi(0,0,y)dy
\right\|_{L^p(\Bbb R^2)}
\leq
C\|f\|_{L^p(\Bbb R^2)}
\end{equation}
for all $f\in C_0^{\infty}(\Bbb R^2)$. Assume $S(x,y)$ is real analytic in the square $(-\rho,\rho)^2$ for some small number $\rho>0$. Without loss of generality, assume $c_{A_L+1,B_L+1}>0$. Take $\rho>0$ sufficiently small such that (i) $\varphi(0,0,y)\neq 0$ for $y\in(-\rho,\rho)$ and (ii) $\sum c_{\alpha,\beta}y^{\beta-B_L-1}
>\frac{c_{A_L+1,B_L+1}}{2}$ for $y\in(-\rho,\rho)$, where the summation is taken over $\alpha= A_L+1,\beta\geq B_L+1$. In (\ref{limit-standard-RT-Section3}), we take smooth $f$ supported in $(-1,1)\times (\rho_1,\rho_2)$ with $0<\rho_1<\rho_2<\rho$. By a change of variables,
\begin{equation*}
u
=h(y)=u(y)y,
\end{equation*}
where $u$ is given by
\begin{equation*}
u(y)=\left(
\sum_{\beta\geq B_L+1} c_{A_L+1,\beta}y^{\beta-B_L-1}
\right)^{\frac{1}{B_L+1}  },~~~y\in (-\rho,\rho).
\end{equation*}
Let $[\epsilon_1,\epsilon_2]=h([\rho_1,\rho_2])$ be the image of $h$ on $[\rho_1,\rho_2]$, and $h^{-1}$ the inverse of $h$. It follows from (\ref{limit-standard-RT-Section3}) that
\begin{equation*}\label{limit-standard-RT}
\left\|
\big|\partial_t\big|^{\delta}
\int_{-\infty}^{\infty}
f(t+x^{A_L+1}u^{B_L+1},h^{-1}(u))
\varphi(0,0,h^{-1}(u))\frac{dh^{-1}(u)}{du}du
\right\|_{L^p(\Bbb R^2)}
\leq
C\|f\|_{L^p(\Bbb R^2)}
\end{equation*}
for all $f\in C_0^{\infty}((-1,1)\times (\rho_1,\rho_2))$. Define a function $\widetilde{f}$ by
$$
\widetilde{f}(t,u)=
f(t,h^{-1}(u))
\varphi(0,0,h^{-1}(u))\frac{dh^{-1}(u)}{du}.
$$
Then $f\rightarrow \widetilde{f}$ is a one-to-one mapping from $C_0^{\infty}((-1,1)\times (\rho_1,\rho_2))$ onto $f\in C_0^{\infty}((-1,1)\times (\epsilon_1,\epsilon_2))$. Moreover, $f$ and $\widetilde{f}$ have comparable $L^p$ norms. As a consequence of the above inequality, we obtain
\begin{equation*}
\left\|
\big|\partial_t\big|^{\delta}
\int_{-\infty}^{\infty}
\widetilde{f}(t+x^{A_L+1}u^{B_L+1},u)du
\right\|_{L^p(\Bbb R^2)}
\leq
C\|\widetilde{f}\|_{L^p(\Bbb R^2)}
\end{equation*}
for all $\widetilde{f}\in C_0^{\infty}((-1,1)\times (\epsilon_1,\epsilon_2))$.

y^{-B_L/(B_L+1)}$ is a smooth function in $C_0^{\infty}((-1,1)\times (\tau_1,\tau_2))$.
This correspondence is also one-to-one. Also, both $\widetilde{f}$ and $g$ have equivalent $L^p$ norms. Hence we get, for all $g\in C_0^{\infty}((-1,1)\times (\tau_1,\tau_2))$,
\begin{equation*}
\left\|
\big|\partial_t\big|^{\delta}
\int_{-\infty}^{\infty}
g(t+x^{A_L+1}y,y)dy
\right\|_{L^p(\Bbb R^2)}
\leq
C\|g\|_{L^p(\Bbb R^2)},
\end{equation*}
where $\frac{A_L+B_L+2}{A_L+1}\leq p<2$ and $\delta=\frac{1}{(A_L+1)p}$. Now we embed the left hand side in an analytic family of operators
\begin{equation*}
V_zg(t,x)=
x_{+}^{az}
\int_{\Bbb R}|\partial_t|^{L(z)}
g(t+x^{A_L+1}y,y)dy,~~~0\leq \,{\rm Re}\,(z) \leq 1,
\end{equation*}
where $a>0$ is to be determined and $L(z)=\frac{1-z}{A_L+B_L+2}+\frac{z}{2N_2}$. Here $N_2>1$ is a number to be fixed. We shall also find a number $N_1>$ such that $\frac{N_1}{N_1+1}
=\frac{\theta}{p}+\frac{1-\theta}{2}$ for some $\theta\in (0,1)$. The first step is to choose a $\theta\in (0,1)$ such that it is sufficiently close to $1$. Then $N_1$ is uniquely determined and sufficiently close to $\frac{1}{p-1}$. The second step is to determine $a>0$ such that $$\frac{A_L+1}{1+a\theta\frac{N_1+1}{N_1}}=N_1,
~~~
N_2=\frac{A_L+1}{1+2a}> 1.$$
The first equation gives
$$a=\frac{1}{\theta}
\left(\frac{A_L+1}{N_1+1}
-
\frac{N_1}{N_1+1}\right).$$
Since $N_1$ is close to $\frac{1}{p-1}>1$, it is true that $a<\frac{A_L}{2}$. Hence $N_2=\frac{A_L+1}{1+2a}>1$ for $\theta$ near one. By complex interpolation, for $z=\theta$, we get
\begin{equation*}
\left\|
\big|\partial_t\big|^{L(\theta)}
\int_{-\infty}^{\infty}
g(t+x^{N_1}y,y)dy
\right\|_{L^{\frac{N_1+1}{N_1}}(\Bbb R\times\Bbb R^+)}
\leq
C\|g\|_{\frac{N_1+1}{N_1}}
\end{equation*}
for all $g\in C_0^{\infty}((-1,1)\times (\tau_1,\tau_2))$. This is impossible by Theorem \ref{Thm-Failure-local-Radon-transform}. The proof of the theorem is complete.

|W_0f(x)
-
V_0f(x)|
\lesssim
\int_{A_0}
|x-r_1(y)|^{-2}
|y^{d_2}||f(y)|dy.
\end{equation*}
If we replace $A_0$ by $|x-r_1(y)|\geq 100 |C_1 y^{d_2}|$, then $W_0-V_0$ is bounded on $L^1$. On the other hand, $|y|\gtrsim |x|$ for $(x,y)\in A_0$. Observe that the kernel of $W_0-V_0$ is bounded by $C|y|^{d_2-2}$ since $|x-r_1(y)|\approx |y|$. Hence $W_0-V_0$ is also bounded on $L^{\infty}$.
\vspace{0.5\baselineskip}

For $W_1$, we use the same argument as above. Similarly,
\begin{equation*}
|W_1f(x)
-
V_1f(x)|
\lesssim
\int_{A_1}
|x-r_1(y)|^{-2}
|y^{d_2}||f(y)|dy.
\end{equation*}
For $(x,y)\in A_1$, it is clear that $|y|\lesssim |x|$. Then $|x-r_1(y)|^{-2}|y^{d_2}|\lesssim |x|^{-2}|y|^{d_2}$. Since $d_2>1$,
$W_1-V_1$ is bounded on $L^{\infty}$. Similarly, we also have $|x-r_1(y)|\gtrsim |y|^{d_2}$ in $A_1$. This implies that $W_1-V_1$ is bounded on $L^1$.
\vspace{0.5\baselineskip}

In view of
\begin{equation*}
|W_2f(x)
-
V_2f(x)|
\lesssim
\int_{A_2  }
|x-r_1(y)|^{-2}
|y^{d_2}||f(y)|dy.
\end{equation*}
It follows immediately that $W_2-V_2$ is bounded on $L^1$. For points in $A_2$, we deduce $|y|\approx |x|$. Then $|y|^{d_2}\approx |x|^{d_2}$ and $|x-r_1(y)|\gtrsim |x|^{d_2}$. Hence $W_2-V_2$ is also bounded on $L^{\infty}$.
\vspace{0.5\baselineskip}

The arguments for estimate of $W_3,W_4$ are somewhat different and more direct. First observe that $|x|\approx |y|$ for points in either $A_3$ or $A_4$. The distance between the singularities $r_i(y)$ is of order $O(|y|^{d_2})$. Each integral is taken over a measurable set of measure $O(|y|^{d_2})$. Hence both $W_3$ and $W_4$ are bounded on $L^{1}$. For example, we see that the kernel of $W_3$ has absolute value bounded by $C|y|^{-m_1d_2/N}
\prod_{i=2}^{M}|x-r_i(y)|^{-m_i/N}$.
A rearrangement inequality yields
$$\int_{A_3}\prod_{i=2}^{M}|x-r_i(y)|^{-\frac{m_i}N}
dx
\leq
\int_{|x|\lesssim |y|^{d_2}}|x|^{-1+\frac{m_1}{N}}
dx
\lesssim
|y|^{m_1d_2/N}.$$
Hence $W_3$ is bounded on $L^{1}$. By the symmetry of $x$ and $y$, it is also bounded on $L^{\infty}$. A similar argument yields that $W_4$ is bounded on $L^p$ for $1\leq p \leq \infty$.
\vspace{0.5\baselineskip}

{\bf Case 2. $r_{1}(y)\neq \sigma_1(y)$}
\vspace{0.5\baselineskip}

As in Case 1, suppose
$\prod_{i=1}^{N}\left(x-r_i(y)\right)=
\prod_{i=1}^{M}\left(x-r_i(y)\right)^{m_i},$
where $m_i$ is the multiplicity of $r_i$.
For $i=1,\cdots,M$, assume $r_i(y)-\sigma_1(y)=C_iy^{d_i}+o(y^{d_i})$ with $C_i\neq 0$ and $d_i>1$ for $i=1,\cdots,M$. Also assume $d_2\leq d_3\leq \cdots \leq d_M$. Also, if $d_i=d_j$ for some $2\leq i <j \leq M$, we also assume $|C_i|\geq |C_j|$. Here we do not know whether $d_1$ is large than $d_2$.

The simplest case is $d_1>d_2$. Then $r_2(y)-r_1(y)=C_2y^{d_2}+o(y^{d_2})$ as $y\rightarrow 0$. The argument in Case 1 is also applicable.

Consider the case $d_1=d_2$. The proof method in Case 1 can be adapted to this case. In the definition of $A_k$, the constant $C_2$ shall be replaced by $\,{\rm max}\{|C_1|,|C_2|\}$. If $d_1$ is not an integer, then $d_1=\frac{v}{u}$ and $u,v>0$ are relatively prime integers.  Since $S$ is real analytic, $\prod_{i=1}^{M}(x-r_i(y))^{m_i}$ is conjugate invariant. This implies that $r_i(y)=\sigma_1(y)+\exp({\frac{2\pi v }{u}}\sqrt{-1})C_1y^{d_1}+o(y^{d_1})$ for some $i\geq 2$. Hence there exists some $i\geq 2$ such that $r_i(y)-r_1(y)=Cy^{d_2}+o(y^{d_2})$
with $C\neq 0$. If $d_1$ is an integer, then $C_1$ is a pure imaginary number. Hence
$|x-r_1(y)|\gtrsim |y|^{d_2}$. All estimates in Case 1 are also true.

The final case is $d_1<d_2$. In view of $d_1>1$,
we use the conjugate invariance to obtain that $d_1$ is an integer and $C_1$ is a pure imaginary number. If $d_1$ were not an integer, then $r_1$ would have at least one conjugate Puiseux $r_j$. Then $r_j(y)-\sigma_1(y)=Cy^{d_1}+o(y^{d_1})$ for some $C\neq 0$. This contradicts our previous assumptions. Hence $d_1>1$ is an integer. By our definition of $\sigma_1(y)$, it is the analytic part of the real part of $r_1(y)$. This implies that $C_1$ is an imaginary number. Now we turn to estimates of $W_k$. The above argument in Case 1 should be modified slightly. Let $A_k$ be defined as Case 1 with $C_1y^{d_1}$ in place of $C_2y^{d_2}$. Our estimates for $W_0,W_1,W_2$ are the same as above. For $W_3,W_4$, first observe $|x-r_1(y)|\gtrsim |y|^{d_1}$ since $C_1y^{d_1}$ is contained in the imaginary part of $r_1(y)$. The argument in Case 1 also holds here.
\vspace{0.5\baselineskip}

Finally, each $V_k$ satisfies the estimate (\ref{truncated-maximal-sing-oper}). Combining this fact with the above results, our proof is complete.

\label{Thm-Maximal-RT}
Assume the maximal operator $T^\ast$ is defined by {\rm (\ref{maximal truncated operator})}. Then for every $1<p<\infty$, there exists a constant $C_p>0$ such that

<!-- proof-11: body_md -->
Choose a $C_0^\infty$ function $\eta$, $supp(\eta)\subseteq\{t\in \Bbb R^k:a/2\leq |t|\leq 2a\}$, such that $\sum_{j\in \Bbb Z}\eta (t/2^j)=1$ for $t\neq 0$. For $j\geq 0$,
\begin{equation}\label{Def-Tj}
T_jf(x)=\psi(x)\int_{\Bbb R^k}f(\gamma_t(x))K(t)\eta (2^jt)dt.
\end{equation}
Write $K_j(t)=K(t)\eta(2^jt), j\geq 0, j\in \Bbb N.$ Let the maximal operator $T^\ast$ be defined by
$$T^\ast f(x)=\sup_{m\geq 0}\left|\sum_{j=0}^m T_j f(x)\right|.$$
We are going to prove that $T^\ast$ is bounded from $L^p(\Bbb R^n)$ into itself for $1<p<\infty$.

For each $m>0$, decompose
\begin{equation}\label{decomposition}
\sum_{j=0}^m T_jf(x)=S_m\left(\sum_{j=0}^m T_jf\right)(x)+\left(I-S_m\right)\sum_{j=0}^mT_jf(x),
\end{equation}
where $I$ is the identity operator. For the first term,
$$S_m\left(\sum_{j=0}^m T_j f\right)=S_m Tf-S_m\left(\sum_{j=m+1}^\infty T_j f\right).$$
Then $S_m(Tf)$ is majorized by a constant multiple of $M(Tf)$, where the maximal operator $M$ is defined with repsect to the quasi-metric $d$. On the other hand, $S_m\left(\sum_{j=m+1}^\infty T_jf\right)$ is also bounded by $O(Mf(x))$, with the implicit constant independent of $m$. To see this,
observe that
\begin{eqnarray*}
&&
S_m\left(\sum_{j=m+1}^\infty T_j\right)f(x)\nonumber\\
&=&\sum_{j=m+1}^\infty \int_{\Bbb R^n} S_m(x,y)\left(\int_{\Bbb R^k}\psi(y)f(\gamma_t(y))K(t)\eta (2^jt)dt\right)dy\nonumber\\
&=&\sum_{j=m+1}^\infty \int_{\Bbb R^n}S_m(x,\gamma_t^{-1}(y)) \left(\int_{\Bbb R^k}\psi(\gamma_t^{-1}(y))f(y)K(t)\eta (2^j t)\det\left(\frac{\partial \gamma_t^{-1}(y)}{\partial y}\right) dt\right)dy.\nonumber
        \end{eqnarray*}
For each $j$, $j\geq m+1$, one has
\begin{eqnarray*}
S_m(x,\gamma_t^{-1}(y)&=&
S_m(x,y)+
 O\left(2^{m(Q+\frac{1}{N})}|t|^{1/N}\right),\\
 \psi(\gamma_t^{-1}(y))&=&\psi(y)+O(|t|),\\
 \det\left(\frac{\partial \gamma_t^{-1}(y)}{\partial y}\right)&=&1+O(|t|).
 \end{eqnarray*}
Using these estimates, we can see that
$$\sup_{m\geq 0}\left|
S_m\left(\sum_{j=m+1}^{\infty}T_j\right)f(x)
 \right|\lesssim Mf(x).$$
Now we turn to the second term in (\ref{decomposition}). Observe that $$(I-S_m)\sum_{j=0}^mT_j
=\sum_{j=-\infty}^0(I-S_m)T_{j+m}.$$
Here $T_k=0$ for $k<0$, by the definition of $T_j$ in (\ref{Def-Tj}). It suffices to show that for every $1<p<\infty,$
$$\left\| \left(\sum_{m=0}^\infty |(I-S_m)T_{j+m}f|^2\right)^{1/2}
 \right\|_{L^p(\Bbb R^n)}\leq C 2^{\epsilon j}\Vert f\Vert_{L^p(\Bbb R^n)},$$
where $C$ and $\epsilon >0$ are independent of $j\leq 0$. Let $\gamma_m(t)$ be the Rademacher functions on $[0,1]$. Then the above inequality can be deduced by
\begin{equation}\label{Esti-ell2-squre-function}
\left\| \sum_{m=0}^\infty \gamma_m(t)(I-S_m)T_{j+m}
f\right\|_{L^p(\Bbb R^n)}\leq C2^{\epsilon j}
\| f\|_{L^p(\Bbb R^n)}
\end{equation}
for $C$ and $\epsilon>0$ independent of $j$ and $t\in [0,1]$.

For every $1<p<\infty$, taking the adjoint operator, we reduce (\ref{Esti-ell2-squre-function}) to
$$\left\| \sum_{m=0}^\infty \gamma_m(t)T_{j+m}^\ast(I-S_m)\right\|_{L^{p'}\rightarrow L^{p'}}\leq C 2^{\epsilon j},~~~p'=\frac{p}{p-1}.$$
Here we make use of $S_m^\ast=S_m$. Since $p'$ can take any value in $(1,\infty)$, it suffices to show
$$\left\| \sum_{m=0}^\infty \gamma_m(t)T_{j+m}^\ast(I-S_m)\right\|_{L^{p}\rightarrow L^{p}}\leq C 2^{\epsilon j},~~~1<p<\infty$$
with $\epsilon>0$ depending on $p$. In view of
$$T_{j+m}^\ast=T_{j+m}'+E_{j+m},$$
with $\Vert E_{j+m}\Vert_{p\rightarrow p}=O(2^{-(j+m)})$, it suffices to show
$$\left\| \sum_{m=0}^\infty \gamma_m(t)T_{j+m}'(I-S_m)\right\|_{L^p\rightarrow L^p}\leq C2^{\epsilon j}.$$
Write $T_{j+m}'$ by $\widetilde{T}_{j+m}$. Then
\begin{eqnarray*}
\sum\limits_{m=0}^\infty \gamma_m(t)\widetilde{T}_{j+m}(I-S_m)f(x)
 &=& \sum\limits_{m=0}^\infty \gamma_m(t) \widetilde{T}_{j+m}
 \left(\sum\limits_{v=0}^{\infty}R_{m+v}\right)
 f(x)\nonumber\\
 &=& \sum\limits_{v=0}^{\infty} \left(\sum\limits_{m=0}^\infty \gamma_m(t)\widetilde{T}_{j+m}R_{m+v}f\right)
 \nonumber\\
 &=& \sum\limits_{v=0}^\infty \left(\sum\limits_{m=0}^\infty \gamma_{m-j}(t)\widetilde{T}_m R_{m+v-j }f\right).\nonumber
 \end{eqnarray*}
By the estimates (16.3) and (16.4) in \cite{Christ-Nagel-Stein-Wainger}, we have
$$\left\| \sum_{m=0}^\infty \gamma_{m-j}(t)\widetilde{T}_m R_{m+v-j}\right\|_{L^2\rightarrow L^2}\leq C2^{-\epsilon(v-j)}$$
for some $\epsilon >0$, and
$$\left\| \sum_{m=0}^\infty \gamma_{m-j}(t)\widetilde{T}_m
 R_{m+v-j}\right\|_{L^p\rightarrow L^p}\leq C_{p,\epsilon'}2^{\epsilon'(v-j)}$$
for some $\epsilon'>0$ and $1<p\leq 2$.
By interpolation, we obtain for $1<p\leq 2$,
\begin{equation}\label{(6)}
\left\| \sum_{m=0}^\infty \gamma_{m-j}(t)
\widetilde{T}_mR_{m+j-v}\right\|_{L^p\rightarrow L^p}\leq C2^{-\epsilon(v-j)}
\end{equation}
for some $\epsilon>0$, where $C$ is independent of $t\in [0,1]$. By a simple duality argument, we see that (\ref{(6)}) is also true for $2<p<\infty$. This completes the proof of the theorem.

f(x)+(X_1f)(x)t+\frac{1}{2!}(X_2f+X_1^2f)(x)t^2+O(t^3),
~~~ t\rightarrow 0.$$
Hence we obtain $$X_1=X(0)=\left(-\frac{\partial}{\partial x_1}+\frac{\partial S}{\partial x_2}(x_1,x_1)\frac{\partial}{\partial x_2}\right)|_x.$$
By direct calculation, it is easy to see that
$$X_2=\frac{\partial^2S}{\partial x_1\partial x_2}(x_1,x_1)\frac{\partial}{\partial x_2}|_x.$$
Since $\frac{\partial^2S}{\partial x_1\partial x_2}$ is a real-analytic function near the origin, we can divide our argument into two cases, depending on whether $\frac{\partial^2S}{\partial x_1\partial x_2}(x_1,x_1)$ vanishes identically.

If $\frac{\partial^2S}{\partial x_1\partial x_2}(x_1,x_1)$ is not identical to zero, then for some $k\in \mathbb{N}$ and small $x_1$,  $$\frac{d^k}{dx_1^k}\left(\frac{\partial^2 S}{\partial x_1\partial x_2}(x_1,x_1)\right)=\left(\frac{\partial}{\partial x_1}+\frac{\partial}{\partial x_2}\right)^k\frac{\partial^2S}{\partial x_1\partial x_2}(x_1,x_1)\neq 0.$$
We see that $X_1$ and $(ad X_1)^k(X_2)=
\left[X_1,\left[X_1,\left[\cdots,\left[X_1,
X_2\right]\cdots\right]\right]\right]$, with $k$ copies of $X_1$, span the tangent space $\mathbb{R}^2$. Hence $\gamma_t$ satisfies the condition $\mathcal{C}_\mathfrak{g}$.

Now consider the case $\frac{\partial^2S}{\partial x_1\partial x_2}(x_1,x_1)\equiv 0$ for small $x_1$. Let $M\geq 0$ be the smallest integer such that
\begin{equation}\label{vanishing-condition}
\frac{\partial^{l+2}S}{\partial x_1^{l+1}\partial x_2}(x_1,x_1)\equiv 0~~~\textrm{for~~$0\leq l \leq M$}.
\end{equation}
By the factorization (\ref{Sxy}), our assumption on $S$ implies that $0\leq M \leq N-1$. For integers $p,q\geq 0$ and $p+q\leq M$, we also have
 \begin{equation*}
\frac{\partial^{p+q+2}S}{\partial x_1^{p+1}\partial x_2^{q+1}}(x_1,x_1)\equiv 0.
\end{equation*}
For two arbitrary constants $C_1,C_2$, let $f(x)=C_1x_1+C_2x_2$. For $m\geq 2$, one can verify easily that
\begin{equation*}
\frac{d^m}{dt^m}\Big( f(\gamma_t(x) )\Big)
=
C_2(-1)^{m-1}\frac{\partial^m S}{\partial x_2^m}
(x_1,x_1-t).
\end{equation*}
By an induction argument, one has $X_k=0$ for $2\leq k \leq M+2$, where the vector fields $X_k$ are given by (\ref{exponential-expansion}). Then $X_{M+3}$ satisfies
$$X_{M+3}f+X_1^{M+3}f=C_2(-1)^{M+2}
\frac{\partial^{M+3} S}{\partial x_2^{M+3}}
(x_1,x_1).$$
Observe that $f(x)=C_1x_1+C_2x_2$. Hence, using (\ref{vanishing-condition}),
\begin{eqnarray*}
X_1^{M+3}f(x)
&= &
C_2(-1)^{M+2}\frac{d^{M+2}}{d x_1^{M+2}}
\Big( \frac{\partial S}{\partial x_2}(x_1,x_1)\Big)\\
&= &
C_2(-1)^{M+2}\frac{\partial^{M+3} S}{\partial x_1^{M+2}\partial x_2}(x_1,x_1)
+
C_2(-1)^{M+2}\frac{\partial^{M+3} S}{\partial x_2^{M+3}}(x_1,x_1).
\end{eqnarray*}
This gives
\begin{equation*}
X_{M+3}=(-1)^{M+3}\frac{\partial^{M+3} S}{\partial x_1^{M+2}\partial x_2}(x_1,x_1)\frac{\partial}{\partial x_2}.
\end{equation*}
By our definition of $M$, the function $\frac{\partial^{M+3}S}{\partial x_1^{M+2}\partial x_2}(x_1,x_1)$ does not vanish identically near the origin. There exists a nonnegative integer $k$ such that $X_1$ and $(adX_1)^{k}(X_{M+3})$ span the full tangent space $\mathbb{R}^2$. This proves that $\gamma_t$ satisfies the condition $\mathcal{C}_\mathfrak{g}$. Hence our proof is complete.

<!-- theorem-18: body_md -->
\label{Thm-Regularity-Unbouned-Faces}
Assume $S$ is a real-valued real analytic function near the origin in the plane. Let the Radon transform $Rf$ be defined as in {\rm(1.1)}, $(A_L,B_L)$ the leftmost extreme point of $N(S_{xy}'')$. Then the following statements are true.
\vspace{0.5\baselineskip}

\noindent {\rm(i)} If $A_L\geq B_L$, then there exits a constant $C>0$ such that
$$\|Rf\|_{L_{\delta}^p(\mathbb{R}^2)}\leq C\|f\|_{L^p(\mathbb{R}^2)},~~~\delta=\frac{1}{(A_L+1)p},
~~~2\leq p \leq \infty.$$
\noindent{\rm(ii)} If $A_L<B_L$, then
$$\|Rf\|_{L_\delta^p(\mathbb{R}^2)}\leq C\|f\|_{L^p(\mathbb{R}^2)},~~~ \delta=\frac{1}{(A_L+1)p},$$
where $\frac{A_L+B_L+2}{A_L+1}<p\leq +\infty$.

As shown in $\S$3, the statement in (i) is not true for all $\frac{A_L+B_L+2}{A_L+1}\leq p <2$. In other words, if $A_L>B_L$ then the critical Sobolev regularity breaks down on a bounded segment of the left unbounded face of $N_r(S)$.

If $A_L \geq B_L,$ then by the Phong-Stein Theorem \ref{THM-PS-L2-Regularity},
$$\|Rf\|_{L_{\frac{1}{2(A_L+1)}}^2}\leq C\|f\|_{L^2}.$$
By a complex interpolation, using $\|Rf\|_\infty\leq C\|f\|_\infty,$ we obtain the desired estimate in (i).\\

The proof of (ii) is more intricate. The $L^2$ Sobolev regularity of $Rf$ does not imply the desired result. A resolution of singularities for the Hessian $S_{xy}''$ is necessary to prove various $L^2$ Sobolev regularity estimates.

Assume $S_{xy}''(0,0)=0.$ Then $S(x,y)=\sum_{k,l}a_{k,l}x^k y^l$ is true for small $(x,y)$. By the Puiseux theorem,
$$S_{xy}''(x,y)=U(x,y)x^m y^s\prod_v(y-\gamma_v(x)),$$
where $U(x,y)\neq 0$ is smooth near the origin, $m,s\geq 0$, and $\gamma_v$ are non-trival Puiseux series. Now we shall use the resolution algorithm of $S_{xy}''$ by a complete classification of $\gamma_v$; see Phong-Stein \cite{PS1997}.

The exponents of $\gamma_v$ are ordered as follows:
$$0<a_1<a_2<\cdots<a_l<a_{l+1}<\cdots<a_n.$$
Denote by $\Psi \begin{bmatrix}
\cdot\\
l
\end{bmatrix}$ by the cluster of nonzero roots $\gamma_v$ of the form
\begin{equation}\label{section 7 formula 1}
\gamma_v(x)=C_l^\alpha x^{a_l}+o(x^{a_l}),\ x\rightarrow 0,
\end{equation}
where $C_l^\alpha\neq 0$. For roots $\gamma_v$ associated with a fixed exponent, we consider their coefficients  $C_l^\alpha$, where $\alpha$ is the index for the coefficient of $x^{a_l}$. Let $\Psi\begin{bmatrix}
\cdot\\
l
\end{bmatrix}=\bigcup_\alpha \Psi\begin{bmatrix}
\alpha\\
l
\end{bmatrix}.$
By induction, we continue this classification process to obtain
$$\Psi\begin{bmatrix}
\alpha_1,&\alpha_2,\dots,\alpha_{N-1},&\cdot\\
l_1,& l_2,\dots,l_{N-1},&l_N
\end{bmatrix}=\bigcup_{\alpha_N}\Psi\begin{bmatrix}
\alpha_1,&\alpha_2,&\dots,&\alpha_{N-1},&\alpha_N\\
l_1,&l_2,&\dots,&l_{N-1},&l_N
\end{bmatrix},$$
where the cluster $\Psi \begin{bmatrix}
\alpha_1,&\dots,&\alpha_N\\
l_1,&\dots,&l_N
\end{bmatrix}$ contains all roots of the form
$$\gamma_v(x)
=C_{l_1}^{\alpha_1}x^{a_1}
+C_{l_1,l_2}^{\alpha_1,\alpha_2}x^{a_{l_1,l_2}^{\alpha_1}}
+
\dots+C_{l_1,\dots,l_N}^{\alpha_1,\dots,
\alpha_N}x^{a_{l_1,\cdots,l_N}^{
\alpha_1,\dots,\alpha_{N-1}}}+
o\left(x^{a_{l_1,\cdots,l_N}^{
\alpha_1,\dots,\alpha_{N-1}}}\right).$$
Let $N\begin{bmatrix}
\alpha_1,\dots,\alpha_{k-1},\cdot\\
l_1,\dots,l_{k-1},l_k
\end{bmatrix}$ and $N\begin{bmatrix}
\alpha_1,\dots,\alpha_{k-1},\alpha_k\\
l_1,\dots,l_{k-1},l_k
\end{bmatrix}$ be the number of roots in the cluster $\Psi \begin{bmatrix}
\alpha_1,\dots,\alpha_{k-1},\cdot\\
l_1,\dots,l_{k-1},l_k
\end{bmatrix}$ and
$\Psi \begin{bmatrix}
\alpha_1,\dots,\alpha_{k-1},\alpha_k\\
l_1,\dots,l_{k-1},l_k
\end{bmatrix}$, respectively. For simplicity, we also use the notation
$$N=\sum_{l=1}^n N\begin{bmatrix}
\cdot\\
l
\end{bmatrix}=\sum_{l=1}^N the\ number\ of\ roots\ in\ \Psi\begin{bmatrix}
\cdot\\
l
\end{bmatrix}.$$
The vertices of the Newton polyhedron of $S_{xy}''$ are denoted by $(A_0,B_0),
(A_1,B_1),\cdots,(A_n,B_n)$ with $A_0<A_1<\cdots<A_n$ and $B_0>B_1>\dots>B_n$. Hence
$(A_0,B_0)$ is the leftmost extreme point, also denoted by $(A_L,B_L)$.

It is easy to verify that
$$A_0=m,~~~ B_0=s+\sum_{l=1}^nN\begin{bmatrix}
\cdot\\
l
\end{bmatrix},$$ and for each $1\leq r\leq n$,
$$A_r=m+\sum_{l=1}^rN\begin{bmatrix}
\cdot\\
l
\end{bmatrix}a_l,~~~ B_r=s+\sum_{l=r+1}^mN\begin{bmatrix}
\cdot\\
l
\end{bmatrix}.$$
Choose a smooth cut-off function $\Phi\in C_0^\infty(1/2,2)$ such that $\sum_{j\in \mathbb{Z}}\Phi(x/2^j)=1$ for all $x>0$. Define a localized Radon transform
\begin{equation}\label{Local-daydic-Radon-transform}
R_{j,k}^{\sigma_1,\sigma_2}f(x)
=
\int_{-\infty}^{\infty}f(t+S(x,y),y)
\varphi(x,y)\Phi(\sigma_12^{-j}x)\Phi(\sigma_22^{-k}y)dy,
\end{equation}
where $\sigma_1,\sigma_2\in\{+,-\}$. Here we only consider $R_{j,k}^{\sigma_1,\sigma_2}$ in the first quadrant, i.e., $\sigma_1=\sigma_2=+$. The same argument also works in other quadrants.\\

\noindent\textbf{Case (i).}\quad $k\geq a_1j+N_0$ for a large $N_0\geq 1$.\\

If $m>0$, then
$$|S_{xy}''(x,y)|\approx 2^{mj}2^{(N+s)k}$$
in the support of $R_{j,k}^{\sigma_1,\sigma_2}$. Using the Fourier transform with respect to $t$, we have
$$\widehat{R_{j,k}^{\sigma_1,\sigma_2}f}(\xi,x)
=\int_{-\infty}^{\infty}e^{2\pi i \xi S(x,y)}\hat{f}(\xi,y)\varphi(x,y)
\Phi(\sigma_1 2^{-j}x)\Phi(\sigma_2 2^{-k}y)dy.$$
Then by the van der Corput lemma,
$$\|\widehat{R_{j,k}^{\sigma_1,\sigma_2}f}(\xi,\cdot)\|_2\lesssim \left(|\xi|2^{mj}2^{(N+s)k}\right)^{-1/2}\|\hat{f}(\xi,\cdot)\|_2.$$
Hence, by an interpolation with the trivial size estimate
$$\|\widehat{R_{j,k}^{\sigma_1,\sigma_2}f}(\xi,\cdot)\|_2\lesssim 2^{j/2}2^{k/2}\|f\|_2,$$
we obtain
$$\|\widehat{R_{j,k}^{\sigma_1,\sigma_2}f}(\xi,\cdot)\|_2\lesssim \left(|\xi|2^{(N+s)k}\right)^{-\frac {1}{2(m+1)}}2^{\frac{mk}{2(m+1)}}\|\hat{f}(\xi,\cdot)\|_2.$$
This yields the Sobolev regularity
$$\||\partial_t|^{\frac{1}{2(m+1)}}
R_{j,k}^{\sigma_1,\sigma_2}f\|_2\lesssim 2^{k\delta}\|f\|_2$$
with $\delta$ given by
\begin{equation}\label{section 7 formula 2}
\delta=\frac{m}{2(m+1)}-\frac{N+s}{2(m+1)}.
\end{equation}
By an almost orthogonality argument, we also have
$$\left\|\sum_{j\leq k/a_1+0(1)}\widehat{R_{j,k}^{\sigma_1,
\sigma_2}f}(\xi,\cdot)\right\|_2\lesssim \left(|\xi|2^{(N+s)k}\right)^{-\frac{1}{2(m+1)}}
2^{\frac{mk}{2(m+1)}}\|\widehat{f}(\xi,\cdot)\|_2.$$
Hence
$$\left \||\partial_t|^{\frac{1}{2(m+1)}}\sum_{j\leq k/a_1+N_0/a_1}R_{j,k}^{\sigma_1,\sigma_2}f\right\|_2
\lesssim
2^{k\delta}\|f\|_2,$$
with the exponent $\delta$ given by (\ref{section 7 formula 2}). On the other hand,
$$\left\|\sum_{j\leq k/a_1+N_0/a_1}R_{j,k}f\right\|_\infty
 \lesssim
 2^k\|f\|_\infty.$$
A complex interpolation gives
$$\left\||\partial_t|^{\frac{1}{(m+1)p}}\sum_{j\leq (k-N_0)/a_1}R_{j,k}f\right\|_p
\lesssim
2^{k\delta_p}\|f\|_p,$$
where the exponent $\delta_p$ is defined by
\begin{equation}\label{delta-p}
\delta_p=-\frac{N+s}{m+1}\cdot \frac{1}{p}+\frac{m}{m+1}\cdot\frac{1}{p}+1-\frac{2}{p}
=1-\frac{m+N+s+2}{(m+1)p}>0
\end{equation}
if $p>\frac{A_L+B_L+2}{A_L+1}=\frac{m+N+s+2}{m+1}$.
This proves
$$\sum_{k\geq 0}\Big\||\partial_t|^{\frac{1}{(m+1)p}}
\sum_{j\ll k_1/a_0}R_{j,k}f\Big\|_p\lesssim \|f\|_p,~~~ p>\frac{A_L+B_L+2}{A_L+1}.$$
The case $m=0$ can be proved similarly without almost orthogonality argument. Consider
$$R_k^{\sigma_1,\sigma_2}
=
\sum_{j\ll k_1/a_1}R_{j,k}^{\sigma_1,\sigma_2}.$$
Then, by an interpolation of the oscillation and size estimates,
$$\|\widehat{R_k^{\sigma_1,\sigma_2}f}(\xi,\cdot)\|_2
\lesssim
\left(|\xi|2^{(N+s)k}\right)^{-\frac{1}{2}}
\|\widehat{f}(\xi,\cdot)\|_2$$
and
$$\||\partial_t|^{1/2}R_k^{\sigma_1,\sigma_2}f\|_2
\lesssim
2^{-(N+s)k/2}\|f\|_2.$$
In view of the size estimate
$$\|R_k^{\sigma_1,\sigma_2} f\|_\infty
\lesssim
2^k\|f\|_\infty.$$
It follows immediately that
$$\left\||\partial_t|^{\frac{1}{(m+1)p}}
R_k^{\sigma_1,\sigma_2}f\right\|_p
\lesssim
2^{k\delta_p}\|f\|_p,~~~ p>\frac{A_L+B_L+2}{A_L+1}$$
with $\delta_p>0$. This also implies the desired estimate.

A similar argument can be used to prove the regularity with respect to $x$, i.e.,
$$\int_{\xi \in \mathbb{R}}\int_{h\in \mathbb{R}}\int_{x\in \mathbb{R}}
\frac{|\widehat{R_k^{\sigma_1,\sigma_2}f}(\xi,x+h)
-\widehat{R_k^{\sigma_1,\sigma_2}f}(\xi,x)|^2}
{|h|^{1+\frac{1}{m+1}}}dxdhd\xi
\lesssim 2^{\frac{m-N-s}{m+1}k}\|\widehat{f}(\xi,\cdot)\|_2^2.$$
By interpolation, we obtain
$$\left\||\partial_x|^{\frac{1}{(A_L+1)p}}
R_k^{\sigma_1,\sigma_2} f\right\|_p
\lesssim
2^{k\delta_p}\|f\|_p,~~~ p>\frac{A_L+B_L+2}{A_L+1},$$
where $\delta_p$ is defined as in (\ref{delta-p}). Since $\delta_p$ is positive, we obtain the desired estimate by taking summation over $k$.\\

\noindent\textbf{Case 2.}\quad $k\leq a_nj-N_0$ for some large $N_0>0$.\\

Consider first the case $s>0$. On the support of $R_{j,k}^{\sigma_1,\sigma_2}$,
$$|S_{xy}''(x,y)|\approx 2^{mj}2^{sk}2^{j\sum_{l=1}^N N\begin{bmatrix}
\begin{smallmatrix}
\cdot\\ l
\end{smallmatrix}
\end{bmatrix}a_l}.$$
The partial Fourier transform $\widehat{R_{j,k}^{\sigma_1,\sigma_2}f}(\xi,\cdot)$ satisfies
$$\|\widehat{R_{j,k}^{\sigma_1,\sigma_2}f}(\xi,\cdot)\|_2
\lesssim
\left(|\xi|2^{mj}2^{sk}2^{j\sum_{l=1}^n N\begin{bmatrix}
\begin{smallmatrix}
\cdot\\ l
\end{smallmatrix}
\end{bmatrix}a_l}\right)^{-1/2}
\|\widehat{f}(\xi,\cdot)\|_2$$
and
$$\|\widehat{R_{j,k}f}(\xi,\cdot)\|_2\lesssim 2^{\frac{j}{2}}2^{\frac{k}{2}}
\|\widehat{f}(\xi,\cdot)\|_2.$$

Let $\theta=\frac{1}{m+1}$. A convex combination of the above two estimates gives
\begin{equation}\label{section 7 formula 3}
\|\widehat{R_{j,k}^{\sigma_1,\sigma_2}f}(\xi,\cdot)\|_2
\lesssim
\left(|\xi|2^{mj}2^{sk}2^{j\sum_{l=1}^N N\begin{bmatrix}
\begin{smallmatrix}
\cdot\\ l
\end{smallmatrix}
\end{bmatrix}a_l}\right)^{-\frac{\theta}{2}}2^{\frac{j}{2}(1-\theta)}2^{\frac{k}{2}(1-\theta)}
\|\hat{f}(\xi,\cdot)\|_2.
\end{equation}
The gives the estimate for $\||\partial_t|^{\frac{1}{2(m+1)}}
R_{j,k}^{\sigma_1,\sigma_2}f\|_2.$ With $\|R_{j,k}^{\sigma_1,\sigma_2}f\|_\infty\lesssim 2^k\|f\|_\infty$, an interpolation gives
\begin{equation}\label{section 7 formula 4}
\left\||\partial_t|^{\frac{1}{(m+1)p}}
R_{j,k}^{\sigma_1,\sigma_2}f\right\|_p
\lesssim
2^{j\delta_1}2^{k\delta_2}\|f\|^p,~~~ 2\leq p< +\infty,
\end{equation}
where $\delta_1$ and $\delta_2$ are given by
$$\delta_1=-\left(m+\sum_{l=1}^n N\begin{bmatrix}
\cdot\\
l
\end{bmatrix}a_l\right)\cdot \frac{\theta}{p}+\frac{1-\theta}{p},$$
$$\delta_2=-\frac{s\theta}{p}+\frac{1-\theta}{p}+1-\frac{2}{p}.$$
If $p>\frac{A_L+B_L+2}{A_L+1},$ then
$\delta_2=1-\frac{s+m+2}{(m+1)p}>0,\ A_L=m,\ B_L=s+\sum_{l=1}^nN\begin{bmatrix}
\begin{smallmatrix}
\cdot\\ l
\end{smallmatrix}
\end{bmatrix}.$
If we take the summation of (\ref{section 7 formula 4}) over $k\leq a_nj-N_0$, then the desired estimate is true provided that $\delta_1+a_n\delta_2>0$. Now we shall verify this simple observation. To see this, observe
\begin{align*}
\delta_1+a_n\delta_2&=-\left(m+\sum_{l=1}^n N
\begin{bmatrix}
\cdot\\
l
\end{bmatrix}a_l\right)\cdot \frac{1}{(m+1)p}+\frac{m}{(m+1)p}+a_n-\frac{(s+m+2)a_n}{(m+1)p}\\
&\geq a_n-a_n\left(s+m+2+\sum_{l=1}^n N
\begin{bmatrix}
\cdot\\ l
\end{bmatrix}\right)\frac{1}{(m+1)p}\\
&=a_n-a_n\frac{A_L+B_L+2}{(A_L+1)p}>0
\end{align*}
for $p>\frac{A_L+B_L+2}{A_L+1}$.
If $s=0$, we consider $R_j^{\sigma_1,\sigma_2}
=\sum_{k:k\leq a_nj-N_0}R_{j,k}^{\sigma_1,\sigma_2}$ instead of each piece $R_{j,k}^{\sigma_1,\sigma_2}$. The estimate for
$R_j^{\sigma_1,\sigma_2}$ is also true, similar to the case $s>0$,
$$\left\||\partial_t|^{\frac{1}{(m+1)p}}
R_j\sigma_1 f\|_p
\lesssim
2^{j(\delta_1+a_n\delta_2)}\right\|f\|_p,~~~ p>\frac{A_L+B_L+2}{A_L+1}.$$
The convergence of the summation over $j\leq 0$ implies the desired estimate.

Now consider the corresponding estimate for the variable $x$. Assume first $s>0$. Then
\begin{eqnarray}\label{section 7 formula 5}
& &
\widehat{R_{j,k}^{\sigma_1,\sigma_2}f}(\xi,x+h)
-\widehat{R_{j,k}^{\sigma_1,\sigma_2}f}(\xi,x)\nonumber\\
&=&
\int_{\mathbb{R}}\left(e^{2\pi i \xi[S(x+h,y)-S(x,y)]}-1\right)e^{2\pi i \xi S(x,y)}\hat{f}(\xi,y)\eta_{j,k}(x+h,y)dy \nonumber\\
& &
+\int_{\mathbb{R}}e^{2\pi i \xi S(x,y)}\hat{f}(x,y)\left(
\eta_{j,k}^{\sigma_1,\sigma_2}(x+h,y)
-
\eta_{j,k}^{\sigma_1,\sigma_2}(x,y)\right)dy,
\end{eqnarray}
where $\eta_{j,k}^{\sigma_1,\sigma_2}(x,y)
=\Phi(\sigma_1x/2^j)\Phi(\sigma_2y/2^k)\varphi(x,y)$. We shall prove that
$\||\partial_x|^{\frac{1}{2(m+1)}}
R_{j,k}^{\sigma_1,\sigma_2}f\|_2$ has the same bound as in (\ref{section 7 formula 4}).

If the integral of $h$ is restricted to $|h\xi|>1$, then we apply (\ref{section 7 formula 3}) to get
\begin{align*}
&\int_{\mathbb{R}}\int_{|h\xi|>1}\int_{\mathbb{R}}
\frac{|\widehat{R_{j,k}^{\sigma_1,\sigma_2}f}(\xi,x+h)-
\widehat{R_{j,k}^{\sigma_1,\sigma_2}f}(\xi,x)|^2}
{|h|^{1+\frac{1}{m+1}}}dxdhd\xi \\
\leq &2\int_{\mathbb{R}}\int_{|h\xi|>1}
|h|^{-1-\frac{1}{m+1}}\|\widehat{f}(\xi,\cdot)\|_2^2
\left(|\xi|~2^{sk}2^{j\sum_{l=1}^{n}N
\begin{bmatrix}
\begin{smallmatrix}
\cdot\\ l
\end{smallmatrix}
\end{bmatrix}a_l
}\right)^{-\frac{1}{m+1}}
2^{\frac{mk}{m+1}}dhd\xi\\
\lesssim &\left(2^{sk}2^{j\sum_{l=1}^n N
\begin{bmatrix}
\begin{smallmatrix}
\cdot\\ l
\end{smallmatrix}
\end{bmatrix}a_l}\right)^{-\frac{1}{m+1}}
2^{\frac{mk}{m+1}}
\|f\|_2^2.
\end{align*}
This bound is the same as that of (\ref{section 7 formula 4}) for $p=2$.

For the integral restricted to $|h\xi|\leq 1$, we write the right hand side of (\ref{section 7 formula 5}) as $E_{j,k}^{\sigma_1,\sigma_2}+F_{j,k}^{\sigma_1,\sigma_2}$. By the operator van der Corput lemma, we obtain
$$\int_{\mathbb{R}}|E_{j,k}^{\sigma_1,\sigma_2}|^2dx
\lesssim |h\xi|^2\left(|\xi|~2^{sk}2^{\sum_{l=1}^nN
\begin{bmatrix}
\begin{smallmatrix}
\cdot\\ l
\end{smallmatrix}
\end{bmatrix}a_l}\right)^{-\frac{1}{m+1}}
2^{\frac{mk}{m+1}}
\|\widehat{f}(\xi,\cdot)\|_2.$$
Hence
$$\int_{\mathbb{R}}\int_{|h\xi|<1}
\int_{\mathbb{R}}
\frac{|E_{j,k}^{\sigma_1,\sigma_2}|^2}{|h|^{1+\frac
{1}{m+1}}}dxdhd\xi
\lesssim
\left(2^{sk}2^{\sum_{l=1}^nN\begin{bmatrix}
\begin{smallmatrix}
\cdot\\ l
\end{smallmatrix}
\end{bmatrix}a_l}\right)^{-\frac{1}{m+1}}
2^{\frac{mk}{m+1}}
\|f\|_2^2.$$
As expected, this estimate is the same as (\ref{section 7 formula 4}) for $p=2$. For $F_{j,k}$, the oscillation estimate shall be replaced by the size estimate, using the Cauchy-Schwarz inequality,
$$\int_{\mathbb{R}}|F_{j,k}^{\sigma_1,\sigma_2}|^2dx
\lesssim
\,{\rm min} \left\{\left(\frac{|h|}{2^j}\right)^2,1\right\}2^j 2^k\|\widehat{f}(\xi,\cdot)\|_2^2.$$
A simple observation is
$$\int_{\mathbb{R}}|h|^{-1-\frac{1}{m+1}}\,{\rm min} \left\{\left(\frac{|h|}{2^j}\right)^2,1\right\}dh \lesssim 2^{-\frac{j}{m+1}}.$$
As a consequence, we obtain
$$\int_{\mathbb{R}}\int_{\mathbb{R}}\int_{\mathbb{R}}
|F_{j,k}^{\sigma_1,\sigma_2}|^2dxdhd\xi\lesssim
2^{mj/(m+1)}2^k\|f\|_2^2.$$
Interpolation with the trivial estimate $\|R_{j,k}f\|_\infty\lesssim 2^k\|f\|_\infty$ gives
$$\left\||\partial_x|^{\frac{1}{2(m+1)}}
R_{j,k}^{\sigma_1,\sigma_2}f\right\|_p
\lesssim A_{j,k}\|f\|_p,~~~ p>\frac{A_L+B_L+2}{A_L+1},$$
where $A_{j,k}$ satisfies $\sum_{j\leq 0}\sum_{k\leq a_nj-N_0}A_{j,k}<+\infty$.\\

\noindent\textbf{Case 3.}\quad $k=a_{t_1}j+O(1)$ for some $1\leq t_1 \leq n$.\\

Consider first $a_{t_1}\geq 1$. The inequality $|k-a_{t_1}j|\leq N_0$ implies that the zeros of $S_{xy}''$, on the support of $R_{j,k}^{\sigma_1,\sigma_2}$, are of form $y=cx^{a_{t_1}}+o(x^{a_{t_1}})$.
For this reason, we insert $\Phi(\sigma_3(y-C_{t_1}^{\alpha_1}x^{a_{t_1}})/2^{\mu_1})$ into the cut-off function of $R_{j,k}^{\sigma_1,\sigma_2}$. The resulting operator is denoted by $R_{j,k,\mu_1}^{\sigma_1,\sigma_2,\sigma_3}$, with smaller support.
If $\mu_1$ satisfies $\mu_1\geq k-N_0$ for some large $N_0$, then the zeros of the Hessian are far away from $y=C_{t_1}^{\alpha_1}x^{a_{t_1}}+o(x^{a_{t_1}})$.
Then choose another index $\alpha_1'\neq \alpha_1$ and further decompose $R_{j,k,\mu_1}^{\sigma_1,\sigma_2,\sigma_3}$ into smaller pieces $R_{j,k,\mu_1,\mu_2}^{\sigma_1,\sigma_2,\sigma_3,\sigma_4}$ by insertion of $\Phi(\sigma_4(y-C_{t_1}^{\alpha_1'}x^{a_{t_1}})/2^{\mu_2})$ into the cut-off function of $R_{j,k,\mu_1}^{\sigma_1,\sigma_2,\sigma_3}$. \\


For example, we consider $\mu_1\ll k$. For points in the support of $R_{j,k,\mu_1}^{\sigma_1,\sigma_2,\sigma_3}$, $$\left|y-C_{t_1}^{\alpha}x^{a_{t_1}}\right|\approx 2^k$$
for $\alpha\neq \alpha_1$. To further decompose $R_{j,k,\mu_1}^{\sigma_1,\sigma_2,\sigma_3}$ into smaller pieces, it is convenient to divide the range of $\mu_1$ into several cases:
\begin{enumerate}
\item[{\rm (i)}] $a_{t_1,1}^{\alpha_1}j\ll \mu_1\ll a_{t_1}j$;
\item[{\rm (ii)}] $a_{t_1,t+1}^{\alpha_1}j\ll \mu_1\ll a_{t_1,t}^{\alpha_1}j,\quad 1\leq t\leq {N}\begin{bmatrix}
    \begin{smallmatrix}
\alpha_1\\
t_1
\end{smallmatrix}
\end{bmatrix}-1$;
\item[{\rm (iii)}] $\mu_1\ll a_{t_1,t}^{\alpha_1}j$ with $t={N}\begin{bmatrix}
    \begin{smallmatrix}
\alpha_1\\
t_1
\end{smallmatrix}
\end{bmatrix}$;
\item[{\rm (iv)}] $\mu_1=a_{t_1,t}^{\alpha_1}j + O(1)$
for some $1\leq t\leq {N}\begin{bmatrix}
\begin{smallmatrix}
\alpha_1\\
t_1
\end{smallmatrix}
\end{bmatrix}$.
\end{enumerate}
For the previous three cases, the Hessian of the phase function behaves like a nonzero constant. We turn to the most delicate case  (iv). More precisely, we assume
$\left|\mu_1-a_{t_1,t_2}^{\alpha_1}j\right|\lesssim 1$. By insertion of the cut-off function $$\Phi(\sigma_4(y-C_{t_1}^{\alpha_1}x^{a_{t_1}}
-C_{t_1,t_2}^{\alpha_1,\alpha_2}
x^{a_{t_1,t_2}^{\alpha_1}})/{2^{\mu_2}}),$$
we decompose $R_{j,k,\mu_1}^{\sigma_1,\sigma_2,\sigma_3}$ into smaller pieces $R_{j,k,\mu_1,\mu_2}^{\sigma_1,\sigma_2,\sigma_3,\sigma_4}$. For the same reason, we may assume $\mu_2\ll \mu_1$
such that for $\alpha\neq \alpha_2$,
$
|y-C_{t_1}^{\alpha_1}x^{a_{t_1}}
-C_{t_1,t_2}^{\alpha_1,\alpha}x^{a_{t_1,t_2}^{\alpha_1}}|\approx 2^{\mu_1}
$
in the support of $R_{j,k,\mu_1,\mu_2}^{\sigma_1,\sigma_2,\sigma_3,\sigma_4}$.\\

By induction, if $R_{j,k,\mu_1,\cdots,\mu_p}^{\sigma_1,\sigma_2,
\cdots,\sigma_{p+2}}$ has been defined and $\mu_p$ lies in the range
$\mu_p=a_{t_1,\cdots,t_{p+1}}^{\alpha_1,\cdots,\alpha_p}j
+O(1),$
we choose $\alpha_{p+1}$ and insert
$$

-\sum_{i=1}^{p+1}C_{t_1,\cdots,t_{i}}^{\alpha_1,
\cdots,\alpha_i}

\cdots,\alpha_{i-1}}})/{2^{\mu_{p+1}}}\Big)

into the cut-off function of $R_{j,k,\mu_1,\cdots,\mu_{p}}^{\sigma_1,
\cdots,\sigma_{p+2}}$. In this way, we obtain a new operator $R_{j,k,\mu_1,\cdots,\mu_{p+1}}^{\sigma_1,\cdots,
\sigma_{p+3}}$. Similar to our discussion for $\mu_1$, there are also four possible ranges for $\mu_{p+1}$.\\

Assume the above procedure stops in the $N$-th step. Then $\mu_N$ must satisfy one of the following conditions:
\begin{enumerate}
\item[{\rm (i)}] $a_{t_1,\cdots,t_N,1}^{\alpha_1,\cdots,\alpha_N}j\ll \mu_N\ll \mu_{N-1}$;
\item[{\rm (ii)}] $a_{t_1,\cdots,t_N,t_{N+1}+1}^{\alpha_1,\cdots,\alpha_N}j\ll \mu_N\ll a_{t_1,\cdots,t_{N+1}}^{\alpha_1,\cdots,\alpha_N}j$ for some
    $1\leq t_{N+1}\leq \textbf{N}\begin{bmatrix}
\begin{smallmatrix}
\alpha_1,\cdots,\alpha_N\\
t_1,\cdots,t_N
\end{smallmatrix}
\end{bmatrix}-1$ and\\
$1\leq \alpha_N\leq \textbf{N}\begin{bmatrix}
\begin{smallmatrix}
\alpha_1,\cdots,\alpha_{N-1}, \cdot\\
t_1,\cdots,t_{N-1}, t_N
\end{smallmatrix}
\end{bmatrix}$;
\item[{\rm (iii)}] $\mu_N\ll a_{t_1,\cdots,t_{N+1}}^{\alpha_1,\cdots,\alpha_N}j$ with $t_{N+1}=\textbf{N}\begin{bmatrix}
\begin{smallmatrix}
\alpha_1,\cdots,\alpha_N\\
t_1,\cdots,t_N
\end{smallmatrix}
\end{bmatrix}$;
\item[{\rm (iv)}] For some $t$ we have $\mu_N= a_{t_1,\cdots,t_N,t}^{\alpha_1,\cdots,\alpha_N}j + O(1)$, and for all $\alpha$
\begin{equation*}

The statements (ii) and (iii) in Theorem \ref{Main-Thm-2} are contained in Theorem \ref{Thm-Regularity-Unbouned-Faces}. Now assume $(m,n)$ lies on the boundary of $N_r(S)$ and $A_L<m<A_R$, and it is not an extreme point of $N_r(S)$. This assumption implies that $N_r(S)$ has at least two extreme points. By duality, we also assume $m>n$. The case $m=n$ is proved in Phong and Stein \cite{PS1997}; see Theorem \ref{THM-PS-L2-Regularity}.
\vspace{0.5 \baselineskip}

\noindent {\bf Case 1. $S$ is not almost translation invariant.}
\vspace{0.5 \baselineskip}

Assume $(m,n)$ lies on the bounded face with endpoints $(A_{k-1},B_{k-1})$ and $(A_k,B_k)$. Then there exists a $\theta\in (0,1)$ such that $(m,n)=\theta(A_{k-1},B_{k-1})+(1-\theta)(A_k,B_k)$. Consider
\begin{equation*}
\mathbf{R}_zf(t,x)
=
\int_{\Bbb R}f(t+S(x,y),y)
|S''_{xy}(x,y)|^z\varphi(t,x,y)dy,
~~~-\frac{1}{m+n-2}\leq \,{\rm Re}\,(z) \leq \frac{1}{2}.
\end{equation*}
For $z$ on the left line $\,{\rm Re}\,(z)=-\frac{1}{m+n-2}$, $R_z$ is bounded on $L^{p_{\theta}}(\mathbb{R}^2)$ with $p_{\theta}=\frac{m+n-2}{m-1}$. On the other hand, for $z$ with real part $\,{\rm Re}\,(z)=\frac12$, it follows from Theorem \ref{Thm-L2-Damped-RT-Hessian} that $R_z$ is bounded from $L^2$ into $L^2_{1/2}$. By a complex interpolation argument, we have that $R=R_0$ is bounded from $L^p$ into $L^p_{1/(m+n)}$ with $p=\frac{m+n}{m}$.
\vspace{0.5 \baselineskip}

\noindent {\bf Case 2. $S$ is almost translation invariant.}
\vspace{0.5 \baselineskip}

Assume the Hessian $S''_{xy}$ is of the form
\begin{equation*}
S''_{xy}(x,y)=U(x,y)\prod_{i=1}^{N}(x-r_i(y)),
\end{equation*}
where $U(x,y)\neq0$ near the origin, and $r_i$ are convergent Puiseux series satisfying $r_i(0)=0$ and $r_i'(0)=c$ for some real number $c\neq 0$. By dilation, assume $c=1$. By our assumption, we have $1<m<N+1$ and $m+n=N+2$. Consider
\begin{equation*}
\mathbf{R}_zf(t,x)
=
\left(z+\frac{1}{N}\right)
\int_{\Bbb R}f(t+S(x,y),y)
\prod_{i=1}^{N}|(x-r_i(y))|^z\varphi(t,x,y)dy,
\end{equation*}
where $z$ lies on the strip $-\frac{1}{N}\leq \,{\rm Re}\,(z) \leq \frac{1}{2}$. A similar argument as in Case 1, we are able to show that $R=R_0$ is bounded from $L^p(\mathbb{R}^2)$ into $L^p_{1/(N+2)}(\mathbb{R}^2)$ for all $\frac{N+2}{N+1}<p<N+2$.
