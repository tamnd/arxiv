\noindent\textbf{Step 1:} Compute $\frac12\int_{M\cap\{r\geq2R_\epsilon\}}|\nabla_gG|^2\,dv_g$.
		
		\
		
		
		
		
		
		Integrating by parts, we get
		\begin{align*}
			\frac12\int_{M\cap\{r\geq2R_\epsilon\}}|\nabla_gG|^2\,dv_g &=-\frac12\int_{M\cap\{r\geq2R_\epsilon\}}\Delta_gG\cdot G\,dv_g +\frac12\int_{M\cap\{r=2R_\epsilon\}}G(-\partial_rG)\,ds_g\\
			&=\frac{2\pi}{|M|}\int_{M\cap\{r<2R_\epsilon\}}G\,dv_g
			+\frac12\int_{M\cap\{r=2R_\epsilon\}}G(-\partial_rG)\,ds_g.
		\end{align*}
		Since the divergence theorem yields
		$$
		\int_{M\cap\{r=2R_\epsilon\}}(-\partial_rG)\,ds_g
		=
		4\pi-\frac{4\pi}{|M|}
		\operatorname{Vol}_g\bigl(M\cap\{r<2R_\epsilon\}\bigr),
		$$
		then
		\begin{align}
			\frac12\int_{M\cap\{r\geq2R_\epsilon\}}|\nabla_gG|^2\,dv_g
			={}&-8\pi\log(2R_\epsilon)+2\pi A(p) \notag\\
			&+\frac12\int_{M\cap\{r=2R_\epsilon\}}
			\bigl[G+4\log(2R_\epsilon)-A(p)\bigr](-\partial_rG)\,ds_g \notag\\
			&+\frac{2\pi}{|M|}\int_{M\cap\{r<2R_\epsilon\}}
			\bigl[G+4\log(2R_\epsilon)-A(p)\bigr]\,dv_g.
			\label{eq:green-energy-exterior-reduction}
		\end{align}
		
		For $r<2R_\epsilon$,
		$$
		G+4\log(2R_\epsilon)-A(p)
		=
		-4\log\frac{r}{2R_\epsilon}+b_1(p)x_1+b_2(p)x_2+O(r^2).
		$$
		Since
		$$
		\int_0^{2R_\epsilon}\left|\log\frac{r}{2R_\epsilon}\right|r\,dr=O(R_\epsilon^2),
		$$
		we have
		$$
		\int_{M\cap\{r<2R_\epsilon\}}
		\bigl[G+4\log(2R_\epsilon)-A(p)\bigr]\,dv_g
		=
		O(R_\epsilon^2)=o(\sqrt{\epsilon}).
		$$
		
		On $r=2R_\epsilon$, we have
		$$
		G+4\log(2R_\epsilon)-A(p)
		=
		2R_\epsilon\bigl(b_1(p)\cos\theta+b_2(p)\sin\theta\bigr)+O(R_\epsilon^2),
		$$
		$$
		-\partial_rG
		=
		\frac{2}{R_\epsilon}-b_1(p)\cos\theta-b_2(p)\sin\theta+O(R_\epsilon),
		\quad
		ds_g=2R_\epsilon\bigl(1+O(R_\epsilon^2)\bigr)\,d\theta.
		$$
		Let $\theta_+$ and $\theta_-$ be defined so that the two endpoints of the circular arc $r=2R_\epsilon$ correspond to the polar angles $\theta_+$ and $\pi-\theta_-$, respectively. Then by the proof of Lemma \ref{lem:04}, we have
		$$
		\theta_+=\kappa(p)R_\epsilon+O(R_\epsilon^2),
		\qquad
		\theta_-=\kappa(p)R_\epsilon+O(R_\epsilon^2),
		$$
		and
		\begin{align*}
			\int_{\theta_+}^{\pi-\theta_-}\cos\theta\,d\theta
			=\sin\theta_- -\sin\theta_+=O(R_\epsilon^2),\
			\int_{\theta_+}^{\pi-\theta_-}\sin\theta\,d\theta
			=\cos\theta_++\cos\theta_-=2+O(R_\epsilon^2).
		\end{align*}
		Therefore,
		\begin{align*}
			&\frac12\int_{M\cap\{r=2R_\epsilon\}}
			\left(G+4\log(2R_\epsilon)-A(p)\right)(-\partial_rG)\,ds_g\\
			&=
			4R_\epsilon\int_{\theta_+}^{\pi-\theta_-}
			\bigl(b_1(p)\cos\theta+b_2(p)\sin\theta\bigr)\,d\theta
			+O(R_\epsilon^2)=
			8b_2(p)R_\epsilon+O(R_\epsilon^2).
		\end{align*}
		Substituting these estimates into
		\eqref{eq:green-energy-exterior-reduction}, and using
		$b_2(p)=-2\kappa(p)$ and $R_\epsilon^2=o(\sqrt{\epsilon})$, we obtain
		\begin{equation}\label{eq:complete-green-energy-exterior-final}
			\frac12\int_{M\cap\{r\geq2R_\epsilon\}}|\nabla_gG|^2\,dv_g
			=
			-8\pi\log(2R_\epsilon)+2\pi A(p)
			+16\bigl(b_2(p)+\kappa(p)\bigr)R_\epsilon
			+o(\sqrt{\epsilon}).
		\end{equation}
		
		
		\
		
		\noindent\textbf{Step 2:} Compute $\frac12\int_{M\cap\{r_\epsilon\geq2R_\epsilon,\ r\leq\delta\}}
		|\nabla_g\phi_{\epsilon,\delta}|^2\,dv_g-\frac12\int_{M\cap\{2R_\epsilon\leq r\leq\delta\}}
		|\nabla_gG|^2\,dv_g $.
		
		\
		
		
		
		
		In $M\cap\{r_\epsilon\geq2R_\epsilon,\ r\leq\delta\}$, we have
		\begin{align*}
			\nabla_g\phi_{\epsilon,\delta}
			=&
			\nabla_g\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr]
			+
			\nabla_g\bigl(G(x,p)+4\log r\bigr),\\
			\nabla_gG
			=&
			\nabla_g(-4\log r)
			+
			\nabla_g\bigl(G(x,p)+4\log r\bigr).
		\end{align*}
		Then
		\begin{align*}
			&\frac12\int_{M\cap\{r_\epsilon\geq2R_\epsilon,\ r\leq\delta\}}
			|\nabla_g\phi_{\epsilon,\delta}|^2\,dv_g
			-
			\frac12\int_{M\cap\{2R_\epsilon\leq r\leq\delta\}}
			|\nabla_gG|^2\,dv_g \notag\\
			&=
			\frac12\int_{M\cap\{r_\epsilon\geq2R_\epsilon,\ r\leq\delta\}}
			\left|\nabla_g\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr]\right|^2\,dv_g
			-
			\frac12\int_{M\cap\{2R_\epsilon\leq r\leq\delta\}}
			|\nabla_g(-4\log r)|^2\,dv_g \notag\\
			&\quad+
			\int_{M\cap\{r_\epsilon\geq2R_\epsilon,\ r\leq\delta\}}
			\left\langle
			\nabla_g\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr],
			\nabla_g\bigl(G(x,p)+4\log r\bigr)
			\right\rangle_g\,dv_g \notag\\
			&\quad-
			\int_{M\cap\{2R_\epsilon\leq r\leq\delta\}}
			\left\langle
			\nabla_g(-4\log r),
			\nabla_g\bigl(G(x,p)+4\log r\bigr)
			\right\rangle_g\,dv_g \notag\\
			&\quad+
			\frac12\int_{M\cap\{r_\epsilon\geq2R_\epsilon,\ r\leq\delta\}}
			\left|\nabla_g\bigl(G(x,p)+4\log r\bigr)\right|^2\,dv_g -
			\frac12\int_{M\cap\{2R_\epsilon\leq r\leq\delta\}}
			\left|\nabla_g\bigl(G(x,p)+4\log r\bigr)\right|^2\,dv_g\\
			&:=\mathbf{I_1}+\mathbf{I_2}+\mathbf{I_3}.
		\end{align*}
		
		For $\mathbf{I_1}$, noting that
		\begin{align*}
			\int_{\{r_\epsilon\leq \delta\}}|g^{-2}-r^{-2}||\frac{\partial}{\partial\theta}(-2\log(\epsilon+r_\epsilon^2))|^2 dv_g&= \int_{\{r_\epsilon\leq \delta\}}\left|\frac{\Pi}{3}+O(r)\right|\left| \frac{4a\sqrt{\epsilon}r\cos\theta}{r_\epsilon^2+\epsilon}\right|^2 dv_g\\&\leq C\int_{\{r_\epsilon\leq \delta\}}\left(\left| \frac{4a\sqrt{\epsilon}r\cos\theta}{r_\epsilon^2+\epsilon}\right|^2 \right)dx\\
			&\leq C\epsilon\int_{\{r_\epsilon\leq \delta\}}\frac{1}{r_\epsilon^2+\epsilon}dx\leq C\epsilon\log\alpha=o(\sqrt{\epsilon})
		\end{align*} and
		$$
		\left|D\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr]\right|^2
		-|D(-4\log r)|^2=O(\frac{\sqrt{\epsilon}x_2+\epsilon}{r^4}),\ for\  r_\epsilon\geq2R_\epsilon,\ 2R_\epsilon\leq  r\leq\delta,
		$$
		we have
		\begin{align*}
			\mathbf{I_1}
			&=\frac12\int_{M\cap\{r_\epsilon\geq2R_\epsilon,\ r\leq\delta\}}
			\left|D\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr]\right|^2\,dv_g
			-\frac12\int_{M\cap\{2R_\epsilon\leq r\leq\delta\}}
			|D(-4\log r)|^2\,dv_g +o(\sqrt{\epsilon})\notag\\
			&=
			\frac12\int_{M\cap\{r_\epsilon\geq2R_\epsilon,\ 2R_\epsilon\leq  r\leq\delta\}}
			\left(
			\left|D\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr]\right|^2
			-|D(-4\log r)|^2
			\right)dv_g +o(\sqrt{\epsilon})\notag\\
			&\quad+
			\frac12\int_{M\cap\{r_\epsilon\geq2R_\epsilon,\ r<2R_\epsilon\}}
			\left|D\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr]\right|^2\,dv_g -
			\frac12\int_{M\cap\{r\geq2R_\epsilon,\ r_\epsilon<2R_\epsilon\}}
			|D(-4\log r)|^2\,dv_g\\
			&=
			\frac12\int_{M\cap\{r_\epsilon\geq2R_\epsilon,\ 2R_\epsilon\leq  r\leq\delta\}}
			\left(
			\left|D\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr]\right|^2
			-|D(-4\log r)|^2
			\right)dx +o(\sqrt{\epsilon})+O(\delta\sqrt{\epsilon})\\
			&\quad+
			\frac12\int_{M\cap\{r_\epsilon\geq2R_\epsilon,\ r<2R_\epsilon\}}
			\left|D\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr]\right|^2\,dx -
			\frac12\int_{M\cap\{r\geq2R_\epsilon,\ r_\epsilon<2R_\epsilon\}}
			|D(-4\log r)|^2\,dx.
		\end{align*}
		
		
		
		
		
		Since
		\begin{align*}
			\{r_\epsilon\geq2R_\epsilon,\ r<2R_\epsilon, \ 0\leq x_2\leq f(x_1)\}\subset \{\sqrt{4R_\epsilon^2-(x_2+a\sqrt{\epsilon})^2}\leq |x_1|\leq 2R_\epsilon, \ 0\leq x_2\leq f(x_1)\}
		\end{align*} and $$2R_\epsilon-\sqrt{4R_\epsilon^2-(x_2+a\sqrt{\epsilon})^2}=O(\frac{\epsilon}{R_\epsilon})=o(\sqrt{\epsilon}),$$
		we get
		\begin{align*}
			&\int_{\{r_\epsilon\geq2R_\epsilon,\ r<2R_\epsilon, \ 0\leq x_2\leq f(x_1)\}}
			\left|D\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr]\right|^2\,dx=o(\sqrt{\epsilon}).
		\end{align*}
		Similarly,
		\begin{align*}
			\int_{\{r\geq2R_\epsilon,\ r_\epsilon<2R_\epsilon, \ 0\leq x_2\leq f(x_1)\}}
			|D(-4\log r)|^2\,dx=o(\sqrt{\epsilon}).
		\end{align*}
		
		Noting that
		\begin{align*}
			&\int_{\{r_\epsilon\geq2R_\epsilon,\ 2R_\epsilon\leq  r\leq\delta,\ 0\leq x_2\leq f(x_1)\}}
			\left|
			\left|D\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr]\right|^2
			-|D(-4\log r)|^2
			\right|dx\\& \leq C\int_{R_\epsilon}^\delta\int_0^{f(x_1)}\frac{\sqrt{\epsilon}x_2+\epsilon}{r^4}dx_2dx_1=O(\delta\sqrt{\epsilon})+o(\sqrt{\epsilon}),
		\end{align*}
		we have
		\begin{align*}
			\mathbf{I_1}
			&=\frac12\int_{\{r_\epsilon\geq2R_\epsilon,\ r\leq\delta,\ x_2\geq 0\}}
			\left|D\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr]\right|^2\,dx
			-\frac12\int_{\{2R_\epsilon\leq r\leq\delta, \ x_2\geq 0\}}
			|D(-4\log r)|^2\,dx \\&\quad +o(\sqrt{\epsilon})+O(\delta\sqrt{\epsilon})\\
			&=8\int_{r_\epsilon\geq2R_\epsilon,\ r\leq\delta,\ x_2\geq 0}
			\frac{r_\epsilon^2}{(\epsilon+r_\epsilon^2)^2}\,dx
			-
			8\int_{2R_\epsilon\leq r\leq\delta,\ x_2\geq 0}
			\frac1{r^2}\,dx +
			O(\delta\sqrt{\epsilon})
			+o(\sqrt{\epsilon}).
		\end{align*}
		
		On one hand, by \eqref{equation:09} and \eqref{equation:10}, we have
		\begin{align*}
			&8\int_{r_\epsilon\leq \alpha\sqrt{\epsilon},\ x_2\geq 0}
			\frac{r_\epsilon^2}{(\epsilon+r_\epsilon^2)^2}\,dx\\ &=8\int_{y\in B^+_\alpha(0),\ y_2\geq a}
			\frac{|y|^2}{(1+|y|^2)^2}\,dy \\
			&=8\pi\log \alpha+\frac{8\pi}{\alpha^2}+16\frac{a}{\alpha}-4\pi-8\pi\log (\sqrt{1+a^2}+a)+4\pi\frac{a}{\sqrt{1+a^2}}+O(\frac{1}{\alpha^3}),
		\end{align*} which implies
		\begin{align*}
			8\int_{2\alpha\sqrt{\epsilon}\leq r_\epsilon\leq\delta,\ x_2\geq 0}
			\frac{r_\epsilon^2}{(\epsilon+r_\epsilon^2)^2}\,dx&=\left(8\pi\log s+\frac{8\pi}{s^2}+16\frac{a}{s}\right)\bigg|_{s=2\alpha}^{s=\frac{\delta}{\sqrt{\epsilon}}}+O(\frac{1}{\alpha^3})\\
			&=8\pi\log \frac{\delta}{2R_\epsilon}+8\pi(\frac{\epsilon}{\delta^2}-\frac{1}{4\alpha^2})+16a(\frac{\sqrt{\epsilon}}{\delta}-\frac{1}{2\alpha}) +O(\frac{1}{\alpha^3})\\
			&=8\pi\log \frac{\delta}{2R_\epsilon}-\frac{2\pi}{\alpha^2}+16a\frac{\sqrt{\epsilon}}{\delta}-\frac{8a}{\alpha} +o(\sqrt{\epsilon}).
		\end{align*}
		
		On the other hand, since
		\begin{align*}
			8\int_{ r_\epsilon\geq \delta,\ r\leq\delta ,\ x_2\geq 0}
			\frac{r_\epsilon^2}{(\epsilon+r_\epsilon^2)^2}\,dx&= 8\int_{r_\epsilon\geq \delta,\ r\leq\delta ,\ x_2\geq 0}
			\frac{1}{r^2}\,dx+O(\frac{\sqrt{\epsilon}\delta+\epsilon}{\delta^4})\int_{r_\epsilon\geq \delta,\ r\leq\delta ,\ x_2\geq 0}
			\,dx\\
			&= 8\int_{|y|\geq \frac{\delta}{\sqrt{\epsilon}},\ y_1^2+(y_2-a)^2\leq\frac{\delta^2}{\epsilon} ,\ y_2\geq a}
			\frac{1}{y_1^2+(y_2-a)^2}\,dy+O(\frac{\sqrt{\epsilon}\delta+\epsilon}{\delta^4})\cdot O(\delta\sqrt{\epsilon})\\
			&=8\int_0^\pi d\theta\int_{-a\sin\theta+\sqrt{\frac{\delta^2}{\epsilon}-a^2\cos^2\theta}}^{\frac{\delta}{\sqrt{\epsilon}}}\frac{1}{t}dt+ o(\sqrt{\epsilon})\\
			&=16a\frac{\sqrt{\epsilon}}{\delta}+ o(\sqrt{\epsilon}),
		\end{align*}
		where $y_1=t\cos\theta$, $y_2=a+t\sin\theta$, then
		\begin{align*}
			8\int_{r_\epsilon\geq2R_\epsilon,\ r\leq\delta,\ x_2\geq 0}
			\frac{r_\epsilon^2}{(\epsilon+r_\epsilon^2)^2}\,dx= 8\pi\log \frac{\delta}{2R_\epsilon}-\frac{2\pi}{\alpha^2}+32a\frac{\sqrt{\epsilon}}{\delta}-\frac{8a}{\alpha} +o(\sqrt{\epsilon}).
		\end{align*}
		
		Noting the fact that $$8\int_{2R_\epsilon\leq r\leq\delta,\ x_2\geq 0}
		\frac1{r^2}\,dx= 8\pi\log \frac{\delta}{2R_\epsilon},$$
		thus,
		\begin{align*}
			\mathbf{I_1}=
			-\frac{8a}{\alpha}-\frac{2\pi}{\alpha^2}+32a\frac{\sqrt{\epsilon}}{\delta} +
			O(\delta\sqrt{\epsilon})
			+o(\sqrt{\epsilon}).
		\end{align*}
		
		\
		
		For $\mathbf{I_2}$, noting that
		\begin{align*}
			&\int_{M\cap\{2R_\epsilon\leq r\leq\delta\}}
			\left\langle
			\nabla_g(-4\log r),
			\nabla_g\bigl(G(x,p)+4\log r\bigr)
			\right\rangle_g\,dv_g\\&=\int_{M\cap\{2R_\epsilon\leq r\leq\delta\}}
			D(-4\log r)D\bigl(G(x,p)+4\log r\bigr)
			\,dv_g
		\end{align*}
		and
		\begin{align*}
			&\int_{M\cap\{r_\epsilon\geq2R_\epsilon,\ r\leq\delta\}}
			\left\langle
			\nabla_g\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr],
			\nabla_g\bigl(G(x,p)+4\log r\bigr)
			\right\rangle_g\,dv_g \\
			&=\int_{M\cap\{r_\epsilon\geq2R_\epsilon,\ r\leq\delta\}}
			D\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr]D\bigl(G(x,p)+4\log r\bigr)
			\,dv_g\\&\quad +\int_{M\cap\{r_\epsilon\geq2R_\epsilon,\ r\leq\delta\}}
			(g^{-2}-r^{-2})\frac{\partial}{\partial\theta}\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr] \cdot\frac{\partial}{\partial\theta}\bigl(G(x,p)+4\log r\bigr)
			\,dv_g\\
			&=\int_{M\cap\{r_\epsilon\geq2R_\epsilon,\ r\leq\delta\}}
			D\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr]D\bigl(G(x,p)+4\log r\bigr)
			\,dv_g+O(\delta\sqrt{\epsilon}),
		\end{align*}
		we have
		\begin{align*}
			\mathbf{I_2}&=\int_{M\cap\{r_\epsilon\geq2R_\epsilon,\ r\leq\delta\}}
			D\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr]D\bigl(G(x,p)+4\log r\bigr)
			\,dv_g\\&\quad -\int_{M\cap\{2R_\epsilon\leq r\leq\delta\}}
			D(-4\log r)D\bigl(G(x,p)+4\log r\bigr)
			\,dv_g+O(\delta\sqrt{\epsilon}).
		\end{align*}
		Following the same argument as in the treatment of $\mathbf{I_1}$, we get
		\begin{align*}
			\mathbf{I_2}&=\int_{M\cap\{r_\epsilon\geq2R_\epsilon,\ r\leq\delta\}}
			D\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr]D\bigl(G(x,p)+4\log r\bigr)
			\,dx\\&\quad -\int_{M\cap\{2R_\epsilon\leq r\leq\delta\}}
			D(-4\log r)D\bigl(G(x,p)+4\log r\bigr)
			\,dx+O(\delta\sqrt{\epsilon})+o(\sqrt{\epsilon})\\
			&=\int_{\{r_\epsilon\geq2R_\epsilon,\ r\leq\delta,\ x_2\geq 0\}}
			D\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr]D\bigl(G(x,p)+4\log r\bigr)
			\,dx\\&\quad -\int_{\{2R_\epsilon\leq r\leq\delta,\ x_2\geq 0\}}
			D(-4\log r)D\bigl(G(x,p)+4\log r\bigr)
			\,dx+O(\delta\sqrt{\epsilon})+o(\sqrt{\epsilon})\\
			&=\int_{\{r_\epsilon\geq2R_\epsilon,\ 2R_\epsilon\leq r\leq\delta,\ x_2\geq 0\}}
			\left(D\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr]-D(-4\log r) \right)D\bigl(G(x,p)+4\log r\bigr)
			\,dx\\&\quad +\int_{\{r_\epsilon\geq2R_\epsilon,\ r\leq 2R_\epsilon,\ x_2\geq 0\}}
			D\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr]D\bigl(G(x,p)+4\log r\bigr)
			\,dx\\&\quad -\int_{\{r\geq 2R_\epsilon, \ r_\epsilon\leq 2R_\epsilon,\ x_2\geq 0\}}
			D(-4\log r)D\bigl(G(x,p)+4\log r\bigr)
			\,dx+O(\delta\sqrt{\epsilon})+o(\sqrt{\epsilon}).
		\end{align*}
		Noting that $$\int_{\{r_\epsilon\geq2R_\epsilon,\ 2R_\epsilon\leq r\leq\delta,\ x_2\geq 0\}}
		\frac{x_1}{\epsilon+r_\epsilon^2}
		\,dx=\int_{\{r_\epsilon\geq2R_\epsilon,\ 2R_\epsilon\leq r\leq\delta,\ x_2\geq 0\}}
		\frac{x_1}{r^2}
		\,dx=0$$
		and
		$$\left|D\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr]-D(-4\log r) \right|\leq C\frac{\epsilon+r\sqrt{\epsilon}}{r^3},$$ we have
		\begin{align*}
			&\int_{\{r_\epsilon\geq2R_\epsilon,\ 2R_\epsilon\leq r\leq\delta,\ x_2\geq 0\}}
			\left(D\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr]-D(-4\log r) \right)D\bigl(G(x,p)+4\log r\bigr)
			\,dx\\
			&= \int_{\{r_\epsilon\geq2R_\epsilon,\ 2R_\epsilon\leq r\leq\delta,\ x_2\geq 0\}}
			\left(\frac{-4x_1}{\epsilon+r_\epsilon^2}-\frac{-4x_1}{r^2}\right)(b_1+O(r))
			\,dx\\
			&\quad + \int_{\{r_\epsilon\geq2R_\epsilon,\ 2R_\epsilon\leq r\leq\delta,\ x_2\geq 0\}}
			\left( \frac{-4(x_2+a\sqrt{\epsilon})}{\epsilon+r_\epsilon^2}-\frac{-4x_2}{r^2}\right)(b_2+O(r))
			\,dx\\
			&=b_2 \int_{\{r_\epsilon\geq2R_\epsilon,\ 2R_\epsilon\leq r\leq\delta,\ x_2\geq 0\}}
			\left( \frac{-4(x_2+a\sqrt{\epsilon})}{\epsilon+r_\epsilon^2}-\frac{-4x_2}{r^2}\right)
			\,dx+O(\delta\sqrt{\epsilon})+o(\sqrt{\epsilon}).
		\end{align*}
		
		
		
		Similarly, we have
		\begin{align*}
			&\int_{\{r_\epsilon\geq2R_\epsilon,\ r\leq 2R_\epsilon,\ x_2\geq 0\}}
			D\bigl[-2\log(\epsilon+r_\epsilon^2)\bigr]D\bigl(G(x,p)+4\log r\bigr)
			\,dx\\&= \int_{\{r_\epsilon\geq2R_\epsilon,\ r\leq 2R_\epsilon,\ x_2\geq 0\}}
			\left(\frac{-4x_1}{\epsilon+r_\epsilon^2}(b_1+O(r))+ \frac{-4(x_2+a\sqrt{\epsilon})}{\epsilon+r_\epsilon^2}(b_2+O(r))\right)dx\\&= b_2\int_{\{r_\epsilon\geq2R_\epsilon,\ r\leq 2R_\epsilon,\ x_2\geq 0\}}
			\frac{-4(x_2+a\sqrt{\epsilon})}{\epsilon+r_\epsilon^2}dx+ o(\sqrt{\epsilon})
		\end{align*}
		and
		\begin{align*}
			-\int_{\{r\geq 2R_\epsilon, \ r_\epsilon\leq 2R_\epsilon,\ x_2\geq 0\}}
			D(-4\log r)D\bigl(G(x,p)+4\log r\bigr)
			\,dx
			=-b_2\int_{\{r\geq2R_\epsilon,\ r_\epsilon\leq 2R_\epsilon,\ x_2\geq 0\}}
			\frac{-4x_2}{r^2}dx+ o(\sqrt{\epsilon}).
		\end{align*}
		Thus,
		\begin{align*}
			\mathbf{I_2}&= b_2\int_{\{r_\epsilon\geq2R_\epsilon,\ r\leq \delta,\ x_2\geq 0\}}
			\frac{-4(x_2+a\sqrt{\epsilon})}{\epsilon+r_\epsilon^2}dx- b_2\int_{\{2R_\epsilon\leq  r\leq \delta,\ x_2\geq 0\}}
			\frac{-4x_2}{r^2}dx+O(\delta\sqrt{\epsilon})+ o(\sqrt{\epsilon}).
		\end{align*}
		
		Setting $x_1=\sqrt{\epsilon}t\cos\theta$,   $x_2=\sqrt{\epsilon}t\sin\theta$, a direct computation yields
		\begin{align*}
			&-4 b_2\int_{\{r_\epsilon\geq2R_\epsilon,\ r\leq \delta,\ x_2\geq 0\}}

	&+4(2\pi-\rho)\log(2\pi-\rho)
	-2\rho\log K(z)-4(2\pi-\rho)\log h(z).
\end{aligned}
$$
Thus,
\begin{equation}\label{eq:joint-fixed-mass-threshold}
	m=\inf_{u\in W^{1,2}(M)}\widehat J(u,\rho_*)
	\geq\min_{z\in\partial M}L(\rho_*,z).
\end{equation}


For every fixed $z\in\partial M$, a direct computation gives
$$
\frac{\partial}{\partial\rho} L(\rho,z)
=2\log\frac{\rho(4\pi-\rho)h(z)^2}
{(2\pi-\rho)^2K(z)},\ \ \frac{\partial^2}{\partial \rho^2} L(\rho,z)>0.
$$
Thus the function $L(\cdot,z)$ has a unique minimum at
$$
\rho_z=2\pi\left(
1-\frac{h(z)}{\sqrt{K(z)+h(z)^2}}
\right).
$$
Therefore, by the choice of $p$, we have
\begin{equation}\label{eq:joint-optimal-blow-up-level}
	L(\rho,z)\geq L(\rho_z,z)=4\pi-8\pi\log\Phi(z)
	\geq4\pi-8\pi\log\Phi(p)
\end{equation}
for every $0<\rho<2\pi$ and $z\in\partial M$.
Then \eqref{eq:joint-fixed-mass-threshold}  gives
$$
m\geq4\pi-8\pi\log\Phi(p),
$$
 which is a contradiction to \eqref{eq:joint-strict-blow-up-level}.       Thus we proved the claim.

Consequently, there exists $u_*\in W^{1,2}(M)$ such that
\begin{equation}\label{eq:joint-limit-minimality}
	\widehat J(u_*,\rho_*)
	=\inf_{u\in W^{1,2}(M)}\widehat J(u,\rho_*)=m,
	\qquad 0<\rho_*<2\pi.
\end{equation}

Write $\gamma(s)=(x_1(s),x_2(s))$ in the above coordinates. Since
		$\gamma(0)=p$, $e_1=\dot{\gamma}(0)$, and $e_2=\nu(0)$, we have
		$$
		x_1(0)=x_2(0)=0,
		\qquad
		x_1'(0)=1,
		\qquad
		x_2'(0)=0.
		$$
		Moreover, geodesic normal coordinates satisfy
		$$
		g_{ij}(p)=\delta_{ij},
		\qquad
		\Gamma_{ij}^m(p)=0,
		$$ where $\Gamma_{ij}^m$ is the Christoffel symbols.
		
		Since
		$\dot{\gamma}=\sum_{i=1}^2x_i'\partial_{x_i}$, the local coordinate
		formula for the covariant derivative gives
		$$
		\left(\nabla_{\dot{\gamma}}\dot{\gamma}\right)^m
		=
		x_m''
		+
		\sum_{i,j=1}^2
		\Gamma_{ij}^m(\gamma(s))x_i'x_j'.
		$$
		Evaluating at $s=0$, all Christoffel symbols vanish, and hence
		$$
		\left(\nabla_{\dot{\gamma}}\dot{\gamma}\right)^m(p)
		=
		x_m''(0).
		$$
		Since
		$\nabla_{\dot{\gamma}}\dot{\gamma}(p)=\kappa(0)e_2$, it follows that
		$$
		x_1''(0)=0,
		\qquad
		x_2''(0)=\kappa(0).
		$$
		
		Metric compatibility and
		$\langle\dot{\gamma},\nu\rangle_g=0$ imply
		$\nabla_{\dot{\gamma}}\nu=-\kappa\dot{\gamma}$. Therefore,
		$$
		\nabla_{\dot{\gamma}}
		\left(\nabla_{\dot{\gamma}}\dot{\gamma}\right)
		=
		\kappa'\nu-\kappa^2\dot{\gamma}.
		$$
		At $s=0$, its two coordinate components are
		$-\kappa(0)^2$ and $\kappa'(0)$.
		
		On the other hand, differentiating the coordinate expression for
		$\nabla_{\dot{\gamma}}\dot{\gamma}$ and using
		$x_1'(0)=1$, $x_2'(0)=0$, and $\Gamma_{ij}^m(p)=0$, we obtain
		$$
		\left[
		\nabla_{\dot{\gamma}}
		\left(\nabla_{\dot{\gamma}}\dot{\gamma}\right)
		\right]^m(p)
		=
		x_m'''(0)
		+
		\partial_1\Gamma_{11}^m(p).
		$$
		The coordinate curve $t\mapsto(t,0)$ is a geodesic, so its geodesic
		equation gives $\Gamma_{11}^m(t,0)=0$. Consequently,
		$\partial_1\Gamma_{11}^m(p)=0$, and hence
		$$
		x_1'''(0)=-\kappa(0)^2,
		\qquad
		x_2'''(0)=\kappa'(0).
		$$
		
		Since $x_1'(0)=1$, the inverse function theorem allows us to write
		$x_2=f(x_1)$. Differentiating
		$x_2(s)=f(x_1(s))$ at $s=0$ gives
		$$
		f(0)=f'(0)=0,
		\qquad
		f''(0)=\kappa(0),
		\qquad
		f'''(0)=\kappa'(0).
		$$
		The conclusion follows from Taylor's theorem.
