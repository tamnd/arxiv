\label{def:decay-function}
    A non-increasing function \(F\colon \intervalco{0,\infty} \to \intervaloo{0,\infty}\) is called a \emph{decay function} on \(\varGamma\) if
    \begin{align}
        \norm{F}
        &\coloneq
        \sup_{x\in \varGamma} \sum_{y \in \varGamma} F\pdist{x,y}
        <
        \infty
        \label{eq:F-norm}
    \shortintertext{and}
        C_F
        &\coloneq
        \sup_{x,y\in \varGamma} \sum_{z\in\varGamma} \frac{F\pdist{x,z} \, F\pdist{z,y}}{F\pdist{x,y}}
        <
        \infty
        \label{eq:convolution-constant}
        .
    \end{align}

\label{def:interaction}
    An \emph{interaction} is a map
    \begin{equation*}
        \varPhi\colon \mathcal{P}_0(\varGamma)\times I \to \mathcal{L}(\mathcal{A}), \ (Z,t) \mapsto \varPhi(Z,t)
        ,
    \end{equation*}
    such that the following conditions are satisfied:
    \begin{thmlist}
        \item
            \label{item:def-interaction:localized}
            \(\varPhi(Z,t)\in \mathcal{B}_Z\).
        \item
            \label{item:def-interaction:dissipative}
            \(\varPhi(Z,t)\) is dissipative 
            \footnote{ 
                An operator \(K\in \mathcal{L}(\mathcal{A})\) is called dissipative if for all \(\lambda>0\) and all \(A\in \mathcal{A}\) it holds that\\ \(\norm{(\lambda \, \operatorname{id} - K) \, A} \geq \lambda \, \norm{A}\).
                This is equivalent to \(\intervalco{0,\infty} \to \mathcal{L}(\mathcal{A}), \, t \mapsto {\mathrm{e}}^{tK}\) being a contraction semigroup by the Lumer-Phillips Theorem.
            } 
            .
        \item
            \label{item:def-interaction:continuous}
            For all \(Z\in \mathcal{P}_0(\varGamma)\) the map \(I \to \mathcal{L}(\mathcal{A}),\ t \mapsto \varPhi(Z,t)\) is norm-continuous and uniformly bounded.
    \end{thmlist}

    \noindent
    We call \(\varPhi\) \emph{finitely supported} if \(\sup_{t\in I} \, \norm{\varPhi(Z,t)} = 0\) for all but finitely many \(Z\in \mathcal{P}_0(\varGamma)\).
    We say it has \emph{finite range}~\(R\geq 0\) if \(\sup_{t\in I} \, \norm{\varPhi(Z,t)} = 0\) unless \(\diam{Z} \leq R\), and it is a \emph{\(k\)-body} interaction for some \(k\in\mathbb{N}_{>0}\) if \(\sup_{t\in I} \, \norm{\varPhi(Z,t)} = 0\) unless \(\abs{Z} \leq k\).

<!-- remark-2: body_md -->
In the usual description of open quantum systems with Markovian dynamics, the interaction \(\varPhi\) corresponds to a time-dependent dual Lindblad generator in the Heisenberg picture.
    The most relevant class of interactions satisfying our assumptions are those whose local terms \(\varPhi(Z,t)\) are given as local Lindbladians in GKSL form
    \begin{equation*}
        \varPhi(Z,t) \, A
        =
        \sum_{j\in J_Z} \mathrm{i} \, \commutator{H_j(Z,t), A} + L_j(Z,t)^* \, A \, L_j(Z,t) - \tfrac{1}{2} \, \acomm[\big]{L_j(Z,t)^* \, L_j(Z,t), A}
        ,
    \end{equation*}
    where \(J_Z\) is a finite index set.
    If the Hamiltonian terms \(H_j(Z,t) \in \mathcal{A}\) and jump operators \(L_j(Z,t)\in \mathcal{A}\) are bounded operators supported in \(Z\), property~ 
    \begingroup                           
    \let\@gobble\@gobble\ref{item:def-interaction:localized} 
    \endgroup
 is automatically fulfilled.
    Furthermore, operators in the above GKSL form with \(H_j(Z,t)\) self-adjoint are always completely dissipative, which implies property~ 
    \begingroup                           
    \let\@gobble\@gobble\ref{item:def-interaction:dissipative} 
    \endgroup
 (cf.~\cite[Theorem~1]{NVZ2011}).
    Property~ 
    \begingroup

            \(\tau_{s,s} = \operatorname{id}\) for all \(s \in I\),

\label{lem:bounding-sum-over-n-long-paths}
    Let \(F\) be a decay function on \(\varGamma\), and let \(\varPhi\) be an interaction with \(F\)-decay.
    Fix \((s,t)\in \varDelta^2_I\).

    \begin{align}
        \label{eq:lem-bounding-sum-over-n-long-paths-from-X-to-Y}
        
    \int_{\mathrlap{#1}}^{\mathrlap{#2}}
_{\varDelta^n_{s,t}} \odif{u}
        \, \ifstrempty{}{\sum_{\substack{\gamma\in\mathrm{P}_n(X,Y)}}}{\smashoperator[]{\sum_{\substack{\gamma\in\mathrm{P}_n(X,Y)}}}}

        \, \norm{\varPhi(\gamma_j,u_j)}
        &\leq
        \sum_{x\in X} \, \sum_{y\in Y} \, F\pdist{x,y}
        \, \frac{C_F^{n-1} \, I_{\varPhi}(s,t)^n}{n!}
    \shortintertext{and}
        \label{eq:lem-bounding-sum-over-n-long-paths-attached-to-X}

_{\varDelta^n_{s,t}} \odif{u}

        \, \prod_{j=1}^n
        \, \norm{\varPhi(\gamma_j,u_j)}
        &\leq
        \abs{X} \, \norm{F}
        \, \frac{C_F^{n-1} \, I_{\varPhi}(s,t)^n}{n!}
        .

For the proof, let
    \begin{equation*}
        K(t)
        \coloneq
        \ifstrempty{lr}{\sum_{\substack{Z\in \mathcal{P}_0(\varGamma)}}}{\smashoperator[lr]{\sum_{\substack{Z\in \mathcal{P}_0(\varGamma)}}}} \, \varPhi(Z,t)
        .
    \end{equation*}
    Since \(\varPhi\) is finitely supported and each map \(t\mapsto\varPhi(Z,t)\) is norm-continuous and uniformly bounded, the same holds for \(t\mapsto K(t)\).
    Moreover, \(K(t)\) is dissipative for all~\(t\in I\), since the sum of two bounded dissipative operators is again dissipative, by Lumer-Phillips and the Trotter product formula.
    Then, for all \((s,t)\in \varDelta^2_I\) we have
    \begin{equation*}
        \tau_{s,t}
        =
        \sum_{n=0}^\infty
        
    \int_{\mathrlap{#1}}^{\mathrlap{#2}}
_{\varDelta^n_{s,t}} \odif{u}
        \, K(u_1) \, \dotsm \, K(u_n)
        ,
    \end{equation*}
    which is bounded by
    \begin{equation}
        \label{eq:proof-prop-finite-evolutions-series-bound}
        \norm{\tau_{s,t}}
        \leq
        \sum_{n=0}^\infty
        \frac{\abs{t-s}^n}{n!}
        \, \paren[\Big]{
            \;\sup_{u\in \intervalcc{s,t}} \, \norm{K(u)}
        }^n
        .
    \end{equation}

    Statement~ 
    \begingroup                           
    \let\@gobble\@gobble\ref{item:prop-finite-evolutions:cocycle} 
    \endgroup
 can be shown by writing out the series on both sides of the equation and using that for \(N\in \mathbb{N}_0\) and \(t \in \intervalcc{s,u}\) the simplex \(\varDelta_{s,u}^N\) can be decomposed into smaller simplices with sizes \(\abs{t-s}\) and \(\abs{u-t}\) as
    \begin{equation*}
        \varDelta_{s,u}^N
        =
        \bigcup_{n=0}^N \varDelta_{s,t}^n \times \varDelta_{t,u}^{N-n}
        ,
    \end{equation*}
    with the union being disjoint up to sets of measure zero.
    Consequently,
    \begin{align*}
        \hspace*{2em}&\hspace*{-2em}
        
    \int_{\mathrlap{#1}}^{\mathrlap{#2}}
_{\varDelta^N_{s,u}} \odif{v}
        \, K(v_1)  \dotsm  K(v_N)
        =
        \sum_{n=0}^N
        \paren[\bigg]{
            
    \int_{\mathrlap{#1}}^{\mathrlap{#2}}
_{\varDelta^n_{s,t}} \odif{v}
            \, K(v_1)  \dotsm  K(v_n)
        }
        \,
        \paren[\bigg]{
            
    \int_{\mathrlap{#1}}^{\mathrlap{#2}}
_{\varDelta^{N-n}_{t,u}} \odif{w}
            \, K(w_1)  \dotsm  K(w_{N-n})
        }
        ,
    \end{align*}
    which, together with a change in summation index, implies the statement. 
    Statement~ 
    \begingroup                           
    \let\@gobble\@gobble\ref{item:prop-finite-evolutions:identity} 
    \endgroup
 follows immediately from the definition.
    For~ 
    \begingroup                           
    \let\@gobble\@gobble\ref{item:prop-finite-evolutions:continuity} 
    \endgroup
, observe that the integrand
    \begin{equation*}
        K(u_1) \, \dotsm \, K(u_n)
    \end{equation*}
    is continuous in \(u_1,\dotsc,u_n\).
    Hence, the integral
    \begin{equation*}
        
    \int_{\mathrlap{#1}}^{\mathrlap{#2}}
_{\varDelta^n_{s,t}} \odif{u}
        \, K(u_1) \, \dotsm \, K(u_n)
    \end{equation*}
    is also continuous in both~\(s\) and~\(t\).
    With the bound~\eqref{eq:proof-prop-finite-evolutions-series-bound} and dominated convergence, one can show that \(\tau_{s,t}\) is also continuous in \(s\) and \(t\).
    For statement~ 
    \begingroup                           
    \let\@gobble\@gobble\ref{item:prop-finite-evolutions:derivative-second} 
    \endgroup
, we first observe that we can reorder the integrals into
    \begin{equation*}
        
    \int_{\mathrlap{#1}}^{\mathrlap{#2}}
_{\varDelta^n_{s,t}} \odif{u}
        \, K(u_1) \, \dotsm \, K(u_n)
        =
        
    \int_{\mathrlap{#1}}^{\mathrlap{#2}}
_s^t \odif{u_1} 
    \int_{\mathrlap{#1}}^{\mathrlap{#2}}
_s^{u_1} \odif{u_2} \, \dotsi 
    \int_{\mathrlap{#1}}^{\mathrlap{#2}}
_{s}^{u_{n-1}} \odif{u_n} \, K(u_n) \, \dotsm \, K(u_1)
        .
    \end{equation*}
    This simplifies taking the derivative
    \begin{align*}
        \hspace*{2em}&\hspace*{-2em}
        \sum_{n=0}^\infty \partial_t 
    \int_{\mathrlap{#1}}^{\mathrlap{#2}}
_s^t \odif{u_1} 
    \int_{\mathrlap{#1}}^{\mathrlap{#2}}
_s^{u_1} \odif{u_2} \, \dotsi 
    \int_{\mathrlap{#1}}^{\mathrlap{#2}}
_{s}^{u_{n-1}} \odif{u_n} \, K(u_n) \, \dotsm \, K(u_1)
        \\&=
        \sum_{n=1}^\infty 
    \int_{\mathrlap{#1}}^{\mathrlap{#2}}
_s^{t} \odif{u_2} \, \dotsi 
    \int_{\mathrlap{#1}}^{\mathrlap{#2}}
_{s}^{u_{n-1}} \odif{u_n} \, K(u_n) \, \dotsm \, K(u_2) \, K(t)
        \\&=
        \tau_{s,t} \, K(t)
        .
    \end{align*}
    It only remains to exchange the sum and derivative, which is justified because for every compact interval \(J=\intervalcc{s,b} \subseteq I\) one has \(\partial_t \, \operatorname{id}_{\mathcal{A}} = 0\) and
    \begin{equation*}
        \sup_{t\in J}
        \, \norm[\bigg]{
            \partial_t 
    \int_{\mathrlap{#1}}^{\mathrlap{#2}}
_s^t \odif{u_1} 
    \int_{\mathrlap{#1}}^{\mathrlap{#2}}
_s^{u_1} \odif{u_2} \, \dotsi 
    \int_{\mathrlap{#1}}^{\mathrlap{#2}}
_{s}^{u_{n-1}} \odif{u_n} \, K(u_n) \, \dotsm \, K(u_1)
        }
        \leq
        \sup_{t\in J} \, \frac{\abs{t-s}^{n-1}}{(n-1)!} \, \paren[\Big]{
            \, \sup_{u\in J} \norm{K(u)}
        }^n
    \end{equation*}
    for \(n \geq 1\), and this bound is summable.
    Statement~ 
    \begingroup                           
    \let\@gobble\@gobble\ref{item:prop-finite-evolutions:derivative-first} 
    \endgroup
 follows in exactly the same way without reordering the integrals.
    To prove the final statement~ 
    \begingroup                           
    \let\@gobble\@gobble\ref{item:prop-finite-evolutions:contractive} 
    \endgroup
, set \(u^l_k \coloneq s + k \, \frac{t-s}{l}\) for \(l\in \mathbb{N}_{>0}\) and \(k\in \List{0,1,\dotsc,l}\) and approximate the map \(\intervalcc{s,t} \to \mathcal{L}(\mathcal{A}), \, u \mapsto K(u)\) by a sequence of piecewise constant functions \((K^l)_{l\in \mathbb{N}_{>0}}\)
    \begin{equation*}
        K^l\colon \intervalcc{s,t} \to \mathcal{L}(\mathcal{A})
        ,\
        u \mapsto K\paren[\big]{\max \, \Set{u^l_k  
         
    \nonscript\,\delimsize\vert
    \allowbreak
    \nonscript\,
    \mathopen{} u^l_k \leq u}}
        .
    \end{equation*}
    For \((u,v)\in \varDelta^2_{s,t}\) we define the time-ordered exponential of \(K^l\) as
    \begin{equation*}
        \tau^l_{u,v}
        \coloneq
        \sum_{n=0}^\infty
        
    \int_{\mathrlap{#1}}^{\mathrlap{#2}}
_{\varDelta^n_{u,v}} \odif{w}
        \, K^l(w_1) \, \dotsm \, K^l(w_n)
        .
    \end{equation*}
    We note that by the same argument used to prove statement~ 
    \begingroup                           
    \let\@gobble\@gobble\ref{item:prop-finite-evolutions:cocycle} 
    \endgroup
 it holds for all \((u,v,w) \in \varDelta^3_{s,t}\) that \(\tau^l_{u,v} \, \tau^l_{v,w} = \tau^l_{u,w}\).
    Moreover, by continuity of \(K\) and dominated convergence it holds that \(\tau^l_{u,v} \to \tau_{u,v}\) as \(l\to \infty\).
    Hence, we obtain
    \begin{equation*}
        \norm{\tau_{s,t}}
        =
        \lim_{l\to \infty} \norm{\tau^l_{s,t}}
        \leq
        \limsup_{l\to \infty} \, \norm{\tau^l_{u^l_0, u^l_1}} \, \norm{\tau^l_{u^l_1, u^l_2}} \, \dotsm \, \norm{\tau^l_{u^l_{l-1}, u^l_l}}
        .
    \end{equation*}
    Since the function \(K^l\) is constant on the intervals \(\intervalco{u^l_k,u^l_{k+1}}\) with \(k\in \List{0,\dotsc,l-1}\), the time-ordered exponential \(\tau^l_{u^l_k,u^l_{k+1}}\) reduces to an ordinary exponential
    \begin{equation*}
        \tau^l_{u^l_k,u^l_{k+1}}
        =
        \Exp[\Big]{\frac{t-s}{l} \, K(u^l_k)}
        .
    \end{equation*}
    As we have observed previously, \(K(u)\) is a bounded dissipative operator for all \(u\in I\).
    Then by the Lumer-Phillips Theorem, it generates a contractive semigroup and the norm of the above exponential is bounded by \(1\).
    Therefore, we can conclude
    \begin{equation*}
        \norm{\tau_{s,t}}
        \leq
        \limsup_{l\to \infty}
        \, \norm[\Big]{\Exp[\Big]{\frac{t-s}{l} \, K(u^l_0)}}
        \, \dotsm
        \, \norm[\Big]{\Exp[\Big]{\frac{t-s}{l} \, K(u^l_{l-1})}}
        \leq
        1
        .
        \qedhere
    \end{equation*}
