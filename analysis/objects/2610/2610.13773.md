<!-- definition-1: body_md -->
\label{def:ca}
$x$ is \emph{$\epsilon$-accurate Pareto stationary} if
$\min_{w\in\Delta_K}\|\nabla F(x)w\|_*\le\epsilon$; $w^\star(x)$ denotes a minimizer
and $d^\star(x):=\nabla F(x)w^\star(x)$. For an iterate--weight pair $(x_t,w_t)$,
the \emph{CA distance} is $\mathrm{CA}_t:=\|\nabla F(x_t)w_t-d^\star(x_t)\|_*$ and the
\emph{inner gap} is
$\mathrm{gap}_t:=\|\nabla F(x_t)w_t\|_*-\min_{w\in\Delta_K}\|\nabla F(x_t)w\|_*\ge0$. By the
reverse triangle inequality $\mathrm{gap}_t\le\mathrm{CA}_t$ for every selection
$w^\star(x_t)$; $\mathrm{gap}_t=0$ iff $w_t$ solves the inner problem; and,
unlike $\mathrm{CA}_t$, the gap involves no selection.
<!-- end definition-1 -->
<!-- proposition-1: body_md -->
\label{prop:unique}
If the dual unit ball is strictly convex, $d^\star(x)$ is the same for every
minimizer $w^\star(x)$, for every instance. If it is not, there is an instance
with $K=2$ in which every $w\in\Delta_K$ is optimal and $d^\star$ ranges over a
whole segment: take $\nabla f_1=a$, $\nabla f_2=b$ for distinct unit vectors
$a,b$ with $\|(a+b)/2\|_*=1$ (for $\ell_1$: $a=(1,0)$, $b=(0,1)$).
<!-- end proposition-1 -->
<!-- proposition-2: body_md -->
\label{prop:scalarization}
If $\bar x$ is a first-order stationary point of $x\mapsto F(x)u$ for a fixed
$u\in\Delta_K$, i.e. $\nabla F(\bar x)u=0$, then $\min_{w\in\Delta_K}\|\nabla F(\bar x)w\|_*=0$,
attained at $w=u$.
<!-- end proposition-2 -->
<!-- head-1: body_md -->
\begin{assumption}[Smoothness in the norm pair]\label{as:smooth}
Each $f_i$ is differentiable and $\|\nabla f_i(x)-\nabla f_i(y)\|_*\le L\|x-y\|$.
\end{assumption}
\begin{assumption}[Lower bound]\label{as:lb}
$f_i^\star:=\inf_xf_i(x)>-\infty$; $\Delta:=\max_i\big(f_i(x_0)-f_i^\star\big)>0$
and $L>0$ (if $\Delta=0$ or $L=0$ every gradient vanishes at $x_0$ and there is
nothing to prove; excluding these keeps $M>0$ and the step sizes defined).
\end{assumption}
\begin{assumption}[Stochastic oracle]\label{as:noise}
With $\mathcal F_t$ the history up to the computation of $x_t,w_t$, the sample
$\xi_t$ is independent of $\mathcal F_t$ and
$\mathbb{E}[\nabla f_i(x_t;\xi_t)\mid\mathcal F_t]=\nabla f_i(x_t)$,
$\mathbb{E}[\|\nabla f_i(x_t;\xi_t)-\nabla f_i(x_t)\|_2^2\mid\mathcal F_t]\le\sigma^2$
for every $i$ (a shared minibatch across tasks is allowed).
\end{assumption}
\begin{assumption}[Gradient bound; stochastic setting only]\label{as:gradbound}
$\max_i\|\nabla f_i(x_t)\|_*\le M$ a.s.\ along the trajectory of Algorithm~\ref{alg:stoch}.
\end{assumption}

norm on $m\times n$ matrices), as
in \citet{pethick2025scion}. In the deterministic analysis $M$ is derived, not
assumed (Lemma~\ref{lem:M}).

<!-- lemma-1: body_md -->
\label{lem:onestep}
Under Assumption~\ref{as:smooth}, for every $w\in\Delta_K$ and every $t$,
\begin{equation}\label{eq:onestep}
F(x_{t+1})w\;\le\;F(x_t)w\;-\;\gamma R\,\|G_t\|_*\;+\;\gamma\langle g_t, w-w_t\rangle\;+\;\tfrac12L\gamma^2R^2 .
\end{equation}
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{lem:md}
Line~6 of Algorithm~\ref{alg:main} satisfies, for every $u\in\Delta_K^\kappa$,
with $\rho=\eta R\tau$ and $r(w):=\operatorname{KL}(w\|u_K)$,
\begin{equation}\label{eq:md}
\eta\langle g_t, u-w_t\rangle\;\le\;\operatorname{KL}(u\|w_t)-(1+\rho)\operatorname{KL}(u\|w_{t+1})
+\rho\big(r(u)-r(w_{t+1})\big)+\tfrac12\eta^2\|g_t\|_\infty^2 .
\end{equation}
For $u=u_K$ the regularization terms are $\le0$; for general $u$ they are
$\le\rho\ln K$; at $\tau=\kappa=0$ this is the standard bound for every

<!-- theorem-1: body_md -->
\label{thm:det}
Under Assumptions~\ref{as:smooth}--\ref{as:lb}, run Algorithm~\ref{alg:main}
with any $\tau\ge0$, $\kappa\in[0,1/K]$, and suppose
$\max_i\|\nabla f_i(x_t)\|_*\le M$ for $t<T$. Then for every $\gamma,\eta>0$,
\begin{equation}\label{eq:det-master}
\frac1T\sum_{t=0}^{T-1}\|\nabla F(x_t)w_t\|_*\;\le\;\frac{\Delta}{\gamma RT}+\frac{\eta RM^2}{2}+\frac{L\gamma R}{2},
\end{equation}
in which neither $\tau$, $\kappa$ nor $K$ appears. With the step sizes of
Lemma~\ref{lem:M},
\begin{equation}\label{eq:det}
\frac1T\sum_{t=0}^{T-1}\|\nabla F(x_t)w_t\|_*
\;\le\;\frac{\sqrt{2\Delta L}+M\sqrt{\log K/2}}{\sqrt T},
\end{equation}
so with $\hat t\in\operatorname*{arg\,min}_{t<T}\|\nabla F(x_t)w_t\|_*$ the output $x_{\hat t}$ is
$\epsilon$-accurate Pareto stationary,
$\min_w\|\nabla F(x_{\hat t})w\|_*\le\|\nabla F(x_{\hat t})w_{\hat t}\|_*\le\epsilon$,
after $T=O\big((\Delta L+M^2\log K)/\epsilon^2\big)$ iterations; for
$\tau=\kappa=0$, $M$ is supplied by Lemma~\ref{lem:M}, and substituting
$M^2=O(L\Delta\log K)$ gives $T=O(L\Delta\log^2K/\epsilon^2)$. No property of the
primal norm beyond Assumption~\ref{as:smooth} is used.

<!-- theorem-2: body_md -->
\label{thm:regret}
Let $A_t:\mathbb{R}^K\to\mathbb{R}^d$, $t<T$, be any sequence of linear maps with
$\max_i\|A_te_i\|_*\le M$, where $M>0$. Start from $w_0=u_K$ and set
$D_t=\operatorname{lmo}(A_tw_t)$, $g_t=A_t^\top D_t$, and
$w_{t+1,i}\propto w_{t,i}e^{\eta g_{t,i}}$ with
$\eta R=\sqrt{2\log K}/(M\sqrt T)$. Then, for every $u\in\Delta_K$,
\begin{equation}\label{eq:regret}
\frac1T\sum_{t=0}^{T-1}\|A_tw_t\|_*
\;\le\;\frac1T\sum_{t=0}^{T-1}\|A_tu\|_*
+M\sqrt{\frac{2\log K}{T}}.
\end{equation}
In particular, this holds for $A_t=\nabla F(x_t)$ under the conditions of
\eqref{eq:det} with $\tau=\kappa=0$.

<!-- corollary-1: body_md -->
\label{cor:gap}
Under the conditions of Theorem~\ref{thm:regret} with $A_t=\nabla F(x_t)$,
\begin{equation}\label{eq:gap}
\frac1T\sum_{t=0}^{T-1}\mathrm{gap}_t\;\le\;M\sqrt{\frac{2\log K}{T}}\;+\;V_T,
\qquad
V_T:=\min_{u\in\Delta_K}\frac1T\sum_{t=0}^{T-1}
\Big[\|\nabla F(x_t)u\|_*-\min_{w\in\Delta_K}\|\nabla F(x_t)w\|_*\Big],
\end{equation}
where the comparator-variation term $V_T\ge0$ is the shortfall of the best
\emph{fixed} weighting against the per-iterate optimum, a property of the
trajectory rather than of the update. This decomposition is informative when
$V_T$ is controlled. Along the trajectory covered by Theorem~\ref{thm:det},
\eqref{eq:gap-stationarity} already gives average-gap convergence at the
stationarity rate, without requiring $V_T\to0$; one may take the smaller of
the two bounds. In the Euclidean case $\mathrm{CA}_t^2\le2M\,\mathrm{gap}_t$, and under the uniform
convexity inequality \eqref{eq:S2} of Appendix~\ref{app:lp},
$\mathrm{CA}_t^q\le(q/\mu)M^{q-1}\mathrm{gap}_t$ (Remark~\ref{rem:hilbert}).

\label{thm:stoch}

\label{thm:iterwise}
Let Assumptions~\ref{as:smooth}--\ref{as:lb} hold and suppose
$\max_i\|\nabla f_i(x_t)\|_*\le M$ along the trajectory of
Algorithm~\ref{alg:main}. Fix a tracking accuracy $\epsilon\in(0,1]$ with
$\epsilon\le2M$ and a bias accuracy $\beta\in(0,\epsilon]$, and set
\[
\tau=\frac{\beta}{2\ln K},\qquad
\kappa=\frac{\beta}{4MK},\qquad
\eta R=\frac{\epsilon^{2}\beta}{768\,M^{4}\ln K},\qquad
\gamma R=\frac{\epsilon^{6}\beta^{6}}{27\cdot2^{28}\,(2+\Lambda)\,L\,M^{11}K\ln^{5}K},
\]
with $\Lambda:=\ln(4MK/\beta)$. Then, for any primal norm:
\begin{enumerate}[leftmargin=1.8em,itemsep=1pt]
\item[(i)] \textbf{Tracking.} For every
$t\ge T_0:=\big\lceil\tfrac{6144\,M^4\ln^2K}{\epsilon^2\beta^2}\ln\tfrac{24M^2\ln K}{\epsilon^2}\big\rceil$:
$\ \|w_t-p_t\|_1\le\epsilon/(2M)$ and
$\mathrm{gap}(p_t):=\|\nabla F(x_t)p_t\|_*-\min_w\|\nabla F(x_t)w\|_*\le\beta$, hence
$\mathrm{gap}_t\le\epsilon/2+\beta$. With $\beta=\epsilon/2$:
$\mathrm{gap}_t\le\epsilon$ for all $t\ge T_0=\tilde O(M^4\ln^2K/\epsilon^4)$.
\item[(ii)] \textbf{Progress.} For every $T$, \eqref{eq:det-master} holds, and
its right-hand side is at most $\epsilon/2$ for
$T\ge T_1:=\lceil4\Delta/(\gamma R\,\epsilon)\rceil$; with $\beta=\epsilon/2$,
$T_1=\tilde O(\Delta LM^{11}K/\epsilon^{13})$.
\item[(iii)] \textbf{Tail output.} For $T\ge\max\{2T_0,T_1\}$ and
$\hat t\in\operatorname*{arg\,min}_{T_0\le t<T}\|\nabla F(x_t)w_t\|_*$, the single iterate
$(x_{\hat t},w_{\hat t})$ satisfies both $\|\nabla F(x_{\hat t})w_{\hat t}\|_*\le\epsilon$,
hence $\min_w\|\nabla F(x_{\hat t})w\|_*\le\epsilon$, and $\mathrm{gap}_{\hat t}\le\epsilon/2+\beta$.
\end{enumerate}
If $\epsilon>2M$ all conclusions are trivial.

\label{lem:descent}
Under Assumption~\ref{as:smooth}, for all $x,y\in\mathbb{R}^d$ and each $i$,
\[
f_i(y)\;\le\;f_i(x)+\langle \nabla f_i(x), y-x\rangle+\tfrac{L}{2}\|y-x\|^2 .

<!-- proof-1: body_md -->
By the fundamental theorem of calculus along the segment,
\[
f_i(y)-f_i(x)-\langle \nabla f_i(x), y-x\rangle
=\int_0^1\langle \nabla f_i(x+t(y-x))-\nabla f_i(x), y-x\rangle\,dt .
\]

\[
\le\int_0^1\|\nabla f_i(x+t(y-x))-\nabla f_i(x)\|_*\|y-x\|\,dt
\le\int_0^1 Lt\|y-x\|^2dt=\tfrac{L}{2}\|y-x\|^2 . \qedhere
\]

<!-- lemma-7: body_md -->
\label{lem:gradbound}
Under Assumptions~\ref{as:smooth}--\ref{as:lb},
$\|\nabla f_i(x)\|_*^2\le 2L\big(f_i(x)-f_i^\star\big)$ for all $x$ and $i$.
<!-- end lemma-7 -->
<!-- proof-2: body_md -->
If $\nabla f_i(x)=0$ there is nothing to prove, so assume $\nabla f_i(x)\ne0$.
Since $\mathbb{R}^d$ is finite-dimensional the primal unit ball is compact, so the
supremum defining $\|\nabla f_i(x)\|_*$ is attained: there is $v$ with $\|v\|=1$
and $\langle \nabla f_i(x), v\rangle=-\|\nabla f_i(x)\|_*$ (negate a maximizer). For $s>0$,
Lemma~\ref{lem:descent} applied at $y=x+sv$ gives
\[
f_i^\star\;\le\;f_i(x+sv)\;\le\;f_i(x)-s\,\|\nabla f_i(x)\|_*+\tfrac{L}{2}s^2 .
\]
Minimizing the right-hand side over $s>0$ at $s=\|\nabla f_i(x)\|_*/L$ yields
$f_i^\star\le f_i(x)-\|\nabla f_i(x)\|_*^2/(2L)$, which rearranges to the claim.

<!-- lemma-8: body_md -->
\label{lem:subgrad}
Fix $x$ and let $\psi(w):=\|Aw\|_*$ on $\mathbb{R}^K$. For every $d\in\operatorname{lmo}(Aw)$ we have
$-A^\top d/R\in\partial\psi(w)$; componentwise, $-g/R\in\partial\psi(w)$ where
$g_i=\langle d, \nabla f_i(x)\rangle$.
<!-- end lemma-8 -->
<!-- proof-3: body_md -->
Because $B=-B$ is symmetric, the dual norm is the support function
$\|h\|_*=\tfrac1R\max_{v\in B}\langle h, v\rangle$, which is convex with
$\partial\|\cdot\|_*(h)=\tfrac1R\operatorname*{arg\,max}_{v\in B}\langle h, v\rangle
=-\tfrac1R\operatorname*{arg\,min}_{v\in B}\langle h, v\rangle=-\tfrac1R\operatorname{lmo}(h)$.
The chain rule for the composition of a convex function with a linear map gives
$A^\top\partial\|\cdot\|_*(Aw)\subseteq\partial\psi(w)$, whence the claim.
<!-- end proof-3 -->
<!-- proof-4: body_md -->
The set $S=\{Aw:w\in\Delta_K\}=\operatorname{conv}\{\nabla f_1(x),\dots,\nabla f_K(x)\}$ is
convex and compact. Let $m:=\min_{h\in S}\|h\|_*$ and suppose $g_1,g_2\in S$ both
attain $m$ with $g_1\ne g_2$. If $m=0$ then $g_1=g_2=0$, a contradiction; so
$m>0$. Then $g_1/m$ and $g_2/m$ are distinct points of the dual unit sphere, and
strict convexity of the dual unit ball gives
$\|\tfrac{g_1+g_2}{2m}\|_*<1$, i.e. $\|\tfrac{g_1+g_2}{2}\|_*<m$. But
$\tfrac{g_1+g_2}{2}\in S$ by convexity, contradicting minimality of $m$. Hence
the minimizing value $Aw^\star$ is unique.

For the converse, suppose the dual unit ball is not strictly convex: there
are distinct $a,b$ with $\|a\|_*=\|b\|_*=1$ and $\|m\|_*=1$ for $m:=(a+b)/2$. Let
$\varphi$ be a norming functional at $m$, i.e. $\varphi(m)=1$ and
$|\varphi(y)|\le\|y\|_*$ for all $y$ (it exists by compactness of the primal
unit ball, since $\|m\|_*=\max_{\|v\|\le1}\langle m, v\rangle$). Then $\varphi(a)\le1$,
$\varphi(b)\le1$ and $\tfrac12(\varphi(a)+\varphi(b))=\varphi(m)=1$ force
$\varphi(a)=\varphi(b)=1$, so for every $\theta\in[0,1]$,
\[
1=\varphi\big(\theta a+(1-\theta)b\big)\le\|\theta a+(1-\theta)b\|_*\le\theta\|a\|_*+(1-\theta)\|b\|_*=1 ,
\]
and the whole segment $[a,b]$ lies on the unit sphere. Take $K=2$ with
$\nabla f_1(x)=a$, $\nabla f_2(x)=b$: then $\|Aw\|_*=1$ for every $w\in\Delta_K$,
every $w$ is optimal, and $Aw^\star$ ranges over $[a,b]$. For $\|\cdot\|_*=\ell_1$
one may take $a=(1,0)$, $b=(0,1)$; for the nuclear norm, $a=E_{11}$,
$b=E_{22}$. Hence strict convexity of the dual ball is necessary and sufficient
for the optimal blended gradient to be unique in every instance.

<!-- proof-5: body_md -->
The function $x\mapsto F(x)u=\sum_i u_if_i(x)$ is differentiable with gradient
$\nabla F(x)u$, so first-order stationarity is $\nabla F(\bar x)u=0$ (a minimizer is a
special case), and then $\min_{w\in\Delta_K}\|\nabla F(\bar x)w\|_*\le\|\nabla F(\bar x)u\|_*=0$,

<!-- proposition-3: body_md -->
\label{prop:selection}
Fix $x$, let $w\in\Delta_K$, $G=\nabla F(x)w$, and let $D\in\operatorname{lmo}(G)$ with
$g_i=\langle \nabla f_i(x), D\rangle$. Then $(w,D)$ is a saddle point of $\Phi_x$ on
$\Delta_K\times B$ if and only if
\begin{equation}\label{eq:star}
\max_{i\in[K]}g_i \;=\; -R\,\|G\|_* .
\end{equation}
When \eqref{eq:star} holds, every objective improves by at least $R\|G\|_*$
along $D$, which by \eqref{eq:saddle} is the optimal guaranteed worst-case
improvement over $B$. If $\|\cdot\|_*$ is differentiable at $G$ then $\operatorname{lmo}(G)$ is a
singleton and \eqref{eq:star} holds automatically for any minimizer $w$ of
$\psi$; if not, \eqref{eq:star} can fail for some elements of $\operatorname{lmo}(G)$.
<!-- end proposition-3 -->
<!-- proof-6: body_md -->
$D\in\operatorname{lmo}(G)$ is exactly the statement that $D$ minimizes $\Phi_x(w,\cdot)$ over
$B$. The remaining saddle condition is that $w$ maximizes
$\Phi_x(\cdot,D)=\langle g, \cdot\rangle$ over $\Delta_K$, i.e. that $w$ is supported on
$\operatorname*{arg\,max}_i g_i$, i.e. that $\langle g, w\rangle=\max_i g_i$. Since
$\langle g, w\rangle=\langle \nabla F(x)w, D\rangle=\langle G, D\rangle=-R\|G\|_*$ by \eqref{eq:lmo}, this is
\eqref{eq:star}. Under \eqref{eq:star} we get $-g_i\ge R\|G\|_*$ for every $i$,
which is the stated improvement. If $\|\cdot\|_*$ is differentiable at $G$ then
$\partial\|\cdot\|_*(G)$ is a singleton, so by Lemma~\ref{lem:subgrad}
$\partial\psi(w)=\{-g/R\}$ and first-order optimality of $w$ on $\Delta_K$
coincides with \eqref{eq:star}.

For the failure, take the instance in the proof of
Proposition~\ref{prop:unique} with $w=e_1$, so $G=(1,0)$ and
$\operatorname{lmo}(G)=\{(-R,d_2):|d_2|\le R\}$, every element of which satisfies
$\langle G, v\rangle=-R=-R\|G\|_1$. Here $g=(-R,d_2)$ and \eqref{eq:star} reads
$\max(-R,d_2)=-R$, forcing $d_2=-R$. The selection $d_2=+R$ gives $g_2=R>0$:
the second objective is actively degraded, while the choice $D=(-R,-R)$
improves both by $R$.

<!-- proof-7: body_md -->
Fix $w\in\Delta_K$. Multiplying Lemma~\ref{lem:descent} (with $y=x_{t+1}$,
$x=x_t$) by $w_i\ge0$ and summing over $i$, using $\sum_iw_i=1$ for the
quadratic term,
\[
F(x_{t+1})w\;\le\;F(x_t)w+\langle \nabla F(x_t)w, x_{t+1}-x_t\rangle+\tfrac{L}{2}\|x_{t+1}-x_t\|^2 .
\]
By line~7 of Algorithm~\ref{alg:main}, $x_{t+1}-x_t=\gamma D_t$ with
$\|D_t\|\le R$, so the last term is at most $\tfrac12L\gamma^2R^2$. For the
inner product, decompose $\nabla F(x_t)w=G_t+\nabla F(x_t)(w-w_t)$ and use
\eqref{eq:lmo}:
\[
\langle \nabla F(x_t)w, D_t\rangle=\langle G_t, D_t\rangle+\sum_i(w_i-w_{t,i})\langle \nabla f_i(x_t), D_t\rangle
=-R\|G_t\|_*+\langle g_t, w-w_t\rangle. \qedhere
\]
<!-- end proof-7 -->
<!-- proof-8: body_md -->
Let $\Phi(w)=\sum_iw_i\ln w_i$ with $\nabla\Phi(w)=(1+\ln w_i)_i$, whose
Bregman divergence is $D_\Phi(u,v)=\operatorname{KL}(u\|v)$ for $u\in\Delta_K$ and $v$ in the
relative interior (with $0\ln0=0$). Write $C:=\Delta_K^\kappa$ and
\[
h(w):=-\eta\langle g_t, w\rangle+\rho\,r(w)+\operatorname{KL}(w\|w_t),\qquad \rho=\eta R\tau,
\]
so that $w_{t+1}=\operatorname*{arg\,min}_Ch$. Both $r=\operatorname{KL}(\cdot\|u_K)$ and $\operatorname{KL}(\cdot\|w_t)$
have Bregman divergence $\operatorname{KL}(u\|v)$ with respect to themselves, and the linear
term contributes nothing, so $h$ is $(1+\rho)$-strongly convex relative to
$\Phi$: for all $u\in\Delta_K$ and $v$ in the relative interior,
\begin{equation}\label{eq:h-relsc}
h(u)=h(v)+\langle \nabla h(v), u-v\rangle+(1+\rho)\operatorname{KL}(u\|v).
\end{equation}
The function $h$ is differentiable at $w_{t+1}$: if $\kappa>0$, every point of
$C$ has all coordinates at least $\kappa>0$, so $h$ is differentiable on all of
$C$, including at points with some $w_i=\kappa$ (where the clipped solution may
well lie); if $\kappa=0$, the minimizer has all coordinates positive because
$\partial_i\operatorname{KL}(w\|w_t)=\ln(w_i/w_{t,i})+1\to-\infty$ as $w_i\downarrow0$ while
$w_t$ has positive coordinates. Hence first-order optimality over the convex
set $C$ gives
$\langle \nabla h(w_{t+1}), u-w_{t+1}\rangle\ge0$ for all $u\in C$. Hence, by
\eqref{eq:h-relsc} at $v=w_{t+1}$,
\[
h(u)\ge h(w_{t+1})+(1+\rho)\operatorname{KL}(u\|w_{t+1})\qquad(u\in C).
\]
Expanding $h$ on both sides and rearranging,
\begin{equation}\label{eq:md1}
\eta\langle g_t, u-w_{t+1}\rangle\le\operatorname{KL}(u\|w_t)-(1+\rho)\operatorname{KL}(u\|w_{t+1})-\operatorname{KL}(w_{t+1}\|w_t)
+\rho\big(r(u)-r(w_{t+1})\big),
\end{equation}
with equality when $\kappa=0$, since then $C=\Delta_K$ and
$\nabla h(w_{t+1})$ is a multiple of $\mathbf 1$, orthogonal to $u-w_{t+1}$.
Next, split $\langle g_t, u-w_t\rangle=\langle g_t, u-w_{t+1}\rangle+\langle g_t, w_{t+1}-w_t\rangle$ and
bound the second term by H\"older, Young's inequality
$ab\le\tfrac12a^2+\tfrac12b^2$ and Pinsker's inequality
$\|p-q\|_1^2\le2\operatorname{KL}(p\|q)$:
\begin{equation}\label{eq:md2}
\eta\langle g_t, w_{t+1}-w_t\rangle\le\eta\|g_t\|_\infty\|w_{t+1}-w_t\|_1
\le\tfrac12\eta^2\|g_t\|_\infty^2+\tfrac12\|w_{t+1}-w_t\|_1^2
\le\tfrac12\eta^2\|g_t\|_\infty^2+\operatorname{KL}(w_{t+1}\|w_t).
\end{equation}
Adding \eqref{eq:md1} and \eqref{eq:md2}, the terms $\pm\operatorname{KL}(w_{t+1}\|w_t)$
cancel and we obtain \eqref{eq:md}. The claims about $u=u_K$ and general $u$
follow from $r(u_K)=0$, $r\ge0$ and $r\le\ln K$ on $\Delta_K$. At
$\tau=\kappa=0$ we have $\rho=0$ and $C=\Delta_K$, and \eqref{eq:md1} is the
three-point identity
$\eta\langle g_t, u-w_{t+1}\rangle=\operatorname{KL}(u\|w_t)-\operatorname{KL}(u\|w_{t+1})-\operatorname{KL}(w_{t+1}\|w_t)$ of
exponentiated gradient; the iterates then stay in the relative interior since
$w_0$ is uniform and the update multiplies by positive factors, so
\eqref{eq:md} holds for every $u\in\Delta_K$, including the vertices $u=e_i$
used in Lemma~\ref{lem:M}.

<!-- proof-9: body_md -->
Set $\bar\gamma:=\gamma R=\sqrt{2\Delta/(LT)}$ and
$\bar\eta:=\eta R=\sqrt{2\ln K}/(M\sqrt T)$, and define
\[
\mathcal C \;:=\; \frac{\bar\gamma\ln K}{\bar\eta}
\;+\;\frac{T\bar\gamma\bar\eta M^2}{2}\;+\;\frac{TL\bar\gamma^2}{2}.
\]
Substituting the step sizes,
\begin{equation}\label{eq:Ccalc}
\frac{\bar\gamma\ln K}{\bar\eta}=M\sqrt{\tfrac{\Delta\ln K}{L}},\qquad
\frac{T\bar\gamma\bar\eta M^2}{2}=M\sqrt{\tfrac{\Delta\ln K}{L}},\qquad
\frac{TL\bar\gamma^2}{2}=\Delta,
\end{equation}
so $\mathcal C=2M\sqrt{\Delta\ln K/L}+\Delta$.

$\|\nabla f_i(x_0)\|_*\le\sqrt{2L\Delta}$. Since $K\ge2$ we have
$4\ln K\ge4\ln2>2$, so $2\sqrt{L\Delta\ln K}\ge\sqrt{2L\Delta}$ and the
hypothesis on $M$ gives $\|\nabla f_i(x_0)\|_*\le M$.

\emph{Inductive step.} Suppose $\|\nabla f_i(x_s)\|_*\le M$ for all $i$ and all
$s\le t$. Then $\|g_s\|_\infty\le\max_i\|D_s\|\|\nabla f_i(x_s)\|_*\le RM$ for
$s\le t$ by H\"older. Apply Lemma~\ref{lem:onestep} with $w=e_i$ and add
$\gamma/\eta$ times \eqref{eq:md} with $u=e_i$; summing over $s=0,\dots,t$ and
using $\operatorname{KL}(e_i\|w_0)=\ln K$, $\operatorname{KL}(e_i\|w_{t+1})\ge0$ and $t+1\le T$,
\[
f_i(x_{t+1})-f_i(x_0)\;\le\;-\bar\gamma\sum_{s=0}^{t}\|G_s\|_*
+\frac{\bar\gamma\ln K}{\bar\eta}+\frac{T\bar\gamma\bar\eta M^2}{2}
+\frac{TL\bar\gamma^2}{2}\;\le\;\mathcal C,
\]
where we dropped the non-positive first sum. Hence
$f_i(x_{t+1})-f_i^\star\le\Delta+\mathcal C=2\Delta+2M\sqrt{\Delta\ln K/L}$, and
Lemma~\ref{lem:gradbound} gives
\[
\|\nabla f_i(x_{t+1})\|_*^2\;\le\;2L\Big(2\Delta+2M\sqrt{\tfrac{\Delta\ln K}{L}}\Big)
\;=\;4L\Delta+4M\sqrt{L\Delta\ln K}.
\]
It therefore suffices that $M^2\ge4L\Delta+4M\sqrt{L\Delta\ln K}$, i.e. that $M$
be at least the positive root of $M^2-4\sqrt{L\Delta\ln K}\,M-4L\Delta=0$, namely
\[
M\;\ge\;2\sqrt{L\Delta\ln K}+\sqrt{4L\Delta\ln K+4L\Delta}
\;=\;2\sqrt{L\Delta\ln K}+2\sqrt{L\Delta(\ln K+1)},
\]
which is the hypothesis. This closes the induction and shows
$M=O(\sqrt{L\Delta\ln K})$ suffices. Note that $M$ is defined by an explicit
formula in $L,\Delta,K$ only, so the step sizes are well defined and there is
no circularity.

<!-- remark-1: body_md -->
\label{rem:hilbert}
In a Hilbert dual norm the constant $2$ can be removed: the first-order
optimality condition for $w^\star$ gives
$\langle \nabla F w^\star, \nabla F w_t-\nabla F w^\star\rangle\ge0$, whence
$\|\nabla F w_t-\nabla F w^\star\|_*^2\le\|\nabla F w_t\|_*^2-\|\nabla F w^\star\|_*^2\le\|\nabla F w_t\|_*^2$.
This is the argument used in the Euclidean literature. It relies on the
parallelogram identity and is unavailable for a general dual norm, which is why
we state Corollary~\ref{cor:ca} with the triangle inequality instead. The same
identity gives the conversion used in Corollary~\ref{cor:gap}: writing
$a=\|\nabla F w_t\|_*$ and $b=\|\nabla F w^\star\|_*$,
\[
\mathrm{CA}_t^2\le a^2-b^2=(a+b)(a-b)\le2M\,\mathrm{gap}_t ,
\]
since $a,b\le M$ along the trajectory (Lemma~\ref{lem:M}) and
$a-b=\mathrm{gap}_t$. More generally, if the dual norm satisfies \eqref{eq:S2}
with power type $q$, then with $w^\star$ minimizing $\varphi_q(\nabla F w)$ over
$\Delta_K$ (the same minimizers as $\|\nabla F w\|_*$), first-order optimality and
\eqref{eq:S2} give $\tfrac{\mu}{q}\mathrm{CA}_t^q\le\varphi_q(\nabla F w_t)-\varphi_q(\nabla F w^\star)
=\tfrac1q(a^q-b^q)\le a^{q-1}(a-b)\le M^{q-1}\mathrm{gap}_t$, i.e.
$\mathrm{CA}_t^q\le(q/\mu)M^{q-1}\mathrm{gap}_t$.
The direct inequalities $\mathrm{gap}_t\le\|\nabla F(x_t)w_t\|_*$ and
$\mathrm{CA}_t\le2\|\nabla F(x_t)w_t\|_*$ imply average-gap and average-CA
convergence whenever the actual blended-gradient residual converges in
average. This includes a fixed scalarization on its own convergent trajectory.
Corollary~\ref{cor:gap} gives an additional comparator-dependent decomposition;
it is not needed to infer average-gap convergence from Theorem~\ref{thm:det}.

reduced to $\Theta(T^{-3/4}/M^2)$ in the stochastic setting.

\label{lem:closedform}
The minimizer in line~6 of Algorithm~\ref{alg:main} is given by
\eqref{eq:regeg}: $w_{t+1,i}=\max\{\kappa,c_i/Z_t\}$ with
$c_i=w_{t,i}^{1/(1+\rho)}e^{\eta g_{t,i}/(1+\rho)}$ and $Z_t>0$ the unique

<!-- proof-20: body_md -->
The objective $h(w)=-\eta\langle g_t, w\rangle+\rho r(w)+\operatorname{KL}(w\|w_t)$ is strictly convex
and differentiable on the relative interior, and the constraints are affine,
so the KKT conditions are necessary and sufficient. With multiplier $\nu$ for
$\sum_iw_i=1$ and $\lambda_i\ge0$ for $w_i\ge\kappa$, and using
$\partial_i r(w)=\ln w_i+\ln K+1$, $\partial_i\operatorname{KL}(w\|w_t)=\ln w_i-\ln w_{t,i}+1$,
stationarity reads
$-\eta g_{t,i}+(1+\rho)\ln w_i+\rho(\ln K+1)+1-\ln w_{t,i}-\nu-\lambda_i=0$,
i.e. $(1+\rho)\ln w_i=\ln w_{t,i}+\eta g_{t,i}+\text{const}+\lambda_i$. If
$\lambda_i=0$ this gives $w_i=c_i/Z_t$ for a constant $Z_t$; if $\lambda_i>0$
then $w_i=\kappa$, and $\lambda_i\ge0$ is equivalent to $c_i/Z_t\le\kappa$.
Hence $w_{t+1,i}=\max\{\kappa,c_i/Z_t\}$. The map $z\mapsto\sum_i\max\{\kappa,c_i/z\}$
is continuous, non-increasing, tends to $K\kappa<1$ as $z\to\infty$ and to
$\infty$ as $z\to0$, and is strictly decreasing wherever its value exceeds
$K\kappa$, so $Z_t$ is unique.

<!-- lemma-12: body_md -->
\label{lem:clip}
Let $\kappa\in[0,1/K)$ and $c\in\mathbb{R}^K_{>0}$, and let $Z^\star$ be the unique
normalizer of Lemma~\ref{lem:closedform}, with
$S^\star=\{i:c_i/Z^\star<\kappa\}$. Along Algorithm~\ref{alg:clip}:
\begin{enumerate}[leftmargin=1.8em]
\item[(i)] $S_0\subseteq S_1\subseteq\cdots\subseteq S^\star$, and $Z_0<Z_1<\cdots\le Z^\star$
while the recurrence continues;
\item[(ii)] the recurrence returns after at most $|S^\star|+1\le K$ rounds, in
the first round $k$ with $A_k=\emptyset$; at that round $S_k=S^\star$ and
$Z_k=Z^\star$, so the returned $w$ is $w_{t+1}$ of \eqref{eq:regeg}. (A
coordinate with $c_i=\kappa Z^\star$ exactly may sit in either set; the
returned vector is the same.)
\end{enumerate}
Each round costs $O(K)$, so the update costs $O(K\min\{K,|S^\star|+1\})$; sorting
$c$ once gives an $O(K\log K)$ implementation, since $S^\star$ is a lower set in
the order of $c$. At $\kappa=0$ the recurrence stops after one round with
$S_0=\emptyset$ and $Z_0=\sum_ic_i$, i.e. the plain softmax.
<!-- end lemma-12 -->
<!-- proof-21: body_md -->
Two facts about $Z(\cdot)$ are used. First, by the definition of $S^\star$ and
$\sum_i\max\{\kappa,c_i/Z^\star\}=1$,
\begin{equation}\label{eq:Zstar}
\sum_{i\notin S^\star}c_i=Z^\star\big(1-|S^\star|\kappa\big),\qquad
c_i<\kappa Z^\star\ \ (i\in S^\star),\qquad
c_i\ge\kappa Z^\star\ \ (i\notin S^\star).
\end{equation}
Second, for $S\subseteq S^\star$,
\[
\sum_{i\notin S}c_i=\sum_{i\notin S^\star}c_i+\sum_{i\in S^\star\setminus S}c_i
\le Z^\star(1-|S^\star|\kappa)+\kappa Z^\star\,(|S^\star|-|S|)=Z^\star(1-|S|\kappa),
\]
with strict inequality if $S\ne S^\star$; dividing by $1-|S|\kappa>0$ gives
\begin{equation}\label{eq:ZS}
Z(S)\le Z^\star\quad\text{for all }S\subseteq S^\star,\qquad\text{with equality iff }S=S^\star .
\end{equation}

\emph{(i) Monotonicity.} We show by induction that $S_k\subseteq S^\star$. This
holds for $S_0=\emptyset$. If $S_k\subseteq S^\star$ then $Z_k=Z(S_k)\le Z^\star$
by \eqref{eq:ZS}, so every $i\in A_k$ satisfies $c_i<\kappa Z_k\le\kappa Z^\star$,
i.e. $i\in S^\star$; hence $S_{k+1}\subseteq S^\star$. The sets are nested by
construction. For the normalizers, if $A_k\ne\emptyset$ then, using
$c_i<\kappa Z_k$ for $i\in A_k$,
\[
\sum_{i\notin S_{k+1}}c_i=Z_k(1-|S_k|\kappa)-\sum_{i\in A_k}c_i
>Z_k(1-|S_k|\kappa)-|A_k|\kappa Z_k=Z_k\big(1-|S_{k+1}|\kappa\big),
\]
and $1-|S_{k+1}|\kappa\ge1-|S^\star|\kappa>0$, so $Z_{k+1}>Z_k$.

\emph{(ii) Termination and correctness.} Since the $S_k$ are nested subsets of
$S^\star$ and strictly grow while $A_k\ne\emptyset$, the recurrence stops after
at most $|S^\star|+1$ rounds. At termination, $A_k=\emptyset$ means
$c_i\ge\kappa Z_k$ for all $i\notin S_k$, while every $i\in S_k$ was added at
some round $j<k$ with $c_i<\kappa Z_j\le\kappa Z_k$ by (i). Hence the returned
vector satisfies $w_i=\max\{\kappa,c_i/Z_k\}$ for every $i$, and it sums to $1$
by the definition of $Z(S_k)$. It is therefore the water-filling vector of
Lemma~\ref{lem:closedform} with normalizer $Z_k$; by the uniqueness of the
normalizer proved there, $Z_k=Z^\star$, and consequently $S_k=S^\star$ and
$w=w_{t+1}$.

For the complexity claim, note that $S^\star$ consists of the coordinates with
the smallest $c_i$: if $c_i\le c_j$ and $j\in S^\star$ then
$c_i/Z^\star\le c_j/Z^\star<\kappa$. After sorting, one can scan the candidates
$S=\{$the $s$ smallest$\}$ for $s=0,1,\dots$ and stop at the first $s$ for which
the $(s+1)$-st smallest $c$ is at least $\kappa Z(S)$; by \eqref{eq:ZS} and the
argument above this is $S^\star$.

<!-- remark-3: body_md -->
\label{rem:unsquared}
In Euclidean geometry, the squared inner objective
$\tfrac12\|\nabla F(x_t)w\|_2^2$ is smooth. The analyses of
\citet{zhang2025mgda} exploit this structure for iteration-wise CA distance.
Our oracle instead supplies a subgradient of the unsquared norm, matching the
linear descent term in Lemma~\ref{lem:onestep}. In Euclidean geometry, the
squared-objective gradient is $\|G_t\|_2$ times the unsquared gradient when
$G_t\ne0$; the latter involves normalization by $\|G_t\|_2$ and is
ill-conditioned near a zero blend.

For general dual norms, nonsmoothness can also occur at nonzero blends: the
$\ell_1$ norm is nonsmooth at vectors with zero coordinates, and the nuclear
norm at rank-deficient matrices. Squaring need not remove this nonsmoothness.
Our analysis therefore uses only convexity, Lipschitz continuity, and the
oracle subgradient identity. It incurs the slack $2\bar\eta^2M^2$ per step in
Lemma~\ref{lem:contract-slack}, a H\"older-$1/2$ sensitivity bound, and the
$1/\kappa$ factor in Lemma~\ref{lem:sens}. These bounds lead to the stated
burn-in and total-horizon exponents; sharper analysis under additional norm
smoothness assumptions remains open.

<!-- lemma-13: body_md -->
\label{lem:contract-slack}
Suppose $\max_i\|\nabla f_i(x_t)\|_*\le M$. For every $u\in\Delta_K^\kappa$,
\[
(1+\rho)\operatorname{KL}(u\|w_{t+1})\;\le\;\operatorname{KL}(u\|w_t)+\bar\eta\big[\tilde\psi_t(u)-\tilde\psi_t(w_{t+1})\big]+2\bar\eta^2M^2 .
\]
In particular, since $p_t$ minimizes $\tilde\psi_t$ over $\Delta_K^\kappa\ni w_{t+1}$,
\[
\operatorname{KL}(p_t\|w_{t+1})\;\le\;\frac{\Phi_t+2\bar\eta^2M^2}{1+\rho}.
\]
<!-- end lemma-13 -->
<!-- proof-22: body_md -->
Start from \eqref{eq:md1} in the proof of Lemma~\ref{lem:md}, written with
$\hat g_t=-g_t/R$, so that $\eta\langle g_t, u-w_{t+1}\rangle=-\bar\eta\langle \hat g_t, u-w_{t+1}\rangle$:
\[
(1+\rho)\operatorname{KL}(u\|w_{t+1})\le\operatorname{KL}(u\|w_t)-\operatorname{KL}(w_{t+1}\|w_t)
+\bar\eta\langle \hat g_t, u-w_{t+1}\rangle+\rho\big(r(u)-r(w_{t+1})\big).
\]
For the inner product, split $u-w_{t+1}=(u-w_t)+(w_t-w_{t+1})$ and use (F2):
the subgradient inequality gives $\langle \hat g_t, u-w_t\rangle\le f_t(u)-f_t(w_t)$, and
H\"older gives $\langle \hat g_t, w_t-w_{t+1}\rangle\le M\|w_{t+1}-w_t\|_1$. Then, by (F1),
$f_t(u)-f_t(w_t)=f_t(u)-f_t(w_{t+1})+\big(f_t(w_{t+1})-f_t(w_t)\big)\le f_t(u)-f_t(w_{t+1})+M\|w_{t+1}-w_t\|_1$.
Hence
\begin{align*}
\bar\eta\langle \hat g_t, u-w_{t+1}\rangle&\le\bar\eta\big(f_t(u)-f_t(w_{t+1})\big)+2\bar\eta M\|w_{t+1}-w_t\|_1\\
&\le\bar\eta\big(f_t(u)-f_t(w_{t+1})\big)+2\bar\eta^2M^2+\tfrac12\|w_{t+1}-w_t\|_1^2 ,
\end{align*}
by Young's inequality $2ab\le2a^2+\tfrac12b^2$ with $a=\bar\eta M$,
$b=\|w_{t+1}-w_t\|_1$. Pinsker's inequality bounds the last term by
$\operatorname{KL}(w_{t+1}\|w_t)$, which cancels the $-\operatorname{KL}(w_{t+1}\|w_t)$ above.
Substituting and using $\rho=\bar\eta\tau$,
\[
(1+\rho)\operatorname{KL}(u\|w_{t+1})\le\operatorname{KL}(u\|w_t)+\bar\eta\big[f_t(u)+\tau r(u)-f_t(w_{t+1})-\tau r(w_{t+1})\big]+2\bar\eta^2M^2 ,
\]
which is the claim. The second statement follows from
$\tilde\psi_t(p_t)\le\tilde\psi_t(w_{t+1})$, as $w_{t+1}\in\Delta_K^\kappa$ by
construction.

<!-- lemma-14: body_md -->
\label{lem:sens}
Let $p(x):=\operatorname*{arg\,min}_{\Delta_K^\kappa}\{\|\nabla F(x)w\|_*+\tau r(w)\}$. For any $x,x'$,
\[
\|p(x)-p(x')\|_1\;\le\;\sqrt{\frac{2L\|x-x'\|}{\tau}} ,
\]
and hence, along the trajectory,
\[
\chi^2(p_t\|p_{t+1}):=\sum_i\frac{(p_{t,i}-p_{t+1,i})^2}{p_{t+1,i}}\;\le\;\frac{2L\bar\gamma}{\tau\kappa}.
\]
<!-- end lemma-14 -->
<!-- proof-23: body_md -->
Write $p=p(x)$, $p'=p(x')$, $\tilde\psi_x=f_x+\tau r$ with $f_x=\|\nabla F(x)\cdot\|_*$.
Since $f_x$ is convex and finite, the first-order optimality condition for the
convex problem $\min_{\Delta_K^\kappa}\tilde\psi_x$ provides
$s\in\partial f_x(p)$ with $\langle s+\tau\nabla r(p), p'-p\rangle\ge0$. Convexity of
$f_x$ and the Bregman identity $r(p')=r(p)+\langle \nabla r(p), p'-p\rangle+\operatorname{KL}(p'\|p)$
give
\[
\tilde\psi_x(p')-\tilde\psi_x(p)\ge\langle s+\tau\nabla r(p), p'-p\rangle+\tau\operatorname{KL}(p'\|p)\ge\tau\operatorname{KL}(p'\|p)\ge\tfrac\tau2\|p'-p\|_1^2 ,
\]
the last step by Pinsker. Symmetrically
$\tilde\psi_{x'}(p)-\tilde\psi_{x'}(p')\ge\tfrac\tau2\|p-p'\|_1^2$. Adding and
cancelling the identical $\tau r$ terms,
\[
\tau\|p-p'\|_1^2\le\big[f_x(p')-f_{x'}(p')\big]-\big[f_x(p)-f_{x'}(p)\big]\le2\sup_{w\in\Delta_K}|f_x(w)-f_{x'}(w)|\le2L\|x-x'\|
\]
by (F3), which gives the first claim. For the second, $\|x_{t+1}-x_t\|\le\bar\gamma$,
and since $p_{t+1,i}\ge\kappa$,
$\chi^2(p_t\|p_{t+1})\le\kappa^{-1}\sum_i(p_{t,i}-p_{t+1,i})^2\le\kappa^{-1}\|p_t-p_{t+1}\|_1^2$.

<!-- lemma-15: body_md -->
\label{lem:klpert}
Let $p,p',q\in\Delta_K^\kappa$ and $\Lambda:=\ln(1/\kappa)$. Then
\[
\operatorname{KL}(p'\|q)\le\operatorname{KL}(p\|q)+\sqrt{(2+\Lambda)\operatorname{KL}(p'\|q)}\cdot\sqrt{\chi^2(p\|p')} .
\]
<!-- end lemma-15 -->
<!-- proof-24: body_md -->
Fix $q$ and let $F(\cdot):=\operatorname{KL}(\cdot\|q)$, which is convex with
$\nabla F(p')_i=\ln(p_i'/q_i)+1$. Convexity at $p'$ gives
$F(p')-F(p)\le\langle \nabla F(p'), p'-p\rangle=\sum_i\ln(p_i'/q_i)(p_i'-p_i)$, the
constant dropping because $\sum_i(p_i'-p_i)=0$. Cauchy--Schwarz with the
splitting $\sqrt{p_i'}\cdot1/\sqrt{p_i'}$ yields
\[
\sum_i\ln\frac{p_i'}{q_i}(p_i'-p_i)
\le\Big(\sum_ip_i'\ln^2\frac{p_i'}{q_i}\Big)^{1/2}
\Big(\sum_i\frac{(p_i'-p_i)^2}{p_i'}\Big)^{1/2}
=\sqrt{V(p'\|q)}\sqrt{\chi^2(p\|p')} .
\]
It remains to show $V(p'\|q)\le(2+\Lambda)\operatorname{KL}(p'\|q)$. Put $r_i:=p_i'/q_i$;
since $p',q\in\Delta_K^\kappa$ we have $r_i\in[\kappa,1/\kappa]$, so
$|\ln r_i|\le\Lambda$. Writing $\psi(r):=r\ln^2r$ and
$d(r):=r\ln r-r+1\ge0$, we have $V(p'\|q)=\sum_iq_i\psi(r_i)$ and, because
$\sum_iq_i(r_i-1)=0$, also $\operatorname{KL}(p'\|q)=\sum_iq_id(r_i)$. So it suffices that
$\psi(r)\le(2+\Lambda)d(r)$ whenever $|\ln r|\le\Lambda$. Substituting
$r=e^s$, this is
\begin{equation}\label{eq:scalar}
e^ss^2\le(2+\Lambda)\,(se^s-e^s+1)\qquad\text{for }|s|\le\Lambda ,
\end{equation}
where $se^s-e^s+1\ge0$ with equality only at $s=0$ (both sides of
\eqref{eq:scalar} vanish there). We prove the stronger statement
$e^ss^2\le(2+\max(s,0))(se^s-e^s+1)$ for all real $s$, which implies
\eqref{eq:scalar} since $\max(s,0)\le\Lambda$ on $[-\Lambda,\Lambda]$.

\emph{Case $s\ge0$.} Let $h(s):=(2+s)(se^s-e^s+1)-e^ss^2$. Dividing by $e^s>0$,
$e^{-s}h(s)=(2+s)(s-1+e^{-s})-s^2=s-2+(2+s)e^{-s}=:k(s)$.
Then $k(0)=0$ and $k'(s)=1-(1+s)e^{-s}\ge0$ because $1+s\le e^s$. Hence
$k\ge0$ on $[0,\infty)$, i.e. $h\ge0$.

\emph{Case $s\le0$.} Put $u:=-s\ge0$. The claim $e^ss^2\le2(se^s-e^s+1)$ reads
$e^{-u}u^2\le2\big(1-(1+u)e^{-u}\big)$, i.e. $u^2+2u+2\le2e^u$, which holds
because $2e^u=2+2u+u^2+\sum_{j\ge3}2u^j/j!\ge u^2+2u+2$.

Hence $e^ss^2\le(2+\max(s,0))(se^s-e^s+1)$ for all $s$, which gives
\eqref{eq:scalar} and therefore $V(p'\|q)\le(2+\Lambda)\operatorname{KL}(p'\|q)$.

\Phi_{t+1}\;\le\;\Big(1-\frac{\rho}{4}\Big)\Phi_t+4\bar\eta^2M^2+\frac{2(2+\Lambda)}{\rho}\,\chi^2(p_t\|p_{t+1}).

<!-- remark-4: body_md -->
\label{rem:M-reg}
Theorem~\ref{thm:iterwise} takes the trajectory gradient bound as a
hypothesis, and we keep it as one: unlike Theorem~\ref{thm:det} at
$\tau=\kappa=0$, we do not derive it. We record what a derivation would
involve. The induction of Lemma~\ref{lem:M} changes in the comparator: Lemma~\ref{lem:M} uses $u=e_i$, which is not in
$\Delta_K^\kappa$, so one applies \eqref{eq:md} at
$\hat e_i=(1-K\kappa)e_i+K\kappa u_K\in\Delta_K^\kappa$ and pays
$\gamma\|g_s\|_\infty\|e_i-\hat e_i\|_1\le2\bar\gamma MK\kappa$ per step for
the difference, together with the regularization term $\rho\ln K/\eta\cdot\gamma=\bar\gamma\tau\ln K$
per step from Lemma~\ref{lem:md}. Since $\tau\ln K+2MK\kappa=\beta\le\epsilon$
and $T_1\bar\gamma\le4\Delta/\epsilon+\bar\gamma$, over $T_1$ steps these add
at most $T_1\bar\gamma\,\beta\le4\Delta+\bar\gamma\epsilon$ to the
function-value increase, so $f_i(x_t)-f_i^\star\le5\Delta+C'$ with $C'$
collecting $\bar\gamma\epsilon$ and the (much smaller) remaining terms of
Lemma~\ref{lem:M} at these step sizes, and Lemma~\ref{lem:gradbound} converts
this into $\|\nabla f_i(x_t)\|_*\le\sqrt{2L(5\Delta+C')}$, which is below the
$M$ of Lemma~\ref{lem:M} ($\ge4\sqrt{L\Delta}$) once $C'\le3\Delta$. Verifying
$C'\le3\Delta$ requires relating the remaining terms, which involve $M$,
$\epsilon$ and $\beta$ but not $\Delta$, to $\Delta$ through the explicit value
of $M$ in Lemma~\ref{lem:M} and the horizon actually run, which is
$\max\{2T_0,T_1\}$ rather than $T_1$. We have not carried this out, so the
gradient bound remains an assumption of Theorem~\ref{thm:iterwise}.

<!-- proposition-4: body_md -->
\label{prop:lp}
Let $\|\cdot\|_*$ be the $\ell_p$ norm on $\mathbb{R}^m$ or the Schatten-$p$ norm on
$\mathbb{R}^{m\times n}$.
\begin{enumerate}[leftmargin=1.8em]
\item[(i)] For $1<p\le2$, \eqref{eq:S2} holds with $q=2$ and $\mu=p-1$: the
function $\psi_p:=\tfrac12\|\cdot\|_*^2$ is $(p-1)$-strongly convex with respect
to $\|\cdot\|_*$.
\item[(ii)] For $2\le p<\infty$, \eqref{eq:S2} holds with $q=p$ and
$\mu=2^{1-p}$.
\item[(iii)] For $p>2$, \eqref{eq:S2} fails with $q=2$ for every $\mu>0$: the
ratio $D_{\psi_p}(y,x)/\|y-x\|_*^2$ can be made arbitrarily small.
\end{enumerate}
For $p=1$ (the $\ell_1$ and nuclear norms) the unit ball is not strictly convex
and no inequality of the form \eqref{eq:S2} can hold.
<!-- end proposition-4 -->
<!-- proof-29: body_md -->
(i) Theorem~1 of \citet{ball1994sharp} (optimal $2$-uniform convexity) states
that for $1\le p\le2$ and all $X,Y$,
\begin{equation}\label{eq:bcl}
\Big(\frac{\|X+Y\|_*^p+\|X-Y\|_*^p}{2}\Big)^{2/p}\;\ge\;\|X\|_*^2+(p-1)\|Y\|_*^2 ,
\end{equation}
both for $\ell_p$ (their case (a), functions in $L_p$) and for Schatten-$p$
(their case (b), matrices in $C_p$). Put $X=\tfrac{a+b}{2}$ and
$Y=\tfrac{a-b}{2}$, so that $X+Y=a$ and $X-Y=b$. Because $2/p\ge1$, the map
$s\mapsto s^{2/p}$ is convex on $[0,\infty)$ and Jensen's inequality gives
$\big(\tfrac{\|a\|_*^p+\|b\|_*^p}{2}\big)^{2/p}\le\tfrac12\|a\|_*^2+\tfrac12\|b\|_*^2$.
Combining with \eqref{eq:bcl},
\[
\Big\|\frac{a+b}{2}\Big\|_*^2+(p-1)\Big\|\frac{a-b}{2}\Big\|_*^2\;\le\;\tfrac12\|a\|_*^2+\tfrac12\|b\|_*^2
\qquad\text{for all }a,b .
\]
Multiplying by $\tfrac12$ gives
$\psi_p(\tfrac{a+b}{2})\le\tfrac12\psi_p(a)+\tfrac12\psi_p(b)-\tfrac{p-1}{8}\|a-b\|_*^2$,
which is midpoint strong convexity with modulus $p-1$; for continuous functions
this upgrades to strong convexity by the usual dyadic-subdivision argument, and
strong convexity of $\psi_p$ is \eqref{eq:S2} with $q=2$, $\mu=p-1$.

(ii) Let $\phi:=\|\cdot\|_*^p$. Clarkson's inequality for
$p\ge2$ states that for all $y,y'$,
\begin{equation}\label{eq:clarkson}
\phi\Big(\frac{y+y'}{2}\Big)+\phi\Big(\frac{y-y'}{2}\Big)\le\frac{\phi(y)+\phi(y')}{2}.
\end{equation}
For $\ell_p$ this follows coordinatewise from the scalar case, which is
$X^{p/2}+Y^{p/2}\le(X+Y)^{p/2}\le\tfrac12(|a|^p+|b|^p)$ for
$X=(\tfrac{a+b}{2})^2$, $Y=(\tfrac{a-b}{2})^2$, $X+Y=\tfrac12(a^2+b^2)$, using
superadditivity of $s\mapsto s^{p/2}$ on $[0,\infty)$ (as $p/2\ge1$) and then
convexity of $s\mapsto s^{p/2}$; for Schatten norms \eqref{eq:clarkson} is
McCarthy's extension of Clarkson's inequality to trace classes
\citep{mccarthy1967cp}. Now let $m:=(y+y')/2$. Convexity of $\phi$ gives
$\langle \nabla\phi(y'), m-y'\rangle\le\phi(m)-\phi(y')$, and $y-y'=2(m-y')$, so
\begin{align*}
D_\phi(y,y')&:=\phi(y)-\phi(y')-\langle \nabla\phi(y'), y-y'\rangle
\ge\phi(y)-\phi(y')-2\big(\phi(m)-\phi(y')\big)\\
&=\phi(y)+\phi(y')-2\phi(m)\ge2\phi\Big(\frac{y-y'}{2}\Big)=2^{1-p}\|y-y'\|_*^p ,
\end{align*}
the last inequality being \eqref{eq:clarkson}. Dividing by $p$ gives
\eqref{eq:S2} for $\varphi_p=\phi/p$ with $q=p$ and $\mu=2^{1-p}$.


(iii) Take $x=e_1$ and $y=e_1+te_2$ with $t\downarrow0$ (for Schatten norms,
$x=E_{11}$, $y=E_{11}+tE_{22}$). Since
$\nabla\psi_p(g)=\|g\|_p^{2-p}\big(\operatorname{sign}(g_i)|g_i|^{p-1}\big)_i$ we have
$\nabla\psi_p(e_1)=e_1$, hence $\langle \nabla\psi_p(x), y-x\rangle=0$ and
$D_{\psi_p}(y,x)=\tfrac12(1+t^p)^{2/p}-\tfrac12=\tfrac1pt^p+O(t^{2p})$ while
$\|y-x\|_p^2=t^2$, so the ratio is $\tfrac1pt^{p-2}+O(t^{2p-2})\to0$ for $p>2$.
Finally, if the unit ball is not strictly convex there are $y\ne y'$ on the
unit sphere with $\|(y+y')/2\|_*=1$, and \eqref{eq:S2} applied at $y$ and $y'$
around the midpoint forces $\mu=0$.

Under Assumptions~\ref{as:smooth}--\ref{as:lb} and \eqref{eq:inexact}, with

\end{align*}

\emph{One-step inequality.} The proof of Lemma~\ref{lem:onestep} gives
$F(x_{t+1})w\le F(x_t)w+\gamma\langle A_tw, D_t\rangle+\tfrac12L\gamma^2\|D_t\|^2$; with
\eqref{eq:inexact}, $\langle A_tw, D_t\rangle=\langle G_t, D_t\rangle+\langle g_t, w-w_t\rangle\le-(1-\vartheta)R\|G_t\|_*+\langle g_t, w-w_t\rangle$
and $\|D_t\|^2\le(1+\vartheta)^2R^2$. Repeating the proof of
Theorem~\ref{thm:det} with $u=u_K$ and $\|g_t\|_\infty\le(1+\vartheta)RM$,
\[
(1-\vartheta)\frac1T\sum_t\|G_t\|_*\le\frac{\Delta}{\gamma RT}+(1+\vartheta)^2\frac{\eta RM^2}{2}+(1+\vartheta)^2\frac{L\gamma R}{2}
\le(1+\vartheta)^2\Big[\frac{\Delta}{\gamma RT}+\frac{\eta RM^2}{2}+\frac{L\gamma R}{2}\Big],
\]
and the step sizes give the first claim.

\emph{Regret.} Write $\psi_t(w)=\|A_tw\|_*$. For any $u\in\Delta_K$, H\"older
gives $\|A_tu\|_*\|D_t\|\ge-\langle A_tu, D_t\rangle=-\langle g_t, u\rangle$, hence
$\psi_t(u)\ge-\langle g_t, u\rangle/((1+\vartheta)R)$ (trivially if the right-hand side is
negative). For $w_t$, \eqref{eq:inexact} gives
$\psi_t(w_t)=\|G_t\|_*\le-\langle G_t, D_t\rangle/((1-\vartheta)R)=-\langle g_t, w_t\rangle/((1-\vartheta)R)$.
Subtracting, with $a:=1/((1-\vartheta)R)$ and $b:=1/((1+\vartheta)R)$,
\begin{align*}
\psi_t(w_t)-\psi_t(u)&\le-a\langle g_t, w_t\rangle+b\langle g_t, u\rangle
=a\langle g_t, u-w_t\rangle-(a-b)\langle g_t, u\rangle\\
&\le\frac{\langle g_t, u-w_t\rangle}{(1-\vartheta)R}+(a-b)(1+\vartheta)RM ,
\end{align*}
using $|\langle g_t, u\rangle|\le\|g_t\|_\infty\le(1+\vartheta)RM$. Since
$(a-b)(1+\vartheta)RM=M(1+\vartheta)\big[\tfrac1{1-\vartheta}-\tfrac1{1+\vartheta}\big]=\tfrac{2\vartheta M}{1-\vartheta}$,
summing and applying Lemma~\ref{lem:md} as in the proof of
Theorem~\ref{thm:regret}, with $\|g_t\|_\infty^2\le(1+\vartheta)^2R^2M^2$, gives
the second claim.
