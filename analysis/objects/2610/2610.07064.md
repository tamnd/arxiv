<!-- definition-1: body_md -->
Let $\{x_n\}$ be a sequence converging to $\bar{x}$. The sequence is said
to converge with order $q\geq 1$ if there exists a constant
$\mu>0$ such that
\[
\lim_{n\to\infty}
\frac{|e_{n+1}|}{|e_n|^q}
=
\mu.
\]
The constant $\mu$ is called the asymptotic error constant. The value of $q$ determines the speed of convergence. If $q=1$ and $0<\mu<1$, the convergence is \emph{linear}. If $q=2$, the convergence is \emph{quadratic}. If $q=3$, the convergence is \emph{cubic} (see \cite{Weerakoon_2000}).
<!-- end definition-1 -->
<!-- definition-2: body_md -->
Following the previous definition of convergence order, a matrix iterative method is said to have a  convergence order of $q$ if there exists a constant $C>0$ such that
\[
\|E_{n+1}\| \leq C\|E_n\|^q,
\]
for sufficiently large values of $n$. The value of $q$ illustrates the order of convergence. (see \cite{Higham_2008}).
<!-- end definition-2 -->
<!-- definition-3: body_md -->
The Moore-Penrose inverse or pseudoinverse $X$ of a rectangular matrix $A\in \mathbb{C}^{M\times N}$ is defined as the unique matrix satisfying the following Penrose equations \cite{Ben-Israel_2003}:
\begin{multicols}{4}
\begin{enumerate}[label=\roman*.]
    \item $AX A = A$, 
    \item $X A X=X$,
    \item $AX = (AX)^*$, 
    \item $X A=(X A)^*$.
\end{enumerate}
\end{multicols}
For rectangular matrices, the Moore-Penrose inverse is a natural extension of the classical inverse. The Moore-Penrose inverse of a matrix $A$  is denoted by $A^\dagger$.
<!-- end definition-3 -->
<!-- definition-4: body_md -->
Let $A \in \mathbb{C}^{N\times N}$ be a square complex matrix.
The index of $A$ is denoted by
$\operatorname{ind}(A),$ 
 which is defined as the smallest non-negative integer \(k\) satisfying,
$\operatorname{rank}(A^{k+1})
=
\operatorname{rank}(A^k).$
The Drazin inverse of $A$, denoted by $A^D$, is the unique matrix satisfying the following conditions \cite{Ben-Israel_2003}:
\begin{multicols}{3}
    \begin{enumerate}[label=(\alph*)]
    \item $A^{k+1}A^D=A^k$,
    \item $A^DAA^D=A^D$,
    \item $AA^D=A^DA$.
\end{enumerate}
\end{multicols}

Here, $k$ is the index of $A$. The Drazin inverse is useful when the matrix $A$ is singular or non-invertible.
<!-- end definition-4 -->
<!-- definition-5: body_md -->
Let $R:\widehat{\mathbb{C}}\to\widehat{\mathbb{C}}$ be a rational operator and let $z_0\in\widehat{\mathbb{C}}$.  
The orbit of $z_0$ under the action of $R$ is the sequence
\begin{equation*}
\mathcal{O}(z_0)=\{z_0,R(z_0),R^2(z_0),\ldots,R^n(z_0),\ldots\}.
\end{equation*}
<!-- end definition-5 -->
<!-- definition-6: body_md -->
A point $z^*\in\widehat{\mathbb{C}}$ is called a fixed point of the rational operator $R$ if
\begin{equation*}
R(z^*)=z^*.
\end{equation*}
<!-- end definition-6 -->
<!-- definition-7: body_md -->
A fixed point of the rational operator that is not a root of the polynomial equation $p(z)=0$ is called a strange fixed point.
<!-- end definition-7 -->
<!-- definition-8: body_md -->
A point $\hat{z}\in\widehat{\mathbb{C}}$ is called a periodic point of period $k \in \mathbb{N}$ if
\begin{equation*}
R^k(\hat{z})=\hat{z},
\end{equation*}
and $\;R^p(\hat{z})\neq \hat{z}, \; \forall\, p\in\mathbb{N},\; 1\leq p<k.$
<!-- end definition-8 -->
<!-- definition-9: body_md -->
Let $z^*$ be a fixed point of the rational operator $R$. The dynamical behavior of $z^*$ is determined by the modulus of the derivative of $R(z^*)$:
\begin{itemize}
    \item \textbf{Attracting}: if $|R'(z^*)|<1$.
    \item \textbf{Repelling}: if $|R'(z^*)|>1$.
    \item \textbf{Parabolic (or Neutral)}: if $|R'(z^*)|=1$.
    \item \textbf{Superattracting}: if $|R'(z^*)|=0$.
\end{itemize}
<!-- end definition-9 -->
<!-- definition-10: body_md -->
A point $z^*\in\widehat{\mathbb{C}}$ is called a critical point of the rational operator $R$ if
\begin{equation*}
R'(z^*)=0.
\end{equation*}
<!-- end definition-10 -->
<!-- definition-11: body_md -->
Critical points that are not roots of the polynomial equation $p(z)=0$ are called free critical points.
<!-- end definition-11 -->
<!-- definition-12: body_md -->
Let $z^*$ be an attracting fixed point of the rational operator $R$.  
Then the basin of attraction of $z^*$ is defined as:
\begin{equation*}
A(z^*)=\left\{z_0\in\widehat{\mathbb{C}}:\lim_{n\to\infty}R^n(z_0)=z^*\right\}.
\end{equation*}
<!-- end definition-12 -->
<!-- definition-13: body_md -->
The immediate basin of attraction of an attractor $z^*$ is the connected component of $A(z^*)$ that contains $z^*$.
<!-- end definition-13 -->

\label{theo:Fatou and julia}
Let $R$ be a rational function.  The immediate basin of attraction of each periodic orbit contains at least one critical point.

Consider the residual matrix at the $n$-th iteration $E_n = I - AX_n.$ At the $(n+1)$-th iteration,  the residual matrix satisfies
\begin{equation}
\label{eq:residual_n+1}
E_{n+1}
= I - AX_{n+1}.
\end{equation}
Substituting \eqref{eq:Heun4_matrix} into the equation \eqref{eq:residual_n+1} yields
\begin{align*}
E_{n+1}
= \frac{1}{729}\Big[&
(5854-1024\beta)AX_n - (18100-4864\beta)(AX_n)^2 + (28920-9408\beta)(AX_n)^3 \\&-(26285-9488\beta)(AX_n)^4 + (13874-5312\beta)(AX_n)^5 - (4134-1632\beta)(AX_n)^6 \\
&+ (640-256\beta)(AX_n)^7 - (40-16\beta)(AX_n)^8
\Big] \; - I.
\end{align*}
Since $AX_n = I - E_n$, substitution gives a polynomial expansion in $E_n$:
\begin{equation}\label{eq:residual_heun4_without_norm}
E_{n+1}
=
\left(-\frac{25}{27}+\frac{16\beta}{27}\right)E_n^4
+ \frac{10}{27}E_n^5
+ \left(\frac{86}{81}-\frac{32\beta}{81}\right)E_n^6 
+ \left(\frac{320}{729}-\frac{128\beta}{729}\right)E_n^7
+ \left(\frac{40}{729}-\frac{16\beta}{729}\right)E_n^8.
\end{equation}
Taking norms on both sides yields
\begin{equation}
\begin{aligned}
\|E_{n+1}\|
\leq&
\left|-\frac{25}{27}+\frac{16\beta}{27}\right|\|E_n\|^4
+ \frac{10}{27}\|E_n\|^5 + \left|\frac{86}{81}-\frac{32\beta}{81}\right|\|E_n\|^6
\\&+ \left|\frac{320}{729}-\frac{128\beta}{729}\right|\|E_n\|^7 + \left|\frac{40}{729}-\frac{16\beta}{729}\right|\|E_n\|^8.
\end{aligned}
\end{equation}
Since $\|E_0\| < 1$,  by induction process $\|E_n\| < 1$ for $n\geq 0$. Hence, by defining
\[
w =
\left|-\frac{25}{27}+\frac{16\beta}{27}\right|
+ \frac{10}{27}
+ \left|\frac{86}{81}-\frac{32\beta}{81}\right|
+ \left|\frac{320}{729}-\frac{128\beta}{729}\right|
+ \left|\frac{40}{729}-\frac{16\beta}{729}\right|,

we obtain
\begin{align*}
\|E_{n+1}\|
&\leq w \|E_n\|^4 \\
&\leq w (w \|E_{n-1}\|^4)^4
= w^{1+4} \|E_{n-1}\|^{4^2} \\
&\leq w^{1+4+4^2} \|E_{n-2}\|^{4^3}\\
&\quad  \vdots\\
& \leq w^{\frac{4^{n+1}-1}{3}}\|E_0\|^{4^{n+1}} ,\quad\quad \forall n \geq 0.
\end{align*}
For $\beta \in \left[\frac{25}{16}, \frac{5}{2}\right]$, a direct evaluation shows that $w = 1$. Because $\|E_0\|<1$, then $\|E_0\|^{4^{n+1}} \to 0$ when $n \to \infty$. Consequently $\|E_{n+1}\| \leq \|E_0\|^{4^{n+1}}$.

Hence,  $\|E_n\| \to 0 \;\; \text{as} \;n \to \infty$, 
which implies $X_n \to A^{-1}.$ So, the convergence of the sequence $\{X_n\}$ is  established.

Additionally, we have
\begin{align*}
\|I-AX_{n+1}\|
\leq\|I-AX_{n}\|^4&\Bigg( \left|-\frac{25}{27}+ \frac{16\beta}{27}\right| +\frac{10}{27}\|I-AX_{n}\| + \left|\frac{86}{81}-\frac{32\beta}{81}\right|\|I-AX_{n}\|^2 \\&+ \left|\frac{320}{729}-\frac{128\beta}{729}\right| \|I-AX_{n}\|^3 + \left|\frac{40}{729}-\frac{16\beta}{729}\right|\|I-AX_{n}\|^4\Bigg),
\end{align*}
and we can write the above expression as
\begin{align*}
\|A(A^{-1}-X_{n+1})\|
\leq\|A(A^{-1}-X_{n})\|^4&\Bigg( \left|-\frac{25}{27}+ \frac{16\beta}{27}\right| +\frac{10}{27}\|A(A^{-1}-X_{n})\| \\&+ \left|\frac{86}{81}-\frac{32\beta}{81}\right|\|A(A^{-1}-X_{n})\|^2 + \\&\left|\frac{320}{729}-\frac{128\beta}{729}\right| \|A(A^{-1}-X_{n})\|^3+ \\&\left|\frac{40}{729}-\frac{16\beta}{729}\right|\|A(A^{-1}-X_{n})\|^4\Bigg).
\end{align*}
Therefore,
\begin{align*}
\|A^{-1}A(A^{-1}-X_{n+1})\|
&\le \|A^{-1}\|\,\|A(A^{-1}-X_{n+1})\| \\
&\le \|A^{-1}\|\,\|A(A^{-1}-X_{n})\|^4 \Bigg( 
\left|-\frac{25}{27}+ \frac{16\beta}{27}\right|  + \frac{10}{27}\|A(A^{-1}-X_{n})\|\\&+ \left|\frac{86}{81}-\frac{32\beta}{81}\right|\|A(A^{-1}-X_{n})\|^2 + \left|\frac{320}{729}-\frac{128\beta}{729}\right|\|A(A^{-1}-X_{n})\|^3\\&  + \left|\frac{40}{729}-\frac{16\beta}{729}\right|\|A(A^{-1}-X_{n})\|^4 
\Bigg)  \\

\|A^{-1}\|\,\|A\|^4 \|A^{-1}-X_{n}\|^4 \\&+\frac{10}{27}\|A^{-1}\|\,\|A\|^5 \|A^{-1}-X_{n}\|^5 \\& + \left|\frac{86}{81}-\frac{32\beta}{81}\right|
\|A^{-1}\|\,\|A\|^6 \|A^{-1}-X_{n}\|^6 \\& + \left|\frac{320}{729}-\frac{128\beta}{729}\right|
\|A^{-1}\|\,\|A\|^7 \|A^{-1}-X_{n}\|^7 \\&+ \left|\frac{40}{729}-\frac{16\beta}{729}\right|
\|A^{-1}\|\,\|A\|^8 \|A^{-1}-X_{n}\|^8.
\end{align*}

\end{equation}

In particular, by choosing $\beta=\frac{25}{16}$, factoring \eqref{eq:Error_eq_heun4} gives

<!-- lemma-3: body_md -->
Let $\{X_n\}$ be the sequence generated by \eqref{eq:Heun4_matrix}, where 
$\beta \in \left[\frac{25}{16}, \frac{5}{2}\right]$, and the initial approximation is given by
$X_0 = \alpha A^*$, where $\alpha \in \mathbb{R}$ is a real constant. Then, the following properties hold for all $n \ge 0$:
\begin{multicols}{2}
\begin{enumerate}[label=i.]
    \item $X_n A A^\dagger = X_n$,
    \item $A^\dagger A X_n = X_n$,
    \item $(A X_n)^* = A X_n$,
    \item $(X_n A)^* = X_n A$.
\end{enumerate}
\end{multicols}

<!-- proof-3: body_md -->
We prove \textup{(ii)} by induction. The proof of \textup{(i)} is analogous.\\
For $n=0$
\[
A^\dagger A X_0
= A^\dagger A (\alpha A^*)
= \alpha (A^\dagger A) A^*.
\]
Using $(A^\dagger A)^* = A^\dagger A$ and properties of the Moore–Penrose inverse,
\[
\alpha (A^\dagger A)^* A^*
= \alpha A^* (A^\dagger)^* A^*
= \alpha (A A^\dagger A)^*
= \alpha A^*
= X_0.
\]
Next, we assume that $A^\dagger A X_n = X_n$. We show that $A^\dagger A X_{n+1} = X_{n+1}$.\\
From \eqref{eq: simplified HEUN4},
\[
A^\dagger AX_{n+1} = \frac{1}{729} A^\dagger A X_n \, \mathcal{P}(A X_n),
\]
where 
\begin{align*}
\mathcal{P}(A X_n)&=
5854I-1024\beta I -18100AX_n
+4864\beta AX_n+28920(AX_n)^2
-9408\beta(AX_n)^2 \\
&-26285(AX_n)^3
+9488\beta(AX_n)^3 +13874(AX_n)^4
-5312\beta(AX_n)^4 -4134(AX_n)^5\\&
+1632\beta(AX_n)^5 +640(AX_n)^6
-256\beta(AX_n)^6 -40(AX_n)^7
+16\beta(AX_n)^7
.
\end{align*}
Then,
\begin{align*}
A^\dagger A X_{n+1}
&= \frac{1}{729} X_n \, \mathcal{P}(A X_n) \quad  \\
&= X_{n+1}.
\end{align*}
Hence, property (ii) holds for all $n \ge 0$.

For $n=0$,

(AX_0)^* = (\alpha A A^*)^* = \alpha A A^* = AX_0.
\]
Assume $(AX_n)^* = AX_n$. Then,
\begin{align*}
(AX_{n+1})^*
&= \left(\frac{1}{729} A X_n \mathcal{P}(A X_n)\right)^* \\

\end{align*}
Since $(AX_n)^* = AX_n$ and $\mathcal{P}(AX_n)$ is a polynomial in $AX_n$, all its powers are Hermitian under the induction assumption, hence

(AX_{n+1})^* = AX_{n+1}.
\]
For (iv), the proof follows similarly by taking adjoints on the right-hand side and using the induction hypothesis, yielding
\[
(X_{n+1}A)^* = X_{n+1}A.

it follows that

(I-AA^\dagger)Ae_n = 0,

the above expression simplifies considerably. In particular,

\big((I-AA^\dagger)-Ae_n\big)^q
=
(I-AA^\dagger) + (-1)^q(Ae_n)^q,

for $q=4,5,6,7,8$.

\Bigg).
\end{aligned}

 \begin{align*}
\|e_{n+1}\|
\leq &
\|A^D\|\,\|A\|^4\,\|e_n\|^4
\Bigg(

\\

\|A\|^3\|e_n\|^3
+\left|\frac{40}{729}-\frac{16\beta}{729}\right|
\|A\|^4\|e_n\|^4
\Bigg).
\end{align*}
In particular, for $\beta=\frac{25}{16}$, the sequence $\{X_n\}$ converges to $A^{D}$ with order $p=5$, and $\{X_n\}$ must satisfy the error estimate for approximating the Drazin inverse of $A$

Let A be a $N \times N$ singular matrix, let $k=\operatorname{ind}(A)$ and $r=\operatorname{rank}(A^k)$; then there exists a non-singular matrix $V$ such that 
\[
A=V\begin{pmatrix}
    P &0\\0 & Q
\end{pmatrix}V^{-1},
\]
where $P$ is a non-singular matrix of size $r \times r$ and $Q$ is a nilpotent matrix of size $(N-r) \times (N-r)$ (see \cite{Ben-Israel_2003}). Its Drazin inverse is 
\[
A^D=V\begin{pmatrix}
    P^{-1} &0\\0 & 0
\end{pmatrix}V^{-1}.
\]
By choosing the initial approximation $X_0=\alpha A^l$, where $l \geq k=ind(A)$, we obtain
\[
X_0=V\begin{pmatrix}
    \alpha P^l &0\\0 & \alpha Q^l
\end{pmatrix}V^{-1}=V\begin{pmatrix}
    X_{0,P} &0 \\0&0
\end{pmatrix}V^{-1},
\]
which shows that the initial approximation $X_0$ preserves the structure of Drazin inverse.  Suppose that 
\[
X_n=V\begin{pmatrix}
     X_{n,P} &0\\0 & 0
\end{pmatrix}V^{-1},
\]
then
\[
AX_n=V\begin{pmatrix}
     PX_{n,P} &0\\0 & 0
\end{pmatrix}V^{-1},
\]
where $P$ is a non-singular matrix. For every power $q$, we have
\[
(AX_n)^q=V\begin{pmatrix}
     (PX_{n,P})^q &0\\0 & 0
\end{pmatrix}V^{-1}.
\]
From \eqref{eq: simplified HEUN4}, we can write
\[\mathcal{P}(A X_n)=V\begin{pmatrix}
    \mathcal{P}(PX_{n,P})&0\\0 &  \frac{5854-1024\beta}{729}I
\end{pmatrix},\]
where 
\begin{align*}
\mathcal{P}(A X_n)=&
5854I_r-1024\beta I_r -18100PX_{n,P}
+4864\beta PX_n+28920(PX_{n,P})^2
\\&-9408\beta(PX_{n,P})^2-26285(PX_{n,P})^3
+9488\beta(PX_{n,P})^3 +13874(PX_{n,P})^4
\\&-5312\beta(PX_{n,P})^4 -4134(PX_{n,P})^5
+1632\beta(PX_{n,P})^5 +640(PX_{n,P})^6
\\&-256\beta(PX_{n,P})^6 -40(PX_{n,P})^7
+16\beta(PX_{n,P})^7.
\end{align*}

Then the final expression of equation \eqref{eq: simplified HEUN4} becomes
\[X_{n+1}=\begin{pmatrix}
    X_{n,P}\mathcal{P}(PX_{n,P})&0\\0 &  0
    \end{pmatrix},\]
    where $X_{n,P}\mathcal{P}(A_1X_{n,P})$ is a non-singular matrix that can be written as
        \begin{align*}
        X_{n+1,P}=\frac{1}{729}X_{n,P}\Big[&5854I_r-1024\beta I_r -18100PX_{n,P}
+4864\beta PX_n\\&+28920(PX_{n,P})^2
-9408\beta(PX_{n,P})^2-26285(PX_{n,P})^3
\\&+9488\beta(PX_{n,P})^3 +13874(PX_{n,P})^4
-5312\beta(PX_{n,P})^4 \\&-4134(PX_{n,P})^5
+1632\beta(PX_{n,P})^5 +640(PX_{n,P})^6\\&
-256\beta(PX_{n,P})^6 -40(PX_{n,P})^7
+16\beta(PX_{n,P})^7\Big].
        \end{align*}
   By considering $E_{n,P}=I_r-PX_{n,P}$, the above expression can be written as
\begin{equation}
\begin{aligned}
E_{n+1,P}
=&
\left(-\frac{25}{27}+\frac{16\beta}{27}\right)E_{n,P}^4
+ \frac{10}{27}E_{n,P}^5
+ \left(\frac{86}{81}-\frac{32\beta}{81}\right)E_{n,P}^6 
\\&+ \left(\frac{320}{729}-\frac{128\beta}{729}\right)E_{n,P}^7
+ \left(\frac{40}{729}-\frac{16\beta}{729}\right)E_{n,P}^8.
\end{aligned}
\end{equation}
By choosing the initial approximation $X_0$ as $\|E_{0,P}\|=\|I_r-PX_{n+1,P}\|\leq 1$, it is easy to prove $\|E_{n,P}\|\leq 1,$ then we have
\begin{equation*}
\|E_{n+1,P}\|\leq w\|E_{n,P}\|^4\leq 
w^{\frac{4^n-1}{3}}\|E_{0,P}\|^{4^{n}} ,\quad\quad \forall n \geq 0,
\end{equation*}
where 
\[
w =
\left|-\frac{25}{27}+\frac{16\beta}{27}\right|
+ \frac{10}{27}
+ \left|\frac{86}{81}-\frac{32\beta}{81}\right|
+ \left|\frac{320}{729}-\frac{128\beta}{729}\right|
+ \left|\frac{40}{729}-\frac{16\beta}{729}\right|.
\]
For $\beta \in \left[\frac{25}{16}, \frac{5}{2}\right]$, a direct evaluation shows that $w = 1$. Because we choose $\|E_{0,P}\|<1$, then $\|E_{0,P}\|^{4^{n+1}} \to 0$ when $n \to \infty$. Consequently $\|E_{n+1,P}\| \leq \|E_{0,P}\|^{4^{n+1}}$.

Hence,  $\|E_{n,P}\| \to 0 \;\; \text{as} \;n \to \infty$, 
which implies $X_{n,P} \to P^{-1}.$
Hence, the convergence of the sequence $\{X_n\}$ is  established.
Since $X_0=\alpha A^l$, where $l \geq k=ind(A)$, we obtain
\[
\mathcal{R}(X_0)=\mathcal{R}(A^l)\subseteq\mathcal{R}(A^k).
\]
Therefore, we can state that
\[
\mathcal{R}(X_n)\subseteq\mathcal{R}(X_{n-1})\subseteq \mathcal{R}(X_{n-2})\subseteq\dots\subseteq\mathcal{R}(X_{0})\subseteq \mathcal{R}(A^k),
\]
and consequently,
\begin{equation}
\mathcal{R}(X_n)\subseteq\mathcal{R}(A^k),
\qquad
\forall n\geq0.
\label{eq:range_drazin_4}
\end{equation}
Similarly
\begin{equation}
\mathcal{N}(X_0)\supseteq\mathcal{N}(A^k),
\qquad
\forall n\geq0.
\end{equation}
Assume that
\[t \in \mathcal{N}(A^k),\]
then, after simplification
\[X_0t=\alpha A^l t =0.\]
This property is preserved by the iteration if
\[X_nt=0, \qquad AX_nt=0.\]
From \eqref{eq: simplified HEUN4}, we deduce $X_{n+1}t=0$. Consequently,
\[t \in \mathcal{N}({X_{n+1}}).\]
By induction,
\begin{equation}
    \label{eq:null_space_drazin_4}
    \mathcal{N}(A^k) \subseteq \mathcal{N}(X_n), \qquad n \geq 0.
\end{equation}
From \cite{Ben-Israel_2003}, we have
\begin{equation}
AA^D=A^DA=
P_{\mathcal{R}(A^k),\mathcal{N}(A^k)},
\label{eq:projector_RN_drazin_4}
\end{equation}
where $P_{\mathcal{R}(A^k),\mathcal{N}(A^k)}$ 
denotes the projector onto $\mathcal{R}(A^k)$ along $\mathcal{N}(A^k)$.\\
By employing Proposition \ref{proposition:projector} together with equations \eqref{eq:range_drazin_4}, \eqref{eq:null_space_drazin_4}, and \eqref{eq:projector_RN_drazin_4}, we conclude that
\begin{equation}\label{eq:Drazin_lemma_proof_1}
X_nAA^D=X_n=A^DAX_n,
\qquad
\forall n\geq0.
\end{equation}
Let $e_n = X_n - A^D$
denote the error matrix at the $n$-th iteration. Then
\begin{equation*}
\begin{aligned}
    e_n&=X_n-A^D\\&
    =A^D(I-AX_n)\\&
    =A^DE_n.
\end{aligned}
\end{equation*}
Taking norms on both sides and simplifying yields
\begin{equation}
    \|e_n\|\leq \|A^D\|\,\|E_n\|\leq w^{\frac{4^n-1}{3}}\|A^D\|\,\|E_0\|^{4^{n}} ,\quad\quad \forall n \geq 0. 
\end{equation}
We can write this as
\[\|A^D-X_n\|\leq w^{\frac{4^n-1}{3}}\|A^D\|\,\|I-AX_0\|^{4^{n}} ,\quad\quad \forall n \geq 0.\]
Hence, the expression stated in the theorem is satisfied. From the initial condition, we have assumed that $\|I-AX_0\|<1$. Then, $\|I-AX_0\|^{4^n} \to 0$ as $n \to \infty$. 
Hence, $\|A^D-X_n\|\to 0,$ and consequently $X_n \to A^D.$ So,  the convergence is proven. 
Moreover,
\begin{align*}
Ae_{n+1}
&= AX_{n+1} - AA^D \\
&= -E_{n+1} + I - AA^D.
\end{align*}
Using \eqref{eq:residual_heun4_without_norm}, we obtain
\begin{align*}
Ae_{n+1}
&={}
\left(\frac{25}{27}-\frac{16\beta}{27}\right)E_n^4
-\frac{10}{27}E_n^5
-\left(\frac{86}{81}-\frac{32\beta}{81}\right)E_n^6
\\
&
-\left(\frac{320}{729}-\frac{128\beta}{729}\right)E_n^7
-\left(\frac{40}{729}-\frac{16\beta}{729}\right)E_n^8
+ I - AA^D.
\end{align*}
Since
\[
E_n = I - AX_n
      = (I-AA^D)-Ae_n,
\]
it follows that
\begin{equation}
\begin{aligned}\label{eq:Ae(n+1)_drazin4}
Ae_{n+1}
&=
\left(\frac{25}{27}-\frac{16\beta}{27}\right)
\big((I-AA^D)-Ae_n\big)^4
-\frac{10}{27}
\big((I-AA^D)-Ae_n\big)^5
\\
&
-\left(\frac{86}{81}-\frac{32\beta}{81}\right)
\big((I-AA^D)-Ae_n\big)^6
-\left(\frac{320}{729}-\frac{128\beta}{729}\right)
\big((I-AA^D)-Ae_n\big)^7
\\
&
-\left(\frac{40}{729}-\frac{16\beta}{729}\right)
\big((I-AA^D)-Ae_n\big)^8
+ I - AA^D.
\end{aligned}
\end{equation}
Using the identities (see \cite{SoleymaniF_2013})
\[
(I-AA^D)^m = I-AA^D,
\qquad
(I-AA^D)Ae_n = 0,
\qquad
\forall m \in \mathbb{N},
\]
we obtain
\[
\big((I-AA^D)-Ae_n\big)^p
=
(I-AA^D)+(-1)^p(Ae_n)^p,
\]
for $p=4,5,6,7,8$.

Substituting these expressions into the equation \eqref{eq:Ae(n+1)_drazin4} gives
\begin{align*}
Ae_{n+1}
&={}
\left(\frac{25}{27}-\frac{16\beta}{27}\right)(Ae_n)^4
+\frac{10}{27}(Ae_n)^5
-\left(\frac{86}{81}-\frac{32\beta}{81}\right)(Ae_n)^6
\\
&
+\left(\frac{320}{729}-\frac{128\beta}{729}\right)(Ae_n)^7
-\left(\frac{40}{729}-\frac{16\beta}{729}\right)(Ae_n)^8.
\end{align*}
Taking norms on both sides yields
\begin{align*}
\|Ae_{n+1}\|
\leq {} &
\|Ae_n\|^4
\Bigg(
\left|\frac{25}{27}-\frac{16\beta}{27}\right|
+\frac{10}{27}\|Ae_n\|
+\left|\frac{86}{81}-\frac{32\beta}{81}\right|
\|Ae_n\|^2
\\
&\quad
+\left|\frac{320}{729}-\frac{128\beta}{729}\right|
\|Ae_n\|^3
+\left|\frac{40}{729}-\frac{16\beta}{729}\right|
\|Ae_n\|^4
\Bigg).
\end{align*}
Since
\[
e_{n+1}=A^DAe_{n+1},
\]
we have
\[
\|e_{n+1}\|
\leq
\|A^D\|\,\|Ae_{n+1}\|.
\]
Moreover,
\[
\|Ae_n\| \leq \|A\|\,\|e_n\|.
\]
Therefore,
\begin{equation}\label{eq:error_Drazin4}
\begin{aligned}
\|e_{n+1}\|
\leq &
\|A^D\|\,\|A\|^4\,\|e_n\|^4
\Bigg(
\left|-\frac{25}{27}+\frac{16\beta}{27}\right|
+\frac{10}{27}\|A\|\|e_n\|
+\left|\frac{86}{81}-\frac{32\beta}{81}\right|
\|A\|^2\|e_n\|^2
\\
&\quad
+\left|\frac{320}{729}-\frac{128\beta}{729}\right|
\|A\|^3\|e_n\|^3
+\left|\frac{40}{729}-\frac{16\beta}{729}\right|
\|A\|^4\|e_n\|^4
\Bigg).
\end{aligned}
\end{equation}

Hence, $\|e_n\| \to 0
\;\;\text{as}\;\; n\to\infty$, which implies that $X_n \to A^D.$\\
Therefore, for $\beta \in \left[\frac{25}{16},\frac{5}{2}\right] $, the sequence $\{X_n\}$ converges to the Drazin inverse $A^D$ with fourth-order convergence.
For $\beta=\frac{25}{16},$ the equation \eqref{eq:error_Drazin4} becomes 
\begin{align*}
\|e_{n+1}\|
\leq {} &
\|A^D\| \, \|A\|^5 \, \|e_n\|^5
\Bigg(
\frac{10}{27}
+\frac{4}{9}\|A\|\|e_n\|
+\frac{120}{729}\|A\|^2\|e_n\|^2
+\frac{15}{729}\|A\|^3\|e_n\|^3
\Bigg),
\end{align*}
which represents the fifth-order of convergence.

we obtain the triple critical point $x=1$ and the real roots of the polynomial $p(x).$ All the real roots of the above expression are considered the free critical points,  except $x=1$, which corresponds to the root of the rational function.
