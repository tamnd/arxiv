<!-- definition-1: body_md -->
Let $\bm{A}\in\ensuremath\mathbb{R}^{n\timesn}$ be the system matrix of an asymptotically stable \gls{lti} system~\cref{eq:full_order_state_space_system}.
	The decomposition of $\bm{A}$ into the form
	\begin{equation*}
	\bm{A} = (\bm{J} - \bm{R})\bm{Q},
	\end{equation*}
	with a skew-symmetric matrix $\bm{J}\in\ensuremath\mathbb{R}^{n\timesn}$, a symmetric positive semidefinite matrix $\bm{R}\in\mathcal{S}^{n}_{\succeq}$, and a symmetric positive definite matrix $\bm{Q}\in\mathcal{S}^{n}_{\succ}$, is referred to as \emph{dissipative Hamiltonian} decomposition.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
An \gls{lti} system of the form
\begin{equation}
\label{eq:linear_ph_system_with_feedthrough}
\begin{aligned}
\dot{\bm{x}}(t) &= (\bm{J}-\bm{R})\bm{Q}\bm{x}(t) + (\bm{G} - \bm{P}) \bm{u}(t), \quad \bm{x}(0) = \bm{x}_{0}, \\
\bm{y}(t) &= (\bm{G} + \bm{P})^{\intercal}\bm{Q}\bm{x}(t) + (\bm{S} + \bm{N})\bm{u}(t),
\end{aligned}
\end{equation}
together with the Hamiltonian
\begin{align}
	\label{eq:hamiltonian_quadratic_form}
	\mathcal{H}\colon \ensuremath\mathbb{R}^{n}\to\ensuremath\mathbb{R}, \quad \bm{x} \mapsto \tfrac{1}{2}\bm{x}^{\intercal}\bm{Q}\bm{x},
\end{align} 
with the matrices $\bm{J},\bm{R},\bm{Q}\in\ensuremath\mathbb{R}^{n\timesn}$, $\bm{G}, \bm{P}\in\ensuremath\mathbb{R}^{n\timesm}$, $\bm{S},\bm{N}\in\ensuremath\mathbb{R}^{m\timesm}$, is called a \gls{ph} system if the following conditions hold:
\begin{enumerate}[label=(\roman*)]
	\item the Hessian of the Hamiltonian $\bm{Q}$ is symmetric positive definite, 
	\item the structure matrix $ \bm{\Gamma} \vcentcolon= \begin{bmatrix}
\bm{J} & \bm{G} \\
-\bm{G}^{\intercal} & \bm{N}
\end{bmatrix}$ is skew-symmetric, and
	\item the dissipation matrix $\bm{\Xi} \vcentcolon= \begin{bmatrix}
\bm{R} & \bm{P} \\
\bm{P}^{\intercal} & \bm{S}
\end{bmatrix}$ is symmetric positive semidefinite.
\end{enumerate}
<!-- end definition-2 -->
<!-- proposition-1: body_md -->
Let $\bm{W}$ be chosen as $\bm{W} = \bm{Q}\bm{V}$. Then, the \gls{rom} is given by 
	\begin{subequations}\label{eq:pg:intrusive:rom:ph}
	\begin{equation}\label{eq:pg:intrusive:rom:ph:sys}
	\begin{aligned}
	\bm{E}_{\mathrm{r}}\dot{\bm{x}}_{\mathrm{r}}(t) 
		&= (\bm{J}_{\mathrm{r}}-\bm{R}_{\mathrm{r}})\bm{x}_{\mathrm{r}}(t) + (\bm{G}_{\mathrm{r}} - \bm{P}_{\mathrm{r}}) \bm{u}(t), 
		\quad \bm{x}_{\mathrm{r}}(0) = \@ifstar{\@ifstar{\@ifstar{\@ifstar{\parentheses}{\parentheses*}}{\@ifstar{\parentheses}{\parentheses*}*}}{\@ifstar{\@ifstar{\parentheses}{\parentheses*}}{\@ifstar{\parentheses}{\parentheses*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\parentheses}{\parentheses*}}{\@ifstar{\parentheses}{\parentheses*}*}}{\@ifstar{\@ifstar{\parentheses}{\parentheses*}}{\@ifstar{\parentheses}{\parentheses*}*}*}*}{\bm{W}^{\intercal}\bm{V}}^{-1}\bm{W}^{\intercal} \bm{x}_{0}, \\
	\bm{y}_{\mathrm{r}}(t) 
		&= (\bm{G}_{\mathrm{r}} + \bm{P}_{\mathrm{r}})^{\intercal}\bm{x}_{\mathrm{r}}(t) + (\bm{S}_{\mathrm{r}} + \bm{N}_{\mathrm{r}})\bm{u}(t),
	\end{aligned}
	\end{equation}
	where the reduced  matrices are defined by
	\begin{equation}\label{eq:pg:intrusive:rom:ph:mats}
		\begin{aligned}
		\bm{E}_{\mathrm{r}} &\vcentcolon= \bm{V}^{\intercal}\bm{Q}^{\intercal}\bm{V},
		&\bm{Q}_{\mathrm{r}} &\vcentcolon= \bm{I}_{r},
		&\bm{J}_{\mathrm{r}} &\vcentcolon= \bm{V}^{\intercal}\bm{Q}^{\intercal}\bm{J}\bm{Q}\bm{V}, 
		&\bm{R}_{\mathrm{r}} &\vcentcolon= \bm{V}^{\intercal}\bm{Q}^{\intercal}\bm{R}\bm{Q}\bm{V}, \\
		\bm{G}_{\mathrm{r}} &\vcentcolon= \bm{V}^{\intercal}\bm{Q}^{\intercal}\bm{G},
		&\bm{P}_{\mathrm{r}} &\vcentcolon= \bm{V}^{\intercal}\bm{Q}^{\intercal}\bm{P},
		&\bm{S}_{\mathrm{r}} &\vcentcolon= \bm{S}, 
		&\bm{N}_{\mathrm{r}} &\vcentcolon= \bm{N}.
	\end{aligned} 
	\end{equation}
	\end{subequations}
	Hence, $\bm{E}_{\mathrm{r}}$ is symmetric positive definite, $\bm{J}_{\mathrm{r}}$ is skew-symmetric, and $\bm{R}_{\mathrm{r}}$ is symmetric positive semidefinite.

<!-- theorem-1: body_md -->
\label{thm:pgopinf}
	Assume that the data matrices~\cref{eq:data_matrices} satisfy~\cref{eq:data_satisfies_system_equations} for some $\Delta_{\bm{x}}\in\ensuremath\mathbb{R}^{n\timesN}$ and $\Delta_{\bm{y}}\in\ensuremath\mathbb{R}^{p\timesN}$.
	Moreover, for some $\bm{V} \in \ensuremath\mathbb{R}^{n \times r}$ and $\bm{W} \in \ensuremath\mathbb{R}^{n \times r}$, consider the Petrov--Galerkin \gls{opinf} problem~\cref{eq:pgopinf} with projected data according to~\cref{eq:projected_data_matrices_petrov_galerkin}.
	If the matrix $\bm{T}_{\mathrm{r}}$ has full row rank, then the minimizers of the Petrov--Galerkin \gls{opinf} problem~\cref{eq:pgopinf} satisfy
	\begin{equation}\label{eq:thm:pgopinf}
		\begin{aligned}
			\begin{bmatrix}
			\tilde{\bm{A}}_{\mathrm{r}}^{*} & \tilde{\bm{B}}_{\mathrm{r}}^{*}\\
			\tilde{\bm{C}}_{\mathrm{r}}^{*} & \tilde{\bm{D}}_{\mathrm{r}}^{*}
			\end{bmatrix}
			&= \begin{bmatrix}
				\bm{A}_{\mathrm{r}} & \bm{B}_{\mathrm{r}}\\
				\bm{C}_{\mathrm{r}} & \bm{D}_{\mathrm{r}}
			\end{bmatrix} + \begin{bmatrix}
		\bm{W}^{\intercal}\bm{A}\bm{X}_{\perp} + \bm{W}^{\intercal}\Delta_{\bm{x}}\\
				\bm{C}\bm{X}_{\perp} + \Delta_{\bm{y}}
			\end{bmatrix}
			\bm{T}_{\mathrm{r}}^{\dagger},
		\end{aligned}
	\end{equation}

The assumption that the data matrices \cref{eq:data_matrices} satisfy \cref{eq:data_satisfies_system_equations} implies
	\begin{equation}
		\label{eq:projected_dynamics_data_equation}
		\begin{aligned}
			\bm{W}^{\intercal}\dot{\bm{X}} &= \bm{W}^{\intercal}\bm{A} \bm{X} + \bm{W}^{\intercal}\bm{B}\bm{U} + \bm{W}^{\intercal}\Delta_{\bm{x}}\\
			\bm{Y} &= \bm{C} \bm{X} + \bm{D} \bm{U} + \Delta_{\bm{y}}.
		\end{aligned}
	\end{equation}
	Define $\bm{E}_{\mathrm{r}}\vcentcolon=\bm{W}^{\intercal}\bm{V}$, then
	\begin{align}\label{eq:relation_descriptor_to_proj_Dstate}
	 	\bm{W}^{\intercal}\dot{\bm{X}} =  (\bm{W}^{\intercal}\bm{V})(\bm{W}^{\intercal}\bm{V})^{-1}\bm{W}^{\intercal}\dot{\bm{X}} = \bm{E}_{\mathrm{r}}\dot{\bm{X}}_{\mathrm{r}}.
	\end{align}
	Hence, together with the data decomposition \cref{eq:data_decomposition}, we can rewrite \cref{eq:projected_dynamics_data_equation} as
	\begin{equation}\label{eq:petrov_galerkin_intrusive_with_residual}
		\begin{aligned}
		\bm{E}_{\mathrm{r}}\dot{\bm{X}}_{\mathrm{r}} 
			&= \bm{W}^{\intercal}\bm{A}\bm{V}\bm{X}_{\mathrm{r}} 		
			+ \bm{W}^{\intercal}\bm{B}\bm{U}
			+ \bm{W}^{\intercal}\bm{A}\bm{X}_{\perp} + \bm{W}^{\intercal}\Delta_{\bm{x}}\\
			\bm{Y} &= \bm{C} \bm{V}\bm{X}_{\mathrm{r}} + \bm{D} \bm{U} + \bm{C} \bm{X}_{\perp} + \Delta_{\bm{y}},
		\end{aligned}
	\end{equation}
	which reads in block form as
	\begin{equation}
		\label{eq:block_form_petrov_galerkin_intrusive_with_residual}
		\begin{bmatrix}
			\bm{E}_{\mathrm{r}}\dot{\bm{X}}_{\mathrm{r}} \\
			\bm{Y}
		\end{bmatrix}
		=
		\begin{bmatrix}
			\bm{A}_{\mathrm{r}} & \bm{B}_{\mathrm{r}} \\
			\bm{C}_{\mathrm{r}} & \bm{D}_{\mathrm{r}}
		\end{bmatrix}
		\begin{bmatrix}
			\bm{X}_{\mathrm{r}} \\
			\bm{U}
		\end{bmatrix}
		+
		              
		\begin{bmatrix}
			\bm{W}^{\intercal}\bm{A}\bm{X}_{\perp} + \bm{W}^{\intercal}\Delta_{\bm{x}}\\
			\bm{C}\bm{X}_{\perp} + \Delta_{\bm{y}}
		\end{bmatrix}.
	\end{equation}

		\begin{bmatrix}
			\tilde{\bm{A}}_{\mathrm{r}}^{*} & \tilde{\bm{B}}_{\mathrm{r}}^{*} \\
			\tilde{\bm{C}}_{\mathrm{r}}^{*} & \tilde{\bm{D}}_{\mathrm{r}}^{*}
		\end{bmatrix} 
		= \bm{Z}_{\mathrm{r}}\bm{T}_{\mathrm{r}}^{\dagger}
		= \begin{bmatrix}

			\bm{Y}
		\end{bmatrix} \bm{T}_{\mathrm{r}}^{\dagger}
		&= \@ifstar{\@ifstar{\@ifstar{\@ifstar{\parentheses}{\parentheses*}}{\@ifstar{\parentheses}{\parentheses*}*}}{\@ifstar{\@ifstar{\parentheses}{\parentheses*}}{\@ifstar{\parentheses}{\parentheses*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\parentheses}{\parentheses*}}{\@ifstar{\parentheses}{\parentheses*}*}}{\@ifstar{\@ifstar{\parentheses}{\parentheses*}}{\@ifstar{\parentheses}{\parentheses*}*}*}*}{
			\begin{bmatrix}
			\bm{A}_{\mathrm{r}} & \bm{B}_{\mathrm{r}} \\
			\bm{C}_{\mathrm{r}} & \bm{D}_{\mathrm{r}}
		\end{bmatrix} 
		\bm{T}_{\mathrm{r}} + 
		\begin{bmatrix}
			\bm{W}^{\intercal}\bm{A}\bm{X}_{\perp} + \bm{W}^{\intercal}\Delta_{\bm{x}}\\
			\bm{C}\bm{X}_{\perp} + \Delta_{\bm{y}}
		\end{bmatrix}
		}\bm{T}_{\mathrm{r}}^{\dagger} \\
		&=\begin{bmatrix}
			\bm{A}_{\mathrm{r}} & \bm{B}_{\mathrm{r}} \\
			\bm{C}_{\mathrm{r}} & \bm{D}_{\mathrm{r}}
		\end{bmatrix} 
		\bm{T}_{\mathrm{r}}\bm{T}_{\mathrm{r}}^{\dagger} + 
		\begin{bmatrix}
			\bm{W}^{\intercal}\bm{A}\bm{X}_{\perp} + \bm{W}^{\intercal}\Delta_{\bm{x}}\\
			\bm{C}\bm{X}_{\perp} + \Delta_{\bm{y}}
		\end{bmatrix} \bm{T}_{\mathrm{r}}^{\dagger}
	\end{align*}
	The full row rank assumption for $\bm{T}_{\mathrm{r}}$ implies $\bm{T}_{\mathrm{r}}\bm{T}_{\mathrm{r}}^{\dagger}=\bm{I}_{r+m}$, which completes the proof.

<!-- remark-1: body_md -->
\label{rem:representation_e}
	From~\cref{eq:block_form_petrov_galerkin_intrusive_with_residual} follows that the terms $\mathcal{E}_{\bm{x}}$ and $\mathcal{E}_{\bm{y}}$ can be equivalently represented as
	\begin{equation*}
		\mathcal{E}_x = \bm{E}_{\mathrm{r}}\dot{\bm{X}}_{\mathrm{r}} - \bm{A}_{\mathrm{r}} \bm{X}_{\mathrm{r}} - \bm{B}_{\mathrm{r}}\bm{U}
		\quad\text{and}\quad
		\mathcal{E}_y = \bm{Y} - \bm{C}_{\mathrm{r}}\bm{X}_{\mathrm{r}} - \bm{D}_{\mathrm{r}}\bm{U}.
	\end{equation*}
	This representation for $\mathcal{E}_{\bm{x}}$ and $\mathcal{E}_{\bm{y}}$ is consistent 
	with the results reported in~\cite{Peherstorfer20}, where the data is obtained via a so-called reprojection algorithm.
	In more detail, for a given $\bm{V}$, the reprojection algorithm described in \cite[Fig.~2, Alg.~1]{Peherstorfer20} 
	constructs the data iteratively. In each step of the sampling process, the current state is first projected onto 
	the reduced space and directly lifted back to the full space. The lifted state is subsequently used to evaluate the 
	full-order dynamics in order to generate the next state sample for one time step. Under certain conditions, the 
	resulting reprojected trajectories coincide with those obtained from the intrusive \gls{rom}. As a consequence, 
	the error terms $\mathcal{E}_{\bm{x}}$ and $\mathcal{E}_{\bm{y}}$ vanish, and the operators identified by \gls{opinf} 
	coincide with the intrusive operators of the Petrov--Galerkin projection, which is consistent with \cite[Cor.~3.2]{Peherstorfer20}.

<!-- proposition-2: body_md -->
The optimization problem~\cref{eq:convex_linear_reduced_ph_optimization_problem} is a convex optimization problem.

<!-- proof-3: body_md -->
Let us define the residual function $\mathcal{R}$ of the optimization problem as
\begin{align}
	\mathcal{R}(\tilde{\bm{E}}_{\mathrm{r}},\tilde{\bm{\Gamma}}_{\mathrm{r}},\tilde{\bm{\Xi}}_{\mathrm{r}}) := \begin{bmatrix}
            \tilde{\bm{E}}_{\mathrm{r}}\dot{\bm{X}}_{\mathrm{r}} \\ -\bm{Y}
        \end{bmatrix} - 
        \@ifstar{\@ifstar{\@ifstar{\@ifstar{\parentheses}{\parentheses*}}{\@ifstar{\parentheses}{\parentheses*}*}}{\@ifstar{\@ifstar{\parentheses}{\parentheses*}}{\@ifstar{\parentheses}{\parentheses*}*}*}}{\@ifstar{\@ifstar{\@ifstar{\parentheses}{\parentheses*}}{\@ifstar{\parentheses}{\parentheses*}*}}{\@ifstar{\@ifstar{\parentheses}{\parentheses*}}{\@ifstar{\parentheses}{\parentheses*}*}*}*}{\tilde{\bm{\Gamma}}_{\mathrm{r}} - \tilde{\bm{\Xi}}_{\mathrm{r}}}
        \begin{bmatrix} 
            \bm{X}_{\mathrm{r}} \\ \bm{U} 
        \end{bmatrix}
\end{align}
and observe that $\mathcal{R}$ is affine in its arguments.
Hence, the residual $\mathcal{R}$ is affine in all decision variables.
Moreover, \cref{eq:convex_linear_reduced_ph_optimization_problem} is convex, since
the Frobenius norm is convex and the composition of a convex function with an affine function is again convex.
Finally, the constraints on the matrices are convex sets: the set of skew-symmetric matrices is a linear subspace, the set of positive semidefinite matrices is a convex cone, and the set of positive definite matrices is an open convex cone.
Consequently, the entire optimization problem \cref{eq:convex_linear_reduced_ph_optimization_problem} is a convex optimization problem.
