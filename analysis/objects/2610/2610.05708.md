<!-- lemma-1: body_md -->
\label{lem:maximum_principle}
		Let $n\ge 2$, $\Omega$, $\mathcal{D}_1$, and $\mathcal{D}_2$ be Lipschitz domains in $\mathbb{R}^n$. Let $u$ be a piecewise $H^1$ subsolution to the equation \eqref{robin}, that is,
		$u_+\in H^1(\widetilde\Omega)$ and $u_- \in H^1(\mathcal{D}_1 \cup \mathcal{D}_2)$ satisfy
		\begin{equation}\label{subrobin}
			\begin{cases}
				\Delta u_+ \ge 0&\mbox{in}~\widetilde\Omega,\\
				\Delta u_- \ge 0&\mbox{in}~\mathcal{D}_1\cup\mathcal{D}_2,\\ 
				\partial_\nu u_+ \le  -\gamma^{-1}(u_+-u_-)&\mbox{on}~\partial \mathcal{D}_i, \quad i=1,2,\\
				k_i \partial_\nu u_- \ge -\gamma^{-1}(u_+-u_-)&\mbox{on}~\partial \mathcal{D}_i, \quad i=1,2,\\
				u_+ \le \varphi&\mbox{on}~\partial\Omega.
			\end{cases}
		\end{equation}
		Then it holds that
		\begin{equation*}            
			u \le \sup_{\partial \Omega} \varphi \quad \mbox{in } \Omega.
		\end{equation*}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
By considering $u_\pm -\sup_{\partial \Omega} \varphi$ in place of $u_\pm$, we may assume $u_+\le 0$ on $\partial\Omega$, and it suffices to show that
		\begin{equation}\label{v_upper}
			u \le 0 \quad \mbox{in } \Omega.
		\end{equation}
		We denote $v_1:=\max\{u_+,0\}$ and $v_2:=\max\{u_-,0\}$. We multiply the first equation in \eqref{subrobin} by $v_1$ and integrate by parts in $\widetilde\Omega$. We have
		\begin{align*}
			0& \ge \int_{\widetilde\Omega} |\nabla v_1|^{2}- \sum_{j=1}^2 \int_{\partial \mathcal{D}_j} v_1\,\partial_\nu u_+ \\
			& \ge \int_{\widetilde\Omega} |\nabla v_1|^{2}+ \gamma^{-1}\sum_{j=1}^2 \int_{\partial \mathcal{D}_j} (v_1^2 -v_1 u_- )\\
			& \ge \int_{\widetilde\Omega} |\nabla v_1|^{2} +  \gamma^{-1}\sum_{j=1}^2 \int_{\partial \mathcal{D}_j} (v_1^2 -v_1 v_2).
		\end{align*}
		We then multiply the second equation in \eqref{subrobin} by $k_j v_2$ and integrate by parts in $\mathcal{D}_j$, $j = 1,2$. We have
		\begin{align*}
			0& \ge \sum_{j=1}^2 \int_{\mathcal{D}_j} k_j |\nabla v_2|^{2} + \sum_{j=1}^2 k_j \int_{\partial \mathcal{D}_j} v_2\,\partial_\nu u_- \\
			& \ge \sum_{j=1}^2 \int_{\mathcal{D}_j} k_j |\nabla v_2|^{2}+ \gamma^{-1}\sum_{j=1}^2 \int_{\partial \mathcal{D}_j} (v_2^2 -v_2 u_+) \\
			& \ge \sum_{j=1}^2 \int_{\mathcal{D}_j} k_j |\nabla v_2|^{2} +  \gamma^{-1}\sum_{j=1}^2 \int_{\partial \mathcal{D}_j} (v_2^2 -v_1 v_2).
		\end{align*}
		Summing up the two inequalities above yields
		$$
		\int_{\widetilde\Omega} |\nabla v_1|^{2} + \sum_{j=1}^2 \int_{\mathcal{D}_j} k_j |\nabla v_2|^{2} +  \gamma^{-1}\sum_{j=1}^2 \int_{\partial \mathcal{D}_j} (v_1- v_2)^2 \le 0.
		$$
		Therefore,
		$$
		\nabla v_1 \equiv 0 ~\mbox{in}~\widetilde\Omega, \quad \nabla v_2 \equiv 0 ~\mbox{in}~\mathcal{D}_1 \cup \mathcal{D}_2, \quad v_1 \equiv v_2 ~\mbox{on}~ \partial \mathcal{D}_1 \cup \partial \mathcal{D}_2.
		$$
		Since $v_1 = 0$ on $\partial\Omega$, it follows that $v_1 \equiv 0$ in $\widetilde \Omega$, and hence $v_2 \equiv 0$ in $\mathcal{D}_1 \cup \mathcal{D}_2$. Therefore, \eqref{v_upper} is proved.

			\|u\|_{{C^{1,\beta}}(\widetilde\Omega \setminus \Omega_{R_0/2})} &\le C \Big( 1 + \frac{1}{k_1} + \frac{1}{k_2}\Big)\|\varphi\|_{C^{1,\beta}(\partial \Omega)}.
		\end{align}

First, \eqref{u_C0} simply follows from the maximum principle Lemma \ref{lem:maximum_principle}. For $i = 1, 2$, $u_-$ satisfies
		$$
		\begin{cases}
			\Delta u_- = 0&\mbox{in}~\mathcal{D}_i,\\ 
			\partial_\nu u_- = -(k_i\gamma)^{-1}(u_+-u_-)&\mbox{on}~\partial \mathcal{D}_i.
		\end{cases}
		$$
		By the $W^{1,p}$ estimate for the Neumann boundary problem (see \cite{DongLi21}*{Theorem 2.6}), we have
		$$
		\| u_- - (u_-)_{\mathcal{D}_i} \|_{W^{1,p}(\mathcal{D}_i)} \le  \frac{C}{k_i} \| u_+ - u_-\|_{ L^{p} (\partial \mathcal{D}_i)},
		$$
		which implies \eqref{u_holder_inside} by \eqref{u_C0} and Morrey's inequality. Recall that $u_+$ satisfies
		$$
		\begin{cases}
			\Delta u_+ = 0&\mbox{in}~\widetilde\Omega,\\ 
			\partial_\nu u_+ = {-}\gamma^{-1}(u_+-u_-)&\mbox{on}~\partial \mathcal{D}_1 \cup \partial \mathcal{D}_2,\\
			u_+ = \varphi &\mbox{on}~\partial \Omega.
		\end{cases}
		$$
		Away from the narrow gap region, one can apply a similar { local $W^{1,p}$ estimate} to see that
		$$
		[u_+]_{{C^{\beta}}(\widetilde\Omega \setminus \Omega_{R_0/4})} \le C \| \varphi\|_{C^\beta (\partial \Omega)}.
		$$
		Then one can apply the $C^{1,\beta}$ estimate for the Neumann boundary problem (see \cite{Lieberman87}*{Theorem 1.2}) to obtain
		$$
		\|u\|_{{C^{1,\beta}}(\widetilde\Omega \setminus \Omega_{R_0/2})} \le C \Big( [u_+]_{C^\beta(\widetilde\Omega \setminus \Omega_{R_0/4})} + {\|u_-\|_{C^\beta(\mathcal{D}_1)} + \|u_-\|_{C^\beta(\mathcal{D}_2)}} + \| \varphi\|_{C^{1,\beta} (\partial \Omega)} \Big),
		$$
		which implies \eqref{u_C1_outside}.

		Let $h_1$ and $h_2$ be $C^{1,1}$ functions satisfying \eqref{fg_0}--\eqref{def:c_2}, $n \ge 2$, $\gamma \ge \gamma_0 > 0$, $\varepsilon \in(0,R_0/4)$, and $u \in H^1({\Omega}_{R_0})$ be a solution of \eqref{main_problem_narrow} for some  functions $g_1,g_2 \in C^{\beta}_{x'}$ for some $\beta \in (0,1)$. Then there exists  a constant $C_1>0$ depending only on $n$, $\gamma_0$, $\beta$, $R_0$, $c_1$, and $c_2$, and a constant $C_2 >0$ further depending on $\gamma$, such that for any $x_0\in \Omega_{R_0/2}$ and $r=\frac{1}{4}(\varepsilon+|x'_0|^2)^{\frac{1}{2}}$, we have

				\Big),

		\begin{equation}\label{gradient-1/2}

				&r\|\nabla u\|_{L^\infty(\Omega_{x_0,r})} + r^{1+\beta} [\nabla u]_{C^\beta(\Omega_{x_0,r})} \\

				\Big),

It suffices to prove estimates \eqref{gradient-1/2a} and \eqref{gradient-1/2} for $r < r_0$, where $r_0$ is a small $\varepsilon$-independent constant to be specified during the proof.
		Without loss of generality, we assume $g_2(x_0')=-g_1(x_0')$, since otherwise we can consider the equation for $u-(g_1(x_0')+g_2(x_0'))/2$ instead of $u$.
		
		\textbf{Step 1:} Subtract a linear function in $x_n$.
		
		{Fix $x_0 \in \Omega_{R_0/2}$ and define}
		$$
		\hat{u}:= u - (A x_n + B).
		$$
		The first step is to find $A$ and $B$ such that $\hat u + \gamma \partial_\nu \hat u = 0$ at $x' = x_0'$ on $\Gamma_\pm$. By direct computations, we have
		on $\Gamma_+$,
		\begin{align*}
			\hat{u} + \gamma \partial_\nu \hat{u} &= g_1(x') - A \left( \frac{\varepsilon}{2} + h_1(x') + \frac{\gamma}{\sqrt{1 + |\nabla h_1|^2}} \right) - B;
		\end{align*}
		on $\Gamma_-$,
		\begin{align*}
			\hat{u} + \gamma \partial_\nu \hat{u} &= g_2(x') - A \left( -\frac{\varepsilon}{2} + h_2(x') - \frac{\gamma}{\sqrt{1 + |\nabla h_2|^2}} \right) - B.
		\end{align*}
		Set
		$$
		\left\{
		\begin{aligned}
			A \left( \frac{\varepsilon}{2} + h_1(x_0') + \frac{\gamma}{\sqrt{1 + |\nabla h_1(x_0')|^2}} \right) {+} B &= g_1(x_0'),\\
			A \left( -\frac{\varepsilon}{2} + h_2(x_0') - \frac{\gamma}{\sqrt{1 + |\nabla h_2(x_0')|^2}} \right) {+} B &= {g_2(x_0')}.
		\end{aligned}
		\right.
		$$
		Then one can solve for $A$ and $B$ explicitly, and see that
		\begin{equation}\label{est:ab}
			|A| \le C |g_1(x_0')|, \quad |B| \le C |x_0'|^2 |g_1(x_0')|.
		\end{equation}
		Thus, we conclude that $\hat{u}$ satisfies
		\begin{equation}\label{main_problem_narrow_2}
			\begin{cases}
				\Delta \hat u =0&\mbox{in}~\Omega_{R_0},\\
				\hat u + \gamma \partial_\nu \hat u = \tilde g_1&\mbox{on}~\Gamma_+,\\
				\hat u + \gamma \partial_\nu \hat u = \tilde g_2&\mbox{on}~\Gamma_-,
			\end{cases}
		\end{equation}
		with $\tilde g_1(x_0') = \tilde g_2(x_0') = 0$.
		
		
		\textbf{Step 2:} Flatten the boundaries.
		
		For any $x_0 \in \Omega_{R_0/2}$, we denote
		$$r:=\frac{1}{4}(\varepsilon+|x_0'|^2)^{1/2},$$
		and
		$$
		Q_{s,t} := \{ y = (y', y_n) \in \mathbb{R}^n:~ |y'-x_0'| < s,
        \,|y_n| < t \},
		$$
		Using the flattening map $\Phi$ defined in \eqref{Phi} {which satisfies
$\Phi(Q_{1.9r, r^2})\subset \Omega_{x_0,2r}$ and $\Omega_{x_0,r} \subset \Phi(Q_{1.1r, r^2})$}, and defining $\tilde{u}(y) = \hat u(\Phi(y))$, where $ \hat u$ is a solution of \eqref{main_problem_narrow_2}, we know that $\tilde{u}(y)$ satisfies the equation
		\begin{equation}\label{tilde_u_equation}
			\left\{
			\begin{aligned}
				\partial_{y_i}(a^{ij}\partial_{y_j}\tilde u)&=0  &&\mbox{in }Q_{1.9r, r^2},\\
				\tilde u+\gamma \,F_1^{-1} \partial_{y_n}\tilde u&= \tilde g_1 &&\mbox{on } \{y_n=r^2\}\cap \overline{Q_{1.9r, r^2}},\\
				\tilde u-\gamma \, F_2^{-1} \partial_{y_n}\tilde u&= \tilde g_2 &&\mbox{on } \{y_n=-r^2\}\cap \overline{Q_{1.9r, r^2}},
			\end{aligned}
			\right.
		\end{equation}
		where 
		$a:=(a^{ij})_{n\times n}=(\nabla \Phi)^{-1}((\nabla\Phi)^{-1})^T{|\det(\nabla \Phi)|}\in C^{0,1}(\overline{Q_{1.9r, r^2}}:\mathbb{R}^{n\times n})$ satisfies
		\begin{equation}\label{est:a}
			\frac{I}{C} \le a \le C I, \quad |\nabla_y a| \le \frac{C}{r},
		\end{equation}
		and
		\begin{equation}\label{ortho}
			{a^{nj}=a^{jn}}=0 \quad \mbox{on } \{y_n=\pm r^2\}\cap \overline{Q_{1.9r, r^2}}
		\end{equation}
		for any  $j\in \{1,2,\ldots, n-1\}$,
		and $F_i\in C^{0,1}(\overline{Q_{1.9r, r^2}})\cap C^{1,1}(Q_{1.9r, r^2})$ ($i=1,2$) are defined as
		\begin{equation*}
			\left\{
			\begin{aligned}
				&F_1(y):=\frac{1}{2r^2}(\varepsilon+h_1(y')-h_2(y'))\sqrt{1+|\nabla_{y'} \widehat h_1(y)|^2},\\
				&F_2(y):=\frac{1}{2r^2}(\varepsilon+h_1(y')-h_2(y'))\sqrt{1+|\nabla_{y'} \widehat h_2(y)|^2},
			\end{aligned}
			\right.
		\end{equation*}
		$\widehat{h}_i$ denotes the mollification defined in \eqref{mollification}. It follows from \eqref{est:mollification} that 
		\begin{equation}\label{F_bound}
			\left\{
			\begin{aligned}
				&\frac{1}{C}\le F_i \le C, \quad |\nabla_{y'} F_i|\le \frac{C}{r}, \quad |\partial_{y_n} F_i|\le C\, r^2,\\
				& |\nabla^2_{y'}F_i(y)|\le \frac{C\,r^2}{r^4-y_n^2},\quad |\partial_{y_n}\nabla_{y'}F_i(y)|\le \frac{C\,r^3}{r^4-y_n^2}, \quad 
				|\partial^2_{y_n}F_i(y)|\le \frac{C\,r^4}{r^4-y_n^2},
			\end{aligned}
			\right.
		\end{equation}
		Furthermore, we have
		\begin{equation}\label{est:a2}
			{a^{ii} \sim 1, \qquad |a^{ij}|} \le Cr \quad \mbox{for}~i \neq j. 
		\end{equation}
		
		\textbf{Step 3:} {Reduce to conormal boundary conditions}.
		
		Let $\tilde v= \tilde u \, e^{\psi}$, where $\psi=\psi(y)$ is a function to be determined. Note that by the chain rule, 
		$$\partial_{y_n}\tilde v=e^{\psi}(\partial_{y_n}\tilde u+\tilde u\,\partial_{y_n} \psi).$$
		We choose $$\psi=(y_n+r^2)(y_n-r^2)(\tilde{A}\,y_n+\tilde{B}),$$ where $\tilde A$ and $\tilde B$ are functions such that
		\begin{equation*}
			\left\{
			\begin{aligned}
				&\partial_{y_n}\psi(y',r^2)=2r^2(\tilde A r^2+\tilde B)=\gamma^{-1} F_1,\\
				&\partial_{y_n}\psi(y',-r^2)=-2r^2(-\tilde A r^2+\tilde B)=-\gamma^{-1} F_2,
			\end{aligned}
			\right.
		\end{equation*}
		namely, 
		$$\tilde A:=(4\gamma r^4)^{-1}(F_1-F_2) \quad \mbox{and}\quad \tilde B:=(4\gamma r^2)^{-1}(F_1+F_2).$$
		The equation \eqref{tilde_u_equation} becomes
		\begin{equation}\label{tilde_v_equation}
			\left\{
			\begin{aligned}
				\partial_{y_i}(a^{ij}\partial_{y_j}\tilde v) + b^i \partial_{y_i}\tilde v + c \tilde v &=0  &&\mbox{in }Q_{1.9r, r^2},\\
				\partial_{y_n}\tilde v&= e^\psi \frac{F_1}{\gamma} \tilde g_1 &&\mbox{on } \{y_n=r^2\}\cap \overline{Q_{1.9r, r^2}},\\
				\partial_{y_n}\tilde v&= -e^\psi \frac{F_2}{\gamma} \tilde g_2 &&\mbox{on } \{y_n=-r^2\}\cap \overline{Q_{1.9r, r^2}},
			\end{aligned}
			\right.
		\end{equation}
		where
		$$
		b^i = - 2a^{ij} \partial_{y_j} \psi,
		$$
		and
		$$
		c = -(\partial_{y_i}a^{ij})(\partial_{y_j} \psi) + a^{ij} (\partial_{y_i} \psi) (\partial_{y_j} \psi) - a^{ij} \partial_{y_i y_j} \psi.
		$$
		In light of \eqref{ortho}, the boundary conditions in \eqref{tilde_v_equation} can also be written as
		$$
		a^{nj} \partial_{y_j}\tilde v =  a^{nn} e^\psi \frac{F_1}{\gamma} \tilde g_1 \quad \mbox{on } \{y_n=r^2\}\cap \overline{Q_{1.9r, r^2}},
		$$
		and
		$$
		a^{nj} \partial_{y_j}\tilde v =  -a^{nn} e^\psi \frac{F_2}{\gamma} \tilde g_2 \quad \mbox{on } \{y_n=-r^2\}\cap \overline{Q_{1.9r, r^2}}.
		$$
		{Following the same computations} as in \cite{DongYangZhu26}, we have
		\begin{equation}\label{est:psi}
			|\psi| \le Cr^2, \quad |\nabla_y \psi| \le C, \quad |\nabla_{y'}^2 \psi| \le C, \quad |\nabla_{y'}\partial_{y_n} \psi| \le Cr^{-1}, \quad |\partial_{y_n}^2 \psi| \le C r^{-2}.
		\end{equation}
		Furthermore, by a direct computation, we have
		$$
		{ \partial_{y_n}^2 \psi  = 6 \tilde A y_n + 2\tilde B + 4y_n (y_n\partial_{y_n} \tilde A  + \partial_{y_n} \tilde B) + (y_n^2 - r^4) (2 \partial_{y_n} \tilde A + \partial_{y_n}^2 \tilde A y_n + \partial_{y_n}^2 \tilde B)}.
		$$
		{
			By \eqref{est:a}, \eqref{F_bound}, \eqref{est:a2}, and \eqref{est:psi}, we can conclude that the leading order term from $c$, denoted by $c_L$, is
			$$
			c_L:= - 2 a^{nn} \tilde B.
			$$
			It satisfies
			$$
			{\frac{1}{C\gamma r^2} \le -c_L \le \frac{C}{\gamma r^2}} \quad \mbox{and} \quad |\nabla c_L| \le \frac{C}{\gamma r^3} .
			$$
			The remaining terms from $c$, denoted by $c_R:= c- c_L$, satisfy
			$$
			|c_R| \le Cr^{-1}.
			$$
			Furthermore, direct computation yields
			$$
			|b| \le C.
			$$
		}
		
		\textbf{Step 4:} Extend and remove the discontinuities of the co-normal boundary conditions.
		
		We define
		$$
		\mathcal{Q}_k : =  \{ y = (y', y_n) \in \mathbb{R}^n:~ |y'-x_0'| < 1.9r,\, (2k-1)r^2 < y_n < (2k+1)r^2 \}
		$$
		for $k \in \mathbb{Z}$,
		and denote the upper boundary of $\mathcal{Q}_k$ by
		$$
		\Gamma_k := \{ y = (y', y_n) \in \mathbb{R}^n:~ |y'-x_0'| < 1.9r,\, y_n = (2k+1)r^2 \}.
		$$
		Now for $i,j = 1,2,\ldots, n-1$, we extend $ a^{ij}$, $ a^{nn}$, $ b^i$, $c$, and $\tilde{v}$ evenly with respect to $y_n = r^2$, $ a^{in}$, $ a^{ni}$, and $b^n$ oddly with respect to $y_n = r^2$, and then periodically along $y_n$ so that the period is equal to $4r^2$. Such extension creates discontinuities of the co-normal derivatives across interfaces $\Gamma_k$. We adapt the idea in \cites{DongXu1,DongXu2} to remove the discontinuities. We denote
		$$
		[a^{nj} \partial_{y_j}\tilde v]_{\Gamma_k}:= a^{nj} \partial_{y_j}\tilde v|_{\Gamma_k}^+ - a^{nj} \partial_{y_j}\tilde v|_{\Gamma_k}^-
		$$
		to be the difference between the {upper and lower traces} of the co-normal derivative of $\tilde{v}$ across $\Gamma_k$. Then it follows from \eqref{tilde_v_equation} that $\tilde{v}$ satisfies
		\begin{equation}\label{tilde_v_equation_extend}
			\left\{
			\begin{aligned}
				\partial_{y_i}(a^{ij}\partial_{y_j}\tilde v) + b^i \partial_{y_i}\tilde v + c \tilde v &=0  &&\mbox{in }\cup_{k=- \infty}^\infty \mathcal{Q}_k,\\
				[a^{nj} \partial_{y_j}\tilde v]_{\Gamma_k}&= \mathcal{G}_k &&\mbox{on } \Gamma_k,
			\end{aligned}
			\right.
		\end{equation}
		where
		$$
		\mathcal{G}_k(y') = \left\{
		\begin{aligned}
			&-2a^{nn} e^\psi \frac{F_1}{\gamma} \tilde g_1, &&\mbox{when}~k~\mbox{is even},\\
			&-2a^{nn} e^\psi \frac{F_2}{\gamma} \tilde g_2, &&\mbox{when}~k~\mbox{is odd}.
		\end{aligned}
		\right.
		$$
		For $m \sim \frac{1}{r}$, we define
		$$
		f^i(y):=  \delta_{in} \sum_{k=-m}^m \mathbbm{1}_{\{y_n > (2k{+}1)r^2\}} \mathcal{G}_k(y') \quad \mbox{for}~i=1,\ldots,n,
		$$
		where $\delta$ is the  Kronecker delta and $\mathbbm{1}_{\bullet}$ is the indicator function. Then it follows from \eqref{tilde_v_equation_extend} that $\tilde{v}$ satisfies
		\begin{equation}\label{tilde_v_equation_extend_new}
			\left\{
			\begin{aligned}
				\partial_{y_i}(a^{ij}\partial_{y_j}\tilde v - f^{i}) + b^i \partial_{y_i}\tilde v + c \tilde v &=0  &&\mbox{in }\cup_{k=-m}^m \mathcal{Q}_k,\\
				[a^{nj} \partial_{y_j}\tilde v - f^{{n}}]_{\Gamma_k}&= 0 &&\mbox{on } \Gamma_k.
			\end{aligned}
			\right.
		\end{equation}

	\textbf{Step 5:} Rescale and estimate.
		
		We consider the following rescaling
		{
			$$
			\begin{aligned}
				&\bar{v}(y)=\tilde v(x_0'+ry',ry_n), \quad \bar{a}(y)=a(x_0'+ry',ry_n), \quad \bar{b}(y)=r b(x_0'+ry',ry_n),\\
				&\bar{c}(y)=r^2 c(x_0'+ry',ry_n), \quad \bar{c}_L(y)=r^2 c_L(x_0'+ry',ry_n), \\
				&\bar{c}_R(y)=r^2 c_R(x_0'+ry',ry_n), \quad
				\bar{f}(y)=r f(x_0'+ry',ry_n).
			\end{aligned}
			$$
			Then by \eqref{tilde_v_equation_extend_new} $\bar{v}$ satisfies the equation
			\begin{equation}\label{eqn:extend}
				\partial_{y_i}(\bar a^{ij}\partial_{y_j}\bar v - \bar f^i) +\bar b^i \partial_{y_i}\bar v + \bar c \bar v =0  \quad \mbox{in}~ {B_{1.9}}:=\{ |y|< 1.9 \},
			\end{equation}
			with the coefficients satisfying
			\begin{equation}\label{est:coefficient}
				\begin{aligned}
					&\frac{I}{C} \le \bar a \le C I, \quad |\nabla_y \bar a| \le C, \quad |\bar{b}| \le Cr, \\
					&\frac{1}{C\gamma} \le -\bar{c}_L \le \frac{C}{\gamma}, \quad |\bar c_R| \le Cr, \quad \mbox{and} \quad |\nabla_y \bar c_L| \le \frac{C}{\gamma}.
				\end{aligned}
			\end{equation}
			By \cite{Dong2012}*{Theorem 2 and Remark 3}, we have
			\begin{equation*}\label{est:2012a}
				\begin{aligned}
					&
                    \| \bar{v}\|_{W^{1,\infty}({B_{1.1}})} 
					+ [\nabla \bar{v}]_{C^\beta_{{y'}}(B_{1.1})}
					+ [\nabla \bar{v}]_{C^\beta({\mathcal{Q}_0})}
					\le C(\|\bar{v} \|_{L^2(B_{1.9})} + \| \bar f \|_{C_{y'}^\beta(B_{1.9})})\\
					&\le C (\|\bar{v} \|_{L^2(B_{1.9})} + \| \bar f \|_{C_{y'}^\beta(B_{1.9})}.
				\end{aligned}
                \end{equation*}
			Since $\bar{f}(0',y_n) = 0$, we have
			\begin{equation}\label{est:f_bar}
				\| \bar{f}\|_{L^\infty(B_{1.9})} + \| \bar f \|_{C_{y'}^\beta(B_{1.9})}  \le C r^\beta ([g_1]_{C^\beta_{x'}} + [g_2]_{C^\beta_{x'}}) {+ Cr^2|g_1(x_0')-g_2(x_0')|}.    
			\end{equation}
		{Reversing the scaling and the change of variables gives}
		$$
		\begin{aligned}
			&\|\hat u\|_{L^\infty(\Omega_{x_0,r})} +r\|\nabla\hat u\|_{L^\infty(\Omega_{x_0,r})} + r^{1+\beta} [\nabla \hat u]_{C^\beta(\Omega_{x_0,r})} \\
			&\le  C \left( \nparallel\kern-.2em \hat u  \Vert_{L^2(\Omega_{x_0,2r})} + r^{\beta}([g_1]_{C^\beta_{x'}} + [g_2]_{C^\beta_{x'}})+ r^2|g_1(x_0')-g_2(x_0')| \right).
		\end{aligned}
		$$
		From \eqref{est:ab}, we can compute that 
		$$\|\nabla(Ax_n + B)\|_{L^\infty({\Omega_{x_0,r}})} \le C |g_1(x_0')|,$$ and 
        $$\nparallel\kern-.2em Ax_n + B \Vert_{L^2(\Omega_{x_0,2r})} \le C r^2|g_1(x_0')|.$$
        Finally, 
        $$
        |g_i(x_0') - g_i(0')| \le r^{\beta}[g_i]_{C^\beta_{x'}}, \quad i = 1,2.
        $$
        Hence \eqref{gradient-1/2a} follows.
        
Next, we show \eqref{gradient-1/2}. By the triangle inequality,
            \begin{equation}\label{est:2012}
				\begin{aligned}
					&\|\bar{v}\|_{W^{1,\infty}({B_{1.1}})} 
					+ [\nabla \bar{v}]_{C^\beta_{{y'}}(B_{1.1})}
					+ [\nabla \bar{v}]_{C^\beta({\mathcal{Q}_0})}
					\le C(\|\bar{v} \|_{L^2(B_{1.9})} + \| \bar f \|_{C_{y'}^\beta(B_{1.9})})\\
					&\le C (\|\bar{v} - (\bar{v})_{B_{1.1}}\|_{L^2(B_{1.9})} + \| \bar f \|_{C_{y'}^\beta(B_{1.9})} + |(\bar{v})_{B_{1.1}}|).
				\end{aligned}
			\end{equation}
			We can also estimate
			\begin{equation}\label{est:mean-oscillation}
				\begin{aligned}
					&\| \bar{v} - (\bar{v})_{B_{1.1}}\|_{L^2(B_{1.9})}\\ & = \left( \int_{B_{1.9}} \big| \fint_{B_{1.1}} \bar{v}(y) - \bar{v}(x) \, dx  \big|^2 \, dy \right)^{1/2}\\
					&\le C \left( \int_{B_{1.9}}  \fint_{B_{1.9}} \big|{\bar{v}(y)-\bar{v}(x)}\big|^2 \,dx\,dy \right)^{1/2}\\

			\begin{aligned}
				L_0 V_{0}:= V_{0}''(r) +\left(\frac {n-2}r +\frac{2r}{\varepsilon+r^2} \right)V_{0}'(r)- \frac{2}{\gamma(\varepsilon + r^2)}  V_{0}(r)=0 \quad \mbox{in}~~(0,R_0).
			\end{aligned}

		Then it follows that $v_0(y):=V_0(|y|)$ is a radial solution of \eqref{equation_v1}. 
		Let $\rho\in (0,R_0)$.
		By the weak maximum principle for $v_0$ in $B_\rho$, we know that $$0\le v_0(0)\le \underset{\partial{B_\rho}}{\max} \,v_0.$$
		This directly implies that
		\begin{equation}\label{V0-lower}
			V_0(\rho) - V_0(0)\ge0, \quad \rho\in(0,R_0).
		\end{equation}
		Next, we prove the upper bound for $V_0(\rho)-V_0(0)$. First, we show that $w(r):=(\varepsilon+r^2)^{\alpha_0/2}$ is a supersolution. By direct computations,
		\begin{equation*}
			w'(r) =  \alpha_0 r (\varepsilon + r^2)^{\alpha_0/2-1},
		\end{equation*}
		\begin{equation*}
			w''(r) = \alpha_0 (\varepsilon + r^2)^{\alpha_0/2-1} +  \alpha_0 (\alpha_0 - 2) r^2 (\varepsilon + r^2)^{\alpha_0/2-2}.
		\end{equation*}
		Thus
		\begin{align*}
			L_0 w&=\alpha_0^2r^2(\varepsilon + r^2)^{\alpha_0/2-2}+\big((n-1)\alpha_0-2\gamma^{-1}\big)(\varepsilon + r^2)^{\alpha_0/2-1}
			\\&=-\alpha_0^2\varepsilon(\varepsilon + r^2)^{\alpha_0/2-2}\le 0,
		\end{align*}
		where in the last line, we used the fact that 
		$$\alpha_0^2+(n-1)\alpha_0-2\gamma^{-1}=0$$
		in view of the definition of $\alpha_0$ in \eqref{alpha_new_zero}.

		\begin{equation*}
			W(r) := V_0(0) + B\,w(r)=V_0(0) + B(\varepsilon + r^2)^{\alpha_0/2},
		\end{equation*}
		where the constant $B$ is defined as
		$$B:=\frac{V_0(R) - V_0(0)}{(\varepsilon + R^2)^{\alpha_0/2}}\ge 0.$$

			{L_0}W = - \frac{2}{\gamma(\varepsilon + r^2)}  V_{0}(0)+B\,L_0w\le 0=L_0 V_0 \quad \mbox{in}~~(0,R_0) .

		is a bounded particular solution to ${L}p_1=H_1$. Using \eqref{ABD_bounds}, we can construct and estimate $|p_2| \le Cr^{1+\min\{\alpha,\beta\}}$ as in the case $\gamma>1$. Then there exists a constant $C_1(\varepsilon)$ such that
		$$
		V_{1,1}(r) = p_1(r) + p_2(r) + C_1(\varepsilon)r.
		$$
		Plugging in $r = 1/2$, we can see that $|C_1(\varepsilon)| \le C$ independent of $\varepsilon$. Therefore, $p_1(r)$ is the leading order term. Namely,
		\begin{equation}\label{V11_lower_2}
		\begin{aligned}
		V_{1,1}(r)&\ge p_1(r)-Cr^{1+\beta}-Cr\\
		&\ge -\frac{{c_n}((1+k_1)^{-1}+(1+k_2)^{-1})}{4\gamma(n+1)}r\log(\varepsilon+r^2),\\
		&\hspace{2em} r\in(0,r_1),\quad {0<\varepsilon<r_1^2}.
		\end{aligned}
		\end{equation}
		where $r_1=r_1(n,\gamma,k_1,k_2)$ is a positive constant independent of $\varepsilon$ {and in the last inequality we used \eqref{est:g_x1_lower}}.
		By \eqref{v_bar_expansion} and \eqref{V11_lower_2}, we have
		\begin{align*}
			\left( \int_{\mathbb{S}^{n-2}} |\bar{v}(\sqrt{\varepsilon},\xi)|^2 \, d\xi \right)^{1/2} &\ge |V_{1,1}(\sqrt{\varepsilon})| \ge -\frac{1}{C} \sqrt{\varepsilon}\log(\varepsilon)\quad \forall 0 < \varepsilon < \varepsilon_0
		\end{align*}
		for some small positive $\varepsilon_0$ depending only on $n$, $\gamma$, $k_1$, $k_2$. Then {arguing as in the case when $\gamma>1/\mu$,}
		{we conclude the proof} when $n \ge 3$.
		
		When $n = 2$, by Lemma \ref{lemma_reduction}, $\bar{v}$ satisfies the equation
		$$
		((\varepsilon+|y_1|^2)\bar v')'- 2\gamma^{-1} \bar{v} = F' + G_1+G_2\quad \mbox{in}~(-3/4,3/4),
		$$
		where $F$ and $G_1$ satisfy the same bounds as in \eqref{FG_bound_3}. By using $\bar{v}$ in place of $V_{1,1}$ and repeating steps 2-4 with minor modifications, we can establish the estimate \eqref{grad_u_lower_bound_new} for $n = 2$.
