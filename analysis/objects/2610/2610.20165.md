<!-- defn-1: body_md -->
\label{def:int-Hil}
The ring ${\mathcal Z}$ is said to be \emph{integrally Hilbertian} if for all $k,n,s, \underline T,\underline Y$ and $P_1,\ldots,P_s$ as above, the following subset is Zariski-dense:
\vskip 1mm

\centerline{${\mathcal H}_{\mathcal Z}(P_1,\ldots,P_s) = \left\{\underline{t} \in {\mathcal Z}^k \hskip 1pt \Bigg\vert \hskip 1pt 
\left\{\begin{matrix}
P_1(\underline{t},\underline{Y}),\dots,P_s(\underline{t},\underline{Y}) \hbox{\it \ are irreducible in } \mathcal{Q}[\underline{Y}], \hbox{\ \it and}\hfill \\
\hbox{\it the product } \prod_{i=1}^s P_i(\underline t,\underline Y) \hbox{\it  \ has no nonunit divisor in } {\mathcal Z}. \hfill \\
\end{matrix}
\right.
\right\}$.}

<!-- thm-1: body_md -->
\label{thm:denposHilIrrThm} Let ${\mathcal Z} ={\mathcal O}_K$ be the ring of integers of a global field $K$. Let $P_1,\ldots,P_s \in \mathcal{Z}[\underline{T},\underline{Y}]$ be some polynomials, irreducible in $K[\underline{T},\underline{Y}]$ ($s\geqslant 1$), of degree $\geqslant 1$ in $\underline{Y}$ and such that the product $P=P_1 \cdots P_s$ has no {fixed divisor \hbox{w.r.t.} $\underline{T}$}. Then the set ${\mathcal H}_{{\mathcal Z}}(P_1,\ldots,P_s)$ contains a subset of $({\mathcal O}_K)^k$ of positive density.

\label{thm:ekpoongen} Let $K$ be a global field and $Q_1,\ldots,Q_\ell \in {\mathcal O}_K[\underline T]$ be $\ell\geqslant 2$ nonzero polynomials, coprime in $K[\underline{T}]$. Then we have:

\mu(\mathcal{R}^\star)=\prod_{\mathfrak{p}\in ({\rm Spec}\hskip 1pt \mathcal{O}_K)^\ast} \left( 1- \frac{c_\mathfrak{p}}{{\rm N}(\mathfrak{p})^{k}} \right).

<!-- defn-2: body_md -->
\label{defn:density}
The \emph{upper density} (resp., \emph{lower density}) of $\mathcal{A} \subset \mathcal{O}_K^k$ is given by the limit
$$
\mu_{{\rm sup}}(\mathcal{A})=\limsup_{B \rightarrow \infty}\frac{\# \mathcal{A}_B}{\#\mathcal{O}^k_{K,B}} \hskip 3mm ( \textrm{resp.},\, \mu_{{\rm inf}}(\mathcal{A})=\liminf_{B \rightarrow \infty}\frac{\# \mathcal{A}_B}{\#\mathcal{O}^k_{K,B}}).
$$ If $\mu_{{\rm sup}}(\mathcal{A})=\mu_{{\rm inf}}(\mathcal{A})$, the set $\mathcal{A}$ is said to \emph{have a density}, given by
$$
\mu(\mathcal{A})=\lim_{B \rightarrow \infty}\frac{\# \mathcal{A}_B}{\#\mathcal{O}^k_{K,B}}.  
$$

<!-- thm-3: body_md -->
\label{prop:crucari} 
\label{lem:bounheigharithprog} Let $\mathfrak{p}$ be a nonzero ideal of $\mathcal{O}_K$ and let $a \in \mathcal{O}_K$.

(2) The subset $\mathfrak{p}+a \subset {\mathcal O}_K$ has density

<!-- cor-1: body_md -->

$$

$$

D_{\varepsilon,B}=

\#(\mathfrak{p}+a)_B \leqslant D_{\varepsilon,B},
$$
for all nonzero ideal $\mathfrak{p} \subset \mathcal{O}_K$ with

\begin{equation}\label{eq:majexpN}
\frac{1}{{\rm N}(\mathfrak{p})^{1/\mathfrak{d}}} \leqslant \frac{\deg(\mathfrak{p})}{{\rm N}(\mathfrak{p})^{1/\mathfrak{d}}} =O\left(\frac{1}{{\rm N}(\mathfrak{p})^{\varepsilon}}\right).
\end{equation}
Then, by Theorem \ref{prop:crucari} and \eqref{eq:majexpN}, there exist constants $c_{K,\varepsilon}^{(1)},c_{K,\varepsilon}^{(2)},c_{K,\varepsilon}^{(3)}>0$ such that
\begin{align*}
                                
\# (\mathfrak{p}+a)_B \leqslant D_{\varepsilon,B}&:=
c_{K,\varepsilon}^{(1)} B(\log_q(B))^{\#\mathcal{S}_\infty-1-\frac{1}{1-\varepsilon}}
+c_{K,\varepsilon}^{(2)} (\log_q(B))^{\#\mathcal{S}_\infty-\frac{1}{1-\varepsilon}}+c_{K,\varepsilon}^{(3)}B(\log_q(B))^{\#\mathcal{S}_\infty-2-\frac{\varepsilon}{1-\varepsilon}}.
\end{align*}
 whenever ${\rm N}(\mathfrak{p}) \geqslant \theta_{\varepsilon}(B) \geqslant 2$. Moreover, $$
 D_{\varepsilon,B}=O_{K,\varepsilon}\left( B(\log_q(B))^{\#\mathcal{S}_\infty-1-\frac{1}{1-\varepsilon}}\right),
 $$ and hence
\vskip 1mm
\hskip 20mm
$$ \frac{D_{\varepsilon,B}}{\#\mathcal{O}_{K,B}}=
O_{K,\varepsilon}(\log_q(B)^{-\frac{1}{1-\varepsilon}}).
$$ 
\noindent

<!-- lem-2: body_md -->
\label{lem:volsm}
The volume of $\mathscr{S}_\mathfrak{b}$ is
$$
{\rm Vol}_{s-1}(\mathscr{S}_\mathfrak{b})=\frac{\Vert \underline{\xi} \Vert \mathfrak{b}^{s-1}}{(s-1)! \xi_1 \cdots \xi_s}.
$$
<!-- end lem-2 -->
<!-- proof-4: body_md -->
Define 
$$
f(x_1,\dots,x_{s-1})=\frac{\mathfrak{b}}{\xi_s}-\frac{\xi_1}{\xi_s}x_1-\dots-\frac{\xi_{s-1}}{\xi_s}x_{s-1}.
$$
The volume of $\mathscr{S}_\mathfrak{b}$ is obtained by integrating the function
$$
\sqrt{1+\left(\frac{\partial f}{\partial x_1}\right)^2+\dots+\left(\frac{\partial f}{\partial x_{s-1}}\right)^2}=\frac{\Vert \underline{\xi} \Vert}{\xi_s}
$$
over a certain coordinate domain. More precisely,
$$
{\rm Vol}_{s-1}(\mathscr{S}_\mathfrak{b})=\frac{\Vert \underline{\xi} \Vert}{\xi_s} 
\int_{0}^{\frac{\mathfrak{b}}{\xi_1}} \int_{0}^{\frac{\mathfrak{b}}{\xi_2}-\frac{\xi_1}{\xi_2}x_1}\cdots \int_{0}^{\frac{\mathfrak{b}}{\xi_{s-1}}-\frac{\xi_1}{\xi_{s-1}}x_1 -\cdots-\frac{\xi_{s-2}}{\xi_{s-1}}x_{s-2}}d{x_{s-2}}\cdots dx_1.
$$
Denote this last integral by $I_\mathfrak{b}(\xi_1,\dots,\xi_s)$ so that
$$
{\rm Vol}_{s-1}(\mathscr{S}_\mathfrak{b})=\frac{\Vert \underline{\xi} \Vert}{\xi_s} I_\mathfrak{b}(\xi_1,\dots,\xi_s).
$$
We need to prove that
$$
I_\mathfrak{b}(\xi_1,\dots,\xi_s)=\frac{\mathfrak{b}^{s-1}}{(s-1)! \xi_1 \cdots \xi_{s-1}}.
$$
This follows  by induction on $s$, using the recursion
$$
I_\mathfrak{b}(\xi_1,\dots,\xi_s)=\int_0^{\frac{\mathfrak{b}}{\xi_1}}I_{\mathfrak{b}-\xi_1 x_1}(\xi_2,\dots,\xi_s)\, dx_1,
$$
with base case ($s=2$) $I_\mathfrak{b}(\xi_1,\xi_2)=\int_0^{\frac{\mathfrak{b}}{\xi_1}}dx_1=\frac{\mathfrak{b}}{\xi_1}$.
