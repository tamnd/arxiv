\begin{equation}\label{holdecomN}
    N=\bigoplus_{0<\theta<2\pi} N_{\theta}
\end{equation}
where $g$ acts by $e^{i\theta}$ on $N_{\theta}$.  In the analytic language we need to pass to the real normal bundle,  its decomposition into real $g$-invariant  blocks is
\begin{equation}\label{realdecomN}
    E=N_{\mathbb{R}}=E_g(-1) \bigoplus_{0<\xi<\pi} E_c(e^{i\xi})
\end{equation}
where $g$ acts by $e^{i\xi}$ on $E_c(e^{i\xi})$. $E_c(e^{i\xi})$ (including $E_c(e^{i\pi}):=E_g(-1)$) is even-dimensional and admits a natural $G$-invariant complex structure, we call this the \textit{analytic} complex structure. For the $(-1)$-eigenspace we use the holomorphic complex structure inherited from $N_\pi$ and 
\[
E_c(e^{i\pi}):=E_c(-1)=N_\pi
\] For ${0<\xi<\pi}$, the relation between the analytic complex structure and the holomorphic complex structure is
\[E_c(e^{i\xi})=N_{\xi} \oplus \overline{N_{2\pi-\xi}}\]

\[\frac{ch_g(\lambda_1N_\xi)}
     {ch_g(\lambda_{-1}N_\xi)}
=
(-1)^{r_\xi}
\frac{ch_g(\lambda_1N_\xi^\vee)}
     {ch_g(\lambda_{-1}N_\xi^\vee)}\]
\[\frac{ch_g(\lambda_1\overline{N_{2\pi-\xi}})}
     {ch_g(\lambda_{-1}\overline{N_{2\pi-\xi}})}
=
\frac{ch_g(\lambda_1N_{2\pi-\xi}^\vee)}
     {ch_g(\lambda_{-1}N_{2\pi-\xi}^\vee)}\]
Hence
\[\frac{ch_g(\lambda_1E_c(e^{i\xi}))}
     {ch_g(\lambda_{-1}E_c(e^{i\xi}))}
=
(-1)^{r_\xi}
\frac{ch_g(\lambda_1N_\xi^\vee)}
     {ch_g(\lambda_{-1}N_\xi^\vee)}
\frac{ch_g(\lambda_1N_{2\pi-\xi}^\vee)}
     {ch_g(\lambda_{-1}N_{2\pi-\xi}^\vee)}\]

\frac{ch_g(\lambda_1E_c(e^{i\pi}))}
     {ch_g(\lambda_{-1}E_c(e^{i\pi}))}
=
(-1)^{r_\pi}
\frac{ch_g(\lambda_1N_\pi^\vee)}
     {ch_g(\lambda_{-1}N_\pi^\vee)}
\end{equation}

r_+:=\sum_{0<\xi<\pi}\operatorname{rk}_{\mathbb C}N_\xi,
\qquad
r_-:=\sum_{\pi<\theta<2\pi}\operatorname{rk}_{\mathbb C}N_\theta
\]

\frac{ch_g(\lambda_1E_c(e^{i\xi}))}

(-1)^{r_++r_\pi}
\frac{ch_g(\lambda_1N^\vee)}
     {ch_g(\lambda_{-1}N^\vee)}
\end{equation}

Hence the analytic orientation of $E$ differs from its holomorphic orientation by
$(-1)^{r_-}$. As mentioned in Remark \ref{comatbileoreination} the orientation on $X^g$ is induced by that of $X$ and $E$; 
\[TX|_{X^g}\simeq TX^g \oplus E\]

\begin{equation*}
L_*(X^g)
=
(-1)^{r_-}L_*(X_{\mathrm{an}}^g)
=
(-1)^{r_-}\Psi_2L^{an}_*(X_{\mathrm{an}}^g)
\end{equation*}
By assumption,
\[
IT_{1*}(X^g)=L_*(X^g)
\]
\begin{equation*}
\Psi_2L^{an}_*(X_{\mathrm{an}}^g)
=
(-1)^{r_-}IT_{1*}(X^g)
\end{equation*}

\begin{equation*}
IT_{1*}(X,g)
=
\frac{ch_{(2),g}(\lambda_1N^\vee)}
     {ch_{(2),g}(\lambda_{-1}N^\vee)}
\cap IT_{1*}(X^g)
\end{equation*}
By Corollary \ref{chernlocal},
\[
L^{an}_*(X,g)
=
(-1)^{\dim_{\mathbb C}N}
\prod_{0<\xi\leq\pi}
\frac{ch_g(\lambda_1E_c(e^{i\xi}))}
     {ch_g(\lambda_{-1}E_c(e^{i\xi}))}
\cap L^{an}_*(X_{\mathrm{an}}^g)
\]
Therefore by (\ref{total-normal-factor}) and $r_++r_-+r_\pi=\dim_{\mathbb C}N$ we get
\[
\begin{aligned}
\Psi_2L^{an}_*(X,g) &= (-1)^{\dim_{\mathbb C}N+r_++r_\pi} \frac{ch_{(2),g}(\lambda_1N^\vee)} {ch_{(2),g}(\lambda_{-1}N^\vee)} \cap \Psi_2L^{an}_*(X_{\mathrm{an}}^g) \\
&= (-1)^{\dim_{\mathbb C}N+r_++r_\pi+r_-} \frac{ch_{(2),g}(\lambda_1N^\vee)} {ch_{(2),g}(\lambda_{-1}N^\vee)} \cap IT_{1*}(X^g) \\
&= \frac{ch_{(2),g}(\lambda_1N^\vee)} {ch_{(2),g}(\lambda_{-1}N^\vee)} \cap IT_{1*}(X^g) \\
&= IT_{1*}(X,g)
\end{aligned}
\]

We make some remarks on how some of the inputs in the proof of \cite[Prop.~6.20]{B} can be seen equivariantly. \cite[Lemma~6.19]{B} implies that there is an isomorphism $\alpha:i^*IC_X^H[-c]\xrightarrow{\cong}IC_Y^H$ in $D^b(\mathrm{MHM}(Y))$, where $c$ is the complex codimension of $Y/X$. Instead, we construct an isomorphism in $D^{b,G}(\mathrm{MHM}(Y))$ as follows. Let $U\xrightarrow{j}X$ and $V=U\cap Y\xrightarrow{j_V}Y$ be the Zariski-open  dense smooth algebraic subvarieties of $X$ and $Y$ appearing in loc. cit. Set $U_G:=\bigcap_{h\in G}h.U$ and $V_G:=\bigcap_{h\in G}h.V$. Since $G$ is finite, $U_G$ and $V_G$ are again smooth dense Zariski-open subsets, and they are $G$-invariant. By functoriality of the intermediate extension, $a^{-1}$ induces an automorphism $j_{V_G!*}(a^{-1}):IC_Y^H\xrightarrow{\cong}IC_Y^H$. Replacing $\alpha$ by $\widetilde{\alpha}:=j_{V_G!*}(a^{-1})\circ\alpha$, we may therefore assume that, under the canonical identifications above, $\widetilde{\alpha}|_{V_G}=\operatorname{id}_{\mathbb{Q}_{V_G}^H[d_Y]}$. Then $\widetilde{\alpha}$ is $G$-equivariant. Indeed, for every $g\in G$, the $g$-translate $g\cdot\widetilde{\alpha}:=\psi^{-1}_g\circ g_*(\widetilde{\alpha})\circ \psi_g$ is another isomorphism $i^*IC_X^H[-c]\xrightarrow{\cong}IC_Y^H$ whose restriction to $V_G$ is the identity. Hence $(g\cdot\widetilde{\alpha})\widetilde{\alpha}^{-1}$ is an automorphism of $IC_Y^H$ which restricts to the identity on $\mathbb{Q}_{V_G}^H[d_Y]$. Since $IC_Y^H$ has no nonzero subobject supported on $Y\setminus V_G$, such an automorphism must be the identity. Thus $g\cdot\widetilde{\alpha}=\widetilde{\alpha}$ for every $g\in G$, and therefore $\widetilde{\alpha}$ is $G$-equivariant. Another input is \cite[Lemma~6.13]{B}, which says that the smooth pullback of (shifted) $IC^H$ is $IC^H$. This is induced by a natural morphism of functors that are all $G$-equivariant, since all the maps in the deformation to the normal cone are \(G\)-equivariant. Since the forgetful functor is conservative, this implies that the equivariant map is also an isomorphism.
