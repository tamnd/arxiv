<!-- remark-1: body_md -->
\label{rmk-iso}
		In the algebraic setting, the role of the Atiyah extension $A(E)$ is played by the sheaf $\mathcal D^1(E)$ of scalar differential operators of order at most one on $E$ (see \cite{Ser06} for the line bundle case and \cite{Mar09} for the general case). Recall that a \textit{differential operator of order $1$} on a holomorphic vector bundle $E$ of rank $r$ is a linear map $P\colon \Omega^0(E)\to \Omega^0(E)$ which, locally, takes the form
		\[
		P = (g_{ij}) + \sum_{k=1}^n (h_{ij}^k) \frac{\partial}{\partial z^k},
		\]
		where $(g_{ij})$ and $(h_{ij}^k)$ are $r\times r$ matrices of holomorphic functions. Such an operator is said to have \textit{scalar principal symbol} if $(h_{ij}^k) = h^k \cdot \mathrm I_{r\times r}$ for some local holomorphic functions $h^k$. There exists an exact sequence
		\[
		0 \longrightarrow \operatorname{End}E \longrightarrow \mathcal D^1(E) \xrightarrow[\quad\,]{\sigma} T_X^{1,0} \longrightarrow 0,
		\]
		where the symbol map $\sigma$ is locally given by $\sigma(P)=\sum_{k=1}^n h^k \frac{\partial}{\partial z_k}$. It was shown in e.g. \cite[Proposition A.2]{CS16} that $\mathcal D^1(E)$ carries a natural holomorphic structure, under which it is isomorphic to the Atiyah extension $A(E)$. Therefore, the DGLA $(A^{0,\bullet}(X,A(E)), \bar\partial_{A(E)}, [\cdot,\cdot])$ above is naturally isomorphic to the one obtained via the algebraic construction.
<!-- end remark-1 -->
<!-- remark-2: body_md -->
The above construction involves the choice of Hermitian metrics, and hence the resulting analytic space \(\operatorname{Kur}(X,E)\) is \textit{not uniquely defined}. Nevertheless, the germs at the origin arising from different choices of Hermitian metrics are (non-canonically) isomorphic. We denote this 
		\textit{Kuranishi germ} by
		$\bigl(\operatorname{Kur}(X,E),0\bigr)$.

<!-- proposition-4: body_md -->
\label{prop:formal-vs-canonical}
		The joint deformations of   $(X,E)$ are unobstructed  if and only if for $\epsilon_1(t)$ defined in \textcolor{Red2}{\hyperref[e1]{(\ref*{e1})}},
		the   obstruction equation
		\[
		\bar\partial_{A(E)}\epsilon(t)
		=
		\frac12\,[\epsilon(t),\epsilon(t)],
		\]
		or equivalently, writing $\epsilon(t) = (A_t,\varphi_t)$, 
		\begin{empheq}[left=\empheqlbrace]{align}
			\bar{\partial}\varphi_t &= \frac{1}{2}[\varphi_t,\varphi_t];  \tag{MC1}\label{MC1} \\[6pt]
			\bar{\partial}_{\operatorname{End} E} A_t &= \varphi_t \lrcorner R_E + \varphi_t \lrcorner D_{\operatorname{End} E}^{1,0} A_t - \frac{1}{2}[A_t,A_t] \tag{MC2}\label{MC2},
		\end{empheq}

<!-- proof-2: body_md -->
The “only if’’ part is clear by setting $\epsilon(t):=\epsilon^{\mathrm{can}}(t)$ as in \textcolor{Red2}{\hyperref[good]{(\ref*{good})}}.
		
		For the converse, the degree-two  part of the Maurer--Cartan equation reads
		\[
		\bar\partial_{A(E)}\epsilon_2(t)
		=
		\frac{1}{2}\,[\epsilon_1(t),\epsilon_1(t)].
		\]
		In particular, $[\epsilon_1(t),\epsilon_1(t)]$ is $\bar\partial_{A(E)}$-exact. By bundle-valued Hodge decomposition,
		\[
		[\epsilon_1(t),\epsilon_1(t)]
		=
		\mathcal H[\epsilon_1(t),\epsilon_1(t)]
		+
		\bar\partial_{A(E)}\bar\partial_{A(E)}^*\mathcal G[\epsilon_1(t),\epsilon_1(t)]
		+
		\bar\partial_{A(E)}^*\bar\partial_{A(E)}\mathcal G[\epsilon_1(t),\epsilon_1(t)],
		\]
		and exactness forces $\mathcal H[\epsilon_1(t),\epsilon_1(t)]=0$, hence
		\[
		[\epsilon_1(t),\epsilon_1(t)]
		=
		\bar\partial_{A(E)}\bar\partial_{A(E)}^*\mathcal G[\epsilon_1(t),\epsilon_1(t)].
		\]
		We then replace $\epsilon_2(t)$ by the degree-two term of the canonical solution of \textcolor{Red2}{\hyperref[good]{(\ref*{good})}}, namely
		\[
		\epsilon_2^{\mathrm{can}}(t)
		=
		\frac{1}{2}\,\bar\partial_{A(E)}^*\mathcal G[\epsilon_1(t),\epsilon_1(t)]
		\]
		presented in \textcolor{Red2}{\hyperref[epsilon2]{(\ref*{epsilon2})}}, which  still solves the  Maurer--Cartan equation in degree-two.

	At each higher order $N \ge 3$, a slightly more involved
	argument shows that one may replace $\epsilon_N(t)$ by
	the ``better'' term $\epsilon_N^{\mathrm{can}}(t)$, so that
	the full (convergent) power series $\epsilon^{\mathrm{can}}(t)$

		\begin{step}{Reduction on the obstruction equations:   Calabi--Yau property \& assumption $H^2(X,\operatorname{End} ^0 E)=0$.}\label{step1}
		\end{step}
		
		Under the assumption that $K_X$ is torsion, i.e., $lK_X$ is holomorphically trivial for some integer $l\geq  1$, write
		\begin{equation} \label{bundle F}
			lK_X = K_X + (l-1)K_X \quad \text{and set} \quad F := (l-1)K_X.
		\end{equation}
		Since $F$ is also torsion, it is  Hermitian  flat. More precisely, we can equip $F$ with a Hermitian metric $h_F$ whose Chern connection $$\nabla_F = D_F^{1,0} + \bar\partial_F$$ is flat, meaning its curvature $R_F$ vanishes.
		
		Moreover, there exists a nowhere vanishing $F$-valued holomorphic $(n,0)$-form
		\[
		\boldsymbol{\Omega} \in H^0(X, lK_X) \simeq H^{n,0}(X, F).
		\]
		
		
		
		We will need the following bundle-valued version of the Tian--Todorov lemma, in which the flatness of $F$ is not required --- it will be needed  later; see Lemma \ref{Fujiki}.
		
		\begin{lemma}[{\cite[Corollary 3.4]{LR12}}]\label{Tian-Todorov}
			For any $\varphi,\psi\in A^{0,1}\bigl(X, T_X^{1,0}\bigr)$ and any $\eta\in A^{\bullet,\bullet}(X,F)$, one has
			\[
			[\varphi,\psi] \lrcorner \eta
			= -D_F^{1,0}\bigl(\psi \lrcorner \bigl(\varphi \lrcorner \eta\bigr)\bigr)
			+ \psi \lrcorner D_F^{1,0}\bigl(\varphi \lrcorner \eta\bigr)
			+ \varphi \lrcorner D_F^{1,0}\bigl(\psi \lrcorner \eta\bigr)-\psi \lrcorner\bigl(\varphi\lrcorner D_F^{1,0} \eta\bigr)
			\]
		\end{lemma}
		
		
		
		Using the Calabi--Yau property of $X$ and following the insightful idea dating back to \cite{LRW19}  (also developed in \cite{Zha26}), we will need another auxiliary formal power series $\Omega(t)$ with $\Omega(0)=\boldsymbol{\Omega}$ to further reduce Proposition \ref{prop:formal-vs-canonical}   to the following:
		
		\begin{proposition}\label{prop-1}
			In the setting of Theorem  \ref{main thm},     for any first-order deformation 
			$\epsilon_1=(A_1,\varphi_1)$, suppose that there exist formal power series
			\[
			\varphi(t)=\sum_{i\geq 1}\varphi_i t^i \in A^{0,1}(X, T_X^{1,0})[[t]],\quad
			a(t)=\sum_{i\geq 1} a_i t^i  \in A^{0,1}(X,\mathcal O_X)[[t]], \quad a_1=\operatorname{tr} A_1,
			\]
			and
			\[
			\Omega(t)=\Omega_0+\sum_{i\geq 1}\Omega_i t^i\in A^{n,0}(X, F)[[t]],\quad	\Omega_0=\boldsymbol{\Omega},
			\]
			satisfying the system formally:
			\begin{equation}\tag{$\dagger$}\label{log 0.2}
				\begin{cases}
					\bigl(\bar\partial_F + D_F^{1,0}\circ \varphi(t)\lrcorner\bigr)\Omega(t)=0,\\[0.8em]
					\bigl(\bar\partial_F + \dfrac{1}{2}D_F^{1,0}\circ \varphi(t)\lrcorner\bigr)\bigl(\varphi(t)\lrcorner\Omega(t)\bigr)=0,\\[0.8em]
					\Bigl(		\bar\partial a(t)-\varphi(t) \lrcorner \operatorname{tr} R_E- \varphi(t)\lrcorner \partial a(t)\Bigr)\wedge\Omega(t)=0.\\
				\end{cases}
			\end{equation}
			Then the Kuranishi space of the pair $(X,E)$ is smooth.
		\end{proposition}
		
		\begin{proof}
			The first two 	equations in  \textcolor{Red2}{\hyperref[log 0.2]{(\ref*{log 0.2})}} together with Lemma \ref{Tian-Todorov} (noting that $\Omega(t)$ is automatically $D_F^{1,0}$-closed for type reasons) imply
			\begin{align*}
				\bar\partial\bigl(\varphi(t)\lrcorner \Omega(t)\bigr)
				&= -\frac{1}{2}\,D^{1,0}_F\Bigl(\varphi(t)\lrcorner\bigl(\varphi(t)\lrcorner\Omega(t)\bigr)\Bigr)\\
				&= \frac{1}{2}\,[\varphi(t),\varphi(t)]\lrcorner \Omega(t)
				- \varphi(t)\lrcorner D^{1,0}_F\bigl(\varphi(t)\lrcorner\Omega(t)\bigr)\\
				&= \frac{1}{2}\,[\varphi(t),\varphi(t)]\lrcorner \Omega(t)
				+ \varphi(t)\lrcorner\bar\partial\Omega(t).
			\end{align*}
			Hence
			\begin{equation}\label{aabb}
				\bigl(\bar\partial\varphi(t)\bigr)\lrcorner \Omega(t)
				= \bar\partial\bigl(\varphi(t)\lrcorner \Omega(t)\bigr)
				- \varphi(t)\lrcorner\bar\partial\Omega(t)
				= \frac{1}{2}\,[\varphi(t),\varphi(t)]\lrcorner \Omega(t).
			\end{equation}
			
			Set 
			\[
			V(t) := \bar\partial\varphi(t) - \frac{1}{2}[\varphi(t),\varphi(t)] = \sum_{i\ge 2} v_i t^i,
			\]
			with \(v_i \in A^{0,2}(X,T_X^{1,0})\). Then \textcolor{Red2}{\hyperref[aabb]{(\ref*{aabb})}} becomes \(V(t)\lrcorner\Omega(t)=0\).
			
			Locally write 
			\[
			\boldsymbol{\Omega} = f(z)\,dz^1\wedge\cdots\wedge dz^n\otimes e
			\]
			with \(f(z)\) nowhere vanishing and \(e\) a local frame of \(F\). Any \(v\in A^{0,2}(X,T_X^{1,0})\) can be expressed locally as
			\[
			v = \sum_{j=1}^n g_j\,\frac{\partial}{\partial z^j},
			\quad\text{where } g_j \text{ are smooth }(0,2)\text{-forms}.
			\]
			Contracting with \(\boldsymbol{\Omega}\) gives
			\[
			\boldsymbol{\Omega}\lrcorner v = f(z)\sum_{j=1}^n (-1)^{j-1} g_j\,
			dz^1\wedge\cdots\wedge\widehat{dz^j}\wedge\cdots\wedge dz^n\otimes e.
			\]
			The \((n-1,2)\)-forms appearing are linearly independent, hence the map \(v\mapsto\boldsymbol{\Omega}\lrcorner v\) is injective.
			
			
			\medskip
			Expanding
			\[
			V(t)\lrcorner\Omega(t)=\sum_{k\ge 2}\Bigl(\sum_{i+j=k}v_i\lrcorner\Omega_j\Bigr)t^k=0,
			\]
			we obtain:
			
			\begin{itemize}	\setlength{\itemsep}{5pt}
				\item[$\centerdot$] For \(k=2\), \(v_2\lrcorner\boldsymbol{\Omega}=0\), so \(v_2=0\) by injectivity.
				\item[$\centerdot$] Assume \(v_2=\dots=v_{N}=0\) for some \(N\ge 2\). Then the coefficient of \(t^{N+1}\) reduces to \(v_{N+1}\lrcorner\boldsymbol{\Omega}=0\), hence \(v_{N+1}=0\).
			\end{itemize}
			Therefore  \(V(t)=0\), i.e., 
			\begin{equation}\label{hhh}
				\bar\partial\varphi(t) = \dfrac{1}{2}[\varphi(t),\varphi(t)],
			\end{equation}
			namely the first obstruction equation \textcolor{Red2}{\hyperref[MC1]{(\ref*{MC1})}}.
			
			
			\medskip
			\medskip
			Via a similar argument, the third equation in \textcolor{Red2}{\hyperref[log 0.2]{(\ref*{log 0.2})}} is equivalent to 
			\begin{equation}
				\bar\partial a(t)=\varphi(t) \lrcorner \operatorname{tr} R_E+ \varphi(t)\lrcorner \partial a(t).
			\end{equation}
			This is obtained by taking the trace of the second obstruction equation \textcolor{Red2}{\hyperref[MC2]{(\ref*{MC2})}} wherein the following identities are used:
			
			\smallskip
			\begin{itemize}
				\setlength{\itemsep}{7.9pt}
				\item[--\quad] $\operatorname{tr}(\bar\partial_{\operatorname{End} E}A(t)) = \bar\partial(\operatorname{tr} A(t))$;
				\item[--\quad] $\operatorname{tr}(\varphi(t)\lrcorner R_E) = \varphi(t)\lrcorner\operatorname{tr} R_E$;
				\item[--\quad]$\operatorname{tr}([A(t),A(t)]) = 0$;
				\item[--\quad] For the term $\operatorname{tr}\Big(\varphi(t)\lrcorner D_{\operatorname{End} E}^{1,0}A(t)\Big)$, we compute locally. In a local trivialisation, $$D_{\operatorname{End} E}^{1,0}A(t) = \partial A(t) + [\omega^{1,0},A(t)],$$ where $\omega^{1,0}$ is a matrix-valued $(1,0)$-form. Then this term equals
				\[
				\varphi(t)\lrcorner\operatorname{tr}\Big(D_{\operatorname{End} E}^{1,0}A(t)\Big) = \varphi(t)\lrcorner\Big(\operatorname{tr}(\partial A(t)) + \operatorname{tr}([\omega^{1,0},A(t)])\Big) = \varphi(t)\lrcorner\partial a(t).
				\]
			\end{itemize}
			
			\medskip
			Consequently, the trace-free part of \textcolor{Red2}{\hyperref[MC2]{(\ref*{MC2})}} becomes
			\begin{equation}\tag{TF}\label{tf}
				\bar\partial_{\operatorname{End} E} A^0(t) = \varphi(t) \lrcorner R_E^0 + \varphi(t) \lrcorner D_{\operatorname{End} E}^{1,0} A^0(t) - \frac{1}{2}[A(t),A(t)].
			\end{equation}
			
			Expanding in $t$, the $N$-th order obstruction equation for $N \ge 2$ takes the form
			\[
			\bar\partial A_N^0 = C_N^{\text{tf}},
			\]
			where $C_N^{\text{tf}}$ is $\bar\partial_{\operatorname{End} E}$-closed, \textit{provided that all obstructions up to order $N-1$ in \textcolor{Red2}{\hyperref[tf]{(\ref*{tf})}} and all obstructions up to order $N$ in \textcolor{Red2}{\hyperref[hhh]{(\ref*{hhh})}} vanish}.  Since $H^2(X,\operatorname{End}^0 E)=0$, we can solve for $A_N^0$ inductively. Hence
			\[
			A(t): = A^0(t) + \frac{a(t)}{r}\,\operatorname{id}_E
			\]
			satisfies \textcolor{Red2}{\hyperref[MC2]{(\ref*{MC2})}}. This completes the proof. 
		\end{proof}
		
		\begin{remark}
			The main differences between Proposition \ref{prop-1} and \cite[Proposition 4.10]{LRW19}, as well as \cite[Proposition I.3]{Zha26} (both with $D=\varnothing$), are two-fold:
			\begin{enumerate}[(i)]
				\item An additional obstruction equation (the third equation in \textcolor{Red2}{\hyperref[log 0.2]{(\ref*{log 0.2})}}) needs to be solved simultaneously;
				\item The solutions $\varphi(t)$, $a(t)$ and $\Omega(t)$ are only required to be formal. 
				\begin{enumerate}[label=(\alph*)]
					\item In the arguments of \cite{LRW19,Zha26}, the smoothness of the corresponding  $\Omega(t)$  was used to ensure that $\Omega(t)$ has no zeros for small $t$ (since $\Omega(0)$ is nowhere vanishing).  However, formal solutions suffice as shown above.
					\item   As for $\varphi(t)$ and $a(t)$, formal solutions are also enough: once they are obtained, we can then choose particularly nice solutions whose smoothness is very standard (see Proposition \ref{prop:formal-vs-canonical}). 
				\end{enumerate}
				This omits the regularity arguments  as in \cite[proof of Theorem 4.12, pp. 36-39]{LRW19} and \cite[Appendix I, Step (III)]{Zha26}.
			\end{enumerate}
		\end{remark}
		
		\begin{step}{Solve the obstruction equations: iteration}\label{step2}
		\end{step}
		
		By \textcolor{Red2}{\hyperref[log 0.2]{(\ref*{log 0.2})}}$_k$ we mean the part of \textcolor{Red2}{\hyperref[log 0.2]{(\ref*{log 0.2})}} formed by the first equation at order $k$ together with the second and third equations at order $k+1$. We now adopt an \textit{iterative} approach to solve \textcolor{Red2}{\hyperref[log 0.2]{(\ref*{log 0.2})}}.
		
		
		
		
		
		Before proceeding with the proof, we recall the following standard
		fact; see, for example, \cite[Proposition A.1]{HZ26} for a proof.
		
		\begin{lemma}\label{Fujiki}
			Let $X$ be a Fujiki manifold, and let $(F,h_F)$ be a Hermitian flat holomorphic
			vector bundle on $X$. Write its Chern connection as
			$
			\nabla_F=D_F^{1,0}+\bar\partial_F.
			$
			Then $(X,F)$ satisfies the
			$D_F^{1,0}\bar\partial_F$-lemma. More precisely, for every
			$\nabla_F$-closed $F$-valued $(p,q)$-form $\alpha$ on $X$, the
			following conditions are equivalent:
			\begin{equation*}
				\nabla_F\text{-exactness}
				\Longleftrightarrow
				D_F^{1,0}\text{-exactness}
				\Longleftrightarrow
				\bar\partial_F\text{-exactness}
				\Longleftrightarrow
				D_F^{1,0}\bar\partial_F\text{-exactness}.
			\end{equation*}
		\end{lemma}
		
		For $k=1$, the system \textcolor{Red2}{\hyperref[log 0.2]{(\ref*{log 0.2})}}$_1$ reads as
		\begin{equation}\label{123}
			\begin{dcases}
				\bar{\partial}_F\Omega_1 + D_F^{1,0} \bigl(\varphi_1 \lrcorner \Omega_0\bigr) = 0, \\[0.6em]
				\bar{\partial}_F\bigl(\varphi_2 \lrcorner \Omega_0 + \varphi_1 \lrcorner \Omega_1\bigr)
				+ \frac{1}{2}\, D_F^{1,0} \bigl(\varphi_1 \lrcorner (\varphi_1 \lrcorner \Omega_0)\bigr) = 0, \\[0.6em]
				\begin{aligned}
					\bar{\partial}a_2 \wedge \Omega_0 
					= &-\bar\partial a_1\wedge\Omega _1 + \bigl(\varphi_2 \lrcorner\operatorname{tr}  R_E \bigr) \wedge \Omega_0  +  \bigl(\varphi_1 \lrcorner \operatorname{tr} R_E \bigr) \wedge \Omega_1 +  \bigl(\varphi_1 \lrcorner \partial a_1 \bigr) \wedge \Omega_0. 
				\end{aligned}
			\end{dcases}
		\end{equation}
		From the first equation in \textcolor{Red2}{\hyperref[123]{(\ref*{123})}}, we have
		\[
		\bar{\partial}_F D_F^{1,0}  (\varphi_1 \lrcorner \Omega_0)
		= - D_F^{1,0} \bar{\partial}_F (\varphi_1 \lrcorner \Omega_0)
		= - D_F^{1,0}  \bigl( (\bar{\partial} \varphi_1) \lrcorner \Omega_0 + \varphi_1 \lrcorner \bar{\partial}_F \Omega_0 \bigr) = 0,
		\]
		since $\bar{\partial}_F \Omega_0 = 0$ and $\bar{\partial} \varphi_1 = 0$. Hence, there exists 
		\begin{equation}\label{exact1}
			\textrm{a solution } \Omega_1
		\end{equation}
		thanks to Lemma \ref{Fujiki}.
		
		
		\medskip
		Likewise in the second equation of \textcolor{Red2}{\hyperref[123]{(\ref*{123})}}, since $D_F^{1,0}\bigl( \varphi_1 \lrcorner (\varphi_1 \lrcorner \Omega_0) \bigr)$ is $\bar{\partial}_F$-closed,  Lemma \ref{Fujiki} can be applied  again to $$\textrm{solve for } \varphi_2 \lrcorner \Omega_0,   \textrm{ and thus for } \varphi_2 \textrm{ itself,}$$ using the non-degeneracy of $\Omega_0$. Furthermore, one can ensure the existence of some $B_2\in A^{n-2,1}(X,F)$ such that 
		\begin{equation}\label{exact2}
			\varphi_2 \lrcorner \Omega_0 + \varphi_1 \lrcorner \Omega_1 = D_F^{1,0} B_2.
		\end{equation}
		
		
		\medskip
		For the third equation in \textcolor{Red2}{\hyperref[123]{(\ref*{123})}}, its right-hand side is $\bar{\partial}_{ F}$-closed, since we have already solved for $\varphi_2$.
		For any $\theta \in A^{0,1}(X, T_X^{1,0})$ and bundle-valued forms $u, v$, we have \cite[p. 292]{Deme}
		\begin{equation}\label{contraction-formula}
			\theta \lrcorner (u \wedge v) = (\theta \lrcorner u) \wedge v + (-1)^{\deg u} u \wedge (\theta \lrcorner v).
		\end{equation}
		Therefore, by type reasons,
		\begin{equation}\label{eq:first}
			\bar{\partial} a_2 \wedge \Omega_0
			= -\bar\partial a_1 \wedge \Omega_1
			- \operatorname{tr} R_E \wedge (\varphi_2 \lrcorner \Omega_0 + \varphi_1 \lrcorner \Omega_1)
			- \partial a_1 \wedge (\varphi_1 \lrcorner \Omega_0).
		\end{equation}
		
		Using \textcolor{Red2}{\hyperref[exact2]{(\ref*{exact2})}}, \textcolor{Red2}{\hyperref[bianchi2]{(\ref*{bianchi2})}}, \textcolor{Red2}{\hyperref[exact1]{(\ref*{exact1})}}, and the Leibniz rule, we obtain
		\begin{equation}\label{eq:second}
			\begin{aligned}
				\bar{\partial}_F \bigl(a_2 \wedge \Omega_0\bigr)
				&= -\bar\partial a_1 \wedge \Omega_1
				- \operatorname{tr} R_E \wedge (D_F^{1,0} B_2)
				- \partial a_1 \wedge (\varphi_1 \lrcorner \Omega_0) \\[0.5em]
				&= -\bar\partial a_1 \wedge \Omega_1
				- D_F^{1,0}\bigl( \operatorname{tr} R_E \wedge B_2 - a_1 \wedge (\varphi_1 \lrcorner \Omega_0) \bigr)
				+ a_1 \wedge \bar\partial_F \Omega_1 \\[0.5em]
				&= -\bar\partial_F (a_1 \wedge \Omega_1)
				- D_F^{1,0}\bigl( \operatorname{tr} R_E \wedge B_2 - a_1 \wedge (\varphi_1 \lrcorner \Omega_0) \bigr).
			\end{aligned}
		\end{equation}
		Therefore, we can \begin{equation}\label{exact3}
			\textrm{solve for $a_2 \wedge \Omega_0 + a_1 \wedge \Omega_1$, and hence for $a_2$},
		\end{equation}  based on Lemma \ref{Fujiki}.
		
		
		
		\medskip
		\medskip
		\medskip
		By induction, assume that \textcolor{Red2}{\hyperref[log 0.2]{(\ref*{log 0.2})}}  is solvable for all $k \le N$, and that we have already found
		\[
		\Omega_i, \, 0 \le i \le N, \quad \varphi_i, \, 1 \le i \le N+1,  \quad \text{and} \quad a_i,  \, 1 \le i \le N+1.
		\]
		Thus, \textcolor{Red2}{\hyperref[hhh]{(\ref*{hhh})}} implies
		\begin{equation}\label{log 0.4}
			\bar{\partial} \varphi_k = \frac{1}{2} \sum_{i+j=k} [\varphi_i, \varphi_j], \quad k \le N+1,
		\end{equation}
		and then 
		\begin{equation}\label{log 0.5}
			\bar{\partial} \Bigl( \sum_{i+j=k} [\varphi_i, \varphi_j] \Bigr) = 0, \quad k \le N+2,
		\end{equation}
		by using $[\alpha,\beta]=-[\beta,\alpha]$ for $\alpha\in A^{0,2} (X,T_X^{1,0})$ and arbitrary $\beta$.
		
		
		\medskip
		\medskip
		For the $(N+1)$-th step, \textcolor{Red2}{\hyperref[log 0.2]{(\ref*{log 0.2})}}$_{N+1}$ together with \textcolor{Red2}{\hyperref[contraction-formula]{(\ref*{contraction-formula})}} yields
		\begin{equation}\label{log 0.6}
			\begin{dcases}
				\bar{\partial}_F \Omega_{N+1} 
				+ \sum_{\substack{i+j=N+1}} D^{1,0}_{F}  \bigl({\varphi_i}\lrcorner \Omega_j \bigr) = 0, \\[0.8em]
				\sum_{\substack{i+j=N+2}} \bar{\partial}_F \bigl( {\varphi_i}\lrcorner \Omega_j \bigr)
				+ \sum_{\substack{i+j+k=N+2}} \frac{1}{2} D^{1,0}_{F}  \Bigl({\varphi_i}\lrcorner \bigl({\varphi_j}\lrcorner \Omega_k \bigr)\Bigr) = 0, \\[0.8em]
				\bar\partial_F \bigl(a_{N+2}\wedge\Omega_0\bigr)=-\sum_{\substack{i+j=N+2}}  \operatorname{tr} R_E\wedge\bigl(\varphi_i\lrcorner\Omega_j\bigr)-\sum_{\substack{i+j+k=N+2}} \partial a_i\wedge\bigl(\varphi_j\lrcorner\Omega_k\bigr)-\sum_{\substack{i+j=N+2,\,  j\geq 1}}\bar\partial a_i\wedge \Omega_j.
			\end{dcases}
		\end{equation}
		For the first equation  in \textcolor{Red2}{\hyperref[log 0.6]{(\ref*{log 0.6})}}, 
		\allowdisplaybreaks
		\begin{align*}
			-\bar\partial_F \sum_{i+j=N+1} D^{1,0}_F(\varphi_i \lrcorner \Omega_j)
			&= D^{1,0}_F \Bigg(
			\sum_{i=1}^{N+1} \bar\partial \varphi_i \lrcorner \Omega_{N+1-i}
			+ \sum_{i=1}^{N+1} \varphi_i \lrcorner \bar\partial_F \Omega_{N+1-i}
			\Bigg) \\[0.9em]
			&= D^{1,0}_F \Bigg(
			\frac12 \sum_{i=1}^{N+1} \sum_{j=1}^{i-1} [\varphi_j,\varphi_{i-j}] \lrcorner \Omega_{N+1-i} \\
			&\qquad\qquad
			- \sum_{i=1}^{N+1} \varphi_i \lrcorner D^{1,0}_F \bigg(
			\sum_{j=1}^{N+1-i} \varphi_j \lrcorner \Omega_{N+1-i-j}
			\bigg)
			\Bigg) \\[0.9em]
			&= D^{1,0}_F \Bigg(
			\frac12 \sum_{i=1}^{N+1} \sum_{j=1}^{i-1} \bigg(
			- D^{1,0}_F \bigl( \varphi_j \lrcorner (\varphi_{i-j} \lrcorner \Omega_{N+1-i}) \bigr) \\
			&\qquad\qquad\qquad
			+ \varphi_j \lrcorner D^{1,0}_F \bigl( \varphi_{i-j} \lrcorner \Omega_{N+1-i} \bigr)
			+ \varphi_{i-j} \lrcorner D^{1,0}_F \bigl( \varphi_j \lrcorner \Omega_{N+1-i} \bigr)
			\bigg) \\
			&\qquad
			- \sum_{i=1}^{N+1} \varphi_i \lrcorner D^{1,0}_F \bigg(
			\sum_{j=1}^{N+1-i} \varphi_j \lrcorner \Omega_{N+1-i-j}
			\bigg)
			\Bigg) \\[0.9em]
			&= D^{1,0}_F \Bigg(
			\sum_{1 \le j < i \le N+1} \varphi_j \lrcorner D^{1,0}_F \bigl( \varphi_{i-j} \lrcorner \Omega_{N+1-i} \bigr) \\
			&\qquad
			- \sum_{i=1}^{N+1} \varphi_i \lrcorner D^{1,0}_F \bigg(
			\sum_{j=1}^{N+1-i} \varphi_j \lrcorner \Omega_{N+1-i-j}
			\bigg)
			\Bigg) \\[0.6em]
			&= 0,
		\end{align*}
		where the third equality holds in view of   Lemma \ref{Tian-Todorov}.
		We then can   \begin{equation}\label{exact1'}
			\textrm{solve for } \Omega_{N+1}
		\end{equation}
		by Lemma \ref{Fujiki}.
		
		
		\medskip\medskip
		Moreover, for the second  equation  in \textcolor{Red2}{\hyperref[log 0.6]{(\ref*{log 0.6})}}, 
		\begin{align*}
			&\quad \bar{\partial}_F \sum_{\substack{i+j+k=N+2}} D^{1,0}_{F}  \Bigl({\varphi_i}\lrcorner \bigl({\varphi_j}\lrcorner \Omega_k \bigr)\Bigr) \\
			&= \sum_{i+j+k=N+2} \bar{\partial}_F \Bigl(
			- [\varphi_j, \varphi_i] \lrcorner \Omega_k
			+ \varphi_i \lrcorner D^{1,0}_F \bigl(\varphi_j \lrcorner \Omega_k\bigr)
			+ \varphi_j \lrcorner D^{1,0}_F \bigl(\varphi_i \lrcorner \Omega_k\bigr)
			\Bigr) \\[0.6em]
			&= - \sum_{i+j+k=N+2} \bar{\partial}_F \bigl([\varphi_j, \varphi_i] \lrcorner \Omega_k\bigr)
			- 2 \sum_{\substack{j+k=N+2\\ k \ge 1}} \bar{\partial}_F \bigl(\varphi_j \lrcorner \bar{\partial}_F \Omega_k\bigr) \\[6pt]
			&= - \sum_{k=0}^{N+2} \Biggl( \bar{\partial} \sum_{i+j=N+2-k} [\varphi_j, \varphi_i] \Biggr) \lrcorner \Omega_k \\[3pt]

			- \bar{\partial} \varphi_{N+2-k} \Biggr) \lrcorner \bar{\partial}_F \Omega_k \\[6pt]
			&= 0,

		Together with $H^0(X,V)=\mathbb C s$, this gives
		$$
		H^0(X_R,V_R)=R\,s_R,\quad s_R|_X=s.
		$$
		Any other extension of $s$ is therefore $u s_R$
		for an invertible element $u\in R$.
		Since $Z(u s_R)=Z(s_R)$, the closed subspace
		$C_R:=Z(s_R)$ is independent of this choice.
		Moreover, $s$ cuts out $C$ in codimension two,
		so $C_R$ is a flat deformation of $C$.
		Hence the given bundle $E_R$ determines $C_R\subset X_R$.
		
				\medskip
		Conversely, suppose that $C_R\subset X_R$ is given.
		Both $\det N_{C_R/X_R}$ and $L_R|_{C_R}$ extend $L|_C$.
		Since $H^1(C,\mathcal O_C)=0$, the same uniqueness
		argument as in \textcolor{Red2}{\hyperref[thomas-picard]{(\ref*{thomas-picard})}} gives
		$$
		\det N_{C_R/X_R}\cong L_R|_{C_R}.
		$$
		The vanishings for $L^{-1}$ in \textcolor{Red2}{\hyperref[thomas-vanishing]{(\ref*{thomas-vanishing})}}
		also hold for $L_R^{-1}$.
		The Serre construction of \cite[\S 3]{Tho99}
		therefore applies over $R$:
		$H^2(X_R,L_R^{-1})=0$ gives a rank-two bundle $V_R$
		in the sequence \textcolor{Red2}{\hyperref[thomas-serre]{(\ref*{thomas-serre})}} over $R$, and
		$H^1(X_R,L_R^{-1})=0$ makes it unique up to isomorphism.
		Twisting by $\mathcal O_{X_R}(1,-1)$ as in \textcolor{Red2}{\hyperref[EEE]{(\ref*{EEE})}}
		recovers $E_R$. This proves \textcolor{Red2}{\hyperref[thomas-correspondence]{(\ref*{thomas-correspondence})}}.
		
		\begin{step}{Lift the curve and the threefold.}
			\label{step:thomas-lift}
		\end{step}
		
		By \textcolor{Red2}{\hyperref[thomas-correspondence]{(\ref*{thomas-correspondence})}}, it remains to lift
		$(X_R,C_R)$.
		
		By \textcolor{Red2}{\hyperref[thomas-picard]{(\ref*{thomas-picard})}}, the line bundles
		$\mathcal O_X(1,0)$ and $\mathcal O_X(0,1)$ extend to $X_R$.
		Moreover, \cite[Equation (3.2) with
		$(k,l)=(1,0)$ \& $(0,1)$]{Tho99}, gives
		$$
		H^1(X,\mathcal O_X(1,0))
		=
		H^1(X,\mathcal O_X(0,1))
		=0.
		$$
		Thus the coordinate sections $x,y,z$ and $u,v,w$
		extend and still generate these line bundles.
		They define a morphism
		$$
		X_R\longrightarrow P_R:=P\times\operatorname{Spec}R
		$$
		extending $X\subset P$,  which remains a closed
		immersion by Nakayama's lemma.
		Since $X_R$ is flat over $R$ and $X\subset P$ is
		a hypersurface, the immersion $X_R\subset P_R$ is
		regular of codimension one
		\cite[Proposition~D.2.5]{Ser06}.
		Its ideal is therefore a line bundle extending
		$\mathcal O_P(-3,-3)$. 
		As $H^1(P,\mathcal O_P)=0$, the same uniqueness argument
		as in \textcolor{Red2}{\hyperref[thomas-picard]{(\ref*{thomas-picard})}} identifies this ideal with
		$\mathcal O_{P_R}(-3,-3)$.
		Hence
		$$
		X_R=\{F_R=0\}\subset P_R
		$$
		for a bicubic polynomial $F_R$ lifting $F$.
		
		\medskip
		In the Addendum to the revised version of \cite{Tho99},
		Thomas described the deformations of $C$ inside $P$
		as a moving line in the first factor together with
		a moving point in the second.
		Writing $\mathfrak m_R$ for the maximal ideal of $R$,
		his description gives
		\begin{equation}\label{thomas-embedded-pair}
			C_R=\{x=ay+bz,\ u=cw,\ v=dw\}
			\subset\{F_R=0\},
			\quad a,b,c,d\in\mathfrak m_R.
		\end{equation}
		
		
				\medskip
		Choose lifts $\widetilde a,\widetilde b,\widetilde c,
		\widetilde d$ to $R'$.
		The same equations define
		$$
		C_{R'}:=
		\{x=\widetilde a y+\widetilde b z,\
		u=\widetilde c w,\
		v=\widetilde d w\}
		\cong\mathbb P^1_{R'}
		$$
		inside $P_{R'}:=P\times\operatorname{Spec}R'$.
		Thus $C_{R'}$ is a flat lift of $C_R$.
		Choose also a bicubic lift $\widetilde F$ of $F_R$ and set
		$$
		Q(y,z):=
		\widetilde F
		\bigl(\widetilde a y+\widetilde b z,\ y,\ z;\
		\widetilde c,\ \widetilde d,\ 1\bigr).
		$$
		The containment in \textcolor{Red2}{\hyperref[thomas-embedded-pair]{(\ref*{thomas-embedded-pair})}} implies
		that every coefficient of the homogeneous cubic $Q$
		lies in $J$.
		Since $\widetilde F$ has degree three in $u,v,w$,
		$$
		\widetilde F|_{C_{R'}}=w^3Q(y,z).
		$$
		We therefore set
		\begin{equation}\label{thomas-correction}
			F_{R'}:=\widetilde F-w^3Q(y,z).
		\end{equation}
		The correction is again bicubic, and
		\textcolor{Red2}{\hyperref[thomas-correction]{(\ref*{thomas-correction})}} gives
		\begin{equation}\label{thomas-lifting}
			F_{R'}\bmod J=F_R,
			\quad
			F_{R'}|_{C_{R'}}=0.
		\end{equation}
		
				\medskip
		Put $X_{R'}:=\{F_{R'}=0\}\subset P_{R'}$.
		Since its central fibre $X$ is a smooth hypersurface,
		the Jacobian criterion shows that $X_{R'}$ is smooth,
		hence flat, over $R'$.
		Thus \textcolor{Red2}{\hyperref[thomas-lifting]{(\ref*{thomas-lifting})}} gives a lift
		$(X_{R'},C_{R'})$ of $(X_R,C_R)$.
		By \textcolor{Red2}{\hyperref[thomas-correspondence]{(\ref*{thomas-correspondence})}}, this lifted pair
		determines a bundle $E_{R'}$ restricting to $E_R$.
		This proves that $\operatorname{Def}(X,E)$ is smooth.

	\setcounter{step}{0}
	
	\begin{step}{An obstructed deformation of an ideal sheaf.}
	\end{step}
	
	Fix $p\in X$ and local holomorphic coordinates
	$z_1,\ldots,z_n$ centred at $p$.
	Let $Z\subset X$ be the zero-dimensional closed
	analytic subspace supported at $p$ with ideal
	$$
	\mathcal I_{Z,p}
	=
	(z_4,\ldots,z_n)+(z_1,z_2,z_3)^2,
	$$
	where the first summand is omitted when $n=3$.
	Then $\operatorname{length}(Z)=4$.
	
	\medskip
	For $n=3$, \cite[Theorem~2.1]{HMS22} identifies
	the germ of the Hilbert scheme at $[Z]$ with
	$(\mathbb C^3,0)\times(C\operatorname{Gr}(2,6),0)$.
	For $n>3$, on the chart with quotient basis
	$1,z_1,z_2,z_3$, each additional coordinate
	contributes four free parameters. Hence
	\begin{equation}\label{eq:simple-hilbert-model}
		\bigl(\operatorname{Hilb}^4(X),[Z]\bigr)
		\simeq
		(\mathbb C^{4n-9},0)\times
		\bigl(C\operatorname{Gr}(2,6),0\bigr).
	\end{equation}
	Here $C\operatorname{Gr}(2,6)\subset\mathbb C^{15}$
	is the affine cone over the Pl\"ucker embedding,
	with coordinates $p_{ij}$, $1\leq i<j\leq6$,
	and quadratic defining relations
	\cite[\S1.3]{HMS22}.
	Choose $v$ with $p_{12}(v)=p_{34}(v)=1$
	and all other coordinates zero.
	Since the relations are quadratic, $\epsilon v$
	defines a first-order \textit{embedded deformation}  of $Z$,
	namely a flat deformation as a closed analytic
	subspace of the fixed manifold $X$;
	see \cite[\S3.2.1]{Ser06}.
	
		\medskip
	The ideal sequence
	$$
	0\longrightarrow\mathcal I_Z
	\longrightarrow\mathcal O_X
	\longrightarrow\mathcal O_Z
	\longrightarrow0
	$$
	induces
	$$
	H^1(X,\mathcal O_X)
	\longrightarrow
	\operatorname{Ext}^1_X(\mathcal I_Z,\mathcal O_X)
	\longrightarrow
	\operatorname{Ext}^2_X(\mathcal O_Z,\mathcal O_X),
	$$
	where $\operatorname{Ext}^q_X$ denotes the global
	extension groups of coherent sheaves.
	The first term vanishes by assumption, while
	Serre duality gives
	$$
	\operatorname{Ext}^2_X(\mathcal O_Z,\mathcal O_X)
	\simeq H^{n-2}(X,\mathcal O_Z)^*=0.
	$$
	Thus
	\begin{equation}\label{eq:simple-ideal-ext-vanishing}
		\operatorname{Ext}^1_X(\mathcal I_Z,\mathcal O_X)=0.
	\end{equation}
	By \textcolor{Red2}{\hyperref[eq:simple-ideal-ext-vanishing]{(\ref*{eq:simple-ideal-ext-vanishing})}} and
	\cite[Corollary~A.6]{Ser06}, a second-order lift
	of the corresponding ideal sheaf would yield
	an embedded lift.
	However, the Pl\"ucker relation
	$$
	q=p_{12}p_{34}-p_{13}p_{24}+p_{14}p_{23}
	$$
	would then require
	$$
	0=q(\epsilon v+\epsilon^2w)
	=\epsilon^2\neq0
	\quad\bmod\epsilon^3,
	$$
	a contradiction.
	Consequently, the corresponding first-order class
	$\alpha\in\operatorname{Ext}^1_X(\mathcal I_Z,\mathcal I_Z)$
	has nonzero quadratic obstruction
	\cite[\S2.3]{CPZ24}:
	\begin{equation}\label{eq:simple-ideal-obstruction}
		\alpha\circ\alpha\neq0
		\quad\text{in}\quad
		\operatorname{Ext}^2_X(\mathcal I_Z,\mathcal I_Z).
	\end{equation}
	
	
		\medskip
	Recall that a coherent sheaf $\mathcal F$ is
	\textit{simple} if
	$\operatorname{Hom}_X(\mathcal F,\mathcal F)
	=\mathbb C\cdot\operatorname{id}_{\mathcal F}$.
	An endomorphism of $\mathcal I_Z$ restricts on
	$X\setminus\{p\}$ to multiplication by a holomorphic
	function, which extends to a constant on $X$
	by Hartogs' theorem and compactness.
	Since $\mathcal I_Z$ is torsion-free, the endomorphism
	is scalar on all of $X$. Thus $\mathcal I_Z$ is simple.
	
	\medskip
	\begin{step}{Construction of the vector bundle.}
	\end{step}
	
	We follow the construction of Joyce--Song in \cite[\S8]{JS12}.
	Fix a very ample line bundle $\mathcal O_X(1)$
	and set $G_0:=\mathcal I_Z$.
	By \cite[Theorems~II.5.17 \& III.5.2]{Har77},
	we may successively choose $m_j\gg0$ such that
	$G_{j-1}(m_j)$ is globally generated and
	\begin{equation}\label{eq:simple-serre-vanishing}
		H^q\bigl(X,G_{j-1}(m_j)\bigr)=0
		\qquad(q>0).
	\end{equation}
	Put $V_j:=H^0(X,G_{j-1}(m_j))$ and consider
	\begin{equation}\label{eq:simple-evaluation}
		0\longrightarrow G_j
		\longrightarrow V_j\otimes\mathcal O_X(-m_j)
		\longrightarrow G_{j-1}
		\longrightarrow0,
		\qquad 1\leq j\leq n-1.
	\end{equation}
	
		\medskip
	Since $X$ is smooth of dimension $n$, the ideal
	sequence and \cite[Proposition~III. 6.11A]{Har77}
	show that $\mathcal I_Z$ has local projective
	dimension at most $n-1$.
	Each sequence \textcolor{Red2}{\hyperref[eq:simple-evaluation]{(\ref*{eq:simple-evaluation})}}
	decreases any positive local projective dimension
	by one. Hence
	$$
	E:=G_{n-1}
	$$
	is locally free.
	
	\medskip
	\begin{step}{Simplicity and the quadratic obstruction.}\label{sssssstep3}
	\end{step}
	
	Denote by $D^b(X)$ the bounded derived category
	of coherent sheaves on $X$, and set
	$M_j:=\mathcal O_X(-m_j)$.
	Since $X$ is strict Calabi--Yau,
	$M_j\otimes K_X\simeq M_j$ and
	$$
	\operatorname{Ext}^q_X(M_j,M_j)
	\simeq H^q(X,\mathcal O_X)
	=
	\begin{cases}
		\mathbb C,&q=0,n,\\
		0,&q\neq0,n.
	\end{cases}
	$$
	Thus $M_j$ is spherical in the sense of Seidel--Thomas
	\cite[Definition~1.1]{ST01};
	see also \cite[Example~3.3]{ST01}.
	By \cite[Theorem~1.2]{ST01}, its Seidel--Thomas
	twist $T_{M_j}$ is an exact autoequivalence
	of $D^b(X)$.
	
		\medskip
	By \textcolor{Red2}{\hyperref[eq:simple-serre-vanishing]{(\ref*{eq:simple-serre-vanishing})}},
	$$
	\operatorname{Hom}_X(M_j,G_{j-1})=V_j,
	\quad
	\operatorname{Ext}^q_X(M_j,G_{j-1})=0
	\,\,(q>0).
	$$
	The evaluation description of the twist
	\cite[p.~44]{ST01} therefore gives the triangle
	$$
	V_j\otimes M_j
	\longrightarrow G_{j-1}
	\longrightarrow T_{M_j}(G_{j-1})
	\longrightarrow (V_j\otimes M_j)[1].
	$$
	Comparing with \textcolor{Red2}{\hyperref[eq:simple-evaluation]{(\ref*{eq:simple-evaluation})}}, we obtain
	$$
	T_{M_j}(G_{j-1})\simeq G_j[1];
	$$
	see also \cite[Lemma~8.2 \& (8.1)]{JS12}.
	Applying these autoequivalences successively gives
	\begin{equation}\label{eq:simple-ext-identification}
		\operatorname{Ext}^{\bullet}_X(\mathcal I_Z,\mathcal I_Z)
		\simeq
		\operatorname{Ext}^{\bullet}_X(E,E),
	\end{equation}
	compatibly with composition.
	
		\medskip
	In degree zero, \textcolor{Red2}{\hyperref[eq:simple-ext-identification]{(\ref*{eq:simple-ext-identification})}}
	shows that $E$ is simple.
	Let $\beta\in\operatorname{Ext}^1_X(E,E)$
	correspond to $\alpha$. Then
	\textcolor{Red2}{\hyperref[eq:simple-ideal-obstruction]{(\ref*{eq:simple-ideal-obstruction})}} gives
	\begin{equation}\label{eq:simple-bundle-obstruction}
		\beta\circ\beta\neq0
		\quad\text{in}\quad
		\operatorname{Ext}^2_X(E,E)
		\simeq H^2(X,\operatorname{End} E).
	\end{equation}
Since $H^2(X,\mathcal O_X)=0$, the trace decomposition gives
$$
H^2(X,\operatorname{End}^0E)\simeq H^2(X,\operatorname{End} E)\neq0.
$$
In particular, $\operatorname{rank}E\geq2$.

	\begin{step}{Persistence under joint deformations.}
	\end{step}
	
	Let $X_R$ be an arbitrary first-order deformation
	of $X$ over $R=\mathbb C[\epsilon]/(\epsilon^2)$.
	Since $H^1(X,\mathcal O_X)=H^2(X,\mathcal O_X)=0$,
	the same argument as in \textcolor{Red2}{\hyperref[thomas-picard]{(\ref*{thomas-picard})}}
	extends $\mathcal O_X(1)$ to $X_R$.
	Keeping the defining ideal constant in a local
	trivialization near $p$ also extends $Z$ to
	a flat subspace $Z_R\subset X_R$.
	
	Starting with $\mathcal I_{Z_R}$, we lift
	\textcolor{Red2}{\hyperref[eq:simple-evaluation]{(\ref*{eq:simple-evaluation})}} successively.
	The sections defining the evaluation maps lift
	by \textcolor{Red2}{\hyperref[eq:simple-serre-vanishing]{(\ref*{eq:simple-serre-vanishing})}}, and the
	lifted maps remain surjective by Nakayama's lemma.
	Their kernels are $R$-flat, and the final kernel
	is locally free since its central fibre is $E$.
	Thus $E$ extends over every first-order deformation of $X$.
	
		\medskip
	In the cohomology sequence of the Atiyah extension,
	$$
	H^1(X,A(E))
	\longrightarrow H^1(X,T_X^{1,0})
	\longrightarrow H^2(X,\operatorname{End} E)
	\xrightarrow{\;\iota\;}H^2(X,A(E)),
	$$
	the first arrow is therefore surjective,
	so $\iota$ is injective.
	
	The class $\beta$ defines a first-order deformation
	of $(X,E)$ with $X$ fixed.
	Its quadratic obstruction is
	$-\iota(\beta\circ\beta)$, which is nonzero by
	\textcolor{Red2}{\hyperref[eq:simple-bundle-obstruction]{(\ref*{eq:simple-bundle-obstruction})}} and the
	injectivity of $\iota$.
	Hence $\operatorname{Kur}(X,E)$ is singular.

We divide the proof into two steps.
		
		
		
		
		\setcounter{step}{0}
		\begin{step}
			Reduction to the Kuranishi space of the bundle
		\end{step}
		Since $X$ is a complex torus, it is compact K\"ahler. Endow $X$
		with a flat translation-invariant K\"ahler metric. Since
		$L\in\operatorname{Pic}^0(X)$, the line bundle $L$ admits a
	flat Hermitian metric. Equip
		\(
		E=L^{\oplus r}
		\)
		with the induced flat metric. 
		The splitting theorem of Kasuya--Purho
		\cite[Proposition 1.1]{KP26} then gives an isomorphism of analytic germs
		\[
		\bigl(\operatorname{Kur}(X,E),0\bigr)
		\simeq
		\bigl(\operatorname{Kur}(X),0\bigr)
		\times
		\bigl(\operatorname{Kur}_X(E),0\bigr),
		\]
		where $\operatorname{Kur}_X(E)$ denotes the Kuranishi space of
		deformations of the holomorphic structure of $E$ with the complex
		structure of $X$ fixed.
		
		Since $X$ has trivial
		canonical bundle, it has unobstructed deformations  thanks to  the
		Bogomolov--Tian--Todorov theorem, it therefore suffices to prove that
		$\operatorname{Kur}_X(E)$ is singular at $0$.
		
		
		\begin{step}
			Identification with the germ of the commuting variety
		\end{step}
		Recall that the infinitesimal deformations of a holomorphic vector
		bundle $E$ over a fixed compact complex manifold $X$ are parametrized
		by $H^1(X,\operatorname{End}E)$, while the obstructions lie in
		$H^2(X,\operatorname{End}E)$. In our case,
		\(
		\operatorname{End}E
		\simeq
		\mathcal O_X\otimes_{\mathbb C}\mathfrak{gl}_r(\mathbb C).
		\)
		Therefore
		\[
		H^1(X,\operatorname{End}E)
		\simeq
		H^{0,1}(X)\otimes\mathfrak{gl}_r(\mathbb C)
		\simeq
		\bigoplus_{i=1}^n
		\mathbb C w_i\otimes\mathfrak{gl}_r(\mathbb C),
		\]
		where $
		\left\{w_1,\ldots,w_n\right\} 
		$
		denotes  a basis of translation-invariant harmonic $(0,1)$-forms.
		
		Denote by $E_{\alpha\beta}$, $1\leq\alpha,\beta\leq r$, the standard
		basis of $\mathfrak{gl}_r(\mathbb C)$. Then the first-order
		term in the Kuranishi construction can be written as
		\[
		\varphi_1(t)
		=
		\sum_{i=1}^n
		\sum_{\alpha,\beta=1}^r
		t_{i,\alpha\beta}\,
		w_i\otimes E_{\alpha\beta}:=\sum_{i=1}^n w_i\otimes A_i(t),
		\]
		where we define
		\(
		A_i(t)
		:=
		\sum_{\alpha,\beta=1}^r
		t_{i,\alpha\beta}E_{\alpha\beta}
		\in\mathfrak{gl}_r(\mathbb C).
		\)
		
		Since the $w_i$ are translation-invariant harmonic forms, their wedge
		products are again harmonic. In particular, the bracket
		\[
		\frac12[\varphi_1(t),\varphi_1(t)]
		=
		\sum_{1\leq i<j\leq n}
		w_i\wedge w_j\otimes[A_i(t),A_j(t)]
		\]
		is harmonic. Consequently, the Green operator annihilates this bracket,
		and all higher-order terms in the canonical Kuranishi series vanish.
		Thus the obstruction map is precisely
		\[
		\operatorname{Ob}_E(t)
		=
		\sum_{1\leq i<j\leq n}
		w_i\wedge w_j\otimes[A_i(t),A_j(t)].
		\]
		Furthermore, since the forms
		$\{w_i\wedge w_j\,|\, 1\leq i<j\leq n\}$
		are linearly independent,
		\[
		\operatorname{Ob}_E(t)=0
		\quad\Longleftrightarrow\quad
		[A_i(t),A_j(t)]=0
		\quad\text{for all }i<j.
		\]
		
		Define the commuting variety of $n$-tuples of $r\times r$ matrices by
		\[
		\mathcal C_{n,r}
		:=
		\left\{
		(A_1,\ldots,A_n)\in\mathfrak{gl}_r(\mathbb C)^n
		\;\middle|\;
		[A_i,A_j]=0
		\text{ for every }i<j
		\right\}.
		\]
		Then there is an isomorphism of analytic germs
		\[
		\bigl(\operatorname{Kur}_X(E),0\bigr)
		\simeq
		\bigl(\mathcal C_{n,r},0\bigr).
		\]
		
		\medskip
		
		It therefore suffices to consider the following two cases:
		\begin{enumerate}[{(i)}]
			\item If $\operatorname{Kur}_X(E)$ is reducible at $0$, then the origin
			cannot be a smooth point.
			
			\item If $\operatorname{Kur}_X(E)$ is irreducible at $0$, 
			then the analytic germ $(\mathcal C_{n,r},0)$ is irreducible.
			
			We then  claim that the underlying topological space
			$\lvert\mathcal C_{n,r}\rvert$ is globally irreducible. Suppose, to
			the contrary, that
			\[
			\lvert\mathcal C_{n,r}\rvert
			=
			Z_1\cup\cdots\cup Z_\ell,
			\quad \ell\geq2,
			\]
			is its decomposition into irreducible components.
			
			Since the defining commutator equations are homogeneous, every
			$Z_\alpha$ is a cone and therefore contains the origin. Set
			\(
			W:=Z_2\cup\cdots\cup Z_\ell.
			\)
			Then
			\[
			\bigl(\mathcal C_{n,r},0\bigr)
			=
			(Z_1,0)\cup(W,0).
			\]
			
			Both $(Z_1,0)$ and $(W,0)$ are proper analytic subgerms. Indeed,
			suppose for instance that
			\(
			(Z_1,0)=(\mathcal C_{n,r},0).
			\)
			Then $Z_1$ agrees with $\lvert\mathcal C_{n,r}\rvert$ in some
			neighborhood $U$ of the origin. Given any
			$z\in\lvert\mathcal C_{n,r}\rvert$, choose
			$\lambda\in\mathbb C^*$ sufficiently small that $\lambda z\in U$.
			Then $\lambda z\in Z_1$. Since $Z_1$ is a cone, it follows that
			\[
			z=\lambda^{-1}(\lambda z)\in Z_1.
			\]
			Thus $\lvert\mathcal C_{n,r}\rvert=Z_1$, contradicting
			$\ell\geq2$. The same argument applies to $W$.
			
			Hence $(\mathcal C_{n,r},0)$ is the union of two proper closed
			analytic subgerms, contradicting its irreducibility. Therefore
			$\ell=1$, and $\lvert\mathcal C_{n,r}\rvert$ is globally
			irreducible.
			
			\medskip
			Applying \cite[Proposition 6]{GS00}, or equivalently
			\cite[Lemma 7.9.1]{OCV11}, we obtain\footnote{We learned of these
				references from Alexander Stasinski's helpful answer to the
				MathOverflow question
				\href{https://mathoverflow.net/questions/499942/dimension-of-the-variety-of-commuting-matrices}
				{\emph{Dimension of the variety of commuting matrices}}.}
			\[
			\dim\mathcal C_{n,r}
			=
			r^2+(n-1)r,
			\]
			and hence the local analytic dimension of $\textrm{Kur}_X(E)$ at $0$ is 
			\[
			\dim_0\operatorname{Kur}_X(E)
			=
			r^2+(n-1)r.
			\]
			On the other hand, the dimension of the Zariski tangent space at
			$0$ is
			\[
			\dim_{\mathbb C}T_0\operatorname{Kur}_X(E)
			=
			h^1(X,\operatorname{End}E)
			=
			nr^2.
			\]
			
			For $n,r\geq2$, we have
			\[
			\begin{aligned}
				\dim_{\mathbb C}T_0\operatorname{Kur}_X(E)
				-
				\dim_0\operatorname{Kur}_X(E)
				&=
				nr^2-\bigl(r^2+(n-1)r\bigr)\\
				&=
				(n-1)r(r-1)\\
				&>0.
			\end{aligned}
			\]
		\end{enumerate}
		
		In either case, $\operatorname{Kur}_X(E)$, and hence
		$\operatorname{Kur}(X,E)$, is singular at $0$. This completes the proof.

\label{rem:wan-huybrechts}
			Huybrechts \cite[\S2]{Huy95} asked whether the tangent
			bundle of a strict Calabi--Yau threefold can have
			obstructed deformations with the threefold fixed.
			Wan \cite[Theorem~1.5]{Wan26} constructs such a
			projective threefold $Z$ with Hermitian flat
			tangent bundle.
			The splitting theorem
			\cite[Proposition~1.1]{KP26} then shows that the
			joint deformations of $(Z,T_Z^{1,0})$ are also
			obstructed, thereby also answering
			Felten's Question~\ref{q2} positively.
