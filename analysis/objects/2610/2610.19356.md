\label{rem:temperature-range}
	The two conditions defining $\beta_{\max}(\kappa)$ enter at different stages of the proof. a)~\textbf{The overlap condition $\beta^2<\alpha_c(\kappa)$}. This condition controls exponential moments of the overlap between uniform configurations with bounded magnetizations (Lemma~\ref{lem:profile-moments}). These bounds yield the mixed second-moment limits for the normalized partition functions and the overlap comparison. This is the only temperature restriction needed for Theorem~\ref{thm:balanced_clt}.
    b)~\textbf{The profile condition $\alpha<\alpha_c(\kappa)$}. This additional condition controls the contribution of profiles with large magnetization when removing the profile constraint. For the truncated disorder, the annealed profile weights on bounded magnetization windows are asymptotically proportional to $P_{\boldsymbol{n}}e^{\alpha\left\Vertm_{\boldsymbol{n}}\right\Vert_2^2/2}$, where $P_{\boldsymbol{n}}$ is the multinomial probability of $\boldsymbol{n}$. These leading weights are those of the Curie--Weiss--Potts model at inverse temperature $\kappa\alpha$, and $\alpha<\alpha_c(\kappa)$ places that model in its high-temperature phase~\cite{EW90}. The resulting tail bounds allow the comparison to extend to all profiles (Proposition~\ref{prop:profile-equiv}).

\label{rem:loop-motivation}
 The graph expansion of the unconstrained partition function motivates the disorder statistics used for the common factor in Step 1. Define
 \begin{align*}
  \mathrm{Ch}_\kappa(x)&:=e^{-x}\big(1+\kappa^{-1}(e^{\kappa x}-1)\big)
  \quad\text{and}\quad
  \mathrm{Th}_\kappa(x):=\frac{e^{\kappa x}-1}{e^{\kappa x}+\kappa-1}.
 \end{align*}
 The identity $e^{xA_e(\boldsymbol{\sigma})}=\mathrm{Ch}_\kappa(x)\big(1+\mathrm{Th}_\kappa(x)A_e(\boldsymbol{\sigma})\big)$ gives the exact factorization
 \begin{align*}
  Z_N(\beta;J)&=\kappa^N Z_N^{\mathrm{ch}}(\beta)Z_N^{\mathrm{cl}}(\beta),
  \quad\text{where}\\
  Z_N^{\mathrm{ch}}(\beta):=\prod_{e\in E_N}\mathrm{Ch}_\kappa(\beta J_e/\sqrt N),\quad &
  Z_N^{\mathrm{cl}}(\beta):=\operatorname{\mathds{E}}_{\boldsymbol{\sigma}}\Big(\prod_{e\in E_N}
  \big(1+ \omega_eA_e(\boldsymbol{\sigma})\big)\Big),
 \end{align*}
 where $\operatorname{\mathds{E}}_{\boldsymbol{\sigma}}$ is the uniform average on $\Sigma_N$ and $\omega_e:=\mathrm{Th}_\kappa(\beta J_e/\sqrt N)$. For $\Gamma \subseteq E_N$ write $\omega(\Gamma):=\prod_{e\in\Gamma}\omega_e$. Expanding the product yields
 \begin{align*}
  Z_N^{\mathrm{cl}}(\beta)=\sum_{\Gamma\subseteq E_N}f_\kappa(\Gamma)\omega(\Gamma),
  \quad f_\kappa(\Gamma):=\operatorname{\mathds{E}}_{\boldsymbol{\sigma}}\Big(\prod_{e\in\Gamma}A_e(\boldsymbol{\sigma})\Big).
 \end{align*}
 If $\Gamma$ has a vertex of degree one, averaging over its color gives $f_\kappa(\Gamma)=0$. For a loop $\gamma$ of length $n$, $f_\kappa(\gamma)=\operatorname{Tr}(\Pi^n)=\kappa-1$.

 To identify the leading random terms, use $\mathrm{Th}_\kappa(x)=x+\mathcal{O}(x^2)$ and replace $\omega_e$ by the centered weight $\beta J_e/\sqrt N$. For a fixed graph with $v$ nonisolated vertices and $r$ edges, the sum over its copies has variance $\mathcal{O}(N^{v-r})$, since distinct edge sets give orthogonal monomials. The contributing nonempty graphs have minimum degree at least two, so $v\leqslant r$, with equality precisely for vertex-disjoint unions of loops. Their coefficients and edge products factor over the loops, suggesting the formal approximation
 \begin{align*}
  Z_N^{\mathrm{cl}}(\beta)\approx Z_{N,\ell}^{\mathrm{loop}}(\beta)
  :=\prod_{\substack{\gamma\text{ loop}\\3\leqslant\left\vert\gamma\right\vert\leqslant\ell}}
  \big(1+(\kappa-1)\omega(\gamma)\big).
 \end{align*}
 The linear term in the formal expansion of $\ln Z_{N,\ell}^{\mathrm{loop}}$, with each $\omega_e$ replaced by $\beta J_e/\sqrt N$, gives $(\kappa-1)\sum_{n=3}^{\ell}\beta^nY_{N,n}(J)$. The expansion $\ln\mathrm{Ch}_\kappa(x)=\frac{\kappa-1}{2}x^2+\mathcal{O}(x^3)$ supplies the centered quadratic term $(\kappa-1)\beta^2X_N(J)/2$, with $X_N$ and $Y_{N,n}$ defined in~\eqref{eq:array-cycle-statistics}. After truncation, these terms give the statistic used in the proof,
 \begin{align*}
  \widetilde G_{N,\ell}
  =\frac{(\kappa-1)\beta^2}{2}\widetilde X_N
  +(\kappa-1)\sum_{n=3}^{\ell}\beta^n\widetilde Y_{N,n},
 \end{align*}
 where $\widetilde X_N$ and $\widetilde Y_{N,n}$ are defined in~\eqref{eq:truncated-loop-statistics}. At $\kappa=2$, these are the quadratic and loop terms of the cluster expansion in~\cite{ALR87}.

\label{lem:bolthausen}
	Let $N\geqslant2$, let $(c(i,j))_{i,j=1}^{N}$ be a doubly centered real array, let $\pi$ be a uniform random permutation of $[N]$, and set $S := \sum_{i=1}^{N} c(i,\pi(i))$. Then
	\begin{align*}
		\operatorname{\mathds{E}}_\pi(S) = 0\quad\text{and}\quad v := \operatorname{Var}_\pi(S) = \frac{1}{N-1}\sum_{i,j} c(i,j)^2.
	\end{align*}
	If $v > 0$, then for an absolute constant $C$ and $\zeta \sim \operatorname{N}(0,1)$,
	\begin{align}
		\sup_{x \in \mathds{R}} \Big\lvert \operatorname{\mathds{P}}_\pi\big( S/\sqrt v \leqslant x \big) - \operatorname{\mathds{P}}\big( \zeta \leqslant x \big) \Big\rvert \leqslant \frac{C}{N v^{3/2}} \sum_{i,j} \left\vertc(i,j)\right\vert^3.
		\label{eq:comb-be}
	\end{align}

<!-- proof-3: body_md -->
The idea is as follows. Conditionally on $\boldsymbol{\tau}$, the overlap $\mathscr{R}(\boldsymbol{\sigma},\boldsymbol{\tau})$ is a linear statistic of a uniform permutation, to which Lemma~\ref{lem:bolthausen} applies. Its conditional variance depends on $\boldsymbol{\tau}$ only through its profile, up to a small error. For three replicas, the conditional covariance of $\mathscr{R}(\boldsymbol{\sigma},\boldsymbol{\tau})$ and $\mathscr{R}(\boldsymbol{\sigma},\boldsymbol{\tau}')$ is governed by $N^{-1/2}\mathscr{R}(\boldsymbol{\tau},\boldsymbol{\tau}')\to0$, which gives asymptotic independence.

	\smallskip\noindent\textbf{Step 1 (permutation representation).} Fix a word $w$ with profile $\boldsymbol{n}_N$ and represent $\sigma_i=w_{\pi(i)}$, where $\pi$ is a uniform permutation independent of $(\boldsymbol{\tau},\boldsymbol{\tau}')$. This represents the uniform law on $\Sigma_N^{(\boldsymbol{n}_N)}$, since every configuration of that profile arises from exactly $\prod_xn_x!$ permutations. For a fixed matrix $s$,
	\begin{align*}
		\langle s,\mathscr{R}(\boldsymbol{\sigma},\boldsymbol{\tau})\rangle_F
		=N^{-1/2}\sum_i c(i,\pi(i)),\quad
		c(i,j):=\phi(w_j)^{\intercal}s\phi(\tau_i).
	\end{align*}
	Write $c_{i\cdot},c_{\cdot j},\bar c$ for the row, column, and overall means, and define the doubly centered array
	\begin{align*}
		d(i,j):=c(i,j)-c_{i\cdot}-c_{\cdot j}+\bar c
		= \left(\phi(w_j)- \frac{m_{\boldsymbol{n}_N}}{\sqrt{N}}\right)^{\intercal}s\left(\phi(\tau_i)- \frac{m_{\boldsymbol{n}'_N}}{\sqrt{N}}\right).
	\end{align*}
	Then
	\begin{align*}
		\langle s,\mathscr{R}(\boldsymbol{\sigma},\boldsymbol{\tau})\rangle_F
		=N^{-1/2}\sum_i \left( d(i,\pi(i)) + c_{i \cdot} + c_{\cdot \pi(i)} - \bar c \right)
		=N^{-1/2}\sum_i d(i,\pi(i)) + N^{1/2} \bar c.
	\end{align*}
	Since $N\bar c=m_{\boldsymbol{n}_N}^{\intercal}s\, m_{\boldsymbol{n}'_N}$, the last term is $\mathcal{O}(N^{-1/2})$ uniformly on the window.

	\smallskip\noindent\textbf{Step 2 (two replicas).}
	Set $a_j :=\phi(w_j)- {m_{\boldsymbol{n}_N}}/{\sqrt{N}}$ and $b_i := \phi(\tau_i)- {m_{\boldsymbol{n}'_N}}/{\sqrt{N}}$. Then
	\begin{align*}
		d(i,j)^2 = (a_j^{\intercal}sb_i)(b_i^{\intercal}s^{\intercal}a_j)
		= \operatorname{Tr}(s^{\intercal}a_ja_j^{\intercal}sb_ib_i^{\intercal}).
	\end{align*}
	It follows that
	\begin{align*}
		\frac{1}{N(N-1)}\sum_{i,j} d(i,j)^2 = \frac{N}{N-1} \operatorname{Tr} \left[ s^{\intercal}\left(\frac{1}{N} \sum_j a_j a_j^{\intercal}\right)s\left(\frac{1}{N} \sum_i b_i b_i^{\intercal}\right)\right].
	\end{align*}
	Let $p = \boldsymbol{n}_N / N$. Using $p = u + (\kappa N)^{-1/2}m_{\boldsymbol{n}_N}$ and $m_{\boldsymbol{n}_N} = \mathcal{O}(1)$, we get
	\begin{align*}
		\frac{1}{N} \sum_j a_j a_j^{\intercal} = \sum_{x=1}^\kappa p_x \phi(x) \phi(x)^{\intercal} - \frac{1}{N} m_{\boldsymbol{n}_N} m_{\boldsymbol{n}_N}^{\intercal}
		\to \Pi.
	\end{align*}
	The same approximation applies to $b_i$, and we obtain the limit of the conditional variance:
	\begin{align*}
		\operatorname{Var}\Big( N^{-1/2}\sum_i d(i,\pi(i)) \,\Big|\, \boldsymbol{\tau} \Big)=\frac1{N(N-1)}\sum_{i,j}d(i,j)^2
		\longrightarrow \operatorname{Tr}(s^{\intercal}\Pi s\Pi)=\left\Vert\Pi s\Pi\right\Vert_F^2.
	\end{align*}
	This limit is deterministic, since it depends on $\boldsymbol{\tau}$ only through its profile. The arrays are uniformly bounded. If the limiting variance is positive, we apply Lemma~\ref{lem:bolthausen} to the array $d(i,j)/\sqrt N$. Since $\sum_{i,j}\left\vertd(i,j)\right\vert^3=\mathcal{O}(N^2)$, the bound~\eqref{eq:comb-be} is $\mathcal{O}(N^{-1/2})$. If the limiting variance is zero, Chebyshev's inequality applies. Since $\left\Vert\Pi s\Pi\right\Vert_F^2=\operatorname{Var}(\langle s,K_0\rangle_F)$, the Cram\'er--Wold device proves the two-replica assertion, uniformly along profile sequences in the window.

	\smallskip\noindent\textbf{Step 3 (three replicas).} Fix matrices $s_1,s_2,s_3$ and write
	\begin{align*}
		T_1:=\langle s_1,\mathscr{R}(\boldsymbol{\sigma},\boldsymbol{\tau})\rangle_F,\quad
		T_2:=\langle s_2,\mathscr{R}(\boldsymbol{\sigma},\boldsymbol{\tau}')\rangle_F,\quad
		T_3:=\langle s_3,\mathscr{R}(\boldsymbol{\tau},\boldsymbol{\tau}')\rangle_F.
	\end{align*}
	Conditionally on $(\boldsymbol{\tau},\boldsymbol{\tau}')$, $T_1+T_2$ is a single permutation statistic:
	\begin{align*}
		T_1 + T_2
		=N^{-1/2}\sum_i c'(i,\pi(i)),
		\quad c'(i,j):=\phi(w_j)^{\intercal}\big(s_1\phi(\tau_i)+s_2\phi(\tau'_i)\big).
	\end{align*}
	As in Steps 1 and 2, define the doubly centered array
	\begin{align*}
		d'(i,j):=c'(i,j)-c'_{i\cdot}-c'_{\cdot j}+\bar c'
		= \left(\phi(w_j)- \frac{m_{\boldsymbol{n}_N}}{\sqrt{N}}\right)^{\intercal} \left[s_1\left(\phi(\tau_i)- \frac{m_{\boldsymbol{n}'_N}}{\sqrt{N}}\right) + s_2\left(\phi(\tau'_i)- \frac{m_{\boldsymbol{n}''_N}}{\sqrt{N}}\right)\right].
	\end{align*}
	Then
	\begin{align*}
		T_1 + T_2
		=N^{-1/2}\sum_i d'(i,\pi(i)) + \mathcal{O}(N^{-1/2}).
	\end{align*}
	Recall $a_j$ and $b_i$ from Step 2 and set $b_i' := \phi(\tau_i')- {m_{\boldsymbol{n}''_N}}/{\sqrt{N}}$. Then
	\begin{align*}
		d'(i,j)^2
		= \operatorname{Tr}(s_1^{\intercal}a_ja_j^{\intercal}s_1b_ib_i^{\intercal})
		+\operatorname{Tr}(s_2^{\intercal}a_ja_j^{\intercal}s_2b_i'(b_i')^{\intercal})
		+2\operatorname{Tr}(s_2^{\intercal}a_ja_j^{\intercal}s_1b_i(b_i')^{\intercal}).
	\end{align*}
	Moreover, $N^{-1}\sum_i b_i (b_i')^{\intercal} =N^{-1/2}\mathscr{R}(\boldsymbol{\tau},\boldsymbol{\tau}')+ \mathcal{O}(1/N)$, which tends to zero in probability by the two-replica assertion. Hence
	\begin{align*}
		\operatorname{Var} \big(T_1 + T_2 \,\big|\, \boldsymbol{\tau}, \boldsymbol{\tau}' \big) \longrightarrow \left\Vert\Pi s_1\Pi\right\Vert_F^2 + \left\Vert\Pi s_2\Pi\right\Vert_F^2\quad\text{in probability}.
	\end{align*}
	If this limit is positive, Lemma~\ref{lem:bolthausen} applies on an event where the conditional variance is bounded below, and the complement of this event has probability tending to zero. The conditional characteristic function of $T_1+T_2$ therefore converges in probability, and in $L^1$ by boundedness, to the deterministic Gaussian characteristic function. If the limit vanishes, the same conclusion follows from the conditional second moment. Since $T_3$ is measurable with respect to $(\boldsymbol{\tau},\boldsymbol{\tau}')$,
	\begin{align*}
		\operatorname{\mathds{E}}\big(e^{\mathrm{i}(T_1+T_2+T_3)}\big)
		=\operatorname{\mathds{E}}\Big(e^{\mathrm{i} T_3}\,\operatorname{\mathds{E}}\big(e^{\mathrm{i}(T_1+T_2)}\,\big|\,\boldsymbol{\tau},\boldsymbol{\tau}'\big)\Big),
	\end{align*}
	and this factorizes in the limit into the product of the two limiting Gaussian characteristic functions. The two-replica result for $T_3$ and the Cram\'er--Wold device complete the proof.

Fix $\boldsymbol{\tau}$. The configurations $\boldsymbol{\sigma}\in\Sigma_N^{(\boldsymbol{n})}$ with pair type $q$ are obtained by choosing, within each color class $\{i:\tau_i=y\}$ of $\boldsymbol{\tau}$, which sites receive each color of $\boldsymbol{\sigma}$, with prescribed counts $(Nq_{xy})_x$. Dividing this count by $\left\vert\Sigma_N^{(\boldsymbol{n})}\right\vert=\binom N{(n_x)_x}$ gives the exact ratio. Write $\mathcal{S}(p):=-\sum_xp_x\ln p_x$. The multinomial bounds of the method of types~\cite[Lemma 2.1.8]{DZ98},
	\begin{align*}
		(n+1)^{-\kappa}e^{n\mathcal{S}(a/n)}\leqslant\binom n{(a_x)_x}\leqslant e^{n\mathcal{S}(a/n)},
	\end{align*}
	and the identity
	\begin{align}
		\sum_y n'_y\mathcal{S}((Nq_{xy}/n'_y)_x)-N\mathcal{S}(p)
		&=-N\sum_y p'_y\sum_x\frac{q_{xy}}{p'_y}\ln\frac{q_{xy}}{p'_y}
		+N\sum_{x,y}q_{xy}\ln p_x\nonumber\\
		&=-ND(q\|p\otimes p')
		\label{eq:profile-type-entropy}
	\end{align}
	prove~\eqref{eq:profile-types}.
	On the core, each $Nq_{xy}$ is a positive integer. Stirling's bounds
	\begin{align*}
		\sqrt{2\pi n}\,(n/e)^n\leqslant n!\leqslant\sqrt{2\pi n}\,(n/e)^n e^{1/(12n)}
	\end{align*}
	yield
	\begin{align*}
		\binom{n'_y}{(Nq_{xy})_x}
		&\leqslant e^{1/12}(2\pi)^{-(\kappa-1)/2}(n'_y)^{-(\kappa-1)/2}
		\left(\prod_x\frac{Nq_{xy}}{n'_y}\right)^{-1/2}
		e^{n'_y\mathcal{S}((Nq_{xy}/n'_y)_x)}\\
		&\leqslant C N^{-(\kappa-1)/2}e^{n'_y\mathcal{S}((Nq_{xy}/n'_y)_x)}.
	\end{align*}
	Here we used $Nq_{xy}/n'_y\geqslant q_{xy}\geqslant1/(2\kappa^2)$ and $n'_y=N\sum_xq_{xy}\geqslant N/(2\kappa)$ on the core. The same computation, with the two Stirling bounds interchanged, bounds the denominator in~\eqref{eq:profile-types} from below:
	\begin{align*}
		\binom N{(n_x)_x}\geqslant cN^{-(\kappa-1)/2}e^{N\mathcal{S}(p)}.
	\end{align*}
	Combining these bounds with the entropy identity~\eqref{eq:profile-type-entropy} proves~\eqref{eq:profile-types-sharp}.

Write $p=\boldsymbol{n}/N$ and $p'=\boldsymbol{n}'/N$, and let $\mathcal{T}_N$ be the set of realizable pair types with these margins. Conditioning on the pair type and using $\mathscr{R}=\kappa\sqrt N\hat q$ in the definition of $\xi$, we obtain the type expansion

		=e^{-\frac{\beta^2}2(\kappa-1)^2}
		\sum_{q\in\mathcal{T}_N}\operatorname{\mathds{P}}_{\boldsymbol{\sigma},\boldsymbol{\tau}}^{(\boldsymbol{n}),(\boldsymbol{n}')}
		\big(p(\boldsymbol{\sigma},\boldsymbol{\tau})=q\big)
		e^{\frac{\beta^2\kappa^2N}{2}\left\Vert\hat q\right\Vert_F^2}.
	\end{align*}

	(i) \smallskip\noindent\textbf{Step 1 (energy and entropy).} Define $w=q-p\otimes p'$ and $\delta=\left\Vertp-u\right\Vert_\infty\vee\left\Vertp'-u\right\Vert_\infty\leqslant L/\sqrt{\kappa N}$. Since $w$ is doubly centered and $\Pi p=p-u$,
	\begin{align*}
		\hat q-w=\Pi(p\otimes p')\Pi=(p-u)(p'-u)^{\intercal}
		\quad\text{and}\quad
		\left\Vert\hat q-w\right\Vert_F\leqslant\kappa\delta^2.
	\end{align*}
	The resulting energy error is controlled by
	\begin{align*}
		\left\Vert\hat q\right\Vert_F^2-\left\Vertw\right\Vert_F^2
		=2\langle\hat q-w,w\rangle_F+\left\Vert\hat q-w\right\Vert_F^2
		\leqslant2\kappa\delta^2\left\Vertw\right\Vert_F+\kappa^2\delta^4.
	\end{align*}
	Set $a:=(\alpha_c(\kappa)-\beta^2)\kappa^2/2>0$. Combining this estimate with Lemma~\ref{lem:near-uniform} and using $\delta\leqslant L/\sqrt{\kappa N}$, we get, for $N\geqslant C(1+L^2)$,
	\begin{align*}
		\frac{\beta^2\kappa^2N}{2}\left\Vert\hat q\right\Vert_F^2-ND(q\|p\otimes p')
		\leqslant-Na\left\Vertw\right\Vert_F^2+C\sqrt N L\left\Vertw\right\Vert_F+CL^2.
	\end{align*}
	Completing the square,
	\begin{align*}
		-\frac{Na}{2}\left\Vertw\right\Vert_F^2+C\sqrt N L\left\Vertw\right\Vert_F
		=-\frac{Na}{2}\left(\left\Vertw\right\Vert_F-\frac{CL}{a\sqrt N}\right)^2
		+\frac{C^2L^2}{2a}
		\leqslant CL^2.
	\end{align*}
	Consequently,
	\begin{align}
		\frac{\beta^2\kappa^2N}{2}\left\Vert\hat q\right\Vert_F^2-ND(q\|p\otimes p')
		\leqslant-\frac{Na}{2}\left\Vertw\right\Vert_F^2+CL^2.

	\smallskip\noindent\textbf{Step 2 (lattice sum over core types).} A table with fixed margins is determined by its entries with $x,y<\kappa$. Each $Nq_{xy}$ is an integer, and $\left\Vertw\right\Vert_F^2\geqslant\sum_{x,y<\kappa}(q_{xy}-p_xp'_y)^2$. Thus~\eqref{eq:profile-types-sharp} and~\eqref{eq:profile-exponent} give

		\sum_{q\text{ core}}\operatorname{\mathds{P}}_{\boldsymbol{\sigma},\boldsymbol{\tau}}^{(\boldsymbol{n}),(\boldsymbol{n}')}
		\big(p(\boldsymbol{\sigma},\boldsymbol{\tau})=q\big)e^{\frac{\beta^2\kappa^2N}{2}\left\Vert\hat q\right\Vert_F^2}
		&\leqslant C e^{CL^2}N^{-(\kappa-1)^2/2}

		&\leqslant C e^{CL^2}N^{-(\kappa-1)^2/2}
		\prod_{x,y<\kappa}\left(\sum_{j\in\mathds{Z}}e^{-\frac a{2N}(j-Np_xp'_y)^2}\right)
		\leqslant Ce^{CL^2}.
	\end{align*}

	\begin{align*}
		\sup_{b\in\mathds{R}}\sum_{j\in\mathds{Z}}e^{-\frac a{2N}(j-b)^2}
		\leqslant2+\sqrt{\frac{2\pi N}{a}}\leqslant C\sqrt N.
	\end{align*}
	There are $(\kappa-1)^2$ such factors, so their powers of $N$ cancel the prefactor, uniformly over the profiles.

	\smallskip\noindent\textbf{Step 3 (remaining types).} If $q$ is not in the core, some cell has $q_{xy}<1/(2\kappa^2)$. 	\begin{align*}
		\sum_{q\text{ not core}}\operatorname{\mathds{P}}_{\boldsymbol{\sigma},\boldsymbol{\tau}}^{(\boldsymbol{n}),(\boldsymbol{n}')}
		\big(p(\boldsymbol{\sigma},\boldsymbol{\tau})=q\big)e^{\frac{\beta^2\kappa^2N}{2}\left\Vert\hat q\right\Vert_F^2}
		\leqslant e^{CL^2}(N+1)^{\kappa+\kappa^2}
		e^{-Na/(32\kappa^4)}\longrightarrow0,
	\end{align*}
	uniformly over the profiles. In particular, this sum is bounded by $C$ uniformly in $N$. For fixed $L$, the finitely many smaller $N$ also give finite expectations, since their configuration spaces are finite. Thus Steps 2 and 3 prove (i).

	(ii) Choose $\beta^2<s<\alpha_c(\kappa)$. Part (i) with $\beta^2$ replaced by $s$ bounds the expectation of $\big(e^{\beta^2\xi(\mathscr{R})}\big)^{s/\beta^2}=e^{s\xi(\mathscr{R})}$ uniformly over $N$ and the profile window. Hence these exponential weights are uniformly integrable. By Lemma~\ref{lem:profile-clt}, $\mathscr{R}\Longrightarrow K_0$ along every sequence of profiles in the window. The continuous mapping theorem, uniform integrability, and Lemma~\ref{lem:Zsecond} therefore give
	\begin{align*}
		\operatorname{\mathds{E}}_{\boldsymbol{\sigma},\boldsymbol{\tau}}^{(\boldsymbol{n}),(\boldsymbol{n}')}\big(e^{\beta^2\xi(\mathscr{R})}\big)
		\longrightarrow\operatorname{\mathds{E}}\big(e^{\beta^2\xi(K_0)}\big)
		=e^{-\frac{\beta^2}2(\kappa-1)^2}(1-\beta^2)^{-(\kappa-1)^2/2}
		=e^{\varphi_\kappa(\beta^2)}.
	\end{align*}
	If the uniform convergence in (ii) failed, some sequence of profile pairs in the window would have errors bounded below by a fixed positive number along a subsequence. This contradicts the convergence just proved.

	\begin{align*}
		q^*_{xx}=\frac{\kappa-1}{\kappa^2},\qquad
		q^*_{xy}=\frac1{\kappa^2(\kappa-1)}\quad(y\ne x).
	\end{align*}

(i) Two distinct loops have an edge that lies in only one of them. Independence and centering of the edges therefore make their monomials orthogonal, which proves~\eqref{eq:Wcov}. Moreover,
	\begin{align*}

	\end{align*}
	where the inner sum runs over the three pairings of $[4]$.

Each pairing term vanishes as well, since a nonzero pairing term requires two pairs of identical loops, and then every edge has even multiplicity. Hence $\mathcal{K}(\boldsymbol{q}) = 0$ unless $\nu_e \geqslant 2$ for all $e$. In that case the joint term and each nonzero pairing term are bounded in absolute value by

		\prod_{e \in \Gamma} C N^{-\nu_e/2} = C^{\left\vert\Gamma\right\vert} N^{-2n} \leqslant C(n) N^{-2n},
	\end{align*}
	so $\left\vert\mathcal{K}(\boldsymbol{q})\right\vert \leqslant C(n)N^{-2n}$. Since $\nu_e \geqslant 2$ for all $e$, we have $\left\vert\Gamma\right\vert \leqslant 2n$. 			\Big( e_N(\boldsymbol{\sigma}) e_N(\boldsymbol{\tau})f\big(\mathscr{R}(\boldsymbol{\sigma},\boldsymbol{\tau})\big)\Big)
		- \mathcal{Z}_N^{(\boldsymbol{n})} \mathcal{Z}_N^{(\boldsymbol{n}')}\,\operatorname{\mathds{E}}\big(f(K)\big)
			\to0\quad\text{in }L^1,
		\label{eq:general-overlap-factorization}
	\end{align}
	where $K\sim\operatorname{N}(0,\Pi\otimes\Pi/(1-\beta^2))$ as in Theorem~\ref{thm:C}. The fourth-cumulant factors have canceled.

	\emph{Balanced replicas.} Take both profiles equal to $\boldsymbol{n}^*$ with $\kappa\mid N$. By~\eqref{eq:general-profile-lognormal}, we may divide~\eqref{eq:general-overlap-factorization} by $\big( \mathcal{Z}_N^{\operatorname{bal}}\big)^2$ in probability. Since $c_N$ is constant on a profile class, the ratio is the overlap average under two independent balanced Gibbs replicas. This gives the balanced overlap limit throughout $\beta^2<\alpha_c(\kappa)$.

	\emph{Unconstrained replicas.} Let $h_1,h_2$ be bounded continuous functions on $\mathds{R}^{\kappa}$, and first restrict to $\left\Vertm_{\boldsymbol{n}}\right\Vert_2,\left\Vertm_{\boldsymbol{n}'}\right\Vert_2\leqslant L$. Divide by $\big(\sum_{\boldsymbol{n}} P_{\boldsymbol{n}} w_{\boldsymbol{n}} \mathcal{Z}_N^{(\boldsymbol{n})}\big)^2$, whose inverse is bounded in probability by~\eqref{eq:general-gibbs-denominator}. 	\begin{align*}
		&\Big\langle h_1(m(\boldsymbol{\sigma}^1))h_2(m(\boldsymbol{\sigma}^2))f\big(\mathscr{R}(\boldsymbol{\sigma}^1,\boldsymbol{\sigma}^2)\big)\chi_L(\boldsymbol{\sigma}^1)\chi_L(\boldsymbol{\sigma}^2)\Big\rangle_{\varrho_{N,\beta}^{\otimes2}}\\
		&\quad-\Big\langle h_1(m)\chi_L\Big\rangle_{\varrho_{N,\beta}}
		\Big\langle h_2(m)\chi_L\Big\rangle_{\varrho_{N,\beta}}\,\operatorname{\mathds{E}}\big(f(K)\big)
		\longrightarrow0.
	\end{align*}
	Removing the cutoffs changes both terms by at most a constant times $\langle\mathds{1}\{\left\Vertm\right\Vert_2>L\}\rangle_{\varrho_{N,\beta}}$. Letting $L\to\infty$ and using the tail bound~\eqref{eq:general-color-tail}, we obtain, in probability,
	\begin{align*}
		\Big\langle h_1(m(\boldsymbol{\sigma}^1))h_2(m(\boldsymbol{\sigma}^2))f\big(\mathscr{R}(\boldsymbol{\sigma}^1,\boldsymbol{\sigma}^2)\big)\Big\rangle_{\varrho_{N,\beta}^{\otimes2}}
		-\Big\langle h_1(m)\Big\rangle_{\varrho_{N,\beta}}
		\Big\langle h_2(m)\Big\rangle_{\varrho_{N,\beta}}\,\operatorname{\mathds{E}}\big(f(K)\big)
		\longrightarrow0.
	\end{align*}
	The color limits~\eqref{eq:general-color-clt} now yield
	\begin{align*}
		\Big\langle h_1(m(\boldsymbol{\sigma}^1))h_2(m(\boldsymbol{\sigma}^2))f\big(\mathscr{R}(\boldsymbol{\sigma}^1,\boldsymbol{\sigma}^2)\big)\Big\rangle_{\varrho_{N,\beta}^{\otimes2}}

	\end{align*}
	with $g,g',K$ independent as in Theorem~\ref{thm:C}.

	It remains to pass from products to general $F$. The marginal limits give tightness in probability of the Gibbs law of $(m(\boldsymbol{\sigma}^1),m(\boldsymbol{\sigma}^2),\mathscr{R}(\boldsymbol{\sigma}^1,\boldsymbol{\sigma}^2))$. Finite sums of products $h_1(x)h_2(y)f(z)$ form an algebra that separates points and contains the constants. By the Stone--Weierstrass theorem they approximate every continuous function uniformly on compact sets. Together with tightness, the factorization therefore proves the joint weak convergence in probability. The identity $m=\sqrt{\kappa N}(p-u)$ also gives the corresponding limit for the color fractions. Finally, the coupling~\eqref{eq:truncation-coupling} transfers all assertions to the original disorder.

\label{lem:expmoment}
	Let $W, W_1, W_2, \ldots$ be i.i.d.~mean-zero random vectors in $\mathds{R}^d$ with $\left\VertW\right\Vert_2 \leqslant b$ almost surely. Set $S_N := N^{-1/2} \sum_{i=1}^{N} W_i$, and define the \emph{optimal sub-Gaussian constant} of $W$ by
	\begin{align}
		\eta_W := \sup_{s \in \mathds{R}^d \setminus \{0\}} \frac{2 \ln \operatorname{\mathds{E}}\big(e^{\langle s, W \rangle}\big)}{\left\Verts\right\Vert_2^2}.
		\label{eq:eta-def}
	\end{align}
	Then $\eta_W \leqslant b^2$. If $\eta_W < 1$, then for every $N \geqslant 1$,
	\begin{align*}
		\operatorname{\mathds{E}}\Big( \exp\Big( \frac{1}{2} \left\VertS_N\right\Vert_2^2 \Big)\Big) \leqslant \left( 1 - \eta_W \right)^{-d/2}.
	\end{align*}

<!-- proof-20: body_md -->
Write $F(s) := \ln \operatorname{\mathds{E}}\big(e^{\langle s, W \rangle}\big)$. By Hoeffding's lemma~\cite{Hoe63}, every mean-zero random variable $X$ with $\left\vertX\right\vert \leqslant M$ almost surely satisfies $\operatorname{\mathds{E}}(e^{X}) \leqslant e^{M^2/2}$. The variable $X = \langle s, W \rangle$ is mean-zero and satisfies $\left\vertX\right\vert \leqslant \left\Verts\right\Vert_2\, b$ by the Cauchy--Schwarz inequality. Hence $F(s) \leqslant \left\Verts\right\Vert_2^2 b^2/2$ and $\eta_W \leqslant b^2 < \infty$. By the definition of $\eta_W$,
	\begin{align}
		F(s) \leqslant \frac{\eta_W}{2} \left\Verts\right\Vert_2^2 \quad \text{for every } s \in \mathds{R}^d.
		\label{eq:subgaussian}
	\end{align}
	Now let $\zeta$ be a standard Gaussian vector in $\mathds{R}^d$, independent of $(W_i)_{i \geqslant 1}$. By the Hubbard--Stratonovich identity~\cite{Hub59}, $e^{\left\Vertx\right\Vert_2^2/2} = \operatorname{\mathds{E}}_\zeta\big(e^{\langle \zeta, x \rangle}\big)$, and by~\eqref{eq:subgaussian} with $s = \zeta/\sqrt{N}$,
	\begin{align*}
		\operatorname{\mathds{E}}\Big( \exp\Big( \frac{1}{2} \left\VertS_N\right\Vert_2^2 \Big)\Big)
		= \operatorname{\mathds{E}}_\zeta\Big(  \operatorname{\mathds{E}}_W \big( e^{\langle \zeta, S_N \rangle} \big)\Big)
		= \operatorname{\mathds{E}}_\zeta\big(  e^{N F(\zeta/\sqrt N)}\big)
		\leqslant \operatorname{\mathds{E}}_\zeta\big(  e^{\frac{\eta_W}{2} \left\Vert\zeta\right\Vert_2^2}\big)
		= \left( 1 - \eta_W \right)^{-d/2}.
	\end{align*}
	The last equality uses $\eta_W < 1$.

\label{lem:eta-dual}
	Let $W$ be a mean-zero random vector on a finite probability space $(\mathcal{X}, \pi)$, and let $\mathcal{P}(\mathcal{X})$ be the set of probability measures on $\mathcal{X}$. Then
	\begin{align*}
		\eta_W = \sup_{q \in\mathcal{P}(\mathcal{X}),\, q\neq \pi} \frac{\left\Vert\operatorname{\mathds{E}}_q(W)\right\Vert_2^2}{2  D(q  \|  \pi)}.
	\end{align*}
<!-- end lem-18 -->
<!-- proof-21: body_md -->
The Gibbs variational principle on the finite space $(\mathcal{X},\pi)$ states that, for every function $f$,
	\begin{align*}
		\ln \operatorname{\mathds{E}}_\pi\big(e^{f}\big) = \sup_{q\in\mathcal{P}(\mathcal{X})} \big[ \operatorname{\mathds{E}}_q(f) - D(q\|\pi) \big],
	\end{align*}
	see~\cite{DZ98}. Apply it to $f = \langle s, W\rangle$. By the definition~\eqref{eq:eta-def}, for every $c>0$,
	\begin{align*}
		\eta_W \leqslant c
		\Leftrightarrow \forall q, s : \langle s, \operatorname{\mathds{E}}_q(W) \rangle - \frac{c}{2}\left\Verts\right\Vert_2^2 \leqslant D(q\|\pi)
		\Leftrightarrow \forall q : \frac{\left\Vert\operatorname{\mathds{E}}_q(W)\right\Vert_2^2}{2c} \leqslant D(q\|\pi).
	\end{align*}
	The last step maximizes the quadratic function of $s$. Since this equivalence holds for every $c > 0$, the claim follows.

\emph{Case $\kappa = 2$.} Here $\phi(\sigma) = \varepsilon (1,-1)/\sqrt 2$ with $\varepsilon$ a Rademacher variable. Thus $\Lambda(t) = \ln\cosh\big( (t_1 - t_2)/\sqrt 2 \big) \leqslant (t_1-t_2)^2/4 \leqslant \left\Vertt\right\Vert_2^2/2$, and the ratio tends to $1$ along $t = (a, -a)$ as $a \to 0$. Hence the supremum equals $1$. Assume $\kappa \geqslant 3$ from now on.

	\smallskip\noindent\textbf{Step 1 (reduction to centered vectors).} We have $\langle t, \phi(\sigma) \rangle = \sqrt\kappa\,  (t_\sigma - \bar t)$ with $\bar t := \kappa^{-1}\sum_x t_x$. So $\Lambda$ depends only on the centered part $\hat t := t - \bar t\,  \mathbf{1} $, while $\left\Vert\hat t\right\Vert_2 \leqslant \left\Vertt\right\Vert_2$. We may therefore restrict to centered $t$. Writing $c := \sqrt\kappa\,  \hat t \in \mathcal{V}$, the claim becomes
	\begin{align*}
		\sup_{c \in \mathcal{V} \setminus \{0\}} Q(c) = \eta_\phi(\kappa), \quad Q(c) := \frac{2\kappa  L(c)}{\left\Vertc\right\Vert_2^2}, \quad L(c) := \ln\Big( \frac1\kappa \sum_{x} e^{c_x} \Big).
	\end{align*}

	\smallskip\noindent\textbf{Step 2 (a binary sub-problem).} For $\rho \in (0,1)$ let $\Lambda_\rho(s) := \ln\big( \rho  e^{(1-\rho)s} + (1-\rho)  e^{-\rho s} \big)$ be the cumulant generating function of a centered Bernoulli$(\rho)$ variable, and define
	\begin{align*}
		h(\rho) := \frac{2\rho(1-\rho)\ln\frac{1-\rho}{\rho}}{1 - 2\rho} \quad (\rho \neq \tfrac12), \quad h(\tfrac12) := 1.
	\end{align*}
	The optimal sub-Gaussian constant of such a variable is known exactly:
	\begin{align}
		\sup_{s \neq 0} \frac{2 \Lambda_\rho(s)}{\rho(1-\rho) s^2} = \frac{1}{h(\rho)}, \quad \text{attained at } s^* = 2\ln\tfrac{1-\rho}{\rho} \text{ for } \rho \neq \tfrac12.
		\label{eq:binary-sup}
	\end{align}
	The inequality $\Lambda_\rho(s) \leqslant \tfrac{\rho(1-\rho)}{2h(\rho)}s^2$, with the constant $\rho(1-\rho)/h(\rho) = (1-2\rho)/\big(2\ln\tfrac{1-\rho}\rho\big)$, is the Kearns--Saul inequality~\cite{KS98}. See~\cite{BK13,BM13} for complete proofs and the optimality of the constant. At $\rho = \tfrac12$ it is the classical bound $\ln\cosh(s/2) \leqslant s^2/8$. That the supremum is attained at $s^*$ follows by substitution. Indeed, $e^{s^*} = \big(\tfrac{1-\rho}\rho\big)^2$ gives $\rho e^{s^*} + 1 - \rho = \tfrac{1-\rho}\rho$, so that
	\begin{align*}
		\Lambda_\rho(s^*) = -\rho s^* + \ln\tfrac{1-\rho}{\rho} = \tfrac{1-2\rho}{2}\,  s^* = \tfrac{\rho(1-\rho)}{2h(\rho)} (s^*)^2,
	\end{align*}
	where the last step uses the definition of $h$. For $\rho = \tfrac12$ the supremum is only approached as $s \to 0$.

	\smallskip\noindent\textbf{Step 3 (two-valued reduction).} We first record the lower bound. Let $c^* := 2\ln(\kappa-1) \big( e_1 - \tfrac1\kappa \mathbf{1}  \big)$. It takes one value on a single coordinate and another value on the remaining $\kappa - 1$ coordinates. The computation of Step 4 below with $j = 1$ and the equality case of~\eqref{eq:binary-sup} give $Q(c^*) = 1/h(1/\kappa) = \eta_\phi(\kappa)$. Moreover $\eta_\phi(\kappa) > 1$ for $\kappa \geqslant 3$. With $x := \kappa - 1 > 1$ this reads $x^2 - 1 > 2x\ln x$, that is, $x - 1/x - 2\ln x > 0$. This holds since the left side vanishes at $x = 1$ and has derivative $(1 - 1/x)^2 > 0$.

	As $c \to 0$ in $\mathcal{V}$, we have $L(c) = \left\Vertc\right\Vert_2^2/(2\kappa) + \mathcal{O}(\left\Vertc\right\Vert_2^3)$ because $\sum_x c_x = 0$, so $Q(c) \to 1$. Also $L(c) \leqslant \max_x c_x \leqslant \left\Vertc\right\Vert_2$, so $Q(c) \to 0$ as $\left\Vertc\right\Vert_2 \to \infty$. Since $\sup Q \geqslant \eta_\phi(\kappa) > 1$, every maximizing sequence stays in a compact subset of $\mathcal{V} \setminus \{0\}$. Hence the supremum is attained at some $c^0 \neq 0$, which is a critical point of $Q$ on $\mathcal{V}$. Let $\nabla_{\mathcal{V}}=\Pi \nabla$ denote the gradient along $\mathcal{V}$. Since $\partial_x L(c) = e^{c_x}/\sum_y e^{c_y}$ and these $\kappa$ numbers sum to one,
	\begin{align*}
		\big( \nabla_{\mathcal{V}} L(c) \big)_x = \frac{e^{c_x}}{\sum_y e^{c_y}} - \frac1\kappa.
	\end{align*}
	The criticality condition for $Q$ reads $\nabla_{\mathcal{V}} L(c^0) = 2\big( L(c^0)/\left\Vertc^0\right\Vert_2^2 \big)\,  c^0$. Multiplying its $x$-th component by $\sum_y e^{c^0_y}$ gives
	\begin{align*}
		e^{c^0_x} = A  c^0_x + B \quad \text{for every } x \in [\kappa],
	\end{align*}
	where $A$ and $B$ depend on $c^0$ but not on $x$.
	By the strict convexity of the exponential, this equation has at most two solutions. Hence $c^0$ takes exactly two distinct values, since a single value would force $c^0 = 0$.

	\smallskip\noindent\textbf{Step 4 (lumping and optimization over $j$).} Let $c \in \mathcal{V}$ take the value $a$ on $j$ coordinates and $b$ on the remaining $\kappa - j$ coordinates, with $1 \leqslant j \leqslant \kappa - 1$ and $j a + (\kappa - j) b = 0$. 	\begin{align*}

	\end{align*}
	whence $Q(c) = 2\Lambda_\rho(s) / (\rho(1-\rho) s^2)$. By~\eqref{eq:binary-sup}, $\sup Q = \max_{1 \leqslant j \leqslant \kappa-1} 1/h(j/\kappa)$. 	\begin{align*}
		\frac{d h}{d w} = \frac1w - \frac{(1+w^2)\tanh^{-1}(w)}{w^2} < 0
		\Leftrightarrow (1+w^2)\tanh^{-1}(w) > w,
	\end{align*}
	which holds for $w \in (0,1)$ since $\tanh^{-1}(w) > w$. Hence the maximum over $j$ is attained at $j = 1$ (equivalently $j = \kappa - 1$), and

	\end{align*}
	The maximizer is the two-valued vector with $j = 1$ and gap $s^* = 2\ln(\kappa-1)$, namely $c^* = 2\ln(\kappa-1)(e_{x_0} - \tfrac1\kappa\mathbf{1} )$. It corresponds to $t = c^*/\sqrt\kappa = \tfrac{2\ln(\kappa-1)}{\kappa}\,  \phi(x_0)$. Its tilted law is proportional to $(e^{c^*_x})_x$, which is exactly $p^*$. The idea is to apply~\eqref{eq:row-dual} to each conditional row and to average, using the chain rule for relative entropy. The error terms come from replacing $u$ by $p'$ and $1/p_x$ by $\kappa$. Write $p^{(x)} := q_{x\cdot}/p_x \in \mathcal{P}([\kappa])$ for the $x$-th conditional row of $q$. Since $\kappa u_y = 1$,
	\begin{align*}
		D(p^{(x)}\|p') - D(p^{(x)}\|u) = - \sum_y p^{(x)}_y \ln(\kappa p'_y) = - \kappa \big\langle p^{(x)}, p'-u \big\rangle - \sum_y p^{(x)}_y \Big( \ln(\kappa p'_y) - \kappa (p'_y - u_y) \Big).
	\end{align*}
	For the first term, the Cauchy--Schwarz inequality and $\langle u, p'-u\rangle = 0$ give
	\begin{align*}
		\left\vert\big\langle p^{(x)}, p'-u \big\rangle\right\vert
		 & = \left\vert\big\langle p^{(x)} - p',  p'-u \big\rangle + \big\langle p',  p'-u \big\rangle\right\vert \\
		 & \leqslant \left\Vertp^{(x)} - p'\right\Vert_2 \left\Vertp' - u\right\Vert_2 + \left\Vertp' - u\right\Vert_2^2
		\leqslant C\delta  \left\Vertp^{(x)} - p'\right\Vert_2 + C\delta^2.
	\end{align*}
	For the second term, note that $\ln(\kappa p'_y) = \ln\big(1 + \kappa(p'_y - u_y)\big)$ and $\left\vert\kappa(p'_y-u_y)\right\vert \leqslant \kappa\delta \leqslant \tfrac12$. Hence $\left\vert\ln(\kappa p'_y) - \kappa(p'_y - u_y)\right\vert \leqslant C\delta^2$, and $\sum_y p^{(x)}_y = 1$.
	We combine these bounds with~\eqref{eq:row-dual} and with $\left\Vertp^{(x)}-u\right\Vert_2^2 \geqslant \left\Vertp^{(x)}-p'\right\Vert_2^2 - 2\sqrt\kappa\, \delta\left\Vertp^{(x)}-p'\right\Vert_2$, which follows from $\left\Vertp'-u\right\Vert_2\leqslant\sqrt\kappa\,\delta$. This gives
	\begin{align*}
		D(p^{(x)}\|p')  \geqslant \frac{\alpha_c\kappa}{2}  \left\Vertp^{(x)}-p'\right\Vert_2^2 - C\delta  \left\Vertp^{(x)} - p'\right\Vert_2 - C\delta^2.
	\end{align*}

	Now average over rows. By the chain rule, $D(q\|p\otimes p') = \sum_x p_x  D(p^{(x)}\|p')$. Since $p^{(x)}_y - p'_y = (q_{xy} - p_x p'_y)/p_x$,
	\begin{align*}
		\sum_x p_x \left\Vertp^{(x)} - p'\right\Vert_2^2 = \sum_{x,y} \frac{w_{xy}^2}{p_x}.
	\end{align*}
	The margin bound $\kappa^{-1}(1 - \kappa\delta) \leqslant p_x \leqslant \kappa^{-1}(1 + \kappa\delta)$ bounds this quantity below by $\kappa(1-\kappa\delta)\left\Vertw\right\Vert_F^2$ and above by $C\left\Vertw\right\Vert_F^2$.
	The lower bound produces the error $\alpha_c\kappa^3\delta\left\Vertw\right\Vert_F^2/2$, which is at most $C\delta\left\Vertw\right\Vert_F$ because $\left\Vertw\right\Vert_F\leqslant\left\Vertq\right\Vert_F+\left\Vertp\otimes p'\right\Vert_F\leqslant2$. The upper bound and the Cauchy--Schwarz inequality in $x$ bound the averaged linear term by $C\delta\left\Vertw\right\Vert_F$. Together these estimates give~\eqref{eq:near-uniform}.
