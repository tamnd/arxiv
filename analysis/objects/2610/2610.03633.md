<!-- assumption-1: body_md -->
\label[assumption]{ass:primitive}
The lower objective $g(x,\cdot)$ is $\mu$-strongly convex.  Moreover,
\[
\left\lVert f_y(x,y)\right\rVert\le C_f,
\qquad
\operatorname{Lip}(\nabla f)\le L_f,
\qquad
\operatorname{Lip}(\nabla g)\le L_g,
\]
and the blocks $g_{yy}$ and $g_{xy}$ are jointly $\rho$-Lipschitz in $(x,y)$.  The initial hyperobjective gap satisfies $F(x_0)-\inf F\le\Delta$.
<!-- end assumption-1 -->
<!-- definition-1: body_md -->
\label{def:sfo}
At an adaptively chosen query $(x,y)$, an SFO returns random vectors $(\widehat\nabla f,\widehat\nabla g)$ satisfying
\[
\mathbb E[\widehat\nabla f\mid\mathcal F, x,y]=\nabla f(x,y),
\qquad
\mathbb E[\widehat\nabla g\mid\mathcal F, x,y]=\nabla g(x,y),
\]
\[
\mathbb E[\left\lVert \widehat\nabla f-\nabla f\right\rVert^2\mid\mathcal F,x,y]\le\sigma_f^2,
\qquad
\mathbb E[\left\lVert \widehat\nabla g-\nabla g\right\rVert^2\mid\mathcal F,x,y]\le\sigma_g^2.
\]
Each call uses a fresh sample and returns only the two gradient vectors at its queried point.  For a conservative SFO, every sample response is the gradient of a globally defined scalar function.
<!-- end definition-1 -->
<!-- theorem-1: body_md -->
\label{thm:main-lower-informal}
Fix $1\le\chi\lesssim\kappa$, a positive Hessian-variation budget $\rho\gtrsim\mu$, and sufficiently small $\epsilon$.  In sufficiently large dimension, every randomized adaptive algorithm has an admissible globally regular NC--SC instance and a conservative SFO requiring
\begin{equation}\label{eq:main-lower}
\Omega\!\left(
\kappa^2L\Delta\epsilon^{-2}
+L^3\chi^2\kappa^8\sigma_g^2\Delta\epsilon^{-6}
\right)
\end{equation}
queries to guarantee $\mathbb E\left\lVert \nabla F(\widehat x)\right\rVert\le\epsilon$.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:main-upper-informal}
Under \cref{ass:primitive} and a finite initialization certificate, there is a fresh-sample fully first-order algorithm with expected complexity
\begin{equation}\label{eq:main-upper}
\widetilde O\!\left[
\left(1+\frac{\Lambda\Delta}{\epsilon^2}\right)
\left\{
1+k+\frac{(1+k)^2}{\epsilon^2}
\left[\sigma_f^2+
\left(\frac{L_f^2}{\mu^2}+1+\frac{E_\rho^2}{\epsilon^2}\right)\sigma_g^2
\right]
\right\}
\right].
\end{equation}
<!-- end theorem-2 -->
<!-- corollary-1: body_md -->
\label{cor:curvature-controlled}
Suppose, in addition, that $\operatorname{Lip}(\nabla F)=O(L\kappa^2)$.  For fixed positive lower-level noise and sufficiently high accuracy,
\[
\mathsf T_{\mathsf{SFO}}(\epsilon)
=\widetilde\Theta\!\left(L^3\chi^2\kappa^8\sigma_g^2\Delta\epsilon^{-6}\right).
\]
This comparison holds in the normalized range $\rho\gtrsim\mu$, $\chi\lesssim\kappa$, and also for conservative SFOs.
<!-- end corollary-1 -->
<!-- theorem-3: body_md -->
\label{thm:small-gap-informal}
When
\[
\Delta=\Theta\!\left(\frac{\epsilon^2}{L\chi\kappa^2}\right),
\]
with a sufficiently large fixed multiplier, the original function and oracle class in the range $\rho\gtrsim\mu$, $\chi\lesssim\kappa$ has leading complexity
\[
\mathsf T_{\mathsf{SFO}}(\epsilon)
=\widetilde\Theta\!\left(L^2\chi^2\kappa^6\sigma_g^2\epsilon^{-4}\right)
\]
for fixed positive $\sigma_g$ and sufficiently high accuracy.
<!-- end theorem-3 -->
<!-- theorem-4: body_md -->
\label{thm:zero-branch-informal}
At $\rho=0$, with $\sigma_f=0$ and a nontrivial unknown lower offset ($R_0>0$), the branch
\[

\]
is parameter-matched under the dominance conditions stated in \cref{app:zero-variation}.  Amplified Chebyshev solves also realize the deterministic term with $\widetilde O(L\kappa^{5/2}\Delta\epsilon^{-2})$ queries.

<!-- lemma-1: body_md -->
\label{lb:lem:residual}
For $k=0,1,2,3$,
\begin{equation}\label{lb:eq:Rbounds}
   \left\lVert D^kR\right\rVert_\infty\le C_kL\beta^{2-k}.
\end{equation}
If $|z_i|\le\beta/4$ for all $i>r$, then
\begin{equation}\label{lb:eq:support-A-R}
   \operatorname{supp}\nabla A(z)\subseteq [r],
   \qquad
   \operatorname{supp}\nabla R(z)\subseteq [r+1].
\end{equation}
Also $R(0)=0$ and $\nabla A(0)=0$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
When a derivative of a tail mask is nonzero, its inner sum is at most one.  Therefore the first three derivatives of the mask are bounded by $C_k\beta^{-k}$ independently of $T$.  At a fixed point, let $j=1+\max\{i:|z_i|>\beta/2\}$, with maximum zero when the set is empty.  Terms before $j$ have a flat-zero tail mask, and terms after $j$ have a flat-zero predecessor gate.  Thus only one summand has a nonzero jet through order three, proving \eqref{lb:eq:Rbounds}.  If the tail coordinates after $r$ are within $\beta/4$, all their cutoffs are flat and the corresponding $H$ and $R$ derivatives cancel in $A=H-R$ after coordinate $r$, while chain locality leaves at most the next coordinate in $R$.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lb:lem:endgame}
For sufficiently large fixed $K$, $\mathcal H_U$ is coercive, attains its minimum, and
\begin{equation}\label{lb:eq:wrapped-gap}
  \mathcal H_U(0)-\inf\mathcal H_U\le C_0Ta.
\end{equation}
Moreover,
\begin{equation}\label{lb:eq:wrapped-endgame}
 |(U^\top b(u))_T|<\beta
 \quad\Longrightarrow\quad
 \left\lVert \nabla\mathcal H_U(u)\right\rVert\ge \eta L\beta/2.
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
The chain $H$ is bounded below and $P(u)\to\infty$, so the wrapped objective is coercive.  Since $P\ge0$, $P(0)=0$, and $H\ge\inf H$, \eqref{lb:eq:wrapped-gap} follows directly from \eqref{lb:eq:chainfacts}.  For the end-game, set $z=U^\top b(u)$ and let $G=C_1\eta L\beta\sqrt T$ bound $\left\lVert \nabla H(z)\right\rVert$.  If $\left\lVert u\right\rVert\le\mathcal R$, then $\alpha=(1+\left\lVert u\right\rVert^2/\mathcal R^2)^{-1/2}\ge1/\sqrt2$, and the coordinate given by \eqref{lb:eq:coordinate-endgame} has ambient gradient component
\[
\left\langle \nabla\mathcal H_U(u),Ue_j\right\rangle
=\alpha\partial_jH(z)+
\left(\nu-\frac{\alpha^2}{\mathcal R^2}\left\langle u,U\nabla H(z)\right\rangle\right)z_j.
\]
The second term has magnitude at most $(\nu+G/\mathcal R)\beta$, which is a small fraction of $\eta L\beta$ by choosing $K$ and $\eta_0$.  If $\left\lVert u\right\rVert>\mathcal R$, test the gradient in the radial direction $u/\left\lVert u\right\rVert$.  The radial term gives at least $\nu\mathcal R/\sqrt2$ while the chain contribution is at most $G$; increasing $K$ again gives the desired margin.
<!-- end proof-2 -->
<!-- lemma-3: body_md -->
\label{lb:lem:flatten}
Uniformly in $T$ and $\delta$,
\[
\left\lVert R_\delta\right\rVert_\infty\le CL\beta^2,
\quad \left\lVert \nabla R_\delta\right\rVert_\infty\le CL\beta,
\quad \left\lVert D^2R_\delta\right\rVert_\infty\le CL\beta/\delta,
\quad \left\lVert D^3R_\delta\right\rVert_\infty\le CL\beta/\delta^2.
\]
Moreover $\left\lVert R-R_\delta\right\rVert_\infty\le CL\beta\delta$, and if $|z_i|\le\delta$ for all $i>k$, then $\operatorname{supp}\nabla R_\delta(z)\subseteq[k]$.
For two resolutions $0<\delta'<\delta$, the jets through order three of $R_{\delta'}-R_\delta$ can be nonzero only when the unique active child coordinate satisfies $\delta'<|z_j|<2\delta$.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Where $c_\delta(t)\ne t$, both $t$ and $c_\delta(t)$ are at most $2\delta\le\beta/8$ in magnitude.  Thus the predecessor gates and tail masks are unchanged; only the outgoing factor $\Phi(\pm z_j/\beta)$ is modified.  There is still only one active summand jet.  The derivative bounds follow from $\left\lVert c_\delta^{(m)}\right\rVert\le C\delta^{1-m}$ and the chain rule.  If a coordinate is below the flattening scale, both its flattened value and derivative vanish.  For the annulus claim, outside $\delta'<|z_j|<2\delta$ either both flattened arguments are zero or both equal the identity, so all jets cancel.
<!-- end proof-3 -->
<!-- theorem-5: body_md -->
\label{lb:thm:mean-lower}
If $\beta\vartheta$ is smaller than a numerical constant, the pair \eqref{lb:eq:f-lower}--\eqref{lb:eq:g-lower} satisfies
\[
 \nabla^2_{yy}g_U\succeq(\mu/2)I,
 \qquad \operatorname{Lip}(\nabla f_U),\operatorname{Lip}(\nabla g_U)\le CL,
 \qquad \operatorname{Lip}(\nabla^2g_U)\le C\mu\vartheta,
\]
and
\[
 \sup\left\lVert \nabla f_U\right\rVert\le C(L+\sqrt{L\Delta})

<!-- lemma-4: body_md -->
\label{lb:lem:genie}
Let $U_1,\ldots,U_T$ be a Haar orthonormal frame in $\mathbb R^d$.  A protocol maintains a known subspace $V$, inserts at most $M$ transcript-measurable query directions, observes all projections $P_VU_i$, and completely reveals at most $T$ columns.  If

 b_*^2=\frac{8\log(4MT/\delta_0)}{d-M-T-2},
 \qquad TMb_*^2\le1/2,

\begin{equation}\label{lb:eq:projection-energy}
 \left\lVert P_VU_i\right\rVert^2\le\frac{Mb_*^2}{1-TMb_*^2}\le2Mb_*^2.
\end{equation}

\begin{equation}\label{lb:eq:dimension}
 d\ge C(M+T)\vartheta^2T\log(4MT/\delta_0).
\end{equation}

\[
\left\langle (I-P_V)U_i,(I-P_V)U_j\right\rangle=-\left\langle P_VU_i,P_VU_j\right\rangle,
\]
and inserting the normalized residual of $U_j$ increases $e_i$ to at most $e_i/(1-e_j)$. F_b(0)-\inf F_b\le\frac{\ell b^2}{2\mu^2}\le\Delta,
\qquad \left\lVert y_b^*(0)\right\rVert=|b|/\mu\le R_0.
\]
These estimates hold over the full Euclidean domain.

<!-- lemma-10: body_md -->
\label{zv:lem:experiment}
Even if all structural parameters and the form of \eqref{zv:eq:offset-f}--\eqref{zv:eq:offset-g} are revealed, except $b$, every adaptive SFO call contains exactly one nontrivial scalar observation about $b$:
\begin{equation}\label{zv:eq:location}
\widehat g_y(x,y,z)-\mu y+ax=b+\xi.
\end{equation}
All other returned components are functions of the query and known parameters. The offset observation has the same signal-to-noise ratio at every query magnitude.
<!-- end lemma-10 -->
<!-- proof-16: body_md -->
The other lower components are $g_x=-ay$ and $g_z=Mz$, while the full upper response is known from $y$. Thus the entire transcript can be simulated using one independent $N(b,\sigma^2)$ observation per call. Conversely, one obtains such an observation by querying at zero.

<!-- theorem-10: body_md -->
\label{zv:thm:offset-risk}
There is a numerical $c>0$ such that
\begin{equation}\label{zv:eq:offset-risk}
 cK_{\rm off}\min\{b_{\max},\sigma/\sqrt N\}
 \ \le\ \mathcal R_N
 \ \le\ K_{\rm off}\min\{b_{\max},\sigma/\sqrt N\}.
\end{equation}
The lower bound holds even when $a,\ell,C,\mu,M$ and the function family are known. It therefore also holds in the original less-informed problem class containing these instances.
<!-- end theorem-10 -->
<!-- proof-17: body_md -->
For the upper bound, either output zero, or average $N$ observations in \eqref{zv:eq:location}, project onto $[-b_{\max},b_{\max}]$ to obtain $\widehat b$, and output $\widehat b/a$. Because $|\tanh t|\le|t|$ and projection cannot increase error from a feasible $b$,
\[
\mathbb E_b|F_b'(\widehat b/a)|
\le K_{\rm off}\mathbb E_b|\widehat b-b|
\le K_{\rm off}\sigma/\sqrt N.
\]
The zero output has risk at most $K_{\rm off}b_{\max}$.

For the lower bound use hypotheses $b=\pm\delta$, where
\[
\delta=\min\{b_{\max},\sigma/(4\sqrt N)\}.
\]
By the conditional KL chain rule, or directly by Lemma~\ref{zv:lem:experiment}, the KL divergence of the complete transcripts is at most
\[
\frac{2N\delta^2}{\sigma^2}\le\frac18.
\]
Include the independent private seed in the transcript. Pinsker's inequality gives total variation at most $1/4$, hence every test has average error at least $3/8$. Classify the sign of $b$ from the sign of $\widehat x$, breaking a zero tie toward the positive sign. On a classification error, $|a\widehat x-b|\ge\delta$. Since $\ell\delta/(\mu C)\le1$ and $\tanh t\ge t/2$ for $0\le t\le1$,
\[
|F_b'(\widehat x)|\ge\frac{aC}{\mu}
\tanh\left(\frac{\ell\delta}{\mu C}\right)
\ge\frac12K_{\rm off}\delta.
\]
The Bayes average expected loss is therefore at least $3K_{\rm off}\delta/16$. Finally $\delta\ge\frac14\min\{b_{\max},\sigma/\sqrt N\}$. This proves \eqref{zv:eq:offset-risk}, for example with $c=3/64$.

<!-- lemma-11: body_md -->
\label{zv:lem:polynomial}
For $k>1$ let $q=(\sqrt k-1)/(\sqrt k+1)$. For every $m\ge1$ there is a polynomial $p_{m-1}$ of degree at most $m-1$ such that
\begin{equation}\label{zv:eq:polyerror}
\sup_{\lambda\in[\mu,M]}|p_{m-1}(\lambda)-1/\lambda|
\le\frac2\mu q^m,
\qquad
\sup_{\lambda\in[\mu,M]}|p_{m-1}(\lambda)|\le\frac3\mu.
\end{equation}
Its coefficients depend only on the public endpoints $\mu,M$.
<!-- end lemma-11 -->
<!-- proof-19: body_md -->
With $T_m$ the Chebyshev polynomial, define
\[
r_m(\lambda)=
\frac{T_m((M+\mu-2\lambda)/(M-\mu))}
{T_m((M+\mu)/(M-\mu))}.
\]
Since $r_m(0)=1$, the function $(1-r_m(\lambda))/\lambda$ is a polynomial of degree at most $m-1$. Take it as $p_{m-1}$. On $[\mu,M]$ the numerator has magnitude at most one. The identity
\[
T_m((M+\mu)/(M-\mu))=\tfrac12(q^{-m}+q^m)
\]
gives $|r_m|\le2q^m$, proving both inequalities.
<!-- end proof-19 -->
<!-- lemma-12: body_md -->
\label{zv:lem:inverse-apply}
For a known vector $v$ with $\left\lVert v\right\rVert\le V$ and target $r>0$, the original oracle can produce $w$ with
\[
\mathbb E[\left\lVert w-H^{-1}v\right\rVert^2\mid v,\text{past}]\le r^2
\]
using
\begin{equation}\label{zv:eq:inverse-cost}
O\!\left(1+\sqrt k\log(2+V/(\mu r))\right)
\end{equation}
original calls. Each amplification is finite and computable from public bounds, the target accuracy, and the input. The displayed query count specifies the spectral and accuracy dependence.
<!-- end lemma-12 -->
<!-- proof-20: body_md -->
Choose $m=O(1+\sqrt k\log(2+V/(\mu r)))$ so that $2q^mV/\mu\le r/4$. Write $p_{m-1}(M t)$ in monomials and evaluate it by Horner's rule using the scaled operator $K=H/M$, whose norm is at most one. Replace each $Kw$ action by \eqref{zv:eq:Haction} divided by $M$, with conditional mean-square action error at most $r^2/(16m)$. An amplification of order $1+\sigma_g\sqrt m/(Mr)$ achieves this tolerance.

The errors in Horner evaluation obey the recurrence $e_j=K e_{j+1}+\eta_j$. Hence the final error is a sum of powers of $K$ applied to conditional martingale differences. Cross inner-product expectations vanish, even though the query vectors are adaptive. Since $\left\lVert K\right\rVert\le1$, its mean square is at most $m\cdot r^2/(16m)=r^2/16$. Combining with the deterministic polynomial error proves the assertion with slack. In this recurrence, powers of $K$ propagate the injected action errors. Every intermediate vector and amplification is finite; the global absolute variance bound is valid there. This explicitly uses the stated real-arithmetic cost model. For $k=1$, $H=\mu I$ and inverse application is direct.

<!-- lemma-13: body_md -->
\label{zv:lem:endpoint-est}
For sufficiently large numerical $C$, conditionally on the calibration and the outer past,
\begin{equation}\label{zv:eq:Gmse}
\mathbb E\left\lVert G-\nabla\bar F(x)\right\rVert^2\le\epsilon^2/256.
\end{equation}
The conditional expected original SFO cost is
\begin{equation}\label{zv:eq:fast-perstep}
\widetilde O\!\left(1+\sqrt k+\frac{(1+k)^2\sigma_f^2}{\epsilon^2}\right).
\end{equation}
Finite amplification controls the lower-gradient variance inside each estimated linear action.
<!-- end lemma-13 -->
<!-- proof-21: body_md -->
Using projection and Lipschitzness of $f_y$,
\[
\mathbb E\left\lVert \widehat a-f_y(x,\bar y)\right\rVert^2
\le 2\sigma_f^2/B_f+2L_f^2 r_y^2.
\]
The same upper batch estimates $f_x$; correlation between its two components is allowed. Expanding the hypergradient difference and using a sum-of-squares bound yields
\[
\mathbb E\left\lVert G-\nabla\bar F\right\rVert^2
\le C_1\left[(1+k)^2(\sigma_f^2/B_f+L_f^2r_y^2)
+L_g^2r_v^2+\epsilon^2/C^2\right].
\]
This proves \eqref{zv:eq:Gmse} with numerical slack. Each inverse application costs \eqref{zv:eq:inverse-cost}; every matrix action uses two original calls. Add the upper batch and the constant number of right-hand-side and final-product calls. Known bounds on the input norms enter the logarithmic factors.

<!-- theorem-11: body_md -->
\label{zv:thm:endpoint-upper}
Under \cref{ass:primitive} with $\rho=0$, an algorithm in the original SFO model achieves $\mathbb E\left\lVert \nabla F(\widehat x)\right\rVert\le\epsilon$ at expected cost
\begin{equation}\label{zv:eq:endpoint-upper}
\boxed{\ \widetilde O\!\left[
q_b+
\left(1+\frac{\Lambda_0\Delta}{\epsilon^2}\right)
\left(1+\sqrt k+\frac{(1+k)^2\sigma_f^2}{\epsilon^2}\right)
\right].\ }
\end{equation}
Here $q_b$ is \eqref{zv:eq:qcal}, or zero when $R_0=0$. The matrices $H,B$ remain unknown. Both general unbiased-vector and conservative sample-gradient SFOs are allowed.

<!-- proof-22: body_md -->
After calibration use Lemma~\ref{zv:lem:endpoint-est} and $x_{t+1}=x_t-G_t/\Lambda_0$. Completing a square in the smoothness inequality gives
\[
\bar F(x_{t+1})\le\bar F(x_t)
-\frac{\left\lVert \nabla\bar F(x_t)\right\rVert^2}{2\Lambda_0}
+\frac{\left\lVert G_t-\nabla\bar F(x_t)\right\rVert^2}{2\Lambda_0}.
\]
Take a fixed $n=\max\{1,\lceil C\Lambda_0\Delta/\epsilon^2\rceil\}$ and output a uniform iterate. Average conditionally and then over the calibration samples; \eqref{zv:eq:auxgap} and \eqref{zv:eq:Gmse} make $\mathbb E\left\lVert \nabla\bar F(\widehat x)\right\rVert^2$ a sufficiently small multiple of $\epsilon^2$. The uniform bound \eqref{zv:eq:gradtransfer} and \eqref{zv:eq:qcal} transfer this guarantee to $F$, including the dependence of the output on the calibrated offset.

For the expected query count, the degree of the first polynomial depends logarithmically on $1+L_g(\left\lVert x\right\rVert+R_0)/(\mu r_y)$. The descent inequality bounds the expected sum of squared steps; Cauchy--Schwarz bounds expected squared displacement over the fixed horizon, and Jensen bounds the expected logarithm. This yields only the stated initialization and accuracy logarithms. Multiplying \eqref{zv:eq:fast-perstep} by $n$ and adding $q_b$ proves \eqref{zv:eq:endpoint-upper}.

<!-- proposition-1: body_md -->
\label{sh:prop:simultaneous}
For $0<\mu\lesssim L$, $\rho>0$, and sufficiently small $\epsilon>0$, there is a globally regular finite-gap NC--SC problem with $\left\lVert f_y\right\rVert=L$, first-order smoothness $O(L)$, and lower Hessian Lipschitzness $O(\rho)$ such that
\begin{equation}\label{zv:eq:coattain}
F'(0)=2\epsilon,\qquad
F''(0)=-\frac{\rho L^3}{\mu^3},\qquad
\lim_{t\downarrow0}\frac{|h_t-F'(0)|}{t}
=\frac{\rho L^3}{2\mu^3}.
\end{equation}
Here $h_t$ denotes the ideal frozen-tilt estimator evaluated at zero.
<!-- end proposition-1 -->
<!-- proof-23: body_md -->
Let $\zeta:\mathbb R^2\to[0,1]$ be a fixed smooth cutoff, equal to one on the unit ball and zero outside the radius-two ball. Put $r=c\mu/\rho$ with small numerical $c$. Let $\psi$ be an odd compactly supported smooth function equal to $x$ near zero. Set
\[
f(x,u,s,z)=Ls+2\epsilon\psi(x),
\]
\begin{equation}\label{zv:eq:sharp-mean}
g(x,u,s,z)=\frac\mu2(u^2+s^2)+\frac L2z^2-Lxu
+\frac\rho2(u^2s+us^2)\zeta((u,s)/r).
\end{equation}
The cubic perturbation has Hessian norm at most $C\rho r\le\mu/2$ and third derivative at most $C\rho$. The mixed $x,u$ block is constant. Thus all global mean assumptions hold, with fixed slack.

The lower solution at zero is $(0,0,0)$. In the cutoff core, implicit differentiation of the lower equations gives $u'(0)=L/\mu$, $s'(0)=0$, and $s''(0)=-\rho L^2/\mu^3$. The outer tilt has derivative $2\epsilon$ and second derivative zero at zero. This proves the first two identities.

For the frozen direction $(0,L,0)$, the tilted lower equations at $x=0$ in the core are
\[
\mu u+\rho us+\tfrac\rho2s^2=0,
\qquad
\mu s+\tfrac\rho2u^2+\rho us+tL=0.
\]
The implicit function theorem yields
\[
s_t=-Lt/\mu+O(t^3),\qquad
u_t=-\rho L^2t^2/(2\mu^3)+O(t^3),
\]
Since $g_x=-Lu$, the estimator obeys
\[
h_t-F'(0)=-Lu_t/t
=\frac{\rho L^3}{2\mu^3}t+O(t^2).
\]
This proves the bias limit. Outside the lower cutoff support, a lower minimizer has $s^*=0$; inside it, $|s^*|\le2r$. The compact outer tilt is bounded. Therefore $F$ is bounded and its initial gap is at most $2Lr+2\epsilon\left\lVert \psi\right\rVert_\infty$. Fixed normalized gap budgets can contain this example after numerical rescaling when $\rho\gtrsim\mu$.
