<!-- theorem-1: body_md -->
\label{theorem:sderivative}
		The $n$-th derivative of the $S$ function when $n \geq 2$ is
		\begin{equation}\label{eq:snthder}
			S^{(n)}(t) = (-1)^{n-1} \frac{(n-2)!}{2\pi}t^{1-n} + O(t^{-n-1}), \ t \to +\infty.
		\end{equation}
  except for its discontinuity points.
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{corollary:thetanderivative}
	Suppose $n \geq 2$. Then, the $n$-th derivative of the Riemann-Siegel theta function is
	\begin{equation}\label{eq:thetanthder}
	    \vartheta^{(n)}(t) = (-1)^{n} \frac{(n-2)!}{2}t^{1-n} + O(t^{-n-1}), \ t \to +\infty.
	\end{equation}
<!-- end corollary-1 -->
<!-- theorem-2: body_md -->
\label{theorem:gramnthderivative}
        The $n$-th derivative of the Gram function is asymptotically
        $$
            t_u^{(n)} = \frac{(-1)^{n + 1} 2 \pi (n - 2)!}{u^{n - 1} \log^2 u} \left( 1 + (2 + o(1)) \frac{\log \log u}{\log u}\right), \ u \to +\infty,
        $$
        for $n = 2, \ldots$.
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{thm:gramunifdist}
        The sequence $\{t_k^n\}_{k = 1}^\infty$ is uniformly distributed modulo one for $n = 1, 2, 3, \ldots$.
<!-- end theorem-3 -->
<!-- corollary-2: body_md -->
\label{prop:gramcontunifdist}
        The function $t_u^n$, $n = 1, 2, 3, \ldots$, is continuously uniformly distributed modulo one.
<!-- end corollary-2 -->
<!-- proof-1: body_md -->
Our goal is to compute the $n$-th derivative of $S(t)$. Recall the formula \eqref{eq:Sfull}. We split the computation into two parts.
        
        First, let us compute
		\begin{displaymath}
			\left(-\frac{t}{2\pi} \log \frac{t}{2\pi} + \frac{t}{2\pi} - \frac{7}{8} \right)^{(n)} = \frac{d^{n - 2}}{dt^{n - 2}}\left(- \frac{1}{2\pi t} \right) = (-1)^{n - 1} \frac{(n-2)!}{2\pi t^{n-1}}.
		\end{displaymath}
		It remains to estimate the error term arising from differentiating the function $\delta$. The terms
		\begin{equation}
                \label{eq:firsttwoterms}
			\frac{t}{4}\log \left(1 + \frac{1}{4t^2} \right) + \frac{1}{4} \arctan \frac{1}{2t}
		\end{equation}
		can be expanded into
            $$
            \frac{3}{16t} + \sum_{k = 3}^\infty \frac{a_k}{t^k}
            $$
        with real coefficients $a_n$. This Laurent series is differentiable since it converges for $|t| > 1/2$ and is analytic. Differentiating the above expression $n$ times term-by-term yields
         \begin{equation}\label{eq:deltafirstterms}
         (-1)^n \frac{3n!}{16 t^{n + 1}} + O(t^{-n - 3}), \ t \to +\infty.
         \end{equation}
For our purposes, the accuracy $O(t^{-n - 1})$ is sufficient. 
				
		Continuing our computation of the $n$-th derivative of $\delta(t)$, let us denote
		\begin{align*}
			j(t) & := \int_0^\infty \frac{\rho(u) \, du}{(u + 1/4)^2 + (t/2)^2},\\
            \sigma(u) & := \int_0^u \rho(z) \, dz.
		\end{align*}
		We compute the $n$-th derivative of $j$. We follow with integration by parts
		\begin{displaymath}
			j(t) = \int_0^\infty \frac{d\sigma(u)}{(u + 1/4)^2 + (t/2)^2} = 2 \int_0^\infty \frac{\sigma(u)(u+1/4) \, du}{((u + 1/4)^2 + (t/2)^2)^2}.
		\end{displaymath}
		Note that $0 \leq \sigma(u) \leq 1/8$.
		
		We need to differentiate $j(t)$ with respect to $t$ $n$ times, which amounts to differentiating under the integral sign. We use the Leibniz rule 
        $$
            \frac{d}{d t} \int_a^\infty f(u, t) \, du = \int_a^\infty \frac{\partial}{\partial t} f(u, t) \, du.
        $$
        The following conditions must be satisfied: $f(u, t)$ and $\partial f/\partial t$ be continuous for $u \geq a$, and there exists an integrable $g(u)$ such that $|\partial f(u, t)/\partial t)| \leq g(u)$ (see Rudin \cite[pp. 180--182]{Rudin1976Analysis}). For integrating the $k$-th derivative, we need an integrable majorant uniform in $t$ on $[T, 
    \infty)$. This condition is satisfied by
        $$
            \left| \frac{\partial^k}{\partial t^k} \left( \frac{1}{((u + 1/4)^2 + (t/2)^2)^2}\right) \right| \ll_k \frac{1}{((u + 1/4)^2 + 1)^2}, \ t \geq 1
        $$
        and the bound $0 \leq \sigma \leq 1/8$.
        
		Applying the Leibniz rule yields
        \begin{align*}
			 \frac{d^n}{dt^n} & \int_0^\infty \frac{(u + 1/4) \sigma(u)\, du}{((u+1/4)^2 + (t/2)^2)^2} 
             = \int_0^\infty \frac{\partial^n}{\partial t^n} \frac{(u + 1/4) \sigma(u) \, du}{((u + 1/4)^2 + (t/2)^2)^2} \\
            & \ll \int_0^\infty\sum_{r=\lceil n/2 \rceil}^n \frac{t^{2r-n} (4u + 1) \sigma(u)\, du}{((4u + 1)^2 + 4t^2)^{r + 2}}, \ t \to +\infty.
		\end{align*}

        In evaluating the above integral, we use the bound $\sigma(u) \le 1/8$ and the substitution $v = (4u+1)^2 + 4t^2$, so that $dv = 8(4u+1) \,  du$. We get

\begin{align*}
& \int_0^\infty \frac{(4u+1)\sigma(u)du}{((4u+1)^2+4t^2)^{r+2}}
\le \frac{1}{8}\int_0^\infty \frac{(4u+1)du}{((4u+1)^2+4t^2)^{r+2}} \\ &
= \frac{1}{64}\int_{1+4t^2}^{\infty} \frac{dv}{v^{r+2}}
= \frac{1}{64(r+1)(1+4t^2)^{r+1}}.
\end{align*}

Multiplying by $t^{2r-n}$ and using $1+4t^2 \ge 4t^2$:

\begin{align*}
& \frac{t^{2r-n}}{64(r+1)(1+4t^2)^{r+1}} \le \frac{t^{2r-n}}{64(r+1)4^{r+1}t^{2r+2}} \\ & = \frac{t^{-n-2}}{64(r+1)4^{r+1}} .
\end{align*}

Summing over $\lceil n/2\rceil \le r \le n$  yields:

$$
j^{(n)}(t) \ll_n t^{-n-2}, \ t \to +\infty.
$$
         
         We need to find the $n$-th derivative of the last term of the function $\delta(t)$ (see \eqref{eq:deltadefinition})
         $$
          - \frac{t}{2} \int_0^\infty \frac{\rho(u) \, du}{(u+1/4)^2 + (t/2)^2} = -\frac{t}{2} j(t).
         $$
          We apply the Leibniz rule to differentiate products. If $u$ and $v$ are $n$ times differentiable functions of $x$, then (see Gradshteyn and Ryzhik \cite[Formula 0.42]{gradshteyn2014table})
		 \begin{displaymath}
		 	\frac{d^n(uv)}{dx^n} = u \frac{d^nv}{dx^n} + \binom{n}{1} \frac{du}{dx}\frac{dv^{n-1}}{dx^{n-1}} + \binom{n}{2}\frac{d^2 u}{dx^2}\frac{d^{n-2}v}{dx^{n-2}} + \ldots + \frac{d^n u}{dx^n}v.
		 \end{displaymath}
	   We get
		 \begin{displaymath}
		 	-\frac{1}{2} \left(t j^{(n)}(t) + \binom{n}{1} j^{(n-1)}(t) \right) = O(t^{-n-1}).
		 \end{displaymath}
		 Collecting this and \eqref{eq:deltafirstterms} completes the proof of Theorem \ref{theorem:sderivative}.

<!-- proof-2: body_md -->
We prove this by induction. Let us make the inductive hypothesis
        \begin{equation} \label{eq:inductivestep} 
            t_u^{(k)} = \frac{(-1)^{k + 1} 2 \pi (k - 2)!}{u^{k - 1} \log^2 u} \left( 1 + (2 + o(1)) \frac{\log \log u}{\log u}\right), \ u \to +\infty.
        \end{equation}
        First, we prove that the theorem statement holds for $k = 2$. Differentiating the equation \eqref{eq:gramimplicit} gives
     $$
     \vartheta''(t_u) t^{\prime 2}_u + \vartheta'(t_u) t''_u = 0.
     $$
     We calculate the term $\vartheta''(t_u) t^{\prime 2}_u$ using Corollary \ref{corollary:thetanderivative}, equations \eqref{eq:gramfirstder} and \eqref{eq:gramestimate}
     \begin{align*}
         \vartheta''(t_u) t^{\prime 2}_u = & \left(\frac{1}{2 t_u} + O(t_u^{-3}) \right) \left(\frac{2\pi}{\log u}\left(1 + (1 + o(1))\frac{\log \log u}{\log u} \right) \right)^2 \\ = &\frac{2 \pi}{2 u \log u}\left( 1 + (1 + o(1)) \frac{\log \log u}{\log u}\right), \ u \to +\infty.
     \end{align*}

     Differentiating \eqref{eq:gramimplicit} and using the expression \eqref{eq:gramfirstder} yields
     $$
     \frac{1}{\vartheta'(t_u)} = \frac{2}{\log u} \left(1 + (1 + o(1))\frac{\log \log u}{\log u} \right), \ u \to +\infty.
     $$
     Combining the above two results gives
     $$
     t''_u = - \frac{2\pi}{u \log^2 u}\left( 1 + (2 + o(1)) \frac{\log \log u}{\log u}\right), \ u \to +\infty.
     $$

     Suppose our hypothesis \eqref{eq:inductivestep} holds for $2 \leq k \leq n - 1$. We show it holds for $k = n$. By Faà di Bruno's formula (see Gradshteyn and Ryzhik \cite[formula (0.430)]{gradshteyn2014table}), the $n$-th derivative of a composite function $h(x) = g(f(x))$ is
\begin{align}\label{bruno}
\frac{d^n}{dx^n} g(f(x)) = & \sum \frac{n!}{m_1! 1!^{m_1} m_2! 2!^{m_2} \cdots m_n! n!^{m_n}} g^{(m_1 + m_2 + \cdots +m_n)}(f(x)) \\ & \times \prod_{j = 1}^n (f^{(j)}(x))^{m_j}. \nonumber
\end{align}
Here, the summation is taken over all partitions
\begin{equation} \label{eq:partition}
 m_1 + 2 m_2 + \cdots + n m_n = n.
\end{equation}
We will need the shorthand notation $\sum m = m_1 + m_2 + \cdots + m_n$ for our purposes.
        
We plug $g(s) = \vartheta(s)$, $f(u) = t_u$ into Faà di Bruno's formula. Then, the relation \eqref{eq:gramimplicit} yields that the $d^n \vartheta(t_u)/d u^n$ equals $0$ and the formula \eqref{bruno} becomes
\begin{align}\label{brunoforgram}
   0 =  \vartheta'(t_u) t_u^{(n)} + \cdots + \vartheta^{(n)}(t_u)(t_u')^n.
\end{align}
The first member in the above equation corresponds to the case $m_n = 1$ and $m_i = 0$ for $i = 1, \ldots, n - 1$. The last member - to the case $m_1 = n$ and $m_i = 0$ for $i = 2, \ldots, n$.
        
In the summation of Faà di Bruno's formula, the first term in the above equation \eqref{brunoforgram} is the only one containing $t_u^{(n)}$, and the last one is asymptotically the largest of the remaining ones as $u \to \infty$. Let us prove this fact. We will show that $\vartheta^{(n)}(t_u)(t'_u)^n$ is asymptotically larger than the terms obtained using Faà di Bruno's formula, except possibly the first one. Hence, we can ignore the constants. Any term is uniquely determined by the values $m_1$, $m_2$, \ldots, $m_n$, which come from the partition \eqref{eq:partition}. For this calculation, we use the asymptotics for the derivatives of $\vartheta(t)$ (Corollary \ref{corollary:thetanderivative}), the asymptotics for $t'_u$ and $t_u$, and our inductive step \eqref{eq:inductivestep}. We skip the term $\vartheta'(t_u) t^{(n)}_u$. In all the remaining terms, we have $m_n = 0$. The arbitrary term in Faà di Bruno's formula, except for the partition $m_n = 1$, $m_i = 0$, $i = 1, \ldots, n - 1$, is
\begin{align}
\nonumber &\left| \vartheta^{(m_1 + m_2 + \cdots +m_{n - 1})}(t_u) \prod_{j = 1}^{n - 1} (t^{(j)}_u)^{m_j} \right| = \Theta \left( \frac{1}{t_u^{\sum m - 1}} \cdot \frac{1}{\log^{m_1} u}\prod_{j = 2}^{n - 1} \left( \frac{1}{u^{j - 1} \log^2 u} \right)^{m_j} \right)
        \\ \nonumber
            & =
\Theta \left(\frac{\log^{\sum m -1}u}{u^{\sum m - 1} \log^{m_1} u} \cdot \frac{1}{u^{\sum_{j = 2}^{n -1} j m_j - \sum m + m_1} \log^{2 \sum m - 2m_1}u}\right) \\ \label{eq:memberofsumfaadibruno}
            & = \Theta \left( \frac{1}{u^{n - 1} \log^{\sum m - m_1 + 1} u} \right).
\end{align}
We say that a function $f(x)$ is $\Theta(g(x))$ if there exist positive constants $c_1$, $c_2$ and a real number $x_0$ such that $c_1 g(x) \leq f(x) \leq c_2 g(x)$, $x \geq x_0$. In the last step above, we used the equation \eqref{eq:partition}. The last expression shows that among all the terms in Faà di Bruno's formula, except $\vartheta'(t_u)t_u^{(n)}$, the asymptotically largest one is $\vartheta^{(n)}(t_u)(t'_u)^n$, corresponding to the partition $m_1 = n$ and $m_i = 0$, $i = 2, \ldots n$. This, together with \eqref{brunoforgram}, leads to
\begin{align*}
\vartheta'(t_u) t_u^{(n)}= -\vartheta^{(n)}(t_u)(t_u')^n +o\left( \vartheta^{(n)}(t_u)(t_u')^n \right)\qquad (u\to\infty).
 \end{align*}
\begin{align*}
\vartheta'(t_u) t_u^{(n)}= -\vartheta^{(n)}(t_u)(t_u')^n +O\left( \frac{1}{u^{n - 1} \log^{2} u} \right)\qquad (u\to\infty),
\end{align*}
It follows
$$
t_u^{(n)} = \frac{-\vartheta^{(n)}(t_u)(t'_u)^n}{\vartheta'(t_u)} + O \left( \frac{1}{\vartheta'(t_u) u^{n-1} \log^2 u}\right).
$$
The dominating term of $\vartheta'(t_u)$ is asymptotically $\log u$, thus the error term in the above expression becomes
$$
    O\left( \frac{1}{u^{n-1} \log^3 u}\right).
$$
Thus Corollary \ref{corollary:thetanderivative} together with the asymptotics for $t'_u$ and $t_u$ leads to
$$
t_u^{(n)} = \frac{(-1)^{n + 1} 2 \pi (n - 2)!}{u^{n - 1} \log^2 u} \left( 1 + (2 + o(1)) \frac{\log \log u}{\log u}\right).
$$
        This completes the induction and yields the theorem.

We have theorem \cite[Chapter 1, Theorem 3.5]{Kuipers2012UniformSequences} stating: if there exists $l \in \mathbb{N}$, the function $f(x)$ is defined for $x \geq 1$, the derivative $f^{(l)}(x)$ tends monotonically to $0$ as $x \to \infty$, and if $x|f^{(l)}(x)| \to \infty$, $x \to \infty$, then the sequence $\{ f(n) \}_{n \in \mathbb{N}}$ is uniformly distributed modulo one.

    Consider the $n$-th power of the Gram function $t^n_u$. By Faà di Bruno's formula, we have
    \begin{equation} \label{eq:tnfaadibruno}
    \frac{d^n t_u^n}{du^n} = \sum \frac{n!}{m_1!1!^{m_1}m_2!2!^{m_2} \cdots m_n!n!^{m_n}}\frac{d^{m_1 + m_2 + \cdots m_n}t_u^n}{d t_u^{m_1 + m_2 + \cdots m_n}} \prod_{j = 1}^n (t_u^{(j)})^{m_j},
    \end{equation}
    where we sum over all partitions
    \begin{equation}\label{eq:npartition}
        m_1 + 2 m_2 + \cdots + n m_n = n.
    \end{equation}
    Let us denote $\sum m := m_1 + m_2 + \cdots + m_n$. An arbitrary term on the right-hand side of the equation \eqref{eq:tnfaadibruno} depends on the value of the vector $(m_1, m_2, \ldots, m_n)$. Ignoring the constant, recalling the asymptotic of $t_u$ given in the formula \eqref{eq:gramestimate}, and the asymptotic of its derivative in Theorem \ref{theorem:gramnthderivative} yields the following asymptotic of an arbitrary term in the formula \eqref{eq:tnfaadibruno}
    \begin{align*}
    & \frac{u^{n - \sum m}}{\log^{n - \sum m} u} \cdot \frac{1}{\log^{m_1} u} \prod_{j = 2}^n \frac{1}{(u^{j-1} \log^2 u)^{m_j}} = \frac{u^{n - \sum m}}{\log^{n - \sum m} u} \cdot \frac{1}{u^{n - \sum m} \log^{-m_1 + 2 \sum m} u} \\ & = \frac{1}{\log^{n -m_1 + \sum m} u},
    \end{align*}
    where we used the formula \eqref{eq:npartition}. This value is maximized when we minimize $-m_1 + \sum m$, which happens when $m_1 = n $ and $m_2 = \cdots = m_n= 0$. Here $\sum m \leq n$ thus no term degenerates.

    To prove monotonicity, we calculate the $n + 1$-st derivative of the $n$-th power of the Gram function. Again, we use Faà di Bruno's formula. Summing over partitions, we have
    $$
    m_1' + 2m_2' + \cdots  + nm_n' + (n + 1)m_{n + 1}' = n + 1,
    $$
    where the integers $m_1'$, $m_2'$, \ldots, $m_{n + 1}'$ appear in our formula now. Let us denote $\sum m' := m_1' + m_2' + \cdots + m_{n + 1}'$. It follows
    \begin{align*}
        & \frac{u^{n - \sum m'}}{\log^{n - \sum m'} u} \cdot \frac{1}{\log^{m_1'} u} \prod_{j = 2}^{n + 1} \frac{1}{(u^{j-1} \log^2 u)^{m_j'}} = \frac{u^{n - \sum m'}}{\log^{n - \sum m'} u} \\ & \times \frac{1}{u^{n + 1- \sum m'} \log^{-m_1' + 2 \sum m'} u} \\ & = \frac{1}{u\log^{n -m_1' + \sum m'} u}
    \end{align*}
    Again, this value is maximized by setting $\sum m' - m'_1 = 1$. 
    
    All in all, we have
    $$
    \frac{d^n t_u^n}{d u^n} \sim  \frac{n!(2\pi)^n}{\log^n u}
    $$
    and
    $$
        \frac{d^{n + 1} t_u^n}{du^{n + 1}} \sim  -\frac{n n!(2\pi)^n}{u \log^{n + 1} u}.
    $$
     Based on the fact that the main terms of the $n + 1$-st derivative's asymptotic is non-zero of a fixed sign, there exists some value $u_0$ for which $d^{n + 1} t_u^n/du^{n + 1}$ does not change its sign provided $u > u_0$. Hence, the $n$-th derivative of $t_u^n$ will monotonically approach $0$ for $u > u_0$. In addition, $u |d^n t_u^n/du^n|$ will monotonically approach $+\infty$. Therefore, by the criterion stated at the beginning of the proof, the sequence $\{t^n_k\}_{k = 1}^\infty$ is uniformly distributed modulo one.

By Kuipers and Niederreiter \cite[Chapter 1, Theorem 9.6]{Kuipers2012UniformSequences}, the function $f(t)$, $t \geq 0$, is continuously uniformly distributed modulo one if 1) the sequence $f(k+t)$, $k = 0, 1, \ldots$, is uniformly distributed modulo one for almost all $t \in [0, 1)$ and 2) the sequence $f(kt)$, $k = 1, 2, \ldots$, is uniformly distributed modulo one for almost all $t \in [0, 1)$. If we take $f$ to be the $n$-th power of the Gram function, both of these facts follow from the fact that there exists real number $u_0$ such that $d^n t^n_u/du^n$ monotonically approaches zero for $u > u_0$, $u|d^n t_u^n/du^n|$ monotonically approaches $+\infty$ for $u > u_0$, and the criterion for the uniform distribution modulo $1$ of sequences, formulated at the beginning of the proof of Theorem \ref{thm:gramunifdist}.
