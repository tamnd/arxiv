<!-- lemma-1: body_md -->
\label{lem:relation-K-A}
The Euler--Mascheroni and Glaisher--Kinkelin constants satisfy
\[
\log A = \frac{1}{12} - \zeta'(-1) = \frac{\operatorname{{\it \gamma}} + \log 2\pi}{12} - \frac{\zeta'(2)}{2\pi^2}.
\]
The Kinkelin--Bendersky function $K$ and the Glaisher--Kinkelin constant $A$ are related by the following equality: for any integer $n\ge 2$, we have
\[
\prod_{j=1}^{n-1} K\left(\frac{j}{n}\right) = 
\frac{A^{\frac{n^2-1}{n}}}{n^{\frac{1}{12n}}e^{\frac{n^2-1}{12n}}}.
\]

Since $\psi\,'$ is the trigamma function defined above,

<!-- lemma-2: body_md -->
\label{lem:gammapro}
Let $d$ be a positive integer and $a_j,b_j$ complex numbers for $j\in\{1,\ldots,d\}$.
If $\sum_{j=1}^d a_j = \sum_{j=1}^d b_j$ and no $b_j$'s belong to $\{-1, -2, -3, \ldots\}$, then
\[
\prod_{n \geq 1}\frac{(n+a_1) \cdots (n+a_d)}{(n+b_1) \cdots (n+b_d)}
= \frac{\Gamma(1+b_1) \cdots \Gamma(1+b_d)}{\Gamma(1+a_1) \cdots \Gamma(1+a_d)}.
\]
<!-- end lemma-2 -->
<!-- remark-2: body_md -->
One can prove the following generalization of
Lemma~\ref{lem:gammapro}. Let $r$ be a positive integer. 
Let $a_1, \ldots, a_r, \theta_1, \ldots \theta_r$ be $2r$ 
complex numbers such that: 
\begin{itemize} 
\item[]{$*$} no $a_j$ belongs to 
$\{0, -1, -2, -3, \ldots\}$, 
\item[ ]{$*$} $\sum_{1 \leq j \leq r} \theta_j = 0$,
\item[ ]{$*$} $\sum_{1 \leq j \leq r} a_j \theta_j = 0$. 
\end{itemize}
Then, building on classically known techniques for the expression of 
products of rational functions with the $\Gamma$-function (see, e.g., 
Chamberland and Straub's applications of such techniques 
\cite{ChamberlandStraub2013}), it can be shown that 
                                                                      
\begin{align*}
\prod_{n \geq 0} 
\prod_{1 \leq j \leq r}(n + a_j)^{\theta_j}
= \left(\prod_{1 \leq j \leq r} \Gamma(a_j)^{\theta_j}\right)^{-1}.
\end{align*}

<!-- theorem-1: body_md -->
\label{B-alpha-beta}
For the function $B_{\alpha, \beta}$ defined in Equation~\eqref{eq:B-alpha-beta}, we have
\begin{multline}\label{variant1}
\log B_{\alpha, \beta}(x) = \displaystyle
(x - \beta) \left(\log \Gamma\left(\frac{x}{\alpha} + 1 \right) 
+ \operatorname{{\it \gamma}}\frac{x}{\alpha}\right) \\
- \ \alpha \int_0^{x/\alpha} \log \Gamma(t+1) \ dt
- \frac{\operatorname{{\it \gamma}}}{2\alpha} x^2 \ \ \ \ \ \ \ \ \ \ 
\end{multline}
and
\begin{equation}\label{variant2}
B_{\alpha, \beta}(x) = 
\frac{G_{B}^{\alpha}\left(\frac{x}{\alpha}+1\right)}
{\Gamma^{\beta} \left(\frac{x}{\alpha}+1\right)}
e^{\frac{1+\operatorname{{\it \gamma}}}{2\alpha} x^2 - \left(\frac{\beta \operatorname{{\it \gamma}}}{\alpha}
+ \frac{\log2\pi -1}{2}\right) x},
\end{equation}
for positive reals $\alpha, \beta, x$.
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
Recalling the definition of $B_{\alpha, \beta}$ in
Equation~\eqref{eq:B-alpha-beta}, we have
\begin{equation}\label{logBalphabeta}
\log B_{\alpha, \beta}(x) =
\sum_{n \geq 1} \left( - x - \frac{\beta x}{\alpha n}
+ \frac{x^2}{2 \alpha n} 
+ (\alpha n + \beta) \log \left(\frac{x}{\alpha n} + 1 \right)\right),
\end{equation}
\bigskip
 Differentiating both sides of Equation~\eqref{logBalphabeta}, 
 and then interchanging the application of differentiation and infinite summation on the resulting right-hand side via the dominated convergence theorem,
 we find that
\begin{equation}\label{expresswdigamma} 
\frac{B_{\alpha, \beta}'(x)}{B_{\alpha, \beta}(x)} 
=
 \frac{x - \beta}{\alpha} 
 \sum_{n \geq 1} \frac{x/\alpha}{n (n + x/\alpha)}.
\end{equation}
 Using Formulas~\eqref{eq:formula for psi} 
 and \eqref{expresswdigamma} together yields 
\begin{equation}\label{BprimeoverB}
\frac{B_{\alpha, \beta}'(x)}{B_{\alpha, \beta}(x)} =
\left(\frac{x-\beta}{\alpha}\right)
\left(\psi\left(\frac{x}{\alpha}+1\right) + \operatorname{{\it \gamma}}\right).
\end{equation}
 Using the definition for the digamma function given in Equation~\eqref{eq:digamma-trigamma} together with 
 the evaluation $B_{\alpha, \beta}(0)=1$, we obtain
 Equation~\eqref{variant1} by integrating 
Equation~\eqref{BprimeoverB}. 

 To obtain the second relation of the theorem, we make use of the relation
\begin{equation}\label{alexeiewsky-barnes}
\int_0^z \log \Gamma(1+t) \ dt = \frac{1}{2}(\log 2\pi - 1)z
- \frac{z^2}{2} + z \log \Gamma(1+z) - \log \operatorname{{\it G_B}}(1+z) 
\end{equation}
 attributed to 
 Alexeiewsky
 (see~\cite[p.\ 281]{Barnes1900}
 and Remark~\ref{rem:attribute} below). 
 In this direction, Equations~\eqref{alexeiewsky-barnes} and~\eqref{variant1} together yield
\begin{multline*}
\log B_{\alpha, \beta}(x) = 
- \beta \log\Gamma\left(\frac{x}{\alpha}+1\right)
+ \frac{1+\operatorname{{\it \gamma}}}{2\alpha} x^2 \\
- \left(\frac{\beta \operatorname{{\it \gamma}}}{\alpha} 
+ \frac{\log 2\pi - 1}{2}\right) x

\label{eq: special value for 1, a half and 1}
The case $\alpha = 1$, $\beta = 1/2$, $x = 1$ of 
Theorem~\ref{B-alpha-beta} above might deserve a special mention.
It reads

$$
This equality can also be proven by a more direct argument. 
Namely, write 
\begin{align*}
\prod_{n \geq 1} \frac{1}{e}
\left(\frac{1}{n}+1\right)^{n + \frac{1}{2}}
&= \displaystyle \lim_{N \to + \infty}
\prod_{1 \leq n \leq N} \frac{1}{e}
\left(\frac{1}{n}+1\right)^{n + \frac{1}{2}} \\
&= 
\displaystyle \lim_{N \to + \infty} \prod_{1 \leq n \leq N}
\frac{1}{e} \frac{(n+1)^{n+1}}{n^n} \frac{1}{\sqrt{n(n+1)}} \\
&= \displaystyle \lim_{N \to + \infty}
\frac{(N+1)^{N + \frac{1}{2}}}{e^N\,N!}, 
\end{align*}
where the last equality is obtained by telescopic simplifications.
Now, the Stirling approximation of $N!$ yields that the last limit is $e/\sqrt{2\pi}$, as desired.

<!-- theorem-3: body_md -->
\label{thm:Sondow-Hadjicostas}
Let $(\theta_n)_{n\ge 1}$ be a periodic sequence in $\mathbb{C}$ 
and let $q$ be a positive multiple of its period.
If $\sum_{j=1}^q \theta_j = 0$, then
 \begin{equation*}
 \label{eq:Sondow-Hadjicostas}
 \sum_{n\ge 1} \theta_n \left( 1 - n\log \frac{n+1}{n} \right) 
 =
 \sum_{j=1}^{q} \theta_j \log \frac{K(\frac{j+1}{q})^q e^{-j/q}}{K(\frac{j}{q})^q \Gamma(\frac{j+1}{q})}.
 \end{equation*}
<!-- end theorem-3 -->
<!-- corollary-1: body_md -->
\label{cor:applications}
As in Theorem~\ref{thm:Sondow-Hadjicostas}, let
$(\theta_n)_{n\ge 1}$ be a sequence of 
reals of period $q$ such that $\sum_{j=1}^q \theta_j = 0$. 
Then, defining $\theta_0 := \theta_q$, we have
\begin{align*}
&\prod_{n\ge 1} 
\prod_{j=0}^{q-1}\left(\frac{qn+j+1}{qn+j}\right)^{\theta_j(qn+j)} \\
&= e^{-\theta_0} \prod_{n=1}^{q-1} 
\left(\frac{n}{n+1}\right)^{n \theta_n} 
\prod_{j=1}^q \left(\frac{K(\frac{j}{q})^q \Gamma(\frac{j+1}{q})}{K(\frac{j+1}{q})^q e^{-j/q}} \right)^{\theta_j} \\
&= e^{-\theta_0} \prod_{n=1}^{q-1} \left(\frac{n}{n+1}\right)^{n \theta_n} 
\prod_{j=1}^q \left(\frac{G_B(\frac{j+1}{q})^q \Gamma(\frac{j+1}{q})^{q-j}}{G_B(\frac{j}{q})^q \Gamma(\frac{j}{q})^{q-j} e^{-j/q}}\right)^{\theta_j}.
\end{align*}
<!-- end corollary-1 -->
<!-- proof-2: body_md -->
We prove the first equality of the statement.
Since the general term of the series in Theorem~\ref{thm:Sondow-Hadjicostas} tends to zero, we can group the terms in each period, and (after extending the
sequence $(\theta_n)_{n\ge 1}$ by setting $\theta_0 := \theta_q$ as in the statement), we may write
\begin{align*}
 & \sum_{n\ge 1} \theta_n \left( 1 - n\log \frac{n+1}{n} \right) \\ 
&= 
 \displaystyle\sum_{n=1}^{q-1} \theta_n \left( 1 - n\log \frac{n+1}{n} \right) + \displaystyle\sum_{n\ge 1} \sum_{j=0}^{q-1}
\theta_{qn+j} \left( 1 - (qn+j)\log \frac{qn+j+1}{qn+j} \right)
\\
&= 
 \displaystyle\sum_{n=1}^{q-1} \theta_n \left( 1 - n\log \frac{n+1}{n} \right) 
 + \displaystyle\sum_{n\ge 1} \sum_{j=0}^{q-1} \theta_j
 \left(1 - (qn+j)\log \frac{qn+j+1}{qn+j} \right),
\end{align*}
where we have used the fact that $q$ is a multiple of the period of $(\theta_n)_{n\ge 1}$. Now using the assumption that $\sum_{j=0}^{q-1} \theta_j = 0$ twice, we obtain
\begin{align*}
& \sum_{n\ge 1} \theta_n \left( 1 - n\log \frac{n+1}{n} \right) \\
& =
- \theta_0 
 - \displaystyle\sum_{n=1}^{q-1} \left(n \theta_n \log \frac{n+1}{n} \right) 
 - \displaystyle\sum_{n\ge 1} \sum_{j=0}^{q-1}
 \left(\theta_j (qn+j)\log \frac{qn+j+1}{qn+j} \right).
\end{align*}
Next, using Theorem~\ref{thm:Sondow-Hadjicostas} (while paying attention to the negative sign above) and exponentiating give
\[
\prod_{j=1}^q \left(\frac{K(\frac{j}{q})^q \Gamma(\frac{j+1}{q})}{K(\frac{j+1}{q})^q e^{-j/q}} \right)^{\theta_j}
=
e^{\theta_0} \prod_{n=1}^{q-1} \left(\frac{n+1}{n}\right)^{n \theta_n} 
\prod_{n\ge 1} \prod_{j=0}^{q-1}\left(\frac{qn+j+1}{qn+j}\right)^{\theta_j(qn+j)}.
\]
The second equality in the statement can be obtained using the relationship in Equation~\eqref{eq:relation-K-Gamma-GB}.

<!-- theorem-4: body_md -->
\label{theoremmotive}
The relation 
 \begin{multline*}
 \prod_{n\ge 0} e^{-x^2} \left( 1 + \frac{x^2}{(2n+1)^2} \right)^{(2n+1)^2} = 
 \\ 
 (1+e^{-\pi x})^{-x^2} \cdot
 \exp\left( \frac{2 x \operatorname{Li}_2\left(-e^{-\pi x}\right)}{\pi}+\frac{2 \operatorname{Li}_3\left(-e^{-\pi x}\right)}{\pi ^2}-\frac{\pi x^3}{6}+\frac{3 \zeta
 (3)}{2 \pi ^2} \right). 
 \end{multline*}
holds for complex $x$ such that $\Re(x) > 0$.

<!-- proof-5: body_md -->
It is possible to prove this theorem by mimicking 
the proof for the evaluation of the Ramanujan's Equation~\eqref{Ramanujan} given in
\cite[p.~488]{Berndt1998}, but we will give a shorter proof 
using Ramanujan's evaluation. Note that a proof for a 
bisection is ``easier'' than for a multisection.

\medskip

\noindent
Define, as in \cite{AlloucheMarkov2026},
$$
{\mathcal S}^{\rm Ram}(x) := \
\prod_{n \geq 1} e^{-x^{2}} \left( 1 + \frac{x^2}{n^2} 
\right)^{n^{2}}.
$$
Then
$$
\left({\mathcal S}^{\rm Ram}\left(\frac{x}{2}\right)\right)^4 = \
\prod_{n \geq 1} e^{-x^{2}} \left( 1 + \frac{x^2}{4 n^2} 
\right)^{4 n^{2}} = \prod_{\substack{n \geq 1 \\ n \ \text{even}}}
e^{-x^{2}} \left( 1 + \frac{x^2}{n^2} \right)^{n^{2}}
$$
so that
$$
\frac{{\mathcal S}^{\rm Ram}(x)}
{\left({\mathcal S}^{\rm Ram}\left(\frac{x}{2}\right)\right)^4}
= \prod\limits_{\substack{n \geq 1 \\ n \ \text{odd}}}
e^{-x^{2}} \left( 1 + \frac{x^2}{n^2} \right)^{n^{2}} =
\prod_{n \geq 0} e^{-x^2} \ \left( 1 + \frac{x^2}{(2n+1)^2} \right)^{(2n+1)^2}.
$$
Now, consider the polylogarithm function $\operatorname{Li}_{s}(z) = \sum_{n\ge 1} \frac{z^n}{n^s}$ defined for $s$ a positive integer
and $|z| <1$ (the function $z \mapsto \operatorname{Li}_{s}(z)$ can be 
extended to the whole complex plane by analytic continuation). The 
polylogarithm satisfies the functional equation
\begin{equation}
\label{eqn:polylog}
\displaystyle \operatorname{Li} _{s}(-z) +
\operatorname{Li} _{s}(z)=2^{1-s}\operatorname {Li} _{s}(z^{2});
\end{equation}
see~\cite{Lewin1981} for a classical reference.
Using Equation~\eqref{Ramanujan} and the cases $s \in\{2, 3\}$ of Equation~\eqref{eqn:polylog}, we obtain that 
\begin{multline*}
 \prod\limits_{\substack{n \geq 1 \\ n \ \text{odd}}} e^{-x^2} \left( 1 + \frac{x^2}{(2n+1)^2} \right)^{(2n+1)^2} = 
 \\
 (1+e^{-\pi x})^{-x^2}
 \exp\left( \frac{2 x \operatorname{Li}_2\left(-e^{-\pi x}\right)}{\pi}+\frac{2 \operatorname{Li}_3\left(-e^{-\pi x}\right)}{\pi ^2}-\frac{\pi x^3}{6} + \frac{3 \zeta
 (3)}{2 \pi ^2} \right), 
\end{multline*}
 as desired.
