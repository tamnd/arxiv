<!-- definition-1: body_md -->
\label{def:phase}
	A point $x$ of the basin $\mathcal A(\Gamma_i)=\{x\in D_i:\operatorname{dist}(\phi_t^i(x),\Gamma_i)\to0\text{ as }t\to+\infty\}$ has \emph{asymptotic phase} $\Theta_i(x)=\theta\in\mathbb S^1$ if $\|\phi_t^i(x)-\gamma_i(\theta+\omega t)\|\to0$ as $t\to+\infty$. The \emph{isochron} of phase $\theta$ is $\mathcal I_i(\theta):=\{x:\Theta_i(x)=\theta\}$.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:emissionsection}
	Oscillator $i$ emits when it crosses an \emph{emission section} $\Sigma_i\subset U_i$: a connected $C^2$ embedded curve segment, transverse to $F_i$, with $p_i=\gamma_i(0)$ in its interior and $\Sigma_i\cap\Gamma_i=\{p_i\}$. Only crossings with a fixed orientation are counted.
<!-- end definition-2 -->
<!-- definition-3: body_md -->
\label{def:response}
	The \emph{first-order phase response} of oscillator $i$ to an impulse received at phase $\theta$ is
	\[
	h_i(\theta)
	:=\mathrm d\Theta_{i,\gamma_i(\theta)}
	\bigl(B_i(\gamma_i(\theta))\bigr)
	=Z_i(\theta)\cdot B_i(\gamma_i(\theta)).
	\]
<!-- end definition-3 -->
<!-- definition-4: body_md -->
\label{def:H}
	Let
	\begin{equation}\label{eq:H}
		H(\psi):=h_2(\psi)-h_1(-\psi),
		\qquad \psi\in\mathbb S^1.
	\end{equation}
	A zero $\psi_*$ of $H$ is \emph{simple} if $H'(\psi_*)\neq0$; it is then \emph{downward}, or \emph{restoring}, if $H'(\psi_*)<0$, and \emph{upward}, or \emph{destabilising}, if $H'(\psi_*)>0$.
<!-- end definition-4 -->
<!-- remark-1: body_md -->
\label{rem:finitezeros}
The terms of Definition~\ref{def:H} are used for the zeros of any $C^1$ function on $\mathbb S^1$. Simple zeros are isolated, so a $C^1$ function on the circle all of whose zeros are simple has finitely many.
<!-- end remark-1 -->
<!-- definition-5: body_md -->
\label{def:locking}
	A \emph{one-to-one locked state}, or simply \emph{locked state}, is a periodic trajectory of the impulsive system along which each oscillator emits exactly once per period. It is \emph{orbitally asymptotically stable} if every sufficiently close trajectory converges to it up to a translation of time; equivalently, if the corresponding fixed point of the return map on a transverse section is attracting \cite{HirschSmaleDevaney2013}.
<!-- end definition-5 -->
<!-- maintheorem-1: body_md -->
\label{thm:reciprocallocal}
	Let $\psi_*\in(0,2\pi)$ be a simple zero of $H$. For every sufficiently small $\varepsilon>0$ there is a locked state within $O(\varepsilon)$ of $\Gamma_1\times\Gamma_2$, locally unique up to translation of time, whose phase difference at the emissions of oscillator~1 is $\psi_\varepsilon=\psi_*+O(\varepsilon)$. It is orbitally asymptotically stable if $\psi_*$ is downward, and unstable in the relative-phase direction if $\psi_*$ is upward.
<!-- end maintheorem-1 -->
<!-- corollary-1: body_md -->
\label{cor:period}
	In the setting of Theorem~\ref{thm:reciprocallocal}, the fixed point of the return map corresponding to the locked state is hyperbolic, with multipliers
	\[
	\lambda_{\rm rel}=1+\varepsilon H'(\psi_*)+o(\varepsilon)
	\quad\text{in the relative-phase direction},
	\qquad
	\lambda_i=\rho_i+o(1)\quad(i=1,2).
	\]
	The locked state has period
	\begin{equation}\label{eq:period}
		\mathcal T_\varepsilon
		=T-\frac{\varepsilon}{\omega}h_1(-\psi_*)+O(\varepsilon^2)
		=T-\frac{\varepsilon}{\omega}h_2(\psi_*)+O(\varepsilon^2),
	\end{equation}
	and, for each fixed sufficiently small $\varepsilon>0$, it persists, together with its stability type and, since $\psi_*\neq0$, the order of the two emissions, under perturbations of the vector fields, emission sections and impulse maps that are sufficiently small in terms of $\varepsilon$.
<!-- end corollary-1 -->
<!-- definition-6: body_md -->
\label{def:winding}
	The \emph{winding number} along $\Gamma_i$ of a continuous nonvanishing vector field $V$ on $\Gamma_i$ is
	\[
	\operatorname{wind}_{\Gamma_i}(V)
	:=\deg\left(
	\theta\longmapsto
	\frac{V(\gamma_i(\theta))}{\left\lVertV(\gamma_i(\theta))\right\rVert}
	\right),
	\]
	the topological degree of a map $\mathbb S^1\to\mathbb S^1$ \cite{Lee2013}; that is, the total change of a continuous argument of $V$ around the cycle, divided by $2\pi$.

\label{thm:reciprocalgeometric}

\label{thm:phaseclassical}

<!-- lemma-1: body_md -->
\label{lem:phaseidentity}
	Let $z=(x_1,x_2)\in\mathcal V$ and let $y_1:=z^{(2)}_1=\phi^1_{\tau_2(z,\varepsilon)}(x_1)$ be the state of oscillator~1 when oscillator~2 emits. Then
	\begin{equation}\label{eq:exactphase}
		\psi\bigl(\mathcal P_\varepsilon(z)\bigr)-\psi(z)
		=\bigl[\Theta_2(K_2^\varepsilon(x_2))-\Theta_2(x_2)\bigr]
		-\bigl[\Theta_1(K_1^\varepsilon(y_1))-\Theta_1(y_1)\bigr]
		\pmod{2\pi}.
	\end{equation}
	In particular $\psi\circ\mathcal P_0=\psi$ on all of $\mathcal V$, not only on the cycles.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
By~\eqref{eq:phaseidentity}, a free flight of duration $t$ increases $\Theta_1$ and $\Theta_2$ by $\omega t$, so $\psi$ is constant along the product flow. The return consists of two free flights and two impulses. The impulse on oscillator~2 changes $\Theta_2$ by the first bracket and leaves $\Theta_1$ unchanged; the impulse on oscillator~1 changes $\Theta_1$ by the second bracket and leaves $\Theta_2$ unchanged.

<!-- proposition-1: body_md -->
\label{prop:returnexp}
	There exist $r_*,\varepsilon_0>0$ such that, for $(\psi,a)\in\mathcal N_{J,r_*}$ and $|\varepsilon|<\varepsilon_0$, the return $\mathcal P_\varepsilon$ is $C^2$ and has the form
	\begin{align}
		\psi'&=\psi+\varepsilon H(\psi)+\mathcal E(\psi,a,\varepsilon),\label{eq:recphase}\\
		a'&=Da+\varepsilon b(\psi)+Q(\psi,a,\varepsilon),\nonumber
	\end{align}
	with $b\in C^1(J;\mathbb R^2)$ and, uniformly on $\mathcal N_{J,r_*}$,
	\begin{align}
		|\mathcal E(\psi,a,\varepsilon)|
		&\le C\bigl(\varepsilon^2+|\varepsilon|\,\|a\|\bigr),\label{eq:Evalue}\\
		\|D_a\mathcal E(\psi,a,\varepsilon)\|
		&\le C|\varepsilon|,\label{eq:Eaderivative}\\
		\|Q(\psi,a,\varepsilon)\|
		&\le C\bigl(\|a\|^2+|\varepsilon|\,\|a\|+\varepsilon^2\bigr),\label{eq:Qvalue}\\
		\|D_aQ(\psi,a,\varepsilon)\|+\|\partial_\psi Q(\psi,a,\varepsilon)\|
		&\le C\bigl(\|a\|+|\varepsilon|\bigr).\label{eq:Qderivatives}
	\end{align}

<!-- proof-2: body_md -->
The return~\eqref{eq:Poincare} is a composition of $C^2$ free-flight maps, impulse maps and hitting-time functions. In the coordinates~\eqref{eq:tubecoordinates} it is therefore $C^2$ jointly in $(\psi,a,\varepsilon)$ on a neighbourhood of the compact tube $\mathcal N_{J,r_*}$, and compactness makes all constants below uniform.
	
	\emph{Phase component.} By Lemma~\ref{lem:phaseidentity}, $\psi'-\psi$ is the difference of the two impulsive phase changes in~\eqref{eq:exactphase}; in particular it vanishes identically at $\varepsilon=0$, whatever the transverse coordinates. Let $G(\psi,a)$ be the derivative of $\psi'$ with respect to $\varepsilon$ at $\varepsilon=0$. Since the return is $C^2$, $G$ is $C^1$ and, as in the preceding subsection,
	\[
	\psi'-\psi=\varepsilon\,G(\psi,a)+\widetilde{\mathcal E}(\psi,a,\varepsilon),
	\qquad
	|\widetilde{\mathcal E}|\le C\varepsilon^2,
	\qquad
	\|\widetilde{\mathcal E}(\cdot,\cdot,\varepsilon)\|_{C^1}=o(\varepsilon).
	\]
	By~\eqref{eq:C2Taylor}, $G$ is the difference of the values of $\mathrm d\Theta_i(B_i)$ at the two points where the impulses are received at $\varepsilon=0$. For $a=0$ these points are $\gamma_2(\psi)$ and $\gamma_1(-\psi)$, so $G(\psi,0)=h_2(\psi)-h_1(-\psi)=H(\psi)$. Hence~\eqref{eq:recphase} holds with
	\[
	\mathcal E(\psi,a,\varepsilon)
	=\varepsilon\bigl[G(\psi,a)-G(\psi,0)\bigr]+\widetilde{\mathcal E}(\psi,a,\varepsilon),
	\]
	and~\eqref{eq:Evalue} and~\eqref{eq:Eaderivative} follow because $G$ is $C^1$.
	
	\emph{Transverse components.} At $\varepsilon=0$ the transverse return has derivative $D$ at $a=0$: over one period the two transverse variations are multiplied by $\rho_1$ and $\rho_2$. The impulses contribute a term of order $\varepsilon$, written $\varepsilon b(\psi)$, and Taylor's theorem in $(a,\varepsilon)$ on the tube gives~\eqref{eq:Qvalue} and~\eqref{eq:Qderivatives}.

<!-- corollary-2: body_md -->
\label{cor:remainder}
	In the setting of Proposition~\ref{prop:returnexp}, for every $C_0>0$,
	\begin{equation}\label{eq:Epsiderivative}
		\sup\bigl\{|\partial_\psi\mathcal E(\psi,a,\varepsilon)|:\psi\in J,\ \|a\|\le C_0|\varepsilon|\bigr\}=o(\varepsilon).
	\end{equation}
	If $a_\varepsilon:J\to\mathbb R^2$ is Lipschitz with $\|a_\varepsilon\|_{\infty}+\operatorname{Lip}(a_\varepsilon)=O(\varepsilon)$, then
	\begin{equation}\label{eq:EcomposedLip}
		\|\mathcal E(\cdot,a_\varepsilon(\cdot),\varepsilon)\|_{C^0(J)}=O(\varepsilon^2),
		\qquad
		\operatorname{Lip}\bigl(\mathcal E(\cdot,a_\varepsilon(\cdot),\varepsilon)\bigr)=o(\varepsilon),
	\end{equation}
	and if $a_\varepsilon$ is $C^1$ with $\|a_\varepsilon\|_{C^1(J)}=O(\varepsilon)$, then
	\begin{equation}\label{eq:Ecomposed}
		\|\mathcal E(\cdot,a_\varepsilon(\cdot),\varepsilon)\|_{C^1(J)}=o(\varepsilon).
	\end{equation}
<!-- end corollary-2 -->
<!-- proof-3: body_md -->
We use the decomposition $\mathcal E=\varepsilon[G(\psi,a)-G(\psi,0)]+\widetilde{\mathcal E}$ from the preceding proof. Since $DG$ is uniformly continuous on the tube, it gives~\eqref{eq:Epsiderivative}. If $\|a_\varepsilon\|_\infty+\operatorname{Lip}(a_\varepsilon)=O(\varepsilon)$, then $G(\psi,a_\varepsilon(\psi))-G(\psi,0)$ has uniform norm $O(\varepsilon)$, and its Lipschitz constant is $o(1)$ because $DG$ is uniformly continuous on the tube and $a_\varepsilon\to0$ uniformly. Multiplying by $\varepsilon$ and adding $\widetilde{\mathcal E}$ gives~\eqref{eq:EcomposedLip}. If $a_\varepsilon$ is $C^1$ with $\|a_\varepsilon\|_{C^1}=O(\varepsilon)$, the same calculation with derivatives gives~\eqref{eq:Ecomposed}.
<!-- end proof-3 -->
<!-- proof-4: body_md -->
Fix a compact interval $J\subset(0,2\pi)$ with $\psi_*$ in its interior. By Proposition~\ref{prop:returnexp}, the fixed points of the return are the solutions of
	\begin{align}
		0&=\varepsilon H(\psi)+\mathcal E(\psi,a,\varepsilon),\label{eq:fixedphaseproof}\\
		0&=(D-I)a+\varepsilon b(\psi)+Q(\psi,a,\varepsilon).\label{eq:fixedampproof}
	\end{align}
	Consider the second equation first. The map $\Pi(a,\psi,\varepsilon):=(I-D)a-\varepsilon b(\psi)-Q(\psi,a,\varepsilon)$ satisfies $D_a\Pi(0,\psi,0)=I-D$, which is invertible because $0<\rho_i<1$. The implicit function theorem, with $\psi$ as a parameter, therefore gives an open neighbourhood of $(\psi_*,0)$ and a unique $C^1$ function
	\begin{equation}\label{eq:afix}
		a=a_\varepsilon^{\rm fix}(\psi),
		\qquad
		\|a_\varepsilon^{\rm fix}\|_{C^1}=O(\varepsilon),
	\end{equation}
	that solves~\eqref{eq:fixedampproof}. Substituting it into~\eqref{eq:fixedphaseproof} and dividing by $\varepsilon\neq0$ gives, by~\eqref{eq:EcomposedLip} and~\eqref{eq:Ecomposed},
	\begin{equation}\label{eq:scalarfix}
		0=H(\psi)+r_\varepsilon(\psi),
		\qquad
		\|r_\varepsilon\|_{C^0}=O(\varepsilon),
		\qquad
		\|r_\varepsilon\|_{C^1}=o(1).
	\end{equation}
	Choose a closed interval $I$ with $\psi_*$ in its interior and a constant $c>0$ such that $|H'|\ge c$ on $I$. For small $\varepsilon$ we have $|r_\varepsilon'|\le c/2$ on $I$, so $H+r_\varepsilon$ is strictly monotone there, and it has opposite signs at the endpoints of $I$ because $H$ does and $\|r_\varepsilon\|_{C^0}=O(\varepsilon)$. Hence $H+r_\varepsilon$ has exactly one zero $\psi_\varepsilon$ in $I$, and by the mean value theorem
	\begin{equation}\label{eq:phasefixedproof}
		|\psi_\varepsilon-\psi_*|\le \frac{2}{c}\,|r_\varepsilon(\psi_*)|=O(\varepsilon).
	\end{equation}
	With~\eqref{eq:afix}, this is the unique fixed point of the return near $(\psi_*,0)$.
	
	At this fixed point $a=O(\varepsilon)$, and Proposition~\ref{prop:returnexp} gives
	\begin{equation}\label{eq:blockderivative}
		D\mathcal P_\varepsilon=
		\begin{pmatrix}
			1+\varepsilon H'(\psi_*)+o(\varepsilon) & O(\varepsilon)\\
			O(\varepsilon) & D+o(1)
		\end{pmatrix}.
	\end{equation}
	Indeed, the first entry is $1+\varepsilon H'(\psi_\varepsilon)+\partial_\psi\mathcal E$, where $H'(\psi_\varepsilon)=H'(\psi_*)+o(1)$ and $\partial_\psi\mathcal E=o(\varepsilon)$ by~\eqref{eq:Epsiderivative}; the remaining entries are $D_a\mathcal E$, $\varepsilon b'+\partial_\psi Q$ and $D+D_aQ$, bounded by~\eqref{eq:Eaderivative} and~\eqref{eq:Qderivatives}.
	
	Write $M_{11}$, $M_{12}$, $M_{21}$, $M_{22}$ for the four blocks. At $\varepsilon=0$ the multiplier $1$ is separated from $\rho_1$ and $\rho_2$, so for $\lambda$ near $1$ the matrix $\lambda I-M_{22}$ is uniformly invertible and
	\[
	\det(\lambda I-D\mathcal P_\varepsilon)
	=\det(\lambda I-M_{22})
	\bigl[\lambda-M_{11}-M_{12}(\lambda I-M_{22})^{-1}M_{21}\bigr].
	\]
	Since $M_{12}$ and $M_{21}$ are $O(\varepsilon)$, the last product is $O(\varepsilon^2)$ uniformly near $\lambda=1$. Hence the eigenvalue near $1$ is
	\begin{equation}\label{eq:relmultproof}
		\lambda_{\rm rel}
		=M_{11}+O(\varepsilon^2)
		=1+\varepsilon H'(\psi_*)+o(\varepsilon),
	\end{equation}
	and the other two converge to $\rho_1$ and $\rho_2$. For small $\varepsilon>0$ all three lie in the open unit disc if $H'(\psi_*)<0$, and $\lambda_{\rm rel}>1$ if $H'(\psi_*)>0$. By Definition~\ref{def:locking} this proves the stability statements.

<!-- proof-5: body_md -->
The multipliers were obtained in the proof of Theorem~\ref{thm:reciprocallocal}. Over one period $\mathcal T_\varepsilon$ of the locked state, the phase of oscillator~1 increases by $\omega\mathcal T_\varepsilon$ along the flow and by $\varepsilon h_1(-\psi_*)+O(\varepsilon^2)$ through the impulse it receives, by~\eqref{eq:C2Taylor}, \eqref{eq:phasefixedproof} and $h_1\in C^1$. The total is $2\pi$:

	2\pi=\omega\mathcal T_\varepsilon+\varepsilon h_1(-\psi_*)+O(\varepsilon^2),

<!-- lemma-2: body_md -->
\label{lem:winding}
	Let $\Gamma$ be a planar limit cycle with phase gradient $Z$ and tangent field $F$, let $W$ be a continuous nonvanishing vector field along $\Gamma$, and put
	\[
	\Psi(\theta):=Z(\theta)\cdot W(\gamma(\theta)),
	\qquad
	m:=\operatorname{wind}_\Gamma(F)-\operatorname{wind}_\Gamma(W).
	\]
	If $m\neq0$, then $\Psi$ takes both positive and negative values and has at least $2|m|$ zeros. If moreover $\Psi$ is $C^1$ and every zero of $\Psi$ is simple, then the zeros are finite in number and equally many are downward and upward; in particular there are at least $|m|$ of each.
<!-- end lemma-2 -->
<!-- proof-6: body_md -->
By~\eqref{eq:normalisation}, $Z(\theta)\cdot F(\gamma(\theta))>0$ for every $\theta$. The straight-line homotopy
	\[
	(1-s)\frac{F}{\|F\|}+s\frac{Z}{\|Z\|},
	\qquad 0\le s\le1,
	\]
	therefore never vanishes, and $Z$ has the same winding number as $F$.
	
	Let $\alpha:\mathbb R\to\mathbb R$ be a continuous lift of the angle from $W$ to $Z$. Then
	\[
	\alpha(\theta+2\pi)-\alpha(\theta)=2\pi m,
	\qquad
	\Psi(\theta)=\|Z(\theta)\|\,\|W(\gamma(\theta))\|\cos\alpha(\theta).
	\]
	Since $m\neq0$, $\alpha$ is not constant, so the two fields are not perpendicular at every phase and we may choose the phase origin with $\Psi(0)\neq0$. Over $[0,2\pi]$ the lift then runs from $\alpha(0)$ to $\alpha(0)+2\pi m$, neither of which is of the form $\pi/2+k\pi$, and the open interval between them contains exactly $2|m|$ such values. By the intermediate value theorem $\alpha$ attains each of them, at distinct phases in $(0,2\pi)$, and $\Psi$ vanishes there. The same interval contains values at which $\cos\alpha=1$ and values at which $\cos\alpha=-1$, so $\Psi$ takes both signs.
	
If $\Psi$ is $C^1$ and every zero is simple, then $\Psi$ changes sign at each zero, and the zeros are finite in number (Remark~\ref{rem:finitezeros}). Around the circle they are therefore alternately downward and upward, so there are equally many of each. Since there are at least $2|m|$ in all, at least $|m|$ are downward and at least $|m|$ upward.
<!-- end proof-6 -->
<!-- proof-7: body_md -->
Apply Lemma~\ref{lem:winding} to $\Gamma_2$, with $Z=Z_2$ and $W$ given by~\eqref{eq:effectivefield}, so that the integer $m$ of the lemma is~\eqref{eq:windingnumber}; by~\eqref{eq:effectiveidentity}, $\Psi=H$. Hence $H$ has at least $|m|$ downward and at least $|m|$ upward zeros. Each of them other than $\psi=0$ lies in $(0,2\pi)$, and Theorem~\ref{thm:reciprocallocal} applies to it. If $H(0)\neq0$ this covers every zero, which gives at least $|m|$ attracting and at least $|m|$ unstable locked states.

<!-- lemma-3: body_md -->
\label{lem:translation}
	Let $\Gamma$ be a hyperbolic attracting planar limit cycle with phase gradient $Z$. For every $d\in\mathbb R^2\setminus\{0\}$ the function
	\[
	h_d(\theta):=Z(\theta)\cdot d
	\]
	takes both positive and negative values. For $d/\|d\|$ in an open, dense subset of the unit circle with complement of measure zero, every zero of $h_d$ is simple.
<!-- end lemma-3 -->
<!-- proof-8: body_md -->
The constant field $d$ has winding number $0$ and the tangent field has winding number $\pm1$. Lemma~\ref{lem:winding} with $W=d$ therefore applies with $m=\pm1$, and $h_d$ takes both signs.
	
	Write $d_\alpha=(\cos\alpha,\sin\alpha)$ and consider the $C^1$ function
	\[
	\Lambda(\theta,\alpha):=Z(\theta)\cdot d_\alpha
	\qquad\text{on }\mathbb S^1\times\mathbb S^1.
	\]
	At a zero of $\Lambda$ the vector $Z(\theta)$ is perpendicular to $d_\alpha$, and hence
	\[
	\partial_\alpha\Lambda(\theta,\alpha)
	=Z(\theta)\cdot(-\sin\alpha,\cos\alpha)\neq0.
	\]
	Thus $0$ is a regular value and $\Lambda^{-1}(0)$ is a compact $C^1$ curve in the torus \cite{Lee2013}. A direction $\alpha$ is a regular value of the projection of this curve to the $\alpha$-coordinate exactly when $\partial_\theta\Lambda\neq0$ at every zero, that is, when every zero of $h_{d_\alpha}$ is simple. By Sard's theorem the critical values have measure zero, and they form a closed set because the curve is compact. The regular directions are therefore open and of full measure, hence dense.
<!-- end proof-8 -->
<!-- proof-9: body_md -->
Choose $d_2$ such that all zeros of $h_2=Z_2\cdot d_2$ are simple and $h_2(0)\neq0$. By Lemma~\ref{lem:translation} this holds for an open, dense set of directions with complement of measure zero, the second condition excluding only the two directions perpendicular to $Z_2(0)$. Since $h_2$ changes sign and its zeros are simple, it has a downward zero $\psi_s\in(0,2\pi)$.
	
	With $d_1$ replaced by $\eta d_1$, $\eta>0$, the function $H$ becomes
	\[
	H_\eta(\psi)=h_2(\psi)-\eta h_1(-\psi),
	\]
	and $H_\eta\to h_2$ in $C^1(\mathbb S^1)$ as $\eta\to0^+$. By the implicit function theorem, $\psi_s$ continues to a zero $\psi_s(\eta)$ of $H_\eta$ for small $\eta$. After decreasing $\eta_0$ if necessary, these zeros lie in a compact interval $J\subset(0,2\pi)$ and satisfy
	\[
	H_\eta'(\psi_s(\eta))\le -c<0,
	\qquad 0\le\eta\le\eta_0.
	\]
	The estimates of Proposition~\ref{prop:returnexp} are uniform on this compact set of phases and ratios $\eta$. Hence a single $\varepsilon_0>0$ can be chosen for which Theorem~\ref{thm:reciprocallocal} gives an orbitally asymptotically stable locked state for all $0<\eta<\eta_0$ and $0<\varepsilon<\varepsilon_0$.
	
	Simplicity of a zero and the sign of its derivative are open conditions in $C^1$, and the compactness argument is uniform under small changes of $d_1$, $d_2$ and $\eta$. This gives the persistence statement.

<!-- lemma-4: body_md -->
\label{lem:circle}
	Let $H\in C^1(\mathbb S^1)$ have at least one zero, all of them simple, and $H(0)\neq0$, and let $J_*\subset(0,2\pi)$ be a compact interval whose interior contains all zeros of $H$. Let $q_\varepsilon$ be orientation-preserving circle homeomorphisms with lifts
	\begin{equation}\label{eq:circleexp}
		\widetilde q_\varepsilon(\theta)
		=\theta+\varepsilon H(\theta)+r_\varepsilon(\theta),
		\qquad
		\|r_\varepsilon\|_\infty=o(\varepsilon),
		\qquad
		\operatorname{Lip}(r_\varepsilon|_{J_*})=o(\varepsilon).
	\end{equation}
	Then, for every sufficiently small $\varepsilon>0$, $q_\varepsilon$ has topologically Morse--Smale dynamics, with exactly one fixed point near each zero of $H$ and no others, attracting at downward zeros and repelling at upward zeros.
<!-- end lemma-4 -->
<!-- proof-10: body_md -->
Since $H(0)\neq0$, there is a closed neighbourhood $U$ of $0$ on which $|H|\ge c_0>0$. By the uniform bound in~\eqref{eq:circleexp} the displacement $\widetilde q_\varepsilon(\theta)-\theta$ has the sign of $H$ on $U$ for small $\varepsilon$, so $q_\varepsilon$ has no fixed point in $U$.
	
	The zeros of $H$ are finite in number (Remark~\ref{rem:finitezeros}). Around each zero $\theta_*$ choose a closed interval $I_{\theta_*}$ in the interior of $J_*$, with $\theta_*$ in its interior, on which $|H'|\ge c_*>0$, these intervals being pairwise disjoint. By the Lipschitz bound in~\eqref{eq:circleexp} the displacement is strictly monotone on $I_{\theta_*}$ for small $\varepsilon$, and by the uniform bound it has the signs of $H$ at the endpoints. It therefore has exactly one zero there, attracting if $H'(\theta_*)<0$ and repelling if $H'(\theta_*)>0$.
	
	On the complement of $U$ and of the interiors of the intervals $I_{\theta_*}$, which is compact, $|H|$ has a positive minimum, and the $o(\varepsilon)$ remainder allows no further fixed point. Between consecutive fixed points the displacement has constant sign, and since $q_\varepsilon$ preserves orientation every orbit there moves monotonically from the repelling endpoint to the attracting one. This gives the Morse--Smale dynamics.

<!-- lemma-5: body_md -->
\label{lem:circleconjugacy}
	Two orientation-preserving circle homeomorphisms with topologically Morse--Smale dynamics are topologically conjugate if they have the same number of fixed points, with the same cyclic order of attracting and repelling types.
<!-- end lemma-5 -->
<!-- proof-11: body_md -->
Let $q$ and $\widehat q$ be the two maps, and match their fixed points in cyclic order. On each open interval between consecutive fixed points both maps have no fixed point and move every point monotonically from the repelling endpoint to the attracting one. Choose a point $x$ in such an interval and a point $\widehat x$ in the corresponding one, choose an increasing homeomorphism from the interval between $x$ and $q(x)$ onto the interval between $\widehat x$ and $\widehat q(\widehat x)$, and extend it to all iterates. Monotone convergence to the endpoints gives continuity there. The conjugacies on the intervals agree at the fixed points and glue to a circle homeomorphism.

\label{def:pairedregular}
The paired-event transition is \emph{regular} if, on a fixed neighbourhood of the simultaneous-emission state and for all sufficiently small $|\varepsilon|$, it is piecewise $C^1$ and, in fixed incoming and outgoing coordinates,
\begin{equation}\label{eq:pairedReg}

\end{equation}

<!-- lemma-6: body_md -->
\label{lem:pairedepisode}
Under Hypotheses~\ref{hyp:cycles} and~\ref{hyp:equalperiods} and the impulse law of Section~\ref{sec:setting}, with emissions counted as in Definition~\ref{def:singleemission}, the switching set is a smooth hypersurface independent of $\varepsilon$ and each episode records exactly one emission of each oscillator. \begin{equation}\label{eq:uniformTimes}
 |s_{12}-u|\le C|\varepsilon|\ \ (u\ge0),
 \qquad
 |s_{21}+u|\le C|\varepsilon|\ \ (u\le0),
 \qquad
 s_{12}=s_{21}=0\ \ (u=0),
\end{equation}
\begin{equation}\label{eq:uniformReceivers}
 \|w_i^\varepsilon-w_i^0\|\le C|\varepsilon|.
\end{equation}
Outside the switching flow box the order of the emissions is fixed, and no kick carries its receiver onto, across or back across its section.

<!-- proof-13: body_md -->
Take real lifts $\theta_i$ of the asymptotic phases on a neighbourhood of the arcs $|\theta_i|\le3\pi/4$ of the cycles, continued along trajectories, put $\psi=\theta_2-\theta_1$ and $\chi=(\theta_1+\theta_2)/2$, and choose $0<2\kappa<\chi_0<\pi/2$. An incoming section is the level $\chi=-\chi_0$ with $|\psi|<3\kappa$, restricted to a small transverse disc. It is transverse to the product free flow because $\dot\chi=\omega$. The larger interval leaves room for the buffered extension of the incoming chart constructed in the proof of Lemma~\ref{lem:gluing}. The free hits of $\Sigma_i$ have $C^2$ times $t_i(z)$ by the implicit function theorem and the positive lower bound for $\mathrm d g_i(F_i)$ on compact subsegments. On the product of the two cycles,
\[
 t_1=(\chi_0+\psi/2)/\omega,\qquad
 t_2=(\chi_0-\psi/2)/\omega,\qquad
 u=-\psi/\omega.
\]
Thus $\mathrm d u\ne0$ near the simultaneous state after reducing the transverse disc. No kick occurs before the first of the two emissions, so $u=0$ is independent of $\varepsilon$. Since $3\kappa/2<\chi_0$, the two emissions occur a fixed positive time after the level $\chi=-\chi_0$ and before the next level $\chi=2\pi-\chi_0$.

At $u=0$ the two free hits coincide; by Definition~\ref{def:singleemission} both emissions are recorded at that instant, the impulses act as in~\eqref{eq:simultaneous}, and $s_{12}=s_{21}=0$, so the estimates below hold trivially there. At $u=0$ the pre-impulse receiver states coincide with their free counterparts, $w_i^\varepsilon=w_i^0$. Suppose $u>0$, so oscillator 1 emits first. Its kick displaces receiver 2 by $O(|\varepsilon|)$ in $C^0$. In a local signed-time extension, the transverse hitting equation
\[
 g_2\bigl(\phi^2_{\widehat s_{12}}K_2^\varepsilon\phi^2_{t_1(z)}(x_2)\bigr)=0
\]
has a unique $C^2$ solution satisfying $\widehat s_{12}(z,0)=u(z)$ and $|\widehat s_{12}(z,\varepsilon)-u(z)|\le C|\varepsilon|$; the derivative with respect to the hitting time is bounded away from zero. If $\widehat s_{12}>0$, the receiver remains incoming after the kick and emits at that hit. If $\widehat s_{12}\le0$, it is carried onto or through its section by the kick and, by Definition~\ref{def:singleemission}, emits immediately. Hence
\[
 s_{12}=\max\{0,\widehat s_{12}\},\qquad
 |s_{12}-u|\le|\widehat s_{12}-u|\le C|\varepsilon|,
\]
since $u\ge0$. The other order is identical, with $-u$ in place of $u$. Subsequent genuine emissions are a fixed positive time away, while rapid recrossings, if any, are suppressed by the convention. This establishes the event count. The actual reception times differ from their free counterparts by at most $C|\varepsilon|$; smoothness and boundedness of the free flows, together with the $O(|\varepsilon|)$ size of the kicks, give~\eqref{eq:uniformReceivers}.

Outside the switching flow box, $|u|\ge u_0$. Suppose $u\ge u_0$. The first kick then reaches receiver 2 at least a time $u_0$ before its free hit of $\Sigma_2$, and the second kick reaches oscillator 1 a time $s_{12}\ge u_0-C|\varepsilon|$ after its emission. Both kicks are of size $O(|\varepsilon|)$, so for small $|\varepsilon|$ neither carries its receiver onto, across or back across its section. Hence $s_{12}=\widehat s_{12}$ is the $C^2$ hitting time given by the implicit function theorem as in Section~\ref{sec:sectionsimpulses}, and the estimates above hold there as well. The case $u\le-u_0$ is identical.

<!-- lemma-7: body_md -->
\label{lem:gluing}
Under the single-emission convention and the regularity of Definition~\ref{def:pairedregular}, for sufficiently small $|\varepsilon|$ there are a smooth fixed-order return $\mathcal P_\varepsilon$ on a section $\mathcal S_A$, a paired-episode return $\mathcal R_{B,\varepsilon}$ on a section $\mathcal S_B$, and a $C^1$ gluing map $\sigma_\varepsilon$, defined on two fixed-order overlap strips and a local diffeomorphism there, such that
\begin{equation}\label{eq:returnconjugacy}
 \mathcal P_\varepsilon\circ\sigma_\varepsilon
 =\sigma_\varepsilon\circ\mathcal R_{B,\varepsilon}
\end{equation}
wherever both sides are defined.

We specify the domains to control iterations that change charts. Choose $\kappa\in(0,\pi/8)$, then $\chi_0\in(2\kappa,\pi/2)$ and $0<\zeta<\kappa/20$. The fixed-order chart A is the part of $\mathcal S_1$ with $\kappa<\psi<2\pi-\kappa$; chart B is the incoming section $\chi=-\chi_0$ with $|\psi|<2\kappa$. In either chart use a sufficiently small transverse disc. The chart A coordinates are $X_A=(\psi,a_1,a_2)$, as in~\eqref{eq:tubecoordinates}. Its coordinate map extends through $\psi=0$, although its physical fixed-order return need not.

Use enlarged phase intervals
\[
 A^+:\ \kappa-3\zeta<\psi<2\pi-\kappa+3\zeta,
 \qquad
 B^+:\ |\psi|<2\kappa+3\zeta
\]
and enlarged transverse discs. The inner charts have transverse radius $\ell_0$ and the extensions radius $\ell_1>\ell_0$. All event times and return formulas are defined on neighbourhoods of the closed inner domains, by Lemma~\ref{lem:pairedepisode} and the fixed-order hitting-time theorem. The image of either inner chart under one return stays in its enlarged chart for $|\varepsilon|$ small: the angular displacement is $O(|\varepsilon|)$, since there are two kicks of size $O(|\varepsilon|)$ and free flight preserves $\psi$, while the transverse images remain in the larger disc after reducing $\ell_0$. Thus an iteration that leaves an inner chart can still be evaluated in its extension. Chart B and its extension $B^+$ lie in the incoming section of Lemma~\ref{lem:pairedepisode}, since $2\kappa+3\zeta<3\kappa$. Since $u=-\psi/\omega$ on the product of the cycles, the transverse discs can be reduced so that $|u+\psi/\omega|<\kappa/(8\omega)$ on the enlarged B-domain. The strips $\kappa/4<|\psi|<2\kappa+3\zeta$ therefore satisfy $|u|>\kappa/(8\omega)>u_0$ and lie outside the switching flow box, where the order of the emissions is fixed.

On $\mathcal S_B$, the local return $\mathcal R_{B,\varepsilon}$ follows the trajectory from the lifted level $\chi=-\chi_0$ to its next level $\chi=2\pi-\chi_0$. The two emissions lie a fixed positive time away from the entry and exit levels. Since the kicks change $\chi$ by only $O(|\varepsilon|)$, this return traverses exactly one paired episode. Near switching, where $|u|<2u_0$, it is the regular transition followed by a smooth free flight; where $|u|\ge u_0$ it is a smooth fixed-order map, by Lemma~\ref{lem:pairedepisode}. It is therefore continuous, piecewise $C^1$, bi-Lipschitz, and $O(|\varepsilon|)$-close in Lipschitz norm to its unperturbed counterpart. On chart A the return $\mathcal P_\varepsilon$ is the smooth fixed-order map of Section~\ref{sec:localreturn}.

If $\psi<0$, oscillator 1 emits first and $\sigma_\varepsilon=\sigma_0$ is a free-flow hitting map. If $\psi>0$, oscillator 2 emits first and $\sigma_\varepsilon$ is the free flight to $\Sigma_2$, followed by $K_1^\varepsilon$ and the free flight to $\Sigma_1$. On both overlap strips the event order is fixed uniformly in $\varepsilon$ and in the transverse coordinates, once the latter have been restricted. Consequently $\sigma_\varepsilon$ is a local diffeomorphism and
\begin{equation}\label{eq:sigmaclose}
 \|\sigma_\varepsilon-\sigma_0\|_{C^1}\le C|\varepsilon|.
\end{equation}

Because the phase shift in one return is $O(|\varepsilon|)$, their common compact overlap can be reduced slightly so that both compositions are defined there for small $|\varepsilon|$. The equality is exact, including the transverse components.

<!-- lemma-8: body_md -->
\label{lem:returncylinder}
Under the assumptions of Lemma~\ref{lem:gluing}, $\sigma_\varepsilon$ is a diffeomorphism onto its image where the charts are glued, and the quotient $\mathcal M_\varepsilon$ of the two sections by $\sigma_\varepsilon$ has coordinates $\mathcal X_\varepsilon:\mathcal M_\varepsilon\to\mathbb S^1\times B_\ell(0)$, for some $\ell>0$, with uniformly bi-Lipschitz $C^1$ coordinate changes and with angular coordinate equal to the physical relative phase on chart A. \begin{align}
 &\|\mathcal R_\varepsilon-\mathcal R_0\|_\infty
 +\operatorname{Lip}(\mathcal R_\varepsilon-\mathcal R_0)=O(|\varepsilon|),\label{eq:globalLipclose}\\
 &Q_0(\vartheta,a)=\vartheta,\qquad A_0(\vartheta,0)=0,\qquad
 D_a A_0(\vartheta,0)=\operatorname{diag}(\rho_1,\rho_2).\label{eq:uncoupledGlobal}
\end{align}

The coordinate map $X_A$ on $\mathcal S_1$ has a smooth continuation to phases near zero. Define $X_B^0=X_A\circ\sigma_0$ on $B^+$, where $\sigma_0$ is the free-flow map to $\mathcal S_1$, defined also at $\psi=0$. Because free flow preserves the relative phase, the angular component of $X_B^0$ is precisely $\psi$. Choose the transverse domains so that the coordinate images of both enlarged charts are products of their indicated phase intervals and a transverse disc. \[
 D_\varepsilon=X_A\circ\sigma_\varepsilon\circ(X_B^0)^{-1}-\operatorname{Id},
 \qquad \|D_\varepsilon\|_{C^1}\le C|\varepsilon|.
\]

\begin{equation}\label{eq:chartcorrection}
 \|d_\varepsilon\|_{C^1}\le C|\varepsilon|.
\end{equation}

Thus~\eqref{eq:globalLipclose} implies
\[
 \|\mathcal R_\varepsilon(z)-\mathcal R_\varepsilon(z')\|
 \ge (c_0-C|\varepsilon|)\|z-z'\|,
\]
which proves the claimed bi-Lipschitz property.

\label{lem:globalphaseexpansion}
Under the assumptions of Lemma~\ref{lem:returncylinder}, let $a_\varepsilon:\mathbb S^1\to\mathbb R^2$ be Lipschitz with $\|a_\varepsilon\|_\infty+\operatorname{Lip}(a_\varepsilon)\le C|\varepsilon|$. For a consistent lift of the angular component of the return,
\begin{equation}\label{eq:globalphaseexp}
 Q_\varepsilon(\vartheta,a_\varepsilon(\vartheta))
 =\vartheta+\varepsilon H(\vartheta)+r_\varepsilon(\vartheta),
 \qquad \|r_\varepsilon\|_\infty=O(\varepsilon^2),
\end{equation}
and $\operatorname{Lip}(r_\varepsilon|_J)=o(|\varepsilon|)$ on every compact interval $J\subset(0,2\pi)$ contained in the fixed-order chart A.

For local real lifts define
\[
 \Delta_i^\varepsilon(x):=\Theta_i(K_i^\varepsilon x)-\Theta_i(x),\qquad
 f_i(x):=\mathrm d\Theta_i|_x(B_i(x)).
\]
The $C^2$ impulse assumption gives $\Delta_i^\varepsilon(x)=\varepsilon f_i(x)+O(\varepsilon^2)$ uniformly on compact subsets. By~\eqref{eq:phaseidentity} the free flow preserves the physical phase difference $\psi=\Theta_2-\Theta_1$. \begin{equation}\label{eq:pairedPhaseIdentity}
 \psi'-\psi=\Delta_2^\varepsilon(w_2^\varepsilon)-\Delta_1^\varepsilon(w_1^\varepsilon),
\end{equation}

 \psi'-\psi
 =\varepsilon\bigl[h_2(\psi)-h_1(-\psi)\bigr]
 +O(\varepsilon^2+|\varepsilon|\,\|a\|)
 =\varepsilon H(\psi)+O(\varepsilon^2+|\varepsilon|\,\|a\|).
\end{equation}
Here the two physical phases refer to the beginning and end of the \emph{same representative return}. We now verify the change to global coordinates, including a return that changes chart.

For a representative $z$ in the inner B-domain, let $w=X_B^0(z)$ and $w'=X_B^0(\mathcal R_{B,\varepsilon}z)$. These expressions are defined even if the image leaves the inner chart, because it belongs to $B^+$.  \|X_B^0(\mathcal R_{B,0}z)-X_B^0(z)\|\le C\|a\|.

 \|X_B^0(\mathcal R_{B,\varepsilon}z)-X_B^0(\mathcal R_{B,0}z)\|
 \le C|\varepsilon|.

\begin{equation}\label{eq:chartmove}
 \|w'-w\|\le C(\|a\|+|\varepsilon|)=O(|\varepsilon|).
\end{equation}
Using~\eqref{eq:chartcorrection},
\begin{equation}\label{eq:chartOeps2}
 \|d_\varepsilon(w')-d_\varepsilon(w)\|
 \le\operatorname{Lip}(d_\varepsilon)\|w'-w\|=O(\varepsilon^2).
\end{equation}
The first component of~\eqref{eq:chartOeps2} is exactly the difference between the global angular increment and the physical phase increment in the B-representative. \[
 \vartheta'-\vartheta=(\psi'-\psi)+O(\varepsilon^2).
\]
Also $\vartheta=\psi+O(|\varepsilon|)$ in B by~\eqref{eq:chartcorrection}, and $\vartheta=\psi$ in A. Since $H\in C^1(\mathbb S^1)$, $\varepsilon H(\psi)=\varepsilon H(\vartheta)+O(\varepsilon^2)$. Combined with~\eqref{eq:physicalphaseexp}, this proves~\eqref{eq:globalphaseexp}.

\[
 Q_\varepsilon(\psi,a)-\psi=\varepsilon G(\psi,a)+\widetilde{\mathcal E}(\psi,a,\varepsilon),
 \quad G(\psi,0)=H(\psi),\quad
 \|\widetilde{\mathcal E}(\cdot,\cdot,\varepsilon)\|_{C^1}=o(|\varepsilon|).

 \operatorname{Lip}\bigl[G(\cdot,a_\varepsilon(\cdot))-G(\cdot,0)\bigr]=o(1).
\]
Multiplying by $|\varepsilon|$ and adding the remainder yields $\operatorname{Lip}(r_\varepsilon|_J)=o(|\varepsilon|)$.

<!-- theorem-3: body_md -->
\label{thm:globalcircle}
Under the assumptions of Lemma~\ref{lem:returncylinder}, there are constants $C_1,C_2>0$, independent of $\varepsilon$, such that for all sufficiently small $|\varepsilon|$ the global return $\mathcal R_\varepsilon$ has a unique invariant graph $\mathcal C_\varepsilon=\{(\vartheta,a_\varepsilon(\vartheta)):\vartheta\in\mathbb S^1\}$ with $\|a_\varepsilon\|_\infty\le C_1|\varepsilon|$ and $\operatorname{Lip}(a_\varepsilon)\le C_2|\varepsilon|$. Write $\mathcal R_\varepsilon=(Q_\varepsilon,A_\varepsilon)$ and let $\operatorname{Lip}_a$ and $\operatorname{Lip}_\vartheta$ denote separate Lipschitz constants. Choose $\max\{\rho_1,\rho_2\}<\lambda_0<1$. By~\eqref{eq:uncoupledGlobal}, reducing the fixed tube if necessary, and then using~\eqref{eq:globalLipclose}, we have
\begin{equation}\label{eq:graphestimates}
 \operatorname{Lip}_a(A_\varepsilon)\le\lambda_0,\quad
 \operatorname{Lip}(Q_\varepsilon-\vartheta)\le M|\varepsilon|,\quad
 \|A_\varepsilon(\vartheta,0)\|\le M|\varepsilon|.
\end{equation}
The constants are now fixed in the following order. First fix $C_1>M/(1-\lambda_0)$. Since $A_0(\vartheta,0)=0$, we have $\partial_\vartheta A_0(\vartheta,0)=0$, and $C^2$ smoothness of $A_0$ gives $\operatorname{Lip}_\vartheta(A_0|_{\|a\|\le C_1|\varepsilon|})\le M'C_1|\varepsilon|$. Adding the Lipschitz perturbation from~\eqref{eq:globalLipclose} yields
\begin{equation}\label{eq:graphangularnormal}
 \operatorname{Lip}_\vartheta(A_\varepsilon|_{\|a\|\le C_1|\varepsilon|})\le M_1|\varepsilon|,
\end{equation}
where $M_1=M_1(C_1)$ depends on $C_1$ but not on $\varepsilon$. Next fix $C_2>M_1/(1-\lambda_0)$. Only then is $|\varepsilon|$ taken small.

Let $\mathcal G_\varepsilon$ be the closed complete class of periodic Lipschitz maps $u:\mathbb S^1\to\mathbb R^2$ satisfying $\|u\|_\infty\le C_1|\varepsilon|$ and $\operatorname{Lip}(u)\le C_2|\varepsilon|$. \[
 \operatorname{Lip}(q_u-\operatorname{Id})\le M|\varepsilon|(1+C_2|\varepsilon|)=:m_\varepsilon<1.
\]
Hence the lift of $q_u$ is increasing and $q_u$ is an orientation-preserving bi-Lipschitz circle homeomorphism with $\operatorname{Lip}(q_u^{-1})\le(1-m_\varepsilon)^{-1}$. Its graph image is the graph of
\[
 (\mathsf G u)(\vartheta')
 :=A_\varepsilon\bigl(q_u^{-1}(\vartheta'),u(q_u^{-1}(\vartheta'))\bigr).
\]

\[
 \|\mathsf G u\|_\infty\le M|\varepsilon|+\lambda_0\|u\|_\infty,
 \qquad
 \operatorname{Lip}(\mathsf G u)\le
 \frac{M_1|\varepsilon|+\lambda_0\operatorname{Lip}(u)}{1-m_\varepsilon}.
\]
Hence $\|\mathsf G u\|_\infty\le(M+\lambda_0C_1)|\varepsilon|\le C_1|\varepsilon|$, by the choice of $C_1$, and $\operatorname{Lip}(\mathsf G u)\le(M_1+\lambda_0C_2)|\varepsilon|/(1-m_\varepsilon)\le C_2|\varepsilon|$ as soon as $m_\varepsilon\le1-\lambda_0-M_1/C_2$. \[
 \|q_u^{-1}-q_v^{-1}\|_\infty
 \le\frac{M|\varepsilon|}{1-m_\varepsilon}\|u-v\|_\infty.
\]

\begin{equation}\label{eq:graphcontraction}
 \|\mathsf G u-\mathsf G v\|_\infty
 \le(\lambda_0+C|\varepsilon|)\|u-v\|_\infty.
\end{equation}
For $|\varepsilon|$ small this coefficient is less than one. Banach's fixed-point theorem provides a unique invariant graph $a_\varepsilon\in\mathcal G_\varepsilon$.

\[
 \|b'\|\le

 \le(\lambda_0+C\varepsilon^2)\|b\|.
\]
This proves metric normal attraction on a sufficiently small invariant tube. The induced circle map is $q_{a_\varepsilon}$, hence is an orientation-preserving bi-Lipschitz homeomorphism. Lemma~\ref{lem:globalphaseexpansion}, applied to the fixed graph, gives the claimed phase expansion and local remainder estimate.

<!-- proposition-2: body_md -->
\label{prop:sufficientregularity}
Suppose that the $C^2$ local impulse diffeomorphisms satisfy
\begin{equation}\label{eq:sufficientregularity}
 K_i^\varepsilon(\Sigma_i)\subseteq\Sigma_i,
 \qquad i=1,2,
\end{equation}
for all $|\varepsilon|$ sufficiently small, locally near $p_i$. Then the paired-event transition, with emissions recorded as in Definition~\ref{def:singleemission}, is regular in the sense of Definition~\ref{def:pairedregular}. In particular, the regularity assumption of Theorem~\ref{thm:globalreciprocal} follows from~\eqref{eq:sufficientregularity}.
<!-- end proposition-2 -->
<!-- proof-19: body_md -->
Near $\Sigma_i$ introduce $C^2$ coordinates $(v,y)$ with $v=g_i(x)$, so that $\Sigma_i=\{v=0\}$. Let $k_i^\varepsilon(v,y)$ be the normal coordinate of $K_i^\varepsilon$. Exact invariance means $k_i^\varepsilon(0,y)=0$. Hence the fundamental theorem of calculus gives
\[
 k_i^\varepsilon(v,y)
 =v\int_0^1\partial_v k_i^\varepsilon(tv,y)\,\mathrm d t
 =:\alpha_i^\varepsilon(v,y)v.
\]
Since $K_i^0=\operatorname{Id}$, we have $\alpha_i^0=1$. By continuity, on a smaller fixed neighbourhood and for small $|\varepsilon|$, $\alpha_i^\varepsilon\ge1/2$. Equivalently,
\begin{equation}\label{eq:sectionfactor}
 g_i(K_i^\varepsilon(x))=\alpha_i^\varepsilon(x)g_i(x),
 \qquad\alpha_i^\varepsilon(x)>0.
\end{equation}
Thus each kick preserves the emission section and its incoming and outgoing sides. In particular, a receiver which has not yet emitted cannot be carried across its emission section by a kick, and a receiver which has just emitted cannot be kicked back across that emission section.

Use the incoming section of Lemma~\ref{lem:pairedepisode}, and write $u=t_2-t_1$. Write $z=(u,w)$, with $u$ as the normal coordinate to the switching hypersurface and $w$ coordinates along it. Suppose $u\ge0$. The signed time $\widehat s_{12}$ of Lemma~\ref{lem:pairedepisode} is the unique solution of
\[
 g_2\bigl(\phi^2_{\widehat s_{12}}K_2^\varepsilon\phi^2_{t_1(z)}x_2\bigr)=0.
\]
It is jointly $C^2$ by the implicit function theorem, since the derivative with respect to $\widehat s_{12}$ is $\mathrm d g_2(F_2)$, uniformly bounded away from zero. On $u=0$, the receiver is already on $\Sigma_2$ before the kick and remains there by~\eqref{eq:sectionfactor}; hence $\widehat s_{12}(0,w,\varepsilon)=0$. At $\varepsilon=0$, $\widehat s_{12}(u,w,0)=u$. Integrating the derivative in $u$ gives
\[
 \widehat s_{12}(u,w,\varepsilon)
 =u\int_0^1\partial_u \widehat s_{12}(tu,w,\varepsilon)\,\mathrm d t
 =u(1+O(|\varepsilon|)),
\]
uniformly up to $u=0$. This is positive for $u>0$, so $s_{12}=\widehat s_{12}$ there. Similarly, on the other side, $s_{21}=\widehat s_{21}=-u(1+O(|\varepsilon|))$. Thus no immediate impulse-induced emission is needed in this section-preserving case, while the natural emissions still occur.

Let $\Phi_t=(\phi_t^1,\phi_t^2)$ and define the product-space kicks
\[
 J_1^\varepsilon(x_1,x_2)=(K_1^\varepsilon x_1,x_2),
 \qquad
 J_2^\varepsilon(x_1,x_2)=(x_1,K_2^\varepsilon x_2).
\]
Immediately after the paired emissions the two event-order formulas are
\begin{align*}
 Y_{12}^\varepsilon(z)&=J_1^\varepsilon\circ\Phi_{s_{12}(z,\varepsilon)}
 \circ J_2^\varepsilon\circ\Phi_{t_1(z)}(z),\qquad u\ge0,\\
 Y_{21}^\varepsilon(z)&=J_2^\varepsilon\circ\Phi_{s_{21}(z,\varepsilon)}
 \circ J_1^\varepsilon\circ\Phi_{t_2(z)}(z),\qquad u\le0.
\end{align*}
Each formula extends jointly $C^2$ across $u=0$, with signed intermediate times. On that boundary $t_1=t_2$ and $s_{12}=s_{21}=0$, so the values coincide \emph{exactly} because $J_1^\varepsilon J_2^\varepsilon=J_2^\varepsilon J_1^\varepsilon$. Compose both formulas with the smooth free hit to the same fixed outgoing section, obtaining two $C^2$ branches, $E_\varepsilon^+$ from $Y_{12}^\varepsilon$ and $E_\varepsilon^-$ from $Y_{21}^\varepsilon$, with equal boundary values.

At $\varepsilon=0$ the kicks are identities and both extended branches reduce, on their entire domains, to the same free transition $E_0$. Uniform $C^2$ regularity on compact closures and the parameter mean-value theorem imply
\[
 \|E_\varepsilon^\pm-E_0\|_{C^1}\le C|\varepsilon|.
\]
Straighten $u=0$ and restrict to a convex box. A line segment crosses that hyperplane at most once. Integrating the derivative estimate on the one or two pieces, using the equality of boundary values, gives
\[
 \|E_\varepsilon-E_0\|_\infty+\operatorname{Lip}(E_\varepsilon-E_0)\le C|\varepsilon|.
\]
Each branch is $C^2$ up to $u=0$, so $E_\varepsilon$ is piecewise $C^1$, and the last estimate is~\eqref{eq:pairedReg}. This is Definition~\ref{def:pairedregular}.

<!-- theorem-5: body_md -->
\label{thm:reciprocalstruct}
	Fix a sufficiently small $\varepsilon>0$ and an equal-period reference system with such a neighbourhood $\mathcal U$. Assume that $H(0)\neq0$ and that $H$ has at least one zero, all of its zeros being simple. Then, after shrinking $\mathcal U$, every system in $\mathcal U$ has a nearby Lipschitz invariant circle, normally attracting in the metric sense. The induced circle homeomorphism has the same number and cyclic order of attracting and repelling locked states as the reference one, and is topologically conjugate to it. An admissible perturbation may change the vector fields, the impulse maps, the emission sections and the natural periods, provided the stated uniform return regularity is retained.
<!-- end theorem-5 -->
<!-- proof-28: body_md -->
The continuity of the local returns in Lipschitz norm and of the gluing map in the $C^1$ norm, with the uniform bounds, keeps the graph transform a uniform contraction, so the invariant circle persists. Let $\mathcal R$ be the return of a perturbed system, $\mathcal C$ its invariant circle, and $z$ a fixed point of $\mathcal R$ in the common tube. Metric normal attraction gives
	\[
	\operatorname{dist}(z,\mathcal C)
	=\operatorname{dist}(\mathcal R^n z,\mathcal C)
	\le C_*\lambda^n\operatorname{dist}(z,\mathcal C)
	\]
	for every $n$, hence $z\in\mathcal C$: every nearby fixed point is a fixed point of the circle map.
	
	For the reference system, the displacement has constant sign near simultaneous emission, because $H(0)\neq0$. Each locked state lies away from simultaneous emission and is a hyperbolic fixed point of a smooth fixed-order return in dimension three. After shrinking $\mathcal U$, the hyperbolic continuation theorem therefore gives exactly one fixed point of the full return near each reference locked state, with the same stability type, and by the preceding paragraph it lies on the perturbed invariant circle.
	
	It remains to exclude other fixed points. The graph transforms depend continuously on the system as operators in the sup norm, and so do their fixed graphs; hence the displacements of the circle maps vary continuously in $C^0$. On a fixed-order interval around a zero of $H$, the two Lipschitz invariant graphs $a$ and $\widehat a$ do not spoil monotonicity: since $\operatorname{Lip}_a(Q)=O(\varepsilon)$ and both graphs have Lipschitz constant $O(\varepsilon)$, their contribution to the Lipschitz constant of the difference of the displacements is at most
	\[
	\operatorname{Lip}_a(Q)\bigl(\operatorname{Lip}(\widehat a)+\operatorname{Lip}(a)\bigr)=O(\varepsilon^2),
	\]
	well below the slope $c\varepsilon$ of the reference displacement at the fixed small coupling. Outside the chosen neighbourhoods of the fixed points, the reference displacement is bounded away from zero, and $C^0$ closeness excludes new fixed points. The number, type and cyclic order of the fixed points are therefore unchanged, and Lemma~\ref{lem:circleconjugacy} gives the conjugacy.
