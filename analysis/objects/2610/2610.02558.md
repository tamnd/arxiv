<!-- definition-1: body_md -->
\label{def:entropies}
\mbox{}
\begin{enumerate}
\item[(a)] The \emph{joint entropy} of the pair \((X,Y)\) is defined as
\[
H(\mathbf{p}_{(X,Y)})=-\sum_{(i,j)\in I\times J}p_{i,j}\log p_{i,j}.
\]
\item[(b)] The \emph{conditional entropy of \(Y\) given \(X\)} is defined as
\[
H(\mathbf{p}_{Y|X})
=-\sum_{(i,j)\in I\times J}p_{i,j}\log\frac{p_{i,j}}{p_{X,i}}.
\]
\item[(c)] The \emph{mutual information between \(X\) and \(Y\)} is defined as
\[
\mathcal{I}(\mathbf{p}_{X:Y})
=\sum_{(i,j)\in I\times J}p_{i,j}\log\frac{p_{i,j}}{p_{X,i}\,p_{Y,j}}.
\]
\end{enumerate}
where the marginal masses are
\[
p_{X,i}=\sum_{j\in J}p_{i,j},
\qquad
p_{Y,j}=\sum_{i\in I}p_{i,j}.
\]

be given by \eqref{eq:plugjoint}. Then
\\
\textbf{a-)} the bias of the joint entropy plug-in estimator 
\(H(\widehat{\mathbf{p}}_{(X,Y)}^{(n)})\) is
\begin{equation}
\mathrm{Bias}\bigl(H(\widehat{\mathbf{p}}_{(X,Y)}^{(n)})\bigr)
=-\frac{rs-1}{2n}+o\!\left(\frac{1}{n}\right).
\label{eq:biasjoint}
\end{equation}
\textbf{b-)} the bias-corrected joint entropy estimator is
\begin{equation}
H^{(\mathrm{corr})}(\widehat{\mathbf{p}}_{(X,Y)}^{(n)})
=H(\widehat{\mathbf{p}}_{(X,Y)}^{(n)})+\frac{rs-1}{2n}.
\label{eq:corrjoint}
\end{equation}
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
By \eqref{eq:HXY=HZ}, the joint entropy estimator equals the Shannon entropy 
estimator for the univariate variable \(Z\) with \(rs\) categories. Applying 
the Miller-Madow bias formula \cite{Miller1955,Basharin1959,Harris1975} gives 
\eqref{eq:biasjoint}. Equation~\eqref{eq:corrjoint} follows by subtracting 
the estimated bias.
<!-- end proof-1 -->
<!-- theorem-2: body_md -->
\label{thm:cond} 
Let $H(\widehat{\mathbf{p}}_{Y|X}^{(n)})$ be given by \eqref{eq:plugcond}. 
Then under the same conditions as in Theorem~\ref{thm:joint},
\\
\textbf{a-)} the bias of the conditional entropy plug-in estimator 
\(H(\widehat{\mathbf{p}}_{Y|X}^{(n)})\) is
\begin{equation}
\mathrm{Bias}\bigl(H(\widehat{\mathbf{p}}_{Y|X}^{(n)})\bigr)
=-\frac{r(s-1)}{2n}+o\!\left(\frac{1}{n}\right).
\label{eq:biascond}
\end{equation}
\textbf{b-)} the bias-corrected conditional entropy estimator is
\begin{equation}
H^{(\mathrm{corr})}(\widehat{\mathbf{p}}_{Y|X}^{(n)})
=H(\widehat{\mathbf{p}}_{Y|X}^{(n)})+\frac{r(s-1)}{2n}.
\label{eq:corrcond}
\end{equation}
<!-- end theorem-2 -->
<!-- proof-2: body_md -->
Using \eqref{eq:cond} and the linearity of the bias,
\[
\mathrm{Bias}\bigl(H(\widehat{\mathbf{p}}_{Y|X}^{(n)})\bigr)
=\mathrm{Bias}\bigl(H(\widehat{\mathbf{p}}_{(X,Y)}^{(n)})\bigr)
-\mathrm{Bias}\bigl(H(\widehat{\mathbf{p}}_{X}^{(n)})\bigr).
\]
From Theorem~\ref{thm:joint},
\(\mathrm{Bias}(H(\widehat{\mathbf{p}}_{(X,Y)}^{(n)}))=-(rs-1)/(2n)\). 
The marginal entropy \(H(\widehat{\mathbf{p}}_{X}^{(n)})\) has bias 
\(-(r-1)/(2n)\). Therefore
\[
\mathrm{Bias}\bigl(H(\widehat{\mathbf{p}}_{Y|X}^{(n)})\bigr)
=-\frac{rs-1}{2n}+\frac{r-1}{2n}+o\!\left(\frac{1}{n}\right)
=-\frac{rs-r}{2n}+o\!\left(\frac{1}{n}\right)
=-\frac{r(s-1)}{2n}+o\!\left(\frac{1}{n}\right).
\]
This proves \eqref{eq:biascond}; \eqref{eq:corrcond} follows by subtracting 
the estimated bias.
<!-- end proof-2 -->
<!-- theorem-3: body_md -->
\label{thm:mi} 
Let $\mathcal{I}(\widehat{\mathbf{p}}_{X:Y}^{(n)})$ be given by 
\eqref{eq:plugmi}. Then under the same conditions as in Theorem~\ref{thm:joint},
\\
\textbf{a-)} the bias of the mutual information plug-in estimator 
\(\mathcal{I}(\widehat{\mathbf{p}}_{X:Y}^{(n)})\) is
\begin{equation}
\mathrm{Bias}\bigl(\mathcal{I}(\widehat{\mathbf{p}}_{X:Y}^{(n)})\bigr)
=\frac{(r-1)(s-1)}{2n}+o\!\left(\frac{1}{n}\right).
\label{eq:biasmi}
\end{equation}
\textbf{b-)} the bias-corrected mutual information estimator is
\begin{equation}
\mathcal{I}^{(\mathrm{corr})}(\widehat{\mathbf{p}}_{X:Y}^{(n)})
=\mathcal{I}(\widehat{\mathbf{p}}_{X:Y}^{(n)})-\frac{(r-1)(s-1)}{2n}.
\label{eq:corrmi}
\end{equation}
<!-- end theorem-3 -->
<!-- proof-3: body_md -->
Using \eqref{eq:mi} and the linearity of the bias,
\[
\mathrm{Bias}\bigl(\mathcal{I}(\widehat{\mathbf{p}}_{X:Y}^{(n)})\bigr)
=\mathrm{Bias}\bigl(H(\widehat{\mathbf{p}}_{X}^{(n)})\bigr)
+\mathrm{Bias}\bigl(H(\widehat{\mathbf{p}}_{Y}^{(n)})\bigr)
-\mathrm{Bias}\bigl(H(\widehat{\mathbf{p}}_{(X,Y)}^{(n)})\bigr).
\]
The marginal entropy \(H(\widehat{\mathbf{p}}_{X}^{(n)})\) has bias 
\(-(r-1)/(2n)\) and \(H(\widehat{\mathbf{p}}_{Y}^{(n)})\) has bias 
\(-(s-1)/(2n)\). Therefore
\[
\mathrm{Bias}\bigl(\mathcal{I}(\widehat{\mathbf{p}}_{X:Y}^{(n)})\bigr)
=-\frac{r-1}{2n}-\frac{s-1}{2n}+\frac{rs-1}{2n}+o\!\left(\frac{1}{n}\right)
=\frac{(r-1)(s-1)}{2n}+o\!\left(\frac{1}{n}\right).
\]
This proves \eqref{eq:biasmi}; \eqref{eq:corrmi} follows by subtracting 
the estimated bias.
<!-- end proof-3 -->
<!-- theorem-4: body_md -->
\label{thm:asymp}
Let $\widehat{\theta}$ denote any of the three plug-in estimators \eqref{eq:plugjoint}, \eqref{eq:plugcond}, \eqref{eq:plugmi} 
and let $\widehat{\theta}^{(\mathrm{corr})}$ be the corresponding 
bias-corrected estimator, as defined in Theorems~\ref{thm:joint}--\ref{thm:mi}.\\
 Then, as $n\to\infty$,
\begin{equation}\label{eq:asympeq1}
\widehat{\theta}^{(\mathrm{corr})}-\widehat{\theta}
=\frac{C}{2n}+o_p\!\left(\frac{1}{n}\right)
=O_p\!\left(\frac{1}{n}\right),
\end{equation}
and consequently
\begin{equation}\label{eq:asympeq2}
\sqrt{n}\,\bigl(\widehat{\theta}^{(\mathrm{corr})}-\widehat{\theta}\bigr)
\;\xrightarrow{\;\mathbb{P}\;}\;0.
\end{equation}

                 
                                                      
                                   
                   
               
              
   
                                                                            
                                                     
   
Thus, for each of the three measures, the uncorrected and bias-corrected 
estimators are asymptotically equivalent. The asymptotic results established 
in \cite{Ba2019a,Ba2020} therefore apply to both versions.
<!-- end theorem-4 -->
<!-- proof-4: body_md -->
For each of the three measures, the correction term takes the form $C/(2n)$, 
where the constant $C$ depends only on the support sizes $r,s$ and belongs 
to the finite set
\[
C \in \Bigl\{\,rs-1,\;\; r(s-1),\;\; -(r-1)(s-1)\,\Bigr\}.
\]
Since $r,s$ are fixed finite integers, $C$ is bounded by a constant $C_{\max}<\infty$ depending only on the support sizes. Hence \[
\bigl|\widehat{\theta}^{(\mathrm{corr})}-\widehat{\theta}\bigr|
=\frac{C}{2n}\le\frac{C_{\max}}{2n},
\]
which is deterministic and $O(1/n)$. Multiplying by $\sqrt{n}$ gives
$O(1/\sqrt{n})\to 0$.
<!-- end proof-4 -->
<!-- corollary-1: body_md -->
\label{cor:transfer}
Let $\theta$ be the true value of one of the three measures, and let
$\widehat{\theta}$, $\widehat{\theta}^{(\mathrm{corr})}$ be the
plug-in and bias-corrected estimators. Then:
\begin{enumerate}
\item \emph{Consistency.} If
$\widehat{\theta}\xrightarrow{\mathbb{P}}\theta$, then
$\widehat{\theta}^{(\mathrm{corr})}\xrightarrow{\mathbb{P}}\theta$.
\item \emph{Asymptotic normality.} If
$\sqrt{n}(\widehat{\theta}-\theta)\xrightarrow{d}\mathcal{N}(0,\sigma^2)$,
then
$\sqrt{n}(\widehat{\theta}^{(\mathrm{corr})}-\theta)
\xrightarrow{d}\mathcal{N}(0,\sigma^2)$ with the \emph{same} variance
$\sigma^2$.
\item \emph{Variance estimation.} If $\widehat{\sigma}^2$ is a
consistent estimator of $\sigma^2$ based on $\widehat{\theta}$, then
$\widehat{\sigma}^2$ is also consistent for the asymptotic variance
of $\widehat{\theta}^{(\mathrm{corr})}$.
\item \emph{Confidence intervals.} If
$[\widehat{\theta}\pm z_{1-\alpha/2}\widehat{\sigma}/\sqrt{n}]$ is an
asymptotically valid $(1-\alpha)$-confidence interval for $\theta$,
then so is
$[\widehat{\theta}^{(\mathrm{corr})}\pm z_{1-\alpha/2}\widehat{\sigma}/\sqrt{n}]$.
\end{enumerate}
<!-- end corollary-1 -->
<!-- proof-5: body_md -->
All four statements are immediate consequences of
Theorem~\ref{thm:asymp} and Slutsky's lemma. For (2), write
\[
\sqrt{n}\bigl(\widehat{\theta}^{(\mathrm{corr})}-\theta\bigr)
=\sqrt{n}\bigl(\widehat{\theta}-\theta\bigr)
+\sqrt{n}\bigl(\widehat{\theta}^{(\mathrm{corr})}-\widehat{\theta}\bigr).
\]
The first term converges in distribution to
$\mathcal{N}(0,\sigma^2)$, the second converges in probability to $0$;
Slutsky's lemma gives the result. Statements (1), (3), (4) follow by
the same argument.
<!-- end proof-5 -->
<!-- remark-1: body_md -->
\label{rem:why}
The asymptotic variance of the corrected estimator is \emph{the same}
as that of the plug-in estimator: the correction does not inflate the
variance. The corrected estimator therefore dominates the plug-in
estimator: it has the same limiting behaviour and a smaller
finite-sample bias. In particular, the practitioner can use the
corrected estimator while continuing to rely on the asymptotic
results established for the plug-in estimator in \cite{Ba2019a,Ba2020}.
<!-- end remark-1 -->
<!-- remark-2: body_md -->
The almost sure convergence and asymptotic normality established in
\cite{Ba2019a,Ba2020} for the plug-in estimators of joint entropy,
conditional entropy, and mutual information carry over to the bias-corrected
estimators. Indeed, by Theorem~\ref{thm:asymp}, the correction is a
deterministic term of order \(O(1/n)\). It therefore vanishes almost surely,
in probability, and under \(\sqrt{n}\)-scaling. Consequently, the corrected
estimators inherit the same almost sure rates and the same asymptotic normal
distributions as the plug-in estimators.
<!-- end remark-2 -->
<!-- head-1: body_md -->
We derived explicit first-order bias formulas for three bivariate 
information-theoretic measures:
\begin{enumerate}
\item the joint entropy $H(\mathbf{p}_{(X,Y)})$;
\item the conditional entropy $H(\mathbf{p}_{Y|X})$;
\item the mutual information $\mathcal{I}(\mathbf{p}_{X:Y})$.
\end{enumerate}
For each of these three measures, we proposed an explicit bias-corrected 
estimator and proved that the uncorrected and corrected versions are 
\emph{asymptotically equivalent}: they differ by a term of order $O(1/n)$, 
which vanishes at the parametric rate under $\sqrt{n}$-scaling 
(Theorem~\ref{thm:asymp}). This equivalence is the structural reason why the 
bias correction does not alter the asymptotic distribution of the estimators; 
it improves only their finite-sample behaviour. In practice, the bias-corrected 
estimators are therefore preferable, since they preserve all asymptotic 
properties of the uncorrected ones while removing the leading bias term.
