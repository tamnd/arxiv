\label{properties}   Let $(\Omega,\Sigma,\mu)$ be a $\sigma$-finite measure and  $p:\Omega\to [1,+\infty]$  an exponent function
 such that $p_+(\Omega\setminus\Omega_\infty)<\infty$.  Then, the following properties are all satisfied  for every $f\in L^{p(\cdot)}(\Omega)$:




(1)  $\rho(f/\Vert f\Vert)=1$. In particular $\rho(f)\leq 1$ if and only if $\Vert f\Vert\leq 1$

(2) The convergence in modulus for $\rho(\cdot)$ is equivalent to the norm convergence.

\label{ucc2}
 Assume $1<p_-\leq p_+<\infty$. For  $2\geq \delta >0$ there exists $\alpha >0$ depending on $\delta, p^+$ and $p_-$ such that for any $u,v\in  L^{p(\cdot)}(\Omega)$ satisfying $\Vert u\Vert \leq 1$,  $\Vert v\Vert \leq 1$, $\Vert u-v\Vert \geq \delta$ we have
$$ \frac{\rho( u) +\rho( v)} {2} \geq  \rho \left( \frac {u+v}{2}\right) +\alpha.$$

<!-- definition-1: body_md -->
\label{PS} Let $(X,\Vert\cdot\Vert)$ be a Banach space.
Given a sequence of bounded linear projections $\mathcal{P}=\{P_n:X\to X: 1\le n\le \omega\}$,
we say that   $\mathcal{P}$ is a \textit{ Projection Scheme (PS)}   on $X$ if the following  conditions are satisfied:

\begin{itemize}

\item[a)]  For all $1\le n,m\le \omega$, $P_n\circ P_m=P_{\min\{n,m\}}$;

\item[b)] For every $x\in X$,  $\lim_n \Vert P_n(x)-P_\omega (x)\Vert=0$.

\item[c)] $I-P_\omega$ has finite rank.

                                                                        

\end{itemize}
<!-- end definition-1 -->
<!-- example-1: body_md -->
\label{Sch}

Let  $X$ be a Banach space endowed with a Schauder basis $(e_n)_n$.  The family $\mathcal{P}$ consisting of the canonical projections $P_n:X\to X$, defined   by $P_n(\sum_{k=1}^\infty x_k e_k)=\sum_{k=1}^n x_k e_k$  together with  $P_\omega=I$  is a projection scheme on $X$.

<!-- example-2: body_md -->
\label{VLSPS}

Let $(\Omega,\Sigma,\mu)$ be a  $\sigma$-finite measure space.  Let $p: \Omega\to (1,+\infty)$ be a measurable function such that  $1=p_-(\Omega)<p_+(\Omega)<\infty$. Fix a nonincreasing sequence $(\gamma_n)\subset (1,+\infty)$  with $\lim_n\gamma_n=1$. For $n\in\mathbb{N}$, define $P_n: L^{p(\cdot)}(\Omega)\to L^{p(\cdot)}(\Omega)$ by $P_n(f)=f\cdot \chi_{p^{-1}(\gamma_n,\infty)}$ and $P_\omega=I$.  Then $\mathcal{P}=\{P_n: 1\le n\le \omega\}$ is a PS on $L^{p(\cdot)}(\Omega)$. More details, together with a slight modification of this construction, will be given in Section \ref{VLSsection}).

We say that a  sequence $(x_n)_n$ in $X$ is $\mathcal{P}$-convergent to some $x\in X$, denoted by $\mathcal{P}-\lim_n x_n=x$,  if $\lim_n P_k(x_n)=P_k(x)$ for all $k\in\mathbb{N}$ and $\lim_n (I-P_\omega)(x_n)=(I-P_\omega)(x)$.

\label{PS2}
Let $\mathcal{P}=\{P_n: 1\le n\le \omega\}$ be a   projection scheme on $X$.
We  say that $\mathcal{P}$  is  boundedly complete if for every  bounded sequence $(x_n)$  such that  $P_n(x_m)=x_n$ if $n\le m<\omega$, there is  a vector  $x\in P_\omega(X)$ with $P_n(x)=x_n$ for all $n\in\mathbb{N}$.

<!-- definition-4: body_md -->
\label{kUC}
Let $A:X\to X$ be a  bounded operator. We say that $X$ is $A$-uniformly convex ($A$-UC) if for all $\epsilon>0$, there exists some $\gamma(A)>0$ such that for all $x,y\in B_X$  either
$$
\Vert A(x-y)\Vert<\epsilon \mbox{ or }\left\Vert {x+y\over 2}\right\Vert<1-\gamma(A).$$

If $\mathcal{A}=\{A_n: n\in\mathbb{N}\}$ is a family of linear bounded operators from $X$ into $X$, we say that $X$ is $\mathcal{A}$-UC if $X$ is $A_n$-UC for all $A_n\in\mathcal{A}$.
<!-- end definition-4 -->
<!-- example-3: body_md -->
\label{puc}
Let $X=Y\oplus_2 Z$ where $Y$ is uniformly convex, $Z$ is any Banach space and the norm on $X$  is given by $\Vert x\Vert_X:=\left(\Vert y\Vert_Y^2+\Vert z\Vert_Z^2\right)^{1\over 2}$ if $x=y+z$ with $y\in Y$ and $z\in Z$.
Define $A:X\to X$ given by $A(x)=y$ if $x=y+z$. Then $X$ is $A$-UC.

\medskip


Indeed, fix some $\epsilon>0$ and $x_1,x_2\in B_X$ with $x_i=y_i+z_i$, $y_i\in Y$, $z_i\in Z$, $i=1,2$ and assume that $\Vert y_1-y_2\Vert\ge \epsilon$. Using that $Y$ is UC, there exists some $\delta_Y(\epsilon)$ such that \cite[Proposition 1, page 190]{Bea}:
$$
\left\Vert{y_1+y_2\over 2}\right\Vert_Y^2\le {\Vert y_1\Vert^2+\Vert y_2\Vert_Y^2\over 2}-\delta_Y(\epsilon).
$$
By the convexity of the function $\Vert \cdot\Vert^2$, when $\Vert\cdot\Vert$ is any norm, we have
$$
\begin{array}{lll}
\left\Vert {x_1+x_2\over 2}\right\Vert_X^2&\le& {\Vert y_1\Vert_Y^2+\Vert y_2\Vert_Y^2\over 2}-\delta_Y(\epsilon)+ {\Vert z_1\Vert_Z^2+\Vert z_2\Vert_Z^2\over 2}\\
  &= & {\Vert x_1\Vert^2_X+\Vert x_2\Vert_X^2\over 2}-\delta_Y(\epsilon).\\
  \end{array}
$$
Taking square root and adjusting the coefficients we obtain  that Definition \ref{kUC} is satisfied.

<!-- definition-5: body_md -->
\label{AUC}
Let $X$ be a Banach space endowed with a projection scheme  $\mathcal{P}=\{P_n: 1\le n\le \omega\}$.  We say that $X$ is $\mathcal{P}$-almost uniformly convex ($\mathcal{P}$-AUC) if for every $n\in\mathbb{N}$, the projection $P_n$ can be split into the sum of two projections $P_n=A(P_n)+B(P_n)$,
 where $X$ is  $\mathcal{A}$-UC for $\mathcal{A}=\{A(P_n): n\in\mathbb{N}\}$  and rank of $B(P_n)$ is finite for every $n\in \mathbb{N}$.

<!-- lemma-3: body_md -->
\label{Pasymp}
Let $X$ be a Banach space endowed with a projection scheme $\mathcal{P}=\{P_n: n\in\mathbb{N}\cup\{\omega\}\}$ which is boundedly complete. 
 Assume that $(Z_s)$ is a sequence of closed bounded subsets of $X$ Then $(Z_s)$ is $\mathcal{P}$-asymptotically compact if and only if $\lim_s \alpha(P_n(Z_s))=0$ for all $n\in\mathbb{N}$.
<!-- end lemma-3 -->
<!-- proof-1: body_md -->
When $(Z_s)$ is $\mathcal{P}$-asymptotically compact, it can  be easily checked that  $\lim_s\alpha(P_n(Z_s))=0$ para todo $n\in\mathbb{N}$ (without the requirement of being boundedly complete).

\medskip

Let us prove the converse: 
Let $(z_s)$ be a sequence in $X$, with $z_s\in Z_s$ for every $s\in\mathbb{N}$. Since $(I-P_\omega)$ has finite rank, we can assume, without loss of generality,  that there exists some $v_0\in (I-P_\omega)(X)$ such that $\lim_s (I-P_\omega)(z_s)=v_0$

Let $(\epsilon_k)$ be a nonnegative null sequence. Using a diagonal argument, we can construct a subsequence $(s_k)_k$ such that
$$
\alpha\left(P_k(Z_{s_k})\right)<\epsilon_k
$$
for all $k\in\mathbb{N}$. We prove that $\alpha(\{P_n(z_{s_k}):k\in\mathbb{N}\})=0$ for all $n\in\mathbb{N}$. Indeed, fix $n\in\mathbb{N}$. Take  $j>n$.
Then
$
\alpha(\{P_n(z_{s_k}):k\in\mathbb{N}\})=\alpha(\{P_n(z_{s_k}): s_k\ge s_j\})$. For $k,k^\prime$ with $s_k, s_{k^\prime}\ge s_j$ we have:
$$
\begin{array}{lll}
\Vert P_n(z_{s_k})-P_n(z_{s_{k^\prime}})\Vert & = & \Vert P_nP_j(z_{s_k})-P_nP_j(z_{s_{k^\prime}})\Vert\\
 & \le & {K}(\mathcal{P})\Vert P_j(z_{s_k})-P_j(z_{s_{k^\prime}})\Vert.
 \end{array}
$$
Therefore:
$$
\begin{array}{lll}
\alpha(\{P_n(z_{s_k}): s_k\ge s_j\})& \le& {K}(\mathcal{P})\alpha\left(\{P_{j}(z_{s_k}): s_k\ge s_j\}\right)\\
 & \le &  \mathcal{K}(\mathcal{P} )\alpha(P_j(Z_{s_{j}}))\le \epsilon_j.\\
\end{array}
$$
Consequently $\alpha(\{P_n(z_{s_k}):k\in\mathbb{N}\})=0$.  Taking a diagonal argument, we can assume, up to a subsequence,  that  $\{P_n(z_{s})\}_s$ is norm-convergent for all $n\in\mathbb{N}$. Then, there is a further sequence $(u_n)$ such that
$$
\lim_s\Vert P_n(z_s)-u_n\Vert=0\ \forall n\in\mathbb{N}.
$$
In particular $(u_n)$ is norm-bounded.
Take $n\le m<\omega$. We have
$$
P_n(u_m)=\lim_s P_nP_m(z_s)=\lim_s P_n(z_s)=u_n.
$$


Since $\mathcal{P}$ is boundedly complete, there exists $v_1\in P_\omega(X)$ with $P_n(v_1)=u_n$ for all $n\in\mathbb{N}$,   which implies  that $\lim_s P_n(z_s)=P_n(v_1)$ for all $n\in\mathbb{N}$. Since $v_1\in P_\omega(X)$ and $v_0\in (I-P_\omega)(X)$, the above implies that the sequence $(z_s)$ is $\mathcal{P}$-convergent to the vector $v_1+v_0\in X$ and the proof is complete.

<!-- lemma-4: body_md -->
\label{gamma}
Let $X$ be a Banach space. Let $A:X\to X$ be a linear bounded operator and assume that   $X$ is $A$-UC.
Let $C$ be a closed convex bounded subset and $(x_k)$ a sequence in $C$. Then
$$
\lim_{\delta\to 0^+}{\rm diam} \left[A(Z(\{x_k\},C,\delta))\right]=0.
$$
<!-- end lemma-4 -->
<!-- proof-2: body_md -->
In case that $r(\{x_k)\},C)=0$, the sequence $(x_n)$ converges to some $x\in C$ and $\{x\}=Z(\{x_k\},C)$. Thus  $Z(\{x_k\},C,\delta)$ is contained in a ball centered at $x$ with radius $\delta$ and there is nothing to prove. Set $r:=r(\{x_k\},C)>0$.

Fix some $n\in\mathbb{N}$. 
Assume,  to the contrary, that there exists some $c>0$ such that:
$$
 0<c< \inf_{\delta>0}{\rm diam} \left[A(Z(\{x_k\},C,\delta))\right]=\lim_{\delta\to 0^+}{\rm diam}\left[A(Z(\{x_k\},C,\delta))\right]
$$
  For $\epsilon:=\frac{c}{ r+1}>0$ consider the corresponding  $\gamma(A)>0$ stated in the Definition \ref{kUC}. Find some $0<\delta<1$ such that $(r+\delta)(1-\gamma(A))<r$.

\medskip

Since $c<{\rm diam} \left[A(Z(\{x_k\},C,\delta/2))\right]$, we can find two points $y_1, y_2\in Z(\{x_k\},C,\delta/2)$ such that $\Vert  A(y_1-y_2)\Vert \geq c$.

 Then, there exists $k_0\in\mathbb{N}$ such that for all $k\ge k_0$:
 $$
 \Vert x_k-y_1\Vert \le r+\delta,\qquad \Vert x_k-y_2\Vert\le r+\delta.
 $$
For $k\ge k_0$, define $\displaystyle{u_k:=\frac{x_k-y_1}{ r+\delta}}$, $\displaystyle{v_k:=\frac{x_k-y_2}{ r+\delta}}$. Then, $\Vert u_k\Vert \le 1$, $\Vert v_k\Vert \le 1$ and $\Vert A(u_k-v_k)\Vert\ge \frac{c}{ r+\delta}>\frac{c}{ r+1}$. Thus,
$$
\left\Vert x_k-\frac{y_1+y_2}{ 2}\right\Vert=(r+\delta)\left\Vert \frac{u_k+v_k}{ 2}\right\Vert\le (r+\delta)(1-\gamma(A)),$$
for all $k\ge k_0$. Taking limit as $k$ goes to infinity:
$$
\limsup_k\left\Vert x_k-\frac{y_1+y_2}{ 2}\right\Vert\le (r+\delta)(1-\gamma(A))<r,
$$
which is a contradiction with the definition of $r$.

<!-- theorem-1: body_md -->
\label{diam}
Assume that $X$ is endowed with a projection scheme $\mathcal{P}=\{P_n:1\le n\le \omega\}$ which is boundedly complete and $X$ is $\mathcal{P}$-AUC.   Then, the sequence of subsets  $\{Z(\{x_n\},C,\delta_s):s\in\mathbb{N}\}$, where $(\delta_s)$ is a nonnegative null sequence, is $\mathcal{P}$-asymptotically compact.
<!-- end theorem-1 -->
<!-- proof-3: body_md -->
Consider $\alpha(\cdot)$ the Kuratowski measure of noncompactness. 
Notice that $\alpha(\cdot)$ is monotone and sublinear in the sense that
 $\alpha(Z_1)\le \alpha(Z_2)$ if $Z_1\subset Z_2$ and $\alpha(Z_1+Z_2)\le \alpha(Z_1)+\alpha(Z_2)$ for all sets $Z_1,Z_2$ in $X$. Additionally
$\alpha(Z)\le {\rm diam}(Z)$ for all $Z\subset X$.

Fix some $n\in\mathbb{N}$.
According to Definition \ref{kUC}, we can assume that  $P_n=A_n+B_n$  where $B_n$ is a finite-rank operator and $X$ is $\mathcal{A}$-UC for $\mathcal{A}=\{A_n: n\in \mathbb{N}\}$.
Notice that for all bounded set $Z\subset X$ we have that $P_n(Z)\subset A_n(Z)+B_n(Z)$ and $B_n(Z)$ is relatively compact. Hence, it suffices to show that

 $$\lim_{s}{\rm diam} \left(A_n(Z(\{x_n\},C,\delta_s)\right)=0$$
for all $n\in\mathbb{N}$ and this follows from Lemma \ref{gamma}.
<!-- end proof-3 -->
<!-- corollary-1: body_md -->
\label{afps}
Let $X$ be endowed with a boundedly complete projection scheme $\mathcal{P}$ for which $X$ is $\mathcal{P}$-AUC.
 Let $C$ be a closed convex bounded subset, and $T:C\to C$ a nonexpansive mapping. Then, there exists an a.f.p.s.  $(y_n)$ in $C$  which is $\mathcal{P}$-convergent.
<!-- end corollary-1 -->
<!-- proof-4: body_md -->
Take $(x_k)$ an a.f.p.s. in $C$.
It is enough to pick $y_s\in Z_s:=Z(\{x_k\},C,\delta_s)$ such that $\Vert y_s-T(y_s)\Vert <{1\over s} $ where $(\delta_s)$ is a decreasing sequence converging to 0,  apply Theorem \ref{diam} and the definition of $\mathcal{P}$-asymptotic compactness.

<!-- proof-6: body_md -->
If $x=0$ or $\lim_n\Vert x_n\Vert=0$, the assertion is immediate. 
 
 Thus, we can assume $x\ne 0$ and  $l:=\limsup_n\Vert x_n\Vert>0$. Take a subsequence $(x_{n_s})$ such that $l:=\lim_s\Vert x_{n_s}\Vert$ and, without loss of generality, we can assume that $\Vert x_{n_s}\Vert\ne 0$ for all $s\in\mathbb{N}$, $\mathcal{P}-\lim_s {x_{n_s}\over \Vert x_{n_s}\Vert}=0$ and, given $\epsilon>0$ that

$$ \left\Vert \frac{x_{n_s}}{\Vert x_{n_s}\Vert }+\frac{x}{\Vert x\Vert}\right\Vert>S(P_k)-\epsilon.$$


Take $f_s\in X^*$ with $\Vert f_s\Vert=1$, such that
$$ \left\Vert \frac{x_{n_s}}{\Vert x_{n_s}\Vert }+\frac{x}{\Vert x\Vert}\right\Vert=f_s\left(\frac{x_{n_s}}{\Vert x_{n_s}\Vert }+\frac{x}{\Vert x\Vert}\right)=f_s\left(\frac{x_{n_s}}{\Vert x_{n_s}\Vert }\right)+f_s\left(\frac{x}{\Vert x\Vert}\right).$$
Then, $$f_s(x_{n_s}/\Vert x_{n_s}\Vert)\geq S(P_k)-\epsilon-1\quad {\rm and} \quad  f_s(x/\Vert x\Vert)\geq S(P_k)-\epsilon-1.$$

For all $s\in\mathbb{N}$, we have
$$
\begin{array}{lll}
\Vert x_{n_s}+x\Vert & \ge & f_s(x_{n_s}+x)
 =\Vert x_{n_s}\Vert f_s\left(\frac{x_{n_s}}{ \Vert x_{n_s}\Vert}\right)+\Vert x\Vert f_s\left(\frac{x}{ \Vert x\Vert}\right)\\
 &\ge & (S(P_k)-\epsilon-1)\left[\Vert x_{n_s}\Vert+\Vert x\Vert\right].\\
\end{array}
$$
Taking upper  limit when $s$ goes to infinity, the arbitrariness  of $\epsilon$ implies that
$$
\limsup_s\Vert x_{n_s}+x\Vert\ge (S(P_k)-1)\left[\lim_s\Vert x_{n_s}\Vert +\Vert x\Vert\right]= (S(P_k)-1)\left[\limsup_n\Vert x_{n}\Vert +\Vert x\Vert\right].
$$
Finally,
$$
\limsup_n\Vert x_n+x\Vert\ge \limsup_s\Vert x_{n_s}+x\Vert\ge (S(P_k)-1)\left[\limsup_n\Vert x_{n}\Vert +\Vert x\Vert\right]
$$
as we wanted to prove.

<!-- lemma-6: body_md -->
\label{bound} Let $X$ be a Banach space endowed with a Projection Scheme
$\mathcal{P}=\{P_k:1\le k\le \omega\}$. Fix some $k\in\mathbb{N}$. Then for every pair of  bounded sequences $(x_n), (y_n)\subset X$ with $\mathcal{P}-\lim_n x_n=x$, $\mathcal{P}-\lim_n y_n=y$, the following inequality holds:
\begin{equation}\label{ka}
\begin{array}{lll}
\displaystyle{\Vert I-P_k\Vert\limsup_m \limsup_n\Vert x_n-y_m\Vert} &\ge& (S(P_k)-1)  \limsup_n\Vert x_n-x\Vert\\
                 & + & (S(P_k)-1)^2 \limsup_m\Vert y_m-y\Vert \\
                  &+ & (S(P_k)-1)^2 \Vert (I-P_k)(x-y)\Vert\\
\end{array}
\end{equation}

In particular, if $\mathcal{P}$ is premonotone and $X$ is of $\mathcal{P}$-asymptotically $\ell_1$-type, we have
\begin{equation}\label{dl}
\limsup_m \limsup_n\Vert x_n-y_m\Vert\ge \limsup_n\Vert x_n-x\Vert+\limsup_n\Vert y_n-y\Vert.
\end{equation}

Notice that $\limsup_n\Vert x_n-x\Vert=\limsup_n\Vert (I-P_k)(x_n-x)\Vert$, $\limsup_n\Vert y_n-y\Vert=\limsup_n\Vert (I-P_k)(y_n-y)\Vert$ and $\mathcal{P}-\lim_n (I-P_k)(x_n-x)=0=\mathcal{P}-\lim_n (I-P_k)(y_n-y)$.


Then, for all $m\in\mathbb{N}$,
$$
\begin{array}{ll}
 &\displaystyle{\Vert I-P_k\Vert \limsup_n\Vert x_n-y_m\Vert} \\
 \ge &\displaystyle{ \limsup_n\Vert (I-P_k)(x_n-x)+(I-P_k)(x-y_m)\Vert}\\
  \ge &\displaystyle{ (S(P_k)-1) \left[ \limsup_n\Vert (I-P_k)(x_n-x)\Vert+\Vert(I-P_k) (x-y_m)\Vert\right]}\\
  = &\displaystyle{ (S(P_k)-1) \left[ \limsup_n\Vert x_n-x\Vert+\Vert(I-P_k) (x-y_m)\Vert\right]}
\end{array}
$$
Likewise, taking limit when $m$ goes to infinity:
$$
\begin{array}{ll}
& \displaystyle{\limsup_m\Vert (I-P_k)(y_m-x)\Vert }\\
= &\displaystyle{ \limsup_m\Vert (I-P_k)(y_m-y)+ (I-P_k)(y-x)\Vert
}\\
 \ge & \displaystyle{ (S(P_k)-1)\limsup_n\Vert y_n-y\Vert+(S(P_k)-1)\Vert (I-P_k)(y-x)\Vert.}
 \end{array}
$$

Combining the above inequalities we obtain (\ref{ka}). Finally, letting $k\to\infty$, we obtain (\ref{dl}).

\begin{enumerate}

\item $X$ is $\mathcal{P}$-AUC


\item  $X$ is of $\mathcal{P}$-asymptotically $\ell_1$-type and

\item  $

\end{enumerate}

Then $X$ has the FPP.

\begin{array}{lll}

                                  & = & \gamma_k \left(\limsup_n\Vert  x_n\Vert + \Vert x\Vert\right)=2\gamma_k\\
                                  \end{array},
$$
which shows that $S(P_k)\ge 2\gamma_k$ and  $X$ is of $\mathcal{P}$-asymptotically $\ell_1$ type (see \cite{todos}).

We next prove that $\limsup_{\lambda\to 0^+}{\Delta_{P_k}(\lambda)\over \lambda}=0$ for all $k\in\mathbb{N}$, which implies that $\Delta(\mathcal{P})=0$.

$$

\label{PSV}  Assume that  $p_+(\Omega\setminus \Omega_\infty) <\infty$ and the set $\Omega_1\cup\Omega_\infty$ only contains finitely many atoms.
Define $P_n(f)=f\cdot 1_{\{\gamma_n\leq p\le \infty\}}$ for all $n\in\mathbb{N}$  and $P_\omega(f)=f\cdot 1_{\{1< p\le \infty\}}$.

<!-- lemma-8: body_md -->
$\mathcal{P}=\{P_n: 1\le n\le \omega\}$ is a  premonotone projection scheme which is boundedly complete.
<!-- end lemma-8 -->
<!-- proof-12: body_md -->
It is clear that $P_n\circ P_m=P_{\min\{n,m\}}$ if $n<m\le \omega$ and that $I-P_\omega$ has rank in a finite-dimensional closed subspace, since $\Omega_1$ is formed  by just finitely many atoms at most.
To prove $\lim_n \Vert P_n(f)-P_\omega(f)\Vert=\lim_n \Vert f\cdot 1_{\{1<p<\gamma_n\}}\Vert=0$,  observe that $(f\cdot 1_{\{1<p<\gamma_n\}})_n$ is a monotonous sequence converging to zero almost everywhere. Since $p_+(\Omega\setminus \Omega_\infty) <\infty$  the convergence for the modulus implies the convergence for the norm. It is trivial that $\Vert I-P_n\Vert\le 1$ for all $n\in\mathbb{N}$.


\medskip

 To prove bounded completeness, consider a bounded sequence  $(f_n)_n$  in  $L^{p(\cdot)}(\Omega)$, say $\Vert f_n\Vert \leq M$ for all $n\in \mathbb{N}$, such that $P_n(f_m)=f_n$ for $m\geq n$.  In particular $P_n(f_n)=f_n$ which implies that $supp(f_n)\subset \{\gamma_n<p\le \infty\}$.
The above implies that, defining $f(t)=f_n(t)$ whether $p(t)\geq \gamma_n$, we obtain a well-defined measurable function on $\{1<p\le +\infty\}$.
Moreover, the sequence $\{P_n(f)\}_n=\{f \cdot 1_{\{\gamma_n<p\le +\infty\}}\}_n$  is nondecreasing and convergent to $f$ almost everywhere. From the Monotone Convergence Theorem and Lemma \ref{properties} we obtain   $$\rho(f/M)=\lim_n \rho(P_n(f)/M)\le \limsup_n\rho(f_n/M) \leq 1$$ and so $f$ belongs to  $L^{p(\cdot)}(\Omega)$.

<!-- lemma-9: body_md -->
\label{kUC2}
$L^{p(\cdot)}(\Omega)$ is $\mathcal{P}$-AUC
<!-- end lemma-9 -->
<!-- proof-13: body_md -->
Notice that for all $n\in\mathbb{N}$ we can write $P_n=A_n+B_n$ where $A_n(f)=f\cdot 1_{\{\gamma_n\leq p< \infty\}}$ and $B_n(f)=f \cdot 1_{\Omega_\infty}$. It is clear that $B_n$ has finite-rank for every $n\in\mathbb{N}$. Let us prove that $L^{p(\cdot)}(\Omega)$ is $\mathcal{A}$-UC where $\mathcal{A}=\{A_n: n\in\mathbb{N}\}$.

\medskip






 Let  $\rho_n:L^{p(\cdot)}(\Omega)\to [0,\infty]$ be the modular defined by
$$\rho_n(f)=\rho(A_n(f))=\int_{\{\gamma_n\le p <+\infty\}} \vert f(t)\vert^{p(t)}dt.$$ Since $p_+(\Omega\setminus \Omega_\infty)<\infty$,  we can use the uniform convexity of the  modular $\rho_n$ given by Lemma \ref{ucc2}. Hence, for a given  $\delta>0$, there exists some $\alpha_n>0$ (depending on $\gamma_n, \delta $ and $p_+(\Omega\setminus \Omega_\infty))$ such that
$$
\rho_n\left(\frac{f+g}{ 2}\right)\le \frac{\rho_n(f)+\rho_n(g)}{ 2}-\alpha_n
$$
whenever $\Vert A_n(f-g)\Vert>\delta$ and $\Vert A_n(f)\Vert\le \Vert f\Vert \le 1$ and $\Vert A_n(g)\Vert\le \Vert g\Vert\le 1$.

\vskip 0.5cm
Using the additivity of the modular   and Proposition \ref{properties} we deduce
$$
\rho\left(\frac{f+g}{ 2}\right)\le \frac{\rho(f)+\rho(g)}{2}-\alpha_n\le 1-\alpha_n.
$$
and
$$\left\Vert \frac{f+g}{ 2}\right\Vert\le (1-\alpha_n)^{\frac{1}{ p_+(\Omega\setminus \Omega_\infty)}}=1-(1-\alpha_n)^{\frac{1}{ p_+(\Omega\setminus \Omega_\infty)}},
$$
and the proof is complete.

<!-- lemma-10: body_md -->
\label{gammaa} Assume $1 < \gamma<\infty$ and let $f,g\in L^{p(\cdot)}(\Omega)$  be normalized disjointedly supported functions with support in $\{1\le p\leq \gamma\}$. Then:
$$
\Vert f+g\Vert\ge 2^{\frac{1}{\gamma}}
$$
<!-- end lemma-10 -->
<!-- proof-14: body_md -->
Write $r=2^{1/\gamma}$. Since $r>1$ we have
$$ \rho\left(\frac{g}{r}\right)=\int_\Omega \left(\frac{1}{r}\right)^{p(t)} \vert g(t)\vert^{p(t)} d\mu \geq  \left(\frac{1}{r}\right)^\gamma\int_\Omega \vert g(t)\vert^{p(t)} d\mu
=\left(\frac{1}{r}\right)^\gamma.$$
Likewise $\rho(f)\geq  \left(\frac{1}{r}\right)^\gamma$ and
since $f$ and $g$ are disjointly supported we have
$$
\rho\left(\frac{f+g}{ r}\right)=\rho\left(\frac{f}{ r}\right) +\rho\left(\frac{g}{ r}\right)\geq
\left(1 \over r\right)^{\gamma} +\left(1\over r\right)^{\gamma}=1
$$
and, using Lemma \ref{properties},  the proof is complete.

<!-- corollary-4: body_md -->
\label{co} Assume that $(\Omega,\Sigma, \mu)$ is a $\sigma$-finite measure space. Let $p:\Omega\to [1,\infty]$ be a measurable function. The following conditions are equivalent:

\begin{itemize}

\item[1)] $L^{p(\cdot)}(\Omega)$ has the FPP.

\item[2)] $L^{p(\cdot)}(\Omega)$ does not contain isometrically $\ell_1$.

\item[3)] $p_+(\Omega\setminus \Omega_\infty) <\infty$ and the set $\Omega_1\cup\Omega_\infty$ only contains finitely many atoms.

\end{itemize}
