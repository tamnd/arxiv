For the general case of mixed boundary conditions, the functional \( F_h \) changes, while the bilinear form \( A_h \) remains unchanged. More precisely, suppose the boundary \(\partial\Omega\) is partitioned into two disjoint subsets \(\Gamma_D\) and \(\Gamma_N\). We impose the conditions  
\[
u = g_D \text{ on } \Gamma_D, \quad \text{and} \quad \rho\nabla u\cdot\bm{n} = g_N \text{ on } \Gamma_N,
\]
where \(g_D \in H^{1/2}(\Gamma_D)\) and \(g_N \in L^2(\Gamma_N)\).  
Let \(\mathcal{E}^D\) and \(\mathcal{E}^N\) denote the sets of boundary edges lying on \(\Gamma_D\) and \(\Gamma_N\), respectively. Then the functional \(F_h(v)\) takes the form  
\begin{equation*}
F_h(v) = \sum_{E \in \mathcal{T}_h} (f, v)_E  
+ \sum_{e \in \mathcal{E}^N} \langle g_N, v\rangle_e  
- \sum_{e \in \mathcal{E}^D} \left\langle g_D, \rho\nabla v \cdot\bm{n}_e - \frac{\gamma_h \rho_e}{|e|}v\right\rangle_e.
\end{equation*}

<!-- lemma-1: body_md -->
\label{lem:auxCoercCont}
There exists $C>0$, depending on only on $\gamma$ and $n_0$ (see Assumptions \ref{ass:A1} and \ref{ass:A3}), such that for all $u,v\in H^s(\mathcal{T}_h)$, $s>3/2$,
\begin{equation*}
    \sum_{e\in\mathcal{E}_h} \langle \{\rho\nabla\Pi_{h,1}^\nabla u\cdot\bm{n}_e\}_w ,[v] \rangle_e \leq C\left(\sum_{e\in\mathcal{E}_h} \frac{\rho_e}{|e|}\|[v]\|_{L^2(e)} ^2\right)^{1/2} \left(\sum_{E\in\mathcal{T}_h} \rho_E \|\nabla v \|_{L^2(E)}^2\right)^{1/2}.
\end{equation*}
In particular, the left-hand side is bounded by $C\|u\|_{1,h}\|v\|_{1,h}.$
\begin{proof}
    For $e\in\mathcal{E}^I$, by \cref{eq:averageWe} and the triangle inequality,
    \begin{align*}
    \| \{\rho\nabla \Pi^{\nabla}_{h, 1}  v \cdot \bm{n}_e\}_w\|_{L^2(e)} &\leq \frac{\rho_e}{2} \left( \|\nabla \Pi^{\nabla}_{E^+}  v \cdot \bm{n}_e \|_{L^2(e)} + \| \nabla \Pi^{\nabla}_{E^-}  v \cdot \bm{n}_e  \|_{L^2(e)} \right).
    \end{align*}    
    Using \cref{trace_inequality}, $\rho_e\leq \sqrt{2}\rho_e^{1/2}(\rho^\pm)^{1/2}$, the stability of $\Pi_E^\nabla$ and Cauchy-Scharwz, we obtain
    \begin{align*}
    \| \{\rho\nabla \Pi^{\nabla}_{h, 1}  v \cdot \bm{n}_e\}_w\|_{L^2(e)} 
                                                                                                                                                                                         
                                                                                                                                                    
    &\leq C\left(\frac{\rho_e}{|e|}\right)^{1/2} \left(\rho^+ \|\nabla v \|_{L^2(E^+)}^2 + \rho^-\| \nabla v\|_{L^2(E^-)}^2 \right)^{1/2}.
    \end{align*}
    A similar bound holds for $e\in \mathcal{E}^B$ with the only element $E^+$. Therefore, using again Cauchy-Schwarz,

<!-- lemma-2: body_md -->
\label{Coercivity}
There exist positive constants $\gamma$ and $\alpha$ depending only on the shape of the elements in $\mathcal{T}_h$, such that if $\gamma_h \geq \gamma$, then
\begin{equation*}\label{continuity}
    B_h(v,v) \geq \alpha\|v\|^2_{1,h} \quad \forall \: v \in \: V_h.
\end{equation*}
\begin{proof}
We need to deduce a lower bound for
\begin{align*}
    B_h(v, v) = \hat{a}_h(v, v) -  2\sum_{e \in \mathcal{E}_h}\langle  \{\rho\nabla \Pi^{\nabla}_{h, 1}  v \cdot \bm{n}_e\}_w, [v]\rangle_{e} +\sum_{e \in \mathcal{E}_h} \frac{\gamma_h\rho_e}{|e|}\langle [v],[v] \rangle_e.
\end{align*}
The first term is easily bounded using \cref{eq:stabIneq}, where we deduce that 
\begin{align*}
    \hat{a}_h(v, v)\geq \hat{C} \sum_{E\in\mathcal{T}_h} (\rho \nabla v,\nabla v)_E,
\end{align*}
where $\hat{C}:= \min\{1,c_1\}$, with $c_1>0$ defined in \cref{eq:stabIneq}. For the second term, we use \cref{lem:auxCoercCont} and the inequality $2ab\leq a^2/\delta + \delta b^2$ for $\delta>0$ to obtain
\begin{align*}
    2\sum_{e \in \mathcal{E}_h}\langle  \{\rho\nabla \Pi^{\nabla}_{h, 1}&  v \cdot \bm{n}_e\}_w, [v]\rangle_{e} \leq \frac{C^2n_0}{\delta} \sum_{e \in \mathcal{E}_h} \frac{\rho_e}{|e|} \|[v]\|_{L^2(e)}^2 + \delta\sum_{E \in \mathcal{T}_h} \rho_E \|\nabla v \|_{L^2(E)}^2.
\end{align*}
                                                                                                          
Adding all terms, we get
\begin{align*}
     B_h(v, v) \geq \left(\hat{C}-\delta\right)\sum_{E \in \mathcal{T}_h} \rho_E \| \nabla v \|_{L^2(E)}^2 + \sum_{e \in \mathcal{E}_h} \left(\gamma_h- \frac{C^2 n_0}{\delta}\right)\frac{\rho_e}{|e|} \|[v]\|_{L^2(e)}^2.
\end{align*}
Taking $\delta = {\hat{C}}/2$, $\gamma_h \geq \dfrac{2C^2n_0}{\hat{C}} + \dfrac{\hat{C}}{2}$, we conclude that $B_h(v,v)\geq \dfrac{\hat{C}}{2} \|v\|_{1, h}$.
\end{proof}

<!-- lemma-3: body_md -->
\label{Continuity}
Given $u, v \in V_h$, there exists a positive constant $C$, depending only on $\gamma_h$, $n_0$ and the shape regularity of the elements, such that 
\begin{equation*}\label{ContInequality}
    |B_h(u, v)| \leq C\|u\|_{1,h}\|v\|_{1,h}.
\end{equation*}
\begin{proof}
We bound each term of $B_h=\hat{a}_h+\hat{b}_h$ separately. Fix $E\in\mathcal{T}_h$. From standard arguments of the VEM, $\hat{a}^E_h(u,v)\leq \max\{1,c_2\} a^E(u,u)^{1/2}a^E(v,v)^{1/2}$, where $c_2$ is defined in \cref{eq:stabIneq}. Adding over all the elements, it holds that 
\begin{equation}\label{eq_lem1}
|\hat{a}_h(u,v)|\leq \max\{1,c_2\} \|u\|_{1,h}\|v\|_{1,h}.
\end{equation}

Applying \cref{lem:auxCoercCont}, we bound directly
\begin{equation} \label{eq_lem2}
    \sum_{e \in \mathcal{E}_h}\langle \{\rho\nabla \Pi^{\nabla}_{h, 1}  u \cdot \bm{n}_e\}, [v]\rangle_{e} + \langle  \{\rho \nabla \Pi^{\nabla}_{h, 1} v \cdot \bm{n}_e\}, [u]\rangle_{e} \leq 2Cn_0^{1/2} \|u\|_{1,h} \|v\|_{1,h}.
\end{equation}

Finally, the penalty term is bounded by the Cauchy-Schwarz inequality 
\begin{align} \label{eq_lem3}
    \sum_{e \in \mathcal{E}_h}\frac{\gamma_h\rho_e }{|e|}\langle [u],[v] \rangle_e &\leq \gamma_h \left(\sum_{e \in \mathcal{E}_h}\frac{\rho_e }{|e|}\|[u]\|_{L^2(e)}^2\right)^{1/2}\left(\sum_{e \in \mathcal{E}_h}\frac{\rho_e }{|e|}\|[v]\|_{L^2(e)}^2\right)^{1/2}\nonumber\\
    &\leq \gamma_h\|u\|_{1,h}\|v\|_{1,h}.
\end{align}
\end{proof}
The lemma follows from \cref{eq_lem1}, \cref{eq_lem2} and \cref{eq_lem3}.

<!-- remark-4: body_md -->
\label{rem:discreteCoarse}
A discontinuous reduced coarse space $V^{R,D}_0$ can be defined in the same way by not identifying the degrees of freedom associated with a subdomain vertex across the subdomains sharing it. We have $V_0^{R,C}\subset V_0^{R,D}\subset V_0$. The analysis of this paper does not cover this variant. We show a single experiment with $V_0^{R,D}$, reported in \textcolor{red}{\cref{tab:discV0}}, and observed no gain: the coarse space is three to four times larger, while the condition numbers and the iteration counts are essentially the same as those obtained with $V_0^{R,C}$. There is a further advantage in the continuous choice. If the coarse solver is taken to be a discontinuous Galerkin problem posed on $\mathcal{T}_H$, with its own bilinear form and penalty parameter, then the fine and the coarse forms have to satisfy a compatibility relation; see \cite{Feng-Karakashian,10.1007/s10915-017-0453-4}. This is why we restrict our attention to the continuous space $V_0^{R,C}$ in what follows.

\label{N_term2}
    Let $\alpha = [\alpha_1,\ldots,\alpha_m]^T\in\mathbb{R}^m$, $\rho = [\rho_1,\ldots,\rho_m]^T\in\mathbb{R}_+^m$. Denote by $\rho_{ij}$ the harmonic mean \cref{eq:harmonicMean} of $\rho_i$ and $\rho_j$. Define the weighted average of $\alpha$ with weights $\rho$ by
    $$ \beta:=\dfrac{\sum_{j=1}^m \rho_j\alpha_j}{\sum_{j=1}^m \rho_j}.$$
    Then, there exists a constant $C>0$, depending on $m$ and possibly $\rho$, such that
    \begin{equation}\label{eq:ineqAverage}
    \sum_{j=1}^m \rho_j |\alpha_j-\beta|^2 \leq C\sum_{j=1}^{m} \rho_{j,j+1} |\alpha_{j+1}-\alpha_j|^2,
    \end{equation}
    where the indices are understood cyclically, that is, $j=m+1$ corresponds to $j=1$.                                                                                                                                
    \begin{proof}
        Set $\gamma=\alpha-\beta\mathbf{1}$. Let $D_\rho$ be the $m\times m$ diagonal matrix $D_\rho = \operatorname{diag}(\rho)$. By the definition of $\beta$ we have $\mathbf{1}^TD_\rho\gamma=\mathbf{1}^TD_\rho\alpha-\beta\,\mathbf{1}^TD_\rho\mathbf{1}=0$. Let $L_\rho$ be the $m\times m$ matrix such that  $$\sum_{j=1}^{m}\rho_{j,j+1}\,|\gamma_{j+1}-\gamma_j|^2=\gamma^{T}L_\rho\,\gamma.$$ 
        The non-zero entries of $L_\rho$ are $(L_\rho)_{j,j}=\rho_{j-1,j}+\rho_{j,j+1}$ and $(L_\rho)_{j,j+1}=(L_\rho)_{j+1,j}=-\rho_{j,j+1}$, with $\rho_{0,1}=\rho_{m,m+1}=0$. Hence, $L_\rho$ is a symmetric positive semi-definite matrix with $L_\rho\mathbf{1}=0$. The right-hand side in \cref{eq:ineqAverage} is
        $$\sum_{j=1}^m\rho_{j,j+1}|\alpha_{j+1}-\alpha_j|^2=\alpha^TL_\rho\alpha=(\gamma+\beta\mathbf{1})^TL_\rho(\gamma+\beta\mathbf{1})=\gamma^TL_\rho\gamma .$$
        Hence \cref{eq:ineqAverage} is equivalent to $\gamma^TD_\rho\gamma\leq C \gamma^TL_\rho\gamma$ for all $\gamma\neq 0$ with $\mathbf{1}^TD_\rho\gamma=0$, and the best constant is the reciprocal of 
        \begin{equation}\label{eq:lambda2}
        \lambda_2(L_\rho,D_\rho):=\min\Bigl\{\tfrac{\gamma^TL_\rho\gamma}{\gamma^TD_\rho\gamma}\ :\ \gamma\neq0,\ \mathbf{1}^TD_\rho\gamma=0\Bigr\}; 
        \end{equation}
        i.e., the smallest non-zero eigenvalue of the generalized eigenvalue problem $L_\rho x=\lambda D_\rho x$. 
    \end{proof}

<!-- lemma-5: body_md -->
\label{N_term_rho}
Let Assumption \ref{ass:monotonoCoef} hold. Then, the optimal constant of \cref{N_term2} satisfies $C_{\mathrm{opt}}\leq m^2$ and is independent of $\rho_1,\ldots,\rho_m$.
\begin{proof}
   Since $\beta$ minimizes $t\mapsto\sum_j\rho_j|\alpha_j-t|^2$ over $t\in\mathbb{R}$, it holds that
    \begin{equation}\label{eq:minProperty}
   \sum_{j=1}^m\rho_j|\alpha_j-\beta|^2\;\leq\;\sum_{j=1}^m\rho_j|\alpha_j-\alpha_{j_{\max}}|^2 .
    \end{equation}
    Fix $j$ and let $j=j_1,\ldots,j_L=j_{\max}$ be a monotone path of indices. Since the harmonic mean of two positive numbers is bounded below by their minimum, $\rho_{j_l}\leq\rho_{j_{l+1}}$ gives
    \begin{equation}\label{eq:harmonicMonotone}
   \rho_j=\rho_{j_1}\leq\rho_{j_l}=\min\{\rho_{j_l},\rho_{j_{l+1}}\}\leq\rho_{j_l,j_{l+1}},
   \qquad 1\leq l\leq L-1 .
    \end{equation}
    Writing $\alpha_j-\alpha_{j_{\max}}$ as the telescoping sum of the increments along the path, the Cauchy-Schwarz inequality and \cref{eq:harmonicMonotone} yield
    $$\rho_j|\alpha_j-\alpha_{j_{\max}}|^2\leq(L-1)\sum_{l=1}^{L-1}\rho_j\bigl|\alpha_{j_{l+1}}-\alpha_{j_l}\bigr|^2 \leq m\sum_{l=1}^{L-1}\rho_{j_l,j_{l+1}}\bigl|\alpha_{j_{l+1}}-\alpha_{j_l}\bigr|^2,$$
    where $j_{\max}$ is an index for which $\rho_{j_{\max}} = \max_{j} \rho_j$. 
    Summing over $j$ and using that each edge of the patch belongs to at most $m$ of the paths, we obtain $\sum_j\rho_j|\alpha_j-\alpha_{j_{\max}}|^2\leq m^2\sum_{j=1}^m\rho_{j,j+1}|\alpha_{j+1}-\alpha_j|^2$, which together with \cref{eq:minProperty} proves the lemma. 
\end{proof}

\label{rem:qmSharp}
Assumption \ref{ass:monotonoCoef} cannot be dispensed with in \cref{N_term_rho}. Let $\bm{x}$ be an interior node surrounded by four elements, with $(\rho_1,\rho_2,\rho_3,\rho_4)=(M,1,M,1)$ and $M\geq1$; this is the configuration produced by a checkerboard coefficient around an interior vertex of a structured partition that is not quasi-monotone. We then have that in \cref{N_term2} the optimal constant is $C_{\mathrm{opt}}=\frac{M+1}{4}$, which grows linearly in the contrast $M$, and no bound independent of the contrast can hold.

We construct $\chi \in V_h\cap H_0^1(\Omega)$ similarly as in \cite{10.1007/s10915-017-0453-4} for the case of triangular meshes, except that the degrees of freedom of $\chi$ are defined as the $\rho$-weighted averages of the nodal values of the discontinuous function $u\in V_h$. 

For $u\in V_h$ we write $u=\sum_{E\in\mathcal{T}_h}\ \sum_{\bm{x}\in S^E_{\mathcal{N}}}\alpha_E^{(\bm{x})}\,\phi_{E,\bm{x}}$ with $\alpha_E^{(\bm{x})}=u|_E(\bm{x})$, so that $\alpha_E^{(\bm{x})}$ is the value that $u$ takes at the node $\bm{x}$ as seen from the element $E$. We define
\begin{equation*}
    \chi:=\sum_{\bm{x}\in\mathcal{N}_v}\beta^{(\bm{x})}\,\phi^h_{\bm{x}}, \qquad \beta^{(\bm{x})}:=\frac{\sum_{E\in\mathcal{T}_{\bm{x},h}}\rho_E\,\alpha_E^{(\bm{x})}}{\sum_{E\in\mathcal{T}_{\bm{x},h}}\rho_E},
\end{equation*}
where $\phi^h_{\bm{x}}:=\sum_{E\,:\,\bm{x}\in S^E_{\mathcal{N}}}\phi_{E,\bm{x}}$ is the nodal continuous function with $\phi^h_{\bm{x}}(\bm{x})=1$ that vanishes at all the remaining vertices of the mesh. Clearly $\chi\in V_h\cap H^1_0(\Omega)$. Setting $\beta^{(\bm{x})}:=0$ for $\bm{x}\in\mathcal{N}_b$, this reads $\chi=\sum_{E\in\mathcal{T}_h}\ \sum_{\bm{x}\in S^E_{\mathcal{N}}}\beta^{(\bm{x})}\,\phi_{E,\bm{x}}$.

From Assumptions \ref{ass:A1} and \ref{ass:A3}, we have that $\|\phi^h_{\bm{x}}\|_{L^2(E)}^2\leq C_1 h_E^2$. Therefore, grouping the terms by nodes,
\begin{align*}
    \sum_{E\in\mathcal{T}_h}\rho_E \|u-\chi\|_{L^2(E)}^2 \leq  C_1 \max_{E \in \mathcal{T}_h} (d_E)
    \sum_{\bm{x}\in S_{\mathcal{N}}} h_{\bm{x}}^2 \sum_{E\in\mathcal{T}_{\bm{x},h}}
    \rho_E\bigl|\alpha_E^{(\bm{x})}-\beta^{(\bm{x})}\bigr|^2,
\end{align*}
with $h_{\bm{x}} = \max_{E\in\mathcal{T}_{\bm{x},h}} h_E$.

Fix an interior node $\bm{x}\in\mathcal{N}_v$ and let $E_1,\ldots,E_m$ be the elements of $\mathcal{T}_{\bm{x},h}$ enumerated cyclically such that $E_i$ and $E_{i+1}$ share an edge $e_i\in\mathcal{E}_h$ having $\bm{x}$ as an endpoint. Writing $\alpha_i:=\alpha_{E_i}^{(\bm{x})}$, by \cref{N_term2} we have that
\begin{equation*}
   \sum_{E\in\mathcal{T}_{\bm{x},h}}\rho_E\bigl|\alpha_E^{(\bm{x})}-\beta^{(\bm{x})}\bigr|^2
   \leq m^2\sum_{i=1}^{m}\rho_{e_i}\bigl|\alpha_{i+1}-\alpha_i\bigr|^2 ,
\end{equation*}
where $\rho_{e_i}$ is the harmonic mean of the coefficients of $E_i$ and $E_{i+1}$. The difference $\alpha_{i+1}-\alpha_i$ is the jump of $u$ at $\bm{x}$ across $e_i$, that is $[u]|_{e_i}(\bm{x})$. For a boundary node $\bm{x}\in\mathcal{N}_b$ we have $\beta^{(\bm{x})}=0$ and the same bound holds. Summing over the nodes and regrouping by edges,
\begin{equation*}
   \sum_{E\in\mathcal{T}_h}\rho_E\|u-\chi\|^2_{L^2(E)}
   \leq C\sum_{e\in\mathcal{E}_h}\rho_e\sum_{\bm{x}\in e}h_{\bm{x}}^2\bigl|[u]|_e(\bm{x})\bigr|^2 .
\end{equation*}
Finally, the two endpoints of $e$ satisfy $h_{\bm{x}}\leq Ch_e$ by Assumption \ref{ass:A2}; since $[u]|_e\in\mathcal{P}_1(e)$, the lower bound in
\cref{eq:P1equiv} together with $|e|\approx h_e$ from Assumption \ref{ass:A3} gives
\begin{equation*}
   \sum_{\bm{x}\in e} h_{\bm{x}}^2\bigl|[u]|_e(\bm{x})\bigr|^2
   \leq C h_e^2\sum_{\bm{x}\in e}\bigl|[u]|_e(\bm{x})\bigr|^2
   \leq C\,\frac{h_e^2}{|e|}\,\|[u]\|_{L^2(e)}^2
   \leq C\,h_e\|[u]\|_{L^2(e)}^2,
\end{equation*}
and we obtain \cref{sumL2}. Since $|\phi^h_{\bm{x}}|_{H^1(E)}^2\leq C_2$, \cref{sumh1} is deduced similarly.

\label{lem:jumpInterp}
Let $\{\theta_i\}_{i=1}^N\subset C^0(\overline{\Omega})$ satisfy $0\leq\theta_i\leq1$ and
$\sum_{i=1}^N\theta_i=1$. For every $v\in V_h$ it holds that
\begin{equation}\label{eq:jumpInterpSum}
   \sum_{i=1}^N\|[I^h(\theta_i v)]\|^2_{L^2(e)}\leq 3\|[v]\|^2_{L^2(e)}
   \qquad\forall e\in\mathcal{E}_h.
\end{equation}
\begin{proof}
Fix and edge $e$ and let $\bm{x}_1,\bm{x}_2$ be its endpoints. Set $q:=[v]|_e$ and $p_i:=[I^h(\theta_i v)]|_e$. Since $v,I^h(\theta_i v)\in V_h$, then $p_i,q\in\mathcal{P}_1(e)$. As $\theta_i$ is single-valued at each node, the degrees of freedom of $I^h(\theta_i v)$ associated with $\bm{x}_l$ and with $E^{\pm}$ are $\theta_i(\bm{x}_l)v^{\pm}(\bm{x}_l)$, so that
$$p_i(\bm{x}_l)=\theta_i(\bm{x}_l)v^+(\bm{x}_l)-\theta_i(\bm{x}_l)v^-(\bm{x}_l)=\theta_i(\bm{x}_l)\,q(\bm{x}_l),\qquad l\in\{1,2\},$$
and likewise for $e\in\mathcal{E}^B$. By the upper bound in \cref{eq:P1equiv} and
$\sum_{i}\theta_i(\bm{x})^2\leq\sum_{i}\theta_i(\bm{x})=1$, then
$$\sum_{i=1}^N\|p_i\|^2_{L^2(e)}\leq\tfrac{|e|}{2}\sum_{l=1}^2\Bigl(\sum_{i=1}^N\theta_i(\bm{x}_l)^2\Bigr)q(\bm{x}_l)^2\leq\tfrac{|e|}{2}\sum_{l=1}^2 q(\bm{x}_l)^2,$$
and the lower bound in \cref{eq:P1equiv} gives
$\dfrac{|e|}{2}\sum_{l}q(\bm{x}_l)^2\leq3\|q\|^2_{L^2(e)}$.
\end{proof}

\label{lem:discreteSobolev}
                             
Let $v\in V_h\cap H^1_0(\Omega)$ have zero mean value on $\Omega_i$. Then
\begin{equation}\label{eq:discreteSobolev}
\|v\|^2_{L^\infty(\Omega_i)}\;\leq\;C\,\omega_i\,|v|^2_{H^1(\Omega_i)},
\end{equation}
with $C$ independent of $H_i$ and $h_i$.

<!-- lemma-11: body_md -->
Given $u \in V_h$, there exist functions $u_i\in V_i$ for $i\in\{0,1,\ldots N\}$ and a constant $C_0$ independent of $H,h,N,\rho$ such that $u= \sum_{i=0}^N R_i^T u_i$ and
\begin{equation}\label{eq:C0}
   \sum_{i=0}^N\tilde{B}_{h,i}(u_i, u_i) \leq C^2_0\,B_h(u,u), \qquad C_0^2\leq C\,\omega\Bigl(1+\frac{H}{\delta}\Bigr).
\end{equation}

<!-- proof-9: body_md -->
Consider $u_0  \in V_0^{R,C}$ as in \cref{u-u0}. Let $w := u - R_0^Tu_0 \in V_h$ and define $u_i=R_i(I^h(\tilde{\theta}_i w )) \in V_i$, $i=1,...,N$, where \(\{\tilde{\theta}_i\}_{i=1}^N\) is a typical partition of unity for the overlapping subdomains \(\Omega'_i\); see \cite{toselli2006domain}. 

For the  coarse component, since it is a continuous function, all jumps vanish and it holds that
\begin{align*}
    \tilde{B}_{h,0}(u_0, u_0) &= B_h(R_0^T u_0, R_0^T u_0)= \sum_{i=1}^N \hat{a}_h^{\Omega_i}(R_0^T u_0, R_0^T u_0)\leq C \sum_{i=1}^N \rho_{\Omega_i} |R_0^T u_0|^2_{H^1(\Omega_i)}\\
    &\leq C\sum_{i=1}^N \rho_{\Omega_i}\omega_i |\chi|^2_{H^1(\Omega_i)}\leq C \omega B_h(u,u),
\end{align*}
where we have used continuity of the VEM bilinear form, \cref{u-u0} and \cref{lemmachi}.

From \cref{Continuity}, we can bound the contribution of the local components by
\begin{equation}\label{local_bi}
    B_h(R_i^T u_i, R_i^T u_i) \leq C \sum_{E \subset \Omega_i'} \rho_E\left|I^h(\tilde{\theta}_i w)\right|_{H^1(E)}^2 + C \sum_{e \in \mathcal{E}_h} \frac{\rho_e}{|e|} \left\|[I^h(\tilde{\theta}_i w)]\right\|^2_{L^2(e)}.
\end{equation}

To bound the first summation, note that for each \(E \subset \Omega_i'\),
\begin{equation}\label{first_term}
    \rho_E\left| \, I^h(\tilde{\theta}_i w)\right|_{H^1(E)}^2 
    \leq C \left(\rho_E\left\|I^h(\tilde{\theta}_i) \nabla I^h(w)\right\|_{L^2(E)}^2 
    + \rho_E\left\|I^h(w) \nabla I^h(\tilde{\theta}_i)\right\|_{L^2(E)}^2 \right).
\end{equation}

Using that $I^hw=w$ and $0\leq I^h(\tilde{\theta}_i)\leq1$, we obtain 
\begin{align*}
\sum_{E \subset \Omega_i'} \rho_E&\left\|I^h(\tilde{\theta}_i)\nabla I^h(w)\right\|_{L^2(E)}^2  \leq \sum_{E \in {\@fontswitch{\relax}{\mathcal} T}_h} \rho_E\left| w\right|_{H^1(E)}^2 \\
&\leq 2\|u\|_{1,h}^2+2\sum_{i=1}^N\rho_{\Omega_i}\left|R_0^Tu_0\right|_{H^1(\Omega_i)}^2
\leq C\omega\|u\|_{1,h}^2.    
\end{align*}

Here the second inequality follows from $w=u-R_0^Tu_0$ and the triangle inequality, while the third one follows from \cref{h1u0} together with $\sum_{i}\rho_{\Omega_i}|\chi|^2_{H^1(\Omega_i)}\leq C\|u\|^2_{1,h}$, which is a consequence of \cref{sumh1}.  

For the second term in \cref{first_term}, let $\Xi_i:=\{j:\overline{\Omega_j}\cap\overline{\Omega_i}\neq\emptyset\}$, so that $i\in\Xi_i$. Then, $\tilde{\theta}_i$ equals one on $\Omega_i\setminus\bigcup_{j\neq i}\Omega'_j$ and vanishes outside $\Omega'_i$. Hence, $\nabla I^h(\tilde{\theta}_i)$ vanishes outside a neighbourhood of $\partial\Omega_i$ of width $\max_{j\in\Xi_i}\delta_j$ and $\|\nabla I^h(\tilde\theta_i)\|_{L^\infty}\leq C/\delta_i$. Since the number of sets $\Omega_j'$ that intersect $\Omega_i$ is uniformly bounded, it suffices to estimate the contribution of a single $j$. We split 
\begin{equation}\label{eq:splitW}
   w = \underbrace{(u-\chi)}_{=:w_1} +\underbrace{(\chi-u_0)}_{=:w_2} +\underbrace{(u_0-R_0^Tu_0)}_{=:w_3},
\end{equation}
and bound the three contributions separately.

For $w_1$, the elements meeting $\Omega'_j\cap\Omega_i$ lie in a layer of width $C\delta_i$, so by the local form of \cref{sumL2} and $h_e\leq h_i\leq\delta_i$,
\begin{align}
  \frac{\rho_{\Omega_i}}{\delta_i^2}\|w_1\|^2_{L^2(\Omega'_j\cap\Omega_i)} &\leq\frac{C}{\delta_i^2}\sum_{e}\rho_eh_e\|[u]\|^2_{L^2(e)} = C\sum_{e}\frac{h_e^2}{\delta_i^2}\frac{\rho_e\|[u]\|^2_{L^2(e)}}{h_e} \nonumber \\
  &\leq C\sum_{e}\frac{\rho_e}{|e|}\|[u]\|^2_{L^2(e)},\label{eq:w1}
\end{align}
the sums being over the edges of the elements meeting $\Omega'_j\cap\Omega_i$.

For $w_2$, let $\hat{\chi}$ be the mean value of $\chi$ over $\Omega_i$. Since $u_0$ is harmonic in $\Omega_i$ and its degrees of freedom are uniformly bounded by the nodal values of $\chi$, we have $\|w_2\|_{L^\infty(\Omega_i)}\leq C\|\chi-\hat{\chi}\|_{L^\infty(\Omega_i)}$, and \cref{lem:discreteSobolev} together with $|\Omega'_j\cap\Omega_i|\leq CH_i\delta_i$ gives
\begin{equation}\label{eq:w2}
  \frac{\rho_{\Omega_i}}{\delta_i^2}\|w_2\|^2_{L^2(\Omega'_j\cap\Omega_i)} \leq\frac{C\rho_{\Omega_i}}{\delta_i^2}\|\chi-\hat{\chi}\|^2_{L^\infty(\Omega_i)}\, |\Omega'_j\cap\Omega_i| \leq C\,\frac{H_i}{\delta_i}\,\omega_i\,\rho_{\Omega_i}|\chi|^2_{H^1(\Omega_i)} .
\end{equation}
This is the only term that produces the factor $H/\delta$.

For $w_3$, we use a similar argument as in \cite[proof of Theorem 4.1]{CalvoJ}. By the definition of $R_0^T$ we have $R_0^Tu_0=u_0$ on $\partial\Omega_i$, so that $w_3$ vanishes on $\partial\Omega_i$. Cover $\Omega'_j\cap\Omega_i$ with square patches $\pi_1,\ldots,\pi_L$ of side of order $\delta_i$ and with bounded overlap; since the region is a layer of width $C\delta_i$ along $\partial\Omega_i$, $L$ is of order $H_i/\delta_i$ and every $\pi_k$ meets $\partial\Omega_i$. Friedrichs' inequality on $\pi_k$, applicable because $w_3$ vanishes on $\partial\pi_k\cap\partial\Omega_i$, yields $\|w_3\|^2_{L^2(\pi_k)}\leq C\delta_i^2|w_3|^2_{H^1(\pi_k)}$, whence
\begin{equation}\label{eq:w3}
  \frac{\rho_{\Omega_i}}{\delta_i^2}\|w_3\|^2_{L^2(\Omega'_j\cap\Omega_i)} \leq\frac{C\rho_{\Omega_i}}{\delta_i^2}\sum_{k=1}^L\|w_3\|^2_{L^2(\pi_k)} \leq C\rho_{\Omega_i}\sum_{k=1}^L|w_3|^2_{H^1(\pi_k)} \leq C\rho_{\Omega_i}|w_3|^2_{H^1(\Omega_i)},
\end{equation}
and $|w_3|^2_{H^1(\Omega_i)}\leq 2|\chi-u_0|^2_{H^1(\Omega_i)}+2|\chi-R_0^Tu_0|^2_{H^1(\Omega_i)}\leq C\omega_i|\chi|^2_{H^1(\Omega_i)}$ by \cref{h1u0}.

On the other hand, the second summation in \cref{local_bi} is handled by \cref{lem:jumpInterp}. After multiplying \cref{eq:jumpInterpSum} by $\rho_e/|e|$ and summing over all edges,
\begin{equation}\label{eq:jumpsAssembled}
  \sum_{i=1}^N\sum_{e\in\mathcal{E}_h} \frac{\rho_e}{|e|}\bigl\|[I^h(\tilde{\theta}_i w)]\bigr\|^2_{L^2(e)} \;\leq\; 3\sum_{e\in\mathcal{E}_h}\frac{\rho_e}{|e|}\bigl\|[w]\bigr\|^2_{L^2(e)}=3\sum_{e}\frac{\rho_e}{|e|}\|[u]\|^2_{L^2(e)}\leq3\|u\|^2_{1,h},
\end{equation}
where we used the fact that $[w]=[u]$ since $R_0^Tu_0$ is continuous on $\Omega$ and vanishes on $\partial\Omega$.

Collecting \cref{eq:w1}, \cref{eq:w2}, \cref{eq:w3} and \cref{eq:jumpsAssembled}, summing over $i$ and using that $$\sum_i\rho_{\Omega_i}|\chi|^2_{H^1(\Omega_i)}\leq C\|u\|^2_{1,h}$$ by \cref{sumh1}, together with the bounded overlap of the sets $\Omega_i'$, we obtain \cref{eq:C0} with $C_0^2\leq C\omega(1+H/\delta)$, and the proof is complete.

\begin{proof}[Proof of \cref{main_thm}]
From the abstract additive Schwarz theory \cite[Theorem 2.7]{toselli2006domain}, we have that 
$$C_0^{-2}\,B_h(u,u)\;\leq\;B_h(P_{ad}u,u)\;\leq\; \bigl(N^C+1\bigr)\,B_h(u,u)\qquad\forall\,u\in V_h,$$
where $N^C$ is the minimum number of colors needed to paint the overlapping subdomains $\Omega_i'$ such that no pair of subdomains of the same color intersect. Therefore $\kappa(B_{ad}^{-1}B)\leq C_0^2(N^C+1)$. The conclusion follows from standard arguments; see \cite[Theorem 2.7, Lemma 2.10]{toselli2006domain}.
\end{proof}
