<!-- definition-1: body_md -->
Let $m\ge1$.  We define
\[
\mathbb D_M^{m,p,\lambda}(I;E_0,E)
:=\{u\in\mathcal M^{p,\lambda}(I;E_0):D_t^m u\in\mathcal M^{p,\lambda}(I;E)\},
\]
where $D_t^m u$ is taken in the sense of $E$-valued distributions.  The graph norm is
\[
\|u\|_{\mathbb D_M^{m,p,\lambda}}
:=\|u\|_{\mathcal M^{p,\lambda}(E_0)}
+\|D_t^m u\|_{\mathcal M^{p,\lambda}(E)}.
\]

<!-- lemma-1: body_md -->
\label{lem:kernel-max}
Let $k\in L^1(\mathbb R)$ satisfy
\[
|k(t)|\le C_N(1+|t|)^{-N}
\]
for some $N>2$, and put $k_h(t)=h^{-1}k(t/h)$.  Then for every locally integrable Banach-valued function $f$,
\[
\|k_h*f(t)\|_X\le C M(\|f\|_X)(t)
\]
for almost every $t$, uniformly in $h>0$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Write $F=\|f\|_X$.  Since
\[
\|k_h*f(t)\|_X
\le \int_{\mathbb R}|k_h(s)|F(t-s)\,ds,
\]
it is enough to estimate the scalar convolution on the right.  Decompose $\mathbb R$ into
\[
A_0:=\{|s|<h\},
\qquad
A_\ell:=\{2^\ell h\le |s|<2^{\ell+1}h\},\quad \ell\ge0.
\]
On $A_0$ we have $|k_h(s)|\le C h^{-1}$, and hence
\[
\int_{A_0}|k_h(s)|F(t-s)\,ds
\le C h^{-1}\int_{t-h}^{t+h}F(\tau)\,d\tau
\le C MF(t).
\]
On $A_\ell$, the decay of $k$ gives
\[
|k_h(s)|
=h^{-1}|k(s/h)|
\le C h^{-1}2^{-N\ell}.
\]
Moreover $t-A_\ell\subset(t-2^{\ell+1}h,t+2^{\ell+1}h)$, so
\[
\int_{A_\ell}F(t-s)\,ds
\le C2^\ell h\,MF(t).
\]
Therefore
\[
\int_{A_\ell}|k_h(s)|F(t-s)\,ds
\le C2^{-(N-1)\ell}MF(t).
\]
The geometric series is summable because $N>2$, and the resulting constant is independent of $h$.  This proves the claimed pointwise bound.

<!-- theorem-1: body_md -->
\label{thm:whole-line}
Let $E_0\hookrightarrow E$ continuously, $m\ge2$, $1<p<\infty$, $0\le\lambda<1$, and $0<j<m$.  Then
\[
D_t^j:\mathbb D_M^{m,p,\lambda}(\mathbb R;E_0,E)
\longrightarrow
\mathcal M^{p,\lambda}(\mathbb R;X_{j/m})
\]
is bounded.  More precisely,
\[
\|D_t^j u\|_{\mathcal M^{p,\lambda}(\mathbb R;X_{j/m})}
\le C
\bigl(
\|u\|_{\mathcal M^{p,\lambda}(\mathbb R;E_0)}
+\|D_t^m u\|_{\mathcal M^{p,\lambda}(\mathbb R;E)}
\bigr).
\]
No geometric assumption on the Banach spaces is required.
<!-- end theorem-1 -->
<!-- proof-2: body_md -->
Set $\theta=j/m$.  The argument is a scale-by-scale decomposition of $D_t^j u$ into an $E_0$-part and an $E$-part, chosen so that the Peetre parameter exactly matches the time-frequency scale.

For $\eta\in\mathbb R$ define
\[
a(\eta):=\frac{(i\eta)^j}{1+\eta^{2m}},
\qquad
c(\eta):=i^{j-m}\frac{\eta^{j+m}}{1+\eta^{2m}}.
\]
Because $0<j<m$, both symbols are proper rational functions.  Their poles are the nonreal roots of $-1$, so the inverse Fourier transforms $k_a$ and $k_c$ are finite linear combinations of exponentially decaying functions on the two half-lines.  In particular, for every $N$,
\[
|k_a(t)|+|k_c(t)|\le C_N(1+|t|)^{-N}.
\]

Fix $h>0$.  Let $A_h$ and $C_h$ be the convolution operators whose multipliers are
\[
h^{-j}a(h\xi)
=\frac{(i\xi)^j}{1+(h\xi)^{2m}},
\]
and
\[
h^{m-j}c(h\xi),
\]
respectively.  If the second multiplier is applied to $D_t^m u$, then
\[
h^{m-j}c(h\xi)(i\xi)^m
=\frac{(i\xi)^j(h\xi)^{2m}}{1+(h\xi)^{2m}}.
\]
Consequently the elementary identity
\[
1=\frac1{1+(h\xi)^{2m}}+
\frac{(h\xi)^{2m}}{1+(h\xi)^{2m}}
\]
gives, first for smooth compactly supported functions and then by distributional continuity,
\begin{equation}\label{eq:rational-splitting}
D_t^j u=A_hu+C_hD_t^m u.
\end{equation}
The first term is $E_0$-valued, because $A_h$ is a scalar convolution operator acting on $u$; the second is $E$-valued.

The kernels of $A_h$ and $C_h$ are, respectively,
\[
h^{-j-1}k_a(t/h),
\qquad
h^{m-j-1}k_c(t/h).
\]
Apply Lemma~\ref{lem:kernel-max} to
\[
f_0(t):=\|u(t)\|_{E_0},
\qquad
f_m(t):=\|D_t^m u(t)\|_E.
\]
For almost every $t$,
\begin{equation}\label{eq:Ah-Ch-pointwise}
\|A_hu(t)\|_{E_0}
\le Ch^{-j}Mf_0(t),
\qquad
\|C_hD_t^m u(t)\|_E
\le Ch^{m-j}Mf_m(t).
\end{equation}

Now use \eqref{eq:rational-splitting} as an admissible decomposition in the $K$-functional.  At the specific Peetre parameter
\[
r=h^{-m},
\]
we obtain
\[
\begin{aligned}
K(r,D_t^j u(t);E_0,E)
&\le \|A_hu(t)\|_{E_0}
+r\|C_hD_t^m u(t)\|_E\\
&\le Ch^{-j}Mf_0(t)
+h^{-m}Ch^{m-j}Mf_m(t)\\
&\le Ch^{-j}\bigl(Mf_0(t)+Mf_m(t)\bigr).
\end{aligned}
\]
Since $r=h^{-m}$ runs through all positive numbers as $h$ runs through $(0,\infty)$ and
\[
r^{-\theta}=h^{m\theta}=h^j,
\]
we may multiply the last inequality by $r^{-\theta}$ and take the supremum over $r>0$.  This yields the pointwise interpolation estimate
\begin{equation}\label{eq:pointwise-weak-interpolation}
\|D_t^j u(t)\|_{X_{j/m}}
\le C\bigl(Mf_0(t)+Mf_m(t)\bigr).
\end{equation}
Finally, the scalar Hardy--Littlewood maximal operator is bounded on $\mathcal M^{p,\lambda}(\mathbb R)$ for $1<p<\infty$ and $0\le\lambda<1$.  Taking the Morrey norm in \eqref{eq:pointwise-weak-interpolation} gives
\[
\|D_t^j u\|_{\mathcal M^{p,\lambda}(X_{j/m})}
\le C\bigl(
\|u\|_{\mathcal M^{p,\lambda}(E_0)}
+\|D_t^m u\|_{\mathcal M^{p,\lambda}(E)}
\bigr),
\]
which proves the theorem.

<!-- proof-3: body_md -->
Put
\[
a=\frac{1-\lambda}{p},
\qquad
A:=\|u\|_{\mathcal M^{p,\lambda}(E_0)},
\qquad
B:=\|D_t^m u\|_{\mathcal M^{p,\lambda}(E)}.
\]
For $s>0$ define $u_s(t)=u(st)$.  A change of variables on each interval in the Morrey supremum gives
\[
\|u_s\|_{\mathcal M^{p,\lambda}(E_0)}=s^{-a}A.
\]
Since $D_t^m u_s(t)=s^m(D_t^m u)(st)$ and $D_t^j u_s(t)=s^j(D_t^j u)(st)$,

<!-- proposition-1: body_md -->
\label{prop:weak-index-sharp}
Let $m\ge2$ and $0<j<m$.  For every $1<p<\infty$, $0\le\lambda<1$, and every finite $1\le q<\infty$, there exist Banach spaces $E_0\hookrightarrow E$ and a function
\[
u\in\mathbb D_M^{m,p,\lambda}(\mathbb R;E_0,E)
\]
such that, on a nonempty time interval,
\[
D_t^j u(t)\in(E_0,E)_{j/m,\infty}
\qquad\text{but}\qquad
D_t^j u(t)\notin(E_0,E)_{j/m,q}.
\]
Consequently the target $(E_0,E)_{j/m,\infty}$ in Theorem~\ref{thm:whole-line} cannot, in general, be replaced by any finite fine-index space at the same interpolation level.
<!-- end proposition-1 -->
<!-- proof-5: body_md -->
It is enough to work over the complex field.  Let
\[
E=c_0,
\qquad
E_0=\left\{x=(x_n)\in c_0:\ (2^{mn}x_n)_{n\ge1}\in c_0\right\},
\]
with the natural norms.  Then $E_0\hookrightarrow E$.  For this diagonal couple one has
\begin{equation}\label{eq:diag-K}
K(r,x;E_0,E)\asymp
\sup_{n\ge1}\min\{2^{mn},r\}|x_n|.
\end{equation}
To see this, write $x=y+z$ with $y\in E_0$ and $z\in E$.  For every coordinate $n$,
\[
\min\{2^{mn},r\}|x_n|
\le 2^{mn}|y_n|+r|z_n|
\le \|y\|_{E_0}+r\|z\|_E.
\]
Taking the supremum in $n$ and then the infimum over all decompositions gives the lower estimate in \eqref{eq:diag-K}.  Conversely, set
\[
y_n=x_n\mathbf1_{\{2^{mn}\le r\}},
\qquad
z_n=x_n\mathbf1_{\{2^{mn}>r\}}.
\]
Then $y\in E_0$, $z\in E$, and
\[
\|y\|_{E_0}+r\|z\|_E
\le 2\sup_n\min\{2^{mn},r\}|x_n|.
\]
This proves \eqref{eq:diag-K} with constants independent of $r$ and $x$.

Choose a decreasing sequence $\varepsilon_n\downarrow0$ such that
\[
\sum_{n\ge2}\varepsilon_n^q=\infty
\qquad\text{for every finite }q;
\]
for instance $\varepsilon_n=1/\log(n+1)$.  Let $\chi\in C_c^\infty(\mathbb R)$ satisfy $\chi=1$ on $(0,1)$ and set
\[
u(t)=\bigl(\varepsilon_n2^{-mn}\chi(t)e^{i2^nt}\bigr)_{n\ge1}.
\]
For each $t$, the weighted coordinates defining the $E_0$ norm tend to zero, and the $E_0$-norm of $u(t)$ is uniformly bounded.  Moreover the map is strongly measurable (indeed continuous) because the $c_0$ tails are uniformly small.  Leibniz' rule gives
\[
D_t^m\bigl(\varepsilon_n2^{-mn}\chi(t)e^{i2^nt}\bigr)
=
\varepsilon_n\sum_{k=0}^m c_{m,k}2^{-kn}\chi^{(k)}(t)e^{i2^nt},
\]
so $D_t^m u$ is a bounded, compactly supported $c_0$-valued function.  Hence $u\in\mathbb D_M^{m,p,\lambda}(\mathbb R;E_0,E)$ for every stated $p,\lambda$.

On $(0,1)$,
\[
D_t^j u(t)=i^j\bigl(\varepsilon_n2^{-n(m-j)}e^{i2^nt}\bigr)_{n\ge1}=:y(t).
\]
Put $\theta=j/m$.  Formula~\eqref{eq:diag-K} implies
\[
K(r,y(t);E_0,E)\asymp
\sup_{n\ge1}\varepsilon_n\min\{2^{mn},r\}2^{-n(m-j)}.
\]
The elementary inequality
\[
\min\{A,r\}A^{-(1-\theta)}\le r^\theta
\]
shows that $\sup_{r>0}r^{-\theta}K(r,y(t))<\infty$.  On the other hand, if
\[
2^{mn}\le r\le 2^{mn+m/2},
\]
then the $n$th coordinate gives
\[
r^{-\theta}K(r,y(t))\ge c_{m,j}\varepsilon_n.
\]
These logarithmic intervals are disjoint and have fixed positive $dr/r$ length.  Consequently, for every finite $q$,
\[
\int_1^\infty\bigl[r^{-\theta}K(r,y(t))\bigr]^q\frac{dr}{r}
\ge c\sum_n\varepsilon_n^q=\infty.
\]
Thus $y(t)\in(E_0,E)_{\theta,\infty}$ and $y(t)\notin(E_0,E)_{\theta,q}$ for every finite $q$ and every $t\in(0,1)$.

<!-- lemma-2: body_md -->
\label{lem:extension}
Let $E_0\hookrightarrow E$, $m\ge1$, $1<p<\infty$, $0\le\lambda<1$, and $T>0$.  There is a bounded linear extension
\[
\mathscr E_T:\mathbb D_M^{m,p,\lambda}(0,T;E_0,E)
\longrightarrow
\mathbb D_M^{m,p,\lambda}(\mathbb R;E_0,E)
\]
such that $(\mathscr E_Tu)|_{(0,T)}=u$.  Its norm depends only on $m,p,\lambda,T$ and the embedding $E_0\hookrightarrow E$.
<!-- end lemma-2 -->
<!-- proof-6: body_md -->
We separate the construction into reflection and cutoff steps.

First, because $(0,T)$ itself is one of the intervals allowed in the Morrey norm,
\[
\|u\|_{L^p(0,T;E_0)}
\le T^{\lambda/p}\|u\|_{\mathcal M^{p,\lambda}(0,T;E_0)},
\]
and similarly
\[
\|D_t^m u\|_{L^p(0,T;E)}
\le T^{\lambda/p}\|D_t^m u\|_{\mathcal M^{p,\lambda}(0,T;E)}.
\]
Since $E_0\hookrightarrow E$, it follows that $u,D_t^m u\in L^p(0,T;E)$.  The one-dimensional distributional characterization of Sobolev spaces therefore yields
\[
u\in W^{m,p}(0,T;E).
\]
In particular, $D_t^k u$ has the standard endpoint traces in $E$ for $0\le k<m$.

Choose coefficients $c_1,\ldots,c_{m+1}$ solving the finite Vandermonde system
\[
\sum_{\ell=1}^{m+1}c_\ell(-\ell)^k=1,
\qquad k=0,\ldots,m-1.
\]
For $-T/(m+1)<t<0$ define
\[
(R_0u)(t)=\sum_{\ell=1}^{m+1}c_\ell u(-\ell t).
\]
Differentiating $k<m$ times and taking $t\uparrow0$ gives
\[
D_t^k(R_0u)(0^-)
=\sum_{\ell=1}^{m+1}c_\ell(-\ell)^kD_t^k u(0^+)
=D_t^k u(0^+),
\]
so no distributional boundary masses are created when the pieces are glued.  Reflect in the same way across $T$ to obtain a function $\widetilde u$ on a fixed larger interval $I_T\supset[0,T]$.

We record the Morrey scaling used for the reflected pieces.  If $g_\ell(t)=g(\ell t)$, then for every interval $J$,
\[
|J|^{-\lambda/p}\|g_\ell\|_{L^p(J)}
=\ell^{(\lambda-1)/p}
|\ell J|^{-\lambda/p}\|g\|_{L^p(\ell J)}.
\]
Thus dilation by a fixed integer $\ell$ changes the Morrey norm only by the fixed factor $\ell^{(\lambda-1)/p}$.  For the highest derivative there is in addition the factor $\ell^m$.  Since only the finitely many integers $1,\ldots,m+1$ occur, the reflection operator satisfies
\[
\|\widetilde u\|_{\mathcal M^{p,\lambda}(I_T;E_0)}
+\|D_t^m\widetilde u\|_{\mathcal M^{p,\lambda}(I_T;E)}
\le C\|u\|_{\mathbb D_M^{m,p,\lambda}(0,T;E_0,E)}.
\]

It remains to turn the finite-interval extension into a whole-line extension.  Let $\eta\in C_c^\infty(I_T)$ be equal to one on $[0,T]$ and put
\[
\mathscr E_Tu:=\eta\widetilde u,
\]
extended by zero outside $I_T$.  The $E_0$-valued Morrey norm of $\eta\widetilde u$ is immediately controlled by that of $\widetilde u$.  For the highest derivative, Leibniz' formula gives
\[
D_t^m(\eta\widetilde u)
=\eta D_t^m\widetilde u
+\sum_{k=0}^{m-1}\binom{m}{k}
D_t^{m-k}\eta\,D_t^k\widetilde u.
\]
The first term is already controlled in $\mathcal M^{p,\lambda}(E)$.  For the remaining terms we use the fixed-interval Sobolev estimate
\[
\sum_{k=0}^{m-1}
\|D_t^k\widetilde u\|_{L^\infty(I_T;E)}
\le C_{m,p,T}
\bigl(
\|\widetilde u\|_{L^p(I_T;E)}
+\|D_t^m\widetilde u\|_{L^p(I_T;E)}
\bigr).
\]
The right-hand side is bounded by the graph-Morrey norm established above.  Each cutoff term is therefore a bounded $E$-valued function supported in a fixed compact set.  Such a function belongs to $\mathcal M^{p,\lambda}(\mathbb R;E)$, with norm bounded by a constant depending only on the size of that fixed support.  Hence
\[
\|\mathscr E_Tu\|_{\mathbb D_M^{m,p,\lambda}(\mathbb R;E_0,E)}
\le C\|u\|_{\mathbb D_M^{m,p,\lambda}(0,T;E_0,E)}.
\]
Linearity follows from the construction, and $\mathscr E_Tu=u$ on $(0,T)$ because $\eta\equiv1$ there.

Set $v=D_t^j u$.  We first verify that $v$ lies in the first-order Morrey evolution class to which the exact trace theorem applies.

If $j=0$, then
\[
v=u\in\mathcal M^{p,\lambda}(0,T;X_0)

by definition.  If $j>0$, Theorem~\ref{thm:finite-interval} gives
\[
v\in\mathcal M^{p,\lambda}(0,T;X_{j/m}).
\]
For the derivative of $v$, if $j+1<m$ we apply Theorem~\ref{thm:finite-interval} with $j+1$ and obtain
\[
v'=D_t^{j+1}u
\in\mathcal M^{p,\lambda}(0,T;X_{(j+1)/m}).
\]
When $j=m-1$, this last statement is simply the defining assumption
\[
v'=D_t^m u\in\mathcal M^{p,\lambda}(0,T;E)
=\mathcal M^{p,\lambda}(0,T;X_1).
\]
Thus in all cases
\[
v\in\mathcal M^{p,\lambda}(0,T;Y_1),
\qquad
v'\in\mathcal M^{p,\lambda}(0,T;Y_0),
\]
with
\[
Y_1:=X_{j/m},
\qquad
Y_0:=X_{(j+1)/m},
\qquad
Y_1\hookrightarrow Y_0.
\]

The exact first-order Morrey trace theorem from \cite{ShahmurovTrace2026} states that for
\[
a=\frac{1-\lambda}{p},
\qquad
\vartheta=1-a,
\]
such a function satisfies
\[
v\in L^\infty\bigl(0,T;(Y_0,Y_1)_{\vartheta,\infty}\bigr)
\cap C^{0,\vartheta}([0,T];Y_0),
\]
with the corresponding norm bounded by
\[
C\bigl(
\|v\|_{\mathcal M^{p,\lambda}(Y_1)}
+\|v'\|_{\mathcal M^{p,\lambda}(Y_0)}
\bigr).
\]
The preceding derivative estimates bound this quantity by $C\|u\|_{\mathbb D_M^{m,p,\lambda}}$.

It remains only to identify the trace space.  By real reiteration,
\[
(Y_0,Y_1)_{\vartheta,\infty}
=(X_{(j+1)/m},X_{j/m})_{\vartheta,\infty}
=(E_0,E)_{\tau_j,\infty},
\]
where the interpolation coordinate is
\[
(1-\vartheta)\frac{j+1}{m}
+\vartheta\frac jm
=a\frac{j+1}{m}+(1-a)\frac jm
=\frac{j+a}{m}=\tau_j.
\]
Hence the $L^\infty$ and endpoint-trace conclusions take values in $X_{\tau_j}$, while the H\"older estimate takes values in $Y_0=X_{(j+1)/m}$.  This is exactly the asserted statement.

<!-- theorem-4: body_md -->
\label{thm:exact-derivative-trace}
Let $0<\lambda<1$ and $0\le j<m$.  Then
\[
\operatorname{Tr}_0 D_t^j\mathbb D_M^{m,p,\lambda}(0,T;E_0,E)
=(E_0,E)_{\tau_j,\infty}=X_{\tau_j}.
\]
More precisely, the trace map $u\mapsto D_t^j u(0)$ is bounded into $X_{\tau_j}$, and for every $x\in X_{\tau_j}$ there exists $u\in\mathbb D_M^{m,p,\lambda}(0,T;E_0,E)$ such that
\[
D_t^j u(0)=x,
\qquad
\|u\|_{\mathbb D_M^{m,p,\lambda}}
\le C\|x\|_{X_{\tau_j}}.
\]
Thus the extension may be chosen as a bounded assignment; linearity is not required.

<!-- proof-9: body_md -->
The boundedness of the trace map is already contained in Theorem~\ref{thm:derivative-traces}; only surjectivity requires construction.

Fix $x\in X_{\tau_j}$ and put
\[
a=\frac{1-\lambda}{p},
\qquad
\tau=\tau_j=\frac{j+a}{m},
\qquad
M:=\|x\|_{(E_0,E)_{\tau,\infty}}.
\]
Because $0<a<1$ and $0\le j<m$, we have $0<\tau<1$.  Choose $T_0<T/2$ and dyadic scales
\[
h_n=T_0 2^{-n},\qquad n\ge0.
\]
For the Peetre parameter $r_n=h_n^{-m}$, the definition of the $K$-functional gives a decomposition
\[
x=x_n+e_n,
\qquad
x_n\in E_0,
\quad e_n\in E,
\]
which may be chosen so that
\[
\|x_n\|_{E_0}+h_n^{-m}\|e_n\|_E
\le 2K(h_n^{-m},x).
\]
Since $x\in(E_0,E)_{\tau,\infty}$,
\[
K(h_n^{-m},x)
\le M(h_n^{-m})^\tau
=Mh_n^{-m\tau}.
\]
Using $m\tau=j+a$ we therefore obtain
\begin{equation}\label{eq:trace-approximants}
\|x_n\|_{E_0}\le 2M h_n^{-j-a},
\qquad
\|e_n\|_E\le 2M h_n^{m-j-a}.
\end{equation}
The second exponent is positive, because $m-j-a\ge1-a>0$.

We now turn these discrete approximants into a smooth function of time.  Choose a nonnegative $\psi\in C_c^\infty((1/4,4))$, positive on $[1/2,2]$, such that the dyadic family
\[
\psi_n(t):=
\frac{\psi(t/h_n)}{\sum_{k\ge0}\psi(t/h_k)}
\]
forms a partition of unity for $0<t<T_1$ with some $T_1<T_0$.  After a harmless smooth completion on $[T_1,T_0]$, we may assume that
\[
\sum_n\psi_n(t)=1\quad(0<t<T_1),
\]
that at each $t$ only a fixed number of indices occur, and that
\[
\operatorname{supp}\psi_n\subset[c h_n,C h_n],
\qquad
|D_t^k\psi_n(t)|\le C_k h_n^{-k}.
\]
Finally multiply the completed family by a cutoff supported in $(0,T_0)$ and equal to one on $(0,T_1)$.  Define
\[
v(t):=\sum_n\psi_n(t)x_n.
\]
The sum is locally finite for $t>0$, hence $v$ is smooth there with values in $E_0$.

If $0<t<T_1$, every active index satisfies $h_n\simeq t$.  The finite overlap and \eqref{eq:trace-approximants} yield
\begin{equation}\label{eq:v-strong-bound}
\|v(t)\|_{E_0}
\le CM t^{-j-a}.
\end{equation}
Moreover, because $\sum_n\psi_n(t)=1$,
\[
x-v(t)=\sum_n\psi_n(t)e_n,
\]
so
\begin{equation}\label{eq:v-approaches-x}
\|x-v(t)\|_E
\le CM t^{m-j-a}.
\end{equation}
Since $m-j-a>0$, this proves $v(t)\to x$ in $E$ as $t\downarrow0$.

Next let $k=m-j$.  Differentiating the identity $\sum_n\psi_n=1$ gives
\[
\sum_nD_t^k\psi_n=0
\qquad (0<t<T_1).
\]
Hence
\[
D_t^k v
=\sum_nD_t^k\psi_n x_n
=-\sum_nD_t^k\psi_n e_n.
\]
For an active index $h_n\simeq t$, and therefore
\[
\|D_t^k\psi_n(t)e_n\|_E
\le C h_n^{-k}Mh_n^{m-j-a}
=CMh_n^{-a}
\le CMt^{-a}.
\]
Finite overlap gives
\begin{equation}\label{eq:v-high-derivative}
\|D_t^{m-j}v(t)\|_E\le CMt^{-a},
\qquad 0<t<T_1.
\end{equation}
On $[T_1,T]$ only finitely many smooth cutoff pieces remain, so both $v$ and all derivatives occurring above are uniformly bounded there.

We now recover a function whose $j$-th derivative is $v$.  If $j=0$, simply put $u=v$.  If $j\ge1$, define
\[
u(t)=\frac{(-1)^j}{(j-1)!}
\int_t^T(s-t)^{j-1}v(s)\,ds.
\]
Because $v$ is supported in $(0,T_0)$, this Bochner integral is well defined for every $t>0$.  Repeated differentiation gives
\[
D_t^j u=v,
\qquad
D_t^m u=D_t^{m-j}v.
\]
For the strong-space norm, \eqref{eq:v-strong-bound} implies, for $0<t<T_1$,
\[
\|u(t)\|_{E_0}
\le CM\int_t^{T_0}(s-t)^{j-1}s^{-j-a}\,ds.
\]
With $s=t(1+y)$,
\[
\int_t^{T_0}(s-t)^{j-1}s^{-j-a}\,ds
\le t^{-a}\int_0^\infty y^{j-1}(1+y)^{-j-a}\,dy
\le C_{j,a}t^{-a},
\]
where the last integral converges because $a>0$.  Thus, also in the case $j=0$ by \eqref{eq:v-strong-bound},
\[
\|u(t)\|_{E_0}\le CM(1+t^{-a}),
\qquad
\|D_t^m u(t)\|_E\le CM(1+t^{-a}).
\]

It remains to check the Morrey integrability of the borderline power.  Since
\[
ap=1-\lambda,
\]
for any interval $J\subset(0,T)$ of length $\ell$ the monotonicity of $t^{-(1-\lambda)}$ gives
\[
\int_J t^{-ap}\,dt
\le \int_0^\ell t^{-(1-\lambda)}\,dt
=\frac{\ell^\lambda}{\lambda}.
\]
Hence
\[
\|t^{-a}\|_{\mathcal M^{p,\lambda}(0,T)}\le \lambda^{-1/p}.
\]
The constant function also belongs to the same finite-interval Morrey space.  Therefore
\[
\|u\|_{\mathbb D_M^{m,p,\lambda}(0,T;E_0,E)}
\le C M.
\]

Finally, $D_t^j u(t)=v(t)\to x$ in $E$ by \eqref{eq:v-approaches-x}.  On the other hand, Theorem~\ref{thm:derivative-traces} gives a continuous representative of $v$ at $t=0$ in the weaker space $X_{(j+1)/m}$ and a trace $v(0)\in X_{\tau_j}$.  Since $X_{(j+1)/m}\hookrightarrow E$, the same representative converges to $v(0)$ in $E$.  Uniqueness of limits in $E$ therefore gives $v(0)=x$.  Thus $D_t^j u(0)=x$, completing the surjectivity proof.

Let $0<\lambda<1$, $0\le j<m$, $1\le q<\infty$, and assume

\qquad
s_0=\frac jm,
\qquad
s_1=\tau_j=\frac{j+a}{m}.
\]
Theorem~\ref{thm:finite-interval} and Theorem~\ref{thm:derivative-traces} give, respectively,

\end{equation}

=\gamma s_0+(1-\gamma)s_1.
\]

\begin{equation}\label{eq:strict-scale-pointwise}
\|x\|_{(E_0,E)_{\sigma,r}}
\le C
\|x\|_{X_{s_0}}^\gamma
\|x\|_{X_{s_1}}^{1-\gamma}.
\end{equation}

\[
A_0=\|x\|_{X_{s_0}},
\qquad
A_1=\|x\|_{X_{s_1}},
\]
then
\[
K(\rho,x)\le
\min\{A_0\rho^{s_0},A_1\rho^{s_1}\}.
\]
Split the defining $d\rho/\rho$ integral for the $(\sigma,r)$ norm at the unique scale where the two powers are equal.  The two geometric integrals give \eqref{eq:strict-scale-pointwise}; for $r=\infty$ the same estimate follows by taking the supremum.

\[
\|v(t)\|_{(E_0,E)_{\sigma,r}}
\le C B^{1-\gamma}A(t)^\gamma.
\]

=\frac{bq}{a}
=p\frac{1-\mu}{1-\lambda}
\le p.
\]

\[
\begin{aligned}
\left(\int_J A(t)^{\gamma q}\,dt\right)^{1/q}
&\le
\left(\int_J A(t)^p\,dt\right)^{\gamma/p}
|J|^{1/q-\gamma/p}\\
&\le
\|A\|_{\mathcal M^{p,\lambda}}^\gamma
|J|^{\lambda\gamma/p+1/q-\gamma/p}.
\end{aligned}
\]
The exponent of $|J|$ is exactly $\mu/q$, because
\[
\frac1q-\gamma\frac{1-\lambda}{p}
=\frac1q-b
=\frac\mu q.
\]
Consequently
\[
|J|^{-\mu/q}
\|v\|_{L^q(J;(E_0,E)_{\sigma,r})}
\le C
B^{1-\gamma}
\|A\|_{\mathcal M^{p,\lambda}}^\gamma.
\]
Taking the supremum over $J$ and using \eqref{eq:pq-two-endpoints} proves the asserted estimate.

If $b=a$, then $\gamma=1$ and $\sigma=s_0$.  In that case no finite fine-index improvement at the critical level is available in general, by Proposition~\ref{prop:weak-index-sharp}; however Theorem~\ref{thm:finite-interval} and Corollary~\ref{cor:weaker-morrey} apply because $b=a$ implies $q\le p$ and the scalar Morrey exponent shift is zero. <!-- proposition-2: body_md -->
\label{prop:mu-threshold}
Fix $0<\lambda<1$, $0\le j<m$, $1\le q<\infty$, $0\le\mu<1$, and $1\le r\le\infty$.  Put
\[
a=\frac{1-\lambda}{p},
\qquad
b=\frac{1-\mu}{q},
\qquad
\sigma=\frac{j+a-b}{m},
\]
and assume the strict-gap condition
\[
0<b<a.
\]
Then the following are equivalent:
\begin{enumerate}[label=\textup{(\roman*)}]
\item for every continuously embedded Banach couple $E_0\hookrightarrow E$, the map
\[
D_t^j:\mathbb D_M^{m,p,\lambda}(0,T;E_0,E)
\longrightarrow
\mathcal M^{q,\mu}\bigl(0,T;(E_0,E)_{\sigma,r}\bigr)
\]
is bounded;
\item $\mu\ge\lambda$.
\end{enumerate}
Moreover, if $\mu<\lambda$, failure already occurs for one fixed densely and compactly embedded Hilbert couple.
<!-- end proposition-2 -->
<!-- proof-12: body_md -->
The implication \textup{(ii)}$\Rightarrow$\textup{(i)} is Corollary~\ref{cor:pq-gain}.  We prove necessity.

An affine change of the time variable only changes the constants in the norms, so it is enough to work on $(0,1)$.  Let
\[
E=\ell^2,
\qquad
E_0=\left\{x=(x_k)_{k\ge1}:\sum_{k\ge1}2^{2mk}|x_k|^2<\infty\right\},
\]
with
\[
\|x\|_{E_0}^2=\sum_{k\ge1}2^{2mk}|x_k|^2.
\]
Then $E_0\mathrel{\hookrightarrow\!\hookrightarrow} E$ densely and compactly.  Denote by $e_n$ the $n$th coordinate vector and set
\[
h_n=2^{-n}.
\]
Thus
\[
\|e_n\|_E=1,
\qquad
\|e_n\|_{E_0}=h_n^{-m}.
\]

Choose a nonzero $\phi\in C_c^\infty((-1/10,1/10))$.  Let
\[
N_n=\lfloor c_0h_n^{-\lambda}\rfloor
\]
with $c_0>0$ small and choose points $t_{1,n},\dots,t_{N_n,n}$ in a fixed interval $J_*\Subset(0,1)$, equally spaced with separation
\[
d_n\asymp N_n^{-1}\asymp h_n^\lambda.
\]
Since $0<\lambda<1$, one has $h_n/d_n\to0$.  Hence, after discarding finitely many $n$, the supports of
\[
\phi\left(\frac{t-t_{k,n}}{h_n}\right),
\qquad 1\le k\le N_n,
\]
are pairwise disjoint and all lie in $J_*$.  Define
\begin{equation}\label{eq:threshold-pulse-def}
u_n(t)=h_n^{m-a}
\sum_{k=1}^{N_n}
\phi\left(\frac{t-t_{k,n}}{h_n}\right)e_n.
\end{equation}

We first show that the graph norms of $u_n$ remain bounded.  Since the pulses are disjoint,
\[
\|u_n(t)\|_{E_0}
=h_n^{-a}
\left|\sum_{k=1}^{N_n}
\phi\left(\frac{t-t_{k,n}}{h_n}\right)\right|,
\]
and
\[
\|D_t^m u_n(t)\|_E
=h_n^{-a}
\left|\sum_{k=1}^{N_n}
\phi^{(m)}\left(\frac{t-t_{k,n}}{h_n}\right)\right|.
\]
Thus it suffices to estimate a scalar pulse train
\[
G_n(t)=h_n^{-a}\sum_{k=1}^{N_n}
\left|\psi\left(\frac{t-t_{k,n}}{h_n}\right)\right|,
\]
where $\psi$ is either $\phi$ or $\phi^{(m)}$.

Let $J\subset(0,1)$ be an interval and write $\ell=|J|$.  Recall that
\[
ap=1-\lambda.
\]
If $\ell\le h_n$, then at most one pulse meets $J$ and
\[
\ell^{-\lambda}\int_JG_n(t)^p\,dt
\le C\ell^{-\lambda}h_n^{-(1-\lambda)}\ell
=C\left(\frac{\ell}{h_n}\right)^{1-\lambda}
\le C.
\]
If $h_n<\ell<d_n$, only a bounded number of pulses meet $J$, and therefore
\[
\ell^{-\lambda}\int_JG_n(t)^p\,dt
\le C\ell^{-\lambda}h_n^{-(1-\lambda)}h_n
=C\left(\frac{h_n}{\ell}\right)^\lambda
\le C.
\]
Finally, if $\ell\ge d_n$, the separation of the centers gives
\[
\#\{k:\operatorname{supp}\phi((\cdot-t_{k,n})/h_n)\cap J\ne\varnothing\}
\le C\frac{\ell}{d_n}.
\]
Consequently,
\[
\ell^{-\lambda}\int_JG_n(t)^p\,dt
\le
C\ell^{-\lambda}h_n^{-(1-\lambda)}h_n\frac{\ell}{d_n}
\le C\ell^{1-\lambda},
\]
because $d_n\asymp h_n^\lambda$.  This is uniformly bounded for $J\subset(0,1)$.  Taking $p$th roots in the three cases yields
\begin{equation}\label{eq:threshold-source-bound}
\sup_n\|u_n\|_{\mathbb D_M^{m,p,\lambda}(0,1;E_0,E)}<\infty.
\end{equation}

It remains to compute the target size.  The coordinate projection onto $\mathbb C e_n$ is contractive on both $E_0$ and $E$, so the $K$-functional of $e_n$ is exactly the one-dimensional $K$-functional:
\[
K(\rho,e_n;E_0,E)=\min\{h_n^{-m},\rho\}.
\]
Hence, for every fixed $1\le r\le\infty$,
\begin{equation}\label{eq:threshold-coordinate-interp}
\|e_n\|_{(E_0,E)_{\sigma,r}}
\asymp_{\sigma,r} h_n^{-m(1-\sigma)}.
\end{equation}
Since
\[
m(1-\sigma)=m-j-a+b,
\]
we obtain from \eqref{eq:threshold-pulse-def} and \eqref{eq:threshold-coordinate-interp}
\[
\|D_t^ju_n(t)\|_{(E_0,E)_{\sigma,r}}
\asymp
h_n^{-b}
\left|\sum_{k=1}^{N_n}
\phi^{(j)}\left(\frac{t-t_{k,n}}{h_n}\right)\right|.
\]
Using the fixed interval $J_*$ containing all pulses and their disjointness,
\[
\begin{aligned}
\|D_t^ju_n\|_{\mathcal M^{q,\mu}(0,1;(E_0,E)_{\sigma,r})}
&\ge
|J_*|^{-\mu/q}
\|D_t^ju_n\|_{L^q(J_*;(E_0,E)_{\sigma,r})}\\
&\ge c\,h_n^{-b}(N_nh_n)^{1/q}\\
&\asymp h_n^{-b+(1-\lambda)/q}
=h_n^{(\mu-\lambda)/q}.
\end{aligned}
\]
Because $\phi^{(j)}\not\equiv0$, the constant $c$ is positive.  If $\mu<\lambda$, the last quantity tends to infinity as $n\to\infty$, contradicting boundedness in \textup{(i)} in view of \eqref{eq:threshold-source-bound}.  Thus \textup{(i)} implies $\mu\ge\lambda$.

<!-- lemma-3: body_md -->
\label{lem:cz-lift}
Let $T$ be an $H$-valued convolution operator on $\mathbb R$ which is bounded on $L^p(\mathbb R;H)$ and whose kernel satisfies
\[
\|K(t)\|_{\mathcal L(H)}\le C|t|^{-1},\qquad t\ne0.
\]
Then
\[
T:\mathcal M^{p,\lambda}(\mathbb R;H)\longrightarrow\mathcal M^{p,\lambda}(\mathbb R;H)
\]
is bounded for $1<p<\infty$, $0\le\lambda<1$.
<!-- end lemma-3 -->
<!-- proof-16: body_md -->
Let
\[
N:=\|f\|_{\mathcal M^{p,\lambda}(\mathbb R;H)}
\]
and fix a bounded interval $J$.  Write
\[
f=f_1+f_2,
\qquad
f_1=f\mathbf1_{2J},
\quad
f_2=f\mathbf1_{(2J)^c},
\]
where $2J$ is the concentric interval of twice the length.

For the local part, the assumed $L^p$ boundedness gives
\[
\|Tf_1\|_{L^p(J;H)}
\le C_p\|f\|_{L^p(2J;H)}
\le C_p(2|J|)^{\lambda/p}N.
\]
Hence
\[
|J|^{-\lambda/p}\|Tf_1\|_{L^p(J;H)}
\le C N.
\]

For the far part, if $t\in J$ and
\[
A_k:=2^{k+1}J\setminus2^kJ,
\qquad k\ge1,
\]
then $|t-s|\simeq2^k|J|$ for $s\in A_k$.  The kernel bound therefore yields
\[
\|Tf_2(t)\|_H
\le C\sum_{k\ge1}(2^k|J|)^{-1}
\int_{2^{k+1}J}\|f(s)\|_H\,ds.
\]
By H\"older's inequality and the Morrey estimate,
\[
\begin{aligned}
\int_{2^{k+1}J}\|f(s)\|_H\,ds
&\le |2^{k+1}J|^{1-1/p}
\|f\|_{L^p(2^{k+1}J;H)}\\
&\le C(2^k|J|)^{1-1/p+\lambda/p}N.
\end{aligned}
\]
Thus the $k$-th annular contribution is bounded by
\[
C2^{-k(1-\lambda)/p}|J|^{-(1-\lambda)/p}N.
\]
Since $\lambda<1$, the series converges and
\[
\|Tf_2(t)\|_H
\le C|J|^{-(1-\lambda)/p}N,
\qquad t\in J.
\]
Taking the $L^p(J)$ norm gives
\[
\|Tf_2\|_{L^p(J;H)}
\le C|J|^{\lambda/p}N.
\]
Combining the local and far estimates, multiplying by $|J|^{-\lambda/p}$, and taking the supremum over $J$ proves the lemma.

<!-- theorem-7: body_md -->
\label{thm:hilbert}
Let $m\ge2$, $0<j<m$, $1<p<\infty$, and $0\le\lambda<1$.  If
\[
u\in\mathcal M^{p,\lambda}(\mathbb R;D(B)),
\qquad
D_t^m u\in\mathcal M^{p,\lambda}(\mathbb R;H),
\]
then
\[
D_t^j u\in\mathcal M^{p,\lambda}\bigl(\mathbb R;D(B^{1-j/m})\bigr)
\]
and
\[
\|B^{1-j/m}D_t^j u\|_{\mathcal M^{p,\lambda}(H)}
\le C
\bigl(
\|Bu\|_{\mathcal M^{p,\lambda}(H)}
+\|D_t^m u\|_{\mathcal M^{p,\lambda}(H)}
\bigr).
\]
The same statement holds on finite intervals.
<!-- end theorem-7 -->
<!-- proof-17: body_md -->
It is enough to prove the whole-line assertion.  Once that is done, apply the extension lemma with the Banach couple $(D(B),H)$ and restrict back to $(0,T)$.

Set $\theta=j/m$.  We split the target multiplier by the positive operator $B^2+|\xi|^{2m}$.  In Fourier variables,
\[
\begin{aligned}
B^{1-\theta}(i\xi)^j\widehat u
&=B^{1-\theta}(i\xi)^j
\frac{B^2}{B^2+|\xi|^{2m}}\widehat u\\
&\quad+
B^{1-\theta}(i\xi)^j
\frac{|\xi|^{2m}}{B^2+|\xi|^{2m}}\widehat u.
\end{aligned}
\]
The first term can be written as $M_0(\xi)\widehat{Bu}$ with
\[
M_0(\xi)=B^{2-\theta}(i\xi)^j(B^2+|\xi|^{2m})^{-1},
\]
and the second as $M_1(\xi)\widehat{D_t^m u}$ with
\[
M_1(\xi)=i^{j-m}B^{1-\theta}\xi^{j+m}
(B^2+\xi^{2m})^{-1}.
\]
Thus
\begin{equation}\label{eq:hilbert-multiplier-split}
B^{1-\theta}D_t^j u
=T_0(Bu)+T_1(D_t^m u),
\end{equation}
where $T_i$ are the Fourier multipliers with symbols $M_i$.

We next verify uniform $L^p$ and kernel bounds.  Let $b\ge1$ be a spectral value of $B$ and put
\[
\rho=b^{1/m}.
\]
For the first multiplier, the scalar spectral symbol is
\[
\frac{b^{2-\theta}(i\xi)^j}{b^2+|\xi|^{2m}}
=
\frac{(i\xi/\rho)^j}{1+(\xi/\rho)^{2m}}
=a(\xi/\rho),
\]
because $b^{j/m}=b^\theta$.  Similarly,
\[
i^{j-m}\frac{b^{1-\theta}\xi^{j+m}}{b^2+\xi^{2m}}
=c(\xi/\rho).
\]
Hence the scalar kernels are precisely the dilates
\[
k_{0,b}(t)=\rho k_a(\rho t),
\qquad
k_{1,b}(t)=\rho k_c(\rho t).
\]
Since $k_a$ and $k_c$ decay exponentially,
\[
\rho|k_i(\rho t)|
=|t|^{-1}\,|\rho t|\,|k_i(\rho t)|
\le \frac{C}{|t|},
\qquad t\ne0,
\]
uniformly in $\rho>0$.  The spectral theorem then gives operator-valued kernels $K_i(t)$ satisfying
\[
\|K_i(t)\|_{\mathcal L(H)}
\le\sup_{b\ge1}|k_{i,b}(t)|
\le C|t|^{-1}.
\]

The same dilation formulas also give the Mikhlin bounds.  For $\ell=0,1$,
\[
\sup_{\rho>0}\sup_{\xi\ne0}
|\xi|^\ell
\left|\frac{d^\ell}{d\xi^\ell}a(\xi/\rho)\right|<\infty,
\]
and likewise for $c$.  By functional calculus this yields the corresponding operator-norm bounds for $M_0$ and $M_1$.  Since $H$ is a Hilbert space, the one-dimensional operator-valued Mikhlin theorem applies, and both $T_0$ and $T_1$ are bounded on $L^p(\mathbb R;H)$, uniformly in the spectral scale.

Lemma~\ref{lem:cz-lift} now upgrades each $T_i$ to a bounded operator on $\mathcal M^{p,\lambda}(\mathbb R;H)$.  Taking the Morrey norm in \eqref{eq:hilbert-multiplier-split} gives
\[
\|B^{1-\theta}D_t^j u\|_{\mathcal M^{p,\lambda}(H)}
\le C\bigl(
\|Bu\|_{\mathcal M^{p,\lambda}(H)}
+\|D_t^m u\|_{\mathcal M^{p,\lambda}(H)}
\bigr).
\]
Finally, $B\ge I$ implies
\[
\|x\|_H\le\|B^{1-\theta}x\|_H,
\]
so the displayed quantity controls the full graph norm of $D(B^{1-\theta})$.  This proves the theorem.

<!-- proposition-3: body_md -->
\label{prop:sharp-hilbert}
The fractional-domain exponent $1-j/m$ in Theorem~\ref{thm:hilbert} cannot be increased uniformly.
<!-- end proposition-3 -->
<!-- proof-18: body_md -->
Take $H=\ell^2$ and $Be_N=N^m e_N$.  Let $0\ne\psi\in C_c^\infty(\mathbb R)$ and set
\[
u_N(t)=\psi(Nt)e_N.
\]
If $a=(1-\lambda)/p$, time dilation gives
\[
\|Bu_N\|_{\mathcal M^{p,\lambda}(H)}
+\|D_t^m u_N\|_{\mathcal M^{p,\lambda}(H)}
\asymp N^{m-a}.
\]
For any $\varepsilon>0$,
\[
\|B^{1-j/m+\varepsilon}D_t^j u_N\|_{\mathcal M^{p,\lambda}(H)}
\asymp N^{m+m\varepsilon-a}.
\]
The ratio diverges like $N^{m\varepsilon}$.  Hence no uniform estimate with a larger fractional-domain exponent is possible.

<!-- corollary-6: body_md -->
\label{cor:hilbert-multiplicative}
Under the assumptions of Theorem~\ref{thm:hilbert} on $\mathbb R$, with
\[
\theta=\frac jm,
\]
one has
\[
\boxed{
\|B^{1-\theta}D_t^j u\|_{\mathcal M^{p,\lambda}(\mathbb R;H)}
\le C
\|Bu\|_{\mathcal M^{p,\lambda}(\mathbb R;H)}^{1-\theta}
\|D_t^m u\|_{\mathcal M^{p,\lambda}(\mathbb R;H)}^{\theta}.
}
\]
Consequently, for every $\varepsilon>0$,
\[
\|B^{1-\theta}D_t^j u\|_{\mathcal M^{p,\lambda}(\mathbb R;H)}
\le
\varepsilon\|D_t^m u\|_{\mathcal M^{p,\lambda}(\mathbb R;H)}
+C_\varepsilon\|Bu\|_{\mathcal M^{p,\lambda}(\mathbb R;H)}.
\]
<!-- end corollary-6 -->
<!-- proof-19: body_md -->
Put $a=(1-\lambda)/p$ and $u_s(t)=u(st)$.  Time dilation gives
\[
\|Bu_s\|_{\mathcal M^{p,\lambda}}=s^{-a}\|Bu\|_{\mathcal M^{p,\lambda}},
\quad
\|D_t^m u_s\|_{\mathcal M^{p,\lambda}}
=s^{m-a}\|D_t^m u\|_{\mathcal M^{p,\lambda}},
\]
and
\[
\|B^{1-\theta}D_t^j u_s\|_{\mathcal M^{p,\lambda}}
=s^{j-a}\|B^{1-\theta}D_t^j u\|_{\mathcal M^{p,\lambda}}.
\]
Apply Theorem~\ref{thm:hilbert} to $u_s$ and divide by $s^{j-a}$:
\[
\|B^{1-\theta}D_t^j u\|_{\mathcal M^{p,\lambda}}
\le C\left(
 s^{-j}\|Bu\|_{\mathcal M^{p,\lambda}}
 +s^{m-j}\|D_t^m u\|_{\mathcal M^{p,\lambda}}
\right).
\]
Optimizing at $s^m=\|Bu\|/\|D_t^m u\|$ gives the multiplicative estimate; the cases in which one endpoint norm vanishes follow by a limiting argument.  The $\varepsilon$-form is Young's inequality.

<!-- proposition-4: body_md -->
\label{prop:periodic-relaxation}
Let $B\ge I$ be positive selfadjoint on $H$, and let $v$ be $T$-periodic with
\[
v\in L^2(0,T;D(B))\cap H^2_{\rm per}(0,T;H).
\]
Then
\begin{equation}\label{eq:periodic-middle}
\|B^{1/2}v_t\|_{L^2(0,T;H)}^2
\le
\|Bv\|_{L^2(0,T;H)}
\|v_{tt}\|_{L^2(0,T;H)}.
\end{equation}
In particular, if $u_t$ is periodic, then
\begin{equation}\label{eq:periodic-acoustic}
\|B^{1/2}u_{tt}\|_{L^2}^2
\le
\|Bu_t\|_{L^2}\|u_{ttt}\|_{L^2},
\end{equation}
and for every relaxation parameter $\tau>0$,
\begin{equation}\label{eq:tau-weight}
\boxed{
\sqrt{\tau}\,\|B^{1/2}u_{tt}\|_{L^2}
\le
\|Bu_t\|_{L^2}^{1/2}
\|\tau u_{ttt}\|_{L^2}^{1/2}.
}
\end{equation}
<!-- end proposition-4 -->
<!-- proof-20: body_md -->
Write the time Fourier series $v(t)=\sum_{n\in\mathbb Z}e^{i\omega_n t}v_n$, where $\omega_n=2\pi n/T$.  Parseval's identity and positivity of $B$ give
\[
\begin{aligned}
\|B^{1/2}v_t\|_{L^2}^2
&=T\sum_n \omega_n^2\langle Bv_n,v_n\rangle_H\\
&\le
\left(T\sum_n\|Bv_n\|_H^2\right)^{1/2}
\left(T\sum_n\omega_n^4\|v_n\|_H^2\right)^{1/2}\\
&=\|Bv\|_{L^2}\|v_{tt}\|_{L^2}.
\end{aligned}
\]
The remaining statements follow by taking $v=u_t$ and multiplying by $\tau^{1/2}$.
