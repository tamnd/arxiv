<!-- assumption-1: body_md -->
\label{ass:one-sided-margin}
There are constants $C_m>0$, $beta>0$, and $t_0>0$ such that
\begin{equation}
\label{eq:one-sided-margin}
\mathbb P\{0<M_q(w^\star)\le t\}
\le C_m t^beta,
\qquad 0<t\le t_0.
\end{equation}

<!-- theorem-1: body_md -->
\label{thm:buffered-main}
Assume \cref{ass:one-sided-margin}, let $\varepsilon_N$ be given by \eqref{eq:epsilon-N}, and suppose $2\varepsilon_N\le t_0$.  Let $\widehat w_{\varepsilon_N}$ be any minimizer in \eqref{eq:buffered-erm}.  Then, with probability at least $1-\delta$ over both sources of randomness,
\begin{equation}
\label{eq:buffered-bound-exact}
R(\widehat w_{\varepsilon_N})-R(w^\star)
\le
2\mathfrak v_Q(\delta)
+C_m(2\varepsilon_N)^beta.
\end{equation}
Substituting \eqref{eq:epsilon-N} and \eqref{eq:vq} gives the explicit rate stated in \eqref{eq:intro-main}.
<!-- end theorem-1 -->
<!-- proof-1: body_md -->
For $t\ge0$, define the clean population and empirical threshold risks
\[
R_t(w):=\mathbb P\{M_q(w)\le t\},
\qquad
\widetilde R_t(w):=
\frac1Q\sum_{j=1}^Q\mathbf 1\{M_{q_j}(w)\le t\}.
\]
In particular, $\widetilde R_0$ is the ordinary empirical risk that would be available if the clean answer distributions were known.  It contains the randomness from sampling $Q$ questions, but not the finite-$N$ generation noise present in $\widehat R_{\varepsilon_N}$.  Write $\mathfrak v_Q=\mathfrak v_Q(\delta)$ for brevity.
The result follows from a chain of six inequalities.  By \cref{lem:clean-risk-deviation}, with probability at least $1-\delta/4$ over the sampled questions,
$|\widetilde R_t(w)-R_t(w)|\le\mathfrak v_Q$ simultaneously for all $w\inDelta_K$ and $t\in\{0,2\varepsilon_N\}$.  In particular,
\[
R_0(\widehat w_{\varepsilon_N}) \stackrel{(a)}{\le} \widetilde R_0(\widehat w_{\varepsilon_N})+\mathfrak v_Q, \quad \widetilde R_{2\varepsilon_N}(w^\star) \stackrel{(e)}{\le} R_{2\varepsilon_N}(w^\star)+\mathfrak v_Q.
\]

Separately, \cref{lem:margin-accuracy} shows, conditionally on the sampled questions, that event \eqref{eq:intro-uniform-event} holds with probability at least $1-\delta/4$ over the generations.  On this event, the buffered sandwich \eqref{eq:intro-sandwich} holds for every $w\inDelta_K$ and $j\in[Q]$.  Summing over $j$ and dividing by $Q$ gives
\[
\widetilde R_0(\widehat w_{\varepsilon_N}) \stackrel{(b)}{\le} \widehat R_{\varepsilon_N}(\widehat w_{\varepsilon_N}) \stackrel{(c)}{\le}  \widehat R_{\varepsilon_N}(w^\star) \stackrel{(d)}{\le} \widetilde R_{2\varepsilon_N}(w^\star).
\]
The middle inequality (c) follows from the minimality of $\widehat w_{\varepsilon_N}$ for the buffered empirical risk.  Finally, \cref{ass:one-sided-margin} gives
\[
R_{2\varepsilon_N}(w^\star)-R_0(w^\star)
=
\mathbb P\{0<M_q(w^\star)\le2\varepsilon_N\}
\stackrel{(f)}{\le} C_m(2\varepsilon_N)^beta.
\]
By a union bound, the two preceding events hold simultaneously with probability at least $1-\delta$.  Combining (a)--(f) in alphabetic order and using $R=R_0$ proves \eqref{eq:buffered-bound-exact}.
<!-- end proof-1 -->
<!-- remark-1: body_md -->
For fixed $\varepsilon$, the buffered objective is piecewise constant on the arrangement of the empirical hyperplanes
\[
\widehat p_w(a^\star(q_j)\mid q_j)
-\widehat p_w(a\mid q_j)
=\varepsilon,
\qquad a\ne a^\star(q_j).
\]
It can be optimized by the same mixed-integer formulation used for weighted Best-of-$infinity$, with the right-hand side shifted by $\varepsilon$.  If a numerical solver returns $\widetilde w_\varepsilon$ satisfying
\[
\widehat R_\varepsilon(\widetilde w_\varepsilon)
\le
\inf_{w\inDelta_K}\widehat R_\varepsilon(w)+\eta,
\]
the same proof adds only $\eta$ to \eqref{eq:buffered-bound-exact}.  Thus a computational tolerance no larger than the statistical bound in \eqref{eq:intro-main} does not change the rate.

<!-- lemma-1: body_md -->
\label{lem:side}
Define
\begin{equation}
\label{eq:theta}
theta(G)
:=
\int_{-1}^1\operatorname{sgn}(z)\,\,\mathrm d G(z)
=
G((0,1])-G([-1,0)).
\end{equation}
Every $t>1/2$ has risk $G([-1,0])$, every $t<1/2$ has risk $G([0,1])$, and $t=1/2$ has risk one.  Hence the optimal side is the sign of $theta(G)$ when $theta(G)\ne0$; when $theta(G)=0$, the two sides have equal risk.  Choosing the wrong side incurs excess risk $|theta(G)|$.
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
The stated risks for the three decisions $t>1/2$ (the $+$ decision), $t<1/2$ (the $-$ decision), and $t=1/2$ follow from~\eqref{eq:scalar-margin}.  The decision $t=1/2$ is weakly dominated.  Moreover,
\[
R(-)-R(+)
=G([0,1])-G([-1,0])
=G((0,1])-G([-1,0))
=theta(G),
\]
where the mass at zero cancels.  Thus, when $theta(G)>0$, the $+$ decision is optimal and choosing $-$ incurs excess risk $theta(G)=|theta(G)|$.  The case $theta(G)<0$ is similar, while the two decisions have equal risk when $theta(G)=0$.
<!-- end proof-2 -->
<!-- lemma-2: body_md -->
\label{lem:moment-equivalence}
For probability measures $G,H$ on $[-1,1]$, the following are equivalent:
\begin{enumerate}[label=(\alph*)]
\item $K_nG=K_nH$;
\item $\int P\,\,\mathrm d G=\int P\,\,\mathrm d H$ for every algebraic polynomial $P$ of degree at most $n$;
\item $\int z^j\,\,\mathrm d G(z)=\int z^j\,\,\mathrm d H(z)$ for $j=0,\dots,n$.
\end{enumerate}
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
Equality of $K_nG$ and $K_nH$ is equivalent to equality of the integrals of $b_{k,n}((1+z)/2)$ for $k=0,\dots,n$.  The Bernstein polynomials form a basis for the polynomials of degree at most $n$ in $p$, and composition with the invertible affine map $p=(1+z)/2$ gives a basis for the polynomials of degree at most $n$ in $z$. Monomials $z^j, j=0,\dots,n$ form another basis.

<!-- lemma-7: body_md -->
\label{lem:weak-sign-patterns}
Fix affine functions $g_1,\dots,g_H:\mathbb R^p\to\mathbb R$.  Their weak sign patterns
are the vectors
\[
\bigl(\operatorname{sgn} g_1(x),\dots,\operatorname{sgn} g_H(x)\bigr)
\in\{-1,0,+1\}^H,
\qquad x\in\mathbb R^p.
\]
The number of distinct weak sign patterns is at most
\begin{equation}
\label{eq:faces-bound}
B(H,p):=\sum_{r=0}^{p}2^r\binom Hr.
\end{equation}
If $H\ge p\ge1$, then $B(H,p)\le(2eH/p)^p$.
If $H<p$, then $B(H,p)=3^H$.  In either case,
$\log B(H,p)\le Cp\log(eH)$ for a universal constant $C$.
<!-- end lemma-7 -->
<!-- proof-8: body_md -->
Constant affine functions may be discarded because they do not increase
the number of patterns.  The hyperplanes $\{g_h=0\}$ partition $\mathbb R^p$
into relatively open faces of all dimensions.  The weak sign vector is
constant on each face, so it suffices to bound the total number of faces.

Let $F(H,p)$ be the largest possible number of such faces.  We prove the
bound by induction on $H$.  After arranging the first $H-1$ hyperplanes,
add the $H$th.  The preceding hyperplanes induce an arrangement of at
most $H-1$ hyperplanes on the new hyperplane, and hence at most
$F(H-1,p-1)$ faces there.  Each induced face can split one old face into
two while itself becoming a new face, producing a net increase of at
most two.  Therefore
\[
F(H,p)\le F(H-1,p)+2F(H-1,p-1),
\]
with $F(0,p)=F(H,0)=1$.  The quantity $B(H,p)$ obeys the same recurrence
by Pascal's identity, so induction proves $F(H,p)\le B(H,p)$.  Finally,
when $H\ge p$,
\[
B(H,p)
\le\sum_{r=0}^{p}\binom{2H}{r}
\le\left(\frac{2eH}{p}\right)^p.
\]
If $H<p$, then $\binom Hr=0$ for $r>H$, and the binomial theorem
gives $B(H,p)=\sum_{r=0}^H2^r\binom Hr=3^H$.  The stated logarithmic
bound follows from these two cases.

<!-- lemma-9: body_md -->
\label{lem:growth-deviation}
Let $\mathcal H$ be a class of binary functions and let $P_Q$ be the
empirical measure of $Q$ independent observations from $P$.  For every
$u>0$,
\begin{equation}
\label{eq:vc-ineq}
\mathbb P\left\{
\sup_{h\in\mathcal H}|P_Qh-Ph|>u

<!-- end lemma-9 -->

Write $Z:=\sup_{h\in\mathcal H}|P_Qh-Ph|$.  Introduce an
independent ghost sample with empirical measure $P'_Q = \frac1Q \sum_{j=1}^Q \delta_{q'_j}$, and write
$Z':=\sup_{h\in\mathcal H}|P_Qh-P'_Qh|$.  Conditional on the original
sample, $P_Qh-Ph=\mathbb E[P_Qh-P'_Qh\mid q_1,\dots,q_Q]$ for every $h$.

gives $\mathbb E e^{\lambda Z}\le\mathbb E e^{\lambda Z'}$ for every $\lambda>0$.

distribution after independent random swaps.  Encode the swaps by
Rademacher signs $\sigma_1,\dots,\sigma_Q$.  Conditional on the combined
sample, a fixed restriction of $h$ gives coefficients

$Q^{-1}\sum_{j=1}^Q\sigma_jd_j$.  There are at most
$\Pi_{\mathcal H}(2Q)$ such restrictions.  Hoeffding's exponential-moment
bound, together with $e^{\lambda|x|}\le e^{\lambda x}+e^{-\lambda x}$,
thus gives
\[
\mathbb E e^{\lambda Z}
\le
2\Pi_{\mathcal H}(2Q)
\exp\left(\frac{\lambda^2}{2Q}\right).
\]
Markov's inequality with $\lambda=Qu$ now proves
\eqref{eq:vc-ineq}.

We first bound the error of the deployment vote on questions with margin
greater than $r$.  The enlarged training buffer will then control the
population mass of the remaining questions.

$a\ne a^\star(q)$, define the vote differences
\[
D_t(a):=\mathbf 1\{Y_t=a^\star(q)\}-\mathbf 1\{Y_t=a\},
\qquad t\in[T].
\]
Conditional on $q$, these variables are independent, take values in
$[-1,1]$, and have mean
$p_w(a^\star(q)\mid q)-p_w(a\mid q)\ge M_q(w)>r$.
Hoeffding's inequality gives
\[
\mathbb P\!\left\{\sum_{t=1}^T D_t(a)\le0\,\middle|\,q\right\}
\le e^{-T M_q(w)^2/2}
\le e^{-Tr^2/2}.
\]

event occurs for some wrong answer.  A union bound over at most $A-1$

$(A-1)e^{-Tr^2/2}$.  On questions with $M_q(w)\le r$, we bound the
conditional error probability by one.  Averaging over questions gives

\label{eq:deployment-vote-bound}

<!-- lemma-10: body_md -->
\label{lem:margin-class-properties}
The class $\mathcal G_{beta,L}$ is convex, invariant under reflection
$G\mapsto G^\dagger$, where $G^\dagger(B)=G(-B)$, and compact for weak convergence.  Every $G\in\mathcal G_{beta,L}$ satisfies $G(\{0\})=0$.  The functional $G\mapstotheta(G)$ and the channel map $G\mapsto K_nG$ are affine and continuous on $\mathcal G_{beta,L}$.  Moreover,
\begin{equation}
\label{eq:reflection-properties}
theta(G^\dagger)=-theta(G),
\qquad
(K_nG^\dagger)(k)=(K_nG)(n-k),
\quad k=0,\dots,n.
\end{equation}
<!-- end lemma-10 -->
<!-- proof-13: body_md -->
Convexity and reflection invariance are immediate from the definition of the class.  Since $[-1,1]$ is compact, the set of all probability measures on it is weakly compact.  It remains to show that the margin class is weakly closed.  Suppose $G_m\Rightarrow G$.  Fix $0<t<1$ and choose $\eta>0$ such that $t+\eta<1$.  By the Portmanteau theorem applied to the open interval $(-t-\eta,t+\eta)$,
\[
G([-t,t])
\le
G((-t-\eta,t+\eta))
\le
\liminf_{m\toinfinity}G_m((-t-\eta,t+\eta))
\le
L(t+\eta)^beta.
\]
Letting $\eta\downarrow0$ proves the margin inequality for $t<1$; at $t=1$ it is automatic because $G([-1,1])=1\le L$.  Thus $\mathcal G_{beta,L}$ is weakly closed and hence compact.  Taking $t\downarrow0$ shows that $G(\{0\})=0$.

For continuity of $theta$, let $\psi_\delta$ be the continuous clipped-sign function that equals $-1$ on $[-1,-\delta]$, equals $z/\delta$ on $[-\delta,\delta]$, and equals $1$ on $[\delta,1]$.  Uniformly over the margin class,
\[
\left|theta(G)-\int\psi_\delta\,\,\mathrm d G\right|
\le
2G([-\delta,\delta])
\le
2L\delta^beta.
\]
The integral of $\psi_\delta$ is weakly continuous.  First letting $m\toinfinity$ and then $\delta\downarrow0$ shows that $theta(G_m)\totheta(G)$.

Each coordinate of $K_nG$ is the integral of a bounded continuous polynomial in $z$, so the channel map is weakly continuous; both the channel and target maps are affine by linearity of integration.  Finally, reflection changes the sign of the target, and the Bernstein identity
\[
b_{k,n}\!\left(\frac{1-z}{2}\right)
=
b_{n-k,n}\!\left(\frac{1+z}{2}\right)
\]
gives the channel identity in \eqref{eq:reflection-properties}.

\label{lem:robust-affinity-test}
Let $\mathcal P_0,\mathcal P_1\subsetDelta_m$ be nonempty compact convex sets, and put

:=
\inf\{h(p,q):p\in\mathcal P_0,\ q\in\mathcal P_1\}.

\le

First suppose that every distribution in $\mathcal P_0\cup\mathcal P_1$ has strictly positive coordinates.  Let $(p_0,q_1)$ maximize the Hellinger affinity
\[
\rho(p,q):=\sum_{i=1}^m\sqrt{p_iq_i}
\]
over $\mathcal P_0\times\mathcal P_1$.  For fixed $q_1$, the map $p\mapsto\rho(p,q_1)$ is concave.  First-order optimality at $p_0$ along every segment from $p_0$ to $p\in\mathcal P_0$ gives
\begin{equation}
\label{eq:affinity-first-order-p}
\sum_{i=1}^m p_i\sqrt{\frac{q_{1i}}{p_{0i}}}
\le
\sum_{i=1}^m p_{0i}\sqrt{\frac{q_{1i}}{p_{0i}}}
=\rho(p_0,q_1).
\end{equation}
Similarly, for every $q\in\mathcal P_1$,
\begin{equation}
\label{eq:affinity-first-order-q}
\sum_{i=1}^m q_i\sqrt{\frac{p_{0i}}{q_{1i}}}
\le\rho(p_0,q_1).
\end{equation}
Use the likelihood-ratio test between $p_0^{\otimes k}$ and $q_1^{\otimes k}$, namely,
$\phi = \mathbf 1 \left\{\prod_{j=1}^k q_1(Y_j)/p_0(Y_j)\ge1 \right\}$.  For $p\in\mathcal P_0$, Markov's inequality and \eqref{eq:affinity-first-order-p} yield
\begin{align*}
\mathbb E_{p^{\otimes k}}\phi
\;\le\;
\mathbb E_{p^{\otimes k}}
\prod_{j=1}^k
\left(\frac{q_1(Y_j)}{p_0(Y_j)}\right)^{1/2}
=
\left[
\sum_{i=1}^m p_i\sqrt{\frac{q_{1i}}{p_{0i}}}
\right]^k
\le\rho(p_0,q_1)^k.
\end{align*}
The inverse likelihood ratio and \eqref{eq:affinity-first-order-q} give the same bound for the other error.  Since maximizing affinity is equivalent to minimizing Hellinger distance,
\[
\rho(p_0,q_1)=1-\frac12r_*^2,
\]
which proves \eqref{eq:robust-test-error} under strict positivity.

To remove that assumption, let $u$ be the uniform law and, for $\eta\in(0,1)$, set
\[
\mathcal P_j^\eta
:=
\{(1-\eta)p+\eta u:p\in\mathcal P_j\},
\qquad j=0,1.
\]
Apply the preceding construction to obtain a randomized test $\phi_\eta$, viewed as a vector in $[0,1]^{m^k}$.  Along a sequence $\eta_\ell\downarrow0$, compactness gives coordinatewise convergence to a randomized test $\phi$.  Uniform continuity of Hellinger affinity shows that the maximal affinities of the smoothed sets converge to the maximal affinity of the original sets.  For each fixed $p\in\mathcal P_0$ and $q\in\mathcal P_1$, the corresponding smoothed product laws converge in total variation to $p^{\otimes k}$ and $q^{\otimes k}$.  Together with coordinatewise convergence of the tests on the finite sample space, this implies convergence of their testing risks.  Passing to the limit proves the result on the boundary of the simplex.

Our goal is to show an estimator that has expected absolute error at most $7\delta/2$, uniformly over $\mathcal X$.

$D/2 = \delta/2$.

In particular, $\delta=0$ would imply $D=0$.  If $\delta\ge D/6$, the midpoint again has error at most $D/2\le3\delta$.  It remains to consider

\end{equation}

We now construct a sample-split interval-search estimator.  It proceeds through stages $s=0,\dots,S-1$ and maintains a nested sequence of intervals $I_s$ intended to localize $\tau$.  Start from $I_0=[\tau_-,\tau_+]$.  The observations are divided into $S$ disjoint blocks, one per stage, with block sizes specified below.  At stage $s$, write $I_s=[a_s,b_s]$ and $\ell_s=b_s-a_s$.  Using only block $s$, test the lower third of $I_s$ against the upper third through the composite hypotheses
\[
H_{s}^-:\ \tau\le a_s+\frac{\ell_s}{3}
\qquad\text{and}\qquad
H_{s}^+:\ \tau\ge a_s+\frac{2\ell_s}{3}.
\]

The interval lengths are deterministic and satisfy
\[

\qquad

\qquad
p_++p_-=1,

\qquad
F_-(t)=G([-t,0)),
\qquad 0<t\le1.
\]

The construction treats the two sides of zero separately.  Starting at
zero, divide as much of the mass on each side as possible into blocks of
mass $1/M$, and replace each block by an atom at its endpoint farther from
zero.  Any atom needed to round the number of blocks is placed at $+1$ or
$-1$.  Moving mass outward in this way is what preserves the margin bound.

$m_-=M-m_+$.  These are the numbers of atoms assigned to the positive and
negative sides.  They satisfy
\begin{equation}
\label{eq:side-mass-rounding}
\left|\frac{m_+}{M}-p_+\right|\le\frac1{2M},
\qquad
\left|\frac{m_-}{M}-p_-\right|\le\frac1{2M}.
\end{equation}

\qquad
k_-=\lfloor Mp_-\rfloor,
\qquad

\qquad
e_-=m_--k_-.

x_j^+
=
\inf\{t\in(0,1]:F_+(t)\ge j/M\},
\qquad
x_j^-
=
\inf\{t\in(0,1]:F_-(t)\ge j/M\}.

=
\frac1M

\sum_{j=1}^{k_+}\delta_{x_j^+}
+
\sum_{j=1}^{k_-}\delta_{-x_j^-}

+
e_-\delta_{-1}
\right\}.
\end{equation}

\frac{\lfloor MF_-(t)\rfloor}{M}.

=

\le

\[
0\le F_+(t)-G^{(M)}((0,t])<\frac1M,
\qquad
0\le F_-(t)-G^{(M)}([-t,0))<\frac1M.
\]

\]

$J_1,\dots,J_Q$ iid uniformly from $[M]$, independently of $(R,q)$, and

\label{lem:jackson}
For every $s>0$, there is $C_s$ such that, for every integer $n\ge1$, there is $P_n\in\Pi_n$ satisfying
\begin{equation}
\label{eq:jackson-bound}
\norm{P_n}_{L_infinity[-1,1]}\le1,
\qquad
|P_n(x)-\operatorname{sgn}(x)|
\le
C_s\min\{1,(n|x|)^{-s}\}
\end{equation}
for every $x\in[-1,1]\setminus\{0\}$.

<!-- proof-25: body_md -->
The key idea is to smooth the discontinuous sign function with a positive, normalized, finite-frequency Jackson kernel.  The resulting convolution is a bounded trigonometric polynomial that closely follows the sign function away from its jumps; evenness then converts it into an algebraic polynomial in $x=\cos\vartheta$.

Let $f(\vartheta):=\operatorname{sgn}(\cos\vartheta)$ on the circle $\mathbb R/(2\pi\mathbb Z)$, and choose an integer $r$ such that $2r-1>s$.  For $m\ge2$, define
\[
J_{m,r}(u)
:=c_{m,r}
\left(\frac{\sin(mu/2)}{\sin(u/2)}\right)^{2r},
\]
where the quotient is interpreted continuously at zero.  The constant $c_{m,r}$ is chosen to give normalization $\frac1{2\pi} \int_{-\pi}^{\pi} J_{m,r}(u)\,\,\mathrm d u=1$.

\emph{Kernel properties.}
Recall that the Fej\'er kernel satisfies
\[
F_m(u)
:=\frac1m\left(\frac{\sin(mu/2)}{\sin(u/2)}\right)^2
=
\sum_{k=-(m-1)}^{m-1}
\left(1-\frac{|k|}{m}\right)e^{iku}.
\]
Since $J_{m,r}=c_{m,r}m^rF_m^r$, it is nonnegative, even, and a trigonometric polynomial of degree $d:=r(m-1)$.

To determine its normalization scale, put $K_{m,r}^0(u):=(\sin(mu/2)/\sin(u/2))^{2r}$.  On $|u|\le1/m$, one has $K_{m,r}^0(u)\asymp_r m^{2r}$, while for $0<|u|\le\pi$, the inequalities $|\sin(u/2)|\ge|u|/\pi$ and $|\sin(mu/2)|\le1$ give $K_{m,r}^0(u)\lesssim_r|u|^{-2r}$.  Splitting the integral at $1/m$ therefore yields
\[
\frac1{2\pi}\int_{-\pi}^{\pi}K_{m,r}^0(u)\,\,\mathrm d u
\asymp_r m^{2r-1},
\qquad
c_{m,r}\asymp_r m^{-(2r-1)}.
\]
Only the resulting upper bound on $c_{m,r}$ is needed below.  We next bound the tail.  If $a\le1/m$, normalized total mass bounds it by one; if $a>1/m$, the preceding pointwise and normalization bounds give
\[
\frac1{2\pi}\int_{|u|\ge a}J_{m,r}(u)\,\,\mathrm d u
\lesssim_r
m^{-(2r-1)}\int_a^\pi u^{-2r}\,\,\mathrm d u
\lesssim_r
(ma)^{-(2r-1)}.
\]
Thus, for every $0<a\le\pi$,
\begin{equation}
\label{eq:jackson-tail}
\frac1{2\pi}\int_{|u|\ge a}J_{m,r}(u)\,\,\mathrm d u
\le
C_r\min\{1,(ma)^{-(2r-1)}\},
\qquad 0<a\le\pi.
\end{equation}

\emph{Finite-frequency smoothing.}
Define
\[
T_m(\vartheta)
:=
\frac1{2\pi}\int_{-\pi}^{\pi}
f(\vartheta-u)J_{m,r}(u)\,\,\mathrm d u.
\]
Since $J_{m,r}\ge0$ and has normalized mass one, $T_m(\vartheta)$ is a weighted average of values in $[-1,1]$, so $|T_m(\vartheta)|\le1$.  Write $J_{m,r}(u)=\sum_{|k|\le d}j_ke^{iku}$.  The convolution identity gives
\[
T_m(\vartheta)
=
\sum_{|k|\le d}j_k\widehat f(k)e^{ik\vartheta},
\]
where $\widehat f(k)$ is the $k$th Fourier coefficient of $f$.  Thus $T_m$ is a trigonometric polynomial of degree at most $d$.  Because both $f$ and $J_{m,r}$ are even, so is $T_m$, and hence it has a cosine expansion $T_m(\vartheta)=a_0+\sum_{k=1}^da_k\cos(k\vartheta)$.  If $\mathsf T_k$ denotes the $k$th Chebyshev polynomial, define $P(x):=a_0+\sum_{k=1}^da_k\mathsf T_k(x)$.  Then $P$ has degree at most $d$ and
\[
T_m(\vartheta)=P(\cos\vartheta),
\qquad
\norm P_{L_infinity[-1,1]}\le1.
\]

\emph{Localized approximation error.}
Let $a(\vartheta)$ denote the circular distance from $\vartheta$ to the nearest discontinuity of $f$, namely $\pi/2$ or $3\pi/2$.  Whenever $|u|<a(\vartheta)$, the values $f(\vartheta-u)$ and $f(\vartheta)$ agree.  Therefore \eqref{eq:jackson-tail} implies
\[
\begin{aligned}
|T_m(\vartheta)-f(\vartheta)|
&\le
\frac2{2\pi}\int_{|u|\ge a(\vartheta)}J_{m,r}(u)\,\,\mathrm d u\\
&\le
C_r\min\{1,(m a(\vartheta))^{-(2r-1)}\}.
\end{aligned}
\]
If $x=\cos\vartheta$, then $|x|=\sin a(\vartheta)\le a(\vartheta)$.  For $n\ge2r$, take $m=1+\lfloor n/r\rfloor$.  Then $d=r(m-1)\le n$ and $m\ge n/r$.  Since $2r-1>s$, the last display implies \eqref{eq:jackson-bound}, after enlarging the constant.  For the finitely many integers $1\le n<2r$, take $P_n\equiv0$ and enlarge $C_s$.

\end{equation}

Consequently, the Hellinger radius in \eqref{eq:spectral-modulus-lower} is at most $Q^{-1/2}$.  Monotonicity of the modulus, \cref{thm:scalar-modulus}, and $d\asymp\sqrt{n\Lambda_{Q,n}}$ yield
\[
\mathcal R^{\mathrm{dec}}_{Q,n}(beta,L)
\gtrsim_{beta,L}
(n\Lambda_{Q,n})^{-beta/2}.
\]
The comparison $\Lambda_{Q,n}\asymp_{beta,L}y_{Q,n}$ proves \eqref{eq:spectral-explicit-lower}.  When $beta>1/2$, apply \eqref{eq:spectral-modulus-lower-L2} instead and set $\Lambda_{Q}:=\log(A_{beta,L}\sqrt Q)$.  Then $\Lambda_Q\asymp_{beta,L}y_Q^{(2)}$, and the same argument proves the refined assertion.

For $d\le n/2$, the upper half of \eqref{eq:kappa-asymp} gives $\log\kappa_{d,n}\le C_\kappa d(d+1)/n$.  Choose
\[
d=\left\lfloor a\sqrt{nx_{Q,n}}\right\rfloor\vee1
\]
with $a>0$ small enough that $C_\kappa d(d+1)/n\le x_{Q,n}/4$.  Reducing $c_0$ if necessary ensures $d\le n/2$, while $x_{Q,n}\ge1$ gives $d\asymp\sqrt{nx_{Q,n}}$.  The spectral upper bound \eqref{eq:spectral-upper} therefore yields a bias term of order $(nx_{Q,n})^{-beta/2}$ and a stochastic term bounded by
\[
\frac{\sqrt{n+1}\,\kappa_{d,n}}{\sqrt Q}
\lesssim
\frac{\sqrt n\,e^{x_{Q,n}/4}}
{n^{(beta+1)/2}e^{x_{Q,n}/2}}
=
n^{-beta/2}e^{-x_{Q,n}/4}.
\]
Since $e^{-x/4}\le C_beta x^{-beta/2}$ for $x\ge1$, the stochastic term has the same stated order, proving \eqref{eq:intermediate-upper}.
