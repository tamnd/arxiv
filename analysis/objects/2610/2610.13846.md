<!-- definition-1: body_md -->
A function  $\widetilde{C}\colon\Delta\to\mathbb{I}$ is called a \emph{discrete copula}, if
\begin{enumerate}[(i)]
    \item $\widetilde{C}(a_i, 0)=\widetilde{C}(0, b_j)=0$ for all $i \in [m], j \in [n]$ ($\widetilde{C}$ is \emph{grounded}),
    \item $\widetilde{C}(a_i, 1)=a_i$ and $\widetilde{C}(1, b_j)=b_j$ for all $i \in [m], j \in [n]$ ($\widetilde{C}$ has \emph{uniform marginals}), and
    \item  $V_{\widetilde{C}}([a_{i-1},a_i]\times[b_{j-1},b_j])  \ge 0$ ($\widetilde{C}$ is \emph{$2$-increasing}).
\end{enumerate}
<!-- end definition-1 -->
<!-- definition-2: body_md -->
Let $A$ be a checkerboard copula with splitting points $\Delta$. 
\begin{enumerate}[(i)]
    \item The \emph{value matrix} $S_A = [s_{ij}]_{ij} \in \mathbb{R}^{m\times n}$ is the matrix with entries $$s_{ij} = A(a_i, b_j).$$
    \item The \emph{mass matrix} $M_A = [m_{ij}]_{ij} \in \mathbb{R}^{m\times n}$ is the matrix with entries $$m_{ij} = V_A([a_{i-1},a_i]\times[b_{j-1},b_j]).$$
    \item The \emph{density matrix} $G_A = [g_{ij}]_{ij} \in \mathbb{R}^{m\times n}$ is the matrix with entries $$g_{ij} = \partial_1\partial_2A(\tfrac{a_{i-1}+a_i}{2},\tfrac{b_{j-1}+b_j}{2}).$$
\end{enumerate}
<!-- end definition-2 -->
<!-- proposition-1: body_md -->
\label{prop:connection}
    Let $A$ be a checkerboard copula with splitting points $\Delta$. Then
$$ M_A = J_m^T S_A J_n = D_{\delta_1} G_A D_{\delta_2}. $$
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
For every $i\in [m], j \in [n]$ we have
\begin{align*}
  m_{ij} &= V_A([a_{i-1},a_i]\times[b_{j-1},b_j]) \\
  &= A(a_i,b_j)-A(a_i,b_{j-1})-A(a_{i-1},b_j)+A(a_{i-1},b_{j-1}) \\
  &= s_{ij}-s_{i,j-1}-s_{i-1,j}+s_{i-1,j-1}, 
\end{align*}
where we adopt the convention $s_{0j}=s_{i0}=0$. It follows that $M_A = J_m^T S_A J_n$. Furthermore, since copula $A$ is absolutely continuous, 
\begin{align*}
  m_{ij} &= V_A([a_{i-1},a_i]\times[b_{j-1},b_j]) = \int_{a_{i-1}}^{a_i} \int_{b_{j-1}}^{b_j} \partial_1\partial_2A(u,v)\,du\,dv\\
  &= \int_{a_{i-1}}^{a_i} \int_{b_{j-1}}^{b_j} g_{ij}\,du\,dv = (a_i-a_{i-1})(b_j-b_{j-1})g_{ij}, 
\end{align*}
thus $M_A = D_{\delta_1} G_A D_{\delta_2}$.
<!-- end proof-1 -->
<!-- lemma-1: body_md -->
\label{lem:checkerboard_hat}
    If $A$ is a checkerboard copula with splitting points $\Delta$, then the survival copula $\widehat{A}$ is also a checkerboard copula with splitting points $\Delta' = \delta'_1 \times \delta'_2$, where
\begin{align*}
\delta'_1 &:= \{1-a_m, 1-a_{m-1}, \ldots, 1-a_1, 1-a_0\}  \subseteq \mathbb{I},\\
\delta'_2 &:= \{1-b_n, 1-b_{n-1}, \ldots, 1-b_1, 1-b_0\}  \subseteq \mathbb{I},
\end{align*}
and 
 $$ S_{\widehat{A}} = J_m^{-T} P_m J_m^T S_A J_n P_n J_n^{-1}, \qquad M_{\widehat{A}} = P_m M_A P_n, \qquad G_{\widehat{A}} = P_m G_A P_n. $$
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
By equation \eqref{eq:Chat} we have 
$$\partial_1\partial_2\widehat{A}(u,v) = \partial_1\partial_2A(1-u,1-v).$$
so $\partial_1\partial_2\widehat{A}(u,v)$ is constant on every rectangle $(1-a_i,1-a_{i-1})\times (1-b_j,1-b_{j-1})$. Thus $\widehat{A}$ is a checkerboard copula with splitting points $\Delta'$. For its density matrix $G_{\widehat{A}} = [g'_{ij}]_{ij}$ we have
$g'_{ij} = g_{m+1-i,n+1-j}$ for any $i\in [m]$ and $j \in [n]$, so $G_{\widehat{A}} = P_m G_A P_n$. Furthermore,
\begin{align*}
    M_{\widehat{A}} &= D_{\delta'_1} G_{\widehat{A}} D_{\delta'_2} = D_{\delta'_1} P_m G_A P_n D_{\delta'_2} = D_{\delta'_1} P_m D_{\delta_1}^{-1} M_A 
    D_{\delta_2}^{-1} P_n D_{\delta'_2} = P_m M_A P_n, \text{ and} \\
    S_{\widehat{A}} &= J_m^{-T} M_{\widehat{A}} J_n^{-1} = J_m^{-T} P_m M_A P_n J_n^{-1} = J_m^{-T} P_m J_m^T S_A J_n P_n J_n^{-1}
\end{align*}
by Proposition~\ref{prop:connection}.

<!-- proposition-2: body_md -->
\label{prop:matrices}
Let $\Delta=\delta_1 \times \delta_2$ be a mesh of splitting points, where $|\delta_1|=m+1$ and $|\delta_2|=n+1$.
\begin{enumerate}[$(i)$]
 \item A matrix $M = [m_{ij}]_{ij} \in \mathbb{R}^{m\times n}$ is the mass matrix of some checkerboard copula with splitting points $\Delta$ if and only if it is entry-wise nonnegative and 
$$ M\mathbf{1}_n = \mathbf{v}_{\delta_1}, \qquad M^T\mathbf{1}_m = \mathbf{v}_{\delta_2}. $$
 \item A matrix $G = [g_{ij}]_{ij} \in \mathbb{R}^{m\times n}$ is the density matrix of some checkerboard copula with splitting points $\Delta$ if and only it is entry-wise nonnegative and 
 $$ G\mathbf{v}_{\delta_2} = \mathbf{1}_m, \qquad G^T\mathbf{v}_{\delta_1} = \mathbf{1}_n. $$
 \item A matrix $S = [s_{ij}]_{ij} \in \mathbb{R}^{m\times n}$ is the value matrix of some checkerboard copula with splitting points $\Delta$ if and only if the matrix $J_m^T S J_n$  is entry-wise nonnegative and 
$$ s_{in} = a_i \text{ for all } i \in [m], \text { and } s_{mj} = b_j \text{ for all } j \in [n].$$
\end{enumerate}
<!-- end proposition-2 -->
<!-- proof-3: body_md -->
$(i)$: Let $M$ be a mass matrix of a checkerboard copula with splitting points $\Delta$. Then $M$ is entry-wise nonnegative and we have for every $i \in [m]$
$$\sum_{j=1}^n m_{ij} = \sum_{j=1}^n V_A([a_{i-1},a_i] \times [b_{j-1},b_j]) = V_A([a_{i-1},a_i] \times [0, 1]) = a_i-a_{i-1},$$
so $M\mathbf{1}_n = \mathbf{v}_{\delta_1}$. The equality $M^T\mathbf{1}_m = \mathbf{v}_{\delta_2}$ is proved analogously. Now suppose that $M$ is entry-wise nonnegative, $M\mathbf{1}_n = \mathbf{v}_{\delta_1}$ and $M^T\mathbf{1}_m = \mathbf{v}_{\delta_2}$. Note that vectors $\mathbf{v}_{\delta_1}$ and $\mathbf{v}_{\delta_2}$ uniquely determine the sets $\delta_1$ and $\delta_2$ and that $\mathbf{1}_m^T\mathbf{v}_{\delta_1} = a_m = \mathbf{1}_n^T\mathbf{v}_{\delta_2} = b_n = 1$. Let $S = J_m^{-T} M_{\widehat{A}} J_n^{-1}$. For every $i\in [m]$ and $j \in [n]$ we have
$$ s_{ij} = \sum_{k=1}^i \sum_{\ell=1}^j m_{k\ell}.$$
Let $\widetilde{A} \colon \Delta \to \mathbb{I}$ be a function defined by $\widetilde{A}(a_i,b_j) = s_{ij}$ and $\widetilde{A}(0,0) = \widetilde{A}(a_i,0) = \widetilde{A}(0, b_j) = 0$ for every $i \in [m]$ and $j \in [n]$. Then $\widetilde{A}$ is grounded, we have $V_{\widetilde{A}}([a_{i-1},a_i]\times[b_{j-1},b_j]) = m_{ij} \ge 0$, 
$$\widetilde{A}(a_i,1) = s_{in} = \sum_{k=1}^i \sum_{\ell=1}^n m_{k\ell} = \sum_{k=1}^i (a_k-a_{k-1}) = a_i,$$
and, similarly, $\widetilde{A}(1,b_j) = b_j$ for every $i\in [m]$ and $j \in [n]$. Hence, $\widetilde{A}$ is a discrete copula. Finally, we extend discrete copula $\widetilde{A}$ to a checkerboard copula $A$ using bilinear functions. By the above, the mass matrix of copula $A$ is $M$.

$(ii)$: Let $G$ be the density matrix of a checkerboard copula with splitting points $\Delta$. Then $M = D_{\delta_1} G D_{\delta_2}$ is its mass matrix by Proposition~\ref{prop:connection}, $M$ is entry-wise nonnegative, $M\mathbf{1}_n = \mathbf{v}_{\delta_1}$ and $M^T\mathbf{1}_m = \mathbf{v}_{\delta_2}$ by $(i)$. It follows that $G=D_{\delta_1}^{-1} M D_{\delta_2}^{-1}$ is entry-wise nonnegative, 
$$G\mathbf{v}_{\delta_2} = (D_{\delta_1}^{-1} M D_{\delta_2}^{-1})(D_{\delta_2} \mathbf{1}_n) = D_{\delta_1}^{-1} \mathbf{v}_{\delta_2} = \mathbf{1}_n,$$ 
and, similarly, $G^T\mathbf{v}_{\delta_1} = \mathbf{1}_m$. Conversely, suppose that matrix $G$ is entry-wise nonnegative, $G\mathbf{v}_{\delta_2} = \mathbf{1}_n$ and $G^T\mathbf{v}_{\delta_1} = \mathbf{1}_m$. Let $M = D_{\delta_1} G D_{\delta_2}$. Then $M$ is entry-wise nonnegative, $M\mathbf{1}_n = D_{\delta_1} G D_{\delta_2}\mathbf{1}_n = D_{\delta_1} G \mathbf{v}_{\delta_2} = D_{\delta_1} \mathbf{1}_m = \mathbf{v}_{\delta_1}$ and $M^T\mathbf{1}_m = \mathbf{v}_{\delta_2}$, so $M$ is the mass matrix of some checkerboard copula with splitting points $\Delta$ by $(i)$, and thus $G$ is its density matrix.

$(iii)$: Let $S$ be the value matrix of a checkerboard copula with splitting points $\Delta$. Then $M = J_m^T S J_n$ is its mass matrix by Proposition~\ref{prop:connection}, $M$ is entry-wise nonnegative, $M\mathbf{1}_n = \mathbf{v}_{\delta_1}$ and $M^T\mathbf{1}_m = \mathbf{v}_{\delta_2}$ by $(i)$. As already proven in $(i)$, we have $s_{in} = a_i$ and $s_{mj} = b_j$ for every $i\in [m]$ and $j \in [n]$. Conversely, suppose that $S$ is a matrix, such that $M = J_m^T S J_n$ is entry-wise nonnegative, $s_{in} = a_i$ and $s_{mj} = b_j$ for every $i\in [m]$ and $j \in [n]$. Then 
$$M\mathbf{1}_n = J_m^T S J_n \mathbf{1}_n = J_m^T S (0, 0, ..., 0, 1)^T = J_m^T (a_1, a_2, ..., a_m)^T = \mathbf{v}_{\delta_1}$$ and $M^T\mathbf{1}_m = \mathbf{v}_{\delta_2}$, so $M$ is the mass matrix of some checkerboard copula with splitting points $\Delta$ by $(i)$, thus $S$ is its value matrix.

<!-- lemma-2: body_md -->
\label{lem:psi_hat}
    Let $C$ be a copula. Then $\Psi(\widehat{C})=\widehat{\Psi(C)}$.
<!-- end lemma-2 -->
<!-- proof-4: body_md -->
Since $\widehat{C}(t,u)=t+u-1+C(1-t,1-u)$, we have $\partial_1 \widehat{C}(t,u)=1-\partial_1 C(1-t,1-u)$ for all $t,u \in \mathbb{I}$. Hence, for all $u,v \in \mathbb{I}$
\begin{align*}
    \Psi(\widehat{C})(u,v)
    &=\int_0^1 \partial_1 \widehat{C}(t,u) \partial_1 \widehat{C}(t,v) dt \\
    &=\int_0^1 \big(1-\partial_1 C(1-t,1-u)\big)\big(1-\partial_1 C(1-t,1-v)\big) dt\\
    &=\int_0^1 \big(1-\partial_1 C(s,1-u)\big)\big(1-\partial_1 C(s,1-v)\big) ds\\
    &=1-\int_0^1 \partial_1 C(s,1-v) ds-\int_0^1 \partial_1 C(s,1-u) ds+\int_0^1 \partial_1 C(s,1-u)\partial_1 C(s,1-v) ds\\
    &=1-(1-v)-(1-u)+\Psi(C)(1-u,1-v)\\
    &=\widehat{\Psi(C)}(u,v),
\end{align*}
which finishes the proof.

<!-- proposition-3: body_md -->
\label{prop:ordinal}
    If $C$ is the ordinal sum of $\{C_i\}_{i \in I}$ with respect to $\{J_i\}_{i \in I}$, then $\Psi(C)$ is the ordinal sum of $\{\Psi(C_i)\}_{i \in I}$ with respect to $\{J_i\}_{i \in I}$.
<!-- end proposition-3 -->
<!-- proof-5: body_md -->
Let $J_i=(a_i,b_i)$ for all $i \in I$. By equation~\eqref{eq:ordinal} we have
\begin{equation}\label{eq:partial_C}
\partial_1C(u,v)=\begin{cases}
    \partial_1 C_i\big(\frac{u-a_i}{b_i-a_i},\frac{v-a_i}{b_i-a_i}\big), & (u,v) \in (a_i,b_i) \text{ for some } i \in I,\\
    \partial_1 M(u,v), & \text{otherwise},
\end{cases}
\end{equation}
whenever the partial derivatives on the right-hand side exist. Note also that $\partial_1 M(u,v)$ equal $1$ if $u<v$ and $0$ if $u>v$. 

Now let $u,v \in (a_i,b_i)$. Then
\begin{align*}
    \Psi(C)(u,v) &= \int_0^1 \partial_1 C(t,u) \partial_1 C(t,v) dt\\
    &= \int_{0}^{a_i} \partial_1 C(t,u) \partial_1 C(t,v) dt
    +\int_{a_i}^{b_i} \partial_1 C(t,u) \partial_1 C(t,v) dt
    +\int_{b_i}^{1} \partial_1 C(t,u) \partial_1 C(t,v) dt\\
    &=a_i+\int_{a_i}^{b_i} \partial_1 C(t,u) \partial_1 C(t,v) dt,
\end{align*}
since $\partial_1 C(t,u)=\partial_1 C(t,v)=1$ for all $t\le a_i$ and $\partial_1 C(t,u)=\partial_1 C(t,v)=0$ for all $t \ge b_i$.
Thus, by equation~\eqref{eq:partial_C}, we obtain
\begin{align*}
    \Psi(C)(u,v) &= a_i+\int_{a_i}^{b_i} \partial_1 C_i\big(\tfrac{t-a_i}{b_i-a_i},\tfrac{u-a_i}{b_i-a_i}\big) \partial_1 C_i\big(\tfrac{t-a_i}{b_i-a_i},\tfrac{v-a_i}{b_i-a_i}\big) dt\\
    &= a_i+(b_i-a_i)\int_0^1 \partial_1 C_i\big(s,\tfrac{u-a_i}{b_i-a_i}\big) \partial_1 C_i\big(s,\tfrac{v-a_i}{b_i-a_i}\big) ds\\
    &=a_i+(b_i-a_i)\Psi(C_1)\big(\tfrac{u-a_i}{b_i-a_i},\tfrac{v-a_i}{b_i-a_i}\big).
\end{align*}
Next, let $u \in (a_i,b_i)$ and $v\ge b_i$. Since $\partial_1 C(t,u)=0$ for all $t \ge b_i$ and $\partial_1 C(t,v)=1$ for all $t \le b_i$, we have
\begin{align*}
    \Psi(C)(u,v) &= \int_0^1 \partial_1 C(t,u) \partial_1 C(t,v) dt= \int_{0}^{b_i} \partial_1 C(t,u) dt=C(b_i,u)=M(b_i,u)=u=M(u,v).
\end{align*}
Now, let $u \in (a_i,b_i)$ and $v \le a_i$. Since $\partial_1 C(t,u)=1$ for all $t \le a_i$ and $\partial_1 C(t,v)=0$ for all $t \ge a_i$, we have
\begin{align*}
    \Psi(C)(u,v) &= \int_0^1 \partial_1 C(t,u) \partial_1 C(t,v) dt= \int_{0}^{a_i} \partial_1 C(t,v) dt=C(a_i,v)=v=M(u,v).
\end{align*}
By symmetry, if $v \in (a_i,b_i)$ and $u \notin (a_i,b_i)$, we get $\Psi(u,v)=M(u,v)$.
Finally, if $u,v \notin \bigcup_{i \in I} (a_i,b_i)$, then $\Psi(C)(u,v)=\Psi(M)(u,v)=M(u,v)$, since $\Psi(M)=M$.
This finishes the proof.

<!-- theorem-1: body_md -->
\label{thm:psd}
    Let $A$ be a checkerboard copula with splitting points in a symmetric mesh $\underline{\Delta}=\delta_2 \times \delta_2$ such that $A = \Psi(C)$ for some copula $C$.
    Then the value matrix, mass matrix, and density matrix of $A$ are all positive semidefinite.
<!-- end theorem-1 -->
<!-- proof-6: body_md -->
Let $S_A=[s_{ij}]_{ij} \in \mathbb{R}^{n \times n}$ be the value matrix of $A$. By equation~\eqref{eq:psi} we have
$$s_{ij}=A(b_i,b_j)=\int_0^1 \partial_1 C(t,b_i) \partial_1 C(t,b_j) dt.$$
For any vector $\mathbf{x}=(x_1,\ldots,x_n) \in \mathbb{R}^n$ we have
\begin{align*}
    \mathbf{x}^T S_A \mathbf{x} &=\sum_{i,j=1}^n s_{ij}x_ix_j=\sum_{i,j=1}^n \int_0^1 \partial_1 C(t,b_i) \partial_1 C(t,b_j) x_i x_j dt\\
    &=\int_0^1 \left(\sum_{i=1}^n \partial_1 C(t,b_i)x_i\right)^2 dt \ge 0.
\end{align*}
This proves that $S_A$ is positive semidefinite. For the mass matrix $M_A$ and the density matrix $G_A$ we have $M_A=J_n^TS_AJ_n$ and $G_A=D_{\delta_2}^{-T} M_A D_{\delta_2}^{-1}$ by Proposition~\ref{prop:connection}, so that both $M_A$ and $G_A$ are positive semidefinite as well.

<!-- theorem-2: body_md -->
\label{thm:cb_cp}
    Let $A_1$ be a checkerboard copula with splitting points $\Delta=\delta_1 \times \delta_2$ and $A = \Psi(A_1)$. Then $A$  is a checkerboard copula with splitting points $\underline{\Delta}=\delta_2 \times \delta_2$ and its mass matrix and density matrix are completely positive.
    Furthermore,
    \begin{equation}\label{eq:G_psi}
        G_{A}=G_{A_1}^TD_{\delta_1}G_{A_1}.
    \end{equation}
<!-- end theorem-2 -->
<!-- proof-7: body_md -->
By equation~\eqref{eq:psi} we have
$$A(u,v)=\int_0^1 \partial_1 A_1(t,u) \partial_1 A_1(t,v) dt.$$
Letting $(u,v) \in [b_{i-1},b_i] \times [b_{j-1},b_j]$ and differentiating with respect to $u$ ad $v$ we obtain
\begin{align*}
    \partial_1\partial_2 A(u,v) &=\int_0^1 \partial_1\partial_2 A_1(t,u) \partial_1\partial_2 A_1(t,v) dt=\sum_{k=1}^n \int_{a_{k-1}}^{a_k} \partial_1\partial_2 A_1(t,u) \partial_1\partial_2 A_1(t,v) dt.
\end{align*}
Since $A_1$ is a checkerboard copula with splitting points $\Delta$, all mixed partial derivatives on the right-hand side are constant, which justifies differentiation under the integral sign. Denoting the density matrix of $A_1$ by $G_{A_1}=[h_{ij}]_{ij} \in \mathbb{R}^{m \times n}$, we obtain
\begin{align*}
    \partial_1\partial_2 A(u,v) =\sum_{k=1}^n \int_{a_{k-1}}^{a_k} h_{ki} h_{kj} dt=\sum_{k=1}^n h_{ki} h_{kj} (a_k-a_{k-1})=(G_{A_1}^TD_{\delta_1}G_{A_1})_{ij}.
\end{align*}
Since this mixed derivative is constant on $[b_{i-1},b_i] \times [b_{j-1},b_j]$ for all $i$ and $j$, we infer that $A$ is a checkerboard copula with splitting points $\underline{\Delta}$. Furthermore, we have shown that
\begin{equation}
    G_{A}=G_{A_1}^TD_{\delta_1}G_{A_1}=\Big(\sqrt{D_{\delta_1}}G_{A_1}\Big)^T\Big(\sqrt{D_{\delta_1}}G_{A_1}\Big).
\end{equation}
Since $\sqrt{D_{\delta_1}}G_{A_1}$ is entry-wise nonnegative matrix, we conclude that $G_A$ is completely positive. Proposition~\ref{prop:connection} implies that
$$M_A=D_{\delta_2} G_A D_{\delta_2}=\Big(\sqrt{D_{\delta_1}}G_{A_1}D_{\delta_2}\Big)^T\Big(\sqrt{D_{\delta_1}}G_{A_1} D_{\delta_2}\Big),$$
which is completely positive as well.

<!-- theorem-3: body_md -->
\label{thm:cp-psi}
    Let $A$ be a checkerboard copula with splitting points in a symmetric mesh $\underline{\Delta}=\delta_2 \times \delta_2$, such that its density matrix is completely positive. Then $A=\Psi(A_1)$ for some checkerboard copula $A_1$ with splitting points in $\Delta=\delta_1 \times \delta_2$ for an appropriately chosen finite set $\delta_1$.

<!-- proof-8: body_md -->
Let $|\delta_2|=n+1$. By assumption there exists a positive integer $m$ and an entry-wise nonnegative matrix $H \in \mathbb{R}^{m \times n}$ such that $G_A=H^TH$.
Furthermore, assume that $m$ is minimal possible, so that $H$ has nonzero rows (any zero row can simply be omitted from $H$).
Let $\mathbf{d}=HD_{\delta_2}\mathbf{1}_n=H\mathbf{v}_{\delta_2} \in \mathbb{R}^m$ and let $\delta_1=\{0,a_1,a_2,\ldots,a_m\}$, where $a_i=d_1^2+d_2^2+\ldots+d_i^2$.
Note that, by Proposition~\ref{prop:connection},
$$a_m=\mathbf{d}^T\mathbf{d}=\mathbf{1}_n^T D_{\delta_2}H^T HD_{\delta_2}\mathbf{1}_n=\mathbf{1}_n^T D_{\delta_2} G_A D_{\delta_2}\mathbf{1}_n=\mathbf{1}_n^T M_A\mathbf{1}_n=1,$$
since the sum of entries of $M_A$ is equal to $1$.
Note also that $D_{\delta_1}=D^2$, where $D=\mathrm{diag}(\mathbf{d})$, so that $\mathbf{v}_{\delta_1}=(d_1^2,d_2^2\ldots,d_m^2)^T$. Since the matrix $H$ has nonzero rows, vector $\mathbf{d}$ has nonzero components, so the matrix $D$ is invertible.
Furthermore, $\mathbf{v}_{\delta_1}=D_{\delta_1}\mathbf{1}_m=D^2\mathbf{1}_m$ and $\mathbf{v}_{\delta_2}=D_{\delta_2}\mathbf{1}_n$.
Now, let $G=D^{-1}H$, which is an entry-wise nonnegative matrix. Then
$$G\mathbf{v}_{\delta_2}=D^{-1}(H\mathbf{v}_{\delta_2})=D^{-1}\mathbf{d} =\mathbf{1}_m$$
and
$$G^T\mathbf{v}_{\delta_1}=(H^TD^{-1})(D^2\mathbf{1}_m)=H^T(D\mathbf{1}_m)=H^T\mathbf{d}=H^TH\mathbf{v}_{\delta_2}=G_A\mathbf{v}_{\delta_2}=\mathbf{1}_n$$
by Proposition~\ref{prop:matrices} $(ii)$.
Proposition~\ref{prop:matrices} $(ii)$ then implies that $G$ is the density matrix of some checkerboard copula $A_1$ with splitting points $\Delta=\delta_1 \times \delta_2$.
Equation~\eqref{eq:G_psi} implies that
$$G_{\Psi(A_1)}=G_{A_1}^TD_{\delta_1}G_{A_1}=G^TD_{\delta_1}G=(H^TD^{-1})D^2(D^{-1}H)=H^TH=G_A.$$
Since both $A$ and $\Psi(A_1)$ are checkerboard copulas (see Theorem~\ref{thm:cb_cp}) and the density matrix uniquely determines the checkerboard copula, we conclude that $A=\Psi(A_1)$.

where

    &= \int_0^1 \Big(\big(1-L_i(u)\big)\partial_1 C_1(t,b_{i-1})+L_i(u)\partial_1 C_1(t,b_i)\Big) \\
    & \hspace{3cm}\cdot \Big(\big(1-L_j(v)\big)\partial_1 C_1(t,b_{j-1})+L_j(v)\partial_1 C_1(t,b_j)\Big) dt\\
    &= \big(1-L_i(u)\big)\big(1-L_j(v)\big) \int_0^1 \partial_1 C_1(t,b_{i-1})\partial_1 C_1(t,b_{j-1}) dt\\

    &\hspace{1cm} +L_i(u)\big(1-L_j(v)\big) \int_0^1 \partial_1 C_1(t,b_i)\partial_1 C_1(t,b_{j-1}) dt\\
    &\hspace{1cm} +L_i(u)L_j(v) \int_0^1 \partial_1 C_1(t,b_i)\partial_1 C_1(t,b_j) dt\\
    &=\big(1-L_i(u)\big)\big(1-L_j(v)\big) C(b_{i-1},b_{j-1})+\big(1-L_i(u)\big)L_j(v) C(b_{i-1},b_j)\\

\end{align*}
The last expression is exactly the formula for the checkerboard extension of $C|_{\underline{\Delta}}$, given in equation~\eqref{eq:CB}, hence, $\Psi(D_1)=A$.

<!-- proposition-5: body_md -->
\label{prop:spodnja1}
    Let $C$ be a copula such that $C = \Psi(C_1)$ for some copula $C_1$, and let $a \in (0,1)$. Let $d =C(a,a) \in (a^2, a]$ be the value of $C$ at a diagonal point and $a^* = \tfrac{d}{a} \in (a,1]$. Then for any point $u \in [a, a^*]$  we have 
    $$ C(u,u) \ge \frac{d}{d-a^2}(u-a)^2 + d = \frac{a^*}{a^*-a}(u-a)^2 + d.$$
<!-- end proposition-5 -->
<!-- proof-10: body_md -->
Let $\underline{\Delta}=\{0, a, u, 1\} \times \{0, a, u, 1\}$ be a symmetric mesh and $A$ a checkerboard extension of $C|_{\underline{\Delta}}$. Then $A = \Psi(D_1)$ for some copula $D_1$ by Proposition \ref{prop:3.6}, so its value matrix $S_A = S_C$ is positive semidefinite by Theorem~\ref{thm:psd}.
We denote by $x = C(a,u) = C(u,a)$ and $y = C(u,u)$.
In this notation we have 
$$ S_C = \begin{bmatrix}
    d & x & a \\
    x & y  & u \\
    a & u & 1 
\end{bmatrix},$$
so 
\begin{equation}\label{eq:detSA}
   \det S_C = dy + 2axu - x^2 - a^2y - du^2 \ge 0. 
\end{equation}
Since $x = C(a,u) \ge C(a,a) = d$ and $d = aa^* \ge au$, we have
$$x+d-2au \ge 2d -2au \ge 0,$$
which implies
\begin{equation}\label{eq:audx}
    (x+d-2au)(x-d) = x^2-2axu +2aud - d^2 \ge 0.
\end{equation}
Adding inequalities \eqref{eq:detSA} and \eqref{eq:audx} we obtain
$$dy - a^2y - du^2 + 2aud - d^2 \ge 0,$$
which is equivalent to 
$$ (y-d)(d-a^2) \ge d(u-a)^2. $$
Since $d > a^2$, this implies
$$ y = C(u,u) \ge \frac{d}{d-a^2}(u-a)^2 + d.$$
which finishes the proof.

    \end{cases}$$

<!-- lemma-3: body_md -->
\label{function_f}
Let $u \in (0,1)$, $w \in [u^2, u]$ and $u^* = \tfrac{w-u^2}{w+1-2u}$. Let a function $f \colon [0, u] \to [0,\infty)$ be defined by
\begin{align}
    f(a)=f_{u,w}(a) &= a^2+\tfrac14 \left(\sqrt{w- (u-2a)^2} + \sqrt{w- u^2}\right)^2 \label{eq:f} \\
    &= au + \tfrac12(w-u^2) +\sqrt{(w-u^2)(a(u-a)+\tfrac14(w-u^2))} \nonumber \\
    &= au + \tfrac12(w-u^2) + \tfrac12\sqrt{(w-u^2)(w- (u-2a)^2)}. \nonumber 
\end{align} 
Then $u^* \in [0,u]$,  $f(0) = w- u^2$, $f(u) = w$, $f(u^*) = u^*$, $f$ is twice differentiable, increasing, concave, $f^\prime(a) \in [0,1)$ for every $a \in [u^*, u]$, and $u-a-w+f(a) \ge 0$ for every $a \in [u^*, u]$.
<!-- end lemma-3 -->
<!-- proof-15: body_md -->
A short calculation reveals that all three forms of function $f$ are equal. Let us use the abbreviation $\hat{x} = 1 - x$ for each $x \in [0,1]$.
Since $w \in [u^2, u]$, there is a number $t \in [0,1]$ such that $w = (1-t)u^2+tu$. It follows that
$$w-u^2 = t u (1-u) = t u \hat{u},$$
so that 
$$ u^* = \frac{w-u^2}{w-u^2+(1-u)^2} = \frac{t u \hat{u}}{t u \hat{u}+\hat{u}^2} = 
\frac{t u}{t u +\hat{u}} = \frac{t u}{t u +1-u} = \frac{t u}{1- \hat{t} u}.$$
Since $1- \hat{t} u \ge 1- \hat{t} = t$, we obtain that $u^* \in [0,u]$.
Now, we have 
\begin{align*}
    f(0) &= \tfrac14 \left(2 \sqrt{w- u^2}\right)^2= w-u^2, \text{ and} \\
    f(u) &= u^2+\tfrac14 \left(2 \sqrt{w- u^2}\right)^2 = w.
\end{align*}
Since 
$$ u- u^* = u - \frac{t u}{1- \hat{t} u} = \frac{u- \hat{t} u^2 - t u}{1- \hat{t} u} = 
\frac{\hat{t} u \hat{u}}{1-\hat{t} u} , $$
we obtain from the second form of $f$
\begin{align*}
     f(u^*) &= \frac{tu^2}{1-\hat{t} u}+ \frac12 t u \hat{u} + \sqrt{tu\hat{u} \left( \frac{t u}{1- \hat{t} u} \cdot \frac{\hat{t} u \hat{u}}{1-\hat{t} u}+\frac14 tu\hat{u}\right)} \\
     &= \frac{tu(2u+\hat{u}-\hat{t}u\hat{u})}{2(1-\hat{t} u)}+\frac{t u \hat{u}}{2(1-\hat{t} u)} \sqrt{4 \hat{t} u + (1-\hat{t}u)^2} \\
     &= \frac{tu}{2(1-\hat{t}u)}\left( 2u+\hat{u}-\hat{t}u\hat{u} + \hat{u} (1+\hat{t} u)\right) \\
     &= \frac{t u}{1- \hat{t} u} = u^*. 
\end{align*}
If $w=u^2$ then $f(a) = ua$, so $f'(a) = u \in (0,1)$, $f''(a) = 0$, and 
$$u-a-w+f(a) = u-a -u^2 +ua = (1-u)(u-a)\ge 0$$
for every $a \le u$.

From now on we suppose that $w > u^2$. We differentiate $f$ in the third form, to obtain that 
$$ f^\prime (a) = u + \sqrt{w- u^2} \, \frac{u-2a}{\sqrt{w-(u-2a)^2}} $$
and 
$$ f^{\prime \prime} (a) = - \frac{2 w \sqrt{w- u^2}}{(w-(u-2a)^2)^{3/2}} . $$
Clearly, $f^{\prime \prime} (a) < 0$ for all $a \in [0,u]$, and so the function $f$ is concave,
and the function $f^\prime$ is decreasing.
Since $f^\prime(u) = 0$, we have $f^\prime (a) \ge 0$ for all $a \in [0,u]$, so that 
the function $f$ is increasing. 

                                                                                
Define a function $f_0 \colon [0, u] \to \mathbb{R}$ by $f_0(a) = f(a) - a$. 
Since $f_0^{\prime \prime} (a) = f^{\prime \prime} (a) < 0$ for all $a \in [0,u]$, the function $f_0$ is also concave. 
Since $f_0(0) = f(0) = w- u^2 > 0$, $f_0(u) = f(u) - u = w- u \le 0$ and 
$f_0(u^*) = f(u^*) - u^* = 0$, the concavity of $f_0$ implies that the number $u^*$ is the only zero of $f_0$ and 
$f_0^\prime (a) < 0$ for all $a \in [u^*, u]$. It follows that 
$f^\prime(a)= f_0^\prime(a) + 1 \in [0,1)$ for all $a \in [u^*, u]$.

Finally, define a function $h \colon [u^*, u] \to \mathbb{R}$ by $h(a) = u-a-w+f(a)$.
Then $h^\prime(a) = f^\prime (a) - 1 < 0$ for all $a \in [u^*, u]$ and 
$h(u) = 0$, so that $h(a) \ge 0$ for all $a \in [u^*, u]$.
This completes the proof of the lemma.

<!-- lemma-4: body_md -->
\label{function_g}
Let $u \in (0,1)$, $w \in [u^2, u]$ and $u^{**} = \tfrac{u^2}{w} \in [u,1]$. Let a function $g \colon [u, 1] \to [0,\infty)$ be defined by
\begin{equation} \label{eq:g}
    g(a) =g_{u,w}(a)= au + a - u + \tfrac12(w-u^2) +\sqrt{(w-u^2)((1-a)(a-u)+\tfrac14(w-u^2))}.
\end{equation}
Then $g(u) = w$, $g(u^{**}) = u^{**}$, $g$ is twice differentiable, increasing, $2$-Lipschitz, concave, $g'(a) \in [1,2]$ for every $a \in [u, u^{**}]$, and $u-a-w+g(a) \ge 0$ for every $a \in [u, u^{**}]$.
<!-- end lemma-4 -->
<!-- proof-16: body_md -->
We apply the survival transformation to the function $f$ from Lemma \ref{function_f}.
Let $\hat{a}=1-a$, $\hat{u}=1-u$, and $\hat{w}=w+1-2u$. Then  $\hat{w}-\hat{u}^2=(w+1-2u) - (1-u)^2 = w- u^2$. 
Note that
\begin{align*}
    g_{u,w}(a) &= 2a-1+(1-a)(1-u)+ \tfrac12(w-u^2) + \sqrt{(w-u^2)((1-a)(a-u)+\tfrac14(w-u^2))} \\
    &= 2a-1+\hat{a}\hat{u}+ \tfrac12(\hat{w}-\hat{u}^2) + \sqrt{(\hat{w}-\hat{u}^2)(\hat{a}(\hat{u}-\hat{a})+\tfrac14(\hat{w}-\hat{u}^2))} \\
    &=2a-1+f_{\hat{u},\hat{w}}(\hat{a})= 2a-1+f_{1-u,w+1-2u}(1-a).
\end{align*}
Furthermore, let $u^* = \tfrac{\hat{w}-\hat{u}^2}{\hat{w}+1-2\hat{u}}$. Then 
\begin{equation}\label{eq:1-u*}
1-u^*=1-\frac{w+1-2u-(1-u)^2}{w+1-2u + 1- 2(1-u)} = 1-\frac{w-u^2}{w} = \frac{u^2}{w}=u^{**} .
\end{equation}
Now, the properties of the function $g$ follow easily from the corresponding properties 
of the function $f$ from Lemma \ref{function_f}, and so we omit the details.

\end{cases}$$
Furthermore, $C(a,a) = a$ for every $a \in [0, u^*]$ and $C(u,u) = w$.

Similarly, by Proposition \ref{prop:real_zgornja2} with $a=u^{**}$, there there exists a checkerboard copula $B$ with splitting points $\underline{\Delta} = \delta_2' \times \delta_2'$, where $\delta_2' = \{0, u, u^{**}, 1\} $, such that $B = \Psi(B_1)$ for some checkerboard copula $B_1$, $B(u^{**},u^{**}) = g(u^{**}) = u^{**}$ and $B(u,u) = w$. We have  $B(x, u^{**}) = B(u^{**},x) = u^{**}$ for any $x \in [u^{**}, 1]$, implying that $B$ is an ordinal sum of two copulas with respect to $\{(0,u^{**}),(u^{**},1)\}$. If we let 
$$ D(x,y) = \begin{cases}
    \min\{x,y\}; &\text{if } x\ge u^{**}, y\ge u^{**}, \\
    B(x,y); &\text{otherwise, }
\end{cases}$$
then $D(a,a) = a$ for every $a \in [u^{**}, 1]$, $D(u,u) = w$, and, similarly as above, $D = \Psi(D_1)$ for some copula $D_1$.

Let $C$ be a copula such that $C = \Psi(C_1)$ for some copula $C_1$ and $u \in (0,1)$. Let $w = C(u,u) \in [u^2, u]$ be the value of $C$ at a diagonal point, let $u^* = \tfrac{w-u^2}{w+1-2u} \in [0,u]$ and $u^{**} = \tfrac{u^2}{w} \in [u,1]$. Then the upper bound for the diagonal section of $C$ is 
    $$C(a,a) \le \begin{cases}
			f(a); & \text{if }u^* \le a \le u, \\

			a; & \text{otherwise},
    \end{cases}$$
    where functions $f$ and $g$ are defined in equations \eqref{eq:f} and \eqref{eq:g}.
    Every point on the upper bound is attained by some copula $C$ of the form $C = \Psi(C_1)$ for some copula $C_1$ and $C(u,u) = w$.
