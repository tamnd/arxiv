<!-- theorem-1: body_md -->
\label{thm:main}
Let $X$ be a connected compact K\"ahler manifold with semi-ample
canonical bundle, and let $[{\omega}]$  be any K\"ahler class.
For all sufficiently small $\varepsilon>0$, let
$\omega_\varepsilon\in 2\pi c_1(K_X)+\varepsilon [{\omega}] $ be the unique cscK metric. If $m=0$, the metrics converge to a point in the Gromov-Hausdorff topology. If $m>0$, then:\begin{enumerate}
    \item For any compact $K\Subset Y^\circ$, 
    \begin{equation}
        \|\omega_\varepsilon-(f^*\omega_{\rm can}+\varepsilon\omega_{\rm SRF})\|_{f^*\omega_{\rm can}+\varepsilon{\omega}}\to 0 \quad \text{uniformly on }f^{-1}(K).
    \end{equation}
    \item Let $d_{\omega_{\varepsilon}}$ be {the distance function induced by the metric} $\omega_{\varepsilon}$ on $X$, \begin{equation}\label{eq:gh}
(X,d_{\omega_\varepsilon})
 \longrightarrow (\overline{Y^\circ},d_{\rm can})
 \qquad\text{in the Gromov--Hausdorff topology}.
 \end{equation}
\end{enumerate}

\label{thm:sobolev}
Let ${(X,\omega)}$ be a fixed compact K\"ahler manifold. For a K\"ahler form ${\eta}$, set
\[
\color{black}
 V_\eta=\int_X\eta^n,\qquad d\mu_\eta=\frac{\eta^n}{V_\eta},\qquad
 I_\eta=[\eta]\cdot[\omega]^{n-1},\qquad V_\omega=\int_X\omega^n.
\]
Fix ${p>n}$ and constants ${K,c>0}$. Suppose that
\[
\color{black}
 \int_X\left|\log\frac{\eta^n/V_\eta}{\omega^n/V_\omega}\right|^p d\mu_\eta\leq K,
 \qquad \frac{\eta^n/V_\eta}{\omega^n/V_\omega}\geq c.
\]
Then there exist constants ${q=q(n,p)>1}$ and ${C_S>0}$, depending only on the fixed background, ${n,p,K,c}$, and ${q}$, such that, for every ${u\in W^{1,2}(X,\eta)}$,
\[
\color{black}
 \left(\int_X|u-\bar u_\eta|^{2q}d\mu_\eta\right)^{1/q}
 \leq C_S I_\eta\int_X|\nabla u|_\eta^2d\mu_\eta,
 \qquad \bar u_\eta=\int_Xu\,d\mu_\eta.
\]

<!-- lemma-1: body_md -->
\label{lem:trace-lp}
Let $\Phi:X\to Z$ be a holomorphic map from a compact complex manifold {$X$}
to a fixed compact K\"ahler manifold $(Z,\omega_Z)$, and put
$\chi=\Phi^*\omega_Z$. Suppose that $\omega_\varphi$, $F$, and a constant
$\underline R$ satisfy
\begin{equation}\label{eq:ht1}
 \operatorname{Ric}(\omega_\varphi)=-\chi-dd^cF,\qquad
 \Delta_\varphi F=-\underline R-\operatorname{tr}_\varphi\chi.
\end{equation}
If $|\underline R|+\operatorname{osc}_X F\leq C_0$, then, for every
finite $p\geq1$,
\begin{equation}\label{eq:ht2}
\int_X(\operatorname{tr}_\varphi\chi)^p\,d\mu\leq C_p.
\end{equation}
The constant {$C_p$} depends only on $p,C_0$ and an upper bound for the
holomorphic bisectional curvature of $\omega_Z$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Integration of \eqref{eq:ht1} gives
$-\underline R=\int_X\operatorname{tr}_\varphi\chi\,d\mu\geq0$.
{Choose source coordinates satisfying} \eqref{eq:estimate-coordinates} {at} $x_0${, and normal target coordinates at} $\Phi(x_0)$, and write
$\omega_Z=\sqrt{-1}h_{\alpha\bar\beta}dw^\alpha\wedge d\bar w^\beta$.
Then
\[
 \chi_{i\bar j}=\sum_{\alpha,\beta}
 h_{\alpha\bar\beta}(\Phi)\Phi_i^\alpha\overline{\Phi_j^\beta},
\qquad \operatorname{tr}_\varphi\chi=\sum_{i,\alpha}|\Phi_i^\alpha|^2 =\sum_i\lambda_i.
\]
With $R_{i\bar j k\bar l}=-\partial_k\partial_{\bar l}
g_{\varphi,i\bar j}$ at the normal point, differentiation gives
\begin{align}
 (\operatorname{tr}_\varphi\chi)_k&=\sum_{i,\alpha}\Phi_{ik}^\alpha\overline{\Phi_i^\alpha},
       \label{eq:trace-first}\\
 \Delta_\varphi\operatorname{tr}_\varphi\chi
 &=\sum_{i,k,\alpha}|\Phi_{ik}^\alpha|^2
   +\sum_i\lambda_i\operatorname{Ric}(\omega_\varphi)_{i\bar i}-\sum_{i,k,\alpha,\beta,\gamma,\delta}R^{\omega_Z}_{\alpha\bar\beta\gamma\bar\delta}\Phi_i^\alpha\overline{\Phi_i^\beta}\Phi_k^\gamma\overline{\Phi_k^\delta}.
       \label{eq:map-bochner-coordinates}
\end{align}
By Cauchy--Schwarz inequality,
\[
|\nabla_\varphi\operatorname{tr}_\varphi\chi|_\varphi^2
\leq(\operatorname{tr}_\varphi\chi)\sum_{i,k,\alpha}|\Phi_{ik}^\alpha|^2.\]
By \eqref{eq:ht1},
\[\sum_i\lambda_i\operatorname{Ric}(\omega_\varphi)_{i\bar i}=-\sum_i\lambda_i^2-\sum_i\lambda_iF_{i\bar i}.
\]
Choose $K\geq1$ so that the bisectional curvature of $\omega_Z$ is at most
$K-1$. Raising both indices of $\chi$ with $g_\varphi$, we obtain
\begin{equation}\label{eq:ht3}
 \Delta_\varphi\operatorname{tr}_\varphi\chi
 \geq\sum_{i,k,\alpha}|\Phi_{ik}^\alpha|^2
       -K(\operatorname{tr}_\varphi\chi)^2-\sum_{i,j}\chi^{i\bar j}F_{i\bar j}.
\end{equation}
Here $\chi^{i\bar j}=g_\varphi^{i\bar b}\chi_{a\bar b}
g_\varphi^{a\bar j}$, so $\chi^{i\bar j}=\lambda_i\delta_{ij}$ at
the normal point.

For $0<\delta\leq1$, put $v=\operatorname{tr}_\varphi\chi+\delta$. Since $v>0$,
\begin{equation}\label{eq:ht4}
 \Delta_\varphi v\geq\frac{|\nabla_\varphi v|_\varphi^2}{v}
       -Kv^2-\sum_{i,j}\chi^{i\bar j}F_{i\bar j},\qquad
 \Delta_\varphi F=-\underline R+\delta-v.
\end{equation}
The closedness of $\chi$ gives, at the normal point,
\begin{equation}\label{eq:ht5}
\sum_i\nabla_{\varphi,i}\chi^{i\bar{j}}=\sum_i\nabla_{\varphi,i}\chi_{j\bar i}=\sum_i\nabla_{\varphi,j}\chi_{i\bar i}=v_j.
\end{equation}

Fix $p\geq1$, and choose $b\in \mathbb R$ later. Testing \eqref{eq:ht4} with
$e^{bF}v^{p-1}$,
\begin{align*}
 \int_Xe^{bF}v^{p-1}\Delta_\varphi v\,d\mu
 &=-\int_Xe^{bF}v^p\sum_i\left[
 (p-1)\left|\frac{v_i}{v}\right|^2+b\operatorname{Re}\left(\frac{v_i}{v}F_{\bar i}\right)\right]d\mu,\\
 \int_Xe^{bF}v^{p-2}|\nabla_\varphi v|_\varphi^2\,d\mu
&=\int_Xe^{bF}v^p\sum_i\left|\frac{v_i}{v}\right|^2d\mu,\\
 -\int_Xe^{bF}v^{p-1}\chi^{i\bar j}F_{i\bar j}\,d\mu
 &=\operatorname{Re}\int_X
 \nabla_{\varphi,i}(e^{bF}v^{p-1}\chi^{i\bar j})F_{\bar j}\,d\mu.
\end{align*}
Repeated indices in the last line are summed. Expanding its derivative
and using \eqref{eq:ht5} gives
\[
\begin{aligned}
 &\operatorname{Re}\sum_{i,j}
 \nabla_{\varphi,i}(e^{bF}v^{p-1}\chi^{i\bar j})F_{\bar j}\\
 &\quad=e^{bF}v^p\sum_i\left[
 b\frac{\lambda_i}{v}|F_i|^2
 +(p-1)\frac{\lambda_i}{v}
              \operatorname{Re}\left(\frac{v_i}{v}F_{\bar i}\right)
 +\operatorname{Re}\left(\frac{v_i}{v}F_{\bar i}\right)\right].
\end{aligned}
\]
Consequently,
\begin{equation}\label{eq:ht6}
\begin{split}
 &\int_Xe^{bF}v^p\sum_i\left[
 p\left|\frac{v_i}{v}\right|^2
 +\left((p-1)\frac{\lambda_i}{v}+b+1\right)
             \operatorname{Re}\left(\frac{v_i}{v}F_{\bar i}\right)
 +b\frac{\lambda_i}{v}|F_i|^2\right]d\mu\\
 &\hspace{45mm}\leq K\int_Xe^{bF}v^{p+1}\,d\mu.
\end{split}
\end{equation}
Testing the equation $\Delta_\varphi F=-\underline{R}+\delta-v$ with $e^{bF}v^p$ similarly gives
\begin{equation}\label{eq:ht7}
\begin{split}
 &\int_Xe^{bF}v^p\sum_i\left[
 p\operatorname{Re}\left(\frac{v_i}{v}F_{\bar i}\right)+b|F_i|^2
 \right]d\mu\\
 &\qquad=\int_Xe^{bF}v^{p+1}\,d\mu
       -(-\underline R+\delta)\int_Xe^{bF}v^p\,d\mu.
\end{split}
\end{equation}

Subtract $\Lambda$ times \eqref{eq:ht7} from \eqref{eq:ht6}:
\begin{equation}\label{eq:trace-combination}
\begin{split}
 \int_Xe^{bF}v^p\mathcal Q\,d\mu
 +(\Lambda-K)\int_Xe^{bF}v^{p+1}\,d\mu
 \leq\Lambda(-\underline R+\delta)\int_Xe^{bF}v^p\,d\mu,
\end{split}
\end{equation}
where
\[
\begin{split}
 \mathcal Q=\sum_i\bigg[
 &p\left|\frac{v_i}{v}\right|^2
 +\left((p-1)\frac{\lambda_i}{v}+b+1-\Lambda p\right)
        \operatorname{Re}\left(\frac{v_i}{v}F_{\bar i}\right)\\
 &+b\left(\frac{\lambda_i}{v}-\Lambda\right)|F_i|^2\bigg].
\end{split}
\]
We seek $\Lambda>K$ and $\mathcal Q\geq0$. At the endpoint
$\lambda_i/v=1$, requiring the summand to be
$p|v_i/v-\alpha F_i|^2$ gives
\[
 \begin{cases}
 p+b-\Lambda p=-2p\alpha,\\
 b(1-\Lambda)=p\alpha^2.
 \end{cases}
 \quad\Longrightarrow\quad
 (\Lambda-1-\alpha)^2=0,
 \qquad \Lambda=\alpha+1,\quad b=-p\alpha.
\]
For these choices, set $u=e^{-\alpha F}v$. Then
\[
 (\log u)_i=\frac{v_i}{v}-\alpha F_i,
 \qquad 0\leq\frac{\lambda_i}{v}\leq1,
\]
and the coefficients of the remaining terms satisfy
\[
\begin{aligned}
 (p-1)\frac{\lambda_i}{v}+b+1-\Lambda p
 &=-2p\alpha-(p-1)\left(1-\frac{\lambda_i}{v}\right),\\
 b\left(\frac{\lambda_i}{v}-\Lambda\right)
 &=p\alpha^2+p\alpha\left(1-\frac{\lambda_i}{v}\right).
\end{aligned}
\]
Substitution, followed by $v_i/v=(\log u)_i+\alpha F_i$, yields
\[
\begin{aligned}
 \mathcal Q
 &=\sum_i p\left|\frac{v_i}{v}-\alpha F_i\right|^2
 +\sum_i\left(1-\frac{\lambda_i}{v}\right)
 \left[-(p-1)\operatorname{Re}\left(\frac{v_i}{v}F_{\bar i}\right)
          +p\alpha|F_i|^2\right]\\
 &=\sum_i p|(\log u)_i|^2
 +\sum_i\left(1-\frac{\lambda_i}{v}\right)
 \left[-(p-1)\operatorname{Re}((\log u)_iF_{\bar i})
          +\alpha|F_i|^2\right].
\end{aligned}
\]
For each $i$, the last bracket equals
\[
 \alpha\left|F_i-\frac{p-1}{2\alpha}(\log u)_i\right|^2
 -\frac{(p-1)^2}{4\alpha}|(\log u)_i|^2.
\]
Hence one gets the complete square identity
\begin{equation}\label{eq:ht8}
\begin{split}
 \mathcal Q
 ={}&\sum_i\alpha\left(1-\frac{\lambda_i}{v}\right)
       \left|F_i-\frac{p-1}{2\alpha}(\log u)_i\right|^2\\
 &+\sum_i\left[p-\frac{(p-1)^2}{4\alpha}
                  \left(1-\frac{\lambda_i}{v}\right)\right]
                  |(\log u)_i|^2.
\end{split}
\end{equation}
Choose
\[
 \alpha=K+p,\qquad \Lambda-K=p+1,\qquad
 p-\frac{(p-1)^2}{4\alpha}\left(1-\frac{\lambda_i}{v}\right)
 \geq\frac p2.
\]
Thus \eqref{eq:trace-combination} gives
\begin{equation}\label{eq:ht9}
\begin{split}
 &\frac p2\int_Xe^{-p\alpha F}v^p
                   |\nabla_\varphi\log u|_\varphi^2\,d\mu
 +(p+1)\int_Xe^{-p\alpha F}v^{p+1}\,d\mu\\
 &\qquad\leq(\alpha+1)(-\underline R+\delta)
                       \int_Xe^{-p\alpha F}v^p\,d\mu.
\end{split}
\end{equation}

H\"older's inequality gives
\[
 \int_Xe^{-p\alpha F}v^p\,d\mu
 \leq\left(\int_Xe^{-p\alpha F}v^{p+1}\,d\mu\right)^{p/(p+1)}
      \left(\int_Xe^{-p\alpha F}\,d\mu\right)^{1/(p+1)}.
\]
Dropping the first nonnegative term in \eqref{eq:ht9} yields
\[
 \int_Xe^{-p\alpha F}v^{p+1}\,d\mu
 \leq\left(\frac{(\alpha+1)(-\underline R+\delta)}{p+1}\right)^{p+1}
       \int_Xe^{-p\alpha F}\,d\mu.
\]
Since $\mu(X)=1$, multiplying by $e^{p\alpha\sup F}$ gives
\begin{equation}\label{eq:ht10}
 \int_X(\operatorname{tr}_\varphi\chi+\delta)^{p+1}\,d\mu
 \leq\left(\frac{(\alpha+1)(-\underline R+\delta)}{p+1}\right)^{p+1}
       e^{p\alpha\operatorname{osc}F},\qquad \alpha=K+p.
\end{equation}
Let $\delta\downarrow0$. This proves \eqref{eq:ht2} for $p\geq2$;
for $1\leq p<2$, use $\|\operatorname{tr}_\varphi\chi\|_p\leq
\|\operatorname{tr}_\varphi\chi\|_2$.

<!-- lemma-2: body_md -->
\label{lem:gradient}
Under the same assumptions in Lemma~\ref{lem:trace-lp},
\begin{equation}\label{eq:c-18b}
 \sup_X|\nabla_\varphi F|_\varphi^2\leq C'.
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Put $w=|\nabla_\varphi F|_\varphi^2$. The Bochner formula gives 
\begin{equation}\label{eq:gradient-bochner}
 \Delta_\varphi w
 =\sum_{i,j}(|F_{ij}|^2+|F_{i\bar j}|^2)-2\operatorname{Re}\sum_i(\operatorname{tr}_\varphi\chi)_iF_{\bar i}-\sum_i\lambda_i|F_i|^2
 -\sum_{i,j}F_{i\bar j}F_jF_{\bar i}.
\end{equation}
Direct computation gives
\[w_i=\sum_j(F_{ji}F_{\bar j}+F_jF_{i\bar j}),\]
 \[e^{-bF}\Delta_\varphi e^{bF}=b(-\underline R-\operatorname{tr}_\varphi\chi)+b^2w,\]
 \[2e^{-bF}\operatorname{Re}\sum_i(e^{bF})_iw_{\bar i}=2b\operatorname{Re}\sum_{i,j}F_{ij}F_{\bar i}F_{\bar j}
   +2b\sum_{i,j}F_{i\bar j}F_jF_{\bar i},\]
\[ \sum_{i,j}|F_{ij}|^2
+2b\operatorname{Re}\sum_{i,j}F_{ij}F_{\bar i}F_{\bar j}+b^2w^2=\sum_{i,j}|F_{ij}+bF_iF_j|^2.
\]
Thus
\[
\begin{split}
 e^{-bF}\Delta_\varphi(e^{bF}w)
 =&\sum_{i,j}|F_{ij}+bF_iF_j|^2+\sum_{i,j}|F_{,i\bar j}|^2+(2b-1)\sum_{i,j}F_{i\bar j}F_jF_{\bar i}-2\operatorname{Re}\sum_i(\operatorname{tr}_\varphi\chi)_iF_{\bar i}\\
 &-\sum_i\lambda_i|F_i|^2+b(-\underline R-\operatorname{tr}_\varphi\chi)w.
\end{split}
\]
The mixed Hessian has no sign; its coefficient vanishes for
$b=\frac12$. This is \cite[equations~(4.3)--(4.4)]{CC17}, with
reference Ricci form $-\chi$. Set
\[
 u=1+e^{F/2}|\nabla_\varphi F|_\varphi^2.
\]
Then
\begin{equation}\label{eq:gradient-weighted-exact}
\begin{split}
 \Delta_\varphi u
 =&e^{F/2}\sum_{i,j}\left(|F_{ij}+\tfrac12F_iF_j|^2+|F_{i\bar j}|^2\right)-2e^{F/2}\operatorname{Re}\sum_i(\operatorname{tr}_\varphi\chi)_iF_{\bar i}-e^{F/2}\sum_i\lambda_i|F_i|^2\\
 &+\tfrac12(-\underline R-\operatorname{tr}_\varphi\chi)(u-1).
\end{split}
\end{equation}
Since $0\leq\lambda_i\leq\operatorname{tr}_\varphi\chi$, dropping the Hessian
squares gives
\begin{equation}\label{eq:c-18c}
 \Delta_\varphi u \geq-2e^{F/2}\operatorname{Re}\langle\nabla_\varphi\operatorname{tr}_\varphi\chi,\nabla_\varphi F\rangle_\varphi
 -\left(\tfrac12\underline R+\tfrac32\operatorname{tr}_\varphi\chi\right)(u-1).
\end{equation}

For $p\geq2$, testing \eqref{eq:c-18c} with $u^{p-1}$ gives
\begin{equation}\label{eq:gradient-tested}
\begin{split}
 &(p-1)\int_Xu^{p-2}|\nabla_\varphi u|_\varphi^2\,d\mu\\
 &\quad\leq2\int_Xe^{F/2}u^{p-1}\operatorname{Re}\langle\nabla_\varphi\operatorname{tr}_\varphi\chi,\nabla_\varphi F\rangle_\varphi\,d\mu\\
&\qquad+\int_X\left(\tfrac12\underline R+\tfrac32\operatorname{tr}_\varphi\chi\right)
                      (u-1)u^{p-1}\,d\mu.
\end{split}
\end{equation}
For the first integral on the right,
\begin{equation}\label{eq:gradient-q-parts}
\begin{split}
 &2\int_Xe^{F/2}u^{p-1}\operatorname{Re}\langle\nabla_\varphi\operatorname{tr}_\varphi\chi,\nabla_\varphi F\rangle_\varphi\,d\mu\\
 &\quad=-2\int_X(\operatorname{tr}_\varphi\chi)e^{F/2}u^{p-1}
                      (-\underline R-\operatorname{tr}_\varphi\chi)\,d\mu\\
 &\qquad-\int_X(\operatorname{tr}_\varphi\chi)(u-1)u^{p-1}\,d\mu\\
 &\qquad-2(p-1)\int_X(\operatorname{tr}_\varphi\chi)e^{F/2}u^{p-2}\operatorname{Re}\langle\nabla_\varphi u,\nabla_\varphi F\rangle_\varphi\,d\mu.
\end{split}
\end{equation}
Because $e^{F/2}|\nabla_\varphi F|_\varphi
=e^{F/4}\sqrt{u-1}\leq C\sqrt u$, Young's inequality gives
\[
\begin{split}
 &2(p-1)\left|\int_X(\operatorname{tr}_\varphi\chi)e^{F/2}u^{p-2}
 \operatorname{Re}\langle\nabla_\varphi u,\nabla_\varphi F\rangle_\varphi\,d\mu\right|\\
 &\quad\leq\frac{p-1}{2}\int_Xu^{p-2}|\nabla_\varphi u|_\varphi^2\,d\mu
       +C(p-1)\int_X(\operatorname{tr}_\varphi\chi)^2u^{p-1}\,d\mu.
\end{split}
\]
The terms without $\nabla_\varphi u$ combine to
\[
\begin{split}
 &2\underline R\int_X(\operatorname{tr}_\varphi\chi)e^{F/2}u^{p-1}\,d\mu
 +2\int_X(\operatorname{tr}_\varphi\chi)^2e^{F/2}u^{p-1}\,d\mu\\
 &\quad+\frac12\int_X(\operatorname{tr}_\varphi\chi)(u-1)u^{p-1}\,d\mu
 +\frac{\underline R}{2}\int_X(u-1)u^{p-1}\,d\mu\\
 &\qquad\leq C\int_X\bigl(1+(\operatorname{tr}_\varphi\chi)^2\bigr)u^p\,d\mu,
\end{split}
\]
using $u\geq1$.
Absorbing the gradient term and using
$|\nabla_\varphi u^{p/2}|_\varphi^2
=\frac{p^2}{4}u^{p-2}|\nabla_\varphi u|_\varphi^2$ yields
\begin{equation}\label{eq:c-18d}
 \int_X|\nabla_\varphi u^{p/2}|_\varphi^2\,d\mu
 \leq Cp^2\int_X\bigl(1+(\operatorname{tr}_\varphi\chi)^2\bigr)u^p\,d\mu.
\end{equation}


Let $\theta=q/[r(q-1)]\in(0,1)$, where $q>1$. H\"older and interpolation inequality give
\begin{equation}\label{eq:iteration-interpolation}
\begin{split}
 \int_X\bigl(1+(\operatorname{tr}_\varphi\chi)^2\bigr)u^p\,d\mu
 &\leq C\|u^{p/2}\|_{L^{2r/(r-1)}}^2
 \leq C\|u^{p/2}\|_{L^{2q}}^{2\theta}\|u^{p/2}\|_{L^2}^{2(1-\theta)}\\
 &\leq\tau\|u^{p/2}\|_{L^{2q}}^2
       +C\tau^{-\theta/(1-\theta)}\|u^{p/2}\|_{L^2}^2,
 \qquad 0<\tau\leq1.
\end{split}
\end{equation}
Apply the uniform Sobolev inequality~\ref{thm:sobolev} to $u^{p/2}$. 
Choosing $\tau=(2Cp^2)^{-1}$ in \eqref{eq:iteration-interpolation}
and absorbing half the left side gives
\begin{equation}\label{eq:iteration-step}
 \|u\|_{L^{pq}}\leq\bigl(Cp^{2/(1-\theta)}\bigr)^{1/p}\|u\|_{L^p},
 \qquad p\geq2.
\end{equation}
For $p_j=2q^j$, both $\sum_jp_j^{-1}$ and
$\sum_jp_j^{-1}\log p_j$ converge. Iteration and interpolation yield
\[
 \|u\|_{L^\infty}\leq C\|u\|_{L^2}
 \leq C\|u\|_{L^\infty}^{1/2}\|u\|_{L^1}^{1/2},
 \qquad \|u\|_{L^\infty}\leq C^2\|u\|_{L^1}.
\]
Finally, by \eqref{eq:ht1},
\[
 \int_Xwd\mu
 =-\int_XF(-\underline R-\operatorname{tr}_\varphi\chi)d\mu
 \leq2|\underline R|\|F\|_{L^\infty},
 \qquad \|u\|_{L^1}\leq1+e^{\|F\|_{L^\infty}/2}\int_Xwd\mu\leq C.
\]
This proves \eqref{eq:c-18b}.

\label{lem:trace}
Under the assumption of Lemma~\ref{lem:trace-lp}, 
\[
 \operatorname{tr}_\varphi\chi\leq C',\qquad \omega_\varphi\geq(C')^{-1}\chi.

Set $v=1+\operatorname{tr}_\varphi\chi$. Taking $\delta=1$, $b=0$ in
\eqref{eq:ht6} gives, for $p\geq2$,
\[
 \int_Xv^p\sum_i\left[
 p\left|\frac{v_i}{v}\right|^2+
 \left((p-1)\frac{\lambda_i}{v}+1\right)
       \operatorname{Re}\left(\frac{v_i}{v}F_{\bar i}\right)\right]d\mu
 \leq K\int_Xv^{p+1}\,d\mu.
\]
Since $0\leq\lambda_i/v\leq1$,
\[
 \left|\sum_i\left((p-1)\frac{\lambda_i}{v}+1\right)
       \operatorname{Re}\left(\frac{v_i}{v}F_{\bar i}\right)\right|
 \leq p\frac{|\nabla_\varphi v|_\varphi}{v}|\nabla_\varphi F|_\varphi
 \leq\frac p2\frac{|\nabla_\varphi v|_\varphi^2}{v^2}+Cp.
\]
Absorption gives
\[
 \frac p2\int_Xv^{p-2}|\nabla_\varphi v|_\varphi^2\,d\mu
 \leq K\int_Xv^{p+1}\,d\mu+Cp\int_Xv^p\,d\mu,
\]
and hence
\begin{equation}\label{eq:q1}
 \int_X|\nabla_\varphi v^{p/2}|_\varphi^2\,d\mu
 \leq Cp^2\int_X(1+\operatorname{tr}_\varphi\chi)v^p\,d\mu.
\end{equation}
For fixed $r>q/(q-1)$, the coefficient $1+\operatorname{tr}_\varphi\chi$ has a
uniform $L^r$ bound. The interpolation and iteration in
\eqref{eq:iteration-interpolation}--\eqref{eq:iteration-step} therefore give
\[
 \|1+\operatorname{tr}_\varphi\chi\|_{L^\infty}
 \leq C\|1+\operatorname{tr}_\varphi\chi\|_{L^2}\leq C'.
\]
Finally, $\lambda_i\leq\sum_j\lambda_j=\operatorname{tr}_\varphi\chi\leq C'$,
so $\chi\leq C'\omega_\varphi$.

\label{lem:compactness}
Suppose
\[
 \|u_\varepsilon\|_{L^\infty}\leq C,\qquad
 \int_X|\nabla u_\varepsilon|_{\omega_\varepsilon}^2\,d\mu_\varepsilon\leq C.
\]
Then:
\begin{enumerate}
\item\label{item:lem-compactness-1} $\|u_\varepsilon\|_{W^{1,1}(X,{\omega})}\leq C$. Hence every sequence has a subsequence converging in every finite
$L^p(X,\Omega)$.
\item\label{item:lem-compactness-2} For any compact subset $K\Subset Y^\circ$, put
\[
 \bar u_\varepsilon(y)=V_F^{-1}\int_{X_y}u_\varepsilon{\omega_y}^d.
\]
Then
\begin{align*}
 \int_{f^{-1}(K)}|\nabla_Vu_\varepsilon|_{{\omega_y}}\Omega
 &\leq C_K\sqrt\varepsilon,\\
 \int_{f^{-1}(K)}|u_\varepsilon-f^*\bar u_\varepsilon|\Omega
 &\leq C_K\sqrt\varepsilon.
\end{align*}
Here $\nabla_V u$ is the covariant derivative in the fiber direction. 
Hence every limit is constant on almost every smooth connected fiber.
For $d=0$, the fibers on $Y^\circ$ are points and this last assertion
is automatic.
\end{enumerate}
<!-- end lemma-5 -->
<!-- proof-5: body_md -->
1. By Cauchy--Schwarz inequality
\[
 |\nabla u_\varepsilon|_{{\omega}}^2
 \leq|\nabla u_\varepsilon|_{\omega_\varepsilon}^2
                 \operatorname{tr}_{{\omega}}\omega_\varepsilon.
\]
Since $\|F_\varepsilon\|_\infty\le C$, $\mu_\varepsilon, \Omega, {\omega}^n$ are all equivalent,
\begin{align*}
\int_X\operatorname{tr}_{{\omega}}\omega_\varepsilon\,\Omega
 &\leq Cn\int_X\omega_\varepsilon\wedge{\omega}^{n-1}
 =Cn(2\pi c_1(K_X)+\varepsilon [{\omega}])\cdot [{\omega}]^{n-1}\leq C,\\
 \int_X|\nabla u_\varepsilon|_{{\omega}}\Omega
 &\leq C\left(\int_X|\nabla u_\varepsilon|_{\omega_\varepsilon}^2d\mu_\varepsilon\right)^{1/2}
       \left(\int_X\operatorname{tr}_{{\omega}}\omega_\varepsilon\,\Omega\right)^{1/2}\leq C.
\end{align*}
Rellich compactness gives $h_{\varepsilon_j}\to h$ in $L^1$; then
\[
 \|h_{\varepsilon_j}-h\|_{L^p}^p
 \leq(2C)^{p-1}\|h_{\varepsilon_j}-h\|_{L^1}\longrightarrow0.
\]

2. Take $K\Subset K'\Subset Y^\circ$.
For $d>0$,
\[
 [\omega_\varepsilon|_{X_y}]=\varepsilon[{\omega_y}],\qquad
 \int_{X_y}\operatorname{tr}_{{\omega_y}}(\omega_\varepsilon|_{X_y}){\omega_y}^d
 =d\varepsilon V_F.
\]
On $f^{-1}(K')$, $\Omega, {\omega}^d\wedge f^*\chi_Y^m$ are equivalent;
hence fiber integration yields
\[
 \int_{f^{-1}(K)}\operatorname{tr}_{{\omega_y}}(\omega_\varepsilon|_{X_y})\Omega
 \leq C_K\varepsilon.
\]
The dual norm decreases on restriction, so
\begin{align*}
 |\nabla_Vu_\varepsilon|_{{\omega_y}}^2
 &\leq|\nabla h_\varepsilon|_{\omega_\varepsilon}^2
              \operatorname{tr}_{{\omega_y}}(\omega_\varepsilon|_{X_y}),\\
 \int_{f^{-1}(K)}|\nabla_Vu_\varepsilon|_{{\omega_y}}\Omega
 &\leq C_K\left(\int_X|\nabla u_\varepsilon|_{\omega_\varepsilon}^2d\mu_\varepsilon\right)^{1/2}
              \varepsilon^{1/2}\leq C_K\sqrt\varepsilon.
\end{align*}
The smooth compact family of connected fibers has a uniform
$L^1$ Poincar\'e inequality:
\[
 \int_{X_y}|u_\varepsilon-\bar u_\varepsilon(y)|{\omega_y}^d
 \leq C_K\int_{X_y}|\nabla_V u_\varepsilon|_{{\omega_y}}{\omega_y}^d.
\]
Integrating proves the second estimate. Passing to distributions,
$\nabla_V u=0$; connectedness gives fiberwise constancy.

\label{prop:volume}

Every sequence $\varepsilon_j\downarrow0$ has a subsequence, still denoted $\varepsilon_j$, with the following properties.
\begin{enumerate}
\item\label{item:volume-potentials}
\[
F_{\varepsilon_j}\longrightarrow F_\infty,\qquad
\varphi_{\varepsilon_j}\longrightarrow\varphi_\infty

\]

\int_XF_{\varepsilon_j}e^{F_{\varepsilon_j}}\Omega\longrightarrow

\end{enumerate}

Apply Lemmas~\ref{lem:energy} and~\ref{lem:compactness} and a common subsequence. Exponentiation and the function $x\mapsto xe^x$ are Lipschitz on the common bounded range. <!-- lemma-6: body_md -->
\label{lem:am-limit}
\leavevmode
\begin{enumerate}
        \item $E_\varepsilon(\varphi_\varepsilon)\to E_Y(u)$.
        \item $u$ maximizes the functional $E_Y(\varphi)-\int_Y\varphi d\nu$.
        \item $\operatorname{MA}_Y(u)=\nu$.
    \end{enumerate}
<!-- end lemma-6 -->
<!-- proof-7: body_md -->
Fix a smooth $\varphi\in \operatorname{PSH}(Y,\chi_Y)$, one has \[
    E_{\varepsilon}(f^*\varphi)=\frac{1}{V_{\varepsilon}}\int_X\int_0^1f^*\varphi(\chi+\varepsilon{\omega}+tdd^c f^*\varphi)^ndt
    \]
    Since powers of a pull-back form above degree $m$ vanish, uniformly for $t\in [0,1]$, \[
\begin{aligned}
        &\int_X f^*\varphi\,
       \bigl(\chi+tdd^c f^*\varphi+\varepsilon{\omega}\bigr)^n\\
 &\quad=\sum_{k=d}^n\binom nk\varepsilon^k
       \int_X f^*\bigl(\varphi(\chi_Y+tdd^c\varphi)^{n-k}\bigr)
                      \wedge{\omega}^k\\
 &\quad=\binom nd\varepsilon^dV_F
       \int_Y\varphi(\chi_Y+tdd^c\varphi)^m+O_\varphi(\varepsilon^{d+1}),\\
 &V_\varepsilon=\binom nd\varepsilon^dV_FV_Y+O(\varepsilon^{d+1}).
    \end{aligned}
    \]
Here we used that $f_*({\omega}^d)=V_F$, so division cancels the factor
$\binom nd\varepsilon^dV_F$:\[
 \begin{aligned}
 E_\varepsilon(f^*\varphi)
 &=\frac1{V_Y}\int_0^1\int_Y
       \varphi\bigl((1-t)\chi_Y+t\chi_\varphi\bigr)^m\,dt+O_\varphi(\varepsilon)\\
 &=\frac1{(m+1)V_Y}\sum_{j=0}^m
       \int_Y\varphi\chi_\varphi^j\wedge\chi_Y^{m-j}
          +O_\varphi(\varepsilon)
 =E_Y(\varphi)+O_\varphi(\varepsilon).
\end{aligned}
\]
A similar argument also shows that \[
\frac{(\chi+\varepsilon{\omega}+dd^c f^*\varphi)^n}{V_\varepsilon}\to \frac{{\omega}^d\wedge f^*(\chi_Y+dd^c\varphi)^m}{V_F V_Y},
\]
and integration along fiber gives \[
f_*\left(\frac{{\omega}^d\wedge f^*(\chi_Y+dd^c\varphi)^m}{V_F V_Y}\right)=V_Y^{-1}(\chi_Y+dd^c\varphi)^m.
\]
Note that the energy $E_\varepsilon$ is concave, so \[
\begin{aligned}
&E_{\varepsilon}(\varphi_\varepsilon)\leq E_\varepsilon(f^*\varphi)+\int_X(\varphi_\varepsilon-f^*\varphi)\frac{(\chi+\varepsilon{\omega}+dd^c f^*\varphi)^n}{V_\varepsilon}\\
&E_{\varepsilon}(\varphi_\varepsilon)\geq E_\varepsilon(f^*\varphi)+\int_X(\varphi_\varepsilon-f^*\varphi)d\mu_\varepsilon.
\end{aligned}
\]
Taking $\varepsilon\to 0$, we have \[
\begin{aligned}
    &\limsup_{\varepsilon\to 0}E_\varepsilon(\varphi_\varepsilon)\leq E_Y(\varphi)+\int_Y(u-\varphi)\operatorname{MA}_Y(\varphi),\\
    &\liminf_{\varepsilon\to 0}E_\varepsilon(\varphi_\varepsilon)\geq E_Y(\varphi)+\int_Y(u-\varphi)d\nu.
\end{aligned}
\]
Here, we used \[
 \begin{aligned}
\left|\int_X(\varphi_\varepsilon-f^*u)\,d\mu_\varepsilon\right|
 &\leq C\|\varphi_\varepsilon-f^*u\|_{L^1(\Omega)}\longrightarrow0,\\
 \left|\int_X f^*(u-\varphi)
        \bigl(e^{F_\varepsilon}-e^{f^*F}\bigr)\Omega\right|
 &\leq\|u-\varphi\|_\infty
       \|e^{F_\varepsilon}-e^{f^*F}\|_{L^1(\Omega)}\longrightarrow0,
 \end{aligned}
\]Since $\varphi_\varepsilon$ and $F_\varepsilon$ are uniformly bounded. 

Now we choose a sequence of bounded smooth $\varphi_j\in \operatorname{PSH}(\mathbb{CP}^N,\ell^{-1}\omega_{FS})$ that $\varphi_j\downarrow u$. First let $\varepsilon\to 0$ with $j$ fixed. Since $\varphi_j-u\geq 0$, we have \[
 \begin{aligned}
 E_Y(\varphi_j)+\int_Y(u-\varphi_j)\,d\nu
 &\leq\liminf_{\varepsilon\to0}E_\varepsilon(\varphi_\varepsilon)\\
 &\leq\limsup_{\varepsilon\to0}E_\varepsilon(\varphi_\varepsilon)\\
 &\leq E_Y(\varphi_j)+\int_Y(u-\varphi_j)\operatorname{MA}_Y(\varphi_j)
 \leq E_Y(\varphi_j).
 \end{aligned}

\label{lem:entropy-to-metric}
Let $\omega,\widehat\omega$ be cohomologous K\"ahler forms on a compact
complex $n$-fold. Set
$V=\int_X\omega^n$, $\mu=\omega^n/V$, and $\nu=\widehat\omega^n/V$.

\[
 \int_X|\widehat\omega-\omega|_\omega\,d\mu
 \leq2\sqrt{n\operatorname{Ent}_\nu(\mu)},\qquad
 \text{ where }\operatorname{Ent}_\nu(\mu)=\int_X\log\frac{d\mu}{d\nu}d\mu.
\]
\item\label{item:entropy-convergence}
Suppose $\mu_j=\rho_j\Omega$, $\nu_j=\widehat\rho_j\Omega$, where
$C^{-1}\leq\rho_j,\widehat\rho_j\leq C$ and
$\|\rho_j-\widehat\rho_j\|_{L^1(\Omega)}\to0$. Then
\[
 \int_X|\widehat\omega_j-\omega_j|_{\omega_j}\,d\mu_j\to0,
 \qquad
 \int_X|\omega_j-\widehat\omega_j|_{\widehat\omega_j}\,d\nu_j\to0.
\]
For every $\eta>0$, outside a set of $\Omega$-measure tending to zero,
\[
 (1-\eta)\omega_j\leq\widehat\omega_j\leq(1+\eta)\omega_j.
\]
\end{enumerate}

Note the following elementary inequality 
\[
 (\sqrt t-1)^2\leq t-1-\log t,\qquad
 (\sqrt t+1)^2\leq2(t+1),\qquad
 |t-1|=|\sqrt t-1|(\sqrt t+1).
\]
It follows that 
\[
 |\widehat\omega-\omega|_\omega^2
 \leq2(\operatorname{tr}_\omega\widehat\omega+n)
 \left(\operatorname{tr}_\omega\widehat\omega-n-
                    \log\frac{\widehat\omega^n}{\omega^n}\right).
\]
The first factor is bounded due to the cohomology class
\[
\int_X\operatorname{tr}_\omega\widehat\omega d\mu
 =\frac{n}{V}\int_X\widehat\omega\wedge\omega^{n-1}=n,
\]
and 
\[
 \int_X\left(\operatorname{tr}_\omega\widehat\omega-n-
                 \log\frac{\widehat\omega^n}{\omega^n}\right)d\mu
 =\operatorname{Ent}_\nu(\mu).
\]
By Cauchy--Schwarz inequality, one gets (i).

\[
 0\leq\operatorname{Ent}_{\nu_j}(\mu_j)
 \leq C^2\|\rho_j-\widehat\rho_j\|_{L^1(\Omega)}\to0,

and likewise with the measures interchanged. Finally,

Markov's inequality gives the last assertion.

Work over $B_{2r}\Subset Y^\circ$, with base coordinates
$y^1,\ldots,y^m$. For
$a=\sum_{\alpha=1}^m a_\alpha\,dy^\alpha$ with constant coefficients
and $|a|_{\omega_{\rm can}}\leq1$ on $B_{2r}$, set
\[\omega_a=\sqrt{-1}a\wedge\bar a,\qquad u_\varepsilon=\operatorname{tr}_{\omega_\varepsilon}f^*\omega_a,\qquad
u_0=\operatorname{tr}_{\omega_{\rm can}}\omega_a.
\]
We have
\[
 0\leq u_\varepsilon\leq C,\qquad
 \|u_\varepsilon-f^*u_0\|_{L^2(f^{-1}(B_{2r}),\mu_\varepsilon)}
 \longrightarrow0.
\]
Suppress $\varepsilon$ and put
\[
 h=\sum_{\alpha=1}^m a_\alpha(y^\alpha\circ f),\qquad
 f^*a=\partial h,\qquad
f^*\omega_a=\sqrt{-1}\,\partial h\wedge\bar\partial\bar h.
\]
Fix $x\in f^{-1}(B_{2r})$ and choose holomorphic coordinates
$z^1,\ldots,z^n$ normal for $\omega$ at $x$:
\[
 \omega=\sqrt{-1}g_{i\bar j}\,dz^i\wedge d\bar z^j,\qquad
 g_{i\bar j}(x)=\delta_{ij},\qquad
 \partial_k g_{i\bar j}(x)=0.
\]
Write $h_i=\partial_i h$, $h_{ik}=\partial_k\partial_i h$, with
$\sum_jg^{i\bar j}g_{k\bar j}=\delta^i_k$. Since $h$ is holomorphic,
\[
 u=\sum_{i,j}g^{i\bar j}h_i\overline{h_j},\qquad
 \partial_{\bar l}h_i=0,\qquad
 \partial_k\overline{h_j}=0.
\]
Straightforward computation gives  
\[
 \partial_k u
 =\sum_{i,j}(\partial_k g^{i\bar j})h_i\overline{h_j}
  +\sum_{i,j}g^{i\bar j}h_{ik}\overline{h_j}.
\]
\[
 u_{k\bar l}=\sum_{i,j}(\partial_k\partial_{\bar l}g^{i\bar j})h_i\overline{h_j}+\sum_ih_{ik}\overline{h_{il}}.
\]
At 
$x$, we have
\[
 \begin{aligned}
\partial_k\partial_{\bar l}g^{i\bar j}
 &=-\partial_k\partial_{\bar l}g_{j\bar i}
   =R_{j\bar i k\bar l}.
 \end{aligned}
\]
Thus, using $\operatorname{Ric}(\omega)=-\chi-dd^c F$,
\[
 \begin{aligned}
 \Delta_\omega u
 &=\sum_{i,k}|h_{ik}|^2+\sum_{i,j}\operatorname{Ric}(\omega)_{j\bar i}h_i\overline{h_j}\\
 &=\sum_{i,k}|h_{ik}|^2
   -\sum_{i,j}\chi_{j\bar i}h_i\overline{h_j}-\sum_{i,j}F_{j\bar i}h_i\overline{h_j}.
 \end{aligned}
\]
Thus
\begin{equation}\label{eq:sharp-schwarz-bochner}
 \begin{split}
 \Delta_\omega u
 &=|\nabla^{1,0}_\omega(f^*a)|_\omega^2
   -\langle f^*\omega_a,\chi\rangle_\omega\\
 &\qquad-\langle f^*\omega_a,dd^c F\rangle_\omega.
 \end{split}
\end{equation}
It is also \eqref{eq:map-bochner-coordinates} for the holomorphic
map $h$ to flat $\mathbb C$. A unitary change of the normal coordinates
diagonalizes $\chi$, with eigenvalues $\lambda_i\geq0$, so
\[
 0\leq\langle f^*\omega_a,\chi\rangle_\omega=\sum_i\lambda_i|h_i|^2\leq\left(\sum_i\lambda_i\right)\left(\sum_i|h_i|^2\right)
 =(\operatorname{tr}_\omega\chi)u\leq Cu.
\]
In particular,
\[
 \Delta_\omega u\geq-Cu-\langle f^*\omega_a,dd^c F\rangle_\omega.
\]
We retain the square in \eqref{eq:sharp-schwarz-bochner} below.
The smoothness of $u_0$ on the fixed chart gives
\[
 |\nabla f^*u_0|_\omega^2\leq C\operatorname{tr}_\omega\chi\leq C,
 \qquad
 |\Delta_\omega f^*u_0|
 =|\operatorname{tr}_\omega f^*(dd^c u_0)|\leq C\operatorname{tr}_\omega\chi\leq C.
\]
Consequently,
\begin{equation}\label{eq:sharp-schwarz-excess}
 \Delta_\omega(u-f^*u_0)\geq|\nabla^{1,0}_\omega(f^*a)|_\omega^2-C-\langle f^*\omega_a,dd^c F\rangle_\omega.
\end{equation}

Set $v=(u-f^*u_0)_+$, $\xi=\xi_{s,t}$ and
$\psi=\xi^2v^{p-1}$, where $p\geq3$.
All following integrals use $d\mu_\varepsilon$ on $f^{-1}(B_t)$,
and all differential norms use $\omega$. Testing
\eqref{eq:sharp-schwarz-excess} with $\psi$,
\[
 \begin{aligned}
 &(p-1)\int\xi^2v^{p-2}|\nabla v|^2+\int\psi|\nabla^{1,0}_\omega(f^*a)|^2\\
 &\quad\leq2\int\xi v^{p-1}|\nabla\xi|\,|\nabla v| +C\int\psi+\int\psi\langle f^*\omega_a,dd^c F\rangle_\omega.
 \end{aligned}
\]
To estimate the last term, note that at the normal point
\[\nabla_k(f^*\omega_a)_{i\bar j}=h_{ik}\overline{h_j},\qquad
 \nabla_{\bar k}(f^*\omega_a)_{i\bar j}=h_i\overline{h_{jk}}.
\]
Hence
\[
|\nabla(f^*\omega_a)|_\omega\leq Cu^{1/2}|\nabla^{1,0}_\omega(f^*a)|_\omega.
\]
Integration by parts,  and the bounds
$u+|\nabla F|_\omega\leq C$ from Lemma~\ref{lem:gradient} and
the trace estimate in base  give
\[
 \begin{aligned}
\left|\int\psi\langle f^*\omega_a,dd^c F\rangle_\omega\right|
 &\leq C\int\bigl(u|\nabla\psi|+\psi|\nabla(f^*\omega_a)|\bigr)|\nabla F|\\
&\leq\frac12\int\psi|\nabla^{1,0}_\omega(f^*a)|^2+C\int(|\nabla\psi|+\psi).
 \end{aligned}
\]
Here the square is absorbed using
\[
C\psi|\nabla^{1,0}_\omega(f^*a)|\leq\frac12\psi|\nabla^{1,0}_\omega(f^*a)|^2+C\psi.
\]
Since
\[
|\nabla\psi|\leq2\xi v^{p-1}|\nabla\xi|+(p-1)\xi^2v^{p-2}|\nabla v|,
\]
the remaining terms satisfy
\[
 \begin{aligned}
 2\xi v^{p-1}|\nabla\xi|\,|\nabla v|&\leq\frac{p-1}{4}\xi^2v^{p-2}|\nabla v|^2+\frac{C}{p-1}v^p|\nabla\xi|^2,\\
 C\xi v^{p-1}|\nabla\xi|&\leq Cv^p|\nabla\xi|^2+C\xi^2v^{p-2},\\
 C(p-1)\xi^2v^{p-2}|\nabla v|&\leq\frac{p-1}{4}\xi^2v^{p-2}|\nabla v|^2+C(p-1)\xi^2v^{p-2}.
 \end{aligned}
\]
After absorption, the cutoff bound and $0\leq v\leq C$ imply

\end{equation}
The uniform Sobolev inequality therefore yields

\]

$\overline{B_r}$, homogeneity gives

\end{equation}

with repeated indices summed, the Bochner identity is
\begin{equation}\label{eq:resolvent-bochner}
 \Delta_\omega u
 =|H_{ij}|^2+|H_{i\bar j}|^2+2\lambda u+2\operatorname{Re}(G_iH_{\bar i})
      -\chi_{i\bar j}H_jH_{\bar i}
      -F_{i\bar j}H_jH_{\bar i}.

$\bar j$ direction gives the invariant identity
\[
 \begin{split}
 I_\psi&:=\int_X\psi F_{i\bar j}H_jH_{\bar i}\,d\mu\\
 &=-\operatorname{Re}\int_XF_i\bigl(\psi_{\bar j}H_jH_{\bar i}+\psi H_{j\bar j}H_{\bar i} +\psi H_jH_{\bar i\bar j}\bigr)\,d\mu .
 \end{split}
\]
Since $|\nabla F|+|\Delta H|\leq C$,
\begin{equation}\label{eq:resolvent-F-term}

 \leq C\int_X\bigl(u|\nabla\psi|+\psi\sqrt u+\psi\sqrt u\,|H_{ij}|\bigr)\,d\mu.
\end{equation}

The two terms requiring absorption satisfy
\[

 &\leq\tfrac{p-1}{4}v^{p-2}|\nabla v|^2+Cp\,v^p,\\
 Cv^{p-1/2}|H_{ij}|
 &\leq\tfrac12v^{p-1}|H_{ij}|^2+Cv^p.
 \end{split}

Also $\chi_{i\bar j}H_jH_{\bar i}\leq Cu$ and
$|G_iH_{\bar i}|\leq C\sqrt u$.

\eqref{eq:resolvent-F-term}, and retaining the positive Hessian term
therefore gives

The common Sobolev inequality yields

\[\|v\|_{L^2}^2\leq\|v\|_{L^\infty}\|v\|_{L^1},\qquad
 \|v\|_{L^1}=1+\int ud\mu\leq C
\]
by \eqref{eq:total-resolvent-energy}. This bounds $\|v\|_{L^\infty}$.

Let $p_0=1$ and $p_{j+1}=q(p_j+2)$. Applying the last inequality
with $p=p_j+2$ gives
\[
 \|z\|_{L^{p_{j+1}}}\leq[C(p_j+2)^2]^{1/(p_j+2)}\|z\|_{L^{p_j}}^{p_j/(p_j+2)}.
\]
Here
\[
p_j=q^j\left(1+\frac{2q}{q-1}\right)-\frac{2q}{q-1},\qquad
 \prod_{j=0}^{k-1}\frac{p_j}{p_j+2}=\frac{q^k}{p_k}\longrightarrow\alpha .
\]
The product of the constants is bounded because
$\sum_j\log(C(p_j+2)^2)/(p_j+2)<\infty$.
Letting $k\to\infty$ proves the estimate.

\label{lem:convergence-approximation} \leavevmode
\begin{enumerate}

        \item For any $K\Subset Y^\circ$, \[
        \int_{f^{-1}(K)}|\nabla (h_\varepsilon-f^*h)|_{\omega_{\varepsilon}}^2d\mu_\varepsilon\to 0.

For simplicity, put $e_\varepsilon:=h_\varepsilon-f^*h$.

(1)
Let $d_{FS}$ be the ambient Fubini-Study metric restricted to $Y$. By lemma~\ref{lem:trace} and the normalization $f^*\omega_{FS}=\ell\chi,$ \[
    0\leq f^*\omega_{FS}\leq \ell (\operatorname{tr}_{\omega_{\varepsilon}}\chi)\omega_{\varepsilon}\leq \ell C\omega_{\varepsilon}.
    \]
    Therefore, for any cuver $\gamma$ in $X$, aftering taking infumum, \[
    d_{FS}(f(x),f(x'))\leq \inf_\gamma \operatorname{Length}_{FS}(f\circ\gamma)\leq \inf_{\gamma}\sqrt{\ell C}\operatorname{Length}_{\omega_{\varepsilon}}(f\circ\gamma)=\sqrt{\ell C}d_{\omega_{\varepsilon}}(x,x').
    \]
Apply Lemma~\ref{lem:gradient-estimate-on-total-space}, we get $|\nabla h_\varepsilon|_{\omega_{\varepsilon}}\leq C$ for a uniform constant $C$.

Let $r=d_{\omega_{\varepsilon}}(x,x')$, then \[
|e_\varepsilon(x)-e_\varepsilon(x')|\leq Cr+\sup_{\substack{y,y'\in Y\\d_{FS}(y,y')\leq\sqrt{\ell C}\,r}}|h(y)-h(y')|
\]
Since $(\overline{Y^\circ},d_{\rm can})$ is homeomorphic to $(Y,d_{FS})$, $h$ is continuous on $Y$. Hence, by compactness of $Y$, for each $\delta>0$, there exits $r_\delta\in (0,1)$, which is independent of 
$\varepsilon$, such that \[
|e_\varepsilon(x)|\ge \delta \quad\Longrightarrow\quad
 |e_\varepsilon|\geq\delta/2
 \text{ on }B_{\omega_\varepsilon}(x,r_\delta).
\]
By the volume noncollapsing estimate in Theorem~\ref{thm:bounds}, \[
\|e_\varepsilon\|_{L^2(\mu_\varepsilon)}^2\geq\frac{\delta^2}{4}\mu_\varepsilon(B_{\omega_{\varepsilon}}(x,r_\delta))\geq\frac{\delta^2r_\delta^{a_{\rm vol}}}{4C}>0.
\]
However, Lemma~\ref{lem:convergence-approximation} gives $e_\varepsilon\to 0$ in $L^2(\Omega)$, and also in $L^2(\mu_\varepsilon)$ since $|F_\varepsilon|\leq C$. 

Hence $\|e_\varepsilon\|_{L^\infty}\to 0.$ 

(2) Choose $\xi\in C^\infty_c(Y^\circ)$ with $0\leq\xi\leq1$
and $\xi=1$ near $K$; write $\xi$ also for $f^*\xi$.
On $\operatorname{supp}\xi$, one has \[
\Delta_{\omega_{\varepsilon}}e_\varepsilon=\lambda e_\varepsilon+f^*\Delta_{\omega_{\rm can}}h-\Delta_{\omega_{\varepsilon}}h.
\]
$\operatorname{tr}_{\omega_{\varepsilon}}\chi\leq C$ implies $|\nabla\xi|_{\omega_{\varepsilon}}\leq C, |\Delta_{\omega_{\varepsilon}}f^*h|\leq C$, $|\Delta_{\omega_{\varepsilon}}e_\varepsilon|\leq C$. Therefore \[\begin{aligned}
\int_X\xi^2|\nabla e_\varepsilon|_{\omega_{\varepsilon}}^2d\mu_\varepsilon=&-\int_X\xi^2e_\varepsilon\Delta_{\omega_{\varepsilon}}e_\varepsilon d\mu_\varepsilon-2\int_X\xi e_\varepsilon\langle\nabla\xi,\nabla e_\varepsilon\rangle_{\omega_{\varepsilon}}d\mu_\varepsilon\\
\leq& C\|e_{\varepsilon}\|_{L^\infty}+\frac{1}{2}\int_X\xi^2|\nabla e_\varepsilon|_{\omega_{\varepsilon}}^2d\mu_\varepsilon+2\int_Xe_\varepsilon|\nabla\xi|_{\omega_{\varepsilon}}^2d\mu_\varepsilon.
\end{aligned}
\]
As a reusult, \[
\int_{f^{-1}(K)}|\nabla e_\varepsilon|_{\omega_{\varepsilon}}^2d\mu_\varepsilon\leq C(\|e_\varepsilon\|_{L^\infty}+\|e_\varepsilon\|_{L^\infty}^2)\to 0. 
\]

\label{lem:krf-uniform}
There exist $q>1$ and constants independent of $t\geq0$ such that:
\begin{enumerate}
\item\label{item:krf-density-gradient}
\[
 C^{-1}\leq e^{F(t)}\leq C,\qquad\operatorname{tr}_{{\tilde\omega}(t)}\chi+|\nabla F(t)|_{{\tilde\omega}(t)}^2\leq C.
\]
\item\label{item:krf-sobolev}
For every $v\in W^{1,2}(X,{\tilde\omega}(t))$,
\begin{equation}\label{eq:krf-sobolev}
 \left(\int_X|v|^{2q}\,d\mu_t\right)^{1/q}
 \leq C\int_X\bigl(|\nabla v|_{{\tilde\omega}(t)}^2+v^2\bigr)\,d\mu_t.
\end{equation}
\item\label{item:krf-volume-diameter}
For $x\in X$ and $0<r\leq1$,
\begin{equation}\label{eq:krf-volume-diameter}
 \mu_t(B_{{\tilde\omega}(t)}(x,r))\geq C^{-1}r^{2q/(q-1)},\qquad
 \operatorname{diam}(X,{\tilde\omega}(t))\leq C.
\end{equation}
\end{enumerate}

(1) The estimates of Song--Tian
\cite[Lemma~2.1 and Propositions~2.1, 2.2, 3.1]{ST16} give

 \operatorname{tr}_{{\tilde\omega}(t)}\chi\leq C,\qquad
 |\nabla({\dot{\tilde\varphi}}+{\tilde\varphi})|_{{\tilde\omega}(t)}^2\leq C.
\]
The last inequality is \cite[equation~(3.19)]{ST16}, obtained by

                          \int_X\chi^k\wedge{\omega}^{n-k},

\label{lem:krf-resolvent-limit}
As $t\to\infty$,
\[
 h_t\longrightarrow f^*h
 \quad\text{locally uniformly on }X^\circ
 \quad\text{and in }L^p(X,\Omega),\qquad 1\leq p<\infty.
\]
<!-- end lemma-33 -->
<!-- proof-36: body_md -->
Take any sequence $t_j\to\infty$.
On a compact subset of $X^\circ$, the bound
${\tilde\omega}(t_j)\leq C_K{\omega}$ and \eqref{eq:krf-resolvent-bound}
give equicontinuity in the fixed smooth metric.
Arzel\`a--Ascoli and a compact exhaustion give a subsequence
converging locally uniformly to a bounded continuous function.
On each fiber over $K\Subset Y^\circ$,
\[
 \operatorname{osc}_{X_y}h_{t_j}
 \leq C\operatorname{diam}(X_y,{\tilde\omega}(t_j)|_{X_y})\to 0.
\]
Thus the limit is $f^*h_\infty$, for a bounded continuous function
$h_\infty$ on $Y^\circ$. 

For $\psi\in C^\infty_c(Y^\circ)$, integration by parts gives
\[
 \int_X h_{t_j}
   \bigl(\lambda f^*\psi-\Delta_{{\tilde\omega}(t_j)}f^*\psi\bigr)d\mu_{t_j}
 =\int_X f^*(g\psi)d\mu_{t_j}.
\]
All integrands are supported in a fixed compact subset of $X^\circ$.
The local uniform convergence implyies
\[
 \int_Y h_\infty
     (\lambda\psi-\Delta^{\mathbb R}_{\omega_{\rm can}}\psi)d\nu_{\rm can}
 =\int_Y g\psi\,d\nu_{\rm can}.
\]
Interior elliptic regularity gives $h_\infty\in C^\infty(Y^\circ)$.
By Lemma~\ref{lem:removable}, $h_\infty=h$.
Every subsequential limit is therefore $f^*h$, which proves local
uniform convergence for the whole family. Since $h_t,h$ are bounded
and $\Omega(X\setminus X^\circ)=0$, dominated convergence gives the
finite-$L^p$ limits.
