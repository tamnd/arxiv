<!-- lemma-1: body_md -->
\label{lem:odeict}
For every fixed $\tau>0$, the empirical distribution $P_t$ tracks the ODE a.s.
\[
\frac{dP}{dt}=\pi^*(P)-P.
\]
Hence, the limiting set of $P_t$ is an ICT of this ODE a.s.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Using (iv), the update is given by 
\[
P_{t+1}-P_t
=
\gamma_t\bigl(\pi^*(P_t)-P_t+\xi_{t+1}+\eta_{t+1}\bigr),
\qquad
\]
where
\[
\xi_{t+1}=M_{P_t}(x_t,\cdot)-\pi^*(P_t),
\enspace
\eta_{t+1}=I(x_{t+1}=\cdot)-M_{P_t}(x_t,\cdot).
\]
Note that $\eta_{t+1}$ is a bounded martingale difference, and so
$\sum_t \gamma_t\eta_{t+1}$ converges (a.s.) as
$\sum_t\gamma_t^2<\infty$ from (i).

For the Markovian term $\xi_{t+1}$, uniform ergodicity from (iii) implies that
there is a uniformly bounded function $g_P$, Lipschitz-continuous in $P$,
satisfying
\[
g_P(x)-\sum_y M_P(x,y) g_P(y)
=
M_P(x,\cdot)-\pi^*(P).
\]
Thus
\[
\xi_{t+1}
=
g_{P_t}(x_t)-g_{P_t}(x_{t+1})+\zeta_{t+1},
\]
where
\[
\zeta_{t+1}
=
g_{P_t}(x_{t+1})
-
\sum_y M_{P_t}(x_t,y) g_{P_t}(y),
\]
is a bounded martingale difference.
After multiplication by $\gamma_t$, the first two terms telescope up to a
boundary term, which vanishes, and errors of order
$|\gamma_t-\gamma_{t-1}|$ and
$\gamma_t\|P_t-P_{t-1}\|=O(\gamma_t^2)$, which are summable.
Also, $\sum_t\gamma_t\zeta_{t+1}$ converges (a.s.). Hence the accumulated
Markovian noise has vanishing tails.

Since $P\mapsto\pi^*(P)-P$ is Lipschitz, the proof follows from stochastic approximation~\citep{benaim1999dynamics}.

<!-- lemma-2: body_md -->
\label{lem:jacobian}
It holds that
    \[
    \frac{\partial \pi^*}{\partial P(v)} = \frac{\partial \pi^*}{\partial P(v)} M_P + \pi^* \frac{\partial M_P}{\partial P(v)}.
\]
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
We trace the chain of derivatives with respect to $P(v)$:

(i) We differentiate the normalization term $Z_u$ in \eqref{eq:attnweight}:
\[
    \frac{\partial Z_u}{\partial P(v)} = \exp(s_{u,v}).
\]
Applying the quotient rule to  $\alpha_{u,k}$ in
\eqref{eq:attnweight}:
\begin{align*}
    \frac{\partial \alpha_{u,k}}{\partial P(v)} & = \frac{\delta_{k,v} \exp(s_{u,k}) Z_u - P(k)\exp(s_{u,k})\exp(s_{u,v})}{Z_u^2} \\
    & = \frac{\exp(s_{u,v})}{Z_u} (\delta_{k,v} - \alpha_{u,k}).
\end{align*}
Summing over all $k$ to find the shift in the attention output $o_u$:
\begin{equation}
\label{eq:attnoutputpd}
    \frac{\partial o_u}{\partial P(v)} = \sum_{k=1}^V \frac{\partial \alpha_{u,k}}{\partial P(v)} \phi_k = \frac{\exp(s_{u,v})}{Z_u} (\phi_v - o_u).
\end{equation}

(ii) Let $z_{u,w} = \langle o_u, E_{\mathrm{out}}[w] \rangle$ be the logit for token $w$. Its derivative is linear: $\frac{\partial z_{u,w}}{\partial P(v)} = \left\langle \frac{\partial o_u}{\partial P(v)}, E_{\mathrm{out}}[w] \right\rangle$.
The transition probability $M_P(u,w)$ is a softmax over these logits scaled by $\tau$. Applying the standard softmax derivative, the chain rule,
and using \eqref{eq:attnoutputpd}, 
 
\begin{align*}
& \frac{\partial M_P(u, w)}{\partial P(v)} = \sum_{y \in V} \frac{\partial M_P(u, w)}{\partial z_{u,y}} \frac{\partial z_{u,y}}{\partial P(v)} \nonumber \\
&= \sum_{y \in V} \frac{1}{\tau} M_P(u, w) (\delta_{w,y} - M_P(u, y)) \left\langle
\frac{\partial o_u}{\partial P(v)},E_{\rm out}[y]
\right\rangle \nonumber \\
    &= \frac{1}{\tau} M_P(u, w) \left\langle
\frac{\partial o_u}{\partial P(v)},E_{\rm out}[w]
\right\rangle \\
    & \qquad - \sum_{y \in V} M_P(u, y) \left\langle
\frac{\partial o_u}{\partial P(v)},E_{\rm out}[y]
\right\rangle.
\end{align*}
 
Let $\bar{e}_u = \sum_{y \in V} M_P(u, y) E_{\mathrm{out}}[y]$ be the expected output embedding over the entire vocabulary for query $u$.  Rewriting, we obtain
 
  \par
  \vspace{-\parskip} 
  \begingroup
    \small
    \setlength{\abovedisplayskip}{0pt} 
    \setlength{\abovedisplayshortskip}{0pt} 
    \setlength{\belowdisplayskip}{4pt} 
    \setlength{\belowdisplayshortskip}{4pt} 
    
\begin{equation}
\label{eq:partialM}
    \frac{\partial M_P(u, w)}{\partial P(v)} = \frac{1}{\tau} M_P(u, w) \frac{\exp(s_{u,v})}{Z_u} \left\langle
\phi_v-o_u,\,
E_{\rm out}[w]-\bar e_u
\right\rangle.
\end{equation}
 
  \endgroup

(iii) Finally, we derive the Jacobian of the stationary distribution $J_{w,v} = \frac{\partial \pi^*(w)}{\partial P(v)}$. Since $\pi^*$ is the stationary distribution of $M_P$, it satisfies the equilibrium condition $\pi^* = \pi^* M_P$. Differentiating,
\[
    \frac{\partial \pi^*}{\partial P(v)} = \frac{\partial \pi^*}{\partial P(v)} M_P + \pi^* \frac{\partial M_P}{\partial P(v)}.
\]

<!-- lemma-3: body_md -->
\label{lem:critical0}
There exist constants $C,\tau_0<\infty$ such that, for all $\tau\geq\tau_0$,
\[
    \sup_{P\in\Delta^{V-1}}\|J(P)\|\leq \frac{C}{\tau}.
\]
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
By \eqref{eq:attnoutputpd}, $\partial o_u/\partial P(v)$ is uniformly bounded over
$P\in\Delta^{V-1}$.  Since all remaining factors in \eqref{eq:partialM} except $1/\tau$ are bounded on the simplex, we obtain
\[
    \left\|\frac{\partial M_P}{\partial P(v)}\right\| = O\left( \frac{1}{\tau} \right).
\]

As $\tau\to\infty$, every row of $M_P$ converges uniformly in $P$ to the
uniform distribution $u$. Since $\pi^*(P)=\pi^*(P)M_P$,
\[
\|\pi^*(P)-u\|
\le \max_x \|M_P(x,\cdot)-u\|,
\]
so $\pi^*(P)\to u$ uniformly as well.
Hence $I-M_P+\mathbf{1}\pi^* \longrightarrow I$ uniformly in $P$, so
$
    F=(I-M_P+\mathbf{1}\pi^*)^{-1}
$
remains uniformly bounded for all sufficiently large $\tau$. 
Using $c_v^T=\pi^*(\partial M_P/\partial P(v))F$, and since
$\|\pi^*\|$ and $\|F\|$ are bounded, we obtain $\|c_v\| = O(1/\tau)$ for every column $c_v$ of $J$. Since there are only $V$ columns,
$\sup_P\|J(P)\|=O(1/\tau)$.

<!-- theorem-1: body_md -->
\label{thm:critical}
There is a temperature
$\tau_c$ such that $P\mapsto\pi^*(P)$ is a
contraction for all $\tau>\tau_c$.  Hence, it has a unique fixed point
$P^*$, which is globally attracting under the ODE.
<!-- end theorem-1 -->
<!-- proof-4: body_md -->
Choose $\tau_c > \max\{C, \tau_0\}$, where $C, \tau_0$ are
the constants in Lemma~\ref{lem:critical0}. Since the
simplex is convex, $P\mapsto\pi^*(P)$ is  a contraction and has
a unique fixed point $P^*$. Writing the ODE as
$\frac{dP}{dt}=\pi^*(P)-P$, 
 let $e(t)=P(t)-P^*$. Since $\pi^*(P^*)=P^*$ and
using
\[
\|\pi^*(P)-\pi^*(P^*)\|
\le q\|P-P^*\|,
\enspace 
q=\sup_P\|J(P)\|<1,
\]
we have
 
  \par
  \vspace{-\parskip} 
  \begingroup
    \small
    \setlength{\abovedisplayskip}{0pt} 
    \setlength{\abovedisplayshortskip}{0pt} 
    \setlength{\belowdisplayskip}{4pt} 
    \setlength{\belowdisplayshortskip}{4pt} 
    
\begin{equation*}
\frac{1}{2}\frac{d}{dt}\|e(t)\|^2
=
\langle e(t),\pi^*(P(t))-\pi^*(P^*)-e(t)\rangle
\le -(1-q)\|e(t)\|^2.
\end{equation*}
 
  \endgroup

Integrating this inequality,
\[
\|P(t)-P^*\|\le e^{-(1-q)t}\|P(0)-P^*\|,
\]
so $P^*$ is globally attracting.  Since $\|P(t)-P^*\|$ decreases
strictly away from $P^*$, $\{P^*\}$ is the only ICT set of the ODE.

<!-- lemma-4: body_md -->
\label{lem:tointegral}
For any continuous function $f:H_{{\boldsymbol\theta}}\to\mathbb C$,
\[
\sup_{\alpha\in H_{\boldsymbol\theta}}
\left|
\frac1T\sum_{n=1}^T f(\alpha+n{\boldsymbol\theta})
-
\int_{H_{\boldsymbol\theta}}f(\omega)\,d\mu_{\boldsymbol\theta}(\omega)
\right|
\longrightarrow0.
\]
<!-- end lemma-4 -->
<!-- proof-5: body_md -->
Consider $\alpha = 0$.  For an integer vector $m$ of the same dimension as ${\boldsymbol\theta}$, let $\chi_m(\omega)=e^{i\langle m,\omega\rangle}$.  Then, 
\[
\frac1T\sum_{n=1}^T \chi_m(n{\boldsymbol\theta})
\longrightarrow
\begin{cases}
1, & \langle m,{\boldsymbol\theta}\rangle\in2\pi\mathbb Z,\\
0, & \text{otherwise}.
\end{cases}
\]
The average of the same function under the uniform distribution on
$H_{\boldsymbol\theta}$ has the same value. Since finite linear combinations of
$\chi_m$ uniformly approximate every continuous function on
$H_{\boldsymbol\theta}$, the result follows.  The same calculation with 
$\alpha+n{\boldsymbol\theta}$ instead of $n{\boldsymbol\theta}$
differs only by the factor $e^{i\langle m,\alpha\rangle}$, whose
absolute value is one, so the convergence is uniform in $\alpha$.

<!-- lemma-5: body_md -->
\label{lem:piQ}
For every fixed $\eta$, (i) there is a unique family
$\{ \pi_\eta(\omega) \}_{\omega \in H_{\boldsymbol\theta}}$ of
probability distributions such that
$\pi_\eta(\omega+{\boldsymbol\theta})
=
\pi_\eta(\omega) \cdot Q_{\eta,\omega}$
and (ii) for every initial distribution $\lambda$,
\[
\left\|
\lambda \cdot Q_{\eta,\omega}^{[m]}
-
\pi_\eta(\omega+m{\boldsymbol\theta})
\right\|_{\rm tv}
\leq
(1-\beta)^{\lfloor m/W\rfloor}.
\]
<!-- end lemma-5 -->
<!-- proof-6: body_md -->
Fix $\omega$ and any distribution $\lambda$ on
$\Sigma^W$. Consider
\begin{align}
\lambda \cdot 
Q_{\eta,\omega-m{\boldsymbol\theta}} \cdot
Q_{\eta,\omega-(m-1){\boldsymbol\theta}} \cdot
\cdots \cdot
Q_{\eta,\omega-{\boldsymbol\theta}}.
\label{eq:chain}
\end{align}
If a longer product is used, its last $m$ transition matrices are the
same as those in \eqref{eq:chain}, but with a possibly different distribution at
their starting point. By \eqref{eq:tv}, changing that starting distribution
changes the result by at most
$(1-\beta)^{\lfloor m/W\rfloor}$. Hence \eqref{eq:chain} converges as
$m\to\infty$, and its limit does not depend on $\lambda$. Define this
limit to be $\pi_\eta(\omega)$.

Shifting $\omega$ by ${\boldsymbol\theta}$ in \eqref{eq:chain} gives (i). Finally, apply
\eqref{eq:tv} to $\lambda$ and $\pi_\eta(\omega)$; (ii) now follows from (i). 
Uniqueness also follows from the same bound.

Combining \eqref{eq:Qsmooth} with \eqref{eq:tv} gives (i): comparing the products defining
$\pi_\eta(\omega)$ and $\pi_{\eta'}(\omega)$ term by term, a change

$Q_{\eta,\omega}$ in $\omega$ implies continuity of

\[

\eta_t+\frac1{t+1}

O\!\left(\frac{k}{p}\right).
\end{align*}

\[

Q_{\eta_p,\omega_{p+j}}(u,\cdot)\|_{\rm tv}
=
O\!\left(\frac{k}{p}\right).
\]
 
Comparing the two $k$-step products one step at a time, the errors
from replacing $\eta_{p+j}$ by $\eta_p$ add up to $O(k^2/p)$.  The $O(1/t)$ error in the actual one-step transition
accumulates to $O(k/p)$, which is smaller. Hence, conditioned on
$\mathcal F_p$, the distribution of $u_{p+k}$ differs by
$O(k^2/p)$ from the distribution obtained by holding the cache fixed
at $\eta_p$. Lemma~\ref{lem:piQ} thus gives

For a Lipschitz function $\varphi: \mathcal{Z}\to\mathbb R$, the empirical cache
update gives
\[
\eta_{t+1}(\varphi)-\eta_t(\varphi)
=
\frac1{t+1}
\bigl(\varphi(z_{t+1})-\eta_t(\varphi)\bigr).
\]
By Theorem~\ref{thm:multi}, over blocks that grow to infinity but remain short
relative to the current time, the average conditional change is $\Gamma(\eta)(\varphi)-\eta(\varphi)+o(1),$
where $\eta$ is the cache state at the start of the block.  Since the
cache changes by $o(1)$ across such a block and $\Gamma$ is Lipschitz,
we can use the current cache state within the block.  Partitioning any fixed interval of the $s$-time scale into such blocks,
Theorem~\ref{thm:multi} therefore shows that the accumulated difference between the
conditional change and $\Gamma(\eta)-\eta$ tends to zero.

The remaining random term is bounded and has conditional mean zero.
After multiplying by $1/(t+1)$, its weighted sum converges (a.s.) because $\sum_t\frac{1}{(t+1)^2}<\infty$.
The $O(1/t)$ approximation error used in Section~\ref{sec:frozencache} contributes only
$O(1/t^2)$ after multiplication by the step size, and is therefore
summable as well.

Since $\mathcal{Z}$ is compact, controlling finitely many $1$-Lipschitz test
functions up to an arbitrarily small error is enough to control $d_1$; hence the same bounds hold uniformly in the definition of $d_1$.
Consequently, over every fixed interval of the $s$-time scale,
$\widehat\eta$ satisfies
\[
\widehat\eta(s_t+s)
=
\eta_t
+
\int_0^s
\bigl(
\Gamma(\widehat\eta(s_t+r))
-
\widehat\eta(s_t+r)
\bigr)\,dr
+
o(1),
\]
where the error tends to zero uniformly for $0\leq s\leq T$ in the
distance $d_1$.

The exact solution $\eta^{(t)}$ satisfies the same equation without the
$o(1)$ term. Since $\Gamma$ is Lipschitz, their distance is at most
\[

+
(L_\Gamma+1)
\int_0^s
d_1\!\left(

Repeated substitution in this inequality shows that, uniformly for

the RHS vanishes (a.s).

\label{thm:multi-high-temp}
For every transformer satisfying the assumptions of this
section, there is a finite $\tau_c$ such that, for every
$\tau>\tau_c$, $\Gamma_\tau$ has a unique fixed point
$\eta_\tau^*$.  Every solution of
$\frac{d\eta}{ds}=\Gamma_\tau(\eta)-\eta$
converges exponentially to $\eta_\tau^*$, and
$d_1(\eta_t,\eta_\tau^*)\longrightarrow0$
\text{a.s.}

Choose $\delta>0$ small enough that $A(\delta)<1/4$, and choose
$\tau_c$ large such that all the preceding large temperature bounds hold and
$\tau_c > 4 C_\delta$. 
By \eqref{eq:gamma-contraction}, for every $\tau>\tau_c$,
$\Gamma_\tau$ is a contraction under $d_{1,\delta}$; let $\kappa<1$ be its contraction
constant.  Since $\mathcal Z$ is compact, iterating
$\Gamma_\tau$ gives a unique fixed point $\eta_\tau^*$.

Let $\eta_s$ be any solution of the ODE.  Its integral form is
\[
\eta_s
=
e^{-s}\eta_0
+
\int_0^s e^{-(s-r)}\Gamma_\tau(\eta_r)\,dr.
\]
Using $\Gamma_\tau(\eta_\tau^*)=\eta_\tau^*$ and the contraction

\[
d_{1,\delta}(\eta_s,\eta_\tau^*)
\leq
e^{-s}d_{1,\delta}(\eta_0,\eta_\tau^*)
+
\kappa
\int_0^s
e^{-(s-r)}
d_{1,\delta}(\eta_r,\eta_\tau^*)\,dr,
\]
and repeated substitution gives
\begin{align}
d_{1,\delta}(\eta_s,\eta_\tau^*)
& \leq
e^{-(1-\kappa)s}

\label{eq:ode-high-temp-contraction}
\end{align}

$d_{1,\delta}$ and $d_1$ are equivalent, Corollary~\ref{cor:slowcache}

let $\rho:=e^{-(1-\kappa)T}<1$; choose
times $t_0<t_1<\cdots$ so that
