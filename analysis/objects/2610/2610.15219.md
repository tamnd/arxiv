<!-- definition-1: body_md -->
For matrices $\bm{M}$ and $\bm{N}$ with full column rank, $\theta_{\max}(\bm{M},\bm{N})$ denotes the largest principal angle between $\operatorname{Range}(\bm{M})$ and $\operatorname{Range}(\bm{N})$, and
  \begin{equation}
    \sin\theta_{\max}(\bm{M},\bm{N}) = \big\lVert\left(\bm{I} - \mathbb{P}_{\bm{M}}\right)\bm{N}\left(\bm{N}^\top\bm{N}\right)^{-1/2}\big\rVert,\quad \mathbb{P}_{\bm{M}}\coloneqq\bm{M}\left(\bm{M}^\top\bm{M}\right)^{-1}\bm{M}^\top.
  \end{equation}
  Moreover, given a symmetric positive-definite matrix $\bm{S}$, we let $\theta^{\bm{S}}_{\max}(\bm{M},\bm{N})\coloneqq\theta_{\max}(\bm{S}^{1/2}\bm{M},\bm{S}^{1/2}\bm{N})$ denote the largest principal angle between $\operatorname{Range}(\bm{M})$ and $\operatorname{Range}(\bm{N})$ measured in the inner product $\inp{\bm{a}}{\bm{b}}_{\bm{S}}\coloneqq\bm{a}^\top\bm{S}\bm{b}$.

<!-- theorem-1: body_md -->
\label{thm:balanced_truncation_bounds}
  Let $\bm{W}_R = \bm{X} \bm{X}^\top$, $\bm{W}_O = \bm{Y} \bm{Y}^\top$, and $\widetilde{\bm{W}}_O = \widetilde{\bm{Y}}\widetilde{\bm{Y}}^\top$ be factorizations of the reachability, observability, and projected observability Gramians, respectively, with $\widetilde{\bm{Y}} = \bm{\Pi}\bm{Y}$ and $\bm{\Pi} \coloneqq \mathbb{P}_{\bm{X}_0}$, so that $\widetilde{\bm{W}}_O = \bm{\Pi}\bm{W}_O\bm{\Pi}$.
  Likewise, let $\bm{H} = \bm{Y}^\top \bm{X}$ and $\widetilde{\bm{H}} = \widetilde{\bm{Y}}^\top \bm{X}$ be the corresponding Hankel matrices, and let $\sigma_k(\bm{H})$ and $\sigma_k(\widetilde{\bm{H}})$ denote their singular values.
  With $\delta^2_R \coloneqq \lVert \bm{\Pi}_\perp \bm{W}_R \bm{\Pi}_\perp\rVert$ and $\delta^2_O \coloneqq \lVert \bm{\Pi}_\perp \bm{W}_O \bm{\Pi}_\perp\rVert$, where $\bm{\Pi}_\perp = \bm{I} - \bm{\Pi}$ and $\lVert \cdot\rVert$ denotes the spectral norm, the following inequality holds
                    
                                                                                                                                                                              
                  
                                                                                        
  \begin{equation}
    \label{eq:weyl_bound}
    \big\lvert\sigma_k\left(\widetilde{\bm{H}}\right) - \sigma_k\left(\bm{H}\right)\big\rvert \leq \delta_O\delta_R,\quad \forall k \in \{1,2,\ldots,\text{rank}(\bm{H})\}.
  \end{equation}
  Moreover, let $\bm{H} = \bm{U}\bm{\Sigma}\bm{V}^\top$ and $\widetilde{\bm{H}} = \widetilde{\bm{U}}\widetilde{\bm{\Sigma}}\widetilde{\bm{V}}^\top$ be singular value decompositions, and define the Balanced Truncation trial and test matrices as
  \begin{equation}
    \bm{\Phi} = \bm{X}\bm{V}_r\bm{\Sigma}_r^{-1/2},\quad \bm{\Psi} = \bm{Y}\bm{U}_r\bm{\Sigma}_r^{-1/2},
  \end{equation}
  where the subscript $r$ denotes truncation to rank $r$, and let $\widetilde{\bm{\Phi}}$ and $\widetilde{\bm{\Psi}}$ be defined as in \eqref{eq:PhiPsi}.
  Then, provided $\varepsilon\coloneqq \sigma_r(\widetilde{\bm{H}}) - \sigma_{r+1}(\bm{H}) > 0$, and assuming that~$\bm{W}_R$ is positive definite, we have
                    
                            
                                                                                                                                                                             
                  
                                                       
  \begin{equation}
    \label{eq:sin_theta_bound}
    \sin\theta^{\bm{W}_R^{-1}}_{\max}\left(\bm{\Phi},\widetilde{\bm{\Phi}}\right) = \sin\theta_{\max}\left(\bm{V}_r,\widetilde{\bm{V}}_r\right) \leq \frac{\delta_O\delta_R}{\varepsilon},\quad \sin\theta^{\bm{W}_R}_{\max}\left(\bm{\Psi},\widetilde{\bm{\Psi}}\right) = \sin\theta_{\max}\left(\bm{V}_r,\widetilde{\bm{V}}_r\right) \leq \frac{\delta_O\delta_R}{\varepsilon}.
  \end{equation}
<!-- end theorem-1 -->

The proof relies on Weyl's inequality and Wedin's $\sin\theta$ theorem, and it is provided in Appendix~\ref{sec:proof_of_thm_1}.

<!-- theorem-2: body_md -->
\label{thm:minimax}
Let $\bm{W}_R = \bm{X}\bm{X}^\top$ be positive definite with eigenvalues $\lambda_1\geq\dots\geq\lambda_n$ and eigenvectors $\bm{u}_1,\dots,\bm{u}_n$, let $\widetilde{\bm{H}}$ denote the Hankel matrix computed from the projected observability Gramian associated with $\bm{X}_0$, and let $\mathcal{O}_\gamma := \{\bm{W}_O\succeq 0 : \|\bm{W}_O\|\leq\gamma\}$. Then
\begin{equation}
\min_{\mathrm{rank}(\bm{X}_0)\leq \ell}\;\sup_{\bm{W}_O\in\mathcal{O}_\gamma}\;\big\|\bm{H}-\widetilde{\bm{H}}\big\| = \gamma^{1/2}\lambda_{\ell+1}^{1/2},
\end{equation}
and the minimum is attained by any $\bm{X}_0$ with $\operatorname{Range}(\bm{X}_0) = \mathrm{span}\{\bm{u}_1,\dots,\bm{u}_\ell\}$.
<!-- end theorem-2 -->
<!-- proof-2: body_md -->
Let $\operatorname{Range}(\bm{X}_0) = \mathrm{span}\{\bm{u}_1,\dots,\bm{u}_\ell\}$ and $\bm{\Pi} := \mathbb{P}_{\bm{X}_0}$. Then $\bm{\Pi}_\perp\bm{W}_R\bm{\Pi}_\perp = \sum_{k>\ell}\lambda_k\bm{u}_k\bm{u}_k^\top$, so that $\delta_R^2 = \lambda_{\ell+1}$. Since $\delta_O^2 = \|\bm{\Pi}_\perp\bm{W}_O\bm{\Pi}_\perp\|\leq\|\bm{W}_O\|\leq\gamma$, the first half of the proof of Theorem~\ref{thm:balanced_truncation_bounds} gives $\|\bm{H}-\widetilde{\bm{H}}\|\leq\delta_O\delta_R\leq\gamma^{1/2}\lambda_{\ell+1}^{1/2}$ for all $\bm{W}_O\in\mathcal{O}_\gamma$.
This establishes the maximum error attained by letting the range of $\bm{X}_0$ coincide with the dominant $\ell$-dimensional eigenspace of $\bm{W}_R$.
We now show that no other choice of $\bm{X}_0$ with $\mathrm{rank}(\bm{X}_0)\leq \ell$ leads to a smaller maximum error.
Let $\mathrm{rank}(\bm{X}_0)\leq \ell$ and $\bm{\Pi} := \mathbb{P}_{\bm{X}_0}$. Since $\dim\operatorname{Range}(\bm{\Pi}_\perp)\geq n-\ell$, there exists a unit vector $\bm{v}\in\operatorname{Range}(\bm{\Pi}_\perp)\cap\mathrm{span}\{\bm{u}_1,\dots,\bm{u}_{\ell+1}\}$, so that $\bm{\Pi}\bm{v} = \mathbf{0}$ and $\bm{v}^\top\bm{W}_R\bm{v}\geq\lambda_{\ell+1}$. Taking $\bm{W}_O = \gamma\bm{v}\bm{v}^\top\in\mathcal{O}_\gamma$, with factor $\bm{Y} = \gamma^{1/2}\bm{v}$, we have $\widetilde{\bm{Y}}\widetilde{\bm{Y}}^\top = \bm{\Pi}\bm{W}_O\bm{\Pi} = 0$, so that $\widetilde{\bm{H}} = 0$ and
\begin{equation*}
\big\|\bm{H}-\widetilde{\bm{H}}\big\| = \|\bm{H}\| = \gamma^{1/2}\|\bm{v}^\top\bm{X}\| = \gamma^{1/2}\big(\bm{v}^\top\bm{W}_R\bm{v}\big)^{1/2}\geq\gamma^{1/2}\lambda_{\ell+1}^{1/2}.
\end{equation*}
