        &=& \frac{1}{m}\sum_{j=1}^m Q(A) \\
        &=& Q(A),

<!-- lemma-3: body_md -->
\label{prop:entropy-curvature}
    Suppose that~\Cref{ass:convex} and the conditions of~\Cref{lemma:existence-uniqueness-full-pro} hold, and that the first variation in~\cref{eq:pro-self-gibbs} is finite $\Pi$-almost surely.
    Define 
    \begin{equation*}
        g_n(\theta)
        =
        \left.\frac{1}{n}\sum_{i=1}^n\frac{\delta S(P_Q, x_i)}{\delta Q}\right|_{Q=Q_n}(\theta),
    \end{equation*}
    and
    \begin{equation}\label{eq:pro-score-bregman}
        \mathcal{B}_n(Q,Q_n)
        =
        S_n(P_Q)-S_n(P_{Q_n})
        -
        \int_\Theta g_n(\theta)\ensuremath{\,\mathrm{d}}(Q-Q_n)(\theta).
    \end{equation}
    Then $\mathcal{B}_n(Q,Q_n)\geq0$, and for every $Q\in\mathcal{P}_\Pi(\Theta)$ for which the terms below are finite, 
    \begin{multline}\label{eq:pro-entropy-curvature}
        \left\{
        S_n(P_Q)
        +
        \frac{d_{\mathrm{KL}}\left(Q;\,\Pi\right)}{\lambda_n}
        \right\}
        -
        \left\{
        S_n(P_{Q_n})
        +
        \frac{d_{\mathrm{KL}}\left(Q_n;\,\Pi\right)}{\lambda_n}
        \right\}
        =
        \mathcal{B}_n(Q,Q_n)
        +
        \frac{d_{\mathrm{KL}}\left(Q;\,Q_n\right)}{\lambda_n}.
    \end{multline}

<!-- proof-2: body_md -->
By~\cref{eq:pro-self-gibbs}, there exists a constant \(c_n\in\mathbb{R}\) such that, for $\Pi$-almost every $\theta$, 
    \begin{equation}\label{eq:pro-fixed-point-log-density}
        \log\frac{\ensuremath{\,\mathrm{d}}Q_n}{\ensuremath{\,\mathrm{d}}\Pi}(\theta)
        =
        -\lambda_n g_n(\theta)-c_n.
    \end{equation}
    Since \(g_n\) is finite \(\Pi\)-almost surely, \(Q_n\) and \(\Pi\) are mutually absolutely continuous, and hence every \(Q\in\mathcal{P}_\Pi(\Theta)\) is absolutely continuous with respect to \(Q_n\).
    By direct computation, we then have
    \begin{IEEEeqnarray*}{rCl}
        d_{\mathrm{KL}}\left(Q;\,\Pi\right)-d_{\mathrm{KL}}\left(Q_n;\,\Pi\right)
        &=&
        d_{\mathrm{KL}}\left(Q;\,Q_n\right)
        +
        \int_\Theta
        \log\frac{\ensuremath{\,\mathrm{d}}Q_n}{\ensuremath{\,\mathrm{d}}\Pi}(\theta)
        \ensuremath{\,\mathrm{d}}(Q-Q_n)(\theta)
        \\
        &=&
        d_{\mathrm{KL}}\left(Q;\,Q_n\right)
        -
        \lambda_n
        \int_\Theta g_n(\theta)\ensuremath{\,\mathrm{d}}(Q-Q_n)(\theta),
    \end{IEEEeqnarray*}
    where the constant $c_n$ vanishes because $Q$ and $Q_n$ are probability measures.
    Dividing both sides by $\lambda_n$ and adding $S_n(P_Q)-S_n(P_{Q_n})$ establishes~\cref{eq:pro-entropy-curvature}.
    Since the map $Q\mapsto P_Q$ is linear and $P\mapsto S_n(P)$ is convex under~\Cref{ass:convex}, the map $Q\mapsto S_n(P_Q)$ is convex.
    The subgradient inequality \citep[Lemma 4.1]{hu_mean-field_2021} applied at $Q_n$ therefore yields $\mathcal{B}_n(Q,Q_n)\geq0$.

<!-- lemma-4: body_md -->
\label{lemma:bound}
    Let $\mathcal{P}_\Pi(\Theta)=\{Q\in\mathcal{P}(\Theta):Q\ll\Pi\}$.
    Under~\Cref{ass:convex}, for a prior $\Pi\in\mathcal{P}(\Theta)$ and any $Q\in\mathcal{P}_\Pi(\Theta)$ such that $Q^{\otimes m}\ll \Pi^{\otimes m}$,
    \begin{equation*}
        S_{n}(P_{\widebar{Q}_{n,m}}) + \frac{d_{\mathrm{KL}}\left(\widebar{Q}_{n,m};\,\Pi\right)}{\lambda_n} \leq S_n(P_{Q}) + \frac{d_{\mathrm{KL}}\left(Q;\,\Pi\right)}{\lambda_n} + \beta_{n,m}(Q).
    \end{equation*}
    In particular, 
    \begin{equation*}
        0\leq \left\{S_{n}(P_{\widebar{Q}_{n,m}}) + \frac{d_{\mathrm{KL}}\left(\widebar{Q}_{n,m};\,\Pi\right)}{\lambda_n}\right\} - \left\{ S_n(P_{Q_n}) + \frac{d_{\mathrm{KL}}\left(Q_n;\,\Pi\right)}{\lambda_n} \right\} \leq \beta_{n,m}(Q_n).
    \end{equation*}
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
Recall that $\widebar{Q}_{m} = \int \widehat Q_m(\theta_{1:m}) \ensuremath{\,\mathrm{d}} Q_m(\theta_{1:m})$, then by the convexity of $P\mapsto S_n(P)$ for all $Q_m\in\mathcal{P}(\Theta^m)$,
    \begin{IEEEeqnarray}{rCl}
        S_n(P_{\widebar{Q}_{m}}) &=& S_n\left\{P_{\int \widehat Q_m(\theta_{1:m}) \ensuremath{\,\mathrm{d}} Q_m(\theta_{1:m})}\right\} \nonumber \\
        &\leq& \int  S_n\left\{P_{\widehat Q_m(\theta_{1:m})}\right\} \ensuremath{\,\mathrm{d}} Q_m(\theta_{1:m}) \nonumber \\
        &=& \int \mathsf{L}_{n,m}(\theta_{1:m}) \ensuremath{\,\mathrm{d}} Q_m(\theta_{1:m}), \label{eq:Sn-bound}
    \end{IEEEeqnarray}
    where the third line follows by the definition of the finite-particle predictive score.
    Elsewhere, we have by standard information processing results that
    \begin{IEEEeqnarray}{rCl}
        d_{\mathrm{KL}}\left(Q_m;\,\Pi^{\otimes m}\right) &=& d_{\mathrm{KL}}\left\{Q_m;\,\bigotimes_{j=1}^m Q_m^{(j)}\right\} + \sum_{j=1}^m d_{\mathrm{KL}}\left\{Q_m^{(j)};\,\Pi\right\} \nonumber \\
        &\geq& \sum_{j=1}^m d_{\mathrm{KL}}\left\{Q_m^{(j)};\,\Pi\right\} \nonumber \\
        &=& \frac{m}{m} \sum_{j=1}^m d_{\mathrm{KL}}\left\{Q_m^{(j)};\,\Pi\right\} \nonumber \\
        &\geq& m\, d_{\mathrm{KL}}\left\{\frac{1}{m}\sum_{j=1}^m Q_m^{(j)};\,\Pi\right\} \nonumber \\
        &=& m\, d_{\mathrm{KL}}\left(\widebar{Q}_{m};\,\Pi\right),\label{eq:kl-bound}
    \end{IEEEeqnarray}
    where the second line follows from the non-negativity of the Kullback--Leibler divergence, the third multiplies by one, the fourth by the convexity of the Kullback--Leibler divergence in its first argument, and the final line by definition of $\widebar{Q}_{m}$.
    As such, we can combine~\cref{eq:Sn-bound,eq:kl-bound} to write 
    \begin{equation*}
        S_n(P_{\widebar{Q}_{m}}) + \frac{d_{\mathrm{KL}}\left(\widebar{Q}_{m};\,\Pi\right)}{\lambda_n} \leq \int \mathsf{L}_{n,m}(\theta_{1:m}) \ensuremath{\,\mathrm{d}} Q_m(\theta_{1:m}) + \frac{d_{\mathrm{KL}}\left(Q_m;\,\Pi^{\otimes m}\right)}{m\cdot\lambda_n}.
    \end{equation*}

    \end{equation*}
    as desired.

    First define

We first show that the marginal distribution of $\theta_{1:m}$ under $\widetilde Q_{n,m,r}$ is exactly $Q_{n,m}$ in~\cref{eq:replica-log-target}.
Fix $\theta_{1:m}$ and consider a single replicated datum $\tilde x_{i,s}$, before conditioning on its value.
Since $z_{i,s}$ is uniform on $\{1,\ldots,m\}$ and $\tilde x_{i,s}\mid z_{i,s},\theta_{1:m}\sim P_{\theta_{z_{i,s}}}$, the law of total probability gives
\begin{equation*}
    p_{\tilde x_{i,s}\mid\theta_{1:m}}(x)
    =\sum_{j=1}^m\mathrm{pr}(z_{i,s}=j)\,p_{\theta_j}(x)
    =\frac{1}{m}\sum_{j=1}^m p_{\theta_j}(x),
\end{equation*}
that is, the equal-weight mixture density of the $m$ components.
By construction, $\tilde x_{i,1:r} = (\tilde x_{i,1},\ldots,\tilde x_{i,r})$ depend on $\theta_{1:m}$ only through their own labels $z_{i,1:r}$, which are mutually independent; 
hence $\tilde x_{i,1:r}$ are conditionally independent given $\theta_{1:m}$.
Since all $r$ of them are conditional on the same value $x_i$, their joint likelihood contribution is the $r$-fold product of the single-slot term above,
\begin{equation*}
    \prod_{s=1}^r p_{\tilde x_{i,s}\mid\theta_{1:m}}(x_i)
    =
    \left\{\frac{1}{m}\sum_{j=1}^m p_{\theta_j}(x_i)\right\}^{r}.
\end{equation*}
The blocks of replicates for different $i$ are independent of one another by the same argument, so multiplying over $i=1,\ldots,n$ gives the joint likelihood of the entire array of replicated data,
\begin{equation*}
    \prod_{i=1}^n
    \left\{
        \frac{1}{m}\sum_{j=1}^m p_{\theta_j}(x_i)
    \right\}^{r}.
\end{equation*}
Multiplying this likelihood by the prior density $\prod_{j=1}^m\pi(\theta_j)$ gives the unnormalised joint density of $\theta_{1:m}$ and the conditioning event,
\begin{equation*}
    \left\{\prod_{j=1}^m\pi(\theta_j)\right\}
    \prod_{i=1}^n
    \left\{
        \frac{1}{m}\sum_{j=1}^m p_{\theta_j}(x_i)
    \right\}^{r}.
\end{equation*}
Since $r=m\lambda_n/n$, this expression is term-for-term identical to the unnormalised density of $Q_{n,m}$ in \cref{eq:replica-log-target}.

    \begin{equation*}
        (f_\# \mu)(A) = \mu\{f^{-1}(A)\}

    For $\sigma\in\mathfrak S_m$, define
    \begin{equation*}
    T_\sigma(\theta_1,\dots,\theta_m)
    =
    \{\theta_{\sigma(1)},\dots,\theta_{\sigma(m)}\},

    Q_m^{\mathrm{sym}}

        \item for some $a\ge1$ and a constant $K>0$, $\mathsf{d}_2(P_Q,P_{Q^\prime})^a\le K d_{\mathcal{S}}(P_Q,P_{Q^\prime})$ when applying~\Cref{thm:predictive-concentration}, or $\mathsf{d}_2(P_Q,P_{Q^\prime})^a\le K d_{\alpha}(P_Q,P_{Q^\prime})$ when applying~\Cref{thm:predictive-concentration-kl}.
        \label{ass:concentration-iii}
    \end{enumerate*}

<!-- proof-13: body_md -->
Begin by noting that by~\ref{ass:concentration-i} there exists $Q^\star\in\mathcal{P}(\Theta)$ such that $P_{Q^\star} = P_0$, so that the first term in~\Cref{thm:predictive-concentration,thm:predictive-concentration-kl} is zero.
    Then, by~\ref{ass:concentration-iii} and Jensen's inequality, either
    \begin{IEEEeqnarray}{rCl}
        \@ifnextchar[{\@Ecurlwith}{\@Ecurlwithout}[0]{\mathsf{d}_2(P_{\widebar{Q}_{n,m}},\,P_{Q^\star})} 
        &\leq&
        K^{1/a}\@ifnextchar[{\@Ecurlwith}{\@Ecurlwithout}[0]{d_{\mathcal{S}}(P_{\widebar{Q}_{n,m}},\,P_{Q^\star})^{1/a}} \nonumber \\
        &\leq&
        K^{1/a}\left[\@ifnextchar[{\@Ecurlwith}{\@Ecurlwithout}[0]{d_{\mathcal{S}}(P_{\widebar{Q}_{n,m}},\,P_{Q^\star})}\right]^{1/a}, \label{eq:posterior-jensen}
    \end{IEEEeqnarray}
    or
    \begin{IEEEeqnarray*}{rCl}
        \@ifnextchar[{\@Esqrwith}{\@Esqrwithout}[0]{\mathsf{d}_2\{P_{\widebar{Q}_{n,m}^{(\alpha)}},\,P_{Q^\star}\}}
        &\leq&
        K^{1/a}\@ifnextchar[{\@Esqrwith}{\@Esqrwithout}[0]{d_{\alpha}\{P_{\widebar{Q}_{n,m}^{(\alpha)}},\,P_{Q^\star}\}^{1/a}} \\
        &\leq&
        K^{1/a}\left(\@ifnextchar[{\@Esqrwith}{\@Esqrwithout}[0]{d_{\alpha}\{P_{\widebar{Q}_{n,m}^{(\alpha)}},\,P_{Q^\star}\}}\right)^{1/a},
    \end{IEEEeqnarray*}
    where the second inequality in each line follows from the concavity of the map $x\mapsto x^{1 / a}$ for $a \ge 1$.
    For simplicity, we shall henceforth just argue in the first case; all arguments apply verbatim to the second.
    For a sequence $M_n\rightarrow\infty$ such that $M_n\gamma_n^{1/a}=o(1)$, by Markov's inequality
    \begin{flalign*}
    P_0\left\{\mathsf{d}_2(P_{\widebar{Q}_{n,m}},\,P_{Q^\star})>M_n\gamma_n^{1/a}\right\}
    &\le \frac{\@ifnextchar[{\@Ecurlwith}{\@Ecurlwithout}[0]{\mathsf{d}_2(P_{\widebar{Q}_{n,m}},\,P_{Q^\star})}}{M_n\gamma_n^{1/a}}\\
    &\le \frac{K^{1/a}\left[\@ifnextchar[{\@Ecurlwith}{\@Ecurlwithout}[0]{d_{\mathcal{S}}(P_{\widebar{Q}_{n,m}},\,P_{Q^\star})}\right]^{1 / a}}{M_n\gamma_n^{1/a}} \lesssim \frac{1}{M_n},
    \end{flalign*}
    where the second inequality follows from~\cref{eq:posterior-jensen}, and the final inequality from~\Cref{thm:predictive-concentration,thm:predictive-concentration-kl}. 
    Since $M_n\rightarrow\infty$, taking the complementary event yields 
    \begin{equation*}
        P_0\left\{\mathsf{d}_2(P_{\widebar{Q}_{n,m}},\,P_{Q^\star})\le M_n\gamma_n^{1/a}\right\}\ge 1-o(1).
    \end{equation*}
    Since $M_n\gamma_n^{1/a}=o(1)$, for all $n$ sufficiently large we have $M_n\gamma_n^{1/a}\leq\varepsilon$, where $\varepsilon$ is the constant in~\ref{ass:concentration-ii}.
    Then, by~\ref{ass:concentration-ii},
    \begin{equation*}
        P_0\{\mathsf{d}_1({\widebar{Q}_{n,m}},\,{Q^\star})\leq CM_n\gamma_n^{1/a}\}
        \geq 
        P_0\{\mathsf{d}_2(P_{\widebar{Q}_{n,m}},\,P_{Q^\star})\leq M_n\gamma_n^{1/a}\}
        \geq 1-o(1).
    \end{equation*}
    Since $M_n$ is arbitrary up to a fixed multiplicative constant, the stated result follows.

<!-- lemma-9: body_md -->
\label{lemma:existence-uniqueness-full-pro}
    Suppose that Assumption~\ref{ass:convex} holds, that the score is continuous and bounded from below, and that the first variation of the predictive score with respect to the posterior exists.
    Assume further that the prior $\Pi$ admits a density with respect to the Lebesgue measure $\ensuremath{\,\mathrm{d}}\Pi(\theta)\propto \exp\{-U(\theta)\}$ for a smooth function $U:\Theta\to\mathbb{R}$ such that:
    \begin{enumerate}[(i),nosep]
        \item there exist constants $c > 0$ and $c^\prime \in \mathbb{R}$ such that for all $\theta\in\Theta$,
        $
            \nabla_\theta U(\theta)\cdot\theta \geq c\lVert\theta\rVert^2 + c^\prime,
        $
        and,\label{ass:prior-coercivity}
        \item $\nabla_\theta U$ is Lipschitz continuous.\label{ass:prior-lipz}
    \end{enumerate}
    Then there exists a unique minimiser to~\cref{eq:pro-posterior-defn} characterised by the implicit equation
    \begin{equation}\label{eq:pro-self-gibbs}
        q_n
        (\theta)\propto 
        \exp\left\{\left. - \frac{\lambda_n}{n}\sum_{i=1}^n \frac{\delta S(P_Q, x_i)}{\delta Q}\right|_{Q = Q_n}(\theta) \right\} 
        \pi 
        (\theta),
    \end{equation}
    where $\{\delta S(P_Q, x)/\delta Q\}|_{Q = Q_n}$ denotes the functional derivative of $S(P_Q, x)$ with respect to $Q$ evaluated at $Q = {Q_n}$, which we assume exists.

<!-- proof-14: body_md -->
We have by Assumption~\ref{ass:convex} that the score $S(\cdot,\,x)$ is convex in its first argument.
    In combination with the fact that the mapping $Q\mapsto P_Q$ is linear, we have the mapping $Q\mapsto S(P_Q,\,x)$ is convex.
    The existence and uniqueness of a minimiser is then provided under the same conditions in Proposition 2.5 of \citet{hu_mean-field_2021}, and we point the reader to their Section 4 for a proof. 
    
    To characterise the form of this minimiser, we take the functional derivative of the objective of~\cref{eq:pro-posterior-defn} with respect to $Q$, set it equal to zero, and solve for $Q$.
    We begin by noting that the functional derivative of the Kullback--Leibler regulariser evaluated at $\theta$ is
    \begin{equation}\label{eq:func-deriv-regulariser}
        1 + \log\left\{\frac{\ensuremath{\,\mathrm{d}} Q}{\ensuremath{\,\mathrm{d}}\Pi}(\theta)\right\},
    \end{equation}
    while that of the score term is
    \begin{equation}\label{eq:func-deriv-score-wrt-Q} 
        \frac{\lambda_n}{n} \sum_{i=1}^n \frac{\delta S(P_Q, x_i)}{\delta Q}(\theta).
    \end{equation}
    By the linearity of functional derivatives, setting the sum of~\cref{eq:func-deriv-regulariser,eq:func-deriv-score-wrt-Q} equal to zero and removing constant terms, we find that $Q_n$ must satisfy
    \begin{equation*}
        0 = \frac{\lambda_n}{n} \sum_{i=1}^n \left.\frac{\delta S(P_Q, x_i)}{\delta Q}\right|_{Q = Q_n}(\theta) + \log\left\{\frac{\ensuremath{\,\mathrm{d}} Q_n}{\ensuremath{\,\mathrm{d}}\Pi}(\theta)\right\}
    \end{equation*}
    whence rearranging yields the stated result.

<!-- assumption-7: body_md -->
\label{ass:poc}
    Let $\mathcal{P}_{2}(\Theta)$ denote the probability measures on $\Theta$ with finite second moment and a Lebesgue density.
    The following conditions hold:
    \begin{enumerate}[
        nosep,
        label=(\roman*), 
        ref=\Cref{ass:poc}(\roman*)
    ]
        \label{ass:poc-convex}
        \item the map $Q\mapsto S_n(P_Q)$ is Wasserstein smooth:
        there exists $L_n<\infty$ such that, for all $\theta,\theta^\prime\in\Theta$ and all $Q,Q^\prime\in\mathcal{P}_{2}(\Theta)$,
        \begin{equation*}
            \left\|
            \nabla_{\mathrm{W}_2}S_n(P_Q)(\theta)
            -
            \nabla_{\mathrm{W}_2}S_n(P_{Q^\prime})(\theta^\prime)
            \right\|
            \leq
            L_n
            \left\{
                \|\theta-\theta'\|
                +
                d_{\mathrm{W}_1}(Q,Q^\prime)
            \right\}.
        \end{equation*}
        Here $\nabla_{\mathrm{W}_2}$ denotes the Wasserstein-2 gradient, $d_{\mathrm{W}_1}$ is the Wasserstein-1 distance, and $\|\cdot\|$ is the Euclidean norm. 
        \label{ass:poc-smooth}
        \item The prior-tilted proximal Gibbs measures 
        satisfy a log-Sobolev inequality with a constant uniform in the proximal centre.
        More precisely, for each $\mu\in\mathcal{P}_{2}(\Theta)$ define
        \begin{equation*} 
            { q_n^\mu }
            (\theta)
            \propto
            \exp\left\{
                -\lambda_n
                \frac{\delta S_n(P_Q)}{\delta Q}\bigg|_{Q = \mu}(\theta)
            \right\}
            { \pi }
            (\theta).
        \end{equation*}
        There exists $\bar C_{\mathrm{LSI},n}<\infty$ such that
        \(
            \max\left\{C_{\mathrm{LSI}}(Q_n)
            ,
            \sup_{\mu\in\mathcal{P}_{2}(\Theta)}
            C_{\mathrm{LSI}}(Q_n^\mu)
            \right\}
            \leq
            \bar C_{\mathrm{LSI},n}.
        \)
        \label{ass:poc-lsi}
    \end{enumerate}
