<!-- theorem-1: body_md -->
\label{theointro:decomposition theorem}
 Let $X$ be a normal toric variety. The Serre subcategory of $\mathrm{Coh}^T(X)$ generated by elementary toric sheaves is $\mathrm{Coh}^T(X)$ itself.
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{corintro:decomposition theorem}
Let $X$ be a normal toric variety.
 Assume that a property $P$ for objects in $\mathrm{Coh}^T(X)$ is satisfied by elementary toric sheaves and is preserved in short exact sequences. Then $P$ is satisfied by all coherent toric sheaves on $X$.
<!-- end corollary-1 -->

 Assume that a property $P$ for objects in $\mathrm{Coh}^T(X)$ is satisfied by toric invertible sheaves and is preserved in short exact sequences. Then $P$ is satisfied by all coherent toric sheaves on $X$.

<!-- definition-1: body_md -->
\label{def:equivariant sheaf}
 Let $\mathcal{E}$ be a sheaf of $\mathcal{O}_X$-modules on $X$. A \textit{$T$-equivariant structure} on $\mathcal{E}$ is an isomorphism
 \[
 \Phi : \alpha^*\mathcal{E}\to\pi_2^*\mathcal{E}
 \]
 satisfying the following cocycle condition :
 \begin{equation*}
  \label{eq:cocycle condition equiv sheaf}
 (\mu\times 1_X)^*\Phi=\pi_{23}^*\Phi\circ(1_T\times\alpha)^*\Phi.
 \end{equation*}
 A \textit{toric sheaf} on $X$ is a pair $(\mathcal{E},\Phi)$ where $\mathcal{E}$ is a sheaf of $\mathcal{O}_X$-modules and $\Phi$ a $T$-equivariant structure on $\mathcal{E}$.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
\label{def:Sigma families}
 A \textit{$\Sigma$-family} for a fan $\Sigma$ is a pair $(E^\Sigma,\eta^\Sigma)$ consisting of a collection $E^\Sigma=(E^\sigma)_{\sigma\in\Sigma}$ of $\sigma$-families and a collection $\eta^\Sigma=(\eta_{\tau\sigma})_{\tau\leq \sigma}$ of isomorphisms $\eta_{\tau\sigma} : \iota_{\tau\sigma}^*E^\sigma \to E^\tau$ such that
 \begin{enumerate}
  \item for each $\sigma\in\Sigma$, $\eta_{\sigma\sigma}=1_{E^\sigma}$,
  \item for each $\rho\leq \tau\leq \sigma$, $\eta_{\rho\sigma}=\eta_{\rho\tau}\circ \iota_{\rho\tau}^*\eta_{\tau\sigma}$.
 \end{enumerate}
 A \textit{morphism of $\Sigma$-families} $\Phi^\Sigma : (E^\Sigma,\eta^\Sigma)\to (F^\Sigma,\nu^\Sigma)$ is a collection of morphisms $(\phi^\sigma : E^\sigma \to F^\sigma)_{\sigma\in\Sigma}$ of $\sigma$-families such that for all $\tau\leq\sigma$,
 \[
 \nu_{\tau\sigma}\circ \iota_{\tau\sigma}^*\phi^\sigma=\phi^\tau\circ \eta_{\tau\sigma}.
 \]
<!-- end definition-2 -->
<!-- theorem-2: body_md -->
\label{theorem:equiv categories Tsheaves and Sigma families}
 For a given fan $\Sigma$, the $\Sigma$-families together with their morphisms form a category that is equivalent to $\mathrm{qCoh}^T(X_\Sigma)$.
<!-- end theorem-2 -->
<!-- definition-3: body_md -->
\label{def:finite sigma families}
 A $\sigma$-family $(E^\sigma,\chi^\sigma)$ is \textit{finite} if it satisfies the following :
 \begin{enumerate}
  \item For each $m\in M$, $E^\sigma_m$ is finite dimensional,
  \item For each strictly increasing sequence $(m_i)_{i\in\mathbb{Z}}\in M^\mathbb{Z}$, i.e. such that 
  \[
\ldots<_\sigma m_{i-1} <_\sigma m_i <_\sigma m_{i+1} <_\sigma \ldots,
\] there exists $i_0\in\mathbb{Z}$ such that $E^\sigma_{m_i}=\lbrace 0\rbrace$ for $i\leq i_0$,
  \item For each $m\in M$, denote by $C^\sigma_m$ the cokernel of the map 
  \[
 \sum_{m'<_\sigma m} \chi^\sigma_{m',m} : \bigoplus_{m'<_\sigma m} E^\sigma_{m'} \to E^\sigma_m.
  \]
  Then the set of classes $[m]\in M/(\sigma^\perp\cap M)$ such $C^\sigma_m\neq\lbrace 0\rbrace$ is finite.
 \end{enumerate}
A $\Sigma$-family $(E^\Sigma,\eta^\Sigma)$ is \textit{finite} if for each $\sigma\in\Sigma$, $E^\sigma$ is finite.

<!-- proposition-1: body_md -->
\label{prop:family of filtrations rank one reflexive sheaves}
 Let $(L^\sigma_m)$ be the family of multifiltrations for $\mathcal{L}$. Then for any $\sigma\in\Sigma$,
 \[
 L^\sigma_m = \left\{
 \begin{array}{ccc}
\lbrace 0\rbrace & \mathrm{ if } & \exists\; \rho\in\sigma(1)\:\vert\:   \langle m, u_\rho \rangle < -a_\rho \\
 \mathbb{K}             & \mathrm{ if } & \forall\; \rho\in\sigma(1)\:\vert\:   \langle m, u_\rho \rangle\geq -a_\rho.
 \end{array}
 \right.
 \]
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Let $\rho\in\Sigma(1)$. By definition,
\[
\Gamma(U_\rho,\mathcal{L})=\lbrace f\in\mathbb{K}(X)^*\:\vert\: \mathrm{div}(f)_{\vert U_\rho}+ a_\rho D_\rho \geq 0\rbrace \cup\lbrace 0\rbrace.
\]
Let $m\in M$. Then, $\mathrm{div}(\chi^m)$ is $T$-invariant. By the orbit-cone correspondence, $\mathrm{div}(\chi^m)_{\vert U_\rho}$ is supported on $D_\rho$ and the valuation of $\chi^m$ is given by $\nu_\rho(\chi^m)=\langle m , u_\rho \rangle$ \cite[Proposition 4.1.1]{CLS}. Hence, $\mathrm{div}(\chi^m)_{\vert U_\rho}+ a_\rho D_\rho \geq 0$ if and only if $\langle m , u_\rho \rangle+a_\rho\geq 0$. We deduce that the degree-$m$ component of $\Gamma(U_\rho,\mathcal{L})$ is given by
\[
\Gamma(U_\rho,\mathcal{L})_m=\left\{
 \begin{array}{ccc}
\lbrace 0\rbrace & \mathrm{ if } & \langle m , u_\rho \rangle < -a_\rho \\
 \mathbb{K}  \cdot\chi^m           & \mathrm{ if } & \langle m , u_\rho \rangle\geq -a_\rho.
 \end{array}
 \right.
\]
The result follows, as by definition:
\begin{equation*}
 \begin{array}{ccc}
L^\sigma_m & \simeq &  \Gamma(U_\sigma,\mathcal{L})_m\\
 & \simeq & \displaystyle\bigcap_{\rho\in\sigma(1)} \, \Gamma(U_\rho,\mathcal{L})_m
 \end{array}
\end{equation*}
where for the last isomorphism we used reflexivity of $\mathcal{L}$ which implies normality, cf \cite[Proposition 1.6]{Har80}.
<!-- end proof-1 -->
<!-- definition-4: body_md -->
\label{def:elementary toric sheaf}
 A coherent toric sheaf $\mathcal{E}$ on a normal toric variety $X$ is \textit{elementary} if it is pure, with reduced and irreducible support, and restricts to a rank one toric reflexive sheaf on its support.
<!-- end definition-4 -->
<!-- lemma-1: body_md -->
\label{lem:elementary toric sheaf geometric}
 Assume that $X$ is simplicial. Let $\mathcal{E}$ be an elementary toric sheaf on $X$. Then there are a cone $\tau\in\Sigma(d)$,
  integers $(a_\rho)_{\rho+\tau\in\Sigma(d+1)}$, and a weight $m\in M$ such that 
  \begin{equation}
   \label{eq:elementary toric sheaf geometric}
  \mathcal{E}\simeq \Big(\big(\iota_*\mathcal{O}_{V(\tau)}\big)\otimes_{\mathcal{O}_X}\mathcal{O}_X\big(\sum_{\tau+\rho\in\Sigma(d+1)}a_\rho \, D_\rho\big) \Big)[m],
  \end{equation}
where $\iota : V(\tau)\to X$ stands for the inclusion map, and $[m]$ denotes the shifted equivariant structure.

By definition, there exists $V=V(\tau)$, for $\tau\in\Sigma(d)$, and a rank one reflexive sheaf $\mathcal{L}$ on $V$ such that 
 $\mathcal{E}\simeq\iota_*\mathcal{L}$. We can find integers $(a_{\overline\rho})_{\overline\rho\in\Sigma_V(1)}$ such that
 \[
 \mathcal{L}\simeq\mathcal{O}_V(\sum_{\overline\rho\in\Sigma_V(1)} a_{\overline\rho} D_{\overline\rho}),
 \]
 where the $D_{\overline\rho}$'s stand for invariant divisors in $V$.
 As $X$ is simplicial, by the orbit-cone correspondence, one has a bijection between $T_X$-invariant divisors $D_\rho$ such that $\tau+\rho\in\Sigma(d+1)$ and $T_V$-invariant divisors on $V$ \cite[Theorem 3.2.6, Proposition 3.2.7]{CLS}. More precisely, for any $\rho\in\Sigma(1)$,
 \[
 D_\rho\cap V(\tau)=\big(\bigcup_{\rho\leq\sigma} O(\sigma)\big) \cap\big(\bigcup_{\tau\leq \sigma'} O(\sigma') \big)=\bigcup_{\rho+\tau\leq\sigma} O(\sigma),
 \]
 and the right hand side defines a divisor in $V(\tau)$ precisely when $\rho+\tau\in \Sigma(d+1)$. For $\rho\in\Sigma(1)$ such that $\tau+\rho\in\Sigma(d+1)$, we set $a_\rho:=a_{\overline{\rho}}$, where $\overline{\rho}$ is the projection of $\tau+\rho\in\Sigma$ to $\Sigma_V$ through the map
 \[
 N_\mathbb{R}\to N_\mathbb{R}/\mathrm{Span}_\mathbb{R}(\tau).
 \]
 As it is supported on $V$, we deduce that the sheaf 
 \[
 \big(\iota_*\mathcal{O}_{V}\big)\otimes_{\mathcal{O}_X}\mathcal{O}_X\big(\sum_{\tau+\rho\in\Sigma(d+1)}a_\rho \, D_\rho\big)
 \]
 is isomorphic, as a coherent sheaf, to 
 \[
 \iota_*\Big(\mathcal{O}_V(\sum_{\overline\rho\in\Sigma_V(1)} a_{\overline\rho} D_{\overline\rho})\Big)\simeq\iota_*\mathcal{L}\simeq\mathcal{E}.
 \]
 We claim that $\mathcal{E}$ is simple, which implies the result, as all $T$-equivariant structures on a simple toric sheaf $(\mathcal{E},\Phi)$ are of the form $\chi^m\cdot \Phi$, for $m\in M$. To prove the claim, it is enough to notice that as $\iota_*\mathcal{O}_V$ is supported on $V$, so is $\operatorname{\mathcal{H\mkern-3mu om}}_{\mathcal{O}_X}(\iota_*\mathcal{O}_V,\iota_*\mathcal{O}_V)$, and then
 \[
 \operatorname{\mathcal{H\mkern-3mu om}}_{\mathcal{O}_X}(\mathcal{E},\mathcal{E})\simeq \operatorname{\mathcal{H\mkern-3mu om}}_{\mathcal{O}_X}(\iota_*\mathcal{O}_V,\iota_*\mathcal{O}_V)\simeq \iota_*\operatorname{\mathcal{H\mkern-3mu om}}_{\mathcal{O}_V}(\mathcal{O}_V,\mathcal{O}_V).
 \]
We deduce that
 \[
 \mathrm{Hom}(\mathcal{E},\mathcal{E})\simeq \mathrm{Hom}(\mathcal{O}_V,\mathcal{O}_V)\simeq \mathbb{K}
 \]
 and the result follows.

<!-- definition-5: body_md -->
\label{def:serre subcategory}
 Let $\mathfrak{A}$ be an abelian category. A full subcategory $\mathfrak{B}$ is a \textit{Serre subcategory} if it contains $0$ and for any exact sequence
 \[
 E_1\to E_2\to E_3
 \]
 of objects in $\mathfrak{A}$, if $E_1$ and $E_3$ are in $\mathfrak{B}$, so is $E_2$.

<!-- proof-3: body_md -->
It is enough to show that the smallest class of objects $\mathfrak{A}$ of $\mathrm{Coh}^T(X)$ containing elementary toric sheaves and stable under subobjects, quotients, and extensions, is the set of all coherent toric sheaves. Taking the full subcategory associated to this class gives the result. Let $\mathcal{E}$ be a coherent toric sheaf. We will show that $\mathcal{E}\in \mathfrak{A}$ by using a step by step decomposition of $\mathcal{E}$ involving a finite number of short exact sequences. First, one can consider the torsion subsheaf $T(\mathcal{E})=T_{n-1}(\mathcal{E})\subset \mathcal{E}$. We have a short exact sequence of toric sheaves :
 \[
 0\to T(\mathcal{E})\to \mathcal{E} \to \mathcal{Q} \to 0
 \]
 where the quotient $\mathcal{Q}$ is torsion-free and equivariant. Hence, it is enough to prove that $\mathfrak{A}$ contains all torsion toric sheaves and all torsion-free toric sheaves. 
 
 We start by the torsion case, and use an induction on the dimension of the support. If $\mathcal{E}$ is torsion and of dimension zero, then it is supported on the torus fixed points of $X$. By induction, we may assume that $\mathcal{E}$ is supported on a single torus fixed point $p\in X$. Denoting $\iota: p\to X$ the inclusion, we have a short exact sequence of coherent toric sheaves
 \[
0\to \mathcal{I}_p\cdot\mathcal{E}\to\mathcal{E}\to\iota_*\iota^*\mathcal{E}\to 0
 \]
 supported on $p$. The multiplicity of $p$ in $\mathcal{I}_p\cdot\mathcal{E}$ is strictly smaller than in $\mathcal{E}$, and by induction on $\mathrm{mult}_p(\mathcal{E})$ we may assume that $\mathcal{E}\simeq \iota_*\iota^*\mathcal{E}$. Then, by coherence, $\mathcal{E}$ is equivalent to a finite dimensional $T$-representation over $p$. It is then a direct sum of elementary toric sheaves by the complete reducibility theorem, and the result follows in that case. Assuming the result for toric torsion sheaves of supports of dimension less or equal to $d-1$, we consider $\mathcal{E}\in\mathrm{Coh}^T(X)$ and assume that it is torsion, with $\dim(\mathrm{Supp}(\mathcal{E}))=d$. Using the torsion filtration for $\mathcal{E}$, we can assume further that $\mathcal{E}$ is pure. Then, we proceed by induction on the number of irreducible components for $\mathrm{Supp}(\mathcal{E})$. Let $V\subset \mathrm{Supp}(\mathcal{E})$ be such a component, and denote by $\iota : V\to X$ the inclusion. Then we have a short exact sequence :
 \[
 0\to \mathcal{H}^0_V(\mathcal{E})\to \mathcal{E} \to\mathcal{Q} \to 0
 \]
 where $\mathcal{H}^0_V(\mathcal{E})$ is the maximal coherent subsheaf supported on $V$. By maximality, $\mathcal{H}^0_V(\mathcal{E})$ is unique, which implies by invariance of $V$ that it is toric. Then, the quotient $\mathcal{Q}$ is either zero, or a toric pure sheaf of dimension $d$ whose support has less irreducible components than the one of $\mathcal{E}$. By induction, we can restrict to the case when $\mathcal{E}$ is pure with irreducible support. We then have a sequence of coherent toric sheaves
 \[
0\to \mathcal{I}_V\cdot\mathcal{E}\to\mathcal{E}\to\iota_*\iota^*\mathcal{E}\to 0
 \]
 supported on $V=\mathrm{Supp}(\mathcal{E})$. As the multiplicity of $V$ in $\mathcal{I}_V\cdot \mathcal{E}$ is strictly smaller than in $\mathcal{E}$, by induction on the multiplicity, we may assume $\mathcal{E}\simeq\iota_*\iota^*\mathcal{E}$, that is the support of $\mathcal{E}$ is reduced.  We then consider $\mathcal{E}_{\vert V}$. This admits a short exact sequence (as toric sheaves on $V$) :
 \[
 0\to \mathcal{E}_{\vert V} \to \mathcal{E}_{\vert V}^{\vee\vee} \to \mathcal{Q}' \to 0
 \]
 where this time the quotient $\mathcal{Q}'$ is a torsion toric sheaf  on $V$. By our induction hypothesis on the dimension of the support, $\iota_*\mathcal{Q}'\in \mathfrak{A}$, and thus we can restrict to the case when $\mathcal{E}_{\vert V}$ is reflexive. Finally, we argue by induction on the rank. If the rank $r$ of $\mathcal{E}_{\vert V}$ is one, then $\mathcal{E}$ is an elementary toric sheaf, and thus $\mathcal{E}\in\mathfrak{A}$. We can always find a saturated rank $1$ toric subsheaf $\mathcal{L}$ of $\mathcal{E}_{\vert V}$. Indeed, if $(E^\sigma_m)$ is the family of multifiltrations for $\mathcal{E}_{\vert V}$, with $E=\displaystyle\lim_{\to} E^\sigma_m$, then the family of multifiltrations $(L\cap E^\sigma_m)$, for $L\subset E$ a line, corresponds to such a saturated subsheaf $\mathcal{L}\subset \mathcal{E}_{\vert V}$ (see e.g. \cite[Lemma 2.15]{NaTip}). The sheaf $\mathcal{L}$ is reflexive on $V$, hence elementary on $X$, and the quotient $\mathcal{Q}_\mathcal{L}$ of $\mathcal{E}_{\vert V}$ by $\mathcal{L}$ is torsion-free, of rank $r-1$. Reducing again to its reflexive hull $\mathcal{Q}_\mathcal{L}^{\vee\vee}$ (the quotient is again a torsion sheaf on $V$), this concludes the proof in that case by induction on the rank.
 
 We move on to the torsion-free case. Assume  that $\mathcal{E}$ is torsion-free. Then it lies in the canonical sequence
 \[
 0\to \mathcal{E} \to \mathcal{E}^{\vee\vee} \to \mathcal{Q} \to 0,
 \]
 where $\mathcal{Q}$ is a toric torsion sheaf. By the previous case, $\mathcal{Q}\in \mathfrak{A}$. Hence, we reduced to the reflexive case, and a similar argument as before on the rank concludes the proof of the theorem.

By Corollary \ref{corintro:decomposition theorem}, it is enough to show that the property $P$ is satisfied by all the elementary toric sheaves on $X$. Let $\mathcal{E}\in\mathrm{Coh}^T(X)$ be elementary. By Lemma \ref{lem:elementary toric sheaf geometric}, we may assume that
\[
\mathcal{E}\simeq (\iota_*\mathcal{O}_{V}\otimes_{\mathcal{O}_X}\mathcal{L})[m],
\]
where $\iota:V\to X$ is the inclusion of some orbit closure $V=V(\tau)$ for $\tau\in\Sigma$, $\mathcal{L}$ is a toric line bundle on $X$, and $m\in M$. We will assume $m=0$, which will not affect the argument. The key fact is that $\iota_*\mathcal{O}_{V}$ admits a Koszul resolution by direct sums of toric line bundles. Indeed, the orbit-cone correspondence gives
\[
V(\tau)=\bigcap_{\rho\in\tau(1)}\, D_\rho.
\]
Then, we can introduce the toric vector bundle of rank $r=\mathrm{codim} V$ :
\[
 \mathcal{F}=\bigoplus_{\rho\in\tau(1)}\, \mathcal{O}_X(D_\rho)
\]
and find a section $s=(s_\rho)_{\rho\in \tau(1)}\in H^0(X,\mathcal{F})$, with $(s_\rho)$ defining sections for the $D_\rho$'s, such that $V(\tau)=\lbrace s= 0 \rbrace$. Convolution by $s$ induces the Koszul complex
\[
 0\to \Lambda^r\mathcal{F}^\vee \to \ldots \to \Lambda^2\mathcal{F}^\vee \to\mathcal{F}^\vee\to \mathcal{O}_X\to \iota_*\mathcal{O}_V\to 0
\]
which corresponds to the toric locally free resolution of $\iota_*\mathcal{O}_V$ :
\[
 0\to \mathcal{O}_X(-\sum_{\rho\in\tau(1)}\, D_\rho) \to \ldots \to \bigoplus_{\sigma\in\tau(2)}\, \mathcal{O}_X(-\sum_{\rho\in\sigma(1)}\, D_\rho)\to\bigoplus_{\rho\in\tau(1)}\, \mathcal{O}_X(-D_\rho) \to \mathcal{O}_X\to \iota_*\mathcal{O}_V\to 0.
\]
Tensoring by $\mathcal{L}$ gives a toric resolution of $\mathcal{E}$ by direct sums of toric line bundles. As the property $P$ is satisfied by toric line bundles, and preserved in short exact sequences, a simple induction shows that it is satisfied by each term in this resolution of $\mathcal{E}$, and then by $\mathcal{E}$ itself. The result follows.

Using Corollary \ref{corintro:decomposition theorem}, and Lemma \ref{lem:exact seq toric sheaves gives exact seq acyclic resolution}, it is enough to show the result when $\mathcal{E}$ is an elementary toric sheaf. Then, there is a torus invariant subvariety $\iota:V\to X$, with fan $\Sigma_V$,  and a rank one reflexive sheaf $\mathcal{L}$ on $V$, such that $\mathcal{E}=\iota_*\mathcal{L}$. We leave to the reader the proof of the following facts :
 \begin{itemize}
  \item For $i>\dim(V)$, $\mathcal{C}^i(\Sigma,\iota_*\mathcal{L})=0$,
  \item For $i\leq \dim(V)$, $\iota^*\mathcal{C}^i(\Sigma,\iota_*\mathcal{L})=\mathcal{C}^i(\Sigma_V,\mathcal{L})$, and $\partial^i\circ\iota^*=\iota^*\circ\partial^i$,
  \item For each $i$, $\mathcal{C}^i(\Sigma,\iota_*\mathcal{L})$ has support in $V$.
 \end{itemize}
 From those properties, one deduces that if the complex $(\mathcal{C}^\bullet(\Sigma_V,\mathcal{L}),\partial^\bullet)$ is exact then  so is $(\mathcal{C}^\bullet(\Sigma,\iota_*\mathcal{L}),\partial^\bullet)$. Hence, we can reduce ourselves to the case when $\mathcal{E}=\mathcal{L}$ is a rank one reflexive toric sheaf on $X$, and there exists integers $(a_\rho)_{\rho\in\Sigma(1)}\in\mathbb{Z}^{\vert\Sigma(1)\vert}$ such that
 \[
 \mathcal{L}\simeq \mathcal{O}_X\big(\sum_{\rho\in\Sigma(1)}\,a_\rho\,D_\rho\big).
 \]
 We set $(L^\sigma_m)$ to be a family of multifiltrations of $\mathcal{L}$. By proposition \ref{prop:family of filtrations rank one reflexive sheaves}, for any $\sigma\in\Sigma$, and any weight $m\in M$, we may assume that
 \[
 L^\sigma_m=\left\{
 \begin{array}{ccc}
  \lbrace 0\rbrace & \mathrm{ if } & \exists\, \rho\in\sigma(1)\,,\: \langle m, u_\rho\rangle < -a_\rho\\
  \mathbb{K} & \mathrm{ if } & \forall\, \rho\in\sigma(1)\,,\: \langle m, u_\rho\rangle \geq -a_\rho.
 \end{array}
 \right.
 \]
 We wish to prove that $(\mathcal{C}^\bullet(\Sigma,\mathcal{L}),\partial^\bullet)$ is an exact sequence of quasicoherent toric sheaves. By Theorem \ref{theorem:equiv categories Tsheaves and Sigma families}, it is enough to prove exactness at the level of $\Sigma$-families. This amounts to showing exactness, for each $\tau\in\Sigma$, and $m\in M$, of the complex $(\Gamma(U_\tau,\mathcal{C}^\bullet(\Sigma,\mathcal{L}))_m,\partial^\bullet)$.
 By \eqref{eq:Gamma Usigma acyclic resolution}, this complex is
 \begin{equation}
 \label{eq:cohomology complex acyclic resolution rank one case}
 0\to L^\tau_m\to \bigoplus_{\sigma\in\Sigma(n)}\, L^{\tau\cap\sigma}_m \xrightarrow{\partial_m^0}  \ldots  \xrightarrow{\partial_m^{n-2}}\bigoplus_{\rho\in\Sigma(1)} L^{\tau\cap\rho}_m \xrightarrow{\partial_m^{n-1}} \mathbb{K} \to 0
\end{equation}
with 
\[
(\partial^i_m (s_\sigma))_{\sigma'}= \sum_{\substack{\sigma'\leq\sigma\\\sigma\in \Sigma(n-i)}}\, \varepsilon_{\sigma'\sigma} \, s_\sigma.
\]
 From the description of $(L^\sigma_m)$, the contribution of $L^{\tau\cap\sigma}_m$ to this latter complex will be non-zero precisely when for all $\rho\in\tau(1)\cap\sigma(1),\, \langle m, u_\rho\rangle \geq -a_\rho$. Let then $\Sigma_{\hat\tau_m}\subset\Sigma$ be the subset of cones $\sigma\in\Sigma$ such that for all $\rho\in \sigma(1)$, either $\rho\notin \tau(1)$, or $\rho\in\tau(1)$ and $\langle m, u_\rho \rangle \geq -a_\rho$. Equivalently, $\Sigma_{\hat\tau_m}$ is the largest subfan of $\Sigma$ that do not contain the set of rays  $\tau_m:=\lbrace \rho\in\tau(1)\:\vert\: \langle m, u_\rho\rangle < -a_\rho \rbrace\subset \tau(1)$. Then, we reduced ourselves to showing exactness of the complex
 \begin{equation}
  \label{eq:augmented homology proof of acyclic resolution}
 0\to L_m^{\tau_m}\to \bigoplus_{\sigma\in\Sigma_{\hat\tau_m}(n)}\, \mathbb{K}_\sigma \xrightarrow{\partial_m^0}  \ldots \to\bigoplus_{\rho\in\Sigma_{\hat\tau_m}(1)} \mathbb{K}_\rho \xrightarrow{\partial_m^{n-1}} \mathbb{K} \to 0,
 \end{equation}
 where each $\mathbb{K}_\sigma$ is a copy of $\mathbb{K}$. We recognize the augmented homology complex for the CW-complex given by $\Sigma_{\hat\tau_m}\cap S^{n-1}$, where $S^{n-1}\subset N_\mathbb{R}$ is the sphere given by a fixed euclidean structure on $N_\mathbb{R}$. We further note that either $\tau_m=\varnothing$, and then $\vert\Sigma_{\hat\tau_m}\vert\cap S^{n-1}=S^{n-1}$, or $\tau_m\neq 0$, and then  $\vert\Sigma_{\hat\tau_m}\vert\cap S^{n-1}$ is contractible. Indeed, for each cone $\sigma\in\Sigma$ such that $\sigma(1)\cap\tau_m\neq\varnothing$, the cell $\sigma\cap S^{n-1}$ admits a deformation retract to a point in $\tau_m\cap S^{n-1}$, and $\tau\cap S^{n-1}$ is contractible.  In both cases, this proves exactness of \eqref{eq:augmented homology proof of acyclic resolution}, which concludes the proof.
