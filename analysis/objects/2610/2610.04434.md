\label{theorem: Intro cohom realization of supercoset}\rm
Let ${\mathfrak{h}}$ be a direct sum of a reductive Lie algebra and Lie superalgebras of type ${\mathfrak{osp}}_{1|2m}$. Suppose that the restriction of ${\kappa}_{\mathfrak{h}} - {\frac{1}{2}}{\overline{\kappa}}_{\mathfrak{h}}$ to each nonabelian simple summand of ${\mathfrak{h}}$ is an irrational level. Then we have
\begin{align*}
{H}^{{\frac{\infty}{2}} + {n}}_{{\,\,} {\text{s, rel}}}({\mathfrak{h}}, {V}^{{\kappa}_{\mathfrak{g}}}_{{\,\,} {\text{s}}}({\mathfrak{g}}) \otimes {V}^{-{\kappa}_{\mathfrak{h}}}_{{\,\,} {\text{s}}}({\mathfrak{h}})) \cong {\delta}_{n,0}{\,}{\text{Com}}({V}^{{\kappa}_{\mathfrak{h}}}_{{\,\,} {\text{s}}}({\mathfrak{h}}), {V}^{{\kappa}_{\mathfrak{g}}}_{{\,\,} {\text{s}}}({\mathfrak{g}})),
\end{align*}
for all $n \in {\mathbb{Z}}$.

<!-- assumption-1: body_md -->
\label{assump: Intro KS condition}\rm
Let ${\mathfrak{g}}$ be a finite-dimensional Lie superalgebra with a nondegenerate consistent supersymmetric invariant bilinear form ${\kappa}_{\mathfrak{g}}$. Suppose that there exists an injective Lie superalgebra homomorphism ${\iota} : {\mathfrak{h}} \rightarrow {\mathfrak{g}}$, and that the bilinear form ${\kappa}_{\mathfrak{h}} := {\kappa}_{\mathfrak{g}} \circ ({\iota} \times {\iota})$ is nondegenerate on ${\mathfrak{h}}$. Assume further that the orthogonal complement of ${\iota}({\mathfrak{h}})$ in ${\mathfrak{g}}$ with respect to ${\kappa}_{\mathfrak{g}}$ admits a vector superspace decomposition ${\iota}({\mathfrak{h}})^{\perp} = {\mathfrak{t}}^{+} \oplus {\mathfrak{t}}^{-}$ such that:
\begin{align*}
{\kappa}_{\mathfrak{g}}({\mathfrak{t}}^{\pm}, {\,} {\mathfrak{t}}^{\pm}) = 0,
{\quad}
[{\mathfrak{t}}^{\pm}, {\iota}({\mathfrak{h}}) \oplus {\mathfrak{t}}^{\pm}] \subseteq {\mathfrak{t}}^{\pm}.
\end{align*}

\label{def: sMP}\rm
Let ${\mathfrak{g}}$ be a finite-dimensional Lie superalgebra over ${\mathbb{C}}$ with a nondegenerate consistent supersymmetric invariant bilinear form ${\kappa}_{\mathfrak{g}}$. Suppose that there is a decomposition ${\mathfrak{g}} = {\mathfrak{a}}^{+} {\oplus} {\,} {\mathfrak{a}}^{-}$ as a vector superspace, where ${\mathfrak{a}}^{+}$ and ${\mathfrak{a}}^{-}$ are isotropic subspaces of ${\mathfrak{g}}$ with respect to ${\kappa}_{\mathfrak{g}}$, i.e., ${\kappa}_{\mathfrak{g}}({\mathfrak{a}}^{\pm}, {\,} {\mathfrak{a}}^{\pm}) = 0$, and ${\mathfrak{a}}^{+}$ is a Lie subalgebra of ${\mathfrak{g}}$. Then $({\mathfrak{g}}, {\mathfrak{a}}^{+}, {\mathfrak{a}}^{-})$ is called a {\textit{super Manin pair}} and we also simply say that ${\mathfrak{g}}$ admits a super Manin pair structure. Moreover, if ${\mathfrak{a}}^{-}$ is also a Lie subalgebra of ${\mathfrak{g}}$, then $({\mathfrak{g}}, {\mathfrak{a}}^{+}, {\mathfrak{a}}^{-})$ is called a {\textit{super Manin triple}}. We can also consider the notion of a {\textit{super Manin subpair}}.

\label{assum: second assumption}\rm
Let ${\mathfrak{g}}$ be a finite-dimensional Lie superalgebra with a nondegenerate consistent supersymmetric invariant bilinear form ${\kappa}_{\mathfrak{g}}$. Suppose that there exists an injective Lie superalgebra homomorphism ${\iota} : {\mathfrak{h}} \rightarrow {\mathfrak{g}}$, and that the bilinear form ${\kappa}_{\mathfrak{h}} := {\kappa}_{\mathfrak{g}} \circ ({\iota} \times {\iota})$ is nondegenerate on ${\mathfrak{h}}$. \begin{align*}
{\kappa}_{\mathfrak{g}}({\mathfrak{t}}^{\pm}, {\,} {\mathfrak{t}}^{\pm}) = 0,
{\quad}
[{\mathfrak{t}}^{\pm}, {\iota}({\mathfrak{h}}) \oplus {\mathfrak{t}}^{\pm}] \subseteq {\mathfrak{t}}^{\pm}.
\end{align*}

\rm
A vertex algebra $V$ is called an {\textit{$N = 2$ superconformal vertex algebra}} if it has two odd vectors ${G}^{+}$, ${G}^{-}$ such that ${T} := {\frac{1}{2}}({G}^{+}_{(0)}{G}^{-} + {G}^{-}_{(0)}{G}^{+})$ is a conformal vector with central charge ${c} \in {\mathbb{C}}$, and the following $N = 2$ super-Virasoro relations hold:
\begin{gather*}
[{T} {\,} {}_{\lambda} {\,} {G}^{\pm}] = ({\partial} + {\frac{3}{2}}{\lambda}){G}^{\pm},
{\quad}
[{G}^{\pm} {\,} {}_{\lambda} {\,} {G}^{\pm}] = 0,
{\quad}
[{G}^{+} {\,} {}_{\lambda} {\,} {G}^{-}] = {T} + {\frac{1}{2}}{\partial}{U} + {\lambda}{U} + {\frac{c}{6}}{\lambda}^{2},
\\
[{T} {\,} {}_{\lambda} {\,} {U}] = ({\partial} + {\lambda}){U},
{\quad}
[{G}^{\pm} {\,} {}_{\lambda} {\,} {U}] = {\mp}{G}^{\pm},
{\quad}
[{U} {\,} {}_{\lambda} {\,} {U}] = {\frac{c}{3}}{\lambda},
\end{gather*}
where ${U} := {G}^{+}_{(1)}{G}^{-}$. In this case, we call the collection ${T}, {G}^{\pm}, {U}$ (or simply ${G}^{\pm}$) an {\textit{$N = 2$ superconformal structure}} of $V$.
\label{def: N=2 sconf VA}

<!-- lemma-4: body_md -->
\label{lem: equiv def of N=2 sconf}\rm
Let $V$ be a vertex algebra with an $N = 1$ superconformal vector $G$ of central charge $c \in {\mathbb{C}}$. Suppose that $U \in V$ is an even primary vector of conformal weight $1$ satisfying the following OPEs:
\begin{align}
[{G} {\,} {}_{\lambda} {\,} {U}] = {G}_{(0)}{U},
{\quad}
[{U} {\,} {}_{\lambda} {\,} {U}] = {\frac{c}{3}}{\lambda},
{\quad}
{U}_{(0)}({G}_{(0)}{U}) = - {G}.
\label{eq: equiv def of N=2 sconf}
\end{align}
Then the vectors ${G}^{\pm} := {\frac{1}{2}}({G} \mp {G}_{(0)}{U})$ form an $N = 2$ superconformal structure of $V$ with central charge $c$.
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
First, if we denote by ${T}$ the conformal vector corresponding to ${G}$, then by using sesquilinearity, skew-symmetry, and the Jacobi identity of vertex algebras, we obtain the following identities:
\begin{gather}
[{T} {\,} {}_{\lambda} {\,} {G}_{(0)}{U}] = ({\partial} +{\frac{3}{2}}{\lambda}){G}_{(0)}{U},
{\quad}
[{G} {\,} {}_{\lambda} {\,} {G}_{(0)}{U}] = ({\partial} + {2}{\lambda}){U},
\label{eq: equiv def of N=2 sconf AB}
\\
[{U} {\,} {}_{\lambda} {\,} {G}_{(0)}{U}] = -{G},
{\quad}
[{G}_{(0)}{U} {\,} {}_{\lambda} {\,} {G}_{(0)}{U}] = -(2{T} + {\frac{c}{3}}{\lambda}^{2}).
\label{eq: equiv def of N=2 sconf CD}
\end{gather}
For example, to prove the first identity in \eqref{eq: equiv def of N=2 sconf CD}, we note that by assumption $[{G} {\,} {}_{\lambda} {\,} {U}] = {G}_{(0)}{U}$, and by skew-symmetry we also have $[{U} {\,} {}_{\lambda} {\,} {G}] = -{G}_{(0)}{U}$. Hence, from the Jacobi identity, we have:
\begin{align*}
[{U} {\,} {}_{\lambda} {\,} {G}_{(0)}{U}] 
= [{U} {\,} {}_{\lambda} {\,} [{G} {\,} {}_{\gamma} {\,} {U}]] 
= [[{U} {\,} {}_{\lambda} {\,} {G}] {\,} {}_{\lambda + \gamma} {\,} {U}] 
+ [{G} {\,} {}_{\gamma} {\,} [{U} {\,} {}_{\lambda} {\,} {U}]] 
= - [{G}_{(0)}{U} {\,} {}_{\lambda + \gamma} {\,} {U}],
\end{align*}
so that substituting ${\gamma} = -{\lambda}$ yields $[{U} {\,} {}_{\lambda} {\,} {G}_{(0)}{U}] = - ({G}_{(0)}{U})_{(0)}{U}$. That is, since the ${\lambda}$-bracket $[{U} {\,} {}_{\lambda} {\,} {G}_{(0)}{U}]$ contains only the constant term in ${\lambda}$, the assumption ${U}_{(0)}({G}_{(0)}{U}) = - {G}$ implies the desired identity $[{U} {\,} {}_{\lambda} {\,} {G}_{(0)}{U}] = -{G}$. Also, the second equality in equation \eqref{eq: equiv def of N=2 sconf CD} follows from the second identity in \eqref{eq: equiv def of N=2 sconf AB} and the first identity in \eqref{eq: equiv def of N=2 sconf CD}.

\vskip 1mm

If we now consider the odd vectors defined by ${G}^{\pm} := {\frac{1}{2}}({G} \mp {G}_{(0)}{U})$, we see from \eqref{eq: equiv def of N=2 sconf AB} and \eqref{eq: equiv def of N=2 sconf CD} that the vectors ${G}^{\pm}$ satisfy all the conditions of an $N = 2$ superconformal structure in Definition \ref{def: N=2 sconf VA}. For example, we have:
\begin{align*}
[{G}^{+} {\,} {}_{\lambda} {\,} {G}^{-}] 
&= {\frac{1}{4}}[{G} - {G}_{(0)}{U} {\,} {}_{\lambda} {\,} {G} + {G}_{(0)}{U}] 
\\
&= {\frac{1}{4}} {\big(}[{G} {\,} {}_{\lambda} {\,} {G}] - [{G}_{(0)}{U} {\,} {}_{\lambda} {\,} {G}] + [{G} {\,} {}_{\lambda} {\,} {G}_{(0)}{U}] - [{G}_{(0)}{U} {\,} {}_{\lambda} {\,} {G}_{(0)}{U}]{\big)}
\\
&= {T} + {\frac{1}{2}}{\partial}{U} + {\lambda}{U} + {\frac{c}{6}}{\lambda}^{2},
\end{align*}
which follows from the second equalities in \eqref{eq: equiv def of N=2 sconf AB} and \eqref{eq: equiv def of N=2 sconf CD}.

[{T} {\,} {}_{\lambda} {\,} {U}_{(0)}({G}_{(0)}{U})] 
&= [{T} {\,} {}_{\lambda} {\,} [{U} {\,} {}_{\gamma} {\,} {G}_{(0)}{U}]] 
= [[{T} {\,} {}_{\lambda} {\,} {U}] {\,} {}_{{\lambda} + {\gamma}} {\,} {G}_{(0)}{U}] + [{U} {\,} {}_{\gamma} {\,} [{T} {\,} {}_{\lambda} {\,} {G}_{(0)}{U}]]
\\
&= [({\partial} + {\lambda}){U} {\,} {}_{{\lambda} + {\gamma}} {\,} {G}_{(0)}{U}] + [{U} {\,} {}_{\gamma} {\,} ({\partial} + {\frac{3}{2}}{\lambda}){G}_{(0)}{U}]
\\
&= -{\gamma}[{U} {\,} {}_{{\lambda} + {\gamma}} {\,} {G}_{(0)}{U}] + ({\partial} + {\frac{3}{2}}{\lambda} +{\gamma})[{U} {\,} {}_{\gamma} {\,} {G}_{(0)}{U}]
\\
&= ({\partial} + {\frac{3}{2}}{\lambda}){U}_{(0)}({G}_{(0)}{U}).
\end{align*}
Hence, ${G}^{\pm}$ and ${\Phi}$ are all odd primary vectors of conformal weight ${\frac{3}{2}}$, defined by
\begin{align*}
{\Phi} := - {\frac{1}{8}}({G} + {U}_{(0)}({G}_{(0)}{U})),
{\quad}
{G}^{+} := {\frac{1}{2}}({G} - {G}_{(0)}{U}) + {\Phi},
{\quad}
{G}^{-} := {\frac{1}{2}}({G} + {G}_{(0)}{U}) - {2}{\Phi}.
\end{align*}

\vskip 1mm

Next, we compute the OPEs of ${G}^{\pm}$ with ${U}$. Using equations \eqref{eqa: G and U}, \eqref{eqa: U and GU}, together with
\begin{align}
\label{eqa: U and Phi}
[{U} {\,} {}_{\lambda} {\,} {\Phi}] = -3{\Phi},
\end{align}
we have:
\begin{align*}
[{U} {\,} {}_{\lambda} {\,} {G}^{+}] 
&= {\frac{1}{2}}[{U} {\,} {}_{\lambda} {\,} {G}] - {\frac{1}{2}}[{U} {\,} {}_{\lambda} {\,} {G}_{(0)}{U}] + [{U} {\,} {}_{\lambda} {\,} {\Phi}]
\\
&= - {\frac{1}{2}}{G}_{(0)}{U} - {\frac{1}{2}}{U}_{(0)}({G}_{(0)}{U}) - {3}{\Phi}
\\
&= - {\frac{1}{2}}{G}_{(0)}{U} - {\frac{1}{2}}(- {8}{\Phi} - {G})- {3}{\Phi} 
= {G}^{+},
\end{align*}
since ${U}_{(0)}({G}_{(0)}{U}) = - {8}{\Phi} - {G}$ by definition of ${\Phi}$. Similarly, using \eqref{eqa: G and U}, \eqref{eqa: U and GU}, and \eqref{eqa: U and Phi}, we also obtain
\begin{align}
\label{eqa: U and G-}
[{U} {\,} {}_{\lambda} {\,} {G}^{-}] = - {G}^{-}.
\end{align}

\vskip 1mm

On the other hand, using the Jacobi identity of vertex algebras together with \eqref{eqa: G and U}, we have
\begin{align}
\label{eqa: G and GU}
[{G} {\,} {}_{\lambda} {\,} {G}_{(0)}{U}] = ({\partial} + {2}{\lambda}){U}.
\end{align}
Hence, using \eqref{eqa: G and U}, \eqref{eqa: U and GU}, \eqref{eqa: G and GU}, and the following OPEs:
\begin{align*}
[{U} {\,} {}_{\lambda} {\,} {U}] = {\frac{c}{3}}{\lambda},
\end{align*}
\begin{align}
\label{eqa: G and Phi}
[{G} {\,} {}_{\lambda} {\,} {\Phi}] = {G}_{(0)}{\Phi},
\end{align}
we obtain
\begin{align}
[{G}_{(0)}{U} {\,} {}_{\lambda} {\,} {G}_{(0)}{U}] 
&= [{G}_{(0)}{U} {\,} {}_{\lambda} {\,} [{G} {\,} {}_{\gamma} {\,} {U}]] 
= [[{G}_{(0)}{U} {\,} {}_{\lambda} {\,} {G}] {\,} {}_{{\lambda} + {\gamma}} {\,} {U}] - [{G} {\,} {}_{\gamma} {\,} [{G}_{(0)}{U} {\,} {}_{\lambda} {\,} {U}]]
\notag
\\
&= ({\gamma} - {\lambda})[{U} {\,} {}_{{\lambda} + {\gamma}} {\,} {U}] - [{G} {\,} {}_{\gamma} {\,} {G}] - {8}[{G} {\,} {}_{\gamma} {\,} {\Phi}]
\label{eqa: GU and GU}
\\
&= {\frac{c}{3}}({\gamma}^{2} - {\lambda}^{2}) - ({2}{T} + {\frac{c}{3}}{\gamma}^{2}) - {8}{G}_{(0)}{\Phi} 
= - ({2}{T} + {\frac{c}{3}}{\lambda}^{2}) - {8}{F},
\notag
\end{align}
where ${F} := {G}_{(0)}{\Phi}$.

\vskip 1mm

Now, using \eqref{eqa: G and GU}, \eqref{eqa: G and Phi}, \eqref{eqa: GU and GU}, together with the following OPEs:
\begin{align*}
[{G}_{(0)}{U} {\,} {}_{\lambda} {\,} {\Phi}] = - {F},
{\quad}
[{\Phi} {\,} {}_{\lambda} {\,} {\Phi}] = 0,
\end{align*}
we have:
\begin{align*}
[{G}^{+} {\,} {}_{\lambda} {\,} {G}^{+}] 
= [{\frac{1}{2}}{G} - {\frac{1}{2}}{G}_{(0)}{U} + {\Phi} {\,} {}_{\lambda} {\,} {\frac{1}{2}}{G} - {\frac{1}{2}}{G}_{(0)}{U} + {\Phi}] = 0,
\end{align*}
by direct expansion. Similarly, we also have the following OPEs
\begin{align}
[{G}^{+} {\,} {}_{\lambda} {\,} {G}^{-}] &= {T} + {\frac{1}{2}}{\partial}{U} + {\lambda}{U} + {\frac{c}{6}}{\lambda}^{2},
\label{eqa: G+ and G-}
\\
[{G}^{+} {\,} {}_{\lambda} {\,} {\Phi}] &= F,
\label{eqa: G+ and Phi}
\\
[{G}^{-} {\,} {}_{\lambda} {\,} {G}^{-}] &= - {2}{F},
\label{eqa: G- and G-}
\\
[{G}^{-} {\,} {}_{\lambda} {\,} {\Phi}] &= 0.
\label{eqa: G- and Phi}
\end{align}

\vskip 1mm

To complete the proof, we will derive the remaining OPEs of the $N = {\frac{3}{2}}$ superconformal algebra from the OPEs obtained so far. First, using \eqref{eqa: G- and G-}, the OPE of ${F}$ with the conformal vector ${T}$ is given by
\begin{align*}
[{T} {\,} {}_{\lambda} {\,} {F}] 
&= - {\frac{1}{2}}[{T} {\,} {}_{\lambda} {\,} [{G}^{-} {\,} {}_{\gamma} {\,} {G}^{-}]] 
= - {\frac{1}{2}}[[{T} {\,} {}_{\lambda} {\,} {G}^{-}] {\,} {}_{{\lambda} + {\gamma}} {\,} {G}^{-}] - {\frac{1}{2}}[{G}^{-} {\,} {}_{\gamma} {\,} [{T} {\,} {}_{\lambda} {\,} {G}^{-}]]
\\
&= - {\frac{1}{2}}[({\partial} + {\frac{3}{2}}{\lambda}){G}^{-} {\,} {}_{{\lambda} + {\gamma}} {\,} {G}^{-}] - {\frac{1}{2}}[{G}^{-} {\,} {}_{\gamma} {\,} ({\partial} + {\frac{3}{2}}{\lambda}){G}^{-}] 
= ({\partial} + {2}{\lambda}){F}.
\end{align*}
The OPE of ${F}$ with ${G}^{+}$ is given by
\begin{align}
[{G}^{+} {\,} {}_{\lambda} {\,} {F}] 
=& {\,} - {\frac{1}{2}}[{G}^{+} {\,} {}_{\lambda} {\,} [{G}^{-} {\,} {}_{\gamma} {\,} {G}^{-}]] 
= - {\frac{1}{2}}[[{G}^{+} {\,} {}_{\lambda} {\,} {G}^{-}] {\,} {}_{{\lambda} + {\gamma}} {\,} {G}^{-}] + {\frac{1}{2}}[{G}^{-} {\,} {}_{\gamma} {\,} [{G}^{+} {\,} {}_{\lambda} {\,} {G}^{-}]]
\notag
\\
=& {\,} - {\frac{1}{2}}[{T} {\,} {}_{{\lambda} + {\gamma}} {\,} {G}^{-}] - {\frac{1}{2}}{\big(}{\frac{1}{2}}{\lambda} - {\frac{1}{2}}{\gamma}{\big)}[{U} {\,} {}_{{\lambda} + {\gamma}} {\,} {G}^{-}]
\label{eqa: G+ and F}
\\
&+ {\frac{1}{2}}[{G}^{-} {\,} {}_{\gamma} {\,} {T}] + {\frac{1}{2}}{\big(}{\frac{1}{2}}{\partial} + {\lambda} + {\frac{1}{2}}{\gamma}{\big)}[{G}^{-} {\,} {}_{\gamma} {\,} {U}] = 0,
\notag
\end{align}
which follows from \eqref{eqa: U and G-}, \eqref{eqa: G+ and G-}, and \eqref{eqa: G- and G-}. To compute the OPE of ${F}$ with ${G}^{-}$, we can use \eqref{eqa: U and Phi}, \eqref{eqa: G+ and G-}, \eqref{eqa: G+ and Phi}, and \eqref{eqa: G- and Phi}, from which we obtain:
\begin{align*}
[{G}^{-} {\,} {}_{\lambda} {\,} {F}] 
&= [{G}^{-} {\,} {}_{\lambda} {\,} [{G}^{+} {\,} {}_{\gamma} {\,} {\Phi}]] 
= [[{G}^{-} {\,} {}_{\lambda} {\,} {G}^{+}] {\,} {}_{{\lambda} + {\gamma}} {\,} {\Phi}] - [{G}^{+} {\,} {}_{\gamma} {\,} [{G}^{-} {\,} {}_{\lambda} {\,} {\Phi}]]
\\
&= [{T} - {\frac{1}{2}}{\partial}{U} - {\lambda}{U} {\,} {}_{{\lambda} + {\gamma}} {\,} {\Phi}] 
= [{T} {\,} {}_{{\lambda} + {\gamma}} {\,} {\Phi}] - {\frac{1}{2}}({\lambda} - {\gamma})[{U} {\,} {}_{{\lambda} + {\gamma}} {\,} {\Phi}] 
= ({\partial} + {3}{\lambda}){\Phi}.
\end{align*}

\vskip 1mm

We now compute the OPEs of ${F}$ with ${U}$ and ${\Phi}$. Using \eqref{eqa: U and G-}, \eqref{eqa: G- and G-}, and \eqref{eqa: G- and G-}, \eqref{eqa: G- and Phi}, respectively, we obtain:
\begin{align*}
[{U} {\,} {}_{\lambda} {\,} {F}] 
&= -{\frac{1}{2}}[{U} {\,} {}_{\lambda} {\,} [{G}^{-} {\,} {}_{\gamma} {\,} {G}^{-}]] 
= -{\frac{1}{2}}[[{U} {\,} {}_{\lambda} {\,} {G}^{-}] {\,} {}_{{\lambda} + {\gamma}} {\,} {G}^{-}] - {\frac{1}{2}}[{G}^{-} {\,} {}_{\gamma} {\,} [{U} {\,} {}_{\lambda} {\,} {G}^{-}]]
\\
&= {\frac{1}{2}}[{G}^{-} {\,} {}_{{\lambda} + {\gamma}} {\,} {G}^{-}] + {\frac{1}{2}}[{G}^{-} {\,} {}_{\gamma} {\,} {G}^{-}] = - {2}{F},
\end{align*}
and
\begin{align}
\begin{split}
[{\Phi} {\,} {}_{\lambda} {\,} {F}] 
&= - {\frac{1}{2}}[{\Phi} {\,} {}_{\lambda} {\,} [{G}^{-} {\,} {}_{\gamma} {\,} {G}^{-}]] 
\\
&= -{\frac{1}{2}}[[{\Phi} {\,} {}_{\lambda} {\,} {G}^{-}] {\,} {}_{{\lambda} + {\gamma}} {\,} {G}^{-}] + {\frac{1}{2}}[{G}^{-} {\,} {}_{\gamma} {\,} [{\Phi} {\,} {}_{\lambda} {\,} {G}^{-}]] = 0.
\end{split}
\label{eqa: Phi and F}
\end{align}
Finally, the OPE of ${F}$ with itself is given by
\begin{align*}
[{F} {\,} {}_{\lambda} {\,} {F}] 
= [{F} {\,} {}_{\lambda} {\,} [{G}^{+} {\,} {}_{\gamma} {\,} {\Phi}]] 
= [[{F} {\,} {}_{\lambda} {\,} {G}^{+}] {\,} {}_{{\lambda} + {\gamma}} {\,} {\Phi}] + [{G}^{+} {\,} {}_{\gamma} {\,} [{F} {\,} {}_{\lambda} {\,} {\Phi}]] = 0,
\end{align*}
which follows from \eqref{eqa: G+ and Phi}, \eqref{eqa: G+ and F}, and \eqref{eqa: Phi and F}. Therefore, we have completed the proof of Proposition \ref{prop: equiv def of N=3/2 sconf}.
