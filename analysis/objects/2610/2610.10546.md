\label{2p1t1d}
	A mapping $\zeta:[0,+\infty)\times[0,+\infty)\rightarrow\mathbb{R}$ is called a simulation function if the following are satisfied:
	\begin{enumerate}[($\zeta_1$)]
		\item $\zeta(0,0)=0$;
		\item $\zeta(t,s)<s-t$ for all $t,s>0$;
		\item if $\{t_n\}, \{s_n\}$ are sequences in $(0,+\infty)$ with $\displaystyle\lim_{n\to+\infty}t_n=\displaystyle\lim_{n\to+\infty}s_n>0$, then $ \ \ \ \ \ $ $\displaystyle\limsup_{n\to+\infty}\zeta(t_n,s_n)<0.$
	\end{enumerate}

	\begin{enumerate}[$\bullet$]
		
		\item $\zeta_1(t,s)=\alpha s-t$ for all $t,s\in[0,+\infty)$, and $\alpha\in[0,1)$.

	\begin{center}
		$\displaystyle\lim_{n\to +\infty}d(S^n{t_0},t)=0$~ for some $t_0\in M$ implies $\displaystyle\lim_{n\to +\infty}d(S(S^n{t_0}),S(t))=0$.

<!-- remark-2: body_md -->
A few facts regarding  Picard-continuous mappings.
	\begin{itemize}
		\item Every continuous mapping is Picard continuous, but the converse need not be true. Picard continuity only requires continuity along sequences generated by iteration of $S$, not along all convergent sequences. For example, consider $M=[0,1]\subset\mathbb{R}$ and define $S:M\to M$ by 
		\begin{equation*}
			S(r)=\left\{
			\begin{array}{ll}
				0, & \text{for} \ \hbox{$r\in[0,1)$}\\
				\frac{1}{k}, & \text{for}~ r=1 ~\ \hbox{(where, $k$ is a fixed natural number.)}
			\end{array}
			\right.
		\end{equation*}
		Then, it is easy to check that $S$ is not continuous (at $r=1$) but it is Picard continuous.
		
		
		
		\item  Sometimes such mappings are called orbitally continuous mappings, since continuity is required only along the orbit $\{r,Sr,S^2r,\cdots\}$
	\end{itemize}
In the next section, we show that our main result ($i.e.,$ Theorem \ref{4p1th}) is expandable for Picard-continuous mappings as well.

<!-- proposition-1: body_md -->
\label{prop1}
Let $(M,d)$ be a metric space, and $S:M\to M$ be a  polynomial $\mathcal{Z}$-contraction for some $\zeta\in\mathcal{Z}$. Assume that the coefficient functions $\alpha_i(r,s)$ are continuous and there exist $j\in \{1,\ldots, k\}$ and a constant $A_j>0$ such that $\alpha_j(r,s)\geq A_j$ for all $r, s\in M$. Furthermore, assume that $\alpha_0(r,s)=0,  r,s\in M$. Then S is a continuous.
\begin{proof}
Let $\{r_n\}\subset M$ be a sequence such that $r_n\to r$ for some $r\in M$. 
As $\alpha_i$ and $d^i$ are continuous, $\mathbb{P}(r,s)=\sum_{i=0}^{k}\alpha_i(r,s)d^{i}(r,s)$ is also continuous in both variables. Thus,
$$ \lim_{n\to +\infty}\mathbb{P}(r_n,r)=\mathbb{P}(r,r)=0.$$
Now, if $Sr_n=Sr$ for all sufficiently large $n\in\mathbb{N}$, then the result is obvious. Otherwise, assume that $\mathbb{P}(Sr_n,Sr)>0$, then by polynomial $\mathcal{Z}$-contraction condition we get 
\begin{eqnarray*}
	0\leq\zeta(\mathbb{P}(Sr_n,Sr), \mathbb{P}(r_n,r))< \mathbb{P}(r_n,r)-\mathbb{P}(Sr_n,Sr)
\end{eqnarray*}
  implies that $\mathbb{P}(Sr_n,Sr)< \mathbb{P}(r_n,r)$ and by taking limit we get $\displaystyle\lim_{n\to +\infty}\mathbb{P}(Sr_n,Sr)=0$.

 Also, from the hypothesis,  $\alpha_j(r,s)\geq A_j>0$ gives the bound  
\begin{equation}
	\mathbb{P}(Sr_n,Sr)\geq A_jd^{j}(Sr_n,Sr)>0.
\end{equation}
Therefore, we must have $\displaystyle\lim_{n\to +\infty}	d^{j}(Sr_n,Sr)=0$ for any such $j$.

As, $\alpha_0(r,s)=0$, for all $r,s\in M$, we can conclude from above that  $\displaystyle\lim_{n\to +\infty}	d(Sr_n,Sr)=0$ and hence $S$ is continuous.
\end{proof}

<!-- theorem-3: body_md -->
\label{4p1th}
	Let $(M,d)$ be a complete metric space, and $S:M\to M$ be a continuous polynomial $\mathcal{Z}$-contraction for some $\zeta\in\mathcal{Z}$. Assume that the coefficient functions $\alpha_i(r,s)$ are uniformly continuous (with respect to the usual product metric) and there exist $j\in \{1,\ldots, k\}$ and a constant $A_j>0$ such that $\alpha_j(r,s)\geq A_j$ for all $r, s\in M$. Then $S$ has a unique fixed point $r^*\in M$ and for every $r\in M$ the sequence $\{S^n{r}\}_{n\in\mathbb{N}}$ converges to $r^*$(where $S^n:= S \underset{n-\text{times}}{\underbrace{\circ \cdots\circ}} S$).
<!-- end theorem-3 -->
<!-- proof-2: body_md -->
Let $r_0$ be an arbitrary point in $M$. Construct a Picard sequence $\{r_n\}$  generated by $r_0$ as $r_n=Sr_{n-1}(=S^nr_0), n\in\mathbb{N}_0$.
Now, if there exists some $n_0\in\mathbb{N}_0$ such that $r_{n_0}=Sr_{n_0}$, then the result follows immediately.

Therefore, without loss of generality, let us assume that for all $n\in\mathbb{N}_0$, $r_n\not=r_{n+1}$   so that $d(r_{n},r_{n+1})>0$. Since $\mathbb{P}(r_n,r_{n+1}):=\sum_{i=0}^{k}\alpha_i(r_n,r_{n+1})d^{i}(r_n,r_{n+1}),$ we have $\mathbb{P}(r_n,r_{n+1})>0$. As $S$ is polynomial $\mathcal Z$-contraction, we have

	\begin{eqnarray}\label{4p1th3eq}
	\nonumber0&\leq&\zeta\big(\mathbb{P}(Sr_{n},Sr_{n+1}),\mathbb{P}(r_n,r_{n+1}))\\
	\nonumber&<&\mathbb{P}(r_n,r_{n+1})-\mathbb{P}(Sr_{n},Sr_{n+1})\\
    \mathbb{P}(Sr_{n},Sr_{n+1})&<&\mathbb{P}(r_n,r_{n+1}),
	\end{eqnarray}
	which implies that $\mathbb{P}(r_{n+1},r_{n+2})<\mathbb{P}(r_n,r_{n+1})$ for all $n\in\mathbb{N}_0$. Therefore, $\{\mathbb{P}(r_n,r_{n+1})\}_{n=0}^{+\infty}$ is a monotonically decreasing sequence of positive real numbers, and hence there exists $l\geq0$, such that $\displaystyle\lim_{n\to+\infty}\mathbb{P}(r_n,r_{n+1})=l.$ We claim that $l=0.$ On contrary, let us assume that $l>0$ then using $(\zeta_3)$, we obtain
	\begin{equation*}
	0\leq\limsup_{n\to+\infty}\zeta\big(\mathbb{P}(Sr_n,Sr_{n+1}),d(r_n,r_{n+1})\big)<0,
	\end{equation*}
	which is a contradiction and hence $l=0, i.e.,$
	\begin{equation}\label{p1th1eq}
	\lim_{n\to+\infty}\mathbb{P}(r_n,r_{n+1})=0.
	\end{equation} 
	Now, from the hypothesis of our theorem,  $\alpha_j(r,s)\geq A_j>0.$ Therefore, 
	\begin{eqnarray*}
		\nonumber \mathbb{P}(r_n,r_{n+1})&=&\sum_{i=0}^{k}\alpha_i(r_n,r_{n+1})d^{i}(r_n,r_{n+1})\\
		\nonumber&\geq&\alpha_j(r_n,r_{n+1})d^{j}(r_n,r_{n+1})\\
		&\geq&A_jd^{j}(r_n,r_{n+1})>0,
	\end{eqnarray*}
	Thus, from \eqref{p1th1eq} we must have $A_jd^{j}(r_n,r_{n+1})\to0$ (as $n\to+\infty$) and consequently
	\begin{equation}\label{12p1th1eq}
		\lim_{n\to+\infty}d(r_n,r_{n+1})=0.
	\end{equation}
	This shows that the sequence $\{r_n\}$ obtained above is asymptotically regular.
	
	We now prove that $\{r_n\}$ is Cauchy. Suppose, to the contrary, that $\{r_n\}$ is not Cauchy. Then there exists $\epsilon>0$ and $l_0\in\mathbb{N}_0$ with $m(l)>n(l)>l \geq l_0,$ such that
	\begin{equation*}
	d(r_{m(l)}, r_{n(l)}) \ge \epsilon, \qquad d(r_{m(l)}, r_{n(l)-1}) < \epsilon.
	\end{equation*}
	

	Using the triangle inequality and \eqref{12p1th1eq}, we obtain
	\begin{equation}\label{12p1th2eq0}
			\lim_{l\to+\infty} d(r_{m(l)}, r_{n(l)}) = \epsilon. 
			     
	\end{equation}
	
	Also, since $d(r_{n+1}, r_n) \to 0$, another application of the triangle inequality gives
	\begin{equation}\label{12p1th3eq0}
		\lim_{l\to+\infty} d(r_{m(l)+1}, r_{n(l)+1}) = \epsilon.
		         
	\end{equation}
	Now, from \eqref{12p1th1eq} we also have
	$$
	d(r_{m(l)+1}, r_{m(l)}) \to 0, \qquad d(r_{n(l)+1}, r_{n(l)}) \to 0.
	$$
	Consider the product metric $d_\times$ on $M \times M$ defined by
	$$
	d_\times\big((x,y),(u,v)\big) := d(x,u) + d(y,v).
	$$
	Then
	\begin{equation}\label{12p1th4eq}
		d_\times\big((r_{m(l)+1}, r_{n(l)+1}), (r_{m(l)}, r_{n(l)})\big)
	= d(r_{m(l)+1}, r_{m(l)}) + d(r_{n(l)+1}, r_{n(l)}) \to 0. 
	        
	\end{equation}
	
	Since each $\alpha_i$ is uniformly continuous on $M \times M$ with respect to $d_\times$, and \eqref{12p1th4eq} holds, we get
	\[
	\lim_{l\to+\infty} \big| \alpha_i(r_{m(l)+1}, r_{n(l)+1}) - \alpha_i(r_{m(l)}, r_{n(l)}) \big| = 0 \quad \text{for each } i.
	\]
	Combining this with \eqref{12p1th2eq0} and \eqref{12p1th3eq0}, and using the fact that $\mathbb{P}(r,s) = \sum_{i=0}^k \alpha_i(r,s) d^i(r,s)$, we conclude
	
	\begin{eqnarray}\label{12p1th5eq}
		\lim_{l\to+\infty} \mathbb{P}(r_{m(l)+1}, r_{n(l)+1})
		&=
		\lim_{l\to+\infty} \mathbb{P}(r_{m(l)}, r_{n(l)}) &=: L_0. 
		        
	\end{eqnarray}
	
	Moreover, since $\alpha_j \ge A_j > 0$ and $d(r_{m(l)}, r_{n(l)}) \to \epsilon > 0$, we have
	$$
	\mathbb{P}(r_{m(l)}, r_{n(l)}) \ge A_j d^j(r_{m(l)}, r_{n(l)}) \to A_j \epsilon^j > 0,
	$$
	so $L_0 > 0$.
	
	Now, using the polynomial $\mathcal{Z}$-contraction condition for the pair $(r_{m(l)}, r_{n(l)})$, we have
	\[
	\zeta\big(\mathbb{P}(r_{m(l)+1}, r_{n(l)+1}), \mathbb{P}(r_{m(l)}, r_{n(l)})\big) \ge 0.
	\]
	Taking the limit superior as $l \to +\infty$ and applying property $(\zeta_3)$ with \eqref{12p1th5eq} and $L_0 > 0$, we obtain
	\[
	0 \le \limsup_{l\to+\infty} \zeta\big(\mathbb{P}(r_{m(l)+1}, r_{n(l)+1}), \mathbb{P}(r_{m(l)}, r_{n(l)})\big) < 0,
	\]
	which is a contradiction. Therefore, $\{r_n\}$ is Cauchy in $M$. Since $(M,d)$ is complete, therefore there exists  $ r^*\in M$ such that $r_n\stackrel{d}\longrightarrow r^*$.\\
	As $S$ is continuous, we have   
	\begin{equation*}
		r^*=\lim_{n\to +\infty}r_{n+1}=\lim_{n\to +\infty}Sr_n=S(\lim_{n\to +\infty}r_n)=Sr^*,
	\end{equation*}
	this shows $r^*$ is a fixed point of $S$. This completes the proof of existence of fixed point.
	
	Finally, we prove uniqueness. Suppose $u, v \in M$ are two fixed points of $S$ with $u \neq v$. Then $d(u,v) > 0$ and $\mathbb{P}(u,v) > 0$. By the contraction condition and property $(\zeta_2)$,
	\[
	0 \le \zeta\big(\mathbb{P}(S u, S v), \mathbb{P}(u,v)\big) = \zeta\big(\mathbb{P}(u,v), \mathbb{P}(u,v)\big) < \mathbb{P}(u,v) - \mathbb{P}(u,v) = 0,
	\]
	a contradiction. Hence, $u = v$, and the fixed point is unique.

<!-- example-2: body_md -->
\label{exm1}
	Let $M=[0,+\infty)$ be endowed with the usual metric $d(r,s)=|r-s|$. Then $(M,d)$ is a complete metric space. Define the mapping $S:M\to M$ by
	$$S(r)=\sqrt r,\qquad r\in M.$$
		
	\medskip
	
	\noindent
	\textbf{Claim 1: ($S$ is not a $\mathcal Z$-contraction)}
	Suppose that $S$ is a $\mathcal Z$-contraction. Then there exists a simulation function $\zeta\in\mathcal Z$ such that

	$$\zeta\bigl(d(Sr,Ss),d(r,s)\bigr)\ge0	\quad \text{for all } r,s\in M.$$
	For $r>s>0$, we have
	\begin{equation*}
			d(Sr,Ss)=\sqrt r-\sqrt s=\frac{r-s}{\sqrt r+\sqrt s}.
	\end{equation*}

	Letting $r=s+\varepsilon$ with $\varepsilon\to0^+$, we obtain
	\begin{equation*}
	\frac{d(Sr,Ss)}{d(r,s)}=\frac{1}{\sqrt r+\sqrt s}\longrightarrow+\infty\quad (\text{as}\ r,s\to0^+).
	\end{equation*}

	Hence the distance between images cannot be uniformly controlled by the usual distance, which contradicts the definition of a $\mathcal Z$-contraction. Therefore, $S$ is not a $\mathcal Z$-contraction.
	\medskip
	
	\noindent
	\textbf{Claim 2: ($S$ is not a polynomial contraction)}
	If $S$ satisfies a polynomial contraction, that is,
	\begin{equation*}
	\sum_{i=0}^{k} \alpha_i(Sr,Ss)\, d^i(Sr,Ss)
	\leq
	\lambda
	\sum_{i=0}^{k} \alpha_i(r,s)\, d^i(r,s)
	\end{equation*}
	

	for some $\lambda\in[0,1)$ and suitable nonnegative coefficient functions $\alpha_i$.
	
	Again, choosing $r=s+\varepsilon$ with $\varepsilon\to0^+$, we have
	\begin{equation*}
	\frac{d(Sr,Ss)}{d(r,s)}\longrightarrow+\infty\quad (\text{as}\ r,s\to0^+),
	\end{equation*}
	which shows that no polynomial expression in $d(Sr,Ss)$ of finite degree can be dominated by a polynomial in $d(r,s)$ with contraction constant $\lambda<1$. Hence $S$ is not a polynomial contraction.
	
	\medskip
	
	\noindent
	\textbf{Claim 3: ($S$ is Polynomial $\mathcal Z$-contraction )}
	Finally, we show that $S$ is a polynomial $\mathcal Z$-contraction for some suitable $\zeta\in\mathcal{Z}$.	Let $k=2$ and define the coefficient functions
	\begin{equation*}
			\alpha_1(r,s)=1,
		\qquad
		\alpha_2(r,s)=\frac{1}{1+r+s},
	\end{equation*}
	which are uniformly continuous on $M\times M$ and satisfy $\alpha_1(r,s)\ge1>0$ for all $r,s\in M$.
	
	Define a simulation function $\zeta:[0,+\infty)\times[0,+\infty)\to\mathbb R$ by
	\begin{equation*}
	\zeta(t,s)=s-t-\frac{s^2}{1+s}.
	\end{equation*}
	Clearly $\zeta\in\mathcal Z$. Then, for $r,s\in M$ with $r\neq s$, we compute
\begin{equation*}
	\begin{aligned}
		\sum_{i=1}^{2} \alpha_i(Sr,Ss)\, d^i(Sr,Ss)
		&=
		(\sqrt r-\sqrt s)
		+\frac{(\sqrt r-\sqrt s)^2}{1+\sqrt r+\sqrt s}.
	\end{aligned}
\end{equation*}

	Using the inequalities
\begin{equation*}
	\sqrt r-\sqrt s < r-s,
	\qquad
	(\sqrt r-\sqrt s)^2\le r-s,
\end{equation*}
	
	we obtain
\begin{equation*}
		\sum_{i=1}^{2} \alpha_i(Sr,Ss)\, d^i(Sr,Ss)
		<
		(r-s)+\frac{(r-s)^2}{1+r-s}.
\end{equation*}
	
	Consequently,
	\begin{equation*}
		\zeta\!\left(
		\sum_{i=1}^{2} \alpha_i(Sr,Ss)\, d^i(Sr,Ss),\;
		\sum_{i=1}^{2} \alpha_i(r,s)\, d^i(r,s)
		\right)\ge0,
	\end{equation*}
	
	i.e.,
	\begin{equation*}
		\zeta(\mathbb{P}(Sr,Ss),\mathbb{P}(r,s))\geq0\ \ \ \text{for all}\ r,s\in M,
	\end{equation*}
	
	\noindent which shows that $S$ is a polynomial $\mathcal Z$-contraction. Also,
	 $S$ is clearly continuous on $M$. Hence, all conditions of Theorem \ref{4p1th} are satisfied and consequently $S$ admits a unique fixed point $r^*\in M$, (namely $r^*=1$). Moreover, for every $r\in M$ the Picard sequence $\{S^n r\}$ converges to $r^*$.
	
	The above example clearly demonstrates that the class of polynomial $\mathcal Z$-contractions strictly contains both the class of polynomial contractions and the class of $\mathcal Z$-contractions, thereby confirming the genuine extension provided in the main result, $i.e.$, Theorem \ref{4p1th}.

On the similar lines of the proof of Theorem \ref{4p1th}, one can show that $\{r_n\}$ is Cauchy sequence. Therefore, due to the completeness of  $(M,d)$, $r_n\stackrel{d}\longrightarrow r^*$ in $M$. Now, as $S$ is Picard continuous, and $\lim_{n\to +\infty}d(r_n,r^*)=0$ (where, $\{r_n\}$ is a Picard sequence generated by $r_0$, i.e., $r_n=S^nr_0$) we must have  $\lim_{n\to +\infty}d(Sr_n,Sr^*)=\lim_{n\to +\infty}d(r_{n+1},Sr^*)=0.$ Then by the uniqueness of limits in metric space, we must have $Sr^*=r^*$. The proof for uniqueness of the fixed point can be established similarly as it is done in Theorem \ref{4p1th}.

\label{exm3} Let $(M,d)=([0,1],|\cdot|)$ and define the mapping $S:M\to M$ by
	\begin{equation*}
			S(r)=
		\begin{cases}
			0, & r\in[0,1),\\[4pt]
			\dfrac12, & r=1.
		\end{cases}
	\end{equation*}
	Clearly $S$ is discontinuous at $r=1$.
                                                                               
                                                                 
	
	To check that $S$ is Picard continuous.	Let $\{r_n\}$ be a Picard sequence defined by $r_{n+1}=S(r_n)$.
	If $r_0\in[0,1)$, then $r_1=0$ and $r_n=0$ for all $n\ge1$.
	If $r_0=1$, then $r_1=\frac12$, $r_2=0$, and again $r_n=0$ for all $n\ge2$.
	Thus every Picard sequence converges to $0$, and
	$$ S(x_n)\to S(0)=0,$$
	which proves that $S$ is Picard continuous.\\
	Here, show that $S$ is in fact a Polynomial $\mathcal{Z}$-contraction.	Let $k=2$ and define the polynomial functional
	$$	P(t)=t+t^2,$$
	corresponding to the constant coefficients $\alpha_0\equiv0$, $\alpha_1\equiv1$ and
	$\alpha_2\equiv1$. Consider the simulation function
	$$	\zeta(t,s)=\frac{s}{1+s}-\frac{t}{1+t}, \qquad t,s\ge0.$$
	
	Clearly, $\zeta\in\mathcal Z$. For arbitrary $r,s\in M$, either $Sr=Ss$, in which case $P(d(Sr,Ss))=0$,
	or $\{r,s\}=\{1,z\}$ with $z<1$, yielding
	$$d(Sr,Ss)=\tfrac12 \le |1-z|=d(r,s).$$
	Hence
	$$P(d(Sr,Ss)) \le P(d(r,s)) \quad  r,s\in M.$$
	Since the function $t\mapsto \frac{t}{1+t}$ is increasing on $[0,+\infty)$,
	it follows that
	$$\zeta\!\left(P(d(Sr,Ss)),P(d(r,s))\right)\ge0,
	\quad  r,s\in M.$$
	Therefore, $S$ is a polynomial $\mathcal Z$-contraction. By Theorem~\ref{4p2th}, $S$ has a unique fixed point $r^*=0$ and for every $r\in M$,	the Picard sequence $\{S^n r\}$ converges to $0$.

 \begin{equation*}

<!-- example-4: body_md -->
Let $\Omega=(0,\pi)$ and let $1\le p<+\infty$. Consider the problem
\begin{equation}\label{BVP}
	\begin{cases}
		-u''(x)	
		=\sin x+\tanh(u(x)-\sin x)
		+\frac{(u(x)-\sin x)^2}{1+(u(x)-\sin x)^2},
		& x\in(0,\pi),\\[6pt]
		u(0)=u(\pi)=0.
	\end{cases}
\end{equation}

Let $M=L^p(\Omega)$ be endowed with the metric $d$ induced by $\|\cdot\|_{L^p(\Omega)}$, then $(M,d)$ is complete.

\medskip

Consider the Green function  $G:\Omega\times\Omega\to\mathbb{R}$  associated with $-u''$ under Dirichlet boundary conditions, given by
\begin{equation}
G(x,y)=
\begin{cases}
	\dfrac{x(\pi-y)}{\pi}, & x\le y,\\[6pt]
	\dfrac{y(\pi-x)}{\pi}, & x>y.
\end{cases}
\end{equation}

Define a operator $S:M\to M$ by
$$(Su)(x):=\int_\Omega G(x,y)\,f(y,u(y))\,dy,$$
where
\begin{equation}\label{eq:fexample}
	f(x,r)=\sin x+\tanh(r-\sin x)
	+\frac{(r-\sin x)^2}{1+(r-\sin x)^2}.
\end{equation}
Then, $f$ is a Carathéodory function since $f(\cdot,r)$ is measurable
for each $r\in\mathbb R$ and $f(x,\cdot)$ is continuous for \ $x\in\Omega$ a.e.
Moreover, the functions $\tanh$ and
$r\mapsto \frac{r^2}{1+r^2}$ are bounded.
Hence,
$$|f(x,u(x))|\le C(1+|u(x)|)\quad\text{a.e.\ in }\Omega,$$
for some constant $C>0$, which implies
$f(\cdot,u(\cdot))\in L^p(\Omega)$ for all $u\in L^p(\Omega)$.

Also, by assumption, the Green operator is bounded on $L^p(\Omega)$, that is,
$$\|Su\|_{L^p(\Omega)}\le C_G\|f(\cdot,u(\cdot))\|_{L^p(\Omega)},
\quad u\in L^p(\Omega),$$ for some constant $C_G>0$.
Now, take
$$\zeta(t,s)=\frac{s}{1+s}-\frac{t}{1+t},
\qquad t,s\ge0,$$
Then, it is easy to check that $\zeta\in\mathcal Z$ and define
$$\alpha_1(r,s)\equiv1,
\qquad
\alpha_2(r,s)\equiv 1,
\quad r,s\in M.$$
For the function
$$f(x,r)=\sin x+\tanh(r-\sin x)
+\frac{(r-\sin x)^2}{1+(r-\sin x)^2}.$$
Since both $\tanh$ and $t\mapsto \frac{t^2}{1+t^2}$ are globally Lipschitz
functions on $\mathbb R$, there exists a constant $L>1$ such that
\begin{equation}\label{eq:Lipschitzf}
	|f(x,r)-f(x,s)|\le L|r-s|,
	\quad
	\text{for all } r,s\in\mathbb R,\ \text{a.e. }x\in\Omega.
\end{equation}

Using \eqref{eq:Lipschitzf}, we obtain
\begin{equation*}
\begin{aligned}
	\sum_{i=1}^2 \alpha_i(r,s)\,|f(x,r)-f(x,s)|^i
	&=
	|f(x,r)-f(x,s)|
	+|f(x,r)-f(x,s)|^2
	\\
	&\le
	L|r-s|+L^2|r-s|^2.
\end{aligned}
\end{equation*}
On the other hand,
$$\sum_{i=1}^2 \alpha_i(r,s)\,|r-s|^i
=|r-s|+|r-s|^2.$$

Setting $t=|r-s|\ge0$, define
$$\Phi(t)=Lt+L^2t^2,\qquad \Psi(t)=t+t^2.$$
Then
$$\sum_{i=1}^2 \alpha_i(r,s)\,|f(x,r)-f(x,s)|^i \le \Phi(t), \qquad
\sum_{i=1}^2 \alpha_i(r,s)\,|r-s|^i = \Psi(t).$$

Next, consider an auxiliary function
$$\phi(u)=\frac{u}{1+u},\qquad u\ge0.$$
The function $\phi$ is strictly increasing on $[0,+\infty)$.
Therefore,
\begin{equation*}
\phi(\Phi(t))=\frac{\Phi(t)}{1+\Phi(t)}\le\frac{\Psi(t)}{1+\Psi(t)}=\phi(\Psi(t)),
\quad t\ge0.
\end{equation*}

Consequently,
\begin{equation*}
\frac{
	\sum_{i=1}^2 \alpha_i(r,s)\,|f(x,r)-f(x,s)|^i
}{
	1+\sum_{i=1}^2 \alpha_i(r,s)\,|f(x,r)-f(x,s)|^i
}
\le
\frac{
	\sum_{i=1}^2 \alpha_i(r,s)\,|r-s|^i
}{
	1+\sum_{i=1}^2 \alpha_i(r,s)\,|r-s|^i
}.
\end{equation*}

Hence,
$$\zeta\!\left(
\sum_{i=1}^2 \alpha_i(r,s)\,|f(x,r)-f(x,s)|^i,\;
\sum_{i=1}^2 \alpha_i(r,s)\,|r-s|^i
\right)\ge0,$$
for all $r,s\in\mathbb R$ and for a.e.\ $x\in\Omega$. Therefore, all the assumptions of Theorem~\ref{4p1th} are satisfied by the boundary value problem \ref{BVP} and the operator $S$ admits a unique fixed point $u^*(x)=\sin x$ and the Picard iteration $u_{n+1}=Su_n$ converges to $u^*$ in $L^p(\Omega)$ for any initial function $u_0\in L^p(\Omega)$(Figure~\ref{fig:PicardConvergence} illustrates the convergence of the Picard iteration). Consequently, $u^*$ is the unique weak solution of \eqref{BVP} in $L^p(\Omega)$.
