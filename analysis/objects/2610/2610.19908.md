<!-- theorem-1: body_md -->
\label{main}
There exists a bounded Lipschitz domain $\Omega\subset\mathbb R^3$, starlike about $0$, whose first positive Neumann eigenvalue is simple. A corresponding real eigenfunction $u$ has a continuous representative on $\overline\Omega$ and, after a choice of sign, satisfies
\begin{equation}\label{exact}
u(0)>u(x)\qquad\bigl(x\in\overline\Omega\setminus\{0\}\bigr).
\end{equation}
In particular,
\begin{equation}\label{target}
u(0)>\max_{x\in\partial\Omega}u(x).
\end{equation}
<!-- end theorem-1 -->
<!-- lemma-1: body_md -->
\label{geom}
For the sets in \eqref{Udef}--\eqref{domain}, $U_\delta$ is open, connected, and satisfies $U_\delta=-U_\delta$. For every continuous $g$ on $S^2$,
\begin{equation}\label{averaging}
\int_{U_\delta}g\,d S\longrightarrow
\theta\int_{S^2}g\,d S.
\end{equation}
The domains $\Omega_\delta$ are bounded connected Lipschitz domains, starlike about $0$, with
\[
B_1\subset\Omega_\delta\subsetB_R,
\qquad \partial\Omega_\delta\subset\{x:1\leq|x|\leq R\}.
\]
For fixed $R,\theta$, there are $c,C,a,L>0$, independent of sufficiently small $\delta$, such that:
\begin{itemize}
\item every removed spherical square has diameter between $c\delta$ and $C\delta$, and the distance between distinct removed squares is at least $c\delta$;
\item near every point of $\partial\Omega_\delta$, in a ball of radius $a\delta$ and suitable orthogonal coordinates, the domain is the region on one side of a graph with Lipschitz constant at most $L$.
\end{itemize}
<!-- end lemma-1 -->
<!-- lemma-2: body_md -->
\label{extension}
For fixed $R,\theta$, there are linear operators
\[
E_\delta:H^1(\Omega_\delta)\longrightarrow H^1(B_R)
\]
such that $E_\delta v=v$ on $\Omega_\delta$ and
\begin{equation}\label{ext}
\|E_\delta v\|_{H^1(B_R)}\leq C\|v\|_{H^1(\Omega_\delta)},
\end{equation}
with $C$ independent of $\delta$. An extension assigns values on $B_R\setminus\Omega_\delta$ to a function initially defined on $\Omega_\delta$. The displayed inequality is the required control on those values and their gradients, including near $r=1$ and $r=R$.
<!-- end lemma-2 -->
<!-- lemma-3: body_md -->
\label{hom}
For the geometry of Lemma~\ref{geom}, the actual-domain Neumann energies
\[
\mathcal E_\delta(v,v)=\int_{\Omega_\delta}|\nabla v|^2\,d x
\]
converge to \eqref{energy}, with mass \eqref{mass}, as follows.
\begin{enumerate}[label=(\roman*)]
\item Every sequence with bounded $L^2$ norm and energy has an extended subsequence converging strongly in $L^2(B_R)$ to $v\in H^1(B_R)$. For every such convergent subsequence,
\[
\liminf_\delta\mathcal E_\delta(v_\delta,v_\delta)\geq\mathcal E(v,v),
\qquad
\int_{\Omega_\delta}v_\delta^2\,d x\longrightarrow\|v\|_m^2.
\]
\item For every $v\in H^1(B_R)$ there is a sequence $v_\delta$ with $E_\delta v_\delta\to v$ in $L^2(B_R)$ and with convergence of its energy to $\mathcal E(v,v)$.
\item The matrix on each tangent plane satisfies \eqref{coercive}; the radial coefficient is exactly $\theta$ and there is no mixed radial--angular term.
\end{enumerate}
Consequently, each indexed Neumann eigenvalue converges to the corresponding limiting eigenvalue, including multiplicities. If the limiting first positive eigenvalue is simple, the normalized first eigenfunctions on the actual domains, after a consistent choice of sign, converge after extension in $L^2(B_R)$ to the normalized limiting eigenfunction. The actual first positive eigenvalue is then simple for sufficiently small $\delta$.
<!-- end lemma-3 -->
<!-- lemma-4: body_md -->
\label{moser}
For fixed $R,\theta$, there are $C,s_0>0$, independent of $\delta$, with the following property. If $x\in\overline\Omega_\delta$, $0<s\leq s_0$, and $v\in H^1(\Omega_\delta\capB(x,2s))$ solves
\[
-\Delta v=F\quad\hbox{in }\Omega_\delta\capB(x,2s)
\]
with zero normal derivative on $\partial\Omega_\delta\capB(x,2s)$ in the weak sense specified in Section~\ref{proofmoser}, then, for every constant $c\in\mathbb R$ and $F\in L^\infty$,
\begin{equation}\label{local}
\|v-c\|_{L^\infty(\Omega_\delta\capB(x,s))}
\leq C\Big[s^{-3/2}\|v-c\|_{L^2(\Omega_\delta\capB(x,2s))}
 +s^2\|F\|_{L^\infty}\Big].
\end{equation}
The estimate holds also when $x$ is near $r=1$, near a coordinate-region boundary, or near $r=R$. Moreover, for every $\Lambda<\infty$ there is a constant $C=C(R,\theta,\Lambda)$ such that every Neumann eigenfunction normalized by $\int_{\Omega_\delta}u_\delta^2\,dx=1$, with eigenvalue at most $\Lambda$, satisfies
\begin{equation}\label{globalbound}
\|u_\delta\|_{L^\infty(\Omega_\delta)}\leq C.
\end{equation}
Each has a continuous representative on $\overline\Omega_\delta$; the same local estimate holds for its boundary values.
<!-- end lemma-4 -->
<!-- lemma-5: body_md -->
\label{trial}
Let $\lambda_{\rm rad}$ denote the first positive eigenvalue of \eqref{energy} restricted to radial functions. With \eqref{theta},
\begin{equation}\label{radbound}
\lambda_{\rm rad}\leq C R^{-5/2}
\end{equation}
for large $R$, with $C$ independent of $R$.
<!-- end lemma-5 -->
<!-- lemma-6: body_md -->
\label{sturm}
Put
\[
p(r)=r^2\ (0<r<1),\qquad p(r)=\theta r^2\ (1<r<R).
\]
The first positive radial eigenvalue is simple. A corresponding real eigenfunction $f$, chosen with $f(0)>0$, has exactly one zero in $(0,R)$, has weighted mean zero, and satisfies \eqref{ode} with equal one-sided values of $pf'$ at $r=1$. In the unit ball,
\begin{equation}\label{sinc}
f(r)=f(0)\frac{\sin(\sqrt\lambda\,r)}{\sqrt\lambda\,r},
\end{equation}
with the value at $r=0$ understood by continuity.
