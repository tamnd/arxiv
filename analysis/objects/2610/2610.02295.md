\label{thm:liu-bound}

 c_{\mathrm{UC}}\ge c_{\mathrm L}>0.382709087918735.

<!-- definition-1: body_md -->
Following Liu~\cite[Definition~1]{liu}, a \emph{protocol} is a family
\[
 \Pi=(\Pi_{s,t})_{(s,t)\in[0,1]^2}
\]
such that $\Pi_{s,t}$ is a coupling of $\operatorname{Ber}(s)$ and $\operatorname{Ber}(t)$.
<!-- end definition-1 -->
<!-- proposition-1: body_md -->
\label{prop:liu-criterion}
Let $c>0$.  For $j=1,\ldots,N$, let $\Pi^{(j)}$ be a protocol and let $w_j\ge0$ with $\sum_jw_j=1$.  For every probability measure $\mu$ on $[0,1]$, let $\mathcal K_j(\mu)$ be a nonempty set of couplings of $\mu$ with itself containing every induced law produced by $\Pi^{(j)}$ with common marginal $\mu$.

Suppose that there is $\gamma>1$ such that, for every probability measure $\mu$ on $[0,1]$ satisfying
\[
 \int s\,\mu(\,\mathrm d s)\le c,
\]
one has
\[
 \sum_{j=1}^N w_j
 \inf_{\rho\in\mathcal K_j(\mu)}
 \iint H\bigl(\Pi^{(j)}_{s,t}(0,0)\bigr)\,\rho(\,\mathrm d s,\,\mathrm d t)
 \ge \gamma\int H(s)\,\mu(\,\mathrm d s).
\]
Then $c\le c_{\mathrm{UC}}$.
<!-- end proposition-1 -->
<!-- lemma-1: body_md -->
\label{lem:positive-correlation}
Let $P$ be a probability measure on $[0,1]$, let $\rho\in\mathcal C_3(P)$, and let $\psi:[0,1]\to\mathbb R$ be continuous.  Then
\[
 \iint \psi(x)\psi(y)\,\rho(\,\mathrm d x,\,\mathrm d y)
 \ge\left(\int\psi\,\,\mathrm d P\right)^2.
\]
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Choose finite mixtures
\[
 \rho_n=\sum_i\lambda_{i,n}Q_{i,n}^{\otimes2}
\]
converging weakly to $\rho$, as permitted by \eqref{eq:C3-def}, and put $P_n:=\sum_i\lambda_{i,n}Q_{i,n}$.  The first marginal of $\rho_n$ is $P_n$, so $P_n$ converges weakly to $P$.  For every $n$,
\[
 \iint\psi(x)\psi(y)\,\rho_n(\,\mathrm d x,\,\mathrm d y)
 =\sum_i\lambda_{i,n}\left(\int\psi\,\,\mathrm d Q_{i,n}\right)^2
 \ge\left(\int\psi\,\,\mathrm d P_n\right)^2
\]
by Jensen's inequality.  Since $\psi$ is continuous and bounded on $[0,1]$, weak convergence allows passage to the limit.

<!-- lemma-2: body_md -->
\label{lem:R}
For every $x,y\in[0,1]$,
\begin{equation}
 H(r(x,y))^2\ge H(g(x))H(g(y)).
 \label{eq:R}
\end{equation}

<!-- lemma-3: body_md -->
\label{lem:S}
For every $x,y\in[0,1]$,
\begin{equation}
 2\alpha H(xy)+2\beta\sqrt{H(g(x))H(g(y))}
 \ge\frac{xH(y)+yH(x)}{m_*}.
 \label{eq:S}
\end{equation}

<!-- proposition-2: body_md -->
\label{prop:fixed-two-construction}
Let $P$ be a probability measure on $[0,1]$ and let $\rho\in\mathcal C_3(P)$.  Then
\begin{equation}
 \alpha\iint H(xy)\,P(\,\mathrm d x)P(\,\mathrm d y)
 +\beta\iint H(r(x,y))\,\rho(\,\mathrm d x,\,\mathrm d y)
 \ge \frac{m(P)}{m_*}\int H(x)\,P(\,\mathrm d x).
 \label{eq:fixed-two-construction}
\end{equation}
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
Taking square roots in \eqref{eq:R} and applying Lemma~\ref{lem:positive-correlation} gives
\[
 \iint H(r(x,y))\,\rho(\,\mathrm d x,\,\mathrm d y)
 \ge\left(\int b\,\,\mathrm d P\right)^2.
\]
Next integrate \eqref{eq:S} with respect to $P(\,\mathrm d x)P(\,\mathrm d y)$.  The right-hand side becomes, by Fubini,
\[
 \frac{2m(P)}{m_*}\int H\,\,\mathrm d P.
\]
Dividing by $2$ gives
\[
 \alpha\iint H(xy)\,\,\mathrm d P\,\mathrm d P
 +\beta\left(\int b\,\,\mathrm d P\right)^2
 \ge\frac{m(P)}{m_*}\int H\,\,\mathrm d P.
\]
Replacing the squared integral by the preceding lower bound for the Example~5 integral proves \eqref{eq:fixed-two-construction}.

<!-- proof-3: body_md -->
Fix $0<c<c_{\mathrm L}$ and let $\mu$ be a probability measure on $[0,1]$ with $\int s\,\mu(\,\mathrm d s)\le c$.  Let $S$ be a $[0,1]$-valued random variable with distribution $\mu$, and let $P$ be the distribution of $1-S$.  By \eqref{eq:mean-reflection},
\[
 m(P)\ge1-c>m_*.
\]
Set
\[
 \gamma:=\frac{1-c}{m_*}>1.
\]
Apply Proposition~\ref{prop:liu-criterion} to the independent protocol and Example~5 with weights $\alpha,\beta$, taking
\[
 \mathcal K_0(\mu)=\{\mu\otimes\mu\},
 \qquad
 \mathcal K_5(\mu)=\mathcal C_3(\mu).
\]
For any $\rho\in\mathcal C_3(\mu)$, reflect both coordinates: if $(X,Y)$ has joint law $\rho$, consider the law of $(1-X,1-Y)$.  Its two marginals are $P$, and it belongs to $\mathcal C_3(P)$ by \eqref{eq:C3-def}: each product measure $Q^{\otimes2}$ is sent to the square of the one-dimensional law of $1-X$ when $X$ has distribution $Q$, and convex combinations and weak limits are preserved.  Moreover $H(1-s)=H(s)$.  Proposition~\ref{prop:fixed-two-construction} therefore bounds the left-hand side in Liu's criterion from below by
\[
 \frac{m(P)}{m_*}\int H\,\,\mathrm d P
 \ge\gamma\int H\,\,\mathrm d\mu.
\]
Thus $c\le c_{\mathrm{UC}}$.  Letting $c\uparrow c_{\mathrm L}$ proves the theorem.

<!-- lemma-4: body_md -->
\label{lem:entropy-one-variable}
For $0<u<1/2$,
\begin{equation}
 -\frac{\Lambda''(u)}{\Lambda'(u)}
 >\frac1u+\frac1{1-u}.
 \label{eq:entropy-one-variable}
\end{equation}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
For $0<u<1/2$, put
\[
 \lambda:=\log\frac{1-u}{u}>0,
 \qquad
 q:=-\log(1-u)>0.
\]
Then
\[
 H'(u)=\lambda,
 \qquad
 H(u)=u\lambda+q,
\]
and therefore
\begin{equation}
 M(u)=\frac{u\lambda}{u\lambda+q}.
 \label{eq:M-ul}
\end{equation}
For $0<u<1/2$, both $M(u)$ and $\Lambda'(u)=H'(u)/H(u)$ are positive.  Since $M=u\Lambda'$, we may take logarithms:
\[
 \log M=\log u+\log\Lambda'.
\]
Differentiating term by term gives
\[
 \frac{M'}M=\frac1u+\frac{\Lambda''}{\Lambda'}.
\]
Hence
\[
 -\frac{\Lambda''}{\Lambda'}=-\frac{M'}M+\frac1u,
\]
and therefore
\[
 -\frac{\Lambda''}{\Lambda'}>\frac1u+\frac1{1-u}
 \quad\Longleftrightarrow\quad
 -\frac{M'}M>\frac1{1-u}.
\]
Thus it remains to prove
\begin{equation}
 -\frac{M'}M>\frac1{1-u}.
 \label{eq:M-target}
\end{equation}

We compute the left-hand side explicitly.  Since
\[
 \lambda'=-\frac1{u(1-u)},
\]
taking logarithms in \eqref{eq:M-ul} gives
\[
 \log M=\log u+\log\lambda-\log H(u).
\]
Differentiating term by term gives
\[
 \frac{M'}M
 =\frac1u+\frac{\lambda'}\lambda-\frac{H'(u)}{H(u)}
 =\frac1u-\frac1{u(1-u)\lambda}-\frac{\lambda}{H(u)}.
\]
Hence
\begin{align}
 -\frac{M'}M-\frac1{1-u}
 &=-\frac1u+\frac1{u(1-u)\lambda}+\frac{\lambda}{H(u)}-\frac1{1-u}\nonumber\\
 &=\frac{1-\lambda}{u(1-u)\lambda}+\frac{\lambda}{H(u)}\nonumber\\
 &=\frac{H(u)(1-\lambda)+u(1-u)\lambda^2}
 {u(1-u)\lambda H(u)}\nonumber\\
 &=\frac{q(1-\lambda)+u\lambda(1-u\lambda)}
 {u(1-u)\lambda H(u)}.
 \label{eq:M-numerator}
\end{align}
The denominator in \eqref{eq:M-numerator} is positive, so it remains to show that the numerator is positive.

If $0<\lambda\le1$, then $q(1-\lambda)\ge0$.  Moreover $0<u\lambda\le u<1/2$, so $u\lambda(1-u\lambda)>0$.  Thus the numerator is positive.

Assume now $\lambda>1$.  The elementary estimate
\[
 -\log(1-u)\le\frac{u}{1-u}
\]
gives $q\le u/(1-u)$.  Because $1-\lambda<0$, multiplying the inequality $q\le u/(1-u)$ by $1-\lambda$ reverses the inequality sign and gives
\[
 q(1-\lambda)\ge\frac{u}{1-u}(1-\lambda).
\]
Therefore
\begin{align}
 q(1-\lambda)+u\lambda(1-u\lambda)
 &\ge \frac{u}{1-u}
 \left(1-u\lambda(\lambda+1)+u^2\lambda^2\right).
 \label{eq:lambda-lower}
\end{align}
Now
\[
 u=\frac1{1+e^\lambda}<e^{-\lambda}.
\]
For $\lambda\ge1$ we have $e^\lambda>\lambda(\lambda+1)$.  To see this without any approximation, use the cubic Taylor polynomial:
\[
 e^\lambda>1+\lambda+\frac{\lambda^2}{2}+\frac{\lambda^3}{6}.
\]
Subtracting $\lambda(\lambda+1)$ gives
\[
 1-\frac{\lambda^2}{2}+\frac{\lambda^3}{6}.
\]
Its derivative is $\lambda(\lambda/2-1)$, so its minimum on $[1,\infty)$ is attained at $\lambda=2$ and equals $1/3$.  Hence it is positive.  Consequently
\[
 u\lambda(\lambda+1)<1.
\]
The parenthesized factor in \eqref{eq:lambda-lower} is therefore strictly positive: the sum of its first two terms is $1-u\lambda(\lambda+1)>0$, and its last term $u^2\lambda^2$ is nonnegative.  This proves \eqref{eq:M-target}, hence \eqref{eq:entropy-one-variable}.

<!-- lemma-7: body_md -->
\label{lem:M-decreasing}
The function $M(u)=uH'(u)/H(u)$ is strictly decreasing on $(0,1)$.
<!-- end lemma-7 -->
<!-- proof-7: body_md -->
Put $u=e^{-v}$.  By the definition above,
\[
 G(v)=\frac{H(u)}u,
 \qquad
 H(e^{-v})=e^{-v}G(v).
\]
Hence
\[
 \log H(e^{-v})=-v+\log G(v).
\]
Using $-\log(1-z)=\sum_{k\ge1}z^k/k$ with $z=e^{-v}$ gives
\[
 G(v)=v+1-\sum_{k\ge1}\frac{e^{-kv}}{k(k+1)}.
\]
Termwise differentiation yields
\[
 G''(v)=-\sum_{k\ge1}\frac{k e^{-kv}}{k+1}<0.
\]
Since $G(v)>0$,
\[
 (\log G)''=\frac{G''}{G}-\left(\frac{G'}G\right)^2<0:
\]
the first term is strictly negative and the second is nonpositive.  Thus $\log G$ is strictly concave.  The function $v\mapsto-v$ is affine, and adding an affine function preserves strict concavity; hence $v\mapsto\log H(e^{-v})=-v+\log G(v)$ is strictly concave.

Its derivatives can be computed directly.  Since $u=e^{-v}$ and $u'=-u$,
\[
 \frac{\,\mathrm d}{\,\mathrm d v}\log H(e^{-v})
 =\frac{H'(u)}{H(u)}(-u)
 =-\frac{uH'(u)}{H(u)}
 =-M(u).
\]
Differentiating once more,
\[
 \frac{\,\mathrm d^2}{\,\mathrm d v^2}\log H(e^{-v})
 =-M'(u)u'
 =uM'(u)
 =e^{-v}M'(e^{-v}).
\]
Strict concavity makes this quantity negative.  Since $e^{-v}>0$, we obtain $M'(e^{-v})<0$, and therefore $M'(u)<0$ for every $0<u<1$.

Both sides of \eqref{eq:R} are continuous on $[0,1]^2$, so the boundary inequality follows once the inequality is proved in the open square. On the diagonal, $g(x)=r(x,x)$ gives equality. Thus it remains only to prove the strict inequality for $x\ne y$ in the open square. By symmetry assume $0<x<y<1$ and define
\[
 \mathcal L(x,y):=\log H(r(x,y))
\]
and
\begin{equation}
 \Delta_x(y):=\mathcal L(x,y)-\frac12\mathcal L(x,x)-\frac12\mathcal L(y,y).
 \label{eq:Deltax-def}
\end{equation}
Since $g(z)=r(z,z)$, the strict inequality in Lemma~\ref{lem:R} is
\[
 2\mathcal L(x,y)>\mathcal L(x,x)+\mathcal L(y,y),
\]
which is exactly $\Delta_x(y)>0$.  Moreover $\Delta_x(x)=0$.  Thus it suffices to prove $\Delta_x'(s)>0$ for every $s\in(x,y]$, because then
\[
 \Delta_x(y)=\Delta_x(x)+\int_x^y\Delta_x'(s)\,\,\mathrm d s>0.
\]
After renaming $s$ as $y$, the remaining goal is therefore
\[
 \Delta_x'(y)>0\qquad(0<x<y<1).
\]

Differentiate \eqref{eq:Deltax-def} with respect to $y$.  The term $\mathcal L(x,x)$ is constant, while symmetry gives $\mathcal L_1(y,y)=\mathcal L_2(y,y)$.  Hence
\begin{equation}
 \Delta_x'(y)=\mathcal L_2(x,y)-\mathcal L_2(y,y).
 \label{eq:Deltax-derivative}
\end{equation}
We split the proof into two cases according to the value of $g(y)$.

\smallskip
\noindent\emph{Case A: $g(y)<1/2$.}  Then $y<y_0$, and for every $s\in[x,y]$ we have $s,y\le y_0$ and
\[
 r(s,y)\le r(y,y)=g(y)<1/2.
\]
By definition, $\mathcal L(x,y)=\log H(r(x,y))=\Lambda(r(x,y))$, so $\mathcal L=\Lambda\circ r$.  The chain rule gives
\[
 \mathcal L_{12}=\Lambda''(r)r_xr_y+\Lambda'(r)r_{xy}
 =\Lambda'(r)r_xr_y
 \left(\frac{r_{xy}}{r_xr_y}+\frac{\Lambda''(r)}{\Lambda'(r)}\right).
\]
Here $\Lambda'(r)=H'(r)/H(r)>0$ because $r<1/2$, and $r_x,r_y>0$.  By Lemmas~\ref{lem:r-identity} and~\ref{lem:WUV},
\[
 \frac{r_{xy}}{r_xr_y}
 =\frac1r+\frac1{WUV}
 \le\frac1r+\frac1{1-r}.
\]
Lemma~\ref{lem:entropy-one-variable} gives
\[
 \frac{\Lambda''(r)}{\Lambda'(r)}
 <-\frac1r-\frac1{1-r}.
\]
Therefore $\mathcal L_{12}(s,y)<0$ for every $s\in[x,y]$.  Apply the fundamental theorem of calculus to the function $s\mapsto\mathcal L_2(s,y)$:
\[
 \mathcal L_2(y,y)-\mathcal L_2(x,y)=\int_x^y\mathcal L_{12}(s,y)\,\,\mathrm d s<0.
\]
Thus $\mathcal L_2(x,y)>\mathcal L_2(y,y)$, and equation~\eqref{eq:Deltax-derivative} gives $\Delta_x'(y)>0$.

\smallskip
\noindent\emph{Case B: $g(y)\ge1/2$.}  By Lemma~\ref{lem:M-decreasing}, $M$ is strictly decreasing on $(0,1)$.

Next, differentiating $\mathcal L$ with respect to its second variable gives
\begin{equation}
 \mathcal L_2(x,y)=\frac{r_y(x,y)}{r(x,y)}M(r(x,y))
 =A_x(y)M(r(x,y)),
 \label{eq:K2-M}
\end{equation}
where
\begin{equation}
 A_x(y):=\frac1y-
 \frac{1-x}{1+(1-x)(1-y)}.
 \label{eq:Ax-def}
\end{equation}
For $0<x<y<1$, the derivative $r_y$ is positive, so $A_x(y)>0$.  Moreover the function
\[
 u\longmapsto\frac{u}{1+u(1-y)}
\]
is strictly increasing for $u\ge0$.  Since $1-x>1-y$, equation \eqref{eq:Ax-def} gives
\begin{equation}
 0<A_x(y)<A_y(y).
 \label{eq:A-order}
\end{equation}
Also $r(x,y)<r(y,y)=g(y)$ because $r$ is strictly increasing in its first variable.

If $r(x,y)<1/2\le g(y)$, then
\[
 M(r(x,y))>0\ge M(g(y)),
\]
so combining \eqref{eq:K2-M} with \eqref{eq:A-order} gives
\[
 \mathcal L_2(x,y)>\mathcal L_2(y,y).
\]
If $1/2\le r(x,y)<g(y)$, then the strict decrease of $M$ gives
\[
 M(g(y))<M(r(x,y))\le0.
\]
Therefore
\[
 A_x(y)M(r(x,y))
 >A_x(y)M(g(y))
 >A_y(y)M(g(y)),
\]
where the second inequality reverses because $M(g(y))<0$ and $A_x(y)<A_y(y)$.  Again $\mathcal L_2(x,y)>\mathcal L_2(y,y)$.  Thus $\Delta_x'(y)>0$ in every case.

Both cases give $\Delta_x'(y)>0$. Hence the integral identity following \eqref{eq:Deltax-def} gives $\Delta_x(y)>0$, which proves \eqref{eq:R} in the open square.  Continuity gives the boundary inequality.

<!-- lemma-8: body_md -->
\label{lem:G-properties}
The function $G$ is increasing and concave on $[0,\infty)$, and strictly increasing and strictly concave on $(0,\infty)$.  For every $u\ge0$,
\[
 u+1-e^{-u}\le G(u)\le u+1.
\]
<!-- end lemma-8 -->
<!-- proof-10: body_md -->
Starting from the definition of $H$ with $x=e^{-u}$,
\[
 G(u)=\frac{-x\log x-(1-x)\log(1-x)}x
 =u+(e^u-1)\bigl[-\log(1-e^{-u})\bigr].
\]
Using $-\log(1-z)=\sum_{k\ge1}z^k/k$ at $z=e^{-u}$ and multiplying by $e^u-1$ gives
\begin{equation}
 G(u)=u+1-\sum_{k\ge1}\frac{e^{-ku}}{k(k+1)}.
 \label{eq:G-series}
\end{equation}
The series and its first two derivatives converge locally uniformly for $u>0$, so termwise differentiation is valid there.  We obtain
\[
 G'(u)=1+\sum_{k\ge1}\frac{e^{-ku}}{k+1}>0,
 \qquad
 G''(u)=-\sum_{k\ge1}\frac{k e^{-ku}}{k+1}<0.
\]
This proves strict monotonicity and strict concavity on $(0,\infty)$.  Together with $G(0)=0$, these properties extend in the non-strict form to $[0,\infty)$.

Finally, $e^{-ku}\le e^{-u}$ for $k\ge1$, and
\[
 \sum_{k\ge1}\frac1{k(k+1)}
 =\sum_{k\ge1}\left(\frac1k-\frac1{k+1}\right)=1.
\]
Hence
\[
 0\le\sum_{k\ge1}\frac{e^{-ku}}{k(k+1)}
 \le e^{-u}\sum_{k\ge1}\frac1{k(k+1)}=e^{-u}.
\]
Writing the series in \eqref{eq:G-series} as $S(u)$, the estimate just obtained is $0\le S(u)\le e^{-u}$. Hence
\[
 u+1-e^{-u}\le u+1-S(u)=G(u)\le u+1,
\]
which gives the two stated bounds.

 W r_{xy}=\frac{r_xr_y}{xy}+\tau_0xy.

 W>0,

 1+\tau_0(1-2x)(1-y)\ge1-\tau_0>0,

 \qquad

Since $\Delta_x(x)=0$, it is enough to prove $\partial_y\Delta_x(y)>0$.  Differentiating the last term along the diagonal gives

 \frac{\,\mathrm d}{\,\mathrm d y}L(y,y)=L_1(y,y)+L_2(y,y)=2L_2(y,y),

 \label{eq:tuned-r-ratio-bound}
\end{equation}

\label{thm:ceiling}
Let $m>0$.  Suppose that, for some $N\ge1$, there are weights
\[
 w_0,w_1,\ldots,w_N\ge0,
 \qquad \sum_{j=0}^Nw_j=1,
\]

\[
 \psi_j(1)=0,
 \qquad
 \psi_j(x)^2\le\log2\quad(0\le x\le1),
\]
such that every probability measure $P$ on $[0,1]$ satisfies
\begin{equation}
 w_0\iint H(xy)\,P(\,\mathrm d x)P(\,\mathrm d y)
 +\sum_{j=1}^Nw_j\left(\int\psi_j\,\,\mathrm d P\right)^2
 \ge \frac{m(P)}m\int H\,\,\mathrm d P.
 \label{eq:mw-certificate}
\end{equation}
Then $m\ge\widehat m$.  Consequently
\[
 1-m\le\widehat c=0.3828852599667\ldots.
\]

Because \eqref{eq:mw-certificate} holds for every probability measure, fix $x\in[0,1]$ and take $P=\delta_x$, the Dirac probability measure assigning mass $1$ to $x$.  For this measure,
\[
 m(P)=x,\qquad
 \iint H(uv)\,P(\,\mathrm d u)P(\,\mathrm d v)=H(x^2),
\]
\[
 \int\psi_j\,\,\mathrm d P=\psi_j(x),
 \qquad
 \int H\,\,\mathrm d P=H(x).
\]
Substitution in \eqref{eq:mw-certificate}, followed by multiplication by $m>0$, gives
\begin{equation}
 m\left[
 w_0H(x^2)+\sum_{j=1}^Nw_j\psi_j(x)^2
 \right]\ge xH(x).
 \label{eq:point-mass-probe}
\end{equation}

We next obtain a lower bound on the independent weight $w_0$.  Fix $y\in(0,1)$ and, for $0<\eta<1$, set
\[
 P_\eta=(1-\eta)\delta_1+\eta\delta_y.
\]
Expanding the product measure $P_\eta\otimes P_\eta$ gives
\begin{align*}
 \iint H(uv)\,P_\eta(\,\mathrm d u)P_\eta(\,\mathrm d v)
 &=(1-\eta)^2H(1)
   +2\eta(1-\eta)H(y)
   +\eta^2H(y^2)\\
 &=2\eta(1-\eta)H(y)+\eta^2H(y^2).
\end{align*}
For each $j$,
\[
 \int\psi_j\,\,\mathrm d P_\eta
 =(1-\eta)\psi_j(1)+\eta\psi_j(y)
 =\eta\psi_j(y),
\]
and also
\[
 m(P_\eta)=1-\eta(1-y),
 \qquad
 \int H\,\,\mathrm d P_\eta=\eta H(y).
\]
Since $H(y)>0$, divide \eqref{eq:mw-certificate} by $\eta H(y)$.  The resulting inequality is
\begin{align*}
 &w_0\left(2(1-\eta)+\eta\frac{H(y^2)}{H(y)}\right)
 +\sum_{j=1}^Nw_j\eta\frac{\psi_j(y)^2}{H(y)}\\
 &\hspace{45mm}\ge
 \frac{1-\eta(1-y)}m.
\end{align*}
Letting $\eta$ tend to $0$ gives $2w_0\ge1/m$, hence
\begin{equation}
 w_0\ge\frac1{2m}.
 \label{eq:w0-floor}
\end{equation}

Return to \eqref{eq:point-mass-probe}.  Since $\psi_j(x)^2\le\log2$ and $\sum_{j=1}^Nw_j=1-w_0$,
\begin{align*}
 w_0H(x^2)+\sum_{j=1}^Nw_j\psi_j(x)^2
 &\le w_0H(x^2)+(1-w_0)\log2\\
 &=\log2+w_0\bigl(H(x^2)-\log2\bigr).
\end{align*}
The coefficient $H(x^2)-\log2$ is nonpositive.  Using \eqref{eq:w0-floor} therefore gives
\[
 w_0H(x^2)+\sum_{j=1}^Nw_j\psi_j(x)^2
 \le \log2+\frac{H(x^2)-\log2}{2m}.
\]
Combining this estimate with \eqref{eq:point-mass-probe} yields
\begin{align*}
 xH(x)
 &\le m\log2+\frac12\bigl(H(x^2)-\log2\bigr)\\
 &=\frac12H(x^2)+\left(m-\frac12\right)\log2.
\end{align*}
Thus
\[
 m\ge\frac12+\frac{xH(x)-\frac12H(x^2)}{\log2}
 =\frac12+\frac{J(x)}{\log2}
 \qquad(0\le x\le1).
\]
Taking the maximum over $x$ proves $m\ge\widehat m$, and hence $1-m\le\widehat c$.

It remains to justify the numerical value of $\widehat c$ stated in the theorem.  For $x\in(0,1)$,
\begin{align*}
 J'(x)&=-(1-x)\log(1-x)-x\log(1+x),\\
 J''(x)&=\log\frac{1-x}{1+x}+\frac1{1+x},\\
 J'''(x)&=\frac{x+3}{(x+1)^2(x-1)}<0.
\end{align*}
Thus $J''$ is strictly decreasing.  Moreover
\[
 \lim_{x\to0^+}J''(x)=1,
 \qquad
 \lim_{x\to1^-}J''(x)=-\infty,
\]
so $J''$ has a unique zero and $J'$ first increases and then decreases.  Since
\[
 \lim_{x\to0^+}J'(x)=0,
 \qquad
 \lim_{x\to1^-}J'(x)=-\log2<0,
\]
and $J''(x)>0$ for all sufficiently small $x>0$, we have $J'(x)>0$ for all sufficiently small $x>0$.  Therefore $J'$ has exactly one zero in $(0,1)$, and $J$ has a unique maximizer $\widehat x\in(0,1)$.  Proposition~\ref{prop:ceiling-certificate} gives rigorous intervals for $\widehat x$, $\widehat m$, and $\widehat c$, in particular
\[
 \widehat c=0.3828852599667\ldots.
\]

\begin{equation}

 \ge\gamma\int H\,\,\mathrm d P.
 \label{eq:general-product-certificate}
\end{equation}

Fix $y\in(0,1)$.  For every
\[

 m(P_\eta)=1-\eta(1-y)>m,

\]

 +\eta\sum_{j=1}^N w_j\frac{\psi_j(y)^2}{H(y)}
 \ge\gamma.
\]
Because $y$ is fixed and $H(y)>0$, all ratios in this display are finite.  Taking the one-sided limit $\eta\to0^+$,
\[
 \lim_{\eta\to0^+}
 w_0\left(2(1-\eta)+\eta\frac{H(y^2)}{H(y)}\right)=2w_0,
\]

\[
 \lim_{\eta\to0^+}
 \eta\sum_{j=1}^N w_j\frac{\psi_j(y)^2}{H(y)}=0.
\]
The right-hand side is the constant $\gamma$, so the limiting inequality is $2w_0\ge\gamma$.  Hence
\[
 w_0\ge\frac\gamma2.

 \gamma H(x)
 \le\frac\gamma2H(x^2)
   +\left(1-\frac\gamma2\right)\log2.

\end{equation}

which is the stated numerical bound.
