<!-- lemma-1: body_md -->
\label{lem:frame-calculus}
Fix \(\alpha\).  For a bounded symbol \(\sigma\),
\(\|T_{\sigma}^{\alpha}\|\leq\|\sigma\|_\infty\).
For \(\eta\in L^1(X,\nu_\alpha)\), the rank-one integral defining
\(T_{\eta}^{\alpha}\) converges in trace norm and
\begin{equation}\label{eq:trace-symbol}
    \|T_{\eta}^{\alpha}\|_{\mathfrak S_1}
       \leq\int_X|\eta|\,d\nu_\alpha,\qquad
    \operatorname{Tr}T_{\eta}^{\alpha}=\int_X\eta\,d\nu_\alpha.
\end{equation}
More generally, for every bounded operator \(B\) on \(\mathcal{H}_\alpha\),
\begin{equation}\label{eq:weighted-trace-identity}
    \operatorname{Tr}(T_{\eta}^{\alpha}B)
       =\int_X\eta(x)\langle Bk_{x}^{\alpha},k_{x}^{\alpha}\rangle_\alpha
           \,d\nu_\alpha(x).
\end{equation}
The positive measure
\begin{equation}\label{eq:overlap-measure}
    d\Pi_\alpha(x,y):=dm_\alpha(x)\,Q_\alpha(x,dy)
      =\frac{|\langlek_{x}^{\alpha},k_{y}^{\alpha}\rangle_\alpha|^2}
              {c(\alpha)}\,d\nu_\alpha(x)d\nu_\alpha(y)
\end{equation}
is symmetric, and both its marginals are \(m_\alpha\).
Consequently the Berezin transform is a contraction on
\(L^p(X,m_\alpha)\) for \(1\leq p\leq\infty\), and preserves integrals
of \(L^1(m_\alpha)\) functions.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
The norm bound follows from \eqref{eq:analysis-compression}.  The operator
\(k_{x}^{\alpha}\otimesk_{x}^{\alpha}\), defined by
\((u\otimes u)f=\langle f,u\rangle_\alpha u\), has trace norm and trace
equal to \(1\).  Integrating it against \(\eta\) proves
\eqref{eq:trace-symbol}; continuity of \(A\mapsto\operatorname{Tr}(AB)\) on
\(\mathfrak S_1\) proves \eqref{eq:weighted-trace-identity}.
Parseval applied to \(k_{x}^{\alpha}\) gives \(Q_\alpha(x,X)=1\).
Symmetry of \eqref{eq:overlap-measure} then identifies both marginals.
Jensen's inequality followed by the marginal identity gives the
\(L^p\) contraction for \(p<\infty\); the probability-kernel identity gives
the \(L^\infty\) bound.  Fubini gives preservation of integrals.

<!-- theorem-1: body_md -->
\label{thm:frame-concentration}
Fix \(\alpha\), let \(A\subset X\) be Borel with
\(\nu_\alpha(A)<\infty\), and let \(0<\varepsilon<1\).  Then \(C_{\alpha,A}\)
is a positive trace-class contraction and
\[
    \operatorname{Tr} C_{\alpha,A}=\nu_\alpha(A),
    \qquad
    \operatorname{Tr}\bigl(C_{\alpha,A}-C_{\alpha,A}^2\bigr)
       =\int_AQ_\alpha(x,X\setminus A)\,d\nu_\alpha(x).
\]
If
\[
    n_{\alpha,A}(\varepsilon)
       =\#\{j:\lambda_j(C_{\alpha,A})\geq1-\varepsilon\},
\]
where eigenvalues are counted with multiplicity, then
\[
\begin{aligned}
 \nu_\alpha(A)
    -\frac1\varepsilon\int_AQ_\alpha(x,X\setminus A)\,d\nu_\alpha(x)
 &\leq n_{\alpha,A}(\varepsilon)\\
 &\leq \nu_\alpha(A)
    +\frac1{1-\varepsilon}\int_AQ_\alpha(x,X\setminus A)\,d\nu_\alpha(x).
\end{aligned}
\]
Moreover, \(n_{\alpha,A}(\varepsilon)\) is the largest dimension of a
subspace \(E\subset\mathcal{H}_\alpha\) such that every unit vector \(f\in E\)
satisfies
\[
    \int_A|\langle f,k_x^\alpha\rangle_\alpha|^2\,d\nu_\alpha(x)
       \geq1-\varepsilon.
\]
In particular, every orthonormal family
\(\varphi_1,\ldots,\varphi_N\) for which each member has at least
\(1-\varepsilon\) of its frame mass in \(A\) satisfies
\[
    N(1-\varepsilon)\leq\nu_\alpha(A).
\]
Consequently, for any sequence of finite-measure sets \(A_\alpha\), if
\[
    \int_{A_\alpha}Q_\alpha(x,X\setminus A_\alpha)
          \,d\nu_\alpha(x)
       =o\bigl(\nu_\alpha(A_\alpha)\bigr),
\]
then, for every fixed \(0<\varepsilon<1\),
\[
    n_{\alpha,A_\alpha}(\varepsilon)
       =\nu_\alpha(A_\alpha)+o\bigl(\nu_\alpha(A_\alpha)\bigr)
       \qquad(\alpha\to\infty).
\]
<!-- end theorem-1 -->
<!-- proof-2: body_md -->
Lemma~\ref{lem:frame-calculus} gives \(0\leq C_{\alpha,A}\leq I\),
trace class, and \(\operatorname{Tr} C_{\alpha,A}=\nu_\alpha(A)\).  Applying
\eqref{eq:weighted-trace-identity} with \(\eta=\mathbf1_A\) and
\(B=C_{\alpha,A}\) gives
\[
    \operatorname{Tr} C_{\alpha,A}^2
       =\int_AQ_\alpha(x,A)\,d\nu_\alpha(x).
\]
The stated trace-defect formula follows because \(Q_\alpha(x,\cdot)\)
is a probability measure.

Write \(\lambda_j=\lambda_j(C_{\alpha,A})\), and separate the
eigenvalues according as \(\lambda_j\geq1-\varepsilon\) or
\(\lambda_j<1-\varepsilon\).  In the latter case,
\(\lambda_j\leq\varepsilon^{-1}\lambda_j(1-\lambda_j)\), whereas in the
former,
\(1-\lambda_j\leq(1-\varepsilon)^{-1}\lambda_j(1-\lambda_j)\).
Summing these two estimates proves the two bounds for
\(n_{\alpha,A}(\varepsilon)\).  The spectral subspace for
\([1-\varepsilon,1]\) has the asserted concentration property; the converse
dimension bound is the min--max principle.  For an orthonormal family,
sum the concentration inequalities and bound the resulting diagonal sum
of \(C_{\alpha,A}\) by its trace.  The last assertion follows immediately
from the two-sided estimate.

<!-- lemma-2: body_md -->
\label{lem:averaged-berezin}
Let \(\sigma\in C_b(X)\), \(1\leq p<\infty\), and \(K\subset X\) be
compact.  Then
\begin{equation}\label{eq:local-oscillation}
    \int_K\int_X|\sigma(y)-\sigma(x)|^p
          \,Q_\alpha(x,dy)\,dm_\alpha(x)\longrightarrow0.
\end{equation}
In particular,
\(\int_K|\widetilde{\sigma}^{\,\alpha}-\sigma|^p\,dm_\alpha\to0\).
If \(\sigma\in C_c(X)\), then
\begin{equation}\label{eq:global-berezin-convergence}
    \|\widetilde{\sigma}^{\,\alpha}-\sigma\|_{L^p(m_\alpha)}\longrightarrow0.
\end{equation}
For real \(\sigma\in C_c(X)\), also
\(\mathcal D_\alpha(\sigma)\to0\).
<!-- end lemma-2 -->
<!-- proof-3: body_md -->
A compactly supported continuous function that is at least \(1\) on \(K\),
together with \eqref{eq:measure-convergence}, bounds \(m_\alpha(K)\)
uniformly for all sufficiently large \(\alpha\).  Set
\[
    \omega_{\sigma,K}(\delta)
      =\sup\{|\sigma(y)-\sigma(x)|:x\in K,\ d(x,y)<\delta\}.
\]
Continuity and compactness imply
\(\omega_{\sigma,K}(\delta)\to0\) as \(\delta\downarrow0\).
The integral in \eqref{eq:local-oscillation} is bounded by
\[
    \omega_{\sigma,K}(\delta)^p m_\alpha(K)
      +(2\|\sigma\|_\infty)^p
        \int_K Q_\alpha(x,X\setminus B(x,\delta))\,dm_\alpha(x).
\]
First let \(\alpha\to\infty\), then \(\delta\downarrow0\).
Jensen's inequality gives the local Berezin assertion.

If \(\sigma\) is compactly supported, take \(K=\operatorname{supp}\sigma\).
The difference \(\sigma(y)-\sigma(x)\) vanishes unless \(x\in K\) or
\(y\in K\).  Symmetry of \(\Pi_\alpha\) therefore bounds its global
\(p\)-th oscillation integral by twice the integral in
\eqref{eq:local-oscillation}.  Jensen gives
\eqref{eq:global-berezin-convergence}, and
\eqref{eq:dirichlet-trace-defect} proves the last assertion.

<!-- proposition-1: body_md -->
\label{prop:berezin-lieb}
Suppose \(\nu_\alpha(X)<\infty\).  Let \(\sigma\) be a bounded real Borel
function taking values in a compact interval \(I\), and let
\(\psi:I\to\mathbb{R}\) be continuous and convex.  Then
\begin{equation}\label{eq:berezin-lieb}
    \int_X\psi(\widetilde{\sigma}^{\,\alpha})\,dm_\alpha
       \leq\frac1{c(\alpha)}\operatorname{Tr}\psi(T_{\sigma}^{\alpha})
       \leq\int_X\psi(\sigma)\,dm_\alpha.
\end{equation}
<!-- end proposition-1 -->
<!-- proof-4: body_md -->
The space is finite-dimensional.  Choose an orthonormal eigenbasis \(e_j\)
of \(T_{\sigma}^{\alpha}\), with eigenvalues \(\lambda_j\in I\).
For each \(x\), the numbers
\(|\langle e_j,k_{x}^{\alpha}\rangle_\alpha|^2\) sum to \(1\), and
\[
    \widetilde{\sigma}^{\,\alpha}(x)
       =\sum_j\lambda_j|\langle e_j,k_{x}^{\alpha}\rangle_\alpha|^2.
\]
Jensen, integration, and Parseval prove the left inequality.
For each \(j\), the measure
\(|\langle e_j,k_{x}^{\alpha}\rangle_\alpha|^2\,d\nu_\alpha(x)\)
is a probability measure, with \(\sigma\)-average \(\lambda_j\).
Applying Jensen to it and summing over \(j\) proves the right inequality.

<!-- theorem-2: body_md -->
\label{thm:finite-volume-szego}
Suppose \(\nu_\alpha(X)<\infty\) for every \(\alpha\), and, in addition to
\eqref{eq:measure-convergence} and \eqref{eq:averaged-localization}, assume
\begin{equation}\label{eq:total-mass-convergence}
    m_\alpha(X)\longrightarrow\nu(X)<\infty.
\end{equation}
Let \(\sigma\in C_b(X;\mathbb{R})\), and let \(I\) be a compact interval
containing its range.  For every \(\psi\in C(I)\),
\begin{equation}\label{eq:finite-volume-szego}
    \frac1{c(\alpha)}\operatorname{Tr}\psi(T_{\sigma}^{\alpha})
       \longrightarrow\int_X\psi(\sigma)\,d\nu.
\end{equation}
In particular, this holds on compact \(X\) under the two standing
convergence assumptions alone.
<!-- end theorem-2 -->
<!-- proof-5: body_md -->
Vague convergence and \eqref{eq:total-mass-convergence} imply asymptotic
tightness and convergence against \(C_b(X)\).  Indeed, choose
\(0\leq\chi\leq1\) in \(C_c(X)\) with
\(\int_X(1-\chi)\,d\nu\) small, and use
\[
    \int_X(1-\chi)\,dm_\alpha
       =m_\alpha(X)-\int_X\chi\,dm_\alpha
       \longrightarrow\int_X(1-\chi)\,d\nu.
\]
Lemma~\ref{lem:averaged-berezin} on \(\operatorname{supp}\chi\), together with
\(|\widetilde{\sigma}^{\,\alpha}-\sigma|\leq2\|\sigma\|_\infty\), now gives
\(\|\widetilde{\sigma}^{\,\alpha}-\sigma\|_{L^1(m_\alpha)}\to0\).
Uniform continuity of \(\psi\) and boundedness of \(m_\alpha(X)\) imply
\[
    \int_X\bigl|\psi(\widetilde{\sigma}^{\,\alpha})-\psi(\sigma)\bigr|\,dm_\alpha
       \longrightarrow0.
\]
For convex \(\psi\), \eqref{eq:berezin-lieb} and convergence against
\(C_b(X)\) prove \eqref{eq:finite-volume-szego}.
Every real continuous function on \(I\) is uniformly approximable by
piecewise linear functions, each a difference of two continuous convex
functions.  The normalized trace error of an approximation of size
\(\varepsilon\) is at most
\(\varepsilon\,\dim\mathcal{H}_\alpha/c(\alpha)=\varepsilon m_\alpha(X)\).
This proves the result for real \(\psi\), and then for complex \(\psi\)
by linearity.  On compact \(X\), the test function \(1\) gives
\eqref{eq:total-mass-convergence}.

\label{thm:localized-szego}
Let \(\sigma\in C_b(X;\mathbb{R})\), let \(I\) be a compact interval containing
its range, and let \(\psi\in C(I)\).  For every compact \(K\subset X\),
\begin{equation}\label{eq:localized-functional-calculus}
    \int_K
      \left|\langle\psi(T_{\sigma}^{\alpha})k_{x}^{\alpha},k_{x}^{\alpha}
                    \rangle_\alpha-\psi(\sigma(x))\right|
                  \,dm_\alpha(x)\longrightarrow0.
\end{equation}
Consequently, for every \(\eta\in C_c(X)\),
\begin{equation}\label{eq:localized-szego}
    \frac1{c(\alpha)}
       \operatorname{Tr}\bigl(T_{\eta}^{\alpha}\psi(T_{\sigma}^{\alpha})\bigr)
       \longrightarrow\int_X\eta(x)\psi(\sigma(x))\,d\nu(x).
\end{equation}
The operators \(T_{\eta}^{\alpha}\) and \(T_{\sigma}^{\alpha}\) need not commute.
<!-- end theorem-3 -->
<!-- proof-6: body_md -->
Write \(A=T_{\sigma}^{\alpha}\).  By \eqref{eq:compression-variance},
\[
\begin{aligned}
    v_\alpha(x)
       &:=\|(A-\sigma(x)I)k_{x}^{\alpha}\|_\alpha^2\\
       &\leq\int_X|\sigma(y)-\sigma(x)|^2\,Q_\alpha(x,dy).
\end{aligned}
\]
Let \(\omega_\psi(\delta)\) be the modulus of continuity of \(\psi\)
on \(I\).  The spectral probability measure of \(A\) in the vector
\(k_{x}^{\alpha}\) assigns mass at most \(v_\alpha(x)/\delta^2\) to
\(\{t:|t-\sigma(x)|>\delta\}\).  Hence
\[
    \left|\langle\psi(A)k_{x}^{\alpha},k_{x}^{\alpha}\rangle_\alpha
                      -\psi(\sigma(x))\right|
       \leq\omega_\psi(\delta)
          +\frac{2\|\psi\|_\infty}{\delta^2}\,v_\alpha(x).
\]
Integrate over \(K\).  Its \(m_\alpha\)-mass stays bounded, while
Lemma~\ref{lem:averaged-berezin} gives
\(\int_Kv_\alpha\,dm_\alpha\to0\).
Letting first \(\alpha\to\infty\), then \(\delta\downarrow0\), proves
\eqref{eq:localized-functional-calculus}.
Formula \eqref{eq:weighted-trace-identity} identifies the left-hand side
of \eqref{eq:localized-szego} with the integral of
\(\eta(x)\langle\psi(A)k_{x}^{\alpha},k_{x}^{\alpha}\rangle_\alpha\)
against \(m_\alpha\).  The preceding convergence and vague convergence
applied to \(\eta\psi(\sigma)\in C_c(X)\) finish the proof.

<!-- remark-1: body_md -->
\label{rem:rough-symbol-counterexample}
The approximation condition cannot be omitted.  On \(X=[0,1]\), take
\(\mathcal H_n=\mathcal P_{n-1}\) with inner product
\(\langle f,g\rangle=\sum_{j=1}^n f(j/n)\overline{g(j/n)}\).
The cardinal Lagrange polynomials form an orthonormal basis; the
normalized reproducing kernels are continuous, and the frame measure is
\(\nu_n=\sum_{j=1}^n\delta_{j/n}\).
The kernels at the nodes are orthonormal, so averaged off-diagonal
leakage is identically zero.  With \(c(n)=n\), the measures \(m_n\)
converge weakly to Lebesgue measure.  Nevertheless, for
\(\sigma=\mathbf1_{\mathbb Q\cap[0,1]}\), one has
\(T_\sigma^n=I\), hence \(n^{-1}\operatorname{Tr} T_\sigma^n=1\), whereas
\(\int_0^1\sigma(x)\,dx=0\).
Thus symbols must be interpreted as Borel functions with specified
representatives when the frame measures have different null sets.

<!-- proposition-2: body_md -->
\label{prop:riesz-sandwich}
For every \(\alpha\) and \(t>0\),
\begin{equation}\label{eq:riesz-sandwich}
    \int_X(t-\widetilde a^{\,\alpha})_+\,dm_\alpha
       \leq\frac{R_{\alpha,a}(t)}{c(\alpha)}
       \leq\int_X(t-a)_+\,dm_\alpha<\infty,
\end{equation}
where \((t-\infty)_+=0\).  In particular, \(T_a^\alpha\) has compact
resolvent.  Neither measure convergence nor averaged localization is
needed for these assertions.
<!-- end proposition-2 -->
<!-- proof-9: body_md -->
For any finite orthonormal family \(u_1,\ldots,u_N\) in the form domain,
put \(s(x)=\sum_j|\langle u_j,k_{x}^{\alpha}\rangle_\alpha|^2\).
Then \(0\leq s\leq1\), \(\int s\,d\nu_\alpha=N\), and
\[
    \sum_{j=1}^N(t-q_{\alpha,a}(u_j))
       =\int_X(t-a)s\,d\nu_\alpha
       \leq\int_X(t-a)_+\,d\nu_\alpha.
\]
The variational formula for the trace of a positive part proves the
upper bound.  The last integral is finite because \((t-a)_+\) is
continuous and compactly supported.

For \(x\) with \(\widetilde a^{\,\alpha}(x)<\infty\), the vector
\(k_{x}^{\alpha}\) belongs to the form domain and has energy
\(q_{\alpha,a}(k_{x}^{\alpha})=\widetilde a^{\,\alpha}(x)\).
Jensen's inequality for its spectral probability measure gives
\[
    (t-\widetilde a^{\,\alpha}(x))_+
       \leq\langle(t-T_a^\alpha)_+k_{x}^{\alpha},k_{x}^{\alpha}\rangle_\alpha.
\]
The same inequality is trivial when the left side is defined to be zero.
Integrate and use \eqref{eq:positive-trace-frame}.
Finally, for \(r\geq0\),
\(\mathbf1_{[0,r]}(T_a^\alpha)\leq(r+1-T_a^\alpha)_+\).
All bounded spectral intervals therefore have finite rank, which proves
compactness of the resolvent.

\label{thm:riesz-limit}
Let \(a:X\to[0,\infty)\) be continuous with compact sublevel sets.
Assume the standing convergence hypotheses, density of the domains of
the forms \eqref{eq:unbounded-toeplitz-form}, and
\eqref{eq:kernel-moment-tail}.  Then, for every \(t>0\),
\begin{equation}\label{eq:riesz-limit}
    \frac1{c(\alpha)}\operatorname{Tr}(t-T_a^\alpha)_+
       \longrightarrow\int_X(t-a(x))_+\,d\nu(x).
\end{equation}

<!-- proposition-3: body_md -->
\label{prop:radial-spectrum}
The operator \(T_a^\alpha\) has compact resolvent.  Its eigenvalues,
repeated with multiplicity, can therefore be ordered as
\[
    0\leq\lambda_{1,\alpha}(a)\leq\lambda_{2,\alpha}(a)
       \leq\cdots,
\]
with \(\lambda_{j,\alpha}(a)\to\infty\) if \(\mathcal{H}_\alpha\) is
infinite dimensional.  Moreover, \(\nu_\alpha(\{e\})\leq1\).
If \(\nu_\alpha(\{e\})<1\), then
\[
    \lambda_{1,\alpha}(a)
       \geq\sup_{r>0}\rho(r)\bigl(1-V_\alpha(r)\bigr)_+>0,
    \qquad V_\alpha(r)=\nu_\alpha(B(e,r)),
\]
and \(T_a^\alpha\) has compact inverse.  If
\(\nu_\alpha(\{e\})=1\), its kernel is exactly
\(\operatorname{span}\{k_{e}^{\alpha}\}\).

Proposition~\ref{prop:riesz-sandwich}.  Parseval applied to
\(k_{e}^{\alpha}\) gives \(\nu_\alpha(\{e\})\leq1\).

       \geq\rho(r)\bigl(1-V_\alpha(r)\bigr).
\]

<!-- lemma-4: body_md -->
\label{lem:filling-duality}
For every finite \(0\leq N\leq\nu_\alpha(X)\),
\[
\begin{aligned}
    \Phi_{\alpha,a}(N)
       &=\inf_{\substack{0\leq h\leq1\\ \int_Xh\,d\nu_\alpha=N}}
                 \int_X a h\,d\nu_\alpha\\
       &=\int_0^\infty\bigl(N-F_{\alpha,a}(t)\bigr)_+\,dt\\
       &=\sup_{t\geq0}
            \left\{Nt-\int_X(t-a)_+\,d\nu_\alpha\right\}.
\end{aligned}
\]
The infimum is attained.  For \(0<N<\nu_\alpha(X)\), one may take
\begin{equation}\label{eq:classical-filling-density}
    h_N=\mathbf1_{\{a<\tau\}}+\theta\mathbf1_{\{a=\tau\}},
    \qquad \tau=F_{\alpha,a}^{\dagger}(N),\quad 0\leq\theta\leq1,
\end{equation}
where \(\theta\) is chosen to make the total mass \(N\).
<!-- end lemma-4 -->
<!-- proof-14: body_md -->
If \(h\) is admissible, layer cake and \(h\leq1\) give
\[
    \int_Xah\,d\nu_\alpha
       =\int_0^\infty\int_{\{a>t\}}h\,d\nu_\alpha\,dt
       \geq\int_0^\infty\bigl(N-F_{\alpha,a}(t)\bigr)_+\,dt.
\]
Tonelli's theorem applied to the generalized inverse identifies the
last expression with \(\int_0^NF_{\alpha,a}^{\dagger}(s)\,ds\).
For an interior mass \(N\), right continuity of \(F_{\alpha,a}\)
and finiteness of its values imply
\[
    \nu_\alpha(\{a<\tau\})\leq N\leq\nu_\alpha(\{a\leq\tau\}).
\]
Thus \eqref{eq:classical-filling-density} is admissible and attains
the bound.  At the endpoints use \(h=0\), or \(h=1\) if
\(N=\nu_\alpha(X)<\infty\).

Finally, layer cake gives
\[
    \int_X(t-a)_+\,d\nu_\alpha=\int_0^tF_{\alpha,a}(u)\,du.
\]
Since \(N-F_{\alpha,a}(u)\) is nonincreasing, the supremum of its
integral over \([0,t]\) is the integral of its positive part over
\([0,\infty)\).  This proves the last equality, including cases
where the value is infinite.

<!-- theorem-5: body_md -->
\label{thm:occupation-filling}
Let \(0\leq\gamma\leq I\) be trace class, with \(N=\operatorname{Tr}\gamma\).  Then
\begin{equation}\label{eq:occupation-filling-chain}
    \mathcal E_{\alpha,a}(\gamma)
       \geq S_{\alpha,a}(N)
       \geq \Phi_{\alpha,a}(N).
\end{equation}
More precisely,
\begin{equation}\label{eq:spectral-filling-duality}
\begin{aligned}
    S_{\alpha,a}(N)
       &=\min_{\substack{0\leq\eta\leq I\\\operatorname{Tr}\eta=N}}
          \mathcal E_{\alpha,a}(\eta)\\
       &=\sup_{t\geq0}\{Nt-\operatorname{Tr}(t-T_a^\alpha)_+\}.
\end{aligned}
\end{equation}
For integer \(N\), the minimum is attained by projection onto the
first \(N\) eigenvectors.  In particular, every orthonormal family
\(\varphi_1,\ldots,\varphi_N\) satisfies
\begin{equation}\label{eq:orthonormal-filling-bound}
    \sum_{j=1}^Nq_{\alpha,a}(\varphi_j)
       \geq S_{\alpha,a}(N)
       \geq \Phi_{\alpha,a}(N).
\end{equation}
<!-- end theorem-5 -->
<!-- proof-15: body_md -->
The operator inequality \(0\leq\gamma\leq I\) gives
\(0\leq h_\gamma^\alpha\leq1\).  Parseval and Tonelli give
\(\int_Xh_\gamma^\alpha\,d\nu_\alpha=\operatorname{Tr}\gamma=N\), so
Lemma~\ref{lem:filling-duality} implies
\(\mathcal E_{\alpha,a}(\gamma)\geq\Phi_{\alpha,a}(N)\).

Let \(v_j\) be an orthonormal eigenbasis of \(T_a^\alpha\), and put
\(b_j=\langle\gamma v_j,v_j\rangle_\alpha\).  Then
\[
    0\leq b_j\leq1,\qquad \sum_jb_j=N,\qquad
    \mathcal E_{\alpha,a}(\gamma)
       =\sum_j\lambda_{j,\alpha}(a)b_j.
\]
Filling this nondecreasing sequence from the bottom minimizes the last
sum.  Equivalently, apply Lemma~\ref{lem:filling-duality} to counting
measure on the eigenvalue indices.  The minimizer has \(b_j=1\)
for \(j\leq n\), \(b_{n+1}=\theta\), and all other entries zero.
Its cost is \eqref{eq:spectral-filling-function}, and the same lemma
gives the Riesz-mean duality \eqref{eq:spectral-filling-duality}.
Applying the classical lower bound to this minimizing operator proves
\(S_{\alpha,a}(N)\geq\Phi_{\alpha,a}(N)\).
Taking \(\gamma\) to be the projection onto the given orthonormal
family proves \eqref{eq:orthonormal-filling-bound}.

\label{thm:semiclassical-filling}
Suppose that the forms \(q_{\alpha,a}\) are densely defined and
\(m_\alpha\rightharpoonup\nu\) as in
\eqref{eq:measure-convergence}.  Let \(0\leq N_\alpha\leq D_\alpha\)
be finite, write \(N_\alpha=n_\alpha+\theta_\alpha\) with
\(n_\alpha\in\mathbb N\cup\{0\}\) and \(0\leq\theta_\alpha<1\), and
assume \(N_\alpha/c(\alpha)\to\varrho\in[0,\infty)\).
Then
\[
    \liminf_{\alpha\to\infty}
       \frac{1}{c(\alpha)}
       \left(\sum_{j=1}^{n_\alpha}\lambda_{j,\alpha}(a)
              +\theta_\alpha\lambda_{n_\alpha+1,\alpha}(a)\right)
       \geq\int_0^\varrho F_a^{\dagger}(s)\,ds.
\]
This lower bound needs neither averaged localization nor a kernel
moment hypothesis.

If, in addition, \eqref{eq:averaged-localization} and
\eqref{eq:kernel-moment-tail} hold and
\(0<\varrho<\nu(X)\), then
\begin{equation}\label{eq:semiclassical-filling-limit}
    \frac{1}{c(\alpha)}
       \left(\sum_{j=1}^{n_\alpha}\lambda_{j,\alpha}(a)
              +\theta_\alpha\lambda_{n_\alpha+1,\alpha}(a)\right)
       \longrightarrow\int_0^\varrho F_a^{\dagger}(s)\,ds.
\end{equation}
The total limiting volume \(\nu(X)\) may be finite or infinite,
and the distribution of \(a\) may have atoms.

<!-- proof-20: body_md -->
Put \(\varrho_\alpha=N_\alpha/c(\alpha)\) and
\[
    G_\alpha(t)=\int_X(t-a)_+\,dm_\alpha,
    \qquad G(t)=\int_X(t-a)_+\,d\nu.
\]
For each fixed \(t\geq0\), the integrand is continuous and compactly
supported, so \(G_\alpha(t)\to G(t)\).  By the Riesz upper bound
and \eqref{eq:spectral-filling-duality},
\[
    \frac{S_{\alpha,a}(N_\alpha)}{c(\alpha)}
       \geq\varrho_\alpha t-G_\alpha(t).
\]
Take the lower limit and then the supremum over \(t\geq0\).
Lemma~\ref{lem:filling-duality}, applied to \(\nu\), identifies the
result as \(\Phi_a(\varrho)\).  If \(\varrho>\nu(X)\), both
this supremum and \(\Phi_a(\varrho)\) are infinite; the same
argument still applies.

For the matching limit set
\(r_\alpha(t)=R_{\alpha,a}(t)/c(\alpha)\).
Theorem~\ref{thm:riesz-limit} gives \(r_\alpha(t)\to G(t)\)
for every \(t\geq0\).  These are nondecreasing functions of \(t\)
with continuous limit, so the convergence is uniform on every compact
interval.  Since \(F_a(t)\uparrow\nu(X)>\varrho\), choose
\(L>0\) such that
\[
    G(L+1)-G(L)=\int_L^{L+1}F_a(u)\,du>\varrho.
\]
For all sufficiently large \(\alpha\),
\(r_\alpha(L+1)-r_\alpha(L)>\varrho_\alpha\).
Convexity implies, for \(t\geq L+1\),
\[
    r_\alpha(t)\geq r_\alpha(L+1)
       +(t-L-1)\bigl(r_\alpha(L+1)-r_\alpha(L)\bigr).
\]
Consequently the supremum of
\(\varrho_\alpha t-r_\alpha(t)\) is attained in \([0,L+1]\).
The same is true of \(\varrho t-G(t)\).  Uniform convergence on
this interval and \eqref{eq:spectral-filling-duality} now give
\[
    \frac{S_{\alpha,a}(N_\alpha)}{c(\alpha)}
       \longrightarrow\sup_{t\geq0}\{\varrho t-G(t)\}
       =\Phi_a(\varrho).
\]

\label{lem:moment-centers}

for any \(y,z\in X\),

\label{thm:mean-dispersion}
Let \(\varphi_1,\ldots,\varphi_N\) be orthonormal in \(\mathcal{H}_\alpha\),
with \(\mu_{\varphi_j}^\alpha\in\mathcal P_p(X)\).  Then
\begin{equation}\label{eq:p-mean-dispersion}
\begin{aligned}
    \sum_{j=1}^N
       \bigl[\Delta_{\alpha,p}(\varphi_j)
                  +d(e,\mu_{\alpha,p}(\varphi_j))\bigr]^p
       &\geq\sum_{j=1}^Nq_{\alpha,d_e^p}(\varphi_j)\\
       &\geq\int_0^N\bigl(V_\alpha^\dagger(s)\bigr)^p\,ds.
\end{aligned}
\end{equation}
Consequently,
\begin{equation}\label{eq:separate-mean-dispersion}
    \sum_{j=1}^N
       \left[\Delta_{\alpha,p}(\varphi_j)^p
                  +d(e,\mu_{\alpha,p}(\varphi_j))^p\right]
       \geq 2^{1-p}\int_0^N\bigl(V_\alpha^\dagger(s)\bigr)^p\,ds.
\end{equation}
If \(q_{\alpha,d_e^p}\) is densely defined, one may replace the
integral on the right of \eqref{eq:p-mean-dispersion} by
\(\sum_{j=1}^N\lambda_{j,\alpha}(d_e^p)\).

\label{cor:bounded-localization-cardinality}

\(A_\alpha,D>0\).  If

    N\leq A_\alpha
            \left(\frac{D+p}{D}\right)^{D/p}(L+R)^D.

orthonormal sequence whose centers stay in a bounded set and whose
\(p\)-dispersions stay bounded.

\label{thm:rearrangement-umbrella}

\eqref{eq:common-frame-envelope}, with
\(\Phi_\alpha\in L^1(\nu_\alpha)\).  Then
\begin{equation}\label{eq:envelope-capacity}
    N\leq\int_X\min\{1,N\Phi_\alpha(x)\}\,d\nu_\alpha(x)
       =\int_0^\infty\min\{1,N\Phi_\alpha^*(s)\}\,ds.
\end{equation}
For every \(0<\delta<1\),
\[
    N(1-\delta)\leq K_{\alpha,\Phi_\alpha}(\delta).
\]
In particular, an infinite orthonormal sequence cannot have a common
integrable frame envelope.
<!-- end theorem-8 -->
<!-- proof-26: body_md -->
The occupation function satisfies both \(s_\alpha\leq1\) and
\(s_\alpha\leq N\Phi_\alpha\).  Integrating proves the inequality
in \eqref{eq:envelope-capacity}.  Its equality follows from
equimeasurability, or directly from layer cake:
\[
    \int_X\min\{1,N\Phi_\alpha\}\,d\nu_\alpha
       =\int_0^1\nu_\alpha(\{\Phi_\alpha>t/N\})\,dt.
\]
If \(E\) is admissible in \eqref{eq:effective-envelope-volume}, each
state has at least \(1-\delta\) of its mass in \(E\).  Therefore
\[
    N(1-\delta)
       \leq\int_Es_\alpha\,d\nu_\alpha\leq\nu_\alpha(E).
\]

\[
    1\leq\int_X\min\{N^{-1},\Phi_\alpha\}\,d\nu_\alpha.
\]
At a fixed parameter the right side tends to zero as \(N\to\infty\),
by dominated convergence.  This excludes an infinite family.

\label{thm:collective-transport}

\begin{equation}\label{eq:center-dependent-transport}

       \geq\sup_{y\in X}

\end{equation}

\(U_\alpha:[0,\infty)\to[0,\infty)\) satisfies
\begin{equation}\label{eq:uniform-ball-volume}
    \nu_\alpha(B(y,r))\leq U_\alpha(r)
       \qquad(y\in X,\ r>0),
\end{equation}
then
\begin{equation}\label{eq:uniform-volume-transport}
    W_p(\overline\mu_\gamma^\alpha,\sigma)
       \geq
       \left[
          \left(\frac1N\int_0^N(U_\alpha^\dagger(s))^p\,ds\right)^{1/p}
          -\Delta_p(\sigma)
       \right]_+.
\end{equation}
Distances are interpreted as \(+\infty\) if
\(\overline\mu_\gamma^\alpha\) has infinite \(p\)-th moment.
<!-- end theorem-9 -->
<!-- proof-28: body_md -->
The density of \(\overline\mu_\gamma^\alpha\) relative to
\(\nu_\alpha\) is at most \(1/N\).  Scalar filling at any fixed
center \(y\) therefore gives
\[

       =\frac1N\int_Xd(x,y)^p h_\gamma^\alpha(x)\,d\nu_\alpha(x)
       \geq\mathcal R_{\alpha,p}(N;y)^p.
\]

\[

       \geq\mathcal R_{\alpha,p}(N;y).

\eqref{eq:center-dependent-transport}.
Under \eqref{eq:uniform-ball-volume}, one has

radius is then independent of \(y\), and minimizing
\(M_p(\sigma;y)\) proves \eqref{eq:uniform-volume-transport}.
If the source has infinite moment, its distance to the finite-moment
target is infinite by the same moment triangle inequality.

\label{rem:bessel-uncertainty}
The occupation argument also applies to normalized families satisfying
\[

       \leq B\|f\|_\alpha^2.

equivalently, the operator
\(B^{-1}\sum_j\langle\,\cdot\,,\varphi_j\rangle_\alpha\varphi_j\)
is a positive contraction.  Thus a finite Bessel bound gives
quantitative control when exact orthogonality is weakened.

\label{prop:fock-application}
For \(p>0\), let \(0\leq\gamma\leq I\) be trace class, with
\(N=\operatorname{Tr}\gamma\).  Then
\begin{equation}\label{eq:fock-filling-bound}
\begin{aligned}
    \mathcal E_{\alpha,|z|^p}(\gamma)
       &\geq S_{\alpha,|z|^p}(N)\\
       &\geq\int_0^N F_{\alpha,|z|^p}^{\dagger}(s)\,ds\\
       &=\frac{2d}{2d+p}\,(d!)^{p/(2d)}
          \alpha^{-p/2}N^{1+p/(2d)}.
\end{aligned}
\end{equation}
The monomials
\[
    e_\beta^\alpha(z)
       =\left(\frac{\alpha^{|\beta|}}{\beta!}\right)^{1/2}z^\beta,
       \qquad \beta\in\{0,1,\ldots\}^d,
\]
form an orthonormal eigenbasis of \(T_{|z|^p}^\alpha\), with
\begin{equation}\label{eq:fock-radial-eigenvalues}
    \lambda_{\beta,\alpha}(|z|^p)
       =\alpha^{-p/2}
          \frac{\Gamma(|\beta|+d+p/2)}{\Gamma(|\beta|+d)}.
\end{equation}
The eigenvalue at degree \(m\) has multiplicity
\(\binom{m+d-1}{d-1}\).  At fixed \(\alpha\), the last expression in
\eqref{eq:fock-filling-bound} is asymptotic to
\(\sum_{j=1}^N\lambda_{j,\alpha}(|z|^p)\) as \(N\to\infty\).
