<!-- proposition-1: body_md -->
\label{lem:edge-potential}
	\label{thm:full-gradient}
	\label{thm:rotation}
	The potential $\mathcal V$ is independent of the edge orientation and
	determined up to an additive constant.  Moreover:
	\begin{enumerate}[label=\textup{(\roman*)}]
		\item The network map is a gradient step for $\mathcal V$:
		\begin{equation}
			\label{eq:full-gradient-components}
			\nabla\Psi(z)=-A\mathbf g(z),\qquad
			\nabla\mathcal V(\theta)=-BA\mathbf g(B^T\theta),
		\end{equation}
		\begin{equation}
			\label{eq:full-gradient-step}
			\widetilde F_h(\theta)=\theta-h\nabla\mathcal V(\theta).
		\end{equation}
		\item For every $\alpha\in\mathbb R$,
		\begin{equation}
			\label{eq:F-rotation}
			\mathcal V(\theta+\alpha\mathbf 1)=\mathcal V(\theta),\qquad
			\widetilde F_h(\theta+\alpha\mathbf 1)=\widetilde F_h(\theta)+\alpha\mathbf 1 .
		\end{equation}
		\item $\mathbf 1^T\nabla\mathcal V(\theta)=0$, and the lifted map preserves
		the phase sum:
		\begin{equation}
			\label{eq:sum-conserved}
			\mathbf 1^T\widetilde F_h(\theta)=\mathbf 1^T\theta .
		\end{equation}
	\end{enumerate}
	In particular, $F_h$ descends to the $N$-dimensional quotient
	$\mathbb T_P^{N+1}/\mathbb T_P\simeq\mathbb T_P^N$ by common-phase shifts.
<!-- end proposition-1 -->
<!-- proof-1: body_md -->
Differentiation and the chain rule give \eqref{eq:full-gradient-components}
	and \eqref{eq:full-gradient-step}.  Reversing an edge changes the signs of
	both $b_e$ and $b_e^T\theta$; evenness of $\Phi_e$ preserves the
	potential, and oddness of $g_e$ preserves the map.  The only freedom in
	the construction is the additive constant in each primitive.  Finally,
	$B^T\mathbf 1=0$ gives \eqref{eq:F-rotation}, and $\mathbf 1^TB=0$ gives the
	orthogonality and \eqref{eq:sum-conserved}.

\label{thm:inherited-potential}
	\label{thm:reduced-gradient}
	There is a unique smooth \(V:\mathbb T_P^N\to\mathbb R\) satisfying
	\(\mathcal V=V\circ Q\). It is given by
	\begin{equation}
		\label{eq:reduced-potential}
		V(x)=\Psi(Cx),\qquad \nabla V(x)=-C^TA\mathbf g(Cx).
	\end{equation}
	The induced map \(\bar F_h:\mathbb T_P^N\to\mathbb T_P^N\) is
	\begin{equation}
		\label{eq:reduced-gradient-map}
		\bar F_h(x)=x-hM\nabla V(x)
		=x-h\operatorname{grad}_{M^{-1}}V(x),
		\qquad \operatorname{grad}_{M^{-1}}V=M\nabla V.
	\end{equation}

	\(Q\); evaluation at \(Sx\) gives the unique factor \(V(x)=\Psi(Cx)\).

	\begin{equation}
		\label{eq:chain-gradient}
		\nabla_\theta\mathcal V(\theta)=Q^T\nabla V(Q\theta).
	\end{equation}
	Apply \(Q\) to \eqref{eq:full-gradient-step} and use \(QQ^T=M\).
	The metric \eqref{eq:metric} then gives the stated Riemannian gradient.

<!-- theorem-1: body_md -->
\label{thm:finite-estimates}
	\label{thm:full-Lyapunov}
	\label{thm:reduced-Lyapunov}
	\label{thm:diffeomorphism}
	For \(0<h<2/\Lambda\), let \(c_h=h(1-h\Lambda/2)>0\). Then
	\begin{equation}
		\label{eq:reduced-descent}
		\begin{aligned}
			\mathcal V(F_h(\theta))
			&\le\mathcal V(\theta)-c_h\|\nabla\mathcal V(\theta)\|^2,\\
			V(\bar F_h(x))
			&\le V(x)-c_h\nabla V(x)^TM\nabla V(x),
		\end{aligned}
	\end{equation}
	so both potentials decrease strictly away from their critical sets.
	If \(0<h<1/\Lambda\), then \(F_h\) and \(\bar F_h\) are smooth global
	diffeomorphisms.
<!-- end theorem-1 -->
<!-- proof-4: body_md -->
Taylor's inequality on the periodic lift, with increment
	\(-h\nabla\mathcal V(\theta)\), and Lemma~\ref{thm:Hessian-bound} give
	the first descent estimate \cite{Nesterov2004}. The identities
	\(\mathcal V=V\circ Q\) and \eqref{eq:chain-gradient} give
	\[
	\|\nabla\mathcal V(\theta)\|^2
	=\nabla V(Q\theta)^TM\nabla V(Q\theta),
	\]
	and hence the reduced estimate. Positive definiteness of \(M\) gives
	strictness.
	
	For \(h<1/\Lambda\), Lemma~\ref{thm:Hessian-bound} shows that every eigenvalue of
	\(D\widetilde F_h=I_{N+1}-hD^2\mathcal V\) is positive. The induced
	torus map is therefore an orientation-preserving local diffeomorphism.
	Its image is open and compact, hence closed, so connectedness makes
	it surjective. The homotopy
	\(\theta\mapsto\theta-th\nabla\mathcal V(\theta)\), \(0\le t\le1\),
	gives degree one. A local diffeomorphism of a compact manifold is a
	covering map, and positive orientation makes this covering one-sheeted
	\cite{Lee2012SmoothManifolds}. Thus \(F_h\) is a global diffeomorphism.
	Its inverse also commutes with common-phase shifts, so it induces
	a diffeomorphism on the quotient.

<!-- proposition-3: body_md -->
\label{thm:fixed-critical}
	\label{cor:no-periodic}
	\label{thm:approach-critical}
	\label{thm:finite-convergence}
	Let \(0<h<2/\Lambda\). Then
	\begin{equation}
		\label{eq:fixed-critical}
		\operatorname{Fix}(\bar F_h)=\operatorname{Crit}(V),
	\end{equation}
	and every periodic orbit of \(\bar F_h\) is fixed. Every forward orbit
	\(x_{n+1}=\bar F_h(x_n)\) satisfies
	\begin{equation}
		\label{eq:distance-critical}
		\nabla V(x_n)\to0,\qquad
		\operatorname{dist}(x_n,\operatorname{Crit}(V))\to0;
	\end{equation}
	in particular, every accumulation point is fixed. If \(\operatorname{Crit}(V)\) is
	finite, each orbit converges to one of its points.
<!-- end proposition-3 -->
<!-- proof-5: body_md -->
Critical points are fixed by \eqref{eq:reduced-gradient-map}.
	Conversely, a fixed point leaves \(V\) unchanged, so strict descent
	forces criticality. A periodic orbit cannot contain a strict decrease
	and is therefore fixed.
	
	Summing \eqref{eq:reduced-descent} gives
	\[
	c_h\sum_{n=0}^{m-1}\nabla V(x_n)^TM\nabla V(x_n)
	\le V(x_0)-V(x_m).
	\]
	Compactness bounds the right-hand side, so positive definiteness of
	\(M\) gives \(\nabla V(x_n)\to0\). Every accumulation point is
	therefore critical. If the distance to \(\operatorname{Crit}(V)\) did not tend to
	zero, a convergent subsequence at a positive distance from it would
	contradict this conclusion. Also,
	\[
		\operatorname{dist}(x_{n+1},x_n)\le h\|M\nabla V(x_n)\|\longrightarrow0.
	\]
	When the critical set is finite, choose balls around its points with
	pairwise disjoint closures. Their mutual distances have a positive
	lower bound. Eventually the orbit lies in their union and its steps
	are smaller than this bound, so it remains in one ball. Its distance
	to the critical set tends to zero, and hence it converges to that
	ball's critical point.

<!-- proposition-4: body_md -->
\label{thm:Morse-hyperbolic}
	Let \(0<h<1/\Lambda\). Each non-degenerate critical point \(p\) of \(V\)
	is a hyperbolic fixed point of \(\bar F_h\), and
	\begin{equation}
		\label{eq:index-equality}
		\dim W^u(p)=\operatorname{ind}_{\mathrm M}(p).
	\end{equation}
	Such a point is a sink exactly when it is a strict local minimum,
	a source exactly when it is a strict local maximum, and a saddle
	otherwise.
<!-- end proposition-4 -->
<!-- proof-6: body_md -->
At a non-degenerate critical point, put \(H_p=D^2V(p)\). Then
	\[
		D\bar F_h(p)=I_N-hMH_p.
	\]
	The matrix \(MH_p\) is similar to \(M^{1/2}H_pM^{1/2}\), which is
	symmetric and has the inertia of \(H_p\)
	\cite{HornJohnson2012}. Its eigenvalues \(\mu\) are thus real and
	nonzero. They are also the nonzero eigenvalues of
	\(D^2\mathcal V(\theta)=Q^TH_pQ\), for \(Q\theta=p\), so
	\(|\mu|\le\Lambda\) by Lemma~\ref{thm:Hessian-bound}. For \(0<h<1/\Lambda\), positive \(\mu\)
	give multipliers \(0<1-h\mu<1\), while negative \(\mu\) give
	\(1-h\mu>1\). This proves hyperbolicity, the index formula and the
	classification.
<!-- end proof-6 -->
<!-- remark-2: body_md -->
The same multiplier calculation gives \(-1<1-h\mu<1\) for positive
	\(\mu\), and \(1-h\mu>1\) for negative \(\mu\), throughout
	\(0<h<2/\Lambda\). Thus non-degenerate critical points retain
	multipliers off the unit circle in the full descent interval.

\label{thm:global-dynamics}
	\label{cor:almost-everywhere}

	of \(\bar F_h\) converges to a critical point, and almost every orbit

A Morse function on the compact torus has finitely many critical points,

	dimension less than \(N\) by Proposition~\ref{thm:Morse-hyperbolic} and

	All other initial conditions converge to local minima.

<!-- lemma-3: body_md -->
\label{lem:adjoint-transversality}
	Let $G$ be a $C^1$ diffeomorphism of $\mathbb T^N$, let $p,q$ be hyperbolic
	fixed points, and let $z_n=G^n(z_0)$ with
	$z_0\in W^u(p)\cap W^s(q)$.  Set $A_n=DG(z_n)$.
	The intersection at $z_0$ is transverse if and only if the only solution of
	\begin{equation}
		\label{eq:adjoint-criterion}
		\eta_n=A_n^T\eta_{n+1},\qquad n\in\mathbb Z,
		\qquad
		\eta_n\longrightarrow0\quad\text{as }n\to\pm\infty,
	\end{equation}
	is $\eta_n=0$ for all $n$.
<!-- end lemma-3 -->
<!-- proof-9: body_md -->
Write $E^s_n=T_{z_n}W^s(q)$ and $E^u_n=T_{z_n}W^u(p)$.
	The adjoint equation transports covectors by
	$\eta_{n+1}=A_n^{-T}\eta_n$ and preserves their pairing with solutions of
	$v_{n+1}=A_nv_n$.
	
	We claim that $\eta_n\to0$ as $n\to+\infty$ precisely when $\eta_0$
	vanishes on $E^s_0$.  Choose $C^1$ local coordinates $(s,u)$ near $q$ in
	which its local stable manifold is $\{u=0\}$, using the stable-manifold
	theorem \cite{HirschPughShub1977}.  For all sufficiently large $n$, the
	derivative in these coordinates has the block form
	\[
	\widehat A_n=\begin{pmatrix}\Sigma_n&\Xi_n\\0&\Upsilon_n\end{pmatrix},
	\]
	where $\Sigma_n$ tends to the stable block and $\Upsilon_n$ to the unstable
	block of $DG(q)$.  If $\eta_0$ vanishes on $E^s_0$, invariance of the stable
	tangent spaces makes the transported covector have the form $(0,\beta_n)^T$,
	with $\beta_{n+1}=\Upsilon_n^{-T}\beta_n$.
	The limiting inverse transpose has all eigenvalues inside the unit circle.
	Choose a norm in which its operator norm is less than one.  A uniform
	bound less than one then holds for all sufficiently large $n$.
	The coordinate changes and their inverses have bounded derivatives along
	the convergent orbit, so $\eta_n$ decays exponentially in the original
	coordinates too.
	Conversely, stable tangent vectors evolve through $\Sigma_n$; the same norm
	argument gives their exponential decay.  If $\eta_n\to0$, the constant
	pairing $\eta_n^Tv_n=\eta_0^Tv_0$ must therefore vanish, so $\eta_0$
	vanishes on $E^s_0$.
	
	Applying the same argument to $G^{-1}$ at $p$ identifies backward decay
	with vanishing on $E^u_0$.  Solutions of \eqref{eq:adjoint-criterion}
	therefore correspond exactly to covectors vanishing on $E^s_0 + E^u_0$.
	A nonzero such covector exists precisely when the intersection is not
	transverse.

By the observation preceding the theorem, $K$ is positive definite.  Put
	$\lambda=\lambda_{\max}(K)$ and
	$D(z)=\operatorname{diag}(\cos z_1,\ldots,\cos z_N)$.
	The derivative $I_N+hKD(z)$ is similar to the symmetric matrix
	$I_N+hK^{1/2}D(z)K^{1/2}$, whose eigenvalues lie in
	$[1-h\lambda,1+h\lambda]\subset(0,2)$.  Thus $G_{K,h}$ is a local
	diffeomorphism.  It is homotopic to the identity through
	$z\mapsto z+thK\sin z$, $0\le t\le1$, and hence has degree one.
	Compactness and positivity of the Jacobian determinant make it a
	one-sheeted covering, so it is a global diffeomorphism. 
	
	The interval \eqref{eq:matrix-ms-range} cannot be enlarged.  Consider the
	diagonal points $z(t)=t\mathbf 1_N$, $t\in[0,\pi]$, where
	$D(z(t))=\cos t\,I_N$ and hence $DG_{K,h}(z(t))=I_N+h\cos t\,K$.  If
	$Kv=\lambda v$ with $v\neq0$, then $DG_{K,h}(z(t))\,v=(1+h\lambda\cos t)\,v$
	for every $t$.  If $h\lambda\ge1$, the number
	$t_*=\arccos\bigl(-1/(h\lambda)\bigr)$ lies in $(\pi/2,\pi]$, and
	$DG_{K,h}(z(t_*))\,v=0$, so $G_{K,h}$ is not a local diffeomorphism.
	
	Since $\|D^2V_0\|_2\le1$ and $\|Kv\|^2\le\lambda\,v^TKv$ for
	$v\in\mathbb R^N$, Taylor's inequality on a real lift gives
	\begin{equation}
		\label{eq:matrix-ms-descent}
		V_0(G_{K,h}(z))-V_0(z)
		\le-h\left(1-\frac{h\lambda}{2}\right)
		(\sin z)^T K\sin z.
	\end{equation}
	Therefore $V_0$ decreases strictly unless $\sin z=0$, and the fixed points
	are exactly $\{0,\pi\}^N$.
	At a fixed point $p$, the matrix $KD(p)$ is similar to
	$K^{1/2}D(p)K^{1/2}$.  By Sylvester's law of inertia, the number of
	positive eigenvalues equals the number of zero coordinates of $p$.
	All eigenvalues $\mu$ are nonzero and satisfy $|\mu|\le\lambda$, so the
	multipliers $1+h\mu$ lie off the unit circle: positive $\mu$ give
	unstable directions and negative $\mu$ stable ones.  The fixed points
	are therefore hyperbolic, with unstable dimension equal to their number
	of zero coordinates.
	
	For a non-fixed point, continuity gives a neighbourhood $U$ such that
	\[
	\sup_{y\in U}V_0(G_{K,h}(y))<\inf_{y\in U}V_0(y).
	\]
	Monotonicity then gives $G_{K,h}^k(U)\cap U=\varnothing$ for every $k\ge1$.
	Hence the nonwandering set is the stated finite fixed set.
	
	It remains to prove transversality.  The idea is to follow the largest
	coordinate of a decaying adjoint solution: diagonal dominance forces
	that coordinate into an invariant slab, where the adjoint equation
	cannot decrease it, which contradicts decay.
	
	Suppose that an intersection is not transverse.  Along its orbit
	$(z_n)_{n\in\mathbb Z}$, Lemma~\ref{lem:adjoint-transversality} supplies
	a nonzero solution tending to zero at both ends, with
	\[
		\eta_n=(I_N+hD_nK)\eta_{n+1},
		\qquad D_n=D(z_n).
	\]
	The maximum
	\[
		\rho=\max_{1\le i\le N,\ n\in\mathbb Z}|\eta_{i,n}|>0
	\]
	is attained.  Fix a time $j$ at which it is attained, and let
	$\mathcal I=\{i:|\eta_{i,j}|=\rho\}$.
	For $i\in\mathcal I$, put $s_i=\eta_{i,j}/\rho\in\{-1,1\}$.
	Diagonal dominance gives
	\[
	s_i(K\eta_j)_i
	\ge\rho\left(k_{ii}-\sum_{\ell\ne i}|k_{i\ell}|\right)\ge0.
	\]
	For at least one $i\in\mathcal I$ the left side is strictly positive.
	Otherwise, equality forces $k_{i\ell}=0$ for every
	$i\in\mathcal I$ and $\ell\notin\mathcal I$: a nonzero such entry
	would multiply a coordinate of absolute value strictly less than $\rho$.
	It would follow that
	$K_{\mathcal I\mathcal I}(\eta_j)_{\mathcal I}=0$,
	contradicting positive definiteness of this principal submatrix.
	
	Choose such an index $i$ and put $s=s_i$.
	Write $c_{i,n}=\cos z_{i,n}$.  The adjoint equation at time $j-1$ gives
	\[
	s\eta_{i,j-1}
	=\rho+hc_{i,j-1}s(K\eta_j)_i\le\rho,
	\]
	so $c_{i,j-1}\le0$.  Lemma~\ref{lem:path-slabs} implies
	$c_{i,n}\le0$ for every $n\ge j-1$.
	Set $c=c_{i,j}\le0$.  The diagonal coefficient $1+hk_{ii}c$ is strictly
	positive, and the adjoint equation at time $j$ gives
	\begin{align*}
		\rho
		&=(1+hk_{ii}c)s\eta_{i,j+1}
		+hc\sum_{\ell\ne i}k_{i\ell}s\eta_{\ell,j+1}\\
		&\le\rho\left[1+hc
		\left(k_{ii}-\sum_{\ell\ne i}|k_{i\ell}|\right)\right]
		\le\rho.
	\end{align*}
	Equality and the strictly positive diagonal coefficient force
	$\eta_{i,j+1}=s\rho$.  This step uses only $|\eta_{i,n}|=\rho$ and
	$c_{i,n}\le0$, so repeating it forces $\eta_{i,n}=s\rho$ for all
	$n\ge j$, contrary to decay.  This proves transversality.
	
	Finally, \eqref{eq:matrix-ms-descent}, compactness and finiteness of the
	critical set give convergence of every orbit to one fixed point by the
	argument of Proposition~\ref{thm:finite-convergence}.
	The unique minimum is $(\pi,\ldots,\pi)$.  All other fixed points have
	stable manifolds of dimension less than $N$, so their basins have measure
	zero, as in Theorem~\ref{thm:global-dynamics}.

Orient every edge from one bipartition class to the other and set
	$K=B^TB$ and $d=(d_e)_{e\in\mathcal E}$.  The diagonal entries of $K$ are
	two, and its off-diagonal entries are one when the edges share a vertex
	and zero otherwise.  For an edge $uv$, \eqref{eq:tree-sign-condition}
	gives
	\[
	(Kd)_{uv}=\sum_{e\ni u}d_e+\sum_{e\ni v}d_e=2.
	\]
	Put $\Delta=\operatorname{diag}(d_e)$ and let $p_e=0$ if $d_e=1$, and
	$p_e=\pi$ if $d_e=-1$.  Writing $m$ for the number of edges and
	$\mathbf 1_m=(1,\ldots,1)^T$, the circle $z(s)=p+s\mathbf 1_m$ is invariant,
	with induced map $s\mapsto s+2h\sin s$.
	The bound on $h$ implies $h<1/2$, since $\lambda_{\max}(K)\ge2$.
	For $0<s<\pi$, the scalar map is strictly increasing and maps this
	interval into itself.  Since $s+2h\sin s>s$ there, every orbit has
	forward limit $\pi$ and backward limit zero.
	Thus the arc joins $p$ to its binary complement
	$q=p+\pi\mathbf 1_m\pmod{2\pi}$.
	
	Along the arc, the derivative is $I_m+h\cos s\,K\Delta$.
	The matrix $K\Delta$ is similar to the symmetric matrix
	$K^{1/2}\Delta K^{1/2}$.  Sylvester's law of inertia shows that
	$K\Delta$ has $m-r$ positive and $r$ negative eigenvalues.  Every
	eigenvalue $\mu$ satisfies $|\mu|\le\lambda_{\max}(K)$.  All variational
	factors $1+h\mu\cos s$ are therefore positive, and the same real
	eigenspaces are preserved along the orbit.  Let $E_+$ be the sum of the
	positive eigenspaces of $K\Delta$.
	
	Tangent vectors to $W^u(p)$ decay under backward iteration of the
	derivative.  A nonzero component in a negative eigenspace instead grows
	backwards, because its forward multiplier tends to $1+h\mu\in(0,1)$
	near $p$.  Hence $T_zW^u(p)\subseteq E_+$, and equality follows from the
	dimensions.  Similarly, negative eigendirections grow forwards near $q$,
	where their multipliers tend to $1-h\mu>1$.  Thus
	\begin{equation}
		\label{eq:tree-obstruction-tangents}
		T_zW^u(p)=T_zW^s(q)=E_+.
	\end{equation}
	A negative sign makes $E_+$ a proper subspace, so the intersection is
	not transverse.
	
	Finally, summing \eqref{eq:tree-sign-condition} over vertices gives
	$2(m-2r)=m+1$, since a tree has $m+1$ vertices.  Therefore $m=4r+1$,
	and Sylvester's law gives $\dim E_+=m-r=3r+1$.

\label{prop:tree-nontransverse}
Consider the tree with two adjacent vertices of degree three, each joined
to two leaves, with unit weights and sinusoidal interaction.
For
\begin{equation}
\label{eq:tree-counterexample-range}
    0<h<\frac{2}{5+\sqrt{17}},
\end{equation}
its reduced map on $\mathbb T^5$ is a diffeomorphism with a non-transverse
heteroclinic intersection.

Label the vertices $1,\ldots,6$ and orient and order the edges as

    (1,2),\quad(1,3),\quad(1,4),\quad(5,2),\quad(6,2),

where the first vertex is the tail and the second is the head.
In edge coordinates,
\begin{equation}
\label{eq:tree-counterexample-map}
    G_{K_*,h}(z)=z+hK_*\sin z\pmod{2\pi},\qquad
    K_*=B^TB=
    \begin{pmatrix}
      2&1&1&1&1\\
      1&2&1&0&0\\
      1&1&2&0&0\\
      1&0&0&2&1\\
      1&0&0&1&2
    \end{pmatrix}.
\end{equation}
The signs $d=(-1,1,1,1,1)^T$ satisfy
\eqref{eq:tree-sign-condition}.  The eigenvalues of $K_*$ are
$1,1,3,(5-\sqrt{17})/2$ and $(5+\sqrt{17})/2$, so
Proposition~\ref{prop:tree-sign-obstruction} applies in the stated range.
Explicitly, with
\[
    p=(\pi,0,0,0,0)^T,\qquad q=(0,\pi,\pi,\pi,\pi)^T,
\]
the arc $z(s)=p+s\mathbf 1_5$, $0<s<\pi$, has scalar dynamics
\begin{equation}
\label{eq:tree-counterexample-circle}
    s(n+1)=s(n)+2h\sin s(n).
\end{equation}

    E_+=\{\xi\in\mathbb R^5:
               4\xi_1=\xi_2+\xi_3+\xi_4+\xi_5\}.
\]
It equals both tangent spaces along this heteroclinic arc by

\label{thm:all-to-all}
	The intrinsic potential is
	\begin{equation}
		\mathcal V_{\mathrm{all}}(\theta)
		=\sum_{1\le i<j\le N+1}\cos(\theta_i-\theta_j)
		=\frac12|Z(\theta)|^2-\frac{N+1}{2}.
		\label{eq:all-potential-order}
	\end{equation}
	Its absolute minima are precisely the \emph{balanced} configurations,
	those with $Z(\theta)=0$, and their potential value is $-(N+1)/2$.
