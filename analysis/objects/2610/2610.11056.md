\label{thm:kelvin}
	Let $u$	be a normal solution of \eqref{eq:euclidean}. Then there is a unique $R>0$ such that
		\begin{equation}\label{u kelvin}
	u(x)=u\!\left(\frac{R^2x}{|x|^2}\right)				-2\alpha\log\frac{|x|}{R},			\qquad x\neq 0.
		\end{equation}
      Here $R$ is the unique radius such that
		\begin{equation}\label{R half mass}
			\int_{B_R}|x|^{n\gamma}e^{nu(x)}\,\mathrm d x=\frac{1}{2}\alpha|\mathbb S^n|.
		\end{equation}

		\begin{equation}\label{V sym}

		\end{equation}

<!-- proof-1: body_md -->
Write $f(x)=|x|^{n\gamma}e^{nu(x)}$. Then $\int_{\mathbb R^n}f(x)\,\mathrm d x=\alpha|\mathbb S^n|$. The normal logarithmic behavior of $u$ implies that, for every sufficiently small $\varepsilon>0$,
	\begin{equation*}
		f(x)\leq C_\varepsilon |x|^{-n(1+\alpha)+\varepsilon},
		\qquad |x|\geq 1.
	\end{equation*}
	We first assume $\gamma>0$. Then since $n\alpha>2$, we can choose $\beta>0$ so that
	\begin{equation}\label{f decay}
		f(x)\leq C(1+|x|)^{-n-2-\beta}.
	\end{equation}
	
	We may therefore absorb the numerator in \eqref{rig:eq:normal-representation} into its constant and write
	\begin{equation}\label{u rep f}
		u(x)=c-\frac{2}{|\mathbb S^n|}\int_{\mathbb R^n}\log|x-y|\,f(y)\,dy
	\end{equation}
	Differentiation is legitimate by \eqref{f decay}. Hence
	\begin{equation}\label{grad u rep f}
		\nabla u(x)=-\frac{2}{|\mathbb S^n|}\int_{\mathbb R^n}\frac{x-y}{|x-y|^2}\,f(y)\,dy.
	\end{equation}
	
	To obtain the behavior at infinity, we rewrite \eqref{u rep f} as
    \begin{equation}\label{uneq 0rep f}
		u(x)=c-\frac{2}{|\mathbb S^n|}\int_{\mathbb R^n}\log\frac{|x-y|}{|x|}\,f(y)\,dy-2\alpha\log|x|, \qquad x\neq 0.
	\end{equation}
    For $r=|x|\geq 2$, we split $\mathbb R^n$ into
	\begin{equation*}
		\Omega_1=\{y: |y|<r/2\},\quad
		\Omega_2=\{y: |y|\geq r/2,\ |x-y|<r/2\},\quad
		\Omega_3=\{y: |y|\geq r/2,\ |x-y|\geq r/2\}.
	\end{equation*}
	On $\Omega_1$,
	\begin{equation*}
		\left|\log\frac{|x-y|}{r}\right|\leq C\frac{|y|}{r},
	\end{equation*}
    so
    \begin{equation*}

			\int_{\Omega_2}
			\left|\log\frac{|x-y|}{r}\right|f(y)\,dy\leq Cr^{-2-\beta}

	\end{equation*}

		\leq C+\log\left(1+\frac{|y|}{r}\right),

	\end{equation*}

\end{equation*}

    Finally, when $-1<\gamma\leq 0$, we take $\delta<\min\{1,n\alpha\}$. Then for $0<\varepsilon<n\alpha-\delta$, we have

            C|x|^{n\alpha-n},\qquad |x|<1,\\
            C_\varepsilon |x|^{-n-n\alpha+\varepsilon}, \qquad |x|\ge 1.
        \end{cases}
    \end{equation*}

    \begin{equation*}
        |u(x)-u(0)|+r|\nabla u(x)|\le Cr^{\delta}.
    \end{equation*}
    At infinity, following the previous argument, we use $\left|\log\frac{|x-y|}{r}\right|\leq C\frac{|y|}{r}\le C(\frac{|y|}{r})^\delta$ to obtain
    \begin{equation*}
        \int_{\Omega_1}\left|\log\frac{|x-y|}{r}\right|f(y)\,dy\le \frac{C}{r^\delta}.
    \end{equation*}

    \end{equation*}
    This gives the weaker asymptotics for $-1<\gamma\le 0$.

We first start the moving plane far to the left. We then show that it must stop at a finite position and prove that the stopping plane is a plane of symmetry by ruling out further continuation.
    \noindent\textbf{Step 1: Start of the moving plane.} 	For a real number $\lambda$, reflect in $t=\lambda$ and set
	\begin{equation*}
	t^\lambda=2\lambda-t,\qquad w_\lambda(t,\omega)=V(t^\lambda,\omega)-V(t,\omega) \quad(t<\lambda).
	\end{equation*}
	Then taking difference of the two representations in \eqref{int rep V} yields
	\begin{equation}\label{int rep w la}
		\begin{aligned}
			w_\lambda(t,\omega)
			&=\alpha\int_{s<\lambda}\int_{\mathbb S^{n-1}}K_\lambda(t,s,\omega\cdot\eta)\bigl(p(s^\lambda,\eta)-p(s,\eta)\bigr)\,d\eta\,ds\\
            &=\alpha\int_{s<\lambda}\int_{\mathbb S^{n-1}}K_\lambda(t,s,\omega\cdot\eta)p(s,\eta)\bigl(e^{nw_\lambda(s,\eta)}-1\bigr)\,d\eta\,ds,
		\end{aligned}
	\end{equation}
	where
	\begin{equation*}
		K_\lambda(t,s,a)
		=\log\frac{\cosh(2\lambda-t-s)-a}{\cosh(t-s)-a}.
	\end{equation*}
	
	For $t,s<\lambda$, $K_\lambda>0$ since $2\lambda-t-s>|t-s|$. To obtain an upper bound, we set $A=2\lambda-t-s$ and $D=|t-s|$. Then	for $a\in[-1,1]$ and $A, D>0$, we have elementary inequalities
	\begin{equation*}
		\begin{aligned}
			\cosh A-a\leq 2e^A,\qquad
			\cosh D-a\geq\cosh D-1\geq\frac{e^D}{2}(1-e^{-1})^2\min\{D^2,1\}.
		\end{aligned}
	\end{equation*}
  Therefore,
	\begin{equation}\label{K upper}
			0<K_\lambda(t,s,a)
			\leq 2(\lambda-s)+C_*+2\log^+\frac{1}{|t-s|},
			\qquad
			C_*=\log\frac{4}{(1-e^{-1})^2}.
	\end{equation}
	Here $\log^+r=\max\{\log r,0\}$. 
	
	We will use the following elementary bounds:
	\begin{equation}\label{log int}
		\int_\mathbb R\log^+\frac{1}{|t-s|}\,\mathrm d s=2,
	\end{equation}
	and, for every interval $I$ of length $0<\ell\leq1$,
	\begin{equation}\label{log int upper}
	\sup_{t\in\mathbb R}\int_I\log^+\frac{1}{|t-s|}\,\mathrm d s\leq\ell\left(1+\log\frac{2}{\ell}\right).
	\end{equation}

	
	To start the moving plane, we define

		N_\lambda=\{(s,\eta):s<\lambda,\ w_\lambda(s,\eta)<0\},
		\qquad w_\lambda^-=\max\{-w_\lambda,0\}.
	\end{equation*}
On $N_\lambda$, we have

1-e^{nw_\lambda(s,\eta)}=1-e^{-nw_\lambda^-(s,\eta)}\leq nw_\lambda^-(s,\eta).
	\end{equation*}
At a point where $w_\lambda(t,\omega)<0$, discarding the integral in \eqref{int rep w la} outside $N_\lambda$ yields

			w_\lambda^-(t,\omega)\leq n\alpha\int_{N_\lambda}K_\lambda(t,s,\omega\cdot\eta)p(s,\eta)
			w_\lambda^-(s,\eta)\,ds\,d\eta.
	\end{equation}
	
	For each fixed $\lambda$, the expansions \eqref{V infty} give $\lim_{t\to-\infty}w_\lambda(t,\omega)=b_+-b_--2\alpha\lambda$ uniformly in $\omega$. Together with continuity and $w_\lambda(\lambda,\omega)=0$ due to its definition, this proves that
	\begin{equation}\label{M la finite}
		M_\lambda:=\sup_{t<\lambda,\,\omega\in\mathbb S^{n-1}}w_\lambda^-(t,\omega)<\infty.
	\end{equation}

	For sufficiently negative $\lambda$, the bound $p(s,\eta)\leq Ce^{n\alpha s}$ holds for all $s<\lambda$. Using \eqref{log int}, the upper bound for $K_\lambda$ \eqref{K upper} gives
\begin{equation*}

	\sup_{t<\lambda,\,\omega}\int_{s<\lambda}\int_{\mathbb S^{n-1}}K_\lambda(t,s,\omega\cdot\eta)p(s,\eta)\,d\eta\,ds\leq C|\mathbb S^{n-1}|e^{n\alpha\lambda}\left(\frac{2}{(n\alpha)^2}+\frac{C_*}{n\alpha}+4\right)\longrightarrow0, \quad \lambda\to-\infty.
		\end{aligned}
\end{equation*}
	Consequently, for $\lambda_0$ negative enough, taking the supremum in \eqref{w- le} gives $M_\lambda\leq\frac{1}{2}M_\lambda$, so $w_\lambda\geq0$ for every $\lambda\leq\lambda_0$.	This starts the moving plane.

    Define the stopping value by
	\begin{equation}\label{def t0}
		t_0=\sup\Bigl\{\lambda\in\mathbb R:w_\mu(t,\omega)\geq0

	\end{equation}
 By \eqref{V infty},	$\lim_{t\to-\infty}w_\lambda(t,\omega)=b_+-b_--2\alpha\lambda<0$	for sufficiently large $\lambda$, so $t_0$ is finite. By continuity,	$w_{t_0}(t,\omega)\geq0$ for all $t<t_0$. Our goal is to prove that $w_{t_0}\equiv 0$ and then the symmetry follows.

	Suppose, for contradiction, that $w_{t_0}$ is not identically zero. It is then positive on a nonempty open subset. On that subset $	p(s,\eta)\bigl(e^{nw_{t_0}(s,\eta)}-1\bigr)>0$. Since $K_{t_0}>0$ by \eqref{K upper}, the integral representation \eqref{int rep w la} then yields

We now show that this contradicts the definition of $t_0$.

    \noindent\textbf{Step 3: Excluding further continuation.}

\end{equation*}

Strict positivity of $w_{t_0}$ and continuity preserve positivity on the middle region. On the other two regions, choosing $T$ large and $\delta$ small makes the kernel integrals sufficiently small to exclude a negative part by \eqref{w- le}.
    	
We first provide a uniform estimate for the integral over $(-\infty,-T)\times\mathbb S^{n-1}$. To this end, we choose $T>|t_0|+2$ sufficiently large that $p(s,\eta)\leq Ce^{n\alpha s}$ for all $s<-T$. For all $\lambda\in[t_0,t_0+1]$, we have

	\leq C(1+T)e^{-n\alpha T}.
	\end{equation*}
	For the logarithmic part, we have

		\int_{s<-T}\int_{\mathbb S^{n-1}}\log^+\frac{1}{|t-s|}\,p(s,\eta)\,d\eta\,ds

	\end{equation*}
uniformly in $t$. Consequently, (\ref{K upper}) implies

		\leq C(1+T)e^{-n\alpha T}.
	\end{equation}

We also need an estimate on $(t_0-\delta,\lambda)\times\mathbb S^{n-1}$. Let $0<\delta<1/4$ and $\lambda\in[t_0,t_0+\delta]$. Set

			&\quad\leq C\delta(1+|\log\delta|).

	Now we fix $T$ large enough and $\delta$ small enough such that the integrals in \eqref{int far left} and \eqref{int strip} are less than $\frac{1}{4n\alpha}$. Then by the compactness of $[-T,t_0-\delta]\times\mathbb S^{n-1}$ and \eqref{w>0 strict}, we get

	|w_\lambda(t,\omega)-w_{t_0}(t,\omega)|=|V(2\lambda-t,\omega)-V(2t_0-t,\omega)|\leq2\|\partial_t V\|_\infty|\lambda-t_0|.

	\end{equation*}
	Therefore its negative set satisfies
	\begin{equation*}
		N_\lambda\subset\bigl(( -\infty,-T)\cup(t_0-\delta,\lambda)\bigr)\times\mathbb S^{n-1}.
	\end{equation*}
	
	
	By our choices of $T$, $\delta$, and $\varepsilon$, we obtain

	\end{equation*}

	\leq\frac{1}{2}M_\lambda.
	\end{equation*}
	Taking the supremum yields $M_\lambda\leq\frac{1}{2}M_\lambda$, so $M_\lambda=0.$ Consequently, $w_\lambda\geq0$ for every $\lambda\in [t_0,t_0+\varepsilon]$, which contradicts the definition of $t_0$ \eqref{def t0}. 
    
    Therefore, we proved that $w_{t_0}\equiv0$, which is exactly \eqref{V sym}. As a result, \eqref{u kelvin} holds for $R=e^{t_0}$. Finally, \eqref{R half mass} follows from a simple change of variable.

	Set $K(x,y)=f_n(x-y)$ and fix $x_1<x_2$ and $y_1<y_2$.

The strict inequality follows because both determinants are

The integrand is nonnegative everywhere and strictly positive

\label{prop:coercivity}
	Let $\rho$ be a probability measure translated so that $0$ is its median, that is,
\begin{equation}\label{0 median}
\rho\bigl((-\infty,0]\bigr)\ge\frac{1}{2}\text{ and } \rho\bigl([0,\infty)\bigr)\ge\frac{1}{2}.
\end{equation}
	 Then
	\begin{equation*}
	\mathcal F_{n,\alpha}(\rho)
	\ge c\int_\mathbb R|t|\rho(t)\,\mathrm d t-C
	\end{equation*}
	for constants $c,C>0$ depending only on $n$ and $\alpha$.
<!-- end proposition-1 -->
<!-- proof-7: body_md -->
By \eqref{0 median}, we obtain
	\begin{equation*}
		\iint_{\mathbb R^2}|t-s|\rho(t)\rho(s)\,\mathrm d t \,\mathrm d s
		\ge\frac{1}{2}\int_\mathbb R|t|\rho(t) \,\mathrm d t.
	\end{equation*}
From \eqref{eq:K-linear-revised}, we have $\mathcal K_n(t)\ge \frac{|t|}{2\beta_n} -C_0$. Therefore,
\begin{equation}\label{int ge}
	\iint_{\mathbb R^2}\mathcal K_n(t-s)\rho(t)\rho(s)\,\mathrm d t\,\mathrm d s \ge \frac{1}{4\beta_n}\int_\mathbb R|t|\rho(t) \,\mathrm d t-C_0.
\end{equation}
	

    
	For $\sigma>0$, we define an auxiliary function
    \begin{equation*}
		q_\sigma(t)=\frac{\sigma}{2}e^{-\sigma|t|},
	\end{equation*} 
    Since $x\mapsto x\log x$ is convex on $(0,\infty)$ and its derivative is $1+\log x$, we have
\begin{equation}\label{xlogx convex}
 x\log x\ge y\log y+(1+\log y)(x-y)
 =x\log y+x-y \text{ for }x\ge0\text{ and }y>0,   
\end{equation}
where the case $x=0$ follows by continuity. Taking $x=\rho(t)$ and $y=q_\sigma(t)>0$ gives
\[
 \rho(t)\log\rho(t)
 \ge \rho(t)\log q_\sigma(t)+\rho(t)-q_\sigma(t).
\]
	Integrating, we obtain
	\begin{equation}\label{ent ge}
			\int_{\mathbb R}\rho\log\rho\,\mathrm d t
		\ge\log\frac\sigma2-\sigma\int_{\mathbb R}|t|\rho(t)\,\mathrm d t.
	\end{equation}
	Combining \eqref{int ge} and \eqref{ent ge}, the assertion follows by choosing $0<\sigma<\frac{C_{n,\alpha}}{8\beta_n}$.

<!-- proposition-2: body_md -->
\label{prop:radial-existence}
For every $n\geq 3$ and $\alpha>0$, the infimum of $\mathcal F_{n,\alpha}$ on $\mathcal{P}_1(\mathbb R)$ is finite and attained.  Every minimizer is positive, satisfies \eqref{eq:density-EL} and gives a radial normal solution of \eqref{eq:euclidean}.
<!-- end proposition-2 -->
<!-- proof-8: body_md -->
Choose a minimizing sequence $\rho_j$ and translate each density so that $0$ is a median, as in \eqref{0 median}. Proposition~\ref{prop:coercivity} gives
\[
 \sup_j\int_\mathbb R |t|\rho_j(t)\,\mathrm d t\le M_0<\infty.
\]
By Prokhorov's theorem \cite[Theorem~3.10.3]{Durrett2019}, after passing to a subsequence, the probability measures $\rho_j(t)\,\mathrm d t$ converge weakly to a probability measure $\mu$, with
$\int_\mathbb R|t|\,\mathrm d\mu(t)\le M_0$.

Since $\mathcal{K}_n\geq 0$, the minimizing sequence satisfies $\operatorname{Ent}(\rho_j)\le \mathcal{F}_{n,\alpha}\le C$. By the lower semicontinuity of the entropy on measures with uniformly bounded first moments (see \cite[Proposition 2.1 (1)--(2)]{Santambrogio2016}), we have
\begin{equation*}
    \operatorname{Ent}(\mu)\le \liminf_{j\to\infty}\operatorname{Ent}(\rho_j)\le C.
\end{equation*}
Therefore $\mu=\rho\,\mathrm d t$ for some density $\rho$ in $\mathcal{P}_1(\mathbb R)$.

Since $\mathcal K_n$ is continuous and nonnegative, weak convergence of the product measures also yields
\begin{equation*}
    \iint_{\mathbb R^2}\mathcal K_n(t-s)\rho(t)\rho(s)\,\mathrm d t\,\mathrm d s
 \le\liminf_{j\to\infty}\iint_{\mathbb R^2}\mathcal K_n(t-s)\rho_j(t)\rho_j(s)\,\mathrm d t\,\mathrm d s.
\end{equation*}
Combining these inequalities shows that $\rho$ is a minimizer with finite entropy and finite first moment. 

Moreover, since
\begin{equation*}
 (\mathcal K_n*\rho)(t)=\frac{|t|}{2\beta_n}+O(1),    
\end{equation*}
the probability density
\begin{equation*}
 \bar\rho(t):=\frac{\exp(-C_{n,\alpha}(\mathcal K_n*\rho)(t))}{\int_\mathbb R\exp(-C_{n,\alpha}(\mathcal K_n*\rho)(s))\,\mathrm d s}    
\end{equation*}
is well defined and satisfies $c e^{-n\alpha|t|}\le\bar\rho(t)\le C e^{-n\alpha|t|}$ for $t\in\mathbb R$ and some $c, C>0$.  For $\rho_\varepsilon=(1-\varepsilon)\rho+\varepsilon\bar\rho$ and $\delta=\bar\rho-\rho$, convexity of the entropy and expansion of the interaction term give
\begin{equation*}
    \begin{aligned}
      0\le\mathcal F_{n,\alpha}(\rho_\varepsilon)
 -\mathcal F_{n,\alpha}(\rho)
 \le&\varepsilon \left(\operatorname{Ent}(\bar\rho)-\operatorname{Ent}(\rho)+C_{n,\alpha}\int_{\mathbb R}(\mathcal K_n*\rho)\delta\,\mathrm d t\right)\\
 &+\frac{C_{n,\alpha}\varepsilon^2}{2} \iint_{\mathbb R^2}\mathcal K_n(t-s)\delta(t)\delta(s)\,\mathrm d t\,\mathrm d s\\
 =&-\varepsilon\int_\mathbb R\rho\log\frac{\rho}{\bar\rho}\,\mathrm d t+O(\varepsilon^2).   
    \end{aligned}
\end{equation*}
The remainder is finite because both densities have finite first moment and $\mathcal K_n$ has at most linear growth. Dividing by $\varepsilon$ and letting $\varepsilon\downarrow0$ and applying inequality \eqref{xlogx convex}, we obtain that  $\rho=\bar\rho$ almost everywhere. Thus $\rho$ has a positive continuous representative and satisfies \eqref{eq:density-EL}.

With $M_\alpha$ as in \eqref{def M rho C}, set $v_\alpha=n^{-1}\log(M_\alpha\rho)$. Then \eqref{eq:K-fundamental-revised} gives \eqref{eq:radial-equation}. The symbol $a_{n,0}$ is elliptic of order $n$, so local elliptic regularity \cite[Theorem~18.1.29]{HormanderIII} first gives $v_\alpha\in H^n_{\mathrm{loc}}(\mathbb R)$, since $e^{nv_\alpha}\in L^2_{\mathrm{loc}}(\mathbb R)$.
Iterating this regularity gain together with Sobolev composition gives $v_\alpha\in H^{kn}_{\mathrm{loc}}(\mathbb R)$ for every integer $k\ge1$ and hence $v_\alpha\in C^\infty(\mathbb R)$. Using \eqref{eq:K-linear-revised} and the finite first moment of $\rho$, we obtain the following asymptotics by the dominated convergence theorem:
\begin{equation}\label{eq:v toinfty}
 v_\alpha(t)=
 \begin{cases}
  \alpha t+c_-+o(1),&t\to-\infty,\\
  -\alpha t+c_++o(1),&t\to+\infty.
 \end{cases}
\end{equation}
Differentiating \eqref{eq:density-EL} also gives
\eqref{eq:v al infty}.
Consequently,
\[
 u_\alpha(x)=v_\alpha(\log|x|)-\alpha\log|x|
\]
is continuous on $\mathbb R^n$ with $u_\alpha(0)=c_-$, has at most logarithmic growth and
\[
 \int_{\mathbb R^n}|x|^{n\gamma}e^{nu_\alpha}\,\mathrm d x=|\mathbb S^{n-1}|M_\alpha<\infty.
\]
The Emden--Fowler transformation gives the equation away from the origin and Lemma~\ref{lem:removability-normality} shows that $u_\alpha$ is a radial normal solution on $\mathbb R^n$.

\label{prop:radial-nondegeneracy}
Let $\phi$ be a bounded radial solution of $A_{n,0}\phi=W_\alpha\phi$. Then $\phi\in\operatorname{span}\{ v_\alpha'\}$.

We first claim that $\int_\mathbb R W_\alpha(t)\phi(t)\,\mathrm d t=0$. In fact, the function $h:=\phi+\mathcal{K}_n*(W_\alpha\phi)$ satisfies $A_{n,0}h=0$ and it is a tempered distribution because of the exponential decay of $W_\alpha$ and the boundedness of $\phi$. By Lemma \ref{lem:tempered-radial-kernel}, $h=at+b$ for some $a, b\in\mathbb R$.

On the other hand, by \eqref{eq:K-linear-revised}, the dominated convergence theorem gives
\begin{equation*}
\frac{(\mathcal{K}_n* (W_\alpha\phi))(t)}{t}\longrightarrow
\begin{cases}
	\frac{1}{2\beta_n} \int_\mathbb R W_\alpha(s)\phi(s)\,\mathrm d s,&t\to+\infty,\\
	-\frac{1}{2\beta_n} \int_\mathbb R W_\alpha(s)\phi(s)\,\mathrm d s,&t\to-\infty.
\end{cases}
\end{equation*}
Since $\phi$ is bounded, we must have $\int_\mathbb R W_\alpha(t)\phi(t)\,\mathrm d t=0$.

Set $\sigma(t)=W_\alpha(t)\phi(t)$. Then $\int_{\mathbb R}\sigma(t)\,\mathrm d t=0$ and the equation for $\phi$ gives
\begin{equation}\label{eq:sigma}
	\frac{\sigma}{\rho_\alpha}
	+C_{n,\alpha}\mathcal{K}_n*\sigma=\text{constant}.
\end{equation}
Define $\zeta(t):=-\frac{1}{\rho_\alpha(t)} \int_{-\infty}^t\sigma(s)\,\mathrm d s$. Then $\sigma=-(\rho_\alpha\zeta)'$. Since $\sigma=C_{n,\alpha}\rho_\alpha\phi$ and $\int_\mathbb R\sigma=0$, for $t\ge0$ we have
\begin{equation*}
    |\zeta(t)|=\frac{\left|\int_t^\infty\sigma(s)\,\mathrm d s\right|}{\rho_\alpha(t)}
 \le C_{n,\alpha}\|\phi\|_\infty\frac{\int_t^\infty\rho_\alpha(s)\,\mathrm d s}{\rho_\alpha(t)}
 \le C.
\end{equation*}
Here the last inequality follows from $\rho_\alpha(t)\sim e^{-n\alpha|t|}$ due to  \eqref{def M rho C} and \eqref{eq:v toinfty}. For $t\le0$, the same estimate follows directly from the defining integral over $(-\infty,t]$. Thus $\zeta$ is bounded.
Differentiating its definition gives
\begin{equation*}
    \zeta'=-\frac{\sigma}{\rho_\alpha}-\frac{\rho_\alpha'}{\rho_\alpha}\zeta
 =-C_{n,\alpha}\phi-nv_\alpha'\zeta.
\end{equation*}
Since $\phi$, $v_\alpha'$ and $\zeta$ are bounded, so is $\zeta'$. 

We have the following key identity.
\begin{equation}\label{Q al id}
	\begin{aligned}
		\mathcal Q_\alpha[\zeta]
		:=&\int_\mathbb R\rho_\alpha(t)|\zeta'(t)|^2\,\mathrm d t+\frac{C_{n,\alpha}}2
		\iint_{\mathbb R^2}g_n(t-s)
		|\zeta(t)-\zeta(s)|^2\rho_\alpha(t)\rho_\alpha(s)\,\mathrm d t\,\mathrm d s\\
		=& \int_\mathbb R \rho_\alpha\zeta(\frac{\sigma}{\rho_\alpha}
		+C_{n,\alpha}\mathcal{K}_n*\sigma)'\,\mathrm d t.
	\end{aligned}
\end{equation}
Indeed, using integration by parts, we can write
\begin{equation}\label{Q al 1}
	\begin{aligned}
	\int_\mathbb R \rho_\alpha\zeta(\frac{\sigma}{\rho_\alpha})'\,\mathrm d t=&\int_\mathbb R \rho_\alpha\zeta\left(-\zeta'-(\log\rho_\alpha)'\zeta\right)'\,\mathrm d t\\
	=&\int_\mathbb R\rho_\alpha|\zeta'|^2\,\mathrm d t-\int_\mathbb R (\log\rho_\alpha)''\rho_\alpha\zeta^2\,\mathrm d t\\
	=&\int_\mathbb R\rho_\alpha|\zeta'|^2\,\mathrm d t+C_{n,\alpha}\iint g_n(t-s)	\rho_\alpha(t)\rho_\alpha(s)\zeta(t)^2\,\mathrm d t\,\mathrm d s,
	\end{aligned}
\end{equation}
where we used the equation $(\log\rho_\alpha)''=-C_{n,\alpha}g_n*\rho_\alpha$ in the last equality.


On the other hand, integration by parts gives
\begin{equation*}
	(\mathcal{K}_n'*\sigma)(t)
	=-\int_\mathbb R g_n(t-s)\rho_\alpha(s)\zeta(s)\,\mathrm d s,
\end{equation*}
so
\begin{equation}\label{Q al 2}
	C_{n,\alpha}\int_\mathbb R\rho_\alpha(t)\zeta(t)(\mathcal{K}_n'*\sigma)(t)\,\mathrm d t
	=-C_{n,\alpha}\iint_{\mathbb R^2} g_n(t-s)
	\rho_\alpha(t)\rho_\alpha(s)\zeta(t)\zeta(s)\,\mathrm d t\,\mathrm d s.
\end{equation}
\eqref{Q al id} then follows from \eqref{Q al 1} and \eqref{Q al 2}.


By \eqref{eq:sigma}, integration by parts once more yields that $\mathcal{Q}_\alpha[\zeta]=0$, so $\zeta$ is constant. Hence $\sigma=-c\rho_\alpha'$. Since $W_\alpha=n!M_\alpha\rho_\alpha$, we obtain that
\begin{equation*}
	\phi=\frac{\sigma}{W_\alpha}=-\frac{c}{n!M_\alpha}\frac{\rho_\alpha'}{\rho_\alpha}=-\frac{c}{(n-1)!M_\alpha}v_\alpha'.
\end{equation*}

\label{prop:mode-green}
For $n\ge3$, $k\ge1$ and $t\ge0$, we have
\begin{align}
 G_{n,k}(t)
 &=\frac{2^{1-n}e^{-kt}}{\Gamma(n/2)^2}
 \int_0^1s^{k-1}(1-s)^{n/2-1}
 (1-e^{-2t}s)^{n/2-1}\,\mathrm d s,
 \label{eq:G-beta}\\
 &=\frac{2^{1-n}\Gamma(k)}{\Gamma(n/2)\Gamma(k+n/2)}e^{-kt}
 \,{}_2F_1\left(1-\frac n2,k;k+\frac n2;e^{-2t}\right).
 \label{eq:G-hyper}
\end{align}
Moreover,
\begin{equation}\label{eq:G-diagonal}
 G_{n,k}(0)=
 \frac{2^{1-n}\Gamma(k)\Gamma(n-1)}
 {\Gamma(n/2)^2\Gamma(n+k-1)}.
\end{equation}
The function $G_{n,k}$ is even, strictly positive and strictly
decreasing on $(0,\infty)$. Moreover,
\begin{equation}\label{eq:G-positive-minors}
 \det\bigl[G_{n,k}(t_i-s_j)\bigr]_{i,j=1}^2>0
 \qquad
 \text{whenever }t_1<t_2\text{ and }s_1<s_2.
\end{equation}

<!-- definition-2: body_md -->
Let $\{e_j\}_{j=1}^\infty$ be an orthonormal basis of a Hilbert space with inner product $\langle \cdot ,\cdot \rangle$. An operator $T$ is \emph{Hilbert--Schmidt} if
\begin{equation*}
	\left\lVert T\right\rVert_{\mathfrak S_2}^2:=\sum_{j=1}^\infty \|Te_j\|^2<\infty.
\end{equation*}
$T$ is \emph{trace class} if
\begin{equation*}
	\left\lVert T\right\rVert_{\mathfrak S_1}:= \operatorname{tr}(\sqrt{T^*T})<\infty,
\end{equation*}
where the \emph{trace} is 
\begin{equation*}
	\operatorname{tr}(T)=\sum_{j=1}^\infty\langle Te_j, e_j\rangle.
\end{equation*}

\label{thm:mu2}
If $\mu_1(\alpha,k)>\mu_2(\alpha,k)\ge\cdots$ are the eigenvalues of $\mathscr K_{\alpha,k}$, then
\begin{equation*}
 \mu_2(\alpha,k)<1 \quad\hbox{for every }n\ge3,\ \alpha>0,\ k\ge1.

The kernel $g_n(|t-s|)-g_n(t+s)$ is strictly positive because $g_n$ is strictly decreasing.

 -\left({W_\alpha}R'\right)'(t)
 ={}&W_\alpha(t)\int_0^\infty

 &-2M(t)\int_0^\infty g_n(t+s)W_\alpha(s)\,\mathrm d s.

At $t=t_0$, the first term on the right is nonnegative, while the second is strictly positive because $M(t_0)<0$.  This is impossible.  Thus $R\ge0$.

If $R(t_0)=0$ for some $t_0\in (0,\infty)$, then $t_0$ would be a global minimum equal to zero. The second term of the right hand side of \eqref{eq:ratio-identity} would vanish, but the first would be strictly positive because $R$ is not identically zero.  This is again impossible. \le C_nR^{n-1}\|\widetilde f_j\|_{L^\infty(B_{2R})}.
 \end{equation*}

For every $\varepsilon>0$, we can choose $R$ such that
\begin{equation*}
     \int_{B_R}e^{n\eta}\,\mathrm d z>|\mathbb S^n|-\varepsilon.
\end{equation*}
For $f_j=|x|^{n\gamma_j}e^{nu_j}$, the local convergence of the rescaled densities \eqref{fj til} gives
\begin{equation*}
 \liminf_{j\to\infty}\int_{B_{Rr_j}(x_j)}f_j(x)\,\mathrm d x

Since $r_j\to0$ and the total mass of $f_j\,\mathrm d x$ is $\alpha_j|\mathbb S^n| \to |\mathbb S^n|$, the mass outside these shrinking balls tends to zero as $\varepsilon\downarrow0$. Consequently,

    f_j\,\mathrm d x\rightharpoonup|\mathbb S^n|\delta_{x_*}
\end{equation}
The median normalization \eqref{rig:eq:median} rules out both $x_*=0$ and $x_*=\infty$, because otherwise the mass of $B_1$ would then tend respectively to the full mass or to zero.  Hence $x_*\in\mathbb R^n\setminus\{0\}$.

\end{equation*}

By Lemma~\ref{rig:lem:no-concentration}, the functions $u_j$ are uniformly bounded above. Hence $f_j$ is uniformly bounded on every fixed ball. Using \eqref{grad u rep f} and splitting the integral into $|x-y|<1$ and its complement, we obtain, for every $R\ge1$,
\begin{equation*}
 \|\nabla u_j\|_{L^\infty(B_R)}
 \le \frac{2}{|\mathbb S^n|}\left(\|f_j\|_{L^\infty(B_{R+1})}\int_{B_1}|z|^{-1}\,\mathrm d z+\|f_j\|_{L^1(\mathbb R^n)}\right)
 \le C_R.
\end{equation*}
On the other hand, since $\gamma_j>0$, the median normalization
\eqref{rig:eq:median} gives
\begin{equation*}
 \frac{|\mathbb S^n|}{2}\le \int_{B_1}f_j\,\mathrm d x\le |B_1|\exp\left(n\sup_{B_1}u_j\right).\end{equation*}
Thus $\sup_{B_1}u_j$ is uniformly bounded below. Together with
the gradient estimate and the uniform upper bound, this yields
\begin{equation*}
 \sup_j\|u_j\|_{W^{1,\infty}(B_R)}<\infty \qquad\text{for every }R\ge1.   
\end{equation*}
The Arzel\`a--Ascoli theorem therefore
gives, after passing to a subsequence,
\begin{equation*}
 u_j\longrightarrow u_0\quad\text{in }C^0_{\mathrm{loc}}(\mathbb R^n).    
\end{equation*}

Let $\Omega'\Subset\Omega\Subset\mathbb R^n\setminus\{0\}$ and $1<p<\infty$. The local potential estimates used in the proof of \cite[Theorem~2.1]{HMM2021}, applied to the normal
representation, give
\[
 \|u_j\|_{W^{n,p}(\Omega')}
 \le C\left(
 \|f_j\|_{L^p(\Omega)}
 +\|f_j\|_{L^1(\mathbb R^n)}
 +\|u_j\|_{L^\infty(\Omega)}
 \right)\le C,
\]
where $C$ is independent of $j$. Since the weights $|x|^{n\gamma_j}$ have uniformly bounded derivatives of every order on $\Omega$, the regularity iteration in the same proof yields
\begin{equation*}
 \sup_j\|u_j\|_{W^{m,p}(\Omega')}<\infty
 \qquad\text{for every integer }m\ge1.   
\end{equation*}
Compact Sobolev embeddings therefore give, up to a subsequence,
\[
 u_j\longrightarrow u_0
 \quad\text{in }C^\ell_{\mathrm{loc}}(\mathbb R^n\setminus\{0\})
 \qquad\text{for every }\ell\ge0.
\]

We next control the mass at infinity. By the uniform bound of $u_j$ in $B_1$ and the Kelvin symmetry, we obtain
\begin{equation*}
     f_j(x)\leq C|x|^{-2n-n\gamma_j}, \qquad |x|>1
\end{equation*}
for some constant $C$ independent of $j$. Consequently,
\begin{equation}\label{int fj uniform}
 \sup_j\int_{|x|>R}(1+|x|)f_j(x)\,\mathrm d x\leq CR^{-(n-1)}.   
\end{equation}
Thus both the masses and the first moments are uniformly integrable. Since \(\gamma_j\to0\) and \(u_j\to u_0\) locally uniformly,
\begin{equation*}
 f_j=|x|^{n\gamma_j}e^{nu_j}
 \longrightarrow e^{nu_0}
 \quad\text{in }L^1_{\mathrm{loc}}(\mathbb R^n).    
\end{equation*}
Combining this local convergence with \eqref{int fj uniform},
we obtain
\begin{equation*}
     \int_{\mathbb R^n}e^{nu_0}\,\mathrm d x=\lim_{j\to\infty}\int_{\mathbb R^n}f_j\,\mathrm d x
 =|\mathbb S^n|.
\end{equation*}

It remains to verify normality. Fix \(x_0\in\mathbb R^n\). Subtracting the normal representations at \(x\) and \(x_0\) gives
\begin{equation}\label{uj rep x x0}
 u_j(x)-u_j(x_0)=\frac{2}{|\mathbb S^n|}\int_{\mathbb R^n}\log\frac{|x_0-y|}{|x-y|}f_j(y)\,\mathrm d y.   
\end{equation}
On each fixed ball, we pass to the limit by local convergence. On its complement,
\begin{equation*}
     \left|\log\frac{|x_0-y|}{|x-y|}\right|\leq\frac{C_{x,x_0}}{|y|}
\end{equation*}
for large $|y|$, so the uniform mass bound controls the integral outside fixed balls. Therefore, we can pass to the limit in \eqref{uj rep x x0} to obtain
\begin{equation*}
    u_0(x)-u_0(x_0)=\frac{2}{|\mathbb S^n|}\int_{\mathbb R^n}\log\frac{|x_0-y|}{|x-y|}e^{nu_0(y)}\,\mathrm d y.
\end{equation*}
Rearranging this identity yields
\begin{equation*}
 u_0(x)=c_0+\frac{2}{|\mathbb S^n|}\int_{\mathbb R^n}\log\frac{1+|y|}{|x-y|}e^{nu_0(y)}\,\mathrm d y   
\end{equation*}
for some $c_0\in\mathbb R$. The fundamental-solution identity
\begin{equation*}
 (-\Delta)^{n/2}\log\frac1{|x|}=\frac{(n-1)!|\mathbb S^n|}{2}\delta_0    
\end{equation*}
therefore gives
\begin{equation*}
    (-\Delta)^{n/2}u_0=(n-1)!e^{nu_0} \quad\text{in }\mathcal D'(\mathbb R^n).
\end{equation*}
Thus $u_0$ is a normal solution of the unweighted equation. The classifications \cite[Theorem~2]{Martinazzi2009} in even dimensions and \cite[Theorem~1.3]{Hyder2019} in odd dimensions now give
\begin{equation*}
 u_0(x)=\log\frac{2\lambda}{1+\lambda^2|x-\xi|^2}   
\end{equation*} 
for some $\lambda>0$ and $\xi\in\mathbb R^n$ . By the uniform integrability \eqref{int fj uniform} and
Corollary \ref{coro:translation-identity},
\begin{equation*}
0=\lim_{j\to\infty}\int_{\mathbb R^n}x f_j(x)\,\mathrm d x=\int_{\mathbb R^n}x e^{nu_0(x)}\,\mathrm d x=|\mathbb S^n|\xi,    
\end{equation*}
so \(\xi=0\). Finally, passing to the limit in \eqref{rig:eq:median} and

The left-hand side is strictly increasing in $\lambda$ and equals $\frac{|\mathbb S^n|}{2}$ at $\lambda=1$. Therefore $\lambda=1$. Hence

By Theorem \ref{thm:kelvin}, we know that every normal solution is Kelvin symmetric. Let $u_\gamma^{\text{rad}}$ be the unique radial normal solution such that it is Kelvin symmetric with respect to $B_1$. 
We argue by contradiction. If the theorem were false, we could choose $\gamma_j\downarrow0$ and a normal solution $u_j$ of equation \eqref{eq:euclidean} with parameter $\gamma_j$, which is different from $u_j^{\text{rad}}:=u_{\gamma_j}^{\text{rad}}$ and also Kelvin symmetric with respect to $B_1$. By \eqref{rig:eq:median}, the Kelvin symmetry is equivalent to
\begin{equation*}
	\int_{B_1}(n-1)!|x|^{n\gamma_j}e^{nu_j^{\text{rad}}}\,dx=
	\int_{B_1}(n-1)!|x|^{n\gamma_j}e^{nu_j}\,dx=\frac{(n-1)!(1+\gamma_j)}{2}|\mathbb S^n|.
\end{equation*}

By Proposition \ref{rig:prop:compactness}, we have
\begin{equation}\label{uj conver U}
 u_j, u_j^{\text{rad}}\longrightarrow U=\log\frac{2}{1+|x|^2} \quad\text{in }C^0_{\mathrm{loc}}(\mathbb R^n)
\end{equation}


Set
\begin{equation*}
h_j:=\frac{u_{j}^{\text{rad}}-u_{j}}{\|u_{j}^{\text{rad}}-u_{j}\|_{L^\infty(\mathbb R^n)}}.
\end{equation*}
Then
\begin{equation}\label{eq:maximum}
h_j(\frac{x}{|x|^2})=h_j(x),\qquad
\|h_j\|_{L^\infty(\mathbb R^n)}=\max_{\overline B_1}|h_j|=1.
\end{equation}
We are going to show $h\equiv 0$ to obtain a contradiction.

Subtracting the equations of $u_j$ and $u_j^{\text{rad}}$ yields
\begin{equation*}\label{eq:linear}
(-\Delta)^{n/2}h_j=a_jh_j,\qquad
a_j(x)=n!|x|^{n\gamma_j}\int_0^1 e^{n((1-s)u_{j}^{\text{rad}}(x)+su_{j}(x))}\,ds.
\end{equation*}
For every fixed $R$,  \eqref{uj conver U} gives
\begin{equation}\label{eq:coeff}
0\le a_j\le C_R\quad\hbox{on }B_R,\qquad
a_j\longrightarrow n!e^{nU}\quad\hbox{in }L^p(B_R)
\quad(1\le p<\infty).
\end{equation}


There is also a uniform global $L^1$ bound requiring only the mass identity: 
\begin{equation}\label{eq:L1}
\int_{\mathbb R^n}a_j\,dx\le\frac{n!}{2}\int_{\mathbb R^n}|x|^{n\gamma_j}
\bigl(e^{nu_{j}^{\text{rad}}}+e^{nu_{j}}\bigr)\,dx=n!(1+\gamma_j)|\mathbb S^n|\le C.
\end{equation}

Subtracting the two normal representations of $u_j$ and $u_j^{\text{rad}}$ gives
\begin{equation}\label{eq:repr}
h_j(x)-h_j(0)=-\frac{2}{(n-1)!|\mathbb S^n|}\int_{\mathbb R^n}\log\frac{|x-y|}{|y|}\,a_j(y)h_j(y)\,dy.
\end{equation}
Hence
\begin{equation}\label{grad hj}
\nabla h_j(x)=-\frac{2}{(n-1)!|\mathbb S^n|}\int_{\mathbb R^n}\frac{x-y}{|x-y|^2}\,a_j(y)h_j(y)\,dy.
\end{equation}

 For $x\in B_R$, we split the integral into $|x-y|<1$ and its complement. Differentiate \eqref{grad hj} once more, by \eqref{eq:coeff} and \eqref{eq:L1}, we obtain
\begin{equation*}
	\begin{aligned}
		|\nabla h_j(x)|+|D^2h_j(x)|\le &C\int_{|x-y|<1}
		\frac{1}{|x-y|^2}\,a_j(y)|h_j(y)|\,dy
		+C\int_{|x-y|\geq 1}
		\frac{1}{|x-y|}\,a_j(y)|h_j(y)|\,dy\\
		\le& C_R\int_{|z|<1}
		\bigl(|z|^{-2}\bigr)\,dz+C\int_{\mathbb R^n}a_j\,dy\\
		\le &C_R.
	\end{aligned}
\end{equation*}
Here $D^2h_j$ is first understood in the weak sense. Thus, by the Arzel\`a--Ascoli theorem, after passing to a subsequence, we have
\begin{equation}\label{eq:convergence}
h_j\longrightarrow h\quad\hbox{in }C^1_{\mathrm{loc}}(\mathbb R^n).
\end{equation}
In particular, convergence is uniform on $\overline B_1$. Equation \eqref{eq:maximum} therefore gives
\begin{equation}\label{max h=1}
\max_{\overline B_1}|h|=1,\qquad \|h\|_{L^\infty(\mathbb R^n)}=1.
\end{equation}

Now we pass to the limit in \eqref{eq:repr}. For fixed $x$ and $R>2|x|+1$, we have
\begin{equation*}
\left|\log\frac{|x-y|}{|y|}\right|\le\frac{C_x}{|y|}, \quad |y|>R,
\end{equation*}
so by \eqref{eq:L1}, 
\begin{equation*}
	\int_{|y|>R}\left|\log\frac{|x-y|}{|y|}\right|a_j(y)|h_j(y)|\,dy\le\frac{C_x}{R}.
\end{equation*}
 On bounded sets, we have the convergence \eqref{eq:coeff} and \eqref{eq:convergence}. Consequently,
\begin{equation}\label{eq:limitintegral}
h(x)-h(0)=-\frac{2}{(n-1)!|\mathbb S^n|}\int_{\mathbb R^n}\log\frac{|x-y|}{|y|}\,n!e^{nU(y)}h(y)\,dy.
\end{equation}


We observe that \eqref{eq:limitintegral} is exactly the integral form of the linearized equation at $U$. In other words,
\begin{equation*}
	(-\Delta)^{n/2}h=n!e^{nU}h\quad\hbox{in }\mathbb R^n.
\end{equation*}
By Corollary \ref{coro:nondegen}, the nondegeneracy of $U$ implies that $h=\sum_{i=0}^n c_iZ_i$ for some $c_i\in\mathbb R$, where $Z_0,\dots, Z_n$ are defined by \eqref{eq:kernel}.


Since $\nabla u_j(0)=\nabla u_j^{\text{rad}}(0)=0$ for all $j$, we have  $\nabla h_j(0)=0$. By \eqref{eq:convergence}, $\nabla h(0)=0$. Since
\begin{equation*}
\nabla Z_0(0)=0,\quad \partial_k Z_i(0)=2\delta_{ik}, \qquad i,k=1,\dots,n,
\end{equation*}
we get $c_1=\dots=c_n=0$.

 For every $x\ne0$, equation \eqref{eq:maximum} passes to the limit and gives $h(\frac{x}{|x|^2})=h(x)$. Since
\begin{equation*}
Z_0(\frac{x}{|x|^2})=-Z_0(x),\quad Z_i(\frac{x}{|x|^2})=Z_i(x),\qquad i=1,\dots,n,
\end{equation*}
we get $c_0=0$.

In conclusion, we have $h\equiv0$, contradicting \eqref{max h=1}. The uniqueness up to a dilation follows immediately. This completes the proof of Theorem \ref{thm:rad small ga}.
