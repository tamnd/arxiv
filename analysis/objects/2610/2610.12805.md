<!-- theorem-1: body_md -->
\label{thm:complete-loglinear-coordinates}
For any strictly positive binary probability tables $p$ and $q$,
\begin{equation}
    p\sim q
    \quad\Longleftrightarrow\quad
    J_S(p)=J_S(q)
    \qquad
    \text{for all }|S|\geq2.
    \label{eq:complete-loglinear-coordinates}
\end{equation}

<!-- theorem-2: body_md -->
\label{thm:positive-binary-sklar}
Fix any strictly positive dependence nucleus $    \mathcal{N}=[p]$.
For every collection of strictly positive one-body marginals, there exists
one and only one joint probability table in $\mathcal{N}$ having those
marginals.
Equivalently,
\begin{equation}
    p
    \longleftrightarrow
    \bigl(
        \text{one-body marginals},
        \mathcal{N}(p)
    \bigr).
    \label{eq:sklar-nucleus-decomposition}
\end{equation}
Using the complete coordinates of Theorem~\ref{thm:complete-loglinear-coordinates},
this can also be written as
\begin{equation}
    \boxed{
    p
    \longleftrightarrow
    \bigl(
        \text{one-body marginals},
        \{J_S:\ |S|\geq2\}
    \bigr).
    }
    \label{eq:sklar-J-decomposition}
\end{equation}
<!-- end theorem-2 -->
