<!-- lem-1: body_md -->
\label{lem:kernel}
Let $a$ and $b$ be positive integers. For every integer $k\in [0,\ ab/2]$, we have
\begin{eqnarray}\label{formal+kernel}
 \dim_{\mathbb Q}\ker\bigl(\mathcal D:V_{b,k}\longrightarrow V_{b,k-1}\bigr)
 &=&[q^k](1-q)\mathcal G_{a,b}(q).
\end{eqnarray}
In particular, $\mathcal G_{a,b}(q)$ is unimodal.
<!-- end lem-1 -->
<!-- proof-1: body_md -->
We will first prove that (\ref{formal+kernel}) holds. For $k=0$, $[q^0](1-q)\mathcal G_{a,b}(q)=1$ and it follows from $V_{b,0}=\mathbb Q x_0^b$ and $\mathcal D x_0^b=0$ that
the dimension of the kernel is one. So (\ref{formal+kernel}) is obvious for $k=0$. In the following it suffices to prove that
$\mathcal D:V_{b,k}\to V_{b,k-1}$ is surjective for
$1\le k\le ab/2$. Indeed, combining surjectivity and
\eqref{eq:kernel-dimension} yields
\[
 \begin{aligned}
 \dim\ker\mathcal D
 &=\dim V_{b,k}-\dim V_{b,k-1}\\
 &=[q^k]\mathcal G_{a,b}(q)-[q^{k-1}]\mathcal G_{a,b}(q)\\
 &=[q^k](1-q)\mathcal G_{a,b}(q).
 \end{aligned}
\]


We now present the proof of the required surjectivity. In the following proof,
we extend the domain $\mathbb Q[x_0,\ldots,x_a]$ of $\mathcal D,\mathcal E,\mathcal H$ and $V_{b,k}$ onto $\mathbb R[x_0,\ldots,x_a]$.
Since the corresponding matrix of the linear operator $\mathcal D$ acting on monomial bases has rational entries, this extension does not change its
rank and kernel dimension. Thus surjectivity and the dimension of its kernel are unaffected by this
change of scalars.

For any $\textbf{x}^{\boldsymbol{\alpha}}\in V_{b,k}$ and $\textbf{x}^{\boldsymbol{\beta}}\in V_{b,k}$, define a bilinear inner product by
\[
 \langle \textbf{x}^{\boldsymbol{\alpha}},\textbf{x}^{\boldsymbol{\beta}}\rangle
 :=\begin{cases}
 \displaystyle\left(\prod_{i=0}^{a}\alpha_i!\right)
   \prod_{i=0}^{a}\left(\frac{i!(a-i)!}{a!}\right)^{\alpha_i},
   &\boldsymbol{\alpha}=\boldsymbol{\beta},\\
 0,&\boldsymbol{\alpha}\ne\boldsymbol{\beta}.
 \end{cases}
\]
Obviously, this inner product is nonnegative.
We write $\|f\|^2=\langle f,f\rangle$.
We first give the next claim.

\begin{cla} The inner product $\langle\cdot,\cdot\rangle$ on $V_{b,k}$ makes $\mathcal E$ adjoint to $\mathcal D$, i.e.,
\begin{eqnarray}\label{inner+product+adjoint}
\langle\mathcal D f,g\rangle=\langle f,\mathcal E g\rangle
\end{eqnarray}
for all $f,g$ in $V_{b,k}$. 
\end{cla}

\begin{proof}
To obtain this identity, we will assign positive
weights to the monomial basis. Let $\textbf{x}^{\boldsymbol{\alpha}}\in V_{b,k}$ be an arbitrary monomial with $\alpha_i>0$
for $1\le i\le a$. And assume that
$\boldsymbol{\beta}=\boldsymbol{\alpha}-\boldsymbol{e}_i+\boldsymbol{e}_{i-1}$, where $\boldsymbol{e}_i$ is the coordinate
vector corresponding to $x_i$. Then in terms of the definition of the inner product $\langle\cdot,\cdot\rangle$, we obtain
\begin{eqnarray}\label{form+two+vectors}
\frac{\|\textbf{x}^{\boldsymbol{\beta}}\|^2}{\|\textbf{x}^{\boldsymbol{\alpha}}\|^2}
 =\frac{\alpha_{i-1}+1}{\alpha_i}\frac{a-i+1}{i},
\end{eqnarray}
In addition, it follows from the definitions of the operator $\mathcal D$ and the inner product $\langle\cdot,\cdot\rangle$ that
\begin{eqnarray}
\langle\mathcal D \textbf{x}^{\boldsymbol{\alpha}},\textbf{x}^{\boldsymbol{\beta}}\rangle
 &=&i\,\alpha_i\,\|\textbf{x}^{\boldsymbol{\beta}}\|^2,\label{inner+product+adjoint+1}\\
 \langle \textbf{x}^{\boldsymbol{\alpha}},\mathcal E \textbf{x}^{\boldsymbol{\beta}}\rangle
 &=&(a-i+1)(\alpha_{i-1}+1)\,\|\textbf{x}^{\boldsymbol{\alpha}}\|^2.\label{inner+product+adjoint+2}
\end{eqnarray}
Thus, by (\ref{form+two+vectors})-(\ref{inner+product+adjoint+2}), we obtain
\begin{eqnarray}
\langle\mathcal D \textbf{x}^{\boldsymbol{\alpha}},\textbf{x}^{\boldsymbol{\beta}}\rangle
 &=& \langle \textbf{x}^{\boldsymbol{\alpha}},\mathcal E \textbf{x}^{\boldsymbol{\beta}}\rangle
\end{eqnarray}
for any $\textbf{x}^{\boldsymbol{\alpha}}\in V_{b,k}$ with all $\alpha_i>0$ and $\boldsymbol{\beta}=\boldsymbol{\alpha}-\boldsymbol{e}_j+\boldsymbol{e}_{j-1}$, where $\boldsymbol{e}_j$ is the coordinate
vector corresponding to $x_j$. In addition, for any pair of monomials not related in this way, their inner
product is zero. Therefore, using the bilinearity of the inner product gives the identity (\ref{inner+product+adjoint}) for all $f,g$ in $V_{b,k}$. This prove this claim.
\end{proof}


Fix $1\le k\le ab/2$. Let $y\in V_{b,k-1}$ be orthogonal
to the image of $\mathcal D:V_{b,k}\to V_{b,k-1}$.
For every $x\in V_{b,k}$, adjointness (\ref{inner+product+adjoint}) gives
$\langle x,\mathcal E y\rangle=\langle\mathcal D x,y\rangle=0$.
Since $\mathcal E y\in V_{b,k}$, taking $x=\mathcal E y$
                                                                                                                                 
we have
$\langle\mathcal D\mathcal E y,y\rangle=0$.
On the other hand, using $\mathcal D\mathcal E-\mathcal E\mathcal D=\mathcal H$,
$\mathcal H y=(ab-2k+2)y$, and adjointness, we obtain
\[
 \begin{aligned}
 \langle\mathcal D\mathcal E y,y\rangle
  &=\langle(\mathcal E\mathcal D +\mathcal H) y,y\rangle\\
  &=\langle\mathcal E\mathcal D y,y\rangle
    +\langle\mathcal H y,y\rangle\\
                                              
                                    
  &=\langle\mathcal D y,\mathcal D y\rangle
    +\langle(ab-2k+2)y,y\rangle\\
  &=\|\mathcal D y\|^2+(ab-2k+2)\|y\|^2.
 \end{aligned}
\]
In consequence, we derive $\|\mathcal D y\|^2+(ab-2k+2)\|y\|^2=0$.
Here $ab-2k+2\ge2$. Thus, in terms of the nonnegativity of the inner product, we have $\|y\|^2=0$, which further implies $y=0$.
Thus $(\operatorname{im}\mathcal D)^\perp=\{0\}$ in $V_{b,k-1}$.
Since $V_{b,k-1}$ is finite-dimensional,
$\operatorname{im}\mathcal D=V_{b,k-1}$, so $\mathcal D$ is surjective.

to show that the coefficients are nondecreasing up to $ab/2$, which is immediate from the nonnegativity of $[q^k](1-q)\mathcal G_{a,b}(q)$ for $1\le k\le ab/2$ in the kernel formula (\ref{formal+kernel}). This completes the proof of this lemma.

\label{lem:root-evaluation}
Let $A,B$ and $r$ be positive integers satisfying $A\ge B$, $r>1$ and $r|B$.
If $\zeta\in\mathbb C$ be a primitive $r$th root of unity, then
\[
  \left.{A-1\brackB-1}\right|_{q\rightarrow\zeta}
  =\begin{cases}
     0,&r\nmid A,\\[2pt]
     \displaystyle\binom{A/r-1}{B/r-1},&r\mid A.
   \end{cases}
\]

<!-- proof-3: body_md -->
By (\ref{def+Gauss+poly}) of Gaussian polynomials, we have
\begin{equation}\label{gass:A-1+B-1}
{A-1\brackB-1}
  =\frac{N(q)}
         {D(q)},
\end{equation}
where
\[
N(q):=\prod_{j=A-B+1}^{A-1}(1-q^j),
\quad
D(q):=\prod_{j=1}^{B-1}(1-q^j).
\]
We will evaluate this quotient (\ref{gass:A-1+B-1}) by separating the factors that vanish at $\zeta$.
Since \(\zeta\) is a primitive \(r\)-th root of unity, we have
\[
\zeta^j=1
\iff
r\mid j.
\]
Thus a factor \(1-q^j\) vanishes at \(q=\zeta\) exactly when \(r\mid j\), and every such zero is simple because
\[
\frac{d}{dq}(1-q^j)\Big|_{q\rightarrow\zeta}
=
-j\zeta^{j-1}\neq 0.
\]

For a set, let
$
Z(I):=|\{j\in I:\ r\mid j\}|
$
denote the number of multiples of \(r\) in \(I\).
We also let $B=k\,r$ since \(r\mid B\). For the denominator $D(q)$, we have
\[
Z([1,B-1])=k-1.
\]
 Hence \(D(q)\) has a zero \(q=\zeta\) of order \(k-1\). For the numerator $N(q)$, we have
\begin{equation}\label{number+zeros:A-1+B-1}
Z([A-B+1,A-1])
=
\left\lfloor \frac{A-1}{r}\right\rfloor
-
\left\lfloor \frac{A-B}{r}\right\rfloor
=
\left\lfloor \frac{A-1}{r}\right\rfloor
-
\left\lfloor \frac{A}{r}\right\rfloor
+
k.
\end{equation}


We now split into the following two cases to consider:

\medskip
\noindent
\textbf{Case 1: \(r\nmid A\).}
Then
\[
\left\lfloor \frac{A-1}{r}\right\rfloor
=
\left\lfloor \frac{A}{r}\right\rfloor,
\]
so
\[
Z([A-B+1,A-1])=k.
\]
Thus  $N(q)$ has one more zero \(q=\zeta\) than $D(q)$. Hence
we obtain
\begin{equation}
\left.{A-1\brackB-1}\right|_{q\rightarrow\zeta}=\left.\frac{N(q)}{D(q)}\right|_{q\rightarrow\zeta}=0.
\end{equation}

\medskip
\noindent
\textbf{Case 2: \(r\mid A\).} Let $A=\ell\, r$. Obviously, $\ell\geq k\geq1$ since $A\geq B$.
Then
\[
\left\lfloor \frac{A-1}{r}\right\rfloor
=
\ell-1,
\quad
\left\lfloor \frac{A}{r}\right\rfloor
=
\ell.
\]
Therefore,  (\ref{number+zeros:A-1+B-1}) gives
\[
Z([A-B+1,A-1])
=
\ell-1-\ell+k
=
k-1.
\]
So the numerator $N(q)$ and denominator $D(q)$ have the zero \(q=\zeta\) of the same order $k-1$.

The nonvanishing factors in $N(q)$ are those with \(r\nmid j\). In the interval
\[
A-B+1\le j\le A-1,
\]
there are \(B-1=k\,r-1\) indices, among which \(k-1\) are multiples of \(r\). Hence each nonzero residue class modulo \(r\) occurs exactly \(k\) times. For the
denominator, it is the same as the numerator. Therefore the products of the nonvanishing factors in \(N(q)\) and \(D(q)\) have the same value at \(q=\zeta\), and their ratio is \(1\).

It remains to compare the vanishing factors. The multiples of \(r\) in $N(q)$ correspond to
\[
j=r\, m,
\quad
m=l-k+1,l-k+2,\dots,l-1.
\]
The multiples of \(r\) in $D(q)$ correspond to
$
j=r\, m,\,
m=1,2,\dots,k-1.
$
Thus, we obtain
\begin{eqnarray*}
\left.\frac{N(q)}{D(q)}\right|_{q\rightarrow\zeta}
&=&\lim_{q\to \zeta}
\frac{
\prod_{m=l-k+1}^{l-1}(1-q^{rm})
}{
\prod_{m=1}^{k-1}(1-q^{rm})
}\\
&
=&\lim_{q\to \zeta}
\frac{
\prod_{m=l-k+1}^{l-1}(1-q^r)\left(1+q^r+q^{2r}+\cdots+q^{(m-1)r}\right)
}{
\prod_{m=1}^{k-1}(1-q^r)\left(1+q^r+q^{2r}+\cdots+q^{(m-1)r}\right)
}\\
&=&\frac{
\prod_{m=l-k+1}^{l-1} m
}{
\prod_{m=1}^{k-1} m
}\\
&
=&\binom{l-1}{k-1}.
\end{eqnarray*}

Substituting \(l=A/r\) and \(k=B/r\), we get
\[
\left.
\begin{bmatrix}
A-1\\
B-1
\end{bmatrix}\right|_{q\rightarrow \zeta}
=
\binom{A/r-1}{B/r-1}.
\]

Combining both two cases yields the proof of this lemma.

$R_d(\zeta).$

\left.c'\right|\frac{m}{d}.

Obviously, the condition \(c'\mid n/d\) is equivalent to
\[
\left.\frac{c}{\gcd(c,d)}\right| \frac nd
\iff
\operatorname{lcm}(c,d)\mid n.

By \(d\mid r\) and \(r\mid m\), this is further equivalent to \(c\mid n\). Hence, we derive
\begin{equation}\label{eq:Rd+zata+2}

We begin to compute $R(\zeta)$ and claim $R(\zeta)=P(\zeta)$. We arrange the proof of this claim as follows.
Now extend the definition by setting \(R_1(q)=P(q)\) and \(\mu(1)=1\). We claim that

\sum_{d\mid r}\mu(d)R_d(\zeta)=0.

By (\ref{def+Rq+new}), for each $k\ge0$, the coefficient of $q^k$ in $R(q)$ is
\begin{eqnarray*}
 R_k&=&
 \sum_{\substack{\lambda\in\mathcal P\\
                  wt(\lambda)=k,\ M(\lambda)>0}}M(\lambda)
 - \sum_{\substack{\lambda\in\mathcal P\\
                  wt(\lambda)=k,\ M(\lambda)<0}}(-M(\lambda))\\
&\geq&\bigl|\{\lambda\in\mathcal P:
       wt(\lambda)=k,\ M(\lambda)=1\}\bigr|
- \sum_{\substack{\lambda\in\mathcal P\\
                  wt(\lambda)=k,\ M(\lambda)<0}}(-M(\lambda)).
\end{eqnarray*}
So in order to show $R(q)\geq_q 0$, it suffices to prove 
\begin{eqnarray}\label{ineq+M}
 \bigl|\{\lambda\in\mathcal P:
       wt(\lambda)=k,\ M(\lambda)=1\}\bigr|
\geq\sum_{\substack{\lambda\in\mathcal P\\
                  wt(\lambda)=k,\ M(\lambda)<0}}-M(\lambda).
\end{eqnarray}


Fix $\lambda\in\mathcal P$ with $M(\lambda)<0$, and set
$c=-M(\lambda)$ and $H=H(\lambda)$. For each such partition with $M(\lambda)<0$, we will use $T_h(\lambda)$ to construct $c$ distinct
partitions $\lambda'$ with $wt(\lambda')=wt(\lambda)$ and $M(\lambda')=1$, with no overlap
between different sources.
Obviously, the partition $\lambda$ is nonempty since $M(\lambda)<0$ and $M(\varnothing)=1$.
Thus, by Lemma~\ref{lem:multiplicity} (iii) and (iv), $H$ has $s\ge4$ distinct prime
divisors and $c\le2^{s-1}-1$. We first prove the next claim.
\begin{cla}
In terms of $T_h(\lambda)$, at least $c$ choices of the splitting parameter
$h$ are available. 
\end{cla}
\begin{proof}Let $p$ be the smallest prime divisor of $H$ and set
\[
 \mathcal A_\lambda:=
 \{h\in\mathbb Z_{\ge1}:p\mid h,\ 2h<H,\
                         \gcd(h/p,H/p)=1\}.
\]
In terms of definition of $H$, the integer $H/p$ is odd. Writing $h=pj$, we have
\[|\mathcal A_\lambda|=|\{
 1\le j<\frac{H/p}{2},\quad \gcd(j,H/p)=1.\}|
\]
The units $j$ modulo $H/p$ occur in pairs $j$ and $H/p-j$. Since
$H/p$ is squarefree, this pairing and the product formula for
Euler's totient function give
\[
 |\mathcal A_\lambda|
 =\frac12\prod_{\substack{r\mid H\\r\ne p}}(r-1)
 \ge\frac{2\cdot4\cdot6}{2}\,2^{s-4}
 =3\cdot2^{s-1}>c,
\]
where for the first inequality from the first three primes dividing $H/p$ are at
least $3,5,7$, and every remaining prime contributes a factor of
at least $2$. This proves this claim.
\end{proof}

From this claim, we assume that

$\mathcal A_\lambda$. In terms of the definition (\ref{def+M+labta}) of $M(\lambda)$, we next verify that each split
$\nu=T_{h_i(\lambda)}(\lambda)$ belongs to $\mathcal S_p$ and satisfies

 \lambda_1\ge H>2h>0.

Lemma~\ref{lem:split} are satisfied. Then by Lemma~\ref{lem:split}, we obtain $\lambda'=\nu=T_{h_i(\lambda)}(\lambda)$. The largest entry of $\nu$ is at most
$\lambda_1\le n-m$. Let $\phi$ be the second-smallest prime divisor
of $H$. By Lemma~\ref{lem:multiplicity} (2),
$\ell(\lambda)+1\le m/\phi$, which combines
$p\phi\mid m$ and $\phi>p$ to give
\[
 \frac mp-\frac m\phi=\frac{m}{p\phi}(\phi-p)\ge1,
 \qquad
 \ell(\nu)=\ell(\lambda)+1\le\frac m\phi\le\frac mp-1.
\]

\[
 \gcd(r,\nu_1,\ldots,\nu_{\ell(\lambda)+1})
 =\gcd(r,\lambda_1,\ldots,\lambda_{\ell(\lambda)},h).
\]
Using $rad(\gcd(u,v))=\gcd(rad (u),v)$ for $u,v\in\mathbb Z_{\ge1}$, we obtain
\[
 \begin{aligned}

 &=rad\bigl(\gcd(r,\lambda_1,\ldots,
                       \lambda_{\ell(\lambda)},h)\bigr)=\gcd(H,h)=p,
 \end{aligned}
\]
where the last equality follows from $h=p(h/p)$,
$\gcd(h/p,H/p)=1$ and the squarefreeness of $H$. Hence, we obtain
$p\mid H(\nu)$ and $\ell(\nu)\le m/p-1$, so
$\nu\in\mathcal S_p$. Since $H(\nu)=p$,

leading monomials. So $C_k\ge R_k$ for $0\le k\le\delta/2.$

Let $a\ge b\ge2$ be integers. For $\lambda\vdash b$, set

Then

 {a+b\brackb}
 =\sum_{\lambda\vdash b}
 q^{2\sum_{i\ge1}\binom{\lambda_i}{2}}

 {j(a+2)-Y_{j-1}-Y_{j+1}\brack\lambda_j-\lambda_{j+1}}
\end{equation}
and every nonzero summand is symmetric and unimodal

<!-- prop-5: body_md -->
\label{prop:domination+m6+12}
Let$ n\geq 12$, $6\mid n$ and $r=\gcd(n,6)>1$.
Then $C_k\ge R_k$ for $0\le k\le5(n-6)/2$.
<!-- end prop-5 -->
<!-- proof-14: body_md -->
We first compute $R(q)$, then obtain a lower bound for $C_k$ from
four summands of \eqref{eq:KOH}. We finish by comparing this bound
with $R_k$ in odd and even degrees.

The hypotheses imply $\gcd(n,m)=6$. In the divisor
formula (\ref{def+R}) for $R(q)$, the divisors greater than $1$ are $2,3,6$, with
M\"obius values $-1,-1,1$, respectively. Thus
\begin{equation}\label{eq:six-correction}
 R(q)=\genfrac{[}{]}{0pt}{}{n/2-1}{2}_{q^{2}}+[n/3-1]_{q^3}-1.
\end{equation}
For the KOH identity, taking $a=n-6$ and $b=5$ yields
\[
 P(q)={n-1\brack5}={(n-6)+5\brack5}.
\]
We only need to use the four summands of \eqref{eq:KOH} indexed by
$(1^5)$, $(2,1^3)$, $(2,2,1)$, and $(3,1,1)$ to bound $C_k$.
They are
\[
\begin{array}{c|l}
 \lambda & \text{summand}\\ \hline
 (1^5)   &[5n-29]_{q}\\
 (2,1^3) &q^2[n-7]_{q}[4n-25]_{q}\\
 (2,2,1) &q^4[2n-15]_{q}[3n-21]_{q}\\
 (3,1,1) &q^6{n-8\brack2}[3n-21]_{q}.
\end{array}
\]
For example, if $\lambda=(3,1,1)$, then
$(Y_0,Y_1,Y_2,Y_3,Y_4)=(0,3,4,5,5)$ and
$(\lambda_1-\lambda_2,\lambda_3-\lambda_4)=(2,1)$,
which gives the last row. To compute their contributions to
$C_k$ for $0\le k\le\delta/2$, multiply each summand by $1-q$:
\[
\begin{aligned}
 (1-q)[5n-29]_{q}
  &=1-q^{5n-29},\\
 (1-q)q^2[n-7]_{q}[4n-25]_{q}
  &=q^2[n-7]_{q}-q^{4n-23}[n-7]_{q},\\
 (1-q)q^4[2n-15]_{q}[3n-21]_{q}
  &=q^4[2n-15]_{q}-q^{3n-17}[2n-15]_{q},\\
 (1-q)q^6{n-8\brack2}[3n-21]_{q}
  &=q^6{n-8\brack2}-q^{3n-15}{n-8\brack2}.
\end{aligned}
\]
For these four first differences, their negative terms begin above the range $0\le k\le5(n-6)/2$, since
$n\ge12$ and
\[
 \frac{5(n-6)}2
 <3n-17
 =\min\{5n-29,4n-23,3n-17,3n-15\}.
\]
So we next restrict these four first differences to consider their former terms.
Every remaining summand of \eqref{eq:KOH} has nonnegative first
differences in the same range by the symmetry and unimodality of Gaussian  polynomials. Hnece, for $0\le k\le5(n-6)/2$, we have
\begin{equation}\label{eq:six-bound}
 C_k\ge \mathbf1_{k=0}
  +\mathbf1_{2\le k\le n-6}
  +\mathbf1_{4\le k\le2n-12}
  +[q^{k-6}]{n-8\brack2}.
\end{equation}

We now compare this lower bound with the coefficients of $R(q)$.
By \eqref{eq:six-correction}, $\deg R(q)=2n-12$, so $R_k=0$ for
$k>2n-12$. As the right-hand side of \eqref{eq:six-bound} is
nonnegative, it remains to consider $0\le k\le2n-12$ in terms of the following two cases.

\textbf{Case 1}: For odd $k$, combining \eqref{eq:six-correction} and \eqref{eq:six-bound} gives
\[
 R_k=\mathbf1_{\{3\mid k,\ 1\le k\le n-6\}}
 \le\mathbf1_{2\le k\le n-6}\le C_k.
\]
For the first three even degrees, the same equations give
$c_0\ge1$, $c_2\ge1$, and $c_4\ge2$, while
$(r_0,r_2,r_4)=(1,1,2)$.

\textbf{Case 2}: For the remaining even degrees, write $k=2h$ with
$3\le h\le n-6$. We compute two Gaussian coefficients in
\eqref{eq:six-bound} and \eqref{eq:six-correction} by
\eqref{eq:coeff+Gauss}. In the two counts, the partitions have at most
two parts and are represented by $(2h-6-v,v)$ and $(h-v,v)$,
respectively, with zero parts omitted:
\begin{align*}
 [q^{2h-6}]{n-8\brack2}
 &=\bigl|\{v\in\mathbb Z:\max(0,2h-n+4)\le v\le h-3\}\bigr|\\
 &=\min(h-2,n-h-6)
   &&(3\le h\le n-7),\\
 [q^{2h}]\genfrac{[}{]}{0pt}{}{n/2-1}{2}_{q^{2}}
 &=\bigl|\{v\in\mathbb Z:\max(0,h-n/2+3)
                     \le v\le\lfloor h/2\rfloor\}\bigr|\\
 &=\left\lfloor\frac{\min(h,n-h-6)}2\right\rfloor+1
   &&(0\le h\le n-6).
\end{align*}
The second count and \eqref{eq:six-correction} now give
\[
 R_{2h}=\left\lfloor\frac{\min(h,n-h-6)}2\right\rfloor+1
          +\mathbf1_{\{3\mid h,\ 2h\le n-6\}}
 \qquad(3\le h\le n-6).
\]
We compare this formula with \eqref{eq:six-bound} by the following three cases:


\textbf{Case 2.1:} If $3\le h\le n/2-3$, then 
$ C_{2h}\ge h \ge \lfloor h/2\rfloor+1+\mathbf1_{3\mid h}= R_{2h}.$



\textbf{Case 2.2:}  If $n/2-2\le h\le n-7$, then
\[
 C_{2h}\ge n-h-5
 \ge\left\lfloor\frac{n-h-6}{2}\right\rfloor+1
 =R_{2h}.
\]



\textbf{Case 2.3:} If $h=n-6$, then combining \eqref{eq:six-correction} and \eqref{eq:six-bound}
gives $C_{2n-12}\ge1=R_{2n-12}$.


This covers every integer $0\le k\le5(n-6)/2$. The proof is complete.

<!-- prop-6: body_md -->
\label{prop:W-positive}
Let $n,m\in\mathbb Z$ satisfy $3\le m\le n/2$ and $r=\gcd(n,m)>1$.
Then $W(q)\ge_q0$.
<!-- end prop-6 -->
<!-- proof-15: body_md -->
Recall $\delta=(n-m)(m-1)$. Note by Lemma~\ref{lem:polynomiality} that $\deg W(q)=\delta-n-m$. If $\delta-n-m<0$, then the polynomial $W(q)$ must be zeros. If $\delta-n-m=0$, then $m=3$ and $n=9$, or $m=4$ and $n=8$. In this case, it is easy to obtain that $W(q)$ is a nonnegative constant. 
So in the following, we assume $\delta-n-m>0$. By Lemma~\ref{lem:polynomiality}, $W(q)=q^{\delta-n-m}W(q^{-1})$.
In consequence, in order to prove $W(q)\ge_q0$, it suffices to show that $[q^k]W(q)$ is nonnegative for $0\leq k\leq (\delta-n-m)/2$.
By (\ref{def+R}), we rewrite
$W(q)$ as
\begin{eqnarray}\label{W+rewrite}
 W(q)=\frac{R(q)}{1-q^m}
 +\sum_{\substack{d\mid r,d>1}}
   \mu(d)\frac{q^{L_d}R_d(q)}{1-q^m}.
\end{eqnarray}
By (\ref{Ld+posi}), we have $L_d\geq0$, where equality holds if and if $d=2$ and $m=4$ for every $d\mid r$ with $d>1$,
\begin{eqnarray}
L_d-\frac{(\delta-n-m)}{2}
 =\frac{m(d-2)(n-m)}{2d}\ge0
\end{eqnarray}
with equality if and only if $d=2$.
Since $R_d(0)=1$, the factor
$q^{L_d}R_d(q)/(1-q^m)$ has the lowest degree $L_d$.
Now we consider nonnegativity of $[q^k]W(q)$ by the following two cases:


\textbf{Case 1}: $0\le k<\frac{\delta-n-m}{2}$.
Combining (\ref{W+rewrite}) and Proposition~\ref{prop:R-positive} gives
\[
 [q^k]W(q)=\sum_{j\ge0}R_{k-jm}\ge0.
\]

\textbf{Case 2}: $k=\frac{\delta-n-m}{2}$.
If $d>2$, then $L_d-\frac{(\delta-n-m)}{2}>0$. Thus, in the second summation of (\ref{W+rewrite}), only the term with $d=2$ contributes in degree $L_2$.
Note $\mu(2)=-1$ and $R_2(0)=1$. It follows from (\ref{W+rewrite}) and Proposition~\ref{prop:R-positive} that
\begin{equation}\label{eq:midpoint-compensation}
 [q^{L_2}]W(q)=\begin{cases}
    \sum_{j\ge0}R_{L_2-jm}\geq0,&2\nmid r,\\[2pt]
    \displaystyle \sum_{j\ge0}R_{L_2-jm}-1\ge R_{L_2}-1,&2\mid r.
  \end{cases} 
\end{equation}
In the following we will prove $R_{L_2}\geq1$.
 
 
 We first consider the case $n-m>6$. Note for every $d\mid r$ with $d\ge3$ that
\[
 \begin{aligned}
 L_2-\deg R_d
 =m(n-m)\left(\frac12-\frac1d\right)-m\ge m\left(\frac{n-m}{6}-1\right)>0.
 \end{aligned}
\]
So the polynomial $R_d(q)$ has no the term $q^{L_2}$ for $d\geq3$.
It follows that $$R_{L_2}=[q^{L_2}]R(q)=[q^{L_2}]R_2(q)=[q^{L_2}]\left.\begin{bmatrix}n/2-1\\ m/2-1\end{bmatrix}\right|_{q\rightarrow q^2}=[q^{L_2/2}]{n/2-1\brackm/2-1}.$$
 By $d=2$, we have $m,n-m\in2\mathbb Z$, which implies $L_2\in 2\mathbb Z_{\ge0}.$ Moreover, we also have
$ 0\le\frac{L_2}{2} \le\deg {n/2-1\brackm/2-1}.$
In consequence, we obtain
\[
R_{L_2}=[q^{L_2/2}]{n/2-1\brackm/2-1}\ge1
\] since the coefficients of Gaussian polynomials are positive integers.


If $n-m\le6$, then $4\le m\le n-m\le6$ and both $m$ and $n-m$
are even. Thus $(n,m)\in\{(8,4),(10,4),(12,6)\}$, and direct
coefficient extraction gives
\[
 \begin{array}{c|c|c|c}
 (n,m)&R(q)&L_2&r_{L_2}\\ \hline
  (8,4)&[3]_{q^2}&0&1\\
  (10,4)&[4]_{q^2}&2&1\\
  (12,6)&\genfrac{[}{]}{0pt}{}{5}{2}_{q^{2}}+[3]_{q^3}-1&6&3
 \end{array}
\]
In consequence, we have $R_{L_2}\ge1$ and
\eqref{eq:midpoint-compensation} is nonnegative. The proof is complete.

<!-- lem-6: body_md -->
\label{lem:coprime}
Let $n$ and $m$ be integers with $0<m<n$ and $\gcd(n,m)=1$.
Then $G_{n,m}(q)\ge_q0$.
<!-- end lem-6 -->
<!-- proof-16: body_md -->
For $m=1$, by (\ref{def+Fnm+Gnm}), we have
$G_{n,1}(q)=1/(1-q^{n+1})$, whose coefficients are obviously nonnegative.
Suppose $m>1$. By
$P(q)={n-1\brackm-1}$, we define a rational function
\begin{eqnarray}
 B(q)=\frac{P(q)}{[m]_{q}}.
\end{eqnarray}
It follows from (\ref{def+Fnm+Gnm}) that
\begin{eqnarray}
  G_{n,m}(q)=\frac{B(q)}{1-q^{n+1}}.
\end{eqnarray}

So, in order to prove that $G_{n,m}(q)\ge_q0$, it suffices to show $B(q)\ge_q0$. In what follows we will prove that $B(q)$ is a polynomial with only nonnegative coefficients.

If $\zeta^m=1$ and $\zeta\ne1$, then the condition $\gcd(n,m)=1$ implies that the order of $\zeta$ divides
$m$ but not $n$. By Lemma~\ref{lem:root-evaluation},
$P(\zeta)=0$. These roots are precisely simple of $[m]_{q}$.
Thus we obtain $[m]_{q}\mid P(q)$ and $B(q)\in\mathbb Z[q]$.


Easy computations give $B(0)=1$,
\begin{eqnarray}
 \deg B(q)=(m-1)(n-m-1)\ge0,
 \qquad q^{\deg B(q)}B(q^{-1})=B(q).
\end{eqnarray}
Thus, in order to prove that all coefficients of $B(q)$ are nonnegative, we only need to show that $[q^k]B(q)$ is nonnegative for $0\le k\le\deg B(q)/2$.
Since $[m]_{q}=(1-q^m)/(1-q)$, we have
\begin{eqnarray}\label{coeff+B}
[q^k]B(q)=[q^k]\frac{(1-q)P(q)}{1-q^m}
 =\sum_{j\ge0}[q^{k-jm}](1-q)P(q)
\end{eqnarray}
for $k\ge0$.
For $0\le k\le\deg B(q)/2$, each nonnegative index in this sum satisfies
\begin{eqnarray}
 0\le k-jm\le k\le\frac{\deg B(q)}{2}
 \le\frac{(m-1)(n-m)}{2}=\frac{\deg P(q)}{2}.
\end{eqnarray}

Hence, applying Lemma~\ref{lem:kernel} with $a=n-m$ and $b=m-1$ to (\ref{coeff+B}) yields nonnegativity of each term. So $[q^k]B(q)$ is nonnegative for $0\le k\le\deg B(q)/2$. We complete the proof.
