<!-- theorem-1: body_md -->
\label{thm1} The set $A$ is a strongly minimal asymptotic basis of order $h$ if
and only if the following two conditions hold:

\begin{enumerate}
\item[(I)] For every $i\in\{1,\ldots,h\}$,
\[
 W_i(m)=\frac{m}{h}+O(1).
\]

\item[(II)] For every $i\in\{1,\ldots,h\}$, every $w\in W_i$, and every
$(k_1,\ldots,k_h)\in\mathcal K_h\setminus\{(1,\ldots,1)\}$, there exists
an integer $t\geq1$ such that
\begin{equation}\label{eq3}
 \sum_{j=1}^h k_jP_j(t)
 <
 \begin{cases}
 \displaystyle\sum_{j\neq i}P_j(t),&t\leq w,\\[2mm]
 \displaystyle\sum_{j\neq i}P_j(t)+2^w,&t>w.
 \end{cases}
\end{equation}
\end{enumerate}

<!-- lemma-3: body_md -->
\label{lema3} Let
\[
 n=b_1+\cdots+b_q,
 \qquad b_\ell\in B_{j_\ell}\quad(1\leq\ell\leq q).

If $n'$ is a nonnegative integer satisfying

 {S}(n')\subseteq{S}(n),

<!-- proof-1: body_md -->
If $n=0$, then $b_1=\cdots=b_q=n'=0$, and the choice
$b'_\ell=0$ proves the assertion. Hence assume $n>0$.
Write
\[
 S(n)=\{y_1,\ldots,y_s\},
 \qquad y_1<\cdots<y_s,
 \qquad n=\sum_{r=1}^{s}2^{y_r}.
\]
For every $1\leq\ell\leq q$, write
\(b_\ell=\sum_{x\in{S}(b_\ell)}2^x\).
Let
\[
 T=\sum_{\ell=1}^{q}\left|{S}(b_\ell)\right|.
\]
List all the binary terms above, with multiplicities, as
\[
 2^{x_1},\ldots,2^{x_T}.
\]
Then
\[
 \sum_{\rho=1}^{T}2^{x_\rho}
 =\sum_{\ell=1}^{q}b_\ell
 =n
 =\sum_{r=1}^{s}2^{y_r}.
\]
Let $\lambda(\rho)$ denote the index of the summand containing
$2^{x_\rho}$, for $1\leq\rho\leq T$. Then, for
$1\leq\ell\leq q$,
\[
 \{x_\rho:\lambda(\rho)=\ell\}={S}(b_\ell).
\]

By Lemma~\ref{lema2}, there exists a partition
\[
 \{1,\ldots,T\}=J_1\sqcup\cdots\sqcup J_s
\]
such that
\[
 2^{y_r}=\sum_{\rho\in J_r}2^{x_\rho}
 \qquad(1\leq r\leq s).
\]
By
\[
 {S}(n')\subseteq{S}(n)=\{y_1,\ldots,y_s\},
\]
let
\[
 R'=\{r\in\{1,\ldots,s\}:y_r\in{S}(n')\}.
\]
Then
\[
 n'
 =\sum_{r\in R'}2^{y_r}
 =\sum_{r\in R'}\sum_{\rho\in J_r}2^{x_\rho}.
\]
For $1\leq\ell\leq q$, define
\[
 b'_\ell
 =\sum_{r\in R'}
   \sum_{\substack{\rho\in J_r\\\lambda(\rho)=\ell}}2^{x_\rho}.
\]
For a fixed $\ell$, only some of the binary terms of $b_\ell$ are retained. Hence
\[
 {S}(b'_\ell)
 \subseteq
 \{x_\rho:\lambda(\rho)=\ell\}
 ={S}(b_\ell).
\]
Since $b_\ell\in B_{j_\ell}$ and
$S(b'_\ell)\subseteq S(b_\ell)$, we have
\[
 S(b'_\ell)\subseteq S(b_\ell)\subseteq W_{j_\ell},
 \qquad b'_\ell\in B_{j_\ell}
 \quad(1\leq\ell\leq q).
\]
Finally, since $J_1,\ldots,J_s$ are pairwise disjoint,
\begin{align*}
 \sum_{\ell=1}^{q}b'_\ell
 &=\sum_{\ell=1}^{q}
   \sum_{r\in R'}
   \sum_{\substack{\rho\in J_r\\\lambda(\rho)=\ell}}2^{x_\rho}\\
 &=\sum_{r\in R'}\sum_{\rho\in J_r}2^{x_\rho}\\
 &=\sum_{r\in R'}2^{y_r}\\
 &=n'.
\end{align*}
This completes the proof of the lemma.

<!-- lemma-4: body_md -->
\label{lema4} Let $M\geq1$, let $q_0,\ldots,q_{M-1}$ be nonnegative integers, and let $n$ be an integer with $0\leq n<2^M$. Then
\[
 n=\sum_{t=0}^{M-1}d_t2^t,
 \qquad
 0\leq d_t\leq q_t,
\]
holds for some integers $d_t$ if and only if
\begin{equation}\label{eq6}
 n\bmod2^r
 \leq
 \sum_{t=0}^{r-1}q_t2^t
 \qquad(1\leq r\leq M).
\end{equation}
<!-- end lemma-4 -->
<!-- proof-2: body_md -->
Suppose first that there exist integers $d_0,\ldots,d_{M-1}$ satisfying
\[
 n=\sum_{t=0}^{M-1}d_t2^t,
 \qquad
 0\leq d_t\leq q_t.
\]
For $1\leq r\leq M$,
\begin{align*}
 n&=\sum_{t=0}^{r-1}d_t2^t
 +2^r\sum_{t=r}^{M-1}d_t2^{t-r},\\
 n\bmod2^r&=\left(\sum_{t=0}^{r-1}d_t2^t\right)\bmod2^r,\\
 0\leq n\bmod2^r
 &\leq\sum_{t=0}^{r-1}d_t2^t
 \leq\sum_{t=0}^{r-1}q_t2^t.
\end{align*}
This proves \eqref{eq6}.

Conversely, we prove by induction on $M$ that the conditions
\begin{equation}\label{eq7}
 \left\{
 \begin{aligned}
 0&\leq n\leq\sum_{t=0}^{M-1}q_t2^t,\\
 n\bmod2^r&\leq\sum_{t=0}^{r-1}q_t2^t
 \qquad(1\leq r<M),
 \end{aligned}
 \right.
\end{equation}
imply the existence of integers $d_0,\ldots,d_{M-1}$ with
\[
 n=\sum_{t=0}^{M-1}d_t2^t,
 \qquad
 0\leq d_t\leq q_t.
\]
For $M=1$, take $d_0=n\in[0,q_0]$.

Now let $M\geq2$ and assume the assertion for $M-1$. Write
\(n=Q2^{M-1}+R\),
where
\[
 Q=\left\lfloor\frac{n}{2^{M-1}}\right\rfloor,
 \qquad
 0\leq R<2^{M-1}.
\]
Define
\[
 d_{M-1}=\min\{q_{M-1},Q\},
 \qquad
 n'=n-d_{M-1}2^{M-1}.
\]
Clearly,
\[
 0\leq d_{M-1}\leq q_{M-1}.
\]

If $Q\leq q_{M-1}$, then
\begin{align*}
 d_{M-1}&=Q,\\
 n'&=R=n\bmod2^{M-1},\\
 0\leq n'&\leq\sum_{t=0}^{M-2}q_t2^t
 \qquad\text{by \eqref{eq7}}.
\end{align*}
If $Q>q_{M-1}$, then
\begin{align*}
 d_{M-1}&=q_{M-1},\\
 n'&=(Q-q_{M-1})2^{M-1}+R>0,\\
 n'&=n-q_{M-1}2^{M-1}
 \leq\sum_{t=0}^{M-2}q_t2^t
 \qquad\text{by \eqref{eq7}}.
\end{align*}
In either case,
\begin{equation}\label{eq8}
 0\leq n'\leq\sum_{t=0}^{M-2}q_t2^t.
\end{equation}

Since
\(n-n'=d_{M-1}2^{M-1}\),
for every $1\leq r<M-1$, we have
\(n'\equiv n\pmod{2^r}\).
Therefore,
\[
 n'\bmod2^r
 =n\bmod2^r
 \leq
 \sum_{t=0}^{r-1}q_t2^t
 \qquad(1\leq r<M-1).
\]
By \eqref{eq8} and the induction hypothesis, there exist integers
$d_0,\ldots,d_{M-2}$ such that
\[
 n'=\sum_{t=0}^{M-2}d_t2^t,
 \qquad
 0\leq d_t\leq q_t
 \quad(0\leq t<M-1).
\]
Consequently,
\begin{align*}
 n
 &=n'+d_{M-1}2^{M-1}\\
 &=\sum_{t=0}^{M-2}d_t2^t+d_{M-1}2^{M-1}\\
 &=\sum_{t=0}^{M-1}d_t2^t,
\end{align*}
and
\[
 0\leq d_t\leq q_t
 \qquad(0\leq t<M).
\]
Thus the stronger assertion holds for every $M$. For $0\leq n<2^M$, \eqref{eq6} implies
\[
 0\leq n=n\bmod2^M\leq\sum_{t=0}^{M-1}q_t2^t,
 \qquad
 n\bmod2^r\leq\sum_{t=0}^{r-1}q_t2^t\quad(1\leq r<M).
\]
Applying \eqref{eq7}, we obtain
\[
 n=\sum_{t=0}^{M-1}d_t2^t,\qquad
 0\leq d_t\leq q_t\quad(0\leq t<M).
\]

<!-- lemma-5: body_md -->
\label{lema5} Fix $i\in\{1,\ldots,h\}$ and $w\in W_i$. The following two assertions are equivalent.

\begin{enumerate}
\item[(a)] For every
$(k_1,\ldots,k_h)\in\mathcal K_h\setminus\{(1,\ldots,1)\}$, there exists an integer $t\geq1$ such that
\begin{equation}\label{eq9}
 \sum_{j=1}^h k_jP_j(t)
 <
 \begin{cases}
 \displaystyle\sum_{j\neq i}P_j(t),&t\leq w,\\[2mm]
 \displaystyle\sum_{j\neq i}P_j(t)+2^w,&t>w.
 \end{cases}
\end{equation}

\item[(b)] There exists an integer $M>w$ such that, for every
$(k_1,\ldots,k_h)\in\mathcal K_h\setminus\{(1,\ldots,1)\}$,
\begin{equation}\label{eq10}
 2^w+\sum_{\substack{0\leq t<M\\t\notin W_i}}2^t
 \notin
 \sum_{j=1}^h k_jB_j.
\end{equation}
\end{enumerate}
<!-- end lemma-5 -->
<!-- proof-3: body_md -->
Fix an integer
\[
 M>w
\]
and a vector
\[
 (k_1,\ldots,k_h)\in\mathcal K_h.
\]
For each $0\leq t<M$, since $W_1,\ldots,W_h$ partition $\mathbb N_0$, there is a unique index
\[
 c(t)\in\{1,\ldots,h\}
\]
with
\(t\in W_{c(t)}\).
Let
\(q_t=k_{c(t)} \qquad(0\leq t<M)\).

Let $0\leq n<2^M$. We first establish the equivalence
\[
 n\in\sum_{j=1}^{h}k_jB_j
 \quad\text{if and only if}\quad
 n=\sum_{t=0}^{M-1}d_t2^t
 \quad\text{for some integers }0\leq d_t\leq q_t.
\]
Indeed, if
\(n\in\sum_{j=1}^{h}k_jB_j\),
then we may write
\[
 n=\sum_{j=1}^{h}\sum_{u=1}^{k_j}b_{j,u},
 \qquad
 b_{j,u}\in B_j.
\]
Since all summands are nonnegative,
\[
 0\leq b_{j,u}\leq n<2^M,
 \qquad S(b_{j,u})\subseteq[0,M)\cap\mathbb Z.
\]
For $0\leq t<M$, define
\[
 d_t
 =\left|\{u:1\leq u\leq k_{c(t)},\ t\in{S}(b_{c(t),u})\}\right|.
\]
Since $t$ belongs to no set other than $W_{c(t)}$,
\[
 0\leq d_t\leq k_{c(t)}=q_t.
\]
Moreover,
\begin{align*}
 n
 &=\sum_{j=1}^{h}\sum_{u=1}^{k_j}
   \sum_{t\in{S}(b_{j,u})}2^t\\
 &=\sum_{t=0}^{M-1}d_t2^t.
\end{align*}

Conversely, suppose that there exist integers $d_t$ satisfying
\[
 n=\sum_{t=0}^{M-1}d_t2^t,
 \qquad
 0\leq d_t\leq k_{c(t)}.
\]
For every $0\leq t<M$, choose a subset $U_t$ of
\[
 \{1,\ldots,k_{c(t)}\}
\]
with $d_t$ elements. For
$1\leq j\leq h$ and $1\leq u\leq k_j$, define
\[
 b_{j,u}
 =\sum_{\substack{0\leq t<M\\t\in W_j\\u\in U_t}}2^t.
\]
By construction,
\[
 S(b_{j,u})\subseteq W_j,
 \qquad b_{j,u}\in B_j.
\]
Moreover, $|U_t|=d_t$, so
\begin{align*}
 \sum_{j=1}^{h}\sum_{u=1}^{k_j}b_{j,u}
 &=\sum_{t=0}^{M-1}d_t2^t\\
 &=n.
\end{align*}
Thus
\[
 n\in\sum_{j=1}^{h}k_jB_j.
\]

By Lemma~\ref{lema4}, this representation exists if and only if
\[
 n\bmod2^r
 \leq
 \sum_{t=0}^{r-1}q_t2^t
 \qquad(1\leq r\leq M).
\]
Since $q_t=k_{c(t)}$ and $t\in W_{c(t)}$, we have
\begin{align*}
 \sum_{t=0}^{r-1}q_t2^t
 &=\sum_{t=0}^{r-1}k_{c(t)}2^t\\
 &=\sum_{j=1}^{h}k_j
   \sum_{t\in W_j\cap[0,r)}2^t\\
 &=\sum_{j=1}^{h}k_jP_j(r).
\end{align*}
Therefore,
\begin{equation}\label{eq11}
\begin{gathered}
n\in\sum_{j=1}^{h}k_jB_j
\quad\text{if and only if}\\[2pt]
n\bmod 2^r\leq\sum_{j=1}^{h}k_jP_j(r)
\qquad(1\leq r\leq M).
\end{gathered}
\end{equation}

Now take
\[
 n_M
 =2^w+\sum_{\substack{0\leq t<M\\t\notin W_i}}2^t.
\]
Since $w\in W_i$ and $w<M$, the binary positions in the expression above are distinct and all are less than $M$. Hence
\[
 0<n_M\leq\sum_{t=0}^{M-1}2^t=2^M-1<2^M.
\]
Fix $1\leq r\leq M$.

If $r\leq w$, then $2^w$ is divisible by $2^r$, and hence
\begin{align*}
 n_M\bmod2^r
 &=\sum_{\substack{0\leq t<r\\t\notin W_i}}2^t\\
 &=\sum_{j\neq i}\sum_{t\in W_j\cap[0,r)}2^t\\
 &=\sum_{j\neq i}P_j(r).
\end{align*}
If $r>w$, then the position $w$ remains among the lowest $r$ binary positions, and hence
\begin{align*}
 n_M\bmod2^r
 &=2^w+\sum_{\substack{0\leq t<r\\t\notin W_i}}2^t\\
 &=2^w+\sum_{j\neq i}P_j(r).
\end{align*}
Thus
\begin{equation}\label{eq12}
 n_M\bmod2^r
 =
 \begin{cases}
 \displaystyle\sum_{j\neq i}P_j(r),&r\leq w,\\[2mm]
 \displaystyle\sum_{j\neq i}P_j(r)+2^w,&r>w.
 \end{cases}
\end{equation}

By \eqref{eq11} and \eqref{eq12},
$n_M\notin\sum_{j=1}^{h}k_jB_j$ holds precisely when there is
an integer $1\leq r\leq M$ such that
\begin{equation}\label{eq13}
 \sum_{j=1}^{h}k_jP_j(r)
 <
 \begin{cases}
 \displaystyle\sum_{j\neq i}P_j(r),&r\leq w,\\[2mm]
 \displaystyle\sum_{j\neq i}P_j(r)+2^w,&r>w.
 \end{cases}
\end{equation}

Suppose that (a) holds. Since
\[
 \mathcal K_h

$E_{a_i}\cap[0,2^m)$ according to whether $S(n)\cap W_i=S(a_i)$.
In the first class,
\[
\begin{gathered}
\varepsilon_t(n)=
\begin{cases}
1,&t\in S(a_i),\\
0,&t\in(W_i\cap[0,m))\setminus S(a_i),
\end{cases}\\[2pt]
\#\bigl(([0,m)\cap\mathbb Z)\setminus W_i\bigr)=m-W_i(m).
\end{gathered}
\]
Therefore
\begin{align*}
 \#\{0\leq n<2^m:S(n)\cap W_i=S(a_i)\}
 &=2^{m-W_i(m)},\\
 \#\{0\leq n<2^m:n\in E_{a_i},\ S(n)\cap W_i\neq S(a_i)\}
 &=o(2^{(h-1)K(m)})
\end{align*}
by Lemma~\ref{lema6}. Hence
\begin{equation}\label{eq39}
 E_{a_i}(2^m-1)
 \leq
 2^{m-W_i(m)}
 +o\bigl(2^{(h-1)K(m)}\bigr).
\end{equation}

Combining \eqref{eq38} and \eqref{eq39}, for each fixed $i$ we have
\[
 c'_i2^{(h-1)K(m)}
 \leq
 2^{m-W_i(m)}
 +o\bigl(2^{(h-1)K(m)}\bigr).
\]
For sufficiently large $m$,
\[
 o\bigl(2^{(h-1)K(m)}\bigr)
 \leq
 \frac{c'_i}{2}2^{(h-1)K(m)}.
\]
It follows that
\[
 2^{m-W_i(m)}
 \geq
 \frac{c'_i}{2}2^{(h-1)K(m)}.
\]
Since there are only finitely many indices $i$, we may choose a positive constant $c$, depending only on the $c'_i$, such that for every $i$ and all sufficiently large $m$,
\begin{equation}\label{eq40}
 2^{m-W_i(m)}
 \geq
 c\,2^{(h-1)K(m)}.
\end{equation}

Choose $i(m)$ with $W_{i(m)}(m)=K(m)$. Since the constant
$c$ in \eqref{eq40} is independent of $i$,
\begin{align*}
 2^{m-K(m)}&\geq c\,2^{(h-1)K(m)},
 \end{align*}
and so
\[ hK(m)\leq m-\log_2c.\]

Consequently,
\begin{equation}\label{eq41}
 K(m)\leq\frac mh+C_1
\end{equation}
for some constant $C_1>0$ and all sufficiently large $m$.

Since $W_i(m)\leq K(m)$,
\[
 W_i(m)\leq\frac mh+C_1
 \qquad(1\leq i\leq h).
\]
By \eqref{eq2},
\[
 \sum_{j=1}^{h}W_j(m)=m.
\]
Fix $i$. Then
\begin{align*}
 W_i(m)
 &=m-\sum_{j\neq i}W_j(m)\\
 &\geq
 m-\sum_{j\neq i}\left(\frac mh+C_1\right)\\
 &=m-(h-1)\frac mh-(h-1)C_1\\
 &=\frac mh-(h-1)C_1.
\end{align*}
Therefore, for all sufficiently large $m$,
\[
 \frac mh-(h-1)C_1
 \leq
 W_i(m)
 \leq
 \frac mh+C_1.
\]
Hence there exists a constant $C_2>0$ such that
\[
 \left|W_i(m)-\frac mh\right|\leq C_2
\]
for all $1\leq i\leq h$ and all sufficiently large $m$. Increasing the constant to cover the finitely many smaller values of $m$, we obtain $C>0$ such that
\[
 \left|W_i(m)-\frac mh\right|\leq C
 \qquad(1\leq i\leq h,\ m\geq0).
\]
This completes the proof of the lemma.

\label{lema9} Suppose that, for some constant $C>0$,

For every interval $I=[s,s+L)\cap\mathbb Z$, where $s\geq0$ and $L\geq1$ are integers, put $m_j(I)=\left|W_j\cap I\right|$. By \eqref{eq42},
\begin{align}
 m_j(I)
 &=W_j(s+L)-W_j(s)
 \nonumber\\
 &=\frac Lh+e_j,
 \qquad
 |e_j|\leq2C.
 \label{eq43}
\end{align}
Let $D=2C$ and choose the integer
\begin{equation}\label{eq44}
 L=
 \left\lfloor
 \max\left\{
 h\bigl((2h+1)D+2(h-1)\bigr),
 h\bigl(2hD+h^2-1\bigr),
 \frac{hD}{h-1}+h(h+1)
 \right\}
 \right\rfloor+1.
\end{equation}
By this choice,
\begin{equation}\label{eq45}
 \frac Lh-D>2hD+2(h-1),
\end{equation}
\begin{equation}\label{eq46}
 \frac Lh-2hD-(h-1)>h(h-1),
\end{equation}
and
\begin{equation}\label{eq47}
 \frac{h-1}{h}L-D-(h-1)>h(h-1).
\end{equation}

Let
\[
 I=[s,s+L)\cap\mathbb Z,
 \qquad
 s>\max S(a),
\]
and suppose that
\[
 n\in\sum_{j=1}^hk_jB_j,
 \qquad
 (k_1,\ldots,k_h)\in\mathcal K_h\setminus\{(1,\ldots,1)\}.
\]
Since
\(\sum_{j=1}^hk_j=h\),
we may choose
\[
 b_{j,r}\in B_j
 \qquad
 (1\leq j\leq h,\ 1\leq r\leq k_j)
\]
such that
\begin{equation}\label{eq48}
 n=\sum_{j=1}^h\sum_{r=1}^{k_j}b_{j,r}.
\end{equation}
The summands $b_{j,r}$ may be $0$. Let
\[
 \mathcal P=
 \{(j,r):1\leq j\leq h,\ 1\leq r\leq k_j\}.
\]
Then
\begin{equation}\label{eq49}
 \left|\mathcal P\right|=\sum_{j=1}^hk_j=h.
\end{equation}

For $(j,r)\in\mathcal P$, since
\(b_{j,r}\in B_j=A(W_j)\cup\{0\}\)
we may write
\[
 b_{j,r}=\sum_{u\geq0}\varepsilon_u(b_{j,r})2^u,
\]
where
\begin{equation}\label{eq50}
 \varepsilon_u(b_{j,r})\in\{0,1\},
 \qquad
 \varepsilon_u(b_{j,r})=0\quad(u\notin W_j).
\end{equation}
Since
\(\mathbb N_0=W_1\sqcup\cdots\sqcup W_h\),
for every $t\geq0$ there exists a unique
\[
 c(t)\in\{1,\ldots,h\}

such that
\begin{equation}\label{eq51}
 t\in W_{c(t)}.
\end{equation}
Let
\begin{align}
 d_t
 &:=\sum_{j=1}^h\sum_{r=1}^{k_j}\varepsilon_t(b_{j,r})
 \nonumber\\
 &=\sum_{r=1}^{k_{c(t)}}\varepsilon_t(b_{c(t),r}).
 \label{eq52}
\end{align}
Then
\begin{align}
 0\leq d_t
 &\leq\sum_{r=1}^{k_{c(t)}}1
 =k_{c(t)}
 \leq\sum_{j=1}^hk_j
 =h.
 \label{eq53}
\end{align}
By \eqref{eq48},
\begin{align}
 n
 &=\sum_{j=1}^h\sum_{r=1}^{k_j}
   \sum_{u\geq0}\varepsilon_u(b_{j,r})2^u
 \nonumber\\
 &=\sum_{u\geq0}
   \left(\sum_{j=1}^h\sum_{r=1}^{k_j}\varepsilon_u(b_{j,r})\right)2^u
 \nonumber\\
 &=\sum_{u\geq0}d_u2^u.
 \label{eq54}
\end{align}
On the other hand, with the notation above, the standard binary expansion of $n$ is
\begin{equation}\label{eq55}
 n=\sum_{u\geq0}\varepsilon_u(n)2^u,
 \qquad
 \varepsilon_u(n)\in\{0,1\}.
\end{equation}
By the hypothesis of the lemma, for $t\in I$,
\begin{equation}\label{eq56}
 \varepsilon_t(n)=
 \begin{cases}
 0,&t\in W_i\cap I,\\
 1,&t\in I\setminus W_i.
 \end{cases}
\end{equation}

\[
 S_t=\sum_{u=0}^{t}d_u2^u,
 \qquad
 T_t=\sum_{u=0}^{t}\varepsilon_u(n)2^u.
\]
Both expansions represent $n$, so
\begin{equation}\label{eq57}
 S_t\equiv n\pmod{2^{t+1}},
\end{equation}
\begin{equation}\label{eq58}
 T_t\equiv n\pmod{2^{t+1}}.
\end{equation}

\[
 S_t\geq0,
 \qquad
 0\leq T_t\leq2^{t+1}-1,
 \qquad
 S_t\equiv T_t\pmod{2^{t+1}},

\begin{equation}\label{eq59}
 \sum_{u=0}^{t}d_u2^u
 =
 \sum_{u=0}^{t}\varepsilon_u(n)2^u
 +2^{t+1}c_{t+1}.
\end{equation}
For $t\geq1$, substituting $t-1$ into
\eqref{eq59} gives
\begin{equation}\label{eq60}
 \sum_{u=0}^{t-1}d_u2^u
 =
 \sum_{u=0}^{t-1}\varepsilon_u(n)2^u
 +2^tc_t.
\end{equation}

\[
 d_t2^t
 =\varepsilon_t(n)2^t+2^{t+1}c_{t+1}-2^tc_t,

\begin{equation}\label{eq61}
 d_t+c_t=\varepsilon_t(n)+2c_{t+1}
 \qquad(t\geq1).
\end{equation}
For $t=0$, the same identity follows from
\eqref{eq59} and $c_0=0$. Thus
\begin{equation}\label{eq62}
 d_t+c_t=\varepsilon_t(n)+2c_{t+1}
 \qquad(t\geq0).
\end{equation}

We next prove that the carries are uniformly bounded. We already have
\(c_0=0\leq h-1\).
If
\(0\leq c_t\leq h-1\),

\begin{align*}
 0\leq2c_{t+1}
 &=d_t+c_t-\varepsilon_t(n)\\
 &\leq d_t+c_t\\

\end{align*}
Since $c_{t+1}$ is an integer,
\[

 \leq\left\lfloor\frac{2h-1}{2}\right\rfloor
 =h-1.

\begin{equation}\label{eq63}
\begin{gathered}

0\leq d_t\leq k_{c(t)},\qquad
0\leq c_t\leq h-1
\qquad(t\geq0).
\end{gathered}
\end{equation}

\begin{align*}

\end{equation}

Using

we obtain
\begin{align*}
 \sum_{j\neq i}(1-k_j)m_j(I)
 &=\frac Lh(k_i-1)
 +\sum_{j\neq i}(1-k_j)e_j\\
 &\geq\frac Lh(k_i-1)
 -D\sum_{j\neq i}|1-k_j|.
\end{align*}
Also,
\begin{align*}
 \sum_{j\neq i}|1-k_j|
 &\leq\sum_{j\neq i}(1+k_j)\\
 &=(h-1)+(h-k_i)\\
 &\leq2h.
\end{align*}
Therefore,
\begin{align}
 \sum_{t\in W_i\cap I}d_t
 &\geq
 (k_i-1)\frac Lh-2hD-(h-1)
 \nonumber\\
 &\geq
 \frac Lh-2hD-(h-1)
 \nonumber\\
 &>h(h-1),
 \label{eq81}
\end{align}
where the last step follows from \eqref{eq46}.

For $1\leq r\leq k_i$, let
\(q_r=\left|{S}(b_{i,r})\cap I\right|\).
By \eqref{eq52},
\begin{align*}
 \sum_{r=1}^{k_i}q_r
 &=\sum_{r=1}^{k_i}\sum_{t\in W_i\cap I}\varepsilon_t(b_{i,r})\\
 &=\sum_{t\in W_i\cap I}\sum_{r=1}^{k_i}\varepsilon_t(b_{i,r})\\
 &=\sum_{t\in W_i\cap I}d_t\\
 &>h(h-1).
\end{align*}
Since $k_i\leq h$, the inequality above gives
$\sum_{r=1}^{k_i}q_r>k_i(h-1)$. Thus there exists
$r_*$ such that
\begin{equation}\label{eq82}
 b:=b_{i,r_*}\in A_i,
 \qquad
 \left|{S}(b)\cap I\right|\geq h.
\end{equation}
Let
\(p_*=(i,r_*)\).
By \eqref{eq48}, \eqref{eq49},
\eqref{eq82}, and Lemma~\ref{lema8} (with
$\ell=i$), we obtain
in the case $k_i\geq2$,
\begin{equation}\label{eq83}
 n\in h(A\setminus\{a\}).
\end{equation}

\medskip
\noindent
\textbf{Case 3. $k_i=0$.} By \eqref{eq72} and \eqref{eq73},
\begin{align*}
 \sum_{t\in I}d_t
 &\geq\sum_{j\neq i}m_j(I)-(h-1)\\
 &=L-m_i(I)-(h-1).
\end{align*}
By \eqref{eq43},
\[
 m_i(I)=\frac Lh+e_i,
 \qquad |e_i|\leq D,
\]
and hence
\begin{align}
 \sum_{t\in I}d_t
 &\geq L-\left(\frac Lh+D\right)-(h-1)
 \nonumber\\
 &=\frac{h-1}{h}L-D-(h-1)
 \nonumber\\
 &>h(h-1),
 \label{eq84}
\end{align}
where the last step follows from \eqref{eq47}.

For every $p=(j,r)\in\mathcal P$, let
\(q_p=\left|{S}(b_p)\cap I\right|\).
By \eqref{eq52},
\begin{align*}
 \sum_{p\in\mathcal P}q_p
 &=\sum_{j=1}^h\sum_{r=1}^{k_j}
   \sum_{t\in I}\varepsilon_t(b_{j,r})\\
 &=\sum_{t\in I}
   \sum_{j=1}^h\sum_{r=1}^{k_j}\varepsilon_t(b_{j,r})\\
 &=\sum_{t\in I}d_t\\
 &>h(h-1).
\end{align*}
Since $|\mathcal P|=h$, the inequality above gives
$\sum_{p\in\mathcal P}q_p>|\mathcal P|(h-1)$. Thus
there exists $p_*=(j_*,r_*)\in\mathcal P$ such that
\begin{equation}\label{eq85}
 b:=b_{p_*}\in A_{j_*},
 \qquad
 \left|{S}(b)\cap I\right|\geq h.
\end{equation}
Since $k_i=0$,
\begin{equation}\label{eq86}
 j_*\neq i.
\end{equation}
For all $p=(j,r)\in\mathcal P$, we have $j\neq i$, and hence
\[
 b_p\in B_j=A_j\cup\{0\},\qquad
 (A_j\cup\{0\})\cap A_i=\varnothing,
 \qquad a\in A_i.
\]
Thus
\begin{equation}\label{eq87}
 \{p\in\mathcal P:b_p=a\}=\varnothing.
\end{equation}
By \eqref{eq48}, \eqref{eq49},
\eqref{eq85}--\eqref{eq87}, and
Lemma~\ref{lema8} (with $\ell=j_*$), in the case $k_i=0$ we obtain
\begin{equation}\label{eq88}
 n\in h(A\setminus\{a\}).
\end{equation}

Since $k_i\in\{0,1,\ldots,h\}$, the case $k_i=1$ is excluded by
\eqref{eq80}. By
\eqref{eq83} and \eqref{eq88}, in each of the other two cases we have
\(n\in h(A\setminus\{a\})\).
This completes the proof of the lemma.

Assume that $A$ is a strongly minimal asymptotic basis of order $h$. By Lemma~\ref{lema7}, fix a constant $C>0$ such that
\[
 \left|W_j(m)-\frac mh\right|\leq C
 \qquad(1\leq j\leq h,\ m\geq0).
\]
In particular,
\begin{equation}\label{eq89}
 W_i(m)=\frac mh+O(1)
 \qquad(1\leq i\leq h).
\end{equation}
In particular,
\[
 K(m)=\max_{1\leq j\leq h}W_j(m)
 =\frac mh+O(1).
\]

Suppose that \eqref{eq3} fails. Then there exist
\[
 i\in\{1,\ldots,h\},
 \qquad
 w\in W_i,
\]
and
\[
 (k_1,\ldots,k_h)
 \in\mathcal K_h\setminus\{(1,\ldots,1)\}
\]
such that, for every $t\geq1$,
\begin{equation}\label{eq90}
 \sum_{j=1}^hk_jP_j(t)
 \geq
 \begin{cases}
 \displaystyle\sum_{j\neq i}P_j(t),&t\leq w,\\[2mm]
 \displaystyle\sum_{j\neq i}P_j(t)+2^w,&t>w.
 \end{cases}
\end{equation}
Let
\[
 a=2^w.
\]
Then
\(a\in A_i, \qquad {S}(a)=\{w\}\).
For any $M>w$, let
\[
 N_M=2^w+\sum_{\substack{0\leq t<M\\t\notin W_i}}2^t.
\]
By \eqref{eq90} and \eqref{eq12}, for every
$M>w$ and $1\leq r\leq M$ we have
\[
 N_M\bmod2^r\leq\sum_{j=1}^hk_jP_j(r).
\]
Since $0<N_M<2^M$, the representation criterion
\eqref{eq11} gives
\begin{equation}\label{eq91}
 N_M\in\sum_{j=1}^hk_jB_j.
\end{equation}

Fix $m>w$ and let
\[
 \mathcal H_m
 =\{0\leq n<2^m:{S}(n)\cap W_i=\{w\}\}.
\]
Let
\[
 \Omega_m=([0,m)\cap\mathbb Z)\setminus W_i.
\]
Then
\(\left|\Omega_m\right|=m-W_i(m)\).
For every $S\subseteq\Omega_m$, let
\(n_S=2^w+\sum_{t\in S}2^t\).
As $S(n_S)=\{w\}\sqcup S$, distinct subsets of $\Omega_m$
give distinct integers $n_S$. Moreover,
$\mathcal H_m=\{n_S:S\subseteq\Omega_m\}$.
Hence
\begin{equation}\label{eq92}
 |\mathcal H_m|=2^{|\Omega_m|}=2^{m-W_i(m)}.
\end{equation}

For every $n\in\mathcal H_m$,
\begin{align*}
 {S}(n)
 &\subseteq
 \{w\}\sqcup\Omega_m\\
 &={S}(N_m).
\end{align*}
Applying \eqref{eq91} with $M=m$ and then Lemma~\ref{lema3}, we obtain
\begin{equation}\label{eq93}
 n\in\sum_{j=1}^hk_jB_j
 \qquad(n\in\mathcal H_m).
\end{equation}

Let $L=L(h,C)$ be the integer in Lemma~\ref{lema9}. Let
\[
 q=\left\lfloor\frac{m-w-1}{L}\right\rfloor.
\]
For sufficiently large $m$, we have $q\geq1$. Define
\[
 I_r=[w+1+(r-1)L,w+1+rL)\cap\mathbb Z
 \qquad(1\leq r\leq q).
\]
Then
\begin{align*}
 \left|I_r\right|&=L,\\
 I_r\cap I_s&=\varnothing\qquad(r\neq s),\\
 \min I_r&>w=\max S(a),\\
 I_r&\subseteq[0,m).
\end{align*}
For $1\leq r\leq q$, let
\[
 J_r=I_r\setminus W_i,
 \qquad
 \ell_r=\left|J_r\right|.
\]
Then
\[
 0\leq\ell_r\leq L,
 \qquad
 J_r\cap J_s=\varnothing\quad(r\neq s).
\]
For $n\in\mathcal H_m$, let $\mathcal C_r(n)$ mean that
$S(n)\cap J_r=J_r$. Since $S(n)\cap W_i=\{w\}$ and
$\min I_r>w$, all binary digits of $n$ on $W_i\cap I_r$
are $0$. If $\mathcal C_r(n)$ holds, all digits on
$J_r=I_r\setminus W_i$ are $1$. Thus
\begin{equation}\label{eq94}
 \varepsilon_t(n)=
 \begin{cases}
 0,&t\in W_i\cap I_r,\\
 1,&t\in I_r\setminus W_i.
 \end{cases}
\end{equation}
By \eqref{eq93},
\[
 n\in\sum_{j=1}^hk_jB_j,
 \qquad
 (k_1,\ldots,k_h)\neq(1,\ldots,1).
\]
Therefore, whenever $\mathcal C_r(n)$ holds, Lemma~\ref{lema9} together with
\eqref{eq94} gives
\begin{equation}\label{eq95}
 n\in h(A\setminus\{a\}).
\end{equation}

Now take
\(n\in E_a\cap\mathcal H_m\).
By $E_a=hA\setminus h(A\setminus\{a\})$,
\[
 n\notin h(A\setminus\{a\}).
\]
By \eqref{eq95}, $\mathcal C_r(n)$ fails for every
$1\leq r\leq q$. Consequently,
\begin{equation}\label{eq1}
 E_a\cap\mathcal H_m
 \subseteq
 \{n\in\mathcal H_m:
 \mathcal C_r(n)\text{ fails for every }1\leq r\leq q\}.
\end{equation}

For each $r$, the $\ell_r$ digits in $J_r$ admit $2^{\ell_r}$
assignments, exactly one of which satisfies $\mathcal C_r$. Hence
\[
 \#\{(\varepsilon_t)_{t\in J_r}:\mathcal C_r(n)\text{ fails}\}
 =2^{\ell_r}-1.
\]
Therefore,
\begin{align*}
 \frac{2^{\ell_r}-1}{2^{\ell_r}}
 &=1-2^{-\ell_r}\\
 &\leq1-2^{-L}.
\end{align*}
Let
\(J=\bigsqcup_{r=1}^qJ_r\).
Then
\begin{align*}
 \left|J\right|&=\sum_{r=1}^q\ell_r,\\
 \left|\Omega_m\setminus J\right|
 &=m-W_i(m)-\sum_{r=1}^q\ell_r.
\end{align*}
Therefore,
\[
\begin{aligned}
&\left|\{n\in\mathcal H_m:
\mathcal C_r(n)\text{ fails for every }1\leq r\leq q\}\right|\\
&\quad=2^{\left|\Omega_m\setminus J\right|}
\prod_{r=1}^q(2^{\ell_r}-1)\\
&\quad=2^{m-W_i(m)-\sum_{r=1}^q\ell_r}
\prod_{r=1}^q(2^{\ell_r}-1)\\
&\quad=2^{m-W_i(m)}\prod_{r=1}^q(1-2^{-\ell_r})\\
&\quad\leq2^{m-W_i(m)}(1-2^{-L})^q.
\end{aligned}
\]
By \eqref{eq1},
\begin{align}
 \left|E_a\cap\mathcal H_m\right|
 &\leq
 2^{m-W_i(m)}(1-2^{-L})^q
 \nonumber\\
 &=
 2^{m-W_i(m)}
 (1-2^{-L})^{\lfloor(m-w-1)/L\rfloor}.
 \label{eq96}
\end{align}

Let
\[
 \rho=1-2^{-L}\in(0,1),
 \qquad
 \theta=\rho^{1/L}\in(0,1).
\]
By
\[
 q=\left\lfloor\frac{m-w-1}{L}\right\rfloor
 \geq\frac{m-w-1}{L}-1,
\]
we obtain
\begin{align*}
 \rho^q
 &\leq
 \rho^{(m-w-1)/L-1}\\
 &=\rho^{-1}\theta^{m-w-1}\\
 &=O_{w,L}(\theta^m)\\
 &=o(1).
\end{align*}
By \eqref{eq89},
\begin{align*}
 m-W_i(m)
 &=\frac{h-1}{h}m+O(1),\\
 2^{m-W_i(m)}
 &\asymp2^{(h-1)m/h}.
\end{align*}
Substituting these two estimates into \eqref{eq96}, we obtain
\begin{equation}\label{eq97}
 \left|E_a\cap\mathcal H_m\right|
 =o\bigl(2^{(h-1)m/h}\bigr).
\end{equation}

On the other hand, by Lemma~\ref{lema6},
\[
 \left|(E_a\cap[0,2^m))\setminus\mathcal H_m\right|
 =o\bigl(2^{(h-1)K(m)}\bigr).
\]
Combining this with $K(m)=m/h+O(1)$ yields
\[
 2^{(h-1)K(m)}
 =2^{(h-1)m/h+O(1)}\asymp2^{(h-1)m/h},
\]
so that
\begin{equation}\label{eq98}
 \left|(E_a\cap[0,2^m))\setminus\mathcal H_m\right|
 =o\bigl(2^{(h-1)m/h}\bigr).
\end{equation}
Moreover, since
\[
 E_a\cap[0,2^m)
 =
 (E_a\cap\mathcal H_m)
 \sqcup
 \bigl((E_a\cap[0,2^m))\setminus\mathcal H_m\bigr),
\]
we have, by \eqref{eq97} and \eqref{eq98},
\begin{align}
 E_a(2^m-1)
 &=\left|E_a\cap[0,2^m)\right|
 \nonumber\\
 &=o\bigl(2^{(h-1)m/h}\bigr).
 \label{eq99}
\end{align}

On the other hand, strong minimality gives a constant $c_a>0$ such that, for all sufficiently large $m$,
\[
 E_a(2^m-1)
 \geq
 c_aA(2^m-1)^{h-1}.
\]
By \eqref{eq5} and \eqref{eq89},
$A(2^m-1)\asymp2^{m/h}$. Hence
\begin{align}
 E_a(2^m-1)
 &\gg_a A(2^m-1)^{h-1}
 \nonumber\\
 &\asymp2^{(h-1)m/h}.
 \label{eq100}
\end{align}
Therefore,
\[
 \frac{E_a(2^m-1)}{2^{(h-1)m/h}}
 \longrightarrow0,
\]
which contradicts
\[
 \frac{E_a(2^m-1)}{2^{(h-1)m/h}}
 \gg_a1.
\]
This contradiction establishes \eqref{eq3} for every
choice of $i$, $w$, and $(k_1,\ldots,k_h)$, and hence proves condition~(II).

Condition~(I) follows from \eqref{eq89}.
