<!-- theorem-1: body_md -->
\label{thm:equiv-main}
	Let $0 < \varepsilon \leq 1/2$, and let $p$ and $q$ be two (unknown) distributions over $[n]$. 
	There exists an algorithm that takes $\tilde O \left(\frac{\log n}{\varepsilon^2}\right)$ non-adaptive conditional samples from $p$ and $q$ and, with probability $\geq 2/3$, distinguishes between the case $p = q$ and $d_{\mathrm{TV}}(p,q) \geq \varepsilon$. 
	Furthermore, any algorithm with such guarantees requires at least $\Omega\left(\frac{\log n}{\varepsilon^2}\right)$ non-adaptive conditional samples.
<!-- end theorem-1 -->
<!-- definition-1: body_md -->
\label{def:oracle}
A conditional sampling oracle for $p$ takes a query set $S\subseteq[n]$ and returns an independent sample from $p_S$. 
If $p(S)=0$, then a symbol $i \in S$ is returned uniformly at random.
<!-- end definition-1 -->
<!-- definition-2: body_md -->
The \emph{total variation distance} between distributions $p$ and $q$ is defined as 
	\[d_{\mathrm{TV}}(p,q) = \frac{1}{2} \sum_{i=1}^n |p(i) - q(i)|.\]
<!-- end definition-2 -->
<!-- lemma-1: body_md -->
\label{lem:test-equal}
	Fix $0 < t, \delta \leq 1$. 
	Suppose we have samples from two distributions $\operatorname{Ber}(u)$ and $\operatorname{Ber}(v)$, where $u$ and $v$ are unknown.
	There exists an algorithm, \textsc{Test-equal}, that, with probability $\geq 1 - \delta$, distinguishes between the cases where $u = v$ and 
	\[ \frac{(u-v)^2}{(u+v)(2-u-v)} \geq t.\]
	It requires $O\left(\frac{\log (1/\delta)}{t}\right)$ samples from each distribution.
<!-- end lemma-1 -->
<!-- corollary-1: body_md -->
\label{cor:test-discrep}
	Fix $0 < t, \delta \leq 1$. 
	Suppose we have samples from two discrete distributions $p$ and $q$ over $[n]$.
	There exists an algorithm that, with probability $\geq 1 - \delta$, distinguishes between the cases where $p = q$ and there exists some $i \in [n]$ such that 
	\[ \frac{(p(i)-q(i))^2}{(p(i)+q(i))(2-p(i)-q(i))} \geq t.\]
	It requires $O\left(\frac{\log(1/t) + \log (1/\delta)}{t}\right)$ samples from each distribution.
<!-- end corollary-1 -->
<!-- lemma-2: body_md -->
\label{bb:tail}
\begin{equation*}
\sum_{i=1}^{n}w_i|v_i-m_i|
\geq\tfrac12d_{\mathrm{TV}}(p,q).
\end{equation*}
<!-- end lemma-2 -->
<!-- lemma-3: body_md -->
\label{lem:energy}
Suppose $d_{\mathrm{TV}}(p,q)\geq\varepsilon$. 
We let $S_{t,\ell}$ be the query set sampled by Algorithm~\ref{alg:main} at repetition $\ell$ with threshold $t$.
Then with probability at least $9/10$, there exists $t, \ell$ such that $\mathcal{F}(S_{t,\ell})\geq t$.
<!-- end lemma-3 -->
<!-- lemma-4: body_md -->
\label{lem:quad-tail}
Define the following set of indices: 
\[
\mathcal{I}=\{i: w_i \geq \varepsilon/16n,\ T_i\geq \varepsilon/16\}.
\]
If $d_{\mathrm{TV}}(p,q)\geq\varepsilon$, then
\begin{equation*}
\sum_{i\in\mathcal{I}}w_iT_i(v_i-m_i)^2
	\geq\frac{\varepsilon^2}{16\left(1+\ln\frac{16}{\varepsilon}\right)}.
\end{equation*}
<!-- end lemma-4 -->
<!-- proof-1: body_md -->
The coordinates with $w_i \leq \varepsilon/16n$ have total probability mass at most $n \cdot \varepsilon/16n = \varepsilon/16$. 
	Similarly, the coordinates with $T_i \leq \varepsilon/16$ have total mass at most $\varepsilon/16$.
	Observing that $v_i, m_i \in [-1,1]$ entrywise, we have that $|v_i - m_i| \leq 2$. 
	Thus, removing $[n] \setminus \mathcal{I}$ from the summation in \cref{bb:tail} gives 
\begin{equation*}
\sum_{i\in\mathcal{I}}w_i|v_i-m_i|
\geq\varepsilon/2-2(\varepsilon/16+\varepsilon/16)=\varepsilon/4.
\end{equation*}
Thus, 
\begin{equation}\label{eq:retained-l1}
	\left(\sum_{i\in\mathcal{I}}w_i|v_i-m_i|\right)^2 \geq \varepsilon^2/16.
\end{equation}
We are left with $\mathcal{I} = [r]$ for some $r \leq n$. 
	Since $w_i=T_i-T_{i+1}$,
	\begin{equation}\label{eq:telescoping}
\sum_{i=1}^r\frac{w_i}{T_i}
	= \sum_{i=1}^r\frac{T_i-T_{i+1}}{T_i} = \sum_{i=1}^r 1- \frac{T_{i+1}}{T_i}
\leq1+\sum_{i=1}^{r-1}\ln\frac{T_i}{T_{i+1}}
=1+\ln\prod_{i=1}^{r-1}\frac{T_i}{T_{i+1}}
=1+\ln\frac1{T_r}
	\leq1+\ln\frac{16}{\varepsilon}.
\end{equation}
The first inequality bounds the $i=r$ term by $1$, and the remaining terms using $1-x\leq - \ln x$. 

\noindent We factor the summand of the summation as follows: 
\[
w_i|v_i-m_i|=
\sqrt{w_iT_i(v_i-m_i)^2}\sqrt{w_i/T_i}.
\]
Cauchy-Schwarz says that 
\[
	\left(\sum_{i \in \mathcal{I}} w_i|v_i-m_i|\right)^2\leq 
	\left(\sum_{i \in \mathcal{I}} w_iT_i(v_i-m_i)^2\right)\left(\sum_{i \in \mathcal{I}} w_i/T_i\right).
\]
Combining this with \eqref{eq:retained-l1} and \eqref{eq:telescoping} gives the desired statement.
<!-- end proof-1 -->
<!-- proposition-1: body_md -->
\label{prop:i-heaviest}
	Fix $i \in [n]$, and let $j$ be the index of the bucket $B_j$ that $i$ belongs to.
	Let $\rho = 2^{-j}$ be a sampling probability, and let $S \subseteq [n]$ be a random set where every symbol in $[n]$ is independently included with probability $\rho$. 
	Then $\mathbb{P}_S(A_i \mid j) \geq \rho/4$.
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
First, we have that $A_i$ occurs precisely when symbol $i$ is selected and none of the previous $i-1$ symbols are selected: $\mathbb{P}_S(A_i \mid j) = \rho (1-\rho)^{i-1}$.
	Observe that $(i-1)w_i \leq iw_i \leq 1$: the latter inequality holds because the $w_i$'s are in decreasing order and form a probability distribution. 
	If this weren't the case, this prefix would contain more than $1$ mass. 
	Note further that $\rho \leq w_i$, based on the definition of our bucketing scheme.

<!-- proof-4: body_md -->
Fix coordinate $i \in [n]$, the index $j$ of the bucket $B_j$ it belongs to, and the sampling probability $\rho=2^{-j}$. 
	We will condition on the event $A_i$, where $i$ is the first symbol selected and every subsequent symbol is selected with probability $\rho$, and the choice of the index $j$. 
	Define the following quantities:
\begin{equation*}
U=\sum_{\ell\in S\setminus\{i\}}w_\ell,\qquad
W=w_i+U,\qquad
Y=\sum_{\ell\in S}w_\ell(v_i-v_\ell).
\end{equation*}
$U$ and $W$ are the total weight of the set $S$, without and with the contribution of symbol $i$. 
$Y$ captures how the relative difference between $p$ and $q$ at symbol $i$ compares to their relative difference (on average) across the rest of the set $S$.


We compute the (conditional) expectation $Y$: 
\begin{equation}\label{eq:EY}
\mathbb{E}_S[Y\mid A_i,j]=\rho\sum_{\ell\geq i}w_\ell(v_i-v_\ell)
=\rho T_i(v_i-m_i),
\end{equation}
	where we recall the definitions $T_i = \sum_{\ell=i}^n w_\ell$ and $m_i = \frac{\sum_{\ell=i}^n w_\ell v_\ell}{T_i}$. 

We establish the pointwise bound
\begin{equation}\label{eq:pointwise}
\mathcal{F}(S)\geq\frac{w_i^2Y^2}{2W^3U}.
\end{equation}
First, we deal with degenerate cases: if $U=0$, $p(S)=0$, or $q(S) = 0$, it is not hard to see that $Y=0$ and the bound holds vacuously.
Therefore, we assume that $p(S), q(S),$ and $U > 0$. 

We inspect the following lower bound for $\mathcal{F}(S)$, examining the contribution at symbol $i$. 
\[
\mathcal{F}(S)\geq\frac{(p_S(i)-q_S(i))^2}{(p_S(i)+q_S(i))(2-p_S(i)-q_S(i))}
\]
We first derive an alternative form for the numerator. 
We write
\[
p_S(i)-q_S(i)=\frac{p(i)q(S)-q(i)p(S)}{p(S)q(S)}.
\]
	Using $p(\ell) = \frac{p(\ell) + q(\ell)}{2}\cdot \frac{(p(\ell) + q(\ell)) + (p(\ell) - q(\ell))}{p(\ell) + q(\ell)}=w_\ell(1+v_\ell)$ and $q(\ell)=w_\ell(1-v_\ell)$, we expand the numerator as
\[
\begin{aligned}
p(i)q(S)-q(i)p(S)
&=\sum_{\ell\in S}\bigl(p(i)q(\ell)-q(i)p(\ell)\bigr)\\
&=w_i\sum_{\ell\in S}w_\ell
\bigl((1+v_i)(1-v_\ell)-(1-v_i)(1+v_\ell)\bigr)\\
&=2w_i\sum_{\ell\in S}w_\ell(v_i-v_\ell)
=2w_iY.
\end{aligned}
\]
Therefore,
\[
	(p_S(i)-q_S(i))^2
	=\frac{4w_i^2Y^2}{(p(S)q(S))^2}.
\]
To deal with the second term in the denominator, write each complement probability in terms of its remaining mass:
\[
\begin{aligned}
2-p_S(i)-q_S(i)
&=(1-p_S(i))+(1-q_S(i))\\
&=\frac{p(S\setminus\{i\})}{p(S)}
+\frac{q(S\setminus\{i\})}{q(S)}\\
&=\frac{p(S\setminus\{i\})q(S)+q(S\setminus\{i\})p(S)}{p(S)q(S)}.
\end{aligned}
\]
Also $p(S)+q(S)=2W$ and $p(S\setminus\{i\})+q(S\setminus\{i\})=2U$, so
\[
p(S)q(S) \leq \frac14\left(p(S)+q(S)\right)^2= W^2
\]
and
\[
p(S\setminus\{i\})q(S)+q(S\setminus\{i\})p(S) \leq (p(S)+q(S))(p(S\setminus\{i\})+q(S\setminus\{i\})) =4UW.
\]
As a last observation, $p_S(i)+q_S(i)\leq2$.
Putting this all together, 
\[
\begin{aligned}
\mathcal{F}(S)
&\geq\frac{(p_S(i)-q_S(i))^2}{(p_S(i)+q_S(i))(2-p_S(i)-q_S(i))}\\
	&\geq \frac{4w_i^2Y^2 \cdot p(S)q(S)}{2\cdot (p(S)q(S))^2\bigl(p(S\setminus\{i\})q(S)+q(S\setminus\{i\})p(S)\bigr)}\\
&\geq\frac{w_i^2Y^2}{2W^3U},
\end{aligned}
\]
which proves \eqref{eq:pointwise} in this case.

We use the following technical lemma to bound the expectation of the quantity in the denominator, whose proof appears in Appendix~\ref{app:moment}. 
\begin{restatable}{lemma}{momentbound}\label{lem:moment}
Let $0<\rho\leq1/2$, let $0<a\leq2\rho$, and let $b_1,\ldots,b_s\in[0,2\rho]$ satisfy
\[
T:=a+\sum_{\ell=1}^s b_\ell\leq1.
\]
Let $\xi_1,\ldots,\xi_s$ be independent $\operatorname{Ber}(\rho)$ variables, and set $U=\sum_\ell b_\ell\xi_\ell$. Then
\begin{equation*}\label{eq:moment}
\mathbb{E}[(a+U)^3U]\leq159\rho^4T.
\end{equation*}
\end{restatable}

We apply Lemma~\ref{lem:moment} with $a=w_i$ and the vector $b$ being the vector $w_\ell$ for $\ell > i$.
It is not hard to verify the lemma conditions using the definition and choice of $\rho$, the fact that $w_\ell$ is non-increasing, and $T$ corresponds to $T_i \leq 1$. 
This gives 
\begin{equation}\label{eq:moment-used}
\mathbb{E}_S[W^3U\mid A_i,j]\leq 159\rho^4T_i \leq 256 \rho^4 T_i.
\end{equation}

We let $Z = W^3 U$, and decompose $Y = \frac{Y}{\sqrt{Z}} \sqrt{Z}$ for the sake of applying Cauchy-Schwarz. 
Specifically: 
\[
\begin{aligned}
(\mathbb{E}_S[Y\mid A_i,j])^2
&=\left(\mathbb{E}_S\!\left[\frac{Y}{\sqrt Z}\sqrt Z\,\middle|\,A_i,j\right]\right)^2\\
&\leq\mathbb{E}_S[Y^2/Z\mid A_i,j]\mathbb{E}_S[Z\mid A_i,j].
\end{aligned}
\]
Returning to the pointwise bound \eqref{eq:pointwise} and taking (conditional) expectation of both sides: 
\begin{equation*}\label{eq:conditional-F}
\begin{aligned}
\mathbb{E}_S[\mathcal{F}(S)\mid A_i,j]
&\geq\frac{w_i^2}{2}\mathbb{E}_S[Y^2/Z\mid A_i,j]\\
&\geq\frac{w_i^2(\mathbb{E}_S[Y\mid A_i,j])^2}{2\mathbb{E}_S[Z\mid A_i,j]}\\
&\geq\frac{w_i^2}{512\rho^2}\,T_i(v_i-m_i)^2.
\end{aligned}
\end{equation*}
The last inequality uses \eqref{eq:EY} and \eqref{eq:moment-used}.
Finally, use $\mathbb{P}_S(A_i\mid j)\geq\rho/4$ from \cref{prop:i-heaviest} and $w_i\geq\rho$:
\[
\mathbb{E}_S[\mathcal{F}(S)\mathbf{1}_{A_i}\mid j]
=\mathbb{P}_S(A_i\mid j)\mathbb{E}_S[\mathcal{F}(S)\mid A_i,j]
\geq\frac{w_i^2}{2048\rho}T_i(v_i-m_i)^2
\geq\frac{w_i}{2048}T_i(v_i-m_i)^2.
\]

<!-- lemma-7: body_md -->
\label{lb:lem:transcript}
Fix $0<\varepsilon\le1/2$, a support $D$, and a list of queries $(S_1,b_1),\ldots,(S_N,b_N)$, where
for each $i \in [N]$, $S_i\subseteq[n]$ is the query set and $b_i\in\{p,q\}$ is the queried distribution.
Let $s_i=|S_i\cap D|$ be the number of support points in the $i$th query.
Let $\mathsf T_0^D$ and $\mathsf T_1^D$ be the distributions of the entire
answer list under $\mathcal H_0$ and $\mathcal H_1$, respectively,
conditional on the support $D$ (but not the random perturbation vector $\sigma$). 

There is an absolute constant $C$ such that
\begin{equation}\label{lb:eq:transcript-bound}
  d_{\mathrm{TV}}(\mathsf T_0^D,\mathsf T_1^D)
  \le \min\left\{1,\ C\varepsilon^2
                   \sum_{i=1}^N h(s_i)\right\}.
\end{equation}
<!-- end lemma-7 -->
<!-- proof-6: body_md -->
We use the second-moment method to bound the total variation distance, by looking at the second moment of the likelihood ratio. 
Specifically, we upper bound the total-variation distance between the transcripts in terms of their chi-squared distance:
\[
  d_{\mathrm{TV}}(\mathsf T_1^D,\mathsf T_0^D)
	\leq \tfrac12\sqrt{\chi^2(\mathsf T_1^D \| \mathsf T_0^D)}
  =\tfrac12\sqrt{\mathbb{E}_{\mathsf T_0^D}[\mathcal L^2]-1},
\]
where $\mathcal L$ is the likelihood ratio 
\[
  \mathcal L(x)
    =\frac{\mathsf T_1^D(x)}{\mathsf T_0^D(x)}.
\]

If a query has $s_i=0$ or $1$, then the answer distribution is identical under the two hypotheses and the likelihood ratio for this query will be $1$, regardless of the sign vector $\sigma$. 
For the sequel, we focus on the remaining queries which have $s_i \geq 2$. 

Focus on the likelihood ratio for a single answer. 
For a fixed sign vector $\sigma$, let $\bar\sigma_i$ be its average
	over the support points in query $i$ -- that is, the fraction of points that are spiked upwards minus the fraction of points that are spiked downwards, which gives a measure of the excess mass assigned to $p$ under $\mathcal{H}_1$. 
\[
  \bar\sigma_i=\frac1{s_i}\sum_{x\in S_i\cap D}\sigma_x,
\]
We let $r_{\sigma,i}(x)$ be the
ratio of the probability of answering $x$ under $\mathcal H_1$ to that
under $\mathcal H_0$.
For distribution $p$ and under $\mathcal{H}_0$, the probability of outputting any $x \in S_i \cap D$ is $1/s_i$. 
Under $\mathcal{H}_1$, the probability is $\frac{1 + \varepsilon\sigma_x}{s_i + \varepsilon s_i \bar \sigma_i}.$
Taking their ratio (and with a similar calculation for $q$), for any $x \in S_i \cap D$, we have:
\[
  r_{\sigma,i}(x)
  =\begin{cases}
    \displaystyle\frac{1+\varepsilon\sigma_x}{1+\varepsilon\bar\sigma_i},&b_i=p,\\[6pt]
    \displaystyle\frac{1-\varepsilon\sigma_x}{1-\varepsilon\bar\sigma_i},&b_i=q.
  \end{cases}
\]
Note that these likelihood ratios are conditioned on a particular $\sigma$, whereas the transcript distribution $\mathsf T_1^D$ chooses $\sigma$ uniformly at random, which averages the likelihood ratio over all $\sigma$.  
For any $x \in S_i \cap D$, a symmetry argument implies that $\mathbb{E}_\sigma[p(x \mid S_i)] = \mathbb{E}_\sigma[q(x \mid S_i)] = 1/s_i$.
	Since the same is true for a sample under $\mathcal{H}_0$ (since the distribution is uniform), we have that 
\begin{equation}
	\mathbb{E}_\sigma[r_{\sigma,i}(x)]=1. \label{lb:eq:one-coordinate}
\end{equation}

The challenge in analysis is that we are doing \emph{several} samples with respect to a single sign vector $\sigma$, not just one sample, and we must account for correlations between the likelihood ratios of individual samples. 
The likelihood ratio for the whole transcript is as follows: 
\[
  \mathcal L(x)
  =\mathbb{E}_\sigma\left[\prod_i r_{\sigma,i}(x_i)\right].
\]
Note that, conditioned on a fixed sign vector $\sigma$, the answers are independent, justifying the product of the $r_{\sigma,i}$'s inside the expectation. 

We look at the square of the likelihood ratio by taking a product with the same expression using independent randomness; we let $\sigma'$ denote an independently sampled sign vector.  
\[
  \mathcal L(x)^2
  =\mathbb{E}_{\sigma}\left[\prod_i r_{\sigma,i}(x_i)\right]\mathbb{E}_{\sigma'}\left[\prod_i r_{\sigma',i}(x_i)\right]
  =\mathbb{E}_{\sigma,\sigma'}\left[\prod_i r_{\sigma,i}(x_i)r_{\sigma',i}(x_i)\right].
\]

Recall our goal is to bound 
	\begin{equation*}
	\begin{aligned}
		\mathbb{E}_{x \sim \mathsf T_0^D}[\mathcal L^2] &= \mathbb{E}_{x \sim \mathsf T_0^D}\left[\mathbb{E}_{\sigma,\sigma'}\left[\prod_i r_{\sigma,i}(x_i)r_{\sigma',i}(x_i)\right]\right] \\ 
		&= \mathbb{E}_{\sigma,\sigma'}\left[\mathbb{E}_{x \sim \mathsf T_0^D}\left[\prod_i r_{\sigma,i}(x_i)r_{\sigma',i}(x_i)\right]\right] \\
		&= \mathbb{E}_{\sigma,\sigma'}\left[\prod_i \mathbb{E}_{x_i \sim \operatorname{Unif}(S_i \cap D)} [r_{\sigma,i}(x_i)r_{\sigma',i}(x_i)]\right].
	\end{aligned}
	\end{equation*}
	We can factor the expectation in the last equality since the $x_i$'s are independent under the null. 

	We rewrite each individual factor in this product as 
	\[
		1 + K_i(\sigma, \sigma') := \mathbb{E}_{x_i \sim \operatorname{Unif}(S_i \cap D)} [r_{\sigma,i}(x_i)r_{\sigma',i}(x_i)].
	\]
	We proceed to derive an expression for 
	\[
	K_i(\sigma,\sigma') = \mathbb{E}_{x_i \sim \operatorname{Unif}(S_i \cap D)} [r_{\sigma,i}(x_i)r_{\sigma',i}(x_i)] -1. 
	\]
We illustrate by focusing on a query to $p$; a query to $q$ will be analogous. 
Fix the sign vectors $\sigma, \sigma'$. 
We substitute the expression for $r_{\sigma,i}$ and $r_{\sigma',i}$ and expand the expectation: 
\[
  K_i(\sigma,\sigma')
  =\frac1{s_i}\sum_{x_i\in S_i\cap D}
    \frac{(1+\varepsilon\sigma_{x_i})(1+\varepsilon\sigma'_{x_i})}
         {(1+\varepsilon\bar\sigma_i)(1+\varepsilon\bar{\sigma}'_i)}-1 
  =\frac1{s_i}\sum_{x_i\in S_i\cap D}
  \frac{(1+\varepsilon\sigma_{x_i})(1+\varepsilon\sigma'_{x_i})
          -(1+\varepsilon\bar\sigma_i)(1+\varepsilon\bar{\sigma}'_i)}
         {(1+\varepsilon\bar\sigma_i)(1+\varepsilon\bar{\sigma}'_i)}.
\]
We expand the numerator as follows:
\begin{align*}
  &(1+\varepsilon\sigma_{x_i})(1+\varepsilon\sigma'_{x_i})
    -(1+\varepsilon\bar\sigma_i)(1+\varepsilon\bar{\sigma}'_i)\\
  &\qquad=\varepsilon(\sigma_{x_i}+\sigma'_{x_i}
                           -\bar\sigma_i-\bar{\sigma}'_i)
       +\varepsilon^2(\sigma_{x_i}\sigma'_{x_i}
                           -\bar\sigma_i\bar{\sigma}'_i).
\end{align*}
The terms which are linear in $\varepsilon$ vanish when we average over $x_i \in S_i \cap D$, as we substitute  $\bar\sigma_i=\frac1{s_i}\sum_{x_i \in S_i\cap D}\sigma_{x_i}$
\[
  \frac1{s_i}\sum_{x_i\in S_i\cap D}
    (\sigma_{x_i}+\sigma'_{x_i}-\bar\sigma_i-\bar{\sigma}'_i)
  =\bar\sigma_i+\bar{\sigma}'_i-\bar\sigma_i-\bar{\sigma}'_i=0.
\]
Only the terms which are quadratic in $\varepsilon$ remain, yielding
\[
  K_i(\sigma,\sigma')
  =\varepsilon^2\,
    \frac{\displaystyle\frac1{s_i}\sum_{x_i\in S_i\cap D}
                         \sigma_{x_i}\sigma'_{x_i}
          -\bar\sigma_i\bar{\sigma}'_i}
         {(1+\varepsilon\bar\sigma_i)(1+\varepsilon\bar{\sigma}'_i)}.
\]
Doing the same calculations for a query to $q$, we obtain an expression where the only difference is the denominator becomes $(1-\varepsilon\bar\sigma_i)(1-\varepsilon\bar{\sigma}'_i)$.

At this point, we have obtained:
\begin{equation}\label{lb:eq:chi-identity}
  \mathbb{E}_{\mathsf T_0^D}[\mathcal L^2]
  =\mathbb{E}_{\sigma,\sigma'}\left[\prod_i(1+K_i(\sigma,\sigma'))\right].
\end{equation}
We further upper bound this using the inequality $1 + x \leq e^x$: 

\begin{equation*}
  \mathbb{E}_{\mathsf T_0^D}[\mathcal L^2]
	\leq\mathbb{E}_{\sigma,\sigma'}\left[\prod_i\exp\left(K_i(\sigma, \sigma')\right)\right]
	=\mathbb{E}_{\sigma,\sigma'}\left[\exp\left(\sum_i K_i(\sigma, \sigma')\right)\right] 
	= \mathbb{E}_{\sigma,\sigma'}\left[e^Z\right],
\end{equation*}
where we have defined
\[
  Z=\sum_i K_i(\sigma,\sigma').
\]
We use the power series expansion of this exponential:
\[
  \mathbb{E}_{\sigma,\sigma'}\left[e^Z\right]
  = \sum_{k=0}^\infty \mathbb{E}_{\sigma,\sigma'}\left[\frac{Z^k}{k!}\right].
\]
The $k = 0$ term evaluates to $1$. 
The $k= 1$ term $\mathbb{E}_{\sigma, \sigma'}[Z]=\sum_i \mathbb{E}_{\sigma, \sigma'} [K_i(\sigma,\sigma')]$ vanishes:
\[
	\mathbb{E}_{\sigma,\sigma'}[K_i(\sigma,\sigma')] = \mathbb{E}_{x_i \sim \operatorname{Unif}(S_i \cap D)} [\mathbb{E}_\sigma [r_{\sigma,i}(x_i)]\mathbb{E}_{\sigma'} [r_{\sigma',i}(x_i)]]  -1 = 1^2 -1 = 0,
\]
where the first equality uses the definition of $K_i$ and independence of $\sigma$ and $\sigma'$, and the second equality uses \eqref{lb:eq:one-coordinate}. 
We then sum over $i$ to get $\mathbb{E}_{\sigma, \sigma'}[Z] = 0$.


Consequently,
\[
  \mathbb{E}_{\sigma,\sigma'}\left[e^Z\right]
  =1+\sum_{k=2}^\infty\frac{\mathbb{E}_{\sigma,\sigma'}[Z^k]}{k!}
  \le1+\sum_{k=2}^\infty\frac{\mathbb{E}_{\sigma,\sigma'}[|Z|^k]}{k!}
  =1+\sum_{k=2}^\infty\frac{\|Z\|_k^k}{k!}.
\]
It remains to bound these absolute moments. We do this by bounding
the moments of each $K_i$ and then applying the triangle inequality.
The proof of the following lemma (a variant of Khintchine inequality) appears in Appendix~\ref{app:balanced-signs}. 

\begin{restatable}{lemma}{balancedsignbound}\label{lb:lem:signs}
Let $D$ have even cardinality, and let $\sigma$ be uniformly sampled from the set of all balanced sign vectors on $D$. 
There is an absolute constant $C_0$ such that, for
every vector $a\in\mathbb{R}^D$ and every $k\ge 2$,
\[
  \left\|\sum_{x\in D}a_x\sigma_x\right\|_k
  \le C_0\sqrt k\left(\sum_{x\in D}a_x^2\right)^{1/2},
  \qquad
  \left\|X\right\|_k=(\mathbb{E}[|X|^k])^{1/k}.
\]
\end{restatable}
Specifically, we apply this bound to the two terms in the numerator of $K_i(\sigma, \sigma')$:
\[
  \frac1{s_i}\sum_{x\in S_i \cap D}\sigma_x\sigma'_x
  \qquad \text{and} \qquad
  \bar\sigma_i\bar{\sigma}'_i.
  \]


For the first term, fix $\sigma'$ and choose the coefficients in
Lemma~\ref{lb:lem:signs} to be
\[
  a_x=\begin{cases}
    \sigma'_x/s_i,&x\in S_i\cap D,\\
    0,&x\in D\setminus S_i.
  \end{cases}
\]
Then
\[
  \sum_{x\in D}a_x\sigma_x
  =\frac1{s_i}\sum_{x\in S_i\cap D}\sigma_x\sigma'_x,
  \qquad
  \sum_{x\in D}a_x^2
  =\frac{s_i}{s_i^2}=\frac1{s_i}.
\]
Thus, for every integer $k\ge2$, the lemma gives
\[
  \mathbb{E}_\sigma\left[
    \left|\frac1{s_i}\sum_{x\in S_i\cap D}\sigma_x\sigma'_x\right|^k
    \,\middle|\,\sigma'\right]
  \le\left(C_0\sqrt{\frac{k}{s_i}}\right)^k.
\]
This bound holds for every fixed $\sigma'$, so averaging over $\sigma'$
gives the same bound on the joint expectation over both sign vectors.

For the second term, choose $a_x=1/s_i$ on $S_i\cap D$ and zero
elsewhere in $D$. Now $\sum_xa_x\sigma_x=\bar\sigma_i$, and the
squared coefficients again sum to $1/s_i$. The lemma gives
\[
  \mathbb{E}_\sigma[|\bar\sigma_i|^k]
  \le\left(C_0\sqrt{\frac{k}{s_i}}\right)^k.
\]
The same bound applies to $\bar{\sigma}'_i$. Independence of the two
sign vectors then gives
\begin{align*}
  \mathbb{E}_{\sigma,\sigma'}[|\bar\sigma_i\bar{\sigma}'_i|^k]
  &=\mathbb{E}_\sigma[|\bar\sigma_i|^k]\,
    \mathbb{E}_{\sigma'}[|\bar{\sigma}'_i|^k]\\
  &\le\left(C_0^2\frac{k}{s_i}\right)^k.
\end{align*}
Taking $k$th roots, we obtain the following bounds, with both norms
over the two independent sign vectors:
\[
  \left\|\frac1{s_i}\sum_{x\in S_i\cap D}\sigma_x\sigma'_x\right\|_k
    \le C_0\sqrt{\frac k{s_i}},
  \qquad
  \left\|\bar\sigma_i\bar{\sigma}'_i\right\|_k
    =\left\|\bar\sigma_i\right\|_k\left\|\bar{\sigma}'_i\right\|_k
    \le C_0^2\frac k{s_i}.
\]

We turn our attention to the denominator of $K_i(\sigma, \sigma')$, which is $(1 \pm \varepsilon \bar \sigma_i)(1 \pm \varepsilon \bar \sigma_i')$, where the sign depends on whether the query is to $p$ or $q$. 
Regardless, based on the range of $\bar \sigma_i$, we can lower bound this expression by $(1 - \varepsilon)^2 \geq 1/4$. 
Using this denominator bound and then the triangle inequality gives
\begin{equation}\label{lb:eq:kernel-moment}
\begin{aligned}
  \|K_i\|_k
  &\le4\varepsilon^2
    \left\|\frac1{s_i}\sum_{x_i\in S_i\cap D}\sigma_{x_i}\sigma'_{x_i}
           -\bar\sigma_i\bar{\sigma}'_i\right\|_k\\
  &\le4\varepsilon^2\left(
    \left\|\frac1{s_i}\sum_{x_i\in S_i\cap D}\sigma_{x_i}\sigma'_{x_i}\right\|_k
    +\|\bar\sigma_i\bar{\sigma}'_i\|_k\right)\\
  &\le4\varepsilon^2\left(C_0\sqrt{\frac{k}{s_i}}
                          +C_0^2\frac{k}{s_i}\right)\\
  &\le C_1\varepsilon^2\frac{k}{\sqrt{s_i}}
   =C_1\varepsilon^2k\,h(s_i).
\end{aligned}
\end{equation}
The last inequality uses $\sqrt{k}\le k$, $1/s_i\le1/\sqrt{s_i}$, and introduces a new absolute constant $C_1 = 4(C_0 + C_0^2)$.
Observe here that the dependence on the size of the intersection $s_i$ is $1/\sqrt{s_i}$, which is precisely how we defined the function $h(s_i)$. 

We can now bound the moments of $Z$.
\begin{equation}\label{lb:eq:Z-moment}
  \|Z\|_k
  =\left\|\sum_iK_i\right\|_k
  \le\sum_i\|K_i\|_k
  \le\sum_i C_1\varepsilon^2k\,h(s_i)
  =C_1\varepsilon^2k\sum_i h(s_i)=\gamma k,
\end{equation}
where we let $\gamma = C_1 \varepsilon^2 \sum_i h(s_i)$. 
The two inequalities are triangle inequality and \eqref{lb:eq:kernel-moment}. 


We now return to the series we were trying to bound.
First, we consider the case $e \gamma \leq 1/2$. 
Equation~\eqref{lb:eq:Z-moment} gives
$\mathbb{E}_{\sigma,\sigma'}[|Z|^k]\le(\gamma k)^k$, and $k!\ge(k/e)^k$
turns the bound into a geometric series:
\begin{align*}
  \mathbb{E}_{\mathsf T_0^D}[\mathcal L^2]
  &\le \mathbb{E}_{\sigma,\sigma'}[e^Z]\\
  &\le 1+\sum_{k=2}^\infty\frac{\mathbb{E}_{\sigma,\sigma'}[|Z|^k]}{k!}\\
  &\le 1+\sum_{k=2}^\infty(e\gamma)^k
   \le 1+2e^2\gamma^2.
\end{align*}
The TV inequality from the start of the proof now gives
\[
  d_{\mathrm{TV}}(\mathsf T_1^D,\mathsf T_0^D)
  \le\tfrac12\sqrt{\mathbb{E}_{\mathsf T_0^D}[\mathcal L^2]-1}
  \le e\gamma/\sqrt{2}= \frac{eC_1}{\sqrt{2}}\varepsilon^2 \sum_i h(s_i),
\]
as desired by the lemma statement.
When $e\gamma>1/2$, we can use the trivial bound $d_{\mathrm{TV}}\le1\le2e\gamma$ to conclude the proof.

\label{lb:lem:scales}
	Let $L = \log_2 n$, choose $J$ uniformly from $\{1, 2, \dots, \log_2 n\}$, and set $s = 2^J$. Choose $D \subseteq [n]$ to be a uniformly random subset of $[n]$ of size $s$.

\end{equation}

Let $b=|S|\ge1$.
For the time being, fix the support size $s = 2^j$ and 
	define
\[
  M=|S\cap D|,\qquad \lambda=\mathbb{E}[M]=bs/n.
\]
The randomness in the following bounds is exclusively over a uniformly random support of this fixed size, and we incorporate randomness in $s$ itself at the very end of the proof. 

We bound $\mathbb{E}[h(M)]$ separately for small and large values of $\lambda$.
We start with $\lambda\le1$: intuitively, this captures the case where the query is unlikely to contain enough support points to contribute to the expectation. 
Since $h(M)=0$ when $M=0$ or $1$, and $h(M)\le1\le M/2$
when $M\ge2$, we have $h(M)\le M/2$. Taking expectations gives
\begin{equation}\label{lb:eq:small-scale}
  \mathbb{E}[h(M)]\le\frac{\mathbb{E}[M]}2=\frac\lambda2.
\end{equation}

If $\lambda>1$ (i.e., the expected intersection is substantial), split according to whether $M\ge\lambda/2$.
If so, then $h(M)$ will be small: $h(M) \leq \sqrt{2/\lambda}$. 
In the other case, $h(M)$ may be large (we again will use the crude bound $h(M) \leq 1$), but we bound the probability that this case happens using Chebyshev's inequality. 
The indicators $\mathbf1_{\{x\in D\}}$, for $x\in S$, are negatively
correlated random variables: conditioning on one point belonging to the
	fixed-size support makes another less likely to belong. Hence, noting that $M = \sum_{x \in S}\mathbf1_{\{x\in D\}}$
\[
  \operatorname{Var}(M)
  \le\sum_{x\in S}\operatorname{Var}(\mathbf1_{\{x\in D\}})
  =\sum_{x\in S}\frac{s}{n}\left(1-\frac{s}{n}\right)
  \le\sum_{x\in S}\frac{s}{n}
  =\frac{bs}{n}=\lambda.
\]
Chebyshev's inequality then gives
$\mathbb{P}(M<\lambda/2)\le4/\lambda$. Consequently,
\begin{equation}\label{lb:eq:large-scale}
  \mathbb{E}[h(M)]
  \le\sqrt{\frac2\lambda}+\mathbb{P}(M<\lambda/2)
  \le\sqrt{\frac2\lambda}+\frac4\lambda
  \le\frac{\sqrt2+4}{\sqrt\lambda}.
\end{equation}

At this point, we sum over the choice of the support size $s = 2^j$. 
The expected intersection size
$\lambda_j=b2^j/n$ doubles at each successive scale.
We split the analysis as we did before, when $\lambda_j \leq 1$ and when $\lambda_j > 1$. 
	In the former case, a geometric series bounds the sum of conditional expectations $\sum_{j : \lambda_j \leq 1} \mathbb{E}\bigl[h(|S\cap D|)\mid s = 2^j]$ by $\frac{1}{2} \sum_{k \geq 0} \left(\frac12\right)^{k} = 1$. 
	In the latter case, we can similarly bound the sum of conditional expectations $\sum_{j : \lambda_j > 1} \mathbb{E}\bigl[h(|S\cap D|)\mid s = 2^j]$ by the geometric series $(\sqrt{2} + 4) \sum_{k \geq 0} \left(\frac{1}{\sqrt{2}}\right)^{k} = \frac{\sqrt{2}+4}{1 - 2^{-1/2}} \leq 19$.
Together,
\[
  \sum_{j=1}^L
    \mathbb{E}\bigl[h(|S\cap D|)\mid s=2^j\bigr]
  \le1+\frac{\sqrt2+4}{1-2^{-1/2}}\leq20.
\]
The support scale is uniform among these $L$ choices, so dividing by
$L$ gives the desired bound on expectation.

Suppose an algorithm makes $N$ queries, and fix its random seed $\omega$. 
	We represent its query list as 
\[
  (S_1(\omega),b_1(\omega)),\ldots,(S_N(\omega),b_N(\omega)),
\]
where $S_i$ denotes the $i$-th query set and $b_i \in \{p,q\}$ denotes which oracle we query. 
Note that since the algorithm is non-adaptive, the query list is independent of the input. 

We augment the transcript by including the identity of the support $D$ and the random seed $\omega$. 
This can only give the testing algorithm more information and thus will not increase its error probability.
Conditioning on $(\omega, D)$, the answer lists under $\mathcal H_0$ and
$\mathcal H_1$ are exactly those compared in
Lemma~\ref{lb:lem:transcript}, where the fixed query list is $(S_i(\omega),b_i(\omega))_{i=1}^{N}$.
Let $\mathsf T_0,\mathsf T_1$ denote the distribution of the augmented transcripts, 
and let $\mathsf T_0^{D,\omega},\mathsf T_1^{D,\omega}$ denote the distribution of the augmented transcripts conditioned on a particular support $D$ and query list determined by a random seed $\omega$. 
Because the augmented transcripts have the
same distribution of $(\omega,D)$ under both $\mathcal H_0$ and $\mathcal H_1$,
\[
  d_{\mathrm{TV}}(\mathsf T_0,\mathsf T_1)
  =\mathbb{E}_{\omega,D}\left[
      d_{\mathrm{TV}}(\mathsf T_0^{D,\omega},\mathsf T_1^{D,\omega})\right].
\]
Applying Lemma~\ref{lb:lem:transcript} for each fixed $(D,\omega)$,
then using linearity of expectation and Lemma~\ref{lb:lem:scales}, gives
\begin{align}
  d_{\mathrm{TV}}(\mathsf T_0,\mathsf T_1)
  &\le C\varepsilon^2\,
       \mathbb{E}_{\omega,D}\left[\sum_{i=1}^{N}
          h(\left|S_i(\omega)\cap D\right|)\right]\nonumber\\
  &= C\varepsilon^2\,
       \mathbb{E}_\omega\left[\sum_{i=1}^{N}
          \mathbb{E}_D[h(\left|S_i(\omega)\cap D\right|)]\right]\nonumber\\
  &\le C\varepsilon^2\,
       \mathbb{E}_\omega\left[\sum_{i=1}^{N}\frac{20}{L}\right]\nonumber\\
  &= \frac{20C\varepsilon^2N}{L}.
     \label{lb:eq:final-tv}
\end{align}

Correctness requires us to output that $p=q$ with probability at least $2/3$ under $\mathcal H_0$ and with probability at most $1/3$ under $\mathcal H_1$. 
These correspond to cases when we are given transcripts drawn from distributions $\mathsf T_0$ and $\mathsf T_1$, respectively.
We can lower bound the total variation distance between these cases by this gap in probabilities, and thus
\[
  \frac13\led_{\mathrm{TV}}(\mathsf T_0,\mathsf T_1)
       \le\frac{20C\varepsilon^2N}{L}.
\]
It follows that
\[
  N\ge\frac{L}{60C\varepsilon^2}
    =\Omega\!\left(\frac{\log n}{\varepsilon^2}\right),
\]
as desired.

\label{lem:khintchine}
Let $D$ be a finite set, and let the signs $(\xi_x)_{x\in D}$ be
independently and uniformly sampled from $\{-1,+1\}$.
There is an absolute constant $C$ such that, for
 every vector $a\in\mathbb{R}^D$ and every $k\ge2$,
\[
  \left\|\sum_{x\in D}a_x\xi_x\right\|_k
  \le C\sqrt k\left(\sum_{x\in D}a_x^2\right)^{1/2},
  \qquad
  \left\|X\right\|_k=(\mathbb{E}[|X|^k])^{1/k}.
\]
<!-- end lemma-10 -->
<!-- proof-10: body_md -->
Choose a uniformly random perfect matching between the elements of the set $D$.
For each pair $\{u,v\}$ in this matching, independently set
$(\sigma_u,\sigma_v)=(-1,+1)$ or $(+1,-1)$ with equal probability.
The resulting sign vector $\sigma$ satisfies the conditions of the lemma. 

Conditional on a fixed matching, each pair $\{u,v\}$ contributes
$\pm(a_u-a_v)$ to the sum $\sum_xa_x\sigma_x$, with independent uniform
	choices of sign across pairs. Applying the Khintchine inequality (Lemma~\ref{lem:khintchine})
to this Rademacher sum gives
\[
  \left(\mathbb{E}\left[\left|\sum_xa_x\sigma_x\right|^k
      \,\middle|\,\text{matching}\right]\right)^{1/k}
  \le C\sqrt{k}
    \left(\sum_{\{u,v\}\text{ matched}}(a_u-a_v)^2\right)^{1/2}.
\]
Since $(a_u-a_v)^2\le2(a_u^2+a_v^2)$ and each point belongs to one pair,
\[
  \sum_{\{u,v\}\text{ matched}}(a_u-a_v)^2
  \le2\sum_{x\in D}a_x^2.
\]
We raise the conditional moment bound to the $k$th power, and note that the bound holds pointwise for every matching, giving
\[
  \mathbb{E}\left[\left|\sum_xa_x\sigma_x\right|^k\right]
  \le\left(C\sqrt{2k}\left(\sum_{x\in D}a_x^2\right)^{1/2}\right)^k.
\]
Taking $k$th roots proves the lemma with $C_0=\sqrt{2}C$.
