<!-- conjecture-1: body_md -->
\label{conj:main}
Fix $\varepsilon>0$. Let $f:\mathbb{N}\to\mathbb{U}$ be multiplicative,
and suppose that $|\alpha-a/q|\leqslant1/(qx)$, where
$(a,q)=1$ and $1\leqslant q\leqslant x/(\log x)^{2+\varepsilon}$.
Then
\begin{equation}\label{eq:conjecture}
 |S_f(x,\alpha)|
 \ll_\varepsilon
 \frac{x}{\log x}+\frac{x}{\sqrt q}.
\end{equation}
<!-- end conjecture-1 -->
<!-- thm-1: body_md -->
\label{thm:main}
Let $f:\mathbb{N}\to\mathbb{U}$ be multiplicative, and suppose that
$|\alpha-a/q|\leqslant1/(qx)$, where $(a,q)=1$ and
$1\leqslant q\leqslant Q_{\rm MV}(x)$. Then
\begin{equation}\label{eq:main-expanded}
 |S_f(x,\alpha)|
 \leqslant
 \left(\frac4{\sqrt6}+o(1)\right)\frac{x}{\sqrt q}
 +O\left(\frac{x}{\log x}\right).
\end{equation}
If $f$ is completely multiplicative, then
\begin{equation}\label{eq:complete-main}
 |S_f(x,\alpha)|
 \leqslant
 (1+o(1))\frac{x}{\sqrt q}
 +O\left(\frac{x}{\log x}\right).
\end{equation}
Both leading constants are best possible, and the
estimates are uniform in $f,a,q$, and $\alpha$.
In particular, Conjecture~\ref{conj:main} holds
for every fixed $\varepsilon>0$.
<!-- end thm-1 -->
<!-- rem-1: body_md -->
Independently, Robles \cite{Robles2026} proved the folklore
conjecture for all $1$-bounded multiplicative functions,
in the wider range $q\leqslant x/(\log x)^2$ under the weaker approximation condition
$|\alpha-a/q|\leqslant q^{-2}$. His result gives the conjectured
upper bound, but does not determine the sharp leading
constants $4/\sqrt6$ and $1$ as in our
Theorem~\ref{thm:main}.

<!-- cor-1: body_md -->
\label{cor:sharp-range}
Suppose that $q=o((\log x)^2)$ and that the hypotheses of
Theorem~\ref{thm:main} hold. Then
\begin{equation}\label{eq:subcritical-general}
 |S_f(x,\alpha)|
 \leqslant
 \left(\frac4{\sqrt6}+o(1)\right)\frac{x}{\sqrt q}.
\end{equation}
For completely multiplicative $f$, the constant
$4/\sqrt6$ can be replaced by $1$.
Both constants are optimal, and neither estimate
can be extended uniformly to $q\asymp(\log x)^2$
with the same leading constant.

<!-- cor-2: body_md -->
\label{cor:logarithmic}
Let $f:\mathbb{N}\to\mathbb{U}$ be multiplicative, and suppose that
$|\alpha-a/q|\leqslant1/(qx)$, where $(a,q)=1$ and
$1\leqslant q\leqslant x$. Then
\begin{equation}\label{eq:logarithmic-corollary}
 \left|
 \sum_{n\leqslant x}\frac{f(n)}n\mathrm e(n\alpha)
 \right|
 \ll \log q+\frac{\log x}{\sqrt q}.
\end{equation}
In particular, conjecture~(1.5) of
\cite{GranvilleLamzouri2026} holds.

Moreover, if
$q=o((\log x/\log\log x)^2)$, then
\begin{equation}\label{eq:general-sharp-log}
 \left|
 \sum_{n\leqslant x}\frac{f(n)}n\mathrm e(n\alpha)
 \right|
 \leqslant
 \left(\frac4{\sqrt6}+o(1)\right)
 \frac{\log x}{\sqrt q}.
\end{equation}
If $f$ is completely multiplicative, the stronger
estimate
\begin{equation}\label{eq:complete-sharp-log}
 \left|
 \sum_{n\leqslant x}\frac{f(n)}n\mathrm e(n\alpha)
 \right|
 \leqslant
 (1+o(1))\frac{\log x}{\sqrt q}
\end{equation}
holds in the same range.

The leading constants $4/\sqrt6$ and $1$ are
best possible for multiplicative and completely
multiplicative functions, respectively.
All estimates are uniform in $f,a,q$, and $\alpha$.

<!-- prop-1: body_md -->
\label{prop:counterexamples}
The following statements hold.

\begin{enumerate}

\item
There are $1$-bounded multiplicative functions $f_j$
and parameters $q_j,x_j\to\infty$, with
$q_j\leqslant(\log x_j)^2$, such that
\[
 |S_{f_j}(x_j,1/q_j)|
 =
 \left(\frac4{\sqrt6}+o(1)\right)
 \frac{x_j}{\sqrt{q_j}}.
\]
Thus the leading constant $1$ in conjecture~(1.3)
is false for general multiplicative functions.

\item
There are squarefree integers $q_j\to\infty$,
real numbers $x_j\geqslant y_j\geqslant q_j^2$ tending to
infinity, reduced numerators $a_j$, and periodic
unimodular multiplicative functions $f_j$ such that
\[
 \frac{
 \left|
 \sum_{\substack{n\leqslant x_j\\P^+(n)\leqslant y_j}}
 f_j(n)\mathrm e(a_jn/q_j)
 \right|
 }{
 \Psi(x_j,y_j)/\sqrt{q_j}
 +\sqrt{x_jy_j}/\log x_j
 }
 \longrightarrow\infty.
\]
Hence conjecture~(1.4) fails in order of magnitude.

\item
For $f\equiv1$ and
$q=\lfloor(\log x/\log\log x)^2\rfloor$, we have
\[
 \left|
 \sum_{n\leqslant x}\frac{\mathrm e(n/q)}n
 \right|
 =(2+o(1))\log\log x.
\]
This exceeds the proposed upper bound with leading
constant $4/\sqrt6$ in conjecture~(1.6).
Thus that conjecture fails at its stated endpoint,
even for completely multiplicative functions.

\end{enumerate}

<!-- prop-2: body_md -->
\label{prop:large-q}
Suppose that $(a,q)=1$, $L^{5/2}\leqslant q\leqslant Q_{\rm MV}(x)$, and
$|\alpha-a/q|\leqslant1/(qx)$. Then
$S_f(x,\alpha)\ll x/L$.
<!-- end prop-2 -->
<!-- proof-2: body_md -->
For these values of $q$, \cite[Theorem~1]{MontgomeryVaughan1977} gives \eqref{eq:MV-intro}. The standard estimate $q/\phi(q)\ll\log\log(3q)$ yields
$x/\sqrt{\phi(q)}\ll x\sqrt{\log\log(3q)/q}=o(x/L)$ when $q\geqslant L^{5/2}$. Lemma~\ref{lem:phase} contributes $O(x/q)=o(x/L)$.

<!-- lem-2: body_md -->
\label{lem:conv}
There is a multiplicative function $h$ such that $f=h*F_q$, with $h(p^k)=0$ for $p\mid q$, $h(p)=0$ for $p\nmid q$, and
$h(p^k)=f(p^k)-f(p)f(p^{k-1})$ for $p\nmid q$ and $k\geqslant2$. In particular,
\begin{equation}\label{eq:h-sums}
 \sum_{d\geqslant1}\frac{|h(d)|}{d}<\infty,
 \qquad
 \sum_{d\geqslant1}\frac{|h(d)|}{d^{1-\delta}}<\infty
 \quad(0<\delta<1/2).
\end{equation}
<!-- end lem-2 -->
<!-- proof-3: body_md -->
Define $h$ multiplicatively by the identity of formal Euler factors
\[
 \sum_{k\geqslant0}\frac{f(p^k)}{p^{ks}}
 =\left(\sum_{k\geqslant0}\frac{h(p^k)}{p^{ks}}\right)
  \left(\sum_{k\geqslant0}\frac{F_q(p^k)}{p^{ks}}\right).
\]
If $p\mid q$, the two factors involving $f$ and $F_q$ are identical, whence $h(p^k)=0$ for every $k\geqslant1$. Suppose that $p\nmid q$. Since $F_q(p^k)=f(p)^k$, comparison of the coefficient of $p^{-s}$ gives $h(p)=0$. For $k\geqslant2$, the coefficient identity reads
$f(p^k)=h(p^k)+f(p)h(p^{k-1})+\cdots+f(p)^k$. Using the same identity at level $k-1$ and multiplying it by $f(p)$, all terms except the first cancel, and therefore
$h(p^k)=f(p^k)-f(p)f(p^{k-1})$.

It follows that $|h(p^k)|\leqslant2$, and that every integer in the support of $h$, apart from $1$, is powerful and coprime to $q$. Hence
\[
 \sum_{d\geqslant1}\frac{|h(d)|}{d}
 \leqslant\prod_{p\nmid q}\left(1+2\sum_{k\geqslant2}\frac1{p^k}\right)
 \leqslant\prod_p\left(1+\frac{2}{p(p-1)}\right)<\infty.
\]
More generally, if $0<\delta<1/2$, then
$\sum_{k\geqslant2}p^{-k(1-\delta)}\ll p^{-2(1-\delta)}$, and the corresponding Euler product converges because $2(1-\delta)>1$. This proves both assertions in \eqref{eq:h-sums}.

<!-- prop-3: body_md -->
\label{prop:reduction}
Suppose that, whenever $X$ is large, $q\leqslant8(\log X)^3$, and
$F:\mathbb{N}\to\mathbb{U}$ is multiplicative and completely multiplicative at every
prime $p\nmid q$, one has
\begin{equation}\label{eq:local-goal}
 S_F(X,\theta)\ll \frac{X}{\log X}+\frac{X}{\sqrt q}
\end{equation}
for $|\theta-b/q|\leqslant1/(qX)$ and $(b,q)=1$. Then the same estimate
holds, under the same approximation hypothesis, for every multiplicative
$f:\mathbb{N}\to\mathbb{U}$ whenever $q\leqslant(\log x)^3$.
<!-- end prop-3 -->
<!-- proof-4: body_md -->
Expanding the convolution $f=h*F_q$ and changing the order of summation gives
\begin{equation}\label{eq:conv-sum}
 S_f(x,\alpha)=\sum_{\substack{d\leqslant x\\(d,q)=1}}h(d)S_{F_q}(x/d,d\alpha).
\end{equation}
The restriction $(d,q)=1$ follows from Lemma~\ref{lem:conv}.
Since $(d,q)=1$, $ad$ is also coprime to $q$, and
\[
 d\alpha=\frac{ad}{q}+d(\alpha-a/q),
 \qquad |d(\alpha-a/q)|\leqslant\frac{d}{qx}=\frac1{q(x/d)}.
\]
Thus every inner sum in \eqref{eq:conv-sum} has a rational approximation with the same denominator $q$. This is the point of the definition \eqref{eq:Fq}.

We split the $d$-sum at $\sqrt x$. If $d\leqslant\sqrt x$, then
$x/d\geqslant\sqrt x$ and $\log x\leqslant2\log(x/d)$. Consequently
$q\leqslant(\log x)^3\leqslant8(\log(x/d))^3$, so \eqref{eq:local-goal}
applies. Since $\log(x/d)\asymp\log x$ in this range,
\begin{align*}
 \sum_{d\leqslant\sqrt x}|h(d)|\left(\frac{x/d}{\log(x/d)}+\frac{x/d}{\sqrt q}\right)
 &\ll \left(\frac{x}{\log x}+\frac{x}{\sqrt q}\right)
       \sum_{d\geqslant1}\frac{|h(d)|}{d}\\
 &\ll \frac{x}{\log x}+\frac{x}{\sqrt q}.
\end{align*}
For $d>\sqrt x$ we use $|S_{F_q}(x/d,d\alpha)|\leqslant x/d$.
Taking $\delta=1/4$ in \eqref{eq:h-sums},
\[
 x\sum_{d>\sqrt x}\frac{|h(d)|}{d}
 \leqslant x^{7/8}\sum_{d\geqslant1}\frac{|h(d)|}{d^{3/4}}
 \ll x^{7/8}=o(x/\log x).
\]
Combining the two ranges proves the proposition.

<!-- prop-4: body_md -->
\label{prop:large-prime}
Uniformly for $q\leqslant8L^3$ and $(a,q)=1$,
\begin{equation}\label{eq:large-prime}
 \sum_{\substack{n\leqslant x\\P^{+}(n)>x/q}}F(n)\mathrm e(an/q)\ll \frac{x}{\log x}.
\end{equation}
<!-- end prop-4 -->
<!-- proof-5: body_md -->
Let $n\leqslant x$ satisfy $P^{+}(n)>x/q$. Since $x/q>\sqrt x$, the largest prime factor of $n$ occurs to the first power and is the unique prime factor of $n$ exceeding $x/q$. Thus $n=pm$ uniquely, where $p>x/q$ is prime and $m<x/p<q$. In particular $p\nmid m$, so ordinary multiplicativity gives $F(pm)=F(p)F(m)$.

For $1\leqslant k<q$, put $I_k=(x/(k+1),x/k]$ and
$A_k(b):=\sum_{m\leqslant k}F(m)\mathrm e(abm/q)$. The condition $p\in I_k$ is equivalent to $\lfloor x/p\rfloor=k$, and therefore
\begin{equation}\label{eq:large-prime-decomp}
 \sum_{\substack{n\leqslant x\\P^{+}(n)>x/q}}F(n)\mathrm e(an/q)
 =\sum_{k<q}\sum_{p\in I_k}F(p)A_k(p).
\end{equation}
The length of $I_k$ is
$|I_k|=x/(k(k+1))\asymp x/k^2$. Since $k<q\leqslant8L^3$, we have
$|I_k|/q\geqslant x/q^3=x^{1-o(1)}$. The short-interval
Brun--Titchmarsh inequality bounds the primes in an interval of length
$H>q$ and a reduced class by
$\ll H/\{\phi(q)\log(H/q)\}$. Applying this to $I_k$ as in
\cite[(3.20)]{GranvilleLamzouri2026} gives
\begin{equation}\label{eq:BT}
 \#\{p\in I_k:p\equiv b\pmod q\}
 \ll \frac{x}{\phi(q)k^2\log x}
\end{equation}
for every reduced residue class $b\pmod q$, uniformly in $k<q$. Here the logarithm in Brun--Titchmarsh is $\log(|I_k|/q)\asymp\log x$.

Define
$\mathcal E_q(k):=\phi(q)^{-1}\sum_{b\bmod q,(b,q)=1}|A_k(b)|^2$. Grouping the primes in the inner sum of \eqref{eq:large-prime-decomp} by residue class, and then using \eqref{eq:BT}, gives
\begin{align*}
 \left|\sum_{p\in I_k}F(p)A_k(p)\right|
 &\leqslant\sum_{\substack{b\bmod q\\(b,q)=1}}|A_k(b)|
       \#\{p\in I_k:p\equiv b\pmod q\}\\
 &\ll \frac{x}{\phi(q)k^2\log x}
       \sum_{\substack{b\bmod q\\(b,q)=1}}|A_k(b)|\\
 &\leqslant \frac{x}{k^2\log x}\sqrt{\mathcal E_q(k)}.
\end{align*}
Thus each $k$-block is bounded by $x\sqrt{\mathcal E_q(k)}/(k^2\log x)$.

We estimate $\mathcal E_q(k)$ in two different ways. Enlarging the average from the reduced residue classes to all additive frequencies and applying Parseval gives
\[
 \mathcal E_q(k)
 \leqslant\frac1{\phi(q)}\sum_{b\bmod q}|A_k(b)|^2
 =\frac q{\phi(q)}\sum_{m\leqslant k}|F(m)|^2
 \leqslant\frac q{\phi(q)}k.
\]
For small $k$ one can do better. Expanding the square over reduced residue classes yields the exact identity
\[
 \mathcal E_q(k)=\frac1{\phi(q)}
 \sum_{m,n\leqslant k}F(m)\overline{F(n)}c_q(m-n).
\]
The diagonal contributes at most $k$, while for each $1\leqslant h<k$ there are at most $k$ pairs with $m-n=h$ and at most $k$ with $n-m=h$. Hence
\begin{equation}\label{eq:energy-ramanujan}
 \mathcal E_q(k)\leqslant k+2k\sum_{1\leqslant h<k}\frac{|c_q(h)|}{\phi(q)}.
\end{equation}
The prime-power formula for Ramanujan sums implies
\begin{equation}\label{eq:ram-support}
 \frac{|c_q(h)|}{\phi(q)}=
 \begin{cases}
  \phi((u,\operatorname{rad}(q)))/\phi(\operatorname{rad}(q)),&h=\frac{q}{\operatorname{rad}(q)}u,\\
  0,&\frac{q}{\operatorname{rad}(q)}\nmid h.
 \end{cases}
\end{equation}
Indeed, for $p^a\Vert q$ the local factor vanishes unless $p^{a-1}\mid h$; after writing $h=(q/\operatorname{rad}(q))u$, it is $1$ if $p\mid u$ and $1/(p-1)$ otherwise. Therefore,
\[
 \sum_{1\leqslant h<k}\frac{|c_q(h)|}{\phi(q)}
 \leqslant\frac1{\phi(\operatorname{rad}(q))}\sum_{u\leqslant k\operatorname{rad}(q)/q}\phi((u,\operatorname{rad}(q)))
 \leqslant\frac{k^2}{\phi(\operatorname{rad}(q))}.
\]
If $k\leqslant(q/\phi(q))^2$, then
\[
 \frac{k^2}{\phi(\operatorname{rad}(q))}\leqslant\frac{(q/\phi(q))^4}{\phi(\operatorname{rad}(q))}
 =\prod_{p\mid q}\frac{p^4}{(p-1)^5}\leqslant\frac{81}{2}.
\]
The last inequality holds because each local factor is less than $1$ for $p\geqslant5$, while the factors at $2$ and $3$ have product $81/2$. Substitution in \eqref{eq:energy-ramanujan} gives
$\mathcal E_q(k)\leqslant82k$ for $k\leqslant(q/\phi(q))^2$.

We now sum the blocks. For $k\leqslant(q/\phi(q))^2$ use the short estimate, and for larger $k$ use Parseval:
\begin{align*}
 \sum_{k<q}\frac{\sqrt{\mathcal E_q(k)}}{k^2}
 &\ll\sum_{k\leqslant(q/\phi(q))^2}k^{-3/2}
   +\sqrt{q/\phi(q)}\sum_{k>(q/\phi(q))^2}k^{-3/2}\\
 &\ll1.
\end{align*}
Summing the preceding block estimate over $k<q$ proves \eqref{eq:large-prime}.

\label{lem:shiu-weight}
Uniformly for $0\leqslant z\leqslant1$ and $u\geqslant1$,
\begin{equation}\label{eq:shiu-total}
 \sum_{m\leqslant u}w_z(m)\ll u\rho_z(u).
\end{equation}
If $u\geqslant q^4$, then uniformly for every $b\pmod q$,
\begin{equation}\label{eq:shiu-progression}
 \sum_{\substack{m\leqslant u\\m\equiv b\pmod q}}w_z(m)
 \ll\frac uq\rho_z(u).
\end{equation}
<!-- end lem-3 -->
<!-- proof-6: body_md -->
Shiu's Brun--Titchmarsh theorem for non-negative multiplicative
functions \cite{Shiu1980}, combined with Mertens' estimate, gives
\[
 \sum_{m\leqslant u}w_z(m)
 \ll\frac{u}{\log(2u)}
   \exp\left(\sum_{p\leqslant u}\frac{w_z(p)}p\right)
 \ll u\rho_z(u).
\]
For small $u\ll T$ the same assertion follows trivially, since
$w_z(m)\leqslant1$ and $\rho_z(u)\asymp1$.
For the progression estimate, let $d=(b,q)$, $q_1=q/d$ and $u_1=u/d$.
Dividing the congruence by $d$ gives a reduced class modulo $q_1$.
Every prime dividing $d$ is below $T$, so $w_z(dm)=w_z(m)$.
Moreover $u\geqslant q^4$ ensures $q_1\leqslant u_1^{1/4}$.
Shiu's theorem, with the prime factors of $q_1$ omitted from its
Euler product, gives
\[
 \sum_{\substack{m\leqslant u_1\\m\equiv b/d\pmod{q_1}}}w_z(m)
 \ll\frac{u_1}{\phi(q_1)}\rho_z(u_1)
       \prod_{p\mid q_1}\left(1-\frac1p\right).
\]
Here $w_z(p)=1$ for $p\mid q_1$; the final product cancels
$q_1/\phi(q_1)$. Since $d\leqslant q\leqslant u^{1/4}$,
$\rho_z(u/d)\ll\rho_z(u)$ uniformly in $z$.
We obtain $u_1\rho_z(u_1)/q_1\ll u\rho_z(u)/q$.

\label{lem:general-primary-kernel}
Uniformly in $\chi\pmod q$ and $(a,q)=1$,
\begin{equation}\label{eq:general-primary-kernel}
 \sum_{d\in\mathcal D_q}\frac{|G_q(ad,\overline\chi)|}{d}
 \leqslant\frac4{\sqrt6}\sqrt q.
\end{equation}
For each fixed $0<\rho<1/2$,
\begin{equation}\label{eq:general-primary-weighted-kernel}
 \sum_{d\in\mathcal D_q}\frac{|G_q(ad,\overline\chi)|}{d^{1-\rho}}
 \ll_\rho q^{1/2+o(1)}.
\end{equation}
If $f$ is completely multiplicative and $|f(n)|\leqslant1$, then, for
all real $t$,
\begin{equation}\label{eq:complete-primary-kernel}
 \left|\sum_{d\in\mathcal D_q}
 \frac{f(d)G_q(ad,\overline\chi)}{d^{1+it}}\right|
 \leqslant\sqrt q.
\end{equation}
<!-- end lem-10 -->
<!-- proof-15: body_md -->
We prove the estimates prime by prime. Factor $q=\prod_{p^e\Vert q}p^e$ and let
$\chi_p$ denote the component of $\chi$ modulo $p^e$.
The Chinese remainder theorem gives a constant $\omega(a,\chi)$ of
modulus one for which
\begin{equation}\label{eq:primary-CRT-factor}
 \sum_{d\in\mathcal D_q}\frac{f(d)G_q(ad,\overline\chi)}{d^{1+it}}
 =\omega(a,\chi)
 \prod_{p^e\Vert q}\left(\sum_{v\geqslant0}
  \frac{f(p^v)\chi^{(p)}(p)^v}{p^{v(1+it)}}
  G_{p^e}(p^v,\overline{\chi_p})\right),
\end{equation}
where $\chi^{(p)}=\prod_{\substack{r^b\Vert q\\r\ne p}}\chi_r$;
thus $|\chi^{(p)}(p)|=1$. More explicitly, the CRT expression for
$G_q$ has local additive coefficient $ad(q/p^e)^{-1}$; its unit part
can be removed from the local Gauss sum and contributes precisely the
character factors displayed in \eqref{eq:primary-CRT-factor}.
The factorization also holds after taking absolute values.

For a local character of conductor $p^j$ with $1\leqslant j\leqslant e$,
$G_{p^e}(p^v,\overline{\chi_p})$ vanishes unless $v=e-j$, in which case
its modulus is $p^{e-j/2}$. The corresponding local contribution
to \eqref{eq:general-primary-kernel} is $p^{j/2}\leqslant p^{e/2}$.
For a principal local character the relevant Ramanujan sums are
\[
 c_{p^e}(p^{e-1})=-p^{e-1},\qquad
 c_{p^e}(p^v)=\phi(p^e)\quad(v\geqslant e).
\]
Consequently the local absolute norm equals $2$.

contributions multiply to $4/\sqrt6$.

where $x_0$ is an absolute threshold for Proposition~\ref{prop:basic-bound}.
For $U\leqslant t\leqslant x$, one has $q\leqslant Q_{\rm MV}(t)$ and
$|\alpha-a/q|\leqslant1/(qt)$, so Proposition~\ref{prop:basic-bound} gives
\[
 |S_f(t,\alpha)|\ll\frac{t}{\log t}+\frac{t}{\sqrt q}.
\]
Using $|S_f(t,\alpha)|\leqslant t$ below $U$, partial summation yields
\[
 \left|\sum_{n\leqslant x}\frac{f(n)}n\mathrm e(n\alpha)\right|
 \ll \log U+\log\log x+\frac{L}{\sqrt q}
 \ll \log q+\frac{L}{\sqrt q}.
\]
For the last inequality, note that $\log U\leqslant\log q+O(\log\log x)$, and
$\log\log x\ll\log q+L/\sqrt q$: use $L/\sqrt q\geqslant\log\log x$ if
$q\leqslant(L/\log\log x)^2$, and $\log q\gg\log\log x$ otherwise. If
$Q_{\rm MV}(x)<q\leqslant x$, then $\log q\asymp L$, and the trivial
harmonic bound proves the same assertion. This establishes
\eqref{eq:logarithmic-corollary}.

For the sharp estimates, put $C_0=4/\sqrt6$ in the general case and
$C_0=1$ in the completely multiplicative case. Under the subcritical
hypothesis, choose $T$ by
\[
 \log T=\left(\frac{L\log\log x}{\sqrt q}\right)^{1/2}.
\]
Then $q\leqslant Q_{\rm MV}(T)$ and
\[
 \log T=o(L/\sqrt q),\qquad
 \log\log x=o(L/\sqrt q).
\]
Theorem~\ref{thm:main} therefore implies,
uniformly for $T\leqslant t\leqslant x$,
\[
 |S_f(t,\alpha)|\leqslant(C_0+o(1))\frac{t}{\sqrt q}
          +O\left(\frac{t}{\log t}\right).
\]
The initial segment $n\leqslant T$ contributes $O(\log T)$; partial
summation on $(T,x]$ contributes at most
\[
 (C_0+o(1))\frac{L}{\sqrt q}+O(\log\log x).
\]
This proves \eqref{eq:general-sharp-log} and
\eqref{eq:complete-sharp-log}. For optimality, the periodic construction
in Proposition~\ref{prop:counterexamples}\emph{(i)} gives the
coefficient $4/\sqrt6$ after partial summation; primitive Dirichlet
characters give the coefficient $1$ in the completely multiplicative
case. A direct treatment of the logarithmic sums and further discussion appear in
\cite{GouLogarithmic2026}.

 (g(0),g(1),\ldots,g(5))=(1,1,-1,-1,-1,1),

Write
$\tau(\chi):=\sum_{v\bmod\ell}\chi(v)\mathrm e(v/\ell)$ for the primitive
Gauss sum; then $|\tau(\chi)|=\sqrt\ell$.

\[
 C_q:=\sum_{b\bmod q}f(b)\mathrm e(b/q).

\[
 C_q=
 \left(\sum_{u\bmod6}g(u)\mathrm e(su/6)\right)
 \left(\sum_{v\bmod\ell}\chi(v)\mathrm e(tv/\ell)\right).

 C_q=4\chi(6)\tau(\chi),\qquad |C_q|=4\sqrt\ell.

 \sum_{n\leqslant x}f(n)\mathrm e(n/q)

Let $q$ be squarefree, assume $P^+(q)\leqslant y$, and define
\[
 f_q(n):=\mu((n,q)).
\]
This is a periodic, unimodular multiplicative function. For
$a\in(\mathbb{Z}/q\mathbb{Z})^\times$, write
\[
 S_q(a;x,y):=
 \sum_{\substack{n\leqslant x\\P^+(n)\leqslant y}}
 f_q(n)\mathrm e(an/q).
\]
Averaging over the reduced numerators gives the exact identity
\begin{equation}\label{eq:counter-average}
 \frac1{\phi(q)}\sum_{a\bmod q}^{*}S_q(a;x,y)
 =\mu(q)M_q(x,y),
\end{equation}
where
\begin{equation}\label{eq:counter-Mq}
 M_q(x,y):=\frac1{\phi(q)}
 \sum_{r\mid q}
 \left(\prod_{p\mid r}(p-2)\right)\Psi(x/r,y)>0.
\end{equation}
Indeed, if $d=(n,q)$, then the squarefreeness of $q$ gives
\[
 c_q(n)=\mu(q/d)\phi(d),
 \qquad \mu(d)c_q(n)=\mu(q)\phi(d).
\]
Also
\[
 \phi((n,q))
 =\prod_{p\mid q}\bigl(1+(p-2)\mathbf1_{p\mid n}\bigr).
\]
Expanding the last product and summing over the $y$-smooth integers
gives \eqref{eq:counter-Mq}.  In particular,
\begin{equation}\label{eq:counter-max-a}
 \max_{(a,q)=1}|S_q(a;x,y)|\geqslant M_q(x,y).
\end{equation}
Averaging in \eqref{eq:counter-average} is essential: the corresponding identity need not hold for a fixed numerator $a$.

Choose $x=x(y)$ by
\begin{equation}\label{eq:counter-saddle-x}
 \log x=\sum_{p\leqslant y}\frac{\log p}{\sqrt p-1}.
\end{equation}
Thus the Hildebrand--Tenenbaum saddle point is $1/2$. Their saddle-point
theorem \cite[Theorems~1 and~3]{HildebrandTenenbaum1986} and its
fixed-dilation consequence give, for each fixed $q$ and each $r\mid q$,
\begin{equation}\label{eq:counter-local-ratio}
 \frac{\Psi(x/r,y)}{\Psi(x,y)}
 =r^{-1/2}(1+o_q(1))
 \qquad(y\to\infty).
\end{equation}
For completeness, let $\sigma_r$ be the saddle point associated with
$(x/r,y)$. The mean-value theorem in the saddle equation gives
\[
 \sigma_r=\frac12+O_q\!\left(
  \left\{\sum_{p\leqslant y}
  \frac{p^{1/2}(\log p)^2}{(p^{1/2}-1)^2}\right\}^{-1}
 \right)=\frac12+o_q(1).
\]
A Taylor expansion of the two saddle-point formulae, first in the
$\log x$ variable and then in the saddle, therefore gives
$\log\Psi(x/r,y)-\log\Psi(x,y)=-(\log r)/2+o_q(1)$, which is
\eqref{eq:counter-local-ratio}. This is the framework recalled in
\cite[\S3.1]{GranvilleLamzouri2026}; crucially, $q$ is fixed before
$y$ tends to infinity. It follows that
\begin{align}
 M_q(x,y)
 &=(1+o_q(1))\frac{\Psi(x,y)}{\phi(q)}
   \prod_{p\mid q}\left(1+\frac{p-2}{\sqrt p}\right) \notag\\
 &=(1+o_q(1))\frac{\Psi(x,y)}{\sqrt q}\mathfrak A(q),
 \label{eq:counter-Aq-main}
\end{align}
where
\begin{equation}\label{eq:counter-Aq}
 \mathfrak A(q):=
 \prod_{p\mid q}\left(1+\frac1{\sqrt p+1}\right).
\end{equation}
The second equality uses
\[
 \frac{\sqrt p}{p-1}\left(1+\frac{p-2}{\sqrt p}\right)
 =1+\frac1{\sqrt p+1}.
\]

The same saddle-point formula, combined with the prime number theorem,
gives $\log x=(2+o(1))\sqrt y$ and
\[
 \log\left(\frac{\Psi(x,y)\log x}{\sqrt{xy}}\right)
 =\left(2+o(1)\right)\frac{\sqrt y}{\log y}.
\]
Consequently, for every fixed $q$,
\begin{equation}\label{eq:counter-secondary-small}
 \frac{\sqrt{xy}}{\log x}
 =o\left(\frac{\Psi(x,y)}{\sqrt q}\right).
\end{equation}

Finally take the primorials $q_z:=\prod_{p\leqslant z}p$. The prime number
theorem gives
\begin{align*}
 \log\mathfrak A(q_z)
 &=\sum_{p\leqslant z}\log\left(1+\frac1{\sqrt p+1}\right)\\
 &=\sum_{p\leqslant z}\left(\frac1{\sqrt p}+O\left(\frac1p\right)\right)
 =\left(2+o(1)\right)\frac{\sqrt z}{\log z}\longrightarrow\infty.
\end{align*}
Choose $z_j\to\infty$ and set $q_j=q_{z_j}$.  With $q_j$ fixed,
choose $y_j\geqslant q_j^2$ sufficiently large, define $x_j$ by
\eqref{eq:counter-saddle-x}, and then choose a reduced numerator $a_j$
using \eqref{eq:counter-max-a}.  Equations
\eqref{eq:counter-Aq-main} and \eqref{eq:counter-secondary-small} give
\[
 \frac{|S_{q_j}(a_j;x_j,y_j)|}
 {\Psi(x_j,y_j)/\sqrt{q_j}+\sqrt{x_jy_j}/\log x_j}
 \longrightarrow\infty.
\]
This disproves \cite[(1.4)]{GranvilleLamzouri2026}.  The order of these choices matters: we use the smooth-number asymptotic only with $q$ fixed, without assuming uniformity as $q$ grows.  Since $f_q(p^2)=f_q(p)=-1$ for
$p\mid q$, this example is not completely multiplicative and does not
settle the completely multiplicative version of that problem.

\label{prop:critical-counterexample}
Let $0<c\leqslant1$ be fixed. There exist arbitrarily large $x$, integers
$q$ satisfying
\[
 q=c(\log x)^2,
\]
and $1$-bounded multiplicative functions $f$ such that, with
$\alpha=1/q$,
\begin{equation}\label{eq:critical-counterexample}
 \left|
 \sum_{n\leqslant x}f(n)\mathrm e(n/q)
 \right|
 \geqslant
 \left(
 \frac4{\sqrt6}+\frac{\sqrt c}{2}+o(1)
 \right)\frac{x}{\sqrt q}.
\end{equation}
In particular, the estimate
\[
 \left|
 \sum_{n\leqslant x}f(n)\mathrm e(n\alpha)
 \right|
 \leqslant
 \left(\frac4{\sqrt6}+o(1)\right)\frac{x}{\sqrt q}
\]
cannot hold uniformly for all $1$-bounded multiplicative $f$ throughout
the range $q\leqslant(\log x)^2$.

Let $\ell>3$ tend to infinity through the primes, and take $q=6\ell$ and
\[
 x=\exp\left(\sqrt{\frac q c}\right).
\]
Thus $q=c(\log x)^2$. Let $\chi$ be a non-principal character modulo
$\ell$; since $\ell$ is prime, $\chi$ is primitive.

Define the multiplicative function $g$ on prime powers by
\[
 g(2^k)=g(3^k)=-1\quad(k\geqslant1),
 \qquad
 g(p^k)=1\quad(p>3,\ k\geqslant1).
\]
Then $g$ is periodic modulo $6$, with
\[
 (g(0),g(1),\ldots,g(5))=(1,1,-1,-1,-1,1).
\]
The multiplicative function
\[
 f_0(n)=g(n)\chi(n)
\]
is therefore periodic modulo $q$.

Let $\overline{\ell}$ denote the inverse of $\ell$ modulo $6$, and
$\overline6$ the inverse of $6$ modulo $\ell$. By the Chinese
remainder theorem,
\begin{align*}
 \sum_{b\bmod q}f_0(b)\mathrm e(b/q)
 &=
 \left(
 \sum_{u\bmod6}g(u)\mathrm e(u\overline{\ell}/6)
 \right)
 \left(
 \sum_{v\bmod\ell}\chi(v)\mathrm e(v\overline6/\ell)
 \right).
\end{align*}
Since $\overline{\ell}\equiv\pm1\pmod6$, the first factor is
\[
 1+\mathrm e(1/6)-\mathrm e(1/3)-\mathrm e(1/2)-\mathrm e(2/3)+\mathrm e(5/6)=4,
\]
up to complex conjugation. The second factor is a Gauss sum of absolute
value $\sqrt\ell$. Hence
\[
 \left|
 \sum_{b\bmod q}f_0(b)\mathrm e(b/q)
 \right|
 =4\sqrt\ell.
\]
Using periodicity,
\[
 \sum_{n\leqslant x}f_0(n)\mathrm e(n/q)
 =
 \frac{x}{q}
 \sum_{b\bmod q}f_0(b)\mathrm e(b/q)+O(q),
\]
and consequently
\begin{equation}\label{eq:f0-major}
 \left|
 \sum_{n\leqslant x}f_0(n)\mathrm e(n/q)
 \right|
 =
 \left(\frac4{\sqrt6}+o(1)\right)\frac{x}{\sqrt q},
\end{equation}
since $q^{3/2}=o(x)$.

We next show that primes in $(x/2,x]$ make a negligible contribution to
\eqref{eq:f0-major}. For such primes $p$ one has $g(p)=1$, and hence
\[
 \sum_{x/2<p\leqslant x}f_0(p)\mathrm e(p/q)
 =
 \sum_{x/2<p\leqslant x}\chi(p)\mathrm e(p/q).
\]
Since $q\ll(\log x)^2$, the Siegel--Walfisz theorem, applied after
splitting the primes into residue classes modulo $q$, gives
\begin{align*}
 \sum_{x/2<p\leqslant x}\chi(p)\mathrm e(p/q)
 &=
 \frac{Li(x)-Li(x/2)}{\phi(q)}
 \sum_{\substack{b\bmod q\\(b,q)=1}}
 \chi(b)\mathrm e(b/q)
 +O_A\left(\frac{x}{(\log x)^A}\right)
\end{align*}
for every fixed $A>0$.

The finite sum again factors by the Chinese remainder theorem:
\begin{align*}
 \sum_{\substack{b\bmod q\\(b,q)=1}}
 \chi(b)\mathrm e(b/q)
 &=
 \left(
 \sum_{u\in(\mathbb{Z}/6\mathbb{Z})^\times}
 \mathrm e(u\overline{\ell}/6)
 \right)
 \left(
 \sum_{v\bmod\ell}\chi(v)\mathrm e(v\overline6/\ell)
 \right).
\end{align*}
The first factor has absolute value $1$, while the second has absolute
value $\sqrt\ell$. Since $\phi(q)=2(\ell-1)$, it follows that
\begin{equation}\label{eq:large-primes-f0}
 \sum_{x/2<p\leqslant x}f_0(p)\mathrm e(p/q)
 =
 o\left(\frac{x}{\sqrt q}\right).
\end{equation}

Consequently, if
\[
 S_*=
 \sum_{\substack{n\leqslant x\\
 n\notin\{p:x/2<p\leqslant x\}}}
 f_0(n)\mathrm e(n/q),
\]
then \eqref{eq:f0-major} and \eqref{eq:large-primes-f0} imply
\begin{equation}\label{eq:Sstar}
 |S_*|
 =
 \left(\frac4{\sqrt6}+o(1)\right)\frac{x}{\sqrt q}.
\end{equation}
For sufficiently large $x$ this is non-zero; write
$S_*=\omega|S_*|$ with $|\omega|=1$.

We now modify $f_0$ only at primes $p\in(x/2,x]$. Define a
multiplicative function $f$ on prime powers by
\[
 f(p)=\omega\mathrm e(-p/q),
 \qquad
 f(p^k)=f(p)^k
 \quad
 \left(\frac x2<p\leqslant x,\ k\geqslant2\right),
\]
and retain
\[
 f(p^k)=f_0(p^k)
\]
at every other prime power. This defines a $1$-bounded multiplicative
function.

If $p>x/2$, then $2p>x$ and $p^2>x$. Thus, among the integers
$n\leqslant x$, changing the value at such a prime affects only the term
$n=p$. Therefore
\begin{align*}
 \sum_{n\leqslant x}f(n)\mathrm e(n/q)
 &=
 S_*+
 \sum_{x/2<p\leqslant x}f(p)\mathrm e(p/q)\\
 &=
 \omega\left(
 |S_*|+\pi(x)-\pi(x/2)
 \right).
\end{align*}
By the prime number theorem,
\[
 \pi(x)-\pi(x/2)
 =
 \left(\frac12+o(1)\right)\frac{x}{\log x}
 =
 \left(\frac{\sqrt c}{2}+o(1)\right)\frac{x}{\sqrt q}.
\]
Combining this with \eqref{eq:Sstar} proves
\eqref{eq:critical-counterexample}.

\label{rem:critical-counterexample}
The construction exhibits two independent sources of large values.
The first comes from the Gauss sums at the primes $2$ and
$3$, which contribute
\[
 \frac4{\sqrt6}\frac{x}{\sqrt q}.
\]
The second comes from the unrestricted values of a multiplicative
function at primes $p>x/2$, and contributes
\[
 \left(\frac12+o(1)\right)\frac{x}{\log x}.
\]
At the critical scale $q\sim c(\log x)^2$ these two quantities have the
same order of magnitude and can be arranged to have the same phase.
Thus any sharp critical-scale estimate of the form
\[
 \left|
 \sum_{n\leqslant x}f(n)\mathrm e(n\alpha)
 \right|
 \leqslant
 \bigl(C(c)+o(1)\bigr)\frac{x}{\sqrt q}
\]
must satisfy
\[
 C(c)\geqslant\frac4{\sqrt6}+\frac{\sqrt c}{2}.
\]
In particular, at the endpoint $q\sim(\log x)^2$ one necessarily has
\[
 C(1)\geqslant\frac4{\sqrt6}+\frac12.
\]
The large-prime contribution is $o(x/\sqrt q)$ when
$q=o((\log x)^2)$, in agreement with
Corollary~\ref{cor:sharp-range}.

A similar construction shows that the range in Corollary~\ref{cor:sharp-range} is also optimal for completely multiplicative functions. Take $q$ prime and use a
primitive character modulo $q$ as the initial completely multiplicative
function. Modifying its values at primes in $(x/2,x]$ and then extending
completely multiplicatively gives
\[
 \left|\sum_{n\leqslant x}f(n)\mathrm e(n/q)\right|
 \geqslant\left(1+\frac{\sqrt c}{2}+o(1)\right)\frac{x}{\sqrt q}
\]
when $q\sim c(\log x)^2$.
