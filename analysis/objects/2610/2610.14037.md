<!-- theorem-1: body_md -->
\label{thm:weighted}
For every integer $d\ge1$, there is a constant $c_d>0$ such that, for every
finite positive Borel measure $\mu$ with compact support satisfying
$\operatorname{diam}(\operatorname{supp}\mu)\le1$ and every $0<\varepsilon<1$,
\begin{equation}
  \mathcal N_\varepsilon(\mu)
  \ge
  c_d\varepsilon^{(d-1)/2}\mathcal A_\varepsilon(\mu).
  \label{eq:weighted-main-bound}
\end{equation}
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{cor:main}
For every integer $d\ge1$, there is a constant $c_d>0$ such that, for every
finite $P\subset\mathbb R^d$ with $\operatorname{diam}(P)\le1$ and every $0<\varepsilon<1$,
\begin{equation}
  \mathcal N_\varepsilon(P)
  \ge
  c_d\varepsilon^{(d-1)/2}\mathcal A_\varepsilon(P).
  \label{eq:main-bound}
\end{equation}
<!-- end corollary-1 -->
<!-- lemma-1: body_md -->
\label{lem:biconvex}
Suppose that $K$ is biconvex and that there is $R>0$ such that
\begin{equation}
  |E|\,\#X\le R
  \label{eq:rectangle-condition}
\end{equation}
whenever $E\subset I$ is measurable, $X\subset\mathcal J$, and $K=1$
on $E\times X$.  Then
\[
  T_Kc(s):=\sum_{a\in\mathcal J}c_aK(s,a)
\]

  \label{eq:biconvex-norm}
\end{equation}

Set $d_a:=|I_a|$ and $n(s):=\#\mathcal J_s$, and discard columns with
$d_a=0$.

\emph{Column tails.}
Fix $a\in\mathcal J$ and an integer $k\ge1$, and put
$\ell=\lceil k/2\rceil$.  Let $B_a^-$ be the block of $\ell$
consecutive elements ending at $a$, when it exists, and let $B_a^+$ be
the block of $\ell$ consecutive elements beginning at $a$, when it
exists.  Any interval in $\mathcal J$ of cardinality at least $k$ that
contains $a$ has at least $\ell$ elements on one side of $a$, counting
$a$ itself, and therefore contains at least one of the two existing
blocks $B_a^-$ and $B_a^+$.  For either existing block $B$, the set
\[
  E_B:=\{s\in I:B\subset\mathcal J_s\}
\]
forms an all-one rectangle $E_B\times B$.  By
\eqref{eq:rectangle-condition},
$|E_B|\le R/\ell\le2R/k$.  Taking the union of the two possibilities and
also using the trivial bound by $d_a$, we obtain
\begin{equation}
  \big|\{s\in I_a:n(s)\ge k\}\big|
  \le
  \min\left(d_a,\frac{4R}{k}\right).
  \label{eq:column-tail}
\end{equation}

\emph{Row tails.}
Conversely, fix $s\in I$ and $r>0$.  For $a\in\mathcal J_s$, let
$L_a$ and $R_a$ be the lengths of the portions of $I_a$ to the left and
right of $s$.  Since a point has measure zero,
$L_a+R_a=d_a$.  Thus, if $d_a\ge r$, then either $L_a\ge r/2$ or
$R_a\ge r/2$.

Let $0<r'<r$ and set
\[
  E_s^-:=(s-r'/2,s),
  \qquad
  E_s^+:=(s,s+r'/2).
\]
If $L_a\ge r/2$, then $E_s^-\subset I_a$; in particular, this interval
lies inside the ambient interval $I$.  Similarly, if $R_a\ge r/2$, then
$E_s^+\subset I_a$.  Split the relevant columns according to these two
possibilities, assigning a column arbitrarily if both occur.  For each
nonempty class, the corresponding fixed interval, together with that class
of columns, is an all-one rectangle.  Since each fixed interval has length
$r'/2$, applying \eqref{eq:rectangle-condition} to the two classes gives
\[
  \#\{a\in\mathcal J_s:d_a\ge r\}
  \le
  \min\left(n(s),\frac{4R}{r'}\right).
\]
Letting $r'\uparrow r$ yields
\begin{equation}
  \#\{a\in\mathcal J_s:d_a\ge r\}
  \le
  \min\left(n(s),\frac{4R}{r}\right).
  \label{eq:row-tail}
\end{equation}

\emph{Layer-cake estimates.}
For $t>0$, take $k=\lfloor t^2\rfloor+1$ in
\eqref{eq:column-tail}.  The layer-cake formula gives
\begin{align}
  \int_{I_a}\sqrt{n(s)}\,ds
  &=\int_0^\infty
    \big|\{s\in I_a:\sqrt{n(s)}>t\}\big|\,dt \notag\\
  &\le\int_0^\infty
    \min\left(d_a,\frac{4R}{t^2}\right)dt
  =4\sqrt{Rd_a}.
  \label{eq:first-schur}
\end{align}
Similarly, using \eqref{eq:row-tail},
\begin{align}
  \sum_{a\in\mathcal J_s}\sqrt{d_a}
  &=\int_0^\infty
    \#\{a\in\mathcal J_s:\sqrt{d_a}>t\}\,dt \notag\\
  &\le\int_0^\infty
    \min\left(n(s),\frac{4R}{t^2}\right)dt
  =4\sqrt{Rn(s)}.
  \label{eq:second-schur}
\end{align}

\emph{Weighted Schur test.}
When $n(s)>0$, set
\[
  w(s,a):=\left(\frac{n(s)}{d_a}\right)^{1/4}
  \qquad(a\in\mathcal J_s).
\]
Weighted Cauchy--Schwarz gives
\[
  |T_Kc(s)|^2
  \le
  \left(\sum_{a\in\mathcal J_s}|c_a|^2w(s,a)^2\right)
  \left(\sum_{a\in\mathcal J_s}w(s,a)^{-2}\right)
\]
\[
  =
  \left(\sum_{a\in\mathcal J_s}|c_a|^2
    \frac{\sqrt{n(s)}}{\sqrt{d_a}}\right)
  \left(\sum_{a\in\mathcal J_s}
    \frac{\sqrt{d_a}}{\sqrt{n(s)}}\right).
\]
By \eqref{eq:second-schur}, the second factor is at most $4\sqrt R$.
Integrating, interchanging the sum and integral, and using
\eqref{eq:first-schur}, we obtain
\[
\begin{aligned}
  \|T_Kc\|_{L^2(I)}^2
  &\le4\sqrt R
    \sum_{a\in\mathcal J}|c_a|^2\frac{1}{\sqrt{d_a}}
    \int_{I_a}\sqrt{n(s)}\,ds\\
  &\le16R\sum_{a\in\mathcal J}|c_a|^2.
\end{aligned}
\]
This proves \eqref{eq:biconvex-norm}.

\label{prop:one-dimensional}

$\Lambda\subset\delta\mathbb Z$ be finite, and let $\eta\ge0$.  Then

  \1_{\{\mathcal F_f(s,a)\le\eta\}}

  \le

\end{equation}

<!-- proof-2: body_md -->
For fixed $a$, the set
\[
  J_a:=\{s\in[-1,1]:\mathcal F_f(s,a)\le\eta\}
\]
is an interval, since $s\mapsto\mathcal F_f(s,a)$ is convex.  For fixed $s$,
the admissible $a$ form an interval in the ordering of $\Lambda$, since
$a\mapsto f_{[-1,1]}^*(a)-as$ is convex.  Thus the incidence kernel is
biconvex.

Suppose that the kernel is one on $E\times X$, and put $k=\#X$.  The
cases $E=\varnothing$ and $k=0$ are trivial, and if $k=1$, then
$k|E|\le2$.  If $k\ge2$, let
$a_-:=\min X$ and $a_+:=\max X$.  For $s,t\in E$, direct cancellation
gives the exact identity
\begin{equation}
  \mathcal F_f(s,a_-)+\mathcal F_f(t,a_+)
  -\mathcal F_f(s,a_+)-\mathcal F_f(t,a_-)
  =(s-t)(a_+-a_-).
  \label{eq:cross-difference}
\end{equation}
All four gaps on the left lie in $[0,\eta]$, so
\[
  |s-t|(a_+-a_-)\le2\eta.
\]
The crucial point is that the cross-difference eliminates $f$ completely;
thus the resulting rectangle estimate is uniform over all convex
functions.
Taking the supremum over $s,t\in E$, using $|E|\le\operatorname{diam}(E)$ and
$a_+-a_-\ge(k-1)\delta$, we find
\[
  k|E|
  \le\frac{2k}{k-1}\cdot\frac{\eta}{\delta}
  \le\frac{4\eta}{\delta}.
\]
Thus the rectangle hypothesis of Lemma \ref{lem:biconvex} holds with
\[
  R_0:=\max\left(2,\frac{4\eta}{\delta}\right)
  \le4\left(1+\frac{\eta}{\delta}\right).
\]
Lemma \ref{lem:biconvex} gives
\[
  \|T_fc\|_2
  \le8\left(1+\frac{\eta}{\delta}\right)^{1/2}\|c\|_2,

<!-- theorem-2: body_md -->
\label{thm:fenchel-bound}
Let $m\ge1$, let $Q=[-1,1]^m$, let $f:Q\to\mathbb R$ be continuous and convex,
let $\delta>0$, let $\Lambda\subset\delta\mathbb Z^m$ be finite, and
let $\eta\ge0$.  Define
\[
  T_fc(p):=\sum_{x\in\Lambda}c_x
  \1_{\{\mathcal F_f(p,x)\le\eta\}}.
\]
Then
\begin{equation}
  \|T_f\|_{\ell^2(\Lambda)\to L^2(Q)}
  \le
  C_m\left(1+\frac{\eta}{\delta}\right)^{m/2},
  \label{eq:fenchel-bound}
\end{equation}
where $C_m$ depends only on $m$.  In particular, if
$0\le\eta\le A\delta$, then the operator norm is bounded by a constant
depending only on $m$ and $A$.
<!-- end theorem-2 -->
<!-- proof-3: body_md -->
We induct on $m$.  Proposition \ref{prop:one-dimensional} is the case
$m=1$.

Write $p=(s,q)$ and $x=(a,b)$, where $s,a\in\mathbb R$ and
$q,b\in\mathbb R^{m-1}$, and put $Q':=[-1,1]^{m-1}$.  For each first
coordinate $a$ occurring in $\Lambda$, define
\[
  g_a(q):=\min_{s\in[-1,1]}\bigl(f(s,q)-as\bigr).
\]
The function $g_a$ is convex.  Indeed, for $q_0,q_1\in Q'$ choose
minimizers $s_0,s_1$, and for $0\le\lambda\le1$ set
$q_\lambda=\lambda q_0+(1-\lambda)q_1$ and
$s_\lambda=\lambda s_0+(1-\lambda)s_1$.  Joint convexity gives
\[
  g_a(q_\lambda)
  \le f(s_\lambda,q_\lambda)-as_\lambda
  \le\lambda g_a(q_0)+(1-\lambda)g_a(q_1).
\]
Moreover,
\[
  |g_a(q)-g_a(q')|
  \le\sup_{s\in[-1,1]}|f(s,q)-f(s,q')|,
\]
so uniform continuity of $f$ on the compact cube shows that $g_a$ is
continuous.

For fixed $q$, let $\phi_q(s):=f(s,q)$.  The relevant conjugates satisfy
\begin{align*}
  g_{a,Q'}^*(b)
  &=\max_{q\in Q'}\max_{s\in[-1,1]}
    \bigl(as+b\cdot q-f(s,q)\bigr)
   =f_Q^*(a,b),\\
  \phi_{q,[-1,1]}^*(a)
  &=\max_{s\in[-1,1]}\bigl(as-f(s,q)\bigr)
   =-g_a(q).
\end{align*}
Consequently, the gap factors exactly as
\begin{equation}
  \mathcal F_f((s,q),(a,b))
  =\mathcal F_{\phi_q}(s,a)+\mathcal F_{g_a}(q,b).
  \label{eq:gap-factorization}
\end{equation}
Both terms are nonnegative.  This exact identity is the mechanism that
permits coordinate iteration.  Hence
\begin{equation}
  \1_{\{\mathcal F_f\le\eta\}}
  \le
  \1_{\{\mathcal F_{\phi_q}\le\eta\}}
  \1_{\{\mathcal F_{g_a}\le\eta\}}.
  \label{eq:indicator-factorization}
\end{equation}

Because the kernel is nonnegative, $|T_fc|\le T_f|c|$.  Set
\[
  u_a(q)
  :=\sum_{b:(a,b)\in\Lambda}|c_{a,b}|
  \1_{\{\mathcal F_{g_a}(q,b)\le\eta\}}.
\]
By \eqref{eq:indicator-factorization},
\[
  |T_fc(s,q)|
  \le
  \sum_a u_a(q)
  \1_{\{\mathcal F_{\phi_q}(s,a)\le\eta\}}.
\]
Apply Proposition \ref{prop:one-dimensional} in the $s$ variable for
each fixed $q$, integrate over $q$, and apply the induction hypothesis
to each $g_a$ on the lattice fiber
$\{b:(a,b)\in\Lambda\}\subset\delta\mathbb Z^{m-1}$.  This gives
\[
\begin{aligned}
  \|T_fc\|_{L^2(Q)}^2
  &\le
  C_1^2\left(1+\frac{\eta}{\delta}\right)
  \sum_a\|u_a\|_{L^2(Q')}^2\\
  &\le
  C_1^2C_{m-1}^2
  \left(1+\frac{\eta}{\delta}\right)^m
  \sum_{(a,b)\in\Lambda}|c_{a,b}|^2.
\end{aligned}
\]
Taking $C_m=C_1C_{m-1}$ proves \eqref{eq:fenchel-bound}.

<!-- proposition-2: body_md -->
\label{prop:cap-bound}
Let $d\ge2$, let $K\subset\mathbb R^d$ be compact and convex, let $\delta>0$,
let $\Gamma\subset\delta\mathbb Z^d\cap K$ be finite, and let $\tau>0$.
Define
\[
  \mathcal C_{K,\tau}c(u)
  :=\sum_{z\in\Gamma}c_z\1_{\Omega_K(z,\tau)}(u).
\]
Then
\begin{equation}
  \|\mathcal C_{K,\tau}c\|_{L^2(S^{d-1},d\sigma)}
  \le
  C_d\left(1+\frac{\tau}{\delta}\right)^{d/2}
  \|c\|_{\ell^2(\Gamma)},
  \label{eq:cap-bound}
\end{equation}
where $C_d$ depends only on $d$.
<!-- end proposition-2 -->
<!-- proof-4: body_md -->
Put $m=d-1$.  Apart from the measure-zero set on which two coordinates
have equal maximal absolute value, the sphere is the disjoint union of the
$2d$ open regions on which one signed coordinate is strictly larger in
absolute value than all the others.  On each region, a signed coordinate
permutation $S$ sends the distinguished coordinate to the positive last
coordinate, and every direction in the region has a unique representation
\begin{equation}
  Su=\frac{(p,1)}{\sqrt{1+|p|^2}},
  \qquad p\in(-1,1)^m.
  \label{eq:gnomonic-chart}
\end{equation}
We enlarge the parameter domain to $Q:=[-1,1]^m$ when applying Theorem
\ref{thm:fenchel-bound}.  In these coordinates,
\[
  d\sigma(u)=(1+|p|^2)^{-d/2}\,dp,
\]
so spherical and Lebesgue measure are comparable on the chart.

Because $S$ is a signed permutation, it preserves
$\delta\mathbb Z^d$.  Write $Sz=(x,t)$ with
$x\in\delta\mathbb Z^m$ and $t\in\delta\mathbb Z$.  Let $K_S:=SK$ and
define
\[
  f(p):=H_{K_S}(p,1),
  \qquad p\in Q.
\]
For $u$ and $p$ related by \eqref{eq:gnomonic-chart}, define the
homogeneous support deficit
\begin{equation}
  \operatorname{def}_z(p)
  :=f(p)-p\cdot x-t
  =\sqrt{1+|p|^2}\,[h_K(u)-z\cdot u].
  \label{eq:homogeneous-deficit}
\end{equation}

Define the chartwise support barrier
\[
  \beta(x):=\min_{p\in Q}\bigl(f(p)-p\cdot x\bigr)=-f_Q^*(x).
\]
Because the slopes are restricted to the chart $Q$, no assertion is made
that $\beta(x)$ equals the actual upper boundary of the vertical fiber of
$K_S$.  What is needed is the support inequality: if
$(x,t)=Sz\in K_S$, then $f(p)\ge p\cdot x+t$ for every $p\in Q$, and hence
$t\le\beta(x)$.

The support deficit decomposes exactly as
\begin{equation}
  \underbrace{\operatorname{def}_z(p)}_{\text{homogeneous support deficit}}
  =
  \underbrace{\mathcal F_f(p,x)}_{\text{Fenchel--Young gap}}
  +
  \underbrace{\beta(x)-t}_{\text{vertical slack}}.
  \label{eq:deficit-decomposition}
\end{equation}
Indeed, $f_Q^*(x)=-\beta(x)$.  Both terms on the right are nonnegative, so
\begin{equation}
  \mathcal F_f(p,x)\le\operatorname{def}_z(p),
  \qquad
  0\le\beta(x)-t\le\operatorname{def}_z(p).
  \label{eq:gap-below-deficit}
\end{equation}
Geometrically, the first term records how far the horizontal coordinate
$x$ is from being dual to the chart direction $p$, while the second records
how far the height $t$ lies below the chartwise supporting barrier.

For each horizontal lattice point $x$, define
\[
  \mathcal T_x
  :=\{t\in\delta\mathbb Z:
      S^{-1}(x,t)\in\Gamma
      \text{ and }
      \Omega_K(S^{-1}(x,t),\tau)
      \text{ meets the current chart}\}.
\]
If $t\in\mathcal T_x$, then \eqref{eq:homogeneous-deficit} is at most
$\sqrt d\,\tau$ at some $p\in(-1,1)^m$.  By
\eqref{eq:gap-below-deficit},
\[
  t\le\beta(x)\le t+\sqrt d\,\tau.
\]
Thus every $t\in\mathcal T_x$ lies in an interval of length
$\sqrt d\,\tau$, and therefore
\begin{equation}
  \#\mathcal T_x
  \le
  \left\lfloor\frac{\sqrt d\,\tau}{\delta}\right\rfloor+1
  \le C_d\left(1+\frac{\tau}{\delta}\right).
  \label{eq:vertical-multiplicity}
\end{equation}

For each $x$, aggregate the coefficients over the relevant heights:
\[
  \widetilde c_x
  :=\sum_{t\in\mathcal T_x}|c_{S^{-1}(x,t)}|.
\]
By Cauchy--Schwarz and \eqref{eq:vertical-multiplicity},
\begin{equation}
  \sum_x|\widetilde c_x|^2
  \le
  C_d\left(1+\frac{\tau}{\delta}\right)
  \sum_{z\in\Gamma}|c_z|^2.
  \label{eq:aggregation-bound}
\end{equation}
Moreover, \eqref{eq:homogeneous-deficit} and
\eqref{eq:gap-below-deficit} give, pointwise on the chart,
\[
  |\mathcal C_{K,\tau}c(u(p))|
  \le
  \sum_x\widetilde c_x
  \1_{\{\mathcal F_f(p,x)\le\sqrt d\,\tau\}}.
\]
Theorem \ref{thm:fenchel-bound}, applied with
$\eta=\sqrt d\,\tau$, yields
\[
  \|\mathcal C_{K,\tau}c\|_{L^2(\text{current chart})}
  \le
  C_d\left(1+\frac{\tau}{\delta}\right)^{(d-1)/2}
  \|\widetilde c\|_{\ell^2}.
\]
Combining this with \eqref{eq:aggregation-bound} gives
\[
  \|\mathcal C_{K,\tau}c\|_{L^2(\text{current chart})}
  \le
  C_d\left(1+\frac{\tau}{\delta}\right)^{d/2}
  \|c\|_{\ell^2(\Gamma)}.
\]
Summing the squared estimates over the $2d$ chart pieces proves
\eqref{eq:cap-bound}.

  <2\varepsilon.
\]
Thus, if $v=(z-w)/|z-w|$, the geodesic ball of radius $\sqrt\varepsilon$ about
$v$ lies in
\[
  \Omega_z\cap(-\Omega_w),
  \qquad
  \Omega_z:=\Omega_B(z,2\varepsilon).
\]
Writing $B_{\mathrm{geo}}(v,\rho)$ for the geodesic ball of radius
$\rho$ about $v$, we have, for $0<\rho\le1$,
\[
  \sigma(B_{\mathrm{geo}}(v,\rho))
  =\sigma(S^{d-2})\int_0^\rho\sin^{d-2}t\,dt
  \gtrsim_d\rho^{d-1}.
\]
Taking $\rho=\sqrt\varepsilon$ gives
\begin{equation}
  1
  \le
  C_d\varepsilon^{-(d-1)/2}
  \int_{S^{d-1}}
  \1_{\Omega_z}(u)\1_{\Omega_w}(-u)\,d\sigma(u)
  \qquad(z\sim w).
  \label{eq:entrywise-incidence}
\end{equation}

Define
\[
  F(u):=\sum_{z\in\Gamma}m_z\1_{\Omega_z}(u)
       =\mathcal C_{B,2\varepsilon}m(u).
\]
Summing \eqref{eq:entrywise-incidence} with weights $m_zm_w$, enlarging
the sum from the relation $z\sim w$ to all ordered cell pairs, and using
Cauchy--Schwarz and the invariance of $\sigma$ under $u\mapsto-u$, we
obtain
\begin{align*}
  \mathcal A_\varepsilon(\mu)
  &\le\sum_{z\sim w}m_zm_w\\
  &\le C_d\varepsilon^{-(d-1)/2}
    \int_{S^{d-1}}F(u)F(-u)\,d\sigma(u)\\
  &\le C_d\varepsilon^{-(d-1)/2}\|F\|_2^2.
\end{align*}
Proposition \ref{prop:cap-bound} applies with $K=B$, $\tau=2\varepsilon$, and
\[
  \frac{\tau}{\delta}=20\sqrt d.
\]
Therefore
\[
  \|F\|_2^2
  \le C_d\sum_{z\in\Gamma}m_z^2.
\]
Combining this with \eqref{eq:neighbor-cell-bound} yields
\[
  \mathcal A_\varepsilon(\mu)
  \le C_d\varepsilon^{-(d-1)/2}\mathcal N_\varepsilon(\mu),
\]
which is equivalent to \eqref{eq:weighted-main-bound}.
