<!-- definition-1: body_md -->
\label{defn:ndpg}
  We denote the objective function in~\eqref{eq:J-min-NonlinDPG} by
  $\mathcal{{R}}_h(\alpha, {\cdot})$, i.e.,
  \begin{equation}
    \label{eq:DPG-loss-defn}
    \mathcal{{R}}_h(\alpha, {{\mathcal{{z}}}_h}) = \| \ell - \mathcal{{A}}_\alpha({\mathcal{{z}}}_h) \|^2_{(\mathcal{{V}}_{\alpha, h})'}, \qquad
    {\mathcal{{z}}}_h \in {\mathcal{{U}}_{\alpha, h}}, \alpha \in \mathcal{{Q}},
  \end{equation}
  and refer to it as the {\bf{nonlinear residual loss function}}
  generated by $\mathcal{{V}}_{\alpha, h}$.  Its two instances, the NFOSLS and the NDPG
  loss functions, are described in \S\ref{ssec:nfosls-abstract}
  and~\S\ref{ssec:ndpg-abstract}, where their computability is also
  discussed.

<!-- assumption-1: body_md -->
\label{asm:NDPGdiff}
  We assume that
  \begin{enumerate}[label=(\alph*)]

  \item for every $\alpha \in \mathcal{{Q}}$, there is a
    ${\mathcal{{u}}} = {\mathcal{{u}}}(\alpha) \in D$ satisfying $\mathcal{{R}}(\alpha, {{\mathcal{{u}}}}) = 0$,

  \item  $\mathcal{{A}}_\alpha$ is G\^ateaux differentiable in the sense
    of~\eqref{eq:gateaux} at each ${\mathcal{{w}}} \in D$ and
    $\alpha \in \mathcal{{Q}}$; the function
    $t \mapsto d\mathcal{{A}}_\alpha({\mathcal{{w}}} + t {\mathcal{{z}}})({\mathcal{{z}}})({\mathcal{{v}}})$ is continuous
    on $[0,1]$ whenever the segment $\{{\mathcal{{w}}} + t {\mathcal{{z}}} : 0 \le t \le 1\}$
    lies in $D$; and there exists $0<C_{d}<\infty$ such that
\begin{equation}
\label{linvc}
\| d\mathcal{{A}}_\alpha({\mathcal{{w}}})({\mathcal{{z}}}) \|_{{\mathcal{{V}}'_\alpha}} \le C_{d} \| {\mathcal{{z}}}\|_{{\mathcal{{U}}_\alpha}},\quad
   \text{ for all } {\mathcal{{w}}} \in D, {\mathcal{{z}}} \in {\mathcal{{U}}_\alpha}, \mbox{and}\,\,  \alpha \in \mathcal{{Q}},
\end{equation}

  \item and there exist constants $0<\delta \le +\infty$ and $c_*>0$ such that for
    every $\alpha\in\mathcal{{Q}}$ and every
    ${\mathcal{{z}}}\in S_\delta({\mathcal{{u}}}(\alpha))\cap D$,
    \begin{equation}
      \label{eq:c-prime}
      c_*\|{\mathcal{{u}}}(\alpha)-{\mathcal{{z}}}\|_{\mathcal{{U}}_\alpha} \le \|\mathcal{{A}}_\alpha({\mathcal{{u}}}(\alpha))-\mathcal{{A}}_\alpha({\mathcal{{z}}})\|_{\mathcal{{V}}'_\alpha}.
    \end{equation}
\end{enumerate}

<!-- theorem-1: body_md -->
\label{thm:NDPG-gen-nonlinear-dpg}
  Suppose Assumptions~\ref{asm:NDPGdiff} and~\ref{asm:fortin-a} hold.
  Then the error estimator (or the loss function) $\mathcal{{R}}_h(\alpha, {\cdot})$
  of~\eqref{eq:DPG-loss-defn}
  satisfies the following local reliability and efficiency estimates:
  there are positive constants $c$ and
  $C$ such that given any $\alpha \in \mathcal{{Q}}$, there exist an $h_0>0$ and
  a set $\tilde D_{h_0} \subseteq D$, such that for all $h < h_0$,
  $\tilde D_{h_0}$ contains a nonempty open subset of ${\mathcal{{U}}_{\alpha, h}}$ and every
  ${{\mathcal{{w}}}_h} \in \tilde D_{h_0} \cap {\mathcal{{U}}_{\alpha, h}}$ satisfies
  \begin{align}
    \label{eq:reliability}
    c & \| {\mathcal{{u}}}  - {\mathcal{{w}}}_h \|_{\mathcal{{U}}_\alpha}^2  \le
    \mathcal{{R}}_h(\alpha, {{\mathcal{{w}}}_h})\,
          \| \varPi \|_{L({\mathcal{{V}}_\alpha})}^2
          + \operatorname{{\ensuremath{\text{osc}}}}(\ell)^2,
    \\
    \label{eq:efficiency}
    \mathcal{{R}}_h(\alpha, {{\mathcal{{w}}}_h}) \le C & \| {\mathcal{{u}}} - {\mathcal{{w}}}_h \|_{\mathcal{{U}}_\alpha}^2,
  \end{align}
  and we may choose $c= c_*^2/2$ and $C=C_{d}^2.$ If \eqref{eq:c-prime}
  holds globally, i.e., if $D = {\mathcal{{U}}_\alpha}$ and $\delta=+\infty$ in
  Assumption~\ref{asm:NDPGdiff}(c), then we may take $h_0=+\infty$ and
  $\tilde D_{h_0} = {\mathcal{{U}}_\alpha}$.  If $\operatorname{{\ensuremath{\text{osc}}}}(\ell)=0$ and
  $\| \varPi \|_{L({\mathcal{{V}}_\alpha})}^2$ is bounded uniformly in $\alpha$ and $h$,
  then~\eqref{eq:reliability}--\eqref{eq:efficiency} imply local
  variational correctness in the sense of~\eqref{ideally} on
  $\mathcal{{H}} = \tilde D_{h_0} \cap {\mathcal{{U}}_{\alpha, h}}$, and if, in addition, $D = {\mathcal{{U}}_\alpha}$
  and $\delta=+\infty$, then the variational correctness holds with
  $\mathcal{{H}}={\mathcal{{U}}_{\alpha, h}}$.

<!-- proof-1: body_md -->
{\em Step 1.}
  Abbreviate ${\mathcal{{u}}} \equiv {\mathcal{{u}}}(\alpha)$.
  Let $\delta$ and $c_*$ be as in Assumption~\ref{asm:NDPGdiff}(c),
  choose $\delta_1 > 0$ so small that $S_{\delta_1}({\mathcal{{u}}}) \subseteq D$,
  and set $\delta_2 = \min(\delta, \delta_1)$.
  We claim that for every ${\mathcal{{z}}} \in S_{\delta_2}({\mathcal{{u}}})$,
  \begin{equation}
    \label{eq:33-prime}
    c_* \|{\mathcal{{u}}} - {\mathcal{{z}}}\|_{{\mathcal{{U}}_\alpha}}
    \;\le\; \|\mathcal{{A}}_\alpha({\mathcal{{u}}}) - \mathcal{{A}}_\alpha({\mathcal{{z}}})\|_{{\mathcal{{V}}'_\alpha}}
    \;\le\; C_{d} \|{\mathcal{{u}}} - {\mathcal{{z}}}\|_{{\mathcal{{U}}_\alpha}}.
  \end{equation}
  The lower bound is immediate from Assumption~\ref{asm:NDPGdiff}(c),
  since $S_{\delta_2}({\mathcal{{u}}}) \subseteq S_\delta({\mathcal{{u}}}) \cap D$.
  For the upper bound, since $S_{\delta_2}({\mathcal{{u}}})$ is convex, the segment
  ${\mathcal{{z}}}_t = {\mathcal{{z}}} + t({\mathcal{{u}}} - {\mathcal{{z}}})$ lies in $S_{\delta_2}({\mathcal{{u}}}) \subseteq D$
  for all $t \in [0,1]$.
  On this line segment, define
  $\psi(t) = a_\alpha({\mathcal{{z}}}_t; {\mathcal{{v}}})$ for a
  ${\mathcal{{v}}} \in {\mathcal{{V}}_\alpha}$.
  The  Fundamental Theorem of Calculus (FTC) applied
  to $\psi$ yields
  \begin{equation}
    \label{eq:ftc}
    \langle \mathcal{{A}}_\alpha({\mathcal{{u}}}) - \mathcal{{A}}_\alpha({\mathcal{{z}}}),\, {\mathcal{{v}}} \rangle
    = \psi(1) - \psi(0) = \int_0^1 \psi'(t) \, dt
    = \int_0^1 d\mathcal{{A}}_\alpha({\mathcal{{z}}}_t)({\mathcal{{u}}} - {\mathcal{{z}}})({\mathcal{{v}}})\, dt.
  \end{equation}
  By Assumption~\ref{asm:NDPGdiff}(b), the integrand is bounded by
  $C_{d} \|{\mathcal{{u}}} - {\mathcal{{z}}}\|_{{\mathcal{{U}}_\alpha}} \|{\mathcal{{v}}}\|_{{\mathcal{{V}}_\alpha}}$, so taking the supremum
  over $\{ {\mathcal{{v}}} \in {\mathcal{{V}}_\alpha}: \|{\mathcal{{v}}}\|_{{\mathcal{{V}}_\alpha}} = 1\}$ yields the upper bound
  in~\eqref{eq:33-prime}.
  (If $\delta=+\infty$ and $D = {\mathcal{{U}}_\alpha}$, then we may take
  $\delta_2 = +\infty$.)

  {\em Step 2.} We now prove the reliability
  estimate~\eqref{eq:reliability}.
  By~\eqref{eq:approx'}, we can find an $h_0>0$ such that for all
  $h < h_0$, there is some ${\mathcal{{p}}_h}$ in
  ${\mathcal{{U}}_{\alpha, h}} \cap S_{\delta_2}({\mathcal{{u}}}) \subset D$, and consequently, there is
  also an open neighborhood $U_h({\mathcal{{p}}_h})$ of ${\mathcal{{p}}_h}$ in $D_h$, which we
  may take to be contained in $S_{\delta_2}({\mathcal{{u}}})$. Then
  $\tilde D_{h_0} = \cup_{h < h_0} U_h({\mathcal{{p}}_h})$ is contained in
  $S_{\delta_2}({\mathcal{{u}}}) \cap D$ and contains  for each $h < h_0$
  the nonempty open subset   $U_h({\mathcal{{p}}_h})$ of ${\mathcal{{U}}_{\alpha, h}}$.
  (If $\delta=+\infty$ and $D = {\mathcal{{U}}_\alpha}$, then $S_{\delta_2}({\mathcal{{u}}}) = {\mathcal{{U}}_\alpha}$,
  so we may take $h_0=+\infty$ and $\tilde D_{h_0} = {\mathcal{{U}}_\alpha}$, giving
  $\tilde D_{h_0} \cap {\mathcal{{U}}_{\alpha, h}} = {\mathcal{{U}}_{\alpha, h}}$ for all $h$.)

  Since $\mathcal{{R}}(\alpha, {{\mathcal{{u}}}}) = 0$ by Assumption~\ref{asm:NDPGdiff}(a), we have
  $\mathcal{{A}}_\alpha({\mathcal{{u}}}) = \ell$, so applying the lower bound in~\eqref{eq:33-prime}
  with ${\mathcal{{z}}} = {{\mathcal{{w}}}_h} \in \tilde D_{h_0} \cap {\mathcal{{U}}_{\alpha, h}} \subseteq S_{\delta_2}({\mathcal{{u}}})$ yields
  \begin{equation}
    \label{eq:34}
    c_* \| {\mathcal{{u}}} - {\mathcal{{w}}}_h \|_{{\mathcal{{U}}_\alpha}}
    \le
    \sup_{0 \ne{\mathcal{{v}}} \in {\mathcal{{V}}_\alpha}}
    \frac{|\ell({\mathcal{{v}}}) - a_\alpha({\mathcal{{w}}}_h; {\mathcal{{v}}})|}{ \| {\mathcal{{v}}} \|_{{\mathcal{{V}}_\alpha}}}.
  \end{equation}
  To bound the numerator, observe that
  \begin{align*}
    \ell({\mathcal{{v}}}) - a_\alpha({\mathcal{{w}}}_h; {\mathcal{{v}}})
    & = \ell(\varPi {\mathcal{{v}}}) - a_\alpha({\mathcal{{w}}}_h; \varPi  {\mathcal{{v}}})
      + \ell({\mathcal{{v}}} - \varPi {\mathcal{{v}}}) - a_\alpha( {\mathcal{{w}}}_h; {\mathcal{{v}}} - \varPi  {\mathcal{{v}}})
    \\
    & = (\mathcal{{e}}_h^{a}(\alpha, {{{\mathcal{{w}}}_h}}), \varPi  {\mathcal{{v}}})_{{\mathcal{{V}}_\alpha}}
      + \ell({\mathcal{{v}}} - \varPi {\mathcal{{v}}})
    \\
    & \le \Big(
      \mathcal{{R}}_h(\alpha, {{{\mathcal{{w}}}_h}})^{1/2}  \| \varPi  \|_{L({\mathcal{{V}}_\alpha})}
      + \operatorname{{\ensuremath{\text{osc}}}} (\ell ) \Big)
      \| {\mathcal{{v}}} \|_{{\mathcal{{V}}_\alpha}}
  \end{align*}
  where we have used
  Assumption~\ref{asm:fortin-a}'s equation~\eqref{eq:Fortin},  \eqref{eq:residual-rep'},
  and~\eqref{eq:L-and-E} in succession. Using this to bound the right-hand
  side of~\eqref{eq:34} after squaring,
  \begin{align*}
    c_*^2 \| {\mathcal{{u}}} - {\mathcal{{w}}}_h \|_{{\mathcal{{U}}_\alpha}}^2
    \le
    2 \| \varPi  \|^2_{L({\mathcal{{V}}_\alpha})}
      \mathcal{{R}}_h(\alpha, {{\mathcal{{w}}}_h})
      + 2 \operatorname{{\ensuremath{\text{osc}}}}(\ell)^2,
  \end{align*}
  proves  the reliability inequality~\eqref{eq:reliability}.

  {\em Step 3.}  To prove the efficiency
  estimate~\eqref{eq:efficiency}, we start
  from~\eqref{eq:residual-rep'} to get
  \begin{align*}
    \| \mathcal{{e}}_h^{a}(\alpha, {{{\mathcal{{w}}}_h}}) \|_{{\mathcal{{V}}_\alpha}}^2 & = \ell ( \mathcal{{e}}_h^{a}(\alpha, {{\mathcal{{w}}}_h})) - a_\alpha( {\mathcal{{w}}}_h; \mathcal{{e}}_h^{a}(\alpha, {{{\mathcal{{w}}}_h}}))
    \\
    & = a_\alpha( {\mathcal{{u}}}; \mathcal{{e}}_h^{a}(\alpha, {{{\mathcal{{w}}}_h}})) - a_\alpha({{\mathcal{{w}}}_h}; \mathcal{{e}}_h^{a}(\alpha, {{{\mathcal{{w}}}_h}}))
    \\
    & \le C_{d} \| {\mathcal{{u}}} - {{\mathcal{{w}}}_h}\|_{{\mathcal{{U}}_\alpha}} \| \mathcal{{e}}_h^{a}(\alpha, {{{\mathcal{{w}}}_h}}) \|_{{\mathcal{{V}}_\alpha}}
  \end{align*}
  where the second equality holds because $\ell = \mathcal{{A}}_\alpha({\mathcal{{u}}})$ by
  Assumption~\ref{asm:NDPGdiff}(a) and, in the last step, we have
  used the upper bound in~\eqref{eq:33-prime}, which applies since
  $\tilde D_{h_0} \subset S_{\delta_2}({\mathcal{{u}}}).$
  In view of \eqref{eq:L-and-E}, the
  resulting bound for $\| \mathcal{{e}}_h^{a}(\alpha, {{{\mathcal{{w}}}_h}}) \|_{{\mathcal{{V}}_\alpha}}$ gives the claimed
  estimate~\eqref{eq:efficiency}.

<!-- remark-1: body_md -->
\label{rem:quasioptimality}
  Theorem~\ref{thm:NDPG-gen-nonlinear-dpg} is an {\it a posteriori}
  statement, but it constrains the loss minimizer {\it a priori} as
  well.  Such a passage from two-sided {\it a posteriori} bounds to
  {\it a priori} error estimates is by now routine: see the linear
  precedent in~\cite[Remark~2.5]{CarstDemkoGopal14}. Repeating the
  idea, let ${\mathcal{{u}}}_h$ realize
  $\inf\{ \mathcal{{R}}_h(\alpha, {{\mathcal{{w}}}}) : {\mathcal{{w}}} \in \tilde D_{h_0} \cap {\mathcal{{U}}_{\alpha, h}}\}$.  Reading
  the reliability bound~\eqref{eq:reliability} at ${\mathcal{{u}}}_h$, replacing
  $\mathcal{{R}}_h(\alpha, {{\mathcal{{u}}}_h})$ by $\mathcal{{R}}_h(\alpha, {{\mathcal{{w}}}})$ for any
  ${\mathcal{{w}}} \in \tilde D_{h_0} \cap {\mathcal{{U}}_{\alpha, h}}$ (which minimality permits), and
  then discarding $\mathcal{{R}}_h(\alpha, {{\mathcal{{w}}}})$ in favor of the error through the
  efficiency bound~\eqref{eq:efficiency}, one arrives at
  \begin{equation}
    \label{eq:quasioptimality}
    \| {\mathcal{{u}}} - {\mathcal{{u}}}_h \|^2_{{\mathcal{{U}}_\alpha}}
    \;\le\;
    \varLambda \!\!\!
    \inf_{{\mathcal{{w}}} \in \tilde D_{h_0} \cap {\mathcal{{U}}_{\alpha, h}}} \!\!\!
    \| {\mathcal{{u}}} - {\mathcal{{w}}} \|^2_{{\mathcal{{U}}_\alpha}}
    \; + \; \frac 1 c\, \operatorname{{\ensuremath{\text{osc}}}}(\ell)^2,
    \qquad
    \varLambda := \frac C c \, \| \varPi \|^2_{L({\mathcal{{V}}_\alpha})} .
  \end{equation}
  An exact minimizer of the loss therefore achieves, up to the factor
  $\varLambda$ and the oscillation term, the best approximation error
  attainable in the admissible trial set.  Two aspects
  of~\eqref{eq:quasioptimality} deserve emphasis.  The first is that
  no linear structure on ${\mathcal{{U}}_{\alpha, h}}$ was used either
  in~\eqref{eq:quasioptimality} or anywhere in the proof of
  Theorem~\ref{thm:NDPG-gen-nonlinear-dpg}.  The infimum
  in~\eqref{eq:quasioptimality} therefore runs over precisely the set
  on which the loss was minimized, whatever that set may be.  For the
  neural network classes of \S\ref{sec:nn}, where the loss is
  minimized in expectation over $\alpha$ rather than for each $\alpha$
  separately, the same argument applied to the integrated
  estimate~\eqref{vc} gives a genuine best-approximation guarantee,
  in the norm~\eqref{eq:Xnorm}, for the minimizer of the expected loss
  over a nonlinear set of realizations.  The amplification $\varLambda$, in turn, is the
  reliability-to-efficiency ratio delivered by variational
  correctness, inflated by the Fortin operator norm. In the
  least-squares instance of \S\ref{ssec:nfosls-abstract}, provided
  $D={\mathcal{{U}}_\alpha}$ and $\delta = +\infty$, this inflation disappears along with
  the oscillation, leaving
  $\| {\mathcal{{u}}} - {\mathcal{{u}}}_h \|^2_{{\mathcal{{U}}_\alpha}} \le (C/c) \inf_{{\mathcal{{w}}} \in {\mathcal{{U}}_{\alpha, h}}} \| {\mathcal{{u}}}
  - {\mathcal{{w}}}\|^2_{{\mathcal{{U}}_\alpha}}$
  for any chosen trial class ${\mathcal{{U}}_{\alpha, h}}\subseteq{\mathcal{{U}}_\alpha}$.

\label{rem:sufficient-c}
  When the neighborhoods $S_r({\mathcal{{u}}}(\alpha))$ are in $D$ for some $r>0$
  independent of $\alpha$, the condition~(c) of
  Assumption~\ref{asm:NDPGdiff} is implied by parts~(a) and~(b)
  together with
  \begin{itemize}
  \item[{\rm{(b$'$)}}]
    a quantitative uniform lower bound on the derivative, i.e.,
    there is a $c_{d} > 0$ such that
    \begin{equation}
      \label{eq:lower-dA}
      c_{d} \|{\mathcal{{z}}}\|_{{\mathcal{{U}}_\alpha}}
      \le \|d\mathcal{{A}}_\alpha({\mathcal{{w}}})({\mathcal{{z}}})\|_{{\mathcal{{V}}'_\alpha}},
      \qquad {\mathcal{{w}}} \in D,\; {\mathcal{{z}}} \in {\mathcal{{U}}_\alpha},\; \alpha \in \mathcal{{Q}},
    \end{equation}
  \item[{\rm{(c$'$)}}] and the continuity of the derivative
    $d\mathcal{{A}}_\alpha : D \to L({\mathcal{{U}}_\alpha}, {\mathcal{{V}}'_\alpha})$, uniformly in $\alpha$, in the sense that
    the quantity
    \begin{equation}
      \label{eq:uniform-cty}
      c(\delta) := \sup_{\alpha\in\mathcal{{Q}}}
      \sup_{{\mathcal{{z}}} \in S_\delta({\mathcal{{u}}}(\alpha))}
      \|d\mathcal{{A}}_\alpha({\mathcal{{z}}}) - d\mathcal{{A}}_\alpha({\mathcal{{u}}}(\alpha))\|_{L({\mathcal{{U}}_\alpha},{\mathcal{{V}}'_\alpha})}
      \to 0 \quad \text{ as } \delta \to 0,
    \end{equation}
  \end{itemize}
  where $L({\mathcal{{U}}_\alpha}, {\mathcal{{V}}'_\alpha})$ denotes the space of bounded linear operators
  from ${\mathcal{{U}}_\alpha}$ to ${\mathcal{{V}}'_\alpha}$ endowed with the operator norm topology. To show this,
  by~\eqref{eq:uniform-cty},
  choose $0< \delta \le r $, independent of $\alpha$, with
  $c(\delta) < c_{d}/2$. Fix $\alpha$, write ${\mathcal{{u}}}={\mathcal{{u}}}(\alpha)$,
  and let ${\mathcal{{z}}}\in S_\delta({\mathcal{{u}}})$. Then the segment
  ${\mathcal{{z}}}_t = {\mathcal{{u}}} + t({\mathcal{{z}}} - {\mathcal{{u}}})$, $t\in [0, 1]$, lies
  in $S_\delta({\mathcal{{u}}}) \subset D$.
  Proceeding as in the proof of
  Theorem~\ref{thm:NDPG-gen-nonlinear-dpg} to get \eqref{eq:ftc} and
  then subtracting $d\mathcal{{A}}_\alpha({\mathcal{{u}}})({\mathcal{{u}}}-{\mathcal{{z}}})({\mathcal{{v}}})$ from both
  sides yields
  \[
     \langle{ \mathcal{{A}}_\alpha({\mathcal{{u}}}) - \mathcal{{A}}_\alpha({\mathcal{{z}}})      - d\mathcal{{A}}_\alpha({\mathcal{{u}}})({\mathcal{{u}}}-{\mathcal{{z}}}), {\mathcal{{v}}}}\rangle
    = \int_0^1 \Big[ d\mathcal{{A}}_\alpha({\mathcal{{z}}}_t)({\mathcal{{u}}}-{\mathcal{{z}}})({\mathcal{{v}}})
    - d\mathcal{{A}}_\alpha({\mathcal{{u}}})({\mathcal{{u}}}-{\mathcal{{z}}})({\mathcal{{v}}})\Big]\,dt.
  \]
  Since every ${\mathcal{{z}}}_t \in S_\delta({\mathcal{{u}}})$, the integrand is bounded by
  $c(\delta)\|{\mathcal{{u}}}-{\mathcal{{z}}}\|_{\mathcal{{U}}_\alpha}\|{\mathcal{{v}}}\|_{\mathcal{{V}}_\alpha}$, so
  \[
    \bigl\|d\mathcal{{A}}_\alpha({\mathcal{{u}}})({\mathcal{{u}}}-{\mathcal{{z}}})
    - \bigl[\mathcal{{A}}_\alpha({\mathcal{{u}}})-\mathcal{{A}}_\alpha({\mathcal{{z}}})\bigr]\bigr\|_{{\mathcal{{V}}'_\alpha}}
    \le c(\delta)\|{\mathcal{{u}}}-{\mathcal{{z}}}\|_{{\mathcal{{U}}_\alpha}}.
  \]
  Combining with~\eqref{eq:lower-dA} and the triangle inequality,
  \[
    \|\mathcal{{A}}_\alpha({\mathcal{{u}}})-\mathcal{{A}}_\alpha({\mathcal{{z}}})\|_{{\mathcal{{V}}'_\alpha}}
    \ge c_{d}\|{\mathcal{{u}}}-{\mathcal{{z}}}\|_{{\mathcal{{U}}_\alpha}} - c(\delta)\|{\mathcal{{u}}}-{\mathcal{{z}}}\|_{{\mathcal{{U}}_\alpha}}
    \ge \frac{c_{d}}{2}\|{\mathcal{{u}}}-{\mathcal{{z}}}\|_{{\mathcal{{U}}_\alpha}},
  \]
  which is~\eqref{eq:c-prime} with $c_* = c_{d}/2$.
  We hasten to add that {\rm(b$'$)} and {\rm(c$'$)} are sufficient,
  but far from necessary, for Assumption~\ref{asm:NDPGdiff}(c).  The
  proof of Theorem~\ref{thm:NDPG-gen-nonlinear-dpg} uses no lower
  bound on $d\mathcal{{A}}_\alpha$ at all, so the uniformity of~\eqref{eq:lower-dA} in
  both ${\mathcal{{w}}}$ and $\alpha$ demands more than the theory requires;
  indeed, even in the above argument, \eqref{eq:lower-dA} was used
  only at ${\mathcal{{w}}} = {\mathcal{{u}}}(\alpha)$.  Moreover, the operator-norm
  continuity demanded by {\rm(c$'$)} may be unavailable for
  Nemytskii-type nonlinearities such as some of the later examples
  (cf.~Remark~\ref{rem:no-frechet}).
  This is why we assumed the weaker~\eqref{eq:c-prime}, which
  will be verified directly for such examples in
  Proposition~\ref{prop:NDPGdiff-verif'}, without recourse to~{\rm(c$'$)}.

\label{rem:mixed-reformulation}
  Suppose again that  ${\mathcal{{U}}_{\alpha, h}}$ is a  linear subspace of
  ${\mathcal{{U}}_\alpha}$, so that the perturbations considered  below stay within the trial
class.  Let us observe what the first-order
optimality conditions of the optimization
in~\eqref{eq:exact-minimization} and~\eqref{eq:J-min-NonlinDPG} yield.
A ${\mathcal{{w}}}_h$ in $D_h$ is  a {\em stationary point} of  $\mathcal{{R}}_h(\alpha, {\cdot})$
if
\[
  \lim_{\delta\to 0 }
  \frac{\mathcal{{R}}_h(\alpha, {{\mathcal{{w}}}_h + {\mathcal{{z}}}_h\, \delta}) - \mathcal{{R}}_h(\alpha, {{\mathcal{{w}}}_h})}{\delta}
  = 0
\]
for all directions ${\mathcal{{z}}}_h \in {\mathcal{{U}}_{\alpha, h}}.$ It is easy to
see~\cite{carstensen2017} that a ${\mathcal{{u}}}_h \in D_h$ is a stationary
point of $\mathcal{{R}}_h(\alpha, {\cdot})$ if and only if $\mathcal{{u}}_h$ together with
$\varepsilon_h
\in \mathcal{{V}}_{\alpha, h}$
solves the nonlinear mixed system
\begin{subequations}
  \label{eq:nonlinear-mixed}
  \begin{align}
    \label{eq:eps-u}
    (\varepsilon_h, {\mathcal{{v}}}_h)_{\mathcal{{V}}_\alpha} + a_\alpha({\mathcal{{u}}}_h; {\mathcal{{v}}}_h)
    &  = \ell( {\mathcal{{v}}}_h),
    && {\mathcal{{v}}}_h \in \mathcal{{V}}_{\alpha, h},
    \\
    da_\alpha  ({\mathcal{{u}}}_h; {\mathcal{{z}}}_h, \varepsilon_h)
    & = 0,
    && {\mathcal{{z}}}_h \in {\mathcal{{U}}_{\alpha, h}}.
  \end{align}
\end{subequations}
This {\em nonlinear mixed DPG reformulation} generalizes the
well-known~\cite{DHSW,DPGacta} reformulation of the DPG method as a mixed
method for linear problems.
The prior equation~\eqref{eq:residual-rep'} is
exactly~\eqref{eq:eps-u}, i.e., $\varepsilon_h = \mathcal{{e}}_h^{a}(\alpha, {{\mathcal{{u}}}_h}) \in
\mathcal{{V}}_{\alpha, h}$. The undiscretized version of~\eqref{eq:nonlinear-mixed} can be
written in operator form yielding the stationarity
conditions for
\eqref{eq:exact-minimization}: a stationary point ${\mathcal{{u}}}$ in $D$,
together with its residual representative $\varepsilon$ in ${\mathcal{{V}}_\alpha}$, satisfies
\begin{equation}\label{eq:mixedcarst}
  \begin{pmatrix}
    \mathcal{{G}}_\alpha & \mathcal{{A}}_\alpha \\
    d\mathcal{{A}}_\alpha'(\mathcal{{u}}) & 0
  \end{pmatrix}
  \begin{pmatrix}
    \varepsilon \\ \mathcal{{u}}
  \end{pmatrix}
  =
  \begin{pmatrix}
    \ell \\ 0
  \end{pmatrix}
  \qquad
  \text{ in } \; {\mathcal{{V}}'_\alpha} \times {\mathcal{{U}}'_\alpha}.
\end{equation}
Here, $\mathcal{{G}}_\alpha : {\mathcal{{V}}_\alpha} \to {\mathcal{{V}}'_\alpha}$ is the Riesz isometry and $d\mathcal{{A}}_\alpha'$ denotes
the dual operator of $d\mathcal{{A}}_\alpha$.  At an exact solution ${\mathcal{{u}}}$, the
residual representative $\varepsilon$ vanishes. Conversely, a solution of
\eqref{eq:mixedcarst} with $\varepsilon=0$ satisfies $\mathcal{{A}}_\alpha({\mathcal{{u}}})=\ell$.
The discrete residual representation $\varepsilon_h$ does not vanish in
general.

\label{lem:rho-exists}
  Let Hypothesis~\ref{hyp:sigma} hold.  Then, for each fixed
  $s \in \mathbb{{R}}$, the map $p \mapsto \sigma_\alpha(p,s)$ is a
  $C^1$-isomorphism of $\mathbb{{R}}^d$ onto itself.  Its inverse
  $\rho_\alpha(\cdot, s)$ satisfying
  \begin{equation}
    \label{eq:rho-defn}
    \sigma_\alpha\bigl( \rho_\alpha(q,s), s\bigr) = q,
    \qquad (q,s) \in \mathbb{{R}}^d \times \mathbb{{R}},
  \end{equation}
  is $C^1$ jointly in $(q,s)$, with
  \begin{equation}
    \label{eq:rho-derivs}

    \qquad
    \partial_s\rho_\alpha(q,s)
    = -\,\partial_q\rho_\alpha(q,s)\,
    \partial_s \sigma_\alpha(\rho_\alpha(q,s), s),
  \end{equation}

  \begin{equation}
    \label{eq:rho-bounds}
    \mu_2^{-1} |\xi|^2

    \le \mu_1^{-1} |\xi|^2,
    \qquad
    |\partial_s\rho_\alpha(q,s)| \le \frac{\lambda}{\mu_1}.
  \end{equation}

  $\|r\|_{L_2} \le \mu_2 ( \|g_1\|_{L_2} + \|\nabla v\|_{L_2})$ and
  $\|\operatorname*{div} r\|_{L_2} \le (1+\lambda^2)^{1/2}\|(g_1,g_2)\|_{L_2} +

  the bound on $\|\nabla v\|_{L_2}$ sum to the $c_T^{-2}$
  of~\eqref{eq:frozen-constants}, which is the lower bound
  in~\eqref{eq:frozen-transposed-two-sided}.  Its upper bound follows
  from $\|S^{-1}\| \le \mu_1^{-1}$, the bounds on $\tau$ and $c$,
  and~\eqref{eq:Friedrichs-std}.

  A similar argument also shows the bijectivity of
  $A_{S,\tau,c}$.
  The first equation of $A_{S,\tau,c}(q,z) = (g_1,g_2)$ gives
  $q = g_1 - S\nabla z + \tau\, z$, and substituting it into the
  second shows that $z$ must solve

    B_{S,\tau,c}(z, \zeta)
    = (g_2, \zeta)_\varOmega + (g_1, \nabla \zeta)_\varOmega,
    \qquad \zeta \in \mathring{H}^1,
  \end{equation}

  $\| g \|_\star := \|g_1\|_{L_2} + c_P^{-1/2}\|g_2\|_{L_2}$, by~\eqref{eq:Friedrichs-std},
  so the argument proceeds with minimal changes.
  For the lower bound in~\eqref{eq:frozen-two-sided},
  put $\zeta = z$ in \eqref{eq:BStc} to get
  \[
    B_{S,\tau,c}(z, z)
    = (S \nabla z, \nabla z)_\varOmega - (\tau\, z, \nabla z)_\varOmega
    + (c\, z, z)_\varOmega
    = (g_1, \nabla z)_\varOmega + (g_2, z)_\varOmega
    \le \| g \|_\star\, \|\nabla z\|_{L_2},
  \]
  where~\eqref{eq:Friedrichs-std} was used.  By
  Lemma~\ref{lem:quasilinear-wellposed} the left-hand side is at least
  $c_0 \|\nabla z\|^2_{L_2}$, so
  $\|\nabla z\|_{L_2} \le \| g \|_\star/c_0$.  Now, we can bound
  $\|q\|_{L_2}$ and $\| \operatorname*{div} q \|_{L_2}$ using
  $q + S \nabla z - \tau\, z= g_1$ and $\operatorname*{div} q + c\, z = g_2$,
  respectively.  Since $\|g_1\|_{L_2} \le \| g \|_\star$ and
  $\|g_2\|_{L_2} \le c_P^{1/2} \| g \|_\star$, the three resulting
  bounds on $\|q\|_{L_2}$, $\|\operatorname*{div} q\|_{L_2}$ and
  $\|\nabla z\|_{L_2}$ are the multiples of $\| g \|_\star$ whose
  squares are summed in the bracket
  of~\eqref{eq:frozen-constants}, while the remaining factor
  $1 + c_P^{-1}$ there accounts for
  $\| g \|^2_\star \le (1 + c_P^{-1}) \|g\|^2_{L_2}$, by the
  Cauchy--Schwarz inequality in $\mathbb{{R}}^2$.  This proves the lower
  bound.
    The upper bound in~\eqref{eq:frozen-two-sided} follows from the
  bounds on $S, \tau, c$ and~\eqref{eq:Friedrichs-std}.

      + \bigl( S^{-1}( \tau\, z - \varsigma), r \bigr)_{{\varOmega_h}}

      + (c\, z, v)_\varOmega .

    \;\ge\; c_{d}\, \|{\mathcal{{z}}}\|_{{\mathcal{{U}}_\alpha}},
    \qquad {\mathcal{{z}}} \in {\mathcal{{U}}_\alpha},
  \end{equation}
  with the $c_{d}$ of~\eqref{eq:uw-cd}, uniformly over all admissible
  triples $S, \tau, c$ and all $\alpha \in \mathcal{{Q}}$.  Granting the claim, \eqref{eq:uw-frozen} gives
  $\|\mathcal{{A}}_\alpha({\mathcal{{z}}}^0) - \mathcal{{A}}_\alpha({\mathcal{{z}}}^1)\|_{{\mathcal{{V}}'_\alpha}} \ge c_{d} \|{\mathcal{{z}}}^0 -
  {\mathcal{{z}}}^1\|_{{\mathcal{{U}}_\alpha}}$ for {\em all} ${\mathcal{{z}}}^0, {\mathcal{{z}}}^1 \in {\mathcal{{U}}_\alpha}$, which
  is~\eqref{eq:c-prime} with $c_* = c_{d}$ and $\delta = +\infty$ upon
  setting ${\mathcal{{z}}}^0 = {\mathcal{{u}}}(\alpha)$; since the estimate holds for all
  pairs, the element ${\mathcal{{u}}}$ of part {\rm(a)} is also the unique one at
  which $\mathcal{{R}}(\alpha, {\cdot})$ vanishes.

  It remains to prove the claim~\eqref{eq:uw-infsup}.  The form $b_{S,\tau,c}$ is the
  ultraweak DPG bilinear form of the linear boundary value
  problem
  \begin{equation}
    \label{eq:uw-frozen-bvp}
    \varsigma = \tau\, z - S \nabla z,
    \qquad
    \operatorname*{div} \varsigma + c\, z = g \; \text{ in } \varOmega,
    \qquad z|_{\partial\varOmega} = 0,
  \end{equation}
  i.e., of $-\operatorname*{div}( S\nabla z - \tau\, z) + c\, z = g$, whose primal weak
  form is $B_{S,\tau,c}$ of~\eqref{eq:frozen-form}.  By the theory of
  broken variational
  formulations (see~\cite[Theorem~3.3]{CarstDemkoGopal16} or \cite[Theorem~4.3]{DPGacta})
  the claim~\eqref{eq:uw-infsup} holds, with a constant depending only on the
  corresponding unbroken constant and on the norms
  in~\eqref{eq:quotient-norms}, as soon as the {\em unbroken}
  ultraweak form (which is \eqref{eq:uw-frozen-form} with interface terms removed)
  \[
    b^0_{S,\tau,c}\bigl( (\varsigma, z), (r,v)\bigr)
    := \bigl( S^{-1}(\tau\,z - \varsigma), r \bigr)_\varOmega
    + (z, \operatorname*{div} r)_\varOmega - (\varsigma, \nabla v)_\varOmega + (c\, z, v)_\varOmega
  \]
  is inf-sup stable on
  $\bigl( L_2^d \times L_2 \bigr) \times \bigl( H(\operatorname*{div}) \times
  \mathring{H}^1\bigr)$.  Since $S$, and hence $S^{-1}$, is symmetric, observe that $
    b^0_{S,\tau,c}\bigl( (\varsigma, z), (r,v) \bigr)
    = \bigl( (\varsigma, z), \, T_{S,\tau,c}(r,v) \bigr)_\varOmega
  $
  with $T_{S,\tau,c}$ as in~\eqref{eq:frozen-transposed}.  The trial
  space being an $L_2$ product, the required stability is therefore
  equivalent to the bijectivity of $T_{S,\tau,c}$ onto
  $L_2^d\times L_2$ with a uniformly bounded inverse, which is
  provided by Lemma~\ref{lem:frozen-operator}.

  This also makes $c_{d}$ explicit.  Indeed,
  \cite[Theorem~3.3]{CarstDemkoGopal16} gives
  $c_{d}^{-2} = c^{-2}_{\rm ub} + \hat c^{-2}( \|b^0\|/c_{\rm ub} +
  1)^2$, where $c_{\rm ub}$ and $\|b^0\|$ are the inf-sup and
  continuity constants of $b^0_{S,\tau,c}$ and $\hat c$ is the
  stability constant of the interface terms, which equals one because
  the quotient norms~\eqref{eq:quotient-norms} are the dual norms of
  those terms.  Taking $(r,v) = T_{S,\tau,c}^{-1}(\varsigma, z)$
  in the supremum, and using
  $\|(r,v)\|_{{\mathcal{{V}}_\alpha}} \le (1+c_P^{-1})^{1/2}\|(r,v)\|_{H(\operatorname*{div})\times\mathring{H}^1}$
  by~\eqref{eq:Friedrichs-std}, the two-sided
  bound~\eqref{eq:frozen-transposed-two-sided} gives
  $c_{\rm ub} \ge c_T (1+c_P^{-1})^{-1/2}$ and $\|b^0\| \le C_T$,
  whence~\eqref{eq:uw-cd}.

\label{rem:no-frechet}
  The present setting, in which $\rho_\alpha = \sigma_\alpha^{-1}$ is a
  function of a single vector argument, is a convenient one in which
  to record why the G\^ateaux differentiability required in
  Assumption~\ref{asm:NDPGdiff}{\rm(b)} was not strengthened to
  Fr{\'{e}}chet\ differentiability: the latter simply fails.  Suppose
  $\rho_\alpha$ is not affine.  Then there are a $V \in \mathbb{{R}}^d$, a
  $\eta>0$, and a direction vector $e \in \mathbb{{R}}^d$ with $|e|=1$,
  such that the vector
  $ R = \rho_\alpha(V + \eta e) - \rho_\alpha( V) -
  d\rho_\alpha(V)(\eta e) \ne 0 $. Let $\varpi_V$ be a smooth flux such
  that $\varpi_V \equiv V$ in a ball $B$ contained in an element $K$ of
  ${\varOmega_h}$.  For small $\varepsilon>0$, let $B_\varepsilon \subset B$ be the
  concentric ball of radius $\varepsilon$.  Let $\chi_\varepsilon$ be a smooth
  scalar function with $0 \le \chi_\varepsilon \le 1$, $\chi_\varepsilon \equiv 1$
  on $B_\varepsilon$, and support contained in $B_{(1+\theta)\varepsilon}$, for a
  $\theta \in (0,1)$ chosen below independently of $\varepsilon$.  Consider
  trial functions ${\mathcal{{w}}} =(\varpi_V, w, \hat w, \hat\varpi_n)$ and
  ${\mathcal{{z}}}_\varepsilon = (\eta e \chi_\varepsilon, 0, 0, 0)$ in ${\mathcal{{U}}_\alpha}$ and test
  function ${\mathcal{{v}}} = (r, 0)$ with $r \in H(\operatorname*{div}, {\varOmega_h})$ supported in
  $B_{2\varepsilon} \subset K$ satisfying $\operatorname{{\ensuremath{\text{div}}}} r = 0$, $r \equiv R$ on
  $B_{(1+\theta)\varepsilon}$, and $|r| \lesssim |R|$ throughout (e.g., make
  $r$ by extending $R$ as the flux of the solution of a Neumann
  problem on $B_{2\varepsilon}\setminus B_{(1+\theta)\varepsilon}$).
  By~\eqref{eq:DPGinfiniteBF}, \eqref{eq:dA}, and that
  $\chi_\varepsilon \equiv 1$ and $r \equiv R$ on $B_\varepsilon$,
  \begin{gather*}
    \nonumber
    \langle{\mathcal{{A}}_\alpha({\mathcal{{w}}} + {\mathcal{{z}}}_\varepsilon) - \mathcal{{A}}_\alpha({\mathcal{{w}}})
      - d\mathcal{{A}}_\alpha({\mathcal{{w}}})({\mathcal{{z}}}_\varepsilon), {\mathcal{{v}}}}\rangle
    \\
    = \big(\rho_\alpha(\varpi_V + \eta e \chi_\varepsilon)
    - \rho_\alpha(\varpi_V)
    - d\rho_\alpha(\varpi_V)(\eta e \chi_\varepsilon), r\big)_{{\varOmega_h}}
    \\
    = (g, r)_{L_2(B_{(1+\theta)\varepsilon})}
    =  |R|^2 |B_\varepsilon| + E_\varepsilon,
  \end{gather*}
  because the function
  $g = \rho_\alpha(\varpi_V + \eta e \chi_\varepsilon)
  - \rho_\alpha(\varpi_V) -
  d\rho_\alpha(\varpi_V)(\eta e \chi_\varepsilon)$ vanishes off
  $\operatorname{{\ensuremath{\text{supp}}}} \chi_\varepsilon$ and equals $R$ in $B_\varepsilon$. Here $E_\varepsilon$
  equals the integral of $g \cdot r$ over the shell
  $B_{(1+\theta)\varepsilon} \setminus B_\varepsilon$, whose volume is
  $\eqsim \theta \varepsilon^d$.  There $|r| = |R|$ and
  $|g| \le 2\alpha_2\beta_1^{-1}\eta$ by~\eqref{eq:globalized-rho}, so
  $|E_\varepsilon| \lesssim \theta\, \eta\, |R| \varepsilon^d$.  As
  $|B_\varepsilon| \eqsim \varepsilon^d$, choosing $\theta$ small enough in terms
  of the fixed ratio $|R|/\eta$ (independently of $\varepsilon$) gives
  $|R|^2 |B_\varepsilon| + E_\varepsilon\gtrsim \varepsilon^d |R|^2$.  Since
  $\| {\mathcal{{v}}}\|_{{\mathcal{{V}}_\alpha}} = \| r \|_{L_2} \lesssim |R| \varepsilon^{d/2}$ and
  $\| {\mathcal{{z}}}_\varepsilon\|_{{\mathcal{{U}}_\alpha}} \lesssim \eta \varepsilon^{d/2}$,
    \[
    \frac{\|\mathcal{{A}}_\alpha({\mathcal{{w}}} + {\mathcal{{z}}}_\varepsilon) - \mathcal{{A}}_\alpha({\mathcal{{w}}})
      - d\mathcal{{A}}_\alpha({\mathcal{{w}}})({\mathcal{{z}}}_\varepsilon)\|_{{\mathcal{{V}}'_\alpha}}}
    {\|{\mathcal{{z}}}_\varepsilon\|_{{\mathcal{{U}}_\alpha}}}
    \;\gtrsim\;
    \frac{\varepsilon^{d}\,|R|^2 }{ |R| \varepsilon^{d/2} }
    \frac{1}{\eta\, \varepsilon^{d/2}}
    \;=\; \frac{|R|}{\eta},
  \]
  which does not vanish as $\varepsilon \to 0$, even though
  $\|{\mathcal{{z}}}_\varepsilon\|_{{\mathcal{{U}}_\alpha}} \to 0$.  Hence $\mathcal{{A}}_\alpha$ is not Fr{\'{e}}chet\
  differentiable, even though it is G\^ateaux differentiable as proved
  in~(b): indeed, the direction ${\mathcal{{z}}}_\varepsilon/\|{\mathcal{{z}}}_\varepsilon\|_{{\mathcal{{U}}_\alpha}}$
  moves in ${\mathcal{{U}}_\alpha}$ with $\varepsilon$, whereas along any fixed direction, the
  derivative exists. This lack of Fr{\'{e}}chet\ differentiability reflects
  a general obstruction: a Nemytskii operator generated by a function
  $f$ is Fr{\'{e}}chet\ differentiable from $L_p$ into $L_p$ only if $f$ is
  affine~\cite{AppellZabrejko90}. Also recall that a G\^ateaux
  differentiable map whose derivative is continuous in the operator
  norm is automatically Fr{\'{e}}chet\ differentiable. Therefore our $d\mathcal{{A}}_\alpha$
  cannot be operator-norm continuous, which is the reason for the
  caveat on~{\rm(c$'$)} raised in Remark~\ref{rem:sufficient-c}.

\label{ex:mu-kirchhoff}
  Let $\kappa_\alpha$ and $\mathcal{{Q}}$ be the $m^2$-dimensional
  piecewise-constant family of Example~\ref{ex:semilin-vhp}, i.e.,
  $\kappa_\alpha = \bigl( \sum_{i=1}^{m^2} \alpha^{(i)} \chi_i \bigr) I$
  with $\alpha^{(i)} \in [\kappa_1, \kappa_2]$, so that~{\rm(H4)}
  holds, and let
  \begin{equation}
    \label{eq:carrier}
    M_\alpha(s) = M(s) := a + \frac{b\, s}{1 + \varepsilon s},
    \qquad a, b, \varepsilon > 0,
  \end{equation}
  independently of $\alpha$.  This is the Kirchhoff--Carrier law
  $a + b s$, saturated at large $s$: the two agree to $O(\varepsilon)$ on
  bounded ranges of $s$ and coincide as $\varepsilon \to 0$.  The saturation
  is what buys~{\rm(H1)}, which the unbounded law $a + bs$ violates,
  while the feature that matters physically, that $M$ increases with
  the elongation, is retained.  Indeed
  $M'(s) = b (1 + \varepsilon s)^{-2} > 0$, so $C_M^- = 0$
  and~{\rm(H2)} holds with no smallness condition whatsoever, for
  every $a, b, \varepsilon$.  The remaining constants are $M_1 = a$,
  $M_2 = a + b/\varepsilon$, and, since $s(1+\varepsilon s)^{-2}$ is maximized at
  $s = 1/\varepsilon$, $C_M^* = b/(4\varepsilon)$.  Finally $|M'| \le b$ and
  $|M''| \le 2\varepsilon b$ on all of $[0,\infty)$, so~{\rm(H3)} holds with
  $C_M(\bar s) = \max(b, 2\varepsilon b)$ for every $\bar s$.

  By contrast,

  $C_M^- = C_M^* = 1/4$, and~{\rm(H2)} then genuinely restricts the

\label{lem:kirchhoff-wellposed}
  Let Hypothesis~\ref{hyp:M} hold and $f \in L_2$.  For
  $\alpha \in \mathcal{{Q}}$ let $w_\alpha \in \mathring{H}^1$ be the solution of the
  linear problem
  $(\kappa_\alpha \nabla w_\alpha, \nabla v)_\varOmega = (f, v)_\varOmega$ for all
  $v \in \mathring{H}^1$, and put $g_\alpha := G_\alpha(w_\alpha)$.  Then the
  function $\psi_\alpha(s) := s\, M_\alpha(s)^2$ is a strictly
  increasing bijection of $[0, \infty)$, and letting
  $s^* = s^*(\alpha) \ge 0$ denote the unique root of
  $\psi_\alpha(s^*) = g_\alpha$, the function
  \begin{equation}
    \label{eq:KH-solution}
    u(\alpha) := M_\alpha(s^*)^{-1}\, w_\alpha
  \end{equation}
  is the unique solution of~\eqref{eq:KH-weak}.  Moreover, the exact
  flux is
  $M_\alpha(G_\alpha(u(\alpha)))\, \kappa_\alpha \nabla u(\alpha) =
  \kappa_\alpha \nabla w_\alpha$, and
  \begin{equation}
    \label{eq:KH-apriori}
    G_\alpha(u(\alpha)) = s^*(\alpha) \le \frac{g_\alpha}{M_1^2},
    \qquad
    g_\alpha \le \frac{\|f\|^2_{L_2}}{c_P\, \kappa_1} =: g_*,
  \end{equation}
  so that $s^*$ and $g_\alpha$ are bounded independently of $\alpha$.

<!-- proof-11: body_md -->
The function $\psi_\alpha$ is continuous, $\psi_\alpha(0) = 0$, and
  $\psi_\alpha(s) \ge M_1^2\, s \to \infty$, so $\psi_\alpha$ maps
  $[0,\infty)$ onto itself.  By Hypothesis~\ref{hyp:M}, using
  $s M_\alpha'(s) \ge - s (M_\alpha'(s))_- \ge -C_M^-$,
  \[
    \psi_\alpha'(s)
    = M_\alpha(s) \bigl( M_\alpha(s) + 2 s M_\alpha'(s) \bigr)
    \ge M_1 (M_1 - 2 C_M^-) > 0,
  \]
  so $\psi_\alpha$ is strictly increasing and the root $s^*$ exists
  uniquely.  For $u(\alpha)$ defined by~\eqref{eq:KH-solution},
  $G_\alpha(u(\alpha))
  = (\kappa_\alpha \nabla u(\alpha), \nabla u(\alpha))_\varOmega
  = M_\alpha(s^*)^{-2} g_\alpha =
  M_\alpha(s^*)^{-2} \psi_\alpha(s^*) = s^*$, so
  \[
    M_\alpha(G_\alpha(u(\alpha)))\,
    (\kappa_\alpha \nabla u(\alpha), \nabla v)_\varOmega
    = M_\alpha(s^*)\, M_\alpha(s^*)^{-1}
    (\kappa_\alpha \nabla w_\alpha, \nabla v)_\varOmega
    = (f, v)_\varOmega,
  \]
  i.e., $u(\alpha)$ solves~\eqref{eq:KH-weak}, and
  $M_\alpha(G_\alpha(u(\alpha)))\kappa_\alpha \nabla u(\alpha) =
  \kappa_\alpha \nabla w_\alpha$.  If $\tilde u$ is any solution
  of~\eqref{eq:KH-weak}, then, since
  $\tilde m := M_\alpha(G_\alpha(\tilde u))$ is positive, the
  uniqueness of $w_\alpha$ gives $\tilde u = \tilde m^{-1} w_\alpha$,
  so
  $\psi_\alpha(G_\alpha(\tilde u)) = \tilde m^2\, G_\alpha(\tilde u) =
  g_\alpha$, and the injectivity of $\psi_\alpha$ yields
  $G_\alpha(\tilde u) = s^*$, hence $\tilde u = u(\alpha)$.  Finally,
  $\psi_\alpha(s) \ge M_1^2 s$ gives $s^* \le g_\alpha/M_1^2$, while
  testing the equation for $w_\alpha$ with $v = w_\alpha$ and
  using~\eqref{eq:Friedrichs-std} and~{\rm(H4)},
  \[
    g_\alpha
    = (f, w_\alpha)_\varOmega
    \le c_P^{-1/2} \|f\|_{L_2} \|\nabla w_\alpha\|_{L_2}
    \le c_P^{-1/2} \|f\|_{L_2} \bigl( g_\alpha/\kappa_1 \bigr)^{1/2},
  \]
  which is the second bound in~\eqref{eq:KH-apriori}.

<!-- lemma-5: body_md -->
\label{lem:kirchhoff-frozen}
  Let Hypothesis~\ref{hyp:M} hold and set
  \begin{equation}
    \label{eq:KH-cK}
    c_K := (M_1 - 2 C_M^-)\, \kappa_1 > 0,
    \qquad
    C_K := (M_2 + 2 C_M^*)\, \kappa_2 .
  \end{equation}
  Then, for every $w, \zeta \in \mathring{H}^1$ and every $\alpha \in \mathcal{{Q}}$,
  \begin{equation}
    \label{eq:KH-coercive}
    \| B_w \zeta \|_{L_2}
    \le C_K\, \| \nabla \zeta \|_{L_2},
    \qquad
    ( B_w \zeta, \nabla \zeta)_\varOmega
    \ge c_K\, \| \nabla \zeta \|^2_{L_2} .
  \end{equation}
  Both constants are independent of $w$ and of $\alpha$.  Consequently
  the same two bounds hold for the averaged map
  $\int_0^1 B_{w^t}\, dt$ generated by an arbitrary family
  $\{ w^t : t \in [0,1] \} \subset \mathring{H}^1$.
<!-- end lemma-5 -->
<!-- proof-12: body_md -->
Since $\kappa_\alpha$ is symmetric and positive definite by
  {\rm(H4)}, the form $(\kappa_\alpha \nabla \cdot, \nabla\cdot)_\varOmega$
  is an inner product on $\mathring{H}^1$, so
  \begin{equation}
    \label{eq:KH-CS}
    | (\kappa_\alpha \nabla w, \nabla \zeta)_\varOmega |
    \le G_\alpha(w)^{1/2} G_\alpha(\zeta)^{1/2},
    \qquad
    \| \kappa_\alpha \nabla w \|_{L_2}^2
    \le \kappa_2\, G_\alpha(w).
  \end{equation}
  Hence the second term
  of~\eqref{eq:KH-frozen-form} has $L_2$ norm at most
  $2 |M_\alpha'(G_\alpha(w))|\, G_\alpha(w)\, \kappa_2^{1/2}
  G_\alpha(\zeta)^{1/2} \le 2 C_M^* \kappa_2 \|\nabla\zeta\|_{L_2}$
  by {\rm(H2)} and {\rm(H4)}, while the first is at most
  $M_2 \kappa_2^{1/2} G_\alpha(\zeta)^{1/2} \le M_2 \kappa_2
  \|\nabla\zeta\|_{L_2}$ by {\rm(H1)}.  This gives the first bound
  of~\eqref{eq:KH-coercive}.
  For the second, \eqref{eq:KH-CS} gives
  \begin{align*}
    ( B_w \zeta, \nabla \zeta)_\varOmega
    & = M_\alpha(G_\alpha(w))\, G_\alpha(\zeta)
      + 2 M_\alpha'(G_\alpha(w))
      (\kappa_\alpha \nabla w, \nabla\zeta)^2_\varOmega
    \\
    & \ge \bigl[ M_\alpha(G_\alpha(w))
      - 2\bigl(M_\alpha'(G_\alpha(w))\bigr)_- G_\alpha(w) \bigr]
      G_\alpha(\zeta).
  \end{align*}
  By {\rm(H1)},
  {\rm(H2)} and {\rm(H4)}, the right-hand side is at least
  $(M_1 - 2C_M^-) \kappa_1 \|\nabla\zeta\|^2_{L_2}$, which is the
  second bound  of~\eqref{eq:KH-coercive}.
  Since both hold pointwise in $t$ with
  $w$-independent constants, they also hold for the average.

{\em Proof of} {\rm(a)}.  Let $u = u(\alpha)$ be the solution
  of~\eqref{eq:KH-weak} given by
  Lemma~\ref{lem:kirchhoff-wellposed}.  Hence  the flux
  $q := M_\alpha(G_\alpha(u))\kappa_\alpha\nabla u = \kappa_\alpha\nabla w_\alpha$ satisfies
   $-\operatorname*{div} (\kappa_\alpha \nabla w_\alpha) = f \in L_2$
  distributionally, so $q \in H(\operatorname*{div})$ and ${\mathcal{{u}}} := (q, u) \in {\mathcal{{U}}_\alpha}$
  satisfies $\mathcal{{A}}_\alpha({\mathcal{{u}}}) = \ell$, i.e., $\mathcal{{R}}(\alpha, {{\mathcal{{u}}}}) = 0$.

  {\em Proof of} {\rm(b)}.  Since $G$ is a continuous quadratic form
  on $\mathring{H}^1$ and $M_\alpha \in C^1$ by {\rm(H1)}, the map
  $u \mapsto M_\alpha(G_\alpha(u))\kappa_\alpha\nabla u$ is Fr{\'{e}}chet\ differentiable from
  $\mathring{H}^1$ into $L_2^d$, with derivative $B_u$
  of~\eqref{eq:KH-frozen-form}.  Hence $\mathcal{{A}}_\alpha$ is Fr{\'{e}}chet\ differentiable, and a
  fortiori G\^ateaux differentiable at every $(q,u) \in {\mathcal{{U}}_\alpha}$, with
  \begin{equation}
    \label{eq:KH-fos-dA}
    d\mathcal{{A}}_\alpha(q,u)(\varsigma,z)
    = \bigl( \varsigma - B_u z, \; -\operatorname*{div} \varsigma \bigr),
    \qquad (\varsigma,z) \in {\mathcal{{U}}_\alpha} .
  \end{equation}
  By the first bound of~\eqref{eq:KH-coercive},
  $\|\varsigma - B_u z\|^2_{L_2} \le 2\|\varsigma\|^2_{L_2} + 2 C_K^2
  \|\nabla z\|^2_{L_2}$, so
  $\|d\mathcal{{A}}_\alpha(q,u)(\varsigma,z)\|^2_{{\mathcal{{V}}'_\alpha}}
  \le C_{d}^2 \|(\varsigma,z)\|^2_{{\mathcal{{U}}_\alpha}}$ with
  $C_{d}$ as claimed, which is~\eqref{linvc}.

  {\em Proof of} {\rm(c)}.  Let ${\mathcal{{z}}}^i = (\varsigma^i, z^i) \in {\mathcal{{U}}_\alpha}$,
  $i = 0,1$, be arbitrary, put $e_\varsigma = \varsigma^0 - \varsigma^1$,
  $e_z = z^0 - z^1$,
  and let $z^t := z^0 - t\, e_z$.  By {\rm(b)} the map
  $t \mapsto M_\alpha(G_\alpha(z^t))\kappa_\alpha \nabla z^t$ is continuously
  differentiable from $[0,1]$ into $L_2^d$, with derivative
  $-B_{z^t} e_z$, so the FTC gives
  \[
    M_\alpha(G_\alpha(z^0))\kappa_\alpha\nabla z^0
    - M_\alpha(G_\alpha(z^1))\kappa_\alpha\nabla z^1
    = \bar B\, e_z,
    \qquad
    \bar B := \int_0^1 B_{z^t}\, dt .
  \]
  Hence $\mathcal{{A}}_\alpha({\mathcal{{z}}}^0) - \mathcal{{A}}_\alpha({\mathcal{{z}}}^1) = (r_1, r_2)$ with
  $r_1 = e_\varsigma - \bar B e_z$ and $r_2 = -\operatorname*{div} e_\varsigma$; set
  $A_*^2 := \|r_1\|^2_{L_2} + \|r_2\|^2_{L_2}$.  Since
  $e_z \in \mathring{H}^1$ and $e_\varsigma \in H(\operatorname*{div})$, integration by parts gives
  \[
    (e_\varsigma, \nabla e_z)_\varOmega = -(\operatorname*{div} e_\varsigma, e_z)_\varOmega = (r_2, e_z)_\varOmega,
  \]
  so that
  \[
    ( \bar B e_z, \nabla e_z)_\varOmega
    = (e_\varsigma, \nabla e_z)_\varOmega - (r_1, \nabla e_z)_\varOmega
    = (r_2, e_z)_\varOmega - (r_1, \nabla e_z)_\varOmega
    \le A\, \|\nabla e_z\|_{L_2},
  \]
  where $A := \|r_1\|_{L_2} + c_P^{-1/2}\|r_2\|_{L_2}$
  and~\eqref{eq:Friedrichs-std} was used.  By the second bound
  of~\eqref{eq:KH-coercive}, applied to the average $\bar B$, the
  left-hand side is at least $c_K \|\nabla e_z\|^2_{L_2}$, whence
  $\|\nabla e_z\|_{L_2} \le A / c_K$.  Consequently
  \[
    \|e_\varsigma\|_{L_2}
    = \| r_1 + \bar B e_z \|_{L_2}
    \le \|r_1\|_{L_2} + C_K \|\nabla e_z\|_{L_2},
    \qquad
    \|\operatorname*{div} e_\varsigma\|_{L_2} = \|r_2\|_{L_2},
  \]
  and, since $\|r_1\|_{L_2} \le A$ and
  $\|r_2\|_{L_2} \le \sqrt{c_P}\, A$, adding the three bounds and
  using $A^2 \le (1 + c_P^{-1}) A_*^2$ gives
  $\|{\mathcal{{z}}}^0 - {\mathcal{{z}}}^1\|^2_{{\mathcal{{U}}_\alpha}} \le c_*^{-2} A_*^2$.  This
  is~\eqref{eq:c-prime} for all ${\mathcal{{z}}}^1 \in {\mathcal{{U}}_\alpha}$, i.e., with
  $\delta = +\infty$; as before, its global validity also shows that
  the ${\mathcal{{u}}}$ of part {\rm(a)} is the unique element of ${\mathcal{{U}}_\alpha}$ at which
  $\mathcal{{R}}(\alpha, {\cdot})$ vanishes.

Since $G$ is a continuous quadratic form on $\mathring{H}^1$, it is
  infinitely differentiable with
  $DG_\alpha(w)(\zeta) = 2 (\kappa_\alpha\nabla w, \nabla \zeta)_\varOmega$.  By {\rm(H1)} and
  the chain and product rules, $\mathcal{{A}}_\alpha$ is Fr{\'{e}}chet\ differentiable at
  every ${\mathcal{{w}}} = (w, \hat r_n) \in {\mathcal{{U}}_\alpha}$ with, for
  ${\mathcal{{z}}} = (\zeta, \hat\varsigma_n) \in {\mathcal{{U}}_\alpha}$ and ${\mathcal{{v}}} \in {\mathcal{{V}}_\alpha}$,
  \begin{equation}
    \label{eq:KH-dA}
    \begin{aligned}
      d\mathcal{{A}}_\alpha({\mathcal{{w}}})({\mathcal{{z}}})({\mathcal{{v}}})
      & = M_\alpha(G_\alpha(w))\, (\kappa_\alpha \nabla \zeta, \nabla {\mathcal{{v}}})_{{\varOmega_h}}
      - \langle{ \hat\varsigma_n, {\mathcal{{v}}}}\rangle_{\partial{\varOmega_h}}
      \\
      & \quad
      + 2 M_\alpha'(G_\alpha(w))\, (\kappa_\alpha \nabla w, \nabla \zeta)_\varOmega\,
      (\kappa_\alpha \nabla w, \nabla {\mathcal{{v}}})_{{\varOmega_h}} .
    \end{aligned}
  \end{equation}

  {\em Proof of {\rm(a)}.}  Since $-\operatorname*{div} (\kappa_\alpha\nabla w_\alpha) = f \in
  L_2$ holds distributionally, the flux $\kappa_\alpha\nabla w_\alpha$ is in
  $H(\operatorname*{div})$, so $\hat q_n = \mathop{\mathrm{tr}_n}(\kappa_\alpha\nabla w_\alpha)$ is well defined.  Using
  $M_\alpha(G_\alpha(u(\alpha)))\, \kappa_\alpha\nabla u(\alpha)
  = \kappa_\alpha\nabla w_\alpha$ from
  Lemma~\ref{lem:kirchhoff-wellposed} and elementwise integration by
  parts,
  $
    a_\alpha( {\mathcal{{u}}}(\alpha); {\mathcal{{v}}})
    = (\kappa_\alpha\nabla w_\alpha, \nabla {\mathcal{{v}}})_{{\varOmega_h}}
    - \langle{ \mathop{\mathrm{tr}_n}(\kappa_\alpha\nabla w_\alpha), {\mathcal{{v}}}}\rangle_{\partial {\varOmega_h}}
    = -(\operatorname*{div} (\kappa_\alpha\nabla w_\alpha), {\mathcal{{v}}})_\varOmega
    = (f, {\mathcal{{v}}})_\varOmega
    = \ell({\mathcal{{v}}})
  $
  for all ${\mathcal{{v}}} \in {\mathcal{{V}}_\alpha}$, so $\mathcal{{R}}(\alpha, {{\mathcal{{u}}}(\alpha)}) = 0$.  For
  uniqueness, suppose ${\mathcal{{w}}} = (\tilde u, \tilde q_n) \in {\mathcal{{U}}_\alpha}$ also
  satisfies $\mathcal{{A}}_\alpha({\mathcal{{w}}}) = \ell$.  Testing with conforming
  ${\mathcal{{v}}} \in \mathring{H}^1 \subset {\mathcal{{V}}_\alpha}$ annihilates the interface term,
  so $\tilde u$ solves~\eqref{eq:KH-weak}, and
  Lemma~\ref{lem:kirchhoff-wellposed} gives $\tilde u = u(\alpha)$.
  Then $\langle{\tilde q_n - \hat q_n, {\mathcal{{v}}}}\rangle_{\partial{\varOmega_h}} = 0$ for all
  ${\mathcal{{v}}} \in {\mathcal{{V}}_\alpha}$, which forces $\tilde q_n = \hat q_n$ by the duality
  underlying the quotient norm in~\eqref{eq:quotient-norms} (see
  also~\cite{CarstDemkoGopal16}).

  {\em Proof of {\rm(b)}.}  By~\eqref{eq:KH-CS} and
  {\rm(H1)}--{\rm(H2)}, exactly as in the first bound
  of~\eqref{eq:KH-coercive},
  \[
    | d\mathcal{{A}}_\alpha({\mathcal{{w}}})({\mathcal{{z}}})({\mathcal{{v}}}) |
    \le C_K\, \|\nabla\zeta\|_{L_2} \|{\mathcal{{v}}}\|_{{\mathcal{{V}}_\alpha}}
    + \|\hat\varsigma_n\|_{H^{-1/2}(\partial{\varOmega_h})} \|{\mathcal{{v}}}\|_{{\mathcal{{V}}_\alpha}}
    \le C_{d} \|{\mathcal{{z}}}\|_{{\mathcal{{U}}_\alpha}} \|{\mathcal{{v}}}\|_{{\mathcal{{V}}_\alpha}}
  \]
  with $C_{d} = [C_K^2 + 1]^{1/2}$,
  proving~\eqref{linvc}.

  {\em Proof of {\rm(c)}.}  Write
  $b_w(\zeta, v) := (B_w \zeta, \nabla v)_{{\varOmega_h}}$ for the
  bilinear form on $\mathring{H}^1 \times H^1({\varOmega_h})$ generated by the
  linearization~\eqref{eq:KH-frozen-form}, so that, by
  Lemma~\ref{lem:kirchhoff-frozen}, $b_w$ is bounded with constant
  $C_K$ and, when restricted to conforming $v$, coercive on
  $\mathring{H}^1$ with constant $c_K$, both uniformly in $w$
  and~$\alpha$.  Comparing with~\eqref{eq:KH-dA},
  $d\mathcal{{A}}_\alpha({\mathcal{{w}}})({\mathcal{{z}}})({\mathcal{{v}}}) = b_w(\zeta,{\mathcal{{v}}}) -
  \langle{\hat\varsigma_n, {\mathcal{{v}}}}\rangle_{\partial{\varOmega_h}}$.

  Now fix ${\mathcal{{z}}}^0 = (w^0, \hat r_n^0)$ and ${\mathcal{{z}}}^1 = (w^1, \hat r_n^1)$ in
  ${\mathcal{{U}}_\alpha}$, arbitrary, and put $e := w^0 - w^1$,
  $\hat e_n := \hat r_n^0 - \hat r_n^1$, and
  ${\mathcal{{z}}}^t := {\mathcal{{z}}}^1 + t({\mathcal{{z}}}^0 - {\mathcal{{z}}}^1)$ for $t \in [0,1]$.  Fix
  ${\mathcal{{v}}} \in {\mathcal{{V}}_\alpha}$.  By~(b) and the continuity of $M_\alpha$ and
  $M_\alpha'$ from {\rm(H1)}, the function
  $t \mapsto a_\alpha({\mathcal{{z}}}^t; {\mathcal{{v}}})$ is continuously differentiable on
  $[0,1]$, with derivative $d\mathcal{{A}}_\alpha({\mathcal{{z}}}^t)({\mathcal{{z}}}^0 - {\mathcal{{z}}}^1)({\mathcal{{v}}})$.
  Hence, by the FTC and~\eqref{eq:KH-dA},
  \begin{equation}
    \label{eq:KH-averaged}
    \langle{ \mathcal{{A}}_\alpha({\mathcal{{z}}}^0) - \mathcal{{A}}_\alpha({\mathcal{{z}}}^1), {\mathcal{{v}}}}\rangle
    = \int_0^1 d\mathcal{{A}}_\alpha({\mathcal{{z}}}^t)({\mathcal{{z}}}^0 - {\mathcal{{z}}}^1)({\mathcal{{v}}})\, dt
    = \bar b(e, {\mathcal{{v}}}) - \langle{ \hat e_n, {\mathcal{{v}}}}\rangle_{\partial{\varOmega_h}},
  \end{equation}
  where $\bar b(\zeta, v) := \int_0^1 b_{w^t}(\zeta, v)\, dt$ is the
  average of these forms along the segment, with $w^t$ denoting the
  $\mathring{H}^1$ component of ${\mathcal{{z}}}^t$.  By the last assertion of
  Lemma~\ref{lem:kirchhoff-frozen}, $\bar b$ is bounded and coercive
  with the {\em same} two constants.

  The right-hand side of~\eqref{eq:KH-averaged} is exactly the broken
  primal DPG form generated by $\bar b$ and the interface variable
  $\hat e_n$.  Therefore the theory of broken variational
  formulations~\cite[Theorem~3.3]{CarstDemkoGopal16} converts the
  Lax--Milgram well-posedness of $\bar b$ into the inf-sup estimate
  \begin{equation}
    \label{eq:inf-sup-kirchhoff}
    \sup_{0 \ne {\mathcal{{v}}} \in {\mathcal{{V}}_\alpha}}
    \frac{ | \langle{ \mathcal{{A}}_\alpha({\mathcal{{z}}}^0) - \mathcal{{A}}_\alpha({\mathcal{{z}}}^1), {\mathcal{{v}}}}\rangle |}{ \|{\mathcal{{v}}}\|_{{\mathcal{{V}}_\alpha}}}
    \ge c_{d}\, \| {\mathcal{{z}}}^0 - {\mathcal{{z}}}^1 \|_{{\mathcal{{U}}_\alpha}},
  \end{equation}
  with $c_{d}$ as in~\eqref{eq:KH-cd}, and in particular independent of
  ${\mathcal{{z}}}^0, {\mathcal{{z}}}^1$ and of $\alpha \in \mathcal{{Q}}$.  That reference gives
  $c_{d}^{-2} = c_{\rm ub}^{-2} + \hat c^{-2} ( \|\bar b\|/c_{\rm ub} + 1)^2$, where
  $c_{\rm ub}$ and $\|\bar b\|$ are the inf-sup and continuity constants of
  $\bar b$ on the {\em unbroken} pair $\mathring{H}^1 \times \mathring{H}^1$ and
  $\hat c$ is the stability constant of the interface term.  Here
  $\hat c = 1$, since the quotient norm~\eqref{eq:quotient-norms} is
  the dual norm of that term, while Lemma~\ref{lem:kirchhoff-frozen}
  and the Friedrichs inequality~\eqref{eq:Friedrichs-std}, which gives
  $\|\zeta\|_{H^1(\varOmega)} \le (1 + c_P^{-1})^{1/2}\|\nabla\zeta\|_{L_2}$,
  yield $\|\bar b\| \le C_K$ and
  $c_{\rm ub} \ge c_K (1 + c_P^{-1})^{-1/2}$, whence~\eqref{eq:KH-cd}.  Taking ${\mathcal{{z}}}^0 = {\mathcal{{u}}}(\alpha)$
  gives~\eqref{eq:c-prime} for every ${\mathcal{{z}}} = {\mathcal{{z}}}^1 \in {\mathcal{{U}}_\alpha}$, that is,
  {\rm(c)} with $c_* = c_{d}$ and $\delta = +\infty$.  (Since the
  estimate holds for all pairs, it also re-derives the uniqueness
  asserted in {\rm(a)}.)

<!-- proof-16: body_md -->
Let ${{\mathcal{{w}}}_h} = (w_h, \hat r_{n,h}) \in {\mathcal{{U}}_{\alpha, h}}$ and ${\mathcal{{v}}} \in {\mathcal{{V}}_\alpha}$.  The
  Fortin condition~\eqref{eq:Fortin} reads
  \[
    M_\alpha( G_\alpha(w_h))\,
    (\kappa_\alpha \nabla w_h, \nabla( {\mathcal{{v}}} - \varPi {\mathcal{{v}}}))_{{\varOmega_h}}
    - \langle{ \hat r_{n,h}, {\mathcal{{v}}} - \varPi {\mathcal{{v}}}}\rangle_{\partial{\varOmega_h}}
    = 0.
  \]
  By~\cite[Lemma~3.2]{GopalQiu14} for $k\ge1$ and
  \eqref{eq:fortin-preservation} for $k=0$, this uniformly bounded
  operator satisfies, on every $K \in {\varOmega_h}$,
  \begin{equation}
    \label{eq:KH-fortin-orth}
    ({\mathcal{{v}}} - \varPi {\mathcal{{v}}}, y_h)_K = 0, \;\;
    \qquad
    \langle{ {\mathcal{{v}}} - \varPi {\mathcal{{v}}}, \mu_h}\rangle_{\partial K} = 0
  \end{equation}
  for all $y_h\in   P_{k-1}(K)$ and $\mu_h \in P_k(\partial K).$
  Since $\hat r_{n,h}|_{\partial K}$ is in $P_k(\partial K)$, the
  second term of the Fortin condition vanishes by the second
  orthogonality.  For the first, $w_h \in U_{k+1}$ and the piecewise
  constancy of $\kappa_\alpha$ put $\kappa_\alpha \nabla w_h$ in
  $P_k^d(K)$, so elementwise integration by parts gives
  \[
    (\kappa_\alpha \nabla w_h, \nabla( {\mathcal{{v}}} - \varPi {\mathcal{{v}}}))_K
    = - (\operatorname*{div}(\kappa_\alpha \nabla w_h), {\mathcal{{v}}} - \varPi {\mathcal{{v}}})_K
    + \langle{ \kappa_\alpha \nabla w_h \cdot n, {\mathcal{{v}}} - \varPi {\mathcal{{v}}}}\rangle_{\partial K},
  \]
  where $\operatorname*{div}(\kappa_\alpha \nabla w_h) \in P_{k-1}(K)$ and
  $\kappa_\alpha \nabla w_h \cdot n \in P_k(\partial K)$, so both
  terms vanish by~\eqref{eq:KH-fortin-orth}.  Hence the two terms of
  the Fortin condition vanish separately---the first for any value of
  the scalar $M_\alpha(G_\alpha(w_h))$---so~\eqref{eq:Fortin} holds
  for every $k \ge 0$, with no restriction on the nonlinearity
  $M_\alpha$.

\label{thm:vc-NDPG-Kirchhoff}
  Let Hypothesis~\ref{hyp:M} hold, adopt the
  setting~\eqref{eq:KH-DPG} and~\eqref{eq:KH-discrete}, and let
  $\kappa_\alpha$ be piecewise constant on ${\varOmega_h}$.  Then the
  NDPG loss function $\mathcal{{R}}_h(\alpha, {\cdot})$ of~\eqref{eq:DPG-loss-defn} is
  {\em globally} variationally correct, with the $\alpha$- and
  mesh-independent constants $c = c_{d}^2/2$ and $C = C_{d}^2 = C_K^2 + 1$,
  where $c_{d}$ is given by~\eqref{eq:KH-cd} in terms of $c_K$, $C_K$
  and $c_P$: the
