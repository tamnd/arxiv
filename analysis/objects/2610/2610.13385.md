\label{def:LR}
		A \emph{linear relation} $T$ in $H$ is a linear subspace of $H \times H$. We denote by $LR(H)$ the set of all linear relations on $H$. The domain, range, and multivalued part of $T$ are defined respectively by:
		\begin{align*}
			D(T) &= \{x \in H \mid (x,y) \in T \text{ for some } y \in H\},\\
			R(T) &= \{y \in H \mid (x,y) \in T \text{ for some } x \in H\},\\
			T(0) &= \{y \in H \mid (0,y) \in T\}.
		\end{align*}
		For $x \in D(T)$, we write $T(x) = \{y \in H \mid (x,y) \in T\}$. The graph of $T$ is $G(T) = T$ itself. $T$ is \emph{closed} if $G(T)$ is closed in $H \times H$.
<!-- end defn-1 -->
<!-- defn-2: body_md -->
\label{def:compact}
		A linear relation $T \in LR(H)$ is \emph{compact} if $QT$ is a compact operator. That is, for every bounded sequence $(x_n)$ in $D(T)$, the sequence $(QT(x_n))$ has a convergent subsequence in $H/\overline{T(0)}$.
<!-- end defn-2 -->
<!-- defn-3: body_md -->
\label{def:norm}
		For $x \in D(T)$, the norm of $T(x)$ is
		\[
		\|T(x)\| = \|(QT)(x)\|_{H/\overline{T(0)}} = \inf_{y \in T(x)} \|y\|.
		\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
		The norm of $T$ is $\|T\| = \sup\{\|T(x)\| : x \in D(T),\, \|x\| \le 1\}$. A linear relation $T$ is \emph{continuous} if $\|T\| < \infty$ and \emph{bounded} if it is continuous with $D(T) = H$.
<!-- end defn-3 -->
<!-- prop-1: body_md -->
\label{prop:compact-seq}
		A linear relation $T \in LR(H)$ is compact if and only if for every bounded sequence $(x_n)$ in $D(T)$, there exist a subsequence $(x_{n_k})$ and a sequence $(y_{n_k})$ with $y_{n_k} \in T(x_{n_k})$ such that $(y_{n_k})$ converges in $H$.
<!-- end prop-1 -->

\label{arens}

		\[
		T = T_s \oplus T_\infty,

			T_s &= T \cap \bigl(T(0)^\perp \times T(0)^\perp\bigr).
		\end{align*}

		\[
		D(T_s) = D(T) \cap T(0)^\perp, \qquad R(T_s) \subset T(0)^\perp,

For a Hermitian relation $T \subset T^*$, we have $T(0) \subset T^*(0)$. To see this, let $x\in T(0)$, then $(x,0)\in T$, and hence $(x,0)\in T^*$, which gives $x\in T^*(0)$.
		This inclusion is essential for the orthogonality of the decomposition.



        
	
		
		Define $T_\infty = \{0\} \times T(0)$ and $T_s = T \cap (T(0)^\perp \times T(0)^\perp)$. 
		
		
		\noindent If $(x,y) \in T_s$ and $(0,z) \in T_\infty$, then $y \in T(0)^\perp$ and $z \in T(0)$, so\\ ${\langle (x,y), (0, z) \rangle}_{H \times H} = 0$. Hence we achieve the orthogonality mentioned in the statement. 		
		
		
		For the decomposition of $T$, consider $(x,y) \in T$. Decompose $y = y_1 + y_2$ with $y_1 \in T(0)^\perp$ and $y_2 \in T(0)$ (closedness of $T(0)$ follows from closedness of $T$). Since $y_2 \in T(0)$, we have $(0, y_2) \in T$. Therefore $(x,y_1)=(x,y)-(0,y_2)\in T$. Moreover, $y_1\in T(0)^\perp$, we have $(x,y_1)\in T\cap(H\times T(0)^\perp)$.\\
        To conclude $(x, y_1) \in T_s$, we need $x \in T(0)^\perp$. For that, let $z \in T(0),$ then $z \in T^*(0)$, that is, $(0, z) \in T^*$   ; since $T$ is Hermitian, for any $(x, y_1) \in T$ , we have $\langle y_1, 0\rangle = \langle x, z\rangle$, that is $ \langle x, z\rangle=0$. This means $x \perp T(0)$, that is, $x \in T(0)^\perp$. Thus $(x, y_1) \in T_s$, and the decomposition $T = T_s \oplus T_\infty$ is proved.
		
		Finally, we show that  $T_s$ is single-valued. If $(x, y_1), (x, y_1') \in T_s$, then $(0, y_1 - y_1') \in T$, so $y_1 - y_1' \in T(0)$. But $y_1, y_1' \in T(0)^\perp$, so $y_1 - y_1' \in T(0) \cap T(0)^\perp = \{0\}$. Thus $y_1 = y_1'$, proving $T_s$ is single-valued.

        
Also, we have
$
D(T_s)=D(T)\cap T(0)^\perp.
$
To prove this, let $x\in D(T_s)$. Then there exists $y_1\in T(0)^\perp$ such that $(x,y_1)\in T_s$. By the definition of $T_s$, we have $(x,y_1)\in T$, and hence $x\in D(T)$. Also, $x\in T(0)^\perp$. Thus
$
x\in D(T)\cap T(0)^\perp,
$
and consequently
$
D(T_s)\subseteq D(T)\cap T(0)^\perp.
$ Conversely, let $x\in D(T)\cap T(0)^\perp$. Then there exists $y\in H$ such that $(x,y)\in T$. Write
$
y=y_1+y_2,\qquad y_1\in T(0)^\perp,\quad y_2\in T(0).
$
As shown above, $(x,y_1)\in T_s$. Hence $x\in D(T_s)$, and therefore
$
D(T)\cap T(0)^\perp\subseteq D(T_s).
$
Thus,
$
D(T_s)=D(T)\cap T(0)^\perp.
$\\
Furthermore, $R(T_s) \subset T(0)^\perp$ follows directly from the definition of $T_s$.

\label{lem:converse-self-adjoint}
	Let $H$ be a Hilbert space, let $V \subseteq H$ be a closed subspace, and let
	$S$ be a self-adjoint operator on $V^\perp$ (that is, $D(S) \subseteq V^\perp$
	is dense in $V^\perp$ and $S = S^*$ computed within $V^\perp$). Define
	\[
	T := \{(x, Sx + v) : x \in D(S),\ v \in V\} \subset H \times H.
	\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
	Then $T$ is a self-adjoint linear relation in $H$, with $D(T) = D(S)$,
	$T(0) = V$, and $T_s = S$.
<!-- end lem-2 -->
<!-- proof-2: body_md -->
$T$ is clearly a linear subspace of $H\times H$, so $T \in LR(H)$.
	
	First we notice that $T(0)=V:$ since $0 \in D(S)$ and $S$ is linear, $S(0)=0$. Hence
	$(0,y) \in T$ if and only if $y = S(0)+v = v$ for some $v \in V$, that is if and only if $y \in V$.
	Thus $T(0)=V$.
	
	Next we see that $T^* = T.$ Let $(f,g) \in T^*$, that is \
	\begin{equation}\label{eq:conv-defn}
		\langle g,x\rangle = \langle f,\,Sx+v\rangle
		\qquad\text{for all } x\in D(S),\ v\in V.
	\end{equation}
	
	Then $f \in V^\perp:$ taking $x=0$ in \eqref{eq:conv-defn} gives
	$0=\langle g,0\rangle=\langle f,v\rangle$ for every $v \in V$.
	
	
	Write $g=g_V+g_{V^\perp}$ with $g_V\in V$, $g_{V^\perp}\in V^\perp$ (using
	$H=V\oplus V^\perp$, $V$ closed). Since $f\in V^\perp$,
	$\langle f,v\rangle=0$ for all $v\in V$, so \eqref{eq:conv-defn} reduces to
	\[
	\langle g,x\rangle=\langle f,Sx\rangle
	\qquad\text{for all } x\in D(S).
	\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
	As $D(S)\subseteq V^\perp$, we have $\langle g,x\rangle=\langle g_{V^\perp},x\rangle$,
	so
	\[
	\langle g_{V^\perp},x\rangle=\langle f,Sx\rangle
	\qquad\text{for all } x\in D(S),
	\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
	that is\ $(f,g_{V^\perp})\in S^*$ (adjoint computed within $V^\perp$). Since $S$
	is self-adjoint, $S=S^*$, so $f\in D(S)$ and $g_{V^\perp}=Sf$.
	
	Therefore, for any $f\in D(S)$ and any $g_V\in V$, the pair
	$(f,\,Sf+g_V)$ satisfies \eqref{eq:conv-defn}. Hence for all $x\in D(S)$, $v\in V$,
	\[
	\langle Sf+g_V,\,x\rangle
	=\langle Sf,x\rangle
	=\langle f,Sx\rangle
	=\langle f,Sx+v\rangle,
	\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
	using $x\in D(S)\subseteq V^\perp \Rightarrow \langle g_V,x\rangle=0$,
	self-adjointness of $S$, and $f\in V^\perp \Rightarrow \langle f,v\rangle=0$.
	
	Combining the above calculations, we get,
	\[
	T^* = \{(f,\,Sf+g_V) : f\in D(S),\ g_V\in V\} = T.
	\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
	
	Finally, $D(T)=D(S)$ is immediate from the definition of $T$, and $T_s = T \cap (T(0)^\perp\times T(0)^\perp) = T\cap(V^\perp\times V^\perp)$
	consists exactly of the pairs $(x,Sx)$ with $x\in D(S)$ (taking $v=0$ in the
	definition of $T$, and noting $Sx+v\in V^\perp$ forces $v=0$ since $Sx\in V^\perp$),
	so $T_s=S$.

\label{compact-selection}
	Let $H$ be a Hilbert space and let $T \in LR(H)$ be compact, bounded, and closed. Then $T$ admits a compact linear selection $A_0: H \to H$.
<!-- end lem-4 -->
<!-- proof-4: body_md -->
Since $T$ is bounded, we have $D(T) = H$. Let $Q: H \to H/\overline{T(0)}$ denote the quotient map.
	
First we show that $T(0)$ is closed.
	
	Indeed, $T(0) = \{y \in H : (0, y) \in T\}$ is the preimage of $T$ under the continuous map $y \mapsto (0, y)$. Since $T$ is closed, its preimage is closed. Alternatively, if $y_n \in T(0)$ and $y_n \to y$, then $(0, y_n) \to (0, y)$ in $H \times H$. Since $T$ is closed, $(0, y) \in T$, so $y \in T(0)$. Therefore, $T(0) = \overline{T(0)}$.
	

	
	Since $T$ is compact, $QT: H \to H/T(0)$ is a compact operator (here we use $T(0) = \overline{T(0)}$). Since $T$ is bounded, $QT$ is bounded. Therefore, $QT$ is a compact bounded linear operator from $H$ to the Hilbert space $H/T(0)$.
	

    
Let $x+T(0)\in H/T(0)$. Since $T(0)$ is closed, by the projection theorem, for every $x\in H$ there exist unique $x_1\in T(0)^\perp$ and $x_2\in T(0)$ such that
$x=x_1+x_2$. Then, $x+T(0)=x_1+x_2+T(0)=x_1+T(0)$.
\begin{center}
    Define, $R:H/T(0)\longrightarrow H$ by
$R(x+T(0))=x_1$.
\end{center}
Then $(QR)(x+T(0))=Q(x_1)=x_1+T(0)=x+T(0)$. Hence, $QR=I_{H/T(0)}$.

	That is, $R$ be the  right inverse of $Q$ mapping $H/T(0)$ onto $T(0)^\perp$. 
	
	Define:
	\[
	A_0 := R \circ QT.
	\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
	Then $A_0: H \to H$ is compact (composition of compact $QT$ with bounded $R$) and linear.
	
	Now we show that $A_0$ is a selection of $T$. Let $x \in H$, since $D(T) = H$, there exists $y \in T(x)$. We have:
	\[
	Qy = QTx = Q(R(QTx)) = QA_0x.
	\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
	Therefore, $y - A_0x \in T(0)$ (since $Q(y - A_0x) = Qy - QA_0x = 0$ in $H/T(0)$).
	
	Since $(x, y) \in T$ and $(0, y - A_0x) \in T$ (because $y - A_0x \in T(0)$), the linearity of $T$ gives:
	\[
	(x, A_0x) = (x, y) - (0, y - A_0x) \in T.
	\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
	Therefore, $A_0x \in T(x)$ for all $x \in H$, proving that $A_0$ is a selection of $T$.
	
	finally we show that $A_0$ is compact.
	
	Since $A_0 = R \circ QT$ where $R$ is bounded and $QT$ is compact, $A_0$ is compact.
	
	Therefore, $A_0: H \to H$ is a compact linear selection of $T$.

<!-- proof-7: body_md -->
Since $T$ is self-adjoint, $T(0) = T^*(0)$ automatically, and by Proposition~ 
	\IfBooleanTF{#1}{\@kernel@sref}{\@kernel@ref} 
{arens} (in its self-adjoint form), $T$ decomposes as $T = T_s \oplus T_\infty$, where $T_s$ is a single-valued compact self-adjoint operator on the Hilbert space $K := T(0)^\perp$, with $D(T_s) = D(T) \cap T(0)^\perp$, and $T_\infty = \{0\} \times T(0)$. By Remark~ 
	\IfBooleanTF{#1}{\@kernel@sref}{\@kernel@ref} 
{rem:key-observation}, $D(T) \subset T(0)^\perp$, so in fact $D(T) = D(T_s)$.
		
		\textbf{(a)} For $x \in D(T) = D(T_s)$, we have $T(x) = T_s(x) + T(0)$, so
		\[
		(I-T)(x) = x - T_s(x) - T(0) = (I - T_s)(x) - T(0).
		\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
		Hence $y \in (I-T)(x)$ if and only if $(I-T_s)(x) = y + z_0$ for some $z_0 \in T(0)$, that is, if and only if $y \in (I-T_s)(x) - T(0)$.
		
		By the classical Fredholm alternative for the compact self-adjoint operator $T_s$ on $K$, exactly one of the following holds:
		\begin{enumerate}
			\@par\hangindent 2em[(i$_s$)] $I - T_s$ is a bijection of $K$ onto $K$;
			\@par\hangindent 2em[(ii$_s$)] $N(I-T_s) \neq \{0\}$ in $K$.
		\end{enumerate}
		
		Suppose (i$_s$) holds. Given any $y \in H$, decompose $y = y_K + y_0$ with $y_K \in K$, $y_0 \in T(0)$. Since $I - T_s$ is onto $K$, there exists $x \in D(T_s)$ with $(I-T_s)(x) = y_K$. Then
		\[
		y = y_K + y_0 = (I-T_s)(x) + y_0 \in (I-T_s)(x) - T(0) = (I-T)(x),
		\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
		since $-y_0 \in T(0)$. This proves (a)(i).\\ For uniqueness modulo $T(0)$: if $x_1, x_2 \in D(T_s)$ both satisfy $y \in (I-T)(x_j)$, then $(I-T_s)(x_1) - (I-T_s)(x_2) \in T(0)$. But $(I-T_s)(x_1) - (I-T_s)(x_2) \in K$ (as $T_s$ maps into $K$ and $x_1, x_2 \in K$), and $K \cap T(0) = \{0\}$, so $(I-T_s)(x_1 - x_2) = 0$. Since $I - T_s$ is injective on $K$ (from (i$_s$)), then $x_1 = x_2$; in particular the solution is unique in $D(T_s)$.
		
		Suppose instead (ii$_s$) holds: there is a nonzero $u \in D(T_s) \subset D(T)$ with $T_s(u) = u$. Then $u \in T_s(u) + T(0) = T(u)$, so $u \in T(u)$, proving (a)(ii). Moreover,
		\[
		N(I-T) = \{x \in D(T) \mid 0 \in (I-T)(x)\} = \{x \in D(T_s) \mid (I-T_s)(x) \in T(0)\}.
		\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
		Since $(I-T_s)(x) \in K$ for $x \in D(T_s)$ and $K \cap T(0) = \{0\}$, this forces $(I-T_s)(x) = 0$, that is $N(I-T) = N(I-T_s)$. By the classical Fredholm alternative for the compact operator $T_s$, $Dim( N(I-T_s)) < \infty$, so $Dim (N(I-T) )< \infty$.
		
		Finally, (i) and (ii) cannot both hold, since (i$_s$) and (ii$_s$) are mutually exclusive and each of (a)(i), (a)(ii) forces the corresponding one of (i$_s$), (ii$_s$).
		
	
		
		\textbf{(b)} By part (a), the inhomogeneous inclusion $y \in (I-T)(x)$ has a solution for every $y \in H$ if and only if alternative (a)(i) holds, that is\ if and only if (i$_s$) holds, that is\ if and only if (ii$_s$) fails, that is\ if and only if $N(I-T_s) = \{0\}$. By the identity $N(I-T) = N(I-T_s)$ from part (a), this is equivalent to $N(I-T) = \{0\}$, that is\ to the homogeneous inclusion $0 \in (I-T)(x)$ having only the zero solution (indeed, since $N(I-T) \subset D(T_s) \subset T(0)^\perp$ meets $T(0)$ only at $0$).

			\@par\hangindent 2em[(iv)] $\sigma(T_s) \setminus \{0\} = \{s_n\}_{n=1}^N$, each $s_n$ has finite multiplicity, and if $N = \infty$, then $s_n \to 0$.
			
		\end{enumerate}
		
		Since $T_s \subset T$, property (i) gives $(u_n, s_n u_n) \in T$, which means $T(u_n) = s_n u_n + T(0)$. Thus $u_n$ is an eigenvector of $T$ with eigenvalue $s_n$. This proves part (a). The orthogonality follows from the self-adjointness of $T_s$. Also property (ii) proves part (b).
		
	
To prove  part (c), consider $x \in D(T) = D(T_s)$ and $y \in T(x)$. Since $T = T_s \oplus T_\infty$, we have $y \in T_s(x) + T(0)$. Thus there exists $y_\infty \in T(0)$ with $y = T_s(x) + y_\infty$. By part (b), this gives (3.1).
		
	Proof of part (d) is as follows.
		By Theorem~ 
	\IfBooleanTF{#1}{\@kernel@sref}{\@kernel@ref} 
{thm:spectrum-decomp}(ii), $\sigma(T)=\sigma(T_s)$. Property (iv) of the classical compact self-adjoint spectral theorem gives
		\[
		\sigma(T_s)\setminus\{0\}=\{s_n\}_{n=1}^N,
		\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
		with each nonzero spectral value having finite multiplicity and with $0$ as the only possible accumulation point. Therefore
		\[
		\sigma(T)\setminus\{0\}=\{s_n\}_{n=1}^N.
		\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
		
	Now we prove part (e). From the classical spectral theorem, $\{u_n\}_{n=1}^N$ is an orthonormal basis for $\overline{R(T_s)} = N(T_s)^\perp$ in $H_s$. Let $\{v_\alpha\}$ be an orthonormal basis of $N(T_s)$. Then $\{u_n\}_{n=1}^N \cup \{v_\alpha\}$ is an orthonormal basis of $H_s = T(0)^\perp$.
		
		We verify that $N(T_s) = N(T)$. For $x \in D(T) = D(T_s)$, the condition $0 \in T(x)$ means $T_s(x) \in T(0)$ (since $T(x) = T_s(x) + T(0)$). But $T_s(x) \in T(0)^\perp$, so $T_s(x) \in T(0) \cap T(0)^\perp = \{0\}$, hence $T_s(x) = 0$. Thus $x \in N(T_s)$. Conversely, if $x \in N(T_s)$, then $(x, 0) \in T_s \subset T$, so $0 \in T(x)$, giving $x \in N(T)$. Therefore $N(T_s) = N(T)$.
		
		Each $v_\alpha \in N(T_s) = N(T)$ satisfies $(v_\alpha, 0) \in T_s \subset T$, so $v_\alpha$ is an eigenvector of $T$ with eigenvalue $0$. Each $u_n$ is an eigenvector of $T$ with eigenvalue $s_n$ as shown previously.
		
		Finally, let $\{w_\beta\}$ be an orthonormal basis of $T(0)$. Since $H = T(0)^\perp \oplus T(0)$, the union $\{u_n\} \cup \{v_\alpha\} \cup \{w_\beta\}$ is an orthonormal basis of $H$. The vectors $u_n$ and $v_\alpha$ are eigenvectors of $T$. The vectors $w_\beta$ form an orthonormal basis of the multivalued part $T(0)$ and are not eigenvectors of $T$ in the usual sense.

<!-- rem-6: body_md -->
The proof shows that the spectral theorem for compact self-adjoint linear relations is a direct consequence of the Arens decomposition and the classical operator theorem. The multivalued part $T(0)$ contributes only the $y_\infty$ term in the representation (3.1) and the additional multivalued directions $\{w_\beta\}$. No finite-dimensionality condition on $T(0)$ is required anywhere in the proof; the only requirement is that $T_s$ be a compact self-adjoint operator, which follows automatically from the compactness and self-adjointness of $T$.

\label{lem:properties}
		The linear relation $T = A^{-1}$ is self-adjoint, compact, and $Dim T(0) = 1$.
<!-- end lem-7 -->
<!-- proof-9: body_md -->
The inverse of a self-adjoint linear relation is self-adjoint \@nx\protect\@empty\@nx\csname cite \endcsname{21}. Explicitly, $T^* = (A^{-1})^* = (A^*)^{-1} = A^{-1} = T$, and hence the relation $T = A^{-1}$ is self-adjoint.
		
To prove the compactness of a linear relation $T$, we have to show the compactness of the induced operator $QT$, where $Q: L^2(\Omega) \to L^2(\Omega)/\overline{T(0)}$ is the quotient map. Since $T(0) = \operatorname{span}\{\mathbf 1\}$ is already closed, $\overline{T(0)} = T(0)$. Let $(f_n)$ be a bounded sequence in $D(T) = R(A)$. For each $n$, choose $u_n \in T(f_n)$; then $-\Delta u_n = f_n$ and $\partial_\nu u_n = 0$. Decompose $u_n = u_n^0 + c_n \mathbf{1}$, where
		\[
		u_n^0 = u_n - \frac{1}{|\Omega|} \int_\Omega u_n \, dx
		\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
		has zero mean and $c_n = \frac{1}{|\Omega|} \int_\Omega u_n \, dx$. Since $Q$ annihilates $\operatorname{span}\{\mathbf 1\}$, we have $QT(f_n) = Q(u_n) = Q(u_n^0)$, so it suffices to produce an $L^2$-convergent subsequence of $(u_n^0)$; the constants $c_n$ play no role in the quotient.
		
		We show $(u_n^0)$ is bounded in $H^1(\Omega)$. Since $f_n \in R(A) = N(A)^\perp$, we have $\int_\Omega f_n \, dx = 0$, so $\langle f_n, \mathbf{1} \rangle = 0$. Using Green's identity and the Neumann boundary condition:
		\[
		\int_\Omega |\nabla u_n|^2 \, dx = \int_\Omega (-\Delta u_n) u_n \, dx = \langle f_n, u_n \rangle = \langle f_n, u_n^0 \rangle + c_n \langle f_n, \mathbf{1} \rangle = \langle f_n, u_n^0 \rangle.
		\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
		By the Cauchy--Schwarz inequality and the Poincar\'e inequality for zero-mean functions ($\|u_n^0\|_{L^2} \le C_P \|\nabla u_n^0\|_{L^2} = C_P \|\nabla u_n\|_{L^2}$), we get
		\[
		\|\nabla u_n\|_{L^2}^2 \le \|f_n\|_{L^2} \|u_n^0\|_{L^2} \le C_P \|f_n\|_{L^2} \|\nabla u_n\|_{L^2}.
		\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
		Since $(f_n)$ is bounded, this gives $\|\nabla u_n\|_{L^2} \le C_P \sup_n \|f_n\|_{L^2} < \infty$. By the Poincar\'e inequality again,
		\[
		\|u_n^0\|_{H^1(\Omega)}^2 = \|u_n^0\|_{L^2}^2 + \|\nabla u_n\|_{L^2}^2 \le (1+C_P^2)\|\nabla u_n\|_{L^2}^2 \le C,
		\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
		so $(u_n^0)$ is bounded in $H^1(\Omega)$.
		
		By the Rellich--Kondrachov theorem, the embedding $H^1(\Omega) \hookrightarrow L^2(\Omega)$ is compact, so there exists a subsequence $(u_{n_k}^0)$ converging in $L^2(\Omega)$ to some $u_*^0 \in L^2(\Omega)$. Consequently $Q(u_{n_k}^0) \to Q(u_*^0)$ in $L^2(\Omega)/T(0)$, that is $QT(f_{n_k}) = QT$ applied to the subsequence converges. Thus $QT$ is compact, and $T$ is compact.
		
		\emph{Dimension:} Clearly $Dim~ T(0) = Dim~ N(A) = 1$, since $N(A) = \operatorname{span}\{\mathbf{1}\}$.

\label{rem:generalization}
		The same technique applies to any self-adjoint elliptic operator with compact resolvent that has a finite-dimensional kernel. For example, the Robin problem
		\[
		\partial_\nu u + \alpha u = 0 \quad \text{on } \partial\Omega, \qquad \alpha \in \mathbb{R},
		\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
		has a non-trivial kernel precisely when $-\alpha$ is an eigenvalue of the associated Steklov problem
		\[
		\Delta v = 0 \text{ in } \Omega, \qquad \partial_\nu v = \sigma v \text{ on } \partial\Omega;
		\relax
    \ifmmode
      \egroup \m@th$\hfil \egroup
      \end{trivlist} 
    \else \@badmath
    \fi
		since the Steklov eigenvalues $\sigma$ satisfy $\sigma \ge 0$ (with $\sigma = 0$ corresponding to the constants), a non-trivial kernel of the Robin realization can only occur for $\alpha \le 0$, and in that case $Dim~ N(A)$ equals the multiplicity of $-\alpha$ as a Steklov eigenvalue. More generally, the theory of boundary triples allows one to prescribe boundary conditions that are themselves linear relations (set-valued), leading to multi-valued self-adjoint realisations of elliptic operators. The inverses of such realisations are again compact self-adjoint linear relations, and under the same type of compactness argument as in Lemma~ 
	\IfBooleanTF{#1}{\@kernel@sref}{\@kernel@ref} 
{lem:properties}, Theorem~ 
	\IfBooleanTF{#1}{\@kernel@sref}{\@kernel@ref} 
{thm:spectral} applies to give a unified framework for eigenfunction expansions in elliptic boundary value problems with set-valued boundary conditions.
