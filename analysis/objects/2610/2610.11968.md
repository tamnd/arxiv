<!-- remark-1: body_md -->
\label{rem_geo}
If $\Theta$ is star-shaped with respect to some $x^0\in\Theta$, then, with $q(x)=x-x^0$ and $\nu$ the outward unit normal to $\Omega_0$, one has $q\cdot\nu>0$ on $\Gamma_1$ and $q\cdot\nu\leq0$ on $\Gamma_2$, so that \eqref{cond_geo} is the classical partition $\Gamma_1=\{q\cdot\nu>0\}$ of Machtyngier \cite{Mach}. Here, however, $\Theta$ is only assumed to be non-trapping. Equivalently, every generalized bicharacteristic leaves any ball $B_R\supset\Theta$; by compactness, there is then a time $L_R$ such that every unit-speed generalized bicharacteristic meets the region $\omega=\{|x|>R\}$ within time $L_R$. Thus the pair $(\Omega,\omega)$ satisfies the geometric control condition (GCC) of Bardos, Lebeau and Rauch \cite{BLR}; see Figure \ref{fig_geo}(b)--(c). The geometric time $L_R$ is not a restriction on the control time: as is usual for the Schr\"odinger equation, our results hold for every $T>0$. It is through this property, encoded in the resolvent bound \eqref{l9} and in the local smoothing effect \eqref{l12}, that the geometry enters our proofs.
<!-- end remark-1 -->
<!-- theorem-1: body_md -->
\label{principal}
Let $\Theta$, $\Omega$, $\Omega_0$, $\Gamma_1$, $\Gamma_2$ be as in \eqref{cond_geo}, with $\Theta$ non-trapping and $\Omega$ connected, and let $T>0$, $n\in\{2,3\}$ and $s\in\{1,2\}$. Set $X^s(\Omega_0):=\{u\in H^s(\Omega_0):\ u=0 \text{ on }\Gamma_2\}$. Then there exists $\delta>0$ such that for any $u_0,u_1\in X^s(\Omega_0)$ satisfying $\|u_0\|_{H^s(\Omega_0)}\leq\delta$ and $\|u_1\|_{H^s(\Omega_0)}\leq\delta$, one can find a boundary control $h\in C([0,T];H^{s-1/2}(\Gamma_1))$ such that \eqref{s1}--\eqref{s2} admits a solution $u\in C([0,T];H^s(\Omega_0))$ satisfying
\begin{equation}\label{1.1}
u(x,0)=u_0(x), \qquad u(x,T)=u_1(x) \quad\text{in }\Omega_0.
\end{equation}
Here \eqref{s1} holds in $\mathcal{D}'(\Omega_0\times(0,T))$ and \eqref{s2} holds in the sense of traces.
<!-- end theorem-1 -->
<!-- remark-2: body_md -->
\label{rem_principal}
The compatibility condition $u_0=u_1=0$ on $\Gamma_2$ is necessary, since no control acts on $\Gamma_2$. Similarly, since $u\in C([0,T];H^s(\Omega_0))$ and $h=u|_{\Gamma_1}$, the control necessarily satisfies $h(0)=u_0|_{\Gamma_1}$ and $h(T)=u_1|_{\Gamma_1}$; the control constructed in the proof (the trace on $\Gamma_1$ of the solution of the extended problem) satisfies these identities. The time $T>0$ is arbitrary, and no star-shapedness of $\Theta$ is required.

<!-- proposition-3: body_md -->
\label{st1}
Assume that $\Theta$ is non-trapping. Let $T>0$, $s\in[0,1]$ and let $(p,q)$, $p>2$, satisfy $\frac2p+\frac nq=\frac n2$. Then, for $\chi\in C^\infty_0(\mathbb{R}^n)$ with $\chi=1$ near $\Theta$,
\begin{equation}\label{se1}

\end{equation}

<!-- lemma-1: body_md -->
\label{compact}
Assume \textup{(H)}, let $T>0$ and $\sigma\in\mathcal I$. If $v_{j,0}\rightharpoonup v_0$ weakly in $H^\sigma_D(\Omega)$, then $$Q_\sigma(v_{j,0}-v_0)\to0.$$
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Set $k=k(\sigma)$, $\tau=\sigma+2k\in[0,2)$, $\rho=\rho(\sigma)$, $y_{j,0}:=\mathcal R^k(v_{j,0}-v_0)$ and $y_j(t):=e^{it\Delta_D}y_{j,0}$. Then $y_{j,0}\rightharpoonup0$ weakly in $H^\tau_D(\Omega)$, hence weakly in $L^2(\Omega)$, and $\|y_{j,0}\|_{H^\tau_D}\leq M$ for some $M$; in particular $\|y_{j,0}\|_{L^2}\leq M$.

\medskip
\noindent\textit{(a) Bounds.} By Proposition \ref{smoothing} with $s=0$ and $\eta=\psi$,
$$
\int_0^T\|\psi y_j(t)\|^2_{H^{1/2}(\Omega')}\,dt\leq C\int_0^T\|\psi y_j(t)\|^2_{H^{1/2}_D(\Omega)}\,dt\leq CM^2 .
$$
Moreover $\partial_t(\psi y_j)=i\psi\Delta_D y_j$, and for $\zeta\in C^\infty_0(\Omega')$ we have $\psi\zeta\in H^2(\Omega)\cap H^1_0(\Omega)$, so that
$$
|\langle \psi\Delta_Dy_j(t),\zeta\rangle|=|\langle y_j(t),\Delta(\psi\zeta)\rangle|\leq C\|y_j(t)\|_{L^2}\|\zeta\|_{H^2(\Omega')}\leq CM\|\zeta\|_{H^2(\Omega')} .
$$
Hence $(\partial_t(\psi y_j))_j$ is bounded in $L^\infty(0,T;H^{-2}(\Omega'))$, where $H^{-2}(\Omega'):=(H^2_0(\Omega'))'$.

\medskip
\noindent\textit{(b) Compactness.} Since $\Omega'$ is bounded and smooth, $H^{1/2}(\Omega')\hookrightarrow L^2(\Omega')$ is compact and $L^2(\Omega')\hookrightarrow H^{-2}(\Omega')$ is continuous. By the Aubin--Lions--Simon lemma \cite{Simon}, $(\psi y_j)_j$ is relatively compact in $L^2(0,T;L^2(\Omega'))$. As $\operatorname{supp}\psi\subset B_{R+3}(0)$, this is the same as relative compactness in $L^2(0,T;L^2(\Omega))$.

\medskip
\noindent\textit{(c) Identification of the limit.} For $g\in L^2(0,T;L^2(\Omega))$,
$$
\int_0^T\langle\psi y_j(t),g(t)\rangle\,dt=\Big\langle y_{j,0},\int_0^Te^{-it\Delta_D}(\psi g(t))\,dt\Big\rangle\longrightarrow0,
$$
since $$\int_0^Te^{-it\Delta_D}(\psi g)\,dt\in L^2(\Omega).$$ Thus $\psi y_j\rightharpoonup0$ weakly in $L^2(0,T;L^2(\Omega))$, and by (b) every subsequence has a further subsequence converging strongly to $0$. Therefore $\psi y_j\to0$ in $L^2(0,T;L^2(\Omega))$, which is the claim if $\rho=0$.

\medskip
\noindent\textit{(d) The case $\rho>0$.} Then $\sigma\in(-1,0)$, $\tau\in(1,2)$ and $\rho=\tau-1$. Since multiplication by $\psi$ is bounded on $H^\tau_D(\Omega)$ and $e^{it\Delta_D}$ is unitary on this space, $\|\psi y_j(t)\|_{H^\tau_D}\leq CM$ for all $t$ and $j$. The spectral theorem and H\"older's inequality give $\|f\|_{H^\rho_D}\leq\|f\|^{1-\rho/\tau}_{L^2}\|f\|^{\rho/\tau}_{H^\tau_D}$, and H\"older's inequality in time yields
$$
Q_\sigma(v_{j,0}-v_0)=\int_0^T\|\psi y_j(t)\|^2_{H^\rho_D}\,dt\leq(CM)^{\frac{2\rho}\tau}\,T^{\frac\rho\tau}\Big(\int_0^T\|\psi y_j(t)\|^2_{L^2}\,dt\Big)^{1-\frac\rho\tau}\longrightarrow0 .
$$

<!-- lemma-2: body_md -->
\label{relaxed}
Assume \textup{(H)} and let $T>0$, $\sigma\in\mathcal I$. There exists $C>0$ such that
\begin{equation}\label{relaxed_est}
\|v_0\|^2_{H^\sigma_D(\Omega)}\leq C\left(\int_0^T\|\varphi v(t)\|^2_{H^\sigma_D(\Omega)}\,dt+Q_\sigma(v_0)\right),\qquad\forall\,v_0\in H^\sigma_D(\Omega).
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
We split the proof in two cases.

\medskip
\noindent\textit{1. Case $\sigma=\tau\in[0,2)$.}
\vspace{0.1cm}

By unitarity of $e^{it\Delta_D}$ on $H^\tau_D(\Omega)$ and $v=\varphi v+\chi v$,
\begin{equation}\label{rel1}
T\|v_0\|^2_{H^\tau_D}=\int_0^T\|v(t)\|^2_{H^\tau_D}\,dt\leq2\int_0^T\|\varphi v(t)\|^2_{H^\tau_D}\,dt+2\int_0^T\|\chi v(t)\|^2_{H^\tau_D}\,dt .
\end{equation}
If $\tau=0$, then by \eqref{psi} $\|\chi v\|_{L^2}\leq\|\psi v\|_{L^2}$, and \eqref{rel1} is \eqref{relaxed_est}. Let $\tau\in(0,2)$ and set
$$
\theta:=0\ \text{ if }\tau\in(0,\tfrac12),\qquad \theta:=\tau-\tfrac12\ \text{ if }\tau\in[\tfrac12,\tfrac32],\qquad \theta:=2\tau-2\ \text{ if }\tau\in(\tfrac32,2),
$$
so that $0\leq\theta<\tau$. We claim that
\begin{equation}\label{rel_chi}
\int_0^T\|\chi v(t)\|^2_{H^\tau_D}\,dt\leq C_1\|v_0\|^2_{H^\theta_D}.
\end{equation}
If $\tau\in[\frac12,\frac32]$ we apply Proposition \ref{smoothing} with $s=\tau-\frac12$ and $\eta=\chi$; if $\tau<\frac12$ we use $\|\chi v\|_{H^\tau_D}\leq\|\chi v\|_{H^{1/2}_D}$ and Proposition \ref{smoothing} with $s=0$. If $\tau\in(\frac32,2)$, we interpolate between the estimates
$$
\|\chi e^{it\Delta_D}v_0\|_{L^2(0,T;H^{3/2}_D)}\leq C\|v_0\|_{H^1_D},\qquad\|\chi e^{it\Delta_D}v_0\|_{L^2(0,T;H^2_D)}\leq C\|v_0\|_{H^2_D},
$$
the first one being Proposition \ref{smoothing} with $s=1$, and the second one a consequence of the unitarity of $e^{it\Delta_D}$ on $H^2_D(\Omega)$ and of the boundedness of multiplication by $\chi$ on this space. Complex interpolation with parameter $a=2\tau-3\in(0,1)$ gives \eqref{rel_chi}, since $[H^1_D,H^2_D]_a=H^{2\tau-2}_D$ and $[H^{3/2}_D,H^2_D]_a=H^{\tau}_D$.

Since $0\leq\theta<\tau$, the spectral theorem gives, for every $\varepsilon>0$, $$\|v_0\|^2_{H^\theta_D}\leq\varepsilon\|v_0\|^2_{H^\tau_D}+C_\varepsilon\|v_0\|^2_{L^2}.$$ Choosing $2C_1\varepsilon=T/2$ in \eqref{rel1} we obtain
$$
\frac T2\|v_0\|^2_{H^\tau_D}\leq2\int_0^T\|\varphi v(t)\|^2_{H^\tau_D}\,dt+2C_1C_\varepsilon\|v_0\|^2_{L^2}.
$$
Finally, by unitarity in $L^2$, $v=\varphi v+\chi v$ and \eqref{psi},
$$
\|v_0\|^2_{L^2}=\frac1T\int_0^T\|v(t)\|^2_{L^2}\,dt\leq\frac2T\left(\int_0^T\|\varphi v(t)\|^2_{H^\tau_D}\,dt+\int_0^T\|\psi v(t)\|^2_{L^2}\,dt\right),
$$
which yields \eqref{relaxed_est}, since $\rho(\sigma)=0$ in this case.

\medskip
\noindent\textit{2. Case $\sigma=\tau-2\in[-2,0)$.}
\vspace{0.1cm}

 Put $y:=\mathcal Rv=e^{it\Delta_D}y_0$, $y_0:=\mathcal Rv_0\in H^\tau_D(\Omega)$, so that $\|y_0\|_{H^\tau_D}=\|v_0\|_{H^\sigma_D}$. By the first case, applied to $y_0$,
\begin{equation}\label{rel2}
\|v_0\|^2_{H^\sigma_D}\leq C\left(\int_0^T\|\varphi y(t)\|^2_{H^\tau_D}\,dt+\int_0^T\|\psi y(t)\|^2_{L^2}\,dt\right).
\end{equation}
We compare $\varphi y=\varphi\mathcal Rv$ with $\mathcal R(\varphi v)$. Since $\varphi$ vanishes near $\partial\Omega$, for $u\in\mathcal D(\Delta_D)$ we have $\varphi u\in\mathcal D(\Delta_D)$ and $(1-\Delta_D)(\varphi u)-\varphi(1-\Delta_D)u=-[\Delta,\varphi]u$, where
$$
[\Delta,\varphi]u:=(\Delta\varphi)u+2\nabla\varphi\cdot\nabla u=2\operatorname{div}(u\nabla\varphi)-(\Delta\varphi)u .
$$
Applying $\mathcal R$ on both sides with $u=\mathcal R f$ and using density, we get $$\varphi\mathcal R f-\mathcal R(\varphi f)=-\mathcal R[\Delta,\varphi]\mathcal R f,$$ for $f\in H^\sigma_D(\Omega)$. For $u\in L^2(\Omega)$ and $\zeta\in H^1_0(\Omega)$,
$$
|\langle[\Delta,\varphi]u,\zeta\rangle|=\Big|-2\int_\Omega u\nabla\varphi\cdot\nabla\overline\zeta\,dx-\int_\Omega(\Delta\varphi)u\overline\zeta\,dx\Big|\leq C\|u\|_{L^2}\|\zeta\|_{H^1_0},
$$
so that $[\Delta,\varphi]$ is bounded from $L^2(\Omega)$ to $H^{-1}_D(\Omega)$. Being a first order operator with smooth coefficients supported in $\operatorname{supp}\nabla\varphi$, it is also bounded from $H^1_D(\Omega)$ to $L^2(\Omega)$, hence, by interpolation, from $H^\alpha_D(\Omega)$ to $H^{\alpha-1}_D(\Omega)$ for every $\alpha\in[0,1]$. Moreover $$[\Delta,\varphi]u=[\Delta,\varphi](\psi u),$$ because $\psi=1$ in a neighbourhood of $\operatorname{supp}\nabla\varphi$ by \eqref{psi}. If $\tau\in[0,1]$, then $\sigma\leq-1$ and $$\|[\Delta,\varphi]y\|_{H^\sigma_D}\leq\|[\Delta,\varphi](\psi y)\|_{H^{-1}_D}\leq C\|\psi y\|_{L^2};$$ if $\tau\in(1,2)$, we take $\alpha=\tau-1=\rho(\sigma)$ and obtain $\|[\Delta,\varphi]y\|_{H^\sigma_D}\leq C\|\psi y\|_{H^{\tau-1}_D}$. In both cases $\|[\Delta,\varphi]y\|_{H^\sigma_D}\leq C\|\psi y\|_{H^{\rho(\sigma)}_D}$, and since $\mathcal R:H^\sigma_D\to H^\tau_D$ is an isometry,
$$
\|\varphi y(t)\|_{H^\tau_D}\leq\|\varphi v(t)\|_{H^\sigma_D}+C\|\psi y(t)\|_{H^{\rho(\sigma)}_D}.
$$
Plugging this into \eqref{rel2}, and using $\|\psi y\|_{L^2}\leq\|\psi y\|_{H^{\rho(\sigma)}_D}$ and $$Q_\sigma(v_0)=\int_0^T\|\psi y(t)\|^2_{H^{\rho(\sigma)}_D}dt,$$ gives \eqref{relaxed_est}.

<!-- lemma-3: body_md -->
\label{UCP}
Assume \textup{(H)}. Then $\mathcal N^\sigma_T=\{0\}$ for every $T>0$ and every $\sigma\in[-2,2]$.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Since $\mathcal N^\sigma_T\subset\mathcal N^{-2}_T$ for every $\sigma\in[-2,2]$, it would be enough to consider $\sigma=-2$; we give the argument for an arbitrary $\sigma\in\mathcal I$.

\textit{(i) $\mathcal N^\sigma_T$ is finite dimensional.} For $v_0\in\mathcal N^\sigma_T$, \eqref{relaxed_est} reads $$\|v_0\|^2_{H^\sigma_D}\leq C\,Q_\sigma(v_0).$$ Let $(v_{j,0})$ be a bounded sequence in $\mathcal N^\sigma_T$. Extracting a subsequence, $v_{j,0}\rightharpoonup v_0$ weakly in $H^\sigma_D(\Omega)$, and $v_0\in\mathcal N^\sigma_T$ because this subspace is weakly closed. Applying \eqref{relaxed_est} to $v_{j,0}-v_0\in\mathcal N^\sigma_T$ and Lemma \ref{compact},
$$
\|v_{j,0}-v_0\|^2_{H^\sigma_D}\leq C\,Q_\sigma(v_{j,0}-v_0)\longrightarrow0 .
$$
Thus the closed unit ball of $\mathcal N^\sigma_T$ is compact and $\dim\mathcal N^\sigma_T<\infty$.

\smallskip
\textit{(ii) Invariance under $\Delta_D$.} Fix $T_0>0$. The map $T\mapsto d(T):=\dim\mathcal N^\sigma_T$ is non-increasing and, by (i), takes finitely many values on $[T_0/2,T_0]$. Hence there are $T_0/2\leq a<b\leq T_0$ such that $d$ is constant on $(a,b)$; since the spaces are nested and finite dimensional, $\mathcal N^\sigma_T=\mathcal N^\sigma_{T'}=:\mathcal N$ for all $T,T'\in(a,b)$. Fix $T_1\in(a,b)$ and $0<\varepsilon<b-T_1$. If $v_0\in\mathcal N=\mathcal N^\sigma_{T_1+\varepsilon}$, then $$\varphi e^{it\Delta_D}(e^{i\varepsilon\Delta_D}v_0)=\varphi e^{i(t+\varepsilon)\Delta_D}v_0=0,$$ for $t\in[0,T_1]$, i.e. $e^{i\varepsilon\Delta_D}v_0\in\mathcal N^\sigma_{T_1}=\mathcal N$. Consequently
$$
\frac{e^{i\varepsilon\Delta_D}v_0-v_0}{\varepsilon}\in\mathcal N\quad\text{and}\quad \frac{e^{i\varepsilon\Delta_D}v_0-v_0}{\varepsilon}\xrightarrow[\varepsilon\to0^+]{}i\Delta_Dv_0\ \text{ in } H^{\sigma-2}_D(\Omega).
$$
Being finite dimensional, $\mathcal N$ is closed in $H^{\sigma-2}_D(\Omega)$; hence $\Delta_D\mathcal N\subset\mathcal N$.

\smallskip
\textit{(iii) Conclusion.} Suppose $\mathcal N\neq\{0\}$. Then the linear map $\Delta_D|_{\mathcal N}$ has an eigenvector: $v_0\in\mathcal N\setminus\{0\}$, $\Delta_Dv_0=\mu v_0$ for some $\mu\in\mathbb C$. We have $\mu\neq1$, since otherwise $(1-\Delta_D)v_0=0$ and $v_0=0$. Thus $v_0=(1-\mu)\mathcal Rv_0$ and, iterating, $v_0\in H^{\sigma+2m}_D(\Omega)$ for every $m\in\mathbb N$; in particular $v_0\in\mathcal D(\Delta_D)$ and $\Delta v_0=\mu v_0$ in $\mathcal D'(\Omega)$. By elliptic regularity, $v_0$ is real analytic in $\Omega$. Taking $t=0$ in the definition of $\mathcal N^\sigma_{T_1}$ we get $\varphi v_0=0$, so $v_0$ vanishes on the nonempty open set $\{x\in\Omega:|x|>R+1\}$, where $\varphi=1$. Since $\Omega$ is connected, $v_0\equiv0$, a contradiction. Therefore $\mathcal N^\sigma_{T_1}=\{0\}$, and $\mathcal N^\sigma_{T_0}\subset\mathcal N^\sigma_{T_1}=\{0\}$ because $T_0\geq T_1$. Finally, $\mathcal N^2_T\subset\mathcal N^0_T=\{0\}$.

\label{obs}
Assume \textup{(H)}, let $T>0$ and $\sigma\in[-2,2]$. There exists $C>0$ such that every solution of \eqref{adj} with $v_0\in H^\sigma_D(\Omega)$ satisfies
\begin{equation}\label{OI_new}
\|v_0\|^2_{H^\sigma_D(\Omega)}\leq C\int_0^T\|\varphi v(t)\|^2_{H^\sigma_D(\Omega)}\,dt .
\end{equation}
<!-- end proposition-6 -->
<!-- proof-4: body_md -->
Let first $\sigma\in\mathcal I$. If \eqref{OI_new} fails, there are $v_{j,0}\in H^\sigma_D(\Omega)$ with
\begin{equation}\label{contr}
\|v_{j,0}\|_{H^\sigma_D}=1,\qquad\int_0^T\|\varphi v_j(t)\|^2_{H^\sigma_D}\,dt\leq\frac1j .
\end{equation}
Extracting a subsequence, $v_{j,0}\rightharpoonup v_0$ weakly in $H^\sigma_D(\Omega)$. Arguing as in Lemma \ref{compact}(c), $\varphi v_j\rightharpoonup\varphi v$ weakly in $L^2(0,T;H^\sigma_D(\Omega))$, while $\varphi v_j\to0$ strongly by \eqref{contr}. Hence $\varphi v=0$ a.e. in $(0,T)$ and, by continuity of $t\mapsto\varphi v(t)\in H^\sigma_D(\Omega)$, for every $t\in[0,T]$. Thus $v_0\in\mathcal N^\sigma_T=\{0\}$ by Lemma \ref{UCP}. By Lemma \ref{compact}, $Q_\sigma(v_{j,0})=Q_\sigma(v_{j,0}-v_0)\to0$, and Lemma \ref{relaxed} gives
$$
1=\|v_{j,0}\|^2_{H^\sigma_D}\leq C\left(\frac1j+Q_\sigma(v_{j,0})\right)\longrightarrow0,
$$
a contradiction.

Let now $\sigma=2$ and set $A:=1-\Delta_D$. For $v_0\in H^2_D(\Omega)$, the function $Av(t)=e^{it\Delta_D}Av_0$ solves \eqref{adj} with initial data $Av_0\in L^2(\Omega)$, and $\|v_0\|_{H^2_D}=\|Av_0\|_{L^2}$. Since $$A(\varphi v)-\varphi Av=-[\Delta,\varphi]v,$$ see the proof of Lemma \ref{relaxed}, $\varphi v(t)\in H^2_D(\Omega)$, and $[\Delta,\varphi]$ is bounded from $H^1_D(\Omega)$ to $L^2(\Omega)$, we have
$$
\|\varphi Av(t)\|_{L^2}\leq\|\varphi v(t)\|_{H^2_D}+C\|v(t)\|_{H^1_D}=\|\varphi v(t)\|_{H^2_D}+C\|v_0\|_{H^1_D}.
$$
Applying \eqref{OI_new} with $\sigma=0$ to $Av$, we obtain
$$
\|v_0\|^2_{H^2_D}\leq C_1\int_0^T\|\varphi v(t)\|^2_{H^2_D}\,dt+C_2T\|v_0\|^2_{H^1_D}.
$$
By the spectral theorem, $$\|v_0\|^2_{H^1_D}\leq\varepsilon\|v_0\|^2_{H^2_D}+C_\varepsilon\|v_0\|^2_{L^2},$$ for every $\varepsilon>0$, and, by \eqref{OI_new} with $\sigma=0$ applied to $v$, $$\|v_0\|^2_{L^2}\leq C\int_0^T\|\varphi v(t)\|^2_{L^2}\,dt\leq C\int_0^T\|\varphi v(t)\|^2_{H^2_D}\,dt.$$ Choosing $C_2T\varepsilon=\frac12$ gives \eqref{OI_new} with $\sigma=2$.

<!-- theorem-6: body_md -->
\label{exact_lin}
Assume \textup{(H)}, let $T>0$ and $s\in[-2,2]$. There exists a bounded linear operator
$$
G:H^s_D(\Omega)\times H^s_D(\Omega)\longrightarrow C([0,T];H^s_D(\Omega))
$$
such that for any $z_0,z_T\in H^s_D(\Omega)$ the solution of \eqref{lin_new} with $h=G(z_0,z_T)$ satisfies $z(T)=z_T$. The control force $\varphi h$ is supported in $\{x\in\Omega:|x|\geq R\}\times[0,T]$.
<!-- end theorem-6 -->
<!-- proof-5: body_md -->
Let $\sigma:=-s\in[-2,2]$ and $\Lambda_s:=(1-\Delta_D)^s$, an isometric isomorphism from $H^s_D(\Omega)$ onto $H^{-s}_D(\Omega)=H^\sigma_D(\Omega)$ with $\langle\Lambda_s^{-1}g,g\rangle=\|g\|^2_{H^\sigma_D}$, for $g\in H^\sigma_D(\Omega)$.

\smallskip
\textit{Null controllability.} For $v_0\in H^\sigma_D(\Omega)$ let $v$ solve \eqref{adj}, set
\begin{equation}\label{HUMcontrol}
h(t):=-i\,\Lambda_s^{-1}\big(\varphi v(t)\big)\in C([0,T];H^s_D(\Omega)),
\end{equation}
let $z$ be the solution of \eqref{lin_new} (backward in time) with $z(T)=0$, and define $\mathcal S v_0:=z(0)\in H^s_D(\Omega)$. For smooth data, $$\frac{d}{dt}\langle z,v\rangle=\langle i\Delta z-i\varphi h,v\rangle+\langle z,i\Delta v\rangle=-i\langle\varphi h,v\rangle,$$ so that, by density,
$$
\langle\mathcal Sv_0,v_0\rangle=i\int_0^T\langle h(t),\varphi v(t)\rangle\,dt=\int_0^T\langle\Lambda_s^{-1}(\varphi v),\varphi v\rangle\,dt=\int_0^T\|\varphi v(t)\|^2_{H^\sigma_D}\,dt .
$$
By Proposition \ref{obs}, $\langle\mathcal Sv_0,v_0\rangle\geq C^{-1}\|v_0\|^2_{H^\sigma_D}$. Since $\mathcal S:H^\sigma_D\to H^s_D=(H^\sigma_D)'$ is bounded and linear, the Lax--Milgram theorem shows that $\mathcal S$ is an isomorphism. Given $z_0\in H^s_D(\Omega)$, the choice $v_0=\mathcal S^{-1}z_0$ in \eqref{HUMcontrol} drives $z_0$ to $0$ at time $T$, by uniqueness of solutions of \eqref{lin_new}.

\smallskip
\textit{Exact controllability.} The solution of \eqref{lin_new} is $$z(t)=e^{it\Delta_D}z_0+L_h(t),$$ with $L_h$ linear in $h$. Thus $z(T)=z_T$ if and only if $h$ drives $z_0-e^{-iT\Delta_D}z_T$ to $0$, and we set $$G(z_0,z_T):=-i\Lambda_s^{-1}\big(\varphi\,e^{it\Delta_D}\mathcal S^{-1}(z_0-e^{-iT\Delta_D}z_T)\big).$$ Boundedness of $G$ follows from the boundedness of $\mathcal S^{-1}$, $\Lambda_s^{-1}$, $e^{it\Delta_D}$ and of multiplication by $\varphi$. Since \eqref{lin_new} coincides with \eqref{int_lin}, this also proves Theorem \ref{lin_th}.

Let us give some remarks.
\begin{itemize}
\item[(1)] The forcing term in \eqref{lin_new} is $\varphi h=A v$, with $A:=-i\varphi(1-\Delta_D)^{-s}\varphi$, which is exactly the HUM operator used by Laurent \cite[Proposition 7.2]{Laurent2}. Only $\varphi h$, not $h$, is localized, as in \cite[Remark 1]{BrCaCaDo}.

\item[(3)] For the nonlinear results we only use $s=1$ (control in $H^1_0(\Omega)$) and $s=2$ (control in $H^2_D(\Omega)$), and $s\in(\frac n2,2]$ in Remark \ref{rem_algebra}. \item[(4)] Proposition \ref{smoothing} is only used with $s\in[0,1]$, which is exactly the range proved in \cite{BGT1}; for $\sigma\in[-2,0)$ it is applied to $y_0=\mathcal Rv_0\in H^{\sigma+2}_D(\Omega)$.

<!-- theorem-7: body_md -->
\label{control_n}
Assume \textup{(H)}. Let $T>0$, $s\in[1,2]$, $m\in\{3,5\}$, and let $X_{T}$ be a Banach space with $X_T\hookrightarrow C([0,T];H^s_D(\Omega))$, that is, $\|u\|_{C([0,T];H^s_D)}\leq\|u\|_{X_T}$. Assume that there exists $C_0\geq1$ such that
\begin{itemize}
\item[(i)] $\|W(\cdot)u_0\|_{X_T}\leq C_0\|u_0\|_{H^s_D(\Omega)}$ for all $u_0\in H^s_D(\Omega)$;
\item[(ii)] $\big\|\displaystyle\int_0^tW(t-r)f(r)\,dr\big\|_{X_T}\leq C_0\|f\|_{L^1(0,T;H^s_D(\Omega))}$, \quad for all $f\in L^1(0,T;H^s_D(\Omega))$;
\item[(iii)] for all $u,v\in X_T$, $g(v)\in L^1(0,T;H^s_D(\Omega))$ and
$$
\|g(v)\|_{L^1(0,T;H^s_D)}\leq C_0\|v\|^m_{X_T},\qquad \|g(v)-g(u)\|_{L^1(0,T;H^s_D)}\leq C_0\big(\|v\|^{m-1}_{X_T}+\|u\|^{m-1}_{X_T}\big)\|v-u\|_{X_T}.
$$
\end{itemize}
Then there exists $\delta>0$ such that for any $v_0,v_T\in H^s_D(\Omega)$ with $\|v_0\|_{H^s_D}\leq\delta$ and $\|v_T\|_{H^s_D}\leq\delta$, there is a control $h\in C([0,T];H^s_D(\Omega))$ such that \eqref{nlinear} admits a solution $v\in X_T$ with $v(0)=v_0$ and $v(T)=v_T$.
<!-- end theorem-7 -->
<!-- proof-6: body_md -->
Let $G$ be the operator given by Theorem \ref{exact_lin}: for $z_0,z_T\in H^s_D(\Omega)$, the solution of \eqref{lin_new} with $h=G(z_0,z_T)$ starts at $z_0$ and satisfies $z(T)=z_T$, and $$\|G(z_0,z_T)\|_{C([0,T];H^s_D)}\leq C_G(\|z_0\|_{H^s_D}+\|z_T\|_{H^s_D}).$$ Recall also that multiplication by $\varphi$ is bounded on $H^s_D(\Omega)$, with norm $C_\varphi$.

For $v\in X_T$ set
$$
N(v)(t):=i\int_0^tW(t-r)g(v)(r)\,dr,\qquad h_v:=G\big(v_0,\,v_T-N(v)(T)\big),
$$
and
$$
\mathcal T(v)(t):=W(t)v_0+N(v)(t)-i\int_0^tW(t-r)(\varphi h_v)(r)\,dr .
$$
By definition of $G$, the linear part $$W(t)v_0-i\int_0^tW(t-r)\varphi h_v\,dr$$ equals $v_0$ at $t=0$ and $v_T-N(v)(T)$ at $t=T$. Hence $\mathcal T(v)(0)=v_0$ and $\mathcal T(v)(T)=v_T$, and by \eqref{duhamel} a fixed point of $\mathcal T$ is a solution of \eqref{nlinear} with control $h=h_v$ and the required properties.

By (ii)--(iii), $$\|N(v)\|_{X_T}\leq C_0^2\|v\|^m_{X_T}$$ and, in particular, $$\|N(v)(T)\|_{H^s_D}\leq C_0^2\|v\|^m_{X_T}.$$ Therefore, using (i), (ii) and $$\|\varphi h_v\|_{L^1(0,T;H^s_D)}\leq TC_\varphi\|h_v\|_{C([0,T];H^s_D)},$$ we get
$$
\|\mathcal T(v)\|_{X_T}\leq C_1\big(\|v_0\|_{H^s_D}+\|v_T\|_{H^s_D}\big)+C_2\|v\|^m_{X_T},
$$
with $$C_1:=C_0+C_0TC_\varphi C_G$$ and $$C_2:=C_0^2(1+C_0TC_\varphi C_G).$$ Similarly, since $h_v-h_u=-G(0,N(v)(T)-N(u)(T))$,
$$
\|\mathcal T(v)-\mathcal T(u)\|_{X_T}\leq C_2\big(\|v\|^{m-1}_{X_T}+\|u\|^{m-1}_{X_T}\big)\|v-u\|_{X_T}.
$$
Let $M:=4C_1\delta$ and choose $\delta>0$ so small that $4C_2M^{m-1}\leq1$. If $\|v_0\|_{H^s_D},\|v_T\|_{H^s_D}\leq\delta$, then for $u,v$ in the closed ball $B_M\subset X_T$ we get $$\|\mathcal T(v)\|_{X_T}\leq\frac M2+C_2M^m\leq M$$ and $$\|\mathcal T(v)-\mathcal T(u)\|_{X_T}\leq2C_2M^{m-1}\|v-u\|_{X_T}\leq\frac12\|v-u\|_{X_T}.$$ The contraction mapping principle gives the fixed point.

<!-- proposition-7: body_md -->
\label{H1-Control}
Assume \textup{(H)} with $n\in\{2,3\}$, and let $T>0$ and $m\in\{3,5\}$. The assumptions of Theorem \ref{control_n} hold with $s=1$ and
$$
X_T:=C([0,T];H^1_0(\Omega))\cap L^4(0,T;L^\infty(\Omega)),\qquad \|u\|_{X_T}:=\|u\|_{L^\infty(0,T;H^1_0)}+\|u\|_{L^4(0,T;L^\infty)} .
$$
Consequently, system \eqref{nlinear} is locally exactly controllable in $H^1_0(\Omega)$.
<!-- end proposition-7 -->
<!-- proof-8: body_md -->
(i) follows from the unitarity of $W(t)$ on $H^1_0(\Omega)$ and Lemma \ref{L4Linfty} with $p=4$.

(ii) For the $C([0,T];H^1_0)$ part we use unitarity. For the $L^4L^\infty$ part, by Minkowski's inequality and Lemma \ref{L4Linfty} applied on the interval $[0,T-r]$ (with a constant uniform in $r\in[0,T]$, since $[0,T-r]\subset[0,T]$),
$$
\Big\|\int_0^t W(t-r)f(r)\,dr\Big\|_{L^4(0,T;L^\infty)}\leq\int_0^T\big\|W(\cdot-r)f(r)\big\|_{L^4(r,T;L^\infty)}\,dr\leq C\|f\|_{L^1(0,T;H^1_0)} .
$$

(iii) If $v\in H^1_0(\Omega)\cap L^\infty(\Omega)$, then $g(v)=\lambda v^{\frac{m+1}2}\overline v^{\frac{m-1}2}\in H^1_0(\Omega)$,
$$
\nabla g(v)=\lambda\Big(\tfrac{m+1}2|v|^{m-1}\nabla v+\tfrac{m-1}2|v|^{m-3}v^2\nabla\overline v\Big)\quad\text{and}\quad \|g(v)\|_{H^1_0}\leq m|\lambda|\,\|v\|^{m-1}_{L^\infty}\|v\|_{H^1_0}.
$$
By H\"older's inequality in time, since $m-1\leq4$,
$$
\|g(v)\|_{L^1(0,T;H^1_0)}\leq m|\lambda|\,\|v\|_{L^\infty(0,T;H^1_0)}\int_0^T\|v(t)\|^{m-1}_{L^\infty}dt\leq m|\lambda|T^{\frac{5-m}4}\|v\|_{X_T}^m .
$$
For the difference, we use the pointwise bounds
$$
\begin{gathered}
|g(v)-g(u)|\leq C(|v|^{m-1}+|u|^{m-1})|v-u|,\\
|\nabla(g(v)-g(u))|\leq C(|v|^{m-1}+|u|^{m-1})|\nabla(v-u)|+C(|v|^{m-2}+|u|^{m-2})(|\nabla v|+|\nabla u|)|v-u|,
\end{gathered}
$$
which give, for a.e. $t$,
$$
\begin{aligned}
\|g(v)-g(u)\|_{H^1_0}\leq{}& C\big(\|v\|^{m-1}_{L^\infty}+\|u\|^{m-1}_{L^\infty}\big)\|v-u\|_{H^1_0}\\
&+C\big(\|v\|^{m-2}_{L^\infty}+\|u\|^{m-2}_{L^\infty}\big)\big(\|v\|_{H^1_0}+\|u\|_{H^1_0}\big)\|v-u\|_{L^\infty}.
\end{aligned}
$$
Integrating in time and using H\"older's inequality (with exponents $\frac{m-2}4+\frac14+\frac{5-m}4=1$ in the last term),
$$
\|g(v)-g(u)\|_{L^1(0,T;H^1_0)}\leq CT^{\frac{5-m}4}\big(\|v\|^{m-1}_{X_T}+\|u\|^{m-1}_{X_T}\big)\|v-u\|_{X_T}.
$$
Theorem \ref{control_n} applies.

<!-- lemma-5: body_md -->
\label{extension}
Let $s\in[1,2]$. There exists a bounded linear operator $E:X^s(\Omega_0)\to H^s_D(\Omega)$ such that $(Eu)|_{\Omega_0}=u$ for every $u\in X^s(\Omega_0)$.

<!-- proof-11: body_md -->

$$
Eu:=\widetilde{(1-\zeta)u}+\tilde\zeta\,E_{\mathcal A}\big((\zeta u)|_{\mathcal A}\big),
$$
where $\widetilde{(1-\zeta)u}$ denotes the extension by zero of $(1-\zeta)u$ to $\Omega$. Since $(1-\zeta)u$ vanishes for $|x|\geq R_0-\varepsilon$, its extension by zero belongs to $H^s(\Omega)$; since $\tilde\zeta$ vanishes near $\Theta$, the second term belongs to $H^s(\Omega)$ and vanishes near $\partial\Omega$. Moreover $Eu=0$ on $\partial\Omega=\Gamma_2$ because $u=0$ there. Hence $$Eu\in H^s(\Omega)\cap H^1_0(\Omega)=H^s_D(\Omega),$$ and $$Eu=(1-\zeta)u+\zeta u=u$$ in $\Omega_0$, because $\tilde\zeta=1$ on $\operatorname{supp}\zeta$ and $E_{\mathcal A}(\zeta u)=\zeta u$ in $\mathcal A$.

Since $\varphi=0$ on $\Omega_0$, $u$ solves \eqref{s1} in $\mathcal{D}'(\Omega_0\times(0,T))$, and $u\in C([0,T];H^s(\Omega_0))$ with $u(0)=u_0$, $u(T)=u_1$ by Lemma \ref{extension}. \frac{d}{dt}\operatorname{Re}\langle z,v\rangle=\operatorname{Re}\langle -i\varphi h,v\rangle ,
$$
because the contributions of $\Delta$, $V_1$ and $V_2$ cancel (this is the reason for the sign of $V_2$ in \eqref{adj_w}). For $v_0\in H^\sigma_D$ let $h:=-i\Lambda_s^{-1}(\varphi v)\in C([0,T];H^s_D)$, let $z$ solve \eqref{linearized_int} backwards with $z(T)=0$, and set $\mathcal S_wv_0:=z(0)$. Then
$$
\operatorname{Re}\langle\mathcal S_wv_0,v_0\rangle=\operatorname{Re}\Big(i\int_0^T\langle h,\varphi v\rangle\,dt\Big)=\int_0^T\|\varphi v(t)\|^2_{H^\sigma_D}\,dt\geq C^{-1}\|v_0\|^2_{H^\sigma_D}
$$
by Proposition \ref{obs_w}. The Lax--Milgram theorem, applied in the real Hilbert spaces $H^\sigma_D$ and $H^s_D=(H^\sigma_D)'$, shows that the real-linear operator $\mathcal S_w$ is an isomorphism, which gives null controllability. Exact controllability follows as in Theorem \ref{exact_lin}, using the invertibility of the propagator of \eqref{linearized_int} with $h=0$. We denote by $G_w:H^s_D\times H^s_D\to C([0,T];H^s_D)$ the resulting bounded real-linear control operator.

Let $X_T$ be the space of Proposition \ref{H1-Control} if $s=1$, and that of Proposition \ref{H2-Control} if $s=2$. Writing $v=w+z$, $z$ solves
\begin{equation}\label{z_eq}
iz_t+\Delta z+V_1z+V_2\overline z+F(w,z)=\varphi h,\qquad\text{with}\quad F(w,z):=\lambda\big(2w|z|^2+\overline wz^2+|z|^2z\big),
\end{equation}
and  $z(0)=z_0:=v_0-w(0)$, we must reach $z(T)=z_T:=v_T-w(T)$. Let $S(t,r)$ be the propagator of \eqref{linearized_int} with $h=0$. Because of the term $V_2\overline z$, $S(t,r)$ is only real-linear, so that in general $S(t,r)(if)\neq iS(t,r)f$; the factor $i$ must therefore be kept inside the propagator, and \eqref{z_eq} is equivalent to
$$
z(t)=S(t,0)z_0+\int_0^tS(t,r)\big(iF(w,z)(r)\big)\,dr+\int_0^tS(t,r)\big(-i\varphi h(r)\big)\,dr .
$$
In the same way, the solution of \eqref{linearized_int} is $$S(t,0)z_0+\int_0^tS(t,r)(-i\varphi h(r))\,dr,$$ and $G_w$ is defined accordingly. For $z\in X_T$ we set
$$
N_w(z)(t):=\int_0^tS(t,r)\big(iF(w,z)(r)\big)\,dr,\qquad h_z:=G_w\big(z_0,\,z_T-N_w(z)(T)\big),
$$
and
$$
\mathcal T_w(z)(t):=S(t,0)z_0+N_w(z)(t)+\int_0^tS(t,r)\big(-i\varphi h_z(r)\big)\,dr .
$$
By definition of $G_w$, the sum of the first and last terms equals $z_0$ at $t=0$ and $z_T-N_w(z)(T)$ at $t=T$, so that $\mathcal T_w(z)(0)=z_0$ and $\mathcal T_w(z)(T)=z_T$, and a fixed point of $\mathcal T_w$ solves \eqref{z_eq} with $h=h_z$. Since $$h_z-h_{\tilde z}=-G_w\big(0,N_w(z)(T)-N_w(\tilde z)(T)\big)$$ and multiplication by $i$ preserves all the norms involved, the estimates of the proof of Theorem \ref{control_n} carry over, with $W(t-r)$ replaced by $S(t,r)$, $g$ by $F(w,\cdot)$ and $G$ by $G_w$ (see also \cite[Proposition 8.1]{Laurent2}), provided that $S$ satisfies (i)--(ii) of Theorem \ref{control_n} and that $F$ satisfies the analogue of (iii). Properties (i)--(ii) for $S$ follow from those for $W$ through the Duhamel formula
$$
S(t,r)u_0=W(t-r)u_0+i\int_r^tW(t-\rho)\big(V_1S(\rho,r)u_0+V_2\overline{S(\rho,r)u_0}\big)\,d\rho
$$
(here $i$ may be taken out of the integral, since $W$ is complex-linear), Gronwall's lemma and the boundedness of $V_1,V_2$ on $H^s_D$; this applies both to $X_T=C([0,T];H^1_0)\cap L^4(0,T;L^\infty)$ ($s=1$) and to $X_T=C([0,T];H^2_D)$ ($s=2$), for $n\in\{2,3\}$. Property (iii) is replaced by
$$
\begin{gathered}
\|F(w,z)\|_{L^1(0,T;H^s_D)}\leq C\big(\|z\|^2_{X_T}+\|z\|^3_{X_T}\big),\\
\|F(w,z)-F(w,\tilde z)\|_{L^1(0,T;H^s_D)}\leq C\big(\|z\|_{X_T}+\|\tilde z\|_{X_T}+\|z\|^2_{X_T}+\|\tilde z\|^2_{X_T}\big)\|z-\tilde z\|_{X_T},
\end{gathered}
$$
which follow as in Propositions \ref{H1-Control} and \ref{H2-Control}, using $w\in C([0,T];W^{1,\infty})$ for the quadratic terms when $s=1$ and the algebra property of $H^2$ when $s=2$. With these estimates, $$\|\mathcal T_w(z)\|_{X_T}\leq C_1\delta'+C_2(\|z\|^2_{X_T}+\|z\|^3_{X_T})$$ and $$\|\mathcal T_w(z)-\mathcal T_w(\tilde z)\|_{X_T}\leq C_2(\|z\|_{X_T}+\|\tilde z\|_{X_T}+\|z\|^2_{X_T}+\|\tilde z\|^2_{X_T})\|z-\tilde z\|_{X_T},$$ where $$\delta':=\|z_0\|_{H^s_D}+\|z_T\|_{H^s_D}\leq2\delta.$$ Hence $\mathcal T_w$ maps the closed ball of radius $M=4C_1\delta$ of $X_T$ into itself and is a contraction there, for $\delta$ small enough.
