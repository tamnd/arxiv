<!-- definition-1: body_md -->
(Evaluated Conditional Mutual Information (eCMI) of an Algorithm)
Let $A:\mathcal{Z}^n\rightarrow \{0,1\}^{\mathcal{X}}$ be a randomized or deterministic algorithm. Let $\mathcal{D}$ be a realizable probability distribution on $\mathcal{Z}$ and let $\tilde{Z}\in \mathcal{Z}^{2n}$ be a realizable supersample consisting of $2n$ i.i.d. samples from $\mathcal{D}$. Let $\boldsymbol{\sigma} = (\boldsymbol{\sigma}(1), \boldsymbol{\sigma}(2), \dots, \boldsymbol{\sigma}(n))$ be independent uniform random variables with $\boldsymbol{\sigma}(i)\sim \text{Unif}(\{i,i+n\})$ for all $i\in[n]$. The independent random swaps $\boldsymbol{\sigma}$ are independent of $\tilde{Z}$ and the randomness of $A$. 
Define $\tilde{Z}_{\boldsymbol{\sigma}} = \{ (X_{\boldsymbol{\sigma}(1)}, Y_{\boldsymbol{\sigma}(1)}), \dots, (X_{\boldsymbol{\sigma}(n)}, Y_{\boldsymbol{\sigma}(n)})\} \in \mathcal{Z}^n$ as the $n$ training samples selected by $\boldsymbol{\sigma}$, we also refer to it as training data variables.
                                                                                                       
Let $\ell(h,(x,y)) = \1[h(x)\neq y]$ be a 0-1 loss function.
Define $\vec{\ell}:\{0,1\}^{\mathcal{X}} \times \mathcal{Z}^{2n} \rightarrow \mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\originalleft\{ 0,1 \aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\originalright\}^{2n}$ as vector of 0-1 losses on $2n$ samples defined by
$\vec{\ell}(A(\tilde{Z}_{\boldsymbol{\sigma}}),\tilde{Z})_{i} = \ell(A(\tilde{Z}_{\boldsymbol{\sigma}}), \tilde{Z}_{i})$ for all $i\in[2n]$.

\noindent
The evaluated conditional mutual information of $A$ with respect to the distribution $\mathcal{D}$ is
\[
\text{eCMI}_{\mathcal{D}}(\ell(A)) = I(\vec{\ell}(A(\tilde{Z}_{\boldsymbol{\sigma}}),\tilde{Z}); \boldsymbol{\sigma}|\tilde{Z}).
\]
The distribution-free, also known as worst-case evaluated conditional mutual information, of $A$ on a worst-case supersample is
\[
\text{eCMI}(\ell(A)) = \sup_{\substack{\tilde{z}\in \mathcal{Z}^{2n} \text{realizable}}} I(\vec{\ell}(A(\tilde{z}_{\boldsymbol{\sigma}}),\tilde{z}); \boldsymbol{\sigma}).
\]
And we have $\text{eCMI}_{\mathcal{D}}(\ell(A)) \leq \text{eCMI}(\ell(A))$ for any algorithm $A$ and distribution $\mathcal{D}$.

<!-- lemma-1: body_md -->
\label{lemma:epy_of_points}
For any supersample $\tilde{z} \in \mathcal{Z}^{2n}$. We have 
$$I\big(\vec{\ell}\big(A(\tilde{z}_{\boldsymbol{\sigma}},\boldsymbol{\beta}),\tilde{z}\big);\boldsymbol{\sigma}\big) \leq H\big(\vec{\ell}\big(A(\tilde{z}_{\boldsymbol{\sigma}},\boldsymbol{\beta}),\tilde{z}\big)\mid \boldsymbol{\beta}\big) \leq \sum_{i\leq 2n} H\big(\ell\big(A(\tilde{z}_{\boldsymbol{\sigma}},\boldsymbol{\beta}),\tilde{z}_{i}\big)\mid \boldsymbol{\beta}\big).$$
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Consider a fixed but arbitrary $\tilde{z} \in \mathcal{Z}^{2n}$. According to the chain rule for mutual information, we have
\begin{align*}
    I\big(\vec{\ell}\big(A(\tilde{z}_{\boldsymbol{\sigma}},\boldsymbol{\beta}),\tilde{z}\big);\boldsymbol{\sigma}\big) &= I\big(\big(\vec{\ell}\big(A(\tilde{z}_{\boldsymbol{\sigma}},\boldsymbol{\beta}),\tilde{z}\big), \boldsymbol{\beta} \big) ; \boldsymbol{\sigma}\big) - I\big(\boldsymbol{\beta}; \boldsymbol{\sigma} \mid \vec{\ell}\big(A(\tilde{z}_{\boldsymbol{\sigma}},\boldsymbol{\beta}),\tilde{z}\big)\big).
\end{align*}
By the non-negativity of mutual information, the above expression is bounded by
$$
   I\big(\big(\vec{\ell}\big(A(\tilde{z}_{\boldsymbol{\sigma}},\boldsymbol{\beta}),\tilde{z}\big), \boldsymbol{\beta} \big) ; \boldsymbol{\sigma}\big).
$$
Applying the chain rule for Shannon entropy, we have the above expression equals
\begin{align*}
    &H\big(\vec{\ell}\big(A(\tilde{z}_{\boldsymbol{\sigma}},\boldsymbol{\beta}),\tilde{z}\big)\mid\boldsymbol{\beta}\big) + H\big(\boldsymbol{\beta}\big) - H\big(\vec{\ell}\big(A(\tilde{z}_{\boldsymbol{\sigma}},\boldsymbol{\beta}),\tilde{z}\big) \mid \boldsymbol{\beta}, \boldsymbol{\sigma} \big) - H\big(\boldsymbol{\beta} \mid \boldsymbol{\sigma}\big).
                                                                                                                                                                                                                                                                                                                                                                   
\end{align*}
Since $H\big(\vec{\ell}\big(A(\tilde{z}_{\boldsymbol{\sigma}},\boldsymbol{\beta}),\tilde{z}\big) \mid \boldsymbol{\beta}, \boldsymbol{\sigma} \big) = 0$ and $H(\boldsymbol{\beta}) = H(\boldsymbol{\beta}\mid \boldsymbol{\sigma})$, 
                                                                         
we have the above expression equals
\begin{align*}
    &H\big(\vec{\ell}\big(A(\tilde{z}_{\boldsymbol{\sigma}},\boldsymbol{\beta}),\tilde{z}\big)\mid\boldsymbol{\beta}\big).
\end{align*}
By the subadditivity of Shannon entropy, we conclude
\begin{align*}
H\big(\vec{\ell}\big(A(\tilde{z}_{\boldsymbol{\sigma}},\boldsymbol{\beta}),\tilde{z}\big)\mid\boldsymbol{\beta}\big) \leq \sum_{i\leq 2n} H\big(\ell\big(A(\tilde{z}_{\boldsymbol{\sigma}},\boldsymbol{\beta}),\tilde{z}_{i}\big)\mid\boldsymbol{\beta}\big),
\end{align*}
as claimed.

<!-- lemma-2: body_md -->
\label{lemma:duplicates_adjust}

Since $S_1^{\boldsymbol{\beta},\boldsymbol{\sigma}},\dots,S_5^{\boldsymbol{\beta},\boldsymbol{\sigma}}$ has the same conditional distribution given $\boldsymbol{\sigma}$, without loss of generality, we consider $k=1$.
To show that the training data set $S_1^{\boldsymbol{\beta},\boldsymbol{\sigma}}$ is conditionally i.i.d. given $\boldsymbol{\sigma}$ from the empirical distribution of $n$ training data points, it suffices to show that the sequence of indices $I^{\boldsymbol{\pi},\boldsymbol{U}}_1$  of the training data set is i.i.d. $\mathrm{Unif}\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\originalleft( \mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left\{ 1, \dots, n \aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right\}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\originalright)$. 

Recall that the algorithm adjusts for duplicates in $I^{\boldsymbol{\pi}}_1$
                                                                                                                
by replacing the $\boldsymbol{N}$ distinct values $(\boldsymbol{v}_1,\dots,\boldsymbol{v}_{\boldsymbol{N}})$ in an i.i.d. sequence $\boldsymbol{U}$ with the first $\boldsymbol{N}$ distinct values $(\boldsymbol{\pi}(1),\dots,\boldsymbol{\pi}(\boldsymbol{N}))$ in $I^{\boldsymbol{\pi}}_1$.
To show that the reconstructed sequence $I_1^{\boldsymbol{\pi},\boldsymbol{U}}$ is i.i.d. $\mathrm{Unif}\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\originalleft( \{1,\dots,n\}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\originalright)$, we first characterize the distribution of $I_1^{\boldsymbol{\pi},\boldsymbol{U}}$.
Indeed, $I_1^{\boldsymbol{\pi},\boldsymbol{U}}$ is constructed by first sampling a random histogram of relative frequencies of distinct values and the ordering of the values from the conditional distribution of $G(\boldsymbol{U})$ given the number of the distinct values $\boldsymbol{N}$. Conditional on $\boldsymbol{N}$, a sequence of $\boldsymbol{N}$ distinct values $(\boldsymbol{\pi}(1),\dots,\boldsymbol{\pi}(\boldsymbol{N}))$ is drawn uniformly at random without replacement from $\{1,\dots,n\}$, independently of $G(\boldsymbol{U})$.
Assigning these distinct values to $G(\boldsymbol U)$ yields the reconstructed sequence $I_1^{\boldsymbol{\pi},\boldsymbol U}$.
                                                             

By the chain rule of probability, the above sampling process can be formalized as sampling sequentially from the following sequence of conditional distributions.
Therefore,
the reconstructed sequence of indices $I_1^{\boldsymbol{\pi},\boldsymbol{U}}$ is indeed sampled from the marginal distribution of the following joint distribution,
\begin{align*}
P_{I_1^{\boldsymbol{\pi},\boldsymbol{U}},G(\boldsymbol{U}),\boldsymbol{N}} = P_{I_1^{\boldsymbol{\pi},\boldsymbol{U}} | G(\boldsymbol{U}),\boldsymbol{N}} \cdot P_{G(\boldsymbol{U}) | \boldsymbol{N}} \cdot P_{\boldsymbol{N}}.
\end{align*}
To show that $I_1^{\boldsymbol{\pi},\boldsymbol{U}}$ has the same distribution as the i.i.d. sequence $\boldsymbol{U}$,
it suffices to establish $P_{I_1^{\boldsymbol{\pi},\boldsymbol{U}} | G(\boldsymbol{U}),\boldsymbol{N}}=P_{\boldsymbol{U}| G(\boldsymbol{U}),\boldsymbol{N}}$. 

Note that because $(\boldsymbol{\pi}(1),\dots,\boldsymbol{\pi}(\boldsymbol{N}))$ consists of the first $\boldsymbol{N}$ elements of a uniform random permutation, they form a uniform random sample drawn without replacement from $\{1,\dots,n\}$ given $\boldsymbol{N}$. 
Similarly, the sequence of distinct values $(\boldsymbol{v}_1, \dots,\boldsymbol{v}_{\boldsymbol{N}})$ of an i.i.d. sequence $\boldsymbol{U}$ also form a uniform random sample without replacement from $\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\originalleft\{ 1,\dots,n \aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\originalright\}$ given $\boldsymbol{N}$. It follows that
\begin{equation}
P_{\boldsymbol{\pi}(1),\dots,\boldsymbol{\pi}(\boldsymbol{N})|\boldsymbol{N}}=P_{\boldsymbol{v}_1, \dots,\boldsymbol{v}_{\boldsymbol{N}}|\boldsymbol{N}}.
\label{eq:pi-v}
\end{equation}
Since $\boldsymbol{U}= ( \boldsymbol{u}_1, \boldsymbol{u}_2, \dots, \boldsymbol{u}_{\frac{n}{5}})$ is a sequence of i.i.d. $\mathrm{Unif}\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\originalleft( \mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left\{ 1, \dots, n \aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right\} \aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\originalright)$ random variables, we know that $G(\boldsymbol{U})$ is conditionally independent of the uniform random choice of the $\boldsymbol{N}$ distinct values $(\boldsymbol{v}_1,\dots,\boldsymbol{v}_{\boldsymbol{N}})$ given $\boldsymbol{N}$. Moreover, by the construction of the algorithm, we also have that $G(\boldsymbol{U})$ is conditionally independent of the uniform random sequence of distinct values $(\boldsymbol{\pi}(1),\dots,\boldsymbol{\pi}(\boldsymbol{N}))$  drawn without replacement given $\boldsymbol{N}$. 
Therefore, \eqref{eq:pi-v} implies that
\begin{align}
\label{eq:cond_indep}
P_{\boldsymbol{\pi}(1),\dots,\boldsymbol{\pi}(\boldsymbol{N})|G(\boldsymbol{U}),\boldsymbol{N}} = P_{\boldsymbol{v}_1, \dots,\boldsymbol{v}_{\boldsymbol{N}}|G(\boldsymbol{U}),\boldsymbol{N}}.
\end{align}
Note that by definition, both $I^{\boldsymbol{\pi},\boldsymbol{U}}_1$ and $\boldsymbol{U}$ are obtained by applying the same function $f$ to $(\boldsymbol{\pi}(1),\dots,\boldsymbol{\pi}(\boldsymbol{N}))$ and $(\boldsymbol{v}_1, \dots,\boldsymbol{v}_{\boldsymbol{N}})$ respectively.  Together with~\eqref{eq:cond_indep}, this immediately implies that
$$
P_{I^{\boldsymbol{\pi},\boldsymbol{U}}_1|G(\boldsymbol{U}),\boldsymbol{N}} = P_{\boldsymbol{U}|G(\boldsymbol{U}),\boldsymbol{N}}.
$$
Altogether, we can conclude that $(I_1^{\boldsymbol{\pi},\boldsymbol{U}},G(\boldsymbol{U}),\boldsymbol{N})$ and $(\boldsymbol{U},G(\boldsymbol{U}),\boldsymbol{N})$ have identical joint distributions, which implies that the reconstructed sequence of indices $I_{1}^{\boldsymbol{\pi, U}}$ has an identical marginal distribution as the i.i.d. sequence $\boldsymbol{U}$. 
                              
 

Since an empirical distribution on $n$ training data points is equivalent to a uniform distribution over $n$ distinct indices,
we have that the training data set $S_1^{\boldsymbol{\beta},\boldsymbol{\sigma}} = \tilde{z}_{\boldsymbol{\sigma}(I^{\boldsymbol{\pi},\boldsymbol{U}}_1)} $ is conditionally i.i.d. from $\mathrm{Unif}\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\originalleft( \mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left\{ x_{\boldsymbol{\sigma}(1)},\dots,x_{\boldsymbol{\sigma}(n)} \aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right\}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\originalright)$ given $\boldsymbol{\sigma}$.
This concludes the proof of the lemma.

we can bound the conditional probability of error over $\boldsymbol{\sigma}$ given $\boldsymbol{\beta}$ of the Majority-of-5 learner at $x_i$, by the probability that at least two of the other four base learners both err at $x_i$. That is, for any $x_i \in \Gamma_1^{\boldsymbol{\pi}}$,
\begin{multline*}
    \mathbb{P}_{\boldsymbol{\sigma} | \boldsymbol{\beta}}\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(A(\tilde{z}_{\boldsymbol{\sigma}},\boldsymbol{\beta})(x_i)\neq h^*(x_i) | \boldsymbol{\beta}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right) \leq
    \mathbb{P}_{\boldsymbol{\sigma} | \boldsymbol{\beta}}\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(\exists\, 2\leq k'<k\leq 5: h_k^{\boldsymbol{\beta},\boldsymbol{\sigma}}(x_i)\neq h^*(x_i) \wedge h_{k'}^{\boldsymbol{\beta},\boldsymbol{\sigma}}(x_i)\neq h^*(x_i) |\boldsymbol{\beta} \aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right).
\end{multline*}
By a union bound, we have
\begin{align*}
    \mathbb{P}_{\boldsymbol{\sigma} | \boldsymbol{\beta}}\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(A(\tilde{z}_{\boldsymbol{\sigma}},\boldsymbol{\beta})(x_i)\neq h^*(x_i) | \boldsymbol{\beta}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right)
    \leq 
    \sum_{\substack{2 \leq k'<k \leq 5}}\mathbb{P}_{\boldsymbol{\sigma}| \boldsymbol{\beta}}\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(L_i(h_k^{\boldsymbol{\beta},\boldsymbol{\sigma}}) \wedge 
    L_i(h_{k'}^{\boldsymbol{\beta},\boldsymbol{\sigma}})|\boldsymbol{\beta}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right).
\end{align*}
By the symmetry of a uniform random permutation over $\tilde{z}_{\boldsymbol{\sigma}}$,
the conditional entropy of errors on $2n$ data points is five times that on the data points in $\Gamma^{\boldsymbol{\pi}}_1$. And $f(p) = p\log \frac{4}{p}$ is monontone non-decreasing on $[0,1]$, we have \eqref{eq:cond_entropy} bounded by
\begin{equation}
     5 \sum_{i\leq 2n}\mathop{\mathbb{E}}_{\boldsymbol{\beta}}\Bigg[\1[x_i\in \Gamma^{\boldsymbol{\pi}}_1 ](\sum_{\substack{2 \leq k'<k \leq 5}}\mathbb{P}_{\boldsymbol{\sigma}| \boldsymbol{\beta}}(L_i(h_k^{\boldsymbol{\beta},\boldsymbol{\sigma}}) \wedge 
    L_i(h_{k'}^{\boldsymbol{\beta},\boldsymbol{\sigma}})|\boldsymbol{\beta})) \log \frac{4}{ \min \{1, \sum_{\substack{2 \leq k'<k \leq 5}}\mathbb{P}_{\boldsymbol{\sigma}| \boldsymbol{\beta}}(L_i\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(h_k^{\boldsymbol{\beta},\boldsymbol{\sigma}}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right) \wedge L_i(h_{k'}^{\boldsymbol{\beta},\boldsymbol{\sigma}})|\boldsymbol{\beta}) \} }\Bigg].
\label{eq:error_first_block}
\end{equation}
Define the $j$-th conditional error region on the data points in the first part of the partition $\Gamma^{\boldsymbol{\pi}}_1$ based on the conditional probability of error of the $k$-th base learner over $\boldsymbol{\sigma}$ given $\boldsymbol{\beta}$:
\begin{align*}
    R^{k,\boldsymbol{\beta},1}_j := \mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left\{x \in \Gamma^{\boldsymbol{\pi}}_1: \mathbb{P}_{\boldsymbol{\sigma}|\boldsymbol{\beta}}(h_k^{\boldsymbol{\beta},\boldsymbol{\sigma}}(x) \neq h^*(x) | \boldsymbol{\beta}) \in (2^{-j}, 2^{-j+1}] \aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right\}.
\end{align*}
Let $j^{\boldsymbol{\beta}}_k(x_i)$ denote the index of the conditional error region such that the conditional probability of error of the $k$-th base learner at $x_i \in \Gamma^{\boldsymbol{\pi}}_1 $, is in the interval $(2^{-j^{\boldsymbol{\beta}}_k(x_i)}, 2^{-j^{\boldsymbol{\beta}}_k(x_i)+1}]$. 
 
By the linearity of expectation and the conditional independence of $S_k^{\boldsymbol{\beta},\boldsymbol{\sigma}}, S_{k'}^{\boldsymbol{\beta},\boldsymbol{\sigma}}$ given $\boldsymbol{\beta}$, we have \eqref{eq:error_first_block} bounded by
                                                                                                                                                      
                                                                                              
\begin{align*}
    &5 \sum_{i\leq 2n}\mathop{\mathbb{E}}_{\boldsymbol{\beta}}\Big[\1[x_i\in \Gamma^{\boldsymbol{\pi}}_1]
    (\sum_{\substack{2 \leq k'<k \leq 5}} 2^{-j^{\boldsymbol{\beta}}_k(x_i) -j^{\boldsymbol{\beta}}_{k'}(x_i)+2}) 
    \log \frac{4}{ \min\{1, \sum_{\substack{2 \leq k'<k \leq 5}} 2^{-j^{\boldsymbol{\beta}}_k(x_i)-j^{\boldsymbol{\beta}}_{k'}(x_i) +2 }\}}\Big]\\
    &\leq 5 \sum_{\substack{2 \leq k'<k \leq 5}} \mathop{\mathbb{E}}_{\boldsymbol{\beta}}\Big[\sum_{x_i \in \Gamma^{\boldsymbol{\pi}}_1 }
    (2^{-j^{\boldsymbol{\beta}}_k(x_i) -j^{\boldsymbol{\beta}}_{k'}(x_i)+2} ) 
    \log \frac{4}{2^{-j^{\boldsymbol{\beta}}_k(x_i)-j^{\boldsymbol{\beta}}_{k'}(x_i)+2}}\Big].
\end{align*}
Combined with the inequality $xy\leq \max\{x^2 , y^2\}$, the above expression is bounded by
\begin{align*}
    &5\cdot \binom{4}{2} \cdot \mathop{\mathbb{E}}_{\boldsymbol{\beta}}\Big[\sum_{x_i \in \Gamma^{\boldsymbol{\pi}}_1 }
    \max\{ 2^{-2j^{\boldsymbol{\beta}}_k(x_i)+2} , 2^{-2j^{\boldsymbol{\beta}}_{k'}(x_i)+2} \}
    \log \frac{4}{ \max\{ 2^{-2j^{\boldsymbol{\beta}}_k(x_i)+2} , 2^{-2j^{\boldsymbol{\beta}}_{k'}(x_i)+2} \} }\Big].
\end{align*}
Since $f(p)= p \log \frac{4}{p}$ is non-negative on $[0,1]$, it follows that $f(\max\{x^2, y^2\})\leq f(x^2)+f(y^2)$.
We have the above expression bounded by
\begin{align*}
    &5 \cdot 6 \cdot \mathop{\mathbb{E}}_{\boldsymbol{\beta}}\Big[\sum_{x_i \in \Gamma^{\boldsymbol{\pi}}_1 }
    2^{-2j^{\boldsymbol{\beta}}_k(x_i)+2} \log \frac{4}{2^{-2j^{\boldsymbol{\beta}}_k(x_i)+2}} +
     2^{-2j^{\boldsymbol{\beta}}_{k'}(x_i)+2} \log \frac{4}{2^{-2j^{\boldsymbol{\beta}}_{k'}(x_i)+2}}\Big]\\
    &\leq 30\cdot\Big( \mathop{\mathbb{E}}_{\boldsymbol{\beta}}[\sum_{j=1}^{\infty}|R^{k,\boldsymbol{\beta},1}_j|\cdot
    2^{-2j+2}\cdot 2j] +
     \mathop{\mathbb{E}}_{\boldsymbol{\beta}}[\sum_{j=1}^{\infty}|R^{k',\boldsymbol{\beta},1}_j|\cdot
    2^{-2j+2}\cdot 2j] \Big)\\
    &= 30 \cdot 4 \cdot 2 \cdot\Big( \sum_{j=1}^{\infty}\mathop{\mathbb{E}}_{\boldsymbol{\beta}}[|R^{k,\boldsymbol{\beta},1}_j|]\cdot
    2^{-2j}\cdot j +
     \sum_{j=1}^{\infty}\mathop{\mathbb{E}}_{\boldsymbol{\beta}}[|R^{k',\boldsymbol{\beta},1}_j|]\cdot
    2^{-2j}\cdot j\Big).
    \end{align*}
    According to Lemma \ref{expected_err_region}, since the $j$-th conditional error region in the first part of the partition $\Gamma^{\boldsymbol{\pi}}_1$ is a subset of the $j$-th conditional error region in the other $\frac{4n}{5}$ pairs of data points, i.e.,
    $R_j^{k,\boldsymbol{\beta},1} \subseteq R_j^{k,\boldsymbol{\beta}}$, we have
    $\mathop{\mathbb{E}}_{\boldsymbol{\beta}}[|R^{k,\boldsymbol{\beta},1}_j|] \leq \mathop{\mathbb{E}}_{\boldsymbol{\beta}}[|R^{k,\boldsymbol{\beta}}_j|] \leq 2^j\cdot 2nR(\frac{n}{5}).$
    Thus, the above expression is bounded by
    \begin{align*}
                                                                                          
                            
     480\cdot \Big(\sum_{j=1}^{\infty} j2^{-j}\Big) \cdot 2nR\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(\frac{n}{5}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right)
    = 1920 \cdot nR\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(\frac{n}{5}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right)
    =O\Big(nR\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(\frac{n}{5}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right)\Big).
\end{align*}
This concludes the proof.

        &\leq 
        \1\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left[h_2^{\tilde{Z},\boldsymbol{\sigma}}(X_1)\neq h^*(X_1)\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right] + \1\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left[h_3^{\tilde{Z},\boldsymbol{\sigma}}(X_1)\neq h^*(X_1)\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right].
    \end{align*}
    Since $S_2^{\tilde{Z},\boldsymbol{\sigma}}, S_3^{\tilde{Z},\boldsymbol{\sigma}}$ are sampled independently of the test sample $X_1\sim D$,
    we have the expected training error rate of the Majority-of-3 learner bounded by the expected true error rate of the other two base learners. That is,
    \begin{align*}
        &\mathop{\mathbb{E}}_{\tilde{Z}_{\boldsymbol{\sigma}}\sim D^n}\Big[ \1[A(\tilde{Z}_{\boldsymbol{\sigma}})(X_1)\neq h^*(X_1)] \Big]\\
                                                                                                                                                 
        &\leq\mathop{\mathbb{E}}_{\tilde{Z}_{\boldsymbol{\sigma}} \sim D^n} \mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left[\1[\mathcal{B}\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(S_2^{\tilde{Z},\boldsymbol{\sigma}}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right)(X_1)\neq h^*(X_1)]\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right] + \mathop{\mathbb{E}}_{\tilde{Z}_{\boldsymbol{\sigma}} \sim D^n} \Big[\1[\mathcal{B}\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(S_3^{\tilde{Z},\boldsymbol{\sigma}}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right)(X_1)\neq h^*(X_1)]\Big] \\
        &= \mathop{\mathbb{E}}_{S_2^{\tilde{Z},\boldsymbol{\sigma}}\sim \mathcal{D}^{\frac{n}{3}}}\Big[\mathbb{P}_{X_1\sim \mathcal{D}}\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(\mathcal{B}\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(S_2^{\tilde{Z},\boldsymbol{\sigma}}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right)(X_1)\neq h^*(X_1)\big| S_2^{\tilde{Z},\boldsymbol{\sigma}} \aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right)\Big] + \mathop{\mathbb{E}}_{S_3^{\tilde{Z},\boldsymbol{\sigma}}\sim \mathcal{D}^{\frac{n}{3}}}\Big[\mathbb{P}_{X_1\sim \mathcal{D}}\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(\mathcal{B}\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\mathopen{}\mathclose\bgroup\left(S_3^{\tilde{Z},\boldsymbol{\sigma}}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right)(X_1)\neq h^*(X_1) \big| S_3^{\tilde{Z},\boldsymbol{\sigma}}\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\aftergroup\egroup\right)\Big]\\
        &\leq 2\cdot R(\frac{n}{3}).
    \end{align*}
This completes the proof of Theorem \ref{thm:distributional eCMI(A)}.
