{\rm(\cite{GLS})}
	 \begin{enumerate}
		\item
		Let $(G, \cdot_G, e_G)$ be a Lie group.  A smooth map $\mathcal{B}: G\to G$ is called  a {\bf Rota-Baxter operator} on $G$ if the following equality holds:
		\begin{equation}\label{defiB}
		\mathcal{B}(g)\cdot_G\mathcal{B}(h)=\mathcal{B}(g\cdot_G\mathrm{Ad}_{\mathcal{B}(g)}h), \quad \forall g, h\in G.
		\end{equation}
		The pair $(G, \mathcal{B})$ is called a Rota-Baxter Lie group.
	\item Let $(G, \mathcal{B})$  and $(G', \mathcal{B}')$ be Rota-Baxter Lie groups. A smooth map $\Psi:G\to G'$ is called a {\bf Rota-Baxter Lie group homomorphism} if $\Psi$ is a Lie group homomorphism and 
    \begin{equation*}
	\Psi\circ \mathcal{B}=\mathcal{B}'\circ \Psi.
	\end{equation*}
	 \end{enumerate}

{\rm(\cite{BGN}, \cite{BD})}
	Let $(\mathfrak g, [\cdot, \cdot]_\mathfrak g)$ be a Lie algebra. A linear map $B:\mathfrak g\to\mathfrak g$ is called a {\bf Rota-Baxter operator of weight $1$} on $\mathfrak g$ if 
	\begin{equation*}
		[B(x), B(y)]_\mathfrak g=B([B(x), y]_\mathfrak g+[x, B(y)]_\mathfrak g+[x, y]_\mathfrak g), \quad \forall x, y\in\mathfrak g.
	\end{equation*}
	The pair $(\mathfrak g, B)$ is called a {\bf Rota-Baxter Lie algebra of weight $1$.}
<!-- end defi-2 -->

In the sequel, we will call Rota-Baxter Lie algebras of weight $1$ simply Rota-Baxter Lie algebras when there is no ambiguity.

{\rm(\cite{DR})}\label{defirep}
	Let $V$ be a vector space and $T: V\to V$ a linear map. A {\bf representation} of a Rota-Baxter Lie group $(G, \mathcal{B})$ on $(V, T)$ is a Lie group homomorphism $\Theta$ from $G$ to $\mathrm{GL}(V)$ such that 
	\begin{equation}\label{rep}
		\Theta(\mathcal{B}(a))T(u)=T\Big(\Theta(a\cdot_G\mathcal{B}(a))u+\Theta(a\cdot_G\mathcal{B}(a))T(u)-\Theta(\mathcal{B}(a))T(u)\Big),\quad \forall a\in G, ~u\in V.
	\end{equation}
	We denote a representation by $(V, T, \Theta)$.

\label{semirb}
Let $(G, \mathcal{B})$ be a Rota-Baxter Lie group and $\Theta: G\to \mathrm{GL}(V)$ a representation of $G$. Let $T: V\to V$ be a linear map. Consider the semidirect product Lie group $G\ltimes_\Theta V$, whose multiplication $\cdot_\ltimes$ is given by
	\begin{equation*}
		(a, u)\cdot_{\ltimes}(b, v)=(a\cdot_G b, u+\Theta(a)v), \quad \forall a, b\in G, u, v\in V,
	\end{equation*}
Define $\mathrm{B}: G\times V\to G\times V$ by
	\begin{equation*}
		\mathrm{B}(a, u)=\Big(\mathcal{B}(a), \Theta(\mathcal{B}(a))T(\Theta(\mathcal{B}(a)^{-1}\cdot_G a^{-1})u)\Big),\quad \forall a\in G, u\in V.
	\end{equation*}
Then $(G\ltimes_\Theta V, \mathrm{B})$ is a Rota-Baxter Lie group if and only if $(V, \Theta, T)$ is a representation of $(G, \mathcal{B})$.
<!-- end pro-1 -->
<!-- proof-1: body_md -->
On the one hand, for any $a, b\in G, u, v\in V$, we have
	\begin{equation*}
		\mathrm{B}(a, u)\cdot_{\ltimes}\mathrm{B}(b, v)=\Big(\mathcal{B}(a)\cdot_G\mathcal{B}(b), \Theta(\mathcal{B}(a))T(\Theta(\mathcal{B}(a)^{-1}\cdot_G a^{-1})u)+\Theta(\mathcal{B}(a)\cdot_G\mathcal{B}(b))T(\Theta(\mathcal{B}(b)^{-1}\cdot_G b^{-1})v)\Big).
	\end{equation*}
On the other hand, 
\begin{align*}
&\quad \mathrm{B}\Big((a, u)\cdot_{\ltimes}\mathrm{B}(a, u)\cdot_{\ltimes}(b, v)\cdot_{\ltimes}\mathrm{B}(a, u)^{-1}\Big)\\
&=\Big(\mathcal{B}(a\cdot_G\mathcal{B}(a)\cdot_G b\cdot_G\mathcal{B}(a)^{-1}), \Theta(\mathcal{B}(a\cdot_G\mathcal{B}(a)\cdot_G b\cdot_G\mathcal{B}(a)^{-1}))T\Big(\Theta(\mathcal{B}(b)^{-1}\cdot_G b^{-1}\cdot_G\mathcal{B}(a)^{-1}\cdot_G a^{-1})u\\
&\quad+\Theta(\mathcal{B}(b)^{-1}\cdot_G b^{-1})T(\Theta(\mathcal{B}(a)^{-1}\cdot_G a^{-1})u)+\Theta(\mathcal{B}(b)^{-1}\cdot_G b^{-1})v-\Theta(\mathcal{B}(b)^{-1})T(\Theta(\mathcal{B}(a)^{-1}\cdot_G a^{-1})u)\Big) \Big).
\end{align*}
By \cite[Proposition 3.5]{JLS}, $\mathcal{B}(b)^{-1}=\mathcal{B}(\mathcal{B}(b)^{-1}\cdot_G b^{-1}\cdot_G\mathcal{B}(b))$, let $w=\Theta(\mathcal{B}(a)^{-1}\cdot_G a^{-1})u$. Then to prove $\mathrm{B}$ is a Rota-Baxter operator if and only if 
\begin{align}\label{eqrepth}
&\quad\Theta(\mathcal{B}\Big(\mathcal{B}(b)^{-1}\cdot_G b^{-1}\cdot_G\mathcal{B}(b))\Big)T(w)\\
\nonumber&=T\Big(\Theta(\mathcal{B}(b)^{-1}\cdot_G b^{-1})w+\Theta(\mathcal{B}(b)^{-1}\cdot_G b^{-1})T(w)-\Theta(\mathcal{B}(b)^{-1})T(w)\Big).
\end{align}
By \cite[Proposition 3.5]{JLS}, for any $g\in G$, there exists a unique $h\in G$ such that 
\begin{equation*}
	g=\mathcal{B}(h)^{-1}\cdot_G h^{-1}\cdot_G\mathcal{B}(h).
\end{equation*}
Thus \eqref{eqrepth} holds if and only if for any $g\in G, r\in V$ such that
\begin{equation*}
\Theta(\mathcal{B}(g))T(r)=T\Big(\Theta(g\cdot_G\mathcal{B}(g))r+\Theta(g\cdot_G\mathcal{B}(g))T(r)-\Theta(\mathcal{B}(g))T(r)\Big).
\end{equation*}
We finish the proof.

\label{repad}
	For a Rota-Baxter Lie group $(G, \mathcal{B})$, $(\mathfrak g, B, \widetilde{\mathrm{Ad}})$ is a representation of $(G, B)$ on $(\mathfrak g, B)$.
<!-- end pro-2 -->
<!-- proof-2: body_md -->
According to \cite[Proposition 3.15]{JSZ}, $\mathcal{B}(a)^{-1}=\mathcal{B}(\mathrm{Ad}_{\mathcal{B}(a)^{-1}}a^{-1})$ for any $a\in G$. Denote the exponential map 
of the Lie group $G$ by $\exp$. By the fact that $\mathcal{B}$ is a Rota-Baxter operator, for any $a\in G, u\in V$, we have
\begin{align*}
\widetilde{\mathrm{Ad}}_{\mathcal{B}(a)}(B(u))	&=\frac{d}{dt}|_{t=0}\mathrm{Ad}_{\mathcal{B}(a)}\mathcal{B}(\exp(tu))\\
	&=\frac{d}{dt}|_{t=0}\mathcal{B}\Big(\mathrm{Ad}_{a\cdot_G\mathcal{B}(a)}(\exp(tu)\cdot_G\mathcal{B}(\exp(tu)))\cdot_G\mathrm{Ad}_{\mathcal{B}(a)}\mathcal{B}(\exp(tu))^{-1}\Big)\\
	&=B\Big(\frac{d}{dt}|_{t=0}\mathrm{Ad}_{a\cdot_G\mathcal{B}(a)}(\exp(tu)\cdot_G\mathcal{B}(\exp(tu)))+\frac{d}{dt}|_{t=0}\mathrm{Ad}_{\mathcal{B}(a)}\mathcal{B}(\exp(tu))^{-1}\Big)\\
	&=B(\widetilde{\mathrm{Ad}}_{a\cdot_G\mathcal{B}(a)}(u+B(u))-\widetilde{\mathrm{Ad}}_{\mathcal{B}(a)}B(u)),
	\end{align*}
	which means that $(\mathfrak g, B, \widetilde{\mathrm{Ad}})$ is a representation of $(G, \mathcal{B})$ on $(\mathfrak g, B)$.

{\rm(\cite{DAS})}
	Let $V$ be a finite dimensional vector space and $T:V\to V$ a linear map. A {\bf representation} of $(\mathfrak g, B)$ on $(V, T)$ is a Lie algebra representation $\theta: \mathfrak g\to \mathfrak {gl}(V)$ satisfying
	\begin{equation}\label{defth}
		\theta(B(x))T(u)=T(\theta(B(x))u+\theta(x)T(u)+\theta(x)u)\quad \forall x\in\mathfrak g, ~u\in V.
	\end{equation}
	We denote a representation by $(V, T, \theta)$.

\label{diffr}
	Let $(V, T, \Theta)$ be a representation of a Rota-Baxter Lie group $(G, \mathcal{B})$. Then $(V, T, \theta)$ is a representation of the Rota-Baxter Lie algebra $(\mathfrak g, \mathcal{B}_{*e_G})$, where $\mathfrak g=T_{e_G}G$ and $\theta=\Theta_{*e_G}$.
<!-- end pro-3 -->
<!-- proof-3: body_md -->
Denote the exponential map of the Lie group $G$ by $\exp$. Then we have
	\begin{align*}
		\theta(B(x))T(u)&=\frac{d}{dt}|_{t=0}\Theta(\mathcal{B}(\exp(tx)))T(u)\\
		&=T\Big(\Theta(\exp(tx)\cdot_G\mathcal{B}(\exp(tx)))u+\Theta(\exp(tx)\cdot_G\mathcal{B}(\exp(tx)))T(u)-\Theta(\mathcal{B}(\exp(tx)))T(u)\Big)\\
		&=T(\theta(B(x))u+\theta(x)T(u)+\theta(x)u),
	\end{align*}
	for any $x\in\mathfrak g, u\in V$. Thus $(V, T, \theta)$ is a representation of the Rota-Baxter Lie algebra $(\mathfrak g, \mathcal{B}_{*e_G})$.

\label{intrep}
	With the above notations, let $\Theta:G\to \mathrm{GL}(V)$ be the representation of $G$ corresponding to $\theta:\mathfrak g\to\mathfrak {gl}(V)$. Then $(V, T, \Theta)$ is a representation of $(G, \mathcal{B})$.
<!-- end pro-4 -->
<!-- proof-4: body_md -->
Since $\mathcal{B}$ is a Rota-Baxter operator on $G$, the set 	
\begin{equation*}
H=\{(a\cdot_G\mathcal{B}(a), \mathcal{B}(a))|\forall a\in G\}\subset G\times G
\end{equation*}
 is a Lie subgroup of the direct product Lie group $G\times G$, with Lie algebra 
 \begin{equation*}
 \mathfrak h=\{({\rm{Id}}+B)(x), B(x)|\forall x\in \mathfrak g\}.
 \end{equation*}
  Moreover, the map $\kappa:H\to G$ defined by $\kappa(a\cdot_G\mathcal{B}(a), \mathcal{B}(a))=a$ is a homeomorphism, consequently $H$ is a connected and simply connected Lie group. 

Define the vector space $W=\{(({\rm{Id}}+T)u, T(u))|\forall u\in V\}$. Consider the representation of the direct product Lie algebra $\mathfrak g\oplus\mathfrak g$ on $V\times V$, given by 
\begin{equation*}
\lambda:\mathfrak g\oplus\mathfrak g\to\mathfrak {gl}(V\times V), \quad	\lambda(x, y)(u, v)=(\theta(x)u, \theta(y)v), \quad x, y\in\mathfrak g, ~\forall u, v\in V,
\end{equation*}
and let 
\begin{equation*}
\Lambda: G\times G\to \mathrm{GL}(V\times V), \quad \Lambda(a, b)(u, v)=(\Theta(a)u, \Theta(b)v), \quad \forall a, b\in G,~u, v\in V.
\end{equation*}
be the corresponding representation of $G\times G$ on $V\times V$. By \eqref{defth}, the restriction $\lambda|_{\mathfrak h}$ preserves $W$, and hence induces a representation of $\mathfrak h$ on $W$. Since $H$ is connected and simply connected, this representation of $\mathfrak h$ integrates to a representation of $H$ on $W$, which is precisely $\Lambda|_H$. It follows that
\begin{equation*}
\Lambda|_{H}(a\cdot_G\mathcal{B}(a), \mathcal{B}(a))(({\rm{Id}}+T)u, T(u))=(\Theta(a\cdot_G\mathcal{B}(a))({\rm{Id}}+T)u, \Theta(\mathcal{B}(a))T(u))\in W,
\end{equation*}
which gives
\begin{equation*}
\Theta(\mathcal{B}(a))T(u)=T\Big(\Theta(a\cdot_G\mathcal{B}(a))u+\Theta(a\cdot_G\mathcal{B}(a))T(u)-\Theta(\mathcal{B}(a))T(u)\Big).
\end{equation*} 
Thus $(V, T, \Theta)$ is a representation of $(G, \mathcal{B})$.

\begin{equation}\label{repB}
\Theta_{\mathcal{B}}(a)u=({\rm{Id}}_V+T)\Theta(\mathcal{B}(a))u-T\Big(\Theta(a\cdot_G\mathcal{B}(a))u\Big),\quad \forall a\in G, u\in V.
\end{equation}

For any $a, b\in G, u\in V$, since 
	\begin{align*}
	\Theta_{\mathcal{B}}(a)\Theta_{\mathcal{B}}(b)u&=({\rm{Id}}_V+T)\Theta(\mathcal{B}(a))({\rm{Id}}_V+T)\Theta(\mathcal{B}(b))u-({\rm{Id}}_V+T)\Theta(\mathcal{B}(a))T(\Theta(b\cdot_G\mathcal{B}(b))u)\\
	&\quad-T\Big(\Theta(a\cdot_G\mathcal{B}(a))({\rm{Id}}_V+T)\Theta(\mathcal{B}(b))u\Big)+T\Big(\Theta(a\cdot_G\mathcal{B}(a))T(\Theta(b\cdot_G\mathcal{B}(b)))u\Big)\\

For any $a_i\in G$ and $f\in C^{n}(G, V)$, by \eqref{repB} and \eqref{dec}, we have
\begin{align*}
&\quad\Big(c_{k+1}(\mathrm{d}^{\Theta}f)+\mathrm{d}^{\mathcal{B}}c_{k}(f)\Big)(a_1,\cdots, a_{k+1})\\
&=(-1)^{k+1}\Big(({\rm{Id}}_V+T)\mathrm{d}^\Theta f(\mathcal{B}(a_1), \cdots, \mathcal{B}(a_{k+1}))-T(\mathrm{d}^\Theta f(a_1\cdot_G\mathcal{B}(a_1), \cdots, a_{k+1}\cdot_G\mathcal{B}(a_{k+1})))\Big)\\

&\quad+(-1)^{k+1}f(\mathcal{B}(a_1),\cdots, \mathcal{B}(a_k))\Big)-(-1)^{k+1}T\Big(\Theta(a_1\cdot_G\mathcal{B}(a_1))f(a_2\cdot_G\mathcal{B}(a_2), \cdots, a_{k+1}\cdot_G\mathcal{B}(a_{k+1}))\\
&\quad+\sum_{i=1}^{k}(-1)^{i}f(a_1\cdot_G\mathcal{B}(a_1),\cdots,a_i\cdot_G\mathcal{B}(a_i)\cdot_G a_{i+1}\cdot_G\mathcal{B}(a_{i+1}),\cdots, a_{k+1}\cdot_G\mathcal{B}(a_{k+1}))\\
&\quad+(-1)^{k+1}f(a_1\cdot_G\mathcal{B}(a_1), \cdots, a_{k}\cdot_G\mathcal{B}(a_{k})))\Big)+(-1)^{k}({\rm{Id}}_V+T)\Theta(\mathcal{B}(a_1))\Big(({\rm{Id}}_V+T)f(\mathcal{B}(a_2), \cdots, \mathcal{B}(a_{k+1}))\\
&\quad-T(f(a_2\cdot_G\mathcal{B}(a_2), \cdots, a_{k+1}\cdot_G\mathcal{B}(a_{k+1})))\Big)-(-1)^{k}T\Big(\Theta(a_1\cdot_G\mathcal{B}(a_1))\Big(({\rm{Id}}_V+T)f(\mathcal{B}(a_2), \cdots, \mathcal{B}(a_{k+1}))\\
&\quad-T(f(a_2\cdot_G\mathcal{B}(a_2), \cdots, a_{k+1}\cdot_G\mathcal{B}(a_{k+1})))\Big)\Big)+\sum_{i=1}^{k}(-1)^{i+k}\Big(({\rm{Id}}_V+T)f(\mathcal{B}(a_1), \cdots, \mathcal{B}(a_i)\cdot_G\mathcal{B}(a_{i+1}), \cdots, \mathcal{B}(a_k))\\
&\quad-T(f(a_1\cdot_G\mathcal{B}(a_1), \cdots, a_i\cdot_G\mathcal{B}(a_i)\cdot_G a_{i+1}\cdot_G\mathcal{B}(a_{i+1}),\cdots, a_k\cdot_G\mathcal{B}(a_k)))\Big)\\

&=(-1)^{k}\Big(({\rm{Id}}_V+T)\Theta(\mathcal{B}(a_1))T\Big(f(\mathcal{B}(a_2), \cdots, \mathcal{B}(a_{k+1}))\Big)+T\Big(\Theta(a_1\cdot_G\mathcal{B}(a_1))f(a_2\cdot_G\mathcal{B}(a_2), \cdots, a_{k+1}\cdot_G\mathcal{B}(a_{k+1}))\Big)\\
&\quad-({\rm{Id}}_V+T)\Theta(\mathcal{B}(a_1))T\Big(f(a_2\cdot_G\mathcal{B}(a_2), \cdots, a_{k+1}\cdot_G\mathcal{B}(a_{k+1}))\Big)\\
&\quad-T\Big(\Theta(a_1\cdot_G\mathcal{B}(a_1))({\rm{Id}}_V+T)f(\mathcal{B}(a_2), \cdots, \mathcal{B}(a_{k+1}))\Big)\\
&\quad+T\Big(\Theta(a_1\cdot_G\mathcal{B}(a_1))T(f(a_2\cdot_G\mathcal{B}(a_2), \cdots, a_{k+1}\cdot_G\mathcal{B}(a_{k+1})))\Big)\Big).
\end{align*}

\begin{equation*}

\end{equation*}

\begin{equation*}

<!-- thm-6: body_md -->
\label{chainhomtopy}
The map $\widetilde{\mathrm{VE}}: \oplus_{n=1}^{+\infty}C^{n}(G, \mathcal{B}, V, T)\,\rightarrow\, \oplus_{n=1}^{+\infty}C^{n}(\mathfrak g, B, V, T)$ is a cochain map, which induces homomorphisms $\widetilde{\mathrm{VE}}_{n*}$ from the cohomology group $H^{n}(G, \mathcal{B}, V, T)$ to $H^{n}(\mathfrak g, B, V, T)$. The map $\widetilde{\mathrm{VE}}$ is called the {\bf van Est map}.

<!-- proof-9: body_md -->
For $n\geq1, \alpha_n\in C^{n}(G, V),  \beta_{n-1}\in C^{n}(\mathcal{B}, T)$, following the classical argument used for the cohomology of Lie groups and Lie algebras, we have
	\begin{eqnarray*}
		\widetilde{\mathrm{VE}}_{n+1}(\mathrm{d}^{\Theta}\alpha_n, c_n(\alpha_n)+\mathrm{d}^\mathcal{B}\beta_{n-1})&=&(\mathrm{VE}_{n+1}(\mathrm{d}^{\Theta}\alpha_n), \mathrm{VE}_{n}(c_n(\alpha_n))+\mathrm{VE}_{n}(\mathrm{d}^\mathcal{B}\beta_{n-1}))\\
		&=&\Big(\mathrm{d}^{\theta}(\mathrm{VE}_{n}(\alpha_n)), \mathrm{VE}_{n}(c_n(\alpha_n))+\mathrm{d}^{\theta_{B}}(\mathrm{VE}_{n-1}(\beta_{n-1}))\Big).
	\end{eqnarray*}
On the other hand, by definition of $\mathrm{d}$, we also have
\begin{equation*}
\mathrm{d}(\widetilde{\mathrm{VE}}_{n}(\alpha_n, \beta_{n-1}))=(\mathrm{d}^{\theta}(\mathrm{VE}_{n}(\alpha_n)), K_n(\mathrm{VE}_{n}(\alpha_n))+\mathrm{d}^{\theta_B}(\mathrm{VE}_{n-1}(\beta_{n-1}))),
\end{equation*}

Belwal, Rathee and Singh \cite{BRS} developed a low-dimensional cohomology theory for relative Rota–Baxter groups and used its second cohomology group to classify abelian and central extensions. When the action is given by the adjoint action, their framework is closely related to that of Rota–Baxter groups. In contrast, we construct a cochain complex in arbitrary degrees via a mapping cone construction, which also can classify the central extensions of Rota-Baxter Lie groups.	It remains unclear whether this chain complex can be used to classify abelian Rota–Baxter group extensions.
