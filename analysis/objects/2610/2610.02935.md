<!-- lem-1: body_md -->
\label{lem:Schwarz-Pick}
Let $\phi(z)=\sum_{n=0}^{\infty} a_n z^n$
be an analytic function in the unit disk $\mathbb{D}=\{z\in\mathbb{C}: |z|<1\},$
such that $|\phi(z)|<1$ for all $z\in\mathbb{D}$. Then the following assertions hold:

\begin{enumerate}
\item For any $z_1, z_2 \in \mathbb{D}$,
\[
\frac{|\phi(z_1)-\phi(z_2)|}{|1-\phi(z_1)\overline{\phi(z_2)}|}
\le
\frac{|z_1-z_2|}{|1-z_1\overline{z_2}|}.
\]
Moreover, equality holds for distinct $z_1, z_2 \in \mathbb{D}$ if and only if
$\phi$ is a M\"{o}bius transformation of $\mathbb{D}$.

\item For any $z\in\mathbb{D}$,
\[
|\phi'(z)| \le \frac{1-|\phi(z)|^2}{1-|z|^2}.
\]
Furthermore, equality holds for some $z\in\mathbb{D}$ if and only if
$\phi$ is a M\"{o}bius transformation.
\end{enumerate}

Suppose that $B_{X}$ and $B_{Y}$ are the unit balls of the complex Banach spaces $X$ and $Y$, respectively. Let $F: B_{X} \to B_{Y}$ be a holomorphic mapping. Then

This estimate is sharp with equality possible for each value of $\|f(0)\|_{Y}$ and for each $z \in B_{X}$.

Let $z\in B_X\setminus\{0\}$ be fixed and define $z_0=\frac{z}{\|z\|_X}\in \partial B_X.$ Also let $f(\zeta)=F(\zeta z_{0})$. Then $f$ is a holomorphic functions in $\mathbb{U}$ and
\begin{eqnarray}\label{te1.0}
f(\zeta)=a+\sum_{s=1}^{\infty}(P_{s})(z_{0})\zeta^{s},\qquad \zeta\in\mathbb{U}.
\end{eqnarray}
Therefore,  we have (see e.g. \cite[p.\,35]{GK2003}), $|P_{s}(z_{0})|\leq 1-|a|^2$ for all $s\geq 1$. Now applying Lemma \ref{L2}, we deduce that 

\begin{eqnarray}\label{te1.1} &&\sum_{s=2}^{\infty}|P_{s}(z_{0})|\,|\zeta|^{s}
+\left(\frac{1}{1+|a|}+\frac{|\zeta|}{1-|\zeta|}\right)
\sum_{s=1}^{\infty} |P_{s}(z_{0})|^2 |\zeta|^{2s}\nonumber\\
& \le & (1-|a|^2)\frac{|\zeta|^2}{1-|\zeta|}.\end{eqnarray}
Let $\zeta=\|z\|_X=r<1$. Then by Eq. (\ref{te1.1}), we deduce that 
\begin{eqnarray}\label{te1.2}\sum_{s=2}^{\infty}|P_{s}(z)|+\left(\frac{1}{1+|a|}+\frac{\|z\|_X}{1-\|z\|_X}\right)
\sum_{s=1}^{\infty} |P_{s}(z)|^2 \leq (1-|a|^2)\frac{\|z\|_X^2}{1-\|z\|_X}.\end{eqnarray}

Note that $t(x)=\frac{x}{1-x}$. Then $t'(x)=\frac{1}{(1-x)^2}>0,$ so $t(x)$ is increasing on $(0,1)$, since $r^k\leq r$ for $k\geq 1$, it follows that 
\[t(r^k)\leq t(r).\]
Hence 
\[\frac{r^k}{1-r^k}\leq \frac{r}{1-r} =\frac{\|z\|_X}{1-\|z\|_X}.\]
That is 
\begin{eqnarray}\label{ne1}\frac{1}{1+|a|}+\frac{r^k}{1-r^k}\leq \frac{1}{1+|a|}+\frac{\|z\|_X}{1-\|z\|_X}\end{eqnarray}
Therefore from (\ref{te1.2}) and (\ref{ne1}), we get 
\begin{eqnarray*} &&\sum_{s=2}^{\infty}|P_{s}(z)|+\left(\frac{1}{1+|a|}+\frac{r^k}{1-r^k}\right)
\sum_{s=1}^{\infty} |P_{s}(z)|^2\\
&\leq &\sum_{s=2}^{\infty}|P_{s}(z)|+\left(\frac{1}{1+|a|}+\frac{\|z\|_X}{1-\|z\|_X}\right)
\sum_{s=1}^{\infty} |P_{s}(z)|^2 \leq (1-|a|^2)\frac{\|z\|_X^2}{1-\|z\|_X}\end{eqnarray*}
and so 
\begin{eqnarray}\label{ne2} \sum_{s=2}^{\infty}|P_{s}(z)|+\left(\frac{1}{1+|a|}+\frac{r^k}{1-r^k}\right)
\sum_{s=1}^{\infty} |P_{s}(z)|^2\leq (1-|a|^2)\frac{\|z\|_X^2}{1-\|z\|_X}.\end{eqnarray}

In view of (\ref{ne2}) and (\ref{eq:Schwarz}), we obtain
\begin{eqnarray}\label{te1.3}&&\sum_{s=2}^{\infty}|P_{s}(\mu_k(z))|+\left(\frac{1}{1+|a|}+\frac{r^k}{1-r^k}\right)
\sum_{s=1}^{\infty} |P_{s}(\mu_k(z))|^2\nonumber\\
 &\leq & (1-|a|^2)\frac{\|\mu_k(z)\|_X^2}{1-\|\mu_k(z)\|_X}\;\left(\text{as $\frac{x}{1-x}$ is increasing in $(0,1)$}\right)\nonumber\\
&\leq &(1-|a|^2)\frac{\|z\|_X^{2k}}{1-\|z\|_X^k}.\end{eqnarray}
Differentiating (\ref{te1.0}), we have
\[f'(\zeta)=D F(\zeta z_{0}) z_0.\]
Then by Schwarz-Pick Lemma, we can conclude that 
\[ |D F(\zeta z_{0}) z_0|=|f'(\zeta)|\leq \frac{1-|f(\zeta)|^2}{1-|\zeta|^2}.\]
Let $\zeta=\|z\|_X=r<1$. Then the above Eq. becomes
\begin{eqnarray}\label{te1.4} |D F(z) z|\leq \left(\frac{1-|F(z)|^2}{1-\|z\|^2}\right) \|z\|_X.\end{eqnarray}
By combining \eqref{te1.4} and \eqref{eq:Schwarz}, we obtain

\begin{eqnarray}\label{te1.5} |D F(\mu_m(z)) \mu_m(z)|\leq \left(\frac{1-|F(\mu_m(z))|}{1-\|\mu_m(z)\|^2}\right) \|\mu_m(z)\|_X\leq \left(\frac{1-|F(\mu_m(z))|}{1-\|z\|_X^{2m}}\right) \|z\|_X^m.\end{eqnarray}
Let $x=|a|\in[0,1]$. If $x=1$, then $P_s(z_0)=0$ for all $s\ge1$, and
\eqref{1.1} holds trivially. Hence, we assume $x\in[0,1)$.  Then from \eqref{te1.4} and \eqref{te1.5}, we obtain
\begin{eqnarray}\label{te1.6} &&|F(\mu_m(z))|+|D F(\mu_m(z)) \mu_m(z)|+\sum_{s=2}^{\infty}|P_{s}(\mu_k(z))|+\left(\frac{1}{1+|a|}+\frac{r^k}{1-r^k}\right)
\sum_{s=1}^{\infty} |P_{s}(\mu_k(z))|^2\nonumber\\
&&\leq |F(\mu_m(z))|+\left(\frac{1-|F(\mu_m(z))|^2}{1-r^{2m}}\right) r^m+(1-x^2)\frac{r^{2k}}{1-r^k}\nonumber\\
&&\leq  \frac{r^{m}+x}{1+r^{m}x}+\left(1-\left(\frac{r^{m}+x}{1+r^{m}x}\right)^2\right)\frac{r^m}{1-r^{2m}}+(1-x^2)\frac{r^{2k}}{1-r^k}\nonumber\\
&&=1+\frac{(1-r^m)(x-1)}{1+xr^m}+\frac{(1-x^2)(1-r^{2m})}{(1+xr^m)^2}\frac{r^m}{1-r^{2m}}+(1-x^2)\frac{r^{2k}}{1-r^k}\nonumber\\
&&=1+\frac{(1-x)D_1(x,r)}{1+xr^m},\end{eqnarray}
where 
\[D_1(x,r)=-(1-r^m)+\frac{(1+x)r^m}{1+xr^m}+\frac{r^{2k}(1+x)(1+xr^m)}{1-r^k}.\]
It is worth pointing out that, in the third inequality of Eq. (\ref{te1.6}), we have used the fact that
\begin{equation}\label{ll1}
\Phi(t)=t+\lambda(1-t^{2})\leq \Phi(t_{0}),
\end{equation}
whenever
\begin{equation*}\label{ll2}
t=|F(\mu_m(z))|\leq t_{0}=\frac{r^{m}+x}{1+r^{m}x}
\quad \text{and} \quad
\lambda=\frac{r^{m}}{1-r^{2m}}.
\end{equation*}
Inequality \eqref{ll1} holds provided that $\frac{2r^{m}}{1-r^{2m}}\leq 1,$ which is satisfied for $0\leq r\leq R_m$, where $R_m$ is the unique positive root of the equation $r^{2m}+2r^{m}-1=0.$\par
Differentiating partially $D_1(x,r)$ with respect to $x$, we deduce that 
\[\frac{\partial D_1(x,r)}{\partial x}=\frac{(1-r^m)r^m}{(1+xr^m)^2}+\frac{r^{2k}(1+r^m+2xr^m)}{1-r^k}\geq 0,\]
which shows that $D_1(x,r)$ is increasing with respect to $x$ and so we deduce that 
\begin{eqnarray*} D_1(x,r)\leq D_1(1,r)=-(1-r^m)+\frac{2r^m}{1+r^m}+\frac{2r^{2k}(1+r^m)}{1-r^k}\leq 0,\end{eqnarray*}
for $r\leq R_{1,m,k}$, where $R_{1,m,k}$ is minimal positive root of (\ref{1.2}). We can see that 
\begin{eqnarray*}
\frac{d}{dr}D_1(1,r)&=&mr^{m-1}+\frac{2mr^{m-1}}{(1+r^m)^2}+\frac{2(2kr^{2k-1}(1+r^m)+mr^{2k+m-1})}{1-r^k}\\
&&+\frac{2kr^{3k-1}(1+r^m)}{(1-r^k)^2}\geq 0.
\end{eqnarray*}
and so $D_1(1,r)$ is increasing in $(0,1)$. Note that $D_1(1,0)=-1$ and $\lim_{r\rightarrow 1^-}D_1(1,r)=+\infty$. Therefore $R_{1,m,k}$ is unique  root of (\ref{1.2}) in $(0,1)$. \par 
We now claim that \( R_{1,m,k} < R_{1,m} \).
Suppose, on the contrary that \( R_{1,m,k} \geq R_{1,m} \).
Then for any \( r \geq R_{1,m} \) we have $-1+2r^{m}+r^{2m}\geq 0,$
and hence
\begin{align*}
-(1-r^m)+\frac{2r^m}{1+r^m}+\frac{2r^{2k}(1+r^m)}{1-r^k}
&=
\frac{-1+2r^{m}+r^{2m}}{1+r^{m}}
+\frac{2r^{2k}(1+r^{m})}{1-r^{k}}  >0.
\end{align*}
This contradicts \( R_{1,m,k} \) is a root of (\ref{1.2}).
Therefore $R_{1,m,k} < R_{1,m}.$\par

{\bf Sharpness of the Radius $R_{1,m,k}$:}
Next, we prove the sharpness for $R_{1,m,k}$. We consider the functions
 \begin{eqnarray}\label{fa}
f_a(z)
= \frac{a + T_v(z)}{1 + a\,T_v(z)}
= a + (1-a^2)\sum_{s=1}^{\infty}(-a)^{\,s-1}\bigl(T_v(z)\bigr)^s,
\qquad z \in B_X,\ a \in (0,1)
\end{eqnarray}
and $\mu_n(z)=T_v(z)^{n-1}z$, where $T_v\in T(v)$ and $v\in \partial B_X$ for $n\geq 1$.
Clearly 
\[ Df_a(z)(z) =\frac{(1-a^2)\,T_v(z)}{\bigl(1+a\,T_v(z)\bigr)^2}.\]

For \(z = r v\), where \(v \in \partial B_X\), we have $T_v(z) = r$ and so $\mu_n(z) = r^n v,  n \in \mathbb{N}$ and $P_s(\mu_n(z))=(1-a^2)(-a)^{s-1}r^{sn}$ for $s\geq 1$.\par
Therefore, by a simple computation it follows that
\begin{eqnarray*} &&\mathcal{A}(f_a(z)):=|f_a(\mu_m(r v))|+|D f_a(\mu_m(r v)) \mu_m(r v)|+\sum_{s=2}^{\infty}|P_{s}(\mu_k(r v))|\\
&&+\left(\frac{1}{1+|a|}+\frac{r^k}{1-r^k}\right)
\sum_{s=1}^{\infty} |P_{s}(\mu_k(r v))|^2\\

&=&1-\frac{(1-a)(1-r^m)}{1+ar^m}+\frac{(1-a^2)r^m}{\bigl(1+ar^m\bigr)^2}+(1-a^2) \frac{a\, r^{2k}}{1-a r^{k}}+\frac{(1-a^2)^2(1+|a|\,r^{k})}{(1+|a|)(1-r^{k})}\frac{r^{2k}}{1-a^{2} r^{2k}}\\
&=&1+(1-a) G_1(a,r),\end{eqnarray*}
where 
\[G_1(a,r)=-\frac{1-r^m}{1+ar^m}+\frac{(1+a)r^m}{\bigl(1+ar^m\bigr)^2}+(1+a)\frac{a\, r^{2k}}{1-a r^{k}}+\frac{(1-a^2)(1+a)(1+|a|\,r^{k})}{(1+|a|)(1-r^{k})}\frac{ r^{2k}}{1-a^{2} r^{2k}}.\]
Let $a\rightarrow 1$. It follows that 

\[\lim_{a\rightarrow 1}G_1(a,r)=\frac{1}{1+r^m}\left[-(1-r^m)+\frac{2r^m}{1+r^m}+\frac{2r^{2k}(1+r^m)}{1- r^{k}}\right]>0,\;\text{for}\;r>R_{1,m,k}.\]
Hence $\mathcal{A}(f_a(z))>1$ for $r>R_{1,m,k}$.

\[

= -\,\frac{1-r^{2m}-r^{m}}{(1+x r^{m})^{2}}

\]

Therefore, we obtain
\[

\end{equation}

\[\frac{\partial D_2(1,r)}{\partial r}=\frac{m r^{m-1}\left(3+2r^m-r^{2m}\right)}

\end{align*}

+\sum_{s=1}^{\infty}\|P_{2s}(\mu_{k}(rz_{0}))\|_{\infty}

&= \left(\frac{\beta+r^m}{1+\beta r^m}\right)^p+(1-\alpha^{2})\alpha r^{2k}+\sum_{s=2}^{\infty}\|P_{2s}(\mu_{k}(rz_{0}))\|_{\infty}\\
&\geq \left(\frac{\beta+r^m}{1+\beta r^m}\right)^p+(1-\alpha^{2})\alpha r^{2k}+\sum_{s=2}^{\infty}(1-\beta^{2})\beta^{2s-1}r^{2ks}\\

Let $z\in B_X\setminus\{0\}$ be fixed and define $z_0=\frac{z}{\|z\|_X}\in \partial B_X.$
We consider the holomorphic functions $f_j : \mathbb{U} \to \overline{\mathbb{U}}$ defined by
\[
f_j(\zeta) = F_j(\zeta z_0), \qquad \zeta \in \mathbb{U}
\]
where $F=(F_1,\ldots,F_n)$. Then 
\begin{equation}\label{eq:fexp}
f_j(\zeta)=a_j+\sum_{s=1}^{\infty}(P_s)_j(z_0)\zeta^s,
\qquad \zeta\in\mathbb{U}.
\end{equation}
Hence, for each fixed $j$ (see, e.g., \cite[p.~35]{GK2003}),
$|(P_s)_j(z_0)|\le 1-|a_j|^2=1-\|a\|_{\infty}^2.$
This yields
\begin{equation}\label{eq:Ps}
\|P_s(z_0)\|_{\infty}\le 1-\|a\|_{\infty}^2 .
\end{equation}

Let $x=\|a\|_{\infty}\in[0,1]$. If $x=1$, then $P_s(z_0)=0$ for all $s\ge1$, and
\eqref{2.1} holds trivially. Hence, we assume $x\in[0,1)$.
Now applying Lemma~\ref{L3}, we obtain
\begin{align}
\mathcal{A}_f(z):=&\|a\|_{\infty}^{p}
+\sum_{j=1}^{\infty}\|P_{qj}(\mu_k(rz_0))\|_{\infty}
+\left(\frac{1}{1+\|a\|_{\infty}}+\frac{r^{qk}}{1-r^{qk}}\right)
\sum_{j=1}^{\infty}\|P_{qj}(\mu_k(rz_0))\|_{\infty}^2\nonumber\\ 
&\le x^p+(1-x^2)\frac{r^{qk}}{1-r^{qk}} \nonumber\\
&=1+(1-x^2)D_4(x,r),
\label{eq:G1}
\end{align}
where
\[
D_4(x,r)
=\frac{r^{qk}}{1-r^{qk}}
-\frac{1-x^p}{1-x^2}.
\]

Suppose that $A(x)=\frac{1-x^p}{1-x^2}$. Since $A(x)$ is decreasing on $[0,1)$ for $p\in(0,2]$, we have
\[
A(x)\ge \lim_{x\to1^-}A(x)=\frac{p}{2}.
\]
Consequently,
\[
D_4(x,r)\le D_4(1,r)
=\frac{r^{qk}}{1-r^{qk}}
-\frac{p}{2}.
\]
Thus $\mathcal{A}_f(z)\le1$ whenever $D_4(1,r)\le0$, which holds for
$r\le R_{1,k,p,q}$, where $R_{1,k,p,q}$ is the smallest positive root in $(0,1)$ of
\begin{equation}\label{eq:R1}
\frac{r^{qk}}{1-r^{qk}}
-\frac{p}{2}=0.
\end{equation}
Let $\Phi(r)=\frac{r^{qk}}{1-r^{qk}}-\frac{p}{2}$. Clearly $\Phi'(r)>0$, this shows that $\Phi(r)$ is increasing in $(0,1)$. Note that 
\[\Phi(0)=-\frac{p}{2}\;\;\text{and}\;\;\lim_{r\rightarrow 1^-}\Phi(r)=+\infty.\]
Therefore $R_{1,k,p,q}$ is the unique  root in $(0,1)$ of $\Phi(r)=0$.

\medskip

{\bf Optimality of the Radius $R_{1,k,p,q}$:} 
We now show that the radius $R_{1,k,p,q}$ is sharp.
Fix $z_0\in\partial B_X$ and $u=(u_1,\ldots,u_n)\in\partial\mathbb{U}_n$ with
$|u_1|=\cdots=|u_n|=1$.
For $b\in(0,1)$, define
\begin{equation}\label{F1}
F_1(z)=f(l_{z_0}(z))u,\qquad z\in B_X,
\end{equation}
where
\[
f(\zeta)=\frac{b-\zeta}{1-b\zeta},\qquad \zeta\in\mathbb{U},
\]
and $l_{z_0}\in T(z_0)$.
Let $\mu_m(z)=-l_{z_0}(z)^{\,m-1}z$ for $m\geq 1$.
Then, for $r\in(0,1)$,
\[
F_1(rz_0)=\frac{b-r}{1-br}\,u
= a+\sum_{n=1}^{\infty}P_n(rz_0),
\]
where $a=bu$ and $P_n(z_0)=-b^{\,n-1}(1-b^2)u, n\ge 1.$
Consequently,
\[
P_n(\mu_k(rz_0))
=-r^{kn}b^{\,n-1}(1-b^2)u.
\]

Thus we have 
\begin{align*}
\mathcal{A}_f(z):=&\|a\|_{\infty}^{p}
+\sum_{j=1}^{\infty}\|P_{qj}(\mu_k(rz_0))\|_{\infty}
+\left(\frac{1}{1+\|a\|_{\infty}}+\frac{r^k}{1-r^k}\right)
\sum_{j=1}^{\infty}\|P_{qj}(\mu_k(rz_0))\|_{\infty}^2 \\
=&1+(1-b^2)G_4(b,r),
\end{align*}
where
\[
G_4(b,r)
=\frac{ b^{q-1} r^{qk}}{1-b^{q} r^{qk}}
+\left(\frac{1}{1+b}+\frac{r^{qk}}{1-r^{qk}}\right)
\frac{(1-b^2)}{b^{2}}\,
\frac{b^{2q} r^{2qk}}{1-b^{2q} r^{2qk}}
-\frac{1-b^p}{1-b^2}.
\]
Letting $b\to1^-$, we obtain
\[
\lim_{b\to1^-}G_4(b,r)
=\frac{r^{qk}}{1-r^{qk}}
-\frac{p}{2}>0,
\qquad r>R_{1,k,p,q}.
\]
Hence, for $r>R_{1,k,p,q}$ and $b$ sufficiently close to $1$, we have
$\mathcal{A}_f(z)>1$.
This proves that the radius $R_{1,k,p,q}$ is best possible.
