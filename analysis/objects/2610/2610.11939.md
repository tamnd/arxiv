<!-- lemma-1: body_md -->
\label{lemma: deltas}
		For $u_-,u_+\geqslant0$ with $u_-+u_+>0$ and $\beta\geqslant0$,
		\begin{align}
			\delta_-(u_-,u_+,\beta)
			&=
			\frac{W^{(\beta)}(u_+)}
			{W^{(\beta)}(u_-+u_+)},
			\label{eq: delta_min}\\
			\delta_+(u_-,u_+,\beta)
			&=
			Z^{(\beta)}(u_+)
			-
			Z^{(\beta)}(u_-+u_+)
			\frac{W^{(\beta)}(u_+)}
			{W^{(\beta)}(u_-+u_+)},
			\label{eq: delta_plus}
		\end{align}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
These identities follow from the standard two-sided exit identities for
		spectrally one-sided L\'evy processes; see, e.g.,
		\cite[Theorem 8.1(iii)]{KYP}. The result there is stated for spectrally negative
		L\'evy processes and arbitrary initial states, so applying it to $-Y$ and
		shifting the boundaries gives \eqref{eq: delta_min}--\eqref{eq: delta_plus}.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lemma: Levy exit probabilities}
		For $u_-, u_+\geqslant0$ and $\beta\geqslant0$,
		\begin{align}
			\mathbb{P}\bigl(\sigma(u_-)\leqslant T_\beta\bigr)
			&=
			e^{-\psi(\beta)u_-}, \label{eq: sigma before killing}\\
			\mathbb{P}\bigl(\tau(u_+)\leqslant T_\beta\bigr)
			&=
			Z^{(\beta)}(u_+)
			-
			\frac{\beta}{\psi(\beta)}W^{(\beta)}(u_+).
			\label{eq: tau before killing}
		\end{align}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
The first identity follows immediately from the Wiener--Hopf
		decomposition; see, e.g., \cite[Equation (8.2)]{KYP}. The second is the
		standard one-sided exit identity in \cite[Theorem 8.1(ii)]{KYP}. In both
		cases, we apply the corresponding results for spectrally negative
		L\'evy processes to $-Y$, with the appropriate adjustment of the
		boundaries.
<!-- end proof-2 -->

\label{lemma: eta}
		For $u_+\geqslant0$ and $\alpha,\beta\geqslant0$,
		\begin{equation}\label{eq: eta}
			\eta(u_+,\alpha,\beta)
			= 
			(\varphi(\alpha)-\beta)
			\int_{u_+}^\infty e^{-\alpha (y-u_+)}W^{(\beta)}(y)\,\mathrm{d}y
			+
			\frac{\varphi(\alpha)-\beta}
			{\psi(\beta)-\alpha}

		\end{equation}

<!-- lemma-4: body_md -->
\label{lemma: eta_tilde}
		For $u_-,u_+\geqslant0$ with $u_-+u_+>0$, $\alpha\geqslant0$ and
		$\beta>0$,
		\begin{equation}\label{eq: eta_tilde}
			\tilde\eta(u_-,u_+,\alpha,\beta)
			=
			\eta(u_+,\alpha,\beta)
			-
			\delta_-(u_-,u_+,\beta)
			\eta(u_-+u_+,\alpha,\beta).
		\end{equation}
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Decompose according to whether the lower boundary is reached before
		$u_+$:
		\begin{align*}
			\tilde\eta(u_-,u_+,\alpha,\beta)
			&=
			\eta(u_+,\alpha,\beta)-
			\mathbb{E}\left[
			e^{-\alpha(Y(\tau(u_+))-u_+)}
			\mathbf{1}_{\{\sigma(u_-)\leqslant\tau(u_+)\leqslant T_\beta\}}
			\right].
		\end{align*}
		On the event in the second term, $Y$ reaches $-u_-$ before $u_+$ and,
		by the strong Markov property, the process after this time is distributed
		as a fresh copy of $Y$ started from $-u_-$. Shifting the process by
		$u_-+u_+$, the remaining overshoot problem is therefore described by
		$\eta(u_-+u_+,\alpha,\beta)$. Hence
		\[
		\mathbb{E}\left[
		e^{-\alpha(Y(\tau(u_+))-u_+)}
		\mathbf{1}_{\{\sigma(u_-)\leqslant\tau(u_+)\leqslant T_\beta\}}
		\right]
		=
		\delta_-(u_-,u_+,\beta)
		\eta(u_-+u_+,\alpha,\beta),
		\]
		which proves \eqref{eq: eta_tilde}.

<!-- lemma-7: body_md -->
\label{lemma: ruin prob v=0}
		For $u>0$ and $\beta\geqslant0$
		\begin{equation}\label{eq: ruin prob v=0}
			p(u,0,\beta) = \frac{\int_0^{u} h_1(0, y, \beta)\,\delta_{2,+}(y, u-y, \beta)\,\mathrm{d}y + \int_{u}^\infty h_1(0, y, \beta)\,\mathrm{d}y}{1 - \int_0^{u} h_1(0, y, \beta)\,\delta_{2,-}(y, u-y,\beta)\,\mathrm{d}y}.
		\end{equation}
<!-- end lemma-7 -->
<!-- proof-6: body_md -->
We decompose according to the first event after the process crosses the
		threshold $v=0$: either the overshoot already exceeds $u$, the process
		subsequently reaches $u$ before returning to the threshold, or it returns
		to the threshold before reaching $u$. By the strong Markov property,
		\begin{align*}
			p(u,0,\beta)
			&=
			\int_u^\infty h_1(0,y,\beta)\,\mathrm{d}y
			+\int_0^u h_1(0,y,\beta)\,
			\delta_{2,+}(y,u-y,\beta)\,\mathrm{d}y \\
			&\quad
			+\int_0^u h_1(0,y,\beta)\,
			\delta_{2,-}(y,u-y,\beta)\,\mathrm{d}y
			\,p(u,0,\beta).
		\end{align*}
		The three terms correspond, respectively, to ruin occurring directly upon
		crossing the threshold, ruin occurring before returning to the threshold,
		and returning to the threshold before ruin. Since
		\[
		0\leqslant
		\int_0^u h_1(0,y,\beta)\,
		\delta_{2,-}(y,u-y,\beta)\,\mathrm{d}y
		<1,
		\]
		we can solve for $p(u,0,\beta)$, yielding \eqref{eq: ruin prob v=0}.
<!-- end proof-6 -->
<!-- lemma-8: body_md -->
\label{lemma: ruin prob v>0}
		For $u>v>0$ and $\beta\geqslant0$
		\begin{align}\label{eq: ruin prob v>0}
			\notag p(u,v,\beta) &= \int_{u-v}^\infty h_1(v,y,\beta) \,\mathrm{d}y + \int_{0}^{u-v} \delta_{2,+}(y, u-v-y, \beta) h_1(v,y,\beta) \,\mathrm{d}y\\
			&\:\:\:\:\:\:\:+ \int_{0}^{u-v} \delta_{2,-}(y, u-v-y, \beta) h_1(v,y,\beta) \,\mathrm{d}y \times p(u-v, 0, \beta)
		\end{align}
		with $p(u-v, 0, \beta)$ as given in \eqref{eq: ruin prob v=0}.
<!-- end lemma-8 -->
<!-- proof-7: body_md -->
Similar to the proof of Lemma~\ref{lemma: ruin prob v=0}, we decompose
		according to the first event after the process crosses the threshold $v$.
		If the overshoot is at least $u-v$, ruin occurs immediately. Otherwise,
		the process either reaches $u$ before returning to $v$, or returns to $v$
		before reaching $u$. In the latter case, the strong Markov property reduces
		the problem to the one with threshold $0$ and initial surplus $u-v$.
		This yields the three terms in \eqref{eq: ruin prob v>0}.
<!-- end proof-7 -->
<!-- lemma-9: body_md -->
\label{lemma: ruin prob v<0}
		For $u \geqslant 0 > v$ and $\beta \geqslant 0$
		\begin{equation}\label{eq: ruin prob v<0}
			p(u,v,\beta) = \delta_{2,+}(-v,u,\beta) + \delta_{2,-}(-v, u, \beta) \times p(u-v,0,\beta)
		\end{equation}
		with $p(u-v,0,\beta)$ as given in \eqref{eq: ruin prob v=0}.
<!-- end lemma-9 -->
<!-- proof-8: body_md -->
For this case we only need to consider two scenarios: (i) $Y_v$ crosses the level $u$ before it reaches the threshold $v$ and before killing, and (ii) $Y_v$ reaches the threshold $v$ before killing and before crossing the level $u$, at which point, by the strong Markov property, we are in a scenario which is initialized at the threshold. These two scenarios correspond to the first and second term on the right-hand side of \eqref{eq: ruin prob v<0}.

\label{rem: all-time ruin prob}
		{\em When $\mathbb{E}(Y_1(1)) > 0$ the all-time ruin probability is always equal to one. Indeed, since $\mathbb{P}(\tau_1(0) < \infty) = 1$ in this case, we have 
			\begin{align*}
				\int_0^{u} &h_1(0, y, 0)\,\delta_{2,+}(y, u-y, 0)\,\mathrm{d}y + \int_{u}^\infty h_1(0, y, 0)\,\mathrm{d}y \\
				=&\: \int_0^{u} h_1(0, y, 0)\,\left(1-\delta_{2,-}(y, u-y, 0)\right)\,\mathrm{d}y + \int_{u}^\infty h_1(0, y, 0)\,\mathrm{d}y \\
				=&\: 1 - \int_0^{u} h_1(0, y, 0)\,\delta_{2,-}(y, u-y, 0)\,\mathrm{d}y,
			\end{align*}

			p(\alpha) = \prod_{i=1}^{n^*}(\alpha - \pi^*_i)^{d^*_i},
		\end{equation*}
		from which it follows that, for $i=1,\dots,n^*$,
		\begin{equation}\label{eq: all deriv p zer0}
			p(\pi^*_i) = p^{(1)}(\pi_i^*) = \dots =
			p^{(d_i-1)}(\pi^*_i) = 0.
		\end{equation}
		Hence, for each $i=1,\dots,n^*$, the first $d_i^*-1$ derivatives of
		$p(\alpha)/\tilde{q}(\alpha)$ also vanish at $\alpha=\pi_i^*$. Using
		the partial fraction decomposition in
		\eqref{eq: partial frac decomp p / tilde q}, we obtain, for
		$\ell=0,\dots,d_i^*-1$,
		\begin{equation}\label{eq: all deriv p / tilde q zero}
			0 =
			\frac{\partial^\ell}{\partial \alpha^\ell}
			\frac{p(\alpha)}{\tilde{q}(\alpha)}
			\Bigg|_{\alpha = \pi^*_i}
			= \sum_{k=1}^{d+1}
			(-1)^\ell \ell!
			\frac{c_k}{(\pi^*_i - \alpha_k)^{\ell+1}}.
		\end{equation}
		Finally, using \eqref{eq: partial frac decomp B_j(alpha)}, we have
		\begin{align*}
			\sum_{k=1}^{d+1}c_k \mathscr{B}_j(\alpha_k)
			&= \sum_{k=1}^{d+1}c_k
			\sum_{i=1}^{n^*}\sum_{\ell=1}^{d^*_i}
			\frac{c^*_{i\ell j}}{(\alpha_k - \pi^*_i)^\ell} \\
			&= \sum_{i=1}^{n^*}\sum_{\ell=1}^{d^*_i}
			\frac{c^*_{i\ell j}}{(-1)^{\ell-1}(\ell-1)!}
			\sum_{k=1}^{d+1}
			(-1)^{\ell-1}(\ell-1)!
			\frac{c_k}{(\alpha_k - \pi^*_i)^\ell}
			= 0,
		\end{align*}
		by \eqref{eq: all deriv p / tilde q zero}. Thus, for all $j\in[d]$,
		\begin{equation*}
			\sum_{k=1}^{d}c_k\mathscr{B}_j(\alpha_k)
			= -c_{d+1}\mathscr{B}_j(\alpha_{d+1}),
		\end{equation*}
		which is equivalent to \eqref{eq: relation B vs B_j}.

<!-- lemma-18: body_md -->
\label{lemma: relation M B_inv}
		For $u_-,u_+\geqslant0$ with $u_-+u_+ >0$, and $\beta > 0$,
		\begin{equation}\label{eq: MB_inv}
			\boldsymbol{M}\boldsymbol{B}^{-1}
			\left(\boldsymbol{\theta}(u_+)
			- \delta_-(u_-,u_+,\beta)\boldsymbol{\theta}(u_-+u_+)\right)
			= e^{\alpha_{d+1}(u_-+u_+)}\boldsymbol{\theta}(u_+)
			- e^{\alpha_{d+1}u_+}\boldsymbol{\theta}(u_-+u_+).
		\end{equation}
<!-- end lemma-18 -->
<!-- proof-18: body_md -->
Starting with
		$\boldsymbol{M}\boldsymbol{B}^{-1}
		=e^{\alpha_{d+1}(u_-+u_+)}\boldsymbol{I}
		-\tilde{\boldsymbol{B}}\boldsymbol{B}^{-1}$,
		we obtain for the left-hand side of \eqref{eq: MB_inv}
		\begin{align}
			\notag e^{\alpha_{d+1}(u_-+u_+)}
			&\boldsymbol{\theta}(u_+)
			-e^{\alpha_{d+1}(u_-+u_+)}
			\delta_-(u_-,u_+,\beta)\boldsymbol{\theta}(u_-+u_+) \\
			&-\tilde{\boldsymbol{B}}\boldsymbol{B}^{-1}\boldsymbol{\theta}(u_+)
			+\delta_-(u_-,u_+,\beta)
			\tilde{\boldsymbol{B}}\boldsymbol{B}^{-1}\boldsymbol{\theta}(u_-+u_+).
			\label{eq: temp MB_inv}
		\end{align}
		Furthermore, by Lemma~\ref{lemma: relation B vs B_j},
		\begin{equation*}
			\tilde{\boldsymbol{B}}\boldsymbol{B}^{-1}
			=\boldsymbol{\theta}(u_-+u_+)
			\left(\boldsymbol{\mathscr{B}}(\alpha_{d+1})^\top\boldsymbol{B}^{-1}\right)
			=-\frac{1}{c_{d+1}}
			\boldsymbol{\theta}(u_-+u_+)\boldsymbol{c}^\top.
		\end{equation*}
		Hence \eqref{eq: temp MB_inv} can be written as
		\begin{align}
			\notag e^{\alpha_{d+1}(u_-+u_+)}
			&\boldsymbol{\theta}(u_+)
			-e^{\alpha_{d+1}(u_-+u_+)}
			\delta_-(u_-,u_+,\beta)\boldsymbol{\theta}(u_-+u_+) \\
			&\:\:\:\:\:
			+\frac{1}{c_{d+1}}
			\left(\boldsymbol{c}^\top\boldsymbol{\theta}(u_+)\right)
			\boldsymbol{\theta}(u_-+u_+)
			-\frac{1}{c_{d+1}}
			\delta_-(u_-,u_+,\beta)
			\left(\boldsymbol{c}^\top\boldsymbol{\theta}(u_-+u_+)\right)
			\boldsymbol{\theta}(u_-+u_+).
		\end{align}
		This can be rewritten as
		\begin{align*}
			&e^{\alpha_{d+1}(u_-+u_+)}\boldsymbol{\theta}(u_+)
			-\frac{1}{c_{d+1}}
			\left(
			c_{d+1}e^{\alpha_{d+1}(u_-+u_+)}
			+\boldsymbol{c}^\top\boldsymbol{\theta}(u_-+u_+)
			\right)
			\delta_-(u_-,u_+,\beta)\boldsymbol{\theta}(u_-+u_+)\\
			&\quad
			+\frac{1}{c_{d+1}}
			\left(\boldsymbol{c}^\top\boldsymbol{\theta}(u_+)\right)
			\boldsymbol{\theta}(u_-+u_+).
		\end{align*}
		With the expression for $\delta_-(u_-,u_+,\beta)$ in
		\eqref{eq: delta_min scale func}, this simplifies to
		\begin{align*}
			&e^{\alpha_{d+1}(u_-+u_+)}\boldsymbol{\theta}(u_+)
			-\frac{1}{c_{d+1}}
			\left(\sum_{k=1}^{d+1}c_ke^{\alpha_k u_+}\right)
			\boldsymbol{\theta}(u_-+u_+)
			+\frac{1}{c_{d+1}}
			\left(\boldsymbol{c}^\top\boldsymbol{\theta}(u_+)\right)
			\boldsymbol{\theta}(u_-+u_+),
		\end{align*}
		which equals the right-hand side of \eqref{eq: MB_inv}.

<!-- lemma-19: body_md -->
\label{lemma: proof LSTs same}
		For $v>0$,
		\begin{equation}\label{eq: proof LSTs same}
			\int_0^v e^{-\alpha x}\mathrm{d} W^{(0)}_1(x) = \frac{\lambda_1}{-r_1} \int_0^v e^{-\alpha y} W^{(0)}_1(y) \left(1- \int_0^{v-y} e^{-\alpha x} \mathbb{P}(B_1\in\mathrm{d}x)\right)\,\mathrm{d}y.
		\end{equation}
<!-- end lemma-19 -->
<!-- proof-19: body_md -->
We show the equality by proving that the Laplace transforms with respect to $v$ of both sides are equal. Multiplying the left-hand side of \eqref{eq: proof LSTs same} by $e^{-\omega v}$, integrating $v$ over $(0,\infty)$ and changing the order of integration yields 
		\begin{align*}
			\int_0^\infty e^{-\alpha x} \left( \int_x^\infty e^{-\omega v}\,\mathrm{d}v\right)\mathrm{d}W^{(0)}_1(x) = \frac{1}{\omega}\int_0^\infty e^{-(\alpha+\omega) x} \mathrm{d} W^{(0)}_1(x).
		\end{align*}
		Integrating by parts then gives
		\begin{equation}\label{eq: proof LSTs same step 1}
			\frac{1}{\omega}\left(-W^{(0)}_1(0) + \frac{\alpha+\omega}{\varphi_1(\alpha+\omega)}\right).
		\end{equation}
		Multiplying the right-hand side of \eqref{eq: proof LSTs same} by $e^{-\omega v}$, integrating $v$ over $(0,\infty)$ and changing the order of integration yields for the first term
		\begin{equation}\label{eq: proof LSTs same step 2a}
			\frac{\lambda_1}{-r_1}\int_0^\infty e^{-\alpha y} \left( \int_y^\infty e^{-\omega v}\,\mathrm{d}v\right) W^{(0)}_1(y) \,\mathrm{d}y =\frac{\lambda_1}{-r_1\omega} \frac{1}{\varphi_1(\alpha + \omega)}.
		\end{equation}
		For the second, double integral, after applying the change in the order of integration $(\mathrm{d}x, \mathrm{d}y, \mathrm{d}v) \to (\mathrm{d}x, \mathrm{d}v, \mathrm{d}y)$, substituting $z:= v-y$, and then changing the order of integration to $(\mathrm{d}x, \mathrm{d}z, \mathrm{d}y) \to (\mathrm{d}z, \mathrm{d}x, \mathrm{d}y)$, we obtain
		\begin{align*}
			\frac{\lambda_1}{-r_1}\int_0^\infty e^{-(\alpha+\omega)y}W^{(0)}_1(y) \int_0^\infty e^{-\alpha x} \int_x^\infty e^{-\omega z}\,\mathrm{d}z\,\mathbb{P}(B_1\in\mathrm{d}x)\,\mathrm{d}y,
		\end{align*}
		which can be simplified to
		\begin{equation}\label{eq: proof LSTs same step 2b}
			\frac{\lambda_1}{-r_1\omega}\frac{\mathscr{B}_1(\alpha+\omega)}{\varphi_1(\alpha+\omega)}.
		\end{equation}
		Taking the difference of \eqref{eq: proof LSTs same step 2a} and \eqref{eq: proof LSTs same step 2b} gives for the Laplace transform of the right-hand side of \eqref{eq: proof LSTs same}
		\begin{align*}
			\frac{\lambda_1}{-r_1\omega} \frac{1-\mathscr{B}_1(\alpha+\omega)}{\varphi_1(\alpha+\omega)}.
		\end{align*}
		Using \eqref{eq: def laplace exp}, this can be rewritten to
		\begin{equation*}
			\frac{1}{\omega}\left(\frac{1}{r_1} + \frac{\alpha+\omega}{\varphi_1(\alpha+\omega)}\right).
		\end{equation*}
		Since $W_1^{(0)}(0) = -r_1^{-1}$ for a compound Poisson process with drift, this is equal to \eqref{eq: proof LSTs same step 1}, thus showing the equivalence of \eqref{eq: proof LSTs same}.

\label{coro: renewal equation}
		For $x\in[0,\infty)$,
		\begin{equation}\label{eq: renewal equation}
			W_1^{(0)}(x) = -\frac{1}{r_1} - \frac{\lambda_1}{r_1}\int_0^x W_1^{(0)}(y)\mathbb{P}(B_1>x-y)\,\mathrm{d}y.
		\end{equation}
