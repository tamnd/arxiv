<!-- definition-1: body_md -->
\label{D:Kminus}
    For $K=(k_0,k_1,k_2)$ with $k_0+k_2>0$, define
    \[
        K^{-} = \begin{cases}
            (k_0,\,k_1+1,\,k_2-1) & \text{if } k_2>0,          \\
            (k_0-1,\,k_1+1,\,0)   & \text{if } k_2=0,\, k_0>0.
        \end{cases}
    \]
    In both cases $K^{-}$ is again at level $k$, and the initial conditions for
    $K^{-}$ have the same shape as the initial conditions for $K$, with the right
    sides of \hyperref[IC2]{IC2} and \hyperref[IC3]{IC3} reduced from $k_0+k_2$ to
    $k_0+k_2-1$. When $k_2=0$, the right side of \hyperref[IC1]{IC1} is also
    reduced from $k_0$ to $k_0-1$.

<!-- proposition-1: body_md -->
\label{P:S(K)}
    Assume $k_0+k_2>0$. The set of degree $-1$ triples
    $(a_1,b_1,c_1)$ of partitions in
    $\Pi_K\setminus\Pi_{K^{-}}$ is
    \[
        \mathcal{S}(K) = \left\{(a_1,b_1,c_1)\in\mathbb{Z}_{\geq 0}^3 \,\middle|\,
        \begin{gathered}
            a_1\leq k_0,\,
            b_1+a_1\leq k_0+k_2,\,
            c_1+b_1\leq k_0+k_2, \\
            b_1+a_1=k_0+k_2\ \textup{or}\ c_1+b_1=k_0+k_2
        \end{gathered}
        \right\}.
    \]

<!-- proof-1: body_md -->
The difference conditions are the same for $K$ and $K^{-}$. Hence a partition
    in $\Pi_K$ fails to lie in $\Pi_{K^{-}}$ precisely when it violates at least
    one initial condition for $K^{-}$. We analyze the two cases in the definition
    of $K^{-}$ separately.

    \textit{Case $k_2>0$:} The initial conditions for $K^{-}=(k_0,k_1+1,k_2-1)$ are
    \begin{align*}
        a_1     & \leq k_0,       \\
        b_1+a_1 & \leq k_0+k_2-1, \\
        c_1+b_1 & \leq k_0+k_2-1.
    \end{align*}
    Since \hyperref[IC1]{IC1} is unchanged, violation requires
    \[
        b_1+a_1\geq k_0+k_2
        \quad\text{or}\quad
        c_1+b_1\geq k_0+k_2.
    \]
    Together with the initial conditions for $K$, this implies $b_1+a_1=k_0+k_2$ or
    $c_1+b_1=k_0+k_2$.

    \textit{Case $k_2=0$:} The initial conditions for $K^{-}=(k_0-1,k_1+1,0)$ are
    \begin{align*}
        a_1     & \leq k_0-1, \\
        b_1+a_1 & \leq k_0-1, \\
        c_1+b_1 & \leq k_0-1.
    \end{align*}
    If $a_1\geq k_0$, then $b_1+a_1\geq k_0$. Hence violation is
    equivalent to
    \[
        b_1+a_1\geq k_0
        \quad\text{or}\quad
        c_1+b_1\geq k_0.
    \]
    Together with the initial conditions for $K$, this implies
    \[
        b_1+a_1=k_0
        \quad\text{or}\quad
        c_1+b_1=k_0.
    \]
    Since $k_2=0$, this is equivalent to $b_1+a_1=k_0+k_2$ or $c_1+b_1=k_0+k_2$.

    Conversely, every triple satisfying the defining inequalities of
    $\mathcal{S}(K)$ is the degree $-1$ triple of some partition in
    $\Pi_K\setminus\Pi_{K^{-}}$. Namely, set $a_j=b_j=c_j=0$ for all $j\geq2$. The
    resulting partition satisfies \eqref{DC} at every $j$, satisfies the initial
    conditions for $K$ by assumption, and violates an initial condition for $K^{-}$
    by the equality $b_1+a_1=k_0+k_2$ or $c_1+b_1=k_0+k_2$.

<!-- lemma-1: body_md -->
\label{L:deletion bijection}
    Fix $(a_1,b_1,c_1)\in\mathcal{S}(K)$. For a partition
    $\pi=(a_j,b_j,c_j)_{j\geq1}$ with degree $-1$ triple
    $(a_1,b_1,c_1)$, define $\pi'=(a'_j,b'_j,c'_j)_{j\geq1}$ by
    \[
        \begin{aligned}
            a'_j & = a_{j+1}, \\
            b'_j & = b_{j+1}, \\
            c'_j & = c_{j+1}.
        \end{aligned}
    \]
    Then $\pi\mapsto\pi'$ is a bijection from the set of all
    $\pi\in\Pi_K\setminus\Pi_{K^{-}}$ with degree $-1$ triple $(a_1,b_1,c_1)$ onto
    $\Pi_{(k_1,c_1,k_0+k_2-c_1)}$.

    If
    \[
        d=\deg(\pi'),\qquad
        n_1=|\pi'|_2,\quad n_2=|\pi'|_0,\quad
        n_3=|\pi'|_{\underline{2}},
    \]
    then
    \[
        \deg(\pi)=(a_1+b_1+c_1)+d+n_1+n_2+n_3.
    \]

<!-- proof-3: body_md -->
We verify that the map $\pi\mapsto\pi'$ is well-defined, injective, and
    surjective, and then compute the degree.

    \textit{Well-defined:} By Proposition~\ref{P:S(K)}, the fixed triple
    $(a_1,b_1,c_1)$ satisfies the initial conditions for $K$ and fails the initial
    conditions for $K^{-}$. The initial conditions for
    $\Pi_{(k_1,c_1,k_0+k_2-c_1)}$ are
    \[
        \begin{aligned}
            a'_1      & \leq k_1,   \\
            b'_1+a'_1 & \leq k-c_1, \\
            c'_1+b'_1 & \leq k-c_1.
        \end{aligned}
    \]
    Since $a'_1=a_2$, $b'_1=b_2$, and $c'_1=c_2$, these are exactly the conditions
    obtained in Proposition~\ref{P:deg2 constraints}. For the conditions \eqref{DC}
    with $j\geq2$, reindexing by $i=j-1$ gives the difference conditions for $\pi'$
    with $i\geq1$.

    \textit{Injective:} The degree $-1$ triple $(a_1,b_1,c_1)$ is fixed, so
    distinct partitions with degree $-1$ triple $(a_1,b_1,c_1)$ have distinct
    images.

    \textit{Surjective:} Given $\pi'\in\Pi_{(k_1,c_1,k_0+k_2-c_1)}$, write
    $\pi'=(a'_j,b'_j,c'_j)_{j\geq 1}$. Let $\pi$ be the partition whose degree $-1$
    triple is $(a_1,b_1,c_1)$ and whose remaining entries are
    \[
        (a_j,b_j,c_j)=(a'_{j-1},b'_{j-1},c'_{j-1})\quad\text{for } j\geq 2.
    \]
    The initial conditions for $\pi'$ are exactly the inequalities obtained from
    \eqref{DC} at $j=1$ by the calculation in Proposition~\ref{P:deg2 constraints}.
    Therefore, the difference conditions for $\pi$ hold at $j=1$. For $j\geq2$, the
    difference conditions for $\pi$ follow from the difference conditions for
    $\pi'$ by reindexing. Because $(a_1,b_1,c_1)\in\mathcal{S}(K)$, the initial
    conditions for $K$ hold and the initial conditions for $K^{-}$ fail. Hence the
    constructed partition lies in $\Pi_K\setminus\Pi_{K^{-}}$ and has image $\pi'$.

    \textit{Degree statement:} The partition $\pi$ has degree
    \begin{align*}
        \sum_{j\geq 1}j(a_j+b_j+c_j)
         & = (a_1+b_1+c_1) + \sum_{j\geq 2}j(a'_{j-1}+b'_{j-1}+c'_{j-1})                      \\
         & = (a_1+b_1+c_1) + \sum_{i\geq 1}(i+1)(a'_i+b'_i+c'_i)                              \\
         & = (a_1+b_1+c_1) + \sum_{i\geq 1}i(a'_i+b'_i+c'_i) + \sum_{i\geq 1}(a'_i+b'_i+c'_i) \\
         & = (a_1+b_1+c_1) + d + (n_1+n_2+n_3).
    \end{align*}

<!-- theorem-1: body_md -->
\label{T:main}
    Let $K=(k_0,k_1,k_2)$ be a level $k$ weight.

    If $k_0+k_2>0$, let $K^{-}$ and $\mathcal{S}(K)$ be as in
    Definition~\ref{D:Kminus} and Proposition~\ref{P:S(K)}. Then
    \begin{equation}\label{E:main}
        \chi_K(\mathbf{z},q)-\chi_{K^{-}}(\mathbf{z},q)
        = \sum_{i=0}^{k_0+k_2}\mathcal{P}_i(\mathbf{z},q)\,
        \chi_{(k_1,i,k_0+k_2-i)}(\mathbf{z}q,q),
    \end{equation}
    where $\mathbf{z}q=(z_1q,z_2q,z_3q)$ and
    \begin{equation}\label{E:Pm}
        \mathcal{P}_i(\mathbf{z},q) =
        \sum_{\substack{(a_1,b_1,c_1)\in\mathcal{S}(K)\\c_1=i}}
        z_1^{a_1}\,z_2^{b_1}\,z_3^{c_1}\,q^{a_1+b_1+c_1}.
    \end{equation}

    If $k_0+k_2=0$, then $K=(0,k,0)$ and
    \begin{equation}\label{E:base}
        \chi_{(0,k,0)}(\mathbf{z},q)
        =
        \chi_{(k,0,0)}(\mathbf{z}q,q).
    \end{equation}
<!-- end theorem-1 -->
<!-- proof-4: body_md -->
Assume first that $k_0+k_2>0$. By \eqref{E:LHS gen fun}, the left side of
    \eqref{E:main} is the generating function of $\Pi_K\setminus\Pi_{K^{-}}$. By
    Lemma~\ref{L:deletion bijection}, the set $\Pi_K\setminus\Pi_{K^{-}}$ is
    partitioned by the degree $-1$ triple $(a_1,b_1,c_1)\in\mathcal{S}(K)$, and for
    each fixed triple the deletion bijection maps the subset with degree $-1$
    triple $(a_1,b_1,c_1)$ onto $\Pi_{(k_1,c_1,k_0+k_2-c_1)}$.

    For a partition with degree $-1$ triple $(a_1,b_1,c_1)$ whose image $\pi'$
    satisfies $|\pi'|_2=n_1$, $|\pi'|_0=n_2$, $|\pi'|_{\underline{2}}=n_3$, and
    $\deg(\pi')=d$, the contribution to the generating function is
    \begin{align*}
        z_1^{a_1+n_1}\,z_2^{b_1+n_2}\,z_3^{c_1+n_3}\,
        q^{(a_1+b_1+c_1)+(d+n_1+n_2+n_3)}
         & = z_1^{a_1} z_2^{b_1} z_3^{c_1} q^{a_1+b_1+c_1}             \\
         & \quad\cdot z_1^{n_1} z_2^{n_2} z_3^{n_3} q^{d+n_1+n_2+n_3}.
    \end{align*}
    Summing over all images gives
    \[
        z_1^{a_1} z_2^{b_1} z_3^{c_1} q^{a_1+b_1+c_1}
        \cdot \chi_{(k_1,c_1,k_0+k_2-c_1)}(\mathbf{z}q,q).
    \]
    Summing over all $(a_1,b_1,c_1)\in\mathcal{S}(K)$ and grouping by $c_1=i$
    yields \eqref{E:main}.

    It remains to consider the case $k_0+k_2=0$. Then $K=(0,k,0)$, and the initial
    conditions
    \begin{align*}
        a_1     & \leq 0, \\
        b_1+a_1 & \leq 0, \\
        c_1+b_1 & \leq 0
    \end{align*}
    force $a_1=b_1=c_1=0$. Evaluating \eqref{DC} at $j=1$ gives
    \begin{align*}
        a_2     & \leq k, \\
        b_2+a_2 & \leq k, \\
        c_2+b_2 & \leq k,
    \end{align*}
    which are the initial conditions for $\Pi_{(k,0,0)}$. Thus deleting
    the zero degree $-1$ triple and reindexing gives a bijection from
    $\Pi_{(0,k,0)}$ to $\Pi_{(k,0,0)}$. After the zero degree $-1$
    triple is deleted and the remaining entries are reindexed, each
    remaining factor contributes one extra power of $q$, and hence
    \[
        \chi_{(0,k,0)}(\mathbf z,q)=\chi_{(k,0,0)}(\mathbf zq,q),
    \]
    which is \eqref{E:base}.

<!-- corollary-1: body_md -->
\label{C:Pm explicit}
    For $i=0,\ldots,k_0+k_2$,
    \begin{equation}\label{E:Pm explicit}
        \mathcal{P}_i(\mathbf{z},q) = z_3^i\bigg[
        q^{k_0+k_2+i}\!\sum_{a_1=i}^{k_0} z_2^{k_0+k_2-a_1}\,z_1^{a_1}
        \,+\, z_2^{k_0+k_2-i}q^{k_0+k_2}\!\sum_{a_1=0}^{\min(i-1,k_0)}(z_1q)^{a_1}
        \bigg],

<!-- theorem-2: body_md -->
\label{T:uniqueness}
    The system \eqref{E:main}--\eqref{E:base}, together with the condition that the constant term of each series $\chi_K(\mathbf z,q)$ is equal to $1$, uniquely determines $\chi_K(\mathbf z,q)$ for all $K$ at level~$k$.

<!-- proof-6: body_md -->
Write
    \[
        \chi_K(\mathbf z,q)=\sum_{d\geq 0}f_K^{(d)}(\mathbf z)q^d.
    \]
    The condition on the constant terms gives $f_K^{(0)}(\mathbf z)=1$ for all
    weights $K$ at level~$k$. We prove by induction on $d$ that $f_K^{(d)}$ is
    uniquely determined for all such weights $K$.

    The recurrence \eqref{E:main} expresses $\chi_K-\chi_{K^{-}}$ in terms of
    products $\mathcal{P}_i(\mathbf{z},q)\cdot\chi_{K'}(\mathbf{z}q,q)$ for various
    $K'$. Since $k_0+k_2>0$, every triple in $\mathcal{S}(K)$ has
    $a_1+b_1+c_1\geq1$. Hence, by \eqref{E:Pm}, each $\mathcal{P}_i(\mathbf z,q)$
    has minimum $q$-degree at least $1$. Under the substitution
    $\mathbf{z}\mapsto\mathbf{z}q$, a monomial $\mathbf{z}^{\mathbf{n}}q^{d'}$ in
    $\chi_{K'}$ becomes $\mathbf{z}^{\mathbf{n}}q^{d'+|\mathbf{n}|}$, where
    $|\mathbf{n}|=n_1+n_2+n_3\geq 0$. Hence the coefficient of $q^d$ in the product
    $\mathcal{P}_i\cdot\chi_{K'}(\mathbf{z}q,q)$ depends only on coefficients
    $f_{K'}^{(d')}$ with $d'<d$.

    Fix $d\geq1$ and assume that all coefficients of degree $<d$ are known. We
    first determine $f_{(0,k,0)}^{(d)}$. By \eqref{E:base},
    \[
        \chi_{(0,k,0)}(\mathbf z,q)=\chi_{(k,0,0)}(\mathbf zq,q).
    \]
    The coefficient of $q^d$ on the right depends only on coefficients
    $f_{(k,0,0)}^{(d')}$ with $d'<d$: the empty monomial contributes only to degree
    $0$, and every non-empty monomial gains a positive power of $q$ under the
    substitution $\mathbf z\mapsto\mathbf zq$. Hence $f_{(0,k,0)}^{(d)}$ is
    determined.

    Now let $K$ be any level $k$ weight with $k_0+k_2>0$. Since each step $K\mapsto
        K^{-}$ decreases $k_0+k_2$ by one, iterating $K\mapsto K^{-}$ reaches $(0,k,0)$
    after finitely many steps. Along this chain, \eqref{E:main} determines
    \[
        f_K^{(d)}-f_{K^{-}}^{(d)}
    \]
    from coefficients of degree $<d$, which are already known. Starting from the
    already determined coefficient for $(0,k,0)$, we determine $f_K^{(d)}$
    successively along the chain. Thus all coefficients $f_K^{(d)}$ are uniquely
    determined.

\label{T:level-one-character-formulas}
    Define
    \[
        F(\mathbf z,q)
        =
        \sum_{n_1,n_2,n_3\geq0}
        \frac{q^{n_1^2+n_3^2+\binom{n_2+1}{2}+n_1n_2+n_2n_3}}

            \chi_{0,0,1}
             & =

By Theorem~\ref{T:uniqueness}, it is enough to show that the three series in
    \eqref{E:level-one-chi} satisfy \eqref{E:lvl1 r1}--\eqref{E:lvl1 r3} and have
    constant term $1$. The first two series have constant term $1$. For the third
    series, the first summand has constant term $1$, while the second summand is
    divisible by $q$. Relation \eqref{E:lvl1 r3} follows directly from
    \eqref{E:level-one-chi}.

    Let $f_{n_1,n_2,n_3}$ denote the coefficient of $z_1^{n_1}z_2^{n_2}z_3^{n_3}$
    in $F(\mathbf z,q)$ for $n_1,n_2,n_3\geq0$. We set $f_{n_1,n_2,n_3}=0$ whenever
    $n_1<0$, $n_2<0$, or $n_3<0$. Thus
    \[
        f_{n_1,n_2,n_3}
        =
        \frac{q^{n_1^2+n_3^2+\binom{n_2+1}{2}+n_1n_2+n_2n_3}}
        {(q)_{n_2}\,(q)_{n_1+n_3}}
        \binom{n_1+n_3}{n_1}_{q^2}.
    \]
    Comparing the powers of $q$ and the factors $(q)_{n_2}$ gives, for $0\leq m\leq
        n_2$,
    \begin{equation}\label{E:f-ratio}
        f_{n_1,n_2-m,n_3}
        =
        q^{\binom{m}{2}-m(n_1+n_2+n_3)}
        \frac{(q)_{n_2}}{(q)_{n_2-m}}\,
        f_{n_1,n_2,n_3}.
    \end{equation}

    \emph{Relation \eqref{E:lvl1 r2}.} First we prove
    \begin{equation}\label{E:level-one-chi-sum}
        \chi_{0,0,1}
        =
        \sum_{m\geq0}
        q^{\binom{m+1}{2}}z_2^m
        \left(
        F(\mathbf zq^{m+1},q)
        +q^{m+1}z_3\,F(\mathbf zq^{m+2},q)
        \right).
    \end{equation}
    The contribution of the terms containing $F(\mathbf zq^{m+1},q)$ to
    the coefficient of $z_1^{n_1}z_2^{n_2}z_3^{n_3}$ on the right side of
    \eqref{E:level-one-chi-sum} equals
    \begin{align*}
         &
        \sum_{m=0}^{n_2}
        q^{\binom{m+1}{2}+(m+1)(n_1+n_2-m+n_3)}
        f_{n_1,n_2-m,n_3}
        \\
         & =
        \sum_{m=0}^{n_2}
        q^{n_1+n_2+n_3-m}
        \frac{(q)_{n_2}}{(q)_{n_2-m}}\,
        f_{n_1,n_2,n_3}
        \\
         & =
        q^{n_1+n_3}\,(q)_{n_2}\,f_{n_1,n_2,n_3}
        \sum_{m=0}^{n_2}
        \frac{q^{n_2-m}}{(q)_{n_2-m}}
        \\
         & =
        q^{n_1+n_3}\,f_{n_1,n_2,n_3}.
    \end{align*}
    The second equality follows from \eqref{E:f-ratio}, and the last one
    from Euler's identity
    \[
        \sum_{m=0}^{n}\frac{q^m}{(q)_m}
        =
        \frac{1}{(q)_n}.
    \]
    This equals the coefficient of $z_1^{n_1}z_2^{n_2}z_3^{n_3}$ in
    \[
        F\bigl((z_1q,\,z_2,\,z_3q),\,q\bigr).
    \]

    For $n_3>0$, the contribution of the terms containing $q^{m+1}z_3F(\mathbf
        zq^{m+2},q)$ to the coefficient of $z_1^{n_1}z_2^{n_2}z_3^{n_3}$ on the right

        \sum_{m=0}^{n_2}
        q^{

                +(m+2)(n_1+n_2-m+n_3-1)

        f_{n_1,n_2-m,n_3-1}
        \\
         & =
        \sum_{m=0}^{n_2}
        q^{2n_1+2n_2+2n_3-1-m}
        \frac{(q)_{n_2}}{(q)_{n_2-m}}\,
        f_{n_1,n_2,n_3-1}
        \\
         & =
        q^{2n_1+n_2+2n_3-1}
        (q)_{n_2}\,f_{n_1,n_2,n_3-1}
        \sum_{m=0}^{n_2}
        \frac{q^{n_2-m}}{(q)_{n_2-m}}
