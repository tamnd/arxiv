<!-- proof-1: body_md -->
Let $1\le p,q < \infty$ (The cases $p=\infty$ or $q=\infty$ are treated analogously using the usual modifications) and $\max\{1/p,2/q-1\}<r<2$.
		
		\smallskip
		\noindent\textit{Step 1:} \textbf{Preliminary work and decomposition of the integral.} We set $s \coloneq \min\{p,q\}$ and for $\ell,L \in \mathbb{N}_0$ we set 
		\begin{align*}
			\mathcal{V}_\ell \coloneq\mleft\{\mathbf{j} \in \mathbb{N}^d_{-1}:\sum_{i=1}^d j_i = \ell-d\mright\}, \quad \mathcal{W}_\ell \coloneq \{ (\mathbf{j},\mathbf{k}): \mathbf{j} \in \mathcal{V}_\ell, \mathbf{k}\in \mathbb{B}_\mathbf{j} \}
		\end{align*}
		and 
		\begin{align}\label{eq:defi:J}
			J_L f \coloneq \sum_{\ell=0}^L 2^{-\ell} \sum_{(\mathbf{j},\mathbf{k})\in \mathcal{W}_\ell} d_{\mathbf{j},\mathbf{k}}(f),
		\end{align}
		where $d_{\mathbf{j},\mathbf{k}}(f)$ are the Faber coefficients in \eqref{eq:fabermultivar}. We note that
		\begin{align}\label{eq:defWl}
			W_\ell \coloneq|\mathcal{W}_\ell| \asymp 2^\ell|\mathcal{V}_\ell|\asymp 2^\ell \ell^{d-1}.
		\end{align}
		Further let $F$ and $\tilde F_\ell$ denote the closed unit balls of ${S^r_{p,q}B(\mathbb{T}^d)}$ and $L_{s}^{\mathcal{W}_\ell}$ respectively.
		
		Now let us fix some parameters for the algorithms. Let $\delta>0$ such that
		\begin{align}\label{eq:defi:delta}
			\delta < \min \mleft( r,\frac{s(1+r)}{2}-1 \mright),
		\end{align}
		and set $K\coloneq 1+2\sum_{\ell=1}^\infty \lceil 8 (2 \ln \ell+\ln 8)\rceil 2^{-\ell\delta}$.
		
		
		Without loss of generality we assume $n \in \mathbb{N}$ with $n \ge 6(6d)^{d-1}K$ and set
		\begin{align}
			{\ell_0} &\coloneq\max\{\ell \in \mathbb{N}: 3^d  \ell^{d-1}2^\ell K \le n\}, \label{defi:lowl}\\
			{\ell_1} &\coloneq\min\{\ell \in \mathbb{N}: \ell^{(d-1)(1-1/q)}2^{-r\ell}\le {\ell_0}^{-(d-1)/q}2^{-(r+1){\ell_0}}\}, \label{eq:defi:highl}
		\end{align}
		and
		\begin{align}\label{defi:ntilde}
			\tilde n \coloneq 3^d{\ell_0} ^{d-1}2^{\ell_0}.
		\end{align}
		Using the approximation $J_L$ defined in \eqref{eq:defi:J}, we decompose the integral into the sum
		\begin{align*}
			I_d f= (I_df-J_{\ell_1} f)+J_{\ell_0} f+(J_{\ell_1} f- J_{\ell_0} f),
		\end{align*}
		and note that $n \ge K \tilde n > \tilde n$.
		So we can apply Lemma 6 (i) from \cite{heinrich2002summation} and Lemma 2 from \cite{heinrich2003sobolev} and obtain the following estimate
		\begin{align}\label{mainerrorest}
			e^\text{qua}_n(I_d,F)\le \sup_{f \in F}|I_df-J_{\ell_1} f|+e^\text{qua}_{\tilde n}(J_{\ell_0},F,0)+e^\text{qua}_{n-\tilde n}(J_{\ell_1} -J_{\ell_0} ,F).
		\end{align}
		In what follows, we evaluate the three terms separately.
		
		\smallskip
		\noindent\textit{Step 2.} \textbf{Estimation of the truncation error.} For $f \in F$ the multivariate Faber expansion \eqref{eq:fabermultivar} converges uniformly \cite{hinrichs2016}. Together with \eqref{eq:multifaber} and Definition \ref{defi:faber-basis} this yields
		\begin{align*}
			|I_df&-J_{\ell_1} f| 
			\le \sum_{\ell={\ell_1}+1}^{\infty}\sum_{(\mathbf{j},\mathbf{k})\in \mathcal{W}_\ell}  | d_{\mathbf{j},\mathbf{k}}(f) | \int_{[0,1)^d} v_{\mathbf{j},\mathbf{k}}(x) \mathrm{d} x \\
			&\lesssim \sum_{\ell={\ell_1}+1}^{\infty}\sum_{\mathbf{j} \in \mathcal{V}_\ell} 2^{-|\mathbf{j}|_1/p} \mleft(\sum_{\mathbf{k}\in \mathbb{B}_\mathbf{j}} | d_{\mathbf{j},\mathbf{k}}(f) |^p\mright)^{1/p} \\
			                                                                                                                                                      
			&\le \mleft( \sum_{\ell={\ell_1}+1}^{\infty}\sum_{\mathbf{j} \in \mathcal{V}_\ell} 2^{-r|\mathbf{j}|_1 q'} \mright)^{1/q'} \mleft[ \sum_{\mathbf{j} \in \mathbb{N}_{-1}^d}  2^{(r-1/p)|\mathbf{j}|_1q} \mleft(\sum_{\mathbf{k}\in \mathbb{B}_\mathbf{j}} | d_{\mathbf{j},\mathbf{k}}(f) |^p\mright)^{q/p} \mright]^{1/q}\\
			&\lesssim\mleft( \sum_{\ell={\ell_1}+1}^\infty |\mathcal{V}_\ell|2^{-r\ell q'}   \mright)^{1/q'}\|d_{\mathbf{j},\mathbf{k}}(f)\|_{s^r_{p,q}b} \\
			&\lesssim \mleft(\sum_{\ell={\ell_1}+1}^\infty  \ell^{d-1} 2^{-r\ell q'}\mright)^{1/q'}\|d_{\mathbf{j},\mathbf{k}}(f)\|_{s^r_{p,q}b} \\
			&\lesssim {\ell_1}^{(d-1)(1-1/q)}2^{-r{\ell_1}}\|d_{\mathbf{j},\mathbf{k}}(f)\|_{s^r_{p,q}b},\\
		\end{align*}
		whereby Hölder's inequality is used in the second estimate for the sum over $\mathbb{B}_\mathbf{j}$ with exponent $p$ and in the third estimate with the conjugate exponents $q$ and $q'$ ($1/q+1/q'=1$). 
		
		Applying Proposition \ref{prop:SBinsb}, \eqref{eq:defi:highl} and \eqref{defi:lowl} we thus obtain the following estimate for the first term in equation \eqref{mainerrorest}
		\begin{align}
			\sup_{f \in F}|I_df-J_{\ell_1} f| 
			&\lesssim {\ell_1}^{(d-1)(1-1/q)}2^{-r{\ell_1}}
			\lesssim {\ell_0}^{-(d-1)/q}2^{-(r+1){\ell_0}}\notag\\
			&= (2^{\ell_0} {\ell_0}^{d-1})^{-(r+1)} {\ell_0}^{(d-1)(r+1-1/q)} \notag\\
			&\lesssim n^{-r-1}(\log n)^{(d-1)(r+1-1/q)}. \label{errortrunc}
		\end{align}
		
		\smallskip
		\noindent\textit{Step 3.} \textbf{Evaluation of the lower levels.} By \eqref{eq:defi:fabercoeff} and \eqref{eq:defi:Uj} every coefficient $d_{\mathbf{j},\mathbf{k}}(f)$ depends on at most $3^d$ sampling points. Therefore the number of samples entering $J_{\ell_0} f$ can be estimated using \eqref{defi:ntilde} by 
		\begin{align*}
			\sum_{\ell=0}^{{\ell_0}} \sum_{\mathbf{j} \in \mathcal{V}_\ell} |\mathbb{B}_\mathbf{j}|| U_\mathbf{j}|
			&\le \frac{3^d}{2^d} \sum_{\ell=0}^{{\ell_0}} 2^\ell |\mathcal{V}_\ell| 
			\le \frac{3^d}{2^d (d-1)!} \sum_{\ell=0}^{{\ell_0}} (\ell+d-1)^{d-1}2^\ell\\
			&\le  3^d\frac{{\ell_0}^{d-1}}{(d-1)!}  2^{{\ell_0}}
			\le 3^d {\ell_0}^{d-1}2^{{\ell_0}}\le \tilde n.
		\end{align*}
		This yields 
		\begin{align}\label{errorlowerlevels}
			e^\text{qua}_{\tilde n}(J_{{\ell_0}},F,0)=0,
		\end{align}
		so $J_{{\ell_0}}$ can be computed to arbitrary precision using sufficiently many qubits.				
	
		\smallskip
		\noindent\textit{Step 4.} \textbf{Reduction of the intermediate levels to quantum summation.}
		This is the technical main part of the proof. To estimate the last term in equation \eqref{mainerrorest}
		we decompose $J_{\ell_1}  -J_{\ell_0}$ into its levels
		\begin{align}\label{eq:tele}
			J_{\ell_1}  -J_{\ell_0} = \sum_{\ell={\ell_0}+1}^{{\ell_1}} J'_\ell,
		\end{align}
		where, by \eqref{eq:defi:J}
		\begin{align}\label{eq:jstroke}
			J'_\ell f \coloneq J_\ell f-J_{\ell-1}f=2^{-\ell} \sum_{(\mathbf{j},\mathbf{k})\in \mathcal{W}_\ell} d_{\mathbf{j},\mathbf{k}}(f).
		\end{align}
		For $\ell \in \{{\ell_0}+1,\cdots,{\ell_1}\}$ we set
		\begin{align}\label{eq:defnuldefnl}
			\nu_\ell \coloneq \lceil 8 (2 \ln(\ell-{\ell_0})+\ln 8)\rceil, \quad
			n_\ell \coloneq 
			2^{\ell_0} {\ell_0}^{d-1}2^{-(\ell-{\ell_0})\delta}
		\end{align}
		to get
		\begin{align*}
			\sum_{\ell={\ell_0}+1}^{{\ell_1}} e^{-\nu_\ell/8} 
			\le \frac{1}{8}\sum_{\ell=1}^{\infty}\frac{1}{\ell^2} 
			< \frac{1}{4}
		\end{align*}
		and with \eqref{defi:lowl}
		\begin{align*}
			2\cdot 3^d\sum_{\ell={\ell_0}+1}^{{\ell_1}} \nu_\ell \lfloor n_\ell \rfloor 
			                   
			&\le 3^d {\ell_0}^{d-1} 2^{\ell_0}  2\sum_{\ell=1}^{\infty} \lceil 8 (2 \ln \ell+\ln 8)\rceil 2^{-\ell\delta}\notag \\
			&\le \tilde n (K-1)
			\le n-\tilde n.
		\end{align*}
		
		By using \eqref{eq:tele} and applying Lemma \ref{lem:errorofsum<sumoferrors} we see 
		\begin{align}\label{eq:errsumJl'}
			e^\text{qua}_{n-\tilde n}(J_{\ell_1}  -J_{\ell_0},F)
			\le \sum_{\ell={\ell_0}+1}^{{\ell_1}} e^\text{qua}_{2\cdot3^d\lfloor n_\ell \rfloor}( J'_{\ell},F).
		\end{align}
		
		At this point we define mappings $\Gamma_\ell: F \to L_s^{\mathcal{W}_\ell}, \ell={\ell_0},\dots,{\ell_1}$ which allow us to reduce the problem (in the sense of Lemma \ref{lem:quantemb}) to the computation of means in $L_s^{\mathcal{W}_\ell}$. For this, let $m^*\in \mathbb{N}$ such that 
		\begin{align}\label{eq:defm*}
			2^{-m^*/2}
			&\le 2^{-r{\ell_1}} {\ell_1}^{-d}\quad \text{and} \quad 
			c \le 2^{m^*/2-1},
		\end{align}
		where $c$ is the constant from \eqref{eq:besovemb}. We first define mappings $\beta, \gamma$ and $\eta_{\ell,\bm{l}}, \varrho_\ell$ for ($\bm{l} \in \{0,1,2\}^d$)  by
		\begin{align}
			\begin{split}\label{eq:def_beta_gamma}
				\beta &: \mathbb{R} \to \mathbb{B}_{m^*}, z \mapsto \min\mleft( (\lfloor 2^{m^*/2}z+2^{m^*-1} \rfloor)_+,2^{m^*}-1 \mright),\\
				\gamma &: \mathbb{B}_{m^*} \to \mathbb{R}, y \mapsto 2^{-m^*/2}y-2^{m^*/2-1},
			\end{split}\\
			\eta_{\ell,\bm{l}}&: \mathcal{W}_\ell \to \mathbb{T}^d, (\mathbf{j},\mathbf{k}) \mapsto 
			\begin{cases}
				x_{\mathbf{j},\mathbf{k},\bm{l}} &\textrm{if } \bm{l} \in U_\mathbf{j} \\
				0 &\textrm{otherwise},
			\end{cases} \nonumber \\
			\varrho_\ell&: \mathcal{W}_\ell \times \mathbb{B}_{m^*}^{\{0,1,2\}^d} \to \mathbb{R}, ((\mathbf{j},\mathbf{k}),y) \mapsto \mleft(-\frac{1}{2}\mright)^{|\{i:j_i\ge 0\}|}\frac{W_\ell}{2^{\ell}}\sum_{\bm{l} \in U_\mathbf{j}}(-2)^{|\{i:\ell_i=1\}|}\gamma(y_{\bm{l}}), \nonumber
		\end{align}
		where $x_{\mathbf{j},\mathbf{k},\bm{l}}$ are defined in (\ref{eq:samplepoints}). Finally we set
		\begin{align*}
			\Gamma_\ell(f)(\mathbf{j},\mathbf{k})\coloneq \varrho_\ell((\mathbf{j},\mathbf{k}),(\beta\circ f\circ \eta_{\ell,\bm{l}}(\mathbf{j},\mathbf{k}))_{\bm{l} \in \{0,1,2\}^d}).
		\end{align*}
		
		For $|z| \le 2^{m^*/2-1}$ we have
		\begin{align}\label{distretizationerror}
			|z-\gamma\circ \beta(z)|\le 2^{-m^*/2},
		\end{align}
		which, due to \eqref{eq:defm*}, applies in particular for $f\in F$. Again with \eqref{eq:defm*} and \eqref{eq:defi:fabercoeff} we get
		\begin{align}
			\label{eq:diffdgamma}
			\mleft|d_{\mathbf{j},\mathbf{k}}(f)W_\ell2^{-\ell}-\Gamma_\ell(f)(\mathbf{j},\mathbf{k})\mright|
			&\le \frac{W_\ell}{2^{\ell}} \sum_{\bm{l} \in U_\mathbf{j}} 2^{|\{i:\ell_i=1\}|}| f(x_{\mathbf{j},\mathbf{k},\bm{l}})-\gamma\circ \beta \circ f(x_{\mathbf{j},\mathbf{k},\bm{l}})| \notag\\ 
			&\lesssim \ell^{d-1} 2^{-m^*/2}
			\le {\ell_1}^{-1} 2^{-r{\ell_1}},
		\end{align}
		and therefore \eqref{eq:jstroke} yields
		\begin{align*}
			| J_\ell'f-S_{\mathcal{W}_\ell} \Gamma_\ell(f) | 
			&\le \frac{1}{W_\ell} \sum_{(\mathbf{j},\mathbf{k})\in \mathcal{W}_\ell} |d_{\mathbf{j},\mathbf{k}}(f)W_\ell 2^{-\ell}-\Gamma_\ell(f)(\mathbf{j},\mathbf{k}) | 
			\lesssim {\ell_1}^{-1} 2^{-r{\ell_1}},
		\end{align*}
		where $S_{\mathcal{W}_\ell}$ is the summation operator, see Definition \ref{defi:summation}. Using \cite[Lem. 6 (i)]{heinrich2002summation}, we thus obtain
		\begin{align}\label{eq:errJl'}
			e^\text{qua}_{2\cdot 3^d\lfloor n_\ell\rfloor}( J'_{\ell},F) & \le
			e^\text{qua}_{2\cdot 3^d\lfloor n_\ell \rfloor}( S_{\mathcal{W}_\ell} \Gamma_\ell,F) + {\ell_1}^{-1} 2^{-r{\ell_1}}.
		\end{align}
		
		Now we estimate the norm of $\Gamma_\ell(f)$ in $L^{\mathcal{W}_\ell}_{s}$ for $f \in F$. By using Hölder's inequality (regarding the outer sum in the case where $s = p$ and the inner sum in the other case) and Proposition \ref{prop:SBinsb} we get
		\begin{align*}
			\| d_{\mathbf{j},\mathbf{k}}(f)\|_{L^{\mathcal{W}_\ell}_{s}} 
			&=\mleft(  \frac{1}{W_\ell}\sum_{\mathbf{j}\in \mathcal{V}_\ell} \sum_{\mathbf{k} \in \mathbb{B}_\mathbf{j}}|d_{\mathbf{j},\mathbf{k}}(f)|^{s} \mright)^{1/s}\\
			&\lesssim 2^{-r\ell}\ell^{-(d-1)/q}\mleft(\sum_{\mathbf{j}\in \mathcal{V}_\ell} 2^{|\mathbf{j}|_1(r-1/p)q}\mleft( \sum_{\mathbf{k} \in \mathbb{B}_\mathbf{j}}|d_{\mathbf{j},\mathbf{k}}(f)|^p \mright)^{q/p}\mright)^{1/q}\\
			&\lesssim 2^{-r\ell}\ell^{-(d-1)/q} \|d_{\mathbf{j},\mathbf{k}}(f)\|_{s^r_{p,q}b}\\
			&\lesssim 2^{-r\ell}\ell^{-(d-1)/q}.
		\end{align*}
		
		In combination with \eqref{eq:diffdgamma},\eqref{eq:defWl} and \eqref{eq:defm*} this results in
		\begin{align*}
			\|\Gamma_\ell(f)\|_{L^{\mathcal{W}_\ell}_{s}}
			&\le   W_\ell 2^{-\ell} \| d_{\mathbf{j},\mathbf{k}}(f)\|_{L^{\mathcal{W}_\ell}_{s}}
			+ \| \Gamma_\ell(f)(\mathbf{j},\mathbf{k}) - d_{\mathbf{j},\mathbf{k}}(f)W_\ell2^{-\ell}   \|_{L_\infty^{\mathcal{W}_\ell}}\\
			&\lesssim 2^{-r\ell}\ell^{(d-1)(1-1/q)} + \ell^{-1}2^{-r\ell}\\
			&\lesssim 2^{-r\ell} \ell^{(d-1)(1-1/q)},
		\end{align*}
		i.e. there is a $c>0$ such that
		\begin{align*}
			\Gamma_\ell(F) \subseteq c 2^{-r\ell}\ell^{(d-1)(1-1/q)} \tilde F_\ell.
		\end{align*}
		
		An application of Lemma \ref{lem:quantemb} yields
		\begin{align*}
			e^\text{qua}_{2\cdot 3^d\lfloor n_\ell \rfloor}( S_{\mathcal{W}_\ell} \Gamma_\ell,F)
			\lesssim 2^{-r\ell}\ell^{(d-1)(1-1/q)} e^\text{qua}_{\lfloor n_\ell \rfloor}( S_{\mathcal{W}_\ell} , \tilde F_\ell )
		\end{align*}
		and together with \eqref{eq:errsumJl'} and \eqref{eq:errJl'} we obtain
		\begin{align*}
			e^\text{qua}_{n-\tilde n}(J_{\ell_1}-J_{\ell_0},F) 
			&\lesssim  2^{-r{\ell_1}}+\sum_{\ell={\ell_0}+1}^{{\ell_1}} 2^{-r\ell}\ell^{(d-1)(1-1/q)} e^\text{qua}_{\lfloor n_\ell \rfloor }( S_{\mathcal{W}_\ell} ,\tilde F_\ell).
		\end{align*}
		With \eqref{eq:defi:highl} we estimate the first term on the right hand side to
		\begin{align*}
			2^{-r{\ell_1}}
			&\le {\ell_1}^{(d-1)(1-1/q)}2^{-r{\ell_1}} 
			\le {\ell_0}^{-(d-1)/q}2^{-(r+1){\ell_0}}\notag \\
			&\lesssim n^{-r-1}(\log n)^{(d-1)(r+1-1/q)}
		\end{align*}
		and get
		\begin{align}\label{eq:errIdfin}
			e^\text{qua}_{n-\tilde n}(J_{\ell_1}-J_{\ell_0},F) 
			&\lesssim  n^{-r-1}(\log n)^{(d-1)(r+1-1/q)}+E(n).
		\end{align}
		with 
		\begin{align*}
			E(n)\coloneq \sum_{\ell={\ell_0}+1}^{{\ell_1}} 2^{-r\ell}\ell^{(d-1)(1-1/q)} e^\text{qua}_{\lfloor n_\ell \rfloor }( S_{\mathcal{W}_\ell} ,\tilde F_\ell)
		\end{align*}
		
		\smallskip
		\noindent\textit{Step 5.} \textbf{Estimation of the summation error.} It remains to find estimates for $E(n)$ in different parameter regimes. We have to distinguish between three cases. In each case the choice of $\delta$ in \eqref{eq:defi:delta} ensures sufficient geometric decay to control the corresponding sum and \eqref{defi:lowl} implies ${\ell_0}^{d-1}2^{\ell_0} \asymp n$. First we consider the case $s>2$. From Proposition \ref{prop:quantsum}\ref{prop:quantsumiii} (if $n_\ell<3$ we use the trivial error bounds) and \eqref{eq:defnuldefnl} we conclude
		\begin{align*}
			e^\text{qua}_{\lfloor n_\ell \rfloor }( S_{\mathcal{W}_\ell} , \tilde F_\ell ) \lesssim n_\ell^{-1}\le {\ell_0}^{-(d-1)}2^{-{\ell_0}}2^{(\ell-{\ell_0})\delta}.
		\end{align*}
		By using this inequality in \eqref{eq:errIdfin} and taking \eqref{defi:lowl} into account we get
		\begin{align*}
			E(n) 
			&\lesssim {\ell_0}^{-(d-1)}2^{-(r+1){\ell_0}}\sum_{\ell={\ell_0}+1}^{{\ell_1}} \ell^{(d-1)(1-1/q)} 2^{-(r-\delta)(\ell-{\ell_0})}\\
			&\lesssim 2^{-(r+1){\ell_0}} {\ell_0}^{-(d-1)/q}\\
			&\lesssim (2^{{\ell_0}}{\ell_0}^{d-1})^{-(1+r)}  ({\ell_0}+\log {\ell_0})^{(d-1)(r+1-1/q)}  \\
			&\lesssim n^{-r-1} (\log n)^{(d-1)(r+1-1/q)}.
		\end{align*}
		
		Next we prove the upper bound in the case $1\le s < 2$. From Proposition \ref{prop:quantsum}\ref{prop:quantsumi}, \eqref{eq:defnuldefnl} and \eqref{eq:defWl} we get
		\begin{align*}
			e^\text{qua}_{\lfloor n_\ell \rfloor }( S_{\mathcal{W}_\ell} ,\tilde F_\ell)
			&\lesssim \min\{n_\ell^{-2(1-1/s)},n_\ell^{-2/s}W_\ell^{2/s-1}\}\max\{\log(n_\ell/\sqrt {W_\ell}),1\}^{2/s-1}\\
			& \lesssim n_\ell^{-2/s}W_\ell^{2/s-1} (1+\max\{\log(n_\ell/\sqrt {W_\ell}),0\})^{2/s-1} \\
			& \lesssim n_\ell^{-2/s}W_\ell^{2/s-1} (1+ {\ell_0} + \log {\ell_0})^{2/s-1} \\
			&\lesssim {\ell_0}^{-2(d-1)/s}\ell^{(d-1)(2/s-1)} 2^{-{\ell_0}} 2^{-(\ell-{\ell_0})(s-2-2\delta)/s}(\log{\ell_0}+{\ell_0})^{2/s-1}.
		\end{align*}
		Together with \eqref{eq:errIdfin}, and \eqref{defi:lowl} this results in the following estimate
		\begin{align*}
			E(n)
			& \lesssim 2^{-(r+1){\ell_0}}{\ell_0}^{-2(d-1)/s}(\log {\ell_0} + {\ell_0})^{2/s-1} \\
			& \phantom{==} \cdot\!  \sum_{\ell={\ell_0}+1}^{{\ell_1}} 2^{-(\ell-{\ell_0})(s+rs-2-2\delta)/s} \ell^{(d-1)(2/s-1/q)}\\
			&\lesssim 2^{-(r+1){\ell_0}} {\ell_0}^{-(d-1)/q}(\log {\ell_0} + {\ell_0})^{2/s-1} \\
			&\lesssim (2^{\ell_0} {\ell_0}^{d-1})^{-(r+1)} ({\ell_0}+\log {\ell_0})^{(d-1)(r+1-1/q)+2/s-1}\\
			&\lesssim n^{-r-1} (\log n)^{(d-1)(r+1-1/q)+2/s-1}\,.  
		\end{align*}
		
		Finally we consider the case $s=2$. An application of Proposition \ref{prop:quantsum}\ref{prop:quantsumii} together with \eqref{eq:defnuldefnl} and \eqref{eq:defWl} yields
		\begin{align*}
			e^\text{qua}_{\lfloor n_\ell \rfloor }( S_{\mathcal{W}_\ell} ,\tilde F_\ell)
			&\lesssim n_\ell^{-1} (\log(W_\ell/n_\ell)+\log\log(n_\ell+1)+2)^{3/2} \\
			&\phantom{==}\cdot \log(\log(W_\ell/n_\ell)+\log\log(n_\ell+1)+2)\\
			&\lesssim {\ell_0}^{-(d-1)}2^{-{\ell_0}}2^{(\ell-{\ell_0})\delta}(\ell-{\ell_0}+\log(\ell/{\ell_0})+\log\log n+1)^{3/2}\\
			&\phantom{==}\cdot \log(\ell-{\ell_0}+\log(\ell/{\ell_0})+\log\log n+1)\,.
		\end{align*}
		We plug this into \eqref{eq:errIdfin}, use again \eqref{defi:lowl} and get the upper bound of the last case:
		\begin{align*}
			E(n)
			&\lesssim 2^{-(1+r){\ell_0}}{\ell_0}^{-(d-1)} \sum_{\ell={\ell_0}+1}^{{\ell_1}}2^{-(r-\delta)(\ell-{\ell_0})}\ell^{(d-1)(1-1/q)}\\
			& \phantom{=============} \cdot (\ell-{\ell_0}+\log(\ell/{\ell_0})+\log\log n+1)^{3/2} \\
			& \phantom{=============} \cdot \log(\ell-{\ell_0}+\log(\ell/{\ell_0})+\log\log n)\\
			&\lesssim 2^{-(1+r){\ell_0}} {\ell_0}^{-(d-1)/q}(\log\log n)^{3/2} \log\log\log n\\
			&\lesssim (2^{{\ell_0}}{\ell_0}^{d-1})^{-1-r}({\ell_0}+\log{\ell_0})^{(d-1)(r+1-1/q)}(\log\log n)^{3/2} \log\log\log n\\
			&\lesssim n^{-1-r}(\log n)^{(d-1)(r+1-1/q)}(\log\log n)^{3/2} \log\log\log n.
		\end{align*}
		In summary, \eqref{mainerrorest}, \eqref{errortrunc}, \eqref{errorlowerlevels}, \eqref{eq:errIdfin}, and the estimates of $E(n)$ in the three different cases yield the claimed rates.
