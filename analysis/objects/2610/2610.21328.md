\label{thm:phase}
Assume \eqref{ass:K} and \eqref{eq:data}. For every $0\le\beta<\infty$, there exists a positive continuous directional amplitude $a_\beta$ on $\overline{\mathbb S_+}$. Define the directional phase by
\[
 \xi_\beta(\theta)=\log a_\beta(\theta).
\]
For every $0<B<\infty$, there exists $C_B>0$ such that
\begin{equation}\label{eq:holder-main}
 |a_\beta(\theta)-a_\beta(\omega)|
 \le C_B|\theta-\omega|^{1/3},
 \qquad 0\le\beta\le B,\quad\theta,\omega\in\overline{\mathbb S_+}.
\end{equation}
For every $0<\delta\le1$ and $0<Z<\infty$, the receiving front satisfies
\begin{equation}\label{eq:cone-wave}
 \sup_{\theta\in\Gamma_\delta,\ |z|\le Z}
 \left|u_+(t,(m(t)+z)\theta)-U_*(z-\xi_\beta(\theta))\right|\longrightarrow0
 \qquad(t\to\infty).
\end{equation}
Along the interface, both traces converge to the same profile:
\begin{equation}\label{eq:eq-wave}
 \sup_{\eta\in\mathbb S^{N-2},\ |z|\le Z}
 \left|u_\pm(t,(m(t)+z)\eta,0)-U_*(z-\xi_\beta(\eta,0))\right|\longrightarrow0
 \qquad(t\to\infty).
\end{equation}
This equatorial convergence also holds in bounded physical neighborhoods transverse to each equatorial ray. Both convergence statements are uniform for $\beta\in[0,B]$, for every finite $B>0$.

For $R>0$, $\theta\in\overline{\mathbb S_+}$, and a threshold $\upsilon\in(0,1)$, define the first arrival time on the receiving side by
\[
 T_\upsilon^\beta(R,\theta)=\inf\{t>0:u_+(t,R\theta)\ge\upsilon\},
\]
where the receiving trace is used when $\theta_N=0$. Then
\begin{equation}\label{eq:arrival}
 T_\upsilon^\beta(R,\theta)-T_\upsilon^0(R,\theta)\longrightarrow
 \tau_\beta(\theta):=\frac12\log\frac{a_0(\theta)}{a_\beta(\theta)}
 \qquad(R\to\infty).
\end{equation}
The limiting arrival-time difference is independent of $\upsilon$. The convergence is uniform for $\beta\in[0,B]$, for $\upsilon$ in compact subsets of $(0,1)$, and for directions in each fixed $\Gamma_\delta$, or separately on the equator.

\label{thm:grazing}
For $\eta\in\mathbb S^{N-2}$ and $0<s<1$, set
\[
 \theta_s=(\sqrt{1-s^2}\,\eta,s).
\]
As $s\downarrow0$,
\begin{equation}\label{eq:grazing-I}
 I_N(\theta_s)=4(4\pi)^{(N-2)/2}a_\mathrm{Neu}(\eta,0)\log(1/s)(1+o(1)),
\end{equation}
and consequently
\begin{equation}\label{eq:grazing-C}
 C_N(\theta_s)=-\frac12\log\log(1/s)
       +\frac12\log\frac{\pi a_0(\eta,0)}{2a_\mathrm{Neu}(\eta,0)}+o(1).
\end{equation}
Both asymptotic formulas hold uniformly for $\eta\in\mathbb S^{N-2}$.

\label{thm:saturation}
As $\beta\to\infty$, uniformly for $\eta\in\mathbb S^{N-2}$,
\begin{equation}\label{eq:saturation}
\begin{aligned}
 a_\beta(\eta,0)&\longrightarrow\frac12a_\mathrm{Neu}(\eta,0),\\
 \tau_\beta(\eta,0)&\longrightarrow
 \tau_*^{\rm eq}(\eta):=\frac12\log\frac{2a_0(\eta,0)}{a_\mathrm{Neu}(\eta,0)}.
\end{aligned}
\end{equation}
If \eqref{ass:C} also holds, then $\tau_\beta(\eta,0)$ increases strictly to this limit as $\beta$ increases, and
\begin{equation}\label{eq:saturation-range}
 0<\tau_*^{\rm eq}(\eta)<\frac12\log2.
\end{equation}
In particular, for each fixed $\eta\in\mathbb S^{N-2}$, with $\theta_s$ as above,
\begin{align*}
 \lim_{s\downarrow0}\lim_{\beta\to\infty}a_\beta(\theta_s)&=0,
 &\lim_{\beta\to\infty}\lim_{s\downarrow0}a_\beta(\theta_s)&=\tfrac12a_\mathrm{Neu}(\eta,0),\\
 \lim_{s\downarrow0}\lim_{\beta\to\infty}\tau_\beta(\theta_s)&=+\infty,
 &\lim_{\beta\to\infty}\lim_{s\downarrow0}\tau_\beta(\theta_s)&=\tau_*^{\rm eq}(\eta).
\end{align*}

<!-- proof-1: body_md -->
Let $\mathsf T_\kappa(t)$ denote the heat semigroup
on $H$ with Robin coefficient $\kappa$.
We use the endpoint conventions
\[
 \mathsf T_0=\mathsf T_{\mathrm N},
 \qquad
 \mathsf T_\infty=\mathsf T_{\mathrm D}.
\]
The kernel inequalities
$0\le r_\kappa\le r_{\mathrm N}$ imply, uniformly
for $\kappa\in[0,\infty]$,
\[
 \|\mathsf T_\kappa(t)\|_{L^\infty\to L^\infty}\le1,
 \qquad
 \|\mathsf T_\kappa(t)\|_{L^2\to L^\infty}
 \le C_Nt^{-N/4}.
\]
These estimates include evaluation on $y=0$,
using the continuous positive-time representatives.

We first verify operator convergence away from
time zero. For $0<\kappa<\infty$, positivity and
\eqref{eq:robin-positive} give
\begin{align*}
 &\sup_{y\ge0}\int_0^\infty
   |r_\kappa(t,y,z)-r_{\mathrm D}(t,y,z)|\,\mathrm d z\\
 &\qquad=
 \sup_{y\ge0}
 2\int_0^\infty\mathrm e^{-\kappa a}G_t^1(y+a)\,\mathrm d a
 \le\frac{C}{\kappa\sqrt t}.
\end{align*}
Similarly, \eqref{eq:robin} gives
\begin{align*}
 &\sup_{y\ge0}\int_0^\infty
   |r_{\mathrm N}(t,y,z)-r_\kappa(t,y,z)|\,\mathrm d z\\
 &\qquad\le
 2\kappa\int_0^\infty\int_0^\infty
       G_t^1(z+a)\,\mathrm d z\,\mathrm d a
 \le C\kappa\sqrt t.
\end{align*}
For a finite positive coefficient $\kappa_*$,
the same formula yields
\[
 \sup_{y\ge0}\int_0^\infty
 |r_\kappa(t,y,z)-r_{\kappa_*}(t,y,z)|\,\mathrm d z
 \le
 \int_0^\infty
 |\kappa\mathrm e^{-\kappa a}
       -\kappa_*\mathrm e^{-\kappa_*a}|\,\mathrm d a,
\]
whose right-hand side tends to zero as
$\kappa\to\kappa_*$.
The tangential heat kernel has mass one.
Consequently, for every $0<h<T$ and every
$\kappa_j\to\kappa_*\in[0,\infty]$,
\[
 \sup_{h\le t\le T}
 \|\mathsf T_{\kappa_j}(t)-\mathsf T_{\kappa_*}(t)
       \|_{L^\infty\to L^\infty}
 \longrightarrow0.
\]
In particular, this convergence is uniform up to
the flat boundary.

Write
\[
 \mathbf U_\beta=(S_\beta,D_\beta),\qquad
 \mathsf S_\beta(t)
 =\operatorname{diag}
   \bigl(\mathsf T_{\mathrm N}(t),
         \mathsf T_{2/\beta}(t)\bigr),
\]
with $2/0=\infty$ and $2/\infty=0$.
The folded reaction vector is
\[
 \mathcal F(S,D)
 =\frac12
 \begin{pmatrix}
 f(S+D)+f(S-D)\\
 f(S+D)-f(S-D)
 \end{pmatrix}.
\]
It is bounded and Lipschitz on the common invariant
region $0\le S-D\le S+D\le1$.

Fix $0<h<t_*/2$. For $t\in[t_*,T]$, apply the
mild formula on the short interval $[t-h,t]$:
\[
 \mathbf U_\beta(t)
 =\mathsf S_\beta(h)\mathbf U_\beta(t-h)
 +\int_{t-h}^t
   \mathsf S_\beta(t-s)\mathcal F(\mathbf U_\beta(s))
   \,\mathrm d s.
\]
Subtract the corresponding formula for $\beta_j$.
The two recent-time integrals have total
$L^\infty$ norm at most $Ch$, uniformly in the
parameters. For the initial term, insert and subtract
$\mathsf S_{\beta_j}(h)\mathbf U_\beta(t-h)$.
The smoothing estimate and operator convergence
then give
\begin{align*}
 &\sup_{t_*\le t\le T}
 \|\mathbf U_{\beta_j}(t)-\mathbf U_\beta(t)\|_\infty\\
 &\quad\le
 C h^{-N/4}
 \sup_{0\le s\le T}
 \|\mathbf U_{\beta_j}(s)-\mathbf U_\beta(s)\|_2\\
 &\qquad+
 \|\mathsf S_{\beta_j}(h)-\mathsf S_\beta(h)
       \|_{L^\infty\to L^\infty}
 \sup_{0\le s\le T}\|\mathbf U_\beta(s)\|_\infty
 +Ch.
\end{align*}
The already established $C([0,T];L^2)$ convergence
makes the first term vanish as $j\to\infty$.
For fixed $h$, operator convergence makes the second
term vanish as well. Thus the limit superior is
bounded by $Ch$. Letting $h\downarrow0$ proves
uniform convergence of the folded pair.
Since $p=S+D$ and $q=S-D$, the asserted convergence
of both densities follows, including at $y=0$.

\label{lem:envelope}
For every $0<\sigma<1/4$, there exist constants
$C_\sigma,c>0$, independent of $\beta\in[0,\infty]$,
such that
\begin{equation}\label{eq:envelope}
 u(t,x)\le
 C_\sigma(z+t^\sigma)\mathrm e^{-z}\mathrm e^{-cz^2/t},
 \qquad z=|x|-m(t)\ge0,\quad t\ge2.
\end{equation}
Both interface traces are included in this estimate.
The same bound holds for the even Neumann extension $P$,
after adjusting the constants.

\label{lem:energytrace}
Fix $0<B<\infty$ and an integer $M>(N+1)/2$. There exist $T_0\ge2$ and $C_{B,M}>0$ such that, for every $T\ge T_0$, $0<\beta\le B$, and $\eta\in\mathbb S^{N-2}$,
\begin{equation}\label{eq:trace-energy}
 \int_{\log T}^{\log(2T)}\int_{\rho\ge2a(t),\ v\in\mathbb R^{N-2}}
 (1+\rho^2+|v|^2)^M W(\tau,\rho,v,0)^2\,\mathrm d\rho\,\mathrm d v\,\mathrm d\tau
 \le C_{B,M}T^{-1/2},\qquad t=\mathrm e^\tau.
\end{equation}
Consequently,
\begin{equation}\label{eq:trace-L1}
 \int_{\log T}^{\log(2T)}\int_{\rho\ge2a(t),\ v\in\mathbb R^{N-2}}
 (1+\rho)W(\tau,\rho,v,0)\,\mathrm d\rho\,\mathrm d v\,\mathrm d\tau
 \le C_{B,M}T^{-1/4}.
\end{equation}
At $\beta=0$, the boundary trace of $W$ vanishes, so both estimates hold trivially.

<!-- proof-3: body_md -->
Choose a smooth cutoff $\chi_0$ with $0\le\chi_0\le1$, $\chi_0=0$ on $(-\infty,1]$, and $\chi_0=1$ on $[2,\infty)$. Set
\[
 \chi(\tau,\rho)=\chi_0(\rho/a(t)),\qquad \Psi(X)=(1+|X|^2)^M.
\]
Testing \eqref{eq:W} with $\chi^2\Psi W$, initially with compact outer cutoffs, produces the nonnegative boundary dissipation
\[
 \kappa\sqrt t\int_{Y=0}\chi^2\Psi W^2.
\]
The polynomial weight satisfies
\[
 |\nabla\Psi|^2/\Psi\le C_M\Psi,\qquad |X\cdot\nabla\Psi|\le C_M\Psi.
\]
The potential is controlled by \eqref{eq:V-small}. Moreover, $\partial_\tau\chi$ is uniformly bounded and supported in $a(t)\le\rho\le2a(t)$. The cutoff terms are therefore controlled by the weighted energy and
\[
 Ca(t)^{-2}\int_{a(t)\le\rho\le2a(t),\ Y>0}\Psi W^2\,\mathrm d X.
\]
To estimate this term, write
\[
 l=m(t)+\rho\sqrt t,\qquad r=\sqrt{l^2+t|(v,Y)|^2},\qquad d=r-l\ge0.
\]
Since $0\le D_\beta\le p_\beta/2$, the Gaussian envelope gives
\[
 W\le C\bigl(\rho+d/\sqrt t+t^{\sigma-1/2}\bigr)\mathrm e^{-c\rho^2}\mathrm e^{-d}.
\]
On the cutoff layer, $l$ is comparable to $t$, and
\[
 d=\frac{t|(v,Y)|^2}{r+l}\ge c\frac{|(v,Y)|^2}{1+|(v,Y)|}.
\]
Hence integration against any fixed polynomial weight yields
\[
 \int_{\mathbb R^{N-2}}\int_0^\infty\Psi W^2\,\mathrm d Y\,\mathrm d v
 \le C_M\bigl(a(t)^2+t^{-1}+t^{2\sigma-1}\bigr).
\]
The layer has width $a(t)$, so
\begin{equation}\label{eq:cutoff-error}
 Ca(t)^{-2}\int_{a(t)\le\rho\le2a(t),\ Y>0}\Psi W^2\,\mathrm d X
 \le C_M\left(a(t)+\frac{t^{-1}}{a(t)}+\frac{t^{2\sigma-1}}{a(t)}\right)\le C_M.
\end{equation}
This proves \eqref{eq:cutoff-error}.

The same envelope gives a uniform bound for
\[
 E(\tau)=\int_{Y>0}\chi^2\Psi W^2\,\mathrm d X.
\]
For large $\rho$, the estimate
\[
 d\ge c\frac{|(v,Y)|^2}{1+\rho+|(v,Y)|}
\]
combined with $\mathrm e^{-c\rho^2}$ controls all polynomial moments and justifies removing the outer cutoffs. After absorbing the gradient cross terms, the energy inequality therefore gives
\[
 E'(\tau)+\kappa\sqrt t\int_{Y=0}\chi^2\Psi W^2\le C_M(E(\tau)+1).
\]
Integrating over $[\log T,\log(2T)]$, whose length is $\log2$, and using the uniform bound on $E$, we obtain
\[
 \int_{\log T}^{\log(2T)}\kappa\sqrt t\int_{Y=0}\chi^2\Psi W^2\,\mathrm d\tau\le C_M.
\]
Since $\kappa\ge2/B$, $t\ge T$, and $\chi=1$ for $\rho\ge2a(t)$, this proves \eqref{eq:trace-energy}.

Finally, $W\ge0$, and the condition $M>(N+1)/2$ ensures
\[
 \int_0^\infty\int_{\mathbb R^{N-2}}\frac{(1+\rho)^2}{(1+\rho^2+|v|^2)^M}\,\mathrm d v\,\mathrm d\rho<\infty.
\]
Cauchy--Schwarz in space and logarithmic time now gives \eqref{eq:trace-L1}.

\label{lem:robin-local}
Let $X_0\in\{Y=0\}$ and $0<r<R\le1$. Define
\[
 B_R^+=B_R(X_0)\cap\{Y>0\},\qquad
 Q_R^+=(s_0-R^2,s_0]\times B_R^+,
\]
and
\[
 \Gamma_R=(s_0-R^2,s_0]\times(B_R(X_0)\cap\{Y=0\}).
\]
Suppose that $w\ge0$ is a subsolution of
\[
 w_s\le\Delta w+b\cdot\nabla w+cw\quad\text{in }Q_R^+,
 \qquad \partial_nw+\lambda w\le0\quad\text{on }\Gamma_R,
\]
where $n=-e_Y$, $|b|+c_+\le M$, and $\lambda\ge0$. The boundary coefficient $\lambda$ may depend on space and time and is locally bounded for each subsolution; no uniform upper bound on $\lambda$ is assumed. Then
\begin{equation}\label{eq:robin-local-sup}
 \|w\|_{L^\infty(Q_r^+)}
 \le\frac{C(N,M)}{(R-r)^{(N+2)/2}}\|w\|_{L^2(Q_R^+)}.
\end{equation}
The constant is independent of $\lambda$. If $w$ is continuous up to the flat boundary, the same bound holds on $\Gamma_r$. The conclusion also holds when the Robin inequality is replaced by homogeneous Dirichlet data on the flat boundary.
<!-- end lemma-4 -->
<!-- proof-4: body_md -->
Choose a space--time cutoff $0\le\zeta\le1$, equal to one on $Q_r^+$, vanishing at the initial time and near the curved spatial boundary of $Q_R^+$, and satisfying
\[
 |\nabla\zeta|\le\frac{C}{R-r},\qquad
 |\partial_s\zeta|\le\frac{C}{(R-r)^2}.
\]
The cutoff may be nonzero on the flat boundary.

For $p\ge2$, test the differential inequality with $\zeta^2w^{p-1}$. The boundary condition gives
\[
 \int_{\{Y=0\}}\zeta^2w^{p-1}\partial_nw
 \le-\int_{\{Y=0\}}\lambda\zeta^2w^p.
\]
Thus the Robin term contributes nonnegative dissipation after it is moved to the left-hand side. Estimate the drift directly by Young's inequality, without differentiating $b$, and use $c\le c_+\le M$. Setting $v=w^{p/2}$, integrating in time, and taking the supremum of the spatial energy yields
\begin{align}\label{eq:robin-local-energy}
 &\sup_s\int_{B_R^+}\zeta^2v^2
 +\int_{Q_R^+}|\nabla(\zeta v)|^2
 +p\int_{\Gamma_R}\lambda\zeta^2v^2\notag\\
 &\qquad\le Cp^2\int_{Q_R^+}
 \bigl(|\partial_s\zeta|+|\nabla\zeta|^2+(1+M^2)\zeta^2\bigr)v^2.
\end{align}
No upper bound or derivative bound on $\lambda$ enters this estimate. Truncated and regularized power tests justify the calculation for weak subsolutions. Under homogeneous Dirichlet conditions, the boundary contribution vanishes.

Extend $\zeta v$ evenly across $\{Y=0\}$. Since it vanishes near the curved boundary, the extension belongs to $H_0^1(B_R(X_0))$. The parabolic Sobolev inequality therefore applies with a constant depending only on $N$; only the test function is extended, not the equation. With $\chi=1+2/N$, the energy estimate \eqref{eq:robin-local-energy} gives
\[
 \|w\|_{L^{p\chi}(Q_r^+)}
 \le\left[\frac{C(N,M)p^2}{(R-r)^2}\right]^{1/p}\|w\|_{L^p(Q_R^+)}.
\]
Here $R-r\le1$ allows the zeroth-order contribution to be included in the same factor.

Iterate this inequality with
\[
 p_j=2\chi^j,\qquad R_j=r+(R-r)2^{-j},\qquad j=0,1,\ldots,
\]
using $R_j$ as the outer radius and $R_{j+1}$ as the inner radius. Since
\[
 \sum_{j=0}^\infty\frac{2}{p_j}=\frac{N+2}{2},\qquad
 \sum_{j=0}^\infty\frac{j+\log p_j}{p_j}<\infty,
\]
the iteration yields \eqref{eq:robin-local-sup}. Continuity up to the flat boundary gives the trace estimate.

<!-- proposition-1: body_md -->
\label{prop:moment}
For every $0<B<\infty$, there exists a limiting moment $E_\beta^\infty(\theta)$ such that
\begin{equation}\label{eq:Etail}
 |E_\beta(t,\theta)-E_\beta^\infty(\theta)|\le C_Bt^{-1/4}.
\end{equation}
for all sufficiently large $t$, uniformly for $0\le\beta\le B$ and $\theta\in\overline{\mathbb S_+}$. The limit is jointly continuous in $(\beta,\theta)$ on $[0,B]\times\overline{\mathbb S_+}$.

At equatorial directions, the stronger estimate
\[
 |E_\beta(t,(\eta,0))-E_\beta^\infty(\eta,0)|\le Ct^{-\mu}(1+\log t),\qquad \mu=\frac12-2\sigma=\frac38,
\]
holds uniformly for $\eta\in\mathbb S^{N-2}$ and $\beta\in[0,\infty]$.
<!-- end proposition-1 -->
<!-- proof-5: body_md -->
The volume contribution is controlled by \eqref{eq:absorption-tail}. It remains to estimate the interface contribution in \eqref{eq:moment-balance} uniformly as the direction approaches the equator.

On each fixed receiving cone, use $D\le p_\infty/2$ and the linear Neumann Gaussian upper bound; the tilted interface term is exponentially integrable. Near the equator write $\theta=(b\eta,s)$, $b=\sqrt{1-s^2}\ge1/2$, and $\lambda=1-b$. For $s>0$, the change of variables \eqref{eq:cartesian} bounds the absolute interface contribution by
\begin{equation}\label{eq:Itau}
 2s t^{-1+k\lambda}\mathrm e^{-2\lambda t}
 \int_{\mathbb R\times\mathbb R^{N-2}} |1-2\lambda t-bk\log t+b\rho\sqrt t|
       \mathrm e^{-\lambda\rho\sqrt t}W(\tau,\rho,v,0)\,\mathrm d\rho\,\mathrm d v.
\end{equation}
On $\rho\ge2a(t)$, maximizing the scalar coefficients over $s$ and using $s^2=\lambda(1+b)$ bounds \eqref{eq:Itau} by
\[
 Ct^{-1}\int_{\rho\ge2a(t)}(1+\rho)W(\tau,\rho,v,0)\,\mathrm d\rho\,\mathrm d v.
\]
The factor $t^{k\lambda}$ is absorbed into $\mathrm e^{-\lambda t}$ for large $t$. Integration over $t\in[T,2T]$, using $\,\mathrm d t/t=\,\mathrm d\tau$ and \eqref{eq:trace-L1}, gives $C_BT^{-1/4}$.

For the complementary part $l=\eta\cdot x'\le m+2t^\gamma$, first restrict to $r'=|x'|\le m+4t^\gamma$. Near the front, tangential angular integration contributes $t^{(N-2)/2}$, so the tilted radial factor is $t^{-2+k\lambda}\mathrm e^{-2\lambda t}$. For $r'=m+z$, $0\le z\le4t^\gamma$, the remaining integral is bounded by
\[
 C\int_0^{4t^\gamma}(z+t^\sigma)(1+\lambda t+\log t+z)\,\mathrm d z.
\]
Multiplication by $s$ and maximization give
$Ct^{-5/2+3\gamma}(1+\log t)$. The part $z<0$ is smaller, using $u\le1$. If $r'>m+4t^\gamma$ but $l\le m+2t^\gamma$, then $r'-l\ge2t^\gamma$; the factor $\mathrm e^{-b(r'-l)}$ makes this remainder superpolynomially small. Since $\gamma=1/4$, the complementary contribution integrated over $[T,2T]$ is $O(T^{-3/4}\log T)$.

Summing the dyadic bounds and the volume tail with $\sigma=1/16$ proves \eqref{eq:Etail}. At each fixed positive time, parameter continuity and the common Gaussian bound make $E_\beta(t,\theta)$ jointly continuous. Uniform convergence therefore gives joint continuity of the limit. At $s=0$, the interface contribution vanishes identically, and integration of \eqref{eq:absorption-tail} gives the stronger equatorial estimate.

\label{lem:wave-tails}
There exist $B_->0$ and $B_+\in\mathbb R$ such that, as $s\to-\infty$,
\begin{align*}
 1-U_*(s)&=B_-\mathrm e^{\lambda_*s}+O(\mathrm e^{2\lambda_*s}),\\
 U_*'(s)&=-\lambda_*B_-\mathrm e^{\lambda_*s}+O(\mathrm e^{2\lambda_*s}),\\
 U_*''(s)&=-\lambda_*^2B_-\mathrm e^{\lambda_*s}+O(\mathrm e^{2\lambda_*s}).
\end{align*}
As $s\to+\infty$,

 U_*(s)&=(s+B_+)\mathrm e^{-s}+O((1+s)^2\mathrm e^{-2s}),\\

\label{lem:coarse}
Fix $0<B<\infty$, $0<\gamma<1/3$, and $\gamma<\delta<1/2$. In the moving Cartesian coordinates

 h(t,z,\theta,\zeta)=z-U_*^{-1}(u(t,x)).

\end{align}
for every $\epsilon>0$ and $Z>0$ in the second estimate. The same bounds hold in equatorial coordinates for each folded component. In \eqref{eq:h-behind}, $C_\epsilon$ is independent of $Z$, although the starting time may depend on $Z$.

\begin{equation}\label{eq:sharp-upper}
 u(t,x)\le C(1+z_r)\mathrm e^{-z_r},\qquad

\end{equation}

Then $\eta_t=t^{-1/2}$, the inner boundary is $z_r=-1$, and the coefficient bounds give bounded $\zeta_{\rm lin}$ and $Q\asymp\eta_t$. Here $\zeta_{\rm lin}$ is the scalar coefficient in \eqref{eq:linear-Phi}. The comparison accommodates both the inner-boundary data and the Gaussian tail at the starting time.  \mathrm e^{-2t}\mathrm e^t\mathrm e^{|\theta'|^2t}
 =\mathrm e^{-\theta_N^2t}.
\]
Thus, for each fixed integer $j\ge0$, the weighted
boundary integral is bounded by a constant times
\[
 t^{-1/2}(1+t)^j
 \mathrm e^{-\delta^2t}\mathrm e^{-d_0^2/(4t)},
\]
where $d_0>0$ is the distance of the initial support
from the interface. This bound is integrable at both
zero and infinity.

Finite-time convergence, followed by dominated convergence,
proves convergence of the weighted input moments.
The domination is uniform for $\theta\in\Gamma_\delta$,
so the convergence is uniform on that cone. This argument
does not assert an $O(\varepsilon)$ rate after integration
over the entire time interval.

Let $U_\varepsilon$ be the corresponding solution. Fix $0<\zeta<1$, put $L=\log(1/\varepsilon)$, and choose
\[
 A=(1-\zeta)L,\qquad T=L^4.
\]
Suppose $\operatorname{supp}\varphi\subset B_{R_0}$ and take $\varepsilon$
sufficiently small that $\varepsilon\|\varphi\|_\infty\le1$.
For each $\theta\in\mathbb S^{N-1}$, introduce the moving half-space
\[
 D_t=\{x:\theta\cdot x>2t-A\},\qquad 0\le t\le T.
\]
We construct a linear Dirichlet lower solution in $D_t$ and
compute its kernel explicitly. All constants below may depend
on $\varphi,N,\zeta$, but are independent of $\theta$ and
sufficiently small $\varepsilon$.

Linear comparison gives $U_\varepsilon(t,x)\le C\varepsilon\mathrm e^t$.
For $t\ge A/2\ge1$ and $x\in D_t$, the compact-data heat bound gives
\[
 U_\varepsilon(t,x)
 \le C\varepsilon t^{-N/2}
 \exp\left(t-\frac{[(2t-A-R_0)_+]^2}{4t}\right)
 \le C\varepsilon\mathrm e^{A+R_0}
 \le C\varepsilon^\zeta.
\]
Indeed, with $B=A+R_0$, the exponent is at most $B$:
if $2t\le B$ it equals $t\le B/2$, and otherwise it equals
$B-B^2/(4t)$. Choose a common constant $C$ so that
\[
 q_\varepsilon(t)=
 \begin{cases}
 C\varepsilon\mathrm e^t,&0\le t\le A/2,\\
 C\varepsilon^\zeta,&A/2<t\le T
 \end{cases}
\]

$\Lambda_\varepsilon(t)=\int_0^t q_\varepsilon(s)\,\mathrm d s$.
Then
\begin{equation}\label{eq:seed-absorption-loss}
 0\le \Lambda_\varepsilon(T)
 \le C\left[\varepsilon^{(1+\zeta)/2}
                 +T\varepsilon^\zeta\right]=o(1).
\end{equation}

Let $v$ solve the linear problem
\[
 \begin{cases}
 v_t=\Delta v+(1-q_\varepsilon(t))v,&x\in D_t,\\
 v=0,&x\in\partial D_t,\\
 v(0,x)=\varepsilon\varphi(x),&x\in D_0.
 \end{cases}

Moreover,
\[
 (\partial_t-\Delta-1+q_\varepsilon)U_\varepsilon
 =(q_\varepsilon-U_\varepsilon)U_\varepsilon\ge0
 \quad\hbox{in }D_t.
\]

One can apply comparison separately on the two time intervals
on which $q_\varepsilon$ is continuous. On each finite time
interval the solutions are bounded. To justify comparison on
the unbounded domain, add
$\delta\mathrm e^{(2N+2)t}(1+|x|^2)$ to $U_\varepsilon$, apply
comparison on $D_t\cap B_R$ for sufficiently large $R$, and
then let $R\to\infty$ and $\delta\downarrow0$. The added
function has nonnegative residual under
$\partial_t-\Delta-1+q_\varepsilon$ and dominates the bounded
solutions on the artificial outer boundary.

To compute $v$, write

$\zeta\downarrow0$. This proves \eqref{eq:small-seed} with the
exact coefficient $c_\varphi$, without requiring any of the
error estimates to be uniform as $\zeta\downarrow0$.

 =\mathrm e^{-2t}\int_\Sigma

\]

\begin{equation}\label{eq:absolute-input-bound}
 j_\varepsilon(t,\theta)

 \le C(1+t)t^{-1/2}
       \exp[-\delta^2t-d_0^2/(4t)],
 \qquad t>0.
\end{equation}
Indeed, the tilted tangential Gaussian has mean $a'+2t\theta'$ for $a$ in the initial support. The expected absolute centered factor is therefore at most $C(1+t+\sqrt t)\le C'(1+t)$.

It follows that, for every $A\ge0$,
\begin{align}\label{eq:projected-source-bounds}
 \mathcal J_{\varepsilon,\theta}(t,\mathbb R)
 &=2\varepsilon j_\varepsilon(t,\theta),\notag\\
 \int_0^\infty\int_\mathbb R(z+A)_+\,
       \mathcal J_{\varepsilon,\theta}(t,\,\mathrm d z)\,\mathrm d t
 &\le2\varepsilon\left[
 A J_\varepsilon(\theta)

       \mathcal J_{\varepsilon,\theta}(t,\mathbb R)\,\mathrm d t
 &\le C_\delta\varepsilon\mathrm e^{-c_\delta S},
 \qquad S\ge1.
\end{align}
The total space--time source mass is $2\varepsilon J_\varepsilon=O_\delta(\varepsilon)$. These estimates involve no division by $|\theta'|$ and are therefore uniform up to the normal direction.

Set $A=(1+\gamma_0)L$ and work on $z>-A$. Let $b_A(t)=\mathfrak f(t,-A)$ be the actual wall trace. Using $Q_\varepsilon\le1$ inside the large ball and \eqref{eq:tilted-gaussian} outside it gives
\begin{equation}\label{eq:wall}
 b_A(t)\le\overline b_A(t):=
 C\varepsilon^{1+\gamma_0}(t+L)^{N-1}
 +C\mathrm e^{-P_0L}\mathrm e^{-ct}.
\end{equation}
The Dirichlet heat kernel and boundary Poisson kernel on this half-line are
\begin{align*}
 D_A(r;z,\zeta)
 &=G_r^1(z-\zeta)-G_r^1(z+\zeta+2A)\ge0,\\
 P_A(r,z)&=\frac{z+A}{r}G_r^1(z+A)\ge0.
\end{align*}

\begin{align}\label{eq:measure-halfline-comparison}
 \mathfrak f(t,z)\le{}&
 \int_0^t\int_{(-A,\infty)}
 D_A(t-s;z,\zeta)
 \mathcal J_{\varepsilon,\theta}(s,\,\mathrm d\zeta)\,\mathrm d s
 \notag\\
 &+\int_0^tP_A(t-s,z)\overline b_A(s)\,\mathrm d s,
 \qquad z>-A.
\end{align}

For completeness, this comparison follows by testing against nonnegative backward Dirichlet heat solutions. The boundary term is the Poisson contribution of the actual trace $b_A$, which is bounded by \eqref{eq:wall}. One may first use positive initial times and finite spatial cutoffs. The Gaussian bound removes the outer cutoff, while \eqref{eq:MB} gives
\[
 0\le M_\varepsilon(s,\theta)
 \le2\varepsilon\int_0^s j_\varepsilon(r,\theta)\,\mathrm d r
 \longrightarrow0
 \qquad(s\downarrow0).
\]
Thus the initial term vanishes. Approximation of nonnegative terminal test functions then gives \eqref{eq:measure-halfline-comparison}, with the kernel integrated directly against the source measure.

Only the source restricted to $(-A,\infty)$ appears in the interior integral. Source at or to the left of the wall affects the half-line through its actual trace, already controlled by \eqref{eq:wall}. For example, the atom in \eqref{eq:normal-projected-source} lies inside the half-line for $s<A/2$ and outside it for $s>A/2$. Its crossing time carries no atom in the time measure.

\begin{align}\label{eq:halfline-kernel-masses}
 \int_{-A}^\infty D_A(r;z,\zeta)\,\mathrm d z
 &=\frac2{\sqrt\pi}
   \int_0^{d/(2\sqrt r)}\mathrm e^{-v^2}\,\mathrm d v
 \le\min\left\{1,\frac d{\sqrt{\pi r}}\right\},
 \notag\\
 \int_{-A}^\infty P_A(r,z)\,\mathrm d z
 &=\frac1{\sqrt{\pi r}}.
\end{align}
Tonelli's theorem applies to these nonnegative integrals, including atomic sources.

 +\varepsilon\mathrm e^{-c_\delta T}\right].
\end{align*}
The wall contribution is bounded by

 \int_0^T
 \frac{\overline b_A(s)}{\sqrt{\pi(T-s)}}\,\mathrm d s
 \le C\left[

 M_\varepsilon(T)\le C\bigl[
 &\varepsilon L/\sqrt T+\varepsilon\mathrm e^{-cT}
 +\varepsilon^{1+\gamma_0}(T+L)^N\notag\\

 \int_T^\infty B_\varepsilon(t)\,\mathrm d t
 =M_\varepsilon(T)
  +2\varepsilon\int_T^\infty j_\varepsilon(t)\,\mathrm d t
 =o(\varepsilon).
\]
This proves both assertions in \eqref{eq:early}.

       g(\omega')\,\mathrm d\omega'

The radial Gaussian envelope controls the remaining longitudinal ranges and supplies an integrable majorant.

 O\bigl(\mathrm e^{-2\lambda t}t^{-2+k\lambda}\bigr).

 r=m(t)+\rho\sqrt t,\qquad

 ={}&\left(\frac{4\pi}{b}\right)^\alpha
       \mathrm e^{-2\lambda t}t^{-1+k\lambda}
       \int_0^\infty\mathrm e^{-\lambda\sqrt t\rho}
       w_P(\log t,\rho,(\eta,0))\,\mathrm d\rho\\
 &+o\bigl(\mathrm e^{-2\lambda t}t^{-1}\bigr).
 \end{aligned}
\end{equation}

For the upper bound, use the angular Gaussian estimate and the radial envelope directly, without imposing $\lambda t\le M$. The factor $\mathrm e^{-\lambda\sqrt t\rho}$ is at most one, and the remaining longitudinal integral is uniformly bounded. Together with the interior estimate, this gives \eqref{eq:tangent-bound}.

Set
\[
 V=\mathrm e^zq,\qquad
 G(u)=\int_0^u g(s)\,\mathrm d s,\qquad
 P(z,V)=\mathrm e^{2z}G(\mathrm e^{-z}V).
\]
The transformed equation is
\[
 V_t-V_{zz}-d(t,z)(V_z-V)+P_V(z,V)=0,
 \qquad
 P_V(z,V)=\mathrm e^zg(\mathrm e^{-z}V)\ge0.
\]
The boundary values become
\[
 V(t,-R(t))=b(t):=\mathrm e^{-R(t)}h_b(t),
 \qquad V(t,R(t))=\alpha R(t).
\]

Define the renormalized energy
\[
 \mathcal F(t)=
 \int_{-R(t)}^{R(t)}
 \left(\frac12V_z^2+P(z,V)\right)\,\mathrm d z
 -\frac12\alpha^2R(t).
\]
Completing the square on the positive half-interval gives
the exact identity
\begin{equation}\label{eq:energy-positive}
 \begin{aligned}
 \mathcal F+\alpha
 ={}&\frac12\int_{-R}^0V_z^2\,\mathrm d z
 +\frac12\int_0^R(V_z-\alpha)^2\,\mathrm d z\\
 &+\int_{-R}^RP(z,V)\,\mathrm d z
 +\alpha(1-V(t,0))
 \ge0.
 \end{aligned}
\end{equation}
Here $V(t,0)=q(t,0)\le1$ and $P\ge0$.

This identity also controls the drift error. On $[0,R]$,
write
\[
 V(z)=V(0)+\alpha z+\int_0^z(V_z-\alpha)\,\mathrm d s.
\]
On $[-R,0]$, use the corresponding expression in terms
of $V_z$. Cauchy--Schwarz and
\eqref{eq:energy-positive} then yield, for $R\ge1$,
\[
 \int_{-R}^R(V_z-V)^2\,\mathrm d z
 \le CR^2(\mathcal F+\alpha+1)+CR^3.
\]

We next differentiate the energy, including the moving
endpoints. At the right boundary,
\[
 V_t(t,R(t))=R'(t)(\alpha-V_z(t,R(t))).
\]
After including the derivative of
$-\alpha^2R/2$, the right endpoint contributes
\[
 -\frac{R'}2(V_z-\alpha)^2
 +R'P(R,\alpha R).
\]
At the left boundary,
\[
 V_t(t,-R(t))=b'(t)+R'(t)V_z(t,-R(t)),
\]
so its contribution is
\[
 -V_zb'-\frac{R'}2V_z^2+R'P(-R,b).
\]
Since $R'>0$, this is bounded above by
\[
 \frac{|b'|^2}{2R'}+R'\mathrm e^{-2R}G(h_b).
\]

All positive boundary remainders are integrable in time.
Indeed, $G(u)\le Cu^3$ on $[0,1]$ gives
\[
 P(R,\alpha R)\le C R^3\mathrm e^{-R},
\]
and
\[
 b'=\mathrm e^{-R}(h_b'-R'h_b).
\]
The boundedness of $h_b'$ and the exponential factors
therefore control the left boundary terms as well.

The interior contribution is
\[
 -\int_{-R}^RV_t^2\,\mathrm d z
 +\int_{-R}^RV_t\,d(t,z)(V_z-V)\,\mathrm d z.
\]
Young's inequality and $|d|\le C/t$ give
\[
 \left|\int_{-R}^RV_t\,d(V_z-V)\,\mathrm d z\right|
 \le\frac12\int_{-R}^RV_t^2\,\mathrm d z
 +Ct^{-2}\int_{-R}^R(V_z-V)^2\,\mathrm d z.
\]
Combining the preceding estimates yields
\begin{equation}\label{eq:energy-dissipation}
 \mathcal F'
 \le-\frac12\int_{-R}^RV_t^2\,\mathrm d z
 +Ct^{2\gamma-2}(\mathcal F+\alpha+1)
 +Ct^{3\gamma-2}
 +C(1+R^3)\mathrm e^{-R}.
\end{equation}
No derivative of $d$ occurs.

The two power errors are integrable because $\gamma<1/3$.
Applying Gronwall's inequality to
$\mathcal F+\alpha+1$, which is positive by
\eqref{eq:energy-positive}, gives a uniform energy bound.
Integrating \eqref{eq:energy-dissipation} and using the
lower bound $\mathcal F\ge-\alpha$ then gives
\[
 \int_{t_1}^\infty\int_{-R(t)}^{R(t)}
 V_t^2\,\mathrm d z\,\mathrm d t<\infty
\]
for any fixed starting time $t_1$ after parabolic
regularization.

Consider a sequence $t_j\to\infty$. On each fixed
space--time cylinder, the translated functions
$V(t_j+s,z)$ are uniformly bounded because
$0\le V\le\mathrm e^z$. Local parabolic compactness gives a
subsequential limit. The total dissipation on every
such translated cylinder tends to zero, so the limit
is independent of $s$. Since $d(t_j+s,z)\to0$ uniformly
on compact sets, the limiting equation is
\[
 V_\infty''=\mathrm e^zg(\mathrm e^{-z}V_\infty)\ge0.
\]
The passage to this equation uses local energy bounds
and does not require uniform derivative bounds for $d$.

Lower semicontinuity in \eqref{eq:energy-positive} gives
\[
 \int_0^\infty(V_\infty'-\alpha)^2\,\mathrm d z<\infty.
\]
Convexity makes $V_\infty'$ nondecreasing. Its square
integrability relative to $\alpha$ therefore forces
\[
 V_\infty'(z)\longrightarrow\alpha,\qquad
 V_\infty(z)=\alpha z+o(z)
 \quad(z\to+\infty).
\]
In particular, the limiting profile is nonzero.

Set $Q=\mathrm e^{-z}V_\infty$. Then
\[
 Q''+2Q'+f(Q)=0,\qquad
 0\le Q\le1,\qquad Q(+\infty)=0.
\]
Its mechanical energy
\[
 \mathcal H(z)=
 \frac12Q'(z)^2+\int_0^{Q(z)}f(s)\,\mathrm d s
\]
satisfies $\mathcal H'=-2Q'^2$.
Boundedness of the entire solution and the ODE give
bounded derivatives. The limiting sets at either end
therefore consist of equilibria, as follows from the
energy identity; since the only zeros of $f$ in $[0,1]$
are $0$ and $1$, the end limits exist and belong to
$\{0,1\}$.

The left limit cannot be zero. Otherwise the mechanical
energy would have the same value zero at both ends,
forcing $Q'\equiv0$, contrary to the nonzero
right-hand tail. Hence $Q(-\infty)=1$.
Uniqueness for the ODE gives $0<Q<1$ at finite $z$.
At any critical point,
\[
 Q''=-f(Q)<0.
\]
Thus every critical point would be a strict local
maximum. The end limits $1$ and $0$, together with
$0<Q<1$, exclude such a point, so $Q$ is strictly
decreasing.

Wave uniqueness now identifies $Q$ as a translate of
$U_*$. Its leading-edge coefficient is $\alpha$, because
$\mathrm e^zQ(z)=V_\infty(z)=\alpha z+o(z)$. The normalization
of $U_*$ therefore gives
\[
 Q(z)=U_*(z-\log\alpha).
\]
Every subsequential limit is the same, proving local
uniform convergence of the full solution.

The energy calculations may be started at a positive
time after the initial boundary corners. Thus no
compatibility between the bounded initial profile
and the boundary data is needed.
