<!-- lemma-1: body_md -->
\label{lem:gaussian-average}
Let $f\in L^1(\mathbb{T}^{d_2})$ and extend it $\mathbb{Z}^{d_2}$-periodically onto $\mathbb{R}^{d_2}$.  Then,
we have
\begin{equation}
 \lim_{\varepsilon\to 0}
 \varepsilon^{d_2/2}\int_{\mathbb{R}^{d_2}}f(x_2)L_\varepsilon (x_2)\, dx_2
 =\int_{\mathbb{T}^{d_2}}f(x_2)\, dx_2.
 \label{eq:gaussian-average}
\end{equation}



\noindent
Moreover, given $0<r<\infty$ and $F\in L^r(\mathbb{R}^{d_1}\times\mathbb{T}^{d_2})$, we have
\begin{equation}
 \lim_{\varepsilon\to 0}
 \varepsilon^{d_2/(2r)}
 \|F(x_1, x_2)L_{\varepsilon/r}(x_2)\|_{L^r(\mathbb{R}^{d_1}\times\mathbb{R}^{d_2})}
 =\|F\|_{L^r(\mathbb{R}^{d_1}\times\mathbb{T}^{d_2})},
 \label{eq:norm-average}
\end{equation}

\noindent
where $F$ on the left-hand side
is the periodic extension
\textup{(}in the second variable $x_2$\textup{)}
of $F$ onto $\mathbb{R}^{d_1}\times\mathbb{R}^{d_2}$
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Using the periodicity of $f$ and Fubini's  theorem, we get
\[
\varepsilon^{d_2/2}\int_{\mathbb{R}^{d_2}}f(x_2)L_\varepsilon(x_2)\, dx_2
 =\int_{[0, 1)^{d_2}} f(x_2)P_\varepsilon(x_2)\, dx_2,
\]
where $P_\varepsilon$ is given by
$$
P_\varepsilon (x_2)=\varepsilon^{d_2/2}\sum_{k\in\mathbb{Z}^{d_2}}L_\varepsilon (x_2+k).
$$
The Poisson summation formula \eqref{Poisson}
with the invariance of the Gaussian under the Fourier transform then yields
\[
P_\varepsilon(x_2)=\sum_{n\in\mathbb{Z}^{d_2}}L_{1/\varepsilon}(n)e^{2\pi i n\cdot x_2}.
\]
Hence, as $\varepsilon\to 0$, we have
\[
 \sup_{x_2\in [0, 1)^{d_2}}\left|P_\varepsilon(x_2)-1\right|
 \leq\sum_{n\in\mathbb{Z}^{d_2}\setminus\{0\}} L_{1/\varepsilon}(n)\longrightarrow 0,
 \]

\noindent
where the last step follows from \eqref{L1}.
This
 proves \eqref{eq:gaussian-average}.
 As for \eqref{eq:norm-average},
 given $F \in L^r( \mathbb{R}^{d_1}\times\mathbb{T}^{d_2})$,
 simply define $f \in L^1( \mathbb{T}^{d_2})$ by
\[
 f(x_2)=\int_{\mathbb{R}^{d_1}}|F(x_1, x_2)|^r\, dx_1,\quad  x_2\in\mathbb{T}^{d_2},
\]

\noindent
and apply
\eqref{eq:gaussian-average} to $f$.

<!-- lemma-2: body_md -->
\label{lem:resonance}
Let $m\in\mathbb N$ and $\alpha_0, \alpha_1,\ldots, \alpha_m>0$ such that
\(
\displaystyle\sum_{j=0}^m \alpha_j=1.
\)
For $x_2\in\mathbb{R}^{d_2}$ and $\boldsymbol y\in\mathbb{R}^{md_2}$, define
\begin{align}
 \Gamma_{\varepsilon, x_2}(\boldsymbol y)
 =\varepsilon^{d_2/2}G_{\varepsilon, \alpha_0}
(x_2- y_{1\cdots m})
\prod_{j=1}^m G_{\varepsilon, \alpha_j}(y_j).
 \label{G2}
\end{align}


\noindent
Then, we have
\begin{equation}
 \int_{\mathbb{R}^{md_2}}\Gamma_{\varepsilon, x_2}(\boldsymbol y)\, d\boldsymbol y
 =G_{\varepsilon,  \alpha_{0\cdots m}}(x_2)=L_{1/\varepsilon}(x_2).
\label{G2a}
\end{equation}


\noindent
In \eqref{G2} and \eqref{G2a}, we used the shorthand notation \eqref{sum1}
and \eqref{sum2}.

<!-- proposition-1: body_md -->
\label{LEM:multitrans1}
Let $m\in\mathbb N$ and $d, d_1,d_2\in\mathbb{N}$ with $d=d_1+d_2$.
Given a bounded and continuous function $\sigma: \mathbb{R}^{md_1}\times\mathbb{R}^{md_2}\to\mathbb C$,
let $T_\sigma$ and $S_\sigma$ be as in \eqref{T1} and \eqref{T2}, respectively.
If $K_{T_\sigma} \in L^1(\mathbb{R}^{md_1}\times\mathbb{R}^{md_2})$, then $K_{S_\sigma}\in
L^1(\mathbb{R}^{md_1}\times\mathbb{T}^{md_2})$
with the  bound\textup{:}
\begin{align}
\|K_{S_\sigma}\|_{L^1(\mathbb{R}^{md_1}\times\mathbb{T}^{md_2})}
\le
\|K_{T_\sigma}\|_{L^1(\mathbb{R}^{md})} < \infty.
\label{multitrans1}
\end{align}

\noindent
In particular,
 given any  \(1\leq p, p_1,\ldots, p_m<\infty\)
satisfying
\begin{align}
\displaystyle\frac{1}{p}=\displaystyle\sum_{j=1}^m \displaystyle\frac{1}{p_j},
\label{T3}
\end{align}
the operator  $S_\sigma$ is bounded from $L^{p_1}(\mathbb{R}^{d_1}\times\mathbb{T}^{d_2})\times\cdots \times L^{p_m}(\mathbb{R}^{d_1}\times\mathbb{T}^{d_2})$ to $L^p(\mathbb{R}^{d_1}\times\mathbb{T}^{d_2})$
with the following operator bound\textup{:}
\begin{align}
\| S_\sigma\|_{L^{p_1}\times\cdots\times L^{p_m}\to L^{p}}
\le
\|K_{T_\sigma}\|_{L^1(\mathbb{R}^{md})}.
\label{multitrans2}
\end{align}
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
Let $K_{T_\sigma}\in L^1(\mathbb R^{md_1}\times
\mathbb R^{md_2})$ be the convolution kernel of $T_\sigma$. Define its
periodization in the second variable by
\begin{align*}
    \widetilde K(\boldsymbol x,\boldsymbol y)
    = \sum_{\boldsymbol k\in\mathbb Z^{md_2}}K_{T_\sigma}(\boldsymbol x,\boldsymbol y+\boldsymbol k),
    \qquad
    (\boldsymbol x,\boldsymbol y)\in\mathbb R^{md_1}\times\mathbb T^{md_2}.
\end{align*}

\noindent
By the Fubini--Tonelli theorem, the sum converges absolutely for almost
every $(\boldsymbol x,\boldsymbol  y)$, and $\widetilde K\in
L^1(\mathbb R^{md_1}\times\mathbb T^{md_2})$. Moreover,
we have
\begin{align}
\begin{split}
    \|\widetilde K\|_{L^1(\mathbb R^{md_1}\times\mathbb T^{md_2})}
    &\le
    \int_{\mathbb R^{md_1}}\int_{\mathbb T^{md_2}}
    \sum_{k\in\mathbb Z^{md_2}}
    |K_{T_\sigma}(\boldsymbol x,\boldsymbol y+\boldsymbol k)|\,d\boldsymbol yd\boldsymbol x \\
    &=
    \int_{\mathbb R^{md_1}}\int_{\mathbb R^{md_2}}
    |K_{T_\sigma}(\boldsymbol x,\boldsymbol y)|\,d\boldsymbol yd\boldsymbol x \\
    &=
    \|K_{T_\sigma}\|_{L^1(\mathbb R^{md})}.
\end{split}
\label{Z1}
\end{align}

We claim that $\widetilde K=K_{S_\sigma}$. Indeed, for every
$n\in\mathbb Z^{md_2}$, the $\boldsymbol n$th Fourier coefficient of
$\widetilde K$ in the periodic variable is given, for almost every
$\boldsymbol  x\in\mathbb R^{md_1}$, by
\begin{align*}
    \int_{\mathbb T^{md_2}}\widetilde K(\boldsymbol x,\boldsymbol y)e^{-2\pi i \boldsymbol n\cdot \boldsymbol y}\,d\boldsymbol y
    &=
    \int_{\mathbb R^{md_2}}
    K_{T_\sigma}(\boldsymbol x,\boldsymbol y)e^{-2\pi i \boldsymbol n\cdot \boldsymbol y}\,d\boldsymbol y\\
    &=
    \mathcal F_{\mathbb R^{md_2}}^{(2)}
    (K_{T_\sigma})(\boldsymbol x,\boldsymbol n).
\end{align*}
Taking the Fourier transform also in
$\boldsymbol x\in\mathbb R^{md_1}$, we obtain
\begin{align*}
    \mathcal F_{\mathbb R^{md_1}}
    \left[
    \int_{\mathbb T^{md_2}}
    \widetilde K(\,\cdot\,,\boldsymbol y)e^{-2\pi i \boldsymbol n\cdot \boldsymbol y}\,d\boldsymbol y
    \right](\boldsymbol \xi)
    =
    \sigma(\boldsymbol \xi,\boldsymbol n).
\end{align*}


\noindent
Thus,  $\widetilde K$ has Fourier transform $\sigma(\boldsymbol \xi,\boldsymbol n)$ on
$\mathbb R^{md_1}\times\mathbb Z^{md_2}$ and
hence
$\widetilde K=K_{S_\sigma}$.
Therefore,
\eqref{multitrans1} follows from \eqref{Z1}.
The bound \eqref{multitrans2} then follows from \eqref{multitrans1} and \eqref{T0}.

 \left(1+\frac{2}{K}\right)^{d_2/p}
 \prod_{j=1}^m
 \|F_j\|_{L^{p_j}(\mathbb{R}^{d_1}\times\mathbb{T}^{d_2})}.
\end{equation}


\noindent
Hence,
from \eqref{eq:norms-1stesti} and \eqref{eq:product-limsup}, we obtain
\begin{equation}
 \lvert\Omega_\lambda\rvert
 \leq
 \left(1+\frac{2}{K}\right)^{d_2}
 \left(\frac{B\prod_{j=1}^m
 \|F_j\|_{L^{p_j}(\mathbb{R}^{d_1}\times\mathbb{T}^{d_2})}
 }{\lambda-\varepsilon}\right)^p.
\label{S12}
\end{equation}
Finally, by taking $K\to\infty$ and $\varepsilon\to 0$ in \eqref{S12},
we obtain
\begin{align}
\lvert\Omega_\lambda\rvert\leq\left(\frac{B\prod_{j=1}^m
 \|F_j\|_{L^{p_j}(\mathbb{R}^{d_1}\times\mathbb{T}^{d_2})}}{\lambda}\right)^p.
\label{S13}
\end{align}


\noindent
Therefore, the bound
\eqref{S1}
follows from
\eqref{S9a}, \eqref{S13}, and \eqref{S2}
with the density argument.
