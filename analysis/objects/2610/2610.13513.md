First we show the $\Gamma$-$\limsup$ inequality, i.e. for any $\rho_0\in H^1(\mathbb{R}^d)$ there exists a sequence of functions  $\rho_\varepsilon$ vanishing on $\mathbb{R}^d\setminus \frac{1}{\sqrt{\varepsilon}}\Omega$, converging to $\rho_0$ in  $L^2(\mathbb{R}^d)$, and such that
		\begin{equation}
		\lim_{\varepsilon\to 0} F_\varepsilon(\rho_\varepsilon)= F_0(\rho_0). 
		\label{3.7}\quad                                               
		                                           
	\end{equation}
It sufficies to prove \eqref{3.7} in the case when $\rho_0\in C_c^\infty(\mathbb{R}^d)$ thanks to 

	\begin{lem} 
	For any                      
	function $\rho_\varepsilon\in H^1_0\big(\frac{1}{\sqrt{\varepsilon}}\Omega\big)$
                      
the following bound holds  
	\begin{equation*}
		\begin{aligned}	
		\frac{1}{2\varepsilon^{d/2+1}} \int\limits_{\mathbb{R}^d}
			\int\limits_{\mathbb{R}^d}
			J\left(\frac{|\xi-\tilde \xi|^2}{\varepsilon}\right)
			(\rho_\varepsilon(\xi)-\rho_\varepsilon(\tilde \xi))^2d\xi d\tilde \xi \, \leq \, \frac{\vartheta}{2}\int\limits_{\mathbb{R}^d} |\nabla\rho_\varepsilon(\xi)|^2 \, d \xi.
	\end{aligned}
	\end{equation*}
         
                  
                 
		                                                                                
	\label{EP}  
\end{lem}
\begin{proof}
First we perform the change of variables $\xi-\tilde \xi=\sqrt{\varepsilon}\tilde\zeta$ and get
	\begin{equation}
	\begin{aligned}	
		\frac{1}{2\varepsilon^{d/2+1}}&\int\limits_{\mathbb{R}^d}
		\int\limits_{\mathbb{R}^d}
		J\left(\frac{|\xi-\tilde \xi|^2}{\varepsilon}\right)
		(\rho_\varepsilon(\xi)-\rho_\varepsilon(\tilde \xi))^2 d\xi d\tilde \xi \\
		&=\frac{1}{2\varepsilon}\int\limits_{\mathbb{R}^d} d\tilde\xi 
		\int\limits_{\mathbb{R}^d}
		J\left(|\tilde \zeta|^2\right)
		(\rho_\varepsilon(\tilde\xi+\sqrt{\varepsilon}\tilde\zeta)-\rho_\varepsilon(\tilde \xi))^2	d\tilde \zeta. 
		\label{3.6} 
	\end{aligned}
\end{equation}
Representing $\rho_\varepsilon(\tilde\xi+\sqrt{\varepsilon}\tilde\zeta)-\rho_\varepsilon(\tilde \xi)=
\sqrt{\varepsilon}\int\limits_0^1|\nabla\rho_\varepsilon(\tilde\xi+\sqrt{\varepsilon}t\tilde\zeta)|^2 \, dt $
 and applying the Cauchy-Schwarz inequality we obtain   
$|\rho_\varepsilon(\tilde\xi+\sqrt{\varepsilon}\tilde\zeta)-\rho_\varepsilon(\tilde \xi)|^2\leq
\varepsilon |\tilde \zeta|^2 \int\limits_0^1|\nabla\rho_\varepsilon(\tilde\xi+\sqrt{\varepsilon}t\tilde\zeta)|^2 \, dt $. 
Then we estimate the right-hand side of \eqref{3.6} as follows 
\begin{equation*}
	\begin{aligned}
		\frac{1}{2\varepsilon}&\int\limits_{\mathbb{R}^d} d\tilde\xi 
		\int\limits_{\mathbb{R}^d}
		J\left(|\tilde \zeta|^2\right)
		(\rho_\varepsilon(\tilde\xi+\sqrt{\varepsilon}\tilde\zeta)-\rho_\varepsilon(\tilde \xi))^2	d\tilde \zeta \\
	&	\leq	
		\frac{1}{2}\int\limits_0^1 dt 
		\int\limits_{\mathbb{R}^d}
J\left(|\tilde\zeta|^2\right) |\tilde \zeta|^2 d\tilde \zeta 
\int\limits_{\mathbb{R}^d} |\nabla\rho_\varepsilon(\tilde\xi+\sqrt{\varepsilon}t\tilde\zeta)|^2 \, d\tilde \xi=	\frac{\vartheta}{2}
\int\limits_{\mathbb{R}^d} |\nabla\rho_\varepsilon(\xi)|^2 \, d \xi.
\end{aligned}
\end{equation*}
	\end{proof}

{\it Proof of Theorem \ref{Th31} (continued).} Given $\rho_0\in C_c^\infty(\mathbb{R}^d)$, 
                                                                            
we set $\rho_\varepsilon(\xi)=\rho_0(\xi)$ when $\xi\in \frac{1}{\sqrt{\varepsilon}}\Omega $ and $\rho_\varepsilon(\xi)=0$ otherwise. Notice that for sufficiently small $\varepsilon$ we have $\rho_\varepsilon(\xi)=\rho_0(\xi)$ on $\mathbb{R}^d$, then
introducing $\zeta=\frac{1}{\sqrt{\varepsilon}}(\tilde\xi-\xi)$ and expanding  
\begin{equation*}
	\rho_0(\tilde\xi)-\rho_0(\xi)=\rho_0(\xi+\sqrt{\varepsilon}\zeta)-\rho_0(\xi)=\sqrt{\varepsilon}\nabla\rho_0(\xi)\cdot \zeta +              
\varepsilon\int\limits_0^1	
	\partial^2_{\xi_j \xi_k}  \rho_0(\xi+\sqrt{\varepsilon}t\zeta)\zeta_j\zeta_k(1-t)dt
\end{equation*}	
 we derive the asymptotic formula for the first term of the functional \eqref{funcFve} 
	\begin{equation*}
	\begin{aligned}	
		\frac{1}{2\varepsilon^{d/2+1}}&\int\limits_{\mathbb{R}^d}
		\int\limits_{\mathbb{R}^d}
		J\left(\frac{|\xi-\tilde \xi|^2}{\varepsilon}\right)
		(\rho_\varepsilon(\xi)-\rho_\varepsilon(\tilde \xi))^2 d\xi d\tilde \xi \\ 
		& = 	\frac{1}{2d}\int\limits_{\mathbb{R}^d}

		 	d\xi +O(\sqrt{\varepsilon}).
	\end{aligned}
\end{equation*}
The limit transition in second term of  \eqref{funcFve} is straightforward. 

It remains to demonstrate the $\Gamma$-$\liminf$ inequality, i.e. 
\begin{equation}
	\liminf_{\varepsilon\to 0} F_\varepsilon(\rho_\varepsilon)\geq F_0(\rho_0), \quad \text{for any } \rho_\varepsilon\to \rho_0\ \text{in}
	\ L^2(\mathbb{R}^d), \text{as} \  \varepsilon\to 0.
	\label{3.12}
\end{equation}
Thanks to Lemma \ref{EP} it is sufficient to establish \eqref{3.12} for $\rho_0\in C_c^\infty(\mathbb{R}^d)$. 
To this end we write
	\begin{equation}
	\begin{aligned}	
	\frac{1}{2\varepsilon^{d/2+1}}&\int\limits_{\mathbb{R}^d}
\int\limits_{\mathbb{R}^d}
J\left(\frac{|\xi-\tilde \xi|^2}{\varepsilon}\right)
(\rho_\varepsilon(\xi)-\rho_\varepsilon(\tilde \xi))^2 d\xi d\tilde \xi \\ 
& =	\frac{1}{2\varepsilon^{d/2+1}}\int\limits_{\mathbb{R}^d}
\int\limits_{\mathbb{R}^d}
J\left(\frac{|\xi-\tilde \xi|^2}{\varepsilon}\right)
(\rho_\varepsilon(\xi)-\rho_0(\xi)+\rho_0(\tilde\xi)-\rho_\varepsilon(\tilde \xi))^2 d\xi d\tilde \xi \\ 
& +		\frac{1}{2\varepsilon^{d/2+1}}\int\limits_{\mathbb{R}^d}
\int\limits_{\mathbb{R}^d}
J\left(\frac{|\xi-\tilde \xi|^2}{\varepsilon}\right)
\big(\rho_0(\xi)-\rho_0(\tilde\xi)\big)^2 d\xi d\tilde \xi\\
&+	\frac{2}{\varepsilon^{d/2+1}}\int\limits_{\mathbb{R}^d}
\int\limits_{\mathbb{R}^d}
J\left(\frac{|\xi-\tilde \xi|^2}{\varepsilon}\right)
\big(\rho_0(\xi)-\rho_0(\tilde\xi)\big)d\tilde \xi \,  \big(\rho_\varepsilon(\xi)-\rho_0(\xi)\big)d\xi.
\label{3.14}
	\end{aligned}
\end{equation}
The first term in the right hand side of \eqref{3.14} is non-negative; the second term has already been dealt with in the proof of $\Gamma$-$\limsup$ inequality; finally, the last term tends to zero since $\rho_\varepsilon-\rho_0\to 0$ in $L^2(\mathbb{R}^d)$ while 	$\frac{2}{\varepsilon^{d/2+1}}
\int\limits_{\mathbb{R}^d}
J\left(\frac{|\xi-\tilde \xi|^2}{\varepsilon}\right)
(\rho_0(\xi)-\rho_0(\tilde\xi))d\tilde \xi$ remains bounded in $L^2(\mathbb{R}^d)$  as $\varepsilon\to 0$. Indeed, 
arguing as in the proof of $\Gamma$-$\limsup$ inequality, we can write 
	\begin{equation*}
		\frac{2}{\varepsilon^{d/2+1}}\int\limits_{\mathbb{R}^d}
		J\left(\frac{|\xi-\tilde \xi|^2}{\varepsilon}\right)
		\big(\rho_0(\xi)-\rho_0(\tilde\xi)\big)d\tilde \xi
			=-2 \int\limits_0^1	\int\limits_{\mathbb{R}^d} 
		J\left(| \zeta|^2\right) 	\partial^2_{\xi_j \xi_k}  \rho_0(\xi+\sqrt{\varepsilon}t\zeta)\zeta_j\zeta_k
		d\zeta dt.
		                                                            
\end{equation*}
Then the sought boundedness follows from the facts that 
$\int\limits_{\mathbb{R}^d} J(|\zeta^2) |\zeta|^2  d\zeta< \infty$ and $\rho_0\in C_c^\infty(\mathbb{R}^d)$.

	\int\limits_{-\infty}^{\infty}
	I_0\big(\xi_m-\tilde\xi\big)U(\xi_m)d\tilde\xi=U(\xi_m)=\int\limits_{0}^{\infty}
	I_0\big(\xi_m-\tilde\xi\big)U(\tilde\xi)d\tilde\xi 
\end{equation*}

\begin{equation}
	\int\limits_{-\infty}^{0}
	I_0\big(\xi_m-\tilde\xi\big)U(\xi_m)d\tilde\xi=\int\limits_{0}^{\infty}
	I_0\big(\xi_m-\tilde\xi\big)\Big(U(\tilde\xi)-U(\xi_m)\Big)d\tilde\xi. 	\label{cor_p4}
\end{equation}
Since  $I_0(z)\geq 0$,  the left-hand side of \eqref{cor_p4} is nonpositive while the right-hand side
