<!-- theorem-1: body_md -->
\label{thm:main}
		Let $\mathcal O$ be an octonion algebra over a field $F$ with $\operatorname{char}F\neq2$, and let $A,B\in\mathcal O$ with $A\neq0$. Consider the map
		$$
		\Phi:\mathcal O\times\mathcal O\longrightarrow\mathcal O,
		\qquad
		\Phi(x,y)=A(xy)-B(yx).
		$$
		Then $\Phi$ is surjective if and only if one of the following conditions holds:
		\begin{enumerate}[label=(\alph*)]
			\item at least one of $A$ and $B$ is invertible, and $A\neq B$;
			\item $A$ and $B$ are nonzero zero divisors and $N(A,B)\neq0$.
		\end{enumerate}
		In the remaining cases, the following statements hold:
		\begin{enumerate}[label=(\roman*)]
			\item if $A=B$ is invertible, then $\operatorname{Im}(\Phi)=A^\perp=\{z\in\mathcal O\,|\,N(A,z)=0\}$;
			\item if $A$ is a nonzero zero divisor and $B\in FA$, then $\operatorname{Im}(\Phi)=A\mathcal O$, which is a four-dimensional maximal totally isotropic subspace of $\mathcal O$;
			\item if $A$ and $B$ are linearly independent nonzero zero divisors with $N(A,B)=0$, then the linear span of $\operatorname{Im}(\Phi)$ has dimension $6$.
		\end{enumerate}
<!-- end theorem-1 -->
<!-- definition-1: body_md -->
A \emph{composition algebra} is a pair $(\mathcal{C},N)$, where $\mathcal{C}$ is a unital algebra over $F$ and $N:\mathcal{C}\to F$ is a non-degenerate quadratic form satisfying
		$$N(xy)=N(x)N(y)$$
		for all $x,y\in\mathcal{C}$. The quadratic form $N$ is called the \emph{norm} of $\mathcal{C}$.
<!-- end definition-1 -->
<!-- lemma-1: body_md -->
\label{lem: identities}
		For all $a,b,c \in\mathcal{C}$, the following identities hold:
		\begin{enumerate}
			\item $a\overline{a}=\overline{a}a=N(a)1$.
			\item $\overline{\overline{a}}=a$.
			\item The element $a$ is invertible if and only if $N(a)\neq 0$. In this case, $a^{-1}=N(a)^{-1}\overline{a}$.
			\item $a(\overline{a}b)=N(a)b$. \label{eqn: identity 4}
			\item $ a(ab)=a^2b,\qquad (ab)b=ab^2,\qquad (ab)a=a(ba)$.
			\item $a^2-Tr(a)a+N(a)1 = 0$
			\item $N(ca,b) = N(a,\overline{c}b)$ and $N(ac,b) = N(a,b\overline{c})$, 
		\end{enumerate}
		The identities in $(5)$ are referred to as the \emph{alternative laws}. The identity $(6)$ is referred to as the \emph{Cayley-Hamilton identity}.
<!-- end lemma-1 -->
<!-- proposition-1: body_md -->
\label{prop: Cayley-Dickson}
		Let $(\mathcal C, N)$ be a composition algebra over a field $F$. Suppose $\mathcal C_0 \subseteq \mathcal C$ is a composition proper subalgebra and let $\omega \in \mathcal C_0^\perp$ be an element such that $N(\omega) \neq 0$.
		Then the space $$\mathcal C_1=\mathcal C_0\oplus \mathcal C_0 \omega$$ forms a composition subalgebra of $\mathcal C$ with $\mathrm{dim}(\mathcal C_1)=2\cdot \mathrm{dim}(\mathcal C_0)$. Moreover, for $a,a',b,b'\in \mathcal C_0$, the operations on $\mathcal C_1$ are given by:
		\begin{align*}
			\text{Multiplication:}\quad & (a+b\omega)(a'+b'\omega)=\big(aa'+ \delta\,\overline{b'}\, b\big)+\big( b'a+b\overline{a'}\big) \omega,\\
			\text{Norm:}\quad           & N(a+b\omega)=N(a)-\delta\,N(b),\\
			\text{Conjugation:}\quad    & \overline{a+b\omega}=\overline{a}-b\omega,
		\end{align*}
		where $\overline{\cdot}$ denotes the canonical involution on $\mathcal C_0$ and $\delta = -\,N(\omega)$.
<!-- end proposition-1 -->
<!-- lemma-2: body_md -->
\label{lem: N(A-B) nonzero}
		Let $A,B\in\mathcal O$ such that $N(A-B)\neq0$. Then the map $\Phi(x,y)=A(xy)-B(yx)$ is surjective on $\mathcal O$.
<!-- end lemma-2 -->
<!-- proof-1: body_md -->
Let $Z\in\mathcal O$ be arbitrary. Taking $y=1_{\mathcal O}$, we have $\Phi(x,1_{\mathcal O})=A(x)-B(x)=(A-B)x$. Since $N(A-B)\neq0$, the element $A-B$ is invertible. Hence, taking
		$x=(A-B)^{-1}Z$ and using equation~(\ref{eqn: identity 4}), we obtain
		$\Phi(x,1_{\mathcal O})=Z$. Thus, every $Z\in\mathcal{O}$ lies in the image of $\Phi$, and consequently $\Phi$ is surjective.
<!-- end proof-1 -->
<!-- lemma-3: body_md -->
\label{lem:sandwich}
		For all $A,B\in\mathcal O$,
		$$
		(A-B)\mathcal O\ \subseteq\ \operatorname{Im}(\Phi)\ \subseteq\ A\mathcal O+B\mathcal O.
		$$
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
For the right inclusion, $\Phi(x,y)=A(xy)-B(yx)\in A\mathcal O+B\mathcal O$ for all $x,y$, since $xy,yx\in\mathcal O$. For the left inclusion, take $y=1_{\mathcal O}$: $\Phi(x,1_{\mathcal O})=A(x)-B(x)=(A-B)x$, so $(A-B)\mathcal O\subseteq\operatorname{Im}(\Phi)$.
<!-- end proof-2 -->
<!-- lemma-4: body_md -->
\label{lem: B is zero}
		Let $A\in\mathcal{O}$ be nonzero. Consider the map
		$\Phi:\mathcal{O}\times\mathcal{O}\longrightarrow\mathcal{O}$ defined by $\Phi(x,y)=A(xy)$.
		Then $\operatorname{Im}(\Phi)=A\mathcal{O}$.
		In particular, if $A$ is invertible, then $\Phi$ is surjective.
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
Since $1_{\mathcal{O}}$ is the identity element of $\mathcal{O}$, every $z\in\mathcal{O}$ can be written as $z=z1_{\mathcal{O}}$. Hence
		$$\{xy:x,y\in\mathcal{O}\}=\mathcal{O}.$$
		Therefore,
		$$\operatorname{Im}(\Phi)=\{A(xy):x,y\in\mathcal{O}\}=\{Az:z\in\mathcal{O}\}=A\mathcal{O}.$$
		If $A$ is invertible, then left multiplication by $A$ is a bijection of $\mathcal{O}$. Thus $A\mathcal{O}=\mathcal{O}$,
		and consequently $\Phi$ is surjective.
<!-- end proof-3 -->
<!-- proposition-2: body_md -->
\label{Image_of_L_A}
		Let $A\in \mathcal O$ be a nonzero zero divisor, i.e., $A\neq 0$ and $N(A) = 0$. Then
		\begin{enumerate}
			\item $\mathrm{Im}(L_A)$ is a totally isotropic subspace of $\mathcal O$, i.e., $N(x,y) = 0$ for any $x,y\in \mathrm{Im}(L_A)$.
			\item $L_A$ satisfies $(L_A)^2 = Tr(A)L_A$
			\item $\dim \mathrm{Im}L_A = 4$.
			\item If $Tr(A) = 0$, then $\mathrm{Im}(L_A) = \ker(L_A)$
			\item If $Tr(A) \neq 0$, then $\mathcal O = \mathrm{Im}(L_A)\oplus \ker(L_A)$, with both subspaces being maximal isotropic.
		\end{enumerate}
<!-- end proposition-2 -->
<!-- proof-4: body_md -->
Items (1) and (3) are known: by \cite[Theorem~3]{MR123622} (see also \cite[Theorem~3.6(b)]{MR4812310}), $\mathrm{Im}(L_A) = A\mathcal O$ is a maximal totally isotropic subspace of $\mathcal O$, and $\dim \mathrm{Im}(L_A) = 4$ by \cite[Lemma~1]{Gow2008}. Item (1) also follows directly from the multiplicativity of the norm: for $x\in \mathcal O$, $N(Ax) = N(A)N(x) = 0$, and since $N(x,y) = N(x+y)-N(x)-N(y)$, we obtain that $N(x,y) = 0$ for any $x,y\in \mathrm{Im}L_A$.
		
		(2) Since $N(A) = 0$, the Cayley-Hamilton identity gives $A^2 = Tr(A)A$. Multiplying by $x$ and using the alternative laws, we obtain $A(Ax) = Tr(A)(Ax)$, for any $x\in \mathcal O$, which means $L_A^2 = Tr(A) L_A$.
		
		(4) If $Tr(A) = 0$, $L_A^2 = 0$, then $Im(L_A)\subseteq \ker(L_A)$. By (3), $\dim \mathrm{Im}(L_A) = 4$ and  by rank–nullity, $8 = \dim \mathcal O = \dim \mathrm{Im}(L_A) + \dim \ker(L_A)$, and we obtain $\dim \ker(L_A) = 4$ and this implies equality.
		
		
		(5) Now we assume $Tr(A) \neq 0$. Then $f = A/Tr(A)$ satisfy $Tr(f) = 1$. Also, $f$ is an idempotent of $\mathcal O$, that is $f^2 = f$ and using the alternative laws, $L_f^2 = L_f$. As a consequence, $\mathcal O = \mathrm{Im}(L_f)\oplus \ker(L_f) = \mathrm{Im}(L_A)\oplus \ker(L_A)$. Notice that $\overline{f} = Tr(f)1-f = 1-f$ and $N(1-f) = N(1) + N(f) - N(1,f)$. Now it follows from the facts that $N(f) =0$, $N(1) = 1$ and $N(1,f) = Tr(f) = 1$ that $N(1-f) = 0$, that is, $1-f$ is a nonzero zero divisor. By item (4) of Lemma \ref{lem: identities}, $f((1-f)y) = f(\overline{f}y) = N(f)y = 0$ for all $y\in \mathcal O$, so $\mathrm{Im}(L_{1-f})\subseteq \ker(L_f)$. By (3) and rank–nullity, both subspaces have dimension $4$, hence $\ker(L_A) = \ker(L_f) = \mathrm{Im}(L_{1-f})$ (see also \cite[Theorem~3.6(a)]{MR4812310}). Finally, by \cite[Theorem~3]{MR123622}, both $\mathrm{Im}(L_A)$ and $\ker(L_A) = \mathrm{Im}(L_{1-f})$ are maximal totally isotropic subspaces of $\mathcal O$.
<!-- end proof-4 -->
<!-- lemma-5: body_md -->
\label{lem:amitsurrowencommutator}
		Let $R$ be a central simple algebra of degree $2$ over a field $F$ with $\operatorname{char}F\neq2$. Then every element of $R$ having reduced trace zero is a commutator.

<!-- corollary-1: body_md -->
\label{cor: trace zero quaternion}
		Let $Q$ be a quaternion algebra over a field $F$ with $\operatorname{char}F\neq2$. Then every element $q\in Q$ satisfying $\operatorname{Trd}(q)=0$ is a commutator in $Q$.
<!-- end corollary-1 -->
<!-- proposition-3: body_md -->
\label{prop: element octonion to quaternion}
		Every element $a$ of an octonion algebra $\mathcal O$ over $F$ is contained in a quaternion subalgebra of $\mathcal O$. If $\mathcal O$ is split, this quaternion subalgebra may be taken to be split as well.
<!-- end proposition-3 -->
<!-- lemma-6: body_md -->
\label{lem:trace-zero-commutator}
		If $\mathcal O$ is an octonion algebra, then the image of $\Phi(x,y) = xy-yx$ is $\mathcal O_0$, the set of pure octonions.
<!-- end lemma-6 -->
<!-- proof-5: body_md -->
First notice that \[Tr(xy) = N(xy,1) = N(x,\overline{y}) = N(1,\overline{x}\overline{y}) = N(1,\overline{yx}) = N(yx,1) = Tr(yx).\] As a consequence, $Tr(xy-yx) = 0$, and we get the first inclusion.

<!-- remark-1: body_md -->
\label{rem:unified-coverage}
		Let us explain how Propositions~\ref{prop: T contained in quaternion} and~\ref{prop:unified} together cover all the cases of Lemma~\ref{lem: unitary subalgebras}, except the Heisenberg-radical case (4)(c), which is treated in Proposition~\ref{prop:T Heisenberg case} below. 			\item If $\dim_F T\leq 2$, then $T=F+Fc$ for some $c\in T$. By Proposition~\ref{prop: element octonion to quaternion}, $c$ lies in a quaternion subalgebra of $\mathcal O$, and hence so does $T$. So Proposition~\ref{prop: T contained in quaternion} applies.
			\item If $T$ is of type (3)(a), then $T\subseteq F+Fp_0+Fn_0\subseteq\mathcal Q$, and Proposition~\ref{prop: T contained in quaternion} applies with $\mathcal H=\mathcal Q$.
			\item If $T$ is of type (3)(b), then $A=a+r\omega$ and $B=b+s\omega$ with $a,b\in F$ and $r,s\in U$, and Proposition~\ref{prop:unified} applies.

\label{thm:quaternion}
		Let $Q$ be a quaternion algebra over a field $F$ with $\operatorname{char}F\neq2$, and let $A,B\in Q$ with $A\neq0$. Then $\operatorname{Im}(\Phi^Q_{A,B})$ is a vector subspace of $Q$. More precisely:
