<!-- theorem-1: body_md -->
\label{main}
Let $(\mathcal{M},\tau)$ be a von Neumann algebra equipped with a faithful normal semifinite trace. There exists a universal constant $C<\infty$ such that, for every $n\in\mathbb N$ and every $f:\Omega_{n}\to L_1(\mathcal{M},\tau)$,
\begin{equation}\label{eqmain}
\mathbb{E}_{\varepsilon,\delta}
\|f(\varepsilon)-f(\delta)\|_{L_1(\mathcal{M})}
\leq C\sum_{j=1}^n
\bigl\|\,\|\partial_j f\|_{L_1(\mathcal{M})}\,\bigr\|_{\psi_{1,1}}.
\end{equation}
Consequently, $L_1(\mathcal{M},\tau)$ has Talagrand type $(1,\psi_{1,1})$, with a constant independent of the algebra. In particular, the Schatten trace class $S_1$ has Talagrand type $(1,\psi_{1,1})$.
<!-- end theorem-1 -->
<!-- corollary-1: body_md -->
\label{cor:centered}
Under the assumptions of Theorem~\ref{main},
\[
\mathbb{E}\|f-\mathbb{E} f\|_{L_1(\mathcal{M})}
\leq C\sum_{j=1}^n
\bigl\|\,\|\partial_j f\|_{L_1(\mathcal{M})}\,\bigr\|_{\psi_{1,1}}.
\]

<!-- theorem-2: body_md -->
\label{thm;maz}
		Let $\mathcal{M}$ be a von Neumann algebra, for $1\le p,q<\infty$, $\mathbb{M}_{p,q}$ is $\min\left\{\frac{p}{q},1\right\}$-Hölder on the unit ball of $L_p(\mathcal{M})$.
<!-- end theorem-2 -->
<!-- corollary-2: body_md -->
\label{cor:mazur}
		There exist universal constants $A,B<\infty$ such that for any $x,y\in L_1(\mathcal M)$ and any $u,v\in L_2(\mathcal M)$
		\begin{align}
			\left\|\mathbb{M}_{1,2}(x)-\mathbb{M}_{1,2}(y)\right\|_{L_{2}(\mathcal{M})}
			&\leq A\|x-y\|_{L_{1}(\mathcal{M})}^{1/2},\label{A}\\
			\left\|\mathbb{M}_{2,1}(u)-\mathbb{M}_{2,1}(v)\right\|_{L_{1}(\mathcal{M})}
			&\leq B\max\{\left\|u\right\|_{L_{2}(\mathcal{M})},\left\|v\right\|_{L_{2}(\mathcal{M})}\}\,\left\|u-v \right\|_{L_{2}(\mathcal{M})}.\label{B}
		\end{align}
<!-- end corollary-2 -->
<!-- proof-1: body_md -->
Ricard's first theorem in \cite{R2015} gives, on the unit ball of $L_{1}(\mathcal M)$,
		\[\|\mathbb{M}_{1,2}(x)-\mathbb{M}_{1,2}(y)\|_{L_{2}(\mathcal{M})}\leq A\|x-y\|_{L_{1}(\mathcal{M})}^{1/2}.\]
		For arbitrary $x,y$, let $r=\max\{\|x\|_{L_{1}(\mathcal{M})},\|y\|_{L_{1}(\mathcal{M})}\}$. When $r=0$, it is trivial. Since $\mathbb{M}_{1,2}(rx)=r^{1/2}\mathbb{M}_{1,2}(x)$ for $r\geq 0$, applying the unit-ball estimate to $x/r$ and $y/r$ gives \eqref{A} with the same constant.
		
		Likewise, Theorem \ref{thm;maz} says that $\mathbb{M}_{2,1}$ is a Lipschitz operation on the unit ball of $L_{2}(\mathcal M)$. If $u=v=0$, \eqref{B} is immediate. For $s =\max\{\|u\|_{L_{2}(\mathcal{M})},\|v\|_{L_{2}(\mathcal{M})}\}>0$, we have
		\[\mathbb{M}_{2,1}(su)=s^2 \mathbb{M}_{2,1}(u).\]
		Then we can obtain \eqref{B} by
		\begin{align*}
			\|\mathbb{M}_{2,1}(u)-\mathbb{M}_{2,1}(v)\|_{L_{1}(\mathcal{M})}
			&=s^2\left\|\mathbb{M}_{2,1}\left(\frac{u}{s}\right)-\mathbb{M}_{2,1}\left(\frac{v}{s}\right)\right\|_{L_{1}(\mathcal{M})}\\
			&\leq Bs^2\left\|\frac{u-v}{s}\right\|_{L_{2}(\mathcal{M})}\\
			&=Bs\|u-v\|_{L_{2}(\mathcal{M})}.
		\end{align*}
<!-- end proof-1 -->
<!-- theorem-3: body_md -->
Let $(E,\|\cdot\|_E)$ be a Banach space with martingale type $2$.
		Then there exists $C(E)\in(0,\infty)$ such that for every
		$n\in\mathbb{N}$, every function $f:\Omega_{n}\to E$ satisfies
		\begin{equation}
			\|f-{\mathbb{E}} f\|_{L_2(\Omega_{n};E)}^2 \leq C(E)\sum_{i=1}^n\|\partial_i f\|_{L_2(\log L)^{-1}(\Omega_{n};E)}^2.
		\end{equation}
<!-- end theorem-3 -->
<!-- theorem-4: body_md -->
\label{eq:cee-original}
		Let $H$ be an arbitrary Hilbert space. There is a constant $C_{\Xi}<\infty$ depending only on the choice of $\Xi$ in $\|\cdot\|_{L_{\Xi}}$ such that every $f:\Omega_{n}\to H$ satisfies
		\begin{equation}
			\|f-{\mathbb{E}} f\|_{L_2(\Omega_{n};H)}^2 \leq C_{\Xi}\sum_{i=1}^n\|\partial_i f\|_{L_{\Xi}}^2.
		\end{equation}
		where $\Xi$ may be taken to be any Young function equivalent to $ \frac{t^2}{\log(e+t)}$, in particular the function $\Phi$, after changing the universal constant.
<!-- end theorem-4 -->
<!-- remark-1: body_md -->
The constant can be chosen uniformly over all Hilbert spaces since the image of a map from the discrete cube spans a finite-dimensional Hilbert space, which embeds isometrically into a fixed $\ell_2$, so the constant for $\ell_2$ works for every such map. Here Theorem 3.3 applies to $\ell_2$ because it has martingale type $2$; the comparison with the chosen $L_\Xi$ norm contributes a constant depending only on $\Xi$.
<!-- end remark-1 -->
<!-- proof-2: body_md -->
For any $f: \Omega_{n} \to L_{1}\left(\mathcal{M} \right)$ with $p\geq1$, $f(\varepsilon)$ is a $L_{1}\left(\mathcal{M} \right)$-valued random variable. Supposing that $\delta$ is an independent copy of $\varepsilon$, then $f(\delta)$ and $f(\varepsilon)$ are i.i.d. We write $f(\varepsilon)$ briefly as $\mathbf{x}$ and $f(\delta)$ as $\mathbf{x}'$.
		Now we derive a bound for the left-hand side of \eqref{eqmain} via the Mazur map.
		Denote the centered variable by
		\[\mathbf{y}=\mathbf{x}-\mathbb{E} \mathbf{x},\quad \mathbf{y}'=\mathbf{x}'-\mathbb{E} \mathbf{x}',\]
		and those under the $\{1,2\}$ Mazur map by
		\[G=\mathbb{M}_{1,2}(\mathbf{y}), \quad G'=\mathbb{M}_{1,2}(\mathbf{y}').\]
		Since $\mathbb{E}\mathbf{y}=\mathbb{E}\mathbf{y}'=0$ and centering does not change pairwise differences, by Jensen\'s inequality we have
		\begin{equation}\label{eq y}
			\mathbb{E}_{\mathbf{y}}\left\| \mathbf{y}\right\|_{L_{1}(\mathcal{M})} = \mathbb{E}_{\mathbf{y}}\left\| \mathbf{y}-\mathbb{E}\mathbf{y}'\right\|_{L_{1}(\mathcal{M})} \leq \mathbb{E}_{\mathbf{y},\mathbf{y}'}\left\| \mathbf{y}-\mathbf{y}'\right\|_{L_{1}(\mathcal{M})}=\mathbb{E}_{\mathbf{x},\mathbf{x}'}\|\mathbf{x}-\mathbf{x}'\|_{L_{1}(\mathcal{M})}.
		\end{equation}
		In the following, if two random variables are present, $\mathbb{E}$ denotes expectation over all random variables unless otherwise indicated by subscripts.
		
		Recall that we know \eqref{mazur relations2}, then
		\[\mathbb{E}\|G\|_{L_{2}(\mathcal{M})}^2=\mathbb E\|\mathbf{y}\|_{L_{1}(\mathcal{M})}.\]
		Moreover, invoking \eqref{mazur relations1} and \eqref{B}, it holds
		\begin{align*}
			\mathbb{E}\|\mathbf{x}-\mathbf{x}'\|_{L_{1}(\mathcal{M})} 
			&= \mathbb{E}\|\mathbf{y}-\mathbf{y}'\|_{L_{1}(\mathcal{M})}\\
			&=\mathbb{E}\|\mathbb{M}_{2,1}(\mathbb{M}_{1,2}(\mathbf{y}))-\mathbb{M}_{2,1}(\mathbb{M}_{1,2}(\mathbf{y}'))\|_{L_{1}(\mathcal{M})}\\
			&= \mathbb{E}\|\mathbb{M}_{2,1}(G)-\mathbb{M}_{2,1}(G')\|_{L_{1}(\mathcal{M})}\\
			&\leq B\,\mathbb{E}\left[\max\{\|G\|_{L_{2}(\mathcal{M})},\|G'\|_{L_{2}(\mathcal{M})}\}\,\|G-G'\|_{L_{2}(\mathcal{M})}\right].
		\end{align*}
		The Cauchy–Schwarz inequality allows us to further estimate the above inequality
		\begin{align*}
			\mathbb{E}\|\mathbf{y}-\mathbf{y}'\|_{L_{1}(\mathcal{M})}
			&\leq B\left(\mathbb{E}\max\{\|G\|_{L_{2}(\mathcal{M})}^2,\|G'\|_{L_{2}(\mathcal{M})}^2\}\right)^{1/2} \left(\mathbb{E} \|G-G'\|_{L_{2}(\mathcal{M})}^2\right) ^{1/2}\\
			&\leq B(2 \mathbb{E} \|G\|_{L_{2}(\mathcal{M})}^2)^{1/2} \left(\mathbb{E} \|G-G'\|_{L_{2}(\mathcal{M})}^2\right)^{1/2}\\
			&= B(2 \mathbb{E} \|\mathbf{y}\|_{L_{1}(\mathcal{M})})^{1/2} \left(\mathbb{E} \|G-G'\|_{L_{2}(\mathcal{M})}^2\right) ^{1/2}.
		\end{align*}
		Since $L_2(\mathcal{M})$ is a Hilbert space, the standard independent-copy identity gives
		\[\mathbb{E} \|G-G'\|_{L_{2}(\mathcal{M})}^2=2 \mathbb{E}\|G-\mathbb{E} G\|_{L_{2}(\mathcal{M})}^2.\]
		Using all the preceding derivations, \eqref{eq y} yields
		\begin{equation}\label{proof1}
			\begin{aligned}
				\mathbb{E}\|\mathbf{x}-\mathbf{x}'\|_{L_{1}(\mathcal{M})}
				&\leq 4B^2 \mathbb{E}\|G-\mathbb{E} G\|_{L_{2}(\mathcal{M})}^2\\
				&=4B^2\mathbb{E} \|\mathbb{M}_{1,2}(\mathbf{y}(\varepsilon))-\mathbb{E} \mathbb{M}_{1,2}(\mathbf{y})\|_{L_{2}(\mathcal{M})}^2.
			\end{aligned}
		\end{equation}
		
		We next reduce the problem to the sum of the norms of the partial derivatives to proceed with the proof. Theorem \ref{eq:cee-original} gives the desired bound
		\begin{equation}\label{proof2}
			\begin{aligned}
				\mathbb{E} \|\mathbb{M}_{1,2}(\mathbf{y})-\mathbb{E} \mathbb{M}_{1,2}(\mathbf{y})\|_{L_{2}(\mathcal{M})}^2
				&\leq C_{\Phi}\sum_{j=1}^n\left\| \partial_j (\mathbb{M}_{1,2} \circ \mathbf{y})\right\|_{L_\Phi}^2\\
				&= C_{\Phi}\sum_{j=1}^n\left\|\,\|\partial_j (\mathbb{M}_{1,2} \circ \mathbf{y})\|_{L_{2}(\mathcal{M})}\,\right\|_{L_\Phi}^2
			\end{aligned}
		\end{equation}
		as $L_2(\mathcal{M})$ is a Hilbert space.
		According to the definition of the derivation, 
		\begin{align}
			\| \partial_j \left( \mathbb{M}_{1,2} \circ \mathbf{y}\right) (\varepsilon)\|_{L_{2}(\mathcal{M})}
			&=\frac{1}{2}\|\mathbb{M}_{1,2}(\mathbf{y}(\varepsilon))-\mathbb{M}_{1,2}(\mathbf{y}(\varepsilon^{(j)}))\|_{L_{2}(\mathcal{M})}\notag\\
			&\leq\frac A2\|\mathbf{y}(\varepsilon)-\mathbf{y}(\varepsilon^{(j)})\|_{L_{1}(\mathcal{M})}^{1/2}\notag =\frac A{\sqrt2}\|\partial_j f(\varepsilon)\|_{L_{1}(\mathcal{M})}^{1/2}.
		\end{align}
		where the first inequality follows from \eqref{A}. 
		By monotonicity and homogeneity of the Orlicz norm, followed by the identity and comparison in \eqref{normequvalience2},
		\begin{equation}\label{proof3}
			\begin{aligned}
				\bigl\|\,\|\partial_j (\mathbb{M}_{1,2} \circ \mathbf{y})\|_{L_{2}(\mathcal{M})}\,\bigr\|_{L_\Phi}^2
				&\leq\frac{A^2}2\bigl\|\|\partial_j f(\varepsilon)\|_{L_{1}(\mathcal{M})}^{1/2}\bigr\|_{L_\Phi}^2\\
				&\asymp\frac{A^2}2\bigl\|\|\partial_j f(\varepsilon)\|_{L_{1}(\mathcal{M})}\bigr\|_{\psi_{1,1}}.
			\end{aligned}
		\end{equation}
		Combining \eqref{proof1}, \eqref{proof2} and \eqref{proof3}, we get
		\[\mathbb{E} \left\|f(\varepsilon)-f(\delta)\right\|_{L_{1}(\mathcal{M})} \lesssim \sum_{j=1}^{n} \bigl\|\|\partial_j f(\varepsilon)\|_{L_{1}(\mathcal{M})}\bigr\|_{\psi_{1,1}}\]
		Hence we complete the whole proof.
