<!-- head-1: body_md -->
While logistic damping is known to suppress blow-up in related
chemotaxis systems
(\cite{Lankeit Wang(2017),Osaki(2002),Winkler(2010CPDE),Zhang(2021)}),
its effect in the present degenerate consumption system has remained
elusive; in particular, whether \(L^\infty\)-boundedness holds has been
an open question, to the best of our knowledge. To this end, we
consider the initial-boundary value problem
\begin{equation}\label{1.1}
	\left\{
	\begin{array}{ll}
		u_t=\Delta(\phi(v)u)+au-bu^2, & x\in\Omega,\ t>0,\\[2mm]
		v_t=\Delta v-vu, & x\in\Omega,\ t>0,\\[2mm]
		\partial_\nu u=\partial_\nu v=0, & x\in\partial\Omega,\ t>0,\\[2mm]
		u(\cdot,0)=u_0,\quad v(\cdot,0)=v_0, & x\in\Omega,
	\end{array}
	\right.
\end{equation}
where \(\Omega\subset\mathbb R^2\) is a bounded domain with smooth
boundary.
Here \(a>0\), \(b>0\), and the initial data \((u_0,v_0)\)
satisfy
\begin{equation}\label{1.8}
	\left\{\begin{array}{ll}
		u_0\in W^{1,\infty}(\Omega)\mbox{ is nonnegative with }u_0\not\equiv0,\\[1mm]
		v_0\in W^{1,\infty}(\Omega)\mbox{ with }v_0>0\mbox{ in }\overline{\Omega}.
	\end{array}
	\right.
\end{equation}
The motility function \(\phi\) is assumed to be algebraically
degenerate near zero in the sense that
\begin{equation}\label{1.5}
    \phi\in C^0([0,\infty))\cap C^3((0,\infty))\mbox{ is such that }\phi(0)=0\mbox{ and }\phi>0\mbox{ on }(0,\infty),
\end{equation}
and that for some $\alpha\geq 1$,
\begin{equation}\label{1.6}
	\liminf_{s\searrow0}\frac{\phi(s)}{s^\alpha}>0,\qquad
	\limsup_{s\searrow0}\frac{|\phi'(s)|}{s^{\alpha-1}}<\infty.
\end{equation}
The existence of a  classical solution to \eqref{1.1} was established in \cite{Wang2025}; see Proposition \ref{pro1.1} below.

<!-- theorem-1: body_md -->
\label{theorem1.1}
Under assumptions \eqref{1.8}--\eqref{1.6}, any globally classical solution $(u,v)$ of problem \eqref{1.1} satisfies
\begin{equation}
    \|u(\cdot,t)\|_{L^{\infty}(\Omega)}+\|v(\cdot,t)\|_{W^{1,\infty}(\Omega)}\leq C~~\mbox{for all}~~t>0,
\end{equation}
where $C>0$ depends only on the initial data and parameters.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{theorem1.2}
Assume that \eqref{1.8}--\eqref{1.6} hold with \(\alpha>1\).
Then there exist constants \(C_v>0\) and \(\lambda>0\) such that
\begin{equation}\label{1.10}
\|v(\cdot,t)\|_{W^{1,\infty}(\Omega)}
\leq C_v e^{-\lambda t}
\qquad\text{for all }t>0.
\end{equation}
Moreover, for every \(1\leq p<\infty\), there exist constants
\(C_p>0\) and \(\eta_p>0\) such that
\begin{equation}\label{1.11a}
\left\|u(\cdot,t)-\frac{a}{b}\right\|_{L^p(\Omega)}
\leq C_p e^{-\eta_p t}
\qquad\text{for all }t>0.
\end{equation}
In particular,
\begin{equation}\label{1.12}
u(\cdot,t)\overset{*}{\rightharpoonup}\frac{a}{b}
\quad\text{in }L^\infty(\Omega)
\quad\text{as }t\to\infty.
\end{equation}

<!-- remark-1: body_md -->
\label{rem:rate}
	The proof provides an admissible choice of the decay rate in
	\eqref{1.11a}. More precisely, fix
	\(r\in(0,\min\{1,\alpha/2\})\), and let \(\lambda>0\) be the
	decay rate furnished by Lemma~\ref{lemma4.4}. Then, for any
	$
	0<\mu<\frac12\min\{2a,(\alpha-2r)\lambda\}$,	one may choose
	\(\eta_p=\mu\) when \(1\le p<2\) and
	\(\eta_p=2\mu/p\) when \(2\le p<\infty\).

<!-- remark-2: body_md -->
\label{rem:macroscopic}
Theorem~\ref{theorem1.2} excludes persistent macroscopic patterns only in the following sense: since \(u(\cdot,t)\to a/b\) in \(L^p(\Omega)\) for every \(1\le p<\infty\), any \(O(1)\)-amplitude non-uniformity must be supported on sets of vanishing measure. However, small-measure heterogeneities are not excluded, and strong \(L^\infty\)-convergence remains open.

Set
	\begin{equation}\label{4.8}
		y(t):=\frac12\int_\Omega\left(u(\cdot,t)-\frac{a}{b}\right)^2,

		h(t):=\int_\Omega v^{\alpha-2}|\nabla v|^2.
	\end{equation}

	parts, we obtain
	\begin{align}\label{4.9}
		y'(t)
		&=-\int_\Omega\phi(v)|\nabla u|^2
		-\int_\Omega u\phi'(v)\nabla u\cdot\nabla v
		-b\int_\Omega u\left(u-\frac{a}{b}\right)^2.
	\end{align}
	By Young's inequality, Theorem

Set $u_*:=a/b$, $s(x,t):=u(x,t)/u_*$, and
	\[
		\Psi(s):=s^{-4}+4s-5,
		\qquad E(t):=\int_\Omega\Psi(s(\cdot,t)).
	\]
	By the positivity of $u(\cdot,t)$ for all $t>0$, the functional $E(t)$ is well defined on $(0,\infty)$.   A direct calculation gives
	\begin{equation}\label{4.16}
		4(s^{-4}-s)(1-s)-2\Psi(s)
		=\frac{2(s-1)^4(2s^2+2s+1)}{s^4}\geq0.
	\end{equation}
	Since $s_t=u_t/u_*$, we get
	\begin{align}\label{4.17}
		E'(t)
		&=\int_\Omega\frac{\Psi'(s)}{u_*}\Delta(\phi(v)u)
		+a\int_\Omega\Psi'(s)s(1-s)\nonumber\\
		&=\int_\Omega\frac{\Psi'(s)}{u_*}\Delta(\phi(v)u)-4a\int_\Omega(s^{-4}-s)(1-s)\\
		&\leq\int_\Omega\frac{\Psi'(s)}{u_*}\Delta(\phi(v)u)-2aE(t)\nonumber
	\end{align}

		-\int_\Omega\frac{\Psi''(s)s}{u_*}\phi'(v)\nabla u\cdot\nabla v\nonumber\\

		\left|\nabla u+\frac{u_*s\phi'(v)}{2\phi(v)}\nabla v\right|^2+\frac14\int_\Omega\Psi''(s)s^2
		\frac{|\phi'(v)|^2}{\phi(v)}|\nabla v|^2\nonumber\\
\leq& 5\int_\Omega s^{-4}
		\frac{|\phi'(v)|^2}{\phi(v)}|\nabla v|^2.\label{4.18}
	\end{align}
