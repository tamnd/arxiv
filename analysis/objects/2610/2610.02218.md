<!-- theorem-1: body_md -->
\label{thm:main}
Let $N\ge1$ be square-free, and let $\varnothing\ne A\subseteq\mathbb Z_N$
have cardinality $m$. The following assertions are equivalent:
\begin{enumerate}[label=\textup{(\arabic*)},leftmargin=2.3em]
\item $A$ is spectral in $\mathbb Z_N$;
\item $A$ tiles $\mathbb Z_N$;
\item $m\mid N$ and reduction modulo $m$ maps $A$ bijectively onto $\mathbb Z_m$.
\end{enumerate}
When these conditions hold,
\[
 A\oplus m\mathbb Z_N=\mathbb Z_N,
 \qquad
 \frac Nm\{0,1,\ldots,m-1\}
 \quad\text{is a spectrum of }A.
\]

<!-- lemma-1: body_md -->
\label{lem:coefficients}
Let $n\ge1$, let $P$ be a prime not dividing $n$, and let
$c\in\mathbb Z_P\setminus\{0\}$. If $b_0,\ldots,b_{P-1}\in\mathbb Q(\zeta_n)$
satisfy
\[
 \sum_{j=0}^{P-1}b_j\zeta_P^{cj}=0,
\]
then $b_0=\cdots=b_{P-1}$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
Since $P\nmid n$, the cyclotomic degree formula gives
\[
 [\mathbb Q(\zeta_n,\zeta_P):\mathbb Q(\zeta_n)]
 =\frac{\varphi(nP)}{\varphi(n)}=P-1.
\]
Thus $\Phi_P(X)=1+X+\cdots+X^{P-1}$ is the minimal polynomial
of the primitive $P$th root $\zeta_P^c$ over $\mathbb Q(\zeta_n)$.
The polynomial $\sum_{j=0}^{P-1}b_jX^j$ has degree at most $P-1$
and vanishes at $\zeta_P^c$, so it is a scalar multiple of $\Phi_P$,
possibly the zero polynomial. Its coefficients are therefore equal.
<!-- end proof-1 -->
<!-- lemma-2: body_md -->
\label{lem:block-rigidity}
Let $R=\mathbb Z[\zeta_n]$ for some $n\ge1$, let $m\ge1$, and let
$P$ be an odd prime not dividing $m$. Fix a common block decomposition
of the row and column indices of $M_m(R)$. Suppose
$D_0,\ldots,D_{P-1}\in M_m(R)$ are block diagonal and
$B\in M_m(R)$ has zero diagonal blocks. If
\begin{equation}\label{eq:rigidity-assumptions}
 \sum_{j=0}^{P-1}D_j=mI_m,
 \qquad
 (D_j+B)^2=m(D_j+B)\quad(0\le j<P),
\end{equation}
then $B=0$.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
Summing the quadratic identities in~\eqref{eq:rigidity-assumptions}
and using
\[
 \sum_j(D_jB+BD_j)
 =\left(\sum_jD_j\right)B+B\left(\sum_jD_j\right)=2mB
\]
gives
\[
 \sum_jD_j^2+2mB+PB^2=m^2I_m+mPB.
\]
The matrices $D_j^2$ are block diagonal. Hence the off-diagonal
blocks of this identity give
\begin{equation}\label{eq:off-diagonal}
 P(B^2)_{ii'}=m(P-2)B_{ii'}\qquad(i\ne i'),
\end{equation}
where $(B^2)_{ii'}=\sum_k B_{ik}B_{ki'}$.

We prove by induction that every entry of $B$ belongs to $P^rR$
for every integer $r\ge0$. The assertion for $r=0$ is immediate.
If it holds for $r$, then every entry of $PB^2$ belongs to
$P^{2r+1}R\subseteq P^{r+1}R$. By~\eqref{eq:off-diagonal},
the same is true of $m(P-2)B_{ii'}$ for $i\ne i'$.
Since $m(P-2)$ is coprime to $P$, it has an integer inverse
modulo $P^{r+1}$. Thus every entry of $B_{ii'}$ belongs to
$P^{r+1}R$. The diagonal blocks of $B$ are zero, completing
the induction.

Finally, $R$ is free over $\mathbb Z$, with basis
$1,\zeta_n,\ldots,\zeta_n^{\varphi(n)-1}$.
Consequently $\bigcap_{r\ge0}P^rR=\{0\}$, and hence $B=0$.

<!-- theorem-2: body_md -->
\label{thm:eliminate}
Let $n\ge1$ and let $P$ be an odd prime with $P\nmid n$. Suppose
$(A,\Lambda)$ is an $m$-point spectral pair in
$\mathbb Z_n\times\mathbb Z_P$ and $P\nmid m$.
Then the projections of $A$ and $\Lambda$ onto $\mathbb Z_n$ are
injective, and their images form a spectral pair in $\mathbb Z_n$.
<!-- end theorem-2 -->
<!-- proof-3: body_md -->
For $i,j\in\mathbb Z_P$, set
\[
 A_i=\{x:(x,i)\in A\},\qquad
 \Lambda_j=\{u:(u,j)\in\Lambda\}.
\]
Let $F$ be the character matrix of $(A,\Lambda)$, and let $W$
be obtained by deleting the $P$-coordinate factor:
\[
 F_{(x,i),(u,j)}=\zeta_n^{xu}\zeta_P^{ij},
 \qquad W_{(x,i),(u,j)}=\zeta_n^{xu}.
\]
Both are $m\times m$ matrices indexed by the original points;
injectivity of the projections is not assumed. We use the row
block decomposition $\mathbb C^A=\bigoplus_i\mathbb C^{A_i}$.

Let $F_j,W_j$ consist of the columns indexed by $\Lambda_j$, and put
$Q_j=W_jW_j^*\in M_m(\mathbb Z[\zeta_n])$. Multiplying row $(x,i)$
of $W_j$ by $\zeta_P^{ij}$ gives $F_j$. This is a unitary row
scaling, so column orthogonality in $F$ gives
\[
 W_j^*W_j=F_j^*F_j=mI_{|\Lambda_j|}.
\]
It follows that
\begin{equation}\label{eq:quadratic}
 Q_j^2=mQ_j\qquad(0\le j<P).
\end{equation}
An empty column block gives $Q_j=0$.

For rows $(x,i),(x',i')$ in distinct layers, define
\[
 b_j=(Q_j)_{(x,i),(x',i')}
     =\sum_{u\in\Lambda_j}\zeta_n^{(x-x')u}.
\]
Their orthogonality in $F$ reads
\[
 \sum_{j=0}^{P-1}b_j\zeta_P^{(i-i')j}=0.
\]
Lemma~\ref{lem:coefficients} implies $b_0=\cdots=b_{P-1}$.
Thus the matrices $Q_j$ have a common off-diagonal block part $B$,
and we can write
\begin{equation}\label{eq:block-parts}
 Q_j=D_j+B,
\end{equation}
where $D_j$ is block diagonal and $B$ has zero diagonal blocks.
Here block indices refer to the layers $A_i$, so each $B_{ii'}$
is an $|A_i|\times|A_{i'}|$ matrix.
For two rows in the same layer, their $P$-coordinate phases cancel
in the inner product. Row orthogonality and the row norms in $F$
therefore yield
\begin{equation}\label{eq:diagonal-sum}
 \sum_{j=0}^{P-1}D_j=mI_m.
\end{equation}

Equations~\eqref{eq:quadratic}--\eqref{eq:diagonal-sum} verify
all the hypotheses of Lemma~\ref{lem:block-rigidity}, so $B=0$.
Consequently
\[
 WW^*=\sum_jQ_j=\sum_jD_j=mI_m.
\]
The square matrix $W$ is nonsingular. Two points of $A$ with the
same projection would give equal rows of $W$, and two points of
$\Lambda$ with the same projection would give equal columns.
Thus both projections are injective, and $W$ is the character
matrix of their images. These images form an $m$-point spectral pair.

<!-- corollary-1: body_md -->
\label{cor:abelian-elimination}
Let $H$ be a finite abelian group of exponent $e$, let $P$ be
an odd prime with $P\nmid e$, and suppose $P\nmid m$.
If $(A,\Lambda)$ is an $m$-point spectral pair in $H\times\mathbb Z_P$,
with $\Lambda\subseteq\widehat H\times\mathbb Z_P$ under the product
pairing, then the projections of $A$ onto $H$ and of $\Lambda$
onto $\widehat H$ are injective and form a spectral pair.
<!-- end corollary-1 -->
<!-- proof-4: body_md -->
Every character of $H$ takes values in $\mathbb Z[\zeta_e]$.
Replace $\zeta_n^{xu}$ in the proof of Theorem~\ref{thm:eliminate}
by $\chi(x)$, with $x\in H$ and $\chi\in\widehat H$.
Lemmas~\ref{lem:coefficients} and~\ref{lem:block-rigidity} apply
with $n=e$, and the same argument gives $WW^*=mI_m$.
The conclusions follow from the nonsingularity of $W$.

<!-- remark-1: body_md -->
\label{rem:two}
For $P=2$, the coefficient $m(P-2)$ in~\eqref{eq:off-diagonal}
vanishes. The block-matrix lemma therefore requires $P$ to be odd.
In Proposition~\ref{prop:spectral-transversal}, we retain a possible
coordinate of order two and handle it by cardinality and parity.

<!-- proposition-1: body_md -->
\label{prop:spectral-transversal}
Let $N\ge1$ be square-free. If $A\subseteq\mathbb Z_N$ is a nonempty
spectral set of cardinality $m$, then $m\mid N$ and reduction
modulo $m$ maps $A$ bijectively onto $\mathbb Z_m$.
<!-- end proposition-1 -->
<!-- proof-5: body_md -->
Let $(A,\Lambda)$ be an $m$-point spectral
pair in $\mathbb Z_N$, with $N$ square-free. Put $g=\gcd(N,m)$.
Thus $g$ is the product of those prime factors of $N$ that divide
$m$. No divisibility relation between $m$ and $N$ is assumed at
this stage.

Apply Theorem~\ref{thm:eliminate} successively to every odd prime
factor of $N$ that does not divide $m$. Each step preserves the
cardinality and spectrality of both projected sets and is injective
on them. More explicitly, after deleting coordinates whose product
is $d$, the spatial and frequency labels in the current cyclic
pairing are $a\bmod(N/d)$ and $d^{-1}\lambda\bmod(N/d)$,
respectively. Thus the spatial map remains the natural quotient
throughout the iteration. The final spectral pair lies in a group of order
\[
 M=\begin{cases}
 2g,&N\text{ even and }m\text{ odd},\\
 g,&\text{otherwise}.
 \end{cases}
\]
Hence $g\mid m\le M$. If $M=g$, this gives $m=g$.
If $M=2g$, then $m/g$ is an odd positive integer at most $2$,
so again $m=g$. In particular, $m\mid N$.

When $M=m$, the final projected set is the whole group $\mathbb Z_m$.
Suppose now that $M=2m$, so $N$ is even and $m$ is odd. The final pair, say
$(A',\Lambda')$, lies in $\mathbb Z_m\times\mathbb Z_2$.
We use the product pairing, relabelling the dual coordinates as
at the beginning of Section~\ref{sec:elimination}. The coordinate
projection onto $\mathbb Z_m$ is injective on $A'$. Indeed, a collision
would give two points $(x,0),(x,1)\in A'$, whose row orthogonality
would imply
\[
 0=\sum_{(u,j)\in\Lambda'}(-1)^j
   =|\Lambda'_0|-|\Lambda'_1|.
\]
This makes $m=|\Lambda'|$ even, a contradiction. Since $|A'|=m$,
its projection is the whole of $\mathbb Z_m$.

In all cases, $m\mid N$, and the composition of these CRT
projections is the natural quotient map $\mathbb Z_N\to\mathbb Z_m$ on the
spatial group. It is bijective on $A$. This proves the proposition.

<!-- lemma-3: body_md -->
\label{lem:dilation}
Suppose $C\oplus D=\mathbb Z_N$ and $|C|=k$. If $r$ is prime and
$r\nmid k$, then the map $c\mapsto rc$ is injective on $C$ and
$rC\oplus D=\mathbb Z_N$.
<!-- end lemma-3 -->
<!-- proof-6: body_md -->
Work in the commutative group ring
$\mathbb Z[\mathbb Z_N]\simeq\mathbb Z[X]/(X^N-1)$, so that coefficients are indexed
by residues modulo $N$. Write
$C(X)=\sum_{c\in C}X^c$, and similarly for $D$, and put
$U=\sum_{x\in\mathbb Z_N}X^x$. The tiling identity is $C(X)D(X)=U$.
For every group-ring element $H$ one has $HU=H(1)U$.
Reducing coefficients modulo $r$ and applying Frobenius gives
\[
 C(X^r)D(X)
 \equiv C(X)^rD(X)
 =C(X)^{r-1}U
 =k^{r-1}U
 \equiv U\pmod r.
\]
The last congruence uses Fermat's theorem and $r\nmid k$.
The coefficients of $C(X^r)D(X)$ in the integral group ring are
nonnegative integers, each congruent to $1$ modulo $r$, and hence
each at least $1$. They sum to $k|D|=N$,
and there are $N$ coefficients. Thus every coefficient is $1$.
This also excludes repeated points in the list $rC$, since any
such repetition would give a coefficient at least $2$ after
addition of an element of the nonempty set $D$.

<!-- proof-7: body_md -->
Proposition~\ref{prop:spectral-transversal} proves
\textup{(1)}$\Rightarrow$\textup{(3)}.

To prove \textup{(2)}$\Rightarrow$\textup{(3)}, suppose that
$A\oplus B=\mathbb Z_N$ and $|A|=m$. Counting gives
$m\mid N$. Put $b=N/m$. Since $N$ is square-free,
$\gcd(b,m)=1$. Apply Lemma~\ref{lem:dilation} successively to
the prime factors of $b$. The dilation $a\mapsto ba$ is injective
on $A$, and $bA$ has $m$ elements. But it is contained in the
subgroup $b\mathbb Z_N$, also of order $m$, so
\[
 bA=b\mathbb Z_N.
\]
The equality $ba=ba'$ in $\mathbb Z_N$ is equivalent to $a=a'$ modulo
$m$. Thus reduction modulo $m$ maps $A$ bijectively onto $\mathbb Z_m$.

Finally, suppose \textup{(3)} holds. The unique representative
of each residue modulo $m$ immediately gives
$A\oplus m\mathbb Z_N=\mathbb Z_N$. To see spectrality directly, set
\[
 \Lambda_0=\left\{\frac Nm j:0\le j<m\right\}\subseteq\mathbb Z_N.
\]
For distinct $j,k\in\{0,\ldots,m-1\}$,
\[
 \sum_{a\in A}e^{2\pi i a(j-k)/m}
   =\sum_{u=0}^{m-1}e^{2\pi i u(j-k)/m}=0.
\]
The $m$ columns indexed by $\Lambda_0$ have squared norm $m$,
so $\Lambda_0$ is a spectrum. This completes the proof of
Theorem~\ref{thm:main}.
