\begin{align*}
\mathcal{H}_{\hbar}(t)&=\frac{1}{2}\mathrm{tr}\left((i\hbar\nabla+A_{\hbar}+u)^{2}R_{\hbar}\right)\\

In the above formula and hereafter we denote by $\vee$ the anti-commutator defined by $$T\vee S\coloneqq TS+ST.$$

\textbf{Step 1. Reduction of the system}. We claim that existence of a solution to \eqref{Euler Maxwell}-\eqref{EMI} would follow from existence of a solution to the system 
\begin{align}
\begin{cases}
 \begin{array}{lc}
        \partial_{t}u+uD_{x}u=E+u\times B\\
       \partial_{t}E-\mathrm{curl}B+u\mathrm{div}E=0\\
       \partial_{t}B+\mathrm{curl}E+u\mathrm{div}B=0\\
       (u(0,\cdot),E(0,\cdot),B(0,\cdot))=(u^{\mathrm{in}},E^{\mathrm{in}},B^{\mathrm{in}}).  
 \end{array}
 \end{cases}
\label{reduced Euler Maxwell}
\end{align}
To see this, suppose $(u,E,B)\in C([0,T];H^{s}(\mathbb{R}^{3}))$ is a solution to \eqref{reduced Euler Maxwell} and set $\rho=\mathrm{div}E$ (so that $\rho(0,\cdot)=\mathrm{div}E^{\mathrm{in}}=\rho^{\mathrm{in}}$). Then,
by taking the divergence in the equation for $B$ and using that $\mathrm{div}(\mathrm{curl})\equiv 0$ we get 
\begin{align*}
\partial_{t}\mathrm{div}B+\mathrm{div}(u\mathrm{div}B)=0,\ \mathrm{div}B(0,\cdot )=0.       
\end{align*}
Thus $\mathrm{div}B$ satisfies a linear transport equation in divergence form with $0$ initial data, and therefore by uniqueness it must be that $\mathrm{div}B\equiv 0$. 
Moreover, setting $\rho=\mathrm{div}E$ the equation for $E$ writes 
\begin{align*}
\partial_{t}E-\mathrm{curl}B=-\rho u.    
\end{align*}
Taking the divergence of the last equation we get 
\begin{align*}
\partial_{t}\rho+\mathrm{div}(\rho u)=0,  \end{align*}
so that we eventually arrive at 
\begin{align}
\begin{cases}
\begin{array}{lc}
\partial_{t}\rho+\mathrm{div}(\rho u)=0\\
\partial_{t}u+uD_{x}u=E+u\times B\\
\partial_{t}E-\mathrm{curl}B=-\rho u\\
\partial_{t}B+\mathrm{curl}E=0\\
\mathrm{div}E=\rho,\ \mathrm{div}(B)=0,
\end{array}
\end{cases}
\end{align}
which is \eqref{Euler Maxwell}. Note also that since $u\in C([0,T];H^{s}(\mathbb{R}^{3}))$ we also have $\rho\in C([0,T];H^{s-1}(\mathbb{R}^{3}))$. 
\textbf{Step 2. The system \eqref{reduced Euler Maxwell} in vector form.} Set $W\coloneqq(u,E,B)$ and $$F(u,E,B)\coloneqq \left(\begin{array}{c}
E+u\times B\\
0\\
0
\end{array}\right). $$
Given a vector $(a^{1},a^{2},a^{3})$ denote 
\begin{align*}
\mathbf{C}(a)\coloneqq \begin{pmatrix}0 & -a^{3} & a^{2}\\
a^{3} & 0 & -a^{1}\\
-a^{2} & a^{1} & 0
\end{pmatrix}    
\end{align*}
and denote by $\mathbf{E}_{j}(a)$ the $3\times 3$ matrix whose $j$-th column is $a$ and $0$ at all other entries. 
Note that $a\times b=\mathbf{C}(a)b$. Finally, 
we designate by  $e_{j}$ the $j$-th unit vector and by $\mathbf{B}^{j}:\mathbb{R}^{9}\rightarrow M_{9\times 9}(\mathbb{R})$ ($j=1,2,3$) the matrix given by 
\begin{align*}
\mathbf{B}^{j}(u,E,B)\coloneqq \begin{pmatrix}u^{j}I & 0 & 0\\
0 & \mathbf{E}_{j}(u) & -\mathbf{C}(e_{j})\\
0 & \mathbf{C}(e_{j}) & \mathbf{E}_{j}(u)
\end{pmatrix}.    \end{align*}
With this notation the system \eqref{reduced Euler Maxwell} writes 
\begin{align}
\partial_{t}W+\sum_{j=1}^{3}\mathbf{B}^{j}\partial_{x_{j}}W=F(W), \ W(0,\cdot)=W^{\mathrm{in}}. \label{vector form system}      
\end{align}
\textbf{Step 3. Symmetrization.} Note that the matrices $\mathbf{B}^{j}$ in \eqref{vector form system} are not symmetric as required by Theorem \ref{Kato type thm}. Therefore we transform the system to match the conditions in Theorem \ref{Kato type thm} as follows. Set 
\begin{align*}
\mathbf{S}(u,E,B)\coloneqq \begin{pmatrix}I & 0 & 0\\
0 & I & \mathbf{C}(u)\\
0 & -\mathbf{C}(u) & I
\end{pmatrix}\in M_{9\times 9}(\mathbb{R}).    
\end{align*}
We start by verifying that $\mathbf{S}\mathbf{B}^{j}$ is symmetric. 
Invoking block-matrix multiplication we compute that: 
\begin{align}
\mathbf{S}(u,E,B)\mathbf{B}^{j}(u,E,B)=\begin{pmatrix}u^{j}I & 0 & 0\\
0 & \mathbf{E}_{j}(u)+\mathbf{C}(u)\mathbf{C}(e_{j}) & -\mathbf{C}(e_{j})+\mathbf{C}(u)\mathbf{E}_{j}(u)\\
0 & \mathbf{C}(e_{j})-\mathbf{C}(u)\mathbf{E}_{j}(u) & \mathbf{E}_{j}(u)+\mathbf{C}(u)\mathbf{C}(e_{j})
\end{pmatrix}. 
\label{Multiplication of S by Aj}
\end{align}
By direct calculation we have 
\begin{align*}
\mathbf{E}_{j}(u)+\mathbf{C}(u)\mathbf{C}(e_{j})=\mathbf{E}_{j}(u)+\mathbf{E}_{j}^{T}(u)-u_{j}I    
\end{align*}
which is readily seen to be symmetric. 
Therefore all the matrices on the diagonal in \eqref{Multiplication of S by Aj} are symmetric. In addition we have 
\begin{align*}
\mathbf{C}(a)^{T}=\begin{pmatrix}0 & a^{3} & -a^{2}\\
-a^{3} & 0 & a^{1}\\
a^{2} & -a^{1} & 0
\end{pmatrix}=-\mathbf{C}(a)   
\end{align*}
and
\begin{align*}
\mathbf{C}(u)\mathbf{E}_{j}(u)=\mathbf{E}_{j}(u\times u)=\mathbf{0}_{3\times 3}.   
\end{align*}
This shows that $\mathbf{S}\mathbf{B}^{j}$ is symmetric, i.e. that condition \eqref{symmetry of Aj} is verified. 
Next we verify the condition \eqref{positivity of A0}. Given $\xi=(v,E,B)\in \mathbb{R}^{9}\setminus\{0\}$ and $\zeta=(u,E',B')$ we have 
\begin{align*}
\xi^{T}\mathbf{S}(u,E',B')\xi&=(v,E,B)\begin{pmatrix}v \\
E+\mathbf{C}(u)B \\
-\mathbf{C}(u)E+B 
\end{pmatrix}=\left\vert v\right\vert^{2}+\left\vert E\right\vert^{2}+E\cdot (u\times B)+\left\vert B\right\vert^{2}-B\cdot (u\times E)\\
&=\left\vert v\right\vert^{2}+\left\vert E\right\vert^{2}+\left\vert B\right\vert^{2}-2u\cdot (E\times B). 
\end{align*}
Note that for all $(v,E,B)\neq 0$ and $\left\vert u\right\vert <1$ it holds that  
\begin{align*}
\left\vert v\right\vert^{2}+\left\vert E\right\vert^{2}+\left\vert B\right\vert^{2}-2u\cdot (E\times B)&\geq \left\vert v\right\vert^{2}+\left\vert E\right\vert^{2}+\left\vert B\right\vert^{2}-\left\vert u\right\vert(\left\vert B\right\vert^{2}+\left\vert E\right\vert^{2})\\
&=\left\vert v\right\vert^{2}+(1-\left\vert u\right\vert)(\left\vert E\right\vert^{2}+\left\vert B\right\vert^{2})>0.         
\end{align*}
Fix $\delta >0$ such that $\left\Vert u^{\mathrm{in}}\right\Vert_{\infty}\leq  1-\delta$. Then $\mathbf{S}$ verifies \eqref{positivity of A0} on $$\Omega\coloneqq \{(u,E,B)\in \mathbb{R}^{9}\vert \left\vert u\right\vert< 1-\frac{\delta}{2}\}.$$ 
Moreover, by assumption we also have $\overline{\{(u^{\mathrm{in}}(x),E^{\mathrm{in}}(x),B^{\mathrm{in}}(x))\vert x\in \mathbb{R}^{3}\}}\Subset \Omega$, which shows that condition iii. is satisfied. To conclude, by Theorem \ref{Kato type thm} we have proved that there exist some $T>0$ and a unique solution $W$ to the system 
\begin{align*}
\mathbf{S}(W)\partial_{t}W+\sum_{j=1}^{3}\mathbf{S}(W)\mathbf{B}^{j}\partial_{x_{j}}W=\mathbf{S}(W)F(W), \ W(0,\cdot)=W^{\mathrm{in}}.    
\end{align*}
Since $\mathbf{S}(W)$ is invertible on $[0,T]$ by applying the inverse we get that $W$ is the requested solution for the system \eqref{vector form system}. Finally we have that $W(t,x)\in \Omega$ for all $(t,x)\in [0,T]\times \mathbb{R}^{3}$ so that $\left\Vert u\right\Vert_{\infty}<1 $.

^{\otimes 2} -\frac{1}{2}\left\vert v\right\vert^{2} I\right)-(\mathrm{div}v)v.   
\end{align*}
Therefore, we find that 
\begin{align}
\int_{\mathbb{R}^{3}} u\cdot (e_{\hbar}\times \mathrm{curl}(e_{\hbar}))\ {\rm d} x&=-\int_{\mathbb{R}^{3}} u\cdot \mathrm{div}\left(e^{\otimes 2}_{\hbar}-\frac{1}{2}\left\vert e_{\hbar}\right\vert^{2}I \right)\ {\rm d} x+\int_{\mathbb{R}^{3}}u\cdot e_{\hbar}\mathrm{div}e_{\hbar}\ {\rm d} x \notag\\
&=\int_{\mathbb{R}^{3}} D_{x}u:\left(e^{\otimes 2}_{\hbar} -\frac{1}{2}\left\vert e_{\hbar}\right\vert^{2}I \right)\ {\rm d} x+\int_{\mathbb{R}^{3}}u\cdot e_{\hbar}\mathrm{div}e_{\hbar}\ {\rm d} x  
\label{eh times curl eh}
\end{align}
and that 
\begin{align}
-\int_{\mathbb{R}^{3}} u\cdot (\mathrm{curl}(b_{\hbar})\times b_{\hbar})\ {\rm d} x&=-\int_{\mathbb{R}^{3}} u\cdot \mathrm{div}\left(b^{\otimes 2}_{\hbar}-\frac{1}{2}\left\vert b_{\hbar}\right\vert^{2}I \right)\ {\rm d} x+\underset{=0}{\underbrace{\int_{\mathbb{R}^{3}}u\cdot b_{\hbar}\mathrm{div}b_{\hbar}\ {\rm d} x}} \notag\\
&=\int_{\mathbb{R}^{3}} D_{x}u:\left(b^{\otimes 2}_{\hbar}-\frac{1}{2}\left\vert b_{\hbar}\right\vert^{2} I\right)\ {\rm d} x. \label{curl bh times bh} \end{align}
Substituting \eqref{eh times curl eh}-\eqref{curl bh times bh} inside \eqref{wedge quantum product term} we get 
\begin{align*}
\mathcal{T}_{5}&=\int_{\mathbb{R}^{3}}D_{x}u:\left(e^{\otimes 2}_{\hbar}-\frac{1}{2}\left\vert e_{\hbar}\right\vert^{2}I+b^{\otimes 2}_{\hbar}-\frac{1}{2}\left\vert b_{\hbar}\right\vert^{2}I  \right)\ {\rm d} x\\
&-\int_{\mathbb{R}^{3}}u\cdot \left((\rho u -J_{\hbar})\times b_{\hbar}\right)\ {\rm d} x
+\int_{\mathbb{R}^{3}}u\cdot e_{\hbar}\mathrm{div}e_{\hbar}\ {\rm d} x. 
\end{align*}
Furthermore, using that $J\cdot (u\times b)=-u\cdot (J\times b)$ we see that  
\begin{align*}
\mathcal{T}_{3}=\int_{\mathbb{R}^{3}}J_{\hbar}\cdot (u\times b_{\hbar})\ {\rm d} x=-\int_{\mathbb{R}^{3}}u\cdot ( J_{\hbar}\times b_{\hbar})\ {\rm d} x . 
\end{align*}
Consequently, it follows that 
\begin{align}
\mathcal{T}_{3}+\mathcal{T}_{5}&=\int_{\mathbb{R}^{3}}D_{x}u:\left(e_{\hbar}^{\otimes 2}-\frac{1}{2}\left\vert e_{\hbar}\right\vert^{2}I+b_{\hbar}^{\otimes 2}-\frac{1}{2}\left\vert b_{\hbar}\right\vert^{2}I\right)\ {\rm d} x \notag\\
&\underset{=0}{\underbrace{-\int_{\mathbb{R}^{3}}u\cdot ( J_{\hbar}\times b_{\hbar})\ {\rm d} x +\int_{\mathbb{R}^{3}}u\cdot (J_{\hbar}\times b_{\hbar})\ {\rm d} x}}+\int_{\mathbb{R}^{3}}u\cdot e_{\hbar}\mathrm{div}e_{\hbar}\ {\rm d} x \notag\\
&=\int_{\mathbb{R}^{3}}D_{x}u:\left(e_{\hbar}^{\otimes 2}-\frac{1}{2}\left\vert e_{\hbar}\right\vert^{2}I+b_{\hbar}^{\otimes 2}-\frac{1}{2}\left\vert b_{\hbar}\right\vert^{2}I\right)\ {\rm d} x+\int_{\mathbb{R}^{3}}u\cdot e_{\hbar}\mathrm{div}e_{\hbar}\ {\rm d} x. \label{sum of quantum T3T5}     
\end{align}
Finally, using that $\mathrm{div}E=\rho$ and that $\mathrm{div}E_{\hbar}=\rho_{\hbar}$ we see that 
\begin{align}
\mathcal{T}_{2}&=-\int_{\mathbb{R}^{3}}(\rho_{\hbar}-\rho)u\cdot e_{\hbar}\ {\rm d} x=-\int_{\mathbb{R}^{3}}u\cdot e_{\hbar}\mathrm{div}e_{\hbar}\ {\rm d} x  \label{quantum T2 identity} 
\end{align}
and hence 
\begin{align}
\mathcal{T}_{2}+\mathcal{T}_{3}+\mathcal{T}_{5}=\int_{\mathbb{R}^{3}}D_{x}u:\left(e_{\hbar}^{\otimes 2}-\frac{1}{2}\left\vert e_{\hbar}\right\vert^{2}I+b_{\hbar}^{\otimes 2}-\frac{1}{2}\left\vert b_{\hbar}\right\vert^{2}I\right)\ {\rm d} x.  \label{T2T3T5}   
\end{align}
Substituting \eqref{T2T3T5} inside 
\eqref{formula for time der of quantum G} we deduce that 
\begin{align}
\frac{{\rm d}}{{\rm d} t}\mathcal{G}_{\hbar}(t)&=-\frac{1}{4}\sum_{1\leq j,k\leq 3}\mathrm{tr}\left((i\hbar\partial_{x_{j}}+A^{j}_{\hbar}+u^{j})\vee \left((i\hbar\partial_{x_{k}}+A^{k}_{\hbar}+u^{k})\vee \partial_{x_{k}}u^{j}\right)R_{\hbar}\right) \notag\\
&+\int_{\mathbb{R}^{3}}D_{x}u:\left(e_{\hbar}^{\otimes 2}-\frac{1}{2}\left\vert e_{\hbar}\right\vert^{2}I+b_{\hbar}^{\otimes 2}-\frac{1}{2}\left\vert b_{\hbar}\right\vert^{2}I\right)\ {\rm d} x \notag
\\
&-\int_{\mathbb{R}^{3}}\partial_{t}u\cdot (e_{\hbar}\times b_{\hbar})\ {\rm d} x\coloneqq \sum_{k=1}^{3}I_{k},\label{right hand side of time der of Gh} 
\end{align}
which is the asserted formula \eqref{Time derivative of quantum corrected modulated energy statement}. 
\\
\textbf{Gr\"onwall estimate on $\mathcal{H}_{\hbar}(t)$.} We start by estimating $I_{1}$.  Set $P^{j}\coloneqq \Pi^{j}+u^{j}$. We have 
\begin{align}
P^{j}\vee (P
^{k}\vee \partial_{x_{k}}u^{j})&=P^{j}\vee (P^{k}\partial_{x_{k}}u^{j}+\partial_{x_{k}}u^{j}P^{k}) \notag\\
&=P^{j}P^{k}\partial_{x_{k}}u^{j}+P^{j}\partial_{x_{k}}u^{j}P^{k}+P^{k}\partial_{x_{k}}u^{j}P^{j}+\partial_{x_{k}}u^{j}P^{k}P^{j}. \label{formula for anticommutator of Pj}  \end{align}
Therefore we have 
\begin{align*}
I_{1}=&-\frac{1}{4}\sum_{1\leq j,k\leq 3}\langle \psi_{\hbar}\vert P^{j}P^{k}\partial_{x_{k}}u^{j}\vert \psi_{\hbar}\rangle -\frac{1}{4}\sum_{1\leq j,k\leq 3}\langle \psi_{\hbar} \vert  P^{j}(\partial_{x_{k}}u^{j}+\partial_{x_{j}}u^{k})P^{k} \vert \psi_{\hbar}\rangle\\
&-\frac{1}{4}\sum_{1\leq j,k\leq 3}\langle \psi_{\hbar} \vert  \partial_{x_{k}}u^{j}P^{k}P^{j}\vert  \psi_{\hbar}\rangle \coloneqq I_{1}^{1}+I_{1}^{2}+I_{1}^{3}.        
\end{align*}
To estimate $I_{1}^{1}$, we apply the self-adjointness of $P^{j}$ to find
\begin{align*}
\sum_{k=1}^{3}\langle \psi_{\hbar}\vert   P^{j}P^{k}\partial_{x_{k}}u^{j}\vert\psi_{\hbar}\rangle  &=\sum_{k=1}^{3}\langle P^{j}\psi_{\hbar}\vert   P^{k}\partial_{x_{k}}u^{j}\vert \psi_{\hbar} \rangle\\
&=\sum_{k=1}^{3} (\langle P^{j}\psi_{\hbar}  \vert  i\hbar \partial_{x_{k}x_{k}}u^{j}\vert \psi_{\hbar}\rangle +\langle P^{j}\psi_{\hbar}\vert  \partial_{x_{k}}u^{j}P^{k}\vert \psi_{\hbar}\rangle)\\
&=\langle P^{j}\psi_{\hbar}\vert  i\hbar \Delta u^{j}\vert \psi_{\hbar}\rangle+\sum_{k=1}^{3}\langle P^{j}\psi_{\hbar}  \vert \partial_{x_{k}}u^{j}P^{k}\vert \psi_{\hbar} \rangle.       
\end{align*}
Observe that
\begin{align*}
\left\vert \langle P^{j}\psi_{\hbar}\vert  \partial_{x_{k}}u^{j}P^{k}\vert \psi_{\hbar}\rangle \right\vert  \leq \frac{1}{2}\left\Vert D_{x}u\right\Vert_{\infty}\left(\left\Vert P^{j}\psi_{\hbar}(t,\cdot)\right\Vert_{2}^{2} +\left\Vert P^{k}\psi_{\hbar}(t,\cdot)\right\Vert_{2}^{2} \right).     
\end{align*}
Furthermore, since the $L^{2}$ norm of $\psi_{\hbar}$ is conserved ($\left\Vert \psi_{\hbar}(t,\cdot)\right\Vert_{2}=\left\Vert \psi_{\hbar}^{\mathrm{in}}\right\Vert_{2}=1$) we have 
\begin{align*}
\left\vert \langle P^{j}\psi_{\hbar} \vert i\hbar \Delta_{x}u^{j}\vert \psi_{\hbar}\rangle\right\vert  \leq \frac{1}{2}\left\Vert P^{j}\psi_{\hbar}(t,\cdot) \right\Vert_{2}^{2}+\frac{\hbar^{2}}{2}\left\Vert  \Delta_{x}u^{j}\psi_{\hbar}(t,\cdot)\right\Vert_{2}^{2}\leq \mathcal{U}_{\hbar}(t)+\frac{\hbar^{2}}{2}\left\Vert \Delta_{x} u\right\Vert_{\infty}^{2}.     
\end{align*}
Therefore, we obtain 
\begin{align}
\left\vert I_{1}^{1}\right\vert &\leq 3\left\Vert D_{x}u\right\Vert_{\infty}\mathcal{U}_{\hbar}(t)+9\mathcal{U}_{\hbar}(t)+9\hbar^{2} \left\Vert \Delta_{x}u\right\Vert_{\infty}^{2} \notag\\
&= (3\left\Vert D_{x}u\right\Vert_{\infty} +9)\mathcal{U}_{\hbar}(t)+9\hbar^{2}\left\Vert \Delta_{x}u \right\Vert_{\infty}^{2}.  \label{Est D1}    
\end{align}
To estimate $I_{1}^{2}$, we have 
\begin{align*}
\left\vert \langle \psi_{\hbar}\vert   P^{j}(\partial_{x_{k}}u^{j}+\partial_{x_{j}}u^{k})P^{k}\vert \psi_{\hbar}\rangle\right\vert &= \left\vert \langle P^{j}\psi_{\hbar}\vert  (\partial_{x_{k}}u^{j}+\partial_{x_{j}}u^{k})P^{k}\vert\psi_{\hbar}\rangle\right\vert \\
&\leq 2\left\Vert D_{x}u\right\Vert_{\infty} \left(\frac{1}{2}\left\Vert P^{j}\psi_{\hbar}(t,\cdot)\right\Vert_{2}^{2}+\frac{1}{2}\left\Vert P^{k}\psi_{\hbar}(t,\cdot)\right\Vert_{2}^{2}\right).      
\end{align*}
Therefore it follows that 
\begin{align}
\left\vert I_{1}^{2}\right\vert \leq 6\left\Vert D_{x}u\right\Vert_{\infty}\mathcal{U}_{\hbar}(t). \label{Est D2}     
\end{align}
To estimate $I_{1}^{3}$ we first note that 
\begin{align*}
&\sum_{k=1}^{3}\langle \psi_{\hbar} \vert  \partial_{x_{k}}u_{j}P^{k}P^{j}\vert \psi_{\hbar}\rangle=\sum_{k=1}^{3}\langle P^{k}\partial_{x_{k}}u^{j}\psi_{\hbar}\vert  P^{j}\vert \psi_{\hbar} \rangle\\
&=\sum_{k=1}^{3}\langle \partial_{x_{k}}u^{j}P^{k}\psi_{\hbar}\vert  P^{j}\vert \psi_{\hbar} \rangle  +\sum_{k=1}^{3}\langle i\hbar \partial_{x_{k}x_{k}}u^{j}\psi_{\hbar}\vert  P^{j}\vert \psi_{\hbar}\rangle \\
&= \sum_{k=1}^{3}\langle \partial_{x_{k}}u^{j}P^{k}\psi_{\hbar}\vert P^{j}\vert \psi_{\hbar}\rangle+\langle i\hbar \Delta u^{j}\psi_{\hbar} \vert P
^{j}\vert \psi_{\hbar}\rangle.    
\end{align*}
Therefore it follows that 
\begin{align}
\left\vert I_{1}^{3}\right\vert&\leq \frac{\left\Vert D_{x}u\right\Vert_{\infty}}{2}\sum_{1\leq j,k\leq 3} \left(\left\Vert P^{k}\psi_{\hbar}(t,\cdot)\right\Vert_{2}^{2} \notag 
+\left\Vert P^{j}\psi_{\hbar}(t,\cdot)\right\Vert_{2}^{2}\right)\\
&+\frac{\hbar^{2}}{2}\sum_{j=1}^{3}\left\Vert \Delta u^{j}\psi_{\hbar}(t,\cdot)\right\Vert_{2}^{2}+\frac{1}{2}\sum_{j=1}^{3}\left\Vert P^{j}\psi_{\hbar}\right\Vert_{2}^{2} \notag\\
&\leq 3(\left\Vert D_{x}u\right\Vert_{\infty}+\frac{1}{2})\mathcal{U}_{\hbar}(t)+\frac{9\hbar^{2}}{2}\left\Vert \Delta u\right\Vert_{\infty}^{2} . \label{EST D3}      
\end{align}
Gathering \eqref{Est D1}-\eqref{EST D3} we conclude that 
\begin{align}
I_{1}\leq c_{1}\mathcal{U}_{\hbar}(t)+c_{2}\hbar^{2},    \label{final est D1} 
\end{align}
where $c_{1}=c_{1}(\left\Vert D_{x}u\right\Vert_{\infty})$ and $c_{2}=c_{2}(\left\Vert \Delta_{x}u\right\Vert_{\infty})$. 
To estimate $I_{2}$ we note that 
\begin{align*}
\left\vert I_{2} \right\vert\leq& \left\vert \int_{\mathbb{R}^{3}}D_{x}u:e_{\hbar}^{\otimes 2}\ {\rm d} x\right\vert+\frac{1}{2}\left\vert \int_{\mathbb{R}^{3}}D_{x}u: \left\vert e_{\hbar}\ \right\vert^{2}I\ {\rm d} x \right\vert \\
&+\left\vert \int_{\mathbb{R}^{3}}D_{x}u:b_{\hbar}^{\otimes 2}\ {\rm d} x\right\vert+\frac{1}{2}\left\vert \int_{\mathbb{R}^{3}}D_{x}u:\left\vert b_{\hbar}\right\vert^{2}I \ {\rm d} x\right\vert \\
&\leq \left(\left\Vert D_{x}u\right\Vert_{\infty}+\frac{1}{2}\left\Vert \mathrm{div}u\right\Vert_{\infty}\right)\int_{\mathbb{R}^{3}}\left\vert e_{\hbar}\right\vert^{2}\ {\rm d} x+\left(\left\Vert D_{x}u\right\Vert_{\infty}+\frac{1}{2}\left\Vert \mathrm{div}u\right\Vert_{\infty}  \right)\int_{\mathbb{R}^{3}}\left\vert b_{\hbar}\right\vert^{2}\ {\rm d} x\\  &\leq (2\left\Vert D_{x}u\right\Vert_{\infty}+\left\Vert \mathrm{div}u\right\Vert_{\infty})(\mathcal{E}_{\hbar}(t)+\mathcal{B}_{\hbar}(t)).        
\end{align*}
As for $I_{3}$, we have 
\begin{align*}
\left\vert I_{3} \right\vert\leq \frac{1}{2}\left\Vert \partial_{t}u\right\Vert_{\infty}\int_{\mathbb{R}^{3}}(\left\vert e_{\hbar}\right\vert^{2}+\left\vert b_{\hbar}\right\vert^{2}) \ {\rm d} x\leq \left\Vert \partial_{t}u\right\Vert_{\infty}\left(\mathcal{E}_{\hbar}(t)+\mathcal{B}_{\hbar}(t)\right).         
\end{align*}
Consequently, it follows that 
\begin{align*}
\frac{{\rm d}}{{\rm d} t}\mathcal{G}_{\hbar}(t)\leq C(\mathcal{H}_{\hbar}(t)+\hbar^{2})    
\end{align*}
for some $C=C(\left\Vert u\right\Vert_{W^{1,\infty}_{t,x}([0,T]\times \mathbb{R}^{3})},\left\Vert u\right\Vert_{L^{\infty}([0,T];W^{2,\infty}(\mathbb{R}^{3}))})$. 
Therefore integrating in time and owing to Lemma \ref{sandwich lemma} we conclude that 
\begin{align*}
(1-\left\Vert u\right\Vert_{\infty})\mathcal{H}_{\hbar}(t)&\leq \mathcal{G}_{\hbar}(t)\leq \mathcal{G}_{\hbar}(0)+C\left(\int_{0}^{t}\mathcal{H}_{\hbar}(\tau)\ {\rm d} \tau+\hbar^{2}t\right)\\
&\leq (1+\left\Vert u^{\mathrm{in}}\right\Vert_{\infty})\mathcal{H}_{\hbar}(0) +C\hbar^{2}t+C\int_{0}^{t}\mathcal{H}_{\hbar}(\tau)\ {\rm d} \tau   
\end{align*}
or equivalently 
\begin{align*}
\mathcal{H}_{\hbar}(t)\leq \frac{1+\left\Vert u^{\mathrm{in}}\right\Vert_{\infty}}{1-\left\Vert u\right\Vert_{\infty}}\mathcal{H}_{\hbar}(0)+\frac{C\hbar^{2}t}{1-\left\Vert u\right\Vert_{\infty}}+\frac{C}{1-\left\Vert u\right\Vert_{\infty}}\int_{0}^{t}\mathcal{H}_{\hbar}(\tau)\ {\rm d} \tau.   \end{align*}
By Gr\"onwall's lemma we deduce 
\begin{align*}
\underset{t\in [0,T]}{\sup}\mathcal{H   }_{\hbar}(t)\leq Ce^{CT}(\mathcal{H}_{\hbar}(0)+\hbar^{2}T),    \end{align*}
which proves \eqref{vanishing of quantum modulated energy statement}. \\
\textbf{The convergence \eqref{convergence of E and B}-\eqref{convergence of rho and J}.} That 
\begin{align}
\underset{t\in [0,T]}{\sup}(\left\Vert (B_{\hbar}-B)(t,\cdot)\right\Vert_{2} +\left\Vert (E_{\hbar}-E)(t,\cdot)\right\Vert_{2})\underset{\hbar\rightarrow 0}{\rightarrow} 0 \label{convergence of Bh and Eh}    
\end{align}
follows immediately from \eqref{vanishing of quantum modulated energy statement}. Since $E_{\hbar}=-\partial_{t}A_{\hbar}-\nabla \varphi_{\hbar}$ and $\mathrm{div}A_{\hbar}=0$, taking the divergence we obtain 
\begin{align*}
\mathrm{div}E_{\hbar}=-\Delta\varphi_{\hbar}=\rho_{\hbar}.    
\end{align*}
Therefore, it follows that 
\begin{align*}
\left\Vert (\rho_{\hbar}-\rho)(t,\cdot)\right\Vert_{H^{-1}}^{2}&=\int_{\mathbb{R}^{3}}\frac{\left\vert \mathcal{F}(\mathrm{div}(E_{\hbar}-E))\right\vert^{2}(\xi)}{1+\left\vert \xi\right\vert^{2} }\ {\rm d} \xi\\
&\leq\int_{\mathbb{R}^{3}}\frac{\left\vert \xi\right\vert^{2} }{1+\left\vert \xi\right\vert^{2}}\left\vert \mathcal{F}(E_{\hbar}-E)\right\vert^{2}(\xi)\ {\rm d} \xi\leq \left\Vert (E_{\hbar}-E)(t,\cdot)\right\Vert_{2}^{2}      
\end{align*}
and hence in view \eqref{convergence of Bh and Eh} it follows that  
\begin{align}
\underset{t\in [0,T]}{\sup}\left\Vert (\rho_{\hbar}-\rho)(t,\cdot)\right\Vert_{H^{-1}}\underset{\hbar\rightarrow 0}{\rightarrow} 0. \label{convergence of rhohbar}     \end{align}
In addition, given a test vector field $\Psi\in W^{1,\infty}(\mathbb{R}^{3};\mathbb{R}^{3})$ with $\left\Vert \Psi\right\Vert_{W^{1,\infty}}\leq 1$ we have 
\begin{align*}
\int_{\mathbb{R}^{3}}(J_{\hbar}-\rho u)\cdot \Psi\ {\rm d} x&=\int_{\mathbb{R}^{3}}(J_{\hbar}-\rho_{\hbar}u)\cdot \Psi\ {\rm d} x+\int_{\mathbb{R}^{3}}(\rho_{\hbar}-\rho)u\cdot \Psi\ {\rm d} x\\
&=-\frac{1}{2}\mathrm{tr}\left((i\hbar\nabla+A_{\hbar}+u)\vee \Psi R_{\hbar}\right)+\int_{\mathbb{R}^{3}}(\rho_{\hbar}-\rho) u\cdot \Psi\ {\rm d} x
\\
&\leq  \sqrt{2\mathcal{U}_{\hbar}(t)}+\left\Vert (u\cdot \Psi)(t,\cdot)\right\Vert_{H^{1}} \left\Vert (\rho_{\hbar}-\rho)(t,\cdot)\right\Vert_{H^{-1}}.      
\end{align*}
Due to the convergence \eqref{convergence of rhohbar} and using that $\underset{t\in [0,T]}{\sup}\mathcal{U}_{\hbar}(t)\underset{\hbar\rightarrow 0}{\rightarrow} 0$ we find  $$\underset{t\in [0,T]}{\sup}\left\Vert (J_{\hbar}-\rho u)(t,\cdot)\right\Vert_{\mathrm{BL}} \underset{\hbar\rightarrow 0}{\rightarrow} 0. $$ 
\qed
