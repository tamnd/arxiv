<!-- head-1: body_md -->
For a positive integer $n$, let $[n]:=\{1,\ldots,n\}$. Let $x=(x_1,\ldots,x_n)$, and let $\mathbb R[x]$ be the set of real polynomials in $x$, with $\mathbb R[x]_d$ the subset of polynomials of degree at most $d$. For $\alpha\in\mathbb N^n$, where $\mathbb N$ includes zero, write $x^\alpha=x_1^{\alpha_1}\cdots x_n^{\alpha_n}$ and $|\alpha|=\sum_i\alpha_i$. Let $\mathbb N_d^n=\{\alpha\in\mathbb N^n:|\alpha|\leq d\}$, and let $v_d(x)=(x^\alpha)_{\alpha\in\mathbb N_d^n}$ be the standard monomial basis. For $J\subseteq[n]$, let $x_J=(x_i)_{i\in J}$ be the corresponding subvector, $\pi_J$ the coordinate projection, and $\mathbb N_d^J=\{\alpha\in\mathbb N_d^n:\alpha_i=0\text{ for }i\notin J\}$. We write $A\succeq0$ if a symmetric matrix $A$ is positive semidefinite. For a set $K$, $\mathcal P(K)$ denotes the Borel probability measures supported on $K$, $\delta_u$ the Dirac measure at $u$, and $(\pi_J)_\#\mu$ the marginal of a measure $\mu$ on the coordinates $J$.

<!-- proposition-1: body_md -->
\label{prop:dense-hierarchy}
For $\kappa\geq d_0$, let $\rho_\kappa^{\mathrm{den}}$ denote the optimal value of
\begin{equation}
\begin{aligned}
    \rho_\kappa^{\mathrm{den}}=\inf_y\quad &L_y(f)\\
    \text{s.t.}\quad &y_0=1,\qquad M_\kappa(y)\succeq0,\\
    &M_{\kappa-d(g_j)}(g_jy)\succeq0,
       &&j=1,\ldots,m,\\
    &L_y(x^\alpha h_j)=0,
       &&|\alpha|\leq2\kappa-\deg h_j,\quad j=1,\ldots,q.
\end{aligned}
\label{eq:dense-relaxation}
\end{equation}
Then $\rho_\kappa^{\mathrm{den}}\leq\rho_{\kappa+1}^{\mathrm{den}}\leq f^\star$. If the constraint description of $K$ is Archimedean (Definition~\ref{def:archimedean}), then $\rho_\kappa^{\mathrm{den}}\uparrow f^\star$ as $\kappa\to\infty$.
<!-- end proposition-1 -->
<!-- definition-1: body_md -->
\label{def:archimedean}
The set $Q(g,h)$ is \emph{Archimedean} if $R-\|x\|^2\in Q(g,h)$ for some $R>0$.
<!-- end definition-1 -->
<!-- example-1: body_md -->
\label{ex:chain}
Consider a trajectory optimization problem over states $s_0,\ldots,s_N$ and controls $u_0,\ldots,u_{N-1}$ that minimizes $\sum_{k=1}^N\ell_k(s_{k-1},u_{k-1},s_k)$ subject to $s_0=s_{\mathrm{init}}$, the discretized dynamics $F_k(s_{k-1},u_{k-1},s_k)=0$, and path constraints $(s_{k-1},u_{k-1},s_k)\in C_k$ for $k\in[N]$, where $\ell_k$, $F_k$, and the inequalities describing $C_k$ are polynomial. Every term and every constraint involves the variables of at most two consecutive steps. Identifying each vector variable with the set of its coordinates, the cliques
\begin{equation}
    \Delta_k=\{s_{k-1},u_{k-1},s_k\},\qquad k\in[N],
    \label{eq:chain-cover}
\end{equation}
therefore form a valid cover, and it satisfies RIP in its natural order, since $\Delta_k\cap(\Delta_1\cup\cdots\cup\Delta_{k-1})=\{s_{k-1}\}\subseteq\Delta_{k-1}$. The number of variables grows with the horizon $N$, whereas the size of each clique does not.
<!-- end example-1 -->
<!-- proposition-2: body_md -->
\label{prop:sparse-hierarchy}
For $\kappa\geq\kappa_0$, let $\rho_\kappa$ denote the optimal value of
\begin{equation}
\begin{aligned}
    \rho_\kappa=\inf_y\quad
        &\sum_{\ell=1}^pL_y(f_\ell)\\
    \text{s.t.}\quad
        &y_0=1,\\
        &M_\kappa^{\Delta_\ell}(y)\succeq0,
            &&\ell=1,\ldots,p,\\
        &M_{\kappa-d(g_j)}^{\Delta_\ell}(g_jy)\succeq0,
            &&j\in G_\ell,\quad\ell=1,\ldots,p,\\
        &L_y(x^\alpha h_j)=0,
            &&\alpha\in\mathbb N_{2\kappa-\deg h_j}^{\Delta_\ell},
              \quad j\in H_\ell,\quad\ell=1,\ldots,p.
\end{aligned}
\label{eq:sparse-relaxation}
\end{equation}
Then $\rho_\kappa\leq\rho_{\kappa+1}$ and $\rho_\kappa\leq\rho_\kappa^{\mathrm{den}}\leq f^\star$. If the cover satisfies RIP and, for each $\ell$, the constraints assigned to $\Delta_\ell$ generate an Archimedean quadratic module in the variables $x_{\Delta_\ell}$ (Definition~\ref{def:archimedean}), then $\rho_\kappa\uparrow f^\star$ as $\kappa\to\infty$~\citep{lasserre2006sparse,yang2024structure}.

<!-- proposition-3: body_md -->
\label{prop:flat-truncation}
Let $y^\star$ be an optimal solution of~\eqref{eq:dense-relaxation}, and suppose that there is an integer $t$ with $d_0\leq t\leq\kappa$ such that
\begin{equation}
    \operatorname{rank}M_t(y^\star)
       =\operatorname{rank}M_{t-d_c}(y^\star)=r.
    \label{eq:dense-flat-truncation}
\end{equation}
Then $y^\star|_{2t}$ has an $r$-atomic representing measure~\eqref{eq:atomic-measure} on $K$, the relaxation is exact, and every atom is a global minimizer.

<!-- proposition-4: body_md -->
\label{prop:sparse-extraction}
Let $y^\star$ be an optimal solution of~\eqref{eq:sparse-relaxation}, and suppose that, at a common order $t$ with $\kappa_0\leq t\leq\kappa$, every clique satisfies~\eqref{eq:local-flat-truncation} and
\begin{equation}
    \operatorname{rank}M_t^{S_{\ell r}}(y^\star)=1
    \qquad\text{for every nonempty overlap }S_{\ell r}.
    \label{eq:sparse-overlap-rank-one}
\end{equation}
Then $\rho_\kappa=f^\star$, and any choice of one atom $a_{\ell i_\ell}$ of $\mu_\ell$ for each $\ell$ assembles into a global minimizer $x^\star$ with $x^\star_{\Delta_\ell}=a_{\ell i_\ell}$ for all $\ell$.

<!-- definition-2: body_md -->
\label{def:partial-flatness}
Let $y$ be feasible for~\eqref{eq:sparse-relaxation} and let $d_\ell\leq t\leq\kappa$. We say that $y$ is \emph{flat on} $\Delta_\ell$ at order $t$ if the flat truncation condition~\eqref{eq:local-flat-truncation} holds for this clique, regardless of the other cliques. We say that $y$ is \emph{partially flat} if it is flat on some, but not necessarily all, of the cliques.

<!-- example-2: body_md -->
\label{ex:counterexample}
Consider the polynomial optimization problem
\begin{equation}
\begin{aligned}
    f^\star=\min_{x,y,u,v}\quad
        &f(x,y,u,v)=(x-y)^2-u^2-v^2\\
    \text{s.t.}\quad
        &x\geq0,\qquad y\geq0,\\
        &1-x^2\geq0,\quad 1-y^2\geq0,\quad
         1-u^2\geq0,\quad 1-v^2\geq0,\\
        &g_{\sigma,\tau}(y,u,v):=
          1+\frac12y-\frac12
          (\sigma u+\tau v+\sigma\tau uv)\geq0,
          \quad (\sigma,\tau)\in\{-1,1\}^2,
\end{aligned}
\label{eq:counterexample-pop}
\end{equation}
with the cliques $\Delta_1=\{x,y\}$ and $\Delta_2=\{y,u,v\}$ and the local objectives $f_1=(x-y)^2$ and $f_2=-u^2-v^2$. The constraints $x\geq0$, $y\geq0$, $1-x^2\geq0$, and $1-y^2\geq0$ are assigned to $\Delta_1$, and the remaining ones to $\Delta_2$, together with the redundant inequality $3-y^2-u^2-v^2\geq0$. This inequality makes the quadratic module of $\Delta_2$ Archimedean. That of $\Delta_1$ is Archimedean because $(1-x^2)+(1-y^2)=2-x^2-y^2$, and the two-clique cover satisfies RIP.

<!-- proposition-5: body_md -->
\label{prop:counterexample}
For Example~\ref{ex:counterexample} and $\kappa=2$, $L$ is an optimal solution of the sparse relaxation, and $\rho_2=L(f)=f^\star$. Moreover, $L$ is flat on $\Delta_1$ with $\operatorname{rank}M_2^{\Delta_1}(L)=\operatorname{rank}M_1^{\Delta_1}(L)=1$, and its unique extracted atom on $\Delta_1$ is $a=(0,0)$.
<!-- end proposition-5 -->
<!-- proof-1: body_md -->
\emph{Feasibility.} Let $v_d^{\Delta}$ denote the vector of monomials of degree at most $d$ in the variables of $\Delta$, for instance $v_1^{\Delta_2}=(1,y,u,v)^\top$. Every constraint has degree at most two, so at $\kappa=2$ all localizing matrices have order one. On $\Delta_\ell$, $L$ is integration against $\mu_\ell$, so by~\eqref{eq:moment-localizing}, for every constraint $p$ assigned to $\Delta_\ell$,
\begin{equation}
    M_2^{\Delta_\ell}(L)
      =\int v_2^{\Delta_\ell}\bigl(v_2^{\Delta_\ell}\bigr)^{\!\top}d\mu_\ell\succeq0,
    \qquad
    M_1^{\Delta_\ell}(pL)
      =\int p\,v_1^{\Delta_\ell}\bigl(v_1^{\Delta_\ell}\bigr)^{\!\top}d\mu_\ell.
    \label{eq:counterexample-matrices}
\end{equation}
Hence $M_1^{\Delta_\ell}(pL)=0$ if $p=0$ on $\operatorname{supp}\mu_\ell$, and $M_1^{\Delta_\ell}(pL)=M_1^{\Delta_\ell}(L)\succeq0$ if $p=1$ on $\operatorname{supp}\mu_\ell$. Since $\operatorname{supp}\mu_1=\{(0,0)\}$ and $\operatorname{supp}\mu_2=\{(0,\pm1,\pm1)\}$, the first case covers $x$, $y$, $1-u^2$, and $1-v^2$, and the second covers $1-x^2$, $1-y^2$, and $3-y^2-u^2-v^2$. The shared moments agree because both measures have marginal $\delta_0$ on $y$. The constraint $g_{\sigma,\tau}$ fits neither case, since $g_{\sigma,\tau}(0,s,t)$ equals $-1/2$ if $(s,t)=(\sigma,\tau)$ and $3/2$ otherwise. With $w_{st}=v_1^{\Delta_2}(0,s,t)=(1,0,s,t)^\top$ and $\frac14\sum_{s,t}w_{st}w_{st}^\top=\mathrm{diag}\left(1,0,1,1\right)$, the second formula in~\eqref{eq:counterexample-matrices} gives
\begin{equation}
    M_1^{\Delta_2}(g_{\sigma,\tau}L)
    =\frac32\mathrm{diag}\left(1,0,1,1\right)-\frac12w_{\sigma\tau}w_{\sigma\tau}^\top
    =
    \begin{pmatrix}
        1 & 0 & -\sigma/2 & -\tau/2\\
        0 & 0 & 0 & 0\\
        -\sigma/2 & 0 & 1 & -\sigma\tau/2\\
        -\tau/2 & 0 & -\sigma\tau/2 & 1
    \end{pmatrix}\succeq0.
    \label{eq:counterexample-localizer}
\end{equation}

\emph{Optimality.} We have $L(f)=-2$. Let $\widetilde L$ be any feasible normalized functional. With $c=(0,1,-1)^\top$, we have $c^\top v_1^{\Delta_1}=x-y$, so $\widetilde L((x-y)^2)=c^\top M_1^{\Delta_1}(\widetilde L)\,c\geq0$. Moreover, $\widetilde L(1-u^2)$ and $\widetilde L(1-v^2)$ are the $(1,1)$ entries of $M_1^{\Delta_2}((1-u^2)\widetilde L)\succeq0$ and $M_1^{\Delta_2}((1-v^2)\widetilde L)\succeq0$. Identity~\eqref{eq:counterexample-lower-certificate} therefore gives
\[
    \widetilde L(f)+2
      =\widetilde L\bigl((x-y)^2\bigr)
       +\widetilde L(1-u^2)+\widetilde L(1-v^2)\geq0.
\]
Hence $\rho_2=L(f)=-2=f^\star$.

\emph{Flatness.} Since $\mu_1=\delta_{(0,0)}$, \eqref{eq:counterexample-matrices} gives $M_d^{\Delta_1}(L)=v_d^{\Delta_1}(0,0)\,v_d^{\Delta_1}(0,0)^\top$ for $d=1,2$, and both matrices have rank one. Since $d_1=1$, $L$ is flat on $\Delta_1$, and $a=(0,0)$ is its unique extracted atom.

<!-- definition-4: body_md -->
\label{def:extendability}
For a variable subset $J\subseteq[n]$, a partial assignment $c\in\mathbb R^{|J|}$ on $J$ is \emph{globally extendable} if $c\in\pi_J(X^\star)$, that is, if some global minimizer agrees with $c$ on $J$. In particular, an extracted atom $a$ on a clique $\Delta$ is globally extendable if $a\in\pi_\Delta(X^\star)$.

\label{def:boundary}
For a selected clique $\Delta\in\mathcal C$, its \emph{boundary} and \emph{interior variable set} are
\begin{equation}
    B_\Delta=\Delta\cap
         \left(\bigcup_{\Gamma\in\mathcal C,\,\Gamma\ne\Delta}\Gamma\right),
    \qquad I_\Delta=\Delta\setminus B_\Delta.
    \label{eq:clique-boundary}
\end{equation}

<!-- lemma-1: body_md -->
\label{lem:common-range}
Let $J$ be contained in a clique and let $y^\circ\in\mathcal F_\kappa$ satisfy
\begin{equation}
    \operatorname{rank}M_\kappa^J(y^\circ)
      =\max_{y\in\mathcal F_\kappa}
         \operatorname{rank}M_\kappa^J(y).
    \label{eq:maximum-block-rank}
\end{equation}
Then, for every $y\in\mathcal F_\kappa$,
\begin{equation}
    \operatorname{range}M_\kappa^J(y)
       \subseteq\operatorname{range}M_\kappa^J(y^\circ),
    \qquad
    \ker M_\kappa^J(y^\circ)\subseteq\ker M_\kappa^J(y).
    \label{eq:common-range}
\end{equation}
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
Put $A=M_\kappa^J(y^\circ)$ and $B=M_\kappa^J(y)$. Convexity gives $(y^\circ+y)/2\in\mathcal F_\kappa$, whose block is $(A+B)/2$. Since $A,B\succeq0$,
\[
    \ker(A+B)=\ker A\cap\ker B.
\]
Thus $\operatorname{rank}(A+B)\geq\operatorname{rank}A$, while maximality of the rank of $A$ gives the reverse inequality. It follows that $\ker A\cap\ker B=\ker A$, which proves the kernel inclusion. Taking orthogonal complements gives the range inclusion.
<!-- end proof-2 -->
<!-- theorem-1: body_md -->
\label{thm:rank-one-extendability}
Assume $\rho_\kappa=f^\star$. Let $J$ be contained in a clique and suppose
\begin{equation}
    \max_{y\in\mathcal F_\kappa}
       \operatorname{rank}M_\kappa^J(y)=1.
    \label{eq:rank-one-optimal-face}
\end{equation}
Then all $y\in\mathcal F_\kappa$ have the same block $M_\kappa^J(y)$, and there is a unique $c\in\mathbb R^{|J|}$ such that
\begin{equation}
    \pi_J(X^\star)=\{c\}.
    \label{eq:unique-extendable-value}
\end{equation}
The value is given by $c_j=L_y(x_j)$ for any $y\in\mathcal F_\kappa$ and $j\in J$. In particular, $c$ is the unique globally extendable assignment on $J$.
<!-- end theorem-1 -->
<!-- proof-3: body_md -->
Choose $y^\circ\in\mathcal F_\kappa$ and write $M_\kappa^J(y^\circ)=ww^\top$, with the entry of $w$ indexed by the constant monomial equal to one. This normalization is possible because $y^\circ_0=1$. Lemma~\ref{lem:common-range} implies that every $M_\kappa^J(y)$ has range contained in $\operatorname{span}\{w\}$. Positive semidefiniteness and $y_0=1$ then force $M_\kappa^J(y)=ww^\top$. In particular, the first moments on $J$ are identical for all optimal sequences; denote their common value by $c$.

For any $x^\star\in X^\star$, its evaluation sequence $y^{x^\star}_\alpha=(x^\star)^\alpha$ is feasible for the sparse relaxation. Value exactness gives $L_{y^{x^\star}}(f)=f^\star=\rho_\kappa$, so $y^{x^\star}\in\mathcal F_\kappa$. Its first moments therefore satisfy $x_J^\star=c$. Since $X^\star$ is nonempty, its projection is exactly $\{c\}$.
<!-- end proof-3 -->
<!-- lemma-2: body_md -->
\label{lem:atomic-replacement}
Let $\bar y\in\mathcal F_\kappa$ satisfy~\eqref{eq:theory-local-measure}--\eqref{eq:theory-full-representation}. Then every atom is conditionally optimal over the interior variables:
\begin{equation}
    z_i\in\operatorname*{arg\,min}_{z\in K_\Delta(b_i)}
                  f_\Delta(z,b_i),
    \qquad i=1,\ldots,r.
    \label{eq:conditional-atomic-optimality}
\end{equation}
<!-- end lemma-2 -->
<!-- proof-4: body_md -->
Suppose that $\widetilde z\in K_\Delta(b_i)$ satisfies $f_\Delta(\widetilde z,b_i)<f_\Delta(z_i,b_i)$ for some $i$. Let $\widetilde\mu_\Delta$ be the measure obtained from $\mu_\Delta$ by moving the atom $(z_i,b_i)$ to $(\widetilde z,b_i)$, and let $\widetilde y$ be the sparse sequence whose moments supported on $\Delta$ are those of $\widetilde\mu_\Delta$ and whose remaining moments are those of $\bar y$.

The sequence $\widetilde y$ is feasible. Every moment shared between $\Delta$ and another clique is supported on $B_\Delta$, where $\widetilde\mu_\Delta$ and $\mu_\Delta$ have the same marginal, so the other cliques and their constraints are untouched. On $\Delta$, the moments come from a probability measure on $K_\Delta$, so the assigned constraints hold. Only the objective term of $\Delta$ changes, and
\begin{equation}
    L_{\widetilde y}(f)
       =\rho_\kappa
         +\lambda_i\bigl(
            f_\Delta(\widetilde z,b_i)-f_\Delta(z_i,b_i)
          \bigr)
       <\rho_\kappa,
    \label{eq:replacement-objective-change}
\end{equation}
contradicting the optimality of $\bar y$.
<!-- end proof-4 -->
<!-- theorem-2: body_md -->
\label{thm:boundary-certificate}
Assume $\rho_\kappa=f^\star$. Let $\bar y\in\mathcal F_\kappa$ admit a locally feasible representation~\eqref{eq:theory-local-measure}--\eqref{eq:theory-full-representation} on $\Delta$. If
\begin{equation}
    \max_{y\in\mathcal F_\kappa}
       \operatorname{rank}M_\kappa^{B_\Delta}(y)=1,
    \label{eq:boundary-rank-one-condition}
\end{equation}
then every atom $(z_i,b_i)$ of $\mu_\Delta$ is globally extendable.
<!-- end theorem-2 -->
<!-- proof-5: body_md -->
Apply Theorem~\ref{thm:rank-one-extendability} to $J=B_\Delta$. There is a value $c$ shared by every global minimizer and every optimal boundary moment block. For each $j\in B_\Delta$, the common rank-one block gives $L_{\bar y}(x_j)=c_j$ and $L_{\bar y}(x_j^2)=c_j^2$. Using the local representation,
\begin{equation}
    \sum_{i=1}^r\lambda_i(b_{i,j}-c_j)^2
       =L_{\bar y}\bigl((x_j-c_j)^2\bigr)=0.
    \label{eq:boundary-zero-variance}
\end{equation}
Since every weight is positive, $b_i=c$ for all $i$.

Now fix an atom $(z_i,c)$ and choose any $x^\star\in X^\star$; by~\eqref{eq:unique-extendable-value}, $x^\star_{B_\Delta}=c$. Write $z^\star=x^\star_{I_\Delta}$ and define $\widetilde x$ by
\[
    \widetilde x_{I_\Delta}=z_i,
    \qquad
    \widetilde x_{[n]\setminus I_\Delta}
       =x^\star_{[n]\setminus I_\Delta}.
\]
The selected clique remains feasible because $(z_i,c)\in K_\Delta$, and every other clique is unchanged because none contains an interior variable of $\Delta$. Hence $\widetilde x\in K$. Lemma~\ref{lem:atomic-replacement} and $z^\star\in K_\Delta(c)$ give $f_\Delta(z_i,c)\leq f_\Delta(z^\star,c)$, and all other objective terms are unchanged, so
\begin{equation}
    f(\widetilde x)
       =f(x^\star)+f_\Delta(z_i,c)-f_\Delta(z^\star,c)
       \leq f^\star.
    \label{eq:certificate-objective}
\end{equation}
As $f^\star$ is the feasible minimum value, $f(\widetilde x) = f^\star$.
Thus $\widetilde x\in X^\star$ and $\widetilde x_\Delta=(z_i,c)=(z_i,b_i)$, so $(z_i,b_i)\in\pi_\Delta(X^\star)$.
