\label{t:main}
Let $F \in \mathbb{Z}[x_1, \ldots, x_n]$ be a homogeneous form of degree $d \geqslant 4$ with $V_F^{*} = \{ \mathbf{0} \}$.

<!-- lemma-1: body_md -->
\label{lem:Weyl}
Let $d\geqslant3$, $X\geqslant1$, and let
\[
 f(x)=\theta x^d+\theta_{d-1}x^{d-1}+\cdots+\theta_1x\in\mathbb{R}[x].
\]
Let $a \in\mathbb{Z}$ and $q \in \mathbb{N}$ satisfy $\gcd(a,q)=1$.
Then
\begin{equation*}                       
 \sum_{1 \leqslant x \leqslant X} e(f(x))
 \ll X^{1+\varepsilon}
 \bigl(X^{-1}+\lambda^{-1}+\lambda X^{-d}\bigr)^{ \frac{1}{d(d-1)}},
\end{equation*}
where $\lambda= q+ X^d|q\theta-a|$, for any $\varepsilon > 0$. Here the implicit constant is independent of $\theta_{d-1}, \cdots, \theta_{1}$.
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
As a consequence of Vinogradov's mean value theorem, it is known (for example, this is recorded in~\cite[Lemma~2.4]{browning-yamagishi}) that
given any $a \in \mathbb{Z}$ and $q \in \mathbb{N}$ such that  $\gcd(a,q) = 1$, $|\theta - a/q| \leqslant q^{-2}$ and $q \leqslant X^d$, we have
$$
\sum_{1 \leqslant x \leqslant X} e(f(x))
\ll X^{1+\varepsilon}
\bigl(X^{-1}+ q^{-1}+ q X^{-d}\bigr)^{ \frac{1}{d(d-1)} }.
$$
The condition $q \leqslant X^d$ is superfluous, because the bound holds trivially if $q > X^d$.
Furthermore, it can be seen from the proof, given in~\cite[Theorem~5.2]{vaughan} and~\cite[Theorem~1.5]{wooleyMVT}, that the implicit constant is independent of
$\theta_{d-1}, \cdots, \theta_{1}$. Thus, the result follows on applying~\cite[Lemma~14.1]{wooley}.
<!-- end proof-1 -->
<!-- remark-1: body_md -->
By simply shifting the polynomial $f$, we can see that the bound in this lemma holds even if the range of summation $[1, X]$ is replaced with any other interval of length $\ll X$.
<!-- end remark-1 -->
<!-- lemma-2: body_md -->
\label{TypeII}
Let $d\geqslant3$, and let
\[
f(x)=\theta x^d+\theta_{d-1}x^{d-1}+\cdots+\theta_1x\in\mathbb{R}[x].
\]
Let $\gamma > 0$, $1\leqslant Q\leqslant P^{1/4}$, $P \ll X \leqslant P$, $P^{1/8} \leqslant M, N \leqslant P$ and $P \ll M N \ll P$.
Let us define
$$
\mathcal{S} = \sum_{ N \leqslant  y <  2 N } \sum_{ \substack{ M \leqslant x <  2 M \\ \gamma P \leqslant  x y \leqslant  X } } \xi_1(x) \xi_2(y) e( f(xy) ),
$$
where $|\xi_1(m)|, |\xi_2(m)| \ll m^{\varepsilon}$ for all $m \in \mathbb{N}$ and any $\varepsilon > 0$. If $\theta\notin\mathfrak M(Q)$,
then
$$
\mathcal{S} \ll  P^{1 + \varepsilon} Q^{- \frac{1}{2 d (d-1)}},
$$
for any $\varepsilon > 0$.
<!-- end lemma-2 -->
<!-- lemma-3: body_md -->
\label{lem:prime-Weyl}
Let $d\geqslant3$, and let
\[
f(x)=\theta x^d+\theta_{d-1}x^{d-1}+\cdots+\theta_1x\in\mathbb{R}[x].
\]
Let $\gamma > 0$.
If $\theta\notin\mathfrak M(Q)$, $1\leqslant Q\leqslant P^{1/4}$ and $P \ll X \leqslant P$, then
\begin{equation*}                      
\sum_{\gamma P \leqslant x\leqslant X}\Lambda(x)e(f(x)) \ll P^{1+\varepsilon}Q^{- \frac{1}{2d(d-1)} },
\end{equation*}
for any $\varepsilon > 0$.
<!-- end lemma-3 -->
<!-- proof-2: body_md -->
By Dirichlet's theorem, there exist $a\in\mathbb{Z}$ and $q \in \mathbb{N}$ such that $\gcd(a,q)=1$,
\[
1\leqslant q\leqslant P^{\frac{d}{2}}
\qquad
\textnormal{and}
\qquad
|q\theta-a|\leqslant P^{- \frac{d}{2}}.
\]
Since $\theta\notin\mathfrak M(Q)$, we have
\begin{equation}\label{eq:mixed-height}
 Q<q+P^d|q\theta-a|\leqslant2P^{\frac{d}{2}}.
\end{equation}
Let us define
\[
 A(y)=\sum_{\substack{rs=y\\ 1 \leqslant r,s\leqslant U}}\mu(r)\Lambda(s)
\qquad
\textnormal{and}
\qquad
 B(y)=\sum_{\substack{r\mid y\\ 1 \leqslant  r\leqslant U}}\mu(r),
\]
where $U=P^{1/8}$. We apply Vaughan's identity~\cite[Section~24]{davenport} to decompose the sum into Type I and Type II sums as follows
\begin{equation}
\label{SSS+++}
\sum_{\gamma P \leqslant x \leqslant X}\Lambda(x)e(f(x)) = T_{\mathrm{I}}-T_{\mathrm{II}},
\end{equation}
where
\[
 T_{\mathrm{I}}=\sum_{1 \leqslant y \leqslant U}\mu(y)
 \sum_{ \gamma P/y \leqslant x \leqslant X/y   }(\log x)e(f(xy))- \sum_{1 \leqslant y \leqslant U^2} A(y)
 \sum_{ \gamma P/y \leqslant x \leqslant X/y  }e(f(xy))
\]
and
\[
T_{\mathrm{II}}=\sum_{ U <  y  \leqslant X } B(y) \sum_{  \substack{ U <  x \leqslant X /y   \\  \gamma P \leqslant xy  } }\Lambda(x)e(f(xy)).
\]
Here $|A(y)|\leqslant\log y$ and $|B(y)|\leqslant\tau(y)$, where $\tau(y)$ is the number of divisors of $y$.

For each $1 \leqslant y \leqslant U^2$, write $ay^d / q = b_y/r_y$ with $r_y \geqslant 1$ and $\gcd(b_y,r_y)=1$.
Let
\[
 \lambda_y = r_y +   \frac{ P^d |r_y  y^d \theta - b_y |}{y^d} =\frac{q+P^d|q\theta-a|}{\gcd(q,y^d)}.
\]
Let $0 < \sigma \leqslant 1 / d$ and $\iota \in \{1, 2 \}$. It follows from \eqref{eq:mixed-height} that
\begin{equation}\label{eq:mixed-height-moments1--}
\lambda_y \leqslant \frac{q  +   P^{d} |q \theta  - a|  }{\gcd(q, y^d)} \leqslant 2 P^{\frac{d}{2}}.
\end{equation}
Similarly, since it follows from \eqref{eq:mixed-height} that
$$
\frac{Q}{2} < q \qquad \textnormal{or} \qquad \frac{Q}{2} < P^d |q  \theta  - a|,
$$
we have
\begin{eqnarray}
\label{eq:mixed-height-moments2--}
\sum_{ M \leqslant y < 2 M }\lambda_y^{-\sigma}
&\ll&
\sum_{ M \leqslant y < 2 M } \frac{\gcd(q, y^d)^{\sigma}}{ Q^{\sigma} }
\\
\notag
&\ll&
\sum_{\ell |  q}  Q^{- \sigma}  \sum_{g | \ell}  \sum_{ \substack{ M \leqslant y < 2 M  \\ \gcd(q, y^d) = \ell  \\ \gcd(q, y) = g  } }
\ell^{\sigma}
\\
\notag
&\ll&
\tau(q) \sum_{\ell |  q}  Q^{- \sigma} \ell^{\sigma} \frac{M}{ \ell^{1/d} }
\\
\notag
&\ll&
M Q^{- \sigma} \tau(q)^2.
\end{eqnarray}
Therefore, by Lemma~\ref{lem:Weyl} (with partial summation when $\iota = 1$), \eqref{eq:mixed-height-moments1--} and \eqref{eq:mixed-height-moments2--}, we obtain
\begin{eqnarray}
\label{typeI}
T_{\mathrm{I}} &\ll&  \sum_{\iota \in \{1, 2\} } (\log P) \sum_{1 \leqslant y \leqslant U^{\iota}}
\left|   \sum_{ \gamma P /y \leqslant x \leqslant X/y} (\log x)^{2 - \iota}  e(f(xy)) \right|
\\
\notag
&\ll&
(\log P)^2 \sum_{1 \leqslant y \leqslant U^2 }
\frac{X^{1+ \varepsilon}}{ y^{1 + \varepsilon }}  \left(   \frac{y}{X} + \lambda_y^{-1}+\lambda_y \frac{y^d}{X^d}  \right)^{ \frac{1}{d(d-1)}}
\\
\notag
&\ll&
(\log P)^2
X^{1 - \frac{1}{d(d-1)} +  \varepsilon} U^{ \frac{ 2}{d(d-1)}  }
\\
\notag
&&
+ (\log P)^3  X^{1 + \varepsilon}
\sup_{ \substack{  [M, 2M) \subseteq [1, U^2] \\ M \geqslant 1 }}
\bigl( Q^{- \frac{1}{d(d-1)}} \tau(q)^2 +
M^{\frac{1}{d-1}} X^{- \frac{1}{d-1}} P^{\frac{1}{2 (d-1)}} \bigr)
\\
\notag
&\ll& P^{1+\varepsilon} \bigl(    Q^{-  \frac{1}{d(d-1)}   } +  P^{-  \frac{1}{4(d-1)} }    \bigr).
\end{eqnarray}

We treat $T_{\mathrm{II}}$ by decomposing it into dyadic sums as follows
\begin{eqnarray*}
T_{\mathrm{II}}
&\ll&
(\log P)^2 \sup \sum_{ \substack{ N \leqslant y < 2 N  \\  U <  y  \leqslant X  }} B(y) \sum_{ \substack{  M \leqslant x < 2 M  \\   U <  x \leqslant X /y \\ \gamma P \leqslant x y } }\Lambda(x) e(f(xy)),
\end{eqnarray*}
where the supremum is over $[M,2M)$ and  $[N, 2N)$ with  $U \leqslant M, N \leqslant X$ and $P \ll M N \ll P$.
Then, by Lemma~\ref{TypeII}, we obtain
$$
T_{\mathrm{II}} \ll  P^{1 + 2 \varepsilon} Q^{- \frac{1}{2 d (d-1)}}.
$$
Finally, on substituting this estimate and \eqref{typeI} into \eqref{SSS+++}, it follows that
$$
\sum_{ \gamma P  \leqslant x \leqslant X}\Lambda(x) e(f(x)) \ll  |T_{\mathrm{I}}| + |T_{\mathrm{II}}|  \ll  P^{1 +  \varepsilon} Q^{- \frac{1}{2 d (d-1)}}.
$$

\label{lem:pure-terminal}
Suppose  $P^{1/100}\leqslant L\leqslant P^{d/2}$ and $\alpha\in\mathfrak n(L)$. Then
\begin{equation*}                         
 \sum_{1 \leqslant x \leqslant P} \omega_1 \left( \frac{x}{P} \right) \Lambda(x) \,  e\bigl(\Phi(x)\bigr)
\ll P^{1 + \varepsilon} L^{-  \frac{1}{ 4d^2(d-1)  } },
\end{equation*}
for any $\varepsilon > 0$. Here the implicit constant is independent of the coefficients of $\Phi(x) - c \alpha x^d$.
<!-- end lemma-4 -->
<!-- proof-3: body_md -->
Since $\alpha\notin\mathfrak M(L)$, we have $c \alpha \notin \mathfrak M(L/|c|)$.
If $P^{1/4} \leqslant L/|c|$, then it follows that $c \alpha \notin \mathfrak M(P^{1/4})$.
Let $Q = \min\{L/|c|,P^{1/4}\}$. Let $\gamma > 0$ be sufficiently small.
Therefore, by partial summation and Lemma~\ref{lem:prime-Weyl}, we have
\begin{eqnarray*}
\left| \sum_{1 \leqslant x \leqslant P} \omega_1 \left( \frac{x}{P} \right) \Lambda(x) \,  e\bigl(\Phi(x)\bigr) \right|
&=&
P^{-1} \left| \int_{1}^P  \omega_1' \left( \frac{t}{P} \right)   \sum_{\gamma P  \leqslant x \leqslant t}  \Lambda(x) \,   e\bigl(\Phi(x)\bigr) \mathrm{d} t \right|
\\
&\ll&
P^{1 + \varepsilon} Q^{- \frac{1}{2d(d-1)}}
\\
&\ll&
\begin{cases}
P^{1 + \varepsilon} L^{- \frac{1}{2d(d-1)}} & \mbox{if } P^{1/100}  \leqslant L  \leqslant |c|P^{1/4},  \\
P^{ 1 + \varepsilon  -  \frac{1}{8d(d-1)}} & \mbox{if } |c|P^{1/4}< L \leqslant P^{d/2}.
\end{cases}
\end{eqnarray*}
In either case, we obtain the bound
$$
\ll P^{1 + \varepsilon} L^{-  \frac{1}{ 4d^2(d-1)  } },
$$
which proves the lemma.

<!-- lemma-5: body_md -->
\label{lem:mixed-terminal}
Suppose $P^{1/100}\leqslant L\leqslant P^{d/2}$ and $\alpha\in\mathfrak n(L)$.  Then
\begin{equation*}                          
 \sum_{ 1 \leqslant  x \leqslant P }\sum_{ 1 \leqslant  y \leqslant P}
 \omega_1 \left( \frac{x}{P} \right) \omega_2 \left( \frac{y}{P} \right) \Lambda(x) \, \Lambda(y) \, e\bigl(\Phi(x,y)\bigr)
 \ll P^{2 + \varepsilon} L^{- \frac{1}{d(d-1)(d-2)} },
\end{equation*}
for any $\varepsilon > 0$. Here the implicit constant is independent of the coefficients of $\Phi(x, y) - c \alpha x^{d-1}y$.
<!-- end lemma-5 -->
<!-- proof-4: body_md -->
Since $\alpha\in\mathfrak M(2L)$, there exist $a_0 \in \mathbb{Z}$ and $q_0 \in \mathbb{N}$ such that $\gcd(a_0,q_0)=1$, $1\leqslant q_0\leqslant2L$ and $P^d|q_0\alpha-a_0|\leqslant2L$.
Write $ca_0/q_0=a/q$ with $q\geqslant1$ and $\gcd(a,q)=1$.                                                  
Then it follows that
$$
1 \leqslant q \leqslant 2L \qquad \textnormal{and} \qquad P^d|q c \alpha - a|\leqslant 2 |c| L.
$$
Since $c \alpha \notin \mathfrak M(L/|c|)$, we have
\begin{equation}\label{eq:mixed-height+}
\frac{L}{|c|} < q +  P^d |q c \alpha  - a| \ll L.
\end{equation}
For each $1\leqslant h\leqslant P$, write $ah/q = b_h/r_h$ with $r_h\geqslant1$ and $\gcd(b_h,r_h)=1$, and put
\[
\lambda_h=r_h+P^{d-1}|r_h h c \alpha - b_h|
 =\frac{q  +   hP^{d-1} |q c \alpha   - a|  }{\gcd(q,h)}.
\]

Let $0 < \sigma < 1$. It follows from \eqref{eq:mixed-height+} that
$$
\lambda_h \leqslant \frac{q  +   P^{d} |q c \alpha   - a|  }{\gcd(q,h)} \ll L.
$$
Therefore, we have
\begin{equation}\label{eq:mixed-height-moments1}
\sum_{1 \leqslant h\leqslant P} \lambda_h^\sigma \ll P L^\sigma.
\end{equation}
Similarly, since it follows from \eqref{eq:mixed-height+} that
$$
\frac{L}{2|c|} < q \qquad \textnormal{or} \qquad \frac{L}{2|c|} < P^d |q c \alpha  - a|,
$$
we obtain
\begin{eqnarray}
\label{eq:mixed-height-moments2}
\sum_{1 \leqslant h\leqslant P}\lambda_h^{-\sigma}
&\ll&
\sum_{1 \leqslant h\leqslant P}
\left( \frac{\gcd(q,h)^{\sigma}}{L^{\sigma}} + \frac{\gcd(q,h)^{\sigma} P^{\sigma}}{ h^{\sigma} L^{\sigma}  } \right)
\\
\notag
&\ll&
\sum_{\ell |  q} \ell^{\sigma} L^{- \sigma} \sum_{ \substack{ 1 \leqslant h\leqslant P \\ \gcd(h, q) = \ell  } }
\left( 1 + \frac{ P^{\sigma}}{ h^{\sigma} } \right)
\\
\notag
&\ll&
\sum_{\ell | q } \ell^{\sigma} L^{- \sigma } \frac{P}{\ell }
\\
\notag
&\ll& P L^{-\sigma} \tau(q)
\\
\notag
&\ll& P^{1 + \varepsilon} L^{-\sigma}.
\end{eqnarray}

Let
\[
S = \sum_{ 1 \leqslant  x \leqslant P }\sum_{ 1 \leqslant  y \leqslant P} \omega_1 \left( \frac{x}{P} \right) \omega_2 \left( \frac{y}{P} \right) \Lambda(x) \, \Lambda(y) \, e\bigl(\Phi(x,y)\bigr).
\]
Let $\gamma > 0$ be sufficiently small.
By the Cauchy--Schwarz inequality, we obtain
\begin{eqnarray*}
|S|^2 &\ll&  P (\log P)^2  \sum_{ \gamma P \leqslant  x \leqslant P } \sum_{ 1 \leqslant  y_1, y_2 \leqslant P}
\\
&&  \omega_2 \left( \frac{y_1}{P} \right) \omega_2 \left( \frac{y_2}{P} \right) \Lambda(y_1) \, \Lambda(y_2) \, e\bigl(\Phi(x,y_1) - \Phi(x,y_2)\bigr)
\\
&\ll& P^3 (\log P)^4 +  P (\log P)^4 \sum_{ \substack{ 1 \leqslant y_1,  y_2 \leqslant P \\ y_1 \neq y_2  } }
\\
&&
\left| \sum_{\gamma P  \leqslant x \leqslant P}  e\bigl(  c \alpha (y_1 - y_2) x^{d-1} + \nu(x, y_1, y_2)  \bigr) \right|,
\end{eqnarray*}
where
$$
\nu(x, y_1, y_2) = \Phi(x,y_1) -  c \alpha y_1  x^{d-1}  - \Phi(x,y_2) + c \alpha y_2  x^{d-1}.
$$
In particular, $\deg_x \nu \leqslant d - 2$. Therefore, it follows from Lemma~\ref{lem:Weyl}, \eqref{eq:mixed-height-moments1} and \eqref{eq:mixed-height-moments2} that
\begin{eqnarray*}
|S|^2 &\ll&  P^3 (\log P)^4 +  P^2 (\log P)^4 \sum_{1 \leqslant h \leqslant P} P^{1+ \varepsilon}  \bigl(P^{-1}+\lambda_h^{-1}+\lambda_h P^{-d + 1}\bigr)^{ \frac{1}{(d-1)(d-2)}}
\\
&\ll&
P^{3 + 2 \varepsilon}
\left( P^{1 - \frac{1}{(d-1)(d-2)} } + P^{1 + \varepsilon}  L^{- \frac{1}{(d-1)(d-2)}} + L^{ \frac{1}{(d-1)(d-2)}} P^{ 1 -  \frac{1}{ d-2 }} \right)
\\
&\ll&
P^{4 + 2 \varepsilon} L^{- \frac{2}{d (d-1)(d-2) }},
\end{eqnarray*}
where the final inequality follows on recalling that $P^{1/100}\leqslant L\leqslant P^{d/2}$ and $d\geqslant4$.
The lemma then follows by taking the square roots.
