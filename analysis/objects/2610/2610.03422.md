<!-- defn-1: body_md -->
\label{def-good}
A complex hyperbolic triangle group $\Gamma=\langle R_1,R_2,R_3\rangle$ is said to
be \emph{good} if the generators satisfy both of the following conditions:
\begin{enumerate}
\item[(a)] ${\rm tr}(R_2R_3)\le {\rm tr}(R_1R_3)\le {\rm tr}(R_1R_2)\le {\rm tr}(R_1R_3R_2R_3)$;
\item[(b)] if any of ${\rm tr}(R_2R_3)$, ${\rm tr}(R_1R_3)$, ${\rm tr}(R_1R_2)$, and ${\rm tr}(R_1R_3R_2R_3)$ is less than $3$, then it has the form $1+2\cos(2\pi/p)$ for some $p\in{\mathbb N}$ with $p\ge 2$.
\end{enumerate}

    Let $F:{\mathbb C}\longrightarrow {\mathbb R}$ be given by
    \begin{eqnarray*}
       F(\tau) & = & |\tau|^4-8\mathrm{Re}(\tau^3)+18|\tau|^2-27 \\
       & = & f_1(\tau)^2+4f_2(\tau)^3,

is coprime to $\tilde{p}_2^{(n)}$, then perform reduce. Denote the resulting ordered triple of reflections by
$\{R_1^{(n+1)},\,R_2^{(n+1)},\,R_3^{(n+1)}\}$.
Note that 
$$
x_{n+1}={\rm tr}(R_1^{(n+1)}R_3^{(n+1)})=1+2\cos(2\pi/\tilde{p}_2).
$$
\item[(3b)] Given $\{\widetilde{R}_1^{(n)},\,\widetilde{R}_2^{(n)},\,\widetilde{R}_3^{(n)}\}$, if 
$$
\tilde{x}_n =1+2\cos(2\pi/\tilde{p}_2^{(n)}),\quad 
\tilde{y}_n=1+2\cos(2k\pi/\tilde{p}_3^{(n)}),
$$ 
where $2 \le k \le \tilde{p}_3^{(n)}/2$
is coprime to $\tilde{p}_3^{(n)}$, then apply $\iota_{23}$ and then perform reduce.
Denote the resulting ordered triple of reflections by
$\{R_1^{(n+1)},\,R_2^{(n+1)},\,R_3^{(n+1)}\}$. Note that 
$$
x_{n+1}={\rm tr}(R_1^{(n+1)}R_3^{(n+1)})=1+2\cos(2\pi/\tilde{p}_3).
$$
\item[(3c)] Given $\{\widetilde{R}_1^{(n)},\,\widetilde{R}_2^{(n)},\,\widetilde{R}_3^{(n)}\}$, if 
$$
\begin{array}{ll}
\tilde{x}_n =1+2\cos(2\pi/\tilde{p}_2^{(n)}),\quad &
\tilde{y}_n =1+2\cos(2\pi/\tilde{p}_3^{(n)}), \\
\tilde{z}_n=1+2\cos(2k\pi/\tilde{p}_4^{(n)}), & 
\end{array}
$$ 
where $2\le k\le \widetilde{p}_4^{(n)}/2$ is coprime to $\tilde{p}_4^{(n)}$, then apply $\iota_3$ followed by $\iota_{23}$ and then perform reduce.
Denote the resulting ordered triple of reflections by
$\{R_1^{(n+1)},\,R_2^{(n+1)},\,R_3^{(n+1)}\}$. Note that 
$$
x_{n+1}={\rm tr}\bigl(R_1^{(n+1)}R_3^{(n+1)}\bigr)=1+2\cos(2\pi/\tilde{p}_4).

\end{array}
$$
then stop.
\end{enumerate}
It remains to show that this algorithm terminates after finitely many steps.



Write $C_j^{(n)}$ for the mirror of $R_j^{(n)}$ and $\widetilde{C}_j^{(n)}$ for the mirror of $\widetilde{R}_j^{(n)}$. 
Observe that, at each stage, $R_2^{(n)}$ is one of $R_2$ or $R_3$ and $R_3^{(n)}$ is the other one. Similarly
for $\widetilde{R}_2^{(n)}$ and $\widetilde{R}_3^{(n)}$. This means that for all $n$ 
$$
C_2^{(n)}\cap C_3^{(n)}=\widetilde{C}_2^{(n)}\cap \widetilde{C}_3^{(n)}=C_2\cap C_3=v_1.
$$
We claim that for all $n$
\begin{equation}\label{eq-claim-good}
\rho\bigl(v_1,\widetilde{C}_1^{(n)}\bigr)=\rho\bigl(v_1,C_1^{(n)}\bigr),\qquad 
\rho\bigl(v_1,C_1^{(n+1)}\bigr)<\rho\bigl(v_1,\widetilde{C}_1^{(n)}\bigr).
\end{equation}
First consider the process reorder. 
If we apply $\iota_{23}$ then $\iota_{23}(R_1)=R_1$, so its mirror is $C_1$ and $\rho(v_1,C_1)$ remains unchanged.
If we apply $\iota_3$ then $\iota_3(R_1)=R_3R_1R_3$ whose mirror is $R_3(C_1)$. We have
$$
\rho\bigl(v_1,R_3(C_1)\bigr)=\rho\bigl(R_3(v_1),R_3(C_1)\bigr)=\rho(v_1,C_1).
$$
For the first equality we used the fact that $R_3$ fixes $v_1$ and for the second equality we use that $R_3$ is an isometry. 
Therefore, when we apply the process reorder the distance between $v_1$ and $C_1$ does not change. This proves
the first identity from \eqref{eq-claim-good}
$$
\rho\bigl(v_1,\widetilde{C}_1^{(n)}\bigr)=\rho\bigl(v_1,C_1^{(n)}\bigr).
$$



Now we consider the process reduce. 
Let 
$$
\tilde{v}_2^{(n)}=\tilde{C}_3^{(n)}\cap \tilde{C}_1^{(n)}
\quad \hbox{ and }\quad v_2^{(n+1)}=C_3^{(n+1)}\cap C_1^{(n+1)}
$$ 
be the fixed
points of $\tilde{R}_1^{(n)}\tilde{R}_3^{(n)}$ and $R_1^{(n+1)}R_3^{(n+1)}$. By construction, 
$R_1^{(n+1)}R_3^{(n+1)}$ is a power of $\tilde{R}_1^{(n)}\tilde{R}_3^{(n)}$ and so their fixed points are the same. 
That is $\tilde{v}_2^{(n)}=v_2^{(n+1)}$ and we choose to simply write it as $\tilde{v}_2^{(n)}$. 



Also, let $\tilde{u}_n$ be the foot of the perpendicular from $v_1$ to $\widetilde{C}_1^{(n)}$ and 
$u_{n+1}$ be the foot of the perpendicular from $v_1$ to $C_{n+1}$. Note that
$$
\rho\bigl(v_1,\widetilde{C}_1^{(n)}\bigr)=\rho(v_1,\tilde{u}_n),\qquad
\rho\bigl(v_1,C_1^{(n+1)}\bigr)=\rho(v_1,u_{n+1}).
$$
We now use the sine rule (see Theorem 3.2.1 of Goldman \cite{gol}) on the
complex triangle with vertices $v_1$, $\tilde{v}_2^{(n)}$ and $\tilde{u}_n$. This triangle has internal complex angles 
$k\pi/\tilde{p}_2^{(n)}$ at $\tilde{v}_2^{(n)}$ and $\pi/2$ at $\tilde{u}_n$. We do the same with the 
complex triangle with vertices $v_1$, $\tilde{v}_2^{(n)}$ and $u_{n+1}$. This has internal complex angles 
of $\pi/\tilde{p}_2^{(n)}$ at $\tilde{v}_2^{(n)}$ and $\pi/2$ at $u_{n+1}$. Therefore
\begin{eqnarray*}
\sinh\left(\frac{\rho(v_1,C_1^{(n+1)})}{2}\right) 
& = & \sinh\left(\frac{\rho(v_1,u_{n+1})}{2}\right) \\
& = & \sin\left(\frac{\pi}{\tilde{p}_2^{(n)}}\right)\sinh\left(\frac{\rho(v_1,\tilde{v}_2^{(n)})}{2}\right) \\
& < & \sin\left(\frac{k\pi}{\tilde{p}_2^{(n)}}\right)\sinh\left(\frac{\rho(v_1,\tilde{v}_2^{(n)})}{2}\right) \\
& = & \sinh\left(\frac{\rho(v_1,\tilde{u}_n)}{2}\right) \\
& = & \sinh\left(\frac{\rho( v_1,\tilde{C}_1^{(n)})}{2}\right).
\end{eqnarray*}
This completes the proof of \eqref{eq-claim-good}. 



We are now in a position to prove that the algorithm terminates after finitely many steps. Suppose for a contradiction
that the algorithm does not terminate. Then it will give an infinite family $\{R_1^{(n)}\}$ of reflections in $\Gamma$ with
mirrors $\{C_1^{(n)}\}$. Let $K$ be the closed ball with centre $v_1$ and radius $r=\rho(v_1,C_1)$. 
From \eqref{eq-claim-good} we have 
$$
0<\ldots<\rho\bigl(v_1,C_1^{(n+1)}\bigr)<\rho\bigl(v_1,C_1^{(n)}\bigr)<\cdots<\rho(v_1,C_1^{(1)})=\rho(v_1,C_1)=r.
$$
Therefore the complex geodesics $\{C_1^{(n)}\}$ are distinct and they all intersect $K$. Since $R_1^{(n)}$ fixes 
$C_1^{(n)}$ this means that $R_1^{(n)}(K)\cap K$ is nonempty for all $n$. Therefore $\Gamma$ does not act
properly discontinuously on ${\bf H}^2_{\mathbb C}$, which is a contradiction to our hypothesis that $\Gamma$ is
discrete.



Therefore, after finitely many steps we obtain an ordered triple $\{\tilde{R}_1^{(n)},\,\tilde{R}_2^{(n)},\,\tilde{R}_3^{(n)}\}$
so that $\tilde{x}_n\le\tilde{y}_n\le\tilde{z}_n$ and 
$$
\begin{array}{ll}
\tilde{x}_n=1+2\cos(2\pi/\tilde{p}_2^{(n)}), & \hbox{ or } \quad \tilde{x}_n\ge 3, \\
\tilde{y}_n=1+2\cos(2\pi/\tilde{p}_3^{(n)}), & \hbox{ or } \quad \tilde{y}_n\ge 3, \\
\tilde{z}_n=1+2\cos(2\pi/\tilde{p}_4^{(n)}), & \hbox{ or } \quad \tilde{z}_n\ge 3.
\end{array}
$$
We also have 
$$
w={\rm tr}(R_2R_3)={\rm tr}\bigl(\tilde{R}_2^{(n)}\tilde{R}_3^{(n)}\bigr)=1+2\cos(2\pi/p_1).
$$
By hypothesis, $p_1$ was chosen to be the smallest order of an elliptic element of $\Gamma$. This means that
$p_1\le \tilde{p}_2^{(n)}$ and so $w\le \tilde{x}_n$. This proves the result.

\tilde{g}_1(u,v,y,z) & = &

\frac{\partial \tilde{g}_1}{\partial u}(u,v,y,z)

8y^2 + 4yz - 24y + 16+ 4v^2(1-y) & \le & 8y^2+12y-24y+16+26-26y \\
& = & 2(y-3)(4y-7)\le 0.
\end{eqnarray*}
Hence
$$
\frac{\partial \tilde{g}_1}{\partial u}(u,v,y,z)<0
$$
and so 
$$
\tilde{g}_1(u,v,y,z)\le \tilde{g}_1(v,v,y,z).

variables $y$ and $z$. It is given by
$$
H_{u,v}(y,z) = \left(\begin{matrix} 
16u+6y+2z+8 & 4u+2y+4 \\ 4u+2y+4 & 8 \end{matrix}\right).
$$ 
Note that $\det(H_{u,v}(y,z))=4(12-2u-y)(2u+y)+16(z-y)+48>0$. Therefore, we see $H_{u,v}(y,z)$
