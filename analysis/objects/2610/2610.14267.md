<!-- thm-1: body_md -->
\label{mainpositive}
Let $p>0$ and $\alpha>-1$, and let $f:\mathbb D\to \Omega$ be a
quasiconformal homeomorphism. Then the following conditions
are equivalent:
\begin{enumerate}
    \item[(1)] $\displaystyle f\in A_{\alpha}^{p}(\mathbb{D})$;
\vspace{2 mm}
\item[(2)] $\displaystyle \int_0^{\infty}r^{p-1} |U_r|^{\frac{\alpha+2}{2}}\,dr<\infty$;
\vspace{2 mm}
    \item[(3)]
        $\displaystyle\int_{0}^{\infty}
        r^{p-1}\mathcal{H}^{1}(E_r)^{\alpha+2}\,dr
        <\infty$;
\vspace{2 mm}
    \item[(4)] $\displaystyle a_f\in A_{\alpha+p}^{p}(\mathbb{D})$.
\end{enumerate}
\vspace{2 mm}
When $-1<\alpha\leq 0$, the above conditions are further
equivalent to:
\vspace{2 mm}
\begin{enumerate}
    \item[(5)]
        $\displaystyle\int_{0}^{\infty}
        r^{p-1}\mathcal{H}^{1}(F_r)^{\alpha+2}\,dr
        <\infty.$
\end{enumerate}
Moreoever, $(5)$ implies $(3)$ for every $\alpha>-1$ and $p>0$.

\label{lem:level-set-slicing}
Let $f:\mathbb D\to \Omega$ be a $K-$quasiconformal homeomorphism, denote $g:=f^{-1}:\Omega\to\mathbb D$, and, for $w\neq0$, let
$\tau(w):=iw/|w|$. We choose a Borel measurable representative of $Dg$ and define
\begin{equation}\label{defLr}
    L(r):=
	\int_{\Omega\cap S(0,r)}
	|Dg(w)\tau(w)|\,d\mathcal H^1(w),
	\quad \text{for}\quad r>0.
\end{equation}
	Then $L:(0,\infty)\to[0,\infty]$ is measurable. Moreover, for almost
	every $r>0$, $F_r$ is countably $1$-rectifiable and
\begin{equation}\label{eq:Hausdorff-F_r-Lr}
    \mathcal H^1(F_r)
	=
	L(r)
	\leq
	\int_{\Omega\cap S(0,r)}
	|Dg(w)|\,d\mathcal H^1(w)<\infty.
\end{equation}
	In particular, $L$ is a measurable representative of the function
	$r\mapsto\mathcal H^1(F_r)$.
<!-- end lem-5 -->
<!-- proof-1: body_md -->
Since $g:\Omega\to\mathbb D$ is also $K$-quasiconformal, the change of variables formula gives
$$\int_\Omega |Dg(w)|^2\,dA(w)
    \le K\int_\Omega J_g(w)\,dA(w)
    =K\pi.$$
Consequently, polar coordinates and the Cauchy--Schwarz
inequality imply that, for almost every $r>0$,
$$
\int_{\Omega\cap S(0,r)}|Dg(w)|\,d\mathcal H^1(w)
\le
(2\pi r)^{1/2}
\left(
\int_{\Omega\cap S(0,r)}
|Dg(w)|^2\,d\mathcal H^1(w)
\right)^{1/2}
<\infty.
$$

By the Sobolev chain rule and slicing in local polar
coordinates \cite[Section~4.9.2]{EG}, together with the
continuity of $g$, for almost every such $r$ the restriction
of $g$ to each component of $\Omega\cap S(0,r)$ is locally
absolutely continuous, with tangential derivative
$Dg(w)\tau(w)$. 

Applying the length formula on compact subarcs, using
injectivity, and exhausting the countably many components,
we obtain that $F_r$ is countably $1$-rectifiable and
$$
\mathcal H^1(F_r)
=
\int_{\Omega\cap S(0,r)}
|Dg(w)\tau(w)|\,d\mathcal H^1(w)
=L(r).
$$
The asserted inequality follows from $|\tau(w)|=1$.

Finally, extending $Dg$ by zero outside $\Omega$, we have
$$
L(r)
=
r\int_0^{2\pi}
|Dg(re^{i\theta})(ie^{i\theta})|\,d\theta,
\quad r>0.
$$
Since the integrand is nonnegative and Borel measurable in
$(r,\theta)$, by Tonelli's theorem $L$ is measurable.

\label{lem:tents}
Let $G\subset\partial\mathbb D$ and $V\subset\mathbb D$
be measurable sets satisfying 
\begin{equation}\label{densityassumption}
    \mathcal H^1\bigl(G\cap S(z)\bigr)
    \ge \frac12\mathcal H^1\bigl(S(z)\bigr)
    \quad\text{for every }z\in V.
\end{equation}
	Then, for every $\alpha>-1$, we have
    \begin{equation}\label{eq:weighted-tent}
    |V|_\alpha
    :=\int_V(1-|z|)^\alpha\,dA(z)
    \le \frac{2}{\alpha+1}\,
        \mathcal H^1(G)^{\alpha+2}.
\end{equation}
<!-- end lem-6 -->
<!-- proof-2: body_md -->
For $z=\rho e^{i\theta}\in\mathbb D$, the arc
$\{e^{i(\theta+t)}:|t|<1-\rho\}$ lies in $S(z)$.
Hence
$$
\mathcal H^1(S(z))\ge 2(1-|z|),
$$
and \eqref{densityassumption} gives
\begin{equation}\label{inclusionofV}
V\subset
\{z\in\mathbb D:1-|z|\le\mathcal H^1(G)\}.
\end{equation}
In particular, if $\mathcal H^1(G)=0$, then $V=\emptyset$. By rotational invariance and Tonelli's theorem, we have
$$
\int_0^{2\pi}
\frac{\mathcal H^1(G\cap S(\rho e^{i\theta}))}
     {\mathcal H^1(S(\rho e^{i\theta}))}\,d\theta
=
\mathcal H^1(G),
\quad 0<\rho<1.
$$
Using \eqref{densityassumption},
\eqref{inclusionofV} and polar coordinates, we obtain
\begin{align*}
|V|_\alpha
&\le
2\int_{\max\{0,1-\mathcal H^1(G)\}}^1
(1-\rho)^\alpha\rho
\int_0^{2\pi}
\frac{\mathcal H^1(G\cap S(\rho e^{i\theta}))}
     {\mathcal H^1(S(\rho e^{i\theta}))}\,d\theta\,d\rho\\
&=
2\mathcal H^1(G)
\int_{\max\{0,1-\mathcal H^1(G)\}}^1
(1-\rho)^\alpha\rho\,d\rho\\
&\le
2\mathcal H^1(G)\int_0^{\mathcal H^1(G)}s^\alpha\,ds\\
&=
\frac{2}{\alpha+1}\mathcal H^1(G)^{\alpha+2}.
\end{align*}

\label{lem:shifted-density}
Let $f:\mathbb{D}\to \Omega$ be a $K$-quasiconformal homeomorphism, and set $d_0:=\operatorname{dist}(0,\partial\Omega)$. Then there exists a constant $\Lambda=\Lambda(K)>1$, such that
for every positive $r\ge d_0$ and every $z\in\mathbb D$ with
$|f(z)|\ge\Lambda r$, we have
    $$\mathcal H^1(E_r\cap S(z))
    \ge \frac12\mathcal H^1(S(z)).$$
<!-- end lem-7 -->
<!-- proof-3: body_md -->
We choose
	$y_0\in\partial\Omega$ with $|y_0|=d_0$ and let $h:=f-y_0$. Then $h:\mathbb D \to \Omega-y_0$ is $K$-quasiconformal and $h(z)\neq0$ for every $z\in \mathbb D$. Let $C(K)$ be the constant in
	Lemma~\ref{lem:Astala-Koskela-annulus-estimate}. Choose a constant $M_0=M_0(K)>1$ such that ${C(K)}/{\log M_0}\leq 1/2$,
		and set $\Lambda=\Lambda(K):=2M_0+2>1$.
    If positive $r\geq d_0$ and $|f(z)|\geq\Lambda r$, then
		$$
			|h(z)|\geq |f(z)|-|y_0|\geq \Lambda r-d_0\geq(2M_0+1)r.$$
	
 For every $\omega\in S(z)\setminus(E_r\cup N_f)$, we therefore have
        $$|h(\omega)|
    =|f(\omega)-y_0|
    \le r+d_0 \le2r
    <\frac{|h(z)|}{M_0}.$$ 
  Applying Lemma \ref{lem:Astala-Koskela-annulus-estimate} to $h$ and using $\mathcal H^1(N_f)=0$ yields that
\begin{align*}
    \mathcal H^1(S(z)\setminus E_r)
    &\le
    \mathcal H^1\left(
    \left\{\omega\in S(z)\setminus N_f:
    |h(\omega)|<\frac{|h(z)|}{M_0}\right\}
    \right)\\
    &\le \frac{C_K}{\log M_0}\mathcal H^1(S(z))\\
    &\le \frac12\mathcal H^1(S(z)).
\end{align*}
Therefore, we conclude that
$$\mathcal H^1(E_r\cap S(z))=\mathcal H^1(S(z))- \mathcal H^1(S(z)\setminus E_r)\ge \frac{1}{2}\mathcal H^1(S(z)),$$
which proves the lemma.

\label{Urestimate}
Let $\alpha>-1$, and let $f:\mathbb D \to \Omega$ be a $K$-quasiconformal homeomorphism. Then there exist constants $\Lambda=\Lambda(K)>1$ such that
\begin{equation}\label{eq:weighted-shifted-superlevel-upper-bound}
			|U_{\Lambda r}|_\alpha
			\leq
			\frac{2}{\alpha+1}\,\mathcal H^1(E_r)^{\alpha+2},\quad \text{for every } r>0,
		\end{equation}
and
\begin{equation}\label{eq:weighted-shifted-superlevel-lower-bound}
			|U_r|_\alpha
			\geq
	C(K,\alpha)\,\mathcal H^1(E_{2r})^{\alpha+2},\quad \text{for every } r>R_0,
		\end{equation}        
        where $$R_0:=\max_{|z|=\frac{1}{2}}|f(z)|,\quad C(K,\alpha)=\frac{1}{2(\alpha+1)}
\left(\frac{\log 2}{4\pi K}\right)^{\alpha+1}.$$
<!-- end lem-8 -->
<!-- proof-4: body_md -->
Recall that $d_0:=\operatorname{dist}(0,\partial\Omega)$. 
By Lemma~\ref{lem:shifted-density}, for $r\ge d_0$, we have
$$
\mathcal H^1(E_r\cap S(z))
\geq
\frac12\mathcal H^1(S(z)),\quad z\in U_{\Lambda r} .
$$
Combining with Lemma \ref{lem:tents}, it follows that
		$$
		|U_{\Lambda r}|_\alpha
		\leq
		\frac{2}{\alpha+1}\,\mathcal H^1(E_r)^{\alpha+2}.
		$$

        On the other hand, if $0<r<d_0$, then $\mathcal H^1(E_r)=2\pi$. Thus,
		$$|U_{\Lambda r}|_\alpha\le
		|\mathbb D|_\alpha
		=
		\frac{2\pi}{(\alpha+1)(\alpha+2)}\le \frac{2}{\alpha+1}\,\mathcal H^1(E_r)^{\alpha+2}.
		$$
		The last inequality follows from $\alpha>-1$. This gives \eqref{eq:weighted-shifted-superlevel-upper-bound} for every $r>0$. 

    To prove the lower bound \eqref{eq:weighted-shifted-superlevel-lower-bound}, fix $r>R_0$. We may assume that $\mathcal H^1(E_{2r})>0$. For $\omega\in E_{2r}$, define 
$$
a(\omega)
:=\max\{t\in[1/2,1):|f(t\omega)|\leq r\},\quad\delta(\omega):=1-a(\omega).
$$
Then $a(\omega)>1/2$, $|f(a(\omega)\omega)|=r$ and the radial segment $\{t\omega:a(\omega)<t<1\}$ lies in $U_r$.

Let $\Gamma$ consist of the radial segments from $a(\omega)\omega$ to the first point $b(\omega)$ where $|f(b(\omega)\omega)|=2r$. Since $f(\gamma_\omega)\subset A(0;r,2r)$ for every $\gamma_\omega\in \Gamma$, by Cauchy--Schwarz and the quasiconformality of $f$, it follows that

\begin{equation}\label{eq:U_r-modulus}
    \int_{E_{2r}}\frac{d\mathcal H^1(\omega)}{\delta(\omega)} \le 2\operatorname{Mod}(\Gamma) \le  2K\operatorname{Mod}(f\Gamma)\le \frac{4\pi K}{\log 2}.
\end{equation}

Integrating over the radial segments and applying change of variables, gives
\begin{equation}\label{eq:U_r-lower-bound}
    |U_r|_{\alpha}\ge \frac12\int_{E_{2r}}\int_{a(\omega)}^1
(1-t)^\alpha\,dt\,d\mathcal H^1(\omega)\ge\frac{1}{2(\alpha+1)}
\int_{E_{2r}}\delta(\omega)^{\alpha+1}\,d\mathcal{H}^1(\omega).
\end{equation}

Finally, since $\alpha>-1$, by H\"older's inequality, we have
\begin{equation}\label{eq:U_r-Holder}
\mathcal{H}^1(E_{2r})^{\alpha+2}
\leq
\left(
\int_{E_{2r}}\delta^{\alpha+1}\,d\mathcal{H}^1
\right)
\left(
\int_{E_{2r}}\delta^{-1}\,d\mathcal{H}^1
\right)^{\alpha+1}.
\end{equation}
Using \eqref{eq:U_r-modulus} and \eqref{eq:U_r-lower-bound} in \eqref{eq:U_r-Holder} gives \eqref{eq:weighted-shifted-superlevel-lower-bound} with $C(K,\alpha)=\frac{1}{2(\alpha+1)}
\left(\frac{\log 2}{4\pi K}\right)^{\alpha+1}$.

\label{thm:(5)implies(3)}
Let $p>0$ and $\alpha>-1$, and let $f:\mathbb{D}\to \Omega$ be a $K$-quasiconformal homeomorphism. Then 
$$
\int_{0}^\infty r^{p-1}\mathcal H^1(F_r)^{\alpha+2}\,dr<\infty\quad\Longrightarrow\quad
\int_0^\infty r^{p-1}\mathcal H^1(E_r)^{\alpha+2}\,dr<\infty.
$$
<!-- end thm-3 -->
<!-- pf-2: body_md -->
Recall that we denote $0<R_0:=\max_{|z|=\frac{1}{2}}|f(z)|<\infty$. Since $f$ is continuous and open, $|f(z)|\le R_0$
for $|z|\le1/2$. Thus, for $r>R_0$,
$$
F_r\subset\{z\in\mathbb D:|z|>1/2\}.
$$
Moreover, for every $\omega\in E_r$, we have
$$
\left|f\left({\omega}/{2}\right)\right|\le R_0<r<|f(\omega)|.
$$
By the definition of radial limits and the intermediate
value theorem, the radial segment $\{s\omega:1/2<s<1\}$
meets $F_r$. Hence $E_r\subset\pi(F_r)$.

Since the radial projection $\pi(z)=z/|z|$ is $4$-Lipschitz on $\{|z|\ge1/2\}\subset \mathbb D\setminus\{0\}$, it follows that
$$
\mathcal H^1(E_r)\le\mathcal H^1(\pi(F_r))\le4\mathcal H^1(F_r),
\quad \text{for every}\,\,\, r>R_0.
$$
By the assumption (5), we know that
$$\int_{R_0}^\infty r^{p-1}\mathcal H^1(F_r)^{\alpha+2}\,dr
<\infty$$
Combining this with $\mathcal H^1(E_r)\le2\pi$, we conclude that
$$
\int_0^\infty r^{p-1}\mathcal H^1(E_r)^{\alpha+2}\,dr
\le
\frac{(2\pi)^{\alpha+2}}{p}R_0^p
+
4^{\alpha+2}
\int_{R_0}^\infty r^{p-1}\mathcal H^1(F_r)^{\alpha+2}\,dr
<\infty,
$$
which proves the theorem.

<!-- pf-3: body_md -->
The claim follows from Lemma \ref{cavalieri} together with Lemma \ref{Urestimate}. Indeed, by integrating \eqref{eq:weighted-shifted-superlevel-upper-bound}
and \eqref{eq:weighted-shifted-superlevel-lower-bound}
against $r^{p-1}\,dr$, a change of variables gives
$$\int_0^\infty r^{p-1}|U_r|_\alpha\,dr
\le
C_1\int_0^\infty r^{p-1}\mathcal H^1(E_r)^{\alpha+2}\,dr,$$
and 
$$\int_{2R_0}^\infty r^{p-1}\mathcal H^1(E_r)^{\alpha+2}\,dr
\le
C_2\int_{R_0}^\infty r^{p-1}|U_r|_\alpha\,dr,$$

$$

\quad\Longleftrightarrow\quad
\int_0^\infty r^{p-1}\mathcal H^1(E_r)^{\alpha+2}\,dr<\infty.
$$
The claim now follows from Lemma \ref{cavalieri}.

<!-- thm-6: body_md -->
\label{thm:f-iff-a_f}
Let $p>0$ and $\alpha>-1$, and let $f:\mathbb{D}\to \Omega$ be a $K$-quasiconformal homeomorphism. Then
$$  \int_\mathbb D
|f(z)|^{p}(1-|z|)^{\alpha}\,dA(z)<\infty \iff \int_\mathbb D a_f(z)^{p}(1-|z|)^{p+\alpha}\,dA(z)<\infty.$$
<!-- end thm-6 -->
<!-- pf-4: body_md -->
Suppose $\int_\mathbb D
|f(z)|^{p}(1-|z|)^{\alpha}\,dA(z)<\infty$. Choose $y_0\in\partial\Omega$ with $|y_0|=d_0:=\operatorname{dist}(0,\partial\Omega)$. By
		Lemma~\ref{lem:koebe}, we have
		$$
a_f(z)(1-|z|)
\le C(K)\operatorname{dist}(f(z),\partial\Omega)
\le C(K)(|f(z)|+d_0).
		$$
		Consequently, since $\alpha>-1$, 
	$$\int_{\mathbb D}a_f(z)^p(1-|z|)^{p+\alpha}\,dA(z)
\le C(K,p)\int_{\mathbb D}
\bigl(|f(z)|^p+d_0^p\bigr)(1-|z|)^\alpha\,dA(z)
<\infty.
$$

Conversely, by the equivalence of (1) and (3) established in Section \ref{thm:(3)iff(1)}, it suffices to prove that $\int_\mathbb D a_f(z)^{p}(1-|z|)^{p+\alpha}\,dA(z)<\infty$ implies (3). By Lemma~\ref{lem:BKL}, we have
\begin{equation}\label{eq:maximum-modulus-integral}
   \int_0^1 M(s)^p(1-s)^{\alpha+1}\,ds<\infty.
\end{equation}
Recall that $R_0:=M(1/2)$ and set $R:=\sup_{\mathbb D}|f|$, allowing
$R=\infty$. Since $f$ is continuous and open, $M$ is continuous and strictly increasing.
Thus, for each $2R_0<t<2R$, there is a unique
$\rho_t\in(1/2,1)$ such that
\begin{equation}\label{Mrhot}
    2M(\rho_t)=t.
\end{equation}

Let $\Gamma_t$ be the family of radial segments from $\rho_t\omega$ to $\omega$,
with $\omega\in E_t$. By a direct computation in polar coordinates,
$$\operatorname{Mod}(\Gamma_t)=\frac{\mathcal H^1(E_t)}{\log({1}/{\rho_t})}.$$
Since $|f(\rho_t\omega)|\le t/2$ and $|f(\omega)|>t$, continuity implies that each curve in $f(\Gamma_t)$ contains a subcurve joining the two boundary circles of $A(0;t/2,t)$.
Combining these with $\rho_t\in(1/2,1)$ and the quasiconformality of $f$, we deduce that
\begin{equation}\label{eq:m-r}
	\mathcal H^1(E_t)\le K \operatorname{Mod}(f\Gamma_t)\log\left(1/{\rho_t}\right)
	\le
	\frac{4\pi K}{\log2}(1-\rho_t).
\end{equation}

Note that 
$$(1-\rho_t)^{\alpha+2}
=(\alpha+2)\int_{\rho_t}^1(1-s)^{\alpha+1}\,ds.$$
By \eqref{eq:maximum-modulus-integral}, \eqref{Mrhot}, \eqref{eq:m-r} and Tonelli's theorem, we conclude that
\begin{align*}
	\int_{2R_0}^{2R}
	r^{p-1}\mathcal H^1(E_r)^{\alpha+2}\,dr
	&\le
	C(K,\alpha)\int_{2R_0}^{2R}r^{p-1}
	\int_{\rho_r}^1(1-s)^{\alpha+1}\,ds\,dr\\
	&=
	C(K,\alpha)\int_{1/2}^1(1-s)^{\alpha+1}
	\int_{2R_0}^{2M(s)}r^{p-1}\,dr\,ds\\
	&\le
	C(K,p,\alpha)\int_{1/2}^1
	M(s)^p(1-s)^{\alpha+1}\,ds
	<\infty.
\end{align*}
The integral over $(0,2R_0)$ is finite by \eqref{int02R0}. If $R<\infty$,
then $E_r=\emptyset$ for every $r\ge R$. Hence
$$
\int_0^\infty r^{p-1}\mathcal H^1(E_r)^{\alpha+2}\,dr<\infty.
$$
This is (3) and the proof is complete.

Suppose that $f\in A_\alpha^p(\mathbb D)$. Let $g=f^{-1}$,
	and let $L$ be the measurable representative of $r\mapsto\mathcal H^1(F_r)$ defined in Lemma \ref{lem:level-set-slicing}. 
	For $r>0$, define
	$$
	\mathcal A(r):=
	\int_{\Omega\cap S(0,r)}
	|Dg(w)|^2(1-|g(w)|)^\alpha\,d\mathcal H^1(w).
	$$
	By polar coordinates, the quasiconformality of $g$,
	and a change of variables, 
	\begin{align}
		\int_0^\infty r^p\mathcal A(r)\,dr
		&=
		\int_\Omega
		|w|^p|Dg(w)|^2(1-|g(w)|)^\alpha\,dA(w)
		\notag\\
		&\le
		K\int_\mathbb D
		|f(z)|^p(1-|z|)^\alpha\,dA(z)
		<\infty.
		\label{eq:weighted-inverse-energy}
	\end{align}
    In particular, $A(r)<\infty$ for almost every $r>0$. By Cauchy–Schwarz inequality and $\alpha\le0$, it follows that
	\begin{equation}\label{eq:weighted-level-CS}
		L(r)^2
		\le
		\mathcal A(r)
		\int_{\Omega\cap S(0,r)}
		(1-|g(w)|)^{-\alpha}\,d\mathcal H^1(w)\le2\pi r\mathcal A(r)
	\end{equation}
	for almost every $r>0$.

	Choose $y\in\partial\Omega$ and fix $R>|y|$.
	For $r>R$, the circle $S(0,r)$ cannot be contained
	in $\Omega$: otherwise, simply connectivity would imply
	$B(0,r)\subset\Omega$, contradicting
	$y\in\partial\Omega$.
	Thus each component $\sigma$ of $\Omega\cap S(0,r)$ is an open circular arc with endpoints in $\partial\Omega$.

    As $w$ approaches either endpoint along $\sigma$, we have $|g(w)|\to 1$; otherwise, compactness and the continuity of $f$ would imply that the limiting endpoint belongs to $\Omega$, a contradiction.
    Consequently, for almost every $r>R$, Lemma~\ref{lem:level-set-slicing} gives
    $$1-|z| \le\mathcal H^1(g(\sigma)) \le L(r) \quad \text{for every}\,\,\,z\in g(\sigma).$$
   Since this estimate holds for every component of $\Omega\cap S(0,r)$, we have
   $$1-|g(w)|\le L(r) \quad \text{for every}\,\,\,w\in \Omega\cap S(0,r).$$
   For $L(r)>0$, combining this with \eqref{eq:weighted-level-CS} and $\alpha\le 0$, it follows that
	$$
	L(r)^2
	\le
	2\pi r\mathcal A(r)L(r)^{-\alpha}.
	$$
	Hence
	\begin{equation*}
		L(r)^{\alpha+2}
		\le2\pi r\mathcal A(r)\quad \text{for almost every}\,\,\, r>R.
	\end{equation*}
 The same inequality is immediate
	when $L(r)=0$.

    For $0<r<R$, by \eqref{eq:weighted-level-CS} and the assumption $1<\alpha+2\le2$, we obtain
	$$
	L(r)^{\alpha+2} \le 1+L(r)^2
	\le1+2\pi r\mathcal A(r) \quad \text{for almost every}\,\,\,0<r<R.
	$$
	 Combining these two estimates with Lemma \ref{lem:level-set-slicing} and
	\eqref{eq:weighted-inverse-energy}, we conclude that
	\begin{align*}
		\int_0^\infty
		r^{p-1}\mathcal H^1(F_r)^{\alpha+2}\,dr
		&=
		\int_0^\infty r^{p-1}L(r)^{\alpha+2}\,dr\\
		&\le
		\frac{R^p}{p}
		+2\pi\int_0^\infty r^p\mathcal A(r)\,dr\\
		&\le
		\frac{R^p}{p}
		+2\pi K\int_\mathbb D
		|f(z)|^p(1-|z|)^\alpha\,dA(z)
		<\infty.
	\end{align*}
    This completes the proof.

\label{rem:K-blowup}
The maximal dilatations of the constructed maps satisfy
$$
K(\Phi)\ge K(f_\alpha)
\ge
\frac{\sqrt{\alpha+1}+1}{\sqrt{\alpha+1}-1}
\longrightarrow\infty
\quad\text{as }\alpha\downarrow0.
$$

Indeed, \eqref{eq:example-holder-parametrization} shows
that $h$ is bi-Lipschitz from
$(\partial\mathbb D,|\cdot|^\beta)$ onto
$(\Gamma_\rho,|\cdot|)$.
Since the subsequent similarity normalization preserves
Hausdorff dimension,
\begin{equation}\label{eq:snowflake-HD}
\dim_H\Phi(\partial\mathbb D)
=
\dim_H\Gamma_\rho
=
\frac1\beta
=
\frac{\alpha+2}{\alpha+1}.
\end{equation}

Recall that $\overline{\mathbb D}\Subset\Omega_\alpha$ and
$f_\alpha^{-1}=\Phi|_{\Omega_\alpha}$.
Extend the Beltrami coefficient of
$\Phi|_{\Omega_\alpha}$ by zero outside $\Omega_\alpha$.
The measurable Riemann mapping theorem gives a global
quasiconformal homeomorphism $\Psi:\mathbb C\to\mathbb C$ with the same Beltrami coefficient and
$$
K(\Psi)
=
K(\Phi|_{\Omega_\alpha})
=
K(f_\alpha).
$$
Since $\Phi$ and $\Psi$ have the same Beltrami coefficient
on $\Omega_\alpha$, the map
$H:=\Phi\circ\Psi^{-1}$ is conformal on
$\Psi(\Omega_\alpha)$.
In particular, it is locally bi-Lipschitz near the compact
curve $\Psi(\partial\mathbb D)$ and therefore preserves its
Hausdorff dimension.

Consequently, \eqref{eq:snowflake-HD} and Smirnov's
quasicircle dimension theorem
\cite[Theorem~1]{Smirnov} yield
$$
\frac{\alpha+2}{\alpha+1}
=
\dim_H\Phi(\partial\mathbb D)
=
\dim_H\Psi(\partial\mathbb D)
\le
1+
\left(
\frac{K(f_\alpha)-1}{K(f_\alpha)+1}
\right)^2.
$$
Rearranging gives the asserted lower bound for
$K(f_\alpha)$, and the bound for $K(\Phi)$ follows by
restriction and invariance of maximal dilatation under
inversion.

\label{rem:Hayman-Wu-and-Hausdorff-to-harmonic}
The failure of the Hayman--Wu bound in the quasiconformal setting was established by \cite[Theorem 1.1]{AstalaFernandezRohde1993}, where the counterexample yields an infinite sum of component diameters. Our snowflake construction exhibits a different way of the failure: the level sets are Jordan curves whose lengths are unbounded, although they are rectifiable for almost every radius.

Indeed, $\Phi(\partial \mathbb D)$ is a snowflake of Hausdorff dimension greater than $1$, hence $\mathcal H^1(\Phi(\partial \mathbb D))=\infty$. As $r\uparrow 1$, the curves $F_r=\Phi(r\partial\mathbb D)=\partial \Phi(r\mathbb D)$ converge to this snowflake, this gives $\mathcal H^1(F_r)\to\infty$, whereas $\mathcal H^1(F_r)<\infty$ for almost every $0<r<1$.
Thus, no finite uniform length bound exists even for this single mapping.

Moreover, since $\Phi(0)=0$ and $ \Phi(\overline{\mathbb D})\Subset\mathbb D$, the curve $F_r=\partial\Phi(r\mathbb D)$
surrounds $0$ and separates it from $\partial\mathbb D$ for every $0<r<1$. 
Therefore, we have $\omega_\mathbb D(0,F_r)=1$. The divergence of $\mathcal H^1(F_r)$ also shows the failure of 
\eqref{eq:Hausdorff-to-harmonic}, even with a 
constant depending on the mapping.
