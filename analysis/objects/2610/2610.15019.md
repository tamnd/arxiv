<!-- theorem-1: body_md -->
\label{T:H2}
Let $(b,a)$ be a Pythagorean pair, and suppose that $\phi:=b/a\in H^2$.
Then
\begin{equation}\label{E:equiv}
f\in{\mathcal H}(b) \iff \Bigl(f\in H^2 \text{~and~}\sum_{m\ge0}\Bigl|\sum_{n\ge0}\overline{\widehat{\phi}(n)}\widehat{f}(m+n)\Bigr|^2<\infty\Bigr),\end{equation}
in which case \eqref{E:formula} holds.
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{T:oldexample}
There exists a Pythagorean pair $(b,a)$  such that 
$\phi:=b/a\in\cap_{p<2}H^p$,
 yet, for some $f\in{\mathcal H}(b)$, the series $\sum_{n\ge0}\overline{\widehat{\phi}(n)}\widehat{f}(n)$ diverges.
<!-- end theorem-2 -->
<!-- theorem-3: body_md -->
\label{T:newexample}
There exist a Pythagorean pair $(b,a)$  such that 
$\phi:=b/a\in\cap_{p<1}H^p$,
and a function $f\in H^2$ such that
$\sum_{m\ge0}\Bigl|\sum_{n\ge0}\overline{\widehat{\phi}(n)}\widehat{f}(m+n)\Bigr|^2<\infty$,
yet $f\notin {\mathcal H}(b)$.
<!-- end theorem-3 -->
<!-- theorem-4: body_md -->
\label{T:positive}
Let $(b,a)$ be a Pythagorean pair, let $\phi:=b/a$, and suppose that
$\inf_{z\in{\mathbb D}}\operatorname{Re}\phi(z)>0$.
If $f\in{\mathcal H}(b)$, then 
\[
\sum_{m\ge0}\Bigl|\sum_{n\ge0}\overline{\widehat{\phi}(n)}\widehat{f}(m+n)\Bigr|^2<\infty,
\]
and the formula \eqref{E:formula} holds.
<!-- end theorem-4 -->
<!-- proof-1: body_md -->
Let $f\in {\mathcal H}(b)$. 
Then, as mentioned in the introduction, there is a unique $g\in H^2$
such that $T_{\overline{b}}f=T_{\overline{a}}g$,
and
\[
\|f\|_{{\mathcal H}(b)}^2=\|f\|_{H^2}^2+\|g\|_{H^2}^2.
\]
Since $1/\phi\in H^\infty$ and $a=b/\phi$, we have
\[
0=T_{\overline{b}}f-T_{\overline{a}}g=T_{\overline{b}}(f-T_{1/\overline{\phi}}(g)).
\]
Hence 
\[
f-T_{1/\overline{\phi}}(g)\in \ker T_{\overline{b}}=(bH^2)^\perp
\subset ((1/\phi)bH^2)^\perp=(aH^2)^\perp=\{0\},
\]
where the last equality holds because $a$ is outer.
Therefore $T_{1/\overline{\phi}}(g)=f$.


For $N\ge0$, set 
\[
q_N(z):=\sum_{k=0}^N\widehat{\phi}(k)z^k.
\]
Then, for each $m\ge0$, we have
\[
\sum_{n=0}^N\overline{\widehat{\phi}(n)}\widehat{f}(n+m)
=\bigl\langle f,\,z^mq_N\bigr\rangle_{H^2}
=\bigl\langle T_{1/\overline{\phi}}(g),\,z^mq_N\bigr\rangle_{H^2}
=\bigl\langle g,\,z^mq_N/\phi\bigr\rangle_{H^2}.
\]
Writing $q_N/\phi=1+r_N$, where $r_N\in z^{N+1}H^2$,
we obtain
\[
\bigl\langle g,\,z^mq_N/\phi\bigr\rangle_{H^2}
=\bigl\langle g,\,z^m+z^mr_N\bigr\rangle_{H^2}
=\widehat{g}(m)+\bigl\langle g,\,z^mr_N\bigr\rangle_{H^2}.
\]
To summarize, we have
\begin{equation}\label{E:summary}
\sum_{n=0}^N\overline{\widehat{\phi}(n)}\widehat{f}(n+m)
=\widehat{g}(m)+\bigl\langle g,\,z^mr_N\bigr\rangle_{H^2}
\quad(m\ge0,\,N\ge0).
\end{equation}
Now, since $r_N\in z^{N+1}H^2$, it is orthogonal to $1$, so
\[
1+\|r_N\|_{H^2}^2=\|1+r_N\|_{H^2}^2
=\|q_N/\phi\|_{H^2}^2
=\bigl\langle q_N/\phi,\,q_N/\phi\bigr\rangle_{H^2}
=\bigl\langle |q_N|^2,\,1/|\phi|^2\bigr\rangle_{L^2({\mathbb T})}.
\]
By hypothesis, there exists $c>0$ such that $\operatorname{Re}\phi\ge c$ on ${\mathbb D}$, and therefore we have
$\operatorname{Re}(1/\phi)\ge c/|\phi|^2$ a.e.\ on ${\mathbb T}$.
Hence
\begin{align*}
\bigl\langle |q_N|^2,\,1/|\phi|^2\bigr\rangle_{L^2({\mathbb T})}
&\le (1/c)\operatorname{Re}\bigl\langle |q_N|^2,\,1/\phi\bigr\rangle_{L^2({\mathbb T})}
=(1/c)\operatorname{Re}\bigl\langle q_N,\,q_N/\phi\bigr\rangle_{H^2}\\
&=(1/c)\operatorname{Re}\bigl\langle q_N,\,1+r_N\rangle_{H^2}
=(1/c)\operatorname{Re}\langle q_N,1\rangle=\operatorname{Re}\widehat{\phi}(0)/c.
\end{align*}
Therefore
\[
\|r_N\|_{H^2}\le (|\widehat{\phi}(0)|/c)^{1/2},
\]
and so, by Cauchy--Schwarz,
\[
\bigl|\langle g,\,z^mr_N\rangle_{H^2}\bigr|
\le \Bigl(\sum_{n\ge m+N+1}|\widehat{g}(n)|^2\Bigr)^{1/2}\Bigl(|\widehat{\phi}(0)|/c\Bigr)^{1/2}.
\]
The right-hand side tends to zero as $N\to\infty$.
Substituting this information into \eqref{E:summary},
we deduce that
\[
\sum_{n=0}^\infty\overline{\widehat{\phi}(n)}\widehat{f}(n+m)
=\widehat{g}(m) \quad(m\ge0).
\]
Hence, finally, 
\[
\sum_{m\ge0}\Bigl|\sum_{n=0}^\infty\overline{\widehat{\phi}(n)}\widehat{f}(n+m)
\Bigr|^2=\sum_{m\ge0}|\widehat{g}(m)|^2=\|g\|_{H^2}^2<\infty,
\]
and
\[
\|f\|_{{\mathcal H}(b)}^2
=\|f\|_{H^2}^2+\|g\|_{H^2}^2
=\sum_{m\ge0}|\widehat{f}(m)|^2
+\sum_{m\ge0}\Bigl|\sum_{n\ge0}\overline{\widehat{\phi}(n)}\widehat{f}(m+n)\Bigr|^2.\qedhere
\]
<!-- end proof-1 -->
<!-- theorem-5: body_md -->
\label{T:counterex}
Define $\phi:{\mathbb D}\to{\mathbb C}$  by
\begin{equation}\label{E:counterex}
\phi(z):=\sum_{j=1}^\infty \frac{2^{-j}}{1-e^{2\pi i/2^j}z}
\quad(z\in{\mathbb D}).
\end{equation}
\begin{enumerate}[\normalfont(i)]
\item
The function $\phi$ is  outer and belongs to $\cap_{p<1}H^p$.
Consequently $\phi=b/a$, where $(b,a)$ is a Pythagorean pair
with $b$ outer.
\item
There exists $f\in H^2$ such that
\[
\sum_{n\ge0}\overline{\widehat{\phi}(n)}\widehat{f}(n+m)=0 \quad(\forall m\ge0),
\]
yet $f\notin{\mathcal H}(b)$.
\end{enumerate}
<!-- end theorem-5 -->
<!-- lemma-1: body_md -->
\label{L:outer}
Let $(b,a)$ be a Pythagorean pair such that $b$ is an outer function.
Then $\|f\|_{{\mathcal H}(b)}>\|f\|_{H^2}$ for all $f\in{\mathcal H}(b)\setminus\{0\}$.
<!-- end lemma-1 -->
<!-- proof-2: body_md -->
Let $f\in{\mathcal H}(b)$. Then there exists a unique $g\in H^2$
such that $T_{\overline{b}}f=T_{\overline{a}}g$,
and $\|f\|_{{\mathcal H}(b)}^2=\|f\|_{H^2}^2+\|g\|_{H^2}^2$.
It follows that $\|f\|_{{\mathcal H}(b)}>\|f\|_{H^2}$ unless $g=0$.
But $g=0$ implies that $T_{\overline{b}}f=T_{\overline{a}}0=0$,
so that $f\in\ker(T_{\overline{b}})=(bH^2)^\perp=\{0\}$,
where the last equality holds because $b$ is outer. 
Thus $\|f\|_{{\mathcal H}(b)}>\|f\|_{H^2}$
unless $f=0$.
<!-- end proof-2 -->
<!-- proof-3: body_md -->
(i) The series defining $\phi$ converges locally uniformly on ${\mathbb D}$,
so $\phi$ is holomorphic in ${\mathbb D}$.
Also, we have
\[
 \operatorname{Re}\frac{1}{1-e^{2\pi i/2^j} z}
 =\frac{1}{2}+\frac{1-|z|^2}{2|1-e^{2\pi i/2^j} z|^2}\ge\frac{1}{2} \quad(j\ge1).
\]
Since $\sum_{j\ge1}2^{-j}=1$, this implies that
\begin{equation}\label{E:realpart}
\operatorname{Re}\phi(z)\ge1/2 \quad(z\in{\mathbb D}).
\end{equation}

For $0<p<1$, the principal power $\phi^p$ is holomorphic and satisfies
\[
 \cos(p\pi/2)|\phi(z)|^p\le\operatorname{Re}\bigl(\phi(z)^p\bigr).
\]
The mean-value property consequently gives
\[
 \sup_{0<r<1}\frac1{2\pi}\int_0^{2\pi}
 |\phi(re^{it})|^p\,dt\le\frac{\operatorname{Re}(\phi(0)^p)}{\cos(p\pi/2)}
=\frac{1}{\cos(p\pi/2)}.
\]
Thus $\phi\in H^p$ for every $0<p<1$, and in particular $\phi\in N^+$.
Furthermore, \eqref{E:realpart} implies that $1/\phi\in H^\infty$.
The memberships $\phi\in H^p$ and $1/\phi\in H^\infty$ together
imply that $\phi$ is outer. 

Since $\phi\in N^+$,
we can write it as  $\phi=b/a$, where $(b,a)$ is a Pythagorean pair.
Then $b=a\phi$, the product of two outer
functions, so $b$ is also outer.

\bigskip

(ii) Let us write $\gamma_k:=\widehat{\phi}(k)$. 
We shall construct a sequence $(\alpha_n)\in\ell^2({\mathbb Z}^+)$
such that $\alpha_0=1$ and
\begin{equation}\label{E:alpha}
\sum_{n=m}^\infty \gamma_{n-m}\alpha_n=0 \quad(m\ge0).
\end{equation}
Assuming that such a sequence exists,
set $f(z):=\sum_{n\ge0}\overline{\alpha}_nz^n$.
Then $f\in H^2$, and \eqref{E:alpha} gives
\[
\sum_{n\ge0}\overline{\widehat{\phi}(n)}\widehat{f}(n+m)=0 \quad(\forall m\ge0).
\]
Further, we have $f\notin {\mathcal H}(b)$. Indeed, suppose, on the contrary,
that $f\in{\mathcal H}(b)$. Since $\operatorname{Re}\phi\ge 1/2$ on ${\mathbb D}$, Theorem~\ref{T:positive} applies, and from \eqref{E:formula}
we get $\|f\|_{{\mathcal H}(b)}=\|f\|_{H^2}$. As $b$ is outer, 
Lemma~\ref{L:outer} tells us that $f=0$. This contradicts the fact that $\widehat{f}(0)=\overline{\alpha}_0=1$. Thus  (ii) is proved,
subject to the construction of the sequence $(\alpha_n)$.

Before carrying out this construction,
we establish an auxiliary bound.
Let us write
$\lambda_j:=\exp(2\pi i/2^j)$.
We claim that 
\begin{equation}\label{E:claim}
\Bigl|\sum_{n=N}^{N+t-1}\gamma_{n-m}\lambda_j^{-n}-t2^{-j}\lambda_j^{-m}\Bigr|\le 2^j 
\qquad(j\ge1,~t\ge0,~N\ge m\ge0).
\end{equation}
To see this, note that, expanding the summands in
\eqref{E:counterex}, we get
\[
\gamma_k=\sum_{j=1}^\infty 2^{-j}\lambda_j^k
\quad(k\ge0).
\]
It follows that, for all integers $N\ge m\ge0$ and $t\ge0$,
\[
\sum_{n=N}^{N+t-1}\gamma_{n-m}\lambda_j^{-n}
-t2^{-j}\lambda_j^{-m}
=\sum_{\substack{\ell=1\\ \ell\ne j}}^\infty2^{-\ell}\lambda_\ell^{-m}\sum_{n=N}^{N+t-1}(\lambda_\ell/\lambda_j)^n.
\]
Now, for distinct integers $j,\ell\ge1$, the angular separation
of $\lambda_j$ and $\lambda_\ell$ lies between $\pi/2^j$
and $\pi$, so
\[
|1-\lambda_\ell/\lambda_j|\ge 2\sin(\pi/2^{j+1})\ge 2^{1-j},
\]
and hence 
\[
\Bigl|\sum_{n=N}^{N+t-1}(\lambda_\ell/\lambda_j)^n\Bigr|
\le \frac{2}{|1-\lambda_\ell/\lambda_j|}\le 2^j.
\]
It follows that
\[
\Bigl|\sum_{n=N}^{N+t-1}\gamma_{n-m}\lambda_j^{-n}-t2^{-j}\lambda_j^{-m}\Bigr|
=\Bigl| \sum_{\substack{\ell=1\\ \ell\ne j}}^\infty2^{-\ell}\lambda_\ell^{-m}\sum_{n=N}^{N+t-1}(\lambda_\ell/\lambda_j)^n\Bigr|
\le \sum_{\ell=1}^\infty 2^{-\ell}2^j=2^j.
\]
This justifies the claim \eqref{E:claim}.

We now construct the sequence $(\alpha_n)_{n\ge0}$ at the same time as a strictly increasing sequence of positive integers $(N_M)_{M\ge0}$.
The construction is inductive, starting with $\alpha_0=1$
and $N_0=1$. Let $M\ge0$, and suppose that
$N_0,\dots,N_M$ and
$\alpha_0,\dots,\alpha_{N_M-1}$  have been chosen.
For $0\le m\le M$, set
\[
R_m:=\sum_{n=m}^{N_M-1}\gamma_{n-m}\alpha_n.
\]
Choose complex numbers $d_1,\dots,d_{M+1}$ to solve the system
\begin{equation}\label{E:system}
\sum_{j=1}^{M+1}2^{-j}\lambda_j^{-m}d_j=-R_m
\quad(0\le m\le M).
\end{equation}
The matrix of this system is a Vandermonde matrix in the distinct 
numbers $\lambda_j^{-1}$, with non-zero column factors $2^{-j}$,
so the solution exists and is unique. Define
\[
C_M:=\sum_{j=1}^{M+1}2^j|d_j|,
\quad
L_M:=2+\lceil 2^{2M+4}C_M^2\rceil
\quad
\text{and}\quad
N_{M+1}:=N_M+L_M,
\]
and set
\begin{equation}\label{E:alphadef}
\alpha_n:=\frac{1}{L_M}\sum_{j=1}^{M+1}d_j\lambda_j^{-n}
\quad(N_M\le n\le N_{M+1}-1).
\end{equation}
Clearly $N_{M+1}>N_M$.
Also, 
using the definitions
of $R_m$ and then $\alpha_n$, we have,
for $0\le m\le M$ and $0\le t\le L_M$, 
\begin{align*}
&\sum_{n=m}^{N_M+t-1}\gamma_{n-m}\alpha_n\\
&=R_m+\sum_{n=N_M}^{N_M+t-1}\gamma_{n-m}\alpha_n\\
&=R_m+\sum_{n=N_M}^{N_M+t-1}\gamma_{n-m}\frac{1}{L_M}\sum_{j=1}^{M+1}d_j\lambda_j^{-n}\\
&=R_m+\frac{1}{L_M}\sum_{j=1}^{M+1}d_j\sum_{n=N_M}^{N_M+t-1}\gamma_{n-m}\lambda_j^{-n}\\
&=R_m+\frac{1}{L_M}\sum_{j=1}^{M+1}d_jt2^{-j}\lambda_j^{-m}
+\frac{1}{L_M}\sum_{j=1}^{M+1}d_j\Bigl(\sum_{n=N_M}^{N_M+t-1}\gamma_{n-m}\lambda_j^{-n}-t2^{-j}\lambda_j^{-m}\Bigr)\\
&=R_m-\frac{t}{L_M}R_m+\frac{1}{L_M}\sum_{j=1}^{M+1}d_j\Bigl(\sum_{n=N_M}^{N_M+t-1}\gamma_{n-m}\lambda_j^{-n}-t2^{-j}\lambda_j^{-m}\Bigr),
\end{align*}
where, in the last line, we used \eqref{E:system}.
Also, using \eqref{E:claim}, we have
\begin{align*}
&\Bigl|\frac{1}{L_M}\sum_{j=1}^{M+1}d_j\Bigl(\sum_{n=N_M}^{N_M+t-1}\gamma_{n-m}\lambda_j^{-n}-t2^{-j}\lambda_j^{-m}\Bigr)\Bigr|\\
&\le \frac{1}{L_M}\sum_{j=1}^{M+1}|d_j|2^j=\frac{C_M}{L_M}\le \frac{C_M}{\sqrt{L_M}}\le 2^{-M-2}.
\end{align*}
Putting everything together, we deduce that,
for $0\le m\le M$ and $0\le t\le L_M$,
\begin{equation}\label{E:overall}
\Bigl|\sum_{n=m}^{N_M+t-1}\gamma_{n-m}\alpha_n\Bigr|\le (1-t/L_M)|R_m|+2^{-M-2}.
\end{equation}
In particular, taking $t=L_M$, and recalling that $N_M+L_M=N_{M+1}$, we get
\begin{equation}\label{E:overall2}
\Bigl|\sum_{n=m}^{N_{M+1}-1}\gamma_{n-m}\alpha_n\Bigr|\le 2^{-M-2}
\quad(0\le m\le M).
\end{equation}

Now fix $m\ge0$.
For every $M>m$, the inequality \eqref{E:overall2} from stage $M-1$
shows that $|R_m|\le 2^{-M-1}$. Substituting this into
\eqref{E:overall}, we deduce that
\[
\Bigl|\sum_{n=m}^{N_M+t-1}\gamma_{n-m}\alpha_n\Bigr|\le (1-t/L_M)|R_m|+2^{-M-2} <2^{-M}
\quad(0\le t\le L_M),
\]
whence
\[
\Bigl|\sum_{n=m}^{N-1}\gamma_{n-m}\alpha_n\Bigr|\le 2^{-M}
\quad(N_M\le N\le N_{M+1}).
\]
This implies that
\[
\sum_{n=m}^\infty \gamma_{n-m}\alpha_n=0.
\]
As this holds for each $m\ge0$, we have established \eqref{E:alpha}.

All that remains is to check that the sequence $(\alpha_n)$ belongs to $\ell^2({\mathbb Z}^+)$. For this, we go back to the definition \eqref{E:alphadef} of $\alpha_n$. For $N_M\le n<N_{M+1}$, we have
\[
|\alpha_n|
=\Bigl|\frac{1}{L_M}\sum_{j=1}^{M+1}d_j\lambda_j^{-n}\Bigr|
\le \frac{1}{L_M}\sum_{j=1}^{M+1}|d_j|\le \frac{C_M}{L_M}.
\]
Hence
\begin{align*}
\sum_{n=1}^\infty|\alpha_n|^2
&=\sum_{M=0}^\infty\sum_{n=N_M}^{N_{M+1}-1}|\alpha_n|^2
\le \sum_{M=0}^\infty(N_{M+1}-N_M)C_M^2/L_M^2\\
&=\sum_{M=0}^\infty C_M^2/L_M
\le\sum_{M=0}^\infty 2^{-2M}<\infty.
\end{align*}
The proof is complete.

Let $(b,a)$ be a Pythagorean pair such that $\phi:=b/a\in H^p$,
where $0<p\le 2$. Let $f\in H^2$ be a function 
whose Taylor coefficients satisfy
\begin{equation}\label{E:pcond}
\sum_{m\ge1}m^{2/p-1}|\widehat{f}(m)|^2<\infty.
\end{equation}
If 
\[
\sum_{m\ge0}\Bigl|\sum_{n\ge0}\overline{\widehat{\phi}(n)}\widehat{f}(m+n)\Bigr|^2<\infty,
\]
then $f\in{\mathcal H}(b)$ and \eqref{E:formula} holds.
