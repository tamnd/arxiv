--- From an experimental perspective, we would like
to probe the chaotic nature of a many-body quantum system, clean or
disordered, at a given time $T$. Specifically, we would like to know
whether the spectrum of the system Hamiltonian is integrable, satisfying
Poisson statistics or chaotic satisfying Wigner-Dyson level statistics,
predicted by the Gaussian ensembles of random matrix theory (RMT)~\citep{haake2001quantum,mehta2004randommatrices}.
However, in experimental platforms such as cold atoms and ion traps,
one cannot directly access to the energy spectrum~\citep{Schreiber2015,Hess2017}.
Here, inspired by the spirit of quantum metrology, we consider to
apply a sensing Hamiltonian $\Pi_{\bm{\theta}}=\sum_{\alpha}\theta_{\alpha}P_{\alpha}$
for a time $T$, the coupling parameter $\theta_{\alpha}$ is known
and can be carefully tuned, and the coupling parameters $P_{\alpha}$
can be engineered. Upon time evolution, we expect quantum chaos signatures
to be encoded in the parameter-dependent states, which then can be
further probed through quantum measurements. \textcolor{black}{Two
protocols naturally arise: the continuous protocol, where system and
sensing Hamiltonians are applied at the same time, and }the quench
protocol, where the system Hamiltonian is applied for a time $T$
followed by the sensing Hamiltonian applied for the same time $T$.
From the perspective of quantum sensing, the system Hamiltonian $H$
can be viewed as a many-body control to create necessary entanglement
for sensing.

--- We consider three models, random matrices from
GUE and two spin$-1/2$ systems with open boundary conditions, the
disordered full-random-field XXX (FRF-XXX) $H_{\mathrm{FRF-XXX}}=J\sum^{L-1}_{i=1}\bm{S}_{i}\cdot\bm{S}_{i+1}+\sum_{i=1}\bm{h}_{i}\cdot\bm{S}_{i}$,
where $\bm{S}_{i}=(\sigma^{x}_{i},\,\sigma^{y}_{i},\,\sigma^{z}_{i})/2$
and $\bm{h}_{i}=(h^{x}_{i},\,h^{y}_{i},\,h^{z}_{i})$ with $h^{\alpha}_{i}\in[-h,h]$,
and the clean mixed-field Ising (MFI) $H_{\mathrm{MFI}}=J\sum^{L-1}_{i=1}S^{x}_{i}S^{x}_{i+1}+g^{x}\sum^{L-1}_{i=1}S^{x}_{i}+g^{z}\sum^{L-1}_{i=1}S^{z}_{i}+d_{L/2}S^{z}_{L/2}$\footnote{The term $d_{L/2}S^{z}_{L/2}$ is added to break the total $S^{z}=\sum^{L}_{i=1}S^{z}_{i}$
conservation for the convenience of the numerical calculation of the
level statistics}. Numerical results are shown in Fig.~\ref{fig:Numerics}, which
validates our findings between the relation of the average QFI and
the SFF in the continuous protocol. Furthermore, as shown in the insets
of Fig.~\ref{fig:Numerics}(b) and~\ref{fig:Numerics}(c), numerical
results suggest that the fluctuations of the time-integrated SFF are
not only negligible at short and long times, but also at intermediate
times, suggesting that the QFI is self-averaging for RMT and for realistic
chaotic Hamiltonians at all timescales. Note that although the correlation
hole, which is the characteristics of quantum chaos, disappears in
QFI due to time integration, one can always recover it upon reconstructing
the SFF.

\textup{It can be shown that 
\begin{equation}
\mathscr{T}^{(k)}_{\mathrm{ISO}}(X_{1},\,t_{1};X_{2},t_{2};\cdots X_{k},t_{k})=\mathbb{E}_{\Lambda\sim P(\Lambda)}\left\{ \mathrm{Tr}_{(1,2,\cdots k)}\left[\otimes^{k}_{\alpha=1}X_{\alpha}\mathscr{T}^{(2k)}_{\mathrm{Haar}}\left((\otimes^{k}_{\beta=1}\mathrm{e}^{-\mathrm{i}\Lambda t_{k}})\otimes\otimes^{k}_{\beta=1}\mathrm{e}^{\mathrm{i}\Lambda t_{k}})\right)\prod^{k}_{\delta=1}\mathbb{F}_{\delta,\,k+\delta}\right]\right\} ,
\end{equation}
where }$V=U\Lambda U^{\dagger}$\textup{ and
\begin{equation}
\mathscr{T}^{(2k)}_{\mathrm{Haar}}(X)\equiv\mathbb{E}_{U\sim\mathrm{Haar}}[U^{\otimes2k}XU^{\dagger\otimes2k}]
\end{equation}
is the twirling operation $\mathbb{F}_{j,\,k}$ is the swap operator
between the $j$-th and $k$-th Hilbert spaces and $\mathrm{Tr}_{(1,2,\cdots,\,k)}$
is the partial trace over the first half of the Hilbert space. }
<!-- end lem-1 -->
<!-- proof-1: body_md -->
The proof is straightforward: Using the partial swap trick $AB=\mathrm{Tr}_{2}(A\otimes B\mathbb{F})$,
we find 
\begin{align}
\otimes^{k}_{\alpha=1}\left(\mathrm{e}^{\mathrm{i}Ht_{\alpha}}X_{\alpha}\mathrm{e}^{-\mathrm{i}Ht_{\alpha}}\right) & =\otimes^{k}_{\alpha=1}\left(U\mathrm{e}^{\mathrm{i}\Lambda t_{\alpha}}U^{\dagger}X_{\alpha}U\mathrm{e}^{-\mathrm{i}\Lambda t_{\alpha}}U^{\dagger}\right)\nonumber \\
 & =\otimes^{k}_{\alpha=1}\left(\mathrm{Tr}_{2\alpha}(\mathbb{I}\otimes X_{\alpha})U^{\otimes2}(\mathrm{e}^{\mathrm{i}\Lambda t_{\alpha}}\otimes\mathrm{e}^{-\mathrm{i}\Lambda t_{\alpha}})U^{\dagger\otimes2}\mathbb{F}_{2\alpha-1,2\alpha}\right)\nonumber \\
 & =\mathrm{Tr}_{2,4\cdots,2k}\left[(\mathbb{I}\otimes X_{1}\otimes\cdots\otimes\mathbb{I}\otimes X_{k})U^{\otimes2k}(\mathrm{e}^{\mathrm{i}\Lambda t_{1}}\otimes\mathrm{e}^{-\mathrm{i}\Lambda t_{1}}\cdots\mathrm{e}^{\mathrm{i}\Lambda t_{k}}\otimes\mathrm{e}^{-\mathrm{i}\Lambda t_{k}})U^{\dagger\otimes2k}(\mathbb{F}_{12}\otimes\mathbb{F}_{34}\cdots\mathbb{F}_{2k-1,2k})\right].
\end{align}
Upon permuting the order of Hilbert spaces, it can be readily found
\begin{equation}
\otimes^{k}_{\alpha=1}\left(\mathrm{e}^{\mathrm{i}Ht_{\alpha}}X_{\alpha}\mathrm{e}^{-\mathrm{i}Ht_{\alpha}}\right)=\mathrm{Tr}_{(1,2,\cdots k)}\left[\left((\otimes^{k}_{\alpha=1}X_{\alpha})U^{\otimes2k}(\otimes^{k}_{\beta=1}\mathrm{e}^{-\mathrm{i}\Lambda t_{\alpha}}\otimes^{k}_{\gamma=1}\mathrm{e}^{\mathrm{i}\Lambda t_{\alpha}})U^{\dagger\otimes2k}\prod^{k}_{\delta=1}\mathbb{F}_{\delta,\,k+\delta}\right)\right],
\end{equation}
which concludes the proof.

\textup{\label{lem:Bochner-Bakry-Emergy}Consider a function $\Phi:M\to\mathbb{R}$,
the probability measure 
\begin{equation}
d\mu_{\Phi}\equiv\frac{1}{Z}\mathrm{e}^{-\Phi(x)}d\mathrm{vol}(x),\label{eq:dmu-Phi}
\end{equation}
where $d\mathrm{vol}(x)=\sqrt{\det g(x)}dx_{1}\wedge dx_{2}\wedge\cdots\wedge dx_{m}$,
satisfies LSI with constant $c_{\mathrm{LSI}}$ if the following inequality
holds
\begin{equation}
[\mathrm{Ric}+\mathrm{Hess}(\Phi)](\mathcal{X},\mathcal{X})\ge\frac{1}{c_{\mathrm{LSI}}}g_{x}(\mathcal{X},\mathcal{X})\label{eq:BBE}
\end{equation}
for any tangent vector $\mathcal{X}\in T_{x}M$at any point $x\in M$,
where $\mathrm{Ric}(\Phi)$ is the Ricci tensor for for the Riemannian
manifold $M$ and $\mathrm{Hess}(\Phi)$ is the Hessian of the function
$\Phi$.}

\textup{\label{lem:LSI-Product}Given $n$-probability measure $\{\mu^{(i)}\}^{n}_{i=1}$
on $n$-Riemannian manifold $\{M_{i}\}^{n}_{i=1}$ with each of them
satisfying the LSI with constant $c^{(i)}_{\mathrm{LSI}}$, the product
measure $\mu^{(M)}=\mu^{(1)}\otimes\mu^{(2)}\otimes\cdots\otimes\mu^{(n)}$
on the manifold $M=M_{1}\times M_{2}\times\cdots\times M_{n}$ also
satisfy the LSI with the constant $c^{(M)}_{\mathrm{LSI}}=\max\{c^{(i)}_{\mathrm{LSI}}\}$.}
