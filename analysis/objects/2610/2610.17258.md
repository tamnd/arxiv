\begin{enumerate}
			\item We first prove the formula for $H_k$. The Selberg identity
			$K(-a,b;c)=\sum_{d\mid(a,b,c)}dK(-ab/d^2,1;c/d)$
			implies the following relation between the coefficients of the Poincaré series:
			$$
			\frac{\pi^{(k)}(-a,b)}{b^{k-1}}
			=-\sum_{d\mid(a,b)}\frac{\gamma^{(k)}(-1,ab/d^2)}{d^{k-1}}.
			$$
			Substituting this identity into the double Fourier series for $H_k$ and writing $a=dr$, $b=ds$, we obtain the sum
			$\sum_{d\ge1}(q_z^rq_w^s)^d/d^{k-1}
			=\operatorname{Li}_{k-1}(q_z^rq_w^s)$.
			Similarly,
			$\sum_{n\ge1}\sigma_{k-1}(n)q_z^n/n^{k-1}
			=\sum_{r\ge1}\operatorname{Li}_{k-1}(q_z^r)$.
			This proves the first assertion.
			
			\item We now apply the Maass raising operators in both variables to \eqref{zag:H}. For $\xi=q_z^rq_w^s$, we have
			$D_z\operatorname{Li}_n(\xi)=r\operatorname{Li}_{n-1}(\xi)$ and
			$D_w\operatorname{Li}_n(\xi)=s\operatorname{Li}_{n-1}(\xi)$.
			The resulting expression is a linear combination of the real parts of the polylogarithms $\operatorname{Li}_j(\xi)$, with coefficients depending on $y_z$ and $y_w$. Since
			$\log|\xi|=-2\pi(ry_z+sy_w)$, the definition of $D_\ell$ allows us to write this combination as
			$$
			\begin{aligned}
				&2(-1)^{k/2+1}
				\Re\left(
				\mathcal R^{2-k}_{[k/2-1],z}
				\mathcal R^{2-k}_{[k/2-1],w}
				\operatorname{Li}_{k-1}(\xi)
				\right)\\
				&=\frac{1}{(4\pi^2y_zy_w)^{k/2-1}}
				\Bigg[
				\frac{(-1)^{k/2}}{2^{k-3}}
				\sum_{\substack{a,b\ge0\\a+b\le k/2-1}}
				\frac{(-1)^{a+b}(k-2-2a)!(k-2-2b)!}
				{a!b!(k/2-1-a)!(k/2-1-b)!}\\
				&\hspace{30mm}\times(2\pi r y_z)^{2a}(2\pi s y_w)^{2b}
				D_{k-1-2a-2b}(\xi)
				-\frac{(2\pi r y_z)^{k-1}+(2\pi s y_w)^{k-1}}{k-1}
				\Bigg].
			\end{aligned}
			$$
			Substituting this expression into \eqref{zag:H} and applying Theorem~\ref{thm:maass-green}, we obtain \eqref{zag:G} for $y_z>y_w$ and $y_zy_w>1$.
			
			It remains to extend \eqref{zag:G} to the full region $y_zy_w>1$. For $r,s\ge1$, the coefficients satisfy the standard estimate $|\gamma^{(k)}(-1,rs)|\le Ce^{4\pi\sqrt{rs}}$, while on every compact subset of this region there exists $\delta>0$ such that
			$ry_z+sy_w-2\sqrt{rs}\ge\delta(r+s)$.
			The polynomial terms in $D_\ell(\xi)$ combine to
			$\bigl((2\pi r y_z)^{k-1}+(2\pi s y_w)^{k-1}\bigr)/(k-1)$
			and therefore cancel with the last term in the square brackets in \eqref{zag:G}. The remaining part decays exponentially, and hence the double series converges uniformly on compact subsets. Finally, since $D_\ell$ is single-valued, the term corresponding to $(r,s)=(1,-1)$ extends to the region $z-w\notin\mathbb Z$. This completes the proof of the second assertion.
		\end{enumerate}

\label{thm:Pi-closed-form-general}
		Let $w\in\Delta_\sigma$ and $w\notin\Gamma\cdot{i,\rho}$. Then
		$$
		\refstepcounter{equation}\label{eq:contour-identity-general}
		\begin{gathered}
			\sum_{j=0}^{k-2}\binom{k-2}{j}\Pi_j^{(k)}(w)
			\left((-1)^jz^{k-2-j}+(z+1)^{k-2-j}z^j-(-1)^j(z+1)^{k-2-j}\right)\\
			=(z-w)^{k-2}+\bigl((w+1)z+1\bigr)^{k-2}+\bigl(wz+w+1\bigr)^{k-2}.
		\end{gathered}\eqno(\theequation)
		$$
<!-- end proposition-3 -->
<!-- proof-3: body_md -->
We evaluate the integral around the boundary of $\Delta_\sigma$ in two ways.
		The identities $j'(t)/(2\pi i)=-E_{14}(t)/\Delta(t)$ and
		$E_{14}=E_kE_{14-k}$ imply that, for
		$\gamma=\begin{psmallmatrix}a&b\\c&d\end{psmallmatrix}\in\Gamma$,
		$\operatorname*{Res}_{t=\gamma w}F_k(t,w)=\frac{(cw+d)^{k-2}}{2\pi i}.$
		The three points of the orbit $\Gamma w$ lying inside $\Delta_\sigma$ are
		$w,\sigma w,\sigma^2w$.  
		Since the boundary is oriented clockwise, the residue theorem gives
		$$
		\begin{aligned}
			\oint_{\partial\Delta_\sigma}(z-t)^{k-2}F_k(t,w)dt
			={}&-(z-w)^{k-2}\\
			&-\bigl((w+1)z+1\bigr)^{k-2}
			-\bigl(wz+w+1\bigr)^{k-2}.
		\end{aligned}
		$$
		
		On the second and third sides, make the substitutions
		$t=-1/(s+1)$ and $t=s-1$, respectively. Modularity and periodicity of $F_k$ give
		$$
		\begin{aligned}
			\oint_{\partial\Delta_\sigma}(z-t)^{k-2}F_k(t,w)dt
			={}&\int_0^{i\infty}\Bigl(-(z-s)^{k-2}\\
			&-(zs+z+1)^{k-2}+(z+1-s)^{k-2}\Bigr)F_k(s,w)ds.
		\end{aligned}
		$$
		Expanding the powers and comparing the two expressions for the contour integral,
		we obtain \eqref{eq:contour-identity-general}.

			\Pi_0^{(k)}(w)
			=C_k^{-1}\left(
			H_k(i,w)-\frac{2\varepsilon_ki}{k-2}\partial_zH_k(i,w)
			\right)-\beta_k(w),
			\eqno(\theequation)
			$$
			where
			$$
			\beta_k(w)=
			\begin{cases}
				-\dfrac12\displaystyle\sum_{j=1}^{k-3}
				(-1)^j\binom{k-2}{j}i^{k-2-j}\Pi_j^{(k)}(w),
				&k=4,8,\\[6pt]
				\dfrac1{k-2}\displaystyle\sum_{j=1}^{k-3}
				(-1)^j\binom{k-2}{j}(k-2-j)i^{k-2-j}\Pi_j^{(k)}(w),
				&k=6,10,14.
			\end{cases}
			$$
			
			Since $g_{-n}^{(k)}$ has weight $2-k$, its transformation under $S$ at the fixed point $i$ gives
			$$
			g_{-n}^{(k)}(i)-\frac{2\varepsilon_ki}{k-2}(g_{-n}^{(k)})'(i)=0.
			$$
			Consequently, the Fourier expansion of $H_k(w,z)$ yields
			$$
			\left.
			\left(
			1-\frac{2\varepsilon_ki}{k-2}\frac{\partial}{\partial z}
			\right)H_k(w,z)
			\right|_{z=i}=0.
			$$
			
			Applying the same differential operator to \eqref{eq:H-reciprocity} and using Lemma~\ref{lemma:Eichler-i-combination}, we find
			$$
			\begin{aligned}
				\Pi_0^{(k)}(w)
				={}&\frac{B_{k-1}(i-w)}{k-1}
				-\frac{2\varepsilon_ki}{k-2}B_{k-2}(i-w)-\beta_k(w)\\
				&+\frac{2k}{B_kC_k}
				\left(
				\frac{\zeta(k-1)}2+
				\sum_{n\ge1}\frac{\sigma_{k-1}(n)}{n^{k-1}}q_w^n
				-r_k\pi^{k-1}
				\right).
			\end{aligned}
			$$
			The explicit formulas for $\Pi_j^{(k)}(w)$ and $r_k$ give
			$$
			\frac{B_{k-1}(i-w)}{k-1}
			-\frac{2\varepsilon_ki}{k-2}B_{k-2}(i-w)
			-\beta_k(w)
			-\frac{2k(k-2)!r_k}{B_k(2i)^{k-1}}
			=
			w+\frac12-\frac{B_{k-1}(w+1)}{k-1}.
			$$
			Hence \eqref{eq:Pi0-closed-form} holds for $\Im w>1$. Since both sides are holomorphic on $\Delta_\sigma$, the identity extends throughout $\Delta_\sigma$. Applying $D_w^{k-1}$ gives
			$D_w^{k-1}\Pi_0^{(k)}(w)=-C_k^{-1}E_k(w)$.
