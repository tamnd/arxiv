\label{thm:easy regime informal}
                                                             
    Fix $\beta\geq 0$ such that $0\leq \Delta (J) \cdot \frac{\tanh(\beta \eta)}{\eta} \leq 1.$ 
    
    Fix $ \epsilon,\delta \in (0,1).$ There exists
    an efficient randomized classical algorithm that runs in time \[\text{poly}(n, \beta, \max_{i,j} |J_{ij}| , \max|h_i^z|,  |\eta|, \epsilon^{-1}) \cdot \log \delta^{-1}\]                                                                                                                                    
    and:
    \begin{enumerate}
\item Computes $ \widehat{Z}$ s.t. 
$\Pr[(1-\epsilon) Z(\beta) \leq \widehat{Z} \leq (1+\epsilon) Z(\beta)]\geq 1-\delta.$
                                                                                                      
                                                                              
\item Given a Pauli string $O=\prod_{i=1}^n P_i,$ computes $ \widehat{\operatorname{Tr}(O\rho_\beta)}$ s.t.
$\Pr\left[\left|\widehat{\operatorname{Tr}(O \rho_\beta)} -\operatorname{Tr}(O \rho_\beta)\right|\leq \epsilon \right] \geq 1-\delta.$

In the special case when $ O$ is also diagonal in the $X$-basis, i.e., $P_i\in \set{X_i, I_i}\forall i\in [n],$ we have the stronger guarantee $ \Pr\left[\left|\widehat{\operatorname{Tr}(O \rho_\beta)} -\operatorname{Tr}(O \rho_\beta)\right|\leq \epsilon|\operatorname{Tr}(O \rho_\beta)| \right] \geq 1-\delta.$                                                                                                                                                                 
                                                                                                                                                                                                                                                                                                         
                                                                                                           
\end{enumerate}

\label{thm:hard regime informal}
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                             
                                                              
Fix $ \eta ,\beta > 0.$ For any $\varepsilon >0,$ there exists a family of symmetric matrices $J\in \mathbb{R}^{n\times n}$ with $ \Delta(J) \cdot \frac{\tanh (\beta \eta) }{\eta} \in (1, 1+\varepsilon),$ such that approximating $\operatorname{Tr}\left(e^{-\beta\left( \frac{1}{2}\sum_{ij} J_{ij} Z_i Z_j + \eta \sum_i X_i \right)}\right)$
    within an $\exp(c n)$-multiplicative factor is $\textnormal{NP}$-hard, and thus unlikely to admit an efficient classical or quantum algorithm assuming $ \textnormal{NP} \not\subseteq \textnormal{BQP}$ and $ \textnormal{NP} \neq \textnormal{RP}$.  
    Here $c$ is a constant dependent only on $\varepsilon, \eta,\beta.$ Moreover, the matrices in this family can be chosen to have zero diagonal and rational entries of $O_{\eta,\beta,\varepsilon}(1)$ bit length (see \cref{remark:hardness finite precision and zero diagonal}).

\label{def:general path definition}

For $\eta\geq 0$ and $p\in\{0,1\}$, define
\begin{equation}\label{eq:normalization factor for each index}
Z_{\eta, (p)}= \frac{1}{2}\operatorname{Tr}(X^p e^{\eta X}) = \sum_{\substack{d\geq 0\\d\equiv p \mod 2}}  \frac{\eta^d }{d!} =  \begin{cases} \cosh\eta, &\text{ if } p = 0\\ \sinh \eta, &\text{ if } p = 1\end{cases}.   
\end{equation}

Construct the probability measure $\nu_{\eta,(p)}$ over functions $ \sigma:\mathbb{S} \to \set{\pm 1}$ as follows:
\begin{enumerate}
    \item Sample $D\sim \mathrm{Pois}(\eta, p),$  where $\mathrm{Pois}(\eta, p)$ is the Poisson distribution $\mathrm{Pois}(\eta)$ conditioned on $D\equiv p \mod 2$. 
    
    For $\eta > 0$, $\mathrm{Pois}(\eta, p) $ is defined by
    \[
    \Pr_{D\sim \mathrm{Pois}(\eta, p) } [D=d]=\frac{\eta^d}{d!\,Z_{\eta, (p)}},
    \qquad d\geq0,\quad d\equiv p\mod2.
    \]
    For $ \eta =0,$ taking the limit as $\eta\to 0$, we have $\Pr_{D\sim \mathrm{Pois}(0, p) } [D=d] = \mathbbm{1}[d = p];$ thus   $\mathrm{Pois}(0, p) $ places all probability mass on $p.$
    \item Conditional on $D=d$, sample $d$ independent uniform

    \sigma(t)=u(-1)^{\#\{j\in [d]:t^{(j)}\leq t\}}
    .
    $  Let
    $
    \mathcal J(\sigma)

    =\{t^{(1)},\ldots,t^{(d)}\}.
    $

\label{def:1D Ising model CBC}
For interaction $\mathbf{K} = (K^{(k)})_{k=0}^{M-1}\in \mathbb{R}^M$ and external field $\mathbf{w} = (w^{(k)})_{k=0}^{M-1} \in \mathbb{R}^M$, the 1D Ising model with closed boundary conditions (CBC) $\pi_{\mathbf{K}, \mathbf{w}}^{\mathrm{1D Ising, CBC}} $ is the probability distribution  over $\set{\pm 1}^M$ defined by: for any configuration $\mathbf{s} = (s^{(0)}, \dots, s^{(M-1)}) \in \set{\pm 1}^M$,
\begin{equation}\label{def:1D Ising with closed boundary}
\pi_{\mathbf{K}, \mathbf{w}}^{\mathrm{1D Ising, CBC}}(\mathbf{s}) 
= \frac{1}{Z_{\mathbf{K}, \mathbf{w}}^{\mathrm{1D Ising, CBC}}}
\exp\left( \sum_{k=0}^{M-1} K^{(k)} s^{(k)} s^{(k+1)} + \sum_{k=0}^{M-1} w^{(k)} s^{(k)} \right), \quad \text{where } s^{(M)} = s^{(0)}.
\end{equation}
This definition extends to allow infinite fields, i.e., $ w^{(k)} \in \set{\pm \infty}$ by taking limits. Infinite fields are interpreted as
conditioning on $s^{(k)} =\operatorname{sign}(w^{(k)})$ whenever
$w^{(k)}\in \set{\pm \infty}.$

\label{thm:main theorem tractable}
Consider a symmetric matrix $J = (J_{ij})_{i,j\in [n]}\in \mathbb{R}^{n\times n}, \boldsymbol{h^z}=(h_i^z)_{i=1}^n\in\mathbb R^n, \boldsymbol{\eta} =(\eta_i)_i\in \mathbb{R}^n.$                                                                  

                                                                                                                                 
Let $ \eta_{\min} =\min_{i} |\eta_i|.$ Let $\alpha(\eta) = \frac{\tanh \eta}{\eta}$\footnote{Here, we follow the convention $\frac{\tanh(x)}{x}= 1$ when $ x=0.$}. 
Suppose $\Delta(J) \cdot \alpha(\eta_{\min}) \leq 1.$

Let  $H =\frac{1}{2}\sum_{i, j} J_{ij} Z_i Z_j +\sum_{i=1}^n h_i^z Z_i + \sum_{i} \eta_i X_i$ and $\rho = e^H/\operatorname{Tr}(e^H).$

Fix $\epsilon,\delta\in (0,1).$ There exists a randomized classical algorithm that runs in $\text{poly}(n, \max_{i,j} |J_{ij}| , \max|h_i^z|,  \max |\eta_i|, \epsilon^{-1}) \cdot \log \delta^{-1}$ time and:
\begin{enumerate}
\item Compute $ \widehat{\operatorname{Tr}( e^H)}$ s.t. 
$\Pr[\widehat{\operatorname{Tr}(e^H)} = e^{\pm \epsilon} \operatorname{Tr}(e^H)]\geq 1-\delta.$
    \item Given $ S\subseteq [n]$ and $X_S=\prod_{i\in S} X_i$, compute $ \widehat{\operatorname{Tr}(X_S \rho)}$ s.t. 
$\Pr[\widehat{\operatorname{Tr}(X_S \rho)} = e^{\pm \epsilon} \operatorname{Tr}(X_S \rho)]\geq 1-\delta.$
\item Given a Pauli string $O=\prod_{i=1}^n P_i,$ compute $ \widehat{\operatorname{Tr}(O\rho)}$ s.t.
$\Pr\left[\left|\widehat{\operatorname{Tr}(O \rho)} -\operatorname{Tr}(O \rho)\right|\leq \epsilon \right] \geq 1-\delta.$
\end{enumerate}

\label{thm:main mixing theorem}
Fix an integer $M\geq 3.$ Let $\Omega = \set{\pm 1}^M .$ 

For $i \in [n],$ consider  $K_i = (K_i^{(k)})_{k=0}^{M-1}, w_i= (w_i^{(k)})_{k=0}^{M-1},$ and let $ \pi_i =\pi_{K_i,w_i}^{\mathrm{1D Ising, CBC}}. $ 

Consider a symmetric matrix $J = (J_{ij})_{i,j\in [n]}\in \mathbb{R}^{n\times n},$ and let $W$ be as in \eqref{eq: VM and W}, i.e.,
$ W (\mathbf{s}) = \frac{1}{2M} \sum_{i,j} J_{ij} \langle \mathbf{s} _i, \mathbf{s}_j \rangle.$ 

Let $ \pi = \bigotimes_{i=1}^n \pi_i$ and for $\lambda\in [0,1],$ let $\mu_\lambda = \mathcal{T}_{\lambda W} \pi.$ We view $\mu_\lambda$ as a distribution over $\Omega^n.$ We let  $\mu \equiv \mu_1.$  

                                                                    
     Suppose $\forall i, k: |K_i^{(k)}|\leq K$ for $K \in (0,+\infty].$ Let  $\eta \geq 0$ be s.t. $ K = \frac{1}{2} \ln \coth (\eta/M)$, and let $\alpha_M(\eta) = \frac{\tanh \eta}{M\tanh(\eta/M)}.$\footnote{Here, we follow the convention $\frac{\tanh(x)}{x}= 1$ when $ x=0,$ so that $ \alpha_M(\eta) = 1 $ when $\eta =0.$} 
                                                                                                                                         
     
     In particular, this condition holds when $ K_i \in \set{\mathbf{K}_{\eta_i, (p)} \mid p\in \set{0,1} }$ with $\eta_i \geq \eta,$ where $ \mathbf{K}_{\eta_i, (p)}$ is as defined in \cref{prop:approximation of path functional}.                                                                                                                                                                                                                                                                                                 
                                                                                                                                                                           
                                            
     
    
   Let $ L = n \lceil \| J\|_\infty \rceil +1.$ 
  The following holds:
  \begin{enumerate}
      \item 
                                                          
Fix $\lambda \in [0,1].$  
Suppose $ \lambda \Delta(J) \alpha_M (\eta) < 1.$   
Let $\rho^{(t)}$ be the distribution of the Glauber dynamics\footnote{Each step of the Glauber dynamics chooses \(i\in[n]\) uniformly and resamples $\mathbf{s}_i\in \set{\pm 1}^M$ conditioned on $\mathbf{s}_{-i} = (\mathbf{s}_j)_{j\neq i}.$} with respect to $\mu_\lambda$ initialized at $\rho^{(0)}\ll \mu_\lambda$ after $t$ steps. Each step of the Glauber dynamics contracts KL-divergence by $\left(1-\frac1{Cn}\right)$, i.e.,
\[
D_{\operatorname{KL}}(\rho^{(t+1)}\|\mu_\lambda)
\leq \left(1-\frac1{Cn}\right)
D_{\operatorname{KL}}(\rho^{(t)}\|\mu_\lambda),
\quad C = \frac{1}{1 - \lambda \Delta(J) \alpha_M (\eta) }.\]  
Let $\rho^{(0)}=\pi.$    
Fix $\epsilon_{TV}\in(0,1).$
If 
$t\geq 2 (C n \log (L/\epsilon_{TV})),$ 
then
$ d_{TV}(\rho^{(t)} , \mu_\lambda) \leq \epsilon_{TV} .$ Sampling from $\pi$ and
   implementing  each step of the Glauber dynamics take $O(n M)$ (classical) time. Hence, there exists a classical algorithm that samples from $ \hat{\mu}_\lambda$ with $ d_{TV}(\hat{\mu}_\lambda, \mu_\lambda)\leq \epsilon_{TV}$ in time $O\left(\frac{n \log (L/\epsilon_{TV})}{1 - \lambda \Delta(J) \alpha_M (\eta) } \cdot nM \right).$
     \item  Suppose $ (1-1/L) \Delta(J) \alpha_M (\eta) < 1.$   
    For $ \epsilon,\delta\in (0,1),$ there exists a classical algorithm that runs in time \[T = O\left(\frac{n \log (L/\epsilon)}{1 - (1-1/L) \Delta(J) \alpha_M (\eta) } \cdot nM \cdot \frac{L^2}{\epsilon^2}\cdot \log\frac{1}{\delta}\right) \] and outputs an approximation $\hat{Z} $ of  $ Z =\mathbb{E}_{\pi}[e^W] $  so that $\Pr[ \hat{Z} = \exp(\pm \epsilon) Z ] \geq 1-\delta.$
  \end{enumerate}

<!-- theorem-5: body_md -->
\label{thm:semilogconcave}
Suppose that $\mu = \bigotimes_{i = 1}^n \mu^{(i)}$ is a product measure where each $\mu^{(i)}$ is a probability measure on $(\mathbb S^{N-1})\subseteq \mathbb{R}^N$. Let $J \succeq 0$, and define the measure $\nu$ by its density
$\frac{d\nu}{d\mu}(x) \propto \exp\left(\frac{1}{2} \sum_{i,j} J_{ij} \langle x_i, x_j \rangle \right). $
                                
Suppose that for all $i \in [n]$, the measure $ \mu^{(i)}$ is $ \rho$ semi-log concave, 
meaning that for all $w ,$ $ \| \operatorname{Cov}(T_w \mu^{(i)})\|_{\operatorname{op}}\leq \rho ,$ with $ \| J\|_{\operatorname{op}} \rho <1.$
Then approximate tensorization of entropy (ATE) for $\nu$ holds with constant at most $ C = \frac{1}{1 - \| J\|_{\operatorname{op}} \rho},$ and thus each step of the Glauber dynamics on $ \nu$ contracts KL-divergence by $\left(1-\frac1{Cn}\right).$ \footnote{For background on ATE and Glauber dynamics, see \cite[Sections 2.2 and 2.3]{AKV24}.}

        \mu_\lambda(\sigma) = \frac{\exp(\lambda V(\sigma))\nu(\sigma)}{Z(\lambda)}$, where $\quad Z(\lambda) = \mathbb{E}_{\nu}[\exp(\lambda V(\sigma))].
    $
    For any $\epsilon,\delta \in (0,1)$, there exists a classical algorithm that outputs a strictly positive estimator $\hat{Z}$ such that
    \[
        \Pr\left[(1-\epsilon) Z(1) \leq \hat{Z} \leq (1+\epsilon) Z(1)\right] \geq 1- \delta.
    \]

Let $ O$ be an arbitrary Pauli string observable.
Conjugating by $U = \prod_{i: \eta_i < 0} Z_i$ yields $U H U^\dag = H' := H_z + \sum_{i=1}^n |\eta_i| X_i$ and $U O U^\dag = s O$, for $s \in \set{\pm 1}$ computable in $O(n)$ time. Thus $\operatorname{Tr}(O e^H) = s \operatorname{Tr}(O e^{H'})$.  Hence, it suffices to estimate $\operatorname{Tr}(O e^{H'}).$ We therefore assume without loss of generality that $\eta_i \geq 0\forall i.$                                                                                                                  

    Let $\eta \equiv \eta_{\min}.$
   We use the same notation and setup as in \cref{prop:derive expression for observables}.

   \begin{equation}\label{eq:set M and L}
      c  =  \| J\|_{\infty} +\| \mathbf{h}^z\|_\infty+ \| \boldsymbol{\eta}\|_\infty+1,\quad L = n \lceil \| J\|_\infty \rceil +1,\quad
    M =\lceil 1000 \cdot \max \left\{\frac{n c^2 }{\epsilon} , \left (\frac{nc}{\epsilon}\right)^{3/2}\right\}\rceil.  
   \end{equation} 
                                                                                   

We note that for parts 1 and 2, it suffices to compute a multiplicative approximation of $ \operatorname{Tr}(X_S e^H) $ given $S\subseteq [n].$

By \cref{prop:multiplicative approx of trotterized X observable} and the choice of $M$, we have $ \operatorname{Tr}(X_S e^H) =\exp(\pm \epsilon/8) \operatorname{Tr}(X_S Q_M). $


    To approximate $\operatorname{Tr}(X_S Q_M) ,$ we recall from \cref{prop:derive expression for observables} that \[\operatorname{Tr}(X_S Q_M) = D\mathbb{E}_{\tilde{\nu}}[e^W], \text{ where } D = \operatorname{Tr} (X_S ( e^{\frac{1}{M}\sum_i h_i^z Z_i } e^{H_x/M})^M ) = \prod_{i\in [n]} \operatorname{Tr}(X^{\mathbbm{1}_{i\in S}} (e^{\frac{h_i^z}{M}  Z} e^{\frac{\eta_i}{M}  X} )^M) \geq 0. \] 
    We compute $D$ in $O(nM)$ time. Next, we use \cref{thm:main mixing theorem} with accuracy parameters $\epsilon/8$ and $\delta/2$ to approximate $\mathbb{E}_{\tilde{\nu}}[e^W]$ and output $\widehat{\operatorname{Tr}(X_S e^H) } = \hat{Z} D$. We have: 
     \[\Pr[\widehat{\operatorname{Tr}(X_S e^H) } =\exp(\pm \epsilon/4) \operatorname{Tr}(X_S e^H)]  \geq \Pr[\hat{Z}= \exp(\pm \epsilon/8) \mathbb{E}_{\tilde{\nu}}[e^W]] \geq 1-\delta/2\]
For part 1, we apply the above procedure to $S =\emptyset.$ For part 2, we apply the above procedure for $ S$ and $\emptyset,$ compute $ \widehat{\operatorname{Tr}(X_S e^H) }$ and $\widehat{\operatorname{Tr}(e^H) },$ and output $ \widehat{\operatorname{Tr}(X_S \rho)} = \frac{\widehat{\operatorname{Tr}(X_S e^H) }}{\widehat{\operatorname{Tr}(e^H) }}.$ We have:
\[\Pr[ \widehat{\operatorname{Tr}(X_S \rho)}  = \exp(\pm \epsilon/2) \operatorname{Tr}(X_S \rho)] \geq 1-\delta. \]

   Next, we prove part 3. Let $\epsilon' = \frac{\epsilon}{100}.$
  Applying \cref{prop:addition apporoximation of Pauli observables}, we have $ \left|\operatorname{Tr} (O \rho) - \frac{\operatorname{Tr}(O Q_M) }{\operatorname{Tr}(Q_M) }\right| \leq \epsilon', $ where we recall that by \cref{prop:multiplicative approx of trotterized X observable}, $ \operatorname{Tr}(Q_M)> 0.$  Fix $ \lambda= 1-1/L.$ Using \cref{prop:derive expression for observables}, we can rewrite

   where $ \mu_\lambda= \mathcal{T}_{\lambda W} \tilde{\nu},$ and $f$ is the real-valued function defined in \eqref{eq: VM and W}. 
   
   By our choice of $M$, and the fact that $ |W| \leq L,$ we have $ e^{\lambda W}\leq e^W  e^{|W|/L}\leq e^{W} \cdot e,$ thus
                                                                                                                                                                 
\[
B\leq e \cdot \frac{D  \mathbb{E}_{\tilde{\nu}}[e^{W}] }{\operatorname{Tr}(Q_M) } = e\cdot \frac{\operatorname{Tr}(X_SQ_M)}{\operatorname{Tr}(Q_M)}
\le e\bigl(\operatorname{Tr}(X_S\rho)+\epsilon'\bigr)
\le e(1+\epsilon')
                     
\]

   By the above argument,  we can compute $\widehat{\mathbb{E}_{\tilde{\nu}}[e^{\lambda W}]} $ and $\widehat{\operatorname{Tr}(Q_M) } $ such that:  \[\Pr[\widehat{\mathbb{E}_{\tilde{\nu}}[e^{\lambda W}]} = \exp(\pm \epsilon')  \mathbb{E}_{\tilde{\nu}}[e^{\lambda W}] \text{ and } \widehat{\operatorname{Tr}(Q_M) } = \exp(\pm\epsilon') \operatorname{Tr}(Q_M)] \geq 1-\delta/2.\] Let $ \hat{B} = \frac{D \widehat{\mathbb{E}_{\tilde{\nu}}[e^{\lambda W}]}}{\widehat{\operatorname{Tr}(Q_M) } }.$ Then, with probability $ \geq 1-\delta/2,$ we have: \[0\leq e^{-2\epsilon'}  B \leq  \hat{B} \leq e^{2\epsilon'}  B .\]
   
   We approximate $A:=\mathbb{E}_{\mu_\lambda}[e^{W/L} f] .$ Note that $ |e^{W/L} f|\leq e,$ thus $|A| \leq e.$

\label{prop:covariance bound 2 spin}

    \begin{align*}

<!-- definition-3: body_md -->
\label{def:LDP}
A sequence of probability measures $\{\rho_n\}_{n\ge 1}$ satisfies the
large deviation principle with rate function $I$\footnote{Here, we implicitly assume that $I$ is a lower semicontinuous map into $[0,\infty]$.} if, for every measurable
set $\Gamma$,
\[
-\inf_{x\in \Gamma^\circ} I(x)
\le
\liminf_{n\to\infty} \frac{1}{n}\log \rho_n(\Gamma)
\le
\limsup_{n\to\infty} \frac{1}{n}\log \rho_n(\Gamma)
\le
-\inf_{x\in \overline{\Gamma}} I(x),
\]
where $\Gamma^\circ$ and $\overline{\Gamma}$ denote  the interior and closure of $\Gamma$, respectively.

\label{lem:application of LDP}
 Let $\sigma_1,\dots,\sigma_n$ be i.i.d.\ samples from $\nu_\eta$, and let $\tilde{\rho}_n$ be the distribution of their average $ \bar{\sigma} = \frac{1}{n} \sum_{i=1}^n \sigma_i.$

 \Lambda_{\nu_\eta}(\lambda) = \log \mathbb{E}_{\sigma \sim \nu_\eta}[\exp(\langle \lambda, \sigma\rangle)].    
 \end{equation}

\tilde{I} -F
\end{equation}

     Since $\mathcal{B}$ is convex, the measures $ (\tilde{\rho}_n)_n$ are also supported on $\mathcal{B}.$ Since $F$ is continuous and bounded on $\mathcal{B},$   
the first statement and Varadhan's theorem (see \cite[Chapter 3, Exercise 3.9]{RS15}) imply that the sequence of measures $(\rho_n)_n$ satisfies an LDP with rate $J (x) = I(x)-\inf I$, where $I = \tilde{I}-F.$

\[ \tilde{I}(x)\geq b \| x\|^2 - \Lambda_{\nu_\eta}( b x)  \]
thus
\[ I(x) = \tilde{I}(x)- F(x) \geq F(x) -  \Lambda_{\nu_\eta}( b x)  = -\Phi_{b,\eta} (bx) \]

\[

+v\langle\sigma,\tau\rangle\right)\right]= (2\cosh\eta)^2

+v B}\right],
\]
where \(A_\sigma= \langle\sigma, \mathbf{1}\rangle=\int_{\mathbb S}\sigma(t)\,dt\),  $B:=\langle\sigma,\tau\rangle
  =\int_{\mathbb S}\sigma(t)\tau(t)\,dt$, and
\(\sigma\) and \(\tau\) are independent. 

Fix $v >0.$
Subtracting the expressions for \(F(v)\) and \(F(-v)\), and using the Taylor series for $\sinh(x)$, we obtain

\right].
\label{eq:F-odd-expansion}

For each \(k\ge1\), we have:
\begin{align*}
B^k
&=\left(\int_{\mathbb S}\sigma(t)\tau(t)\,dt\right)^k=\int_{\mathbb S^k}
\prod_{j=1}^k \sigma(t_j)\tau(t_j)\,
