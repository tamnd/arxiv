<!-- definition-1: body_md -->
\label{def:Cartan_component}
 Let $\mu,\nu\in\mathbf{P}^+$.
  The \emph{Cartan component} of $\mathcal{B}(\mu)\otimes\mathcal{B}(\nu)$ is the unique irreducible subcrystal of highest weight $\mu+\nu$.
  More generally, for $\mu_n,\ldots,\mu_0\in\mathbf{P}^+$, the Cartan component of $\mathcal{B}(\mu_n)\otimes\cdots\otimes\mathcal{B}(\mu_0)$ is the unique irreducible subcrystal of highest weight $\sum_{j=0}^n\mu_j$.

<!-- lemma-1: body_md -->
The truncation and shift maps satisfy the following relations, whenever $\mu+\nu\leq\lambda$:
\begin{align}
   & \mathsf{R}_\nu \circ \mathsf{R}_{\mu+\nu} = \mathsf{R}_{\nu} :  \mathcal{B}(\lambda) \to \mathcal{B}(\nu),
   \label{eq:RR} \\
   & \mathsf{S}_\nu \circ \mathsf{S}_{\mu} = \mathsf{S}_{\mu+\nu} :  \mathcal{B}(\lambda) \to \mathcal{B}(\lambda-\mu-\nu),
   \label{eq:SS} \\
   & \mathsf{S}_{\mu}\circ\mathsf{R}_{\mu+\nu} = \mathsf{R}_{\nu}\circ\mathsf{S}_{\mu} : \mathcal{B}(\lambda) \to \mathcal{B}(\nu).
   \label{eq:SR}
\end{align}

One can directly check these relations on an element $b_2\otimes b_1\otimes b_0$ in the Cartan component of $\mathcal{B}(\lambda-\mu-\nu) \otimes \mathcal{B}(\mu)\otimes\mathcal{B}(\nu)$.

<!-- definition-2: body_md -->
\label{def:B_infty}
    The \emph{semi-infinite crystal} $\overleftarrow{\mathcal{B}}\!(\infty)$ is the inverse limit of the system of finite crystals,
    \[
      \overleftarrow{\mathcal{B}}\!(\infty) = \underset{\lambda\in\mathbf{P}^+}{\underleftarrow{\lim}}\mathcal{B}(\lambda),
    \]
    with connecting maps~\eqref{eq:directed_system}.
    This means that an element of $\overleftarrow{\mathcal{B}}\!(\infty)$ is a family $(b_\lambda)_{\lambda\in\mathbf{P}^+}$  with $b_\lambda\in\mathcal{B}(\lambda)$ for every $\lambda$ and such that $\mathsf{R}_\mu(b_{\lambda}) = b_\mu$ for all $\lambda\geq\mu$.

    We equip $\overleftarrow{\mathcal{B}}\!(\infty)$ with the inverse limit topology.  So a base of clopen sets is the collection of cylinder sets $\mathsf{R}_\mu^{-1}(a)$ indexed by $\mu\in\mathbf{P}^+$ and $a\in\mathcal{B}(\mu)$.

<!-- definition-3: body_md -->
\label{def:crystal_groupoid}
    The \emph{crystal groupoid} of the simply connected compact semisimple Lie group $K$ is the groupoid $\mathcal{G}(K) \rightrightarrows \overleftarrow{\mathcal{B}}\!(\infty)$ given by
    \[
        \mathcal{G}(K) = \{ (x,\mu-\nu,y) \in \overleftarrow{\mathcal{B}}\!(\infty) \times \mathbf{P} \times \overleftarrow{\mathcal{B}}\!(\infty) \mid  \mu,\nu\in\mathbf{P}^+,\ \mathsf{S}_{\mu}(x) = \mathsf{S}_{\nu}(y) \}
    \]
    with range, source, product and inverse given by
    \begin{align*}
     & r(x,\mu,y) = x, &
     & s(x,\mu,y) = y, \\
     & (x,\mu,y)(y,\nu,z) = (x, \mu+\nu, z), &
     & (x,\mu,y)^{-1} = (y,-\mu,x).
    \end{align*}
<!-- end definition-3 -->
<!-- lemma-2: body_md -->
\label{lem:topology}

<!-- definition-4: body_md -->
\label{def:Cartan_words}
  Let $(\mu_j)_{j=0}^\infty$ be a sequence of dominant weights.  We will be particularly interested in the case where all the $\mu_j$ are equal, and are strictly dominant, but we do not want to limit ourselves to that situation.

  A finite word
  \begin{equation}
  \label{eq:finite_word}
    x_nx_{n-1} \ldots x_2x_1x_0 \ \in\  \prod_{j=0}^n \mathcal{B}(\mu_j)
  \end{equation}
  will be called a \emph{Cartan word} if $x_n\otimes\cdots\otimes x_0$ belongs to the Cartan component of $\mathcal{B}(\mu_n)\otimes\cdots\otimes\mathcal{B}(\mu_0)$.
  A semi-infinite word
  \begin{equation}
  \label{eq:semi-infinite_word}
    x= \cdots x_3x_2x_1x_0 \ \in\ \prod_{j=0}^\infty \mathcal{B}(\mu_j)
  \end{equation}
  will be called a \emph{semi-infinite Cartan word} if every finite subword is a Cartan word.
  We will write $\emptyset$ for the unique element of the trivial crystal $\mathcal{B}(0)$, and view it as the empty word.

<!-- proposition-1: body_md -->
\label{prop:words}
Let $(\mu_j)_{j\in\mathbb{N}}$ be a sequence of dominant weights which is {additively cofinal}, in the sense that
\begin{equation}
  \label{eq:cofinal}
  \forall \lambda \in \mathbf{P}^+, ~ \exists n\in\mathbb{N} \text{ such that } \lambda \leq \sum_{j=0}^n \mu_j.
\end{equation}
There is a one-to-one correspondence between the set of semi-infinite Cartan words in $\prod_{j\in\mathbb{N}}\mathcal{B}(\mu_j)$ and the semi-infinite crystal $\overleftarrow{\mathcal{B}}\!(\infty)$, given by
\begin{equation}
  \label{eq:word-decomposition}
  \ldots x_3 x_2 x_1 x_0 \ \mapsto\  x=(b_\lambda)_{\lambda\in\mathbf{P}^+},
\end{equation}
where
\begin{equation}
  b_\lambda = \mathsf{R}_\lambda(x_nx_{n-1}\ldots x_1x_0)
\end{equation}
for any $n$ large enough so that $\lambda \leq \sum_{j=0}^n \mu_j$.
<!-- end proposition-1 -->
<!-- proof-2: body_md -->
Let $x = (b_\lambda)_{\lambda\in\mathbf{P}^+}\in\overleftarrow{\mathcal{B}}\!(\infty)$.  Let us write $\lambda_n = \sum_{j=0}^n\mu_j$. Then for any $n$, we can realise $b_{\lambda_n}$ as a finite Cartan word,
\[
  b_{\lambda_n} = x_n\ldots x_1 x_0 \qquad \text{with }x_j \in \mathcal{B}(\mu_j),
\]
where according to Equation~\eqref{eq:letter},
\[
  x_k = \mathsf{R}_{\mu_k}\circ\mathsf{S}_{\sum_{j=0}^{k-1}\mu_j}(b_{\lambda_n}).
\]
The compatibility relations \eqref{eq:RR}--\eqref{eq:SR} ensure that this  $x_k$ is independent of the word length $n$.  It is then straightforward to check that the map $x\mapsto \ldots x_3x_2x_1x_0$ is inverse to~\eqref{eq:word-decomposition}.

  \label{def:Cartan_braiding}
  Let $\mu,\nu\in\mathbf{P}^+$, and let us write
  \[
    \iota_{\mu,\nu}:\mathcal{B}(\mu+\nu) \to \mathcal{B}(\mu)\otimes\mathcal{B}(\nu)
  \]
  for the inclusion of the Cartan component.  The \emph{Cartan braiding} is the map
  \begin{equation*}
    \sigma = \sigma_{\mathcal{B}(\mu),\mathcal{B}(\nu)}:(\mathcal{B}(\mu) \otimes \mathcal{B}(\nu))\sqcup\{0\}  \to (\mathcal{B}(\nu) \otimes \mathcal{B}(\mu))\sqcup\{0\}
  \end{equation*}
  defined by
  \begin{equation*}
    \sigma: a\otimes b  \mapsto
    \begin{cases}
      \iota_{\nu,\mu}\circ\iota_{\mu,\nu}^{-1}(a\otimes b) &
        \text{if $a\otimes b$ is in the Cartan component,} \\
      0 & \text{otherwise.}\\

    \end{cases}
  \end{equation*}

  If $\mu_n,\ldots,\mu_0 \in \mathbf{P}^+$, we write
  \begin{multline*}
    \sigma_j : \mathcal{B}(\mu_n) \otimes \cdots \otimes \mathcal{B}(\mu_{j}) \otimes \mathcal{B}(\mu_{j-1}) \otimes \cdots \otimes \mathcal{B}(\mu_0) \\
    \to
    \mathcal{B}(\mu_n) \otimes \cdots \otimes \mathcal{B}(\mu_{j-1}) \otimes \mathcal{B}(\mu_{j}) \otimes \cdots \otimes \mathcal{B}(\mu_0) \; \sqcup\{0\}
  \end{multline*}
  for the action
  \[
  \sigma_j = \operatorname{id}_{\mu_n} \otimes \cdots \otimes\operatorname{id}_{\mu_{j+1}} \otimes \sigma \otimes \operatorname{id}_{\mu_{j-2}} \otimes \cdots \otimes\operatorname{id}_{\mu_0}
  \]
  of the Cartan braiding on the $j$th and $(j\!-\!1)$th tensor legs.

<!-- lemma-3: body_md -->
\label{lem:relations}
Let $a\in\mathcal{B}(\mu)$, $b\in\mathcal{B}(\nu)$.  We have:
\begin{enumerate}
    \item[(a)]
    $\mathbf{1}_{Z(\emptyset,\emptyset)}=1$,
    \item[(b)]
    $\mathbf{1}_{Z(a,b)}^* = \mathbf{1}_{Z(b,a)}$,
    \item[(c)]
    $\mathbf{1}_{Z(\emptyset,a)}\mathbf{1}_{Z(\emptyset,b)} = \mathbf{1}_{Z(\emptyset,ab)}$ if $ab$ is a Cartan word, and $0$ otherwise.
    \item[(d)]
    $\mathbf{1}_{Z(a,\emptyset)}\mathbf{1}_{Z(\emptyset,b)} = \mathbf{1}_{Z(a,b)}$,
    \item[(e)]
    \(
      \displaystyle\mathbf{1}_{Z(\emptyset,a)}\mathbf{1}_{Z(b,\emptyset)} = \!\!\! \sum_{a'a=b'b \in \operatorname{CCE}(a,b)} \!\!\!\!\!\!\mathbf{1}_{Z(a',b')}.
    \)
\end{enumerate}
Thus, $\bigcup_{\mu \in \mathbf{P}^+} \big\{\mathbf{1}_{Z(\emptyset, a)} \mid a \in \mathcal{B}(\mu)\big\}$ generates $C_\mathrm{c}^\infty(\mathcal{G}(K))$ as a $*$-algebra.
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
These are elementary calculations.  For (a), note that $Z(\emptyset,\emptyset)$ is the set of units of $\mathcal{G}(K)$.  Part (b) is direct from the definitions.  For (c), note that the composable pairs in $Z(\emptyset,a) \times Z(\emptyset,b)$ are precisely those of the form $((z,-\mu,za),(za,-\nu,zab))$ with $z\in\overleftarrow{\mathcal{B}}\!(\infty)$. The proof of (d) is similar.

For (e), we observe that elements in $Z(\emptyset,a)$ and $Z(b,\emptyset)$ can be written, respectively, in the form
\begin{align*}
  &(za',-\mu,za'a), && \text{for some $z\in\overleftarrow{\mathcal{B}}\!(\infty)$, $a'\in\mathcal{B}(\nu)$,}\\
  &(z'b'b,\nu,z'b'), && \text{for some $z'\in\overleftarrow{\mathcal{B}}\!(\infty)$, $b'\in\mathcal{B}(\mu)$.}
\end{align*}
These elements are composable in $\mathcal{G}(K)$ if and only if $a'a=b'b$ and $z=z'$.  In this case, the composition is
\begin{equation}
\label{eq:decomposition}
(za',-\mu,za'a)(za'a,\nu,zb') = (za',\nu-\mu,zb').
\end{equation}
To complete the calculation, we need to check that every element $(za',\nu-\mu,zb')\in Z(a',b')$ admits a unique decomposition as a product in $Z(\emptyset,a)\times Z(b,\emptyset)$.  Certainly, if such a decomposition exists it is unique, since it must have the form~\eqref{eq:decomposition}.  Existence amounts to proving that $za'a$ is a Cartan word, \emph{i.e} that $z_\lambda a'a$ is a Cartan word for any finite crystal element $z_\lambda\in\mathcal{B}(\lambda)$ such that $z_\lambda a'$ and $z_\lambda b'$ are both Cartan words.  For this, we use the Cartan braiding criterion of Proposition~\ref{prop:Cartan_test}.  Our hypothesis is equivalent to $\sigma(z_\lambda\otimes a')\neq0$ and $\sigma(z_\lambda\otimes b')\neq 0$.  But then
\[
  \sigma_2(z_\lambda\otimes a' \otimes a) \neq0,\quad\text{and}\quad
  \sigma_2\circ\sigma_1(z_\lambda\otimes a'\otimes a) = \sigma_2(z_\lambda\otimes b'\otimes b) \neq 0,
\]
from which we see that $z_\lambda\otimes a'\otimes a$ is in the Cartan component.  The result follows.

<!-- theorem-2: body_md -->
\label{thm:isomorphism}
There is a $\mathbf{P}$-graded $*$-isomorphism
    $\mathcal{O}(K_0)\cong C_\mathrm{c}^\infty(\mathcal{G}(K))$ and $C(K_0)\cong C^*(\mathcal{G}(K))$, where $\mathcal{O}(K_0)$ and $C(K_0)$ are equipped with the opposite $\mathbf{P}$-grading to~\eqref{eq:OKq_gauge_action}.  The isomorphism is given on generators by
    \begin{equation}
    \label{eq:isomorphism}
      f_a  \mapsto \mathbf{1}_{Z(\emptyset,a)}
    \end{equation}
    for every $a\in\mathcal{B}(\mu)$.
<!-- end theorem-2 -->
<!-- proof-4: body_md -->
By \cite[\S{}8-9]{MatYun:crystals}, $\mathcal{O}(K_0)$ is the universal $*$-algebra with generators $f_a$, with $a$ varying over all finite crystals, and the following relations:
    \begin{enumerate}
        \item[(1)]
        for any $a\in\mathcal{B}(\mu)$, $b\in\mathcal{B}(\nu)$,
        $f_af_b = f_{ab}$ if $a\otimes b$ is in the Cartan component, and $f_af_b=0$ otherwise;
        \item[(2)]
        for any $a\in\mathcal{B}(\mu)$, $b\in\mathcal{B}(\nu)$,
        $f_af_b^* = \!\!\!\displaystyle\sum_{a'a=b'b \in \operatorname{CCE}(a,b)} \!\!\!\!\!\! f_{a'}^*f_{b'}$;
        \item[(3)]
        for any $\mu\in\mathbf{P}^+$,
        $\displaystyle\sum_{a\in\mathcal{B}(\mu)} f_a^*f_a = 1.$
    \end{enumerate}
    It is straightforward to check, using Lemma~\ref{lem:relations}, that the generators $\mathbf{1}_{Z(\emptyset,a)}$ of $C_\mathrm{c}^\infty(\mathcal{G}(K))$ satisfy the same relations, so the formula~\eqref{eq:isomorphism} does define a $*$-morphism $\varphi_0 : \mathcal{O}(K_0) \to C_\mathrm{c}^\infty(\mathcal{G}(K))$.    Moreover, if $a\in\mathcal{B}(\mu)$ then the generators $f_a\in\mathcal{O}(K_0)$ and $\mathbf{1}_{Z(\emptyset,a)}\inC_\mathrm{c}^\infty(\mathcal{G}(K))$ are of degree $\mu$ and $-\mu$ for their respective $\mathbf{P}$-gradings, so $\varphi_0$ is grading-preserving, with the opposite $\mathbf{P}$-grading action on $\mathcal{O}(K_0)$.

    By \cite[Theorem~9.4]{MatYun:crystals}, the algebra $\mathcal{O}(K_0)$ is graded isomorphic to the Kumjian-Pask algebra \cite{MatYun:crystals} of a row-finite higher-rank graph $\Lambda$ without sources, again with the opposite $\mathbf{P}$-grading  on $\mathcal{O}(K_0)$.  Therefore, once we show that the images under $\varphi_0$ of the vertex projections are all nonzero, it will follow from the graded uniqueness theorem \cite[Theorem~4.1]{Aranda:Kumjian-Pask} that the map~\eqref{eq:isomorphism} is an isomorphism.  By \cite[Corollary 8.15]{MatYun:crystals}, the vertex projections can be written as sums of projections of the form $p_a= f_a^*f_a$ with $a\in\mathcal{B}(\mu)$ for any strictly dominant weight $\mu$.  But
    \[
      \varphi_0(p_a) = \varphi_0(f_a^*f_a) = \mathbf{1}_{Z(a,a)},
    \]
    and $Z(a,a)$ is non-empty because it contains $(b_{w_0}(\infty)a,0,b_{w_0}(\infty)a)$ where $w_0$ is the longest Weyl group element, see \cite[Lemma 5.2]{MatYun:crystals}.  Therefore, the images of the vertex projections are nonzero.

    This completes the proof for $\mathcal{O}(K_0)$. By \cite[Theorem~9.4]{MatYun:crystals} again, the aforementioned graded isomorphism $\mathcal{O}(K_0) \cong KP_\mathbb{C}(\Lambda)$ extends to an isomorphism $C(K_0) \cong C^*(\Lambda)$. The universal property of $C^*(\Lambda)$ therefore ensures that $\varphi_0$ extends to a homomorphism $\varphi : C(K_0) \to C^*(\mathcal{G}(K))$. This $\varphi$ intertwines the gauge actions because $\varphi_0$ is graded, and we already saw that each $\varphi(p_a) = \varphi_0(p_a) \not= 0$. So the Gauge-Invariant Uniqueness Theorem \cite[Theorem~3.4]{KumPas} shows that $\varphi$ is injective. It is surjective because its range contains the dense subalgebra $C_\mathrm{c}^\infty(\mathcal{G}(K))$ of $C^*(\mathcal{G}(K))$.

<!-- proposition-4: body_md -->
\label{prop:extremal_in_Demazure}
  Let $\lambda\in\mathbf{P}^+$ with support $I$, and let $v,w\in W$.  Then $b_v(\lambda) \in \mathcal{B}_w(\lambda)$ if and only if $[v]_I\leq[w]_I$ for the Bruhat order.
<!-- end proposition-4 -->
<!-- proof-7: body_md -->
If $[v]_I\leq[w]_I$ then it follows from the definitions that $\mathcal{B}_v(\lambda) \subseteq \mathcal{B}_w(\lambda)$ and hence $b_v(\lambda)\in\mathcal{B}_w(\lambda)$.

  For the converse, we work by induction on the length of a minimal coset representative of $[w]_I$.  Indeed, without loss of generality, we may suppose that $w$ is the minimal length element in its coset.
  If $\ell(w)=0$ then $\mathcal{B}_w(\lambda)=\{b_e(\lambda)\}$ so the result is trivial.  Now suppose $\ell(w)>0$ and assume we have proven the result for all $w'\in W$ with coset representative of length less than $\ell(w)$.  Let $s_i$ be a simple reflection such that $\ell(s_iw) = \ell(w) - 1$, meaning that $s_i$ is the first simple reflection in a reduced expression for $w$.  Since $b_v(\lambda)\in\mathcal{B}_{w}(\lambda)$ we have
  \begin{equation}
    \label{eq:Demazure_induction}
    b_v(\lambda) = f_i^m b \qquad \text{for some }b\in\mathcal{B}_{s_iw}(\lambda).
  \end{equation}
  Since $b_v(\lambda)$ is extremal, it is either at the top or the bottom of its $\alpha_i$-string.  If it is at the top then $m=0$ in Equation~\eqref{eq:Demazure_induction}, which means $b_v(\lambda)\in\mathcal{B}_{s_iw}(\lambda)$ and by induction we get $[v]_I\leq [s_iw]_I < [w]_I$.  If it is at the bottom, then $b_{s_iv}(\lambda) = e_i^M (b_{v}(\lambda))$, where $M\in\mathbb{N}$ is the largest integer such that $e_i^M(b_v(\lambda))\neq0$.  From this and~\eqref{eq:Demazure_induction} we get
  \[
    b_{s_iv}(\lambda) = e_i^{M-m} b \in \mathcal{B}_{s_iw}(\lambda),
  \]
  since Demazure crystals are stable under $e_i$.  By induction, we get $[s_iv]_I\leq [s_iw]_I$, \emph{i.e.} $s_iv' \leq s_iw$, where $s_iv'$ is the minimal length coset representative of $[s_iv]_I$.  
  Since $\ell(w) = \ell(s_iw)+1$, this implies $v' \leq w$. Since $[v']_I = [v]_I$, this completes the proof.
<!-- end proof-7 -->

<!-- theorem-4: body_md -->
\label{thm:orbit_topology}
  Under the bijection
  \[
    W \to \overleftarrow{\mathcal{B}}\!(\infty) / \mathcal{G}(K); \qquad w\mapsto[b_w(\infty)]
  \]
  of Theorem~\ref{thm:orbits}, the topology on $\overleftarrow{\mathcal{B}}\!(\infty)/\mathcal{G}(K)$ is determined by the fact that, for any $v,w\in W$,
  \begin{equation}
  \label{eq:orbit_topology}
    [b_v(\infty)] \in \overline{[b_w(\infty)]}
    \quad \Leftrightarrow \quad
    v \leq w
\end{equation}
<!-- end theorem-4 -->
<!-- proof-10: body_md -->
Firstly, suppose that $v \leq w$.  Fix a strictly dominant weight $\lambda\in\mathbf{P}^+$.
  By Proposition \ref{prop:extremal_products},
  \[
    z_n := b_w(\lambda)^\infty b_v(\lambda)^n
  \]
  is a semi-infinite Cartan word for every $n$.  These $z_n$ belong to the orbit $[b_w(\infty)]$ and converge to $b_v(\infty)$.

  Conversely, suppose that $[b_v(\infty)] \in \overline{[b_w(\infty)]}$, so there is a sequence $(z_n)$ in $[b_w(\infty)]$ which converges to an element $z$ of $[b_v(\infty)]$.  By applying the shift $\mathsf{S}_\mu$ to this sequence, for $\mu\in\mathbf{P}^+$ sufficiently large, we may assume that $z=b_v(\infty)$, see Remark~\ref{rmk:shift-invariant}.  Then for $n$ sufficiently large, the expression of $z_n$ as a semi-infinite Cartan word in $\mathcal{B}(\lambda)^\infty$ has the form
  \[
    z_n = b_w(\lambda)^\infty x b_v(\lambda)
  \]
  for some finite Cartan word $x\in\mathcal{B}(\lambda)^k$. 
  By Proposition \ref{prop:Demazure}, the image of $x\otimes b_v(\lambda)$ in $\mathcal{B}((k+1)\lambda)$ lives in the Demazure crystal $\mathcal{B}_w((k+1)\lambda)$, and from the definition of the Demazure crystal it follows that $b_v(\lambda)\in\mathcal{B}_w(\lambda)$.  Proposition \ref{prop:extremal_in_Demazure} then gives $v\leq w$.

<!-- proposition-7: body_md -->
\label{prop:infinite_shift}
  Fix $I\subseteq\mathbf{\Delta}$. For any semi-infinite crystal $x\in\overleftarrow{\mathcal{B}}\!(\infty)$, the net of shifts
  \[
    (\mathsf{S}_\lambda(x))_{\lambda\in\mathbf{P}^+_I},
  \]
  is eventually constant, in the sense that there exists $\lambda_0\in\mathbf{P}^+_I$ such that
  \[
  \mathsf{S}_\lambda(x) = \mathsf{S}_{\lambda_0}(x) \qquad \text{for all }\lambda\in\mathbf{P}^+_I,\  \lambda\geq\lambda_0.
  \]
<!-- end proposition-7 -->
<!-- proof-13: body_md -->
Thanks to Theorem~\ref{thm:orbits} and Remark \ref{rmk:shift-invariant}, we can write
\[
  x= b_w(\infty) x_0
\]
for some $x_0\in\mathcal{B}(\mu)$ and some $w\in W$.   Then for every $\lambda\in\mathbf{P}^+_I$, writing
\[
  x_\lambda := \mathsf{R}_\mu\circ\mathsf{S}_\lambda(x) = \mathsf{S}_\lambda(b_w(\lambda)x_0) \in \mathcal{B}(\mu),
\]
we have
\[
  \mathsf{S}_\lambda(x) = b_w(\infty) x_\lambda.
\]
For $\nu\in\mathbf{P}^+_I$, the elements $x_\lambda$ and $x_{\lambda+\nu}$ are related by
\(
  x_{\lambda+\nu} = \mathsf{S}_\nu(b_w(\nu)x_\lambda)
\),
which is equivalent to the Cartan braiding relation
\[
  \sigma (b_w(\nu) \otimes x_\lambda) = x_{\lambda+\nu} \otimes b'
\]
for some $b'\in\mathcal{B}(\nu)$.  By the ordering property of the Cartan braiding, see Lemma~\ref{lem:braiding_order}, the net $(x_\lambda)_{\lambda\in\mathbf{P}^+_I}$ is decreasing, in the sense that
\[
  \operatorname{wt}(x_{\lambda+\nu}) \preceq \operatorname{wt}(x_\lambda), \qquad \forall \nu\in\mathbf{P}^+_I.
\]
Moreover, whenever these weights are equal we have $x_{\lambda+\nu}=x_{\lambda}$.  It follows that $(x_\lambda)_{\lambda\in\mathbf{P}^+_I}$ is eventually constant, and so there is $\lambda_0\in\mathbf{P}^+_I$ such that
\[
  \mathsf{S}_{\lambda}(x) = b_w(\infty) x_{\lambda_0}, \qquad
  \text{for all }\lambda\in\mathbf{P}^+_I,\ \lambda\geq\lambda_0.
  \qedhere
\]

<!-- proposition-8: body_md -->
\label{prop:SxR}
Let $I\subseteq\mathbf{\Delta}$.
\begin{itemize}
  \item[(i)]
The map
\begin{align}
  \label{eq:SxR}
  (\mathsf{S}_I,\mathsf{R}_I) : \overleftarrow{\mathcal{B}}\!(\infty) & \to \overleftarrow{\mathcal{B}}\!(\infty) \times \overleftarrow{\mathcal{B}}\!_I(\infty); \\
  x &\mapsto (\mathsf{S}_{I}(x), \mathsf{R}_{I}(x))
  \nonumber
\end{align}
is injective.

\item[(ii)]
If a pair $(z,y) \in \overleftarrow{\mathcal{B}}\!(\infty)\times\overleftarrow{\mathcal{B}}\!_I(\infty)$ belongs to the image of the decomposition map $(\mathsf{S}_I,\mathsf{R}_I)$ then so does $(z,y')$, for any $y'\in\overleftarrow{\mathcal{B}}\!_I(\infty)$ which is tail equivalent to $y$, meaning that $\mathsf{S}_\mu(y) = \mathsf{S}_\nu(y')$ for some $\mu,\nu\in\mathbf{P}^+_I$.
\end{itemize}
<!-- end proposition-8 -->
<!-- proof-14: body_md -->
\begin{itemize}
    \item[\emph{(i)}]
  Suppose $x,x'\in\overleftarrow{\mathcal{B}}\!(\infty)$ have the same image under $(\mathsf{S}_I,\mathsf{R}_I)$.  Put
  \[
    z=\mathsf{S}_{I}(x) = \mathsf{S}_{I}(x').
  \]
  Fixing $\lambda_0\in\mathbf{P}^+_I$ sufficiently large as in Proposition~\ref{prop:infinite_shift}, we get
  \[
    x= zy_0, \qquad x' =  zy'_0
  \]
  for some $y_0,y'_0 \in \mathcal{B}(\lambda_0)$.  Using $\mathsf{R}_{I}(x)=\mathsf{R}_{I}(x')$, we get $y_0=y'_0$.

  \item[\emph{(ii)}]
  Let $(z,y)$, $(z,y')$ and $\mu,\nu$ be as in the statement, and suppose that there exists $x\in\overleftarrow{\mathcal{B}}\!(\infty)$ such that
  \[
    (\mathsf{S}_I(x),\mathsf{R}_I(x)) = (z,y).
  \]
  Fix $\lambda\in\mathbf{P}^+$. Decompose $\mathsf{R}_{\lambda+\nu+\mu}(x)$ as a Cartan word
  \begin{equation}
    \label{eq:SxR1}
    \mathsf{R}_{\lambda+\nu+\mu}(x) = x_2 x_1 x_0 \quad\in \mathcal{B}(\lambda)\times\mathcal{B}(\nu)\times\mathcal{B}(\mu).
  \end{equation}
  Then
  \[
    \mathsf{R}_{\nu+\mu}(y) = \mathsf{R}_{\nu+\mu}(x) = x_1x_0,
  \]
  and since $\mathsf{S}_\mu(y) = \mathsf{S}_\nu(y')$ we get $\mathsf{R}_{\nu}(\mathsf{S}_\nu(y')) = \mathsf{R}_{\nu}(\mathsf{S}_\mu(y))= x_1$ and hence
  \begin{equation}
    \label{eq:SxR2}
    \mathsf{R}_{2\nu}(y') = x_1x'_0
  \end{equation}
  for some $x'_0\in\mathcal{B}(\nu)$.  From~\eqref{eq:SxR1} and~\eqref{eq:SxR2} we deduce that both $x_2x_1$ and $x_1x'_0$ are Cartan words.  But then
  \begin{align*}
    \sigma_{\mathcal{B}(\lambda+\nu),\mathcal{B}(\nu)}&((x_2\otimes x_1) \otimes x'_0) \\
    &= (\sigma_{\mathcal{B}(\lambda),\mathcal{B}(\nu)}\otimes\operatorname{id})(\operatorname{id}\otimes\sigma_{\mathcal{B}(\nu),\mathcal{B}(\nu)})(x_2\otimes x_1\otimes x'_0) \\
    &= (\sigma_{\mathcal{B}(\lambda),\mathcal{B}(\nu)}\otimes\operatorname{id})(x_2\otimes x_1\otimes x'_0) \\
    &\neq 0,
  \end{align*}
  and hence $x_2x_1x'_0 = \mathsf{R}_{\lambda+\nu}(\mathsf{S}_{\mu}(x))x'_0$ is a Cartan word, for any $\lambda\in\mathbf{P}^+$.  Therefore, the semi-infinite word
  \[
    x' = \mathsf{S}_{\mu}(x)x'_0
  \]
  is a Cartan word and it satisfies
  \begin{align*}
    \mathsf{S}_I(x') &= \mathsf{S}_I(x) = z, \\
    \mathsf{R}_I(x') &= \mathsf{R}_I(\mathsf{S}_\mu(x))x'_0 = \mathsf{S}_\mu(y)x'_0 = y'.
  \end{align*}
  \end{itemize}
  This completes the proof.

<!-- proof-16: body_md -->
It is immediate that $\mathcal{G}_I(K)$ is a subgroupoid.  To see that it is open, let $(x,\lambda,y)\in\mathcal{G}_I(K)$.  Then

It follows from the definition of $\mathcal{G}_I(K)$ that $\pi_I(\mathcal{G}_I(K))\subseteq\mathcal{G}(Y^I)$, and it is clearly a groupoid morphism.  To show that it is continuous, we will prove that
  \begin{equation}
  \label{eq:pi_I_continuous}
    \pi_I^{-1}(Z_I(a,b)) = Z(a,b),
  \end{equation}
  where $Z_I(a,b)$ is any open set from the neighbourhood basis of Equation \eqref{eq:Z_I} for $a\in\mathcal{B}(\mu)$, $b\in\mathcal{B}(\nu)$ with $\mu,\nu\in\mathbf{P}^+_I$.
  Certainly $\pi_I(Z(a,b)) \subseteq Z_I(a,b)$.  Conversely, suppose that $(x,\lambda,y)\in\mathcal{G}_I(K)$ with $\pi_I(x,\lambda,y)\in Z_I(a,b)$.  Then we can write
  \[
    x = x'a, \quad y=y'b
  \]
  for some $x',y'\in\overleftarrow{\mathcal{B}}\!(\infty)$ with
  \[
  \mathsf{R}_I(x') = \mathsf{R}_I(y').
  \]
  But also,
  \[
  \mathsf{S}_I(x') = \mathsf{S}_I(x) = \mathsf{S}_I(y)= \mathsf{S}_I(y'),
  \]
  so by Proposition~\ref{prop:SxR}, we have $x'=y'$.  Therefore $(x,\lambda,y)\in Z(a,b)$, which proves the claim.

Finally, for surjectivity, let $(x',\lambda,y')\in\mathcal{G}(Y^I)$.  By the surjectivity of $\mathsf{R}_I$ in Corollary~\ref{cor:R_I_surjective}, there is $x\in\overleftarrow{\mathcal{B}}\!(\infty)$ with $\mathsf{R}_I(x)=x'$.  Since $x'$ and $y'$ are tail equivalent in $\overleftarrow{\mathcal{B}}\!_I(\infty)$, Proposition~\ref{prop:SxR}\emph{(ii)} implies that there is also $y\in\overleftarrow{\mathcal{B}}\!(\infty)$ such that $\mathsf{R}_I(y)=y'$ and $\mathsf{S}_I(x)=\mathsf{S}_I(y)$.  The latter equation means that $\mathsf{S}_\mu(x) = \mathsf{S}_\nu(y)$ for all $\mu,\nu\in\mathbf{P}^+_I$ sufficiently large, and we may take them so that $\lambda = \mu-\nu$.  It follows that $(x,\lambda,y)\in\mathcal{G}_I(K)$, which completes the proof.

An element $(x,\lambda,y)\in\mathcal{G}(K)$ belongs to $\mathcal{G}_I(K)$ if and only if $\lambda\in\mathbf{P}_I$ and $\mathsf{S}_I(x) = \mathsf{S}_I(y)$.

\label{lem:GIK_decomposition}
  Let $\mathcal{G}_I(K)$ be the open subgroupoid of Lemma~\ref{lem:GIK}, take $(x,\lambda,y) \in \mathcal{G}_I(K)$, and write $\pi_I(x,\lambda,y) = (x',\lambda,y')\in\mathcal{G}(Y^I)$.  If $z'\in\overleftarrow{\mathcal{B}}\!_I(\infty)$ and $\mu,\nu\in\mathbf{P}_I$ are such that
  \[
     (x',\lambda,y') = (x',\mu,z')(z',\nu,y'),
  \]
  then there exists a unique preimage $z\in\mathsf{R}_I^{-1}(z')$ such that $(x,\mu,z)$ and $(z,\nu,y)$ are in $\mathcal{G}_I(K)$.
<!-- end lemma-8 -->
<!-- proof-17: body_md -->
By the definition of $\mathcal{G}(Y^I)$, we have
  \[
    \mathsf{S}_{\mu'}(x') = \mathsf{S}_{\mu''}(z') \text{ and }
    \mathsf{S}_{\nu'}(z') = \mathsf{S}_{\nu''}(y'),
  \]
  for some $\mu',\mu'',\nu',\nu''\in\mathbf{P}^+_I$ with $\mu=\mu'-\mu''$ and $\nu=\nu'-\nu''$.
  By Proposition~\ref{prop:SxR}\emph{(ii)}, there exists $z\in\overleftarrow{\mathcal{B}}\!(\infty)$ such that
  \[
    (\mathsf{S}_I(z),\mathsf{R}_I(z)) = (\mathsf{S}_I(x), z').
  \]
  By Remark~\ref{rmk:GIK_characterization}, $(x,\mu,z)$ and $(z,\nu,y)$ are in $\mathcal{G}_I(K)$.  This proves existence.  Uniqueness follows from the injectivity of $(\mathsf{R}_I,\mathsf{S}_I)$, see Proposition~\ref{prop:SxR}\emph{(i)}.

<!-- lemma-9: body_md -->
\label{lem:pi_I_star}
Fix $I\subseteq\mathbf{\Delta}$.  The map
\begin{equation}
  \label{eq:pi_I_star}
  C_\mathrm{c}^\infty(\mathcal{G}(Y^I))
   \stackrel{\pi_I^*}{\longrightarrow} C_\mathrm{c}^\infty(\mathcal{G}_I(K)) \hookrightarrow C_\mathrm{c}^\infty(\mathcal{G}(K)),
\end{equation}
where the first arrow is the pullback by $\pi_I$ from Equation~\eqref{eq:pi_I} and the second map is extension by zero, is an injective graded $*$-algebra morphism.  Its image is isomorphic to $\mathcal{O}(Y^I_0)$ under the isomorphism of Lemma~\ref{lem:flag_subalgebra}.
<!-- end lemma-9 -->
<!-- proof-18: body_md -->
Extension by zero from the open subgroupoid $\mathcal{G}_I(K)$ to $\mathcal{G}(K)$ is certainly a graded $*$-morphism, so it suffices to consider the map $\pi_I^*$.  To find the image of $\pi_I^*$, first recall that $C_\mathrm{c}^\infty(\mathcal{G}(Y^I))$ is spanned by the indicator functions $\mathbf{1}_{Z_I(a,b)}$, where $Z_I(a,b)$ are the clopen subsets of~\eqref{eq:Z_I}, for $a\in\mathcal{B}(\mu)$, $b\in\mathcal{B}(\nu)$ with $\mu,\nu\in\mathbf{P}^+_I$.
  But $\pi_I^*\mathbf{1}_{Z_I(a,b)} = \mathbf{1}_{Z(a,b)}$ thanks to Equation \eqref{eq:pi_I_continuous}.  This confirms that its image is contained in $C_\mathrm{c}^\infty(\mathcal{G}_I(K))$ and under the isomorphism of Theorem \ref{thm:isomorphism} it corresponds to $\mathcal{O}(Y^I_0)$ by Lemma \ref{lem:flag_subalgebra}.
  
  The fact that $\pi_I^*$ is $*$-preserving and graded is straightforward from the definitions.  It is an algebra morphism by the unique lifting property of Lemma~\ref{lem:GIK_decomposition}, and it is injective by the surjectivity of $\pi_I$ from Lemma~\ref{lem:GIK}.
  This completes the proof.

<!-- proof-19: body_md -->
The map described in Lemma~\ref{lem:pi_I_star} is a graded $*$-isomorphism
  \begin{equation*}
    \mathcal{O}(Y^I_0) \cong C_\mathrm{c}^\infty(\mathcal{G}(Y^I)),
  \end{equation*}
  where $\mathcal{O}(Y^I_0)$ has the opposite $\mathbf{P}$-grading.  This is Theorem~\ref{thm:isomorphism2} in the case where $L=\mathbf{P}_I$.  Since it preserves the $\mathbf{P}$-grading, it restricts to an isomorphism between the subalgebras
  \[
    \mathcal{O}(X^{I,L}_0) \cong C_\mathrm{c}^\infty(\mathcal{G}(X^{I,L})),
  \]
  for every $L\leq\mathbf{P}_I$.  The isomorphism of the $C^*$-completions can be proven similarly; we leave the details to the reader.
