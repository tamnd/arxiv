<!-- definition-1: body_md -->
\label{def:concrete-hilbert}
Let $\Omega\subset P_{\mathrm{ext}}\times X_{\mathrm{ext}}$ be compact and
let $\mu$ be a finite positive Borel measure with full support on
$\Omega$ (i.e., $\operatorname{supp}\mu=\Omega$). Set $\mathcal H=L^2(\Omega,\mu)$,
with inner product and norm
\[
\langle f,g\rangle_{\mathcal H}
\coloneqq\int_\Omega fg\,\mathrm d\mu,
\qquad
\|f\|_{\mathcal H}\coloneqq\langle f,f\rangle_{\mathcal H}^{1/2}.
\]

<!-- definition-2: body_md -->
\label{def:QR}
Let $\mathscr H$ be a real Hilbert space. Given
$\mathbf F=(f_1,\dots,f_q)^\top\in\mathscr H^q$ and
$\mathbf G=(g_1,\dots,g_m)^\top\in\mathscr H^m$, set
$V\coloneqq\operatorname{span}\{f_1,\dots,f_q\}\subset\mathscr H$.
Define the Gram matrix and cross-Gram matrix by
\begin{equation}
\begin{aligned}
R(\mathbf F)
\coloneqq
\bigl(\langle f_i,f_j\rangle_{\mathscr H}\bigr)_{i,j=1}^q
\in\mathbb{R}^{q\times q},
\label{eq:R-general}
\end{aligned}
\end{equation}
and
\begin{equation}
\begin{aligned}
Q(\mathbf G,\mathbf F)
\coloneqq
\bigl(\langle g_i,f_j\rangle_{\mathscr H}\bigr)_{\substack{i=1,\dots,m\\ j=1,\dots,q}}
\in\mathbb{R}^{m\times q}.
\label{eq:Q-general}
\end{aligned}
\end{equation}

<!-- lemma-1: body_md -->
\label{lem:gram-invertible}
\label{thm:general-proj}
Let $\mathscr H$ be a real Hilbert space and
$\mathbf F=(f_1,\dots,f_q)^\top\in\mathscr H^q$.
Then $R(\mathbf F)\succ0$ if and only if $f_1,\dots,f_q$ are linearly
independent in $\mathscr H$. In this case, $V=\operatorname{span}\{f_1,\dots,f_q\}$
is closed and, for every $\mathbf G\in\mathscr H^m$,
\begin{equation}
\Pi_{V^m}\mathbf G
=
Q(\mathbf G,\mathbf F)R(\mathbf F)^{-1}\mathbf F.
\label{eq:general-proj}
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
For $a\in\mathbb{R}^q$, $a^\top R(\mathbf F)a=\|\sum_i a_if_i\|_{\mathscr H}^2$,
which proves the first claim \cite[Theorem~6.5-1]{kreyszig1978introductory}.
Finite-dimensional subspaces are closed. The projection formula follows
componentwise from the Gram normal equations
\cite[Ch.~3, p.~80]{arbogastbona}.

<!-- theorem-1: body_md -->
\label{thm:reduced-family}
Under Assumption~\ref{ass:CCK-membership}, the raw lift family can be
reduced to a nested family that is linearly independent of the
actuator-coordinate subspace. More precisely, there exist
$M\in\mathbb{N}\cup\{0,\infty\}$, a strictly increasing sequence of indices
$\{j_k\}_{k=1}^{M}\subset\mathbb{N}$ (empty if $M=0$), and an
$\mathcal H$-orthonormal family
$\{\widehat\phi_k\}_{k=1}^{M}\subset C(\Omega,\mathbb{R})$ such that, for every
$N$ with $1\le N\le M$,
\begin{equation}
\begin{aligned}
V_N
\coloneqq
\operatorname{span}\{\widehat\phi_1,\dots,\widehat\phi_N\}
=
\operatorname{span}\{\phi_{j_1}^{\mathrm{raw}},\dots,\phi_{j_N}^{\mathrm{raw}}\}
\subset\mathcal H,
\label{eq:VCCK-WN}
\end{aligned}
\end{equation}
and $U_p\cap V_N=\{0\}$.
Consequently,
\begin{equation}
\begin{aligned}
W_N
\coloneqq
U_p\oplus V_N
=
U_p+\operatorname{span}\{\phi_{j_1}^{\mathrm{raw}},\dots,\phi_{j_N}^{\mathrm{raw}}\},
\label{eq:WN-characterization}
\end{aligned}
\end{equation}
and $W_N\subset W_{N+1}$ whenever $1\le N<M$. In particular, each
$\widehat\phi_k$ is a finite real linear combination of raw lifts.

If $M=0$, then all raw lifts belong to $U_p$. If
$1\le M<\infty$, then
\begin{equation}
\begin{aligned}
U_p+\operatorname{span}\{\phi_k^{\mathrm{raw}}:k\ge1\}
=
U_p\oplus V_M.
\label{eq:finite-M}
\end{aligned}
\end{equation}
Hence, if $\bigl(U_p+\operatorname{span}\{\phi_k^{\mathrm{raw}}:k\ge1\}\bigr)/U_p$
is infinite-dimensional, then $M=\infty$.
<!-- end theorem-1 -->
<!-- proof-2: body_md -->
If every raw lift belongs to $U_p$, set $M=0$ and the conclusion is
immediate. Otherwise, let $j_1$ be the smallest index such that
$\phi_{j_1}^{\mathrm{raw}}\notin U_p$ and set
\[
\widehat\phi_1
\coloneqq
\frac{\phi_{j_1}^{\mathrm{raw}}}
{\|\phi_{j_1}^{\mathrm{raw}}\|_{\mathcal H}}.
\]
Then $V_1=\operatorname{span}\{\phi_{j_1}^{\mathrm{raw}}\}$ and
$U_p\cap V_1=\{0\}$.

Suppose inductively that $j_1<\cdots<j_N$ and
$\widehat\phi_1,\dots,\widehat\phi_N$ have been constructed so that
they are orthonormal,
\[
V_N
=
\operatorname{span}\{\phi_{j_1}^{\mathrm{raw}},\dots,\phi_{j_N}^{\mathrm{raw}}\},
\\
U_p\cap V_N=\{0\}.
\]
If every $j>j_N$ satisfies $\phi_j^{\mathrm{raw}}\in U_p\oplus V_N$, the recursion terminates and
we set $M=N$. Otherwise, let $j_{N+1}>j_N$ be the smallest index for
which $\phi_{j_{N+1}}^{\mathrm{raw}}\notin U_p\oplus V_N$,
and define
\[
\widetilde\phi_{N+1}
\coloneqq
\phi_{j_{N+1}}^{\mathrm{raw}}
-
\Pi_{V_N}\phi_{j_{N+1}}^{\mathrm{raw}},
\\
\widehat\phi_{N+1}
\coloneqq
\frac{\widetilde\phi_{N+1}}
{\|\widetilde\phi_{N+1}\|_{\mathcal H}}.
\]
Since $\phi_{j_{N+1}}^{\mathrm{raw}}\notin V_N$, we have $\widetilde\phi_{N+1}\neq0$. Also, $\widetilde\phi_{N+1}\in V_N^\perp$, so the enlarged family is orthonormal. The new lift is continuous, being a finite combination of raw lifts. Moreover, the induction hypothesis
gives
\[
\Pi_{V_N}\phi_{j_{N+1}}^{\mathrm{raw}}
\in
\operatorname{span}\{\phi_{j_1}^{\mathrm{raw}},\dots,\phi_{j_N}^{\mathrm{raw}}\},
\]
and therefore
\[
V_{N+1}
=
\operatorname{span}\{\phi_{j_1}^{\mathrm{raw}},\dots,
       \phi_{j_{N+1}}^{\mathrm{raw}}\}.
\]

It remains to verify directness. If $u+v+a\widehat\phi_{N+1}=0,
u\in U_p,\ v\in V_N$, and $a\neq0$, then
$\widehat\phi_{N+1}\in U_p\oplus V_N$. Since
\[
\phi_{j_{N+1}}^{\mathrm{raw}}
=
\|\widetilde\phi_{N+1}\|_{\mathcal H}\widehat\phi_{N+1}
+
\Pi_{V_N}\phi_{j_{N+1}}^{\mathrm{raw}},
\]
this would imply
$\phi_{j_{N+1}}^{\mathrm{raw}}\in U_p\oplus V_N$, contradicting the
choice of $j_{N+1}$. Hence $a=0$, and then
$u=-v\in U_p\cap V_N=\{0\}$. Thus
$U_p\cap V_{N+1}=\{0\}$. The nesting
$V_N\subset V_{N+1}$ immediately gives $W_N\subset W_{N+1}$.

If the construction never terminates, set $M=\infty$. If it stops
after $M<\infty$ steps, every selected raw lift belongs to $V_M$,
every skipped index belongs to the space available when it was skipped
and hence to $U_p\oplus V_M$, and every subsequent raw lift belongs to
$U_p\oplus V_M$ by the stopping criterion. This proves
\eqref{eq:finite-M}. A finite $M$ makes the corresponding quotient
finite-dimensional, so an infinite-dimensional quotient forces
$M=\infty$.

<!-- lemma-2: body_md -->
\label{lem:lower-block-exactB}
For each $k$ with $1\le k\le M$,
$\widehat\gamma_k\in C(\Omega,\mathbb{R})$ and
\begin{equation}
\begin{aligned}
D_p\widehat\phi_k(p,x)B_p=0
\ \text{for all }(p,x)\in\Omega.
\label{eq:exactB-scalar}
\end{aligned}
\end{equation}
Consequently, for every $1\le N\le M$, $D_p\mathbf\Phi_N(p,x)B_p=0$
on $\Omega$, and $\boldsymbol\gamma_N$ and $\mathbf\Gamma_N$ are continuous.

<!-- proof-3: body_md -->
Fix $k$ and use its defining finite expansion
\[
\widehat\phi_k
=
\sum_{r=1}^{m_k}c_{kr}\phi_{n_{kr}}^{\mathrm{raw}}.
\]
For each raw lift,
\[
\begin{aligned}
D_p\phi_{n_{kr}}^{\mathrm{raw}}(p,x)&=D_u\psi_{n_{kr}}(x,C_pp)C_p,\\
D_x\phi_{n_{kr}}^{\mathrm{raw}}(p,x)&=D_x\psi_{n_{kr}}(x,C_pp).
\end{aligned}
\]
Hence
\[
D_p\phi_{n_{kr}}^{\mathrm{raw}}A_pp
+
D_x\phi_{n_{kr}}^{\mathrm{raw}}f(x,C_pp)
=
\gamma_{n_{kr}}^{\mathrm{raw}},
\]
and therefore, by linearity,
\[
\widehat\gamma_k
=
\sum_{r=1}^{m_k}c_{kr}\gamma_{n_{kr}}^{\mathrm{raw}}
\]
is continuous on $\Omega$ by the stated regularity assumptions. Moreover,
\[
D_p\widehat\phi_k(p,x)B_p
=
\sum_{r=1}^{m_k}
c_{kr}D_u\psi_{n_{kr}}(x,C_pp)C_pB_p
=
0
\]
by \eqref{eq:CpBp}, proving \eqref{eq:exactB-scalar}.
Stacking gives the claimed Jacobian identity and continuity of
$\boldsymbol\gamma_N$ and $\mathbf\Gamma_N$.

\coloneqq
(sI-A_{\mathrm{cl},N})^{-1}B_{\mathrm{cl},N}$.

$\varepsilon>0$,
\[
\left\|
\begin{bmatrix}
T_{zw,N}\\
\sqrt{\varepsilon}\,T_{\chi w,N}
\end{bmatrix}
\right\|_\infty^2
\le
\|T_{zw,N}\|_\infty^2
+
\varepsilon\|T_{\chi w,N}\|_\infty^2.
\]
If $T_{\chi w,N}\ne0$, choose
\[
0<\varepsilon_N
<
\frac{
\gamma_{N,\mathrm a}^2-\|T_{zw,N}\|_\infty^2

\|T_{\chi w,N}\|_\infty^2

\coloneqq
\operatorname{col}\bigl(z_N,\sqrt{\varepsilon_N}\chi_N\bigr)$ has $\mathcal{H}_\infty$ norm strictly below $\gamma_{N,\mathrm a}$.

\[
\|D_{\mathrm{cl},N}\|_{2\to2}
\le
\|T_{zw,N}\|_\infty
<
\gamma_{N,\mathrm a}.
\]
Therefore,
\[
R_{\gamma,N}
\coloneqq
\gamma_{N,\mathrm a}^2I
-
D_{\mathrm{cl},N}^\top D_{\mathrm{cl},N}
\succ0.
\]
Because $A_{\mathrm{cl},N}$ is Hurwitz,
$(A_{\mathrm{cl},N},B_{\mathrm{cl},N})$ is stabilizable; the augmented
output contains $\sqrt{\varepsilon_N}I$, so detectability is
automatic. The continuous-time strict bounded-real lemma in
Riccati form~\cite[Ch.~12]{ZhouDoyle1998} therefore gives a stabilizing solution
$P_N=P_N^\top\succeq0$ of

\begin{aligned}

A_{\mathrm{cl},N}^\top P_N
+
P_NA_{\mathrm{cl},N}
+
C_{\mathrm{cl},N}^\top C_{\mathrm{cl},N}
+
\varepsilon_NI\\
&+

\end{aligned}
\label{eq:analysis-ARE-s6}
\end{equation}

\coloneqq
P_NB_{\mathrm{cl},N}
+
C_{\mathrm{cl},N}^\top D_{\mathrm{cl},N}$.
Equation~\eqref{eq:analysis-ARE-s6} is a single post-synthesis Riccati
equation for the fixed closed loop.

\|C_{\mathrm{cl},N}q\|_2^2
+
\varepsilon_N\|q\|_2^2
+
\left\|
R_{\gamma,N}^{-1/2}
D_{\mathrm{cl},N}^\top C_{\mathrm{cl},N}q
\right\|_2^2.
\]
Thus $q=0$, and hence $P_N\succ0$.

\eqref{eq:lyapunov-function-s6} along
\eqref{eq:closed-state-s6} and completing the square yields
\begin{align}

\|z_N\|_2^2
+
\varepsilon_N\|\chi_N\|_2^2
-
\gamma_{N,\mathrm a}^2\|w_N\|_2^2
\notag\\
&\quad=
-\left\|R_{\gamma,N}^{1/2}
\bigl(w_N-R_{\gamma,N}^{-1}\Theta_N^\top\chi_N\bigr)
\right\|_2^2\le0,
\label{eq:square-completion-s6}
\end{align}
which proves \eqref{eq:dissipation-s6}.

The range of $\widetilde\Pi_{W_N}$ is contained in $W_N$, and

\[

so $\widetilde\Pi_{W_N}$ is bounded.

$m=1$, and the integral cross-Gram formula give

\Pi_{W_N}g
=
\left(
\int_\Omega g Z_N^\top\,d\mu
\right)
R_N^{-1}Z_N
=
\widetilde\Pi_{W_N}g
\]
in $\mathcal H$, proving \eqref{eq:Pi-coincides}. For $w\in W_N$, the standing convention therefore gives

\widetilde\Pi_{W_N}w=w
\quad\text{on }\Omega.
\]
Thus $\widetilde\Pi_{W_N}$ is a projection with range $W_N$.

Equation~\eqref{eq:Pi-coincides} and the standing convention identify
the residual components as
\[
\mathcal R_{W_N}\widehat\gamma_k
=
(I-\widetilde\Pi_{W_N})\widehat\gamma_k
\quad\text{on }\Omega.
\]
The standard projection estimate
\cite[Sec.~4.4, Thm.~5, p.~193]{cheney2001analysis} yields
\[
\begin{aligned}
&\|\mathcal R_{W_N}\widehat\gamma_k\|_{L^\infty(\Omega)}\\
&\quad\le
\|I-\widetilde\Pi_{W_N}\|_{C(\Omega)\to C(\Omega)}
\inf_{w\in W_N}
\|\widehat\gamma_k-w\|_{L^\infty(\Omega)}\\
&\quad\le
(1+\Lambda_N)
\inf_{w\in W_N}
\|\widehat\gamma_k-w\|_{L^\infty(\Omega)}.
\end{aligned}
\]
Since residuals on $W_N^N$ act componentwise,
$(\mathbf r_N)_k(\vartheta)
=
\bigl(\mathcal R_{W_N}\widehat\gamma_k\bigr)(\vartheta)$ where $\vartheta\in\Omega$,
for $k=1,\dots,N$. Hence, for every $\vartheta\in\Omega$,

\begin{aligned}
\|\mathbf r_N(\vartheta)\|_2^2
&=
\sum_{k=1}^N

\bigl(\mathcal R_{W_N}\widehat\gamma_k\bigr)(\vartheta)

\|\mathcal R_{W_N}\widehat\gamma_k\|_{L^\infty(\Omega)}^2\\

\sum_{k=1}^{N}
\left(
\inf_{w\in W_N}
\|\widehat\gamma_k-w\|_{L^\infty(\Omega)}
\right)^2.
\end{aligned}
\]
