<!-- theorem-1: body_md -->
\label{main theorem} 
For each $\ell=1,2$, let
\begin{align*}
\mathcal{A}^{(\ell)}= {\rm diag} \Big(A^{(\ell),1}_0,\,A^{(\ell),2}_0,\,\cdots,\,A^{(\ell),m}_0\Big)~\mbox{and}~
\mathcal{B}^{(\ell)}= {\rm diag} \Big(\overrightarrow{A}^{(\ell),1},\,\overrightarrow{A}^{(\ell),2},\,\cdots,\,\overrightarrow{A}^{(\ell),m}\Big).
\end{align*}
Let $(\mathcal{A}^{(\ell)},\mathcal{B}^{(\ell)},q^{(\ell)})\in \mathcal{T}(M_0,\alpha)$ 
such that
\begin{align}\label{Def: extra condition}
    \lVert (A^{(2),j}_0, \overrightarrow{A}^{{(2),j}})- (A^{(2),k}_0, \overrightarrow{A}^{{(2),k}})\rVert_{L^{\infty}(Q)}\leq \lVert(\mathcal{A}^{(1)},\mathcal{B}^{(1)})-(\mathcal{A}^{(2)},\mathcal{B}^{(2)})\rVert_{\mathbf{L}^{\infty}(Q)},
\end{align}
 for all $1\leq j,k\leq m$ and $\Lambda_\ell$ be the partial input-output operator corresponding to $\mathcal{L}_{\mathcal{A}^{(\ell)},\mathcal{B}^{(\ell)},q^{(\ell)}}$. 
Furthermore,
assume that \[ {{\rm div}}_{(t,x)}(\mathcal{A}^{(1)}, \mathcal{B}^{(1)})= {{\rm div}}_{(t,x)}(\mathcal{A}^{(2)}, \mathcal{B}^{(2)}).\] Then the following estimates	
\begin{align} 
&\|(\mathcal{A}^{(2)},\mathcal{B}^{(2)})-(\mathcal{A}^{(1)},\mathcal{B}^{(1)})\|_{\mathbf{L}^{\infty}(Q)}\le C\left(\|\Lambda_1-\Lambda_2\|^{\mu_1}+\big|\log|\log\|\Lambda_1-\Lambda_2\||\big|^{-\mu_2}\right),\label{main result 1}\\
&\|q^{(2)}-q^{(1)}\|_{\mathbf{L}^{\infty}(Q)}\le C\left(\|\Lambda_1-\Lambda_2\|^{\mu_3}+\Big|\log\big|\log|\log\|\Lambda_1-\Lambda_2\||\big|\Big|^{-\mu_4}\right),\label{main result 2}
\end{align}	
 hold, where constants $C,\mu_1,\mu_2,\mu_3$ and $\mu_4>0$ are depending only on  $Q,m,\,\alpha\,\mbox{and}~M_0$.

<!-- theorem-2: body_md -->
\label{Theorem:well_posedness}
   Let $\overrightarrow{\psi_0}\in \mathbf{H}^1(\Omega),\,\overrightarrow{\psi_1}\in \mathbf{L}^2(\Omega)$ and $\overrightarrow{f}\in \mathbf{H}^1(\Sigma)$ and satisfies the compatibility criteria, $\overrightarrow{\psi_0}|_{\partial\Omega}=\overrightarrow{f}(0,\cdot)|_{\partial\Omega}$. Then the IBVP \eqref{Govern IBVP} has a unique solution $$\overrightarrow{u}\in C^1\big([0,T];\mathbf{L}^2(\Omega)\big)\cap C\big([0,T];\mathbf{H}^1(\Omega)\big)$$ and there exists a constant $C>0$ depending on $\Omega,\,T,\,\mathcal{A},\,\mathcal{B}$ and $q$, such that for any $t\in[0,T]$, the following estimate holds
\begin{align}\label{regularity}
\|\partial_\nu \overrightarrow{u}\|_{\mathbf{L}^2(\Sigma)}+\|\partial_t \overrightarrow{u}(t,\cdot)\|_{\mathbf{L}^2(\Omega)}+\|\overrightarrow{u}(t,\cdot)\|_{\mathbf{H}^1(\Omega)}\leq C\left(\|\overrightarrow{\psi_0}\|_{\mathbf{H}^1(\Omega)}+\|\overrightarrow{\psi_1}\|_{\mathbf{L}^2(\Omega)}+\|\overrightarrow{f}\|_{\mathbf{H}^1(\Sigma)}\right).
\end{align}

<!-- proof-1: body_md -->
To establish the existence and uniqueness of the solution to the initial-boundary value problem (IBVP) \eqref{Govern IBVP}, we first decompose
\[
\overrightarrow{u}(t,x) := \overrightarrow{u}^{(1)}(t,x) + \overrightarrow{u}^{(2)}(t,x).
\]
We then determine the functions $\overrightarrow{u}^{(1)}$ and $\overrightarrow{u}^{(2)}$ as solutions of the following IBVPs.
\begin{equation}\label{gov_eqn_for_u1}
\begin{cases}
  \square \overrightarrow{u}^{(1)} + \mathcal{A}(t,x) \,  \partial_t\overrightarrow{u}^{(1)} -\mathcal{B} (t,x)\cdot \nabla_x\overrightarrow{u}^{(1)}+\widetilde{\mathcal{A}}(t,x)\overrightarrow{u}^{(1)}=\overrightarrow{0},&(t,x)\in Q,\\
  \overrightarrow {u}^{(1)}(0,x)=\overrightarrow {\psi_0}(x),~~ \partial_t \overrightarrow {u}^{(1)}(0,x) = \overrightarrow {\psi_1}(x), &\quad x\in\Omega,\\
\overrightarrow {u}^{(1)}(t,x) = \overrightarrow {f}(t,x), & (t,x)\in  \Sigma,
\end{cases}
\end{equation}
and 
\begin{equation}\label{gov_eqn_for_u2}
\begin{cases}
  \mathcal{L}_{\mathcal{A},\mathcal{B},q}\overrightarrow{u}^{(2)}=-q(t,x)\overrightarrow{u}^{(1)},&(t,x)\in Q,\\
  \overrightarrow {u}^{(2)}(0,x)=\overrightarrow {0},~~ \partial_t \overrightarrow {u}^{(2)}(0,x) = \overrightarrow {0}, &\quad x\in\Omega,\\
\overrightarrow {u}^{(2)}(t,x) = \overrightarrow {0}, & (t,x)\in  \Sigma,
\end{cases}
\end{equation}
respectively. The existence and uniqueness of the solution $\overrightarrow{u}^{(1)}$ for the hyperbolic IBVP \eqref{gov_eqn_for_u1} can be guaranteed by \cite{kachalov2001inverse}. In addition, we have
\begin{align}\label{Existence: u_1}
\overrightarrow{u}^{(1)}\in C^1\big([0,T];\mathbf{L}^2(\Omega)\big)\cap C\big([0,T];\mathbf{H}^1(\Omega)\big),\quad \partial_\nu \overrightarrow{u}^{(1)}\in \mathbf{L}^2(\Sigma)\end{align} and there exists a constant $C=C(\mathcal{A},\mathcal{B},\Omega,T)>0$ such that
\begin{align}\label{energy_estimate_u1}
\|\partial_\nu \overrightarrow{u}^{(1)}\|_{\mathbf{L}^2(\Sigma)}+\|\overrightarrow{u}^{(1)}\|_{\mathbf{H}^1(Q)}\leq C\left(\|\overrightarrow{\psi_0}\|_{\mathbf{H}^1(\Omega)}+\|\overrightarrow{\psi_1}\|_{\mathbf{L}^2(\Omega)}+\|\overrightarrow{f}\|_{\mathbf{H}^1(\Sigma)}\right).\end{align}
Furthermore, for the existence and uniqueness of $\overrightarrow{u}^{(2)}$, we now introduce the sesquilinear form
$B(t;\cdot,\cdot)$ associated to IBVP
\eqref{gov_eqn_for_u2}.  Define
$B(t;\cdot,\cdot):\mathbf{H}_0^1(\Omega)\times \mathbf{H}_0^1(\Omega)\rightarrow \mathbb{R}$ by
\begin{align}\label{bilinear_map}
B[t;\overrightarrow{g},\overrightarrow{h}]:=\int_{\Omega}\left[\nabla_x\overrightarrow{g}\cdot\overline{\nabla_x\overrightarrow{h}}-\big(\mathcal{B}(t,x)\cdot\nabla_x\overrightarrow{g}\big)\cdot\overline{\overrightarrow{h}}+\big(\widetilde{\mathcal{A}}(t,x)+q(t,x)\big)\overrightarrow{g}\cdot\overline{\overrightarrow{h}}\right]dx.
\end{align}
Using the Cauchy-Schwartz inequality, one can observe that the sesquilinear form $B[t;\cdot,\cdot]$ is continuous on $\mathbf{H}_0^1(\Omega)\times \mathbf{H}_0^1(\Omega)$. More precisely, we have 
\begin{align}\label{Bilinear_is_bdd}
|B[t;\overrightarrow{g},\overrightarrow{h}]|\leq C\|\overrightarrow{g}\|^2_{\mathbf{H}_0^1(\Omega)}\|\overrightarrow{h}\|^2_{\mathbf{H}_0^1(\Omega)},
\end{align}
where constant $C=C(\mathcal{A},\,\mathcal{B},\,q,\,\Omega,\,T)>0$.
Also, we have
\begin{align*}
\Re B[t;\overrightarrow{g},\overrightarrow{g}]\geq\|\nabla_x\overrightarrow{g}\|^2_{\mathbf{L}^2(\Omega)}-\Big(\|\widetilde{\mathcal{A}}\|_{\mathbf L^\infty(Q)}+\| q\|_{\mathbf L^\infty(Q)}\Big)\|\overrightarrow{g}\|^2_{\mathbf{L}^2(\Omega)},
\end{align*}
where $\Re$ denotes the real part of the underlying function.
Choosing $\lambda>\|\widetilde{\mathcal{A}}\|_{\mathbf L^\infty(Q)}+\|q\|_{\mathbf L^\infty(Q)}$ in the above equation, we get 
\begin{align}\label{bilinear_coercivity}
\Re B[t;\overrightarrow{g},\overrightarrow{g}]+\lambda\|\overrightarrow{g}\|^2_{\mathbf{L}^2(\Omega)}\geq\alpha\|\overrightarrow{g}\|^2_{\mathbf{H}^1(\Omega)}\quad \text{for some constant}\,\alpha>0.
\end{align} 
Therefore, using the Theorem $8.1$ together with Remark $8.1$ of Chapter $3$  in \cite{lions2012non} (see also \cite{lions2013equations}), we observe that the IBVP \eqref{gov_eqn_for_u2} admits a unique solution $\overrightarrow{u}^{(2)}\in C^1\big([0,T];\mathbf{L}^2(\Omega)\big)\cap C\big([0,T];\mathbf{H}^1(\Omega)\big)$ and it satisfies the following estimate 
\begin{align}\label{H1_estimate_for_u2}
\lVert\overrightarrow{u}^{(2)}\rVert_{\mathbf{H}^1(Q)}\leq C \left( \lVert\overrightarrow{\psi_0}\rVert_{\mathbf{H}^1(\Omega)}+ \lVert\overrightarrow{\psi_1}\rVert_{\mathbf{L}^2(\Omega)}+\lVert\overrightarrow {f}\rVert_{\mathbf{H}^1(\Sigma)}\right).
\end{align}
To complete the proof of Theorem \ref{Theorem:well_posedness}, it remains to verify that $\partial_\nu\overrightarrow{u}^{(2)}\in \mathbf{L}^2(\Sigma)$. To this end, we follow the approach proposed in \cite{mishra2021determining}. Fix an arbitrary point $x\in\partial\Omega$ and denote by $\nu(x)$ the outward unit normal vector to $\partial\Omega$ at $x$. We then extend $\nu$ to a (sufficiently smooth) vector field on $\overline{\Omega}$, and denote this extension by $\vartheta(x)$. Next, we multiply the left-hand side of IBVP
\eqref{gov_eqn_for_u2} with $(T-t)\vartheta(x)\cdot\nabla_{x}\overrightarrow{u}^{(2)}$ and integrate it over $Q$, we have
\begin{align*}
\int\limits_{Q}\left( T-t\right)  \mathcal{L}_{\mathcal{A},\mathcal{B},q}&\overrightarrow{u}^{(2)}(t,x) \cdot \left( \vartheta(x)\cdot\nabla_{x}\overrightarrow{u}^{(2)}(t,x)\right)\mathrm{d} x \mathrm{d} t\\
=&\int\limits_{Q}\left( T
-t\right)\partial_{t}^{2}\overrightarrow{u}^{(2)}(t,x)\cdot \left( \vartheta(x)\cdot\nabla_{x}\overrightarrow{u}^{(2)}(t,x)\right)\mathrm{d} x \mathrm{d} t \\
&\quad -\int\limits_{Q}\left( T-t\right)\Delta_{x}\overrightarrow{u}^{(2)}(t,x)\cdot \left( \vartheta(x)\cdot\nabla_{x}\overrightarrow{u}^{(2)}(t,x)\right)\mathrm{d} x \mathrm{d} t\\
&\quad +\int\limits_{Q}\left( T-t\right) \mathcal{A}(t,x)\partial_t\overrightarrow{u}^{(2)}(t,x)\cdot \left( \vartheta(x)\cdot\nabla_{x}\overrightarrow{u}^{(2)}(t,x)\right)\mathrm{d} x \mathrm{d} t\\
&\quad -\int\limits_{Q}\left( T-t\right) \left(\mathcal{B}(t,x)\cdot\nabla_x\overrightarrow{u}^{(2)}(t,x)\right)\cdot \left( \vartheta(x)\cdot\nabla_{x}\overrightarrow{u}^{(2)}(t,x)\right)\mathrm{d} x \mathrm{d} t\\
&\quad +\int\limits_{Q}\left( T-t\right) \left( \widetilde{\mathcal{A}}+q(t,x)\right)\overrightarrow{u}^{(2)}(t,x)\cdot \left( \vartheta(x)\cdot\nabla_{x}\overrightarrow{u}^{(2)}(t,x)\right)\mathrm{d} x \mathrm{d} t\\
=& \sum_{j=1}^{m}\int\limits_{Q}\left( T-t\right)\partial_{t}^{2}u^{(2)}_j(t,x)\left( \vartheta(x)\cdot\nabla_{x}u^{(2)}_j(t,x)\right)\mathrm{d} x\mathrm{d} t\\
&\quad -\sum_{j=1}^{m}\int\limits_{Q}\left( T-t\right)\Delta_{x}u^{(2)}_j(t,x)\left( \vartheta(x)\cdot\nabla_{x}u^{(2)}_j(t,x)\right)\mathrm{d} x\mathrm{d} t\\
&\quad +\sum_{j=1}^{m}\int\limits_{Q}\left( T-t\right) A^j_0(t,x)\partial_t u^{(2)}_j(t,x)\left( \vartheta(x)\cdot\nabla_{x}u^{(2)}_j(t,x)\right)\mathrm{d} x\mathrm{d} t\\
&\quad -\sum_{j=1}^{m}\int\limits_{Q}\left( T-t\right)\left( \overrightarrow{A^{j}}(t,x)\cdot\nabla_{x}u^{(2)}_j(t,x)\right)\left( \vartheta(x)\cdot\nabla_{x}u^{(2)}_j(t,x)\right)\mathrm{d} x\mathrm{d} t\\
&\quad +\sum_{j=1}^{m}\int\limits_{Q}\left( T-t\right)  \widetilde{A^j}(t,x) u^{(2)}_j(t,x)\left( \vartheta(x)\cdot\nabla_{x}{u^{(2)}_j}(t,x)\right)\mathrm{d} x\mathrm{d} t\\
&\quad +\sum_{i,j=1}^{m}\int\limits_{Q}\left( T-t\right) q_{ij}(t,x)u^{(2)}_j(t,x)\left( \vartheta(x)\cdot\nabla_{x}{u^{(2)}_i}(t,x)\right)\mathrm{d} x\mathrm{d} t\\
:\triangleq& I+II+III+IV+V+VI.
\end{align*}
By multiplying the right-hand side of the IBVP \eqref{gov_eqn_for_u2} by $(T - t)\,\vartheta(x)\cdot\nabla_{x}\overrightarrow{u}^{(2)}$, integrating the resulting expression over $Q$, and subsequently comparing it with the preceding identity, we obtain
\begin{align}\label{integral_identity_1}
I+II+III+IV+V+VI=-\int\limits_{Q}\left( T-t\right) q(t,x)\overrightarrow{u}^{(1)}(t,x)\cdot \left( \vartheta(x)\cdot\nabla_{x}\overrightarrow{u}^{(2)}(t,x)\right)\mathrm{d} x \mathrm{d} t.
\end{align}
Using integration by parts, we observe that $I$ reduces to
\begin{align*}
\begin{aligned}
I&=-T\sum_{j=1}^{m}\int\limits_{\Omega} \partial_{t}u^{(2)}_j(0,x)\left( \vartheta(x)\cdot\nabla_{x}u^{(2)}_j(0,x)\right)\mathrm{d} x+\sum_{j=1}^{m}\int\limits_{Q}\partial_{t}u^{(2)}_j(t,x)\left( \vartheta(x)\cdot\nabla_{x}u^{(2)}_j(t,x)\right) \mathrm{d} x\mathrm{d} t\\
&\qquad \qquad-\sum_{j=1}^{m}\int\limits_{Q}\left( T-t\right)\partial_{t}u^{(2)}_j(t,x)\left( \vartheta(x)\cdot\nabla_{x}\partial_{t}u^{(2)}_j(t,x)\right) \mathrm{d} x\mathrm{d} t\\
&=-T\int\limits_{\Omega} \partial_{t}\overrightarrow{u}^{(2)}(0,x)\cdot\left( \vartheta(x)\cdot\nabla_{x}\overrightarrow{u}^{(2)}(0,x)\right)\mathrm{d} x+\int\limits_{Q}\partial_{t}\overrightarrow{u}^{(2)}(t,x)\cdot\left( \vartheta(x)\cdot\nabla_{x}\overrightarrow{u}^{(2)}(t,x)\right)\mathrm{d} x\mathrm{d} t\\
&\qquad \qquad  -\int\limits_{Q}\frac{(T-t)}{2} \nabla_{x}\cdot\left( \vartheta(x)\lvert\partial_{t}\overrightarrow{u}^{(2)}(t,x)\rvert^{2}\right)\mathrm{d} x\mathrm{d} t +\int\limits_{Q}\frac{(T-t)}{2}\lvert\partial_{t}\overrightarrow{u}^{(2)}(t,x)\rvert^{2}\nabla_{x}\cdot\vartheta(x)\mathrm{d} x \mathrm{d} t.
\end{aligned}
\end{align*}
Furthermore, by applying the Gauss divergence theorem along with the initial and boundary conditions prescribed in the IBVP \eqref{gov_eqn_for_u2}, the foregoing expression simplifies to
\begin{align}\label{integral_identity_A1}
\begin{aligned}
I&=\int\limits_{Q}\partial_{t}\overrightarrow{u}^{(2)}(t,x)\cdot\left( \vartheta(x)\cdot\nabla_{x}\overrightarrow{u}^{(2)}(t,x)\right)\mathrm{d} x\mathrm{d} t +\int\limits_{Q}\frac{T-t}{2}\lvert\partial_{t}\overrightarrow{u}^{(2)}(t,x)\rvert^{2}\nabla_{x}\cdot\vartheta(x)\mathrm{d} x \mathrm{d} t.
\end{aligned}
\end{align}
Similarly, using integration by parts, $II$ reduces to
\begin{align*}
II&=-\sum_{j=1}^{m}\int\limits_{Q}\left( T-t\right)\Delta_{x}u^{(2)}_j(t,x)\left( \vartheta(x)\cdot\nabla_{x}u^{(2)}_j(t,x)\right)\mathrm{d} x\mathrm{d} t\\
&=-\sum_{j,k=1}^{m}\int\limits_{Q}(T-t)\sum_{l=1}^{n}\partial_{k}^{2}u^{(2)}_j(t,x)\vartheta_{l}(x)\partial_{l}u^{(2)}_j(t,x)\mathrm{d} x\mathrm{d} t\\
&=-\sum_{j=1}^{m}\int\limits_{Q}(T-t)\nabla_{x}\cdot \left( \nabla_{x}u^{(2)}_j(t,x)\vartheta(x)\cdot\nabla_{x}u^{(2)}_j(t,x)\right)\mathrm{d} x\mathrm{d} t\\
&\qquad-\int\limits_{Q}\frac{T-t}{2}\nabla_{x}\cdot\vartheta(x)\lvert\nabla_{x}\overrightarrow{u}^{(2)}(t,x)\rvert^{2}\mathrm{d} x\mathrm{d} t\\
&\qquad+\sum_{j,k=1}^{m}\int\limits_{Q}(T-t)\sum_{l=1}^{n}\partial_{k}u^{(2)}_j(t,x)\partial_{k}\vartheta_{l}(x)\partial_{l}u^{(2)}_j(t,x)\mathrm{d} x\mathrm{d} t\\
&\qquad+\int\limits_{Q}\frac{T-t}{2}\nabla_{x}\cdot\left( \vartheta(x)\lvert\nabla_{x}\overrightarrow{u}^{(2)}(t,x)\rvert^{2}\right)\mathrm{d} x \mathrm{d} t.\\
\end{align*}
Next, we impose the Gauss divergence theorem in the above expression and use the vanishing boundary condition  to arrive at
\begin{equation}\label{integral_identity_A2}
\begin{aligned}
II&=-\int\limits_{\Sigma}\frac{T-t}{2}\lvert\partial_{\nu}\overrightarrow{u}^{(2)}(t,x)\rvert^{2}\mathrm{d} S_{x}\mathrm{d} t+\sum_{j,k=1}^{m}\int\limits_{Q}(T-t)\sum_{l=1}^{n}\partial_{k}u^{(2)}_j(t,x)\partial_{k}\vartheta_{l}(x)\partial_{l}u^{(2)}_j(t,x)\mathrm{d} x\mathrm{d} t\\
&\qquad \qquad -\int\limits_{Q}\frac{T-t}{2}\nabla_{x}\cdot\vartheta(x)\lvert\nabla_{x}\overrightarrow{u}^{(2)}(t,x)\rvert^{2}\mathrm{d} x\mathrm{d} t.
\end{aligned}
\end{equation}
Taking into account \eqref{integral_identity_A1}, \eqref{integral_identity_A2} in \eqref{integral_identity_1}, we have
\begin{align*}
    -&\int\limits_{\Sigma}\frac{T-t}{2}\lvert\partial_{\nu}\overrightarrow{u}^{(2)}(t,x)\rvert^{2}\mathrm{d} S_{x}\mathrm{d} t \\
    & \quad=-\sum_{j,k=1}^{m}\int\limits_{Q}(T-t)\sum_{l=1}^{n}\partial_{k}u^{(2)}_j(t,x)\partial_{k}\vartheta_{l}(x)\partial_{l}u^{(2)}_j(t,x)\mathrm{d} x\mathrm{d} t
    + \int\limits_{Q}\frac{T-t}{2}\nabla_{x}\cdot\vartheta(x)\lvert\nabla_{x}\overrightarrow{u}^{(2)}(t,x)\rvert^{2}\mathrm{d} x\mathrm{d} t\\
    &\qquad -\int\limits_{Q}\partial_{t}\overrightarrow{u}^{(2)}(t,x)\cdot\left( \vartheta(x)\cdot\nabla_{x}\overrightarrow{u}^{(2)}(t,x)\right)\mathrm{d} x\mathrm{d} t -\int\limits_{Q}\frac{T-t}{2}\lvert\partial_{t}\overrightarrow{u}^{(2)}(t,x)\rvert^{2}\nabla_{x}\cdot\vartheta(x)\mathrm{d} x \mathrm{d} t\\
    &\qquad -\int\limits_{Q}\left( T-t\right) q(t,x)\overrightarrow{u}^{(1)}(t,x)\cdot \left( \vartheta(x)\cdot\nabla_{x}\overrightarrow{u}^{(2)}(t,x)\right)\mathrm{d} x \mathrm{d} t -III-IV-V-VI.
\end{align*}
Afterwards, using the Cauchy-Schwartz inequality in the right-hand side of the above identity, we end up with
\begin{align}\label{L2_norm_normal_derivative}
\begin{aligned}
\Big|\int\limits_{\Sigma}&\frac{T-t}{2}\lvert\partial_{\nu}\overrightarrow{u}^{(2)}(t,x)\rvert^{2}\mathrm{d} S_{x}\mathrm{d} t\Big|\\
&\leq C\int\limits_{Q}\left(\lvert\overrightarrow{u}^{(1)}(t,x)\rvert^{2}+\lvert\overrightarrow{u}^{(2)}(t,x)\rvert^{2}+ \lvert\partial_{t}\overrightarrow{u}^{(2)}(t,x)\rvert^{2}+\lvert\nabla_{x}\overrightarrow{u}^{(2)}(t,x)\rvert^{2}\right)\mathrm{d} x\mathrm{d} t.
\end{aligned}
\end{align}
Combining \eqref{L2_norm_normal_derivative} with \eqref{energy_estimate_u1} and \eqref{H1_estimate_for_u2}, to get
\begin{align*}
 \|\partial_\nu \overrightarrow{u}^{(2)}\|_{\mathbf{L}^2(\Sigma)} \leq C \left( \lVert\overrightarrow{\psi_0}\rVert_{\mathbf{H}^1(\Omega)}+ \lVert\overrightarrow{\psi_1}\rVert_{\mathbf{L}^2(\Omega)}+\lVert\overrightarrow {f}\rVert_{\mathbf{H}^1(\Sigma)}\right).
\end{align*}
As a consequence, we have 
\begin{align}\label{Existence: u_2}
\overrightarrow{u}^{(2)}\in \mathbf{C}^{1}\left( [0,T];\mathbf{L}^{2}(\Omega)\right)\cap \mathbf{C}\left( [0,T];\mathbf{H}^{1}(\Omega)\right)  \mbox{and}\  \partial_{\nu}\overrightarrow{u}^{(2)}\in \mathbf{L}^{2}(\Sigma).
\end{align}
Combining \eqref{Existence: u_1} and \eqref{Existence: u_2}, we conclude that
\begin{align*}
&\overrightarrow{u}\in \mathbf{C}^{1}\left( [0,T];\mathbf{L}^{2}(\Omega)\right)\cap \mathbf{C}\left( [0,T];\mathbf{H}^{1}(\Omega)\right)  \mbox{and}\  \partial_{\nu}\overrightarrow{u}\in \mathbf{L}^{2}(\Sigma).
\end{align*}
Also, we have
\begin{align*}
 \lVert\partial_{\nu}\overrightarrow{u}\rVert_{\mathbf{L}^{2}(\Sigma)}+\lVert\vec{u}\rVert_{\mathbf{H}^{1}(Q)}\leq C \left( \lVert\overrightarrow{\psi_0}\rVert_{\mathbf{H}^1(\Omega)}+ \lVert\overrightarrow{\psi_1}\rVert_{\mathbf{L}^2(\Omega)}+\lVert\overrightarrow {f}\rVert_{\mathbf{H}^1(\Sigma)}\right).
\end{align*}
Hence the result follows.

\label{Theorem: Boundary Carleman estimate}
Let  $(\mathcal{A},\mathcal{B},q)\in \mathcal{T}(M_0,\alpha)$ and $\omega\in \mathbb{S}^{n-1}$. Furthermore, let $\overrightarrow{v}\in \mathbf{C}^{2}(\overline{Q})$ be an arbitrary vector field satisfying
\begin{align*}
\overrightarrow{v}|_{\Sigma}=\overrightarrow{0},\ \overrightarrow{v}|_{t=0}=\partial_{t}\overrightarrow{v}|_{t=0}=\overrightarrow{0}.
\end{align*}
Then there exist constants $\lambda_{0}, C > 0$, both depending only on $m, M_{0}, \alpha \text{ and } Q$, such that for every $\lambda \geq \lambda_{0}$, the following holds:
\begin{align}\label{boundary carleman estimate}
\begin{aligned}
&\int_{Q}e^{-2\lambda(t+x\cdot \omega)}\left(\lambda^2|\overrightarrow{v}(t,x)|^2+|\nabla_{(t,x)}\overrightarrow{v}(t,x)|^2\right) dxdt+\lambda \int\limits_{\Sigma_{+,\omega}}e^{-2\lambda(t+x\cdot \omega)}|\omega\cdot \nu(x)||\partial_{\nu}\overrightarrow{v}|^2dS_xdt\\
&\quad +\lambda\int_{\Omega} e^{-2\lambda(T+x\cdot \omega)}|\partial_t\overrightarrow{v}(T,x)|^2\ dx\le C\int_{\Omega} e^{-2\lambda(T+x\cdot\omega)}\left(\lambda^3|\overrightarrow{v}(T,x)|^2+\lambda|\nabla_x{\overrightarrow{v}(T,x)}|^2\right)dx\\
&\quad+C \bigg(\int_{Q}e^{-2\lambda(t+x\cdot\omega)}|\mathcal{L}_{\mathcal{A},\mathcal{B},q}\overrightarrow{v}(t,x)|^2dxdt+\lambda\int\limits_{\Sigma_{-,\omega}}{ e^{-2\lambda(t+x\cdot\omega)}|\omega\cdot \nu(x)||\partial_{\nu}\overrightarrow{v}|^2}~dS_xdt \bigg).
\end{aligned}
\end{align}

For $(t,x)\in Q$, we denote $\varphi(t,x):=t+x\cdot\omega$ and $\varphi_\lambda(t,x):=\lambda\,(t+x\cdot\omega)$, where $\omega\in \mathbb{S}^{n-1}$. To absorb the first-order perturbation, we next define a perturbed weight function depending on an auxiliary parameter $\sigma$, where $\sigma$ is taken to be independent of $\lambda$ and will be fixed at a later stage. Specifically, for parameters satisfying $\lambda>\sigma>1$, we set
\[
 \varphi_{\lambda,\sigma}(t,x):=\varphi_{\lambda}(t,x)-\frac{\sigma t^2}{2}.
\]
Now if we denote $\overrightarrow{u}=e^{-\phi_{\lambda,\sigma}}\overrightarrow{v}$, we then have $\overrightarrow{u}\in \textbf{C}^2(\overline{Q})$. Also, we have
\begin{align*}
\overrightarrow{u}|_{\Sigma}=\overrightarrow{0},\ \overrightarrow{u}|_{t=0}=\partial_{t}\overrightarrow{u}|_{t=0}=\overrightarrow{0}.
\end{align*}
Next, we define the conjugated operator $\mathcal{L}_{\varphi_{\lambda,\sigma}}$ of $\mathcal{L}_{\mathcal{A},\mathcal{B},q}$ by \begin{align}
\mathcal{L}_{\varphi_{\lambda,\sigma}}\overrightarrow{u}:&=e^{-\varphi_{\lambda,\sigma}}\mathcal{L}_{\mathcal{A},\mathcal{B},q}e^{\varphi_{\lambda,\sigma}}\overrightarrow{u}\\&=\square \overrightarrow{u}-\sigma \overrightarrow{u}+(\lambda-\sigma t)^2\overrightarrow{u}-\lambda^2 \overrightarrow{u}\\&\qquad+2(\lambda -\sigma t)\partial_t\overrightarrow{u}-2\lambda \omega\cdot \nabla_x\overrightarrow{u}\\&\qquad+\mathcal{A}\partial_t\overrightarrow{u}+(\lambda-\sigma t)\mathcal{A}\overrightarrow{u}-\mathcal{B}\cdot \nabla_x \overrightarrow{u}-\lambda(\omega\cdot \mathcal{B})\overrightarrow{u}+(\widetilde{\mathcal{A}}+q)\overrightarrow{u}\\:& \triangleq I_1 \overrightarrow{u}+I_2 \overrightarrow{u}+I_3\overrightarrow{u}.
\end{align}
Now, using the triangle inequality, we observe that
\begin{align}\label{Def:Norm conjugated operator}
\begin{aligned}
\lVert\mathcal{L}_{\phi_{\lambda,\sigma}}\overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)}&=\int_{Q} \lvert(I_1+I_2+I_3)\overrightarrow{u}\rvert^2~dxdt \\&\geq \dfrac{1}{2}\lVert(I_1+I_2) \overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)}- \lVert I_3 \overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)}\\&\geq \int_{Q} \Re\left(\left(I_1 \overrightarrow{u}\right)\cdot \overline{\left(
I_2 \overrightarrow{u}\right)}\right)~dxdt- \lVert I_3 \overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)}.
\end{aligned}
\end{align}
We then further simplify the terms in the above expression to obtain a suitable lower bound for $\lVert\mathcal{L}_{\phi_{\lambda,\sigma}}\overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)}$. To this end, we first simplify the first term on the right-hand side of the inequality above. Taking into account  $\overrightarrow{u}|_{\Sigma}=\overrightarrow{0},\ \overrightarrow{u}|_{t=0}=\partial_{t}\overrightarrow{u}|_{t=0}=\overrightarrow{0}$ and the divergence theorem, we achieve
\begin{align}
    J_1:&=\int_{Q} \Re\left(\left(I_1 \overrightarrow{u}\right)\cdot \overline{\left(
I_2 \overrightarrow{u}\right)}\right)~dxdt\\&=
 \int_Q (\lambda-\sigma t)\partial_t\left( \lvert \partial_t \overrightarrow{u}\rvert^2+\lvert \nabla_x \overrightarrow{u}\rvert^2
 \right)~dxdt-2\lambda \Re \int_{\Omega}\partial_t\overrightarrow{u}(T,x)\cdot (\overline{\omega\cdot\nabla_x\overrightarrow{u}(T,x)})~dx\\&\qquad +\lambda \int_{Q}\omega\cdot \nabla_x(\lvert \partial_t \overrightarrow{u}\rvert^2)~dxdt+\lambda \int_{\Sigma} (\omega\cdot \nu)\lvert \partial_\nu \overrightarrow{u}\rvert^2~dS_x dt
 \\&\qquad +\int_{Q} \left(-\lambda \sigma +\sigma^2 t-\sigma^3t^3+3\lambda \sigma^2t^2-2\lambda^2\sigma t
 \right)\partial_t(\lvert \overrightarrow{u}\rvert^2) ~dxdt.
\end{align}
A further simplification will give us 
\begin{align}
    J_1&=\int_{\Omega}(\lambda-\sigma T)(\lvert \partial_t \overrightarrow{u}(T,x) \rvert^2+\lvert \nabla_x \overrightarrow{u}(T,x)\rvert^2)~dx +\sigma \int_{Q} (\lvert \partial_t \overrightarrow{u}\rvert^2+\lvert \nabla_x \overrightarrow{u}\rvert^2)~dxdt
\\ &\qquad -2\lambda \Re \int_{\Omega}\partial_t\overrightarrow{u}(T,x)\cdot (\overline{\omega\cdot\nabla_x\overrightarrow{u}(T,x)})~dx+\lambda \int_{\Sigma} (\omega\cdot \nu)\lvert \partial_\nu \overrightarrow{u}\rvert^2~dS_x dt\\ &\qquad +\int_{\Omega} (-\sigma^3T^3+3\lambda \sigma^2 T^2 +\sigma^2 T-2\lambda^2\sigma T-\lambda\sigma)\lvert \overrightarrow{u}(T,x)\rvert^2~dx\\ &\qquad -\int_{Q} (-3\sigma^3 t^2 +6\lambda \sigma^2 t+\sigma^2-2\lambda^2 \sigma)\lvert \overrightarrow{u}\rvert^2~dxdt.
\end{align}
Next, we determine suitable lower bounds for the coefficients appearing in the preceding expression, which are then used to derive a lower bound for $J_1$. Observe that
\begin{enumerate}
    \item If $\lambda\geq (\sigma +6\lambda \sigma T)^{1/2}$, then
    \[ 
  - ( -3\sigma^3 t^2 +6\lambda \sigma^2 t+\sigma^2-2\lambda^2 \sigma)\geq \lambda^2 \sigma.
    \]
    \item If $\lambda \geq \left(\sigma^2T^2+\frac{\lambda}{T}\right)^{1/2}$, then 
    \[-\sigma^3T^3+3\lambda \sigma^2 T^2 +\sigma^2 T-2\lambda^2\sigma T-\lambda\sigma\geq -3\lambda^2 \sigma T.\]
    \item If $\lambda \geq 2 \sigma T$, then
    \[\lambda -\sigma T\geq \frac{\lambda}{2}.\]
\end{enumerate}
Thus, we have 
\begin{align}\label{Bound: J_1}
\begin{aligned}
    J_1&\geq \frac{\lambda}{4}\int_{\Omega} \lvert \partial_t \overrightarrow{u}(T,x) \rvert^2~dx -\frac{7\lambda}{2}\int_{\Omega}\lvert \nabla_x \overrightarrow{u}(T,x)\rvert^2~dx+\sigma\int_{Q} (\lvert \partial_t \overrightarrow{u}\rvert^2+\lvert \nabla_x \overrightarrow{u}\rvert^2)~dxdt\\&\qquad+\lambda \int_{\Sigma} (\omega\cdot \nu)\lvert \partial_\nu \overrightarrow{u}\rvert^2~dS_x dt+\lambda^2 \sigma \int_{Q} \lvert \overrightarrow{u}\rvert^2~dxdt-3\lambda^2 \sigma T \int_{\Omega}\lvert \overrightarrow{u}(T,x)\rvert^2~dx
    \end{aligned}
\end{align}
for $\lambda> \max \left\{(\sigma +6\lambda \sigma T)^{1/2}, \left(\sigma^2T^2+\frac{\lambda}{T}\right)^{1/2}, 2 \sigma T\right\}$.
Now, we seek an upper bound for the second term on the right-hand side of \eqref{Def:Norm conjugated operator}. Following Sobolev embedding, we obtain
\begin{align}\label{Bound: I3}
\begin{aligned}
    \lVert I_3 \overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)}:&= \lVert\mathcal{A}\partial_t\overrightarrow{u}+(\lambda-\sigma t)\mathcal{A}\overrightarrow{u}-\mathcal{B}\cdot \nabla_x \overrightarrow{u}-\lambda(\omega\cdot \mathcal{B})\overrightarrow{u}+(\widetilde{\mathcal{A}}+q)\overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)}\\& \leq C\bigg(  M_0^2\lVert \partial_t\overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)}+32\lambda^2M_0^2\lVert \overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)}+32 m^2M_0^2\lVert \overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)}\\&\qquad\quad +8M_0^2\lVert \nabla_x \overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)}+326 M_0^2 \lVert \overrightarrow{u}\rVert^2_{\textbf{L}^2{(Q)}}+384M_0^4\lVert  \overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)}\bigg)\\& \leq C(m,M_0,Q,\alpha)(\lVert \partial_t\overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)}+\lambda^2\lVert \overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)}+\lVert \overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)}+\lVert \nabla_x \overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)})
    \end{aligned}
\end{align}
for $\lambda \geq 2\sigma T.$ We then combine \eqref{Bound: J_1} and \eqref{Bound: I3} to achieve
\begin{align}
    \begin{aligned}
\lVert\mathcal{L}_{\phi_{\lambda,\sigma}}\overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)}&\geq  \frac{\lambda}{4}\int_{\Omega} \lvert \partial_t \overrightarrow{u}(T,x) \rvert^2~dx -\frac{7\lambda}{2}\int_{\Omega}\lvert \nabla_x \overrightarrow{u}(T,x)\rvert^2~dx+\sigma\int_{Q} (\lvert \partial_t \overrightarrow{u}\rvert^2+\lvert \nabla_x \overrightarrow{u}\rvert^2)~dxdt\\&\qquad+\lambda \int_{\Sigma} (\omega\cdot \nu)\lvert \partial_\nu \overrightarrow{u}\rvert^2~dS_x dt+\lambda^2 \sigma \int_{Q} \lvert \overrightarrow{u}\rvert^2~dxdt-3\lambda^2 \sigma T \int_{\Omega}\lvert \overrightarrow{u}(T,x)\rvert^2~dx\\&\qquad-C(m,M_0,Q,\alpha)(\lVert \partial_t\overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)}+\lambda^2\lVert \overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)}+\lVert \overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)}+\lVert \nabla_x \overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)}).
    \end{aligned}
\end{align}
By choosing $\sigma = C(m,M_0,Q,\alpha) + 6$, we consequently obtain the following bound:
\begin{align}
    \begin{aligned}
\lVert\mathcal{L}_{\phi_{\lambda,\sigma}}\overrightarrow{u}\rVert^2_{\textbf{L}^2(Q)}&\geq  \frac{\lambda}{4}\int_{\Omega} \lvert \partial_t \overrightarrow{u}(T,x) \rvert^2~dx -7\lambda\int_{\Omega}\lvert \nabla_x \overrightarrow{u}(T,x)\rvert^2~dx+2\int_{Q} (\lvert \partial_t \overrightarrow{u}\rvert^2+\lvert \nabla_x \overrightarrow{u}\rvert^2)~dxdt\\&\qquad+\lambda \int_{\Sigma} (\omega\cdot \nu)\lvert \partial_\nu \overrightarrow{u}\rvert^2~dS_x dt+5\lambda^2  \int_{Q} \lvert \overrightarrow{u}\rvert^2~dxdt-6\lambda^3 \sigma T \int_{\Omega}\lvert \overrightarrow{u}(T,x)\rvert^2~dx.
    \end{aligned}
\end{align}
Subsequently, we substitute $\overrightarrow{u}$ by $e^{-\varphi_{\lambda,\sigma}}\overrightarrow{v}$ in the preceding expression and simplify to obtain
\begin{align}
    \begin{aligned}
       e^{\sigma T^2} \lVert e^{-\varphi_{\lambda}}\mathcal{L}_{\mathcal{A},\mathcal{B},q}\overrightarrow{v}\rVert^2_{\textbf{L}^2(Q)}&\geq \lambda \int_{\Sigma_+} e^{-2\varphi_\lambda}(\omega\cdot \nu)\lvert \partial_\nu \overrightarrow{v}\rvert^2~dS_x dt-\lambda e^{\sigma T^2}\int_{\Sigma_-} e^{-2\varphi_\lambda}(\omega\cdot \nu)\lvert \partial_\nu \overrightarrow{v}\rvert^2~dS_x dt\\&\quad+\frac{\lambda}{8}\int_{\Omega}e^{-2\varphi_{\lambda}}\lvert \partial_t \overrightarrow{v}(T,x)\rvert^2~dx-(6\sigma T+14 e^{\sigma T^2} +1)\lambda^3 \int_\Omega e^{-2\varphi_{\lambda}}\lvert \overrightarrow{v}(T,x)\rvert^2~dx\\&\quad+\int_{Q}e^{-2\varphi_{\lambda}} (\lvert \partial_t \overrightarrow{v}\rvert^2+\lvert \nabla_x \overrightarrow{v}\rvert^2)~dxdt+\lambda^2\int_{Q}e^{-2\varphi_{\lambda}} \lvert \overrightarrow{v}\rvert^2~dxdt\\&\quad-14\lambda e^{\sigma T^2}\int_\Omega e^{-2\varphi_{\lambda}}\lvert \nabla_x\cdot\overrightarrow{v}(T,x)\rvert^2~dx.
    \end{aligned}
\end{align}
Upon further simplification, and invoking the uniform bound $e^{\sigma T^{2}}$, the above expression reduces to
\begin{align}
\begin{aligned}
&\int_{Q}e^{-2\lambda(t+x\cdot \omega)}\left(\lambda^2|\overrightarrow{v}(t,x)|^2+|\nabla_{(t,x)}\overrightarrow{v}(t,x)|^2\right) dxdt+\lambda \int\limits_{\Sigma_{+,\omega}}e^{-2\lambda(t+x\cdot \omega)}|\omega\cdot \nu(x)||\partial_{\nu}\overrightarrow{v}|^2dSdt\\
&\quad +\lambda\int_{\Omega} e^{-2\lambda(T+x\cdot \omega)}|\partial_t\overrightarrow{v}(T,x)|^2\ dx\le C\int_{\Omega} e^{-2\lambda(T+x\cdot\omega)}\left(\lambda^3|\overrightarrow{v}(T,x)|^2+\lambda|\nabla_x{\overrightarrow{v}(T,x)}|^2\right)dx\\
&\quad+C \bigg(\int_{Q}e^{-2\lambda(t+x\cdot\omega)}|\mathcal{L}_{\mathcal{A},\mathcal{B},q}\overrightarrow{v}(t,x)|^2dxdt+\lambda\int\limits_{\Sigma_{-,\omega}}{ e^{-2\lambda(t+x\cdot\omega)}|\omega\cdot \nu(x)||\partial_{\nu}\overrightarrow{v}|^2}dS \bigg).
\end{aligned}
\end{align}
Thus the desired estimate \eqref{boundary carleman estimate} is obtained.

(\textbf{Interior Carleman estimates})

<!-- lemma-1: body_md -->
\cite[Lemma 5.4]{kian2016recovery}\label{Lemma for GO sol}
For $(\mathcal{A},\mathcal{B},q)\in \mathcal{T}(M_0,\alpha)$, let us consider the conjugated operator $$P_{\mathcal{A},\mathcal{B},\lambda,\omega}=e^{-\lambda(x\cdot\omega+t)}(\mathcal{L}_{\mathcal{A},\mathcal{B},q}-q)e^{\lambda(x\cdot\omega+t)}.$$ There exist constants $C,\lambda_0>0$ depending on $M_0,\alpha$ and $Q$ such that for $\overrightarrow{u}\in \mathbf{C}_c^{\infty}(Q)$ and $\lambda\ge\lambda_0$

<!-- proof-3: body_md -->
The proof is based on the Hahn-Banach and  Reis\'z representation theorem. 
 For this purpose, we consider the subspace 
 \[
 \mathcal{P}:=\{P_{-\mathcal{A},-\mathcal{B},-\lambda,\omega}\overrightarrow{v}:\ \overrightarrow{v}\in \mathbf{C}_c^{\infty}(Q)\}
 \]
and 
define a linear map $\mathcal{S}: \mathcal{P}\to\mathbb{R}$ by
\begin{align}\label{Def: operator S}
\mathcal{S}(P_{-\mathcal{A},-\mathcal{B},-\lambda,\omega}\overrightarrow{v})=\int_{Q}\overrightarrow{v}(t,x)\cdot\overrightarrow{f}(t,x)\ dxdt,\;~\overrightarrow{f}\in \mathbf{H}^1_\lambda(Q).
\end{align}
Following  Lemma \ref{Lemma for GO sol}, we conclude that  
\begin{align*}
|\mathcal{S}(P_{-\mathcal{A},-\mathcal{B},-\lambda,\omega}\overrightarrow{v})|\le \lVert \overrightarrow{v}\rVert_{\mathbf{H}^{-1}_{\lambda}(\mathbb{R}^{1+n})}\lVert \overrightarrow{f}\rVert_{\mathbf{H}^{1}_{\lambda}(Q)} \leq C
\|P_{-\mathcal{A},-\mathcal{B},-\lambda,\omega}\overrightarrow{v}\|_{\mathbf{H}^{-2}_{\lambda}(\mathbb{R}^{1+n})}\|\overrightarrow{f}\|_{\mathbf{H}^{1}_{\lambda}(Q)}.
\end{align*}
Thus, $\mathcal{S}$ is a continuous linear operator.
Using the Hahn-Banach theorem, we now extend $\mathcal{S}$ to a continuous functional on $\mathbf{H}^{-2}_{\lambda}(\mathbb{R}^{1+n})$ and  still denoted by $\mathcal{S}$. Also, we have  $\|\mathcal{S}\|\le C\|\overrightarrow{f}\|_{\mathbf{H}^{1}_{\lambda}(Q)} $. Following the Reis\'z representation theorem, we have a unique $\overrightarrow{u}\in{\mathbf{H}^{2}_{\lambda}(\mathbb{R}^{1+n})}$ such that 
\begin{align}\label{exist1}
\mathcal{S}(\overrightarrow{v})=(\overrightarrow{v},\overrightarrow{u})_{{\mathbf{H}^{-2}_{\lambda}(\mathbb{R}^{1+n})},{\mathbf{H}^{2}_{\lambda}(\mathbb{R}^{1+n})}},~~ \overrightarrow{v}\in \mathbf{C}_c^{\infty}(Q)
\end{align}
and 
\begin{align}
    \|\overrightarrow{u}\|_{{\mathbf{H}^{2}_{\lambda}(\mathbb{R}^{1+n})}}\le \|\mathcal{S}\|\le C\|\overrightarrow{f}\|_{\mathbf{H}^{1}_{\lambda}(Q)}.
\end{align}
Combining \eqref{Def: operator S} and \eqref{exist1} together, we have  
\begin{align}\label{exist2}
\begin{aligned}
\mathcal{S}(P_{-\mathcal{A},-\mathcal{B},-\lambda,\omega}\overrightarrow{v})=\int_{Q}\overrightarrow{v}(t,x)\cdot\overrightarrow{f}(t,x)\ dxdt=(P_{-\mathcal{A},-\mathcal{B},-\lambda,\omega}\overrightarrow{v},\overrightarrow{u})_{{\mathbf{H}^{-2}_{\lambda}(\mathbb{R}^{1+n})},{\mathbf{H}^{2}_{\lambda}(\mathbb{R}^{1+n})}},
\end{aligned}
\end{align}
for all $\overrightarrow{v}\in \mathbf{C}_c^{\infty}(Q)$.
From \eqref{exist2}, we conclude that there exist $\overrightarrow{u}\in{\mathbf{H}^{2}_{\lambda}(\mathbb{R}^{1+n})}$ such that 
\begin{align*}
P_{\mathcal{A},\mathcal{B},\lambda,\omega}\overrightarrow{u}=\overrightarrow{f}\mbox{ with  }\|\overrightarrow{u}\|_{{\mathbf{H}^{2}_{\lambda}(\mathbb{R}^{1+n})}}C\le\|\overrightarrow{f}\|_{\mathbf{H}^{1}_{\lambda}(Q)}.
\end{align*} 
Now, we seek for $\overrightarrow{u}_g$ solves $\mathcal{L}_{\mathcal{A}, \mathcal{B},q}\overrightarrow{u}=\overrightarrow{0}$, having the form
\begin{align}
&\quad \quad \overrightarrow{u}_g(t,x)=e^{\lambda(x\cdot\omega+t)}(\overrightarrow{B_g}(t,x)+\overrightarrow{R}_{g,\lambda}(t,x)).\nonumber
\end{align}
To do so, it is sufficient to find $\overrightarrow{B_g}$ and $\overrightarrow{R}_{g,\lambda}$ satisfying \[ e^{-\lambda(x\cdot\omega+t)}\mathcal{L}_{\mathcal{A},\mathcal{B},q}e^{\lambda(x\cdot\omega+t)}(\overrightarrow{B_g}(t,x)+\overrightarrow{R}_{g,\lambda}(t,x))=\overrightarrow{0}.\]
Equivalently,  \[P_{\mathcal{A},\mathcal{B},\lambda,\omega}\overrightarrow{R}_{g,\lambda}=-q\overrightarrow{R}_{g,\lambda}-P_{\mathcal{A},\mathcal{B},\lambda,\omega}\overrightarrow{B_g}-q\overrightarrow{B_g}=-q\overrightarrow{R}_{g,\lambda}-\Big(2\lambda\big\{1,-\omega\big\}\cdot\big\{\nabla_{t,x}B_g^j+\mathcal{C}^jB_g^j\big\}\Big)_{j=1}^m-\mathcal{L}_{\mathcal{A},\mathcal{B},q}\overrightarrow{B_g}.\]
Next, we set the term involving $\lambda$ to be 0, results in the following transport equation
\begin{align}
\qquad\qquad\Big(1,-\omega\Big)\cdot\Big(\nabla_{t,x}B_g^j(t,x)+\mathcal{C}^j(t,x)B_g^j(t,x)\Big)=0,\label{equation for GO solution}
\end{align}
for each $j=1,2,\dots,m$.
A particular solution to the above transport equation is given by 
\begin{align*}
B_g^j(t,x)=K^j\phi_j(x+t\omega)e^{-\int_0^t (1,-\omega)\cdot\mathcal{C}^{j}(s,x+(t-s)\omega)\,ds},
\end{align*}
where $K^j$ is $j^{th}$ component of an arbitrary constant vector $\overrightarrow{K}\in\mathbb{R}^m$ and $\phi_j\in C_c^\infty(\mathbb{R}^n)$ for each $1\leq j\leq m$. 
Thus, it remains to solve 
\begin{align}\label{exist3}
P_{\mathcal{A},\mathcal{B},\lambda,\omega}\overrightarrow{R}_{g,\lambda}=-q\overrightarrow{R}_{g,\lambda}-\mathcal{L}_{\mathcal{A},\mathcal{B},q}\overrightarrow{B_g},
\end{align}
where $\mathcal{L}_{\mathcal{A},\mathcal{B},q}\overrightarrow{B_g}\in \mathbf{H}^1_\lambda(Q)$ with $\|\mathcal{L}_{\mathcal{A},\mathcal{B},q}\overrightarrow{B_g}\|_{\mathbf{H}^1_\lambda(Q)}\le C\lambda\|\overrightarrow{\phi}\|_{\mathbf{H}^3(\mathbb{R}^n)}$.
Define $T(\overrightarrow{f})=\overrightarrow{u}$, where $P_{\mathcal{A},\mathcal{B},\lambda,\omega}\overrightarrow{u}=\overrightarrow{f}$. Then $\|T\|_{\mathbf{H}^1_\lambda(Q)\to \mathbf{H}^1_\lambda(Q)}\le \frac{C}{\lambda}$ and $\|T\|_{\mathbf{H}^1_\lambda(Q)\to \mathbf{H}^2_\lambda(Q)}\le C$.
Thus the problem \eqref{exist3} reduces to finding $\overrightarrow{\tilde{f}}\in \mathbf{H}^1_\lambda(Q)$ such that 
\begin{align*}
    (I+qT)\overrightarrow{\tilde{f}}=-\mathcal{L}_{\mathcal{A},\mathcal{B},q}\overrightarrow{B_g}.
\end{align*}
The invertibility of $(I+qT)$ in $\mathbf{H}^1_\lambda(Q)$ is guaranteed for $\lambda$  large enough. So, we can find $\overrightarrow{\tilde{f}}\in \mathbf{H}^1_\lambda(Q)$ and hence $\overrightarrow{R}_{g,\lambda}\in \mathbf{H}^2_\lambda(Q)$ which satisfies 
\begin{align}\label{remainder}
\|\overrightarrow{R}_{g,\lambda}\|_{\mathbf{H}^2_\lambda(Q)}\le C\|\overrightarrow{\tilde{f}}\|_{\mathbf{H}^1_\lambda(Q)}\le C\lambda\|\overrightarrow{\phi}\|_{\mathbf{H}^3(Q)}.
\end{align}
This completes the proof.

Assume that the hypotheses of Lemma \ref{first estimate for matrix potential} are satisfied. Then there exists an open cone $\mathcal{S}\subset \mathbb{R}^{1+n}$ such that, for every $(\tau,\xi)\in \mathcal{S}$, one has

=\frac{\varepsilon}{2}.\nonumber

Because $(\tau_0,\xi_0)\in\mathcal{S}$ was arbitrary, it follows that for every $(\tau,\xi)\in\mathcal{S}$ there exists $\omega\in\mathbb{S}^{n-1}$ satisfying $|\omega-\omega_0|<\varepsilon/2$ and $\tau=\omega\cdot\xi$, and consequently
\[
\bigl|(1,-\omega)\cdot\widehat{\mathcal{C}^{j}}(\tau,\xi)\bigr|
\le C\left(\frac{1}{\lambda^\delta}+ e^{\beta\lambda}\|\Lambda_1-\Lambda_2\|\right).
\]
Hence the result follows.

Let $\tilde{\mathcal{S}}$ denote the set of parameters $(\tau,\xi)\in\mathbb{S}^{n}$ contained in a sufficiently small neighborhood of $(\tau_0,\xi_0)$ such that the hyperplane
$
\{x\in\mathbb{R}^n;\ \xi\cdot x=\tau\}
$
depends continuously on $(\tau,\xi)$. More precisely, as $(\tau,\xi)$ varies continuously in $\tilde{\mathcal{S}}$, one may choose a family of $n$ linearly independent vectors $\{\omega_k(\tau,\xi)\}_{1\le k\le n}\subset \mathcal{B}_{\tau,\xi}$ depending continuously on $(\tau,\xi)$. Consequently, the matrix $M_{(\tau,\xi)}$ varies continuously with $(\tau,\xi)$ and retains its non-singular behaviors on $\tilde{\mathcal{S}}$, i.e. $\det M_{(\tau,\xi)}\neq 0$ for all $(\tau,\xi)\in\tilde{\mathcal{S}}$.
Moreover, due to the extreme-value theorem, there exists a constant  $c>0$ such that
\[
0<c\le |\det M_{(\tau,\xi)}|,\quad \text{for all }(\tau,\xi)\in\tilde{\tilde{\mathcal{S}}}\Subset \tilde{\mathcal{S}}.
\]
For $(\tau,\xi)\in\tilde{\tilde{\mathcal{S}}}$, consider a linear system
\begin{align}
\label{system1}(1,-\omega_k(\tau,\xi))\cdot \widehat{\mathcal{C}^{j}}(\tau,\xi)&=G(\xi,\omega_k(\tau,\xi)),\quad\mbox{ for }k=1,\ldots,n,\\
\frac{1}{\sqrt{\tau^2+\xi^2}}(\tau,\xi)\cdot \widehat{\mathcal{C}^{j}}(\tau,\xi)&=0.
\end{align}
We can write the above system in matrix form as $M_{(\tau,\xi)} X=B$, where 
\begin{align*}
M_{(\tau,\xi)}=\begin{pmatrix}
1 & -\omega_1^1(\tau,\xi) & \cdots & -\omega_1^n(\tau,\xi) \\
1 & -\omega_2^1(\tau,\xi) & \cdots & -\omega_2^n(\tau,\xi) \\
\vdots  & \vdots  & \ddots & \vdots  \\
1 & -\omega_n^1(\tau,\xi) & \cdots & -\omega_n^n(\tau,\xi) \\
\frac{\tau}{\sqrt{{\tau}^2+|\xi|^2}} & \frac{\xi^1}{\sqrt{{\tau}^2+|\xi|^2}} & \cdots & \frac{\xi^n}{\sqrt{{\tau}^2+|\xi|^2}} 
\end{pmatrix},\quad X=\begin{pmatrix}
    \widehat{C^j}_0\\\widehat{C^j}_1\\ \vdots\\\widehat{C^j}_n
\end{pmatrix},\quad B=\begin{pmatrix}
    G(\xi, \omega_1\\G(\xi, \omega_1)\\ \vdots\\ G(\xi, \omega_n)\\0
\end{pmatrix}.
\end{align*} 
Corollary \ref{existence of a cone} provides an upper bound for $G$ on the cone $\mathcal{S}$. Also,  $M_{(\tau,\xi)}$ is a homogeneous matrix of degree zero that is invertible on $\tilde{\tilde{\mathcal{S}}}$. Combining these properties, one can transfer local estimates from $\tilde{\tilde{\mathcal{S}}}$ to corresponding estimates on the open cone in $\mathbb{R}^{1+n}$ generated by $\tilde{\tilde{\mathcal{S}}}$.
In particular, by Lemma \ref{set of equation and bound for G}, for $(\tau,\xi)\in\tilde{\tilde{\mathcal{S}}}$ and $r>0$ we obtain
\begin{align*}
\widehat{C^j}_i(r\tau,r\xi)&=\sum_{k=1}^{n}c_{k,i}(r\tau,r\xi)G(r\xi,\omega^k(r\tau,r\xi)),\quad i\in\{0,\ldots,n\}\\
&=\sum_{k=1}^{n}c_{k,i}(\tau,\xi)G(r\xi,\omega^k(\tau,\xi)),
\end{align*}
where
$
c_{k,i}(\tau,\xi)=\frac{1}{\det M(\tau,\xi)}\,C_{i,k}(\tau,\xi),
$
and $C_{i,k}(\tau,\xi)$ denotes the $(i,k)^{\mathrm{th}}$ cofactor of $M_{(\tau,\xi)}$. Now for the open cone $\mathcal{S}_0\ (\equiv\cup_{r>0}r\tilde{\tilde{\mathcal{S}}})$ in $\mathbb{R}^{1+n}$, Corollary \ref{existence of a cone} along with previous calculation ensures the existence of $C>0$ such that the estimate
\begin{align}
|\widehat{C^j}_i(\tau,\xi)|\le C\left(\frac{1}{\lambda^\delta}+ e^{\beta\lambda}\|\Lambda_1-\Lambda_2\|\right),\label{oncone}
\end{align}
holds for all $i\in\{0,\ldots,n\}$.
This completes the proof.
