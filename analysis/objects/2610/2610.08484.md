<!-- proposition-1: body_md -->
\label{the:viscosity}
  For initial values $u_0 \in C^3_b(\mathbb{R}^d\setminus S) \cap C^{0,1}(\mathbb{R}^d)$ that are constant outside a bounded set and where $S \subset \mathbb{R}^d$ is closed with $\overline{\mathbb{R}^d\setminus S} = \mathbb{R}^d$, let $u_h$ be the unique classic solution of the median filter~\eqref{def:semi_discrete_scheme} starting from  $u_0$. Then $u_h \rightarrow u$ uniformly as $h \downarrow 0$ and $u$ is the unique viscosity solution of the level-set mean curvature flow equation
  \begin{equation}\label{eq:level_set_equation}
    \partial_t u = \nabla \cdot \left(\frac{\nabla u}{|\nabla u|} \right)|\nabla u|
  \end{equation}
   in the sense of  Definition~\eqref{def:viscosity}.

<!-- theorem-1: body_md -->
\label{cor:energy_conv}
Let $u_h$ be the solution of~\eqref{def:semi_discrete_scheme} starting from $u_0$, where $u_0$ satisfies the assumption of Proposition~\ref{the:viscosity} and
\begin{equation}\label{def:well_prepared_init}
     \sup_{h\in(0,1]}E_h(u_0)<\infty,
\quad     \sup_{h \in (0,1]} \int_{\mathbb{R}^d} |H_{0,h}(x)| \, dx < \infty.
\end{equation}
 Then it holds, with $u$ the viscosity solution of~\eqref{eq:level_set_equation}, that
  \begin{equation*}
     \lim_{h\to0}E_h(u_h(\cdot,t)) =  E(u(\cdot,t)) \quad \text{for a.e.\ }t>0.
  \end{equation*}

<!-- lemma-1: body_md -->
\label{lem:comp_princ}
    Assume that $\underline{w}$ and $\overline{w}$ are a sub- and super-solution to~\eqref{def:generalized_median}, in the sense that
    \begin{equation*}
      \partial_t \underline{w} \leq \frac{\operatorname{\mathop \textup{med}}_{K, \Psi} \underline{w} - \underline{w}}{h} \quad \quad \text{ and }\quad \quad\partial_t \overline{w} \geq \frac{\operatorname{\mathop \textup{med}}_{K, \Psi} \overline{w} - \overline{w}}{h},
    \end{equation*}
    respectively, with bounded continuous initial data satisfying  $\underline{w}(\cdot, 0) \leq \overline{w}(\cdot, 0)$, and such that $\lim_{|x|\to \infty} \underline{w}(x,t) \leq \lim_{|x|\to \infty} \overline{w}(x,t)$ exist for all $t \geq 0$. Then it holds for all $t \geq 0$ that                                                                                                               
    \begin{equation*}
        \underline{w}(\cdot, t) \leq \overline{w}(\cdot, t).
    \end{equation*}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
\emph{Step 1 -- Strict Inequalities:} First, we prove the claim if it holds
    \begin{equation*} 
      \underline{w}(\cdot, 0) < \overline{w}(\cdot, 0) \quad \quad \text{ and } \quad\quad \lim_{|x|\to \infty} \underline{w}(x,t) < \lim_{|x|\to \infty} \overline{w}(x,t)
    \end{equation*}
     for all $t \geq 0$, and if $\overline{w}$ is a strict super-solution, i.e, 
    \begin{equation*}
      \partial_t \overline{w} > \frac{\operatorname{\mathop \textup{med}}_{K, \Psi} \overline{w} - \overline{w}}{h}.
    \end{equation*}

    We argue by contradiction: Assume that $\underline{w} \geq \overline{w}$ at some point.  This, together with the continuity of $\underline{w}$ and $\overline{w}$ on $\mathbb{R}^d \times (0, \infty)$ implies that we can pick the smallest $t_0 > 0$ such that there exists an $x_0 \in \mathbb{R}^d$ with $\underline{w}(x_0,t_0) = \overline{w}(x_0,t_0)$. By construction, it also holds $\underline{w}(x,t_0) \leq \overline{w}(x,t_0)$ for all $x \in \mathbb{R}^d$. As $\underline{w}(x_0, t_0 - \varepsilon) - \overline{w}(x_0, t_0 -\varepsilon ) \leq 0 = \underline{w}(x_0, t_0 ) - \overline{w}(x_0, t_0 )$ for small $\varepsilon$ it holds 
    \begin{align*}
        0 \leq \partial_t (\underline{w}(x_0, t_0) - \overline{w}(x_0, t_0)) h &< \operatorname{\mathop \textup{med}}_{K, \Psi} \underline{w}(x_0,t_0) - \operatorname{\mathop \textup{med}}_{K, \Psi} \overline{w}(x_0,t_0) + \overline{w}(x_0, t_0) - \underline{w}(x_0, t_0) \\ 
        &= \operatorname{\mathop \textup{med}}_{K, \Psi} \underline{w}(x_0,t_0) - \operatorname{\mathop \textup{med}}_{K, \Psi}\overline{w}(x_0,t_0).
    \end{align*}
    But this is a contradiction, because the monotonicity of the median operator by Lemma~\ref{lem:uniqueness_median}, together with $\underline{w}(x,t_0) \leq \overline{w}(x,t_0)$ for all $x \in \mathbb{R}^d$, implies 
    \begin{equation*}
        \operatorname{\mathop \textup{med}}_{K, \Psi} \underline{w}(x_0,t_0) - \operatorname{\mathop \textup{med}}_{K, \Psi} \overline{w}(x_0,t_0) \leq 0.
    \end{equation*}
    
    \emph{Step 2 -- Conclusion:} 
    Define $\tilde{w}(\cdot, t) := \overline{w}(\cdot, t) + \delta (t +1)$ for any $t \geq 0$ and some $\delta > 0$.
    Then $\tilde{w}$ is a strict super-solution as
    \begin{align*}
      \partial_t \tilde{w}(\cdot, t) = \partial_t \overline{w} + \delta &> \frac{\operatorname{\mathop \textup{med}}_{K, \Psi} \overline{w}(\cdot, t) - \overline{w}(\cdot, t)}{h} \\ &= \frac{\operatorname{\mathop \textup{med}}_{K, \Psi} (\overline{w}(\cdot, t) + \delta (t + 1)) - (\overline{w}(\cdot, t) + \delta (t +1))}{h} = \frac{\operatorname{\mathop \textup{med}}_{K, \Psi} \tilde{w}(\cdot, t) - \tilde{w}(\cdot,t)}{h},
    \end{align*}
        where we used that for any $c \in \mathbb{R}$ and any continuous function $f$ it holds $\operatorname{\mathop \textup{med}}_{K,\psi}(f + c) = \operatorname{\mathop \textup{med}}_{K, \Psi}(f) + c$.
    Additionally, one has $\underline{w}(\cdot, 0) < \tilde{w}(\cdot, 0)$ and $\lim_{|x|\to \infty} \underline{w}(x,t) < \lim_{|x|\to \infty} \tilde{w}(x,t)$. This brings us in the position to apply \emph{Step 1} to the pair $\underline{w}, \tilde{w}$ yielding
    \begin{equation*}
      \underline{w}(x, t) \leq \tilde{w}(x, t) = \overline{w}(x, t) + \delta t \quad \quad \text{ for any } (x,t) \in \mathbb{R}^d \times [0, \infty).
    \end{equation*}
    As $\delta \geq 0$ was arbitrary, we can pass to the limit $\delta \to 0$ which finishes this proof.

<!-- proposition-2: body_md -->
For any initial value $w_0\in C^{0,1}(\mathbb{R}^d)$ that is constant outside a bounded set there exists a classic solution to~\eqref{def:generlized_scheme}.
<!-- end proposition-2 -->
<!-- proof-2: body_md -->
We want to approximate the equation~\eqref{def:generlized_scheme} by replacing the time derivative with a difference quotient. To that aim define iteratively
    \begin{equation}\label{def:app_scheme}
        w_\tau^{n+1} = \frac{\tau}{h}(\operatorname{\mathop \textup{med}}_{K, \Psi} w_\tau^n -w_\tau^n) + w_\tau^n
    \end{equation}
    with the initial condition $w_\tau^0 = w_0$ as given in~\eqref{def:semi_discrete_scheme}.
    Rearranging immediately yields the implicit Euler formula
    \begin{equation}\label{eq:discrete_equation}
        \frac{w_\tau^{n+1} - w_\tau^n}{\tau} = \frac{\operatorname{\mathop \textup{med}}_{K, \Psi} w_\tau^n - w_\tau^n}{h}.
    \end{equation}
    \emph{Step 1 -- Monotonicity in $L^\infty$: } We first claim that if $\tau < h$, then it holds
    \begin{equation}\label{eq:mon_Linfty_app_scheme}
        \sup_{x \in \mathbb{R}^d} |w_\tau^{n+1}| \leq \sup_{x \in \mathbb{R}^d} |w_\tau^n|,
    \end{equation}
    which implies that $w_\tau^n$ is uniformly bounded.

    Indeed, as it holds $0 \leq 1- \frac{h}{\tau} \leq 1$ and $|\operatorname{\mathop \textup{med}}_{K, \Psi} v(x_0)| \leq \sup_{y \in R^d} |v(y)|$ for any function $v$ by Lemma~\ref{lem:uniqueness_median} (comparing to a constant function) and $x_0 \in \mathbb{R}^d$ we simply compute
    \begin{align*}
        |w_\tau^{n+1}(x_0)| &= |\frac{\tau}{h} \operatorname{\mathop \textup{med}}_{K, \Psi} w_\tau^n(x_0) + (1- \frac{h}{\tau} ) w_\tau^n(x_0)|\\
        & \leq \frac{\tau}{h}| \operatorname{\mathop \textup{med}}_{K, \Psi} w_\tau^n(x_0)| + (1- \frac{h}{\tau} ) |w_\tau^n(x_0)| \\
        & \leq \frac{\tau}{h} \sup_{x \in \mathbb{R}^d} |w_\tau^n(x)| + (1- \frac{h}{\tau} ) \sup_{x \in R^d}|w_\tau^n(x)|\\
        &\leq \sup_{x \in \mathbb{R}^d} |w_\tau^n(x)|.
    \end{align*}
    As $x_0$ was arbitrary, the claim follows.
    
    \emph{Step 2 -- Linear Interpolation:} We introduce the linear interpolation $w_\tau: \mathbb{R}^d \times (0, \infty) \rightarrow \mathbb{R}$, i.e., $w_\tau$ is piecewise affine in time and it holds 
    \begin{equation*}
      w_\tau(x, n \tau) = w_\tau^n(x) \quad \text{ for all } n \in \mathbb{N}. 
    \end{equation*}

    \emph{Step 3 -- Equicontinuity in Space and Time:} We claim that the family of functions $\{w_\tau\}_\tau$ of the linear interpolations is equicontinuous. 

    We start with the equicontinuity in time: Notice the time-slope of the linear interpolation is bounded as it holds by \emph{Step 1} that
    \begin{align*}
      \left| \frac{w_\tau( \cdot, (n+1) \tau) - w_\tau( \cdot, n\tau)}{\tau} \right| = \left| \frac{\operatorname{\mathop \textup{med}}_{K, \Psi} w_\tau^n(\cdot) - w_\tau^n(\cdot)}{h} \right| \leq \frac{2  \|w_0\|_\infty}{h}.
    \end{align*}
    As $h$ is fixed this immediately implies the claimed equicontinuity in time.

    We continue with the equicontinuity in space:
   Later, in Lemma~\ref{lem:contraction} we will prove that the generalized median operator is 1-Lipschitz, i.e.,
    \begin{equation}\label{eq:med_contraction}
      |\operatorname{\mathop \textup{med}}_{K, \Psi} f(x) - \operatorname{\mathop \textup{med}}_{K, \Psi} f(y)| \leq L(f) |x-y| \quad \text{ for all } x,y \in \mathbb{R}^d,
    \end{equation}
    for any continuous, bounded $f$ and the Lipschitz constant $L(f)$.
    
    Equipped with~\eqref{eq:med_contraction} we can now compute that the Lipschitz constant of $w_\tau$ is also non-increasing, i.e.,
    \begin{align*}
      |w_\tau^{n+1}(x) - w_\tau^{n+1}(y)| &\leq \frac{\tau}{h} |\operatorname{\mathop \textup{med}}_{K, \Psi} w_\tau^n (x) - \operatorname{\mathop \textup{med}}_{K, \Psi} w_\tau^n(y)| + (1- \frac{\tau}{h}) |w_\tau^n(x) - w_\tau^n(y)| \\
      &\leq \frac{\tau}{h} L(w_\tau^n)|x-y| + (1- \frac{\tau}{h}) L(w_\tau^n) |x-y|\\
      &= L(w_\tau^n) |x-y|.
    \end{align*}
    By induction, it follows
    \begin{equation} \label{eq:bound_lipschitz}
      L(w_\tau^{n+1}) \leq L(w_\tau^{n}) \leq \dots \leq L(w_0),
    \end{equation}
    which implies the space-equicontinuity of $w_\tau$.

    \emph{Step 4 -- Compactness of $w_\tau$:} The requirements of the Arzel{\`a}-Ascoli theorem are fulfilled by \emph{Step~1} and \emph{Step~3}. Hence, there exists a $w$ such that $w_\tau \rightarrow w$ uniformly on $B_R(0) \times (0, T)$ for any $R, T > 0$. Furthermore, $w$ is Lipschitz in space with 
    \begin{equation}\label{eq:Lip_u_to_u_0}
      \operatorname*{Lip}(w) \leq \operatorname*{Lip}(w_0).
    \end{equation}

    \emph{Step 5 -- Equicontinuity of Difference Quotient:}
    We show that the difference quotient 
    \begin{equation}\label{def:difference_quotient}
      \frac{w_\tau(x, t + \tau) - w_\tau(x, t)}{\tau} 
    \end{equation}
    is equicontinuous in $x \in \mathbb{R}^d$ and $t \in [0, \infty)$. 

    We start with the time-equicontinuity: By \emph{Step 3} it holds for any $s,t \geq 0$ and $z \in \mathbb{R}^d$ that $|w_\tau(z, t) - w_\tau(z, s)| \leq \frac{2\|w_0\|_\infty}{h} |t-s|$. Again, as the median operator is $1$-Lipschitz by Lemma~\ref{lem:contraction} that implies
    \begin{equation*}
      |\operatorname{\mathop \textup{med}}_{K, \Psi} w_\tau (x, t) - \operatorname{\mathop \textup{med}}_{K, \Psi} w_\tau(x, s)| \leq \sup_{z \in \mathbb{R}^d} |w_\tau(z, t) - w_\tau(z, s)| \leq \frac{2\|w_0\|_\infty}{h} |t-s|.
    \end{equation*} 
    By our definition of the scheme~\eqref{def:app_scheme} it follows now for $t = n \tau$ and $n \in \mathbb{N}$
    \begin{align*}
      &\left|\frac{w_\tau(x, t + \tau) - w_\tau(x, t)}{\tau} - \frac{w_\tau(x, s + \tau) - w_\tau(x, s)}{\tau} \right| \\
      &= \left|\frac{\operatorname{\mathop \textup{med}}_{K, \Psi} w_\tau(x,t) - \operatorname{\mathop \textup{med}}_{K, \Psi} w_\tau(x,s)}{h} + \frac{w_\tau(x, s) - w_\tau(x,t)}{h} \right|\\
      &\leq \frac{4\|w_0\|_\infty}{h^2} |t-s|.
    \end{align*}
    As the difference quotient is linear in $w_\tau$ the result follows by the properties of the linear interpolation.
    For the space-equicontinuity we use again~\eqref{eq:med_contraction} and \eqref{eq:bound_lipschitz} of \emph{Step 3} to compute for $t = n \tau$ and $n \in \mathbb{N}$
    \begin{align*}
      &\left|\frac{w_\tau(x, t + \tau) - w_\tau(x, t)}{\tau} - \frac{w_\tau(y, t + \tau) - w_\tau(y, t)}{\tau} \right| \\
      &= \left|\frac{\operatorname{\mathop \textup{med}}_{K, \Psi} w_\tau(x, t) - \operatorname{\mathop \textup{med}}_{K, \Psi} w_\tau(y, t)}{h} + \frac{w_\tau(y, t) - w_\tau(x, t)}{h} \right|\\
      &\leq \frac{2 L(w_0)}{h}|x-y|.
    \end{align*}
    Again, the result for general $t$ follows by the properties of the linear interpolation, i.e., for any Lipschitz continuous $f^0, f^1$ and the linear interpolation $f$ between them it holds
    \begin{align*}
      |f(x,t) - f(y,t)| &= |(1-t) f^0(x ) + t f^1(x) - (1-t) f^0(y) - t f^1(y)| \\
      &\leq (1-t) |f^0(x) - f^0(y)| + t|f^1(x) - f^1(y)| \\
      &\leq \max\{L(f^0), L (f^1)\} |x - y|.
    \end{align*}

    \emph{Step 6 –- Compactness of the Difference Quotient:} We claim that the difference quotient converges to the time derivative of $w$, the limit of \emph{Step 4}.

    The equicontinuity in time of $w$ form \emph{Step~3} gives the equiboundedness of the difference quotients~\eqref{def:difference_quotient} and \emph{Step~5} gives the equicontinuity. By the Arzel{\`a}-Ascoli theorem (up to subsequences)
    \begin{equation}\label{eq:lim_differnce_quotient}
        \frac{w_\tau(x, t + \tau) - w_\tau(x, t)}{\tau} \rightarrow f(x,t)
    \end{equation}
    uniformly as $\tau \downarrow 0$ for any $x \in B_R(0)$ and $t \in [0,T]$. It is also clear that $f$ is Lipschitz-continuous.
    By the linear interpolation in time it holds
    \begin{equation*}
      \frac{w_\tau(x, t + \tau) - w_\tau(x, t)}{\tau}  = \partial_t w_\tau(x,t) \quad \quad \text{ for } t \in [0, \infty) \setminus \mathbb{N}\tau.
    \end{equation*}
    This lets us conclude, by the fundamental theorem of calculus and the continuity of $w_\tau$ in time, that for any $0 \leq s_1 \leq s_2 \leq T$
    \begin{equation*}
      \int_{s_1}^{s_2} \frac{w_\tau(x, t + \tau) - w_\tau(x, t)}{\tau}\, dt = w_\tau(s_2) - w_\tau(s_1).
    \end{equation*}
    Now we can use~\eqref{eq:lim_differnce_quotient} to pass to the limit $\tau \downarrow 0$ to obtain
    \begin{equation*}
      \frac{1}{s_2-s_1}\int_{s_1}^{s_2} f(t) \, dt = \frac{w(s_2) - w(s_1)}{s_2 - s_1}.
    \end{equation*}
    As $f$ is continuous we can pass $s_1 \rightarrow s_2$ to obtain
    \begin{equation*}
      f(t) = \partial_t w(t)
    \end{equation*}
    for any $t \geq 0$. Hence, $w$ is differentiable in time and it holds
    \begin{equation*}
        \frac{w_\tau(t + \tau, x) - w_\tau(t, x)}{\tau} \rightarrow \partial_t w(t,x).
    \end{equation*}

\emph{Step 7 -- Convergence of the Median:} We claim, for any $x \in \mathbb{R}^d$, $t \geq 0$ it holds
\begin{equation}\label{eq:conv_med_tau}
  \lim_{\tau \downarrow 0} \operatorname{\mathop \textup{med}}_{K, \Psi} u_{\tau}(x,t) = \operatorname{\mathop \textup{med}}_{K, \Psi} w(x,t),
\end{equation}
where $w$ is the limit of $\emph{Step 4}$.

To prove this, fix $x \in \mathbb{R}^d$, $t\geq 0$ and $\varepsilon > 0$. By Lemma~\ref{lem:cuttable} combined with $\max\{\operatorname*{Lip}(w_\tau), \operatorname*{Lip}(w)\} \leq \operatorname*{Lip}(w_0)$ there exists an $R > 0$ such that for the cut-off kernel $K_R(x,y) = \mathbbold{1}_{|x-y|\leq R} K(x,y)$ it holds
\begin{equation*}
  |\operatorname{\mathop \textup{med}}_{K_R, \Psi} w_\tau(x,t) - \operatorname{\mathop \textup{med}}_{K, \Psi} w_\tau(x,t)| \leq \frac{\varepsilon}{3} \quad \quad \text{ and } \quad \quad |\operatorname{\mathop \textup{med}}_{K_R, \Psi} w(x,t) - \operatorname{\mathop \textup{med}}_{K, \Psi} w(x,t)| \leq \frac{\varepsilon}{3}.
\end{equation*}
Furthermore, by the local uniform convergence of \emph{Step 4} there exists a $\tau_0$ such that for $\tau < \tau_0$ it holds 
\begin{equation*}
\|w_\tau(\cdot,t) - w(\cdot, t)\|_{L^\infty(B_R(0))} \leq \frac{\varepsilon}{3}.
\end{equation*}
This yields together with Lemma~\ref{lem:contraction} the bound
\begin{equation*}
   |\operatorname{\mathop \textup{med}}_{K_R, \Psi} w_\tau(x,t) - \operatorname{\mathop \textup{med}}_{K_R, \Psi} w(x,t)| \leq \|w_\tau(\cdot, t) - w(\cdot, t)\|_{L^\infty(\mathop \textup{supp} K_R(x, \cdot))} \leq \frac{\varepsilon}{3}.
\end{equation*}
To summarize, for $\tau \leq \tau_0$ it holds
\begin{align*}
&\left|\operatorname{\mathop \textup{med}}_{K,\Psi} w_\tau(x,t)-\operatorname{\mathop \textup{med}}_{K,\Psi} w(x,t)\right| \\
&\leq \left|\operatorname{\mathop \textup{med}}_{K,\Psi} w_\tau(x,t)-\operatorname{\mathop \textup{med}}_{K_R,\Psi} w_\tau(x,t)\right|
\,+\, \left|\operatorname{\mathop \textup{med}}_{K_R,\Psi} w_\tau(x,t)-\operatorname{\mathop \textup{med}}_{K_R,\Psi} w(x,t)\right| \\
&\qquad + \left|\operatorname{\mathop \textup{med}}_{K_R,\Psi} w(x,t)-\operatorname{\mathop \textup{med}}_{K,\Psi} w(x,t)\right| \\
&\leq \varepsilon.
\end{align*}
As $\varepsilon$ was, arbitrary this finishes \emph{Step 7}.

    \emph{Step 8 -- Conclusion:} We claim we can pass to the limit in~\eqref{eq:discrete_equation}.

    By the local uniform convergence of $w_\tau$ from \emph{Step 4}, the local uniform convergence of the difference quotient from \emph{Step 6}, and the pointwise convergence of the median from \emph{Step 7}, we can indeed pass to the pointwise limit $\tau \to 0$ in~\eqref{eq:discrete_equation} to obtain
    \begin{equation*}
      \partial_t w(x,t) = \frac{\operatorname{\mathop \textup{med}}_{K, \Psi} w(x,t) - w(x,t) }{h}.
    \end{equation*}
    Thus, $w$ is a classical solution to \eqref{def:generlized_scheme}.

<!-- lemma-2: body_md -->
\label{lem:mon_time_derivative}
  Let $w$ be a solution of~\eqref{def:generlized_scheme} with initial value $w_0\in C_b^{0,1}(\mathbb{R}^d)$ then it holds for $0 \leq s \leq t < \infty$
  \begin{equation*}

  \end{equation*}

<!-- proof-3: body_md -->
Let $w$ be a solution of~\eqref{def:generlized_scheme} and $\tau > 0$ be arbitrary, then it holds
  \begin{equation*}
    w(x,t + \tau) = w(x,t) + \tau \partial_t w(x,t) + \int_{t}^{t + \tau} \partial_t w(x,s) - \partial_t w(x,t) \, ds.
  \end{equation*}
  We use the contraction property of Lemma~\ref{lem:contraction} to compute
  \begin{multline}\label{eq:partial_t_contraction}
    |\partial_t w(x,s) - \partial_t w(x,t)| \leq \left| \frac{\operatorname{\mathop \textup{med}}_{K, \Psi} w(x,s) - \operatorname{\mathop \textup{med}}_{K, \Psi} w(x,t)}{h} \right| + \left| \frac{w(x,s) - w(x,t)}{h} \right| \\
    \leq \frac{2}{h} \|w(\cdot,s) - w(\cdot,t)\|_{C^0(\mathbb{R}^d)}.
  \end{multline}
  Furthermore, we crudely estimate, using $|\operatorname{\mathop \textup{med}}_{K, \Psi} w(\cdot,t)| \leq \|w(\cdot, t)\|_\infty \leq \|w_0\|_\infty$ by~\eqref{eq:mon_Linfty_app_scheme},
  \begin{equation}\label{eq:partial_t_contraction2}
    \|w(\cdot,s) - w(\cdot,t)\|_{C^0(\mathbb{R}^d)} \leq |s-t|\|\partial_t w\|_{C^0(\mathbb{R}^d \times [0,\infty))} \leq |s-t| \frac{2\|w_0\|_{\infty}}{h}.
  \end{equation}
  Combining the above computations yields
  \begin{equation*}
    w(t+\tau) = w(t) + \tau \partial_t w(t) + \mathcal{O}\big(\frac{\tau^2}{h^2}\big).
  \end{equation*}
  So for two solutions $v, w$ of~\eqref{def:generlized_scheme} we have again by Lemma~\ref{lem:contraction}
  \begin{align}\label{eq:difference_quotient_second_order}
    &\|w(\cdot, t+\tau) - v(\cdot, t+\tau)\|_\infty \\
    &\leq (1-\frac{\tau}{h})\|w(\cdot, t) - v(\cdot, t)\|_\infty + \frac{\tau}{h}\|\operatorname{\mathop \textup{med}}_{K, \Psi} w(\cdot, t) - \operatorname{\mathop \textup{med}}_{K, \Psi} v(\cdot, t)\|_\infty + \mathcal{O}\big(\frac{\tau^2}{h^2}\big) \nonumber\\
    &\leq \|w(\cdot, t) - v(\cdot, t)\|_\infty + \mathcal{O}\big(\frac{\tau^2}{h^2}\big) \nonumber.
  \end{align}
  Denote $\zeta(t):= \|w(\cdot, t) - v(\cdot, t)\|_\infty$ and compute for $0 \leq s \leq t < \infty$ that
  \begin{align*}
    \zeta(t) - \zeta(s) = \sum_{i = 1}^N \zeta\Big(t - \frac{i-1}{N}(t-s)\Big) - \zeta\Big(t - \frac{i}{N}(t-s)\Big) \leq \sum_{i = 1}^N \mathcal{O}\Big(\frac{|s-t|^2 }{(Nh)^2}\Big) = \mathcal{O}\Big(\frac{|s-t|^2 }{N h^2}\Big).
  \end{align*}
  As $s,t$ and $h$ are fixed we can pass $N \to \infty$ to obtain
  \begin{equation*}
    \|w(\cdot, t) - v(\cdot, t)\|_\infty \leq \|w(\cdot, s) - v(\cdot, s)\|_\infty.
  \end{equation*}
  Now using $v(\cdot, t) := w(\cdot, t + \tau)$ we obtain
  \begin{equation}\label{eq:before_pass_tau}
    \frac{1}{\tau}\|w(\cdot, t) - w(\cdot, t + \tau)\|_\infty \leq \frac{1}{\tau}\|w(\cdot, 0) - w(\cdot, \tau)\|_\infty.
  \end{equation}
  We can pass to the limit in the above estimate if we can show that for arbitrary $t \geq 0$ 
  \begin{equation*}
    \lim_{\tau \to 0} \frac{1}{\tau}\|w(\cdot, t) - w(\cdot, t + \tau)\| _\infty = \|\partial_t w(\cdot, t)\|_\infty.
  \end{equation*}
  But this follows by the inverse triangle inequality and the above computations~\eqref{eq:partial_t_contraction} and \eqref{eq:partial_t_contraction2}:
  \begin{align*}
    \left|\frac{1}{\tau}\|w(\cdot, t + \tau) - w(\cdot, t)\|_\infty - \|\partial_t w(\cdot, t)\|_\infty \right| &\leq \left\|\frac{w(\cdot, t + \tau) - w(\cdot, t)}{\tau} - \partial_t w(\cdot, t) \right\|_\infty\\
    &= \left\| \frac{1}{\tau} \int_{t}^{t + \tau} \partial_t w(\cdot, s) - \partial_t w(\cdot, t)\,ds\right\|_\infty\\
    &\lesssim \frac{1}{\tau} \int_{t}^{t + \tau} \frac{|s-t|}{h^2} \,ds\\
    & = \frac{\tau}{h^2}.
  \end{align*}
  Thus, we can pass to the limit in~\eqref{eq:before_pass_tau} and finish the proof.

\label{def:viscosity}
    A \emph{viscosity super-solution} is a function $u \in C^0(\mathbb{R}^d \times [0, T])$ such that for any $\phi \in C^\infty(\mathbb{R}^d \times [0, T])$ and any minimum point $(x_0, t_0) \in \mathbb{R}^d \times (0,T)$ of $u - \phi$ it holds
    \begin{equation*}
      \partial_t \phi (x_0, t_0) \geq F_*(\nabla \phi(x_0, t_0), \nabla^2\phi(x_0, t_0)).
    \end{equation*}
    Furthermore, $u$ is a \emph{viscosity sub-solution} if $-u$ is a super solution. Finally, for an initial value $u_0 \in C^0(\mathbb{R}^d \times [0, T])$, the function $u$ is a \emph{viscosity solution} to level-set mean curvature flow if $u$ is a viscosity super- and sub-solution and if $u(\cdot, 0) = u_0$.

First we record that the comparison principle of~\cite[Theorem 3.2]{evans_spruck_I} yields the uniqueness of viscosity solutions to level-set mean curvature flow.

  Define the two limits 
  \begin{align*}
    \overline{u}(x,t)= \limsup_{y \rightarrow x, s \rightarrow t, h \downarrow 0} u_h(y,s),\\
    \underline{u}(x,t) = \liminf_{y \rightarrow x, s \rightarrow t, h \downarrow 0} u_h(y,s).
  \end{align*}
  We claim that $\overline{u}$ and $\underline{u}$ are viscosity sub- and super-solution, respectively, with $\overline{u}(\cdot, 0) = \underline{u}(\cdot, 0) = u_0$. If the claim is valid, the comparison principle for the viscosity solutions~\cite[Theorem 3.2]{evans_spruck_I} yields $\overline{u} \leq \underline{u}$. But, the opposite inequality is also true by definition of $\underline{u}$ and $\overline{u}$ implying
  \begin{equation*}
    u = \underline{u} = \overline{u}.
  \end{equation*}
  This means $u$ is the unique viscosity solution and by standard arguments~\cite[Remark 6.4]{zbMATH00089226} one concludes the local uniform convergence $u_h \rightarrow u$ in space and time. As $u_h = u = c$ outside $B_L(0)$ this local convergence is upgraded to global uniform convergence on $\mathbb{R}^d \times [0,T]$.

  So it remains to justify the claim.
  
  \emph{Step 1 -- Initial data $\overline{u}(\cdot, 0) = \underline{u}(\cdot, 0) = u_0$:} We focus on the proof of $\overline{u}(\cdot, 0) = u_0$ as the other case is treated similarly. As (see for example~\eqref{eq:Lip_non_increasing} later)
  \begin{equation}\label{eq:uniform_lipschitz_bound}
    \sup_{h \in (0,1], t \in [0, \infty)}\operatorname*{Lip}( u_h(\cdot, t)) \leq \operatorname*{Lip}(u_0) \leq C_1
  \end{equation}
  holds, we immediately get $\overline{u}(x, 0) = \limsup_{h \downarrow 0, t \downarrow 0} u_h(x,t)$. Furthermore, we have the uniform bound (see for example~\eqref{eq:max_non_increasing} later)
  \begin{equation*}
    \sup_{h \in (0,1], t \in [0, \infty)} \|u_h(\cdot,t)\|_\infty \leq \|u_0(\cdot)\|_\infty \leq C_2.
  \end{equation*}
  Thus, for any sequence $(h_k, t_k) \to (0,0)$ as $k \to \infty$ we can apply the Arzel{\`a}-Ascoli theorem to get a subsequence $(h_{k_n}, t_{k_n}) \to (0,0)$ as $n \to \infty$ such that
  \begin{equation}\label{eq:uniform_v_limit}
    \lim_{n \to \infty} u_{h_{k_n}}(\cdot, t_{k_n}) = v \quad \quad \text{ uniformly},
  \end{equation}
  for some function $v \in C_b^{0,1}(\mathbb{R}^d)$. We claim it has to hold $v = u_0$. By assumption, we have that $u_0 \in C^3_b(\mathbb{R}^d \setminus S)$. As $\mathbb{R}^d \setminus S$ is dense in $\mathbb{R}^d$, the continuity of $v$ and $u_0$ implies that it is enough to prove that $v = u_0$ on $\mathbb{R}^d \setminus S$. So let $x \in \mathbb{R}^d \setminus S$ be given. As $\mathbb{R}^d \setminus S$ is open by assumption, we have that $u_0$ is twice differentiable in a neighborhood of $x$. Furthermore, $u_0$ is bounded. Hence, there exists $\xi_0, \zeta_0 \in C^3_b(\mathbb{R}^d)$ with
  \begin{align*}
    \xi_0 \leq u_0 \leq \zeta_0 \quad \text{ on } \mathbb{R}^d \quad\quad \quad \text{ and } \quad \quad \quad\xi_0(x) = u_0(x) = \zeta_0(x).
  \end{align*} 
  Denote by $\xi_h$ and $\zeta_h$ the solutions to~\eqref{def:semi_discrete_scheme} with initial values $\xi_0$ and $\zeta_0$, respectively.
  By the comparison principle of Lemma~\ref{lem:comp_princ} we get
  \begin{equation*}
    \xi_h \leq u_h \leq \zeta_h.
  \end{equation*}
  Thus, if we can show
  \begin{equation}\label{eq:upper_lower_limits_match}
    \lim_{n \to \infty} \xi_{h_{k_n}}(x, t_{k_n}) = \lim_{n \to \infty} \zeta_{h_{k_n}}(x, t_{k_n}) = u_0(x)
  \end{equation}
  it follows $v(x) = u_0(x)$. We focus on $\lim_{n \to \infty} \xi_{h_{k_n}}(x, t_{k_n}) =$ the version for $\zeta$ is analogous.

  We want to show that the time derivative of $\xi_h$ is uniformly bounded. By Lemma~\ref{lem:mon_time_derivative} it is enough to check this for $t = 0$. Fix $y \in \mathbb{R}^d$ arbitrary and distinguish the following two cases:

  \emph{Case 1 -- Upper Bounded Gradient $|\nabla \xi_0(y)| \leq \sqrt{h}$:}
  Let $R > 0$ be the constant radius with
  \begin{equation*}
    \int_{\{z > R\}} G_h(0,z) \,dz = \frac 12.
  \end{equation*}
  Then we can assume that there exists a $|z_0| \leq R$ with $\operatorname{\mathop \textup{med}}_h \xi_0 (x) = \xi_0(x + z_0)$ as otherwise the Euler-Lagrange inequality~\eqref{eq:ELI1} would be violated. One checks that there is a constant $C$ such that $R \leq C\sqrt{h}$ which yields $|z_0| \leq C \sqrt{h}$.
   By the case distinction and as $\xi_0 \in C^3_b(\mathbb{R}^d)$ we have  
  \begin{equation*}
    |\operatorname{\mathop \textup{med}}_h \xi_0(y) - \xi_0(y)| =  |\xi_0(y+ z_0) - \xi_0(y)| \leq |\nabla \xi_0(y)| |z_0| + \mathcal{O}(|z_0|^2) = \mathcal{O}(h).
  \end{equation*}
  Thus, we have that 
  \begin{equation*}
    |\partial_t \xi_h(y, 0)| = \left| \frac{\operatorname{\mathop \textup{med}}_h \xi_0(y) - \xi_0(y)}{h}\right| \lesssim 1.
  \end{equation*}

  \emph{Case 2 -- Lower Bounded Gradient $|\nabla \xi_0(y)| > \sqrt{h}$:} In this case we apply~\eqref{eq:strong_consistency} to immediately obtain
  \begin{align*}
    |\partial_t \xi_h(y, 0)| \lesssim \frac{1 + \|\nabla^3 \xi_0\|_\infty}{|\nabla \xi_0(y)|} \sqrt{h} + |F(\nabla \xi_0(y), \nabla^2 \xi_0(y))| \leq 1 + \|\nabla^3 \xi_0\|_\infty + \|\nabla^2 \xi_0\|_\infty < \infty.
  \end{align*}

  Combining Case 1, Case 2 and Lemma~\ref{lem:mon_time_derivative} we find
  \begin{equation*}
    \sup_{h \in (0,1], s \in [0, \infty)}\|\partial_t \xi_h(\cdot, s)\|_\infty \lesssim 1.
  \end{equation*}
  But now, as $\xi_{h_{k_n}}(x,0) = u_0(x)$, we simply have
  \begin{equation*}
    |u_0(x) - \xi_{h_{k_n}}(x, t_{k_n})| = \left| \int_{0}^{t_{k_n}} \partial_t \xi_{h_{k_n}}(x, s)\, ds\right| \leq t_{k_n} \sup_{h \in (0,1], s \in [0, \infty)} |\partial_t \xi_h(\cdot, s)| \lesssim t_{k_n}.
  \end{equation*}
  Taking the limit $n \to \infty$ yields~\eqref{eq:upper_lower_limits_match}.

  Now we know that $v = u_0$ and as implication by~\eqref{eq:uniform_v_limit} that
  \begin{equation}
    \lim_{n \to \infty} u_{h_{k_n}}(\cdot, t_{k_n}) = u_0 \quad \quad \text{ uniformly}.
  \end{equation}
  As we have identified the limit, we can get rid of the subsequences which yields
  \begin{equation*}
    \lim_{k \to \infty} u_{h_{k}}(\cdot, t_{k}) = u_0 \quad \quad \text{ uniformly}.
  \end{equation*}
  The sequence $(h_k, t_k) \to 0$ was arbitrary which gives us
  \begin{equation*}
    \limsup_{h \to 0, t \to 0} u_h(\cdot, t) = u_0 \quad \quad \text{ uniformly}.
  \end{equation*}
  Finally, by the Lipschitz continuity of $u_0$ it holds
  \begin{equation*}
    |u_h(y,t) - u_0(x)| \leq |u_h(y,t) - u_0(y)| + \operatorname*{Lip}(u_0)|x-y|,
  \end{equation*}
  or in other words 
  \begin{equation*}
    \overline{u}(x,t) = \limsup_{h \to 0, y \to x, t \to 0} u_h(y,t) = u_0(x).
  \end{equation*}

  \emph{Step 2 -- Argument for $\overline{u}$ and $\underline{u}$ are viscosity sub- and super-solutions, respectively:}
  We focus on the proof for $\overline{u}$; the other case is identical. Let $\phi$ be a smooth test function and $(x_0, t_0)$ be a strict local maximum of $\overline{u} - \phi$. W.l.o.g.\ we can assume that $\overline{u}(x_0, t_0) = \phi(x_0, t_0)$, i.e., 
  \begin{equation*}
    \overline{u}(x,t) - \phi(x,t) \leq \overline{u}(x_0, t_0) - \phi(x_0, t_0) = 0 \quad \quad \text{ in } B_r(x_0, t_0),
  \end{equation*}
  and that $\phi \geq 2 \sup_{h} \|u_h\|_{C^0(\mathbb{R}^d \times [0,\infty))}$ outside the ball $B_r(x_0, t_0)$.
  Define the auxiliary functions 
  \begin{align*}
    \psi_\delta (x,y,t) = \overline{u}(x,t) - \frac{|x-y|^4}{\delta} - \phi(y,t),\\
    \psi_{h,\delta} (x,y,t) = u_h(x,t) - \frac{|x-y|^4}{\delta} - \phi(y,t).
  \end{align*}
  We claim: 
  \begin{quote}
    \emph{There exists a sequence $(x_{h,\delta}, y_{h,\delta}, t_{h,\delta})$ of local maximum points of $\psi_{h,\delta}$ that converge to $(x_\delta, x_\delta, t_\delta)$, a maximum point of $\psi_\delta$, as $h \to 0$. We furthermore claim that $(x_\delta, x_\delta, t_\delta)$ converges to $(x_0, x_0, t_0)$ as $\delta \to 0$.}
  \end{quote}
  To see this, we first note that due to $\phi \geq 2 \sup_{h} \|u_h\|_{C^0(\mathbb{R}^d \times [0,T])}$ outside the ball $B_r(x_0, t_0)$ we have for any $y, t \in \mathbb{R}^d \times [0,\infty)$ and $x \in \mathbb{R}^d$ that
  \begin{align*}
    u_h(x_0, t_0) - \frac{|x_0-x_0|^4}{\delta}-\phi(x_0, t_0) = 0 &\geq u_h(x, t) - 2\sup_h \|u_h\|_{C^0(\mathbb{R}^d \times [0,\infty))}\\
     &\geq  u_h(x, t) - \frac{|x-y|^4}{\delta} - \phi(y, t)
  \end{align*} 
  Thus, we can restrict our domain to $(x,y,t) \in \mathbb{R}^d \times B_r(x_0, t_0)$. Additionally, we have for any $(x,y,t) \in B_{r+1}(x_0) \times B_r(x_0, t_0)$ and any $\delta \leq \frac{1}{\|u_h\|_{\infty} + \|\phi\|_{B_r(x_0, t_0)}}$ that
  \begin{equation*}
    u_h(x,t) - \frac{|x-y|^4}{\delta} - \phi(y,t) \leq \|u_h\|_{\infty} - \frac{1}{\delta} + \|\phi\|_{B_r(x_0, t_0)} \leq 0 = u_h(x_0, t_0) - \frac{|x_0-x_0|^4}{\delta}-\phi(x_0, t_0).
  \end{equation*} 
  Hence, there have to exist global minima $(x_{h, \delta}, y_{h, \delta},t_{h, \delta})$ of $\psi_{h, \delta}$ with 
  \begin{equation*}
    (x_{h, \delta}, y_{h, \delta} t_{h, \delta}) \in B_{r+1}(x_0, x_0, t_0).
  \end{equation*}
  This implies (after passing to a subsequence) that $(x_{h, \delta}, y_{h, \delta}, t_{h, \delta}) \to (x_\delta, y_\delta, t_\delta) \in  B_{r+ 1}(x_0, x_0, t_0)$ as $h \to 0$. Taking a further subsequence we can assume that $(x_\delta, y_\delta, t_\delta) \to (\overline{x}, \overline{x}, \overline{t})$ as $\delta \to 0$.
  Let $(\tilde{x}, \tilde{y}, \tilde{t}) \in \mathbb{R}^d \times \mathbb{R}^d \times [0, \infty)$ be arbitrary and choose $(\tilde{x}_h, \tilde{t}_h) \to (\tilde{x}, \tilde{t})$ such that
  \begin{equation}\label{eq:recovery_overline_u}
    \lim_{h \to 0} u_h(\tilde{x}_h, \tilde{t}_h) = \overline{u}(\tilde{x}, \tilde{t}).
  \end{equation}
  By the maximality property of $(x_{h, \delta}, y_{h, \delta}, t_{h, \delta})$ we have
  \begin{equation*}
    u_h(\tilde{x}_h, \tilde{t}_h) -  \frac{|\tilde{x}_h-\tilde{y}|^4}{\delta}-\phi(\tilde{y},\tilde{t}_h)  \leq u_h(x_{h, \delta}, t_{h, \delta}) -  \frac{|x_{h, \delta}-y_{h, \delta}|^4}{\delta}-\phi(y_{h, \delta}, t_{h, \delta}).
  \end{equation*}
  Passing to the limit, using~\eqref{eq:recovery_overline_u} on the left and the definition of $\overline{u}$ on the right yields
  \begin{equation*}
    \overline{u}(\tilde{x}, \tilde{t}) -  \frac{|\tilde{x}-\tilde{y}|^4}{\delta}-\phi(\tilde{y}, \tilde{t}) \leq\overline{u}(x_\delta, t_\delta) -  \frac{|x_\delta-y_\delta|^4}{\delta}-\phi(y_\delta, t_\delta)   ,
  \end{equation*}
  which shows that $(x_\delta, y_\delta, t_\delta)$ is indeed a maximum point of $\psi_\delta$. 
  
  If $(\overline{x}, \overline{t}) \neq (x_0, t_0)$ then the upper semi-continuity of $\overline u$ and the strict maximality of $(x_0, t_0)$ yield the contradiction
  \begin{align*}
    \overline{u}(x_0, t_0) - \phi(x_0, t_0) \leq \limsup_{\delta \to 0}\overline{u}(x_\delta, t_\delta)  - \frac{|x_\delta-y_\delta|^4}{\delta} - \phi(y_\delta,t_\delta) &\leq \overline{u}(\overline{x}, \overline{t}) - \phi(\overline{x}, \overline{t}) \\
    &<  \overline{u}(x_0, t_0) - \phi(x_0, t_0).
  \end{align*}
  This proves our claim. 

  \medskip

  Next, we claim that it is enough to prove
  \begin{equation}\label{eq:sub_solutio_delta}
    \partial_t \phi(y_\delta, t_\delta) \leq F^*(\nabla \phi(y_\delta, t_\delta), \nabla^2 \phi(y_\delta, t_\delta)).
  \end{equation}
  Indeed, after establishing~\eqref{eq:sub_solutio_delta} one can simply pass to the limit $\delta \to 0$ and use the upper semi-continuity of $F^*$ together with the continuity of $\partial_t \phi, \nabla \phi$ and $\nabla^2  \phi$ to conclude
  \begin{equation*}
    \partial_t \phi(y_0, t_0) = \lim_{\delta \to 0} \partial_t \phi(y_\delta, t_\delta) \leq \limsup_{\delta \to 0} F^*(\nabla \phi(y_\delta, t_\delta), \nabla^2 \phi(y_\delta, t_\delta)) \leq  F^*(\nabla \phi(x_0, t_0), \nabla^2 \phi(x_0, t_0)).
  \end{equation*}

  It remains to prove~\eqref{eq:sub_solutio_delta} for which we distinguish to cases:

  \emph{Case 1 -- Non Singular $\nabla_y \phi (y_{\delta}, t_{\delta}) \neq 0$:} In this case we can assume that $|\nabla \phi (y, t)| > a > 0$ for $y, t$ close to $y_\delta, t_\delta$. By~\eqref{eq:strong_consistency}, we know that the convergence speed of the consistency~\eqref{eq:consistency_upper} only depends on $\frac{1}{|\nabla \phi|}$ which is here uniformly bounded. Hence, the space consistency~\eqref{eq:consistency_upper} is even a space time consistency
  \begin{equation}\label{eq:space_time_consistency}
    \liminf_{y \to y_\delta, h \to 0, t \to t_\delta} \frac{\operatorname{\mathop \textup{med}}_h \phi(\cdot, t) - \phi(\cdot, t)}{h}(\tilde{y}) \leq F^*(\nabla \phi(y_\delta), \nabla^2 \phi(y_\delta)).
  \end{equation}

  Note that $(y_{h, \delta}, t_{h, \delta})$ is a maximum point of
  \begin{equation*}
    (y,t) \mapsto u(y - (y_{h, \delta} - x_{h, \delta}),t) - \frac{|x_{h, \delta}- y_{h, \delta}|^4}{\delta} - \phi(y, t).
  \end{equation*}
  This implies
  \begin{align*}
    \partial_t u_h(x_{h,\delta}, t_{h,\delta}) &=  \partial_t \phi(y_{h,\delta}, t_{h,\delta}),\\
    u(y - (y_{h, \delta} - x_{h, \delta}),t_{h,\delta}) &\leq \phi(y, t_{h,\delta}) + \xi_\delta \quad \quad \text{ for all } y \in \mathbb{R}^d,\\
    u(x_{h, \delta},t_{h,\delta}) &= \phi(y_{h, \delta}, t_{h,\delta}) + \xi_\delta,
  \end{align*}
  where we denoted $\xi_\delta :=  u(x_{h, \delta},t_{h,\delta}) - \phi(y_{h, \delta}, t_{h,\delta})$. By definition of our scheme~\eqref{def:semi_discrete_scheme}, the monotonicity~\eqref{eq:monotonicity_median}, and the invariance of constant translation~\eqref{eq:med_const} it holds
  \begin{align*}
    0 &= \partial_t u_h(x_{h, \delta}, t_{h, \delta}) - \frac{\operatorname{\mathop \textup{med}}_h u_h(\cdot - (y_{h, \delta} - x_{h, \delta}), t_{h, \delta}) - u_h(\cdot - (y_{h, \delta} - x_{h, \delta}), t_{h, \delta})}{h}(y_{h, \delta})\\
    & \geq \partial_t \phi(y_{h, \delta}, t_{h, \delta}) - \frac{\operatorname{\mathop \textup{med}}_h(\phi(\cdot, t_{h, \delta}) -\xi_\delta) - (\phi(\cdot, t_{h, \delta}) + \xi_\delta)}{h}(y_{h, \delta})\\
    &= \partial_t \phi(y_{h, \delta}, t_{h, \delta}) - \frac{\operatorname{\mathop \textup{med}}_h \phi(\cdot, t_{h, \delta})  - \phi(\cdot, t_{h, \delta}) }{h}(y_{h, \delta}).
  \end{align*}
  Taking the limes inferior over $h$ and utilizing~\eqref{eq:space_time_consistency} yields the desired~\eqref{eq:sub_solutio_delta}:
  \begin{align*}
    0 &\geq \liminf_{h \downarrow 0} \partial_t \phi(y_{h, \delta}, t_{h, \delta}) - \frac{\operatorname{\mathop \textup{med}}_h \phi(\cdot, t_{h, \delta})  - \phi(\cdot, t_{h, \delta}) }{h}(y_{h, \delta})\\
    &\geq \liminf_{y \to y_\delta, h \to 0, t \to t_\delta} \partial_t \phi(y, t) - \frac{\operatorname{\mathop \textup{med}}_h \phi(\cdot, t)  - \phi(\cdot, t) }{h}(y)\\
    &\geq \partial_t \phi(y_\delta, t_\delta) - F^*(\nabla \phi(y_\delta), \nabla^2 \phi(y_\delta)).
  \end{align*}
  
  \emph{Case 2 -- Singularity $\nabla_y \phi (y_{\delta}, t_{\delta}) = 0$:}
  As $(x_{ \delta}, y_{ \delta}, t_{ \delta})$ is a local maximum point of $\psi_{ \delta}$ we have $0 = \nabla_y \psi_{ \delta}(x_{ \delta}, y_{ \delta}, t_{ \delta})$ and $0 \geq  \nabla^2_y \psi_{ \delta}(x_{ \delta}, y_{ \delta}, t_{ \delta})$ which imply
  \begin{equation}\label{eq:first_order_after_limit}
    \nabla_y \phi(y_{ \delta}, t_{ \delta}) = \frac{4 (x_{ \delta}- y_{\delta})|x_{ \delta}- y_{\delta}|^2}{\delta}
  \end{equation}
  and 
  \begin{equation}\label{eq:second_order_after_limit}
    \nabla^2_y \phi(y_{ \delta}, t_{\delta}) \geq - \frac{4 |x_{ \delta}- y_{\delta}|^2}{\delta}\mathop{Id} - \frac{8 (x_{ \delta}- y_{h,\delta})\otimes (x_{ \delta}- y_{\delta})}{\delta}.
  \end{equation}
  Hence, our case $\nabla_y \phi (y_{ \delta}, t_{ \delta}) = 0$ yields first $x_{\delta} = y_{ \delta}$ and then
  \begin{equation*}
    \nabla_y^2 \phi(y_{ \delta}, t_{\delta}) \geq 0.
  \end{equation*} 
  Define the function $\tilde{\phi}(x, t) := \frac{|x- y_{h, \delta}|^4}{\delta} + \phi(y_{h, \delta}, t)$. Recall that $(x_{h, \delta}, t_{h, \delta})$ is a local maximum point of $\psi_{h, \delta}(\cdot, y_{h,\delta}, \cdot ) = u_h -\tilde{\phi}$. Thus, we get
  \begin{align*}
    \partial_t u_h(y_{h, \delta}, t_{h,\delta}) &= \phi(y_{h, \delta}, t_{h,\delta}),\\
    u_h(x,t) &\leq \frac{|x- y_{h, \delta}|^4}{\delta} + \phi(y_{h, \delta}, t) + \xi_{\delta} \quad \quad \text{ for all } (x,t) \in \mathbb{R}^d \times [0,\infty),\\
    u_h(x_{h, \delta}, t_{h, \delta}) &=  \frac{|x_{h, \delta}- y_{h, \delta}|^4}{\delta} + \phi(y_{h, \delta}, t_{h,\delta})  +\xi_{\delta},
  \end{align*}
  where we denoted $\xi_{\delta} := u_h(x_{h, \delta}, t_{h, \delta}) - \frac{|x_{h, \delta}- y_{h, \delta}|^4}{\delta} -\phi(y_{h, \delta}, t_{h, \delta})$. The trick is now that the space variable $x$ and time variable $t$ are split in $\frac{|x- y_{h, \delta}|^4}{\delta} + \phi(y_{h, \delta}, t) $, so we can pull the $\phi(y_{h, \delta}, t) $ term out of the median. To be more precise, similar to the first case we compute
  \begin{align*}
    0 &= \partial_t u_h(x_{h, \delta}, t_{h, \delta}) - \frac{\operatorname{\mathop \textup{med}}_h u_h(\cdot, t_{h, \delta}) - u_h(\cdot, t_{h, \delta})}{h}(x_{h, \delta})\\
    & \geq \partial_t \phi(x_{h, \delta}, t_{h, \delta}) - \frac{\operatorname{\mathop \textup{med}}_h(\frac{|\cdot- y_{h, \delta}|^4}{\delta} + \phi(y_{h, \delta},  t_{h, \delta}) + \xi_{\delta}) - (\frac{|\cdot- y_{h, \delta}|^4}{\delta} + \phi(y_{h, \delta},  t_{h, \delta}) + \xi_{\delta})}{h}(x_{h, \delta})\\
    &= \partial_t \phi(x_{h, \delta}, t_{h, \delta}) - \frac{\operatorname{\mathop \textup{med}}_h \frac{|\cdot- y_{h, \delta}|^4}{\delta}  - \frac{|\cdot- y_{h, \delta}|^4}{\delta} }{h}(x_{h, \delta}).
  \end{align*}
  Now we simply take the limes inferior in $h$ again and use the only in space dependent consistency~\eqref{eq:consistency_upper} together with 
  \begin{equation}\label{eq:vanish_diff_x_y}
    \lim_{h \to 0} x_{h, \delta} - y_{h, \delta} = x_\delta - y_\delta = 0, 
  \end{equation}
  and 
  \begin{equation}\label{eq:F_mon_happy}
    F^*(0,0) \stackrel{\eqref{eq:first_order_after_limit}}{=} F^*(\nabla_y \phi(y_\delta, t_\delta), 0) \stackrel{\eqref{eq:second_order_after_limit}}{\leq} F^*(\nabla_y \phi(y_\delta, t_\delta), \nabla^2_y \phi(y_\delta, t_\delta))
  \end{equation}
  to conclude
  \begin{align*}
    0 &\geq \liminf_{h \to 0} \partial_t \phi(x_{h, \delta}, t_{h, \delta}) - \frac{\operatorname{\mathop \textup{med}}_h \frac{|\cdot|^4}{\delta}  - \frac{|\cdot|^4}{\delta} }{h}(x_{h, \delta} - y_{h, \delta})\\
    &\stackrel{\eqref{eq:vanish_diff_x_y}}{\geq} \liminf_{z \to 0, h \to 0}\partial_t \phi(x_{h, \delta}, t_{h, \delta}) - \frac{\operatorname{\mathop \textup{med}}_h \frac{|\cdot|^4}{\delta}  - \frac{|\cdot|^4}{\delta} }{h}(z)\\
    &\stackrel{\eqref{eq:consistency_upper}}{\geq} \partial_t \phi(x_{ \delta}, t_{ \delta}) - F^*(0, 0)\\
    &\stackrel{\eqref{eq:F_mon_happy}}{\geq} \partial_t \phi(y_{ \delta}, t_{ \delta}) - F^*(\nabla_y \phi(y_\delta, t_\delta), \nabla^2_y \phi(y_\delta, t_\delta)).
  \end{align*}
  This finishes the second case and with it the whole proof.

\emph{Step 1 -- The $\varepsilon$-regularized case}: We claim the lemma is true when we use the energy and median scheme with the $\varepsilon$ regularized modulus, i.e.,
    \begin{equation}\label{eq:energy_mono_eps}
        \frac{d}{dt} \frac{1}{\sqrt{h}} \int_{\mathbb{R}^d}\int_{\mathbb{R}^d} G_{h,R}(x,y) \Psi_\varepsilon(u_{h,\varepsilon, R}(x) - u_{h, \varepsilon, R}(y)) \, dy \, dx   \leq 0.
    \end{equation}

    Indeed, computing the time derivative yields using non-local integration by parts
    \begin{multline*}
         \frac{d}{dt} \frac{1}{\sqrt{h}} \int_{\mathbb{R}^d}\int_{\mathbb{R}^d} G_{h,R}(x,y) \Psi_\varepsilon(u_{h,\varepsilon, R}(x) - u_{h, \varepsilon, R}(y)) \, dy \, dx \\
         =  \int_{\mathbb{R}^d} H_{h, \varepsilon, R}(x)  \frac{\operatorname{\mathop \textup{med}}_{h,\varepsilon, R} u_{h,\varepsilon,R}(x) - u_{h, \varepsilon, R}(x)}{h}\, dx.
    \end{multline*}
    We now claim that for any $x \in \mathbb{R}^d$ we have the pointwise estimate
    \begin{equation}\label{eq:different_signs_eps_h_r}
        H_{h, \varepsilon, R}(x)  \big(\operatorname{\mathop \textup{med}}_{h,\varepsilon, R} u_{h,\varepsilon,R}(x) - u_{h, \varepsilon, R}(x)\big) \leq 0
    \end{equation}
    
To see that~\eqref{eq:different_signs_eps_h_r} is indeed true, fix $x \in \mathbb{R}^d$. Assume (the other case is similar) that 
\begin{equation*}
  \operatorname{\mathop \textup{med}}_{h,\varepsilon,R} u_{h,\varepsilon,R}(x) - u_{h,\varepsilon,R}(x) \geq 0.
\end{equation*}
Then, it has to hold $H_{h,\varepsilon,R}(x) \leq 0$ as otherwise one would have the contradiction (using the monotonicity of $\psi_\varepsilon$with $\operatorname{\mathop \textup{med}}_{h, \varepsilon, R} u_{h,\varepsilon,R}(x) > u_{h,\varepsilon,R}(x)$)
\begin{align*}
  0 < H_{h,\varepsilon,R}(x)& = \frac{2}{\sqrt{h}} \int_{\mathbb{R}^d} G_{h, R}(x,y) \psi_\varepsilon(u_{h,\varepsilon,R}(x) - u_{h,\varepsilon,R}(y)) \,dy\\
  &\leq \frac{2}{\sqrt{h}} \int_{\mathbb{R}^d} G_{h,R}(x,y)  \psi_\varepsilon(\operatorname{\mathop \textup{med}}_{h, \varepsilon, R} u_{h,\varepsilon,R}(x) - u_{h,\varepsilon,R}(y)) \,dy =0.
\end{align*} 
Here we used in the last equation the Euler-Lagrange equation, resulting from the differentiability of $\Psi_\varepsilon$ in~\eqref{eq:ELI1} and~\eqref{eq:ELI2}.
Thus,~\eqref{eq:different_signs_eps_h_r} has to hold, which in turn implies~\eqref{eq:energy_mono_eps}.

\emph{Step 2 -- Taking the limit $\varepsilon \downarrow 0$}: Integrating~\eqref{eq:energy_mono_eps} in time yields for any $0 \leq t_1 \leq t_2 < \infty$
\begin{align}
     \frac{1}{\sqrt{h}} \int_{\mathbb{R}^d}\int_{\mathbb{R}^d} G_{h,R}(x,y) \Psi_\varepsilon(u_{h,\varepsilon, R}(x, t_1) - u_{h, \varepsilon, R}(y, t_1)) \, dy \, dx \label{eq:integrated_mono}\\
     \geq  \frac{1}{\sqrt{h}} \int_{\mathbb{R}^d}\int_{\mathbb{R}^d} G_{h,R}(x,y) \Psi_\varepsilon(u_{h,\varepsilon, R}(x, t_2) - u_{h, \varepsilon, R}(y, t_2)) \, dy \, dx. \nonumber
\end{align}
Since $\lim_{\varepsilon \downarrow 0} u_{h,\varepsilon,R}  = u_{h,R}$ uniformly on $\mathbb{R}^d \times \{t = t_i\}$, $i = 1,2$, by Lemma~\ref{lem:uniform_convergenceReps}, $\Psi_\varepsilon \rightarrow |\cdot|$ uniformly, and $u_{h,\varepsilon,R}$ has exponential decay by Lemma~\ref{lem:exp_decay}, we have
\begin{multline*}
    \lim_{\varepsilon \downarrow 0} \int_{\mathbb{R}^d}\int_{\mathbb{R}^d} G_{h,R}(x,y) \Psi_\varepsilon(u_{h,\varepsilon, R}(x, t_i) - u_{h, \varepsilon, R}(y, t_i)) \, dy \, dx \\
    = \int_{\mathbb{R}^d}\int_{\mathbb{R}^d} G_{h,R}(x,y) |u_{h, R}(x, t_i) - u_{h, R}(y, t_i)| \, dy \, dx.
\end{multline*}
Similarly, using $\lim_{R\uparrow \infty} u_{h, R} = u_{h}$ uniformly by Lemma~\ref{lem:uniform_convergenceReps} and $u_{h,R}$ is constant outside a large ball $B_L(0)$ by Lemma~\ref{lem:const_outside_ball} we have
\begin{equation*}
    \lim_{R\uparrow \infty} \int_{\mathbb{R}^d}\int_{\mathbb{R}^d} G_{h,R}(x,y) |u_{h, R}(x, t_i) - u_{h, R}(y, t_i)| \, dy \, dx\\
     = \int_{\mathbb{R}^d}\int_{\mathbb{R}^d} G_{h}(x,y) |u_{h}(x, t_i) - u_{h}(y, t_i)| \, dy \, dx.
\end{equation*}
Hence, in~\eqref{eq:integrated_mono} we can pass first to the limit $\varepsilon \to 0$ and then $R \to \infty$ which concludes the proof of this lemma.

\emph{Step 1 -- Introducing $\delta$:} Take $\eta_\delta \in C^1(\mathbb{R}, \mathbb{R})$ with $\eta_\delta \to |\cdot|$ uniformly as $\delta \downarrow 0$ and $\eta'(s) = \mathop \mathrm{sign}(s)$ for $|s| > \delta$ as well as $|\eta'| \leq 1$. For simpler readability we drop the $h,\varepsilon,R$ subscript for $H_{h, \varepsilon, R}$, $u_{h,\varepsilon,R}$ and $\operatorname{\mathop \textup{med}}_{h,\varepsilon,R}$, i.e., we write $H$ for $H_{h, \varepsilon, R}$, $u$ for $u_{h,\varepsilon,R}$ and $\operatorname{\mathop \textup{med}}$ for $\operatorname{\mathop \textup{med}}_{h,\varepsilon,R}$. Then the derivative of the $L^1$-norm with the modulus replaced by $\eta_\delta$ computes as (suppressing the $t$-dependence on the right-hand side for notational convenience)
  \begin{align*}
    &\frac{d}{dt} \int_{\mathbb{R}^d} \eta_\delta (H(x,t)) \, dx \\
    &= \int_{\mathbb{R}^d} \eta_\delta '(H(x)) \int_{\mathbb{R}^d} \frac{1}{\sqrt{h}} G_{h,R}(x,y) \psi_\varepsilon'\big(u(x) - u(y)\big) \left(\frac{\operatorname{\mathop \textup{med}} u(x) - u(x)}{h} - \frac{\operatorname{\mathop \textup{med}} u(y) - u(y)}{h} \right)\!\! \, dy \, dx\\

  \end{align*}

  \begin{equation}\label{eq:sign_of_integrand}

  \end{equation}
  and if $|H(x,t)| = 0 $ then $\operatorname{\mathop \textup{med}}  u (x) = u(x)$.

  |H_{h, \varepsilon, R}(x, t = 0) | &\leq \frac{1}{\sqrt{h}} \int_{\mathbb{R}^d} G_h(x,y) |\psi_\varepsilon(u_0(x) - u_0(y))|\,dy \\
  &\leq \frac{1}{\sqrt{h}} \int_{\mathbb{R}^d} G_h(x,y) \mathbbold{1}_{\{x\in B_L(0) \vee y \in B_L(0)\}} \,dy \\
  &=: g(x).
  \end{align*}
  As
  \begin{equation*}
    \int_{\mathbb{R}^d} g(x) \,dx \leq \frac{2}{\sqrt{h}} \int_{B_L(0)} \int_{\mathbb{R}^d} G_h(x,y)  \,dy \, dx \leq \frac{2}{\sqrt{h}} |B_L(0)|,
  \end{equation*} 
  we have that $g$ is integrable. Thus, the dominated convergence theorem yields
  \begin{equation}\label{eq:limb}
    \lim_{R \uparrow \infty} \lim_{\varepsilon \downarrow 0}  \int_{\mathbb{R}^d} |H_{h, \varepsilon, R}(x, t = 0) |\,dx = \int_{\mathbb{R}^d} |H_{h}(x, t= 0) |\,dx.
  \end{equation}
  By Lemma~\ref{lem:uniform_convergenceReps} we also have the uniform convergence on $\mathbb{R}^d \times [0, T]$
  \begin{equation}\label{eq:limc}
    \lim_{h \downarrow 0} \lim_{R \uparrow \infty} \lim_{\varepsilon \downarrow 0} u_{h,\varepsilon,R} = u.
  \end{equation}
  Hence, there exists for every $h \in (0,1]$ both $\varepsilon_h \in (0,1]$ and $R_h > 1$ such that by~\eqref{eq:lima}
  \begin{equation}\label{eq:delta_2_bound}
     \int_{\mathbb{R}^d} H_h u_h \, dx \leq \int_{\mathbb{R}^d} H_{h, \varepsilon_h, R_h}\, u_{h,\varepsilon_h, R_h} \,dx + \delta
  \end{equation}
  and by~\eqref{eq:limb}
  \begin{equation}\label{eq:init_bound_eRh}
    \sup_{h, \varepsilon_h, R_h} \int_{\mathbb{R}^d} |H_{h, \varepsilon_h, R_h}(x, t = 0) |\,dx \leq \sup_{h}  \int_{\mathbb{R}^d} |H_{h}(x, t= 0) |\,dx + \delta
  \end{equation}
  and by~\eqref{eq:limc}
  \begin{equation}\label{eq:limd}
    \lim_{h \downarrow 0} u_{h,\varepsilon_h, R_h} = u \quad \quad \text{ uniformly in } \mathbb{R}^d \times [0,T].
  \end{equation}
                                                              
                
                                                                                                                                                                                              
                                                                                                                      
                                             
              
                                                            
                    
                                                                                                                                                                                   
                  
                                                                          
                    
                                                                                                                                                                                                                                                                     
                  
  The bound~\eqref{eq:init_bound_eRh}, combined with the properties~\eqref{def:well_prepared_init} of the initial condition, allows applying Lemma~\ref{lem:limit_of_Hh} to deduce the existence of $\sigma$ with $|\sigma|\leq 2c_0$ such that
  \begin{equation}\label{eq:lime}
  H_{h, \varepsilon_h, R_h} \, dx \, dt \xrightharpoonup{*} \nabla \cdot \sigma \,dx\,dt  \quad \quad \text{ weakly-$*$ as measures as } h \downarrow 0.
  \end{equation} 
  Combing the weak-$*$ convergence~\eqref{eq:lime} of $H_{h, \varepsilon_h, R_h}$ with the uniform convergence~\eqref{eq:limd} of $u_{h,\varepsilon_h, R_h}$ lets us pass to the limit to obtain:
  \begin{align*}
    \limsup_{h \downarrow 0} \int_0^T E_h(u_h) \, dt &= \limsup_{h \downarrow 0} \int_0^T \int u_h H_h \,dx \, dt\\
    &\leq \limsup_{h \downarrow 0} \int_0^T \int u_{h, \varepsilon_h, R_h} H_{h, \varepsilon_h, R_h} \,dx \, dt + T\delta\\
    &= \int_0^T \int u \nabla \cdot \sigma \, dx\,dt + T\delta \\
    &= - \int_0^T \int \nabla u \cdot \sigma \, dx\,dt + T\delta \\
    &\leq 2c_0 \int_0^T \int |\nabla u| \, dx \,dt + T\delta .
  \end{align*}
  Taking the limit $\delta \downarrow 0$ yields the desired upper bound
  \begin{equation*}
    \limsup_{h \downarrow 0} \int_0^T E_h(u_h) \, dt \leq 2c_0 \int_0^T \int |\nabla u| \,dt.
  \end{equation*}
  For the lower bound fix first $z$ and take a smooth test function $\xi$ with $|\xi| \leq 1$ to compute 
  \begin{equation*}
    \int |u_h(x) - u_h(x - \sqrt{h}z)| \,dx \geq  \int (u_h(x) - u_h(x - \sqrt{h}z)) \xi(x) \,dx = \int u_h(x) (\xi(x+ \sqrt{h}z) - \xi(x)) \, dx.
  \end{equation*}
  Hence, it holds
\begin{align*}
   \liminf_{h \downarrow 0} \int |u_h(x) - u_h(x - \sqrt{h}z)| \,dx \geq \sup_{|\xi|\leq 1} \int u(x) \nabla \xi(x) \cdot z = \int |\nu(x) \cdot z| |\nabla u|.
\end{align*}
It follows by Fatou's Lemma and using the normal $\nu(x) = \frac{\nabla u(x)}{|\nabla u(x)|}$, where it exists, that
\begin{align*}
  \liminf_{h \downarrow 0} \int_0^T E_h(u_h) \, dt &= \liminf_{h \downarrow 0} \int_0^T \int_{\mathbb{R}^d} G_1(z) \int |u_h(x) - u_h(x - \sqrt{h}z)| \,dx \,dz \, dt \\
  &\geq \int_0^T \int_{\mathbb{R}^d} \int_{\mathbb{R}^d} G_1(z) |\nu(x) \cdot z| \, dz |\nabla u| \, dx \, dt.
\end{align*}
The result follows now as 
\begin{equation*}
  \int_{\mathbb{R}^d} G_1(z) |\nu(x) \cdot z| \, dz = 2c_0
\end{equation*}
for any $x$.

\label{lem:uniqueness_median}
  Let $f: \mathbb{R}^d \to \mathbb{R}^d$ by continuous and bounded. There exists a unique generalized median and it satisfies the following two Euler-Lagrange inequalities
  \begin{multline}\label{eq:ELI1}
    - \int_{\mathbb{R}^d} K(x,y) \psi(\operatorname{\mathop \textup{med}}_{K,\Psi} f(x) - f(y))\mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{K,\Psi} f(x) < f(y)\}} \, dy  \\
     \leq \int_{\mathbb{R}^d} K(x,y) \psi^*(\operatorname{\mathop \textup{med}}_{K,\Psi} f(x) - f(y))\mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{K, \Psi} f(x) \geq f(y)\}} \, dy, 
     \end{multline}
     and 
     \begin{multline} \label{eq:ELI2}
           - \int_{\mathbb{R}^d} K(x,y) \psi_*(\operatorname{\mathop \textup{med}}_{K,\Psi} f(x) - f(y))\mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{K,\Psi} f(x) \leq f(y)\}} \, dy\\
    \geq \int_{\mathbb{R}^d} K(x,y) \psi(\operatorname{\mathop \textup{med}}_{K,\Psi} f(x) - f(y))\mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{K,\Psi} f(x) > f(y)\}} \, dy.
     \end{multline}
  If $f_1, f_2: \mathbb{R}^d \rightarrow \mathbb{R}$ are continuous and bounded satisfying $f_1 \leq f_2$ then it holds
  \begin{equation}\label{eq:monotonicity_generalized_median}
    \operatorname{\mathop \textup{med}}_{K, \Psi} f_1 \leq \operatorname{\mathop \textup{med}}_{K, \Psi} f_2.
  \end{equation}

<!-- proof-9: body_md -->
We start with existence of a generalized median. As $\Psi$ is convex and $\psi$ is antisymmetric and $\psi \neq \{0\}$ on $\mathbb{R}\setminus 0$ it holds that $\Psi$ is minimal at $0$. Thus,
  \begin{equation*}
    F(c) := \int_{\mathbb{R}^d} K(x, y) \Psi(c-f(y)) \, dy
  \end{equation*}
  is also convex and as $f$ is continuous and $K(z)\,dz < \infty$ it also holds
  \begin{equation*}
    \lim_{|c|\rightarrow \infty} F(c) = \infty.
  \end{equation*}
  So $F(c)$ is convex and coercive implying there exists a minimizer.

  Next, we prove the Euler-Lagrange inequalities. As $F$ is convex we use the subdifferential for the optimality condition 
  \begin{equation*}
    0\in \partial F(c) = \partial\left(\int_{\mathbb{R}^d} K(x, y)  \Psi(\cdot - f(y)) \,dy\right)(c).
  \end{equation*}
  By differentiability of $\Phi$ except at $0$ we have
  \begin{equation*}
    \partial \Psi(c) = \Psi'(c) = \psi(c) \quad \text{ on } \mathbb{R} \setminus\{0\} \quad \quad \text{ and }  \quad \quad \partial \Psi(0) = [\lim_{c \rightarrow 0^-} \psi(c), \lim_{c \rightarrow 0^+} \psi(c)].
  \end{equation*}
  Hence, as $0 \in \partial F(\operatorname{\mathop \textup{med}}_{K, \Psi} f(x))$ we have
  \begin{equation*}
    \int_{\mathbb{R}^d} K(x, y) \psi_*(\operatorname{\mathop \textup{med}}_{K, \Psi} f(x)-f(y)) \,dy \leq 0 \leq \int_{\mathbb{R}^d} K(x, y) \psi^*(\operatorname{\mathop \textup{med}}_{K, \Psi} f(x)-f(y)) \, dy.
  \end{equation*}
  Using $\psi_* = \psi = \psi^*$ on $\mathbb{R} \setminus \{0\}$ yields the Euler-Lagrange inequalities~\eqref{eq:ELI1} and~\eqref{eq:ELI2}.

  Next we prove the monotonicity~\eqref{eq:monotonicity_generalized_median}. Assume~\eqref{eq:monotonicity_generalized_median} is false. Then there exist continuous continuous, bounded functions $f_1 \geq f_2$ and a point $x \in \mathbb{R}^d$ wih two generalized medians $c_1:= \operatorname{\mathop \textup{med}}_{K, \Psi} f_1(x)$ and $c_2:= \operatorname{\mathop \textup{med}}_{K, \Psi} f_2(x)$ with $c_1 < c_2$. One easily checks that $c_i \in [\inf_{y \in \mathop \textup{supp} K(x, \cdot)} f_i(y), \sup_{y \in \mathop \textup{supp} K(x, \cdot)} f_i(y)]$ for $i =1,2$. If $c_1 = \inf_{y \in \mathop \textup{supp} K(x,\cdot)} f_1(y)$ then by the definition~\eqref{def:generalized_median} of the generalized median it already has to hold that $f_1$ is constant in $\mathop \textup{supp} K(x,\cdot)$ and equal to $c_1$. But then it has to hold $c_1 = f_1(x) \geq \sup_{y \in \mathop \textup{supp} K(x,\cdot)} f_2(y) \geq c_2$, a contradiction to $c_1 < c_2$. So, by continuity of $f_1$ there must exist $y^* \in \mathop \textup{supp} K(x, \cdot)$ such that $c_1 < f_1(y^*) < c_2$. Because of $f_1 \geq f_2$ it also holds $c_2 > f_1(y^*) \geq f_2(y^*)$. Now, by continuity of $f_1$ and $f_2$ there exists an $r > 0$ such that
  \begin{equation*}
    B_r(y^*) \subset \{y: c_1 < f_1(y)\} \cap \{y: c_2 > f_2(y)\}.
  \end{equation*} 
 Applying the Euler-Lagrange inequality~\eqref{eq:ELI1} to $c_1$ and~\eqref{eq:ELI2} to $c_2$ as well as using the monotonicity of $\psi$ yields the contradiction
  \begin{align*}
    &- \int_{\mathbb{R}^d} K(x,y) \psi(c_1 - f_1(y))\mathbbold{1}_{\{c_1< f_1(y)\}} \, dy\\
     &\stackrel{\eqref{eq:ELI1}}{\leq} \int_{\mathbb{R}^d} K(x,y) \psi^*(c_1 - f_1(y))\mathbbold{1}_{\{c_1\geq f_1(y)\}} \, dy\\
    &\stackrel{\stackrel{c_1 < c_2}{ f_1 \geq f_2}}{\leq} \int_{\mathbb{R}^d} K(x,y) \psi(c_2 - f_2(y))\mathbbold{1}_{\{c_2 > f_2(y)\}} \, dy - \int_{\mathbb{R}^d} K(x,y) \psi^*(c_1 - f_1(y))\mathbbold{1}_{\{c_1 < f_1(y) \wedge c_2 > f_2(y)\}} \, dy\\
     &\stackrel{\eqref{eq:ELI2}}{\leq} -\int_{\mathbb{R}^d} K(x,y) \psi_*(c_2 - f_2(y))\mathbbold{1}_{\{c_2 \leq f_2(y)\}} \, dy - \int_{B_r(y^*)} K(x,y) \psi^*(c_1 - f_1(y)) \, dy\\
     &\stackrel{c_1 < c_2, f_1 \geq f_2}{<} -\int_{\mathbb{R}^d} K(x,y) \psi(c_1 - f_1(y))\mathbbold{1}_{\{c_1 < f_1(y)\}} \, dy.
  \end{align*}
  We conclude that the generalized median is a monotone operator. This immediately yields the uniqueness of the generalized median which finishes this proof.

\label{lem:contraction}
  The generalized median operator~\eqref{def:generalized_median} satisfies the following contraction property for any continuous, bounded functions $f,g$
  \begin{equation}\label{eq:contraction}
    |\operatorname{\mathop \textup{med}}_{K, \Psi} f(x) - \operatorname{\mathop \textup{med}}_{K, \Psi} g(x)| \leq \max_{y \in \mathop \textup{supp} K(x, \cdot)} |f(y) - g(y)|.
  \end{equation}
  Furthermore, it holds for any $x,y \in \mathbb{R}^d$ and $f \in C^{0,1}(\mathbb{R}^d)$
  \begin{equation}\label{eq:lipschitz_mon}
  |\operatorname{\mathop \textup{med}}_{K, \Psi} f(x) - \operatorname{\mathop \textup{med}}_{K, \Psi} f(y)| \leq \operatorname*{Lip}(f) |x-y|.
  \end{equation}

One easily checks~\eqref{eq:homogenity}. For any $x \in \mathbb{R}^d$ we simply have $f \leq g + \|f-g\|_{L^\infty(\mathop \textup{supp}(K(x, \cdot)))}$ on $\mathop \textup{supp}(K(x, \cdot))$, which immediately yields by~\eqref{eq:monotonicity_generalized_median} and~\eqref{eq:homogenity} that
  \begin{equation*}
    \operatorname{\mathop \textup{med}}_{K, \Psi} f(x) \leq \operatorname{\mathop \textup{med}}_{K, \Psi} (g + \|f-g\|_{L^\infty(\mathop \textup{supp}(K(x, \cdot)))})(x) = \operatorname{\mathop \textup{med}}_{K, \Psi} g(x) + \|f-g\|_{L^\infty(\mathop \textup{supp}(K(x, \cdot)))}.
  \end{equation*}
  Changing the roles of $f$ and $g$ yields the other inequality which proves~\eqref{eq:contraction}.

It remains to prove~\eqref{eq:lipschitz_mon}. Let $x,y \in \mathbb{R}^d$ be fixed. Define $g(z) = f(z +(y-x))$ such that it holds
$g(x) = f(y)$. Then it holds
\begin{align*}
  \operatorname{\mathop \textup{med}}_{K, \Psi} g(x) &\in \operatorname*{\mathop \textup{arg\,min}}_{c \in \mathbb{R}} \int_{\mathbb{R}^d} K(x-z) \Psi(c - g(z))\,dz \\
  &= \operatorname*{\mathop \textup{arg\,min}}_{c \in \mathbb{R}} \int_{\mathbb{R}^d} K(x-z) \Psi(c - f(z +(y-x)))\,dz \\
  &=\operatorname*{\mathop \textup{arg\,min}}_{c \in \mathbb{R}} \int_{\mathbb{R}^d} K(y-\tilde{z}) \Psi(c - f(\tilde{z}))\,d\tilde{z},
\end{align*}
which implies $\operatorname{\mathop \textup{med}}_{K, \Psi} g(x) = \operatorname{\mathop \textup{med}}_{K, \Psi} f(y)$ by the uniqueness of the generalized median of Lemma~\ref{lem:uniqueness_median}. Consequently, applying~\eqref{eq:contraction} to the pair $f,g$ lets us deduce
\begin{align*}
  |\operatorname{\mathop \textup{med}}_{K, \Psi} f(x) - \operatorname{\mathop \textup{med}}_{K, \Psi} f(y)| &= |\operatorname{\mathop \textup{med}}_{K, \Psi} f(x) - \operatorname{\mathop \textup{med}}_{K, \Psi} g(x)|
  \\& \leq \|f - f(\cdot + (y-x))\|_\infty \leq \operatorname*{Lip}(f) |x-y|,
\end{align*}
which is exactly the claimed inequality~\eqref{eq:lipschitz_mon}.

\label{lem:cuttable}
  For any kernel $K$ satisfying~\eqref{def:K_kernel}
  define the cut-off kernel
  \begin{equation*}
    K_R(x,y) = \mathbbold{1}_{|x-y|<R} K(x,y).
  \end{equation*}
  Then, for sufficiently large $R > 0$, any Lipschitz continuous functions $f$ and any $x \in \mathbb{R}^d$ it holds
  \begin{equation}\label{eq:cuttable}
    |\operatorname{\mathop \textup{med}}_{K_R, \Psi} f(x) - \operatorname{\mathop \textup{med}}_{K, \Psi} f(x)| \leq 2\operatorname*{Lip}(f)\left( C(R) \right)^{\frac{1}{2d}}.
  \end{equation}
  Here $C(R)$ is a constant only depending on $R$ and the kernel $K$ given by
  \begin{equation*}
    C(R) := \int_{\{z > R\}} K(0,z)\, dz,
  \end{equation*}
  which in particular satisfies
  \begin{equation*}
    C(R) \to 0 \quad\quad \text{ as } R \uparrow \infty.
  \end{equation*}

<!-- proof-11: body_md -->
If $K$ has compact support there is nothing to prove. So by~\eqref{def:K_kernel} we know
  \begin{equation}\label{eq:inf_kernel_pos}
    \inf_{|z| \in B_R(0)} K(z) > 0 \quad\quad\text{ for any } R > 0.
  \end{equation}
  We argue by contradiction: Assume that~\eqref{eq:cuttable} is false, i.e., there exist $f$ and $x \in \mathbb{R}^d$ such that for every $R > 0$ and $h > 0$, w.l.o.g. (the other case is treated similarly), 
  \begin{equation}\label{eq:not_cuttable}
    \operatorname{\mathop \textup{med}}_{K, \Psi} f(x)  - \operatorname{\mathop \textup{med}}_{K_R, \Psi} f(x) >  2\operatorname*{Lip}(f) \left( C(R) \right)^{\frac{1}{2d}} =: \tilde{C} > 0.
  \end{equation} 
  As $f$ is continuous there exists a $z_0 \in \mathbb{R}^d$ such that $\operatorname{\mathop \textup{med}}_{K, \Psi} f(x) = f(x + z_0)$. Let $R_0> 0$ be the fixed radius with
  \begin{equation*}
    \int_{\{|z| > R_0\}} K(0,z) \,dz = \frac{\int_{\mathbb{R}^d}K(0,z)\, dz}{2}.
  \end{equation*}
  Then we can assume that $|z_0| \leq R_0$ as otherwise the Euler-Lagrange inequality~\eqref{eq:ELI1} would be violated. Similarly, there is $|z_R| \leq R_0$ (with the same constant $R_0$ as $K_R$ has its mass more centered than $K$) such that $\operatorname{\mathop \textup{med}}_{K, \Psi} f(x) - \tilde{C} > \operatorname{\mathop \textup{med}}_{K_R, \Psi} f(x) = f(x + z_R)$. By continuity of $f$ there is a $z^*$ on the segment $[z_0, z_R] \subset B_{C\sqrt{h}}(0)$ such that $f(y^*) = \operatorname{\mathop \textup{med}}_{K, \Psi} f(x) - \frac{\tilde{C}}{2}$ for $y^*=x + z^*$. Now define the radius $r = \frac{\tilde{C}}{2 \operatorname*{Lip}(f)}= \left(C(R) \right)^{\frac{1}{2d}}$. Then it holds 
  \begin{equation}\label{eq:ball_eps_arg}
     B_r(y^*) \subset \{y: \operatorname{\mathop \textup{med}}_{K_R, \Psi} f(x) < f(y) < \operatorname{\mathop \textup{med}}_{K, \Psi} f(x)\}.
  \end{equation}
  Therefore, by assumption, we have
  \begin{equation} \label{eq:f_g_prop_eps}
    \{y: \operatorname{\mathop \textup{med}}_{K_R, \Psi} f(x) \geq f(y) \} \subset \{y: \operatorname{\mathop \textup{med}}_{K, \Psi} f(x) \geq f(y)\},
  \end{equation}
  and by~\eqref{eq:ball_eps_arg}
  \begin{equation}\label{eq:ball_prop_eps}
    B_r(y^*) \subset \{y: \operatorname{\mathop \textup{med}}_{K, \Psi}f(y) \geq f(y)\}\setminus \{y: \operatorname{\mathop \textup{med}}_{K_R, \Psi}f(y) \geq f(y)\}.
  \end{equation}
  Hence, our construction~\eqref{eq:ball_prop_eps} and the Euler-Lagrange inequality~\eqref{eq:ELI1} for the cut-off kernel imply
  \begin{align}
    -\int_{\mathbb{R}^d} &K_R(x,y) \psi(\operatorname{\mathop \textup{med}}_{K_R, \Psi} f(x) - f(y)) \mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{K_R, \Psi} f(x) < f(y)\}} \,dy \nonumber\\
    &\stackrel{\eqref{eq:ELI1}}{\leq} \int_{\mathbb{R}^d} K_R (x,y) \psi^*( \operatorname{\mathop \textup{med}}_{K_R, \Psi} f(x) - f(y) )\mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{K_R, \Psi} f(x) \geq f(y)\}} \, dy \nonumber\\
    & \stackrel{K_R \leq K}{\leq}\int_{\mathbb{R}^d} K (x,y) \psi^*( \operatorname{\mathop \textup{med}}_{K_R, \Psi} f(x) - f(y) )\mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{K_R, \Psi} f(x) \geq f(y)\}} \, dy\nonumber\\
    & \stackrel{\eqref{eq:ball_prop_eps}, \eqref{eq:f_g_prop_eps}}{\leq} \int_{\mathbb{R}^d} K (x,y) \psi( \operatorname{\mathop \textup{med}}_{K, \Psi} f(x) - f(y) )\mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{K, \Psi} f(x) > f(y)\}} \, dy- \int_{B_r(y^*)} K(x,y) \, dy . \label{eq:cont_up_to_constant}
  \end{align}
  As by construction $|z^*| \leq R_0$ and $r \leq 1$ we can bound the good, negative term from below by 
  \begin{align*}
  \int_{B_r(y^*)} K(x,y) \, dy &\geq |B_r(y^*)| \inf_{y \in B_r(y^*)} K(x,y)\\
  &\geq r^{d} \omega_d \inf_{z \in B_r(z^*)} K(0,z) \\
  &\geq \sqrt{C(R)} \omega_d \inf_{z \in B_{R_0+1}(0)} K(0,z).
  \end{align*}
  The infimum is positive due to~\eqref{eq:inf_kernel_pos}; thus, for $R$ big enough it holds $ \omega_d\inf_{z \in B_{R_0+1}(0)} K(0,z) \geq 2\sqrt{C(R)}$ and in this case
  \begin{equation*}
    \int_{B_r(y^*)} K(x,y) \, dy \geq 2C(R).
  \end{equation*}
  Plugging this into~\eqref{eq:cont_up_to_constant}, using the Euler-Lagrange inequality~\eqref{eq:ELI2} and the definition of $C(R)$ with $\psi^* \leq 1$ yields the contradiction
  \begin{align*}
    -&\int_{\mathbb{R}^d} K_R(x,y) \psi(\operatorname{\mathop \textup{med}}_{K_R, \Psi} f(x) - f(y)) \mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{K_R, \Psi} f(x) < f(y)\}} \,dy \\
    &\stackrel{\eqref{eq:ELI2}, \eqref{eq:cont_up_to_constant}}{\leq} -\int_{\mathbb{R}^d} K (x,y) \psi_*( \operatorname{\mathop \textup{med}}_{K, \Psi} f(x) - f(y) )\mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{K, \Psi} f(x) \leq f(y)\}} \, dy - 2C(R)\\
    &\stackrel{\eqref{eq:not_cuttable}, \eqref{eq:f_g_prop_eps}}{\leq} -\int_{\mathbb{R}^d} K (x,y) \psi( \operatorname{\mathop \textup{med}}_{K_R, \Psi} f(x) - f(y) )\mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{K_R, \Psi} f(x) < f(y)\}} \, dy - 2C(R)\\
    &\stackrel{|\psi| \leq 1}{\leq} -\int_{\mathbb{R}^d} K_R (x,y) \psi( \operatorname{\mathop \textup{med}}_{K_R, \Psi} f(x) - f(y) )\mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{K_R, \Psi} f(x) < f(y)\}} \, dy + \int_{B_R(0)^c}\!\! K(z) \, dz - 2C(R)\\
    &\stackrel{\text{ def. } C(R)}{<} -\int_{\mathbb{R}^d} K_R (x,y) \psi( \operatorname{\mathop \textup{med}}_{K_R, \Psi} f(x) - f(y) )\mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{K_R, \Psi} f(x) < f(y)\}} \, dy.
  \end{align*}
  This proves the claimed~\eqref{eq:cuttable} and finishes the proof.

\emph{Step 1 -- $C^{0,1}$-contraction:} Let us first prove the following useful claim:  
\emph{All median schemes are contractions in the sense that the $C^{0,1}$ norm is non-increasing, i.e, for any $0\leq t_1 \leq t_2 < \infty$ let $v$ represent any of the functions $u_h, u_{h, R}$ or $u_{h, \varepsilon, R}$ then it holds
    \begin{align}
      \|v(\cdot, t_1)\|_{C^0} &\geq \|v(\cdot, t_2)\|_{C^0}\label{eq:max_non_increasing} \\
      \operatorname*{Lip}(v(\cdot, t_1))&\geq \operatorname*{Lip}(v(\cdot, t_2)) \label{eq:Lip_non_increasing}.
    \end{align}}

  Indeed, \eqref{eq:max_non_increasing} follows immediately from the comparison principle of Lemma~\ref{lem:comp_princ} when comparing $v$ to the constant functions with values $\min_{y \in \mathbb{R}^d} v(y,t_1)$ and $\max_{y \in \mathbb{R}^d} v(y,t_1)$. Also for  \label{eq:Lip_non_increasing} one can use the comparison principle of Lemma~\ref{lem:comp_princ}; this time between $v$ and the function $ z \mapsto v(z + x - y, t_1) \pm \operatorname*{Lip}(v) |x-y|$.

\emph{Step 2 -- Argument for $\lim_{R\uparrow \infty} u_{h, R} = u_h$:}
  By Lemma~\ref{lem:cuttable} it holds: For any bounded, Lipschitz continuous function $f$, any $x \in \mathbb{R}^d$, and sufficiently large $R > 0$ it holds
  \begin{equation*}
    |\operatorname{\mathop \textup{med}}_{h,R} f(x) - \operatorname{\mathop \textup{med}}_h f(x)| \leq 2\operatorname*{Lip}(f)\left( C(R,h) \right)^{\frac{1}{2d}}.
  \end{equation*}
  Here $C(R,h)$ is a constant only depending on $R, h$ given by
  \begin{equation*}
    C(R,h) := \int_{\{|z| > R\}} G_h(0,z)\, dz,
  \end{equation*}
  which in particular satisfies
  \begin{equation*}
    C(R,h) \to 0 \quad\quad \text{ as } R \uparrow \infty.
  \end{equation*}
  Thus, comparing $u_h$ to $v(x,t) = u_{h,R}(x,t) \pm \frac{2t}{h} \operatorname*{Lip}(u_0) C(R)^{\frac{1}{2d}}$ in Lemma~\ref{lem:comp_princ} yields 
  \begin{equation*}
    u_h(x,t) - u_{h,R} \lessgtr \pm \frac{2t\operatorname*{Lip}(u_0) C(R, h)^{\frac{1}{2d}}}{h}.
  \end{equation*}
  This means that $u_{h,R}$ converges uniformly to $u_h$ as $R \uparrow \infty$ on any compact time interval, because $h$ is fixed and $C(R, h) \to 0$ as $R \uparrow \infty$.

    \emph{Step 3 -- Argument for $\lim_{\varepsilon \downarrow 0} u_{h,\varepsilon,R} = u_{h, R}$:}
  We will first prove the following claim: For any continuous function $f$ it holds 
\begin{equation}\label{eq:eps_bound_epsmed}
  |\operatorname{\mathop \textup{med}}_{h,R} f(x)  - \operatorname{\mathop \textup{med}}_{h, \varepsilon, R} f(x)| \leq \varepsilon \quad \quad \text{ for all } x \in \mathbb{R}^d.
\end{equation}
Once ~\eqref{eq:eps_bound_epsmed} is shown, we can infer~\eqref{eq:convergence_uhepsR}. Indeed, comparing $u_{h,R}$ to $v(x,t) = u_{h, \varepsilon, R}(x,t) \pm \frac{t}{h}\varepsilon$ in Lemma~\ref{lem:comp_princ} yield
\begin{equation*}
  u_{h,R} - u_{h, \varepsilon, R} \lessgtr \pm \frac{t}{h}\varepsilon.
\end{equation*}

To prove the claimed~\eqref{eq:eps_bound_epsmed}, assume the contrary is true. So there exists a fixed $x$ such that $\operatorname{\mathop \textup{med}}_{h, \varepsilon, R} f (x) > \operatorname{\mathop \textup{med}}_{h,R} f(x) + \varepsilon$ (the other case is handled analogously).
Note, by the properties~\eqref{def:psi_eps} of $\psi_\varepsilon$ it holds $\psi_\varepsilon(\operatorname{\mathop \textup{med}}_{h, \varepsilon, R} f(x) - f(y)) = 1$ in the set $\{y: \operatorname{\mathop \textup{med}}_{h,R} f(x) \geq f(y) \}$. Hence, using the Euler-Lagrange inequality~\eqref{eq:ELI1} we have
\begin{align}
  \int_{\mathbb{R}^d} G_{h, R}(x,y) \mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{h,R} f(x) < f(y) \}} \, dy &\leq \int_{\mathbb{R}^d} G_{h, R}(x,y) \mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{h,R} f(x) \geq f(y) \}} \, dy \nonumber\\
  &= \int_{\mathbb{R}^d} G_{h, R}(x,y) \mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{h,R} f(x) \geq f(y) \}} \psi_\varepsilon(\operatorname{\mathop \textup{med}}_{h, \varepsilon, R} f(x) - f(y)) \, dy.\label{eq:R_prop}
\end{align}
As $f$ is continuous, we may make a similar construction as in the proof of Lemma~\ref{lem:contraction} (see~\eqref{eq:ball_prop_eps} for more details) to obtain a point $y^* \in \mathbb{R}^d$ and $r > 0$ satisfying
      \begin{equation*}
      B_r(y^*) \subset \{y: \operatorname{\mathop \textup{med}}_{h, \varepsilon, R} f(x) > f(y)\} \cap \{y: \operatorname{\mathop \textup{med}}_{h,R} f(x) \geq f(y) \}^c.
    \end{equation*}
Thus, we get the strict inequality 
\begin{multline*}
  \int_{\mathbb{R}^d} G_{h, R}(x,y) \mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{h,R} f(x) \geq f(y) \}}  \psi_\varepsilon(\operatorname{\mathop \textup{med}}_{h, \varepsilon, R} f(x) - f(y)) \, dy\\
  <  \int_{\mathbb{R}^d} G_{h, R}(x,y) \mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{h, \varepsilon, R} f(x) > f(y) \}} \psi_\varepsilon(\operatorname{\mathop \textup{med}}_{h, \varepsilon, R} f(x) - f(y)) \, dy.
\end{multline*}
Plugging this into~\eqref{eq:R_prop} and using $\psi_\varepsilon \geq -1$ as well as $\operatorname{\mathop \textup{med}}_{h, \varepsilon, R} f(x) > \operatorname{\mathop \textup{med}}_{h,R} f(x)$ once more yields the desired contradiction
\begin{align*}
  &\int_{\mathbb{R}^d} G_{h, R}(x,y) \mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{h,R} f(x) < f(y) \}} \, dy \\
  &<  \int_{\mathbb{R}^d} G_{h, R}(x,y) \mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{h, \varepsilon, R} f(x) > f(y) \}} \psi_\varepsilon(\operatorname{\mathop \textup{med}}_{h, \varepsilon, R} f(x) - f(y)) \, dy\\
  &\leq -\int_{\mathbb{R}^d} G_{h, R}(x,y) \mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{h, \varepsilon, R} f(x) \leq f(y) \}} \psi_\varepsilon(\operatorname{\mathop \textup{med}}_{h, \varepsilon, R} f(x) - f(y)) \, dy\\
  &\leq \int_{\mathbb{R}^d} G_{h, R}(x,y) \mathbbold{1}_{\{\operatorname{\mathop \textup{med}}_{h,R} f(x) < f(y) \}} \, dy.
\end{align*}
This proves the claim~\eqref{eq:eps_bound_epsmed} and finishes the proof of this lemma.

\label{lem:exp_decay}
If $u_0 = c$ outside $B_L(0)$ then $u_{h, \varepsilon, R}$, $\operatorname{\mathop \textup{med}}_{h, \varepsilon, R} u_{h,\varepsilon, R}$ and $H_{h, \varepsilon, R}$ have exponential decay, i.e., for any $t \in [0,T]$ and any $x \in \mathbb{R}^d$

  |u_{h,\varepsilon, R}(x,t) - c| &\leq C_2 e^{- (|x| - L)}, \label{eq:exp_u}\\
  |H_{h,\varepsilon, R}(x,t) | &\leq C_3(\varepsilon) e^{- (|x| - L)} \label{eq:exp_H},\\
  |\operatorname{\mathop \textup{med}}_{h, \varepsilon, R} u_{h,\varepsilon, R}(x,t) - c| &\leq C_4 e^{- (|x| - L)} \label{eq:exp_medu}.
\end{align}
The constants $C_2$ and $C_4$ do not depend on $\varepsilon$, while $C_3(\varepsilon)$ may depend on $\varepsilon$. They may depend on $h, R$ or $T$.

W.l.o.g.\ we may assume that $c = 0$. For easier readability we drop the $h$, $\varepsilon$ and $R$ subscripts in $u_{h,\varepsilon, R}$, $H_{h,\varepsilon, R}$ and $\operatorname{\mathop \textup{med}}_{h, \varepsilon, R}$. 

\emph{Step 1 -- Median estimate:} We claim there exists a constant $C$ only depending on $\operatorname*{Lip}(u_0)$, $h$ and $R$ but not on $\varepsilon$ such that
\begin{equation}\label{eq:med_u_by_u}
|\operatorname{\mathop \textup{med}} u(x)|  \leq C \left(|u(x)| + \int_{\mathbb{R}^d} G_h(x,y) |u(y)|\,dy\right).
\end{equation}
Indeed, define the function 
\begin{equation*}
  F(m; x):= \int_{\mathbb{R}^d} G_{h,R}(x,y) \Psi_\varepsilon(m- u(y)) \, dy.
\end{equation*}
The Euler-Lagrange equation resulting from the definition of $\operatorname{\mathop \textup{med}}$ reads
\begin{equation}\label{eq:epsEL}
  0 = F'(\operatorname{\mathop \textup{med}} u(x); x).
\end{equation}
Choose $z$ with $\operatorname{\mathop \textup{med}} u(x) = u(z)$. W.l.o.g\ assume that $\operatorname{\mathop \textup{med}} u(x) \geq u(x)$. By the continuity of $u$, for any $m \in [u(x), u(z)]$ there exists a $z_m$ such that $m=u(z_m)$ and $|x-z_m| \leq R$. As $\operatorname*{Lip}(u) \leq \operatorname*{Lip}(u_0)$ by~\eqref{eq:Lip_non_increasing}, there exists an $r$ independent of $m$, $h$, $R$ and $\varepsilon$, with
\begin{equation*}
  u(y) \in \left[u(z_m)- \frac{\varepsilon}{2}, u(z_m) + \frac{\varepsilon}{2}\right] \quad \quad \text{ for all } y \in B_r(z_m).
\end{equation*}
Combining this with the previous observation $|x-z_m| \leq R$ yields with $\psi_\varepsilon' > c_1/\varepsilon$ on the compact interval $[-\frac{\varepsilon}{2}, \frac{\varepsilon}{2}]$ for all $y \in B_r(z_m)$ that
\begin{equation*}
  G_{h,R}(x,y) \psi''_\varepsilon(m - u(y))\,dy \geq G_{h}(0, e_1 R) \frac{c_1}{\varepsilon}  =: \frac{\lambda}{\varepsilon} > 0.
\end{equation*}
This lets us bound the second derivative of $F(\cdot; x)$ for sufficiently large $R$ and all $m \in [u(x), \operatorname{\mathop \textup{med}} u(x)]$ by
\begin{align*}
  F''(m; x) = \int_{\mathbb{R}^d} G_h(x,y) \psi''_\varepsilon(m - u(y)) \,dy \geq \frac{1}{4}|B_r(z_m)| \frac{\lambda}{\varepsilon} > 0.
\end{align*}
But now by the mean value theorem and~\eqref{eq:epsEL} it follows with
\begin{equation*}
    F'(u(x); x) \leq \int_{\mathbb{R}^d} G_h(x,y) \psi_\varepsilon(u(x) - u(y)) \,dy \leq \frac{C_1}{\varepsilon} \left( |u(x)| + \int_{\mathbb{R}^d} G_h(x,y)|u(y)| \, dy\right),
\end{equation*}
where we used $|\psi_\varepsilon'|\leq C_1/\varepsilon$ and $\psi_\varepsilon(0) = 0$, that
\begin{align*}
  (|\operatorname{\mathop \textup{med}} u(x)| - |u(x)|)\ \frac{\lambda}{4 \varepsilon}|B_r(0)| &\leq |\operatorname{\mathop \textup{med}} u(x) - u(x)|\ \frac{\lambda}{4 \varepsilon}|B_r(0)|\\
  & \leq |F'(u(x) ; x) - F'(\operatorname{\mathop \textup{med}} u(x); x)| \\
  &\leq \frac{C_1}{\varepsilon} \left(|u(x)| + \int_{\mathbb{R}^d} G_h(x,y)|u(y)| \, dy\right).
\end{align*}
Rearranging yields the claim~\eqref{eq:med_u_by_u}.

\emph{Step 2 -- Argument for~\eqref{eq:exp_u}}: Let us introduce the weighted norm 
\begin{equation*}
  \|f\|_{L} := \sup_{x \in \mathbb{R}^d} e^{(|x| - L)_+} |f(x)|.
\end{equation*}
Recall that we assumed $u_0$ is bounded and equal to zero for $|x|>L$ and thus it holds
\begin{equation*}
  \|u_0\|_L \leq \|u_0\|_\infty < \infty.
\end{equation*}
Integrating~\eqref{def:u_heps} from $0$ to $t$ yields
\begin{equation*}
  u(t) = u_0 + \int_0^t \frac{\operatorname{\mathop \textup{med}} u(s) - u(s)}{h} \, ds.
\end{equation*}
As $\|\cdot\|_L$ is a norm we can use the triangle and Jensen's inequality and finally \emph{Step 1} to estimate
\begin{align*}
  \|u(t)\|_L &\leq \|u_0\|_L + \frac{1}{h} \int_0^t \|\operatorname{\mathop \textup{med}} u(s)\|_L + \| u(s)\|_L \,ds \\
  &\leq \|u_0\|_L + \frac{1}{h} \int_0^t C\left\|\int_{\mathbb{R}^d}  G_h(\cdot,y) u(y,s)\,dy + |u(s)| \right\|_L + \| u(s)\|_L \,ds  \\
  &\leq \|u_0\|_L + \frac{1}{h} \int_0^t \int_{\mathbb{R}^d} C G_h(\cdot,y) \|u(s)\|_L\,dy  + (1+C)\| u(s)\|_L \,ds\\
    &\leq \|u_0\|_L + \frac{1}{h} \int_0^t (1+2C)\| u(s)\|_L \,ds.
\end{align*}
Thus, by Gr{\"o}nwall's inequality we conclude
\begin{equation*}
  \|u(t)\|_L \leq e^{t(1+2C)}\|u_0\|_L \lesssim e^{t(1+2C)}.  
\end{equation*}
But now by the definition of $\|\cdot\|_L$ we can finish this step, as
\begin{equation}\label{eq:exp_decay_u}
  |u(x,t)| \leq e^{-(|x| - L)_+} \|u(t)\|_L \lesssim e^{-(|x| - L)_+} e^{t(1+2C)}.
\end{equation}

\emph{Step 3 -- Argument for~\eqref{eq:exp_H}:} Simply using $\psi'_\varepsilon(0) = 0$ and $|\psi''_\varepsilon| \leq C_\varepsilon$ we crudely estimate
\begin{align}
  |H(x)| &\leq \frac{1}{\sqrt{h}} \int_{\mathbb{R}^d} G_h(x,y) |\psi'_\varepsilon(u(x) - u(y))|\,dy \nonumber \\
  &\leq \frac{C_\varepsilon}{\sqrt{h}} \int_{\mathbb{R}^d} G_h(x,y) \big(|u(x)| + |u(y)|\big)\,dy \nonumber\\
  &\leq C(\varepsilon, h)\left( e^{-(|x| - L)_+} + \int_{\mathbb{R}^d} G_h(x,y) e^{-(|x| - L)_+} \,dy\right). \label{eq:H_exp_est}
\end{align}
We use the general inequality
\begin{equation*}
  (|y| - L)_+ \geq (|x| - L)_+ - |x-y| \quad \quad \text{ for any } x,y \in \mathbb{R}^d,
\end{equation*}
to get the control
\begin{align}\label{eq:exp_control_convolution}
  \int_{\mathbb{R}^d} G_h(x,y) e^{-(|y| - L)_+} \,dy &\leq e^{-(|x| - L)_+}\int_{\mathbb{R}^d} G_h(x,y) e^{|x-y|} \,dy \\
  &= e^{-(|x| - L)_+} \int_{\mathbb{R}^d} G_h(0,z) e^{|z|} \,dz \lesssim  e^{-(|x| - L)_+} \nonumber.
\end{align}
Plugging this into~\eqref{eq:H_exp_est} yields~\eqref{eq:exp_H}.

\emph{Step 4 -- Argument for~\eqref{eq:exp_medu}:} Combining~\eqref{eq:med_u_by_u}, \eqref{eq:exp_decay_u} and~\eqref{eq:exp_control_convolution} yields
\begin{equation*}
  |\operatorname{\mathop \textup{med}} u (x)| \lesssim |u(x)| + \int_{\mathbb{R}^d} G_h(x,y) |u(y)|\,dy \lesssim e^{-(|x| - L)_+} + \int_{\mathbb{R}^d} G_h(x,y) e^{-(|y| - L)_+}\,dy \lesssim e^{-(|x| - L)_+}.
\end{equation*}
This finishes the proof.

Denote by $d$ the signed distance function to $\Omega_0$, i.e.,
  \begin{equation*}
    d(x) = \begin{cases}
      -\textup{dist}(x, \Omega_0) & \text{ if } x \in \mathbb{R}^d \setminus \Omega_0,\\
      \textup{dist}(x, \mathbb{R}^d \setminus \Omega_0) & \text{ if } x \in \Omega_0.
    \end{cases}
  \end{equation*}
  Furthermore, as $\Omega_0$ is bounded we know that the reach $2\delta_0$ of the manifold $\partial \Omega_0$ is positive. Let $\phi \in C^\infty(\mathbb{R}^d)$ be a linear interpolation on $[-\delta_0, \delta_0]$ that is constantly extended outside:
  \begin{equation*}
    \phi(s) = \begin{cases}
      -1 & \text{ if } s \leq \delta_0,\\
      s &\text{ if } s \in [-\delta_0, \delta_0],\\
      1 & \text{ if } s \geq \delta_0.
    \end{cases}
  \end{equation*}
Now we define our initial conditions by
\begin{equation*}
  u_0 = \phi \circ d.
\end{equation*}
By construction, \eqref{eq:zero_level_of_u_0} and~\eqref{eq:const_out_u_0} hold. Also, the regularity of the boundary of $\Omega_0$ implies that $d \in C^3_b(\mathbb{R}^d)$ which means also $u_0 \in C^3_b(\{x: |d(x)| \neq \delta_0\} )$; so setting $S= \{x: |d(x)| = \delta_0\}$ yields the desired regularity properties.

Next we prove~\eqref{eq:L1_u_0}. We begin by showing the convergence
\begin{equation}\label{eq:H_h_to_H}
  \lim_{h \downarrow 0} \frac{1}{\sqrt{h}} \int_{B_{\sqrt{h} R_h}(x)} G_h(x,y) \mathop \mathrm{sign}(u_0(x) - u_0(y)) \,dy = c_0 \nabla \cdot \left(\frac{\nabla u_0}{|\nabla u_0}\right),
\end{equation}
if $u_0 \in C^3(B_{\delta_0}(x))$ and $R_h = \sqrt{6|\log (h)|}$ and $|\nabla u_0(x)| > 0$ . Moreover, the rate at which this convergence holds depends on $x$ only through $|\nabla^2 u_0(x)|/|\nabla u_0(x)|$, i.e., it is uniform on sets where the Hessian is bounded from above and the gradient is bounded from below.

Since $\operatorname{sign}\big(u_0(x) - u_0(y)\big)$ is invariant under multiplying $u_0$ by a
positive constant, and since the normalized gradient $\nabla u_0 / |\nabla u_0|$ is invariant
under the same rescaling, both sides of \eqref{eq:H_h_to_H} are unchanged if we replace $u_0$
by
\[
  v := \frac{u_0}{|\nabla u_0(x)|}.
\]
This gives $|\nabla v(x)| = 1$, so it suffices to prove \eqref{eq:H_h_to_H} under this
normalization. The only quantity affected is the Hessian, which rescales as
\[
  \nabla^2 v = \frac{\nabla^2 u_0}{|\nabla u_0(x)|}.
\]

We make a change of variables $z = \frac{y-x}{\sqrt{h}}$ to obtain
\begin{align*}
  \frac{1}{\sqrt{h}} \int_{B_{\sqrt{h}R_h}(x)} G_h(x,y) \mathop \mathrm{sign}(u_0(x) - u_0(y)) \,dy &=\frac{1}{\sqrt{h}} \int_{B_{R_h}(x)} G_1(0,z) \mathop \mathrm{sign}(u_0(x) - u_0(x + \sqrt{h} z)) \,dz.
\end{align*}

W.l.o.g.\ we can assume that $u_0(x) = 0, \nabla u_0(x) = e_d$. 
We are interested in understanding the behavior of $\mathop \mathrm{sign}(u_0(x) - u_0(x + \sqrt{h} z))$. Thus, we need to find all point $z$ at which $u_0(x) - u_0(x + \sqrt{h} z) = 0$. In particular, if we denote $(z', z_d) = z$, then we claim that for $h$ small enough, there is a unique $z_d = g_h(z')$ with $u_0(x) - u_0(x + \sqrt{h} z) = 0$. 
By Taylor expansion, since $u_0 \in C^3(B_{\sqrt{h}R_h}(x))$ (for $h$ small enough), we have
\begin{equation*}
  0 \stackrel{!}{=} u_0(x) - u_0(x+ \sqrt{h}z) = \sqrt{h} z_d + \frac{h}{2} z^\intercal \nabla^2 u_0(x) z + \mathcal{O}(h^{\frac{3}{2}} |z|^3).
\end{equation*}
Rearranging and dividing by $\sqrt{h}$ turns the previous equation to 
\begin{equation}\label{eq:zero_sign_level}
  z_d = -\frac{\sqrt{h}}{2} z^T \nabla^2 u_0(x) z + \mathcal{O}(h |z|^3).
\end{equation}
Using $\|\nabla^2 u_0\|_\infty < \infty$ yields the uniform bound
\begin{equation*}
  z_d = \mathcal{O}(\sqrt{h}|z|^2).
\end{equation*}
So there is a zero of $\mathop \mathrm{sign}(u_0(x) - u_0(x + \sqrt{h} z))$ close to $0$; remember that $z_d= 0$ meant $x_d = y_d$ before the coordinate transform. 

Next we want to see that there is no second zero. Therefore, define the function
\begin{equation*}
  F(z_d) := u_0(x + \sqrt{h} z) - u_0(x).
\end{equation*}
Differentiating yields 
\begin{equation*}
  \partial_{z_d} F(z_d) = \partial_{z_d} u(x + \sqrt{h}z) = \partial_{x_d} u_0(x+ \sqrt{h}z) \sqrt{h}.  
\end{equation*}
As $\partial_{x_d} u_0(x) = 1$ and $\|\nabla^2 u_0\|_\infty < \infty$ we have that $\partial_{z_d} F(z_d) > 0$ for $|\sqrt{h} z| < \delta$ and some $\delta > 0$ independent of $x$. In other words, $F(z_d)$ is increasing for $|z| \leq \frac{\delta}{\sqrt{h}}$. Remember we cut of the domain so that we have $|z| \leq R_h = \sqrt{6 |\log h|}$. This means for $h$ small enough it holds that $F(z_d)$ is increasing for all $z$ in our cut-off domain. Hence, $F$ can only have one zero in our domain. 

Furthermore, plugging $z_d = \mathcal{O}(\sqrt{h}|z|^2)$ back into~\eqref{eq:zero_sign_level} and remembering that due to our cut-off it holds $\mathcal{O}(h|z|^4) = \mathcal{O}(h\, (3|\log(h)|)^2) = o(\sqrt{h})$, gives the looked-for characterization of the unique zero
\begin{equation*}
  z_d = -\frac{\sqrt{h}}{2} z'^\intercal \nabla'^2 u_0(x) z' + \mathcal{O}(h |z|^4) = -\frac{\sqrt{h}}{2} z'^\intercal \nabla'^2 u_0(x) z' + o(\sqrt{h}) =: g_h(z').
\end{equation*}

Recall that $z_d = \mathcal{O}(\sqrt{h}|z|^2) = \mathcal{O}(h^{\frac{1}{4}})$. Now using the properties of the heat kernel we have
\begin{align*}
  &\lim_{h \downarrow 0} \frac{1}{\sqrt{h}} \int_{ B_{R_h}(0)} G_1(0,z) \mathop \mathrm{sign}(u_0(x) - u_0(x + \sqrt{h} z)) \,dz \\
  &= \lim_{h \downarrow 0} \frac{1}{\sqrt{h}} \int_{B^{d-1}_{R_h - \mathcal{O}(h^{\frac{1}{4}})}(0)}G_1^{d-1}(0,z') \int_{-(R_h -|z'|)}^{R_h -|z'|} G_1^1(0,z_d) \mathop \mathrm{sign}(u_0(x) - u_0(x + \sqrt{h} z)) \,d z_d \, dz'.
\end{align*}
As the zero $g_h(z')$ is in $[-(R_h -|z'|), R_h -|z'|]$ we have by symmetry of $G_1^1$
\begin{align*}
  &\int_{-(R_h -|z'|)}^{R_h -|z'|} G_1^1(0,z_d) \mathop \mathrm{sign}(u_0(x) - u_0(x + \sqrt{h} z)) \,dz_d\\ 
  &= -\int_{-(R_h - |z'|)}^{g_h(z')} G_1^1(0,z_d) \, d z_d + \int_{g_h(z')}^{R_h - |z'|} G_1^1(0,z_d) \, d z_d\\
  &= -2 \int_{0}^{g_h(z')} G_1^1(0,z_d) \, d z_d\\
  &= -\operatorname*{erf}(\frac{g_h(z')}{2}).
\end{align*}
Furthermore, Taylor-expanding the error function gives
\begin{equation*}
  \operatorname*{erf}(\frac{g_h(z')}{2}) = \operatorname*{erf}(0) + \operatorname*{erf}\nolimits'(0) \frac{g_h(z')}{2} + \mathcal{O}(|g_h|^2) = 2c_0 \frac{g_h(z')}{2} + o(\sqrt{h}).
\end{equation*}
Combing the previous computations leaves us with 
\begin{align*}
  &\lim_{h \downarrow 0} \frac{1}{\sqrt{h}} \int_{ B_{R_h}(0)} G_1(0,z) \mathop \mathrm{sign}(u_0(x) - u_0(x + \sqrt{h} z)) \,dz \\
  &=\lim_{h \downarrow 0} \frac{1}{\sqrt{h}} \int_{B^{d-1}_{R_h - \mathcal{O}(h^{\frac{1}{4}})}(0)}G_1^{d-1}(0,z') \big(-c_0 g_h(z') + o(\sqrt{h})\big)\, dz'\\
  &= \lim_{h \downarrow 0} \int_{B^{d-1}_{R_h - \mathcal{O}(h^{\frac{1}{4}})}(0)} G_1^{d-1}(0,z') \big( \frac{c_0}{2} z'^\intercal \nabla'^2 u_0(x) z' + o(1)\big)\, dz'\\
  &= \lim_{h \downarrow 0} \frac{c_0}{2} \int_{\mathbb{R}^{d-1}} G_1^{d-1}(0,z')  z'^\intercal \nabla'^2 u_0(x) z'\, dz' + o(1),
\end{align*}
where we used in the last equation the exponential decay of the heat kernel (see~\eqref{eq:cut_off_u_tilde} for more details).
The symmetry of the heat kernel yield
\begin{align*}
    \int_{\mathbb{R}^{d-1}} z_i z_j G_1^{d-1}(0, z') \,dz' = \delta_{ij} \int_{\mathbb{R}} z_1^2 G_1^1(0, z_1) \, d z_1
\end{align*}
and
\begin{align*}
    \int_{\mathbb{R}} - 2 z_1 \partial_{z_1} G_1^1(0,z_1) \, dz_1 = \int_{\mathbb{R}} 2 G_1^1(0, z_1) \, dz_1 = 2
\end{align*}
and hence
\begin{equation*}
    \int_{\mathbb{R}^{d-1}} z_i z_j G_1^{d-1}(0, z') \, dz' = 2 \delta_{ij}.
\end{equation*}
This let's us compute
\begin{align*}
   \int_{\mathbb{R}^{d-1}} G_1^{d-1}(0,z')  z'^\intercal \nabla'^2 u_0(x) z'\, dz' &= \sum_{i = 1}^{d-1} \int_{\mathbb{R}} z_i^2 \partial_{x_i} \partial_{x_j}u_0(x) G_1^{d-1}(0, z') \,dz'\\
   &= 2 (\Delta u_0(x) - \partial_{x_d}\partial_{x_d} u_0(x)).
\end{align*}
Note, as $\nabla u_0(x) = e_d$ we also have
\begin{equation*}
  \Delta u_0(x) -  \partial_{x_d}\partial_{x_d} u_0(x) = \Delta u_0(x) - \frac{\nabla u_0(x)^\intercal \nabla^2 u_0(x) \nabla u_0(x)}{|\nabla u_0(x)|^2} = \nabla \cdot \left(\frac{\nabla u_0(x)}{|\nabla u_0(x)|} \right)|\nabla u_0(x)|.
\end{equation*}
Combing the above observations yields the claimed~\eqref{eq:H_h_to_H}.

Denote by
\begin{align*}
  \Omega_- &:= \{x \in \mathbb{R}^d: d(x) \leq - \delta_0\},\\
  \Omega_+ &:= \{x \in \mathbb{R}^d: d(x) \geq + \delta_0\},\\
  \Omega_{:)} &:= \{x \in \mathbb{R}^d:-\delta_0 < d(x) < \delta_0\},
\end{align*}
the sets of points with vanishing gradient $\Omega_-$ and $\Omega_+$, as well as the set of not-vanishing gradient $\Omega_{:)}$.  We now claim that 
\begin{align}\label{eq:conv_to_per}
  \lim_{h \downarrow 0}\int_{\Omega_- \cup \Omega_+} |H_h(x)|\,dx = \textup{Per}(\Omega_-) + \textup{Per}(\Omega_+)
\end{align}  
and 
\begin{align}\label{eq:conv_to_H}
    \lim_{h \downarrow 0}  H_h(x) = 2 c_0 \nabla \cdot \left( \frac{\nabla u_0}{|\nabla u_0|} \right) \quad \quad \text{uniformly on $\Omega_{:)}$}.
\end{align}
Showing this will be enough as $\Omega_-$ and $\Omega_+$ have compact $C^3$ boundary by construction and are thus sets of finite perimeter; furthermore, it holds $|\nabla u_0| =1$, $\phi' \circ d = 1$, $\phi''\circ d =0$ on $\Omega_{:)}$ so that
\begin{equation*}
  \nabla \cdot \left( \frac{\nabla u_0}{|\nabla u_0|} \right) = \Delta u_0 = \Delta d \ \phi' \circ \delta + |d|^2 \phi'' \circ d = \Delta d \quad \quad \text{ on } \Omega_{:)}.
\end{equation*}
This yields with a generic constant $C$
\begin{align*}
  \sup_{h \in (0,1]} \int_{\mathbb{R}^d}|H_h| \,dx &\leq C + \textup{Per}(\Omega_-) + \textup{Per}(\Omega_+) + 2c_0 \int_{\Omega_{:)}}(\nabla \cdot \left( \frac{\nabla u_0}{|\nabla u_0|} \right) + C)\,dx\\
  & \leq C + \textup{Per}(\Omega_-) + \textup{Per}(\Omega_+)+ 2c_0|\Omega_{:)}|(C + \|\nabla^2 d\|_\infty) \\
  &< \infty.
\end{align*}
 
So, it remains to prove~\eqref{eq:conv_to_per} and~\eqref{eq:conv_to_H}. Let us first prove~\eqref{eq:conv_to_per} and concentrate on the set $\Omega_-$ as the argument for $\Omega_+$ is analogous:

As $u_0$ is constantly equal to $-1$ on $\Omega_-$ it holds for all $x, y \in \Omega_-$ and $z \in \Omega_-^c$
\begin{equation*}
  \mathop \mathrm{sign}(u_0(x) - u_0(y)) = 0 \quad \quad \text{ and } \quad \quad \mathop \mathrm{sign}(u_0(x) - u_0(z)) = -1.
\end{equation*}
This lets us compute
\begin{align*}
  \int_{\Omega_-} |H_h(x)|\,dx = \frac1{\sqrt{h}}\int_{\Omega_-} \int_{\Omega_-^c} G_h(x,y)\,dx.
\end{align*}
The right-hand side is exactly the thresholding energy which is known to converge to the perimeter for sets of finite perimeter, see for example~\cite[Lemma A.3]{EsedogluOtto}.

In order to prove~\eqref{eq:conv_to_H} we want to apply the uniform convergence~\eqref{eq:H_h_to_H}. To do this, fix $x \in \Omega_{:)}$. We need to ensure that $|\nabla u_0(x)|$ is bounded from below, $|\nabla^2 u_0|$ is bounded from above and $u_0 \in C^3(B_{\delta_0}(x))$. We first notice that $|\nabla u_0| = |\phi'(d)| |\nabla d| = \delta_0$; thus the gradient is bounded from below. Furthermore, the Hessian of $u_0$ is bounded from above on $\Omega_{:)}$ as $\nabla^2 u_0 = \nabla (\phi'(d) \nabla d) = \phi''(d) + \phi'(d) \nabla^2 d = \nabla^2 d$ and $\|\nabla^2 d\|_{L^\infty(B_{\delta_0}(x))} < \infty$.

As for $y \in \Omega_+$ (and similarly for $\Omega_-$) it always holds $\mathop \mathrm{sign}(u_0(x) - u_0(y)) = -1$ we can thus replace $u_0$ by any function that is smaller on $\Omega_+$ with out changing the $\mathop \mathrm{sign}$. But now we have for $\sqrt{h}R_h \leq \delta_0$ that
\begin{equation*}
   \frac{1}{\sqrt{h}} \int_{B_{\sqrt{h}R_h}(x)} G_h(x,y) \mathop \mathrm{sign}(u_0(x) - u_0(y)) \,dy = \frac{1}{\sqrt{h}} \int_{B_{\sqrt{h}R_h}(x)}  G_h(x,y) \mathop \mathrm{sign}(\tilde{u}_0(x) - \tilde{u}_0(y)) \,dy,
\end{equation*}
where $\tilde{u}_0 := 2 \phi(\frac{u_0}{2})$.
So $\tilde{u}_0$ is in $C^3(B_{\sqrt{h}R_h }(x))$ for $\sqrt{h}R_h \leq \delta_0$. Thus, we can apply~\eqref{eq:H_h_to_H} to obtain the, for all $x \in \Omega_{:)}$ uniform, convergence
\begin{equation}\label{eq:u_tilde}
  \lim_{h \downarrow 0}\frac{2}{\sqrt{h}} \int_{B_{\sqrt{h}R_h}(x)} G_h(x,y) \mathop \mathrm{sign}(u_0(x) - u_0(y)) \,dy = 2c_0\nabla \cdot \left(\frac{\nabla \tilde{u}_0(x)}{|\nabla \tilde{u}_0(x)|}\right) =  2c_0 \nabla \cdot \left(\frac{\nabla u_0(x)}{|\nabla u_0(x)|}\right).
\end{equation}

At last, we cut off the far field term at the radius $R_h = \sqrt{6|\log (h)|}$; this is possible as 
\begin{align}
  &\lim_{h \downarrow 0}\left|\frac{1}{\sqrt{h}} \int_{\mathbb{R}^d \setminus B_{\sqrt{h}R_h}(x)} G_1(x,y) \mathop \mathrm{sign}(u_0(x) - u_0(y)) \,dz\right| \nonumber\\
  &=
  \lim_{h \downarrow 0}\left|\frac{1}{\sqrt{h}} \int_{\mathbb{R}^d \setminus B_{R_h}(0)} G_1(0,z) \mathop \mathrm{sign}(u_0(x) - u_0(x + \sqrt{h} z)) \,dz\right|  \nonumber\\
  &\lesssim \lim_{h \downarrow 0}\frac{1}{\sqrt{h}} e^{-\frac{R_h^2}{8}}= \lim_{h \downarrow 0}\frac{h^{\frac{3}{4}}}{\sqrt{h}} = 0 \label{eq:cut_off_u_tilde}.
\end{align}
Combing~\eqref{eq:u_tilde} with~\eqref{eq:cut_off_u_tilde} yields~\eqref{eq:conv_to_H}.

It remains to prove~\eqref{eq:energy_bound_u_0}. To that aim we use that  $u_0$ is constant outside $B_L(0)$ to compute 
\begin{align*}
    E_h(u_0) = \frac{2}{\sqrt{h}} \int_{B_{L}(0)} \int_{\mathbb{R}^d} G_h(x,y) |u_0(x) - u_0(y)|\, dy \, dx.
\end{align*}
Next we use a coordinate transform $z = \frac{y-x}{\sqrt{h}}$ to obtain
\begin{equation*}
      \frac{2}{\sqrt{h}} \int_{B_{L}(0)} \int_{\mathbb{R}^d} G_h(x,y) |u_0(x) - u_0(y)|\, dy \, dx = \frac{2}{\sqrt{h}} \int_{B_L(0)} \int_{\mathbb{R}^d} G_1(0,z) |u_0(x) - u_0(x + \sqrt{h}z)|\, dz \, dx.
\end{equation*}
Finally, we use that $u_0$ is $1$-Lipschitz to get the control
\begin{multline*}
     \frac{2}{\sqrt{h}} \int_{B_L(0)} \int_{\mathbb{R}^d} G_1(0,z) |u_0(x) - u_0(x + \sqrt{h}z)|\, dz \, dx \\\leq 2 \int_{B_L(0)} \int_{\mathbb{R}^d} G_1(0,z) |z|\, dz \, dx  = 4c_0|B_L(0)| < \infty.
\end{multline*}
Combining the previous computations yields~\eqref{eq:energy_bound_u_0} and finishes this proof.
