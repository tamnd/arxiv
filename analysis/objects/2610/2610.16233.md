All logarithms are natural and all information quantities are in nats.  Let the binary entropy be $h(a)\coloneqq -a\log a-(1-a)\log(1-a)$. For probability measures $P,Q$ on a common measurable space, $\mathrm{D_{KL}}\left(P \|Q \right)$ denotes  the KL divergence.

<!-- lem-1: body_md -->
\label{cmi-js}
Under the preceding notation, for any fixed $i\in[N]$, we have
\begin{align}\label{eq:cmi-js}
    I(U_i;T_i)=J_\rho(P,Q).
\end{align}

<!-- rem-1: body_md -->
\label{rem:cmi-membership}
Eq.~(\ref{eq:cmi-js}) identifies the information that $T_i$ reveals about
membership with the prior weighted JS divergence between the two conditional output laws. Equivalently, $I(U_i;T_i)=h(\rho)-H(U_i\mid T_i)$, where $h$ denotes binary entropy. Thus, selector information measures the
expected reduction in membership uncertainty after observing $T_i$.
If $P=Q$, then $I(U_i;T_i)=0$, that is, the output provides no membership information beyond the prior. At the opposite extreme, if $P$ and $Q$ are mutually
singular, membership can be recovered perfectly from $T_i$, and
$I(U_i;T_i)=h(\rho)$. The selector density therefore determines 
the maximum information that
an individual output can reveal.

<!-- thm-3: body_md -->
\label{thm:fpr-random-fact}
In the random-fact setting, suppose that $\frac{1}{n}I(T^N;K)\le R$.
Then in the sparse limit $\rho\to0$ and $n \to \infty$,
 for every $\tau$ s.t. $\alpha_\tau>0$, we have
$\beta_\tau
 \geq
 \exp\left(-\frac{R+h(\alpha_\tau)}{\alpha_\tau}\right)$.
In particular, perfect recall $\alpha_\tau=1$ implies $\beta_\tau\geq e^{-R}$.

<!-- rem-2: body_md -->
\label{rem:selector-gen-hall}
Lemma~\ref{cmi-js} gives both (generalization and hallucination) analyses a membership-testing interpretation through $I(U;T)=J_\rho(P,Q)$.
Small selector information in \emph{losses} controls the generalization gap, whereas high fact recall with few false positives requires sufficient selector information in \emph{factuality scores}. Indeed, $P=Q$ implies $\alpha_\tau=\beta_\tau$ for every threshold. 
In the random-fact setting, hallucination can be viewed as \emph{unwanted generalization of factual confidence} from observed facts to unseen nonfacts.

\label{thm:joint-information-budget}
Suppose that $I(W;S)\le C$ and that
$\Theta\sim\operatorname{Unif}([N_\Theta])$, where $N_\Theta\ge2$. Assume that there exists a decoder $\widehat\Theta(W)$ satisfying $\mathbb P(\widehat\Theta\neq\Theta)\le\varepsilon_\Theta$. Define
\begin{align*}
C_{\mathrm{rule}}
\coloneqq
\log N_\Theta
-h(\varepsilon_\Theta)
-\varepsilon_\Theta\log(N_\Theta-1), \quad\text{ and }
C_{\mathrm{gen}}
\coloneqq
\frac{2nm\max\{n,m\}}{N^2}
|\mathcal E(\mathcal A)|^2. 
\end{align*}
Then $I(W;\Theta)\ge C_{\mathrm{rule}}$ and
$I(W;K\mid\widetilde Z)\ge C_{\mathrm{gen}}$.

Moreover, for a threshold decision on atomic claims, let $\alpha_\tau$ denote fact recall and $\beta_\tau$ the false-positive hallucination rate. In the sparse atomic-fact limit $\rho_1=\frac{n_1}{N_1}\to0$, perfect atomic recall, $\alpha_\tau=1$, implies
\[
\beta_\tau
\ge
\exp\left(
-\frac{C-C_{\mathrm{rule}}-C_{\mathrm{gen}}}{n_1}
\right).
\]

\label{thm:gaussian-recall}
Fix $\theta>0$ and condition on $\Theta=\theta$. Let $\sigma_{\mathrm c}\coloneqq\sqrt{\sigma_B^2+\sigma_\theta^2}$. Suppose that
$n_1,n_2,n\to\infty$, $a_N\to a\in[0,1]$,
$N\delta_N\to0$, and that $W_N-\widehat W_N\xrightarrow{\mathbb P}0$.
Then $W_N\xrightarrow{\mathbb P}a\theta$.
Moreover, for every fixed $r\geq0$,
\begin{align}
    \alpha_N^{\mathrm{ato}}(r)
    &\xrightarrow{\mathbb P}
    \alpha^{\mathrm{ato}}(r;a)
    \coloneqq
    \Phi\!\left(\frac{r-a\theta}{\sigma_B}\right)
    +
    \Phi\!\left(\frac{r+a\theta}{\sigma_B}\right)-1,
    \label{eq:gaussian-atomic-limit}\\
    \alpha_N^{\mathrm{comp}}(r)
    &\xrightarrow{\mathbb P}
    \alpha^{\mathrm{comp}}(r;a)\coloneqq
    \Phi\!\left(\frac{r-(1-a)\theta}{\sigma_{\mathrm c}}\right)
    +
    \Phi\!\left(\frac{r+(1-a)\theta}{\sigma_{\mathrm c}}\right)-1,
    \label{eq:gaussian-composite-limit}\\
    \beta_N(r)
    &\xrightarrow{\mathbb P}
    \beta(r)\coloneqq2\Phi(r)-1.
    \label{eq:gaussian-fpr-limit}
\end{align}

<!-- cor-1: body_md -->
\label{cor:optimal-mixture-recall}
Under the conditions of Theorem~\ref{thm:gaussian-recall},
fix a target recall $\alpha_0\in(1/2,1)$. For $a\in[0,1]$, define $r_g(a)\coloneqq\inf\{r\geq0:\alpha^g(r;a)\geq\alpha_0\}$ for $g\in\{\mathrm{ato},\mathrm{comp}\}$, the smallest radius at which the limiting recall of population $g$ reaches $\alpha_0$.
Let $r_{\alpha_0}(a)\coloneqq\max\{r_{\mathrm{ato}}(a),r_{\mathrm{comp}}(a)\}$,
and let $\beta_{\alpha_0}(a)\coloneqq\beta(r_{\alpha_0}(a))$ be its limiting false-positive rate.
Then
\begin{equation}
\label{eq:gaussian-optimal-fpr-lower}
    \min_{a\in[0,1]}\beta_{\alpha_0}(a)
    \geq
    2\Phi\!\left(
        \frac{\theta}{2}
        +
        \frac{\sigma_B+\sigma_{\mathrm c}}{2}
        \Phi^{-1}(\alpha_0)
    \right)-1.
\end{equation}
<!-- end cor-1 -->
<!-- rem-4: body_md -->
If $r_{\mathrm{ato}}(1)>r_{\mathrm{comp}}(1)$, there is a unique
$a_*\in(0,1)$ satisfying $r_{\mathrm{ato}}(a_*)=r_{\mathrm{comp}}(a_*)$. In this case, $\beta_{\alpha_0}(a)$ is strictly decreasing on $[0,a_*]$ and strictly increasing on $[a_*,1]$. Otherwise, its unique minimizer is $a_*=1$.
Moreover, $\sigma_{\mathrm c}>\sigma_B$ and $\alpha_0>1/2$ imply $r_{\mathrm{comp}}(1/2)>r_{\mathrm{ato}}(1/2)$, and hence $a_*>1/2$. Thus, equal recall targets do not imply equal training proportions, rather, the optimal mixture favors composite facts to compensate for their greater dispersion.

<!-- cor-2: body_md -->
\label{cor:calibrated-roc}
Under perfect calibration, for $\tau\in(0,1]$ with $\mathbb{P}(T\geq\tau)>0$,
the precision of the accepted claims satisfies $\mathbb{P}(U=1\mid T\geq\tau)
\geq\tau$,
and therefore
\[
\left[\alpha_{\tau}-\sqrt{\frac{h(\rho)-\mathbb{E}_{T}[h(T)]}{2\rho(1-\rho)}}\right]_{+}\leq\beta_\tau\leq\frac{\rho(1-\tau)}{(1-\rho)\tau}\alpha_\tau.
\]
<!-- end cor-2 -->
<!-- proof-1: body_md -->
We first establish the change-of-measure identities used below.
For any measurable set $A\subseteq[0,1]$, calibration gives
\begin{align*}
\rho P(A)
&=\mathbb E[\mathbf 1\{T\in A\}U]
 =\mathbb E[\mathbf 1\{T\in A\}\mathbb E[U\mid T]]
 =\int_A t\,M_\rho(dt),\\
(1-\rho)Q(A)
&=\mathbb E[\mathbf 1\{T\in A\}(1-U)]
 =\mathbb E[\mathbf 1\{T\in A\}(1-T)]
 =\int_A(1-t)\,M_\rho(dt).
\end{align*}
Thus $\frac{dP}{dM_\rho}(t)=t/\rho$ and
$\frac{dQ}{dM_\rho}(t)=(1-t)/(1-\rho)$, so
\[
\alpha_\tau=\frac1\rho\int_{[\tau,1]}t\,M_\rho(dt),
\qquad
\beta_\tau=\frac1{1-\rho}\int_{[\tau,1]}(1-t)\,M_\rho(dt).
\]
Perfect calibration and the tower property give
\begin{align*}
\mathbb P(U=1\mid T\geq\tau)
&=\mathbb E[U\mid T\geq\tau]\\
&=\mathbb E\!\left[\mathbb E[U\mid T]\mid T\geq\tau\right]\\
&=\mathbb E[T\mid T\geq\tau]\geq\tau.
\end{align*}
Bayes' rule then yields
\[
\mathbb P(U=1\mid T\geq\tau)
=\frac{\rho\alpha_\tau}
{\rho\alpha_\tau+(1-\rho)\beta_\tau}\geq\tau,
\]
which gives the stated upper bound on $\beta_\tau$.

For the lower bound, perfect calibration implies
$U\mid T\sim\mathrm{Bern}(T)$. Therefore,
\[
I(U;T)=H(U)-H(U\mid T)
=h(\rho)-\mathbb E_T[h(T)].
\]
We also need the weighted Pinsker inequality. For
$M_\rho=\rho P+(1-\rho)Q$, Pinsker's inequality gives
\begin{align*}
\mathrm{TV}(P,M_\rho)&=(1-\rho)\mathrm{TV}(P,Q)
 \leq\sqrt{\frac{\mathrm{D_{KL}}\left(P \|M_\rho \right)}2},\\
\mathrm{TV}(Q,M_\rho)&=\rho\mathrm{TV}(P,Q)
 \leq\sqrt{\frac{\mathrm{D_{KL}}\left(Q \|M_\rho \right)}2},
\end{align*}
where $\mathrm{TV}$ denotes total variation distance. Multiplying the squared
inequalities by $2\rho$ and $2(1-\rho)$, respectively, and adding gives
\[
J_\rho(P,Q)\geq
2\rho(1-\rho)^2\mathrm{TV}^2(P,Q)
+2(1-\rho)\rho^2\mathrm{TV}^2(P,Q)
=2\rho(1-\rho)\mathrm{TV}^2(P,Q).
\]
For $A_\tau=\mathbf 1\{T\geq\tau\}$, data processing therefore implies
\[
I(U;T)\geq I(U;A_\tau)
=J_\rho\bigl(\mathrm{Bern}(\alpha_\tau),\mathrm{Bern}(\beta_\tau)\bigr)
\geq2\rho(1-\rho)(\alpha_\tau-\beta_\tau)^2.
\]
Combining this with the entropy identity yields
\[
|\alpha_\tau-\beta_\tau|
\leq\sqrt{\frac{h(\rho)-\mathbb E_T[h(T)]}{2\rho(1-\rho)}}.
\]
The stated lower bound follows because $\beta_\tau\geq0$.

Since $U_i\sim\mathrm{Bern}(\rho)$, the marginal law of $T_i$ is

Let $J\sim\mathrm{Unif}([N])$ be independent of
$(\widetilde Z,K,W)$, and define
$U\coloneqq\mathbf 1\{J\in K\}$ and $T\coloneqq T_J$. Then

\le I(U;T\mid\widetilde Z,J)

\label{eq:sharp-js}
\end{align}
In particular,
\begin{equation}
\left|\mathcal E(\mathcal A)\right|
\le
\sqrt{\frac{I(U;T\mid\widetilde Z,J)}{2\rho(1-\rho)}}.

<!-- proof-3: body_md -->
A direct calculation gives
\begin{align*}
\mathbb E[\hat L_n]
&=\frac1n\sum_{i=1}^N\mathbb E[U_iT_i]
 =\frac1N\sum_{i=1}^N\mathbb E[T_i\mid U_i=1]
 =\mathbb E[T\mid U=1],\\
\mathbb E[\hat L_m]
&=\frac1m\sum_{i=1}^N\mathbb E[(1-U_i)T_i]
 =\frac1N\sum_{i=1}^N\mathbb E[T_i\mid U_i=0]
 =\mathbb E[T\mid U=0].
\end{align*}
To obtain the binary projection, given $T\in[0,1]$, output one with
probability $T$ and zero otherwise, using randomness independent of $U$.
The conditional output laws are
$\mathrm{Bern}(\mathbb E[T\mid U=1])$ and
$\mathrm{Bern}(\mathbb E[T\mid U=0])$.
By Lemma~\ref{cmi-js}, the mutual information between $U$ and this
binary output is the weighted JS divergence between these laws.
Since this randomization uses only $T$, data processing gives
\[
d_{\mathrm{JS},\rho}\bigl(\mathbb E[\hat L_n],\mathbb E[\hat L_m]\bigr)
\leq I(U;T).
\]

Although $U$ is a function of $(J,K)$, uniform subset selection and
$K\perp\!\!\!\perp\widetilde Z$ imply
$\mathbb P(U=1\mid\widetilde Z,J)=n/N=\rho$.
Thus $U\perp\!\!\!\perp(\widetilde Z,J)$, and
\[
I(U;T)\leq I(U;T,\widetilde Z,J)
=I(U;T\mid\widetilde Z,J).
\]
Conditioning on the independent uniform index $J$ gives
\[
I(U;T\mid\widetilde Z,J)
=\frac1N\sum_{i=1}^N I(U_i;T_i\mid\widetilde Z),
\]
completing~Eq.~(\ref{eq:sharp-js}).

For the final bound, we establish the weighted JS--TV inequality.
For any $P,Q$ and $M_\rho=\rho P+(1-\rho)Q$, Pinsker's inequality gives
\begin{align*}
\mathrm{TV}(P,M_\rho)&=(1-\rho)\mathrm{TV}(P,Q)
 \leq\sqrt{\frac{\mathrm{D_{KL}}\left(P \|M_\rho \right)}2},\\
\mathrm{TV}(Q,M_\rho)&=\rho\mathrm{TV}(P,Q)
 \leq\sqrt{\frac{\mathrm{D_{KL}}\left(Q \|M_\rho \right)}2},
\end{align*}
where $\mathrm{TV}$ denotes total variation distance. Consequently,
\begin{align*}
J_\rho(P,Q)
&=\rho\mathrm{D_{KL}}\left(P \|M_\rho \right)+(1-\rho)\mathrm{D_{KL}}\left(Q \|M_\rho \right)\\
&\geq2\rho(1-\rho)^2\mathrm{TV}^2(P,Q)
     +2(1-\rho)\rho^2\mathrm{TV}^2(P,Q)\\
&=2\rho(1-\rho)\mathrm{TV}^2(P,Q).
\end{align*}
Applying this inequality to the two Bernoulli risk laws yields
\[
d_{\mathrm{JS},\rho}\bigl(\mathbb E[\hat L_n],\mathbb E[\hat L_m]\bigr)
\geq2\rho(1-\rho)
\bigl(\mathbb E[\hat L_n]-\mathbb E[\hat L_m]\bigr)^2.
\]
Combining this with~Eq.~(\ref{eq:sharp-js}) and
$\mathcal E(\mathcal A)=\mathbb E[\hat L_m]-\mathbb E[\hat L_n]$
proves~Eq.~(\ref{eq:random-coordinate-bound-full})
and~Eq.~(\ref{eq:random-coordinate-bound}).

For $t^N\in[0,1]^N$ and $k\subseteq[N]$ with $|k|=n$, let
\[
G(t^N,k)=\frac1m\sum_{i\notin k}t_i-\frac1n\sum_{i\in k}t_i.
\]
Fix $\widetilde Z=\tilde z$ and compare the true conditional joint
law of $(T^N,K)$ with the decoupled law
\[
Q_{\tilde z}=P_{T^N\mid\widetilde Z=\tilde z}\otimes P_K.
\]
Under $Q_{\tilde z}$, $K$ is a uniform subset independent of $T^N$,
and $\mathbb E_{Q_{\tilde z}}[G\mid T^N]=0$.
The sampling-without-replacement comparison and Hoeffding's
lemma~\citep{hoeffding1963probability} give, uniformly in $t^N$,
\begin{align*}
\log\mathbb E_K[e^{\lambda G(t^N,K)}]
&\leq\frac{\lambda^2N^2}{8nm^2},\\
\log\mathbb E_K[e^{\lambda G(t^N,K)}]
&\leq\frac{\lambda^2N^2}{8mn^2}.
\end{align*}
Indeed,
\[
G(t^N,K)
=\frac Nm\left(\frac1N\sum_i t_i-\frac1n\sum_{i\in K}t_i\right)
=\frac Nn\left(\frac1m\sum_{i\notin K}t_i-\frac1N\sum_i t_i\right),
\]
so the two bounds follow by applying the sample-mean exponential-moment
bound to the selected and unselected subsets, respectively.

Since $K\perp\!\!\!\perp\widetilde Z$, the KL divergence between the true
conditional law and $Q_{\tilde z}$ equals
$I(T^N;K\mid\widetilde Z=\tilde z)$.
The Donsker--Varadhan variational inequality~\citep{donsker1983asymptotic},
applied to $\lambda G$ and $-\lambda G$, gives, for $\lambda>0$,
\[
\left|\mathbb E[G\mid\widetilde Z=\tilde z]\right|
\leq
\frac{I(T^N;K\mid\widetilde Z=\tilde z)}{\lambda}
+\frac{\lambda N^2}{8nm^2}.
\]
Optimizing over $\lambda>0$ yields
\[
\left|\mathbb E[G\mid\widetilde Z=\tilde z]\right|
\leq\frac Nm
\sqrt{\frac{I(T^N;K\mid\widetilde Z=\tilde z)}{2n}}.
\]
The second exponential-moment bound yields the complementary inequality
with $n,m$ interchanged. Averaging each bound over $\widetilde Z$
and applying Jensen's inequality gives
\[
|\mathcal E(\mathcal A)|
\leq\min\left\{
\frac Nm\sqrt{\frac{\mathrm{eCMI}^\ell_\rho(\mathcal A)}{2n}},
\frac Nn\sqrt{\frac{\mathrm{eCMI}^\ell_\rho(\mathcal A)}{2m}}
\right\},
\]
as claimed.

<!-- rem-5: body_md -->
\label{rem:existing-cmi-bounds}
Our fixed-cardinality selector is close to the leave-$m$-out
supersample setting~\citep{lu2026unified}.
When $N=2n$ and $m=n$, Theorem~\ref{thm:global-ecmi} reduces to
$\left|\mathcal E(\mathcal A)\right|
    \le
    \sqrt{
        \frac{2\mathrm{eCMI}^\ell_{1/2}(\mathcal A)}{n}
    }$, which has the same numerical form as the evaluated pairwise-CMI bound~\citep{harutyunyan2021informationtheoretic,hellstrom2022a,wang2023tighter}. The underlying selector laws, however, are different. The original CMI construction selects exactly one instance from each of $n$ prescribed pairs using independent Bernoulli bits~\citep{steinke2020reasoning}, whereas our selector is uniform over all size-$n$ subsets of $[2n]$. It may therefore select both instances from
one pair and neither from another. Thus, $\rho=1/2$ recovers the balanced JS weighting, but not the same joint selector law or, in general, the same CMI quantity.
Our purpose here is to establish a common selector-information perspective, rather than to improve upon existing generalization bounds. For a comprehensive treatment, including tighter bounds, fast-rate results, and extensions beyond bounded losses, we refer to~\cite{hellstrom2025generalization}.

Hold the conditional score laws $P,Q$ fixed as $\rho\downarrow0$, and
let $M_\rho=\rho P+(1-\rho)Q$. Since
\[
\mathrm{TV}(M_\rho,Q)=\rho\mathrm{TV}(P,Q)\longrightarrow0,
\]
lower semicontinuity of KL divergence gives
\[
\mathrm{D_{KL}}\left(P \|Q \right)
\leq\liminf_{\rho\downarrow0}\mathrm{D_{KL}}\left(P \|M_\rho \right).
\]
By nonnegativity of KL divergence,
\[
\frac{J_\rho(P,Q)}{\rho}
=\mathrm{D_{KL}}\left(P \|M_\rho \right)
+\frac{1-\rho}{\rho}\mathrm{D_{KL}}\left(Q \|M_\rho \right)
\geq\mathrm{D_{KL}}\left(P \|M_\rho \right).
\]
Consequently,
\begin{equation}
\label{eq:sparse-js-kl-liminf}
\liminf_{\rho\downarrow0}\frac{J_\rho(P,Q)}{\rho}
\geq\mathrm{D_{KL}}\left(P \|Q \right).
\end{equation}
If $\mathrm{D_{KL}}\left(P \|Q \right)=\infty$, this already proves the claim in the
extended-real sense.

Now suppose that $\mathrm{D_{KL}}\left(P \|Q \right)<\infty$, so $P\ll Q$.
Expanding the divergences and using $M_\rho=\rho P+(1-\rho)Q$ yields
\begin{align*}
\rho\mathrm{D_{KL}}\left(P \|Q \right)-J_\rho(P,Q)
&=\rho\int\log\frac{dM_\rho}{dQ}\,dP\\
&\quad +(1-\rho)\int\log\frac{dM_\rho}{dQ}\,dQ\\
&=\int\log\frac{dM_\rho}{dQ}\,dM_\rho\\
&=\mathrm{D_{KL}}\left(M_\rho \|Q \right).
\end{align*}
Thus
\[
\frac{J_\rho(P,Q)}{\rho}
=\mathrm{D_{KL}}\left(P \|Q \right)-\frac{\mathrm{D_{KL}}\left(M_\rho \|Q \right)}{\rho}
\leq\mathrm{D_{KL}}\left(P \|Q \right).
\]
Together with~Eq.~(\ref{eq:sparse-js-kl-liminf}), this proves
\[
\lim_{\rho\to0}\frac{J_\rho(P,Q)}{\rho}
=\mathrm{D_{KL}}\left(P \|Q \right).
\]
Lemma~\ref{cmi-js} gives $I(U;T)=J_\rho(P,Q)$, completing the stated
limit. In the finite-KL case, the same identity also implies
\[
\frac{\mathrm{D_{KL}}\left(M_\rho \|Q \right)}{\rho}
=\mathrm{D_{KL}}\left(P \|Q \right)-\frac{J_\rho(P,Q)}{\rho}
\longrightarrow0,
\]
so $\mathrm{D_{KL}}\left(M_\rho \|Q \right)=o(\rho)$ follows from the limit rather than being
assumed.

<!-- proof-6: body_md -->
The claim universe $\widetilde Z$ is fixed, so conditioning on it is
implicit. Recall that $U_i=\mathbf 1\{i\in K\}$, $U=U_J$, and $T=T_J$.
Since $K$ is uniform over the size-$n$ subsets of $[N]$,
$\mathbb P(U_i=1)=\binom{N-1}{n-1}/\binom{N}{n}=\rho$ for every $i$.
Thus $U\perp\!\!\!\perp J$, and the independence of $J$ from $(K,T^N)$ gives
\begin{align*}
J_\rho(P,Q)
&=I(U;T)
\leq I(U;T,J)
=I(U;T\mid J)\\
&=\frac1N\sum_{i=1}^N I(U_i;T_i).
\end{align*}
To bound this sum by the information in the full score vector, note
that $(U_1,\ldots,U_N)$ determines $K$ and conversely. Conditional
entropy subadditivity and the fact that $T_i$ is a coordinate of $T^N$
therefore imply
\[
H(K\mid T^N)
=H(U_1,\ldots,U_N\mid T^N)
\leq\sum_{i=1}^N H(U_i\mid T^N)
\leq\sum_{i=1}^N H(U_i\mid T_i).
\]
Using $H(U_i)=h(\rho)$ and $H(K)=\log\binom{N}{n}$, we obtain
\begin{align*}
\sum_{i=1}^N I(U_i;T_i)
&=Nh(\rho)-\sum_{i=1}^N H(U_i\mid T_i)\\
&\leq Nh(\rho)-H(K\mid T^N)\\
&=I(T^N;K)+Nh(\rho)-\log\binom{N}{n}.
\end{align*}
Since $N\rho=n$, the assumed information budget yields
\begin{equation}
\label{eq:random-fact-score-information-bound}
\frac{J_\rho(P,Q)}{\rho}
\leq\frac{I(T^N;K)}{n}
+\frac{Nh(\rho)-\log\binom{N}{n}}{n}
\leq R+\frac{Nh(\rho)-\log\binom{N}{n}}{n}.
\end{equation}
The correction accounts for the dependence among the fixed-cardinality
membership indicators. As $n\to\infty$ and $\rho\to0$, we also have
$N-n\to\infty$, and Stirling's formula gives
\[
\log\binom{N}{n}
=Nh(\rho)-\frac12\log\bigl(2\pi n(1-\rho)\bigr)
+O\!\left(\frac1n+\frac1{N-n}\right).
\]
Consequently,
\[
\frac{Nh(\rho)-\log\binom{N}{n}}{n}
=\frac{\log\bigl(2\pi n(1-\rho)\bigr)}{2n}
+O\!\left(\frac1{n^2}+\frac1{n(N-n)}\right)
=o(1),
\]
so Eq.~(\ref{eq:random-fact-score-information-bound}) gives
$J_\rho(P,Q)/\rho\leq R+o(1)$.

Thresholding is a post-processing of the score, so the data-processing inequality gives
\[
\frac{J_\rho(P,Q)}{\rho}
\geq \frac{J_\rho\bigl(\mathrm{Bern}(\alpha_\tau),\mathrm{Bern}(\beta_\tau)\bigr)}{\rho}.
\]
Combining this with the preceding bound and dropping the nonnegative
second term in the definition of $J_\rho$ yields
\[
\mathrm{D_{KL}}\left(\mathrm{Bern}(\alpha_\tau) \|\mathrm{Bern}\bigl(\rho\alpha_\tau+(1-\rho)\beta_\tau\bigr) \right)
\leq
\frac{J_\rho\bigl(\mathrm{Bern}(\alpha_\tau),\mathrm{Bern}(\beta_\tau)\bigr)}{\rho}
\leq R+o(1).
\]
For any limiting recall--false-positive pair $(\alpha_\tau,\beta_\tau)$,
lower semicontinuity of KL divergence therefore gives
$\mathrm{D_{KL}}\left(\mathrm{Bern}(\alpha_\tau) \|\mathrm{Bern}(\beta_\tau) \right)\leq R$.
This argument also allows the score laws and threshold to vary along
the sparse sequence.
For $\alpha_\tau>0$, the exponential lower bound follows from
\[
\mathrm{D_{KL}}\left(\mathrm{Bern}(\alpha_\tau) \|\mathrm{Bern}(\beta_\tau) \right)
=-h(\alpha_\tau)-\alpha_\tau\log\beta_\tau
-(1-\alpha_\tau)\log(1-\beta_\tau)
\geq -h(\alpha_\tau)-\alpha_\tau\log\beta_\tau.
\]
Rearranging gives
$\beta_\tau\geq\exp\bigl(-(R+h(\alpha_\tau))/\alpha_\tau\bigr)$.
For $\alpha_\tau=1$, $h(1)=0$ gives $\beta_\tau\geq e^{-R}$.

We first lower-bound the information used for rule recovery and sample selection, then relate the remaining atomic-fact budget to the false-positive rate under perfect recall.
By Fano's inequality,
\begin{align}
I(W;\Theta)
&=
H(\Theta)-H(\Theta\mid W) \notag\\
&\geq
\log N_\Theta
-h(\varepsilon_\Theta)
-\varepsilon_\Theta\log(N_\Theta-1)
=
C_{\mathrm{rule}}.
\label{eq:proof-rule-lower-bound}
\end{align}

Let
\[
\mathrm{eCMI}^\ell_\rho(\mathcal A)
=
I(T^N;K\mid\widetilde Z)
\]
denote the evaluated selector-density CMI associated with the
bounded loss. The global generalization bound gives
\[
|\mathcal E(\mathcal A)|
\leq
\min\left\{
\frac{N}{m}
\sqrt{\frac{\mathrm{eCMI}^\ell_\rho(\mathcal A)}{2n}},
\frac{N}{n}
\sqrt{\frac{\mathrm{eCMI}^\ell_\rho(\mathcal A)}{2m}}
\right\}.
\]
Inverting both inequalities yields
\begin{align}
\mathrm{eCMI}^\ell_\rho(\mathcal A)
&\geq
\max\left\{
\frac{2nm^2}{N^2},
\frac{2mn^2}{N^2}
\right\}
|\mathcal E(\mathcal A)|^2 \notag\\
&=
\frac{2nm\max\{n,m\}}{N^2}
|\mathcal E(\mathcal A)|^2
=
C_{\mathrm{gen}}.
\label{eq:proof-gen-lower-bound}
\end{align}
Since $T^N$ is obtained from $(W,\widetilde Z)$, the data processing
inequality gives
\[
I(W;K\mid\widetilde Z)
\geq
I(T^N;K\mid\widetilde Z)
\geq
C_{\mathrm{gen}}.
\]

By Lemma~\ref{3MI},
\[
I(W;S)
=
I(W;\Theta)
+
I(W;B\mid\Theta)
+
I(W;K\mid\widetilde Z).
\]
Combining this decomposition with $I(W;S)\leq C$,
Eq.~(\ref{eq:proof-rule-lower-bound}), and
Eq.~(\ref{eq:proof-gen-lower-bound}), we obtain
\begin{equation}
I(W;B\mid\Theta)
\leq
C-C_{\mathrm{rule}}-C_{\mathrm{gen}}.
\label{eq:proof-residual-fact-budget}
\end{equation}

It remains to lower-bound the information required for perfect
atomic recall. Let $K_1$ be the atomic-fact selector, so that
\[
K_1
\sim
\operatorname{Unif}
\left\{
A\subseteq[N_1]:|A|=n_1
\right\},
\qquad
\rho_1=\frac{n_1}{N_1}.
\]
Because the atomic candidate pool is fixed and contains distinct
claims, $B$ and $K_1$ are in one-to-one correspondence. Hence
\begin{equation}
I(W;B\mid\Theta)
=
I(W;K_1\mid\Theta).
\label{eq:proof-B-K1-equivalence}
\end{equation}

For a threshold $\tau$ and the model's factuality score $T_j$ of atomic claim $b_j$, define the set of accepted atomic claims by
\[
\widehat K_1
\coloneqq
\{j\in[N_1]:T_j\geq\tau\},
\qquad
L\coloneqq|\widehat K_1|.
\]
The fact recall satisfies
\[
\alpha_\tau
=
\frac{1}{n_1}
\mathbb E\!\left[
|K_1\cap\widehat K_1|
\right].
\]
Therefore, under perfect atomic recall, $\alpha_\tau=1$,
\[
\mathbb E\!\left[
|K_1\setminus\widehat K_1|
\right]
=
n_1(1-\alpha_\tau)
=
0.
\]
Since $|K_1\setminus\widehat K_1|$ is nonnegative, it follows that
\begin{equation}
K_1\subseteq\widehat K_1
\qquad\text{almost surely}.
\label{eq:proof-perfect-recall-containment}
\end{equation}
This is the step at which perfect recall is essential.

The false-positive hallucination rate is
\[
\beta_\tau
=
\frac{1}{N_1-n_1}
\mathbb E\!\left[
|\widehat K_1\setminus K_1|
\right].
\]
By Eq.~(\ref{eq:proof-perfect-recall-containment}),
\[
L
=
n_1+|\widehat K_1\setminus K_1|
\qquad\text{almost surely},
\]
and consequently
\begin{equation}
\mathbb E[L]
=
n_1+(N_1-n_1)\beta_\tau.
\label{eq:proof-accepted-size}
\end{equation}

We now bound $H(K_1\mid W,\Theta)$ by counting the $n_1$-subsets compatible with the accepted set.
Since $K_1$ is independent of $\Theta$ and is uniform over all
$n_1$-subsets of $[N_1]$,
\[
H(K_1\mid\Theta)
=
\log\binom{N_1}{n_1}.
\]
Moreover, conditional on $(W,\Theta)$, the accepted set
$\widehat K_1$ is known. By
Eq.~(\ref{eq:proof-perfect-recall-containment}), $K_1$ must be an
$n_1$-subset of $\widehat K_1$. Thus,
\[
H(K_1\mid W,\Theta)
\leq
\mathbb E\!\left[
\log\binom{L}{n_1}
\right].
\]
Using Eq.~(\ref{eq:proof-B-K1-equivalence}), we obtain
\begin{align}
I(W;B\mid\Theta)
&\geq
\log\binom{N_1}{n_1}
-
\mathbb E\!\left[
\log\binom{L}{n_1}
\right] \notag\\
&=
\mathbb E\!\left[
\log
\frac{\binom{N_1}{n_1}}
     {\binom{L}{n_1}}
\right].
\label{eq:proof-combinatorial-information}
\end{align}

For every $n_1\leq L\leq N_1$,
\begin{align}
\frac{\binom{N_1}{n_1}}{\binom{L}{n_1}}
&=
\prod_{j=0}^{n_1-1}
\frac{N_1-j}{L-j}
\geq
\left(\frac{N_1}{L}\right)^{n_1}.
\end{align}
Therefore,
\begin{align}
I(W;B\mid\Theta)
&\geq
n_1\,
\mathbb E\!\left[
\log\frac{N_1}{L}
\right] \notag\\
&\geq
n_1
\log\frac{N_1}{\mathbb E[L]} \notag\\
&=
n_1
\log
\frac{1}
{\rho_1+(1-\rho_1)\beta_\tau},
\label{eq:proof-fact-information-lower-bound}
\end{align}
where the second inequality follows from Jensen's inequality and
the last equality follows from
Eq.~(\ref{eq:proof-accepted-size}).

Combining Eq.~(\ref{eq:proof-residual-fact-budget}) and
Eq.~(\ref{eq:proof-fact-information-lower-bound}) gives
\[
n_1
\log
\frac{1}
{\rho_1+(1-\rho_1)\beta_\tau}
\leq
C-C_{\mathrm{rule}}-C_{\mathrm{gen}}.
\]
Equivalently,
\[
\rho_1+(1-\rho_1)\beta_\tau
\geq
\exp\left(
-\frac{
C-C_{\mathrm{rule}}-C_{\mathrm{gen}}
}{n_1}
\right),
\]
and hence the following finite-density bound holds:
\begin{equation}
\beta_\tau
\geq
\max\left\{
0,\,
\frac{
\exp\left(
-\frac{
C-C_{\mathrm{rule}}-C_{\mathrm{gen}}
}{n_1}
\right)
-\rho_1
}
{1-\rho_1}
\right\}.
\label{eq:proof-finite-density-fpr}
\end{equation}

Finally, taking the sparse atomic-fact limit $\rho_1\to0$ in
Eq.~(\ref{eq:proof-finite-density-fpr}) yields
\[
\beta_\tau
\geq
\exp\left(
-\frac{
C-C_{\mathrm{rule}}-C_{\mathrm{gen}}
}{n_1}
\right),
\]
which proves the result.

Throughout the proof, all probabilities and expectations are conditional
on $\Theta=\theta>0$, unless specified otherwise. In particular,
$\sigma_{\mathrm c}$ is fixed. We assume $\sigma_B>0$ and use the independence of $K$ from
$(\Theta,B,X)$.

\medskip
\noindent\textit{Convergence of the learned parameter.}
Let
\[
\overline Z_N\coloneqq\frac1N\sum_{i=1}^{n_1}\sum_{j=1}^{n_2}
\widetilde Z_{ij},
\qquad
\overline Z_{K,N}\coloneqq\frac1n\sum_{i=1}^{n_1}\sum_{j=1}^{n_2}
K_{ij}\widetilde Z_{ij}
\]
be the averages of the full and selected composite pools.
Since $\widetilde Z_{ij}=B_i+X_j$, the laws of large numbers give
\[
\overline Z_N
=\frac1{n_1}\sum_{i=1}^{n_1}B_i
+\frac1{n_2}\sum_{j=1}^{n_2}X_j
\xrightarrow{\mathbb P}\theta.
\]
Conditional on $(B,X)$, the selected entries form a simple random sample
without replacement from the $N$ composite values. Hence, for $N>1$,
\[
\mathbb E[\overline Z_{K,N}\mid B,X]=\overline Z_N,\qquad
\operatorname{Var}(\overline Z_{K,N}\mid B,X)
=\frac{1-n/N}{n(N-1)}
\sum_{i=1}^{n_1}\sum_{j=1}^{n_2}
(\widetilde Z_{ij}-\overline Z_N)^2.
\]
The empirical variance of the full Cartesian pool is the sum of the
empirical variances of the atomic values and the rule instances: the
cross term vanishes when summed over all $(i,j)$. Taking expectations,
\[
\mathbb E\!\left[\frac1N\sum_{i,j}
(\widetilde Z_{ij}-\overline Z_N)^2\right]
=\left(1-\frac1{n_1}\right)\sigma_B^2
+\left(1-\frac1{n_2}\right)\sigma_\theta^2
\le\sigma_{\mathrm c}^2.
\]
Consequently,
\[
\mathbb E[(\overline Z_{K,N}-\overline Z_N)^2]
\le\frac{N\sigma_{\mathrm c}^2}{n(N-1)}\longrightarrow0.
\]
Thus $\overline Z_{K,N}\xrightarrow{\mathbb P}\theta$. It follows that
\[
\widehat W_N
=(1-a_N)\frac1{n_1}\sum_{i=1}^{n_1}B_i
+a_N\overline Z_{K,N}
\xrightarrow{\mathbb P}a\theta.
\]
The assumption $W_N-\widehat W_N\xrightarrow{\mathbb P}0$ now yields
$W_N\xrightarrow{\mathbb P}a\theta$. No independence or distributional
assumption on $W_N-\widehat W_N$ is needed.

\medskip
\noindent\textit{Convergence of the two recalls.}
Define the empirical distribution functions of the atomic values, rule
instances, and full composite pool, respectively, by
\begin{align*}
F_N^{\mathrm{ato}}(x)&\coloneqq\frac1{n_1}\sum_{i=1}^{n_1}
\mathbf 1\{B_i\le x\},\qquad
F_{X,N}(x)\coloneqq\frac1{n_2}\sum_{j=1}^{n_2}\mathbf 1\{X_j\le x\},\\
F_N^{\mathrm{comp}}(x)&\coloneqq\frac1N\sum_{i=1}^{n_1}\sum_{j=1}^{n_2}
\mathbf 1\{\widetilde Z_{ij}\le x\}.
\end{align*}
The Glivenko--Cantelli theorem applied separately to the two marginal
samples gives
\[
\sup_x\left|F_N^{\mathrm{ato}}(x)-\Phi(x/\sigma_B)\right|
\xrightarrow{\mathbb P}0,\qquad
\sup_x\left|F_{X,N}(x)-\Phi((x-\theta)/\sigma_\theta)\right|
\xrightarrow{\mathbb P}0.
\]
These conclusions hold jointly by the union bound and do not require
independence between the two samples.

The Cartesian structure of the composite pool gives
\[
F_N^{\mathrm{comp}}(x)=\frac1{n_2}\sum_{j=1}^{n_2}F_N^{\mathrm{ato}}(x-X_j).
\]
Replacing $F_N^{\mathrm{ato}}$ by its Gaussian limit in this average
incurs an error bounded by
$\sup_y|F_N^{\mathrm{ato}}(y)-\Phi(y/\sigma_B)|$.
For the resulting average, the identity
\[
\frac1{n_2}\sum_{j=1}^{n_2}\Phi\!\left(\frac{x-X_j}{\sigma_B}\right)
=\int_{\mathbb R}F_{X,N}(x-b)\frac1{\sigma_B}\phi(b/\sigma_B)\,db
\]
and the Gaussian convolution formula yield
\begin{align*}
\sup_x\left|F_N^{\mathrm{comp}}(x)
-\Phi\!\left(\frac{x-\theta}{\sigma_{\mathrm c}}\right)\right|
&\le\sup_x\left|F_N^{\mathrm{ato}}(x)-\Phi(x/\sigma_B)\right|\\
&\quad+\sup_x\left|F_{X,N}(x)-\Phi((x-\theta)/\sigma_\theta)\right|
\xrightarrow{\mathbb P}0.
\end{align*}
Thus the empirical distribution of the composite pool converges to
$\mathcal N(\theta,\sigma_{\mathrm c}^2)$ even without assuming that
$B$ and $X$ are independent given $\Theta$.

To evaluate recall at the data-dependent center, let $F_N$ denote either
$F_N^{\mathrm{ato}}$ or $F_N^{\mathrm{comp}}$, and let $F$ denote its
continuous Gaussian limit established above. The corresponding empirical
proportion in $[W_N-r,W_N+r]$ equals
$F_N(W_N+r)-F_N((W_N-r)^-)$, where $F_N(t^-)$ is the left limit at $t$.
Continuity of $F$ gives the pathwise bound
\[
\left|F_N(W_N+r)-F_N((W_N-r)^-)
-\bigl(F(W_N+r)-F(W_N-r)\bigr)\right|
\le2\sup_x|F_N(x)-F(x)|.
\]
Combining uniform convergence with $W_N\xrightarrow{\mathbb P}a\theta$
therefore proves, for every fixed $r\ge0$,
\begin{align*}
\alpha_N^{\mathrm{ato}}(r)
&\xrightarrow{\mathbb P}
\Phi\!\left(\frac{a\theta+r}{\sigma_B}\right)
-\Phi\!\left(\frac{a\theta-r}{\sigma_B}\right)
=\alpha^{\mathrm{ato}}(r;a),\\
\alpha_N^{\mathrm{comp}}(r)
&\xrightarrow{\mathbb P}
\Phi\!\left(\frac{a\theta+r-\theta}{\sigma_{\mathrm c}}\right)
-\Phi\!\left(\frac{a\theta-r-\theta}{\sigma_{\mathrm c}}\right)
=\alpha^{\mathrm{comp}}(r;a).
\end{align*}
The final identities use $\Phi(-x)=1-\Phi(x)$ and also cover $r=0$.
No independence between $W_N$ and the facts is used. Since recall is
computed over the full composite pool, no condition on $N-n$ is needed.

\medskip
\noindent\textit{Convergence of the false-positive rate.}
For this paragraph, probabilities involving $Z$ refer only to the fresh
draw $Z\sim\mathcal N(W_N,1)$ with $(B,X,W_N)$ held fixed, as in the
definition of $\beta_N$.
Define $\varepsilon_N\coloneqq\mathbb P(Z\in\mathcal T_N)$.
The total length of $\mathcal T_N$ is at most
$2(n_1+N)\delta_N$, and the density of $Z$ is everywhere at most
$1/\sqrt{2\pi}$. Since $n_1\le N$, it follows that
\[
0\le\varepsilon_N
\le\frac{2(n_1+N)\delta_N}{\sqrt{2\pi}}
\le\frac{4N\delta_N}{\sqrt{2\pi}}
\longrightarrow0.
\]
Moreover, $\mathbb P(|Z-W_N|\le r)=2\Phi(r)-1=\beta(r)$ for every
realized $W_N$. For all sufficiently large $N$, the definition of
$\beta_N$ therefore gives
\begin{align*}
|\beta_N(r)-\beta(r)|
&=\frac{\left|\beta(r)\varepsilon_N
-\mathbb P\bigl(|Z-W_N|\le r,\ Z\in\mathcal T_N\bigr)\right|}
{1-\varepsilon_N}\\
&\le\frac{\varepsilon_N}{1-\varepsilon_N}
\longrightarrow0.
\end{align*}
The bound is uniform over the realized facts and learned parameter.
It proves the stated convergence in probability and completes the proof.

For each $a\in[0,1]$ and $g\in\{\mathrm{ato},\mathrm{comp}\}$,
the function $r\mapsto\alpha^g(r;a)$ is continuous and strictly
increasing from $0$ to $1$. Consequently, $r_g(a)$ is finite and is

In particular, $r_{\alpha_0}(a)$ satisfies both recall constraints.
The recall formulas are continuously differentiable and their
derivatives with respect to $r$ are positive, so the implicit function
theorem also makes $r_g(a)$ continuous in $a$.
Thus $\beta_{\alpha_0}(a)$ is continuous on $[0,1]$ and attains its
minimum there.

Fix any $a\in[0,1]$ and any radius $r$ satisfying both recall
constraints. Since $\Phi(x)\le1$, the limiting recall formulas imply
\[
\alpha_0\le\alpha^{\mathrm{ato}}(r;a)
\le\Phi\!\left(\frac{r-a\theta}{\sigma_B}\right),

\alpha_0\le\alpha^{\mathrm{comp}}(r;a)
\le\Phi\!\left(\frac{r-(1-a)\theta}{\sigma_{\mathrm c}}\right).
\]
Applying the increasing function $\Phi^{-1}$ yields

r\ge a\theta+\sigma_B\Phi^{-1}(\alpha_0),

Adding these inequalities and dividing by two eliminates $a$:
