<!-- theorem-1: body_md -->
\label{thm:main}
Suppose that all $N_i^j$ are convex on $[0,\infty)$ and normalized by
\begin{equation}\label{eq:normalization}
                    \inf\{t\ge0:N_i^j(t)\ge1\}=1.
\end{equation}
Then, for every real array $A$, $d\ge1$ and $p\ge2$,
\begin{equation}\label{eq:general-natural-moment-formula}
 \|S_A(X)\|_p\asymp_d
                   \sum_{\mathcal J\in\mathcal P_d}\|A\|_{\mathcal J,p}^{N}.
\end{equation}
<!-- end theorem-1 -->
<!-- theorem-2: body_md -->
\label{thm:regular}
The conclusion of the preceding theorem also holds for independent
symmetric variables satisfying
\begin{equation}\label{eq:regular-growth}
 \|X_i^j\|_{2p}\le K\|X_i^j\|_p\quad(p\ge1),\qquad
                         \|X_i^j\|_2=1/e ,
\end{equation}
with constants depending only on $d,K$:
\begin{equation}\label{eq:regular-moments}
 \|S_A(X)\|_p\asymp_{d,K}
                   \sum_{\mathcal J\in\mathcal P_d}\|A\|_{\mathcal J,p}^{N},
 \qquad p\ge2.
\end{equation}
All moments are assumed finite. The functions $N_i^j$ in this formula
are the actual tail exponents of $X_i^j$ and need not be convex.

<!-- lemma-1: body_md -->
\label{lem:dual-budget}
Let $\phi_i$ be nonnegative even convex functions with $\phi_i(0)=0$,
possibly taking the value $+\infty$. Put
\[
 F(u)=\sum_i\phi_i(u_i),\qquad
 K_p(v)=\sup_{F(u)\le p}\langle v,u\rangle,\qquad
 \phi_i^*(t)=\sup_{u\in\mathbb R}\{tu-\phi_i(u)\}.
\]
If $K_p(v)\le M$ and $M>0$, then
\begin{equation}\label{eq:dual-budget}
                  \sum_i\phi_i^*(pv_i/M)\le p.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
If $F(u)\le p$, then $(p/M)\langle v,u\rangle-F(u)\le p$.
If $p<F(u)<\infty$, convexity gives $F(pu/F(u))\le p$, and consequently
$(p/M)\langle v,u\rangle\le F(u)$.
Taking the supremum over $u$ with finite $F(u)$ proves the claim.

<!-- lemma-2: body_md -->
\label{lem:root-mgf}
Let $\psi(t)=\log\mathbb E e^{tY}$. For every $t\in\mathbb R$, real random variable
$H\in L_2$ and $\lambda\ge0$,
\begin{equation}\label{eq:root-mgf-l2}
 \psi(t)\le C\phi^*(t),\qquad
 \mathbb E\psi(\lambda H)\le C\phi^*(\lambda\|H\|_2),
\end{equation}
where $C$ is universal.
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
The key point is that $q\mapsto\phi^*(\sqrt q)$ is concave.
This will allow Jensen's inequality to replace $H$ by its $L_2$ norm.
To prove concavity, we use the identity
\begin{equation}\label{eq:root-tail-duality}
 \phi^*(t)=\inf_{a\ge1}
       \left\{\frac{t^2}{4a}+F^*(a)\right\},
 \qquad F^*(a)=\sup_{s\ge0}\{as-F(s)\}.
\end{equation}
The upper bound follows from $F(u^2)\ge au^2-F^*(a)$.
For $t>0$, the maximum of $tu-F(u^2)$ is attained at some $u>0$,
because $F(u^2)\ge u^2$ and $F(u^2)=u^2$ near zero.
The optimality condition gives $a\in\partial F(u^2)$ with $t=2au$.
Here $a\ge1$, and equality in convex duality gives
$F^*(a)=au^2-F(u^2)$. Substitution yields the opposite inequality.
At $t=0$, choose $a=1$; negative $t$ follow by symmetry.

Thus $q\mapsto\phi^*(\sqrt q)$ is concave: the right side of
\eqref{eq:root-tail-duality} is an infimum of affine functions of $q=t^2$.
Since this function vanishes at zero, we have
\begin{equation}\label{eq:root-jensen}
 \phi^*(2t)\le4\phi^*(t),\qquad
 \mathbb E\phi^*(\lambda H)\le\phi^*(\lambda\|H\|_2).
\end{equation}
Also $\phi^*(t)\le t^2/4$, with equality for $|t|\le2$.

The tail definition implies
$\mathbb P(\phi(Y)>s)\le e^{-s}$ for $s\ge1$, including at possible atoms.
Consequently
\[
 \mathbb E e^{\phi(Y)/2}
 =1+\frac12\int_0^\infty e^{s/2}\mathbb P(\phi(Y)>s)\,ds
 \le e^{1/2}+e^{-1/2}<3.
\]
Young's inequality now gives
$\psi(t)\le\log3+\phi^*(2t)/2$.
For $|t|\ge1$, combine this with \eqref{eq:root-jensen} and
$\phi^*(t)\ge1/4$.
For $|t|\le1$, symmetry and the power series of the hyperbolic cosine give
\[
 \psi(t)\le\mathbb E(\cosh(tY)-1)
          \le t^2(\mathbb E\cosh Y-1)\le Ct^2=4C\phi^*(t).
\]
The bound on $\mathbb E\cosh Y=\mathbb E e^Y$ follows from the preceding estimate
at $t=1$ and $\phi^*(2)=1$. This proves the first part of
\eqref{eq:root-mgf-l2}; the second follows from \eqref{eq:root-jensen}.

<!-- proposition-1: body_md -->
\label{prop:root-contraction}
Writing $k=|\mathcal J|$, we have
\begin{equation}\label{eq:root-contraction}
 \mathbb E\|B\|^{Q}_{\mathcal J,p;\mathbf s}
 \le(C+k/2)
        \sum_{\mathcal R\subseteq\mathcal J}
          \|A\|^{Q}_{\mathcal J_{\mathcal R},p;\mathbf s_{\mathcal R}}.
\end{equation}
In particular,
\begin{equation}\label{eq:expected-contraction}
                 \mathbb E \|B\|^{Q}_{p}\le C_m \|A\|^{Q}_{p}.
\end{equation}
<!-- end proposition-1 -->
<!-- proof-3: body_md -->
\noindent\textbf{1. Shift Gaussian parameters to the chosen blocks.}
Let $\mu$ be the product law of independent standard Gaussian arrays
$G^I$, one array for each block $I\in\mathcal J$.
For admissible parameters $x=(x^I)_{I\in\mathcal J}$, let $\rho_x$ be the law
of $(x^I+G^I)_{I\in\mathcal J}$. Since $\widehat Q(t)\ge t^2$,
\begin{equation}\label{eq:shift-cost}
       D(\rho_x\Vert\mu)=\frac12\sum_{I\in\mathcal J}\|x^I\|_2^2\le kp/2.
\end{equation}
Define
\[
 h_\ell(\theta)=
       \sum_{\mathbf i_{[m-1]}}a_{\mathbf i_{[m-1]},\ell}
                          \prod_{I\in\mathcal J}\theta^I_{\mathbf i_I},
 \qquad v_\ell(x)^2=\mathbb E_{\rho_x}h_\ell^2.
\]
Independence and centering give $\mathbb E_{\rho_x}h_\ell=h_\ell(x)$.
Thus averaging preserves the coefficient at the chosen point.

\par\smallskip\noindent\textbf{2. Identify the norms created by the second moment.}
In each block expand $x^I+G^I$, choosing either $x^I$ or $G^I$.
The subfamily $\mathcal R$ records the blocks in which the Gaussian
term was selected. This gives
\begin{equation}\label{eq:variance-expansion}
 v_\ell(x)^2=\sum_{\mathcal R\subseteq\mathcal J}c_{\mathcal R,\ell}(x)^2,
\end{equation}
where, with $U=\bigcup_{I\in\mathcal R}I$,
\[
 c_{\mathcal R,\ell}(x)=
 \left(\sum_{\mathbf i_U}
   \left[\sum_{\mathbf i_{[m-1]\setminus U}}
      a_{\mathbf i_{[m-1]},\ell}
            \prod_{I\in\mathcal J\setminus\mathcal R}x^I_{\mathbf i_I}
   \right]^2\right)^{1/2}.
\]
Terms with different $\mathcal R$ are orthogonal, because some independent
centered Gaussian block occurs in just one of them.
For fixed $\mathcal R$, the covariances of the Gaussian coordinates
are Kronecker deltas, which gives the displayed sum of squares.

Let $K_p$ in Lemma~\ref{lem:dual-budget} use
$\phi_\ell=\widehat Q_\ell^m$.
The Gaussian blocks contribute a Euclidean norm.
They can therefore be joined to the last index in the block
$J_{\mathcal R}$, with that last index distinguished.
Rowwise Euclidean duality gives
\begin{equation}\label{eq:merged-block}
 K_p(c_{\mathcal R}(x))
       \le\|A\|^{Q}_{\mathcal J_{\mathcal R},p;\mathbf s_{\mathcal R}}.
\end{equation}
To see this, take $u_\ell\ge0$ with $\sum_\ell\phi_\ell(u_\ell)\le p$.
In the new block $J_{\mathcal R}$ choose the row indexed by $\ell$
in the direction of the vector inside the brackets defining
$c_{\mathcal R,\ell}$, with length $u_\ell$; use zero when this vector
vanishes. This is an admissible new block and gives the contraction
$\sum_\ell u_\ell c_{\mathcal R,\ell}(x)$.
Nonnegative $u$ suffice, since the constraint is invariant under
coordinatewise changes of signs.

\par\smallskip\noindent\textbf{3. Choose one scale for all admissible parameters.}
Denote the sum on the right of \eqref{eq:root-contraction} by $M_0$.
Since $v(x)\le\sum_{\mathcal R}c_{\mathcal R}(x)$ coordinatewise,
monotonicity and subadditivity of $K_p$ imply $K_p(v(x))\le M_0$.
If $M_0=0$, the term $\mathcal R=\mathcal J$ forces $A=0$.
Otherwise set $\lambda=p/M_0$. This choice depends only on $A,p$
and the partition, not on $x$.
Lemmas~\ref{lem:dual-budget} and \ref{lem:root-mgf} then give,
uniformly in admissible $x$,
\begin{equation}\label{eq:mgf-penalty}
 \mathbb E_{\rho_x}\sum_\ell\psi_\ell(\lambda h_\ell)
 \le C\sum_\ell\phi_\ell^*(\lambda v_\ell(x))\le Cp,
 \qquad \psi_\ell(t)=\log\mathbb E e^{tY_\ell^m}.
\end{equation}

\par\smallskip\noindent\textbf{4. Apply the entropy inequality before taking the supremum.}
On the original probability space of $Y^m$, set
\[
 Z(Y^m)=\int\exp\left\{\lambda\sum_\ell Y_\ell^m h_\ell(\theta)
                  -\sum_\ell\psi_\ell(\lambda h_\ell(\theta))\right\}
                                                    \,d\mu(\theta).
\]
Independence and Tonelli's theorem give $\mathbb E Z=1$, so $0<Z<\infty$
almost surely. Moreover $\mathbb E_\mu h_\ell=0$ and $\psi_\ell(t)\le Ct^2$,
while $h_\ell$ is a Gaussian polynomial. Jensen's inequality therefore
gives the finite deterministic lower bound
\[
 \log Z\ge-\sum_\ell\mathbb E_\mu\psi_\ell(\lambda h_\ell).
\]
Also $(\log Z)_+\le Z$. Thus $\log Z$ is integrable and
$\mathbb E\log Z\le0$.

Apply \eqref{eq:entropy} to the exponent defining $Z$, with
$(\rho,\mu)=(\rho_x,\mu)$. Its integral under $\rho_x$ is finite,
by the polynomial form of $h_\ell$ and the quadratic upper bound on
$\psi_\ell$. Equations \eqref{eq:shift-cost} and \eqref{eq:mgf-penalty}
give, on the single event $\{Z<\infty\}$, for every admissible $x$,
\[
 \lambda\sum_\ell Y_\ell^m h_\ell(x)
                     \le(k/2+C)p+\log Z(Y^m).
\]
The right-hand side no longer depends on $x$.
We may therefore take the supremum over all admissible blocks on the left.
Taking expectation uses $\mathbb E\log Z\le0$; dividing by $\lambda=p/M_0$
proves \eqref{eq:root-contraction}.
Summing over the finitely many partitions and distinguished indices
proves \eqref{eq:expected-contraction}; their number depends only on $m$.

<!-- theorem-3: body_md -->
\label{thm:root-moments}
For the independent variables with tails $e^{-N_i^j(t^2)}$,
\begin{equation}\label{eq:root-tail-moments}
                    \|S_A(Y)\|_p\asymp_d \|A\|^{Q}_{p},\qquad p\ge2.
\end{equation}
In particular Corollary~\ref{thm:rademacher} holds in every order.
<!-- end theorem-3 -->
<!-- proof-4: body_md -->
The lower bound is \eqref{eq:known-lower}. For the upper bound in order
one, apply \eqref{eq:strong-weak} to a single coefficient vector.
The mean is at most $C\|A\|_2$ by independence, symmetry and the uniform
second-moment bound. The unit Euclidean ball is contained in
$\mathcal B^Q_{\{1\},1}(p)$, so this term is bounded by $C\|A\|_p^Q$.

Suppose the upper bound holds in order $m-1$, and condition on $Y^m$.
The induction hypothesis and Minkowski's inequality give
\[
 \|S_A(Y)\|_p
       \le C_{m-1}\sum_{\mathcal J,\mathbf s}
                    \big\|\|B\|^{Q}_{\mathcal J,p;\mathbf s}\big\|_p.
\]
For a fixed partition and choice of distinguished indices, the coefficient set
$\{(h_\ell(x))_\ell:x^I\in\mathcal B^Q_{I,s_I}(p)\}$ is compact and symmetric.
By \eqref{eq:strong-weak},
\[
 \big\|\|B\|^{Q}_{\mathcal J,p;\mathbf s}\big\|_p
 \le L\left[\mathbb E\|B\|^{Q}_{\mathcal J,p;\mathbf s}
       +\|A\|^{Q}_{\mathcal J\cup\{\{m\}\},p;(\mathbf s,m)}\right].
\]
The second term is exactly the weak term in \eqref{eq:strong-weak}:
it adds a vector with $\sum_\ell\widehat Q_\ell^m(z_\ell)\le p$.
Summing and using \eqref{eq:expected-contraction} yields
\[
 \|S_A(Y)\|_p
 \le LC_{m-1}\big[\mathbb E \|B\|^{Q}_{p}+\|A\|^{Q}_{p}\big]
 \le C_m \|A\|^{Q}_{p}.
\]
This completes the induction independently of the general-tail upper bound.

<!-- lemma-3: body_md -->
\label{lem:product}
For every real tensor $A$ and $p\ge1$,
\begin{equation}\label{eq:product-comparison}
 \|S_A(X)\|_p
 \le2^d\left\|
       \sum_{\mathbf i}a_{\mathbf i}
            \prod_{j=1}^d Y_{i_j}^{j,0}Y_{i_j}^{j,1}
                                      \right\|_p.
\end{equation}
<!-- end lemma-3 -->
<!-- proof-5: body_md -->
For one pair, independence gives
\[
 \mathbb P(|Y^0Y^1|\ge t)
 \ge\mathbb P(|Y^0|\ge\sqrt t,\ |Y^1|\ge\sqrt t)
 =e^{-2N(t)}\ge e^{-N(2t)}
 =\mathbb P(|X|\ge2t),
\]
using $N(2t)\ge2N(t)$.
Thus $|X|\le_{\mathrm{st}}2|Y^0Y^1|$.
Couple these nonnegative variables in this order separately for each
$(i,j)$, and realize symmetric variables by independent signs times
their absolute values. Conditional on the coupled absolute values,
the contraction principle for signs bounds the smaller coefficients
by the larger ones in $L_p$.

More explicitly, if $0\le a_i\le b_i$, auxiliary independent signs
with means $a_i/b_i$ express the sum with coefficients $a_i$
as a conditional average of sums with coefficients $b_i$.
Jensen's inequality and invariance of independent uniform signs under
multiplication by independent signs give the contraction; zero $b_i$
may be omitted. Apply this argument to one family at a time, conditioning
on the other families. The factor two is paid once per family, giving
$2^d$, rather than once per coordinate.
The products of the two independent signs in $Y^0Y^1$ are again
independent uniform signs. This proves \eqref{eq:product-comparison}.

<!-- proposition-2: body_md -->
\label{prop:diagonal-collapse}
For every normalized family $N_i^j$, every real tensor $A$ and $p\ge2$,
\begin{equation}\label{eq:diagonal-collapse-proved}
                  \|\widetilde A\|^{Q}_{p}\le C_d \|A\|^{N}_{p}.
\end{equation}
<!-- end proposition-2 -->
<!-- proof-6: body_md -->
Fix a partition $\mathcal Q$ of $\Omega$, distinguished indices
$s_B=(j_B,a_B)\in B$, and admissible parameters
$x^B\in\mathcal B^Q_{B,s_B}(p)$ for $B\in\mathcal Q$.
Let $r_i^B$ be the Euclidean norm of row $i$ of $x^B$ along $s_B$.
Its constraint is
\[
 \sum_i\phi_i^{j_B}(r_i^B)\le p,\qquad
 \phi_i^j(t)=
 \begin{cases}t^2,&0\le t\le1,\\N_i^j(t^2),&t>1.\end{cases}
\]

\par\smallskip\noindent\textbf{1. Separate row lengths from row directions.}
The cutoff is one because that is the Rademacher row constraint.
For each row set
\[
 u_i^B=r_i^B\mathbf 1_{\{r_i^B>1\}},\qquad
 v^{B,0}_{\mathbf i_B}
       =x^B_{\mathbf i_B}\mathbf 1_{\{r_{i_{s_B}}^B\le1\}},
 \qquad
 v^{B,1}_{\mathbf i_B}=
 \begin{cases}
 x^B_{\mathbf i_B}/r_{i_{s_B}}^B,&r_{i_{s_B}}^B>1,\\
 0,&r_{i_{s_B}}^B\le1 .
 \end{cases}
\]
Then $x^B=v^{B,0}+u^Bv^{B,1}$, with $u^B$ acting only in
the distinguished coordinate. Both $v^{B,0}$ and $v^{B,1}$ belong
to $C^\varepsilon_{B,s_B}(p)$: their row norms are at most one,
and their squared Euclidean norms are at most $p$.
For $v^{B,1}$, the latter assertion follows from
\[
 \#\{i:r_i^B>1\}\le
                 \sum_{i:r_i^B>1}N_i^{j_B}((r_i^B)^2)\le p.
\]
The multipliers satisfy
\begin{equation}\label{eq:row-multipliers}
 u_i^B\in\{0\}\cup(1,\infty),\qquad
                    \sum_iN_i^{j_B}((u_i^B)^2)\le p.
\end{equation}

\par\smallskip\noindent\textbf{2. Check the tail constraint for each combined length.}
Expand the product over $B\in\mathcal Q$. A term selects
$u^Bv^{B,1}$ from some blocks $\mathcal S\subseteq\mathcal Q$
and $v^{B,0}$ from the others.
On the support of $\widetilde A$, the indices $i_{j,0}$ and $i_{j,1}$
are equal. Combine all selected multipliers with original coordinate
$j$ into $w_i^j$. There are at most two such multipliers: each of the
two copies of $j$ belongs to one block and can be distinguished only there.
Let $J\subseteq[d]$ consist of coordinates with at least one multiplier;
for the other coordinates set $w_i^j=1$.

For every $j\in J$,
\begin{equation}\label{eq:collapsed-budget}
                         \sum_i\widehat N_i^j(w_i^j)\le p.
\end{equation}
With one multiplier, this follows from $N_i^j(u_i)\le N_i^j(u_i^2)$.
With two, convexity and monotonicity give
\[
 N_i^j(u_iv_i)
 \le N_i^j((u_i^2+v_i^2)/2)
 \le\tfrac12N_i^j(u_i^2)+\tfrac12N_i^j(v_i^2).
\]
Every product is zero or greater than one, so
$\widehat N_i^j(w_i^j)=N_i^j(w_i^j)$, also at zero.
Now sum and use \eqref{eq:row-multipliers}.
This is an exact budget bound; no factor depending on $p$ is lost.

\par\smallskip\noindent\textbf{3. Remove the extra indices using random signs.}
Set $a^w_{\mathbf i}=a_{\mathbf i}\prod_{j\in J}w_{i_j}^j$.
The chosen term is a contraction of $\widetilde A^w$ against
Rademacher-admissible blocks $v^{B,0}$ or $v^{B,1}$.
The known lower bound \eqref{eq:known-lower}, in order $2d$, gives
\begin{align}\label{eq:bernoulli-collapse}
 |\text{term}|
 &\le\|\widetilde A^w\|^{\varepsilon}_{\mathcal Q,p;\mathbf s}
 \le L_{2d}
       \left\|\sum_{\mathbf i}a^w_{\mathbf i}
            \prod_{j=1}^d\varepsilon_{i_j}^{j,0}
                              \varepsilon_{i_j}^{j,1}\right\|_p
   =L_{2d}\|S_{A^w}(\eta)\|_p .
\end{align}
Here the superscript $\varepsilon$ denotes the functional with
$C^\varepsilon(p)$, and
$\eta_i^j=\varepsilon_i^{j,0}\varepsilon_i^{j,1}$
are independent Rademacher variables.
The lower bound applies to arbitrary real decoupled tensors, including
$\widetilde A^w$; neither its diagonal support nor its coefficient signs
require a change.

\par\smallskip\noindent\textbf{4. Absorb the multipliers into singleton blocks.}
Put $L=[d]\setminus J$. For fixed signs
$\sigma^j\in\{-1,1\}^n$, $j\in J$, define
\[
 C^{w,\sigma}_{\mathbf i_L}
      =\sum_{\mathbf i_J}a_{\mathbf i}
                      \prod_{j\in J}w_{i_j}^j\sigma_{i_j}^j.
\]
Condition on the signs in the coordinates carrying multipliers.
The conditional $p$th moment is at most its maximum over these signs,
so
\begin{equation}\label{eq:freezing}
 \|S_{A^w}(\eta)\|_p
       \le\max_{\sigma^J}\|S_{C^{w,\sigma}}(\eta^L)\|_p
       \le C_d \|A\|^{N}_{p}.
\end{equation}
For the second inequality, suppose first that $L\ne\varnothing$ and
apply the already proved Rademacher upper bound in order $|L|$.
By \eqref{eq:collapsed-budget}, each $w^j\sigma^j$ is admissible
in the original singleton body $\mathcal B^N_{\{j\},j}(p)$.
Moreover,
\[
 C^\varepsilon_{I,s}(p)\subseteq\mathcal B^N_{I,s}(p),
\]
because all its row lengths are at most one, where every modified
tail exponent equals the square. Consequently, for a
partition with distinguished indices $(\mathcal J,\mathbf t)$ of $L$,
\[
 \|C^{w,\sigma}\|^{\varepsilon}_{\mathcal J,p;\mathbf t}
 \le\|A\|^{N}_{\mathcal J\cup\{\{j\}:j\in J\},p;(\mathbf t,(j)_{j\in J})}.
\]
Their sum is a subsum of $\|A\|^{N}_{p}$, uniformly in $w,\sigma$.
If $L=\varnothing$, $C^{w,\sigma}$ is a scalar and is bounded in
absolute value by the all-singleton functional of $A$.
This proves \eqref{eq:freezing}.
The maximum over signs costs no cardinality factor, since every
choice is covered by the same deterministic budget.

There are $2^{|\mathcal Q|}\le2^{2d}$ expanded terms.
Equations \eqref{eq:bernoulli-collapse} and \eqref{eq:freezing}
bound each one uniformly in the original parameters $x^B$.
Take their supremum and sum over $\mathcal Q$ and its distinguished
indices. The number of these choices depends only on $d$, which
proves \eqref{eq:diagonal-collapse-proved}.

\label{cor:fourteen}
For the variables in Corollary~\ref{thm:weibull}, if $1\le r\le2$, then
\begin{equation}\label{eq:weibull-polynomial}
 \|S_A(X)\|_p\asymp_d

repeat identical terms, with multiplicities depending only on $d$.

<!-- lemma-4: body_md -->
\label{lem:sorting}
Let $2\le r\le\infty$, $q=r/(r-1)$, with $q=1$ if $r=\infty$,
and $m=\min(n,\lfloor p\rfloor)$. Write
$b_1^*\ge\cdots\ge b_n^*$ for the decreasing rearrangement of $(|b_i|)$.
Then, with absolute constants,
\begin{equation}\label{eq:sorting}
 \|b\|_{r,p}\asymp
 p^{1/r}\left(\sum_{i\le m}(b_i^*)^q\right)^{1/q}
       +\sqrt p\left(\sum_{i>m}(b_i^*)^2\right)^{1/2}.
\end{equation}
<!-- end lemma-4 -->
<!-- proof-9: body_md -->
Denote the two terms on the right by $T$ and $R$.
H\"older's inequality on the first $m$ coordinates and
Cauchy--Schwarz on the rest give $\|b\|_{r,p}\le T+R$.
An optimizer for $T$ on the $\ell_r$ ball, supported on those $m$
coordinates, has Euclidean norm at most
$m^{1/2-1/r}p^{1/r}\le\sqrt p$. Consequently $\|b\|_{r,p}\ge T$.

If $m=n$, this finishes the proof. Otherwise let
$a=(\sum_{i>m}(b_i^*)^2)^{1/2}$.
When $a\ge\sqrt p\,b_m^*$ and $a>0$, the tail vector
$u_i=\sqrt p\,b_i^*/a$ has all coordinates at most one.
It has squared Euclidean norm $p$ and $\ell_r$ budget at most $p$
(for $r=\infty$, its maximum is at most one). Thus $\|b\|_{r,p}\ge R$.
If $a<\sqrt p\,b_m^*$, then
$R<pb_m^*\le2p^{1/r}m^{1/q}b_m^*\le2T$,
because $p/m\le2$. The zero-tail case is immediate.
In all cases $T+R\le3\|b\|_{r,p}$.

<!-- corollary-4: body_md -->
\label{cor:tails}
Under the assumptions of Theorem~\ref{thm:main}, if $A\ne0$,
there are positive constants $c_d,C_d$ such that, for $p\ge2$,
\begin{equation}\label{eq:tail-bounds}
 \mathbb P\{|S_A(X)|\ge C_d\|A\|_p^N\}\le e^{-p},
 \qquad
 \mathbb P\{|S_A(X)|\ge c_d\|A\|_p^N\}\ge e^{-C_dp}.
\end{equation}
<!-- end corollary-4 -->
<!-- proof-10: body_md -->
The upper bound follows from Markov's inequality and
Theorem~\ref{thm:main}, increasing its upper constant by a factor $e$.
For $a\ge1$, convexity of $N$ and \eqref{eq:tail-basic} imply
$\widehat N(t/a)\le\widehat N(t)/a$.
When $t>1$ and $t/a\le1$, for instance,
$(t/a)^2\le t/a\le N(t)/a$; the other cases follow from the
quadratic formula or convexity. Rescaling each block therefore gives
\begin{equation}\label{eq:moment-scaling}
                  \|A\|^{N}_{ap}\le a^d\|A\|_p^N.
\end{equation}
Thus $\|S_A(X)\|_{2p}\le K_d\|S_A(X)\|_p$.
Paley--Zygmund applied to $|S_A(X)|^p$ yields
\[
 \mathbb P\{|S_A(X)|\ge\tfrac12\|S_A(X)\|_p\}
 \ge(1-2^{-p})^2
       \left(\frac{\|S_A(X)\|_p}{\|S_A(X)\|_{2p}}\right)^{2p}
 \ge e^{-C_dp}.
\]
Use the lower moment bound to obtain the second assertion.
The norm is nonzero for $A\ne0$, since the independent centered
families have positive variances and the distinct tensor monomials
are orthogonal in $L_2$.
