\end{align*}
where $\phi(z)=D^2 F(z)(z^2)+DF(z) z$.	
Let $z\in \mathbb{B}\backslash\{0\}$ be fixed and let $z_0=z/\|z\|$. We consider the function $g:\mathbb{U}\to \mathbb{C}$ defined by \eqref{Th-31.0}.
Clearly $g\in\mathcal{H}(\mathbb{U})$ and $g(0)=1$.
Therefore for $\zeta\in\mathbb{U}\backslash\{0\}$, we have
\begin{align*}
\displaystyle
g(\zeta)=\frac{l_{\zeta z_0}\left([DF(\zeta z_0)]^{-1}\left(D^2 F(\zeta z_0)(\zeta^2z_0^2)+DF(\zeta z_0) \zeta z_0\right)\right)}{||\zeta z_0||}.
\end{align*}	
Consequently $\Re\{g(\zeta)\}>0$ for all $\zeta\in\mathbb{U}$ and so we may assume that
\begin{align}\label{Th-32.1}
\displaystyle
g(\zeta)=1+p_1\zeta+p_2\zeta^2+\ldots
\end{align}
Now using \eqref{Th-31.8} and \eqref{Th-32.1}, we get
\begin{align}\label{Th-32.2}
\displaystyle
\left(1+\sum\limits_{k=1}^{\infty}p_k\zeta^k\right)\left(1+\sum\limits_{k=2}^{\infty}ka_{k}\zeta^{k-1}\right)=1+\sum\limits_{k=2}^{\infty}k^2a_k\zeta^{k-1},	
\end{align}
where $a_k$ is defined by \eqref{Th-31.5a}.
Equating coefficients on the both sides of \eqref{Th-32.2}, we get
\begin{align}\label{Th-32.3}
k^2a_k=ka_k+(k-1)p_1a_{k-1}+\ldots+2p_{k-2}a_2+p_{k-1}.
\end{align}
We now use mathematical induction and suppose $|a_i|\leq 1$ for $i=2,3,\ldots, k-1$.
Applying Lemma \ref{Lem1} to \eqref{Th-32.3}, we get
\begin{align*}
(k^2-k)|a_k|\leq 2\sum\limits_{i=1}^{k-1}i=k(k-1),
\end{align*}
which shows that $|a_k|\leq 1$.
Then by induction, we have the first inequality
\begin{align*}
	\left|l_{z_0}\left(\frac{1}{k!}D^{k}F(0)(z_0^{k})\right)\right|\leq 1.
\end{align*}
Since $\Re\{g(\zeta)\}>0$, there exists a Schwarz function $\omega(\zeta)$ with $\omega(\zeta)=\sum_{k=1}^{\infty}\omega_k\zeta^k$ such that $g(\zeta)=(1+\omega(\zeta))/(1-\omega(\zeta))$ and so from \eqref{Th-31.8} and \eqref{Th-32.1}, we get
\begin{align}\label{Th-32.4}
\left(2+\sum\limits_{k=2}^{\infty}k(k+1)a_{k} \zeta^{k-1} \right)\omega(\zeta)=\sum\limits_{k=2}^{\infty}k(k-1)a_{k}\zeta^{k-1}.
\end{align}
Equating coefficients on the both sides of \eqref{Th-32.4}, one can easily get
\begin{align*}
	2\omega_{k-1}+6a_2\omega_{k-2}+\ldots+k(k-1)a_{k-1}\omega_1=k(k-1)a_{k}.
\end{align*}
This shows that the coefficient $a_{k}$ depends only on $a_2$, $a_3\ldots$,$a_{k-1}$. Therefore we can rewrite \eqref{Th-32.4} as follows
\begin{align}\label{Th-32.5}
	\left(2+\sum\limits_{k=2}^{n-1}k(k+1)a_{k} \zeta^{k-1} \right)\omega(\zeta)=\sum\limits_{k=2}^{n}k(k-1)a_{k}\zeta^{k-1}+\sum\limits_{k=n+1}^{\infty}b_k\zeta^{k-1}
\end{align}
for some $b_k$.
Let $\zeta=re^{i\theta}$. Then from \eqref{Th-32.5}, we get
\begin{align}\label{Th-32.6}
\left(2+\sum\limits_{k=2}^{n-1}k(k+1)a_{k} r^{k-1}e^{i(k-1)\theta} \right)\omega(re^{i\theta})=&\sum\limits_{k=2}^{n}k(k-1)a_{k}r^{k-1}e^{i(k-1)\theta}\\&+\sum\limits_{k=n+1}^{\infty}b_kr^{k-1}e^{i(k-1)\theta}\nonumber
\end{align}
and
\begin{align}\label{Th-32.7}
\left(2+\sum\limits_{k=2}^{n-1}k(k+1)\overline a_{k} r^{k-1}e^{-i(k-1)\theta} \right)\omega(re^{-i\theta})=&\sum\limits_{k=2}^{n}k(k-1)\overline a_{k}r^{k-1}e^{-i(k-1)\theta}\\&+\sum\limits_{k=n+1}^{\infty}\overline b_kr^{k-1}e^{-i(k-1)\theta}\nonumber.
\end{align}

Note that $|\omega(\zeta)|<1$ for all $\zeta\in\mathbb{U}$. We first multiply \eqref{Th-32.6} and \eqref{Th-32.7} and then integrating with respect to $\theta$ from $0$ to $2\pi$, we can find that
\begin{align*}
\sum\limits_{k=2}^{n}k^2(k-1)^2|a_{k}|^2r^{2(k-1)}+\sum\limits_{k=n+1}^{\infty}|b_k|^2r^{2(k-1)}<4+\sum\limits_{k=2}^{n-1}k^2(k+1)^2|a_{k}|^2r^{2(k-1)}.
\end{align*}
Letting $r\to 1$, we obtain
\begin{align*}
	\sum\limits_{k=2}^{n}k^2(k-1)^2|a_{k}|^2\leq 4+\sum\limits_{k=2}^{n-1}k^2(k+1)^2|a_{k}|^2,
\end{align*}
i.e.,
\begin{align*}
n^2(n-1)^2|a_{n}|^2\leq 4\left(1+\sum\limits_{k=2}^{n-1}k^3|a_{k}|^2\right).
\end{align*}
Now adding $4n^3|a_{n}|^2$ on the both sides and then using \eqref{Th-31.5a}, we get the required conclusion \eqref{Th-32t}.\vspace{1.9mm}

{\bf Sharpness part:} We have $F(z)=\frac{f(l_u(z))}{l_u(z)}z$, where $f(l_u(z))=l_u(z)(1-l_u(z))^{-1}$, $z\in\mathbb{B}$ and $||u||=1$. Clearly $f\in \mathcal{K}(\mathbb{U})$.
According to Lemma \ref{Lem2}, it is clear that $F\in\mathcal{K}_B(\mathbb{B})$. Note that
$F(z)=\sum_{n=0}^{\infty}(l_u(z))^{n}\;z$, $z\in\mathbb{B}$.
A straightforward computation shows that
\begin{align*}
\frac{D^{k}F(0)\left(z^{k}\right)}{k!}=(l_u(z))^{k-1} z.
\end{align*}
Taking $z=ru\;(0<r<1)$, we get
\begin{align*}
\frac{D^{k}F(0)\left(z^{k}\right)}{k!||z||^k}=1
\end{align*}
and so equality of \eqref{Th-32t} holds.
