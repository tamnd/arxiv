<!-- theorem-1: body_md -->
\label{thm:3.1}
Assume $0<\beta<1$, $\alpha>0$, $\mu\neq0$. Then the following statements hold.
\begin{enumerate}[label=(\roman*)]
\item There is a constant $C_{\alpha,\beta,\mu}<\infty$ such that
\begin{equation}\label{eq:3.1}
c_T\leq C_{\alpha,\beta,\mu}T^{2\beta+1},\qquad T\geq1.
\end{equation}
Consequently every admissible family $\mathbb H=\{H^T\}_{T\geq1}$ satisfies
\begin{equation}\label{eq:3.2}
C(\mathbb H)\leq2\beta+1.
\end{equation}
\item For each fixed $\varepsilon>0$, let $\varphi_\varepsilon$ be the $C^2$ smoothing of
$x\mapsto\operatorname{sgn}(x)|x|^\beta$ constructed in
Lemma~5.1. Define, for $0\leq t\leq T$,
\begin{equation}\label{eq:3.3}
H_t^{\beta,\varepsilon,T}:=\bigl((T+1-t)^\beta-1\bigr)\varphi_\varepsilon(S_t).
\end{equation}
Then $H^{\beta,\varepsilon,T}\in\mathcal A_T$ for every $T\geq1$.
\item The admissible family $\mathbb H^{\beta,\varepsilon}:=\{H^{\beta,\varepsilon,T}\}_{T\geq1}$ satisfies
\begin{equation}\label{eq:3.4}
C(\mathbb H^{\beta,\varepsilon})=2\beta+1.
\end{equation}
\end{enumerate}

<!-- lemma-1: body_md -->
\label{lem:5.1}
Fix $0<\beta<1$ and $\varepsilon>0$. Define the odd
function $\varphi_\varepsilon$ by
\begin{equation}\label{eq:5.1}
\varphi_\varepsilon(x)=Ax+Bx^3+Cx^5\quad (|x|\leq\varepsilon),
\qquad
\varphi_\varepsilon(x)=\operatorname{sgn}(x)|x|^\beta\quad (|x|\geq\varepsilon),
\end{equation}
where
\begin{equation}\label{eq:5.2}
A=\frac{15-8\beta+\beta^2}{8}\varepsilon^{\beta-1},
\qquad
B=\frac{(\beta-1)(5-\beta)}{4}\varepsilon^{\beta-3},
\qquad
C=\frac{(\beta-1)(\beta-3)}{8}\varepsilon^{\beta-5}.
\end{equation}
Then $\varphi_\varepsilon\in C^2(\mathbb{R})$. If
\begin{equation}\label{eq:5.3}
F_\varepsilon(x):=\int_0^x\varphi_\varepsilon(u)\,du,
\end{equation}
then $F_\varepsilon\in C^3(\mathbb{R})$, $F_\varepsilon$ is even, and
\begin{equation}\label{eq:5.4}
F_\varepsilon(x)=\frac{A}{2}x^2+\frac{B}{4}x^4+\frac{C}{6}x^6,
\qquad |x|\leq\varepsilon,
\end{equation}
while, for $|x|\geq\varepsilon$,
\begin{equation}\label{eq:5.5}
F_\varepsilon(x)=\frac{|x|^{\beta+1}}{\beta+1}+D_\varepsilon,
\qquad
D_\varepsilon=\varepsilon^{\beta+1}\frac{(\beta-5)(\beta-3)(\beta-1)}{48(\beta+1)}<0.
\end{equation}
Moreover, there are finite constants $K_\varepsilon$ and $M_\varepsilon$ such that, for all $x\in\mathbb{R}$,
\begin{equation}\label{eq:5.6}
F_\varepsilon(x)\geq0,
\qquad
F_\varepsilon(x)\geq\frac{|x|^{\beta+1}}{\beta+1}-K_\varepsilon,
\qquad
\|F_\varepsilon''\|_\infty\leq M_\varepsilon.
\end{equation}
<!-- end lemma-1 -->
<!-- proof-1: body_md -->
The coefficients in \eqref{eq:5.2} are obtained by imposing $C^2$ matching at $x=\varepsilon$:
\begin{align}
A\varepsilon+B\varepsilon^3+C\varepsilon^5&=\varepsilon^\beta,\label{eq:5.7}\\
A+3B\varepsilon^2+5C\varepsilon^4&=\beta\varepsilon^{\beta-1},\label{eq:5.8}\\
6B\varepsilon+20C\varepsilon^3&=\beta(\beta-1)\varepsilon^{\beta-2}.\label{eq:5.9}
\end{align}
Oddness gives the same matching at $x=-\varepsilon$, so $\varphi_\varepsilon\in C^2(\mathbb{R})$. The formula \eqref{eq:5.4} follows by
integration on $[-\varepsilon,\varepsilon]$. For $|x|\geq\varepsilon$,
\[
F_\varepsilon(x)=F_\varepsilon(\varepsilon)+\int_\varepsilon^{|x|}u^\beta\,du
=\frac{|x|^{\beta+1}}{\beta+1}+D_\varepsilon,
\]
with $D_\varepsilon$ as in \eqref{eq:5.5}. This additive constant is important: the primitive does not equal
$|x|^{\beta+1}/(\beta+1)$ outside the smoothing interval unless the inner spline is chosen differently.

It remains to prove nonnegativity and boundedness of the second derivative. For $0\leq x\leq\varepsilon$,
set $y=(x/\varepsilon)^2$. Then
\[
\varphi_\varepsilon(x)=x\varepsilon^{\beta-1}q(y),
\]
where
\[
q(y)=\frac{15-8\beta+\beta^2}{8}
+\frac{(\beta-1)(5-\beta)}{4}y
+\frac{(\beta-1)(\beta-3)}{8}y^2.
\]
We have $q(1)=1$, and for $0\leq y\leq1$,
\begin{equation}\label{eq:5.10}
q'(y)=\frac{(\beta-1)(5-\beta)}{4}
+\frac{(\beta-1)(\beta-3)}{4}y
\leq\frac{\beta-1}{2}<0.
\end{equation}
Hence $q(y)\geq q(1)=1$, so $\varphi_\varepsilon(x)\geq0$ for $x\geq0$. Since $\varphi_\varepsilon$ is odd, $F_\varepsilon$ is even and nonnegative.
The lower bound in \eqref{eq:5.6} follows from \eqref{eq:5.5} outside $[-\varepsilon,\varepsilon]$ and from compactness on $[-\varepsilon,\varepsilon]$.
Finally, $F_\varepsilon''=\varphi_\varepsilon'$ is bounded on $[-\varepsilon,\varepsilon]$ by the polynomial formula, while for $|x|\geq\varepsilon$,
\begin{equation}\label{eq:5.11}
|F_\varepsilon''(x)|=\beta|x|^{\beta-1}\leq\beta\varepsilon^{\beta-1}.
\end{equation}
Thus $\|F_\varepsilon''\|_\infty<\infty$.

<!-- lemma-2: body_md -->
\label{lem:6.1}
Fix $\varepsilon>0$ and $T>0$, and
write $\varphi=\varphi_\varepsilon$ and $F=F_\varepsilon$. Define
\begin{equation}\label{eq:6.1}
H_t^{\beta,\varepsilon,T}:=w_T(t)\varphi(S_t),
\qquad 0\leq t\leq T,
\end{equation}
where
\begin{equation}\label{eq:6.2}
w_T(t):=(T+1-t)^\beta-1.
\end{equation}
Then $H^{\beta,\varepsilon,T}\in\mathcal{A}_T$. Its terminal wealth admits the decomposition
\begin{equation}\label{eq:6.3}
V_T^{\beta,\varepsilon}:=(H^{\beta,\varepsilon,T}\cdot S)_T=I_1(T)-I_2(T),
\end{equation}
where
\begin{equation}\label{eq:6.4}
I_1(T):=\beta\int_0^T(T+1-t)^{\beta-1}F(S_t)\,dt
\end{equation}
and
\begin{equation}\label{eq:6.5}
I_2(T):=\frac{1}{2}\int_0^Tw_T(t)F''(S_t)\,dt.
\end{equation}
Moreover,
\begin{equation}\label{eq:6.6}
|I_2(T)|\leq B_\varepsilon T^{\beta+1},
\qquad
B_\varepsilon:=\frac{1}{2(\beta+1)}\|F''\|_\infty.
\end{equation}
<!-- end lemma-2 -->
<!-- proof-2: body_md -->
The process $H^{\beta,\varepsilon,T}$ is predictable because it is continuous and adapted, and it is
$S$-integrable on finite horizons because $\varphi(S)$ is locally bounded. Let
\[
u(t,x):=w_T(t)F(x),\qquad 0\leq t\leq T.
\]
Then $u\in C^{1,2}([0,T]\times\mathbb{R})$, $w_T(T)=0$, and $F(S_0)=F(0)=0$. It\^o's formula on $[0,T]$ gives
\[
du(t,S_t)=-\beta(T+1-t)^{\beta-1}F(S_t)\,dt+w_T(t)\varphi(S_t)\,dS_t
+\tfrac{1}{2}w_T(t)F''(S_t)\,dt.
\]
The boundary terms vanish, and hence
\begin{equation}\label{eq:6.7}
0=-\beta\int_0^T(T+1-t)^{\beta-1}F(S_t)\,dt
+\int_0^Tw_T(t)\varphi(S_t)\,dS_t
+\frac{1}{2}\int_0^Tw_T(t)F''(S_t)\,dt.
\end{equation}
This gives \eqref{eq:6.3}--\eqref{eq:6.5}. The bound \eqref{eq:6.6} follows from Lemma~\ref{lem:5.1} and the subadditivity inequality $(1+s)^\beta\leq1+s^\beta$, valid for $s\geq0$ and $0<\beta<1$:
\begin{align*}
|I_2(T)|&\leq\frac{1}{2}\|F''\|_\infty\int_0^T\bigl((T+1-t)^\beta-1\bigr)\,dt\\
&=\frac{1}{2}\|F''\|_\infty\int_0^T\bigl((1+s)^\beta-1\bigr)\,ds
\leq\frac{1}{2(\beta+1)}\|F''\|_\infty T^{\beta+1}.
\end{align*}
It remains to show admissibility. The same calculation on $[0,r]$, $0\leq r\leq T$, gives
\begin{align*}
(H^{\beta,\varepsilon,T}\cdot S)_r
&=w_T(r)F(S_r)+\beta\int_0^r(T+1-t)^{\beta-1}F(S_t)\,dt\\
&\quad-\frac{1}{2}\int_0^rw_T(t)F''(S_t)\,dt.
\end{align*}
The first two terms are nonnegative by Lemma~\ref{lem:5.1}, and the last term is bounded below by
$-B_\varepsilon T^{\beta+1}$. Hence
\begin{equation}\label{eq:6.8}
(H^{\beta,\varepsilon,T}\cdot S)_r\geq-B_\varepsilon T^{\beta+1},
\qquad 0\leq r\leq T.\qedhere
\end{equation}

<!-- lemma-3: body_md -->
\label{lem:7.1}
Let
\begin{equation}\label{eq:7.1}
\Xi_t:=\int_0^t|X_s|^{2\beta}\,ds.
\end{equation}
There are constants $\delta_0,c_0,C_0>0$, depending only on $\alpha$ and $\beta$, such that, for every $t\geq0$ and
every tail level $r>0$,
\begin{equation}\label{eq:7.2}
\mathbb{E}\exp\{\delta_0\Xi_t\}\leq C_0e^{c_0t},
\qquad
\mathbb{P}(\Xi_t\geq r)\leq C_0\exp\{c_0t-\delta_0r\}.
\end{equation}
<!-- end lemma-3 -->
<!-- proof-3: body_md -->
Let
\[
Lf(x)=-\alpha\operatorname{sgn}(x)|x|^\beta f'(x)+\frac{1}{2}f''(x)
\]
be the generator of the diffusion process $X$. Choose an even $C^2$ function $V\geq1$ such that
$V(x)=1+a|x|^{\beta+1}$ for $|x|\geq2$, with the function smoothly modified on $[-2,2]$. This smoothing
avoids the singularity of the second derivative at the origin. For $|x|\geq2$,
\[
V'(x)=a(\beta+1)\operatorname{sgn}(x)|x|^\beta,
\qquad
V''(x)=a\beta(\beta+1)|x|^{\beta-1},
\]
and therefore
\[
LV(x)=-\alpha a(\beta+1)|x|^{2\beta}+\frac{1}{2}a\beta(\beta+1)|x|^{\beta-1}.
\]
Since $2\beta>\beta-1$, and compact sets can be absorbed into constants, there are constants
$c_1,c_2,c_3,c_4>0$ such that, for all $x\in\mathbb{R}$,
\[
LV(x)\leq-c_1|x|^{2\beta}+c_2,
\qquad
(V'(x))^2\leq c_3|x|^{2\beta}+c_4.
\]
For $W_\lambda=e^{\lambda V}$,
\[
\frac{LW_\lambda}{W_\lambda}=\lambda LV+\frac{\lambda^2}{2}(V')^2
\leq-\left(\lambda c_1-\frac{\lambda^2c_3}{2}\right)|x|^{2\beta}
+\lambda c_2+\frac{\lambda^2c_4}{2}.
\]
Choose $\lambda>0$ so small that
\[
\delta_0:=\lambda c_1-\frac{\lambda^2c_3}{2}>0,
\qquad
c_0:=\lambda c_2+\frac{\lambda^2c_4}{2}.
\]
Then
\begin{equation}\label{eq:7.3}
(L+\delta_0|x|^{2\beta})W_\lambda(x)\leq c_0W_\lambda(x),
\qquad x\in\mathbb{R}.
\end{equation}
Let $\tau_n:=\inf\{t\geq0:|X_t|\geq n\}$. Since the SDE has a nonexplosive strong solution, $\tau_n\uparrow\infty$
a.s. Applying It\^o's formula to the stopped process
\[
\exp\left\{\delta_0\int_0^{t\wedge\tau_n}|X_s|^{2\beta}\,ds-c_0(t\wedge\tau_n)\right\}
W_\lambda(X_{t\wedge\tau_n})
\]
is legitimate, because all coefficients are bounded up to $\tau_n$. By \eqref{eq:7.3} its drift is nonpositive;
hence this stopped process is a nonnegative supermartingale. Since $W_\lambda\geq1$,
\[
\mathbb{E}\exp\left\{\delta_0\int_0^{t\wedge\tau_n}|X_s|^{2\beta}\,ds\right\}
\leq e^{c_0t}W_\lambda(0).
\]
Letting $n\to\infty$ and using Fatou's lemma gives $\mathbb{E}e^{\delta_0\Xi_t}\leq W_\lambda(0)e^{c_0t}$.

Thus \eqref{eq:7.2} holds with $C_0:=W_\lambda(0)$, and the tail estimate follows immediately from
Markov's inequality.
