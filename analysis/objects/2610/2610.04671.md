<!-- assumption-1: body_md -->
\label{as:crowd}
$f_i=f_0+h_i$, where $h_1,\dots,h_n$ are independent random $C^2$ functions with
$\mathbb{E}\nabla h_i(x)=0$ for every $x$, drawn once and then fixed. We write
$\beta_i:=\nabla h_i(x_0^\star)$ for client $i$'s realised tilt at the truth's
optimum, $q_i:=\lVert \beta_i \rVert^2$ for its realised bias energy, and
$\zeta_i^2:=\mathbb{E} q_i$ for the expected one. We also write $H_w=\sum_iw_ih_i$ and
$\lambda_w:=\sup_{\mathcal{X}}\lVert \nabla^2H_w \rVert_{\mathrm{op}}$, bounded by
$\ell:=\max_i\ell_i$ when $\ell_i$ bounds $\lVert \nabla^2h_i \rVert_{\mathrm{op}}$. The
\emph{linear-tilt} case is $h_i=\langle \beta_i ,\, \cdot \rangle$, i.e.\ $\ell=0$.
<!-- end assumption-1 -->
<!-- assumption-2: body_md -->
\label{as:reg}
$f_0$ is $\mu$-strongly convex and $L$-smooth on $\mathcal{X}$, $0<\mu\le L$, with
$f_0^\star=\min_\mathcal{X} f_0$ attained at $x_0^\star$.
<!-- end assumption-2 -->
<!-- assumption-3: body_md -->
\label{as:oracle}
At round $k$ the server broadcasts $x^{k-1}$ and each client returns
$\widehat{g}_i^k=\nabla f_i(x^{k-1})+\varepsilon_i^k$ computed from $b$ fresh samples, with
$\mathbb{E}[\varepsilon^k_i\mid\mathcal F_{k-1}]=0$,
$\mathbb{E}[\lVert \varepsilon^k_i \rVert^2\mid\mathcal F_{k-1}]\le\sigma_i^2/b$ and conditional
independence across $i$. Query points may be adaptive. We write
$\tau_i^2=\sigma_i^2+\tfrac1{n^2}\sum_j\sigma_j^2$, which bounds the noise level of
client $i$'s report after the crowd mean has been subtracted.
<!-- end assumption-3 -->
<!-- proposition-1: body_md -->
\label{prop:rigid}
Let $h_1,\dots,h_n$ satisfy Assumption~\ref{as:crowd} on an open convex set. If each
$h_i$ is almost surely convex, then almost surely every $h_i$ is affine.

<!-- lemma-1: body_md -->
\label{lem:transfer}
\label{lem:transfer-full}
Let Assumptions~\ref{as:crowd}--\ref{as:reg} hold, $w\in\Delta_n$, $\mathcal{X}=\mathbb{R}^d$ and
$\lambda_w<\mu$, and write $\Gamma_w:=\lVert b(w) \rVert$.
\begin{enumerate}[label=(\roman*),itemsep=1pt,topsep=2pt]
\item $F_w$ is $(\mu-\lambda_w)$-strongly convex and $(L+\lambda_w)$-smooth, and its
minimiser obeys $\lVert x_w^\star-x_0^\star \rVert\le\Gamma_w/(\mu-\lambda_w)$.
\item For every $x$, the excess risk on the truth is at most
$\tfrac{2L}{\mu-\lambda_w}[F_w(x)-\min F_w]+\tfrac{L\Gamma_w^2}{(\mu-\lambda_w)^2}$, and
in the linear-tilt case the bias constant can be made sharp: for every $\epsilon>0$,
\[
f_0(x)-f_0^\star\ \le\ (1+\epsilon)\,\frac{\Gamma_w^2}{2\mu}\ +\ \Big(1+\tfrac1\epsilon\Big)\frac{L}{\mu}\big[F_w(x)-\min F_w\big].
\]
\item If $\lambda_w<\mu/2$ then $f_0(x_w^\star)-f_0^\star\ge
\tfrac{1}{2L}(\tfrac{\mu-2\lambda_w}{\mu-\lambda_w})^2\Gamma_w^2$, which is
$\Gamma_w^2/(2L)$ in the linear case.
\end{enumerate}

<!-- lemma-2: body_md -->
\label{lem:transfer-c}
Let Assumptions~\ref{as:crowd}--\ref{as:reg} hold on a closed convex $\mathcal{X}$ whose
interior contains $x_0^\star$, let $w\in\Delta_n$ with $\lambda_w<\mu$, and let

<!-- proposition-2: body_md -->
\label{prop:oracle}
Under Assumption~\ref{as:crowd}, for every $w\in\Delta_n$ independent of the tilts,
$\mathbb{E}\Gamma_w^2=\sum_iw_i^2\zeta_i^2$. For any $c\in\mathbb{R}^n_{>0}$ the minimum of
$\sum_iw_i^2c_i$ over $\Delta_n$, and also over the affine hull $\mathbf{1}^\top w=1$,
equals $(\sum_ic_i^{-1})^{-1}$ and is attained at $w_i^\star\propto c_i^{-1}>0$.
With $c_i=\zeta_i^2$ this gives $\Psi_0=\mathrm{HM}/n$ against $\mathrm{AM}/n$ for uniform weights.
Moreover \emph{(i)} the ratio is $\mathrm{AM}/\mathrm{HM}\ge1$, with equality iff all $\zeta_i$ are
equal; \emph{(ii)} $\Psi_0\le\min_i\zeta_i^2$; \emph{(iii)} the ratio is unbounded.

<!-- lemma-3: body_md -->
\label{lem:harmonic}
$H(c):=(\sum_ic_i^{-1})^{-1}$ is concave and positively homogeneous of degree one on
$\mathbb{R}^n_{>0}$; hence $H(a+b)\ge H(a)+H(b)$, with equality if and only if $b$ is
proportional to $a$.

<!-- proposition-3: body_md -->
\label{prop:ident-neg}
Under Assumptions~\ref{as:crowd}--\ref{as:oracle}: (i) for every $c\in\mathbb{R}^d$, replacing
$f_0$ by $f_0+\langle c ,\, \cdot \rangle$ and each $h_i$ by $h_i-\langle c ,\, \cdot \rangle$ leaves every
client risk, and hence the law of the transcript of any protocol, unchanged, while for
$c\ne0$ and $\mathcal{X}=\mathbb{R}^d$ the target moves; (ii) subtracting from each client's
gradients its deviation from the crowd mean makes the aggregate an unbiased gradient of
$F_{w_{\mathrm{unif}}}$ for every $w\in\Delta_n$, with expected bias energy $\mathrm{AM}/n$.

<!-- lemma-4: body_md -->
\label{lem:residual}
Under Assumptions~\ref{as:crowd} and \ref{as:oracle}, the residuals
$r_i^k=\widehat{g}_i^k-\frac1n\sum_j\widehat{g}_j^k=\widetilde{\nabla h_i}(x^{k-1})+\widetilde{\varepsilon}_i^k$, with
$\widetilde{\nabla h_i}:=\nabla h_i-\tfrac1n\sum_j\nabla h_j$ and

$\nabla f_0(x^{k-1})$, whatever the query point and whatever $f_0$ is; the array
$(r^k_i)$ is a maximal invariant of $(\widehat{g}^k_i)$ under per-round common shifts; and
$\mathbb{E}[\widetilde{\varepsilon}^k_i\mid\mathcal F_{k-1}]=0$,
$\mathbb{E}[\lVert \widetilde{\varepsilon}^k_i \rVert^2\mid\mathcal F_{k-1}]\le\tau_i^2/b$.

satisfy $s=T\zeta^2$, and $T$ is invertible with

(T^{-1}s)_i=\frac{n}{n-2}\Big(s_i-\frac{1}{n(n-1)}\sum_js_j\Big),
\qquad
\lVert T^{-1} \rVert_{\infty\to\infty}=\frac{n^2-2}{(n-1)(n-2)}=:C_n .
\]
Hence $\zeta^2$, and with it the oracle weights, are identified by the law of the
disagreement residuals. Moreover $C_3=3.5$, $C_{20}\le1.17$, $C_n\to1$, and $T$ is
singular for $n=2$.

<!-- proposition-4: body_md -->
\label{prop:corr}
For jointly Gaussian centred crowds with $\operatorname{Cov}(\beta_i,\beta_j)=\Sigma_{ij}I_d/d$,
$\mathbb{E}\lVert b(w) \rVert^2=w^\top\Sigma w$ for deterministic $w$, minimised over
$\{\mathbf{1}^\top w=1\}$ by the minimum-variance portfolio $w\propto\Sigma^{-1}\mathbf{1}$
\citep{markowitz1952}, which is also the minimiser over $\Delta_n$ when
$\Sigma^{-1}\mathbf{1}\ge0$. The map $\Sigma\mapsto J\Sigma J$ has kernel exactly
$\{\mathbf{1} u^\top+u\mathbf{1}^\top:u\in\mathbb{R}^n\}$, and for $\Sigma=I_3$ versus
$\Sigma'=I_3+\mathbf{1} u^\top+u\mathbf{1}^\top$, $u=(0.1,0,0)^\top$ -- both positive definite --
the disagreement residuals have identical laws while the optimal weights differ,
$(\tfrac13,\tfrac13,\tfrac13)$ versus $(0.267,0.367,0.367)$. Under $\Sigma'$ the
uniform weights attain $0.4000$ against the optimum $0.3933$, and under $\Sigma$ the
weights optimal for $\Sigma'$ attain $0.3400$ against $\tfrac13$. Hence no rule that
is a function of the residuals is optimal for both.

<!-- lemma-9: body_md -->
\label{thm:est}\label{thm:est-full}
Let Assumption~\ref{as:oracle} hold with arbitrary tilts, $n\ge3$, $K_0$ even, and
suppose the disagreement measured at the query points differs from $\widetilde{\beta}_i$ by at
most $\delta_i$ almost surely, for all $k\le K_0$. Put
$\vartheta_i:=\tau_i\sqrt{2/(bK_0)}$ and $\widetilde{\zeta}_i:=\lVert \widetilde{\beta}_i \rVert$.
\begin{enumerate}[label=(\roman*),itemsep=1pt,topsep=2pt]
\item \emph{(Linear crowd, $\delta_i=0$.)} $\widehat{s}_i$ is exactly unbiased for
$\lVert \widetilde{\beta}_i \rVert^2$ given the tilts, for every adaptive query sequence, with
conditional variance at most
$8\widetilde{\zeta}_i^2\tau_i^2/(bK_0)+8\tau_i^4/(bK_0)^2$; the bound is dimension-free.
\item \emph{(General.)} $\mathbb{E}|\widehat{s}_i-\lVert \widetilde{\beta}_i \rVert^2|\le2\widetilde{\zeta}_i(\delta_i+\vartheta_i)+(\delta_i+\vartheta_i)^2$.
\item \emph{(Weights.)} With $\widehat{q}=(T^{-1}\widehat{s})_+$ and Assumption~\ref{as:incoh},
$\Delta:=\max_i|\widehat{q}_i-q_i|\le C_n(\max_i|\widehat{s}_i-\lVert \widetilde{\beta}_i \rVert^2|+3\kappa\bar q)$,
which converts into a high-probability bound by Chebyshev and a union bound, improving
to a $\sqrt{\log(dn/\delta)}$ dependence under sub-Gaussian noise
(Remark~\ref{rem:subgauss}).
\end{enumerate}

<!-- remark-1: body_md -->
\label{rem:subgauss}
Suppose the centred noises are conditionally norm-sub-Gaussian,
$\mathbb{P}(\lVert \widetilde{\varepsilon}^k_i \rVert\ge t\mid\mathcal F_{k-1})\le2e^{-bt^2/(2\tau_i^2)}$. The
concentration inequality for vector martingales with norm-sub-Gaussian increments of
\citet{jin2019subgaussian} then gives, with probability at least $1-\delta$ and for
all $i$ and $a$ simultaneously,
$\lVert \nu^{(a)}_i \rVert\le C\tau_i\sqrt{\log(8dn/\delta)/(bK_0)}$ with an absolute $C$.
On that event the bound of case (ii) with $\delta_i=0$ gives
$|\widehat{s}_i-\lVert \widetilde{\beta}_i \rVert^2|\le2\widetilde{\zeta}_iC\tau_i\sqrt{\log(8dn/\delta)/(bK_0)}+C^2\tau_i^2\log(8dn/\delta)/(bK_0)$,
so the factors $\sqrt{n/\delta}$ and $n/\delta$ of the moment version become
$\sqrt{\log(dn/\delta)}$ and $\log(dn/\delta)$. We keep the moment version in the main
text because it assumes nothing beyond Assumption~\ref{as:oracle}.

<!-- lemma-11: body_md -->
\label{lem:sgd}
Let $F$ be $\mu_F$-strongly convex and $L_F$-smooth on a closed convex $\mathcal{X}$ with
minimiser $x^\star$ over $\mathcal{X}$, and $x_t=\Pi_{\mathcal{X}}(x_{t-1}-\eta g_t)$ with
$\mathbb{E}[g_t\mid\mathcal F_{t-1}]=\nabla F(x_{t-1})$ and

$a_t\le(1-\mu_F\eta)a_{t-1}+\eta^2V$, hence $a_T\le e^{-\mu_F\eta T}a_0+\eta V/\mu_F$.

$\eta:=\min\{\tfrac1{2L_F},\tfrac{\ell}{\mu_FT}\}$. Then
\[
a_T\ \le\ a_0e^{-\mu_FT/(2L_F)}+\Big(\frac{a_0}{R^2}+\ell\Big)\frac{V}{\mu_F^2T},
\]
which is at most $a_0e^{-\mu_FT/(2L_F)}+(1+\ell)V/(\mu_F^2T)$ when $a_0\le R^2$.

\label{thm:anytime}
Let the assumptions of Theorem~\ref{thm:endtoend} hold, fix $\gamma>1$, and run

\label{thm:lower}
Let $\mu,d,n$, the energies $\zeta_i$, the noise levels $\sigma_i$ and a per-client
query budget $K$ all be given, and consider the following instances: the truth is the
quadratic $f_0=\tfrac\mu2\lVert x-x_c \rVert^2$ with unknown centre $x_c$, the clients are

<!-- remark-5: body_md -->
\label{rem:guard}
Compute $\widehat{\kappa}$ at the refresh rounds. If $\widehat{\kappa}<1$, use \textsc{Crowd}{} and report
$\tfrac{1+\widehat{\kappa}}{1-\widehat{\kappa}}$ as an estimate of the competitive ratio of
Proposition~\ref{thm:free}. If $\widehat{\kappa}\ge1$, the diagonal proxy is uninformative on
this instance and the bound is vacuous; fall back to uniform weighting. The guard
indicates which regime the deployment is likely in but does not certify it.
Proposition~\ref{prop:check} shows that $\widehat{\kappa}$ is not consistent for $\kappa$:
its population error is of the same order as $\kappa$ itself. The reported ratio is
therefore an estimate, not a bound. In place of a guarantee the guard comes with a
measured error rate. It is cheap, and in the linear regime it gives the deployment an
estimate of the quantity Assumption~\ref{as:incoh} bounds.
Appendix~\ref{app:diag} measures $\widehat{\kappa}$ against $\kappa$ on three mechanisms, and
Appendix~\ref{app:regime} calibrates the threshold: the empirical probability that
$\kappa\ge1$ given $\widehat{\kappa}<1$ is $4.2\%$. The restriction to a near-linear crowd
matters. The estimator inherits the staleness of Lemma~\ref{thm:est}, so when probing
happens far from the optimum $\widehat{\kappa}$ under-reports, by a factor three to five in
Appendix~\ref{app:regime}. That failure is
detectable and partly repairable, and Remark~\ref{rem:staleguard} does both. In the
real-data study of Appendix~\ref{app:gain}, where the probe is warm, $\widehat{\kappa}=0.52$
against a realised $\kappa=0.48$. The guard is also conservative: \textsc{Crowd}{} often still
improves on uniform weighting when $\kappa$ exceeds one, at $\kappa=1.28$ in
Appendix~\ref{app:region}, $\kappa\approx2$ in Appendix~\ref{app:femnist} and
$\kappa\approx2$ to $3.9$ in Appendix~\ref{app:mlp}.

<!-- remark-6: body_md -->
\label{rem:staleguard}
The restriction to a near-linear crowd can be checked on the trajectory the algorithm
already produces. Split the probing window into two disjoint
\emph{time}-halves, compute $\widehat{\kappa}$ on each from its own parity split, and read the
drift between them. A crowd that is effectively linear over the probing region returns
the same value twice; a curved one drifts, because the two halves measure the Gram at
different points. Scaling that drift by how far the iterate still has to travel, which
the probing displacement itself estimates, gives
\[
\widehat{\kappa}_{\mathrm{corr}}
=\widehat{\kappa}_{\text{late}}+\big|\widehat{\kappa}_{\text{late}}-\widehat{\kappa}_{\text{early}}\big|
\cdot\frac{\lVert x^{\text{end}}-x^{\text{start}} \rVert}{\lVert x^{\text{end}}-x^{\text{mid}} \rVert} ,
\]
at the cost of one extra pair of accumulators and no extra samples. The rule is a
heuristic. The supplementary materials measure it on two mechanisms and three
probing budgets:

\begin{center}\small
\begin{tabular}{@{}llcccc@{}}
\toprule
Mechanism & Probe & $\kappa$ & $\widehat{\kappa}$ & drift & $\widehat{\kappa}_{\mathrm{corr}}$ \\
\midrule
low rank  & cold, short & $2.19$ & $1.48$ & $0.054$ & $1.88$ \\
low rank  & cold, long  & $2.19$ & $1.48$ & $0.007$ & $1.53$ \\
low rank  & warm, long  & $2.19$ & $1.48$ & $0.001$ & $1.48$ \\
full rank & cold, short & $0.49$ & $0.49$ & $0.019$ & $0.71$ \\
full rank & cold, long  & $0.49$ & $0.49$ & $0.004$ & $0.57$ \\
full rank & warm, long  & $0.49$ & $0.49$ & $0.002$ & $0.49$ \\
\bottomrule
\end{tabular}
\end{center}

Three observations follow. The drift falls by one to two orders of magnitude as the probe warms
and lengthens, so it is a working detector of a stale measurement. The correction is
inert on a settled probe, leaving both estimates untouched, so it does not penalise a
good measurement with a false refusal. And on a cold short probe it inflates in the
safe direction, moving the low-rank case from $1.48$ towards the true $2.19$ while
leaving the full-rank case at $0.71$, still below the threshold. The gap does not close
entirely: even on the best probe $\widehat{\kappa}=1.48$ against $\kappa=2.19$. The remaining
gap comes from the change of evaluation point. The estimate is of the incoherence \emph{at the probing point}, which is the quantity
Lemma~\ref{lem:realised} needs -- the inversion is performed on the disagreement
measured there -- whereas Proposition~\ref{thm:free} compares bias energies at the truth's
own optimum. The practical rule therefore has two parts. First, probe after a burn-in and
check the drift, which supports the inversion. Second, treat the bound of
Proposition~\ref{thm:free} as conditional on the drift being small, since only then are
the two evaluation points interchangeable.

\label{prop:partial}
Let Assumptions~\ref{as:crowd}, \ref{as:oracle} (linear tilts) and~\ref{as:part} hold,
$n\ge3$, and put $p:=m/n$ and $\rho:=\frac{(m-1)n}{m(n-1)}\in[\tfrac12,1]$.
\begin{enumerate}[label=(\roman*),itemsep=1pt,topsep=2pt]
\item \emph{(The disagreement shrinks by a known factor.)} For every $i$ and $k$,
$\mathbb{E}[r^k_i\mid\mathcal F_{k-1},\,i\in P_k]=\rho\,\widetilde{\beta}_i$, with $\widetilde{\beta}_i$ the
full-crowd disagreement of Theorem~\ref{thm:ident}.
\item \emph{(An unbiased statistic.)} For the odd and even halves $\mathcal K_a$ of the
probing rounds let $N^{(a)}_i:=|\{k\in\mathcal K_a:i\in P_k\}|$ and, when both are
positive, $\widehat{\beta}^{(a)}_i:=\frac1{N^{(a)}_i}\sum_{k\in\mathcal K_a,\,i\in P_k}r^k_i$
and $\widehat{s}_i:=\langle \widehat{\beta}^{(1)}_i ,\, \widehat{\beta}^{(2)}_i \rangle$. Conditionally on the tilts
and on the participation indicators of client $i$, $\mathbb{E}\widehat{s}_i=\rho^2\lVert \widetilde{\beta}_i \rVert^2$
and
$\mathbb{E}(\widehat{s}_i-\rho^2\lVert \widetilde{\beta}_i \rVert^2)^2\le2\rho^2\lVert \widetilde{\beta}_i \rVert^2(\vartheta_{i,1}^2+\vartheta_{i,2}^2)+2\vartheta_{i,1}^2\vartheta_{i,2}^2$,
where
\[
\vartheta_{i,a}^2:=\frac{1}{N^{(a)}_i}\Big(\psi_{m,n}\bar q+\frac{\tau_{i,P}^2}{b}\Big),
\]
$\psi_{m,n}:=\frac{n(n-m)}{m(n-1)(n-2)}$ and
$\tau_{i,P}^2:=\sigma_i^2+\frac{1}{m(n-1)}\sum_{j\ne i}\sigma_j^2$. With probability at
least $1-2ne^{-pK_0/16}$, every $N^{(a)}_i\ge pK_0/4$.
\item \emph{(Weights.)} $\widehat{q}:=(T^{-1}\widehat{s}/\rho^2)_+$ satisfies
$\max_i|\widehat{q}_i-q_i|\le C_n\big(\max_i|\widehat{s}_i/\rho^2-\lVert \widetilde{\beta}_i \rVert^2|+3\kappa\bar q\big)$.
\end{enumerate}
The energies are therefore identified for every $m\ge2$ once $n\ge3$. The price of
partial participation is a factor of order $1/p$ in the number of reports, a factor
$\rho^{-2}\le4$ from rescaling, and the variance $\psi_{m,n}\bar q$ of drawing the
participants, which vanishes at $m=n$.

<!-- proposition-9: body_md -->
\label{prop:part-corr}
Suppose that the clients that report are those in a set $\mathcal R$, and that their
tilts are $\beta_i=c+\beta_i'$, $i\in\mathcal R$, with $c\in\mathbb{R}^d$ fixed and the
$\beta'_i$ independent and centred, as happens when tilts are drawn i.i.d.\ and
selection depends on each client's own tilt. Then:
(i) the transcript of any protocol has the same law under
$(f_0,(h_i)_{i\in\mathcal R})$ and under
$(f_0+\langle c ,\, \cdot \rangle,(h_i-\langle c ,\, \cdot \rangle)_{i\in\mathcal R})$, and the second
instance satisfies Assumption~\ref{as:crowd} on $\mathcal R$;
(ii) for every $w$ in the simplex on $\mathcal R$, $b(w)=c+b'(w)$ with
$b'(w):=\sum_iw_i\beta'_i$, so no weighting removes $c$;
(iii) the disagreements of the two instances coincide, so \textsc{Crowd}{} computes the same
statistics under both, and they estimate the energies $\lVert \beta'_i \rVert^2$ of the
second.

<!-- proof-1: body_md -->
Let $U$ be the open convex set, fix $i$, and write $h=h_i$.

\emph{Step 1: the monotonicity defect vanishes pointwise.} For $x,y\in U$ put
$M(x,y):=\langle \nabla h(x)-\nabla h(y) ,\, x-y \rangle$. A differentiable convex function has a
monotone gradient, so $M(x,y)\ge0$ almost surely. By linearity of expectation and
Assumption~\ref{as:crowd},
$\mathbb{E} M(x,y)=\langle \mathbb{E}\nabla h(x)-\mathbb{E}\nabla h(y) ,\, x-y \rangle=0$. A non-negative random variable
with zero mean vanishes almost surely, so $M(x,y)=0$ almost surely, for each fixed pair.

\emph{Step 2: simultaneously for all pairs.} Let $U_0\subseteq U$ be countable and
dense. By countable additivity there is an event of probability one on which
$M(x,y)=0$ for all $x,y\in U_0$. On that event, continuity of $\nabla h$ ($h\in C^2$ by
Assumption~\ref{as:crowd}; $C^1$ would suffice) makes $M$ continuous on $U\times U$, so $M(x,y)=0$ for all $x,y\in U$.

\emph{Step 3: a vanishing defect forces affinity.} Fix $x_0\in U$ and $x\in U$, and put
$g(t):=h(x_0+t(x-x_0))$ for $t\in[0,1]$. Then $g'(t)-g'(0)=\langle \nabla
h(x_0+t(x-x_0))-\nabla h(x_0) ,\, x-x_0 \rangle=M(x_0+t(x-x_0),x_0)/t=0$ for $t\in(0,1]$. Hence
$g$ is affine and $h(x)=h(x_0)+\langle \nabla h(x_0) ,\, x-x_0 \rangle$ for every $x\in U$: $h$ is
affine on $U$. Since $i$ was arbitrary and there are finitely many clients, the
conclusion holds for all of them on one event of probability one.

\label{lem:crossmart}
Let $(X_k)$ and $(Y_k)$ be $\mathbb{R}^d$-valued martingale difference sequences with respect
to $(\mathcal F_k)$, with $\mathbb{E}[\lVert X_k \rVert^2\mid\mathcal F_{k-1}]\le v_X$ and

Fix $\rho>0$ and put the prior $x_c\sim\mathcal N(0,\tfrac{\rho^2}{d}I_d)$,
independent of $\beta$ and of all noises. Since the minimax risk over $x_c\in\mathbb{R}^d$
dominates the Bayes risk for this prior, it suffices to lower bound the latter.

\emph{Step 1: reduction to a location model.} The algorithm interacts with client
$i$ only through at most $K$ gradient queries; the $m$-th query at a point $x$ returns
$\nabla f_i(x)+\varepsilon_i^m=\mu(x-x_c)+\beta_i+\varepsilon_i^m$, with the $\varepsilon_i^m$ independent
$\mathcal N(0,\tfrac{\sigma_i^2}dI_d)$. The algorithm chooses $x$ and
knows $\mu$, so the transcript is a measurable function of the private seed $U$ and
of the array
\[
\xi_i^m:=-\mu x_c+\beta_i+\varepsilon_i^m,\qquad i\in[n],\ m\in[K],
\]
and conversely (by induction over rounds, the algorithm's state after each report
is a measurable function of $U$ and the $\xi$'s seen so far, and adding the known
$\mu x$ recovers the report). Note that the law of $\xi_i^m$ does not depend on the
query point: adaptivity provides no information. Hence
$\sigma(\text{transcript})\subseteq\sigma(U,\{\xi^m_i\})$, and it suffices to bound
the Bayes risk of estimating $x_c$ from $(U,\{\xi^m_i\})$.

\emph{Step 2: sufficiency.} Conditionally on $(x_c,\beta_i)$ the vectors
$\xi^1_i,\dots,\xi^K_i$ are i.i.d.\ Gaussian with mean $-\mu x_c+\beta_i$ and
covariance $\tfrac{\sigma_i^2}{d}I_d$; hence
$\bar\xi_i=\tfrac1K\sum_m\xi_i^m$ is sufficient, and
\[
\bar\xi_i=-t+\theta_i,\qquad t:=\mu x_c,\qquad
\theta_i:=\beta_i+\bar\varepsilon_i\sim\mathcal N\Big(0,\frac{v_i}{d}I_d\Big),\qquad
v_i:=\zeta_i^2+\frac{\sigma_i^2}{K},
\]
with $\theta_1,\dots,\theta_n$ independent of each other and of $t$, and
$t\sim\mathcal N(0,\tfrac{\mu^2\rho^2}{d}I_d)$.

\emph{Step 3: Gaussian Bayes risk.} All quantities decouple across the $d$
coordinates. For a single coordinate, $t$ has prior variance
$P:=\mu^2\rho^2/d$ and we observe $n$ independent measurements $-t+\theta_i$ with
noise variances $v_i/d$; the posterior of $t$ is Gaussian with variance
$\big(P^{-1}+\sum_i d/v_i\big)^{-1}$. The Bayes risk for squared error is the
posterior variance, and independence of $U$ from everything else cannot reduce it.
Summing over $d$ coordinates,
\[
\mathbb{E}\lVert \widehat{t}-t \rVert^2\ \ge\ d\Big(\frac{d}{\mu^2\rho^2}+\sum_i\frac{d}{v_i}\Big)^{-1}
=\Big(\frac{1}{\mu^2\rho^2}+\sum_i\frac1{v_i}\Big)^{-1}.
\]

=\tfrac1{2\mu}\lVert \mu\widehat{x}-t \rVert^2$, and $\mu\widehat{x}$ is an admissible estimator of

$\tfrac1{2\mu}\big(\tfrac{1}{\mu^2\rho^2}+\sum_iv_i^{-1}\big)^{-1}$; letting

\emph{Step 5: the algorithm may even be told $\widetilde{\beta}$.} Condition on
$\widetilde{\beta}$. By Proposition~\ref{prop:orth}(iii), $\bar\beta=m(\widetilde{\beta})+e$ with

independent. The $z_i$ share the mean $e-t$, so their precision-weighted average
$y$ is sufficient for $e-t$, and $y=e-t+\bar\varepsilon$ with
$\bar\varepsilon\sim\mathcal N(0,\tfrac{s^2}dI_d)$, $s^2:=(\sum_iK/\sigma_i^2)^{-1}$ (if some
