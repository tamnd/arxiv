<!-- mydef-1: body_md -->
\label{def:directed_information} 
Let $X^n \triangleq (X_1, X_2, \cdots, X_n )$ and $Z^n \triangleq (Z_1, Z_2, \cdots, Z_n)$ be two temporally correlated random sequences. The information flow from $X^n$ to $Z^n$ is measured by directed information, defined as
\begin{equation}
    I(X^n \rightarrow Z^n) = \sum_{k=1}^n I(X^k;Z_k \mid Z^{k-1}).
\end{equation}
A more intuitive interpretation of directed information $I(X^n \rightarrow Z^n)$ is the amount of information that the random sequence $(X_1, X_2, \cdots, X_n)$ causally provides about another random sequence $(Z_1, Z_2, \cdots, Z_n)$.
<!-- end mydef-1 -->
<!-- myrem-1: body_md -->
\label{remark:directed_advantage}
Let $I_{j,k}^{X,Z} = I(X_j; Z_k \mid X^{j-1}, Z^{k-1})$. We define the information matrix $I^{X,Z}$, whose $(j,k)$-th entry is given by $I_{j,k}^{X, Z}$, characterizing the temporal dependence structure between $X^n$ and $Z^n$ as
\begin{equation}\label{eq:informtation_matrix}
I^{X,Z} = 
\begin{pmatrix}
I_{1,1}^{X,Z} & I_{1,2}^{X,Z} & \cdots & I_{1,n}^{X,Z} \\
I_{2,1}^{X,Z} & I_{2,2}^{X,Z} & \cdots & I_{2,n}^{X,Z}\\
\vdots & \vdots & \ddots & \vdots \\
I_{n,1}^{X,Z} & I_{n,2}^{X,Z} & \cdots & I_{n,n}^{X,Z}
\end{pmatrix}_{n \times n.}
\end{equation}
The mutual information $I(X^n; Z^n)$ and the conditional mutual information $\sum_{j=1}^n I(X^n; Z^n | D_{i-1}^r)$ admit matrix representations based on $I^{X,Z}$, where $D_{j-1}^r$ is constructed from historical information up to timestamp $t_{j-1}$. 
Accordingly, the mutual information $I(X^n; Z^n)$ can be decomposed as
\begin{align}
    I(X^n; Z^n) &= \sum_{1 \leq j \leq k \leq n} I_{j,k}^{X,Z} + \sum_{1\leq k < j \leq n} I_{j,k}^{X,Z} \\
    &=I(X^n \rightarrow Z^n) + I(Z^{n-1} \rightarrow X^n), \label{eq:mutual_directed}
\end{align}
where the term $I(Z^{n-1} \rightarrow X^n)$ captures potential backward (non-causal) dependencies.
However, as shown in Fig. \ref{fig:relationship}, the transmitter encodes the $n$-th raw data $x_n$ into the semantic encoding $z_n$ with the aid of the knowledge base $D_{n-1}^r = \{x_1, x_2, \cdots, x_{n-1} \} \cup \{z_1, z_2, \cdots, z_{n-1} \}$, which implies that no causal influence exists from $z^{n-1}$ to $x_n$.
\textit{Consequently, the mutual information includes non-causal statistical dependencies and could overestimate the effective task-relevant information flow compared with directed information.}
In contrast, the conditional mutual information $\sum_{j=1}^n I(X_j; Z_j \mid D_{j-1}^r)$ depends only on the diagonal entries of $I^{X,Z}$ and thus ignores inter-temporal dependencies between $X^{n-1}$ and $Z^n$.
Moreover, directed information can be decomposed as
\begin{align}
I(X^n \rightarrow Z^n) &= \sum_{1\leq j \leq n} I_{j,j}^{X, Z} + \sum_{1 \leq j < k \leq n} I_{j,k}^{X,Z} \\
&=\sum_{1 \leq j \leq n} I(X_j; Z_j \mid D_{j-1}^r) + \sum_{1 \leq j < k \leq n} I_{j,k}^{X, Z}.
\end{align}
\textit{As a result, the conditional mutual information captures only instantaneous dependencies and could underestimate the effective task-relevant information flow compared with directed information.}

<!-- mythe-1: body_md -->
\label{them:data_inequality}
Let the temporally correlated random sequences $Y^n$, $X^n$, $Z^n$, and $\widehat{Z}^n$ satisfy the causal relations illustrated in Fig. \ref{fig:relationship}. Then, the following data processing inequalities hold
\begin{equation}\label{eq:data_proc}
\begin{aligned}
    I(Y^n \rightarrow \widehat{Z}^n) \leq I(X^n \rightarrow \widehat{Z}^n) & \leq \sum_{k=1}^n I(\widehat{Z}_k; Z_k | \widehat{Z}^{k-1}) \\
    & \leq \sum_{k=1}^n I(\widehat{Z}_k;Z_k) \\
    &\leq n\cdot C,
\end{aligned}
\end{equation}
where $C$ denotes the Shannon channel capacity of the extended channel, given by
\begin{equation}
C = \max_{k \in [1,n]}  \max_{P(z_k)} I(\widehat{Z}_k;Z_k).
\end{equation}
In particular, directed information quantifies the total causal information flow from one random sequence to another.
Dividing both sides of Eq. \eqref{eq:data_proc} by $n$ yields the following data processing inequality for the average information flow
\begin{equation}\label{eq:them_2}
    \frac{1}{n}I(Y^n \rightarrow \widehat{Z}^n) \leq \frac{1}{n}I(X^n \rightarrow \widehat{Z}^n) \leq \max_{k \in [1,n] } I(\widehat{Z}_k; Z_k) \leq C.
\end{equation}

<!-- myrem-2: body_md -->
\label{rem:beta_DIB} To gain further insight into the role of $\beta$, we decompose the channel capacity $n\cdot C$ ($n$ corresponds to timestamp $t_n$) into task-relevant semantic information, task-irrelevant redundancy information, and residual channel capacity, as follows  
\begin{equation}
    \begin{aligned}
    &n \cdot C \\
    =&\underbrace{I(Y^n \rightarrow \widehat{Z}^n)}_{\text{Semantic Relevance}} + \underbrace{I(X^n \rightarrow \widehat{Z}^n) - I(Y^n \rightarrow \widehat{Z}^n)}_{\text{Redundant Information}} \\
    &+ \underbrace{n\cdot C - I(X^n \rightarrow \widehat{Z}^n)}_{\text{Residual Channel Capacity}}.
    \end{aligned}
\end{equation}
Particularly, the term $I(Y^n \rightarrow \widehat{Z}^n)$ quantifies the amount of information contained in the semantic encodings $\widehat{Z}^n$ that is relevant to the targets $Y^n$, the term $I(X^n \rightarrow \widehat{Z}^n) - I(Y^n \rightarrow \widehat{Z}^n)$ measures the amount of transmitted information that is irrelevant to $Y^n$, and the term $n\cdot C - I(X^n \rightarrow \widehat{Z}^n)$ denotes the channel capacity not utilized for transmitting information about the source information $X^n$, with their relationships illustrated in Fig. \ref{fig:optimization}(a).
Moreover, the problem \eqref{eq:DIB} indicates that a smaller value of $\beta$ places greater emphasis on semantic relevance (i.e., $I(Y^n \rightarrow \widehat{Z}^n)$), whereas a larger value of $\beta$ assigns a higher penalty to the compression cost (i.e., $I(X^n \rightarrow \widehat{Z}^n)$) and therefore favors compression, as illustrated in Fig. \ref{fig:optimization}(b) and Fig. \ref{fig:optimization}(c).
Note that regardless of the value of $\beta$, maximizing problem \eqref{eq:DIB} encourages the retention of more task-relevant semantic information and the removal of more task-irrelevant redundancy.
<!-- end myrem-2 -->
<!-- mythe-2: body_md -->
\label{them:bound_DIB}The directed information bottleneck (DIB) objective is given by
\begin{equation}
    \mathcal{L}_{\mathrm{DIB}} = -I(Y^n \rightarrow \widehat{Z}^n) + \beta I(X^n \rightarrow \widehat{Z}^n).
\end{equation}
Let $\mathcal{A}_{\theta}(\widehat z_k|x^k,\widehat z^{k-1})$, $\mathcal{B}_{\phi}(y_k|\widehat z^k)$, and $r_{\psi}(\widehat z_k|\widehat z^{k-1})$ denote the causal semantic encoder, the variational decoder, and the autoregressive prior, respectively.
Then, the DIB objective is upper-bounded by the following variational objective
\begin{equation}\label{eq:VDIB}
\begin{aligned}
    \mathcal{L}_{\mathrm{VDIB}} &= \sum_{k=1}^n \underbrace{\mathbb{E}_{P(x^k,y^k)} \{ \mathbb{E}_{\mathcal{A}_{\theta}(\widehat{z}^k \| x^k)} [ -\log \mathcal{B}_{\phi}(y_k | \widehat{z}^k) ] \}}_{\text{Task Loss}} \\
    &+ \beta \sum_{k=1}^n \underbrace{D_{\mathrm{KL}} (\mathcal{A}_{\theta}(\widehat{z}_k | x^k,\widehat{z}^{k-1})\| r_{\psi}(\widehat{z}_k \mid \widehat{z}^{k-1}) )}_{\text{Information Regularization}},
\end{aligned}
\end{equation}
where $D_{\mathrm{KL}}(* \| *)$ denotes the Kullback--Leibler (KL) divergence, the causally conditioned distribution $\mathcal{A}_{\theta}(\widehat{z}^k \| x^k)$ is given by
\begin{equation}
    \mathcal{A}_{\theta}(\widehat{z}^k \| x^k) = \prod_{t=1}^k \mathcal{A}_{\theta}(\widehat{z}_t | x^t, \widehat{z}^{t-1}), 
\end{equation}
and the received semantic encoding over the AWGN channel is expressed as $\widehat{z}_k = z_k + \varepsilon_k,~\varepsilon_k \sim \mathcal{N}(\bm{0}, \sigma^2 \bm{I})$. 
Particularly, the resulting upper bound admits a rate-distortion decomposition, where the task loss acts as the distortion term and the KL regularizer quantifies the information rate, thus capturing the trade-off between task relevance and causal representation compression.
